ZDRVSP

الغرض:

!>
!> ZDRVSP تختبر الروتينات التوجيهية ZSPSV و -SVX.
!>

المعلمات

DOTYPE 
!>          DOTYPE هو مصفوفة LOGICAL، بُعدها (NTYPES)
!>          أنواع المصفوفات التي سيتم استخدامها للاختبار. يتم استخدام المصفوفات من النوع j
!>          (لـ 1 <= j <= NTYPES) للاختبار إذا كان DOTYPE(j) =
!>          .TRUE.؛ إذا كان DOTYPE(j) = .FALSE.، فلن يتم استخدام النوع j.
!>

NN

!>          NN هو INTEGER
!>          عدد قيم N الموجودة في المتجه NVAL.
!>

NVAL

!>          NVAL هي مصفوفة INTEGER، بُعدها (NN)
!>          قيم بُعد المصفوفة N.
!>

NRHS

!>          NRHS هو INTEGER
!>          عدد متجهات الطرف الأيمن التي سيتم إنشاؤها لكل نظام خطي.
!>

THRESH

!>          THRESH هو DOUBLE PRECISION
!>          القيمة الحدية لنسب نتائج الاختبار. يتم تضمين نتيجة
!>          في ملف الإخراج إذا كان RESULT >= THRESH. لطباعة كل نسبة اختبار،
!>          استخدم THRESH = 0.
!>

TSTERR

!>          TSTERR هو LOGICAL
!>          علامة تشير إلى ما إذا كان سيتم اختبار مخارج الأخطاء.
!>

NMAX

!>          NMAX هو INTEGER
!>          الحد الأقصى للقيمة المسموح بها لـ N، المستخدم في تحديد أبعاد
!>          مصفوفات العمل.
!>

A

!>          A هي مصفوفة COMPLEX*16، بُعدها
!>                      (NMAX*(NMAX+1)/2)
!>

AFAC

!>          AFAC هي مصفوفة COMPLEX*16، بُعدها
!>                      (NMAX*(NMAX+1)/2)
!>

AINV

!>          AINV هي مصفوفة COMPLEX*16، بُعدها
!>                      (NMAX*(NMAX+1)/2)
!>

B

!>          B هي مصفوفة COMPLEX*16، بُعدها (NMAX*NRHS)
!>

X

!>          X هي مصفوفة COMPLEX*16، بُعدها (NMAX*NRHS)
!>

XACT

!>          XACT هي مصفوفة COMPLEX*16، بُعدها (NMAX*NRHS)
!>

WORK

!>          WORK هي مصفوفة COMPLEX*16، بُعدها
!>                      (NMAX*max(2,NRHS))
!>

RWORK

!>          RWORK هي مصفوفة DOUBLE PRECISION، بُعدها (NMAX+2*NRHS)
!>

IWORK

!>          IWORK هي مصفوفة INTEGER، بُعدها (NMAX)
!>

NOUT

!>          NOUT هو INTEGER
!>          رقم الوحدة للإخراج.
!>
