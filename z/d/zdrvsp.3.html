subroutine zdrvsp (logical, dimension( * ) dotype, integer nn, integer, dimension( * ) nval, integer nrhs, double precision thresh, logical tsterr, integer nmax, complex*16, dimension( * ) a, complex*16, dimension( * ) afac, complex*16, dimension( * ) ainv, complex*16, dimension( * ) b, complex*16, dimension( * ) x, complex*16, dimension( * ) xact, complex*16, dimension( * ) work, double precision, dimension( * ) rwork, integer, dimension( * ) iwork, integer nout)

ZDRVSP

الغرض:

!>
!> ZDRVSP تختبر الروتينات الفرعية ZSPSV و -SVX.
!>

المعلمات

DOTYPE 
!>          DOTYPE هو مصفوفة منطقية، أبعادها (NTYPES)
!>          أنواع المصفوفات التي ستستخدم للاختبار. المصفوفات من النوع j
!>          (لـ 1 <= j <= NTYPES) تستخدم للاختبار إذا كان DOTYPE(j) =
!>          .TRUE.؛ إذا كان DOTYPE(j) = .FALSE.، فلن يتم استخدام النوع j.
!>

NN

!>          NN هو عدد صحيح
!>          عدد القيم لـ N الموجودة في المتجه NVAL.
!>

NVAL

!>          NVAL هي مصفوفة أعداد صحيحة، أبعادها (NN)
!>          قيم بعد المصفوفة N.
!>

NRHS

!>          NRHS هو عدد صحيح
!>          عدد متجهات الجانب الأيمن التي سيتم إنشاؤها لكل نظام خطي.
!>

THRESH

!>          THRESH هو دقة مزدوجة
!>          القيمة الحدية لنسب الاختبار. يتم تضمين النتيجة
!>          في ملف الإخراج إذا كانت RESULT >= THRESH. لطباعة
!>          كل نسبة اختبار، استخدم THRESH = 0.
!>

TSTERR

!>          TSTERR هو منطقي
!>          علامة تشير إلى ما إذا كان سيتم اختبار مخارج الخطأ.
!>

NMAX

!>          NMAX هو عدد صحيح
!>          الحد الأقصى للقيمة المسموح بها لـ N، المستخدمة في تحديد أبعاد
!>          مصفوفات العمل.
!>

A

!>          A هي مصفوفة COMPLEX*16، أبعادها
!>                      (NMAX*(NMAX+1)/2)
!>

AFAC

!>          AFAC هي مصفوفة COMPLEX*16، أبعادها
!>                      (NMAX*(NMAX+1)/2)
!>

AINV

!>          AINV هي مصفوفة COMPLEX*16، أبعادها
!>                      (NMAX*(NMAX+1)/2)
!>

B

!>          B هي مصفوفة COMPLEX*16، أبعادها (NMAX*NRHS)
!>

X

!>          X هي مصفوفة COMPLEX*16، أبعادها (NMAX*NRHS)
!>

XACT

!>          XACT هي مصفوفة COMPLEX*16، أبعادها (NMAX*NRHS)
!>

WORK

!>          WORK هي مصفوفة COMPLEX*16، أبعادها
!>                      (NMAX*max(2,NRHS))
!>

RWORK

!>          RWORK هي مصفوفة دقة مزدوجة، أبعادها (NMAX+2*NRHS)
!>

IWORK

!>          IWORK هي مصفوفة أعداد صحيحة، أبعادها (NMAX)
!>

NOUT

!>          NOUT هو عدد صحيح
!>          رقم الوحدة للإخراج.
!>
