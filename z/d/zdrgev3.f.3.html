; سيتم تخطي القيم الصفرية. NN >= 0. !>

NTYPES

!>          NTYPES هو عدد صحيح
!>          عدد العناصر في DOTYPE. إذا كان صفرًا، فإن ZDRGEV3
!>          لا تفعل شيئًا. يجب أن يكون على الأقل صفرًا. إذا كان MAXTYP+1
!>          و NSIZES هو 1، فسيتم تعريف نوع إضافي، MAXTYP+1،
!>          وهو استخدام أي مصفوفة موجودة في A. هذا
!>          مفيد فقط إذا كان DOTYPE(1:MAXTYP) هو .FALSE. و
!>          DOTYPE(MAXTYP+1) هو .TRUE. .
!>

DOTYPE

!>          DOTYPE هو مصفوفة منطقية، بُعدها (NTYPES)
!>          إذا كان DOTYPE(j) هو .TRUE.، فلكل حجم في NN
!>          سيتم إنشاء مصفوفة بهذا الحجم ومن النوع j.
!>          إذا كان NTYPES أصغر من الحد الأقصى لعدد الأنواع
!>          المعرفة (PARAMETER MAXTYP)، فسيتم تجاهل الأنواع NTYPES+1 حتى
!>          MAXTYP. إذا كان NTYPES أكبر
!>          من MAXTYP، فسيتم تجاهل DOTYPE(MAXTYP+1) حتى DOTYPE(NTYPES).
!>

ISEED

!>          ISEED هي مصفوفة أعداد صحيحة، بُعدها (4)
!>          عند الدخول، تحدد ISEED بذرة مولد الأرقام العشوائية. يجب أن تكون عناصر المصفوفة بين 0 و 4095؛
!>          إذا لم تكن كذلك، فسيتم تقليلها بمعيار 4096. أيضًا، يجب أن يكون ISEED(4)
!>          فرديًا. يستخدم مولد الأرقام العشوائية تسلسل خطي متطابق محدود بأعداد صحيحة صغيرة، وبالتالي
!>          ينبغي أن ينتج أرقامًا عشوائية مستقلة عن الجهاز. قيم ISEED تتغير عند الخروج، ويمكن استخدامها في
!>          الاستدعاء التالي لـ ZDRGES لمواصلة نفس تسلسل الأرقام العشوائية.
!>

THRESH

!>          THRESH هو دقة مزدوجة
!>          سيتم احتساب الاختبار كـ إذا كان ، المحسوب كـ
!>          تم وصفه أعلاه، يتجاوز THRESH. لاحظ أن الخطأ
!>          مُقاس ليكون O(1)، لذا يجب أن يكون THRESH مضاعفًا صغيرًا بشكل معقول
!>          من 1، على سبيل المثال، 10 أو 100. على وجه الخصوص، يجب
!>          ألا يعتمد على الدقة (أحادية مقابل مزدوجة) أو حجم
!>          المصفوفة. يجب أن يكون على الأقل صفرًا.
!>

NOUNIT

!>          NOUNIT هو عدد صحيح
!>          رقم الوحدة FORTRAN لطباعة رسائل الخطأ
!>          (على سبيل المثال، إذا أعادت روتينية قيمة IERR غير صفرية.)
!>

A

!>          A هي مصفوفة COMPLEX*16، بُعدها (LDA, max(NN))
!>          تستخدم لتخزين المصفوفة الأصلية A. تستخدم كمدخل فقط
!>          إذا كان NTYPES=MAXTYP+1، و DOTYPE(1:MAXTYP)=.FALSE.، و
!>          DOTYPE(MAXTYP+1)=.TRUE.
!>

LDA

!>          LDA هو عدد صحيح
!>          البعد الرائد لمصفوفات A، B، S، و T.
!>          يجب أن يكون على الأقل 1 وعلى الأقل max( NN ).
!>

B

!>          B هي مصفوفة COMPLEX*16، بُعدها (LDA, max(NN))
!>          تستخدم لتخزين المصفوفة الأصلية B. تستخدم كمدخل فقط
!>          إذا كان NTYPES=MAXTYP+1، و DOTYPE(1:MAXTYP)=.FALSE.، و
!>          DOTYPE(MAXTYP+1)=.TRUE.
!>

S

!>          S هي مصفوفة COMPLEX*16، بُعدها (LDA, max(NN))
!>          مصفوفة الشكل شتور (Schur form matrix) التي حسبتها ZGGEV3 من A. عند الخروج،
!>          تحتوي S على مصفوفة الشكل شتور المقابلة للمصفوفة
!>          في A.
!>

T

!>          T هي مصفوفة COMPLEX*16، بُعدها (LDA, max(NN))
!>          المصفوفة المثلثية العليا التي حسبتها ZGGEV3 من B.
!>

Q

!>          Q هي مصفوفة COMPLEX*16، بُعدها (LDQ, max(NN))
!>          مصفوفة المتجهات الذاتية (اليسرى) التي حسبتها ZGGEV3.
!>

LDQ

!>          LDQ هو عدد صحيح
!>          البعد الرائد لمصفوفتي Q و Z. يجب
!>          أن يكون على الأقل 1 وعلى الأقل max( NN ).
!>

Z

!>          Z هي مصفوفة COMPLEX*16، بُعدها ( LDQ, max(NN) )
!>          المصفوفة المتعامدة (اليمنى) التي حسبتها ZGGEV3.
!>

QE

!>          QE هي مصفوفة COMPLEX*16، بُعدها ( LDQ, max(NN) )
!>          تحتوي QE على المتجهات الذاتية اليمنى أو اليسرى المحسوبة.
!>

LDQE

!>          LDQE هو عدد صحيح
!>          البعد الرائد لمصفوفة QE. LDQE >= max(1,max(NN)).
!>

ALPHA

!>          ALPHA هي مصفوفة COMPLEX*16، بُعدها (max(NN))
!>

BETA

!>          BETA هي مصفوفة COMPLEX*16، بُعدها (max(NN))
!>
!>          القيم الذاتية المعممة لـ (A,B) التي حسبتها ZGGEV3.
!>          ( ALPHAR(k)+ALPHAI(k)*i ) / BETA(k) هي القيمة الذاتية المعممة k-th
!>          للمصفوفتين A و B.
!>

ALPHA1

!>          ALPHA1 هي مصفوفة COMPLEX*16، بُعدها (max(NN))
!>

BETA1

!>          BETA1 هي مصفوفة COMPLEX*16، بُعدها (max(NN))
!>
!>          مثل ALPHAR، ALPHAI، BETA، تحتوي هذه المصفوفات على
!>          القيم الذاتية للمصفوفتين A و B، ولكن تلك المحسوبة عندما تقوم ZGGEV3
!>          بحساب تحليل جزئي للقيم الذاتية فقط، أي ليس
!>          القيم الذاتية والمتجهات الذاتية اليسرى واليمنى.
!>

WORK

!>          WORK هي مصفوفة COMPLEX*16، بُعدها (LWORK)
!>

LWORK

!>          LWORK هو عدد صحيح
!>          عدد العناصر في WORK. LWORK >= N*(N+1)
!>

RWORK

!>          RWORK هي مصفوفة دقة مزدوجة، بُعدها (8*N)
!>          مساحة عمل حقيقية.
!>

RESULT

!>          RESULT هي مصفوفة دقة مزدوجة، بُعدها (2)
!>          القيم المحسوبة بواسطة الاختبارات الموصوفة أعلاه.
!>          القيم حاليًا محدودة إلى 1/ulp، لتجنب تجاوز السعة.
!>

INFO

!>          INFO هو عدد صحيح
!>          = 0: خروج ناجح
!>          < 0: إذا كان INFO = -i، فإن الوسيط i كان له قيمة غير قانونية.
!>          > 0: روتينية أعادت رمز خطأ. INFO هو
!>                القيمة المطلقة لقيمة INFO المعادة.
!>
