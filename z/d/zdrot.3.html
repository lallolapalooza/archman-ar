subroutine zdrot (integer n, complex*16, dimension( * ) zx, integer incx, complex*16, dimension( * ) zy, integer incy, double precision c, double precision s)

ZDROT

الغرض:

!>
!> يطبق دوران مستوى، حيث تكون cos و sin (c و s) أعدادًا حقيقية
!> والمتجهات cx و cy أعدادًا مركبة.
!> جاك دونغارا، linpack، 3/11/78.
!>

المعلمات

N 
!>          N هو عدد صحيح (INTEGER)
!>           عند الإدخال، يحدد N رتبة المتجهين cx و cy.
!>           يجب أن تكون N على الأقل صفر.
!>

ZX

!>          ZX هو مصفوفة أعداد مركبة (COMPLEX*16)، بأبعاد لا تقل عن
!>           ( 1 + ( N - 1 )*abs( INCX ) ).
!>           قبل الإدخال، يجب أن تحتوي المصفوفة ZX المتزايدة على متجه cx المكون من n
!>           عنصر. عند الخروج، يتم استبدال ZX بمتجه cx المحدث.
!>

INCX

!>          INCX هو عدد صحيح (INTEGER)
!>           عند الإدخال، يحدد INCX الزيادة لعناصر ZX.
!>           يجب ألا تكون INCX صفرًا.
!>

ZY

!>          ZY هو مصفوفة أعداد مركبة (COMPLEX*16)، بأبعاد لا تقل عن
!>           ( 1 + ( N - 1 )*abs( INCY ) ).
!>           قبل الإدخال، يجب أن تحتوي المصفوفة ZY المتزايدة على متجه cy المكون من n
!>           عنصر. عند الخروج، يتم استبدال ZY بمتجه cy المحدث.
!>

INCY

!>          INCY هو عدد صحيح (INTEGER)
!>           عند الإدخال، يحدد INCY الزيادة لعناصر ZY.
!>           يجب ألا تكون INCY صفرًا.
!>

C

!>          C هو دقة مزدوجة (DOUBLE PRECISION)
!>           عند الإدخال، يحدد C جيب التمام (cosine)، cos.
!>

S

!>          S هو دقة مزدوجة (DOUBLE PRECISION)
!>           عند الإدخال، يحدد S جيب الزاوية (sine)، sin.
!>
