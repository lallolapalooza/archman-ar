subroutine ztplqt (integer m, integer n, integer l, integer mb, complex*16, dimension( lda, * ) a, integer lda, complex*16, dimension( ldb, * ) b, integer ldb, complex*16, dimension( ldt, * ) t, integer ldt, complex*16, dimension( * ) work, integer info)

ZTPLQT

الغرض:

!>
!> ZTPLQT يحسب تحليل LQ محظور لمصفوفة C المعقدة، والتي تتكون من
!> مصفوفة مثلثية A ومصفوفة خماسية B، باستخدام تمثيل WY المدمج لـ Q.
!>

المعلمات

M 
!>          M هو عدد صحيح (INTEGER)
!>          عدد صفوف المصفوفة B، ورتبة المصفوفة المثلثية A.
!>          M >= 0.
!>

N

!>          N هو عدد صحيح (INTEGER)
!>          عدد أعمدة المصفوفة B.
!>          N >= 0.
!>

L

!>          L هو عدد صحيح (INTEGER)
!>          عدد صفوف الجزء شبه المنحرف السفلي من B.
!>          MIN(M,N) >= L >= 0. انظر التفاصيل الإضافية.
!>

MB

!>          MB هو عدد صحيح (INTEGER)
!>          حجم الكتلة المستخدم في QR المحظور. M >= MB >= 1.
!>

A

!>          A هي مصفوفة COMPLEX*16، ذات البعد (LDA,M)
!>          عند الإدخال، تحتوي المصفوفة المثلثية السفلية A بحجم M×M على.
!>          عند الخروج، تحتوي العناصر على القطر وفي الأسفل منه في المصفوفة
!>          على المصفوفة المثلثية السفلية L.
!>

LDA

!>          LDA هو عدد صحيح (INTEGER)
!>          البعد الرائد للمصفوفة A. LDA >= max(1,M).
!>

B

!>          B هي مصفوفة COMPLEX*16، ذات البعد (LDB,N)
!>          عند الإدخال، تحتوي المصفوفة الخماسية B بحجم M×N على. يحتوي أول N-L عمود
!>          على مصفوفة مستطيلة، وآخر L عمود على مصفوفة شبه منحرف سفلية.
!>          عند الخروج، تحتوي B على المصفوفة الخماسية V. انظر التفاصيل الإضافية.
!>

LDB

!>          LDB هو عدد صحيح (INTEGER)
!>          البعد الرائد للمصفوفة B. LDB >= max(1,M).
!>

T

!>          T هي مصفوفة COMPLEX*16، ذات البعد (LDT,N)
!>          المُعكسات الكتلية المثلثية السفلية المخزنة بالشكل المدمج
!>          كسلسلة من الكتل المثلثية العلوية. انظر التفاصيل الإضافية.
!>

LDT

!>          LDT هو عدد صحيح (INTEGER)
!>          البعد الرائد للمصفوفة T. LDT >= MB.
!>

WORK

!>          WORK هي مصفوفة COMPLEX*16، ذات البعد (MB*M)
!>

INFO

!>          INFO هو عدد صحيح (INTEGER)

!>          < 0:  إذا كان INFO = -i، فإن المعامل i كان له قيمة غير قانونية

تفاصيل إضافية:

!>
!>  المصفوفة المدخلة C هي مصفوفة بحجم M × (M+N)
!>
!>               C = [ A ] [ B ]
!>
!>
!>  حيث A هي مصفوفة مثلثية سفلية بحجم M×M، و B هي مصفوفة خماسية بحجم M×N
!>  تتكون من مصفوفة مستطيلة B1 بحجم M×(N-L) في اليسار ومصفوفة شبه منحرف علوية B2 بحجم M×L:
!>          [ B ] = [ B1 ] [ B2 ]
!>                   [ B1 ]  <- مستطيلة بحجم M×(N-L)
!>                   [ B2 ]  <- شبه منحرف سفلية بحجم M×L.
!>
!>  تتكون المصفوفة شبه المنحرفية B2 من أول L أعمدة لمصفوفة مثلثية سفلية بحجم M×M، حيث 0 <= L <= MIN(M,N). إذا كانت L=0،
!>  فإن B تكون مستطيلة بحجم M×N؛ إذا كانت M=L=N، فإن B تكون مثلثية سفلية.
!>
!>  تخزن المصفوفة W المُعكسات الأولية H(i) في الصف i
!>  فوق القطر (لـ A) في المصفوفة المدخلة C بحجم M×(M+N)
!>            [ C ] = [ A ] [ B ]
!>                   [ A ]  <- مثلثية سفلية بحجم M×M
!>                   [ B ]  <- خماسية بحجم M×N
!>
!>  بحيث يمكن تمثيل W كـ
!>            [ W ] = [ I ] [ V ]
!>                   [ I ]  <- متطابقة بحجم M×M
!>                   [ V ]  <- خماسية بحجم M×N، بنفس شكل B.
!>
!>  وبالتالي، فإن جميع المعلومات اللازمة لـ W موجودة عند الخروج في B،
!>  والتي نسميها V أعلاه. لاحظ أن V لها نفس شكل B؛ أي،
!>            [ V ] = [ V1 ] [ V2 ]
!>                   [ V1 ] <- مستطيلة بحجم M×(N-L)
!>                   [ V2 ] <- شبه منحرف سفلية بحجم M×L.
!>
!>  تمثل صفوف V المتجهات التي تحدد H(i)'s.
!>
!>  عدد الكتل هو B = سقف(M/MB)، حيث كل
!>  كتلة من الرتبة MB باستثناء الكتلة الأخيرة، والتي تكون من الرتبة
!>  IB = M - (M-1)*MB. لكل من كتل B، يتم حساب عامل المُعكس الكتلي المثلثي العلوي
!>  : T1, T2, ..., TB. يتم تخزين T's بحجم MB×MB (و IB×IB
!>  للأخيرة) في المصفوفة T بحجم MB×N كـ
!>
!>               T = [T1 T2 ... TB].
!>
