subroutine ztrmm (character side, character uplo, character transa, character diag, integer m, integer n, complex*16 alpha, complex*16, dimension(lda,*) a, integer lda, complex*16, dimension(ldb,*) b, integer ldb)

ZTRMM

الغرض:

!>
!> ZTRMM  تقوم بإجراء إحدى عمليات المصفوفات-المصفوفات
!>
!>    B := alpha*op( A )*B,   أو   B := alpha*B*op( A )
!>
!> حيث  alpha  عدد قياسي،  B  مصفوفة  m  في  n ،  A  مصفوفة مثلثية عليا أو سفلى،
!> وحدوية أو غير وحدوية،  و op( A )  هي إحدى العمليات التالية:
!>
!>    op( A ) = A   أو   op( A ) = A**T   أو   op( A ) = A**H.
!>

المعلمات

SIDE 
!>          SIDE هو CHARACTER*1
!>           عند الدخول،  SIDE  تحدد ما إذا كانت  op( A )  تضرب  B  من
!>           اليسار أو اليمين كما يلي:
!>
!>              SIDE = 'L' أو 'l'   B := alpha*op( A )*B.
!>
!>              SIDE = 'R' أو 'r'   B := alpha*B*op( A ).
!>

UPLO

!>          UPLO هو CHARACTER*1
!>           عند الدخول، UPLO  تحدد ما إذا كانت المصفوفة A  مثلثية عليا أو
!>           سفلى كما يلي:
!>
!>              UPLO = 'U' أو 'u'   A  مصفوفة مثلثية عليا.
!>
!>              UPLO = 'L' أو 'l'   A  مصفوفة مثلثية سفلى.
!>

TRANSA

!>          TRANSA هو CHARACTER*1
!>           عند الدخول، TRANSA  تحدد شكل  op( A )  المستخدم في
!>           ضرب المصفوفات كما يلي:
!>
!>              TRANSA = 'N' أو 'n'   op( A ) = A.
!>
!>              TRANSA = 'T' أو 't'   op( A ) = A**T.
!>
!>              TRANSA = 'C' أو 'c'   op( A ) = A**H.
!>

DIAG

!>          DIAG هو CHARACTER*1
!>           عند الدخول، DIAG  تحدد ما إذا كانت A  وحدوية أم لا كما يلي:
!>
!>              DIAG = 'U' أو 'u'   يفترض أن A  وحدوية.
!>
!>              DIAG = 'N' أو 'n'   لا يفترض أن A  وحدوية.
!>

M

!>          M هو INTEGER
!>           عند الدخول،  M  تحدد عدد صفوف  B.  يجب أن يكون  M  أكبر من أو يساوي صفر.
!>

N

!>          N هو INTEGER
!>           عند الدخول،  N  تحدد عدد أعمدة  B.  يجب أن يكون  N  أكبر من أو يساوي صفر.
!>

ALPHA

!>          ALPHA هو COMPLEX*16
!>           عند الدخول،  ALPHA  يحدد العدد القياسي  alpha.  عندما يكون  alpha
!>           صفرًا،  لا يتم الرجوع إلى A  ولا يلزم تعيين B  قبل الدخول.
!>

A

!>          A  هي مصفوفة COMPLEX*16،  البعد ( LDA, k )، حيث k  هو m
!>           عندما  SIDE = 'L' أو 'l'  و n  عندما  SIDE = 'R' أو 'r'.
!>           قبل الدخول مع  UPLO = 'U' أو 'u'،  يجب أن يحتوي الجزء العلوي
!>           المثلثي الذي حجمه  k  في  k  في المصفوفة A  على المصفوفة
!>           المثلثية العليا،  ولا يتم الرجوع إلى الجزء المثلثي السفلي الصارم
!>           من A.
!>           قبل الدخول مع  UPLO = 'L' أو 'l'،  يجب أن يحتوي الجزء السفلي
!>           المثلثي الذي حجمه  k  في  k  في المصفوفة A  على المصفوفة
!>           المثلثية السفلى،  ولا يتم الرجوع إلى الجزء المثلثي العلوي الصارم
!>           من A.
!>           لاحظ أنه عندما  DIAG = 'U' أو 'u'،  لا يتم الرجوع إلى العناصر القطرية
!>           لـ A  أيضًا،  ولكن يفترض أنها الوحدة.
!>

LDA

!>          LDA هو INTEGER
!>           عند الدخول، LDA  تحدد البعد الأول لـ A  كما تم تعريفه
!>           في البرنامج المستدعي (الفرعي).  عندما  SIDE = 'L' أو 'l'،
!>           يجب أن يكون LDA  على الأقل  max( 1, m )،  عندما  SIDE = 'R' أو 'r'،
!>           يجب أن يكون LDA  على الأقل max( 1, n ).
!>

B

!>          B  هي مصفوفة COMPLEX*16،  البعد ( LDB, N ).
!>           قبل الدخول،  يجب أن يحتوي الجزء الرئيسي الذي حجمه  m  في  n
!>           من المصفوفة  B  على المصفوفة  B ،  وعند الخروج  يتم استبدالها
!>           بالمصفوفة المحولة.
!>

LDB

!>          LDB هو INTEGER
!>           عند الدخول، LDB  تحدد البعد الأول لـ B  كما تم تعريفه
!>           في  البرنامج المستدعي (الفرعي).  يجب أن يكون  LDB  على الأقل
!>           max( 1, m ).
!>
