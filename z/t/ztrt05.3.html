توثيق الوظيفة/الروتين الفرعي

subroutine ztrt05 (character uplo, character trans, character diag, integer n, integer nrhs, complex*16, dimension( lda, * ) a, integer lda, complex*16, dimension( ldb, * ) b, integer ldb, complex*16, dimension( ldx, * ) x, integer ldx, complex*16, dimension( ldxact, * ) xact, integer ldxact, double precision, dimension( * ) ferr, double precision, dimension( * ) berr, double precision, dimension( * ) reslts)

ZTRT05

الغرض:

!>
!> ZTRT05 تختبر حدود الخطأ من التنقيح التكراري للحل المحسوب لنظام المعادلات
!> A*X = B، حيث A هي مصفوفة قطرية n × n.
!>
!> RESLTS(1) = اختبار حد الخطأ
!>           = norm(X - XACT) / ( norm(X) * FERR )
!>
!> يتم إرجاع قيمة كبيرة إذا لم تكن هذه النسبة أقل من واحد.
!>
!> RESLTS(2) = الخطأ المتبقي من روتين التنقيح التكراري
!>           = الحد الأقصى لـ BERR / ( (n+1)*EPS + (*) )، حيث
!>             (*) = (n+1)*UNFL / (min_i (abs(A)*abs(X) +abs(b))_i )
!>

المعلمات

UPLO 
!>          UPLO هو CHARACTER*1
!>          يحدد ما إذا كانت المصفوفة A علوية أم سفلية.
!>          = 'U':  علوية
!>          = 'L':  سفلية
!>

TRANS

!>          TRANS هو CHARACTER*1
!>          يحدد شكل نظام المعادلات.
!>          = 'N':  A * X = B  (بدون تبديل)
!>          = 'T':  A'* X = B  (تبديل)
!>          = 'C':  A'* X = B  (تبديل مصاحب = تبديل)
!>

DIAG

!>          DIAG هو CHARACTER*1
!>          يحدد ما إذا كانت المصفوفة A قطرية وحدوية أم لا.
!>          = 'N':  غير وحدوية
!>          = 'U':  وحدوية
!>

N

!>          N هو INTEGER
!>          عدد صفوف المصفوفات X، B، و XACT، وترتيب المصفوفة A.  N >= 0.
!>

NRHS

!>          NRHS هو INTEGER
!>          عدد أعمدة المصفوفات X، B، و XACT.  NRHS >= 0.
!>

A

!>          A هو مصفوفة COMPLEX*16، أبعاد (LDA,N)
!>          المصفوفة القطرية A.  إذا كان UPLO = 'U'، فإن الجزء العلوي
!>          الأمامي من المصفوفة بمقاس n × n يحتوي على المصفوفة العلوية
!>          القطرية، والجزء السفلي القطري الصارم من A غير مرجعي.
!>          إذا كان UPLO = 'L'، فإن الجزء السفلي الأمامي من المصفوفة بمقاس
!>          n × n يحتوي على المصفوفة السفلية القطرية، والجزء العلوي
!>          القطري الصارم من A غير مرجعي.  إذا كان DIAG = 'U'، فإن العناصر
!>          القطرية لـ A غير مرجعية أيضًا ويُفترض أنها 1.
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرئيسي للمصفوفة A.  LDA >= max(1,N).
!>

B

!>          B هو مصفوفة COMPLEX*16، أبعاد (LDB,NRHS)
!>          متجهات الطرف الأيمن لنظام المعادلات الخطية.
!>

LDB

!>          LDB هو INTEGER
!>          البعد الرئيسي للمصفوفة B.  LDB >= max(1,N).
!>

X

!>          X هو مصفوفة COMPLEX*16، أبعاد (LDX,NRHS)
!>          متجهات الحل المحسوبة.  يتم تخزين كل متجه كعمود في المصفوفة X.
!>

LDX

!>          LDX هو INTEGER
!>          البعد الرئيسي للمصفوفة X.  LDX >= max(1,N).
!>

XACT

!>          XACT هو مصفوفة COMPLEX*16، أبعاد (LDXACT,NRHS)
!>          متجهات الحل الدقيقة.  يتم تخزين كل متجه كعمود في المصفوفة XACT.
!>

LDXACT

!>          LDXACT هو INTEGER
!>          البعد الرئيسي للمصفوفة XACT.  LDXACT >= max(1,N).
!>

FERR

!>          FERR هي مصفوفة DOUBLE PRECISION، أبعاد (NRHS)
!>          تقديرات حدود الخطأ الأمامي لكل متجه حل X.  إذا كان XTRUE هو الحل
!>          الحقيقي، فإن FERR يحد من حجم أكبر إدخال في (X - XTRUE) مقسومًا
!>          على حجم أكبر إدخال في X.
!>

BERR

!>          BERR هي مصفوفة DOUBLE PRECISION، أبعاد (NRHS)
!>          الخطأ الخلفي النسبي المكوني لكل متجه حل (أي، أصغر تغيير نسبي في
!>          أي إدخال في A أو B يجعل X حلاً دقيقًا).
!>

RESLTS

!>          RESLTS هي مصفوفة DOUBLE PRECISION، أبعاد (2)
!>          الحد الأقصى لمتجهات الحل NRHS للنسب التالية:
!>          RESLTS(1) = norm(X - XACT) / ( norm(X) * FERR )
!>          RESLTS(2) = BERR / ( (n+1)*EPS + (*) )
!>
