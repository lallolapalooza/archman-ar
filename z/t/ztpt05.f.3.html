subroutine ztpt05 (character uplo, character trans, character diag, integer n, integer nrhs, complex*16, dimension( * ) ap, complex*16, dimension( ldb, * ) b, integer ldb, complex*16, dimension( ldx, * ) x, integer ldx, complex*16, dimension( ldxact, * ) xact, integer ldxact, double precision, dimension( * ) ferr, double precision, dimension( * ) berr, double precision, dimension( * ) reslts)

ZTPT05

الغرض:

!>
!> ZTPT05 تختبر حدود الخطأ من التكرار التكراري لحل النظام
!> AX = B، حيث A مصفوفة مثلثية بتخزين مضغوط.
!>
!> RESLTS(1) = اختبار حد الخطأ
!>           = norm(X - XACT) / ( norm(X) * FERR )
!>
!> يتم إرجاع قيمة كبيرة إذا لم تكن هذه النسبة أقل من واحد.
!>
!> RESLTS(2) = الباقي من روتين التكرار التكراري
!>           = الحد الأقصى لـ BERR / ( (n+1)*EPS + (*) )، حيث
!>             (*) = (n+1)*UNFL / (min_i (abs(A)*abs(X) +abs(b))_i )
!>

المعلمات

UPLO 
!>          UPLO هي CHARACTER*1
!>          تحدد ما إذا كانت المصفوفة A مثلثية علوية أم سفلية.
!>          = 'U':  مثلثية علوية
!>          = 'L':  مثلثية سفلية
!>

TRANS

!>          TRANS هي CHARACTER*1
!>          تحدد شكل نظام المعادلات.
!>          = 'N':  A * X = B  (بدون تبديل)
!>          = 'T':  A'* X = B  (تبديل)
!>          = 'C':  A'* X = B  (تبديل مرافق = تبديل)
!>

DIAG

!>          DIAG هي CHARACTER*1
!>          تحدد ما إذا كانت المصفوفة A مثلثية وحدوية أم لا.
!>          = 'N':  غير وحدوية
!>          = 'U':  وحدوية
!>

N

!>          N هو INTEGER
!>          عدد الصفوف في المصفوفات X، B، و XACT، وترتيب المصفوفة A.
!>          N >= 0.
!>

NRHS

!>          NRHS هو INTEGER
!>          عدد الأعمدة في المصفوفات X، B، و XACT.
!>          NRHS >= 0.
!>

AP

!>          AP هي مصفوفة COMPLEX*16، البعد (N*(N+1)/2)
!>          المصفوفة المثلثية العلوية أو السفلية A، مضغوطة حسب الأعمدة في
!>          مصفوفة خطية. يتم تخزين العمود j من A في المصفوفة
!>          AP كالتالي:
!>          إذا كان UPLO = 'U'، فإن AP(i + (j-1)*j/2) = A(i,j) لـ 1<=i<=j؛
!>          إذا كان UPLO = 'L'، فإن AP(i + (j-1)*(2n-j)/2) = A(i,j) لـ j<=i<=n.
!>          إذا كان DIAG = 'U'، فإن العناصر القطرية لـ A لا تتم الإشارة إليها
!>          ويفترض أنها 1.
!>

B

!>          B هي مصفوفة COMPLEX*16، البعد (LDB,NRHS)
!>          متجهات الطرف الأيمن لنظام المعادلات الخطية.
!>

LDB

!>          LDB هو INTEGER
!>          البعد الرائد للمصفوفة B. LDB >= max(1,N).
!>

X

!>          X هي مصفوفة COMPLEX*16، البعد (LDX,NRHS)
!>          متجهات الحل المحسوبة. يتم تخزين كل متجه كعمود في
!>          المصفوفة X.
!>

LDX

!>          LDX هو INTEGER
!>          البعد الرائد للمصفوفة X. LDX >= max(1,N).
!>

XACT

!>          XACT هي مصفوفة COMPLEX*16، البعد (LDXACT,NRHS)
!>          متجهات الحل الدقيقة. يتم تخزين كل متجه كعمود في
!>          المصفوفة XACT.
!>

LDXACT

!>          LDXACT هو INTEGER
!>          البعد الرائد للمصفوفة XACT. LDXACT >= max(1,N).
!>

FERR

!>          FERR هي مصفوفة DOUBLE PRECISION، البعد (NRHS)
!>          حدود الخطأ الأمامي المقدرة لكل متجه حل
!>          X. إذا كان XTRUE هو الحل الحقيقي، فإن FERR يحدد حجم
!>          أكبر مدخل في (X - XTRUE) مقسومًا على حجم
!>          أكبر مدخل في X.
!>

BERR

!>          BERR هي مصفوفة DOUBLE PRECISION، البعد (NRHS)
!>          الخطأ الراجعي النسبي المكوني لكل متجه حل
!>          (أي، أصغر تغيير نسبي في أي مدخل من A
!>          أو B يجعل X حلاً دقيقًا).
!>

RESLTS

!>          RESLTS هي مصفوفة DOUBLE PRECISION، البعد (2)
!>          الحد الأقصى عبر متجهات الحل NRHS للنسب:
!>          RESLTS(1) = norm(X - XACT) / ( norm(X) * FERR )
!>          RESLTS(2) = BERR / ( (n+1)*EPS + (*) )
!>
