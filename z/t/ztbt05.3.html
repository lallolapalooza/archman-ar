توثيق الوظيفة/الإجراء الفرعي

subroutine ztbt05 (character uplo, character trans, character diag, integer n, integer kd, integer nrhs, complex*16, dimension( ldab, * ) ab, integer ldab, complex*16, dimension( ldb, * ) b, integer ldb, complex*16, dimension( ldx, * ) x, integer ldx, complex*16, dimension( ldxact, * ) xact, integer ldxact, double precision, dimension( * ) ferr, double precision, dimension( * ) berr, double precision, dimension( * ) reslts)

ZTBT05

الغرض:

!>
!> ZTBT05 تختبر حدود الخطأ من التنقيح التكراري للحل المحسوب لنظام المعادلات A*X = B،
!> حيث A هي مصفوفة نطاق مثلثية.
!>
!> RESLTS(1) = اختبار حد الخطأ
!>           = norm(X - XACT) / ( norm(X) * FERR )
!>
!> يتم إرجاع قيمة كبيرة إذا لم تكن هذه النسبة أقل من واحد.
!>
!> RESLTS(2) = الأثر المتبقي من روتين التنقيح التكراري
!>           = الحد الأقصى لـ BERR / ( NZ*EPS + (*) )، حيث
!>             (*) = NZ*UNFL / (min_i (abs(A)*abs(X) +abs(b))_i )
!>             و NZ = الحد الأقصى. عدد العناصر غير الصفرية في أي صف من A، بالإضافة إلى 1
!>

المعلمات

UPLO 
!>          UPLO هو CHARACTER*1
!>          يحدد ما إذا كانت المصفوفة A مثلثية عليا أم سفلى.
!>          = 'U':  مثلثية عليا
!>          = 'L':  مثلثية سفلى
!>

TRANS

!>          TRANS هو CHARACTER*1
!>          يحدد شكل نظام المعادلات.
!>          = 'N':  A * X = B  (بدون تبديل)
!>          = 'T':  A'* X = B  (تبديل)
!>          = 'C':  A'* X = B  (تبديل مرافق = تبديل)
!>

DIAG

!>          DIAG هو CHARACTER*1
!>          يحدد ما إذا كانت المصفوفة A مثلثية وحدوية أم لا.
!>          = 'N':  غير وحدوية
!>          = 'U':  وحدوية
!>

N

!>          N هو INTEGER
!>          عدد صفوف المصفوفات X، B، و XACT، ورتبة المصفوفة A.  N >= 0.
!>

KD

!>          KD هو INTEGER
!>          عدد الأقطار الفائقة للمصفوفة A إذا كان UPLO = 'U'،
!>          أو عدد الأقطار الفرعية إذا كان UPLO = 'L'.  KD >= 0.
!>

NRHS

!>          NRHS هو INTEGER
!>          عدد أعمدة المصفوفات X، B، و XACT.
!>          NRHS >= 0.
!>

AB

!>          AB هي مصفوفة COMPLEX*16، بأبعاد (LDAB,N)
!>          المصفوفة A المثلثية النطاقية العلوية أو السفلية، مخزنة في
!>          أول kd+1 صفوف من المصفوفة. يتم تخزين العمود j من A
!>          في العمود j من المصفوفة AB على النحو التالي:
!>          إذا كان UPLO = 'U'، فإن AB(kd+1+i-j,j) = A(i,j) لـ max(1,j-kd)<=i<=j؛
!>          إذا كان UPLO = 'L'، فإن AB(1+i-j,j)    = A(i,j) لـ j<=i<=min(n,j+kd).
!>          إذا كان DIAG = 'U'، فلا يتم الرجوع إلى العناصر القطرية لـ A
!>          ويفترض أنها 1.
!>

LDAB

!>          LDAB هو INTEGER
!>          البعد الرائد للمصفوفة AB.  LDAB >= KD+1.
!>

B

!>          B هي مصفوفة COMPLEX*16، بأبعاد (LDB,NRHS)
!>          متجهات الطرف الأيمن لنظام المعادلات الخطية.
!>

LDB

!>          LDB هو INTEGER
!>          البعد الرائد للمصفوفة B.  LDB >= max(1,N).
!>

X

!>          X هي مصفوفة COMPLEX*16، بأبعاد (LDX,NRHS)
!>          متجهات الحل المحسوبة. يتم تخزين كل متجه كـ
!>          عمود في المصفوفة X.
!>

LDX

!>          LDX هو INTEGER
!>          البعد الرائد للمصفوفة X.  LDX >= max(1,N).
!>

XACT

!>          XACT هي مصفوفة COMPLEX*16، بأبعاد (LDXACT,NRHS)
!>          متجهات الحل الدقيقة. يتم تخزين كل متجه كـ
!>          عمود في المصفوفة XACT.
!>

LDXACT

!>          LDXACT هو INTEGER
!>          البعد الرائد للمصفوفة XACT.  LDXACT >= max(1,N).
!>

FERR

!>          FERR هي مصفوفة DOUBLE PRECISION، بأبعاد (NRHS)
!>          حدود الخطأ الأمامي المقدرة لكل متجه حل
!>          X. إذا كان XTRUE هو الحل الحقيقي، فإن FERR يحد من حجم
!>          أكبر عنصر في (X - XTRUE) مقسومًا على حجم
!>          أكبر عنصر في X.
!>

BERR

!>          BERR هي مصفوفة DOUBLE PRECISION، بأبعاد (NRHS)
!>          الخطأ الخلفي النسبي المكون من كل متجه حل
!>          (أي، أصغر تغيير نسبي في أي عنصر من A
!>          أو B يجعل X حلاً دقيقًا).
!>

RESLTS

!>          RESLTS هي مصفوفة DOUBLE PRECISION، بأبعاد (2)
!>          الحد الأقصى لمتجهات الحل NRHS للنسب:
!>          RESLTS(1) = norm(X - XACT) / ( norm(X) * FERR )
!>          RESLTS(2) = BERR / ( NZ*EPS + (*) )
!>
