وثائق الوظيفة/الإجراء الفرعي

subroutine ztftri (character transr, character uplo, character diag, integer n, complex*16, dimension( 0: * ) a, integer info)

ZTFTRI

الغرض:

!>
!> ZTFTRI يحسب معكوس مصفوفة مثلثية A مخزنة بتنسيق RFP
!>
!> هذا إصدار من المستوى 3 BLAS للخوارزمية.
!>

المعاملات

TRANSR 
!>          TRANSR هو CHARACTER*1
!>          = 'N':  تنسيق RFP A العادي مخزن؛
!>          = 'C':  تنسيق RFP A المنقول بالاقتران مخزن.
!>

UPLO

!>          UPLO هو CHARACTER*1
!>          = 'U':  A مثلثية علوية؛
!>          = 'L':  A مثلثية سفلية.
!>

DIAG

!>          DIAG هو CHARACTER*1
!>          = 'N':  A غير موحدة مثلثية؛
!>          = 'U':  A موحدة مثلثية.
!>

N

!>          N هو INTEGER
!>          رتبة المصفوفة A.  N >= 0.
!>

A

!>          A هو مصفوفة COMPLEX*16، البعد ( N*(N+1)/2 )؛
!>          عند الدخول، المصفوفة المثلثية A بتنسيق RFP. تنسيق RFP
!>          موصوف بواسطة TRANSR و UPLO و N كما يلي: إذا كان TRANSR =
!>          'N' فإن RFP A هو (0:N,0:k-1) عندما يكون N زوجيًا؛ k=N/2. RFP A هو
!>          (0:N-1,0:k) عندما يكون N فرديًا؛ k=N/2. إذا كان TRANSR = 'C' فإن RFP هو
!>          المنقول بالاقتران لـ RFP A المعرف عندما يكون
!>          TRANSR = 'N'. محتويات RFP A معرفة بواسطة UPLO كما يلي: إذا كان UPLO =
!>          'U' فإن RFP A يحتوي على العناصر nt للمصفوفة A العلوية المعبأة؛ إذا كان UPLO =
!>          'L' فإن RFP A يحتوي على العناصر nt للمصفوفة A السفلية المعبأة.
!>          LDA لـ RFP A هو (N+1)/2 عندما يكون TRANSR = 'C'. عندما يكون TRANSR = 'N' فإن LDA هو N+1 عندما يكون N زوجيًا وعندما يكون N فرديًا. انظر الملاحظة أدناه لمزيد من التفاصيل.
!>
!>          عند الخروج، معكوس المصفوفة الأصلية (المثلثية)، بنفس تنسيق التخزين.
!>

INFO

!>          INFO هو INTEGER
!>          = 0: خروج ناجح
!>          < 0: إذا كان INFO = -i، فإن المعامل i كان له قيمة غير قانونية
!>          > 0: إذا كان INFO = i، فإن A(i,i) يساوي صفرًا تمامًا. المصفوفة
!>               المثلثية منفردة ولا يمكن حساب معكوسها.
!>

!>
!>  ننظر أولاً إلى التنسيق المعبأ القياسي عندما يكون N زوجيًا.
!>  نعطي مثالًا حيث N = 6.
!>
!>      AP هو علوي             AP هو سفلي
!>
!>   00 01 02 03 04 05       00
!>      11 12 13 14 15       10 11
!>         22 23 24 25       20 21 22
!>            33 34 35       30 31 32 33
!>               44 45       40 41 42 43 44
!>                  55       50 51 52 53 54 55
!>
!>
!>  لتكن TRANSR = 'N'. RFP يحتوي على AP كما يلي:
!>  بالنسبة لـ UPLO = 'U'، الرباعي العلوي A(0:5,0:2) يتكون من الأعمدة الثلاثة الأخيرة
!>  من AP العلوي. المثلث السفلي A(4:6,0:2) يتكون من
!>  المنقول بالاقتران للأعمدة الثلاثة الأولى من AP العلوي.
!>  بالنسبة لـ UPLO = 'L'، الرباعي السفلي A(1:6,0:2) يتكون من الأعمدة الثلاثة الأولى
!>  من AP السفلي. المثلث العلوي A(0:2,0:2) يتكون من
!>  المنقول بالاقتران للأعمدة الثلاثة الأخيرة من AP السفلي.
!>  للدلالة على الاقتران، نضع -- فوق العنصر. هذا يغطي
!>  الحالة N زوجي و TRANSR = 'N'.
!>
!>         RFP A                   RFP A
!>
!>                                -- -- --
!>        03 04 05                33 43 53
!>                                   -- --
!>        13 14 15                00 44 54
!>                                      --
!>        23 24 25                10 11 55
!>
!>        33 34 35                20 21 22
!>        --
!>        00 44 45                30 31 32
!>        -- --
!>        01 11 55                40 41 42
!>        -- -- --
!>        02 12 22                50 51 52
!>
!>  الآن لتكن TRANSR = 'C'. RFP A في كلتا حالتي UPLO هو ببساطة المنقول
!>  بالاقتران لـ RFP A أعلاه. لذلك نحصل على:
!>
!>
!>           RFP A                   RFP A
!>
!>     -- -- -- --                -- -- -- -- -- --
!>     03 13 23 33 00 01 02    33 00 10 20 30 40 50
!>     -- -- -- -- --                -- -- -- -- --
!>     04 14 24 34 44 11 12    43 44 11 21 31 41 51
!>     -- -- -- -- -- --                -- -- -- --
!>     05 15 25 35 45 55 22    53 54 55 22 32 42 52
!>
!>
!>  ننظر بعد ذلك إلى التنسيق المعبأ القياسي عندما يكون N فرديًا.
!>  نعطي مثالًا حيث N = 5.
!>
!>     AP هو علوي                 AP هو سفلي
!>
!>   00 01 02 03 04              00
!>      11 12 13 14              10 11
!>         22 23 24              20 21 22
!>            33 34              30 31 32 33
!>               44              40 41 42 43 44
!>
!>
!>  لتكن TRANSR = 'N'. RFP يحتوي على AP كما يلي:
!>  بالنسبة لـ UPLO = 'U'، الرباعي العلوي A(0:4,0:2) يتكون من الأعمدة الثلاثة الأخيرة
!>  من AP العلوي. المثلث السفلي A(3:4,0:1) يتكون من
!>  المنقول بالاقتران للأعمدة الاثنتين الأولى من AP العلوي.
!>  بالنسبة لـ UPLO = 'L'، الرباعي السفلي A(0:4,0:2) يتكون من الأعمدة الثلاثة الأولى
!>  من AP السفلي. المثلث العلوي A(0:1,1:2) يتكون من
!>  المنقول بالاقتران للأعمدة الاثنتين الأخيرة من AP السفلي.
!>  للدلالة على الاقتران، نضع -- فوق العنصر. هذا يغطي
!>  الحالة N فردي و TRANSR = 'N'.
!>
!>         RFP A                   RFP A
!>

!>        02 03 04                00 33 43

!>        12 13 14                10 11 44
