وثائق الوظيفة/الروتين الفرعي

subroutine ztbcon (character norm, character uplo, character diag, integer n, integer kd, complex*16, dimension( ldab, * ) ab, integer ldab, double precision rcond, complex*16, dimension( * ) work, double precision, dimension( * ) rwork, integer info)

ZTBCON

الغرض:

!>
!> ZTBCON يقدر مقلوب عدد الشرط لمصفوفة فرقة مثلثية A، إما في المعيار 1 أو المعيار اللانهائي.
!>
!> يتم حساب معيار A ويتم الحصول على تقدير لـ
!> norm(inv(A))، ثم يتم حساب مقلوب عدد الشرط على أنه
!>    RCOND = 1 / ( norm(A) * norm(inv(A)) ).
!>

المعلمات

NORM 
!>          NORM هو CHARACTER*1
!>          يحدد ما إذا كان عدد الشرط 1-norm أو infinity-norm مطلوبًا:
!>          = '1' أو 'O':  1-norm؛
!>          = 'I':         Infinity-norm.
!>

UPLO

!>          UPLO هو CHARACTER*1
!>          = 'U':  A مثلثية علوية؛
!>          = 'L':  A مثلثية سفلية.
!>

DIAG

!>          DIAG هو CHARACTER*1
!>          = 'N':  A غير أحادية مثلثية؛
!>          = 'U':  A أحادية مثلثية.
!>

N

!>          N هو INTEGER
!>          ترتيب المصفوفة A.  N >= 0.
!>

KD

!>          KD هو INTEGER
!>          عدد الأقطار الفائقة أو الأقطار الفرعية للمصفوفة الفرقة المثلثية A.  KD >= 0.
!>

AB

!>          AB هي مصفوفة COMPLEX*16، البعد (LDAB,N)
!>          المصفوفة المثلثية الفرقة العلوية أو السفلية A، مخزنة في
!>          الصفوف kd+1 الأولى من المصفوفة. يتم تخزين العمود j من A
!>          في العمود j من المصفوفة AB كما يلي:
!>          إذا كان UPLO = 'U'، فإن AB(kd+1+i-j,j) = A(i,j) لـ max(1,j-kd)<=i<=j؛
!>          إذا كان UPLO = 'L'، فإن AB(1+i-j,j)    = A(i,j) لـ j<=i<=min(n,j+kd).
!>          إذا كان DIAG = 'U'، فلا يتم الرجوع إلى العناصر القطرية لـ A
!>          ويفترض أنها 1.
!>

LDAB

!>          LDAB هو INTEGER
!>          البعد الرئيسي للمصفوفة AB.  LDAB >= KD+1.
!>

RCOND

!>          RCOND هو DOUBLE PRECISION
!>          مقلوب عدد شرط المصفوفة A،
!>          محسوبًا كـ RCOND = 1/(norm(A) * norm(inv(A))).
!>

WORK

!>          WORK هي مصفوفة COMPLEX*16، البعد (2*N)
!>

RWORK

!>          RWORK هي مصفوفة DOUBLE PRECISION، البعد (N)
!>

INFO

!>          INFO هو INTEGER
!>          = 0:  انتهاء ناجح
!>          < 0:  إذا كان INFO = -i، فإن المعلمة i كان لها قيمة غير قانونية
!>
