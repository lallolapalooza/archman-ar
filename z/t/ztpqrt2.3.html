subroutine ztpqrt2 (integer m, integer n, integer l, complex*16, dimension( lda, * ) a, integer lda, complex*16, dimension( ldb, * ) b, integer ldb, complex*16, dimension( ldt, * ) t, integer ldt, integer info)

ZTPQRT2 تحسب تحليلاً QR لمصفوفة "مثلثة-خماسية" حقيقية أو معقدة، تتكون من كتلة مثلثة وكتلة خماسية، باستخدام تمثيل WY المدمج لـ Q.

الغرض:

!>
!> ZTPQRT2 تحسب تحليلاً QR لمصفوفة معقدة C، تتكون من كتلة مثلثة A وكتلة خماسية B،
!> باستخدام تمثيل WY المدمج لـ Q.
!>

تفاصيل إضافية:

!>
!> المصفوفة المدخلة C هي مصفوفة (N+M) × N
!>
!>               C = [ A ]
!>                   [ B ]
!>
!> حيث A هي مصفوفة مثلثة عليا N × N، و B هي مصفوفة خماسية M × N تتكون من مصفوفة مستطيلة (M-L) × N تسمى B1 في الأعلى مصفوفة شبه منحرفة عليا L × N تسمى B2:
!>
!>               B = [ B1 ]  <- (M-L) × N مستطيلة
!>                   [ B2 ]  <-     L × N شبه منحرفة عليا.
!>
!> تتكون المصفوفة شبه المنحرفة العليا B2 من أول L صفوف لمصفوفة مثلثة عليا N × N، حيث 0 <= L <= MIN(M,N). إذا كان L=0، فإن
!> B مستطيلة M × N؛ إذا كان M=L=N، فإن B مثلثة عليا.
!>
!> المصفوفة W تخزن الموجهات الأولية H(i) في العمود i
!> أسفل القطر (في A) في المصفوفة المدخلة C ذات الأبعاد (M+N) × N
!>
!>               C = [ A ]  <- مثلثة عليا N × N
!>                   [ B ]  <- خماسية M × N
!>
!> بحيث يمكن تمثيل W على النحو التالي:
!>
!>               W = [ I ]  <- هوية، N × N
!>                   [ V ]  <- M × N، بنفس شكل B.
!>
!> وبالتالي، فإن جميع المعلومات اللازمة لـ W موجودة عند الخروج في B،
!> والتي نسميها V أعلاه. لاحظ أن V لها نفس شكل B؛ أي،
!>
!>               V = [ V1 ] <- (M-L) × N مستطيلة
!>                   [ V2 ] <-     L × N شبه منحرفة عليا.
!>
!> تمثل أعمدة V المتجهات التي تحدد H(i).
!> الموجه الكتلي H ذو الأبعاد (M+N) × (M+N) يُعطى بواسطة
!>
!>               H = I - W * T * W**H
!>
!> حيث W**H هو المرافق المتعامد لـ W و T هو العامل المثلث العلوي
!> للموجه الكتلي.
!>
