توثيق الوظيفة/الوحدة الفرعية

subroutine ztrcon (character norm, character uplo, character diag, integer n, complex*16, dimension( lda, * ) a, integer lda, double precision rcond, complex*16, dimension( * ) work, double precision, dimension( * ) rwork, integer info)

ZTRCON

الغرض:

!>
!> ZTRCON تقدّر مقلوب رقم الشرط لمصفوفة مثلثية A، إما بمعيار 1 أو معيار اللانهاية.
!>
!> يتم حساب معيار A ويتم الحصول على تقدير لـ
!> norm(inv(A))، ثم يتم حساب مقلوب رقم الشرط على أنه
!> RCOND = 1 / ( norm(A) * norm(inv(A)) ).
!>

المعلمات

NORM 
!>          NORM is CHARACTER*1
!>          تحدد ما إذا كان مطلوبًا رقم الشرط بمعيار 1 أو رقم الشرط بمعيار اللانهاية:
!>          = '1' or 'O':  معيار 1؛
!>          = 'I':         معيار اللانهاية.
!>

UPLO

!>          UPLO is CHARACTER*1
!>          = 'U':  A مثلثية عليا؛
!>          = 'L':  A مثلثية سفلى.
!>

DIAG

!>          DIAG is CHARACTER*1
!>          = 'N':  A غير وحيدة مثلثية؛
!>          = 'U':  A وحيدة مثلثية.
!>

N

!>          N is INTEGER
!>          رتبة المصفوفة A.  N >= 0.
!>

A

!>          A is COMPLEX*16 array, dimension (LDA,N)
!>          المصفوفة المثلثية A.  إذا كان UPLO = 'U'، فإن الجزء المثلثي العلوي
!>          الرئيسي N-by-N من المصفوفة A يحتوي على المصفوفة المثلثية العليا، ولا تتم
!>          الإشارة إلى الجزء المثلثي السفلي الصارم من A.  إذا كان UPLO = 'L'، فإن الجزء
!>          المثلثي السفلي الرئيسي N-by-N من المصفوفة A يحتوي على المصفوفة المثلثية
!>          السفلى، ولا تتم الإشارة إلى الجزء المثلثي العلوي الصارم من A.  إذا كان
!>          DIAG = 'U'، فلا تتم الإشارة إلى العناصر القطرية لـ A ويُفترض أنها 1.
!>

LDA

!>          LDA is INTEGER
!>          البعد الرئيسي للمصفوفة A.  LDA >= max(1,N).
!>

RCOND

!>          RCOND is DOUBLE PRECISION
!>          مقلوب رقم الشرط للمصفوفة A،
!>          محسوبًا على أنه RCOND = 1/(norm(A) * norm(inv(A))).
!>

WORK

!>          WORK is COMPLEX*16 array, dimension (2*N)
!>

RWORK

!>          RWORK is DOUBLE PRECISION array, dimension (N)
!>

INFO

!>          INFO is INTEGER
!>          = 0:  خروج ناجح
!>          < 0:  إذا كان INFO = -i، فإن المعلمة i كانت لها قيمة غير قانونية
!>
