subroutine ztbtrs (character uplo, character trans, character diag, integer n, integer kd, integer nrhs, complex*16, dimension( ldab, * ) ab, integer ldab, complex*16, dimension( ldb, * ) b, integer ldb, integer info)

ZTBTRS

الغرض:

!>
!> ZTBTRS يحل نظامًا مثلثيًا بالشكل:
!>
!>    A * X = B,  A**T * X = B,  أو  A**H * X = B,
!>
!> حيث A هي مصفوفة مثلثية نطاقية من الدرجة N، و B هي مصفوفة N-by-NRHS.
!>
!> يتحقق هذا الروتين الفرعي من أن A غير قابلة للانعكاس، ولكن يجب على المستدعين ملاحظة أن
!> الانفرادية المطلقة فقط هي التي يتم اكتشافها. من الممكن أن تكون عناصر قطرية واحدة أو أكثر
!> من A أرقامًا صغيرة جدًا دون أن يشير هذا الروتين الفرعي إلى خطأ.
!>
!> إذا كان فقدان الدقة العددية بسبب المصفوفات شبه المنفردة مدعاة للقلق،
!> يجب على المستدعي التحقق من أن A غير قابلة للانعكاس ضمن هامش معين قبل استدعاء هذا الروتين الفرعي.
!>

المعلمات

UPLO 
!>          UPLO هو CHARACTER*1
!>          = 'U':  A هي مثلثية عليا؛
!>          = 'L':  A هي مثلثية سفلى.
!>

TRANS

!>          TRANS هو CHARACTER*1
!>          يحدد شكل نظام المعادلات:
!>          = 'N':  A * X = B     (بدون تبديل)
!>          = 'T':  A**T * X = B  (تبديل)
!>          = 'C':  A**H * X = B  (تبديل مترافق)
!>

DIAG

!>          DIAG هو CHARACTER*1
!>          = 'N':  A هي مثلثية غير وحدوية؛
!>          = 'U':  A هي مثلثية وحدوية.
!>

N

!>          N هو INTEGER
!>          درجة المصفوفة A.  N >= 0.
!>

KD

!>          KD هو INTEGER
!>          عدد الأقطار الفائقة أو الأقطار الفرعية للمصفوفة
!>          المثلثية النطاقية A.  KD >= 0.
!>

NRHS

!>          NRHS هو INTEGER
!>          عدد الأطراف اليمنى، أي عدد أعمدة
!>          المصفوفة B.  NRHS >= 0.
!>

AB

!>          AB هي مصفوفة COMPLEX*16، البعد (LDAB,N)
!>          المصفوفة المثلثية العليا أو السفلى A، مخزنة في
!>          الأسطر kd+1 الأولى من AB.  يتم تخزين العمود j من A
!>          في العمود j من المصفوفة AB على النحو التالي:
!>          إذا كان UPLO = 'U'، فإن AB(kd+1+i-j,j) = A(i,j) لـ max(1,j-kd)<=i<=j؛
!>          إذا كان UPLO = 'L'، فإن AB(1+i-j,j)    = A(i,j) لـ j<=i<=min(n,j+kd).
!>          إذا كان DIAG = 'U'، فلا يتم الرجوع إلى العناصر القطرية لـ A
!>          ويفترض أنها 1.
!>

LDAB

!>          LDAB هو INTEGER
!>          البعد الأمامي للمصفوفة AB.  LDAB >= KD+1.
!>

B

!>          B هي مصفوفة COMPLEX*16، البعد (LDB,NRHS)
!>          عند الإدخال، مصفوفة الطرف الأيمن B.
!>          عند الخروج، إذا كان INFO = 0، فإن مصفوفة الحل X.
!>

LDB

!>          LDB هو INTEGER
!>          البعد الأمامي للمصفوفة B.  LDB >= max(1,N).
!>

INFO

!>          INFO هو INTEGER
!>          = 0:  خروج ناجح
!>          < 0:  إذا كان INFO = -i، فإن المعلمة i كانت لها قيمة غير قانونية
!>          > 0:  إذا كان INFO = i، فإن العنصر القطري i لـ A هو صفر تمامًا،
!>                مما يشير إلى أن المصفوفة منفردة و
!>                لم يتم حساب الحلول X.
!>
