subroutine ztzrqf (integer m, integer n, complex*16, dimension( lda, * ) a, integer lda, complex*16, dimension( * ) tau, integer info)

ZTZRQF

الغرض:

!>
!> تم إهمال هذا الإجراء الفرعي واستبداله بالإجراء الفرعي ZTZRZF.
!>
!> ZTZRQF يقلل مصفوفة A شبه المنحرفة العلوية المعقدة من الحجم M × N ( M<=N )
!> إلى شكل مثلثي علوي بواسطة تحويلات وحدوية.
!>
!> يتم تحليل مصفوفة A شبه المنحرفة على النحو التالي:
!>
!>    A = ( R  0 ) * Z,
!>
!> حيث Z هي مصفوفة وحدوية N × N و R هي مصفوفة مثلثية علوية M × M.
!>

المعلمات

M 
!>          M هو عدد صحيح (INTEGER)
!>          عدد الصفوف في المصفوفة A.  M >= 0.
!>

N

!>          N هو عدد صحيح (INTEGER)
!>          عدد الأعمدة في المصفوفة A.  N >= M.
!>

A

!>          A هو مصفوفة من النوع COMPLEX*16، البعد (LDA,N)
!>          عند الإدخال، يجب أن يحتوي الجزء العلوي شبه المنحرف من المصفوفة A
!>          بحجم M × N على المصفوفة المطلوب تحليلها.
!>          عند الخروج، يحتوي الجزء المثلثي العلوي من A بحجم M × M على
!>          المصفوفة المثلثية العلوية R، والعناصر من M+1 إلى N في الصفوف M الأولى من A،
!>          مع المصفوفة TAU، تمثل المصفوفة الوحدوية Z كناتج لـ M محولات أساسية.
!>

LDA

!>          LDA هو عدد صحيح (INTEGER)
!>          البعد الرائد للمصفوفة A.  LDA >= max(1,M).
!>

TAU

!>          TAU هي مصفوفة من النوع COMPLEX*16، البعد (M)
!>          العوامل القياسية للمحولات الأساسية.
!>

INFO

!>          INFO هو عدد صحيح (INTEGER)
!>          = 0: خروج ناجح
!>          < 0: إذا كان INFO = -i، كان المعامل i غير قانوني
!>

تفاصيل إضافية:

!>
!>  يتم الحصول على التحليل بطريقة هاوسهولدر. مصفوفة التحويل k، Z( k )،
!>  والتي يتم استخدام مترافقها النقلي لإدخال الأصفار في الصف (m - k + 1) من A،
!>  تُعطى بالشكل التالي:
!>
!>     Z( k ) = ( I     0   ),
!>              ( 0  T( k ) )
!>
!>  حيث:
!>
!>     T( k ) = I - tau*u( k )*u( k )**H,   u( k ) = (   1    ),
!>                                                   (   0    )
!>                                                   ( z( k ) )
!>
!>  tau هو عدد قياسي و z( k ) هو متجه من ( n - m ) عنصر.
!>  يتم اختيار tau و z( k ) لإلغاء عناصر الصف k من X.
!>
!>  يتم إرجاع العدد القياسي tau في العنصر k من TAU والمتجه u( k )
!>  في الصف k من A، بحيث تكون عناصر z( k ) موجودة في a( k, m + 1 )، ...، a( k, n ).
!>  يتم إرجاع عناصر R في الجزء المثلثي العلوي من A.
!>
!>  Z مُعطاة بواسطة:
!>
!>     Z =  Z( 1 ) * Z( 2 ) * ... * Z( m ).
!>
