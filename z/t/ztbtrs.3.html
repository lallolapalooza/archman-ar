subroutine ztbtrs (character uplo, character trans, character diag, integer n, integer kd, integer nrhs, complex*16, dimension( ldab, * ) ab, integer ldab, complex*16, dimension( ldb, * ) b, integer ldb, integer info)

ZTBTRS

الغرض:

!>
!> ZTBTRS يحل نظامًا مثلثيًا بالشكل
!>
!>    A * X = B,  A**T * X = B,  أو  A**H * X = B,
!>
!> حيث A هي مصفوفة نطاق مثلثي من الدرجة N، و B هي مصفوفة N-by-NRHS.
!>
!> يتحقق هذا الروتين الفرعي من أن A غير مفردة، ولكن يجب أن يلاحظ المتصلون أنه لا يتم اكتشاف
!> التفرد إلا بدقة. من الممكن أن تكون عناصر قطرية واحدة أو أكثر لـ A أرقامًا صغيرة جدًا
!> دون أن يشير هذا الروتين الفرعي إلى خطأ.
!>
!> إذا كان فقدان الدقة العددية بسبب المصفوفات شبه المفردة يمثل مصدر قلق، فيجب على
!> المتصل التأكد من أن A غير مفردة ضمن هامش تسامح معين قبل استدعاء هذا الروتين الفرعي.
!>

المعلمات

UPLO 
!>          UPLO is CHARACTER*1
!>          = 'U':  A is upper triangular; (A مثلثة علوية)
!>          = 'L':  A is lower triangular. (A مثلثة سفلية)
!>

TRANS

!>          TRANS is CHARACTER*1
!>          Specifies the form of the system of equations: (يحدد شكل نظام المعادلات)
!>          = 'N':  A * X = B     (No transpose) (بدون تبديل)
!>          = 'T':  A**T * X = B  (Transpose) (تبديل)
!>          = 'C':  A**H * X = B  (Conjugate transpose) (تبديل مترافق)
!>

DIAG

!>          DIAG is CHARACTER*1
!>          = 'N':  A is non-unit triangular; (A مثلثية غير موحدة)
!>          = 'U':  A is unit triangular. (A مثلثية موحدة)
!>

N

!>          N is INTEGER
!>          The order of the matrix A.  N >= 0. (درجة المصفوفة A. N >= 0.)
!>

KD

!>          KD is INTEGER
!>          The number of superdiagonals or subdiagonals of the
!>          triangular band matrix A.  KD >= 0. (عدد الأقطار الفوقية أو السفلية
!>          للمصفوفة النطاقية المثلثية A. KD >= 0.)
!>

NRHS

!>          NRHS is INTEGER
!>          The number of right hand sides, i.e., the number of columns
!>          of the matrix B.  NRHS >= 0. (عدد الأجزاء اليمنى، أي عدد أعمدة
!>          المصفوفة B. NRHS >= 0.)
!>

AB

!>          AB is COMPLEX*16 array, dimension (LDAB,N)
!>          The upper or lower triangular band matrix A, stored in the
!>          first kd+1 rows of AB.  The j-th column of A is stored
!>          in the j-th column of the array AB as follows:
!>          if UPLO = 'U', AB(kd+1+i-j,j) = A(i,j) for max(1,j-kd)<=i<=j;
!>          if UPLO = 'L', AB(1+i-j,j)    = A(i,j) for j<=i<=min(n,j+kd).
!>          If DIAG = 'U', the diagonal elements of A are not referenced
!>          and are assumed to be 1.
!>          (مصفوفة AB من النوع COMPLEX*16، البعد (LDAB, N)
!>          المصفوفة النطاقية المثلثية العلوية أو السفلية A، مخزنة في
!>          أول kd+1 صفوف من AB. يتم تخزين العمود j من A
!>          في العمود j من المصفوفة AB على النحو التالي:
!>          إذا كان UPLO = 'U'، فإن AB(kd+1+i-j,j) = A(i,j) لـ max(1,j-kd)<=i<=j؛
!>          إذا كان UPLO = 'L'، فإن AB(1+i-j,j) = A(i,j) لـ j<=i<=min(n,j+kd).
!>          إذا كان DIAG = 'U'، لا يتم الرجوع إلى العناصر القطرية لـ A
!>          ويفترض أنها 1.)
!>

LDAB

!>          LDAB is INTEGER
!>          The leading dimension of the array AB.  LDAB >= KD+1. (البعد الرئيسي للمصفوفة AB. LDAB >= KD+1.)
!>

B

!>          B is COMPLEX*16 array, dimension (LDB,NRHS)
!>          On entry, the right hand side matrix B. (عند الدخول، مصفوفة الجزء الأيمن B.)
!>          On exit, if INFO = 0, the solution matrix X. (عند الخروج، إذا كان INFO = 0، مصفوفة الحل X.)
!>

LDB

!>          LDB is INTEGER
!>          The leading dimension of the array B.  LDB >= max(1,N). (البعد الرئيسي للمصفوفة B. LDB >= max(1,N).)
!>

INFO

!>          INFO is INTEGER
!>          = 0:  successful exit (خروج ناجح)
!>          < 0:  if INFO = -i, the i-th argument had an illegal value (إذا كان INFO = -i، فإن المعامل i كان له قيمة غير قانونية)
!>          > 0:  if INFO = i, the i-th diagonal element of A is exactly zero,
!>                indicating that the matrix is singular and the
!>                solutions X have not been computed.
!>                (إذا كان INFO = i، فإن العنصر القطري i للمصفوفة A يساوي صفرًا بالضبط،
!>                مما يشير إلى أن المصفوفة مفردة ولم يتم حساب الحلول X.)
!>
