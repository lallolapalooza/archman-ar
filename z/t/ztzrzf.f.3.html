توثيق الوظيفة/الروتين الفرعي

subroutine ztzrzf (integer m, integer n, complex*16, dimension( lda, * ) a, integer lda, complex*16, dimension( * ) tau, complex*16, dimension( * ) work, integer lwork, integer info)

ZTZRZF

الغرض:

!>
!> ZTZRZF يقلل المصفوفة العلوية شبه المنحرفة A ذات الأبعاد M × N ( M<=N )
!> إلى شكل مثلثي علوي بواسطة تحويلات وحدوية.
!>
!> يتم تحليل المصفوفة شبه المنحرفة العلوية A على النحو التالي:
!>
!>    A = ( R  0 ) * Z,
!>
!> حيث Z مصفوفة وحدوية N×N و R مصفوفة مثلثية علوية M×M.
!>

المعلمات

M 
!>          M هو عدد صحيح (INTEGER)
!>          عدد الصفوف في المصفوفة A.  M >= 0.
!>

N

!>          N هو عدد صحيح (INTEGER)
!>          عدد الأعمدة في المصفوفة A.  N >= M.
!>

A

!>          A هي مصفوفة معقدة *16 (COMPLEX*16)، ذات أبعاد (LDA,N)
!>          عند الإدخال، يجب أن يحتوي الجزء العلوي شبه المنحرف من A
!>          بأبعاد M×N على المصفوفة المراد تحليلها.
!>          عند الخروج، يحتوي الجزء المثلثي العلوي M×M من A
!>          على المصفوفة المثلثية العلوية R، والعناصر من M+1 إلى
!>          N في الصفوف الأولى من A، مع المصفوفة TAU، تمثل
!>          المصفوفة الوحدوية Z كناتج لـ M عاكسات أولية.
!>

LDA

!>          LDA هو عدد صحيح (INTEGER)
!>          البعد الرائد للمصفوفة A.  LDA >= max(1,M).
!>

TAU

!>          TAU هي مصفوفة معقدة *16 (COMPLEX*16)، ذات أبعاد (M)
!>          العوامل العددية للعواكس الأولية.
!>

WORK

!>          WORK هي مصفوفة معقدة *16 (COMPLEX*16)، ذات أبعاد (MAX(1,LWORK))
!>          عند الخروج، إذا كان INFO = 0، فإن WORK(1) يعيد LWORK المثلى.
!>

LWORK

!>          LWORK هو عدد صحيح (INTEGER)
!>          بعد المصفوفة WORK.  LWORK >= max(1,M).
!>          للأداء الأمثل LWORK >= M*NB، حيث NB هو
!>          حجم الكتلة الأمثل.
!>
!>          إذا كان LWORK = -1، فسيتم افتراض استعلام عن مساحة العمل؛
!>          يقوم الروتين فقط بحساب الحجم الأمثل للمصفوفة WORK، ويعيد
!>          هذه القيمة كالإدخال الأول في المصفوفة WORK، ولا تصدر رسالة خطأ
!>          متعلقة بـ LWORK بواسطة XERBLA.
!>

INFO

!>          INFO هو عدد صحيح (INTEGER)
!>          = 0:  الخروج بنجاح
!>          < 0:  إذا كان INFO = -i، فإن الوسيط i كان له قيمة غير قانونية
!>

تفاصيل إضافية:

!>
!> يمكن حساب المصفوفة N×N Z بواسطة
!>
!>    Z =  Z(1)*Z(2)* ... *Z(M)
!>
!> حيث كل Z(k) بحجم N×N تعطى بواسطة
!>
!>    Z(k) = I - tau(k)*v(k)*v(k)**H
!>
!> مع v(k) هو المتجه الصف k للمصفوفة M×N
!>
!>    V = ( I   A(:,M+1:N) )
!>
!> I هي المصفوفة الهوية M×M، A(:,M+1:N)
!> هي المخرجات المخزنة في A عند الخروج من ZTZRZF،
!> و tau(k) هو العنصر k من المصفوفة TAU.
!>
!>
