توثيق الوظيفة/الروتين الفرعي

subroutine ztrt02 (character uplo, character trans, character diag, integer n, integer nrhs, complex*16, dimension( lda, * ) a, integer lda, complex*16, dimension( ldx, * ) x, integer ldx, complex*16, dimension( ldb, * ) b, integer ldb, complex*16, dimension( * ) work, double precision, dimension( * ) rwork, double precision resid)

ZTRT02

الغرض:

!>
!> ZTRT02 يحسب الباقي للحل المحسوب لنظام المعادلات الخطية المثلثية op(A)*X = B، حيث A مصفوفة مثلثية. نسبة الاختبار هي القيمة القصوى عبر
!>    norm(b - op(A)*x) / ( ||op(A)||_1 * norm(x) * EPS )،
!> حيث op(A) = A, A**T, أو A**H، b هو عمود B، x هو متجه الحل، و EPS هو معامل الآلة.
!>

المعلمات

UPLO 
!>          UPLO هو CHARACTER*1
!>          يحدد ما إذا كانت المصفوفة A علوية أو سفلية مثلثية.
!>          = 'U':  مثلثية علوية
!>          = 'L':  مثلثية سفلية
!>

TRANS

!>          TRANS هو CHARACTER*1
!>          يحدد العملية المطبقة على A.
!>          = 'N':  A    * X = B  (بدون تبديل)
!>          = 'T':  A**T * X = B  (تبديل)
!>          = 'C':  A**H * X = B  (تبديل مترافق)
!>

DIAG

!>          DIAG هو CHARACTER*1
!>          يحدد ما إذا كانت المصفوفة A مثلثية وحدة أم لا.
!>          = 'N':  مثلثية غير وحدة
!>          = 'U':  مثلثية وحدة
!>

N

!>          N هو INTEGER
!>          رتبة المصفوفة A.  N >= 0.
!>

NRHS

!>          NRHS هو INTEGER
!>          عدد الأطراف اليمنى، أي عدد أعمدة
!>          المصفوفتين X و B.  NRHS >= 0.
!>

A

!>          A هو مصفوفة COMPLEX*16، البعد (LDA,N)
!>          المصفوفة المثلثية A.  إذا كان UPLO = 'U'، فإن الجزء العلوي المثلثي
!>          المربع n*n الرائد في المصفوفة A يحتوي على المصفوفة المثلثية العلوية،
!>          ويتم تجاهل الجزء المثلثي السفلي الصارم لـ A.  إذا كان UPLO = 'L'، فإن الجزء
!>          المثلثي السفلي الرائد المربع n*n في المصفوفة A يحتوي على المصفوفة
!>          المثلثية السفلية، ويتم تجاهل الجزء المثلثي العلوي الصارم لـ A.
!>          إذا كان DIAG = 'U'، فسيتم أيضًا تجاهل العناصر القطرية لـ A
!>          ويفترض أنها 1.
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرائد للمصفوفة A.  LDA >= max(1,N).
!>

X

!>          X هو مصفوفة COMPLEX*16، البعد (LDX,NRHS)
!>          متجهات الحل المحسوبة لنظام المعادلات الخطية.
!>

LDX

!>          LDX هو INTEGER
!>          البعد الرائد للمصفوفة X.  LDX >= max(1,N).
!>

B

!>          B هو مصفوفة COMPLEX*16، البعد (LDB,NRHS)
!>          متجهات الطرف الأيمن لنظام المعادلات الخطية.
!>

LDB

!>          LDB هو INTEGER
!>          البعد الرائد للمصفوفة B.  LDB >= max(1,N).
!>

WORK

!>          WORK هي مصفوفة COMPLEX*16، البعد (N)
!>

RWORK

!>          RWORK هي مصفوفة DOUBLE PRECISION، البعد (N)
!>

RESID

!>          RESID هو DOUBLE PRECISION
!>          القيمة القصوى على عدد الأطراف اليمنى لـ
!>          norm(op(A)*X - B) / ( norm(op(A)) * norm(X) * EPS ).
!>
