توثيق الوظيفة/الروتين الفرعي

subroutine ztgevc (character side, character howmny, logical, dimension( * ) select, integer n, complex*16, dimension( lds, * ) s, integer lds, complex*16, dimension( ldp, * ) p, integer ldp, complex*16, dimension( ldvl, * ) vl, integer ldvl, complex*16, dimension( ldvr, * ) vr, integer ldvr, integer mm, integer m, complex*16, dimension( * ) work, double precision, dimension( * ) rwork, integer info)

ZTGEVC

Purpose:

!>
!> ZTGEVC يحسب بعض أو كل المتجهات الذاتية اليمنى واليسرى لزوج من المصفوفات المركبة (S,P)، حيث S و P مثلثيتان علويتان.
!> يتم إنتاج أزواج المصفوفات من هذا النوع عن طريق التفكيك الشورطي المعمم لزوج المصفوفات المركبة (A,B):
!>
!>    A = Q*S*Z**H,  B = Q*P*Z**H
!>
!> كما تم حسابه بواسطة ZGGHRD + ZHGEQZ.
!>
!> المتجه الذاتي الأيمن x والمتجه الذاتي الأيسر y لـ (S,P) المقابل لقيمة ذاتية w معرفان بواسطة:
!>
!>    S*x = w*P*x,  (y**H)*S = w*(y**H)*P,
!>
!> حيث يمثل y**H منقولات y المترافقة.
!> القيم الذاتية ليست مدخلات لهذه الروتين، بل يتم حسابها مباشرة من العناصر القطرية لـ S و P.
!>
!> تعيد هذه الروتين المصفوفات X و/أو Y للمتجهات الذاتية اليمنى واليسرى لـ (S,P)، أو المنتجات Z*X و/أو Q*Y،
!> حيث Z و Q هي مصفوفات الإدخال.
!> إذا كانت Q و Z هي عوامل الوحدة من التفكيك الشورطي المعمم لزوج مصفوفات (A,B)، فإن Z*X و Q*Y
!> هي مصفوفات المتجهات الذاتية اليمنى واليسرى لـ (A,B).
!>

Parameters

SIDE 
!>          SIDE هو CHARACTER*1
!>          = 'R': حساب المتجهات الذاتية اليمنى فقط؛
!>          = 'L': حساب المتجهات الذاتية اليسرى فقط؛
!>          = 'B': حساب كل المتجهات الذاتية اليمنى واليسرى.
!>

HOWMNY

!>          HOWMNY هو CHARACTER*1
!>          = 'A': حساب جميع المتجهات الذاتية اليمنى و/أو اليسرى؛
!>          = 'B': حساب جميع المتجهات الذاتية اليمنى و/أو اليسرى،
!>                 معاد تحويلها بواسطة المصفوفات في VR و/أو VL؛
!>          = 'S': حساب المتجهات الذاتية اليمنى و/أو اليسرى المحددة،
!>                 المحددة بواسطة المصفوفة المنطقية SELECT.
!>

SELECT

!>          SELECT هي مصفوفة LOGICAL، بعد (N)
!>          إذا كان HOWMNY='S'، فإن SELECT تحدد المتجهات الذاتية التي سيتم
!>          حسابها. يتم حساب المتجه الذاتي المقابل للقيمة الذاتية j إذا كان SELECT(j) = .TRUE..
!>          لا يتم الاستشهاد به إذا كان HOWMNY = 'A' أو 'B'.
!>

N

!>          N هو INTEGER
!>          ترتيب المصفوفات S و P. N >= 0.
!>

S

!>          S هي مصفوفة COMPLEX*16، بعد (LDS,N)
!>          المصفوفة المثلثية العليا S من تفكيك شورتي معمم،
!>          كما تم حسابه بواسطة ZHGEQZ.
!>

LDS

!>          LDS هو INTEGER
!>          البعد الرائد للمصفوفة S. LDS >= max(1,N).
!>

P

!>          P هي مصفوفة COMPLEX*16، بعد (LDP,N)
!>          المصفوفة المثلثية العليا P من تفكيك شورتي معمم،
!>          كما تم حسابه بواسطة ZHGEQZ. يجب أن تحتوي P على عناصر قطرية حقيقية.
!>

LDP

!>          LDP هو INTEGER
!>          البعد الرائد للمصفوفة P. LDP >= max(1,N).
!>

VL

!>          VL هي مصفوفة COMPLEX*16، بعد (LDVL,MM)
!>          عند الدخول، إذا كان SIDE = 'L' أو 'B' و HOWMNY = 'B'، يجب أن
!>          تحتوي VL على مصفوفة الوحدة Q (عادةً ما تكون مصفوفة الوحدة Q
!>          للأشعة الشورتية اليسرى التي أعادتها ZHGEQZ).
!>          عند الخروج، إذا كان SIDE = 'L' أو 'B'، تحتوي VL على:
!>          إذا كان HOWMNY = 'A'، فإن مصفوفة Y للمتجهات الذاتية اليسرى لـ (S,P)؛
!>          إذا كان HOWMNY = 'B'، فإن مصفوفة Q*Y؛
!>          إذا كان HOWMNY = 'S'، فإن المتجهات الذاتية اليسرى لـ (S,P) المحددة بواسطة
!>                      SELECT، مخزنة بشكل متتالي في أعمدة
!>                      VL، بنفس ترتيب قيمها الذاتية.
!>          لا يتم الاستشهاد به إذا كان SIDE = 'R'.
!>

LDVL

!>          LDVL هو INTEGER
!>          البعد الرائد للمصفوفة VL. LDVL >= 1، وإذا كان
!>          SIDE = 'L' أو 'l' أو 'B' أو 'b'، فإن LDVL >= N.
!>

VR

!>          VR هي مصفوفة COMPLEX*16، بعد (LDVR,MM)
!>          عند الدخول، إذا كان SIDE = 'R' أو 'B' و HOWMNY = 'B'، يجب أن
!>          تحتوي VR على مصفوفة الوحدة Z (عادةً ما تكون مصفوفة الوحدة Z
!>          للأشعة الشورتية اليمنى التي أعادتها ZHGEQZ).
!>          عند الخروج، إذا كان SIDE = 'R' أو 'B'، تحتوي VR على:
!>          إذا كان HOWMNY = 'A'، فإن مصفوفة X للمتجهات الذاتية اليمنى لـ (S,P)؛
!>          إذا كان HOWMNY = 'B'، فإن مصفوفة Z*X؛
!>          إذا كان HOWMNY = 'S'، فإن المتجهات الذاتية اليمنى لـ (S,P) المحددة بواسطة
!>                      SELECT، مخزنة بشكل متتالي في أعمدة
!>                      VR، بنفس ترتيب قيمها الذاتية.
!>          لا يتم الاستشهاد به إذا كان SIDE = 'L'.
!>

LDVR

!>          LDVR هو INTEGER
!>          البعد الرائد للمصفوفة VR. LDVR >= 1، وإذا كان
!>          SIDE = 'R' أو 'B'، فإن LDVR >= N.
!>

MM

!>          MM هو INTEGER
!>          عدد الأعمدة في المصفوفات VL و/أو VR. MM >= M.
!>

M

!>          M هو INTEGER
!>          عدد الأعمدة في المصفوفات VL و/أو VR المستخدمة فعليًا
!>          لتخزين المتجهات الذاتية. إذا كان HOWMNY = 'A' أو 'B'، يتم تعيين M
!>          إلى N. كل متجه ذاتي محدد يشغل عمودًا واحدًا.
!>

WORK

!>          WORK هي مصفوفة COMPLEX*16، بعد (2*N)
!>

RWORK

!>          RWORK هي مصفوفة DOUBLE PRECISION، بعد (2*N)
!>

INFO

!>          INFO هو INTEGER
!>          = 0:  انتهاء ناجح.
!>          < 0:  إذا كان INFO = -i، فإن الوسيط i كان لديه قيمة غير قانونية.
!>
