subroutine ztpv (character uplo, character trans, character diag, integer n, complex*16, dimension(*) ap, complex*16, dimension(*) x, integer incx)

ZTPMV

الغرض:

!>
!> ZTPMV  يجري إحدى عمليات ضرب المصفوفة بالمتجه:
!>
!>    x := A*x,   أو   x := A**T*x,   أو   x := A**H*x,
!>
!> حيث x هو متجه من n عنصر و A هي مصفوفة مثلثة عليا أو سفلى،
!> ذات وحدة واحدة أو غير ذلك، بحجم n × n، مقدمة في شكل مضغوط.
!>

المعلمات

UPLO 
!>          UPLO هو CHARACTER*1
!>           عند الإدخال، يحدد UPLO ما إذا كانت المصفوفة مثلثة عليا أو سفلى
!>           كما يلي:
!>
!>              UPLO = 'U' أو 'u'   A هي مصفوفة مثلثة عليا.
!>
!>              UPLO = 'L' أو 'l'   A هي مصفوفة مثلثة سفلى.
!>

TRANS

!>          TRANS هو CHARACTER*1
!>           عند الإدخال، يحدد TRANS العملية التي سيتم إجراؤها كما يلي:
!>
!>              TRANS = 'N' أو 'n'   x := A*x.
!>
!>              TRANS = 'T' أو 't'   x := A**T*x.
!>
!>              TRANS = 'C' أو 'c'   x := A**H*x.
!>

DIAG

!>          DIAG هو CHARACTER*1
!>           عند الإدخال، يحدد DIAG ما إذا كانت A ذات وحدة واحدة أو لا
!>           مثلثة كما يلي:
!>
!>              DIAG = 'U' أو 'u'   A تعتبر مصفوفة ذات وحدة واحدة.
!>
!>              DIAG = 'N' أو 'n'   A لا تعتبر ذات وحدة واحدة.
!>

N

!>          N هو INTEGER
!>           عند الإدخال، يحدد N رتبة المصفوفة A.
!>           يجب أن تكون N على الأقل صفر.
!>

AP

!>          AP هو مصفوفة COMPLEX*16، بأبعاد لا تقل عن
!>           ( ( n*( n + 1 ) )/2 ).
!>           قبل الإدخال مع UPLO = 'U' أو 'u'، يجب أن تحتوي المصفوفة AP
!>           على المصفوفة المثلثة العليا مضغوطة بشكل متسلسل،
!>           عمودًا بعمود، بحيث تحتوي AP( 1 ) على a( 1, 1 )،
!>           وتحتوي AP( 2 ) و AP( 3 ) على a( 1, 2 ) و a( 2, 2 )
!>           على التوالي، وهكذا.
!>           قبل الإدخال مع UPLO = 'L' أو 'l'، يجب أن تحتوي المصفوفة AP
!>           على المصفوفة المثلثة السفلى مضغوطة بشكل متسلسل،
!>           عمودًا بعمود، بحيث تحتوي AP( 1 ) على a( 1, 1 )،
!>           وتحتوي AP( 2 ) و AP( 3 ) على a( 2, 1 ) و a( 3, 1 )
!>           على التوالي، وهكذا.
!>           لاحظ أنه عندما يكون DIAG = 'U' أو 'u'، فإن عناصر القطر
!>           لـ A لا يتم الرجوع إليها، ولكن يفترض أنها الوحدة.
!>

X

!>          X هي مصفوفة COMPLEX*16، بأبعاد لا تقل عن
!>           ( 1 + ( n - 1 )*abs( INCX ) ).
!>           قبل الإدخال، يجب أن تحتوي المصفوفة X المعدلة على متجه x
!>           المكون من n عنصر. عند الخروج، يتم الكتابة فوق X بالمتجه
!>           المحول x.
!>

INCX

!>          INCX هو INTEGER
!>           عند الإدخال، يحدد INCX الزيادة لعناصر X. يجب أن لا يكون INCX
!>           صفرًا.
!>
