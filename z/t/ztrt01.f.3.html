وثائق الوظيفة/الإجراء الفرعي

subroutine ztrt01 (character uplo, character diag, integer n, complex*16, dimension( lda, * ) a, integer lda, complex*16, dimension( ldainv, * ) ainv, integer ldainv, double precision rcond, double precision, dimension( * ) rwork, double precision resid)

ZTRT01

الغرض:

!>
!> ZTRT01 يحسب الباقي للمصفوفة المثلثية A مضروبة في معكوسها:
!>    RESID = norm( A*AINV - I ) / ( N * norm(A) * norm(AINV) * EPS ),
!> حيث EPS هو الخطأ المطبعي.
!>

المعلمات

UPLO 
!>          UPLO is CHARACTER*1
!>          يحدد ما إذا كانت المصفوفة A مثلثية عليا أم سفلى.
!>          = 'U':  مثلثية عليا
!>          = 'L':  مثلثية سفلى
!>

DIAG

!>          DIAG is CHARACTER*1
!>          يحدد ما إذا كانت المصفوفة A مثلثية الوحدة أم لا.
!>          = 'N':  مثلثية غير الوحدة
!>          = 'U':  مثلثية الوحدة
!>

N

!>          N is INTEGER
!>          مرتبة المصفوفة A.  N >= 0.
!>

A

!>          A is COMPLEX*16 array, dimension (LDA,N)
!>          المصفوفة المثلثية A.  إذا كان UPLO = 'U'، فإن الجزء العلوي
!>          المثلثي العلوي للمصفوفة A يحتوي على المصفوفة المثلثية العليا،
!>          ولا يتم الرجوع إلى الجزء السفلي المثلثي الصارم للمصفوفة A.
!>          إذا كان UPLO = 'L'، فإن الجزء السفلي المثلثي السفلي للمصفوفة A
!>          يحتوي على المصفوفة المثلثية السفلى، ولا يتم الرجوع إلى الجزء
!>          المثلثي العلوي الصارم للمصفوفة A.
!>          إذا كان DIAG = 'U'، فلا يتم الرجوع إلى عناصر القطر لـ A
!>          ويفترض أنها 1.
!>

LDA

!>          LDA is INTEGER
!>          البعد الرئيسي للمصفوفة A.  LDA >= max(1,N).
!>

AINV

!>          AINV is COMPLEX*16 array, dimension (LDAINV,N)
!>          عند الدخول، معكوس المصفوفة A (المثلثية)، بنفس تنسيق التخزين
!>          مثل A.
!>          عند الخروج، يتم تدمير محتويات AINV.
!>

LDAINV

!>          LDAINV is INTEGER
!>          البعد الرئيسي للمصفوفة AINV.  LDAINV >= max(1,N).
!>

RCOND

!>          RCOND is DOUBLE PRECISION
!>          المعكوس الشرطي للمصفوفة A، محسوبًا كـ
!>          1/(norm(A) * norm(AINV)).
!>

RWORK

!>          RWORK is DOUBLE PRECISION array, dimension (N)
!>

RESID

!>          RESID is DOUBLE PRECISION
!>          norm(A*AINV - I) / ( N * norm(A) * norm(AINV) * EPS )
!>
