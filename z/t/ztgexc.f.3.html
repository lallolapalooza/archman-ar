توثيق الوظيفة/الروتين الفرعي

subroutine ztgexc (logical wantq, logical wantz, integer n, complex*16, dimension( lda, * ) a, integer lda, complex*16, dimension( ldb, * ) b, integer ldb, complex*16, dimension( ldq, * ) q, integer ldq, complex*16, dimension( ldz, * ) z, integer ldz, integer ifst, integer ilst, integer info)

ZTGEXC

الغرض:

!>
!> ZTGEXC يعيد ترتيب تفكيك Schur المعمم لمصفوفة زوجية معقدة (A,B)،
!> باستخدام تحويل مساوٍ وحدوي (A, B) := Q * (A, B) * Z**H، بحيث
!> يتم نقل الكتلة القطرية لـ (A, B) ذات رقم الصف IFST إلى الصف ILST.
!>
!> يجب أن تكون (A, B) في شكل Schur القانوني المعمم، أي أن A و
!> B كلاهما مثلثي علوي.
!>
!> اختياريًا، يتم تحديث مصفوفتي متجهات Schur المعممة Q و Z.
!>
!>        Q(in) * A(in) * Z(in)**H = Q(out) * A(out) * Z(out)**H
!>        Q(in) * B(in) * Z(in)**H = Q(out) * B(out) * Z(out)**H
!>

المعاملات

WANTQ 
!>          WANTQ هو LOGICAL
!>          .TRUE. : تحديث مصفوفة التحويل اليسرى Q؛
!>          .FALSE.: لا تقم بتحديث Q.
!>

WANTZ

!>          WANTZ هو LOGICAL
!>          .TRUE. : تحديث مصفوفة التحويل اليمنى Z؛
!>          .FALSE.: لا تقم بتحديث Z.
!>

N

!>          N هو INTEGER
!>          رتبة المصفوفتين A و B. N >= 0.
!>

A

!>          A هي مصفوفة COMPLEX*16، البعد (LDA,N)
!>          عند الإدخال، المصفوفة المثلثية العلوية A في الزوج (A, B).
!>          عند الخروج، المصفوفة المحدثة A.
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرئيسي للمصفوفة A. LDA >= max(1,N).
!>

B

!>          B هي مصفوفة COMPLEX*16، البعد (LDB,N)
!>          عند الإدخال، المصفوفة المثلثية العلوية B في الزوج (A, B).
!>          عند الخروج، المصفوفة المحدثة B.
!>

LDB

!>          LDB هو INTEGER
!>          البعد الرئيسي للمصفوفة B. LDB >= max(1,N).
!>

Q

!>          Q هي مصفوفة COMPLEX*16، البعد (LDQ,N)
!>          عند الإدخال، إذا كان WANTQ = .TRUE.، المصفوفة الوحدوية Q.
!>          عند الخروج، المصفوفة المحدثة Q.
!>          إذا كان WANTQ = .FALSE.، لا يتم الإشارة إلى Q.
!>

LDQ

!>          LDQ هو INTEGER
!>          البعد الرئيسي للمصفوفة Q. LDQ >= 1؛
!>          إذا كان WANTQ = .TRUE.، LDQ >= N.
!>

Z

!>          Z هي مصفوفة COMPLEX*16، البعد (LDZ,N)
!>          عند الإدخال، إذا كان WANTZ = .TRUE.، المصفوفة الوحدوية Z.
!>          عند الخروج، المصفوفة المحدثة Z.
!>          إذا كان WANTZ = .FALSE.، لا يتم الإشارة إلى Z.
!>

LDZ

!>          LDZ هو INTEGER
!>          البعد الرئيسي للمصفوفة Z. LDZ >= 1؛
!>          إذا كان WANTZ = .TRUE.، LDZ >= N.
!>

IFST

!>          IFST هو INTEGER
!>

ILST

!>          ILST هو INTEGER
!>          تحديد إعادة ترتيب الكتل القطرية لـ (A, B).
!>          يتم نقل الكتلة ذات رقم الصف IFST إلى الصف ILST، عن طريق
!>          تسلسل من التبديلات بين الكتل المتجاورة.
!>

INFO

!>          INFO هو INTEGER
!>           =0:  الخروج بنجاح.
!>           <0:  إذا كان INFO = -i، كان المعامل i له قيمة غير قانونية.
!>           =1:  ستكون الزوجة المصفوفية المحولة (A, B) بعيدة جدًا
!>                عن شكل Schur المعمم؛ المشكلة غير مستقرة.
!>                قد تكون (A, B) قد أعيد ترتيبها جزئيًا،
!>                و ILST تشير إلى الصف الأول من الموضع الحالي
!>                للكتلة التي يتم نقلها.
!>
