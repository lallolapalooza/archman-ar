توثيق الوظيفة/الروتين الفرعي

subroutine ztrsyl (character trana, character tranb, integer isgn, integer m, integer n, complex*16, dimension( lda, * ) a, integer lda, complex*16, dimension( ldb, * ) b, integer ldb, complex*16, dimension( ldc, * ) c, integer ldc, double precision scale, integer info)

ZTRSYL

الغرض:

!>
!> ZTRSYL يحل معادلة المصفوفة السيلفسترية المعقدة:
!>
!>    op(A)*X + X*op(B) = scale*C أو
!>    op(A)*X - X*op(B) = scale*C,
!>
!> حيث op(A) = A أو A**H، و A و B كلاهما مثلثيتان علويتان. A بحجم
!> M-by-M و B بحجم N-by-N؛ الطرف الأيمن C والحل X هما بحجم M-by-N؛
!> و scale هو عامل قياس للخرج، يتم تعيينه <= 1 لتجنب تجاوز السعة في X.
!>

المعاملات

TRANA 
!>          TRANA هو CHARACTER*1
!>          يحدد خيار op(A):
!>          = 'N': op(A) = A    (بدون تبديل)
!>          = 'C': op(A) = A**H (تبديل مترافق)
!>

TRANB

!>          TRANB هو CHARACTER*1
!>          يحدد خيار op(B):
!>          = 'N': op(B) = B    (بدون تبديل)
!>          = 'C': op(B) = B**H (تبديل مترافق)
!>

ISGN

!>          ISGN هو INTEGER
!>          يحدد الإشارة في المعادلة:
!>          = +1: حل op(A)*X + X*op(B) = scale*C
!>          = -1: حل op(A)*X - X*op(B) = scale*C
!>

M

!>          M هو INTEGER
!>          رتبة المصفوفة A، وعدد الصفوف في المصفوفتين X و C. M >= 0.
!>

N

!>          N هو INTEGER
!>          رتبة المصفوفة B، وعدد الأعمدة في المصفوفتين X و C. N >= 0.
!>

A

!>          A هي مصفوفة COMPLEX*16، حجمها (LDA,M)
!>          المصفوفة المثلثية العلوية A.
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرئيسي للمصفوفة A. LDA >= max(1,M).
!>

B

!>          B هي مصفوفة COMPLEX*16، حجمها (LDB,N)
!>          المصفوفة المثلثية العلوية B.
!>

LDB

!>          LDB هو INTEGER
!>          البعد الرئيسي للمصفوفة B. LDB >= max(1,N).
!>

C

!>          C هي مصفوفة COMPLEX*16، حجمها (LDC,N)
!>          عند الإدخال، المصفوفة C الطرف الأيمن بحجم M-by-N.
!>          عند الخرج، يتم استبدال C بالمصفوفة الحل X.
!>

LDC

!>          LDC هو INTEGER
!>          البعد الرئيسي للمصفوفة C. LDC >= max(1,M)
!>

SCALE

!>          SCALE هو DOUBLE PRECISION
!>          عامل القياس، scale، يتم تعيينه <= 1 لتجنب تجاوز السعة في X.
!>

INFO

!>          INFO هو INTEGER
!>          = 0: انتهاء ناجح
!>          < 0: إذا كان INFO = -i، كان المعامل i غير صالح
!>          = 1: A و B لهما قيم ذاتية مشتركة أو قريبة جدًا؛ تم استخدام قيم
!>               مضطربة لحل المعادلة (لكن المصفوفتين A و B لم تتغير).
!>
