subroutine ztprfs (character uplo, character trans, character diag, integer n, integer nrhs, complex*16, dimension( * ) ap, complex*16, dimension( ldb, * ) b, integer ldb, complex*16, dimension( ldx, * ) x, integer ldx, double precision, dimension( * ) ferr, double precision, dimension( * ) berr, complex*16, dimension( * ) work, double precision, dimension( * ) rwork, integer info)

ZTPRFS

الغرض:

!>
!> ZTPRFS يوفر حدود الخطأ وتقديرات الخطأ الرجعي لحل نظام
!> المعادلات الخطية مع مصفوفة معاملات مثلثة محزمة.
!>
!> يجب حساب مصفوفة الحل X بواسطة ZTPTRS أو بأي وسيلة أخرى
!> قبل الدخول إلى هذا الروتين. ZTPRFS لا يقوم بالتحسين التكراري
!> لأنه لا يمكن تحسين الخطأ الرجعي.
!>

المعلمات

UPLO 
!>          UPLO is CHARACTER*1
!>          = 'U':  A is upper triangular; (A مثلثة علوية)
!>          = 'L':  A is lower triangular. (A مثلثة سفلية)
!>

TRANS

!>          TRANS is CHARACTER*1
!>          Specifies the form of the system of equations: (يحدد شكل نظام المعادلات)
!>          = 'N':  A * X = B     (No transpose) (بدون تبديل)
!>          = 'T':  A**T * X = B  (Transpose) (تبديل)
!>          = 'C':  A**H * X = B  (Conjugate transpose) (تبديل مترافق)
!>

DIAG

!>          DIAG is CHARACTER*1
!>          = 'N':  A is non-unit triangular; (A مثلثة غير وحدة)
!>          = 'U':  A is unit triangular. (A مثلثة وحدة)
!>

N

!>          N is INTEGER
!>          The order of the matrix A.  N >= 0. (رتبة المصفوفة A. N >= 0)
!>

NRHS

!>          NRHS is INTEGER
!>          The number of right hand sides, i.e., the number of columns
!>          of the matrices B and X.  NRHS >= 0. (عدد الجوانب اليمنى، أي عدد أعمدة
!>          المصفوفتين B و X. NRHS >= 0)
!>

AP

!>          AP is COMPLEX*16 array, dimension (N*(N+1)/2)
!>          The upper or lower triangular matrix A, packed columnwise in
!>          a linear array.  The j-th column of A is stored in the array
!>          AP as follows: (المصفوفة A المثلثية العلوية أو السفلية، محزمة عمودياً في
!>          مصفوفة خطية. يتم تخزين العمود j من A في المصفوفة
!>          AP على النحو التالي:)
!>          if UPLO = 'U', AP(i + (j-1)*j/2) = A(i,j) for 1<=i<=j; (إذا كان UPLO = 'U')
!>          if UPLO = 'L', AP(i + (j-1)*(2n-j)/2) = A(i,j) for j<=i<=n. (إذا كان UPLO = 'L')
!>          If DIAG = 'U', the diagonal elements of A are not referenced
!>          and are assumed to be 1. (إذا كان DIAG = 'U'، فإن العناصر القطرية لـ A لا تتم الإشارة إليها
!>          ويفترض أنها 1.)
!>

B

!>          B is COMPLEX*16 array, dimension (LDB,NRHS)
!>          The right hand side matrix B. (المصفوفة B للجوانب اليمنى.)
!>

LDB

!>          LDB is INTEGER
!>          The leading dimension of the array B.  LDB >= max(1,N). (البعد الرئيسي للمصفوفة B. LDB >= max(1,N))
!>

X

!>          X is COMPLEX*16 array, dimension (LDX,NRHS)
!>          The solution matrix X. (مصفوفة الحل X.)
!>

LDX

!>          LDX is INTEGER
!>          The leading dimension of the array X.  LDX >= max(1,N). (البعد الرئيسي للمصفوفة X. LDX >= max(1,N))
!>

FERR

!>          FERR is DOUBLE PRECISION array, dimension (NRHS)
!>          The estimated forward error bound for each solution vector
!>          X(j) (the j-th column of the solution matrix X).
!>          If XTRUE is the true solution corresponding to X(j), FERR(j)
!>          is an estimated upper bound for the magnitude of the largest
!>          element in (X(j) - XTRUE) divided by the magnitude of the
!!>          largest element in X(j).  The estimate is as reliable as
!>          the estimate for RCOND, and is almost always a slight
!>          overestimate of the true error. (حد الخطأ الأمامي المقدر لكل متجه حل
!>          X(j) (العمود j من مصفوفة الحل X).
!>          إذا كان XTRUE هو الحل الحقيقي المقابل لـ X(j)، فإن FERR(j)
!>          هو حد أعلى مقدر لقيمة أكبر عنصر في (X(j) - XTRUE) مقسومًا على قيمة
!>          أكبر عنصر في X(j). التقدير موثوق به مثل
!>          تقدير RCOND، وهو دائمًا تقريبًا تقدير مبالغ فيه للخطأ الحقيقي.)
!>

BERR

!>          BERR is DOUBLE PRECISION array, dimension (NRHS)
!>          The componentwise relative backward error of each solution
!>          vector X(j) (i.e., the smallest relative change in
!>          any element of A or B that makes X(j) an exact solution). (خطأ الرجوع النسبي على مستوى المكون لكل متجه حل
!>          X(j) (أي، أصغر تغيير نسبي في
!>          أي عنصر من A أو B يجعل X(j) حلاً دقيقًا).)
!>

WORK

!>          WORK is COMPLEX*16 array, dimension (2*N)
!>

RWORK

!>          RWORK is DOUBLE PRECISION array, dimension (N)
!>

INFO

!>          INFO is INTEGER
!>          = 0:  successful exit (خروج ناجح)
!>          < 0:  if INFO = -i, the i-th argument had an illegal value (إذا كان INFO = -i، فإن الوسيط i كان له قيمة غير قانونية)
!>
