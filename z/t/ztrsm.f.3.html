subroutine ztrsm (character side, character uplo, character transa, character diag, integer m, integer n, complex*16 alpha, complex*16, dimension(lda,*) a, integer lda, complex*16, dimension(ldb,*) b, integer ldb)

ZTRSM

الغرض:

!>
!> ZTRSM  يحل إحدى معادلات المصفوفات التالية:
!>
!>    op( A )*X = alpha*B,   أو   X*op( A ) = alpha*B,
!>
!> حيث alpha عدد قياسي، X و B مصفوفتان من البعد m × n، A مصفوفة مثلثية موحدة، أو غير موحدة، علوية أو سفلية، و op( A ) هي إحدى الصيغ التالية:
!>
!>    op( A ) = A   أو   op( A ) = A**T   أو   op( A ) = A**H.
!>
!> يتم استبدال المصفوفة X فوق B.
!>

المعاملات

SIDE 
!>          SIDE هو CHARACTER*1
!>           عند الإدخال، تحدد SIDE ما إذا كانت op( A ) تظهر على اليسار
!>           أو اليمين من X كما يلي:
!>
!>              SIDE = 'L' أو 'l'   op( A )*X = alpha*B.
!>
!>              SIDE = 'R' أو 'r'   X*op( A ) = alpha*B.
!>

UPLO

!>          UPLO هو CHARACTER*1
!>           عند الإدخال، تحدد UPLO ما إذا كانت المصفوفة A علوية أو
!>           سفلية كما يلي:
!>
!>              UPLO = 'U' أو 'u'   A مصفوفة مثلثية علوية.
!>
!>              UPLO = 'L' أو 'l'   A مصفوفة مثلثية سفلية.
!>

TRANSA

!>          TRANSA هو CHARACTER*1
!>           عند الإدخال، تحدد TRANSA شكل op( A ) المستخدم في
!>           ضرب المصفوفات كما يلي:
!>
!>              TRANSA = 'N' أو 'n'   op( A ) = A.
!>
!>              TRANSA = 'T' أو 't'   op( A ) = A**T.
!>
!>              TRANSA = 'C' أو 'c'   op( A ) = A**H.
!>

DIAG

!>          DIAG هو CHARACTER*1
!>           عند الإدخال، تحدد DIAG ما إذا كانت A موحدة أم لا
!>           مثلثية كما يلي:
!>
!>              DIAG = 'U' أو 'u'   يُفترض أن A مصفوفة مثلثية موحدة.
!>
!>              DIAG = 'N' أو 'n'   لا يُفترض أن A موحدة
!>                                  مثلثية.
!>

M

!>          M هو INTEGER
!>           عند الإدخال، يحدد M عدد صفوف B. يجب أن يكون M
!>           على الأقل صفر.
!>

N

!>          N هو INTEGER
!>           عند الإدخال، يحدد N عدد أعمدة B. يجب أن يكون
!>           على الأقل صفر.
!>

ALPHA

!>          ALPHA هو COMPLEX*16
!>           عند الإدخال، يحدد ALPHA العدد القياسي alpha. عندما يكون alpha
!>           صفرًا، لا يتم الرجوع إلى A ولا يلزم تعيين B قبل
!>           الدخول.
!>

A

!>          A هي مصفوفة COMPLEX*16، بأبعاد ( LDA, k )،
!>           حيث k هو m عندما SIDE = 'L' أو 'l'
!>             و k هو n عندما SIDE = 'R' أو 'r'.
!>           قبل الدخول عند UPLO = 'U' أو 'u'، الجزء العلوي
!>           المثلثي العلوي من البعد k × k في المصفوفة A يجب أن يحتوي على المصفوفة
!>           المثلثية العلوية والجزء السفلي المثلثي السفلي
!>           من A لا يتم الرجوع إليه.
!>           قبل الدخول عند UPLO = 'L' أو 'l'، الجزء السفلي
!>           المثلثي السفلي من البعد k × k في المصفوفة A يجب أن يحتوي على المصفوفة
!>           المثلثية السفلية والجزء العلوي المثلثي العلوي
!>           من A لا يتم الرجوع إليه.
!>           لاحظ أنه عند DIAG = 'U' أو 'u'، عناصر القطر
!>           من A لا يتم الرجوع إليها أيضًا، ولكنها تُفترض أن تكون وحدة.
!>

LDA

!>          LDA هو INTEGER
!>           عند الإدخال، يحدد LDA البعد الأول لـ A كما تم تعريفه
!>           في البرنامج (الفرعي) المستدعي. عندما SIDE = 'L' أو 'l'،
!>           يجب أن يكون LDA على الأقل max( 1, m )، عندما SIDE = 'R' أو 'r'،
!>           يجب أن يكون LDA على الأقل max( 1, n ).
!>

B

!>          B هي مصفوفة COMPLEX*16، بأبعاد ( LDB, N )
!>           قبل الإدخال، يجب أن يحتوي الجزء الأمامي من البعد m × n في المصفوفة B على
!>           مصفوفة الجانب الأيمن B، وعند الخروج يتم استبدالها بالمصفوفة
!>           الحل X.
!>

LDB

!>          LDB هو INTEGER
!>           عند الإدخال، يحدد LDB البعد الأول لـ B كما تم تعريفه
!>           في البرنامج (الفرعي) المستدعي. يجب أن يكون LDB على الأقل
!>           max( 1, m ).
!>
