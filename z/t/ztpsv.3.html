subroutine ztpsv (character uplo, character trans, character diag, integer n, complex*16, dimension(*) ap, complex*16, dimension(*) x, integer incx)

ZTPSV

الغرض:

!>
!> ZTPSV  يحل أحد أنظمة المعادلات:
!>
!>    A*x = b,   أو   A**T*x = b,   أو   A**H*x = b,
!>
!> حيث b و x متجهان يتكون كل منهما من n عنصر، و A مصفوفة مثلثة عليا أو سفلى،
!> وحدوية أو غير وحدوية، بحجم n × n، مقدمة بالشكل المضغوط.
!>
!> لا تتضمن هذه الروتينات اختبارًا للعزل أو القرب من العزل. يجب إجراء مثل هذه
!> الاختبارات قبل استدعاء هذا الروتين.
!>

المعلمات

UPLO 
!>          UPLO هو CHARACTER*1
!>           عند الدخول، تحدد UPLO ما إذا كانت المصفوفة مثلثة عليا أو سفلى
!>           بالشكل التالي:
!>
!>              UPLO = 'U' أو 'u'   A مصفوفة مثلثة عليا.
!>
!>              UPLO = 'L' أو 'l'   A مصفوفة مثلثة سفلى.
!>

TRANS

!>          TRANS هو CHARACTER*1
!>           عند الدخول، تحدد TRANS المعادلات المراد حلها بالشكل التالي:
!>
!>              TRANS = 'N' أو 'n'   A*x = b.
!>
!>              TRANS = 'T' أو 't'   A**T*x = b.
!>
!>              TRANS = 'C' أو 'c'   A**H*x = b.
!>

DIAG

!>          DIAG هو CHARACTER*1
!>           عند الدخول، يحدد DIAG ما إذا كانت A وحدوية أو غير وحدوية
!>           الشكل التالي:
!>
!>              DIAG = 'U' أو 'u'   A تفترض أنها مثلثة وحدوية.
!>
!>              DIAG = 'N' أو 'n'   A لا تفترض أنها مثلثة وحدوية.
!>

N

!>          N هو INTEGER
!>           عند الدخول، يحدد N رتبة المصفوفة A.
!>           يجب أن يكون N على الأقل صفرًا.
!>

AP

!>          AP هو مصفوفة COMPLEX*16، ذات حجم لا يقل عن
!>           ( ( n*( n + 1 ) )/2 ).
!>           قبل الدخول مع UPLO = 'U' أو 'u'، يجب أن تحتوي المصفوفة AP
!>           على المصفوفة المثلثة العليا مضغوطة بالتتابع،
!>           عمودًا بعمود، بحيث تحتوي AP( 1 ) على a( 1, 1 )،
!>           وتحتوي AP( 2 ) و AP( 3 ) على a( 1, 2 ) و a( 2, 2 )
!>           على التوالي، وهكذا.
!>           قبل الدخول مع UPLO = 'L' أو 'l'، يجب أن تحتوي المصفوفة AP
!>           على المصفوفة المثلثة السفلى مضغوطة بالتتابع،
!>           عمودًا بعمود، بحيث تحتوي AP( 1 ) على a( 1, 1 )،
!>           وتحتوي AP( 2 ) و AP( 3 ) على a( 2, 1 ) و a( 3, 1 )
!>           على التوالي، وهكذا.
!>           لاحظ أنه عندما يكون DIAG = 'U' أو 'u'، لا يتم الإشارة إلى العناصر
!>           القطرية لـ A، ولكن يفترض أنها تساوي الواحد.
!>

X

!>          X هي مصفوفة COMPLEX*16، ذات حجم لا يقل عن
!>           ( 1 + ( n - 1 )*abs( INCX ) ).
!>           قبل الدخول، يجب أن تحتوي المصفوفة X المعززة بالزيادة على المتجه
!>           b ذي الـ n عنصر الموجود في الجانب الأيمن. عند الخروج، يتم استبدال X
!>           بالحل x.
!>

INCX

!>          INCX هو INTEGER
!>           عند الدخول، يحدد INCX الزيادة لعناصر X. يجب ألا يكون INCX صفرًا.
!>
