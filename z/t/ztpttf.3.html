subroutine ztpttf (transr, uplo, n, ap, arf, info)
ZTPTTF ينسخ مصفوفة مثلثية من الصيغة المكتظة القياسية (TP) إلى صيغة التعبئة الكاملة المستطيلة (TF).

توثيق الوظيفة/الإجراء الفرعي

subroutine ztpttf (character transr, character uplo, integer n, complex*16, dimension( 0: * ) ap, complex*16, dimension( 0: * ) arf, integer info)

ZTPTTF ينسخ مصفوفة مثلثية من الصيغة المكتظة القياسية (TP) إلى صيغة التعبئة الكاملة المستطيلة (TF).

الغرض:

!>
!> ZTPTTF ينسخ مصفوفة مثلثية A من الصيغة المكتظة القياسية (TP)
!> إلى صيغة التعبئة الكاملة المستطيلة (TF).
!>

المعلمات

TRANSR 
!>          TRANSR هو CHARACTER*1
!>          = 'N':  مطلوب ARF بالتنسيق العادي؛
!>          = 'C':  مطلوب ARF بالتنسيق المعكوس المزدوج.
!>

UPLO

!>          UPLO هو CHARACTER*1
!>          = 'U':  A مصفوفة مثلثية علوية؛
!>          = 'L':  A مصفوفة مثلثية سفلية.
!>

N

!>          N هو INTEGER
!>          رتبة المصفوفة A.  N >= 0.
!>

AP

!>          AP هي مصفوفة COMPLEX*16، البعد ( N*(N+1)/2 )،
!>          عند الدخول، المصفوفة المثلثية العلوية أو السفلية A، مكتظة
!>          عمودياً في مصفوفة خطية. يتم تخزين العمود j من A
!>          في المصفوفة AP كما يلي:
!>          إذا كان UPLO = 'U'، فإن AP(i + (j-1)*j/2) = A(i,j) لـ 1<=i<=j؛
!>          إذا كان UPLO = 'L'، فإن AP(i + (j-1)*(2n-j)/2) = A(i,j) لـ j<=i<=n.
!>

ARF

!>          ARF هي مصفوفة COMPLEX*16، البعد ( N*(N+1)/2 )،
!>          عند الخروج، المصفوفة المثلثية العلوية أو السفلية A مخزنة
!>          بصيغة RFP. لمناقشة أعمق انظر ملاحظات أدناه.
!>

INFO

!>          INFO هو INTEGER
!>          = 0:  الانتهاء بنجاح
!>          < 0:  إذا كان INFO = -i، فإن الوسيط i كان له قيمة غير قانونية
!>

!>

!>  بالنسبة لـ UPLO = 'U'، يتكون شبه المنحرف العلوي A(0:5,0:2) من الأعمدة الثلاثة الأخيرة
!>  من AP العلوية. يتكون المثلث السفلي A(4:6,0:2) من
!>  المعكوس المزدوج للأعمدة الثلاثة الأولى من AP العلوية.
!>  بالنسبة لـ UPLO = 'L'، يتكون شبه المنحرف السفلي A(1:6,0:2) من الأعمدة الثلاثة الأولى
!>  من AP السفلية. يتكون المثلث العلوي A(0:2,0:2) من
!>  المعكوس المزدوج للأعمدة الثلاثة الأخيرة من AP السفلية.
!>  للإشارة إلى المزدوج نضع -- فوق العنصر. هذا يغطي
!>  الحالة N زوجية و TRANSR = 'N'.

!>         RFP A                   RFP A
