subroutine ztbsv (character uplo, character trans, character diag, integer n, integer k, complex*16, dimension(lda,*) a, integer lda, complex*16, dimension(*) x, integer incx)

ZTBSV

الغرض:

!>
!> ZTBSV  يحل أحد أنظمة المعادلات
!>
!>    A*x = b,   أو   A**T*x = b,   أو   A**H*x = b,
!>
!> حيث b و x هما متجهان من n عنصر و A هي مصفوفة نطاقية مثلثية علوية أو سفلية
!> بحجم n × n، ذات وحدة، أو غير وحدة، مع ( k + 1 ) قطرات.
!>
!> لا يتضمن هذا الإجراء اختبارًا للتعيين أو التقارب الوثيق. يجب
!> إجراء مثل هذه الاختبارات قبل استدعاء هذا الإجراء.
!>

المعلمات

UPLO 
!>          UPLO هو CHARACTER*1
!>           عند الإدخال، يحدد UPLO ما إذا كانت المصفوفة مثلثية علوية أو سفلية
!>           على النحو التالي:
!>
!>              UPLO = 'U' أو 'u'   A هي مصفوفة مثلثية علوية.
!>
!>              UPLO = 'L' أو 'l'   A هي مصفوفة مثلثية سفلية.
!>

TRANS

!>          TRANS هو CHARACTER*1
!>           عند الإدخال، يحدد TRANS المعادلات التي سيتم حلها على النحو التالي:
!>
!>              TRANS = 'N' أو 'n'   A*x = b.
!>
!>              TRANS = 'T' أو 't'   A**T*x = b.
!>
!>              TRANS = 'C' أو 'c'   A**H*x = b.
!>

DIAG

!>          DIAG هو CHARACTER*1
!>           عند الإدخال، يحدد DIAG ما إذا كانت A ذات وحدة أم لا
!>           مثلثية على النحو التالي:
!>
!>              DIAG = 'U' أو 'u'   A يفترض أنها ذات وحدة مثلثية.
!>
!>              DIAG = 'N' أو 'n'   A لا يفترض أنها ذات وحدة
!>                                  مثلثية.
!>

N

!>          N هو INTEGER
!>           عند الإدخال، يحدد N رتبة المصفوفة A.
!>           يجب أن يكون N على الأقل صفرًا.
!>

K

!>          K هو INTEGER
!>           عند الإدخال مع UPLO = 'U' أو 'u'، يحدد K عدد
!>           القطرات الفوقية للمصفوفة A.
!>           عند الإدخال مع UPLO = 'L' أو 'l'، يحدد K عدد
!>           القطرات التحتية للمصفوفة A.
!>           يجب أن يحقق K: 0 .le. K.
!>

A

!>          A هو مصفوفة COMPLEX*16، ذات بُعد ( LDA, N )
!>           قبل الإدخال مع UPLO = 'U' أو 'u'، يجب أن يحتوي الجزء الرائد
!>           ( k + 1 ) × n من المصفوفة A على الجزء النطاقي المثلثي العلوي
!>           للمصفوفة للمعاملات، المقدمة عمودًا بعمود، مع القطرة الرائدة
!>           للمصفوفة في الصف ( k + 1 ) من المصفوفة، تبدأ القطرة الفوقية الأولى
!>           من الموضع 2 في الصف k، وهكذا. لا تتم الإشارة إلى المثلث العلوي الأيسر
!>           بحجم k × k من المصفوفة A.
!>           سيقوم مقطع البرنامج التالي بنقل مصفوفة نطاقية مثلثية علوية
!>           من تخزين المصفوفة الكاملة التقليدية إلى تخزين النطاق:
!>
!>                 DO 20, J = 1, N
!>                    M = K + 1 - J
!>                    DO 10, I = MAX( 1, J - K ), J
!>                       A( M + I, J ) = matrix( I, J )
!>              10    CONTINUE
!>              20 CONTINUE
!>
!>           قبل الإدخال مع UPLO = 'L' أو 'l'، يجب أن يحتوي الجزء الرائد
!>           ( k + 1 ) × n من المصفوفة A على الجزء النطاقي المثلثي السفلي
!>           للمصفوفة للمعاملات، المقدمة عمودًا بعمود، مع القطرة الرائدة
!>           للمصفوفة في الصف 1 من المصفوفة، تبدأ القطرة التحتية الأولى
!>           من الموضع 1 في الصف 2، وهكذا. لا تتم الإشارة إلى المثلث السفلي الأيمن
!>           بحجم k × k من المصفوفة A.
!>           سيقوم مقطع البرنامج التالي بنقل مصفوفة نطاقية مثلثية سفلية
!>           من تخزين المصفوفة الكاملة التقليدية إلى تخزين النطاق:
!>
!>                 DO 20, J = 1, N
!>                    M = 1 - J
!>                    DO 10, I = J, MIN( N, J + K )
!>                       A( M + I, J ) = matrix( I, J )
!>              10    CONTINUE
!>              20 CONTINUE
!>
!>           لاحظ أنه عندما يكون DIAG = 'U' أو 'u'، لا تتم الإشارة إلى عناصر
!>           المصفوفة A المقابلة لعناصر القطرة للمصفوفة، ولكن يفترض أنها وحدات.
!>

LDA

!>          LDA هو INTEGER
!>           عند الإدخال، يحدد LDA البعد الأول لـ A كما تم تعريفه
!>           في البرنامج المتصل (الفرعي). يجب أن يكون LDA على الأقل
!>           ( k + 1 ).
!>

X

!>          X هو مصفوفة COMPLEX*16، ذات بُعد على الأقل
!>           ( 1 + ( n - 1 )*abs( INCX ) ).
!>           قبل الإدخال، يجب أن تحتوي المصفوفة X المعززة على المتجه الأيمن
!>           b المكون من n عنصر. عند الخروج، يتم الكتابة فوق X
!>           بالمتجه الحل x.
!>

INCX

!>          INCX هو INTEGER
!>           عند الإدخال، يحدد INCX الزيادة لعناصر
!>           X. يجب ألا يكون INCX صفرًا.
!>
