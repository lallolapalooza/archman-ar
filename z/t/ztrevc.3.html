subroutine ztrevc (character side, character howmny, logical, dimension( * ) select, integer n, complex*16, dimension( ldt, * ) t, integer ldt, complex*16, dimension( ldvl, * ) vl, integer ldvl, complex*16, dimension( ldvr, * ) vr, integer ldvr, integer mm, integer m, complex*16, dimension( * ) work, double precision, dimension( * ) rwork, integer info)

ZTREVC

الغرض:

!>
!> ZTREVC يحسب بعض أو كل المتجهات الذاتية اليمنى واليسرى لمصفوفة مثلثة علوية معقدة T.
!> يتم إنتاج المصفوفات من هذا النوع عن طريق التحلل المقياسي لمصفوفة عامة معقدة: A = Q*T*Q**H، كما تم حسابه بواسطة ZHSEQR.
!>
!> المتجه الذاتي الأيمن x والمتجه الذاتي الأيسر y للمصفوفة T المقابلة لقيمة ذاتية w معرفة بواسطة:
!>
!>              T*x = w*x,     (y**H)*T = w*(y**H)
!>
!> حيث تشير y**H إلى المقترن المتناسق للمتجه y.
!> القيم الذاتية ليست مدخلات لهذه الروتين، ولكن يتم قراءتها مباشرة
!> من قطر T.
!>
!> تعيد هذه الروتين المصفوفات X و/أو Y للمتجهات الذاتية اليمنى واليسرى
!> لـ T، أو المنتجات Q*X و/أو Q*Y، حيث Q هي مصفوفة إدخال. إذا كانت Q هي العامل الوحدوي الذي يقلل المصفوفة A إلى
!> الشكل المقياسي T، فإن Q*X و Q*Y هما مصفوفتان للمتجهات الذاتية اليمنى واليسرى
!> لـ A.
!>

المعاملات

SIDE 
!>          SIDE هو CHARACTER*1
!>          = 'R':  حساب المتجهات الذاتية اليمنى فقط؛
!>          = 'L':  حساب المتجهات الذاتية اليسرى فقط؛
!>          = 'B':  حساب المتجهات الذاتية اليمنى واليسرى معًا.
!>

HOWMNY

!>          HOWMNY هو CHARACTER*1
!>          = 'A':  حساب جميع المتجهات الذاتية اليمنى و/أو اليسرى؛
!>          = 'B':  حساب جميع المتجهات الذاتية اليمنى و/أو اليسرى،
!>                  تم تحويلها عكسيًا باستخدام المصفوفات المحددة في
!>                  VR و/أو VL؛
!>          = 'S':  حساب المتجهات الذاتية اليمنى و/أو اليسرى المختارة،
!>                  كما هو محدد بواسطة المصفوفة المنطقية SELECT.
!>

SELECT

!>          SELECT هي مصفوفة LOGICAL (منطقية)، ذات بعد (N)
!>          إذا كان HOWMNY = 'S'، فإن SELECT تحدد المتجهات الذاتية التي سيتم
!>          حسابها.
!>          يتم حساب المتجه الذاتي المقابل للقيمة الذاتية j-th
!>          إذا كان SELECT(j) = .TRUE..
!>          غير مشار إليها إذا كان HOWMNY = 'A' أو 'B'.
!>

N

!>          N هو INTEGER (عدد صحيح)
!>          ترتيب المصفوفة T. N >= 0.
!>

T

!>          T هي مصفوفة COMPLEX*16، ذات بعد (LDT,N)
!>          المصفوفة المثلثية العليا T. يتم تعديل T، ولكن يتم استعادتها
!>          عند الخروج.
!>

LDT

!>          LDT هو INTEGER (عدد صحيح)
!>          البعد الرئيسي للمصفوفة T. LDT >= max(1,N).
!>

VL

!>          VL هي مصفوفة COMPLEX*16، ذات بعد (LDVL,MM)
!>          عند الإدخال، إذا كان SIDE = 'L' أو 'B' و HOWMNY = 'B'، يجب أن
!>          تحتوي VL على مصفوفة Q بحجم N*N (عادةً ما تكون المصفوفة الوحدوية Q لـ
!>          المتجهات المقياسية التي تم إرجاعها بواسطة ZHSEQR).
!>          عند الخروج، إذا كان SIDE = 'L' أو 'B'، تحتوي VL على:
!>          إذا كان HOWMNY = 'A'، المصفوفة Y للمتجهات الذاتية اليسرى لـ T؛
!>          إذا كان HOWMNY = 'B'، المصفوفة Q*Y؛
!>          إذا كان HOWMNY = 'S'، المتجهات الذاتية اليسرى لـ T المحددة بواسطة
!>                           SELECT، مخزنة بشكل متتالي في أعمدة
!>                           VL، بنفس ترتيب قيمها الذاتية.
!>          غير مشار إليها إذا كان SIDE = 'R'.
!>

LDVL

!>          LDVL هو INTEGER (عدد صحيح)
!>          البعد الرئيسي للمصفوفة VL. LDVL >= 1، وإذا
!>          كان SIDE = 'L' أو 'B'، فإن LDVL >= N.
!>

VR

!>          VR هي مصفوفة COMPLEX*16، ذات بعد (LDVR,MM)
!>          عند الإدخال، إذا كان SIDE = 'R' أو 'B' و HOWMNY = 'B'، يجب أن
!>          تحتوي VR على مصفوفة Q بحجم N*N (عادةً ما تكون المصفوفة الوحدوية Q لـ
!>          المتجهات المقياسية التي تم إرجاعها بواسطة ZHSEQR).
!>          عند الخروج، إذا كان SIDE = 'R' أو 'B'، تحتوي VR على:
!>          إذا كان HOWMNY = 'A'، المصفوفة X للمتجهات الذاتية اليمنى لـ T؛
!>          إذا كان HOWMNY = 'B'، المصفوفة Q*X؛
!>          إذا كان HOWMNY = 'S'، المتجهات الذاتية اليمنى لـ T المحددة بواسطة
!>                           SELECT، مخزنة بشكل متتالي في أعمدة
!>                           VR، بنفس ترتيب قيمها الذاتية.
!>          غير مشار إليها إذا كان SIDE = 'L'.
!>

LDVR

!>          LDVR هو INTEGER (عدد صحيح)
!>          البعد الرئيسي للمصفوفة VR. LDVR >= 1، وإذا
!>          كان SIDE = 'R' أو 'B'؛ LDVR >= N.
!>

MM

!>          MM هو INTEGER (عدد صحيح)
!>          عدد الأعمدة في المصفوفات VL و/أو VR. MM >= M.
!>

M

!>          M هو INTEGER (عدد صحيح)
!>          عدد الأعمدة في المصفوفات VL و/أو VR المستخدمة فعليًا
!>          لتخزين المتجهات الذاتية. إذا كان HOWMNY = 'A' أو 'B'، يتم تعيين M
!>          إلى N. كل متجه ذاتي محدد يشغل عمودًا واحدًا.
!>

WORK

!>          WORK هي مصفوفة COMPLEX*16، ذات بعد (2*N)
!>

RWORK

!>          RWORK هي مصفوفة DOUBLE PRECISION، ذات بعد (N)
!>

INFO

!>          INFO هو INTEGER (عدد صحيح)
!>          = 0:  خروج ناجح
!>          < 0:  إذا كان INFO = -i، فإن المعامل i كان له قيمة غير قانونية
!>
