توثيق الوظيفة/الإجراء الفرعي

subroutine ztgex2 (logical wantq, logical wantz, integer n, complex*16, dimension( lda, * ) a, integer lda, complex*16, dimension( ldb, * ) b, integer ldb, complex*16, dimension( ldq, * ) q, integer ldq, complex*16, dimension( ldz, * ) z, integer ldz, integer j1, integer info)

ZTGEX2 تبادل الكتل القطرية المجاورة في زوج من المصفوفات المثلثية (شبه) العليا عن طريق تحويل مكافئ وحدوي.

الغرض:

!>
!> ZTGEX2 تبادل كتل قطرية متجاورة بحجم 1x1 (A11,B11) و (A22,B22)
!> في زوج مصفوفات مثلثية عليا (A, B) عن طريق تحويل مكافئ وحدوي.
!>
!> يجب أن يكون (A, B) في شكل Schur القانوني المعمم، أي أن A و
!> B كلاهما مثلثيان علويان.
!>
!> اختياريًا، يتم تحديث مصفوفتي متجهات Schur المعممة Q و Z.
!>
!>        Q(in) * A(in) * Z(in)**H = Q(out) * A(out) * Z(out)**H
!>        Q(in) * B(in) * Z(in)**H = Q(out) * B(out) * Z(out)**H
!>
!>

المعلمات

WANTQ 
!>          WANTQ هو LOGICAL
!>          .TRUE. : تحديث مصفوفة التحويل اليسرى Q؛
!>          .FALSE.: لا تقم بتحديث Q.
!>

WANTZ

!>          WANTZ هو LOGICAL
!>          .TRUE. : تحديث مصفوفة التحويل اليمنى Z؛
!>          .FALSE.: لا تقم بتحديث Z.
!>

N

!>          N هو INTEGER
!>          رتبة المصفوفتين A و B. N >= 0.
!>

A

!>          A هو مصفوفة COMPLEX*16، أبعاد (LDA,N)
!>          عند الدخول، المصفوفة A في الزوج (A, B).
!>          عند الخروج، المصفوفة A المحدثة.
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرئيسي للمصفوفة A. LDA >= max(1,N).
!>

B

!>          B هو مصفوفة COMPLEX*16، أبعاد (LDB,N)
!>          عند الدخول، المصفوفة B في الزوج (A, B).
!>          عند الخروج، المصفوفة B المحدثة.
!>

LDB

!>          LDB هو INTEGER
!>          البعد الرئيسي للمصفوفة B. LDB >= max(1,N).
!>

Q

!>          Q هي مصفوفة COMPLEX*16، أبعاد (LDQ,N)
!>          إذا كان WANTQ = .TRUE، عند الدخول، المصفوفة الوحدوية Q. عند الخروج،
!>          المصفوفة Q المحدثة.
!>          لا يتم الإشارة إليها إذا كان WANTQ = .FALSE..
!>

LDQ

!>          LDQ هو INTEGER
!>          البعد الرئيسي للمصفوفة Q. LDQ >= 1؛
!>          إذا كان WANTQ = .TRUE.، LDQ >= N.
!>

Z

!>          Z هي مصفوفة COMPLEX*16، أبعاد (LDZ,N)
!>          إذا كان WANTZ = .TRUE، عند الدخول، المصفوفة الوحدوية Z. عند الخروج،
!>          المصفوفة Z المحدثة.
!>          لا يتم الإشارة إليها إذا كان WANTZ = .FALSE..
!>

LDZ

!>          LDZ هو INTEGER
!>          البعد الرئيسي للمصفوفة Z. LDZ >= 1؛
!>          إذا كان WANTZ = .TRUE.، LDZ >= N.
!>

J1

!>          J1 هو INTEGER
!>          الفهرس للكتلة الأولى (A11, B11).
!>

INFO

!>          INFO هو INTEGER
!>           =0: خروج ناجح.
!>           =1: ستكون زوج المصفوفات المحولة (A, B) بعيدة جدًا
!>                عن شكل Schur المعمم؛ المشكلة غير مستقرة.
!>
