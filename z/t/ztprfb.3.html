subroutine ztprfb (side, trans, direct, storev, m, n, k, l, v, ldv, t, ldt, a, lda, b, ldb, work, ldwork)
ZTPRFB يطبق "عاكس كتلة مثلثي خماسي" معقد على مصفوفة معقدة، والتي تتكون من كتلتين.

توثيق الوظيفة/الروتين الفرعي

subroutine ztprfb (character side, character trans, character direct, character storev, integer m, integer n, integer k, integer l, complex*16, dimension( ldv, * ) v, integer ldv, complex*16, dimension( ldt, * ) t, integer ldt, complex*16, dimension( lda, * ) a, integer lda, complex*16, dimension( ldb, * ) b, integer ldb, complex*16, dimension( ldwork, * ) work, integer ldwork)

ZTPRFB يطبق "عاكس كتلة مثلثي خماسي" معقد H أو معكوسه المرافق H**H على مصفوفة معقدة C، والتي تتكون من كتلتين A و B، إما من اليسار أو اليمين.

الغرض:

!>
!> ZTPRFB يطبق عاكس كتلة معقد H أو معكوسه المرافق H**H على مصفوفة معقدة C، والتي تتكون من كتلتين A و B، إما من اليسار أو اليمين.
!>
!>

المعلمات

SIDE 
!>          SIDE is CHARACTER*1
!>          = 'L': تطبيق H أو H**H من اليسار
!>          = 'R': تطبيق H أو H**H من اليمين
!>

TRANS

!>          TRANS is CHARACTER*1
!>          = 'N': تطبيق H (بدون تبديل)
!>          = 'C': تطبيق H**H (تبديل مرافق)
!>

DIRECT

!>          DIRECT is CHARACTER*1
!>          يشير إلى كيفية تشكيل H من حاصل ضرب العواكس الأولية
!>          = 'F': H = H(1) H(2) . . . H(k) (أمامي)
!>          = 'B': H = H(k) . . . H(2) H(1) (خلفي)
!>

STOREV

!>          STOREV is CHARACTER*1
!>          يشير إلى كيفية تخزين المتجهات التي تحدد العواكس الأولية:
!>          = 'C': الأعمدة
!>          = 'R': الصفوف
!>

M

!>          M is INTEGER
!>          عدد صفوف المصفوفة B.
!>          M >= 0.
!>

N

!>          N is INTEGER
!>          عدد أعمدة المصفوفة B.
!>          N >= 0.
!>

K

!>          K is INTEGER
!>          رتبة المصفوفة T، أي عدد العواكس الأولية التي يحدد حاصل ضربها
!>          العاكس الكتلي.
!>          K >= 0.
!>

L

!>          L is INTEGER
!>          رتبة الجزء شبه المنحرف من V.
!>          K >= L >= 0. انظر التفاصيل الإضافية.
!>

V

!>          V is COMPLEX*16 array, dimension
!>                                (LDV,K) if STOREV = 'C'
!>                                (LDV,M) if STOREV = 'R' and SIDE = 'L'
!>                                (LDV,N) if STOREV = 'R' and SIDE = 'R'
!>          المصفوفة الخماسية V، التي تحتوي على العواكس الأولية
!>          H(1), H(2), ..., H(K). انظر التفاصيل الإضافية.
!>

LDV

!>          LDV is INTEGER
!>          البعد الرئيسي للمصفوفة V.
!>          If STOREV = 'C' and SIDE = 'L', LDV >= max(1,M);
!>          if STOREV = 'C' and SIDE = 'R', LDV >= max(1,N);
!>          if STOREV = 'R', LDV >= K.
!>

T

!>          T is COMPLEX*16 array, dimension (LDT,K)
!>          المصفوفة المثلثية K-by-K T في تمثيل العاكس الكتلي.
!>

LDT

!>          LDT is INTEGER
!>          البعد الرئيسي للمصفوفة T.
!>          LDT >= K.
!>

A

!>          A is COMPLEX*16 array, dimension
!>          (LDA,N) if SIDE = 'L' or (LDA,K) if SIDE = 'R'
!>          عند الدخول، المصفوفة A بحجم K-by-N أو M-by-K.
!>          عند الخروج، يتم استبدال A بالجزء المقابل من
!>          H*C أو H**H*C أو C*H أو C*H**H. انظر التفاصيل الإضافية.
!>

LDA

!>          LDA is INTEGER
!>          البعد الرئيسي للمصفوفة A.
!>          If SIDE = 'L', LDA >= max(1,K);
!>          If SIDE = 'R', LDA >= max(1,M).
!>

B

!>          B is COMPLEX*16 array, dimension (LDB,N)
!>          عند الدخول، المصفوفة B بحجم M-by-N.
!>          عند الخروج، يتم استبدال B بالجزء المقابل من
!>          H*C أو H**H*C أو C*H أو C*H**H. انظر التفاصيل الإضافية.
!>

LDB

!>          LDB is INTEGER
!>          البعد الرئيسي للمصفوفة B.
!>          LDB >= max(1,M).
!>

WORK

!>          WORK is COMPLEX*16 array, dimension
!>          (LDWORK,N) if SIDE = 'L',
!>          (LDWORK,K) if SIDE = 'R'.
!>

LDWORK

!>          LDWORK is INTEGER
!>          البعد الرئيسي للمصفوفة WORK.
!>          If SIDE = 'L', LDWORK >= K;
!>          if SIDE = 'R', LDWORK >= M.
!>

التفاصيل الإضافية:

!>
!>  المصفوفة C هي مصفوفة مركبة تتكون من الكتلتين A و B.
!>  الكتلة B بحجم M-by-N؛ إذا كانت SIDE = 'R'، فإن A بحجم M-by-K،
!>  وإذا كانت SIDE = 'L'، فإن A بحجم K-by-N.
!>
!>  إذا كانت SIDE = 'R' و DIRECT = 'F'، فإن C = [A B].
!>
!>  إذا كانت SIDE = 'L' و DIRECT = 'F'، فإن C = [A]
!>                                      [B].
!>
!>  إذا كانت SIDE = 'R' و DIRECT = 'B'، فإن C = [B A].
!>
!>  إذا كانت SIDE = 'L' و DIRECT = 'B'، فإن C = [B]
!>                                      [A].
!>
!>  المصفوفة الخماسية V تتكون من كتلة مستطيلة V1 وكتلة
!>  شبه منحرفة V2. يتم تحديد حجم الكتلة شبه المنحرفة بواسطة
!>  المعلمة L، حيث 0<=L<=K. إذا كانت L=K، فإن كتلة V2 في V مثلثة؛
!>  إذا كانت L=0، فلا توجد كتلة شبه منحرفة، وبالتالي V = V1 مستطيلة.
!>
!>  إذا كانت DIRECT = 'F' و STOREV = 'C': V = [V1]
!>                                         [V2]
!>     - V2 شبه منحرف علوي (أول L صفوف من مصفوفة علوية مثلثية K-by-K)
!>
!>  إذا كانت DIRECT = 'F' و STOREV = 'R': V = [V1 V2]
!>
!>     - V2 شبه منحرف سفلي (أول L أعمدة من مصفوفة سفلية مثلثية K-by-K)
!>
!>  إذا كانت DIRECT = 'B' و STOREV = 'C': V = [V2]
!>                                         [V1]
!>     - V2 شبه منحرف سفلي (آخر L صفوف من مصفوفة سفلية مثلثية K-by-K)
!>
!>  إذا كانت DIRECT = 'B' و STOREV = 'R': V = [V2 V1]
!>
!>     - V2 شبه منحرف علوي (آخر L أعمدة من مصفوفة علوية مثلثية K-by-K)
!>
!>  إذا كانت STOREV = 'C' و SIDE = 'L'، فإن V بحجم M-by-K مع V2 بحجم L-by-K.
!>
!>  إذا كانت STOREV = 'C' و SIDE = 'R'، فإن V بحجم N-by-K مع V2 بحجم L-by-K.
!>
!>  إذا كانت STOREV = 'R' و SIDE = 'L'، فإن V بحجم K-by-M مع V2 بحجم K-by-L.
!>
!>  إذا كانت STOREV = 'R' و SIDE = 'R'، فإن V بحجم K-by-N مع V2 بحجم K-by-L.
!>
