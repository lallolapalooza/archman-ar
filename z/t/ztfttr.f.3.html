subroutine ztfttr (transr, uplo, n, arf, a, lda, info)
ZTFTTR تقوم بنسخ مصفوفة مثلثة من تنسيق الحزمة الكاملة المستطيل (TF) إلى التنسيق الكامل القياسي (TR).

توثيق الوظيفة/الإجراء الفرعي

subroutine ztfttr (character transr, character uplo, integer n, complex*16, dimension( 0: * ) arf, complex*16, dimension( 0: lda-1, 0: * ) a, integer lda, integer info)

ZTFTTR تقوم بنسخ مصفوفة مثلثة من تنسيق الحزمة الكاملة المستطيل (TF) إلى التنسيق الكامل القياسي (TR).

الغرض:

!>
!> ZTFTTR تقوم بنسخ مصفوفة مثلثة A من تنسيق الحزمة الكاملة المستطيل
!> (TF) إلى التنسيق الكامل القياسي (TR).
!>

المعلمات

TRANSR 
!>          TRANSR هو CHARACTER*1
!>          = 'N':  ARF في تنسيق عادي؛
!>          = 'C':  ARF في تنسيق تبديل المرافق؛
!>

UPLO

!>          UPLO هو CHARACTER*1
!>          = 'U':  A مثلثة علوية؛
!>          = 'L':  A مثلثة سفلية.
!>

N

!>          N هو INTEGER
!>          مرتبة المصفوفة A.  N >= 0.
!>

ARF

!>          ARF هو مصفوفة COMPLEX*16، البعد ( N*(N+1)/2 )،
!>          عند الإدخال، المصفوفة المثلثة العلوية أو السفلية A مخزنة
!>          بتنسيق RFP. لمزيد من المناقشة، انظر الملاحظات أدناه.
!>

A

!>          A هو مصفوفة COMPLEX*16، البعد ( LDA, N )
!>          عند الخروج، المصفوفة المثلثة A.  إذا كان UPLO = 'U'، فإن
!>          الجزء العلوي المثلثي المتقدم N-by-N من المصفوفة A يحتوي
!>          على المصفوفة المثلثة العلوية، والجزء السفلي المثلثي بدقة
!>          من A غير مشار إليه.  إذا كان UPLO = 'L'، فإن
!>          الجزء السفلي المثلثي المتقدم N-by-N من المصفوفة A يحتوي
!>          على المصفوفة المثلثة السفلية، والجزء العلوي المثلثي بدقة
!>          من A غير مشار إليه.
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرائد للمصفوفة A.  LDA >= max(1,N).
!>

INFO

!>          INFO هو INTEGER
!>          = 0:  انتهاء ناجح
!>          < 0:  إذا كان INFO = -i، فقد كان الوسيط i له قيمة غير قانونية
!>

!>
!>      AP هو علوي             AP هو سفلي
!>
!>   00 01 02 03 04 05       00
!>      11 12 13 14 15       10 11

!>            33 34 35       30 31 32 33
!>               44 45       40 41 42 43 44
!>                  55       50 51 52 53 54 55

!>
!>  لتكن TRANSR = 'N'. RFP يحتفظ بـ AP كما يلي:
!>  بالنسبة لـ UPLO = 'U'، الجزء شبه المنحرف العلوي A(0:5,0:2) يتكون من
!>  الأعمدة الثلاثة الأخيرة من AP العلوي. المثلث السفلي A(4:6,0:2) يتكون
!>  من تبديل المرافق للأعمدة الثلاثة الأولى من AP العلوي.
!>  بالنسبة لـ UPLO = 'L'، الجزء شبه المنحرف السفلي A(1:6,0:2) يتكون من
!>  الأعمدة الثلاثة الأولى من AP السفلي. المثلث العلوي A(0:2,0:2) يتكون
!>  من تبديل المرافق للأعمدة الثلاثة الأخيرة من AP السفلي.
!>  للتعبير عن المرافق، نضع -- فوق العنصر. هذا يغطي
!>  الحالة N زوجي و TRANSR = 'N'.
!>
!>         RFP A                   RFP A
!>
!>                                -- -- --
!>        03 04 05                33 43 53
!>                                   -- --
!>        13 14 15                00 44 54
!>                                      --
!>        23 24 25                10 11 55

!>

!>
!>     AP هو علوي                 AP هو سفلي
!>
!>   00 01 02 03 04              00
!>      11 12 13 14              10 11
!>         22 23 24              20 21 22
