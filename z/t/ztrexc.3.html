توثيق الوظيفة/الروتين الفرعي

subroutine ztrexc (character compq, integer n, complex*16, dimension( ldt, * ) t, integer ldt, complex*16, dimension( ldq, * ) q, integer ldq, integer ifst, integer ilst, integer info)

ZTREXC

الغرض:

!>
!> ZTREXC يعيد ترتيب صيغة شور لمصفوفة مركبة
!> A = Q*T*Q**H، بحيث يتم نقل العنصر القطري لـ T الذي يحمل فهرس الصف IFST
!> إلى الصف ILST.
!>
!> تتم إعادة ترتيب صيغة شور T عن طريق تحويل تشابه وحدوي
!> Z**H*T*Z، ويتم اختياريًا تحديث مصفوفة Q لمتجهات شور عن طريق
!> ضربها لاحقًا بـ Z.
!>

المعلمات

COMPQ 
!>          COMPQ هو CHARACTER*1
!>          = 'V':  تحديث مصفوفة Q لمتجهات شور؛
!>          = 'N':  لا تقم بتحديث Q.
!>

N

!>          N هو INTEGER
!>          رتبة المصفوفة T. N >= 0.
!>          إذا كان N == 0، يمكن أن تكون الوسائط ILST و IFST أي قيمة.
!>

T

!>          T هو مصفوفة COMPLEX*16، البعد (LDT,N)
!>          عند الإدخال، المصفوفة المثلثية العلوية T.
!>          عند الخروج، المصفوفة المثلثية العلوية المعاد ترتيبها.
!>

LDT

!>          LDT هو INTEGER
!>          البعد الرائد للمصفوفة T. LDT >= max(1,N).
!>

Q

!>          Q هو مصفوفة COMPLEX*16، البعد (LDQ,N)
!>          عند الإدخال، إذا كان COMPQ = 'V'، فإن مصفوفة Q لمتجهات شور.
!>          عند الخروج، إذا كان COMPQ = 'V'، تم ضرب Q لاحقًا بـ
!>          مصفوفة التحويل الوحدوي Z التي تعيد ترتيب T.
!>          إذا كان COMPQ = 'N'، لم يتم الرجوع إلى Q.
!>

LDQ

!>          LDQ هو INTEGER
!>          البعد الرائد للمصفوفة Q. LDQ >= 1، وإذا
!>          كان COMPQ = 'V'، LDQ >= max(1,N).
!>

IFST

!>          IFST هو INTEGER
!>

ILST

!>          ILST هو INTEGER
!>
!>          تحديد إعادة ترتيب العناصر القطرية لـ T:
!>          يتم نقل العنصر الذي يحمل فهرس الصف IFST إلى الصف ILST عن طريق
!>          سلسلة من التبادلات بين العناصر المتجاورة.
!>          1 <= IFST <= N؛ 1 <= ILST <= N.
!>

INFO

!>          INFO هو INTEGER
!>          = 0:  خروج ناجح
!>          < 0:  إذا كان INFO = -i، فإن الوسيط i كان له قيمة غير قانونية
!>
