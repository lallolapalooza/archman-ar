subroutine ztfttp (character transr, character uplo, integer n, complex*16, dimension( 0: * ) arf, complex*16, dimension( 0: * ) ap, integer info)

ZTFTTP تنسخ مصفوفة مثلثية من التنسيق الكامل المعبأ المستطيل (TF) إلى التنسيق المعبأ القياسي (TP).

الغرض:

!>
!> ZTFTTP تنسخ مصفوفة مثلثية A من التنسيق الكامل المعبأ المستطيل
!> (TF) إلى التنسيق المعبأ القياسي (TP).
!>

المعلمات

TRANSR 
!>          TRANSR هو CHARACTER*1
!>          = 'N':  ARF بالتنسيق الطبيعي؛
!>          = 'C':  ARF بالتنسيق المرافق-منقول؛
!>

UPLO

!>          UPLO هو CHARACTER*1
!>          = 'U':  A مثلثية علوية؛
!>          = 'L':  A مثلثية سفلية.
!>

N

!>          N هو INTEGER
!>          ترتيب المصفوفة A. N >= 0.
!>

ARF

!>          ARF هو مصفوفة COMPLEX*16، البعد ( N*(N+1)/2 )،
!>          عند الدخول، المصفوفة المثلثية العلوية أو السفلية A مخزنة
!>          بتنسيق RFP. لمزيد من المناقشة، انظر الملاحظات أدناه.
!>

AP

!>          AP هو مصفوفة COMPLEX*16، البعد ( N*(N+1)/2 )،
!>          عند الخروج، المصفوفة المثلثية العلوية أو السفلية A، معبأة
!>          عمودياً في مصفوفة خطية. يتم تخزين العمود j من A
!>          في المصفوفة AP كالتالي:
!>          إذا UPLO = 'U'، فإن AP(i + (j-1)*j/2) = A(i,j) لـ 1<=i<=j؛
!>          إذا UPLO = 'L'، فإن AP(i + (j-1)*(2n-j)/2) = A(i,j) لـ j<=i<=n.
!>

INFO

!>          INFO هو INTEGER
!>          = 0:  انتهاء ناجح
!>          < 0:  إذا كان INFO = -i، فإن المعلمة i كانت لها قيمة غير قانونية
!>

تفاصيل إضافية:

!>
!>  ننظر أولاً إلى التنسيق المعبأ القياسي عندما يكون N زوجيًا.
!>  نعطي مثالًا حيث N = 6.
!>
!>      AP هو علوي             AP هو سفلي
!>
!>   00 01 02 03 04 05       00
!>      11 12 13 14 15       10 11
!>         22 23 24 25       20 21 22
!>            33 34 35       30 31 32 33
!>               44 45       40 41 42 43 44
!>                  55       50 51 52 53 54 55
!>
!>
!>  لنفترض TRANSR = 'N'. RFP يحتوي على AP كما يلي:
!>  بالنسبة لـ UPLO = 'U'، فإن شبه المنحرف العلوي A(0:5,0:2) يتكون من
!>  الأعمدة الثلاثة الأخيرة من AP العلوي. المثلث السفلي A(4:6,0:2) يتكون من
!>  منقول المرافق للأعمدة الثلاثة الأولى من AP العلوي.
!>  بالنسبة لـ UPLO = 'L'، فإن شبه المنحرف السفلي A(1:6,0:2) يتكون من
!>  الأعمدة الثلاثة الأولى من AP السفلي. المثلث العلوي A(0:2,0:2) يتكون من
!>  منقول المرافق للأعمدة الثلاثة الأخيرة من AP السفلي.
!>  للإشارة إلى المرافق، نضع -- فوق العنصر. هذا يغطي
!>  الحالة N زوجي و TRANSR = 'N'.
!>
!>         RFP A                   RFP A
!>
!>                                -- -- --
!>        03 04 05                33 43 53
!>                                   -- --
!>        13 14 15                00 44 54
!>                                      --
!>        23 24 25                10 11 55
!>
!>        33 34 35                20 21 22
!>        --
!>        00 44 45                30 31 32
!>        -- --
!>        01 11 55                40 41 42
!>        -- -- --
!>        02 12 22                50 51 52
!>
!>  الآن لنفترض TRANSR = 'C'. RFP A في كلتا حالتي UPLO هو مجرد مرافق-
!>  منقول RFP A أعلاه. وبالتالي نحصل على:
!>
!>
!>           RFP A                   RFP A
!>
!>     -- -- -- --                -- -- -- -- -- --
!>     03 13 23 33 00 01 02    33 00 10 20 30 40 50
!>     -- -- -- -- --                -- -- -- -- --
!>     04 14 24 34 44 11 12    43 44 11 21 31 41 51
!>     -- -- -- -- -- --                -- -- -- --
!>     05 15 25 35 45 55 22    53 54 55 22 32 42 52
!>
!>
!>  ننظر بعد ذلك إلى التنسيق المعبأ القياسي عندما يكون N فرديًا.
!>  نعطي مثالًا حيث N = 5.
!>
!>     AP هو علوي                 AP هو سفلي
!>
!>   00 01 02 03 04              00
!>      11 12 13 14              10 11
!>         22 23 24              20 21 22
!>            33 34              30 31 32 33
!>               44              40 41 42 43 44
!>
!>
!>  لنفترض TRANSR = 'N'. RFP يحتوي على AP كما يلي:
!>  بالنسبة لـ UPLO = 'U'، فإن شبه المنحرف العلوي A(0:4,0:2) يتكون من
!>  الأعمدة الثلاثة الأخيرة من AP العلوي. المثلث السفلي A(3:4,0:1) يتكون من
!>  منقول المرافق للأعمدة الاثنتين الأوليين من AP العلوي.
!>  بالنسبة لـ UPLO = 'L'، فإن شبه المنحرف السفلي A(0:4,0:2) يتكون من
!>  الأعمدة الثلاثة الأولى من AP السفلي. المثلث العلوي A(0:1,1:2) يتكون من
!>  منقول المرافق للأعمدة الاثنتين الأخيرتين من AP السفلي.
!>  للإشارة إلى المرافق، نضع -- فوق العنصر. هذا يغطي
!>  الحالة N فردي و TRANSR = 'N'.
!>
!>         RFP A                   RFP A
!>
!>                                   -- --
!>        02 03 04                00 33 43
!>                                      --
!>        12 13 14                10 11 44
!>
!>        22 23 24                20 21 22
!>        --
!>        00 33 34                30 31 32
!>        -- --
!>        01 11 44                40 41 42
!>
!>  الآن لنفترض TRANSR = 'C'. RFP A في كلتا حالتي UPLO هو مجرد مرافق-
!>  منقول RFP A أعلاه. وبالتالي نحصل على:
!>
!>
!>           RFP A                   RFP A
!>

!>     02 12 22 00 01             00 10 20 30 40 50
!>     -- -- -- --                   -- -- -- -- --
