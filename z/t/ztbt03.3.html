subroutine ztbt03 (character uplo, character trans, character diag, integer n, integer kd, integer nrhs, complex*16, dimension( ldab, * ) ab, integer ldab, double precision scale, double precision, dimension( * ) cnorm, double precision tscal, complex*16, dimension( ldx, * ) x, integer ldx, complex*16, dimension( ldb, * ) b, integer ldb, complex*16, dimension( * ) work, double precision resid)

ZTBT03

الغرض:

!>
!> ZTBT03 يحسب الباقي لحل نظام المعادلات المثلثي المُقيَّس  A*x = s*b,  A**T *x = s*b,  أو  A**H *x = s*b
!> عندما تكون A مصفوفة نطاق مثلثية. هنا A**T  تمثل منقولة A، A**H  تمثل منقولة A المترافقة، s هو عدد قياسي،
!> و x و b هما مصفوفتان N × NRHS. نسبة الاختبار هي الحد الأقصى على
!> عدد الأطراف اليمنى من
!>    norm(s*b - op(A)*x) / ( norm(op(A)) * norm(x) * EPS )،
!> حيث op(A) تمثل A، A**T، أو A**H، و EPS هو خطأ الآلة.
!>

المعاملات

UPLO 
!>          UPLO هو CHARACTER*1
!>          يحدد ما إذا كانت المصفوفة A مثلثية عليا أم سفلى.
!>          = 'U':  مثلثية عليا
!>          = 'L':  مثلثية سفلى
!>

TRANS

!>          TRANS هو CHARACTER*1
!>          يحدد العملية المطبقة على A.
!>          = 'N':  A *x = s*b     (بدون منقولة)
!>          = 'T':  A**T *x = s*b  (منقولة)
!>          = 'C':  A**H *x = s*b  (منقولة مترافقة)
!>

DIAG

!>          DIAG هو CHARACTER*1
!>          يحدد ما إذا كانت المصفوفة A مثلثية وحدوية أم لا.
!>          = 'N':  غير وحدوية
!>          = 'U':  وحدوية
!>

N

!>          N هو INTEGER
!>          رتبة المصفوفة A.  N >= 0.
!>

KD

!>          KD هو INTEGER
!>          عدد الأقطار فوق القطر الرئيسية أو تحت القطر الرئيسي للمصفوفة
!>          المثلثية النطاقية A.  KD >= 0.
!>

NRHS

!>          NRHS هو INTEGER
!>          عدد الأطراف اليمنى، أي عدد أعمدة المصفوفتين X و B.  NRHS >= 0.
!>

AB

!>          AB هو مصفوفة COMPLEX*16، البعد (LDAB,N)
!>          المصفوفة النطاقية المثلثية العليا أو السفلى A، مخزنة في
!>          الصفين kd+1 الأوليين من المصفوفة. يتم تخزين العمود j من A
!>          في العمود j من المصفوفة AB على النحو التالي:
!>          إذا كان UPLO = 'U'، فإن AB(kd+1+i-j,j) = A(i,j) لـ max(1,j-kd) <= i <= j؛
!>          إذا كان UPLO = 'L'، فإن AB(1+i-j,j)    = A(i,j) لـ j <= i <= min(n,j+kd).
!>

LDAB

!>          LDAB هو INTEGER
!>          البعد الرائد للمصفوفة AB.  LDAB >= KD+1.
!>

SCALE

!>          SCALE هو DOUBLE PRECISION
!>          عامل القياس s المستخدم في حل النظام المثلثي.
!>

CNORM

!>          CNORM هو مصفوفة DOUBLE PRECISION، البعد (N)
!>          الأعمدة ذات المعيار 1 للمصفوفة A، بدون احتساب القطر.
!>

TSCAL

!>          TSCAL هو DOUBLE PRECISION
!>          عامل القياس المستخدم في حساب المعايير ذات المعيار 1 في CNORM.
!>          CNORM يحتوي فعليًا على معايير أعمدة TSCAL*A.
!>

X

!>          X هو مصفوفة COMPLEX*16، البعد (LDX,NRHS)
!>          متجهات الحل المحسوبة لنظام المعادلات الخطية.
!>

LDX

!>          LDX هو INTEGER
!>          البعد الرائد للمصفوفة X.  LDX >= max(1,N).
!>

B

!>          B هو مصفوفة COMPLEX*16، البعد (LDB,NRHS)
!>          متجهات الطرف الأيمن لنظام المعادلات الخطية.
!>

LDB

!>          LDB هو INTEGER
!>          البعد الرائد للمصفوفة B.  LDB >= max(1,N).
!>

WORK

!>          WORK هو مصفوفة COMPLEX*16، البعد (N)
!>

RESID

!>          RESID هو DOUBLE PRECISION
!>          الحد الأقصى على عدد الأطراف اليمنى من
!>          norm(op(A)*x - s*b) / ( norm(op(A)) * norm(x) * EPS ).
!>
