subroutine ztrtrs (character uplo, character trans, character diag, integer n, integer nrhs, complex*16, dimension( lda, * ) a, integer lda, complex*16, dimension( ldb, * ) b, integer ldb, integer info)

ZTRTRS

الغرض:

!>
!> ZTRTRS يحل نظاماً مثلثياً من الشكل
!>
!>    A * X = B,  A**T * X = B,  أو  A**H * X = B,
!>
!> حيث A مصفوفة مثلثية من الرتبة N، و B مصفوفة N-by-NRHS.
!>
!> يتحقق هذا الإجراء الفرعي من أن A غير مفردة، ولكن يجب ملاحظة أن التفرد الدقيق فقط هو ما يتم اكتشافه.
!> من الممكن أن تكون عناصر قطرية واحدة أو أكثر من A أرقامًا صغيرة جدًا دون أن يشير هذا الإجراء الفرعي إلى خطأ.
!>
!> إذا كان فقدان الدقة الرقمية المحتمل بسبب المصفوفات القريبة من المفردة يمثل مصدر قلق،
!> فيجب على المتصل التحقق من أن A غير مفردة ضمن بعض الحدود قبل استدعاء هذا الإجراء الفرعي.
!>

المعاملات

UPLO 
!>          UPLO هو CHARACTER*1
!>          = 'U':  A مصفوفة مثلثة علوية؛
!>          = 'L':  A مصفوفة مثلثة سفلية.
!>

TRANS

!>          TRANS هو CHARACTER*1
!>          يحدد شكل نظام المعادلات:
!>          = 'N':  A * X = B     (بدون تبديل)
!>          = 'T':  A**T * X = B  (تبديل)
!>          = 'C':  A**H * X = B  (تبديل مترافق)
!>

DIAG

!>          DIAG هو CHARACTER*1
!>          = 'N':  A مصفوفة غير وحدوية؛
!>          = 'U':  A مصفوفة وحدوية.
!>

N

!>          N هو INTEGER
!>          رتبة المصفوفة A.  N >= 0.
!>

NRHS

!>          NRHS هو INTEGER
!>          عدد الأطراف اليمنى، أي عدد أعمدة
!>          المصفوفة B.  NRHS >= 0.
!>

A

!>          A هي مصفوفة COMPLEX*16، البعد (LDA,N)
!>          المصفوفة المثلثية A.  إذا كان UPLO = 'U'، فإن الجزء العلوي المثلثي
!>          الرئيسي N-by-N من المصفوفة A يحتوي على المصفوفة المثلثية العلوية،
!>          الجزء السفلي المثلثي الحقيقي من A لا يتم الرجوع إليه.  إذا كان UPLO = 'L'،
!>          الجزء السفلي المثلثي الرئيسي N-by-N من المصفوفة A يحتوي على المصفوفة المثلثية السفلية،
!>          الجزء العلوي المثلثي الحقيقي من A لا يتم الرجوع إليه.  إذا كان DIAG = 'U'،
!>          فإن العناصر القطرية لـ A لا يتم الرجوع إليها أيضًا وتُفترض أنها 1.
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرئيسي للمصفوفة A.  LDA >= max(1,N).
!>

B

!>          B هي مصفوفة COMPLEX*16، البعد (LDB,NRHS)
!>          عند الدخول، مصفوفة الطرف الأيمن B.
!>          عند الخروج، إذا كان INFO = 0، فإن مصفوفة الحل X.
!>

LDB

!>          LDB هو INTEGER
!>          البعد الرئيسي للمصفوفة B.  LDB >= max(1,N).
!>

INFO

!>          INFO هو INTEGER
!>          = 0:  انتهاء ناجح
!>          < 0: إذا كان INFO = -i، فإن المعامل i كان له قيمة غير قانونية
!>          > 0:  إذا كان INFO = i، فإن العنصر القطري i لـ A هو صفر تمامًا،
!>               مما يشير إلى أن المصفوفة مفردة ولم يتم حساب الحلول X.
!>
