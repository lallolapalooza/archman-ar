توثيق الوظيفة/الإجراء الفرعي

subroutine ztrti2 (character uplo, character diag, integer n, complex*16, dimension( lda, * ) a, integer lda, integer info)

ZTRTI2 تحسب معكوس مصفوفة مثلثة (خوارزمية غير محظورة).

الغرض:

!>
!> ZTRTI2 تحسب معكوس مصفوفة مثلثة عليا أو سفلى معقدة.
!>
!> هذه هي النسخة من المستوى 2 من الخوارزمية (BLAS).
!>

المعاملات

UPLO 
!>          UPLO is CHARACTER*1
!>          يحدد ما إذا كانت المصفوفة A علوية أم سفلية.
!>          = 'U':  مثلثة عليا
!>          = 'L':  مثلثة سفلى
!>

DIAG

!>          DIAG is CHARACTER*1
!>          يحدد ما إذا كانت المصفوفة A مثلثة وحدية أم لا.
!>          = 'N':  مثلثة غير وحدية
!>          = 'U':  مثلثة وحدية
!>

N

!>          N is INTEGER
!>          رتبة المصفوفة A.  N >= 0.
!>

A

!>          A is COMPLEX*16 array, dimension (LDA,N)
!>          عند الإدخال، المصفوفة المثلثة A.  إذا كان UPLO = 'U'، فإن
!>          الجزء العلوي المثلثي العلوي الذي يبلغ حجمه n × n في المصفوفة A يحتوي على
!>          المصفوفة المثلثة العليا، ولا يتم الرجوع إلى الجزء السفلي
!>          المثلثي تمامًا من A.  إذا كان UPLO = 'L'، فإن
!>          الجزء السفلي المثلثي السفلي الذي يبلغ حجمه n × n في المصفوفة A يحتوي على
!>          المصفوفة المثلثة السفلى، ولا يتم الرجوع إلى الجزء العلوي
!>          المثلثي تمامًا من A.  إذا كان DIAG = 'U'، فإن
!>          العناصر القطرية لـ A لا تتم الإشارة إليها أيضًا وتُفترض أن تكون 1.
!>
!>          عند الخروج، معكوس (المثلثي) للمصفوفة الأصلية، بنفس تنسيق التخزين.
!>

LDA

!>          LDA is INTEGER
!>          البعد الرائد للمصفوفة A.  LDA >= max(1,N).
!>

INFO

!>          INFO is INTEGER
!>          = 0: خروج ناجح
!>          < 0: إذا كان INFO = -k، فإن المعامل k كان له قيمة غير قانونية
!>
