توثيق الوظيفة/الإجراء الفرعي

subroutine ztpt05 (character uplo, character trans, character diag, integer n, integer nrhs, complex*16, dimension( * ) ap, complex*16, dimension( ldb, * ) b, integer ldb, complex*16, dimension( ldx, * ) x, integer ldx, complex*16, dimension( ldxact, * ) xact, integer ldxact, double precision, dimension( * ) ferr, double precision, dimension( * ) berr, double precision, dimension( * ) reslts)

ZTPT05

الغرض:

!>
!> ZTPT05 تختبر حدود الخطأ من التحسين التكراري للحل المحسوب لنظام المعادلات A*X = B،
!> حيث A مصفوفة مثلثية بتنسيق تخزين مضغوط.
!>
!> RESLTS(1) = اختبار حد الخطأ
!>           = norm(X - XACT) / ( norm(X) * FERR )
!>
!> يتم إرجاع قيمة كبيرة إذا لم تكن هذه النسبة أقل من واحد.
!>
!> RESLTS(2) = المتبقي من روتين التحسين التكراري
!>           = الحد الأقصى لـ BERR / ( (n+1)*EPS + (*) )، حيث
!>             (*) = (n+1)*UNFL / (min_i (abs(A)*abs(X) +abs(b))_i )
!>

المعلمات

UPLO 
!>          UPLO is CHARACTER*1
!>          تحدد ما إذا كانت المصفوفة A مثلثية علوية أو سفلية.
!>          = 'U':  مثلثية علوية
!>          = 'L':  مثلثية سفلية
!>

TRANS

!>          TRANS is CHARACTER*1
!>          تحدد شكل نظام المعادلات.
!>          = 'N':  A * X = B  (بدون تبديل)
!>          = 'T':  A'* X = B  (تبديل)
!>          = 'C':  A'* X = B  (تبديل مرافق = تبديل)
!>

DIAG

!>          DIAG is CHARACTER*1
!>          تحدد ما إذا كانت المصفوفة A مثلثية موحدة أم لا.
!>          = 'N':  غير موحدة
!>          = 'U':  موحدة
!>

N

!>          N is INTEGER
!>          عدد صفوف المصفوفات X، B، و XACT، وترتيب المصفوفة A.  N >= 0.
!>

NRHS

!>          NRHS is INTEGER
!>          عدد أعمدة المصفوفات X، B، و XACT.  NRHS >= 0.
!>

AP

!>          AP is COMPLEX*16 array, dimension (N*(N+1)/2)
!>          المصفوفة المثلثية العلوية أو السفلية A، مضغوطة عمودياً في
!>          مصفوفة خطية.  يتم تخزين العمود j من A في المصفوفة
!>          AP كالتالي:
!>          if UPLO = 'U', AP(i + (j-1)*j/2) = A(i,j) for 1<=i<=j;
!>          if UPLO = 'L', AP(i + (j-1)*(2n-j)/2) = A(i,j) for j<=i<=n.
!>          إذا كان DIAG = 'U'، لا يتم الإشارة إلى عناصر القطر الرئيسي لـ A
!>          ويفترض أنها 1.
!>

B

!>          B is COMPLEX*16 array, dimension (LDB,NRHS)
!>          متجهات الطرف الأيمن لنظام المعادلات الخطية.
!>

LDB

!>          LDB is INTEGER
!>          البعد الرائد للمصفوفة B.  LDB >= max(1,N).
!>

X

!>          X is COMPLEX*16 array, dimension (LDX,NRHS)
!>          متجهات الحل المحسوبة.  يتم تخزين كل متجه كعمود في
!>          المصفوفة X.
!>

LDX

!>          LDX is INTEGER
!>          البعد الرائد للمصفوفة X.  LDX >= max(1,N).
!>

XACT

!>          XACT is COMPLEX*16 array, dimension (LDX,NRHS)
!>          متجهات الحل الدقيقة.  يتم تخزين كل متجه كعمود في
!>          المصفوفة XACT.
!>

LDXACT

!>          LDXACT is INTEGER
!>          البعد الرائد للمصفوفة XACT.  LDXACT >= max(1,N).
!>

FERR

!>          FERR is DOUBLE PRECISION array, dimension (NRHS)
!>          حدود الخطأ الأمامي المقدرة لكل متجه حل
!>          X.  إذا كان XTRUE هو الحل الحقيقي، فإن FERR يحد من حجم
!>          أكبر مدخل في (X - XTRUE) مقسومًا على حجم
!>          أكبر مدخل في X.
!>

BERR

!>          BERR is DOUBLE PRECISION array, dimension (NRHS)
!>          الخطأ الخلفي النسبي من حيث المكونات لكل متجه حل
!>          (أي، أصغر تغيير نسبي في أي مدخل في A
!>          أو B يجعل X حلاً دقيقًا).
!>

RESLTS

!>          RESLTS is DOUBLE PRECISION array, dimension (2)
!>          الحد الأقصى لمتجهات الحل NRHS من النسب التالية:
!>          RESLTS(1) = norm(X - XACT) / ( norm(X) * FERR )
!>          RESLTS(2) = BERR / ( (n+1)*EPS + (*) )
!>
