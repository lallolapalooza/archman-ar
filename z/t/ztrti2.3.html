subroutine ztrti2 (uplo, diag, n, a, lda, info)
ZTRTI2 تحسب معكوس مصفوفة مثلثية (خوارزمية غير محظورة).

توثيق الوظيفة/الروتين الفرعي

subroutine ztrti2 (character uplo, character diag, integer n, complex*16, dimension( lda, * ) a, integer lda, integer info)

ZTRTI2 تحسب معكوس مصفوفة علوية أو سفلية مثلثية.

الغرض:

!>
!> ZTRTI2 تحسب معكوس مصفوفة مثلثية علوية أو سفلية معقدة.
!>
!> هذا هو الإصدار من المستوى 2 من BLAS للخوارزمية.
!>

المعلمات

UPLO 
!>          UPLO is CHARACTER*1
!>          تحدد ما إذا كانت المصفوفة A علوية أم سفلية مثلثية.
!>          = 'U':  علوية مثلثية
!>          = 'L':  سفلية مثلثية
!>

DIAG

!>          DIAG is CHARACTER*1
!>          تحدد ما إذا كانت المصفوفة A مثلثية موحدة أم لا.
!>          = 'N':  غير موحدة مثلثية
!>          = 'U':  موحدة مثلثية
!>

N

!>          N is INTEGER
!>          رتبة المصفوفة A.  N >= 0.
!>

A

!>          A is COMPLEX*16 array, dimension (LDA,N)
!>          عند الإدخال، المصفوفة المثلثية A.  إذا كان UPLO = 'U'، فإن
!>          الجزء العلوي المثلثي من المصفوفة A الذي حجمه n × n يحتوي على
!>          المصفوفة المثلثية العلوية، والجزء السفلي المثلثي الحاد
!>          من A لا يتم الرجوع إليه.  إذا كان UPLO = 'L'، فإن
!>          الجزء السفلي المثلثي من المصفوفة A الذي حجمه n × n يحتوي على
!>          المصفوفة المثلثية السفلية، والجزء العلوي المثلثي الحاد
!>          من A لا يتم الرجوع إليه.  إذا كان DIAG = 'U'، فإن
!>          العناصر القطرية لـ A لا يتم الرجوع إليها أيضًا وتُفترض أنها 1.
!>
!>          عند الخروج، معكوس (مثلثي) المصفوفة الأصلية، بنفس تنسيق التخزين.
!>

LDA

!>          LDA is INTEGER
!>          البعد الرائد للمصفوفة A.  LDA >= max(1,N).
!>

INFO

!>          INFO is INTEGER
!>          = 0: خروج ناجح
!>          < 0: إذا كان INFO = -k، فإن المعلمة k لها قيمة غير قانونية
!>
