subroutine ztpqrt (integer m, integer n, integer l, integer nb, complex*16, dimension( lda, * ) a, integer lda, complex*16, dimension( ldb, * ) b, integer ldb, complex*16, dimension( ldt, * ) t, integer ldt, complex*16, dimension( * ) work, integer info)

ZTPQRT

الغرض:

!>
!> ZTPQRT يحسب تفكيك QR مسدود لمصفوفة C المركبة، والتي تتكون من
!> كتلة مثلثة A وكتلة خماسية B، باستخدام تمثيل WY المضغوط لـ Q.
!>

التفاصيل الإضافية:

!>
!>  المصفوفة المدخلة C هي مصفوفة (N+M)-by-N
!>
!>               C = [ A ]
!>                   [ B ]
!>
!>  حيث A هي مصفوفة مثلثة علوية N-by-N، و B هي مصفوفة خماسية M-by-N
!>  تتكون من مصفوفة مستطيلة B1 ذات الأبعاد (M-L)-by-N في الأعلى
!>  ومصفوفة شبه منحرفة علوية L-by-N تسمى B2:
!>
!>               B = [ B1 ]  <- (M-L)-by-N مستطيلة
!>                   [ B2 ]  <-     L-by-N شبه منحرفة علوية.
!>
!>  تتكون المصفوفة شبه المنحرفة B2 من أول L صفوف من مصفوفة مثلثة علوية
!>  N-by-N، حيث 0 <= L <= MIN(M,N). إذا كان L=0، فإن B مستطيلة M-by-N؛
!>  إذا كان M=L=N، فإن B مثلثة علوية.
!>
!>  تخزن المصفوفة W المُنعكسات الأولية H(i) في العمود i
!>  تحت القطر (في A) في المصفوفة المدخلة C ذات الأبعاد (N+M)-by-N
!>
!>               C = [ A ]  <- مثلثة علوية N-by-N
!>                   [ B ]  <- خماسية M-by-N
!>
!>  بحيث يمكن تمثيل W على النحو التالي
!>
!>               W = [ I ]  <- هوية، N-by-N
!>                   [ V ]  <- M-by-N، نفس شكل B.
!>
!>  وبالتالي، فإن جميع المعلومات اللازمة لـ W موجودة عند الخروج في B،
!>  والتي نسميها V أعلاه. لاحظ أن V لها نفس شكل B؛ أي،
!>
!>               V = [ V1 ] <- (M-L)-by-N مستطيلة
!>                   [ V2 ] <-     L-by-N شبه منحرفة علوية.
!>
!>  تمثل أعمدة V المتجهات التي تحدد H(i).
!>
!>  عدد الكتل هو B = ceiling(N/NB)، حيث كل
!>  كتلة من الرتبة NB باستثناء الكتلة الأخيرة، والتي هي من الرتبة
!>  IB = N - (B-1)*NB. لكل كتلة من الكتل B، يتم حساب مُنعكس كتلة مثلثية علوية
!>  : T1, T2, ..., TB. يتم تخزين T ذات الأبعاد NB-by-NB (و IB-by-IB
!>  للأخيرة) في المصفوفة T ذات الأبعاد NB-by-N على النحو التالي
!>
!>               T = [T1 T2 ... TB].
!>
