subroutine ztrsyl3 (character trana, character tranb, integer isgn, integer m, integer n, complex*16, dimension( lda, * ) a, integer lda, complex*16, dimension( ldb, * ) b, integer ldb, complex*16, dimension( ldc, * ) c, integer ldc, double precision scale, double precision, dimension( ldswork, * ) swork, integer ldswork, integer info)

ZTRSYL3

الغرض:

!>
!>  ZTRSYL3 تحل معادلة مصفوفة سيلفستر المعقدة:
!>
!>     op(A)*X + X*op(B) = scale*C أو
!>     op(A)*X - X*op(B) = scale*C,
!>
!>  حيث op(A) = A أو A**H، و A و B كلاهما مثلثي علوي. A
!>  هو M في M و B هو N في N؛ C في الطرف الأيمن و X الحل هما
!>  M في N؛ و scale هو عامل قياس خرج، تم تعيينه <= 1 لتجنب
!>  التجاوز في X.
!>
!>  هذه هي النسخة الكتلة من الخوارزمية.
!>

المعلمات

TRANA 
!>          TRANA هو CHARACTER*1
!>          يحدد خيار op(A):
!>          = 'N': op(A) = A    (لا يوجد تبديل)
!>          = 'C': op(A) = A**H (تبديل مترافق)
!>

TRANB

!>          TRANB هو CHARACTER*1
!>          يحدد خيار op(B):
!>          = 'N': op(B) = B    (لا يوجد تبديل)
!>          = 'C': op(B) = B**H (تبديل مترافق)
!>

ISGN

!>          ISGN هو INTEGER
!>          يحدد الإشارة في المعادلة:
!>          = +1: حل op(A)*X + X*op(B) = scale*C
!>          = -1: حل op(A)*X - X*op(B) = scale*C
!>

M

!>          M هو INTEGER
!>          رتبة المصفوفة A، وعدد الصفوف في
!>          المصفوفتين X و C. M >= 0.
!>

N

!>          N هو INTEGER
!>          رتبة المصفوفة B، وعدد الأعمدة في
!>          المصفوفتين X و C. N >= 0.
!>

A

!>          A هو COMPLEX*16 array، dimension (LDA,M)
!>          المصفوفة المثلثية العلوية A.
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرائد للمصفوفة A. LDA >= max(1,M).
!>

B

!>          B هو COMPLEX*16 array، dimension (LDB,N)
!>          المصفوفة المثلثية العلوية B.
!>

LDB

!>          LDB هو INTEGER
!>          البعد الرائد للمصفوفة B. LDB >= max(1,N).
!>

C

!>          C هو COMPLEX*16 array، dimension (LDC,N)
!>          عند الدخول، مصفوفة الطرف الأيمن C بحجم M في N.
!>          عند الخروج، يتم استبدال C بالمصفوفة الحل X.
!>

LDC

!>          LDC هو INTEGER
!>          البعد الرائد للمصفوفة C. LDC >= max(1,M)
!>

SCALE

!>          SCALE هو DOUBLE PRECISION
!>          عامل القياس، scale، يتم تعيينه <= 1 لتجنب التجاوز في X.
!>

SWORK

!>          SWORK هو DOUBLE PRECISION array، dimension (MAX(2, ROWS),
!>          MAX(1,COLS)).
!>          عند الخروج، إذا كان INFO = 0، ترجع SWORK(1) قيمة ROWS المثلى
!>          و SWORK(2) ترجع COLS المثلى.
!>

LDSWORK

!>          LDSWORK هو INTEGER
!>          LDSWORK >= MAX(2,ROWS)، حيث ROWS = ((M + NB - 1) / NB + 1)
!>          و NB هو حجم الكتلة الأمثل.
!>
!>          إذا كان LDSWORK = -1، فسيتم افتراض استعلام مساحة العمل؛ الإجراء
!!>          يحسب فقط الأبعاد المثلى لمصفوفة SWORK،
!>          يرجع هذه القيم كأول وثاني إدخال في مصفوفة SWORK،
!>          ولا يصدر XERBLA أي رسالة خطأ تتعلق بـ LWORK.
!>

INFO

!>          INFO هو INTEGER
!>          = 0: خروج ناجح
!>          < 0: إذا كان INFO = -i، فإن الوسيط i كان له قيمة غير قانونية
!>          = 1: A و B لهما قيم ذاتية مشتركة أو قريبة جدًا؛ تم استخدام
!>               قيم مشوهة لحل المعادلة (لكن المصفوفتين
!>               A و B لم تتغيرا).
!>
