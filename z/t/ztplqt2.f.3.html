subroutine ztplqt2 (m, n, l, a, lda, b, ldb, t, ldt, info)
ZTPLQT2 تحسب تحليل LQ لمصفوفة "مثلثة خماسية" حقيقية أو مركبة، وهي تتكون من كتلة مثلثة وكتلة خماسية، باستخدام تمثيل WY المدمج لـ Q.

توثيق الوظيفة/الروتين الفرعي

subroutine ztplqt2 (integer m, integer n, integer l, complex*16, dimension( lda, * ) a, integer lda, complex*16, dimension( ldb, * ) b, integer ldb, complex*16, dimension( ldt, * ) t, integer ldt, integer info)

ZTPLQT2 تحسب تحليل LQ لمصفوفة مركبة "مثلثة خماسية"، والتي تتكون من كتلة مثلثة وكتلة خماسية، باستخدام تمثيل WY المدمج لـ Q.

الغرض:

!>
!> ZTPLQT2 تحسب تحليل LQ لمصفوفة مركبة C، والتي تتكون من كتلة مثلثة A وكتلة خماسية B،
!> باستخدام تمثيل WY المدمج لـ Q.
!>

تفاصيل إضافية:

!>
!> المصفوفة المدخلة C هي مصفوفة مقاس M × (M+N)
!>
!>               C = [ A ][ B ]
!>
!>
!> حيث A هي مصفوفة مثلثة سفلى مقاس M × M، و B هي مصفوفة خماسية مقاس M × N
!> تتكون من مصفوفة مستطيلة B1 مقاس M × (N-L) على يسار مصفوفة شبه منحرف سفلى B2 مقاس M × L:
!>
!>               B = [ B1 ][ B2 ]
!>                   [ B1 ]  <-     مستطيلة مقاس M × (N-L)
!>                   [ B2 ]  <-     شبه منحرفة سفلى مقاس M × L.
!>
!> المصفوفة شبه المنحرفة السفلى B2 تتكون من الأعمدة L الأولى لمصفوفة
!> مثلثة سفلى مقاس N × N، حيث 0 <= L <= MIN(M,N). إذا كانت L=0،
!> تكون B مستطيلة مقاس M × N؛ إذا كانت M=L=N، تكون B مثلثة سفلى.
!>
!> المصفوفة W تخزن المنعكسات الأولية H(i) في الصف i
!> فوق القطر (في A) في المصفوفة المدخلة C مقاس M × (M+N)
!>
!>               C = [ A ][ B ]
!>                   [ A ]  <- مثلثة سفلى مقاس M × M
!>                   [ B ]  <- خماسية مقاس M × N
!>
!> بحيث يمكن تمثيل W كـ
!>
!>               W = [ I ][ V ]
!>                   [ I ]  <- هوية، مقاس M × M
!>                   [ V ]  <- مقاس M × N، نفس شكل B.
!>
!> وبالتالي، فإن جميع المعلومات المطلوبة لـ W موجودة عند الخروج في B،
!> والتي نسميها V أعلاه. لاحظ أن V لها نفس شكل B؛ أي،
!>
!>               W = [ V1 ][ V2 ]
!>                   [ V1 ] <-     مستطيلة مقاس M × (N-L)
!>                   [ V2 ] <-     شبه منحرفة سفلى مقاس M × L.
!>
!> صفوف V تمثل المتجهات التي تحدد H(i).
!> المنعكس الكتلي H مقاس (M+N) × (M+N) يُعطى بعد ذلك بـ
!>
!>               H = I - W**T * T * W
!>
!> حيث W^H هو المرافق المتعامد لـ W و T هو العامل المثلث العلوي
!> للمنعكس الكتلي.
!>
