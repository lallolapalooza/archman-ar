subroutine ztrttf (transr, uplo, n, a, lda, arf, info)
ZTRTTF تنسخ مصفوفة مثلثية من التنسيق الكامل القياسي (TR) إلى التنسيق المعبأ الكامل المستطيلي (TF).

وثائق الوظيفة/الروتين الفرعي

subroutine ztrttf (character transr, character uplo, integer n, complex*16, dimension( 0: lda-1, 0: * ) a, integer lda, complex*16, dimension( 0: * ) arf, integer info)

ZTRTTF تنسخ مصفوفة مثلثية A من التنسيق الكامل القياسي (TR) إلى التنسيق المعبأ الكامل المستطيلي (TF).

الغرض:

!>
!> ZTRTTF تنسخ مصفوفة مثلثية A من التنسيق الكامل القياسي (TR)
!> إلى التنسيق المعبأ الكامل المستطيلي (TF).
!>

المعلمات

TRANSR 
!>          TRANSR هو CHARACTER*1
!>          = 'N':  ARF في الوضع العادي مطلوب؛
!>          = 'C':  ARF في وضع التبديل المترافق مطلوب؛
!>

UPLO

!>          UPLO هو CHARACTER*1
!>          = 'U':  A هي مثلثية عليا؛
!>          = 'L':  A هي مثلثية سفلى.
!>

N

!>          N هو INTEGER
!>          رتبة المصفوفة A.  N >= 0.
!>

A

!>          A هي مصفوفة COMPLEX*16، البعد ( LDA, N )
!>          عند الدخول، تحتوي الجزء العلوي المثلثي الرئيسي N-by-N من المصفوفة A على
!>          المصفوفة المثلثية العليا، ولا يتم الرجوع إلى الجزء المثلثي السفلي الصارم من A.  إذا كان UPLO = 'L'، فإن
!>          الجزء السفلي المثلثي الرئيسي N-by-N من المصفوفة A يحتوي على
!>          المصفوفة المثلثية السفلى، ولا يتم الرجوع إلى الجزء العلوي المثلثي الصارم من A.
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرئيسي للمصفوفة A.  LDA >= max(1,N).
!>

ARF

!>          ARF هي مصفوفة COMPLEX*16، البعد ( N*(N+1)/2 )،
!>          عند الخروج، تحتوي المصفوفة المثلثية العليا أو السفلى A المخزنة بتنسيق
!>          RFP. لمزيد من المناقشة، انظر الملاحظات أدناه.
!>

INFO

!>          INFO هو INTEGER
!>          = 0:  انتهاء ناجح
!>          < 0:  إذا كان INFO = -i، فإن المعلمة i كانت ذات قيمة غير قانونية
!>
