subroutine ztgsy2 (character trans, integer ijob, integer m, integer n, complex*16, dimension( lda, * ) a, integer lda, complex*16, dimension( ldb, * ) b, integer ldb, complex*16, dimension( ldc, * ) c, integer ldc, complex*16, dimension( ldd, * ) d, integer ldd, complex*16, dimension( lde, * ) e, integer lde, complex*16, dimension( ldf, * ) f, integer ldf, double precision scale, double precision rdsum, double precision rdscal, integer info)

ZTGSY2 تحل معادلة سيلفستر المعممة (خوارزمية غير مجمعة).

الغرض:

!>
!> ZTGSY2 تحل معادلة سيلفستر المعممة
!>
!>             A * R - L * B = scale * C               (1)
!>             D * R - L * E = scale * F
!>
!> باستخدام BLAS من المستوى 1 و 2، حيث R و L هما مصفوفتان غير معروفتين بحجم M-by-N،
!> (A, D)، (B, E) و (C, F) هي أزواج مصفوفات معطاة بحجم M-by-M،
!> N-by-N و M-by-N، على التوالي. A، B، D و E مصفوفات مثلثية عليا
!> (أي، (A,D) و (B,E) في شكل شوري معمّم).
!>
!> الحل (R, L) يحل محل (C, F). 0 <= SCALE <= 1 هو عامل القياس الناتج
!> الذي تم اختياره لتجنب تجاوز السعة.
!>
!> في التدوين المصفوفي، يطابق حل المعادلة (1) الحل
!> Zx = scale * b، حيث Z معرفة كـ
!>
!>        Z = [ kron(In, A)  -kron(B**H, Im) ]             (2)
!>            [ kron(In, D)  -kron(E**H, Im) ],
!>
!> Ik هي المصفوفة الهوية بحجم k و X**H هي المرافقة المنقولة لـ X.
!> kron(X, Y) هو حاصل ضرب كرونيكر بين المصفوفتين X و Y.
!>
!> إذا كان TRANS = 'C'، يتم حل y في النظام المرافق المنقول Z**H*y = scale*b
!>، والذي يعادل الحل لـ R و L في
!>
!>             A**H * R  + D**H * L   = scale * C           (3)
!>             R  * B**H + L  * E**H  = scale * -F
!>
!> يتم استخدام هذه الحالة لحساب تقدير لـ Dif[(A, D), (B, E)] =
!> = sigma_min(Z) باستخدام الاتصال العكسي مع ZLACON.
!>
!> ZTGSY2 أيضًا (IJOB >= 1) تساهم في حساب ZTGSYL
!> لحد أعلى على الفصل بين أزواج المصفوفات. ثم
!> (A, D)، (B, E) هي أجزاء فرعية من زوجين مصفوفيين في
!> ZTGSYL.
!>

المعاملات

TRANS 
!>          TRANS is CHARACTER*1
!>          = 'N': حل معادلة سيلفستر المعممة (1).
!>          = 'T': حل النظام 'المنقول' (3).
!>

IJOB

!>          IJOB is INTEGER
!>          يحدد نوع الوظيفة المراد تنفيذها.
!>          =0: حل (1) فقط.
!>          =1: يتم حساب مساهمة من هذا النظام الفرعي لتقدير قائم على معيار فروبنيوس
!>              للفصل بين زوجين مصفوفيين. (يتم استخدام استراتيجية التطلع).
!>          =2: يتم حساب مساهمة من هذا النظام الفرعي لتقدير قائم على معيار فروبنيوس
!>              للفصل بين زوجين مصفوفيين. (يتم استخدام DGECON على الأنظمة الفرعية.)
!>          غير مشار إليه إذا كان TRANS = 'T'.
!>

M

!>          M is INTEGER
!>          عند الإدخال، يحدد M ترتيب A و D، وبعد الصف
!>          بعد C، F، R و L.
!>

N

!>          N is INTEGER
!>          عند الإدخال، يحدد N ترتيب B و E، وبعد العمود
!>          بعد C، F، R و L.
!>

A

!>          A is COMPLEX*16 array, dimension (LDA, M)
!>          عند الإدخال، يحتوي A على مصفوفة مثلثية عليا.
!>

LDA

!>          LDA is INTEGER
!>          البعد الرائد للمصفوفة A. LDA >= max(1, M).
!>

B

!>          B is COMPLEX*16 array, dimension (LDB, N)
!>          عند الإدخال، يحتوي B على مصفوفة مثلثية عليا.
!>

LDB

!>          LDB is INTEGER
!>          البعد الرائد للمصفوفة B. LDB >= max(1, N).
!>

C

!>          C is COMPLEX*16 array, dimension (LDC, N)
!>          عند الإدخال، يحتوي C على الجانب الأيمن للمعادلة المصفوفية الأولى
!>          في (1).
!>          عند الخروج، إذا كان IJOB = 0، فقد تم استبدال C بالحل
!>          R.
!>

LDC

!>          LDC is INTEGER
!>          البعد الرائد للمصفوفة C. LDC >= max(1, M).
!>

D

!>          D is COMPLEX*16 array, dimension (LDD, M)
!>          عند الإدخال، يحتوي D على مصفوفة مثلثية عليا.
!>

LDD

!>          LDD is INTEGER
!>          البعد الرائد للمصفوفة D. LDD >= max(1, M).
!>

E

!>          E is COMPLEX*16 array, dimension (LDE, N)
!>          عند الإدخال، يحتوي E على مصفوفة مثلثية عليا.
!>

LDE

!>          LDE is INTEGER
!>          البعد الرائد للمصفوفة E. LDE >= max(1, N).
!>

F

!>          F is COMPLEX*16 array, dimension (LDF, N)
!>          عند الإدخال، يحتوي F على الجانب الأيمن للمعادلة المصفوفية الثانية
!>          في (1).
!>          عند الخروج، إذا كان IJOB = 0، فقد تم استبدال F بالحل
!>          L.
!>

LDF

!>          LDF is INTEGER
!>          البعد الرائد للمصفوفة F. LDF >= max(1, M).
!>

SCALE

!>          SCALE is DOUBLE PRECISION
!>          عند الخروج، 0 <= SCALE <= 1. إذا كان 0 < SCALE < 1، فإن الحلول
!>          R و L (C و F عند الإدخال) ستحمل الحلول لـ
!>          نظام مضطرب قليلاً ولكن المصفوفات المدخلة A، B، D و
!>          E لم تتغير. إذا كان SCALE = 0، ستحمل R و L
!>          الحلول للنظام المتجانس مع C = F = 0.
!>          عادة، SCALE = 1.
!>

RDSUM

!>          RDSUM is DOUBLE PRECISION
!>          عند الإدخال، مجموع مربعات المساهمات المحسوبة لـ
!>          تقدير Dif قيد الحساب بواسطة ZTGSYL، حيث
!>          تم استخراج عامل القياس RDSCAL (انظر أدناه).
!>          عند الخروج، تم تحديث مجموع المربعات المقابل مع
!>          المساهمات من النظام الفرعي الحالي.
!>          إذا كان TRANS = 'T'، لا يتم لمس RDSUM.
!>          ملاحظة: RDSUM له معنى فقط عندما يتم استدعاء ZTGSY2 بواسطة
!>          ZTGSYL.
!>

RDSCAL

!>          RDSCAL is DOUBLE PRECISION
!>          عند الإدخال، عامل القياس المستخدم لمنع تجاوز السعة في RDSUM.
!>          عند الخروج، يتم تحديث RDSCAL بالنسبة للمساهمات الحالية
!>          في RDSUM.
!>          إذا كان TRANS = 'T'، لا يتم لمس RDSCAL.
!>          ملاحظة: RDSCAL له معنى فقط عندما يتم استدعاء ZTGSY2 بواسطة
!>          ZTGSYL.
!>

INFO

!>          INFO is INTEGER
!>          عند الخروج، إذا تم تعيين INFO إلى
!>            =0: خروج ناجح
!>            <0: إذا كان INFO = -i، فإن وسيط الإدخال رقم i غير قانوني.
!>            >0: أزواج المصفوفات (A, D) و (B, E) لها قيم ذاتية مشتركة أو قريبة جدًا.
!>
