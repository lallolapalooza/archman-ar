subroutine ztbsv (character uplo, character trans, character diag, integer n, integer k, complex*16, dimension(lda,*) a, integer lda, complex*16, dimension(*) x, integer incx)

ZTBSV

الغرض:

!>
!> ZTBSV  يحصل على واحد من أنظمة المعادلات
!>
!>    A*x = b,   أو   A**T*x = b,   أو   A**H*x = b,
!>
!> حيث b و x متجهان مكونان من n عنصر، و A مصفوفة مربعة من الدرجة n،
!> مثلثية علوية أو سفلية، موحدة أو غير موحدة، بها ( k + 1 ) قطر.
!>
!> لا تتضمن هذه الروتينات أي اختبار للانفراد أو شبه الانفراد.
!> يجب إجراء مثل هذه الاختبارات قبل استدعاء هذا الروتين.
!>

المعلمات

UPLO 
!>          UPLO is CHARACTER*1
!>           عند الإدخال، يحدد UPLO ما إذا كانت المصفوفة علوية أم سفلية
!>           مثلثية على النحو التالي:
!>
!>              UPLO = 'U' or 'u'   A مصفوفة مثلثية علوية.
!>
!>              UPLO = 'L' or 'l'   A مصفوفة مثلثية سفلية.
!>

TRANS

!>          TRANS is CHARACTER*1
!>           عند الإدخال، يحدد TRANS المعادلات التي سيتم حلها على النحو التالي:
!>
!>              TRANS = 'N' or 'n'   A*x = b.
!>
!>              TRANS = 'T' or 't'   A**T*x = b.
!>
!>              TRANS = 'C' or 'c'   A**H*x = b.
!>

DIAG

!>          DIAG is CHARACTER*1
!>           عند الإدخال، يحدد DIAG ما إذا كانت A موحدة أم لا
!>           مثلثية على النحو التالي:
!>
!>              DIAG = 'U' or 'u'   A مفترضة أنها موحدة مثلثية.
!>
!>              DIAG = 'N' or 'n'   A ليست مفترضة أنها موحدة
!>                                  مثلثية.
!>

N

!>          N is INTEGER
!>           عند الإدخال، يحدد N درجة المصفوفة A.
!>           يجب أن يكون N على الأقل صفر.
!>

K

!>          K is INTEGER
!>           عند الإدخال مع UPLO = 'U' أو 'u'، يحدد K عدد
!>           القطريات الفائقة للمصفوفة A.
!>           عند الإدخال مع UPLO = 'L' أو 'l'، يحدد K عدد
!>           القطريات الفرعية للمصفوفة A.
!>           يجب أن يحقق K  0 .le. K.
!>

A

!>          A is COMPLEX*16 array, dimension ( LDA, N )
!>           قبل الدخول مع UPLO = 'U' أو 'u'، يجب أن يحتوي الجزء العلوي ( k + 1 )
!>           في المصفوفة A على الجزء المثلثي العلوي الشريط للمصفوفة المعاملة،
!>           المقدمة عمودًا بعمود، مع القطر الرئيسي للمصفوفة في الصف
!>           ( k + 1 ) من المصفوفة، ويبدأ القطر الفائق الأول عند
!>           الموضع 2 في الصف k، وهكذا. لا يتم الرجوع إلى المثلث العلوي الأيسر k × k
!>           للمصفوفة A.
!>           سيقوم جزء البرنامج التالي بنقل مصفوفة شريطية مثلثة علوية
!>           من تخزين المصفوفة الكاملة التقليدي إلى تخزين الشريط:
!>
!>                 DO 20, J = 1, N
!>                    M = K + 1 - J
!>                    DO 10, I = MAX( 1, J - K ), J
!>                       A( M + I, J ) = matrix( I, J )
!>              10    CONTINUE
!>              20 CONTINUE
!>
!>           قبل الدخول مع UPLO = 'L' أو 'l'، يجب أن يحتوي الجزء العلوي ( k + 1 )
!>           في المصفوفة A على الجزء المثلثي السفلي الشريط للمصفوفة المعاملة،
!>           المقدمة عمودًا بعمود، مع القطر الرئيسي للمصفوفة في الصف 1
!>           للمصفوفة، ويبدأ القطر الفرعي الأول عند الموضع 1 في
!>           الصف 2، وهكذا. لا يتم الرجوع إلى المثلث السفلي الأيمن k × k
!>           للمصفوفة A.
!>           سيقوم جزء البرنامج التالي بنقل مصفوفة شريطية مثلثة سفلية
!>           من تخزين المصفوفة الكاملة التقليدي إلى تخزين الشريط:
!>
!>                 DO 20, J = 1, N
!>                    M = 1 - J
!>                    DO 10, I = J, MIN( N, J + K )
!>                       A( M + I, J ) = matrix( I, J )
!>              10    CONTINUE
!>              20 CONTINUE
!>
!>           لاحظ أنه عند DIAG = 'U' أو 'u'، فإن عناصر المصفوفة A
!>           المقابلة للعناصر القطرية للمصفوفة لا يتم الرجوع إليها، ولكنها مفترضة
!>           أنها وحدة.
!>

LDA

!>          LDA is INTEGER
!>           عند الإدخال، يحدد LDA البعد الأول لـ A كما هو معلن
!>           في البرنامج المستدعي. يجب أن يكون LDA على الأقل
!>           ( k + 1 ).
!>

X

!>          X is COMPLEX*16 array, dimension at least
!>           ( 1 + ( n - 1 )*abs( INCX ) ).
!>           قبل الدخول، يجب أن تحتوي المصفوفة X المتزايدة على المتجه الأيمن
!>           b المكون من n عنصر. عند الخروج، يتم استبدال X
!>           بالمتجه الحل x.
!>

INCX

!>          INCX is INTEGER
!>           عند الإدخال، يحدد INCX الزيادة لعناصر X.
!>           يجب ألا يكون INCX صفرًا.
!>
