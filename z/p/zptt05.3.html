توثيق الدالة/الروتين الفرعي

subroutine zptt05 (integer n, integer nrhs, double precision, dimension( * ) d, complex*16, dimension( * ) e, complex*16, dimension( ldb, * ) b, integer ldb, complex*16, dimension( ldx, * ) x, integer ldx, complex*16, dimension( ldxact, * ) xact, integer ldxact, double precision, dimension( * ) ferr, double precision, dimension( * ) berr, double precision, dimension( * ) reslts)

ZPTT05

الغرض:

!>
!> ZPTT05 تختبر حدود الخطأ من التنقيح التكراري للحل المحسوب لنظام المعادلات A*X = B، حيث A هي مصفوفة هيرميتية ثلاثية القطر من الرتبة n.
!>
!> RESLTS(1) = اختبار حد الخطأ
!>           = norm(X - XACT) / ( norm(X) * FERR )
!>
!> يتم إرجاع قيمة كبيرة إذا لم تكن هذه النسبة أقل من واحد.
!>
!> RESLTS(2) = الباقي من روتين التنقيح التكراري
!>           = الحد الأقصى لـ BERR / ( NZ*EPS + (*) )، حيث
!>             (*) = NZ*UNFL / (min_i (abs(A)*abs(X) +abs(b))_i )
!>             و NZ = الحد الأقصى لعدد العناصر غير الصفرية في أي صف من A، بالإضافة إلى 1
!>

المعلمات

N 
!>          N هو عدد صحيح (INTEGER)
!>          عدد صفوف المصفوفات X، B، و XACT، ورتبة المصفوفة A.  N >= 0.
!>

NRHS

!>          NRHS هو عدد صحيح (INTEGER)
!>          عدد أعمدة المصفوفات X، B، و XACT.
!>          NRHS >= 0.
!>

D

!>          D هو مصفوفة دقة مزدوجة (DOUBLE PRECISION array)، البعد (N)
!>          العناصر القطرية n للمصفوفة ثلاثية القطر A.
!>

E

!>          E هي مصفوفة COMPLEX*16، البعد (N-1)
!>          العناصر تحت القطرية (n-1) للمصفوفة ثلاثية القطر A.
!>

B

!>          B هي مصفوفة COMPLEX*16، البعد (LDB,NRHS)
!>          متجهات الطرف الأيمن لنظام المعادلات الخطية.
!>

LDB

!>          LDB هو عدد صحيح (INTEGER)
!>          البعد الرائد للمصفوفة B.  LDB >= max(1,N).
!>

X

!>          X هي مصفوفة COMPLEX*16، البعد (LDX,NRHS)
!>          متجهات الحل المحسوبة.  يتم تخزين كل متجه كعمود في المصفوفة X.
!>

LDX

!>          LDX هو عدد صحيح (INTEGER)
!>          البعد الرائد للمصفوفة X.  LDX >= max(1,N).
!>

XACT

!>          XACT هي مصفوفة COMPLEX*16، البعد (LDX,NRHS)
!>          متجهات الحل الدقيقة.  يتم تخزين كل متجه كعمود في المصفوفة XACT.
!>

LDXACT

!>          LDXACT هو عدد صحيح (INTEGER)
!>          البعد الرائد للمصفوفة XACT.  LDXACT >= max(1,N).
!>

FERR

!>          FERR هي مصفوفة دقة مزدوجة (DOUBLE PRECISION array)، البعد (NRHS)
!>          حدود الخطأ الأمامي المقدرة لكل متجه حل X.  إذا كان XTRUE هو الحل الحقيقي،
!>          فإن FERR يحد من قيمة أكبر إدخال في (X - XTRUE) مقسومًا على أكبر قيمة
!>          في X.
!>

BERR

!>          BERR هي مصفوفة دقة مزدوجة (DOUBLE PRECISION array)، البعد (NRHS)
!>          الخطأ الخلفي النسبي المكون لكل متجه حل (أي، أصغر تغيير نسبي في أي إدخال
!>          في A أو B يجعل X حلاً دقيقًا).
!>

RESLTS

!>          RESLTS هي مصفوفة دقة مزدوجة (DOUBLE PRECISION array)، البعد (2)
!>          الحد الأقصى عبر متجهات الحل NRHS للنسب التالية:
!>          RESLTS(1) = norm(X - XACT) / ( norm(X) * FERR )
!>          RESLTS(2) = BERR / ( NZ*EPS + (*) )
!>
