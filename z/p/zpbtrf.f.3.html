subroutine zpbtrf (character uplo, integer n, integer kd, complex*16, dimension( ldab, * ) ab, integer ldab, integer info)

ZPBTRF

الغرض:

!>
!> ZPBTRF يحسب تحليل Cholesky لمصفوفة شريطية هرميتية موجبة محددة A.
!>
!> التحليل له الشكل
!>    A = U**H * U,  إذا كان UPLO = 'U', أو
!>    A = L  * L**H,  إذا كان UPLO = 'L',
!> حيث U مصفوفة ثلاثية علوية و L مصفوفة ثلاثية سفلية.
!>

المعلمات

UPLO 
!>          UPLO هو CHARACTER*1
!>          = 'U':  يتم تخزين المثلث العلوي لـ A؛
!>          = 'L':  يتم تخزين المثلث السفلي لـ A.
!>

N

!>          N هو INTEGER
!>          رتبة المصفوفة A.  N >= 0.
!>

KD

!>          KD هو INTEGER
!>          عدد الأقطار الفائقة للمصفوفة A إذا كان UPLO = 'U',
!>          أو عدد الأقطار الفرعية إذا كان UPLO = 'L'.  KD >= 0.
!>

AB

!>          AB هو مصفوفة COMPLEX*16, البعد (LDAB,N)
!>          عند الإدخال، المثلث العلوي أو السفلي للمصفوفة الشريطيّة الهرميتية
!!>          A، يتم تخزينها في أول KD+1 صفوف من المصفوفة.
!>          يتم تخزين العمود j للمصفوفة A في العمود j للمصفوفة AB
!>          كما يلي:
!>          إذا كان UPLO = 'U', AB(kd+1+i-j,j) = A(i,j) لـ max(1,j-kd)<=i<=j؛
!>          إذا كان UPLO = 'L', AB(1+i-j,j)    = A(i,j) لـ j<=i<=min(n,j+kd).
!>
!>          عند الخروج، إذا كان INFO = 0، فإن العامل الثلاثي U أو L من
!>          تحليل Cholesky A = U**H*U أو A = L*L**H للمصفوفة الشريطية
!>          A، بنفس صيغة التخزين مثل A.
!>

LDAB

!>          LDAB هو INTEGER
!>          البعد الرائد للمصفوفة AB.  LDAB >= KD+1.
!>

INFO

!>          INFO هو INTEGER
!>          = 0:  انتهاء ناجح
!>          < 0:  إذا كان INFO = -i, فإن المعلمة i كانت ذات قيمة غير قانونية
!>          > 0:  إذا كان INFO = i، فإن القطاع الرئيسي الرائد من الرتبة i
!>                ليس موجباً، ولم يكن من الممكن إكمال التحليل.
!>

!>  N = 6، KD = 2، و UPLO = 'U':
!>
!>  عند الإدخال:                       عند الخروج:
!>
!>      *    *   a13  a24  a35  a46      *    *   u13  u24  u35  u46
!>      *   a12  a23  a34  a45  a56      *   u12  u23  u34  u45  u56
!>     a11  a22  a33  a44  a55  a66     u11  u22  u33  u44  u55  u66
!>
!>  بالمثل، إذا كان UPLO = 'L' فإن صيغة A كالتالي:
!>
!>  عند الإدخال:                       عند الخروج:
!>
!>     a11  a22  a33  a44  a55  a66     l11  l22  l33  l44  l55  l66
!>     a21  a32  a43  a54  a65   *      l21  l32  l43  l54  l65   *
!>     a31  a42  a53  a64   *    *      l31  l42  l53  l64   *    *
!>
!>  العناصر المصفوفة المشار إليها بـ * لا تستخدمها الروتين.
!>
