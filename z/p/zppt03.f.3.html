subroutine zppt03 (character uplo, integer n, complex*16, dimension( * ) a, complex*16, dimension( * ) ainv, complex*16, dimension( ldwork, * ) work, integer ldwork, double precision, dimension( * ) rwork, double precision rcond, double precision resid)

ZPPT03

الغرض:

!>
!> ZPPT03 يحسب الباقي لمصفوفة معبأة متراسية مضروبة في معكوسها:
!>    norm( I - A*AINV ) / ( N * norm(A) * norm(AINV) * EPS ),
!> حيث EPS هو خطأ الآلة.
!>

المعاملات

UPLO 
!>          UPLO is CHARACTER*1
!>          يحدد ما إذا كان الجزء العلوي أو السفلي الثلاثي للمصفوفة المتراسية A
!>          مخزن:
!>          = 'U':  مثلثي علوي
!>          = 'L':  مثلثي سفلي
!>

N

!>          N is INTEGER
!>          عدد الصفوف والأعمدة للمصفوفة A.  N >= 0.
!>

A

!>          A is COMPLEX*16 array, dimension (N*(N+1)/2)
!>          المصفوفة الأصلية المتراسية A، مخزنة كمصفوفة
!>          مثلثية معبأة.
!>

AINV

!>          AINV is COMPLEX*16 array, dimension (N*(N+1)/2)
!>          المعكوس (المتراسي) للمصفوفة A، مخزن كمصفوفة
!>          مثلثية معبأة.
!>

WORK

!>          WORK is COMPLEX*16 array, dimension (LDWORK,N)
!>

LDWORK

!>          LDWORK is INTEGER
!>          البعد الرئيسي للمصفوفة WORK.  LDWORK >= max(1,N).
!>

RWORK

!>          RWORK is DOUBLE PRECISION array, dimension (N)
!>

RCOND

!>          RCOND is DOUBLE PRECISION
!>          مقلوب رقم الشرط للمصفوفة A، محسوب كـ
!>          ( 1/norm(A) ) / norm(AINV).
!>

RESID

!>          RESID is DOUBLE PRECISION
!>          norm(I - A*AINV) / ( N * norm(A) * norm(AINV) * EPS )
!>
