subroutine zposv (character uplo, integer n, integer nrhs, complex*16, dimension( lda, * ) a, integer lda, complex*16, dimension( ldb, * ) b, integer ldb, integer info)

ZPOSV تحسب حل نظام المعادلات الخطية A * X = B لمصفوفات PO

الغرض:

!>
!> ZPOSV تحسب حل نظام المعادلات الخطية المعقد A * X = B،
!> حيث A مصفوفة هرميتية موجبة محددة N×N، و X و B مصفوفتان N×NRHS.
!>
!> يستخدم تحليل Cholesky لتحليل A إلى
!>    A = U**H* U، إذا كان UPLO = 'U'، أو
!>    A = L * L**H، إذا كان UPLO = 'L'،
!> حيث U مصفوفة مثلثة عليا و L مصفوفة مثلثة سفلى. ثم يستخدم الشكل المحلل لـ A لحل نظام المعادلات A * X = B.
!>

المعلمات

UPLO 
!>          UPLO هو CHARACTER*1
!>          = 'U':  المثلث العلوي لـ A مخزن؛
!>          = 'L':  المثلث السفلي لـ A مخزن.
!>

N

!>          N هو INTEGER
!>          عدد المعادلات الخطية، أي ترتيب المصفوفة A. N >= 0.
!>

NRHS

!>          NRHS هو INTEGER
!>          عدد الأطراف اليمنى، أي عدد أعمدة المصفوفة B. NRHS >= 0.
!>

A

!>          A هي مصفوفة COMPLEX*16، البعد (LDA,N)
!>          عند الدخول، المصفوفة الهرميتية A. إذا كان UPLO = 'U'، فإن الجزء العلوي
!>          المثلثي العلوي N×N من A يحتوي على الجزء المثلثي العلوي
!>          للمصفوفة A، والجزء السفلي القطري
!>          لـ A لا تتم الإشارة إليه. إذا كان UPLO = 'L'، فإن الجزء
!>          المثلثي السفلي العلوي N×N من A يحتوي على الجزء المثلثي السفلي
!>          للمصفوفة A، والجزء العلوي القطري البحت
!>          لـ A لا تتم الإشارة إليه.
!>
!>          عند الخروج، إذا كان INFO = 0، فإن العامل U أو L من تحليل Cholesky
!>          A = U**H *U أو A = L*L**H.
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرئيسي للمصفوفة A. LDA >= max(1,N).
!>

B

!>          B هي مصفوفة COMPLEX*16، البعد (LDB,NRHS)
!>          عند الدخول، المصفوفة B ذات الطرف الأيمن N×NRHS.
!>          عند الخروج، إذا كان INFO = 0، المصفوفة X الحل N×NRHS.
!>

LDB

!>          LDB هو INTEGER
!>          البعد الرئيسي للمصفوفة B. LDB >= max(1,N).
!>

INFO

!>          INFO هو INTEGER
!>          = 0:  انتهاء ناجح
!>          < 0:  إذا كان INFO = -i، فإن المعلمة i لها قيمة غير قانونية
!>          > 0:  إذا كان INFO = i، فإن أصغر بادئة رئيسية من الرتبة i
!>                لـ A ليست موجبة، وبالتالي لا يمكن إكمال التحليل،
!>                ولم يتم حساب الحل.
!>
