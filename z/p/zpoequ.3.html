توثيق الوظيفة/الروتين الفرعي

subroutine zpoequ (integer n, complex*16, dimension( lda, * ) a, integer lda, double precision, dimension( * ) s, double precision scond, double precision amax, integer info)

ZPOEQU

الغرض:

!>
!> ZPOEQU يحسب مقاييس الصف والعمود المعدة لموازنة مصفوفة هرمية موجبة محددة A
!> وتقليل رقم حالتها (بالنسبة للمعيار الثنائي). S يحتوي على عوامل القياس،
!> S(i) = 1/sqrt(A(i,i))، المختار بحيث تكون المصفوفة المقاسة B ذات
!> العناصر B(i,j) = S(i)*A(i,j)*S(j) لها وحدات على القطر. هذا
!> الاختيار لـ S يضع رقم حالة B ضمن عامل N من
!> أصغر رقم حالة ممكن عبر جميع المقاييس القطرية الممكنة.
!>

المعاملات

N 
!>          N هو عدد صحيح (INTEGER)
!>          ترتيب المصفوفة A. N >= 0.
!>

A

!>          A هو مصفوفة COMPLEX*16، البعد (LDA,N)
!>          المصفوفة الهرمية الموجبة المحددة N×N التي سيتم حساب عوامل قياسها.
!>          يتم الرجوع فقط إلى العناصر القطرية لـ A.
!>

LDA

!>          LDA هو عدد صحيح (INTEGER)
!>          البعد الرائد للمصفوفة A. LDA >= max(1,N).
!>

S

!>          S هي مصفوفة DOUBLE PRECISION، البعد (N)
!>          إذا كان INFO = 0، فإن S تحتوي على عوامل القياس لـ A.
!>

SCOND

!>          SCOND هو DOUBLE PRECISION
!>          إذا كان INFO = 0، فإن S تحتوي على نسبة أصغر S(i) إلى
!>          أكبر S(i). إذا كان SCOND >= 0.1 وكان AMAX ليس كبيرًا جدًا
!>          أو صغيرًا جدًا، فلا يستحق القياس بواسطة S.
!>

AMAX

!>          AMAX هو DOUBLE PRECISION
!>          القيمة المطلقة لأكبر عنصر في المصفوفة. إذا كان AMAX قريبًا جدًا
!>          من الفائض أو قريبًا جدًا من النقصان، فيجب قياس المصفوفة.
!>

INFO

!>          INFO هو عدد صحيح (INTEGER)
!>          = 0:  انتهاء ناجح
!>          < 0:  إذا كان INFO = -i، فإن المعامل i كان له قيمة غير قانونية
!>          > 0:  إذا كان INFO = i، فإن العنصر القطري i غير موجب.
!>
