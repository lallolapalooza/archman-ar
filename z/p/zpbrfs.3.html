subroutine zpbrfs (character uplo, integer n, integer kd, integer nrhs, complex*16, dimension( ldab, * ) ab, integer ldab, complex*16, dimension( ldafb, * ) afb, integer ldafb, complex*16, dimension( ldb, * ) b, integer ldb, complex*16, dimension( ldx, * ) x, integer ldx, double precision, dimension( * ) ferr, double precision, dimension( * ) berr, complex*16, dimension( * ) work, double precision, dimension( * ) rwork, integer info)

ZPBRFS

الغرض:

!>
!> ZPBRFS يحسن الحل المحسوب لنظام المعادلات الخطية
!> عندما تكون مصفوفة المعاملات هيرميتية موجبة محددة
!> وذات نطاق، ويوفر حدود خطأ وتقديرات خطأ خلفية
!> للحل.
!>

المعلمات

UPLO 
!>          UPLO is CHARACTER*1
!>          = 'U':  الجزء العلوي من A مخزن؛
!>          = 'L':  الجزء السفلي من A مخزن.
!>

N

!>          N is INTEGER
!>          رتبة المصفوفة A. N >= 0.
!>

KD

!>          KD is INTEGER
!>          عدد الأقطار الفائقة للمصفوفة A إذا كان UPLO = 'U'،
!>          أو عدد الأقطار الفرعية إذا كان UPLO = 'L'. KD >= 0.
!>

NRHS

!>          NRHS is INTEGER
!>          عدد الأطراف اليمنى، أي عدد أعمدة
!>          المصفوفتين B و X. NRHS >= 0.
!>

AB

!>          AB is COMPLEX*16 array, dimension (LDAB,N)
!>          الجزء العلوي أو السفلي من المصفوفة الهيرميتية ذات النطاق A،
!>          مخزن في الصفوف KD+1 الأولى من المصفوفة. العمود j
!>          من A مخزن في العمود j من المصفوفة AB على النحو التالي:
!>          إذا كان UPLO = 'U'، فإن AB(kd+1+i-j,j) = A(i,j) عندما يكون max(1,j-kd)<=i<=j؛
!>          إذا كان UPLO = 'L'، فإن AB(1+i-j,j) = A(i,j) عندما يكون j<=i<=min(n,j+kd).
!>

LDAB

!>          LDAB is INTEGER
!>          البعد الرائد للمصفوفة AB. LDAB >= KD+1.
!>

AFB

!>          AFB is COMPLEX*16 array, dimension (LDAFB,N)
!>          العامل المثلثي U أو L من تحليل Cholesky
!>          A = U**H*U أو A = L*L**H للمصفوفة ذات النطاق A كما تم حسابه بواسطة
!>          ZPBTRF، بنفس تنسيق التخزين مثل A (انظر AB).
!>

LDAFB

!>          LDAFB is INTEGER
!>          البعد الرائد للمصفوفة AFB. LDAFB >= KD+1.
!>

B

!>          B is COMPLEX*16 array, dimension (LDB,NRHS)
!>          مصفوفة الطرف الأيمن B.
!>

LDB

!>          LDB is INTEGER
!>          البعد الرائد للمصفوفة B. LDB >= max(1,N).
!>

X

!>          X is COMPLEX*16 array, dimension (LDX,NRHS)
!>          عند الدخول، مصفوفة الحل X، كما تم حسابها بواسطة ZPBTRS.
!>          عند الخروج، مصفوفة الحل المحسنة X.
!>

LDX

!>          LDX is INTEGER
!>          البعد الرائد للمصفوفة X. LDX >= max(1,N).
!>

FERR

!>          FERR is DOUBLE PRECISION array, dimension (NRHS)
!>          حد الخطأ الأمامي المقدر لكل متجه حل
!>          X(j) (العمود j من مصفوفة الحل X).
!>          إذا كان XTRUE هو الحل الحقيقي المقابل لـ X(j)، فإن FERR(j)
!>          هو حد علوي مقدر لحجم أكبر
!>          عنصر في (X(j) - XTRUE) مقسومًا على حجم
!>          أكبر عنصر في X(j). التقدير موثوق به مثل
!>          التقدير لـ RCOND، وهو دائمًا تقريبًا زيادة طفيفة
!>          في الخطأ الحقيقي.
!>

BERR

!>          BERR is DOUBLE PRECISION array, dimension (NRHS)
!>          الخطأ الخلفي النسبي المكوني لكل متجه حل
!>          X(j) (أي، أصغر تغيير نسبي في
!>          أي عنصر من A أو B يجعل X(j) حلاً دقيقًا).
!>

WORK

!>          WORK is COMPLEX*16 array, dimension (2*N)
!>

RWORK

!>          RWORK is DOUBLE PRECISION array, dimension (N)
!>

INFO

!>          INFO is INTEGER
!>          = 0:  انتهاء ناجح
!>          < 0:  إذا كان INFO = -i، فإن الوسيط i كان له قيمة غير قانونية
!>
