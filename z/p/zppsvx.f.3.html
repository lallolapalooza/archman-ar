subroutine zppsvx (character fact, character uplo, integer n, integer nrhs, complex*16, dimension( * ) ap, complex*16, dimension( * ) afp, character equed, double precision, dimension( * ) s, complex*16, dimension( ldb, * ) b, integer ldb, complex*16, dimension( ldx, * ) x, integer ldx, double precision rcond, double precision, dimension( * ) ferr, double precision, dimension( * ) berr, complex*16, dimension( * ) work, double precision, dimension( * ) rwork, integer info)

ZPPSVX يحسب حل نظام المعادلات الخطية A * X = B للمصفوفات الأخرى

الغرض:

!>
!> ZPPSVX يستخدم تحليل Cholesky A = U**H * U أو A = L * L**H لحساب حل نظام
!> المعادلات الخطية المعقدة
!>    A * X = B,
!> حيث A هي مصفوفة N×N هرميتية موجبة محددة مخزنة بتنسيق مضغوط، و X و B
!> هما مصفوفتان N×NRHS.
!>
!> يتم أيضًا توفير حدود الخطأ للحل وتقدير للحالة.
!>

الوصف:

!>
!> يتم تنفيذ الخطوات التالية:
!>
!> 1. إذا كانت FACT = 'E'، يتم حساب عوامل قياس حقيقية لتحقيق التوازن
!>    للنظام:
!>       diag(S) * A * diag(S) * inv(diag(S)) * X = diag(S) * B
!>    يعتمد ما إذا كان سيتم تحقيق توازن النظام أم لا على قياس المصفوفة A،
!>    ولكن إذا تم استخدام تحقيق التوازن، فإن A يتم استبدالها بـ diag(S)*A*diag(S)
!>    و B بـ diag(S)*B.
!>
!> 2. إذا كانت FACT = 'N' أو 'E'، يتم استخدام تحليل Cholesky لتحليل
!>    المصفوفة A (بعد تحقيق التوازن إذا كانت FACT = 'E') كـ
!>       A = U**H * U ، إذا كان UPLO = 'U'، أو
!>       A = L * L**H ، إذا كان UPLO = 'L'،
!>    حيث U مصفوفة مثلثة عليا، L مصفوفة مثلثة سفلى، و **H تشير إلى المترافقة
!>    المنقولة.
!>
!> 3. إذا لم يكن المحدد الرئيسي الرائد من الرتبة i موجبًا، فإن الروتين
!>    يعود بـ INFO = i. بخلاف ذلك، يتم استخدام الشكل المحلل لـ A لتقدير
!>    رقم حالة المصفوفة A. إذا كان مقلوب رقم الحالة أقل من دقة الآلة،
!>    يتم إرجاع INFO = N+1 كتحذير، ولكن الروتين لا يزال يستمر لحساب X
!>    وحساب حدود الخطأ كما هو موضح أدناه.
!>
!> 4. يتم حل نظام المعادلات لـ X باستخدام الشكل المحلل لـ A.
!>
!> 5. يتم تطبيق تحسين تكراري لتحسين مصفوفة الحل المحسوبة وحساب حدود
!>    الخطأ وتقديرات الخطأ الخلفي لها.
!>
!> 6. إذا تم استخدام تحقيق التوازن، يتم ضرب المصفوفة X مسبقًا بـ diag(S)
!>    بحيث تحل النظام الأصلي قبل تحقيق التوازن.
!>

المعلمات

FACT 
!>          FACT هو CHARACTER*1
!>          يحدد ما إذا كان الشكل المحلل للمصفوفة A موجودًا في المدخل، وإذا لم يكن
!>          كذلك، فما إذا كان يجب تحقيق توازن المصفوفة A قبل تحليلها.
!>          = 'F':  في المدخل، يحتوي AFP على الشكل المحلل لـ A.
!>                  إذا كان EQUED = 'Y'، فقد تم تحقيق توازن المصفوفة A
!>                  بعوامل قياس معطاة بواسطة S. لن يتم تعديل AP و AFP.
!>          = 'N':  سيتم نسخ المصفوفة A إلى AFP وتحليلها.
!>          = 'E':  سيتم تحقيق توازن المصفوفة A إذا لزم الأمر، ثم نسخها
!>                  إلى AFP وتحليلها.
!>

UPLO

!>          UPLO هو CHARACTER*1
!>          = 'U':  يتم تخزين المثلث العلوي لـ A؛
!>          = 'L':  يتم تخزين المثلث السفلي لـ A.
!>

N

!>          N هو INTEGER
!>          عدد المعادلات الخطية، أي رتبة المصفوفة A. N >= 0.
!>

NRHS

!>          NRHS هو INTEGER
!>          عدد الأطراف اليمنى، أي عدد أعمدة المصفوفتين B و X. NRHS >= 0.
!>

AP

!>          AP هو COMPLEX*16 array, dimension (N*(N+1)/2)
!>          في المدخل، المثلث العلوي أو السفلي للمصفوفة الهرميتية
!>          A، مضغوط عمودياً في مصفوفة خطية، باستثناء إذا كان FACT = 'F'
!>          و EQUED = 'Y'، فإن A يجب أن يحتوي على المصفوفة المتوازنة
!>          diag(S)*A*diag(S). يتم تخزين العمود j من A في
!>          المصفوفة AP كالتالي:
!>          إذا كان UPLO = 'U'، AP(i + (j-1)*j/2) = A(i,j) لـ 1<=i<=j؛
!>          إذا كان UPLO = 'L'، AP(i + (j-1)*(2n-j)/2) = A(i,j) لـ j<=i<=n.
!>          انظر أدناه لمزيد من التفاصيل. لا يتم تعديل A إذا كان
!>          FACT = 'F' أو 'N'، أو إذا كان FACT = 'E' و EQUED = 'N' عند الخروج.
!>
!>          عند الخروج، إذا كان FACT = 'E' و EQUED = 'Y'، يتم استبدال A بـ
!>          diag(S)*A*diag(S).
!>

AFP

!>          AFP هو COMPLEX*16 array, dimension (N*(N+1)/2)
!>          إذا كان FACT = 'F'، فإن AFP هو وسيط مدخل ويحتوي عند المدخل
!>          على المثلث المحلل U أو L من تحليل Cholesky
!>          A = U**H*U أو A = L*L**H، بنفس تنسيق التخزين لـ A.
!>          إذا كان EQUED .ne. 'N'، فإن AFP هو الشكل المحلل للمصفوفة المتوازنة A.
!>
!>          إذا كان FACT = 'N'، فإن AFP هو وسيط مخرج ويعود عند الخروج
!>          بالمثلث المحلل U أو L من تحليل Cholesky
!>          A = U**H * U أو A = L * L**H للمصفوفة الأصلية A.
!>
!>          إذا كان FACT = 'E'، فإن AFP هو وسيط مخرج ويعود عند الخروج
!>          بالمثلث المحلل U أو L من تحليل Cholesky
!>          A = U**H * U أو A = L * L**H للمصفوفة المتوازنة A (انظر وصف AP
!>          لشكل المصفوفة المتوازنة).
!>

EQUED

!>          EQUED هو CHARACTER*1
!>          يحدد شكل تحقيق التوازن الذي تم.
!>          = 'N':  لم يتم تحقيق توازن (صحيح دائمًا إذا كان FACT = 'N').
!>          = 'Y':  تم تحقيق توازن، أي تم استبدال A بـ
!>                  diag(S) * A * diag(S).
!>          EQUED هو وسيط مدخل إذا كان FACT = 'F'؛ بخلاف ذلك، فهو وسيط
!>          مخرج.
!>

S

!>          S هو DOUBLE PRECISION array, dimension (N)
!>          عوامل القياس لـ A؛ لا يتم الوصول إليها إذا كان EQUED = 'N'. S هو
!>          وسيط مدخل إذا كان FACT = 'F'؛ بخلاف ذلك، S هو وسيط
!>          مخرج. إذا كان FACT = 'F' و EQUED = 'Y'، يجب أن يكون كل عنصر من S
!>          موجبًا.
!>

B

!>          B هو COMPLEX*16 array, dimension (LDB,NRHS)
!>          في المدخل، مصفوفة الطرف الأيمن B ذات الأبعاد N×NRHS.
!>          عند الخروج، إذا كان EQUED = 'N'، لا يتم تعديل B؛ إذا كان EQUED = 'Y'،
!!>          يتم استبدال B بـ diag(S) * B.
!>

LDB

!>          LDB هو INTEGER
!>          البعد الرائد للمصفوفة B. LDB >= max(1,N).
!>

X

!>          X هو COMPLEX*16 array, dimension (LDX,NRHS)
!>          إذا كان INFO = 0 أو INFO = N+1، فإن مصفوفة الحل X ذات الأبعاد N×NRHS
!>          لنظام المعادلات الأصلي. لاحظ أنه إذا كان EQUED = 'Y'،
!>          يتم تعديل A و B عند الخروج، والحل للنظام المتوازن هو inv(diag(S))*X.
!>

LDX

!>          LDX هو INTEGER
!>          البعد الرائد للمصفوفة X. LDX >= max(1,N).
!>

RCOND

!>          RCOND هو DOUBLE PRECISION
!>          تقدير معامل الح
