توثيق الوظيفة/الروتين الفرعي

subroutine zpttrf (integer n, double precision, dimension( * ) d, complex*16, dimension( * ) e, integer info)

ZPTTRF

الغرض:

!>
!> ZPTTRF يحسب تفكيك L*D*L**H لمصفوفة هيرميتية معرفة موجبة ثلاثية الأقطار A. قد يُعتبر التفكيك أيضًا على شكل A = U**H *D*U.
!>

المعلمات

N 
!>          N هو عدد صحيح (INTEGER)
!>          رتبة المصفوفة A.  N >= 0.
!>

D

!>          D هي مصفوفة من النوع DOUBLE PRECISION، بأبعاد (N)
!>          عند الدخول، العناصر القطرية n للمصفوفة ثلاثية الأقطار A. عند الخروج، العناصر القطرية n للمصفوفة القطرية D من تفكيك L*D*L**H للمصفوفة A.
!>

E

!>          E هي مصفوفة من النوع COMPLEX*16، بأبعاد (N-1)
!>          عند الدخول، عناصر ما دون القطر n-1 للمصفوفة ثلاثية الأقطار A. عند الخروج، عناصر ما دون القطر n-1 للعامل شبه القطري الوحدي L من تفكيك L*D*L**H للمصفوفة A.
!>          يمكن اعتبار E أيضًا كقطر فوقي للعامل شبه القطري الوحدي U من تفكيك U**H *D*U للمصفوفة A.
!>

INFO

!>          INFO هو عدد صحيح (INTEGER)
!>          = 0: خروج ناجح
!>          < 0: إذا كان INFO = -k، فإن المعامل k كان له قيمة غير قانونية
!>          > 0: إذا كان INFO = k، فإن المحدد الرئيسي الرائد من الرتبة k ليس موجبًا؛ إذا كان k < N، لم يكن من الممكن إكمال التفكيك، بينما إذا كان k = N، فقد تم إكمال التفكيك، ولكن D(N) <= 0.
!>
