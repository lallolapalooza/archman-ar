وثائق الوظيفة/الإجراء الفرعي

subroutine zpftrf (character transr, character uplo, integer n, complex*16, dimension( 0: * ) a, integer info)

ZPFTRF

الغرض:

!>
!> ZPFTRF تحسب التحلل Cholesky لمصفوفة A هرميتية موجبة محددة معقدة.
!>
!> التحليل له الشكل
!>    A = U**H * U،  إذا UPLO = 'U'، أو
!>    A = L  * L**H،  إذا UPLO = 'L'،
!> حيث U مصفوفة مثلثية عليا و L مصفوفة مثلثية سفلى.
!>
!> هذه هي النسخة القطاعية من الخوارزمية، وتستدعي BLAS المستوى 3.
!>

المعلمات

TRANSR 
!>          TRANSR هو CHARACTER*1
!>          = 'N':  يتم تخزين Normal TRANSR للمصفوفة RFP A؛
!>          = 'C':  يتم تخزين Conjugate-transpose TRANSR للمصفوفة RFP A.
!>

UPLO

!>          UPLO هو CHARACTER*1
!>          = 'U':  يتم تخزين المثلث العلوي للمصفوفة RFP A؛
!>          = 'L':  يتم تخزين المثلث السفلي للمصفوفة RFP A.
!>

N

!>          N هو INTEGER
!>          رتبة المصفوفة A.  N >= 0.
!>

A

!>          A هي مصفوفة COMPLEX*16، أبعادها ( N*(N+1)/2 )؛
!>          عند الدخول، المصفوفة الهرميتية A بتنسيق RFP. يتم وصف تنسيق RFP
!>          بـ TRANSR، UPLO، و N كما يلي: إذا كان TRANSR = 'N'
!>          فإن RFP A هي (0:N,0:k-1) عندما N زوجي؛ k=N/2. RFP A هي
!>          (0:N-1,0:k) عندما N فردي؛ k=N/2. إذا كان TRANSR = 'C' فإن RFP هي
!>          المصفوفة المنقولة المترافقة (Conjugate-transpose) لـ RFP A كما هي معرفة عندما
!>          يكون TRANSR = 'N'. يتم تعريف محتويات RFP A بواسطة UPLO كما يلي:
!>          إذا كان UPLO = 'U' فإن RFP A تحتوي على عناصر nt للجزء العلوي المضغوط A.
!>          إذا كان UPLO = 'L' فإن RFP A تحتوي على عناصر الجزء السفلي المضغوط A.
!>          LDA للمصفوفة RFP A هي (N+1)/2 عندما يكون TRANSR = 'C'.
!>          عندما يكون TRANSR هو 'N' فإن LDA هو N+1 عندما يكون N زوجيًا و N
!>          يكون فرديًا. انظر الملاحظة أدناه لمزيد من التفاصيل.
!>
!>          عند الخروج، إذا كان INFO = 0، فإن العامل U أو L من تحليل Cholesky
!>          RFP A = U**H*U أو RFP A = L*L**H.
!>

INFO

!>          INFO هو INTEGER
!>          = 0:  الخروج بنجاح
!>          < 0:  إذا كان INFO = -i، فإن المعامل i كان له قيمة غير قانونية
!>          > 0:  إذا كان INFO = i، فإن الجزء الرئيسي المحدد من الرتبة i
!>                ليس موجبًا، ولم يكن من الممكن إكمال التحليل.
!>
!>  ملاحظات إضافية حول تنسيق RFP:
!>  ============================
!>
!>  ننظر أولاً إلى التنسيق القياسي المضغوط عندما يكون N زوجيًا.
!>  نعطي مثالاً حيث N = 6.
!>
!>     AP هو العلوي             AP هو السفلي
!>
!>   00 01 02 03 04 05       00
!>      11 12 13 14 15       10 11
!>         22 23 24 25       20 21 22
!>            33 34 35       30 31 32 33
!>               44 45       40 41 42 43 44
!>                  55       50 51 52 53 54 55
!>
!>  إذا كان TRANSR = 'N'. تحتفظ RFP بـ AP كما يلي:
!>  بالنسبة لـ UPLO = 'U' فإن شبه المنحرف العلوي A(0:5,0:2) يتكون من
!>  الأعمدة الثلاثة الأخيرة من AP العلوي. المثلث السفلي A(4:6,0:2) يتكون من
!>  المصفوفة المنقولة المترافقة للأعمدة الثلاثة الأولى من AP العلوي.
!>  بالنسبة لـ UPLO = 'L' فإن شبه المنحرف السفلي A(1:6,0:2) يتكون من
!>  الأعمدة الثلاثة الأولى من AP السفلي. المثلث العلوي A(0:2,0:2) يتكون من
!>  المصفوفة المنقولة المترافقة للأعمدة الثلاثة الأخيرة من AP السفلي.
!>  للإشارة إلى الترافق نضع -- فوق العنصر. هذا يغطي
!>  الحالة N زوجي و TRANSR = 'N'.
!>
!>         RFP A                   RFP A
!>
!>                                -- -- --
!>        03 04 05                33 43 53
!>                                   -- --
!>        13 14 15                00 44 54
!>                                      --
!>        23 24 25                10 11 55
!>
!>        33 34 35                20 21 22
!>        --
!>        00 44 45                30 31 32
!>        -- --
!>        01 11 55                40 41 42
!>        -- -- --
!>        02 12 22                50 51 52
!>
!>  الآن إذا كان TRANSR = 'C'. RFP A في كلتا الحالتين UPLO هي ببساطة
!>  المصفوفة المنقولة المترافقة لـ RFP A أعلاه. وبالتالي نحصل على:
!>
!>           RFP A                   RFP A
!>
!>     -- -- -- --                -- -- -- -- -- --
!>     03 13 23 33 00 01 02    33 00 10 20 30 40 50
!>     -- -- -- -- --                -- -- -- -- --
!>     04 14 24 34 44 11 12    43 44 11 21 31 41 51
!>     -- -- -- -- -- --                -- -- -- --
!>     05 15 25 35 45 55 22    53 54 55 22 32 42 52
!>
!>  ننظر بعد ذلك إلى التنسيق القياسي المضغوط عندما يكون N فرديًا.
!>  نعطي مثالاً حيث N = 5.
!>
!>     AP هو العلوي                 AP هو السفلي
!>
!>   00 01 02 03 04              00
!>      11 12 13 14              10 11
!>         22 23 24              20 21 22
!>            33 34              30 31 32 33
!>               44              40 41 42 43 44
!>
!>  إذا كان TRANSR = 'N'. تحتفظ RFP بـ AP كما يلي:
!>  بالنسبة لـ UPLO = 'U' فإن شبه المنحرف العلوي A(0:4,0:2) يتكون من
!>  الأعمدة الثلاثة الأخيرة من AP العلوي. المثلث السفلي A(3:4,0:1) يتكون من
!>  المصفوفة المنقولة المترافقة للعمودين الأولين من AP العلوي.
!>  بالنسبة لـ UPLO = 'L' فإن شبه المنحرف السفلي A(0:4,0:2) يتكون من
!>  الأعمدة الثلاثة الأولى من AP السفلي. المثلث العلوي A(0:1,1:2) يتكون من
!>  المصفوفة المنقولة المترافقة للعمودين الأخيرين من AP السفلي.
!>  للإشارة إلى الترافق نضع -- فوق العنصر. هذا يغطي
!>  الحالة N فردي و TRANSR = 'N'.
!>
!>         RFP A                   RFP A
!>
!>                                   -- --
!>        02 03 04                00 33 43
!>                                      --
!>        12 13 14                10 11 44
!>
!>        22 23 24                20 21 22
!>        --
!>        00 33 34                30 31 32
!>        -- --
!>        01 11 44                40 41 42
!>
!>  الآن إذا كان TRANSR = 'C'. RFP A في كلتا الحالتين UPLO هي ببساطة
!>  المصفوفة المنقولة المترافقة لـ RFP A أعلاه. وبالتالي نحصل على:
!>
!>           RFP A                   RFP A
!>
!>     -- -- --                   -- -- -- -- -- --
!>     02 12 22 00 01             00 10 20 30 40 50
!>     -- -- -- --                   -- -- -- -- --
!>     03 13 23 33 11             33 11 21 31 41 51
!>     -- -- -- -- --                   -- -- -- --
!>     04 14 24 34 44             43 44 22 32 42 52
!>
