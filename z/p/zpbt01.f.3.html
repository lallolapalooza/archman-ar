subroutine zpbt01 (character uplo, integer n, integer kd, complex*16, dimension( lda, * ) a, integer lda, complex*16, dimension( ldafac, * ) afac, integer ldafac, double precision, dimension( * ) rwork, double precision resid)

ZPBT01

الغرض:

!>
!> ZPBT01 يعيد بناء مصفوفة هيرميتية موجبة محددة A من
!> تحليلها L*L' أو U'*U ويحسب الباقي
!>    norm( L*L' - A ) / ( N * norm(A) * EPS ) أو
!>    norm( U'*U - A ) / ( N * norm(A) * EPS ),
!> حيث EPS هو الخطأ النسبي للجهاز، L' هو منقول A المرافق لـ L،
!> و U' هو منقول A المرافق لـ U.
!>

المعلمات

UPLO 
!>          UPLO is CHARACTER*1
!>          يحدد ما إذا كان الجزء المثلثي العلوي أو السفلي للمصفوفة
!>          الهيرميتية A مخزنًا:
!>          = 'U':  مثلثي علوي
!>          = 'L':  مثلثي سفلي
!>

N

!>          N is INTEGER
!>          عدد الصفوف والأعمدة للمصفوفة A.  N >= 0.
!>

KD

!>          KD is INTEGER
!>          عدد الأقطار الفائقة للمصفوفة A إذا كان UPLO = 'U'،
!>          أو عدد الأقطار الفرعية إذا كان UPLO = 'L'.  KD >= 0.
!>

A

!>          A is COMPLEX*16 array, dimension (LDA,N)
!>          المصفوفة الهيرميتية الأصلية A.  إذا كان UPLO = 'U'،
!>          يتم تخزين الجزء المثلثي العلوي من A كمصفوفة نطاق؛ إذا كان
!>          UPLO = 'L'، يتم تخزين الجزء المثلثي السفلي من A.
!>          يتم تخزين أعمدة المثلث المناسب في أعمدة
!>          A وتخزين أقطار المثلث في صفوف
!>          A.  راجع ZPBTRF لمزيد من التفاصيل.
!>

LDA

!>          LDA is INTEGER.
!>          البعد الرائد للمصفوفة A.  LDA >= max(1,KD+1).
!>

AFAC

!>          AFAC is COMPLEX*16 array, dimension (LDAFAC,N)
!>          الشكل المحلل للمصفوفة A.  AFAC يحتوي على العامل
!>          L أو U من تحليل L*L' أو U'*U بتنسيق تخزين النطاق،
!>          كما تم حسابه بواسطة ZPBTRF.
!>

LDAFAC

!>          LDAFAC is INTEGER
!>          البعد الرائد للمصفوفة AFAC.
!>          LDAFAC >= max(1,KD+1).
!>

RWORK

!>          RWORK is DOUBLE PRECISION array, dimension (N)
!>

RESID

!>          RESID is DOUBLE PRECISION
!>          إذا كان UPLO = 'L'، norm(L*L' - A) / ( N * norm(A) * EPS )
!>          إذا كان UPLO = 'U'، norm(U'*U - A) / ( N * norm(A) * EPS )
!>
