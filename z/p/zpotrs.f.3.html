توثيق الوظيفة/الروتين الفرعي

subroutine zpotrs (character uplo, integer n, integer nrhs, complex*16, dimension( lda, * ) a, integer lda, complex*16, dimension( ldb, * ) b, integer ldb, integer info)

ZPOTRS

الغرض:

!>
!> ZPOTRS يحل نظام المعادلات الخطية A*X = B باستخدام التحلل Cholesky
!> A = U**H * U أو A = L * L**H الذي تم حسابه بواسطة ZPOTRF.
!>

المعلمات

UPLO 
!>          UPLO هو CHARACTER*1
!>          = 'U':  يتم تخزين المثلث العلوي من A؛
!>          = 'L':  يتم تخزين المثلث السفلي من A.
!>

N

!>          N هو INTEGER
!>          رتبة المصفوفة A.  N >= 0.
!>

NRHS

!>          NRHS هو INTEGER
!>          عدد الأيادي اليمنى، أي عدد أعمدة
!>          المصفوفة B.  NRHS >= 0.
!>

A

!>          A هو COMPLEX*16 array, dimension (LDA,N)
!>          العامل المثلثي U أو L من تحلل Cholesky
!>          A = U**H * U أو A = L * L**H، كما تم حسابه بواسطة ZPOTRF.
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرائد للمصفوفة A.  LDA >= max(1,N).
!>

B

!>          B هو COMPLEX*16 array, dimension (LDB,NRHS)
!>          عند الدخول، مصفوفة الأيادي اليمنى B.
!>          عند الخروج، مصفوفة الحل X.
!>

LDB

!>          LDB هو INTEGER
!>          البعد الرائد للمصفوفة B.  LDB >= max(1,N).
!>

INFO

!>          INFO هو INTEGER
!>          = 0:  الانتهاء بنجاح
!>          < 0:  إذا كان INFO = -i، فإن المعامل i كان له قيمة غير قانونية
!>
