subroutine zpftrs (character transr, character uplo, integer n, integer nrhs, complex*16, dimension( 0: * ) a, complex*16, dimension( ldb, * ) b, integer ldb, integer info)

ZPFTRS

الغرض:

!>
!> ZPFTRS يحل نظام المعادلات الخطية A*X = B مع مصفوفة هرميتية موجبة محددة A
!> باستخدام تحليل Cholesky A = U**H*U أو A = L*L**H المحسوب بواسطة ZPFTRF.
!>

المعلمات

TRANSR 
!>          TRANSR هو CHARACTER*1
!>          = 'N':  يتم تخزين تنسيق RFP A العادي؛
!>          = 'C':  يتم تخزين تنسيق RFP A منقول بالمرافق.
!>

UPLO

!>          UPLO هو CHARACTER*1
!>          = 'U':  يتم تخزين المثلث العلوي لـ RFP A؛
!>          = 'L':  يتم تخزين المثلث السفلي لـ RFP A.
!>

N

!>          N هو INTEGER
!>          رتبة المصفوفة A. N >= 0.
!>

NRHS

!>          NRHS هو INTEGER
!>          عدد الأطراف اليمنى، أي عدد أعمدة المصفوفة B. NRHS >= 0.
!>

A

!>          A هو مصفوفة COMPLEX*16، أبعادها ( N*(N+1)/2 )؛
!>          العامل المثلثي U أو L من تحليل Cholesky لـ RFP A = U**H*U أو RFP A = L*L**H،
!>          كما تم حسابه بواسطة ZPFTRF.
!>          انظر الملاحظة أدناه لمزيد من التفاصيل حول RFP A.
!>

B

!>          B هي مصفوفة COMPLEX*16، أبعادها (LDB,NRHS)
!>          عند الإدخال، مصفوفة الطرف الأيمن B.
!>          عند الخروج، مصفوفة الحل X.
!>

LDB

!>          LDB هو INTEGER
!>          البعد الرائد للمصفوفة B. LDB >= max(1,N).
!>

INFO

!>          INFO هو INTEGER
!>          = 0:  انتهاء ناجح
!>          < 0:  إذا كان INFO = -i، فإن المعلمة i-th كانت لها قيمة غير قانونية
!>

!>  من AP العلوي. يتكون المثلث السفلي A(4:6,0:2) من
!>  منقول بالمرافق للأعمدة الثلاثة الأولى من AP العلوي.
!>  بالنسبة لـ UPLO = 'L'، يتكون شبه المنحرف السفلي A(1:6,0:2) من الأعمدة الثلاثة الأولى
