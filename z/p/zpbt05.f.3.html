subroutine zpbt05 (character uplo, integer n, integer kd, integer nrhs, complex*16, dimension( ldab, * ) ab, integer ldab, complex*16, dimension( ldb, * ) b, integer ldb, complex*16, dimension( ldx, * ) x, integer ldx, complex*16, dimension( ldxact, * ) xact, integer ldxact, double precision, dimension( * ) ferr, double precision, dimension( * ) berr, double precision, dimension( * ) reslts)

ZPBT05

الغرض:

!>
!> ZPBT05 تختبر حدود الخطأ من التحسين التكراري للحل المحسوب لنظام
!> المعادلات A*X = B، حيث A هي مصفوفة نطاق هرميتية.
!>
!> RESLTS(1) = اختبار حد الخطأ
!>           = norm(X - XACT) / ( norm(X) * FERR )
!>
!> يتم إرجاع قيمة كبيرة إذا لم تكن هذه النسبة أقل من واحد.
!>
!> RESLTS(2) = الخطأ المتبقي من روتين التحسين التكراري
!>           = الحد الأقصى لـ BERR / ( NZ*EPS + (*) )، حيث
!>             (*) = NZ*UNFL / (min_i (abs(A)*abs(X) +abs(b))_i )
!>             و NZ = الحد الأقصى لعدد العناصر غير الصفرية في أي صف من A، بالإضافة إلى 1
!>

المعاملات

UPLO 
!>          UPLO هو CHARACTER*1
!>          يحدد ما إذا كان الجزء العلوي أو السفلي المثلثي للمصفوفة A
!>          الهرميتية مخزنًا.
!>          = 'U':  مثلثي علوي
!>          = 'L':  مثلثي سفلي
!>

N

!>          N هو INTEGER
!>          عدد الصفوف في المصفوفات X، B، و XACT، ورتبة المصفوفة A. N >= 0.
!>

KD

!>          KD هو INTEGER
!>          عدد الأقطار الفائقة للمصفوفة A إذا كان UPLO = 'U'،
!>          أو عدد الأقطار الفرعية إذا كان UPLO = 'L'. KD >= 0.
!>

NRHS

!>          NRHS هو INTEGER
!>          عدد الأعمدة في المصفوفات X، B، و XACT. NRHS >= 0.
!>

AB

!>          AB هو مصفوفة COMPLEX*16، الأبعاد (LDAB,N)
!>          الجزء العلوي أو السفلي من مصفوفة النطاق الهرميتية A،
!>          مخزن في الصفوف KD+1 الأولى من المصفوفة. يتم تخزين العمود j
!>          من A في العمود j من المصفوفة AB كما يلي:
!>          إذا كان UPLO = 'U'، AB(kd+1+i-j,j) = A(i,j) لـ max(1,j-kd)<=i<=j؛
!>          إذا كان UPLO = 'L'، AB(1+i-j,j)    = A(i,j) لـ j<=i<=min(n,j+kd).
!>

LDAB

!>          LDAB هو INTEGER
!>          البعد الرئيسي للمصفوفة AB. LDAB >= KD+1.
!>

B

!>          B هي مصفوفة COMPLEX*16، الأبعاد (LDB,NRHS)
!>          متجهات الطرف الأيمن لأنظمة المعادلات الخطية.
!>

LDB

!>          LDB هو INTEGER
!>          البعد الرئيسي للمصفوفة B. LDB >= max(1,N).
!>

X

!>          X هي مصفوفة COMPLEX*16، الأبعاد (LDX,NRHS)
!>          متجهات الحلول المحسوبة. يتم تخزين كل متجه كعمود في المصفوفة X.
!>

LDX

!>          LDX هو INTEGER
!>          البعد الرئيسي للمصفوفة X. LDX >= max(1,N).
!>

XACT

!>          XACT هي مصفوفة COMPLEX*16، الأبعاد (LDX,NRHS)
!>          متجهات الحلول الدقيقة. يتم تخزين كل متجه كعمود في المصفوفة XACT.
!>

LDXACT

!>          LDXACT هو INTEGER
!>          البعد الرئيسي للمصفوفة XACT. LDXACT >= max(1,N).
!>

FERR

!>          FERR هي مصفوفة DOUBLE PRECISION، الأبعاد (NRHS)
!>          تقديرات حدود الخطأ الأمامي لكل متجه حل X. إذا كانت XTRUE هي الحل الحقيقي،
!>          فإن FERR يحد من حجم أكبر عنصر في (X - XTRUE) مقسومًا على حجم أكبر عنصر في X.
!>

BERR

!>          BERR هي مصفوفة DOUBLE PRECISION، الأبعاد (NRHS)
!>          الخطأ الخلفي النسبي المكون من كل متجه حل (أي، أصغر تغيير نسبي في أي عنصر
!>          في A أو B يجعل X حلاً دقيقًا).
!>

RESLTS

!>          RESLTS هي مصفوفة DOUBLE PRECISION، الأبعاد (2)
!>          الحد الأقصى على متجهات الحل NRHS للنسب:
!>          RESLTS(1) = norm(X - XACT) / ( norm(X) * FERR )
!>          RESLTS(2) = BERR / ( NZ*EPS + (*) )
!>
