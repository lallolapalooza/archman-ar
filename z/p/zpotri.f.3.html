وثائق الوظيفة/الإجراء الفرعي

subroutine zpotri (character uplo, integer n, complex*16, dimension( lda, * ) a, integer lda, integer info)

ZPOTRI

الغرض:

!>
!> ZPOTRI تحسب معكوس مصفوفة هرميتية موجبة محددة معقدة A باستخدام تفكيك Cholesky A = U**H*U أو A = L*L**H
!> الذي تم حسابه بواسطة ZPOTRF.
!>

المعلمات

UPLO 
!>          UPLO هو CHARACTER*1
!>          = 'U':  المثلث العلوي من A مخزن؛
!>          = 'L':  المثلث السفلي من A مخزن.
!>

N

!>          N هو INTEGER
!>          ترتيب المصفوفة A.  N >= 0.
!>

A

!>          A هو مصفوفة COMPLEX*16، ذات بعد (LDA,N)
!>          عند الإدخال، العامل المثلث U أو L من تفكيك Cholesky
!>          A = U**H*U أو A = L*L**H، كما تم حسابه بواسطة
!>          ZPOTRF.
!>          عند الخروج، المثلث العلوي أو السفلي للمصفوفة العكسية (الهرميتية)
!>          لـ A، للكتابة فوق العامل المدخل U أو L.
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرائد للمصفوفة A.  LDA >= max(1,N).
!>

INFO

!>          INFO هو INTEGER
!>          = 0:  انتهاء ناجح
!>          < 0:  إذا كان INFO = -i، فإن الوسيط i كان له قيمة غير قانونية
!>          > 0:  إذا كان INFO = i، فإن العنصر (i,i) للعامل U أو L هو
!>                صفر، ولم يكن من الممكن حساب المعكوس.
!>
