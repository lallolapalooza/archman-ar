وثائق الوظيفة/الإجراء الفرعي

subroutine zpbcon (character uplo, integer n, integer kd, complex*16, dimension( ldab, * ) ab, integer ldab, double precision anorm, double precision rcond, complex*16, dimension( * ) work, double precision, dimension( * ) rwork, integer info)

ZPBCON

الغرض:

!>
!> ZPBCON يقدّر مقلوب عدد الشرط (بمعيار 1-norm) لمصفوفة هيرميتية موجبة محددة (complex Hermitian positive definite band matrix)
!> باستخدام تحليل Cholesky A = U**H*U أو A = L*L**H الذي تم حسابه بواسطة ZPBTRF.
!>
!> يتم الحصول على تقدير لـ norm(inv(A))، ويتم حساب مقلوب عدد الشرط كـ RCOND = 1 / (ANORM * norm(inv(A))).
!>

المعاملات

UPLO 
!>          UPLO هو CHARACTER*1
!>          = 'U':  المعامل المثلثي العلوي مخزن في AB؛
!>          = 'L':  المعامل المثلثي السفلي مخزن في AB.
!>

N

!>          N هو INTEGER
!>          رتبة المصفوفة A.  N >= 0.
!>

KD

!>          KD هو INTEGER
!>          عدد الأقطار الفوقية للمصفوفة A إذا كان UPLO = 'U'،
!>          أو عدد الأقطار الفرعية إذا كان UPLO = 'L'.  KD >= 0.
!>

AB

!>          AB هو COMPLEX*16 array، dimension (LDAB,N)
!>          المعامل المثلثي U أو L من تحليل Cholesky
!>          A = U**H*U أو A = L*L**H للمصفوفة الشريطية A، مخزنة في
!>          الصفوف الـ KD+1 الأولى من المصفوفة.  يتم تخزين العمود j من U أو L
!>          في العمود j من المصفوفة AB كالتالي:
!>          إذا كان UPLO ='U'، فإن AB(kd+1+i-j,j) = U(i,j) عندما max(1,j-kd)<=i<=j؛
!>          إذا كان UPLO ='L'، فإن AB(1+i-j,j)    = L(i,j) عندما j<=i<=min(n,j+kd).
!>

LDAB

!>          LDAB هو INTEGER
!>          البعد الرائد للمصفوفة AB.  LDAB >= KD+1.
!>

ANORM

!>          ANORM هو DOUBLE PRECISION
!>          المعيار 1 (أو المعيار اللانهائي) للمصفوفة الهيرميتية الشريطية A.
!>

RCOND

!>          RCOND هو DOUBLE PRECISION
!>          مقلوب عدد الشرط للمصفوفة A،
!>          محسوب كـ RCOND = 1/(ANORM * AINVNM)، حيث AINVNM هو
!>          تقدير لمعيار 1 للمصفوفة المعكوسة inv(A) الذي تم حسابه في هذا الروتين.
!>

WORK

!>          WORK هو COMPLEX*16 array، dimension (2*N)
!>

RWORK

!>          RWORK هو DOUBLE PRECISION array، dimension (N)
!>

INFO

!>          INFO هو INTEGER
!>          = 0:  انتهاء ناجح
!>          < 0:  إذا كان INFO = -i، فإن المعامل i كان له قيمة غير قانونية
!>
