subroutine zpbstf (character uplo, integer n, integer kd, complex*16, dimension( ldab, * ) ab, integer ldab, integer info)

ZPBSTF

الغرض:

!>
!> ZPBSTF يحسب تفكيك Cholesky مقسوم لمصفوفة A الهرمية الموجبة المحددة ذات النطاق.
!>
!> تم تصميم هذا الروتين للاستخدام بالاقتران مع ZHBGST.
!>
!> التفكيك له الشكل A = S**H*S حيث S هي مصفوفة نطاق
!> بنفس عرض النطاق لـ A والهيكل التالي:
!>
!>   S = ( U    )
!>       ( M  L )
!>
!> حيث U مثلثية علوية من الرتبة m = (n+kd)/2، و L مثلثية سفلية
!> من الرتبة n-m.
!>

المعلمات

UPLO 
!>          UPLO هو CHARACTER*1
!>          = 'U':  يتم تخزين المثلث العلوي لـ A؛
!>          = 'L':  يتم تخزين المثلث السفلي لـ A.
!>

N

!>          N هو INTEGER
!>          رتبة المصفوفة A.  N >= 0.
!>

KD

!>          KD هو INTEGER
!>          عدد الأقطار الفائقة للمصفوفة A إذا كان UPLO = 'U'،
!>          أو عدد الأقطار الفرعية إذا كان UPLO = 'L'.  KD >= 0.
!>

AB

!>          AB هو مصفوفة COMPLEX*16، ذات الأبعاد (LDAB,N)
!>          عند الدخول، المثلث العلوي أو السفلي للمصفوفة الهرمية ذات النطاق
!>          A، المخزنة في الصفوف kd+1 الأولى من المصفوفة.  الـ
!>          العمود j من A مخزن في العمود j من المصفوفة AB
!>          كالتالي:
!>          إذا كان UPLO = 'U'، فإن AB(kd+1+i-j,j) = A(i,j) حيث max(1,j-kd)<=i<=j؛
!>          إذا كان UPLO = 'L'، فإن AB(1+i-j,j)    = A(i,j) حيث j<=i<=min(n,j+kd).
!>
!>          عند الخروج، إذا كان INFO = 0، فإن العامل S من تفكيك Cholesky
!>          المقسم A = S**H*S. انظر تفاصيل إضافية.
!>

LDAB

!>          LDAB هو INTEGER
!>          البعد الأمامي للمصفوفة AB.  LDAB >= KD+1.
!>

INFO

!>          INFO هو INTEGER
!>          = 0: الخروج بنجاح
!>          < 0: إذا كان INFO = -i، فإن المعلمة i كانت قيمة غير قانونية
!>          > 0: إذا كان INFO = i، لم يكن من الممكن إكمال التفكيك،
!>               لأن العنصر المحدث a(i,i) كان سالبًا؛
!>               المصفوفة A ليست موجبة محددة.
!>

تفاصيل إضافية:

!>
!>  مخطط تخزين النطاق موضح بالمثال التالي، عندما
!>  N = 7, KD = 2:
!>
!>  S = ( s11  s12  s13                     )
!>      (      s22  s23  s24                )
!>      (           s33  s34                )
!>      (                s44                )
!>      (           s53  s54  s55           )
!>      (                s64  s65  s66      )
!>      (                     s75  s76  s77 )
!>
!>  إذا كان UPLO = 'U'، فإن المصفوفة AB تحتوي على:
!>
!>  عند الدخول:                          عند الخروج:
!>
!>   *    *   a13  a24  a35  a46  a57   *    *   s13  s24  s53**H s64**H s75**H
!>   *   a12  a23  a34  a45  a56  a67   *   s12  s23  s34  s54**H s65**H s76**H
!>  a11  a22  a33  a44  a55  a66  a77  s11  s22  s33  s44  s55    s66    s77
!>
!>  إذا كان UPLO = 'L'، فإن المصفوفة AB تحتوي على:
!>
!>  عند الدخول:                          عند الخروج:
!>
!>  a11  a22  a33  a44  a55  a66  a77  s11    s22    s33    s44  s55  s66  s77
!>  a21  a32  a43  a54  a65  a76   *   s12**H s23**H s34**H s54  s65  s76   *
!>  a31  a42  a53  a64  a64   *    *   s13**H s24**H s53    s64  s75   *    *
!>  العناصر المميزة * في المصفوفة لا تستخدم بواسطة الروتين؛ s12**H تعني
!>  conjg(s12)؛ العناصر القطرية لـ S حقيقية.
!>
