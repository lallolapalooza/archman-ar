subroutine zptrfs (character uplo, integer n, integer nrhs, double precision, dimension( * ) d, complex*16, dimension( * ) e, double precision, dimension( * ) df, complex*16, dimension( * ) ef, complex*16, dimension( ldb, * ) b, integer ldb, complex*16, dimension( ldx, * ) x, integer ldx, double precision, dimension( * ) ferr, double precision, dimension( * ) berr, complex*16, dimension( * ) work, double precision, dimension( * ) rwork, integer info)

ZPTRFS

الغرض:

!>
!> ZPTRFS يحسن الحل المحسوب لنظام المعادلات الخطية
!> عندما تكون مصفوفة المعاملات هيرميتية موجبة محددة
!> وثلاثية الأقطار، ويوفر حدود خطأ وتقديرات خطأ خلفية
!> للحل.
!>

المعلمات

UPLO 
!>          UPLO is CHARACTER*1
!>          يحدد ما إذا كان القطر العلوي أو السفلي للمصفوفة
!>          ثلاثية الأقطار A مخزناً وشكل التحليل:
!>          = 'U':  E هو القطر العلوي لـ A، و A = U**H*D*U؛
!>          = 'L':  E هو القطر السفلي لـ A، و A = L*D*L**H.
!>          (الشكلان متكافئان إذا كانت A حقيقية.)
!>

N

!>          N is INTEGER
!>          رتبة المصفوفة A.  N >= 0.
!>

NRHS

!>          NRHS is INTEGER
!>          عدد الأطراف اليمنى، أي عدد أعمدة
!>          المصفوفة B.  NRHS >= 0.
!>

D

!>          D is DOUBLE PRECISION array, dimension (N)
!>          العناصر القطرية الحقيقية n للمصفوفة ثلاثية الأقطار A.
!>

E

!>          E is COMPLEX*16 array, dimension (N-1)
!>          العناصر خارج القطر n-1 للمصفوفة ثلاثية الأقطار A
!>          (انظر UPLO).
!>

DF

!>          DF is DOUBLE PRECISION array, dimension (N)
!>          العناصر القطرية n للمصفوفة القطرية D من
!>          التحليل الذي تم حسابه بواسطة ZPTTRF.
!>

EF

!>          EF is COMPLEX*16 array, dimension (N-1)
!>          العناصر خارج القطر n-1 لوحدة الم صفوفة المزدوجة
!>          العامل U أو L من التحليل الذي تم حسابه بواسطة ZPTTRF
!>          (انظر UPLO).
!>

B

!>          B is COMPLEX*16 array, dimension (LDB,NRHS)
!>          مصفوفة الطرف الأيمن B.
!>

LDB

!>          LDB is INTEGER
!>          البعد الرئيسي للمصفوفة B.  LDB >= max(1,N).
!>

X

!>          X is COMPLEX*16 array, dimension (LDX,NRHS)
!>          عند الدخول، مصفوفة الحل X، كما تم حسابها بواسطة ZPTTRS.
!>          عند الخروج، مصفوفة الحل المحسنة X.
!>

LDX

!>          LDX is INTEGER
!>          البعد الرئيسي للمصفوفة X.  LDX >= max(1,N).
!>

FERR

!>          FERR is DOUBLE PRECISION array, dimension (NRHS)
!>          حد الخطأ الأمامي لكل متجه حل
!>          X(j) (العمود j من مصفوفة الحل X).
!>          إذا كانت XTRUE هي الحل الحقيقي المقابل لـ X(j)، فإن FERR(j)
!>          هو تقدير لحد أعلى لحجم أكبر
!>          عنصر في (X(j) - XTRUE) مقسوماً على حجم
!>          أكبر عنصر في X(j).
!>

BERR

!>          BERR is DOUBLE PRECISION array, dimension (NRHS)
!>          الخطأ الخلفي النسبي المكون لكل متجه حل
!>          X(j) (أي، أصغر تغيير نسبي في
!>          أي عنصر في A أو B يجعل X(j) حلاً دقيقاً).
!>

WORK

!>          WORK is COMPLEX*16 array, dimension (N)
!>

RWORK

!>          RWORK is DOUBLE PRECISION array, dimension (N)
!>

INFO

!>          INFO is INTEGER
!>          = 0:  خروج ناجح
!>          < 0:  إذا كان INFO = -i، فإن المعامل i كان له قيمة غير قانونية
!>
