subroutine zppsv (character uplo, integer n, integer nrhs, complex*16, dimension( * ) ap, complex*16, dimension( ldb, * ) b, integer ldb, integer info)

ZPPSV تحسب الحل لمعادلات المصفوفة الخطية المعقدة A * X = B،

الغرض:

!>
!> ZPPSV تحسب الحل لنظام المعادلات الخطية المعقدة
!>    A * X = B,
!> حيث A مصفوفة هرميتية موجبة محددة من الحجم N×N مخزنة بتنسيق مضغوط، و X و B مصفوفتان N×NRHS.
!>
!> يتم استخدام تحليل Cholesky لتحليل A إلى:
!>    A = U**H * U،  إذا كان UPLO = 'U'، أو
!>    A = L * L**H،  إذا كان UPLO = 'L'،
!> حيث U مصفوفة مثلثية عليا و L مصفوفة مثلثية سفلية. ثم يتم استخدام الشكل المحلل لـ A لحل نظام المعادلات A * X = B.
!>

المعاملات

UPLO 
!>          UPLO هو CHARACTER*1
!>          = 'U':  يتم تخزين المثلث العلوي لـ A؛
!>          = 'L':  يتم تخزين المثلث السفلي لـ A.
!>

N

!>          N هو INTEGER
!>          عدد المعادلات الخطية، أي، رتبة المصفوفة A.  N >= 0.
!>

NRHS

!>          NRHS هو INTEGER
!>          عدد الأطراف اليمنى، أي، عدد أعمدة المصفوفة B.  NRHS >= 0.
!>

AP

!>          AP هي مصفوفة COMPLEX*16، ذات بعد (N*(N+1)/2)
!>          عند الإدخال، المثلث العلوي أو السفلي للمصفوفة الهرميتية A، مضغوطة حسب العمود في مصفوفة خطية. يتم تخزين العمود j من A
!>          في المصفوفة AP كما يلي:
!>          إذا كان UPLO = 'U'، فإن AP(i + (j-1)*j/2) = A(i,j) لـ 1<=i<=j؛
!>          إذا كان UPLO = 'L'، فإن AP(i + (j-1)*(2n-j)/2) = A(i,j) لـ j<=i<=n.
!>          انظر التفاصيل أدناه.
!>
!>          عند الخروج، إذا كان INFO = 0، فإن العامل U أو L من تحليل Cholesky
!>          A = U**H*U أو A = L*L**H، بنفس تنسيق التخزين مثل A.
!>

B

!>          B هي مصفوفة COMPLEX*16، ذات بعد (LDB,NRHS)
!>          عند الإدخال، المصفوفة B ذات الطرف الأيمن N×NRHS.
!>          عند الخروج، إذا كان INFO = 0، فإن مصفوفة الحل X ذات البعد N×NRHS.
!>

LDB

!>          LDB هو INTEGER
!>          البعد الرائد للمصفوفة B.  LDB >= max(1,N).
!>

INFO

!>          INFO هو INTEGER
!>          = 0:  خروج ناجح
!>          < 0:  إذا كان INFO = -i، فإن المعامل i كان له قيمة غير قانونية
!>          > 0:  إذا كان INFO = i، فإن الحد الرئيسي الأولي من الرتبة i
!>                لـ A ليس موجبًا، لذلك لم يكن من الممكن إكمال التحليل،
!>                ولم يتم حساب الحل.
!>

!>
!>  التخزين ثنائي الأبعاد للمصفوفة الهرميتية A:
