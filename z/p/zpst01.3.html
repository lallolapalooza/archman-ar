subroutine zpst01 (character uplo, integer n, complex*16, dimension( lda, * ) a, integer lda, complex*16, dimension( ldafac, * ) afac, integer ldafac, complex*16, dimension( ldperm, * ) perm, integer ldperm, integer, dimension( * ) piv, double precision, dimension( * ) rwork, double precision resid, integer rank)

ZPST01

الغرض:

!>
!> ZPST01 يعيد بناء مصفوفة هيرميتية موجبة شبه محددة A
!> من عواملها L أو U ومصفوفة التبديل P ويحسب
!> الباقي
!>    norm( P*L*L'*P' - A ) / ( N * norm(A) * EPS ) أو
!>    norm( P*U'*U*P' - A ) / ( N * norm(A) * EPS ),
!> حيث EPS هو خطأ الآلة، L' هو منقولة الاقتران لـ L،
!> و U' هو منقولة الاقتران لـ U.
!>

المعلمات

UPLO 
!>          UPLO هو CHARACTER*1
!>          يحدد ما إذا كان الجزء العلوي أو السفلي المثلثي من المصفوفة
!>          الهيرميتية A مخزنًا:
!>          = 'U':  مثلثي علوي
!>          = 'L':  مثلثي سفلي
!>

N

!>          N هو INTEGER
!>          عدد الصفوف والأعمدة للمصفوفة A.  N >= 0.
!>

A

!>          A هو مصفوفة COMPLEX*16، البعد (LDA,N)
!>          المصفوفة الهيرميتية الأصلية A.
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرائد للمصفوفة A.  LDA >= max(1,N)
!>

AFAC

!>          AFAC هو مصفوفة COMPLEX*16، البعد (LDAFAC,N)
!>          العامل L أو U من تحليل L*L' أو U'*U
!>          للمصفوفة A.
!>

LDAFAC

!>          LDAFAC هو INTEGER
!>          البعد الرائد للمصفوفة AFAC.  LDAFAC >= max(1,N).
!>

PERM

!>          PERM هو مصفوفة COMPLEX*16، البعد (LDPERM,N)
!>          سيتم استبدالها بالمصفوفة المعاد بناؤها، ثم بالفرق
!>          P*L*L'*P' - A (أو P*U'*U*P' - A)
!>

LDPERM

!>          LDPERM هو INTEGER
!>          البعد الرائد للمصفوفة PERM.
!>          LDAPERM >= max(1,N).
!>

PIV

!>          PIV هي مصفوفة INTEGER، البعد (N)
!>          PIV بحيث تكون المدخلات غير الصفرية
!>          P( PIV( K ), K ) = 1.
!>

RWORK

!>          RWORK هي DOUBLE PRECISION، البعد (N)
!>

RESID

!>          RESID هي DOUBLE PRECISION
!>          إذا كان UPLO = 'L'، norm(L*L' - A) / ( N * norm(A) * EPS )
!>          إذا كان UPLO = 'U'، norm(U'*U - A) / ( N * norm(A) * EPS )
!>

RANK

!>          RANK هو INTEGER
!>          عدد القيم المفردة غير الصفرية للمصفوفة A.
!>
