توثيق الوظيفة/الإجراء الفرعي

subroutine zpotrs (character uplo, integer n, integer nrhs, complex*16, dimension( lda, * ) a, integer lda, complex*16, dimension( ldb, * ) b, integer ldb, integer info)

ZPOTRS

الغرض:

!>
!> ZPOTRS يحل نظام المعادلات الخطية A*X = B مع مصفوفة هرميتية موجبة محددة A
!> باستخدام تحليل Cholesky
!> A = U**H * U أو A = L * L**H الذي تم حسابه بواسطة ZPOTRF.
!>

المعلمات

UPLO 
!>          UPLO هو CHARACTER*1
!>          = 'U':  المثلث العلوي لـ A مخزن؛
!>          = 'L':  المثلث السفلي لـ A مخزن.
!>

N

!>          N هو INTEGER
!>          رتبة المصفوفة A.  N >= 0.
!>

NRHS

!>          NRHS هو INTEGER
!>          عدد الأطراف اليمنى، أي عدد الأعمدة
!>          للمصفوفة B.  NRHS >= 0.
!>

A

!>          A هي مصفوفة COMPLEX*16، ذات أبعاد (LDA,N)
!>          العامل المثلث U أو L من تحليل Cholesky
!>          A = U**H * U أو A = L * L**H، كما تم حسابه بواسطة ZPOTRF.
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرائد للمصفوفة A.  LDA >= max(1,N).
!>

B

!>          B هي مصفوفة COMPLEX*16، ذات أبعاد (LDB,NRHS)
!>          عند الإدخال، مصفوفة الطرف الأيمن B.
!>          عند الخروج، مصفوفة الحل X.
!>

LDB

!>          LDB هو INTEGER
!>          البعد الرائد للمصفوفة B.  LDB >= max(1,N).
!>

INFO

!>          INFO هو INTEGER
!>          = 0:  انتهاء ناجح
!>          < 0:  إذا كان INFO = -i، فقد كان الوسيط i غير قانوني
!>
