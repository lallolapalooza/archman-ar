subroutine zpteqr (character compz, integer n, double precision, dimension( * ) d, double precision, dimension( * ) e, complex*16, dimension( ldz, * ) z, integer ldz, double precision, dimension( * ) work, integer info)

ZPTEQR

الغرض:

!>
!> ZPTEQR يحسب جميع القيم الذاتية، واختيارياً، المتجهات الذاتية لمصفوفة متماثلة موجبة محددة ثلاثية الأقطار عن طريق تحليل المصفوفة أولاً باستخدام DPTTRF ثم استدعاء ZBDSQR لحساب قيم التفرد للعامل ثنائي الأقطار.
!>
!> يقوم هذا الروتين الفرعي بحساب القيم الذاتية للمصفوفة ثلاثية الأقطار الموجبة المحددة بدقة نسبية عالية. هذا يعني أنه إذا تراوحت القيم الذاتية عبر العديد من رتب الحجم في الحجم، فسيتم حساب القيم الذاتية الصغيرة والمتجهات الذاتية المقابلة بدقة أكبر من، على سبيل المثال، الطريقة القياسية QR.
!>
!> يمكن أيضًا العثور على المتجهات الذاتية لمصفوفة هرميتية كاملة أو نطاقية موجبة المحددة إذا تم استخدام ZHETRD أو ZHPTRD أو ZHBTRD لتقليل هذه المصفوفة إلى شكل ثلاثي الأقطار. (ومع ذلك، فإن التخفيض إلى شكل ثلاثي الأقطار قد يمنع إمكانية الحصول على دقة نسبية عالية في القيم الذاتية الصغيرة للمصفوفة الأصلية، إذا تراوحت هذه القيم الذاتية عبر العديد من رتب الحجم.)
!>

المعلمات

COMPZ 
!>          COMPZ is CHARACTER*1
!>          = 'N':  حساب القيم الذاتية فقط.
!>          = 'V':  حساب المتجهات الذاتية للمصفوفة الهرميتية الأصلية
!>                  أيضًا. يحتوي المصفوفة Z على المصفوفة الوحدوية
!>                  المستخدمة لتقليل المصفوفة الأصلية إلى شكل ثلاثي الأقطار.
!>          = 'I':  حساب المتجهات الذاتية للمصفوفة ثلاثية الأقطار أيضًا.
!>

N

!>          N is INTEGER
!>          رتبة المصفوفة. N >= 0.
!>

D

!>          D is DOUBLE PRECISION array, dimension (N)
!>          عند الإدخال، العناصر القطرية n للمصفوفة ثلاثية الأقطار.
!>          عند الخروج الطبيعي، يحتوي D على القيم الذاتية، بترتيب تنازلي.
!>

E

!>          E is DOUBLE PRECISION array, dimension (N-1)
!>          عند الإدخال، العناصر الفرعية السفلية (n-1) للمصفوفة ثلاثية الأقطار.
!>          عند الخروج، تم تدمير E.
!>

Z

!>          Z is COMPLEX*16 array, dimension (LDZ, N)
!>          عند الإدخال، إذا كان COMPZ = 'V'، فإن المصفوفة الوحدوية المستخدمة في
!>          التخفيض إلى شكل ثلاثي الأقطار.
!>          عند الخروج، إذا كان COMPZ = 'V'، فإن المتجهات الذاتية المتعامدة لـ
!>          المصفوفة الهرميتية الأصلية؛
!>          إذا كان COMPZ = 'I'، فإن المتجهات الذاتية المتعامدة لـ
!>          المصفوفة ثلاثية الأقطار.
!>          إذا كان INFO > 0 عند الخروج، فإن Z يحتوي على المتجهات الذاتية المرتبطة
!>          بالقيم الذاتية المخزنة فقط.
!>          إذا كان COMPZ = 'N'، فلا يتم الرجوع إلى Z.
!>

LDZ

!>          LDZ is INTEGER
!>          البعد الرئيسي للمصفوفة Z. LDZ >= 1، وإذا
!>          كان COMPZ = 'V' أو 'I'، فإن LDZ >= max(1,N).
!>

WORK

!>          WORK is DOUBLE PRECISION array, dimension (4*N)
!>

INFO

!>          INFO is INTEGER
!>          = 0:  خروج ناجح.
!>          < 0:  إذا كان INFO = -i، فإن الوسيط i كان له قيمة غير قانونية.
!>          > 0:  إذا كان INFO = i، و i هو:
!>                <= N  لم يكن من الممكن إجراء تحليل Cholesky للمصفوفة
!>                      لأن العامل الرئيسي الرائد من الرتبة i لم يكن إيجابياً.
!>                > N   فشلت خوارزمية SVD في التقارب؛
!>                      إذا كان INFO = N+i، فإن i من العناصر غير القطرية لـ
!>                      العامل ثنائي الأقطار لم تتقارب إلى الصفر.
!>
