subroutine zptsvx (character fact, integer n, integer nrhs, double precision, dimension( * ) d, complex*16, dimension( * ) e, double precision, dimension( * ) df, complex*16, dimension( * ) ef, complex*16, dimension( ldb, * ) b, integer ldb, complex*16, dimension( ldx, * ) x, integer ldx, double precision rcond, double precision, dimension( * ) ferr, double precision, dimension( * ) berr, complex*16, dimension( * ) work, double precision, dimension( * ) rwork, integer info)

ZPTSVX يحسب حل نظام المعادلات الخطية A * X = B لمصفوفات PT

الغرض:

!>
!> ZPTSVX يستخدم التحليل A = L*D*L**H لحساب الحل
!> لنظام المعادلات الخطية المعقد A*X = B ، حيث A هي
!> مصفوفة N*N هرميتية موجبة محددة ثلاثية القطر و X و B
!> هما مصفوفتان N*NRHS.
!>
!> يتم أيضًا توفير حدود الخطأ للحل وتقدير للشرط.
!>

الوصف:

!>
!> يتم تنفيذ الخطوات التالية:
!>
!> 1. إذا كان FACT = 'N'، يتم تحليل المصفوفة A إلى A = L*D*L**H ، حيث L
!>    هي مصفوفة ثنائية القطر سفلية وحدوية و D هي مصفوفة قطرية.  يمكن
!>    أيضًا اعتبار التحليل على شكل
!>    A = U**H*D*U.
!>
!> 2. إذا كان المصغّر الرئيسي الأولي من الرتبة i ليس موجبًا،
!>    فإن الروتين يعود بـ INFO = i. بخلاف ذلك، يتم استخدام
!>    الشكل المحلل لـ A لتقدير رقم شرط المصفوفة A.
!>    إذا كان مقلوب رقم الشرط أقل من دقة الآلة، يتم إرجاع INFO = N+1
!>    كتحذير، ولكن الروتين يستمر في الحل لـ X وحساب حدود الخطأ كما
!>    هو موضح أدناه.
!>
!> 3. يتم حل نظام المعادلات لـ X باستخدام الشكل المحلل لـ A.
!>
!> 4. يتم تطبيق التنقيح التكراري لتحسين الحل المحسوب
!>    وتقدير حدود الخطأ والأخطاء الخلفية له.
!>

المعلمات

FACT 
!>          FACT هو CHARACTER*1
!>          يحدد ما إذا كان الشكل المحلل للمصفوفة
!>          A متوفرًا عند الدخول أم لا.
!>          = 'F':  عند الدخول، تحتوي DF و EF على الشكل المحلل لـ A.
!>                  لن يتم تعديل D و E و DF و EF.
!>          = 'N':  سيتم نسخ المصفوفة A إلى DF و EF
!>                  وسيتم تحليلها.
!>

N

!>          N هو INTEGER
!>          رتبة المصفوفة A.  N >= 0.
!>

NRHS

!>          NRHS هو INTEGER
!>          عدد الأطراف اليمنى، أي عدد أعمدة
!>          المصفوفتين B و X.  NRHS >= 0.
!>

D

!>          D هو مصفوفة DOUBLE PRECISION، البعد (N)
!>          العناصر القطرية N للمصفوفة الثلاثية القطر A.
!>

E

!>          E هو مصفوفة COMPLEX*16، البعد (N-1)
!>          عناصر تحت القطر N-1 للمصفوفة الثلاثية القطر A.
!>

DF

!>          DF هو مصفوفة DOUBLE PRECISION، البعد (N)
!>          إذا كان FACT = 'F'، فإن DF هي معلمة إدخال وتحتوي عند الدخول
!>          على العناصر القطرية N للمصفوفة القطرية D
!>          من تحليل L*D*L**H لـ A.
!>          إذا كان FACT = 'N'، فإن DF هي معلمة إخراج وتحتوي عند الخروج
!>          على العناصر القطرية N للمصفوفة القطرية D
!>          من تحليل L*D*L**H لـ A.
!>

EF

!>          EF هو مصفوفة COMPLEX*16، البعد (N-1)
!>          إذا كان FACT = 'F'، فإن EF هي معلمة إدخال وتحتوي عند الدخول
!>          على عناصر تحت القطر N-1 للعامل ثنائي القطر الوحدوي L
!>          من تحليل L*D*L**H لـ A.
!>          إذا كان FACT = 'N'، فإن EF هي معلمة إخراج وتحتوي عند الخروج
!>          على عناصر تحت القطر N-1 للعامل ثنائي القطر الوحدوي L
!>          من تحليل L*D*L**H لـ A.
!>

B

!>          B هي مصفوفة COMPLEX*16، البعد (LDB,NRHS)
!>          مصفوفة الطرف الأيمن B بحجم N*NRHS.
!>

LDB

!>          LDB هو INTEGER
!>          البعد الرئيسي للمصفوفة B.  LDB >= max(1,N).
!>

X

!>          X هي مصفوفة COMPLEX*16، البعد (LDX,NRHS)
!>          إذا كان INFO = 0 أو INFO = N+1، فإن مصفوفة الحل X بحجم N*NRHS.
!>

LDX

!>          LDX هو INTEGER
!>          البعد الرئيسي للمصفوفة X.  LDX >= max(1,N).
!>

RCOND

!>          RCOND هو DOUBLE PRECISION
!>          مقلوب رقم شرط المصفوفة A.  إذا كان RCOND
!>          أقل من دقة الآلة (خاصة، إذا
!>          كان RCOND = 0)، فإن المصفوفة مفردة بدقة العمل.
!>          يتم الإشارة إلى هذا الشرط برمز عودة INFO > 0.
!>

FERR

!>          FERR هي مصفوفة DOUBLE PRECISION، البعد (NRHS)
!>          حد الخطأ الأمامي لكل متجه حل
!>          X(j) (العمود j للمصفوفة الحل X).
!>          إذا كان XTRUE هو الحل الحقيقي المقابل لـ X(j)، فإن FERR(j)
!>          هو حد أعلى مقدر لحجم أكبر
!>          عنصر في (X(j) - XTRUE) مقسومًا على حجم أكبر
!>          عنصر في X(j).
!>

BERR

!>          BERR هي مصفوفة DOUBLE PRECISION، البعد (NRHS)
!>          الخطأ النسبي المكوني لكل متجه حل
!>          X(j) (أي، أصغر تغيير نسبي في أي
!>          عنصر في A أو B يجعل X(j) حلاً دقيقًا).
!>

WORK

!>          WORK هي مصفوفة COMPLEX*16، البعد (N)
!>

RWORK

!>          RWORK هي مصفوفة DOUBLE PRECISION، البعد (N)
!>

INFO

!>          INFO هو INTEGER
!>          = 0:  انتهاء ناجح
!>          < 0:  إذا كان INFO = -i، فإن المعلمة i كانت قيمة غير قانونية
!>          > 0:  إذا كان INFO = i، وكان i
!>                <= N:  المصغّر الرئيسي الأولي من الرتبة i للمصفوفة A
!>                       ليس موجبًا، لذلك لم يكن من الممكن إكمال التحليل،
!>                       ولم يتم حساب الحل. يتم إرجاع RCOND = 0.
!>                = N+1: U غير مفردة، ولكن RCOND أقل من دقة
!>                       الآلة، مما يعني أن المصفوفة مفردة
!>                       بدقة العمل. ومع ذلك، يتم
!>                       حساب الحل وحدود الخطأ لأنه
!>                       توجد عدد من المواقف التي يمكن
!>                       فيها أن يكون الحل المحسوب أكثر دقة من
!>                       قيمة RCOND.
!>
