zposvx(3)

الحزمة: lapack-doc

الاسم

zposvx - حل نظام خطي خطي مع مصفوفة إيجابية محددة باستخدام التحليل LU.

الملخص

 SUBROUTINE ZPOSVX(
     FACT,     UPLO,     N,      NRHS,   A,      LDA,    AF,     LDAF,
     IPIV,     EQUED,  RCOND,  FERR,   BERR,   X,      LDX,    WORK,
     RWORK,  INFO
 )


REAL*8 FUNCTION ZPOSVX(
     CHARACTER(1)  FACT,   CHARACTER(1)  UPLO,   INTEGER  N,      INTEGER  NRHS,
     COMPLEX*16    A( LDA, * ),     INTEGER  LDA,    COMPLEX*16 AF( LDAF, * ),
     INTEGER  LDAF,  INTEGER  IPIV( * ),     CHARACTER(1)  EQUED,  REAL*8   RCOND,
     REAL*8   FERR( * ),     REAL*8   BERR( * ),     COMPLEX*16 X( LDX, * ),
     INTEGER  LDX,    COMPLEX*16 WORK( * ),     REAL*8   RWORK( * ),  INTEGER  INFO
 )

توثيق الدالة/الإجراء الفرعي

ZPOSVX هو إجراء فرعي عام.

يستخدم هذا الإجراء الفرعي تحليل LU لتحليل مصفوفة مربعة A، ويحل نظام خطي واحد أو أكثر من الأشكال A*X = B أو A'*X = B أو A**H*X = B.

تُفترض المصفوفة A أن تكون إيجابية محددة. يجب أن تكون المصفوفة A إيجابية محددة بالفعل.

الاستخدام:

 CALL ZPOSVX( FACT, UPLO, N, NRHS, A, LDA, AF, LDAF, IPIV, EQUED, RCOND,
               FERR, BERR, X, LDX, WORK, RWORK, INFO )

المعلمات

FACT
CHARACTER(1)

FACT = 'N': قم بمعالجة الجزء الأول (تحليل المصفوفة).

FACT = 'F': قم بمعالجة الجزء الأول (تحليل المصفوفة).

تُستخدم القيمة FACT = 'N' فقط إذا كانت المصفوفة A قابلة للتبديل.

UPLO
CHARACTER(1)

UPLO = 'U': الجزء العلوي.

UPLO = 'L': الجزء السفلي.

N
INTEGER

رتبة المصفوفة A. N >= 0.

NRHS
INTEGER

عدد الأنظمة الفرعية المراد حلها، أي عدد الأعمدة في المصفوفة B. 0 <= NRHS.

A
COMPLEX*16 A( LDA, * )

مصفوفة المصفوفة A بحجم LDA × N.

LDA
INTEGER

البعد الرئيسي للمصفوفة A. LDA >= max(1,N).

AF
COMPLEX*16 AF( LDAF, * )

تخزن التحليل LU للمصفوفة A. حجم AF هو LDAF × N.

LDAF
INTEGER

البعد الرئيسي للمصفوفة AF. LDAF >= max(1,N).

IPIV
INTEGER IPIV( * )

متجه التبديل، حجمه N. يحتوي هذا المتجه على تفاصيل التبديل الأساسي الذي تم إجراؤه.

EQUED
CHARACTER(1)

EQUED = 'N': لم يتم إجراء الموازنة.

EQUED = 'R': تم إجراء الموازنة من اليمين. تعديل المصفوفة A.

EQUED = 'C': تم إجراء الموازنة من اليسار. تعديل المصفوفة A.

EQUED = 'B': تم إجراء الموازنة من كلا الجانبين. تعديل المصفوفة A.

إذا كان EQUED = 'Y'، فإن A و B يتم تعديلهما عند الخروج، وحل النظام المتوازن هو inv(diag(S))*X.

RCOND
REAL*8

تقدير المقلوب لرقم الشرط للمصفوفة A بعد الموازنة (إذا تم إجراؤه). إذا كان RCOND أقل من دقة الآلة (بشكل خاص، إذا كان RCOND = 0)، فإن المصفوفة منفردة بالنسبة لدقة العمل. يتم الإشارة إلى هذا الشرط بعودة INFO > 0.

FERR
REAL*8 FERR( * )

حد الخطأ الأمامي المقدر لكل متجه حل X(j) (العمود j للمصفوفة الحل X). إذا كان XTRUE هو الحل الحقيقي المقابل لـ X(j)، فإن FERR(j) هو حد علوي مقدر للقيمة المطلقة لأكبر عنصر في (X(j) - XTRUE) مقسومًا على القيمة المطلقة لأكبر عنصر في X(j). التقدير موثوق به مثل تقدير RCOND، وهو دائمًا ما يزيد قليلاً عن الخطأ الحقيقي.

BERR
REAL*8 BERR( * )

الخطأ الخلفي النسبي المكون لكل متجه حل X(j) (أي، أصغر تغيير نسبي في أي عنصر في A أو B يجعل X(j) حلاً دقيقًا).

X
COMPLEX*16 X( LDX, * )

المصفوفة الناتجة بحجم LDX × NRHS. يحتوي كل عمود من X على حل النظام الفرعي المقابل.

LDX
INTEGER

البعد الرئيسي للمصفوفة X. LDX >= max(1,N).

WORK
COMPLEX*16 WORK( * )

مصفوفة عمل. حجمها هو 2*N.

RWORK
REAL*8 RWORK( * )

مصفوفة عمل حقيقية. حجمها هو N.

INFO
INTEGER

= 0: خروج ناجح.

< 0: إذا كان INFO = -i، فقد كان الوسيط i قيمة غير قانونية.

> 0: إذا كان INFO = i، و i هو:

<= N: الجزئي الرئيسي البادئ من الرتبة i للمصفوفة A ليس موجبًا، لذا لم يكن من الممكن إكمال التحليل، ولم يتم حساب الحل. يتم إرجاع RCOND = 0.

= N+1: U غير مفردة، ولكن RCOND أقل من دقة الآلة، مما يعني أن المصفوفة منفردة بالنسبة لدقة العمل. ومع ذلك، يتم حساب الحل وحدود الخطأ لأنه في عدد من المواقف يمكن أن يكون الحل المحسوب أكثر دقة مما يوحي به قيمة RCOND.
