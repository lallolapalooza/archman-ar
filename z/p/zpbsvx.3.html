توثيق الوظيفة/الروتين الفرعي

subroutine zpbsvx (character fact, character uplo, integer n, integer kd, integer nrhs, complex*16, dimension( ldab, * ) ab, integer ldab, complex*16, dimension( ldafb, * ) afb, integer ldafb, character equed, double precision, dimension( * ) s, complex*16, dimension( ldb, * ) b, integer ldb, complex*16, dimension( ldx, * ) x, integer ldx, double precision rcond, double precision, dimension( * ) ferr, double precision, dimension( * ) berr, complex*16, dimension( * ) work, double precision, dimension( * ) rwork, integer info)

ZPBSVX computes the solution to system of linear equations A * X = B for OTHER matrices

الغرض:

!>
!> يستخدم ZPBSVX تحليل Cholesky A = U**H*U أو A = L*L**H لحساب حل نظام
!> المعادلات الخطية المعقدة
!>    A * X = B,
!> حيث A مصفوفة نطاقية هرميتية موجبة محددة ذات حجم N×N، و X و B مصفوفتان بحجم N×NRHS.
!>
!> يتم أيضًا توفير حدود الخطأ للحل وتقدير للشرط.
!>

الوصف:

!>
!> يتم تنفيذ الخطوات التالية:
!>
!> 1. إذا كان FACT = 'E'، يتم حساب عوامل قياس حقيقية لمعادلة النظام:
!>       diag(S) * A * diag(S) * inv(diag(S)) * X = diag(S) * B
!>    سواء تم معادلة النظام أم لا يعتمد على قياس المصفوفة A، ولكن إذا تم استخدام
!>    المعادلة، يتم استبدال A بـ diag(S)*A*diag(S) و B بـ diag(S)*B.
!>
!> 2. إذا كان FACT = 'N' أو 'E'، يتم استخدام تحليل Cholesky لتحليل المصفوفة A
!>    (بعد المعادلة إذا كان FACT = 'E') كـ
!>       A = U**H * U,  إذا كان UPLO = 'U', أو
!>       A = L * L**H,  إذا كان UPLO = 'L',
!>    حيث U مصفوفة نطاقية مثلثة عليا، و L مصفوفة نطاقية مثلثة سفلى.
!>
!> 3. إذا كان الحد الرئيسي الأدنى من الرتبة i غير موجب،
!>    فإن الروتين يعود بـ INFO = i. بخلاف ذلك، يتم استخدام الشكل المحلل لـ A
!>    لتقدير رقم شرط المصفوفة A. إذا كان مقلوب رقم الشرط أقل من الدقة الآلية،
!>    يتم إرجاع INFO = N+1 كتحذير، ولكن الروتين يستمر في الحل لـ X
!>    وحساب حدود الخطأ كما هو موضح أدناه.
!>
!> 4. يتم حل نظام المعادلات لـ X باستخدام الشكل المحلل لـ A.
!>
!> 5. يتم تطبيق التنقيح التكراري لتحسين الحل المحسوب
!>    وحساب حدود الخطأ وتقديرات الخطأ العكسي له.
!>
!> 6. إذا تم استخدام المعادلة، يتم ضرب المصفوفة X مسبقًا
!>    بـ diag(S) بحيث تحل النظام الأصلي قبل
!>    المعادلة.
!>

المعلمات

FACT 
!>          FACT هو CHARACTER*1
!>          يحدد ما إذا كان الشكل المحلل للمصفوفة A مقدمًا عند الإدخال، وإذا لم يكن الأمر كذلك،
!>          فهل يجب معادلة المصفوفة A قبل تحليلها.
!>          = 'F':  عند الإدخال، يحتوي AFB على الشكل المحلل لـ A.
!>                  إذا كان EQUED = 'Y'، فقد تمت معادلة المصفوفة A
!>                  بعوامل قياس معطاة بواسطة S. لن يتم تعديل AB و AFB.
!>          = 'N':  سيتم نسخ المصفوفة A إلى AFB وتحليلها.
!>          = 'E':  سيتم معادلة المصفوفة A إذا لزم الأمر، ثم
!>                  نسخها إلى AFB وتحليلها.
!>

UPLO

!>          UPLO هو CHARACTER*1
!>          = 'U':  يتم تخزين المثلث العلوي لـ A؛
!>          = 'L':  يتم تخزين المثلث السفلي لـ A.
!>

N

!>          N هو INTEGER
!>          عدد المعادلات الخطية، أي رتبة المصفوفة A. N >= 0.
!>

KD

!>          KD هو INTEGER
!>          عدد الأقطار العلوية للمصفوفة A إذا كان UPLO = 'U'،
!>          أو عدد الأقطار السفلية إذا كان UPLO = 'L'. KD >= 0.
!>

NRHS

!>          NRHS هو INTEGER
!>          عدد الأطراف اليمنى، أي عدد أعمدة المصفوفتين B و X. NRHS >= 0.
!>

AB

!>          AB هي مصفوفة COMPLEX*16، أبعادها (LDAB,N)
!>          عند الإدخال، المثلث العلوي أو السفلي للمصفوفة النطاقية الهرميتية
!>          A، مخزنة في أول KD+1 صفوف من المصفوفة، باستثناء
!>          إذا كان FACT = 'F' و EQUED = 'Y'، فإن A يجب أن تحتوي على
!>          المصفوفة المعادلة diag(S)*A*diag(S). يتم تخزين العمود j من A
!>          في العمود j من المصفوفة AB كالتالي:
!>          إذا كان UPLO = 'U'، AB(KD+1+i-j,j) = A(i,j) لـ max(1,j-KD)<=i<=j؛
!>          إذا كان UPLO = 'L'، AB(1+i-j,j)    = A(i,j) لـ j<=i<=min(N,j+KD).
!>          انظر أدناه لمزيد من التفاصيل.
!>
!>          عند الخروج، إذا كان FACT = 'E' و EQUED = 'Y'، يتم استبدال A بـ
!>          diag(S)*A*diag(S).
!>

LDAB

!>          LDAB هو INTEGER
!>          البعد الرائد للمصفوفة A. LDAB >= KD+1.
!>

AFB

!>          AFB هي مصفوفة COMPLEX*16، أبعادها (LDAFB,N)
!>          إذا كان FACT = 'F'، فإن AFB هي معلمة إدخال وتحتوي عند الإدخال
!>          على العامل المثلثي U أو L من تحليل Cholesky
!>          A = U**H *U أو A = L*L**H للمصفوفة النطاقية
!>          A، بنفس تنسيق التخزين لـ A (انظر AB). إذا كان EQUED = 'Y'،
!>          فإن AFB هو الشكل المحلل للمصفوفة المعادلة A.
!>
!>          إذا كان FACT = 'N'، فإن AFB هي معلمة إخراج وتحتوي عند الخروج
!>          على العامل المثلثي U أو L من تحليل Cholesky
!>          A = U**H *U أو A = L*L**H.
!>
!>          إذا كان FACT = 'E'، فإن AFB هي معلمة إخراج وتحتوي عند الخروج
!>          على العامل المثلثي U أو L من تحليل Cholesky
!>          A = U**H *U أو A = L*L**H للمصفوفة المعادلة
!>          A (انظر وصف A لشكل المصفوفة المعادلة).
!>

LDAFB

!>          LDAFB هو INTEGER
!>          البعد الرائد للمصفوفة AFB. LDAFB >= KD+1.
!>

EQUED

!>          EQUED هو CHARACTER*1
!>          يحدد شكل المعادلة التي تم إجراؤها.
!>          = 'N':  لم تتم المعادلة (صحيح دائمًا إذا كان FACT = 'N').
!>          = 'Y':  تمت المعادلة، أي تم استبدال A بـ
!>                  diag(S) * A * diag(S).
!>          EQUED هي معلمة إدخال إذا كان FACT = 'F'. بخلاف ذلك، فهي معلمة
!>          إخراج.
!>

S

!>          S هي مصفوفة DOUBLE PRECISION، أبعادها (N)
!>          عوامل القياس لـ A؛ لا يتم الوصول إليها إذا كان EQUED = 'N'. S هي
!>          معلمة إدخال إذا كان FACT = 'F'. بخلاف ذلك، S هي معلمة
!>          إخراج. إذا كان FACT = 'F' و EQUED = 'Y'، يجب أن يكون كل عنصر في S
!>          موجبًا.
!>

B

!>          B هي مصفوفة COMPLEX*16، أبعادها (LDB,NRHS)
!>          عند الإدخال، مصفوفة الطرف الأيمن N×NRHS B.
!>          عند الخروج، إذا كان EQUED = 'N'، لا يتم تعديل B؛ إذا كان EQUED = 'Y'،
!>          يتم استبدال B بـ diag(S) * B.
!>

LDB

!>          LDB هو INTEGER
!>          البعد الرائد للمصفوفة B. LDB >= max(1,N).
!>

X

!>          X هي مصفوفة COMPLEX*16، أبعادها
