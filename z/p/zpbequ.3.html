subroutine zpbequ (character uplo, integer n, integer kd, complex*16, dimension( ldab, * ) ab, integer ldab, double precision, dimension( * ) s, double precision scond, double precision amax, integer info)

ZPBEQU

الغرض:

!>
!> ZPBEQU يحسب عوامل قياس للصف والأعمدة تهدف إلى موازنة مصفوفة هيرميتية موجبة محددة A وتقليل عدد شرطتها
!> (بالنسبة للمعيار الثنائي). يحتوي S على عوامل القياس،
!> S(i) = 1/sqrt(A(i,i))، مختارة بحيث يكون للمصفوفة المقاسة B ذات
!> العناصر B(i,j) = S(i)*A(i,j)*S(j) واحدات على القطر. هذا
!> الاختيار لـ S يضع عدد شرطة B في حدود عامل N من
!> أصغر عدد شرطة ممكن عبر جميع القياسات القطرية الممكنة.
!>

المعاملات

UPLO 
!>          UPLO هو CHARACTER*1
!>          = 'U':  المثلث العلوي لـ A مخزن؛
!>          = 'L':  المثلث السفلي لـ A مخزن.
!>

N

!>          N هو INTEGER
!>          ترتيب المصفوفة A.  N >= 0.
!>

KD

!>          KD هو INTEGER
!>          عدد الأقطار الفائقة للمصفوفة A إذا كان UPLO = 'U'،
!>          أو عدد الأقطار الفرعية إذا كان UPLO = 'L'.  KD >= 0.
!>

AB

!>          AB هو COMPLEX*16 array, dimension (LDAB,N)
!>          المثلث العلوي أو السفلي للمصفوفة الهيرميتية النطاقية A،
!>          مخزن في الصفوف KD+1 الأولى من المصفوفة. العمود j
!>          لـ A مخزن في العمود j من المصفوفة AB كما يلي:
!>          إذا كان UPLO = 'U'، AB(kd+1+i-j,j) = A(i,j) لـ max(1,j-kd)<=i<=j؛
!>          إذا كان UPLO = 'L'، AB(1+i-j,j)    = A(i,j) لـ j<=i<=min(n,j+kd).
!>

LDAB

!>          LDAB هو INTEGER
!>          البعد الرائد للمصفوفة A.  LDAB >= KD+1.
!>

S

!>          S هو DOUBLE PRECISION array, dimension (N)
!>          إذا كان INFO = 0، يحتوي S على عوامل القياس لـ A.
!>

SCOND

!>          SCOND هو DOUBLE PRECISION
!>          إذا كان INFO = 0، يحتوي S على نسبة أصغر S(i) إلى
!>          أكبر S(i). إذا كان SCOND >= 0.1 و AMAX ليس كبيرًا جدًا
!>          أو صغيرًا جدًا، فلا داعي للقياس بواسطة S.
!>

AMAX

!>          AMAX هو DOUBLE PRECISION
!>          القيمة المطلقة لأكبر عنصر في المصفوفة. إذا كان AMAX قريبًا جدًا
!>          من تجاوز السعة أو قريبًا جدًا من السعة الضئيلة، يجب قياس المصفوفة.
!>

INFO

!>          INFO هو INTEGER
!>          = 0:  انتهاء ناجح
!>          < 0:  إذا كان INFO = -i، كان المعامل i ذو قيمة غير قانونية.
!>          > 0:  إذا كان INFO = i، فإن العنصر القطري i غير موجب.
!>
