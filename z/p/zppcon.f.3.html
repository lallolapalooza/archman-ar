subroutine zppcon (character uplo, integer n, complex*16, dimension( * ) ap, double precision anorm, double precision rcond, complex*16, dimension( * ) work, double precision, dimension( * ) rwork, integer info)

ZPPCON

الغرض:

!>
!> ZPPCON يقدر المعكوس لعدد الشرط (بمعيار 1-norm) لمصفوفة هرميتية
!> معقدة موجبة محددة مضغوطة باستخدام تحليل Cholesky A = U**H*U أو
!> A = L*L**H الذي تم حسابه بواسطة ZPPTRF.
!>
!> يتم الحصول على تقدير لـ norm(inv(A))، ويتم حساب معكوس عدد الشرط
!> كـ RCOND = 1 / (ANORM * norm(inv(A))).
!>

المعلمات

UPLO 
!>          UPLO is CHARACTER*1
!>          = 'U':  المثلث العلوي لـ A مخزن؛
!>          = 'L':  المثلث السفلي لـ A مخزن.
!>

N

!>          N is INTEGER
!>          رتبة المصفوفة A. N >= 0.
!>

AP

!>          AP is COMPLEX*16 array, dimension (N*(N+1)/2)
!>          العامل المثلثي U أو L من تحليل Cholesky
!>          A = U**H*U أو A = L*L**H، مضغوط حسب العمود في مصفوفة خطية.
!>          يتم تخزين العمود j من U أو L في المصفوفة AP
!>          على النحو التالي:
!>          if UPLO = 'U', AP(i + (j-1)*j/2) = U(i,j) for 1<=i<=j;
!>          if UPLO = 'L', AP(i + (j-1)*(2n-j)/2) = L(i,j) for j<=i<=n.
!>

ANORM

!>          ANORM is DOUBLE PRECISION
!>          معيار 1 (أو معيار اللانهاية) للمصفوفة الهرميتية A.
!>

RCOND

!>          RCOND is DOUBLE PRECISION
!>          معكوس عدد شرط المصفوفة A،
!>          محسوب كـ RCOND = 1/(ANORM * AINVNM)، حيث AINVNM هو
!>          تقدير لمعيار 1 لـ inv(A) المحسوب في هذا الإجراء.
!>

WORK

!>          WORK is COMPLEX*16 array, dimension (2*N)
!>

RWORK

!>          RWORK is DOUBLE PRECISION array, dimension (N)
!>

INFO

!>          INFO is INTEGER
!>          = 0:  إنهاء ناجح
!>          < 0:  if INFO = -i, the i-th argument had an illegal value
!>
