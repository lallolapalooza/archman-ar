توثيق الوظيفة/الوحدة الفرعية

subroutine zpftri (character transr, character uplo, integer n, complex*16, dimension( 0: * ) a, integer info)

ZPFTRI

الغرض:

!>
!> ZPFTRI تحسب معكوس المصفوفة الهرميتية المعرفة موجبة A باستخدام تحلل Cholesky A = U**H*U أو A = L*L**H
!> الذي تم حسابه بواسطة ZPFTRF.
!>

المعاملات

TRANSR 
!>          TRANSR هو CHARACTER*1
!>          = 'N':  يتم تخزين TRANSR العادي لـ RFP A؛
!>          = 'C':  يتم تخزين TRANSR المنقول المترافق لـ RFP A.
!>

UPLO

!>          UPLO هو CHARACTER*1
!>          = 'U':  يتم تخزين المثلث العلوي لـ A؛
!>          = 'L':  يتم تخزين المثلث السفلي لـ A.
!>

N

!>          N هو INTEGER
!>          ترتيب المصفوفة A.  N >= 0.
!>

A

!>          A هي مصفوفة COMPLEX*16، أبعادها ( N*(N+1)/2 )؛
!>          عند الدخول، المصفوفة الهرميتية A بتنسيق RFP. يتم وصف تنسيق RFP
!>          بواسطة TRANSR و UPLO و N كما يلي: إذا كان TRANSR = 'N'
!>          فإن RFP A هي (0:N,0:k-1) عندما تكون N زوجية؛ k=N/2. RFP A هي
!>          (0:N-1,0:k) عندما تكون N فردية؛ k=N/2. إذا كان TRANSR = 'C' فإن RFP هي
!>          المنقول المترافق لـ RFP A كما تم تعريفه عندما
!>          يكون TRANSR = 'N'. يتم تعريف محتويات RFP A بواسطة UPLO كما
!>          يلي: إذا كان UPLO = 'U' فإن RFP A تحتوي على nt عنصر من
!>          المثلث العلوي المعبأ A. إذا كان UPLO = 'L' فإن RFP A تحتوي على عناصر
!>          المثلث السفلي المعبأ A. LDA لـ RFP A هو (N+1)/2 عندما يكون TRANSR =
!>          'C'. عندما يكون TRANSR هو 'N' فإن LDA هو N+1 عندما تكون N زوجية و
!>          N فردية. انظر الملاحظة أدناه لمزيد من التفاصيل.
!>
!>          عند الخروج، معكوس المصفوفة الهرميتية للمصفوفة الأصلية، بنفس
!>          تنسيق التخزين.
!>

INFO

!>          INFO هو INTEGER
!>          = 0:  انتهاء ناجح
!>          < 0:  إذا كان INFO = -i، فإن المعامل i كان له قيمة غير قانونية
!>          > 0:  إذا كان INFO = i، فإن العنصر (i,i) للعامل U أو L هو
!>                صفر، ولم يكن من الممكن حساب المعكوس.
!>
