subroutine zptsv (integer n, integer nrhs, double precision, dimension( * ) d, complex*16, dimension( * ) e, complex*16, dimension( ldb, * ) b, integer ldb, integer info)

ZPTSV يحسب حل نظام المعادلات الخطية A * X = B لمصفوفات PT

الغرض:

!>
!> ZPTSV يحسب حل نظام المعادلات الخطية المعقدة A*X = B، حيث A هي مصفوفة
!> ثلاثية القطر متماثلة موجبة محددة من الرتبة N، و X و B هما مصفوفتان من
!> الرتبة N*NRHS.
!>
!> يتم تحليل A إلى A = L*D*L**H، ثم يتم استخدام الشكل المحلل لـ A لحل
!> نظام المعادلات.
!>

المعاملات

N 
!>          N هو عدد صحيح (INTEGER)
!>          رتبة المصفوفة A.  N >= 0.
!>

NRHS

!>          NRHS هو عدد صحيح (INTEGER)
!>          عدد الأطراف اليمنى، أي عدد أعمدة المصفوفة B.  NRHS >= 0.
!>

D

!>          D هي مصفوفة ذات دقة مزدوجة (DOUBLE PRECISION array)، ذات بعد (N)
!>          عند الإدخال، العناصر القطرية n للمصفوفة الثلاثية القطر A.
!>          عند الخروج، العناصر القطرية n للمصفوفة القطرية D من تحليل
!>          A = L*D*L**H.
!>

E

!>          E هي مصفوفة ذات نوع مركب معقد (COMPLEX*16 array)، ذات بعد (N-1)
!>          عند الإدخال، العناصر شبه القطرية (n-1) للمصفوفة الثلاثية القطر A.
!>          عند الخروج، العناصر شبه القطرية (n-1) للعامل الموحد ثنائي القطر L
!>          من تحليل L*D*L**H للمصفوفة A. يمكن اعتبار E أيضًا القطر العلوي
!>          للعامل الموحد ثنائي القطر U من تحليل U**H*D*U للمصفوفة A.
!>

B

!>          B هي مصفوفة ذات نوع مركب معقد (COMPLEX*16 array)، ذات بعد (LDB,NRHS)
!>          عند الإدخال، مصفوفة الطرف الأيمن B ذات الرتبة N*NRHS.
!>          عند الخروج، إذا كانت INFO = 0، فإن مصفوفة الحل X ذات الرتبة N*NRHS.
!>

LDB

!>          LDB هو عدد صحيح (INTEGER)
!>          البعد الرائد للمصفوفة B.  LDB >= max(1,N).
!>

INFO

!>          INFO هو عدد صحيح (INTEGER)
!>          = 0:  خروج ناجح
!>          < 0:  إذا كانت INFO = -i، فإن المعامل i لديه قيمة غير قانونية
!>          > 0:  إذا كانت INFO = i، فإن المحدد الرئيسي الرائد من الرتبة i
!>                ليس موجبًا، ولم يتم حساب الحل. لم يكتمل التحليل إلا إذا
!>                كان i = N.
!>
