توثيق الوظيفة/الإجراء الفرعي

subroutine zpstf2 (character uplo, integer n, complex*16, dimension( lda, * ) a, integer lda, integer, dimension( n ) piv, integer rank, double precision tol, double precision, dimension( 2*n ) work, integer info)

ZPSTF2 تحسب تحلل Cholesky مع محورية كاملة لمصفوفة مركبة هرميتية موجبة شبه محددة.

الغرض:

!>
!> ZPSTF2 تحسب تحلل Cholesky مع محورية كاملة
!> لمصفوفة مركبة هرميتية موجبة شبه محددة A.
!>
!> التحلل له الشكل
!>    P**T * A * P = U**H * U ,  إذا كان UPLO = 'U',
!>    P**T * A * P = L  * L**H,  إذا كان UPLO = 'L',
!> حيث U مصفوفة مثلثية عليا و L مصفوفة مثلثية سفلى، و
!> P مخزنة كمتجه PIV.
!>
!> هذه الخوارزمية لا تحاول التحقق من أن A موجبة
!> شبه محددة. هذا الإصدار من الخوارزمية يستدعي BLAS مستوى 2.
!>

المعاملات

UPLO 
!>          UPLO هو CHARACTER*1
!>          يحدد ما إذا كان الجزء المثلثي العلوي أو السفلي من المصفوفة
!>          المتماثلة A مخزنًا.
!>          = 'U':  مثلثي علوي
!>          = 'L':  مثلثي سفلي
!>

N

!>          N هو INTEGER
!>          رتبة المصفوفة A.  N >= 0.
!>

A

!>          A هي مصفوفة COMPLEX*16، ذات أبعاد (LDA,N)
!>          عند الدخول، المصفوفة المتماثلة A. إذا كان UPLO = 'U'، فإن الجزء
!>          العلوي المثلثي n × n من A يحتوي على الجزء
!>          المثلثي العلوي للمصفوفة A، ولا يتم الرجوع إلى الجزء
!>          المثلثي السفلي الصارم من A. إذا كان UPLO = 'L'، فإن الجزء
!>          السفلي المثلثي n × n من A يحتوي على الجزء
!>          المثلثي السفلي للمصفوفة A، ولا يتم الرجوع إلى الجزء
!>          المثلثي العلوي الصارم من A.
!>
!>          عند الخروج، إذا كان INFO = 0، فإن العامل U أو L من تحلل
!>          Cholesky أعلاه.
!>

PIV

!>          PIV هي مصفوفة INTEGER، ذات أبعاد (N)
!>          PIV بحيث تكون المدخلات غير الصفرية P( PIV(K), K ) = 1.
!>

RANK

!>          RANK هو INTEGER
!>          رتبة A المعطاة بعدد الخطوات التي أكملتها الخوارزمية.
!>

TOL

!>          TOL هو DOUBLE PRECISION
!>          تحمل مسموح به محدد من قبل المستخدم. إذا كان TOL < 0، فسيتم استخدام
!>          N*U*MAX( A( K,K ) ). تتوقف الخوارزمية عند الخطوة (K-1)
!>          إذا كان المحور <= TOL.
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرئيسي للمصفوفة A.  LDA >= max(1,N).
!>

WORK

!>          WORK هي مصفوفة DOUBLE PRECISION، ذات أبعاد (2*N)
!>          مساحة عمل.
!>

INFO

!>          INFO هو INTEGER
!>          < 0: إذا كان INFO = -K، فقد كان الوسيط K له قيمة غير قانونية،
!>          = 0: اكتملت الخوارزمية بنجاح، و
!>          > 0: المصفوفة A إما ناقصة الرتبة مع رتبة محسوبة
!>               كما تم إرجاعها في RANK، أو ليست موجبة شبه محددة. انظر
!>               القسم 7 من ملاحظة العمل #161 في LAPACK لمزيد من
!>               المعلومات.
!>
