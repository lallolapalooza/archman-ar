subroutine zpbt02 (character uplo, integer n, integer kd, integer nrhs, complex*16, dimension( lda, * ) a, integer lda, complex*16, dimension( ldx, * ) x, integer ldx, complex*16, dimension( ldb, * ) b, integer ldb, double precision, dimension( * ) rwork, double precision resid)

ZPBT02

الغرض:

!>
!> ZPBT02 يحسب الباقي لحل نظام معادلات هيرميتية متدرجة  A*x = b:
!>    RESID = norm( B - A*X ) / ( norm(A) * norm(X) * EPS)
!> حيث EPS هي الدقة الآلية.
!>

المعلمات

UPLO 
!>          UPLO هو CHARACTER*1
!>          يحدد ما إذا كان الجزء العلوي أو السفلي المثلثي للمصفوفة الهيرميتية A مخزنًا:
!>          = 'U':  مثلثي علوي
!>          = 'L':  مثلثي سفلي
!>

N

!>          N هو INTEGER
!>          عدد الصفوف والأعمدة للمصفوفة A.  N >= 0.
!>

KD

!>          KD هو INTEGER
!>          عدد الأقطار الفوقية للمصفوفة A إذا كان UPLO = 'U'،
!>          أو عدد الأقطار الفرعية إذا كان UPLO = 'L'.  KD >= 0.
!>

NRHS

!>          NRHS هو INTEGER
!>          عدد الأجزاء اليمنى. NRHS >= 0.
!>

A

!>          A هو مصفوفة COMPLEX*16، البعد (LDA,N)
!>          المصفوفة المتدرجة الهيرميتية الأصلية A.  إذا كان UPLO = 'U'،
!>          يتم تخزين الجزء المثلثي العلوي من A كمصفوفة متدرجة؛ إذا
!>          كان UPLO = 'L'، يتم تخزين الجزء المثلثي السفلي من A.  الأعمدة
!>          للمثلث المناسب مخزنة في أعمدة A والأقطار للمثلث مخزنة في صفوف
!>          A.  انظر ZPBTRF لمزيد من التفاصيل.
!>

LDA

!>          LDA هو INTEGER.
!>          البعد الرئيسي للمصفوفة A.  LDA >= max(1,KD+1).
!>

X

!>          X هي مصفوفة COMPLEX*16، البعد (LDX,NRHS)
!>          متجهات الحل المحسوبة لنظام المعادلات الخطية.
!>

LDX

!>          LDX هو INTEGER
!>          البعد الرئيسي للمصفوفة X.   LDX >= max(1,N).
!>

B

!>          B هي مصفوفة COMPLEX*16، البعد (LDB,NRHS)
!>          عند الإدخال، متجهات الجزء الأيمن لنظام المعادلات الخطية.
!>          عند الخروج، يتم استبدال B بـ B - A*X.
!>

LDB

!>          LDB هو INTEGER
!>          البعد الرئيسي للمصفوفة B.  LDB >= max(1,N).
!>

RWORK

!>          RWORK هي مصفوفة DOUBLE PRECISION، البعد (N)
!>

RESID

!>          RESID هو DOUBLE PRECISION
!>          الحد الأقصى لعدد الأجزاء اليمنى لـ
!>          norm(B - A*X) / ( norm(A) * norm(X) * EPS ).
!>
