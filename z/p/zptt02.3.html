توثيق الوظيفة/الروتين الفرعي

subroutine zptt02 (character uplo, integer n, integer nrhs, double precision, dimension( * ) d, complex*16, dimension( * ) e, complex*16, dimension( ldx, * ) x, integer ldx, complex*16, dimension( ldb, * ) b, integer ldb, double precision resid)

ZPTT02

الغرض:

!>
!> ZPTT02 يحسب الباقي لحل نظام المعادلات المتري الثلاثي الأقطار المتماثل:
!>    RESID = norm(B - A*X) / (norm(A) * norm(X) * EPS),
!> حيث EPS هو خطأ الآلة.
!>

المعاملات

UPLO 
!>          UPLO هو CHARACTER*1
!>          يحدد ما إذا كان القطر فوقي أو تحت القطر الرئيسي للمصفوفة
!>          المثلثة A مخزنًا.
!>          = 'U':  E هو القطر فوقي لـ A
!>          = 'L':  E هو تحت القطر الرئيسي لـ A
!>

N

!>          N هو INTEGER
!>          رتبة المصفوفة A.
!>

NRHS

!>          NRHS هو INTEGER
!>          عدد الأطراف اليمنى، أي عدد أعمدة
!>          المصفوفتين B و X. NRHS >= 0.
!>

D

!>          D هي مصفوفة DOUBLE PRECISION، أبعادها (N)
!>          العناصر القطرية n للمصفوفة المثلثة A.
!>

E

!>          E هي مصفوفة COMPLEX*16، أبعادها (N-1)
!>          العناصر تحت القطر الرئيسي n-1 للمصفوفة المثلثة A.
!>

X

!>          X هي مصفوفة COMPLEX*16، أبعادها (LDX,NRHS)
!>          المصفوفة n × nrhs لمتجهات الحل X.
!>

LDX

!>          LDX هو INTEGER
!>          البعد الرائد للمصفوفة X. LDX >= max(1,N).
!>

B

!>          B هي مصفوفة COMPLEX*16، أبعادها (LDB,NRHS)
!>          عند الإدخال، المصفوفة n × nrhs لمتجهات الطرف الأيمن B.
!>          عند الخروج، يتم استبدال B بالفرق B - A*X.
!>

LDB

!>          LDB هو INTEGER
!>          البعد الرائد للمصفوفة B. LDB >= max(1,N).
!>

RESID

!>          RESID هو DOUBLE PRECISION
!>          norm(B - A*X) / (norm(A) * norm(X) * EPS)
!>
