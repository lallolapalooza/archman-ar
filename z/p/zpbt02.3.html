subroutine zpbt02 (character uplo, integer n, integer kd, integer nrhs, complex*16, dimension( lda, * ) a, integer lda, complex*16, dimension( ldx, * ) x, integer ldx, complex*16, dimension( ldb, * ) b, integer ldb, double precision, dimension( * ) rwork, double precision resid)

ZPBT02

الغرض:

!>
!> ZPBT02 تحسب الباقي لحل نظام معادلات هيرميتية متجاورة A*x = b:
!>    RESID = norm( B - A*X ) / ( norm(A) * norm(X) * EPS)
!> حيث EPS هي دقة الآلة.
!>

المعلمات

UPLO 
!>          UPLO is CHARACTER*1
!>          يحدد ما إذا كان الجزء العلوي أو السفلي من المصفوفة الهيرميتية A مخزنًا:
!>          = 'U':  المثلث العلوي
!>          = 'L':  المثلث السفلي
!>

N

!>          N is INTEGER
!>          عدد الصفوف والأعمدة للمصفوفة A. N >= 0.
!>

KD

!>          KD is INTEGER
!>          عدد الأقطار الفائقة للمصفوفة A إذا كان UPLO = 'U'،
!>          أو عدد الأقطار الفرعية إذا كان UPLO = 'L'. KD >= 0.
!>

NRHS

!>          NRHS is INTEGER
!>          عدد الأطراف اليمنى. NRHS >= 0.
!>

A

!>          A is COMPLEX*16 array, dimension (LDA,N)
!>          المصفوفة الهيرميتية المتجاورة الأصلية A. إذا كان UPLO = 'U'،
!>          يتم تخزين الجزء المثلثي العلوي من A كمصفوفة متجاورة؛ إذا كان
!>          UPLO = 'L'، يتم تخزين الجزء المثلثي السفلي من A.
!>          يتم تخزين أعمدة المثلث المناسب في أعمدة
!>          A ويتم تخزين أقطار المثلث في صفوف
!>          A. انظر ZPBTRF لمزيد من التفاصيل.
!>

LDA

!>          LDA is INTEGER.
!>          البعد الرئيسي للمصفوفة A. LDA >= max(1,KD+1).
!>

X

!>          X is COMPLEX*16 array, dimension (LDX,NRHS)
!>          متجهات الحل المحسوبة لنظام المعادلات الخطية.
!>

LDX

!>          LDX is INTEGER
!>          البعد الرئيسي للمصفوفة X. LDX >= max(1,N).
!>

B

!>          B is COMPLEX*16 array, dimension (LDB,NRHS)
!>          عند الدخول، متجهات الطرف الأيمن لنظام المعادلات الخطية.
!>          عند الخروج، يتم استبدال B بالفرق B - A*X.
!>

LDB

!>          LDB is INTEGER
!>          البعد الرئيسي للمصفوفة B. LDB >= max(1,N).
!>

RWORK

!>          RWORK is DOUBLE PRECISION array, dimension (N)
!>

RESID

!>          RESID is DOUBLE PRECISION
!>          الحد الأقصى لعدد الأطراف اليمنى
!>          norm(B - A*X) / ( norm(A) * norm(X) * EPS ).
!>
