subroutine zpoequb (integer n, complex*16, dimension( lda, * ) a, integer lda, double precision, dimension( * ) s, double precision scond, double precision amax, integer info)

ZPOEQUB

الغرض:

!>
!> ZPOEQUB يحسب عوامل القياس للصف والأعمدة المخصصة لموازنة مصفوفة هيرميتية موجبة محددة A وتقليل عدد شرطها
!> (بالنسبة للمعيار الثنائي). S يحتوي على عوامل القياس،
!> S(i) = 1/sqrt(A(i,i))، المختارة بحيث أن المصفوفة المقاسة B بعناصر
!> B(i,j) = S(i)*A(i,j)*S(j) لها وحدات على القطر. هذا الاختيار لـ S
!> يضع عدد شرط B في حدود عامل N من أصغر عدد شرط ممكن عبر جميع
!> القياسات القطرية الممكنة.
!>
!> هذا الإجراء يختلف عن ZPOEQU بتقييد عوامل القياس إلى قوة من الأساس.
!> مع استبعاد الفائض والنقص، فإن القياس بهذه العوامل لا ينتج عنه أخطاء تقريب إضافية.
!> ومع ذلك، فإن العناصر القطرية المقاسة لم تعد تقريبًا 1 ولكنها تقع
!> بين sqrt(radix) و 1/sqrt(radix).
!>

المعاملات

N 
!>          N هو عدد صحيح (INTEGER)
!>          مرتبة المصفوفة A.  N >= 0.
!>

A

!>          A هو مصفوفة COMPLEX*16، البعد (LDA,N)
!>          المصفوفة الهيرميتية الموجبة المحددة N*N التي سيتم حساب عوامل القياس لها.
!>          يتم الرجوع فقط إلى العناصر القطرية لـ A.
!>

LDA

!>          LDA هو عدد صحيح (INTEGER)
!>          البعد الرئيسي للمصفوفة A.  LDA >= max(1,N).
!>

S

!>          S هي مصفوفة DOUBLE PRECISION، البعد (N)
!>          إذا كان INFO = 0، فإن S تحتوي على عوامل القياس لـ A.
!>

SCOND

!>          SCOND هو DOUBLE PRECISION
!>          إذا كان INFO = 0، فإن S تحتوي على نسبة أصغر S(i) إلى
!>          أكبر S(i).  إذا كان SCOND >= 0.1 وكان AMAX ليس كبيرًا جدًا أو صغيرًا جدًا،
!>          فلا يستحق القياس بواسطة S.
!>

AMAX

!>          AMAX هو DOUBLE PRECISION
!>          القيمة المطلقة لأكبر عنصر في المصفوفة. إذا كان AMAX قريبًا جدًا
!>          من الفائض أو قريبًا جدًا من النقص، فيجب قياس المصفوفة.
!>

INFO

!>          INFO هو عدد صحيح (INTEGER)
!>          = 0:  انتهاء ناجح
!>          < 0:  إذا كان INFO = -i، فإن المعامل i كان له قيمة غير قانونية
!>          > 0:  إذا كان INFO = i، فإن العنصر القطري i غير موجب.
!>
