subroutine zpptri (character uplo, integer n, complex*16, dimension( * ) ap, integer info)

ZPPTRI

الغرض:

!>
!> ZPPTRI يحسب معكوس مصفوفة هرميتية موجبة محددة مركبة A باستخدام تحليل Cholesky A = U**H*U أو A = L*L**H
!> الذي تم حسابه بواسطة ZPPTRF.
!>

المعلمات

UPLO 
!>          UPLO is CHARACTER*1
!>          = 'U':  عامل المثلث العلوي مخزن في AP؛
!>          = 'L':  عامل المثلث السفلي مخزن في AP.
!>

N

!>          N is INTEGER
!>          رتبة المصفوفة A.  N >= 0.
!>

AP

!>          AP is COMPLEX*16 array, dimension (N*(N+1)/2)
!>          عند الإدخال، عامل المثلث U أو L من تحليل Cholesky
!>          A = U**H*U أو A = L*L**H، مكدس عموديًا كمصفوفة خطية.
!>          يتم تخزين العمود j من U أو L في المصفوفة AP كما يلي:
!>          إذا كان UPLO = 'U'، فإن AP(i + (j-1)*j/2) = U(i,j) لـ 1<=i<=j؛
!>          إذا كان UPLO = 'L'، فإن AP(i + (j-1)*(2n-j)/2) = L(i,j) لـ j<=i<=n.
!>
!>          عند الخروج، المثلث العلوي أو السفلي للم معكوس (الهرميتي) للمصفوفة A،
!>          يتم الكتابة فوق عامل الإدخال U أو L.
!>

INFO

!>          INFO is INTEGER
!>          = 0:  انتهاء ناجح
!>          < 0:  إذا كان INFO = -i، فإن المعلمة i كانت لها قيمة غير قانونية
!>          > 0:  إذا كان INFO = i، فإن العنصر (i,i) للعامل U أو L هو
!>                صفر، ولم يتمكن من حساب المعكوس.
!>
