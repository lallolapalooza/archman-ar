subroutine zppcon (character uplo, integer n, complex*16, dimension( * ) ap, double precision anorm, double precision rcond, complex*16, dimension( * ) work, double precision, dimension( * ) rwork, integer info)

ZPPCON

الغرض:

!>
!> ZPPCON يقدر مقلوب رقم الشرط (بالنسبة لـ
!> 1-norm) لمصفوفة هيرميتية مركبة موجبة محددة مجمعة باستخدام
!> تحليل Cholesky A = U**H*U أو A = L*L**H المحسوب بواسطة
!> ZPPTRF.
!>
!> يتم الحصول على تقدير لـ norm(inv(A))، ويتم حساب مقلوب
!> رقم الشرط على أنه RCOND = 1 / (ANORM * norm(inv(A))).
!>

المعاملات

UPLO 
!>          UPLO is CHARACTER*1
!>          = 'U':  يتم تخزين المثلث العلوي لـ A؛
!>          = 'L':  يتم تخزين المثلث السفلي لـ A.
!>

N

!>          N is INTEGER
!>          رتبة المصفوفة A.  N >= 0.
!>

AP

!>          AP is COMPLEX*16 array, dimension (N*(N+1)/2)
!>          العامل المثلث U أو L من تحليل Cholesky
!>          A = U**H*U أو A = L*L**H، مجمع عمودياً في مصفوفة خطية
!>          . يتم تخزين العمود j من U أو L في المصفوفة AP
!>          كما يلي:
!>          if UPLO = 'U', AP(i + (j-1)*j/2) = U(i,j) for 1<=i<=j;
!>          if UPLO = 'L', AP(i + (j-1)*(2n-j)/2) = L(i,j) for j<=i<=n.
!>

ANORM

!>          ANORM is DOUBLE PRECISION
!>          1-norm (أو infinity-norm) للمصفوفة الهيرميتية A.
!>

RCOND

!>          RCOND is DOUBLE PRECISION
!>          مقلوب رقم شرط المصفوفة A،
!>          المحسوب كـ RCOND = 1/(ANORM * AINVNM)، حيث AINVNM هو
!>          تقدير لـ 1-norm لـ inv(A) محسوب في هذا الروتين.
!>

WORK

!>          WORK is COMPLEX*16 array, dimension (2*N)
!>

RWORK

!>          RWORK is DOUBLE PRECISION array, dimension (N)
!>

INFO

!>          INFO is INTEGER
!>          = 0:  انتهاء ناجح
!>          < 0:  إذا كان INFO = -i، فإن المعامل i كان له قيمة غير قانونية
!>
