subroutine zpttrf (integer n, double precision, dimension( * ) d, complex*16, dimension( * ) e, integer info)

ZPTTRF

الغرض:

!>
!> ZPTTRF تحسب تحليل L*D*L**H لمصفوفة هرميتية موجبة محددة ثلاثية الأقطار A. يمكن أيضًا اعتبار التحليل على أنه A = U**H *D*U.
!>

المعلمات

N 
!>          N هو عدد صحيح (INTEGER)
!>          رتبة المصفوفة A.  N >= 0.
!>

D

!>          D هو مصفوفة دقة مزدوجة (DOUBLE PRECISION)، البعد (N)
!>          عند الإدخال، عناصر القطر الرئيسية n للمصفوفة ثلاثية الأقطار A. عند الخروج، عناصر القطر الرئيسية n للمصفوفة القطرية D من تحليل L*D*L**H للمصفوفة A.
!>

E

!>          E هو مصفوفة عقدية *16 (COMPLEX*16)، البعد (N-1)
!>          عند الإدخال، عناصر القطر الفرعية (n-1) للمصفوفة ثلاثية الأقطار A. عند الخروج، عناصر القطر الفرعية (n-1) للعامل ثنائي الأقطار الأحادي L من تحليل L*D*L**H للمصفوفة A.
!>          يمكن أيضًا اعتبار E القطر العلوي للعامل ثنائي الأقطار الأحادي U من تحليل U**H *D*U للمصفوفة A.
!>

INFO

!>          INFO هو عدد صحيح (INTEGER)
!>          = 0: خروج ناجح
!>          < 0: إذا كان INFO = -k، فإن المعلمة k كانت لها قيمة غير قانونية
!>          > 0: إذا كان INFO = k، فإن المحدد الرئيسي الرائد من الرتبة k
!>               ليس موجبًا؛ إذا كان k < N، لم يكن من الممكن إكمال التحليل، بينما إذا كان k = N، فقد تم إكمال التحليل، ولكن D(N) <= 0.
!>
