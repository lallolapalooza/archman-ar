توثيق الدالة/الإجراء الفرعي

subroutine zrqt02 (integer m, integer n, integer k, complex*16, dimension( lda, * ) a, complex*16, dimension( lda, * ) af, complex*16, dimension( lda, * ) q, complex*16, dimension( lda, * ) r, integer lda, complex*16, dimension( * ) tau, complex*16, dimension( lwork ) work, integer lwork, double precision, dimension( * ) rwork, double precision, dimension( * ) result)

ZRQT02

الغرض:

!>
!> ZRQT02 تختبر ZUNGRQ، التي تقوم بإنشاء مصفوفة Q بحجم m × n بصفوف متعامدة
!> ويتم تعريفها كناتج لمُبدِلات أولية k.
!>
!> بالنظر إلى تحليل RQ لمصفوفة A بحجم m × n، تقوم ZRQT02 بإنشاء
!> المصفوفة المتعامدة Q المعرفة بتحليل الصفوف k الأخيرة من A؛
!> تقارن R(m-k+1:m,n-m+1:n) مع
!> A(m-k+1:m,1:n)*Q(n-m+1:n,1:n)'، وتتحقق من أن صفوف Q متعامدة.
!>

المعلمات

M 
!>          M هو عدد صحيح (INTEGER)
!>          عدد صفوف المصفوفة Q المراد إنشاؤها. M >= 0.
!>

N

!>          N هو عدد صحيح (INTEGER)
!>          عدد أعمدة المصفوفة Q المراد إنشاؤها.
!>          N >= M >= 0.
!>

K

!>          K هو عدد صحيح (INTEGER)
!>          عدد المُبدِلات الأولية التي يُعرّف حاصل ضربها المصفوفة Q.
!>          M >= K >= 0.
!>

A

!>          A هو مصفوفة COMPLEX*16، بأبعاد (LDA,N)
!>          المصفوفة A بحجم m × n التي تم تحليلها بواسطة ZRQT01.
!>

AF

!>          AF هو مصفوفة COMPLEX*16، بأبعاد (LDA,N)
!>          تفاصيل تحليل RQ للمصفوفة A، كما تم إرجاعه بواسطة ZGERQF.
!>          راجع ZGERQF لمزيد من التفاصيل.
!>

Q

!>          Q هو مصفوفة COMPLEX*16، بأبعاد (LDA,N)
!>

R

!>          R هو مصفوفة COMPLEX*16، بأبعاد (LDA,M)
!>

LDA

!>          LDA هو عدد صحيح (INTEGER)
!>          البعد الرائد للمصفوفات A، AF، Q و L. LDA >= N.
!>

TAU

!>          TAU هو مصفوفة COMPLEX*16، بأبعاد (M)
!>          العوامل القياسية للمُبدِلات الأولية المقابلة لتحليل RQ في AF.
!>

WORK

!>          WORK هو مصفوفة COMPLEX*16، بأبعاد (LWORK)
!>

LWORK

!>          LWORK هو عدد صحيح (INTEGER)
!>          بعد المصفوفة WORK.
!>

RWORK

!>          RWORK هو مصفوفة DOUBLE PRECISION، بأبعاد (M)
!>

RESULT

!>          RESULT هي مصفوفة DOUBLE PRECISION، بأبعاد (2)
!>          نسب الاختبار:
!>          RESULT(1) = norm( R - A*Q' ) / ( N * norm(A) * EPS )
!>          RESULT(2) = norm( I - Q*Q' ) / ( N * EPS )
!>
