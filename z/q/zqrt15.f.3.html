!>

!>

المعلمات

SCALE 
!>          SCALE هو صحيح (INTEGER)
!>          SCALE = 1: مصفوفة مقاسة عادةً
!>          SCALE = 2: مصفوفة مقاسة لأعلى
!>          SCALE = 3: مصفوفة مقاسة لأسفل
!>

RKSEL

!>          RKSEL هو صحيح (INTEGER)
!>          RKSEL = 1: مصفوفة كاملة الرتبة
!>          RKSEL = 2: مصفوفة ذات رتبة ناقصة
!>

M

!>          M هو صحيح (INTEGER)
!>          عدد صفوف المصفوفة A.
!>

N

!>          N هو صحيح (INTEGER)
!>          عدد أعمدة A.
!>

NRHS

!>          NRHS هو صحيح (INTEGER)
!>          عدد أعمدة B.
!>

A

!>          A هي مصفوفة COMPLEX*16، البعد (LDA,N)
!>          المصفوفة A بحجم M×N.
!>

LDA

!>          LDA هو صحيح (INTEGER)
!>          البعد الرئيسي للمصفوفة A.
!>

B

!>          B هي مصفوفة COMPLEX*16، البعد (LDB, NRHS)
!>          مصفوفة تقع في نطاق المصفوفة A.
!>

LDB

!>          LDB هو صحيح (INTEGER)
!>          البعد الرئيسي للمصفوفة B.
!>

S

!>          S هي مصفوفة DOUBLE PRECISION، البعد MIN(M,N)
!>          القيم المفردة لـ A.
!>

RANK

!>          RANK هو صحيح (INTEGER)
!>          عدد القيم المفردة غير الصفرية لـ A.
!>

NORMA

!>          NORMA هي DOUBLE PRECISION
!>          مقياس الواحد للمصفوفة A.
!>

NORMB

!>          NORMB هي DOUBLE PRECISION
!>          مقياس الواحد للمصفوفة B.
!>

ISEED

!>          ISEED هي مصفوفة صحيحة، البعد (4)
!>          البذور لتوليد الأرقام العشوائية.
!>

WORK

!>          WORK هي مصفوفة COMPLEX*16، البعد (LWORK)
!>

LWORK

!>          LWORK هو صحيح (INTEGER)
!>          طول مساحة العمل المطلوبة.
!>          LWORK >= MAX(M+MIN(M,N),NRHS*MIN(M,N),2*N+M)
!>
