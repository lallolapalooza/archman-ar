توثيق الوظيفة/الإجراء الفرعي

subroutine zqrt02 (integer m, integer n, integer k, complex*16, dimension( lda, * ) a, complex*16, dimension( lda, * ) af, complex*16, dimension( lda, * ) q, complex*16, dimension( lda, * ) r, integer lda, complex*16, dimension( * ) tau, complex*16, dimension( lwork ) work, integer lwork, double precision, dimension( * ) rwork, double precision, dimension( * ) result)

ZQRT02

الغرض:

!>
!> ZQRT02 تختبر ZUNGQR، التي تنشئ مصفوفة Q بحجم m×n بأعمدة متعامدة
!> تُعرّف على أنها حاصل ضرب k عاكس أولي.
!>
!> بالنظر إلى تحليل QR لمصفوفة A بحجم m×n، تنشئ ZQRT02
!> المصفوفة المتعامدة Q التي يُعرفها تحليل الأعمدة k الأولى
!> لـ A؛ وتقارن R(1:n,1:k) مع Q(1:m,1:n)'*A(1:m,1:k)،
!> وتتحقق من أن أعمدة Q متعامدة.
!>

المعاملات

M 
!>          M هو عدد صحيح (INTEGER)
!>          عدد صفوف المصفوفة Q التي سيتم إنشاؤها. M >= 0.
!>

N

!>          N هو عدد صحيح (INTEGER)
!>          عدد أعمدة المصفوفة Q التي سيتم إنشاؤها.
!>          M >= N >= 0.
!>

K

!>          K هو عدد صحيح (INTEGER)
!>          عدد العواكس الأولية التي يُعرّف حاصل ضربها المصفوفة Q.
!>          N >= K >= 0.
!>

A

!>          A هي مصفوفة COMPLEX*16، بُعدها (LDA,N)
!>          المصفوفة A بحجم m×n التي تم تحليلها بواسطة ZQRT01.
!>

AF

!>          AF هي مصفوفة COMPLEX*16، بُعدها (LDA,N)
!>          تفاصيل تحليل QR لـ A، كما تم إرجاعها بواسطة ZGEQRF.
!>          راجع ZGEQRF لمزيد من التفاصيل.
!>

Q

!>          Q هي مصفوفة COMPLEX*16، بُعدها (LDA,N)
!>

R

!>          R هي مصفوفة COMPLEX*16، بُعدها (LDA,N)
!>

LDA

!>          LDA هو عدد صحيح (INTEGER)
!>          البعد الرائد للمصفوفات A، AF، Q، و R. LDA >= M.
!>

TAU

!>          TAU هي مصفوفة COMPLEX*16، بُعدها (N)
!>          العوامل العددية للعواكس الأولية المقابلة لتحليل QR في AF.
!>

WORK

!>          WORK هي مصفوفة COMPLEX*16، بُعدها (LWORK)
!>

LWORK

!>          LWORK هو عدد صحيح (INTEGER)
!>          بُعد المصفوفة WORK.
!>

RWORK

!>          RWORK هي مصفوفة DOUBLE PRECISION، بُعدها (M)
!>

RESULT

!>          RESULT هي مصفوفة DOUBLE PRECISION، بُعدها (2)
!>          نسب الاختبار:
!>          RESULT(1) = norm( R - Q'*A ) / ( M * norm(A) * EPS )
!>          RESULT(2) = norm( I - Q'*Q ) / ( M * EPS )
!>
