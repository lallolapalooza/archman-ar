توثيق الوظيفة/الروتين الفرعي

subroutine zqlt02 (integer m, integer n, integer k, complex*16, dimension( lda, * ) a, complex*16, dimension( lda, * ) af, complex*16, dimension( lda, * ) q, complex*16, dimension( lda, * ) l, integer lda, complex*16, dimension( * ) tau, complex*16, dimension( lwork ) work, integer lwork, double precision, dimension( * ) rwork, double precision, dimension( * ) result)

ZQLT02

الغرض:

!>
!> ZQLT02 تختبر ZUNGQL، والتي تنشئ مصفوفة Q بحجم m × n بأعمدة متعامدة
!> محددة كناتج لـ k عاكسات أولية.
!>
!> بالنظر إلى تحليل QL لمصفوفة A بحجم m × n، تنشئ ZQLT02
!> المصفوفة المتعامدة Q المحددة بواسطة تحليل الأعمدة k الأخيرة
!> من A؛ تقارن L(m-n+1:m,n-k+1:n) مع
!> Q(1:m,m-n+1:m)'*A(1:m,n-k+1:n)، وتتحقق من أن أعمدة Q
!> متعامدة.
!>

المعلمات

M 
!>          M هو عدد صحيح (INTEGER)
!>          عدد الصفوف للمصفوفة Q التي سيتم إنشاؤها.  M >= 0.
!>

N

!>          N هو عدد صحيح (INTEGER)
!>          عدد الأعمدة للمصفوفة Q التي سيتم إنشاؤها.
!>          M >= N >= 0.
!>

K

!>          K هو عدد صحيح (INTEGER)
!>          عدد العاكسات الأولية التي يحدد ناتجها المصفوفة Q.
!>          N >= K >= 0.
!>

A

!>          A هي مصفوفة COMPLEX*16، بحجم (LDA,N)
!>          المصفوفة A بحجم m × n التي تم تحليلها بواسطة ZQLT01.
!>

AF

!>          AF هي مصفوفة COMPLEX*16، بحجم (LDA,N)
!>          تفاصيل تحليل QL للمصفوفة A، كما تم إرجاعها بواسطة ZGEQLF.
!>          راجع ZGEQLF لمزيد من التفاصيل.
!>

Q

!>          Q هي مصفوفة COMPLEX*16، بحجم (LDA,N)
!>

L

!>          L هي مصفوفة COMPLEX*16، بحجم (LDA,N)
!>

LDA

!>          LDA هو عدد صحيح (INTEGER)
!>          البعد الرائد للمصفوفات A، AF، Q، و L. LDA >= M.
!>

TAU

!>          TAU هي مصفوفة COMPLEX*16، بحجم (N)
!>          العوامل القياسية للعاكسات الأولية المقابلة لتحليل QL في AF.
!>

WORK

!>          WORK هي مصفوفة COMPLEX*16، بحجم (LWORK)
!>

LWORK

!>          LWORK هو عدد صحيح (INTEGER)
!>          بعد مصفوفة WORK.
!>

RWORK

!>          RWORK هي مصفوفة DOUBLE PRECISION، بحجم (M)
!>

RESULT

!>          RESULT هي مصفوفة DOUBLE PRECISION، بحجم (2)
!>          نسب الاختبار:
!>          RESULT(1) = norm( L - Q'*A ) / ( M * norm(A) * EPS )
!>          RESULT(2) = norm( I - Q'*Q ) / ( M * EPS )
!>
