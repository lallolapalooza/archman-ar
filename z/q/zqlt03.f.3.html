توثيق الدالة/الإجراء الفرعي

subroutine zqlt03 (integer m, integer n, integer k, complex*16, dimension( lda, * ) af, complex*16, dimension( lda, * ) c, complex*16, dimension( lda, * ) cc, complex*16, dimension( lda, * ) q, integer lda, complex*16, dimension( * ) tau, complex*16, dimension( lwork ) work, integer lwork, double precision, dimension( * ) rwork, double precision, dimension( * ) result)

ZQLT03

الغرض:

!>
!> ZQLT03 يختبر ZUNMQL، الذي يحسب Q*C، Q'*C، C*Q أو C*Q'.
!>
!> ZQLT03 يقارن نتائج استدعاء ZUNMQL مع نتائج
!> تشكيل Q صراحةً باستدعاء ZUNGQL ثم إجراء الضرب المصفوفي
!> باستدعاء ZGEMM.
!>

المعاملات

M 
!>          M هو عدد صحيح (INTEGER)
!>          رتبة المصفوفة المتعامدة Q.  M >= 0.
!>

N

!>          N هو عدد صحيح (INTEGER)
!>          عدد الصفوف أو الأعمدة للمصفوفة C؛ C هي m × n إذا
!>          تم تطبيق Q من اليسار، أو n × m إذا تم تطبيق Q من
!>          اليمين.  N >= 0.
!>

K

!>          K هو عدد صحيح (INTEGER)
!>          عدد المنعكسات الأولية التي يُعرّف حاصل ضربها
!>          المصفوفة المتعامدة Q.  M >= K >= 0.
!>

AF

!>          AF هي مصفوفة COMPLEX*16، حجمها (LDA,N)
!>          تفاصيل عامل QL لمصفوفة m × n، كما
!>          تم إرجاعها بواسطة ZGEQLF. انظر CGEQLF لمزيد من التفاصيل.
!>

C

!>          C هي مصفوفة COMPLEX*16، حجمها (LDA,N)
!>

CC

!>          CC هي مصفوفة COMPLEX*16، حجمها (LDA,N)
!>

Q

!>          Q هي مصفوفة COMPLEX*16، حجمها (LDA,M)
!>

LDA

!>          LDA هو عدد صحيح (INTEGER)
!>          البعد الرائد للمصفوفات AF، C، CC، و Q.
!>

TAU

!>          TAU هي مصفوفة COMPLEX*16، حجمها (min(M,N))
!>          عوامل القياس للمنعكسات الأولية المقابلة
!>          لعامل QL في AF.
!>

WORK

!>          WORK هي مصفوفة COMPLEX*16، حجمها (LWORK)
!>

LWORK

!>          LWORK هو عدد صحيح (INTEGER)
!>          طول WORK.  يجب أن يكون LWORK على الأقل M، ويجب أن يكون
!>          M*NB، حيث NB هو حجم الكتلة لهذه البيئة.
!>

RWORK

!>          RWORK هي مصفوفة DOUBLE PRECISION، حجمها (M)
!>

RESULT

!>          RESULT هي مصفوفة DOUBLE PRECISION، حجمها (4)
!>          تقارن نسب الاختبار بين تقنيتين لضرب مصفوفة عشوائية C
!>          في مصفوفة متعامدة m × m (Q).
!>          RESULT(1) = norm( Q*C - Q*C )  / ( M * norm(C) * EPS )
!>          RESULT(2) = norm( C*Q - C*Q )  / ( M * norm(C) * EPS )
!>          RESULT(3) = norm( Q'*C - Q'*C )/ ( M * norm(C) * EPS )
!>          RESULT(4) = norm( C*Q' - C*Q' )/ ( M * norm(C) * EPS )
!>
