توثيق الوظيفة/الروتين الفرعي

subroutine zhsein (character side, character eigsrc, character initv, logical, dimension( * ) select, integer n, complex*16, dimension( ldh, * ) h, integer ldh, complex*16, dimension( * ) w, complex*16, dimension( ldvl, * ) vl, integer ldvl, complex*16, dimension( ldvr, * ) vr, integer ldvr, integer mm, integer m, complex*16, dimension( * ) work, double precision, dimension( * ) rwork, integer, dimension( * ) ifaill, integer, dimension( * ) ifailr, integer info)

ZHSEIN

الغرض:

!>
!> يستخدم ZHSEIN التكرار العكسي للعثور على المتجهات الذاتية اليمنى و/أو اليسرى المحددة
!> للمصفوفة هسنبرغ العلوية المعقدة H.
!>
!> المتجه الذاتي الأيمن x والمتجه الذاتي الأيسر y للمصفوفة H
!> المقابل لقيمة ذاتية w معرفة بواسطة:
!>
!>              H * x = w * x,     y**h * H = w * y**h
!>
!> حيث تشير y**h إلى منقولة y المقترنة.
!>

المعلمات

SIDE 
!>          SIDE هي CHARACTER*1
!>          = 'R': حساب المتجهات الذاتية اليمنى فقط؛
!>          = 'L': حساب المتجهات الذاتية اليسرى فقط؛
!>          = 'B': حساب المتجهات الذاتية اليمنى واليسرى.
!>

EIGSRC

!>          EIGSRC هي CHARACTER*1
!>          تحدد مصدر القيم الذاتية المقدمة في W:
!>          = 'Q': تم العثور على القيم الذاتية باستخدام ZHSEQR؛ لذلك، إذا
!>                 كانت H تحتوي على عناصر قطر فرعي صفرية، وبالتالي هي
!>                 مصفوفة مثلثية كتلية، فيمكن افتراض أن القيمة الذاتية j
!>                 هي قيمة ذاتية للكتلة التي تحتوي على الصف/العمود j.
!>                 تسمح هذه الخاصية لـ ZHSEIN بإجراء تكرار عكسي على كتلة قطرية واحدة فقط.
!>          = 'N': لا توجد افتراضات حول الارتباط
!>                 بين القيم الذاتية والكتل القطرية. في هذه
!>                 الحالة، يجب على ZHSEIN دائمًا إجراء تكرار عكسي
!>                 باستخدام المصفوفة H بأكملها.
!>

INITV

!>          INITV هي CHARACTER*1
!>          = 'N': لم يتم تقديم متجهات أولية؛
!>          = 'U': تم تقديم متجهات أولية من قبل المستخدم في المصفوفات
!>                 VL و/أو VR.
!>

SELECT

!>          SELECT هي مصفوفة LOGICAL، بحجم (N)
!>          تحدد المتجهات الذاتية التي سيتم حسابها. لتحديد
!>          المتجه الذاتي المقابل للقيمة الذاتية W(j)،
!>          يجب تعيين SELECT(j) إلى .TRUE..
!>

N

!>          N هو INTEGER
!>          ترتيب المصفوفة H. N >= 0.
!>

H

!>          H هي مصفوفة COMPLEX*16، بحجم (LDH,N)
!>          مصفوفة هسنبرغ العلوية H.
!>          إذا تم اكتشاف NaN في H، فستعود الروتين مع INFO=-6.
!>

LDH

!>          LDH هو INTEGER
!>          البعد الرائد للمصفوفة H. LDH >= max(1,N).
!>

W

!>          W هي مصفوفة COMPLEX*16، بحجم (N)
!>          عند الدخول، القيم الذاتية لـ H.
!>          عند الخروج، قد تم تغيير الأجزاء الحقيقية من W لأن
!>          القيم الذاتية القريبة يتم اضطرابها قليلاً في البحث عن
!>          متجهات ذاتية مستقلة.
!>

VL

!>          VL هي مصفوفة COMPLEX*16، بحجم (LDVL,MM)
!>          عند الدخول، إذا كانت INITV = 'U' و SIDE = 'L' أو 'B'، يجب أن
!>          تحتوي VL على متجهات بداية للتكرار العكسي للمتجهات الذاتية
!>          اليسرى؛ يجب أن يكون المتجه البادئ لكل متجه ذاتي
!>          في نفس العمود الذي سيتم تخزين المتجه الذاتي فيه.
!>          عند الخروج، إذا كانت SIDE = 'L' أو 'B'، سيتم تخزين المتجهات الذاتية
!>          اليسرى المحددة بواسطة SELECT بالتتابع في
!>          أعمدة VL، بنفس ترتيب قيمها الذاتية.
!>          إذا كانت SIDE = 'R'، لا يتم استخدام VL.
!>

LDVL

!>          LDVL هو INTEGER
!>          البعد الرائد للمصفوفة VL.
!>          LDVL >= max(1,N) إذا كانت SIDE = 'L' أو 'B'؛ LDVL >= 1 خلاف ذلك.
!>

VR

!>          VR هي مصفوفة COMPLEX*16، بحجم (LDVR,MM)
!>          عند الدخول، إذا كانت INITV = 'U' و SIDE = 'R' أو 'B'، يجب أن
!>          تحتوي VR على متجهات بداية للتكرار العكسي للمتجهات الذاتية
!>          اليمنى؛ يجب أن يكون المتجه البادئ لكل متجه ذاتي
!>          في نفس العمود الذي سيتم تخزين المتجه الذاتي فيه.
!>          عند الخروج، إذا كانت SIDE = 'R' أو 'B'، سيتم تخزين المتجهات الذاتية
!>          اليمنى المحددة بواسطة SELECT بالتتابع في
!>          أعمدة VR، بنفس ترتيب قيمها الذاتية.
!>          إذا كانت SIDE = 'L'، لا يتم استخدام VR.
!>

LDVR

!>          LDVR هو INTEGER
!>          البعد الرائد للمصفوفة VR.
!>          LDVR >= max(1,N) إذا كانت SIDE = 'R' أو 'B'؛ LDVR >= 1 خلاف ذلك.
!>

MM

!>          MM هو INTEGER
!>          عدد الأعمدة في المصفوفات VL و/أو VR. MM >= M.
!>

M

!>          M هو INTEGER
!>          عدد الأعمدة في المصفوفات VL و/أو VR المطلوبة لتخزين
!>          المتجهات الذاتية (= عدد العناصر .TRUE. في
!>          SELECT).
!>

WORK

!>          WORK هي مصفوفة COMPLEX*16، بحجم (N*N)
!>

RWORK

!>          RWORK هي مصفوفة DOUBLE PRECISION، بحجم (N)
!>

IFAILL

!>          IFAILL هي مصفوفة INTEGER، بحجم (MM)
!>          إذا كانت SIDE = 'L' أو 'B'، فإن IFAILL(i) = j > 0 إذا فشل المتجه الذاتي
!>          الأيسر في العمود i من VL (المقابل للقيمة الذاتية
!>          w(j)) في التقارب؛ IFAILL(i) = 0 إذا تقارب المتجه الذاتي بشكل مرضٍ.
!>          إذا كانت SIDE = 'R'، لا يتم استخدام IFAILL.
!>

IFAILR

!>          IFAILR هي مصفوفة INTEGER، بحجم (MM)
!>          إذا كانت SIDE = 'R' أو 'B'، فإن IFAILR(i) = j > 0 إذا فشل المتجه الذاتي
!>          الأيمن في العمود i من VR (المقابل للقيمة الذاتية
!>          w(j)) في التقارب؛ IFAILR(i) = 0 إذا تقارب المتجه الذاتي بشكل مرضٍ.
!>          إذا كانت SIDE = 'L'، لا يتم استخدام IFAILR.
!>

INFO

!>          INFO هو INTEGER
!>          = 0:  إنهاء ناجح
!>          < 0:  إذا كانت INFO = -i، فإن المعلمة i كانت ذات قيمة غير قانونية
!>          > 0:  إذا كانت INFO = i، فإن i هو عدد المتجهات الذاتية التي
!>                فشلت في التقارب؛ انظر IFAILL و IFAILR لمزيد من
!>                التفاصيل.
!>
