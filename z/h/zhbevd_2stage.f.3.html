توثيق الوظيفة/الإجراء الفرعي

subroutine zhbevd_2stage (character jobz, character uplo, integer n, integer kd, complex*16, dimension( ldab, * ) ab, integer ldab, double precision, dimension( * ) w, complex*16, dimension( ldz, * ) z, integer ldz, complex*16, dimension( * ) work, integer lwork, double precision, dimension( * ) rwork, integer lrwork, integer, dimension( * ) iwork, integer liwork, integer info)

ZHBEVD_2STAGE يحسب القيم الذاتية، واختيارياً، المتجهات الذاتية اليسرى و/أو اليمنى لمصفوفات هرميتية ذات نطاق معقد A باستخدام تقنية المرحلتين

الغرض:

!>
!> ZHBEVD_2STAGE يحسب جميع القيم الذاتية، واختيارياً، المتجهات الذاتية لمصفوفة نطاق هرميتية معقدة A باستخدام تقنية المرحلتين للاختزال إلى مصفوفة ثلاثية الأقطار. إذا تم طلب المتجهات الذاتية، فإنها تستخدم خوارزمية تقسيم وحل.
!>
!>

المعلمات

JOBZ 
!>          JOBZ is CHARACTER*1
!>          = 'N':  حساب القيم الذاتية فقط؛
!>          = 'V':  حساب القيم الذاتية والمتجهات الذاتية.
!>                  غير متوفر في هذا الإصدار.
!>

UPLO

!>          UPLO is CHARACTER*1
!>          = 'U':  يتم تخزين المثلث العلوي للمصفوفة A؛
!>          = 'L':  يتم تخزين المثلث السفلي للمصفوفة A.
!>

N

!>          N is INTEGER
!>          رتبة المصفوفة A. N >= 0.
!>

KD

!>          KD is INTEGER
!>          عدد الأقطار الفائقة للمصفوفة A إذا كان UPLO = 'U'،
!>          أو عدد الأقطار الفرعية إذا كان UPLO = 'L'. KD >= 0.
!>

AB

!>          AB is COMPLEX*16 array, dimension (LDAB, N)
!>          عند الإدخال، المثلث العلوي أو السفلي للمصفوفة الهرميتية ذات النطاق
!>          A، مخزن في أول KD+1 صف من المصفوفة. العمود j للمصفوفة A
!>          مخزن في العمود j للمصفوفة AB كما يلي:
!>          إذا كان UPLO = 'U'، فإن AB(kd+1+i-j,j) = A(i,j) حيث max(1,j-kd)<=i<=j؛
!>          إذا كان UPLO = 'L'، فإن AB(1+i-j,j)    = A(i,j) حيث j<=i<=min(n,j+kd).
!>
!>          عند الخروج، يتم استبدال AB بقيم تم إنشاؤها أثناء
!>          الاختزال إلى شكل ثلاثي الأقطار. إذا كان UPLO = 'U'، فإن القطر الفائق
!>          الأول والقطر الرئيسي للمصفوفة ثلاثية الأقطار T
!>          يتم إرجاعهما في الصفين KD و KD+1 من AB، وإذا كان UPLO = 'L'،
!>          يتم إرجاع القطر الرئيسي والقطر الفائق الأول لـ T في
!>          الصفين الأولين من AB.
!>

LDAB

!>          LDAB is INTEGER
!>          البعد الرائد للمصفوفة AB. LDAB >= KD + 1.
!>

W

!>          W is DOUBLE PRECISION array, dimension (N)
!>          إذا كان INFO = 0، فالقيم الذاتية بترتيب تصاعدي.
!>

Z

!>          Z is COMPLEX*16 array, dimension (LDZ, N)
!>          إذا كان JOBZ = 'V'، فعندما يكون INFO = 0، تحتوي Z على المتجهات الذاتية
!>          المتعامدة للمصفوفة A، مع احتواء العمود i من Z
!>          على المتجه الذاتي المرتبط بـ W(i).
!>          إذا كان JOBZ = 'N'، فلن يتم استخدام Z.
!>

LDZ

!>          LDZ is INTEGER
!>          البعد الرائد للمصفوفة Z. LDZ >= 1، وإذا كان
!>          JOBZ = 'V'، فإن LDZ >= max(1,N).
!>

WORK

!>          WORK is COMPLEX*16 array, dimension (MAX(1,LWORK))
!>          عند الخروج، إذا كان INFO = 0، فإن WORK(1) ترجع LWORK المثلى.
!>

LWORK

!>          LWORK is INTEGER
!>          طول المصفوفة WORK. LWORK >= 1، عندما تكون N <= 1؛
!>          بخلاف ذلك
!>          إذا كان JOBZ = 'N' و N > 1، يجب الاستعلام عن LWORK.
!>                                   LWORK = MAX(1, dimension) حيث
!>                                   dimension = (2KD+1)*N + KD*NTHREADS
!>                                   حيث KD هو حجم النطاق.
!>                                   NTHREADS هو عدد الخيوط المستخدمة عند
!>                                   تم تمكين الترجمة المفتوحة لـ OpenMP، وإلا = 1.
!>          إذا كان JOBZ = 'V' و N > 1، يجب الاستعلام عن LWORK. غير متوفر بعد.
!>
!>          إذا كان LWORK = -1، فسيتم افتراض استعلام عن مساحة العمل؛ يقوم الإجراء
!>          بحساب أحجام مساحات العمل المثلى لمصفوفات WORK، RWORK، و
!>          IWORK فقط، ويعيد هذه القيم كمدخلات أولى
!>          لمصفوفات WORK، RWORK، و IWORK، ولا يتم إصدار أي رسالة خطأ
!>          متعلقة بـ LWORK أو LRWORK أو LIWORK بواسطة XERBLA.
!>

RWORK

!>          RWORK is DOUBLE PRECISION array,
!>                                         dimension (LRWORK)
!>          عند الخروج، إذا كان INFO = 0، فإن RWORK(1) ترجع LRWORK المثلى.
!>

LRWORK

!>          LRWORK is INTEGER
!>          بعد المصفوفة RWORK.
!>          إذا كان N <= 1، فيجب أن يكون LRWORK على الأقل 1.
!>          إذا كان JOBZ = 'N' و N > 1، فيجب أن يكون LRWORK على الأقل N.
!>          إذا كان JOBZ = 'V' و N > 1، فيجب أن يكون LRWORK على الأقل
!>                        1 + 5*N + 2*N**2.
!>
!>          إذا كان LRWORK = -1، فسيتم افتراض استعلام عن مساحة العمل؛ يقوم الإجراء
!>          بحساب أحجام مساحات العمل المثلى لمصفوفات WORK، RWORK، و
!>          IWORK فقط، ويعيد هذه القيم كمدخلات أولى
!>          لمصفوفات WORK، RWORK، و IWORK، ولا يتم إصدار أي رسالة خطأ
!>          متعلقة بـ LWORK أو LRWORK أو LIWORK بواسطة XERBLA.
!>

IWORK

!>          IWORK is INTEGER array, dimension (MAX(1,LIWORK))
!>          عند الخروج، إذا كان INFO = 0، فإن IWORK(1) ترجع LIWORK المثلى.
!>

LIWORK

!>          LIWORK is INTEGER
!>          بعد المصفوفة IWORK.
!>          إذا كان JOBZ = 'N' أو N <= 1، فيجب أن يكون LIWORK على الأقل 1.
!>          إذا كان JOBZ = 'V' و N > 1، فيجب أن يكون LIWORK على الأقل 3 + 5*N .
!>
!>          إذا كان LIWORK = -1، فسيتم افتراض استعلام عن مساحة العمل؛ يقوم الإجراء
!>          بحساب أحجام مساحات العمل المثلى لمصفوفات WORK، RWORK، و
!>          IWORK فقط، ويعيد هذه القيم كمدخلات أولى
!>          لمصفوفات WORK، RWORK، و IWORK، ولا يتم إصدار أي رسالة خطأ
!>          متعلقة بـ LWORK أو LRWORK أو LIWORK بواسطة XERBLA.
!>

INFO

!>          INFO is INTEGER
!>          = 0:  انتهاء ناجح.
!>          < 0:  إذا كان INFO = -i، فإن الوسيط i كان له قيمة غير قانونية.
!>          > 0:  إذا كان INFO = i، فشلت الخوارزمية في التقارب؛ i
!>                العناصر خارج القطر الرئيسي لمصفوفة ثلاثية الأقطار وسيطة
!>                لم تتقارب إلى الصفر.
!>
