subroutine zhbgv (character jobz, character uplo, integer n, integer ka, integer kb, complex*16, dimension( ldab, * ) ab, integer ldab, complex*16, dimension( ldbb, * ) bb, integer ldbb, double precision, dimension( * ) w, complex*16, dimension( ldz, * ) z, integer ldz, complex*16, dimension( * ) work, double precision, dimension( * ) rwork, integer info)

ZHBGV

الغرض:

!>
!> ZHBGV يحسب جميع القيم الذاتية، واختياريًا، المتجهات الذاتية
!> لمشكلة قيمة ذاتية هرميتية محددة إيجابية معقدة، من
!> الشكل A*x=(lambda)*B*x. هنا يُفترض أن A و B هرميتيتان
!> ونطاقيتان، و B أيضًا موجبة محددة.
!>

المعلمات

JOBZ 
!>          JOBZ هو CHARACTER*1
!>          = 'N':  حساب القيم الذاتية فقط؛
!>          = 'V':  حساب القيم الذاتية والمتجهات الذاتية.
!>

UPLO

!>          UPLO هو CHARACTER*1
!>          = 'U':  يتم تخزين المثلثات العلوية لـ A و B؛
!>          = 'L':  يتم تخزين المثلثات السفلية لـ A و B.
!>

N

!>          N هو INTEGER
!>          رتبة المصفوفتين A و B.  N >= 0.
!>

KA

!>          KA هو INTEGER
!>          عدد الأقطار الفائقة للمصفوفة A إذا كانت UPLO = 'U'،
!>          أو عدد الأقطار الفرعية إذا كانت UPLO = 'L'. KA >= 0.
!>

KB

!>          KB هو INTEGER
!>          عدد الأقطار الفائقة للمصفوفة B إذا كانت UPLO = 'U'،
!>          أو عدد الأقطار الفرعية إذا كانت UPLO = 'L'. KB >= 0.
!>

AB

!>          AB هو مصفوفة COMPLEX*16، حجمها (LDAB, N)
!>          عند الإدخال، المثلث العلوي أو السفلي للمصفوفة النطاقية الهرميتية
!>          A، مخزنة في أول ka+1 صفوف من المصفوفة.
!>          العمود j من A مخزن في العمود j من المصفوفة AB
!>          على النحو التالي:
!>          إذا كانت UPLO = 'U'، فإن AB(ka+1+i-j,j) = A(i,j) لـ max(1,j-ka)<=i<=j؛
!>          إذا كانت UPLO = 'L'، فإن AB(1+i-j,j)    = A(i,j) لـ j<=i<=min(n,j+ka).
!>
!>          عند الخروج، يتم تدمير محتويات AB.
!>

LDAB

!>          LDAB هو INTEGER
!>          البعد القيادي للمصفوفة AB.  LDAB >= KA+1.
!>

BB

!>          BB هو مصفوفة COMPLEX*16، حجمها (LDBB, N)
!>          عند الإدخال، المثلث العلوي أو السفلي للمصفوفة النطاقية الهرميتية
!>          B، مخزنة في أول kb+1 صفوف من المصفوفة.
!>          العمود j من B مخزن في العمود j من المصفوفة BB
!>          على النحو التالي:
!>          إذا كانت UPLO = 'U'، فإن BB(kb+1+i-j,j) = B(i,j) لـ max(1,j-kb)<=i<=j؛
!>          إذا كانت UPLO = 'L'، فإن BB(1+i-j,j)    = B(i,j) لـ j<=i<=min(n,j+kb).
!>
!>          عند الخروج، العامل S من تفكيك تشوليسكي المنقسم
!>          B = S**H*S، كما تم إرجاعه بواسطة ZPBSTF.
!>

LDBB

!>          LDBB هو INTEGER
!>          البعد القيادي للمصفوفة BB.  LDBB >= KB+1.
!>

W

!>          W هي مصفوفة DOUBLE PRECISION، حجمها (N)
!>          إذا كان INFO = 0، فإن القيم الذاتية مرتبة تصاعديًا.
!>

Z

!>          Z هي مصفوفة COMPLEX*16، حجمها (LDZ, N)
!>          إذا كانت JOBZ = 'V'، فعندما يكون INFO = 0، تحتوي Z على المصفوفة Z للمتجهات
!>          الذاتية، حيث يحمل العمود i من Z المتجه
!>          الذاتي المرتبط بـ W(i). يتم تطبيع المتجهات الذاتية
!>          بحيث Z**H*B*Z = I.
!>          إذا كانت JOBZ = 'N'، فلن يتم الرجوع إلى Z.
!>

LDZ

!>          LDZ هو INTEGER
!>          البعد القيادي للمصفوفة Z.  LDZ >= 1، وإذا
!>          كانت JOBZ = 'V'، فإن LDZ >= N.
!>

WORK

!>          WORK هي مصفوفة COMPLEX*16، حجمها (N)
!>

RWORK

!>          RWORK هي مصفوفة DOUBLE PRECISION، حجمها (3*N)
!>

INFO

!>          INFO هو INTEGER
!>          = 0:  انتهاء ناجح
!>          < 0:  إذا كان INFO = -i، فإن الوسيط i كان له قيمة غير قانونية
!>          > 0:  إذا كان INFO = i، وكان i هو:
!>             <= N:  فشلت الخوارزمية في التقارب:
!>                    i عنصر خارج القطر في شكل ثلاثي الأقطار وسيط
!>                    لم يتقارب إلى الصفر؛
!>             > N:   إذا كان INFO = N + i، لـ 1 <= i <= N، فإن ZPBSTF
!>                    أرجعت INFO = i: B ليست موجبة محددة.
!>                    لم يتمكن تفكيك B من الاكتمال و
!>                    لم يتم حساب أي قيم ذاتية أو متجهات ذاتية.
!>
