subroutine zhetri (character uplo, integer n, complex*16, dimension( lda, * ) a, integer lda, integer, dimension( * ) ipiv, complex*16, dimension( * ) work, integer info)

ZHETRI

الغرض:

!>
!> ZHETRI تحسب معكوس مصفوفة هرميتية غير محددة معقدة A باستخدام التحليل A = U*D*U**H أو A = L*D*L**H الذي تم حسابه بواسطة
!> ZHETRF.
!>

المعلمات

UPLO 
!>          UPLO is CHARACTER*1
!>          يحدد ما إذا كانت تفاصيل التحليل مخزنة
!>          كمصفوفة مثلثة علوية أو سفلية.
!>          = 'U':  مثلثة علوية، الشكل هو A = U*D*U**H؛
!>          = 'L':  مثلثة سفلية، الشكل هو A = L*D*L**H.
!>

N

!>          N is INTEGER
!>          ترتيب المصفوفة A.  N >= 0.
!>

A

!>          A is COMPLEX*16 array, dimension (LDA,N)
!>          عند الإدخال، مصفوفة القطر الجزئي D والمعاملات
!>          المستخدمة للحصول على العامل U أو L كما تم حسابه بواسطة ZHETRF.
!>
!>          عند الخروج، إذا كان INFO = 0، فإن معكوس المصفوفة الأصلية (الهرميتية).
!>          إذا كان UPLO = 'U'، يتم تشكيل الجزء المثلث العلوي من المعكوس ولا
!>          يتم الرجوع إلى الجزء السفلي من A؛ إذا كان UPLO = 'L' يتم تشكيل الجزء
!>          المثلث السفلي من المعكوس ولا يتم الرجوع إلى الجزء العلوي من A.
!>

LDA

!>          LDA is INTEGER
!>          البعد الرائد للمصفوفة A.  LDA >= max(1,N).
!>

IPIV

!>          IPIV is INTEGER array, dimension (N)
!>          تفاصيل التبادلات وهيكل Q للمصفوفة D
!>          كما تم تحديده بواسطة ZHETRF.
!>

WORK

!>          WORK is COMPLEX*16 array, dimension (N)
!>

INFO

!>          INFO is INTEGER
!>          = 0: خروج ناجح
!>          < 0: إذا كان INFO = -i، فإن المعلمة i كانت ذات قيمة غير قانونية
!>          > 0: إذا كان INFO = i، فإن D(i,i) = 0؛ المصفوفة مفردة ولا يمكن حساب معكوسها.
!>
