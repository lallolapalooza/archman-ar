وثائق الوظيفة/الروتين الفرعي

subroutine zhsein (character side, character eigsrc, character initv, logical, dimension( * ) select, integer n, complex*16, dimension( ldh, * ) h, integer ldh, complex*16, dimension( * ) w, complex*16, dimension( ldvl, * ) vl, integer ldvl, complex*16, dimension( ldvr, * ) vr, integer ldvr, integer mm, integer m, complex*16, dimension( * ) work, double precision, dimension( * ) rwork, integer, dimension( * ) ifaill, integer, dimension( * ) ifailr, integer info)

ZHSEIN

الغرض:

!>
!> يستخدم ZHSEIN التكرار العكسي لإيجاد متجهات ذاتية يمينية و/أو يسارية محددة
!> لمصفوفة هسنبرغ عليا معقدة H.
!>
!> المتجه الذاتي الأيمن x والمتجه الذاتي الأيسر y للمصفوفة H
!> المقابلين لقيمة ذاتية w يعرفان بالعلاقة:
!>
!>              H * x = w * x,     y**h * H = w * y**h
!>
!> حيث تمثل y**h منقول المتجه y المصاحب.
!>

المعلمات

SIDE 
!>          SIDE هو CHARACTER*1
!>          = 'R': حساب المتجهات الذاتية اليمنى فقط؛
!>          = 'L': حساب المتجهات الذاتية اليسرى فقط؛
!>          = 'B': حساب المتجهات الذاتية اليمنى واليسرى معاً.
!>

EIGSRC

!>          EIGSRC هو CHARACTER*1
!>          يحدد مصدر القيم الذاتية المقدمة في W:
!>          = 'Q': تم العثور على القيم الذاتية باستخدام ZHSEQR؛ بالتالي، إذا
!>                 كانت H تحتوي على عناصر دون قطرية صفرية، وبالتالي فهي
!>                 مصفوفة ثلاثية قطاعات، فيمكن افتراض أن القيمة الذاتية j
!>                 هي قيمة ذاتية للقطاع الذي يحتوي على الصف/العمود j.
!>                 تسمح هذه الخاصية لـ ZHSEIN بتنفيذ التكرار العكسي على قطاع قطري واحد فقط.
!>          = 'N': لا توجد افتراضات حول العلاقة
!>                 بين القيم الذاتية والقطاعات القطرية. في هذه
!>                 الحالة، يجب على ZHSEIN دائمًا إجراء تكرار عكسي
!>                 باستخدام المصفوفة H بأكملها.
!>

INITV

!>          INITV هو CHARACTER*1
!>          = 'N': لم يتم تقديم متجهات أولية؛
!>          = 'U': تم تخزين المتجهات الأولية التي قدمها المستخدم في المصفوفات
!>                 VL و/أو VR.
!>

SELECT

!>          SELECT هي مصفوفة LOGICAL، أبعادها (N)
!>          يحدد المتجهات الذاتية التي سيتم حسابها. لاختيار
!>          المتجه الذاتي المقابل للقيمة الذاتية W(j)،
!>          يجب تعيين SELECT(j) إلى .TRUE..
!>

N

!>          N هو INTEGER
!>          رتبة المصفوفة H. N >= 0.
!>

H

!>          H هي مصفوفة COMPLEX*16، أبعادها (LDH,N)
!>          مصفوفة هسنبرغ العليا H.
!>          إذا تم اكتشاف قيمة NaN في H، فسوف تعود الروتين بـ INFO=-6.
!>

LDH

!>          LDH هو INTEGER
!>          البعد الرائد للمصفوفة H. LDH >= max(1,N).
!>

W

!>          W هي مصفوفة COMPLEX*16، أبعادها (N)
!>          عند الإدخال، القيم الذاتية لـ H.
!>          عند الخروج، قد تكون الأجزاء الحقيقية من W قد تغيرت لأن
!>          القيم الذاتية القريبة تتغير قليلاً عند البحث عن
!>          متجهات ذاتية مستقلة.
!>

VL

!>          VL هي مصفوفة COMPLEX*16، أبعادها (LDVL,MM)
!>          عند الإدخال، إذا كان INITV = 'U' و SIDE = 'L' أو 'B'، فيجب أن تحتوي VL
!>          على متجهات بداية للتكرار العكسي للمتجهات الذاتية اليسرى؛ يجب أن يكون المتجه
!>          البادئ لكل متجه ذاتي في نفس العمود الذي سيتم تخزين المتجه الذاتي فيه.
!>          عند الخروج، إذا كان SIDE = 'L' أو 'B'، فسيتم تخزين المتجهات الذاتية اليسرى
!>          المحددة بواسطة SELECT بشكل متتابع في أعمدة VL، بنفس ترتيب القيم الذاتية الخاصة بها.
!>          إذا كان SIDE = 'R'، فلن يتم الرجوع إلى VL.
!>

LDVL

!>          LDVL هو INTEGER
!>          البعد الرائد للمصفوفة VL.
!>          LDVL >= max(1,N) إذا كان SIDE = 'L' أو 'B'؛ LDVL >= 1 بخلاف ذلك.
!>

VR

!>          VR هي مصفوفة COMPLEX*16، أبعادها (LDVR,MM)
!>          عند الإدخال، إذا كان INITV = 'U' و SIDE = 'R' أو 'B'، فيجب أن تحتوي VR
!>          على متجهات بداية للتكرار العكسي للمتجهات الذاتية اليمنى؛ يجب أن يكون المتجه
!>          البادئ لكل متجه ذاتي في نفس العمود الذي سيتم تخزين المتجه الذاتي فيه.
!>          عند الخروج، إذا كان SIDE = 'R' أو 'B'، فسيتم تخزين المتجهات الذاتية اليمنى
!>          المحددة بواسطة SELECT بشكل متتابع في أعمدة VR، بنفس ترتيب القيم الذاتية الخاصة بها.
!>          إذا كان SIDE = 'L'، فلن يتم الرجوع إلى VR.
!>

LDVR

!>          LDVR هو INTEGER
!>          البعد الرائد للمصفوفة VR.
!>          LDVR >= max(1,N) إذا كان SIDE = 'R' أو 'B'؛ LDVR >= 1 بخلاف ذلك.
!>

MM

!>          MM هو INTEGER
!>          عدد الأعمدة في مصفوفتي VL و/أو VR. MM >= M.
!>

M

!>          M هو INTEGER
!>          عدد الأعمدة في مصفوفتي VL و/أو VR المطلوبة لتخزين المتجهات الذاتية (= عدد العناصر .TRUE. في
!>          SELECT).
!>

WORK

!>          WORK هي مصفوفة COMPLEX*16، أبعادها (N*N)
!>

RWORK

!>          RWORK هي مصفوفة DOUBLE PRECISION، أبعادها (N)
!>

IFAILL

!>          IFAILL هي مصفوفة INTEGER، أبعادها (MM)
!>          إذا كان SIDE = 'L' أو 'B'، فإن IFAILL(i) = j > 0 إذا فشل المتجه الذاتي الأيسر
!>          في العمود i من VL (المقابل للقيمة الذاتية w(j)) في التقارب؛ IFAILL(i) = 0 إذا
!>          تقارب المتجه الذاتي بشكل مرضٍ.
!>          إذا كان SIDE = 'R'، فلن يتم الرجوع إلى IFAILL.
!>

IFAILR

!>          IFAILR هي مصفوفة INTEGER، أبعادها (MM)
!>          إذا كان SIDE = 'R' أو 'B'، فإن IFAILR(i) = j > 0 إذا فشل المتجه الذاتي الأيمن
!>          في العمود i من VR (المقابل للقيمة الذاتية w(j)) في التقارب؛ IFAILR(i) = 0 إذا
!>          تقارب المتجه الذاتي بشكل مرضٍ.
!>          إذا كان SIDE = 'L'، فلن يتم الرجوع إلى IFAILR.
!>

INFO

!>          INFO هو INTEGER
!>          = 0:  انتهاء ناجح
!>          < 0:  إذا كان INFO = -i، فإن المعامل i كان لديه قيمة غير قانونية
!>          > 0:  إذا كان INFO = i، فإن i هو عدد المتجهات الذاتية التي
!>                فشلت في التقارب؛ انظر IFAILL و IFAILR لمزيد من
!>                التفاصيل.
!>
