subroutine zhpsvx (character fact, character uplo, integer n, integer nrhs, complex*16, dimension( * ) ap, complex*16, dimension( * ) afp, integer, dimension( * ) ipiv, complex*16, dimension( ldb, * ) b, integer ldb, complex*16, dimension( ldx, * ) x, integer ldx, double precision rcond, double precision, dimension( * ) ferr, double precision, dimension( * ) berr, complex*16, dimension( * ) work, double precision, dimension( * ) rwork, integer info)

ZHPSVX تحسب حل نظام المعادلات الخطية A * X = B للمصفوفات الأخرى

الغرض:

!>
!> تستخدم ZHPSVX تحليل التبديل القطري A = U*D*U**H أو
!> A = L*D*L**H لحساب حل نظام خطي معقد A * X = B، حيث A هي مصفوفة هيرميتية N×N مخزنة
!> بتنسيق مضغوط و X و B هما مصفوفتان N×NRHS.
!>
!> يتم أيضًا توفير حدود الخطأ للحل وتقدير للشرط.
!>

الوصف:

!>
!> يتم تنفيذ الخطوات التالية:
!>
!> 1. إذا كان FACT = 'N'، يتم استخدام طريقة التبديل القطري لتحليل A إلى
!>       A = U * D * U**H، إذا كان UPLO = 'U'، أو
!>       A = L * D * L**H، إذا كان UPLO = 'L'،
!>    حيث U (أو L) هو ناتج عن مصفوفات التبديل ومصفوفات مثلثية علوية (سفلية)
!>    ووحدة، و D هي مصفوفة هيرميتية وقطرية كتلة مع كتل قطرية 1×1 و 2×2.
!>
!> 2. إذا كان D(i,i)=0، بحيث تكون D مفردة تمامًا، فإن الإجراء الفرعي
!>    يعود بـ INFO = i. بخلاف ذلك، يتم استخدام الشكل المحلل لـ A
!>    لتقدير معامل شرط المصفوفة A. إذا كان مقلوب
!>    معامل الشرط أقل من دقة الآلة،
!>    يتم إرجاع INFO = N+1 كتحذير، ولكن الإجراء الفرعي يستمر
!>    في الحل لـ X وحساب حدود الخطأ كما هو موضح أدناه.
!>
!> 3. يتم حل نظام المعادلات لـ X باستخدام الشكل المحلل لـ A.
!>
!> 4. يتم تطبيق التنقيح التكراري لتحسين الحل المحسوب
!>    وحساب حدود الخطأ وتقديرات الخطأ الخلفي له.
!>

المعلمات

FACT 
!>          FACT هو CHARACTER*1
!>          يحدد ما إذا كان الشكل المحلل لـ A قد تم توفيره عند الإدخال أم لا.
!>          = 'F': عند الإدخال، تحتوي AFP و IPIV على الشكل المحلل لـ
!>                  A. لن يتم تعديل AFP و IPIV.
!>          = 'N': سيتم نسخ المصفوفة A إلى AFP وتحليلها.
!>

UPLO

!>          UPLO هو CHARACTER*1
!>          = 'U': يتم تخزين المثلث العلوي لـ A؛
!>          = 'L': يتم تخزين المثلث السفلي لـ A.
!>

N

!>          N هو INTEGER
!>          عدد المعادلات الخطية، أي رتبة المصفوفة A. N >= 0.
!>

NRHS

!>          NRHS هو INTEGER
!>          عدد الأطراف اليمنى، أي عدد أعمدة المصفوفتين B و X. NRHS >= 0.
!>

AP

!>          AP هو مصفوفة COMPLEX*16، البعد (N*(N+1)/2)
!>          المثلث العلوي أو السفلي للمصفوفة الهيرميتية A، مكدس
!>          حسب الأعمدة في مصفوفة خطية. يتم تخزين العمود j للمصفوفة A
!>          في المصفوفة AP كما يلي:
!>          إذا كان UPLO = 'U'، فإن AP(i + (j-1)*j/2) = A(i,j) لـ 1<=i<=j؛
!>          إذا كان UPLO = 'L'، فإن AP(i + (j-1)*(2*n-j)/2) = A(i,j) لـ j<=i<=n.
!>          انظر أدناه لمزيد من التفاصيل.
!>

AFP

!>          AFP هو مصفوفة COMPLEX*16، البعد (N*(N+1)/2)
!>          إذا كان FACT = 'F'، فإن AFP هو معلمة إدخال وعند الإدخال
!>          يحتوي على المصفوفة القطرية الكتلية D والمضاعفات المستخدمة
!>          للحصول على العامل U أو L من التحليل
!>          A = U*D*U**H أو A = L*D*L**H كما تم حسابه بواسطة ZHPTRF، مخزن كـ
!>          مصفوفة مثلثية مكدسة بنفس تنسيق التخزين مثل A.
!>
!>          إذا كان FACT = 'N'، فإن AFP هي معلمة إخراج وعند الخروج
!>          تحتوي على المصفوفة القطرية الكتلية D والمضاعفات المستخدمة
!>          للحصول على العامل U أو L من التحليل
!>          A = U*D*U**H أو A = L*D*L**H كما تم حسابه بواسطة ZHPTRF، مخزن كـ
!>          مصفوفة مثلثية مكدسة بنفس تنسيق التخزين مثل A.
!>

IPIV

!>          IPIV هو مصفوفة INTEGER، البعد (N)
!>          إذا كان FACT = 'F'، فإن IPIV هي معلمة إدخال وعند الإدخال
!>          تحتوي على تفاصيل التبادلات وهيكل الكتلة لـ D، كما تم تحديده بواسطة ZHPTRF.
!>          إذا كان IPIV(k) > 0، فقد تم تبديل الصفوف والأعمدة k و IPIV(k)
!>          و D(k,k) هي كتلة قطرية 1×1.
!>          إذا كان UPLO = 'U' و IPIV(k) = IPIV(k-1) < 0، فقد تم تبديل الصفوف والأعمدة k-1 و -IPIV(k)
!>          و D(k-1:k,k-1:k) هي كتلة قطرية 2×2. إذا كان UPLO = 'L' و IPIV(k) =
!>          IPIV(k+1) < 0، فقد تم تبديل الصفوف والأعمدة k+1 و -IPIV(k)
!>          و D(k:k+1,k:k+1) هي كتلة قطرية 2×2.
!>
!>          إذا كان FACT = 'N'، فإن IPIV هي معلمة إخراج وعند الخروج
!>          تحتوي على تفاصيل التبادلات وهيكل الكتلة لـ D، كما تم تحديده بواسطة ZHPTRF.
!>

B

!>          B هي مصفوفة COMPLEX*16، البعد (LDB,NRHS)
!>          المصفوفة B ذات الطرف الأيمن N×NRHS.
!>

LDB

!>          LDB هو INTEGER
!>          البعد الرئيسي للمصفوفة B. LDB >= max(1,N).
!>

X

!>          X هي مصفوفة COMPLEX*16، البعد (LDX,NRHS)
!>          إذا كان INFO = 0 أو INFO = N+1، فإن مصفوفة الحل X ذات البعد N×NRHS.
!>

LDX

!>          LDX هو INTEGER
!>          البعد الرئيسي للمصفوفة X. LDX >= max(1,N).
!>

RCOND

!>          RCOND هو DOUBLE PRECISION
!>          التقدير لمقلوب معامل الشرط للمصفوفة A. إذا كان RCOND أقل من دقة الآلة
!>          (على وجه الخصوص، إذا كان RCOND = 0)، فإن المصفوفة مفردة
!>          بنسبة دقة العمل. يتم الإشارة إلى هذا الشرط برمز عودة
!>          INFO > 0.
!>

FERR

!>          FERR هي مصفوفة DOUBLE PRECISION، البعد (NRHS)
!>          تقدير حد الخطأ الأمامي لكل متجه حل X(j) (العمود j للمصفوفة X).
!>          إذا كان XTRUE هو الحل الحقيقي المقابل لـ X(j)، فإن FERR(j)
!>          هو حد علوي مقدر لحجم أكبر عنصر في (X(j) - XTRUE) مقسومًا على حجم
!>          أكبر عنصر في X(j). التقدير موثوق به مثل
!>          التقدير لـ RCOND، وهو دائمًا تقريبًا تقدير زائد للخطأ الحقيقي.
!>

BERR

!>          BERR هي مصفوفة DOUBLE PRECISION، البعد (NRHS)
!>          الخطأ النسبي المكون من مكون واحد لحل كل متجه X(j) (أي، أصغر تغيير نسبي في
!>          أي عنصر من A أو B يجعل X(j) حلاً دقيقًا).
!>

WORK

!>          WORK هي مصفوفة COMPLEX*16، البعد (2*N)
!>

RWORK

!>          RWORK هي مصفوفة DOUBLE PRECISION، البعد (N)
!>

INFO

!>          INFO هو INTEGER
!>          = 0: خروج ناجح
!>          < 0: إذا كان INFO = -i، فإن

zhpsvx - Schur decomposition of a Hermitian matrix with iterative refinement

SUBROUTINE ZHPSVX(FACT, TRANS, N, NRHS, AB, LDA, IPIV, B, LDB, X, LDX,
                  RCOND, FERR, BERR, WORK, RWORK, INFO)
  CHARACTER(LEN=1) :: FACT, TRANS
  INTEGER :: INFO, LDA, LDB, LDB, N, NRHS
  INTEGER :: IPIV( * )
  REAL(KIND=8) :: RCOND, RWORK( * )
  COMPLEX(KIND=16) :: AB( * ), B( * ), WORK( * ), X( * )
  REAL(KIND=8) :: BERR( * ), FERR( * )

ZHPSVX uses the diagonal pivoting LU factorization to compute the solution to a system of linear equations.

Arguments

FACT
CHARACTER*1.

Specifies the form of the system of equations:

  = 'E': The LU factorization of A is supplied on the entry to the
         routine.
  = 'N': The LU factorization of A will be computed by the routine.
TRANS
CHARACTER*1.

Specifies the form of the system of equations:

  = 'N': Solve A * X = B         (No transpose)
  = 'C': Solve A' * X = B        (Conjugate transpose)
N
INTEGER.

The order of the matrix A. N >= 0.

NRHS
INTEGER.

The number of right-hand sides, i.e., the number of columns in the arrays B, X, FERR, and BERR. NRHS >= 0.

AB
COMPLEX*16 array, dimension (LDAB,N) if using standard storage, or (N*(N+1)/2) if using packed storage.

Before entry, if FACT = 'E', the main diagonal of AB must contain the diagonal elements of U, and the elements above the diagonal must contain the corresponding elements of U. The elements below the diagonal must contain the corresponding elements of L. For details of the packed storage format see Further Details.

LDA
INTEGER.

The leading dimension of the array AB when using standard storage. LDA >= max(1,N).

IPIV
INTEGER array, dimension (N).

Details of the pivoting.

B
COMPLEX*16 array, dimension (LDB,NRHS).

The right-hand side matrix B.

LDB
INTEGER.

The leading dimension of the array B. LDB >= max(1,N).

X
COMPLEX*16 array, dimension (LDX,NRHS).

Before entry, the array X may contain initial estimates of the solution matrix X. If FACT = 'E', then X must contain the matrix R computed by ZHPTRF.

LDX
INTEGER.

The leading dimension of the array X. LDX >= max(1,N).

RCOND
DOUBLE PRECISION.

The estimate of the reciprocal condition number of the matrix A. The scaling of the matrix is controlled by this parameter.

FERR
DOUBLE PRECISION array, dimension (NRHS).

The estimated forward error bound for each relative error in the solution matrix X.

BERR
DOUBLE PRECISION array, dimension (NRHS).

The estimated backward error bound for each relative error in the solution matrix X.

WORK
COMPLEX*16 array, dimension (2*N).
RWORK
DOUBLE PRECISION array, dimension (N).
INFO
INTEGER.

= 0: Successful exit. < 0: If INFO = -i, the i-th argument had an illegal value. > > 0: if INFO = i, and i is !> <= N: D(i,i) is exactly zero. The factorization !> has been completed but the factor D is exactly !> singular, so the solution and error bounds could !> not be computed. RCOND = 0 is returned. !> = N+1: D is nonsingular, but RCOND is less than machine !> precision, meaning that the matrix is singular !> to working precision. Nevertheless, the !> solution and error bounds are computed because !> there are a number of situations where the !> computed solution can be more accurate than the !> value of RCOND would suggest.
