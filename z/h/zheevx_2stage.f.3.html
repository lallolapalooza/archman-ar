subroutine zheevx_2stage (character jobz, character range, character uplo, integer n, complex*16, dimension( lda, * ) a, integer lda, double precision vl, double precision vu, integer il, integer iu, double precision abstol, integer m, double precision, dimension( * ) w, complex*16, dimension( ldz, * ) z, integer ldz, complex*16, dimension( * ) work, integer lwork, double precision, dimension( * ) rwork, integer, dimension( * ) iwork, integer, dimension( * ) ifail, integer info)

ZHEEVX_2STAGE يحسب القيم الذاتية، اختياريًا، المتجهات الذاتية اليسرى و/أو اليمنى لمصفوفات HE

الغرض:

!>
!> ZHEEVX_2STAGE يحسب القيم الذاتية المحددة، اختياريًا، والمتجهات الذاتية
!> لمصفوفة هرميتية معقدة A باستخدام تقنية المرحلتين (2stage)
!> للاختزال إلى مصفوفة ثلاثية الأقطار. يمكن تحديد القيم الذاتية والمتجهات الذاتية
!> عن طريق تحديد نطاق من القيم أو نطاق من
!> المؤشرات للقيم الذاتية المطلوبة.
!>

المعلمات

JOBZ 
!>          JOBZ is CHARACTER*1
!>          = 'N':  حساب القيم الذاتية فقط؛
!>          = 'V':  حساب القيم الذاتية والمتجهات الذاتية.
!>                  غير متاح في هذا الإصدار.
!>

RANGE

!>          RANGE is CHARACTER*1
!>          = 'A': سيتم العثور على جميع القيم الذاتية.
!>          = 'V': سيتم العثور على جميع القيم الذاتية في النطاق نصف المفتوح (VL,VU].
!>          = 'I': سيتم العثور على القيم الذاتية من IL إلى IU.
!>

UPLO

!>          UPLO is CHARACTER*1
!>          = 'U':  يتم تخزين الجزء العلوي من A؛
!>          = 'L':  يتم تخزين الجزء السفلي من A.
!>

N

!>          N is INTEGER
!>          رتبة المصفوفة A. N >= 0.
!>

A

!>          A is COMPLEX*16 array, dimension (LDA, N)
!>          عند الدخول، المصفوفة الهرميتية A. إذا كان UPLO = 'U'، فإن
!>          الجزء العلوي المثلثي من A بحجم N × N يحتوي على
!>          الجزء العلوي المثلثي للمصفوفة A. إذا كان UPLO = 'L'، فإن
!>          الجزء السفلي المثلثي من A بحجم N × N يحتوي على
!>          الجزء السفلي المثلثي للمصفوفة A.
!>          عند الخروج، يتم تدمير الجزء السفلي المثلثي (إذا كان UPLO='L') أو الجزء العلوي
!>          المثلثي (إذا كان UPLO='U') من A، بما في ذلك القطر.
!>

LDA

!>          LDA is INTEGER
!>          البعد الرئيسي للمصفوفة A. LDA >= max(1,N).
!>

VL

!>          VL is DOUBLE PRECISION
!>          إذا كان RANGE='V'، فإن الحد الأدنى للنطاق المراد
!>          البحث فيه عن القيم الذاتية. VL < VU.
!>          غير مستخدم إذا كان RANGE = 'A' أو 'I'.
!>

VU

!>          VU is DOUBLE PRECISION
!>          إذا كان RANGE='V'، فإن الحد الأعلى للنطاق المراد
!>          البحث فيه عن القيم الذاتية. VL < VU.
!>          غير مستخدم إذا كان RANGE = 'A' أو 'I'.
!>

IL

!>          IL is INTEGER
!>          إذا كان RANGE='I'، فإن مؤشر
!>          أصغر قيمة ذاتية سيتم إرجاعها.
!>          1 <= IL <= IU <= N، إذا كان N > 0؛ IL = 1 و IU = 0 إذا كان N = 0.
!>          غير مستخدم إذا كان RANGE = 'A' أو 'V'.
!>

IU

!>          IU is INTEGER
!>          إذا كان RANGE='I'، فإن مؤشر
!>          أكبر قيمة ذاتية سيتم إرجاعها.
!>          1 <= IL <= IU <= N، إذا كان N > 0؛ IL = 1 و IU = 0 إذا كان N = 0.
!>          غير مستخدم إذا كان RANGE = 'A' أو 'V'.
!>

ABSTOL

!>          ABSTOL is DOUBLE PRECISION
!>          تفاوت الخطأ المطلق للقيم الذاتية.
!>          يتم قبول قيمة ذاتية تقريبية على أنها متقاربة
!>          عندما يتم تحديد أنها تقع في فاصل [a,b]
!>          بعرض يساوي أو أقل من
!>
!>                  ABSTOL + EPS *   max( |a|,|b| ) ,
!>
!>          حيث EPS هي دقة الآلة. إذا كان ABSTOL أقل من
!>          أو يساوي صفرًا، فسيتم استخدام EPS*|T| بدلاً منه،
!>          حيث |T| هو معيار 1 للمصفوفة ثلاثية الأقطار التي تم الحصول عليها
!>          عن طريق اختزال A إلى شكل ثلاثي الأقطار.
!>
!>          سيتم حساب القيم الذاتية بأكبر دقة عندما يتم
!>          تعيين ABSTOL إلى ضعف حد التدفق السفلي 2*DLAMCH('S')، وليس صفرًا.
!>          إذا عادت هذه الوظيفة مع INFO>0، مما يشير إلى أن بعض
!>          المتجهات الذاتية لم تتقارب، حاول تعيين ABSTOL إلى
!>          2*DLAMCH('S').
!>
!>          انظر  بواسطة Demmel و
!>          Kahan، مذكرة عمل LAPACK #3.
!>

M

!>          M is INTEGER
!>          العدد الإجمالي للقيم الذاتية التي تم العثور عليها. 0 <= M <= N.
!>          إذا كان RANGE = 'A'، فإن M = N، وإذا كان RANGE = 'I'، فإن M = IU-IL+1.
!>

W

!>          W is DOUBLE PRECISION array, dimension (N)
!>          عند الخروج العادي، تحتوي العناصر M الأولى على القيم الذاتية المحددة
!>          مرتبة تصاعديًا.
!>

Z

!>          Z is COMPLEX*16 array, dimension (LDZ, max(1,M))
!>          إذا كان JOBZ = 'V'، فعندما يكون INFO = 0، تحتوي الأعمدة M الأولى من Z
!>          على المتجهات الذاتية المتعامدة للمصفوفة A
!>          المقابلة للقيم الذاتية المحددة، حيث يحمل العمود i
!>          من Z المتجه الذاتي المرتبط بـ W(i).
!>          إذا فشل متجه ذاتي في التقارب، فإن العمود المقابل من Z
!>          يحتوي على أحدث تقريب للمتجه الذاتي، ويتم إرجاع مؤشر
!>          المتجه الذاتي في IFAIL.
!>          إذا كان JOBZ = 'N'، فلن يتم استخدام Z.
!>          ملاحظة: يجب على المستخدم التأكد من توفير ما لا يقل عن max(1,M) أعمدة
!>          في المصفوفة Z؛ إذا كان RANGE = 'V'، فإن القيمة الدقيقة لـ M
!>          غير معروفة مسبقًا ويجب استخدام حد أعلى.
!>

LDZ

!>          LDZ is INTEGER
!>          البعد الرئيسي للمصفوفة Z. LDZ >= 1، وإذا
!>          كان JOBZ = 'V'، فإن LDZ >= max(1,N).
!>

WORK

!>          WORK is COMPLEX*16 array, dimension (MAX(1,LWORK))
!>          عند الخروج، إذا كان INFO = 0، فإن WORK(1) ترجع LWORK الأمثل.
!>

LWORK

!>          LWORK is INTEGER
!>          طول المصفوفة WORK. LWORK >= 1، عندما N <= 1؛
!>          غير ذلك
!>          إذا كان JOBZ = 'N' و N > 1، يجب الاستعلام عن LWORK.
!>                                   LWORK = MAX(1, 8*N, dimension) حيث
!>                                   dimension = max(stage1,stage2) + (KD+1)*N + N
!>                                             = N*KD + N*max(KD+1,FACTOPTNB)
!>                                               + max(2*KD*KD, KD*NTHREADS)
!>                                               + (KD+1)*N + N
!>                                   حيث KD هو حجم التجميع للاختزال،
!>                                   FACTOPTNB هو التجميع المستخدم بواسطة خوارزمية QR أو LQ
!>                                   عادةً ما يكون FACTOPTNB=128 خيارًا جيدًا
!>                                   NTHREADS هو عدد الخيوط المستخدمة عند
!>                                   تم تمكين تصريف OpenMP، وإلا =1.
!>          إذا كان JOBZ = 'V' و N > 1، يجب الاستعلام عن LWORK. غير متاح بعد
!>
!>          إذا كان LWORK = -1، فيفترض استعلام مساحة العمل؛ الوظيفة
