subroutine zhesvx (character fact, character uplo, integer n, integer nrhs, complex*16, dimension( lda, * ) a, integer lda, complex*16, dimension( ldaf, * ) af, integer ldaf, integer, dimension( * ) ipiv, complex*16, dimension( ldb, * ) b, integer ldb, complex*16, dimension( ldx, * ) x, integer ldx, double precision rcond, double precision, dimension( * ) ferr, double precision, dimension( * ) berr, complex*16, dimension( * ) work, integer lwork, double precision, dimension( * ) rwork, integer info)

ZHESVX تحسب حل نظام المعادلات الخطية A * X = B لمصفوفات HE

الغرض:

!>
!> ZHESVX تستخدم طريقة تحليل الـ pivoting القطري لحساب
!> الحل لنظام المعادلات الخطية المعقدة A * X = B،
!> حيث A مصفوفة هيرميتية N×N و X و B مصفوفتان N×NRHS.
!>
!> يتم أيضًا توفير حدود الخطأ للحل وتقدير للشرط.
!>

الوصف:

!>
!> يتم تنفيذ الخطوات التالية:
!>
!> 1. إذا كان FACT = 'N'، يتم استخدام طريقة الـ pivoting القطري لتحليل A.
!>    شكل التحليل هو
!>       A = U * D * U**H، إذا كان UPLO = 'U'، أو
!>       A = L * D * L**H، إذا كان UPLO = 'L'،
!>    حيث U (أو L) هو حاصل ضرب مصفوفات التبديل والمثلثية العلوية (السفلية) الوحدوية،
!>    و D هي مصفوفة هيرميتية ومقسمة إلى كتل قطريًا مع كتل قطرية بحجم 1×1 و 2×2.
!>
!> 2. إذا كان D(i,i)=0 لبعض i، مما يجعل D مفردة تمامًا، فإن الروتين
!>    يعود بـ INFO = i. بخلاف ذلك، يتم استخدام الشكل المحلل لـ A
!>    لتقدير رقم شرط المصفوفة A. إذا كان مقلوب رقم الشرط أقل من دقة الآلة،
!>    يتم إرجاع INFO = N+1 كتحذير، ولكن الروتين يستمر في حل X وحساب حدود الخطأ كما هو موضح أدناه.
!>
!> 3. يتم حل نظام المعادلات لـ X باستخدام الشكل المحلل لـ A.
!>
!> 4. يتم تطبيق التنقيح التكراري لتحسين الحل المحسوب
!>    وتقدير حدود الخطأ والأخطاء الخلفية له.
!>

المعلمات

FACT 
!>          FACT هو CHARACTER*1
!>          يحدد ما إذا كان الشكل المحلل لـ A قد تم توفيره عند الدخول أم لا.
!>          = 'F':  عند الدخول، يحتوي AF و IPIV على الشكل المحلل لـ A.
!>                  لن يتم تعديل A و AF و IPIV.
!>          = 'N':  سيتم نسخ المصفوفة A إلى AF وتحليلها.
!>

UPLO

!>          UPLO هو CHARACTER*1
!>          = 'U':  يتم تخزين المثلث العلوي لـ A؛
!>          = 'L':  يتم تخزين المثلث السفلي لـ A.
!>

N

!>          N هو INTEGER
!>          عدد المعادلات الخطية، أي رتبة المصفوفة A. N >= 0.
!>

NRHS

!>          NRHS هو INTEGER
!>          عدد الأطراف اليمنى، أي عدد أعمدة المصفوفتين B و X. NRHS >= 0.
!>

A

!>          A هو COMPLEX*16 array، dimension (LDA,N)
!>          المصفوفة الهيرميتية A. إذا كان UPLO = 'U'، فإن الجزء العلوي المثلثي N×N
!>          من A يحتوي على الجزء العلوي المثلثي للمصفوفة A، ولا تتم الإشارة إلى الجزء المثلثي السفلي الصارم لـ A.
!>          إذا كان UPLO = 'L'، فإن الجزء السفلي المثلثي N×N من A يحتوي على الجزء السفلي المثلثي للمصفوفة A،
!>          ولا تتم الإشارة إلى الجزء المثلثي العلوي الصارم لـ A.
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرئيسي للمصفوفة A. LDA >= max(1,N).
!>

AF

!>          AF هو COMPLEX*16 array، dimension (LDAF,N)
!>          إذا كان FACT = 'F'، فإن AF هو معامل إدخال ويحتوي عند الدخول
!>          على المصفوفة القطرية الكتلية D والمضاعفات المستخدمة
!>          للحصول على العامل U أو L من تحليل
!>          A = U*D*U**H أو A = L*D*L**H كما تم حسابه بواسطة ZHETRF.
!>
!>          إذا كان FACT = 'N'، فإن AF هو معامل إخراج ويعود عند الخروج
!>          بالمصفوفة القطرية الكتلية D والمضاعفات المستخدمة
!>          للحصول على العامل U أو L من تحليل
!>          A = U*D*U**H أو A = L*D*L**H.
!>

LDAF

!>          LDAF هو INTEGER
!>          البعد الرئيسي للمصفوفة AF. LDAF >= max(1,N).
!>

IPIV

!>          IPIV هو INTEGER array، dimension (N)
!>          إذا كان FACT = 'F'، فإن IPIV هو معامل إدخال ويحتوي عند الدخول
!>          على تفاصيل التبادلات وهيكل الكتلة لـ D، كما تم تحديده بواسطة ZHETRF.
!>          إذا كان IPIV(k) > 0، فإنه تم تبديل الصفوف والأعمدة k و IPIV(k)
!>          و D(k,k) هي كتلة قطرية بحجم 1×1.
!>          إذا كان UPLO = 'U' و IPIV(k) = IPIV(k-1) < 0، فإنه تم تبديل الصفوف والأعمدة k-1 و -IPIV(k)
!>          و D(k-1:k,k-1:k) هي كتلة قطرية بحجم 2×2. إذا كان UPLO = 'L' و IPIV(k) =
!>          IPIV(k+1) < 0، فإنه تم تبديل الصفوف والأعمدة k+1 و -IPIV(k)
!>          و D(k:k+1,k:k+1) هي كتلة قطرية بحجم 2×2.
!>
!>          إذا كان FACT = 'N'، فإن IPIV هو معامل إخراج ويعود عند الخروج
!>          بتفاصيل التبادلات وهيكل الكتلة لـ D، كما تم تحديده بواسطة ZHETRF.
!>

B

!>          B هو COMPLEX*16 array، dimension (LDB,NRHS)
!>          مصفوفة الطرف الأيمن B بحجم N×NRHS.
!>

LDB

!>          LDB هو INTEGER
!>          البعد الرئيسي للمصفوفة B. LDB >= max(1,N).
!>

X

!>          X هو COMPLEX*16 array، dimension (LDX,NRHS)
!>          إذا كان INFO = 0 أو INFO = N+1، فإن مصفوفة الحل X بحجم N×NRHS.
!>

LDX

!>          LDX هو INTEGER
!>          البعد الرئيسي للمصفوفة X. LDX >= max(1,N).
!>

RCOND

!>          RCOND هو DOUBLE PRECISION
!>          تقدير رقم الشرط المقلوب للمصفوفة A. إذا كان RCOND أقل من دقة الآلة
!>          (خاصة، إذا كان RCOND = 0)، فإن المصفوفة مفردة بدقة العمل.
!>          يتم الإشارة إلى هذا الشرط برمز عودة INFO > 0.
!>

FERR

!>          FERR هو DOUBLE PRECISION array، dimension (NRHS)
!>          تقدير حد الخطأ الأمامي لكل متجه حل X(j) (العمود j للمصفوفة الحل X).
!>          إذا كان XTRUE هو الحل الحقيقي المقابل لـ X(j)، فإن FERR(j)
!>          هو حد علوي مقدر لحجم أكبر عنصر في (X(j) - XTRUE) مقسومًا على حجم
!>          أكبر عنصر في X(j). التقدير موثوق به مثل
!>          التقدير لـ RCOND، وهو دائمًا تقريبًا زيادة طفيفة في الخطأ الحقيقي.
!>

BERR

!>          BERR هو DOUBLE PRECISION array، dimension (NRHS)
!>          الخطأ النسبي المكوني الخلفي لكل متجه حل X(j)
!>          (أي، أصغر تغيير نسبي في أي عنصر من A أو B يجعل X(j) حلاً دقيقًا).
!>

WORK!> WORK هو COMPLEX*16 array، dimension (LWORK) !> مساحة عمل مؤقتة. !>

LWORK

!>          LWORK هو INTEGER
!>          حجم مصفوفة العمل WORK. LWORK >= max(1, 2*N).
!>          للحصول على تقدير مثالي للشرط وحدود الخطأ، يجب أن يكون LWORK >= N + 2*N.
!>

RWORK

!>          RWORK هو DOUBLE PRECISION array، dimension (N)
!>          مساحة عمل مؤقتة.
!>

INFO

!>          INFO هو INTEGER
!>          = 0: التنفيذ الصحيح.
!>          < 0: إذا كان المعامل i هو -k، فإن العنصر k-th من المعامل كان غير صحيح.
!>          = k > 0: إذا كان k هو مؤشر الكتلة القطرية الأولى التي كانت مفردة،
!>                 فإن المصفوفة مفردة ورقم الشرط هو 0.
!>          = N+1: المصفوفة مفردة تقريبًا. تم استعادة الحل ولكن مع بعض
!>                 التحذيرات.
!>
