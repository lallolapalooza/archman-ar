subroutine zhetrf (character uplo, integer n, complex*16, dimension( lda, * ) a, integer lda, integer, dimension( * ) ipiv, complex*16, dimension( * ) work, integer lwork, integer info)

ZHETRF

الغرض:

!>
!> ZHETRF يحسب تحليل مصفوفة هرميتية مركبة A
!> باستخدام طريقة التنقيط القطري لبونش-كوفمان. شكل
!> التحليل هو
!>
!>    A = U*D*U**H  أو  A = L*D*L**H
!>
!> حيث U (أو L) هو ناتج ضرب مصفوفة تبديل ومصفوفة مثلثة علوية (سفلية)
!>، و D هي مصفوفة هرميتية ومصفوفة قطرية مقسمة إلى كتل ذات كتل قطرية
!> بحجم 1x1 و 2x2.
!>
!> هذا هو الإصدار المكدس من الخوارزمية، والذي يستدعي BLAS من المستوى 3.
!>

المعاملات

UPLO 
!>          UPLO هو CHARACTER*1
!>          = 'U':  يتم تخزين المثلث العلوي لـ A؛
!>          = 'L':  يتم تخزين المثلث السفلي لـ A.
!>

N

!>          N هو INTEGER
!>          رتبة المصفوفة A.  N >= 0.
!>

A

!>          A هو مصفوفة COMPLEX*16، البعد (LDA,N)
!>          عند الدخول، المصفوفة الهرميتية A.  إذا كان UPLO = 'U'، فإن الجزء
!>          المثلثي العلوي الذي حجمه N*N في A يحتوي على الجزء
!>          المثلثي العلوي من المصفوفة A، والجزء المثلثي السفلي الصارم
!>          لـ A لا يتم الرجوع إليه.  إذا كان UPLO = 'L'، فإن الجزء
!>          المثلثي السفلي الذي حجمه N*N في A يحتوي على الجزء
!>          المثلثي السفلي من المصفوفة A، والجزء المثلثي العلوي الصارم
!>          لـ A لا يتم الرجوع إليه.
!>
!>          عند الخروج، المصفوفة القطرية المقسمة إلى كتل D والمعاملات المستخدمة
!>          للحصول على العامل U أو L (انظر أدناه لمزيد من التفاصيل).
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرائد للمصفوفة A.  LDA >= max(1,N).
!>

IPIV

!>          IPIV هو مصفوفة INTEGER، البعد (N)
!>          تفاصيل التبادلات وهيكل الكتل لـ D.
!>          إذا كان IPIV(k) > 0، فذلك يعني أنه تم تبديل الصفوف والأعمدة k و IPIV(k)
!>          وأن D(k,k) هي كتلة قطرية بحجم 1x1.
!>          إذا كان UPLO = 'U' و IPIV(k) = IPIV(k-1) < 0، فذلك يعني أنه تم تبديل الصفوف
!>          والأعمدة k-1 و -IPIV(k) وأن D(k-1:k,k-1:k)
!>          هي كتلة قطرية بحجم 2x2.  إذا كان UPLO = 'L' و IPIV(k) =
!>          IPIV(k+1) < 0، فذلك يعني أنه تم تبديل الصفوف والأعمدة k+1 و -IPIV(k)
!>          وأن D(k:k+1,k:k+1) هي كتلة قطرية بحجم 2x2.
!>

WORK

!>          WORK هو مصفوفة COMPLEX*16، البعد (MAX(1,LWORK))
!>          عند الخروج، إذا كان INFO = 0، فإن WORK(1) يعيد LWORK الأمثل.
!>

LWORK

!>          LWORK هو INTEGER
!>          طول WORK. LWORK >= 1. للحصول على أفضل أداء
!>          LWORK >= N*NB، حيث NB هو حجم الكتلة الذي تم إرجاعه بواسطة ILAENV.
!>

INFO

!>          INFO هو INTEGER
!>          = 0:  خروج ناجح
!>          < 0:  إذا كان INFO = -i، فإن المعامل i كان له قيمة غير قانونية
!>          > 0:  إذا كان INFO = i، فإن D(i,i) يساوي صفرًا تمامًا.  لقد تم
!>                إكمال التحليل، ولكن المصفوفة القطرية المقسمة إلى كتل D
!>                هي مصفوفة منفردة تمامًا، وسيحدث قسمة على صفر إذا تم
!>                استخدامها لحل نظام من المعادلات.
!>

تفاصيل إضافية:

!>
!>  إذا كان UPLO = 'U'، فإن A = U*D*U**H، حيث
!>     U = P(n)*U(n)* ... *P(k)U(k)* ...,
!>  أي، U هو ناتج ضرب المصطلحات P(k)*U(k)، حيث k يتناقص من n إلى
!>  1 بخطوات 1 أو 2، و D هي مصفوفة قطرية مقسمة إلى كتل ذات كتل قطرية
!>  بحجم 1x1 و 2x2 D(k).  P(k) هي مصفوفة تبديل كما
!>  هو محدد بواسطة IPIV(k)، و U(k) هي مصفوفة مثلثة علوية وحدوية، بحيث
!>  إذا كانت الكتلة القطرية D(k) من الرتبة s (s = 1 أو 2)، فإن
!>
!>             (   I    v    0   )   k-s
!>     U(k) =  (   0    I    0   )   s
!>             (   0    0    I   )   n-k
!>                k-s   s   n-k
!>
!>  إذا كان s = 1، فإن D(k) تحل محل A(k,k)، و v تحل محل A(1:k-1,k).
!>  إذا كان s = 2، فإن المثلث العلوي لـ D(k) يحل محل A(k-1,k-1)، A(k-1,k)،
!>  و A(k,k)، و v تحل محل A(1:k-2,k-1:k).
!>
!>  إذا كان UPLO = 'L'، فإن A = L*D*L**H، حيث
!>     L = P(1)*L(1)* ... *P(k)*L(k)* ...,
!>  أي، L هو ناتج ضرب المصطلحات P(k)*L(k)، حيث k يزداد من 1 إلى
!>  n بخطوات 1 أو 2، و D هي مصفوفة قطرية مقسمة إلى كتل ذات كتل قطرية
!>  بحجم 1x1 و 2x2 D(k).  P(k) هي مصفوفة تبديل كما
!>  هو محدد بواسطة IPIV(k)، و L(k) هي مصفوفة مثلثة سفلية وحدوية، بحيث
!>  إذا كانت الكتلة القطرية D(k) من الرتبة s (s = 1 أو 2)، فإن
!>
!>             (   I    0     0   )  k-1
!>     L(k) =  (   0    I     0   )  s
!>             (   0    v     I   )  n-k-s+1
!>                k-1   s  n-k-s+1
!>
!>  إذا كان s = 1، فإن D(k) تحل محل A(k,k)، و v تحل محل A(k+1:n,k).
!>  إذا كان s = 2، فإن المثلث السفلي لـ D(k) يحل محل A(k,k)، A(k+1,k)،
!>  و A(k+1,k+1)، و v تحل محل A(k+2:n,k:k+1).
!>
