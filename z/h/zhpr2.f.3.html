subroutine zhpr2 (character uplo, integer n, complex*16 alpha, complex*16, dimension(*) x, integer incx, complex*16, dimension(*) y, integer incy, complex*16, dimension(*) ap)

ZHPR2

الغرض:

!>
!> ZHPR2  يقوم بعملية الرتبة 2 الهرمية
!>
!>    A := alpha*x*y**H + conjg( alpha )*y*x**H + A,
!>
!> حيث alpha هو قيمة قياسية، x و y هما متجهان مكونان من n عنصر، و A هي
!> مصفوفة هرمية بحجم n × n، مقدمة بشكل مدمج.
!>

المعلمات

UPLO 
!>          UPLO هو CHARACTER*1
!>           عند الدخول، يحدد UPLO ما إذا كان الجزء العلوي أو السفلي
!>           من المثلثي للمصفوفة A مقدم في المصفوفة AP كما يلي:
!>
!>              UPLO = 'U' أو 'u'   يتم توفير الجزء المثلثي العلوي من A
!>                                  في AP.
!>
!>              UPLO = 'L' أو 'l'   يتم توفير الجزء المثلثي السفلي من A
!>                                  في AP.
!>

N

!>          N هو INTEGER
!>           عند الدخول، يحدد N ترتيب المصفوفة A.
!>           يجب أن يكون N أكبر من أو يساوي صفر.
!>

ALPHA

!>          ALPHA هو COMPLEX*16
!>           عند الدخول، يحدد ALPHA القيمة القياسية alpha.
!>

X

!>          X هي مصفوفة COMPLEX*16، ذات بعد لا يقل عن
!>           ( 1 + ( n - 1 )*abs( INCX ) ).
!>           قبل الدخول، يجب أن تحتوي المصفوفة X ذات الفواصل المضافة على
!>           المتجه x ذي n عنصر.
!>

INCX

!>          INCX هو INTEGER
!>           عند الدخول، يحدد INCX الفاصل بين عناصر
!>           X. يجب ألا يكون INCX صفرًا.
!>

Y

!>          Y هي مصفوفة COMPLEX*16، ذات بعد لا يقل عن
!>           ( 1 + ( n - 1 )*abs( INCY ) ).
!>           قبل الدخول، يجب أن تحتوي المصفوفة Y ذات الفواصل المضافة على
!>           المتجه y ذي n عنصر.
!>

INCY

!>          INCY هو INTEGER
!>           عند الدخول، يحدد INCY الفاصل بين عناصر
!>           Y. يجب ألا يكون INCY صفرًا.
!>

AP

!>          AP هي مصفوفة COMPLEX*16، ذات بعد لا يقل عن
!>           ( ( n*( n + 1 ) )/2 ).
!>           قبل الدخول مع UPLO = 'U' أو 'u'، يجب أن تحتوي المصفوفة AP
!>           على الجزء المثلثي العلوي للمصفوفة الهرمية
!>           مدمجًا بالتتابع، عمودًا بعمود، بحيث يحتوي AP( 1 )
!>           على a( 1, 1 )، ويحتوي AP( 2 ) و AP( 3 ) على a( 1, 2 )
!>           و a( 2, 2 ) على التوالي، وهكذا. عند الخروج، يتم الكتابة فوق المصفوفة
!>           AP بالجزء المثلثي العلوي من المصفوفة المحدثة.
!>           قبل الدخول مع UPLO = 'L' أو 'l'، يجب أن تحتوي المصفوفة AP
!>           على الجزء المثلثي السفلي للمصفوفة الهرمية
!>           مدمجًا بالتتابع، عمودًا بعمود، بحيث يحتوي AP( 1 )
!>           على a( 1, 1 )، ويحتوي AP( 2 ) و AP( 3 ) على a( 2, 1 )
!>           و a( 3, 1 ) على التوالي، وهكذا. عند الخروج، يتم الكتابة فوق المصفوفة
!>           AP بالجزء المثلثي السفلي من المصفوفة المحدثة.
!>           لاحظ أنه لا يلزم تعيين الأجزاء التخيلية للعناصر القطرية،
!>           فمن المفترض أنها صفر، وعند الخروج يتم تعيينها إلى صفر.
!>
