subroutine zhegvx (integer itype, character jobz, character range, character uplo, integer n, complex*16, dimension( lda, * ) a, integer lda, complex*16, dimension( ldb, * ) b, integer ldb, double precision vl, double precision vu, integer il, integer iu, double precision abstol, integer m, double precision, dimension( * ) w, complex*16, dimension( ldz, * ) z, integer ldz, complex*16, dimension( * ) work, integer lwork, double precision, dimension( * ) rwork, integer, dimension( * ) iwork, integer, dimension( * ) ifail, integer info)

ZHEGVX

الغرض:

!>
!> ZHEGVX يحسب قيم ذاتية مختارة، واختيارياً، متجهات ذاتية
!> لمشكلة تعميم هرميتية محددة، من الشكل
!> A*x=(lambda)*B*x,  A*Bx=(lambda)*x,  أو B*A*x=(lambda)*x. هنا A و
!> B مفترض أنهما هرميتيان و B موجب محدد أيضاً.
!> يمكن تحديد القيم الذاتية والمتجهات الذاتية عن طريق تحديد إما نطاق
!> قيم أو نطاق مؤشرات للقيم الذاتية المطلوبة.
!>

المعاملات

ITYPE 
!>          ITYPE هو عدد صحيح (INTEGER)
!>          يحدد نوع المشكلة التي سيتم حلها:
!>          = 1:  A*x = (lambda)*B*x
!>          = 2:  A*B*x = (lambda)*x
!>          = 3:  B*A*x = (lambda)*x
!>

JOBZ

!>          JOBZ هو حرف (CHARACTER*1)
!>          = 'N':  حساب القيم الذاتية فقط؛
!>          = 'V':  حساب القيم الذاتية والمتجهات الذاتية.
!>

RANGE

!>          RANGE هو حرف (CHARACTER*1)
!>          = 'A': سيتم العثور على جميع القيم الذاتية.
!>          = 'V': سيتم العثور على جميع القيم الذاتية في الفاصل الزمني شبه المفتوح (VL,VU].
!>          = 'I': سيتم العثور على القيم الذاتية من المؤشر IL إلى IU.
!>

UPLO

!>          UPLO هو حرف (CHARACTER*1)
!>          = 'U':  المثلثات العلوية من A و B مخزنة؛
!>          = 'L':  المثلثات السفلية من A و B مخزنة.
!>

N

!>          N هو عدد صحيح (INTEGER)
!>          رتبة المصفوفتين A و B.  N >= 0.
!>

A

!>          A هو مصفوفة الأعداد المركبة *16 (COMPLEX*16) ، بأبعاد (LDA, N)
!>          عند الإدخال، المصفوفة الهرميتية A. إذا كان UPLO = 'U'، فإن
!>          الجزء العلوي المثلثي من A بأبعاد N*N يحتوي على
!>          الجزء العلوي المثلثي للمصفوفة A. إذا كان UPLO = 'L'، فإن
!>          الجزء السفلي المثلثي من A بأبعاد N*N يحتوي على
!>          الجزء السفلي المثلثي للمصفوفة A.
!>
!>          عند الخروج، الجزء السفلي المثلثي (إذا كان UPLO='L') أو الجزء العلوي
!>          المثلثي (إذا كان UPLO='U') من A، بما في ذلك القطر، سيتم تدميره.
!>

LDA

!>          LDA هو عدد صحيح (INTEGER)
!>          البعد الرائد للمصفوفة A. LDA >= max(1,N).
!>

B

!>          B هو مصفوفة الأعداد المركبة *16 (COMPLEX*16) ، بأبعاد (LDB, N)
!>          عند الإدخال، المصفوفة الهرميتية B. إذا كان UPLO = 'U'، فإن
!>          الجزء العلوي المثلثي من B بأبعاد N*N يحتوي على
!>          الجزء العلوي المثلثي للمصفوفة B. إذا كان UPLO = 'L'، فإن
!>          الجزء السفلي المثلثي من B بأبعاد N*N يحتوي على
!>          الجزء السفلي المثلثي للمصفوفة B.
!>
!>          عند الخروج، إذا كان INFO <= N، فإن الجزء من B الذي يحتوي على المصفوفة
!>          سيتم استبداله بالعامل المثلثي U أو L من تحليل Cholesky
!>          B = U**H*U أو B = L*L**H.
!>

LDB

!>          LDB هو عدد صحيح (INTEGER)
!>          البعد الرائد للمصفوفة B. LDB >= max(1,N).
!>

VL

!>          VL هو عدد ذو دقة مضاعفة (DOUBLE PRECISION)
!>
!>          إذا كان RANGE='V'، الحد الأدنى للفاصل الزمني
!>          الذي سيتم البحث فيه عن القيم الذاتية. VL < VU.
!>          لا يتم الرجوع إليه إذا كان RANGE = 'A' أو 'I'.
!>

VU

!>          VU هو عدد ذو دقة مضاعفة (DOUBLE PRECISION)
!>
!>          إذا كان RANGE='V'، الحد الأعلى للفاصل الزمني
!>          الذي سيتم البحث فيه عن القيم الذاتية. VL < VU.
!>          لا يتم الرجوع إليه إذا كان RANGE = 'A' أو 'I'.
!>

IL

!>          IL هو عدد صحيح (INTEGER)
!>
!>          إذا كان RANGE='I'، مؤشر
!>          أصغر قيمة ذاتية سيتم إرجاعها.
!>          1 <= IL <= IU <= N، إذا كان N > 0؛ IL = 1 و IU = 0 إذا كان N = 0.
!>          لا يتم الرجوع إليه إذا كان RANGE = 'A' أو 'V'.
!>

IU

!>          IU هو عدد صحيح (INTEGER)
!>
!>          إذا كان RANGE='I'، مؤشر
!>          أكبر قيمة ذاتية سيتم إرجاعها.
!>          1 <= IL <= IU <= N، إذا كان N > 0؛ IL = 1 و IU = 0 إذا كان N = 0.
!>          لا يتم الرجوع إليه إذا كان RANGE = 'A' أو 'V'.
!>

ABSTOL

!>          ABSTOL هو عدد ذو دقة مضاعفة (DOUBLE PRECISION)
!>          التفاوت المطلق في الدقة للقيم الذاتية.
!>          يتم قبول القيمة الذاتية التقريبية على أنها متقاربة
!>          عندما يتم تحديد أنها تقع في فاصل زمني [a,b]
!>          بعرض يساوي أو أقل من
!>
!>                  ABSTOL + EPS *   max( |a|,|b| ) ,
!>
!>          حيث EPS هي دقة الآلة. إذا كان ABSTOL أقل من
!>          أو يساوي صفرًا، فسيتم استخدام EPS*|T| بدلاً منه،
!>          حيث |T| هو معيار 1 للمصفوفة الثلاثية الأقطار التي تم الحصول عليها
!>          عن طريق اختزال C إلى شكل ثلاثي الأقطار، حيث C هي المصفوفة المتماثلة
!>          للمشكلة المتماثلة القياسية التي تحولت إليها المشكلة المعممة.
!>
!>          سيتم حساب القيم الذاتية بأكبر دقة عندما يتم تعيين ABSTOL إلى
!>          ضعف حد التدفق السفلي 2*DLAMCH('S')، وليس صفرًا.
!>          إذا عادت هذه الروتينية بمعلومة > 0، مما يشير إلى أن بعض
!>          المتجهات الذاتية لم تتقارب، حاول تعيين ABSTOL إلى
!>          2*DLAMCH('S').
!>

M

!>          M هو عدد صحيح (INTEGER)
!>          إجمالي عدد القيم الذاتية التي تم العثور عليها. 0 <= M <= N.
!>          إذا كان RANGE = 'A'، فإن M = N، وإذا كان RANGE = 'I'، فإن M = IU-IL+1.
!>

W

!>          W هي مصفوفة ذات دقة مضاعفة (DOUBLE PRECISION)، بأبعاد (N)
!>          العناصر M الأولى تحتوي على القيم الذاتية المختارة
!>          بترتيب تصاعدي.
!>

Z

!>          Z هي مصفوفة الأعداد المركبة *16 (COMPLEX*16) ، بأبعاد (LDZ, max(1,M))
!>          إذا كان JOBZ = 'N'، فلن يتم الرجوع إلى Z.
!>          إذا كان JOBZ = 'V'، وإذا كان INFO = 0، فإن الأعمدة M الأولى من Z
!>          تحتوي على المتجهات الذاتية المتعامدة للمصفوفة A
!>          المقابلة للقيم الذاتية المختارة، مع العمود i
!>          من Z يحمل المتجه الذاتي المرتبط بـ W(i).
!>          يتم تطبيع المتجهات الذاتية كالتالي:
!>          إذا كان ITYPE = 1 أو 2، Z**T*B*Z = I؛
!>          إذا كان ITYPE = 3، Z**T*inv(B)*Z = I.
!>
!>          إذا فشل متجه ذاتي في التقارب، فإن ذلك العمود من Z
!>          يحتوي على أحدث تقريب للمتجه الذاتي، وسيتم إرجاع
!>          مؤشر المتجه الذاتي في IFAIL.
!>          ملاحظة: يجب على المستخدم التأكد من توفير عمود واحد على الأقل (max(1,M))
!>          في المصفوفة Z؛ إذا كان RANGE = 'V'، فإن القيمة الدقيقة لـ M
!>          غير معروفة مسبقًا ويجب استخدام حد أعلى.
!>

LDZ

!>          LDZ هو عدد صحيح (INTEGER)
!>          البعد الرائد للمصفوفة Z. LDZ >= 1، وإذا
!>          كان JOBZ = 'V'، فإن LDZ >= max(1,N).
!>

WORK

!>          WORK هي مصفوفة الأعداد المركبة *16 (COMPLEX*16) ، بأبعاد
on (MAX(1,LWORK))
!>          عند الخروج، إذا كانت INFO = 0، فإن WORK(1) ترجع قيمة LWORK المثلى.
!>

LWORK

!>          LWORK هو عدد صحيح
!>          طول المصفوفة WORK.  LWORK >= max(1,2*N).
!>          لتحقيق أقصى قدر من الكفاءة، LWORK >= (NB+1)*N،
!>          حيث NB هي حجم الكتلة لـ ZHETRD التي تم إرجاعها بواسطة ILAENV.
!>
!>          إذا كانت LWORK = -1، فسيتم افتراض استعلام عن مساحة العمل؛ يقوم الروتين
!>          بحساب حجم المصفوفة WORK الأمثل فقط، ويرجع
!>          هذه القيمة كالإدخال الأول للمصفوفة WORK، ولا يصدر رسالة خطأ
!>          متعلقة بـ LWORK بواسطة XERBLA.
!>

RWORK

!>          RWORK هي مصفوفة ذات دقة مزدوجة، ذات بعد (7*N)
!>

IWORK

!>          IWORK هي مصفوفة عدد صحيح، ذات بعد (5*N)
!>

IFAIL

!>          IFAIL هي مصفوفة عدد صحيح، ذات بعد (N)
!>          إذا كان JOBZ = 'V'، فعندما تكون INFO = 0، فإن العناصر M الأولى من
!>          IFAIL تكون صفرًا.  إذا كانت INFO > 0، فإن IFAIL تحتوي على
!>          فهارس المتجهات الذاتية التي فشلت في التقارب.
!>          إذا كان JOBZ = 'N'، فلن يتم الرجوع إلى IFAIL.
!>

INFO

!>          INFO هو عدد صحيح
!>          = 0:  انتهاء ناجح
!>          < 0:  إذا كانت INFO = -i، فإن الوسيط i كان له قيمة غير قانونية
!>          > 0:  أرجعت ZPOTRF أو ZHEEVX رمز خطأ:
!>             <= N:  إذا كانت INFO = i، فشلت ZHEEVX في التقارب؛
!>                    فشلت i من المتجهات الذاتية في التقارب.  فهارسها
!!>                    مخزنة في المصفوفة IFAIL.
!>             > N:   إذا كانت INFO = N + i، لـ 1 <= i <= N، فإن
!>                    المعامل الرئيسي الجزئي من الرتبة i لـ B ليس موجبًا.
!>                    لم يكن من الممكن إكمال تحليل B و
!>                    لم يتم حساب أي قيم ذاتية أو متجهات ذاتية.
!>
