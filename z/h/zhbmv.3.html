subroutine zhbmv (character uplo, integer n, integer k, complex*16 alpha, complex*16, dimension(lda,*) a, integer lda, complex*16, dimension(*) x, integer incx, complex*16 beta, complex*16, dimension(*) y, integer incy)

ZHBMV

الغرض:

!>
!> ZHBMV  يقوم بعملية المصفوفة والمتجه
!>
!>    y := alpha*A*x + beta*y,
!>
!> حيث alpha و beta عددان قياسيان، x و y متجهان من n عنصر، و
!> A مصفوفة هيرميتية نطاقية بحجم n × n، مع k نطاق فوق قطري.
!>

المعاملات

UPLO 
!>          UPLO هو CHARACTER*1
!>           عند الإدخال، يحدد UPLO ما إذا كان الجزء العلوي أو السفلي
!>           المثلثي من المصفوفة A قد تم توفيره كما يلي:
!>
!>              UPLO = 'U' or 'u'   تم توفير الجزء العلوي المثلثي من A.
!>
!>              UPLO = 'L' or 'l'   تم توفير الجزء السفلي المثلثي من A.
!>

N

!>          N هو INTEGER
!>           عند الإدخال، يحدد N رتبة المصفوفة A.
!>           يجب أن يكون N على الأقل صفرًا.
!>

K

!>          K هو INTEGER
!>           عند الإدخال، يحدد K عدد النطاقات الفوق قطرية للمصفوفة A.
!>           يجب أن يحقق K  0 .le. K.
!>

ALPHA

!>          ALPHA هو COMPLEX*16
!>           عند الإدخال، يحدد ALPHA العدد القياسي alpha.
!>

A

!>          A هو مصفوفة COMPLEX*16، البعد ( LDA, N )
!>           قبل الإدخال مع UPLO = 'U' أو 'u'، يجب أن يحتوي الجزء
!>           العلوي ( k + 1 ) × n من المصفوفة A على الجزء العلوي
!>           المثلثي النطاقي للمصفوفة الهيرميتية، يتم توفيره عمودًا
!>           بعمود، مع القطر الرئيسي للمصفوفة في الصف
!>           ( k + 1 ) من المصفوفة، ويبدأ النطاق الفوق قطري الأول في
!>           الموضع 2 في الصف k، وهكذا. لا يتم الرجوع إلى المثلث العلوي
!>           من A بحجم k × k.
!>           سيقوم مقطع البرنامج التالي بنقل الجزء العلوي المثلثي
!>           من مصفوفة هيرميتية نطاقية من التخزين التقليدي الكامل للمصفوفة
!>           إلى تخزين النطاق:
!>
!>                 DO 20, J = 1, N
!>                    M = K + 1 - J
!>                    DO 10, I = MAX( 1, J - K ), J
!>                       A( M + I, J ) = matrix( I, J )
!>              10    CONTINUE
!>              20 CONTINUE
!>
!>           قبل الإدخال مع UPLO = 'L' أو 'l'، يجب أن يحتوي الجزء
!>           السفلي ( k + 1 ) × n من المصفوفة A على الجزء السفلي
!>           المثلثي النطاقي للمصفوفة الهيرميتية، يتم توفيره عمودًا
!>           بعمود، مع القطر الرئيسي للمصفوفة في الصف 1
!>           من المصفوفة، ويبدأ النطاق الفرعي الأول في الموضع 1 في
!>           الصف 2، وهكذا. لا يتم الرجوع إلى المثلث السفلي الأيمن
!>           من A بحجم k × k.
!>           سيقوم مقطع البرنامج التالي بنقل الجزء السفلي المثلثي
!>           من مصفوفة هيرميتية نطاقية من التخزين التقليدي الكامل للمصفوفة
!>           إلى تخزين النطاق:
!>
!>                 DO 20, J = 1, N
!>                    M = 1 - J
!>                    DO 10, I = J, MIN( N, J + K )
!>                       A( M + I, J ) = matrix( I, J )
!>              10    CONTINUE
!>              20 CONTINUE
!>
!>           لاحظ أنه لا يلزم تعيين الأجزاء التخيلية للعناصر القطرية
!>           وتفترض أنها صفر.
!>

LDA

!>          LDA هو INTEGER
!>           عند الإدخال، يحدد LDA البعد الأول لـ A كما تم تعريفه
!>           في البرنامج المستدعي (الإجراء الفرعي). يجب أن يكون LDA على
!>           الأقل ( k + 1 ).
!>

X

!>          X هي مصفوفة COMPLEX*16، البعد على الأقل
!>           ( 1 + ( n - 1 )*abs( INCX ) ).
!>           قبل الإدخال، يجب أن تحتوي المصفوفة X ذات الزيادة على
!>           المتجه x.
!>

INCX

!>          INCX هو INTEGER
!>           عند الإدخال، يحدد INCX الزيادة لعناصر X.
!>           يجب ألا يكون INCX صفرًا.
!>

BETA

!>          BETA هو COMPLEX*16
!>           عند الإدخال، يحدد BETA العدد القياسي beta.
!>

Y

!>          Y هي مصفوفة COMPLEX*16، البعد على الأقل
!>           ( 1 + ( n - 1 )*abs( INCY ) ).
!>           قبل الإدخال، يجب أن تحتوي المصفوفة Y ذات الزيادة على
!>           المتجه y. عند الخروج، يتم استبدال Y بالمتجه y المحدث.
!>

INCY

!>          INCY هو INTEGER
!>           عند الإدخال، يحدد INCY الزيادة لعناصر Y.
!>           يجب ألا يكون INCY صفرًا.
!>
