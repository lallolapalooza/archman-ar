subroutine zher2k (character uplo, character trans, integer n, integer k, complex*16 alpha, complex*16, dimension(lda,*) a, integer lda, complex*16, dimension(ldb,*) b, integer ldb, double precision beta, complex*16, dimension(ldc,*) c, integer ldc)

ZHER2K

الغرض:

!>
!> ZHER2K  تقوم بإحدى عمليات الصف المتهادفة من الرتبة 2k
!>
!>    C := alpha*A*B**H + conjg( alpha )*B*A**H + beta*C,
!>
!> أو
!>
!>    C := alpha*A**H*B + conjg( alpha )*B**H*A + beta*C,
!>
!> حيث  alpha و beta  هما كميات قياسية و beta  حقيقية،  C هي مصفوفة متهادفة
!> حجمها n × n  و A و B  هما مصفوفتان حجمهما n × k  في الحالة الأولى
!> وحجمهما k × n  في الحالة الثانية.
!>

المعلمات

UPLO 
!>          UPLO هو CHARACTER*1
!>           عند الدخول،  UPLO  تحدد ما إذا كان الجزء العلوي أو السفلي
!>           من المصفوفة  C  سيتم الرجوع إليه كالتالي:
!>
!>              UPLO = 'U' or 'u'   يتم الرجوع فقط إلى الجزء العلوي المثلثي من C
!>                                  فقط.
!>
!>              UPLO = 'L' or 'l'   يتم الرجوع فقط إلى الجزء السفلي المثلثي من C
!>                                  فقط.
!>

TRANS

!>          TRANS هو CHARACTER*1
!>           عند الدخول،  TRANS  تحدد العملية التي سيتم إجراؤها كالتالي:
!>
!>              TRANS = 'N' or 'n'    C := alpha*A*B**H          +
!>                                         conjg( alpha )*B*A**H +
!>                                         beta*C.
!>
!>              TRANS = 'C' or 'c'    C := alpha*A**H*B          +
!>                                         conjg( alpha )*B**H*A +
!>                                         beta*C.
!>

N

!>          N هو INTEGER
!>           عند الدخول،  N  تحدد رتبة المصفوفة C.  يجب أن تكون N
!>           أكبر من أو تساوي الصفر.
!>

K

!>          K هو INTEGER
!>           عند الدخول مع  TRANS = 'N' or 'n',  K  تحدد عدد
!>           الأعمدة  للمصفوفتين  A و B,  وعند الدخول مع
!>           TRANS = 'C' or 'c',  K  تحدد عدد الصفوف للمصفوفتين
!>           A و B.  يجب أن تكون K  أكبر من أو تساوي الصفر.
!>

ALPHA

!>          ALPHA هو COMPLEX*16 .
!>           عند الدخول،  ALPHA  تحدد الكمية القياسية alpha.
!>

A

!>          A هو مصفوفة COMPLEX*16 ، بأبعاد ( LDA, ka )، حيث ka هي
!>           k  عندما  TRANS = 'N' or 'n',  و n  بخلاف ذلك.
!>           قبل الدخول مع  TRANS = 'N' or 'n',  يجب أن يحتوي الجزء الرئيسي n × k
!>           من المصفوفة  A  على المصفوفة  A,  بخلاف ذلك
!>           يجب أن يحتوي الجزء الرئيسي k × n  من المصفوفة  A  على
!>           المصفوفة A.
!>

LDA

!>          LDA هو INTEGER
!>           عند الدخول،  LDA  تحدد البعد الأول لـ A كما هو معلن
!>           في  البرنامج (الفرعي) المستدعي.   عندما يكون TRANS = 'N' or 'n'
!>           فيجب أن يكون LDA  أكبر من أو يساوي  max( 1, n )، بخلاف ذلك
!>           يجب أن يكون LDA  أكبر من أو يساوي  max( 1, k ).
!>

B

!>          B هو مصفوفة COMPLEX*16 ، بأبعاد ( LDB, kb )، حيث kb هي
!>           k  عندما  TRANS = 'N' or 'n',  و n  بخلاف ذلك.
!>           قبل الدخول مع  TRANS = 'N' or 'n',  يجب أن يحتوي الجزء الرئيسي n × k
!>           من المصفوفة  B  على المصفوفة  B,  بخلاف ذلك
!>           يجب أن يحتوي الجزء الرئيسي k × n  من المصفوفة  B  على
!>           المصفوفة B.
!>

LDB

!>          LDB هو INTEGER
!>           عند الدخول،  LDB  تحدد البعد الأول لـ B كما هو معلن
!>           في  البرنامج (الفرعي) المستدعي.   عندما يكون TRANS = 'N' or 'n'
!>           فيجب أن يكون LDB  أكبر من أو يساوي  max( 1, n )، بخلاف ذلك
!>           يجب أن يكون LDB  أكبر من أو يساوي  max( 1, k ).
!>           لم يتغير عند الخروج.
!>

BETA

!>          BETA هو DOUBLE PRECISION .
!>           عند الدخول،  BETA  تحدد الكمية القياسية beta.
!>

C

!>          C هو مصفوفة COMPLEX*16 ، بأبعاد ( LDC, N )
!>           قبل الدخول مع  UPLO = 'U' or 'u',  يجب أن يحتوي الجزء الرئيسي n × n
!>           العلوي المثلثي من المصفوفة C على الجزء العلوي
!>           المثلثي من المصفوفة المتهادفة  والجزء السفلي المثلثي تمامًا
!>           من C لا يتم الرجوع إليه.   عند الخروج، يتم الكتابة فوق الجزء العلوي
!>           المثلثي من المصفوفة C بالجزء العلوي
!>           المثلثي من المصفوفة المحدثة.
!>           قبل الدخول مع  UPLO = 'L' or 'l',  يجب أن يحتوي الجزء الرئيسي n × n
!>           السفلي المثلثي من المصفوفة C على الجزء السفلي
!>           المثلثي من المصفوفة المتهادفة  والجزء العلوي المثلثي تمامًا
!>           من C لا يتم الرجوع إليه.   عند الخروج، يتم الكتابة فوق الجزء السفلي
!>           المثلثي من المصفوفة C بالجزء السفلي
!>           المثلثي من المصفوفة المحدثة.
!>           لاحظ أن الأجزاء التخيلية للعناصر القطرية لا
!>           يجب تعيينها،  يفترض أنها صفر،  وعند الخروج
!>           يتم تعيينها إلى الصفر.
!>

LDC

!>          LDC هو INTEGER
!>           عند الدخول،  LDC  تحدد البعد الأول لـ C كما هو معلن
!>           في  البرنامج (الفرعي) المستدعي.   يجب أن يكون LDC
!>           أكبر من أو يساوي  max( 1, n ).
!>
