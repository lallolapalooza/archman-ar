subroutine zhetf2_rk (character uplo, integer n, complex*16, dimension( lda, * ) a, integer lda, complex*16, dimension( * ) e, integer, dimension( * ) ipiv, integer info)

ZHETF2_RK تقوم بحساب تحليل مصفوفة هرميتية غير محددة معقدة باستخدام طريقة التنقيط القطري لبونش-كوفمان (rook) (أسلوب غير مجمّع BLAS2).

الغرض:

!> ZHETF2_RK تقوم بحساب تحليل مصفوفة هرميتية معقدة A
!> باستخدام طريقة التنقيط القطري لبونش-كوفمان (rook) المحدود:
!>
!>    A = P*U*D*(U**H)*(P**T) أو A = P*L*D*(L**H)*(P**T),
!>
!> حيث U (أو L) هي مصفوفة مثلثية عليا (أو سفلى) أحادية،
!> U**H (أو L**H) هي مرافق U (أو L)، P هي مصفوفة تبديل
!> (permutation matrix)، P**T هي منقولة P، و D هي مصفوفة هرميتية وقطرية مجمّعة
!> بكتل قطرية بحجم 1x1 و 2x2.
!>
!> هذا هو الأسلوب غير المجمّع للخوارزمية، والذي يستدعي BLAS من المستوى 2.
!> لمزيد من المعلومات انظر قسم التفاصيل الإضافية.
!>

المعلمات

UPLO 
!>          UPLO هو CHARACTER*1
!>          يحدد ما إذا كان الجزء العلوي أو السفلي المثلثي من المصفوفة الهرميتية A
!>          مخزنًا:
!>          = 'U':  مثلثي علوي
!>          = 'L':  مثلثي سفلي
!>

N

!>          N هو INTEGER
!>          رتبة المصفوفة A. N >= 0.
!>

A

!>          A هو مصفوفة COMPLEX*16، البعد (LDA,N)
!>          عند الدخول، المصفوفة الهرميتية A.
!>            إذا كان UPLO = 'U': الجزء العلوي المثلثي من حجم N*N
!>            في A يحتوي على الجزء العلوي المثلثي للمصفوفة A،
!>            والجزء السفلي المثلثي الصارم لـ A لا يتم الرجوع إليه.
!>
!>            إذا كان UPLO = 'L': الجزء السفلي المثلثي من حجم N*N
!>            في A يحتوي على الجزء السفلي المثلثي للمصفوفة A،
!>            والجزء العلوي المثلثي الصارم لـ A لا يتم الرجوع إليه.
!>
!>          عند الخروج، تحتوي على:
!>            أ) فقط العناصر القطرية للمصفوفة الهرميتية القطرية المجمّعة
!>               D على قطر A، أي D(k,k) = A(k,k)؛
!>               (العناصر فوق القطرية (أو تحت القطرية) لـ D
!>                مخزنة عند الخروج في المصفوفة E)، و
!>            ب) إذا كان UPLO = 'U': العامل U في الجزء فوق القطري لـ A.
!>               إذا كان UPLO = 'L': العامل L في الجزء تحت القطري لـ A.
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرائد للمصفوفة A. LDA >= max(1,N).
!>

E

!>          E هي مصفوفة COMPLEX*16، البعد (N)
!>          عند الخروج، تحتوي على العناصر فوق القطرية (أو تحت القطرية)
!>          للمصفوفة الهرميتية القطرية المجمّعة D
!>          بكتل قطرية بحجم 1x1 أو 2x2، حيث
!>          إذا كان UPLO = 'U': E(i) = D(i-1,i), i=2:N, E(1) يتم ضبطها على 0؛
!>          إذا كان UPLO = 'L': E(i) = D(i+1,i), i=1:N-1, E(N) يتم ضبطها على 0.
!>
!>          ملاحظة: بالنسبة لكتلة قطرية بحجم 1x1 D(k)، حيث
!>          1 <= k <= N، يتم ضبط العنصر E(k) على 0 في كل من
!>          حالات UPLO = 'U' أو UPLO = 'L'.
!>

IPIV

!>          IPIV هي مصفوفة INTEGER، البعد (N)
!>          تصف IPIV مصفوفة التبديل P في تحليل
!>          المصفوفة A كالتالي. القيمة المطلقة لـ IPIV(k)
!>          تمثل فهرس الصف والعمود اللذين تم
!>          تبديلهما مع الصف والعمود k. قيمة UPLO
!>          تصف ترتيب تطبيق التبادلات.
!>          أيضًا، إشارة IPIV تمثل البنية المجمّعة
!>          للمصفوفة الهرميتية القطرية المجمّعة D بكتل قطرية بحجم 1x1 أو 2x2
!>          والتي تتوافق مع تبديل واحد أو اثنين
!>          في كل خطوة تحليل. لمزيد من المعلومات انظر قسم التفاصيل الإضافية.
!>
!>          إذا كان UPLO = 'U'،
!>          (بترتيب التحليل، k ينقص من N إلى 1):
!>            أ) إدخال موجب فردي IPIV(k) > 0 يعني:
!>               D(k,k) هي كتلة قطرية بحجم 1x1.
!>               إذا كان IPIV(k) != k، فقد تم تبديل الصفوف والأعمدة k و IPIV(k)
!>               في المصفوفة A(1:N,1:N)؛
!>               إذا كان IPIV(k) = k، لم يحدث تبديل.
!>
!>            ب) زوج من الإدخالات المتتالية السالبة
!>               IPIV(k) < 0 و IPIV(k-1) < 0 يعني:
!>               D(k-1:k,k-1:k) هي كتلة قطرية بحجم 2x2.
!>               (ملاحظة: الإدخالات السالبة في IPIV تظهر فقط في أزواج).
!>               1) إذا كان -IPIV(k) != k، فقد تم تبديل الصفوف والأعمدة
!>                  k و -IPIV(k)
!>                  في المصفوفة A(1:N,1:N).
!>                  إذا كان -IPIV(k) = k، لم يحدث تبديل.
!>               2) إذا كان -IPIV(k-1) != k-1، فقد تم تبديل الصفوف والأعمدة
!>                  k-1 و -IPIV(k-1)
!>                  في المصفوفة A(1:N,1:N).
!>                  إذا كان -IPIV(k-1) = k-1، لم يحدث تبديل.
!>
!>            ج) في كلتا الحالتين أ) وب)، دائمًا ABS( IPIV(k) ) <= k.
!>
!>            د) ملاحظة: أي إدخال IPIV(k) غير صفري دائمًا عند الخروج.
!>
!>          إذا كان UPLO = 'L'،
!>          (بترتيب التحليل، k يزيد من 1 إلى N):
!>            أ) إدخال موجب فردي IPIV(k) > 0 يعني:
!>               D(k,k) هي كتلة قطرية بحجم 1x1.
!>               إذا كان IPIV(k) != k، فقد تم تبديل الصفوف والأعمدة k و IPIV(k)
!>               في المصفوفة A(1:N,1:N).
!>               إذا كان IPIV(k) = k، لم يحدث تبديل.
!>
!>            ب) زوج من الإدخالات المتتالية السالبة
!>               IPIV(k) < 0 و IPIV(k+1) < 0 يعني:
!>               D(k:k+1,k:k+1) هي كتلة قطرية بحجم 2x2.
!>               (ملاحظة: الإدخالات السالبة في IPIV تظهر فقط في أزواج).
!>               1) إذا كان -IPIV(k) != k، فقد تم تبديل الصفوف والأعمدة
!>                  k و -IPIV(k)
!>                  في المصفوفة A(1:N,1:N).
!>                  إذا كان -IPIV(k) = k، لم يحدث تبديل.
!>               2) إذا كان -IPIV(k+1) != k+1، فقد تم تبديل الصفوف والأعمدة
!>                  k-1 و -IPIV(k-1)
!>                  في المصفوفة A(1:N,1:N).
!>                  إذا كان -IPIV(k+1) = k+1، لم يحدث تبديل.
!>
!>            ج) في كلتا الحالتين أ) وب)، دائمًا ABS( IPIV(k) ) >= k.
!>
!>            د) ملاحظة: أي إدخال IPIV(k) غير صفري دائمًا عند الخروج.
!>

INFO

!>          INFO هو INTEGER
!>          = 0: تم الانتهاء بنجاح
!>
!>          < 0: إذا كان INFO = -k، فإن الوسيط k كان له قيمة غير قانونية
!>
!>          > 0: إذا كان INFO = k، فإن المصفوفة A منفردة (singular)، بسبب:
!>                 إذا كان UPLO = 'U': العمود k في الجزء
!>                 المثلثي العلوي من A يحتوي على جميع الأصفار.
!>                 إذا كان UPLO = 'L': العمود k في الجزء
!>                 المثلثي السفلي من A يحتوي على جميع الأصفار.
!>
!>               لذلك فإن D(k,k) صفر تمامًا، والعناصر فوق القطرية
!>               للعمود k من U (أو العناصر تحت القطرية
!>               للعمود k من L) كلها أصفار. تم إكمال التحليل،
!>               لكن المصفوفة القطرية المجمّعة D
!>               منفردة تمامًا، وسيحدث قسمة على صفر إذا
!>               تم استخدامها لحل نظام معادلات.
!>
!>               ملاحظة: INFO يخزن فقط أول حدوث
!>               للانفراد، أي حدوث لاحق للانفراد
!>               لا يتم تخزينه في INFO على الرغم من أن التحليل
!>               يكتمل دائمًا.
!>

SUBROUTINE ZHETF2_RK (UPLO, N, K)
  CHARACTER * 1 UPLO
  INTEGER K, N

•
UPLO CHARACTER*1. Must be 'U' or 'L'.
  On entry, UPLO specifies whether the upper or lower triangular part of the matrix A is to be computed.
•
N INTEGER.
On entry, N is the order of the matrix A. N >= 0.
•
K INTEGER.
On entry, K is the number of columns of U, R, P, and K that are computed.
