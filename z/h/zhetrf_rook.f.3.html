subroutine zhetrf_rook (character uplo, integer n, complex*16, dimension( lda, * ) a, integer lda, integer, dimension( * ) ipiv, complex*16, dimension( * ) work, integer lwork, integer info)

ZHETRF_ROOK تحسب تحليل مصفوفة هرميتية غير محددة معقدة باستخدام طريقة تجميع Bunch-Kaufman ("rook") pivot قطري (خوارزمية مجمعة، تستدعي BLAS المستوى 3).

الغرض:

!>
!> ZHETRF_ROOK تحسب تحليل مصفوفة هرميتية معقدة A
!> باستخدام طريقة تجميع Bunch-Kaufman () pivot قطري.
!> شكل التحليل هو
!>
!>    A = U*D*U**T  أو  A = L*D*L**T
!>
!> حيث U (أو L) هو ناتج ضرب مصفوفات التبديل والمصفوفات المثلثية العلوية (السفلية)
!> الوحدوية، و D هي مصفوفة هرميتية ومصفوفة قطرية مجمعة مع كتل قطرية بحجم
!> 1x1 و 2x2.
!>
!> هذا هو الإصدار المجمع للخوارزمية، والذي يستدعي BLAS المستوى 3.
!>

المعلمات

UPLO 
!>          UPLO is CHARACTER*1
!>          = 'U':  الجزء العلوي من A مخزن؛
!>          = 'L':  الجزء السفلي من A مخزن.
!>

N

!>          N is INTEGER
!>          رتبة المصفوفة A.  N >= 0.
!>

A

!>          A is COMPLEX*16 array, dimension (LDA,N)
!>          عند الدخول، المصفوفة الهرميتية A.  إذا كان UPLO = 'U'، فإن الجزء
!>          المثلثي العلوي بحجم N×N في A يحتوي على الجزء المثلثي العلوي
!>          للمصفوفة A، والجزء المثلثي السفلي الصارم من A لا تتم الإشارة إليه.
!>          إذا كان UPLO = 'L'، فإن الجزء المثلثي السفلي بحجم N×N في A يحتوي
!>          على الجزء المثلثي السفلي للمصفوفة A، والجزء المثلثي العلوي الصارم
!>          من A لا تتم الإشارة إليه.
!>
!>          عند الخروج، المصفوفة القطرية المجمعة D والمضاعفات المستخدمة
!>          للحصول على العامل U أو L (انظر أدناه لمزيد من التفاصيل).
!>

LDA

!>          LDA is INTEGER
!>          البعد الرائد للمصفوفة A.  LDA >= max(1,N).
!>

IPIV

!>          IPIV is INTEGER array, dimension (N)
!>          تفاصيل التبادلات والهيكل المجمع لـ D.
!>
!>          إذا كان UPLO = 'U':
!>             يتم تعيين آخر KB عنصرًا فقط من IPIV.
!>
!>             إذا كان IPIV(k) > 0، فقد تم تبديل الصفوف والأعمدة k و IPIV(k)
!>             و D(k,k) هي كتلة قطرية بحجم 1×1.
!>
!>             إذا كان IPIV(k) < 0 و IPIV(k-1) < 0، فقد تم تبديل الصفوف
!>             والأعمدة k و -IPIV(k) وتم تبديل الصفوف
!>             والأعمدة k-1 و -IPIV(k-1)،
!>             D(k-1:k,k-1:k) هي كتلة قطرية بحجم 2×2.
!>
!>          إذا كان UPLO = 'L':
!>             يتم تعيين أول KB عنصرًا فقط من IPIV.
!>
!>             إذا كان IPIV(k) > 0، فقد تم تبديل الصفوف والأعمدة k و IPIV(k)
!>             و D(k,k) هي كتلة قطرية بحجم 1×1.
!>
!>             إذا كان IPIV(k) < 0 و IPIV(k+1) < 0، فقد تم تبديل الصفوف
!>             والأعمدة k و -IPIV(k) وتم تبديل الصفوف
!>             والأعمدة k+1 و -IPIV(k+1)،
!>             D(k:k+1,k:k+1) هي كتلة قطرية بحجم 2×2.
!>

WORK

!>          WORK is COMPLEX*16 array, dimension (MAX(1,LWORK)).
!>          عند الخروج، إذا كان INFO = 0، فإن WORK(1) يعيد LWORK الأمثل.
!>

LWORK

!>          LWORK is INTEGER
!>          طول WORK.  LWORK >= 1.  للحصول على أفضل أداء
!>          LWORK >= N*NB، حيث NB هو حجم الكتلة الذي تعيده ILAENV.
!>
!>          إذا كان LWORK = -1، فمن المفترض وجود استعلام للمساحة، و
!>          تقوم الوحدة الفرعية بحساب حجم مساحة العمل الأمثل فقط، وتعيد
!>          هذه القيمة كأول إدخال في مساحة العمل، ولا يصدر XERBLA أي رسالة
!>          خطأ متعلقة بـ LWORK.
!>

INFO

!>          INFO is INTEGER
!>          = 0:  خروج ناجح
!>          < 0:  إذا كان INFO = -i، فإن الوسيط i كان له قيمة غير قانونية
!>          > 0:  إذا كان INFO = i، فإن D(i,i) يساوي صفرًا بالضبط.  تم إكمال
!>                التحليل، ولكن المصفوفة القطرية المجمعة D
!>                فردية تمامًا، وسيحدث قسمة على صفر إذا
!>                تم استخدامها لحل نظام معادلات.
!>

تفاصيل إضافية:

!>
!>  إذا كان UPLO = 'U'، فإن A = U*D*U**T، حيث
!>     U = P(n)*U(n)* ... *P(k)U(k)* ...,
!>  أي، U هو ناتج ضرب المصطلحات P(k)*U(k)، حيث k تتناقص من n إلى
!>  1 بخطوات 1 أو 2، و D هي مصفوفة قطرية مجمعة مع كتل قطرية بحجم
!>  1×1 و 2×2 وهي D(k). P(k) هي مصفوفة تبديل كما
!>  هو محدد بواسطة IPIV(k)، و U(k) هي مصفوفة مثلثة علوية وحدوية، بحيث
!>  إذا كانت الكتلة القطرية D(k) من الرتبة s (s = 1 أو 2)، فإن
!>
!>             (   I    v    0   )   k-s
!>     U(k) =  (   0    I    0   )   s
!>             (   0    0    I   )   n-k
!>                k-s   s   n-k
!>
!>  إذا كان s = 1، فإن D(k) تحل محل A(k,k)، و v تحل محل A(1:k-1,k).
!>  إذا كان s = 2، فإن الجزء العلوي من D(k) يحل محل A(k-1,k-1)، A(k-1,k)،
!>  و A(k,k)، و v تحل محل A(1:k-2,k-1:k).
!>
!>  إذا كان UPLO = 'L'، فإن A = L*D*L**T، حيث
!>     L = P(1)*L(1)* ... *P(k)*L(k)* ...,
!>  أي، L هو ناتج ضرب المصطلحات P(k)*L(k)، حيث k تزداد من 1 إلى
!>  n بخطوات 1 أو 2، و D هي مصفوفة قطرية مجمعة مع كتل قطرية بحجم
!>  1×1 و 2×2 وهي D(k). P(k) هي مصفوفة تبديل كما
!>  هو محدد بواسطة IPIV(k)، و L(k) هي مصفوفة مثلثة سفلية وحدوية، بحيث
!>  إذا كانت الكتلة القطرية D(k) من الرتبة s (s = 1 أو 2)، فإن
!>
!>             (   I    0     0   )  k-1
!>     L(k) =  (   0    I     0   )  s
!>             (   0    v     I   )  n-k-s+1
!>                k-1   s  n-k-s+1
!>
!>  إذا كان s = 1، فإن D(k) تحل محل A(k,k)، و v تحل محل A(k+1:n,k).
!>  إذا كان s = 2، فإن الجزء السفلي من D(k) يحل محل A(k,k)، A(k+1,k)،
!>  و A(k+1,k+1)، و v تحل محل A(k+2:n,k:k+1).
!>
