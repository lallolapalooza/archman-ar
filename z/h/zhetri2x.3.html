subroutine zhetri2x (character uplo, integer n, complex*16, dimension( lda, * ) a, integer lda, integer, dimension( * ) ipiv, complex*16, dimension( n+nb+1,* ) work, integer nb, integer info)

ZHETRI2X

الغرض:

!>
!> ZHETRI2X يحسب معكوس مصفوفة هرميتية غير محددة من النوع COMPLEX*16
!> A باستخدام التحلل A = U*D*U**H أو A = L*D*L**H الذي تم حسابه بواسطة
!> ZHETRF.
!>

المعاملات

UPLO 
!>          UPLO هو CHARACTER*1
!>          يحدد ما إذا كانت تفاصيل التحلل مخزنة
!>          كمصفوفة مثلثة علوية أو سفلية.
!>          = 'U':  مثلثة علوية، الشكل هو A = U*D*U**H؛
!>          = 'L':  مثلثة سفلية، الشكل هو A = L*D*L**H.
!>

N

!>          N هو INTEGER
!>          ترتيب المصفوفة A. N >= 0.
!>

A

!>          A هي مصفوفة COMPLEX*16، البعد (LDA,N)
!>          عند الإدخال، مصفوفة القطر D ذات الأبعاد NNB والعوامل
!>          المستخدمة للحصول على العامل U أو L كما تم حسابه بواسطة ZHETRF.
!>
!>          عند الخروج، إذا كان INFO = 0، فإن المعكوس (المتماثل) للمصفوفة
!>          الأصلية. إذا كان UPLO = 'U'، يتم تشكيل الجزء المثلث العلوي
!>          من المعكوس ولا يتم الرجوع إلى جزء A أسفل القطر؛ إذا كان UPLO = 'L'
!>          يتم تشكيل الجزء المثلث السفلي من المعكوس ولا يتم الرجوع إلى جزء A
!>          فوق القطر.
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرائد للمصفوفة A. LDA >= max(1,N).
!>

IPIV

!>          IPIV هي مصفوفة INTEGER، البعد (N)
!>          تفاصيل التبادلات وهيكل D ذي الأبعاد NNB
!>          كما تم تحديده بواسطة ZHETRF.
!>

WORK

!>          WORK هي مصفوفة COMPLEX*16، البعد (N+NB+1,NB+3)
!>

NB

!>          NB هو INTEGER
!>          حجم الكتلة
!>

INFO

!>          INFO هو INTEGER
!>          = 0: خروج ناجح
!>          < 0: إذا كان INFO = -i، فإن المعامل i كان له قيمة غير قانونية
!>          > 0: إذا كان INFO = i، فإن D(i,i) = 0؛ المصفوفة منفردة ولا يمكن
!>               حساب معكوسها.
!>
