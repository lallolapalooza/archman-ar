subroutine zhetri_rook (uplo, n, a, lda, ipiv, work, info)
ZHETRI_ROOK تحسب معكوس مصفوفة HE باستخدام التحلل الناتج عن طريقة الـ Bunch-Kaufman المحصورة (rook")diagonalpivotingmethod.

توثيق الوظيفة/الإجراء الفرعي

subroutine zhetri_rook (character uplo, integer n, complex*16, dimension( lda, * ) a, integer lda, integer, dimension( * ) ipiv, complex*16, dimension( * ) work, integer info)

ZHETRI_ROOK تحسب معكوس مصفوفة هيرميتية غير محددة باستخدام التحلل الناتج عن طريقة الـ Bunch-Kaufman المحصورة (روك) للـ pivoting القطري.

الغرض:

!>
!> ZHETRI_ROOK تحسب معكوس مصفوفة هيرميتية غير محددة معقدة
!> A باستخدام التحلل A = U*D*U**H أو A = L*D*L**H الذي تم حسابه بواسطة
!> ZHETRF_ROOK.
!>

المعاملات

UPLO 
!>          UPLO هو CHARACTER*1
!>          يحدد ما إذا كانت تفاصيل التحلل مخزنة
!>          كمصفوفة مثلثة عليا أو سفلى.
!>          = 'U':  مثلثة عليا، الشكل هو A = U*D*U**H؛
!>          = 'L':  مثلثة سفلى، الشكل هو A = L*D*L**H.
!>

N

!>          N هو INTEGER
!>          ترتيب المصفوفة A.  N >= 0.
!>

A

!>          A هي مصفوفة COMPLEX*16، ذات بعد (LDA,N)
!>          عند الإدخال، المصفوفة القطرية الفرعية D والمعاملات
!>          المستخدمة للحصول على العامل U أو L كما تم حسابها بواسطة ZHETRF_ROOK.
!>
!>          عند الخروج، إذا كان INFO = 0، فإن معكوس المصفوفة الأصلية
!>          (الهيرميتية). إذا كان UPLO = 'U'، يتم تشكيل الجزء المثلث العلوي
!>          من المعكوس ولا يتم الرجوع إلى الجزء السفلي من A؛ إذا كان UPLO = 'L'
!>          يتم تشكيل الجزء المثلث السفلي من المعكوس ولا يتم الرجوع إلى الجزء العلوي
!>          من A.
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرائد للمصفوفة A.  LDA >= max(1,N).
!>

IPIV

!>          IPIV هي مصفوفة INTEGER، ذات بعد (N)
!>          تفاصيل التبادلات وهيكل Q القطري كما تم تحديده بواسطة ZHETRF_ROOK.
!>

WORK

!>          WORK هي مصفوفة COMPLEX*16، ذات بعد (N)
!>

INFO

!>          INFO هو INTEGER
!>          = 0: خروج ناجح
!>          < 0: إذا كان INFO = -i، فإن المعامل i كان له قيمة غير قانونية
!>          > 0: إذا كان INFO = i، فإن D(i,i) = 0؛ المصفوفة منفردة ولا يمكن
!>               حساب معكوسها.
!>
