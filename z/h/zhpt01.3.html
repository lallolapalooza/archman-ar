subroutine zhpt01 (character uplo, integer n, complex*16, dimension( * ) a, complex*16, dimension( * ) afac, integer, dimension( * ) ipiv, complex*16, dimension( ldc, * ) c, integer ldc, double precision, dimension( * ) rwork, double precision resid)

ZHPT01

الغرض:

!>
!> ZHPT01 يعيد بناء مصفوفة هرميتية غير محددة A من تفكيكها L*D*L' أو U*D*U'
!> ويحسب الباقي
!>    norm( C - A ) / ( N * norm(A) * EPS ),
!> حيث C هي المصفوفة المعاد بناؤها، EPS هي القيمة التقريبية للآلة،
!> L' هي النقل المرافق لـ L، و U' هي النقل المرافق لـ U.
!>

المعاملات

UPLO 
!>          UPLO is CHARACTER*1
!>          يحدد ما إذا كان الجزء الثلاثي العلوي أو السفلي للمصفوفة
!>          الهرميتية A مخزنًا:
!>          = 'U':  ثلاثي علوي
!>          = 'L':  ثلاثي سفلي
!>

N

!>          N is INTEGER
!>          عدد الصفوف والأعمدة للمصفوفة A.  N >= 0.
!>

A

!>          A is COMPLEX*16 array, dimension (N*(N+1)/2)
!>          المصفوفة الهرميتية الأصلية A، مخزنة كمصفوفة ثلاثية
!>          مكدسة.
!>

AFAC

!>          AFAC is COMPLEX*16 array, dimension (N*(N+1)/2)
!>          الشكل المفكك للمصفوفة A، مخزن كمصفوفة ثلاثية مكدسة.
!>          يحتوي AFAC على المصفوفة القطرية الكتلية D والمضاعفات المستخدمة
!>          للحصول على العامل L أو U من تفكيك L*D*L' أو U*D*U' كما تم حسابه
!>          بواسطة ZHPTRF.
!>

IPIV

!>          IPIV is INTEGER array, dimension (N)
!>          مؤشرات المحور من ZHPTRF.
!>

C

!>          C is COMPLEX*16 array, dimension (LDC,N)
!>

LDC

!>          LDC is INTEGER
!>          البعد الرئيسي للمصفوفة C.  LDC >= max(1,N).
!>

RWORK

!>          RWORK is DOUBLE PRECISION array, dimension (N)
!>

RESID

!>          RESID is DOUBLE PRECISION
!>          If UPLO = 'L', norm(L*D*L' - A) / ( N * norm(A) * EPS )
!>          If UPLO = 'U', norm(U*D*U' - A) / ( N * norm(A) * EPS )
!>
