subroutine zhetf2_rook (uplo, n, a, lda, ipiv, info)
ZHETF2_ROOK يحسب تحليل مصفوفة هيرميتية غير محددة معقدة باستخدام طريقة التنقيط القطري المحصور لـ Bunch-Kaufman ("rook") (خوارزمية غير محظورة).

توثيق الوظيفة/الروتين الفرعي

subroutine zhetf2_rook (character uplo, integer n, complex*16, dimension( lda, * ) a, integer lda, integer, dimension( * ) ipiv, integer info)

ZHETF2_ROOK يحسب تحليل مصفوفة هيرميتية غير محددة معقدة باستخدام طريقة التنقيط القطري المحصور لـ Bunch-Kaufman ("rook") (خوارزمية غير محظورة).

الغرض:

!>
!> ZHETF2_ROOK يحسب تحليل مصفوفة هيرميتية معقدة A
!> باستخدام طريقة التنقيط القطري المحصور لـ Bunch-Kaufman:
!>
!>    A = U*D*U**H  أو  A = L*D*L**H
!>
!> حيث U (أو L) هو ناتج ضرب مصفوفة تبديل ومصفوفة مثلثية عليا (سفلى)
!> وحدوية، U**H هو المرافق المتخالف لـ U، و D هي مصفوفة هيرميتية وقطرية
!> كتلية ذات كتل قطرية بحجم 1x1 و 2x2.
!>
!> هذه هي النسخة غير المحظورة من الخوارزمية، وتستدعي BLAS المستوى 2.
!>

المعاملات

UPLO 
!>          UPLO هو CHARACTER*1
!>          يحدد ما إذا كان الجزء المثلثي العلوي أو السفلي للمصفوفة الهيرميتية A
!>          مخزنًا:
!>          = 'U':  مثلثي علوي
!>          = 'L':  مثلثي سفلي
!>

N

!>          N هو INTEGER
!>          رتبة المصفوفة A.  N >= 0.
!>

A

!>          A هي مصفوفة COMPLEX*16، البعد (LDA,N)
!>          عند الإدخال، المصفوفة الهيرميتية A.  إذا كان UPLO = 'U'، فإن الجزء
!>          المثلثي العلوي من A بحجم n × n يحتوي على الجزء المثلثي العلوي
!>          للمصفوفة A، والجزء المثلثي السفلي الصارم لـ A لا تتم الإشارة إليه.
!>          إذا كان UPLO = 'L'، فإن الجزء المثلثي السفلي من A بحجم n × n يحتوي
!>          على الجزء المثلثي السفلي للمصفوفة A، والجزء المثلثي العلوي الصارم
!>          لـ A لا تتم الإشارة إليه.
!>
!>          عند الخروج، مصفوفة القطر الكتلي D والعوامل المستخدمة
!>          للحصول على العامل U أو L (انظر أدناه لمزيد من التفاصيل).
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرائد للمصفوفة A.  LDA >= max(1,N).
!>

IPIV

!>          IPIV هي مصفوفة INTEGER، البعد (N)
!>          تفاصيل التبادلات والبنية الكتلية لـ D.
!>
!>          إذا كان UPLO = 'U':
!>             إذا كان IPIV(k) > 0، فقد تم تبديل الصفوف والأعمدة k و IPIV(k)
!>             وكان D(k,k) كتلة قطرية بحجم 1x1.
!>
!>             إذا كان IPIV(k) < 0 و IPIV(k-1) < 0، فقد تم تبديل الصفوف
!>             والأعمدة k و -IPIV(k) وتم تبديل الصفوف
!>             والأعمدة k-1 و -IPIV(k-1)،
!>             وكان D(k-1:k,k-1:k) كتلة قطرية بحجم 2x2.
!>
!>          إذا كان UPLO = 'L':
!>             إذا كان IPIV(k) > 0، فقد تم تبديل الصفوف والأعمدة k و IPIV(k)
!>             وكان D(k,k) كتلة قطرية بحجم 1x1.
!>
!>             إذا كان IPIV(k) < 0 و IPIV(k+1) < 0، فقد تم تبديل الصفوف
!>             والأعمدة k و -IPIV(k) وتم تبديل الصفوف
!>             والأعمدة k+1 و -IPIV(k+1)،
!>             وكان D(k:k+1,k:k+1) كتلة قطرية بحجم 2x2.
!>

INFO

!>          INFO هو INTEGER
!>          = 0: خروج ناجح
!>          < 0: إذا كان INFO = -k، فقد كان الوسيط k ذو قيمة غير قانونية
!>          > 0: إذا كان INFO = k، فإن D(k,k) يساوي صفرًا تمامًا.  تم إكمال
!>               التحليل، ولكن المصفوفة القطرية الكتلية D منفردة تمامًا،
!>               وسينتج عن القسمة على صفر إذا تم استخدامها لحل نظام
!>               من المعادلات.
!>

تفاصيل إضافية:

!>
!>  إذا كان UPLO = 'U'، فإن A = U*D*U**H، حيث
!>     U = P(n)*U(n)* ... *P(k)U(k)* ...,
!>  أي أن U هو ناتج ضرب الحدود P(k)*U(k)، حيث k ينقص من n إلى
!>  1 بخطوات 1 أو 2، و D هي مصفوفة قطرية كتلية ذات كتل قطرية
!>  بحجم 1x1 و 2x2 هي D(k).  P(k) هي مصفوفة تبديل كما
!>  تم تعريفها بواسطة IPIV(k)، و U(k) هي مصفوفة مثلثية عليا وحدوية، بحيث
!>  إذا كانت الكتلة القطرية D(k) من الرتبة s (s = 1 أو 2)، فإن
!>
!>             (   I    v    0   )   k-s
!>     U(k) =  (   0    I    0   )   s
!>             (   0    0    I   )   n-k
!>                k-s   s   n-k
!>
!>  إذا كان s = 1، فإن D(k) تحل محل A(k,k)، و v تحل محل A(1:k-1,k).
!>  إذا كان s = 2، فإن المثلث العلوي لـ D(k) يحل محل A(k-1,k-1)، A(k-1,k)،
!>  و A(k,k)، و v تحل محل A(1:k-2,k-1:k).
!>
!>  إذا كان UPLO = 'L'، فإن A = L*D*L**H، حيث
!>     L = P(1)*L(1)* ... *P(k)*L(k)* ...,
!>  أي أن L هو ناتج ضرب الحدود P(k)*L(k)، حيث k يزداد من 1 إلى
!>  n بخطوات 1 أو 2، و D هي مصفوفة قطرية كتلية ذات كتل قطرية
!>  بحجم 1x1 و 2x2 هي D(k).  P(k) هي مصفوفة تبديل كما
!>  تم تعريفها بواسطة IPIV(k)، و L(k) هي مصفوفة مثلثية سفلى وحدوية، بحيث
!>  إذا كانت الكتلة القطرية D(k) من الرتبة s (s = 1 أو 2)، فإن
!>
!>             (   I    0     0   )  k-1
!>     L(k) =  (   0    I     0   )  s
!>             (   0    v     I   )  n-k-s+1
!>                k-1   s  n-k-s+1
!>
!>  إذا كان s = 1، فإن D(k) تحل محل A(k,k)، و v تحل محل A(k+1:n,k).
!>  إذا كان s = 2، فإن المثلث السفلي لـ D(k) يحل محل A(k,k)، A(k+1,k)،
!>  و A(k+1,k+1)، و v تحل محل A(k+2:n,k:k+1).
!>
