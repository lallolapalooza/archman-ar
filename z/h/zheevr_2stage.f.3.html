الإجراء الفرعي zheevr_2stage (jobz, range, uplo, n, a, lda, vl, vu, il, iu, abstol, m, w, z, ldz, isuppz, work, lwork, rwork, lrwork, iwork, liwork, info)
ZHEEVR_2STAGE يحسب القيم الذاتية، واختياريًا، المتجهات الذاتية اليسرى و/أو اليمنى لمصفوفات HE

وثائق الدالة/الإجراء الفرعي

الإجراء الفرعي zheevr_2stage (character jobz, character range, character uplo, integer n, complex*16, dimension( lda, * ) a, integer lda, double precision vl, double precision vu, integer il, integer iu, double precision abstol, integer m, double precision, dimension( * ) w, complex*16, dimension( ldz, * ) z, integer ldz, integer, dimension( * ) isuppz, complex*16, dimension( * ) work, integer lwork, double precision, dimension( * ) rwork, integer lrwork, integer, dimension( * ) iwork, integer liwork, integer info)

ZHEEVR_2STAGE يحسب القيم الذاتية، واختياريًا، المتجهات الذاتية اليسرى و/أو اليمنى لمصفوفات HE

الغرض:

!>
!> ZHEEVR_2STAGE يحسب القيم الذاتية المختارة، واختياريًا، المتجهات الذاتية
!> لمصفوفة هيرميتية معقدة A باستخدام تقنية المرحلتين
!> للاختزال إلى الشكل ثلاثي الأقطار. يمكن اختيار القيم الذاتية والمتجهات الذاتية
!> بتحديد إما نطاق من القيم أو نطاق من
!> الفهارس للقيم الذاتية المطلوبة.
!>
!> يقوم ZHEEVR_2STAGE أولاً باختزال المصفوفة A إلى الشكل ثلاثي الأقطار T باستدعاء
!> ZHETRD. ثم، كلما أمكن، يستدعي ZHEEVR_2STAGE الدالة ZSTEMR لحساب
!> الطيف الذاتي باستخدام التمثيلات القوية نسبيًا. تحسب ZSTEMR
!> القيم الذاتية بخوارزمية dqds، بينما تُحسب المتجهات الذاتية
!> المتعامدة من تمثيلات L D L^T مختلفة
!> (المعروفة أيضًا بالتمثيلات القوية نسبيًا). يتم تجنب
!> تعامد جرام-شميدت قدر الإمكان. وبشكل أكثر تحديدًا،
!> خطوات الخوارزمية المختلفة هي كما يلي.
!>
!> لكل كتلة (مصفوفة فرعية) غير مختزلة من T،
!>    (أ) احسب T - sigma I  = L D L^T، بحيث تحدد L و D
!>        جميع القيم الذاتية المطلوبة بدقة نسبية عالية.
!>        هذا يعني أن التغييرات النسبية الصغيرة في عناصر D و L
!>        تسبب فقط تغييرات نسبية صغيرة في القيم الذاتية و
!>        المتجهات الذاتية. التمثيل القياسي (غير المُحلَّل) للمصفوفة
!>        ثلاثية الأقطار T لا يمتلك هذه الخاصية بشكل عام.
!>    (ب) احسب القيم الذاتية بدقة مناسبة.
!>        إذا كانت المتجهات الذاتية مطلوبة، تحقق الخوارزمية الدقة الكاملة
!>        للقيم الذاتية المحسوبة فقط قبل
!>        حساب المتجهات المقابلة مباشرة، انظر الخطوتين ج) و د).
!>    (ج) لكل مجموعة من القيم الذاتية المتقاربة، اختر إزاحة جديدة
!>        قريبة من المجموعة، وابحث عن تحليل جديد، وقم بتحسين
!>        القيم الذاتية المُزاحة إلى دقة مناسبة.
!>    (د) لكل قيمة ذاتية ذات فصل نسبي كبير بما فيه الكفاية، احسب
!>        المتجه الذاتي المقابل بتكوين تحليل ملتوٍ
!>        كاشف للرتبة. عد إلى (ج) لأي مجموعات متبقية.
!>
!> يمكن تحديد الدقة المطلوبة للمخرجات بواسطة معامل الإدخال
!> ABSTOL.
!>
!> لمزيد من التفاصيل، انظر وثائق ZSTEMR و:
!> - Inderjit S. Dhillon and Beresford N. Parlett: 
!>   Linear Algebra and its Applications, 387(1), pp. 1-28, August 2004.
!> - Inderjit Dhillon and Beresford Parlett:  SIAM Journal on Matrix Analysis and Applications, Vol. 25,
!>   2004.  Also LAPACK Working Note 154.
!> - Inderjit Dhillon: ,
!>   Computer Science Division Technical Report No. UCB/CSD-97-971,
!>   UC Berkeley, May 1997.
!>
!>
!> ملاحظة 1 : يستدعي ZHEEVR_2STAGE الدالة ZSTEMR عندما يُطلب الطيف الكامل
!> على الأجهزة التي تتوافق مع معيار الفاصلة العائمة ieee-754.
!> يستدعي ZHEEVR_2STAGE الدالتين DSTEBZ و ZSTEIN على الأجهزة غير المتوافقة مع ieee
!> وعند تقديم طلبات طيف جزئي.
!>
!> قد يؤدي التنفيذ العادي لـ ZSTEMR إلى إنشاء قيم NaN وقيم لا نهائية و
!> بالتالي قد يتوقف بسبب استثناء الفاصلة العائمة في البيئات
!> التي لا تتعامل مع قيم NaN والقيم اللانهائية بطريقة ieee الافتراضية
!> القياسية.
!>

المعاملات

JOBZ 
!>          JOBZ هو CHARACTER*1
!>          = 'N':  حساب القيم الذاتية فقط؛
!>          = 'V':  حساب القيم الذاتية والمتجهات الذاتية.
!>                  غير متاح في هذا الإصدار.
!>

RANGE

!>          RANGE هو CHARACTER*1
!>          = 'A': سيتم إيجاد جميع القيم الذاتية.
!>          = 'V': سيتم إيجاد جميع القيم الذاتية في الفترة نصف المفتوحة (VL,VU]
!>                 .
!>          = 'I': سيتم إيجاد القيم الذاتية من IL-th إلى IU-th.
!>          بالنسبة لـ RANGE = 'V' أو 'I' و IU - IL < N - 1، يتم استدعاء DSTEBZ و
!>          ZSTEIN
!>

UPLO

!>          UPLO هو CHARACTER*1
!>          = 'U':  يتم تخزين المثلث العلوي من A؛
!>          = 'L':  يتم تخزين المثلث السفلي من A.
!>

N

!>          N هو INTEGER
!>          رتبة المصفوفة A.  N >= 0.
!>

A

!>          A هو مصفوفة COMPLEX*16، البعد (LDA, N)
!>          عند الإدخال، المصفوفة الهيرميتية A.  إذا كان UPLO = 'U'، فإن
!>          الجزء العلوي المثلثي N-by-N الرئيسي من A يحتوي على
!>          الجزء العلوي المثلثي من المصفوفة A.  إذا كان UPLO = 'L'،
!>          فإن الجزء السفلي المثلثي N-by-N الرئيسي من A يحتوي على
!>          الجزء السفلي المثلثي من المصفوفة A.
!>          عند الخروج، يتم تدمير المثلث السفلي (إذا كان UPLO='L') أو المثلث
!>          العلوي (إذا كان UPLO='U') من A، بما في ذلك القطر.
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرئيسي للمصفوفة A.  LDA >= max(1,N).
!>

VL

!>          VL هو DOUBLE PRECISION
!>          إذا كان RANGE='V'، الحد الأدنى للفترة
!>          التي سيتم البحث فيها عن القيم الذاتية. VL < VU.
!>          لا يُشار إليه إذا كان RANGE = 'A' أو 'I'.
!>

VU

!>          VU هو DOUBLE PRECISION
!>          إذا كان RANGE='V'، الحد الأعلى للفترة
!>          التي سيتم البحث فيها عن القيم الذاتية. VL < VU.
!>          لا يُشار إليه إذا كان RANGE = 'A' أو 'I'.
!>

IL

!>          IL هو INTEGER
!>          إذا كان RANGE='I'، فهرس
!>          أصغر قيمة ذاتية سيتم إرجاعها.
!>          1 <= IL <= IU <= N، إذا كان N > 0؛ IL = 1 و IU = 0 إذا كان N = 0.
!>          لا يُشار إليه إذا كان RANGE = 'A' أو 'V'.
!>

IU

!>          IU هو INTEGER
!>          إذا كان RANGE='I'، فهرس
!>          أكبر قيمة ذاتية سيتم إرجاعها.
!>          1 <= IL <= IU <= N، إذا كان N > 0؛ IL = 1 و IU = 0 إذا كان N = 0.
!>          لا يُشار إليه إذا كان RANGE = 'A' أو 'V'.
!>

ABSTOL

!>          ABSTOL هو DOUBLE PRECISION
!>          تفاوت الخطأ المطلق للقيم الذاتية.
!>          تُقبل القيمة الذاتية التقريبية على أنها متقاربة
!>          عندما يُحدد أنها تقع في فترة [a,b]
!>          بعرض أقل من أو يساوي
!>
!>                  ABSTOL + EPS *   max( |a|,|b| ) ,
!>
!>          حيث EPS هي دقة الآلة.  إذا كان ABSTOL أقل من
!>          أو يساوي الصفر، فسيتم استخدام EPS*|T| بدلاً منه،
!>          حيث |T| هو المعيار 1 للمصفوفة ثلاثية الأقطار التي تم الحصول عليها
!>          باختزال A إلى الشكل ثلاثي الأقطار.
!>
!>          انظر  بواسطة Demmel و
!>          Kahan، LAPACK Working Note #3.
!>
!>          إذا كانت الدقة النسبية العالية مهمة، فاضبط ABSTOL على
!>          DLAMCH( 'Safe minimum' ).  القيام بذلك سيضمن أن
!>          القيم الذاتية تُحسب بدقة عالية

هذا الموقع غير مرتبط بـ Arch Linux ولا تابع له بأي شكل من الأشكال. المحتوى هو ترجمة فقط لصفحات دليل Arch المنشورة على man.archlinux.org.

الدقة النسبية عندما !> يكون ذلك ممكنًا في الإصدارات المستقبلية. الكود الحالي لا !> يقدم أي ضمانات بشأن الدقة النسبية العالية، لكن !> الإصدارات المستقبلية ستفعل. انظر J. Barlow وJ. Demmel، !> ، LAPACK Working Note #7، لمناقشة !> المصفوفات التي تعرّف قيمها الذاتية بدقة نسبية !> عالية. !>

M

!>          M هو INTEGER
!>          العدد الإجمالي للقيم الذاتية التي تم العثور عليها.  0 <= M <= N.
!>          إذا كان RANGE = 'A'، فإن M = N، وإذا كان RANGE = 'I'، فإن M = IU-IL+1.
!>

W

!>          W هو مصفوفة DOUBLE PRECISION، البُعد (N)
!>          تحتوي العناصر M الأولى على القيم الذاتية المحددة بترتيب
!>          تصاعدي.
!>

Z

!>          Z هو مصفوفة COMPLEX*16، البُعد (LDZ, max(1,M))
!>          إذا كان JOBZ = 'V'، فإنه إذا كان INFO = 0، فإن الأعمدة M الأولى من Z
!>          تحتوي على المتجهات الذاتية المتعامدة المعيارية للمصفوفة A
!>          المقابلة للقيم الذاتية المحددة، حيث يحمل العمود i-th
!>          من Z المتجه الذاتي المرتبط بـ W(i).
!>          إذا كان JOBZ = 'N'، فإن Z لا يُشار إليه.
!>          ملاحظة: يجب على المستخدم التأكد من توفير max(1,M) عمودًا على الأقل
!>          في المصفوفة Z؛ إذا كان RANGE = 'V'، فإن القيمة الدقيقة لـ M
!>          غير معروفة مسبقًا ويجب استخدام حد أعلى.
!>

LDZ

!>          LDZ هو INTEGER
!>          البُعد الرئيسي للمصفوفة Z.  LDZ >= 1، وإذا كان
!>          JOBZ = 'V'، فإن LDZ >= max(1,N).
!>

ISUPPZ

!>          ISUPPZ هو مصفوفة INTEGER، البُعد ( 2*max(1,M) )
!>          دعم المتجهات الذاتية في Z، أي المؤشرات
!>          التي تشير إلى العناصر غير الصفرية في Z. المتجه الذاتي i-th
!>          غير صفري فقط في العناصر ISUPPZ( 2*i-1 ) حتى
!>          ISUPPZ( 2*i ). هذا مخرج من ZSTEMR (مصفوفة
!>          ثلاثية الأقطار). دعم المتجهات الذاتية للمصفوفة A عادةً
!>          يكون 1:N بسبب التحويلات الوحدوية التي يطبقها ZUNMTR.
!>          مُنفَّذ فقط لـ RANGE = 'A' أو 'I' و IU - IL = N - 1
!>

WORK

!>          WORK هو مصفوفة COMPLEX*16، البُعد (MAX(1,LWORK))
!>          عند الخروج، إذا كان INFO = 0، فإن WORK(1) يُرجع LWORK الأمثل.
!>

LWORK

!>          LWORK هو INTEGER
!>          بُعد المصفوفة WORK.
!>          إذا كان N <= 1،               يجب أن يكون LWORK على الأقل 1.
!>          إذا كان JOBZ = 'N' و N > 1، يجب الاستعلام عن LWORK.
!>                                   LWORK = MAX(1, 26*N, dimension) حيث
!>                                   dimension = max(stage1,stage2) + (KD+1)*N + N
!>                                             = N*KD + N*max(KD+1,FACTOPTNB)
!>                                               + max(2*KD*KD, KD*NTHREADS)
!>                                               + (KD+1)*N + N
!>                                   حيث KD هو حجم التجميع للاختزال،
!>                                   FACTOPTNB هو التجميع المستخدم بواسطة خوارزمية QR أو LQ،
!>                                   عادةً FACTOPTNB=128 خيار جيد
!>                                   NTHREADS هو عدد الخيوط المستخدمة عند
!>                                   تمكين ترجمة openMP، وإلا =1.
!>          إذا كان JOBZ = 'V' و N > 1، يجب الاستعلام عن LWORK. غير متاح بعد
!>
!>          إذا كان LWORK = -1، فسيُفترض استعلام عن مساحة العمل؛ الروتين
!>          يحسب فقط الأحجام المثلى لمصفوفات WORK وRWORK و
!>          IWORK، ويُرجع هذه القيم كأول مدخلات في
!>          مصفوفات WORK وRWORK وIWORK، ولا تُصدر رسالة خطأ
!>          متعلقة بـ LWORK أو LRWORK أو LIWORK بواسطة XERBLA.
!>

RWORK

!>          RWORK هو مصفوفة DOUBLE PRECISION، البُعد (MAX(1,LRWORK))
!>          عند الخروج، إذا كان INFO = 0، فإن RWORK(1) يُرجع LRWORK الأمثل
!>          (والأدنى).
!>

LRWORK

!>          LRWORK هو INTEGER
!>          طول المصفوفة RWORK.
!>          إذا كان N <= 1، فإن LRWORK >= 1، وإلا LRWORK >= 24*N.
!>
!>          إذا كان LRWORK = -1، فسيُفترض استعلام عن مساحة العمل؛ الروتين
!>          يحسب فقط الأحجام المثلى لمصفوفات WORK وRWORK
!>          وIWORK، ويُرجع هذه القيم كأول مدخلات
!>          في مصفوفات WORK وRWORK وIWORK، ولا تُصدر رسالة خطأ
!>          متعلقة بـ LWORK أو LRWORK أو LIWORK بواسطة XERBLA.
!>

IWORK

!>          IWORK هو مصفوفة INTEGER، البُعد (MAX(1,LIWORK))
!>          عند الخروج، إذا كان INFO = 0، فإن IWORK(1) يُرجع LIWORK الأمثل
!>          (والأدنى).
!>

LIWORK

!>          LIWORK هو INTEGER
!>          بُعد المصفوفة IWORK.
!>          إذا كان N <= 1، فإن LIWORK >= 1، وإلا LIWORK >= 10*N.
!>
!>          إذا كان LIWORK = -1، فسيُفترض استعلام عن مساحة العمل؛ الروتين
!>          يحسب فقط الأحجام المثلى لمصفوفات WORK وRWORK
!>          وIWORK، ويُرجع هذه القيم كأول مدخلات
!>          في مصفوفات WORK وRWORK وIWORK، ولا تُصدر رسالة خطأ
!>          متعلقة بـ LWORK أو LRWORK أو LIWORK بواسطة XERBLA.
!>

INFO

!>          INFO هو INTEGER
!>          = 0:  خروج ناجح
!>          < 0:  إذا كان INFO = -i، فإن الوسيطة i-th لها قيمة غير قانونية
!>          > 0:  خطأ داخلي
!>
