subroutine zhegv_2stage (integer itype, character jobz, character uplo, integer n, complex*16, dimension( lda, * ) a, integer lda, complex*16, dimension( ldb, * ) b, integer ldb, double precision, dimension( * ) w, complex*16, dimension( * ) work, integer lwork, double precision, dimension( * ) rwork, integer info)

ZHEGV_2STAGE

الغرض:

!>
!> ZHEGV_2STAGE يحسب جميع القيم الذاتية، واختياريًا، المتجهات الذاتية
!> لمشكلة القيم الذاتية الهرمايتية المعممة المحددة، بالشكل
!> A*x=(lambda)*B*x,  A*Bx=(lambda)*x,  أو B*A*x=(lambda)*x.
!> هنا يُفترض أن A و B هرميتيان وأن B أيضًا
!> موجب محدد.
!> يستخدم هذا الروتين الفرعي تقنية المرحلتين (2stage) للاختزال إلى شكل ثلاثي القطر
!> الذي أظهر أداءً أعلى على البنى الحديثة وللأحجام الكبيرة N>2000.
!>

المعلمات:

ITYPE 
!>          ITYPE هو عدد صحيح (INTEGER)
!>          يحدد نوع المشكلة المطلوب حلها:
!>          = 1:  A*x = (lambda)*B*x
!>          = 2:  A*B*x = (lambda)*x
!>          = 3:  B*A*x = (lambda)*x
!>

JOBZ

!>          JOBZ هو حرف (CHARACTER*1)
!>          = 'N':  احسب القيم الذاتية فقط؛
!>          = 'V':  احسب القيم الذاتية والمتجهات الذاتية.
!>                  غير متوفر في هذا الإصدار.
!>

UPLO

!>          UPLO هو حرف (CHARACTER*1)
!>          = 'U':  المثلث العلوي للمصفوفتين A و B مخزن؛
!>          = 'L':  المثلث السفلي للمصفوفتين A و B مخزن.
!>

N

!>          N هو عدد صحيح (INTEGER)
!>          رتبة المصفوفتين A و B. N >= 0.
!>

A

!>          A هو مصفوفة ذات أعداد مركبة مزدوجة الدقة (COMPLEX*16)
!>          بحجم (LDA, N).
!>          عند الدخول، تحتوي المصفوفة A الهرمايتية. إذا كان UPLO = 'U'،
!>          فإن الجزء العلوي الثلاثي من A بأبعاد N*N يحتوي على
!>          الجزء العلوي الثلاثي للمصفوفة A. إذا كان UPLO = 'L'،
!>          فإن الجزء السفلي الثلاثي من A بأبعاد N*N يحتوي على
!>          الجزء السفلي الثلاثي للمصفوفة A.
!>
!>          عند الخروج، إذا كان JOBZ = 'V'، وإذا كان INFO = 0، فإن A تحتوي على
!>          المصفوفة Z للمتجهات الذاتية. يتم تطبيع المتجهات الذاتية كالتالي:
!>          إذا كان ITYPE = 1 أو 2، فإن Z**H*B*Z = I؛
!>          إذا كان ITYPE = 3، فإن Z**H*inv(B)*Z = I.
!>          إذا كان JOBZ = 'N'، فعند الخروج، يتم تدمير المثلث العلوي (إذا كان UPLO='U')
!>          أو المثلث السفلي (إذا كان UPLO='L') من A، بما في ذلك القطر.
!>

LDA

!>          LDA هو عدد صحيح (INTEGER)
!>          البعد الرائد للمصفوفة A. LDA >= max(1,N).
!>

B

!>          B هو مصفوفة ذات أعداد مركبة مزدوجة الدقة (COMPLEX*16)
!>          بحجم (LDB, N).
!>          عند الدخول، المصفوفة B الهرمايتية والموجبة المحددة.
!>          إذا كان UPLO = 'U'، فإن الجزء العلوي الثلاثي من B بأبعاد N*N يحتوي على
!>          الجزء العلوي الثلاثي للمصفوفة B.
!>          إذا كان UPLO = 'L'، فإن الجزء السفلي الثلاثي من B بأبعاد N*N يحتوي على
!>          الجزء السفلي الثلاثي للمصفوفة B.
!>
!>          عند الخروج، إذا كان INFO <= N، فإن الجزء من B الذي يحتوي على المصفوفة
!>          يتم استبداله بالعامل الثلاثي U أو L من تحليل Cholesky
!>          B = U**H*U أو B = L*L**H.
!>

LDB

!>          LDB هو عدد صحيح (INTEGER)
!>          البعد الرائد للمصفوفة B. LDB >= max(1,N).
!>

W

!>          W هي مصفوفة ذات أعداد حقيقية مزدوجة الدقة (DOUBLE PRECISION)
!>          بحجم (N).
!>          إذا كان INFO = 0، فإن القيم الذاتية مرتبة تصاعديًا.
!>

WORK

!>          WORK هي مصفوفة ذات أعداد مركبة مزدوجة الدقة (COMPLEX*16)
!>          بحجم (MAX(1,LWORK)).
!>          عند الخروج، إذا كان INFO = 0، فإن WORK(1) يعيد LWORK الأمثل.
!>

LWORK

!>          LWORK هو عدد صحيح (INTEGER)
!>          طول المصفوفة WORK. LWORK >= 1، عندما N <= 1؛
!>          بخلاف ذلك
!>          إذا كان JOBZ = 'N' و N > 1، يجب الاستعلام عن LWORK.
!>                                   LWORK = MAX(1, dimension) حيث
!>                                   dimension = max(stage1,stage2) + (KD+1)*N + N
!>                                             = N*KD + N*max(KD+1,FACTOPTNB)
!>                                               + max(2*KD*KD, KD*NTHREADS)
!>                                               + (KD+1)*N + N
!>                                   حيث KD هو حجم التجميع للاختزال،
!>                                   FACTOPTNB هو التجميع المستخدم بواسطة خوارزمية QR أو LQ
!>                                   عادةً ما يكون FACTOPTNB=128 اختيارًا جيدًا.
!>                                   NTHREADS هو عدد الخيوط المستخدمة عند
!>                                   تمكين الترجمة المسبقة (openMP)، وإلا =1.
!>          إذا كان JOBZ = 'V' و N > 1، يجب الاستعلام عن LWORK. غير متوفر حاليًا.
!>
!>          إذا كان LWORK = -1، فيُفترض وجود استعلام مساحة عمل؛ الروتين الفرعي
!>          يحسب فقط الحجم الأمثل لمصفوفة WORK، ويعيد
!>          هذه القيمة كالإدخال الأول لمصفوفة WORK، ولا يتم إصدار رسالة خطأ
!>          متعلقة بـ LWORK بواسطة XERBLA.
!>

RWORK

!>          RWORK هي مصفوفة ذات أعداد حقيقية مزدوجة الدقة (DOUBLE PRECISION)
!>          بحجم (max(1, 3*N-2)).
!>

INFO

!>          INFO هو عدد صحيح (INTEGER)
!>          = 0:  انتهاء ناجح
!>          < 0:  إذا كان INFO = -i، فإن المعامل i كان له قيمة غير قانونية
!>          > 0:  عادت ZPOTRF أو ZHEEV برمز خطأ:
!>             <= N:  إذا كان INFO = i، فإن ZHEEV فشلت في التقارب؛
!>                    عناصر خارج القطر i لمصفوفة ثلاثية الأقطار وسيطة
!>                    لم تتقارب إلى الصفر؛
!>             > N:   إذا كان INFO = N + i، لـ 1 <= i <= N، فإن
!>                    المُحدِّد الرئيسي الرائد من الرتبة i للمصفوفة B ليس موجبًا.
!>                    لم يكن من الممكن إكمال تحليل B
!>                    ولم يتم حساب أي قيم ذاتية أو متجهات ذاتية.
!>
