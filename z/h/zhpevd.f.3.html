توثيق الوظيفة/الإجراء الفرعي

subroutine zhpevd (character jobz, character uplo, integer n, complex*16, dimension( * ) ap, double precision, dimension( * ) w, complex*16, dimension( ldz, * ) z, integer ldz, complex*16, dimension( * ) work, integer lwork, double precision, dimension( * ) rwork, integer lrwork, integer, dimension( * ) iwork, integer liwork, integer info)

ZHPEVD يحسب القيم الذاتية، وبشكل اختياري، المتجهات الذاتية اليسرى و/أو اليمنى لمصفوفة هيرميتية معقدة A بتخزين مكدس. إذا تم طلب المتجهات الذاتية، فإنها تستخدم خوارزمية الانقسام والفتح.

الغرض:

!>
!> ZHPEVD يحسب جميع القيم الذاتية، وبشكل اختياري، المتجهات الذاتية لمصفوفة هيرميتية معقدة A بتخزين مكدس. إذا تم طلب المتجهات الذاتية، فإنها تستخدم خوارزمية الانقسام والفتح.
!>
!>

المعلمات

JOBZ 
!>          JOBZ هو CHARACTER*1
!>          = 'N':  حساب القيم الذاتية فقط؛
!>          = 'V':  حساب القيم الذاتية والمتجهات الذاتية.
!>

UPLO

!>          UPLO هو CHARACTER*1
!>          = 'U':  يتم تخزين المثلث العلوي من A؛
!>          = 'L':  يتم تخزين المثلث السفلي من A.
!>

N

!>          N هو INTEGER
!>          رتبة المصفوفة A.  N >= 0.
!>

AP

!>          AP هو مصفوفة COMPLEX*16، بُعدها (N*(N+1)/2)
!>          عند الإدخال، المثلث العلوي أو السفلي للمصفوفة الهيرميتية
!>          A، مكدس عمودياً في مصفوفة خطية. يتم تخزين العمود j من A
!>          في المصفوفة AP كما يلي:
!>          إذا كان UPLO = 'U'، فإن AP(i + (j-1)*j/2) = A(i,j) لـ 1<=i<=j؛
!>          إذا كان UPLO = 'L'، فإن AP(i + (j-1)*(2*n-j)/2) = A(i,j) لـ j<=i<=n.
!>
!>          عند الخروج، يتم استبدال AP بقيم تم إنشاؤها أثناء
!>          الاختزال إلى شكل ثلاثي الأقطار. إذا كان UPLO = 'U'، فإن الأقطار
!>          والفوق قطري الأول للمصفوفة ثلاثية الأقطار T تحل محل
!>          العناصر المقابلة في A، وإذا كان UPLO = 'L'، فإن
!>          الأقطار والفوق قطري الأول لـ T تحل محل
!>          العناصر المقابلة في A.
!>

W

!>          W هي مصفوفة DOUBLE PRECISION، بُعدها (N)
!>          إذا كان INFO = 0، فإن القيم الذاتية مرتبة تصاعديًا.
!>

Z

!>          Z هي مصفوفة COMPLEX*16، بُعدها (LDZ, N)
!>          إذا كان JOBZ = 'V'، فعندما يكون INFO = 0، تحتوي Z على المتجهات الذاتية
!>          المتعامدة للمصفوفة A، حيث يحمل العمود i من Z
!>          المتجه الذاتي المرتبط بـ W(i).
!>          إذا كان JOBZ = 'N'، فلا يتم الرجوع إلى Z.
!>

LDZ

!>          LDZ هو INTEGER
!>          البعد الرائد للمصفوفة Z. LDZ >= 1، وإذا كان
!>          JOBZ = 'V'، فإن LDZ >= max(1,N).
!>

WORK

!>          WORK هي مصفوفة COMPLEX*16، بُعدها (MAX(1,LWORK))
!>          عند الخروج، إذا كان INFO = 0، فإن WORK(1) يعيد LWORK المطلوب.
!>

LWORK

!>          LWORK هو INTEGER
!>          بُعد المصفوفة WORK.
!>          إذا كان N <= 1، يجب أن تكون LWORK على الأقل 1.
!>          إذا كان JOBZ = 'N' و N > 1، يجب أن تكون LWORK على الأقل N.
!>          إذا كان JOBZ = 'V' و N > 1، يجب أن تكون LWORK على الأقل 2*N.
!>
!>          إذا كان LWORK = -1، فسيتم افتراض استعلام مساحة العمل؛ تقوم الإجراءات الفرعية
!>          بحساب أحجام مساحات العمل المطلوبة للمصفوفات WORK و RWORK و
!>          IWORK فقط، وتعيد هذه القيم كإدخالات أولى في
!>          المصفوفات WORK و RWORK و IWORK، ولا يتم إصدار أي رسالة خطأ
!>          متعلقة بـ LWORK أو LRWORK أو LIWORK بواسطة XERBLA.
!>

RWORK

!>          RWORK هي مصفوفة DOUBLE PRECISION، بُعدها (MAX(1,LRWORK))
!>          عند الخروج، إذا كان INFO = 0، فإن RWORK(1) يعيد LRWORK المطلوب.
!>

LRWORK

!>          LRWORK هو INTEGER
!>          بُعد المصفوفة RWORK.
!>          إذا كان N <= 1، يجب أن تكون LRWORK على الأقل 1.
!>          إذا كان JOBZ = 'N' و N > 1، يجب أن تكون LRWORK على الأقل N.
!>          إذا كان JOBZ = 'V' و N > 1، يجب أن تكون LRWORK على الأقل
!>                    1 + 5*N + 2*N**2.
!>
!>          إذا كان LRWORK = -1، فسيتم افتراض استعلام مساحة العمل؛ تقوم الإجراءات الفرعية
!>          بحساب أحجام مساحات العمل المطلوبة للمصفوفات WORK و RWORK و
!>          IWORK فقط، وتعيد هذه القيم كإدخالات أولى في
!>          المصفوفات WORK و RWORK و IWORK، ولا يتم إصدار أي رسالة خطأ
!>          متعلقة بـ LWORK أو LRWORK أو LIWORK بواسطة XERBLA.
!>

IWORK

!>          IWORK هي مصفوفة INTEGER، بُعدها (MAX(1,LIWORK))
!>          عند الخروج، إذا كان INFO = 0، فإن IWORK(1) يعيد LIWORK المطلوب.
!>

LIWORK

!>          LIWORK هو INTEGER
!>          بُعد المصفوفة IWORK.
!>          إذا كان JOBZ  = 'N' أو N <= 1، يجب أن تكون LIWORK على الأقل 1.
!>          إذا كان JOBZ  = 'V' و N > 1، يجب أن تكون LIWORK على الأقل 3 + 5*N.
!>
!>          إذا كان LIWORK = -1، فسيتم افتراض استعلام مساحة العمل؛ تقوم الإجراءات الفرعية
!>          بحساب أحجام مساحات العمل المطلوبة للمصفوفات WORK و RWORK و
!>          IWORK فقط، وتعيد هذه القيم كإدخالات أولى في
!>          المصفوفات WORK و RWORK و IWORK، ولا يتم إصدار أي رسالة خطأ
!>          متعلقة بـ LWORK أو LRWORK أو LIWORK بواسطة XERBLA.
!>

INFO

!>          INFO هو INTEGER
!>          = 0:  خروج ناجح
!>          < 0:  إذا كان INFO = -i، فإن الوسيط i كان له قيمة غير قانونية.
!>          > 0:  إذا كان INFO = i، فشلت الخوارزمية في التقارب؛ i
!>                عناصر خارج القطر لمصفوفة ثلاثية الأقطار وسيطة
!>                لم تتقارب إلى الصفر.
!>
