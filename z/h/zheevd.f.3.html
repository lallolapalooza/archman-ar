توثيق الوظيفة/الإجراء الفرعي

subroutine zheevd (character jobz, character uplo, integer n, complex*16, dimension( lda, * ) a, integer lda, double precision, dimension( * ) w, complex*16, dimension( * ) work, integer lwork, double precision, dimension( * ) rwork, integer lrwork, integer, dimension( * ) iwork, integer liwork, integer info)

ZHEEVD يحسب القيم الذاتية، واختياريًا، المتجهات الذاتية اليسرى و/أو اليمنى لمصفوفات HE

الغرض:

!>
!> ZHEEVD يحسب جميع القيم الذاتية، واختياريًا، المتجهات الذاتية لمصفوفة هيرميتية A. إذا كانت المتجهات الذاتية مطلوبة، فإنه يستخدم خوارزمية تقسيم وغزو.
!>
!>

المعلمات

JOBZ 
!>          JOBZ is CHARACTER*1
!>          = 'N':  حساب القيم الذاتية فقط؛
!>          = 'V':  حساب القيم الذاتية والمتجهات الذاتية.
!>

UPLO

!>          UPLO is CHARACTER*1
!>          = 'U':  يتم تخزين الجزء العلوي من A؛
!>          = 'L':  يتم تخزين الجزء السفلي من A.
!>

N

!>          N is INTEGER
!>          رتبة المصفوفة A.  N >= 0.
!>

A

!>          A is COMPLEX*16 array, dimension (LDA, N)
!>          عند الدخول، المصفوفة الهيرميتية A.  إذا كان UPLO = 'U'، فإن
!>          الجزء العلوي المثلثي N-by-N الرائد من A يحتوي على
!>          الجزء العلوي المثلثي للمصفوفة A.  إذا كان UPLO = 'L'، فإن
!>          الجزء السفلي المثلثي N-by-N الرائد من A يحتوي على
!>          الجزء السفلي المثلثي للمصفوفة A.
!>          عند الخروج، إذا كان JOBZ = 'V'، وإذا كان INFO = 0، فإن A تحتوي على
!>          المتجهات الذاتية المتعامدة للمصفوفة A.
!>          إذا كان JOBZ = 'N'، فعند الخروج يتم تدمير الجزء السفلي المثلثي (إذا كان UPLO='L')
!>          أو الجزء العلوي المثلثي (إذا كان UPLO='U') من A، بما في ذلك
!>          القطر.
!>

LDA

!>          LDA is INTEGER
!>          البعد الرائد للمصفوفة A.  LDA >= max(1,N).
!>

W

!>          W is DOUBLE PRECISION array, dimension (N)
!>          إذا كان INFO = 0، فإن القيم الذاتية مرتبة تصاعديًا.
!>

WORK

!>          WORK is COMPLEX*16 array, dimension (MAX(1,LWORK))
!>          عند الخروج، إذا كان INFO = 0، فإن WORK(1) يعيد LWORK الأمثل.
!>

LWORK

!>          LWORK is INTEGER
!>          طول المصفوفة WORK.
!>          إذا كان N <= 1،                يجب أن يكون LWORK على الأقل 1.
!>          إذا كان JOBZ  = 'N' و N > 1، يجب أن يكون LWORK على الأقل N + 1.
!>          إذا كان JOBZ  = 'V' و N > 1، يجب أن يكون LWORK على الأقل 2*N + N**2.
!>
!>          إذا كان LWORK = -1، فسيتم افتراض استعلام عن مساحة العمل؛ يقوم الروتين
!>          فقط بحساب الأحجام المثلى لمصفوفات WORK و RWORK و
!>          IWORK، ويعيد هذه القيم كمدخلات أولى
!>          للمصفوفات WORK و RWORK و IWORK، ولا يتم إصدار أي رسالة خطأ
!>          متعلقة بـ LWORK أو LRWORK أو LIWORK بواسطة XERBLA.
!>

RWORK

!>          RWORK is DOUBLE PRECISION array, dimension (MAX(1,LRWORK))
!>          عند الخروج، إذا كان INFO = 0، فإن RWORK(1) يعيد LRWORK الأمثل.
!>

LRWORK

!>          LRWORK is INTEGER
!>          بعد المصفوفة RWORK.
!>          إذا كان N <= 1،                يجب أن يكون LRWORK على الأقل 1.
!>          إذا كان JOBZ  = 'N' و N > 1، يجب أن يكون LRWORK على الأقل N.
!>          إذا كان JOBZ  = 'V' و N > 1، يجب أن يكون LRWORK على الأقل
!>                         1 + 5*N + 2*N**2.
!>
!>          إذا كان LRWORK = -1، فسيتم افتراض استعلام عن مساحة العمل؛ يقوم الروتين
!>          فقط بحساب الأحجام المثلى لمصفوفات WORK و RWORK و
!>          IWORK، ويعيد هذه القيم كمدخلات أولى
!>          للمصفوفات WORK و RWORK و IWORK، ولا يتم إصدار أي رسالة خطأ
!>          متعلقة بـ LWORK أو LRWORK أو LIWORK بواسطة XERBLA.
!>

IWORK

!>          IWORK is INTEGER array, dimension (MAX(1,LIWORK))
!>          عند الخروج، إذا كان INFO = 0، فإن IWORK(1) يعيد LIWORK الأمثل.
!>

LIWORK

!>          LIWORK is INTEGER
!>          بعد المصفوفة IWORK.
!>          إذا كان N <= 1،                يجب أن يكون LIWORK على الأقل 1.
!>          إذا كان JOBZ  = 'N' و N > 1، يجب أن يكون LIWORK على الأقل 1.
!>          إذا كان JOBZ  = 'V' و N > 1، يجب أن يكون LIWORK على الأقل 3 + 5*N.
!>
!>          إذا كان LIWORK = -1، فسيتم افتراض استعلام عن مساحة العمل؛ يقوم الروتين
!>          فقط بحساب الأحجام المثلى لمصفوفات WORK و RWORK و
!>          IWORK، ويعيد هذه القيم كمدخلات أولى
!>          للمصفوفات WORK و RWORK و IWORK، ولا يتم إصدار أي رسالة خطأ
!>          متعلقة بـ LWORK أو LRWORK أو LIWORK بواسطة XERBLA.
!>

INFO

!>          INFO is INTEGER
!>          = 0:  خروج ناجح
!>          < 0:  إذا كان INFO = -i، فإن الوسيط i كان له قيمة غير قانونية
!>          > 0:  إذا كان INFO = i و JOBZ = 'N'، فإن الخوارزمية فشلت
!>                في التقارب؛ لم تتقارب i من العناصر القطرية غير القطرية لمصفوفة ثلاثية وسيطة
!>                إلى الصفر؛
!>                إذا كان INFO = i و JOBZ = 'V'، فإن الخوارزمية فشلت
!>                في حساب قيمة ذاتية أثناء العمل على المصفوفة الفرعية
!>                التي تقع في الصفوف والأعمدة INFO/(N+1) حتى
!>                mod(INFO,N+1).
!>
