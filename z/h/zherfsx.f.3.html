ZHERFSX

الغرض:

!>
!>    ZHERFSX يحسن الحل المحسوب لنظام المعادلات الخطية
!>    عندما تكون مصفوفة المعاملات هيرميتية غير محددة، و
!>    يوفر حدود الخطأ وتقديرات الخطأ الخلفي للحل. بالإضافة إلى حد الخطأ المعياري، الكود يوفر
!>    حدود خطأ مركبة قصوى إذا أمكن. انظر التعليقات لـ
!>    ERR_BNDS_NORM و ERR_BNDS_COMP لتفاصيل حدود الخطأ.
!>
!>    قد تم موازنة نظام المعادلات الخطية الأصلي
!>    قبل استدعاء هذا الروتين، كما هو موضح بواسطة الوسائط EQUED و S
!>    أدناه. في هذه الحالة، يكون الحل وحدود الخطأ المعادة
!>    للنظام الأصلي غير المتوازن.
!> 
!>     بعض المعلمات الاختيارية مجمعة في مصفوفة PARAMS. هذه
!>     الإعدادات تحدد كيفية إجراء التحسين، ولكن غالبًا ما تكون
!>     القيم الافتراضية مقبولة. إذا كانت القيم الافتراضية مقبولة، يمكن للمستخدمين
!>     تمرير NPARAMS = 0 مما يمنع الكود المصدري من الوصول
!>     إلى الوسيط PARAMS.
!>

المعلمات

UPLO 
!>          UPLO هو CHARACTER*1
!>       = 'U':  المثلث العلوي من A مخزن؛
!>       = 'L':  المثلث السفلي من A مخزن.
!>

EQUED

!>          EQUED هو CHARACTER*1
!>     يحدد شكل الموازنة الذي تم إجراؤه على A
!>     قبل استدعاء هذا الروتين. هذا مطلوب لحساب
!>     الحل وحدود الخطأ بشكل صحيح.
!>       = 'N':  لا توجد موازنة
!>       = 'Y':  موازنة الصفوف والأعمدة، أي تم استبدال A
!>               بـ diag(S) * A * diag(S).
!>               تم تغيير الجانب الأيمن B وفقًا لذلك.
!>

N

!>          N هو INTEGER
!>     ترتيب المصفوفة A. N >= 0.
!>

NRHS

!>          NRHS هو INTEGER
!>     عدد الجوانب اليمنى، أي عدد أعمدة
!>     المصفوفتين B و X. NRHS >= 0.
!>

A

!>          A هي مصفوفة COMPLEX*16، البعد (LDA,N)
!>     المصفوفة الهيرميتية A. إذا كان UPLO = 'U'، فإن الجزء العلوي
!>     الأمامي بحجم N-by-N من A يحتوي على الجزء العلوي
!>     المثلثي للمصفوفة A، ولا يتم الرجوع إلى الجزء المثلثي
!>     السفلي الصارم لـ A. إذا كان UPLO = 'L'، فإن الجزء السفلي
!>     الأمامي بحجم N-by-N من A يحتوي على الجزء السفلي
!>     المثلثي للمصفوفة A، ولا يتم الرجوع إلى الجزء المثلثي
!>     العلوي الصارم لـ A.
!>

LDA

!>          LDA هو INTEGER
!>     البعد الأمامي للمصفوفة A. LDA >= max(1,N).
!>

AF

!>          AF هي مصفوفة COMPLEX*16، البعد (LDAF,N)
!>     الشكل المفكك للمصفوفة A. AF يحتوي على المصفوفة القطرية
!>     الكتلية D والمعاملات المستخدمة للحصول على U أو L من
!>     التحليل A = U*D*U**H أو A = L*D*L**H كما تم حسابه بواسطة ZHETRF.
!>

LDAF

!>          LDAF هو INTEGER
!>     البعد الأمامي للمصفوفة AF. LDAF >= max(1,N).
!>

IPIV

!>          IPIV هي مصفوفة INTEGER، البعد (N)
!>     تفاصيل التبادلات والهيكل الكتلي لـ D
!>     كما تم تحديده بواسطة ZHETRF.
!>

S

!>          S هي مصفوفة DOUBLE PRECISION، البعد (N)
!>     عوامل القياس لـ A. إذا كان EQUED = 'Y'، يتم ضرب A
!>     من اليسار واليمين بواسطة diag(S). S هي وسيط إدخال إذا كان FACT =
!>     'F'؛ بخلاف ذلك، S هي وسيط إخراج. إذا كان FACT = 'F' و EQUED
!>     = 'Y'، يجب أن يكون كل عنصر من S موجبًا. إذا كان S هو إخراج، فإن كل
!>     عنصر من S هو قوة للجذر. إذا كان S هو إدخال، يجب أن يكون كل
!>     عنصر من S قوة للجذر لضمان حل موثوق به و
!>     تقديرات خطأ. القياس بقوى الجذر لا يسبب
!>     أخطاء تقريب إلا إذا كان الناتج يتجاوز الحدود الدنيا أو العليا.
!>     أخطاء التقريب أثناء القياس تؤدي إلى تحسين بمصفوفة ليست
!>     مكافئة للمصفوفة المدخلة، مما ينتج عنه تقديرات خطأ
!>     قد لا تكون موثوقة.
!>

B

!>          B هي مصفوفة COMPLEX*16، البعد (LDB,NRHS)
!>     مصفوفة الجانب الأيمن B.
!>

LDB

!>          LDB هو INTEGER
!>     البعد الأمامي للمصفوفة B. LDB >= max(1,N).
!>

X

!>          X هي مصفوفة COMPLEX*16، البعد (LDX,NRHS)
!>     عند الإدخال، مصفوفة الحل X، كما تم حسابها بواسطة ZHETRS.
!>     عند الخروج، مصفوفة الحل المحسنة X.
!>

LDX

!>          LDX هو INTEGER
!>     البعد الأمامي للمصفوفة X. LDX >= max(1,N).
!>

RCOND

!>          RCOND هو DOUBLE PRECISION
!>     مقلوب رقم التكييف المعياري. هذا تقدير لـ
!>     مقلوب رقم تكييف Skeel للمصفوفة A بعد
!>     الموازنة (إذا تم إجراؤها). إذا كان هذا أقل من دقة الجهاز
!>     (خاصة، إذا كان صفرًا)، فإن المصفوفة منفردة
!>     لأغراض العمل. لاحظ أن الخطأ قد لا يزال صغيرًا حتى
!>     لو كان هذا الرقم صغيرًا جدًا وكانت المصفوفة تبدو غير
!>     متكيفة.
!>

BERR

!>          BERR هي مصفوفة DOUBLE PRECISION، البعد (NRHS)
!>     الخطأ النسبي المركب الخلفي. هذا هو
!>     الخطأ النسبي المركب الخلفي لكل متجه حل X(j)
!>     (أي، أصغر تغيير نسبي في أي عنصر من A أو B يجعل X(j) حلاً دقيقًا).
!>

N_ERR_BNDS

!>          N_ERR_BNDS هو INTEGER
!>     عدد حدود الخطأ لإرجاعها لكل جانب أيمن
!>     ولكل نوع (معياري أو مركب). انظر ERR_BNDS_NORM و
!>     ERR_BNDS_COMP أدناه.
!>

ERR_BNDS_NORM

!>          ERR_BNDS_NORM هي مصفوفة DOUBLE PRECISION، البعد (NRHS, N_ERR_BNDS)
!>     لكل جانب أيمن، تحتوي هذه المصفوفة على معلومات حول
!>     حدود الخطأ وأرقام التكييف المختلفة المقابلة لـ
!>     الخطأ النسبي المعياري، والذي يتم تعريفه على النحو التالي:
!>
!>     الخطأ النسبي المعياري في متجه الحل i:
!>             max_j (abs(XTRUE(j,i) - X(j,i)))
!>            ------------------------------
!>                  max_j abs(X(j,i))
!>
!>     المصفوفة هي
