subroutine zhegvd (integer itype, character jobz, character uplo, integer n, complex*16, dimension( lda, * ) a, integer lda, complex*16, dimension( ldb, * ) b, integer ldb, double precision, dimension( * ) w, complex*16, dimension( * ) work, integer lwork, double precision, dimension( * ) rwork, integer lrwork, integer, dimension( * ) iwork, integer liwork, integer info)

ZHEGVD

الغرض:

!>
!> ZHEGVD تحسب جميع القيم الذاتية، واختيارياً، المتجهات الذاتية
!> لمشكلة القياس الهرميتية المعرفة المعقدة، من الشكل
!> A*x=(lambda)*B*x,  A*Bx=(lambda)*x,  أو B*A*x=(lambda)*x. هنا A و
!> B مفترضتان هرميتين و B موجبة قطعية أيضًا.
!> إذا تم طلب المتجهات الذاتية، فإنها تستخدم خوارزمية تقسيم الغزو.
!>
!>

المعلمات

ITYPE 
!>          ITYPE هو عدد صحيح (INTEGER)
!>          يحدد نوع المشكلة المراد حلها:
!>          = 1:  A*x = (lambda)*B*x
!>          = 2:  A*B*x = (lambda)*x
!>          = 3:  B*A*x = (lambda)*x
!>

JOBZ

!>          JOBZ هو حرف واحد (CHARACTER*1)
!>          = 'N':  حساب القيم الذاتية فقط؛
!>          = 'V':  حساب القيم الذاتية والمتجهات الذاتية.
!>

UPLO

!>          UPLO هو حرف واحد (CHARACTER*1)
!>          = 'U':  المثلثات العليا من A و B مخزنة؛
!>          = 'L':  المثلثات السفلى من A و B مخزنة.
!>

N

!>          N هو عدد صحيح (INTEGER)
!>          رتبة المصفوفتين A و B.  N >= 0.
!>

A

!>          A هي مصفوفة أعداد مركبة (COMPLEX*16)، حجمها (LDA, N)
!>          عند الإدخال، المصفوفة الهرميتية A.  إذا كان UPLO = 'U'، فإن
!>          الجزء العلوي المثلثي من A بحجم N-بـ N يحتوي على
!>          الجزء العلوي المثلثي للمصفوفة A.  إذا كان UPLO = 'L'، فإن
!>          الجزء السفلي المثلثي من A بحجم N-بـ N يحتوي على
!>          الجزء السفلي المثلثي للمصفوفة A.
!>
!>          عند الخروج، إذا كان JOBZ = 'V'، فعندما يكون INFO = 0، فإن A تحتوي على
!>          المصفوفة Z للمتجهات الذاتية.  المتجهات الذاتية مطبعة
!>          بالشكل التالي:
!>          إذا كان ITYPE = 1 أو 2، Z**H*B*Z = I؛
!>          إذا كان ITYPE = 3، Z**H*inv(B)*Z = I.
!>          إذا كان JOBZ = 'N'، فإن عند الخروج، الجزء العلوي (إذا كان UPLO='U')
!>          أو الجزء السفلي (إذا كان UPLO='L') من A، بما في ذلك
!>          القطر الرئيسي، مدمر.
!>

LDA

!>          LDA هو عدد صحيح (INTEGER)
!>          البعد الرئيسي للمصفوفة A.  LDA >= max(1,N).
!>

B

!>          B هي مصفوفة أعداد مركبة (COMPLEX*16)، حجمها (LDB, N)
!>          عند الإدخال، المصفوفة الهرميتية B.  إذا كان UPLO = 'U'، فإن
!>          الجزء العلوي المثلثي من B بحجم N-بـ N يحتوي على
!>          الجزء العلوي المثلثي للمصفوفة B.  إذا كان UPLO = 'L'، فإن
!>          الجزء السفلي المثلثي من B بحجم N-بـ N يحتوي على
!>          الجزء السفلي المثلثي للمصفوفة B.
!>
!>          عند الخروج، إذا كان INFO <= N، فإن الجزء من B الذي يحتوي على المصفوفة
!>          يتم استبداله بالعامل المثلثي U أو L من تحليل Cholesky
!>          B = U**H*U أو B = L*L**H.
!>

LDB

!>          LDB هو عدد صحيح (INTEGER)
!>          البعد الرئيسي للمصفوفة B.  LDB >= max(1,N).
!>

W

!>          W هي مصفوفة أعداد حقيقية مزدوجة الدقة (DOUBLE PRECISION)، حجمها (N)
!>          إذا كان INFO = 0، فإن القيم الذاتية مرتبة تصاعدياً.
!>

WORK

!>          WORK هي مصفوفة أعداد مركبة (COMPLEX*16)، حجمها (MAX(1,LWORK))
!>          عند الخروج، إذا كان INFO = 0، فإن WORK(1) تحتوي على LWORK الأمثل.
!>

LWORK

!>          LWORK هو عدد صحيح (INTEGER)
!>          طول المصفوفة WORK.
!>          إذا كان N <= 1،                LWORK >= 1.
!>          إذا كان JOBZ  = 'N' و N > 1، LWORK >= N + 1.
!>          إذا كان JOBZ  = 'V' و N > 1، LWORK >= 2*N + N**2.
!>
!>          إذا كان LWORK = -1، فعندئذ يُفترض استعلام مساحة العمل؛ تقوم الوحدة الفرعية
!>          فقط بحساب الأحجام المثلى للمصفوفات WORK و RWORK و
!>          IWORK، وتعيد هذه القيم كأول إدخالات في
!>          المصفوفات WORK و RWORK و IWORK، ولا يتم إصدار أي رسالة خطأ
!>          متعلقة بـ LWORK أو LRWORK أو LIWORK بواسطة XERBLA.
!>

RWORK

!>          RWORK هي مصفوفة أعداد حقيقية مزدوجة الدقة (DOUBLE PRECISION)، حجمها (MAX(1,LRWORK))
!>          عند الخروج، إذا كان INFO = 0، فإن RWORK(1) تحتوي على LRWORK الأمثل.
!>

LRWORK

!>          LRWORK هو عدد صحيح (INTEGER)
!>          بعد المصفوفة RWORK.
!>          إذا كان N <= 1،                LRWORK >= 1.
!>          إذا كان JOBZ  = 'N' و N > 1، LRWORK >= N.
!>          إذا كان JOBZ  = 'V' و N > 1، LRWORK >= 1 + 5*N + 2*N**2.
!>
!>          إذا كان LRWORK = -1، فعندئذ يُفترض استعلام مساحة العمل؛ تقوم الوحدة الفرعية
!>          فقط بحساب الأحجام المثلى للمصفوفات WORK و RWORK و
!>          IWORK، وتعيد هذه القيم كأول إدخالات في
!>          المصفوفات WORK و RWORK و IWORK، ولا يتم إصدار أي رسالة خطأ
!>          متعلقة بـ LWORK أو LRWORK أو LIWORK بواسطة XERBLA.
!>

IWORK

!>          IWORK هي مصفوفة أعداد صحيحة (INTEGER)، حجمها (MAX(1,LIWORK))
!>          عند الخروج، إذا كان INFO = 0، فإن IWORK(1) تحتوي على LIWORK الأمثل.
!>

LIWORK

!>          LIWORK هو عدد صحيح (INTEGER)
!>          بعد المصفوفة IWORK.
!>          إذا كان N <= 1،                LIWORK >= 1.
!>          إذا كان JOBZ  = 'N' و N > 1، LIWORK >= 1.
!>          إذا كان JOBZ  = 'V' و N > 1، LIWORK >= 3 + 5*N.
!>
!>          إذا كان LIWORK = -1، فعندئذ يُفترض استعلام مساحة العمل؛ تقوم الوحدة الفرعية
!>          فقط بحساب الأحجام المثلى للمصفوفات WORK و RWORK و
!>          IWORK، وتعيد هذه القيم كأول إدخالات في
!>          المصفوفات WORK و RWORK و IWORK، ولا يتم إصدار أي رسالة خطأ
!>          متعلقة بـ LWORK أو LRWORK أو LIWORK بواسطة XERBLA.
!>

INFO

!>          INFO هو عدد صحيح (INTEGER)
!>          = 0:  انتهاء ناجح
!>          < 0:  إذا كان INFO = -i، فإن المعلمة i كانت ذات قيمة غير قانونية
!>          > 0:  أعادت ZPOTRF أو ZHEEVD رمز خطأ:
!>             <= N:  إذا كان INFO = i و JOBZ = 'N'، فشلت الخوارزمية
!>                    في التقارب؛ لم تتقارب i عنصر قطري غير قطري في
!>                    شكل ثلاثي قطري وسيط إلى
!>                    صفر؛
!>                    إذا كان INFO = i و JOBZ = 'V'، فشلت الخوارزمية
!>                    في حساب قيمة ذاتية أثناء العمل على
!>                    المصفوفة الفرعية الموجودة في الصفوف والأعمدة INFO/(N+1)
!>                    حتى mod(INFO,N+1)؛
!>             > N:   إذا كان INFO = N + i، لـ 1 <= i <= N، فإن
!>                    المinal الرئيسي الرائد من الدرجة i لـ B ليس موجباً.
!>                    تعذر إكمال تحليل B ولم يتم حساب أي قيم ذاتية أو متجهات ذاتية.
!>
