subroutine zhetrd (character uplo, integer n, complex*16, dimension( lda, * ) a, integer lda, double precision, dimension( * ) d, double precision, dimension( * ) e, complex*16, dimension( * ) tau, complex*16, dimension( * ) work, integer lwork, integer info)

ZHETRD

الغرض:

!>
!> تقوم ZHETRD بتقليص مصفوفة هيرميتية معقدة A إلى شكل ثلاثي الأقطار متناظر حقيقي T عن طريق تحويل تشابه وحدوي:
!> Q**H * A * Q = T.
!>

المعلمات

UPLO 
!>          UPLO هو CHARACTER*1
!>          = 'U':  يتم تخزين المثلث العلوي من A؛
!>          = 'L':  يتم تخزين المثلث السفلي من A.
!>

N

!>          N هو INTEGER
!>          رتبة المصفوفة A.  N >= 0.
!>

A

!>          A هو مصفوفة COMPLEX*16، البعد (LDA,N)
!>          عند الإدخال، مصفوفة هيرميتية A.  إذا كان UPLO = 'U'، فإن الجزء العلوي المثلثي الذي أبعاده N × N من A يحتوي على الجزء العلوي المثلثي من المصفوفة A، والجزء السفلي المثلثي الصارم من A لا تتم الإشارة إليه.  إذا كان UPLO = 'L'، فإن الجزء السفلي المثلثي الذي أبعاده N × N من A يحتوي على الجزء السفلي المثلثي من المصفوفة A، والجزء العلوي المثلثي الصارم من A لا تتم الإشارة إليه.
!>          عند الخروج، إذا كان UPLO = 'U'، يتم تجاوز العناصر القطرية والعناصر فوق القطرية الأولى من A بالعناصر المقابلة للمصفوفة ثلاثية الأقطار T، والعناصر الموجودة فوق العناصر فوق القطرية الأولى، مع المصفوفة TAU، تمثل المصفوفة الوحدوية Q كحاصل ضرب الانعكاسات الأولية؛ إذا كان UPLO
!>          = 'L'، يتم تجاوز العناصر القطرية والعناصر تحت القطرية الأولى من A بالعناصر المقابلة للمصفوفة ثلاثية الأقطار T، والعناصر الموجودة تحت العناصر تحت القطرية الأولى، مع المصفوفة TAU، تمثل المصفوفة الوحدوية Q كحاصل ضرب الانعكاسات الأولية. انظر التفاصيل الإضافية.
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرائد للمصفوفة A.  LDA >= max(1,N).
!>

D

!>          D هي مصفوفة DOUBLE PRECISION، البعد (N)
!>          العناصر القطرية للمصفوفة ثلاثية الأقطار T:
!>          D(i) = A(i,i).
!>

E

!>          E هي مصفوفة DOUBLE PRECISION، البعد (N-1)
!>          العناصر خارج القطر للمصفوفة ثلاثية الأقطار T:
!>          E(i) = A(i,i+1) إذا كان UPLO = 'U'، E(i) = A(i+1,i) إذا كان UPLO = 'L'.
!>

TAU

!>          TAU هي مصفوفة COMPLEX*16، البعد (N-1)
!>          العوامل القياسية للانعكاسات الأولية (انظر التفاصيل الإضافية).
!>

WORK

!>          WORK هي مصفوفة COMPLEX*16، البعد (MAX(1,LWORK))
!>          عند الخروج، إذا كان INFO = 0، فإن WORK(1) يعيد LWORK الأمثل.
!>

LWORK

!>          LWORK هو INTEGER
!>          بعد المصفوفة WORK.  LWORK >= 1.
!>          للحصول على أفضل أداء LWORK >= N*NB، حيث NB هو
!>          حجم الكتلة الأمثل.
!>
!>          إذا كان LWORK = -1، فسيتم افتراض استعلام مساحة العمل؛ الروتين
!>          يحسب فقط الحجم الأمثل لمصفوفة WORK، ويعيد
!>          هذه القيمة كالإدخال الأول لمصفوفة WORK، ولا يتم إصدار أي رسالة خطأ
!>          تتعلق بـ LWORK بواسطة XERBLA.
!>

INFO

!>          INFO هو INTEGER
!>          = 0:  خروج ناجح
!>          < 0:  إذا كان INFO = -i، فإن المعامل i كان له قيمة غير قانونية
!>

تفاصيل إضافية:

!>
!>  إذا كان UPLO = 'U'، يتم تمثيل المصفوفة Q كحاصل ضرب الانعكاسات الأولية
!>
!>     Q = H(n-1) . . . H(2) H(1).
!>
!>  كل H(i) لها الشكل
!>
!>     H(i) = I - tau * v * v**H
!>
!>  حيث tau هو عامل قياسي معقد، و v هو متجه معقد مع
!>  v(i+1:n) = 0 و v(i) = 1؛ يتم تخزين v(1:i-1) عند الخروج في
!>  A(1:i-1,i+1)، و tau في TAU(i).
!>
!>  إذا كان UPLO = 'L'، يتم تمثيل المصفوفة Q كحاصل ضرب الانعكاسات الأولية
!>
!>     Q = H(1) H(2) . . . H(n-1).
!>
!>  كل H(i) لها الشكل
!>
!>     H(i) = I - tau * v * v**H
!>
!>  حيث tau هو عامل قياسي معقد، و v هو متجه معقد مع
!>  v(1:i) = 0 و v(i+1) = 1؛ يتم تخزين v(i+2:n) عند الخروج في A(i+2:n,i)،
!>  و tau في TAU(i).
!>
!>  محتويات A عند الخروج موضحة بالأمثلة التالية
!>  مع n = 5:
!>
!>  إذا كان UPLO = 'U':                       إذا كان UPLO = 'L':
!>
!>    (  d   e   v2  v3  v4 )              (  d                  )
!>    (      d   e   v3  v4 )              (  e   d              )
!>    (          d   e   v4 )              (  v1  e   d          )
!>    (              d   e  )              (  v1  v2  e   d      )
!>    (                  d  )              (  v1  v2  v3  e   d  )
!>
!>  حيث d و e يمثلان العناصر القطرية وخارج القطر لـ T، و vi
!>  يمثل عنصرًا من المتجه الذي يحدد H(i).
!>
