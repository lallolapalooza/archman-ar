subroutine zhetrf_aa (character uplo, integer n, complex*16, dimension( lda, * ) a, integer lda, integer, dimension( * ) ipiv, complex*16, dimension( * ) work, integer lwork, integer info)

ZHETRF_AA

الغرض:

!>
!> ZHETRF_AA يحسب تحليل مصفوفة هرميتية مركبة A
!> باستخدام خوارزمية Aasen. شكل التحليل هو
!>
!>    A = U**H*T*U  أو  A = L*T*L**H
!>
!> حيث U (أو L) هو ناتج ضرب مصفوفات التبديل والمصفوفات المثلثية العليا (السفلى) الوحدوية.
!> T هي مصفوفة هرميتية ثلاثية الأقطار.
!>
!> هذه هي النسخة المجمعة من الخوارزمية، تستدعي BLAS المستوى 3.
!>

المعلمات

UPLO 
!>          UPLO هي CHARACTER*1
!>          = 'U':  يتم تخزين المثلث العلوي من A؛
!>          = 'L':  يتم تخزين المثلث السفلي من A.
!>

N

!>          N هو INTEGER
!>          رتبة المصفوفة A.  N >= 0.
!>

A

!>          A هي مصفوفة COMPLEX*16، ذات بعد (LDA,N)
!>          عند الإدخال، المصفوفة الهرميتية A. إذا كان UPLO = 'U'، فإن الجزء
!>          المثلثي العلوي N*N الرائد من A يحتوي على الجزء المثلثي العلوي
!>          للمصفوفة A، ولا يتم الرجوع إلى الجزء المثلثي السفلي الصارم من A. إذا كان UPLO = 'L'، فإن الجزء
!>          المثلثي السفلي N*N الرائد من A يحتوي على الجزء المثلثي السفلي
!>          للمصفوفة A، ولا يتم الرجوع إلى الجزء المثلثي العلوي الصارم من A.
!>
!>          عند الخروج، يتم تخزين المصفوفة ثلاثية الأقطار في الأقطار
!>          وشبه الأقطار أسفل (أو أعلى) الأقطار،
!>          ويتم تخزين L أسفل (أو أعلى) شبه الأقطار، عندما يكون UPLO
!>          هو 'L' (أو 'U').
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرائد للمصفوفة A.  LDA >= max(1,N).
!>

IPIV

!>          IPIV هي مصفوفة INTEGER، ذات بعد (N)
!>          عند الخروج، تحتوي على تفاصيل التبادلات، أي
!>          تم تبديل الصف والعمود k من A مع
!>          الصف والعمود IPIV(k).
!>

WORK

!>          WORK هي مصفوفة COMPLEX*16، ذات بعد (MAX(1,LWORK))
!>          عند الخروج، إذا كان INFO = 0، فإن WORK(1) يعيد LWORK المثلى.
!>

LWORK

!>          LWORK هو INTEGER
!>          طول WORK.
!>          LWORK >= 1، إذا كان N >= 1، و LWORK >= 2*N، خلاف ذلك.
!>          لأداء مثالي LWORK >= N*(1+NB)، حيث NB
!>          هو حجم الكتلة الأمثل، الذي يتم إرجاعه بواسطة ILAENV.
!>
!>          إذا كان LWORK = -1، فمن المفترض استعلام مساحة العمل؛ الروتين
!>          يحسب فقط الحجم الأمثل لمصفوفة WORK، ويعيد
!>          هذه القيمة كأول مدخل لمصفوفة WORK، ولا يتم إصدار أي رسالة خطأ
!>          متعلقة بـ LWORK بواسطة XERBLA.
!>

INFO

!>          INFO هو INTEGER
!>          = 0:  انتهاء ناجح
!>          < 0:  إذا كان INFO = -i، فإن الوسيط i كان له قيمة غير قانونية.
!>
