subroutine zhesv_rk (character uplo, integer n, integer nrhs, complex*16, dimension( lda, * ) a, integer lda, complex*16, dimension( * ) e, integer, dimension( * ) ipiv, complex*16, dimension( ldb, * ) b, integer ldb, complex*16, dimension( * ) work, integer lwork, integer info)

ZHESV_RK يحسب حل نظام المعادلات الخطية A * X = B لمصفوفات SY

الغرض:

!> ZHESV_RK يحسب حل نظام المعادلات الخطية المعقدة A * X = B، حيث A مصفوفة هرميتية N × N
!> و X و B مصفوفتان N × NRHS.
!>
!> تُستخدم طريقة التنقل القطري غير المقيد BunChe-Kaufman (rook) لتحليل A إلى
!>    A = P*U*D*(U**H)*(P**T)، إذا UPLO = 'U'، أو
!>    A = P*L*D*(L**H)*(P**T)، إذا UPLO = 'L'،
!> حيث U (أو L) مصفوفة مثلثية علوية (أو سفلية) أحادية،
!> U**H (أو L**H) هي المرافق لـ U (أو L)، P مصفوفة تبديل،
!> P**T هي منقولة P، و D مصفوفة هرميتية وقطرية مقطعية مع كتل قطرية 1 × 1 و 2 × 2.
!>
!> يتم استدعاء ZHETRF_RK لحساب تحليل مصفوفة هرميتية معقدة.
!> ثم يتم استخدام الشكل المحلل لـ A لحل نظام المعادلات A * X = B عن طريق استدعاء روتين BLAS3 ZHETRS_3.
!>

المعلمات

UPLO 
!>          UPLO هو CHARACTER*1
!>          يحدد ما إذا كان الجزء المثلثي العلوي أو السفلي من المصفوفة الهرميتية A مخزنًا:
!>          = 'U':  يتم تخزين المثلث العلوي لـ A؛
!>          = 'L':  يتم تخزين المثلث السفلي لـ A.
!>

N

!>          N هو INTEGER
!>          عدد المعادلات الخطية، أي رتبة المصفوفة A. N >= 0.
!>

NRHS

!>          NRHS هو INTEGER
!>          عدد الأطراف اليمنى، أي عدد أعمدة المصفوفة B. NRHS >= 0.
!>

A

!>          A هي مصفوفة COMPLEX*16، أبعاد (LDA,N)
!>          عند الإدخال، المصفوفة الهرميتية A.
!>            إذا كان UPLO = 'U': الجزء المثلثي العلوي N × N الرائد
!>            من A يحتوي على الجزء المثلثي العلوي للمصفوفة A،
!>            والجزء السفلي الصارم من A غير مستخدم.
!>
!>            إذا كان UPLO = 'L': الجزء المثلثي السفلي N × N الرائد
!>            من A يحتوي على الجزء المثلثي السفلي للمصفوفة A،
!>            والجزء العلوي الصارم من A غير مستخدم.
!>
!>          عند الخروج، إذا كان INFO = 0، قطر المصفوفة القطرية المقطعية
!>          D والعوامل U أو L كما تم حسابها بواسطة ZHETRF_RK:
!>            أ) عناصر القطر فقط للمصفوفة الهرميتية القطرية المقطعية
!>               D على قطر A، أي D(k,k) = A(k,k)؛
!>               (عناصر ما فوق القطر (أو ما تحت القطر) لـ D
!>                مخزنة عند الخروج في المصفوفة E)، و
!>            ب) إذا كان UPLO = 'U': العامل U في الجزء فوق القطري لـ A.
!>               إذا كان UPLO = 'L': العامل L في الجزء تحت القطري لـ A.
!>
!>          لمزيد من المعلومات انظر وصف روتين ZHETRF_RK.
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرائد للمصفوفة A. LDA >= max(1,N).
!>

E

!>          E هي مصفوفة COMPLEX*16، أبعاد (N)
!>          عند الخروج، تحتوي على المخرجات التي تم حسابها بواسطة روتين التحليل
!>          ZHETRF_RK، أي عناصر ما فوق القطر (أو ما تحت القطر)
!>          للمصفوفة الهرميتية القطرية المقطعية D
!>          مع كتل قطرية 1 × 1 أو 2 × 2، حيث
!>          إذا كان UPLO = 'U': E(i) = D(i-1,i)، i=2:N، E(1) مضبوطة على 0؛
!>          إذا كان UPLO = 'L': E(i) = D(i+1,i)، i=1:N-1، E(N) مضبوطة على 0.
!>
!>          ملاحظة: بالنسبة لكتلة قطرية 1 × 1 D(k)، حيث
!>          1 <= k <= N، يتم ضبط العنصر E(k) على 0 في كل من
!>          حالات UPLO = 'U' أو UPLO = 'L'.
!>
!>          لمزيد من المعلومات انظر وصف روتين ZHETRF_RK.
!>

IPIV

!>          IPIV هي مصفوفة INTEGER، أبعاد (N)
!>          تفاصيل التبادلات وهيكل الكتلة لـ D،
!>          كما تم تحديده بواسطة ZHETRF_RK.
!>
!>          لمزيد من المعلومات انظر وصف روتين ZHETRF_RK.
!>

B

!>          B هي مصفوفة COMPLEX*16، أبعاد (LDB,NRHS)
!>          عند الإدخال، مصفوفة الطرف الأيمن B ذات الأبعاد N × NRHS.
!>          عند الخروج، إذا كان INFO = 0، مصفوفة الحل X ذات الأبعاد N × NRHS.
!>

LDB

!>          LDB هو INTEGER
!>          البعد الرائد للمصفوفة B. LDB >= max(1,N).
!>

WORK

!>          WORK هي مصفوفة COMPLEX*16، أبعاد ( MAX(1,LWORK) ).
!>          مصفوفة العمل المستخدمة في مرحلة التحليل.
!>          عند الخروج، إذا كان INFO = 0، تقوم WORK(1) بإرجاع LWORK الأمثل.
!>

LWORK

!>          LWORK هو INTEGER
!>          طول WORK. LWORK >= 1. للحصول على أفضل أداء
!>          لمرحلة التحليل LWORK >= max(1,N*NB)، حيث NB هو
!>          حجم الكتلة الأمثل لـ ZHETRF_RK.
!>
!>          إذا كان LWORK = -1، فسيتم افتراض استعلام عن مساحة العمل؛
!>          تقوم الروتين فقط بحساب الحجم الأمثل لمصفوفة WORK
!>          لمرحلة التحليل، وتعيد هذا الحجم كـ
!>          الإدخال الأول لمصفوفة WORK، ولا يتم إصدار أي رسالة خطأ
!>          متعلقة بـ LWORK بواسطة XERBLA.
!>

INFO

!>          INFO هو INTEGER
!>          = 0: خروج ناجح
!>
!>          < 0: إذا كان INFO = -k، فإن الوسيط k كان له قيمة غير قانونية
!>
!>          > 0: إذا كان INFO = k، فإن المصفوفة A منفردة، وذلك بسبب:
!>                 إذا كان UPLO = 'U': العمود k في
!>                 الجزء المثلثي العلوي لـ A يحتوي على جميع الأصفار.
!>                 إذا كان UPLO = 'L': العمود k في
!>                 الجزء المثلثي السفلي لـ A يحتوي على جميع الأصفار.
!>
!>               لذلك D(k,k) يساوي صفر تمامًا، وعناصر ما فوق القطر
!>               للعمود k من U (أو عناصر ما تحت القطر لـ
!>               العمود k من L) كلها أصفار. تم إكمال التحليل،
!>               ولكن المصفوفة القطرية المقطعية D
!>               منفردة تمامًا، وسيحدث قسمة على صفر إذا
!>               تم استخدامها لحل نظام معادلات.
!>
!>               ملاحظة: INFO يخزن فقط أول حدوث
!>               للانفراد، أي حدوث انفراد لاحق
!>               لا يتم تخزينه في INFO حتى لو كان التحليل
!>               يكتمل دائمًا.
!>
