ZHEEVX تحسب القيم الذاتية، اختيارياً، المتجهات الذاتية اليسرى و/أو اليمنى لمصفوفات HE

توثيق الوظيفة/الروتين الفرعي

subroutine zheevx (character jobz, character range, character uplo, integer n, complex*16, dimension( lda, * ) a, integer lda, double precision vl, double precision vu, integer il, integer iu, double precision abstol, integer m, double precision, dimension( * ) w, complex*16, dimension( ldz, * ) z, integer ldz, complex*16, dimension( * ) work, integer lwork, double precision, dimension( * ) rwork, integer, dimension( * ) iwork, integer, dimension( * ) ifail, integer info)

ZHEEVX تحسب القيم الذاتية، اختيارياً، المتجهات الذاتية اليسرى و/أو اليمنى لمصفوفات HE

الغرض:

!>
!> ZHEEVX تحسب قيماً ذاتية محددة، واختيارياً، متجهات ذاتية
!> لمصفوفة هرميتية معقدة A. يمكن تحديد القيم الذاتية والمتجهات الذاتية
!> عن طريق تحديد نطاق من القيم أو نطاق من
!> المؤشرات للقيم الذاتية المطلوبة.
!>

المعلمات

JOBZ 
!>          JOBZ is CHARACTER*1
!>          = 'N':  حساب القيم الذاتية فقط؛
!>          = 'V':  حساب القيم الذاتية والمتجهات الذاتية.
!>

RANGE

!>          RANGE is CHARACTER*1
!>          = 'A': سيتم العثور على جميع القيم الذاتية.
!>          = 'V': سيتم العثور على جميع القيم الذاتية في النطاق شبه المفتوح (VL,VU].
!>          = 'I': سيتم العثور على القيم الذاتية من IL-th إلى IU-th.
!>

UPLO

!>          UPLO is CHARACTER*1
!>          = 'U':  يتم تخزين المثلث العلوي من A؛
!>          = 'L':  يتم تخزين المثلث السفلي من A.
!>

N

!>          N is INTEGER
!>          رتبة المصفوفة A.  N >= 0.
!>

A

!>          A is COMPLEX*16 array, dimension (LDA, N)
!>          عند الإدخال، المصفوفة الهرميتية A.  إذا كان UPLO = 'U'، فإن
!>          الجزء المثلثي العلوي N-by-N من A يحتوي على
!>          الجزء المثلثي العلوي للمصفوفة A.  إذا كان UPLO = 'L'، فإن
!>          الجزء المثلثي السفلي N-by-N من A يحتوي على
!>          الجزء المثلثي السفلي للمصفوفة A.
!>          عند الخروج، المثلث السفلي (إذا كان UPLO='L') أو المثلث
!>          العلوي (إذا كان UPLO='U') من A، بما في ذلك القطر، يتم تدميره.
!>

LDA

!>          LDA is INTEGER
!>          البعد الرائد للمصفوفة A.  LDA >= max(1,N).
!>

VL

!>          VL is DOUBLE PRECISION
!>          إذا كان RANGE='V'، فإن الحد الأدنى للنطاق للبحث عن القيم الذاتية. VL < VU.
!>          لا يتم الرجوع إليه إذا كان RANGE = 'A' أو 'I'.
!>

VU

!>          VU is DOUBLE PRECISION
!>          إذا كان RANGE='V'، فإن الحد الأعلى للنطاق للبحث عن القيم الذاتية. VL < VU.
!>          لا يتم الرجوع إليه إذا كان RANGE = 'A' أو 'I'.
!>

IL

!>          IL is INTEGER
!>          إذا كان RANGE='I'، فإن مؤشر أصغر قيمة ذاتية سيتم إرجاعها.
!>          1 <= IL <= IU <= N، إذا كان N > 0؛ IL = 1 و IU = 0 إذا كان N = 0.
!>          لا يتم الرجوع إليه إذا كان RANGE = 'A' أو 'V'.
!>

IU

!>          IU is INTEGER
!>          إذا كان RANGE='I'، فإن مؤشر أكبر قيمة ذاتية سيتم إرجاعها.
!>          1 <= IL <= IU <= N، إذا كان N > 0؛ IL = 1 و IU = 0 إذا كان N = 0.
!>          لا يتم الرجوع إليه إذا كان RANGE = 'A' أو 'V'.
!>

ABSTOL

!>          ABSTOL is DOUBLE PRECISION
!>          التسامح المطلق للخطأ في القيم الذاتية.
!>          يتم قبول قيمة ذاتية تقريبية على أنها متقاربة
!>          عندما يتم تحديد أنها تقع في نطاق [a,b]
!>          بعرض أقل من أو يساوي
!>
!>          ABSTOL + EPS *   max( |a|,|b| ) ,
!>
!>          حيث EPS هي الدقة الآلية.  إذا كانت ABSTOL أقل من
!>          أو تساوي الصفر، فسيتم استخدام EPS*|T| بدلاً منها،
!>          حيث |T| هو المعيار 1 للمصفوفة الثلاثية القطرية التي تم الحصول عليها
!>          عن طريق تقليل A إلى شكل ثلاثي القطر.
!>
!>          سيتم حساب القيم الذاتية بأكبر دقة عندما يتم تعيين ABSTOL إلى
!>          ضعف عتبة التدفق الصفري 2*DLAMCH('S')، وليس الصفر.
!>          إذا عادت هذه الروتين الفرعي مع INFO>0، مما يشير إلى أن بعض
!>          المتجهات الذاتية لم تتقارب، حاول تعيين ABSTOL إلى
!>          2*DLAMCH('S').
!>
!>          انظر  بواسطة Demmel و
!>          Kahan، مذكرة عمل LAPACK #3.
!>

M

!>          M is INTEGER
!>          إجمالي عدد القيم الذاتية التي تم العثور عليها.  0 <= M <= N.
!>          إذا كان RANGE = 'A'، فإن M = N، وإذا كان RANGE = 'I'، فإن M = IU-IL+1.
!>

W

!>          W is DOUBLE PRECISION array, dimension (N)
!>          عند الخروج الطبيعي، تحتوي العناصر M الأولى على القيم الذاتية
!>          المحددة بترتيب تصاعدي.
!>

Z

!>          Z is COMPLEX*16 array, dimension (LDZ, max(1,M))
!>          إذا كان JOBZ = 'V'، فعندما يكون INFO = 0، تحتوي الأعمدة M الأولى من Z
!>          على المتجهات الذاتية المتعامدة للمصفوفة A
!>          المقابلة للقيم الذاتية المحددة، حيث يحمل العمود i-th من Z
!>          المتجه الذاتي المرتبط بـ W(i).
!>          إذا فشل متجه ذاتي في التقارب، فإن هذا العمود من Z
!>          يحتوي على أحدث تقريب للمتجه الذاتي، ويتم إرجاع مؤشر
!>          المتجه الذاتي في IFAIL.
!>          إذا كان JOBZ = 'N'، فلا يتم الرجوع إلى Z.
!>          ملاحظة: يجب على المستخدم التأكد من توفير عمود واحد على الأقل كحد أقصى (M,1)
!>          في المصفوفة Z؛ إذا كان RANGE = 'V'، فإن القيمة الدقيقة لـ M
!>          غير معروفة مسبقًا ويجب استخدام حد أعلى.
!>

LDZ

!>          LDZ is INTEGER
!>          البعد الرائد للمصفوفة Z.  LDZ >= 1، وإذا
!>          كان JOBZ = 'V'، فإن LDZ >= max(1,N).
!>

WORK

!>          WORK is COMPLEX*16 array, dimension (MAX(1,LWORK))
!>          عند الخروج، إذا كان INFO = 0، فإن WORK(1) يعيد LWORK الأمثل.
!>

LWORK

!>          LWORK is INTEGER
!>          طول المصفوفة WORK.  LWORK >= 1، عندما يكون N <= 1؛
!>          بخلاف ذلك 2*N.
!>          للكفاءة المثلى، LWORK >= (NB+1)*N،
!>          حيث NB هو الحد الأقصى لحجم الكتلة لـ ZHETRD و
!>          ZUNMTR كما تم إرجاعه بواسطة ILAENV.
!>
!>          إذا كان LWORK = -1، فسيتم افتراض استعلام مساحة العمل؛ الروتين
!>          يحسب فقط الحجم الأمثل للمصفوفة WORK، ويعيد
!>          هذه القيمة كالإدخال الأول للمصفوفة WORK، ولا يتم إصدار رسالة خطأ
!>          متعلقة بـ LWORK بواسطة XERBLA.
!>

RWORK

!>          RWORK is DOUBLE PRECISION array, dimension (7*N)
!>

IWORK

!>          IWORK is INTEGER array, dimension (5*N)
!>

IFAIL

!>          IFAIL is INTEGER array, dimension (N)
!>          إذا كان JOBZ = 'V'، فعندما يكون INFO = 0، فإن العناصر M الأولى من
!>          IFAIL تكون صفرًا.  إذا كان INFO > 0، فإن IFAIL يحتوي على
!>          مؤشرات المتجهات الذاتية التي فشلت في التقارب.
!>          إذا كان JOBZ = 'N'، فلا يتم الرجوع إلى IFAIL.
!>

INFO

!>          INFO is INTEGER
!>          = 0:  خروج ناجح
!>          < 0:  إذا كان INFO = -i، فإن المعامل i-th كان لديه
