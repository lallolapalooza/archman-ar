subroutine zhst01 (integer n, integer ilo, integer ihi, complex*16, dimension( lda, * ) a, integer lda, complex*16, dimension( ldh, * ) h, integer ldh, complex*16, dimension( ldq, * ) q, integer ldq, complex*16, dimension( lwork ) work, integer lwork, double precision, dimension( * ) rwork, double precision, dimension( 2 ) result)

ZHST01

الغرض:

!>
!> ZHST01 يختبر اختزال مصفوفة عامة A إلى شكل هسنبرج العلوي: A = Q*H*Q'. يتم حساب نسبتي اختبار؛
!>
!> RESULT(1) = norm( A - Q*H*Q' ) / ( norm(A) * N * EPS )
!> RESULT(2) = norm( I - Q'*Q ) / ( N * EPS )
!>
!> يُفترض أن تكون المصفوفة Q معطاة بشكل صريح كما ستكون بعد ZGEHRD + ZUNGHR.
!>
!> في هذا الإصدار، لا يتم استخدام ILO و IHI، ولكن يمكن استخدامهما لتوفير بعض العمل إذا كان هذا مرغوبًا.
!>

المعلمات

N 
!>          N هو عدد صحيح
!>          رتبة المصفوفة A. N >= 0.
!>

ILO

!>          ILO هو عدد صحيح
!>

IHI

!>          IHI هو عدد صحيح
!>
!>          يُفترض أن تكون A مثلثية علوية في الصفوف والأعمدة
!>          1:ILO-1 و IHI+1:N، لذا تختلف Q عن الهوية فقط في
!>          الصفوف والأعمدة ILO+1:IHI.
!>

A

!>          A هي مصفوفة COMPLEX*16، البعد (LDA,N)
!>          المصفوفة الأصلية A بحجم N × N.
!>

LDA

!>          LDA هو عدد صحيح
!>          البعد الرئيسي للمصفوفة A. LDA >= max(1,N).
!>

H

!>          H هي مصفوفة COMPLEX*16، البعد (LDH,N)
!>          مصفوفة هسنبرج العلوية H من اختزال A = Q*H*Q'
!>          كما تم حسابها بواسطة ZGEHRD. يُفترض أن تكون H صفرية أسفل
!>          القطر الفرعي الأول.
!>

LDH

!>          LDH هو عدد صحيح
!>          البعد الرئيسي للمصفوفة H. LDH >= max(1,N).
!>

Q

!>          Q هي مصفوفة COMPLEX*16، البعد (LDQ,N)
!>          المصفوفة المتعامدة Q من اختزال A = Q*H*Q' كما
!>          تم حسابها بواسطة ZGEHRD + ZUNGHR.
!>

LDQ

!>          LDQ هو عدد صحيح
!>          البعد الرئيسي للمصفوفة Q. LDQ >= max(1,N).
!>

WORK

!>          WORK هي مصفوفة COMPLEX*16، البعد (LWORK)
!>

LWORK

!>          LWORK هو عدد صحيح
!>          طول المصفوفة WORK. LWORK >= 2*N*N.
!>

RWORK

!>          RWORK هي مصفوفة DOUBLE PRECISION، البعد (N)
!>

RESULT

!>          RESULT هي مصفوفة DOUBLE PRECISION، البعد (2)
!>          RESULT(1) = norm( A - Q*H*Q' ) / ( norm(A) * N * EPS )
!>          RESULT(2) = norm( I - Q'*Q ) / ( N * EPS )
!>
