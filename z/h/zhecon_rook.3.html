subroutine zhecon_rook (uplo, n, a, lda, ipiv, anorm, rcond, work, info)


ZHECON_ROOK تقدر مقلوب رقم الشرط لمصفوفات HE باستخدام التحلل الناتج عن إحدى طرق التنقيط المحدودة (بحد أقصى 2 تبديل)

توثيق الدالة/الروتين الفرعي

subroutine zhecon_rook (character uplo, integer n, complex*16, dimension( lda, * ) a, integer lda, integer, dimension( * ) ipiv, double precision anorm, double precision rcond, complex*16, dimension( * ) work, integer info)

ZHECON_ROOK تقدر مقلوب رقم الشرط لمصفوفة هيرميتية معقدة A باستخدام التحلل A = U*D*U**H أو A = L*D*L**H المحسوب بواسطة CHETRF_ROOK.

الغرض: 

!>
!> ZHECON_ROOK تقدر مقلوب رقم الشرط لمصفوفة هيرميتية معقدة A باستخدام التحلل A = U*D*U**H أو
!> A = L*D*L**H المحسوب بواسطة CHETRF_ROOK.
!>
!> يتم الحصول على تقدير لـ norm(inv(A))، ويتم حساب مقلوب رقم الشرط كـ RCOND = 1 / (ANORM * norm(inv(A))).
!>

المعلمات

UPLO 
!>          UPLO هو CHARACTER*1
!>          يحدد ما إذا كانت تفاصيل التحلل مخزنة
!>          كمصفوفة مثلثية علوية أو سفلية.
!>          = 'U':  مثلثية علوية، الشكل هو A = U*D*U**H؛
!>          = 'L':  مثلثية سفلية، الشكل هو A = L*D*L**H.
!>

N

!>          N هو INTEGER
!>          ترتيب المصفوفة A. N >= 0.
!>

A

!>          A هو مصفوفة COMPLEX*16، البعد (LDA,N)
!>          المصفوفة القطرية الجزئية D والمضاعفات المستخدمة
!>          للحصول على العامل U أو L كما تم حسابه بواسطة CHETRF_ROOK.
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرائد للمصفوفة A. LDA >= max(1,N).
!>

IPIV

!>          IPIV هو مصفوفة INTEGER، البعد (N)
!>          تفاصيل التبادلات وهيكل الكتلة لـ D
!>          كما تم تحديده بواسطة CHETRF_ROOK.
!>

ANORM

!>          ANORM هو DOUBLE PRECISION
!>          المعيار 1 للمصفوفة الأصلية A.
!>

RCOND

!>          RCOND هو DOUBLE PRECISION
!>          مقلوب رقم شرط المصفوفة A،
!>          محسوب كـ RCOND = 1/(ANORM * AINVNM)، حيث AINVNM هو
!>          تقدير للمعيار 1 لـ inv(A) المحسوب في هذا الروتين.
!>

WORK

!>          WORK هو مصفوفة COMPLEX*16، البعد (2*N)
!>

INFO

!>          INFO هو INTEGER
!>          = 0:  انتهاء ناجح
!>          < 0:  إذا كان INFO = -i، كانت المعلمة i قيمة غير قانونية
!>
