توثيق الوظيفة/الإجراء الفرعي

subroutine zhbtrd (character vect, character uplo, integer n, integer kd, complex*16, dimension( ldab, * ) ab, integer ldab, double precision, dimension( * ) d, double precision, dimension( * ) e, complex*16, dimension( ldq, * ) q, integer ldq, complex*16, dimension( * ) work, integer info)

ZHBTRD

الغرض:

!>
!> ZHBTRD تقلل مصفوفة هرمية نطاقية A إلى شكل ثلاثي قطري متناظر حقيقي T
!> بواسطة تحويل تشابه وحدوي:
!> Q**H * A * Q = T.
!>

المعلمات

VECT 
!>          VECT هو CHARACTER*1
!>          = 'N':  لا تشكل Q؛
!>          = 'V':  شكل Q؛
!>          = 'U':  حدث مصفوفة X، بتشكيل X*Q.
!>

UPLO

!>          UPLO هو CHARACTER*1
!>          = 'U':  المثلث العلوي من A مخزن؛
!>          = 'L':  المثلث السفلي من A مخزن.
!>

N

!>          N هو INTEGER
!>          رتبة المصفوفة A. N >= 0.
!>

KD

!>          KD هو INTEGER
!>          عدد الأقطار الفائقة للمصفوفة A إذا كان UPLO = 'U'،
!>          أو عدد الأقطار الفرعية إذا كان UPLO = 'L'. KD >= 0.
!>

AB

!>          AB هو مصفوفة COMPLEX*16، البعد (LDAB,N)
!>          عند الإدخال، المثلث العلوي أو السفلي للمصفوفة الهرمية النطاقية
!>          A، مخزنة في أول KD+1 صفوف من المصفوفة. ال
!>          العمود j من A مخزن في العمود j من المصفوفة AB
!>          كما يلي:
!>          إذا كان UPLO = 'U'، AB(kd+1+i-j,j) = A(i,j) لـ max(1,j-kd)<=i<=j؛
!>          إذا كان UPLO = 'L'، AB(1+i-j,j) = A(i,j) لـ j<=i<=min(n,j+kd).
!>          عند الخروج، عناصر القطر للمصفوفة AB يتم الكتابة فوقها بعناصر
!>          القطر للمصفوفة ثلاثية الأقطار T؛ إذا كان KD > 0، ال
!>          عناصر على القطر الفائق الأول (إذا كان UPLO = 'U') أو ال
!>          القطر الفرعي الأول (إذا كان UPLO = 'L') يتم الكتابة فوقها بـ
!>          العناصر خارج القطر لـ T؛ الباقي من AB يتم الكتابة فوقه بـ
!>          القيم المولدة أثناء الاختزال.
!>

LDAB

!>          LDAB هو INTEGER
!>          البعد الرائد للمصفوفة AB. LDAB >= KD+1.
!>

D

!>          D هي مصفوفة DOUBLE PRECISION، البعد (N)
!>          عناصر القطر للمصفوفة ثلاثية الأقطار T.
!>

E

!>          E هي مصفوفة DOUBLE PRECISION، البعد (N-1)
!>          عناصر خارج القطر للمصفوفة ثلاثية الأقطار T:
!>          E(i) = T(i,i+1) إذا كان UPLO = 'U'؛ E(i) = T(i+1,i) إذا كان UPLO = 'L'.
!>

Q

!>          Q هي مصفوفة COMPLEX*16، البعد (LDQ,N)
!>          عند الإدخال، إذا كان VECT = 'U'، فإن Q يجب أن تحتوي على مصفوفة
!>          N-by-N X؛ إذا كان VECT = 'N' أو 'V'، فلا يلزم تعيين Q.
!>
!>          عند الخروج:
!>          إذا كان VECT = 'V'، فإن Q تحتوي على المصفوفة الوحدوية N-by-N Q؛
!>          إذا كان VECT = 'U'، فإن Q تحتوي على حاصل الضرب X*Q؛
!>          إذا كان VECT = 'N'، فإن المصفوفة Q غير مستخدمة.
!>

LDQ

!>          LDQ هو INTEGER
!>          البعد الرائد للمصفوفة Q.
!>          LDQ >= 1، و LDQ >= N إذا كان VECT = 'V' أو 'U'.
!>

WORK

!>          WORK هي مصفوفة COMPLEX*16، البعد (N)
!>

INFO

!>          INFO هو INTEGER
!>          = 0:  خروج ناجح
!>          < 0:  إذا كان INFO = -i، فإن الوسيط i كان له قيمة غير قانونية
!>
