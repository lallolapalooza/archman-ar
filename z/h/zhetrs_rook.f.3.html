subroutine zhetrs_rook (uplo, n, nrhs, a, lda, ipiv, b, ldb, info)
ZHETRS_ROOK يحسب الحل لنظام المعادلات الخطية A * X = B لمصفوفات HE باستخدام تحليل تم الحصول عليه بإحدى طرق التقليب القطري المحدود (بحد أقصى 2 تبديل)

توثيق الوظيفة/الروتين الفرعي

subroutine zhetrs_rook (uplo حرفي، n صحيح، nrhs صحيح، complex*16، dimension( lda, * ) a، lda صحيح، ipiv صحيح، dimension( * )، complex*16، dimension( ldb, * ) b، ldb صحيح، info صحيح)

ZHETRS_ROOK يحسب الحل لنظام المعادلات الخطية A * X = B لمصفوفات HE باستخدام تحليل تم الحصول عليه بإحدى طرق التقليب القطري المحدود (بحد أقصى 2 تبديل)

الغرض:

!>
!> ZHETRS_ROOK يحل نظام المعادلات الخطية A*X = B بمصفوفة A هرميتية معقدة
!> باستخدام التحليل A = U*D*U**H أو A = L*D*L**H الذي تم حسابه بواسطة ZHETRF_ROOK.
!>

المعاملات

UPLO 
!>          UPLO هو CHARACTER*1
!>          يحدد ما إذا كانت تفاصيل التحليل مخزنة
!>          كمصفوفة مثلثية عليا أو سفلى.
!>          = 'U':  مثلثية عليا، الشكل هو A = U*D*U**H؛
!>          = 'L':  مثلثية سفلى، الشكل هو A = L*D*L**H.
!>

N

!>          N هو INTEGER
!>          رتبة المصفوفة A.  N >= 0.
!>

NRHS

!>          NRHS هو INTEGER
!>          عدد الأطراف اليمنى، أي عدد أعمدة
!>          المصفوفة B.  NRHS >= 0.
!>

A

!>          A هي مصفوفة COMPLEX*16، ذات بعد (LDA,N)
!>          المصفوفة القطرية الكتلية D والمضاعفات المستخدمة
!>          للحصول على العامل U أو L كما تم حسابه بواسطة ZHETRF_ROOK.
!>

LDA

!>          LDA هو INTEGER
!>          البعد الأمامي للمصفوفة A.  LDA >= max(1,N).
!>

IPIV

!>          IPIV هي مصفوفة INTEGER، ذات بعد (N)
!>          تفاصيل التبادلات والبنية الكتلية لـ D
!>          كما تم تحديده بواسطة ZHETRF_ROOK.
!>

B

!>          B هي مصفوفة COMPLEX*16، ذات بعد (LDB,NRHS)
!>          عند الإدخال، مصفوفة الطرف الأيمن B.
!>          عند الخروج، مصفوفة الحل X.
!>

LDB

!>          LDB هو INTEGER
!>          البعد الأمامي للمصفوفة B.  LDB >= max(1,N).
!>

INFO

!>          INFO هو INTEGER
!>          = 0:  انتهاء ناجح
!>          < 0:  إذا كان INFO = -i، فإن المعامل i كان له قيمة غير قانونية
!>
