subroutine zhpgv (integer itype, character jobz, character uplo, integer n, complex*16, dimension( * ) ap, complex*16, dimension( * ) bp, double precision, dimension( * ) w, complex*16, dimension( ldz, * ) z, integer ldz, complex*16, dimension( * ) work, double precision, dimension( * ) rwork, integer info)

ZHPGV

الغرض:

!>
!> ZHPGV تحسب جميع القيم الذاتية، واختيارياً، المتجهات الذاتية
!> لمشكلة هيرميتية محددة عامة معقدة، من الشكل
!> A*x=(lambda)*B*x,  A*Bx=(lambda)*x,  أو B*A*x=(lambda)*x.
!> هنا يُفترض أن A و B هيرميتيتان، مخزنتان بتنسيق مضغوط،
!> وأن B موجب محددة أيضًا.
!>

المعاملات

ITYPE 
!>          ITYPE هو عدد صحيح (INTEGER)
!>          يحدد نوع المشكلة المراد حلها:
!>          = 1:  A*x = (lambda)*B*x
!>          = 2:  A*B*x = (lambda)*x
!>          = 3:  B*A*x = (lambda)*x
!>

JOBZ

!>          JOBZ هو حرف (CHARACTER*1)
!>          = 'N':  حساب القيم الذاتية فقط؛
!>          = 'V':  حساب القيم الذاتية والمتجهات الذاتية.
!>

UPLO

!>          UPLO هو حرف (CHARACTER*1)
!>          = 'U':  تُخزن المثلثات العليا لـ A و B؛
!>          = 'L':  تُخزن المثلثات السفلى لـ A و B.
!>

N

!>          N هو عدد صحيح (INTEGER)
!>          رتبة المصفوفتين A و B.  N >= 0.
!>

AP

!>          AP هي مصفوفة أعداد مركبة (COMPLEX*16) ذات بعد (N*(N+1)/2)
!>          عند الإدخال، المثلث العلوي أو السفلي للمصفوفة الهيرميتية
!>          A، مضغوطة عمودياً في مصفوفة خطية.  يتم تخزين العمود j من A
!>          في المصفوفة AP كما يلي:
!>          إذا كان UPLO = 'U'، فإن AP(i + (j-1)*j/2) = A(i,j) لـ 1<=i<=j؛
!>          إذا كان UPLO = 'L'، فإن AP(i + (j-1)*(2*n-j)/2) = A(i,j) لـ j<=i<=n.
!>
!>          عند الخروج، يتم تدمير محتويات AP.
!>

BP

!>          BP هي مصفوفة أعداد مركبة (COMPLEX*16) ذات بعد (N*(N+1)/2)
!>          عند الإدخال، المثلث العلوي أو السفلي للمصفوفة الهيرميتية
!>          B، مضغوطة عمودياً في مصفوفة خطية.  يتم تخزين العمود j من B
!>          في المصفوفة BP كما يلي:
!>          إذا كان UPLO = 'U'، فإن BP(i + (j-1)*j/2) = B(i,j) لـ 1<=i<=j؛
!>          إذا كان UPLO = 'L'، فإن BP(i + (j-1)*(2*n-j)/2) = B(i,j) لـ j<=i<=n.
!>
!>          عند الخروج، يبقى العامل المثلث U أو L من التحلل Cholesky
!>          B = U**H*U أو B = L*L**H، بنفس تنسيق التخزين مثل B.
!>

W

!>          W هي مصفوفة ذات دقة مزدوجة (DOUBLE PRECISION) ذات بعد (N)
!>          إذا كان INFO = 0، فإن القيم الذاتية مرتبة تصاعدياً.
!>

Z

!>          Z هي مصفوفة أعداد مركبة (COMPLEX*16) ذات بعد (LDZ, N)
!>          إذا كان JOBZ = 'V'، فعندما يكون INFO = 0، تحتوي Z على المصفوفة Z للمتجهات الذاتية.
!>          يتم تطبيع المتجهات الذاتية كالتالي:
!>          إذا كان ITYPE = 1 أو 2، فإن Z**H*B*Z = I؛
!>          إذا كان ITYPE = 3، فإن Z**H*inv(B)*Z = I.
!>          إذا كان JOBZ = 'N'، فلا يتم استخدام Z.
!>

LDZ

!>          LDZ هو عدد صحيح (INTEGER)
!>          البعد الرائد للمصفوفة Z.  LDZ >= 1، وإذا كان
!>          JOBZ = 'V'، فإن LDZ >= max(1,N).
!>

WORK

!>          WORK هي مصفوفة أعداد مركبة (COMPLEX*16) ذات بعد (max(1, 2*N-1))
!>

RWORK

!>          RWORK هي مصفوفة ذات دقة مزدوجة (DOUBLE PRECISION) ذات بعد (max(1, 3*N-2))
!>

INFO

!>          INFO هو عدد صحيح (INTEGER)
!>          = 0:  اكتمال ناجح
!>          < 0:  إذا كان INFO = -i، فإن المعامل i كان له قيمة غير قانونية
!>          > 0:  فشل ZPPTRF أو ZHPEV في التقارب؛
!>             <= N:  إذا كان INFO = i، فقد فشل ZHPEV في التقارب؛
!>                    i عنصر خارج القطر لمصفوفة ثلاثية وسيطة لم يتقارب إلى الصفر؛
!>             > N:   إذا كان INFO = N + i، لـ 1 <= i <= n، فإن
!>                    المُقسّم الرئيسي الأيسر من الرتبة i للمصفوفة B ليس موجبًا.
!>                    لم يكن من الممكن إكمال تحلل B ولم يتم حساب أي قيم ذاتية أو متجهات ذاتية.
!>
