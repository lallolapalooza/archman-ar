توثيق الوظيفة/الوحدة الفرعية

subroutine zhptrs (character uplo, integer n, integer nrhs, complex*16, dimension( * ) ap, integer, dimension( * ) ipiv, complex*16, dimension( ldb, * ) b, integer ldb, integer info)

ZHPTRS

الغرض:

!>
!> ZHPTRS تحل نظام المعادلات الخطية A*X = B مع مصفوفة هرميتية معقدة A مخزنة بتنسيق مكدس باستخدام التحلل
!> A = U*D*U**H أو A = L*D*L**H المحسوب بواسطة ZHPTRF.
!>

المعلمات

UPLO 
!>          UPLO هو CHARACTER*1
!>          يحدد ما إذا كانت تفاصيل التحلل مخزنة
!>          كمصفوفة مثلثية علوية أو سفلية.
!>          = 'U':  مثلثية علوية، الشكل هو A = U*D*U**H؛
!>          = 'L':  مثلثية سفلية، الشكل هو A = L*D*L**H.
!>

N

!>          N هو INTEGER
!>          رتبة المصفوفة A.  N >= 0.
!>

NRHS

!>          NRHS هو INTEGER
!>          عدد الأطراف اليمنى، أي عدد أعمدة
!>          المصفوفة B.  NRHS >= 0.
!>

AP

!>          AP هي مصفوفة COMPLEX*16، البعد (N*(N+1)/2)
!>          المصفوفة القطرية الكتلية D والمعاملات المستخدمة
!>          للحصول على العامل U أو L كما تم حسابه بواسطة ZHPTRF، مخزنة كـ
!>          مصفوفة مثلثية مكدسة.
!>

IPIV

!>          IPIV هي مصفوفة INTEGER، البعد (N)
!>          تفاصيل التبادلات والبنية الكتلية لـ D
!>          كما تم تحديده بواسطة ZHPTRF.
!>

B

!>          B هي مصفوفة COMPLEX*16، البعد (LDB,NRHS)
!>          عند الإدخال، مصفوفة الطرف الأيمن B.
!>          عند الخروج، مصفوفة الحل X.
!>

LDB

!>          LDB هو INTEGER
!>          البعد الرائد للمصفوفة B.  LDB >= max(1,N).
!>

INFO

!>          INFO هو INTEGER
!>          = 0:  انتهاء ناجح
!>          < 0: إذا كان INFO = -i، فإن المعلمة i كان لها قيمة غير قانونية
!>
