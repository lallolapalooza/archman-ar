وثائق الوظيفة/الروتين الفرعي

subroutine zhesv (character uplo, integer n, integer nrhs, complex*16, dimension( lda, * ) a, integer lda, integer, dimension( * ) ipiv, complex*16, dimension( ldb, * ) b, integer ldb, complex*16, dimension( * ) work, integer lwork, integer info)

ZHESV تحسب حل نظام المعادلات الخطية A * X = B لمصفوفات HE

الغرض:

!>
!> ZHESV تحسب حل نظام المعادلات الخطية المركب
!>    A * X = B,
!> حيث A مصفوفة هيرميتية N × N و X و B مصفوفتان N × NRHS.
!>
!> تُستخدم طريقة التنقيط القطري لتحليل A إلى:
!>    A = U * D * U**H،  إذا كان UPLO = 'U'، أو
!>    A = L * D * L**H،  إذا كان UPLO = 'L'،
!> حيث U (أو L) هو ناتج ضرب مصفوفة تبديل ومصفوفات مثلثية عليا (سفلى) أحادية،
!> و D مصفوفة هيرميتية وقطرية كتلة بأقطار 1 × 1 و 2 × 2.  بعد ذلك،
!> يُستخدم الشكل المحلل لـ A لحل نظام المعادلات A * X = B.
!>

المعاملات

UPLO 
!>          UPLO هو CHARACTER*1
!>          = 'U':  يتم تخزين المثلث العلوي لـ A؛
!>          = 'L':  يتم تخزين المثلث السفلي لـ A.
!>

N

!>          N هو INTEGER
!>          عدد المعادلات الخطية، أي ترتيب المصفوفة A.  N >= 0.
!>

NRHS

!>          NRHS هو INTEGER
!>          عدد الأطراف اليمنى، أي عدد أعمدة المصفوفة B.  NRHS >= 0.
!>

A

!>          A هي مصفوفة COMPLEX*16، البعد (LDA,N)
!>          عند الدخول، المصفوفة الهيرميتية A.  إذا كان UPLO = 'U'، فإن الجزء
!>          المثلثي العلوي N × N الرئيسي للمصفوفة A يحتوي على الجزء
!>          المثلثي العلوي للمصفوفة A، ولا يتم الرجوع إلى الجزء المثلثي
!>          السفلي الصارم لـ A.  إذا كان UPLO = 'L'، فإن الجزء
!>          المثلثي السفلي N × N الرئيسي للمصفوفة A يحتوي على الجزء
!>          المثلثي السفلي للمصفوفة A، ولا يتم الرجوع إلى الجزء المثلثي
!>          العلوي الصارم لـ A.
!>
!>          عند الخروج، إذا كان INFO = 0، المصفوفة القطرية الكتلية D والمُضاعِفات
!>          المستخدمة للحصول على العامل U أو L من التحليل A = U*D*U**H أو
!>          A = L*D*L**H كما تم حسابه بواسطة ZHETRF.
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرئيسي للمصفوفة A.  LDA >= max(1,N).
!>

IPIV

!>          IPIV هي مصفوفة INTEGER، البعد (N)
!>          تفاصيل التبادلات وبنية الكتلة لـ D، كما تم تحديدها بواسطة ZHETRF.
!>          إذا كان IPIV(k) > 0، تم تبديل الصفوف والأعمدة k و IPIV(k)، و D(k,k)
!>          هي كتلة قطرية 1 × 1.  إذا كان UPLO = 'U' و IPIV(k) = IPIV(k-1) < 0،
!>          تم تبديل الصفوف والأعمدة k-1 و -IPIV(k)، و D(k-1:k,k-1:k) هي كتلة
!>          قطرية 2 × 2.  إذا كان UPLO = 'L' و IPIV(k) = IPIV(k+1) < 0،
!>          تم تبديل الصفوف والأعمدة k+1 و -IPIV(k)، و D(k:k+1,k:k+1) هي كتلة
!>          قطرية 2 × 2.
!>

B

!>          B هي مصفوفة COMPLEX*16، البعد (LDB,NRHS)
!>          عند الدخول، المصفوفة B ذات الطرف الأيمن N × NRHS.
!>          عند الخروج، إذا كان INFO = 0، المصفوفة الحل X ذات الأبعاد N × NRHS.
!>

LDB

!>          LDB هو INTEGER
!>          البعد الرئيسي للمصفوفة B.  LDB >= max(1,N).
!>

WORK

!>          WORK هي مصفوفة COMPLEX*16، البعد (MAX(1,LWORK))
!>          عند الخروج، إذا كان INFO = 0، تعيد WORK(1) قيمة LWORK المثلى.
!>

LWORK

!>          LWORK هو INTEGER
!>          طول WORK.  LWORK >= 1، ولأفضل أداء LWORK >= max(1,N*NB)، حيث NB
!>          هو حجم الكتلة الأمثل لـ ZHETRF.
!>          لـ LWORK < N، سيتم إجراء TRS باستخدام المستوى BLAS 2
!>          لـ LWORK >= N، سيتم إجراء TRS باستخدام المستوى BLAS 3
!>
!>          إذا كان LWORK = -1، فسيتم افتراض استعلام مساحة العمل؛ تقوم الروتين
!>          فقط بحساب الحجم الأمثل لمصفوفة WORK، وتعيد هذا الحجم كالإدخال الأول
!>          للمصفوفة WORK، ولا يتم إصدار أي رسالة خطأ متعلقة بـ LWORK بواسطة
!>          XERBLA.
!>

INFO

!>          INFO هو INTEGER
!>          = 0: خروج ناجح
!>          < 0: إذا كان INFO = -i، فإن المعامل i كان له قيمة غير قانونية
!>          > 0: إذا كان INFO = i، فإن D(i,i) يساوي صفرًا تمامًا.  تم إكمال التحليل،
!>               ولكن المصفوفة القطرية الكتلية D مفردة تمامًا، لذلك لم يكن من الممكن
!>               حساب الحل.
!>
