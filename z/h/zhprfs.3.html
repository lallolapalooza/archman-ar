subroutine zhprfs (character uplo, integer n, integer nrhs, complex*16, dimension( * ) ap, complex*16, dimension( * ) afp, integer, dimension( * ) ipiv, complex*16, dimension( ldb, * ) b, integer ldb, complex*16, dimension( ldx, * ) x, integer ldx, double precision, dimension( * ) ferr, double precision, dimension( * ) berr, complex*16, dimension( * ) work, double precision, dimension( * ) rwork, integer info)

ZHPRFS

الغرض:

!>
!> ZHPRFS يحسن الحل المحسوب لنظام المعادلات الخطية
!> عندما تكون مصفوفة المعاملات هرميتية غير محددة
!> ومعبأة، ويوفر حدود الخطأ وتقديرات الخطأ الخلفي
!> للحل.
!>

المعلمات

UPLO 
!>          UPLO is CHARACTER*1
!>          = 'U':  يتم تخزين المثلث العلوي لـ A؛
!>          = 'L':  يتم تخزين المثلث السفلي لـ A.
!>

N

!>          N is INTEGER
!>          رتبة المصفوفة A. N >= 0.
!>

NRHS

!>          NRHS is INTEGER
!>          عدد الأطراف اليمنى، أي عدد أعمدة
!>          المصفوفتين B و X. NRHS >= 0.
!>

AP

!>          AP is COMPLEX*16 array, dimension (N*(N+1)/2)
!>          المثلث العلوي أو السفلي للمصفوفة الهرميتية A، معبأة
!>          حسب العمود في مصفوفة خطية. يتم تخزين العمود j
!>          من A في المصفوفة AP كما يلي:
!>          if UPLO = 'U', AP(i + (j-1)*j/2) = A(i,j) for 1<=i<=j;
!>          if UPLO = 'L', AP(i + (j-1)*(2*n-j)/2) = A(i,j) for j<=i<=n.
!>

AFP

!>          AFP is COMPLEX*16 array, dimension (N*(N+1)/2)
!>          الشكل المُحلل للمصفوفة A. تحتوي AFP على المصفوفة
!>          القطرية الفرعية D والمضاعفات المستخدمة للحصول على
!>          العامل U أو L من التحليل A = U*D*U**H أو
!>          A = L*D*L**H كما تم حسابه بواسطة ZHPTRF، مخزن كمصفوفة
!>          مثلثة معبأة.
!>

IPIV

!>          IPIV is INTEGER array, dimension (N)
!>          تفاصيل التبادلات والبنية الفرعية لـ D
!>          كما تم تحديده بواسطة ZHPTRF.
!>

B

!>          B is COMPLEX*16 array, dimension (LDB,NRHS)
!>          مصفوفة الطرف الأيمن B.
!>

LDB

!>          LDB is INTEGER
!>          البعد الرئيسي للمصفوفة B. LDB >= max(1,N).
!>

X

!>          X is COMPLEX*16 array, dimension (LDX,NRHS)
!>          عند الإدخال، مصفوفة الحل X، كما تم حسابها بواسطة ZHPTRS.
!>          عند الخروج، مصفوفة الحل المحسنة X.
!>

LDX

!>          LDX is INTEGER
!>          البعد الرئيسي للمصفوفة X. LDX >= max(1,N).
!>

FERR

!>          FERR is DOUBLE PRECISION array, dimension (NRHS)
!>          حد الخطأ الأمامي المقدر لكل متجه حل
!>          X(j) (العمود j من مصفوفة الحل X).
!>          إذا كان XTRUE هو الحل الحقيقي المقابل لـ X(j)، فإن FERR(j)
!>          هو حد علوي مقدر لحجم أكبر
!>          عنصر في (X(j) - XTRUE) مقسومًا على حجم أكبر
!>          عنصر في X(j). التقدير موثوق به مثل
!>          التقدير لـ RCOND، وهو دائمًا تقريبًا تقدير أعلى قليلاً
!>          للخطأ الحقيقي.
!>

BERR

!>          BERR is DOUBLE PRECISION array, dimension (NRHS)
!>          الخطأ النسبي المكون لكل متجه حل
!>          X(j) (أي أصغر تغيير نسبي في
!>          أي عنصر في A أو B يجعل X(j) حلاً دقيقًا).
!>

WORK

!>          WORK is COMPLEX*16 array, dimension (2*N)
!>

RWORK

!>          RWORK is DOUBLE PRECISION array, dimension (N)
!>

INFO

!>          INFO is INTEGER
!>          = 0:  انتهاء ناجح
!>          < 0:  إذا كان INFO = -i، فإن المعلمة i كانت قيمة غير قانونية
!>
