subroutine zhet01_3 (character uplo, integer n, complex*16, dimension( lda, * ) a, integer lda, complex*16, dimension( ldafac, * ) afac, integer ldafac, complex*16, dimension( * ) e, integer, dimension( * ) ipiv, complex*16, dimension( ldc, * ) c, integer ldc, double precision, dimension( * ) rwork, double precision resid)

ZHET01_3

الغرض:

!>
!> ZHET01_3 يعيد بناء مصفوفة هيرميتية غير محددة A من تحليلها
!> L*D*L' أو U*D*U' الكتلية التي تم حسابها بواسطة ZHETRF_RK
!> (أو ZHETRF_BK) ويحسب الباقي
!>    norm( C - A ) / ( N * norm(A) * EPS ),
!> حيث C هي المصفوفة المعاد بناؤها و EPS هي دقة الآلة.
!>

المعلمات

UPLO 
!>          UPLO هو CHARACTER*1
!>          يحدد ما إذا كان الجزء المثلثي العلوي أو السفلي للمصفوفة
!>          الهيرميتية A مخزنًا:
!>          = 'U':  مثلثي علوي
!>          = 'L':  مثلثي سفلي
!>

N

!>          N هو INTEGER
!>          عدد الصفوف والأعمدة للمصفوفة A.  N >= 0.
!>

A

!>          A هو مصفوفة COMPLEX*16، البعد (LDA,N)
!>          المصفوفة الهيرميتية الأصلية A.
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرائد للمصفوفة A.  LDA >= max(1,N)
!>

AFAC

!>          AFAC هو مصفوفة COMPLEX*16، البعد (LDAFAC,N)
!>          قطر المصفوفة القطرية الكتلية D والعوامل U أو L
!>          كما تم حسابها بواسطة ZHETRF_RK و ZHETRF_BK:
!>            أ) عناصر القطر فقط للمصفوفة الهيرميتية القطرية الكتلية
!>               D على قطر A، أي D(k,k) = A(k,k);
!>               (يجب توفير عناصر فوق القطر (أو تحت القطر) لـ D
!>                عند الدخول في المصفوفة E)، و
!>            ب) إذا كان UPLO = 'U': العامل U في الجزء فوق القطر لـ A.
!>               إذا كان UPLO = 'L': العامل L في الجزء تحت القطر لـ A.
!>

LDAFAC

!>          LDAFAC هو INTEGER
!>          البعد الرائد للمصفوفة AFAC.
!>          LDAFAC >= max(1,N).
!>

E

!>          E هي مصفوفة COMPLEX*16، البعد (N)
!>          عند الدخول، تحتوي على عناصر فوق القطر (أو تحت القطر)
!>          للمصفوفة الهيرميتية القطرية الكتلية D
!>          مع كتل قطرية بحجم 1×1 أو 2×2، حيث
!>          إذا كان UPLO = 'U': E(i) = D(i-1,i)، i=2:N، E(1) غير مستخدمة؛
!>          إذا كان UPLO = 'L': E(i) = D(i+1,i)، i=1:N-1، E(N) غير مستخدمة.
!>

IPIV

!>          IPIV هي مصفوفة INTEGER، البعد (N)
!>          مؤشرات المحورية من ZHETRF_RK (أو ZHETRF_BK).
!>

C

!>          C هي مصفوفة COMPLEX*16، البعد (LDC,N)
!>

LDC

!>          LDC هو INTEGER
!>          البعد الرائد للمصفوفة C.  LDC >= max(1,N).
!>

RWORK

!>          RWORK هي مصفوفة DOUBLE PRECISION، البعد (N)
!>

RESID

!>          RESID هو DOUBLE PRECISION
!>          إذا كان UPLO = 'L'، norm(L*D*L' - A) / ( N * norm(A) * EPS )
!>          إذا كان UPLO = 'U'، norm(U*D*U' - A) / ( N * norm(A) * EPS )
!>
