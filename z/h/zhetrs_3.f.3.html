subroutine zhetrs_3 (character uplo, integer n, integer nrhs, complex*16, dimension( lda, * ) a, integer lda, complex*16, dimension( * ) e, integer, dimension( * ) ipiv, complex*16, dimension( ldb, * ) b, integer ldb, integer info)

ZHETRS_3

الغرض:

!> ZHETRS_3 يحل نظام المعادلات الخطية A * X = B مع مصفوفة هرميتية معقدة A باستخدام
!> التحليل الذي تم حسابه بواسطة ZHETRF_RK أو ZHETRF_BK:
!>
!>    A = P*U*D*(U**H)*(P**T) أو A = P*L*D*(L**H)*(P**T)،
!>
!> حيث U (أو L) هي مصفوفة مثلثية عليا (أو سفلى) موحدة،
!> U**H (أو L**H) هي المرافق لـ U (أو L)، P هي مصفوفة تبديل،
!> P**T هي منقولة P، و D هي هرميتية وقطرية كتلة مع كتل قطرية 1x1 و 2x2.
!>
!> هذه الخوارزمية تستخدم BLAS المستوى 3.
!>

المعلمات

UPLO 
!>          UPLO هي CHARACTER*1
!>          تحدد ما إذا كانت تفاصيل التحليل مخزنة كمصفوفة مثلثية عليا أو سفلى:
!>          = 'U':  مثلثية عليا، الشكل هو A = P*U*D*(U**H)*(P**T)؛
!>          = 'L':  مثلثية سفلى، الشكل هو A = P*L*D*(L**H)*(P**T).
!>

N

!>          N هو INTEGER
!>          رتبة المصفوفة A.  N >= 0.
!>

NRHS

!>          NRHS هو INTEGER
!>          عدد الأطراف اليمنى، أي عدد أعمدة المصفوفة B.  NRHS >= 0.
!>

A

!>          A هي مصفوفة COMPLEX*16، حجم (LDA,N)
!>          قطر المصفوفة القطرية الكتلة الهرميتية D وعوامل U أو L
!>          كما تم حسابها بواسطة ZHETRF_RK و ZHETRF_BK:
!>            أ) العناصر القطرية فقط للمصفوفة القطرية الكتلة الهرميتية
!>               D على قطر A، أي D(k,k) = A(k,k)؛
!>               (يجب توفير عناصر فوق القطر (أو تحت القطر) لـ D
!>                عند الإدخال في المصفوفة E)، و
!>            ب) إذا كان UPLO = 'U': العامل U في الجزء فوق القطري لـ A.
!>               إذا كان UPLO = 'L': العامل L في الجزء تحت القطري لـ A.
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرئيسي للمصفوفة A.  LDA >= max(1,N).
!>

E

!>          E هي مصفوفة COMPLEX*16، حجم (N)
!>          عند الإدخال، تحتوي على عناصر فوق القطر (أو تحت القطر)
!>          للمصفوفة القطرية الكتلة الهرميتية D
!>          مع كتل قطرية 1x1 أو 2x2، حيث
!>          إذا كان UPLO = 'U': E(i) = D(i-1,i)،i=2:N، E(1) غير مرجعي؛
!>          إذا كان UPLO = 'L': E(i) = D(i+1,i)،i=1:N-1، E(N) غير مرجعي.
!>
!>          ملاحظة: لكتلة قطرية 1x1 D(k)، حيث
!>          1 <= k <= N، العنصر E(k) غير مرجعي في كلتا
!>          حالات UPLO = 'U' أو UPLO = 'L'.
!>

IPIV

!>          IPIV هي مصفوفة INTEGER، حجم (N)
!>          تفاصيل التبادلات وهيكل الكتلة لـ D
!>          كما تم تحديده بواسطة ZHETRF_RK أو ZHETRF_BK.
!>

B

!>          B هي مصفوفة COMPLEX*16، حجم (LDB,NRHS)
!>          عند الإدخال، مصفوفة الطرف الأيمن B.
!>          عند الخروج، مصفوفة الحل X.
!>

LDB

!>          LDB هو INTEGER
!>          البعد الرئيسي للمصفوفة B.  LDB >= max(1,N).
!>

INFO

!>          INFO هو INTEGER
!>          = 0:  الخروج بنجاح
!>          < 0:  إذا كان INFO = -i، فإن المعلمة i كانت ذات قيمة غير قانونية
!>
