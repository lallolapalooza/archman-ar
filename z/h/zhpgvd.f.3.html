subroutine zhpgvd (integer itype, character jobz, character uplo, integer n, complex*16, dimension( * ) ap, complex*16, dimension( * ) bp, double precision, dimension( * ) w, complex*16, dimension( ldz, * ) z, integer ldz, complex*16, dimension( * ) work, integer lwork, double precision, dimension( * ) rwork, integer lrwork, integer, dimension( * ) iwork, integer liwork, integer info)

ZHPGVD

الغرض:

!>
!> ZHPGVD يحسب جميع القيم الذاتية، واختياريًا، المتجهات الذاتية
!> لمشكلة القيم الذاتية الهرميتية المعممة المعقدة، بالشكل
!> A*x=(lambda)*B*x,  A*Bx=(lambda)*x,  أو B*A*x=(lambda)*x. هنا A و
!> B مفترضان أن تكون هرميتية، مخزنة بتنسيق مضغوط، و B أيضًا
!> موجبة محددة.
!> إذا كانت المتجهات الذاتية مطلوبة، فإنه يستخدم خوارزمية تقسيم و conquista.
!>
!>

المعلمات

ITYPE 
!>          ITYPE هو عدد صحيح (INTEGER)
!>          يحدد نوع المشكلة المراد حلها:
!>          = 1:  A*x = (lambda)*B*x
!>          = 2:  A*B*x = (lambda)*x
!>          = 3:  B*A*x = (lambda)*x
!>

JOBZ

!>          JOBZ هو حرف واحد (CHARACTER*1)
!>          = 'N':  حساب القيم الذاتية فقط؛
!>          = 'V':  حساب القيم الذاتية والمتجهات الذاتية.
!>

UPLO

!>          UPLO هو حرف واحد (CHARACTER*1)
!>          = 'U':  المثلثات العلوية لـ A و B مخزنة؛
!>          = 'L':  المثلثات السفلية لـ A و B مخزنة.
!>

N

!>          N هو عدد صحيح (INTEGER)
!>          رتبة المصفوفتين A و B.  N >= 0.
!>

AP

!>          AP هو مصفوفة مركبة*16 (COMPLEX*16)
!>          عند الدخول، المثلث العلوي أو السفلي للمصفوفة الهرميتية
!>          A، مضغوطة عمودياً في مصفوفة خطية.  العمود j من A
!>          مخزن في المصفوفة AP كالتالي:
!>          إذا كان UPLO = 'U'، فإن AP(i + (j-1)*j/2) = A(i,j) لـ 1<=i<=j؛
!>          إذا كان UPLO = 'L'، فإن AP(i + (j-1)*(2*n-j)/2) = A(i,j) لـ j<=i<=n.
!>
!>          عند الخروج، محتويات AP مدمرة.
!>

BP

!>          BP هو مصفوفة مركبة*16 (COMPLEX*16)
!>          عند الدخول، المثلث العلوي أو السفلي للمصفوفة الهرميتية
!>          B، مضغوطة عمودياً في مصفوفة خطية.  العمود j من B
!>          مخزن في المصفوفة BP كالتالي:
!>          إذا كان UPLO = 'U'، فإن BP(i + (j-1)*j/2) = B(i,j) لـ 1<=i<=j؛
!>          إذا كان UPLO = 'L'، فإن BP(i + (j-1)*(2*n-j)/2) = B(i,j) لـ j<=i<=n.
!>
!>          عند الخروج، العامل المثلثي U أو L من تحليل Cholesky
!>          B = U**H*U أو B = L*L**H، بنفس تنسيق التخزين
!>          مثل B.
!>

W

!>          W هي مصفوفة دقة مزدوجة (DOUBLE PRECISION)
!>          إذا كان INFO = 0، فإن القيم الذاتية بترتيب تصاعدي.
!>

Z

!>          Z هي مصفوفة مركبة*16 (COMPLEX*16)
!>          إذا كان JOBZ = 'V'، فعندما يكون INFO = 0، تحتوي Z على المصفوفة Z للمتجهات الذاتية.
!>          المتجهات الذاتية معيارية كالتالي:
!>          إذا كان ITYPE = 1 أو 2، فإن Z**H*B*Z = I؛
!>          إذا كان ITYPE = 3، فإن Z**H*inv(B)*Z = I.
!>          إذا كان JOBZ = 'N'، فلن يتم استخدام Z.
!>

LDZ

!>          LDZ هو عدد صحيح (INTEGER)
!>          البعد الرئيسي للمصفوفة Z.  LDZ >= 1، وإذا
!>          كان JOBZ = 'V'، فإن LDZ >= max(1,N).
!>

WORK

!>          WORK هي مصفوفة مركبة*16 (COMPLEX*16)
!>          عند الخروج، إذا كان INFO = 0، فإن WORK(1) تعيد LWORK المطلوبة.
!>

LWORK

!>          LWORK هو عدد صحيح (INTEGER)
!>          بعد المصفوفة WORK.
!>          إذا كان N <= 1،               LWORK >= 1.
!>          إذا كان JOBZ = 'N' و N > 1، LWORK >= N.
!>          إذا كان JOBZ = 'V' و N > 1، LWORK >= 2*N.
!>
!>          إذا كان LWORK = -1، فإنه يُفترض استعلام عن مساحة العمل؛ الروتين
!>          يحسب فقط الأحجام المطلوبة للمصفوفات WORK، RWORK، و
!>          IWORK، ويعيد هذه القيم كأول إدخالات في
!>          المصفوفات WORK، RWORK، و IWORK، ولا يصدر أي رسالة خطأ
!>          متعلقة بـ LWORK أو LRWORK أو LIWORK بواسطة XERBLA.
!>

RWORK

!>          RWORK هي مصفوفة دقة مزدوجة (DOUBLE PRECISION)
!>          عند الخروج، إذا كان INFO = 0، فإن RWORK(1) تعيد LRWORK المطلوبة.
!>

LRWORK

!>          LRWORK هو عدد صحيح (INTEGER)
!>          بعد المصفوفة RWORK.
!>          إذا كان N <= 1،               LRWORK >= 1.
!>          إذا كان JOBZ = 'N' و N > 1، LRWORK >= N.
!>          إذا كان JOBZ = 'V' و N > 1، LRWORK >= 1 + 5*N + 2*N**2.
!>
!>          إذا كان LRWORK = -1، فإنه يُفترض استعلام عن مساحة العمل؛ الروتين
!>          يحسب فقط الأحجام المطلوبة للمصفوفات WORK، RWORK، و
!>          IWORK، ويعيد هذه القيم كأول إدخالات في
!>          المصفوفات WORK، RWORK، و IWORK، ولا يصدر أي رسالة خطأ
!>          متعلقة بـ LWORK أو LRWORK أو LIWORK بواسطة XERBLA.
!>

IWORK

!>          IWORK هي مصفوفة أعداد صحيحة (INTEGER)
!>          عند الخروج، إذا كان INFO = 0، فإن IWORK(1) تعيد LIWORK المطلوبة.
!>

LIWORK

!>          LIWORK هو عدد صحيح (INTEGER)
!>          بعد المصفوفة IWORK.
!>          إذا كان JOBZ  = 'N' أو N <= 1، LIWORK >= 1.
!>          إذا كان JOBZ  = 'V' و N > 1، LIWORK >= 3 + 5*N.
!>
!>          إذا كان LIWORK = -1، فإنه يُفترض استعلام عن مساحة العمل؛ الروتين
!>          يحسب فقط الأحجام المطلوبة للمصفوفات WORK، RWORK، و
!>          IWORK، ويعيد هذه القيم كأول إدخالات في
!>          المصفوفات WORK، RWORK، و IWORK، ولا يصدر أي رسالة خطأ
!>          متعلقة بـ LWORK أو LRWORK أو LIWORK بواسطة XERBLA.
!>

INFO

!>          INFO هو عدد صحيح (INTEGER)
!>          = 0:  خروج ناجح
!>          < 0:  إذا كان INFO = -i، فإن المعامل i كان له قيمة غير قانونية
!>          > 0:  ZPPTRF أو ZHPEVD أعاد رمز خطأ:
!>             <= N:  إذا كان INFO = i، فشل ZHPEVD في التقارب؛
!>                    i عنصر خارج القطر لنموذج ثلاثي الأقطار وسيط
!>                    لم يتقارب إلى الصفر؛
!>             > N:   إذا كان INFO = N + i، لـ 1 <= i <= n، فإن
!>                    المصغر الرئيسي الرائد من الرتبة i لـ B ليس موجبًا.
!>                    لم يكن من الممكن إكمال تحليل B ولم يتم حساب أي قيم
!>                    ذاتية أو متجهات ذاتية.
!>
