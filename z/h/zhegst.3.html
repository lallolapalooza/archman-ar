subroutine zhegst (integer itype, character uplo, integer n, complex*16, dimension( lda, * ) a, integer lda, complex*16, dimension( ldb, * ) b, integer ldb, integer info)

ZHEGST

الغرض:

!>
!> ZHEGST تختزل مشكلة القيم الذاتية المعممة الهرميتية المحددة إلى شكل قياسي.
!>
!> إذا كان ITYPE = 1، تكون المشكلة A*x = lambda*B*x،
!> ويتم الكتابة فوق A بـ inv(U**H)*A*inv(U) أو inv(L)*A*inv(L**H)
!>
!> إذا كان ITYPE = 2 أو 3، تكون المشكلة A*B*x = lambda*x أو
!> B*A*x = lambda*x، ويتم الكتابة فوق A بـ U*A*U**H أو L**H*A*L.
!>
!> يجب أن يكون B قد تم تحليله سابقًا على صورة U**H*U أو L*L**H بواسطة ZPOTRF.
!>

المعاملات

ITYPE 
!>          ITYPE هو عدد صحيح
!>          = 1: حساب inv(U**H)*A*inv(U) أو inv(L)*A*inv(L**H);
!>          = 2 أو 3: حساب U*A*U**H أو L**H*A*L.
!>

UPLO

!>          UPLO هو حرف واحد
!>          = 'U':  يتم تخزين المثلث العلوي لـ A ويتم تحليل B إلى
!>                  U**H*U;
!>          = 'L':  يتم تخزين المثلث السفلي لـ A ويتم تحليل B إلى
!>                  L*L**H.
!>

N

!>          N هو عدد صحيح
!>          ترتيب المصفوفتين A و B.  N >= 0.
!>

A

!>          A هي مصفوفة COMPLEX*16، حجمها (LDA,N)
!>          عند الإدخال، المصفوفة الهرميتية A.  إذا كان UPLO = 'U'، فإن
!>          الجزء العلوي المثلثي N-by-N من A يحتوي على الجزء العلوي
!>          المثلثي للمصفوفة A، والجزء السفلي المثلثي الصارم لـ A غير مشار إليه.
!>          إذا كان UPLO = 'L'، فإن الجزء السفلي المثلثي N-by-N من A يحتوي على الجزء السفلي
!>          المثلثي للمصفوفة A، والجزء العلوي المثلثي الصارم لـ A غير مشار إليه.
!>
!>          عند الخروج، إذا كان INFO = 0، فإن المصفوفة المحولة، مخزنة في
!>          نفس تنسيق A.
!>

LDA

!>          LDA هو عدد صحيح
!>          البعد الرئيسي للمصفوفة A.  LDA >= max(1,N).
!>

B

!>          B هي مصفوفة COMPLEX*16، حجمها (LDB,N)
!>          العامل المثلثي من تحليل Cholesky لـ B،
!>          كما تم إرجاعه بواسطة ZPOTRF.
!>          يتم تعديل B بواسطة الروتين ولكن يتم استعادتها عند الخروج.
!>

LDB

!>          LDB هو عدد صحيح
!>          البعد الرئيسي للمصفوفة B.  LDB >= max(1,N).
!>

INFO

!>          INFO هو عدد صحيح
!>          = 0:  خروج ناجح
!>          < 0:  إذا كان INFO = -i، فإن المعامل i كان له قيمة غير قانونية
!>
