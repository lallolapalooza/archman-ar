subroutine zhbevd (jobz, uplo, n, kd, ab, ldab, w, z, ldz, work, lwork, rwork, lrwork, iwork, liwork, info)
ZHBEVD يحسب القيم الذاتية، واختياريًا، المتجهات الذاتية اليسرى و/أو اليمنى لمصفوفات أخرى

توثيق الوظيفة/الإجراء الفرعي

subroutine zhbevd (character jobz, character uplo, integer n, integer kd, complex*16, dimension( ldab, * ) ab, integer ldab, double precision, dimension( * ) w, complex*16, dimension( ldz, * ) z, integer ldz, complex*16, dimension( * ) work, integer lwork, double precision, dimension( * ) rwork, integer lrwork, integer, dimension( * ) iwork, integer liwork, integer info)

ZHBEVD يحسب القيم الذاتية، واختياريًا، المتجهات الذاتية لمصفوفة نطاق هيرميتية معقدة A. إذا كانت المتجهات الذاتية مطلوبة، فإنها تستخدم خوارزمية الانقسام والغزو.

الغرض:

!>
!> ZHBEVD يحسب جميع القيم الذاتية، واختياريًا، المتجهات الذاتية
!> لمصفوفة نطاق هيرميتية معقدة A. إذا كانت المتجهات الذاتية مطلوبة،
!> فإنها تستخدم خوارزمية الانقسام والغزو.
!>
!>

المعلمات

JOBZ 
!>          JOBZ هو CHARACTER*1
!>          = 'N':  حساب القيم الذاتية فقط؛
!>          = 'V':  حساب القيم الذاتية والمتجهات الذاتية.
!>

UPLO

!>          UPLO هو CHARACTER*1
!>          = 'U':  يتم تخزين المثلث العلوي من A؛
!>          = 'L':  يتم تخزين المثلث السفلي من A.
!>

N

!>          N هو INTEGER
!>          ترتيب المصفوفة A.  N >= 0.
!>

KD

!>          KD هو INTEGER
!>          عدد الأقطار الفائقة للمصفوفة A إذا كان UPLO = 'U'،
!>          أو عدد الأقطار الفرعية إذا كان UPLO = 'L'.  KD >= 0.
!>

AB

!>          AB هو مصفوفة COMPLEX*16، البعد (LDAB, N)
!>          عند الإدخال، المثلث العلوي أو السفلي للمصفوفة النطاقية الهيرميتية
!>          A، مخزنة في أول KD+1 صفوف من المصفوفة.  يتم تخزين العمود
!>          j من A في العمود j من المصفوفة AB
!>          كما يلي:
!>          إذا كان UPLO = 'U'، فإن AB(kd+1+i-j,j) = A(i,j) للقيمة max(1,j-kd)<=i<=j؛
!>          إذا كان UPLO = 'L'، فإن AB(1+i-j,j)    = A(i,j) للقيمة j<=i<=min(n,j+kd).
!>
!>          عند الخروج، يتم الكتابة فوق AB بقيم تم إنشاؤها أثناء
!>          الاختزال إلى شكل ثلاثي الأقطار.  إذا كان UPLO = 'U'، يتم إرجاع
!>          القطر الفائق الأول والقطر الرئيسي للمصفوفة ثلاثية الأقطار T
!>          في الصفوف KD و KD+1 من AB، وإذا كان UPLO = 'L'، يتم إرجاع
!>          القطر الرئيسي والقطر الفرعي الأول لـ T في
!>          الصفين الأولين من AB.
!>

LDAB

!>          LDAB هو INTEGER
!>          البعد الرئيسي للمصفوفة AB.  LDAB >= KD + 1.
!>

W

!>          W هي مصفوفة DOUBLE PRECISION، البعد (N)
!>          إذا كان INFO = 0، فإن القيم الذاتية مرتبة تصاعديًا.
!>

Z

!>          Z هي مصفوفة COMPLEX*16، البعد (LDZ, N)
!>          إذا كان JOBZ = 'V'، فعندما يكون INFO = 0، تحتوي Z على المتجهات الذاتية
!>          المتعامدة للمصفوفة A، مع احتواء العمود i من Z
!>          على المتجه الذاتي المرتبط بـ W(i).
!>          إذا كان JOBZ = 'N'، فلا يتم الرجوع إلى Z.
!>

LDZ

!>          LDZ هو INTEGER
!>          البعد الرئيسي للمصفوفة Z.  LDZ >= 1، وإذا
!>          كان JOBZ = 'V'، فإن LDZ >= max(1,N).
!>

WORK

!>          WORK هي مصفوفة COMPLEX*16، البعد (MAX(1,LWORK))
!>          عند الخروج، إذا كان INFO = 0، فإن WORK(1) تُرجع LWORK الأمثل.
!>

LWORK

!>          LWORK هو INTEGER
!>          بعد المصفوفة WORK.
!>          إذا كان N <= 1، يجب أن يكون LWORK على الأقل 1.
!>          إذا كان JOBZ = 'N' و N > 1، يجب أن يكون LWORK على الأقل N.
!>          إذا كان JOBZ = 'V' و N > 1، يجب أن يكون LWORK على الأقل 2*N**2.
!>
!>          إذا كان LWORK = -1، فافترض أن هناك استعلامًا عن مساحة العمل؛
!>          تقوم الإجراءات الفرعية فقط بحساب الأحجام المثلى لمصفوفات
!>          WORK و RWORK و IWORK، وتُرجع هذه القيم كإدخالات أولى للمصفوفات
!>          WORK و RWORK و IWORK، ولا يتم إصدار أي رسالة خطأ
!>          متعلقة بـ LWORK أو LRWORK أو LIWORK بواسطة XERBLA.
!>

RWORK

!>          RWORK هي مصفوفة DOUBLE PRECISION،
!>                                         البعد (LRWORK)
!>          عند الخروج، إذا كان INFO = 0، فإن RWORK(1) تُرجع LRWORK الأمثل.
!>

LRWORK

!>          LRWORK هو INTEGER
!>          بعد المصفوفة RWORK.
!>          إذا كان N <= 1، يجب أن يكون LRWORK على الأقل 1.
!>          إذا كان JOBZ = 'N' و N > 1، يجب أن يكون LRWORK على الأقل N.
!>          إذا كان JOBZ = 'V' و N > 1، يجب أن يكون LRWORK على الأقل
!>                        1 + 5*N + 2*N**2.
!>
!>          إذا كان LRWORK = -1، فافترض أن هناك استعلامًا عن مساحة العمل؛
!>          تقوم الإجراءات الفرعية فقط بحساب الأحجام المثلى لمصفوفات
!>          WORK و RWORK و IWORK، وتُرجع هذه القيم كإدخالات أولى للمصفوفات
!>          WORK و RWORK و IWORK، ولا يتم إصدار أي رسالة خطأ
!>          متعلقة بـ LWORK أو LRWORK أو LIWORK بواسطة XERBLA.
!>

IWORK

!>          IWORK هي مصفوفة INTEGER، البعد (MAX(1,LIWORK))
!>          عند الخروج، إذا كان INFO = 0، فإن IWORK(1) تُرجع LIWORK الأمثل.
!>

LIWORK

!>          LIWORK هو INTEGER
!>          بعد المصفوفة IWORK.
!>          إذا كان JOBZ = 'N' أو N <= 1، يجب أن يكون LIWORK على الأقل 1.
!>          إذا كان JOBZ = 'V' و N > 1، يجب أن يكون LIWORK على الأقل 3 + 5*N .
!>
!>          إذا كان LIWORK = -1، فافترض أن هناك استعلامًا عن مساحة العمل؛
!>          تقوم الإجراءات الفرعية فقط بحساب الأحجام المثلى لمصفوفات
!>          WORK و RWORK و IWORK، وتُرجع هذه القيم كإدخالات أولى للمصفوفات
!>          WORK و RWORK و IWORK، ولا يتم إصدار أي رسالة خطأ
!>          متعلقة بـ LWORK أو LRWORK أو LIWORK بواسطة XERBLA.
!>

INFO

!>          INFO هو INTEGER
!>          = 0:  انتهاء ناجح.
!>          < 0:  إذا كان INFO = -i، فإن المعلمة i كانت لها قيمة غير قانونية.
!>          > 0:  إذا كان INFO = i، فشلت الخوارزمية في التقارب؛ لم تتقارب
!>                العناصر غير القطرية لشكل ثلاثي الأقطار وسيط إلى الصفر.
!>
