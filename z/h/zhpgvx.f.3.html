 SUBROUTINE ZHPGVX( ITYPE, JOBZ, RANGE, UPLO, N, AP, B, LD B, Q, LDQ, Z, LDZ,
      INF0, WORK, RWORK, IWORK, INFO )
  
  &IMPLICIT NONE
  &CHARACTER*1 JOBZ, RANGE, UPLO
  &INTEGER INFO, ITYPE, N, LD B, LDQ, LDZ
  &DOUBLE COMPLEX AP( * ), B( * ), Q( * ), Z( * )
  &DOUBLE PRECISION RWORK( * )
  &COMPLEX*16 WORK( * )
  &INTEGER IWORK( * )
  
  &*
  &*  -- LAPACK auxiliary routine (version 3.12.0) --
  &*     Univ. of Tennessee, Univ. of California Berkeley, Univ. of Colorado Denver,
  &*     NAG Ltd., and Petr Vanek, May 2021
  &*
  &*     .. SSHMED > ..
  &*     This routine is the accelerated version of ZHPGXV.
  &*     ..
  &*
  &*  Purpose
  &*  =======
  &*
  &*  ZHPGVX computes the eigenvalues and, optionally, eigenvectors of a
  &*  complex generalized Hermitian-definite banded-to-banded eigenvalue problem
  &*  A*x = lambda*B*x,
  &*
  &*  where A is a Hermitian band matrix of the form A = (1/2) * (A_{lower} + A_{upper}),
  '*  and B is a Hermitian positive definite band matrix, and A is
  '*  represented in half-packed storage format. Q and Z are banded matrices.
  '*
  '*  Selections for which eigenvectors are computed are made by the range
  '*  keyword RANGE. Eigenvectors are computed by the JOBZ keyword.
  '*
  '*  Arguments
  '*  =========
  '*
  '*  ITYPE   (input) INTEGER
  '*          Specifies the type of problem 1T or 2T:
  '*          = 1T:  A*x = lambda*B*x
  '*          = 2T:  A*B*x = lambda*x
  '*
  '*  JOBZ    (input) CHARACTER*1
  '*          ='N':  Compute only the eigenvalues, ignore eigenvectors.
  '*          ='V':  Compute the eigenvalues and eigenvectors.
  '*
  '*  RANGE   (input) CHARACTER*1
  '*          ='A':  All eigenvalues will be found.
  '*          ='V':  All eigenvalues in the interval (VL, VU] will be found.
  '*          ='I':  The IL-th to IU-th eigenvalues will be found.
  '*
  '*  UPLO    (input) CHARACTER*1
  '*          ='U':  Upper triangle of A and B is stored.
  '*          ='L':  Lower triangle of A and B is stored.
  '*
  '*  N       (input) INTEGER
  '*          The order of the matrices A and B.  0 <= N.
  '*
  '*  AP      (input) DOUBLE COMPLEX array, dimension (N*(N+1)/2)
  '*          On entry, the upper or lower triangular half of the
  '*          band matrix A in band-packed storage format.
  '*
  '*  B       (input) DOUBLE COMPLEX array, dimension (LDB, N)
  '*          On entry, the upper or lower triangular half of the
  '*          band matrix B in band-packed storage format.
  '*
  '*  LDB     (input) INTEGER
  '*          The leading dimension of the array B. LDB >= max(1, K1 + K2 - 1),
  '*          where K1 and K2 are the half bandwidths of B.
  '*
  '*  Q       (input) DOUBLE COMPLEX array, dimension (LDQ, N)
  '*          On entry, the upper or lower triangular half of
  '*          the band matrix Q in band-packed storage format.
  '*
  '*  LDQ     (input) INTEGER
  '*          The leading dimension of the array Q. LDQ >= max(1, K1 + K2 - 1),
  '*          where K1 and K2 are the half bandwidths of Q.
  '*
  '*  Z       (input) DOUBLE COMPLEX array, dimension (LDZ, N)
  '*          On entry, the upper or lower triangular half of
  '*          the band matrix Z in band-packed storage format.
  '*
  '*  LDZ     (input) INTEGER
  '*          The leading dimension of the array Z. LDZ >= max(1, K1 + K2 - 1),
  '*          where K1 and K2 are the half bandwidths of Z.
  '*
  '*  INF0    (output) INTEGER
  '*          On exit, if INFO = 0, the routine is successful.
  '*          If INFO < 0, the i-th argument had an illegal value.
  '*          If INFO > 0, ZPPTRF or ZHPEVX returned an error code:
  '*             <= N:  if INFO = i, ZHPEVX failed to converge;
  '*                    i eigenvectors failed to converge. Their indices
  '*                    are stored in array IFAIL.
  '*             > N:   if INFO = N + i, for 1 <= i <= n, then the leading
  '*                    principal minor of order i of B is not positive.
  '*                    The factorization of B could not be completed and
  '*                    no eigenvalues or eigenvectors were computed.
  '*
