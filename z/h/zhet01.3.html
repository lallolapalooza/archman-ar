توثيق الوظيفة/الروتين الفرعي

subroutine zhet01 (character uplo, integer n, complex*16, dimension( lda, * ) a, integer lda, complex*16, dimension( ldafac, * ) afac, integer ldafac, integer, dimension( * ) ipiv, complex*16, dimension( ldc, * ) c, integer ldc, double precision, dimension( * ) rwork, double precision resid)

ZHET01

الغرض:

!>
!> ZHET01 يعيد بناء مصفوفة هرميتية غير محددة A من تحليلها L*D*L' أو U*D*U'
!> ويحسب معيار الباقي
!>    norm( C - A ) / ( N * norm(A) * EPS ),
!> حيث C هي المصفوفة المعاد بناؤها، EPS هو الخطأ المطبعي للآلة،
!> L' هو تبديل مترافق لـ L، و U' هو تبديل مترافق لـ U.
!>

المعلمات

UPLO 
!>          UPLO هو CHARACTER*1
!>          يحدد ما إذا كان الجزء العلوي أو السفلي الثلاثي للمصفوفة
!>          الهرميتية A مخزنًا:
!>          = 'U':  علوي ثلاثي
!>          = 'L':  سفلي ثلاثي
!>

N

!>          N هو INTEGER
!>          عدد الصفوف والأعمدة للمصفوفة A.  N >= 0.
!>

A

!>          A هو COMPLEX*16 array, dimension (LDA,N)
!>          المصفوفة الهرميتية الأصلية A.
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرائد للمصفوفة A.  LDA >= max(1,N)
!>

AFAC

!>          AFAC هو COMPLEX*16 array, dimension (LDAFAC,N)
!>          الشكل المحلل للمصفوفة A.  AFAC يحتوي على المصفوفة القطرية
!>          الكتلية D والمضاعفات المستخدمة للحصول على العامل L أو U
!>          من التحليل الكتلي L*D*L' أو U*D*U' كما هو محسوب بواسطة ZHETRF.
!>

LDAFAC

!>          LDAFAC هو INTEGER
!>          البعد الرائد للمصفوفة AFAC.  LDAFAC >= max(1,N).
!>

IPIV

!>          IPIV هو INTEGER array, dimension (N)
!>          مؤشرات المحور من ZHETRF.
!>

C

!>          C هو COMPLEX*16 array, dimension (LDC,N)
!>

LDC

!>          LDC هو INTEGER
!>          البعد الرائد للمصفوفة C.  LDC >= max(1,N).
!>

RWORK

!>          RWORK هو DOUBLE PRECISION array, dimension (N)
!>

RESID

!>          RESID هو DOUBLE PRECISION
!>          إذا كان UPLO = 'L'، فإن norm(L*D*L' - A) / ( N * norm(A) * EPS )
!>          إذا كان UPLO = 'U'، فإن norm(U*D*U' - A) / ( N * norm(A) * EPS )
!>
