توثيق الوظيفة/الإجراء الفرعي

subroutine zhetrd_2stage (character vect, character uplo, integer n, complex*16, dimension( lda, * ) a, integer lda, double precision, dimension( * ) d, double precision, dimension( * ) e, complex*16, dimension( * ) tau, complex*16, dimension( * ) hous2, integer lhous2, complex*16, dimension( * ) work, integer lwork, integer info)

ZHETRD_2STAGE

الغرض:

!>
!> ZHETRD_2STAGE تختزل مصفوفة هيرميتية معقدة A إلى شكل متماثل حقيقي ثلاثي الأقطار T عن طريق تحويل تشابه موحد:
!> Q1**H Q2**H* A * Q2 * Q1 = T.
!>

المعلمات

VECT 
!>          VECT is CHARACTER*1
!>          = 'N':  لا حاجة لتمثيل هاوسهولدر،
!>                  خاصة للمرحلة الثانية (نطاق إلى
!>                  ثلاثي الأقطار) وبالتالي LHOUS2 بحجم max(1, 4*N);
!>          = 'V':  يلزم تمثيل هاوسهولدر لـ
!>                  إما توليد Q1 Q2 أو لتطبيق Q1 Q2،
!>                  ثم يتم الاستعلام عن LHOUS2 وحسابه.
!>                  (غير متاح في هذا الإصدار).
!>

UPLO

!>          UPLO is CHARACTER*1
!>          = 'U':  يتم تخزين الجزء العلوي من A؛
!>          = 'L':  يتم تخزين الجزء السفلي من A.
!>

N

!>          N is INTEGER
!>          ترتيب المصفوفة A.  N >= 0.
!>

A

!>          A is COMPLEX*16 array, dimension (LDA,N)
!>          عند الدخول، مصفوفة هيرميتية A.  إذا كان UPLO = 'U'، فإن الجزء العلوي
!>          من A بحجم N-by-N يحتوي على الجزء العلوي
!>          من المصفوفة A، ولا يتم الرجوع إلى الجزء السفلي الصارم
!>          من A. إذا كان UPLO = 'L'، فإن الجزء السفلي
!>          من A بحجم N-by-N يحتوي على الجزء السفلي
!>          من المصفوفة A، ولا يتم الرجوع إلى الجزء العلوي الصارم
!>          من A.
!>          عند الخروج، إذا كان UPLO = 'U'، فإن الحافة الفوقية
!>          من A يتم استبدالها بالعناصر المقابلة للمصفوفة الداخلية AB
!>          نطاقية القطر، والعناصر فوق
!>          الحافة الفوقية KD، مع المصفوفة TAU، تمثل الوحدة
!>          المصفوفة Q1 كناتج من الانعكاسات الأولية؛ إذا كان UPLO
!>          = 'L'، يتم استبدال القطر والحافة تحت القطرية من A
!>          بالعناصر المقابلة للمصفوفة الداخلية AB
!>          نطاقية القطر، والعناصر تحت الحافة تحت القطرية KD، مع
!>          المصفوفة TAU، تمثل الوحدة
!>          المصفوفة Q1 كناتج من الانعكاسات الأولية. انظر المزيد من التفاصيل.
!>

LDA

!>          LDA is INTEGER
!>          البعد الرائد للمصفوفة A.  LDA >= max(1,N).
!>

D

!>          D is DOUBLE PRECISION array, dimension (N)
!>          عناصر القطر للمصفوفة الثلاثية T.
!>

E

!>          E is DOUBLE PRECISION array, dimension (N-1)
!>          عناصر القطر غير القطرية للمصفوفة الثلاثية T.
!>

TAU

!>          TAU is COMPLEX*16 array, dimension (N-KD)
!>          العوامل القياسية للانعكاسات الأولية
!>          للمرحلة الأولى (انظر المزيد من التفاصيل).
!>

HOUS2

!>          HOUS2 is COMPLEX*16 array, dimension (MAX(1,LHOUS2))
!>          يخزن تمثيل هاوسهولدر للمرحلة الثانية
!>          نطاق إلى ثلاثي الأقطار.
!>

LHOUS2

!>          LHOUS2 is INTEGER
!>          بعد المصفوفة HOUS2.
!>          LHOUS2 >= 1.
!>
!>          إذا كان LWORK = -1، أو LHOUS2 = -1،
!>          فإن الاستعلام مفترض؛ الروتين
!>          يحسب فقط الحجم الأمثل لمصفوفة HOUS2، ويعيد
!>          هذه القيمة كأول إدخال في مصفوفة HOUS2، ولا يصدر خطأ
!>          رسالة تتعلق بـ LHOUS2 بواسطة XERBLA.
!>          إذا كان VECT='N'، LHOUS2 = max(1, 4*n)؛
!>          إذا كان VECT='V'، الخيار غير متاح بعد.
!>

WORK

!>          WORK is COMPLEX*16 array, dimension (MAX(1,LWORK))
!>          عند الخروج، إذا كان INFO = 0، يعيد WORK(1) الحجم الأمثل لـ LWORK.
!>

LWORK

!>          LWORK is INTEGER
!>          بعد المصفوفة WORK.
!>          إذا كان N = 0، LWORK >= 1، وإلا LWORK = MAX(1, dimension).
!>
!>          إذا كان LWORK = -1، أو LHOUS2 = -1،
!>          فإن الاستعلام عن مساحة العمل مفترض؛ الروتين
!>          يحسب فقط الحجم الأمثل لمصفوفة WORK، ويعيد
!>          هذه القيمة كأول إدخال في مصفوفة WORK، ولا يصدر خطأ
!>          رسالة تتعلق بـ LWORK بواسطة XERBLA.
!>          LWORK = MAX(1, dimension) حيث
!>          dimension   = max(stage1,stage2) + (KD+1)*N
!>                      = N*KD + N*max(KD+1,FACTOPTNB)
!>                        + max(2*KD*KD, KD*NTHREADS)
!>                        + (KD+1)*N
!>          حيث KD هو حجم التجميع للاختزال،
!>          FACTOPTNB هو التجميع المستخدم بواسطة
!>          خوارزمية QR أو LQ، وعادة ما يكون FACTOPTNB=128 خيارًا جيدًا
!>          NTHREADS هو عدد الأوتار المستخدمة عندما
!>          يتم تمكين تجميع openMP، وإلا =1.
!>

INFO

!>          INFO is INTEGER
!>          = 0:  نجاح الخروج
!>          < 0:  إذا كان INFO = -i، فإن الوسيط i كان له قيمة غير قانونية
!>
