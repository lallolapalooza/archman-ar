subroutine zhseqr (character job, character compz, integer n, integer ilo, integer ihi, complex*16, dimension( ldh, * ) h, integer ldh, complex*16, dimension( * ) w, complex*16, dimension( ldz, * ) z, integer ldz, complex*16, dimension( * ) work, integer lwork, integer info)

ZHSEQR

الغرض:

!>
!>    ZHSEQR يحسب القيم الذاتية لمصفوفة هسنبرغ H
!>    وبشكل اختياري، مصفوفتي T و Z من تحليل شور
!>    H = Z T Z**H، حيث T مصفوفة مثلثة عليا (شكل شور)،
!>    و Z هي مصفوفة المتجهات الشورية الوحدوية.
!>
!>    بشكل اختياري يمكن ضرب Z في مصفوفة وحدوية مدخلة Q بحيث
!>    يمكن لهذه الوظيفة إعطاء تحليل شور لمصفوفة A التي تم
!>    اختزالها إلى شكل هسنبرغ H بواسطة المصفوفة الوحدوية Q:
!>    A = Q*H*Q**H = (QZ)*T*(QZ)**H.
!>

المعلمات

JOB 
!>          JOB هو CHARACTER*1
!>           = 'E':  حساب القيم الذاتية فقط؛
!>           = 'S':  حساب القيم الذاتية وشكل شور T.
!>

COMPZ

!>          COMPZ هو CHARACTER*1
!>           = 'N':  لا يتم حساب متجهات شور؛
!>           = 'I':  يتم تهيئة Z إلى مصفوفة الوحدة ويتم إرجاع مصفوفة
!>                   متجهات شور H المسماة Z؛
!>           = 'V':  يجب أن تحتوي Z على مصفوفة وحدوية Q عند الدخول،
!>                   ويتم إرجاع حاصل ضرب Q*Z.
!>

N

!>          N هو INTEGER
!>           رتبة المصفوفة H.  N >= 0.
!>

ILO

!>          ILO هو INTEGER
!>

IHI

!>          IHI هو INTEGER
!>
!>           يفترض أن H بالفعل مثلثة عليا في الصفوف والأعمدة
!>           1:ILO-1 و IHI+1:N. عادة ما يتم تعيين ILO و IHI بواسطة
!>           استدعاء سابق لـ ZGEBAL، ثم يتم تمريرهما إلى ZGEHRD
!>           عندما يتم اختزال المصفوفة الناتجة عن ZGEBAL إلى شكل هسنبرغ.
!>           بخلاف ذلك، يجب تعيين ILO و IHI إلى 1 و N على التوالي.
!>           إذا كانت N > 0، فإن 1 <= ILO <= IHI <= N.
!>           إذا كانت N = 0، فإن ILO = 1 و IHI = 0.
!>

H

!>          H هي مصفوفة COMPLEX*16، البعد (LDH,N)
!>           عند الدخول، مصفوفة هسنبرغ العليا H.
!>           عند الخروج، إذا كان INFO = 0 و JOB = 'S'، فإن H تحتوي على
!>           المصفوفة المثلثة العليا T من تحليل شور (شكل شور).
!>           إذا كان INFO = 0 و JOB = 'E'، فإن محتويات H غير محددة عند الخروج.
!>           (القيمة الناتجة لـ H عندما يكون INFO > 0 معطاة تحت وصف INFO أدناه.)
!>
!>           على عكس الإصدارات السابقة من ZHSEQR، قد تقوم هذه الوظيفة
!>           بشكل صريح H(i,j) = 0 إذا كان i > j و j = 1، 2، ... ILO-1
!>           أو j = IHI+1، IHI+2، ... N.
!>

LDH

!>          LDH هو INTEGER
!>           البعد الرئيسي للمصفوفة H. LDH >= max(1,N).
!>

W

!>          W هي مصفوفة COMPLEX*16، البعد (N)
!>           القيم الذاتية المحسوبة. إذا كان JOB = 'S'، يتم تخزين القيم الذاتية
!>           بنفس الترتيب الذي تظهر به على القطر الرئيسي للشكل الشوري
!>           المُرجع في H، مع W(i) = H(i,i).
!>

Z

!>          Z هي مصفوفة COMPLEX*16، البعد (LDZ,N)
!>           إذا كان COMPZ = 'N'، فلا يتم استخدام Z.
!>           إذا كان COMPZ = 'I'، فلا يلزم ضبط Z عند الدخول، وعند الخروج،
!>           إذا كان INFO = 0، فإن Z تحتوي على مصفوفة المتجهات الشورية Z لـ H.
!>           إذا كان COMPZ = 'V'، يجب أن تحتوي Z عند الدخول على مصفوفة وحدوية
!>           Q بحجم N×N، والتي يفترض أنها متطابقة مع مصفوفة الوحدة
!>           باستثناء المصفوفة الفرعية Z(ILO:IHI,ILO:IHI). عند الخروج،
!>           إذا كان INFO = 0، فإن Z تحتوي على Q*Z.
!>           عادة ما تكون Q هي المصفوفة الوحدوية التي تم إنشاؤها بواسطة ZUNGHR
!>           بعد استدعاء ZGEHRD الذي شكل مصفوفة هسنبرغ H.
!>           (القيمة الناتجة لـ Z عندما يكون INFO > 0 معطاة تحت وصف INFO أدناه.)
!>

LDZ

!>          LDZ هو INTEGER
!>           البعد الرئيسي للمصفوفة Z.  إذا كان COMPZ = 'I' أو
!>           COMPZ = 'V'، فإن LDZ >= MAX(1,N).  بخلاف ذلك، LDZ >= 1.
!>

WORK

!>          WORK هي مصفوفة COMPLEX*16، البعد (LWORK)
!>           عند الخروج، إذا كان INFO = 0، فإن WORK(1) ترجع تقديرًا
!>           للقيمة المثلى لـ LWORK.
!>

LWORK

!>          LWORK هو INTEGER
!>           بعد المصفوفة WORK.  LWORK >= max(1,N)
!>           كافٍ ويوفر أداءً جيدًا جدًا وأحيانًا مثاليًا.
!>           ومع ذلك، قد يلزم LWORK بحجم يصل إلى 11*N
!>           للحصول على الأداء الأمثل.  يُنصح بالاستعلام عن مساحة العمل
!>           لتحديد حجم مساحة العمل الأمثل.
!>
!>           إذا كان LWORK = -1، فإن ZHSEQR تقوم بالاستعلام عن مساحة العمل.
!>           في هذه الحالة، تتحقق ZHSEQR من معلمات الإدخال وتقدر
!>           حجم مساحة العمل الأمثل للقيم المعطاة لـ N و ILO و IHI.
!>           يتم إرجاع التقدير في WORK(1).  لا يتم إصدار رسائل خطأ متعلقة بـ LWORK
!>           بواسطة XERBLA.  لا يتم الوصول إلى H ولا Z.
!>

INFO

!>          INFO هو INTEGER
!>             = 0:  انتهاء ناجح
!>             < 0:  إذا كان INFO = -i، فإن الوسيط i كان له قيمة غير قانونية
!>             > 0:  إذا كان INFO = i، فشلت ZHSEQR في حساب جميع
!>                القيم الذاتية.  العناصر 1:ilo-1 و i+1:n من W
!>                تحتوي على تلك القيم الذاتية التي تم حسابها بنجاح.
!>                (الفشل نادر.)
!>
!>                إذا كان INFO > 0 وكان JOB = 'E'، فعند الخروج،
!>                القيم الذاتية غير المتقاربة المتبقية هي القيم الذاتية
!>                لمصفوفة هسنبرغ العليا في الصفوف والأعمدة من ILO
!>                إلى INFO للقيمة النهائية للمصفوفة H.
!>
!>                إذا كان INFO > 0 وكان JOB = 'S'، فعند الخروج
!>
!>           (*)  (القيمة الأولية لـ H)*U  = U*(القيمة النهائية لـ H)
!>
!>                حيث U مصفوفة وحدوية.  القيمة النهائية لـ H
!>                هي هسنبرغ عليا ومثلثة في الصفوف والأعمدة
!>                من INFO+1 إلى IHI.
!>
!>                إذا كان INFO > 0 وكان COMPZ = 'V'، فعند الخروج
!>
!>                  (القيمة النهائية لـ Z)  =  (القيمة الأولية لـ Z)*U
!>
!>                حيث U هي المصفوفة الوحدوية في (*) (بغض النظر عن قيمة JOB).
!>
!>                إذا كان INFO > 0 وكان COMPZ = 'I'، فعند الخروج
!>                      (القيمة النهائية لـ Z)  = U
!>                حيث U هي المصفوفة الوحدوية في (*) (بغض النظر عن قيمة JOB).
!>
!>                إذا كان INFO > 0 وكان COMPZ = 'N'، فلا يتم الوصول إلى Z.
!>

NAME

zhseqr - computes the eigenvalues and Schur decomposition of a complex upper quasi-triangular matrix.

SYNOPSIS

   JOB, WANTT, N, ILO, IHI, A, LDA, B, LDB, WR, WI, WORK, LWORK, INFO
