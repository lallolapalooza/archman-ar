subroutine zhetrd_2stage (character vect, character uplo, integer n, complex*16, dimension( lda, * ) a, integer lda, double precision, dimension( * ) d, double precision, dimension( * ) e, complex*16, dimension( * ) tau, complex*16, dimension( * ) hous2, integer lhous2, complex*16, dimension( * ) work, integer lwork, integer info)

ZHETRD_2STAGE

الغرض:

!>
!> ZHETRD_2STAGE تختزل مصفوفة هرميتية مركبة A إلى شكل ثلاثي الأقطار حقيقي متناظر T عن طريق تحويل تشابه وحدوي:
!> Q1**H Q2**H* A * Q2 * Q1 = T.
!>

المعلمات

VECT 
!>          VECT هو CHARACTER*1
!>          = 'N':  لا حاجة لتمثيل Householder،
!>                  خاصة للمرحلة الثانية (شريط إلى
!>                  ثلاثي الأقطار) وبالتالي LHOUS2 هو بحجم max(1, 4*N);
!>          = 'V':  تمثيل Householder مطلوب لـ
!>                  إما إنشاء Q1 Q2 أو لتطبيق Q1 Q2،
!>                  ثم يجب الاستعلام عن LHOUS2 وحسابه.
!>                  (غير متاح في هذا الإصدار).
!>

UPLO

!>          UPLO هو CHARACTER*1
!>          = 'U':  يتم تخزين الجزء العلوي من A;
!>          = 'L':  يتم تخزين الجزء السفلي من A.
!>

N

!>          N هو INTEGER
!>          رتبة المصفوفة A.  N >= 0.
!>

A

!>          A هو مصفوفة COMPLEX*16، البعد (LDA,N)
!>          عند الإدخال، المصفوفة الهرميتية A.  إذا كان UPLO = 'U'، فإن الجزء العلوي
!>          من A بالحجم N*N يحتوي على الجزء العلوي
!>          من المصفوفة A، والجزء السفلي تمامًا
!>          من A لا تتم الإشارة إليه.  إذا كان UPLO = 'L'، فإن الجزء السفلي
!>          من A بالحجم N*N يحتوي على الجزء السفلي
!>          من المصفوفة A، والجزء العلوي تمامًا
!>          من A لا تتم الإشارة إليه.
!>          عند الخروج، إذا كان UPLO = 'U'، فإن الأقطار الفائقة
!>          لـ A يتم استبدالها بالعناصر المقابلة للمصفوفة الداخلية
!>          ثنائية الأقطار AB، والعناصر فوق
!>          القطر الفائق KD، مع المصفوفة TAU، تمثل المصفوفة الوحدوية
!>          Q1 كمنتج للمنكسات الأولية؛ إذا كان UPLO
!>          = 'L'، فإن الأقطار والأقطار الفرعية السفلية لـ A يتم استبدالها
!>          بالعناصر المقابلة للمصفوفة الداخلية ثنائية الأقطار
!>          AB، والعناصر أسفل الأقطار الفرعية KD، مع
!>          المصفوفة TAU، تمثل المصفوفة الوحدوية Q1 كمنتج
!>          للمنكسات الأولية. انظر التفاصيل الإضافية.
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرائد للمصفوفة A.  LDA >= max(1,N).
!>

D

!>          D هي مصفوفة DOUBLE PRECISION، البعد (N)
!>          العناصر القطرية للمصفوفة ثلاثية الأقطار T.
!>

E

!>          E هي مصفوفة DOUBLE PRECISION، البعد (N-1)
!>          العناصر خارج القطر للمصفوفة ثلاثية الأقطار T.
!>

TAU

!>          TAU هي مصفوفة COMPLEX*16، البعد (N-KD)
!>          العوامل القياسية للمنكسات الأولية
!>          للمرحلة الأولى (انظر التفاصيل الإضافية).
!>

HOUS2

!>          HOUS2 هي مصفوفة COMPLEX*16، البعد (MAX(1,LHOUS2))
!>          تخزن تمثيل Householder للمرحلة الثانية
!>          من الشريط إلى ثلاثي الأقطار.
!>

LHOUS2

!>          LHOUS2 هو INTEGER
!>          بعد المصفوفة HOUS2.
!>          LHOUS2 >= 1.
!>
!>          إذا كان LWORK = -1، أو LHOUS2 = -1،
!>          فيعتبر ذلك استعلامًا؛ الوظيفة
!>          تحسب فقط الحجم الأمثل لمصفوفة HOUS2، وتعيد
!>          هذه القيمة كأول إدخال في مصفوفة HOUS2، ولا يتم إصدار أي رسالة خطأ
!>          متعلقة بـ LHOUS2 بواسطة XERBLA.
!>          إذا كان VECT='N'، فإن LHOUS2 = max(1, 4*n);
!>          إذا كان VECT='V'، الخيار غير متاح بعد.
!>

WORK

!>          WORK هي مصفوفة COMPLEX*16، البعد (MAX(1,LWORK))
!>          عند الخروج، إذا كان INFO = 0، فإن WORK(1) يعيد LWORK الأمثل.
!>

LWORK

!>          LWORK هو INTEGER
!>          بعد المصفوفة WORK.
!>          إذا كان N = 0، فإن LWORK >= 1، وإلا فإن LWORK = MAX(1, dimension).
!>
!>          إذا كان LWORK = -1، أو LHOUS2 = -1،
!>          فيعتبر ذلك استعلامًا لمساحة العمل؛ الوظيفة
!>          تحسب فقط الحجم الأمثل للمصفوفة WORK، وتعيد
!>          هذه القيمة كأول إدخال في مصفوفة WORK، ولا يتم إصدار أي رسالة خطأ
!>          متعلقة بـ LWORK بواسطة XERBLA.
!>          LWORK = MAX(1, dimension) حيث
!>          dimension   = max(stage1,stage2) + (KD+1)*N
!>                      = N*KD + N*max(KD+1,FACTOPTNB)
!>                        + max(2*KD*KD, KD*NTHREADS)
!>                        + (KD+1)*N
!>          حيث KD هو حجم التجميع للاختزال،
!>          FACTOPTNB هو التجميع المستخدم بواسطة
!>          خوارزمية QR أو LQ، عادةً ما يكون FACTOPTNB=128 خيارًا جيدًا
!>          NTHREADS هو عدد الخيوط المستخدمة عندما
!>          يكون تجميع openMP ممكنًا، وإلا =1.
!>

INFO

!>          INFO هو INTEGER
!>          = 0:  انتهاء ناجح
!>          < 0:  إذا كان INFO = -i، فإن الوسيط i كان له قيمة غير قانونية
!>
