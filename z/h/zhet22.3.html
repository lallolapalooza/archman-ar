subroutine zhet22 (integer itype, character uplo, integer n, integer m, integer kband, complex*16, dimension( lda, * ) a, integer lda, double precision, dimension( * ) d, double precision, dimension( * ) e, complex*16, dimension( ldu, * ) u, integer ldu, complex*16, dimension( ldv, * ) v, integer ldv, complex*16, dimension( * ) tau, complex*16, dimension( * ) work, double precision, dimension( * ) rwork, double precision, dimension( 2 ) result)

الغرض:

!>
!>      ZHET22  بشكل عام يتحقق من تفكيك للصيغة
!>
!>              A U = U S
!>
!>      حيث A مصفوفة هيرميتية معقدة، وأعمدة U متعامدة،
!>      و S مصفوفة قطرية (إذا كان KBAND=0) أو مصفوفة متماثلة ثلاثية الأقطار (إذا كان
!>      KBAND=1). إذا كان ITYPE=1، يتم تمثيل U كمصفوفة كثيفة،
!>      وإلا يتم التعبير عن U كناتج تحويلات هاوسهولدر،
!>      وتخزن متجهات هذه التحويلات في المصفوفة  و
!>      الثوابت القياسية في  . سنستخدم الحرف
!>       للإشارة إلى ناتج تحويلات هاوسهولدر
!>      (والذي يجب أن يساوي U).
!>
!>      على وجه التحديد، إذا كان ITYPE=1، فإن:
!>
!>              RESULT(1) = | U**H A U - S | / ( |A| m ulp ) و
!>              RESULT(2) = | I - U**H U | / ( m ulp )
!> 
!>  ITYPE   INTEGER
!>          يحدد نوع الاختبارات التي سيتم إجراؤها.
!>          1: U ممثلة كمصفوفة متعامدة كثيفة:
!>             RESULT(1) = | A - U S U**H | / ( |A| n ulp )   *و
!>             RESULT(2) = | I - U U**H | / ( n ulp )
!>
!>  UPLO    CHARACTER
!>          إذا كان UPLO='U'، سيتم استخدام المثلث العلوي من A ولن
!>          يتم الإشارة إلى المثلث السفلي (الصارم). إذا كان
!>          UPLO='L'، سيتم استخدام المثلث السفلي من A ولن
!>          يتم الإشارة إلى المثلث العلوي (الصارم).
!>          لم يتم تعديله.
!>
!>  N       INTEGER
!>          حجم المصفوفة. إذا كان صفرًا، لا تفعل ZHET22 شيئًا.
!>          يجب أن يكون على الأقل صفرًا.
!>          لم يتم تعديله.
!>
!>  M       INTEGER
!>          عدد أعمدة U. إذا كان صفرًا، لا تفعل ZHET22 شيئًا.
!>          يجب أن يكون على الأقل صفرًا.
!>          لم يتم تعديله.
!>
!>  KBAND   INTEGER
!>          عرض النطاق للمصفوفة. قد يكون صفرًا أو واحدًا فقط.
!>          إذا كان صفرًا، فإن S مصفوفة قطرية، ولا يتم الإشارة إلى E. إذا كان
!>          واحدًا، فإن S مصفوفة ثلاثية الأقطار متماثلة.
!>          لم يتم تعديله.
!>
!>  A       مصفوفة COMPLEX*16، ذات بعد (LDA , N)
!>          المصفوفة الأصلية (غير المفككة). يُفترض أنها
!>          متماثلة، وسيتم الإشارة فقط إلى الجزء العلوي (UPLO='U') أو الجزء السفلي (UPLO='L')
!>          فقط.
!>          لم يتم تعديله.
!>
!>  LDA     INTEGER
!>          البعد الرائد للمصفوفة A. يجب أن يكون على الأقل 1
!>          وعلى الأقل N.
!>          لم يتم تعديله.
!>
!>  D       مصفوفة DOUBLE PRECISION، ذات بعد (N)
!>          القطر للمصفوفة (ثلاثية الأقطار متماثلة).
!>          لم يتم تعديله.
!>
!>  E       مصفوفة DOUBLE PRECISION، ذات بعد (N)
!>          العناصر خارج القطر للمصفوفة (ثلاثية الأقطار متماثلة).
!>          يتم تجاهل E(1)، E(2) هو العنصر (1,2) و (2,1)، إلخ.
!>          لا يتم الإشارة إليه إذا كان KBAND=0.
!>          لم يتم تعديله.
!>
!>  U       مصفوفة COMPLEX*16، ذات بعد (LDU, N)
!>          إذا كان ITYPE=1، تحتوي هذه على المصفوفة المتعامدة في
!>          التفكيك، ممثلة كمصفوفة كثيفة.
!>          لم يتم تعديله.
!>
!>  LDU     INTEGER
!>          البعد الرائد للمصفوفة U. يجب أن يكون LDU على الأقل N
!>          وعلى الأقل 1.
!>          لم يتم تعديله.
!>
!>  V       مصفوفة COMPLEX*16، ذات بعد (LDV, N)
!>          إذا كان ITYPE=2 أو 3، يحتوي المثلث السفلي لهذه المصفوفة على
!>          متجهات هاوسهولدر المستخدمة لوصف المصفوفة المتعامدة في
!>          التفكيك. إذا كان ITYPE=1، فلا يتم الإشارة إليه.
!>          لم يتم تعديله.
!>
!>  LDV     INTEGER
!>          البعد الرائد للمصفوفة V. يجب أن يكون LDV على الأقل N
!>          وعلى الأقل 1.
!>          لم يتم تعديله.
!>
!>  TAU     مصفوفة COMPLEX*16، ذات بعد (N)
!>          إذا كان ITYPE >= 2، فإن TAU(j) هو العامل القياسي لـ
!>          v(j) v(j)**H في تحويل هاوسهولدر H(j) لـ
!>          الناتج  U = H(1)...H(n-2)
!>          إذا كان ITYPE < 2، فلا يتم الإشارة إلى TAU.
!>          لم يتم تعديله.
!>
!>  WORK    مصفوفة COMPLEX*16، ذات بعد (2*N**2)
!>          مساحة العمل.
!>          تم التعديل.
!>
!>  RWORK   مصفوفة DOUBLE PRECISION، ذات بعد (N)
!>          مساحة العمل.
!>          تم التعديل.
!>
!>  RESULT  مصفوفة DOUBLE PRECISION، ذات بعد (2)
!>          القيم المحسوبة بواسطة الاختبارين الموصوفين أعلاه.
!>          القيم محدودة حاليًا بـ 1/ulp، لتجنب الفيضان.
!>          يتم تعديل RESULT(1) دائمًا. يتم تعديل RESULT(2) فقط
!>          إذا كان LDU أكبر من أو يساوي N.
!>          تم التعديل.
!>
