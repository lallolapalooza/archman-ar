subroutine zhesv_aa (character uplo, integer n, integer nrhs, complex*16, dimension( lda, * ) a, integer lda, integer, dimension( * ) ipiv, complex*16, dimension( ldb, * ) b, integer ldb, complex*16, dimension( * ) work, integer lwork, integer info)

ZHESV_AA يحسب حل نظام المعادلات الخطية A * X = B للمصفوفات HE

الغرض:

!>
!> ZHESV_AA يحسب حل نظام المعادلات الخطية المركب
!>    A * X = B،
!> حيث A مصفوفة هيرميتية N×N و X و B مصفوفتان N×NRHS.
!>
!> يستخدم خوارزمية Aasen لتحليل A إلى
!>    A = U**H * T * U،  إذا كان UPLO = 'U'، أو
!>    A = L * T * L**H،  إذا كان UPLO = 'L'،
!> حيث U (أو L) هو ناتج تبديلات ومصفوفات قطرية علوية (سفلية) وحدة. ثم يتم استخدام الشكل المحلل لـ A لحل نظام المعادلات A * X = B.
!>

المعلمات

UPLO 
!>          UPLO هو CHARACTER*1
!>          = 'U':  يتم تخزين المثلث العلوي من A؛
!>          = 'L':  يتم تخزين المثلث السفلي من A.
!>

N

!>          N هو INTEGER
!>          عدد المعادلات الخطية، أي، ترتيب المصفوفة A.  N >= 0.
!>

NRHS

!>          NRHS هو INTEGER
!>          عدد الأطراف اليمنى، أي، عدد أعمدة المصفوفة B.  NRHS >= 0.
!>

A

!>          A هو مصفوفة COMPLEX*16، البعد (LDA,N)
!>          عند الإدخال، المصفوفة الهيرميتية A.  إذا كان UPLO = 'U'، فإن الجزء العلوي
!>          القطري من A بحجم N×N يحتوي على الجزء العلوي القطري للمصفوفة A، والجزء
!>          السفلي القطري الصارم من A لا يتم الرجوع إليه. إذا كان UPLO = 'L'، فإن الجزء
!>          السفلي القطري من A بحجم N×N يحتوي على الجزء السفلي القطري للمصفوفة A، والجزء
!>          العلوي القطري الصارم من A لا يتم الرجوع إليه.
!>
!>          عند الخروج، إذا كان INFO = 0، فإن المصفوفة الثلاثية T والمضاعفات
!>          المستخدمة للحصول على العامل U أو L من التحليل A = U**H*T*U أو A = L*T*L**H كما تم حسابها بواسطة
!>          ZHETRF_AA.
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرئيسي للمصفوفة A.  LDA >= max(1,N).
!>

IPIV

!>          IPIV هي مصفوفة INTEGER، البعد (N)
!>          عند الخروج، تحتوي على تفاصيل التبادلات، أي، تم تبديل الصف والعمود k من A
!>          مع الصف والعمود IPIV(k).
!>

B

!>          B هي مصفوفة COMPLEX*16، البعد (LDB,NRHS)
!>          عند الإدخال، المصفوفة الطرفية اليمنى B بحجم N×NRHS.
!>          عند الخروج، إذا كان INFO = 0، المصفوفة الحل X بحجم N×NRHS.
!>

LDB

!>          LDB هو INTEGER
!>          البعد الرئيسي للمصفوفة B.  LDB >= max(1,N).
!>

WORK

!>          WORK هي مصفوفة COMPLEX*16، البعد (MAX(1,LWORK))
!>          عند الخروج، إذا كان INFO = 0، تعيد WORK(1) قيمة LWORK المثلى.
!>

LWORK

!>          LWORK هو INTEGER
!>          طول WORK.  LWORK >= MAX(1,2*N,3*N-2)، وللحصول على أفضل أداء
!>          LWORK >= max(1,N*NB)، حيث NB هو حجم الكتلة الأمثل لـ ZHETRF_AA.
!>
!>          إذا كان LWORK = -1، فسيتم افتراض استعلام عن مساحة العمل؛ الروتين
!>          يحسب فقط الحجم الأمثل لمصفوفة WORK، ويعيد
!>          هذه القيمة كالإدخال الأول لمصفوفة WORK، ولا يتم إصدار أي رسالة خطأ
!>          متعلقة بـ LWORK بواسطة XERBLA.
!>

INFO

!>          INFO هو INTEGER
!>          = 0: خروج ناجح
!>          < 0: إذا كان INFO = -i، كان الوسيط i غير قانوني
!>          > 0: إذا كان INFO = i، فإن D(i,i) يساوي صفرًا تمامًا. تم اكتمال التحليل،
!>               ولكن المصفوفة القطرية الفرعية D مفردة تمامًا، لذا لم يكن من الممكن حساب الحل.
!>
