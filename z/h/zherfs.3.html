subroutine zherfs (character uplo, integer n, integer nrhs, complex*16, dimension( lda, * ) a, integer lda, complex*16, dimension( ldaf, * ) af, integer ldaf, integer, dimension( * ) ipiv, complex*16, dimension( ldb, * ) b, integer ldb, complex*16, dimension( ldx, * ) x, integer ldx, double precision, dimension( * ) ferr, double precision, dimension( * ) berr, complex*16, dimension( * ) work, double precision, dimension( * ) rwork, integer info)

ZHERFS

الغرض:

!>
!> ZHERFS يحسن الحل المحسوب لنظام المعادلات الخطية
!> عندما تكون مصفوفة المعاملات هيرميتية غير محددة، و
!> يوفر حدود خطأ وتقديرات خطأ خلفية للحل.
!>

المعلمات

UPLO 
!>          UPLO is CHARACTER*1
!>          = 'U':  يتم تخزين المثلث العلوي لـ A؛
!>          = 'L':  يتم تخزين المثلث السفلي لـ A.
!>

N

!>          N is INTEGER
!>          رتبة المصفوفة A.  N >= 0.
!>

NRHS

!>          NRHS is INTEGER
!>          عدد الأطراف اليمنى، أي عدد الأعمدة
!>          للمصفوفتين B و X.  NRHS >= 0.
!>

A

!>          A is COMPLEX*16 array, dimension (LDA,N)
!>          المصفوفة الهيرميتية A.  إذا كان UPLO = 'U'، فإن الجزء العلوي
!>          المثلثي العلوي الذي أبعاده N-by-N من A يحتوي على الجزء العلوي
!>          المثلثي للمصفوفة A، والجزء السفلي المثلثي الصارم من A
!>          غير مستخدم.  إذا كان UPLO = 'L'، فإن الجزء السفلي
!>          المثلثي السفلي الذي أبعاده N-by-N من A يحتوي على الجزء السفلي
!>          المثلثي للمصفوفة A، والجزء العلوي المثلثي الصارم من A
!>          غير مستخدم.
!>

LDA

!>          LDA is INTEGER
!>          البعد الرائد للمصفوفة A.  LDA >= max(1,N).
!>

AF

!>          AF is COMPLEX*16 array, dimension (LDAF,N)
!>          الشكل المفسر للمصفوفة A.  AF يحتوي على المصفوفة القطرية الكتلية D
!>          والمضاعفات المستخدمة للحصول على العامل U أو L من التحليل A = U*D*U**H أو
!>          A = L*D*L**H كما تم حسابه بواسطة ZHETRF.
!>

LDAF

!>          LDAF is INTEGER
!>          البعد الرائد للمصفوفة AF.  LDAF >= max(1,N).
!>

IPIV

!>          IPIV is INTEGER array, dimension (N)
!>          تفاصيل التبادلات وهيكل D الكتلي كما تم تحديده بواسطة ZHETRF.
!>

B

!>          B is COMPLEX*16 array, dimension (LDB,NRHS)
!>          مصفوفة الطرف الأيمن B.
!>

LDB

!>          LDB is INTEGER
!>          البعد الرائد للمصفوفة B.  LDB >= max(1,N).
!>

X

!>          X is COMPLEX*16 array, dimension (LDX,NRHS)
!>          عند الدخول، مصفوفة الحل X، كما تم حسابها بواسطة ZHETRS.
!>          عند الخروج، مصفوفة الحل المحسنة X.
!>

LDX

!>          LDX is INTEGER
!>          البعد الرائد للمصفوفة X.  LDX >= max(1,N).
!>

FERR

!>          FERR is DOUBLE PRECISION array, dimension (NRHS)
!>          حد الخطأ الأمامي المقدر لكل متجه حل
!>          X(j) (العمود j من مصفوفة الحل X).
!>          إذا كان XTRUE هو الحل الحقيقي المقابل لـ X(j)، فإن FERR(j)
!>          هو حد علوي مقدر لحجم أكبر
!>          عنصر في (X(j) - XTRUE) مقسومًا على حجم أكبر
!>          عنصر في X(j).  التقدير موثوق به مثل
!>          التقدير لـ RCOND، وهو دائمًا تقريبًا
!>          تقدير زائد للخطأ الحقيقي.
!>

BERR

!>          BERR is DOUBLE PRECISION array, dimension (NRHS)
!>          الخطأ الرجعي النسبي المكون لكل متجه حل
!>          X(j) (أي، أصغر تغيير نسبي في
!>          أي عنصر من A أو B يجعل X(j) حلاً دقيقًا).
!>

WORK

!>          WORK is COMPLEX*16 array, dimension (2*N)
!>

RWORK

!>          RWORK is DOUBLE PRECISION array, dimension (N)
!>

INFO

!>          INFO is INTEGER
!>          = 0:  خروج ناجح
!>          < 0:  إذا كان INFO = -i، فإن المعلمة i كانت ذات قيمة غير قانونية
!>
