subroutine zhemm (character side, character uplo, integer m, integer n, complex*16 alpha, complex*16, dimension(lda,*) a, integer lda, complex*16, dimension(ldb,*) b, integer ldb, complex*16 beta, complex*16, dimension(ldc,*) c, integer ldc)

ZHEMM

الغرض:

!>
!> ZHEMM  يقوم بواحدة من عمليات المصفوفة-المصفوفة
!>
!>    C := alpha*A*B + beta*C,
!>
!> أو
!>
!>    C := alpha*B*A + beta*C,
!>
!> حيث alpha و beta قيمتان قياسيتان، A مصفوفة هيرميتية و B و
!> C مصفوفتان m × n.
!>

المعلمات

SIDE 
!>          SIDE is CHARACTER*1
!>           عند الإدخال، تحدد  SIDE  ما إذا كانت المصفوفة الهيرميتية A
!>           تظهر على اليسار أو اليمين في العملية كما يلي:
!>
!>              SIDE = 'L' or 'l'   C := alpha*A*B + beta*C,
!>
!>              SIDE = 'R' or 'r'   C := alpha*B*A + beta*C,
!>

UPLO

!>          UPLO is CHARACTER*1
!>           عند الإدخال، تحدد  UPLO  ما إذا كان سيتم الرجوع إلى
!>           الجزء المثلثي العلوي أو السفلي من المصفوفة الهيرميتية A كما يلي:
!>
!>              UPLO = 'U' or 'u'   سيتم الرجوع فقط إلى الجزء المثلثي العلوي من
!>                                  المصفوفة الهيرميتية.
!>
!>              UPLO = 'L' or 'l'   سيتم الرجوع فقط إلى الجزء المثلثي السفلي من
!>                                  المصفوفة الهيرميتية.
!>

M

!>          M is INTEGER
!>           عند الإدخال، تحدد M عدد صفوف المصفوفة C.
!>           يجب أن تكون M أكبر من أو تساوي صفر.
!>

N

!>          N is INTEGER
!>           عند الإدخال، تحدد N عدد أعمدة المصفوفة C.
!>           يجب أن تكون N أكبر من أو تساوي صفر.
!>

ALPHA

!>          ALPHA is COMPLEX*16
!>           عند الإدخال، تحدد ALPHA القيمة العددية alpha.
!>

A

!>          A is COMPLEX*16 array, dimension ( LDA, ka ), where ka is
!>           m  when  SIDE = 'L' or 'l'  and is n  otherwise.
!>           قبل الإدخال مع  SIDE = 'L' or 'l',  يجب أن يحتوي الجزء
!>           m × m من المصفوفة A على المصفوفة الهيرميتية، بحيث
!>           عندما  UPLO = 'U' or 'u',  يجب أن يحتوي الجزء المثلثي العلوي الرئيسي
!>           m × m من المصفوفة A على الجزء المثلثي العلوي من
!>           المصفوفة الهيرميتية والجزء السفلي الصارم من A غير مشار إليه، وعندما
!>           UPLO = 'L' or 'l',  يجب أن يحتوي الجزء المثلثي السفلي الرئيسي
!>           m × m من المصفوفة A على الجزء المثلثي السفلي من
!>           المصفوفة الهيرميتية والجزء العلوي الصارم من A غير مشار إليه.
!>           قبل الإدخال مع  SIDE = 'R' or 'r',  يجب أن يحتوي الجزء
!>           n × n من المصفوفة A على المصفوفة الهيرميتية، بحيث
!>           عندما  UPLO = 'U' or 'u',  يجب أن يحتوي الجزء المثلثي العلوي الرئيسي
!>           n × n من المصفوفة A على الجزء المثلثي العلوي من
!>           المصفوفة الهيرميتية والجزء السفلي الصارم من A غير مشار إليه، وعندما
!>           UPLO = 'L' or 'l',  يجب أن يحتوي الجزء المثلثي السفلي الرئيسي
!>           n × n من المصفوفة A على الجزء المثلثي السفلي من
!>           المصفوفة الهيرميتية والجزء العلوي الصارم من A غير مشار إليه.
!>           لاحظ أن الأجزاء التخيلية للعناصر القطرية لا يلزم تعيينها، فهي تفترض أنها صفر.
!>

LDA

!>          LDA is INTEGER
!>           عند الإدخال، تحدد LDA البعد الأول لـ A كما تم تعريفه في
!>           البرنامج (الفرعي) المستدعي. عندما  SIDE = 'L' or 'l'  فإن
!>           LDA يجب أن يكون على الأقل  max( 1, m )، وإلا فإن LDA يجب أن يكون
!>           على الأقل max( 1, n ).
!>

B

!>          B is COMPLEX*16 array, dimension ( LDB, N )
!>           قبل الإدخال، يجب أن يحتوي الجزء الرئيسي m × n من المصفوفة B على
!>           المصفوفة B.
!>

LDB

!>          LDB is INTEGER
!>           عند الإدخال، تحدد LDB البعد الأول لـ B كما تم تعريفه في
!>           البرنامج (الفرعي) المستدعي. يجب أن يكون LDB على الأقل
!>           max( 1, m ).
!>

BETA

!>          BETA is COMPLEX*16
!>           عند الإدخال، تحدد BETA القيمة العددية beta. عندما يتم
!>           توفير BETA كصفر، فلا يلزم تعيين C عند الإدخال.
!>

C

!>          C is COMPLEX*16 array, dimension ( LDC, N )
!>           قبل الإدخال، يجب أن يحتوي الجزء الرئيسي m × n من المصفوفة C على
!>           المصفوفة C، إلا عندما يكون beta صفرًا، وفي هذه الحالة لا يلزم تعيين C عند الإدخال.
!>           عند الخروج، يتم استبدال المصفوفة C بالمصفوفة المحدثة m × n.
!>

LDC

!>          LDC is INTEGER
!>           عند الإدخال، تحدد LDC البعد الأول لـ C كما تم تعريفه في
!>           البرنامج (الفرعي) المستدعي. يجب أن يكون LDC على الأقل
!>           max( 1, m ).
!>
