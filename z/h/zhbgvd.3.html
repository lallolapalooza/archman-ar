subroutine zhbgvd (character jobz, character uplo, integer n, integer ka, integer kb, complex*16, dimension( ldab, * ) ab, integer ldab, complex*16, dimension( ldbb, * ) bb, integer ldbb, double precision, dimension( * ) w, complex*16, dimension( ldz, * ) z, integer ldz, complex*16, dimension( * ) work, integer lwork, double precision, dimension( * ) rwork, integer lrwork, integer, dimension( * ) iwork, integer liwork, integer info)

ZHBGVD

الغرض:

!>
!> ZHBGVD يحسب جميع القيم الذاتية، وربما، المتجهات الذاتية
!> لمشكلة القيم الذاتية المعممة الهيرميتية المحددة لنطاق مركب،
!> من الشكل A*x=(lambda)*B*x. هنا A و B يفترض أنهما هيرميتيان
!> ومحددان، و B موجب تمامًا أيضًا. إذا كانت المتجهات الذاتية
!> مطلوبة، فإنه يستخدم خوارزمية الانقسام والغزو.
!>
!>

المعلمات

JOBZ 
!>          JOBZ هو CHARACTER*1
!>          = 'N':  حساب القيم الذاتية فقط؛
!>          = 'V':  حساب القيم الذاتية والمتجهات الذاتية.
!>

UPLO

!>          UPLO هو CHARACTER*1
!>          = 'U':  تخزين المثلثات العلوية لـ A و B؛
!>          = 'L':  تخزين المثلثات السفلية لـ A و B.
!>

N

!>          N هو INTEGER
!>          ترتيب المصفوفتين A و B. N >= 0.
!>

KA

!>          KA هو INTEGER
!>          عدد الأقطار الفائقة للمصفوفة A إذا كان UPLO = 'U'،
!>          أو عدد الأقطار الفرعية إذا كان UPLO = 'L'. KA >= 0.
!>

KB

!>          KB هو INTEGER
!>          عدد الأقطار الفائقة للمصفوفة B إذا كان UPLO = 'U'،
!>          أو عدد الأقطار الفرعية إذا كان UPLO = 'L'. KB >= 0.
!>

AB

!>          AB هو مصفوفة COMPLEX*16، البعد (LDAB, N)
!>          عند الدخول، المثلث العلوي أو السفلي للمصفوفة الهيرميتية
!>          المحددة، مخزنة في الصفوف ka+1 الأولى من المصفوفة.
!>          يتم تخزين العمود j من A في العمود j من المصفوفة AB
!>          على النحو التالي:
!>          إذا كان UPLO = 'U'، فإن AB(ka+1+i-j,j) = A(i,j) لـ max(1,j-ka)<=i<=j؛
!>          إذا كان UPLO = 'L'، فإن AB(1+i-j,j) = A(i,j) لـ j<=i<=min(n,j+ka).
!>
!>          عند الخروج، يتم تدمير محتويات AB.
!>

LDAB

!>          LDAB هو INTEGER
!>          البعد الرائد للمصفوفة AB. LDAB >= KA+1.
!>

BB

!>          BB هو مصفوفة COMPLEX*16، البعد (LDBB, N)
!>          عند الدخول، المثلث العلوي أو السفلي للمصفوفة الهيرميتية
!>          المحددة، مخزنة في الصفوف kb+1 الأولى من المصفوفة.
!>          يتم تخزين العمود j من B في العمود j من المصفوفة BB
!>          على النحو التالي:
!>          إذا كان UPLO = 'U'، فإن BB(kb+1+i-j,j) = B(i,j) لـ max(1,j-kb)<=i<=j؛
!>          إذا كان UPLO = 'L'، فإن BB(1+i-j,j) = B(i,j) لـ j<=i<=min(n,j+kb).
!>
!>          عند الخروج، العامل S من تفكيك Cholesky المنقسم
!>          B = S**H*S، كما تم إرجاعه بواسطة ZPBSTF.
!>

LDBB

!>          LDBB هو INTEGER
!>          البعد الرائد للمصفوفة BB. LDBB >= KB+1.
!>

W

!>          W هي مصفوفة DOUBLE PRECISION، البعد (N)
!>          إذا كان INFO = 0، فإن القيم الذاتية مرتبة تصاعديًا.
!>

Z

!>          Z هي مصفوفة COMPLEX*16، البعد (LDZ, N)
!>          إذا كان JOBZ = 'V'، فعندما يكون INFO = 0، تحتوي Z على المصفوفة Z
!>          للمتجهات الذاتية، مع احتواء العمود i من Z على
!>          المتجه الذاتي المرتبط بـ W(i). يتم تطبيع المتجهات الذاتية
!>          بحيث Z**H*B*Z = I.
!>          إذا كان JOBZ = 'N'، فلن يتم الرجوع إلى Z.
!>

LDZ

!>          LDZ هو INTEGER
!>          البعد الرائد للمصفوفة Z. LDZ >= 1، وإذا كان
!>          JOBZ = 'V'، فإن LDZ >= N.
!>

WORK

!>          WORK هي مصفوفة COMPLEX*16، البعد (MAX(1,LWORK))
!>          عند الخروج، إذا كان INFO=0، فإن WORK(1) يعيد LWORK الأمثل.
!>

LWORK

!>          LWORK هو INTEGER
!>          بعد المصفوفة WORK.
!>          إذا كان N <= 1، فإن LWORK >= 1.
!>          إذا كان JOBZ = 'N' و N > 1، فإن LWORK >= N.
!>          إذا كان JOBZ = 'V' و N > 1، فإن LWORK >= 2*N**2.
!>
!>          إذا كان LWORK = -1، فيفترض استعلام عن مساحة العمل؛ الروتين
!>          يحسب فقط الأحجام المثلى للمصفوفات WORK و RWORK و
!>          IWORK، ويعيد هذه القيم كمدخلات أولى
!>          للمصفوفات WORK و RWORK و IWORK، ولا يصدر أي رسالة خطأ
!>          متعلقة بـ LWORK أو LRWORK أو LIWORK بواسطة XERBLA.
!>

RWORK

!>          RWORK هي مصفوفة DOUBLE PRECISION، البعد (MAX(1,LRWORK))
!>          عند الخروج، إذا كان INFO=0، فإن RWORK(1) يعيد LRWORK الأمثل.
!>

LRWORK

!>          LRWORK هو INTEGER
!>          بعد المصفوفة RWORK.
!>          إذا كان N <= 1، فإن LRWORK >= 1.
!>          إذا كان JOBZ = 'N' و N > 1، فإن LRWORK >= N.
!>          إذا كان JOBZ = 'V' و N > 1، فإن LRWORK >= 1 + 5*N + 2*N**2.
!>
!>          إذا كان LRWORK = -1، فيفترض استعلام عن مساحة العمل؛ الروتين
!>          يحسب فقط الأحجام المثلى للمصفوفات WORK و RWORK و
!>          IWORK، ويعيد هذه القيم كمدخلات أولى
!>          للمصفوفات WORK و RWORK و IWORK، ولا يصدر أي رسالة خطأ
!>          متعلقة بـ LWORK أو LRWORK أو LIWORK بواسطة XERBLA.
!>

IWORK

!>          IWORK هي مصفوفة INTEGER، البعد (MAX(1,LIWORK))
!>          عند الخروج، إذا كان INFO=0، فإن IWORK(1) يعيد LIWORK الأمثل.
!>

LIWORK

!>          LIWORK هو INTEGER
!>          بعد المصفوفة IWORK.
!>          إذا كان JOBZ = 'N' أو N <= 1، فإن LIWORK >= 1.
!>          إذا كان JOBZ = 'V' و N > 1، فإن LIWORK >= 3 + 5*N.
!>
!>          إذا كان LIWORK = -1، فيفترض استعلام عن مساحة العمل؛ الروتين
!>          يحسب فقط الأحجام المثلى للمصفوفات WORK و RWORK و
!>          IWORK، ويعيد هذه القيم كمدخلات أولى
!>          للمصفوفات WORK و RWORK و IWORK، ولا يصدر أي رسالة خطأ
!>          متعلقة بـ LWORK أو LRWORK أو LIWORK بواسطة XERBLA.
!>

INFO

!>          INFO هو INTEGER
!>          = 0:  انتهاء ناجح
!>          < 0:  إذا كان INFO = -i، فإن المعلمة i كانت ذات قيمة غير قانونية
!>          > 0:  إذا كان INFO = i، و i هو:
!>             <= N:  فشلت الخوارزمية في التقارب:
!>                    i عناصر خارج القطر لمصفوفة ثلاثية الأقطار وسيطة
!>                    لم تتقارب إلى الصفر؛
!>             > N:   إذا كان INFO = N + i، لـ 1 <= i <= N، فإن ZPBSTF
!>                    أعاد INFO = i: B ليست موجبة تمامًا.
!>                    لم يكن من الممكن إكمال تفكيك B و
!>                    لم يتم حساب أي قيم ذاتية أو متجهات ذاتية.
!>
