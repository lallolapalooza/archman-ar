توثيق الوظيفة/الروتين الفرعي

subroutine zhesv_rook (character uplo, integer n, integer nrhs, complex*16, dimension( lda, * ) a, integer lda, integer, dimension( * ) ipiv, complex*16, dimension( ldb, * ) b, integer ldb, complex*16, dimension( * ) work, integer lwork, integer info)

ZHESV_ROOK تحسب حل نظام المعادلات الخطية A * X = B لمصفوفات HE باستخدام طريقة التبديل القطري المحدود لـ Bunch-Kaufman ("rook")

الغرض:

!>
!> ZHESV_ROOK تحسب حل نظام المعادلات الخطية المعقدة
!>    A * X = B,
!> حيث A مصفوفة هيرميتية N-by-N و X و B مصفوفتان N-by-NRHS.
!>
!> يتم استخدام طريقة التبديل القطري المحدود لـ Bunch-Kaufman () لتعيين
!> A إلى
!>    A = U * D * U**T,  إذا كان UPLO = 'U', أو
!>    A = L * D * L**T,  إذا كان UPLO = 'L',
!> حيث U (أو L) هو حاصل ضرب مصفوفات التبديل والمصفوفات المثلثية العلوية (السفلية)
!> الوحدة، و D مصفوفة هيرميتية وقطرية مجمعة مع كتل قطرية 1-by-1 و 2-by-2.
!>
!> يتم استدعاء ZHETRF_ROOK لحساب تحليل مصفوفة هيرميتية معقدة A باستخدام
!> طريقة التبديل القطري المحدود لـ Bunch-Kaufman ().
!>
!> يتم استخدام الشكل المحلل لـ A لحل نظام
!> المعادلات A * X = B عن طريق استدعاء ZHETRS_ROOK (يستخدم BLAS 2).
!>

المعاملات

UPLO 
!>          UPLO هو CHARACTER*1
!>          = 'U':  المثلث العلوي من A مخزن؛
!>          = 'L':  المثلث السفلي من A مخزن.
!>

N

!>          N هو INTEGER
!>          عدد المعادلات الخطية، أي ترتيب
!>          المصفوفة A.  N >= 0.
!>

NRHS

!>          NRHS هو INTEGER
!>          عدد الأطراف اليمنى، أي عدد الأعمدة
!>          للمصفوفة B.  NRHS >= 0.
!>

A

!>          A هي مصفوفة COMPLEX*16، البعد (LDA,N)
!>          عند الإدخال، المصفوفة الهيرميتية A.  إذا كان UPLO = 'U'، فإن الجزء
!>          العلوي المثلثي الرئيسي N-by-N من A يحتوي على الجزء
!>          المثلثي العلوي للمصفوفة A، والجزء السفلي المثلثي الصارم
!>          من A لا يتم الرجوع إليه.  إذا كان UPLO = 'L'، فإن الجزء
!>          السفلي المثلثي الرئيسي N-by-N من A يحتوي على الجزء
!>          المثلثي السفلي للمصفوفة A، والجزء العلوي المثلثي الصارم
!>          من A لا يتم الرجوع إليه.
!>
!>          عند الخروج، إذا كان INFO = 0، فإن المصفوفة القطرية المجمعة D و
!>          المضروبات المستخدمة للحصول على العامل U أو L من
!>          التحليل A = U*D*U**H أو A = L*D*L**H كما تم حسابها بواسطة
!>          ZHETRF_ROOK.
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرئيسي للمصفوفة A.  LDA >= max(1,N).
!>

IPIV

!>          IPIV هي مصفوفة INTEGER، البعد (N)
!>          تفاصيل التبادلات وهيكل الكتلة لـ D.
!>
!>          إذا كان UPLO = 'U':
!>             يتم تعيين آخر KB عنصر من IPIV فقط.
!>
!>             إذا كان IPIV(k) > 0، فقد تم تبديل الصفوف والأعمدة k و IPIV(k)
!>             و D(k,k) هي كتلة قطرية 1-by-1.
!>
!>             إذا كان IPIV(k) < 0 و IPIV(k-1) < 0، فقد تم تبديل الصفوف و
!>             الأعمدة k و -IPIV(k) وتم تبديل الصفوف و
!>             الأعمدة k-1 و -IPIV(k-1)،
!>             D(k-1:k,k-1:k) هي كتلة قطرية 2-by-2.
!>
!>          إذا كان UPLO = 'L':
!>             يتم تعيين أول KB عنصر من IPIV فقط.
!>
!>             إذا كان IPIV(k) > 0، فقد تم تبديل الصفوف والأعمدة k و IPIV(k)
!>             و D(k,k) هي كتلة قطرية 1-by-1.
!>
!>             إذا كان IPIV(k) < 0 و IPIV(k+1) < 0، فقد تم تبديل الصفوف و
!>             الأعمدة k و -IPIV(k) وتم تبديل الصفوف و
!>             الأعمدة k+1 و -IPIV(k+1)،
!>             D(k:k+1,k:k+1) هي كتلة قطرية 2-by-2.
!>

B

!>          B هي مصفوفة COMPLEX*16، البعد (LDB,NRHS)
!>          عند الإدخال، المصفوفة الطرفية اليمنى B ذات البعد N-by-NRHS.
!>          عند الخروج، إذا كان INFO = 0، فإن مصفوفة الحل X ذات البعد N-by-NRHS.
!>

LDB

!>          LDB هو INTEGER
!>          البعد الرئيسي للمصفوفة B.  LDB >= max(1,N).
!>

WORK

!>          WORK هي مصفوفة COMPLEX*16، البعد (MAX(1,LWORK))
!>          عند الخروج، إذا كان INFO = 0، فإن WORK(1) يعيد LWORK الأمثل.
!>

LWORK

!>          LWORK هو INTEGER
!>          طول WORK.  LWORK >= 1، ولأفضل أداء
!>          LWORK >= max(1,N*NB)، حيث NB هو حجم الكتلة الأمثل لـ
!>          ZHETRF_ROOK.
!>          إذا كان LWORK < N، فسيتم تنفيذ TRS باستخدام المستوى BLAS 2
!>          إذا كان LWORK >= N، فسيتم تنفيذ TRS باستخدام المستوى BLAS 3
!>
!>          إذا كان LWORK = -1، فسيتم افتراض استعلام مساحة العمل؛ الروتين
!>          يحسب فقط الحجم الأمثل لمصفوفة WORK، ويعيد
!>          هذا الحجم كأول عنصر في مصفوفة WORK، ولا يتم إصدار رسالة خطأ
!>          متعلقة بـ LWORK بواسطة XERBLA.
!>

INFO

!>          INFO هو INTEGER
!>          = 0: انتهاء ناجح
!>          < 0: إذا كان INFO = -i، فقد كان المعامل i له قيمة غير قانونية
!>          > 0: إذا كان INFO = i، فإن D(i,i) يساوي صفرًا بالضبط.  التحليل
!>               تم الانتهاء منه، لكن المصفوفة القطرية المجمعة D
!>               فردية تمامًا، لذلك لم يكن من الممكن حساب الحل.
!>
