توثيق الوظيفة/الإجراء الفرعي

subroutine zhetrs_aa (character uplo, integer n, integer nrhs, complex*16, dimension( lda, * ) a, integer lda, integer, dimension( * ) ipiv, complex*16, dimension( ldb, * ) b, integer ldb, complex*16, dimension( * ) work, integer lwork, integer info)

ZHETRS_AA

الغرض:

!>
!> ZHETRS_AA يحل نظام المعادلات الخطية A*X = B مع مصفوفة هرميتية معقدة
!> A باستخدام التحليل A = U**H*T*U أو
!> A = L*T*L**H المحسوب بواسطة ZHETRF_AA.
!>

المعلمات

UPLO 
!>          UPLO هو CHARACTER*1
!>          يحدد ما إذا كانت تفاصيل التحليل مخزنة
!>          كمصفوفة علوية أو سفلية.
!>          = 'U':  مثلثية علوية، الشكل هو A = U**H*T*U;
!>          = 'L':  مثلثية سفلية، الشكل هو A = L*T*L**H.
!>

N

!>          N هو INTEGER
!>          رتبة المصفوفة A.  N >= 0.
!>

NRHS

!>          NRHS هو INTEGER
!>          عدد الأطراف اليمنى، أي عدد الأعمدة
!>          من المصفوفة B.  NRHS >= 0.
!>

A

!>          A هي مصفوفة COMPLEX*16، البعد (LDA,N)
!>          تفاصيل العوامل التي تم حسابها بواسطة ZHETRF_AA.
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرائد للمصفوفة A.  LDA >= max(1,N).
!>

IPIV

!>          IPIV هي مصفوفة INTEGER، البعد (N)
!>          تفاصيل المبادلات كما تم حسابها بواسطة ZHETRF_AA.
!>

B

!>          B هي مصفوفة COMPLEX*16، البعد (LDB,NRHS)
!>          عند الإدخال، مصفوفة الطرف الأيمن B.
!>          عند الخروج، مصفوفة الحل X.
!>

LDB

!>          LDB هو INTEGER
!>          البعد الرائد للمصفوفة B.  LDB >= max(1,N).
!>

WORK

!>          WORK هي مصفوفة COMPLEX*16، البعد (MAX(1,LWORK))
!>

LWORK

!>          LWORK هو INTEGER
!>          بعد المصفوفة WORK.
!>          إذا كان MIN(N,NRHS) = 0، فإن LWORK >= 1، وإلا فإن LWORK >= 3*N-2.
!>
!>          إذا كان LWORK = -1، فإنه يُفترض استعلام عن مساحة العمل؛ تقوم الروتين
!>          بحساب الحد الأدنى لحجم مصفوفة WORK فقط، وتعيد
!>          هذه القيمة كإدخال أول للمصفوفة WORK، ولا يتم إصدار أي رسالة خطأ
!>          متعلقة بـ LWORK بواسطة XERBLA.
!>

INFO

!>          INFO هو INTEGER
!>          = 0:  خروج ناجح
!>          < 0:  إذا كان INFO = -i، فإن المعلمة i كانت ذات قيمة غير قانونية
!>
