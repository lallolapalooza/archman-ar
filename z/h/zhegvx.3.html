subroutine zhegvx (integer itype, character jobz, character range, character uplo, integer n, complex*16, dimension( lda, * ) a, integer lda, complex*16, dimension( ldb, * ) b, integer ldb, double precision vl, double precision vu, integer il, integer iu, double precision abstol, integer m, double precision, dimension( * ) w, complex*16, dimension( ldz, * ) z, integer ldz, complex*16, dimension( * ) work, integer lwork, double precision, dimension( * ) rwork, integer, dimension( * ) iwork, integer, dimension( * ) ifail, integer info)

ZHEGVX

الغرض:

!>
!> ZHEGVX يحسب القيم الذاتية المختارة، واختياريًا، المتجهات الذاتية
!> لمشكلة عامة هرمية محددة، من الشكل
!> A*x=(lambda)*B*x,  A*Bx=(lambda)*x,  أو B*A*x=(lambda)*x. هنا A و
!> B مفترض أن تكون هرمية و B موجبة تمامًا أيضًا.
!> يمكن اختيار القيم الذاتية والمتجهات الذاتية عن طريق تحديد نطاق
!> من القيم أو نطاق من المؤشرات للقيم الذاتية المطلوبة.
!>

المعلمات

ITYPE 
!>          ITYPE هو عدد صحيح
!>          يحدد نوع المشكلة المراد حلها:
!>          = 1:  A*x = (lambda)*B*x
!>          = 2:  A*B*x = (lambda)*x
!>          = 3:  B*A*x = (lambda)*x
!>

JOBZ

!>          JOBZ هو حرف واحد
!>          = 'N':  حساب القيم الذاتية فقط؛
!>          = 'V':  حساب القيم الذاتية والمتجهات الذاتية.
!>

RANGE

!>          RANGE هو حرف واحد
!>          = 'A': سيتم العثور على جميع القيم الذاتية.
!>          = 'V': سيتم العثور على جميع القيم الذاتية في الفترة نصف المفتوحة (VL,VU].
!>          = 'I': سيتم العثور على القيم الذاتية من IL إلى IU.
!>

UPLO

!>          UPLO هو حرف واحد
!>          = 'U':  يتم تخزين المثلثات العليا لـ A و B؛
!>          = 'L':  يتم تخزين المثلثات السفلى لـ A و B.
!>

N

!>          N هو عدد صحيح
!>          رتبة المصفوفات A و B.  N >= 0.
!>

A

!>          A هي مصفوفة COMPLEX*16، ذات أبعاد (LDA, N)
!>          عند الإدخال، المصفوفة الهرمية A.  إذا كان UPLO = 'U'، فإن
!>          الجزء العلوي المثلثي N*N الرائد في A يحتوي على
!>          الجزء العلوي المثلثي للمصفوفة A.  إذا كان UPLO = 'L'، فإن
!>          الجزء السفلي المثلثي N*N الرائد في A يحتوي على
!>          الجزء السفلي المثلثي للمصفوفة A.
!>
!>          عند الخروج، المثلث السفلي (إذا كان UPLO='L') أو المثلث العلوي
!>          (إذا كان UPLO='U') من A، بما في ذلك القطر، يتم تدميره.
!>

LDA

!>          LDA هو عدد صحيح
!>          البعد الرائد للمصفوفة A.  LDA >= max(1,N).
!>

B

!>          B هي مصفوفة COMPLEX*16، ذات أبعاد (LDB, N)
!>          عند الإدخال، المصفوفة الهرمية B.  إذا كان UPLO = 'U'، فإن
!>          الجزء العلوي المثلثي N*N الرائد في B يحتوي على
!>          الجزء العلوي المثلثي للمصفوفة B.  إذا كان UPLO = 'L'، فإن
!>          الجزء السفلي المثلثي N*N الرائد في B يحتوي على
!>          الجزء السفلي المثلثي للمصفوفة B.
!>
!>          عند الخروج، إذا كان INFO <= N، فإن الجزء من B الذي يحتوي على المصفوفة
!>          يتم الكتابة فوقه بواسطة العامل المثلثي U أو L من تحليل Cholesky
!>          B = U**H*U أو B = L*L**H.
!>

LDB

!>          LDB هو عدد صحيح
!>          البعد الرائد للمصفوفة B.  LDB >= max(1,N).
!>

VL

!>          VL هو DOUBLE PRECISION
!>
!>          إذا كان RANGE='V'، الحد الأدنى للفترة التي
!>          سيتم البحث فيها عن القيم الذاتية. VL < VU.
!>          لا يتم الرجوع إليها إذا كان RANGE = 'A' أو 'I'.
!>

VU

!>          VU هو DOUBLE PRECISION
!>
!>          إذا كان RANGE='V'، الحد الأعلى للفترة التي
!>          سيتم البحث فيها عن القيم الذاتية. VL < VU.
!>          لا يتم الرجوع إليها إذا كان RANGE = 'A' أو 'I'.
!>

IL

!>          IL هو عدد صحيح
!>
!>          إذا كان RANGE='I'، مؤشر
!>          أصغر قيمة ذاتية سيتم إرجاعها.
!>          1 <= IL <= IU <= N، إذا كان N > 0؛ IL = 1 و IU = 0 إذا كان N = 0.
!>          لا يتم الرجوع إليها إذا كان RANGE = 'A' أو 'V'.
!>

IU

!>          IU هو عدد صحيح
!>
!>          إذا كان RANGE='I'، مؤشر
!>          أكبر قيمة ذاتية سيتم إرجاعها.
!>          1 <= IL <= IU <= N، إذا كان N > 0؛ IL = 1 و IU = 0 إذا كان N = 0.
!>          لا يتم الرجوع إليها إذا كان RANGE = 'A' أو 'V'.
!>

ABSTOL

!>          ABSTOL هو DOUBLE PRECISION
!>          تسامح الخطأ المطلق للقيم الذاتية.
!>          يتم قبول قيمة ذاتية تقريبية على أنها متقاربة
!>          عندما يتم تحديد أنها تقع في فترة [a,b]
!>          بعرض أقل من أو يساوي
!>
!>          ABSTOL + EPS *   max( |a|,|b| ) ,
!>
!>          حيث EPS هي دقة الآلة.  إذا كان ABSTOL أقل من
!>          أو يساوي الصفر، فسيتم استخدام EPS*|T| بدلاً منه،
!>          حيث |T| هو المعيار 1 للمصفوفة ثلاثية القطر التي تم الحصول عليها
!>          عن طريق اختزال C إلى شكل ثلاثي القطر، حيث C هي المصفوفة المتناظرة
!>          للمشكلة المتناظرة القياسية التي يتم تحويل المشكلة
!>          العامة إليها.
!>
!>          سيتم حساب القيم الذاتية بأكبر دقة عندما يتم تعيين ABSTOL إلى
!>          ضعف حد الاستنزاف 2*DLAMCH('S')، وليس صفرًا.
!>          إذا عادت هذه الوظيفة مع INFO>0، مما يشير إلى أن بعض
!>          المتجهات الذاتية لم تتقارب، حاول تعيين ABSTOL إلى
!>          2*DLAMCH('S').
!>

M

!>          M هو عدد صحيح
!>          إجمالي عدد القيم الذاتية التي تم العثور عليها.  0 <= M <= N.
!>          إذا كان RANGE = 'A'، فإن M = N، وإذا كان RANGE = 'I'، فإن M = IU-IL+1.
!>

W

!>          W هي مصفوفة DOUBLE PRECISION، ذات أبعاد (N)
!>          تحتوي العناصر M الأولى على
!>          القيم الذاتية المختارة بترتيب تصاعدي.
!>

Z

!>          Z هي مصفوفة COMPLEX*16، ذات أبعاد (LDZ, max(1,M))
!>          إذا كان JOBZ = 'N'، فلن يتم الرجوع إلى Z.
!>          إذا كان JOBZ = 'V'، وإذا كان INFO = 0، فإن الأعمدة M الأولى من Z
!>          تحتوي على المتجهات الذاتية المتعامدة للمصفوفة A
!>          المقابلة للقيم الذاتية المختارة، مع احتواء العمود i من Z
!>          على المتجه الذاتي المرتبط بـ W(i).
!>          يتم تطبيع المتجهات الذاتية على النحو التالي:
!>          إذا كان ITYPE = 1 أو 2، فإن Z**T*B*Z = I؛
!>          إذا كان ITYPE = 3، فإن Z**T*inv(B)*Z = I.
!>
!>          إذا فشل متجه ذاتي في التقارب، فإن هذا العمود من Z
!>          يحتوي على أحدث تقريب للمتجه الذاتي، وسيتم
!>          إرجاع مؤشر المتجه الذاتي في IFAIL.
!>          ملاحظة: يجب على المستخدم التأكد من توفير عمود واحد على الأقل (max(1,M))
!>          في المصفوفة Z؛ إذا كان RANGE = 'V'، فإن القيمة الدقيقة لـ M
!>          غير معروفة مسبقًا ويجب استخدام حد علوي.
!>

LDZ

!>          LDZ هو عدد صحيح
!>          البعد الرائد للمصفوفة Z.  LDZ >= 1، وإذا
!>          كان JOBZ = 'V'، فإن LDZ >= max(1,N).
!>

WORK

!>          WORK هي مصفوفة COMPLEX*16، ذات أبعاد
