subroutine zhbgv (character jobz, character uplo, integer n, integer ka, integer kb, complex*16, dimension( ldab, * ) ab, integer ldab, complex*16, dimension( ldbb, * ) bb, integer ldbb, double precision, dimension( * ) w, complex*16, dimension( ldz, * ) z, integer ldz, complex*16, dimension( * ) work, double precision, dimension( * ) rwork, integer info)

ZHBGV

الغرض:

!>
!> ZHBGV تحسب جميع القيم الذاتية، واختياريا، المتجهات الذاتية
!> لمشكلة القيم الذاتية الهرميتية المعرفة والمحدبة المعقدة،
!> بالشكل A*x=(lambda)*B*x. هنا A و B يفترض أنهما هرميتيتان
!> ومحدبتان، و B موجبة قطعية.
!>

المعلمات

JOBZ 
!>          JOBZ is CHARACTER*1
!>          = 'N':  حساب القيم الذاتية فقط؛
!>          = 'V':  حساب القيم الذاتية والمتجهات الذاتية.
!>

UPLO

!>          UPLO is CHARACTER*1
!>          = 'U':  المثلثات العليا لـ A و B مخزنة؛
!>          = 'L':  المثلثات السفلى لـ A و B مخزنة.
!>

N

!>          N is INTEGER
!>          رتبة المصفوفتين A و B.  N >= 0.
!>

KA

!>          KA is INTEGER
!>          عدد الأقطار فوق القطر الرئيسي للمصفوفة A إذا كان UPLO = 'U'،
!>          أو عدد الأقطار تحت القطر الرئيسي إذا كان UPLO = 'L'. KA >= 0.
!>

KB

!>          KB is INTEGER
!>          عدد الأقطار فوق القطر الرئيسي للمصفوفة B إذا كان UPLO = 'U'،
!>          أو عدد الأقطار تحت القطر الرئيسي إذا كان UPLO = 'L'. KB >= 0.
!>

AB

!>          AB is COMPLEX*16 array, dimension (LDAB, N)
!>          عند الإدخال، المثلث العلوي أو السفلي للمصفوفة المحدبة الهرميتية
!>          A، مخزنة في أول ka+1 صف من المصفوفة.  العمود j للمصفوفة A
!>          مخزن في العمود j للمصفوفة AB
!>          كما يلي:
!>          إذا كان UPLO = 'U'، AB(ka+1+i-j,j) = A(i,j) لـ max(1,j-ka)<=i<=j؛
!>          إذا كان UPLO = 'L'، AB(1+i-j,j)    = A(i,j) لـ j<=i<=min(n,j+ka).
!>
!>          عند الخروج، محتويات AB مدمرة.
!>

LDAB

!>          LDAB is INTEGER
!>          البعد الرئيسي للمصفوفة AB.  LDAB >= KA+1.
!>

BB

!>          BB is COMPLEX*16 array, dimension (LDBB, N)
!>          عند الإدخال، المثلث العلوي أو السفلي للمصفوفة المحدبة الهرميتية
!>          B، مخزنة في أول kb+1 صف من المصفوفة.  العمود j للمصفوفة B
!>          مخزن في العمود j للمصفوفة BB
!>          كما يلي:
!>          إذا كان UPLO = 'U'، BB(kb+1+i-j,j) = B(i,j) لـ max(1,j-kb)<=i<=j؛
!>          إذا كان UPLO = 'L'، BB(1+i-j,j)    = B(i,j) لـ j<=i<=min(n,j+kb).
!>
!>          عند الخروج، العامل S من تحلل Cholesky المنفصل
!>          B = S**H*S، كما تم إرجاعه بواسطة ZPBSTF.
!>

LDBB

!>          LDBB is INTEGER
!>          البعد الرئيسي للمصفوفة BB.  LDBB >= KB+1.
!>

W

!>          W is DOUBLE PRECISION array, dimension (N)
!>          إذا كان INFO = 0، القيم الذاتية بترتيب تصاعدي.
!>

Z

!>          Z is COMPLEX*16 array, dimension (LDZ, N)
!>          إذا كان JOBZ = 'V'، فعندما يكون INFO = 0، تحتوي Z على مصفوفة Z
!>          للمتجهات الذاتية، حيث يحمل العمود i للمصفوفة Z المتجه الذاتي
!>          المرتبط بـ W(i). المتجهات الذاتية قياسية بحيث Z**H*B*Z = I.
!>          إذا كان JOBZ = 'N'، فلن يتم الرجوع إلى Z.
!>

LDZ

!>          LDZ is INTEGER
!>          البعد الرئيسي للمصفوفة Z.  LDZ >= 1، وإذا
!>          كان JOBZ = 'V'، فإن LDZ >= N.
!>

WORK

!>          WORK is COMPLEX*16 array, dimension (N)
!>

RWORK

!>          RWORK is DOUBLE PRECISION array, dimension (3*N)
!>

INFO

!>          INFO is INTEGER
!>          = 0:  انتهاء ناجح
!>          < 0:  إذا كان INFO = -i، فإن الوسيط i كان له قيمة غير قانونية
!>          > 0:  إذا كان INFO = i، و i هو:
!>             <= N:  فشل الخوارزمية في التقارب:
!>                    i عنصر خارج القطر الرئيسي لشكل ثلاثي الأقطار وسيط
!>                    لم يتقارب إلى الصفر؛
!>             > N:   إذا كان INFO = N + i، لـ 1 <= i <= N، فإن ZPBSTF
!>                    أرجع INFO = i: B ليست موجبة قطعية.
!>                    لم يتمكن من إكمال تحليل B ولم يتم حساب أي قيم ذاتية أو متجهات ذاتية.
!>
