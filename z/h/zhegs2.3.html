توثيق الوظيفة/الروتين الفرعي

subroutine zhegs2 (integer itype, character uplo, integer n, complex*16, dimension( lda, * ) a, integer lda, complex*16, dimension( ldb, * ) b, integer ldb, integer info)

ZHEGS2 تقلل مشكلة قيمة ذاتية معمم هرميتية محددة إلى الشكل القياسي، باستخدام نتائج التحليل التي تم الحصول عليها من cpotrf (خوارزمية غير محظورة).

الغرض:

!>
!> ZHEGS2 تقلل مشكلة قيمة ذاتية معمم هرميتية محددة معقدة إلى الشكل القياسي.
!>
!> إذا كانت ITYPE = 1، فالمشكلة هي A*x = lambda*B*x,
!> ويتم الكتابة فوق A بـ inv(U**H)*A*inv(U) أو inv(L)*A*inv(L**H)
!>
!> إذا كانت ITYPE = 2 أو 3، فالمشكلة هي A*B*x = lambda*x أو
!> B*A*x = lambda*x، ويتم الكتابة فوق A بـ U*A*U**H أو L**H *A*L.
!>
!> يجب أن تكون B قد تم تحليلها مسبقًا كـ U**H *U أو L*L**H بواسطة ZPOTRF.
!>

المعاملات

ITYPE 
!>          ITYPE هو عدد صحيح
!>          = 1: حساب inv(U**H)*A*inv(U) أو inv(L)*A*inv(L**H);
!>          = 2 أو 3: حساب U*A*U**H أو L**H *A*L.
!>

UPLO

!>          UPLO هو حرف واحد
!>          يحدد ما إذا كان الجزء العلوي أو السفلي المثلثي للمصفوفة A الهرميتية مخزنًا، وكيف تم تحليل B.
!>          = 'U':  مثلثي علوي
!>          = 'L':  مثلثي سفلي
!>

N

!>          N هو عدد صحيح
!>          ترتيب المصفوفات A و B. N >= 0.
!>

A

!>          A هو مصفوفة COMPLEX*16، البعد (LDA,N)
!>          عند الإدخال، المصفوفة الهرميتية A. إذا كان UPLO = 'U'، فإن الجزء العلوي المثلثي n x n من A يحتوي على الجزء العلوي
!>          المثلثي للمصفوفة A، والجزء السفلي المثلثي الصارم من A لا يتم الرجوع إليه. إذا كان UPLO = 'L'، فإن
!>          الجزء السفلي المثلثي n x n من A يحتوي على الجزء السفلي
!>          المثلثي للمصفوفة A، والجزء العلوي المثلثي الصارم من A لا يتم الرجوع إليه.
!>
!>          عند الخروج، إذا كان INFO = 0، فإن المصفوفة المحولة، المخزنة في
!>          نفس تنسيق A.
!>

LDA

!>          LDA هو عدد صحيح
!>          البعد الرئيسي للمصفوفة A. LDA >= max(1,N).
!>

B

!>          B هي مصفوفة COMPLEX*16، البعد (LDB,N)
!>          العامل المثلثي من التحليل Cholesky لـ B،
!>          كما تم إرجاعه بواسطة ZPOTRF.
!>          يتم تعديل B بواسطة الروتين الفرعي ولكن يتم استعادتها عند الخروج.
!>

LDB

!>          LDB هو عدد صحيح
!>          البعد الرئيسي للمصفوفة B. LDB >= max(1,N).
!>

INFO

!>          INFO هو عدد صحيح
!>          = 0:  خروج ناجح.
!>          < 0:  إذا كان INFO = -i، فإن المعامل i كان له قيمة غير قانونية.
!>
