subroutine zhegv (integer itype, character jobz, character uplo, integer n, complex*16, dimension( lda, * ) a, integer lda, complex*16, dimension( ldb, * ) b, integer ldb, double precision, dimension( * ) w, complex*16, dimension( * ) work, integer lwork, double precision, dimension( * ) rwork, integer info)

ZHEGV

الغرض:

!>
!> ZHEGV يحسب جميع القيم الذاتية، واختيارياً، المتجهات الذاتية
!> لمشكلة القيمة الذاتية المعممة الهرميتية المحددة، من الشكل
!> A*x=(lambda)*B*x,  A*Bx=(lambda)*x,  or B*A*x=(lambda)*x.
!> هنا يُفترض أن A و B هرميتيان وأن B أيضًا
!> موجب محدد.
!>

المعلمات

ITYPE 
!>          ITYPE هو عدد صحيح
!>          يحدد نوع المشكلة المراد حلها:
!>          = 1:  A*x = (lambda)*B*x
!>          = 2:  A*B*x = (lambda)*x
!>          = 3:  B*A*x = (lambda)*x
!>

JOBZ

!>          JOBZ هو حرف واحد
!>          = 'N':  حساب القيم الذاتية فقط؛
!>          = 'V':  حساب القيم الذاتية والمتجهات الذاتية.
!>

UPLO

!>          UPLO هو حرف واحد
!>          = 'U':  المثلثات العليا لـ A و B مخزنة؛
!>          = 'L':  المثلثات السفلى لـ A و B مخزنة.
!>

N

!>          N هو عدد صحيح
!>          رتبة المصفوفات A و B.  N >= 0.
!>

A

!>          A هي مصفوفة COMPLEX*16، الأبعاد (LDA, N)
!>          عند الإدخال، المصفوفة الهرميتية A.  إذا كان UPLO = 'U'، فإن
!>          الجزء العلوي المثلثي من N*N في A يحتوي على
!>          الجزء العلوي المثلثي للمصفوفة A.  إذا كان UPLO = 'L'، فإن
!>          الجزء السفلي المثلثي من N*N في A يحتوي على
!>          الجزء السفلي المثلثي للمصفوفة A.
!>
!>          عند الخروج، إذا كان JOBZ = 'V'، فإذا كان INFO = 0، فإن A تحتوي على
!>          المصفوفة Z للمتجهات الذاتية.  يتم تطبيع المتجهات الذاتية
!>          على النحو التالي:
!>          إذا كان ITYPE = 1 أو 2، Z**H*B*Z = I؛
!>          إذا كان ITYPE = 3، Z**H*inv(B)*Z = I.
!>          إذا كان JOBZ = 'N'، فإن الجزء العلوي عند الخروج (إذا كان UPLO='U')
!>          أو الجزء السفلي (إذا كان UPLO='L') من A، بما في ذلك
!>          القطر، يتم تدميره.
!>

LDA

!>          LDA هو عدد صحيح
!>          البعد الرائد للمصفوفة A.  LDA >= max(1,N).
!>

B

!>          B هي مصفوفة COMPLEX*16، الأبعاد (LDB, N)
!>          عند الإدخال، المصفوفة الهرميتية الموجبة المحددة B.
!>          إذا كان UPLO = 'U'، فإن الجزء العلوي المثلثي من N*N في B
!>          يحتوي على الجزء العلوي المثلثي للمصفوفة B.
!>          إذا كان UPLO = 'L'، فإن الجزء السفلي المثلثي من N*N في B
!>          يحتوي على الجزء السفلي المثلثي للمصفوفة B.
!>
!>          عند الخروج، إذا كان INFO <= N، فإن الجزء من B الذي يحتوي على المصفوفة
!>          مُستبدل بالعامل المثلثي U أو L من تحليل Cholesky
!>          B = U**H*U أو B = L*L**H.
!>

LDB

!>          LDB هو عدد صحيح
!>          البعد الرائد للمصفوفة B.  LDB >= max(1,N).
!>

W

!>          W هي مصفوفة DOUBLE PRECISION، الأبعاد (N)
!>          إذا كان INFO = 0، فإن القيم الذاتية بترتيب تصاعدي.
!>

WORK

!>          WORK هي مصفوفة COMPLEX*16، الأبعاد (MAX(1,LWORK))
!>          عند الخروج، إذا كان INFO = 0، فإن WORK(1) يُعيد LWORK الأمثل.
!>

LWORK

!>          LWORK هو عدد صحيح
!>          طول المصفوفة WORK.  LWORK >= max(1,2*N-1).
!>          للكفاءة المثلى، LWORK >= (NB+1)*N,
!>          حيث NB هو حجم الكتلة لـ ZHETRD الذي تم إرجاعه بواسطة ILAENV.
!>
!>          إذا كان LWORK = -1، فيُفترض استعلام عن مساحة العمل؛ الروتين
!>          يحسب فقط الحجم الأمثل للمصفوفة WORK، ويعيد
!>          هذه القيمة كإدخال أول للمصفوفة WORK، ولا يتم إصدار أي رسالة خطأ
!>          متعلقة بـ LWORK بواسطة XERBLA.
!>

RWORK

!>          RWORK هي مصفوفة DOUBLE PRECISION، الأبعاد (max(1, 3*N-2))
!>

INFO

!>          INFO هو عدد صحيح
!>          = 0:  الخروج الناجح
!>          < 0:  إذا كان INFO = -i، فإن الوسيط i كان له قيمة غير قانونية
!>          > 0:  ZPOTRF أو ZHEEV أعاد رمز خطأ:
!>             <= N:  إذا كان INFO = i، فشلت ZHEEV في التقارب؛
!>                    i عناصر غير قطرية لشكل ثلاثي قطري وسيط
!>                    لم تتقارب إلى الصفر؛
!>             > N:   إذا كان INFO = N + i، لـ 1 <= i <= N، فإن
!>                    المُحدد الرئيسي الرائد من الرتبة i لـ B ليس موجبًا.
!>                    لم يكن من الممكن إكمال تحليل B
!>                    ولم يتم حساب أي قيم ذاتية أو متجهات ذاتية.
!>
