توثيق الوظيفة/الروتين الفرعي

subroutine zhetrd_he2hb (character uplo, integer n, integer kd, complex*16, dimension( lda, * ) a, integer lda, complex*16, dimension( ldab, * ) ab, integer ldab, complex*16, dimension( * ) tau, complex*16, dimension( * ) work, integer lwork, integer info)

ZHETRD_HE2HB

الغرض:

!>
!> تقوم ZHETRD_HE2HB بتقليل مصفوفة هرميتية هرميتية A إلى شكل نطاق-قطري هرميتي AB عن طريق تحويل وحدوي مشابه:
!> Q**H * A * Q = AB.
!>

المعلمات

UPLO 
!>          UPLO is CHARACTER*1
!>          = 'U':  يتم تخزين المثلث العلوي لـ A؛
!>          = 'L':  يتم تخزين المثلث السفلي لـ A.
!>

N

!>          N is INTEGER
!>          ترتيب المصفوفة A. N >= 0.
!>

KD

!>          KD is INTEGER
!>          عدد الأقطار الفائقة للمصفوفة المخفضة إذا كان UPLO = 'U'،
!>          أو عدد الأقطار الفرعية إذا كان UPLO = 'L'. KD >= 0.
!>          يتم تخزين المصفوفة المخفضة في الصف AB.
!>

A

!>          A is COMPLEX*16 array, dimension (LDA,N)
!>          عند الإدخال، مصفوفة هرميتية A. إذا كان UPLO = 'U'، فإن الجزء العلوي
!>          من A بحجم N x N يحتوي على الجزء العلوي
!>          من المصفوفة A، ولا يتم الرجوع إلى الجزء القطري السفلي الصارم لـ A.
!>          إذا كان UPLO = 'L'، فإن الجزء السفلي
!>          من A بحجم N x N يحتوي على الجزء السفلي
!>          من المصفوفة A، ولا يتم الرجوع إلى الجزء القطري العلوي الصارم لـ A.
!>          عند الخروج، إذا كان UPLO = 'U'، فإن القطر والأقطار الفائقة الأولى
!>          لـ A يتم الكتابة فوقها بالعناصر المقابلة للمصفوفة القطرية T،
!>          والعناصر فوق القطر الفائق الأول، مع الصف TAU، تمثل المصفوفة
!>          الوحدوية Q كحاصل ضرب عاكسات أولية؛ إذا كان UPLO
!>          = 'L'، يتم الكتابة فوق القطر والأقطار الفرعية الأولى لـ A
!>          بالعناصر المقابلة للمصفوفة القطرية T، والعناصر تحت القطر الفرعي الأول،
!>          مع الصف TAU، تمثل المصفوفة الوحدوية Q كحاصل ضرب
!>          عاكسات أولية. انظر التفاصيل الإضافية.
!>

LDA

!>          LDA is INTEGER
!>          البعد الرائد للصف A. LDA >= max(1,N).
!>

AB

!>          AB is COMPLEX*16 array, dimension (LDAB,N)
!>          عند الخروج، المثلث العلوي أو السفلي للمصفوفة النطاقية الهرميتية
!>          A، مخزنة في أول KD+1 صفوف من الصف.
!>          يتم تخزين العمود j من A في العمود j من الصف AB
!>          على النحو التالي:
!>          إذا كان UPLO = 'U'، AB(kd+1+i-j,j) = A(i,j) لـ max(1,j-kd)<=i<=j؛
!>          إذا كان UPLO = 'L'، AB(1+i-j,j) = A(i,j) لـ j<=i<=min(n,j+kd).
!>

LDAB

!>          LDAB is INTEGER
!>          البعد الرائد للصف AB. LDAB >= KD+1.
!>

TAU

!>          TAU is COMPLEX*16 array, dimension (N-KD)
!>          عوامل قياسية للعواكس الأولية (انظر التفاصيل الإضافية).
!>

WORK

!>          WORK is COMPLEX*16 array, dimension (MAX(1,LWORK))
!>          عند الخروج، إذا كان INFO = 0، أو إذا كان LWORK = -1،
!>          تُرجع WORK(1) حجم LWORK.
!>

LWORK

!>          LWORK is INTEGER
!>          بعد الصف WORK الذي يجب حسابه
!>          باستعلام مساحة العمل.
!>          إذا كان N <= KD+1، LWORK >= 1، وإلا LWORK = MAX(1, LWORK_QUERY).
!>
!>          إذا كان LWORK = -1، فيُفترض استعلام مساحة عمل؛ يقوم الروتين
!>          فقط بحساب الحجم الأمثل للصف WORK، ويرجع
!>          هذه القيمة كالإدخال الأول للصف WORK، ولا يتم إصدار أي رسالة خطأ
!>          متعلقة بـ LWORK بواسطة XERBLA.
!>          LWORK_QUERY = N*KD + N*max(KD,FACTOPTNB) + 2*KD*KD
!>          حيث FACTOPTNB هو الحجب المستخدم بواسطة خوارزمية QR أو LQ
!>          ، عادةً ما يكون FACTOPTNB=128 خيارًا جيدًا، وإلا
!>          فإن وضع LWORK=-1 سيوفر حجم WORK.
!>

INFO

!>          INFO is INTEGER
!>          = 0:  انتهاء ناجح
!>          < 0:  إذا كان INFO = -i، كان الوسيط i غير قانوني
!>

!>
!>  إذا كان UPLO = 'U'، يتم تمثيل المصفوفة Q كحاصل ضرب عاكسات أولية
!>
!>     Q = H(k)**H . . . H(2)**H H(1)**H, حيث k = n-kd.
!>
!>  كل H(i) له الشكل
!>
!>     H(i) = I - tau * v * v**H
!>
!>  حيث tau هو عامل قياسي معقد، و v هو متجه معقد مع
!>  v(1:i+kd-1) = 0 و v(i+kd) = 1؛ conjg(v(i+kd+1:n)) مخزن عند الخروج في
!>  A(i,i+kd+1:n)، و tau في TAU(i).
!>
!>  إذا كان UPLO = 'L'، يتم تمثيل المصفوفة Q كحاصل ضرب عاكسات أولية
!>
!>     Q = H(1) H(2) . . . H(k), حيث k = n-kd.
!>
!>  كل H(i) له الشكل
!>
!>     H(i) = I - tau * v * v**H
!>
!>  حيث tau هو عامل قياسي معقد، و v هو متجه معقد مع
!>  v(kd+1:i) = 0 و v(i+kd+1) = 1؛ v(i+kd+2:n) مخزن عند الخروج في
!>  A(i+kd+2:n,i)، و tau في TAU(i).
!>
!>  محتويات A عند الخروج موضحة بالأمثلة التالية
!>  مع n = 5:
!>
!>  إذا كان UPLO = 'U':                       إذا كان UPLO = 'L':
!>
!>    (  ab  ab/v1  v1      v1     v1    )              (  ab                            )
!>    (      ab     ab/v2   v2     v2    )              (  ab/v1  ab                     )
!>    (             ab      ab/v3  v3    )              (  v1     ab/v2  ab              )
!>    (                     ab     ab/v4 )              (  v1     v2     ab/v3  ab       )
!>    (                            ab    )              (  v1     v2     v3     ab/v4 ab )
!>
!>  حيث d و e يمثلان عناصر قطرية وغير قطرية لـ T، و vi
!>  يمثل عنصرًا من المتجه الذي يحدد H(i).
!> .fi
