توثيق الوظيفة/الروتين الفرعي

subroutine zhetrd (character uplo, integer n, complex*16, dimension( lda, * ) a, integer lda, double precision, dimension( * ) d, double precision, dimension( * ) e, complex*16, dimension( * ) tau, complex*16, dimension( * ) work, integer lwork, integer info)

ZHETRD

الغرض:

!>
!> ZHETRD تختزل المصفوفة الهرميتية المعقدة A إلى شكل ثلاثي الأقطار متناظر حقيقي T
!> بتحويل تشابه وحدوي:
!> Q**H * A * Q = T.
!>

المعلمات

UPLO 
!>          UPLO is CHARACTER*1
!>          = 'U':  يتم تخزين المثلث العلوي من A؛
!>          = 'L':  يتم تخزين المثلث السفلي من A.
!>

N

!>          N is INTEGER
!>          رتبة المصفوفة A.  N >= 0.
!>

A

!>          A is COMPLEX*16 array, dimension (LDA,N)
!>          عند الإدخال، المصفوفة الهرميتية A. إذا كان UPLO = 'U'، فإن الجزء
!>          المثلثي العلوي من A بحجم N*N يحتوي على الجزء المثلثي العلوي
!>          للمصفوفة A، والجزء السفلي تمامًا من A لا يتم الرجوع إليه. إذا كان
!>          UPLO = 'L'، فإن الجزء المثلثي السفلي من A بحجم N*N يحتوي على الجزء
!>          المثلثي السفلي للمصفوفة A، والجزء العلوي تمامًا من A لا يتم الرجوع إليه.
!>          عند الخروج، إذا كان UPLO = 'U'، فإن الأقطار والقطر العلوي الأول
!>          في A يتم استبدالها بالعناصر المقابلة للمصفوفة ثلاثية الأقطار T،
!>          والعناصر الموجودة فوق القطر العلوي الأول، مع المصفوفة TAU، تمثل
!>          المصفوفة الوحدوية Q كناتج انعكاسات أولية؛ إذا كان UPLO = 'L'،
!>          فإن الأقطار والقطر السفلي الأول في A يتم استبدالها بالعناصر
!>          المقابلة للمصفوفة ثلاثية الأقطار T، والعناصر الموجودة أسفل القطر
!>          السفلي الأول، مع المصفوفة TAU، تمثل المصفوفة الوحدوية Q كناتج
!>          انعكاسات أولية. انظر التفاصيل الإضافية.
!>

LDA

!>          LDA is INTEGER
!>          البعد الرائد للمصفوفة A.  LDA >= max(1,N).
!>

D

!>          D is DOUBLE PRECISION array, dimension (N)
!>          عناصر القطر للمصفوفة ثلاثية الأقطار T:
!>          D(i) = A(i,i).
!>

E

!>          E is DOUBLE PRECISION array, dimension (N-1)
!>          عناصر خارج القطر للمصفوفة ثلاثية الأقطار T:
!>          E(i) = A(i,i+1) إذا كان UPLO = 'U'، E(i) = A(i+1,i) إذا كان UPLO = 'L'.
!>

TAU

!>          TAU is COMPLEX*16 array, dimension (N-1)
!>          العوامل القياسية للانعكاسات الأولية (انظر التفاصيل الإضافية).
!>

WORK

!>          WORK is COMPLEX*16 array, dimension (MAX(1,LWORK))
!>          عند الخروج، إذا كان INFO = 0، فإن WORK(1) تعيد الحجم الأمثل لـ LWORK.
!>

LWORK

!>          LWORK is INTEGER
!>          بعد المصفوفة WORK.  LWORK >= 1.
!>          لأداء أمثل LWORK >= N*NB، حيث NB هو حجم الكتلة الأمثل.
!>
!>          إذا كان LWORK = -1، فسيتم افتراض استعلام عن مساحة العمل؛ الروتين
!>          يحسب فقط الحجم الأمثل لمصفوفة WORK، ويعيد هذا الحجم كإدخال أول
!>          في مصفوفة WORK، ولا يصدر رسالة خطأ متعلقة بـ LWORK بواسطة XERBLA.
!>

INFO

!>          INFO is INTEGER
!>          = 0:  خروج ناجح
!>          < 0:  إذا كان INFO = -i، فإن الوسيط i كان له قيمة غير قانونية
!>

تفاصيل إضافية:

!>
!>  إذا كان UPLO = 'U'، يتم تمثيل المصفوفة Q كناتج للانعكاسات الأولية
!>
!>     Q = H(n-1) . . . H(2) H(1).
!>
!>  كل H(i) لها الشكل
!>
!>     H(i) = I - tau * v * v**H
!>
!>  حيث tau هو عامل قياسي معقد، و v هو متجه معقد مع
!>  v(i+1:n) = 0 و v(i) = 1؛ يتم تخزين v(1:i-1) عند الخروج في
!>  A(1:i-1,i+1)، و tau في TAU(i).
!>
!>  إذا كان UPLO = 'L'، يتم تمثيل المصفوفة Q كناتج للانعكاسات الأولية
!>
!>     Q = H(1) H(2) . . . H(n-1).
!>
!>  كل H(i) لها الشكل
!>
!>     H(i) = I - tau * v * v**H
!>
!>  حيث tau هو عامل قياسي معقد، و v هو متجه معقد مع
!>  v(1:i) = 0 و v(i+1) = 1؛ يتم تخزين v(i+2:n) عند الخروج في A(i+2:n,i)،
!>  و tau في TAU(i).
!>
!>  محتويات A عند الخروج موضحة بالأمثلة التالية
!>  مع n = 5:
!>
!>  إذا كان UPLO = 'U':                       إذا كان UPLO = 'L':
!>
!>    (  d   e   v2  v3  v4 )              (  d                  )
!>    (      d   e   v3  v4 )              (  e   d              )
!>    (          d   e   v4 )              (  v1  e   d          )
!>    (              d   e  )              (  v1  v2  e   d      )
!>    (                  d  )              (  v1  v2  v3  e   d  )
!>
!>  حيث d و e تمثل عناصر القطر والعناصر خارج القطر لـ T، و vi
!>  تمثل عنصراً من المتجه الذي يحدد H(i).
!>
