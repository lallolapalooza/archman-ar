subroutine zhbgvx (character jobz, character range, character uplo, integer n, integer ka, integer kb, complex*16, dimension( ldab, * ) ab, integer ldab, complex*16, dimension( ldbb, * ) bb, integer ldbb, complex*16, dimension( ldq, * ) q, integer ldq, double precision vl, double precision vu, integer il, integer iu, double precision abstol, integer m, double precision, dimension( * ) w, complex*16, dimension( ldz, * ) z, integer ldz, complex*16, dimension( * ) work, double precision, dimension( * ) rwork, integer, dimension( * ) iwork, integer, dimension( * ) ifail, integer info)

ZHBGVX

الغرض:

!>
!> ZHBGVX يحسب جميع القيم الذاتية، وبالاختيار، المتجهات الذاتية
!> لمشكلة القيم الذاتية الهرميتية المعممة المعرفة بشكل خاص،
!> من الشكل A*x=(lambda)*B*x. هنا A و B مفترضان أنهما هرميتيان
!> ونطاقيان، و B موجب محدد أيضًا. يمكن تحديد القيم الذاتية و
!> المتجهات الذاتية عن طريق تحديد إما جميع القيم الذاتية،
!> أو نطاق من القيم، أو نطاق من المؤشرات للقيم الذاتية المطلوبة.
!>

المعاملات

JOBZ 
!>          JOBZ هو CHARACTER*1
!>          = 'N':  حساب القيم الذاتية فقط؛
!>          = 'V':  حساب القيم الذاتية والمتجهات الذاتية.
!>

RANGE

!>          RANGE هو CHARACTER*1
!>          = 'A': سيتم العثور على جميع القيم الذاتية؛
!>          = 'V': سيتم العثور على جميع القيم الذاتية في الفترة نصف المفتوحة (VL,VU]
!>                 ؛
!>          = 'I': سيتم العثور على القيم الذاتية من IL-th إلى IU-th.
!>

UPLO

!>          UPLO هو CHARACTER*1
!>          = 'U':  يتم تخزين المثلثات العلوية للمصفوفتين A و B؛
!>          = 'L':  يتم تخزين المثلثات السفلية للمصفوفتين A و B.
!>

N

!>          N هو INTEGER
!>          رتبة المصفوفتين A و B. N >= 0.
!>

KA

!>          KA هو INTEGER
!>          عدد الأقطار الفائقة للمصفوفة A إذا كان UPLO = 'U'،
!>          أو عدد الأقطار الفرعية إذا كان UPLO = 'L'. KA >= 0.
!>

KB

!>          KB هو INTEGER
!>          عدد الأقطار الفائقة للمصفوفة B إذا كان UPLO = 'U'،
!>          أو عدد الأقطار الفرعية إذا كان UPLO = 'L'. KB >= 0.
!>

AB

!>          AB هي مصفوفة COMPLEX*16، أبعادها (LDAB, N)
!>          عند الإدخال، المثلث العلوي أو السفلي للمصفوفة النطاقية الهرميتية
!>          A، مخزنة في الصفوف ka+1 الأولى من المصفوفة. يتم
!>          تخزين العمود j من A في العمود j من المصفوفة AB
!>          كما يلي:
!>          إذا كان UPLO = 'U'، فإن AB(ka+1+i-j,j) = A(i,j) لـ max(1,j-ka)<=i<=j؛
!>          إذا كان UPLO = 'L'، فإن AB(1+i-j,j) = A(i,j) لـ j<=i<=min(n,j+ka).
!>
!>          عند الخروج، يتم تدمير محتويات AB.
!>

LDAB

!>          LDAB هو INTEGER
!>          البعد الرائد للمصفوفة AB. LDAB >= KA+1.
!>

BB

!>          BB هي مصفوفة COMPLEX*16، أبعادها (LDBB, N)
!>          عند الإدخال، المثلث العلوي أو السفلي للمصفوفة النطاقية الهرميتية
!>          B، مخزنة في الصفوف kb+1 الأولى من المصفوفة. يتم
!>          تخزين العمود j من B في العمود j من المصفوفة BB
!>          كما يلي:
!>          إذا كان UPLO = 'U'، فإن BB(kb+1+i-j,j) = B(i,j) لـ max(1,j-kb)<=i<=j؛
!>          إذا كان UPLO = 'L'، فإن BB(1+i-j,j) = B(i,j) لـ j<=i<=min(n,j+kb).
!>
!>          عند الخروج، يتم إرجاع العامل S من تحليل Cholesky المنفصل
!>          B = S**H*S، كما يتم إرجاعه بواسطة ZPBSTF.
!>

LDBB

!>          LDBB هو INTEGER
!>          البعد الرائد للمصفوفة BB. LDBB >= KB+1.
!>

Q

!>          Q هي مصفوفة COMPLEX*16، أبعادها (LDQ, N)
!>          إذا كان JOBZ = 'V'، فإن المصفوفة Q هي المصفوفة n*n المستخدمة في اختزال
!>          A*x = (lambda)*B*x إلى الشكل القياسي، أي C*x = (lambda)*x،
!>          وبالتالي اختزال C إلى الشكل ثلاثي الأقطار.
!>          إذا كان JOBZ = 'N'، فإن المصفوفة Q غير مرجعية.
!>

LDQ

!>          LDQ هو INTEGER
!>          البعد الرائد للمصفوفة Q. إذا كان JOBZ = 'N'،
!>          فإن LDQ >= 1. إذا كان JOBZ = 'V'، فإن LDQ >= max(1,N).
!>

VL

!>          VL هو DOUBLE PRECISION
!>
!>          إذا كان RANGE='V'، فإن الحد الأدنى للفترة
!>          التي سيتم البحث فيها عن القيم الذاتية. VL < VU.
!>          غير مرجعي إذا كان RANGE = 'A' أو 'I'.
!>

VU

!>          VU هو DOUBLE PRECISION
!>
!>          إذا كان RANGE='V'، فإن الحد الأعلى للفترة
!>          التي سيتم البحث فيها عن القيم الذاتية. VL < VU.
!>          غير مرجعي إذا كان RANGE = 'A' أو 'I'.
!>

IL

!>          IL هو INTEGER
!>
!>          إذا كان RANGE='I'، فإن مؤشر
!>          أصغر قيمة ذاتية سيتم إرجاعها.
!>          1 <= IL <= IU <= N، إذا كان N > 0؛ IL = 1 و IU = 0 إذا كان N = 0.
!>          غير مرجعي إذا كان RANGE = 'A' أو 'V'.
!>

IU

!>          IU هو INTEGER
!>
!>          إذا كان RANGE='I'، فإن مؤشر
!>          أكبر قيمة ذاتية سيتم إرجاعها.
!>          1 <= IL <= IU <= N، إذا كان N > 0؛ IL = 1 و IU = 0 إذا كان N = 0.
!>          غير مرجعي إذا كان RANGE = 'A' أو 'V'.
!>

ABSTOL

!>          ABSTOL هو DOUBLE PRECISION
!>          التفاوت المطلق في الخطأ للقيم الذاتية.
!>          يتم قبول قيمة ذاتية تقريبية على أنها متقاربة
!>          عندما يتم تحديد أنها تقع في فترة [a,b]
!>          بعرض أقل من أو يساوي
!>
!>                  ABSTOL + EPS *   max( |a|,|b| ) ,
!>
!>          حيث EPS هي دقة الآلة. إذا كانت ABSTOL أقل من
!>          أو تساوي الصفر، فسيتم استخدام EPS*|T| بدلاً منها،
!>          حيث |T| هو المعيار 1 للمصفوفة ثلاثية الأقطار التي تم الحصول عليها
!>          عن طريق اختزال AP إلى شكل ثلاثي الأقطار.
!>
!>          سيتم حساب القيم الذاتية بأكبر قدر من الدقة عندما تكون ABSTOL
!>          مضبوطة على ضعف عتبة تحت التدفق 2*DLAMCH('S')، وليس صفرًا.
!>          إذا عادت هذه الروتين الفرعي بـ INFO>0، مما يشير إلى أن بعض
!>          المتجهات الذاتية لم تتقارب، حاول ضبط ABSTOL على
!>          2*DLAMCH('S').
!>

M

!>          M هو INTEGER
!>          إجمالي عدد القيم الذاتية التي تم العثور عليها. 0 <= M <= N.
!>          إذا كان RANGE = 'A'، فإن M = N، وإذا كان RANGE = 'I'، فإن M = IU-IL+1.
!>

W

!>          W هي مصفوفة DOUBLE PRECISION، أبعادها (N)
!>          إذا كان INFO = 0، فإن القيم الذاتية بترتيب تصاعدي.
!>

Z

!>          Z هي مصفوفة COMPLEX*16، أبعادها (LDZ, N)
!>          إذا كان JOBZ = 'V'، فعندما يكون INFO = 0، فإن Z تحتوي على المصفوفة Z للمتجهات
!>          الذاتية، حيث يمثل العمود i من Z المتجه الذاتي
!>          المرتبط بـ W(i). يتم تطبيع المتجهات الذاتية بحيث Z**H*B*Z = I.
!>          إذا كان JOBZ = 'N'، فإن Z غير مرجعي.
!>

LDZ

!>          LDZ هو INTEGER
!>          البعد الرائد للمصفوفة Z. LDZ >= 1، وإذا كان
!>          JOBZ = 'V'، فإن LDZ >= N.
!>

WORK
