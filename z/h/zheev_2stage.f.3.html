توثيق الوظيفة/الروتين الفرعي

subroutine zheev_2stage (character jobz, character uplo, integer n, complex*16, dimension( lda, * ) a, integer lda, double precision, dimension( * ) w, complex*16, dimension( * ) work, integer lwork, double precision, dimension( * ) rwork, integer info)

ZHEEV_2STAGE يحسب القيم الذاتية، اختياريًا، المتجهات الذاتية اليسرى و/أو اليمنى لمصفوفات HE

الغرض:

!>
!> ZHEEV_2STAGE يحسب جميع القيم الذاتية، واختياريًا، المتجهات الذاتية لمصفوفة متوافقة متخيلة A
!> باستخدام تقنية المرحلة 2 (2stage) للتقليل إلى مصفوفة ثلاثية الأقطار.
!>

المعلمات

JOBZ 
!>          JOBZ هو CHARACTER*1
!>          = 'N':  حساب القيم الذاتية فقط؛
!>          = 'V':  حساب القيم الذاتية والمتجهات الذاتية.
!>                  غير متوفر في هذا الإصدار.
!>

UPLO

!>          UPLO هو CHARACTER*1
!>          = 'U':  يتم تخزين المثلث العلوي من A؛
!>          = 'L':  يتم تخزين المثلث السفلي من A.
!>

N

!>          N هو INTEGER
!>          رتبة المصفوفة A.  N >= 0.
!>

A

!>          A هو COMPLEX*16 array, dimension (LDA, N)
!>          عند الدخول، المصفوفة المتوافقة المتخيلة A.  إذا كان UPLO = 'U'، فإن
!>          الجزء العلوي المثلثي من A بحجم N*N يحتوي على
!>          الجزء العلوي المثلثي من المصفوفة A.  إذا كان UPLO = 'L'، فإن
!>          الجزء السفلي المثلثي من A بحجم N*N يحتوي على
!>          الجزء السفلي المثلثي من المصفوفة A.
!>          عند الخروج، إذا كان JOBZ = 'V'، ثم إذا كان INFO = 0، تحتوي A على
!>          المتجهات الذاتية المتعامدة للمصفوفة A.
!>          إذا كان JOBZ = 'N'، فإن عند الخروج يتم تدمير المثلث السفلي (إذا كان UPLO='L')
!>          أو المثلث العلوي (إذا كان UPLO='U') من A، بما في ذلك
!>          القطر.
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرائد للمصفوفة A.  LDA >= max(1,N).
!>

W

!>          W هو DOUBLE PRECISION array, dimension (N)
!>          إذا كان INFO = 0، فإن القيم الذاتية مرتبة تصاعديًا.
!>

WORK

!>          WORK هو COMPLEX*16 array, dimension (MAX(1,LWORK))
!>          عند الخروج، إذا كان INFO = 0، فإن WORK(1) يعيد LWORK الأمثل.
!>

LWORK

!>          LWORK هو INTEGER
!>          طول المصفوفة WORK. LWORK >= 1، عندما N <= 1؛
!>          خلاف ذلك
!>          إذا كان JOBZ = 'N' و N > 1، يجب الاستعلام عن LWORK.
!>                                   LWORK = MAX(1, dimension) حيث
!>                                   dimension = max(stage1,stage2) + (KD+1)*N + N
!>                                             = N*KD + N*max(KD+1,FACTOPTNB)
!>                                               + max(2*KD*KD, KD*NTHREADS)
!>                                               + (KD+1)*N + N
!>                                   حيث KD هو حجم التجميع للتقليل،
!>                                   FACTOPTNB هو التجميع المستخدم بواسطة خوارزمية QR أو LQ
!>                                   عادةً ما يكون FACTOPTNB=128 اختيارًا جيدًا
!>                                   NTHREADS هو عدد الخيوط المستخدمة عندما
!>                                   تم تمكين تجميع openMP، وإلا =1.
!>          إذا كان JOBZ = 'V' و N > 1، يجب الاستعلام عن LWORK. غير متوفر بعد
!>
!>          إذا كان LWORK = -1، فيفترض استعلام عن مساحة العمل؛ الروتين
!>          يحسب فقط الحجم الأمثل للمصفوفة WORK، ويعيد
!>          هذه القيمة كإدخال أول للمصفوفة WORK، ولا يصدر أي رسالة خطأ
!>          متعلقة بـ LWORK بواسطة XERBLA.
!>

RWORK

!>          RWORK هو DOUBLE PRECISION array, dimension (max(1, 3*N-2))
!>

INFO

!>          INFO هو INTEGER
!>          = 0:  خروج ناجح
!>          < 0:  إذا كان INFO = -i، فإن الوسيط i لديه قيمة غير قانونية
!>          > 0:  إذا كان INFO = i، فشلت الخوارزمية في التقارب؛ i
!>                العناصر خارج القطر لمصفوفة ثلاثية الأقطار وسيطة
!>                لم تتقارب إلى الصفر.
!>
