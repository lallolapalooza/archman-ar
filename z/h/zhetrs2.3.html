subroutine zhetrs2 (character uplo, integer n, integer nrhs, complex*16, dimension( lda, * ) a, integer lda, integer, dimension( * ) ipiv, complex*16, dimension( ldb, * ) b, integer ldb, complex*16, dimension( * ) work, integer info)

ZHETRS2

الغرض:

!>
!> ZHETRS2 يحل نظام المعادلات الخطية A*X = B مع مصفوفة هرميتية معقدة A باستخدام
!> تحليل A = U*D*U**H أو A = L*D*L**H الذي تم حسابه بواسطة ZHETRF وتم تحويله بواسطة ZSYCONV.
!>

المعلمات

UPLO 
!>          UPLO هو CHARACTER*1
!>          يحدد ما إذا كانت تفاصيل التحليل مخزنة
!>          كمصفوفة مثلثة علوية أو سفلية.
!>          = 'U':  مثلثة علوية، الشكل هو A = U*D*U**H;
!>          = 'L':  مثلثة سفلية، الشكل هو A = L*D*L**H.
!>

N

!>          N هو INTEGER
!>          ترتيب المصفوفة A. N >= 0.
!>

NRHS

!>          NRHS هو INTEGER
!>          عدد الجوانب اليمنى، أي عدد الأعمدة
!>          من المصفوفة B. NRHS >= 0.
!>

A

!>          A هي مصفوفة COMPLEX*16، البعد (LDA,N)
!>          المصفوفة القطرية المقطعية D والمضاعفات المستخدمة
!>          للحصول على العامل U أو L كما تم حسابه بواسطة ZHETRF.
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرائد للمصفوفة A. LDA >= max(1,N).
!>

IPIV

!>          IPIV هي مصفوفة INTEGER، البعد (N)
!>          تفاصيل التبادلات والبنية المقطعية لـ D
!>          كما تم تحديده بواسطة ZHETRF.
!>

B

!>          B هي مصفوفة COMPLEX*16، البعد (LDB,NRHS)
!>          عند الإدخال، مصفوفة الجانب الأيمن B.
!>          عند الخروج، مصفوفة الحل X.
!>

LDB

!>          LDB هو INTEGER
!>          البعد الرائد للمصفوفة B. LDB >= max(1,N).
!>

WORK

!>          WORK هي مصفوفة COMPLEX*16، البعد (N)
!>

INFO

!>          INFO هو INTEGER
!>          = 0:  انتهاء ناجح
!>          < 0:  إذا كان INFO = -i، فإن المعلمة i كانت ذات قيمة غير قانونية
!>
