subroutine zhet01_3 (character uplo, integer n, complex*16, dimension( lda, * ) a, integer lda, complex*16, dimension( ldafac, * ) afac, integer ldafac, complex*16, dimension( * ) e, integer, dimension( * ) ipiv, complex*16, dimension( ldc, * ) c, integer ldc, double precision, dimension( * ) rwork, double precision resid)

ZHET01_3

الغرض:

!>
!> ZHET01_3 يعيد بناء مصفوفة هيرميتية غير محددة A من مفكوكها
!> L*D*L' أو U*D*U' الذي تم حسابه بواسطة ZHETRF_RK
!> (أو ZHETRF_BK) ويحسب الباقي
!>    norm( C - A ) / ( N * norm(A) * EPS )،
!> حيث C هي المصفوفة المعاد بناؤها و EPS هو معامل التفاضل العددي الدقيق.
!>

المعاملات

UPLO 
!>          UPLO هو CHARACTER*1
!>          يحدد ما إذا كان الجزء العلوي أو السفلي المثلثي للمصفوفة
!>          الهيرميتية A مخزنًا:
!>          = 'U':  مثلث علوي
!>          = 'L':  مثلث سفلي
!>

N

!>          N هو INTEGER
!>          عدد الصفوف والأعمدة للمصفوفة A.  N >= 0.
!>

A

!>          A هو مصفوفة COMPLEX*16، ذات البعد (LDA,N)
!>          المصفوفة الهيرميتية الأصلية A.
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرائد للمصفوفة A.  LDA >= max(1,N)
!>

AFAC

!>          AFAC هو مصفوفة COMPLEX*16، ذات البعد (LDAFAC,N)
!>          قطر المصفوفة القطرية الكتلية D والعوامل U أو L
!>          كما تم حسابها بواسطة ZHETRF_RK و ZHETRF_BK:
!>            أ) فقط العناصر القطرية للمصفوفة القطرية الكتلية الهيرميتية
!>               D على قطر A، أي D(k,k) = A(k,k)؛
!>               (يجب توفير عناصر فوق القطر (أو تحت القطر) لـ D
!>                عند الإدخال في المصفوفة E)، و
!>            ب) إذا كان UPLO = 'U': العامل U في الجزء العلوي من القطر في A.
!>               إذا كان UPLO = 'L': العامل L في الجزء السفلي من القطر في A.
!>

LDAFAC

!>          LDAFAC هو INTEGER
!>          البعد الرائد للمصفوفة AFAC.
!>          LDAFAC >= max(1,N).
!>

E

!>          E هو مصفوفة COMPLEX*16، ذات البعد (N)
!>          عند الإدخال، تحتوي على العناصر فوق القطر (أو تحت القطر)
!>          للمصفوفة القطرية الكتلية الهيرميتية D
!>          مع كتل قطرية بحجم 1x1 أو 2x2، حيث
!>          إذا كان UPLO = 'U': E(i) = D(i-1,i),i=2:N, E(1) غير مشار إليها؛
!>          إذا كان UPLO = 'L': E(i) = D(i+1,i),i=1:N-1, E(N) غير مشار إليها.
!>

IPIV

!>          IPIV هو مصفوفة INTEGER، ذات البعد (N)
!>          مؤشرات المحور من ZHETRF_RK (أو ZHETRF_BK).
!>

C

!>          C هو مصفوفة COMPLEX*16، ذات البعد (LDC,N)
!>

LDC

!>          LDC هو INTEGER
!>          البعد الرائد للمصفوفة C.  LDC >= max(1,N).
!>

RWORK

!>          RWORK هو مصفوفة DOUBLE PRECISION، ذات البعد (N)
!>

RESID

!>          RESID هو DOUBLE PRECISION
!>          إذا كان UPLO = 'L'، norm(L*D*L' - A) / ( N * norm(A) * EPS )
!>          إذا كان UPLO = 'U'، norm(U*D*U' - A) / ( N * norm(A) * EPS )
!>
