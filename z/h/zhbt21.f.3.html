subroutine zhbt21 (character uplo, integer n, integer ka, integer ks, complex*16, dimension( lda, * ) a, integer lda, double precision, dimension( * ) d, double precision, dimension( * ) e, complex*16, dimension( ldu, * ) u, integer ldu, complex*16, dimension( * ) work, double precision, dimension( * ) rwork, double precision, dimension( 2 ) result)

ZHBT21

الغرض:

!>
!> ZHBT21  بشكل عام تتحقق من تحليل بالشكل
!>
!>         A = U S U**H
!>
!> حيث **H تعني منقول مترافق، A مصفوفة هرميتية ذات نطاق، U مصفوفة موحدة،
!> و S مصفوفة قطرية (إذا كان KS=0) أو مصفوفة شبه قطرية متماثلة (إذا كان KS=1).
!>
!> على وجه التحديد:
!>
!>         RESULT(1) = | A - U S U**H | / ( |A| n ulp ) و
!>         RESULT(2) = | I - U U**H | / ( n ulp )
!>

المعلمات

UPLO 
!>          UPLO هي CHARACTER
!>          إذا كان UPLO='U'، فسيتم استخدام المثلث العلوي من A و V ولن تتم
!>          الإشارة إلى المثلث السفلي (الصارم).
!>          إذا كان UPLO='L'، فسيتم استخدام المثلث السفلي من A و V ولن تتم
!>          الإشارة إلى المثلث العلوي (الصارم).
!>

N

!>          N هو INTEGER
!>          حجم المصفوفة. إذا كان صفرًا، فإن ZHBT21 لا تفعل شيئًا.
!>          يجب أن يكون على الأقل صفرًا.
!>

KA

!>          KA هو INTEGER
!>          عرض النطاق للمصفوفة A. يجب أن يكون على الأقل صفرًا. إذا
!>          كان أكبر من N-1، فسيتم استخدام max( 0, N-1 ).
!>

KS

!>          KS هو INTEGER
!>          عرض النطاق للمصفوفة S. يمكن أن يكون صفرًا أو واحدًا فقط.
!>          إذا كان صفرًا، فإن S مصفوفة قطرية، ولن يتم الإشارة إلى E. إذا
!>          كان واحدًا، فإن S مصفوفة شبه قطرية متماثلة.
!>

A

!>          A هي مصفوفة COMPLEX*16، البعد (LDA, N)
!>          المصفوفة الأصلية (غير المحللة). يُفترض أنها
!>          هرميتية، وسيتم الإشارة فقط إلى الجزء العلوي (UPLO='U') أو الجزء السفلي
!>          (UPLO='L') فقط.
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرئيسي لـ A. يجب أن يكون على الأقل 1
!>          وعلى الأقل min( KA, N-1 ).
!>

D

!>          D هي مصفوفة DOUBLE PRECISION، البعد (N)
!>          القطر الرئيسي للمصفوفة (شبه القطرية المتماثلة) S.
!>

E

!>          E هي مصفوفة DOUBLE PRECISION، البعد (N-1)
!>          العناصر خارج القطر الرئيسي للمصفوفة (شبه القطرية المتماثلة) S.
!>          E(1) هو العنصر (1,2) و (2,1)، E(2) هو العنصر (2,3) و
!>          (3,2)، وهكذا.
!>          لا يتم الإشارة إليه إذا كان KS=0.
!>

U

!>          U هي مصفوفة COMPLEX*16، البعد (LDU, N)
!>          المصفوفة الموحدة في التحليل، معبرًا عنها كمصفوفة كثيفة (أي،
!>          ليست كناتج تحويلات هاوسهولدر، تحويلات جيفنز، إلخ.)
!>

LDU

!>          LDU هو INTEGER
!>          البعد الرئيسي لـ U. يجب أن يكون LDU على الأقل N
!>          وعلى الأقل 1.
!>

WORK

!>          WORK هي مصفوفة COMPLEX*16، البعد (N**2)
!>

RWORK

!>          RWORK هي مصفوفة DOUBLE PRECISION، البعد (N)
!>

RESULT

!>          RESULT هي مصفوفة DOUBLE PRECISION، البعد (2)
!>          القيم التي تم حسابها بواسطة الاختبارين الموصوفين أعلاه.
!>          القيم محدودة حاليًا بـ 1/ulp، لتجنب تجاوز السعة.
!>
