subroutine zher2 (character uplo, integer n, complex*16 alpha, complex*16, dimension(*) x, integer incx, complex*16, dimension(*) y, integer incy, complex*16, dimension(lda,*) a, integer lda)

ZHER2

الغرض:

!>
!> ZHER2  يقوم بعملية هرميتية من الرتبة 2
!>
!>    A := alpha*x*y**H + conjg( alpha )*y*x**H + A,
!>
!> حيث alpha هو عدد قياسي، x و y هما متجهان يحتويان على n عنصر، و A هي مصفوفة هرميتية بحجم n
!> في n.
!>

المعاملات

UPLO 
!>          UPLO هو CHARACTER*1
!>           عند الإدخال، يحدد UPLO ما إذا كان الجزء العلوي أو السفلي
!>           المثلثي من المصفوفة A سيتم الرجوع إليه على النحو التالي:
!>
!>              UPLO = 'U' أو 'u'   سيتم الرجوع فقط إلى الجزء العلوي المثلثي من A
!>                                  .
!>
!>              UPLO = 'L' أو 'l'   سيتم الرجوع فقط إلى الجزء السفلي المثلثي من A
!>                                  .
!>

N

!>          N هو INTEGER
!>           عند الإدخال، يحدد N ترتيب المصفوفة A.
!>           يجب أن تكون N على الأقل صفر.
!>

ALPHA

!>          ALPHA هو COMPLEX*16
!>           عند الإدخال، يحدد ALPHA العدد القياسي alpha.
!>

X

!>          X هي مصفوفة COMPLEX*16، حجمها على الأقل
!>           ( 1 + ( n - 1 )*abs( INCX ) ).
!>           قبل الإدخال، يجب أن تحتوي المصفوفة X المضاعفة على المتجه x المكون من n عنصر.
!>

INCX

!>          INCX هو INTEGER
!>           عند الإدخال، يحدد INCX الزيادة لعناصر X.
!>           يجب ألا يكون INCX صفرًا.
!>

Y

!>          Y هي مصفوفة COMPLEX*16، حجمها على الأقل
!>           ( 1 + ( n - 1 )*abs( INCY ) ).
!>           قبل الإدخال، يجب أن تحتوي المصفوفة Y المضاعفة على المتجه y المكون من n عنصر.
!>

INCY

!>          INCY هو INTEGER
!>           عند الإدخال، يحدد INCY الزيادة لعناصر Y.
!>           يجب ألا يكون INCY صفرًا.
!>

A

!>          A هي مصفوفة COMPLEX*16، حجمها ( LDA, N )
!>           عند الإدخال مع UPLO = 'U' أو 'u'، يجب أن يحتوي الجزء العلوي الرئيسي بحجم n × n
!>           من المصفوفة A على الجزء العلوي المثلثي من المصفوفة الهرميتية والجزء السفلي
!>           المثلثي الصارم من A لا تتم الإشارة إليه. عند الخروج،
!>           يتم الكتابة فوق الجزء العلوي المثلثي من المصفوفة A بالجزء العلوي
!>           المثلثي للمصفوفة المحدثة.
!>           عند الإدخال مع UPLO = 'L' أو 'l'، يجب أن يحتوي الجزء السفلي الرئيسي بحجم n × n
!>           من المصفوفة A على الجزء السفلي المثلثي من المصفوفة الهرميتية والجزء العلوي
!>           المثلثي الصارم من A لا تتم الإشارة إليه. عند الخروج،
!>           يتم الكتابة فوق الجزء السفلي المثلثي من المصفوفة A بالجزء السفلي
!>           المثلثي للمصفوفة المحدثة.
!>           لاحظ أنه لا يلزم تعيين الأجزاء التخيلية للعناصر القطرية،
!>           يفترض أنها صفر، وعند الخروج يتم تعيينها إلى صفر.
!>

LDA

!>          LDA هو INTEGER
!>           عند الإدخال، يحدد LDA البعد الأول للمصفوفة A كما تم تعريفه
!>           في البرنامج (الفرعي) المستدعي. يجب أن يكون LDA على الأقل
!>           max( 1, n ).
!>
