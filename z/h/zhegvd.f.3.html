subroutine zhegvd (integer itype, character jobz, character uplo, integer n, complex*16, dimension( lda, * ) a, integer lda, complex*16, dimension( ldb, * ) b, integer ldb, double precision, dimension( * ) w, complex*16, dimension( * ) work, integer lwork, double precision, dimension( * ) rwork, integer lrwork, integer, dimension( * ) iwork, integer liwork, integer info)

ZHEGVD

الغرض:

!>
!> ZHEGVD يحسب جميع القيم الذاتية، وبشكل اختياري، المتجهات الذاتية
!> لمشكلة القيم الذاتية الهرمية المعممة المحددة، من الشكل
!> A*x=(lambda)*B*x,  A*Bx=(lambda)*x,  أو B*A*x=(lambda)*x. هنا A و
!> B مفترض أن تكونا هرميتين و B موجبة محددة أيضًا.
!> إذا تم طلب المتجهات الذاتية، فإنها تستخدم خوارزمية قسمة وفتح.
!>
!>

المعلمات

ITYPE 
!>          ITYPE هو عدد صحيح
!>          يحدد نوع المشكلة المراد حلها:
!>          = 1:  A*x = (lambda)*B*x
!>          = 2:  A*B*x = (lambda)*x
!>          = 3:  B*A*x = (lambda)*x
!>

JOBZ

!>          JOBZ هو حرف واحد
!>          = 'N':  حساب القيم الذاتية فقط؛
!>          = 'V':  حساب القيم الذاتية والمتجهات الذاتية.
!>

UPLO

!>          UPLO هو حرف واحد
!>          = 'U':  المثلثات العلوية لـ A و B مخزنة؛
!>          = 'L':  المثلثات السفلية لـ A و B مخزنة.
!>

N

!>          N هو عدد صحيح
!>          رتبة المصفوفتين A و B.  N >= 0.
!>

A

!>          A هي مصفوفة COMPLEX*16، ذات البعد (LDA, N)
!>          عند الدخول، المصفوفة الهرمية A.  إذا كان UPLO = 'U'، فإن
!>          الجزء العلوي المثلثي الرئيسي N-by-N من A يحتوي على
!>          الجزء العلوي المثلثي للمصفوفة A.  إذا كان UPLO = 'L'، فإن
!>          الجزء السفلي المثلثي الرئيسي N-by-N من A يحتوي على
!>          الجزء السفلي المثلثي للمصفوفة A.
!>
!>          عند الخروج، إذا كان JOBZ = 'V'، وإذا كان INFO = 0، فإن A تحتوي على
!>          المصفوفة Z للمتجهات الذاتية.  يتم تطبيع المتجهات الذاتية
!>          كالتالي:
!>          إذا كان ITYPE = 1 أو 2، Z**H*B*Z = I؛
!>          إذا كان ITYPE = 3، Z**H*inv(B)*Z = I.
!>          إذا كان JOBZ = 'N'، فعند الخروج، المثلث العلوي (إذا كان UPLO='U')
!>          أو المثلث السفلي (إذا كان UPLO='L') من A، بما في ذلك
!>          القطر، يتم تدميره.
!>

LDA

!>          LDA هو عدد صحيح
!>          البعد الرئيسي للمصفوفة A.  LDA >= max(1,N).
!>

B

!>          B هي مصفوفة COMPLEX*16، ذات البعد (LDB, N)
!>          عند الدخول، المصفوفة الهرمية B.  إذا كان UPLO = 'U'، فإن
!>          الجزء العلوي المثلثي الرئيسي N-by-N من B يحتوي على
!>          الجزء العلوي المثلثي للمصفوفة B.  إذا كان UPLO = 'L'، فإن
!>          الجزء السفلي المثلثي الرئيسي N-by-N من B يحتوي على
!>          الجزء السفلي المثلثي للمصفوفة B.
!>
!>          عند الخروج، إذا كان INFO <= N، فإن الجزء من B الذي يحتوي على المصفوفة
!>          يتم استبداله بالعامل المثلثي U أو L من تحلل Cholesky
!>          B = U**H*U أو B = L*L**H.
!>

LDB

!>          LDB هو عدد صحيح
!>          البعد الرئيسي للمصفوفة B.  LDB >= max(1,N).
!>

W

!>          W هي مصفوفة DOUBLE PRECISION، ذات البعد (N)
!>          إذا كان INFO = 0، فإن القيم الذاتية مرتبة تصاعديًا.
!>

WORK

!>          WORK هي مصفوفة COMPLEX*16، ذات البعد (MAX(1,LWORK))
!>          عند الخروج، إذا كان INFO = 0، فإن WORK(1) يعيد LWORK المثلى.
!>

LWORK

!>          LWORK هو عدد صحيح
!>          طول المصفوفة WORK.
!>          إذا كان N <= 1،                LWORK >= 1.
!>          إذا كان JOBZ  = 'N' و N > 1، LWORK >= N + 1.
!>          إذا كان JOBZ  = 'V' و N > 1، LWORK >= 2*N + N**2.
!>
!>          إذا كان LWORK = -1، فيفترض استعلام عن مساحة العمل؛ الروتين
!>          يحسب فقط الأحجام المثلى لمصفوفات WORK و RWORK و
!>          IWORK، ويعيد هذه القيم كإدخالات أولى لـ
!>          مصفوفات WORK و RWORK و IWORK، ولا يصدر XERBLA أي رسالة خطأ
!>          متعلقة بـ LWORK أو LRWORK أو LIWORK.
!>

RWORK

!>          RWORK هي مصفوفة DOUBLE PRECISION، ذات البعد (MAX(1,LRWORK))
!>          عند الخروج، إذا كان INFO = 0، فإن RWORK(1) يعيد LRWORK المثلى.
!>

LRWORK

!>          LRWORK هو عدد صحيح
!>          بعد المصفوفة RWORK.
!>          إذا كان N <= 1،                LRWORK >= 1.
!>          إذا كان JOBZ  = 'N' و N > 1، LRWORK >= N.
!>          إذا كان JOBZ  = 'V' و N > 1، LRWORK >= 1 + 5*N + 2*N**2.
!>
!>          إذا كان LRWORK = -1، فيفترض استعلام عن مساحة العمل؛ الروتين
!>          يحسب فقط الأحجام المثلى لمصفوفات WORK و RWORK و
!>          IWORK، ويعيد هذه القيم كإدخالات أولى لـ
!>          مصفوفات WORK و RWORK و IWORK، ولا يصدر XERBLA أي رسالة خطأ
!>          متعلقة بـ LWORK أو LRWORK أو LIWORK.
!>

IWORK

!>          IWORK هي مصفوفة INTEGER، ذات البعد (MAX(1,LIWORK))
!>          عند الخروج، إذا كان INFO = 0، فإن IWORK(1) يعيد LIWORK المثلى.
!>

LIWORK

!>          LIWORK هو عدد صحيح
!>          بعد المصفوفة IWORK.
!>          إذا كان N <= 1،                LIWORK >= 1.
!>          إذا كان JOBZ  = 'N' و N > 1، LIWORK >= 1.
!>          إذا كان JOBZ  = 'V' و N > 1، LIWORK >= 3 + 5*N.
!>
!>          إذا كان LIWORK = -1، فيفترض استعلام عن مساحة العمل؛ الروتين
!>          يحسب فقط الأحجام المثلى لمصفوفات WORK و RWORK و
!>          IWORK، ويعيد هذه القيم كإدخالات أولى لـ
!>          مصفوفات WORK و RWORK و IWORK، ولا يصدر XERBLA أي رسالة خطأ
!>          متعلقة بـ LWORK أو LRWORK أو LIWORK.
!>

INFO

!>          INFO هو عدد صحيح
!>          = 0:  انتهاء ناجح
!>          < 0:  إذا كان INFO = -i، فإن الوسيط i كان له قيمة غير قانونية
!>          > 0:  ZPOTRF أو ZHEEVD أرجعت رمز خطأ:
!>             <= N:  إذا كان INFO = i و JOBZ = 'N'، فإن الخوارزمية
!>                    فشلت في التقارب؛ i عناصر خارج القطر لمصفوفة
!>                    وسطية ثلاثية لم تتقارب إلى
!>                    صفر؛
!>                    إذا كان INFO = i و JOBZ = 'V'، فإن الخوارزمية
!>                    فشلت في حساب قيمة ذاتية أثناء العمل على
!>                    المصفوفة الفرعية التي تقع في الصفوف والأعمدة INFO/(N+1)
!>                    حتى mod(INFO,N+1)؛
!>             > N:   إذا كان INFO = N + i، لـ 1 <= i <= N، فإن
!>                    الرئيسية الجزئية الرئيسية من الرتبة i للمصفوفة B ليست موجبة.
!>                    لم يتمكن من إكمال تحلل B
!>                    ولم يتم حساب أي قيم ذاتية أو متجهات ذاتية.
!>

NAME

zhegvd - computes the eigenvalues and eigenvectors of a complex Hermitian positive-definite-definite generalized eigenvalue problem.

SYNOPSIS

                          iu, lwork, rwork, rwork2, iwork, liwork, info)

    character jobz, range, uplo
    integer n, lda, ldb, il, iu, lwork, liwork, info
    complex*16 a(lda, *), b(ldb, *), vl, vu

    integer iwork(*)
