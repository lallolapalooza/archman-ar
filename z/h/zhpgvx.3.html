subroutine zhpgvx (integer itype, character jobz, character range, character uplo, integer n, complex*16, dimension( * ) ap, complex*16, dimension( * ) bp, double precision vl, double precision vu, integer il, integer iu, double precision abstol, integer m, double precision, dimension( * ) w, complex*16, dimension( ldz, * ) z, integer ldz, complex*16, dimension( * ) work, double precision, dimension( * ) rwork, integer, dimension( * ) iwork, integer, dimension( * ) ifail, integer info)

ZHPGVX

الغرض:

!>
!> ZHPGVX يحسب قيمًا ذاتية مختارة، وبشكل اختياري، متجهات ذاتية
!> لمشكلة القيم الذاتية الهرميتية المحددة المعممة المعقدة، من الشكل
!> A*x=(lambda)*B*x,  A*Bx=(lambda)*x,  أو B*A*x=(lambda)*x. هنا يُفترض أن A و
!> B هما هرميتيان، مخزنان بتنسيق مضغوط، و B أيضًا
!> محدد موجب. يمكن اختيار القيم الذاتية والمتجهات الذاتية عن طريق
!> تحديد نطاق من القيم أو نطاق من المؤشرات
!> للقيم الذاتية المطلوبة.
!>

المعلمات

ITYPE 
!>          ITYPE هو عدد صحيح
!>          يحدد نوع المشكلة المراد حلها:
!>          = 1:  A*x = (lambda)*B*x
!>          = 2:  A*B*x = (lambda)*x
!>          = 3:  B*A*x = (lambda)*x
!>

JOBZ

!>          JOBZ هو حرف واحد
!>          = 'N':  حساب القيم الذاتية فقط؛
!>          = 'V':  حساب القيم الذاتية والمتجهات الذاتية.
!>

RANGE

!>          RANGE هو حرف واحد
!>          = 'A': سيتم العثور على جميع القيم الذاتية؛
!>          = 'V': سيتم العثور على جميع القيم الذاتية في الفترة نصف المفتوحة (VL,VU]
!>                 ؛
!>          = 'I': سيتم العثور على القيم الذاتية من IL إلى IU.
!>

UPLO

!>          UPLO هو حرف واحد
!>          = 'U':  المثلثات العليا لـ A و B مخزنة؛
!>          = 'L':  المثلثات السفلى لـ A و B مخزنة.
!>

N

!>          N هو عدد صحيح
!>          رتبة المصفوفات A و B.  N >= 0.
!>

AP

!>          AP هو مصفوفة COMPLEX*16، البعد (N*(N+1)/2)
!>          عند الإدخال، المثلث العلوي أو السفلي للمصفوفة الهرميتية
!>          A، مضغوطة حسب العمود في مصفوفة خطية. يتم تخزين العمود j من A
!>          في المصفوفة AP على النحو التالي:
!>          إذا كان UPLO = 'U'، فإن AP(i + (j-1)*j/2) = A(i,j) لـ 1<=i<=j؛
!>          إذا كان UPLO = 'L'، فإن AP(i + (j-1)*(2*n-j)/2) = A(i,j) لـ j<=i<=n.
!>
!>          عند الخروج، يتم تدمير محتويات AP.
!>

BP

!>          BP هو مصفوفة COMPLEX*16، البعد (N*(N+1)/2)
!>          عند الإدخال، المثلث العلوي أو السفلي للمصفوفة الهرميتية
!>          B، مضغوطة حسب العمود في مصفوفة خطية. يتم تخزين العمود j من B
!>          في المصفوفة BP على النحو التالي:
!>          إذا كان UPLO = 'U'، فإن BP(i + (j-1)*j/2) = B(i,j) لـ 1<=i<=j؛
!>          إذا كان UPLO = 'L'، فإن BP(i + (j-1)*(2*n-j)/2) = B(i,j) لـ j<=i<=n.
!>
!>          عند الخروج، العامل المثلثي U أو L من التحلل Cholesky
!>          B = U**H*U أو B = L*L**H، بنفس تنسيق التخزين
!>          مثل B.
!>

VL

!>          VL هو double precision
!>
!>          إذا كان RANGE='V'، الحد الأدنى للفترة المراد
!>          البحث فيها عن القيم الذاتية. VL < VU.
!>          لا يتم الرجوع إليها إذا كان RANGE = 'A' أو 'I'.
!>

VU

!>          VU هو double precision
!>
!>          إذا كان RANGE='V'، الحد الأعلى للفترة المراد
!>          البحث فيها عن القيم الذاتية. VL < VU.
!>          لا يتم الرجوع إليها إذا كان RANGE = 'A' أو 'I'.
!>

IL

!>          IL هو عدد صحيح
!>
!>          إذا كان RANGE='I'، مؤشر
!>          أصغر قيمة ذاتية سيتم إرجاعها.
!>          1 <= IL <= IU <= N، إذا كان N > 0؛ IL = 1 و IU = 0 إذا كان N = 0.
!>          لا يتم الرجوع إليها إذا كان RANGE = 'A' أو 'V'.
!>

IU

!>          IU هو عدد صحيح
!>
!>          إذا كان RANGE='I'، مؤشر
!>          أكبر قيمة ذاتية سيتم إرجاعها.
!>          1 <= IL <= IU <= N، إذا كان N > 0؛ IL = 1 و IU = 0 إذا كان N = 0.
!>          لا يتم الرجوع إليها إذا كان RANGE = 'A' أو 'V'.
!>

ABSTOL

!>          ABSTOL هو double precision
!>          التسامح المطلق للخطأ في القيم الذاتية.
!>          يتم قبول قيمة ذاتية تقريبية على أنها متقاربة
!>          عندما يتم تحديد أنها تقع في فترة [a,b]
!>          بعرض أقل من أو يساوي
!>
!>          ABSTOL + EPS *   max( |a|,|b| ) ،
!>
!>          حيث EPS هي دقة الآلة. إذا كان ABSTOL أقل من
!>          أو يساوي الصفر، فسيتم استخدام EPS * |T| بدلاً منه،
!>          حيث |T| هو المعيار 1 للمصفوفة القطرية الثلاثية التي تم الحصول عليها
!>          عن طريق تقليل AP إلى شكل قطري ثلاثي.
!>
!>          سيتم حساب القيم الذاتية بأكبر دقة عندما يتم
!>          تعيين ABSTOL إلى ضعف عتبة الانخفاض 2*DLAMCH('S')، وليس صفرًا.
!>          إذا عادت هذه الوظيفة الفرعية بـ INFO>0، مما يشير إلى أن بعض
!>          المتجهات الذاتية لم تتقارب، حاول تعيين ABSTOL إلى
!>          2*DLAMCH('S').
!>

M

!>          M هو عدد صحيح
!>          إجمالي عدد القيم الذاتية التي تم العثور عليها.  0 <= M <= N.
!>          إذا كان RANGE = 'A'، فإن M = N، وإذا كان RANGE = 'I'، فإن M = IU-IL+1.
!>

W

!>          W هو مصفوفة double precision، البعد (N)
!>          عند الخروج العادي، تحتوي العناصر الأولى M على القيم الذاتية
!>          المختارة بترتيب تصاعدي.
!>

Z

!>          Z هي مصفوفة COMPLEX*16، البعد (LDZ, N)
!>          إذا كان JOBZ = 'N'، فلا يتم الرجوع إلى Z.
!>          إذا كان JOBZ = 'V'، فإذا كان INFO = 0، فإن الأعمدة M الأولى من Z
!>          تحتوي على المتجهات الذاتية المتعامدة للمصفوفة A
!>          المقابلة للقيم الذاتية المختارة، مع احتواء العمود i من Z
!>          على المتجه الذاتي المرتبط بـ W(i).
!>          يتم تطبيع المتجهات الذاتية على النحو التالي:
!>          إذا كان ITYPE = 1 أو 2، فإن Z**H*B*Z = I؛
!>          إذا كان ITYPE = 3، فإن Z**H*inv(B)*Z = I.
!>
!>          إذا فشل متجه ذاتي في التقارب، فإن هذا العمود من Z
!>          يحتوي على أحدث تقريب للمتجه الذاتي، ويتم إرجاع
!>          مؤشر المتجه الذاتي في IFAIL.
!>          ملاحظة: يجب على المستخدم التأكد من توفير ما لا يقل عن M عمود (بحد أدنى 1)
!>          في المصفوفة Z؛ إذا كان RANGE = 'V'، فإن القيمة الدقيقة لـ M
!>          غير معروفة مسبقًا ويجب استخدام حد أعلى.
!>

LDZ

!>          LDZ هو عدد صحيح
!>          البعد الرئيسي للمصفوفة Z. LDZ >= 1، وإذا كان
!>          JOBZ = 'V'، فإن LDZ >= max(1,N).
!>

WORK

!>          WORK هي مصفوفة COMPLEX*16، البعد (2*N)
!>

RWORK

!>          RWORK هي مصفوفة double precision، البعد (7*N)
!>

IWORK

!>          IWORK هي مصفوفة INTEGER، البعد (5*N)
!>

IFAIL

!>          IFAIL هي مصفوفة INTEGER، البعد (N)
!>          إذا كان JOBZ = 'V'، فإذا كان INFO = 0، فإن العناصر M الأولى من
!>          IFAIL هي صفر. إذا كان INFO > 0، فإن IFAIL تحتوي على
!>          مؤشرات المتجهات الذاتية التي فشلت في التقارب.
!>          إذا كان JOBZ = 'N'، فلا يتم الرجوع إلى IFAIL.
!>

INFO

!>
