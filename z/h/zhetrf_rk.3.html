subroutine zhetrf_rk (character uplo, integer n, complex*16, dimension( lda, * ) a, integer lda, complex*16, dimension( * ) e, integer, dimension( * ) ipiv, complex*16, dimension( * ) work, integer lwork, integer info)

ZHETRF_RK تحسب تفكيك مصفوفة هرميتية غير محددة مركبة باستخدام طريقة محورية قطرية مقيدة لـ Bunch-Kaufman (rook) (خوارزمية BLAS3 المجمعة).

الغرض:

!> ZHETRF_RK تحسب تفكيك مصفوفة هرميتية مركبة A
!> باستخدام طريقة محورية قطرية مقيدة لـ Bunch-Kaufman (rook):
!>
!>    A = P*U*D*(U**H)*(P**T) أو A = P*L*D*(L**H)*(P**T)،
!>
!> حيث U (أو L) هي مصفوفة مثلثية علوية (أو سفلية) وحدوية،
!> U**H (أو L**H) هي مترافقة U (أو L)، P هي مصفوفة تبديل،
!> P**T هي منقولة P، و D هي هرميتية وقطرية كتلة مع كتل قطرية 1x1 و 2x2.
!>
!> هذه هي النسخة المجمعة من الخوارزمية، وتستدعي Level 3 BLAS.
!> لمزيد من المعلومات، انظر قسم التفاصيل الإضافية.
!>

المعلمات

UPLO 
!>          UPLO هو CHARACTER*1
!>          يحدد ما إذا كان الجزء العلوي أو السفلي المثلثي للمصفوفة الهرميتية A مخزنًا:
!>          = 'U':  مثلثي علوي
!>          = 'L':  مثلثي سفلي
!>

N

!>          N هو INTEGER
!>          رتبة المصفوفة A.  N >= 0.
!>

A

!>          A هو مصفوفة COMPLEX*16، البعد (LDA,N)
!>          عند الإدخال، المصفوفة الهرميتية A.
!>            إذا كان UPLO = 'U': الجزء العلوي المثلثي الذي حجمه N×N من
!>            A يحتوي على الجزء العلوي المثلثي للمصفوفة A،
!>            والجزء السفلي المثلثي الصارم من A لا يتم الرجوع إليه.
!>
!>            إذا كان UPLO = 'L': الجزء السفلي المثلثي الذي حجمه N×N من
!>            A يحتوي على الجزء السفلي المثلثي للمصفوفة A،
!>            والجزء العلوي المثلثي الصارم من A لا يتم الرجوع إليه.
!>
!>          عند الخروج، يحتوي على:
!>            أ) فقط العناصر القطرية للمصفوفة القطرية الكتلية الهرميتية D
!>               على القطر للمصفوفة A، أي D(k,k) = A(k,k)؛
!>               (العناصر فوق القطرية (أو تحت القطرية) للمصفوفة D
!>                مخزنة عند الخروج في المصفوفة E)، و
!>            ب) إذا كان UPLO = 'U': العامل U في الجزء فوق القطري من A.
!>               إذا كان UPLO = 'L': العامل L في الجزء تحت القطري من A.
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرائد للمصفوفة A.  LDA >= max(1,N).
!>

E

!>          E هو مصفوفة COMPLEX*16، البعد (N)
!>          عند الخروج، تحتوي على العناصر فوق القطرية (أو تحت القطرية)
!>          للمصفوفة القطرية الكتلية الهرميتية D
!>          مع كتل قطرية 1x1 أو 2x2، حيث
!>          إذا كان UPLO = 'U': E(i) = D(i-1,i)، i=2:N، E(1) مضبوطة على 0؛
!>          إذا كان UPLO = 'L': E(i) = D(i+1,i)، i=1:N-1، E(N) مضبوطة على 0.
!>
!>          ملاحظة: للكتلة القطرية 1x1 D(k)، حيث
!>          1 <= k <= N، العنصر E(k) مضبوط على 0 في كلتا الحالتين
!>          UPLO = 'U' أو UPLO = 'L'.
!>

IPIV

!>          IPIV هي مصفوفة INTEGER، البعد (N)
!>          يصف IPIV مصفوفة التبديل P في تفكيك
!>          المصفوفة A على النحو التالي. القيمة المطلقة لـ IPIV(k)
!>          تمثل فهرس الصف والعمود اللذين تم تبديلهما
!>          مع الصف والعمود k. قيمة UPLO
!>          تصف ترتيب تطبيق التبادلات.
!>          أيضًا، إشارة IPIV تمثل البنية الكتلية
!>          للمصفوفة القطرية الكتلية الهرميتية D مع كتل قطرية 1x1 أو 2x2
!>          والتي تتوافق مع تبادل واحد أو اثنين
!>          في كل خطوة تفكيك. لمزيد من المعلومات، انظر قسم التفاصيل الإضافية.
!>
!>          إذا كان UPLO = 'U'،
!>          (بترتيب التفكيك، k ينقص من N إلى 1):
!>            أ) إدخال موجب واحد IPIV(k) > 0 يعني:
!>               D(k,k) هي كتلة قطرية 1x1.
!>               إذا كان IPIV(k) != k، تم تبديل الصفوف والأعمدة k و IPIV(k)
!>               في المصفوفة A(1:N,1:N)؛
!>               إذا كان IPIV(k) = k، لم يحدث تبديل.
!>
!>            ب) زوج من الإدخالات المتتالية السالبة
!>               IPIV(k) < 0 و IPIV(k-1) < 0 يعني:
!>               D(k-1:k,k-1:k) هي كتلة قطرية 2x2.
!>               (ملاحظة: الإدخالات السالبة في IPIV تظهر فقط في أزواج).
!>               1) إذا كان -IPIV(k) != k، تم تبديل الصفوف والأعمدة
!>                  k و -IPIV(k)
!>                  في المصفوفة A(1:N,1:N).
!>                  إذا كان -IPIV(k) = k، لم يحدث تبديل.
!>               2) إذا كان -IPIV(k-1) != k-1، تم تبديل الصفوف والأعمدة
!>                  k-1 و -IPIV(k-1)
!>                  في المصفوفة A(1:N,1:N).
!>                  إذا كان -IPIV(k-1) = k-1، لم يحدث تبديل.
!>
!>            ج) في كلتا الحالتين أ) وب)، دائمًا ABS( IPIV(k) ) <= k.
!>
!>            د) ملاحظة: أي إدخال IPIV(k) هو دائمًا غير صفري عند الخروج.
!>
!>          إذا كان UPLO = 'L'،
!>          (بترتيب التفكيك، k يزداد من 1 إلى N):
!>            أ) إدخال موجب واحد IPIV(k) > 0 يعني:
!>               D(k,k) هي كتلة قطرية 1x1.
!>               إذا كان IPIV(k) != k، تم تبديل الصفوف والأعمدة k و IPIV(k)
!>               في المصفوفة A(1:N,1:N).
!>               إذا كان IPIV(k) = k، لم يحدث تبديل.
!>
!>            ب) زوج من الإدخالات المتتالية السالبة
!>               IPIV(k) < 0 و IPIV(k+1) < 0 يعني:
!>               D(k:k+1,k:k+1) هي كتلة قطرية 2x2.
!>               (ملاحظة: الإدخالات السالبة في IPIV تظهر فقط في أزواج).
!>               1) إذا كان -IPIV(k) != k، تم تبديل الصفوف والأعمدة
!>                  k و -IPIV(k)
!>                  في المصفوفة A(1:N,1:N).
!>                  إذا كان -IPIV(k) = k، لم يحدث تبديل.
!>               2) إذا كان -IPIV(k+1) != k+1، تم تبديل الصفوف والأعمدة
!>                  k-1 و -IPIV(k-1)
!>                  في المصفوفة A(1:N,1:N).
!>                  إذا كان -IPIV(k+1) = k+1، لم يحدث تبديل.
!>
!>            ج) في كلتا الحالتين أ) وب)، دائمًا ABS( IPIV(k) ) >= k.
!>
!>            د) ملاحظة: أي إدخال IPIV(k) هو دائمًا غير صفري عند الخروج.
!>

WORK

!>          WORK هي مصفوفة COMPLEX*16، البعد (MAX(1,LWORK)).
!>          عند الخروج، إذا كان INFO = 0، فإن WORK(1) ترجع LWORK الأمثل.
!>

LWORK

!>          LWORK هو INTEGER
!>          طول WORK.  LWORK >= 1.  للحصول على أفضل أداء
!>          LWORK >= N*NB، حيث NB هو حجم الكتلة الذي ترجعه
!>          ILAENV.
!>
!>          إذا كان LWORK = -1، فسيتم افتراض استعلام مساحة العمل؛
!>          تقوم الروتين الفرعي بحساب حجم مساحة العمل المثلى فقط،
!>          تُرجع هذا القيمة كأول إدخال في مساحة العمل WORK،
!>          ولا يتم إصدار أي رسالة خطأ تتعلق بـ LWORK بواسطة XERBLA.
!>

INFO

!>          INFO هو INTEGER
!>          = 0: خروج ناجح
!>
!>          < 0: إذا كان INFO = -k، فإن المعلمة k كانت لها قيمة غير قانونية
!>
!>          > 0: إذا كان INFO = k، فإن المصفوفة A مفردة، وذلك بسبب:
!>                 إذا كان UPLO = 'U': العمود k في
!>                 الجزء المثلثي العلوي من A يحتوي على جميع الأصفار.
!>                 إذا كان UPLO = 'L': العمود k في
!>                 الجزء المثلثي السفلي من A يحتوي على جميع الأصفار.
!>
!>               لذلك D(k,k) تساوي صفرًا تمامًا، والعناصر فوق القطرية
!>               للعمود k من U (أو العناصر تحت القطرية لـ
!>               العمود k من L) هي جميعها أصفار. تم إكمال التفكيك،
!>               ولكن المصفوفة القطرية الكتلية D
!>               مفردة تمامًا، و

توثيق الوظيفة/الإجراء الفرعي

order
CHARACTER*1.
يحدد ترتيب تخزين المصفوفة.
uplo
CHARACTER*1.
يحدد ما إذا كان الجزء العلوي أو السفلي من A سيتم استخدامه.
n
INTEGER.
رتبة المصفوفة.
a
COMPLEX*16 array, dimension at least (max(1, n), max(1, n)).
قبل الإدخال ، يحتوي الجزء المحدد من المصفوفة A على البيانات التي سيتم تحليلها.
lda
INTEGER.
LDG of the array A.
ipiv
INTEGER array, dimension at least (max(1, n)).
يحتوي على تفاصيل التبديل.
tau
COMPLEX*16 array, dimension at least (max(1, n)).
يحتوي على عوامل انعكاس المتجهات.
work
COMPLEX*16 array, dimension at least (max(1, lwork)).
مساحة عمل.
lwork
INTEGER.
عدد العناصر في مساحة العمل WORK.
info
INTEGER.

Info = 0: النجاح.

Info < 0: المعامل i-th الذي تم استدعاؤه كان غير صالح.

Info > 0: العنصر a(i,i) يساوي صفر.

يمكن استخدام التحليل لتشكيل مصفوفة عكسية أو حل نظام معادلات خطية A*X = B.

يمكن أن يحدث القسمة على صفر إذا تم استخدامها لحل نظام معادلات.

ملاحظة: يقوم INFO بتخزين التكرار الأول فقط لـ المفردات ، أي تكرار لاحق للمفردات لا يتم تخزينه في INFO على الرغم من أن التحليل يكتمل دائمًا.
