subroutine zherk (character uplo, character trans, integer n, integer k, double precision alpha, complex*16, dimension(lda,*) a, integer lda, double precision beta, complex*16, dimension(ldc,*) c, integer ldc)

ZHERK

الغرض:

!>
!> ZHERK  يقوم بواحدة من عمليات الرتبة k الهرمية
!>
!>    C := alpha*A*A**H + beta*C,
!>
!> أو
!>
!>    C := alpha*A**H*A + beta*C,
!>
!> حيث  alpha  و beta  هي  ثوابت حقيقية،  C  هي مصفوفة هرمية بحجم n × n
!> و A  هي مصفوفة بحجم n × k في الحالة الأولى ومصفوفة بحجم k × n
!> في الحالة الثانية.
!>

المعلمات

UPLO 
!>          UPLO هو CHARACTER*1
!>           عند الإدخال،  UPLO  تحدد ما إذا كان الجزء المثلثي العلوي أو السفلي
!>           من المصفوفة C  سيتم الرجوع إليه على النحو التالي:
!>
!>              UPLO = 'U' أو 'u'   سيتم الرجوع فقط إلى الجزء المثلثي العلوي من C.
!>
!>              UPLO = 'L' أو 'l'   سيتم الرجوع فقط إلى الجزء المثلثي السفلي من C.
!>

TRANS

!>          TRANS هو CHARACTER*1
!>           عند الإدخال،  TRANS  تحدد العملية التي سيتم إجراؤها على النحو التالي:
!>
!>              TRANS = 'N' أو 'n'   C := alpha*A*A**H + beta*C.
!>
!>              TRANS = 'C' أو 'c'   C := alpha*A**H*A + beta*C.
!>

N

!>          N هو INTEGER
!>           عند الإدخال،  N  تحدد رتبة المصفوفة C.  يجب أن تكون N
!>           على الأقل صفر.
!>

K

!>          K هو INTEGER
!>           عند الإدخال مع  TRANS = 'N' أو 'n'،  K  تحدد عدد
!>           الأعمدة  للمصفوفة A، وعند الإدخال مع
!>           TRANS = 'C' أو 'c'،  K  تحدد عدد الصفوف للمصفوفة A.  يجب أن تكون K
!>           على الأقل صفر.
!>

ALPHA

!>          ALPHA هو DOUBLE PRECISION .
!>           عند الإدخال،  ALPHA  تحدد الثابت alpha.
!>

A

!>          A هي مصفوفة COMPLEX*16، بأبعاد ( LDA, ka )، حيث ka هي
!>           k  عندما  TRANS = 'N' أو 'n'،  و n  بخلاف ذلك.
!>           قبل الإدخال مع  TRANS = 'N' أو 'n'،  يجب أن يحتوي الجزء الرائد بحجم n × k
!>           من المصفوفة A  على المصفوفة A، وإلا
!>           فيجب أن يحتوي الجزء الرائد بحجم k × n  من المصفوفة A  على
!>           المصفوفة A.
!>

LDA

!>          LDA هو INTEGER
!>           عند الإدخال،  LDA  تحدد البعد الأول لـ A كما تم تعريفه
!>           في (الإجراء الفرعي) المستدعي.   عندما يكون TRANS = 'N' أو 'n'
!>           فيجب أن يكون LDA  على الأقل  max( 1, n )، وإلا
!>           فيجب أن يكون LDA  على الأقل  max( 1, k ).
!>

BETA

!>          BETA هو DOUBLE PRECISION.
!>           عند الإدخال،  BETA  يحدد الثابت beta.
!>

C

!>          C هي مصفوفة COMPLEX*16، بأبعاد ( LDC, N )
!>           قبل الإدخال مع  UPLO = 'U' أو 'u'،  يجب أن يحتوي الجزء الرائد بحجم n × n
!>           الجزء المثلثي العلوي من المصفوفة C  على الجزء المثلثي العلوي
!>           من المصفوفة الهرمية  والجزء المثلثي السفلي الصارم لـ C  لا يتم الرجوع إليه.  عند الخروج،
!>           يتم الكتابة فوق الجزء المثلثي العلوي من المصفوفة C  بالجزء المثلثي العلوي
!>           من المصفوفة المحدثة.
!>           قبل الإدخال مع  UPLO = 'L' أو 'l'،  يجب أن يحتوي الجزء الرائد بحجم n × n
!>           الجزء المثلثي السفلي من المصفوفة C  على الجزء المثلثي السفلي
!>           من المصفوفة الهرمية  والجزء المثلثي العلوي الصارم لـ C  لا يتم الرجوع إليه.  عند الخروج،
!>           يتم الكتابة فوق الجزء المثلثي السفلي من المصفوفة C  بالجزء المثلثي السفلي
!>           من المصفوفة المحدثة.
!>           لاحظ أن الأجزاء التخيلية للعناصر القطرية ليست
!>           ضرورية لتعيينها،  يُفترض أنها صفر، وعند الخروج تكون
!>           تعيينها إلى صفر.
!>

LDC

!>          LDC هو INTEGER
!>           عند الإدخال،  LDC  تحدد البعد الأول لـ C كما تم تعريفه
!>           في (الإجراء الفرعي) المستدعي.   يجب أن يكون LDC
!>           على الأقل  max( 1, n ).
!>
