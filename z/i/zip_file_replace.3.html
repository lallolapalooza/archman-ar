zip_file_replace(3)

الحزمة: libzip

التراخيص: BSD

ZIP_FILE_ADD(3) دليل وظائف المكتبة ZIP_FILE_ADD(3)

الاسم

zip_file_add، zip_file_replace — إضافة ملف إلى أرشيف zip أو استبدال ملف في أرشيف zip

المكتبة

libzip (-lzip)

ملخص

#include <zip.h>

zip_int64_t
zip_file_add(zip_t *archive, const char *name, zip_source_t *source, zip_flags_t flags);

int
zip_file_replace(zip_t *archive, zip_uint64_t index, zip_source_t *source, zip_flags_t flags);

الوصف

تقوم الدالة zip_file_add() بإضافة ملف إلى أرشيف zip، بينما zip_file_replace() تستبدل ملفًا موجودًا في أرشيف zip. يحدد الوسيط archive أرشيف zip الذي يجب إضافة الملف إليه. name هو اسم الملف في أرشيف zip (لـ zip_file_add())، بينما index يحدد أي ملف يجب استبداله (لـ zip_file_replace()). يمكن أن يكون الوسيط flags أي مزيج من ZIP_FL_OVERWRITE مع أحد ZIP_FL_ENC_*:

ZIP_FL_OVERWRITE
استبدال أي ملف موجود بنفس الاسم. لـ zip_file_add فقط.
ZIP_FL_ENC_GUESS
تخمين ترميز name (افتراضي). (يتم التعرف على CP-437 و UTF-8 فقط.)
ZIP_FL_ENC_UTF_8
تفسير name كـ UTF-8.
ZIP_FL_ENC_CP437
تفسير name كصفحة ترميز 437 (CP-437).
يتم الحصول على البيانات من الوسيط source، انظر zip_source(5).

ملاحظة: لا يجب استدعاء zip_source_free(3) على source بعد استخدامه بنجاح في استدعاء zip_file_add أو zip_file_replace.

يرجى ملاحظة أيضًا أنه عند استخدام zip_replace، سيتم حذف معلومات الحقل الإضافي للملف المستهدف لأنها تعتمد عادةً على محتويات الملف. إذا كنت ترغب في الاحتفاظ بها، قم بالاستعلام عنها مسبقًا باستخدام zip_file_extra_field_get(3) واستعادتها بعد zip_replace باستخدام zip_file_extra_field_set(3).

قيم الإرجاع

عند النجاح، ترجع الدالة zip_file_add() فهرس الملف الجديد في الأرشيف، وترجع الدالة zip_file_replace() القيمة 0. وإلا، ترجع -1 ويتم تعيين رمز الخطأ في archive للإشارة إلى الخطأ.

أمثلة

zip_source_t *s;
const char buf[]="teststring";

if ((s=zip_source_buffer(archive, buf, sizeof(buf), 0)) == NULL ||
    zip_file_add(archive, name, s, ZIP_FL_ENC_UTF_8) < 0) {
    zip_source_free(s);
    printf("error adding file: %s\n", zip_strerror(archive));
}

الأخطاء

تفشل الدالتان zip_file_add() و zip_file_replace() إذا:

[ZIP_ER_EXISTS]
يوجد بالفعل ملف بالاسم name في الأرشيف. (ينطبق فقط على zip_file_add()، وعندما لا يتم توفير ZIP_FL_OVERWRITE).
[ZIP_ER_INVAL]
source أو name هما NULL، أو index غير صالح.
[ZIP_ER_MEMORY]
لم يتمكن من تخصيص الذاكرة المطلوبة.
[ZIP_ER_RDONLY]
تم فتح الأرشيف في وضع القراءة فقط.

انظر أيضًا

libzip(3)، zip_source(5)
