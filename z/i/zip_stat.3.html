zip_stat(3)

الحزمة: libzip

التراخيص: BSD

ZIP_STAT(3) دليل وظائف المكتبة ZIP_STAT(3)

الاسم

zip_stat، zip_stat_index — الحصول على معلومات حول ملف

المكتبة

libzip (-lzip)

الملخص

#include <zip.h>

int
zip_stat(zip_t *archive, const char *fname, zip_flags_t flags, zip_stat_t *sb);

int
zip_stat_index(zip_t *archive, zip_uint64_t index, zip_flags_t flags, zip_stat_t *sb);

الوصف

تقوم الدالة zip_stat() بالحصول على معلومات حول الملف المسمى fname في archive. تحدد الوسيطة flags كيفية إجراء البحث عن الاسم. يتم وصف قيمها في zip_name_locate(3). بالإضافة إلى ذلك، يمكن أوتحديد ZIP_FL_UNCHANGED بها لطلب معلومات حول الملف الأصلي في الأرشيف، متجاهلاً أي تغييرات تم إجراؤها.

تقوم الدالة zip_stat_index() بالحصول على معلومات حول الملف في الموضع index.

الوسيطة sb هي مؤشر إلى struct zip_stat (معروضة أدناه)، يتم وضع معلومات حول الملف فيها.

struct zip_stat {
    zip_uint64_t valid;                 /* أي الحقول لها قيم صالحة */
    const char *name;                   /* اسم الملف */
    zip_uint64_t index;                 /* الفهرس داخل الأرشيف */
    zip_uint64_t size;                  /* حجم الملف (غير مضغوط) */
    zip_uint64_t comp_size;             /* حجم الملف (مضغوط) */
    time_t mtime;                       /* وقت التعديل */
    zip_uint32_t crc;                   /* CRC لبيانات الملف */
    zip_uint16_t comp_method;           /* طريقة الضغط المستخدمة */
    zip_uint16_t encryption_method;     /* طريقة التشفير المستخدمة */
    zip_uint32_t flags;                 /* محجوز للاستخدام المستقبلي */
};
يجب تخصيص الهيكل الذي يشير إليه sb قبل استدعاء zip_stat() أو zip_stat_index().

يحدد حقل valid في الهيكل الحقول الأخرى الصالحة. تحقق مما إذا كانت العلامة المعرفة بواسطة التعريفات التالية موجودة في valid قبل الوصول إلى الحقول:

ZIP_STAT_NAME
name
ZIP_STAT_INDEX
index
ZIP_STAT_SIZE
size
ZIP_STAT_COMP_SIZE
comp_size
ZIP_STAT_MTIME
mtime
ZIP_STAT_CRC
crc
ZIP_STAT_COMP_METHOD
comp_method
ZIP_STAT_ENCRYPTION_METHOD
encryption_method
ZIP_STAT_FLAGS
flags

قيم الإرجاع

عند الانتهاء بنجاح، يتم إرجاع 0. خلاف ذلك، يتم إرجاع -1 ويتم تعيين معلومات الخطأ في archive للإشارة إلى الخطأ.

الأخطاء

يمكن أن تفشل الدالة zip_stat() لأي من الأخطاء المحددة للإجراء zip_name_locate(3).

تفشل الدالة zip_stat_index() وتعين معلومات الخطأ إلى ZIP_ER_INVAL إذا كان index غير صالح. إذا لم يتم تعيين ZIP_FL_UNCHANGED ولم يتم الحصول على أي معلومات من رد الاتصال المصدر، فسيتم تعيين معلومات الخطأ إلى ZIP_ER_CHANGED.

انظر أيضاً

libzip(3)، zip_get_num_entries(3)، zip_name_locate(3)، zip_stat_init(3)

التاريخ

تمت إضافة zip_stat() في libzip 0.6. في libzip 0.11، تم تغيير نوع flags من int إلى zip_flags_t.

تمت إضافة zip_stat_index() في libzip 0.6. في libzip 0.10، تم تغيير نوع index من int إلى zip_uint64_t. في libzip 0.11، تم تغيير نوع flags من int إلى zip_flags_t.
