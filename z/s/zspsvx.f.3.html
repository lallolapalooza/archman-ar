subroutine zspsvx (character fact, character uplo, integer n, integer nrhs, complex*16, dimension( * ) ap, complex*16, dimension( * ) afp, integer, dimension( * ) ipiv, complex*16, dimension( ldb, * ) b, integer ldb, complex*16, dimension( ldx, * ) x, integer ldx, double precision rcond, double precision, dimension( * ) ferr, double precision, dimension( * ) berr, complex*16, dimension( * ) work, double precision, dimension( * ) rwork, integer info)

ZSPSVX يحسب الحل لنظام المعادلات الخطية A * X = B للمصفوفات الأخرى

الغرض:

!>
!> يستخدم ZSPSVX تحليل التقليب القطري A = U*D*U**T أو
!> A = L*D*L**T لحساب حل نظام المعادلات الخطية المعقدة A * X = B، حيث A مصفوفة متماثلة بحجم N*N مخزنة
!> بتنسيق مكدس و X و B مصفوفتان بحجم N*NRHS.
!>
!> كما يتم توفير حدود الخطأ على الحل وتقدير للشرط.
!>

الوصف:

!>
!> يتم تنفيذ الخطوات التالية:
!>
!> 1. إذا كانت FACT = 'N'، يتم استخدام طريقة التقليب القطري لتحليل A بالشكل
!>       A = U * D * U**T، إذا كان UPLO = 'U'، أو
!>       A = L * D * L**T، إذا كان UPLO = 'L'،
!>    حيث U (أو L) هو حاصل ضرب مصفوفات التبديل والمثلثات العليا (السفلى) الوحدوية
!>    و D متماثلة وقطرية كتلية بكتل قطرية بحجم 1*1 و 2*2.
!>
!> 2. إذا كانت أي D(i,i)=0، مما يجعل D مفردة تمامًا، فإن الإجراء الفرعي
!>    يعود بـ INFO = i. بخلاف ذلك، يتم استخدام الشكل المحلل لـ A لتقدير
!>    رقم شرط المصفوفة A. إذا كان مقلوب رقم الشرط أقل من دقة الآلة،
!>    يتم إرجاع INFO = N+1 كتحذير، ولكن الإجراء الفرعي يواصل
!>    حل X وحساب حدود الخطأ كما هو موضح أدناه.
!>
!> 3. يتم حل نظام المعادلات لـ X باستخدام الشكل المحلل لـ A.
!>
!> 4. يتم تطبيق التنقيح التكراري لتحسين الحل المحسوب
!>    وحساب حدود الخطأ وتقديرات الخطأ الخلفي له.
!>

المعاملات

FACT 
!>          FACT هو CHARACTER*1
!>          يحدد ما إذا كان الشكل المحلل لـ A قد تم توفيره عند الدخول أم لا.
!>          = 'F':  عند الدخول، تحتوي AFP و IPIV على الشكل المحلل لـ
!>                  A. لن يتم تعديل AP و AFP و IPIV.
!>          = 'N':  سيتم نسخ المصفوفة A إلى AFP وتحليلها.
!>

UPLO

!>          UPLO هو CHARACTER*1
!>          = 'U':  يتم تخزين المثلث العلوي لـ A؛
!>          = 'L':  يتم تخزين المثلث السفلي لـ A.
!>

N

!>          N هو INTEGER
!>          عدد المعادلات الخطية، أي رتبة المصفوفة A. N >= 0.
!>

NRHS

!>          NRHS هو INTEGER
!>          عدد الأطراف اليمنى، أي عدد أعمدة المصفوفتين B و X. NRHS >= 0.
!>

AP

!>          AP هو مصفوفة COMPLEX*16، البعد (N*(N+1)/2)
!>          المثلث العلوي أو السفلي للمصفوفة المتماثلة A، مكدس
!>          عمودياً في مصفوفة خطية. يتم تخزين العمود j من A
!>          في المصفوفة AP على النحو التالي:
!>          إذا كان UPLO = 'U'، AP(i + (j-1)*j/2) = A(i,j) لـ 1<=i<=j؛
!>          إذا كان UPLO = 'L'، AP(i + (j-1)*(2*n-j)/2) = A(i,j) لـ j<=i<=n.
!>          انظر أدناه لمزيد من التفاصيل.
!>

AFP

!>          AFP هو مصفوفة COMPLEX*16، البعد (N*(N+1)/2)
!>          إذا كانت FACT = 'F'، فإن AFP هي معامل إدخال وتحتوي عند الدخول
!>          على المصفوفة القطرية الكتلية D والعوامل المستخدمة
!>          للحصول على العامل U أو L من التحليل
!>          A = U*D*U**T أو A = L*D*L**T كما تم حسابه بواسطة ZSPTRF، مخزنة كـ
!>          مصفوفة مثلثة مكدسة بنفس تنسيق التخزين مثل A.
!>
!>          إذا كانت FACT = 'N'، فإن AFP هي معامل إخراج وتحتوي عند الخروج
!>          على المصفوفة القطرية الكتلية D والعوامل المستخدمة
!>          للحصول على العامل U أو L من التحليل
!>          A = U*D*U**T أو A = L*D*L**T كما تم حسابه بواسطة ZSPTRF، مخزنة كـ
!>          مصفوفة مثلثة مكدسة بنفس تنسيق التخزين مثل A.
!>

IPIV

!>          IPIV هو مصفوفة INTEGER، البعد (N)
!>          إذا كانت FACT = 'F'، فإن IPIV هي معامل إدخال وتحتوي عند الدخول
!>          على تفاصيل التبادلات وهيكل الكتل لـ D، كما تم تحديده بواسطة ZSPTRF.
!>          إذا كان IPIV(k) > 0، تم تبديل الصفوف والأعمدة k و IPIV(k)
!>          و D(k,k) هي كتلة قطرية بحجم 1*1.
!>          إذا كان UPLO = 'U' و IPIV(k) = IPIV(k-1) < 0، تم تبديل الصفوف و
!>          الأعمدة k-1 و -IPIV(k) و D(k-1:k,k-1:k)
!>          هي كتلة قطرية بحجم 2*2. إذا كان UPLO = 'L' و IPIV(k) =
!>          IPIV(k+1) < 0، تم تبديل الصفوف والأعمدة k+1 و -IPIV(k)
!>          و D(k:k+1,k:k+1) هي كتلة قطرية بحجم 2*2.
!>
!>          إذا كانت FACT = 'N'، فإن IPIV هي معامل إخراج وتحتوي عند الخروج
!>          على تفاصيل التبادلات وهيكل الكتل لـ D، كما تم تحديده بواسطة ZSPTRF.
!>

B

!>          B هي مصفوفة COMPLEX*16، البعد (LDB,NRHS)
!>          مصفوفة الطرف الأيمن B بحجم N*NRHS.
!>

LDB

!>          LDB هو INTEGER
!>          البعد الرئيسي للمصفوفة B. LDB >= max(1,N).
!>

X

!>          X هي مصفوفة COMPLEX*16، البعد (LDX,NRHS)
!>          إذا كان INFO = 0 أو INFO = N+1، فإن مصفوفة الحل X بحجم N*NRHS.
!>

LDX

!>          LDX هو INTEGER
!>          البعد الرئيسي للمصفوفة X. LDX >= max(1,N).
!>

RCOND

!>          RCOND هو DOUBLE PRECISION
!>          تقدير المقلوب لرقم شرط المصفوفة A. إذا كان RCOND أقل من دقة الآلة
!>          (على وجه الخصوص، إذا كان RCOND = 0)، فإن المصفوفة مفردة بالنسبة لدقة العمل.
!>          يتم الإشارة إلى هذا الشرط برمز عودة INFO > 0.
!>

FERR

!>          FERR هي مصفوفة DOUBLE PRECISION، البعد (NRHS)
!>          حدود الخطأ الأمامية المقدرة لكل متجه حل
!>          X(j) (العمود j من مصفوفة الحل X).
!>          إذا كان XTRUE هو الحل الحقيقي المقابل لـ X(j)، فإن FERR(j)
!>          هو حد أعلى مقدر للقيمة المطلقة لأكبر عنصر في (X(j) - XTRUE) مقسومة على القيمة المطلقة لـ
!>          أكبر عنصر في X(j). التقدير موثوق به مثل
!>          تقدير RCOND، وهو دائمًا تقريبًا تقدير زائد للخطأ الحقيقي.
!>

BERR

!>          BERR هي مصفوفة DOUBLE PRECISION، البعد (NRHS)
!>          الخطأ الخلفي النسبي المكوني لكل متجه حل
!>          X(j) (أي، أصغر تغيير نسبي في
!>          أي عنصر من A أو B يجعل X(j) حلاً دقيقًا).
!>

WORK

!>          WORK هي مصفوفة COMPLEX*16، البعد (2*N)
!>

RWORK

!>          RWORK هي مصفوفة DOUBLE PRECISION، البعد (N)
!>

INFO

!>          INFO هو INTEGER
!>          = 0: خروج ناجح
!>          < 0: إذا كان INFO = -i
