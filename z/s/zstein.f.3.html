توثيق الوظيفة/الإجراء الفرعي

subroutine zstein (integer n, double precision, dimension( * ) d, double precision, dimension( * ) e, integer m, double precision, dimension( * ) w, integer, dimension( * ) iblock, integer, dimension( * ) isplit, complex*16, dimension( ldz, * ) z, integer ldz, double precision, dimension( * ) work, integer, dimension( * ) iwork, integer, dimension( * ) ifail, integer info)

ZSTEIN

الغرض:

!>
!> ZSTEIN تحسب المتجهات الذاتية لمصفوفة حقيقية متناظرة ثلاثية الأقطار
!> T المقابلة للقيم الذاتية المحددة، باستخدام التكرار العكسي.
!>
!> يتم تحديد الحد الأقصى لعدد التكرارات لكل متجه ذاتي بواسطة
!> معلمة داخلية MAXITS (حاليًا مضبوطة على 5).
!>
!> على الرغم من أن المتجهات الذاتية حقيقية، إلا أنه يتم تخزينها في مصفوفة مركبة،
!> والتي يمكن تمريرها إلى ZUNMTR أو ZUPMTR للتحويل العكسي
!> إلى المتجهات الذاتية لمصفوفة هرميتية مركبة تم اختزالها إلى شكل ثلاثي الأقطار.
!>
!>

المعلمات

N 
!>          N هو عدد صحيح (INTEGER)
!>          رتبة المصفوفة.  N >= 0.
!>

D

!>          D هي مصفوفة دقة مزدوجة (DOUBLE PRECISION)، البعد (N)
!>          العناصر القطرية n للمصفوفة ثلاثية الأقطار T.
!>

E

!>          E هي مصفوفة دقة مزدوجة (DOUBLE PRECISION)، البعد (N-1)
!>          العناصر دون القطرية (n-1) للمصفوفة ثلاثية الأقطار
!>          T، مخزنة في العناصر من 1 إلى N-1.
!>

M

!>          M هو عدد صحيح (INTEGER)
!>          عدد المتجهات الذاتية المطلوب حسابها.  0 <= M <= N.
!>

W

!>          W هي مصفوفة دقة مزدوجة (DOUBLE PRECISION)، البعد (N)
!>          تحتوي العناصر M الأولى من W على القيم الذاتية
!>          التي سيتم حساب المتجهات الذاتية المقابلة لها.  يجب أن تكون القيم الذاتية
!>          مجمعة حسب الكتلة المنفصلة ومرتبة من الأصغر إلى الأكبر
!>          داخل الكتلة.  ( مصفوفة الإخراج W من DSTEBZ مع ORDER = 'B' متوقعة هنا. )
!>

IBLOCK

!>          IBLOCK هي مصفوفة أعداد صحيحة (INTEGER)، البعد (N)
!>          مؤشرات المصفوفة الفرعية المرتبطة بالقيم الذاتية المقابلة
!>          في W؛ IBLOCK(i)=1 إذا كانت القيمة الذاتية W(i) تنتمي إلى
!>          المصفوفة الفرعية الأولى من الأعلى، =2 إذا كانت W(i) تنتمي إلى
!>          المصفوفة الفرعية الثانية، إلخ.  ( مصفوفة الإخراج IBLOCK
!>          من DSTEBZ متوقعة هنا. )
!>

ISPLIT

!>          ISPLIT هي مصفوفة أعداد صحيحة (INTEGER)، البعد (N)
!>          نقاط الفصل، التي تتجزأ عندها T إلى مصفوفات فرعية.
!>          تتكون المصفوفة الفرعية الأولى من الصفوف/الأعمدة 1 إلى
!>          ISPLIT( 1 )، والثانية من الصفوف/الأعمدة ISPLIT( 1 )+1
!>          حتى ISPLIT( 2 )، وهكذا.
!>          ( مصفوفة الإخراج ISPLIT من DSTEBZ متوقعة هنا. )
!>

Z

!>          Z هي مصفوفة مركبة (COMPLEX*16)، البعد (LDZ, M)
!>          المتجهات الذاتية المحسوبة.  يتم تخزين المتجه الذاتي المرتبط
!>          بالقيمة الذاتية W(i) في العمود i من
!>          Z.  يتم تعيين أي متجه يفشل في التقارب إلى تكراره الحالي
!>          بعد MAXITS تكرار.
!>          يتم تعيين الأجزاء التخيلية للمتجهات الذاتية إلى الصفر.
!>

LDZ

!>          LDZ هو عدد صحيح (INTEGER)
!>          البعد الرئيسي للمصفوفة Z.  LDZ >= max(1,N).
!>

WORK

!>          WORK هي مصفوفة دقة مزدوجة (DOUBLE PRECISION)، البعد (5*N)
!>

IWORK

!>          IWORK هي مصفوفة أعداد صحيحة (INTEGER)، البعد (N)
!>

IFAIL

!>          IFAIL هي مصفوفة أعداد صحيحة (INTEGER)، البعد (M)
!>          عند الخروج العادي، تكون جميع عناصر IFAIL صفرًا.
!>          إذا فشل متجه ذاتي واحد أو أكثر في التقارب بعد
!>          MAXITS تكرار، فسيتم تخزين مؤشراتها في
!>          المصفوفة IFAIL.
!>

INFO

!>          INFO هو عدد صحيح (INTEGER)
!>          = 0: خروج ناجح
!>          < 0: إذا كان INFO = -i، فإن المعلمة i كانت لها قيمة غير قانونية
!>          > 0: إذا كان INFO = i، فإن i متجه ذاتي فشل في التقارب خلال
!>               MAXITS تكرار.  يتم تخزين مؤشراتها في
!>               المصفوفة IFAIL.
!>

المعلمات الداخلية:

!>  MAXITS  INTEGER، الافتراضي = 5
!>          الحد الأقصى لعدد التكرارات المنفذة.
!>
!>  EXTRA   INTEGER، الافتراضي = 2
!>          عدد التكرارات المنفذة بعد تلبية معيار نمو المعيار، يجب أن يكون 1 على الأقل.
!>
