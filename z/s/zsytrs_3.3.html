subroutine zsytrs_3 (character uplo, integer n, integer nrhs, complex*16, dimension( lda, * ) a, integer lda, complex*16, dimension( * ) e, integer, dimension( * ) ipiv, complex*16, dimension( ldb, * ) b, integer ldb, integer info)

ZSYTRS_3

الغرض:

!> ZSYTRS_3 تحل نظام المعادلات الخطية A * X = B مع مصفوفة متناظرة مركبة A باستخدام التحليل الذي تم حسابه
!> بواسطة ZSYTRF_RK أو ZSYTRF_BK:
!>
!>    A = P*U*D*(U**T)*(P**T) أو A = P*L*D*(L**T)*(P**T),
!>
!> حيث U (أو L) مصفوفة مثلثية عليا (أو سفلى) وحدوية،
!> U**T (أو L**T) هي منقولة U (أو L)، P هي مصفوفة تبديل، P**T هي منقولة P، و D مصفوفة متناظرة وقطرية متدرجة مع كتل 1x1 و 2x2.
!>
!> هذه الخوارزمية تستخدم BLAS المستوى 3.
!>

المعاملات

UPLO 
!>          UPLO هو CHARACTER*1
!>          يحدد ما إذا كانت تفاصيل التحليل مخزنة كمصفوفة مثلثية عليا أو سفلى:
!>          = 'U':  مثلثية عليا، الشكل هو A = P*U*D*(U**T)*(P**T);
!>          = 'L':  مثلثية سفلى، الشكل هو A = P*L*D*(L**T)*(P**T).
!>

N

!>          N هو INTEGER
!>          رتبة المصفوفة A.  N >= 0.
!>

NRHS

!>          NRHS هو INTEGER
!>          عدد الأطراف اليمنى، أي عدد أعمدة المصفوفة B.  NRHS >= 0.
!>

A

!>          A هو مصفوفة COMPLEX*16، البعد (LDA,N)
!>          قطر المصفوفة القطرية المتدرجة المتناظرة D وعوامل U أو L
!>          كما تم حسابها بواسطة ZSYTRF_RK و ZSYTRF_BK:
!>            أ) عناصر القطر فقط للمصفوفة القطرية المتدرجة المتناظرة D
!>               على قطر A، أي D(k,k) = A(k,k);
!>               (يجب توفير عناصر القطر العلوي (أو السفلي) لـ D
!>                في المدخلات في المصفوفة E)، و
!>            ب) إذا كان UPLO = 'U': العامل U في الجزء العلوي من A.
!>               إذا كان UPLO = 'L': العامل L في الجزء السفلي من A.
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرائد للمصفوفة A.  LDA >= max(1,N).
!>

E

!>          E هو مصفوفة COMPLEX*16، البعد (N)
!>          عند الدخول، تحتوي على عناصر القطر العلوي (أو السفلي)
!>          للمصفوفة القطرية المتدرجة المتناظرة D
!>          مع كتل 1x1 أو 2x2، حيث
!>          إذا كان UPLO = 'U': E(i) = D(i-1,i),i=2:N, E(1) غير مستخدمة؛
!>          إذا كان UPLO = 'L': E(i) = D(i+1,i),i=1:N-1, E(N) غير مستخدمة.
!>
!>          ملاحظة: بالنسبة لكتلة قطرية 1x1 D(k)، حيث
!>          1 <= k <= N، العنصر E(k) غير مستخدم في كلتا
!>          حالتين UPLO = 'U' أو UPLO = 'L'.
!>

IPIV

!>          IPIV هو مصفوفة INTEGER، البعد (N)
!>          تفاصيل التبادلات وهيكل الكتلة لـ D
!>          كما تم تحديده بواسطة ZSYTRF_RK أو ZSYTRF_BK.
!>

B

!>          B هو مصفوفة COMPLEX*16، البعد (LDB,NRHS)
!>          عند الدخول، مصفوفة الطرف الأيمن B.
!>          عند الخروج، مصفوفة الحل X.
!>

LDB

!>          LDB هو INTEGER
!>          البعد الرائد للمصفوفة B.  LDB >= max(1,N).
!>

INFO

!>          INFO هو INTEGER
!>          = 0:  انتهاء ناجح
!>          < 0:  إذا كان INFO = -i، فإن المعامل i كان له قيمة غير قانونية
!>
