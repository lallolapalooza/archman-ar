subroutine zspsv (character uplo, integer n, integer nrhs, complex*16, dimension( * ) ap, integer, dimension( * ) ipiv, complex*16, dimension( ldb, * ) b, integer ldb, integer info)

zspsv تحسب حل نظام المعادلات الخطية A * X = B للمصفوفات الأخرى

الغرض:

!>
!> zspsv تحسب حل نظام معادلات خطية معقد
!>    A * X = B,
!> حيث A مصفوفة متماثلة بحجم N × N مخزنة بتنسيق مضغوط و X و B مصفوفتان بحجم N × NRHS.
!>
!> تستخدم طريقة المحورية القطرية لتحليل A إلى:
!>    A = U * D * U**T، إذا كان UPLO = 'U'، أو
!>    A = L * D * L**T، إذا كان UPLO = 'L'،
!> حيث U (أو L) هو حاصل ضرب مصفوفة تبديلية ومصفوفة مثلثة عليا (سفلى) وحدوية.
!> D مصفوفة متماثلة وقطرية جزئية بكتل قطرية بحجم 1 × 1 و 2 × 2. ثم يستخدم الشكل المحلل لـ A
!> لحل نظام المعادلات A * X = B.
!>

المعلمات

UPLO 
!>          UPLO هو CHARACTER*1
!>          = 'U': يتم تخزين المثلث العلوي لـ A؛
!>          = 'L': يتم تخزين المثلث السفلي لـ A؛
!>

N

!>          N هو INTEGER
!>          عدد المعادلات الخطية، أي ترتيب المصفوفة A. N >= 0.
!>

NRHS

!>          NRHS هو INTEGER
!>          عدد الأطراف اليمنى، أي عدد أعمدة المصفوفة B. NRHS >= 0.
!>

AP

!>          AP هي مصفوفة COMPLEX*16، بأبعاد (N*(N+1)/2)
!>          عند الإدخال، المثلث العلوي أو السفلي للمصفوفة المتماثلة A، مضغوط عموديًا في مصفوفة خطية.
!>          يتم تخزين العمود j للمصفوفة A في المصفوفة AP كما يلي:
!>          إذا كان UPLO = 'U'، AP(i + (j-1)*j/2) = A(i,j) لـ 1<=i<=j؛
!>          إذا كان UPLO = 'L'، AP(i + (j-1)*(2n-j)/2) = A(i,j) لـ j<=i<=n.
!>          انظر أدناه لمزيد من التفاصيل.
!>
!>          عند الخروج، مصفوفة القطر الجزئي D والمعاملات المستخدمة للحصول على العامل U أو L
!>          من التحليل A = U*D*U**T أو A = L*D*L**T كما تم حسابها بواسطة ZSPTRF،
!>          مخزنة كمصفوفة مثلثة مضغوطة بنفس تنسيق التخزين مثل A.
!>

IPIV

!>          IPIV هي مصفوفة INTEGER، بأبعاد (N)
!>          تفاصيل التبادلات وهيكل الكتلة لـ D، كما تم تحديدها بواسطة ZSPTRF.
!>          إذا كان IPIV(k) > 0، فقد تم تبديل الصفوف والأعمدة k و IPIV(k)، و D(k,k) هي كتلة قطرية 1 × 1.
!>          إذا كان UPLO = 'U' و IPIV(k) = IPIV(k-1) < 0، فقد تم تبديل الصفوف والأعمدة k-1 و -IPIV(k)،
!>          و D(k-1:k,k-1:k) هي كتلة قطرية 2 × 2.
!>          إذا كان UPLO = 'L' و IPIV(k) = IPIV(k+1) < 0، فقد تم تبديل الصفوف والأعمدة k+1 و -IPIV(k)،
!>          و D(k:k+1,k:k+1) هي كتلة قطرية 2 × 2.
!>

B

!>          B هي مصفوفة COMPLEX*16، بأبعاد (LDB,NRHS)
!>          عند الإدخال، مصفوفة الطرف الأيمن B بحجم N × NRHS.
!>          عند الخروج، إذا كان INFO = 0، مصفوفة الحل X بحجم N × NRHS.
!>

LDB

!>          LDB هو INTEGER
!>          البعد الرئيسي للمصفوفة B. LDB >= max(1,N).
!>

INFO

!>          INFO هو INTEGER
!>          = 0: الخروج ناجح
!>          < 0: إذا كان INFO = -i، فإن الوسيط i كان له قيمة غير قانونية
!>          > 0: إذا كان INFO = i، فإن D(i,i) يساوي صفرًا تمامًا. تم إكمال التحليل،
!>               ولكن المصفوفة القطرية الجزئية D منفردة تمامًا، لذلك لم يكن من الممكن حساب الحل.
!>
