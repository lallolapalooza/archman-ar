subroutine zsyequb (character uplo, integer n, complex*16, dimension( lda, * ) a, integer lda, double precision, dimension( * ) s, double precision scond, double precision amax, complex*16, dimension( * ) work, integer info)

ZSYEQUB

الغرض:

!>
!> ZSYEQUB يحسب عوامل قياس للصف والأعمدة المقصود بها موازنة مصفوفة متماثلة A
!> (بالنسبة لمعيار إقليدس) وتقليل رقم شرطها. يتم حساب عوامل القياس S بواسطة
!> خوارزمية BIN (انظر المراجع) بحيث تكون المصفوفة المقاسة B ذات العناصر
!> B(i,j) = S(i)*A(i,j)*S(j) لها رقم شرط ضمن عامل N من
!> أقل رقم شرط ممكن على جميع مقاييس الأقطار الممكنة.
!>

المعاملات

UPLO 
!>          UPLO هو CHARACTER*1
!>          = 'U':  المثلث العلوي من A مخزن؛
!>          = 'L':  المثلث السفلي من A مخزن.
!>

N

!>          N هو INTEGER
!>          ترتيب المصفوفة A. N >= 0.
!>

A

!>          A هي مصفوفة COMPLEX*16، ذات أبعاد (LDA,N)
!>          المصفوفة المتماثلة N×N التي سيتم حساب عوامل القياس لها.
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرئيسي للمصفوفة A. LDA >= max(1,N).
!>

S

!>          S هي مصفوفة DOUBLE PRECISION، ذات أبعاد (N)
!>          إذا كان INFO = 0، تحتوي S على عوامل القياس للمصفوفة A.
!>

SCOND

!>          SCOND هي DOUBLE PRECISION
!>          إذا كان INFO = 0، تحتوي S على نسبة أصغر S(i) إلى
!>          أكبر S(i). إذا كان SCOND >= 0.1 وكان AMAX ليس كبيرًا جدًا أو صغيرًا جدًا،
!>          فلا داعي للقياس باستخدام S.
!>

AMAX

!>          AMAX هي DOUBLE PRECISION
!>          أكبر قيمة مطلقة لأي عنصر في المصفوفة. إذا كان AMAX قريبًا جدًا
!>          من تجاوز سعة التخزين أو قريبًا جدًا من النقصان، فيجب قياس
!>          المصفوفة.
!>

WORK

!>          WORK هي مصفوفة COMPLEX*16، ذات أبعاد (2*N)
!>

INFO

!>          INFO هو INTEGER
!>          = 0:  انتهاء ناجح
!>          < 0:  إذا كان INFO = -i، فإن المعامل i له قيمة غير قانونية
!>          > 0:  إذا كان INFO = i، فإن عنصر القطر i غير موجب.
!>
