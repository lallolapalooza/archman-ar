توثيق الوظيفة/الروتين الفرعي

subroutine zsgt01 (integer itype, character uplo, integer n, integer m, complex*16, dimension( lda, * ) a, integer lda, complex*16, dimension( ldb, * ) b, integer ldb, complex*16, dimension( ldz, * ) z, integer ldz, double precision, dimension( * ) d, complex*16, dimension( * ) work, double precision, dimension( * ) rwork, double precision, dimension( * ) result)

ZSGT01

الغرض:

!>
!> CDGT01 يتحقق من تحليل بالشكل
!>
!>    A Z   =  B Z D أو
!>    A B Z =  Z D أو
!>    B A Z =  Z D
!>
!> حيث A مصفوفة هيرميتية، B مصفوفة هيرميتية موجبة قطعية،
!> Z مصفوفة وحدوية، و D مصفوفة قطرية.
!>
!> يتم حساب إحدى نسب الاختبار التالية:
!>
!> ITYPE = 1:  RESULT(1) = | A Z - B Z D | / ( |A| |Z| n ulp )
!>
!> ITYPE = 2:  RESULT(1) = | A B Z - Z D | / ( |A| |Z| n ulp )
!>
!> ITYPE = 3:  RESULT(1) = | B A Z - Z D | / ( |A| |Z| n ulp )
!>

المعلمات

ITYPE 
!>          ITYPE هو عدد صحيح
!>          شكل مشكلة القيمة الذاتية المعممة الهيرميتية.
!>          = 1:  A*z = (lambda)*B*z
!>          = 2:  A*B*z = (lambda)*z
!>          = 3:  B*A*z = (lambda)*z
!>

UPLO

!>          UPLO هو حرف واحد
!>          يحدد ما إذا كان الجزء العلوي أو السفلي المثلثي للمصفوفات
!>          الهيرميتية A و B مخزنًا.
!>          = 'U':  مثلثي علوي
!>          = 'L':  مثلثي سفلي
!>

N

!>          N هو عدد صحيح
!>          رتبة المصفوفة A.  N >= 0.
!>

M

!>          M هو عدد صحيح
!>          عدد القيم الذاتية التي تم العثور عليها.  M >= 0.
!>

A

!>          A هي مصفوفة COMPLEX*16، أبعاد (LDA, N)
!>          المصفوفة الهيرميتية الأصلية A.
!>

LDA

!>          LDA هو عدد صحيح
!>          البعد الرائد للمصفوفة A.  LDA >= max(1,N).
!>

B

!>          B هي مصفوفة COMPLEX*16، أبعاد (LDB, N)
!>          المصفوفة الهيرميتية الموجبة القطعية الأصلية B.
!>

LDB

!>          LDB هو عدد صحيح
!>          البعد الرائد للمصفوفة B.  LDB >= max(1,N).
!>

Z

!>          Z هي مصفوفة COMPLEX*16، أبعاد (LDZ, M)
!>          المتجهات الذاتية المحسوبة للمشكلة الذاتية المعممة.
!>

LDZ

!>          LDZ هو عدد صحيح
!>          البعد الرائد للمصفوفة Z.  LDZ >= max(1,N).
!>

D

!>          D هي مصفوفة DOUBLE PRECISION، أبعاد (M)
!>          القيم الذاتية المحسوبة للمشكلة الذاتية المعممة.
!>

WORK

!>          WORK هي مصفوفة COMPLEX*16، أبعاد (N*N)
!>

RWORK

!>          RWORK هي مصفوفة DOUBLE PRECISION، أبعاد (N)
!>

RESULT

!>          RESULT هي مصفوفة DOUBLE PRECISION، أبعاد (1)
!>          نسبة الاختبار كما هو موضح أعلاه.
!>
