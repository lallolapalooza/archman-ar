subroutine zsycon (character uplo, integer n, complex*16, dimension( lda, * ) a, integer lda, integer, dimension( * ) ipiv, double precision anorm, double precision rcond, complex*16, dimension( * ) work, integer info)

ZSYCON

الغرض:

!>
!> ZSYCON تقدّر مقلوب رقم الشرط (بالنسبة للقاعدة 1) لمصفوفة متناظرة مركبة A باستخدام التحلل
!> A = U*D*U**T أو A = L*D*L**T المحسوب بواسطة ZSYTRF.
!>
!> يتم الحصول على تقدير لـ norm(inv(A))، ويتم حساب مقلوب رقم الشرط كـ RCOND = 1 / (ANORM * norm(inv(A))).
!>

المعلمات

UPLO 
!>          UPLO هو CHARACTER*1
!>          يحدد ما إذا كانت تفاصيل التحلل مخزنة
!>          كمصفوفة مثلثية عليا أو سفلى.
!>          = 'U':  مثلثية عليا، الشكل هو A = U*D*U**T؛
!>          = 'L':  مثلثية سفلى، الشكل هو A = L*D*L**T.
!>

N

!>          N هو INTEGER
!>          رتبة المصفوفة A.  N >= 0.
!>

A

!>          A هي مصفوفة COMPLEX*16، البعد (LDA,N)
!>          المصفوفة القطرية القطاعية D والعوامل المضاعفة المستخدمة
!>          للحصول على العامل U أو L كما تم حسابه بواسطة ZSYTRF.
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرئيسي للمصفوفة A.  LDA >= max(1,N).
!>

IPIV

!>          IPIV هي مصفوفة INTEGER، البعد (N)
!>          تفاصيل التبادلات وهيكل القطاع لـ D
!>          كما تم تحديده بواسطة ZSYTRF.
!>

ANORM

!>          ANORM هو DOUBLE PRECISION
!>          معيار القاعدة 1 للمصفوفة الأصلية A.
!>

RCOND

!>          RCOND هو DOUBLE PRECISION
!>          مقلوب رقم الشرط للمصفوفة A،
!>          محسوب كـ RCOND = 1/(ANORM * AINVNM)، حيث AINVNM هو تقدير
!>          لمعيار القاعدة 1 لـ inv(A) محسوب في هذا الإجراء.
!>

WORK

!>          WORK هي مصفوفة COMPLEX*16، البعد (2*N)
!>

INFO

!>          INFO هو INTEGER
!>          = 0:  انتهاء ناجح
!>          < 0:  إذا كان INFO = -i، فإن المعلمة i-th كانت لها قيمة غير قانونية
!>
