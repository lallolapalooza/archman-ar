subroutine zsptrf (character uplo, integer n, complex*16, dimension( * ) ap, integer, dimension( * ) ipiv, integer info)

ZSPTRF

الغرض:

!>
!> ZSPTRF يحسب تحليل مصفوفة متناظرة مركبة A مخزنة بتنسيق مكدس باستخدام طريقة
!> تبديل المحاور القطري لبنش-كوفمان:
!>
!>    A = U*D*U**T  أو  A = L*D*L**T
!>
!> حيث U (أو L) هو حاصل ضرب مصفوفات الوحدة المثلثية العلوية (السفلية) ومصفوفات التبديل،
!> و D مصفوفة متناظرة وقطرية كتلة ذات كتل قطرية بحجم 1x1 و 2x2.
!>

المعلمات

UPLO 
!>          UPLO هو CHARACTER*1
!>          = 'U':  المثلث العلوي لـ A مخزن؛
!>          = 'L':  المثلث السفلي لـ A مخزن.
!>

N

!>          N هو INTEGER
!>          ترتيب المصفوفة A. N >= 0.
!>

AP

!>          AP هو مصفوفة COMPLEX*16، البعد (N*(N+1)/2)
!>          عند الإدخال، المثلث العلوي أو السفلي للمصفوفة المتناظرة A،
!>          مكدسة عمودياً في مصفوفة خطية. يتم تخزين العمود j من A
!>          في المصفوفة AP كما يلي:
!>          إذا كان UPLO = 'U'، فإن AP(i + (j-1)*j/2) = A(i,j) لـ 1<=i<=j؛
!>          إذا كان UPLO = 'L'، فإن AP(i + (j-1)*(2n-j)/2) = A(i,j) لـ j<=i<=n.
!>
!>          عند الخروج، المصفوفة القطرية الكتلة D والمعاملات المستخدمة
!>          للحصول على العامل U أو L، مخزنة كمصفوفة مثلثية مكدسة
!>          تحل محل A (انظر أدناه لمزيد من التفاصيل).
!>

IPIV

!>          IPIV هو مصفوفة INTEGER، البعد (N)
!>          تفاصيل التبادلات وهيكل الكتلة لـ D.
!>          إذا كان IPIV(k) > 0، فإن الصفوف والأعمدة k و IPIV(k) تم تبديلها
!>          و D(k,k) هي كتلة قطرية بحجم 1x1.
!>          إذا كان UPLO = 'U' وكان IPIV(k) = IPIV(k-1) < 0، فإن الصفوف والأعمدة k-1 و -IPIV(k) تم تبديلها
!>          و D(k-1:k,k-1:k) هي كتلة قطرية بحجم 2x2. إذا كان UPLO = 'L' وكان IPIV(k) =
!>          IPIV(k+1) < 0، فإن الصفوف والأعمدة k+1 و -IPIV(k) تم تبديلها
!>          و D(k:k+1,k:k+1) هي كتلة قطرية بحجم 2x2.
!>

INFO

!>          INFO هو INTEGER
!>          = 0: خروج ناجح
!>          < 0: إذا كان INFO = -i، فإن المعلمة i كانت ذات قيمة غير قانونية
!>          > 0: إذا كان INFO = i، فإن D(i,i) يساوي صفر تماماً. تم إكمال التحليل،
!>               ولكن المصفوفة القطرية الكتلة D مفردة تماماً، وسيحدث قسمة على صفر إذا
!>               تم استخدامها لحل نظام معادلات.
!>

!>         Company
!>
!>  إذا كان UPLO = 'U'، فإن A = U*D*U**T، حيث
!>     U = P(n)*U(n)* ... *P(k)U(k)* ...,
!>  أي، U هو حاصل ضرب الحدود P(k)*U(k)، حيث k ينقص من n إلى
!>  1 بخطوات 1 أو 2، و D هي مصفوفة قطرية كتلة ذات كتل قطرية 1x1
!>  و 2x2 D(k). P(k) هي مصفوفة تبديل كما
!>  تم تعريفها بواسطة IPIV(k)، و U(k) هي مصفوفة وحدة مثلثية علوية، بحيث
!>  إذا كانت الكتلة القطرية D(k) من الرتبة s (s = 1 أو 2)، فإن
!>
!>             (   I    v    0   )   k-s
!>     U(k) =  (   0    I    0   )   s
!>             (   0    0    I   )   n-k
!>                k-s   s   n-k
!>
!>  إذا كان s = 1، فإن D(k) تحل محل A(k,k)، و v تحل محل A(1:k-1,k).
!>  إذا كان s = 2، فإن المثلث العلوي لـ D(k) يحل محل A(k-1,k-1)، A(k-1,k)،
!>  و A(k,k)، و v يحل محل A(1:k-2,k-1:k).
!>
!>  إذا كان UPLO = 'L'، فإن A = L*D*L**T، حيث
!>     L = P(1)*L(1)* ... *P(k)*L(k)* ...,
!>  أي، L هو حاصل ضرب الحدود P(k)*L(k)، حيث k يزيد من 1 إلى
!>  n بخطوات 1 أو 2، و D هي مصفوفة قطرية كتلة ذات كتل قطرية 1x1
!>  و 2x2 D(k). P(k) هي مصفوفة تبديل كما
!>  تم تعريفها بواسطة IPIV(k)، و L(k) هي مصفوفة وحدة مثلثية سفلية، بحيث
!>  إذا كانت الكتلة القطرية D(k) من الرتبة s (s = 1 أو 2)، فإن
!>
!>             (   I    0     0   )  k-1
!>     L(k) =  (   0    I     0   )  s
!>             (   0    v     I   )  n-k-s+1
!>                k-1   s  n-k-s+1
!>
!>  إذا كان s = 1، فإن D(k) تحل محل A(k,k)، و v تحل محل A(k+1:n,k).
!>  إذا كان s = 2، فإن المثلث السفلي لـ D(k) يحل محل A(k,k)، A(k+1,k)،
!>  و A(k+1,k+1)، و v يحل محل A(k+2:n,k:k+1).
!>
