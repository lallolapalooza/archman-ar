توثيق الوظيفة/الروتين الفرعي

subroutine zsycon (character uplo, integer n, complex*16, dimension( lda, * ) a, integer lda, integer, dimension( * ) ipiv, double precision anorm, double precision rcond, complex*16, dimension( * ) work, integer info)

ZSYCON

الغرض:

!>
!> ZSYCON تقدر المقلوب لرقم الحالة (بمعيار 1-norm) لمصفوفة متناظرة مركبة A
!> باستخدام التحليل A = U*D*U**T أو A = L*D*L**T الذي تم حسابه بواسطة ZSYTRF.
!>
!> يتم الحصول على تقدير لـ norm(inv(A))، ويتم حساب مقلوب رقم الحالة
!> كـ RCOND = 1 / (ANORM * norm(inv(A))).
!>

المعلمات

UPLO 
!>          UPLO هو CHARACTER*1
!>          يحدد ما إذا كانت تفاصيل التحليل مخزنة
!>          كمصفوفة مثلثة عليا أو سفلى.
!>          = 'U':  مثلثة عليا، الشكل هو A = U*D*U**T;
!>          = 'L':  مثلثة سفلى، الشكل هو A = L*D*L**T.
!>

N

!>          N هو INTEGER
!>          ترتيب المصفوفة A.  N >= 0.
!>

A

!>          A هو مصفوفة COMPLEX*16، البعد (LDA,N)
!>          المصفوفة القطرية الفرعية D والمضاعفات المستخدمة
!>          للحصول على العامل U أو L كما تم حسابه بواسطة ZSYTRF.
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرائد للمصفوفة A.  LDA >= max(1,N).
!>

IPIV

!>          IPIV هو مصفوفة INTEGER، البعد (N)
!>          تفاصيل التبادلات وهيكل D الفرعي
!>          كما تم تحديده بواسطة ZSYTRF.
!>

ANORM

!>          ANORM هو DOUBLE PRECISION
!>          معيار 1-norm للمصفوفة الأصلية A.
!>

RCOND

!>          RCOND هو DOUBLE PRECISION
!>          مقلوب رقم الحالة للمصفوفة A،
!>          تم حسابه كـ RCOND = 1/(ANORM * AINVNM)، حيث AINVNM هو
!>          تقدير لمعيار 1-norm للمصفوفة المعكوسة AINV(A) والذي تم حسابه في هذا الروتين.
!>

WORK

!>          WORK هي مصفوفة COMPLEX*16، البعد (2*N)
!>

INFO

!>          INFO هو INTEGER
!>          = 0:  انتهاء ناجح
!>          < 0:  إذا كان INFO = -i، فإن المعلمة i كانت لها قيمة غير قانونية
!>
