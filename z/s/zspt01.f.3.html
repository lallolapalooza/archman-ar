توثيق الدالة/الروتين الفرعي

subroutine zspt01 (character uplo, integer n, complex*16, dimension( * ) a, complex*16, dimension( * ) afac, integer, dimension( * ) ipiv, complex*16, dimension( ldc, * ) c, integer ldc, double precision, dimension( * ) rwork, double precision resid)

ZSPT01

الغرض:

!>
!> ZSPT01 يعيد بناء مصفوفة متماثلة غير محددة A من تجزئة المحورية القطرية الخاصة بها A = U*D*U' أو A = L*D*L' ويحسب
!> الباقي
!>    norm( C - A ) / ( N * norm(A) * EPS ),
!> حيث C هي المصفوفة المعاد بناؤها و EPS هي وحدة الآلة.
!>

المعلمات

UPLO 
!>          UPLO هو CHARACTER*1
!>          يحدد ما إذا كان الجزء العلوي أو السفلي المثلثي للمصفوفة المتماثلة
!>          A مخزن:
!>          = 'U':  مثلثي علوي
!>          = 'L':  مثلثي سفلي
!>

N

!>          N هو INTEGER
!>          رتبة المصفوفة A. N >= 0.
!>

A

!>          A هو مصفوفة COMPLEX*16، البعد (N*(N+1)/2)
!>          المصفوفة المتماثلة الأصلية A، مخزنة كمصفوفة
!>          مثلثية مكدسة.
!>

AFAC

!>          AFAC هو مصفوفة COMPLEX*16، البعد (N*(N+1)/2)
!>          الشكل المجزأ للمصفوفة A، مخزن كمصفوفة
!>          مثلثية مكدسة. تحتوي AFAC على مصفوفة D القطرية الكتلية
!>          والمضاعفات المستخدمة للحصول على العامل L أو U من
!>          تجزئة L*D*L' أو U*D*U' كما تم حسابها بواسطة ZSPTRF.
!>

IPIV

!>          IPIV هي مصفوفة INTEGER، البعد (N)
!>          مؤشرات المحورية من ZSPTRF.
!>

C

!>          C هي مصفوفة COMPLEX*16، البعد (LDC,N)
!>

LDC

!>          LDC هو INTEGER
!>          البعد الرائد للمصفوفة C. LDC >= max(1,N).
!>

RWORK

!>          RWORK هي مصفوفة DOUBLE PRECISION، البعد (N)
!>

RESID

!>          RESID هو DOUBLE PRECISION
!>          إذا كان UPLO = 'L'، norm(L*D*L' - A) / ( N * norm(A) * EPS )
!>          إذا كان UPLO = 'U'، norm(U*D*U' - A) / ( N * norm(A) * EPS )
!>
