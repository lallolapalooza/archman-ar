subroutine zsytri2 (character uplo, integer n, complex*16, dimension( lda, * ) a, integer lda, integer, dimension( * ) ipiv, complex*16, dimension( * ) work, integer lwork, integer info)

ZSYTRI2

الغرض:

!>
!> ZSYTRI2 تحسب معكوس المصفوفة المتماثلة غير المحددة COMPLEX*16
!> A باستخدام التحليل A = U*D*U**T أو A = L*D*L**T المحسوب بواسطة
!> ZSYTRF. ZSYTRI2 يحدد البعد الرائد لمساحة العمل
!> قبل استدعاء ZSYTRI2X الذي يحسب المعكوس فعليًا.
!>

المعلمات

UPLO 
!>          UPLO هو CHARACTER*1
!>          يحدد ما إذا كانت تفاصيل التحليل مخزنة
!>          كمصفوفة مثلثية عليا أو سفلية.
!>          = 'U':  مثلثية عليا، الشكل هو A = U*D*U**T؛
!>          = 'L':  مثلثية سفلية، الشكل هو A = L*D*L**T.
!>

N

!>          N هو INTEGER
!>          رتبة المصفوفة A.  N >= 0.
!>

A

!>          A هو مصفوفة COMPLEX*16، البعد (LDA,N)
!>          عند الإدخال، مصفوفة القطر الرئيسي الكتلة D والمضاعفات
!>          المستخدمة للحصول على العامل U أو L كما هو محسوب بواسطة ZSYTRF.
!>
!>          عند الخروج، إذا كان INFO = 0، فإن المعكوس (المتماثل) للمصفوفة الأصلية
!>          . إذا كان UPLO = 'U'، فإن الجزء المثلثي العلوي من
!>          المعكوس يتكون والجزء السفلي من A تحت القطر غير
!>          مرجعي؛ إذا كان UPLO = 'L' فإن الجزء المثلثي السفلي من
!>          المعكوس يتكون والجزء العلوي من A فوق القطر غير
!>          مرجعي.
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرائد للمصفوفة A.  LDA >= max(1,N).
!>

IPIV

!>          IPIV هو مصفوفة INTEGER، البعد (N)
!>          تفاصيل التبادلات والبنية القطاعية لـ D
!>          كما تم تحديدها بواسطة ZSYTRF.
!>

WORK

!>          WORK هو مصفوفة COMPLEX*16، البعد (N+NB+1)*(NB+3)
!>

LWORK

!>          LWORK هو INTEGER
!>          بعد المصفوفة WORK.
!>          WORK هو حجم >= (N+NB+1)*(NB+3)
!>          إذا كان LDWORK = -1، فيفترض استعلام عن مساحة العمل؛ الروتين
!>           يحسب:
!>              - الحجم الأمثل لمصفوفة WORK، يعيد
!>          هذه القيمة كإدخال أول لمصفوفة WORK،
!>              - ولا يصدر أي رسالة خطأ متعلقة بـ LDWORK بواسطة XERBLA.
!>

INFO

!>          INFO هو INTEGER
!>          = 0: خروج ناجح
!>          < 0: إذا كان INFO = -i، فإن الوسيط i كان له قيمة غير قانونية
!>          > 0: إذا كان INFO = i، D(i,i) = 0؛ المصفوفة شاذة ومعكوسها
!>               لم يتمكن من الحساب.
!>
