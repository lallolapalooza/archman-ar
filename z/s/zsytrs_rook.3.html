توثيق الوظيفة/الإجراء الفرعي

subroutine zsytrs_rook (character uplo, integer n, integer nrhs, complex*16, dimension( lda, * ) a, integer lda, integer, dimension( * ) ipiv, complex*16, dimension( ldb, * ) b, integer ldb, integer info)

ZSYTRS_ROOK

الغرض:

!>
!> ZSYTRS_ROOK يحل نظام المعادلات الخطية A*X = B
!> بمصفوفة متناظرة معقدة A باستخدام التحليل A = U*D*U**T أو
!> A = L*D*L**T الذي تم حسابه بواسطة ZSYTRF_ROOK.
!>

المعلمات

UPLO 
!>          UPLO هو CHARACTER*1
!>          يحدد ما إذا كانت تفاصيل التحليل مخزنة
!>          كمصفوفة مثلثة عليا أو سفلى.
!>          = 'U':  مثلثة عليا، الشكل هو A = U*D*U**T؛
!>          = 'L':  مثلثة سفلى، الشكل هو A = L*D*L**T.
!>

N

!>          N هو INTEGER
!>          رتبة المصفوفة A.  N >= 0.
!>

NRHS

!>          NRHS هو INTEGER
!>          عدد الأطراف اليمنى، أي عدد الأعمدة
!>          للمصفوفة B.  NRHS >= 0.
!>

A

!>          A هي مصفوفة COMPLEX*16، البعد (LDA,N)
!>          المصفوفة القطرية الجزئية D والمضاعفات المستخدمة
!>          للحصول على العامل U أو L كما تم حسابه بواسطة ZSYTRF_ROOK.
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرئيسي للمصفوفة A.  LDA >= max(1,N).
!>

IPIV

!>          IPIV هي مصفوفة INTEGER، البعد (N)
!>          تفاصيل التبادلات والهيكل الجزئي لـ D
!>          كما تم تحديده بواسطة ZSYTRF_ROOK.
!>

B

!>          B هي مصفوفة COMPLEX*16، البعد (LDB,NRHS)
!>          عند الإدخال، مصفوفة الطرف الأيمن B.
!>          عند الخروج، مصفوفة الحل X.
!>

LDB

!>          LDB هو INTEGER
!>          البعد الرئيسي للمصفوفة B.  LDB >= max(1,N).
!>

INFO

!>          INFO هو INTEGER
!>          = 0:  خروج ناجح
!>          < 0:  إذا كان INFO = -i، فإن المعلمة i كان لها قيمة غير قانونية
!>
