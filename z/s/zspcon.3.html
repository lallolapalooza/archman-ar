subroutine zspcon (character uplo, integer n, complex*16, dimension( * ) ap, integer, dimension( * ) ipiv, double precision anorm, double precision rcond, complex*16, dimension( * ) work, integer info)

ZSPCON

الغرض:

!>
!> ZSPCON تقدّر مقلوب عدد الشرط (في معيار 1) لمصفوفة A متماثلة مركبة محزمة
!> باستخدام التفكيك A = U*D*U**T أو A = L*D*L**T الذي تم حسابه بواسطة ZSPTRF.
!>
!> يتم الحصول على تقدير لـ norm(inv(A))، ويتم حساب مقلوب عدد الشرط على أنه
!> RCOND = 1 / (ANORM * norm(inv(A))).
!>

المعلمات

UPLO 
!>          UPLO is CHARACTER*1
!>          يحدد ما إذا كانت تفاصيل التفكيك مخزنة
!>          كمصفوفة مثلثية علوية أو سفلية.
!>          = 'U':  مثلثية علوية، الشكل هو A = U*D*U**T;
!>          = 'L':  مثلثية سفلية، الشكل هو A = L*D*L**T.
!>

N

!>          N is INTEGER
!>          رتبة المصفوفة A.  N >= 0.
!>

AP

!>          AP is COMPLEX*16 array, dimension (N*(N+1)/2)
!>          المصفوفة القطرية الجزئية D والمضاعفات المستخدمة
!>          للحصول على العامل U أو L كما تم حسابه بواسطة ZSPTRF، مخزنة كـ
!>          مصفوفة مثلثية محزمة.
!>

IPIV

!>          IPIV is INTEGER array, dimension (N)
!>          تفاصيل التبادلات وهيكل D الجزئي
!>          كما تم تحديده بواسطة ZSPTRF.
!>

ANORM

!>          ANORM is DOUBLE PRECISION
!>          معيار 1 للمصفوفة الأصلية A.
!>

RCOND

!>          RCOND is DOUBLE PRECISION
!>          مقلوب عدد الشرط للمصفوفة A،
!>          محسوب كـ RCOND = 1/(ANORM * AINVNM)، حيث AINVNM هو
!>          تقدير لمعيار 1 لـ inv(A) الذي تم حسابه في هذا الإجراء.
!>

WORK

!>          WORK is COMPLEX*16 array, dimension (2*N)
!>

INFO

!>          INFO is INTEGER
!>          = 0:  انتهاء ناجح
!>          < 0:  إذا كان INFO = -i، فإن المعلمة i كانت لها قيمة غير صالحة
!>
