توثيق الوظيفة/الإجراء الفرعي

subroutine zsymv (character uplo, integer n, complex*16 alpha, complex*16, dimension( lda, * ) a, integer lda, complex*16, dimension( * ) x, integer incx, complex*16 beta, complex*16, dimension( * ) y, integer incy)

ZSYMV يحسب حاصل ضرب مصفوفة-متجه لمصفوفة متماثلة معقدة.

الغرض:

!>
!> ZSYMV  ينفذ عملية المصفوفة-المتجه
!>
!>    y := alpha*A*x + beta*y,
!>
!> حيث alpha و beta قيم عددية، x و y متجهان يتكونان من n عنصر،
!> و A مصفوفة متماثلة بحجم n × n.
!>

المعلمات

UPLO 
!>          UPLO هو CHARACTER*1
!>           عند الدخول، تحدد UPLO ما إذا كان الجزء المثلثي العلوي أو السفلي
!>           من المصفوفة A سيتم الرجوع إليه على النحو التالي:
!>
!>              UPLO = 'U' أو 'u'   يتم الرجوع فقط إلى الجزء المثلثي العلوي من A.
!>
!>              UPLO = 'L' أو 'l'   يتم الرجوع فقط إلى الجزء المثلثي السفلي من A.
!>
!>           لا يتغير عند الخروج.
!>

N

!>          N هو INTEGER
!>           عند الدخول، يحدد N رتبة المصفوفة A.
!>           يجب أن يكون N صفرًا على الأقل.
!>           لا يتغير عند الخروج.
!>

ALPHA

!>          ALPHA هو COMPLEX*16
!>           عند الدخول، يحدد ALPHA القيمة العددية alpha.
!>           لا يتغير عند الخروج.
!>

A

!>          A هي مصفوفة COMPLEX*16، بأبعاد ( LDA, N )
!>           قبل الدخول، مع UPLO = 'U' أو 'u'، يجب أن يحتوي الجزء المثلثي العلوي
!>           الرئيسي بحجم n × n من المصفوفة A على الجزء المثلثي العلوي
!>           للمصفوفة المتماثلة، ولا يتم الرجوع إلى الجزء المثلثي السفلي الصارم من A.
!>           قبل الدخول، مع UPLO = 'L' أو 'l'، يجب أن يحتوي الجزء المثلثي السفلي
!>           الرئيسي بحجم n × n من المصفوفة A على الجزء المثلثي السفلي
!>           للمصفوفة المتماثلة، ولا يتم الرجوع إلى الجزء المثلثي العلوي الصارم من A.
!>           لا يتغير عند الخروج.
!>

LDA

!>          LDA هو INTEGER
!>           عند الدخول، يحدد LDA البعد الأول لـ A كما هو معلن
!>           في البرنامج المستدعي (الإجراء الفرعي). يجب أن يكون LDA على الأقل
!>           max( 1, N ).
!>           لا يتغير عند الخروج.
!>

X

!>          X هي مصفوفة COMPLEX*16، بأبعاد لا تقل عن
!>           ( 1 + ( N - 1 )*abs( INCX ) ).
!>           قبل الدخول، يجب أن تحتوي المصفوفة X المتزايدة على المتجه x
!>           المكون من N عنصر.
!>           لا يتغير عند الخروج.
!>

INCX

!>          INCX هو INTEGER
!>           عند الدخول، يحدد INCX الزيادة لعناصر X. يجب ألا يكون INCX صفرًا.
!>           لا يتغير عند الخروج.
!>

BETA

!>          BETA هو COMPLEX*16
!>           عند الدخول، يحدد BETA القيمة العددية beta. عندما يتم
!>           توريد BETA بصفر، لا يلزم تعيين Y عند الإدخال.
!>           لا يتغير عند الخروج.
!>

Y

!>          Y هي مصفوفة COMPLEX*16، بأبعاد لا تقل عن
!>           ( 1 + ( N - 1 )*abs( INCY ) ).
!>           قبل الدخول، يجب أن تحتوي المصفوفة Y المتزايدة على المتجه y
!>           المكون من N عنصر. عند الخروج، يتم استبدال Y بالمتجه y المحدث.
!>

INCY

!>          INCY هو INTEGER
!>           عند الدخول، يحدد INCY الزيادة لعناصر Y. يجب ألا يكون INCY صفرًا.
!>           لا يتغير عند الخروج.
!>
