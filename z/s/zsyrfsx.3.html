ZSYRFSX

الغرض:

!>
!>    ZSYRFSX يحسن الحل المحسوب لنظام المعادلات الخطية
!>    عندما تكون مصفوفة المعاملات متناظرة وغير محددة، و
!>    يوفر حدود الخطأ وتقديرات الخطأ الرجعي لـ
!>    الحل. بالإضافة إلى حد خطأ المعيار، يقوم البرنامج بتوفير
!>    حد خطأ أقصى مكوني إذا كان ذلك ممكناً. انظر التعليقات لـ
!>    ERR_BNDS_NORM و ERR_BNDS_COMP للحصول على تفاصيل حدود الخطأ.
!>
!>    قد تم موازنة نظام المعادلات الأصلي
!>    قبل استدعاء هذا الروتين، كما هو موضح بواسطة الوسائط EQUED و S
!>    أدناه. في هذه الحالة، فإن الحل وحدود الخطأ المعادة
!>    هي للنظام الأصلي غير المتوازن.
!> 
!>     بعض المعاملات الاختيارية مجمعة في مصفوفة PARAMS. هذه
!>     الإعدادات تحدد كيفية إجراء التحسين، ولكن غالبًا ما تكون
!>     الإعدادات الافتراضية مقبولة. إذا كانت الإعدادات الافتراضية مقبولة،
!>     يمكن للمستخدمين تمرير NPARAMS = 0 مما يمنع الكود المصدري من الوصول
!>     إلى الوسيط PARAMS.
!>

المعاملات

UPLO 
!>          UPLO هو CHARACTER*1
!>       = 'U':  المثلث العلوي لـ A مخزن؛
!>       = 'L':  المثلث السفلي لـ A مخزن.
!>

EQUED

!>          EQUED هو CHARACTER*1
!>     يحدد شكل الموازنة الذي تم إجراؤه على A
!>     قبل استدعاء هذا الروتين. هذا ضروري لحساب
!>     الحل وحدود الخطأ بشكل صحيح.
!>       = 'N':  لا يوجد موازنة
!>       = 'Y':  موازنة الصفوف والأعمدة، أي تم استبدال A
!>               بـ diag(S) * A * diag(S).
!>               تم تغيير الجانب الأيمن B وفقًا لذلك.
!>

N

!>          N هو INTEGER
!>     ترتيب المصفوفة A.  N >= 0.
!>

NRHS

!>          NRHS هو INTEGER
!>     عدد الجوانب اليمنى، أي عدد أعمدة
!>     المصفوفتين B و X.  NRHS >= 0.
!>

A

!>          A هي مصفوفة COMPLEX*16، ذات بعد (LDA,N)
!>     المصفوفة المتناظرة A.  إذا كان UPLO = 'U'، فإن الجزء العلوي
!>     المثلثي من A بحجم N*N يحتوي على الجزء العلوي
!>     المثلثي للمصفوفة A، ولا يتم الرجوع إلى الجزء السفلي
!>     المثلثي الصارم لـ A.  إذا كان UPLO = 'L'، فإن الجزء السفلي
!>     المثلثي من A بحجم N*N يحتوي على الجزء السفلي
!>     المثلثي للمصفوفة A، ولا يتم الرجوع إلى الجزء العلوي
!>     المثلثي الصارم لـ A.
!>

LDA

!>          LDA هو INTEGER
!>     البعد الرئيسي للمصفوفة A.  LDA >= max(1,N).
!>

AF

!>          AF هي مصفوفة COMPLEX*16، ذات بعد (LDAF,N)
!>     الشكل المفكك للمصفوفة A.  تحتوي AF على المصفوفة القطرية
!>     الكتلية D والمضاعفات المستخدمة للحصول على
!>     المثلث U أو L من تحليل A = U*D*U**T أو A =
!>     L*D*L**T كما تم حسابه بواسطة ZSYTRF.
!>

LDAF

!>          LDAF هو INTEGER
!>     البعد الرئيسي للمصفوفة AF.  LDAF >= max(1,N).
!>

IPIV

!>          IPIV هي مصفوفة INTEGER، ذات بعد (N)
!>     تفاصيل التبادلات والبنية الكتلية لـ D
!>     كما تم تحديده بواسطة ZSYTRF.
!>

S

!>          S هي مصفوفة DOUBLE PRECISION، ذات بعد (N)
!>     عوامل القياس لـ A.  إذا كان EQUED = 'Y'، يتم ضرب A
!>     على اليسار واليمين بـ diag(S).  S هو معامل مدخل إذا كان FACT =
!>     'F'؛ بخلاف ذلك، S هو معامل مخرج.  إذا كان FACT = 'F' و EQUED
!>     = 'Y'، يجب أن يكون كل عنصر في S موجبًا.  إذا كان S مخرجًا، فإن كل
!>     عنصر في S هو قوة للجذر. إذا كان S مدخلاً، يجب أن يكون كل
!>     عنصر في S قوة للجذر لضمان حل موثوق به و
!>     تقديرات خطأ. الموازنة بقوى الجذر لا تسبب
!>     أخطاء تقريب إلا إذا كان الناتج أقل من الحد الأدنى أو أكبر من الحد الأعلى.
!>     تؤدي أخطاء التقريب أثناء الموازنة إلى التحسين بمصفوفة
!>     غير مكافئة للمصفوفة المدخلة، مما ينتج عنه تقديرات خطأ
!>     قد لا تكون موثوقة.
!>

B

!>          B هي مصفوفة COMPLEX*16، ذات بعد (LDB,NRHS)
!>     المصفوفة اليمنى B.
!>

LDB

!>          LDB هو INTEGER
!>     البعد الرئيسي للمصفوفة B.  LDB >= max(1,N).
!>

X

!>          X هي مصفوفة COMPLEX*16، ذات بعد (LDX,NRHS)
!>     عند الإدخال، مصفوفة الحل X، كما تم حسابها بواسطة ZGETRS.
!>     عند الخروج، مصفوفة الحل المحسنة X.
!>

LDX

!>          LDX هو INTEGER
!>     البعد الرئيسي للمصفوفة X.  LDX >= max(1,N).
!>

RCOND

!>          RCOND هو DOUBLE PRECISION
!>     مقلوب رقم الشرط المقياس. هذا تقدير لـ
!>     مقلوب رقم شرط Skeel للمصفوفة A بعد
!>     الموازنة (إذا تم إجراؤها). إذا كان هذا أقل من دقة الآلة
!>     (على وجه الخصوص، إذا كان صفرًا)، فإن المصفوفة مفردة
!>     لدقة العمل. لاحظ أن الخطأ قد لا يزال صغيرًا حتى
!>     إذا كان هذا الرقم صغيرًا جدًا وتبدو المصفوفة غير مستقرة.
!>

BERR

!>          BERR هي مصفوفة DOUBLE PRECISION، ذات بعد (NRHS)
!>     الخطأ الرجعي النسبي المكوني. هذا هو
!>     الخطأ الرجعي النسبي المكوني لكل متجه حل X(j)
!>     (أي، أصغر تغيير نسبي في أي عنصر في A أو B يجعل X(j)
!>     حلاً دقيقًا).
!>

N_ERR_BNDS

!>          N_ERR_BNDS هو INTEGER
!>     عدد حدود الخطأ المطلوب إرجاعها لكل جانب أيمن
!>     ولكل نوع (معيار أو مكوني). انظر ERR_BNDS_NORM و
!>     ERR_BNDS_COMP أدناه.
!>

ERR_BNDS_NORM

!>          ERR_BNDS_NORM هي مصفوفة DOUBLE PRECISION، ذات بعد (NRHS, N_ERR_BNDS)
!>     لكل جانب أيمن، تحتوي هذه المصفوفة على معلومات حول
!>     حدود الخطأ المختلفة وأرقام الشرط المقابلة لـ
!>     الخطأ النسبي المعياري، والذي يتم تعريفه على النحو التالي:
!>
!>     الخطأ النسبي المعياري في متجه الحل i:
!>             max_j (abs(XTRUE(j,i) - X(j,i)))
!>            ------------------------------
!>                  max_j abs(X(j,i))
!>
!>     المصفوفة i

Function/Subroutine Documentation

FERR

!>          FERR is DOUBLE PRECISION array, dimension (NRHS)
!>     The estimated forward error, each entry FERR(i) is an estimate
!>     of the relative error in the ith solution computed by this
!>     routine, FERR(i) = (norm(X - XTRUE, inf)) / (norm(X, inf)).
!>
!>     FERR is only computed if it is requested by the value of
!>     PARAMS(LA_LINRX_ITREF_I).
!>
!>     See Lapack Working Note 165 for further details and extra
!>     cautions.
!>

BERR

!>          BERR is DOUBLE PRECISION array, dimension (NRHS)
!>     The componentwise relative error, each entry BERR(i) is an estimate
!>     of the componentwise relative error in the ith solution computed by
!>     this routine, BERR(i) = max_j |X(j,i) - XTRUE(j,i)| / |X(j,i)|.
!>
!>     BERR is only computed if it is requested by the value of
!>     PARAMS(LA_LINRX_CWISE_I).
!>
!>     See Lapack Working Note 165 for further details and extra
!>     cautions.
!>

CBE

!>          CBE is DOUBLE PRECISION array, dimension (NRHS, N_ERR_BNDS)
!>     For each right-hand side, this array contains information about
!>     various error bounds and condition numbers corresponding to the
!>     normwise relative error, which is defined as follows:
!>
!>     Normwise relative error in the ith solution vector:
!>              norm(X - XTRUE, inf)
!>            ----------------------
!>                 norm(X, inf)
!>
!>     The array is indexed by the right-hand side i (on which the
!>     normwise relative error depends), and the type of error
!>     information as described below. There currently are up to three
!>     pieces of information returned.
!>
!>     The first index in CBE(i,:) corresponds to the ith
!>     right-hand side.
!>
!>     The second index in CBE(:,err) contains the following
!>     three fields:
!>     err = 1  boolean. Trust the answer if the
!>              reciprocal condition number is less than the threshold
!>              sqrt(n) * dlamch('Epsilon').
!>
!>     err = 2  error bound: The estimated forward error,
!>              almost certainly within a factor of 10 of the true error
!>              so long as the next entry is greater than the threshold
!>              sqrt(n) * dlamch('Epsilon'). This error bound should only
!>              be trusted if the previous boolean is true.
!>
!>     err = 3  Reciprocal condition number: Estimated normwise
!>              reciprocal condition number.  Compared with the threshold
!>              sqrt(n) * dlamch('Epsilon') to determine if the error
!>              estimate is . These reciprocal condition
!>              numbers are 1 / (norm(Z^{-1},inf) * norm(Z,inf)) for some
!>              appropriately scaled matrix Z.
!>              Let Z = S*A, where S scales each row by a power of the
!>              radix so all absolute row sums of Z are approximately 1.
!>
!>     See Lapack Working Note 165 for further details and extra
!>     cautions.
!>

ERR_BNDS_NORM

!>     various error bounds and condition numbers corresponding to the

!>
!>     Normwise relative error in the ith solution vector:
!>              norm(X - XTRUE, inf)
!>            ----------------------
!>                 norm(X, inf)

!>     The array is indexed by the right-hand side i (on which the
!>     normwise relative error depends), and the type of error

!>     pieces of information returned.
!>
!>     The first index in ERR_BNDS_NORM(i,:) corresponds to the ith
!>     right-hand side.
!>
!>     The second index in ERR_BNDS_NORM(:,err) contains the following

!>     err = 2  error bound: The estimated forward error,

ERR_BNDS_COMP

!>          ERR_BNDS_COMP is DOUBLE PRECISION array, dimension (NRHS, N_ERR_BNDS)
!>     For each right-hand side, this array contains information about
!>     various error bounds and condition numbers corresponding to the
!>     componentwise relative error, which is defined as follows:
!>
!>     Componentwise relative error in the ith solution vector:
!>                    abs(XTRUE(j,i) - X(j,i))
!>             max_j ----------------------
!>                         abs(X(j,i))
!>
!>     The array is indexed by the right-hand side i (on which the
!>     componentwise relative error depends), and the type of error
!>     information as described below. There currently are up to three
!>     pieces of information returned for each right-hand side. If
!>     componentwise accuracy is not requested (PARAMS(3) = 0.0), then
!>     ERR_BNDS_COMP is not accessed.  If N_ERR_BNDS < 3, then at most
!>     the first (:,N_ERR_BNDS) entries are returned.
!>
!>     The first index in ERR_BNDS_COMP(i,:) corresponds to the ith
!>     right-hand side.
!>
!>     The second index in ERR_BNDS_COMP(:,err) contains the following
!>     three fields:
!>     err = 1  boolean. Trust the answer if the
!>              reciprocal condition number is less than the threshold
!>              sqrt(n) * dlamch('Epsilon').
!>
!>     err = 2  error bound: The estimated forward error,
!>              almost certainly within a factor of 10 of the true error
!>              so long as the next entry is greater than the threshold
!>              sqrt(n) * dlamch('Epsilon'). This error bound should only
!>              be trusted if the previous boolean is true.
!>
!>     err = 3  Reciprocal condition number: Estimated componentwise
!>              reciprocal condition number.  Compared with the threshold
!>              sqrt(n) * dlamch('Epsilon') to determine if the error
!>              estimate is . These reciprocal condition
!>              numbers are 1 / (norm(Z^{-1},inf) * norm(Z,inf)) for some
!>              appropriately scaled matrix Z.
!>              Let Z = S*(A*diag(x)), where x is the solution for the
!>              current right-hand side and S scales each row of
!>              A*diag(x) by a power of the radix so all absolute row
!>              sums of Z are approximately 1.
!>
!>     See Lapack Working Note 165 for further details and extra
!>     cautions.
!>

NPARAMS

!>          NPARAMS is INTEGER
!>     Specifies the number of parameters set in PARAMS.  If <= 0, the
!>     PARAMS array is never referenced and default values are used.
!>

PARAMS

!>          PARAMS is DOUBLE PRECISION array, dimension NPARAMS
!>     Specifies algorithm parameters.  If an entry is < 0.0, then
!>     that entry will be filled with default value used for that
!>     parameter.  Only positions up to NPARAMS are accessed; defaults
!>     are used for higher-numbered parameters.
!>
!>       PARAMS(LA_LINRX_ITREF_I = 1) : Whether to perform iterative
!>            refinement or not.
!>         Default: 1.0D+0
!>            = 0.0:  No refinement is performed, and no error bounds are
!>                    computed.
!>            = 1.0:  Use the double-precision refinement algorithm,
!>                    possibly with doubled-single computations if the
!>                    compilation environment does not support DOUBLE
!>                    PRECISION.
!>              (other values are reserved for future use)
!>
!>       PARAMS(LA_LINRX_ITHRESH_I = 2) : Maximum number of residual
!>            computations allowed for refinement.
!>         Default: 10
!>         Aggressive: Set to 100 to permit convergence using approximate
!>                     factorizations or factorizations other than LU. If
!>                     the factorization uses a technique other than
!>                     Gaussian elimination, the guarantees in
!>                     err_bnds_norm and err_bnds_comp may no longer be
!!>                     trustworthy.
!>
!>       PARAMS(LA_LINRX_CWISE_I = 3) : Flag determining if the code

!>         Default: 1.0 (attempt componentwise convergence)

RWORK

INFO

!>          INFO is INTEGER
!>       = 0:  Successful exit. The solution to every right-hand side is

!>       < 0:  If INFO = -i, the i-th argument had an illegal value
!>       > 0 and <= N:  U(INFO,INFO) is exactly zero.  The factorization
