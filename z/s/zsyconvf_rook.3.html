توثيق الوظيفة/الروتين الفرعي

subroutine zsyconvf_rook (character uplo, character way, integer n, complex*16, dimension( lda, * ) a, integer lda, complex*16, dimension( * ) e, integer, dimension( * ) ipiv, integer info)

ZSYCONVF_ROOK

الغرض:

!> إذا كان المعامل WAY = 'C':
!> ZSYCONVF_ROOK يقوم بتحويل تنسيق إخراج التحليل الذي تم إدخاله في المعامل A
!> بواسطة ZSYTRF_ROOK إلى تنسيق إخراج التحليل المستخدم في ZSYTRF_RK
!> (أو ZSYTRF_BK) والذي يتم تخزينه عند الخروج في المعاملين A و E.
!> تنسيق IPIV لـ ZSYTRF_ROOK و ZSYTRF_RK (أو ZSYTRF_BK) هو نفسه ولا يتم تحويله.
!>
!> إذا كان المعامل WAY = 'R':
!> ZSYCONVF_ROOK يقوم بالتحويل في الاتجاه المعاكس، أي.
!> يقوم بتحويل تنسيق إخراج التحليل المستخدم في ZSYTRF_RK
!> (أو ZSYTRF_BK) الذي تم إدخاله في المعاملين A و E إلى
!> تنسيق إخراج التحليل المستخدم في ZSYTRF_ROOK والذي يتم تخزينه
!> عند الخروج في المعامل A. تنسيق IPIV لـ ZSYTRF_ROOK و
!> ZSYTRF_RK (أو ZSYTRF_BK) هو نفسه ولا يتم تحويله.
!>
!> يمكن لـ ZSYCONVF_ROOK أيضًا التحويل في حالة المصفوفة الهرميتية، أي. بين
!> التنسيقات المستخدمة في ZHETRF_ROOK و ZHETRF_RK (أو ZHETRF_BK).
!>

المعاملات

UPLO 
!>          UPLO هو CHARACTER*1
!>          يحدد ما إذا كانت تفاصيل التحليل مخزنة كمصفوفة مثلثية علوية أو سفلية A.
!>          = 'U':  مثلثي علوي
!>          = 'L':  مثلثي سفلي
!>

WAY

!>          WAY هو CHARACTER*1
!>          = 'C': تحويل
!>          = 'R': عكس
!>

N

!>          N هو INTEGER
!>          ترتيب المصفوفة A.  N >= 0.
!>

A

!>          A هي مصفوفة COMPLEX*16، ذات أبعاد (LDA,N)
!>
!>          1) إذا كان WAY = 'C':
!>
!>          عند الإدخال، تحتوي على تفاصيل التحليل بتنسيق مستخدم في
!>          ZSYTRF_ROOK:
!>            أ) جميع عناصر المصفوفة القطرية الفرعية المتماثلة
!>               D على القطر الرئيسي للمصفوفة A وعلى القطر الفرعي العلوي
!>               (أو الفرعي السفلي) للمصفوفة A، و
!>            ب) إذا كان UPLO = 'U': المضاعفات المستخدمة للحصول على العامل U
!>               في الجزء الفرعي العلوي للمصفوفة A.
!>               إذا كان UPLO = 'L': المضاعفات المستخدمة للحصول على العامل L
!>               في الجزء الفرعي العلوي للمصفوفة A.
!>
!>          عند الخروج، تحتوي على تفاصيل التحليل بتنسيق مستخدم في
!>          ZSYTRF_RK أو ZSYTRF_BK:
!>            أ) فقط عناصر القطر الرئيسي للمصفوفة القطرية الفرعية المتماثلة
!>               D على القطر الرئيسي للمصفوفة A، أي. D(k,k) = A(k,k);
!>               (عناصر القطر الفرعي العلوي (أو الفرعي السفلي) لـ D
!>                يتم تخزينها عند الخروج في المصفوفة E)، و
!>            ب) إذا كان UPLO = 'U': العامل U في الجزء الفرعي العلوي للمصفوفة A.
!>               إذا كان UPLO = 'L': العامل L في الجزء الفرعي السفلي للمصفوفة A.
!>
!>          2) إذا كان WAY = 'R':
!>
!>          عند الإدخال، تحتوي على تفاصيل التحليل بتنسيق مستخدم في
!>          ZSYTRF_RK أو ZSYTRF_BK:
!>            أ) فقط عناصر القطر الرئيسي للمصفوفة القطرية الفرعية المتماثلة
!>               D على القطر الرئيسي للمصفوفة A، أي. D(k,k) = A(k,k);
!>               (عناصر القطر الفرعي العلوي (أو الفرعي السفلي) لـ D
!>                يتم تخزينها عند الخروج في المصفوفة E)، و
!>            ب) إذا كان UPLO = 'U': العامل U في الجزء الفرعي العلوي للمصفوفة A.
!>               إذا كان UPLO = 'L': العامل L في الجزء الفرعي السفلي للمصفوفة A.
!>
!>          عند الخروج، تحتوي على تفاصيل التحليل بتنسيق مستخدم في
!>          ZSYTRF_ROOK:
!>            أ) جميع عناصر المصفوفة القطرية الفرعية المتماثلة
!>               D على القطر الرئيسي للمصفوفة A وعلى القطر الفرعي العلوي
!>               (أو الفرعي السفلي) للمصفوفة A، و
!>            ب) إذا كان UPLO = 'U': المضاعفات المستخدمة للحصول على العامل U
!>               في الجزء الفرعي العلوي للمصفوفة A.
!>               إذا كان UPLO = 'L': المضاعفات المستخدمة للحصول على العامل L
!>               في الجزء الفرعي العلوي للمصفوفة A.
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرائد للمصفوفة A.  LDA >= max(1,N).
!>

E

!>          E هي مصفوفة COMPLEX*16، ذات أبعاد (N)
!>
!>          1) إذا كان WAY = 'C':
!>
!>          عند الإدخال، مساحة عمل فقط.
!>
!>          عند الخروج، تحتوي على عناصر القطر الفرعي العلوي (أو الفرعي السفلي)
!>          للمصفوفة القطرية الفرعية المتماثلة D
!>          بمصفوفات قطرية فرعية بحجم 1x1 أو 2x2، حيث
!>          إذا كان UPLO = 'U': E(i) = D(i-1,i), i=2:N, E(1) تم تعيينها إلى 0;
!>          إذا كان UPLO = 'L': E(i) = D(i+1,i), i=1:N-1, E(N) تم تعيينها إلى 0.
!>
!>          2) إذا كان WAY = 'R':
!>
!>          عند الإدخال، تحتوي على عناصر القطر الفرعي العلوي (أو الفرعي السفلي)
!>          للمصفوفة القطرية الفرعية المتماثلة D
!>          بمصفوفات قطرية فرعية بحجم 1x1 أو 2x2، حيث
!>          إذا كان UPLO = 'U': E(i) = D(i-1,i),i=2:N, E(1) غير مستخدمة;
!>          إذا كان UPLO = 'L': E(i) = D(i+1,i),i=1:N-1, E(N) غير مستخدمة.
!>
!>          عند الخروج، لم يتم تغييرها.
!>

IPIV

!>          IPIV هي مصفوفة INTEGER، ذات أبعاد (N)
!>          عند الإدخال، تفاصيل التبادلات وهيكل الكتلة D كما تم تحديده:
!>          1) بواسطة ZSYTRF_ROOK، إذا كان WAY = 'C';
!>          2) بواسطة ZSYTRF_RK (أو ZSYTRF_BK)، إذا كان WAY = 'R'.
!>          تنسيق IPIV هو نفسه لجميع هذه الروتينات.
!>
!>          عند الخروج، لم يتم تغييرها.
!>

INFO

!>          INFO هو INTEGER
!>          = 0:  انتهاء ناجح
!>          < 0:  إذا كان INFO = -i، فإن المعامل i كان له قيمة غير قانونية.
!>
