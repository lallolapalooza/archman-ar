توثيق الوظيفة/الإجراء الفرعي

subroutine zsytrf (character uplo, integer n, complex*16, dimension( lda, * ) a, integer lda, integer, dimension( * ) ipiv, complex*16, dimension( * ) work, integer lwork, integer info)

ZSYTRF

الغرض:

!>
!> ZSYTRF يحسب تحليل مصفوفة متناظرة مركبة A
!> باستخدام طريقة أخذ العينات القطرية لبونش-كوفمان. شكل
!> التحليل هو
!>
!>    A = U*D*U**T  أو  A = L*D*L**T
!>
!> حيث U (أو L) هو حاصل ضرب تبادلات ومصفوفات مثلثة علوية (سفلية)
!> وحدوية، و D متناظرة وقطرية كتلية مع كتل قطرية 1*1 و 2*2.
!>
!> هذا هو الإصدار المجمع للخوارزمية، والذي يستدعي BLAS المستوى 3.
!>

المعلمات

UPLO 
!>          UPLO هو CHARACTER*1
!>          = 'U':  المثلث العلوي من A مخزن؛
!>          = 'L':  المثلث السفلي من A مخزن.
!>

N

!>          N هو INTEGER
!>          رتبة المصفوفة A.  N >= 0.
!>

A

!>          A هو COMPLEX*16 array, dimension (LDA,N)
!>          عند الدخول، المصفوفة المتناظرة A.  إذا كان UPLO = 'U'، فإن الجزء
!>          المثلثي العلوي 100*N من A يحتوي على الجزء
!>          المثلثي العلوي للمصفوفة A، والجزء السفلي
!>          المثلثي الصارم لـ A لا يتم الرجوع إليه. إذا كان UPLO = 'L'، فإن الجزء
!>          المثلثي السفلي 100*N من A يحتوي على الجزء
!>          المثلثي السفلي للمصفوفة A، والجزء العلوي
!>          المثلثي الصارم لـ A لا يتم الرجوع إليه.
!>
!>          عند الخروج، المصفوفة القطرية الكتلية D والمعاملات المستخدمة
!>          للحصول على العامل U أو L (انظر أدناه لمزيد من التفاصيل).
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرئيسي للمصفوفة A.  LDA >= max(1,N).
!>

IPIV

!>          IPIV هو INTEGER array, dimension (N)
!>          تفاصيل التبادلات وهيكل الكتلة D.
!>          إذا كان IPIV(k) > 0، فقد تم تبديل الصفوف والأعمدة k و IPIV(k)
!>          و D(k,k) هي كتلة قطرية 1*1.
!>          إذا كان UPLO = 'U' و IPIV(k) = IPIV(k-1) < 0، فقد تم تبديل الصفوف
!>          والأعمدة k-1 و -IPIV(k) و D(k-1:k,k-1:k)
!>          هي كتلة قطرية 2*2. إذا كان UPLO = 'L' و IPIV(k) =
!>          IPIV(k+1) < 0، فقد تم تبديل الصفوف والأعمدة k+1 و -IPIV(k)
!>          و D(k:k+1,k:k+1) هي كتلة قطرية 2*2.
!>

WORK

!>          WORK هو COMPLEX*16 array, dimension (MAX(1,LWORK))
!>          عند الخروج، إذا كان INFO = 0، فإن WORK(1) يعيد LWORK الأمثل.
!>

LWORK

!>          LWORK هو INTEGER
!>          طول WORK.  LWORK >=1.  لأفضل أداء
!>          LWORK >= N*NB، حيث NB هو حجم الكتلة الذي تعيده ILAENV.
!>
!>          إذا كان LWORK = -1، فسيتم افتراض استعلام عن مساحة العمل؛ تقوم الروتين
!>          فقط بحساب الحجم الأمثل للمصفوفة WORK، وإرجاع
!>          هذه القيمة كإدخال أول للمصفوفة WORK، ولا يتم إصدار أي رسالة خطأ
!>          متعلقة بـ LWORK بواسطة XERBLA.
!>

INFO

!>          INFO هو INTEGER
!>          = 0:  انتهاء ناجح
!>          < 0:  إذا كان INFO = -i، فإن وسيطة i كانت ذات قيمة غير قانونية
!>          > 0:  إذا كان INFO = i، فإن D(i,i) تساوي صفرًا تمامًا. التحليل
!>                تم الانتهاء منه، لكن المصفوفة القطرية الكتلية D
!>                فردية تمامًا، وسيحدث قسمة على صفر إذا
!>                تم استخدامها لحل نظام من المعادلات.
!>

تفاصيل إضافية:

!>
!>  إذا كان UPLO = 'U'، فإن A = U*D*U**T، حيث
!>     U = P(n)*U(n)* ... *P(k)U(k)* ...,
!>  أي، U هو حاصل ضرب الحدود P(k)*U(k)، حيث k يتناقص من n إلى
!>  1 بخطوات 1 أو 2، و D هي مصفوفة قطرية كتلية مع كتل قطرية 1*1
!>  و 2*2 D(k). P(k) هي مصفوفة تبديل كما
!>  يتم تعريفها بواسطة IPIV(k)، و U(k) هي مصفوفة مثلثة علوية وحدوية، بحيث
!>  إذا كانت الكتلة القطرية D(k) من الرتبة s (s = 1 أو 2)، فإن
!>
!>             (   I    v    0   )   k-s
!>     U(k) =  (   0    I    0   )   s
!>             (   0    0    I   )   n-k
!>                k-s   s   n-k
!>
!>  إذا كان s = 1، فإن D(k) تستبدل A(k,k)، و v تستبدل A(1:k-1,k).
!>  إذا كان s = 2، فإن المثلث العلوي من D(k) يستبدل A(k-1,k-1)، A(k-1,k)،
!>  و A(k,k)، و v تستبدل A(1:k-2,k-1:k).
!>
!>  إذا كان UPLO = 'L'، فإن A = L*D*L**T، حيث
!>     L = P(1)*L(1)* ... *P(k)*L(k)* ...,
!>  أي، L هو حاصل ضرب الحدود P(k)*L(k)، حيث k يتزايد من 1 إلى
!>  n بخطوات 1 أو 2، و D هي مصفوفة قطرية كتلية مع كتل قطرية 1*1
!>  و 2*2 D(k). P(k) هي مصفوفة تبديل كما
!>  يتم تعريفها بواسطة IPIV(k)، و L(k) هي مصفوفة مثلثة سفلية وحدوية، بحيث
!>  إذا كانت الكتلة القطرية D(k) من الرتبة s (s = 1 أو 2)، فإن
!>
!>             (   I    0     0   )  k-1
!>     L(k) =  (   0    I     0   )  s
!>             (   0    v     I   )  n-k-s+1
!>                k-1   s  n-k-s+1
!>
!>  إذا كان s = 1، فإن D(k) تستبدل A(k,k)، و v تستبدل A(k+1:n,k).
!>  إذا كان s = 2، فإن المثلث السفلي من D(k) يستبدل A(k,k)، A(k+1,k)،
!>  و A(k+1,k+1)، و v تستبدل A(k+2:n,k:k+1).
!>
