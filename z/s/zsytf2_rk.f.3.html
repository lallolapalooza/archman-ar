subroutine zsytf2_rk (uplo, n, a, lda, e, ipiv, info)
ZSYTF2_RK تحسب تحليل مصفوفة متناظرة معقدة غير محددة باستخدام طريقة المحورية القطرية لبونش-كوفمان المحدودة (التسلسل الهرمي للقلم الرصاص غير المجمّع BLAS2).

توثيق الوظيفة/الروتين الفرعي

subroutine zsytf2_rk (character uplo, integer n, complex*16, dimension( lda, * ) a, integer lda, complex*16, dimension( * ) e, integer, dimension( * ) ipiv, integer info)

ZSYTF2_RK تحسب تحليل مصفوفة متناظرة معقدة غير محددة باستخدام طريقة المحورية القطرية لبونش-كوفمان المحدودة (التسلسل الهرمي للقلم الرصاص غير المجمّع BLAS2).

الغرض:

!> ZSYTF2_RK تحسب تحليل مصفوفة متناظرة معقدة A
!> باستخدام طريقة المحورية القطرية لبونش-كوفمان المحدودة (قلم الرصاص):
!>
!>    A = P*U*D*(U**T)*(P**T) أو A = P*L*D*(L**T)*(P**T)،
!>
!> حيث U (أو L) هي مصفوفة مثلثة عليا (أو سفلية) وحدوية،
!> U**T (أو L**T) هي منقولة U (أو L)، P هي مصفوفة تبديل،
!> P**T هي منقولة P، و D مصفوفة متناظرة وقطرية جزئية بقطاعات قطرية بحجم 1x1 و 2x2.
!>
!> هذه هي النسخة غير المجمّعة للخوارزمية، والتي تستدعي BLAS المستوى 2.
!> لمزيد من المعلومات انظر قسم التفاصيل الإضافية.
!>

المعاملات

UPLO 
!>          UPLO هي CHARACTER*1
!>          يحدد ما إذا كان الجزء المثلثي العلوي أو السفلي من المصفوفة المتناظرة A مخزنًا:
!>          = 'U':  مثلثي علوي
!>          = 'L':  مثلثي سفلي
!>

N

!>          N هي INTEGER
!>          ترتيب المصفوفة A.  N >= 0.
!>

A

!>          A هي مصفوفة COMPLEX*16، بأبعاد (LDA,N)
!>          عند الدخول، المصفوفة المتناظرة A.
!>            إذا كان UPLO = 'U': الجزء العلوي المثلثي الذي حجمه N*N في A يحتوي على الجزء العلوي المثلثي من المصفوفة A،
!>            والجزء السفلي المثلثي الصارم من A لا يتم الرجوع إليه.
!>
!>            إذا كان UPLO = 'L': الجزء السفلي المثلثي الذي حجمه N*N في A يحتوي على الجزء السفلي المثلثي من المصفوفة A،
!>            والجزء العلوي المثلثي الصارم من A لا يتم الرجوع إليه.
!>
!>          عند الخروج، تحتوي على:
!>            أ) عناصر قطرية فقط من المصفوفة القطرية الجزئية المتناظرة D على قطر A، أي D(k,k) = A(k,k)؛
!>               (عناصر فوق القطر (أو تحت القطر) لـ D مخزنة عند الخروج في المصفوفة E)، و
!>            ب) إذا كان UPLO = 'U': العامل U في الجزء فوق القطر من A.
!>               إذا كان UPLO = 'L': العامل L في الجزء تحت القطر من A.
!>

LDA

!>          LDA هي INTEGER
!>          البعد الرئيسي للمصفوفة A.  LDA >= max(1,N).
!>

E

!>          E هي مصفوفة COMPLEX*16، بأبعاد (N)
!>          عند الخروج، تحتوي على عناصر فوق القطر (أو تحت القطر)
!>          للمصفوفة القطرية الجزئية المتناظرة D
!>          مع قطاعات قطرية بحجم 1x1 أو 2x2، حيث
!>          إذا كان UPLO = 'U': E(i) = D(i-1,i), i=2:N, E(1) يتم تعيينه إلى 0؛
!>          إذا كان UPLO = 'L': E(i) = D(i+1,i), i=1:N-1, E(N) يتم تعيينه إلى 0.
!>
!>          ملاحظة: لقطاع قطري بحجم 1x1 D(k)، حيث
!>          1 <= k <= N، يتم تعيين العنصر E(k) إلى 0 في كلتا الحالتين
!>          UPLO = 'U' أو UPLO = 'L'.
!>

IPIV

!>          IPIV هي مصفوفة INTEGER، بأبعاد (N)
!>          IPIV تصف مصفوفة التبديل P في تحليل
!>          المصفوفة A على النحو التالي. القيمة المطلقة لـ IPIV(k)
!>          تمثل فهرس الصف والعمود اللذين تم تبديلهما
!>          مع الصف والعمود k. قيمة UPLO
!>          تصف ترتيب تطبيق التبديلات.
!>          أيضًا، علامة IPIV تمثل الهيكل الجزئي
!>          للمصفوفة القطرية الجزئية المتناظرة D مع قطاعات قطرية بحجم 1x1 أو 2x2
!>          والتي تتوافق مع تبديل واحد أو اثنين
!>          في كل خطوة تحليل. لمزيد من المعلومات انظر قسم التفاصيل الإضافية.
!>
!>          إذا كان UPLO = 'U'،
!>          (في ترتيب التحليل، k يتناقص من N إلى 1):
!>            أ) إدخال إيجابي فردي IPIV(k) > 0 يعني:
!>               D(k,k) هو قطاع قطري بحجم 1x1.
!>               إذا كان IPIV(k) != k، فقد تم تبديل الصفوف والأعمدة k و IPIV(k)
!>               في المصفوفة A(1:N,1:N)؛
!>               إذا كان IPIV(k) = k، لم يحدث أي تبديل.
!>
!>            ب) زوج من الإدخالات السالبة المتتالية
!>               IPIV(k) < 0 و IPIV(k-1) < 0 يعني:
!>               D(k-1:k,k-1:k) هو قطاع قطري بحجم 2x2.
!>               (ملاحظة: الإدخالات السالبة في IPIV تظهر فقط في أزواج).
!>               1) إذا كان -IPIV(k) != k، فقد تم تبديل الصفوف والأعمدة
!>                  k و -IPIV(k)
!>                  في المصفوفة A(1:N,1:N).
!>                  إذا كان -IPIV(k) = k، لم يحدث أي تبديل.
!>               2) إذا كان -IPIV(k-1) != k-1، فقد تم تبديل الصفوف والأعمدة
!>                  k-1 و -IPIV(k-1)
!>                  في المصفوفة A(1:N,1:N).
!>                  إذا كان -IPIV(k-1) = k-1، لم يحدث أي تبديل.
!>
!>            ج) في كلتا الحالتين أ) و ب)، دائمًا ABS( IPIV(k) ) <= k.
!>
!>            د) ملاحظة: أي إدخال IPIV(k) غير صفري دائمًا عند الخروج.
!>
!>          إذا كان UPLO = 'L'،
!>          (في ترتيب التحليل، k يزداد من 1 إلى N):
!>            أ) إدخال إيجابي فردي IPIV(k) > 0 يعني:
!>               D(k,k) هو قطاع قطري بحجم 1x1.
!>               إذا كان IPIV(k) != k، فقد تم تبديل الصفوف والأعمدة k و IPIV(k)
!>               في المصفوفة A(1:N,1:N).
!>               إذا كان IPIV(k) = k، لم يحدث أي تبديل.
!>
!>            ب) زوج من الإدخالات السالبة المتتالية
!>               IPIV(k) < 0 و IPIV(k+1) < 0 يعني:
!>               D(k:k+1,k:k+1) هو قطاع قطري بحجم 2x2.
!>               (ملاحظة: الإدخالات السالبة في IPIV تظهر فقط في أزواج).
!>               1) إذا كان -IPIV(k) != k، فقد تم تبديل الصفوف والأعمدة
!>                  k و -IPIV(k)
!>                  في المصفوفة A(1:N,1:N).
!>                  إذا كان -IPIV(k) = k، لم يحدث أي تبديل.
!>               2) إذا كان -IPIV(k+1) != k+1، فقد تم تبديل الصفوف والأعمدة
!>                  k-1 و -IPIV(k-1)
!>                  في المصفوفة A(1:N,1:N).
!>                  إذا كان -IPIV(k+1) = k+1، لم يحدث أي تبديل.
!>
!>            ج) في كلتا الحالتين أ) و ب)، دائمًا ABS( IPIV(k) ) >= k.
!>
!>            د) ملاحظة: أي إدخال IPIV(k) غير صفري دائمًا عند الخروج.
!>

INFO

!>          INFO هي INTEGER
!>          = 0: خروج ناجح
!>
!>          < 0: إذا كان INFO = -k، كان المعامل k غير قانوني
!>
!>          > 0: إذا كان INFO = k، المصفوفة A منفردة، بسبب:
!>                 إذا كان UPLO = 'U': العمود k في الجزء
!>                 المثلثي العلوي من A يحتوي على جميع الأصفار.
!>                 إذا كان UPLO = 'L': العمود k في الجزء
!>                 المثلثي السفلي من A يحتوي على جميع الأصفار.
!>
!>               لذلك D(k,k) يساوي صفرًا تمامًا، وعناصر فوق القطر
!>               للعمود k من U (أو عناصر تحت القطر
!>               للعمود k من L) جميعها أصفار. تم إكمال التحليل،
!>               لكن المصفوفة القطرية الجزئية D منفردة تمامًا، وسيحدث
!>               قسمة على صفر إذا تم استخدامها لحل نظام معادلات.
!>
!>               ملاحظة: INFO يخزن فقط أول حدوث
!>               للانفراد، أي حدوث لاحق للانفراد
!>               لا يتم تخزينه في INFO على الرغم من أن التحليل
!>               يكتمل دائمًا.
!>
