subroutine zsyr (character uplo, integer n, complex*16 alpha, complex*16, dimension( * ) x, integer incx, complex*16, dimension( lda, * ) a, integer lda)

ZSYR تقوم بتحديث مصفوفة معقدة متناظرة بترتيب 1.

الغرض:

!>
!> ZSYR   تقوم بعملية التحديث المتناظر بترتيب 1:
!>
!>    A := alpha*x*x**H + A,
!>
!> حيث alpha هو عدد قياسي معقد، x هو متجه بحجم n، و A هي مصفوفة
!> متناظرة بحجم n × n.
!>

المعلمات

UPLO 
!>          UPLO هو CHARACTER*1
!>           عند الإدخال، تحدد UPLO ما إذا كان الجزء العلوي أو السفلي
!>           من المصفوفة A سيتم الرجوع إليه كما يلي:
!>
!>              UPLO = 'U' or 'u'   يتم الرجوع فقط إلى الجزء العلوي من A.
!>
!>              UPLO = 'L' or 'l'   يتم الرجوع فقط إلى الجزء السفلي من A.
!>
!>           لا تتغير عند الخروج.
!>

N

!>          N هو INTEGER
!>           عند الإدخال، يحدد N رتبة المصفوفة A.
!>           يجب أن تكون N على الأقل صفر.
!>           لا تتغير عند الخروج.
!>

ALPHA

!>          ALPHA هو COMPLEX*16
!>           عند الإدخال، يحدد ALPHA العدد القياسي alpha.
!>           لا يتغير عند الخروج.
!>

X

!>          X هو مصفوفة COMPLEX*16، ذات بعد لا يقل عن
!>           ( 1 + ( N - 1 )*abs( INCX ) ).
!>           قبل الدخول، يجب أن يحتوي المتجه X المكبر على المتجه x المكون من N عنصر.
!>           لا يتغير عند الخروج.
!>

INCX

!>          INCX هو INTEGER
!>           عند الإدخال، يحدد INCX الزيادة لعناصر X. يجب ألا يكون INCX صفرًا.
!>           لا يتغير عند الخروج.
!>

A

!>          A هي مصفوفة COMPLEX*16، ذات بعد ( LDA, N )
!>           قبل الدخول، مع UPLO = 'U' أو 'u'، يجب أن يحتوي الجزء العلوي من
!>           المصفوفة A على الجزء العلوي من المصفوفة المتناظرة، والجزء السفلي
!>           الصارم من A لا يتم الرجوع إليه. عند الخروج، يتم الكتابة فوق الجزء العلوي
!>           من المصفوفة A بالجزء العلوي من المصفوفة المحدثة.
!>           قبل الدخول، مع UPLO = 'L' أو 'l'، يجب أن يحتوي الجزء السفلي من
!>           المصفوفة A على الجزء السفلي من المصفوفة المتناظرة، والجزء العلوي
!>           الصارم من A لا يتم الرجوع إليه. عند الخروج، يتم الكتابة فوق الجزء السفلي
!>           من المصفوفة A بالجزء السفلي من المصفوفة المحدثة.
!>

LDA

!>          LDA هو INTEGER
!>           عند الإدخال، يحدد LDA البعد الأول لـ A كما هو معلن في البرنامج
!>           (الفرعي) المستدعي. يجب أن يكون LDA على الأقل
!>           max( 1, N ).
!>           لا يتغير عند الخروج.
!>
