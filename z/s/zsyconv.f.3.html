توثيق الوظيفة/الروتين الفرعي

subroutine zsyconv (character uplo, character way, integer n, complex*16, dimension( lda, * ) a, integer lda, integer, dimension( * ) ipiv, complex*16, dimension( * ) e, integer info)

ZSYCONV

الغرض:

!>
!> ZSYCONV يحول A المعطى بواسطة ZHETRF إلى L و D أو العكس.
!> الحصول على العناصر غير القطرية من D (المُرجعة في مساحة العمل)
!> وتطبيق أو عكس التبديل الذي تم في TRF.
!>

المعلمات

UPLO 
!>          UPLO هو CHARACTER*1
!>          يحدد ما إذا كانت تفاصيل التحليل مخزنة
!>          كمصفوفة مثلثية علوية أو سفلية.
!>          = 'U':  مثلثية علوية، الشكل هو A = U*D*U**T؛
!>          = 'L':  مثلثية سفلية، الشكل هو A = L*D*L**T.
!>

WAY

!>          WAY هو CHARACTER*1
!>          = 'C': تحويل
!>          = 'R': تراجع
!>

N

!>          N هو INTEGER
!>          رتبة المصفوفة A.  N >= 0.
!>

A

!>          A هي مصفوفة COMPLEX*16، الأبعاد (LDA,N)
!>          المصفوفة القطرية الفرعية D والمعاملات المستخدمة
!>          للحصول على العامل U أو L كما حسبها ZSYTRF.
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرائد للمصفوفة A.  LDA >= max(1,N).
!>

IPIV

!>          IPIV هي مصفوفة INTEGER، الأبعاد (N)
!>          تفاصيل التبادلات وهيكل D الفرعي
!>          كما حددها ZSYTRF.
!>

E

!>          E هي مصفوفة COMPLEX*16، الأبعاد (N)
!>          E يخزن المصفوفة الفرعية/المصفوفة الفرعية للمصفوفة القطرية
!>          المتناظرة 1x1 أو 2x2 في LDLT.
!>

INFO

!>          INFO هو INTEGER
!>          = 0:  انتهاء ناجح
!>          < 0:  إذا كان INFO = -i، فإن المعامل i كان له قيمة غير قانونية
!>
