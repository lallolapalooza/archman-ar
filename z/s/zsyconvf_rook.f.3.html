subroutine zsyconvf_rook (character uplo, character way, integer n, complex*16, dimension( lda, * ) a, integer lda, complex*16, dimension( * ) e, integer, dimension( * ) ipiv, integer info)

ZSYCONVF_ROOK

الغرض:

!> إذا كانت المعلمة WAY = 'C':
!> تحول ZSYCONVF_ROOK تنسيق مخرجات التحليل المستخدم في ZSYTRF_ROOK
!> المقدم في المعلمة A إلى تنسيق مخرجات التحليل المستخدم في ZSYTRF_RK
!> (أو ZSYTRF_BK) والذي يتم تخزينه عند الخروج في المعلمتين A و E.
!> تنسيق IPIV لكل من ZSYTRF_ROOK و ZSYTRF_RK (أو ZSYTRF_BK) هو نفسه
!> ولا يتم تحويله.
!>
!> إذا كانت المعلمة WAY = 'R':
!> تقوم ZSYCONVF_ROOK بإجراء التحويل في الاتجاه العكسي، أي تحويل
!> تنسيق مخرجات التحليل المستخدم في ZSYTRF_RK (أو ZSYTRF_BK) المقدم
!> في المعلمتين A و E إلى تنسيق مخرجات التحليل المستخدم في ZSYTRF_ROOK
!> والذي يتم تخزينه عند الخروج في المعلمة A. تنسيق IPIV لكل من
!> ZSYTRF_ROOK و ZSYTRF_RK (أو ZSYTRF_BK) هو نفسه ولا يتم تحويله.
!>
!> يمكن لـ ZSYCONVF_ROOK أيضًا التحويل في حالة المصفوفة المترافقة، أي بين
!> التنسيقات المستخدمة في ZHETRF_ROOK و ZHETRF_RK (أو ZHETRF_BK).
!>

المعلمات

UPLO 
!>          UPLO هي CHARACTER*1
!>          تحدد ما إذا كانت تفاصيل التحليل مخزنة كمصفوفة مثلثية عليا
!>          أو سفلية A.
!>          = 'U':  مثلثية عليا
!>          = 'L':  مثلثية سفلية
!>

WAY

!>          WAY هي CHARACTER*1
!>          = 'C': تحويل
!>          = 'R': استعادة
!>

N

!>          N هي INTEGER
!>          رتبة المصفوفة A. N >= 0.
!>

A

!>          A هي مصفوفة COMPLEX*16، الأبعاد (LDA,N)
!>
!>          1) إذا كانت WAY ='C':
!>
!>          عند الدخول، تحتوي على تفاصيل التحليل بتنسيق يستخدم في
!>          ZSYTRF_ROOK:
!>            أ) جميع عناصر المصفوفة القطرية المتماثلة المتوازية D
!>               على القطر الرئيسي لـ A وعلى القطر الرئيسي الفوقي
!>               (أو الرئيسي السفلي) لـ A، و
!>            ب) إذا كانت UPLO = 'U': العوامل المستخدمة للحصول على
!>               العامل U في الجزء الرئيسي الفوقي لـ A.
!>               إذا كانت UPLO = 'L': العوامل المستخدمة للحصول على
!>               العامل L في الجزء الرئيسي السفلي لـ A.
!>
!>          عند الخروج، تحتوي على تفاصيل التحليل بتنسيق يستخدم في
!>          ZSYTRF_RK أو ZSYTRF_BK:
!>            أ) فقط عناصر القطر الرئيسي للمصفوفة القطرية المتماثلة
!>               المتوازية D على القطر الرئيسي لـ A، أي D(k,k) = A(k,k)؛
!>               (عناصر القطر الفوقي (أو السفلي) لـ D مخزنة عند
!>                الخروج في المصفوفة E)، و
!>            ب) إذا كانت UPLO = 'U': العامل U في الجزء الرئيسي الفوقي
!>               لـ A.
!>               إذا كانت UPLO = 'L': العامل L في الجزء الرئيسي السفلي
!>               لـ A.
!>
!>          2) إذا كانت WAY = 'R':
!>
!>          عند الدخول، تحتوي على تفاصيل التحليل بتنسيق يستخدم في
!>          ZSYTRF_RK أو ZSYTRF_BK:
!>            أ) فقط عناصر القطر الرئيسي للمصفوفة القطرية المتماثلة
!>               المتوازية D على القطر الرئيسي لـ A، أي D(k,k) = A(k,k)؛
!>               (عناصر القطر الفوقي (أو السفلي) لـ D مخزنة عند
!>                الخروج في المصفوفة E)، و
!>            ب) إذا كانت UPLO = 'U': العامل U في الجزء الرئيسي الفوقي
!>               لـ A.
!>               إذا كانت UPLO = 'L': العامل L في الجزء الرئيسي السفلي
!>               لـ A.
!>
!>          عند الخروج، تحتوي على تفاصيل التحليل بتنسيق يستخدم في
!>          ZSYTRF_ROOK:
!>            أ) جميع عناصر المصفوفة القطرية المتماثلة المتوازية D
!>               على القطر الرئيسي لـ A وعلى القطر الرئيسي الفوقي
!>               (أو الرئيسي السفلي) لـ A، و
!>            ب) إذا كانت UPLO = 'U': العوامل المستخدمة للحصول على
!>               العامل U في الجزء الرئيسي الفوقي لـ A.
!>               إذا كانت UPLO = 'L': العوامل المستخدمة للحصول على
!>               العامل L في الجزء الرئيسي السفلي لـ A.
!>

LDA

!>          LDA هي INTEGER
!>          البعد الرائد للمصفوفة A. LDA >= max(1,N).
!>

E

!>          E هي مصفوفة COMPLEX*16، الأبعاد (N)
!>
!>          1) إذا كانت WAY ='C':
!>
!>          عند الدخول، هي مجرد مساحة عمل.
!>
!>          عند الخروج، تحتوي على عناصر القطر الرئيسي الفوقي (أو السفلي)
!>          للمصفوفة القطرية المتماثلة المتوازية D
!>          مع كتل قطرية بحجم 1x1 أو 2x2، حيث
!>          إذا كانت UPLO = 'U': E(i) = D(i-1,i), i=2:N, E(1) تم تعيينها إلى 0؛
!>          إذا كانت UPLO = 'L': E(i) = D(i+1,i), i=1:N-1, E(N) تم تعيينها إلى 0.
!>
!>          2) إذا كانت WAY = 'R':
!>
!>          عند الدخول، تحتوي على عناصر القطر الرئيسي الفوقي (أو السفلي)
!>          للمصفوفة القطرية المتماثلة المتوازية D
!>          مع كتل قطرية بحجم 1x1 أو 2x2، حيث
!>          إذا كانت UPLO = 'U': E(i) = D(i-1,i),i=2:N, E(1) لا تتم الإشارة إليها؛
!>          إذا كانت UPLO = 'L': E(i) = D(i+1,i),i=1:N-1, E(N) لا تتم الإشارة إليها.
!>
!>          عند الخروج، لا تتغير.
!>

IPIV

!>          IPIV هي مصفوفة INTEGER، الأبعاد (N)
!>          عند الدخول، تفاصيل التبادلات وهيكل الكتل لـ D كما تم تحديده:
!>          1) بواسطة ZSYTRF_ROOK، إذا كانت WAY ='C'؛
!>          2) بواسطة ZSYTRF_RK (أو ZSYTRF_BK)، إذا كانت WAY ='R'.
!>          تنسيق IPIV هو نفسه لجميع هذه الروتينات.
!>
!>          عند الخروج، لا تتغير.
!>

INFO

!>          INFO هي INTEGER
!>          = 0:  انتهاء ناجح
!>          < 0:  إذا كان INFO = -i، كانت المعلمة i غير قانونية
!>
