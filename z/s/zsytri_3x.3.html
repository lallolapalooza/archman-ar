توثيق الدالة/الإجراء الفرعي

subroutine zsytri_3x (character uplo, integer n, complex*16, dimension( lda, * ) a, integer lda, complex*16, dimension( * ) e, integer, dimension( * ) ipiv, complex*16, dimension( n+nb+1, * ) work, integer nb, integer info)

ZSYTRI_3X

الغرض:

!> ZSYTRI_3X يحسب معكوس مصفوفة معقدة متماثلة غير محددة
!> A باستخدام التحليل الذي تم حسابه بواسطة ZSYTRF_RK أو ZSYTRF_BK:
!>
!>     A = P*U*D*(U**T)*(P**T) أو A = P*L*D*(L**T)*(P**T)،
!>
!> حيث U (أو L) هي مصفوفة مثلثية علوية (أو سفلية) موحدة،
!> U**T (أو L**T) هي منقولة U (أو L)، P هي مصفوفة تبديل،
!> P**T هي منقولة P، و D هي مصفوفة متماثلة وقطرية كتلية
!> بأقطار 1x1 و 2x2.
!>
!> هذا هو الإصدار المجمّع للخوارزمية، يستدعي BLAS المستوى 3.
!>

المعاملات

UPLO 
!>          UPLO هو CHARACTER*1
!>          يحدد ما إذا كانت تفاصيل التحليل مخزنة كمصفوفة مثلثية
!>          علوية أم سفلية.
!>          = 'U':  يتم تخزين المثلث العلوي لـ A؛
!>          = 'L':  يتم تخزين المثلث السفلي لـ A.
!>

N

!>          N هو INTEGER
!>          رتبة المصفوفة A.  N >= 0.
!>

A

!>          A هو مصفوفة COMPLEX*16، الأبعاد (LDA,N)
!>          عند الإدخال، قطر المصفوفة القطرية الكتلية D والعوامل
!>          U أو L كما تم حسابها بواسطة ZSYTRF_RK و ZSYTRF_BK:
!>            أ) عناصر الأقطار فقط للمصفوفة القطرية الكتلية المتماثلة
!>               D على قطر A، أي D(k,k) = A(k,k)؛
!>               (يجب توفير عناصر الأقطار الفرعية (أو تحت القطر الفرعي)
!>                لـ D في المصفوفة E عند الإدخال)، و
!>            ب) إذا كان UPLO = 'U': العامل U في الجزء فوق القطري لـ A.
!>               إذا كان UPLO = 'L': العامل L في الجزء تحت القطري لـ A.
!>
!>          عند الخروج، إذا كان INFO = 0، معكوس المصفوفة الأصلية
!>          المتماثل.
!>             إذا كان UPLO = 'U': يتم تشكيل الجزء المثلثي العلوي
!>             للمعكوس ولا يتم الرجوع إلى الجزء السفلي من A؛
!>             إذا كان UPLO = 'L': يتم تشكيل الجزء المثلثي السفلي
!>             للمعكوس ولا يتم الرجوع إلى الجزء العلوي من A.
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرائد للمصفوفة A.  LDA >= max(1,N).
!>

E

!>          E هي مصفوفة COMPLEX*16، الأبعاد (N)
!>          عند الإدخال، تحتوي على عناصر الأقطار الفرعية (أو تحت
!>          القطر الفرعي) للمصفوفة القطرية الكتلية المتماثلة D
!>          ذات الأقطار 1x1 أو 2x2، حيث
!>          إذا كان UPLO = 'U': E(i) = D(i-1,i)، i=2:N، E(1) لا تتم الإشارة
!>          إليها؛
!>          إذا كان UPLO = 'L': E(i) = D(i+1,i)، i=1:N-1، E(N) لا تتم الإشارة
!>          إليها.
!>
!>          ملاحظة: بالنسبة للقطر الكتلية 1x1 D(k)، حيث
!>          1 <= k <= N، لا تتم الإشارة إلى العنصر E(k) في كلتا
!>          الحالتين UPLO = 'U' أو UPLO = 'L'.
!>

IPIV

!>          IPIV هي مصفوفة INTEGER، الأبعاد (N)
!>          تفاصيل التبادلات وهيكل القطر D كما تم تحديدهما
!>          بواسطة ZSYTRF_RK أو ZSYTRF_BK.
!>

WORK

!>          WORK هي مصفوفة COMPLEX*16، الأبعاد (N+NB+1,NB+3).
!>

NB

!>          NB هو INTEGER
!>          حجم الكتلة.
!>

INFO

!>          INFO هو INTEGER
!>          = 0: خروج ناجح
!>          < 0: إذا كان INFO = -i، كان المعامل i غير قانوني
!>          > 0: إذا كان INFO = i، فإن D(i,i) = 0؛ المصفوفة مفردة ولا يمكن
!>               حساب معكوسها.
!>
