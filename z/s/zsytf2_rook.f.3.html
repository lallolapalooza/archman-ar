توثيق الوظيفة/الإجراء الفرعي

subroutine zsytf2_rook (character uplo, integer n, complex*16, dimension( lda, * ) a, integer lda, integer, dimension( * ) ipiv, integer info)

ZSYTF2_ROOK تحسب تحليل مصفوفة معقدة ومتناظرة غير محددة باستخدام طريقة التبديل القطري المحدود لـ Bunch-Kaufman ("rook") (خوارزمية غير محظورة).

الغرض:

!>
!> ZSYTF2_ROOK تحسب تحليل مصفوفة متناظرة معقدة A
!> باستخدام طريقة التبديل القطري المحدود لـ Bunch-Kaufman:
!>
!>    A = U*D*U**T  أو  A = L*D*L**T
!>
!> حيث U (أو L) هو حاصل ضرب مصفوفات الوحدة المثلثية العلوية (السفلية)
!> ومصفوفات التبديل، U**T هو منقول U، و D متناظرة و
!> قطرية كتلة مع كتل قطرية بحجم 1x1 و 2x2.
!>
!> هذه هي النسخة غير المحظورة من الخوارزمية، والتي تستدعي BLAS المستوى 2.
!>

المعلمات

UPLO 
!>          UPLO هو CHARACTER*1
!>          يحدد ما إذا كان الجزء المثلثي العلوي أو السفلي للمصفوفة المتناظرة A
!>          مخزنًا:
!>          = 'U':  مثلثي علوي
!>          = 'L':  مثلثي سفلي
!>

N

!>          N هو INTEGER
!>          ترتيب المصفوفة A.  N >= 0.
!>

A

!>          A هي مصفوفة COMPLEX*16، البعد (LDA,N)
!>          عند الإدخال، المصفوفة المتناظرة A.  إذا كان UPLO = 'U'، فإن الجزء
!>          المثلثي العلوي بحجم n x n من A يحتوي على الجزء المثلثي العلوي
!>          للمصفوفة A، والجزء المثلثي السفلي الصارم من A غير مستخدم.  إذا كان
!>          UPLO = 'L'، فإن الجزء المثلثي السفلي بحجم n x n من A يحتوي على الجزء
!>          المثلثي السفلي للمصفوفة A، والجزء المثلثي العلوي الصارم من A غير مستخدم.
!>
!>          عند الخروج، مصفوفة الوحدة القطرية D والمعاملات المستخدمة
!>          للحصول على العامل U أو L (انظر أدناه لمزيد من التفاصيل).
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرائد للمصفوفة A.  LDA >= max(1,N).
!>

IPIV

!>          IPIV هي مصفوفة INTEGER، البعد (N)
!>          تفاصيل التبادلات وهيكل كتلة D.
!>
!>          إذا كان UPLO = 'U':
!>             إذا كان IPIV(k) > 0، فإن الصفوف والأعمدة k و IPIV(k)
!>             تم تبديلها وكانت D(k,k) كتلة قطرية بحجم 1x1.
!>
!>             إذا كان IPIV(k) < 0 و IPIV(k-1) < 0، فإن الصفوف
!>             والأعمدة k و -IPIV(k) تم تبديلها والصفوف
!>             والأعمدة k-1 و -IPIV(k-1) تم تبديلها،
!>             D(k-1:k,k-1:k) هي كتلة قطرية بحجم 2x2.
!>
!>          إذا كان UPLO = 'L':
!>             إذا كان IPIV(k) > 0، فإن الصفوف والأعمدة k و IPIV(k)
!>             تم تبديلها وكانت D(k,k) كتلة قطرية بحجم 1x1.
!>
!>             إذا كان IPIV(k) < 0 و IPIV(k+1) < 0، فإن الصفوف
!>             والأعمدة k و -IPIV(k) تم تبديلها والصفوف
!>             والأعمدة k+1 و -IPIV(k+1) تم تبديلها،
!>             D(k:k+1,k:k+1) هي كتلة قطرية بحجم 2x2.
!>

INFO

!>          INFO هو INTEGER
!>          = 0: خروج ناجح
!>          < 0: إذا كان INFO = -k، فإن المعامل k كان له قيمة غير قانونية
!>          > 0: إذا كان INFO = k، فإن D(k,k) يساوي صفرًا بالضبط.  تم إكمال
!>               التحليل، ولكن مصفوفة الكتلة القطرية D
!>               متفردة تمامًا، وسيحدث قسمة على صفر إذا
!>               تم استخدامها لحل نظام معادلات.
!>

تفاصيل إضافية:

!>
!>  إذا كان UPLO = 'U'، فإن A = U*D*U**T، حيث
!>     U = P(n)*U(n)* ... *P(k)U(k)* ...,
!>  أي، U هو حاصل ضرب الحدود P(k)*U(k)، حيث تقل k من n إلى
!>  1 بخطوات 1 أو 2، و D هي مصفوفة كتلة قطرية ذات كتل قطرية
!>  بحجم 1x1 و 2x2 هي D(k).  P(k) هي مصفوفة تبديل كما
!>  هو محدد بواسطة IPIV(k)، و U(k) هي مصفوفة وحدة علوية، بحيث إذا كانت
!>  الكتلة القطرية D(k) من الرتبة s (s = 1 أو 2)، فإن
!>
!>             (   I    v    0   )   k-s
!>     U(k) =  (   0    I    0   )   s
!>             (   0    0    I   )   n-k
!>                k-s   s   n-k
!>
!>  إذا كان s = 1، فإن D(k) يتجاوز A(k,k)، و v يتجاوز A(1:k-1,k).
!>  إذا كان s = 2، فإن الجزء العلوي من D(k) يتجاوز A(k-1,k-1)، A(k-1,k)،
!>  و A(k,k)، و v يتجاوز A(1:k-2,k-1:k).
!>
!>  إذا كان UPLO = 'L'، فإن A = L*D*L**T، حيث
!>     L = P(1)*L(1)* ... *P(k)*L(k)* ...,
!>  أي، L هو حاصل ضرب الحدود P(k)*L(k)، حيث تزيد k من 1 إلى
!>  n بخطوات 1 أو 2، و D هي مصفوفة كتلة قطرية ذات كتل قطرية
!>  بحجم 1x1 و 2x2 هي D(k).  P(k) هي مصفوفة تبديل كما
!>  هو محدد بواسطة IPIV(k)، و L(k) هي مصفوفة وحدة سفلية، بحيث إذا كانت
!>  الكتلة القطرية D(k) من الرتبة s (s = 1 أو 2)، فإن
!>
!>             (   I    0     0   )  k-1
!>     L(k) =  (   0    I     0   )  s
!>             (   0    v     I   )  n-k-s+1
!>                k-1   s  n-k-s+1
!>
!>  إذا كان s = 1، فإن D(k) يتجاوز A(k,k)، و v يتجاوز A(k+1:n,k).
!>  إذا كان s = 2، فإن الجزء السفلي من D(k) يتجاوز A(k,k)، A(k+1,k)،
!>  و A(k+1,k+1)، و v يتجاوز A(k+2:n,k:k+1).
!>
