توثيق الوظيفة/الروتين الفرعي

subroutine zsymm (character side, character uplo, integer m, integer n, complex*16 alpha, complex*16, dimension(lda,*) a, integer lda, complex*16, dimension(ldb,*) b, integer ldb, complex*16 beta, complex*16, dimension(ldc,*) c, integer ldc)

ZSYMM

الغرض:

!>
!> ZSYMM  يقوم بواحدة من عمليات ضرب المصفوفات:
!>
!>    C := alpha*A*B + beta*C,
!>
!> أو
!>
!>    C := alpha*B*A + beta*C,
!>
!> حيث  alpha  و beta  هما عددان سكالار، A هي مصفوفة متماثلة و  B و
!> C هما مصفوفتان من الحجم m × n.
!>

المعاملات

SIDE 
!>          SIDE is CHARACTER*1
!>           عند الإدخال،  SIDE  يحدد ما إذا كانت المصفوفة المتماثلة  A
!>           تظهر على  اليسار  أو اليمين  في العملية كما يلي:
!>
!>              SIDE = 'L' or 'l'   C := alpha*A*B + beta*C,
!>
!>              SIDE = 'R' or 'r'   C := alpha*B*A + beta*C,
!>

UPLO

!>          UPLO is CHARACTER*1
!>           عند الإدخال،   UPLO  يحدد ما إذا كان الجزء  العلوي  أو  السفلي
!>           المثلثي  من المصفوفة المتماثلة   A  سيتم  الرجوع إليه كما يلي:
!>
!>              UPLO = 'U' or 'u'   فقط الجزء العلوي المثلثي من المصفوفة
!>                                  المتماثلة سيتم الرجوع إليه.
!>
!>              UPLO = 'L' or 'l'   فقط الجزء السفلي المثلثي من المصفوفة
!>                                  المتماثلة سيتم الرجوع إليه.
!>

M

!>          M is INTEGER
!>           عند الإدخال،  M  يحدد عدد صفوف المصفوفة  C.
!>           يجب أن يكون  M  على الأقل صفرًا.
!>

N

!>          N is INTEGER
!>           عند الإدخال، N يحدد عدد أعمدة المصفوفة C.
!>           يجب أن يكون N  على الأقل صفرًا.
!>

ALPHA

!>          ALPHA is COMPLEX*16
!>           عند الإدخال،  ALPHA  يحدد العدد السكالار alpha.
!>

A

!>          A is COMPLEX*16 array, dimension ( LDA, ka ), where ka is
!>           m  when  SIDE = 'L' or 'l'  and is n  otherwise.
!>           قبل الإدخال مع  SIDE = 'L' or 'l'،  يجب أن يحتوي الجزء  m × m
!>           من المصفوفة  A  على المصفوفة المتماثلة، بحيث
!>           عندما  UPLO = 'U' or 'u'،  يجب أن يحتوي الجزء العلوي المثلثي
!>           الرائد  m × m  من المصفوفة  A  على الجزء العلوي المثلثي
!>           من المصفوفة المتماثلة والجزء  السفلي  المثلثي  تمامًا
!>           من  A  لا يتم الرجوع إليه، وعندما  UPLO = 'L' or 'l'،
!>           يجب أن يحتوي الجزء السفلي المثلثي  الرائد  m × m  من المصفوفة  A
!>           على الجزء السفلي المثلثي  من المصفوفة المتماثلة والجزء
!>           العلوي  المثلثي  تمامًا  من  A  لا يتم الرجوع إليه.
!>           قبل الإدخال مع  SIDE = 'R' or 'r'،  يجب أن يحتوي الجزء  n × n
!>           من المصفوفة  A  على المصفوفة المتماثلة، بحيث
!>           عندما  UPLO = 'U' or 'u'،  يجب أن يحتوي الجزء العلوي المثلثي
!>           الرائد  n × n  من المصفوفة  A  على الجزء العلوي المثلثي
!>           من المصفوفة المتماثلة والجزء  السفلي  المثلثي  تمامًا
!>           من  A  لا يتم الرجوع إليه، وعندما  UPLO = 'L' or 'l'،
!>           يجب أن يحتوي الجزء السفلي المثلثي  الرائد  n × n  من المصفوفة  A
!>           على الجزء السفلي المثلثي  من المصفوفة المتماثلة والجزء
!>           العلوي  المثلثي  تمامًا  من  A  لا يتم الرجوع إليه.
!>

LDA

!>          LDA is INTEGER
!>           عند الإدخال،  LDA  يحدد البعد الأول لـ A كما تم تعريفه
!>           في  البرنامج  (الفرعي)  المستدعي. عندما  SIDE = 'L' or 'l'  then
!>           يجب أن يكون LDA  على الأقل  max( 1, m )، وإلا  يجب أن يكون LDA
!>           على الأقل max( 1, n ).
!>

B

!>          B is COMPLEX*16 array, dimension ( LDB, N )
!>           قبل الإدخال، يجب أن يحتوي الجزء  m × n  الرائد  من المصفوفة  B
!>           على المصفوفة B.
!>

LDB

!>          LDB is INTEGER
!>           عند الإدخال،  LDB  يحدد البعد الأول لـ B كما تم تعريفه
!>           في  البرنامج  (الفرعي)  المستدعي.   يجب أن يكون LDB
!>           على الأقل  max( 1, m ).
!>

BETA

!>          BETA is COMPLEX*16
!>           عند الإدخال،  BETA  يحدد العدد السكالار  beta.  عندما
!>           يتم توفير BETA  كصفر  فإن C  لا تحتاج إلى تعيين عند الإدخال.
!>

C

!>          C is COMPLEX*16 array, dimension ( LDC, N )
!>           قبل الإدخال، يجب أن يحتوي الجزء  m × n  الرائد  من المصفوفة  C
!>           على المصفوفة  C،  باستثناء عندما يكون  beta  صفرًا، في
!>           هذه الحالة  لا تحتاج C  إلى تعيين عند الإدخال.
!>           عند الخروج، يتم الكتابة فوق المصفوفة  C  بالمصفوفة  المحدثة
!>           من الحجم  m × n.
!>

LDC

!>          LDC is INTEGER
!>           عند الإدخال،  LDC  يحدد البعد الأول لـ C كما تم تعريفه
!>           في  البرنامج  (الفرعي)  المستدعي.   يجب أن يكون LDC
!>           على الأقل  max( 1, m ).
!>
