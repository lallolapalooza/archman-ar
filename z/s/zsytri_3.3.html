subroutine zsytri_3 (character uplo, integer n, complex*16, dimension( lda, * ) a, integer lda, complex*16, dimension( * ) e, integer, dimension( * ) ipiv, complex*16, dimension( * ) work, integer lwork, integer info)

ZSYTRI_3

الغرض:

!> ZSYTRI_3 يحسب معكوس مصفوفة متماثلة غير محددة معقدة A باستخدام التحليل الذي تم حسابه بواسطة ZSYTRF_RK أو ZSYTRF_BK:
!>
!>     A = P*U*D*(U**T)*(P**T) أو A = P*L*D*(L**T)*(P**T),
!>
!> حيث U (أو L) هي مصفوفة قطرية علوية (أو سفلية) الوحدة،
!> U**T (أو L**T) هي منقولة U (أو L)، P هي مصفوفة تبديل، P**T هي منقولة P، و D هي مصفوفة متماثلة وقطرية كتلة مع كتل قطرية 1x1 و 2x2.
!>
!> ZSYTRI_3 يضبط البعد الرائد لمساحة العمل قبل استدعاء ZSYTRI_3X التي تحسب بالفعل معكوس المصفوفة. هذا هو الإصدار المكتل من الخوارزمية، يستدعي BLAS المستوى 3.
!>

المعلمات

UPLO 
!>          UPLO هو CHARACTER*1
!>          يحدد ما إذا كانت تفاصيل التحليل مخزنة كمصفوفة مثلثة علوية أو سفلية.
!>          = 'U':  يتم تخزين المثلث العلوي لـ A؛
!>          = 'L':  يتم تخزين المثلث السفلي لـ A.
!>

N

!>          N هو INTEGER
!>          رتبة المصفوفة A.  N >= 0.
!>

A

!>          A هي مصفوفة COMPLEX*16، البعد (LDA,N)
!>          عند الإدخال، القطر للمصفوفة القطرية الكتلة D والعوامل U أو L كما تم حسابها بواسطة ZSYTRF_RK و ZSYTRF_BK:
!>            أ) فقط العناصر القطرية للمصفوفة القطرية الكتلة المتماثلة D على قطر A، أي D(k,k) = A(k,k)؛
!>               (يجب توفير العناصر فوق القطرية (أو تحت القطرية) لـ D في المصفوفة E عند الإدخال)، و
!>            ب) إذا كان UPLO = 'U': العامل U في الجزء فوق القطري لـ A.
!>               إذا كان UPLO = 'L': العامل L في الجزء تحت القطري لـ A.
!>
!>          عند الخروج، إذا كان INFO = 0، فإن معكوس المصفوفة الأصلية المتماثلة.
!>             إذا كان UPLO = 'U': يتم تشكيل الجزء المثلث العلوي للمعكوس ولا يتم الرجوع إلى الجزء السفلي من A؛
!>             إذا كان UPLO = 'L': يتم تشكيل الجزء المثلث السفلي للمعكوس ولا يتم الرجوع إلى الجزء العلوي من A.
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرائد للمصفوفة A.  LDA >= max(1,N).
!>

E

!>          E هي مصفوفة COMPLEX*16، البعد (N)
!>          عند الإدخال، تحتوي على العناصر فوق القطرية (أو تحت القطرية) للمصفوفة القطرية الكتلة المتماثلة D
!>          مع كتل قطرية 1x1 أو 2x2، حيث
!>          إذا كان UPLO = 'U': E(i) = D(i-1,i)، i=2:N، E(1) لا يتم الرجوع إليها؛
!>          إذا كان UPLO = 'L': E(i) = D(i+1,i)، i=1:N-1، E(N) لا يتم الرجوع إليها.
!>
!>          ملاحظة: بالنسبة لكتلة قطرية 1x1 D(k)، حيث
!>          1 <= k <= N، لا يتم الرجوع إلى العنصر E(k) في كل من
!>          حالات UPLO = 'U' أو UPLO = 'L'.
!>

IPIV

!>          IPIV هي مصفوفة INTEGER، البعد (N)
!>          تفاصيل التبادلات وهيكل الكتلة لـ D كما تم تحديده بواسطة ZSYTRF_RK أو ZSYTRF_BK.
!>

WORK

!>          WORK هي مصفوفة COMPLEX*16، البعد (N+NB+1)*(NB+3).
!>          عند الخروج، إذا كان INFO = 0، فإن WORK(1) ترجع LWORK المثلى.
!>

LWORK

!>          LWORK هو INTEGER
!>          طول WORK. LWORK >= (N+NB+1)*(NB+3).
!>
!>          إذا كان LDWORK = -1، فسيتم افتراض استعلام مساحة العمل؛
!>          يحسب الروتين فقط الحجم الأمثل لمساحة العمل المثلى
!>          مصفوفة WORK، ويرجع هذا الحجم كإدخال أول
!>          لمصفوفة WORK، ولا يتم إصدار أي رسالة خطأ متعلقة بـ
!>          LWORK بواسطة XERBLA.
!>

INFO

!>          INFO هو INTEGER
!>          = 0: خروج ناجح
!>          < 0: إذا كان INFO = -i، فإن الوسيط i كان له قيمة غير قانونية
!>          > 0: إذا كان INFO = i، D(i,i) = 0؛ المصفوفة مفردة ولا يمكن حساب معكوسها.
!>
