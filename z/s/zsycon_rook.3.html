توثيق الوظيفة/الروتين الفرعي

subroutine zsycon_rook (character uplo, integer n, complex*16, dimension( lda, * ) a, integer lda, integer, dimension( * ) ipiv, double precision anorm, double precision rcond, complex*16, dimension( * ) work, integer info)

ZSYCON_ROOK

الغرض:

!>
!> ZSYCON_ROOK تقدر مقلوب عدد الشرط (في المعيار 1) للمصفوفة المتماثلة المعقدة A
!> باستخدام التحليل A = U*D*U**T أو A = L*D*L**T المحسوب بواسطة ZSYTRF_ROOK.
!>
!> يتم الحصول على تقدير لـ norm(inv(A))، ويتم حساب مقلوب عدد الشرط على أنه
!> RCOND = 1 / (ANORM * norm(inv(A))).
!>

المعلمات

UPLO 
!>          UPLO is CHARACTER*1
!>          يحدد ما إذا كانت تفاصيل التحليل مخزنة
!>          كمصفوفة مثلثية عليا أو سفلى.
!>          = 'U':  مثلثية عليا، الشكل هو A = U*D*U**T;
!>          = 'L':  مثلثية سفلى، الشكل هو A = L*D*L**T.
!>

N

!>          N is INTEGER
!>          ترتيب المصفوفة A.  N >= 0.
!>

A

!>          A is COMPLEX*16 array, dimension (LDA,N)
!>          المصفوفة القطرية المكتلة D والمضاعفات المستخدمة
!>          للحصول على العامل U أو L كما تم حسابه بواسطة ZSYTRF_ROOK.
!>

LDA

!>          LDA is INTEGER
!>          البعد الرئيسي للمصفوفة A.  LDA >= max(1,N).
!>

IPIV

!>          IPIV is INTEGER array, dimension (N)
!>          تفاصيل التبادلات والبنية المكتلة لـ D
!>          كما تم تحديده بواسطة ZSYTRF_ROOK.
!>

ANORM

!>          ANORM is DOUBLE PRECISION
!>          المعيار 1 للمصفوفة الأصلية A.
!>

RCOND

!>          RCOND is DOUBLE PRECISION
!>          مقلوب عدد الشرط للمصفوفة A،
!>          محسوب كـ RCOND = 1/(ANORM * AINVNM)، حيث AINVNM هو
!>          تقدير للمعيار 1 لـ inv(A) المحسوب في هذا الروتين.
!>

WORK

!>          WORK is COMPLEX*16 array, dimension (2*N)
!>

INFO

!>          INFO is INTEGER
!>          = 0:  انتهاء ناجح
!>          < 0:  إذا كان INFO = -i، فإن المعامل i كان له قيمة غير قانونية
!>
