subroutine zsytrs_aa (character uplo, integer n, integer nrhs, complex*16, dimension( lda, * ) a, integer lda, integer, dimension( * ) ipiv, complex*16, dimension( ldb, * ) b, integer ldb, complex*16, dimension( * ) work, integer lwork, integer info)

ZSYTRS_AA

الغرض:

!>
!> ZSYTRS_AA يحل نظام المعادلات الخطية A*X = B مع مصفوفة متناظرة مركبة A باستخدام
!> التحلل A = U**T*T*U أو A = L*T*L**T المحسوبة بواسطة ZSYTRF_AA.
!>

المعلمات

UPLO 
!>          UPLO هو CHARACTER*1
!>          يحدد ما إذا كانت تفاصيل التحلل مخزنة
!>          كمصفوفة مثلثة علوية أو سفلية.
!>          = 'U':  مثلثة علوية، الشكل هو A = U**T*T*U;
!>          = 'L':  مثلثة سفلية، الشكل هو A = L*T*L**T.
!>

N

!>          N هو INTEGER
!>          ترتيب المصفوفة A.  N >= 0.
!>

NRHS

!>          NRHS هو INTEGER
!>          عدد الأطراف اليمنى، أي عدد الأعمدة
!>          للمصفوفة B.  NRHS >= 0.
!>

A

!>          A هو مصفوفة COMPLEX*16، البعد (LDA,N)
!>          تفاصيل العوامل المحسوبة بواسطة ZSYTRF_AA.
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرائد للمصفوفة A.  LDA >= max(1,N).
!>

IPIV

!>          IPIV هو مصفوفة INTEGER، البعد (N)
!>          تفاصيل التبادلات المحسوبة بواسطة ZSYTRF_AA.
!>

B

!>          B هو مصفوفة COMPLEX*16، البعد (LDB,NRHS)
!>          عند الدخول، مصفوفة الطرف الأيمن B.
!>          عند الخروج، مصفوفة الحل X.
!>

LDB

!>          LDB هو INTEGER
!>          البعد الرائد للمصفوفة B.  LDB >= max(1,N).
!>

WORK

!>          WORK هو مصفوفة COMPLEX*16، البعد (MAX(1,LWORK))
!>

LWORK

!>          LWORK هو INTEGER
!>          بعد المصفوفة WORK. LWORK >= max(1,3*N-2).
!>

INFO

!>          INFO هو INTEGER
!>          = 0:  انتهاء ناجح
!>          < 0:  إذا كان INFO = -i، فإن المعامل i كان له قيمة غير قانونية
!>
