subroutine zsytri2 (character uplo, integer n, complex*16, dimension( lda, * ) a, integer lda, integer, dimension( * ) ipiv, complex*16, dimension( * ) work, integer lwork, integer info)

ZSYTRI2

الغرض:

!>
!> ZSYTRI2 يحسب معكوس مصفوفة متناظرة غير محددة COMPLEX*16
!> A باستخدام التحليل A = U*D*U**T أو A = L*D*L**T الذي تم حسابه بواسطة
!> ZSYTRF. ZSYTRI2 يضبط البعد الرئيسي لمساحة العمل
!> قبل استدعاء ZSYTRI2X الذي يحسب المعكوس فعليًا.
!>

المعلمات

UPLO 
!>          UPLO هو CHARACTER*1
!>          يحدد ما إذا كانت تفاصيل التحليل مخزنة
!>          كمصفوفة مثلثية علوية أو سفلية.
!>          = 'U':  مثلثية علوية، الشكل هو A = U*D*U**T؛
!>          = 'L':  مثلثية سفلية، الشكل هو A = L*D*L**T.
!>

N

!>          N هو INTEGER
!>          رتبة المصفوفة A.  N >= 0.
!>

A

!>          A هو مصفوفة COMPLEX*16، البعد (LDA,N)
!>          عند الإدخال، المصفوفة القطرية الكتلية D والعوامل
!>          المستخدمة للحصول على العامل U أو L كما هو محسوب بواسطة ZSYTRF.
!>
!>          عند الخروج، إذا كان INFO = 0، فإن معكوس (المتناظر) للمصفوفة
!>          الأصلية.  إذا كان UPLO = 'U'، فإن الجزء المثلثي العلوي
!>          للمعكوس مشكل والجزء من A أسفل القطر غير مستخدم؛ إذا كان UPLO = 'L' فإن الجزء المثلثي السفلي
!>          للمعكوس مشكل والجزء من A فوق القطر غير مستخدم.
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرئيسي للمصفوفة A.  LDA >= max(1,N).
!>

IPIV

!>          IPIV هو مصفوفة INTEGER، البعد (N)
!>          تفاصيل التبادلات وهيكل الكتل لـ D
!>          كما تم تحديده بواسطة ZSYTRF.
!>

WORK

!>          WORK هو مصفوفة COMPLEX*16، البعد (N+NB+1)*(NB+3)
!>

LWORK

!>          LWORK هو INTEGER
!>          بعد مصفوفة WORK.
!>          WORK حجمها >= (N+NB+1)*(NB+3)
!>          إذا كان LDWORK = -1، فإنه يفترض استعلام مساحة العمل؛ الروتين
!>           يحسب:
!>              - الحجم الأمثل لمصفوفة WORK، يعيد
!>          هذه القيمة كالإدخال الأول في مصفوفة WORK،
!>              - ولا يتم إصدار أي رسالة خطأ متعلقة بـ LDWORK بواسطة XERBLA.
!>

INFO

!>          INFO هو INTEGER
!>          = 0: خروج ناجح
!>          < 0: إذا كان INFO = -i، فإن الوسيط i كان له قيمة غير قانونية
!>          > 0: إذا كان INFO = i، فإن D(i,i) = 0؛ المصفوفة منفردة ولا يمكن حساب معكوسها.
!>
