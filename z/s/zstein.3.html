توثيق الوظيفة/الروتين الفرعي

subroutine zstein (integer n, double precision, dimension( * ) d, double precision, dimension( * ) e, integer m, double precision, dimension( * ) w, integer, dimension( * ) iblock, integer, dimension( * ) isplit, complex*16, dimension( ldz, * ) z, integer ldz, double precision, dimension( * ) work, integer, dimension( * ) iwork, integer, dimension( * ) ifail, integer info)

ZSTEIN

الغرض:

!>
!> ZSTEIN يحسب المتجهات الذاتية لمصفوفة ثلاثية القطر متناظرة حقيقية T
!> المقابلة للقيم الذاتية المحددة، باستخدام التكرار العكسي.
!>
!> يتم تحديد الحد الأقصى لعدد التكرارات لكل متجه ذاتي بواسطة
!> معلمة داخلية MAXITS (محددة حاليًا بـ 5).
!>
!> على الرغم من أن المتجهات الذاتية حقيقية، إلا أنها مخزنة في مصفوفة
!> مركبة، والتي يمكن تمريرها إلى ZUNMTR أو ZUPMTR للتحويل العكسي
!> إلى المتجهات الذاتية لمصفوفة هرميتية مركبة تم اختزالها إلى شكل ثلاثي
!> القطر.
!>
!>

المعلمات

N 
!>          N هو عدد صحيح
!>          رتبة المصفوفة.  N >= 0.
!>

D

!>          D هي مصفوفة DOUBLE PRECISION
!>          العناصر القطرية n للمصفوفة ثلاثية القطر T.
!>

E

!>          E هي مصفوفة DOUBLE PRECISION، حجم (N-1)
!>          العناصر تحت القطرية (n-1) للمصفوفة ثلاثية القطر
!>          T، مخزنة في العناصر من 1 إلى N-1.
!>

M

!>          M هو عدد صحيح
!>          عدد المتجهات الذاتية المراد العثور عليها.  0 <= M <= N.
!>

W

!>          W هي مصفوفة DOUBLE PRECISION، حجم (N)
!>          العناصر M الأولى من W تحتوي على القيم الذاتية
!>          التي سيتم حساب المتجهات الذاتية لها.  يجب تجميع القيم الذاتية
!>          حسب الكتلة المنفصلة وترتيبها من الأصغر إلى الأكبر ضمن الكتلة.
!>          (يتوقع هنا مصفوفة الإخراج W من DSTEBZ مع ORDER = 'B'.)
!>

IBLOCK

!>          IBLOCK هي مصفوفة عدد صحيح، حجم (N)
!>          فهارس المصفوفات الفرعية المرتبطة بالقيم الذاتية المقابلة
!>          في W؛ IBLOCK(i)=1 إذا كانت القيمة الذاتية W(i) تنتمي إلى
!>          المصفوفة الفرعية الأولى من الأعلى، =2 إذا كانت W(i) تنتمي إلى
!>          المصفوفة الفرعية الثانية، إلخ.
!>          (يتوقع هنا مصفوفة الإخراج IBLOCK من DSTEBZ.)
!>

ISPLIT

!>          ISPLIT هي مصفوفة عدد صحيح، حجم (N)
!>          نقاط التقسيم، التي تنقسم عندها T إلى مصفوفات فرعية.
!>          المصفوفة الفرعية الأولى تتكون من الصفوف/الأعمدة من 1 إلى
!>          ISPLIT( 1 )، والثانية من الصفوف/الأعمدة ISPLIT( 1 )+1
!>          حتى ISPLIT( 2 )، إلخ.
!>          (يتوقع هنا مصفوفة الإخراج ISPLIT من DSTEBZ.)
!>

Z

!>          Z هي مصفوفة COMPLEX*16، حجم (LDZ, M)
!>          المتجهات الذاتية المحسوبة.  المتجه الذاتي المرتبط
!>          بالقيمة الذاتية W(i) مخزن في العمود i من
!>          Z.  أي متجه يفشل في التقارب يتم تعيينه إلى تكراره الحالي
!>          بعد MAXITS تكرار.
!>          الأجزاء التخيلية للمتجهات الذاتية يتم تعيينها إلى الصفر.
!>

LDZ

!>          LDZ هو عدد صحيح
!>          البعد الرائد للمصفوفة Z.  LDZ >= max(1,N).
!>

WORK

!>          WORK هي مصفوفة DOUBLE PRECISION، حجم (5*N)
!>

IWORK

!>          IWORK هي مصفوفة عدد صحيح، حجم (N)
!>

IFAIL

!>          IFAIL هي مصفوفة عدد صحيح، حجم (M)
!>          عند الخروج الطبيعي، تكون جميع عناصر IFAIL صفرًا.
!>          إذا فشل متجه ذاتي واحد أو أكثر في التقارب بعد
!>          MAXITS تكرار، فسيتم تخزين فهارسها في
!>          المصفوفة IFAIL.
!>

INFO

!>          INFO هو عدد صحيح
!>          = 0: خروج ناجح
!>          < 0: إذا كان INFO = -i، فإن المعامل i كان له قيمة غير قانونية
!>          > 0: إذا كان INFO = i، فإن i متجه ذاتي فشل في التقارب
!>               في MAXITS تكرار.  يتم تخزين فهارسها في
!>               المصفوفة IFAIL.
!>

المعلمات الداخلية:

!>  MAXITS  INTEGER, default = 5
!>          الحد الأقصى لعدد التكرارات التي يتم إجراؤها.
!>
!>  EXTRA   INTEGER, default = 2
!>          عدد التكرارات التي يتم إجراؤها بعد استيفاء معيار نمو المعيار
!>          ، يجب أن يكون على الأقل 1.
!>
