subroutine zstegr (character jobz, character range, integer n, double precision, dimension( * ) d, double precision, dimension( * ) e, double precision vl, double precision vu, integer il, integer iu, double precision abstol, integer m, double precision, dimension( * ) w, complex*16, dimension( ldz, * ) z, integer ldz, integer, dimension( * ) isuppz, double precision, dimension( * ) work, integer lwork, integer, dimension( * ) iwork, integer liwork, integer info)

ZSTEGR

الغرض:

!>
!> ZSTEGR يحسب القيم الذاتية المختارة، واختياريًا، المتجهات الذاتية
!> لمصفوفة ثلاثية الأقطار متماثلة حقيقية T. أي مصفوفة غير مختزلة كهذه لديها
!> مجموعة محددة جيدًا من القيم الذاتية الحقيقية المختلفة زوجيًا، والمتجهات الذاتية
!> الحقيقية المقابلة متعامدة زوجيًا.
!>
!> يمكن حساب الطيف إما بالكامل أو جزئيًا عن طريق تحديد
!> إما فاصل زمني (VL,VU] أو نطاق من الفهارس IL:IU للقيم الذاتية المطلوبة.
!>
!> ZSTEGR هو غلاف توافق حول الروتين المحسن ZSTEMR.
!> انظر ZSTEMR لمزيد من التفاصيل.
!>
!> تغيير مهم واحد هو أن المعلمة ABSTOL لم تعد تقدم أي فائدة وبالتالي
!> لم تعد مستخدمة.
!>
!> ملاحظة: تعمل ZSTEGR و ZSTEMR فقط على الآلات التي تتبع
!> معيار الفاصلة العائمة IEEE-754 في معالجتها لللانهايات و
!> NaNs. قد يؤدي التنفيذ العادي إلى إنشاء هذه القيم الاستثنائية وبالتالي
!> قد يتوقف بسبب استثناء فاصلة عائمة في بيئات لا
!> تتوافق مع معيار IEEE-754.
!>

المعلمات

JOBZ 
!>          JOBZ هو CHARACTER*1
!>          = 'N':  حساب القيم الذاتية فقط؛
!>          = 'V':  حساب القيم الذاتية والمتجهات الذاتية.
!>

RANGE

!>          RANGE هو CHARACTER*1
!>          = 'A': سيتم العثور على جميع القيم الذاتية.
!>          = 'V': سيتم العثور على جميع القيم الذاتية في الفاصل الزمني نصف المفتوح (VL,VU]
!>                 .
!>          = 'I': سيتم العثور على القيم الذاتية من IL إلى IU.
!>

N

!>          N هو INTEGER
!>          رتبة المصفوفة.  N >= 0.
!>

D

!>          D هو مصفوفة DOUBLE PRECISION، البعد (N)
!>          عند الدخول، العناصر القطرية N للمصفوفة ثلاثية الأقطار
!>          T. عند الخروج، يتم الكتابة فوق D.
!>

E

!>          E هو مصفوفة DOUBLE PRECISION، البعد (N)
!>          عند الدخول، العناصر تحت القطرية (N-1) للمصفوفة ثلاثية الأقطار
!>          T في العناصر من 1 إلى N-1 من E. لا يلزم تعيين E(N) عند
!>          المدخل، ولكن يتم استخدامه داخليًا كمساحة عمل.
!>          عند الخروج، يتم الكتابة فوق E.
!>

VL

!>          VL هو DOUBLE PRECISION
!>
!>          إذا كان RANGE='V'، الحد الأدنى للفاصل الزمني لـ
!>          البحث عن القيم الذاتية. VL < VU.
!>          غير مرجعي إذا كان RANGE = 'A' أو 'I'.
!>

VU

!>          VU هو DOUBLE PRECISION
!>
!>          إذا كان RANGE='V'، الحد الأعلى للفاصل الزمني لـ
!>          البحث عن القيم الذاتية. VL < VU.
!>          غير مرجعي إذا كان RANGE = 'A' أو 'I'.
!>

IL

!>          IL هو INTEGER
!>
!>          إذا كان RANGE='I'، فهرس
!>          أصغر قيمة ذاتية سيتم إرجاعها.
!>          1 <= IL <= IU <= N، إذا كان N > 0.
!>          غير مرجعي إذا كان RANGE = 'A' أو 'V'.
!>

IU

!>          IU هو INTEGER
!>
!>          إذا كان RANGE='I'، فهرس
!>          أكبر قيمة ذاتية سيتم إرجاعها.
!>          1 <= IL <= IU <= N، إذا كان N > 0.
!>          غير مرجعي إذا كان RANGE = 'A' أو 'V'.
!>

ABSTOL

!>          ABSTOL هو DOUBLE PRECISION
!>          غير مستخدم. كان التفاوت في الخطأ المطلق لـ
!>          القيم الذاتية/المتجهات الذاتية في الإصدارات السابقة.
!>

M

!>          M هو INTEGER
!>          إجمالي عدد القيم الذاتية التي تم العثور عليها.  0 <= M <= N.
!>          إذا كان RANGE = 'A'، فإن M = N، وإذا كان RANGE = 'I'، فإن M = IU-IL+1.
!>

W

!>          W هو مصفوفة DOUBLE PRECISION، البعد (N)
!>          تحتوي العناصر الأولى M على القيم الذاتية المحددة في
!>          ترتيب تصاعدي.
!>

Z

!>          Z هو مصفوفة COMPLEX*16، البعد (LDZ, max(1,M) )
!>          إذا كان JOBZ = 'V'، وإذا كان INFO = 0، فإن الأعمدة الأولى لـ Z
!>          تحتوي على المتجهات الذاتية المتعامدة للمصفوفة T
!>          المقابلة للقيم الذاتية المحددة، مع العمود i-th
!>          لـ Z يحمل المتجه الذاتي المرتبط بـ W(i).
!>          إذا كان JOBZ = 'N'، فلا يتم الرجوع إلى Z.
!>          ملاحظة: يجب على المستخدم التأكد من أن ما لا يقل عن max(1,M) أعمدة
!>          مقدمة في المصفوفة Z؛ إذا كان RANGE = 'V'، فإن القيمة الدقيقة لـ M
!>          غير معروفة مسبقًا ويجب استخدام حد أعلى.
!>          توفير N أعمدة آمن دائمًا.
!>

LDZ

!>          LDZ هو INTEGER
!>          البعد الرائد للمصفوفة Z.  LDZ >= 1، وإذا
!>          كان JOBZ = 'V'، فإن LDZ >= max(1,N).
!>

ISUPPZ

!>          ISUPPZ هو مصفوفة INTEGER، البعد ( 2*max(1,M) )
!>          دعم المتجهات الذاتية في Z، أي الفهارس
!>          تشير إلى العناصر غير الصفرية في Z. المتجه الذاتي المحسوب i-th
!>          غير صفري فقط في العناصر ISUPPZ( 2*i-1 ) حتى
!>          ISUPPZ( 2*i ). هذا ذو صلة في حالة عندما تكون المصفوفة
!>          مقسومة. يتم الوصول إلى ISUPPZ فقط عندما يكون JOBZ هو 'V' و N > 0.
!>

WORK

!>          WORK هو مصفوفة DOUBLE PRECISION، البعد (LWORK)
!>          عند الخروج، إذا كان INFO = 0، فإن WORK(1) يرجع المثلى
!>          (والحد الأدنى) LWORK.
!>

LWORK

!>          LWORK هو INTEGER
!>          بعد المصفوفة WORK. LWORK >= max(1,18*N)
!>          إذا كان JOBZ = 'V'، و LWORK >= max(1,12*N) إذا كان JOBZ = 'N'.
!>          إذا كان LWORK = -1، فمن المفترض وجود استعلام مساحة عمل؛ الـ
!>          روتين يحسب فقط الحجم الأمثل لمصفوفة WORK، ويعيد
!>          هذه القيمة كأول إدخال في مصفوفة WORK، ولا يوجد خطأ
!>          يتم إصدار رسالة متعلقة بـ LWORK بواسطة XERBLA.
!>

IWORK

!>          IWORK هو مصفوفة INTEGER، البعد (LIWORK)
!>          عند الخروج، إذا كان INFO = 0، فإن IWORK(1) يرجع المثلى LIWORK.
!>

LIWORK

!>          LIWORK هو INTEGER
!>          بعد المصفوفة IWORK. LIWORK >= max(1,10*N)
!>          إذا كانت المتجهات الذاتية مطلوبة، و LIWORK >= max(1,8*N)
!>          إذا كان سيتم حساب القيم الذاتية فقط.
!>          إذا كان LIWORK = -1، فمن المفترض وجود استعلام مساحة عمل؛ الـ
!>          روتين يحسب فقط الحجم الأمثل لمصفوفة IWORK،
!>          يعيد هذه القيمة كأول إدخال في مصفوفة IWORK، و
!>          لا يتم إصدار رسالة خطأ متعلقة بـ LIWORK بواسطة XERBLA.
!>

INFO

!>          INFO هو INTEGER
!>          عند الخروج، INFO
!>          = 0:  خروج ناجح
!>          < 0:  إذا كان INFO = -i، فإن المعلمة i-th كانت قيمة غير قانونية
!>          > 0:  إذا كان INFO = 1X، خطأ داخلي في DLARRE،
!>                إذا كان INFO = 2X، خطأ داخلي في ZLARRV.
!>                هنا، الرقم X = ABS( IINFO ) < 10، حيث IINFO هو
!>                رمز الخطأ غير الصفري الذي تم إرجاعه بواسطة DLARRE أو
!>
