توثيق الوظيفة/الإجراء الفرعي

subroutine zsyt01_aa (character uplo, integer n, complex*16, dimension( lda, * ) a, integer lda, complex*16, dimension( ldafac, * ) afac, integer ldafac, integer, dimension( * ) ipiv, complex*16, dimension( ldc, * ) c, integer ldc, double precision, dimension( * ) rwork, double precision resid)

ZSYT01

الغرض:

!>
!> ZSYT01 يعيد بناء مصفوفة هيرميتية غير محددة A من عاملها L*D*L' أو U*D*U'
!> ويحسب معيار الخطأ
!>    norm( C - A ) / ( N * norm(A) * EPS )،
!> حيث C هي المصفوفة المعاد بناؤها و EPS هي دقة الآلة.
!>

المعلمات

UPLO 
!>          UPLO هو CHARACTER*1
!>          يحدد ما إذا كان الجزء العلوي أو السفلي المثلثي للمصفوفة
!>          الهيرميتية A مخزنًا:
!>          = 'U':  مثلثي علوي
!>          = 'L':  مثلثي سفلي
!>

N

!>          N هو INTEGER
!>          عدد الصفوف والأعمدة للمصفوفة A.  N >= 0.
!>

A

!>          A هو مصفوفة COMPLEX*16، حجمها (LDA,N)
!>          المصفوفة الهيرميتية الأصلية A.
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرائد للمصفوفة A.  LDA >= max(1,N)
!>

AFAC

!>          AFAC هو مصفوفة COMPLEX*16، حجمها (LDAFAC,N)
!>          الشكل المفسور للمصفوفة A.  AFAC يحتوي على المصفوفة القطرية الكتلية D
!>          والمضاعفات المستخدمة للحصول على العامل L أو U من عامل L*D*L'
!>          أو U*D*U' كما تم حسابه بواسطة ZSYTRF.
!>

LDAFAC

!>          LDAFAC هو INTEGER
!>          البعد الرائد للمصفوفة AFAC.  LDAFAC >= max(1,N).
!>

IPIV

!>          IPIV هي مصفوفة INTEGER، حجمها (N)
!>          فهارس المحور من ZSYTRF.
!>

C

!>          C هي مصفوفة COMPLEX*16، حجمها (LDC,N)
!>

LDC

!>          LDC هو INTEGER
!>          البعد الرائد للمصفوفة C.  LDC >= max(1,N).
!>

RWORK

!>          RWORK هي مصفوفة COMPLEX*16، حجمها (N)
!>

RESID

!>          RESID هو COMPLEX*16
!>          إذا كان UPLO = 'L'، norm(L*D*L' - A) / ( N * norm(A) * EPS )
!>          إذا كان UPLO = 'U'، norm(U*D*U' - A) / ( N * norm(A) * EPS )
!>
