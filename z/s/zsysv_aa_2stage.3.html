subroutine zsysv_aa_2stage (character uplo, integer n, integer nrhs, complex*16, dimension( lda, * ) a, integer lda, complex*16, dimension( * ) tb, integer ltb, integer, dimension( * ) ipiv, integer, dimension( * ) ipiv2, complex*16, dimension( ldb, * ) b, integer ldb, complex*16, dimension( * ) work, integer lwork, integer info)

ZSYSV_AA_2STAGE يحسب حل نظام المعادلات الخطية A * X = B للمصفوفات SY

الغرض:

!>
!> ZSYSV_AA_2STAGE يحسب حل نظام خطي معقد
!>    A * X = B,
!> حيث A مصفوفة متناظرة N × N و X و B مصفوفتان N × NRHS.
!>
!> يتم استخدام خوارزمية Aasen ذات المرحلتين لتحليل A إلى
!>    A = U**T * T * U,  إذا كان UPLO = 'U', أو
!>    A = L * T * L**T,  إذا كان UPLO = 'L',
!> حيث U (أو L) هي ناتج ضرب مصفوفة التبديل ومصفوفة مثلثية علوية (سفلية) وحدوية، و T مصفوفة متناظرة ونطاقية. ثم يتم تحليل T نطاقيًا باستخدام LU مع تحيز جزئي. ثم يتم استخدام الشكل المحلل لـ A لحل نظام المعادلات A * X = B.
!>
!> هذه هي النسخة المجمعة من الخوارزمية، والتي تستدعي BLAS المستوى 3.
!>

المعلمات

UPLO 
!>          UPLO هو CHARACTER*1
!>          = 'U':  يتم تخزين المثلث العلوي لـ A؛
!>          = 'L':  يتم تخزين المثلث السفلي لـ A.
!>

N

!>          N هو INTEGER
!>          رتبة المصفوفة A.  N >= 0.
!>

NRHS

!>          NRHS هو INTEGER
!>          عدد الأطراف اليمنى، أي عدد أعمدة
!>          المصفوفة B.  NRHS >= 0.
!>

A

!>          A هي مصفوفة COMPLEX*16، أبعادها (LDA,N)
!>          عند الإدخال، المصفوفة المتناظرة A.  إذا كان UPLO = 'U'، فإن الجزء
!>          العلوي المثلثي N × N الرائد في A يحتوي على الجزء
!>          المثلثي العلوي للمصفوفة A، والجزء السفلي
!>          المثلثي الصارم لـ A لا تتم الإشارة إليه.  إذا كان UPLO = 'L'، فإن الجزء
!>          السفلي المثلثي N × N الرائد في A يحتوي على الجزء
!>          المثلثي السفلي للمصفوفة A، والجزء العلوي
!>          المثلثي الصارم لـ A لا تتم الإشارة إليه.
!>
!>          عند الخروج، يتم تخزين L أسفل (أو أعلى) كتل الأقطار الفرعية،
!>          عندما يكون UPLO  هو 'L' (أو 'U').
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرائد للمصفوفة A.  LDA >= max(1,N).
!>

TB

!>          TB هي مصفوفة COMPLEX*16، أبعادها (LTB)
!>          عند الخروج، تفاصيل تحليل LU للمصفوفة النطاقية.
!>

LTB

!>          LTB هو INTEGER
!>          حجم المصفوفة TB. LTB >= 4*N، يستخدم داخليًا
!>          لاختيار NB بحيث يكون LTB >= (3*NB+1)*N.
!>
!>          إذا كان LTB = -1، فسيتم افتراض استعلام عن مساحة العمل؛ الروتين
!>          يحسب فقط الحجم الأمثل لـ LTB،
!>          ويعيد هذا القيمة كمدخل أول لـ TB،
!>          ولا يصدر رسالة خطأ متعلقة بـ LTB بواسطة XERBLA.
!>

IPIV

!>          IPIV هي مصفوفة INTEGER، أبعادها (N)
!>          عند الخروج، تحتوي على تفاصيل التبادلات، أي
!>          تم تبديل الصف والعمود k من A مع
!>          الصف والعمود IPIV(k).
!>

IPIV2

!>          IPIV2 هي مصفوفة INTEGER، أبعادها (N)
!>          عند الخروج، تحتوي على تفاصيل التبادلات، أي
!>          تم تبديل الصف والعمود k من T مع
!>          الصف والعمود IPIV(k).
!>

B

!>          B هي مصفوفة COMPLEX*16، أبعادها (LDB,NRHS)
!>          عند الإدخال، مصفوفة الطرف الأيمن B.
!>          عند الخروج، مصفوفة الحل X.
!>

LDB

!>          LDB هو INTEGER
!>          البعد الرائد للمصفوفة B.  LDB >= max(1,N).
!>

WORK

!>          WORK هي مساحة عمل COMPLEX*16 بحجم LWORK
!>

LWORK

!>          LWORK هو INTEGER
!>          حجم WORK. LWORK >= N، يستخدم داخليًا لاختيار NB
!>          بحيث يكون LWORK >= N*NB.
!>
!>          إذا كان LWORK = -1، فسيتم افتراض استعلام عن مساحة العمل؛ الروتين
!>          يحسب فقط الحجم الأمثل لمصفوفة WORK،
!>          ويعيد هذه القيمة كمدخل أول لمصفوفة WORK،
!>          ولا يصدر رسالة خطأ متعلقة بـ LWORK بواسطة XERBLA.
!>

INFO

!>          INFO هو INTEGER
!>          = 0:  انتهاء ناجح
!>          < 0:  إذا كان INFO = -i، فإن الوسيط i كان له قيمة غير قانونية.
!>          > 0:  إذا كان INFO = i، فشل تحليل LU النطاقي في العمود i
!>
