توثيق الوظيفة/الوحدة الفرعية

subroutine zsptrs (character uplo, integer n, integer nrhs, complex*16, dimension( * ) ap, integer, dimension( * ) ipiv, complex*16, dimension( ldb, * ) b, integer ldb, integer info)

ZSPTRS

الغرض:

!>
!> ZSPTRS يحل نظام المعادلات الخطية A*X = B مع مصفوفة متناظرة مركبة A مخزنة بصيغة مضغوطة باستخدام التحلل
!> A = U*D*U**T أو A = L*D*L**T المحسوب بواسطة ZSPTRF.
!>

المعلمات

UPLO 
!>          UPLO هو CHARACTER*1
!>          يحدد ما إذا كانت تفاصيل التحلل مخزنة كمصفوفة مثلثية علوية أو سفلية.
!>          = 'U':  مثلثية علوية، الشكل هو A = U*D*U**T؛
!>          = 'L':  مثلثية سفلية، الشكل هو A = L*D*L**T.
!>

N

!>          N هو INTEGER
!>          رتبة المصفوفة A.  N >= 0.
!>

NRHS

!>          NRHS هو INTEGER
!>          عدد الأطراف اليمنى، أي عدد أعمدة المصفوفة B.  NRHS >= 0.
!>

AP

!>          AP هي مصفوفة COMPLEX*16، البعد (N*(N+1)/2)
!>          المصفوفة القطرية الفرعية D والعوامل المستخدمة للحصول على العامل U أو L كما تم حسابه بواسطة ZSPTRF، مخزنة كمصفوفة مثلثية مضغوطة.
!>

IPIV

!>          IPIV هي مصفوفة INTEGER، البعد (N)
!>          تفاصيل التبادلات والبنية الفرعية لـ D كما تم تحديدها بواسطة ZSPTRF.
!>

B

!>          B هي مصفوفة COMPLEX*16، البعد (LDB,NRHS)
!>          عند الدخول: مصفوفة الطرف الأيمن B.
!>          عند الخروج: مصفوفة الحل X.
!>

LDB

!>          LDB هو INTEGER
!>          البعد الرئيسي للمصفوفة B.  LDB >= max(1,N).
!>

INFO

!>          INFO هو INTEGER
!>          = 0:  خروج ناجح
!>          < 0: إذا كان INFO = -i، فقد كان الوسيط i غير قانوني
!>
