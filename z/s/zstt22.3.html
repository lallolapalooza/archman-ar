subroutine zstt22 (integer n, integer m, integer kband, double precision, dimension( * ) ad, double precision, dimension( * ) ae, double precision, dimension( * ) sd, double precision, dimension( * ) se, complex*16, dimension( ldu, * ) u, integer ldu, complex*16, dimension( ldwork, * ) work, integer ldwork, double precision, dimension( * ) rwork, double precision, dimension( 2 ) result)

ZSTT22

الغرض:

!>
!> ZSTT22  يتحقق من مجموعة من M قيم ذاتية ومتجهات ذاتية،
!>
!>     A U = U S
!>
!> حيث A مصفوفة هيرميتية ثلاثية الأقطار، وأعمدة U وحدوية،
!> و S قطرية (إذا كان KBAND=0) أو مصفوفة هيرميتية ثلاثية الأقطار (إذا كان KBAND=1).
!> يتم إجراء اختبارين:
!>
!>    RESULT(1) = | U* A U - S | / ( |A| m ulp )
!>
!>    RESULT(2) = | I - U*U | / ( m ulp )
!>

المعلمات

N 
!>          N هو عدد صحيح (INTEGER)
!>          حجم المصفوفة. إذا كان صفرًا، فإن ZSTT22 لا تفعل شيئًا.
!>          يجب أن يكون أكبر من أو يساوي صفرًا.
!>

M

!>          M هو عدد صحيح (INTEGER)
!>          عدد أزواج القيم الذاتية التي يجب التحقق منها. إذا كان صفرًا، فإن ZSTT22
!!>          لا تفعل شيئًا. يجب أن يكون أكبر من أو يساوي صفرًا.
!>

KBAND

!>          KBAND هو عدد صحيح (INTEGER)
!>          عرض النطاق للمصفوفة S. يمكن أن يكون فقط صفرًا أو واحدًا.
!>          إذا كان صفرًا، فإن S قطرية، ولا يتم الاستشهاد بـ SE. إذا كان
!>          واحدًا، فإن S هي مصفوفة هيرميتية ثلاثية الأقطار.
!>

AD

!>          AD هو مصفوفة (DOUBLE PRECISION)، البعد (N)
!>          القطر للمصفوفة الأصلية (غير المفككة) A. A
!>          يفترض أنها مصفوفة هيرميتية ثلاثية الأقطار.
!>

AE

!>          AE هو مصفوفة (DOUBLE PRECISION)، البعد (N)
!>          ما فوق القطر للمصفوفة الأصلية (غير المفككة) A. A
!>          يفترض أنها مصفوفة هيرميتية ثلاثية الأقطار. AE(1) يتم تجاهلها،
!>          AE(2) هو العنصر (1,2) و (2,1)، وما إلى ذلك.
!>

SD

!>          SD هو مصفوفة (DOUBLE PRECISION)، البعد (N)
!>          القطر للمصفوفة القطرية (هيرميتية ثلاثية) S.
!>

SE

!>          SE هو مصفوفة (DOUBLE PRECISION)، البعد (N)
!>          ما فوق القطر للمصفوفة القطرية (هيرميتية ثلاثية) S.
!>          لا يتم الاستشهاد بها إذا كانت KBSND=0. إذا كان KBAND=1، فإن AE(1)
!>          يتم تجاهلها، SE(2) هو العنصر (1,2) و (2,1)، وما إلى ذلك.
!>

U

!>          U هو مصفوفة (DOUBLE PRECISION)، البعد (LDU, N)
!>          المصفوفة الوحدوية في التحلل.
!>

LDU

!>          LDU هو عدد صحيح (INTEGER)
!>          البعد الرئيسي لـ U. يجب أن يكون LDU أكبر من أو يساوي N.
!>

WORK

!>          WORK هو مصفوفة (COMPLEX*16)، البعد (LDWORK, M+1)
!>

LDWORK

!>          LDWORK هو عدد صحيح (INTEGER)
!>          البعد الرئيسي لـ WORK. يجب أن يكون LDWORK أكبر من أو يساوي
!>          max(1,M).
!>

RWORK

!>          RWORK هو مصفوفة (DOUBLE PRECISION)، البعد (N)
!>

RESULT

!>          RESULT هي مصفوفة (DOUBLE PRECISION)، البعد (2)
!>          القيم المحسوبة بواسطة الاختبارين الموصوفين أعلاه.
!>          القيم محدودة حاليًا بـ 1/ulp، لتجنب تجاوز السعة.
!>
