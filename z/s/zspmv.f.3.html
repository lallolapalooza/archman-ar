subroutine zspmv (character uplo, integer n, complex*16 alpha, complex*16, dimension( * ) ap, complex*16, dimension( * ) x, integer incx, complex*16 beta, complex*16, dimension( * ) y, integer incy)

ZSPMV تحسب ناتج ضرب مصفوفة-متجه للناقلات المعقدة باستخدام مصفوفة متناظرة معبأة معقدة

الغرض:

!>
!> ZSPMV  يقوم بعملية ضرب المصفوفة بالمتجه
!>
!>    y := alpha*A*x + beta*y,
!>
!> حيث alpha و beta هما عددان، x و y هما متجهان مكونان من n عنصر،
!> و A هي مصفوفة متناظرة بحجم n × n، مقدمة في شكل مضغوط.
!>

المعاملات

UPLO 
!>          UPLO هو CHARACTER*1
!>           عند الدخول، تحدد UPLO ما إذا كان الجزء العلوي أو السفلي
!>           من المثلثي للمصفوفة A مقدمًا في المصفوفة AP على النحو التالي:
!>
!>              UPLO = 'U' أو 'u'   يتم تقديم الجزء العلوي من المثلثي للمصفوفة A
!>                                  في AP.
!>
!>              UPLO = 'L' أو 'l'   يتم تقديم الجزء السفلي من المثلثي للمصفوفة A
!>                                  في AP.
!>
!>           لا يتغير عند الخروج.
!>

N

!>          N هو INTEGER
!>           عند الدخول، تحدد N ترتيب المصفوفة A.
!>           يجب أن تكون N صفرًا على الأقل.
!>           لا يتغير عند الخروج.
!>

ALPHA

!>          ALPHA هو COMPLEX*16
!>           عند الدخول، تحدد ALPHA العدد alpha.
!>           لا يتغير عند الخروج.
!>

AP

!>          AP هي مصفوفة COMPLEX*16، أبعادها على الأقل
!>           ( ( N*( N + 1 ) )/2 ).
!>           قبل الدخول، مع UPLO = 'U' أو 'u'، يجب أن تحتوي المصفوفة AP
!>           على الجزء العلوي من المثلثي للمصفوفة المتناظرة
!>           معبأة بالتتابع، عمودًا بعمود، بحيث تحتوي AP( 1 )
!>           على a( 1, 1 )، وتحتوي AP( 2 ) و AP( 3 ) على a( 1, 2 )
!>           و a( 2, 2 ) على التوالي، وهكذا.
!>           قبل الدخول، مع UPLO = 'L' أو 'l'، يجب أن تحتوي المصفوفة AP
!>           على الجزء السفلي من المثلثي للمصفوفة المتناظرة
!>           معبأة بالتتابع، عمودًا بعمود، بحيث تحتوي AP( 1 )
!>           على a( 1, 1 )، وتحتوي AP( 2 ) و AP( 3 ) على a( 2, 1 )
!>           و a( 3, 1 ) على التوالي، وهكذا.
!>           لا تتغير عند الخروج.
!>

X

!>          X هي مصفوفة COMPLEX*16، أبعادها على الأقل
!>           ( 1 + ( N - 1 )*abs( INCX ) ).
!>           قبل الدخول، يجب أن تحتوي المصفوفة المضاعفة X على المتجه x
!>           المكون من N عنصر.
!>           لا تتغير عند الخروج.
!>

INCX

!>          INCX هو INTEGER
!>           عند الدخول، تحدد INCX الزيادة لعناصر X. يجب أن تكون INCX غير صفرية.
!>           لا تتغير عند الخروج.
!>

BETA

!>          BETA هو COMPLEX*16
!>           عند الدخول، تحدد BETA العدد beta. عندما يتم توفير BETA على أنها صفر،
!>           فلن تحتاج Y إلى تعيين عند الدخول.
!>           لا تتغير عند الخروج.
!>

Y

!>          Y هي مصفوفة COMPLEX*16، أبعادها على الأقل
!>           ( 1 + ( N - 1 )*abs( INCY ) ).
!>           قبل الدخول، يجب أن تحتوي المصفوفة المضاعفة Y على المتجه y
!>           المكون من N عنصر. عند الخروج، سيتم استبدال Y بالمتجه y المحدث.
!>

INCY

!>          INCY هو INTEGER
!>           عند الدخول، تحدد INCY الزيادة لعناصر Y. يجب أن تكون INCY غير صفرية.
!>           لا تتغير عند الخروج.
!>
