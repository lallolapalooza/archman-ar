subroutine zsysvxx (fact، uplo، n، nrhs، complex*16، dimension( lda, * ) a, integer lda, complex*16، dimension( ldaf, * ) af، integer ldaf، integer، dimension( * ) ipiv، character equed، double precision، dimension( * ) s، complex*16، dimension( ldb, * ) b, integer ldb، complex*16، dimension( ldx, * ) x, integer ldx, double precision rcond, double precision rpvgrw, double precision, dimension( * ) berr, integer n_err_bnds, double precision, dimension( nrhs, * ) err_bnds_norm, double precision, dimension( nrhs, * ) err_bnds_comp, integer nparams, double precision, dimension( * ) params, complex*16, dimension( * ) work, double precision, dimension( * ) rwork, integer info)

ZSYSVXX تحسب حل نظام المعادلات الخطية A * X = B لمصفوفات SY

الغرض:

!>
!>    تستخدم ZSYSVXX تحليل العوامل المحورية القطرية لحساب حل لنظام
!>    المعادلات الخطية المعقد*16 A * X = B، حيث A مصفوفة متماثلة N×N و X و B
!>    هما مصفوفتان N×NRHS.
!>
!>    إذا طُلب ذلك، يتم إرجاع حدود خطأ قياسية وقصوى في المكونات.
!>    ستُرجع ZSYSVXX حلاً بخطأ ضئيل مضمون (O(eps) حيث eps هو الدقة
!>    الآلية العاملة) ما لم تكن المصفوفة غير موصوفة بشكل سيء للغاية،
!>    وفي هذه الحالة يتم إرجاع تحذير. يتم أيضًا حساب وإرجاع أرقام
!>    شروط ذات صلة.
!>
!>    تقبل ZSYSVXX عوامل تحليل وتوازنات مقدمة من المستخدم؛ انظر
!>    تعريفات خيارات FACT و EQUED. حل النظام مع التنقيح واستخدام
!>    عوامل تحليل من استدعاء ZSYSVXX سابق سينتج أيضًا حلاً بخطأ
!>    "O(eps)" أو تحذيرات، ولكن لا يمكننا تقديم هذا الادعاء للعوامل
!>    التحليلية والتوازنات العامة المقدمة من المستخدم إذا كانت تختلف
!>    عما تنتجه ZSYSVXX بنفسها.
!>

الوصف:

!>
!>    تُنفذ الخطوات التالية:
!>
!>    1. إذا كانت FACT = 'E'، يتم حساب عوامل قياس دقة مزدوجة لموازنة
!>    النظام:
!>
!>      diag(S)*A*diag(S)     *inv(diag(S))*X = diag(S)*B
!>
!>    يعتمد ما إذا كان سيتم موازنة النظام أم لا على قياس المصفوفة A،
!>    ولكن إذا تم استخدام الموازنة، فسيتم الكتابة فوق A بواسطة
!>    diag(S)*A*diag(S) و B بواسطة diag(S)*B.
!>
!>    2. إذا كانت FACT = 'N' أو 'E'، يتم استخدام تحليل LU لتحليل
!>    المصفوفة A (بعد الموازنة إذا كانت FACT = 'E') بالشكل:
!>
!>       A = U * D * U**T,  إذا كانت UPLO = 'U'، أو
!>       A = L * D * L**T,  إذا كانت UPLO = 'L'،
!>
!>    حيث U (أو L) هو حاصل ضرب مصفوفات تبديل ومصفوفات مثلثية عليا (دنيا)
!>    وحدوية، و D مصفوفة متماثلة وقطرية كتلية ذات كتل قطرية
!>    بحجم 1×1 و 2×2.
!>
!>    3. إذا كان بعض D(i,i)=0، بحيث تكون D مفردة تمامًا، فإن الروتين
!>    يعود بـ INFO = i. بخلاف ذلك، يتم استخدام الشكل المحلل لـ A لتقدير
!>    رقم شرط المصفوفة A (انظر الوسيط RCOND). إذا كان مقلوب رقم
!>    الشرط أقل من دقة الآلة، يستمر الروتين في الحل لـ X وحساب
!>    حدود الخطأ كما هو موضح أدناه.
!>
!>    4. يتم حل نظام المعادلات لـ X باستخدام الشكل المحلل لـ A.
!>
!>    5. افتراضيًا (ما لم يتم تعيين PARAMS(LA_LINRX_ITREF_I) إلى صفر)،
!>    سيستخدم الروتين التنقيح التكراري لمحاولة الحصول على خطأ صغير
!>    وحدود خطأ. يقوم التنقيح بحساب الباقي بدقة مضاعفة على الأقل
!>    للدقة العاملة.
!>
!>    6. إذا تم استخدام الموازنة، يتم ضرب المصفوفة X مسبقًا بـ
!>    diag(R) بحيث تحل النظام الأصلي قبل الموازنة.
!> 
!>     بعض المعلمات الاختيارية مجمعة في مصفوفة PARAMS. تحدد هذه
!>     الإعدادات كيفية إجراء التنقيح، ولكن في كثير من الأحيان تكون
!>     القيم الافتراضية مقبولة. إذا كانت القيم الافتراضية مقبولة،
!>     يمكن للمستخدمين تمرير NPARAMS = 0 مما يمنع الكود المصدري من
!>     الوصول إلى الوسيط PARAMS.
!>

المعلمات

FACT 
!>          FACT is CHARACTER*1
!>     يحدد ما إذا كان الشكل المحلل للمصفوفة A مقدمًا عند الإدخال،
!>     وإذا لم يكن كذلك، فما إذا كان يجب موازنة المصفوفة A قبل
!>     تحليلها.
!>       = 'F':  عند الإدخال، تحتوي AF و IPIV على الشكل المحلل لـ A.
!>               إذا كان EQUED لا يساوي 'N'، فقد تم موازنة المصفوفة A
!>               بعوامل قياس معطاة بواسطة S.
!>               A، AF، و IPIV لا يتم تعديلها.
!>       = 'N':  سيتم نسخ المصفوفة A إلى AF وتحليلها.
!>       = 'E':  سيتم موازنة المصفوفة A إذا لزم الأمر، ثم نسخها
!>               إلى AF وتحليلها.
!>

UPLO

!>          UPLO is CHARACTER*1
!>       = 'U':  يتم تخزين المثلث العلوي لـ A؛
!>       = 'L':  يتم تخزين المثلث السفلي لـ A.
!>

N

!>          N is INTEGER
!>     عدد المعادلات الخطية، أي رتبة المصفوفة A. N >= 0.
!>

NRHS

!>          NRHS is INTEGER
!>     عدد الأطراف اليمنى، أي عدد أعمدة المصفوفتين B و X. NRHS >= 0.
!>

A

!>          A is COMPLEX*16 array, dimension (LDA,N)
!>     المصفوفة المتماثلة A. إذا كانت UPLO = 'U'، فإن الجزء المثلثي
!>     العلوي الرائد N×N من A يحتوي على الجزء المثلثي العلوي للمصفوفة
!>     A، ولا تتم الإشارة إلى الجزء المثلثي السفلي الصارم من A.
!>     إذا كانت UPLO = 'L'، فإن الجزء المثلثي السفلي الرائد N×N من A
!>     يحتوي على الجزء المثلثي السفلي للمصفوفة A، ولا تتم الإشارة إلى
!>     الجزء المثلثي العلوي الصارم من A.
!>
!>     عند الخروج، إذا كانت FACT = 'E' و EQUED = 'Y'، يتم الكتابة فوق A
!>     بواسطة diag(S)*A*diag(S).
!>

LDA

!>          LDA is INTEGER
!>     البعد الرائد للمصفوفة A. LDA >= max(1,N).
!>

AF

!>          AF is COMPLEX*16 array, dimension (LDAF,N)
!>     إذا كانت FACT = 'F'، فإن AF هي وسيط إدخال وعند الإدخال
!>     تحتوي على المصفوفة القطرية الكتلية D والمضاعفات المستخدمة
!>     للحصول على العامل U أو L من تحليل A = U*D*U**T أو A = L*D*L**T
!>     كما تم حسابه بواسطة ZSYTRF.
!>
!>     إذا كانت FACT = 'N'، فإن AF هي وسيط إخراج وعند الخروج
!>     تعيد المصفوفة القطرية الكتلية D والمضاعفات المستخدمة للحصول
!>     على العامل U أو L من تحليل A = U*D*U**T أو A = L*D*L**T.
!>

LDAF

!>          LDAF is INTEGER
!>     البعد الرائد للمصفوفة AF. LDAF >= max(1,N).
!>

IPIV

!>          IPIV is INTEGER array, dimension (N)
!>     إذا كانت FACT = 'F'، فإن IPIV هي وسيط إدخال وعند الإدخال
!>     تحتوي على تفاصيل التبادلات و الهيكل الكتلي

EQUED

!>          EQUED هو CHARACTER*1
!>     يحدد شكل المعادلة التي تم إجراؤها.
!>       = 'N':  لا يوجد معادلة (دائماً صحيح إذا كان FACT = 'N').
!>       = 'Y':  تمت معادلة كل من الصفوف والأعمدة، أي تم استبدال A بـ
!>               diag(S) * A * diag(S).
!>     EQUED هو وسيط إدخال إذا كان FACT = 'F'. بخلاف ذلك، فهو وسيط
!>     إخراج.
!>

S


!>          S هو مصفوفة DOUBLE PRECISION، البعد (N)
!>     عوامل القياس لـ A. إذا كان EQUED = 'Y'، يتم ضرب A من
!>     اليسار واليمين بواسطة diag(S). S هو وسيط إدخال إذا كان FACT =
!>     'F'. بخلاف ذلك، S هو وسيط إخراج. إذا كان FACT = 'F' و EQUED
!>     = 'Y'، يجب أن يكون كل عنصر من S موجبًا. إذا كان S هو الإخراج، فإن كل
!>     عنصر من S هو قوة للجذر. إذا كان S هو الإدخال، يجب أن يكون كل
!>     عنصر من S قوة للجذر لضمان حل موثوق و
!>     تقديرات الخطأ. القياس باستخدام قوى الجذر لا يسبب
!>     أخطاء تقريب إلا إذا كانت النتيجة تحت التدفق أو تجاوزت التدفق.
!>     تؤدي أخطاء التقريب أثناء القياس إلى تحسين بمصفوفة
!>     غير مكافئة للمصفوفة المدخلة، مما ينتج عنه تقديرات خطأ
!>     قد لا تكون موثوقة.
!>

B


!>          B هي مصفوفة COMPLEX*16، البعد (LDB,NRHS)
!>     عند الإدخال، مصفوفة الطرف الأيمن B بحجم N-by-NRHS.
!>     عند الخروج،
!>     إذا كان EQUED = 'N'، لم يتم تعديل B؛
!>     إذا كان EQUED = 'Y'، تم استبدال B بـ diag(S)*B؛
!>

LDB


!>          LDB هو INTEGER
!>     البعد الرئيسي للمصفوفة B. LDB >= max(1,N).
!>

X


!>          X هي مصفوفة COMPLEX*16، البعد (LDX,NRHS)
!>     إذا كان INFO = 0، فإن مصفوفة الحل X بحجم N-by-NRHS لنظام
!>     المعادلات الأصلي. لاحظ أن A و B يتم تعديلهما عند الخروج إذا
!>     كان EQUED .ne. 'N'، وحل نظام المعادلة المحسّن هو
!>     inv(diag(S))*X.
!>

LDX


!>          LDX هو INTEGER
!>     البعد الرئيسي للمصفوفة X. LDX >= max(1,N).
!>

RCOND


!>          RCOND هو DOUBLE PRECISION
!>     مقلوب رقم الشرط المقاس. هذا تقدير لـ
!>     مقلوب رقم شرط Skeel للمصفوفة A بعد
!>     المعادلة (إذا تم إجراؤها). إذا كان هذا أقل من دقة الآلة
!>     (على وجه الخصوص، إذا كان صفرًا)، فإن المصفوفة منفردة
!>     بالنسبة لدقة العمل. لاحظ أن الخطأ قد لا يزال صغيرًا حتى
!>     لو كان هذا الرقم صغيرًا جدًا وبدت المصفوفة مشروطة بشكل سيء.
!>

RPVGRW


!>          RPVGRW هو DOUBLE PRECISION
!>     مقلوب نمو المحور. عند الخروج، يحتوي هذا على مقلوب
!>     عامل نمو المحور norm(A)/norm(U). يتم استخدام المعيار
!>     . إذا كان هذا أقل بكثير من 1، فإن استقرار
!>     تحليل LU للمصفوفة (المعادلة) A قد يكون سيئًا.
!>     هذا يعني أيضًا أن الحل X، والأرقام الشرطية المقدرة،
!>     وحدود الخطأ قد لا تكون موثوقة. إذا فشل التحليل مع
!>     0 < INFO <= N، فإن هذا يحتوي على مقلوب عامل نمو المحور
!>     للأعمدة INFO الرئيسية لـ A.
!>

BERR


!>          BERR هي مصفوفة DOUBLE PRECISION، البعد (NRHS)
!>     خطأ رجعي نسبي مكون بمكون. هذا هو
!>     خطأ رجعي نسبي مكون بمكون لكل متجه حل X(j)
!>     (أي، أصغر تغيير نسبي في أي عنصر من A أو B يجعل X(j) حلاً دقيقًا).
!>

N_ERR_BNDS


!>          N_ERR_BNDS هو INTEGER
!>     عدد حدود الخطأ المراد إرجاعها لكل جانب أيمن
!>     ولكل نوع (شامل للمعايير أو مكون بمكون). انظر ERR_BNDS_NORM و
!>     ERR_BNDS_COMP أدناه.
!>

ERR_BNDS_NORM


!>          ERR_BNDS_NORM هي مصفوفة DOUBLE PRECISION، البعد (NRHS, N_ERR_BNDS)
!>     لكل جانب أيمن، تحتوي هذه المصفوفة على معلومات حول
!>     حدود الخطأ المختلفة وأرقام الشرط المقابلة لـ
!>     الخطأ النسبي الشامل للمعايير، والذي يتم تعريفه على النحو التالي:
!>
!>     الخطأ النسبي الشامل للمعايير في متجه الحل i:
!>             max_j (abs(XTRUE(j,i) - X(j,i)))
!>            ------------------------------
!>                  max_j abs(X(j,i))
!>
!>     يتم فهرسة المصفوفة حسب نوع معلومات الخطأ كما هو موضح
!>     أدناه. يوجد حاليًا ما يصل إلى ثلاث قطع من المعلومات
!>     يتم إرجاعها.
!>
!>     الفهرس الأول في ERR_BNDS_NORM(i,:) يقابل الجانب الأيمن i.
!>
!>     الفهرس الثاني في ERR_BNDS_NORM(:,err) يحتوي على المجالات الثلاثة التالية:
!>     err = 1  قيمة منطقية. الثقة في الإجابة إذا كان
!>              مقلوب رقم الشرط أقل من الحد
!>              sqrt(n) * dlamch('Epsilon').
!>
!>     err = 2  حد الخطأ: الخطأ الأمامي المقدر،
!>              من المؤكد تقريبًا أنه ضمن عامل 10 من الخطأ الحقيقي
!>              طالما أن المدخل التالي أكبر من الحد
!>              sqrt(n) * dlamch('Epsilon'). يجب الثقة في حد الخطأ هذا فقط
!>              إذا كانت القيمة المنطقية السابقة صحيحة.
!>
!>     err = 3  مقلوب رقم الشرط: مقلوب الشرط الشامل المقدر
!>              . تتم مقارنته بالحد
!>              sqrt(n) * dlamch('Epsilon') لتحديد ما إذا كان الخطأ
!>              التقدير . هذه الأرقام الشرطية المقلوبة
!>              هي 1 / (norm(Z^{-1},inf) * norm(Z,inf)) لبعض
!>              المصفوفات Z المناسبة القياس.
!>              لتكن Z = S*A، حيث تقوم S بقياس كل صف بواسطة قوة للجذر
!>              بحيث تكون جميع مجاميع الصفوف المطلقة لـ Z تقريبًا 1.
!>
!>     انظر Lapack Working Note 165 لمزيد من التفاصيل والاحتياطات الإضافية.
!>

ERR_BNDS_COMP


!>          ERR_BNDS_COMP هي مصفوفة DOUBLE PRECISION، البعد (NRHS, N_ERR_BNDS)
!>     لكل جانب أيمن، تحتوي هذه المصفوفة على معلومات حول
!>     حدود الخطأ المختلفة وأرقام الشرط المقابلة لـ
!>     الخطأ النسبي المكون بمكون، والذي يتم تعريفه على النحو التالي:
!>
!>     الخطأ النسبي المكون بمكون في متجه الحل i:
!>                    abs(XTRUE(j,i) - X(j,i))
!>             max_j ----------------------
!>                         abs(X(j,i))
!>
!>     يتم فهرسة المصفوفة حسب الجانب الأيمن i (الذي يعتمد عليه
!>     الخطأ النسبي المكون بمكون)، ونوع معلومات الخطأ كما هو موضح
!>     أدناه. يوجد حاليًا ما يصل إلى ثلاثة قطع من المعلومات
!>     يتم إرجاعها لكل جانب أيمن. إذا
!>     لم يتم طلب دقة مكون بمكون (PARAMS(3) = 0.0)، فإن
!>     ERR_BNDS_COMP لا يتم الوصول إليه. إذا كان N_ERR_BNDS < 3، فسيتم إرجاع
!>     أول (:,N_ERR_BNDS) إدخالات كحد أقصى.
!>
!>     الفهرس الأول في ERR_BNDS_COMP(i,:) يقابل الجانب الأيمن i.
!>
!>     الفهرس الثاني في ERR_BNDS_COMP(:,err) يحتوي على المجالات الثلاثة التالية:
!>     err = 1  قيمة منطقية. الثقة في الإجابة إذا كان
!>              مقلوب رقم الشرط أقل من الحد
!>              sqrt(n) * dlamch('Epsilon').
!>
!>     err = 2  حد الخطأ: الخطأ الأمامي المقدر،
!>              من المؤكد تقريبًا أنه ضمن عامل 10 من الخطأ الحقيقي
!>              طالما أن المدخل التالي أكبر من الحد
!>              sqrt(n) * dlamch('Epsilon'). يجب الثقة في حد الخطأ هذا فقط
