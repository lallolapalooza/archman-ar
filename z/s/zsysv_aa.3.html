subroutine zsysv_aa (character uplo, integer n, integer nrhs, complex*16, dimension( lda, * ) a, integer lda, integer, dimension( * ) ipiv, complex*16, dimension( ldb, * ) b, integer ldb, complex*16, dimension( * ) work, integer lwork, integer info)

ZSYSV_AA تحسب حل نظام المعادلات الخطية A * X = B لمصفوفات SY

الغرض:

!>
!> ZSYSV يحسب حل نظام المعادلات الخطية المعقدة
!>    A * X = B,
!> حيث A مصفوفة متناظرة N × N و X و B مصفوفتان N × NRHS.
!>
!> تستخدم خوارزمية Aasen لتحليل A كـ
!>    A = U**T * T * U,  إذا كان UPLO = 'U', أو
!>    A = L * T * L**T,  إذا كان UPLO = 'L',
!> حيث U (أو L) هو ناتج ضرب مصفوفات التبديل والمصفوفات المثلثية العلوية (السفلية) الأحادية،
!> و T مصفوفة متناظرة ثلاثية الأقطار. ثم يستخدم الشكل المحلل لـ A لحل نظام المعادلات A * X = B.
!>

المعلمات

UPLO 
!>          UPLO هو CHARACTER*1
!>          = 'U':  يتم تخزين المثلث العلوي لـ A؛
!>          = 'L':  يتم تخزين المثلث السفلي لـ A؛
!>

N

!>          N هو INTEGER
!>          عدد المعادلات الخطية، أي رتبة المصفوفة A. N >= 0.
!>

NRHS

!>          NRHS هو INTEGER
!>          عدد الأطراف اليمنى، أي عدد أعمدة المصفوفة B. NRHS >= 0.
!>

A

!>          A هي مصفوفة COMPLEX*16، ذات بعد (LDA,N)
!>          عند الدخول، المصفوفة المتناظرة A. إذا كان UPLO = 'U'، فإن الجزء العلوي
!>          المثلثي من A الذي يصل إلى N × N يحتوي على الجزء العلوي
!>          المثلثي للمصفوفة A، ولا يتم الإشارة إلى الجزء السفلي
!>          المثلثي بشكل صارم من A. إذا كان UPLO = 'L'، فإن الجزء السفلي
!>          المثلثي من A الذي يصل إلى N × N يحتوي على الجزء السفلي
!>          المثلثي للمصفوفة A، ولا يتم الإشارة إلى الجزء العلوي
!>          المثلثي بشكل صارم من A.
!>
!>          عند الخروج، إذا كان INFO = 0، فإن المصفوفة ثلاثية الأقطار T والمضاعفات
!>          المستخدمة للحصول على العامل U أو L من تحليل
!>          A = U**T*T*U أو A = L*T*L**T كما تم حسابها بواسطة
!>          ZSYTRF.
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرائد للمصفوفة A. LDA >= max(1,N).
!>

IPIV

!>          IPIV هي مصفوفة INTEGER، ذات بعد (N)
!>          عند الخروج، تحتوي على تفاصيل التبادلات، أي
!>          تم تبديل الصف والعمود k من A مع
!>          الصف والعمود IPIV(k).
!>

B

!>          B هي مصفوفة COMPLEX*16، ذات بعد (LDB,NRHS)
!>          عند الدخول، مصفوفة الطرف الأيمن B التي تصل إلى N × NRHS.
!>          عند الخروج، إذا كان INFO = 0، فإن مصفوفة الحل X التي تصل إلى N × NRHS.
!>

LDB

!>          LDB هو INTEGER
!>          البعد الرائد للمصفوفة B. LDB >= max(1,N).
!>

WORK

!>          WORK هي مصفوفة COMPLEX*16، ذات بعد (MAX(1,LWORK))
!>          عند الخروج، إذا كان INFO = 0، فإن WORK(1) يعيد LWORK الأمثل.
!>

LWORK

!>          LWORK هو INTEGER
!>          طول WORK. LWORK >= MAX(1,2*N,3*N-2)، ولأفضل
!>          أداء، LWORK >= MAX(1,N*NB)، حيث NB هو
!>          حجم الكتلة الأمثل لـ ZSYTRF_AA.
!>
!>          إذا كان LWORK = -1، فسيتم افتراض استعلام مساحة العمل؛ الروتين
!>          يحسب فقط الحجم الأمثل للمصفوفة WORK، ويعيد
!>          هذه القيمة كالإدخال الأول للمصفوفة WORK، ولا يتم إصدار أي رسالة خطأ
!>          متعلقة بـ LWORK بواسطة XERBLA.
!>

INFO

!>          INFO هو INTEGER
!>          = 0: انتهاء ناجح
!>          < 0: إذا كان INFO = -i، فإن الوسيط i كان له قيمة غير قانونية
!>          > 0: إذا كان INFO = i، فإن D(i,i) يساوي صفرًا تمامًا. تم إكمال التحليل،
!>               ولكن المصفوفة القطرية الفرعية D فريدة تمامًا، لذلك لم يكن بالإمكان حساب الحل.
!>
