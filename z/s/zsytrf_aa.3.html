subroutine zsytrf_aa (character uplo, integer n, complex*16, dimension( lda, * ) a, integer lda, integer, dimension( * ) ipiv, complex*16, dimension( * ) work, integer lwork, integer info)

ZSYTRF_AA

الغرض:

!>
!> ZSYTRF_AA تحسب تحليل مصفوفة متماثلة معقدة A
!> باستخدام خوارزمية Aasen. شكل التحليل هو
!>
!>    A = U**T*T*U  أو  A = L*T*L**T
!>
!> حيث U (أو L) هو حاصل ضرب تباديل ومصفوفات عليا (سفلى) مثلثة وحدة.
!> و T هي مصفوفة ثلاثية قطرية متماثلة معقدة.
!>
!> هذا هو الإصدار المجمع للخوارزمية، يستدعي BLAS المستوى 3.
!>

المعلمات

UPLO 
!>          UPLO هو CHARACTER*1
!>          = 'U':  يتم تخزين المثلث العلوي للمصفوفة A؛
!>          = 'L':  يتم تخزين المثلث السفلي للمصفوفة A.
!>

N

!>          N هو INTEGER
!>          رتبة المصفوفة A. N >= 0.
!>

A

!>          A هي مصفوفة COMPLEX*16، ذات أبعاد (LDA,N)
!>          عند الدخول، المصفوفة المتماثلة A. إذا كانت UPLO = 'U'، فإن
!>          الجزء المثلث العلوي N-by-N من A يحتوي على الجزء المثلث العلوي
!>          للمصفوفة A، والجزء المثلث السفلي الصارم من A لا يتم الإشارة إليه. إذا كانت UPLO = 'L'، فإن
!>          الجزء المثلث السفلي N-by-N من A يحتوي على الجزء المثلث السفلي
!>          للمصفوفة A، والجزء المثلث العلوي الصارم من A لا يتم الإشارة إليه.
!>
!>          عند الخروج، يتم تخزين المصفوفة الثلاثية القطرية في الأقطار
!>          وشبه الأقطار أسفل (أو أعلى) الأقطار،
!>          وتخزن L أسفل (أو أعلى) شبه الأقطار، عندما تكون UPLO
!>          هي 'L' (أو 'U').
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرائد للمصفوفة A. LDA >= max(1,N).
!>

IPIV

!>          IPIV هي مصفوفة INTEGER، ذات أبعاد (N)
!>          عند الخروج، تحتوي على تفاصيل التبادلات، أي،
!>          تم تبديل الصف والعمود k من A مع
!>          الصف والعمود IPIV(k).
!>

WORK

!>          WORK هي مصفوفة COMPLEX*16، ذات أبعاد (MAX(1,LWORK))
!>          عند الخروج، إذا كان INFO = 0، فإن WORK(1) يعيد LWORK الأمثل.
!>

LWORK

!>          LWORK هو INTEGER
!>          طول WORK. LWORK >=MAX(1,2*N). للحصول على أفضل أداء
!>          LWORK >= N*(1+NB)، حيث NB هو حجم الكتلة الأمثل.
!>
!>          إذا كان LWORK = -1، فمن المفترض أن يكون هناك استعلام للمساحة؛ الروتين
!>          يحسب فقط الحجم الأمثل لمصفوفة WORK، ويعيد
!>          هذه القيمة كالإدخال الأول لمصفوفة WORK، ولا يصدر أي رسالة خطأ
!>          متعلقة بـ LWORK بواسطة XERBLA.
!>

INFO

!>          INFO هو INTEGER
!>          = 0:  خروج ناجح
!>          < 0:  إذا كان INFO = -i، فإن المعلمة i كانت قيمة غير قانونية.
!>
