subroutine zsytri2x (character uplo, integer n, complex*16, dimension( lda, * ) a, integer lda, integer, dimension( * ) ipiv, complex*16, dimension( n+nb+1,* ) work, integer nb, integer info)

ZSYTRI2X

الغرض:

!>
!> ZSYTRI2X تحسب معكوس مصفوفة معقدة متماثلة غير محددة
!> A باستخدام التحليل A = U*D*U**T أو A = L*D*L**T المحسوبة بواسطة
!> ZSYTRF.
!>

المعلمات

UPLO 
!>          UPLO هو CHARACTER*1
!>          يحدد ما إذا كانت تفاصيل التحليل مخزنة
!>          كمصفوفة مثلثية علوية أو سفلية.
!>          = 'U':  مثلثية علوية، الشكل هو A = U*D*U**T;
!>          = 'L':  مثلثية سفلية، الشكل هو A = L*D*L**T.
!>

N

!>          N هو INTEGER
!>          رتبة المصفوفة A. N >= 0.
!>

A

!>          A هي مصفوفة COMPLEX*16، البعد (LDA,N)
!>          عند الإدخال، مصفوفة D القطرية NNB والمضاعفات
!>          المستخدمة للحصول على العامل U أو L كما تم حسابه بواسطة ZSYTRF.
!>
!>          عند الخروج، إذا كان INFO = 0، فإن معكوس المصفوفة الأصلية
!>          (المتماثلة). إذا كان UPLO = 'U'، يتم تشكيل الجزء المثلثي العلوي
!>          للمعكوس ولا يتم الرجوع إلى الجزء من A أسفل القطر؛ إذا كان UPLO = 'L'
!>          يتم تشكيل الجزء المثلثي السفلي للمعكوس ولا يتم الرجوع إلى الجزء
!>          من A فوق القطر.
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرئيسي للمصفوفة A. LDA >= max(1,N).
!>

IPIV

!>          IPIV هي مصفوفة INTEGER، البعد (N)
!>          تفاصيل التبادلات وبنية D الأساسية NNB
!>          كما تم تحديده بواسطة ZSYTRF.
!>

WORK

!>          WORK هي مصفوفة COMPLEX*16، البعد (N+NB+1,NB+3)
!>

NB

!>          NB هو INTEGER
!>          حجم الكتلة
!>

INFO

!>          INFO هو INTEGER
!>          = 0: خروج ناجح
!>          < 0: إذا كان INFO = -i، فقد كان الوسيط i له قيمة غير قانونية
!>          > 0: إذا كان INFO = i، فإن D(i,i) = 0؛ المصفوفة منفردة ولا يمكن
!>               حساب معكوسها.
!>
