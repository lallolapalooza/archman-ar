وثائق الوظيفة/الروتين الفرعي

subroutine zsytf2_rook (character uplo, integer n, complex*16, dimension( lda, * ) a, integer lda, integer, dimension( * ) ipiv, integer info)

ZSYTF2_ROOK تقوم بحساب تحليل مصفوفة مركبة متناظرة غير محددة باستخدام طريقة ارتكاز القطر المحصور لبنش-كوفمان ("rook") (خوارزمية غير مجمعة).

الغرض:

!>
!> ZSYTF2_ROOK تقوم بحساب تحليل مصفوفة مركبة متناظرة A
!> باستخدام طريقة ارتكاز القطر المحصور لبنش-كوفمان ():
!>
!>    A = U*D*U**T  أو  A = L*D*L**T
!>
!> حيث U (أو L) هو حاصل ضرب مصفوفات التبديل والمصفوفات المثلثية العليا (السفلى)
!> الوحدوية، U**T هو منقول U، و D متناظرة ومقسمة قطرية مع كتل قطرية 1x1 و 2x2.
!>
!> هذه هي النسخة غير المجمعة من الخوارزمية، والتي تستدعي BLAS من المستوى 2.
!>

المعلمات

UPLO 
!>          UPLO هو CHARACTER*1
!>          يحدد ما إذا كان الجزء العلوي أو السفلي المثلثي للمصفوفة المتناظرة A
!>          مخزنًا:
!>          = 'U':  مثلثي علوي
!>          = 'L':  مثلثي سفلي
!>

N

!>          N هو INTEGER
!>          رتبة المصفوفة A.  N >= 0.
!>

A

!>          A هو مصفوفة COMPLEX*16، البعد (LDA,N)
!>          عند الإدخال، المصفوفة المتناظرة A.  إذا كان UPLO = 'U'، فإن الجزء
!>          المثلثي العلوي n x n من A يحتوي على الجزء المثلثي العلوي للمصفوفة A،
!>          ولا يتم الإشارة إلى الجزء المثلثي السفلي الصارم من A.  إذا كان UPLO = 'L'،
!>          فإن الجزء المثلثي السفلي n x n من A يحتوي على الجزء المثلثي السفلي
!>          للمصفوفة A، ولا يتم الإشارة إلى الجزء المثلثي العلوي الصارم من A.
!>
!>          عند الخروج، المصفوفة القطرية المقسمة D والمضاعفات المستخدمة
!>          للحصول على العامل U أو L (انظر أدناه لمزيد من التفاصيل).
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرئيسي للمصفوفة A.  LDA >= max(1,N).
!>

IPIV

!>          IPIV هو مصفوفة INTEGER، البعد (N)
!>          تفاصيل التبادلات وهيكل كتلة D.
!>
!>          إذا كان UPLO = 'U':
!>             إذا كان IPIV(k) > 0، فقد تم تبديل الصفوف والأعمدة k و IPIV(k)
!>             و D(k,k) هي كتلة قطرية 1x1.
!>
!>             إذا كان IPIV(k) < 0 و IPIV(k-1) < 0، فقد تم تبديل الصفوف
!>             والأعمدة k و -IPIV(k) وتم تبديل الصفوف
!>             والأعمدة k-1 و -IPIV(k-1)،
!>             D(k-1:k,k-1:k) هي كتلة قطرية 2x2.
!>
!>          إذا كان UPLO = 'L':
!>             إذا كان IPIV(k) > 0، فقد تم تبديل الصفوف والأعمدة k و IPIV(k)
!>             و D(k,k) هي كتلة قطرية 1x1.
!>
!>             إذا كان IPIV(k) < 0 و IPIV(k+1) < 0، فقد تم تبديل الصفوف
!>             والأعمدة k و -IPIV(k) وتم تبديل الصفوف
!>             والأعمدة k+1 و -IPIV(k+1)،
!>             D(k:k+1,k:k+1) هي كتلة قطرية 2x2.
!>

INFO

!>          INFO هو INTEGER
!>          = 0: خروج ناجح
!>          < 0: إذا كان INFO = -k، فإن الوسيط k كان له قيمة غير قانونية
!>          > 0: إذا كان INFO = k، فإن D(k,k) يساوي صفرًا تمامًا.  تم إكمال
!>               التحليل، لكن المصفوفة القطرية المقسمة D متفردة تمامًا، وسيحدث
!>               القسمة على صفر إذا تم استخدامها لحل نظام من المعادلات.
!>

تفاصيل إضافية:

!>
!>  إذا كان UPLO = 'U'، فإن A = U*D*U**T، حيث
!>     U = P(n)*U(n)* ... *P(k)U(k)* ...,
!>  أي، U هو حاصل ضرب الحدود P(k)*U(k)، حيث k ينقص من n إلى
!>  1 بخطوات 1 أو 2، و D هي مصفوفة قطرية مقسمة بكتل قطرية 1x1
!>  و 2x2 D(k).  P(k) هي مصفوفة تبديل كما هي معرفة بواسطة IPIV(k)، و U(k) هي مصفوفة
!>  مثلثية علوية وحدوية، بحيث إذا كانت الكتلة القطرية D(k) من الرتبة s (s = 1 أو 2)،
!>  فإن
!>
!>             (   I    v    0   )   k-s
!>     U(k) =  (   0    I    0   )   s
!>             (   0    0    I   )   n-k
!>                k-s   s   n-k
!>
!>  إذا كان s = 1، فإن D(k) يكتب فوق A(k,k)، و v يكتب فوق A(1:k-1,k).
!>  إذا كان s = 2، فإن المثلث العلوي لـ D(k) يكتب فوق A(k-1,k-1)، A(k-1,k)،
!>  و A(k,k)، و v يكتب فوق A(1:k-2,k-1:k).
!>
!>  إذا كان UPLO = 'L'، فإن A = L*D*L**T، حيث
!>     L = P(1)*L(1)* ... *P(k)*L(k)* ...,
!>  أي، L هو حاصل ضرب الحدود P(k)*L(k)، حيث k يزيد من 1 إلى
!>  n بخطوات 1 أو 2، و D هي مصفوفة قطرية مقسمة بكتل قطرية 1x1
!>  و 2x2 D(k).  P(k) هي مصفوفة تبديل كما هي معرفة بواسطة IPIV(k)، و L(k) هي مصفوفة
!>  مثلثية سفلية وحدوية، بحيث إذا كانت الكتلة القطرية D(k) من الرتبة s (s = 1 أو 2)،
!>  فإن
!>
!>             (   I    0     0   )  k-1
!>     L(k) =  (   0    I     0   )  s
!>             (   0    v     I   )  n-k-s+1
!>                k-1   s  n-k-s+1
!>
!>  إذا كان s = 1، فإن D(k) يكتب فوق A(k,k)، و v يكتب فوق A(k+1:n,k).
!>  إذا كان s = 2، فإن المثلث السفلي لـ D(k) يكتب فوق A(k,k)، A(k+1,k)،
!>  و A(k+1,k+1)، و v يكتب فوق A(k+2:n,k:k+1).
!>
