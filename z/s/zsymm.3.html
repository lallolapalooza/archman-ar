توثيق الوظيفة/الروتين الفرعي

subroutine zsymm (character side, character uplo, integer m, integer n, complex*16 alpha, complex*16, dimension(lda,*) a, integer lda, complex*16, dimension(ldb,*) b, integer ldb, complex*16 beta, complex*16, dimension(ldc,*) c, integer ldc)

ZSYMM

الغرض:

!>
!> ZSYMM  تقوم بإحدى عمليات المصفوفة-المصفوفة
!>
!>    C := alpha*A*B + beta*C,
!>
!> أو
!>
!>    C := alpha*B*A + beta*C,
!>
!> حيث  alpha  و beta  هما عددان قياسيان، A  مصفوفة متناظرة، و B و
!> C  هما مصفوفتان بحجم m × n.
!>

المعلمات

SIDE 
!>          SIDE is CHARACTER*1
!>           عند الإدخال،  SIDE  يحدد ما إذا كانت المصفوفة المتناظرة  A
!>           تظهر على  اليسار  أو اليمين  في العملية كما يلي:
!>
!>              SIDE = 'L' or 'l'   C := alpha*A*B + beta*C,
!>
!>              SIDE = 'R' or 'r'   C := alpha*B*A + beta*C,
!>

UPLO

!>          UPLO is CHARACTER*1
!>           عند الإدخال،  UPLO  يحدد ما إذا كان  الجزء العلوي  أو  السفلي
!>           المثلثي  من المصفوفة المتناظرة  A  سيتم الرجوع إليه كما يلي:
!>
!>              UPLO = 'U' or 'u'   يتم الرجوع فقط إلى الجزء العلوي
!>                                  المثلثي من المصفوفة المتناظرة.
!>
!>              UPLO = 'L' or 'l'   يتم الرجوع فقط إلى الجزء السفلي
!>                                  المثلثي من المصفوفة المتناظرة.
!>

M

!>          M is INTEGER
!>           عند الإدخال،  M  يحدد عدد صفوف المصفوفة  C.
!>           يجب أن يكون  M  على الأقل صفر.
!>

N

!>          N is INTEGER
!>           عند الإدخال، N  يحدد عدد أعمدة المصفوفة C.
!>           يجب أن يكون  N  على الأقل صفر.
!>

ALPHA

!>          ALPHA is COMPLEX*16
!>           عند الإدخال،  ALPHA  يحدد العدد القياسي alpha.
!>

A

!>          A is COMPLEX*16 array, dimension ( LDA, ka ), where ka is
!>           m  when  SIDE = 'L' or 'l'  and is n  otherwise.
!>           قبل الدخول مع  SIDE = 'L' or 'l',  يجب أن يحتوي الجزء
!>           m × m  من المصفوفة  A  على المصفوفة المتناظرة، بحيث
!>           عندما يكون  UPLO = 'U' or 'u', يجب أن يحتوي الجزء العلوي
!>           المثلثي  الرائد  m × m  من المصفوفة  A  على الجزء العلوي
!>           المثلثي من المصفوفة المتناظرة، ولا يتم الرجوع إلى الجزء
!>           المثلثي السفلي الصارم  من  A، وعندما يكون  UPLO = 'L' or 'l',
!>           يجب أن يحتوي الجزء السفلي
!>           المثلثي  الرائد  m × m  من المصفوفة  A  على الجزء السفلي
!>           المثلثي من المصفوفة المتناظرة، ولا يتم الرجوع إلى الجزء
!>           المثلثي العلوي الصارم  من  A.
!>           قبل الدخول مع  SIDE = 'R' or 'r',  يجب أن يحتوي الجزء
!>           n × n  من المصفوفة  A  على المصفوفة المتناظرة، بحيث
!>           عندما يكون  UPLO = 'U' or 'u', يجب أن يحتوي الجزء العلوي
!>           المثلثي  الرائد  n × n  من المصفوفة  A  على الجزء العلوي
!>           المثلثي من المصفوفة المتناظرة، ولا يتم الرجوع إلى الجزء
!>           المثلثي السفلي الصارم  من  A، وعندما يكون  UPLO = 'L' or 'l',
!>           يجب أن يحتوي الجزء السفلي
!>           المثلثي  الرائد  n × n  من المصفوفة  A  على الجزء السفلي
!>           المثلثي من المصفوفة المتناظرة، ولا يتم الرجوع إلى الجزء
!>           المثلثي العلوي الصارم  من  A.
!>

LDA

!>          LDA is INTEGER
!>           عند الإدخال،  LDA  يحدد البعد الأول لـ A كما تم تعريفه
!>           في  البرنامج (الفرعي) الذي يستدعيها. عندما يكون  SIDE = 'L' or 'l'
!>           يجب أن يكون  LDA  على الأقل  max( 1, m )، وإلا فإن  LDA
!>           يجب أن يكون على الأقل max( 1, n ).
!>

B

!>          B is COMPLEX*16 array, dimension ( LDB, N )
!>           قبل الدخول، يجب أن يحتوي الجزء الرائد m × n  من المصفوفة  B
!>           على المصفوفة B.
!>

LDB

!>          LDB is INTEGER
!>           عند الإدخال،  LDB  يحدد البعد الأول لـ B كما تم تعريفه
!>           في  البرنامج (الفرعي) الذي يستدعيها. يجب أن يكون  LDB
!>           على الأقل max( 1, m ).
!>

BETA

!>          BETA is COMPLEX*16
!>           عند الإدخال،  BETA  يحدد العدد القياسي  beta.  عندما يتم
!>           توفير  BETA  كصفر، فلا يلزم تعيين C عند الإدخال.
!>

C

!>          C is COMPLEX*16 array, dimension ( LDC, N )
!>           قبل الدخول، يجب أن يحتوي الجزء الرائد m × n  من المصفوفة  C
!>           على المصفوفة C، باستثناء عندما يكون beta صفرًا، في هذه الحالة
!>           لا يلزم تعيين C عند الإدخال.
!>           عند الخروج، يتم الكتابة فوق المصفوفة C بالمصفوفة المحدثة
!>           بحجم m × n.
!>

LDC

!>          LDC is INTEGER
!>           عند الإدخال،  LDC  يحدد البعد الأول لـ C كما تم تعريفه
!>           في  البرنامج (الفرعي) الذي يستدعيها. يجب أن يكون  LDC
!>           على الأقل max( 1, m ).
!>
