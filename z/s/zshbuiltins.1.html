zshbuiltins(1)

الحزمة: zsh

التراخيص: custom

من fno، سيتم استعادة وضع المحاكاة وجميع الخيارات إلى الحالة التي كانت عليها قبل الدخول إلى المحاكاة المؤقتة.

الوثائق المذكورة أعلاه كافية عادةً للغرض المقصود من تنفيذ التعليمات البرمجية المصممة لأصداف أخرى في بيئة مناسبة. تتضمن القواعد الأكثر تفصيلاً ما يلي:

1.
بيئة المحاكاة اللاصقة التي يوفرها `emulate shell -c' مطابقة لتلك التي يوفرها الدخول إلى دالة تم تمييزها للمحاكاة اللاصقة نتيجة لتعريفها في مثل هذه البيئة. لذلك، على سبيل المثال، يتم توريث المحاكاة اللاصقة للدوال الفرعية المعرفة داخل الدوال ذات المحاكاة اللاصقة.
2.
لا يحدث تغيير في الخيارات عند الدخول أو الخروج من الدوال التي لم يتم تمييزها للمحاكاة اللاصقة، بخلاف تلك التي ستحدث عادةً، حتى لو تم استدعاء تلك الدوال داخل محاكاة لاصقة.
3.
لا يتم توفير معالجة خاصة للدوال المميزة لـ autoload ولا للدوال الموجودة في wordcode الذي تم إنشاؤه بواسطة الأمر zcompile.
4.
يمثل وجود أو عدم وجود المفتاح -R إلى emulate أوضاع محاكاة لاصقة مختلفة، لذلك على سبيل المثال `emulate sh -c` و `emulate -R sh -c` و `emulate csh -c` تُعامل على أنها ثلاث محاكيات لاصقة متميزة.
5.
الاختلاف في خيارات الصدفة المقدمة بالإضافة إلى المحاكاة الأساسية يعني أيضًا أن المحاكيات اللاصقة مختلفة، لذلك على سبيل المثال `emulate zsh -c` و `emulate zsh -o cbases -c` تُعامل على أنها محاكيات لاصقة متميزة.
enable [ -afmprs ] name ...
تمكين عناصر جدول الهاش المسماة، والتي تم تعطيلها سابقًا على الأرجح باستخدام disable. الافتراضي هو تمكين الأوامر المضمنة. يتسبب الخيار -a في أن تعمل enable على الأسماء المستعارة العادية أو العالمية. يتسبب الخيار -s في أن تعمل enable على الأسماء المستعارة اللاحقة. يتسبب الخيار -f في أن تعمل enable على دوال الصدفة. يتسبب الخيار -r في أن تعمل enable على الكلمات المحجوزة. بدون وسيطات، تتم طباعة جميع عناصر جدول الهاش الممكّنة من جدول الهاش المقابل. مع العلامة -m، تُؤخذ الوسيطات كأنماط (يجب اقتباسها) ويتم تمكين جميع عناصر جدول الهاش من جدول الهاش المقابل التي تطابق هذه الأنماط. يمكن تعطيل الكائنات الممكّنة باستخدام الأمر المضمن disable.

enable -p يعيد تمكين الأنماط التي تم تعطيلها باستخدام disable -p. لاحظ أنه لا يتجاوز خيارات العولمة (globbing)؛ على سبيل المثال، `enable -p "~"` لا يتسبب في تنشيط حرف النمط ~ ما لم يتم تعيين خيار EXTENDED_GLOB أيضًا. لتمكين جميع الأنماط الممكنة (حتى يمكن تعطيلها بشكل فردي باستخدام disable -p)، استخدم `setopt EXTENDED_GLOB KSH_GLOB NO_SH_GLOB'.

eval [ arg ... ]
اقرأ الوسيطات كمدخلات للصدفة ونفذ الأوامر الناتجة في عملية الصدفة الحالية. يكون حالة الخروج هي نفسها كما لو تم تنفيذ الأوامر مباشرة بواسطة الصدفة؛ إذا لم تكن هناك args أو لم تحتوِ على أي أوامر (أي أنها سلسلة فارغة أو مسافة بيضاء) تكون حالة الخروج صفرًا.
exec [ -cl ] [ -a argv0 ] [ command [ arg ... ] ]
استبدل الصدفة الحالية بـ command بدلاً من إنشاء فرع. إذا كان command أمرًا مضمنًا في الصدفة أو دالة صدفة، فإن الصدفة تنفذه، وتخرج عند اكتمال الأمر.

مع -c، امسح البيئة؛ مع -l، قم بإلحاق - بسلسلة argv[0] للأمر الذي تم تنفيذه (لمحاكاة صدفة تسجيل الدخول)؛ مع -a argv0، قم بتعيين سلسلة argv[0] للأمر الذي تم تنفيذه. انظر القسم `Precommand Modifiers` في zshmisc(1).

إذا تم تعيين الخيار POSIX_BUILTINS، فلن يتم تفسير command أبدًا كأمر مضمن في الصدفة أو دالة صدفة. هذا يعني أن المعدلات المسبقة للأوامر مثل builtin و noglob لا يتم تفسيرها أيضًا داخل الصدفة. لذلك يتم دائمًا العثور على command عن طريق البحث في مسار الأوامر.

إذا تم حذف command ولكن تم تحديد أي إعادة توجيه، فستصبح إعادة التوجيه سارية المفعول في الصدفة الحالية.

exit [ n ]
اخرج من الصدفة بحالة الخروج المحددة بواسطة التعبير الحسابي n؛ إذا لم يتم تحديد أي شيء، استخدم حالة الخروج من آخر أمر تم تنفيذه. سيؤدي شرط نهاية الملف (EOF) أيضًا إلى خروج الصدفة، ما لم يتم تعيين الخيار IGNORE_EOF.

انظر الملاحظات في نهاية قسم JOBS في zshmisc(1) لبعض التفاعلات غير المتوقعة المحتملة للأمر exit مع الوظائف.

export [ name[=value] ... ]
يتم تمييز الأسماء name المحددة للتصدير التلقائي إلى بيئة الأوامر المنفذة لاحقًا. ما يعادل typeset -gx. إذا كان المعامل المحدد غير موجود بالفعل، يتم إنشاؤه في النطاق العام.
false [ arg ... ]
لا تفعل شيئًا وأرجع حالة خروج 1.
fc [ -e ename ] [ -s ] [ -LI ] [ -m match ] [ old=new ... ] [ first [ last ] ]
fc -l [ -LI ] [ -nrdfEiD ] [ -t timefmt ] [ -m match ]
[ old=new ... ] [ first [ last ] ]
fc -p [ -a ] [ filename [ histsize [ savehistsize ] ] ]
fc -P
fc -ARWI [ filename ]
يتحكم الأمر fc في آلية سجل الأوامر التفاعلية. لاحظ أنه يتم قراءة وكتابة خيارات سجل الأوامر فقط إذا كانت الصدفة تفاعلية. عادةً ما يتم اكتشاف ذلك تلقائيًا، ولكن يمكن فرضه عن طريق تعيين خيار interactive عند بدء تشغيل الصدفة.

الشكلان الأولان لهذا الأمر يحددان نطاقًا من الأحداث من first إلى last من قائمة سجل الأوامر. يمكن تحديد الوسيطات first و last كرقم أو كسلسلة. يتم استخدام رقم سالب كإزاحة لرقم حدث سجل الأوامر الحالي. تحدد السلسلة أحدث حدث يبدأ بالسلسلة المعطاة. ثم يتم إجراء جميع الاستبدالات old=new، إن وجدت، على نص الأحداث.

يمكن تضييق نطاق الأحداث المحدد بالأرقام بشكل أكبر بواسطة العلامات التالية.

-I
يقصر على الأحداث الداخلية فقط (ليست من $HISTFILE)
-L
يقصر على الأحداث المحلية فقط (ليست من أصداف أخرى، انظر SHARE_HISTORY في zshoptions(1) -- لاحظ أن $HISTFILE يعتبر محليًا عند قراءته عند بدء التشغيل)
-m
تأخذ الوسيطة الأولى كنمط (يجب اقتباسه) ويتم النظر فقط في أحداث سجل الأوامر التي تطابق هذا النمط

إذا لم يتم تحديد first، فسيتم تعيينها على -1 (أحدث حدث)، أو على -16 إذا تم إعطاء العلامة -l. إذا لم يتم تحديد last، فسيتم تعيينها على first، أو على -1 إذا تم إعطاء العلامة -l. ومع ذلك، إذا أضاف الحدث الحالي إدخالات إلى سجل الأوامر باستخدام `print -s` أو `fc -R`، فإن last الافتراضي لـ -l يشمل جميع إدخالات سجل الأوامر الجديدة منذ بدء الحدث الحالي.

عند إعطاء العلامة -l، يتم سرد الأحداث الناتجة على المخرج القياسي. بخلاف ذلك، يتم استدعاء برنامج المحرر المحدد بواسطة -e ename على ملف يحتوي على أحداث سجل الأوامر هذه. إذا لم يتم إعطاء -e، يتم استخدام قيمة المتغير FCEDIT؛ إذا لم يتم تعيينها، يتم استخدام قيمة المتغير EDITOR؛ إذا لم يتم تعيينها، يتم استخدام قيمة افتراضية مضمنة، وعادةً ما تكون `vi`. إذا كان ename هو `-`، فلا يتم استدعاء محرر.

عند اكتمال التحرير، يتم تنفيذ الأمر المحرر.

العلامة `-s` تعادل `-e -`. العلامة -r تعكس ترتيب الأحداث والعلامة -n تقمع أرقام الأحداث عند سردها.

أيضًا عند السرد،

-d
يطبع الطوابع الزمنية لكل حدث
-f
يطبع الطوابع الزمنية الكاملة للتاريخ والوقت بالولايات المتحدة بتنسيق `MM/DD/YY hh:mm`
-E
يطبع الطوابع الزمنية الكاملة للتاريخ والوقت بتنسيق أوروبي `dd.mm.yyyy hh:mm`
-i
يطبع الطوابع الزمنية الكاملة للتاريخ والوقت بتنسيق ISO8601 `yyyy-mm-dd hh:mm`
-t fmt
يطبع طوابع الوقت والتاريخ بالتنسيق المحدد؛ يتم تنسيق fmt باستخدام دالة strftime مع امتدادات zsh الموصوفة لتنسيق الموجه %D{string} في قسم EXPANSION OF PROMPT SEQUENCES في zshmisc(1). يجب ألا يزيد طول السلسلة المنسقة الناتجة عن 256 حرفًا أو لن يتم طباعتها
-D
يطبع الفترات الزمنية المنقضية؛ يمكن دمجه مع أحد الخيارات المذكورة أعلاه

`fc -p` يضع قائمة سجل الأوامر الحالية على مكدس وينتقل إلى قائمة سجل أوامر جديدة. إذا تم تحديد الخيار -a أيضًا، فسيتم إخراج قائمة سجل الأوامر هذه تلقائيًا عند الخروج من نطاق الدالة الحالي، وهو حل أفضل بكثير من إنشاء وظيفة اعتراض لاستدعاء `fc -P` يدويًا. إذا لم يتم تحديد أي وسيطات، تكون قائمة سجل الأوامر فارغة، ويتم إلغاء تعيين $HISTFILE، ويتم تعيين $HISTSIZE و $SAVEHIST إلى قيمها الافتراضية. إذا تم توفير وسيطة واحدة، يتم تعيين $HISTFILE إلى اسم الملف هذا، وتظل $HISTSIZE و $SAVEHIST دون تغيير، ويتم قراءة ملف سجل الأوامر (إذا كان موجودًا) لتهيئة القائمة الجديدة. إذا تم تحديد وسيطة ثانية، يتم بدلاً من ذلك تعيين $HISTSIZE و $SAVEHIST إلى القيمة الرقمية المحددة.

أخيرًا، إذا تم تحديد وسيطة ثالثة، فإن

$SAVEHIST هي قيمة منفصلة عن $HISTSIZE. لك مطلق الحرية في تغيير قيم هذه المتغيرات البيئية لقائمة السجل الجديدة كما تشاء لمعالجة قائمة السجل الجديدة.

`fc -P' تعيد قائمة السجل إلى قائمة أقدم تم حفظها بواسطة `fc -p'. يتم حفظ القائمة الحالية في $HISTFILE قبل تدميرها (بافتراض أن $HISTFILE و $SAVEHIST تم تعيينهما بشكل مناسب، بالطبع). يتم استعادة قيم $HISTFILE و $HISTSIZE و $SAVEHIST إلى القيم التي كانت عليها عندما تم استدعاء `fc -p'. لاحظ أن هذه الاستعادة يمكن أن تتعارض مع جعل هذه المتغيرات "محلية"، لذا فإن أفضل ما يمكنك فعله هو تجنب التصريحات المحلية لهذه المتغيرات في الدوال التي تستخدم `fc -p'. المزيج الوحيد الآخر الآمن بشكل مؤكد هو التصريح عن هذه المتغيرات لتكون محلية في بداية دالتك واستخدام خيار التشغيل التلقائي (-a) مع `fc -p'. أخيراً، لاحظ أنه من القانوني إلغاء تعيين سجل يدويًا تم وضع علامة عليه للإلغاء التلقائي إذا احتجت إلى القيام بذلك قبل انتهاء الدالة.

`fc -R' تقرأ السجل من الملف المحدد، `fc -W' تكتب السجل إلى الملف المحدد، و `fc -A' تلحق السجل بالملف المحدد. إذا لم يتم تحديد اسم ملف، فسيتم افتراض $HISTFILE. إذا تمت إضافة الخيار -I إلى -R، فسيتم إضافة الأحداث التي لا تحتوي بالفعل على قائمة السجل الداخلية فقط. إذا تمت إضافة الخيار -I إلى -A أو -W، فسيتم إلحاق/كتابة الأحداث الجديدة فقط منذ آخر إلحاق/كتابة تدريجية إلى ملف السجل. في أي حال، لن يحتوي الملف الذي تم إنشاؤه على أكثر من $SAVEHIST مدخل.

fg [ job ... ]
job ...
إحضار كل مهمة محددة job بالتناوب إلى المقدمة. إذا لم يتم تحديد job، فاستأنف المهمة الحالية.
float [ {+|-}Hghlprtux ] [ {+|-}EFLRZ [ n ] ] [ name[=value] ... ]
يعادل typeset -E، باستثناء أن الخيارات غير ذات الصلة بالأرقام العائمة غير مسموح بها.
functions [ {+|-}UkmtTuWz ] [ -x num ] [ name ... ]
functions -c oldfn newfn
functions -M [-s] mathfn [ min [ max [ shellfn ] ] ]
functions -M [ -m pattern ... ]
functions +M [ -m ] mathfn ...
يعادل typeset -f، باستثناء الخيارات -c و -x و -M و -W. بالنسبة لـ functions -u و functions -U، انظر autoload، التي توفر خيارات إضافية. بالنسبة لـ functions -t و functions -T، انظر typeset -f.

يشير الخيار -x إلى أن أي مخرجات للدوال ستحتوي على كل علامة جدولة بادئة للمسافة البادئة، والتي أضافها الـ shell لإظهار البنية التركيبية، وسيتم توسيعها إلى العدد المعطى num من المسافات. يمكن أن يكون num أيضًا 0 لإزالة كل المسافات البادئة.

الخيار -W يشغل الخيار WARN_NESTED_VAR للدالة المسماة أو الدوال فقط. يتم إيقاف تشغيل الخيار في بداية الدوال المتداخلة (باستثناء الدوال المجهولة) ما لم يكن لدى الدالة المسماة أيضًا سمة -W.

الخيار -c يتسبب في نسخ oldfn إلى newfn. يتم التعامل مع النسخ بكفاءة داخليًا عن طريق عد المراجع. إذا تم وضع علامة على oldfn للتحميل التلقائي، فسيتم تحميلها أولاً وإذا فشل ذلك، يفشل النسخ. يمكن إعادة تعريف أي من الدالتين لاحقًا دون التأثير على الأخرى. عادةً ما يكون oldfn هو اسم دالة shell مكتبية يتم إعادة تعريفها بعد ذلك لاستدعاء newfn، وبالتالي تثبيت نسخة معدلة من الدالة.

الأعلام -M و +M

لا يمكن دمج استخدام الخيار -M مع أي من الخيارات التي يتم التعامل معها بواسطة typeset -f.

functions -M mathfn تعرف mathfn كاسم لدالة رياضية معترف بها في جميع أشكال التعبيرات الحسابية؛ انظر القسم `تقييم الحساب' في zshmisc(1). افتراضيًا، يمكن أن تأخذ mathfn أي عدد من الوسائط المفصولة بفواصل. إذا تم إعطاء min، فيجب أن تحتوي على min وسيطة بالضبط؛ إذا تم إعطاء كل من min و max، فيجب أن تحتوي على min على الأقل و max كحد أقصى. يمكن أن يكون max -1 للإشارة إلى عدم وجود حد أعلى.

افتراضيًا، يتم تنفيذ الدالة بواسطة دالة shell بنفس الاسم؛ إذا تم تحديد shellfn، فإنه يعطي اسم دالة shell المقابلة بينما يظل mathfn هو الاسم المستخدم في التعبيرات الحسابية. اسم الدالة في $0 هو mathfn (وليس shellfn كما هو الحال عادة)، بشرط أن يكون الخيار FUNCTION_ARGZERO فعالاً. تتوافق المعلمات الموضعية في دالة shell مع وسائط استدعاء الدالة الرياضية.

نتيجة آخر تعبير حسابي تم تقييمه داخل دالة shell تعطي نتيجة الدالة الرياضية. هذا لا يقتصر على استبدالات الحساب من النوع $((...))، ولكنه يشمل أيضًا التعبيرات الحسابية التي تم تقييمها بأي طريقة أخرى، بما في ذلك بواسطة الأمر المدمج let، بواسطة عبارات ((...))، وحتى بواسطة الأمر المدمج return وعناوين الصفائف. لذلك، يجب توخي الحذر لعدم استخدام بناء جملي يقوم بتقييم حسابي بعد تقييم ما يجب أن يكون نتيجة الدالة. على سبيل المثال:

# خطأ
zmath_cube() {
  (( $1 * $1 * $1 ))
  return 0
}
functions -M cube 1 1 zmath_cube
print $(( cube(3) ))

سيقوم هذا بطباعة `0' بسبب return.

التعليق على return سيؤدي إلى مشكلة مختلفة: سيصبح بيان ((...)) البيان الأخير في الدالة، وبالتالي فإن *حالة الإرجاع* ($?) للدالة ستكون غير صفرية (تشير إلى فشل) كلما كانت *النتيجة الحسابية* للدالة تساوي صفرًا (عدديًا):

# خطأ
zmath_cube() {
  (( $1 * $1 * $1 ))
}
functions -M cube 1 1 zmath_cube
print $(( cube(0) ))

بدلاً من ذلك، يمكن استخدام الأمر المدمج true:

# صحيح
zmath_cube() {
  (( $1 * $1 * $1 ))
  true
}
functions -M cube 1 1 zmath_cube
print $(( cube(3) ))

إذا تم إعطاء الخيار الإضافي -s لـ functions -M، فإن الوسيط للدالة هو سلسلة واحدة: يتم تمرير أي شيء بين القوس المفتوح والقوس المغلق المتطابق إلى الدالة كوسيط واحد، حتى لو كان يتضمن فواصل أو مسافات بيضاء. لذلك، يجب أن تكون محددات الوسيط الأدنى والأقصى 1 إذا تم تقديمها. يتم تمرير قائمة وسائط فارغة كسلسلة صفرية الطول. وبالتالي، فإن دالة السلسلة التالية تأخذ وسيطًا واحدًا، بما في ذلك الفواصل، وتطبع 11:

stringfn() { (( $#1 )); true }
functions -Ms stringfn
print $(( stringfn(foo,bar,rod) ))

getopts optstring name [ arg ... ]
يتحقق من args للخيارات القانونية. إذا تم حذف args، استخدم المعلمات الموضعية. يبدأ الوسيط الخيار الصالح بحرف '+' أو '-'. الوسيط الذي لا يبدأ بحرف '+' أو '-'، أو الوسيط '--'، ينهي الخيارات. لاحظ أن '-' واحد لا يعتبر وسيط خيار صالحًا. optstring يحتوي على الأحرف التي يتعرف عليها getopts. إذا تبع حرف علامة ':'، فإن هذا الخيار يتطلب وسيطًا. يمكن فصل الخيارات عن الوسيط بمسافات.

في كل مرة يتم فيها استدعاؤها، يضع getopts حرف الخيار الذي يجده في معامل الـ shell name، مسبوقًا بحرف '+' عندما يبدأ arg بحرف '+'. يتم تخزين فهرس arg التالي في OPTIND. يتم تخزين وسيط الخيار، إن وجد، في OPTARG.

يمكن تغيير الخيار الأول الذي يتم فحصه عن طريق التعيين بشكل صريح لـ OPTIND. OPTIND له قيمة أولية 1، ويتم تعيينه عادةً إلى 1 عند الدخول إلى دالة shell ويتم استعادته عند الخروج. (الخيار POSIX_BUILTINS يعطل هذا، ويغير أيضًا طريقة حساب القيمة لتتوافق مع أصداف أخرى.) لا يتم إعادة تعيين OPTARG ويحتفظ بقيمته من آخر استدعاء لـ getopts. إذا تم إلغاء تعيين أي من OPTIND أو OPTARG بشكل صريح، فإنه يظل غير معين، ولا يتم تخزين الفهرس أو وسيط الخيار. لا يزال الخيار نفسه مخزنًا في name في هذه الحالة.

وجود علامة ':' في بداية optstring يتسبب في قيام getopts بتخزين حرف أي خيار غير صالح في OPTARG، وتعيين name إلى '?' لخيار غير معروف وإلى ':' عند فقدان وسيط مطلوب. بخلاف ذلك، يقوم getopts بتعيين name إلى '?' ويطبع رسالة خطأ عند عدم صلاحية الخيار. حالة الخروج تكون غير صفرية عندما لا توجد خيارات أخرى.

لاحظ أن read لا تحاول تغيير وضع معالجة الإدخال. الوضع الافتراضي هو الإدخال القياسي، والذي يتم فيه قراءة سطر كامل في كل مرة، لذا عادةً ما لن يقرأ `read -t' أي شيء حتى يتم كتابة سطر كامل. ومع ذلك، عند القراءة من الجهاز الطرفي باستخدام -k، يتم معالجة الإدخال حرفًا واحدًا في كل مرة؛ في هذه الحالة، يتم اختبار توفر الحرف الأول فقط، بحيث يمكن أن يعلق `read -t -k 2' على الحرف الثاني. استخدم مثيلين من `read -t -k' إذا لم يكن هذا هو المطلوب.

إذا كان الوسيط الأول يحتوي على `?'، فإن باقي هذه الكلمة سيتم استخدامه كـ prompt على المخرج القياسي عندما تكون الصدفة تفاعلية.

باستثناء ما هو مذكور أدناه لعلامات التحكم التي تغير السلوك، يتم إنشاء معلمة لكل name لا تشير بالفعل إلى واحدة. عند التواجد داخل دالة، يتم إنشاء معلمة جديدة لكل name (حتى تلك التي توجد بالفعل)، ويتم إلغاء تعيينها مرة أخرى عند اكتمال الدالة. انظر 'المعلمات المحلية' في zshparam(1). تنطبق نفس القواعد على المعلمات الخاصة بـ shell، والتي تحتفظ بسماتها الخاصة عند جعلها محلية.

لكل تعيين name=value، يتم تعيين المعلمة name إلى value. إذا تم حذف التعيين و name لا يشير إلى معلمة موجودة، يتم تهيئة معلمة جديدة بسلسلة فارغة، أو صفر، أو مصفوفة فارغة (حسب الاقتضاء)، *ما لم* يتم تعيين خيار shell TYPESET_TO_UNSET. عند تعيين هذا الخيار، يتم تسجيل سمات المعلمة ولكن تظل المعلمة غير معينة.

إذا لم يتم تعيين خيار shell TYPESET_SILENT، فلكل name المتبقية التي تشير إلى معلمة تم تعيينها بالفعل، يتم طباعة اسم وقيمة المعلمة في شكل تعيين. لا تتم طباعة أي شيء للمعلمات التي تم إنشاؤها حديثًا، أو عند إعطاء أي من علامات السمات المدرجة أدناه مع name. استخدام `+' بدلاً من ناقص لتقديم سمة يقوم بإلغاء تفعيلها.

إذا لم يكن name موجودًا، يتم طباعة أسماء وقيم جميع المعلمات. في هذه الحالة، تقوم علامات سمة بتقييد العرض إلى تلك المعلمات التي لديها السمات المحددة فقط، واستخدام `+' بدلاً من `-' لتقديم العلامة يقوم بتثبيط طباعة قيم المعلمات عندما لا يكون هناك اسم معلمة.

جميع أشكال الأمر تعالج تعيين القيم العددية. تعيين المصفوفة ممكن إذا تمت مطابقة أي من الكلمات المحجوزة declare، export، float، integer، local، readonly أو typeset عند تحليل السطر (ملاحظة: ليس عند تنفيذه). في هذه الحالة، يتم تحليل الوسائط كتعيينات، باستثناء أن بناء الجملة `+=' وخيار GLOB_ASSIGN غير مدعومين، والقيم العددية بعد = *لا* يتم تقسيمها بشكل أكبر إلى كلمات، حتى لو تم توسيعها (بغض النظر عن تعيين خيار KSH_TYPESET؛ هذا الخيار قديم).

أمثلة على الاختلافات بين تحليل الأوامر والكلمات المحجوزة:

# تحليل الكلمات المحجوزة
typeset svar=$(echo one word) avar=(several words)

يقوم ما سبق بإنشاء معلمة عددية svar ومعلمة مصفوفة avar كما لو كانت التعيينات هي

svar="one word"
avar=(several words)

من ناحية أخرى:

# واجهة مدمجة عادية
builtin typeset svar=$(echo two words)

يسبب الكلمة الرئيسية builtin أن يستخدم ما سبق الواجهة المدمجة القياسية لـ typeset حيث يتم تحليل الوسائط بنفس طريقة الأوامر الأخرى. هذا المثال ينشئ معلمة عددية svar تحتوي على القيمة two ومعلمة عددية أخرى words بدون قيمة. قيمة المصفوفة في هذه الحالة إما أن تسبب خطأ أو يتم التعامل معها كمجموعة غامضة من مؤهلات glob.

يتم السماح بالوسائط العشوائية إذا كانت تأخذ شكل تعيينات بعد توسيع سطر الأوامر؛ ومع ذلك، فإن هذه تقوم فقط بتعيين قيم عددية:

var='svar=val'
typeset $var

يقوم ما سبق بتعيين المعلمة العددية svar إلى القيمة val. الأقواس حول القيمة داخل var لن تسبب تعيين المصفوفة حيث سيتم التعامل معها كأحرف عادية عند استبدال $var. أي توسيع غير بديهي في جزء الاسم من التعيين يسبب التعامل مع الوسيط بهذه الطريقة:

typeset {var1,var2,var3}=name

بناء الجملة أعلاه صالح، وله التأثير المتوقع بتعيين المعلمات الثلاث إلى نفس القيمة، ولكن يتم تحليل سطر الأوامر كمجموعة من ثلاث وسائط عادية لسطر الأوامر لـ typeset بعد التوسيع. لذلك لا يمكن التعيين لمصفوفات متعددة بهذه الوسيلة.

لاحظ أنه يمكن تعطيل كل واجهة من واجهات الأوامر بشكل منفصل. على سبيل المثال، `disable -r typeset' يعطل الواجهة المحجوزة لـ typeset، ويكشف عن الواجهة المدمجة، بينما `disable typeset' يعطل المدمجة. لاحظ أن تعطيل الواجهة المحجوزة لـ typeset قد يسبب مشاكل في إخراج `typeset -p'، والذي يفترض أن الواجهة المحجوزة متاحة لاستعادة قيم المصفوفة والمصفوفة الترابطية.

بخلاف عبارات تعيين المعلمات، فإن حالة الخروج لـ typeset عند تعيين يتضمن استبدال أمر لا يعكس حالة خروج استبدال الأمر. لذلك، لاختبار خطأ في استبدال أمر، افصل تعريف المعلمة عن تهيئتها:

# خاطئ
typeset var1=$(exit 1) || echo "Trouble with var1"
# صحيح
typeset var1 && var1=$(exit 1) || echo "Trouble with var1"

لتهيئة معلمة param لمخرجات أمر وجعلها للقراءة فقط، استخدم typeset -r param أو readonly param بعد عبارة تعيين المعلمة.

إذا لم يتم إعطاء أي علامات سمات، وإما لا توجد وسائط name أو تم استخدام العلامة +m، فسيتم تقديم كل اسم معلمة مطبوعة بقائمة سمات تلك المعلمة (array، association، exported، float، integer، readonly، أو undefined للمعلمات المحملة تلقائيًا لم يتم تحميلها بعد). إذا تم استخدام +m مع علامات السمات، وتم تقديم جميع تلك العلامات مع بادئة `+'، يتم طباعة أسماء المعلمات المطابقة ولكن لا يتم طباعة قيمها.

تغير علامات التحكم التالية سلوك typeset:

+
إذا ظهر `+' بنفسه في كلمة منفصلة كآخر خيار، فسيتم طباعة أسماء جميع المعلمات (الدوال مع -f)، ولكن القيم (هياكل الدوال) لن تتم طباعتها. لا يمكن أن تظهر أي وسائط name، وخطأ لأي خيارات أخرى لتتبع `+'. تأثير `+' كما لو أن جميع علامات السمات التي تسبقه تم تقديمها مع بادئة `+'. على سبيل المثال، `typeset -U +' يعادل `typeset +U' ويعرض أسماء جميع المصفوفات التي لها سمة التفرد، بينما `typeset -f -U +' يعرض أسماء جميع الدوال القابلة للتحميل تلقائيًا. إذا كان + هو الخيار الوحيد، فسيتم أيضًا طباعة معلومات النوع (مصفوفة، للقراءة فقط، إلخ) لكل معلمة، بنفس طريقة `typeset +m "*"'.
-g
-g (عام) يعني أن أي معلمة ناتجة لن تكون مقيدة بنطاق محلي. لاحظ أن هذا لا يعني بالضرورة أن المعلمة ستكون عامة، حيث أن العلامة ستطبق على أي معلمة موجودة (حتى لو لم تكن معينة) من دالة محيطة. لا تؤثر هذه العلامة على المعلمة بعد الإنشاء، لذلك ليس لها تأثير عند سرد المعلمات الحالية، ولا يكون للعلامة +g أي تأثير إلا بالاقتران مع -m (انظر أدناه).
-m
إذا تم إعطاء العلامة -m، يتم اعتبار وسائط name كأنماط (استخدم علامات الاقتباس لمنع تفسيرها كأنماط ملفات). بدون علامات سمات، يتم طباعة جميع المعلمات (أو الدوال مع العلامة -f) ذات الأسماء المطابقة (لا يتم استخدام خيار shell TYPESET_SILENT في هذه الحالة).

إذا تم دمج العلامة +g مع -m، يتم إنشاء معلمة محلية جديدة لكل معلمة مطابقة ليست محلية بالفعل. بخلاف ذلك، تطبق -m جميع العلامات أو التعيينات الأخرى على المعلمات الموجودة.

باستثناء عند إجراء تعيينات مع name=value، فإن استخدام +m يجبر على طباعة المعلمات المطابقة وسماتها، حتى داخل دالة. لاحظ أن -m يتم تجاهلها إذا لم يتم تقديم أنماط، لذلك `typeset -m' يعرض السمات ولكن `typeset -a +m' لا يفعل ذلك.

-p [ n ]
إذا تم إعطاء الخيار -p، يتم طباعة المعلمات والقيم في شكل أمر typeset مع تعيين، بغض النظر عن العلامات والخيارات الأخرى. لاحظ أن العلامة -H على المعلمات يتم احترامها؛ لن يتم عرض أي قيمة لهذه المعلمات.

يمكن أن يتبع -p وسيط عدد صحيح اختياري. حاليًا، يتم دعم القيمة 1 فقط. في هذه الحالة، يتم طباعة المصفوفات والمصفوفات الترابطية مع أسطر جديدة بين العناصر المتداخلة لسهولة القراءة.

-T [ scalar[=value] array[=(value ...)] [ sep ] ]
هذه العلامة لها معنى مختلف عند استخدامها مع -f؛ انظر أدناه. بخلاف ذلك، يتطلب الخيار -T وجود صفر أو اثنين أو ثلاثة وسائط. بدون وسائط، يتم عرض قائمة المعلمات التي تم إنشاؤها بهذه الطريقة. مع وسيطين أو ثلاثة، الأول هو اسم معلمة عددية والثاني هو اسم معلمة مصفوفة (بهذا الترتيب) سيتم ربطهما ببعضهما البعض بطريقة $PATH و $path. الوسيط الثالث الاختياري هو فاصل أحادي الحرف سيتم استخدامه لربط عناصر المصفوفة لتشكيل العدد؛ إذا كان غائبًا، يتم استخدام نقطتين رأسيتين، كما هو الحال مع $PATH. الحرف الأول فقط من الفاصل له أهمية؛ أي أحرف متبقية يتم تجاهلها. متعدد البايتات

WHENCE

print information about command implementations

SYNOPSIS

whence [ -c | -v | -f | -p | -a | -m | -s | -S | -x num ] name ...

DESCRIPTION

For each name, whence prints information about how the shell would find and execute the command. This is done by checking for the command in the following order:

  1. Aliases defined with the alias builtin.
  2. Builtin commands.
  3. External commands defined with the hash builtin.
  4. Shell functions.
  5. External commands found in the directories listed in the PATH environment variable.
  6. Reserved words.

The options modify the behaviour of whence:

-c
Prints information about the location and nature of the command corresponding to name. The output is one of the following: alias, builtin, external, function, hash, reserved, or none, according as name corresponds to an alias, a built-in command, an external command, a shell function, a command defined with the hash builtin, a reserved word, or is not recognised. This takes precedence over -v and -c.
-f
Causes the contents of a shell function to be displayed, which would otherwise not happen unless the -c flag were used.
-p
Do a path search for name even if it is an alias, reserved word, shell function or builtin.
-a
Do a search for all occurrences of name throughout the command path. Normally only the first occurrence is printed.
-m
The arguments are taken as patterns (pattern characters should be quoted), and the information is displayed for each command matching one of these patterns.
-s
If a pathname contains symlinks, print the symlink-free pathname as well.
-S
As -s, but if the pathname had to be resolved by following multiple symlinks, the intermediate steps are printed, too. The symlink resolved at each step might be anywhere in the path.
-x num
Expand tabs when outputting shell functions using the -c option. This has the same effect as the -x option to the functions builtin.

zcompile [ -U ] [ -z | -k ] [ -R | -M ] file [ name ... ]
zcompile -ca [ -m ] [ -R | -M ] file [ name ... ]
zcompile -t file [ name ... ]
This builtin command can be used to compile functions or scripts, storing the compiled form in a file, and to examine files containing the compiled form. This allows faster autoloading of functions and sourcing of scripts by avoiding parsing of the text when the files are read.

The first form (without the -c, -a or -t options) creates a compiled file. If only the file argument is given, the output file has the name `file.zwc' and will be placed in the same directory as the file. The shell will load the compiled file instead of the normal function file when the function is autoloaded; see the section `Autoloading Functions' in zshmisc(1) for a description of how autoloaded functions are searched. The extension .zwc stands for `zsh word code'.

If there is at least one name argument, all the named files are compiled into the output file given as the first argument. If file does not end in .zwc, this extension is automatically appended. Files containing multiple compiled functions are called `digest' files, and are intended to be used as elements of the FPATH/fpath special array.

The second form, with the -c or -a options, writes the compiled definitions for all the named functions into file. For -c, the names must be functions currently defined in the shell, not those marked for autoloading. Undefined functions that are marked for autoloading may be written by using the -a option, in which case the fpath is searched and the contents of the definition files for those functions, if found, are compiled into file. If both -c and -a are given, names of both defined functions and functions marked for autoloading may be given. In either case, the functions in files written with the -c or -a option will be autoloaded as if the KSH_AUTOLOAD option were unset.

The reason for handling loaded and not-yet-loaded functions with different options is that some definition files for autoloading define multiple functions, including the function with the same name as the file, and, at the end, call that function. In such cases the output of `zcompile -c' does not include the additional functions defined in the file, and any other initialization code in the file is lost. Using `zcompile -a' captures all this extra information.

If the -m option is combined with -c or -a, the names are used as patterns and all functions whose names match one of these patterns will be written. If no name is given, the definitions of all functions currently defined or marked as autoloaded will be written.

Note the second form cannot be used for compiling functions that include redirections as part of the definition rather than within the body of the function; for example

fn1() { { ... } >~/logfile }

can be compiled but

fn1() { ... } >~/logfile

cannot. It is possible to use the first form of zcompile to compile autoloadable functions that include the full function definition instead of just the body of the function.

The third form, with the -t option, examines an existing compiled file. Without further arguments, the names of the original files compiled into it are listed. The first line of output shows the version of the shell which compiled the file and how the file will be used (i.e. by reading it directly or by mapping it into memory). With arguments, nothing is output and the return status is set to zero if definitions for all names were found in the compiled file, and non-zero if the definition for at least one name was not found.

Other options:

-U
Aliases are not expanded when compiling the named files.
-R
When the compiled file is read, its contents are copied into the shell's memory, rather than memory-mapped (see -M). This happens automatically on systems that do not support memory mapping.

When compiling scripts instead of autoloadable functions, it is often desirable to use this option; otherwise the whole file, including the code to define functions which have already been defined, will remain mapped, consequently wasting memory.

-M
The compiled file is mapped into the shell's memory when read. This is done in such a way that multiple instances of the shell running on the same host will share this mapped file. If neither -R nor -M is given, the zcompile builtin decides what to do based on the size of the compiled file.
-k
-z
These options are used when the compiled file contains functions which are to be autoloaded. If -z is given, the function will be autoloaded as if the KSH_AUTOLOAD option is not set, even if it is set at the time the compiled file is read, while if the -k is given, the function will be loaded as if KSH_AUTOLOAD is set. These options also take precedence over any -k or -z options specified to the autoload builtin. If neither of these options is given, the function will be loaded as determined by the setting of the KSH_AUTOLOAD option at the time the compiled file is read.

These options may also appear as many times as necessary between the listed names to specify the loading style of all following functions, up to the next -k or -z.

The created file always contains two versions of the compiled format, one for big-endian machines and one for small-endian machines. The upshot of this is that the compiled file is machine independent and if it is read or mapped, only one half of the file is actually used (and mapped).

Performs operations relating to zsh's loadable modules. Loading of modules while the shell is running (`dynamical loading') is not available on all operating systems, or on all installations on a particular operating system, although the zmodload command itself is always available and can be used to manipulate modules built into versions of the shell executable without dynamical loading.

Without arguments the names of all currently loaded binary modules are printed. The -L option causes this list to be in the form of a series of zmodload commands. Forms with arguments are:

zmodload [ -is ] name ...
zmodload -u [ -i ] name ...
In the simplest case, zmodload loads a binary module. The module must be in a file with a name consisting of the specified name followed by a standard suffix, usually `.so' (`.sl' on HPUX). If the module to be loaded is already loaded the duplicate module is ignored. If zmodload detects an inconsistency, such as an invalid module name or circular dependency list, the current code block is aborted. If it is available, the module is loaded if necessary, while if it is not available, non-zero status is silently returned. The option -i is accepted for compatibility but has no effect.

مع -u، تقوم zmodload بإلغاء تحميل الوحدات النمطية. يجب إعطاء نفس الاسم الذي تم إعطاؤه عند تحميل الوحدة النمطية، ولكن ليس من الضروري أن تكون الوحدة النمطية موجودة في نظام الملفات. الخيار -i يمنع الخطأ إذا كانت الوحدة النمطية غير محملة بالفعل (أو لم يتم تحميلها قط).

كل وحدة نمطية لها وظيفة بدء ووظيفة تنظيف. لن يتم تحميل الوحدة النمطية إذا فشلت وظيفة البدء الخاصة بها. وبالمثل، لا يمكن إلغاء تحميل الوحدة النمطية إلا إذا تم تشغيل وظيفة التنظيف الخاصة بها بنجاح.

zmodload -F [ -almLe -P param ] module [ [+-]feature ... ]
zmodload -F يسمح بتحكم أكثر انتقائية في الميزات التي توفرها الوحدات النمطية. بدون خيارات بخلاف -F، يتم تحميل الوحدة النمطية المسماة module، إذا لم تكن محملة بالفعل، ويتم تعيين قائمة features إلى الحالة المطلوبة. إذا لم يتم تحديد أي features، يتم تحميل الوحدة النمطية، إذا لم تكن محملة بالفعل، ولكن حالة الميزات لا تتغير. يمكن أن يسبق كل ميزة + لتشغيل الميزة، أو - لإيقاف تشغيلها؛ يُفترض + إذا لم يكن هناك حرف موجود. تُترك أي ميزة لم يتم ذكرها صراحةً في حالتها الحالية؛ إذا لم تكن الوحدة النمطية محملة سابقًا، فهذا يعني أن أي ميزات من هذا القبيل ستظل معطلة. حالة الإرجاع هي صفر إذا تم تعيين جميع الميزات، 1 إذا فشلت الوحدة النمطية في التحميل، و 2 إذا لم يتمكن من تعيين بعض الميزات (على سبيل المثال، لم يكن من الممكن إضافة معلمة لأن هناك معلمة مختلفة بنفس الاسم) ولكن تم تحميل الوحدة النمطية.

الميزات القياسية هي الوظائف المدمجة، والشروط، والمعلمات، ووظائف الرياضيات؛ يتم الإشارة إليها بالبادئات `b:'، `c:' (`C:' لشرط وسيط)، `p:'، و `f:'، على التوالي، متبوعة بالاسم الذي ستحصل عليه الميزة المقابلة في shell. على سبيل المثال، `b:strftime' تشير إلى وظيفة مدمجة تسمى strftime و p:EPOCHSECONDS تشير إلى معلمة تسمى EPOCHSECONDS. قد توفر الوحدة النمطية ميزات أخرى (`مجردة') خاصة بها كما هو موضح في وثائقها؛ هذه ليس لها بادئة.

مع -l أو -L، يتم سرد الميزات التي توفرها الوحدة النمطية. مع -l وحده، يتم عرض قائمة بالميزات مع حالاتها، ميزة واحدة في كل سطر. مع -L وحده، يتم عرض أمر zmodload -F الذي من شأنه أن يؤدي إلى تشغيل الميزات الممكّنة للوحدة النمطية. مع -L وحده، يتم عرض أمر zmodload -F الذي من شأنه أن يؤدي إلى تشغيل الميزات الممكّنة للوحدة النمطية. مع -lL، يتم عرض أمر zmodload -F الذي من شأنه أن يؤدي إلى تعيين جميع الميزات إلى حالتها الحالية. إذا تم إعطاء أحد هذه التركيبات مع الخيار -P param، فسيتم تعيين المعلمة param إلى مصفوفة من الميزات، إما الميزات مع حالتها أو (إذا تم إعطاء -L وحده) الميزات الممكّنة.

مع الخيار -L، يمكن حذف اسم الوحدة النمطية؛ عندئذٍ يتم طباعة قائمة بجميع الميزات الممكّنة لجميع الوحدات النمطية التي توفر ميزات في شكل أوامر zmodload -F. إذا تم إعطاء -l أيضًا، فسيتم إخراج حالة كل من الميزات الممكّنة والمعطلة بهذا الشكل. قد يتم توفير مجموعة من الميزات مع -l أو -L واسم الوحدة النمطية؛ في هذه الحالة، يتم فقط النظر في حالة تلك الميزات. يمكن أن يسبق كل ميزة + أو - ولكن الحرف ليس له تأثير. إذا لم يتم توفير مجموعة من الميزات، فسيتم اعتبار جميع الميزات.

مع -e، يقوم الأمر أولاً باختبار أن الوحدة النمطية محملة؛ إذا لم تكن كذلك، يتم إرجاع الحالة 1. إذا كانت الوحدة النمطية محملة، يتم فحص قائمة الميزات المعطاة كوسيط. يتم ببساطة اختبار أي ميزة معطاة بدون بادئة لمعرفة ما إذا كانت الوحدة النمطية توفرها؛ يتم اختبار أي ميزة معطاة ببادئة + أو - لمعرفة ما إذا كانت متوفرة وفي الحالة المعطاة. إذا نجحت اختبارات جميع الميزات في القائمة، يتم إرجاع الحالة 0، وإلا فالحالة 1.

مع -m، يتم أخذ كل إدخال في قائمة الميزات المعطاة كنمط لمطابقته مع قائمة الميزات التي توفرها الوحدة النمطية. يجب إعطاء بادئة + أو - بشكل صريح. لا يمكن دمج هذا مع الخيار -a لأن التحميل التلقائي يجب تحديده بشكل صريح.

مع -a، يتم وضع علامة على قائمة الميزات المعطاة للتحميل التلقائي من الوحدة النمطية المحددة، والتي قد لا تكون محملة بعد. يمكن أن يظهر + اختياري قبل اسم الميزة. إذا كانت الميزة مسبوقة بـ -، يتم إزالة أي تحميل تلقائي موجود. يمكن استخدام الخيارين -l و -L لسرد التحميلات التلقائية. التحميل التلقائي خاص بالميزات الفردية؛ عند تحميل الوحدة النمطية، يتم تمكين الميزة المطلوبة فقط. يتم الاحتفاظ بطلبات التحميل التلقائي إذا تم إلغاء تحميل الوحدة النمطية لاحقًا حتى يتم إصدار `zmodload -Fa module -feature' بشكل صريح. ليس خطأً طلب تحميل تلقائي لميزة وحدة نمطية محملة بالفعل.

عند تحميل الوحدة النمطية، يتم فحص كل تحميل تلقائي مقابل الميزات التي توفرها الوحدة النمطية فعليًا؛ إذا لم يتم توفير الميزة، يتم حذف طلب التحميل التلقائي. يتم إخراج رسالة تحذير؛ إذا تم تحميل الوحدة النمطية لتوفير ميزة مختلفة، وكان هذا التحميل التلقائي ناجحًا، فلا يوجد تأثير على حالة الأمر الحالي. إذا كانت الوحدة النمطية محملة بالفعل في الوقت الذي تم فيه تشغيل zmodload -Fa، تتم طباعة رسالة خطأ ويتم إرجاع الحالة 1.

يمكن استخدام zmodload -Fa مع الخيارات -l، -L، -e، و -P لسرد واختبار وجود الميزات القابلة للتحميل التلقائي. في هذه الحالة، يتم تجاهل -l إذا تم تحديد -L. zmodload -FaL بدون اسم وحدة نمطية يسرد التحميلات التلقائية لجميع الوحدات النمطية.

لاحظ أنه لا يمكن تحميل الميزات القياسية فقط كما هو موضح أعلاه تلقائيًا؛ تتطلب الميزات الأخرى تحميل الوحدة النمطية قبل التمكين.

zmodload -d [ -L ] [ name ]
zmodload -d name dep ...
zmodload -ud name [ dep ... ]
يمكن استخدام الخيار -d لتحديد تبعيات الوحدة النمطية. سيتم تحميل الوحدات النمطية المسماة في الحجج الثانية وما يليها قبل الوحدة النمطية المسماة في الحجة الأولى.

مع -d وحجة واحدة، يتم سرد جميع التبعيات لهذه الوحدة النمطية. مع -d وبدون حجج، يتم سرد جميع تبعيات الوحدة النمطية. يتم هذا الإدراج بشكل افتراضي بتنسيق شبيه بـ Makefile. الخيار -L يغير هذا التنسيق إلى قائمة أوامر zmodload -d.

إذا تم استخدام كل من -d و -u، فسيتم إزالة التبعيات. إذا تم تقديم حجة واحدة فقط، فسيتم إزالة جميع التبعيات لهذه الوحدة النمطية.

zmodload -ab [ -L ]
zmodload -ab [ -i ] name [ builtin ... ]
zmodload -ub [ -i ] builtin ...
الخيار -ab يحدد الوظائف المدمجة المحملة تلقائيًا. يحدد الوظائف builtins المحددة. عند استدعاء أي من هذه الوظائف المدمجة، يتم تحميل الوحدة النمطية المحددة في الحجة الأولى ويتم تمكين جميع ميزاتها (للتحكم الانتقائي في الميزات، استخدم `zmodload -F -a' كما هو موضح أعلاه). إذا تم تقديم الاسم name فقط، فسيتم تعريف وظيفة مدمجة واحدة، بنفس اسم الوحدة النمطية. -i يمنع الخطأ إذا كانت الوظيفة المدمجة محددة بالفعل أو محملة تلقائيًا، ولكن ليس إذا كانت هناك وظيفة مدمجة أخرى بنفس الاسم محددة بالفعل.

مع -ab وبدون حجج، يتم سرد جميع الوظائف المدمجة المحملة تلقائيًا، مع عرض اسم الوحدة النمطية (إذا كان مختلفًا) بين قوسين بعد اسم الوظيفة المدمجة. الخيار -L يغير هذا التنسيق إلى قائمة أوامر zmodload -a.

إذا تم استخدام -b مع الخيار -u، فإنه يزيل الوظائف المدمجة المعرفة سابقًا باستخدام -ab. هذا ممكن فقط إذا لم يتم تحميل الوظيفة المدمجة بعد. -i يمنع الخطأ إذا تم إزالة الوظيفة المدمجة بالفعل (أو لم تكن موجودة قط).

يتم الاحتفاظ بطلبات التحميل التلقائي إذا تم إلغاء تحميل الوحدة النمطية لاحقًا حتى يتم إصدار `zmodload -ub builtin' بشكل صريح.

zmodload -ac [ -IL ]
zmodload -ac [ -iI ] name [ cond ... ]
zmodload -uc [ -iI ] cond ...
يستخدم الخيار -ac لتحديد رموز الشروط المحملة تلقائيًا. تعطي السلاسل cond أسماء الشروط التي تحددها الوحدة النمطية. يستخدم الخيار الاختياري -I لتحديد أسماء شروط الوسيط. بدون هذا الخيار، يتم تحديد أسماء شروط البادئة.

إذا لم يتم تقديم أسماء شروط، فسيتم سرد جميع الأسماء المحددة ( كسلسلة من أوامر zmodload إذا تم إعطاء الخيار -L).

يزيل الخيار -uc التعريفات للشروط المحملة تلقائيًا.

zmodload -ap [ -L ]
zmodload -ap [ -i ] name [ parameter ... ]
zmodload -up [ -i ] parameter ...
الخيار -p يشبه الخيارين -b و -c، ولكنه يجعل zmodload يعمل على المعلمات المحملة تلقائيًا بدلاً من ذلك.
zmodload -af [ -L ]
zmodload -af [ -i ] name [ function ... ]
zmodload -uf [ -i ] function ...
الخيار -f يشبه الخيارات -b، -p، و -c، ولكنه يجعل zmodload يعمل على وظائف الرياضيات المحملة تلقائيًا بدلاً من ذلك.
zmodload -a [ -L ]
zmodload -a [ -i ] name [ builtin ... ]
zmodload -ua [ -i ] builtin ...
مكافئ لـ -ab و -ub.
zmodload -e [ -A ] [ string ... ]
الخيار -e بدون حجج يسرد جميع الوحدات النمطية المحملة؛ إذا تم إعطاء الخيار -A أيضًا، فسيتم أيضًا عرض أسماء مستعارة للوحدات النمطية المقابلة للوحدات النمطية المحملة.
