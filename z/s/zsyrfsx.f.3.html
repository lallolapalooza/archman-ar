الغرض:

!>
!>    ZSYRFSX يحسن الحل المحسوب لنظام المعادلات الخطية
!>    عندما تكون مصفوفة المعاملات متناظرة غير محددة،
!>    ويوفر حدودًا للخطأ وتقديرات للخطأ الخلفي للحل.
!>    بالإضافة إلى حد الخطأ المعياري، يوفر الكود
!>    حد خطأ أقصى مكوني إن أمكن. انظر التعليقات لـ
!>    ERR_BNDS_NORM و ERR_BNDS_COMP لتفاصيل حدود الخطأ.
!>
!>    قد تم موازنة نظام المعادلات الأصلي
!>    قبل استدعاء هذا الروتين، كما هو موضح بواسطة الوسيطات EQUED و S
!>    أدناه. في هذه الحالة، يتم إرجاع الحل وحدود الخطأ
!>    للنظام الأصلي غير المتوازن.
!> 
!>     بعض المعلمات الاختيارية مجمعة في مصفوفة PARAMS. هذه
!>     الإعدادات تحدد كيفية إجراء التحسين، ولكن غالبًا ما تكون
!>     الإعدادات الافتراضية مقبولة. إذا كانت الإعدادات الافتراضية مقبولة،
!>     يمكن للمستخدمين تمرير NPARAMS = 0 مما يمنع الكود المصدري من الوصول
!>     إلى الوسيط PARAMS.
!>

المعلمات

UPLO 
!>          UPLO هو CHARACTER*1
!>       = 'U':  المثلث العلوي لـ A مخزن؛
!>       = 'L':  المثلث السفلي لـ A مخزن.
!>

EQUED

!>          EQUED هو CHARACTER*1
!>     يحدد شكل الموازنة التي تم إجراؤها على A
!>     قبل استدعاء هذا الروتين. هذا ضروري لحساب
!>     الحل وحدود الخطأ بشكل صحيح.
!>       = 'N':  لا توجد موازنة
!>       = 'Y':  موازنة الصفوف والأعمدة، أي تم استبدال A بـ
!>               diag(S) * A * diag(S).
!>               تم تغيير الجانب الأيمن B وفقًا لذلك.
!>

N

!>          N هو INTEGER
!>     رتبة المصفوفة A.  N >= 0.
!>

NRHS

!>          NRHS هو INTEGER
!>     عدد الأجناب اليمنى، أي عدد أعمدة
!>     المصفوفتين B و X.  NRHS >= 0.
!>

A

!>          A هي مصفوفة COMPLEX*16، البعد (LDA,N)
!>     المصفوفة المتناظرة A.  إذا كان UPLO = 'U'، فإن الجزء العلوي
!>     المثلثي بحجم N×N من A يحتوي على الجزء العلوي
!>     المثلثي للمصفوفة A، ولا يتم الإشارة إلى الجزء السفلي
!>     المثلثي الصارم لـ A.  إذا كان UPLO = 'L'، فإن الجزء السفلي
!>     المثلثي بحجم N×N من A يحتوي على الجزء السفلي
!>     المثلثي للمصفوفة A، ولا يتم الإشارة إلى الجزء العلوي
!>     المثلثي الصارم لـ A.
!>

LDA

!>          LDA هو INTEGER
!>     البعد الرائد للمصفوفة A.  LDA >= max(1,N).
!>

AF

!>          AF هي مصفوفة COMPLEX*16، البعد (LDAF,N)
!>     الشكل المحلل للمصفوفة A.  AF يحتوي على المصفوفة القطرية الكتلة D
!>     والمضاعفات المستخدمة للحصول على U أو L من تحليل A = U*D*U**T أو A =
!>     L*D*L**T كما تم حسابه بواسطة ZSYTRF.
!>

LDAF

!>          LDAF هو INTEGER
!>     البعد الرائد للمصفوفة AF.  LDAF >= max(1,N).
!>

IPIV

!>          IPIV هي مصفوفة INTEGER، البعد (N)
!>     تفاصيل التبادلات والبنية الكتلية لـ D
!>     كما تم تحديده بواسطة ZSYTRF.
!>

S

!>          S هي مصفوفة DOUBLE PRECISION، البعد (N)
!>     عوامل القياس لـ A.  إذا كان EQUED = 'Y'، يتم ضرب A
!>     على اليسار واليمين بـ diag(S).  S هي وسيطة إدخال إذا كان FACT =
!>     'F'؛ بخلاف ذلك، S هي وسيطة إخراج.  إذا كان FACT = 'F' و EQUED
!>     = 'Y'، يجب أن يكون كل عنصر في S موجبًا.  إذا كان S هو إخراج، فإن كل
!>     عنصر في S هو قوة للجذر. إذا كان S هو إدخال، يجب أن يكون كل
!>     عنصر في S قوة للجذر لضمان حل موثوق
!>     وتقديرات خطأ.  القياس بقوى الجذر لا يسبب
!>     أخطاء تقريب إلا إذا كانت النتيجة أقل من الحد الأدنى أو تجاوزت الحد الأعلى.
!>     تؤدي أخطاء التقريب أثناء القياس إلى التحسين بمصفوفة
!>     غير مكافئة للمصفوفة المدخلة، مما ينتج عنه تقديرات خطأ
!>     قد لا تكون موثوقة.
!>

B

!>          B هي مصفوفة COMPLEX*16، البعد (LDB,NRHS)
!>     مصفوفة الجانب الأيمن B.
!>

LDB

!>          LDB هو INTEGER
!>     البعد الرائد للمصفوفة B.  LDB >= max(1,N).
!>

X

!>          X هي مصفوفة COMPLEX*16، البعد (LDX,NRHS)
!>     عند الإدخال، مصفوفة الحل X، كما تم حسابها بواسطة ZGETRS.
!>     عند الخروج، مصفوفة الحل المحسنة X.
!>

LDX

!>          LDX هو INTEGER
!>     البعد الرائد للمصفوفة X.  LDX >= max(1,N).
!>

RCOND

!>          RCOND هو DOUBLE PRECISION
!>     مقلوب رقم الحالة المقاس. هذا تقدير لـ
!>     مقلوب رقم الحالة Skeel للمصفوفة A بعد
!>     الموازنة (إذا تم إجراؤها).  إذا كان هذا أقل من دقة الآلة
!>     (خاصة، إذا كان صفرًا)، فإن المصفوفة منفردة
!>     لأغراض الدقة العاملة.  لاحظ أن الخطأ قد لا يزال صغيرًا حتى
!>     لو كان هذا الرقم صغيرًا جدًا وتبدو المصفوفة غير مستقرة
!>     بسبب حالتها.
!>

BERR

!>          BERR هي مصفوفة DOUBLE PRECISION، البعد (NRHS)
!>     خطأ خلفي نسبي مكوني. هذا هو
!>     الخطأ الخلفي النسبي المكوني لكل متجه حل X(j)
!>     (أي، أصغر تغيير نسبي في أي عنصر من A أو B يجعل X(j) حلاً دقيقًا).
!>

N_ERR_BNDS

!>          N_ERR_BNDS هو INTEGER
!>     عدد حدود الخطأ المراد إرجاعها لكل جانب أيمن
!>     ولكل نوع (معياري أو مكوني). انظر ERR_BNDS_NORM و
!>     ERR_BNDS_COMP أدناه.
!>

ERR_BNDS_NORM

!>          ERR_BNDS_NORM هي مصفوفة DOUBLE PRECISION، البعد (NRHS, N_ERR_BNDS)
!>     لكل جانب أيمن، تحتوي هذه المصفوفة على معلومات حول
!>     حدود خطأ مختلفة وأرقام حالات مقابلة لـ
!>     الخطأ النسبي المعياري، والذي يتم تعريفه على النحو التالي:
!>
!>     الخطأ النسبي المعياري في متجه الحل i:
!>             max_j (abs(XTRUE(j,i) - X(j,i)))
!>            ------------------------------
!>                  max_j abs(X(j,i))
!>
!>     المصفوفة هي
