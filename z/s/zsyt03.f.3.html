توثيق الدالة/الروتين الفرعي

subroutine zsyt03 (character uplo, integer n, complex*16, dimension( lda, * ) a, integer lda, complex*16, dimension( ldainv, * ) ainv, integer ldainv, complex*16, dimension( ldwork, * ) work, integer ldwork, double precision, dimension( * ) rwork, double precision rcond, double precision resid)

ZSYT03

الغرض:

!>
!> ZSYT03 يحسب الباقي لمصفوفة متناظرة مركبة مضروبة في معكوسها:
!>    norm( I - A*AINV ) / ( N * norm(A) * norm(AINV) * EPS )
!> حيث EPS هو خطأ الآلة.
!>

المعلمات

UPLO 
!>          UPLO is CHARACTER*1
!>          يحدد ما إذا كان الجزء العلوي أو السفلي المثلثي من المصفوفة المتناظرة المركبة A مخزنًا:
!>          = 'U':  مثلثي علوي
!>          = 'L':  مثلثي سفلي
!>

N

!>          N is INTEGER
!>          عدد الصفوف والأعمدة للمصفوفة A.  N >= 0.
!>

A

!>          A is COMPLEX*16 array, dimension (LDA,N)
!>          المصفوفة الأصلية المتناظرة المركبة A.
!>

LDA

!>          LDA is INTEGER
!>          البعد الرائد للمصفوفة A.  LDA >= max(1,N)
!>

AINV

!>          AINV is COMPLEX*16 array, dimension (LDAINV,N)
!>          عند الدخول، معكوس المصفوفة A، مخزن كمصفوفة متناظرة
!>          بنفس تنسيق A.
!>          في هذا الإصدار، يتم توسيع AINV إلى مصفوفة كاملة ويتم
!>          ضربها في A، لذلك سيتم تغيير المثلث المقابل لـ AINV؛
!>          أي، إذا كان الجزء المثلثي العلوي من AINV مخزنًا،
!>          فسيتم استخدام الجزء المثلثي السفلي كمساحة عمل.
!>

LDAINV

!>          LDAINV is INTEGER
!>          البعد الرائد للمصفوفة AINV.  LDAINV >= max(1,N).
!>

WORK

!>          WORK is COMPLEX*16 array, dimension (LDWORK,N)
!>

LDWORK

!>          LDWORK is INTEGER
!>          البعد الرائد للمصفوفة WORK.  LDWORK >= max(1,N).
!>

RWORK

!>          RWORK is DOUBLE PRECISION array, dimension (N)
!>

RCOND

!>          RCOND is DOUBLE PRECISION
!>          المقلوب للمعامل الشرطي لـ A، محسوب كـ
!>          RCOND = 1/ (norm(A) * norm(AINV)).
!>

RESID

!>          RESID is DOUBLE PRECISION
!>          norm(I - A*AINV) / ( N * norm(A) * norm(AINV) * EPS )
!>
