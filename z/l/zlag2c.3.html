subroutine zlag2c (m, n, a, lda, sa, ldsa, info)
ZLAG2C تحول مصفوفة معقدة ذات دقة مزدوجة إلى مصفوفة معقدة ذات دقة مفردة.

توثيق الوظيفة/الإجراء الفرعي

subroutine zlag2c (integer m, integer n, complex*16, dimension( lda, * ) a, integer lda, complex, dimension( ldsa, * ) sa, integer ldsa, integer info)

ZLAG2C تحول مصفوفة معقدة ذات دقة مزدوجة إلى مصفوفة معقدة ذات دقة مفردة.

الغرض:

!>
!> ZLAG2C تحول مصفوفة COMPLEX*16، SA، إلى مصفوفة COMPLEX، A.
!>
!> RMAX هو تجاوز حسابات الدقة المفردة
!> ZLAG2C تتحقق من أن جميع مدخلات A تقع بين -RMAX و
!> RMAX. إذا لم يكن الأمر كذلك، يتم إيقاف التحويل ورفع علم.
!>
!> هذا إجراء فرعي مساعد لذلك لا يوجد فحص للمدخلات.
!>

المعلمات

M 
!>          M هو INTEGER
!>          عدد الأسطر في المصفوفة A.  M >= 0.
!>

N

!>          N هو INTEGER
!>          عدد الأعمدة في المصفوفة A.  N >= 0.
!>

A

!>          A هي مصفوفة COMPLEX*16، dimension (LDA,N)
!>          عند الدخول، مصفوفة المعاملات M-by-N A.
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرائد للمصفوفة A.  LDA >= max(1,M).
!>

SA

!>          SA هي مصفوفة COMPLEX، dimension (LDSA,N)
!>          عند الخروج، إذا كان INFO=0، فإن مصفوفة المعاملات M-by-N SA؛ إذا
!>          كان INFO>0، فإن محتوى SA غير محدد.
!>

LDSA

!>          LDSA هو INTEGER
!>          البعد الرائد للمصفوفة SA.  LDSA >= max(1,M).
!>

INFO

!>          INFO هو INTEGER
!>          = 0:  خروج ناجح.
!>          = 1:  إحدى مدخلات المصفوفة A أكبر من الحد الأقصى للتجاوز
!>                للدقة المفردة، في هذه الحالة، المحتوى
!>                لـ SA عند الخروج غير محدد.
!>
