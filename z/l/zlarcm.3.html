subroutine zlarcm (integer m, integer n, double precision, dimension( lda, * ) a, integer lda, complex*16, dimension( ldb, * ) b, integer ldb, complex*16, dimension( ldc, * ) c, integer ldc, double precision, dimension( * ) rwork)

ZLARCM تقوم بنسخ جزء أو كل مصفوفة ثنائية الأبعاد حقيقية إلى مصفوفة مركبة.

الغرض:

!>
!> ZLARCM تقوم بضرب مصفوفة بسيطة للغاية:
!>          C := A * B,
!> حيث A هي M في M وحقيقية؛ B هي M في N ومركبة؛
!> C هي M في N ومركبة.
!>

المعلمات

M 
!>          M هو عدد صحيح (INTEGER)
!>          عدد صفوف المصفوفة A والمصفوفة C.
!>          M >= 0.
!>

N

!>          N هو عدد صحيح (INTEGER)
!>          عدد الأعمدة والصفوف للمصفوفة B و
!>          عدد الأعمدة للمصفوفة C.
!>          N >= 0.
!>

A

!>          A هي مصفوفة مزدوجة الدقة (DOUBLE PRECISION)، أبعادها (LDA, M)
!>          عند الإدخال، تحتوي A على المصفوفة A بحجم M في M.
!>

LDA

!>          LDA هو عدد صحيح (INTEGER)
!>          البعد الرئيسي للمصفوفة A. LDA >=max(1,M).
!>

B

!>          B هي مصفوفة مركبة*16 (COMPLEX*16)، أبعادها (LDB, N)
!>          عند الإدخال، تحتوي B على المصفوفة B بحجم M في N.
!>

LDB

!>          LDB هو عدد صحيح (INTEGER)
!>          البعد الرئيسي للمصفوفة B. LDB >=max(1,M).
!>

C

!>          C هي مصفوفة مركبة*16 (COMPLEX*16)، أبعادها (LDC, N)
!>          عند الخروج، تحتوي C على المصفوفة C بحجم M في N.
!>

LDC

!>          LDC هو عدد صحيح (INTEGER)
!>          البعد الرئيسي للمصفوفة C. LDC >=max(1,M).
!>

RWORK

!>          RWORK هي مصفوفة مزدوجة الدقة (DOUBLE PRECISION)، أبعادها (2*M*N)
!>
