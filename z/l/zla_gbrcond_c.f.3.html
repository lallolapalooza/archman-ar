الوظائف/الإجراءات الفرعية


double precision function zla_gbrcond_c (trans, n, kl, ku, ab, ldab, afb, ldafb, ipiv, c, capply, info, work, rwork)
ZLA_GBRCOND_C تحسب معامل الشرط للمعيار اللانهائي لـ op(A)*inv(diag(c)) للمصفوفات العامة النطاقية.

وثائق الوظيفة/الإجراء الفرعي

double precision function zla_gbrcond_c (character trans, integer n, integer kl, integer ku, complex*16, dimension( ldab, * ) ab, integer ldab, complex*16, dimension( ldafb, * ) afb, integer ldafb, integer, dimension( * ) ipiv, double precision, dimension( * ) c, logical capply, integer info, complex*16, dimension( * ) work, double precision, dimension( * ) rwork)

ZLA_GBRCOND_C تحسب معامل الشرط للمعيار اللانهائي لـ op(A)*inv(diag(c)) للمصفوفات العامة النطاقية.

الغرض:

!>
!>    ZLA_GBRCOND_C تحسب معامل الشرط للمعيار اللانهائي لـ
!>    op(A) * inv(diag(C)) حيث C هو متجه DOUBLE PRECISION.
!>

المعاملات

TRANS 
!>          TRANS is CHARACTER*1
!>     يحدد شكل نظام المعادلات:
!>       = 'N':  A * X = B     (بدون تبديل)
!>       = 'T':  A**T * X = B  (تبديل)
!>       = 'C':  A**H * X = B  (تبديل مترافق = تبديل)
!>

N

!>          N is INTEGER
!>     عدد معادلات الخطية، أي، ترتيب
!>     المصفوفة A.  N >= 0.
!>

KL

!>          KL is INTEGER
!>     عدد الأقطار الفرعية داخل نطاق A.  KL >= 0.
!>

KU

!>          KU is INTEGER
!>     عدد الأقطار الفائقة داخل نطاق A.  KU >= 0.
!>

AB

!>          AB is COMPLEX*16 array, dimension (LDAB,N)
!>     عند الإدخال، المصفوفة A بتخزين النطاق، في الصفوف من 1 إلى KL+KU+1.
!>     يتم تخزين العمود j من A في العمود j من
!>     المصفوفة AB كالتالي:
!>     AB(KU+1+i-j,j) = A(i,j) for max(1,j-KU)<=i<=min(N,j+kl)
!>

LDAB

!>          LDAB is INTEGER
!>     البعد الرائد للمصفوفة AB.  LDAB >= KL+KU+1.
!>

AFB

!>          AFB is COMPLEX*16 array, dimension (LDAFB,N)
!>     تفاصيل تحليل LU للمصفوفة النطاقية A، كما
!>     تم حسابها بواسطة ZGBTRF.  يتم تخزين U كمصفوفة
!>     مثلثة عليا نطاقية بأقطار فائقة KL+KU في الصفوف من 1 إلى KL+KU+1،
!>     وتُخزن العوامل المستخدمة أثناء التحليل
!>     في الصفوف KL+KU+2 إلى 2*KL+KU+1.
!>

LDAFB

!>          LDAFB is INTEGER
!>     البعد الرائد للمصفوفة AFB.  LDAFB >= 2*KL+KU+1.
!>

IPIV

!>          IPIV is INTEGER array, dimension (N)
!>     مؤشرات المحور من التحليل A = P*L*U
!>     كما تم حسابه بواسطة ZGBTRF؛ تم تبديل الصف i من المصفوفة
!>     مع الصف IPIV(i).
!>

C

!>          C is DOUBLE PRECISION array, dimension (N)
!>     المتجه C في الصيغة op(A) * inv(diag(C)).
!>

CAPPLY

!>          CAPPLY is LOGICAL
!>     إذا كان .TRUE.، فقم بالوصول إلى المتجه C في الصيغة أعلاه.
!>

INFO

!>          INFO is INTEGER
!>       = 0:  الخروج الناجح.
!>     i > 0:  المعامل i غير صالح.
!>

WORK

!>          WORK is COMPLEX*16 array, dimension (2*N).
!>     مساحة العمل.
!>

RWORK

!>          RWORK is DOUBLE PRECISION array, dimension (N).
!>     مساحة العمل.
!>
