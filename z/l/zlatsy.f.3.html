subroutine zlatsy (character uplo, integer n, complex*16, dimension( ldx, * ) x, integer ldx, integer, dimension( * ) iseed)

ZLATSY

الغرض:

!>
!> ZLATSY ينشئ مصفوفة اختبار خاصة للتحليل المتماثل (غير المحدد) المركب.
!> ستكون كتل المحور للمصفوفة التي تم إنشاؤها بالترتيب التالي:
!> كتلة محور 2x2، غير قابلة للقطر
!> كتلة محور 1x1
!> كتلة محور 2x2، قابلة للقطر
!> (تتكرر الدورة)
!> مطلوب تبديل صف لكل كتلة 2x2 غير قابلة للتحويل.
!>

المعلمات

UPLO 
!>          UPLO هو CHARACTER
!>          يحدد ما إذا كانت المصفوفة التي سيتم إنشاؤها ستكون عليا أم
!>          سفلية.
!>          = 'U':  عليا
!>          = 'L':  سفلية
!>

N

!>          N هو INTEGER
!>          بعد المصفوفة التي سيتم إنشاؤها.
!>

X

!>          X هو مصفوفة COMPLEX*16، البعد (LDX,N)
!>          المصفوفة التي تم إنشاؤها، تتكون من كتل قطرية 3x3 و 2x2
!>          التي تنتج تسلسل المحور المذكور أعلاه.
!>          المصفوفة خارج هذه الكتل القطرية صفرية.
!>

LDX

!>          LDX هو INTEGER
!>          البعد الرئيسي للمصفوفة X.
!>

ISEED

!>          ISEED هي مصفوفة INTEGER، البعد (4)
!>          عند الإدخال، البذرة لمولد الأرقام العشوائية. يجب أن يكون
!>          العدد الصحيح الأخير من الأعداد الأربعة فرديًا. (تم تعديله عند الخروج)
!>
