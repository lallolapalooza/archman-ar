ZLAQR5 تنفذ دورة QR واحدة متعددة الإزاحات مع فقاعة صغيرة.

توثيق الوظيفة/الوحدة الفرعية

subroutine zlaqr5 (logical wantt, logical wantz, integer kacc22, integer n, integer ktop, integer kbot, integer nshfts, complex*16, dimension( * ) s, complex*16, dimension( ldh, * ) h, integer ldh, integer iloz, integer ihiz, complex*16, dimension( ldz, * ) z, integer ldz, complex*16, dimension( ldv, * ) v, integer ldv, complex*16, dimension( ldu, * ) u, integer ldu, integer nv, complex*16, dimension( ldwv, * ) wv, integer ldwv, integer nh, complex*16, dimension( ldwh, * ) wh, integer ldwh)

ZLAQR5 تنفذ دورة QR واحدة متعددة الإزاحات مع فقاعة صغيرة.

الغرض:

!>
!>    ZLAQR5، التي تستدعيها ZLAQR0، تنفذ دورة
!>    QR واحدة متعددة الإزاحات مع فقاعة صغيرة.
!>

المعلمات

WANTT 
!>          WANTT هو LOGICAL
!>             WANTT = .true. إذا كان عامل Schur المثلثي
!>             يتم حسابه.  WANTT يتم تعيينه إلى .false. خلاف ذلك.
!>

WANTZ

!>          WANTZ هو LOGICAL
!>             WANTZ = .true. إذا كان عامل Schur الوحدوي يتم
!>             حسابه.  WANTZ يتم تعيينه إلى .false. خلاف ذلك.
!>

KACC22

!>          KACC22 هو INTEGER بقيمة 0، 1، أو 2.
!>             يحدد وضع حساب التحديثات المتعامدة البعيدة عن القطر.
!>        = 0: ZLAQR5 لا يجمع الانعكاسات ولا يستخدم
!>             ضرب المصفوفات لتحديث مدخلات المصفوفة البعيدة عن القطر.
!>        = 1: ZLAQR5 يجمع الانعكاسات ويستخدم ضرب المصفوفات
!>             لتحديث مدخلات المصفوفة البعيدة عن القطر.
!>        = 2: نفس KACC22 = 1. كان هذا الخيار يستخدم للاستفادة
!>             من الهيكل 2x2 أثناء عمليات الضرب على المصفوفات، ولكن
!>             لم يعد هذا مدعومًا.
!>

N

!>          N هو INTEGER
!>             N هو ترتيب المصفوفة الهيسنبرغية H التي تعمل عليها هذه
!>             الوحدة الفرعية.
!>

KTOP

!>          KTOP هو INTEGER
!>

KBOT

!>          KBOT هو INTEGER
!>             هذه هي الصفوف والأعمدة الأولى والأخيرة من كتلة قطرية
!>             معزولة عليها سيتم تطبيق دورة QR. يفترض بدون تحقق أن
!>                       إما KTOP = 1  أو   H(KTOP,KTOP-1) = 0
!>             و
!>                       إما KBOT = N  أو   H(KBOT+1,KBOT) = 0.
!>

NSHFTS

!>          NSHFTS هو INTEGER
!>             NSHFTS يعطي عدد الإزاحات المتزامنة. يجب أن يكون NSHFTS
!>             موجبًا وزوجيًا.
!>

S

!>          S هو مصفوفة COMPLEX*16، البعد (NSHFTS)
!>             تحتوي S على إزاحات الأصل التي تحدد دورة QR متعددة الإزاحات.
!>             قد يتم إعادة ترتيب S عند الإخراج.
!>

H

!>          H هي مصفوفة COMPLEX*16، البعد (LDH,N)
!>             عند الإدخال، تحتوي H على مصفوفة هيسنبرغية. عند الإخراج،
!>             يتم تطبيق دورة QR متعددة الإزاحات مع إزاحات SR(J)+i*SI(J)
!>             على الكتلة القطرية المعزولة في الصفوف والأعمدة من KTOP
!>             إلى KBOT.
!>

LDH

!>          LDH هو INTEGER
!>             LDH هو البعد الرائد للمصفوفة H كما تم تعريفه في
!>             الإجراء المستدعي. LDH >= MAX(1,N).
!>

ILOZ

!>          ILOZ هو INTEGER
!>

IHIZ

!>          IHIZ هو INTEGER
!>             تحديد الصفوف من Z التي يجب تطبيق التحويلات عليها
!>             إذا كانت WANTZ = .TRUE.. 1 <= ILOZ <= IHIZ <= N
!>

Z

!>          Z هي مصفوفة COMPLEX*16، البعد (LDZ,IHIZ)
!>             إذا كانت WANTZ = .TRUE.، يتم تجميع التحويل التشابهي الوحدوي
!>             لدورة QR في Z(ILOZ:IHIZ,ILOZ:IHIZ) من اليمين.
!>             إذا كانت WANTZ = .FALSE.، فإن Z غير مستخدمة.
!>

LDZ

!>          LDZ هو INTEGER
!>             LDZ هو البعد الرائد لـ Z كما تم تعريفه في
!>             الإجراء المستدعي. LDZ >= N.
!>

V

!>          V هي مصفوفة COMPLEX*16، البعد (LDV,NSHFTS/2)
!>

LDV

!>          LDV هو INTEGER
!>             LDV هو البعد الرائد لـ V كما تم تعريفه في
!>             الإجراء المستدعي. LDV >= 3.
!>

U

!>          U هي مصفوفة COMPLEX*16، البعد (LDU,2*NSHFTS)
!>

LDU

!>          LDU هو INTEGER
!>             LDU هو البعد الرائد لـ U كما تم تعريفه في
!>             الإجراء المستدعي. LDU >= 2*NSHFTS.
!>

NV

!>          NV هو INTEGER
!>             NV هو عدد الصفوف في WV المتاحة لمساحة العمل.
!>             NV >= 1.
!>

WV

!>          WV هي مصفوفة COMPLEX*16، البعد (LDWV,2*NSHFTS)
!>

LDWV

!>          LDWV هو INTEGER
!>             LDWV هو البعد الرائد لـ WV كما تم تعريفه في
!>             الإجراء المستدعي. LDWV >= NV.
!>

NH

!>          NH هو INTEGER
!>             NH هو عدد الأعمدة في المصفوفة WH المتاحة
!>             لمساحة العمل. NH >= 1.
!>

WH

!>          WH هي مصفوفة COMPLEX*16، البعد (LDWH,NH)
!>

LDWH

!>          LDWH هو INTEGER
!>             البعد الرائد لـ WH كما تم تعريفه في
!>             الإجراء المستدعي. LDWH >= 2*NSHFTS.
!>
