subroutine zlamswlq (character side, character trans, integer m, integer n, integer k, integer mb, integer nb, complex*16, dimension( lda, * ) a, integer lda, complex*16, dimension( ldt, * ) t, integer ldt, complex*16, dimension( ldc, * ) c, integer ldc, complex*16, dimension( * ) work, integer lwork, integer info)

ZLAMSWLQ

الغرض:

!>
!>    ZLAMSWLQ يستبدل المصفوفة المعقدة C ذات الأبعاد M-by-N بـ
!>
!>
!>                    SIDE = 'L'     SIDE = 'R'
!>    TRANS = 'N':      Q * C          C * Q
!>    TRANS = 'C':      Q**H * C       C * Q**H
!>    حيث Q هي مصفوفة موحدة معقدة معرفة كحاصل ضرب لمصفوفات عاكسة أولية
!>    محسوبة بواسطة تحليل LQ قصير واسع (ZLASWLQ)
!>

المعلمات:

SIDE 
!>          SIDE هو CHARACTER*1
!>          = 'L': تطبيق Q أو Q**H من اليسار؛
!>          = 'R': تطبيق Q أو Q**H من اليمين.
!>

TRANS

!>          TRANS هو CHARACTER*1
!>          = 'N':  عدم النقل، تطبيق Q؛
!>          = 'C':  النقل المرافق، تطبيق Q**H.
!>

M

!>          M هو INTEGER
!>          عدد صفوف المصفوفة C.  M >=0.
!>

N

!>          N هو INTEGER
!>          عدد أعمدة المصفوفة C. N >= 0.
!>

K

!>          K هو INTEGER
!>          عدد مصفوفات العاكسات الأولية التي تعرف حاصل الضرب
!>          للمصفوفة Q.
!>          M >= K >= 0;
!>
!>

MB

!>          MB هو INTEGER
!>          حجم كتلة الصفوف التي ستستخدم في LQ المكتل.
!>          M >= MB >= 1
!>

NB

!>          NB هو INTEGER
!>          حجم كتلة الأعمدة التي ستستخدم في LQ المكتل.
!>          NB > M.
!>

A

!>          A هو COMPLEX*16 array, dimension
!>                               (LDA,M) if SIDE = 'L',
!>                               (LDA,N) if SIDE = 'R'
!>          يجب أن يحتوي الصف i على المتجه الذي يعرف مصفوفة العاكس
!>          الأولية المكتلة H(i)، لـ i = 1,2,...,k، كما تم إرجاعه بواسطة
!>          ZLASWLQ في أول k صفوف من مصفوفة الوسيط A.
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرائد للمصفوفة A. LDA >= MAX(1,K).
!>

T

!>          T هو COMPLEX*16 array, dimension
!>          ( M * Number of blocks(CEIL(N-K/NB-K)),
!>          مصفوفات العاكسات القطاعية المثلثية العليا المكتلة مخزنة بشكل مدمج
!>          كسلسلة من القطاعات المثلثية العليا. انظر أدناه
!>          لمزيد من التفاصيل.
!>

LDT

!>          LDT هو INTEGER
!>          البعد الرائد للمصفوفة T.  LDT >= MB.
!>

C

!>          C هو COMPLEX*16 array, dimension (LDC,N)
!>          عند الدخول، المصفوفة C ذات الأبعاد M-by-N.
!>          عند الخروج، يتم استبدال C بـ Q*C أو Q**H*C أو C*Q**H أو C*Q.
!>

LDC

!>          LDC هو INTEGER
!>          البعد الرائد للمصفوفة C. LDC >= max(1,M).
!>

WORK

!>          (مساحة عمل) COMPLEX*16 array, dimension (MAX(1,LWORK))
!>          عند الخروج، إذا كان INFO = 0، ترجع WORK(1) الحد الأدنى لـ LWORK.
!>

LWORK

!>          LWORK هو INTEGER
!>          بعدد مصفوفة WORK.
!>          إذا كان MIN(M,N,K) = 0، فإن LWORK >= 1.
!>          إذا كان SIDE = 'L'، فإن LWORK >= max(1,NB*MB).
!>          إذا كان SIDE = 'R'، فإن LWORK >= max(1,M*MB).
!>
!>          إذا كان LWORK = -1، فسيتم افتراض استعلام عن مساحة العمل؛ تقوم الروتين
!>          بحساب الحد الأدنى لحجم مصفوفة WORK فقط، وترجع
!>          هذه القيمة كالإدخال الأول لمصفوفة WORK، ولا يتم إصدار أي رسالة خطأ
!>          متعلقة بـ LWORK بواسطة XERBLA.
!>

INFO

!>          INFO هو INTEGER
!>          = 0:  الخروج الناجح
!>          < 0:  إذا كان INFO = -i، فإن المعلمة i كانت قيمة غير قانونية
!>

تفاصيل إضافية:

!> يقوم تحليل LQ قصير واسع (SWLQ) بإجراء تحليل LQ بواسطة سلسلة من التحويلات الوحدوية،
!> ويمثل Q كحاصل ضرب لمصفوفات وحدوية أخرى
!>   Q = Q(1) * Q(2) * . . . * Q(k)
!> حيث تقوم كل Q(i) بتصفير الإدخالات القطرية العليا لكتلة من صفوف NB من A:
!>   Q(1) تقوم بتصفير الإدخالات القطرية العليا للصفوف 1:NB من A
!>   Q(2) تقوم بتصفير الصفوف السفلية MB-N من الصفوف [1:M,NB+1:2*NB-M] من A
!>   Q(3) تقوم بتصفير الصفوف السفلية MB-N من الصفوف [1:M,2*NB-M+1:3*NB-2*M] من A
!>   . . .
!>
!> يتم حساب Q(1) بواسطة GELQT، الذي يمثل Q(1) بواسطة متجهات هاوسهولدر
!> مخزنة تحت القطر في الصفوف 1:MB من A، وبواسطة مصفوفات العاكسات القطاعية
!> المثلثية العليا، مخزنة في المصفوفة T(1:LDT,1:N).
!> لمزيد من المعلومات، انظر التفاصيل الإضافية في GELQT.
!>
!> يتم حساب Q(i) لـ i>1 بواسطة TPLQT، الذي يمثل Q(i) بواسطة متجهات هاوسهولدر
!> مخزنة في الأعمدة [(i-1)*(NB-M)+M+1:i*(NB-M)+M] من A، وبواسطة مصفوفات العاكسات
!> القطاعية المثلثية العليا، مخزنة في المصفوفة T(1:LDT,(i-1)*M+1:i*M).
!> قد تستخدم Q(k) الأخيرة عددًا أقل من الصفوف.
!> لمزيد من المعلومات، انظر التفاصيل الإضافية في TPLQT.
!>
!> لمزيد من التفاصيل حول الخوارزمية الشاملة، انظر وصف
!> TSQR المتسلسل في القسم 2.2 من [1].
!>
!> [1] “Communication-Optimal Parallel and Sequential QR and LU Factorizations,”
!>     J. Demmel, L. Grigori, M. Hoemmen, J. Langou,
!>     SIAM J. Sci. Comput, vol. 34, no. 1, 2012
!>
