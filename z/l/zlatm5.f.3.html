subroutine zlatm5 (integer prtype, integer m, integer n, complex*16, dimension( lda, * ) a, integer lda, complex*16, dimension( ldb, * ) b, integer ldb, complex*16, dimension( ldc, * ) c, integer ldc, complex*16, dimension( ldd, * ) d, integer ldd, complex*16, dimension( lde, * ) e, integer lde, complex*16, dimension( ldf, * ) f, integer ldf, complex*16, dimension( ldr, * ) r, integer ldr, complex*16, dimension( ldl, * ) l, integer ldl, double precision alpha, integer qblcka, integer qblckb)

ZLATM5

الغرض:

!>
!> ZLATM5 يولد المصفوفات المتضمنة في معادلة سيلفستر المعممة:
!>
!>     A * R - L * B = C
!>     D * R - L * E = F
!>
!> وهي تحقق أيضًا (شرط التعظيم)
!>
!>  [ I -L ] ( [ A  -C ], [ D -F ] ) [ I  R ] = ( [ A    ], [ D    ] )
!>  [    I ] ( [     B ]  [    E ] ) [    I ]   ( [    B ]  [    E ] )
!>
!>

المعلمات

PRTYPE 
!>          PRTYPE هو عدد صحيح
!>           إلى نوع معين من المصفوفات لتوليدها
!>          (انظر مزيدًا من التفاصيل).
!>

M

!>          M هو عدد صحيح
!>          يحدد ترتيب A و D وعدد الصفوف في
!>          C و F و R و L.
!>

N

!>          N هو عدد صحيح
!>          يحدد ترتيب B و E وعدد الأعمدة في
!>          C و F و R و L.
!>

A

!>          A هي مصفوفة COMPLEX*16، البعد (LDA, M).
!>          عند الخروج A بحجم M-by-M يتم تهيئتها وفقًا لـ PRTYPE.
!>

LDA

!>          LDA هو عدد صحيح
!>          البعد الرئيسي لـ A.
!>

B

!>          B هي مصفوفة COMPLEX*16، البعد (LDB, N).
!>          عند الخروج B بحجم N-by-N يتم تهيئتها وفقًا لـ PRTYPE.
!>

LDB

!>          LDB هو عدد صحيح
!>          البعد الرئيسي لـ B.
!>

C

!>          C هي مصفوفة COMPLEX*16، البعد (LDC, N).
!>          عند الخروج C بحجم M-by-N يتم تهيئتها وفقًا لـ PRTYPE.
!>

LDC

!>          LDC هو عدد صحيح
!>          البعد الرئيسي لـ C.
!>

D

!>          D هي مصفوفة COMPLEX*16، البعد (LDD, M).
!>          عند الخروج D بحجم M-by-M يتم تهيئتها وفقًا لـ PRTYPE.
!>

LDD

!>          LDD هو عدد صحيح
!>          البعد الرئيسي لـ D.
!>

E

!>          E هي مصفوفة COMPLEX*16، البعد (LDE, N).

تفاصيل إضافية:

!>
!>  PRTYPE = 1: A و B هي كتل جوردان، D و E هي مصفوفات الوحدة
!>
!>             A : إذا كان (i == j) فإن A(i, j) = 1.0
!>                 إذا كان (j == i + 1) فإن A(i, j) = -1.0
!>                 else A(i, j) = 0.0,            i, j = 1...M
!>
!>             B : إذا كان (i == j) فإن B(i, j) = 1.0 - ALPHA
!>                 إذا كان (j == i + 1) فإن B(i, j) = 1.0
!>                 else B(i, j) = 0.0,            i, j = 1...N
!>
!>             D : إذا كان (i == j) فإن D(i, j) = 1.0
!>                 else D(i, j) = 0.0,            i, j = 1...M
!>
!>             E : إذا كان (i == j) فإن E(i, j) = 1.0
!>                 else E(i, j) = 0.0,            i, j = 1...N
!>
!>             L =  R يتم اختيارهما من [-10...10]،
!>                  مما يحدد الطرف الأيمن (C، F).
!>
!>  PRTYPE = 2 أو 3: مثلثية و/أو شبه مثلثية.
!>
!>             A : إذا كان (i <= j) فإن A(i, j) = [-1...1]
!>                 else A(i, j) = 0.0,             i, j = 1...M
!>
!>                 إذا كان (PRTYPE = 3) فإن
!>                    A(k + 1, k + 1) = A(k, k)
!>                    A(k + 1, k) = [-1...1]
!>                    sign(A(k, k + 1) = -(sin(A(k + 1, k))
!>                        k = 1, M - 1, QBLCKA
!>
!>             B : إذا كان (i <= j) فإن B(i, j) = [-1...1]
!>                 else B(i, j) = 0.0,            i, j = 1...N
!>
!>                 إذا كان (PRTYPE = 3) فإن
!>                    B(k + 1, k + 1) = B(k, k)
!>                    B(k + 1, k) = [-1...1]
!>                    sign(B(k, k + 1) = -(sign(B(k + 1, k))
!>                        k = 1, N - 1, QBLCKB
!>
!>             D : إذا كان (i <= j) فإن D(i, j) = [-1...1].
!>                 else D(i, j) = 0.0,            i, j = 1...M
!>
!>
!>             E : إذا كان (i <= j) فإن D(i, j) = [-1...1]
!>                 else E(i, j) = 0.0,            i, j = 1...N
!>
!>                 L، R يتم اختيارهما من [-10...10]،
!>                 مما يحدد الطرف الأيمن (C، F).
!>
!>  PRTYPE = 4 كامل
!>             A(i, j) = [-10...10]
!>             D(i, j) = [-1...1]    i,j = 1...M
!>             B(i, j) = [-10...10]
!>             E(i, j) = [-1...1]    i,j = 1...N
!>             R(i, j) = [-10...10]
!>             L(i, j) = [-1...1]    i = 1..M ,j = 1...N
!>
!>             L، R يحددان الطرف الأيمن (C، F).
!>
!>  PRTYPE = 5 حالة خاصة مع قيم ذاتية مشتركة و/أو قريبة.
!>
