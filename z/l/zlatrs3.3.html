subroutine zlatrs3 (character uplo, character trans, character diag, character normin, integer n, integer nrhs, complex*16, dimension( lda, * ) a, integer lda, complex*16, dimension( ldx, * ) x, integer ldx, double precision, dimension( * ) scale, double precision, dimension( * ) cnorm, double precision, dimension( * ) work, integer lwork, integer info)

ZLATRS3 تحل نظامًا مثلثيًا للمعادلات مع ضبط معاملات القياس لمنع تجاوز السعة.

الغرض:

!>
!> ZLATRS3 يحل أحد الأنظمة المثلثية
!>
!>    A * X = B * diag(scale),  A**T * X = B * diag(scale), or
!>    A**H * X = B * diag(scale)
!>
!> مع القياس لمنع تجاوز السعة. هنا A هي مصفوفة علوية أو سفلية مثلثية، A**T تمثل منقول A، A**H تمثل المترافق المنقول لـ A. X و B هما مصفوفتان n-by-nrhs و scale هو متجه من معاملات القياس بطول nrhs. معامل القياس scale(j) يكون عادة أقل من أو يساوي 1، ويتم اختياره بحيث يكون X(:,j) أقل من حد تجاوز السعة. إذا كانت المصفوفة A مفردة (A(j,j) = 0 لبعض j)، يتم إرجاع حل غير تافه لـ A*X = 0. إذا كان النظام مقاسًا بشكل سيء لدرجة أنه لا يمكن تمثيل الحل بـ (1/scale(k))*X(:,k)، فإن x(:,k) = 0 ويتم إرجاع scale(k).
!>
!> هذه نسخة BLAS-3 من LATRS لحل عدة جوانب يمنى في وقت واحد.
!>
!>

المعلمات

UPLO 
!>          UPLO is CHARACTER*1
!>          يحدد ما إذا كانت المصفوفة A علوية أم سفلية مثلثية.
!>          = 'U':  علوية مثلثية
!>          = 'L':  سفلية مثلثية
!>

TRANS

!>          TRANS is CHARACTER*1
!>          يحدد العملية المطبقة على A.
!>          = 'N':  حل A * x = s*b  (بدون تبديل)
!>          = 'T':  حل A**T* x = s*b  (تبديل)
!>          = 'C':  حل A**T* x = s*b  (تبديل مرافق)
!>

DIAG

!>          DIAG is CHARACTER*1
!>          يحدد ما إذا كانت المصفوفة A وحدات مثلثية أم لا.
!>          = 'N':  غير وحدات مثلثية
!>          = 'U':  وحدات مثلثية
!>

NORMIN

!>          NORMIN is CHARACTER*1
!>          يحدد ما إذا كان CNORM قد تم تعيينه أم لا.
!>          = 'Y':  CNORM يحتوي على معيار الأعمدة عند الدخول
!>          = 'N':  لم يتم تعيين CNORM عند الدخول. عند الخروج، سيتم
!>                  حساب المعايير وتخزينها في CNORM.
!>

N

!>          N is INTEGER
!>          رتبة المصفوفة A.  N >= 0.
!>

NRHS

!>          NRHS is INTEGER
!>          عدد أعمدة X.  NRHS >= 0.
!>

A

!>          A is COMPLEX*16 array, dimension (LDA,N)
!>          المصفوفة المثلثية A.  إذا كان UPLO = 'U'، فإن الجزء العلوي
!>          المثلثي العلوي بحجم n × n في المصفوفة A يحتوي على المصفوفة
!>          المثلثية العلوية، ولا يتم الرجوع إلى الجزء السفلي المثلثي
!>          الصارم لـ A.  إذا كان UPLO = 'L'، فإن الجزء السفلي المثلثي
!>          السفلي بحجم n × n في المصفوفة A يحتوي على المصفوفة المثلثية
!>          السفلية، ولا يتم الرجوع إلى الجزء العلوي المثلثي الصارم لـ A.
!>          إذا كان DIAG = 'U'، فلن يتم الرجوع إلى العناصر القطرية لـ A
!>          ويفترض أنها 1.
!>

LDA

!>          LDA is INTEGER
!>          البعد الرائد للمصفوفة A.  LDA >= max (1,N).
!>

X

!>          X is COMPLEX*16 array, dimension (LDX,NRHS)
!>          عند الدخول، الجانب الأيمن B للنظام المثلثي.
!>          عند الخروج، يتم الكتابة فوق X بالمصفوفة الحل X.
!>

LDX

!>          LDX is INTEGER
!>          البعد الرائد للمصفوفة X.  LDX >= max (1,N).
!>

SCALE

!>          SCALE is DOUBLE PRECISION array, dimension (NRHS)
!>          معامل القياس s(k) هو للنظام المثلثي
!>          A * x(:,k) = s(k)*b(:,k)  or  A**T* x(:,k) = s(k)*b(:,k).
!>          إذا كان SCALE = 0، فإن المصفوفة A مفردة أو مقاسة بشكل سيء.
!>          إذا تم العثور على A(j,j) = 0، يتم إرجاع متجه غير تافه x(:,k)
!>          يكون حلاً دقيقًا أو تقريبيًا لـ A*x(:,k) = 0. إذا كان النظام
!>          مقاسًا بشكل سيء لدرجة أن الحل لا يمكن تقديمه كـ x(:,k) * 1/s(k)،
!>          فإنه يتم إرجاع x(:,k) = 0.
!>

CNORM

!>          CNORM is DOUBLE PRECISION array, dimension (N)
!>
!>          إذا كان NORMIN = 'Y'، فإن CNORM هو وسيط إدخال و CNORM(j)
!>          يحتوي على معيار الجزء غير القطري للعمود j
!>          من A.  إذا كان TRANS = 'N'، يجب أن يكون CNORM(j) أكبر من أو
!>          يساوي المعيار اللانهائي، وإذا كان TRANS = 'T' أو 'C'، يجب أن
!>          يكون CNORM(j) أكبر من أو يساوي المعيار 1.
!>
!>          إذا كان NORMIN = 'N'، فإن CNORM هو وسيط إخراج و CNORM(j)
!>          يرجع المعيار 1 للجزء غير القطري للعمود j
!>          من A.
!>

WORK

!>          WORK is DOUBLE PRECISION array, dimension (LWORK).
!>          عند الخروج، إذا كان INFO = 0، فإن WORK(1) يرجع الحجم الأمثل
!>          لـ WORK.
!>

LWORK

!>          LWORK is INTEGER
!>          بعد المصفوفة WORK.
!>
!>          إذا كان MIN(N,NRHS) = 0، LWORK >= 1، وإلا
!>          LWORK >= MAX(1, 2*NBA * MAX(NBA, MIN(NRHS, 32)))، حيث
!>          NBA = (N + NB - 1)/NB و NB هو حجم الكتلة الأمثل.
!>
!>          إذا كان LWORK = -1، يفترض استعلام عن مساحة العمل؛ الروتين
!>          يحسب فقط الأبعاد المثلى لمصفوفة WORK، ويرجع
!>          هذه القيمة كإدخال أول لمصفوفة WORK، ولا يصدر رسالة خطأ
!>          متعلقة بـ LWORK بواسطة XERBLA.
!>

INFO

!>          INFO is INTEGER
!>          = 0:  الخروج بنجاح
!>          < 0:  إذا كان INFO = -k، فإن الوسيط k كان له قيمة غير قانونية
!>
