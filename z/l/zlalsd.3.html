subroutine zlalsd (character uplo, integer smlsiz, integer n, integer nrhs, double precision, dimension( * ) d, double precision, dimension( * ) e, complex*16, dimension( ldb, * ) b, integer ldb, double precision rcond, integer rank, complex*16, dimension( * ) work, double precision, dimension( * ) rwork, integer, dimension( * ) iwork, integer info)

ZLALSD يستخدم التحلل القيم المنفردة للمصفوفه A لحل مشكلة المربعات الصغرى.

الغرض:

!>
!> ZLALSD يستخدم التحلل القيم المنفردة للمصفوفه A لحل مشكلة المربعات الصغرى
!> لإيجاد X التي تقلل معيار اقليدس لكل عمود من AX-B ، حيث A هي مصفوفه علوية ثنائية قطرية بحجم N*N ، و X و B
!> هما بحجم N*NRHS. الحل X يحل محل B.
!>
!> القيم المنفردة للمصفوفه A التي تكون أصغر من RCOND مضروبه في أكبر قيمة منفردة
!> تعتبر صفرًا عند حل مشكلة المربعات الصغرى ؛ في هذه الحالة يتم إرجاع حل الحد الأدنى للمعايير.
!> القيم المنفردة الفعلية يتم إرجاعها في D بترتيب تصاعدي.
!>
!>

المعلمات

UPLO 
!>          UPLO is CHARACTER*1
!>         = 'U': D و E تحدد مصفوفة ثنائية قطرية علوية.
!>         = 'L': D و E تحدد مصفوفة ثنائية قطرية سفلية.
!>

SMLSIZ

!>          SMLSIZ is INTEGER
!>         الحد الأقصى لحجم المشاكل الفرعية في أسفل شجرة الحساب.
!>

N

!>          N is INTEGER
!>         بعد المصفوفة الثنائية القطرية. N >= 0.
!>

NRHS

!>          NRHS is INTEGER
!>         عدد أعمدة B. يجب أن يكون NRHS على الأقل 1.
!>

D

!>          D is DOUBLE PRECISION array, dimension (N)
!>         عند الإدخال، يحتوي D على القطر الرئيسي للمصفوفة الثنائية القطرية.
!>         عند الخروج، إذا كان INFO = 0، يحتوي D على قيمه المنفردة.
!>

E

!>          E is DOUBLE PRECISION array, dimension (N-1)
!>         يحتوي على مدخلات القطر العلوي للمصفوفة الثنائية القطرية.
!>         عند الخروج، تم تدمير E.
!>

B

!>          B is COMPLEX*16 array, dimension (LDB,NRHS)
!>         عند الإدخال، يحتوي B على الأطراف اليمنى لمشكلة المربعات الصغرى.
!>         عند الخروج، يحتوي B على الحل X.
!>

LDB

!>          LDB is INTEGER
!>         البعد الرئيسي لـ B في البرنامج الفرعي المستدعي.
!>         يجب أن يكون LDB على الأقل max(1,N).
!>

RCOND

!>          RCOND is DOUBLE PRECISION
!>         القيم المنفردة للمصفوفه A التي تقل أو تساوي RCOND مضروبة في
!>         أكبر قيمة منفردة يتم اعتبارها صفرًا عند حل مشكلة المربعات الصغرى.
!>         إذا كانت RCOND سالبة،
!>         يتم استخدام الدقة الآلية بدلاً من ذلك.
!>         على سبيل المثال، إذا كان diag(S)*X=B هو مشكلة المربعات الصغرى،
!>         حيث diag(S) هي مصفوفة قطرية للقيم المنفردة،
!>         فإن الحل سيكون X(i) = B(i) / S(i) إذا كانت S(i) أكبر من
!>         RCOND*max(S)، و X(i) = 0 إذا كانت S(i) أقل أو تساوي
!>         RCOND*max(S).
!>

RANK

!>          RANK is INTEGER
!>         عدد القيم المنفردة للمصفوفه A التي تكون أكبر من RCOND مضروبة في
!>         أكبر قيمة منفردة.
!>

WORK

!>          WORK is COMPLEX*16 array, dimension (N * NRHS)
!>

RWORK

!>          RWORK is DOUBLE PRECISION array, dimension at least
!>         (9*N + 2*N*SMLSIZ + 8*N*NLVL + 3*SMLSIZ*NRHS +
!>         MAX( (SMLSIZ+1)**2, N*(1+NRHS) + 2*NRHS ),
!>         where
!>         NLVL = MAX( 0, INT( LOG_2( MIN( M,N )/(SMLSIZ+1) ) ) + 1 )
!>

IWORK

!>          IWORK is INTEGER array, dimension at least
!>         (3*N*NLVL + 11*N).
!>

INFO

!>          INFO is INTEGER
!>         = 0:  نجاح في الخروج.
!>         < 0:  إذا كان INFO = -i، فإن المعلمة i كانت ذات قيمة غير قانونية.
!>         > 0:  فشل الخوارزمية في حساب قيمة منفردة أثناء العمل على المصفوفة الفرعية
!>               الواقعة في الصفوف والأعمدة
!>               INFO/(N+1) حتى MOD(INFO,N+1).
!>
