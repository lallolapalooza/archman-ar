توثيق الوظيفة/الروتين الفرعي

subroutine zlatrz (integer m, integer n, integer l, complex*16, dimension( lda, * ) a, integer lda, complex*16, dimension( * ) tau, complex*16, dimension( * ) work)

ZLATRZ تقوم بتحليل مصفوفة شبه منحرفة علوية عن طريق تحويلات وحدوية.

الغرض:

!>
!> ZLATRZ تقوم بتحليل المصفوفة شبه المنحرفة العلوية المعقدة M-by-(M+L)
!> [ A1 A2 ] = [ A(1:M,1:M) A(1:M,N-L+1:N) ] إلى ( R  0 ) * Z بواسطة تحويلات
!> وحدوية، حيث Z هي مصفوفة وحدوية (M+L)-by-(M+L)، و R و A1 هي مصفوفات
!> مثلثة علوية M-by-M.
!>

المعاملات

M 
!>          M هو عدد صحيح (INTEGER)
!>          عدد صفوف المصفوفة A.  M >= 0.
!>

N

!>          N هو عدد صحيح (INTEGER)
!>          عدد أعمدة المصفوفة A.  N >= 0.
!>

L

!>          L هو عدد صحيح (INTEGER)
!>          عدد أعمدة المصفوفة A التي تحتوي على الجزء الهادف
!>          من متجهات هاوسهولدر. N-M >= L >= 0.
!>

A

!>          A هي مصفوفة معقدة*16 (COMPLEX*16)، بأبعاد (LDA,N)
!>          عند الإدخال، يجب أن يحتوي الجزء العلوي شبه المنحرف
!>          من المصفوفة A والذي هو M-by-N على المصفوفة المراد تحليلها.
!>          عند الخروج، يحتوي الجزء العلوي المثلثي M-by-M من A
!>          على المصفوفة المثلثية العلوية R، والعناصر من N-L+1 إلى
!>          N في الصفوف M الأولى من A، مع المصفوفة TAU، على المصفوفة
!>          الوحدوية Z كناتج لـ M عاكسات ابتدائية.
!>

LDA

!>          LDA هو عدد صحيح (INTEGER)
!>          البعد الرئيسي للمصفوفة A.  LDA >= max(1,M).
!>

TAU

!>          TAU هي مصفوفة معقدة*16 (COMPLEX*16)، بأبعاد (M)
!>          العوامل القياسية للعواكس الابتدائية.
!>

WORK

!>          WORK هي مصفوفة معقدة*16 (COMPLEX*16)، بأبعاد (M)
!>

تفاصيل إضافية:

!>
!>  يتم الحصول على التحليل بطريقة هاوسهولدر. مصفوفة التحويل k،
!>  Z( k )، التي تستخدم لإدخال أصفار في الصف ( m - k + 1 ) من A،
!>  معطاة بالشكل
!>
!>     Z( k ) = ( I     0   )،
!>              ( 0  T( k ) )
!>
!>  حيث
!>
!>     T( k ) = I - tau*u( k )*u( k )**H،   u( k ) = (   1    )،
!>                                                 (   0    )
!>                                                 ( z( k ) )
!>
!>  يتم اختيار tau و z( k ) لإفناء عناصر الصف k من A2.
!>
!>  يتم إرجاع العامل tau في العنصر k من TAU والمتجه
!>  u( k ) في الصف k من A2، بحيث تكون عناصر z( k )
!>  في a( k, l + 1 )، ...، a( k, n ). يتم إرجاع عناصر R
!>  في الجزء المثلثي العلوي من A1.
!>
!>  Z معطاة بواسطة
!>
!>     Z =  Z( 1 ) * Z( 2 ) * ... * Z( m ).
!>
