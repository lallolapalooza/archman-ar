subroutine zlaed7 (integer n, integer cutpnt, integer qsiz, integer tlvls, integer curlvl, integer curpbm, double precision, dimension( * ) d, complex*16, dimension( ldq, * ) q, integer ldq, double precision rho, integer, dimension( * ) indxq, double precision, dimension( * ) qstore, integer, dimension( * ) qptr, integer, dimension( * ) prmptr, integer, dimension( * ) perm, integer, dimension( * ) givptr, integer, dimension( 2, * ) givcol, double precision, dimension( 2, * ) givnum, complex*16, dimension( * ) work, double precision, dimension( * ) rwork, integer, dimension( * ) iwork, integer info)

ZLAED7 يستخدم بواسطة ZSTEDC. يحسب النظام المميز المحدث لمصفوفة قطرية بعد تعديلها بواسطة مصفوفة متناظرة ذات رتبة واحد. يستخدم عندما تكون المصفوفة الأصلية كثيفة.

الغرض:

!>
!> ZLAED7 يحسب النظام المميز المحدث لمصفوفة قطرية
!> بعد تعديلها بواسطة مصفوفة متناظرة ذات رتبة واحد. هذه
!> الروتين الفرعي يستخدم فقط لمشكلة القيم المميزة التي تتطلب جميع
!> القيم المميزة، وبشكل اختياري، المتجهات المميزة لمصفوفة هرميتية كثيفة أو
!> نطاقية تم اختزالها إلى شكل ثلاثي الأقطار.
!>
!>   T = Q(in) ( D(in) + RHO * Z*Z**H ) Q**H(in) = Q(out) * D(out) * Q**H(out)
!>
!>   حيث Z = Q**Hu، u هو متجه طوله N مع واحدات في العناصر
!>   CUTPNT و CUTPNT + 1 وأصفار في مكان آخر.
!>
!>    تخزن المتجهات المميزة للمصفوفة الأصلية في Q، والقيم المميزة
!>    في D. تتكون الخوارزمية من ثلاث مراحل:
!>
!>       المرحلة الأولى تتكون من تفكيك حجم المشكلة
!>       عندما تكون هناك قيم مميزة متعددة أو إذا كان هناك صفر في
!>       المتجه Z. لكل حدوث من هذا القبيل، يتم تقليل بعد مشكلة
!>       المعادلة العلمانية بمقدار واحد. يتم تنفيذ هذه المرحلة
!>       بواسطة الروتين الفرعي DLAED2.
!>
!>       المرحلة الثانية تتكون من حساب القيم المميزة
!>       المحدثة. يتم ذلك عن طريق إيجاد جذور المعادلة
!>       العلمانية عبر الروتين الفرعي DLAED4 (كما تم استدعاؤه بواسطة SLAED3).
!>       يقوم هذا الروتين الفرعي أيضًا بحساب المتجهات المميزة
!>       للمشكلة الحالية.
!>
!>       المرحلة النهائية تتكون من حساب المتجهات المميزة
!>       المحدثة مباشرة باستخدام القيم المميزة المحدثة. يتم ضرب
!>       المتجهات المميزة للمشكلة الحالية مع المتجهات المميزة
!>       من المشكلة الكلية.
!>

المعاملات

N 
!>          N هو عدد صحيح (INTEGER)
!>         بعد المصفوفة المتناظرة ثلاثية الأقطار. N >= 0.
!>

CUTPNT

!>          CUTPNT هو عدد صحيح (INTEGER)
!>         يحتوي على موقع آخر قيمة مميزة في المصفوفة الجزئية الرائدة.
!>         min(1,N) <= CUTPNT <= N.
!>

QSIZ

!>          QSIZ هو عدد صحيح (INTEGER)
!>         بعد المصفوفة الوحدوية المستخدمة لاختزال
!>         المصفوفة الكاملة إلى شكل ثلاثي الأقطار. QSIZ >= N.
!>

TLVLS

!>          TLVLS هو عدد صحيح (INTEGER)
!>         العدد الإجمالي لمستويات الدمج في شجرة التقسيم والتغلب
!>         الكلية.
!>

CURLVL

!>          CURLVL هو عدد صحيح (INTEGER)
!>         المستوى الحالي في الروتين الفرعي للدمج الكلي،
!>         0 <= curlvl <= tlvls.
!>

CURPBM

!>          CURPBM هو عدد صحيح (INTEGER)
!>         المشكلة الحالية في المستوى الحالي في الروتين الفرعي
!>         للدمج الكلي (العد من اليسار العلوي إلى اليمين السفلي).
!>

D

!>          D هو مصفوفة دقة مزدوجة (DOUBLE PRECISION)، البعد (N)
!>         عند الإدخال، القيم المميزة للمصفوفة المعدلة ذات الرتبة 1.
!>         عند الخروج، القيم المميزة للمصفوفة التي تم إصلاحها.
!>

Q

!>          Q هو مصفوفة مركبة مزدوجة الدقة (COMPLEX*16)، البعد (LDQ,N)
!>         عند الإدخال، المتجهات المميزة للمصفوفة المعدلة ذات الرتبة 1.
!>         عند الخروج، المتجهات المميزة للمصفوفة ثلاثية الأقطار المحدثة.
!>

LDQ

!>          LDQ هو عدد صحيح (INTEGER)
!>         البعد الرئيسي للمصفوفة Q. LDQ >= max(1,N).
!>

RHO

!>          RHO هو دقة مزدوجة (DOUBLE PRECISION)
!>         يحتوي على العنصر تحت القطري المستخدم لإنشاء تعديل الرتبة 1.
!>

INDXQ

!>          INDXQ هي مصفوفة أعداد صحيحة (INTEGER)، البعد (N)
!>         تحتوي هذه على التبديل الذي سيعيد دمج المشكلة الفرعية التي تم حلها
!>         للتو في ترتيبها الصحيح، أي D( INDXQ( I = 1, N ) ) ستكون بترتيب تصاعدي.
!>

IWORK

!>          IWORK هي مصفوفة أعداد صحيحة (INTEGER)، البعد (4*N)
!>

RWORK

!>          RWORK هي مصفوفة دقة مزدوجة (DOUBLE PRECISION)،
!>                                 البعد (3*N+2*QSIZ*N)
!>

WORK

!>          WORK هي مصفوفة مركبة مزدوجة الدقة (COMPLEX*16)، البعد (QSIZ*N)
!>

QSTORE

!>          QSTORE هي مصفوفة دقة مزدوجة (DOUBLE PRECISION)، البعد (N**2+1)
!>         تخزن المتجهات المميزة للمصفوفات الجزئية التي تم مواجهتها أثناء
!>         التقسيم والتغلب، مجمعة معًا. QPTR يشير إلى بداية المصفوفات الجزئية.
!>

QPTR

!>          QPTR هي مصفوفة أعداد صحيحة (INTEGER)، البعد (N+2)
!>         قائمة بمؤشرات تشير إلى بداية المصفوفات الجزئية المخزنة
!>         في QSTORE. يتم ترقيم المصفوفات الجزئية بدءًا من
!>         الزاوية السفلية اليسرى لشجرة التقسيم والتغلب، من اليسار إلى
!>         اليمين ومن الأسفل إلى الأعلى.
!>

PRMPTR

!>          PRMPTR هي مصفوفة أعداد صحيحة (INTEGER)، البعد (N lg N)
!>         تحتوي على قائمة بالمؤشرات التي تشير إلى مكان تخزين تبديل
!>         المستوى في PERM. يشير PRMPTR(i+1) - PRMPTR(i)
!>         إلى حجم التبديل وكذلك حجم المشكلة الكاملة غير المفككة.
!>

PERM

!>          PERM هي مصفوفة أعداد صحيحة (INTEGER)، البعد (N lg N)
!>         تحتوي على التبديلات (من التفكيك والفرز) التي سيتم
!>         تطبيقها على كل كتلة مميزة.
!>

GIVPTR

!>          GIVPTR هي مصفوفة أعداد صحيحة (INTEGER)، البعد (N lg N)
!>         تحتوي على قائمة بالمؤشرات التي تشير إلى مكان تخزين دورانات
!>         Givens للمستوى في GIVCOL. يشير GIVPTR(i+1) - GIVPTR(i)
!>         إلى عدد دورانات Givens.
!>

GIVCOL

!>          GIVCOL هي مصفوفة أعداد صحيحة (INTEGER)، البعد (2, N lg N)
!>         كل زوج من الأرقام يشير إلى زوج من الأعمدة للمشاركة
!>         في دوران Givens.
!>

GIVNUM

!>          GIVNUM هي مصفوفة دقة مزدوجة (DOUBLE PRECISION)، البعد (2, N lg N)
!>         كل رقم يشير إلى قيمة S التي سيتم استخدامها في
!>         دوران Givens المقابل.
!>

INFO

!>          INFO هو عدد صحيح (INTEGER)
!>          = 0:  الخروج بنجاح.
!>          < 0:  إذا كان INFO = -i، فإن الوسيط i كان له قيمة غير قانونية.
!>          > 0:  إذا كان INFO = 1، لم تتقارب قيمة مميزة.
!>
