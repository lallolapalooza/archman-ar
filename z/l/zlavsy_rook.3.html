subroutine zlavsy_rook (character uplo, character trans, character diag, integer n, integer nrhs, complex*16, dimension( lda, * ) a, integer lda, integer, dimension( * ) ipiv, complex*16, dimension( ldb, * ) b, integer ldb, integer info)

ZLAVSY_ROOK

الغرض:

!>
!> ZLAVSY_ROOK يقوم بإحدى عمليات المصفوفة والمتجه
!>    x := A*x  أو  x := A'*x,
!> حيث x متجه من N عنصر و A هو أحد عوامل
!> تحليل الكتلة U*D*U' أو L*D*L' الذي تم حسابه بواسطة ZSYTRF_ROOK.
!>
!> إذا كان TRANS = 'N'، يضرب بـ U  أو U * D  (أو L  أو L * D)
!> إذا كان TRANS = 'T'، يضرب بـ U' أو D * U' (أو L' أو D * L')
!>

المعلمات

UPLO 
!>          UPLO هو CHARACTER*1
!>          يحدد ما إذا كانت الكتلة المخزنة في A علوية أم سفلية
!>          مثلثة.
!>          = 'U':  مثلثية علوية
!>          = 'L':  مثلثية سفلية
!>

TRANS

!>          TRANS هو CHARACTER*1
!>          يحدد العملية المراد تنفيذها:
!>          = 'N':  x := A*x
!>          = 'T':  x := A'*x
!>

DIAG

!>          DIAG هو CHARACTER*1
!>          يحدد ما إذا كانت الكتل القطرية مصفوفات وحدة أم لا.
!>          إذا تم افتراض أن الكتل القطرية مصفوفات وحدة،
!>          فإن A = U أو A = L، وإلا فإن A = U*D أو A = L*D.
!>          = 'U':  يتم افتراض أن الكتل القطرية مصفوفات وحدة.
!>          = 'N':  يتم افتراض أن الكتل القطرية مصفوفات غير وحدة.
!>

N

!>          N هو INTEGER
!>          عدد الصفوف والأعمدة للمصفوفة A. N >= 0.
!>

NRHS

!>          NRHS هو INTEGER
!>          عدد الأطراف اليمنى، أي عدد المتجهات
!>          x المطلوب ضربها في A. NRHS >= 0.
!>

A

!>          A هو مصفوفة COMPLEX*16، البعد (LDA,N)
!>          المصفوفة القطرية الكتلة D والعوامل المستخدمة
!>          للحصول على الكتلة U أو L كما تم حسابها بواسطة ZSYTRF_ROOK.
!>          مخزنة كمصفوفة مثلثية ثنائية الأبعاد.
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرائد للمصفوفة A. LDA >= max(1,N).
!>

IPIV

!>          IPIV هو مصفوفة INTEGER، البعد (N)
!>          تفاصيل التبادلات وهيكل الكتلة لـ D،
!>          كما تم تحديده بواسطة ZSYTRF_ROOK.
!>
!>          إذا كان UPLO = 'U':
!>               إذا كان IPIV(k) > 0، فتم تبادل الصفوف والأعمدة k و IPIV(k)
!>               وكان D(k,k) كتلة قطرية 1*1.
!>               (إذا كان IPIV( k ) = k، لم يتم إجراء تبادل).
!>
!>               إذا كان IPIV(k) < 0 و IPIV(k-1) < 0، فتم تبادل الصفوف
!>               والأعمدة k و -IPIV(k) وتم تبادل الصفوف
!>               والأعمدة k-1 و -IPIV(k-1)،
!>               D(k-1:k,k-1:k) هي كتلة قطرية 2*2.
!>
!>          إذا كان UPLO = 'L':
!>               إذا كان IPIV(k) > 0، فتم تبادل الصفوف والأعمدة k و IPIV(k)
!>               وكان D(k,k) كتلة قطرية 1*1.
!>               (إذا كان IPIV( k ) = k، لم يتم إجراء تبادل).
!>
!>               إذا كان IPIV(k) < 0 و IPIV(k+1) < 0، فتم تبادل الصفوف
!>               والأعمدة k و -IPIV(k) وتم تبادل الصفوف
!>               والأعمدة k+1 و -IPIV(k+1)،
!>               D(k:k+1,k:k+1) هي كتلة قطرية 2*2.
!>

B

!>          B هي مصفوفة COMPLEX*16، البعد (LDB,NRHS)
!>          عند الدخول، تحتوي B على NRHS متجه بطول N.
!>          عند الخروج، يتم استبدال B بمنتج A * B.
!>

LDB

!>          LDB هو INTEGER
!>          البعد الرائد للمصفوفة B. LDB >= max(1,N).
!>

INFO

!>          INFO هو INTEGER
!>          = 0: خروج ناجح
!>          < 0: إذا كان INFO = -k، فإن المعامل k كان له قيمة غير قانونية
!>
