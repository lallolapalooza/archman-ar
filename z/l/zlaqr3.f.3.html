subroutine zlaqr3 (logical wantt, logical wantz, integer n, integer ktop, integer kbot, integer nw, complex*16, dimension( ldh, * ) h, integer ldh, integer iloz, integer ihiz, complex*16, dimension( ldz, * ) z, integer ldz, integer ns, integer nd, complex*16, dimension( * ) sh, complex*16, dimension( ldv, * ) v, integer ldv, integer nh, complex*16, dimension( ldt, * ) t, integer ldt, integer nv, complex*16, dimension( ldwv, * ) wv, integer ldwv, complex*16, dimension( * ) work, integer lwork)

ZLAQR3 تقوم بالتحويل المتشابه الوحدوي لمصفوفة هسنبرغ للكشف عن القيم الذاتية المترافقة بالكامل من مصفوفة جزئية رئيسية لاحقة (نزع الانحراف المبكر العدواني).

الغرض:

!>
!>    نزع الانحراف المبكر العدواني:
!>
!>    تقبل ZLAQR3 كمُدخل مصفوفة هسنبرغ علوية
!>    H وتقوم بتحويل متشابه وحدوي
!>    مصمم للكشف عن القيم الذاتية المترافقة بالكامل ونزعها من
!>    مصفوفة جزئية رئيسية لاحقة. عند الخروج، تم استبدال H بـ
!>    مصفوفة هسنبرغ جديدة هي اضطراب لـ
!>    تحويل متشابه وحدوي لـ H. يُؤمل أن يكون الإصدار النهائي من H به
!>    العديد من الإدخالات الفرعية قطرية صفرية.
!>
!>

المعلمات

WANTT 
!>          WANTT هو LOGICAL
!>          إذا كانت .TRUE.، فسيتم تحديث مصفوفة هسنبرغ H بالكامل
!>          بحيث يمكن حساب العامل الثلاثي لشور (بالتعاون مع الروتين الفرعي المستدعي).
!>          إذا كانت .FALSE.، فلن يتم تحديث سوى ما يكفي من H للحفاظ
!>          على القيم الذاتية.
!>

WANTZ

!>          WANTZ هو LOGICAL
!>          إذا كانت .TRUE.، فسيتم تحديث المصفوفة الوحدوية Z
!>          بحيث يمكن حساب العامل الثلاثي لشور (بالتعاون مع الروتين الفرعي المستدعي).
!>          إذا كانت .FALSE.، فلن تتم الإشارة إلى Z.
!>

N

!>          N هو INTEGER
!>          ترتيب المصفوفة H، وترتيب المصفوفة الوحدوية Z (إذا كان WANTZ هو .TRUE.).
!>

KTOP

!>          KTOP هو INTEGER
!>          من المفترض أن KTOP = 1 أو H(KTOP,KTOP-1)=0.
!>          يحدد KBOT و KTOP معًا كتلة معزولة
!>          على طول قطر مصفوفة هسنبرغ.
!>

KBOT

!>          KBOT هو INTEGER
!>          من المفترض بدون فحص أن KBOT = N أو H(KBOT+1,KBOT)=0.
!>          يحدد KBOT و KTOP معًا كتلة معزولة
!>          على طول قطر مصفوفة هسنبرغ.
!>

NW

!>          NW هو INTEGER
!>          حجم نافذة نزع الانحراف. 1 <= NW <= (KBOT-KTOP+1).
!>

H

!>          H هو مصفوفة COMPLEX*16، البعد (LDH,N)
!>          عند الإدخال، يخزن القسم N-by-N الأولي من H
!>          مصفوفة هسنبرغ التي تخضع لنزع الانحراف المبكر العدواني.
!>          عند الخروج، تم تحويل H بواسطة تحويل متشابه وحدوي،
!>          تم اضطرابها، وأعيدت
!>          إلى شكل هسنبرغ الذي (يؤمل أن يكون به) بعض
!>          الإدخالات الفرعية القطرية الصفرية.
!>

LDH

!>          LDH هو INTEGER
!>          البعد الرئيسي لـ H كما تم الإعلان عنه في الروتين الفرعي المستدعي.
!>          N <= LDH
!>

ILOZ

!>          ILOZ هو INTEGER
!>

IHIZ

!>          IHIZ هو INTEGER
!>          حدد صفوف Z التي يجب تطبيق التحويلات عليها إذا كان WANTZ هو .TRUE..
!>          1 <= ILOZ <= IHIZ <= N.
!>

Z

!>          Z هو مصفوفة COMPLEX*16، البعد (LDZ,N)
!>          إذا كان WANTZ هو .TRUE.، فعند الخروج، تم تجميع التحويل المتشابه الوحدوي المذكور أعلاه
!>          في Z(ILOZ:IHIZ,ILOZ:IHIZ) من اليمين.
!>          إذا كان WANTZ هو .FALSE.، فلن تتم الإشارة إلى Z.
!>

LDZ

!>          LDZ هو INTEGER
!>          البعد الرئيسي لـ Z كما تم الإعلان عنه في
!>          الروتين الفرعي المستدعي. 1 <= LDZ.
!>

NS

!>          NS هو INTEGER
!>          عدد القيم الذاتية غير المترافقة (أي التقريبية)
!>          التي تم إرجاعها في SR و SI والتي يمكن استخدامها كتحولات بواسطة
!>          الروتين الفرعي المستدعي.
!>

ND

!>          ND هو INTEGER
!>          عدد القيم الذاتية المترافقة التي اكتشفتها هذه
!>          الروتين الفرعي.
!>

SH

!>          SH هو مصفوفة COMPLEX*16، البعد (KBOT)
!>          عند الخروج، القيم الذاتية التقريبية التي يمكن
!>          استخدامها للتحولات مخزنة في SH(KBOT-ND-NS+1)
!>          حتى SR(KBOT-ND). القيم الذاتية المترافقة
!>          مخزنة في SH(KBOT-ND+1) حتى SH(KBOT).
!>

V

!>          V هو مصفوفة COMPLEX*16، البعد (LDV,NW)
!>          مصفوفة عمل NW-by-NW.
!>

LDV

!>          LDV هو INTEGER
!>          البعد الرئيسي لـ V كما تم الإعلان عنه في
!>          الروتين الفرعي المستدعي. NW <= LDV
!>

NH

!>          NH هو INTEGER
!>          عدد أعمدة T. NH >= NW.
!>

T

!>          T هو مصفوفة COMPLEX*16، البعد (LDT,NW)
!>

LDT

!>          LDT هو INTEGER
!>          البعد الرئيسي لـ T كما تم الإعلان عنه في
!>          الروتين الفرعي المستدعي. NW <= LDT
!>

NV

!>          NV هو INTEGER
!>          عدد صفوف مصفوفة العمل WV المتاحة لـ
!>          مساحة العمل. NV >= NW.
!>

WV

!>          WV هو مصفوفة COMPLEX*16، البعد (LDWV,NW)
!>

LDWV

!>          LDWV هو INTEGER
!>          البعد الرئيسي لـ W كما تم الإعلان عنه في
!>          الروتين الفرعي المستدعي. NW <= LDV
!>

WORK

!>          WORK هو مصفوفة COMPLEX*16، البعد (LWORK)
!>          عند الخروج، يتم تعيين WORK(1) إلى تقدير للقيمة المثلى
!>          لـ LWORK للقيم المعطاة لـ N، NW، KTOP، و KBOT.
!>

LWORK

!>          LWORK هو INTEGER
!>          بعد مصفوفة العمل WORK. LWORK = 2*NW
!>          كافٍ، ولكن قد ينتج عن قيم أكبر لـ LWORK كفاءة أكبر.
!>
!>          إذا كان LWORK = -1، فسيتم افتراض استعلام مساحة عمل؛ ZLAQR3
!>          يقدر فقط حجم مساحة العمل الأمثل للقيم المعطاة
!>          لـ N، NW، KTOP، و KBOT. يتم إرجاع التقدير
!>          في WORK(1). لا يتم إصدار رسالة خطأ متعلقة بـ LWORK
!>          بواسطة XERBLA. لا تتم الإشارة إلى H ولا Z.
!>
