subroutine zla_lin_berr (integer n, integer nz, integer nrhs, complex*16, dimension( n, nrhs ) res, double precision, dimension( n, nrhs ) ayb, double precision, dimension( nrhs ) berr)

ZLA_LIN_BERR تحسب الخطأ الخلفي النسبي المكون تلو الآخر.

الغرض:

!>
!>    ZLA_LIN_BERR تحسب الخطأ الخلفي النسبي المكون تلو الآخر
!>    من الصيغة
!>        max(i) ( abs(R(i)) / ( abs(op(A_s))*abs(Y) + abs(B_s) )(i) )
!>    حيث abs(Z) هي القيمة المطلقة المكونة تلو الأخرى للمصفوفة
!>    أو المتجه Z.
!>

المعلمات

N 
!>          N هو عدد صحيح (INTEGER)
!>     عدد المعادلات الخطية، أي، رتبة
!>     المصفوفة A.  N >= 0.
!>

NZ

!>          NZ هو عدد صحيح (INTEGER)
!>     نضيف (NZ+1)*SLAMCH( 'Safe minimum' ) إلى R(i) في البسط
!>     للحماية من الأخطاء المتبقية الصفرية بشكل خاطئ. القيمة الافتراضية هي N.
!>

NRHS

!>          NRHS هو عدد صحيح (INTEGER)
!>     عدد الأطراف اليمنى، أي، عدد أعمدة
!>     المصفوفات AYB، RES، و BERR.  NRHS >= 0.
!>

RES

!>          RES هي مصفوفة COMPLEX*16، بأبعاد (N,NRHS)
!>     مصفوفة الخطأ المتبقي، أي، المصفوفة R في صيغة الخطأ الخلفي النسبي
!>     أعلاه.
!>

AYB

!>          AYB هي مصفوفة DOUBLE PRECISION، بأبعاد (N, NRHS)
!>     المقام في صيغة الخطأ الخلفي النسبي أعلاه، أي،
!>     المصفوفة abs(op(A_s))*abs(Y) + abs(B_s). المصفوفات A، Y، و B
!>     تأتي من التنقيح التكراري (انظر zla_gerfsx_extended.f).
!>

BERR

!>          BERR هي مصفوفة DOUBLE PRECISION، بأبعاد (NRHS)
!>     الخطأ الخلفي النسبي المكون تلو الآخر من الصيغة أعلاه.
!>
