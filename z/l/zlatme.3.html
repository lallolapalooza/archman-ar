subroutine zlatme (integer n, character dist, integer, dimension( 4 ) iseed, complex*16, dimension( * ) d, integer mode, double precision cond, complex*16 dmax, character rsign, character upper, character sim, double precision, dimension( * ) ds, integer modes, double precision conds, integer kl, integer ku, double precision anorm, complex*16, dimension( lda, * ) a, integer lda, complex*16, dimension( * ) work, integer info)

الغرض:

!>
!>    ZLATME يولد مصفوفات مربعة عشوائية غير متماثلة ذات
!>    قيم ذاتية محددة لاختبار برامج LAPACK.
!>
!>    تعمل ZLATME عن طريق تطبيق تسلسل العمليات التالي:
!>
!>    1. تعيين القطر إلى D ، حيث يمكن إدخال D أو
!>         حسابها وفقًا للوضع (MODE)، الشرط (COND)، الأقصى (DMAX)، والإشارة (RSIGN)
!>         كما هو موضح أدناه.
!>
!>    2. إذا كان UPPER='T'، يتم تعيين المثلث العلوي من A إلى قيم عشوائية
!>         من التوزيع DIST.
!>
!>    3. إذا كان SIM='T'، يتم ضرب A من اليسار بمصفوفة عشوائية
!>         X، التي قيمها الذاتية محددة بواسطة DS، MODES، و
!>         CONDS، ومن اليمين بمعكوس X.
!>
!>    4. إذا كان KL < N-1، يتم تقليل النطاق السفلي إلى KL باستخدام
!>         تحويلات هاوسهولدر. إذا كان KU < N-1، يتم تقليل النطاق العلوي
!>         إلى KU.
!>
!>    5. إذا كان ANORM غير سالب، يتم قياس المصفوفة بحيث يكون لها
!>         أكبر قيمة قياسية للعنصر ANORM.
!>
!>    (ملاحظة: نظرًا لأنه لا يمكن تقليل المصفوفة إلى ما هو أبعد من الشكل الهيسنبرغي،
!>     لا تتوفر خيارات التعبئة.)
!>

المعلمات

N 
!>          N هو عدد صحيح (INTEGER).
!>           عدد الأعمدة (أو الصفوف) للمصفوفة A. لم يتم تعديلها.
!>

DIST

!>          DIST هو حرف (CHARACTER*1).
!>           عند الدخول، يحدد DIST نوع التوزيع المستخدم
!>           لتوليد القيم الذاتية/القيم الذاتية العشوائية، وعلى
!>           المثلث العلوي (انظر UPPER).
!>           'U' => منتظم (UNIFORM) (0, 1) ( 'U' لـ uniform )
!>           'S' => منتظم (UNIFORM) (-1, 1) ( 'S' لـ symmetric )
!>           'N' => طبيعي (NORMAL) (0, 1) ( 'N' لـ normal )
!>           'D' => منتظم على القرص المركب |z| < 1.
!>           لم يتم تعديلها.
!>

ISEED

!>          ISEED هو مصفوفة أعداد صحيحة (INTEGER array)، البعد ( 4 ).
!>           عند الدخول، يحدد ISEED بذرة مولد الأرقام العشوائية.
!>           يجب أن تكون الأرقام بين 0 و 4095 شاملة، ويجب أن يكون ISEED(4) فرديًا.
!>           يستخدم مولد الأرقام العشوائية تسلسلًا خطيًا متطابقًا محصورًا بأعداد صحيحة صغيرة،
!>           لذلك يجب أن ينتج أرقامًا عشوائية مستقلة عن الآلة.
!>           تتغير قيم ISEED عند الخروج، ويمكن استخدامها في الاستدعاء التالي لـ ZLATME
!>           لمواصلة نفس تسلسل الأرقام العشوائية.
!>           تتغير عند الخروج.
!>

D

!>          D هي مصفوفة أعداد مركبة *16 (COMPLEX*16 array)، البعد ( N ).
!>           تستخدم هذه المصفوفة لتحديد القيم الذاتية للمصفوفة A. إذا
!>           كان MODE=0، فمن المفترض أن تحتوي D على القيم الذاتية.
!>           بخلاف ذلك، سيتم حسابها وفقًا لـ MODE، COND،
!>           DMAX، و RSIGN ووضعها في D.
!>           تتغير إذا كان MODE غير صفري.
!>

MODE

!>          MODE هو عدد صحيح (INTEGER).
!>           عند الدخول، يصف هذا كيفية تحديد القيم الذاتية:
!>           MODE = 0 يعني استخدام D كمدخل.
!>           MODE = 1 يضبط D(1)=1 و D(2:N)=1.0/COND.
!>           MODE = 2 يضبط D(1:N-1)=1 و D(N)=1.0/COND.
!>           MODE = 3 يضبط D(I)=COND**(-(I-1)/(N-1)).
!>           MODE = 4 يضبط D(i)=1 - (i-1)/(N-1)*(1 - 1/COND).
!>           MODE = 5 يضبط D إلى أرقام عشوائية في النطاق
!>                    ( 1/COND , 1 ) بحيث تكون لوغاريتماتها
!>                    موزعة بشكل منتظم.
!>           MODE = 6 يضبط D إلى أرقام عشوائية من نفس التوزيع
!>                    مثل بقية المصفوفة.
!>           MODE < 0 له نفس المعنى مثل ABS(MODE)، باستثناء أن
!>              ترتيب عناصر D معكوس.
!>           لذلك، إذا كان MODE بين 1 و 4، فإن عناصر D تتراوح
!>              من 1 إلى 1/COND. إذا كان بين -1 و -4، فإن عناصر
!>              D تتراوح من 1/COND إلى 1.
!>           لم يتم تعديلها.
!>

COND

!>          COND هو رقم عشري مزدوج الدقة (DOUBLE PRECISION).
!>           عند الدخول، يستخدم كما هو موضح تحت MODE أعلاه.
!>           إذا تم استخدامه، يجب أن يكون >= 1. لم يتم تعديله.
!>

DMAX

!>          DMAX هو عدد مركب *16 (COMPLEX*16).
!>           إذا لم يكن MODE هو -6 أو 0 أو 6، فسيتم قياس محتويات D، كما
!>           تم حسابها وفقًا لـ MODE و COND، بواسطة
!>           DMAX / max(abs(D(i))). لاحظ أن DMAX لا يلزم أن يكون
!>           موجبًا أو حقيقيًا: إذا كان DMAX سالبًا أو مركبًا (أو صفرًا)،
!>           فسيتم قياس D بعدد سالب أو مركب (أو صفر).
!>           إذا كان RSIGN='F'، فسيكون أكبر قيمة ذاتية (من حيث القيمة المطلقة)
!>           مساوية لـ DMAX.
!>           لم يتم تعديلها.
!>

RSIGN

!>          RSIGN هو حرف (CHARACTER*1).
!>           إذا لم يكن MODE هو 0 أو 6 أو -6، و RSIGN='T'، فسيتم
!>           ضرب عناصر D، كما تم حسابها وفقًا لـ MODE و COND،
!>           برقم مركب عشوائي من دائرة الوحدة |z| = 1. إذا كان RSIGN='F'،
!>           فلن يتم ضربها. يمكن أن يحتوي RSIGN فقط على القيم 'T' أو 'F'.
!>           لم يتم تعديلها.
!>

UPPER

!>          UPPER هو حرف (CHARACTER*1).
!>           إذا كان UPPER='T'، فسيتم تعيين عناصر A فوق القطر
!>           إلى أرقام عشوائية من DIST. إذا كان UPPER='F'،
!>           فلن يتم ذلك. يمكن أن يحتوي UPPER فقط على القيم 'T' أو 'F'.
!>           لم يتم تعديلها.
!>

SIM

!>          SIM هو حرف (CHARACTER*1).
!>           إذا كان SIM='T'، فسيتم تشغيل A بواسطة ، أي ضربها من اليسار بمصفوفة X و
!>           من اليمين بمعكوس X. X = U S V، حيث U و V
!>           مصفوفتان وحدويتان عشوائيتان و S مصفوفة (قطرية)
!>           من القيم الذاتية المحددة بواسطة DS، MODES، و CONDS. إذا
!>           كان SIM='F'، فلن يتم تحويل A.
!>           لم يتم تعديلها.
!>

DS

!>          DS هي مصفوفة أرقام عشرية مزدوجة الدقة (DOUBLE PRECISION array)، البعد ( N ).
!>           تستخدم هذه المصفوفة لتحديد القيم الذاتية لـ X،
!>           بنفس الطريقة التي تحدد بها D القيم الذاتية لـ A.
!>           إذا كان MODE=0، تحتوي DS على القيم الذاتية، والتي
!>           قد لا تكون صفرًا.
!>           تتغير إذا كان MODE غير صفري.
!>

MODES

!>          MODES هو عدد صحيح (INTEGER).
!>

CONDS

!>          CONDS هو رقم عشري مزدوج الدقة (DOUBLE PRECISION).
!>           مشابه لـ MODE و COND، ولكن لتحديد قطر S.
!>           MODES=-6 و +6 غير مسموح بهما (لأنهما قد يؤديان إلى
!>           قيم ذاتية سيئة التكييف عشوائيًا).
!>

KL

!>          KL هو عدد صحيح (INTEGER).
!>           يحدد هذا النطاق السفلي للمصفوفة. KL=1
!>           يحدد الشكل الهيسنبرغي العلوي. إذا كان KL على الأقل N-1،
!>           فسيكون للمصفوفة A نطاق سفلي كامل.
!>           لم يتم تعديلها.
!>

KU

!>          KU هو عدد صحيح (INTEGER).
!>           يحدد هذا النطاق العلوي للمصفوفة. KU=1
!>           يحدد الشكل الهيسنبرغي السفلي. إذا كان KU على الأقل N-1،
!>           فسيكون للمصفوفة A نطاق علوي كامل؛ إذا كان KU و KL
!>           كلاهما عند

NAME

zlatme — The purpose of zlatme is to generate a complex double-precision matrix A which has a prescribed singular value decomposition (SVD), or a prescribed set of eigenvalues.

SYNOPSIS

 SUBROUTINE ZLATME(
  N, DIST, MODE, SEED, RATE, SYM, ASPECT,
  D, Z, ALPHA, BETA, ANORM, A, LDA, WORK, INFO
 )
 
 INTEGER N
 CHARACTER DIST, MODE, SEED, RATE, SYM, ASPECT
 DOUBLE PRECISION D( * ), Z( * ), ALPHA( * ), BETA( * ), ANORM
 COMPLEX*16 A( LDA, * ), WORK( * )
 INTEGER LDA, INFO

Function/Subroutine Documentation

N

!>          N is INTEGER
!>           The number of columns of A.
!>           Not modified.

DIST

!>          DIST is CHARACTER
!>           The distribution of the singular values or eigenvalues.
!>           DIST = 'U' => uniform distribution
!>           DIST = 'R' => random distribution
!>           DIST = 'S' => singular values specified by D
!>           Not modified.

MODE

!>          MODE is CHARACTER
!>           The condition of the matrix A.
!>           MODE = 'N' => normal random matrix
!>           MODE = 'S' => sparse random matrix
!>           MODE = 'H' => Householder matrix
!>           MODE = 'P' => permutation matrix
!>           MODE = 'L' => lower triangular matrix
!>           MODE = 'U' => upper triangular matrix
!>           Not modified.

SEED

!>          SEED is CHARACTER*1
!>           The seed for the random number generator.
!>           Not modified.

RATE

!>          RATE is CHARACTER*1
!>           The rate of acceptance/rejection.
!>           Not modified.

SYM

!>          SYM is CHARACTER*1
!>           The symmetry of the matrix.
!>           Not modified.

ASPECT

!>          ASPECT is DOUBLE PRECISION
!>           The aspect ratio of the matrix.
!>           Not modified.

D

!>          D is DOUBLE PRECISION array, dimension ( N )
!>           If DIST = 'S', then the singular values of A are specified by D.
!>           Not modified.

Z

!>          Z is DOUBLE PRECISION array, dimension ( N )
!>           Not modified.

ALPHA

!>          ALPHA is DOUBLE PRECISION array, dimension ( N )
!>           Not modified.

BETA

!>          BETA is DOUBLE PRECISION array, dimension ( N )
!>           Not modified.

ANORM

!>          ANORM is DOUBLE PRECISION
!>           If ANORM is not negative, then A will be scaled by a non-
!>           negative real number to make the maximum-element-norm of A
!>           to be ANORM.
!>           Not modified.

A

!>          A is COMPLEX*16 array, dimension ( LDA, N )
!>           On exit A is the desired test matrix.
!>           Modified.

LDA

!>          LDA is INTEGER
!>           LDA specifies the first dimension of A as declared in the
!>           calling program.  LDA must be at least M.
!>           Not modified.

WORK

!>          WORK is COMPLEX*16 array, dimension ( 3*N )
!>           Workspace.
!>           Modified.

INFO

!>          INFO is INTEGER
!>           Error code.  On exit, INFO will be set to one of the
!>           following values:
!>             0 => normal return
!>            -1 => N negative
!>            -2 => DIST illegal string
!>            -5 => MODE not in range -6 to 6
!>            -6 => COND less than 1.0, and MODE neither -6, 0 nor 6
!>            -9 => RSIGN is not 'T' or 'F'
!>           -10 => UPPER is not 'T' or 'F'
!>           -11 => SIM   is not 'T' or 'F'
!>           -12 => MODES=0 and DS has a zero singular value.
!>           -13 => MODES is not in the range -5 to 5.
!>           -14 => MODES is nonzero and CONDS is less than 1.
!>           -15 => KL is less than 1.
!>           -16 => KU is less than 1, or KL and KU are both less than
!>                  N-1.
!>           -19 => LDA is less than M.
!>            1  => Error return from ZLATM1 (computing D)
!>            2  => Cannot scale to DMAX (max. eigenvalue is 0)
!>            3  => Error return from DLATM1 (computing DS)
!>            4  => Error return from ZLARGE
!>            5  => Zero singular value from DLATM1.
