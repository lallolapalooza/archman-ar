recursive subroutine zlaunhr_col_getrfnp2 (integer m, integer n, complex*16, dimension( lda, * ) a, integer lda, complex*16, dimension( * ) d, integer info)

ZLAUNHR_COL_GETRFNP2

الغرض:

!>
!> ZLAUNHR_COL_GETRFNP2 يحسب عامل LU المعدل بدون
!> تبديل مصفوفة عامة معقدة M-by-N، A. العامل له الشكل:
!>
!>     A - S = L * U,
!>
!> حيث:
!>    S هي مصفوفة إشارة قطرية m-by-n مع القطر D، بحيث
!>    D(i) = S(i,i)، 1 <= i <= min(M,N). القطر D يُنشأ
!>    كـ D(i)=-SIGN(A(i,i))، حيث A(i,i) هي القيمة بعد إجراء
!>    خطوات الاستبعاد الغاوسي i-1. هذا يعني أن العنصر القطري
!>    في كل خطوة من خطوات الاستبعاد الغاوسي هو
!>    واحد على الأقل في القيمة المطلقة (بحيث لا يكون القسمة على صفر
!>    ممكنة أثناء القسمة على العنصر القطري)؛
!>
!>    L هي مصفوفة ثلاثية سفلى M-by-N مع عناصر قطرية وحدة
!>    (شبه منحرف سفلي إذا كان M > N)؛
!>
!>    و U هي مصفوفة ثلاثية عليا M-by-N
!>    (شبه منحرف علوي إذا كان M < N).
!>
!> هذا الروتين هو روتين مساعد يستخدم في روتين إعادة بناء هاوسهولدر
!> ZUNHR_COL. في ZUNHR_COL، يتم تطبيق هذا الروتين على مصفوفة A
!> M-by-N بأعمدة متعامدة، حيث يتم تحديد كل عنصر
!> بوحدة في القيمة المطلقة. مع اختيار
!> المصفوفة S أعلاه، يمكن إثبات أن العنصر القطري في كل
!> خطوة من خطوات الاستبعاد الغاوسي هو الأكبر (في القيمة المطلقة) في
!> العمود على أو تحت القطر، بحيث لا تكون هناك حاجة للتبديل
!> للاستقرار العددي [1].
!>
!> لمزيد من التفاصيل حول خوارزمية إعادة بناء هاوسهولدر،
!> بما في ذلك عامل LU المعدل، انظر [1].
!>
!> هذا هو الإصدار التكراري من خوارزمية عامل LU.
!> دع B = A - S. تقسم الخوارزمية المصفوفة B إلى أربع
!> مصفوفات فرعية:
!>
!>        [  B11 | B12  ]  حيث B11 هي n1 × n1،
!>    B = [ -----|----- ]        B21 هي (m-n1) × n1،
!>        [  B21 | B22  ]        B12 هي n1 × n2،
!>                               B22 هي (m-n1) × n2،
!>                               مع n1 = min(m,n)/2, n2 = n-n1.
!>
!>
!> يستدعي الروتين الفرعي نفسه لعامل B11، ويحل لـ B21،
!> ويحل لـ B12، ويحدث B22، ثم يستدعي نفسه لعامل B22.
!>
!> لمزيد من التفاصيل حول خوارزمية LU التكرارية، انظر [2].
!>
!> يتم استدعاء ZLAUNHR_COL_GETRFNP2 لعامل كتلة بواسطة
!> الروتين المجمع ZLAUNHR_COL_GETRFNP، الذي يستخدم رمزًا مجمعًا يستدعي
!> BLAS المستوى 3 لتحديث المصفوفة الفرعية. ومع ذلك، فإن ZLAUNHR_COL_GETRFNP2
!> مكتفٍ ذاتيًا ويمكن استخدامه بدون ZLAUNHR_COL_GETRFNP.
!>
!> [1] ,
!>     G. Ballard, J. Demmel, L. Grigori, M. Jacquelin, H.D. Nguyen,
!>     E. Solomonik, J. Parallel Distrib. Comput.,
!>     vol. 85, pp. 3-31, 2015.
!>
!> [2] , F. Gustavson, IBM J. of Res. and Dev.,
!>     vol. 41, no. 6, pp. 737-755, 1997.
!>

المعلمات

M 
!>          M هو صحيح (INTEGER)
!>          عدد صفوف المصفوفة A. M >= 0.
!>

N

!>          N هو صحيح (INTEGER)
!>          عدد أعمدة المصفوفة A. N >= 0.
!>

A

!>          A هو مصفوفة COMPLEX*16، ذات البعد (LDA,N)
!>          عند الدخول، المصفوفة M-by-N المراد تحليلها.
!>          عند الخروج، عوامل L و U من التحليل
!>          A-S=L*U؛ لا يتم تخزين عناصر القطر الأحادية لـ L.
!>

LDA

!>          LDA هو صحيح (INTEGER)
!>          البعد الأمامي للمصفوفة A. LDA >= max(1,M).
!>

D

!>          D هي مصفوفة COMPLEX*16، ذات البعد min(M,N)
!>          العناصر القطرية للمصفوفة القطرية M-by-N S،
!>          D(i) = S(i,i)، حيث 1 <= i <= min(M,N). العناصر يمكن أن تكون
!>          فقط ( +1.0, 0.0 ) أو (-1.0, 0.0 ).
!>

INFO

!>          INFO هو صحيح (INTEGER)
!>          = 0:  انتهاء ناجح
!>          < 0:  إذا كان INFO = -i، فإن المعامل i كان له قيمة غير قانونية
!>
