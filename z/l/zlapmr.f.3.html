subroutine zlapmr (logical forwrd, integer m, integer n, complex*16, dimension( ldx, * ) x, integer ldx, integer, dimension( * ) k)

ZLAPMR يعيد ترتيب صفوف المصفوفة كما هو محدد بواسطة متجه التبديل.

الغرض:

!>
!> ZLAPMR يعيد ترتيب صفوف المصفوفة M × N، X، كما هو محدد
!> بواسطة التبديل K(1),K(2),...,K(M) للأعداد الصحيحة 1,...,M.
!> إذا كان FORWRD = .TRUE.، تبديل أمامي:
!>
!>      يتم نقل X(K(I),*) إلى X(I,*) من أجل I = 1,2,...,M.
!>
!> إذا كان FORWRD = .FALSE.، تبديل خلفي:
!>
!>      يتم نقل X(I,*) إلى X(K(I),*) من أجل I = 1,2,...,M.
!>

المعلمات

FORWRD 
!>          FORWRD هو LOGICAL (منطقي)
!>          = .TRUE.، تبديل أمامي
!>          = .FALSE.، تبديل خلفي
!>

M

!>          M هو INTEGER (عدد صحيح)
!>          عدد صفوف المصفوفة X. M >= 0.
!>

N

!>          N هو INTEGER (عدد صحيح)
!>          عدد أعمدة المصفوفة X. N >= 0.
!>

X

!>          X هو مصفوفة COMPLEX*16
!>          عند الإدخال، مصفوفة X ذات الأبعاد M × N.
!>          عند الخروج، تحتوي X على المصفوفة X بعد إعادة الترتيب.
!>

LDX

!>          LDX هو INTEGER (عدد صحيح)
!>          البعد الرائد للمصفوفة X، LDX >= MAX(1,M).
!>

K

!>          K هي مصفوفة INTEGER (عدد صحيح)، البعد (M)
!>          عند الإدخال، تحتوي K على متجه التبديل. يتم استخدام K
!>          كمساحة عمل داخلية، ولكن يتم إعادة تعيين قيمتها الأصلية عند
!>          الخروج.
!>
