وثائق الوظيفة/الروتين الفرعي

subroutine zlakf2 (integer m, integer n, complex*16, dimension( lda, * ) a, integer lda, complex*16, dimension( lda, * ) b, complex*16, dimension( lda, * ) d, complex*16, dimension( lda, * ) e, complex*16, dimension( ldz, * ) z, integer ldz)

ZLAKF2

الغرض:

!>
!> تشكيل المصفوفة 2*M*N × 2*M*N
!>
!>        Z = [ kron(In, A)  -kron(B', Im) ]
!>            [ kron(In, D)  -kron(E', Im) ],
!>
!> حيث In هي مصفوفة الوحدة بالحجم n و X' هو منقول X. kron(X, Y) هو حاصل ضرب كرونيكر بين المصفوفات X
!> و Y.
!>

المعلمات

M 
!>          M هو عدد صحيح (INTEGER)
!>          حجم المصفوفة، يجب أن يكون >= 1.
!>

N

!>          N هو عدد صحيح (INTEGER)
!>          حجم المصفوفة، يجب أن يكون >= 1.
!>

A

!>          A هو عدد مركب معقد*16 (COMPLEX*16)، البعد ( LDA, M )
!>          المصفوفة A في المصفوفة الناتجة Z.
!>

LDA

!>          LDA هو عدد صحيح (INTEGER)
!>          البعد الرائد للمصفوفات A, B, D, و E. ( LDA >= M+N )
!>

B

!>          B هو عدد مركب معقد*16 (COMPLEX*16)، البعد ( LDA, N )
!>

D

!>          D هو عدد مركب معقد*16 (COMPLEX*16)، البعد ( LDA, M )
!>

E

!>          E هو عدد مركب معقد*16 (COMPLEX*16)، البعد ( LDA, N )
!>
!>          المصفوفات المستخدمة في تشكيل المصفوفة الناتجة Z.
!>

Z

!>          Z هو عدد مركب معقد*16 (COMPLEX*16)، البعد ( LDZ, 2*M*N )
!>          المصفوفة الناتجة كرونيكر M*N*2 × M*N*2 (انظر أعلاه.)
!>

LDZ

!>          LDZ هو عدد صحيح (INTEGER)
!>          البعد الرائد للمصفوفة Z. ( LDZ >= 2*M*N )
!>
