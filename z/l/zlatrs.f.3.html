subroutine zlatrs (character uplo, character trans, character diag, character normin, integer n, complex*16, dimension( lda, * ) a, integer lda, complex*16, dimension( * ) x, double precision scale, double precision, dimension( * ) cnorm, integer info)

ZLATRS يحل نظام معادلات مثلثي مع تعيين عامل القياس لمنع التدفق الزائد.

الغرض:

!>
!> ZLATRS يحل أحد الأنظمة المثلثية
!>
!>    A * x = s*b,  A**T * x = s*b,  أو  A**H * x = s*b,
!>
!> مع القياس لمنع التدفق الزائد. هنا A هي مصفوفة علوية أو سفلية
!> مثلثة، A**T ترمز إلى منقول A، A**H ترمز إلى منقول A المرافق، x و b هما متجهان من n عنصر، و s هو
!> عامل قياس، عادة ما يكون أقل من أو يساوي 1، يتم اختياره بحيث
!> ستكون مكونات x أقل من عتبة التدفق الزائد. إذا كان
!> المشكلة غير المقاسة لن تسبب تدفقًا زائدًا، يتم استدعاء روتين BLAS من المستوى 2
!> ZTRSV. إذا كانت المصفوفة A منفردة (A(j,j) = 0 لبعض j)،
!> عندها يتم تعيين s إلى 0 ويتم إرجاع حل غير تافه لـ A*x = 0.
!>

المعلمات

UPLO 
!>          UPLO هو CHARACTER*1
!>          يحدد ما إذا كانت المصفوفة A مثلثة علوية أو سفلية.
!>          = 'U':  مثلثة علوية
!>          = 'L':  مثلثة سفلية
!>

TRANS

!>          TRANS هو CHARACTER*1
!>          يحدد العملية المطبقة على A.
!>          = 'N':  حل A * x = s*b     (بدون منقول)
!>          = 'T':  حل A**T * x = s*b  (منقول)
!>          = 'C':  حل A**H * x = s*b  (منقول مرافق)
!>

DIAG

!>          DIAG هو CHARACTER*1
!>          يحدد ما إذا كانت المصفوفة A مثلثة موحدة أم لا.
!>          = 'N':  غير موحدة مثلثة
!>          = 'U':  موحدة مثلثة
!>

NORMIN

!>          NORMIN هو CHARACTER*1
!>          يحدد ما إذا تم تعيين CNORM أم لا.
!>          = 'Y':  تحتوي CNORM على معايير الأعمدة عند الدخول
!>          = 'N':  لم يتم تعيين CNORM عند الدخول. عند الخروج، ستكون المعايير
!>                  محسوبة ومخزنة في CNORM.
!>

N

!>          N هو INTEGER
!>          رتبة المصفوفة A.  N >= 0.
!>

A

!>          A هو مصفوفة COMPLEX*16، البعد (LDA,N)
!>          المصفوفة المثلثية A.  إذا كان UPLO = 'U'، فإن الجزء العلوي المثلثي الرئيسي N × N
!>          من المصفوفة A يحتوي على المصفوفة المثلثية العلوية، ولا يتم الرجوع إلى الجزء المثلثي السفلي الصارم
!>          من A.  إذا كان UPLO = 'L'، فإن الجزء السفلي المثلثي الرئيسي N × N
!>          من المصفوفة A يحتوي على المصفوفة المثلثية السفلية، ولا يتم الرجوع إلى الجزء المثلثي العلوي الصارم
!>          من A.  إذا كان DIAG = 'U'، فإن العناصر القطرية لـ A
!>          لا تتم الإشارة إليها أيضًا وتُفترض أنها 1.
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرئيسي للمصفوفة A.  LDA >= max (1,N).
!>

X

!>          X هو مصفوفة COMPLEX*16، البعد (N)
!>          عند الدخول، الجانب الأيمن b للنظام المثلثي.
!>          عند الخروج، يتم الكتابة فوق X بواسطة متجه الحل x.
!>

SCALE

!>          SCALE هو DOUBLE PRECISION
!>          عامل القياس s للنظام المثلثي
!>             A * x = s*b,  A**T * x = s*b,  أو  A**H * x = s*b.
!>          إذا كان SCALE = 0، فإن المصفوفة A منفردة أو ذات قياس سيء،
!>          والمتجه x هو حل دقيق أو تقريبي لـ A*x = 0.
!>

CNORM

!>          CNORM هو مصفوفة DOUBLE PRECISION، البعد (N)
!>
!>          إذا كان NORMIN = 'Y'، فإن CNORM هو وسيط إدخال و CNORM(j)
!>          يحتوي على معيار الجزء غير القطري للعمود j
!>          من A.  إذا كان TRANS = 'N'، يجب أن يكون CNORM(j) أكبر من أو يساوي
!>          المعيار اللانهائي، وإذا كان TRANS = 'T' أو 'C'، فيجب أن يكون CNORM(j)
!>          أكبر من أو يساوي المعيار 1.
!>
!>          إذا كان NORMIN = 'N'، فإن CNORM هو وسيط إخراج و CNORM(j)
!>          يعيد المعيار 1 للجزء غير القطري للعمود j
!>          من A.
!>

INFO

!>          INFO هو INTEGER
!>          = 0:  خروج ناجح
!>          < 0:  إذا كان INFO = -k، فقد كان الوسيط k ذا قيمة غير قانونية
!>

تفاصيل إضافية:

!>
!> يتم حساب حد تقريبي لـ x؛ إذا كان هذا الحد أقل من التدفق الزائد، يتم استدعاء ZTRSV
!> وإلا، يتم استخدام كود محدد يتحقق من احتمال
!> التدفق الزائد أو القسمة على صفر في كل عملية.
!>
!> يتم استخدام مخطط عمودي لحل A*x = b. الخوارزمية الأساسية
!> إذا كانت A مثلثة سفلية هي
!>
!>       x[1:n] := b[1:n]
!>       for j = 1, ..., n
!>            x(j) := x(j) / A(j,j)
!>            x[j+1:n] := x[j+1:n] - x(j) * A[j+1:n,j]
!>       end
!>
!> تحديد حدود لمكونات x بعد j تكرار للحلقة:
!>     M(j) = حد لـ x[1:j]
!>     G(j) = حد لـ x[j+1:n]
!> بشكل مبدئي، نضع M(0) = 0 و G(0) = max{x(i), i=1,...,n}.
!>
!> ثم للتكرار j+1 لدينا
!>     M(j+1) <= G(j) / | A(j+1,j+1) |
!>     G(j+1) <= G(j) + M(j+1) * | A[j+2:n,j+1] |
!>            <= G(j) ( 1 + CNORM(j+1) / | A(j+1,j+1) | )
!>
!> حيث CNORM(j+1) أكبر من أو يساوي المعيار اللانهائي
!> للعمود j+1 من A، باستثناء القطر. وبالتالي
!>
!>     G(j) <= G(0) product ( 1 + CNORM(i) / | A(i,i) | )
!>                  1<=i<=j
!> و
!>
!>     |x(j)| <= ( G(0) / |A(j,j)| ) product ( 1 + CNORM(i) / |A(i,i)| )
!>                                   1<=i< j
!>
!> بما أن |x(j)| <= M(j)، نستخدم روتين BLAS من المستوى 2 ZTRSV إذا كان
!> معكوس أكبر M(j)، j=1,..,n، أكبر من
!> max(underflow, 1/overflow).
!>
!> يستخدم الحد على x(j) أيضًا لتحديد متى يمكن إجراء خطوة في
!> المخطط العمودي دون خوف من التدفق الزائد. إذا
!> الحد المحسوب أكبر من ثابت كبير، يتم قياس x لمنع
!> التدفق الزائد، ولكن إذا تدفق الحد الزائد، يتم تعيين x إلى 0، x(j) إلى
!> 1، و scale إلى 0، ويتم العثور على حل غير تافه لـ A*x = 0.
!>
!> وبالمثل، يتم استخدام مخطط صفي لحل A**T *x = b أو
!> A**H *x = b. الخوارزمية الأساسية لـ A علوية مثلثية هي
!>
!>       for j = 1, ..., n
!>            x(j) := ( b(j) - A[1:j-1,j]' * x[1:j-1] ) / A(j,j)
!>       end
!>
!> نقوم بحساب حدين في نفس الوقت
!>       G(j) = حد لـ ( b(i) - A[1:i-1,i]' * x[1:i-1] )، 1<=i<=j
!>       M(j) = حد لـ x(i)، 1<=i<=j
!>
!> القيم الأولية هي G(0) = 0، M(0) = max{b(i)، i=1,..,n}، و نضيف
!> القيد G(j) >= G(j-1) و M(j) >= M(j-1) لـ j >= 1.
!> ثم حد x(j) هو
!>
!>       M(j) <= M(j-1) * ( 1 + CNORM(j) ) / | A(j,j) |
!>
!>            <= M(0) * product ( ( 1 + CNORM(i) ) / |A(i,i)| )
!>                      1<=i<=j
!>
!> ويمكننا استدعاء ZTRSV بأمان إذا كان 1/M(n) و 1/G(n) كلاهما أكبر
!> من max(underflow, 1/overflow).
!>
