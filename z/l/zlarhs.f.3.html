subroutine zlarhs (character*3 path, character xtype, character uplo, character trans, integer m, integer n, integer kl, integer ku, integer nrhs, complex*16, dimension( lda, * ) a, integer lda, complex*16, dimension( ldx, * ) x, integer ldx, complex*16, dimension( ldb, * ) b, integer ldb, integer, dimension( 4 ) iseed, integer info)

ZLARHS

الغرض:

!>
!> ZLARHS يختار مجموعة من NRHS ناقلات حل عشوائية ويقوم بإعداد
!> الأطراف اليمنى لنظام الخطية
!>    op(A) * X = B,
!> حيث op(A) = A، A**T، أو A**H، اعتمادًا على TRANS.
!>

المعلمات

PATH 
!>          PATH هو CHARACTER*3
!>          نوع المصفوفة المركبة A. يمكن تقديم PATH بأي
!>          مزيج من الأحرف الكبيرة والصغيرة. المسارات الصالحة تشمل
!>             xGE: مصفوفة عامة m x n
!>             xGB: مصفوفة نطاقية عامة
!>             xPO: مصفوفة هرمية موجبة محددة، تخزين ثنائي الأبعاد
!>             xPP: مصفوفة هرمية موجبة محددة مضغوطة
!>             xPB: مصفوفة هرمية موجبة محددة نطاقية
!>             xHE: مصفوفة هرمية غير محددة، تخزين ثنائي الأبعاد
!>             xHP: مصفوفة هرمية غير محددة مضغوطة
!>             xHB: مصفوفة هرمية غير محددة نطاقية
!>             xSY: مصفوفة متماثلة غير محددة، تخزين ثنائي الأبعاد
!>             xSP: مصفوفة متماثلة غير محددة مضغوطة
!>             xSB: مصفوفة متماثلة غير محددة نطاقية
!>             xTR: مصفوفة مثلثية
!>             xTP: مصفوفة مثلثية مضغوطة
!>             xTB: مصفوفة مثلثية نطاقية
!>             xQR: مصفوفة عامة m x n
!>             xLQ: مصفوفة عامة m x n
!>             xQL: مصفوفة عامة m x n
!>             xRQ: مصفوفة عامة m x n
!>          حيث يشير الحرف الأول إلى الدقة.
!>

XTYPE

!>          XTYPE هو CHARACTER*1
!>          يحدد كيف سيتم تحديد الحل الدقيق X:
!>          = 'N': حل جديد؛ قم بإنشاء X عشوائي.
!>          = 'C': محسوب؛ استخدم قيمة X عند الدخول.
!>

UPLO

!>          UPLO هو CHARACTER*1
!>          يستخدم فقط إذا كانت A متماثلة أو مثلثية؛ يحدد ما إذا كان
!>          الجزء العلوي أو السفلي المثلثي من المصفوفة A مخزنًا.
!>          = 'U': مثلثية علوية
!>          = 'L': مثلثية سفلية
!>

TRANS

!>          TRANS هو CHARACTER*1
!>          يستخدم فقط إذا كانت A غير متماثلة؛ يحدد العملية
!>          المطبقة على المصفوفة A.
!>          = 'N': B := A * X (لا يوجد تبديل)
!>          = 'T': B := A**T * X (تبديل)
!>          = 'C': B := A**H * X (تبديل مترافق)
!>

M

!>          M هو INTEGER
!>          عدد صفوف المصفوفة A. M >= 0.
!>

N

!>          N هو INTEGER
!>          عدد أعمدة المصفوفة A. N >= 0.
!>

KL

!>          KL هو INTEGER
!>          يستخدم فقط إذا كانت A مصفوفة نطاقية؛ يحدد عدد
!>          الأقطار الفرعية لـ A إذا كانت A مصفوفة نطاقية عامة أو إذا كانت A
!>          متماثلة أو مثلثية و UPLO = 'L'؛ يحدد عدد
!>          الأقطار الفرعية لـ A إذا كانت A متماثلة أو مثلثية و
!>          UPLO = 'U'. 0 <= KL <= M-1.
!>

KU

!>          KU هو INTEGER
!>          يستخدم فقط إذا كانت A مصفوفة نطاقية عامة أو إذا كانت A
!>          مثلثية.
!>
!>          إذا كان PATH = xGB، فإنه يحدد عدد الأقطار الفرعية لـ A،
!>          و 0 <= KU <= N-1.
!>
!>          إذا كان PATH = xTR، xTP، أو xTB، فإنه يحدد ما إذا كانت
!>          المصفوفة لها قطر موحد أم لا:
!>          = 1: المصفوفة لها قطر غير موحد (افتراضي)
!>          = 2: المصفوفة لها قطر موحد
!>

NRHS

!>          NRHS هو INTEGER
!>          عدد متجهات الطرف الأيمن في النظام A*X = B.
!>

A

!>          A هو مصفوفة COMPLEX*16، البعد (LDA,N)
!>          مصفوفة الاختبار التي يحدد نوعها PATH.
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرائد للمصفوفة A.
!>          إذا كان PATH = xGB، LDA >= KL+KU+1.
!>          إذا كان PATH = xPB، xSB، xHB، أو xTB، LDA >= KL+1.
!>          بخلاف ذلك، LDA >= max(1,M).
!>

X

!>          X هي أو (خرج) مصفوفة COMPLEX*16، البعد (LDX,NRHS)
!>          عند الدخول، إذا كان XTYPE = 'C' (لـ 'محسوب')، فإن X تحتوي على
!>          الحل الدقيق لنظام المعادلات الخطية.
!>          عند الخروج، إذا كان XTYPE = 'N' (لـ 'جديد')، يتم تهيئة X
!>          بقِيم عشوائية.
!>

LDX

!>          LDX هو INTEGER
!>          البعد الرائد للمصفوفة X. إذا كان TRANS = 'N'،
!>          LDX >= max(1,N)؛ إذا كان TRANS = 'T'، LDX >= max(1,M).
!>

B

!>          B هي مصفوفة COMPLEX*16، البعد (LDB,NRHS)
!>          متجه (متجهات) الطرف الأيمن لنظام المعادلات،
!>          محسوبة من B = op(A) * X، حيث يتم تحديد op(A) بواسطة
!>          TRANS.
!>

LDB

!>          LDB هو INTEGER
!>          البعد الرائد للمصفوفة B. إذا كان TRANS = 'N'،
!>          LDB >= max(1,M)؛ إذا كان TRANS = 'T'، LDB >= max(1,N).
!>

ISEED

!>          ISEED هي مصفوفة INTEGER، البعد (4)
!>          متجه البذرة لمولد الأرقام العشوائية (المستخدم في
!>          ZLATMS). يتم تعديله عند الخروج.
!>

INFO

!>          INFO هو INTEGER
!>          = 0: خروج ناجح
!>          < 0: إذا كان INFO = -i، فإن المعلمة i-th كانت لها قيمة غير قانونية
!>
