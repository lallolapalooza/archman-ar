subroutine zlasyf_rook (uplo, n, nb, kb, a, lda, ipiv, w, ldw, info)
ZLASYF_ROOK تحسب تجزئة جزئية لمصفوفة متناظرة معقدة باستخدام طريقة التبديل القطري المحدود لـ Bunch-Kaufman ( "rook" ).

توثيق الوظيفة/الروتين الفرعي

subroutine zlasyf_rook (character uplo, integer n, integer nb, integer kb, complex*16, dimension( lda, * ) a, integer lda, integer, dimension( * ) ipiv, complex*16, dimension( ldw, * ) w, integer ldw, integer info)

ZLASYF_ROOK تحسب تجزئة جزئية لمصفوفة متناظرة معقدة باستخدام طريقة التبديل القطري المحدود ( "rook" ) لـ Bunch-Kaufman.

الغرض:

!>
!> ZLASYF_ROOK تحسب تجزئة جزئية لمصفوفة متناظرة معقدة A باستخدام طريقة
!> التبديل القطري المحدود ( "rook" ) لـ Bunch-Kaufman. التجزئة الجزئية لها الشكل:
!>
!> A  =  ( I  U12 ) ( A11  0  ) (  I       0    )  إذا كان UPLO = 'U'، أو:
!>       ( 0  U22 ) (  0   D  ) ( U12**T U22**T )
!>
!> A  =  ( L11  0 ) (  D   0  ) ( L11**T L21**T )  إذا كان UPLO = 'L'
!>       ( L21  I ) (  0  A22 ) (  0       I    )
!>
!> حيث يكون ترتيب D لا يزيد عن NB. يتم إرجاع الترتيب الفعلي في
!> المتغير KB، وهو إما NB أو NB-1، أو N إذا كان N <= NB.
!>
!> ZLASYF_ROOK هي روتينية مساعدة تستدعيها ZSYTRF_ROOK. تستخدم
!> الشفرة المجمعة (تستدعي BLAS المستوى 3) لتحديث المصفوفة الفرعية
!> A11 (إذا كان UPLO = 'U') أو A22 (إذا كان UPLO = 'L').
!>

المتغيرات

UPLO 
!>          UPLO هو CHARACTER*1
!>          تحدد ما إذا كان الجزء العلوي أو السفلي المثلثي من المصفوفة المتناظرة A مخزناً:
!>          = 'U':  مثلثي علوي
!>          = 'L':  مثلثي سفلي
!>

N

!>          N هو INTEGER
!>          ترتيب المصفوفة A. N >= 0.
!>

NB

!>          NB هو INTEGER
!>          الحد الأقصى لعدد الأعمدة من المصفوفة A التي يجب
!>          تجزئتها. يجب أن يكون NB على الأقل 2 للسماح بكتل محورية بحجم 2x2.
!>

KB

!>          KB هو INTEGER
!>          عدد الأعمدة من A التي تم تجزئتها بالفعل.
!>          KB هو إما NB-1 أو NB، أو N إذا كان N <= NB.
!>

A

!>          A هو مصفوفة COMPLEX*16، بالبعد (LDA,N)
!>          عند الدخول، المصفوفة المتناظرة A. إذا كان UPLO = 'U'، فإن الجزء
!>          العلوي المثلثي الرئيسي بحجم n*n من A يحتوي على الجزء
!>          العلوي المثلثي للمصفوفة A، والجزء السفلي المثلثي الصارم
!>          من A لا تتم الإشارة إليه. إذا كان UPLO = 'L'، فإن الجزء
!>          السفلي المثلثي الرئيسي بحجم n*n من A يحتوي على الجزء
!>          السفلي المثلثي للمصفوفة A، والجزء العلوي المثلثي الصارم
!>          من A لا تتم الإشارة إليه.
!>          عند الخروج، تحتوي A على تفاصيل التجزئة الجزئية.
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرائد للمصفوفة A. LDA >= max(1,N).
!>

IPIV

!>          IPIV هو مصفوفة INTEGER، بالبعد (N)
!>          تفاصيل التبديلات وهيكل الكتلة لـ D.
!>
!>          إذا كان UPLO = 'U':
!>             يتم تعيين العناصر KB الأخيرة فقط من IPIV.
!>
!>             إذا كان IPIV(k) > 0، فقد تم تبديل الصفوف والأعمدة k و IPIV(k)
!>             و D(k,k) هي كتلة قطرية بحجم 1x1.
!>
!>             إذا كان IPIV(k) < 0 و IPIV(k-1) < 0، فقد تم تبديل الصفوف
!>             والأعمدة k و -IPIV(k) و تم تبديل الصفوف
!>             والأعمدة k-1 و -IPIV(k-1)،
!>             D(k-1:k,k-1:k) هي كتلة قطرية بحجم 2x2.
!>
!>          إذا كان UPLO = 'L':
!>             يتم تعيين العناصر KB الأولى فقط من IPIV.
!>
!>             إذا كان IPIV(k) > 0، فقد تم تبديل الصفوف والأعمدة k و IPIV(k)
!>             و D(k,k) هي كتلة قطرية بحجم 1x1.
!>
!>             إذا كان IPIV(k) < 0 و IPIV(k+1) < 0، فقد تم تبديل الصفوف
!>             والأعمدة k و -IPIV(k) و تم تبديل الصفوف
!>             والأعمدة k+1 و -IPIV(k+1)،
!>             D(k:k+1,k:k+1) هي كتلة قطرية بحجم 2x2.
!>

W

!>          W هي مصفوفة COMPLEX*16، بالبعد (LDW,NB)
!>

LDW

!>          LDW هو INTEGER
!>          البعد الرائد للمصفوفة W. LDW >= max(1,N).
!>

INFO

!>          INFO هو INTEGER
!>          = 0: خروج ناجح
!>          > 0: إذا كان INFO = k، فإن D(k,k) تساوي صفرًا تمامًا. اكتملت التجزئة،
!>               ولكن المصفوفة القطرية الكتلة D متفردة تمامًا.
!>
