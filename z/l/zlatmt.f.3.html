subroutine zlatmt (integer m, integer n, character dist, integer, dimension( 4 ) iseed, character sym, double precision, dimension( * ) d, integer mode, double precision cond, double precision dmax, integer rank, integer kl, integer ku, character pack, complex*16, dimension( lda, * ) a, integer lda, complex*16, dimension( * ) work, integer info)

ZLATMT

الغرض:

!>
!>    ZLATMT تولد مصفوفات عشوائية بقيم فردية محددة
!>    (أو مصفوفات هرميتية بقيم ذاتية محددة)
!>    لاختبار برامج LAPACK.
!>
!>    تعمل ZLATMT عن طريق تطبيق التسلسل التالي من
!>    العمليات:
!>
!>      تعيين القطر إلى D، حيث قد تكون D مدخلة أو
!>         محسوبة وفقًا لـ MODE، COND، DMAX، و SYM
!>         كما هو موضح أدناه.
!>
!>      توليد مصفوفة بالبنية النطاقية المناسبة، بواسطة إحدى
!>         الطريقتين:
!>
!>      الطريقة أ:
!>          توليد مصفوفة كثيفة M x N بضرب D على اليسار
!>              وعلى اليمين بمصفوفات وحدوية عشوائية، ثم:
!>
!>          تقليل عرض النطاق وفقًا لـ KL و KU، باستخدام
!>              تحويلات هاوسهولدر.
!>
!>      الطريقة ب:
!>          تحويل المصفوفة ذات عرض النطاق الصفري (أي، قطرية) إلى
!>              مصفوفة ذات عرض نطاق 1 باستخدام دورانات جيفنز،
!>              عناصر خارج النطاق مرة أخرى، تمامًا مثل في QR؛ ثم تحويل
!>              عرض النطاق 1 إلى عرض نطاق 2، وهكذا. لاحظ
!>              أنه لعروض النطاقات الصغيرة بشكل معقول (مقارنة بـ M و
!>              N) يتطلب هذا مساحة تخزين أقل، حيث لا يتم توليد مصفوفة كثيفة.
!>              أيضًا، للمصفوفات الهرميتية أو المتماثلة،
!>              يتم توليد مثلث واحد فقط.
!>
!>      يتم اختيار الطريقة أ إذا كان عرض النطاق يمثل جزءًا كبيرًا من
!>          رتبة المصفوفة، و LDA على الأقل M (بحيث يمكن تخزين مصفوفة كثيفة.)
!>          يتم اختيار الطريقة ب إذا كان عرض النطاق
!>          صغيرًا (< 1/2 N للمصفوفات الهرميتية أو المتماثلة، < .3 N+M لغير
!>          المتماثلة)، أو LDA أقل من M وليس أقل من
!>          عرض النطاق.
!>
!>      تعبئة المصفوفة إذا رغبت. الخيارات التي يحددها PACK هي:
!>         لا يوجد تعبئة
!>         تصفير النصف العلوي (إذا كانت هرميتية)
!>         تصفير النصف السفلي (إذا كانت هرميتية)
!>         تخزين النصف العلوي عموديًا (إذا كانت هرميتية أو
!>               مثلثة علوية)
!>         تخزين النصف السفلي عموديًا (إذا كانت هرميتية أو
!>               مثلثة سفلية)
!>         تخزين المثلث السفلي بتنسيق نطاقي (إذا كانت هرميتية أو
!>               مثلثة سفلية)
!>         تخزين المثلث العلوي بتنسيق نطاقي (إذا كانت هرميتية أو
!>               مثلثة علوية)
!>         تخزين المصفوفة بأكملها بتنسيق نطاقي
!>      إذا تم اختيار الطريقة ب، وتم تحديد تنسيق النطاق، فسيتم
!>         توليد المصفوفة بتنسيق النطاق، لذلك لن تكون هناك حاجة لإعادة التعبئة.
!>

!>           'U' => منتظم ( 0, 1 ) ( 'U' للـ uniform )
!>           'S' => منتظم ( -1, 1 ) ( 'S' للمتماثل )
!>           'N' => طبيعي ( 0, 1 ) ( 'N' للـ normal )
!>           لم يتم تعديله.

!>           يجب أن تكون القيم بين 0 و 4095 شاملة، ويجب أن يكون ISEED(4) فرديًا.
!>           يستخدم مولد الأرقام العشوائية تسلسلًا خطيًا توافقيًا محدودًا بالأعداد
!>           الصحيحة الصغيرة، وبالتالي يجب أن ينتج أرقامًا عشوائية مستقلة عن الجهاز.

!>           تم التغيير عند الخروج.

SYM

!>          SYM هو حرف (CHARACTER*1)
!>           إذا كان SYM='H'، فإن المصفوفة المولدة تكون هرميتية، مع
!>             قيم ذاتية محددة بواسطة D، COND، MODE، و DMAX؛ قد تكون
!>             موجبة، سالبة، أو صفرية.
!>           إذا كان SYM='P'، فإن المصفوفة المولدة تكون هرميتية، مع
!>             قيم ذاتية (= قيم فردية) محددة بواسطة D، COND،
!>             MODE، و DMAX؛ لن تكون سالبة.
!>           إذا كان SYM='N'، فإن المصفوفة المولدة تكون غير متماثلة، مع
!>             قيم فردية محددة بواسطة D، COND، MODE، و DMAX؛
!>             لن تكون سالبة.
!>           إذا كان SYM='S'، فإن المصفوفة المولدة تكون متماثلة (مركبة)،
!>             مع قيم فردية محددة بواسطة D، COND، MODE، و
!>             DMAX؛ لن تكون سالبة.
!>           لم يتم تعديله.
!>

D

!>          D هي مصفوفة دقة مزدوجة (DOUBLE PRECISION array)، ذات بعد ( MIN( M, N ) )
!>           تستخدم هذه المصفوفة لتحديد القيم الفردية أو
!>           القيم الذاتية لـ A (انظر SYM، أعلاه). إذا كان MODE=0، فمن المفترض أن
!>           تحتوي D على القيم الفردية/الذاتية، وإلا
!>           فسيتم حسابها وفقًا لـ MODE، و COND، و DMAX،
!>           وسيتم وضعها في D.
!>           تم التعديل إذا كان MODE غير صفري.
!>

MODE

!>          MODE هو عدد صحيح (INTEGER)
!>           عند الإدخال، يصف هذا كيفية تحديد القيم الفردية/الذاتية:
!>           MODE = 0 يعني استخدام D كمدخل
!>           MODE = 1 يحدد D(1)=1 و D(2:RANK)=1.0/COND
!>           MODE = 2 يحدد D(1:RANK-1)=1 و D(RANK)=1.0/COND
!>           MODE = 3 يحدد D(I)=COND**(-(I-1)/(RANK-1))
!>           MODE = 4 يحدد D(i)=1 - (i-1)/(N-1)*(1 - 1/COND)
!>           MODE = 5 يحدد D لأرقام عشوائية في النطاق
!>                    ( 1/COND , 1 ) بحيث تكون لوغاريتماتها
!>                    موزعة بشكل منتظم.
!>           MODE = 6 يحدد D لأرقام عشوائية من نفس التوزيع
!>                    مثل بقية المصفوفة.
!>           MODE < 0 له نفس المعنى مثل ABS(MODE)، باستثناء أن
!>              ترتيب عناصر D معكوس.
!>           وبالتالي، إذا كان MODE موجبًا، فإن D تحتوي على عناصر تتراوح من
!>              1 إلى 1/COND، وإذا كان سالبًا، من 1/COND إلى 1،
!>           إذا كان SYM='H'، وكان MODE ليس 0، 6، ولا -6، فسيتم
!>              أيضًا ضرب عناصر D بإشارة عشوائية (أي، +1 أو -1.)
!>           لم يتم تعديله.
!>

COND

!>          COND هو دقة مزدوجة (DOUBLE PRECISION)
!>           عند الإدخال، يستخدم كما هو موضح تحت MODE أعلاه.
!>           إذا تم استخدامه، يجب أن يكون >= 1. لم يتم تعديله.
!>

DMAX

!>          DMAX هو دقة مزدوجة (DOUBLE PRECISION)
!>           إذا كان MODE ليس -6، 0، ولا 6، فسيتم تحجيم محتويات D، كما
!>           تم حسابها وفقًا لـ MODE و COND، بواسطة
!>           DMAX / max(abs(D(i))); وبالتالي، فإن أقصى قيمة مطلقة ذاتية أو
!>           فردية (وهي المعيار) ستكون abs(DMAX).
!>           لاحظ أن DMAX ليس بالضرورة موجبًا: إذا كان DMAX سالبًا
!>           (أو صفريًا)، فسيتم تحجيم D بعدد سالب (أو صفر).
!>           لم يتم تعديله.
!>

RANK

Parameters




 U 




U is CHARACTER*1


Specifies the upper/lower triangular storage scheme.


 'U' => upper triangular
 'L' => lower triangular



Not modified.




 DIM 




DIM is CHARACTER*1


Specifies whether to store the matrix in a packed
    form.


 'N' => store in the full matrix A
 'Y' => store in packed form in A



Not modified.




 M 




M is INTEGER


The number of rows of the matrix A. If SYM is not
    'N', then M must equal N.


Not modified.




 N 




N is INTEGER


The number of columns of the matrix A. If SYM is not
    'N', then N must equal M.


Not modified.




 DIST 




DIST is CHARACTER*1


Specifies the distribution of the elements of the matrix.


 'U' => uniformly distributed
 'S' => normally distributed
 'H' => normally distributed, with difference of the order of
        the square root of the machine precision
 'R' => normally distributed, with difference of the order of
        the square root of the machine precision



Not modified.




 SYM 




SYM is CHARACTER*1


Specifies the symmetry of the matrix.


 'N' => not symmetric
 'S' => symmetric
 'H' => symmetric (Hermitian)
 'P' => symmetric (positive definite)



Not modified.




 MODE 




MODE is INTEGER


Specifies the condition of the matrix.


 MODE = 0, 1, 2, 3, 4, 5, 6 (see DLATMS)


Not modified.




 COND 




COND is REAL*8


Specifies the condition number of the matrix.


Not modified.




 DMAX 




DMAX is REAL*8


Specifies the maximum absolute value of the diagonal
    entries of the matrix.


Not modified.




 KL 




KL is INTEGER


This specifies the lower bandwidth of the matrix. For
    example, KL=0 implies upper triangular, KL=1 implies upper
    Hessenberg, and KL being at least M-1 means that the matrix
    has full lower bandwidth. KL must equal KU if the matrix
    is symmetric or hermitian.


Not modified.




 KU 




KU is INTEGER


This specifies the upper bandwidth of the matrix. For
    example, KU=0 implies lower triangular, KU=1 implies lower
    Hessenberg, and KU being at least N-1 means that the matrix
    has full upper bandwidth. KL must equal KU if the matrix
    is symmetric or hermitian.


Not modified.




 PACK 




PACK is CHARACTER*1


This specifies packing of matrix as follows:


 'N' => no packing
 'U' => zero out all subdiagonal entries (if symmetric
        or hermitian)
 'L' => zero out all superdiagonal entries (if symmetric
        or hermitian)
 'C' => store the upper triangle columnwise (only if the
        matrix is symmetric, hermitian, or upper triangular)
 'R' => store the lower triangle columnwise (only if the
        matrix is symmetric, hermitian, or lower triangular)
 'B' => store the lower triangle in band storage scheme
        (only if the matrix is symmetric, hermitian, or
        lower triangular)
 'Q' => store the upper triangle in band storage scheme
        (only if the matrix is symmetric, hermitian, or
        upper triangular)
 'Z' => store the entire matrix in band storage scheme
                    (pivoting can be provided for by using this
                    option to store A in the trailing rows of
                    the allocated storage)



Using these options, the various LAPACK packed and banded
    storage schemes can be obtained:


 GB                    - use 'Z'
 PB, SB, HB, or TB     - use 'B' or 'Q'
 PP, SP, HB, or TP     - use 'C' or 'R'



If two calls to ZLATMT differ only in the PACK parameter,
    they will generate mathematically equivalent matrices.


Not modified.




 A 




A is COMPLEX*16 array, dimension ( LDA, N )


On exit A is the desired test matrix. A is first generated
    in full (unpacked) form, and then packed, if so specified
    by PACK. Thus, the first M elements of the first N
    columns will always be modified. If PACK specifies a
    packed or banded storage scheme, all LDA elements of the
    first N columns will be modified; the elements of the
    array which do not correspond to elements of the generated
    matrix are set to zero.


Modified.




 LDA 




LDA is INTEGER


LDA specifies the first dimension of A as declared in the
    calling program. If PACK='N', 'U', 'L', 'C', or 'R', then
    LDA must be at least M. If PACK='B' or 'Q', then LDA must
    be at least MIN( KL, M-1) (which is equal to MIN(KU,N-1)).
    If PACK='Z', LDA must be large enough to hold the packed
    array: MIN( KU, N-1) + MIN( KL, M-1) + 1.


Not modified.




 WORK 




WORK is COMPLEX*16 array, dimension ( 3*MAX( N, M ) )


Workspace.


Modified.




 INFO 




INFO is INTEGER


Error code. On exit, INFO will be set to one of the
    following values:


  0 => normal return
 -1 => M negative or unequal to N and SYM='S', 'H', or 'P'
 -2 => N negative
 -3 => DIST illegal string
 -5 => SYM illegal string
 -7 => MODE not in range -6 to 6
 -8 => COND less than 1.0, and MODE neither -6, 0 nor 6
-10 => KL negative
-11 => KU negative, or SYM is not 'N' and KU is not equal to
       KL
-12 => PACK illegal string, or PACK='U' or 'L', and SYM='N';
       or PACK='C' or 'Q' and SYM='N' and KL is not zero;
       or PACK='R' or 'B' and SYM='N' and KU is not zero;
       or PACK='U', 'L', 'C', 'R', 'B', or 'Q', and M is not
       N.
-14 => LDA is less than M, or PACK='Z' and LDA is less than
       MIN(KU,N-1) + MIN(KL,M-1) + 1.
  1  => Error return from DLATM7
  2  => Cannot scale to DMAX (max. sing. value is 0)
  3  => Error return from ZLAGGE, ZLAGHE or ZLAGSY
