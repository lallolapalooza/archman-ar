توثيق الوظيفة/الإجراء الفرعي

subroutine zlqt03 (integer m, integer n, integer k, complex*16, dimension( lda, * ) af, complex*16, dimension( lda, * ) c, complex*16, dimension( lda, * ) cc, complex*16, dimension( lda, * ) q, integer lda, complex*16, dimension( * ) tau, complex*16, dimension( lwork ) work, integer lwork, double precision, dimension( * ) rwork, double precision, dimension( * ) result)

ZLQT03

الغرض:

!>
!> ZLQT03 يختبر ZUNMLQ، الذي يقوم بحساب Q*C، Q'*C، C*Q أو C*Q'.
!>
!> ZLQT03 يقارن نتائج استدعاء ZUNMLQ بنتائج
!> تشكيل Q بشكل صريح عن طريق استدعاء ZUNGLQ ثم إجراء
!> ضرب المصفوفات عن طريق استدعاء ZGEMM.
!>

المعلمات

M 
!>          M هو عدد صحيح (INTEGER)
!>          عدد الصفوف أو الأعمدة للمصفوفة C؛ C هي n-by-m إذا
!>          تم تطبيق Q من اليسار، أو m-by-n إذا
!>          تم تطبيق Q من اليمين.  M >= 0.
!>

N

!>          N هو عدد صحيح (INTEGER)
!>          ترتيب المصفوفة المتعامدة Q.  N >= 0.
!>

K

!>          K هو عدد صحيح (INTEGER)
!>          عدد المقسّمات الأولية التي يشكل حاصل ضربها
!>          المصفوفة المتعامدة Q.  N >= K >= 0.
!>

AF

!>          AF هي مصفوفة COMPLEX*16، بأبعاد (LDA,N)
!>          تفاصيل تحليل LQ لمصفوفة m-by-n، كما
!>          تم إرجاعها بواسطة ZGELQF. انظر CGELQF لمزيد من التفاصيل.
!>

C

!>          C هي مصفوفة COMPLEX*16، بأبعاد (LDA,N)
!>

CC

!>          CC هي مصفوفة COMPLEX*16، بأبعاد (LDA,N)
!>

Q

!>          Q هي مصفوفة COMPLEX*16، بأبعاد (LDA,N)
!>

LDA

!>          LDA هو عدد صحيح (INTEGER)
!>          البعد الرئيسي للمصفوفات AF، C، CC، و Q.
!>

TAU

!>          TAU هي مصفوفة COMPLEX*16، بأبعاد (min(M,N))
!>          العوامل القياسية للمقسّمات الأولية المقابلة
!>          لتحليل LQ في AF.
!>

WORK

!>          WORK هي مصفوفة COMPLEX*16، بأبعاد (LWORK)
!>

LWORK

!>          LWORK هو عدد صحيح (INTEGER)
!>          طول WORK. يجب أن يكون LWORK على الأقل M، ويجب أن يكون
!>          M*NB، حيث NB هو حجم الكتلة لهذه البيئة.
!>

RWORK

!>          RWORK هي مصفوفة DOUBLE PRECISION، بأبعاد (M)
!>

RESULT

!>          RESULT هي مصفوفة DOUBLE PRECISION، بأبعاد (4)
!>          تقارن نسب الاختبار بين تقنيتين لضرب مصفوفة عشوائية C
!>          بمصفوفة متعامدة n-by-n، Q.
!>          RESULT(1) = norm( Q*C - Q*C )  / ( N * norm(C) * EPS )
!>          RESULT(2) = norm( C*Q - C*Q )  / ( N * norm(C) * EPS )
!>          RESULT(3) = norm( Q'*C - Q'*C )/ ( N * norm(C) * EPS )
!>          RESULT(4) = norm( C*Q' - C*Q' )/ ( N * norm(C) * EPS )
!>
