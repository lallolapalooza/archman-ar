subroutine zlagsy (integer n, integer k, double precision, dimension( * ) d, complex*16, dimension( lda, * ) a, integer lda, integer, dimension( 4 ) iseed, complex*16, dimension( * ) work, integer info)

ZLAGSY

الغرض:

!>
!> ZLAGSY يولد مصفوفة متماثلة مركبة A، عن طريق الضرب المسبق واللاحق
!> لمصفوفة قطرية حقيقية D بمصفوفة وحدوية عشوائية:
!> A = U*D*U**T. قد يتم تقليل شبه النطاق بعد ذلك إلى k عن طريق
!> تحويلات وحدوية إضافية.
!>

المعلمات

N 
!>          N هو عدد صحيح
!>          رتبة المصفوفة A.  N >= 0.
!>

K

!>          K هو عدد صحيح
!>          عدد الأقطار الفرعية غير الصفرية داخل نطاق A.
!>          0 <= K <= N-1.
!>

D

!>          D هو مصفوفة ذات دقة مزدوجة، بعد (N)
!>          عناصر القطر للمصفوفة القطرية D.
!>

A

!>          A هي مصفوفة مركبة*16، بعد (LDA,N)
!>          المصفوفة المتماثلة N × N المولدة (يتم تخزين المصفوفة الكاملة).
!>

LDA

!>          LDA هو عدد صحيح
!>          البعد الرائد للمصفوفة A.  LDA >= N.
!>

ISEED

!>          ISEED هي مصفوفة أعداد صحيحة، بعد (4)
!>          عند الدخول، البذرة لمولد الأرقام العشوائية؛ يجب أن تكون عناصر المصفوفة
!>          بين 0 و 4095، ويجب أن تكون ISEED(4)
!>          فردية.
!>          عند الخروج، يتم تحديث البذرة.
!>

WORK

!>          WORK هي مصفوفة مركبة*16، بعد (2*N)
!>

INFO

!>          INFO هو عدد صحيح
!>          = 0: خروج ناجح
!>          < 0: إذا كان INFO = -i، فإن الوسيط i له قيمة غير قانونية
!>
