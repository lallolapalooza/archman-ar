subroutine zlatme (integer n, character dist, integer, dimension( 4 ) iseed, complex*16, dimension( * ) d, integer mode, double precision cond, complex*16 dmax, character rsign, character upper, character sim, double precision, dimension( * ) ds, integer modes, double precision conds, integer kl, integer ku, double precision anorm, complex*16, dimension( lda, * ) a, integer lda, complex*16, dimension( * ) work, integer info)

ZLATME

الغرض:

!>
!>    ZLATME يقوم بإنشاء مصفوفات مربعة عشوائية غير متماثلة بقيم ذاتية
!>    محددة لاختبار برامج LAPACK.
!>
!>    تعمل ZLATME عن طريق تطبيق تسلسل العمليات التالي:
!>
!>    1. ضبط القطر إلى D، حيث يمكن إدخال D أو
!>         حسابه وفقًا لـ MODE و COND و DMAX و RSIGN
!>         كما هو موضح أدناه.
!>
!>    2. إذا كان UPPER='T'، فسيتم تعيين الجزء العلوي من A إلى قيم عشوائية
!>         من التوزيع DIST.
!>
!>    3. إذا كان SIM='T'، فسيتم ضرب A من اليسار بمصفوفة عشوائية
!>         X، وقيمها الذاتية محددة بواسطة DS و MODES و
!>         CONDS، ومن اليمين بمقلوب X.
!>
!>    4. إذا كان KL < N-1، فسيتم تقليل عرض النطاق السفلي إلى KL باستخدام
!>         تحويلات هاوسهولدر. إذا كان KU < N-1، فسيتم تقليل عرض النطاق العلوي
!>         إلى KU.
!>
!>    5. إذا كان ANORM غير سالب، فسيتم قياس المصفوفة ليكون لها
!>         قيمة قصوى للمعيار ANORM.
!>
!>    (ملاحظة: نظرًا لأنه لا يمكن تقليل المصفوفة إلى ما هو أبعد من الشكل الهيسنبرغي،
!>     لا تتوفر خيارات تعبئة.)
!>

المعلمات

N 
!>          N هو عدد صحيح (INTEGER)
!>           عدد الأعمدة (أو الصفوف) للمصفوفة A. لم يتم تعديله.
!>

DIST

!>          DIST هو حرف (CHARACTER*1)
!>           عند الإدخال، يحدد DIST نوع التوزيع الذي سيتم استخدامه
!>           لتوليد القيم الذاتية/القيم المفردة العشوائية، وعلى
!>           المثلث العلوي (انظر UPPER).
!>           'U' => منتظم (UNIFORM) (0, 1) ( 'U' لمنتظم )
!>           'S' => منتظم (UNIFORM) (-1, 1) ( 'S' لمتماثل )
!>           'N' => طبيعي (NORMAL) (0, 1) ( 'N' لطبيعي )
!>           'D' => منتظم على القرص المعقد |z| < 1.
!>           لم يتم تعديله.
!>

ISEED

!>          ISEED هو مصفوفة أعداد صحيحة (INTEGER array)، البعد (4)
!>           عند الإدخال، تحدد ISEED البذرة لمولد الأرقام العشوائية.
!>           يجب أن تكون بين 0 و 4095 شاملة، و ISEED(4) يجب أن تكون فردية.
!>           يستخدم مولد الأرقام العشوائية تسلسلًا خطيًا توافقيًا محدودًا بأعداد
!>           صحيحة صغيرة، وبالتالي يجب أن ينتج أرقامًا عشوائية مستقلة عن الجهاز.
!>           تتغير قيم ISEED عند الخروج، ويمكن استخدامها في الاستدعاء التالي لـ ZLATME
!>           لمواصلة نفس تسلسل الأرقام العشوائية.
!>           تم تعديله عند الخروج.
!>

D

!>          D هي مصفوفة أعداد مركبة (COMPLEX*16 array)، البعد ( N )
!>           تستخدم هذه المصفوفة لتحديد القيم الذاتية للمصفوفة A. إذا
!>           كان MODE=0، فمن المفترض أن تحتوي D على القيم الذاتية
!>           وإلا فسيتم حسابها وفقًا لـ MODE و COND و
!>           DMAX و RSIGN ووضعها في D.
!>           تم تعديله إذا كان MODE غير صفري.
!>

MODE

!>          MODE هو عدد صحيح (INTEGER)
!>           عند الإدخال، يصف هذا كيفية تحديد القيم الذاتية:
!>           MODE = 0 يعني استخدام D كمدخل
!>           MODE = 1 يضبط D(1)=1 و D(2:N)=1.0/COND
!>           MODE = 2 يضبط D(1:N-1)=1 و D(N)=1.0/COND
!>           MODE = 3 يضبط D(I)=COND**(-(I-1)/(N-1))
!>           MODE = 4 يضبط D(i)=1 - (i-1)/(N-1)*(1 - 1/COND)
!>           MODE = 5 يضبط D إلى أرقام عشوائية في النطاق
!>                    ( 1/COND , 1 ) بحيث تكون لوغاريتماتها
!>                    موزعة بشكل منتظم.
!>           MODE = 6 يضبط D إلى أرقام عشوائية من نفس التوزيع
!>                    مثل بقية المصفوفة.
!>           MODE < 0 له نفس المعنى مثل ABS(MODE)، باستثناء أن
!>              ترتيب عناصر D معكوس.
!>           لذلك، إذا كان MODE بين 1 و 4، فإن D يحتوي على مدخلات تتراوح
!>              من 1 إلى 1/COND، إذا كان بين -1 و -4، فإن D يحتوي على مدخلات
!>              تتراوح من 1/COND إلى 1،
!>           لم يتم تعديله.
!>

COND

!>          COND هو عدد حقيقي مزدوج الدقة (DOUBLE PRECISION)
!>           عند الإدخال، يتم استخدامه كما هو موضح تحت MODE أعلاه.
!>           إذا تم استخدامه، يجب أن يكون >= 1. لم يتم تعديله.
!>

DMAX

!>          DMAX هو عدد مركب مزدوج الدقة (COMPLEX*16)
!>           إذا كان MODE ليس -6 أو 0 أو 6، فسيتم قياس محتويات D، كما
!>           تم حسابها وفقًا لـ MODE و COND، بواسطة
!>           DMAX / max(abs(D(i))). لاحظ أنه ليس من الضروري أن يكون DMAX
!>           موجبًا أو حقيقيًا: إذا كان DMAX سالبًا أو مركبًا (أو صفرًا)،
!>           فسيتم قياس D بعدد سالب أو مركب (أو صفر).
!>           إذا كان RSIGN='F'، فإن أكبر قيمة ذاتية (من حيث القيمة المطلقة) ستكون
!>           مساوية لـ DMAX.
!>           لم يتم تعديله.
!>

RSIGN

!>          RSIGN هو حرف (CHARACTER*1)
!>           إذا لم يكن MODE هو 0 أو 6 أو -6، و RSIGN='T'، فسيتم ضرب عناصر D، كما
!>           تم حسابها وفقًا لـ MODE و COND، برقم مركب عشوائي من الدائرة
!>           الوحدة |z| = 1. إذا كان RSIGN='F'، فلن يتم ضربها. يمكن أن يكون RSIGN
!>           فقط القيم 'T' أو 'F'.
!>           لم يتم تعديله.
!>

UPPER

!>          UPPER هو حرف (CHARACTER*1)
!>           إذا كان UPPER='T'، فسيتم تعيين عناصر A فوق القطر
!>           إلى أرقام عشوائية من DIST. إذا كان UPPER='F'،
!>           فلن يتم ذلك. يمكن أن يكون UPPER فقط القيم 'T' أو 'F'.
!>           لم يتم تعديله.
!>

SIM

!>          SIM هو حرف (CHARACTER*1)
!>           إذا كان SIM='T'، فسيتم تشغيل A، أي ضربه من اليسار بمصفوفة X و
!>           من اليمين بمقلوب X. X = U S V، حيث U و V
!>           هما مصفوفتان وحدويتان عشوائيتان و S مصفوفة (قطرية)
!>           من القيم المفردة المحددة بواسطة DS و MODES و
!>           CONDS. إذا كان SIM='F'، فلن يتم تحويل A.
!>           لم يتم تعديله.
!>

DS

!>          DS هي مصفوفة أعداد حقيقية مزدوجة الدقة (DOUBLE PRECISION array)، البعد ( N )
!>           تستخدم هذه المصفوفة لتحديد القيم المفردة لـ X،
!>           بنفس الطريقة التي تحدد بها D القيم الذاتية لـ A.
!>           إذا كان MODE=0، فإن DS تحتوي على القيم المفردة، والتي
!>           يجب ألا تكون صفرًا.
!>           تم تعديله إذا كان MODE غير صفري.
!>

MODES

!>          MODES هو عدد صحيح (INTEGER)
!>

CONDS

!>          CONDS هو عدد حقيقي مزدوج الدقة (DOUBLE PRECISION)
!>           مشابه لـ MODE و COND، ولكن لتحديد القطر لـ S.
!>           MODES=-6 و +6 غير مسموح بهما (لأنهما قد يؤديان إلى
!>           قيم ذاتية ذات حالة سيئة بشكل عشوائي).
!>

KL

!>          KL هو عدد صحيح (INTEGER)
!>           يحدد هذا عرض النطاق السفلي للمصفوفة A. KL=1
!>           يحدد الشكل الهيسنبرغي العلوي. إذا كان KL على الأقل N-1،
!>           فإن A سيكون له عرض نطاق سفلي كامل.
!>           لم يتم تعديله.
!>

KU

!>          KU هو عدد صحيح (INTEGER)
!>           يحدد هذا عرض النطاق العلوي للمصفوفة A. KU=1
!>           يحدد الشكل الهيسنبرغي السفلي. إذا كان KU على الأقل N-1،
!>           فإن A سيكون له عرض نطاق علوي كامل؛ إذا كان KU و KL
!>           كلاهما على الأقل N-1، فإن A سيكون لديه عرض نطاق كامل.
!>           لم يتم تعديله.
!>

ANORM

!>          ANORM هو عدد حقيقي مزدوج الدقة (DOUBLE PRECISION)
!>           إذا كان ANORM >= 0، يتم قياس المصفوفة A بحيث يكون أكبر
!>           عنصر في القيمة المطلقة يساوي ANORM.
!>           لم يتم تعديله.
!>

A

!>          A هي مصفوفة أعداد مركبة مزدوجة الدقة (COMPLEX*16 array)، البعد ( LDA, * )
!>           المصفوفة التي سيتم إنشاؤها.
!>           تم تعديله.
!>

LDA

!>          LDA هو عدد صحيح (INTEGER)
!>           يحدد عرض النطاق الأمامي للمصفوفة A ( LDA).
!>           لم يتم تعديله.
!>

WORK

!>          WORK هي مصفوفة أعداد مركبة مزدوجة الدقة (COMPLEX*16 array)، البعد ( * )
!>           مساحة عمل.
!>

INFO

!>          INFO هو عدد صحيح (INTEGER)
!>           معلومات الخروج.
!>           = 0: صحيح
!>           > 0: خطأ
!>
