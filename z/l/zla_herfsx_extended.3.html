ZLA_HERFSX_EXTENDED تحسن الحل المحسوب لنظام المعادلات الخطية للمصفوفات الهرميتية غير المحددة عن طريق إجراء تنقيح تكراري فائق الدقة وتوفر حدودًا للأخطاء وتقديرات للأخطاء الخلفية للحل.

الغرض:

!>
!> ZLA_HERFSX_EXTENDED يحسن الحل المحسوب لنظام المعادلات الخطية عن طريق إجراء تنقيح تكراري فائق الدقة
!> ويوفر حدودًا للأخطاء وتقديرات للأخطاء الخلفية للحل.
!> يتم استدعاء هذا الروتين الفرعي بواسطة ZHERFSX لإجراء التنقيح التكراري.
!> بالإضافة إلى حد الخطأ المعياري، يقدم الكود حد الخطأ الأقصى للمكونات إذا كان ذلك ممكنًا.
!> انظر التعليقات الخاصة بـ ERR_BNDS_NORM و ERR_BNDS_COMP للحصول على تفاصيل حدود الخطأ.
!> لاحظ أن هذا الروتين الفرعي مسؤول فقط عن تعيين الحقول الثانية لـ
!> ERR_BNDS_NORM و ERR_BNDS_COMP.
!>

المعلمات

PREC_TYPE 
!>          PREC_TYPE هو INTEGER
!>     يحدد الدقة الوسيطة التي سيتم استخدامها في التنقيح.
!>     القيمة معرفة بواسطة ILAPREC(P) حيث P هو CHARACTER و P
!>          = 'S':  Single (أحادي)
!>          = 'D':  Double (مزدوج)
!>          = 'I':  Indigenous (أصلي)
!>          = 'X' or 'E':  Extra (إضافي)
!>

UPLO

!>          UPLO هو CHARACTER*1
!>       = 'U':  يتم تخزين الجزء العلوي من A؛
!>       = 'L':  يتم تخزين الجزء السفلي من A.
!>

N

!>          N هو INTEGER
!>     عدد المعادلات الخطية، أي رتبة
!>     المصفوفة A. N >= 0.
!>

NRHS

!>          NRHS هو INTEGER
!>     عدد الأطراف اليمنى، أي عدد الأعمدة في
!>     المصفوفة B.
!>

A

!>          A هو مصفوفة COMPLEX*16، البعد (LDA,N)
!>     عند الدخول، المصفوفة A بحجم N*N.
!>

LDA

!>          LDA هو INTEGER
!>     البعد الرئيسي للمصفوفة A. LDA >= max(1,N).
!>

AF

!>          AF هو مصفوفة COMPLEX*16، البعد (LDAF,N)
!>     المصفوفة القطرية الجزئية D والمضاعفات المستخدمة
!>     للحصول على العامل U أو L كما تم حسابه بواسطة ZHETRF.
!>

LDAF

!>          LDAF هو INTEGER
!>     البعد الرئيسي للمصفوفة AF. LDAF >= max(1,N).
!>

IPIV

!>          IPIV هي مصفوفة INTEGER، البعد (N)
!>     تفاصيل التبادلات والبنية الجزئية لـ D
!>     كما تم تحديدها بواسطة ZHETRF.
!>

COLEQU

!>          COLEQU هو LOGICAL
!>     إذا كانت .TRUE. فتم إجراء موازنة الأعمدة للمصفوفة A قبل استدعاء
!>     هذا الروتين. هذا ضروري لحساب الحل وحدود الخطأ
!>     بشكل صحيح.
!>

C

!>          C هي مصفوفة DOUBLE PRECISION، البعد (N)
!>     عوامل قياس الأعمدة لـ A. إذا كانت COLEQU = .FALSE.،
!>     فإن C لا يتم الوصول إليها. إذا تم إدخال C، فيجب أن يكون كل عنصر في C قوة
!>     من الأساس لضمان حل موثوق وتقديرات للخطأ.
!>     القياس بقوى الأساس لا يسبب أخطاء تقريب إلا إذا
!>     كانت النتيجة تحت التدفق أو تجاوزته. تؤدي أخطاء التقريب أثناء القياس
!>     إلى التنقيح بمصفوفة غير مكافئة للمصفوفة
!>     المدخلة، مما ينتج عنه تقديرات خطأ قد لا تكون
!>     موثوقة.
!>

B

!>          B هي مصفوفة COMPLEX*16، البعد (LDB,NRHS)
!>     المصفوفة الطرفية اليمنى B.
!>

LDB

!>          LDB هو INTEGER
!>     البعد الرئيسي للمصفوفة B. LDB >= max(1,N).
!>

Y

!>          Y هي مصفوفة COMPLEX*16، البعد (LDY,NRHS)
!>     عند الدخول، مصفوفة الحل X، كما تم حسابها بواسطة ZHETRS.
!>     عند الخروج، مصفوفة الحل المحسنة Y.
!>

LDY

!>          LDY هو INTEGER
!>     البعد الرئيسي للمصفوفة Y. LDY >= max(1,N).
!>

BERR_OUT

!>          BERR_OUT هي مصفوفة DOUBLE PRECISION، البعد (NRHS)
!>     عند الخروج، تحتوي BERR_OUT(j) على الخطأ الخلفي النسبي للمكونات
!>     للطرف الأيمن j من الصيغة
!>         max(i) ( abs(RES(i)) / ( abs(op(A_s))*abs(Y) + abs(B_s) )(i) )
!>     حيث abs(Z) هي القيمة المطلقة المكونة للمصفوفة
!>     أو المتجه Z. يتم حساب هذا بواسطة ZLA_LIN_BERR.
!>

N_NORMS

!>          N_NORMS هو INTEGER
!>     يحدد حدود الخطأ التي سيتم إرجاعها (انظر ERR_BNDS_NORM
!>     و ERR_BNDS_COMP).
!>     إذا كانت N_NORMS >= 1، قم بإرجاع حدود الخطأ المعيارية.
!>     إذا كانت N_NORMS >= 2، قم بإرجاع حدود الخطأ للمكونات.
!>

ERR_BNDS_NORM

!>          ERR_BNDS_NORM هي مصفوفة DOUBLE PRECISION، البعد (NRHS, N_ERR_BNDS)
!>     لكل طرف أيمن، تحتوي هذه المصفوفة على معلومات حول
!>     حدود الخطأ المختلفة وأرقام الشروط المقابلة لـ
!>     الخطأ المعياري النسبي، والذي يتم تعريفه على النحو التالي:
!>
!>     الخطأ المعياري النسبي في متجه الحل i:
!>             max_j (abs(XTRUE(j,i) - X(j,i)))
!>            ------------------------------
!>                  max_j abs(X(j,i))
!>
!>     يتم فهرسة المصفوفة حسب نوع معلومات الخطأ كما هو موضح
!>     أدناه. يوجد حاليًا ما يصل إلى ثلاث قطع من المعلومات
!>     التي يتم إرجاعها.
!>
!>     الفهرس الأول في ERR_BNDS_NORM(i,:) يتوافق مع الطرف الأيمن i.
!>
!>     الفهرس الثاني في ERR_BNDS_NORM(:,err) يحتوي على المجالات الثلاثة التالية:
!>     err = 1  boolean. ثق بالإجابة إذا كان
!>              رقم الشرط المقلوب أقل من الحد
!>              sqrt(n) * slam
