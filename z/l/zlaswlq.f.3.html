توثيق الوظيفة/الإجراء الفرعي

subroutine zlaswlq (integer m, integer n, integer mb, integer nb, complex*16, dimension( lda, * ) a, integer lda, complex*16, dimension( ldt, * ) t, integer ldt, complex*16, dimension( * ) work, integer lwork, integer info)

ZLASWLQ

الغرض:

!>
!> ZLASWLQ يحسب تحلل LQ المكدس الضيق (Tall-Skinny LQ factorization) للمصفوفة المعقدة A ذات الأبعاد M × N حيث M <= N:
!>
!>    A = ( L 0 ) *  Q,
!>
!> حيث:
!>
!>    Q هي مصفوفة متعامدة N × N، مخزنة عند الخروج في شكل ضمني
!>    فوق القطر في الصفيف A والعناصر في الصفيف T؛
!>    L هي مصفوفة مثلثية سفلية M × M مخزنة عند الخروج في
!>    العناصر على وتحت القطر في الصفيف A.
!>    0 هي مصفوفة صفرية M × (N-M)، إذا كان M < N، ولا يتم تخزينها.
!>
!>

المعلمات:

M 
!>          M هو عدد صحيح (INTEGER)
!>          عدد صفوف المصفوفة A.  M >= 0.
!>

N

!>          N هو عدد صحيح (INTEGER)
!>          عدد أعمدة المصفوفة A.  N >= M >= 0.
!>

MB

!>          MB هو عدد صحيح (INTEGER)
!>          حجم الكتلة الصفية المستخدم في التحلل المكدس (blocked factorization).
!>          M >= MB >= 1
!>

NB

!>          NB هو عدد صحيح (INTEGER)
!>          حجم الكتلة العمودية المستخدم في التحلل المكدس (blocked factorization).
!>          NB > 0.
!>

A

!>          A هو صفيف الأعداد المركبة * 16 (COMPLEX*16)، الأبعاد (LDA,N)
!>          عند الدخول، المصفوفة A ذات الأبعاد M × N.
!>          عند الخروج، تحتوي العناصر على وتحت القطر
!>          للمصفوفة على المصفوفة المثلثية السفلية L ذات الأبعاد N × N؛
!>          العناصر فوق القطر تمثل Q بواسطة صفوف
!>          V المكدسة (انظر التفاصيل الإضافية).
!>
!>

LDA

!>          LDA هو عدد صحيح (INTEGER)
!>          البعد الرائد لـ الصفيف A.  LDA >= max(1,M).
!>

T

!>          T هو صفيف الأعداد المركبة * 16 (COMPLEX*16)،
!>          الأبعاد (LDT, N * Number_of_row_blocks)
!>          حيث Number_of_row_blocks = CEIL((N-M)/(NB-M))
!>          المعكوسات المكدسة المثلثية العلوية المخزنة بشكل مضغوط
!>          كسلسلة من الكتل المثلثية العلوية.
!>          انظر التفاصيل الإضافية أدناه.
!>

LDT

!>          LDT هو عدد صحيح (INTEGER)
!>          البعد الرائد لـ الصفيف T.  LDT >= MB.
!>

WORK

!>          (مساحة عمل) صفيف الأعداد المركبة * 16 (COMPLEX*16)، الأبعاد (MAX(1,LWORK))
!>          عند الخروج، إذا كان INFO = 0، فإن WORK(1) ترجع الحد الأدنى LWORK.
!>

LWORK

!>          LWORK هو عدد صحيح (INTEGER)
!>          بعد الصفيف WORK.
!>          LWORK >= 1، إذا كان MIN(M,N) = 0، و LWORK >= MB*M، بخلاف ذلك.
!>
!>          إذا كان LWORK = -1، فسيتم افتراض استعلام مساحة العمل؛ يقوم الإجراء الفرعي
!>          بحساب الحد الأدنى لحجم الصفيف WORK فقط، ويرجع
!>          هذه القيمة كمدخل أول للصفيف WORK، ولا يصدر أي رسالة خطأ
!>          متعلقة بـ LWORK بواسطة XERBLA.
!>

INFO

!>          INFO هو عدد صحيح (INTEGER)
!>          = 0:  انتهاء ناجح
!>          < 0:  إذا كان INFO = -i، فإن الوسيط i كان له قيمة غير قانونية
!>

تفاصيل إضافية:

!> يقوم تحليل LQ الضيق الطويل (Short-Wide LQ - SWLQ) بإجراء LQ بواسطة سلسلة من التحويلات المتعامدة،
!> تمثل Q كحاصل ضرب لمصفوفات متعامدة أخرى
!>   Q = Q(1) * Q(2) * . . . * Q(k)
!> حيث تقوم كل Q(i) بتصفير إدخالات القطر العلوي لكتلة من M صفوف A:
!>   Q(1) تقوم بتصفير إدخالات القطر العلوي للصفوف 1:MB من A
!>   Q(2) تقوم بتصفير الصفوف السفلية MB-N من الصفوف [1:M,NB+1:2*NB-M] من A
!>   Q(3) تقوم بتصفير الصفوف السفلية MB-N من الصفوف [1:M,2*NB-M+1:3*NB-2*M] من A
!>   . . .
!>
!> يتم حساب Q(1) بواسطة GELQT، والذي يمثل Q(1) بواسطة متجهات هاوسهولدر
!> المخزنة تحت القطر لصفوف 1:MB من A، وبواسطة معكوسات مكدسة
!> مثلثة علوية، مخزنة في الصفيف T(1:LDT,1:N).
!> لمزيد من المعلومات انظر التفاصيل الإضافية في GELQT.
!>
!> Q(i) لـ i>1 يتم حسابها بواسطة TPLQT، والذي يمثل Q(i) بواسطة متجهات هاوسهولدر
!> المخزنة في الأعمدة [(i-1)*(NB-M)+M+1:i*(NB-M)+M] من A، وبواسطة معكوسات مكدسة
!> مثلثة علوية، مخزنة في الصفيف T(1:LDT,(i-1)*M+1:i*M).
!> قد تستخدم Q(k) الأخيرة عددًا أقل من الصفوف.
!> لمزيد من المعلومات انظر التفاصيل الإضافية في TPQRT.
!>
!> لمزيد من التفاصيل حول الخوارزمية الشاملة، انظر وصف
!> TSQR المتسلسل في القسم 2.2 من [1].
!>
!> [1] "تحليلات QR و LU المتوازية والمتسلسلة المثلى للاتصال،"
!>     J. Demmel، L. Grigori، M. Hoemmen، J. Langou،
!>     SIAM J. Sci. Comput، المجلد 34، العدد 1، 2012
!>
