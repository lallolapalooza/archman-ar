توثيق الوظيفة/الإجراء الفرعي

subroutine zla_gbamv (integer trans, integer m, integer n, integer kl, integer ku, double precision alpha, complex*16, dimension( ldab, * ) ab, integer ldab, complex*16, dimension( * ) x, integer incx, double precision beta, double precision, dimension( * ) y, integer incy)

ZLA_GBAMV تنفذ عملية مصفوفة-متجه لحساب حدود الخطأ.

الغرض:

!>
!> ZLA_GBAMV  تنفذ إحدى عمليات المصفوفة-المتجه
!>
!>         y := alpha*abs(A)*abs(x) + beta*abs(y),
!>    أو   y := alpha*abs(A)**T*abs(x) + beta*abs(y),
!>
!> حيث alpha و beta هما عددان قياسيان، x و y متجهان، و A مصفوفة
!> حجمها m في n.
!>
!> هذه الوظيفة تستخدم بشكل أساسي في حساب حدود الخطأ.
!> للحماية ضد التدفق السفلي أثناء التقييم، يتم إزاحة المكونات في
!> المتجه الناتج عن (N+1) مرة من حد التدفق السفلي. لمنع الأخطاء غير الضرورية
!> للمباني المضمنة في المصفوفات العامة،
!> لا يتم إزاحة المكونات الصفرية. يعتبر الإدخال الصفري
!> إذا كان جميع عمليات الضرب المتضمنة
!> في حساب ذلك الإدخال تحتوي على عامل ضرب صفري واحد على الأقل.
!>

المعلمات

TRANS 
!>          TRANS هو INTEGER
!>           عند الإدخال، يحدد TRANS العملية المراد تنفيذها كما يلي:
!>
!>             BLAS_NO_TRANS      y := alpha*abs(A)*abs(x) + beta*abs(y)
!>             BLAS_TRANS         y := alpha*abs(A**T)*abs(x) + beta*abs(y)
!>             BLAS_CONJ_TRANS    y := alpha*abs(A**T)*abs(x) + beta*abs(y)
!>
!>           لا تتغير عند الخروج.
!>

M

!>          M هو INTEGER
!>           عند الإدخال، يحدد M عدد صفوف المصفوفة A.
!>           يجب أن يكون M صفراً على الأقل.
!>           لا تتغير عند الخروج.
!>

N

!>          N هو INTEGER
!>           عند الإدخال، يحدد N عدد أعمدة المصفوفة A.
!>           يجب أن يكون N صفراً على الأقل.
!>           لا تتغير عند الخروج.
!>

KL

!>          KL هو INTEGER
!>           عدد الأقطار الفرعية داخل نطاق A. KL >= 0.
!>

KU

!>          KU هو INTEGER
!>           عدد الأقطار فوق الرئيسية داخل نطاق A. KU >= 0.
!>

ALPHA

!>          ALPHA هو DOUBLE PRECISION
!>           عند الإدخال، يحدد ALPHA العدد القياسي alpha.
!>           لا يتغير عند الخروج.
!>

AB

!>          AB هو مصفوفة COMPLEX*16، أبعادها ( LDAB, n )
!>           قبل الدخول، يجب أن يحتوي الجزء الرائد m في n من المصفوفة AB
!>           على مصفوفة المعاملات.
!>           لا تتغير عند الخروج.
!>

LDAB

!>          LDAB هو INTEGER
!>           عند الإدخال، يحدد LDAB البعد الأول لـ AB كما تم تعريفه
!>           في البرنامج (الفرعي) المستدعي. يجب أن يكون LDAB على الأقل
!>           max( 1, m ).
!>           لا يتغير عند الخروج.
!>

X

!>          X هو مصفوفة COMPLEX*16، أبعادها
!>           ( 1 + ( n - 1 )*abs( INCX ) ) عندما TRANS = 'N' أو 'n'
!>           وعلى الأقل
!>           ( 1 + ( m - 1 )*abs( INCX ) ) بخلاف ذلك.
!>           قبل الدخول، يجب أن تحتوي المصفوفة X ذات الزيادات على
!>           المتجه x.
!>           لا تتغير عند الخروج.
!>

INCX

!>          INCX هو INTEGER
!>           عند الإدخال، يحدد INCX الزيادة للعناصر في
!>           X. يجب ألا يكون INCX صفراً.
!>           لا يتغير عند الخروج.
!>

BETA

!>          BETA هو DOUBLE PRECISION
!>           عند الإدخال، يحدد BETA العدد القياسي beta. عندما يكون BETA
!>           مقدماً كصفر، لا يلزم تعيين Y عند الإدخال.
!>           لا يتغير عند الخروج.
!>

Y

!>          Y هو مصفوفة DOUBLE PRECISION، أبعادها
!>           ( 1 + ( m - 1 )*abs( INCY ) ) عندما TRANS = 'N' أو 'n'
!>           وعلى الأقل
!>           ( 1 + ( n - 1 )*abs( INCY ) ) بخلاف ذلك.
!>           قبل الدخول مع BETA غير صفري، يجب أن تحتوي المصفوفة Y ذات الزيادات
!>           على المتجه y. عند الخروج، يتم الكتابة فوق Y بالمتجه y
!>           المحدث.
!>           إذا كان m أو n صفراً، فلن يتم الرجوع إلى Y وستقوم الوظيفة
!>           بالعودة بسرعة.
!>

INCY

!>          INCY هو INTEGER
!>           عند الإدخال، يحدد INCY الزيادة للعناصر في
!>           Y. يجب ألا يكون INCY صفراً.
!>           لا يتغير عند الخروج.
!>
!>  إجراء BLAS من المستوى 2.
!>
