subroutine zlaqr2 (logical wantt, logical wantz, integer n, integer ktop, integer kbot, integer nw, complex*16, dimension( ldh, * ) h, integer ldh, integer iloz, integer ihiz, complex*16, dimension( ldz, * ) z, integer ldz, integer ns, integer nd, complex*16, dimension( * ) sh, complex*16, dimension( ldv, * ) v, integer ldv, integer nh, complex*16, dimension( ldt, * ) t, integer ldt, integer nv, complex*16, dimension( ldwv, * ) wv, integer ldwv, complex*16, dimension( * ) work, integer lwork)

ZLAQR2 تقوم بتحويل التشابه الوحدوي لمصفوفة هسنبرغ للكشف عن القيم الذاتية المتقاربة بالكامل وتفريغها من مصفوفة فرعية رئيسية لاحقة (تفريغ مبكر شره).

الغرض:

!>
!>    ZLAQR2 مطابق لـ ZLAQR3 باستثناء أنه يتجنب
!>    الاستدعاء التكراري عن طريق استدعاء ZLAHQR بدلاً من ZLAQR4.
!>
!>    التفريغ المبكر الشره:
!>
!>    يقبل ZLAQR2 كمصفوفة هسنبرغ عليا مدخلة
!>    H ويقوم بتحويل تشابه وحدوي
!>    مصمم للكشف عن القيم الذاتية المتقاربة بالكامل وتفريغها من
!>    مصفوفة فرعية رئيسية لاحقة. على المخرجات، تم استبدال H
!>    بمصفوفة هسنبرغ جديدة هي اضطراب لتحويل تشابه وحدوي لـ H.
!>    من المأمول أن يكون الإصدار النهائي لـ H يحتوي على العديد من الإدخالات الفرعية القطرية الصفرية.
!>
!>

المعلمات

WANTT 
!>          WANTT هو LOGICAL
!>          إذا كان .TRUE.، فسيتم تحديث مصفوفة هسنبرغ H بالكامل
!>          بحيث يمكن حساب عامل Schur المثلثي (بالتعاون مع الروتين الفرعي المتصل).
!>          إذا كان .FALSE.، فلن يتم تحديث سوى ما يكفي من H للحفاظ على
!>          القيم الذاتية.
!>

WANTZ

!>          WANTZ هو LOGICAL
!>          إذا كان .TRUE.، فسيتم تحديث المصفوفة الوحدوية Z
!>          بحيث يمكن حساب عامل Schur الوحدوي (بالتعاون مع الروتين الفرعي المتصل).
!>          إذا كان .FALSE.، فلن يتم استخدام Z.
!>

N

!>          N هو INTEGER
!>          رتبة المصفوفة H (وإذا كان WANTZ هو .TRUE.) رتبة المصفوفة الوحدوية Z.
!>

KTOP

!>          KTOP هو INTEGER
!>          من المفترض إما أن KTOP = 1 أو H(KTOP,KTOP-1)=0.
!>          يحدد KBOT و KTOP معًا كتلة معزولة
!>          على طول قطر مصفوفة هسنبرغ.
!>

KBOT

!>          KBOT هو INTEGER
!>          من المفترض دون تحقق إما
!>          KBOT = N أو H(KBOT+1,KBOT)=0. يحدد KBOT و KTOP معًا
!>          كتلة معزولة على طول قطر
!>          مصفوفة هسنبرغ.
!>

NW

!>          NW هو INTEGER
!>          حجم نافذة التفريغ. 1 <= NW <= (KBOT-KTOP+1).
!>

H

!>          H هو مصفوفة COMPLEX*16، ذات البعد (LDH,N)
!>          عند الإدخال، القسم N × N الأولي من H يخزن
!>          مصفوفة هسنبرغ التي تخضع للتفريغ المبكر الشره.
!>          عند المخرجات، تم تحويل H بواسطة تشابه وحدوي
!>          ، تم اضطرابه، وأعيد
!>          إلى شكل هسنبرغ الذي (يؤمل أن يحتوي على بعض
!>          الإدخالات الفرعية القطرية الصفرية.
!>

LDH

!>          LDH هو INTEGER
!>          البعد الرائد لـ H كما هو معلن في الروتين الفرعي المتصل.
!>          N <= LDH
!>

ILOZ

!>          ILOZ هو INTEGER
!>

IHIZ

!>          IHIZ هو INTEGER
!>          حدد صفوف Z التي يجب تطبيق التحويلات عليها
!>          إذا كان WANTZ هو .TRUE.. 1 <= ILOZ <= IHIZ <= N.
!>

Z

!>          Z هو مصفوفة COMPLEX*16، ذات البعد (LDZ,N)
!>          إذا كان WANTZ هو .TRUE.، فعند المخرجات، تم
!>          تجميع تحويل التشابه الوحدوي المذكور أعلاه
!>          في Z(ILOZ:IHIZ,ILOZ:IHIZ) من اليمين.
!>          إذا كان WANTZ هو .FALSE.، فلن يتم استخدام Z.
!>

LDZ

!>          LDZ هو INTEGER
!>          البعد الرائد لـ Z كما هو معلن في
!>          الروتين الفرعي المتصل. 1 <= LDZ.
!>

NS

!>          NS هو INTEGER
!>          عدد القيم الذاتية غير المتقاربة (أي التقريبية)
!>          التي تم إرجاعها في SR و SI والتي يمكن استخدامها كتحولات بواسطة
!>          الروتين الفرعي المتصل.
!>

ND

!>          ND هو INTEGER
!>          عدد القيم الذاتية المتقاربة التي كشفت عنها هذه
!>          الروتين الفرعي.
!>

SH

!>          SH هو مصفوفة COMPLEX*16، ذات البعد (KBOT)
!>          عند المخرجات، القيم الذاتية التقريبية التي قد
!>          تُستخدم للتحولات مخزنة في SH(KBOT-ND-NS+1)
!>          حتى SR(KBOT-ND). القيم الذاتية المتقاربة
!>          مخزنة في SH(KBOT-ND+1) حتى SH(KBOT).
!>

V

!>          V هو مصفوفة COMPLEX*16، ذات البعد (LDV,NW)
!>          مصفوفة عمل بحجم NW × NW.
!>

LDV

!>          LDV هو INTEGER
!>          البعد الرائد لـ V كما هو معلن في
!>          الروتين الفرعي المتصل. NW <= LDV
!>

NH

!>          NH هو INTEGER
!>          عدد أعمدة T. NH >= NW.
!>

T

!>          T هو مصفوفة COMPLEX*16، ذات البعد (LDT,NW)
!>

LDT

!>          LDT هو INTEGER
!>          البعد الرائد لـ T كما هو معلن في
!>          الروتين الفرعي المتصل. NW <= LDT
!>

NV

!>          NV هو INTEGER
!>          عدد صفوف مصفوفة العمل WV المتاحة
!>          للمساحة. NV >= NW.
!>

WV

!>          WV هو مصفوفة COMPLEX*16، ذات البعد (LDWV,NW)
!>

LDWV

!>          LDWV هو INTEGER
!>          البعد الرائد لـ W كما هو معلن في
!>          الروتين الفرعي المتصل. NW <= LDV
!>

WORK

!>          WORK هو مصفوفة COMPLEX*16، ذات البعد (LWORK)
!>          عند المخرجات، يتم تعيين WORK(1) لتقدير القيمة المثلى
!>          لـ LWORK للقيم المعطاة لـ N، NW، KTOP، و KBOT.
!>

LWORK

!>          LWORK هو INTEGER
!>          بعد مصفوفة العمل WORK. LWORK = 2*NW
!>          كافٍ، ولكن قد ينتج عن قيم أكبر لـ LWORK كفاءة أكبر.
!>
!>          إذا كان LWORK = -1، فيُفترض استعلام عن مساحة العمل؛ ZLAQR2
!>          يقدر فقط حجم مساحة العمل المثلى للقيم المعطاة
!>          لـ N، NW، KTOP، و KBOT. يتم إرجاع التقدير
!>          في WORK(1). لا يتم إصدار أي رسالة خطأ متعلقة بـ LWORK
!>          بواسطة XERBLA. لا يتم الوصول إلى H ولا Z.
!>
