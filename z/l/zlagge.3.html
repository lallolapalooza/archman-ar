توثيق الوظيفة/الروتين الفرعي

subroutine zlagge (integer m, integer n, integer kl, integer ku, double precision, dimension( * ) d, complex*16, dimension( lda, * ) a, integer lda, integer, dimension( 4 ) iseed, complex*16, dimension( * ) work, integer info)

ZLAGGE

الغرض:

!>
!> ZLAGGE تولد مصفوفة معقدة عامة A بحجم m x n، عن طريق الضرب المسبق واللاحق
!> لمصفوفة قطرية حقيقية D بمصفوفات وحدوية عشوائية:
!> A = U*D*V. قد يتم تقليل النطاق السفلي والعليا بعد ذلك إلى
!> kl و ku عن طريق تحويلات وحدوية إضافية.
!>

المعلمات

M 
!>          M هو عدد صحيح (INTEGER)
!>          عدد صفوف المصفوفة A.  M >= 0.
!>

N

!>          N هو عدد صحيح (INTEGER)
!>          عدد أعمدة المصفوفة A.  N >= 0.
!>

KL

!>          KL هو عدد صحيح (INTEGER)
!>          عدد الأقطار الفرعية غير الصفرية داخل نطاق A.
!>          0 <= KL <= M-1.
!>

KU

!>          KU هو عدد صحيح (INTEGER)
!>          عدد الأقطار الفرعية غير الصفرية داخل نطاق A.
!>          0 <= KU <= N-1.
!>

D

!>          D هو مصفوفة من نوع double precision، البعد (min(M,N))
!>          العناصر القطرية للمصفوفة القطرية D.
!>

A

!>          A هو مصفوفة من نوع complex*16، البعد (LDA,N)
!>          المصفوفة المعقدة A التي تم إنشاؤها بحجم m x n.
!>

LDA

!>          LDA هو عدد صحيح (INTEGER)
!>          البعد الرائد للمصفوفة A.  LDA >= M.
!>

ISEED

!>          ISEED هي مصفوفة أعداد صحيحة (INTEGER array)، البعد (4)
!>          عند الإدخال، بذرة مولد الأرقام العشوائية؛ يجب أن تكون عناصر المصفوفة
!>          بين 0 و 4095، ويجب أن يكون ISEED(4) فرديًا.
!>          عند الخروج، يتم تحديث البذرة.
!>

WORK

!>          WORK هي مصفوفة من نوع complex*16، البعد (M+N)
!>

INFO

!>          INFO هو عدد صحيح (INTEGER)
!>          = 0: خروج ناجح
!>          < 0: إذا كان INFO = -i، فإن المعلمة i كانت ذات قيمة غير قانونية
!>
