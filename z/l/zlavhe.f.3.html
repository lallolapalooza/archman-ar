subroutine zlavhe (character uplo, character trans, character diag, integer n, integer nrhs, complex*16, dimension( lda, * ) a, integer lda, integer, dimension( * ) ipiv, complex*16, dimension( ldb, * ) b, integer ldb, integer info)

ZLAVHE

الغرض:

!>
!> ZLAVHE يقوم بإحدى عمليات المصفوفة والمتجه
!>    x := A*x  أو  x := A^H*x,
!> حيث x هو متجه ذو N عنصر و A هي إحدى العوامل
!> من التحليل U*D*U' أو L*D*L' التي حسبها ZHETRF.
!>
!> إذا كان TRANS = 'N'، الضرب في U  أو U * D  (أو L  أو L * D)
!> إذا كان TRANS = 'C'، الضرب في U' أو D * U' (أو L' أو D * L')
!>

المعلمات

UPLO 
!>          UPLO هو CHARACTER*1
!>          يحدد ما إذا كان العامل المخزن في A علويًا أم سفليًا
!>          مثلثيًا.
!>          = 'U':  مثلثي علوي
!>          = 'L':  مثلثي سفلي
!>

TRANS

!>          TRANS هو CHARACTER*1
!>          يحدد العملية التي سيتم تنفيذها:
!>          = 'N':  x := A*x
!>          = 'C':  x := A'*x
!>

DIAG

!>          DIAG هو CHARACTER*1
!>          يحدد ما إذا كانت كتل القطر هي مصفوفات وحدة أم لا.
!>          إذا كان من المفترض أن تكون كتل القطر مصفوفات وحدة،
!>          فإن A = U أو A = L، وإلا فإن A = U*D أو A = L*D.
!>          = 'U':  يُفترض أن تكون كتل القطر مصفوفات وحدة.
!>          = 'N':  يُفترض أن تكون كتل القطر مصفوفات غير وحدة.
!>

N

!>          N هو INTEGER
!>          عدد الصفوف والأعمدة للمصفوفة A. N >= 0.
!>

NRHS

!>          NRHS هو INTEGER
!>          عدد الأطراف اليمنى، أي عدد المتجهات
!>          x التي سيتم ضربها في A. NRHS >= 0.
!>

A

!>          A هو مصفوفة COMPLEX*16، البعد (LDA,N)
!>          الكتلة القطرية D والمعاملات المستخدمة
!>          للحصول على العامل U أو L كما حسبها ZHETRF.
!>          مخزنة كمصفوفة مثلثية ثنائية الأبعاد.
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرائد للمصفوفة A. LDA >= max(1,N).
!>

IPIV

!>          IPIV هو مصفوفة INTEGER، البعد (N)
!>          تفاصيل التبادلات وهيكل الكتلة لـ D،
!>          كما حددها ZHETRF.
!>
!>          إذا كان UPLO = 'U':
!>               إذا كان IPIV(k) > 0، فتم تبديل الصفوف والأعمدة k و IPIV(k)
!>               وكان D(k,k) كتلة قطرية بحجم 1x1.
!>               (إذا كان IPIV( k ) = k، فلم يتم إجراء تبديل).
!>
!>               إذا كان IPIV(k) = IPIV(k-1) < 0، فتم تبديل الصفوف
!>               والأعمدة k-1 و -IPIV(k)،
!>               وكان D(k-1:k,k-1:k) كتلة قطرية بحجم 2x2.
!>
!>          إذا كان UPLO = 'L':
!>               إذا كان IPIV(k) > 0، فتم تبديل الصفوف والأعمدة k و IPIV(k)
!>               وكان D(k,k) كتلة قطرية بحجم 1x1.
!>               (إذا كان IPIV( k ) = k، فلم يتم إجراء تبديل).
!>
!>               إذا كان IPIV(k) = IPIV(k+1) < 0، فتم تبديل الصفوف
!>               والأعمدة k+1 و -IPIV(k)،
!>               وكان D(k:k+1,k:k+1) كتلة قطرية بحجم 2x2.
!>

B

!>          B هو مصفوفة COMPLEX*16، البعد (LDB,NRHS)
!>          عند الإدخال، تحتوي B على NRHS متجهات بطول N.
!>          عند الخروج، يتم استبدال B بمنتج A * B.
!>

LDB

!>          LDB هو INTEGER
!>          البعد الرائد للمصفوفة B. LDB >= max(1,N).
!>

INFO

!>          INFO هو INTEGER
!>          = 0: خروج ناجح
!>          < 0: إذا كان INFO = -k، كان المعامل k ذو قيمة غير قانونية
!>
