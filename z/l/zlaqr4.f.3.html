subroutine zlaqr4 (logical wantt, logical wantz, integer n, integer ilo, integer ihi, complex*16, dimension( ldh, * ) h, integer ldh, complex*16, dimension( * ) w, integer iloz, integer ihiz, complex*16, dimension( ldz, * ) z, integer ldz, complex*16, dimension( * ) work, integer lwork, integer info)

ZLAQR4 تحسب القيم الذاتية لمصفوفة هسنبرغ، وبشكل اختياري، المصفوفات من تحليل شتور.

الغرض:

!>
!>    ZLAQR4 تنفذ مستوى واحد من العودية لـ ZLAQR0.
!>    إنها تنفيذ كامل لخوارزمية QR متعددة التحولات ذات الفقاعة الصغيرة.
!>    قد يتم استدعاؤها بواسطة ZLAQR0، ولحجم نافذة إزالة كافٍ، قد يتم استدعاؤها
!>    بواسطة ZLAQR3. هذا الإجراء الفرعي مطابق لـ ZLAQR0 باستثناء أنه يستدعي
!>    ZLAQR2 بدلاً من ZLAQR3.
!>
!>    ZLAQR4 تحسب القيم الذاتية لمصفوفة هسنبرغ H، وبشكل اختياري،
!>    المصفوفتين T و Z من تحليل شتور H = Z T Z**H، حيث T هي مصفوفة
!>    مثلثة عليا (شكل شتور)، و Z هي مصفوفة وحدوية من متجهات شتور.
!>
!>    بشكل اختياري، يمكن ضرب Z في المصفوفة الوحدوية المدخلة Q بحيث يمكن لهذا
!>    الإجراء الفرعي إعطاء تحليل شتور لمصفوفة A التي تم اختزالها إلى شكل هسنبرغ H
!>    بواسطة المصفوفة الوحدوية Q: A = Q*H*Q**H = (QZ)*H*(QZ)**H.
!>

المعلمات

WANTT 
!>          WANTT هو LOGICAL
!>          = .TRUE. : شكل شتور الكامل T مطلوب؛
!>          = .FALSE.: القيم الذاتية فقط مطلوبة.
!>

WANTZ

!>          WANTZ هو LOGICAL
!>          = .TRUE. : مصفوفة متجهات شتور Z مطلوبة؛
!>          = .FALSE.: متجهات شتور غير مطلوبة.
!>

N

!>          N هو INTEGER
!>           ترتيب المصفوفة H. N >= 0.
!>

ILO

!>          ILO هو INTEGER
!>

IHI

!>          IHI هو INTEGER
!>           من المفترض أن H بالفعل مثلثة عليا في الصفوف والأعمدة 1:ILO-1 و
!>           IHI+1:N، وإذا كان ILO > 1، فإن H(ILO,ILO-1) يساوي صفرًا.
!>           عادةً ما يتم تعيين ILO و IHI بواسطة استدعاء سابق لـ ZGEBAL، ثم
!>           يتم تمريرهما إلى ZGEHRD عند اختزال المصفوفة الناتجة عن ZGEBAL
!>           إلى شكل هسنبرغ. بخلاف ذلك، يجب تعيين ILO و IHI إلى 1 و N، على
!>           التوالي. إذا كان N > 0، فإن 1 <= ILO <= IHI <= N.
!>           إذا كان N = 0، فإن ILO = 1 و IHI = 0.
!>

H

!>          H هو مصفوفة COMPLEX*16، البعد (LDH,N)
!>           عند الدخول، مصفوفة هسنبرغ العليا H.
!>           عند الخروج، إذا كان INFO = 0 و WANTT هو .TRUE.، فإن H تحتوي
!>           على المصفوفة المثلثية العليا T من تحليل شتور (شكل شتور).
!>           إذا كان INFO = 0 و WANT هو .FALSE.، فإن محتويات H غير محددة
!>           عند الخروج. (قيمة الخرج لـ H عندما يكون INFO > 0 معطاة تحت
!>           وصف INFO أدناه.)
!>
!>           قد يقوم هذا الإجراء الفرعي صراحةً بتعيين H(i,j) = 0 لـ i > j و
!>           j = 1, 2, ... ILO-1 أو j = IHI+1, IHI+2, ... N.
!>

LDH

!>          LDH هو INTEGER
!>           البعد الرئيسي للمصفوفة H. LDH >= max(1,N).
!>

W

!>          W هو مصفوفة COMPLEX*16، البعد (N)
!>           يتم تخزين القيم الذاتية المحسوبة لـ H(ILO:IHI,ILO:IHI)
!>           في W(ILO:IHI). إذا كان WANTT هو .TRUE.، فسيتم تخزين القيم
!>           الذاتية بنفس ترتيب الأقطار في شكل شتور المعادة في H، مع
!>           W(i) = H(i,i).
!>

ILOZ

!>          ILOZ هو INTEGER
!>

IHIZ

!>          IHIZ هو INTEGER
!>           تحديد الصفوف في Z التي يجب تطبيق التحويلات عليها إذا كان WANTZ
!>           هو .TRUE..
!>           1 <= ILOZ <= ILO; IHI <= IHIZ <= N.
!>

Z

!>          Z هو مصفوفة COMPLEX*16، البعد (LDZ,IHI)
!>           إذا كان WANTZ هو .FALSE.، فلا يتم الاستفادة من Z.
!>           إذا كان WANTZ هو .TRUE.، فسيتم استبدال Z(ILO:IHI,ILOZ:IHIZ)
!>           بـ Z(ILO:IHI,ILOZ:IHIZ)*U حيث U هو عامل شتور الوحدوي
!>           لـ H(ILO:IHI,ILO:IHI).
!>           (قيمة الخرج لـ Z عندما يكون INFO > 0 معطاة تحت وصف INFO أدناه.)
!>

LDZ

!>          LDZ هو INTEGER
!>           البعد الرئيسي للمصفوفة Z. إذا كان WANTZ هو .TRUE.
!>           فإن LDZ >= MAX(1,IHIZ). بخلاف ذلك، LDZ >= 1.
!>

WORK

!>          WORK هو مصفوفة COMPLEX*16، البعد LWORK
!>           عند الخروج، إذا كان LWORK = -1، فإن WORK(1) يعيد تقديرًا
!>           للقيمة المثلى لـ LWORK.
!>

LWORK

!>          LWORK هو INTEGER
!>           بعد المصفوفة WORK. LWORK >= max(1,N) كافٍ، ولكن LWORK
!>           الذي يكون كبيرًا بقدر 6*N قد يكون مطلوبًا للحصول على أداء
!>           مثلى. يوصى باستعلام مساحة العمل لتحديد حجم مساحة العمل
!>           المثلى.
!>
!>           إذا كان LWORK = -1، فإن ZLAQR4 يقوم باستعلام مساحة العمل.
!>           في هذه الحالة، يتحقق ZLAQR4 من معلمات الإدخال ويقدر حجم مساحة
!>           العمل المثلى للقيم المعطاة لـ N، ILO، و IHI. يتم إرجاع
!>           التقدير في WORK(1). لا يتم إصدار رسالة خطأ متعلقة بـ LWORK
!>           بواسطة XERBLA. لا يتم الوصول إلى H أو Z.
!>

INFO

!>          INFO هو INTEGER
!>             =  0:  خروج ناجح
!>             > 0:  إذا كان INFO = i، فشل ZLAQR4 في حساب جميع
!>                القيم الذاتية. العناصر 1:ilo-1 و i+1:n في WR و WI
!>                تحتوي على تلك القيم الذاتية التي تم حسابها بنجاح.
!>                (الإخفاقات نادرة.)
!>
!>                إذا كان INFO > 0 و WANT هو .FALSE.، فعند الخروج،
!>                القيم الذاتية غير المتقاربة المتبقية هي القيم الذاتية
!>                لمصفوفة هسنبرغ العليا في الصفوف والأعمدة ILO حتى INFO
!>                من القيمة النهائية، الخرج لـ H.
!>
!>                إذا كان INFO > 0 و WANTT هو .TRUE.، فعند الخروج
!>
!>           (*)  (القيمة الأولية لـ H)*U  = U*(القيمة النهائية لـ H)
!>
!>                حيث U هي مصفوفة وحدوية. القيمة النهائية لـ H هي
!>                هسنبرغ عليا ومثلثة في الصفوف والأعمدة INFO+1 حتى IHI.
!>
!>                إذا كان INFO > 0 و WANTZ هو .TRUE.، فعند الخروج
!>
!>                  (القيمة النهائية لـ Z(ILO:IHI,ILOZ:IHIZ)
!>                   =  (القيمة الأولية لـ Z(ILO:IHI,ILOZ:IHIZ)*U
!>
!>                حيث U هي المصفوفة الوحدوية في (*) (بغض النظر عن قيمة WANTT).
!>
!>                إذا كان INFO > 0 و WANTZ هو .FALSE.، فلا يتم الوصول إلى Z.
!>
