subroutine zlalsa (integer icompq, integer smlsiz, integer n, integer nrhs, complex*16, dimension( ldb, * ) b, integer ldb, complex*16, dimension( ldbx, * ) bx, integer ldbx, double precision, dimension( ldu, * ) u, integer ldu, double precision, dimension( ldu, * ) vt, integer, dimension( * ) k, double precision, dimension( ldu, * ) difl, double precision, dimension( ldu, * ) difr, double precision, dimension( ldu, * ) z, double precision, dimension( ldu, * ) poles, integer, dimension( * ) givptr, integer, dimension( ldgcol, * ) givcol, integer ldgcol, integer, dimension( ldgcol, * ) perm, double precision, dimension( ldu, * ) givnum, double precision, dimension( * ) c, double precision, dimension( * ) s, double precision, dimension( * ) rwork, integer, dimension( * ) iwork, integer info)

ZLALSA يحسب SVD للمصفوفة المعاملة في صيغة مدمجة. تستخدم بواسطة sgelsd.

الغرض:

!>
!> ZLALSA هي خطوة وسيطة في حل مشكلة المربعات الصغرى
!> عن طريق حساب SVD للمصفوفة المعاملة في صيغة مدمجة (تُحسب المتجهات المفردة
!> كمنتجات لمصفوفات متعامدة بسيطة.).
!>
!> إذا كان ICOMPQ = 0، تطبق ZLALSA معكوس مصفوفة المتجهات المفردة اليسرى
!> لمصفوفة ثنائية قطرية علوية على الطرف الأيمن؛ وإذا كان
!> ICOMPQ = 1، تطبق ZLALSA مصفوفة المتجهات المفردة اليمنى على
!> الطرف الأيمن. تم إنشاء مصفوفات المتجهات المفردة في
!> صيغة مدمجة بواسطة ZLALSA.
!>

المعلمات

ICOMPQ 
!>          ICOMPQ هو عدد صحيح
!>         يحدد ما إذا كانت مصفوفة المتجهات المفردة اليسرى أم اليمنى
!>         متضمنة.
!>         = 0: مصفوفة المتجهات المفردة اليسرى
!>         = 1: مصفوفة المتجهات المفردة اليمنى
!>

SMLSIZ

!>          SMLSIZ هو عدد صحيح
!>         أقصى حجم للمشاكل الفرعية في أسفل شجرة الحساب.
!>

N

!>          N هو عدد صحيح
!>         أبعاد الصف والعمود للمصفوفة ثنائية القطرية العلوية.
!>

NRHS

!>          NRHS هو عدد صحيح
!>         عدد أعمدة B و BX. يجب أن يكون NRHS على الأقل 1.
!>

B

!>          B هو مصفوفة COMPLEX*16، البعد ( LDB, NRHS )
!>         عند الإدخال، تحتوي B على الطرف الأيمن لمشكلة المربعات الصغرى
!>         في الصفوف من 1 إلى M.
!>         عند الإخراج، تحتوي B على الحل X في الصفوف من 1 إلى N.
!>

LDB

!>          LDB هو عدد صحيح
!>         البعد الرئيسي لـ B في البرنامج الفرعي المستدعي.
!>         يجب أن يكون LDB على الأقل max(1,MAX( M, N ) ).
!>

BX

!>          BX هو مصفوفة COMPLEX*16، البعد ( LDBX, NRHS )
!>         عند الخروج، نتيجة تطبيق مصفوفة المتجهات المفردة اليسرى أو اليمنى
!>         على B.
!>

LDBX

!>          LDBX هو عدد صحيح
!>         البعد الرئيسي لـ BX.
!>

U

!>          U هو مصفوفة DOUBLE PRECISION، البعد ( LDU, SMLSIZ ).
!>         عند الإدخال، تحتوي U على مصفوفات المتجهات المفردة اليسرى لجميع
!>         المشاكل الفرعية في المستوى السفلي.
!>

LDU

!>          LDU هو عدد صحيح، LDU = > N.
!>         البعد الرئيسي للمصفوفات U، VT، DIFL، DIFR،
!>         POLES، GIVNUM، و Z.
!>

VT

!>          VT هو مصفوفة DOUBLE PRECISION، البعد ( LDU, SMLSIZ+1 ).
!>         عند الإدخال، تحتوي VT**H على مصفوفات المتجهات المفردة اليمنى لجميع
!>         المشاكل الفرعية في المستوى السفلي.
!>

K

!>          K هي مصفوفة عدد صحيح، البعد ( N ).
!>

DIFL

!>          DIFL هي مصفوفة DOUBLE PRECISION، البعد ( LDU, NLVL ).
!>         حيث NLVL = INT(log_2 (N/(SMLSIZ+1))) + 1.
!>

DIFR

!>          DIFR هي مصفوفة DOUBLE PRECISION، البعد ( LDU, 2 * NLVL ).
!>         عند الإدخال، تسجل DIFL(*, I) و DIFR(*, 2 * I -1)
!>         المسافات بين القيم المفردة في المستوى I
!>         والقيم المفردة في المستوى (I -1)، وتسجل DIFR(*, 2 * I)
!>         عوامل التطبيع لمصفوفات المتجهات المفردة اليمنى
!>         للمشاكل الفرعية في المستوى I.
!>

Z

!>          Z هي مصفوفة DOUBLE PRECISION، البعد ( LDU, NLVL ).
!>         عند الإدخال، يحتوي Z(1, I) على مركبات المتجه الصفّي المحدث
!>         المعدّل للانكماش للمشاكل الفرعية في المستوى I.
!>

POLES

!>          POLES هي مصفوفة DOUBLE PRECISION، البعد ( LDU, 2 * NLVL ).
!>         عند الإدخال، تحتوي POLES(*, 2 * I -1: 2 * I) على القيم المفردة
!>         الجديدة والقديمة المتضمنة في المعادلات العلمانية في المستوى I.
!>

GIVPTR

!>          GIVPTR هي مصفوفة عدد صحيح، البعد ( N ).
!>         عند الإدخال، يسجل GIVPTR( I ) عدد دوران جيفنز
!>         التي تم إجراؤها على المشكلة I في شجرة الحساب.
!>

GIVCOL

!>          GIVCOL هي مصفوفة عدد صحيح، البعد ( LDGCOL, 2 * NLVL ).
!>         عند الإدخال، لكل I، يسجل GIVCOL(*, 2 * I - 1: 2 * I)
!>         مواقع دورانات جيفنز التي تم إجراؤها في المستوى I على
!>         شجرة الحساب.
!>

LDGCOL

!>          LDGCOL هو عدد صحيح، LDGCOL = > N.
!>         البعد الرئيسي لمصفوفتي GIVCOL و PERM.
!>

PERM

!>          PERM هي مصفوفة عدد صحيح، البعد ( LDGCOL, NLVL ).
!>         عند الإدخال، يسجل PERM(*, I) التباديل التي تم إجراؤها في المستوى I
!>         من شجرة الحساب.
!>

GIVNUM

!>          GIVNUM هي مصفوفة DOUBLE PRECISION، البعد ( LDU, 2 * NLVL ).
!>         عند الإدخال، تسجل GIVNUM(*, 2 *I -1 : 2 * I) القيم C و S
!>         لدورانات جيفنز التي تم إجراؤها في المستوى I على
!>         شجرة الحساب.
!>

C

!>          C هي مصفوفة DOUBLE PRECISION، البعد ( N ).
!>         عند الإدخال، إذا كانت المشكلة الفرعية I ليست مربعة،
!>         يحتوي C( I ) على قيمة C لدوران جيفنز يتعلق
!>         بالمجال الصفري الأيمن للمشكلة الفرعية I.
!>

S

!>          S هي مصفوفة DOUBLE PRECISION، البعد ( N ).
!>         عند الإدخال، إذا كانت المشكلة الفرعية I ليست مربعة،
!>         يحتوي S( I ) على قيمة S لدوران جيفنز يتعلق
!>         بالمجال الصفري الأيمن للمشكلة الفرعية I.
!>

RWORK

!>          RWORK هي مصفوفة DOUBLE PRECISION، البعد على الأقل
!>         MAX( (SMLSZ+1)*NRHS*3, N*(1+NRHS) + 2*NRHS ).
!>
