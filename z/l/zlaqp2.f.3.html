subroutine zlaqp2 (integer m, integer n, integer offset, complex*16, dimension( lda, * ) a, integer lda, integer, dimension( * ) jpvt, complex*16, dimension( * ) tau, double precision, dimension( * ) vn1, double precision, dimension( * ) vn2, complex*16, dimension( * ) work)

ZLAQP2 يحسب تحليل QR مع تثبيت الأعمدة للكتلة المصفوفة.

الغرض:

!>
!> ZLAQP2 يحسب تحليل QR مع تثبيت الأعمدة للكتلة A(OFFSET+1:M,1:N).
!> يتم تثبيت الكتلة A(1:OFFSET,1:N) وفقًا لذلك، ولكن لم يتم تحليلها.
!>

المعلمات

M 
!>          M هو عدد صحيح (INTEGER)
!>          عدد الصفوف في المصفوفة A. M >= 0.
!>

N

!>          N هو عدد صحيح (INTEGER)
!>          عدد الأعمدة في المصفوفة A. N >= 0.
!>

OFFSET

!>          OFFSET هو عدد صحيح (INTEGER)
!>          عدد الصفوف في المصفوفة A التي يجب تثبيتها
!>          ولكن لم يتم تحليلها. OFFSET >= 0.
!>

A

!>          A هي مصفوفة ذات أرقام مركبة *16 (COMPLEX*16)، أبعادها (LDA,N)
!>          عند الإدخال، المصفوفة A بحجم M × N.
!>          عند الخروج، المثلث العلوي للكتلة A(OFFSET+1:M,1:N) هو
!>          العامل المثلثي الذي تم الحصول عليه؛ العناصر في الكتلة
!>          A(OFFSET+1:M,1:N) تحت القطر، جنبًا إلى جنب مع المصفوفة TAU،
!>          تمثل المصفوفة المتعامدة Q كناتج للمنكسات الأولية. تم تثبيت الكتلة A(1:OFFSET,1:N)
!>          وفقًا لذلك، ولكن لم يتم تحليلها.
!>

LDA

!>          LDA هو عدد صحيح (INTEGER)
!>          البعد الرائد للمصفوفة A. LDA >= max(1,M).
!>

JPVT

!>          JPVT هي مصفوفة أعداد صحيحة (INTEGER)، أبعادها (N)
!>          عند الإدخال، إذا كان JPVT(i) .ne. 0، يتم تبديل العمود i من A
!>          إلى مقدمة A*P (عمود رائد)؛ إذا كان JPVT(i) = 0،
!>          فإن العمود i من A هو عمود حر.
!>          عند الخروج، إذا كان JPVT(i) = k، فإن العمود i من A*P
!>          كان العمود k من A.
!>

TAU

!>          TAU هي مصفوفة ذات أرقام مركبة *16 (COMPLEX*16)، أبعادها (min(M,N))
!>          العوامل العددية للمنكسات الأولية.
!>

VN1

!>          VN1 هي مصفوفة ذات دقة مزدوجة (DOUBLE PRECISION)، أبعادها (N)
!>          المتجه الذي يحتوي على معايير الأعمدة الجزئية.
!>

VN2

!>          VN2 هي مصفوفة ذات دقة مزدوجة (DOUBLE PRECISION)، أبعادها (N)
!>          المتجه الذي يحتوي على معايير الأعمدة الدقيقة.
!>

WORK

!>          WORK هي مصفوفة ذات أرقام مركبة *16 (COMPLEX*16)، أبعادها (N)
!>
