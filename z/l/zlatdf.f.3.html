subroutine zlatdf (ijob, n, z, ldz, rhs, rdsum, rdscal, ipiv, jpiv)
ZLATDF يستخدم التحلل LU للمصفوفة n-by-n التي تم حسابها بواسطة sgetc2 ويحسب مساهمة في تقدير Dif العكسي.

توثيق الوظيفة/الإجراء الفرعي

subroutine zlatdf (integer ijob, integer n, complex*16, dimension( ldz, * ) z, integer ldz, complex*16, dimension( * ) rhs, double precision rdsum, double precision rdscal, integer, dimension( * ) ipiv, integer, dimension( * ) jpiv)

ZLATDF يستخدم التحلل LU للمصفوفة n-by-n التي تم حسابها بواسطة sgetc2 ويحسب مساهمة في تقدير Dif العكسي.

الغرض:

!>
!> ZLATDF يحسب المساهمة في تقدير Dif العكسي
!> عن طريق حل x في Z * x = b، حيث يتم اختيار b بحيث يكون معيار
!> x أكبر ما يمكن. يفترض أن تحلل LU
!> للمصفوفة Z تم حسابه بواسطة ZGETC2. عند الدخول RHS = f يحمل
!> المساهمة من الأنظمة الفرعية التي تم حلها سابقًا، وعند الخروج RHS = x.
!>
!> التحلل للمصفوفة Z الذي تم إرجاعه بواسطة ZGETC2 له الشكل
!> Z = P * L * U * Q، حيث P و Q هي مصفوفات تبديل. L هي مصفوفة
!> مثلثية سفلية ذات عناصر قطرية واحدية و U هي مصفوفة مثلثية علوية.
!>

المعلمات

IJOB 
!>          IJOB هو عدد صحيح
!>          IJOB = 2: أولاً، قم بحساب متجه صفري تقريبي e
!>              للمصفوفة Z باستخدام ZGECON، يتم تطبيع e وحل
!>              Zx = +-e - f مع الإشارة التي تعطي القيمة الأكبر لـ
!>              2-norm(x). أغلى بحوالي 5 مرات من الافتراضي.
!>          IJOB .ne. 2: استراتيجية التطلع الأمامي المحلية حيث
!>              يتم اختيار جميع مدخلات الجانب الأيمن b لتكون إما +1 أو
!>              -1. الافتراضي.
!>

N

!>          N هو عدد صحيح
!>          عدد أعمدة المصفوفة Z.
!>

Z

!>          Z هو مصفوفة COMPLEX*16، أبعاد (LDZ, N)
!>          عند الدخول، جزء LU من تحلل المصفوفة n-by-n
!>          Z التي تم حسابها بواسطة ZGETC2: Z = P * L * U * Q
!>

LDZ

!>          LDZ هو عدد صحيح
!>          البعد الرئيسي للمصفوفة Z. LDA >= max(1, N).
!>

RHS

!>          RHS هي مصفوفة COMPLEX*16، أبعاد (N).
!>          عند الدخول، يحتوي RHS على مساهمات من أنظمة فرعية أخرى.
!>          عند الخروج، يحتوي RHS على حل النظام الفرعي بمدخلات
!>          وفقًا لقيمة IJOB (انظر أعلاه).
!>

RDSUM

!>          RDSUM هو double precision
!>          عند الدخول، مجموع مربعات المساهمات المحسوبة لـ
!>          تقدير Dif قيد الحساب بواسطة ZTGSYL، حيث تم استبعاد
!>          عامل القياس RDSCAL (انظر أدناه).
!>          عند الخروج، مجموع المربعات المقابل المحدث بـ
!>          المساهمات من النظام الفرعي الحالي.
!>          إذا كان TRANS = 'T' فإن RDSUM لا يتم المساس به.
!>          ملاحظة: RDSUM له معنى فقط عندما يتم استدعاء ZTGSY2 بواسطة CTGSYL.
!>

RDSCAL

!>          RDSCAL هو double precision
!>          عند الدخول، عامل القياس المستخدم لمنع تجاوز التدفق في RDSUM.
!>          عند الخروج، يتم تحديث RDSCAL فيما يتعلق بالمساهمات الحالية
!>          في RDSUM.
!>          إذا كان TRANS = 'T'، فلا يتم المساس بـ RDSCAL.
!>          ملاحظة: RDSCAL له معنى فقط عندما يتم استدعاء ZTGSY2 بواسطة
!>          ZTGSYL.
!>

IPIV

!>          IPIV هي مصفوفة عدد صحيح، أبعاد (N).
!>          مؤشرات المحور؛ لـ 1 <= i <= N، تم تبديل الصف i من
!>          المصفوفة بالصف IPIV(i).
!>

JPIV

!>          JPIV هي مصفوفة عدد صحيح، أبعاد (N).
!>          مؤشرات المحور؛ لـ 1 <= j <= N، تم تبديل العمود j من
!>          المصفوفة بالعمود JPIV(j).
!>
