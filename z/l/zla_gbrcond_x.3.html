الوظائف/الروتينات الفرعية


double precision function zla_gbrcond_x (trans, n, kl, ku, ab, ldab, afb, ldafb, ipiv, x, info, work, rwork)
ZLA_GBRCOND_X يحسب رقم شرط المعيار اللانهائي لـ op(A)*diag(x) للمصفوفات الشريطية العامة.

توثيق الوظيفة/الروتين الفرعي

double precision function zla_gbrcond_x (character trans, integer n, integer kl, integer ku, complex*16, dimension( ldab, * ) ab, integer ldab, complex*16, dimension( ldafb, * ) afb, integer ldafb, integer, dimension( * ) ipiv, complex*16, dimension( * ) x, integer info, complex*16, dimension( * ) work, double precision, dimension( * ) rwork)

ZLA_GBRCOND_X يحسب رقم شرط المعيار اللانهائي لـ op(A)*diag(x) للمصفوفات الشريطية العامة.

الغرض:

!>
!>    ZLA_GBRCOND_X يحسب رقم شرط المعيار اللانهائي لـ
!>    op(A) * diag(X) حيث X متجه COMPLEX*16.
!>

المعلمات

TRANS 
!>          TRANS is CHARACTER*1
!>     يحدد شكل نظام المعادلات:
!>       = 'N':  A * X = B     (بدون تبديل)
!>       = 'T':  A**T * X = B  (تبديل)
!>       = 'C':  A**H * X = B  (تبديل مرافق = تبديل)
!>

N

!>          N is INTEGER
!>     عدد المعادلات الخطية، أي، ترتيب المصفوفة A.  N >= 0.
!>

KL

!>          KL is INTEGER
!>     عدد الأقطار الفرعية داخل نطاق A.  KL >= 0.
!>

KU

!>          KU is INTEGER
!>     عدد الأقطار الفائقة داخل نطاق A.  KU >= 0.
!>

AB

!>          AB is COMPLEX*16 array, dimension (LDAB,N)
!>     عند الإدخال، المصفوفة A بتخزين النطاق، في الصفوف 1 إلى KL+KU+1.
!>     يتم تخزين العمود j من A في العمود j من المصفوفة
!>     AB كالتالي:
!>     AB(KU+1+i-j,j) = A(i,j) لـ max(1,j-KU)<=i<=min(N,j+kl)
!>

LDAB

!>          LDAB is INTEGER
!>     البعد الرائد للمصفوفة AB.  LDAB >= KL+KU+1.
!>

AFB

!>          AFB is COMPLEX*16 array, dimension (LDAFB,N)
!>     تفاصيل تحليل LU للمصفوفة النطاقية A، كما
!>     تم حسابها بواسطة ZGBTRF.  يتم تخزين U كمصفوفة
!>     مثلثة عليا في النطاق مع KL+KU أقطار فائقة في الصفوف 1 إلى KL+KU+1،
!>     وتُخزن المضاعفات المستخدمة أثناء التحليل
!>     في الصفوف KL+KU+2 إلى 2*KL+KU+1.
!>

LDAFB

!>          LDAFB is INTEGER
!>     البعد الرائد للمصفوفة AFB.  LDAFB >= 2*KL+KU+1.
!>

IPIV

!>          IPIV is INTEGER array, dimension (N)
!>     مؤشرات المحور من تحليل A = P*L*U
!>     كما تم حسابها بواسطة ZGBTRF؛ تم تبديل الصف i من المصفوفة
!>     مع الصف IPIV(i).
!>

X

!>          X is COMPLEX*16 array, dimension (N)
!>     المتجه X في الصيغة op(A) * diag(X).
!>

INFO

!>          INFO is INTEGER
!>       = 0:  انتهاء ناجح.
!>     i > 0:  المعلمة i غير صالحة.
!>

WORK

!>          WORK is COMPLEX*16 array, dimension (2*N).
!>     مساحة عمل.
!>

RWORK

!>          RWORK is DOUBLE PRECISION array, dimension (N).
!>     مساحة عمل.
!>
