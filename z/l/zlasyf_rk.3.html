subroutine zlasyf_rk (character uplo, integer n, integer nb, integer kb, complex*16, dimension( lda, * ) a, integer lda, complex*16, dimension( * ) e, integer, dimension( * ) ipiv, complex*16, dimension( ldw, * ) w, integer ldw, integer info)

ZLASYF_RK يحسب تحليل جزئي لمصفوفة متماثلة معقدة غير محددة باستخدام طريقة التقليب القطري المحدود لـ Bunch-Kaufman (rook).

الغرض:

!> ZLASYF_RK يحسب تحليل جزئي لمصفوفة متماثلة معقدة
!> باستخدام طريقة التقليب القطري المحدود لـ Bunch-Kaufman (rook).
!> التحليل الجزئي يأخذ الشكل التالي:
!>
!> A  =  ( I  U12 ) ( A11  0  ) (  I       0    )  إذا كان UPLO = 'U'، أو:
!>       ( 0  U22 ) (  0   D  ) ( U12**T U22**T )
!>
!> A  =  ( L11  0 ) (  D   0  ) ( L11**T L21**T )  إذا كان UPLO = 'L'،
!>       ( L21  I ) (  0  A22 ) (  0       I    )
!>
!> حيث أن رتبة D لا تتجاوز NB. الرتبة الفعلية تُعاد في
!> المتغير KB، وهي إما NB أو NB-1، أو N إذا كان N <= NB.
!>
!> ZLASYF_RK هو روتين مساعد يتم استدعاؤه بواسطة ZSYTRF_RK. يستخدم
!> الكود المجمع (استدعاء BLAS المستوى 3) لتحديث المصفوفة الفرعية
!> A11 (إذا كان UPLO = 'U') أو A22 (إذا كان UPLO = 'L').
!>

المعاملات

UPLO 
!>          UPLO هو CHARACTER*1
!>          يحدد ما إذا كان الجزء العلوي أو السفلي من المثلث المتماثل للمصفوفة A
!>          مخزنًا:
!>          = 'U':  مثلثي علوي
!>          = 'L':  مثلثي سفلي
!>

N

!>          N هو INTEGER
!>          رتبة المصفوفة A.  N >= 0.
!>

NB

!>          NB هو INTEGER
!>          الحد الأقصى لعدد الأعمدة في المصفوفة A التي يجب
!>          تحليلها.  يجب أن يكون NB على الأقل 2 للسماح بكتل محورية 2x2.
!>

KB

!>          KB هو INTEGER
!>          عدد الأعمدة في A التي تم تحليلها بالفعل.
!>          KB هو إما NB-1 أو NB، أو N إذا كان N <= NB.
!>

A

!>          A هو مصفوفة COMPLEX*16، البعد (LDA,N)
!>          عند الدخول، المصفوفة المتماثلة A.
!>            إذا كان UPLO = 'U': الجزء العلوي المثلثي NxN الرائد
!>            من A يحتوي على الجزء العلوي المثلثي للمصفوفة A،
!>            والجزء السفلي المثلثي الصارم من A لا يتم الرجوع إليه.
!>
!>            إذا كان UPLO = 'L': الجزء السفلي المثلثي N-by-N الرائد
!>            من A يحتوي على الجزء السفلي المثلثي للمصفوفة A،
!>            والجزء العلوي المثلثي الصارم من A لا يتم الرجوع إليه.
!>
!>          عند الخروج، تحتوي على:
!>            أ) عناصر القطر فقط للمصفوفة القطرية المتماثلة المجمعة
!>               D على قطر A، أي D(k,k) = A(k,k)؛
!>               (العناصر فوق القطرية (أو تحت القطرية) في D
!>                مخزنة عند الخروج في المصفوفة E)، و
!>            ب) إذا كان UPLO = 'U': العامل U في الجزء فوق القطري من A.
!>               إذا كان UPLO = 'L': العامل L في الجزء تحت القطري من A.
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرائد للمصفوفة A.  LDA >= max(1,N).
!>

E

!>          E هي مصفوفة COMPLEX*16، البعد (N)
!>          عند الخروج، تحتوي على العناصر فوق القطرية (أو تحت القطرية)
!>          للمصفوفة القطرية المتماثلة المجمعة D
!>          مع كتل قطرية 1x1 أو 2x2، حيث:
!>          إذا كان UPLO = 'U': E(i) = D(i-1,i)، i=2:N، E(1) يتم تعيينه إلى 0؛
!>          إذا كان UPLO = 'L': E(i) = D(i+1,i)، i=1:N-1، E(N) يتم تعيينه إلى 0.
!>
!>          ملاحظة: لكتلة قطرية 1x1 D(k)، حيث
!>          1 <= k <= N، يتم تعيين العنصر E(k) إلى 0 في كل من
!>          حالات UPLO = 'U' أو UPLO = 'L'.
!>

IPIV

!>          IPIV هي مصفوفة INTEGER، البعد (N)
!>          IPIV تصف مصفوفة التبديل P في تحليل
!>          المصفوفة A على النحو التالي. القيمة المطلقة لـ IPIV(k)
!>          تمثل فهرس الصف والعمود اللذين تم تبديلهما مع الصف والعمود k.
!>          قيمة UPLO تصف ترتيب تطبيق التبادلات.
!>          أيضًا، إشارة IPIV تمثل البنية المجمعة
!>          للمصفوفة القطرية المتماثلة المجمعة D بكتل قطرية 1x1 أو 2x2
!>          والتي تتوافق مع تبديل أو تبديلين في كل خطوة تحليل.
!>
!>          إذا كان UPLO = 'U'،
!>          ( بترتيب التحليل، k يتناقص من N إلى 1):
!>            أ) إدخال موجب فردي IPIV(k) > 0 يعني:
!>               D(k,k) هي كتلة قطرية 1x1.
!>               إذا كان IPIV(k) != k، تم تبديل الصفوف والأعمدة k و IPIV(k)
!>               في المصفوفة الفرعية A(1:N,N-KB+1:N)؛
!>               إذا كان IPIV(k) = k، لم يحدث تبديل.
!>
!>
!>            ب) زوج من الإدخالات المتتالية السالبة
!>               IPIV(k) < 0 و IPIV(k-1) < 0 يعني:
!>               D(k-1:k,k-1:k) هي كتلة قطرية 2x2.
!>               (ملاحظة: الإدخالات السالبة في IPIV تظهر فقط في أزواج).
!>               1) إذا كان -IPIV(k) != k، تم تبديل الصفوف والأعمدة
!>                  k و -IPIV(k)
!>                  في المصفوفة الفرعية A(1:N,N-KB+1:N).
!>                  إذا كان -IPIV(k) = k، لم يحدث تبديل.
!>               2) إذا كان -IPIV(k-1) != k-1، تم تبديل الصفوف والأعمدة
!>                  k-1 و -IPIV(k-1)
!>                  في المصفوفة الفرعية A(1:N,N-KB+1:N).
!>                  إذا كان -IPIV(k-1) = k-1، لم يحدث تبديل.
!>
!>            ج) في كلتا الحالتين أ) وب) يكون دائمًا | IPIV(k) | <= k.
!>
!>            د) ملاحظة: أي إدخال IPIV(k) غير صفري دائمًا عند الخروج.
!>
!>          إذا كان UPLO = 'L'،
!>          ( بترتيب التحليل، k يزيد من 1 إلى N):
!>            أ) إدخال موجب فردي IPIV(k) > 0 يعني:
!>               D(k,k) هي كتلة قطرية 1x1.
!>               إذا كان IPIV(k) != k، تم تبديل الصفوف والأعمدة k و IPIV(k)
!>               في المصفوفة الفرعية A(1:N,1:KB).
!>               إذا كان IPIV(k) = k، لم يحدث تبديل.
!>
!>            ب) زوج من الإدخالات المتتالية السالبة
!>               IPIV(k) < 0 و IPIV(k+1) < 0 يعني:
!>               D(k:k+1,k:k+1) هي كتلة قطرية 2x2.
!>               (ملاحظة: الإدخالات السالبة في IPIV تظهر فقط في أزواج).
!>               1) إذا كان -IPIV(k) != k، تم تبديل الصفوف والأعمدة
!>                  k و -IPIV(k)
!>                  في المصفوفة الفرعية A(1:N,1:KB).
!>                  إذا كان -IPIV(k) = k، لم يحدث تبديل.
!>               2) إذا كان -IPIV(k+1) != k+1، تم تبديل الصفوف والأعمدة
!>                  k-1 و -IPIV(k-1)
!>                  في المصفوفة الفرعية A(1:N,1:KB).
!>                  إذا كان -IPIV(k+1) = k+1، لم يحدث تبديل.
!>
!>            ج) في كلتا الحالتين أ) وب) يكون دائمًا | IPIV(k) | >= k.
!>
!>            د) ملاحظة: أي إدخال IPIV(k) غير صفري دائمًا عند الخروج.
!>

W

!>          W هي مصفوفة COMPLEX*16، البعد (LDW,NB)
!>

LDW

!>          LDW هو INTEGER
!>          البعد الرائد للمصفوفة W.  LDW >= max(1,N).
!>

INFO

!>          INFO هو INTEGER
!>          = 0: خروج ناجح
!>
!>          < 0: إذا كان INFO = -k، فإن المعامل k كان له قيمة غير قانونية
!>
!>          > 0: إذا كان INFO = k، فإن المصفوفة A منفردة، بسبب:
!>                 إذا كان UPLO = 'U': العمود k في الجزء
!>                 المثلثي العلوي من A يحتوي على جميع الأصفار.
!>                 إذا كان UPLO = 'L': العمود k في الجزء
!>                 المثلثي السفلي من A يحتوي على جميع الأصفار.
!>
!>               لذلك D(k,k) هو صفر بالضبط، والعناصر فوق القطرية
!>               للعمود k من U (أو العناصر تحت القطرية لـ
!>               العمود
