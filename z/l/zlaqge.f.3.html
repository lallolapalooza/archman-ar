وثائق الوظيفة/الروتين الفرعي

subroutine zlaqge (integer m, integer n, complex*16, dimension( lda, * ) a, integer lda, double precision, dimension( * ) r, double precision, dimension( * ) c, double precision rowcnd, double precision colcnd, double precision amax, character equed)

ZLAQGE يقوم بتوسيع مصفوفة مستطيلة عامة، باستخدام عوامل توسيع الصفوف والأعمدة التي حسبتها sgeequ.

الغرض:

!>
!> ZLAQGE يوازن مصفوفة عامة M في N باستخدام عوامل توسيع الصفوف والأعمدة في المتجهين R و C.
!>

المعلمات

M 
!>          M هو عدد صحيح (INTEGER)
!>          عدد صفوف المصفوفة A.  M >= 0.
!>

N

!>          N هو عدد صحيح (INTEGER)
!>          عدد أعمدة المصفوفة A.  N >= 0.
!>

A

!>          A هي مصفوفة معقدة مزدوجة الدقة (COMPLEX*16)، أبعادها (LDA,N)
!>          عند الإدخال، المصفوفة A ذات الأبعاد M في N.
!>          عند الخروج، المصفوفة المتوازنة. راجع EQUED لشكل المصفوفة المتوازنة.
!>

LDA

!>          LDA هو عدد صحيح (INTEGER)
!>          البعد الرائد للمصفوفة A.  LDA >= max(M,1).
!>

R

!>          R هي مصفوفة دقة مزدوجة (DOUBLE PRECISION)، أبعادها (M)
!>          عوامل توسيع الصفوف للمصفوفة A.
!>

C

!>          C هي مصفوفة دقة مزدوجة (DOUBLE PRECISION)، أبعادها (N)
!>          عوامل توسيع الأعمدة للمصفوفة A.
!>

ROWCND

!>          ROWCND هو دقة مزدوجة (DOUBLE PRECISION)
!>          نسبة أصغر R(i) إلى أكبر R(i).
!>

COLCND

!>          COLCND هو دقة مزدوجة (DOUBLE PRECISION)
!>          نسبة أصغر C(i) إلى أكبر C(i).
!>

AMAX

!>          AMAX هو دقة مزدوجة (DOUBLE PRECISION)
!>          القيمة المطلقة لأكبر عنصر في المصفوفة.
!>

EQUED

!>          EQUED هو حرف (CHARACTER*1)
!>          يحدد شكل الموازنة التي تم إجراؤها.
!>          = 'N':  لا توجد موازنة
!>          = 'R':  موازنة الصفوف، أي تم ضرب A مسبقاً بـ
!>                  diag(R).
!>          = 'C':  موازنة الأعمدة، أي تم ضرب A لاحقاً بـ
!>                  diag(C).
!>          = 'B':  موازنة الصفوف والأعمدة، أي تم استبدال A بـ
!>                  diag(R) * A * diag(C).
!>

المعلمات الداخلية:

!>  THRESH هو قيمة حدية تستخدم لتحديد ما إذا كان يجب إجراء توسيع الصفوف أو الأعمدة
!>  بناءً على نسبة عوامل توسيع الصفوف أو الأعمدة. إذا كان ROWCND < THRESH، يتم إجراء توسيع الصفوف، وإذا كان
!>  COLCND < THRESH، يتم إجراء توسيع الأعمدة.
!>
!>  LARGE و SMALL هما قيم حدية تستخدم لتحديد ما إذا كان يجب إجراء توسيع الصفوف
!>  بناءً على الحجم المطلق لأكبر عنصر في المصفوفة. إذا كان AMAX > LARGE أو AMAX < SMALL، يتم إجراء توسيع الصفوف.
!>
