subroutine zlavhe_rook (character uplo, character trans, character diag, integer n, integer nrhs, complex*16, dimension( lda, * ) a, integer lda, integer, dimension( * ) ipiv, complex*16, dimension( ldb, * ) b, integer ldb, integer info)

ZLAVHE_ROOK

الغرض:

تقوم ZLAVHE_ROOK بإحدى عمليات المصفوفة والمتجه x := A*x أو x := A^H*x، حيث x هو متجه ذو N عنصر و A هي واحدة من العوامل من التحليل الكتلي U*D*U' أو L*D*L' الذي تم حسابه بواسطة ZHETRF_ROOK.

إذا كان TRANS = 'N'، فإنها تضرب بـ U أو U * D (أو L أو L * D). إذا كان TRANS = 'C'، فإنها تضرب بـ U' أو D * U' (أو L' أو D * L').

المعلمات

UPLO 
!>          UPLO is CHARACTER*1
!>          يحدد ما إذا كان العامل المخزن في A مثلثيًا علويًا أم سفليًا.
!>          = 'U':  مثلثي علوي
!>          = 'L':  مثلثي سفلي
!>

TRANS

!>          TRANS is CHARACTER*1
!>          يحدد العملية التي سيتم إجراؤها:
!>          = 'N':  x := A*x
!>          = 'C':   x := A^H*x
!>

DIAG

!>          DIAG is CHARACTER*1
!>          يحدد ما إذا كانت الكتل القطرية عبارة عن مصفوفات وحدة أم لا. إذا تم افتراض أن الكتل القطرية هي مصفوفات وحدة، فإن A = U أو A = L، وإلا فإن A = U*D أو A = L*D.
!>          = 'U':  يُفترض أن تكون الكتل القطرية مصفوفات وحدة.
!>          = 'N':  يُفترض أن تكون الكتل القطرية مصفوفات غير وحدة.
!>

N

!>          N is INTEGER
!>          عدد الصفوف والأعمدة في المصفوفة A. N >= 0.
!>

NRHS

!>          NRHS is INTEGER
!>          عدد الأطراف اليمنى، أي عدد المتجهات x التي سيتم ضربها بـ A. NRHS >= 0.
!>

A

!>          A is COMPLEX*16 array, dimension (LDA,N)
!>          المصفوفة القطرية الكتلة D والعوامل المستخدمة للحصول على العامل U أو L كما تم حسابه بواسطة ZHETRF_ROOK.
!>          مخزنة كمصفوفة مثلثية ثنائية الأبعاد.
!>

LDA

!>          LDA is INTEGER
!>          البعد الرئيسي للمصفوفة A. LDA >= max(1,N).
!>

IPIV

!>          IPIV is INTEGER array, dimension (N)
!>          تفاصيل التبادلات والبنية الكتلية لـ D، كما تم تحديدها بواسطة ZHETRF_ROOK.
!>          إذا كان UPLO = 'U':
!>             يتم تعيين العناصر KB الأخيرة فقط من IPIV.
!>
!>             إذا كان IPIV(k) > 0، فقد تم تبديل الصفوف والأعمدة k و IPIV(k) و D(k,k) هي كتلة قطرية 1 × 1.
!>
!>             إذا كان IPIV(k) < 0 و IPIV(k-1) < 0، فقد تم تبديل الصفوف والأعمدة k و -IPIV(k) وتم تبديل الصفوف والأعمدة k-1 و -IPIV(k-1)،
!>             D(k-1:k,k-1:k) هي كتلة قطرية 2 × 2.
!>
!>          إذا كان UPLO = 'L':
!>             يتم تعيين العناصر KB الأولى فقط من IPIV.
!>
!>             إذا كان IPIV(k) > 0، فقد تم تبديل الصفوف والأعمدة k و IPIV(k)
!>             و D(k,k) هي كتلة قطرية 1 × 1.
!>
!>             إذا كان IPIV(k) < 0 و IPIV(k+1) < 0، فقد تم تبديل الصفوف والأعمدة k و -IPIV(k) وتم تبديل الصفوف والأعمدة k+1 و -IPIV(k+1)،
!>             D(k:k+1,k:k+1) هي كتلة قطرية 2 × 2.
!>

B

!>          B is COMPLEX array, dimension (LDB,NRHS)
!>          عند الإدخال، يحتوي B على NRHS متجه بطول N.
!>          عند الخروج، يتم استبدال B بمنتج A * B.
!>

LDB

!>          LDB is INTEGER
!>          البعد الرئيسي للمصفوفة B. LDB >= max(1,N).
!>

INFO

!>          INFO is INTEGER
!>          = 0: خروج ناجح
!>          < 0: إذا كان INFO = -k، فإن المعلمة k كانت لها قيمة غير قانونية
!>
