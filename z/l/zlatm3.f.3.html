توثيق الوظيفة/الروتين الفرعي

complex*16 function zlatm3 (integer m, integer n, integer i, integer j, integer isub, integer jsub, integer kl, integer ku, integer idist, integer, dimension( 4 ) iseed, complex*16, dimension( * ) d, integer igrade, complex*16, dimension( * ) dl, complex*16, dimension( * ) dr, integer ipvtng, integer, dimension( * ) iwork, double precision sparse)

ZLATM3

الغرض:

!>
!>    ZLATM3 تُرجع الإدخال (ISUB,JSUB) لمصفوفة عشوائية ذات
!>    البعد (M, N) موصوفة بالمعلمات الأخرى. (ISUB,JSUB)
!>    هو الموضع النهائي للإدخال (I,J) بعد التبديل
!>    وفقًا لـ IPVTNG و IWORK. يتم استدعاء ZLATM3 بواسطة
!>    ZLATMR لبناء مصفوفات اختبار عشوائية. لا يتم إجراء فحص للأخطاء
!>    على المعلمات، لأن هذا الروتين يتم استدعاؤه في
!>    حلقة ضيقة بواسطة ZLATMR الذي قام بالفعل بفحص المعلمات.
!>
!>    يختلف استخدام ZLATM3 عن CLATM2 في ترتيب توليد
!>    الأرقام العشوائية لملء إدخالات المصفوفة العشوائية.
!>    مع ZLATM2، يتم استدعاء المولد لملء المصفوفة المبدلة
!>    عمودياً. مع ZLATM3، يتم استدعاء المولد لملء
!>    المصفوفة عمودياً، وبعد ذلك يتم تبديلها. وبالتالي، يمكن لـ ZLATM3
!>    استخدامها لبناء مصفوفات عشوائية تختلف فقط في ترتيب
!>    صفوفها و/أو أعمدتها. يتم استخدام ZLATM2 لبناء
!>    مصفوفات النطاق مع تجنب استدعاء مولد الأرقام العشوائية
!>    للإدخالات خارج النطاق (وبالتالي توليد أرقام عشوائية
!>    بترتيبات مختلفة لترتيبات تبديل مختلفة).
!>
!>    المصفوفة التي يتم إرجاع إدخالها (ISUB,JSUB) يتم بناؤها كما يلي
!>    (يقوم هذا الروتين بحساب إدخال واحد فقط):
!>
!>      إذا كان ISUB خارج (1..M) أو JSUB خارج (1..N)، يتم إرجاع صفر
!>         (هذا مناسب لإنشاء مصفوفات بتنسيق النطاق).
!>
!>      توليد مصفوفة A بإدخالات عشوائية من التوزيع IDIST.
!>
!>      تعيين القطر إلى D.
!>
!>      ترتيب المصفوفة، إذا رغبت، من اليسار (بواسطة DL) و/أو
!>         من اليمين (بواسطة DR أو DL) كما هو محدد بواسطة IGRADE.
!>
!>      تبديل، إذا رغبت، الصفوف و/أو الأعمدة كما هو محدد بواسطة
!>         IPVTNG و IWORK.
!>
!>      تحويل المصفوفة إلى نطاق سفلي KL ونطاق علوي
!>         KU.
!>
!>      تعيين الإدخالات العشوائية إلى صفر كما هو محدد بواسطة SPARSE.
!>

المعلمات

M 
!>          M هو INTEGER
!>           عدد صفوف المصفوفة. لا يتم تعديله.
!>

N

!>          N هو INTEGER
!>           عدد أعمدة المصفوفة. لا يتم تعديله.
!>

I

!>          I هو INTEGER
!>           صف الإدخال غير المبدل الذي سيتم إرجاعه. لا يتم تعديله.
!>

J

!>          J هو INTEGER
!>           عمود الإدخال غير المبدل الذي سيتم إرجاعه. لا يتم تعديله.
!>

ISUB

!>          ISUB هو INTEGER
!>           صف الإدخال المبدل الذي سيتم إرجاعه. يتم تعديله عند الخروج.
!>

JSUB

!>          JSUB هو INTEGER
!>           عمود الإدخال المبدل الذي سيتم إرجاعه. يتم تعديله عند الخروج.
!>

KL

!>          KL هو INTEGER
!>           عرض النطاق السفلي. لا يتم تعديله.
!>

KU

!>          KU هو INTEGER
!>           عرض النطاق العلوي. لا يتم تعديله.
!>

IDIST

!>          IDIST هو INTEGER
!>           عند الدخول، يحدد IDIST نوع التوزيع الذي سيتم
!>           استخدامه لتوليد مصفوفة عشوائية.
!>           1 => الأجزاء الحقيقية والتخيلية كل منها UNIFORM( 0, 1 )
!>           2 => الأجزاء الحقيقية والتخيلية كل منها UNIFORM( -1, 1 )
!>           3 => الأجزاء الحقيقية والتخيلية كل منها NORMAL( 0, 1 )
!>           4 => رقم معقد موحد في DISK( 0 , 1 )
!>           لا يتم تعديله.
!>

ISEED

!>          ISEED هو مصفوفة INTEGER ذات البعد ( 4 )
!>           بذرة مولد الأرقام العشوائية.
!>           يتم تعديله عند الخروج.
!>

D

!>          D هو مصفوفة COMPLEX*16 ذات البعد ( MIN( I , J ) )
!>           عناصر القطر للمصفوفة. لا يتم تعديلها.
!>

IGRADE

!>          IGRADE هو INTEGER
!>           يحدد ترتيب المصفوفة كما يلي:
!>           0  => لا يوجد ترتيب.
!>           1  => المصفوفة مضروبة مسبقًا بـ diag( DL ).
!>           2  => المصفوفة مضروبة لاحقًا بـ diag( DR ).
!>           3  => المصفوفة مضروبة مسبقًا بـ diag( DL ) و
!>                         مضروبة لاحقًا بـ diag( DR ).
!>           4  => المصفوفة مضروبة مسبقًا بـ diag( DL ) و
!>                         مضروبة لاحقًا بـ inv( diag( DL ) ).
!>           5  => المصفوفة مضروبة مسبقًا بـ diag( DL ) و
!>                         مضروبة لاحقًا بـ diag( CONJG(DL) ).
!>           6  => المصفوفة مضروبة مسبقًا بـ diag( DL ) و
!>                         مضروبة لاحقًا بـ diag( DL ).
!>           لا يتم تعديله.
!>

DL

!>          DL هو مصفوفة COMPLEX*16 ( I أو J، حسب الاقتضاء )
!>           عوامل قياس اليسار لترتيب المصفوفة. لا يتم تعديلها.
!>

DR

!>          DR هو مصفوفة COMPLEX*16 ( I أو J، حسب الاقتضاء )
!>           عوامل قياس اليمين لترتيب المصفوفة. لا يتم تعديلها.
!>

IPVTNG

!>          IPVTNG هو INTEGER
!>           عند الدخول يحدد تبديلات التبديل كما يلي:
!>           0 => لا شيء.
!>           1 => تبديل الصفوف.
!>           2 => تبديل الأعمدة.
!>           3 => تبديل كامل، أي على الجانبين.
!>           لا يتم تعديله.
!>

IWORK

!>          IWORK هي مصفوفة INTEGER ( I أو J، حسب الاقتضاء )
!>           تحدد هذه المصفوفة التبديل المستخدم. الصف
!>           (أو العمود) الموجود أصلاً في الموضع K يكون في
!>           الموضع IWORK( K ) بعد التبديل.
!>           هذا يختلف عن IWORK لـ ZLATM2. لا يتم تعديله.
!>

SPARSE

!>          SPARSE هو DOUBLE PRECISION بين 0. و 1.
!>           عند الدخول يحدد ندرة المصفوفة
!>           إذا كان سيتم إنشاء مصفوفة نادرة.
!>           يجب أن تكون SPARSE بين 0 و 1.
!>           يتم توليد رقم عشوائي موحد ( 0, 1 ) x
!>           ومقارنته بـ SPARSE؛ إذا كان x أكبر، يظل إدخال المصفوفة
!>           دون تغيير، وإذا كان x أصغر، يتم تعيين الإدخال
!>           إلى صفر. وبالتالي، في المتوسط، سيتم تعيين جزء fracción SPARSE من
!>           الإدخالات إلى صفر.
!>           لا يتم تعديله.
!>
