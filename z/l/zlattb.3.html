subroutine zlattb (integer imat, character uplo, character trans, character diag, integer, dimension( 4 ) iseed, integer n, integer kd, complex*16, dimension( ldab, * ) ab, integer ldab, complex*16, dimension( * ) b, complex*16, dimension( * ) work, double precision, dimension( * ) rwork, integer info)

ZLATTB

الغرض:

!>
!> ZLATTB تولد مصفوفة مثلثية اختبارية في تخزين ثنائي الأبعاد.
!> IMAT و UPLO يحددان بشكل فريد خصائص مصفوفة الاختبار،
!> والتي يتم إرجاعها في المصفوفة A.
!>

المعلمات

IMAT 
!>          IMAT هو عدد صحيح
!>          مفتاح عدد صحيح يصف أي مصفوفة سيتم إنشاؤها لهذا
!>          المسار.
!>

UPLO

!>          UPLO هو حرف*1
!>          يحدد ما إذا كانت المصفوفة A ستكون علوية أو سفلية
!>          مثلثية.
!>          = 'U':  مثلثية علوية
!>          = 'L':  مثلثية سفلية
!>

TRANS

!>          TRANS هو حرف*1
!>          يحدد ما إذا كانت المصفوفة أو منقولتها سيتم استخدامها.
!>          = 'N':  لا يوجد تبديل
!>          = 'T':  تبديل
!>          = 'C':  تبديل مترافق (= تبديل)
!>

DIAG

!>          DIAG هو حرف*1
!>          يحدد ما إذا كانت المصفوفة A مثلثة وحدوية أم لا.
!>          = 'N':  مثلثية غير وحدوية
!>          = 'U':  مثلثية وحدوية
!>

ISEED

!>          ISEED هي مصفوفة أعداد صحيحة، حجمها (4)
!>          متجه البذر لمولد الأرقام العشوائية (يستخدم في
!>          ZLATMS).  يتم تعديلها عند الخروج.
!>

N

!>          N هو عدد صحيح
!>          رتبة المصفوفة المراد إنشاؤها.
!>

KD

!>          KD هو عدد صحيح
!>          عدد الأقطار الفائقة أو الأقطار الفرعية للمصفوفة المثلثية
!>          الشريطية A.  KD >= 0.
!>

AB

!>          AB هي مصفوفة COMPLEX*16، حجمها (LDAB,N)
!>          المصفوفة الشريطية المثلثية العلوية أو السفلية A، مخزنة في
!>          الصفوف KD+1 الأولى من AB.  لتكن j عمودًا في A، 1<=j<=n.
!>          إذا كان UPLO = 'U'، فإن AB(kd+1+i-j,j) = A(i,j) لـ max(1,j-kd)<=i<=j.
!>          إذا كان UPLO = 'L'، فإن AB(1+i-j,j)    = A(i,j) لـ j<=i<=min(n,j+kd).
!>

LDAB

!>          LDAB هو عدد صحيح
!>          البعد الرئيسي للمصفوفة AB.  LDAB >= KD+1.
!>

B

!>          B هي مصفوفة COMPLEX*16، حجمها (N)
!>

WORK

!>          WORK هي مصفوفة COMPLEX*16، حجمها (2*N)
!>

RWORK

!>          RWORK هي مصفوفة DOUBLE PRECISION، حجمها (N)
!>

INFO

!>          INFO هو عدد صحيح
!>          = 0:  انتهاء ناجح
!>          < 0:  إذا كان INFO = -i، فإن المعلمة i لها قيمة غير قانونية
!>
