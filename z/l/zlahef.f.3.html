subroutine zlahef (uplo, n, nb, kb, a, lda, ipiv, w, ldw, info)
ZLAHEF يحسب تحليل جزئي لمصفوفة هيرميتية معقدة غير محددة باستخدام طريقة ارتكاز الزاوية لبوش-كوفمان (خوارزمية محظورة، تستدعي BLAS المستوى 3).

توثيق الوظيفة/الإجراء الفرعي

subroutine zlahef (character uplo, integer n, integer nb, integer kb, complex*16, dimension( lda, * ) a, integer lda, integer, dimension( * ) ipiv, complex*16, dimension( ldw, * ) w, integer ldw, integer info)

ZLAHEF يحسب تحليل جزئي لمصفوفة هيرميتية معقدة غير محددة باستخدام طريقة ارتكاز الزاوية لبوش-كوفمان (خوارزمية محظورة، تستدعي BLAS المستوى 3).

الغرض:

!>
!> ZLAHEF يحسب تحليل جزئي للمصفوفة الهيرميتية A باستخدام طريقة ارتكاز الزاوية لبوش-كوفمان.
!> التحليل الجزئي له الشكل:
!>
!> A  =  ( I  U12 ) ( A11  0  ) (  I      0     )  إذا كان UPLO = 'U'، أو:
!>       ( 0  U22 ) (  0   D  ) ( U12**H U22**H )
!>
!> A  =  ( L11  0 ) (  D   0  ) ( L11**H L21**H )  إذا كان UPLO = 'L'
!>       ( L21  I ) (  0  A22 ) (  0      I     )
!>
!> حيث يكون ترتيب D بحد أقصى NB. يتم إرجاع الترتيب الفعلي في
!> الوسيط KB، وهو إما NB أو NB-1، أو N إذا كان N <= NB.
!> لاحظ أن U**H يمثل النقل المترافق لـ U.
!>
!> ZLAHEF هو إجراء مساعد يتم استدعاؤه بواسطة ZHETRF. يستخدم شفرة محظورة
!> (تستدعي BLAS المستوى 3) لتحديث المصفوفة الفرعية A11 (إذا كان UPLO = 'U') أو
!> A22 (إذا كان UPLO = 'L').
!>

المعلمات

UPLO 
!>          UPLO هو CHARACTER*1
!>          يحدد ما إذا كان الجزء العلوي أو السفلي الثلاثي للمصفوفة الهيرميتية A مخزنًا:
!>          = 'U':  مثلث علوي
!>          = 'L':  مثلث سفلي
!>

N

!>          N هو INTEGER
!>          ترتيب المصفوفة A. N >= 0.
!>

NB

!>          NB هو INTEGER
!>          الحد الأقصى لعدد الأعمدة في المصفوفة A التي يجب تحليلها.
!>          يجب أن يكون NB على الأقل 2 للسماح بكتل ارتكاز بحجم 2x2.
!>

KB

!>          KB هو INTEGER
!>          عدد الأعمدة في A التي تم تحليلها بالفعل.
!>          KB هو إما NB-1 أو NB، أو N إذا كان N <= NB.
!>

A

!>          A هو مصفوفة COMPLEX*16، ذات الأبعاد (LDA,N)
!>          عند الإدخال، المصفوفة الهيرميتية A. إذا كان UPLO = 'U'، فإن الجزء
!>          العلوي الثلاثي من A الذي حجمه n*n يحتوي على الجزء العلوي
!>          الثلاثي للمصفوفة A، ولا يتم الإشارة إلى الجزء السفلي الثلاثي
!>          بدقة للمصفوفة A. إذا كان UPLO = 'L'، فإن الجزء
!>          السفلي الثلاثي من A الذي حجمه n*n يحتوي على الجزء السفلي
!>          الثلاثي للمصفوفة A، ولا يتم الإشارة إلى الجزء العلوي الثلاثي
!>          بدقة للمصفوفة A.
!>          عند الخروج، تحتوي A على تفاصيل التحليل الجزئي.
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرائد للمصفوفة A. LDA >= max(1,N).
!>

IPIV

!>          IPIV هي مصفوفة INTEGER، ذات الأبعاد (N)
!>          تفاصيل التبادلات والبنية الكتلية لـ D.
!>
!>          إذا كان UPLO = 'U':
!>             فقط العناصر KB الأخيرة في IPIV يتم تعيينها.
!>
!>             إذا كان IPIV(k) > 0، فقد تم تبديل الصفوف والأعمدة k و IPIV(k)
!>             و D(k,k) هي كتلة قطرية بحجم 1x1.
!>
!>             إذا كان IPIV(k) = IPIV(k-1) < 0، فقد تم تبديل الصفوف والأعمدة
!>             k-1 و -IPIV(k) و D(k-1:k,k-1:k)
!>             هي كتلة قطرية بحجم 2x2.
!>
!>          إذا كان UPLO = 'L':
!>             فقط العناصر KB الأولى في IPIV يتم تعيينها.
!>
!>             إذا كان IPIV(k) > 0، فقد تم تبديل الصفوف والأعمدة k و IPIV(k)
!>             و D(k,k) هي كتلة قطرية بحجم 1x1.
!>
!>             إذا كان IPIV(k) = IPIV(k+1) < 0، فقد تم تبديل الصفوف والأعمدة
!>             k+1 و -IPIV(k) و D(k:k+1,k:k+1)
!>             هي كتلة قطرية بحجم 2x2.
!>

W

!>          W هي مصفوفة COMPLEX*16، ذات الأبعاد (LDW,NB)
!>

LDW

!>          LDW هو INTEGER
!>          البعد الرائد للمصفوفة W. LDW >= max(1,N).
!>

INFO

!>          INFO هو INTEGER
!>          = 0: خروج ناجح
!>          > 0: إذا كان INFO = k، فإن D(k,k) هو صفر تمامًا. تم إكمال
!>               التحليل، ولكن المصفوفة القطرية الكتلية D فريدة تمامًا.
!>
