subroutine zlagsy (integer n, integer k, double precision, dimension( * ) d, complex*16, dimension( lda, * ) a, integer lda, integer, dimension( 4 ) iseed, complex*16, dimension( * ) work, integer info)

ZLAGSY

الغرض:

!>
!> ZLAGSY يقوم بإنشاء مصفوفة متماثلة مركبة A، عن طريق الضرب المسبق واللاحق
!> لمصفوفة قطرية حقيقية D بمصفوفة وحدوية عشوائية:
!> A = U*D*U**T. يمكن بعد ذلك تقليل عرض النطاق شبه إلى k عن طريق
!> تحويلات وحدوية إضافية.
!>

المعاملات

N 
!>          N هو عدد صحيح
!>          ترتيب المصفوفة A. N >= 0.
!>

K

!>          K هو عدد صحيح
!>          عدد الأقطار الفرعية غير الصفرية داخل نطاق A.
!>          0 <= K <= N-1.
!>

D

!>          D هو مصفوفة ذات دقة مزدوجة، أبعادها (N)
!>          العناصر القطرية للمصفوفة القطرية D.
!>

A

!>          A هي مصفوفة مركبة*16، أبعادها (LDA,N)
!>          المصفوفة المتماثلة N × N التي تم إنشاؤها (يتم تخزين المصفوفة الكاملة).
!>

LDA

!>          LDA هو عدد صحيح
!>          البعد الرئيسي للمصفوفة A. LDA >= N.
!>

ISEED

!>          ISEED هي مصفوفة عدد صحيح، أبعادها (4)
!>          عند الإدخال، بذرة مولد الأرقام العشوائية؛ يجب أن تكون عناصر المصفوفة
!>          بين 0 و 4095، ويجب أن تكون ISEED(4) فردية.
!>          عند الخروج، يتم تحديث البذرة.
!>

WORK

!>          WORK هي مصفوفة مركبة*16، أبعادها (2*N)
!>

INFO

!>          INFO هو عدد صحيح
!>          = 0: خروج ناجح
!>          < 0: إذا كان INFO = -i، فإن المعامل i كان له قيمة غير قانونية
!>
