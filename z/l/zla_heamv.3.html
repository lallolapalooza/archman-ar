subroutine zla_heamv (uplo, n, alpha, a, lda, x, incx, beta, y, incy)

ZLA_HEAMV تحسب ناتج ضرب مصفوفة-متجه باستخدام مصفوفة هرمية غير محددة لحساب حدود الخطأ.

وثائق الوظيفة/الإجراء الفرعي

subroutine zla_heamv (integer uplo, integer n, double precision alpha, complex*16, dimension( lda, * ) a, integer lda, complex*16, dimension( * ) x, integer incx, double precision beta, double precision, dimension( * ) y, integer incy)

ZLA_HEAMV تحسب ناتج ضرب مصفوفة-متجه باستخدام مصفوفة هرمية غير محددة لحساب حدود الخطأ.

الغرض:

!>
!> ZLA_SYAMV  تقوم بعملية المصفوفة-المتجه
!>
!>         y := alpha*abs(A)*abs(x) + beta*abs(y),
!>
!> حيث alpha و beta عددان قياسيان، x و y متجهان، و A مصفوفة متماثلة
!> بحجم n × n.
!>
!> هذه الوظيفة تستخدم بشكل أساسي في حساب حدود الخطأ.
!> للحماية من النقصان أثناء التقييم، يتم إزاحة المكونات في
!> المتجه الناتج بعيداً عن الصفر بمقدار (N+1)
!> مضروبة في عتبة النقصان. لمنع الأخطاء غير الضرورية للمصفوفات
!> المضمنة في هياكل كتلية ضمن المصفوفات العامة،
!> لا يتم إزاحة المكونات الصفرية. يعتبر المدخل
!> صفر إذا كانت جميع عمليات الضرب المتضمنة
!> في حساب هذا المدخل تحتوي على عامل مضروب صفري واحد على الأقل.
!>

!>           دون تغيير عند الخروج.

A

!>          A هي مصفوفة معقدة*16 (COMPLEX*16)، ذات بعد ( LDA, n ).
!>           قبل الدخول، يجب أن يحتوي الجزء الرئيسي بحجم m × n من المصفوفة A
!>           على مصفوفة المعاملات.
!>           دون تغيير عند الخروج.
!>

LDA

!>          LDA هو عدد صحيح (INTEGER)
!>           عند الدخول، يحدد LDA البعد الأول للمصفوفة A كما هو معلن
!>           في البرنامج (الفرعي) المستدعي. يجب أن يكون LDA على الأقل
!>           max( 1, n ).
!>           دون تغيير عند الخروج.
!>

X

!>          X هي مصفوفة معقدة*16 (COMPLEX*16)، ذات بعد على الأقل

!>           قبل الدخول، يجب أن تحتوي المصفوفة X ذات الزيادات على

!>           دون تغيير عند الخروج.
!>

INCX

!>          INCX هو عدد صحيح (INTEGER)
!>           عند الدخول، يحدد INCX الزيادة لعناصر X.
!>           يجب ألا يكون INCX صفرًا.
!>           دون تغيير عند الخروج.
!>

BETA

!>          BETA هو دقة مزدوجة (DOUBLE PRECISION).
!>           عند الدخول، يحدد BETA العدد القياسي beta. عندما يكون BETA
!>           مقدمًا كصفر، لا يلزم تعيين Y عند الإدخال.
!>           دون تغيير عند الخروج.
!>

Y

!>          Y هي مصفوفة دقة مزدوجة (DOUBLE PRECISION)، ذات بعد
!>           ( 1 + ( n - 1 )*abs( INCY ) )
!>           قبل الدخول مع BETA غير صفري، يجب أن تحتوي المصفوفة Y ذات الزيادات على
!>           المتجه y. عند الخروج، يتم استبدال Y بـ
!>           المتجه y المحدث.
!>

INCY
