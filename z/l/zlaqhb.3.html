subroutine zlaqhb (character uplo, integer n, integer kd, complex*16, dimension( ldab, * ) ab, integer ldab, double precision, dimension( * ) s, double precision scond, double precision amax, character equed)

ZLAQHB تقوم بقياس مصفوفة نطاق هيرميتية، باستخدام عوامل القياس المحسوبة بواسطة cpbequ.

الغرض:

!>
!> ZLAQHB تعادل مصفوفة نطاق هيرميتية A
!> باستخدام عوامل القياس في المتجه S.
!>

المعلمات

UPLO 
!>          UPLO هو CHARACTER*1
!>          تحدد ما إذا كان الجزء العلوي أو السفلي المثلثي من المصفوفة
!>          المتناظرة A مخزنًا.
!>          = 'U':  مثلثي علوي
!>          = 'L':  مثلثي سفلي
!>

N

!>          N هو INTEGER
!>          رتبة المصفوفة A.  N >= 0.
!>

KD

!>          KD هو INTEGER
!>          عدد الأقطار الفائقة للمصفوفة A إذا كان UPLO = 'U',
!>          أو عدد الأقطار الفرعية إذا كان UPLO = 'L'.  KD >= 0.
!>

AB

!>          AB هو مصفوفة COMPLEX*16، البعد (LDAB,N)
!>          عند الإدخال، الجزء العلوي أو السفلي المثلثي من مصفوفة النطاق
!>          المتناظرة A، مخزن في أول KD+1 صفوف من المصفوفة.  العمود j
!>          من A مخزن في العمود j من المصفوفة AB
!>          كما يلي:
!>          إذا كان UPLO = 'U'، فإن AB(kd+1+i-j,j) = A(i,j) للقيمة max(1,j-kd)<=i<=j؛
!>          إذا كان UPLO = 'L'، فإن AB(1+i-j,j) = A(i,j) للقيمة j<=i<=min(n,j+kd).
!>
!>          عند الخروج، إذا كان INFO = 0، فإن العامل المثلثي U أو L من
!>          تحليل تشوليسكي A = U**H *U أو A = L*L**H لمصفوفة النطاق
!>          A، بنفس تنسيق التخزين مثل A.
!>

LDAB

!>          LDAB هو INTEGER
!>          البعد الرائد للمصفوفة AB.  LDAB >= KD+1.
!>

S

!>          S هو مصفوفة DOUBLE PRECISION، البعد (N)
!>          عوامل القياس للمصفوفة A.
!>

SCOND

!>          SCOND هو DOUBLE PRECISION
!>          نسبة أصغر S(i) إلى أكبر S(i).
!>

AMAX

!>          AMAX هو DOUBLE PRECISION
!>          القيمة المطلقة لأكبر عنصر في المصفوفة.
!>

EQUED

!>          EQUED هو CHARACTER*1
!>          تحدد ما إذا كان قد تم إجراء القياس أم لا.
!>          = 'N':  لم يتم إجراء أي قياس.
!>          = 'Y':  تم إجراء القياس، أي تم استبدال A بـ
!>                  diag(S) * A * diag(S).
!>

المعلمات الداخلية:

!>  THRESH هو قيمة عتبة تستخدم لتحديد ما إذا كان يجب إجراء القياس
!>  بناءً على نسبة عوامل القياس.  إذا كان SCOND < THRESH،
!>  يتم إجراء القياس.
!>
!>  LARGE و SMALL هي قيم عتبة تستخدم لتحديد ما إذا كان يجب إجراء القياس
!>  بناءً على الحجم المطلق لأكبر عنصر في المصفوفة.
!>  إذا كان AMAX > LARGE أو AMAX < SMALL، يتم إجراء القياس.
!>
