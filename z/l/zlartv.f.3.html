توثيق الوظيفة/الإجراء الفرعي

subroutine zlartv (integer n, complex*16, dimension( * ) x, integer incx, complex*16, dimension( * ) y, integer incy, double precision, dimension( * ) c, complex*16, dimension( * ) s, integer incc)

ZLARTV تطبق متجهًا من دورانات المستوى مع جيب التمام الحقيقية وجيب التمام المركبة على عناصر زوج من المتجهات.

الغرض:

!>
!> ZLARTV تطبق متجهًا من دورانات المستوى المركبة مع جيب تمام حقيقي
!> على عناصر المتجهين x و y. لكل i = 1,2,...,n
!>
!>    ( x(i) ) := (        c(i)   s(i) ) ( x(i) )
!>    ( y(i) )    ( -conjg(s(i))  c(i) ) ( y(i) )
!>

المعلمات

N 
!>          N هو عدد صحيح (INTEGER)
!>          عدد دورانات المستوى المراد تطبيقها.
!>

X

!>          X هو مصفوفة مركبة (COMPLEX*16)، ذات أبعاد (1+(N-1)*INCX)
!>          المتجه x.
!>

INCX

!>          INCX هو عدد صحيح (INTEGER)
!>          الزيادة بين عناصر X. INCX > 0.
!>

Y

!>          Y هي مصفوفة مركبة (COMPLEX*16)، ذات أبعاد (1+(N-1)*INCY)
!>          المتجه y.
!>

INCY

!>          INCY هو عدد صحيح (INTEGER)
!>          الزيادة بين عناصر Y. INCY > 0.
!>

C

!>          C هي مصفوفة ذات دقة مزدوجة (DOUBLE PRECISION)، ذات أبعاد (1+(N-1)*INCC)
!>          جيب التمام لدورانات المستوى.
!>

S

!>          S هي مصفوفة مركبة (COMPLEX*16)، ذات أبعاد (1+(N-1)*INCC)

!>

INCC

!>          INCC هو عدد صحيح (INTEGER)
!>          الزيادة بين عناصر C و S. INCC > 0.
