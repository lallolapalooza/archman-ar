توثيق الوظيفة/الروتين الفرعي

subroutine zlarcm (integer m, integer n, double precision, dimension( lda, * ) a, integer lda, complex*16, dimension( ldb, * ) b, integer ldb, complex*16, dimension( ldc, * ) c, integer ldc, double precision, dimension( * ) rwork)

ZLARCM تقوم بنسخ كل أو جزء من مصفوفة ثنائية الأبعاد حقيقية إلى مصفوفة مركبة.

الغرض:

!>
!> ZLARCM تنفذ عملية ضرب مصفوفات بسيطة جداً:
!>          C := A * B,
!> حيث A هي مصفوفة M × M وحقيقية؛ B هي مصفوفة M × N ومركبة؛
!> C هي مصفوفة M × N ومركبة.
!>

المعلمات

M 
!>          M هو عدد صحيح (INTEGER)
!>          عدد صفوف المصفوفة A والمصفوفة C.
!>          M >= 0.
!>

N

!>          N هو عدد صحيح (INTEGER)
!>          عدد الأعمدة وعدد الصفوف للمصفوفة B،
!>          وعدد الأعمدة للمصفوفة C.
!>          N >= 0.
!>

A

!>          A هي مصفوفة ذات دقة مزدوجة (DOUBLE PRECISION)، بأبعاد (LDA, M)
!>          عند الإدخال، تحتوي A على المصفوفة A ذات الأبعاد M × M.
!>

LDA

!>          LDA هو عدد صحيح (INTEGER)
!>          البعد الرئيسي للمصفوفة A. LDA >=max(1,M).
!>

B

!>          B هي مصفوفة مركبة*16 (COMPLEX*16)، بأبعاد (LDB, N)
!>          عند الإدخال، تحتوي B على المصفوفة B ذات الأبعاد M × N.
!>

LDB

!>          LDB هو عدد صحيح (INTEGER)
!>          البعد الرئيسي للمصفوفة B. LDB >=max(1,M).
!>

C

!>          C هي مصفوفة مركبة*16 (COMPLEX*16)، بأبعاد (LDC, N)
!>          عند الخروج، تحتوي C على المصفوفة C ذات الأبعاد M × N.
!>

LDC

!>          LDC هو عدد صحيح (INTEGER)
!>          البعد الرئيسي للمصفوفة C. LDC >=max(1,M).
!>

RWORK

!>          RWORK هي مصفوفة ذات دقة مزدوجة (DOUBLE PRECISION)، بأبعاد (2*M*N)
!>
