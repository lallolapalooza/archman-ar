subroutine zlaqhp (character uplo, integer n, complex*16, dimension( * ) ap, double precision, dimension( * ) s, double precision scond, double precision amax, character equed)

ZLAQHP تقوم بتوسيع مصفوفة هيرميتية مخزنة بالشكل المكدس.

الغرض:

!>
!> ZLAQHP توحد مصفوفة هيرميتية A باستخدام عوامل القياس
!> في المتجه S.
!>

المعلمات

UPLO 
!>          UPLO هو CHARACTER*1
!>          يحدد ما إذا كان الجزء العلوي أو السفلي المثلثي للمصفوفة
!>          الهيرميتية A مخزنًا.
!>          = 'U':  مثلثي علوي
!>          = 'L':  مثلثي سفلي
!>

N

!>          N هو INTEGER
!>          ترتيب المصفوفة A.  N >= 0.
!>

AP

!>          AP هو مصفوفة COMPLEX*16، البعد (N*(N+1)/2)
!>          عند الإدخال، الجزء العلوي أو السفلي المثلثي للمصفوفة
!>          الهيرميتية A، مكدسة حسب الأعمدة في مصفوفة خطية.  العمود j للمصفوفة A
!>          مخزن في المصفوفة AP كالتالي:
!>          إذا كان UPLO = 'U'، فإن AP(i + (j-1)*j/2) = A(i,j) لـ 1<=i<=j؛
!>          إذا كان UPLO = 'L'، فإن AP(i + (j-1)*(2n-j)/2) = A(i,j) لـ j<=i<=n.
!>
!>          عند الخروج، المصفوفة الموحدة:  diag(S) * A * diag(S)، بنفس
!>          تنسيق التخزين الخاص بـ A.
!>

S

!>          S هي مصفوفة DOUBLE PRECISION، البعد (N)
!>          عوامل القياس لـ A.
!>

SCOND

!>          SCOND هو DOUBLE PRECISION
!>          نسبة أصغر S(i) إلى أكبر S(i).
!>

AMAX

!>          AMAX هو DOUBLE PRECISION
!>          القيمة المطلقة لأكبر عنصر في المصفوفة.
!>

EQUED

!>          EQUED هو CHARACTER*1
!>          يحدد ما إذا تم إجراء التوحيد أم لا.
!>          = 'N':  لم يتم إجراء التوحيد.
!>          = 'Y':  تم إجراء التوحيد، أي تم استبدال A بـ
!>                  diag(S) * A * diag(S).
!>

المعلمات الداخلية:

!>  THRESH هو قيمة حدية تستخدم لتحديد ما إذا كان يجب إجراء القياس
!>  بناءً على نسبة عوامل القياس.  إذا كان SCOND < THRESH،
!>  يتم إجراء القياس.
!>
!>  LARGE و SMALL هما قيمتان حديتان تستخدمان لتحديد ما إذا كان يجب
!>  إجراء القياس بناءً على القيمة المطلقة لأكبر عنصر في المصفوفة.
!>  إذا كان AMAX > LARGE أو AMAX < SMALL، يتم إجراء القياس.
!>
