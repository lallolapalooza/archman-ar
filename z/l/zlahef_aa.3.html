subroutine zlahef_aa (character uplo, integer j1, integer m, integer nb, complex*16, dimension( lda, * ) a, integer lda, integer, dimension( * ) ipiv, complex*16, dimension( ldh, * ) h, integer ldh, complex*16, dimension( * ) work)

ZLAHEF_AA

الغرض:

!>
!> DLAHEF_AA تقوم بتحليل مصفوفة جزئية هيرميتية معقدة A باستخدام
!> خوارزمية Aasen. تتكون المصفوفة الجزئية من مجموعة من صفوف NB من A
!> عندما يكون UPLO هو U، أو مجموعة من أعمدة NB عندما يكون UPLO هو L.
!>
!> لتحليل المصفوفة الجزئية، تتطلب خوارزمية Aasen الصف الأخير، أو العمود الأخير،
!> من المصفوفة الجزئية السابقة. يتم تعيين الصف الأول، أو العمود الأول،
!> من A ليكون الصف الأول، أو العمود الأول، لمصفوفة الوحدة،
!> والتي تستخدم لتحليل المصفوفة الجزئية الأولى.
!>
!> يتم تخزين الصف J الناتج من U، أو العمود J من L، في
!> الصف (J-1)، أو العمود (J-1) من A (بدون أقطار الوحدة)، بينما
!> يتم الكتابة فوق القطر الرئيسي والأقطار الفرعية لـ A بتلك الخاصة بـ T.
!>
!>

المعلمات

UPLO 
!>          UPLO هو CHARACTER*1
!>          = 'U':  يتم تخزين المثلث العلوي لـ A؛
!>          = 'L':  يتم تخزين المثلث السفلي لـ A.
!>

J1

!>          J1 هو INTEGER
!>          موقع الصف الأول، أو العمود الأول، للمصفوفة الجزئية
!>          داخل المصفوفة الفرعية لـ A، التي تم تمريرها إلى هذا الروتين، على سبيل المثال،
!>          عند الاستدعاء بواسطة ZHETRF_AA، بالنسبة للمصفوفة الجزئية الأولى، J1 هو 1،
!>          بينما بالنسبة للمصفوفات الجزئية المتبقية، J1 هو 2.
!>

M

!>          M هو INTEGER
!>          بعد المصفوفة الفرعية. M >= 0.
!>

NB

!>          NB هو INTEGER
!>          بعد المصفوفة الجزئية التي سيتم تحليلها.
!>

A

!>          A هو مصفوفة COMPLEX*16، البعد (LDA,M) لـ
!>          المصفوفة الجزئية الأولى، بينما البعد (LDA,M+1) لـ
!>          المصفوفات الجزئية المتبقية.
!>
!>          عند الدخول، تحتوي A على الصف الأخير، أو العمود الأخير،
!>          من المصفوفة الجزئية السابقة، والمصفوفة الفرعية اللاحقة لـ A
!>          التي سيتم تحليلها، باستثناء المصفوفة الجزئية الأولى، فقط
!>          يتم تمرير المصفوفة الجزئية.
!>
!>          عند الخروج، يتم تحليل المصفوفة الجزئية الرائدة.
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرائد للمصفوفة A. LDA >= max(1,N).
!>

IPIV

!>          IPIV هو مصفوفة INTEGER، البعد (N)
!>          تفاصيل تبديلات الصفوف والأعمدة،
!>          تم تبديل الصف والعمود k مع الصف والعمود IPIV(k).
!>

H

!>          H هي مساحة عمل COMPLEX*16، البعد (LDH,NB).
!>
!>

LDH

!>          LDH هو INTEGER
!>          البعد الرائد لمساحة العمل H. LDH >= max(1,M).
!>

WORK

!>          WORK هي مساحة عمل COMPLEX*16، البعد (M).
!>
