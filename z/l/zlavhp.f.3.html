subroutine zlavhp (character uplo, character trans, character diag, integer n, integer nrhs, complex*16, dimension( * ) a, integer, dimension( * ) ipiv, complex*16, dimension( ldb, * ) b, integer ldb, integer info)

ZLAVHP

الغرض:

!>
!>    ZLAVHP  تقوم بإحدى عمليات المصفوفة-المتجه
!>       x := A*x  أو  x := A^H*x,
!>    حيث x متجه من N عنصر و A هو أحد العوامل
!>    من التحلل المتناظر الذي تم حسابه بواسطة ZHPTRF.
!>    ينتج ZHPTRF تحللاً من الشكل
!>         U * D * U^H     أو     L * D * L^H,
!>    حيث U (أو L) هو ناتج ضرب مصفوفات تبديل ومصفوفات مثلثية عليا (سفلى) وحدة،
!>    U^H (أو L^H) هو المرافق المتبادل لـ U (أو L)، و D هي مصفوفة متناظرة هرمية وشبه قطرية مع كتل 1x1 و 2x2.
!>    يتم تخزين المضاعفات للتحويلات والأجزاء المثلثية العليا أو السفلى من الكتل القطرية
!>    عمودياً في صيغة مضغوطة في المصفوفة الخطية A.
!>
!>    إذا كان TRANS = 'N' أو 'n'، تضرب ZLAVHP إما بـ U أو U * D
!>    (أو L أو L * D).
!>    إذا كان TRANS = 'C' أو 'c'، تضرب ZLAVHP إما بـ U^H أو D * U^H
!>    (أو L^H أو D * L^H).
!> 
!>  UPLO   - CHARACTER*1
!>           عند الإدخال، يحدد UPLO ما إذا كانت المصفوفة المثلثية
!>           المخزنة في A عليا أم سفلى.
!>              UPLO = 'U' أو 'u'   المصفوفة مثلثة عليا.
!>              UPLO = 'L' أو 'l'   المصفوفة مثلثة سفلى.
!>           لم تتغير عند الخروج.
!>
!>  TRANS  - CHARACTER*1
!>           عند الإدخال، يحدد TRANS العملية المراد تنفيذها على النحو التالي:
!>              TRANS = 'N' أو 'n'   x := A*x.
!>              TRANS = 'C' أو 'c'   x := A^H*x.
!>           لم تتغير عند الخروج.
!>
!>  DIAG   - CHARACTER*1
!>           عند الإدخال، يحدد DIAG ما إذا كانت الكتل القطرية
!>           تُفترض أنها مصفوفات وحدة، على النحو التالي:
!>              DIAG = 'U' أو 'u'   الكتل القطرية مصفوفات وحدة.
!>              DIAG = 'N' أو 'n'   الكتل القطرية ليست وحدات.
!>           لم تتغير عند الخروج.
!>
!>  N      - INTEGER
!>           عند الإدخال، يحدد N رتبة المصفوفة A.
!>           يجب أن يكون N أكبر من أو يساوي صفر.
!>           لم يتغير عند الخروج.
!>
!>  NRHS   - INTEGER
!>           عند الإدخال، يحدد NRHS عدد الأطراف اليمنى،
!>           أي عدد المتجهات x التي سيتم ضربها بـ A.
!>           يجب أن يكون NRHS أكبر من أو يساوي صفر.
!>           لم يتغير عند الخروج.
!>
!>  A      - COMPLEX*16 array, dimension( N*(N+1)/2 )
!>           عند الإدخال، تحتوي A على مصفوفة شبه قطرية و
!>           مضاعفات التحويلات المستخدمة للحصول عليها،
!>           مخزنة كمصفوفة مثلثة مضغوطة.
!>           لم تتغير عند الخروج.
!>
!>  IPIV   - INTEGER array, dimension( N )
!>           عند الإدخال، تحتوي IPIV على متجه مؤشرات المحور كما
!>           تم تحديده بواسطة ZSPTRF أو ZHPTRF.
!>           إذا كان IPIV( K ) = K، لم يتم إجراء أي تبديل.
!>           إذا كان IPIV( K ) <> K وكان IPIV( K ) > 0، فتم تبديل الصف K مع
!>           الصف IPIV( K ) وتم استخدام كتلة محور 1x1.
!>           إذا كان IPIV( K ) < 0 وكان UPLO = 'U'، فتم تبديل الصف K-1 مع
!>           الصف | IPIV( K ) | وتم استخدام كتلة محور 2x2.
!>           إذا كان IPIV( K ) < 0 وكان UPLO = 'L'، فتم تبديل الصف K+1 مع
!>           الصف | IPIV( K ) | وتم استخدام كتلة محور 2x2.
!>
!>  B      - COMPLEX*16 array, dimension( LDB, NRHS )
!>           عند الإدخال، تحتوي B على NRHS متجهات بطول N.
!>           عند الخروج، يتم استبدال B بـ A * B.
!>
!>  LDB    - INTEGER
!>           عند الإدخال، يحتوي LDB على البعد الرئيسي لـ B كما
!>           هو معلن في البرنامج المستدعي. يجب أن يكون LDB على الأقل
!>           max( 1, N ).
!>           لم يتغير عند الخروج.
!>
!>  INFO   - INTEGER
!>           INFO هو رمز الخطأ.
!>           عند الخروج، يشير القيمة 0 إلى خروج ناجح.
!>           قيمة سالبة، مثل -K، تشير إلى أن الوسيط K
!>           له قيمة غير قانونية.
!>
