subroutine zlaqr0 (logical wantt, logical wantz, integer n, integer ilo, integer ihi, complex*16, dimension( ldh, * ) h, integer ldh, complex*16, dimension( * ) w, integer iloz, integer ihiz, complex*16, dimension( ldz, * ) z, integer ldz, complex*16, dimension( * ) work, integer lwork, integer info)

ZLAQR0 تحسب القيم الذاتية لمصفوفة هسنبرغ، وبشكل اختياري، المصفوفات من تحلل شور.

الغرض:

!>
!>    ZLAQR0 تحسب القيم الذاتية لمصفوفة هسنبرغ H
!>    وبشكل اختياري، المصفوفات T و Z من تحلل شور
!>    H = Z T Z**H، حيث T مصفوفة مثلثة عليا (شكل شور)،
!>    و Z هي المصفوفة الوحدوية لمتجهات شور.
!>
!>    بشكل اختياري، يمكن ضرب Z في المصفوفة الوحدوية المدخلة Q
!>    بحيث يمكن لهذا الروتين الفرعي إعطاء تحليل شور
!>    لمصفوفة A تم اختزالها إلى شكل هسنبرغ H
!>    بواسطة المصفوفة الوحدوية Q:  A = Q*H*Q**H = (QZ)*H*(QZ)**H.
!>

المعلمات

WANTT 
!>          WANTT هو LOGICAL
!>          = .TRUE. : مطلوب شكل شور الكامل T؛
!>          = .FALSE.: مطلوبة القيم الذاتية فقط.
!>

WANTZ

!>          WANTZ هو LOGICAL
!>          = .TRUE. : مطلوبة مصفوفة متجهات شور Z؛
!>          = .FALSE.: متجهات شور غير مطلوبة.
!>

N

!>          N هو INTEGER
!>           ترتيب المصفوفة H.  N >= 0.
!>

ILO

!>          ILO هو INTEGER
!>

IHI

!>          IHI هو INTEGER
!>
!>           يفترض أن H بالفعل مثلثية عليا في الصفوف والأعمدة 1:ILO-1 و IHI+1:N،
!>           وإذا كان ILO > 1، فإن H(ILO,ILO-1) يساوي صفرًا.
!>           يتم عادةً تعيين ILO و IHI بواسطة استدعاء سابق لـ ZGEBAL،
!>           ثم يتم تمريرهما إلى ZGEHRD عندما يتم اختزال المصفوفة الناتجة
!>           من ZGEBAL إلى شكل هسنبرغ.
!>           خلاف ذلك، يجب تعيين ILO و IHI إلى 1 و N على التوالي.
!>           إذا كان N > 0، فإن 1 <= ILO <= IHI <= N.
!>           إذا كان N = 0، فإن ILO = 1 و IHI = 0.
!>

H

!>          H هي مصفوفة COMPLEX*16، ذات البعد (LDH,N)
!>           عند الإدخال، مصفوفة هسنبرغ العليا H.
!>           عند الخروج، إذا كان INFO = 0 و WANTT هو .TRUE.، فإن H
!>           تحتوي على المصفوفة المثلثية العليا T من تحلل شور
!>           (شكل شور). إذا كان INFO = 0 و WANT هو .FALSE.، فإن محتويات H
!>           غير محددة عند الخروج.
!>           (قيمة الخرج لـ H عندما يكون INFO > 0 معطاة تحت وصف INFO أدناه.)
!>
!>           قد يقوم هذا الروتين الفرعي بتعيين H(i,j) = 0 صراحةً لـ i > j و
!>           j = 1، 2، ... ILO-1 أو j = IHI+1، IHI+2، ... N.
!>

LDH

!>          LDH هو INTEGER
!>           البعد الرائد للمصفوفة H. LDH >= max(1,N).
!>

W

!>          W هي مصفوفة COMPLEX*16، ذات البعد (N)
!>           القيم الذاتية المحسوبة لـ H(ILO:IHI,ILO:IHI) مخزنة
!>           في W(ILO:IHI). إذا كان WANTT هو .TRUE.، فإن القيم الذاتية
!>           مخزنة بنفس الترتيب الذي تظهر به على القطر الرئيسي لشكل شور
!>           المعاد في H، مع W(i) = H(i,i).
!>

ILOZ

!>          ILOZ هو INTEGER
!>

IHIZ

!>          IHIZ هو INTEGER
!>           تحديد الصفوف لـ Z التي يجب تطبيق التحويلات عليها إذا كان WANTZ هو .TRUE..
!>           1 <= ILOZ <= ILO؛ IHI <= IHIZ <= N.
!>

Z

!>          Z هي مصفوفة COMPLEX*16، ذات البعد (LDZ,IHI)
!>           إذا كان WANTZ هو .FALSE.، فلن يتم الرجوع إلى Z.
!>           إذا كان WANTZ هو .TRUE.، يتم استبدال Z(ILO:IHI,ILOZ:IHIZ)
!>           بـ Z(ILO:IHI,ILOZ:IHIZ)*U حيث U هي
!>           عامل شور الوحدوي لـ H(ILO:IHI,ILO:IHI).
!>           (قيمة الخرج لـ Z عندما يكون INFO > 0 معطاة تحت وصف INFO أدناه.)
!>

LDZ

!>          LDZ هو INTEGER
!>           البعد الرائد للمصفوفة Z. إذا كان WANTZ هو .TRUE.
!>           فإن LDZ >= MAX(1,IHIZ). خلاف ذلك، LDZ >= 1.
!>

WORK

!>          WORK هي مصفوفة COMPLEX*16، ذات البعد LWORK
!>           عند الخروج، إذا كان LWORK = -1، فإن WORK(1) يعيد تقديرًا
!>           للقيمة المثلى لـ LWORK.
!>

LWORK

!>          LWORK هو INTEGER
!>           بعد المصفوفة WORK.  LWORK >= max(1,N)
!>           كافٍ، ولكن LWORK بحجم 6*N قد يكون مطلوبًا للحصول على أداء مثالي.
!>           يوصى باستعلام مساحة العمل لتحديد حجم مساحة العمل المثلى.
!>
!>           إذا كان LWORK = -1، فإن ZLAQR0 يقوم باستعلام مساحة العمل.
!>           في هذه الحالة، يتحقق ZLAQR0 من معلمات الإدخال ويقدر
!>           حجم مساحة العمل الأمثل للقيم N، ILO، و IHI المعطاة.
!>           يتم إرجاع التقدير في WORK(1). لا يتم إصدار رسائل خطأ متعلقة بـ LWORK
!>           بواسطة XERBLA. لا يتم الوصول إلى H أو Z.
!>

INFO

!>          INFO هو INTEGER
!>             = 0:  انتهاء ناجح
!>             > 0:  إذا كان INFO = i، فشلت ZLAQR0 في حساب جميع
!>                القيم الذاتية. العناصر 1:ilo-1 و i+1:n من WR
!>                و WI تحتوي على تلك القيم الذاتية التي تم
!>                حسابها بنجاح. (الفشل نادر.)
!>
!>                إذا كان INFO > 0 وكان WANT هو .FALSE.، فعند الخروج،
!>                القيم الذاتية المتبقية غير المتقاربة هي القيم الذاتية
!>                لمصفوفة هسنبرغ العليا في الصفوف والأعمدة ILO إلى INFO
!>                من القيمة النهائية لـ H.
!>
!>                إذا كان INFO > 0 وكان WANTT هو .TRUE.، فعند الخروج
!>
!>           (*)  (القيمة الأولية لـ H)*U  = U*(القيمة النهائية لـ H)
!>
!>                حيث U هي مصفوفة وحدوية. القيمة النهائية لـ H
!>                هي هسنبرغ عليا ومثلثة في الصفوف والأعمدة INFO+1 إلى IHI.
!>
!>                إذا كان INFO > 0 وكان WANTZ هو .TRUE.، فعند الخروج
!>
!>                  (القيمة النهائية لـ Z(ILO:IHI,ILOZ:IHIZ)
!>                   =  (القيمة الأولية لـ Z(ILO:IHI,ILOZ:IHIZ)*U
!>
!>                حيث U هي المصفوفة الوحدوية في (*) (بغض النظر عن قيمة WANTT).
!>
!>                إذا كان INFO > 0 وكان WANTZ هو .FALSE.، فلن يتم الوصول إلى Z.
!>
