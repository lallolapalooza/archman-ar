subroutine zlatmr (integer m, integer n, character dist, integer, dimension( 4 ) iseed, character sym, complex*16, dimension( * ) d, integer mode, double precision cond, complex*16 dmax, character rsign, character grade, complex*16, dimension( * ) dl, integer model, double precision condl, complex*16, dimension( * ) dr, integer moder, double precision condr, character pivtng, integer, dimension( * ) ipivot, integer kl, integer ku, double precision sparse, double precision anorm, character pack, complex*16, dimension( lda, * ) a, integer lda, integer, dimension( * ) iwork, integer info)

ZLATMR

الغرض:

!>
!>    ZLATMR يولد مصفوفات عشوائية من أنواع مختلفة لاختبار برامج LAPACK.
!>
!>    يعمل ZLATMR عن طريق تطبيق التسلسل التالي من
!>    العمليات:
!>
!>      توليد مصفوفة A بإدخالات عشوائية من التوزيع DIST
!>         التي تكون متناظرة إذا كان SYM='S'، هرميتية إذا كان SYM='H'،
!>         وغير متناظرة إذا كان SYM='N'.
!>
!>      ضبط القطر إلى D، حيث يمكن أن يكون D مدخلًا أو
!>         محسوبًا وفقًا لـ MODE، COND، DMAX و RSIGN
!>         كما هو موضح أدناه.
!>
!>      تدرج المصفوفة، إذا رغبت في ذلك، من اليسار و/أو اليمين
!>         كما هو محدد بواسطة GRADE. المدخلات DL، MODEL، CONDL، DR،
!>         MODER و CONDR تحدد أيضًا التدرج كما هو موضح
!>         أدناه.
!>
!>      تبديل الصفوف و/أو الأعمدة، إذا رغبت في ذلك، كما هو محدد بواسطة
!>         PIVTNG و IPIVOT.
!>
!>      تعيين إدخالات عشوائية إلى الصفر، إذا رغبت في ذلك، للحصول على مصفوفة
!>         عشوائية متفرقة كما هو محدد بواسطة SPARSE.
!>
!>      اجعل A مصفوفة نطاقية، إذا رغبت في ذلك، عن طريق تصفير المصفوفة
!>         خارج نطاق النطاق السفلي KL والنطاق العلوي KU.
!>
!>      قياس A، إذا رغبت في ذلك، للحصول على الحد الأقصى للإدخال ANORM.
!>
!>      حزم المصفوفة إذا رغبت في ذلك. الخيارات المحددة بواسطة PACK هي:
!>         لا يوجد حزم
!>         تصفير النصف العلوي (إذا كان متناظرًا أو هرميًا)
!>         تصفير النصف السفلي (إذا كان متناظرًا أو هرميًا)
!>         تخزين النصف العلوي عموديًا (إذا كان متناظرًا أو هرميًا
!>             أو مصفوفة مربعة مثلثية علوية)
!>         تخزين النصف السفلي عموديًا (إذا كان متناظرًا أو هرميًا
!>             أو مصفوفة مربعة مثلثية سفلية)
!>             نفس النصف العلوي صفيًا إذا كان متناظرًا
!>             نفس النصف العلوي المرافق صفيًا إذا كان هرميًا
!>         تخزين المثلث السفلي بتنسيق نطاقي
!>             (إذا كان متناظرًا أو هرميًا)
!>         تخزين المثلث العلوي بتنسيق نطاقي
!>             (إذا كان متناظرًا أو هرميًا)
!>         تخزين المصفوفة بأكملها بتنسيق نطاقي
!>
!>    ملاحظة: إذا كان هناك استدعاءان لـ ZLATMR يختلفان فقط في المعلمة PACK،
!>          فسيولدون مصفوفات مكافئة رياضيًا.
!>
!>          إذا كان هناك استدعاءان لـ ZLATMR بنفس النطاق الكامل (KL = M-1
!>          و KU = N-1)، ويختلفان فقط في معلمتي PIVTNG و PACK،
!>          فستختلف المصفوفات المولدة فقط
!>          في ترتيب الصفوف و/أو الأعمدة، وتحتوي بخلاف ذلك
!>          على نفس البيانات. لا يمكن الحفاظ على هذه الاتساق
!>          ولا يتم الحفاظ عليه بنطاق أقل من النطاق الكامل.
!>

المعلمات

M 
!>          M هو عدد صحيح
!>           عدد صفوف A. لم يتم تعديله.
!>

N

!>          N هو عدد صحيح
!>           عدد أعمدة A. لم يتم تعديله.
!>

DIST

!>          DIST هو حرف*1
!>           عند الإدخال، يحدد DIST نوع التوزيع الذي سيتم استخدامه
!>           لتوليد مصفوفة عشوائية.
!>           'U' => الأجزاء الحقيقية والخيالية مستقلة
!>                  موزعة بانتظام (0, 1) ( 'U' للتوزيع المنتظم )
!>           'S' => الأجزاء الحقيقية والخيالية مستقلة
!>                  موزعة بانتظام (-1, 1) ( 'S' للمتناظر )
!>           'N' => الأجزاء الحقيقية والخيالية مستقلة
!>                  عادية (0, 1) ( 'N' للعادي )
!>           'D' => موزع بانتظام على داخل القرص الوحدوي ( 'D' للقرص )
!>           لم يتم تعديله.
!>

ISEED

!>          ISEED هي مصفوفة عدد صحيح، بعد (4)
!>           عند الإدخال، يحدد ISEED بذرة مولد الأرقام العشوائية.
!>           يجب أن تكون بين 0 و 4095 شاملة، ويجب أن يكون ISEED(4) فرديًا.
!>           يستخدم مولد الأرقام العشوائية تسلسلًا خطيًا توافقيًا محدودًا بأعداد
!>           صحيحة صغيرة، وبالتالي يجب أن ينتج أرقامًا عشوائية مستقلة عن الآلة.
!>           تتغير قيم ISEED عند الخروج، ويمكن استخدامها في الاستدعاء التالي لـ
!>           ZLATMR لمواصلة نفس تسلسل الأرقام العشوائية.
!>           تغير عند الخروج.
!>

SYM

!>          SYM هو حرف*1
!>           إذا كان SYM='S'، فستكون المصفوفة المولدة متناظرة.
!>           إذا كان SYM='H'، فستكون المصفوفة المولدة هرمية.
!>           إذا كان SYM='N'، فستكون المصفوفة المولدة غير متناظرة.
!>           لم يتم تعديله.
!>

D

!>          D هي مصفوفة COMPLEX*16، بعد (min(M,N))
!>           عند الإدخال، تحدد هذه المصفوفة إدخالات القطر
!>           للقطر الرئيسي لـ A. قد يتم تحديد D
!>           عند الإدخال، أو يتم تعيينه وفقًا لـ MODE و COND كما هو موضح
!>           أدناه. إذا كانت المصفوفة هرمية، سيتم أخذ الجزء الحقيقي من D.
!>           قد يتغير عند الخروج إذا كان MODE غير صفري.
!>

MODE

!>          MODE هو عدد صحيح
!>           عند الإدخال يصف كيفية استخدام D:
!>           MODE = 0 يعني استخدام D كمدخل
!>           MODE = 1 يضبط D(1)=1 و D(2:N)=1.0/COND
!>           MODE = 2 يضبط D(1:N-1)=1 و D(N)=1.0/COND
!>           MODE = 3 يضبط D(I)=COND**(-(I-1)/(N-1))
!>           MODE = 4 يضبط D(i)=1 - (i-1)/(N-1)*(1 - 1/COND)
!>           MODE = 5 يضبط D لأرقام عشوائية في النطاق
!>                    ( 1/COND , 1 ) بحيث تكون لوغاريتماتها
!>                    موزعة بانتظام.
!>           MODE = 6 يضبط D لأرقام عشوائية من نفس التوزيع
!>                    مثل بقية المصفوفة.
!>           MODE < 0 له نفس المعنى مثل ABS(MODE)، باستثناء أن
!>              ترتيب عناصر D يتم عكسه.
!>           لذلك، إذا كان MODE موجبًا، فإن D يحتوي على إدخالات تتراوح من
!>              1 إلى 1/COND، وإذا كان سالبًا، من 1/COND إلى 1،
!>           لم يتم تعديله.
!>

COND

!>          COND هو DOUBLE PRECISION
!>           عند الإدخال، يستخدم كما هو موضح تحت MODE أعلاه.
!>           إذا تم استخدامه، يجب أن يكون >= 1. لم يتم تعديله.
!>

DMAX

!>          DMAX هو COMPLEX*16
!>           إذا كان MODE ليس -6 أو 0 أو 6، يتم قياس القطر بـ
!>           DMAX / max(abs(D(i)))، بحيث يكون الحد الأقصى للقيمة المطلقة
!>           للقطر هو abs(DMAX). إذا كان DMAX مركبًا (أو صفرًا)،
!>           سيتم قياس القطر برقم مركب (أو صفر).
!>

RSIGN

!>          RSIGN هو حرف*1
!>           إذا كان MODE ليس -6 أو 0 أو 6، يحدد إشارة القطر
!>           على النحو التالي:
!>           'T' => يتم ضرب إدخالات القطر برقم مركب عشوائي
!>                  موزع بانتظام بقيمة مطلقة 1
!>           'F' => القطر غير معدل
!>           لم يتم تعديله.
!>

GRADE

!>          GRADE هو حرف*1
!>           إذا كان GRADE='L'، يتم تدرج المصفوفة من اليسار.
!>           إذا كان GRADE='R'، يتم تدرج المصفوفة من اليمين.
!>           إذا كان GRADE='B'، يتم تدرج المصفوفة من اليسار واليمين.
!>           إذا كان GRADE='N'، لا يتم تدرج المصفوفة.
!>           لم يتم تعديله.
!>

DL

!>          DL هي COMPLEX*16 array, dimension (M)
!>           إذا كان GRADE='L' أو 'B'، يحدد DL عوامل التدرج اليسرى.
!>           لم يتم تعديله.
!>

MODEL

!>          MODEL هو INTEGER
!>           إذا كان GRADE='L' أو 'B'، يحدد MODEL كيفية استخدام DL.
!>           MODE=0 يعني استخدام DL كمدخل.
!>           MODE=1 يعني تعيين DL إلى أرقام عشوائية.
!>           MODE<0 له نفس المعنى مثل ABS(MODE)، باستثناء أن
!>              ترتيب عناصر DL يتم عكسه.
!>           لم يتم تعديله.
!>

CONDL

!>          CONDL هو DOUBLE PRECISION
!>           إذا كان GRADE='L' أو 'B'، يتم استخدامه لتحديد نطاق DL
!>           عندما يكون MODEL=1. يجب أن يكون >= 1.
!>           لم يتم تعديله.
!>

DR

!>          DR هي COMPLEX*16 array, dimension (N)
!>           إذا كان GRADE='R' أو 'B'، يحدد DR عوامل التدرج اليمنى.
!>           لم يتم تعديله.
!>

MODER

!>          MODER هو INTEGER
!>           إذا كان GRADE='R' أو 'B'، يحدد MODER كيفية استخدام DR.
!>           MODE=0 يعني استخدام DR كمدخل.
!>           MODE=1 يعني تعيين DR إلى أرقام عشوائية.
!>           MODE<0 له نفس المعنى مثل ABS(MODE)، باستثناء أن
!>              ترتيب عناصر DR يتم عكسه.
!>           لم يتم تعديله.
!>

CONDR

!>          CONDR هو DOUBLE PRECISION
!>           إذا كان GRADE='R' أو 'B'، يتم استخدامه لتحديد نطاق DR
!>           عندما يكون MODER=1. يجب أن يكون >= 1.
!>           لم يتم تعديله.
!>

PIVTNG

!>          PIVTNG هو CHARACTER*1
!>           إذا كان PIVTNG='R'، يتم تبديل الصفوف عشوائيًا.
!>           إذا كان PIVTNG='C'، يتم تبديل الأعمدة عشوائيًا.
!>           إذا كان PIVTNG='B'، يتم تبديل الصفوف والأعمدة عشوائيًا.
!>           إذا كان PIVTNG='N'، لا يتم تبديل الصفوف أو الأعمدة.
!>           لم يتم تعديله.
!>

IPIVOT

!>          IPIVOT هو INTEGER array, dimension (max(M,N))
!>           إذا كان PIVTNG='R' أو 'C' أو 'B'، يتم تخزين التباديل
!>           في IPIVOT.
!>           لم يتم تعديله.
!>

KL

!>          KL هو INTEGER
!>           إذا كان KL > 0، يتم وضع المصفوفة في شكل نطاقي
!>           مع نطاق سفلي KL.
!>           لم يتم تعديله.
!>

KU

!>          KU هو INTEGER
!>           إذا كان KU > 0، يتم وضع المصفوفة في شكل نطاقي
!>           مع نطاق علوي KU.
!>           لم يتم تعديله.
!>

SPARSE

!>          SPARSE هو DOUBLE PRECISION
!>           إذا كان SPARSE > 0، يتم وضع نسبة SPARSE من الإدخالات
!>           بشكل عشوائي إلى الصفر.
!>           لم يتم تعديله.
!>

ANORM

!>          ANORM هو DOUBLE PRECISION
!>           إذا كان ANORM /= 0، يتم قياس المصفوفة بحيث يكون الحد الأقصى
!>           للقيمة المطلقة للإدخالات هو ANORM.
!>           لم يتم تعديله.
!>

PACK

!>          PACK هو CHARACTER*1
!>           إذا كان PACK='U'، يتم تخزين النصف العلوي (بما في ذلك القطر)
!>           في الجزء السفلي من المصفوفة A.
!>           إذا كان PACK='L'، يتم تخزين النصف السفلي (بما في ذلك القطر)
!>           في الجزء العلوي من المصفوفة A.
!>           لم يتم تعديله.
!>

A

!>          A هي COMPLEX*16 array, dimension (LDA, N)
!>           المصفوفة المراد توليدها.
!>           يتم تعديله.
!>

LDA

!>          LDA هو INTEGER
!>           عدد الصفوف الفعلي للمصفوفة A. لم يتم تعديله.
!>

IWORK

!>          IWORK هو INTEGER array, dimension (max(M,N))
!>           مساحة عمل.
!>           لم يتم تعديله.
!>

INFO

!>          INFO هو INTEGER
!>           = 0: صحيح
!>           < 0: إذا كان INFO = -i، فإن المعلمة i-th في الاستدعاء
!>                كانت غير صالحة.
!>           > 0: إذا كان INFO = 1، فإن مصفوفة القطر كانت فارغة.
!>

هذا الموقع غير مرتبط بـ Arch Linux ولا تابع له بأي شكل من الأشكال. المحتوى هو ترجمة فقط لصفحات دليل Arch المنشورة على man.archlinux.org.

anged !> لم يتم التعديل. !>

GRADE

!>          GRADE هو CHARACTER*1
!>           يحدد تصنيف المصفوفة على النحو التالي:
!>           'N'  => لا تصنيف
!>           'L'  => المصفوفة مضروبة مسبقًا في diag( DL )
!>                   (فقط إذا كانت المصفوفة غير متناظرة)
!>           'R'  => المصفوفة مضروبة لاحقًا في diag( DR )
!>                   (فقط إذا كانت المصفوفة غير متناظرة)
!>           'B'  => المصفوفة مضروبة مسبقًا في diag( DL ) و
!>                         مضروبة لاحقًا في diag( DR )
!>                   (فقط إذا كانت المصفوفة غير متناظرة)
!>           'H'  => المصفوفة مضروبة مسبقًا في diag( DL ) و
!>                         مضروبة لاحقًا في diag( CONJG(DL) )
!>                   (فقط إذا كانت المصفوفة مترافقة أو غير متناظرة)
!>           'S'  => المصفوفة مضروبة مسبقًا في diag( DL ) و
!>                         مضروبة لاحقًا في diag( DL )
!>                   (فقط إذا كانت المصفوفة متناظرة أو غير متناظرة)
!>           'E'  => المصفوفة مضروبة مسبقًا في diag( DL ) و
!>                         مضروبة لاحقًا في inv( diag( DL ) )
!>                         ( 'S' للتشابه )
!>                   (فقط إذا كانت المصفوفة غير متناظرة)
!>                   ملاحظة: إذا كانت GRADE='S'، فيجب أن يساوي M يساوي N.
!>           لم يتم التعديل.
!>

DL

!>          DL هو COMPLEX*16 array, dimension (M)
!>           إذا كان MODEL=0، فعند الدخول تحدد هذه المصفوفة المدخلات القطرية
!>           لمصفوفة قطرية تستخدم كما هو موضح تحت GRADE
!>           أعلاه. إذا لم يكن MODEL صفرًا، فسيتم تعيين DL
!>           وفقًا لـ MODEL و CONDL، بشكل مشابه للطريقة التي يتم بها تعيين D
!>           وفقًا لـ MODE و COND (باستثناء عدم وجود معلمة DMAX لـ DL).
!>           إذا كانت GRADE='E'، فلا يمكن أن تحتوي DL على مدخلات صفرية.
!>           لا تتم الإشارة إليها إذا كانت GRADE = 'N' أو 'R'. تم تغييرها عند الخروج.
!>

MODEL

!>          MODEL هو INTEGER
!>           يحدد هذا كيف سيتم حساب مصفوفة القطر DL،
!>           تمامًا كما يحدد MODE كيف سيتم حساب D.
!>           لم يتم التعديل.
!>

CONDL

!>          CONDL هو DOUBLE PRECISION
!>           عندما لا يكون MODEL صفرًا، يحدد هذا رقم الشرط
!>           لـ DL المحسوب. لم يتم التعديل.
!>

DR

!>          DR هو COMPLEX*16 array, dimension (N)
!>           إذا كان MODER=0، فعند الدخول تحدد هذه المصفوفة المدخلات القطرية
!>           لمصفوفة قطرية تستخدم كما هو موضح تحت GRADE
!>           أعلاه. إذا لم يكن MODER صفرًا، فسيتم تعيين DR
!>           وفقًا لـ MODER و CONDR، بشكل مشابه للطريقة التي يتم بها تعيين D
!>           وفقًا لـ MODE و COND (باستثناء عدم وجود معلمة DMAX لـ DR).
!>           لا تتم الإشارة إليها إذا كانت GRADE = 'N' أو 'L' أو 'H' أو 'S'.
!>           تم تغييرها عند الخروج.
!>

MODER

!>          MODER هو INTEGER
!>           يحدد هذا كيف سيتم حساب مصفوفة القطر DR،
!>           تمامًا كما يحدد MODE كيف سيتم حساب D.
!>           لم يتم التعديل.
!>

CONDR

!>          CONDR هو DOUBLE PRECISION
!>           عندما لا يكون MODER صفرًا، يحدد هذا رقم الشرط
!>           لـ DR المحسوب. لم يتم التعديل.
!>

PIVTNG

!>          PIVTNG هو CHARACTER*1
!>           عند الدخول يحدد تباديل المحاور على النحو التالي:
!>           'N' أو ' ' => لا يوجد.
!>           'L' => محاور يسارية أو صفية (يجب أن تكون المصفوفة غير متناظرة).
!>           'R' => محاور يمينية أو عمودية (يجب أن تكون المصفوفة
!>                  غير متناظرة).
!>           'B' أو 'F' => كلاهما أو محاور كاملة، أي على كلا الجانبين.
!>                         في هذه الحالة، يجب أن يساوي M يساوي N
!>
!>           إذا كان استدعاء ZLATMR مرتين لهما نفس النطاق الكامل (KL = M-1
!!>           و KU = N-1)، ويختلفان فقط في معلمتي PIVTNG و PACK
!>           فستنتج المصفوفات المتولدة فقط في ترتيب الصفوف و/أو الأعمدة، وبخلاف ذلك
!>           تحتوي على نفس البيانات. لا يمكن الحفاظ على هذا الاتساق
!>           مع نطاق أقل من النطاق الكامل.
!>

IPIVOT

!>          IPIVOT هو INTEGER array, dimension (N or M)
!>           تحدد هذه المصفوفة التباديل المستخدمة. بعد
!!>           يتم إنشاء المصفوفة الأساسية، يتم تبديل الصفوف أو الأعمدة أو كليهما.
!>           إذا، على سبيل المثال، تم تحديد تبديل الصفوف، يبدأ ZLATMR
!>           من الصف *الأخير* ويتبادل الصف M و IPIVOT(M)،
!>           ثم ينتقل إلى الصف قبل الأخير،
!>           يتبادل الصف (M-1) و IPIVOT(M-1)،
!>           وهكذا. من حيث ، التبديل هو
!>           (1 IPIVOT(1)) (2 IPIVOT(2)) ... (M IPIVOT(M))
!>           حيث يتم تطبيق الدورة الأقصى يمينًا أولاً. هذا هو
!>           *معكوس* تأثير التبديل في LINPACK. الفكرة
!>           هي أن عامل (مع التبديل) مصفوفة هوية
!>           التي تم تبديلها عكسيًا بهذه الطريقة يجب
!>           أن ينتج متجه محور مطابق لـ IPIVOT.
!>           لا تتم الإشارة إليها إذا كان PIVTNG = 'N'. لم يتم التعديل.
!>

KL

!>          KL هو INTEGER
!>           عند الدخول يحدد النطاق السفلي للمصفوفة. على سبيل المثال،
!>           KL=0 يعني مثلثي علوي، KL=1 يعني هسنبرج علوي، و KL على الأقل M-1
!>           يعني أن المصفوفة ليست نطاقية. يجب أن يساوي KU إذا كانت المصفوفة متناظرة أو مترافقة.
!>           لم يتم التعديل.
!>

KU

!>          KU هو INTEGER
!>           عند الدخول يحدد النطاق العلوي للمصفوفة. على سبيل المثال،
!>           KU=0 يعني مثلثي سفلي، KU=1 يعني هسنبرج سفلي، و KU على الأقل N-1
!>           يعني أن المصفوفة ليست نطاقية. يجب أن يساوي KL إذا كانت المصفوفة متناظرة أو مترافقة.
!>           لم يتم التعديل.
!>

SPARSE

!>          SPARSE هو DOUBLE PRECISION
!>           عند الدخول يحدد ندرة المصفوفة إذا كان سيتم إنشاء مصفوفة نادرة. يجب أن تكون SPARSE بين
!>           0 و 1. لإنشاء مصفوفة نادرة، لكل مدخل مصفوفة
!>           يتم إنشاء رقم عشوائي موحد (0، 1) x ويتم
!>           مقارنته بـ SPARSE؛ إذا كان x أكبر، فإن مدخل المصفوفة
!>           يبقى دون تغيير، وإذا كان x أصغر، يتم تعيين المدخل
!>           إلى الصفر. وبالتالي، في المتوسط، سيتم تعيين نسبة fração من
!>           المدخلات إلى الصفر.
!>           لم يتم التعديل.
!>

ANORM

!>          ANORM هو DOUBLE PRECISION
!>           عند الدخول يحدد الحد الأقصى لمدخل المصفوفة الناتجة
!>           (سيتم ضرب المصفوفة الناتجة في ثابت بحيث
!>           يساوي أكبر مدخل لها المطلق ANORM)
!>           إذا كان ANORM غير سالب. إذا كان ANORM سالبًا، فلا يتم
!>           تطبيق أي قياس. لم يتم التعديل.
!>

PACK

!>          PACK هو CHARACTER*1
!>           عند الدخول يحدد تعبئة المصفوفة على النحو التالي:
!>           'N' => لا تعبئة
!>           'U' => تصفير جميع المدخلات تحت القطر
!>                  (إذا كانت متناظرة أو مترافقة)
!>           'L' => تصفير جميع المدخلات فوق القطر
!>                  (إذا كانت متناظرة أو مترافقة)
!>           'C' => تخزين المثلث العلوي عموديًا
!>                  (فقط إذا كانت المصفوفة متناظرة أو مترافقة أو
!>                   مثلثية علوية مربعة)
!>           'R' => تخزين المثلث السفلي عموديًا
!>                  (فقط إذا كانت المصفوفة متناظرة أو مترافقة أو
!>                   مثلثية سفلية مربعة)
!>                  (نفس النصف العلوي صفياً إذا كانت متناظرة)
!>                  (نفس النصف العلوي المترافق صفياً إذا كانت مترافقة)
!>           'B' => تخزين المثلث السفلي في مخطط تخزين نطاقي
!>                  (فقط إذا كانت المصفوفة متناظرة أو مترافقة)
!>           'Q' => تخزين المثلث العلوي في مخطط تخزين نطاقي
!>                  (فقط إذا كانت المصفوفة متناظرة أو مترافقة)
!>           'Z' => تخزين المصفوفة بأكملها في مخطط تخزين نطاقي
!>                      (يمكن توفير التبديل باستخدام هذا
!>                      الخيار لتخزين A في الصفوف اللاحقة من
!>                      التخزين المخصص)
!>
!>           باستخدام هذه الخيارات، يمكن الحصول على مخططات التخزين المعبأة والنطاقية المختلفة في LAPACK:
!>           GB               - استخدام 'Z'
!>           PB، HB أو TB     - استخدام 'B' أو 'Q'
!>           PP، HP أو TP     - استخدام 'C' أو 'R'
!>
!>           إذا كان استدعاء ZLATMR مرتين يختلفان فقط في معلمة PACK،
!>           فسينتجان مصفوفات متكافئة رياضيًا.
!>           لم يتم التعديل.
!>

A

!>          A هو COMPLEX*16 array, dimension (LDA,N)
!>           عند الخروج A هي مصفوفة الاختبار المطلوبة. فقط
!>           سيتم الإشارة إلى المدخلات المهمة لـ A عند الخروج
!>           (حتى لو كانت A في صيغة تخزين معبأة أو نطاقية).
!>           سيتم تصفير "الزوايا غير المشغولة" لـ A في
!>           تنسيق النطاق.
!>

LDA

!>          LDA هو INTEGER
!>           عند الدخول يحدد LDA البعد الأول لـ A كما هو
!>           معلن في البرنامج المستدعي.
!>           إذا كان PACK='N' أو 'U' أو 'L'، فيجب أن يكون LDA على الأقل max(1، M).
!>           إذا كان PACK='C' أو 'R'، فيجب أن يكون LDA على الأقل 1.
!>           إذا كان PACK='B' أو 'Q'، فيجب أن يكون LDA هو MIN(KU+1، N).
!>           إذا كان PACK='Z'، فيجب أن يكون LDA على الأقل KUU+KLL+1، حيث
!>           KUU = MIN(KU، N-1) و KLL = MIN(KL، M-1)
!>           لم يتم التعديل.
!>

IWORK

!>          IWORK هو INTEGER array, dimension (N or M)
!>           مساحة العمل. لا تتم الإشارة إليها إذا كان PIVTNG = 'N'. تم تغييرها عند الخروج.
!>

INFO

!>          INFO هو INTEGER
!>           معلمة الخطأ عند الخروج:
!>             0 => عودة طبيعية
!>            -1 => M سالب أو غير مساوٍ لـ N و SYM='S' أو 'H'
!>            -2 => N سالب
!>            -3 => DIST سلسلة غير صالحة
!>            -5 => SYM سلسلة غير صالحة
!>            -7 => MODE خارج النطاق -6 إلى 6
!>            -8 => COND أقل من 1.0، و MODE ليس -6 أو 0 أو 6
!>           -10 => MODE ليس -6 أو 0 أو 6 و RSIGN سلسلة غير صالحة
!>           -11 => GRADE سلسلة غير صالحة، أو GRADE='E' و
!>                  M غير مساوٍ لـ N، أو GRADE='L' أو 'R' أو 'B' أو 'S' أو 'E'
!>                  و SYM = 'H'، أو GRADE='L' أو 'R' أو 'B' أو 'H' أو 'E'
!>                  و SYM = 'S'
!>           -12 => GRADE = 'E' و DL يحتوي على صفر
!>           -13 => MODEL خارج النطاق -6 إلى 6 و GRADE= 'L' أو 'B' أو 'H' أو
!>                  'S' أو 'E'
!>           -14 => CONDL أقل من 1.0، GRADE='L' أو 'B' أو 'H' أو 'S' أو 'E'،
!>                  و MODEL ليس -6 أو 0 أو 6
!>           -16 => MODER خارج النطاق -6 إلى 6 و GRADE= 'R' أو 'B'
!>           -17 => CONDR أقل من 1.0، GRADE='R' أو 'B'، و
!>                  MODER ليس -6 أو 0 أو 6
!>           -18 => PIVTNG سلسلة غير صالحة، أو PIVTNG='B' أو 'F' و
!>                  M غير مساوٍ لـ N، أو PIVTNG='L' أو 'R' و SYM='S'
!>                  أو 'H'
!>           -19 => IPIVOT يحتوي على رقم خارج النطاق و
!>                  PIVTNG ليس 'N'
!>           -20 => KL سالب
!>           -21 => KU سالب، أو SYM='S
