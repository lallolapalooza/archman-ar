subroutine zlaqhe (character uplo, integer n, complex*16, dimension( lda, * ) a, integer lda, double precision, dimension( * ) s, double precision scond, double precision amax, character equed)

ZLAQHE يقوم بقياس مصفوفة متناظرة هرميتية.

الغرض:

!>
!> ZLAQHE يعادل مصفوفة متناظرة هرميتية A باستخدام عوامل القياس
!> في المتجه S.
!>

المعلمات

UPLO 
!>          UPLO هو CHARACTER*1
!>          يحدد ما إذا كان الجزء العلوي أو السفلي من المصفوفة الهرميتية A
!>          مخزنًا.
!>          = 'U':  مثلثي علوي
!>          = 'L':  مثلثي سفلي
!>

N

!>          N هو INTEGER
!>          رتبة المصفوفة A.  N >= 0.
!>

A

!>          A هو مصفوفة COMPLEX*16، البعد (LDA,N)
!>          عند الإدخال، المصفوفة الهرميتية A.  إذا كان UPLO = 'U'، فإن الجزء
!>          المثلثي العلوي الأيسر بحجم n*n من A يحتوي على الجزء
!>          المثلثي العلوي للمصفوفة A، والجزء المثلثي السفلي الصارم
!>          من A لا تتم الإشارة إليه.  إذا كان UPLO = 'L'، فإن الجزء
!>          المثلثي السفلي الأيسر بحجم n*n من A يحتوي على الجزء
!>          المثلثي السفلي للمصفوفة A، والجزء المثلثي العلوي الصارم
!>          من A لا تتم الإشارة إليه.
!>
!>          عند الخروج، إذا كان EQUED = 'Y'، فإن المصفوفة المعادلة:
!>          diag(S) * A * diag(S).
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرائد للمصفوفة A.  LDA >= max(N,1).
!>

S

!>          S هو مصفوفة DOUBLE PRECISION، البعد (N)
!>          عوامل القياس لـ A.
!>

SCOND

!>          SCOND هو DOUBLE PRECISION
!>          نسبة أصغر S(i) إلى أكبر S(i).
!>

AMAX

!>          AMAX هو DOUBLE PRECISION
!>          القيمة المطلقة لأكبر عنصر في المصفوفة.
!>

EQUED

!>          EQUED هو CHARACTER*1
!>          يحدد ما إذا كان قد تم إجراء القياس أم لا.
!>          = 'N':  لم يتم إجراء قياس.
!>          = 'Y':  تم إجراء القياس، أي تم استبدال A بـ
!>                  diag(S) * A * diag(S).
!>

المعلمات الداخلية:

!>  THRESH هو قيمة حد فاصل تستخدم لتحديد ما إذا كان ينبغي إجراء القياس
!>  بناءً على نسبة عوامل القياس.  إذا كان SCOND < THRESH،
!>  يتم إجراء القياس.
!>
!>  LARGE و SMALL هي قيم حدود فاصلة تستخدم لتحديد ما إذا كان ينبغي إجراء القياس
!>  بناءً على الحجم المطلق لأكبر عنصر في المصفوفة.
!>  إذا كان AMAX > LARGE أو AMAX < SMALL، يتم إجراء القياس.
!>
