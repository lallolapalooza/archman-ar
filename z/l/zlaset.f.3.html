subroutine zlaset (character uplo, integer m, integer n, complex*16 alpha, complex*16 beta, complex*16, dimension( lda, * ) a, integer lda)

ZLASET تقوم بتهيئة العناصر خارج القطر والعناصر القطرية للمصفوفة إلى قيم معينة.

الغرض:

!>
!> ZLASET تقوم بتهيئة مصفوفة ثنائية الأبعاد A إلى BETA على القطر
!> و ALPHA على العناصر خارج القطر.
!>

المعلمات

UPLO 
!>          UPLO هو CHARACTER*1
!>          يحدد جزء المصفوفة A الذي سيتم تعيينه.
!>          = 'U':      يتم تعيين الجزء العلوي المثلثي. الجزء السفلي
!>                      يبقى دون تغيير.
!>          = 'L':      يتم تعيين الجزء السفلي المثلثي. الجزء العلوي
!>                      يبقى دون تغيير.
!>          غير ذلك:    يتم تعيين المصفوفة A بأكملها.
!>

M

!>          M هو INTEGER
!>          عند الإدخال، M يحدد عدد صفوف A.
!>

N

!>          N هو INTEGER
!>          عند الإدخال، N يحدد عدد أعمدة A.
!>

ALPHA

!>          ALPHA هو COMPLEX*16
!>          يتم تعيين جميع عناصر المصفوفة خارج القطر إلى ALPHA.
!>

BETA

!>          BETA هو COMPLEX*16
!>          يتم تعيين جميع عناصر المصفوفة القطرية إلى BETA.
!>

A

!>          A هي مصفوفة COMPLEX*16، ذات البعد (LDA,N)
!>          عند الإدخال، المصفوفة A بحجم m × n.
!>          عند الخروج، A(i,j) = ALPHA, 1 <= i <= m, 1 <= j <= n, i.ne.j;
!>                   A(i,i) = BETA , 1 <= i <= min(m,n)
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرائد للمصفوفة A.  LDA >= max(1,M).
!>
