subroutine zlaqr0 (logical wantt, logical wantz, integer n, integer ilo, integer ihi, complex*16, dimension( ldh, * ) h, integer ldh, complex*16, dimension( * ) w, integer iloz, integer ihiz, complex*16, dimension( ldz, * ) z, integer ldz, complex*16, dimension( * ) work, integer lwork, integer info)

ZLAQR0 يحسب القيم الذاتية لمصفوفة هسنبرغ، وبشكل اختياري، المصفوفات من تحليل شور.

الغرض:

!>
!>    ZLAQR0 يحسب القيم الذاتية لمصفوفة هسنبرغ H
!>    وبشكل اختياري، المصفوفات T و Z من تحليل شور
!>    H = Z T Z**H، حيث T مصفوفة مثلثية علوية (شكل شور)،
!>    و Z هي المصفوفة الوحدوية لمتجهات شور.
!>
!>    بشكل اختياري، يمكن ضرب Z في مصفوفة وحدوية مدخلة Q بحيث يمكن لهذا الإجراء الفرعي
!>    إعطاء تحليل شور لمصفوفة A تم اختزالها إلى شكل هسنبرغ H
!>    بواسطة المصفوفة الوحدوية Q:  A = Q*H*Q**H = (QZ)*H*(QZ)**H.
!>

المعلمات

WANTT 
!>          WANTT هو LOGICAL (منطقي)
!>          = .TRUE. : مطلوب شكل شور الكامل T؛
!>          = .FALSE.: مطلوبة القيم الذاتية فقط.
!>

WANTZ

!>          WANTZ هو LOGICAL (منطقي)
!>          = .TRUE. : مطلوب مصفوفة متجهات شور Z؛
!>          = .FALSE.: لا يتم طلب متجهات شور.
!>

N

!>          N هو INTEGER (عدد صحيح)
!>           رتبة المصفوفة H.  N >= 0.
!>

ILO

!>          ILO هو INTEGER (عدد صحيح)
!>

IHI

!>          IHI هو INTEGER (عدد صحيح)
!>
!>           من المفترض أن H بالفعل مثلثية علوية في الصفوف
!>           والأعمدة 1:ILO-1 و IHI+1:N، وإذا كان ILO > 1،
!>           فإن H(ILO,ILO-1) يساوي صفرًا. عادةً ما يتم تعيين ILO و IHI بواسطة
!>           استدعاء سابق لـ ZGEBAL، ثم تمريرهما إلى ZGEHRD عندما
!>           يتم اختزال مصفوفة الإخراج من ZGEBAL إلى شكل هسنبرغ.
!>           بخلاف ذلك، يجب تعيين ILO و IHI إلى 1 و N،
!>           على التوالي.  إذا كان N > 0، فإن 1 <= ILO <= IHI <= N.
!>           إذا كان N = 0، فإن ILO = 1 و IHI = 0.
!>

H

!>          H هي مصفوفة COMPLEX*16، البعد (LDH,N)
!>           عند الإدخال، مصفوفة هسنبرغ العلوية H.
!>           عند الخروج، إذا كان INFO = 0 و WANTT هو .TRUE.، فإن H
!>           تحتوي على المصفوفة المثلثية العلوية T من تحليل شور
!>           (شكل شور). إذا كان INFO = 0 و WANT هو
!>           .FALSE.، فإن محتويات H غير محددة عند الخروج.
!>           (قيمة الإخراج لـ H عندما يكون INFO > 0 معطاة تحت
!>           وصف INFO أدناه.)
!>
!>           قد يقوم هذا الإجراء الفرعي بتعيين H(i,j) = 0 بشكل صريح لـ i > j و
!>           j = 1، 2، ... ILO-1 أو j = IHI+1، IHI+2، ... N.
!>

LDH

!>          LDH هو INTEGER (عدد صحيح)
!>           البعد الرائد للمصفوفة H. LDH >= max(1,N).
!>

W

!>          W هي مصفوفة COMPLEX*16، البعد (N)
!>           القيم الذاتية المحسوبة لـ H(ILO:IHI,ILO:IHI) مخزنة
!>           في W(ILO:IHI). إذا كان WANTT هو .TRUE.، فإن القيم الذاتية
!>           مخزنة بنفس الترتيب الموجود على القطر الرئيسي لشكل شور
!>           الذي تم إرجاعه في H، مع W(i) = H(i,i).
!>

ILOZ

!>          ILOZ هو INTEGER (عدد صحيح)
!>

IHIZ

!>          IHIZ هو INTEGER (عدد صحيح)
!>           تحديد صفوف Z التي يجب تطبيق التحويلات عليها
!>           إذا كان WANTZ هو .TRUE..
!>           1 <= ILOZ <= ILO؛ IHI <= IHIZ <= N.
!>

Z

!>          Z هي مصفوفة COMPLEX*16، البعد (LDZ,IHI)
!>           إذا كان WANTZ هو .FALSE.، فلا يتم الرجوع إلى Z.
!>           إذا كان WANTZ هو .TRUE.، فسيتم استبدال Z(ILO:IHI,ILOZ:IHIZ)
!>           بـ Z(ILO:IHI,ILOZ:IHIZ)*U حيث U هي
!>           عامل شور الوحدوي لـ H(ILO:IHI,ILO:IHI).
!>           (قيمة الإخراج لـ Z عندما يكون INFO > 0 معطاة تحت
!>           وصف INFO أدناه.)
!>

LDZ

!>          LDZ هو INTEGER (عدد صحيح)
!>           البعد الرائد للمصفوفة Z.  إذا كان WANTZ هو .TRUE.
!>           فإن LDZ >= MAX(1,IHIZ).  بخلاف ذلك، LDZ >= 1.
!>

WORK

!>          WORK هي مصفوفة COMPLEX*16، البعد LWORK
!>           عند الخروج، إذا كان LWORK = -1، فإن WORK(1) ترجع تقديرًا لـ
!>           القيمة المثلى لـ LWORK.
!>

LWORK

!>          LWORK هو INTEGER (عدد صحيح)
!>           بعد مصفوفة WORK.  LWORK >= max(1,N)
!>           كافٍ، ولكن LWORK عادةً ما يكون كبيرًا مثل 6*N قد
!>           يكون مطلوبًا للحصول على أداء مثالي.  يُوصى باستعلام مساحة العمل
!>           لتحديد حجم مساحة العمل المثلى.
!>
!>           إذا كان LWORK = -1، فسيقوم ZLAQR0 بإجراء استعلام مساحة العمل.
!>           في هذه الحالة، يتحقق ZLAQR0 من معلمات الإدخال
!>           ويقدر حجم مساحة العمل الأمثل للقيم
!>           المعطاة لـ N و ILO و IHI.  يتم إرجاع التقدير
!>           في WORK(1).  لا يتم إصدار أي رسالة خطأ متعلقة بـ LWORK
!>           بواسطة XERBLA.  لا يتم الوصول إلى H ولا Z.
!>

INFO

!>          INFO هو INTEGER (عدد صحيح)
!>             = 0:  انتهاء ناجح
!>             > 0:  إذا كان INFO = i، فقد فشل ZLAQR0 في حساب جميع
!>                القيم الذاتية.  العناصر 1:ilo-1 و i+1:n من WR
!>                و WI تحتوي على تلك القيم الذاتية التي تم
!>                حسابها بنجاح.  (حالات الفشل نادرة.)
!>
!>                إذا كان INFO > 0 و WANT هو .FALSE.، فعند الخروج،
!>                القيم الذاتية المتبقية غير المتقاربة هي القيم
!>                الذاتية لمصفوفة هسنبرغ العلوية للصفوف
!>                والأعمدة ILO إلى INFO من القيمة النهائية،
!>                المخرجة لـ H.
!>
!>                إذا كان INFO > 0 و WANTT هو .TRUE.، فعند الخروج
!>
!>           (*)  (القيمة الأولية لـ H)*U  = U*(القيمة النهائية لـ H)
!>
!>                حيث U هي مصفوفة وحدوية.  القيمة النهائية لـ
!>                H هي هسنبرغ علوية ومثلثية في
!>                الصفوف والأعمدة INFO+1 إلى IHI.
!>
!>                إذا كان INFO > 0 و WANTZ هو .TRUE.، فعند الخروج
!>
!>                  (القيمة النهائية لـ Z(ILO:IHI,ILOZ:IHIZ)
!>                   =  (القيمة الأولية لـ Z(ILO:IHI,ILOZ:IHIZ)*U
!>
!>                حيث U هي المصفوفة الوحدوية في (*) (بغض النظر
!>                عن قيمة WANTT.)
!>
!>                إذا كان INFO > 0 و WANTZ هو .FALSE.، فلا يتم
!>                الوصول إلى Z.
!>
