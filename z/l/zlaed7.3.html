subroutine zlaed7 (integer n, integer cutpnt, integer qsiz, integer tlvls, integer curlvl, integer curpbm, double precision, dimension( * ) d, complex*16, dimension( ldq, * ) q, integer ldq, double precision rho, integer, dimension( * ) indxq, double precision, dimension( * ) qstore, integer, dimension( * ) qptr, integer, dimension( * ) prmptr, integer, dimension( * ) perm, integer, dimension( * ) givptr, integer, dimension( 2, * ) givcol, double precision, dimension( 2, * ) givnum, complex*16, dimension( * ) work, double precision, dimension( * ) rwork, integer, dimension( * ) iwork, integer info)

ZLAED7 تستخدم بواسطة ZSTEDC. تحسب النظام الفرعي المحدث لمصفوفة قطرية بعد التعديل بواسطة مصفوفة متناظرة ذات رتبة واحدة. تستخدم عندما تكون المصفوفة الأصلية كثيفة.

الغرض:

!>
!> ZLAED7 تحسب النظام الفرعي المحدث لمصفوفة قطرية بعد التعديل بواسطة مصفوفة متناظرة ذات رتبة واحدة. يستخدم هذا
!> الروتين الفرعي فقط لمشكلة القيمة الذاتية التي تتطلب جميع القيم الذاتية وربما المتجهات الذاتية لمصفوفة متناظرة
!> كثيفة أو نطاقية تم اختزالها إلى شكل ثلاثي القطر.
!>
!>   T = Q(in) ( D(in) + RHO * Z*Z**H ) Q**H(in) = Q(out) * D(out) * Q**H(out)
!>
!>   حيث Z = Q**Hu، u هو متجه بطول N بعناصر واحد في المواضع CUTPNT و CUTPNT + 1 وأصفار في المواضع الأخرى.
!>
!>    يتم تخزين المتجهات الذاتية للمصفوفة الأصلية في Q، والقيم الذاتية في D. تتكون الخوارزمية من ثلاث مراحل:
!>
!>       تتكون المرحلة الأولى من تخفيض حجم المشكلة
!>       عند وجود قيم ذاتية متعددة أو إذا كان هناك صفر في
!>       المتجه Z. لكل حدوث من هذا القبيل، يتم تقليل بعد مشكلة المعادلة العلمانية
!>       بمقدار واحد. يتم تنفيذ هذه المرحلة بواسطة الروتين الفرعي DLAED2.
!>
!>       تتكون المرحلة الثانية من حساب القيم الذاتية المحدثة. يتم ذلك عن طريق إيجاد جذور
!>       المعادلة العلمانية عبر الروتين الفرعي DLAED4 (كما تم استدعاؤه بواسطة SLAED3).
!>       يحسب هذا الروتين الفرعي أيضًا المتجهات الذاتية للمشكلة الحالية.
!>
!>       تتكون المرحلة النهائية من حساب المتجهات الذاتية المحدثة
!>       مباشرة باستخدام القيم الذاتية المحدثة. يتم ضرب المتجهات الذاتية للمشكلة
!>       الحالية مع المتجهات الذاتية للمشكلة الشاملة.
!>

المعلمات

N 
!>          N هو عدد صحيح (INTEGER)
!>         بعد المصفوفة المتناظرة ثلاثية القطر. N >= 0.
!>

CUTPNT

!>          CUTPNT هو عدد صحيح (INTEGER)
!>         يحتوي على موقع آخر قيمة ذاتية في المصفوفة الفرعية الرائدة. min(1,N) <= CUTPNT <= N.
!>

QSIZ

!>          QSIZ هو عدد صحيح (INTEGER)
!>         بعد المصفوفة الوحدوية المستخدمة لاختزال
!>         المصفوفة الكاملة إلى شكل ثلاثي القطر. QSIZ >= N.
!>

TLVLS

!>          TLVLS هو عدد صحيح (INTEGER)
!>         العدد الإجمالي لمستويات الدمج في شجرة التقسيم والغزو الشاملة.
!>

CURLVL

!>          CURLVL هو عدد صحيح (INTEGER)
!>         المستوى الحالي في روتين الدمج الشامل،
!>         0 <= curlvl <= tlvls.
!>

CURPBM

!>          CURPBM هو عدد صحيح (INTEGER)
!>         المشكلة الحالية في المستوى الحالي في الروتين الشامل
!>         للدمج (العد من أعلى اليسار إلى أسفل اليمين).
!>

D

!>          D هو مصفوفة مزدوجة الدقة (DOUBLE PRECISION)، البعد (N)
!>         عند الدخول، القيم الذاتية للمصفوفة المعدلة بالرتبة 1.
!>         عند الخروج، القيم الذاتية للمصفوفة المصلحة.
!>

Q

!>          Q هو مصفوفة مركبة مزدوجة الدقة (COMPLEX*16)، البعد (LDQ,N)
!>         عند الدخول، المتجهات الذاتية للمصفوفة المعدلة بالرتبة 1.
!>         عند الخروج، المتجهات الذاتية للمصفوفة ثلاثية القطر المصلحة.
!>

LDQ

!>          LDQ هو عدد صحيح (INTEGER)
!>         البعد الرائد للمصفوفة Q. LDQ >= max(1,N).
!>

RHO

!>          RHO هي دقة مزدوجة (DOUBLE PRECISION)
!>         تحتوي على العنصر دون القطري المستخدم لإنشاء التعديل ذو الرتبة 1.
!>

INDXQ

!>          INDXQ هي مصفوفة عدد صحيح (INTEGER)، البعد (N)
!>         تحتوي هذه على التبديل الذي سيعيد دمج المشكلة الفرعية التي تم حلها للتو مرة أخرى بترتيب مرتب،
!>         أي أن D( INDXQ( I = 1, N ) ) سيكون بترتيب تصاعدي.
!>

IWORK

!>          IWORK هي مصفوفة عدد صحيح (INTEGER)، البعد (4*N)
!>

RWORK

!>          RWORK هي مصفوفة مزدوجة الدقة (DOUBLE PRECISION)،
!>                                 البعد (3*N+2*QSIZ*N)
!>

WORK

!>          WORK هي مصفوفة مركبة مزدوجة الدقة (COMPLEX*16)، البعد (QSIZ*N)
!>

QSTORE

!>          QSTORE هي مصفوفة مزدوجة الدقة (DOUBLE PRECISION)، البعد (N**2+1)
!>         تخزن المتجهات الذاتية للمصفوفات الفرعية التي تمت مواجهتها أثناء
!>         التقسيم والغزو، مجمعة معًا. يشير QPTR إلى
!>         بداية المصفوفات الفرعية.
!>

QPTR

!>          QPTR هي مصفوفة عدد صحيح (INTEGER)، البعد (N+2)
!>         قائمة بالمؤشرات التي تشير إلى بداية المصفوفات الفرعية المخزنة
!>         في QSTORE. يتم ترقيم المصفوفات الفرعية بدءًا من
!>         أسفل اليسار في شجرة التقسيم والغزو، من اليسار إلى
!>         اليمين ومن الأسفل إلى الأعلى.
!>

PRMPTR

!>          PRMPTR هي مصفوفة عدد صحيح (INTEGER)، البعد (N lg N)
!>         تحتوي على قائمة بالمؤشرات التي تشير إلى مكان تخزين تبديل
!>         مستوى معين في PERM. يشير PRMPTR(i+1) - PRMPTR(i)
!>         إلى حجم التبديل وكذلك حجم
!>         المشكلة الكاملة غير المخفضة.
!>

PERM

!>          PERM هي مصفوفة عدد صحيح (INTEGER)، البعد (N lg N)
!>         تحتوي على التباديل (من التخفيض والفرز) التي سيتم
!>         تطبيقها على كل كتلة ذاتية.
!>

GIVPTR

!>          GIVPTR هي مصفوفة عدد صحيح (INTEGER)، البعد (N lg N)
!>         تحتوي على قائمة بالمؤشرات التي تشير إلى مكان تخزين دوران
!>         Givens لمستوى معين في GIVCOL. يشير GIVPTR(i+1) - GIVPTR(i)
!>         إلى عدد دورانات Givens.
!>

GIVCOL

!>          GIVCOL هي مصفوفة عدد صحيح (INTEGER)، البعد (2, N lg N)
!>         يشير كل زوج من الأرقام إلى زوج من الأعمدة التي سيتم إجراء
!>         دوران Givens عليها.
!>

GIVNUM

!>          GIVNUM هي مصفوفة مزدوجة الدقة (DOUBLE PRECISION)، البعد (2, N lg N)
!>         يشير كل رقم إلى قيمة S التي سيتم استخدامها في
!>         دوران Givens المقابل.
!>

INFO

!>          INFO هو عدد صحيح (INTEGER)
!>          = 0:  انتهاء ناجح.
!>          < 0:  إذا كان INFO = -i، فإن المعلمة i كانت ذات قيمة غير قانونية.
!>          > 0:  إذا كان INFO = 1، فإن القيمة الذاتية لم تتقارب
!>
