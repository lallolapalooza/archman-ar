توثيق الوظيفة/الإجراء الفرعي

subroutine zlagge (integer m, integer n, integer kl, integer ku, double precision, dimension( * ) d, complex*16, dimension( lda, * ) a, integer lda, integer, dimension( 4 ) iseed, complex*16, dimension( * ) work, integer info)

ZLAGGE

الغرض:

!>
!> ZLAGGE يقوم بإنشاء مصفوفة معقدة عامة M × N تسمى A، عن طريق الضرب المسبق واللاحق
!> لمصفوفة قطرية حقيقية D بمصفوفات وحدوية عشوائية:
!> A = U*D*V. قد يتم تقليل النطاقات السفلية والعلوية بعد ذلك إلى
!> kl و ku عن طريق تحويلات وحدوية إضافية.
!>

المعاملات

M 
!>          M هو عدد صحيح (INTEGER)
!>          عدد صفوف المصفوفة A.  M >= 0.
!>

N

!>          N هو عدد صحيح (INTEGER)
!>          عدد أعمدة المصفوفة A.  N >= 0.
!>

KL

!>          KL هو عدد صحيح (INTEGER)
!>          عدد الأقطار الفرعية غير الصفرية داخل نطاق A.
!>          0 <= KL <= M-1.
!>

KU

!>          KU هو عدد صحيح (INTEGER)
!>          عدد الأقطار فوق الفرعية غير الصفرية داخل نطاق A.
!>          0 <= KU <= N-1.
!>

D

!>          D هو مصفوفة دقة مزدوجة (DOUBLE PRECISION)، البعد (min(M,N))
!>          عناصر القطر للمصفوفة القطرية D.
!>

A

!>          A هي مصفوفة معقدة (COMPLEX*16)، البعد (LDA,N)
!>          المصفوفة M × N التي تم إنشاؤها A.
!>

LDA

!>          LDA هو عدد صحيح (INTEGER)
!>          البعد الرئيسي للمصفوفة A.  LDA >= M.
!>

ISEED

!>          ISEED هي مصفوفة أعداد صحيحة (INTEGER)، البعد (4)
!>          عند الإدخال، البذرة لمولد الأرقام العشوائية؛ يجب أن تكون عناصر
!>          المصفوفة بين 0 و 4095، ويجب أن يكون ISEED(4) فرديًا.
!>          عند الخروج، يتم تحديث البذرة.
!>

WORK

!>          WORK هي مصفوفة معقدة (COMPLEX*16)، البعد (M+N)
!>

INFO

!>          INFO هو عدد صحيح (INTEGER)
!>          = 0: خروج ناجح
!>          < 0: إذا كان INFO = -i، فإن المعامل i كان له قيمة غير قانونية
!>
