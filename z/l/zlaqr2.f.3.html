subroutine zlaqr2 (logical wantt, logical wantz, integer n, integer ktop, integer kbot, integer nw, complex*16, dimension( ldh, * ) h, integer ldh, integer iloz, integer ihiz, complex*16, dimension( ldz, * ) z, integer ldz, integer ns, integer nd, complex*16, dimension( * ) sh, complex*16, dimension( ldv, * ) v, integer ldv, integer nh, complex*16, dimension( ldt, * ) t, integer ldt, integer nv, complex*16, dimension( ldwv, * ) wv, integer ldwv, complex*16, dimension( * ) work, integer lwork)

ZLAQR2 تقوم بالتحويل المتشابه الوحدوي لمصفوفة هسنبرغ للكشف عن القيم الذاتية المتقاربة بالكامل من مصفوفة فرعية رئيسية لاحقة (انكماش مبكر طموح).

الغرض:

!>
!>    ZLAQR2 مطابق لـ ZLAQR3 باستثناء أنه يتجنب
!>    الاستدعاء التكراري عن طريق استدعاء ZLAHQR بدلاً من ZLAQR4.
!>
!>    انكماش مبكر طموح:
!>
!>    تقبل ZLAQR2 مصفوفة هسنبرغ علوية
!>    H وتقوم بتحويل متشابه وحدوي
!>    مصمم للكشف عن القيم الذاتية المتقاربة بالكامل وانكماشها من
!>    مصفوفة فرعية رئيسية لاحقة. عند الخرج، تم استبدال H بـ
!>    مصفوفة هسنبرغ جديدة هي اضطراب لـ
!>    تحويل متشابه وحدوي لـ H. ومن المأمول أن تكون النسخة النهائية من H تحتوي على العديد من
!>    الإدخالات الفرعية الصفرية.
!>
!>

المعلمات

WANTT 
!>          WANTT هو LOGICAL
!>          إذا كانت .TRUE.، فسيتم تحديث مصفوفة هسنبرغ H بالكامل
!>          بحيث يمكن حساب عامل Schur المثلثي
!>          (بالتعاون مع الإجراء الفرعي المستدعي).
!>          إذا كانت .FALSE.، فلن يتم تحديث سوى ما يكفي من H للحفاظ على
!>          القيم الذاتية.
!>

WANTZ

!>          WANTZ هو LOGICAL
!>          إذا كانت .TRUE.، فسيتم تحديث المصفوفة الوحدوية Z
!>          بحيث يمكن حساب عامل Schur الوحدوي
!>          (بالتعاون مع الإجراء الفرعي المستدعي).
!>          إذا كانت .FALSE.، فلن يتم استخدام Z.
!>

N

!>          N هو INTEGER
!>          رتبة المصفوفة H (وإذا كانت WANTZ هي .TRUE.)
!>          رتبة المصفوفة الوحدوية Z.
!>

KTOP

!>          KTOP هو INTEGER
!>          من المفترض أن KTOP = 1 أو H(KTOP,KTOP-1)=0.
!>          تحدد KBOT و KTOP معًا كتلة معزولة
!>          على طول قطر مصفوفة هسنبرغ.
!>

KBOT

!>          KBOT هو INTEGER
!>          من المفترض دون فحص أن KBOT = N أو H(KBOT+1,KBOT)=0.
!>          تحدد KBOT و KTOP معًا كتلة معزولة
!>          على طول قطر مصفوفة هسنبرغ.
!>

NW

!>          NW هو INTEGER
!>          حجم نافذة الانكماش. 1 <= NW <= (KBOT-KTOP+1).
!>

H

!>          H هو مصفوفة COMPLEX*16، ذات بعد (LDH,N)
!>          عند الإدخال، يخزن القسم الأولي N×N من H مصفوفة
!>          هسنبرغ التي تخضع للانكماش المبكر الطموح.
!>          عند الخرج، تم تحويل H بواسطة تحويل متشابه وحدوي،
!>          تم اضطرابه، وأعيدت إلى شكل هسنبرغ الذي (نأمل) يحتوي على بعض
!>          الإدخالات الفرعية الصفرية.
!>

LDH

!>          LDH هو INTEGER
!>          البعد الرائد لـ H تمامًا كما تم الإعلان عنه في الإجراء الفرعي المستدعي.
!>          N <= LDH
!>

ILOZ

!>          ILOZ هو INTEGER
!>

IHIZ

!>          IHIZ هو INTEGER
!>          تحديد صفوف Z التي يجب تطبيق التحويلات عليها إذا كانت WANTZ هي .TRUE..
!>          1 <= ILOZ <= IHIZ <= N.
!>

Z

!>          Z هو مصفوفة COMPLEX*16، ذات بعد (LDZ,N)
!>          إذا كانت WANTZ هي .TRUE.، فعند الخرج، تم تجميع التحويل المتشابه الوحدوي المذكور أعلاه
!>          في Z(ILOZ:IHIZ,ILOZ:IHIZ) من اليمين.
!>          إذا كانت WANTZ هي .FALSE.، فلن يتم استخدام Z.
!>

LDZ

!>          LDZ هو INTEGER
!>          البعد الرائد لـ Z تمامًا كما تم الإعلان عنه في الإجراء الفرعي المستدعي.
!>          1 <= LDZ.
!>

NS

!>          NS هو INTEGER
!>          عدد القيم الذاتية غير المتقاربة (أي التقريبية)
!>          التي تم إرجاعها في SR و SI والتي يمكن استخدامها كتحولات بواسطة
!>          الإجراء الفرعي المستدعي.
!>

ND

!>          ND هو INTEGER
!>          عدد القيم الذاتية المتقاربة التي تم الكشف عنها بواسطة هذا
!>          الإجراء الفرعي.
!>

SH

!>          SH هو مصفوفة COMPLEX*16، ذات بعد (KBOT)
!>          عند الخرج، القيم الذاتية التقريبية التي قد
!>          تُستخدم كتحولات مخزنة في SH(KBOT-ND-NS+1)
!>          حتى SR(KBOT-ND). القيم الذاتية المتقاربة
!>          مخزنة في SH(KBOT-ND+1) حتى SH(KBOT).
!>

V

!>          V هي مصفوفة COMPLEX*16، ذات بعد (LDV,NW)
!>          مصفوفة عمل بحجم NW×NW.
!>

LDV

!>          LDV هو INTEGER
!>          البعد الرائد لـ V تمامًا كما تم الإعلان عنه في الإجراء الفرعي المستدعي.
!>          NW <= LDV
!>

NH

!>          NH هو INTEGER
!>          عدد أعمدة T. NH >= NW.
!>

T

!>          T هي مصفوفة COMPLEX*16، ذات بعد (LDT,NW)
!>

LDT

!>          LDT هو INTEGER
!>          البعد الرائد لـ T تمامًا كما تم الإعلان عنه في الإجراء الفرعي المستدعي.
!>          NW <= LDT
!>

NV

!>          NV هو INTEGER
!>          عدد صفوف مصفوفة العمل WV المتاحة لـ
!>          مساحة العمل. NV >= NW.
!>

WV

!>          WV هي مصفوفة COMPLEX*16، ذات بعد (LDWV,NW)
!>

LDWV

!>          LDWV هو INTEGER
!>          البعد الرائد لـ W تمامًا كما تم الإعلان عنه في الإجراء الفرعي المستدعي.
!>          NW <= LDV
!>

WORK

!>          WORK هي مصفوفة COMPLEX*16، ذات بعد (LWORK)
!>          عند الخروج، يتم تعيين WORK(1) إلى تقدير للقيمة المثلى
!>          لـ LWORK للقيم المعطاة لـ N، NW، KTOP، و KBOT.
!>

LWORK

!>          LWORK هو INTEGER
!>          بعد مصفوفة العمل WORK. LWORK = 2*NW
!>          كافٍ، ولكن قد ينتج عن قيم أكبر لـ LWORK كفاءة أكبر.
!>
!>          إذا كانت LWORK = -1، فسيتم افتراض استعلام مساحة العمل؛ ZLAQR2
!>          يقدر فقط حجم مساحة العمل الأمثل للقيم المعطاة
!>          لـ N، NW، KTOP، و KBOT. يتم إرجاع التقدير
!>          في WORK(1). لا يتم إصدار رسالة خطأ متعلقة بـ LWORK
!>          بواسطة XERBLA. لا يتم الوصول إلى H أو Z.
!>
