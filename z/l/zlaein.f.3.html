subroutine zlaein (logical rightv, logical noinit, integer n, complex*16, dimension( ldh, * ) h, integer ldh, complex*16 w, complex*16, dimension( * ) v, complex*16, dimension( ldb, * ) b, integer ldb, double precision, dimension( * ) rwork, double precision eps3, double precision smlnum, integer info)

ZLAEIN تحسب متجهًا ذاتيًا أيمن أو أيسر محدد لمصفوفة هسنبرج علوية عن طريق التكرار العكسي.

الغرض:

!>
!> ZLAEIN تستخدم التكرار العكسي للعثور على متجه ذاتي أيمن أو أيسر
!> مطابق للقيمة الذاتية W لمصفوفة هسنبرج العلوية H.
!>

المعلمات

RIGHTV 
!>          RIGHTV هو LOGICAL (منطقي)
!>          = .TRUE. : حساب المتجه الذاتي الأيمن؛
!>          = .FALSE.: حساب المتجه الذاتي الأيسر.
!>

NOINIT

!>          NOINIT هو LOGICAL (منطقي)
!>          = .TRUE. : لم يتم توفير متجه أولي في V
!>          = .FALSE.: تم توفير متجه أولي في V.
!>

N

!>          N هو INTEGER (عدد صحيح)
!>          رتبة المصفوفة H. N >= 0.
!>

H

!>          H هو مصفوفة COMPLEX*16 (عدد مركب)، البعد (LDH,N)
!>          مصفوفة هسنبرج العلوية H.
!>

LDH

!>          LDH هو INTEGER (عدد صحيح)
!>          البعد الرئيسي للمصفوفة H. LDH >= max(1,N).
!>

W

!>          W هو COMPLEX*16 (عدد مركب)
!>          القيمة الذاتية لـ H التي يجب حساب متجهها الذاتي الأيمن أو الأيسر.
!>

V

!>          V هي مصفوفة COMPLEX*16 (عدد مركب)، البعد (N)
!>          عند الإدخال، إذا كان NOINIT = .FALSE.، يجب أن تحتوي V على متجه بدء
!>          للتكرار العكسي؛ وإلا فإن V لا تحتاج إلى تعيين.
!>          عند الخروج، تحتوي V على المتجه الذاتي المحسوب، يتم تطبيعه بحيث
!>          يكون للمكون ذي القيمة المطلقة الأكبر قيمة مطلقة تساوي 1؛ هنا
!>          تؤخذ القيمة المطلقة لعدد مركب (x,y) على أنها
!>          |x| + |y|.
!>

B

!>          B هي مصفوفة COMPLEX*16 (عدد مركب)، البعد (LDB,N)
!>

LDB

!>          LDB هو INTEGER (عدد صحيح)
!>          البعد الرئيسي للمصفوفة B. LDB >= max(1,N).
!>

RWORK

!>          RWORK هي مصفوفة DOUBLE PRECISION (عدد عشري مضاعف الدقة)، البعد (N)
!>

EPS3

!>          EPS3 هو DOUBLE PRECISION (عدد عشري مضاعف الدقة)
!>          قيمة صغيرة تعتمد على الآلة تستخدم لتنقيح القيم الذاتية القريبة،
!>          ولاستبدال المحاور الصفرية.
!>

SMLNUM

!>          SMLNUM هو DOUBLE PRECISION (عدد عشري مضاعف الدقة)
!>          قيمة تعتمد على الآلة قريبة من عتبة النقصان.
!>

INFO

!>          INFO هو INTEGER (عدد صحيح)
!>          = 0:  الخروج بنجاح
!>          = 1:  لم يتقارب التكرار العكسي؛ يتم تعيين V إلى التكرار الأخير.
!>
