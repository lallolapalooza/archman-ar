subroutine zlarhs (character*3 path, character xtype, character uplo, character trans, integer m, integer n, integer kl, integer ku, integer nrhs, complex*16, dimension( lda, * ) a, integer lda, complex*16, dimension( ldx, * ) x, integer ldx, complex*16, dimension( ldb, * ) b, integer ldb, integer, dimension( 4 ) iseed, integer info)

ZLARHS

الغرض:

!>
!> ZLARHS تختار مجموعة من متجهات الحل العشوائي NRHS وتقوم بإعداد
!> الجانب الأيمن للنظام الخطي
!>    op(A) * X = B,
!> حيث op(A) = A, A**T, أو A**H، اعتمادًا على TRANS.
!>

المعلمات

PATH 
!>          PATH هي CHARACTER*3
!>          نوع المصفوفة المعقدة A. يمكن إعطاء PATH بأي
!>          مزيج من الأحرف الكبيرة والصغيرة. المسارات الصالحة تشمل
!>             xGE:  مصفوفة عامة m x n
!>             xGB:  مصفوفة نطاق عامة
!>             xPO:  مصفوفة هرميتية موجبة محددة، تخزين ثنائي الأبعاد
!>             xPP:  مصفوفة هرميتية موجبة محددة مضغوطة
!>             xPB:  مصفوفة هرميتية موجبة محددة نطاق
!>             xHE:  مصفوفة هرميتية غير محددة، تخزين ثنائي الأبعاد
!>             xHP:  مصفوفة هرميتية غير محددة مضغوطة
!>             xHB:  مصفوفة هرميتية غير محددة نطاق
!>             xSY:  مصفوفة متناظرة غير محددة، تخزين ثنائي الأبعاد
!>             xSP:  مصفوفة متناظرة غير محددة مضغوطة
!>             xSB:  مصفوفة متناظرة غير محددة نطاق
!>             xTR:  مصفوفة مثلثة
!>             xTP:  مصفوفة مثلثة مضغوطة
!>             xTB:  مصفوفة مثلثة نطاق
!>             xQR:  مصفوفة عامة m x n
!>             xLQ:  مصفوفة عامة m x n
!>             xQL:  مصفوفة عامة m x n
!>             xRQ:  مصفوفة عامة m x n
!>          حيث يشير الحرف الأول إلى الدقة.
!>

XTYPE

!>          XTYPE هي CHARACTER*1
!>          تحدد كيفية تحديد الحل الدقيق X:
!>          = 'N':  حل جديد؛ إنشاء X عشوائي.
!>          = 'C':  محسوب؛ استخدام قيمة X عند الإدخال.
!>

UPLO

!>          UPLO هي CHARACTER*1
!>          تستخدم فقط إذا كانت A متناظرة أو مثلثة؛ تحدد ما إذا كان
!>          الجزء العلوي أو السفلي المثلث من المصفوفة A مخزنًا.
!>          = 'U':  مثلث علوي
!>          = 'L':  مثلث سفلي
!>

TRANS

!>          TRANS هي CHARACTER*1
!>          تستخدم فقط إذا كانت A غير متناظرة؛ تحدد العملية
!>          المطبقة على المصفوفة A.
!>          = 'N':  B := A    * X  (بدون تبديل)
!>          = 'T':  B := A**T * X  (تبديل)
!>          = 'C':  B := A**H * X  (تبديل مترافق)
!>

M

!>          M هي INTEGER
!>          عدد صفوف المصفوفة A.  M >= 0.
!>

N

!>          N هي INTEGER
!>          عدد أعمدة المصفوفة A.  N >= 0.
!>

KL

!>          KL هي INTEGER
!>          تستخدم فقط إذا كانت A مصفوفة نطاق؛ تحدد عدد
!>          الأقطار الفرعية لـ A إذا كانت A مصفوفة نطاق عامة أو إذا كانت A
!>          متناظرة أو مثلثة و UPLO = 'L'؛ تحدد عدد
!>          الأقطار الفرعية لـ A إذا كانت A متناظرة أو مثلثة و
!>          UPLO = 'U'.  0 <= KL <= M-1.
!>

KU

!>          KU هي INTEGER
!>          تستخدم فقط إذا كانت A مصفوفة نطاق عامة أو إذا كانت A
!>          مثلثة.
!>
!>          إذا كان PATH = xGB، تحدد عدد الأقطار الفائقة لـ A،
!>          و 0 <= KU <= N-1.
!>
!>          إذا كان PATH = xTR، xTP، أو xTB، تحدد ما إذا كانت
!>          المصفوفة لها قطر وحدات أم لا:
!>          = 1:  المصفوفة لها قطر غير وحدات (افتراضي)
!>          = 2:  المصفوفة لها قطر وحدات
!>

NRHS

!>          NRHS هي INTEGER
!>          عدد متجهات الجانب الأيمن في النظام A*X = B.
!>

A

!>          A هي مصفوفة COMPLEX*16، البعد (LDA,N)
!>          مصفوفة الاختبار التي نوعها يعطى بواسطة PATH.
!>

LDA

!>          LDA هي INTEGER
!>          البعد الرائد للمصفوفة A.
!>          إذا كان PATH = xGB، LDA >= KL+KU+1.
!>          إذا كان PATH = xPB، xSB، xHB، أو xTB، LDA >= KL+1.
!>          بخلاف ذلك، LDA >= max(1,M).
!>

X

!>          X هي (أو خرج) مصفوفة COMPLEX*16، البعد (LDX,NRHS)
!>          عند الإدخال، إذا كان XTYPE = 'C' (لـ 'محسوب')، فإن X يحتوي على
!>          الحل الدقيق لنظام المعادلات الخطية.
!>          عند الخروج، إذا كان XTYPE = 'N' (لـ 'جديد')، فإن X مهيأة
!>          بقِيم عشوائية.
!>

LDX

!>          LDX هي INTEGER
!>          البعد الرائد للمصفوفة X. إذا كان TRANS = 'N'،
!>          LDX >= max(1,N)؛ إذا كان TRANS = 'T'، LDX >= max(1,M).
!>

B

!>          B هي مصفوفة COMPLEX*16، البعد (LDB,NRHS)
!>          متجه/متجهات الجانب الأيمن لنظام المعادلات،
!>          محسوبة من B = op(A) * X، حيث يتم تحديد op(A) بواسطة
!>          TRANS.
!>

LDB

!>          LDB هي INTEGER
!>          البعد الرائد للمصفوفة B. إذا كان TRANS = 'N'،
!>          LDB >= max(1,M)؛ إذا كان TRANS = 'T'، LDB >= max(1,N).
!>

ISEED

!>          ISEED هي مصفوفة INTEGER، البعد (4)
!>          متجه البذرة لمولد الأرقام العشوائي (المستخدم في
!>          ZLATMS).  يتم تعديله عند الخروج.
!>

INFO

!>          INFO هي INTEGER
!>          = 0: خروج ناجح
!>          < 0: إذا كان INFO = -i، فإن المعلمة i-th كانت لها قيمة غير قانونية
!>
