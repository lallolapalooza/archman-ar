ZLA_GBRFSX_EXTENDED يحسن الحل المحسوب لنظام المعادلات الخطية للمصفوفات ذات النطاق العام عن طريق إجراء تنقيح تكراري فائق الدقة ويوفر حدود الخطأ وتقديرات الخطأ الخلفي للحل.

الغرض:

!>
!> ZLA_GBRFSX_EXTENDED يحسن الحل المحسوب لنظام المعادلات الخطية عن طريق
!> إجراء تنقيح تكراري فائق الدقة ويوفر حدود الخطأ وتقديرات الخطأ الخلفي للحل.
!> يتم استدعاء هذا الإجراء الفرعي بواسطة ZGBRFSX لإجراء التنقيح التكراري.
!> بالإضافة إلى حد الخطأ المعياري، يوفر الكود حد الخطأ المكوني النسبي الأقصى
!> إذا كان ذلك ممكنًا. راجع التعليقات الخاصة بـ ERR_BNDS_NORM
!> و ERR_BNDS_COMP للحصول على تفاصيل حدود الخطأ. لاحظ أن هذا
!> الإجراء الفرعي مسؤول فقط عن تعيين الحقول الثانية لـ
!> ERR_BNDS_NORM و ERR_BNDS_COMP.
!>

المعلمات

PREC_TYPE 
!>          PREC_TYPE هو INTEGER
!>     يحدد الدقة الوسيطة المراد استخدامها في التنقيح.
!>     يتم تحديد القيمة بواسطة ILAPREC(P) حيث P هو CHARACTER و P
!>          = 'S':  Single (فردي)
!>          = 'D':  Double (مزدوج)
!>          = 'I':  Indigenous (أصلي)
!>          = 'X' or 'E':  Extra (إضافي)
!>

TRANS_TYPE

!>          TRANS_TYPE هو INTEGER
!>     يحدد عملية النقل على A.
!>     يتم تحديد القيمة بواسطة ILATRANS(T) حيث T هو CHARACTER و T
!>          = 'N':  No transpose (لا يوجد نقل)
!>          = 'T':  Transpose (نقل)
!>          = 'C':  Conjugate transpose (نقل مترافق)
!>

N

!>          N هو INTEGER
!>     عدد المعادلات الخطية، أي ترتيب
!>     المصفوفة A.  N >= 0.
!>

KL

!>          KL هو INTEGER
!>     عدد الأقطار الفرعية داخل نطاق A.  KL >= 0.
!>

KU

!>          KU هو INTEGER
!>     عدد الأقطار الفائقة داخل نطاق A.  KU >= 0
!>

NRHS

!>          NRHS هو INTEGER
!>     عدد الأطراف اليمنى، أي عدد أعمدة
!>     المصفوفة B.
!>

AB

!>          AB هو COMPLEX*16 array, dimension (LDAB,N)
!>     عند الدخول، المصفوفة A ذات البعد N*N.
!>

LDAB

!>          LDAB هو INTEGER
!>     البعد الرئيسي للمصفوفة A.  LDAB >= max(1,N).
!>

AFB

!>          AFB هو COMPLEX*16 array, dimension (LDAF,N)
!>     العوامل L و U من تحليل
!>     A = P*L*U كما تم حسابه بواسطة ZGBTRF.
!>

LDAFB

!>          LDAFB هو INTEGER
!>     البعد الرئيسي للمصفوفة AF.  LDAF >= max(1,N).
!>

IPIV

!>          IPIV هو INTEGER array, dimension (N)
!>     مؤشرات المحور من تحليل A = P*L*U
!>     كما تم حسابه بواسطة ZGBTRF؛ تم تبديل الصف i من المصفوفة
!>     مع الصف IPIV(i).
!>

COLEQU

!>          COLEQU هو LOGICAL
!>     إذا كانت .TRUE. فقد تم إجراء موازنة الأعمدة لـ A قبل استدعاء
!>     هذا الإجراء. هذا ضروري لحساب الحل وحدود الخطأ
!>     بشكل صحيح.
!>

C

!>          C هو DOUBLE PRECISION array, dimension (N)
!>     عوامل قياس الأعمدة لـ A. إذا كانت COLEQU = .FALSE.،
!>     فلا يتم الوصول إلى C. إذا تم إدخال C، فيجب أن يكون كل عنصر من C
!>     قوة من الأساس لضمان حل موثوق به وتقديرات للخطأ.
!>     الموازنة بقوى الأساس لا تسبب أخطاء تقريب إلا إذا
!>     كانت النتيجة تقل أو تفيض. تؤدي أخطاء التقريب أثناء الموازنة
!>     إلى التنقيح بمصفوفة ليست مكافئة للمدخلات
!>     ، مما ينتج تقديرات خطأ قد لا تكون
!>     موثوقة.
!>

B

!>          B هو COMPLEX*16 array, dimension (LDB,NRHS)
!>     مصفوفة الطرف الأيمن B.
!>

LDB

!>          LDB هو INTEGER
!>     البعد الرئيسي للمصفوفة B.  LDB >= max(1,N).
!>

Y

!>          Y هو COMPLEX*16 array, dimension (LDY,NRHS)
!>     عند الدخول، مصفوفة الحل X، كما تم حسابها بواسطة ZGBTRS.
!>     عند الخروج، مصفوفة الحل المحسنة Y.
!>

LDY

!>          LDY هو INTEGER
!>     البعد الرئيسي للمصفوفة Y.  LDY >= max(1,N).
!>

BERR_OUT

!>          BERR_OUT هو DOUBLE PRECISION array, dimension (NRHS)
!>     عند الخروج، يحتوي BERR_OUT(j) على الخطأ الخلفي النسبي المكوني
!>     للجانب الأيمن j من الصيغة
!>         max(i) ( abs(RES(i)) / ( abs(op(A_s))*abs(Y) + abs(B_s) )(i) )
!>     حيث abs(Z) هي القيمة المطلقة المكونية للمصفوفة
!>     أو المتجه Z. يتم حساب هذا بواسطة ZLA_LIN_BERR.
!>

N_NORMS

!>          N_NORMS هو INTEGER
!>     يحدد حدود الخطأ التي سيتم إرجاعها (انظر ERR_BNDS_NORM
!>     و ERR_BNDS_COMP).
!>     إذا كان N_NORMS >= 1، فسيتم إرجاع حدود الخطأ المعيارية.
!>     إذا كان N_NORMS >= 2، فسيتم إرجاع حدود الخطأ المكونية.
!>

ERR_BNDS_NORM

!>          ERR_BNDS_NORM هو DOUBLE PRECISION array, dimension (NRHS, N_ERR_BNDS)
!>     لكل جانب أيمن، تحتوي هذه المصفوفة على معلومات حول
!>     حدود الخطأ المختلفة وأرقام الشرط المقابلة لـ
!>     الخطأ النسبي المعياري، والذي يتم تعريفه على النحو التالي:
!>
!>     الخطأ النسبي المعياري في متجه الحل i:
!>             max_j (abs(XTRUE(j,i) - X(j,i)))
!>            ------------------------------
