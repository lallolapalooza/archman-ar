subroutine zlarfb_gett (character ident, integer m, integer n, integer k, complex*16, dimension( ldt, * ) t, integer ldt, complex*16, dimension( lda, * ) a, integer lda, complex*16, dimension( ldb, * ) b, integer ldb, complex*16, dimension( ldwork, * ) work, integer ldwork)

ZLARFB_GETT

الغرض:

!>
!> ZLARFB_GETT يطبق مصفوفة انعكاس كتل هوت هاوسر معقدة H من اليسار
!> على مصفوفة معقدة (K+M)-by-N تتكون من مصفوفتين كتلتين: مصفوفة
!> شبه منحرفة علوية K-by-N مخزنة في المصفوفة A، ومصفوفة مستطيلة
!> M-by-(N-K) مخزنة في المصفوفة B. مصفوفة الانعكاس الكتلي H مخزنة
!> بتمثيل WY مضغوط، حيث توجد مصفوفات الانعكاس الأولية في
!> المصفوفات A، B، و T. انظر قسم التفاصيل الإضافية.
!>

المعلمات

IDENT 
!>          IDENT is CHARACTER*1
!>          إذا كان IDENT = ليس 'I'، أو ليس 'i'، فإن V1 هي مصفوفة
!>             وحدوية مثلثة سفلية ومخزنة في الكتلة اليسرى K-by-K
!>             للمصفوفة المدخلة A،
!>          إذا كان IDENT = 'I' أو 'i'، فإن V1 هي مصفوفة الهوية
!>             وغير مخزنة.
!>          انظر قسم التفاصيل الإضافية.
!>

M

!>          M is INTEGER
!>          عدد الصفوف في المصفوفة B.
!>          M >= 0.
!>

N

!>          N is INTEGER
!>          عدد الأعمدة في المصفوفتين A و B.
!>          N >= 0.
!>

K

!>          K is INTEGER
!>          عدد صفوف المصفوفة A.
!>          K هو أيضًا رتبة المصفوفة T، أي عدد
!>          مصفوفات الانعكاس الأولية التي يحدد منتجها
!>          مصفوفة الانعكاس الكتلي. 0 <= K <= N.
!>

T

!>          T is COMPLEX*16 array, dimension (LDT,K)
!>          المصفوفة المثلثية العلوية K-by-K T في تمثيل
!>          مصفوفة الانعكاس الكتلي.
!>

LDT

!>          LDT is INTEGER
!>          البعد الرائد للمصفوفة T. LDT >= K.
!>

A

!>          A is COMPLEX*16 array, dimension (LDA,N)
!>
!>          عند الإدخال:
!>           أ) في الجزء العلوي الشبه منحرف K-by-N A: المصفوفة المدخلة A.
!>           ب) في الأعمدة تحت القطر: أعمدة V1
!>              (الآحاد غير مخزنة على القطر).
!>
!>          عند الخروج:
!>            يتم استبدال A بالناتج المستطيل K-by-N H*A.
!>
!>          انظر قسم التفاصيل الإضافية.
!>

LDA

!>          LDB is INTEGER
!>          البعد الرائد للمصفوفة A. LDA >= max(1,K).
!>

B

!>          B is COMPLEX*16 array, dimension (LDB,N)
!>
!>          عند الإدخال:
!>            أ) في الكتلة اليمنى M-by-(N-K): المصفوفة المدخلة B.
!>            ب) في الكتلة اليسرى M-by-N: أعمدة V2.
!>
!>          عند الخروج:
!>            يتم استبدال B بالناتج المستطيل M-by-N H*B.
!>
!>          انظر قسم التفاصيل الإضافية.
!>

LDB

!>          LDB is INTEGER
!>          البعد الرائد للمصفوفة B. LDB >= max(1,M).
!>

WORK

!>          WORK is COMPLEX*16 array,
!>          dimension (LDWORK,max(K,N-K))
!>

LDWORK

!>          LDWORK is INTEGER
!>          البعد الرائد للمصفوفة WORK. LDWORK>=max(1,K).
!>
!>

تفاصيل إضافية:

!>
!>    (1) وصف العملية الجبرية.
!>
!>    المصفوفة A هي مصفوفة K-by-N تتكون من مصفوفتين كتلتين عموديتين،
!>    A1، وهي K-by-K، و A2، وهي K-by-(N-K):
!>    A = ( A1, A2 ).
!>    المصفوفة B هي مصفوفة M-by-N تتكون من مصفوفتين كتلتين عموديتين،
!>    B1، وهي M-by-K، و B2، وهي M-by-(N-K):
!>    B = ( B1, B2 ).
!>
!>    قم بتنفيذ العملية:
!>
!>       ( A_out ) := H * ( A_in ) = ( I - V * T * V**H ) * ( A_in ) =
!>       ( B_out )        ( B_in )                          ( B_in )
!>                  = ( I - ( V1 ) * T * ( V1**H, V2**H ) ) * ( A_in )
!>                          ( V2 )                            ( B_in )
!>     عند الإدخال:
!>
!>    أ) ( A_in ) تتكون من كتلتين عموديتين:
!>       ( B_in )
!>
!>       ( A_in ) = (( A1_in ) ( A2_in )) = (( A1_in ) ( A2_in ))
!>       ( B_in )   (( B1_in ) ( B2_in ))   ((     0 ) ( B2_in )),
!>
!>       حيث الكتل العمودية هي:
!>
!>       (  A1_in )  هي مصفوفة شبه منحرفة علوية K-by-K مخزنة في
!>                   الجزء العلوي الشبه منحرف من المصفوفة A(1:K,1:K).
!>       (  B1_in )  هي مصفوفة صفرية مستطيلة M-by-K وغير مخزنة.
!>
!>       ( A2_in )  هي مصفوفة مستطيلة K-by-(N-K) مخزنة
!>                  في المصفوفة A(1:K,K+1:N).
!>       ( B2_in )  هي مصفوفة مستطيلة M-by-(N-K) مخزنة
!>                  في المصفوفة B(1:M,K+1:N).
!>
!>    ب) V = ( V1 )
!>           ( V2 )
!>
!>       حيث:
!>       1) إذا كان IDENT == 'I'، فإن V1 هي مصفوفة هوية K-by-K، غير مخزنة؛
!!>       2) إذا كان IDENT != 'I'، فإن V1 هي مصفوفة مثلثة سفلية وحدوية K-by-K،
!>          مخزنة في الجزء السفلي المثلثي من المصفوفة
!>          A(1:K,1:K) (الآحاد غير مخزنة)،
!>       و V2 هي مصفوفة مستطيلة M-by-K مخزنة في المصفوفة B(1:M,1:K)،
!>                 (لأن B1_in عند الإدخال هي مصفوفة صفرية مستطيلة غير مخزنة ويتم استخدام المساحة
!>                  لتخزين V2).
!>
!>    ج) T هي مصفوفة مثلثية علوية K-by-K مخزنة
!>       في المصفوفة T(1:K,1:K).
!>
!>    عند الخروج:
!>
!>    أ) ( A_out ) تتكون من كتلتين عموديتين:
!>       ( B_out )
!>
!>       ( A_out ) = (( A1_out ) ( A2_out ))
!>       ( B_out )   (( B1_out ) ( B2_out )),
!>
!>       حيث الكتل العمودية هي:
!>
!>       ( A1_out )  هي مصفوفة مربعة K-by-K، أو مصفوفة
!>                   مثلثية علوية K-by-K، إذا كانت V1
!>                   مصفوفة هوية. يتم تخزين AiOut في
!>                   المصفوفة A(1:K,1:K).
!>       ( B1_out )  هي مصفوفة مستطيلة M-by-K مخزنة
!>                   في المصفوفة B(1:M,K:N).
!>
!>       ( A2_out )  هي مصفوفة مستطيلة K-by-(N-K) مخزنة
!>                   في المصفوفة A(1:K,K+1:N).
!>       ( B2_out )  هي مصفوفة مستطيلة M-by-(N-K) مخزنة
!>                   في المصفوفة B(1:M,K+1:N).
!>
!>
!>    يمكن تمثيل العملية المذكورة أعلاه كعملية متماثلة
!>    على كل كتلة عمودية:
!>
!>       ( A1_out ) := H * ( A1_in ) = ( I - V * T * V**H ) * ( A1_in )
!>       ( B1_out )        (     0 )                          (     0 )
!>
!>       ( A2_out ) := H * ( A2_in ) = ( I - V * T * V**H ) * ( A2_in )
!>       ( B2_out )        ( B2_in )                          ( B2_in )
!>
!>    إذا كان IDENT != 'I':
!>
!>       الحساب للكتلة العمودية 1:
!>
!>       A1_out: = A1_in - V1*T*(V1**H)*A1_in
!>
!>       B1_out: = - V2*T*(V1**H)*A1_in
!>
!>       الحساب للكتلة العمودية 2، والتي توجد إذا كان N > K:
!>
!>       A2_out: = A2_in - V1*T*( (V1**H)*A2_in + (V2**H)*B2_in )
!>
!>       B2_out: = B2_in - V2*T*( (V1**H)*A2_in + (V2**H)*B2_in )
!>
!>    إذا كان IDENT == 'I':
!>
!>       العملية للكتلة العمودية 1:
!>
!>       A1_out: = A1_in - V1*T*A1_in
!>
!>       B1_out: = - V2*T*A1_in
!>
!>       الحساب للكتلة العمودية 2، والتي توجد إذا كان N > K:
!>
!>       A2_out: = A2_in - V1*T*( (V1**H)*A2_in + (V2**H)*B2_in )
!>
!>       B2_out: = B2_in - V2*T*( (V1**H)*A2_in + (V2**H)*B2_in )
!>
!>
