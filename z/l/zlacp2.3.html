subroutine zlacp2 (character uplo, integer m, integer n, double precision, dimension( lda, * ) a, integer lda, complex*16, dimension( ldb, * ) b, integer ldb)

ZLACP2 تنسخ جزءًا أو كل مصفوفة حقيقية ثنائية الأبعاد إلى مصفوفة معقدة.

الغرض:

!>
!> ZLACP2 تنسخ مصفوفة حقيقية ثنائية الأبعاد A، كلها أو جزء منها، إلى مصفوفة
!> معقدة B.
!>

المعلمات

UPLO 
!>          UPLO هو CHARACTER*1
!>          يحدد الجزء من المصفوفة A الذي سيتم نسخه إلى B.
!>          = 'U':      الجزء العلوي المثلثي
!>          = 'L':      الجزء السفلي المثلثي
!>          غير ذلك:   كل المصفوفة A
!>

M

!>          M هو INTEGER
!>          عدد صفوف المصفوفة A.  M >= 0.
!>

N

!>          N هو INTEGER
!>          عدد أعمدة المصفوفة A.  N >= 0.
!>

A

!>          A هي مصفوفة DOUBLE PRECISION، ذات بُعد (LDA,N)
!>          المصفوفة A ذات الأبعاد m × n.  إذا كان UPLO = 'U'، يتم الوصول
!>          فقط إلى شبه المنحرف العلوي؛ إذا كان UPLO = 'L'، يتم الوصول
!>          فقط إلى شبه المنحرف السفلي.
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرائد للمصفوفة A.  LDA >= max(1,M).
!>

B

!>          B هي مصفوفة COMPLEX*16، ذات بُعد (LDB,N)
!>          عند الخروج، B = A في المواقع المحددة بواسطة UPLO.
!>

LDB

!>          LDB هو INTEGER
!>          البعد الرائد للمصفوفة B.  LDB >= max(1,M).
!>
