توثيق الوظيفة/الروتين الفرعي

subroutine zlaipd (integer n, complex*16, dimension( * ) a, integer inda, integer vinda)

ZLAIPD

الغرض:

!>
!> ZLAIPD تضبط الجزء التخيلي للعناصر القطرية للمصفوفة المعقدة A بقيمة كبيرة.
!> هذا يستخدم لاختبار روتينات LAPACK للمصفوفات الهرميتية المعقدة،
!> والتي لا يُفترض أن تصل إلى أو تستخدم الأجزاء التخيلية للأقطار.
!>

المعلمات

N 
!>          N هو عدد صحيح (INTEGER)
!>         عدد العناصر القطرية للمصفوفة A.
!>

A

!>          A هي مصفوفة COMPLEX*16، البعد
!>                        (1+(N-1)*INDA+(N-2)*VINDA)
!>         عند الدخول، المصفوفة المعقدة (الهرميتية) A.
!>         عند الخروج، تم ضبط الأجزاء التخيلية للعناصر القطرية
!>         إلى BIGNUM = EPS / SAFMIN، حيث EPS هو إبسيلون الآلة
!>         و SAFMIN هو الحد الأدنى الآمن.
!>

INDA

!>          INDA هو عدد صحيح (INTEGER)
!>         الزيادة بين A(1) والعنصر القطري التالي للمصفوفة A.
!>         القيم النموذجية هي
!>         = LDA+1:  المصفوفات المربعة ذات البعد الرائد LDA
!>         = 2:  مصفوفة مثلثية علوية مضغوطة، تبدأ من A(1,1)
!>         = N:  مصفوفة مثلثية سفلية مضغوطة، تبدأ من A(1,1)
!>

VINDA

!>          VINDA هو عدد صحيح (INTEGER)
!>         التغيير في الزيادة القطرية بين أعمدة A.
!>         القيم النموذجية هي
!>         = 0:  لا يوجد تغيير، فواصل الصف والعمود في A ثابتة
!>         = 1:  مصفوفة مثلثية علوية مضغوطة
!>         = -1:  مصفوفة مثلثية سفلية مضغوطة
!>
