subroutine zlatzm (character side, integer m, integer n, complex*16, dimension( * ) v, integer incv, complex*16 tau, complex*16, dimension( ldc, * ) c1, complex*16, dimension( ldc, * ) c2, integer ldc, complex*16, dimension( * ) work)

ZLATZM

الغرض:

!>
!> تم إهمال هذا الإجراء الفرعي واستبداله بالإجراء الفرعي ZUNMRZ.
!>
!> ZLATZM يطبق مصفوفة هاوسهولدر تم إنشاؤها بواسطة ZTZRQF على مصفوفة.
!>
!> لتكن P = I - tau*u*u**H،   u = ( 1 )،
!>                               ( v )
!> حيث v هو متجه (m-1) إذا كان SIDE = 'L'، أو متجه (n-1) إذا كان
!> SIDE = 'R'.
!>
!> إذا كان SIDE يساوي 'L'، لتكن
!>        C = [ C1 ] 1
!>            [ C2 ] m-1
!>              n
!> إذن C يتم استبداله بواسطة P*C.
!>
!> إذا كان SIDE يساوي 'R'، لتكن
!>        C = [ C1, C2 ] m
!>               1  n-1
!> إذن C يتم استبداله بواسطة C*P.
!>

المعلمات

SIDE 
!>          SIDE هو CHARACTER*1
!>          = 'L': تشكيل P * C
!>          = 'R': تشكيل C * P
!>

M

!>          M هو INTEGER
!>          عدد صفوف المصفوفة C.
!>

N

!>          N هو INTEGER
!>          عدد أعمدة المصفوفة C.
!>

V

!>          V هو مصفوفة COMPLEX*16، الأبعاد
!>                  (1 + (M-1)*abs(INCV)) إذا كان SIDE = 'L'
!>                  (1 + (N-1)*abs(INCV)) إذا كان SIDE = 'R'
!>          المتجه v في تمثيل P. لا يتم استخدام V
!>          إذا كان TAU = 0.
!>

INCV

!>          INCV هو INTEGER
!>          الزيادة بين عناصر v. INCV <> 0
!>

TAU

!>          TAU هو COMPLEX*16
!>          القيمة tau في تمثيل P.
!>

C1

!>          C1 هو مصفوفة COMPLEX*16، الأبعاد
!>                         (LDC,N) إذا كان SIDE = 'L'
!>                         (M,1)   إذا كان SIDE = 'R'
!>          عند الدخول، المتجه n C1 إذا كان SIDE = 'L'، أو المتجه m C1
!>          إذا كان SIDE = 'R'.
!>
!>          عند الخروج، الصف الأول من P*C إذا كان SIDE = 'L'، أو العمود
!>          الأول من C*P إذا كان SIDE = 'R'.
!>

C2

!>          C2 هو مصفوفة COMPLEX*16، الأبعاد
!>                         (LDC, N)   إذا كان SIDE = 'L'
!>                         (LDC, N-1) إذا كان SIDE = 'R'
!>          عند الدخول، المصفوفة C2 التي أبعادها (m - 1) x n إذا كان SIDE = 'L'، أو
!>          المصفوفة C2 التي أبعادها m x (n - 1) إذا كان SIDE = 'R'.
!>
!>          عند الخروج، الصفوف 2:m من P*C إذا كان SIDE = 'L'، أو الأعمدة 2:m من C*P
!>          إذا كان SIDE = 'R'.
!>

LDC

!>          LDC هو INTEGER
!>          البعد الرئيسي للمصفوفتين C1 و C2.
!>          LDC >= max(1,M).
!>

WORK

!>          WORK هو مصفوفة COMPLEX*16، الأبعاد
!>                      (N) إذا كان SIDE = 'L'
!>                      (M) إذا كان SIDE = 'R'
!>
