subroutine zlagtm (trans, n, nrhs, alpha, dl, d, du, x, ldx, beta, b, ldb)
ZLAGTM تنفذ ضرب مصفوفة-مصفوفة بالشكل C = αAB+βC، حيث A هي مصفوفة ثلاثية الأقطار، B و C هما مصفوفتان مستطيلتان، و α و β هما عددان قياسيان، والتي قد تكون 0، 1، أو -1.

subroutine zlagtm (character trans, integer n, integer nrhs, double precision alpha, complex*16, dimension( * ) dl, complex*16, dimension( * ) d, complex*16, dimension( * ) du, complex*16, dimension( ldx, * ) x, integer ldx, double precision beta, complex*16, dimension( ldb, * ) b, integer ldb)

ZLAGTM تنفذ ضرب مصفوفة-مصفوفة بالشكل C = αAB+βC، حيث A هي مصفوفة ثلاثية الأقطار، B و C هما مصفوفتان مستطيلتان، و α و β هما عددان قياسيان، والتي قد تكون 0، 1، أو -1.

الغرض:

!>
!> ZLAGTM تنفذ ضرب مصفوفة-مصفوفة بالشكل
!>
!>    B := alpha * A * X + beta * B
!>
!> حيث A هي مصفوفة ثلاثية الأقطار من الرتبة N، و B و X هما مصفوفتان N × NRHS،
!> و alpha و beta هما عددان قياسيان حقيقيان، وكلاهما قد يكون
!> 0.، 1.، أو -1.
!>

المعلمات

TRANS 
!>          TRANS هو CHARACTER*1
!>          يحدد العملية المطبقة على A.
!>          = 'N':  بدون تبديل، B := alpha * A * X + beta * B
!>          = 'T':  تبديل،    B := alpha * A**T * X + beta * B
!>          = 'C':  تبديل مترافق، B := alpha * A**H * X + beta * B
!>

N

!>          N هو INTEGER
!>          رتبة المصفوفة A.  N >= 0.
!>

NRHS

!>          NRHS هو INTEGER
!>          عدد الأطراف اليمنى، أي عدد أعمدة
!>          المصفوفتين X و B.
!>

ALPHA

!>          ALPHA هو DOUBLE PRECISION
!>          العدد القياسي alpha.  يجب أن يكون ALPHA 0.، 1.، أو -1.؛ وإلا،
!>          يُفترض أنه 0.
!>

DL

!>          DL هو COMPLEX*16 array، البعد (N-1)
!>          عناصر القطر السفلي (N-1) للمصفوفة T.
!>

D

!>          D هو COMPLEX*16 array، البعد (N)
!>          عناصر القطر الرئيسي للمصفوفة T.
!>

DU

!>          DU هو COMPLEX*16 array، البعد (N-1)
!>          عناصر القطر العلوي (N-1) للمصفوفة T.
!>

X

!>          X هو COMPLEX*16 array، البعد (LDX,NRHS)
!>          المصفوفة X ذات الأبعاد N × NRHS.
!>

LDX

!>          LDX هو INTEGER
!>          البعد الرئيسي للمصفوفة X.  LDX >= max(N,1).
!>

BETA

!>          BETA هو DOUBLE PRECISION
!>          العدد القياسي beta.  يجب أن يكون BETA 0.، 1.، أو -1.؛ وإلا،
!>          يُفترض أنه 1.
!>

B

!>          B هو COMPLEX*16 array، البعد (LDB,NRHS)
!>          عند الدخول، المصفوفة B ذات الأبعاد N × NRHS.
!>          عند الخروج، يتم استبدال B بالتعبير المصفوفي
!>          B := alpha * A * X + beta * B.
!>

LDB

!>          LDB هو INTEGER
!>          البعد الرئيسي للمصفوفة B.  LDB >= max(N,1).
!>
