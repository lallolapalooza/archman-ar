subroutine zlatmt (integer m, integer n, character dist, integer, dimension( 4 ) iseed, character sym, double precision, dimension( * ) d, integer mode, double precision cond, double precision dmax, integer rank, integer kl, integer ku, character pack, complex*16, dimension( lda, * ) a, integer lda, complex*16, dimension( * ) work, integer info)

ZLATMT

الغرض:

!>
!>    ZLATMT يقوم بتوليد مصفوفات عشوائية بقيم مفردة محددة
!>    (أو هيرميتية بقيم ذاتية محددة)
!>    لاختبار برامج LAPACK.
!>
!>    تعمل ZLATMT عن طريق تطبيق التسلسل التالي من
!>    العمليات:
!>
!>      تعيين القطر إلى D، حيث يمكن أن يكون D مدخلاً أو
!>         محسوبًا وفقًا لـ MODE، COND، DMAX، و SYM
!>         كما هو موضح أدناه.
!>
!>      توليد مصفوفة ذات هيكل النطاق المناسب، بواسطة إحدى
!>         الطريقتين:
!>
!>      الطريقة A:
!>          توليد مصفوفة كثيفة M x N عن طريق ضرب D من اليسار
!>              ومن اليمين بمصفوفات وحدوية عشوائية، ثم:
!>
!>          تقليل عرض النطاق وفقًا لـ KL و KU، باستخدام
!>              تحويلات هاوسهولدر.
!>
!>      الطريقة B:
!>          تحويل مصفوفة عرض النطاق 0 (أي قطرية) إلى
!>              مصفوفة عرض النطاق 1 باستخدام دورانات جيفنز،
!>              العناصر خارج النطاق للخلف، بشكل مشابه لـ QR؛ ثم تحويل
!>              عرض النطاق 1 إلى عرض النطاق 2، إلخ. لاحظ
!>              أنه بالنسبة لعروض النطاقات الصغيرة نسبيًا (مقارنة بـ M و
!>              N) هذا يتطلب تخزينًا أقل، حيث لا يتم توليد مصفوفة كثيفة.
!>              أيضًا، بالنسبة للمصفوفات الهيرميتية أو المتماثلة،
!>              يتم توليد مثلث واحد فقط.
!>
!>      يتم اختيار الطريقة A إذا كان عرض النطاق يمثل نسبة كبيرة من
!>          رتبة المصفوفة، وكان LDA على الأقل M (بحيث يمكن تخزين
!>          مصفوفة كثيفة). يتم اختيار الطريقة B إذا كان عرض النطاق
!>          صغيرًا (< 1/2 N للمصفوفات الهيرميتية أو المتماثلة، < .3 N+M
!>          للمصفوفات غير المتماثلة)، أو كان LDA أقل من M وليس أقل من
!>          عرض النطاق.
!>
!>      تعبئة المصفوفة إذا لزم الأمر. الخيارات المحددة بواسطة PACK هي:
!>         لا يوجد تعبئة
!>         صفر النصف العلوي (إذا كانت هيرميتية)
!>         صفر النصف السفلي (إذا كانت هيرميتية)
!>         تخزين النصف العلوي في شكل عمودي (إذا كانت هيرميتية أو
!>               مثلثة علوية)
!>         تخزين النصف السفلي في شكل عمودي (إذا كانت هيرميتية أو
!>               مثلثة سفلية)
!>         تخزين المثلث السفلي بتنسيق نطاق (إذا كانت هيرميتية أو
!>               مثلثة سفلية)
!>         تخزين المثلث العلوي بتنسيق نطاق (إذا كانت هيرميتية أو
!>               مثلثة علوية)
!>         تخزين المصفوفة بأكملها بتنسيق نطاق
!>      إذا تم اختيار الطريقة B، وتم تحديد تنسيق النطاق، فسيتم
!>         توليد المصفوفة بتنسيق النطاق، لذلك لن تكون هناك حاجة لإعادة التعبئة.
!>

المعلمات

M 
!>          M هو عدد صحيح (INTEGER)
!>           عدد صفوف A. لا يتم تعديله.
!>

N

!>          N هو عدد صحيح (INTEGER)
!>           عدد أعمدة A. يجب أن يساوي N قيمة M إذا كانت المصفوفة
!>           متماثلة أو هيرميتية (أي، إذا كانت SYM ليست 'N')
!>           لا يتم تعديله.
!>

DIST

!>          DIST هو حرف واحد (CHARACTER*1)
!>           عند الإدخال، يحدد DIST نوع التوزيع الذي سيتم استخدامه
!>           لتوليد القيم الذاتية/المفردة العشوائية.
!>           'U' => منتظم( 0, 1 )  ( 'U' تعني منتظم )
!>           'S' => منتظم( -1, 1 ) ( 'S' تعني متماثل )
!>           'N' => طبيعي( 0, 1 )   ( 'N' تعني طبيعي )
!>           لا يتم تعديله.
!>

ISEED

!>          ISEED هو مصفوفة أعداد صحيحة (INTEGER array)، البعد ( 4 )
!>           عند الإدخال، يحدد ISEED بذرة مولد الأرقام العشوائية.
!>           يجب أن تتراوح القيم بين 0 و 4095 شاملة، ويجب أن يكون ISEED(4) فرديًا.
!>           يستخدم مولد الأرقام العشوائية تسلسلًا خطيًا متطابقًا محدودًا بأعداد صحيحة صغيرة،
!>           ويجب أن ينتج أرقامًا عشوائية مستقلة عن الآلة.
!>           تتغير قيم ISEED عند الخروج، ويمكن استخدامها في الاستدعاء التالي لـ ZLATMT
!>           لمواصلة نفس تسلسل الأرقام العشوائية.
!>           تم تعديله عند الخروج.
!>

SYM

!>          SYM هو حرف واحد (CHARACTER*1)
!>           إذا كانت SYM='H'، فإن المصفوفة المولدة تكون هيرميتية،
!>             بقيمة ذاتية محددة بواسطة D، COND، MODE، و DMAX؛ يمكن أن تكون
!>             موجبة، سالبة، أو صفرية.
!>           إذا كانت SYM='P'، فإن المصفوفة المولدة تكون هيرميتية،
!>             بقيمة ذاتية ( = قيمة مفردة) محددة بواسطة D، COND،
!>             MODE، و DMAX؛ لن تكون سالبة.
!>           إذا كانت SYM='N'، فإن المصفوفة المولدة تكون غير متماثلة،
!>             بقيمة مفردة محددة بواسطة D، COND، MODE، و DMAX؛
!>             لن تكون سالبة.
!>           إذا كانت SYM='S'، فإن المصفوفة المولدة تكون متماثلة (مركبة)،
!>             بقيمة مفردة محددة بواسطة D، COND، MODE، و
!>             DMAX؛ لن تكون سالبة.
!>           لا يتم تعديله.
!>

D

!>          D هي مصفوفة ذات دقة مزدوجة (DOUBLE PRECISION array)، البعد ( MIN( M, N ) )
!>           تستخدم هذه المصفوفة لتحديد القيم المفردة أو
!>           القيم الذاتية لـ A (انظر SYM، أعلاه). إذا كان MODE=0، فسيتم
!>           افتراض أن D تحتوي على القيم الذاتية/المفردة، وإلا
!>           سيتم حسابها وفقًا لـ MODE، COND، و DMAX،
!>           وسيتم وضعها في D.
!>           تم تعديلها إذا كان MODE غير صفري.
!>

MODE

!>          MODE هو عدد صحيح (INTEGER)
!>           عند الإدخال، يصف هذا كيف سيتم تحديد القيم الذاتية/المفردة:
!>           MODE = 0 يعني استخدام D كمدخل
!>           MODE = 1 يعين D(1)=1 و D(2:RANK)=1.0/COND
!>           MODE = 2 يعين D(1:RANK-1)=1 و D(RANK)=1.0/COND
!>           MODE = 3 يعين D(I)=COND**(-(I-1)/(RANK-1))
!>           MODE = 4 يعين D(i)=1 - (i-1)/(N-1)*(1 - 1/COND)
!>           MODE = 5 يعين D لأرقام عشوائية في النطاق
!>                    ( 1/COND , 1 ) بحيث تكون لوغاريتماتها
!>                    موزعة بشكل منتظم.
!>           MODE = 6 يعين D لأرقام عشوائية من نفس التوزيع
!>                    مثل بقية المصفوفة.
!>           MODE < 0 له نفس المعنى مثل ABS(MODE)، باستثناء
!>              أن ترتيب عناصر D يتم عكسه.
!>           وبالتالي، إذا كان MODE موجبًا، فإن D تحتوي على مدخلات تتراوح من
!>              1 إلى 1/COND، وإذا كان سالبًا، من 1/COND إلى 1،
!>           إذا كانت SYM='H'، وكان MODE ليس 0، 6، أو -6، فإن
!>              سيتم أيضًا ضرب عناصر D بإشارة عشوائية (أي +1 أو -1).
!>           لا يتم تعديله.
!>

COND

!>          COND هو دقة مزدوجة (DOUBLE PRECISION)
!>           عند الإدخال، يتم استخدامه كما هو موضح تحت MODE أعلاه.
!>           إذا تم استخدامه، يجب أن يكون >= 1. لا يتم تعديله.
!>

DMAX

!>          DMAX هو دقة مزدوجة (DOUBLE PRECISION)
!>           إذا كان MODE ليس -6، 0، أو 6، سيتم قياس محتويات D، كما
!>           تم حسابها وفقًا لـ MODE و COND، بواسطة
!>           DMAX / max(abs(D(i))); وبالتالي، فإن أقصى قيمة ذاتية أو
!>           مفردة مطلقة (وهو المعيار) ستكون abs(DMAX).
!>           لاحظ أن DMAX ليست بالضرورة موجبة: إذا كانت DMAX سالبة
!>           (أو صفرية)، فسيتم قياس D برقم سالب (أو صفر).
!>           لا يتم تعديله.
!>

RANK

!>          RANK هو عدد صحيح (INTEGER)
!>           رتبة المصفوفة المراد توليدها للأوضاع 1، 2، 3 فقط.
!>           D( RANK+1:N ) = 0.
!>           لا يتم تعديله.
!>
