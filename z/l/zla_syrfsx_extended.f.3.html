subroutine zla_syrfsx_extended (integer prec_type, character uplo, integer n, integer nrhs, complex*16, dimension( lda, * ) a, integer lda, complex*16, dimension( ldaf, * ) af, integer ldaf, integer, dimension( * ) ipiv, logical colequ, double precision, dimension( * ) c, complex*16, dimension( ldb, * ) b, integer ldb, complex*16, dimension( ldy, * ) y, integer ldy, double precision, dimension( * ) berr_out, integer n_norms, double precision, dimension( nrhs, * ) err_bnds_norm, double precision, dimension( nrhs, * ) err_bnds_comp, complex*16, dimension( * ) res, double precision, dimension( * ) ayb, complex*16, dimension( * ) dy, complex*16, dimension( * ) y_tail, double precision rcond, integer ithresh, double precision rthresh, double precision dz_ub, logical ignore_cwise, integer info)

ZLA_SYRFSX_EXTENDED تحسّن الحل المحسوب لنظام المعادلات الخطية للمصفوفات المتماثلة غير المحددة عن طريق إجراء تحسين تكراري فائق الدقة وتوفر حدود الخطأ وتقديرات الخطأ العكسي للحل.

الغرض:

!>
!> ZLA_SYRFSX_EXTENDED تحسّن الحل المحسوب لنظام المعادلات الخطية عن طريق إجراء تحسين تكراري فائق الدقة
!> وتوفر حدود الخطأ وتقديرات الخطأ العكسي للحل.
!> يتم استدعاء هذه الوحدة الفرعية بواسطة ZSYRFSX لإجراء التحسين التكراري.
!> بالإضافة إلى حد الخطأ المعياري، يوفر الكود حد الخطأ الأقصى المكون من مكونات إذا كان ذلك ممكنًا.
!> انظر التعليقات حول ERR_BNDS_NORM و ERR_BNDS_COMP لتفاصيل حدود الخطأ.
!> لاحظ أن هذه الوحدة الفرعية مسؤولة فقط عن تعيين الحقول الثانية
!> لـ ERR_BNDS_NORM و ERR_BNDS_COMP.
!>

المعلمات

PREC_TYPE 
!>          PREC_TYPE هو INTEGER
!>     يحدد الدقة الوسيطة التي سيتم استخدامها في التحسين.
!>     يتم تعريف القيمة بواسطة ILAPREC(P) حيث P هو CHARACTER و P
!>          = 'S':  Single (فردي)
!>          = 'D':  Double (مزدوج)
!>          = 'I':  Indigenous (أصلي)
!>          = 'X' أو 'E':  Extra (إضافي)
!>

UPLO

!>          UPLO هو CHARACTER*1
!>       = 'U':  يتم تخزين المثلث العلوي للمصفوفة A؛
!>       = 'L':  يتم تخزين المثلث السفلي للمصفوفة A.
!>

N

!>          N هو INTEGER
!>     عدد المعادلات الخطية، أي رتبة
!>     المصفوفة A.  N >= 0.
!>

NRHS

!>          NRHS هو INTEGER
!>     عدد الأطراف اليمنى، أي عدد أعمدة
!>     المصفوفة B.
!>

A

!>          A هو مصفوفة COMPLEX*16، البعد (LDA,N)
!>     عند الإدخال، المصفوفة A بحجم N*N.
!>

LDA

!>          LDA هو INTEGER
!>     البعد الرئيسي للمصفوفة A.  LDA >= max(1,N).
!>

AF

!>          AF هو مصفوفة COMPLEX*16، البعد (LDAF,N)
!>     المصفوفة القطرية الجزئية D والعوامل المستخدمة
!>     للحصول على العامل U أو L كما تم حسابه بواسطة ZSYTRF.
!>

LDAF

!>          LDAF هو INTEGER
!>     البعد الرئيسي للمصفوفة AF.  LDAF >= max(1,N).
!>

IPIV

!>          IPIV هو مصفوفة INTEGER، البعد (N)
!>     تفاصيل التبادلات والبنية الجزئية لـ D
!>     كما تم تحديدها بواسطة ZSYTRF.
!>

COLEQU

!>          COLEQU هو LOGICAL
!>     إذا كانت .TRUE. فقد تم إجراء موازنة أعمدة للمصفوفة A قبل استدعاء
!>     هذه الوحدة الفرعية. هذا ضروري لحساب الحل وحدود الخطأ
!>     بشكل صحيح.
!>

C

!>          C هو مصفوفة DOUBLE PRECISION، البعد (N)
!>     عوامل قياس الأعمدة لـ A. إذا كانت COLEQU = .FALSE.،
!>     لم يتم الوصول إلى C. إذا تم إدخال C، يجب أن يكون كل عنصر من C قوة
!>     للأساس لضمان حل وتقديرات خطأ موثوقة.
!>     القياس بقوى الأساس لا يسبب أخطاء تقريب إلا إذا
!>     كانت النتيجة تحت التدفق أو تجاوزت التدفق. أخطاء التقريب أثناء القياس
!>     تؤدي إلى تحسين بمصفوفة غير مكافئة للمصفوفة
!>     المدخلة، مما ينتج تقديرات خطأ قد لا تكون
!>     موثوقة.
!>

B

!>          B هو مصفوفة COMPLEX*16، البعد (LDB,NRHS)
!>     المصفوفة الطرفية اليمنى B.
!>

LDB

!>          LDB هو INTEGER
!>     البعد الرئيسي للمصفوفة B.  LDB >= max(1,N).
!>

Y

!>          Y هو مصفوفة COMPLEX*16، البعد (LDY,NRHS)
!>     عند الإدخال، مصفوفة الحل X، كما تم حسابها بواسطة ZSYTRS.
!>     عند الخروج، مصفوفة الحل المحسنة Y.
!>

LDY

!>          LDY هو INTEGER
!>     البعد الرئيسي للمصفوفة Y.  LDY >= max(1,N).
!>

BERR_OUT

!>          BERR_OUT هو مصفوفة DOUBLE PRECISION، البعد (NRHS)
!>     عند الخروج، تحتوي BERR_OUT(j) على الخطأ العكسي النسبي المكون من مكونات
!>     للطرف الأيمن j من الصيغة
!>         max(i) ( abs(RES(i)) / ( abs(op(A_s))*abs(Y) + abs(B_s) )(i) )
!>     حيث abs(Z) هي القيمة المطلقة المكونة من مكونات المصفوفة
!>     أو المتجه Z. يتم حساب هذا بواسطة ZLA_LIN_BERR.
!>

N_NORMS

!>          N_NORMS هو INTEGER
!>     يحدد حدود الخطأ التي سيتم إرجاعها (انظر ERR_BNDS_NORM
!>     و ERR_BNDS_COMP).
!>     إذا كانت N_NORMS >= 1، فارجع حدود الخطأ المعيارية.
!>     إذا كانت N_NORMS >= 2، فارجع حدود الخطأ المكونة من مكونات.
!>

ERR_BNDS_NORM

!>          ERR_BNDS_NORM هو مصفوفة DOUBLE PRECISION، البعد (NRHS, N_ERR_BNDS)
!>     لكل طرف أيمن، تحتوي هذه المصفوفة على معلومات حول
!>     حدود الخطأ المختلفة وأرقام الشرط المقابلة للخطأ
!>     المعياري النسبي، والذي يتم تعريفه على النحو التالي:
!>
!>     الخطأ المعياري النسبي في متجه الحل i:
!>             max_j (abs(XTRUE(j,i) - X(j,i)))
!>            ------------------------------
!>                  max_j abs(X(j,i))
!>
!>     يتم فهرسة المصفوفة حسب نوع معلومات الخطأ على النحو الموضح
!>     أدناه. يوجد حاليًا ما يصل إلى ثلاثة قطع من المعلومات
!>     يتم إرجاعها.
!>
!>     الفهرس الأول في ERR_BNDS_NORM(i,:) يتوافق مع الطرف
!>     الأيمن i.
!>
!>     الفهرس الثاني في ERR_BNDS_NORM(:,err) يحتوي على الحقول الثلاثة التالية:
!>     err = 1  منطقية. ثق بالإجابة إذا كان
!>              معكوس رقم الشرط أقل من العتبة
!>              sqrt(n) * sl

ERR_BNDS_COMP

!>          ERR_BNDS_COMP هو مصفوفة DOUBLE PRECISION، البعد (NRHS, N_ERR_BNDS)
!>     لكل طرف أيمن، تحتوي هذه المصفوفة على معلومات حول
!>     حدود الخطأ المختلفة وأرقام الشرط المقابلة للخطأ
!>     المكون من مكونات، والذي يتم تعريفه على النحو التالي:
!>
!>     الخطأ المكون من مكونات النسبي في متجه الحل i:
!>             max_j (abs(XTRUE(j,i) - X(j,i)))
!>            ------------------------------
!>                  max_j abs(X(j,i))
!>
!>     يتم فهرسة المصفوفة حسب نوع معلومات الخطأ على النحو الموضح
!>     أدناه. يوجد حاليًا ما يصل إلى ثلاثة قطع من المعلومات
!>     يتم إرجاعها.
!>
!>     الفهرس الأول في ERR_BNDS_COMP(i,:) يتوافق مع الطرف
!>     الأيمن i.
!>
!>     الفهرس الثاني في ERR_BNDS_COMP(:,err) يحتوي على الحقول الثلاثة التالية:
!>     err = 1  منطقية. ثق بالإجابة إذا كان
!>              معكوس رقم الشرط أقل من العتبة
!>              sqrt(n) * sl

RES

!>          RES هو مصفوفة COMPLEX*16، البعد (N)
!>     المتجه residual، أي `op(A)*Y - B`.
!>

AYB

!>          AYB هو مصفوفة DOUBLE PRECISION، البعد (N)
!>     القيمة المطلقة المكونة من مكونات للمتجه `op(A)*Y`.
!>

DY

!>          DY هو مصفوفة COMPLEX*16، البعد (N)
!>     الفرق بين الحل الجديد والقديم.
!>

Y_TAIL

!>          Y_TAIL هو مصفوفة COMPLEX*16، البعد (N)
!>     إذا تم استخدام التحسين التكراري، فإن هذا هو الحل المحسن.
!>

RCOND

!>          RCOND هو DOUBLE PRECISION
!>     تقدير معكوس رقم شرط المصفوفة.
!>

ITHRESH

!>          ITHRESH هو INTEGER
!>     عتبة للتحكم في ما إذا كان سيتم إجراء التحسين التكراري.
!>

RTHRESH

!>          RTHRESH هو DOUBLE PRECISION
!>     عتبة لتحديد ما إذا كان سيتم قبول التحسين.
!>

DZ_UB

!>          DZ_UB هو DOUBLE PRECISION
!>     الحد الأعلى لـ |x - xtrue| / |x| (الخطأ النسبي).
!>

IGNORE_CWISE

!>          IGNORE_CWISE هو LOGICAL
!>     إذا كانت .TRUE.، فسيتم تجاهل حدود الخطأ المكونة من مكونات.
!>

INFO

!>          INFO هو INTEGER
!>     = 0:  نجاح.
!>     < 0:  إذا كانت القيمة -i، فإن المعلمة i-th في الاستدعاء
!>          كانت غير صحيحة.
!>     > 0:  العدد الفعلي للتكرارات التي تم إجراؤها.
!>
