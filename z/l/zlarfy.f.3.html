subroutine zlarfy (character uplo, integer n, complex*16, dimension( * ) v, integer incv, complex*16 tau, complex*16, dimension( ldc, * ) c, integer ldc, complex*16, dimension( * ) work)

ZLARFY

الغرض:

!>
!> ZLARFY يطبق عاكس أولي، أو مصفوفة هاوسهولدر، H،
!> على مصفوفة هيرميتية n x n وهي C، من اليسار واليمين.
!>
!> H ممثلة بالشكل
!>
!>    H = I - tau * v * v'
!>
!> حيث tau قيمة عددية و v متجه.
!>
!> إذا كان tau صفرًا، يتم اعتبار H مصفوفة الوحدة.
!>

المعاملات

UPLO 
!>          UPLO هو CHARACTER*1
!>          يحدد ما إذا كان الجزء العلوي أو السفلي من المثلث المصفوفة الهيرميتية C
!>          مخزنًا.
!>          = 'U':  الجزء العلوي
!>          = 'L':  الجزء السفلي
!>

N

!>          N هو INTEGER
!>          عدد الصفوف والأعمدة للمصفوفة C. N >= 0.
!>

V

!>          V هو مصفوفة COMPLEX*16، ذات البعد
!>                  (1 + (N-1)*abs(INCV))
!>          المتجه v كما هو موصوف أعلاه.
!>

INCV

!>          INCV هو INTEGER
!>          الزيادة بين العناصر المتتالية لـ v. يجب ألا تكون INCV صفرًا.
!>

TAU

!>          TAU هو COMPLEX*16
!>          القيمة tau كما هو موصوف أعلاه.
!>

C

!>          C هو مصفوفة COMPLEX*16، ذات البعد (LDC, N)
!>          عند الدخول، المصفوفة C.
!>          عند الخروج، يتم الكتابة فوق C بـ H * C * H'.
!>

LDC

!>          LDC هو INTEGER
!>          البعد الرائد للمصفوفة C. LDC >= max( 1, N ).
!>

WORK

!>          WORK هو مصفوفة COMPLEX*16، ذات البعد (N)
!>
