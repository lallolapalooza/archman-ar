subroutine zlatm1 (integer mode, double precision cond, integer irsign, integer idist, integer, dimension( 4 ) iseed, complex*16, dimension( * ) d, integer n, integer info)

ZLATM1

الغرض:

!>
!>    ZLATM1 يحسب مدخلات D(1..N) كما هو محدد بواسطة
!>    MODE, COND و IRSIGN. IDIST و ISEED يحددان توليد
!>    الأرقام العشوائية. ZLATM1 يتم استدعاؤه بواسطة ZLATMR لتوليد
!>    مصفوفات اختبار عشوائية لبرامج LAPACK.
!>

المعلمات

MODE 
!>          MODE هو عدد صحيح (INTEGER)
!>           عند الإدخال يصف كيف سيتم حساب D:
!>           MODE = 0 يعني لا تغير D.
!>           MODE = 1 يضبط D(1)=1 و D(2:N)=1.0/COND
!>           MODE = 2 يضبط D(1:N-1)=1 و D(N)=1.0/COND
!>           MODE = 3 يضبط D(I)=COND**(-(I-1)/(N-1))
!>           MODE = 4 يضبط D(i)=1 - (i-1)/(N-1)*(1 - 1/COND)
!>           MODE = 5 يضبط D إلى أرقام عشوائية في النطاق
!>                    ( 1/COND , 1 ) بحيث تكون لوغاريتماتها
!>                    موزعة بشكل موحد.
!>           MODE = 6 يضبط D إلى أرقام عشوائية من نفس التوزيع
!>                    مثل بقية المصفوفة.
!>           MODE < 0 له نفس المعنى مثل ABS(MODE)، باستثناء أن
!>              ترتيب عناصر D معكوس.
!>           لذلك إذا كان MODE موجبًا، فإن عناصر D تتراوح من
!>              1 إلى 1/COND، وإذا كان سالبًا، فمن 1/COND إلى 1،
!>           غير معدل.
!>

COND

!>          COND هو دقة مزدوجة (DOUBLE PRECISION)
!>           عند الإدخال، يستخدم كما هو موضح تحت MODE أعلاه.
!>           إذا تم استخدامه، يجب أن يكون >= 1. غير معدل.
!>

IRSIGN

!>          IRSIGN هو عدد صحيح (INTEGER)
!>           عند الإدخال، إذا كان MODE ليس -6، 0 أو 6، يحدد إشارة
!>           مدخلات D
!>           0 => ترك مدخلات D دون تغيير
!>           1 => ضرب كل مدخل من D برقم مركب عشوائي
!>                موزع بشكل موحد بقيمة مطلقة 1
!>

IDIST

!>          IDIST هو عدد صحيح (INTEGER)
!>           عند الإدخال، يحدد IDIST نوع التوزيع الذي سيتم
!>           استخدامه لتوليد مصفوفة عشوائية.
!>           1 => الجزء الحقيقي والتخيلي كل منهما UNIFORM( 0, 1 )
!>           2 => الجزء الحقيقي والتخيلي كل منهما UNIFORM( -1, 1 )
!>           3 => الجزء الحقيقي والتخيلي كل منهما NORMAL( 0, 1 )
!>           4 => عدد مركب موزع بشكل موحد في DISK( 0, 1 )
!>           غير معدل.
!>

ISEED

!>          ISEED هو مصفوفة أعداد صحيحة، بُعدها ( 4 )
!>           عند الإدخال، يحدد ISEED البذرة لمولد الأرقام
!>           العشوائية. يستخدم مولد الأرقام العشوائية تسلسل
!>           خطي متوافق محدود بأعداد صحيحة صغيرة، ولذلك يجب أن
!>           ينتج أرقامًا عشوائية مستقلة عن الآلة. تتغير قيم ISEED
!>           عند الخروج، ويمكن استخدامها في الاستدعاء التالي لـ ZLATM1
!>           لمواصلة نفس تسلسل الأرقام العشوائية.
!>           تتغير عند الخروج.
!>

D

!>          D هي مصفوفة COMPLEX*16، بُعدها ( N )
!>           المصفوفة المطلوب حسابها وفقًا لـ MODE و COND و IRSIGN.
!>           قد تتغير عند الخروج إذا كان MODE غير صفري.
!>

N

!>          N هو عدد صحيح (INTEGER)
!>           عدد مدخلات D. غير معدل.
!>

INFO

!>          INFO هو عدد صحيح (INTEGER)
!>            0  => إنهاء طبيعي
!>           -1  => إذا كان MODE خارج النطاق -6 إلى 6
!>           -2  => إذا كان MODE ليس -6، 0 أو 6، و
!>                  IRSIGN ليس 0 أو 1
!>           -3  => إذا كان MODE ليس -6، 0 أو 6 و COND أقل من 1
!>           -4  => إذا كان MODE يساوي 6 أو -6 و IDIST خارج النطاق 1 إلى 4
!>           -7  => إذا كان N سالبًا
!>
