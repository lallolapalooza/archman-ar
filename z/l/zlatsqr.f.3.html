توثيق الوظيفة/الروتين الفرعي

subroutine zlatsqr (integer m, integer n, integer mb, integer nb, complex*16, dimension( lda, * ) a, integer lda, complex*16, dimension( ldt, * ) t, integer ldt, complex*16, dimension( * ) work, integer lwork, integer info)

ZLATSQR

الغرض:

!>
!> ZLATSQR يحسب عامل QR عمودي نحيف (Tall-Skinny QR) محظور لمصفوفة معقدة M × N
!> A حيث M >= N:
!>
!>    A = Q * ( R ),
!>            ( 0 )
!>
!> حيث:
!>
!>    Q مصفوفة متعامدة M × M، مخزنة عند الخروج في شكل ضمني
!>    في العناصر أسفل القطر في المصفوفة A وعناصر المصفوفة T؛
!>
!>    R مصفوفة علوية مثلثة N × N، مخزنة عند الخروج
!>    في العناصر على القطر وفوقه في المصفوفة A.
!>
!>    0 مصفوفة صفرية (M-N) × N، وغير مخزنة.
!>
!>

المعاملات

M 
!>          M هو عدد صحيح (INTEGER)
!>          عدد صفوف المصفوفة A.  M >= 0.
!>

N

!>          N هو عدد صحيح (INTEGER)
!>          عدد أعمدة المصفوفة A. M >= N >= 0.
!>

MB

!>          MB هو عدد صحيح (INTEGER)
!>          حجم الكتلة الصفية المستخدمة في QR المحظور.
!>          MB > N.
!>

NB

!>          NB هو عدد صحيح (INTEGER)
!>          حجم الكتلة العمودية المستخدمة في QR المحظور.
!>          N >= NB >= 1.
!>

A

!>          A هي مصفوفة COMPLEX*16، أبعادها (LDA,N)
!>          عند الدخول، المصفوفة A التي أبعادها M × N.
!>          عند الخروج، تحتوي العناصر على القطر وفوقه
!>          من المصفوفة على المصفوفة المثلثية العلوية R التي أبعادها N × N؛
!>          العناصر أسفل القطر تمثل Q بواسطة أعمدة
!>          V المحظورة (انظر التفاصيل الإضافية).
!>

LDA

!>          LDA هو عدد صحيح (INTEGER)
!>          البعد الرئيسي للمصفوفة A.  LDA >= max(1,M).
!>

T

!>          T هي مصفوفة COMPLEX*16،
!>          أبعادها (LDT, N * عدد_الكتل_الصفية)
!>          حيث عدد_الكتل_الصفية = CEIL((M-N)/(MB-N))
!>          تعكس الكتل المثلثية العلوية المحظورة المخزنة في شكل مضغوط
!>          كسلسلة من الكتل المثلثية العلوية.
!>          انظر التفاصيل الإضافية أدناه.
!>

LDT

!>          LDT هو عدد صحيح (INTEGER)
!>          البعد الرئيسي للمصفوفة T.  LDT >= NB.
!>

WORK

!>          (مساحة عمل) مصفوفة COMPLEX*16، أبعادها (MAX(1,LWORK))
!>          عند الخروج، إذا كان INFO = 0، فإن WORK(1) يعيد الحد الأدنى LWORK.
!>

LWORK

!>          LWORK هو عدد صحيح (INTEGER)
!>          بعد المصفوفة WORK.
!>          LWORK >= 1، إذا كان MIN(M,N) = 0، وإلا فإن LWORK >= NB*N.
!>
!>          إذا كان LWORK = -1، فسيتم افتراض استعلام مساحة عمل؛ يقوم الروتين
!>          فقط بحساب الحد الأدنى لحجم مصفوفة WORK، ويعيد
!>          هذه القيمة كمدخل أول لمصفوفة WORK، ولا يتم إصدار أي رسالة خطأ
!>          متعلقة بـ LWORK بواسطة XERBLA.
!>

INFO

!>          INFO هو عدد صحيح (INTEGER)
!>          = 0:  خروج ناجح
!>          < 0:  إذا كان INFO = -i، فإن المعامل i كان لديه قيمة غير قانونية
!>

تفاصيل إضافية:

!> QR العمودي النحيف (TSQR) يقوم بإجراء QR بواسطة سلسلة من التحويلات المتعامدة،
!> تمثل Q كمنتج لمصفوفات متعامدة أخرى
!>   Q = Q(1) * Q(2) * . . . * Q(k)
!> حيث يقوم كل Q(i) بتصفير الإدخالات تحت القطر لمجموعة من MB صفوف من A:
!>   Q(1) يصفر الإدخالات تحت القطر للصفوف 1:MB من A
!>   Q(2) يصفر الصفوف السفلية MB-N من الصفوف [1:N,MB+1:2*MB-N] من A
!>   Q(3) يصفر الصفوف السفلية MB-N من الصفوف [1:N,2*MB-N+1:3*MB-2*N] من A
!>   . . .
!>
!> يتم حساب Q(1) بواسطة GEQRT، الذي يمثل Q(1) بواسطة متجهات هاوسهولدر
!> مخزنة أسفل القطر لصفوف 1:MB من A، وبواسطة كتل مثلثة علوية
!> الانعكاسات، مخزنة في المصفوفة T(1:LDT,1:N).
!> لمزيد من المعلومات، انظر التفاصيل الإضافية في GEQRT.
!>
!> يتم حساب Q(i) لـ i>1 بواسطة TPQRT، الذي يمثل Q(i) بواسطة متجهات هاوسهولدر
!> مخزنة في الصفوف [(i-1)*(MB-N)+N+1:i*(MB-N)+N] من A، وبواسطة كتل مثلثة علوية
!> الانعكاسات، مخزنة في المصفوفة T(1:LDT,(i-1)*N+1:i*N).
!> قد تستخدم Q(k) الأخيرة عددًا أقل من الصفوف.
!> لمزيد من المعلومات، انظر التفاصيل الإضافية في TPQRT.
!>
!> لمزيد من التفاصيل حول الخوارزمية العامة، انظر وصف
!> TSQR المتسلسل في القسم 2.2 من [1].
!>
!> [1] "عوامل QR وLU المتوازية والمتسلسلة المثلى من حيث الاتصال،"
!>     J. Demmel، L. Grigori، M. Hoemmen، J. Langou،
!>     SIAM J. Sci. Comput، المجلد 34، رقم 1، 2012
!>
