subroutine zla_geamv (integer trans, integer m, integer n, double precision alpha, complex*16, dimension( lda, * ) a, integer lda, complex*16, dimension( * ) x, integer incx, double precision beta, double precision, dimension( * ) y, integer incy)

ZLA_GEAMV تحسب حاصل ضرب مصفوفة-متجه باستخدام مصفوفة عامة لحساب حدود الخطأ.

الغرض:

!>
!> ZLA_GEAMV  تقوم بإحدى عمليات المصفوفة-المتجه التالية:
!>
!>         y := alpha*abs(A)*abs(x) + beta*abs(y),
!>    أو   y := alpha*abs(A)**T*abs(x) + beta*abs(y),
!>
!> حيث alpha و beta هما قيمتان قياسيتان، x و y هما متجهان، و A مصفوفة
!> بحجم m × n.
!>
!> هذه الوظيفة تستخدم بشكل أساسي في حساب حدود الخطأ.
!> للحماية من التدفق السفلي أثناء التقييم، يتم إزاحة المكونات في
!> المتجه الناتج بعيدًا عن الصفر بمقدار (N+1)
!> ضعف عتبة التدفق السفلي. لمنع الأخطاء الكبيرة بشكل غير ضروري
!> بالنسبة للهياكل المتداخلة في المصفوفات العامة،
!>  لا يتم إزاحة المكونات الصفرية. يتم اعتبار الإدخال الصفري
!> إذا كانت جميع عمليات الضرب المتضمنة
!> في حساب هذا الإدخال تحتوي على عامل ضرب صفري واحد على الأقل.
!>

المعلمات

TRANS 
!>          TRANS هو عدد صحيح (INTEGER)
!>           عند الإدخال، يحدد TRANS العملية التي سيتم تنفيذها كما يلي:
!>
!>             BLAS_NO_TRANS      y := alpha*abs(A)*abs(x) + beta*abs(y)
!>             BLAS_TRANS         y := alpha*abs(A**T)*abs(x) + beta*abs(y)
!>             BLAS_CONJ_TRANS    y := alpha*abs(A**T)*abs(x) + beta*abs(y)
!>
!>           لا يتغير عند الخروج.
!>

M

!>          M هو عدد صحيح (INTEGER)
!>           عند الإدخال، يحدد M عدد صفوف المصفوفة A.
!>           يجب أن يكون M أكبر من أو يساوي صفر.
!>           لا يتغير عند الخروج.
!>

N

!>          N هو عدد صحيح (INTEGER)
!>           عند الإدخال، يحدد N عدد أعمدة المصفوفة A.
!>           يجب أن يكون N أكبر من أو يساوي صفر.
!>           لا يتغير عند الخروج.
!>

ALPHA

!>          ALPHA هو عدد عشري مزدوج الدقة (DOUBLE PRECISION)
!>           عند الإدخال، يحدد ALPHA القيمة القياسية alpha.
!>           لا يتغير عند الخروج.
!>

A

!>          A هو مصفوفة أعداد مركبة بدقة مزدوجة (COMPLEX*16)، ذات بُعد ( LDA, n )
!>           قبل الإدخال، يجب أن يحتوي الجزء العلوي بحجم m × n من المصفوفة A على
!>           المعاملات.
!>           لا يتغير عند الخروج.
!>

LDA

!>          LDA هو عدد صحيح (INTEGER)
!>           عند الإدخال، يحدد LDA البعد الأول لـ A كما هو معلن
!>           في البرنامج (الفرعي) المستدعي. يجب أن يكون LDA على الأقل
!>           max( 1, m ).
!>           لا يتغير عند الخروج.
!>

X

!>          X هو مصفوفة أعداد مركبة بدقة مزدوجة (COMPLEX*16)، ذات بُعد
!>           ( 1 + ( n - 1 )*abs( INCX ) ) عندما تكون TRANS = 'N' أو 'n'
!>           وعلى الأقل
!>           ( 1 + ( m - 1 )*abs( INCX ) ) بخلاف ذلك.
!>           قبل الإدخال، يجب أن تحتوي المصفوفة X المتزايدة على
!>           المتجه x.
!>           لا يتغير عند الخروج.
!>

INCX

!>          INCX هو عدد صحيح (INTEGER)
!>           عند الإدخال، يحدد INCX الزيادة لعناصر X.
!>           يجب ألا يكون INCX صفراً.
!>           لا يتغير عند الخروج.
!>

BETA

!>          BETA هو عدد عشري مزدوج الدقة (DOUBLE PRECISION)
!>           عند الإدخال، يحدد BETA القيمة القياسية beta. عندما يتم
!>           توفير BETA بصفر، لا يلزم تعيين Y عند الإدخال.
!>           لا يتغير عند الخروج.
!>

Y

!>          Y هو مصفوفة أعداد عشرية مزدوجة الدقة (DOUBLE PRECISION)، ذات بُعد
!>           ( 1 + ( m - 1 )*abs( INCY ) ) عندما تكون TRANS = 'N' أو 'n'
!>           وعلى الأقل
!>           ( 1 + ( n - 1 )*abs( INCY ) ) بخلاف ذلك.
!>           قبل الإدخال مع BETA غير صفري، يجب أن تحتوي المصفوفة Y المتزايدة
!>           على المتجه y. عند الخروج، يتم الكتابة فوق Y بالمتجه y المحدث.
!>           إذا كان m أو n صفراً، فلن يتم الرجوع إلى Y وستقوم الوظيفة
!>           بالعودة بسرعة.
!>

INCY

!>          INCY هو عدد صحيح (INTEGER)
!>           عند الإدخال، يحدد INCY الزيادة لعناصر Y.
!>           يجب ألا يكون INCY صفراً.
!>           لا يتغير عند الخروج.
!>
!>  إجراء فرعي من المستوى 2 من BLAS.
!>
