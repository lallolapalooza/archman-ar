subroutine zlarzt (direct, storev, n, k, v, ldv, tau, t, ldt)
ZLARZT تشكل العامل المثلثي T للمنعكس القطاعي H = I - vtvH.

توثيق الوظيفة/الروتين الفرعي

subroutine zlarzt (character direct, character storev, integer n, integer k, complex*16, dimension( ldv, * ) v, integer ldv, complex*16, dimension( * ) tau, complex*16, dimension( ldt, * ) t, integer ldt)

ZLARZT تشكل العامل المثلثي T للمنعكس القطاعي H.

الغرض:

!>
!> ZLARZT تشكل العامل المثلثي T للمنعكس القطاعي المعقد H الذي ترتيبه > n،
!> والذي يُعرّف كحاصل ضرب k من المنعكسات الأولية.
!>
!> إذا كان DIRECT = 'F'، فإن H = H(1) H(2) . . . H(k) و T مثلثية عليا؛
!>
!> إذا كان DIRECT = 'B'، فإن H = H(k) . . . H(2) H(1) و T مثلثية سفلى.
!>
!> إذا كان STOREV = 'C'، يتم تخزين المتجه الذي يعرّف المنعكس الأولي H(i)
!> في العمود i من المصفوفة V، و
!>
!>    H  =  I - V * T * V**H
!>
!> إذا كان STOREV = 'R'، يتم تخزين المتجه الذي يعرّف المنعكس الأولي H(i)
!> في الصف i من المصفوفة V، و
!>
!>    H  =  I - V**H * T * V
!>
!> حاليًا، يتم دعم STOREV = 'R' و DIRECT = 'B' فقط.
!>

المعاملات

DIRECT 
!>          DIRECT is CHARACTER*1
!>          يحدد ترتيب ضرب المنعكسات الأولية لتشكيل المنعكس القطاعي:
!>          = 'F': H = H(1) H(2) . . . H(k) (أمامي، غير مدعوم بعد)
!>          = 'B': H = H(k) . . . H(2) H(1) (خلفي)
!>

STOREV

!>          STOREV is CHARACTER*1
!>          يحدد كيفية تخزين المتجهات التي تعرّف المنعكسات الأولية (انظر أيضًا
!>          تفاصيل إضافية):
!>          = 'C': عمودياً                        (غير مدعوم بعد)
!>          = 'R': أفقياً
!>

N

!>          N is INTEGER
!>          ترتيب المنعكس القطاعي H. N >= 0.
!>

K

!>          K is INTEGER
!>          ترتيب العامل المثلثي T (= عدد المنعكسات الأولية). K >= 1.
!>

V

!>          V is COMPLEX*16 array, dimension
!>                               (LDV,K) if STOREV = 'C'
!>                               (LDV,N) if STOREV = 'R'
!>          المصفوفة V. انظر تفاصيل إضافية.
!>

LDV

!>          LDV is INTEGER
!>          البعد الرائد للمصفوفة V.
!>          إذا كان STOREV = 'C'، فإن LDV >= max(1,N)؛ إذا كان STOREV = 'R'، فإن LDV >= K.
!>

TAU

!>          TAU is COMPLEX*16 array, dimension (K)
!>          يجب أن تحتوي TAU(i) على العامل القياسي للمنعكس الأولي H(i).
!>

T

!>          T is COMPLEX*16 array, dimension (LDT,K)
!>          العامل المثلثي k × k للمنعكس القطاعي.
!>          إذا كان DIRECT = 'F'، فإن T مثلثية عليا؛ إذا كان DIRECT = 'B'، فإن T
!>          مثلثية سفلى. بقية المصفوفة غير مستخدمة.
!>

LDT

!>          LDT is INTEGER
!>          البعد الرائد للمصفوفة T. LDT >= K.
!>

تفاصيل إضافية:

!>
!>  يتم توضيح شكل المصفوفة V وتخزين المتجهات التي تعرّف H(i) بأفضل شكل
!>  من خلال المثال التالي مع n = 5 و k = 3. العناصر التي تساوي 1 غير مخزنة؛
!>  يتم تعديل عناصر المصفوفة المقابلة ولكن يتم استعادتها عند الخروج.
!>  بقية المصفوفة غير مستخدمة.
!>
!>  DIRECT = 'F' و STOREV = 'C':         DIRECT = 'F' و STOREV = 'R':
!>
!>                                              ______V_____
!>         ( v1 v2 v3 )                        /            \
!>         ( v1 v2 v3 )                      ( v1 v1 v1 v1 v1 . . . . 1 )
!>     V = ( v1 v2 v3 )                      ( v2 v2 v2 v2 v2 . . . 1   )
!>         ( v1 v2 v3 )                      ( v3 v3 v3 v3 v3 . . 1     )
!>         ( v1 v2 v3 )
!>            .  .  .
!>            .  .  .
!>            1  .  .
!>               1  .
!>                  1
!>
!>  DIRECT = 'B' و STOREV = 'C':         DIRECT = 'B' و STOREV = 'R':
!>
!>                                                        ______V_____
!>            1                                          /            \
!>            .  1                           ( 1 . . . . v1 v1 v1 v1 v1 )
!>            .  .  1                        ( . 1 . . . v2 v2 v2 v2 v2 )
!>            .  .  .                        ( . . 1 . . v3 v3 v3 v3 v3 )
!>            .  .  .
!>         ( v1 v2 v3 )
!>         ( v1 v2 v3 )
!>     V = ( v1 v2 v3 )
!>         ( v1 v2 v3 )
!>         ( v1 v2 v3 )
!>
