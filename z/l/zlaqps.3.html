subroutine zlaqps (integer m, integer n, integer offset, integer nb, integer kb, complex*16, dimension( lda, * ) a, integer lda, integer, dimension( * ) jpvt, complex*16, dimension( * ) tau, double precision, dimension( * ) vn1, double precision, dimension( * ) vn2, complex*16, dimension( * ) auxv, complex*16, dimension( ldf, * ) f, integer ldf)

ZLAQPS يحسب خطوة من تحليل QR مع تبديل الأعمدة لمصفوفة حقيقية m-by-n A باستخدام BLAS المستوى 3.

الغرض:

!>
!> ZLAQPS يحسب خطوة من تحليل QR مع تبديل الأعمدة
!> لمصفوفة مركبة M-by-N A باستخدام BLAS-3. يحاول تحليل
!> NB عمودًا من A بدءًا من الصف OFFSET+1، ويحدث كل
!> المصفوفة باستخدام BLAS-3 xGEMM.
!>
!> في بعض الحالات، بسبب الإلغاءات الكارثية، لا يمكنه
!> تحليل NB عمودًا.
!> وبالتالي، يتم إرجاع العدد الفعلي للأعمدة المحللة في KB.
!>
!> الكتلة A(1:OFFSET,1:N) يتم تبديلها وفقًا لذلك، ولكن لم يتم تحليلها.
!>

المعلمات

M 
!>          M هو عدد صحيح
!>          عدد صفوف المصفوفة A. M >= 0.
!>

N

!>          N هو عدد صحيح
!>          عدد أعمدة المصفوفة A. N >= 0
!>

OFFSET

!>          OFFSET هو عدد صحيح
!>          عدد صفوف A التي تم تحليلها في
!>          الخطوات السابقة.
!>

NB

!>          NB هو عدد صحيح
!>          عدد الأعمدة المطلوب تحليلها.
!>

KB

!>          KB هو عدد صحيح
!>          عدد الأعمدة التي تم تحليلها فعليًا.
!>

A

!>          A هو مصفوفة COMPLEX*16، البعد (LDA,N)
!>          عند الإدخال، المصفوفة A ذات الأبعاد M-by-N.
!>          عند الخروج، الكتلة A(OFFSET+1:M,1:KB) هي المصفوفة المثلثية
!>          الناتجة والكتلة A(1:OFFSET,1:N) تم تبديلها وفقًا لذلك، ولكن لم يتم تحليلها.
!>          تم تحديث بقية المصفوفة، الكتلة A(OFFSET+1:M,KB+1:N).
!>

LDA

!>          LDA هو عدد صحيح
!>          البعد الرئيسي للمصفوفة A. LDA >= max(1,M).
!>

JPVT

!>          JPVT هي مصفوفة أعداد صحيحة، البعد (N)
!>          JPVT(I) = K <==> تم تبديل العمود K من المصفوفة الكاملة A
!>          إلى الموضع I في AP.
!>

TAU

!>          TAU هي مصفوفة COMPLEX*16، البعد (KB)
!>          العوامل القياسية للموصلات الأولية.
!>

VN1

!>          VN1 هي مصفوفة DOUBLE PRECISION، البعد (N)
!>          المتجه الذي يحتوي على معايير الأعمدة الجزئية.
!>

VN2

!>          VN2 هي مصفوفة DOUBLE PRECISION، البعد (N)
!>          المتجه الذي يحتوي على معايير الأعمدة الدقيقة.
!>

AUXV

!>          AUXV هي مصفوفة COMPLEX*16، البعد (NB)
!>          متجه مساعد.
!>

F

!>          F هي مصفوفة COMPLEX*16، البعد (LDF,NB)
!>          المصفوفة F**H = L * Y**H * A.
!>

LDF

!>          LDF هو عدد صحيح
!>          البعد الرئيسي للمصفوفة F. LDF >= max(1,N).
!>
