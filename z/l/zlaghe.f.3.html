subroutine zlaghe (integer n, integer k, double precision, dimension( * ) d, complex*16, dimension( lda, * ) a, integer lda, integer, dimension( 4 ) iseed, complex*16, dimension( * ) work, integer info)

ZLAGHE

الغرض:

!>
!> ZLAGHE تولد مصفوفة هيرميتية معقدة A، عن طريق الضرب المسبق واللاحق
!> لمصفوفة قطرية حقيقية D بمصفوفة وحدوية عشوائية:
!> A = U*D*U'. يمكن بعد ذلك تقليل شبه النطاق إلى k عن طريق تحويلات وحدوية إضافية.
!>

المعلمات

N 
!>          N هو عدد صحيح (INTEGER)
!>          رتبة المصفوفة A. N >= 0.
!>

K

!>          K هو عدد صحيح (INTEGER)
!>          عدد الأقطار الفرعية غير الصفرية داخل نطاق A.
!>          0 <= K <= N-1.
!>

D

!>          D هو مصفوفة دقة مزدوجة (DOUBLE PRECISION)، البعد (N)
!>          العناصر القطرية للمصفوفة القطرية D.
!>

A

!>          A هي مصفوفة معقدة *16 (COMPLEX*16)، البعد (LDA,N)
!>          المصفوفة الهيرميتية n × n المولدة A (يتم تخزين المصفوفة الكاملة).
!>

LDA

!>          LDA هو عدد صحيح (INTEGER)
!>          البعد الرئيسي للمصفوفة A. LDA >= N.
!>

ISEED

!>          ISEED هي مصفوفة أعداد صحيحة (INTEGER)، البعد (4)
!>          عند الإدخال، البذرة لمولد الأرقام العشوائية؛ يجب أن تكون عناصر المصفوفة بين 0 و 4095، ويجب أن تكون ISEED(4)
!>          فردية.
!>          عند الخروج، يتم تحديث البذرة.
!>

WORK

!>          WORK هي مصفوفة معقدة *16 (COMPLEX*16)، البعد (2*N)
!>

INFO

!>          INFO هو عدد صحيح (INTEGER)
!>          = 0: خروج ناجح
!>          < 0: إذا كان INFO = -i، فإن المعلمة i لها قيمة غير قانونية
!>
