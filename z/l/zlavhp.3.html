subroutine zlavhp (character uplo, character trans, character diag, integer n, integer nrhs, complex*16, dimension( * ) a, integer, dimension( * ) ipiv, complex*16, dimension( ldb, * ) b, integer ldb, integer info)

ZLAVHP

الغرض:

!>
!>    ZLAVHP  ينفذ إحدى عمليات المصفوفة-المتجه
!>       x := A*x  أو  x := A^H*x,
!>    حيث x متجه مكون من N عنصر و A هي أحد العوامل
!>    من التحليل المتماثل الذي تم حسابه بواسطة ZHPTRF.
!>    ينتج ZHPTRF تحليلاً بالشكل
!>         U * D * U^H     أو     L * D * L^H,
!>    حيث U (أو L) هو ناتج ضرب مصفوفات تبديل ومصفوفات ثلاثية علوية (سفلية) موحدة،
!>    U^H (أو L^H) هو المرافق المترافق لـ U (أو L)، و D هي مصفوفة هيرميتية وقطرية مكدسة مع كتل 1x1 و 2x2.
!>    يتم تخزين المضاعفات للتحويلات والأجزاء المثلثية العلوية أو السفلية من الكتل القطرية عموديًا في صيغة مكدسة في المصفوفة الخطية A.
!>
!>    إذا كانت TRANS = 'N' أو 'n'، تضرب ZLAVHP إما بـ U أو U * D
!>    (أو L أو L * D).
!>    إذا كانت TRANS = 'C' أو 'c'، تضرب ZLAVHP إما بـ U^H أو D * U^H
!>    (أو L^H أو D * L^H).
!> 
!>  UPLO   - CHARACTER*1
!>           عند الإدخال، يحدد UPLO ما إذا كانت المصفوفة المثلثية
!>           المخزنة في A علوية أو سفلية.
!>              UPLO = 'U' أو 'u'   المصفوفة علوية.
!>              UPLO = 'L' أو 'l'   المصفوفة سفلية.
!>           لا تتغير عند الخروج.
!>
!>  TRANS  - CHARACTER*1
!>           عند الإدخال، يحدد TRANS العملية المراد تنفيذها كما يلي:
!>              TRANS = 'N' أو 'n'   x := A*x.
!>              TRANS = 'C' أو 'c'   x := A^H*x.
!>           لا تتغير عند الخروج.
!>
!>  DIAG   - CHARACTER*1
!>           عند الإدخال، يحدد DIAG ما إذا كانت الكتل القطرية
!>           يفترض أنها مصفوفات وحدة، كما يلي:
!>              DIAG = 'U' أو 'u'   الكتل القطرية هي مصفوفات وحدة.
!>              DIAG = 'N' أو 'n'   الكتل القطرية ليست وحدات.
!>           لا تتغير عند الخروج.
!>
!>  N      - INTEGER
!>           عند الإدخال، يحدد N رتبة المصفوفة A.
!>           يجب أن يكون N على الأقل صفرًا.
!>           لا يتغير عند الخروج.
!>
!>  NRHS   - INTEGER
!>           عند الإدخال، يحدد NRHS عدد الأطراف اليمنى،
!>           أي عدد المتجهات x المراد ضربها في A.
!>           يجب أن يكون NRHS على الأقل صفرًا.
!>           لا يتغير عند الخروج.
!>
!>  A      - COMPLEX*16 array, dimension( N*(N+1)/2 )
!>           عند الإدخال، تحتوي A على مصفوفة قطرية مكدسة والمضاعفات
!>           للتحويلات المستخدمة للحصول عليها،
!>           مخزنة كمصفوفة مثلثية مكدسة.
!>           لا تتغير عند الخروج.
!>
!>  IPIV   - INTEGER array, dimension( N )
!>           عند الإدخال، تحتوي IPIV على متجه مؤشرات المحور كما
!>           تم تحديده بواسطة ZSPTRF أو ZHPTRF.
!>           إذا كانت IPIV( K ) = K، لم يتم إجراء أي تبديل.
!>           إذا كانت IPIV( K ) <> K ولكن IPIV( K ) > 0، فقد تم تبديل الصف K
!>           مع الصف IPIV( K ) وتم استخدام كتلة محور 1x1.
!>           إذا كانت IPIV( K ) < 0 و UPLO = 'U'، فقد تم تبديل الصف K-1
!>           مع الصف | IPIV( K ) | وتم استخدام كتلة محور 2x2.
!>           إذا كانت IPIV( K ) < 0 و UPLO = 'L'، فقد تم تبديل الصف K+1
!>           مع الصف | IPIV( K ) | وتم استخدام كتلة محور 2x2.
!>
!>  B      - COMPLEX*16 array, dimension( LDB, NRHS )
!>           عند الإدخال، تحتوي B على NRHS متجهات بطول N.
!>           عند الخروج، يتم الكتابة فوق B بـ A * B.
!>
!>  LDB    - INTEGER
!>           عند الإدخال، يحتوي LDB على البعد الرائد لـ B كما
!>           تم تعريفه في البرنامج المستدعي. يجب أن يكون LDB على الأقل
!>           max( 1, N ).
!>           لا يتغير عند الخروج.
!>
!>  INFO   - INTEGER
!>           INFO هو رمز الخطأ.
!>           عند الخروج، يشير القيمة 0 إلى خروج ناجح.
!>           تشير قيمة سالبة، مثل -K، إلى أن الوسيط K
!>           له قيمة غير قانونية.
!>
