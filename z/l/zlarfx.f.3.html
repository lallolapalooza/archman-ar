subroutine zlarfx (side, m, n, v, tau, c, ldc, work)
ZLARFX يطبق عاكس أولي على مصفوفة مستطيلة عامة، مع فتح الحلقات عندما يكون ترتيب العاكس ≤ 10.

وثائق الوظيفة/الإجراء الفرعي

subroutine zlarfx (character side, integer m, integer n, complex*16, dimension( * ) v, complex*16 tau, complex*16, dimension( ldc, * ) c, integer ldc, complex*16, dimension( * ) work)

ZLARFX يطبق عاكس أولي مركب H على مصفوفة مركبة m × n C، من اليسار أو اليمين. يتم تمثيل H بالشكل 

       H = I - tau * v * v**H

حيث tau هو ثابت مركب و v هو متجه مركب.

إذا كان tau = 0، فيعتبر H مصفوفة الوحدة.

هذه النسخة تستخدم التعليمات البرمجية المضمنة إذا كان ترتيب H < 11.

الغرض:

!>
!> ZLARFX يطبق عاكس أولي مركب H على مصفوفة مركبة m × n
!> C، من اليسار أو اليمين. يتم تمثيل H بالشكل
!>
!>       H = I - tau * v * v**H
!>
!> حيث tau هو ثابت مركب و v هو متجه مركب.
!>
!> إذا كان tau = 0، فيعتبر H مصفوفة الوحدة.
!>
!> هذه النسخة تستخدم التعليمات البرمجية المضمنة إذا كان ترتيب H < 11.
!>

المعلمات

SIDE

!>          = 'L': form  H * C
!>          = 'R': form  C * H
!>

M

!>          M is INTEGER
!>          عدد صفوف المصفوفة C.
!>

N

!>          N is INTEGER
!>          عدد أعمدة المصفوفة C.
!>

V

!>          V is COMPLEX*16 array, dimension (M) if SIDE = 'L'
!>                                        or (N) if SIDE = 'R'
!>          المتجه v في تمثيل H.
!>

TAU

!>          TAU is COMPLEX*16
!>          القيمة tau في تمثيل H.
!>

C

!>          C is COMPLEX*16 array, dimension (LDC,N)
!>          عند الإدخال، المصفوفة C بحجم m × n.
!>          عند الخروج، يتم استبدال C بالمصفوفة H * C إذا كان SIDE = 'L'،
!>          أو C * H إذا كان SIDE = 'R'.
!>

LDC

!>          LDC is INTEGER
!>          البعد الرائد للمصفوفة C. LDC >= max(1,M).
!>

WORK

!>          WORK is COMPLEX*16 array, dimension (N) if SIDE = 'L'
!>                                            or (M) if SIDE = 'R'
!>          لا تتم الإشارة إلى WORK إذا كان ترتيب H < 11.
!>
