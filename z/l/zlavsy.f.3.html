subroutine zlavsy (character uplo, character trans, character diag, integer n, integer nrhs, complex*16, dimension( lda, * ) a, integer lda, integer, dimension( * ) ipiv, complex*16, dimension( ldb, * ) b, integer ldb, integer info)

ZLAVSY

الغرض:

!>
!> ZLAVSY تنفذ إحدى عمليات المتجه المصفوفي
!>    x := A*x  أو  x := A'*x,
!> حيث x هو متجه ذو N عنصر و A هي إحدى العوامل
!> من تحليل الكتلة U*D*U' أو L*D*L' الذي تم حسابه بواسطة ZSYTRF.
!>
!> إذا كانت TRANS = 'N'، اضرب في U  أو U * D  (أو L  أو L * D)
!> إذا كانت TRANS = 'T'، اضرب في U' أو D * U' (أو L' أو D * L')
!>

المعلمات

UPLO 
!>          UPLO هو CHARACTER*1
!>          يحدد ما إذا كان العامل المخزن في A علويًا أم سفليًا
!>          مثلثيًا.
!>          = 'U':  مثلثي علوي
!>          = 'L':  مثلثي سفلي
!>

TRANS

!>          TRANS هو CHARACTER*1
!>          يحدد العملية المراد تنفيذها:
!>          = 'N':  x := A*x
!>          = 'T':  x := A'*x
!>

DIAG

!>          DIAG هو CHARACTER*1
!>          يحدد ما إذا كانت كتل القطر موحدات أم لا.
!>          إذا افترض أن كتل القطر هي كتل موحدة،
!>          فإن A = U أو A = L، وإلا فإن A = U*D أو A = L*D.
!>          = 'U':  يُفترض أن تكون كتل القطر وحدات.
!>          = 'N':  يُفترض أن تكون كتل القطر غير موحدات.
!>

N

!>          N هو INTEGER
!>          عدد الصفوف والأعمدة للمصفوفة A.  N >= 0.
!>

NRHS

!>          NRHS هو INTEGER
!>          عدد الأطراف اليمنى، أي عدد المتجهات
!>          x المطلوب ضربها في A.  NRHS >= 0.
!>

A

!>          A هو COMPLEX*16 array, dimension (LDA,N)
!>          المصفوفة القطرية الكتلية D والمعاملات المستخدمة
!>          للحصول على العامل U أو L كما تم حسابه بواسطة ZSYTRF.
!>          مخزنة كمصفوفة مثلثية ثنائية الأبعاد.
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرائد للمصفوفة A.  LDA >= max(1,N).
!>

IPIV

!>          IPIV هو INTEGER array, dimension (N)
!>          تفاصيل التبادلات والبنية الكتلية لـ D،
!>          كما تم تحديده بواسطة ZSYTRF.
!>
!>          إذا كانت UPLO = 'U':
!>               إذا كانت IPIV(k) > 0، فتم تبديل الصفوف والأعمدة k و IPIV(k)
!>               وكان D(k,k) كتلة قطرية بحجم 1x1.
!>               (إذا كانت IPIV( k ) = k، لم يتم إجراء أي تبديل).
!>
!>               إذا كانت IPIV(k) = IPIV(k-1) < 0، فتم تبديل الصفوف
!>               والأعمدة k-1 و -IPIV(k)،
!>               وكان D(k-1:k,k-1:k) كتلة قطرية بحجم 2x2.
!>
!>          إذا كانت UPLO = 'L':
!>               إذا كانت IPIV(k) > 0، فتم تبديل الصفوف والأعمدة k و IPIV(k)
!>               وكان D(k,k) كتلة قطرية بحجم 1x1.
!>               (إذا كانت IPIV( k ) = k، لم يتم إجراء أي تبديل).
!>
!>               إذا كانت IPIV(k) = IPIV(k+1) < 0، فتم تبديل الصفوف
!>               والأعمدة k+1 و -IPIV(k)،
!>               وكان D(k:k+1,k:k+1) كتلة قطرية بحجم 2x2.
!>

B

!>          B هو COMPLEX*16 array, dimension (LDB,NRHS)
!>          عند الإدخال، تحتوي B على NRHS متجه بطول N.
!>          عند الخروج، سيتم الكتابة فوق B بمنتج A * B.
!>

LDB

!>          LDB هو INTEGER
!>          البعد الرائد للمصفوفة B.  LDB >= max(1,N).
!>

INFO

!>          INFO هو INTEGER
!>          = 0: خروج ناجح
!>          < 0: إذا كان INFO = -k، فإن المعلمة k-th كانت ذات قيمة غير قانونية
!>
