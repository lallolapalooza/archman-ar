ZLARZB تطبق عاكس كتلة أو مرافقها المتحول على مصفوفة عامة.

توثيق الوظيفة/الإجراء الفرعي

subroutine zlarzb (character side, character trans, character direct, character storev, integer m, integer n, integer k, integer l, complex*16, dimension( ldv, * ) v, integer ldv, complex*16, dimension( ldt, * ) t, integer ldt, complex*16, dimension( ldc, * ) c, integer ldc, complex*16, dimension( ldwork, * ) work, integer ldwork)

ZLARZB تطبق عاكس كتلة معقد H أو مرافقها المتحول H**H على مصفوفة عامة معقدة موزعة M-by-N C من اليسار أو اليمين.

الغرض:

!>
!> ZLARZB تطبق عاكس كتلة معقد H أو مرافقها المتحول H**H
!> على مصفوفة عامة معقدة موزعة M-by-N C من اليسار أو اليمين.
!>
!> حاليًا، يتم دعم STOREV = 'R' و DIRECT = 'B' فقط.
!>

المعلمات

SIDE 
!>          SIDE هو CHARACTER*1
!>          = 'L': تطبيق H أو H**H من اليسار
!>          = 'R': تطبيق H أو H**H من اليمين
!>

TRANS

!>          TRANS هو CHARACTER*1
!>          = 'N': تطبيق H (بدون تحويل)
!>          = 'C': تطبيق H**H (مرافق متحول)
!>

DIRECT

!>          DIRECT هو CHARACTER*1
!>          يشير إلى كيفية تشكيل H من حاصل ضرب عواكس أولية
!>          = 'F': H = H(1) H(2) . . . H(k) (أمامي، غير مدعوم بعد)
!>          = 'B': H = H(k) . . . H(2) H(1) (خلفي)
!>

STOREV

!>          STOREV هو CHARACTER*1
!>          يشير إلى كيفية تخزين المتجهات التي تحدد العواكس الأولية:
!>          = 'C': عموديًا (غير مدعوم بعد)
!>          = 'R': أفقيًا
!>

M

!>          M هو INTEGER
!>          عدد صفوف المصفوفة C.
!>

N

!>          N هو INTEGER
!>          عدد أعمدة المصفوفة C.
!>

K

!>          K هو INTEGER
!>          رتبة المصفوفة T (= عدد العواكس الأولية التي يشكل حاصل ضربها
!>          العاكس الكتلي).
!>

L

!>          L هو INTEGER
!>          عدد أعمدة المصفوفة V التي تحتوي على الجزء ذي المعنى
!>          من عواكس هاوسهولدر.
!>          إذا كانت SIDE = 'L'، فإن M >= L >= 0، وإذا كانت SIDE = 'R'،
!>          فإن N >= L >= 0.
!>

V

!>          V هي مصفوفة COMPLEX*16، البعد (LDV,NV).
!>          إذا كانت STOREV = 'C'، فإن NV = K؛ إذا كانت STOREV = 'R'،
!>          فإن NV = L.
!>

LDV

!>          LDV هو INTEGER
!>          البعد الرائد للمصفوفة V.
!>          إذا كانت STOREV = 'C'، فإن LDV >= L؛ إذا كانت STOREV = 'R'،
!>          فإن LDV >= K.
!>

T

!>          T هي مصفوفة COMPLEX*16، البعد (LDT,K)
!>          المصفوفة المثلثية K-by-K T في تمثيل العاكس الكتلي.
!>

LDT

!>          LDT هو INTEGER
!>          البعد الرائد للمصفوفة T. LDT >= K.
!>

C

!>          C هي مصفوفة COMPLEX*16، البعد (LDC,N)
!>          عند الإدخال، المصفوفة C ذات الأبعاد M-by-N.
!>          عند الخروج، يتم الكتابة فوق C بـ H*C أو H**H*C أو C*H أو
!>          C*H**H.
!>

LDC

!>          LDC هو INTEGER
!>          البعد الرائد للمصفوفة C. LDC >= max(1,M).
!>

WORK

!>          WORK هي مصفوفة COMPLEX*16، البعد (LDWORK,K)
!>

LDWORK

!>          LDWORK هو INTEGER
!>          البعد الرائد للمصفوفة WORK.
!>          إذا كانت SIDE = 'L'، فإن LDWORK >= max(1,N)؛
!>          إذا كانت SIDE = 'R'، فإن LDWORK >= max(1,M).
!>
