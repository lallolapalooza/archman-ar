subroutine zlargv (integer n, complex*16, dimension( * ) x, integer incx, complex*16, dimension( * ) y, integer incy, double precision, dimension( * ) c, integer incc)

ZLARGV تولد متجهًا من الدورانات المستوية مع جيب التمام الحقيقي وجيب المعقد.

الغرض:

!>
!> ZLARGV تولد متجهًا من الدورانات المستوية المعقدة مع جيب التمام الحقيقي،
!> يتم تحديدها بواسطة عناصر المتجهات المعقدة x و y.
!> من أجل i = 1,2,...,n
!>
!>    (        c(i)   s(i) ) ( x(i) ) = ( r(i) )
!>    ( -conjg(s(i))  c(i) ) ( y(i) ) = (   0  )
!>
!>    حيث c(i)**2 + ABS(s(i))**2 = 1
!>
!> يتم استخدام الاتفاقيات التالية (وهي نفس الاتفاقيات في ZLARTG،
!> لكنها تختلف عن روتين BLAS1 ZROTG):
!>    إذا كان y(i)=0، فإن c(i)=1 و s(i)=0.
!>    إذا كان x(i)=0، فإن c(i)=0 ويتم اختيار s(i) بحيث يكون r(i) حقيقيًا.
!>

المعلمات

N 
!>          N هو عدد صحيح (INTEGER).
!>          عدد الدورانات المستوية التي سيتم توليدها.
!>

X

!>          X هو مصفوفة COMPLEX*16، بأبعاد (1+(N-1)*INCX).
!>          عند الإدخال، المتجه x.
!>          عند الخروج، يتم الكتابة فوق x(i) بواسطة r(i)، من أجل i = 1,...,n.
!>

INCX

!>          INCX هو عدد صحيح (INTEGER).
!>          الزيادة بين عناصر X. INCX > 0.
!>

Y

!>          Y هو مصفوفة COMPLEX*16، بأبعاد (1+(N-1)*INCY).
!>          عند الإدخال، المتجه y.
!>          عند الخروج، جيوب الدورانات المستوية.
!>

INCY

!>          INCY هو عدد صحيح (INTEGER).
!>          الزيادة بين عناصر Y. INCY > 0.
!>

C

!>          C هو مصفوفة DOUBLE PRECISION، بأبعاد (1+(N-1)*INCC).
!>          جيب التمام للدورانات المستوية.
!>

INCC

!>          INCC هو عدد صحيح (INTEGER).
!>          الزيادة بين عناصر C. INCC > 0.
!>
