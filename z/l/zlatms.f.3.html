subroutine zlatms (integer m, integer n, character dist, integer, dimension( 4 ) iseed, character sym, double precision, dimension( * ) d, integer mode, double precision cond, double precision dmax, integer kl, integer ku, character pack, complex*16, dimension( lda, * ) a, integer lda, complex*16, dimension( * ) work, integer info)

ZLATMS

الغرض:

!>
!>    تُنشئ ZLATMS مصفوفات عشوائية بقيم مفردة محددة
!>    (أو هيرميتية بقيم ذاتية محددة)
!>    لاختبار برامج LAPACK.
!>
!>    تعمل ZLATMS بتطبيق التسلسل التالي من
!>    العمليات:
!>
!>      تعيين القطر إلى D، حيث قد تكون D مُدخلة أو
!>         محسوبة وفقًا لـ MODE وCOND وDMAX وSYM
!>         كما هو موضح أدناه.
!>
!>      إنشاء مصفوفة ذات بنية نطاق مناسبة، بإحدى
!>         الطريقتين:
!>
!>      الطريقة أ:
!>          إنشاء مصفوفة كثيفة M x N بضرب D من اليسار
!>              واليمين بمصفوفات وحدوية عشوائية، ثم:
!>
!>          تقليل عرض النطاق وفقًا لـ KL وKU، باستخدام
!>              تحويلات Householder.
!>
!>      الطريقة ب:
!>          تحويل مصفوفة عرض النطاق-0 (أي القطرية) إلى
!>              مصفوفة عرض النطاق-1 باستخدام دورانات Givens،
!>              مع إرجاع العناصر خارج النطاق، كما في QR؛ ثم تحويل
!>              عرض النطاق-1 إلى مصفوفة عرض النطاق-2، وهكذا. لاحظ
!>              أنه بالنسبة لعروض النطاق الصغيرة نسبيًا (بالنسبة لـ M و
!>              N) يتطلب هذا تخزينًا أقل، حيث لا يتم
!>              إنشاء مصفوفة كثيفة. أيضًا، بالنسبة للمصفوفات الهيرميتية أو المتماثلة،
!>              يتم إنشاء مثلث واحد فقط.
!>
!>      يتم اختيار الطريقة أ إذا كان عرض النطاق جزءًا كبيرًا من
!>          رتبة المصفوفة، وكان LDA على الأقل M (بحيث يمكن تخزين
!>          مصفوفة كثيفة.) يتم اختيار الطريقة ب إذا كان عرض النطاق
!>          صغيرًا (< 1/2 N للهيرميتية أو المتماثلة، < .3 N+M
!>          لغير المتماثلة)، أو كان LDA أقل من M وليس أقل من
!>          عرض النطاق.
!>
!>      حزم المصفوفة إذا رغبت في ذلك. الخيارات المحددة بواسطة PACK هي:
!>         بدون حزم
!>         تصفير النصف العلوي (إذا كانت هيرميتية)
!>         تصفير النصف السفلي (إذا كانت هيرميتية)
!>         تخزين النصف العلوي عموديًا (إذا كانت هيرميتية أو مثلثية
!>               عليا)
!>         تخزين النصف السفلي عموديًا (إذا كانت هيرميتية أو مثلثية
!>               سفلى)
!>         تخزين المثلث السفلي بتنسيق النطاق (إذا كانت هيرميتية أو
!>               مثلثية سفلى)
!>         تخزين المثلث العلوي بتنسيق النطاق (إذا كانت هيرميتية أو
!>               مثلثية عليا)
!>         تخزين المصفوفة بأكملها بتنسيق النطاق
!>      إذا تم اختيار الطريقة ب، وتم تحديد تنسيق النطاق، فسيتم
!>         إنشاء المصفوفة بتنسيق النطاق، لذا لن تكون هناك حاجة
!>         لإعادة الحزم.
!>

المعاملات

M 
!>          M هو INTEGER
!>           عدد صفوف A. غير معدّل.
!>

N

!>          N هو INTEGER
!>           عدد أعمدة A. يجب أن يساوي N العدد M إذا كانت المصفوفة
!>           متماثلة أو هيرميتية (أي إذا كان SYM ليس 'N')
!>           غير معدّل.
!>

DIST

!>          DIST هو CHARACTER*1
!>           عند الإدخال، يحدد DIST نوع التوزيع المستخدم
!>           لإنشاء القيم الذاتية/المفردة العشوائية.
!>           'U' => UNIFORM( 0, 1 )  ( 'U' لـ uniform )
!>           'S' => UNIFORM( -1, 1 ) ( 'S' لـ symmetric )
!>           'N' => NORMAL( 0, 1 )   ( 'N' لـ normal )
!>           غير معدّل.
!>

ISEED

!>          ISEED هو INTEGER array, dimension ( 4 )
!>           عند الإدخال يحدد ISEED بذرة مولّد الأرقام
!>           العشوائية. يجب أن تقع بين 0 و4095 شاملة،
!>           ويجب أن يكون ISEED(4) فرديًا. يستخدم مولّد الأرقام
!>           العشوائية تسلسلًا خطيًا متطابقًا محدودًا بأعداد
!>           صحيحة صغيرة، ولذا يجب أن ينتج أرقامًا عشوائية
!>           مستقلة عن الآلة. يتم تغيير قيم ISEED عند
!>           الخروج، ويمكن استخدامها في الاستدعاء التالي لـ ZLATMS
!>           لمتابعة نفس تسلسل الأرقام العشوائية.
!>           معدّل عند الخروج.
!>

SYM

!>          SYM هو CHARACTER*1
!>           إذا كان SYM='H'، فإن المصفوفة المُنشأة هيرميتية، مع
!>             قيم ذاتية محددة بواسطة D وCOND وMODE وDMAX؛ قد
!>             تكون موجبة أو سالبة أو صفرية.
!>           إذا كان SYM='P'، فإن المصفوفة المُنشأة هيرميتية، مع
!>             قيم ذاتية (= قيم مفردة) محددة بواسطة D وCOND
!>             وMODE وDMAX؛ لن تكون سالبة.
!>           إذا كان SYM='N'، فإن المصفوفة المُنشأة غير متماثلة، مع
!>             قيم مفردة محددة بواسطة D وCOND وMODE وDMAX؛
!>             لن تكون سالبة.
!>           إذا كان SYM='S'، فإن المصفوفة المُنشأة (مركبة) متماثلة،
!>             مع قيم مفردة محددة بواسطة D وCOND وMODE و
!>             DMAX؛ لن تكون سالبة.
!>           غير معدّل.
!>

D

!>          D هو DOUBLE PRECISION array, dimension ( MIN( M, N ) )
!>           تُستخدم هذه المصفوفة لتحديد القيم المفردة أو
!>           القيم الذاتية لـ A (انظر SYM أعلاه.) إذا كان MODE=0، فسيُفترض
!>           أن D تحتوي على القيم المفردة/الذاتية، وإلا
!>           فسيتم حسابها وفقًا لـ MODE وCOND وDMAX،
!>           ووضعها في D.
!>           معدّلة إذا كان MODE غير صفري.
!>

MODE

!>          MODE هو INTEGER
!>           عند الإدخال يصف هذا كيف يتم تحديد القيم
!>           المفردة/الذاتية:
!>           MODE = 0 يعني استخدام D كمدخل
!>           MODE = 1 يعيّن D(1)=1 وD(2:N)=1.0/COND
!>           MODE = 2 يعيّن D(1:N-1)=1 وD(N)=1.0/COND
!>           MODE = 3 يعيّن D(I)=COND**(-(I-1)/(N-1))
!>           MODE = 4 يعيّن D(i)=1 - (i-1)/(N-1)*(1 - 1/COND)
!>           MODE = 5 يعيّن D إلى أرقام عشوائية في النطاق
!>                    ( 1/COND , 1 ) بحيث تكون لوغاريتماتها
!>                    موزعة بشكل منتظم.
!>           MODE = 6 يعيّن D إلى أرقام عشوائية من نفس التوزيع
!>                    كبقية المصفوفة.
!>           MODE < 0 له نفس معنى ABS(MODE)، باستثناء أن
!>              ترتيب عناصر D معكوس.
!>           وبالتالي إذا كان MODE موجبًا، فإن D تحتوي على مدخلات تتراوح من
!>              1 إلى 1/COND، وإذا كان سالبًا، من 1/COND إلى 1،
!>           إذا كان SYM='H'، وكان MODE ليس 0 ولا 6 ولا -6، فإن
!>              عناصر D ستُضرب أيضًا بإشارة عشوائية
!>              (أي +1 أو -1.)
!>           غير معدّل.
!>

COND

!>          COND هو DOUBLE PRECISION
!>           عند الإدخال، يُستخدم كما هو موضح تحت MODE أعلاه.
!>           إذا استُخدم، يجب أن يكون >= 1. غير معدّل.
!>

DMAX

!>          DMAX هو DOUBLE PRECISION
!>           إذا كان MODE ليس -6 ولا 0 ولا 6، فإن محتويات D، كما
!>           تُحسب وفقًا لـ MODE وCOND، ستُقاس بواسطة
!>           DMAX / max(abs(D(i)))؛ وبالتالي، فإن القيمة الذاتية أو
!>           المفردة المطلقة القصوى (أي القاعدة) ستكون abs(DMAX).
!>           لاحظ أن DMAX لا يلزم أن يكون موجبًا: إذا كان DMAX سالبًا
!>           (أو صفريًا)، فسيتم قياس D بعدد سالب (أو صفر).
!>           غير معدّل.
!>

KL

!>          KL هو INTEGER
!>           يحدد هذا عرض النطاق السفلي للمصفوفة. على
!>           سبيل المثال، KL=0 يعني مثلثية عليا، KL=1 يعني
!>           هيسنبرغ عليا، وكون KL على الأقل M-1 يعني أن المصفو
ix !> has full lower bandwidth. KL must equal KU if the matrix !> is symmetric or hermitian. !> Not modified. !>

KU

!>          KU is INTEGER
!>           This specifies the upper bandwidth of the  matrix. For
!>           example, KU=0 implies lower triangular, KU=1 implies lower
!>           Hessenberg, and KU being at least N-1 means that the matrix
!>           has full upper bandwidth.  KL must equal KU if the matrix
!>           is symmetric or hermitian.
!>           Not modified.
!>

PACK

!>          PACK is CHARACTER*1
!>           This specifies packing of matrix as follows:
!>           'N' => no packing
!>           'U' => zero out all subdiagonal entries (if symmetric
!>                  or hermitian)
!>           'L' => zero out all superdiagonal entries (if symmetric
!>                  or hermitian)
!>           'C' => store the upper triangle columnwise (only if the
!>                  matrix is symmetric, hermitian, or upper triangular)
!>           'R' => store the lower triangle columnwise (only if the
!>                  matrix is symmetric, hermitian, or lower triangular)
!>           'B' => store the lower triangle in band storage scheme
!>                  (only if the matrix is symmetric, hermitian, or
!>                  lower triangular)
!>           'Q' => store the upper triangle in band storage scheme
!>                  (only if the matrix is symmetric, hermitian, or
!>                  upper triangular)
!>           'Z' => store the entire matrix in band storage scheme
!>                      (pivoting can be provided for by using this
!>                      option to store A in the trailing rows of
!>                      the allocated storage)
!>
!>           Using these options, the various LAPACK packed and banded
!>           storage schemes can be obtained:
!>           GB                    - use 'Z'
!>           PB, SB, HB, or TB     - use 'B' or 'Q'
!>           PP, SP, HB, or TP     - use 'C' or 'R'
!>
!>           If two calls to ZLATMS differ only in the PACK parameter,
!>           they will generate mathematically equivalent matrices.
!>           Not modified.
!>

A

!>          A is COMPLEX*16 array, dimension ( LDA, N )
!>           On exit A is the desired test matrix.  A is first generated
!>           in full (unpacked) form, and then packed, if so specified
!>           by PACK.  Thus, the first M elements of the first N
!>           columns will always be modified.  If PACK specifies a
!>           packed or banded storage scheme, all LDA elements of the
!>           first N columns will be modified; the elements of the
!>           array which do not correspond to elements of the generated
!>           matrix are set to zero.
!>           Modified.
!>

LDA

!>          LDA is INTEGER
!>           LDA specifies the first dimension of A as declared in the
!>           calling program.  If PACK='N', 'U', 'L', 'C', or 'R', then
!>           LDA must be at least M.  If PACK='B' or 'Q', then LDA must
!>           be at least MIN( KL, M-1) (which is equal to MIN(KU,N-1)).
!>           If PACK='Z', LDA must be large enough to hold the packed
!>           array: MIN( KU, N-1) + MIN( KL, M-1) + 1.
!>           Not modified.
!>

WORK

!>          WORK is COMPLEX*16 array, dimension ( 3*MAX( N, M ) )
!>           Workspace.
!>           Modified.
!>

INFO

!>          INFO is INTEGER
!>           Error code.  On exit, INFO will be set to one of the
!>           following values:
!>             0 => normal return
!>            -1 => M negative or unequal to N and SYM='S', 'H', or 'P'
!>            -2 => N negative
!>            -3 => DIST illegal string
!>            -5 => SYM illegal string
!>            -7 => MODE not in range -6 to 6
!>            -8 => COND less than 1.0, and MODE neither -6, 0 nor 6
!>           -10 => KL negative
!>           -11 => KU negative, or SYM is not 'N' and KU is not equal to
!>                  KL
!>           -12 => PACK illegal string, or PACK='U' or 'L', and SYM='N';
!>                  or PACK='C' or 'Q' and SYM='N' and KL is not zero;
!>                  or PACK='R' or 'B' and SYM='N' and KU is not zero;
!>                  or PACK='U', 'L', 'C', 'R', 'B', or 'Q', and M is not
!>                  N.
!>           -14 => LDA is less than M, or PACK='Z' and LDA is less than
!>                  MIN(KU,N-1) + MIN(KL,M-1) + 1.
!>            1  => Error return from DLATM1
!>            2  => Cannot scale to DMAX (max. sing. value is 0)
!>            3  => Error return from ZLAGGE, CLAGHE or CLAGSY
!>
