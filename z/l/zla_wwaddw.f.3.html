subroutine zla_wwaddw (integer n, complex*16, dimension( * ) x, complex*16, dimension( * ) y, complex*16, dimension( * ) w)

ZLA_WWADDW تضيف متجهًا إلى متجه مضاعف.

الغرض:

!>
!>    ZLA_WWADDW تضيف المتجه W إلى متجه مضاعف (X، Y).
!>
!>    يعمل هذا لجميع حسابات الأرقام العشرية السداسية والثنائية الموجودة في IBM،
!>    ولكن ليس للأرقام العشرية.
!>

المعاملات

N 
!>          N هو عدد صحيح
!>            طول المتجهات X، Y، و W.
!>

X

!>          X هو مصفوفة COMPLEX*16، الأبعاد (N)
!>            الجزء الأول من متجه التجميع المضاعف.
!>

Y

!>          Y هو مصفوفة COMPLEX*16، الأبعاد (N)
!>            الجزء الثاني من متجه التجميع المضاعف.
!>

W

!>          W هو مصفوفة COMPLEX*16، الأبعاد (N)
!>            المتجه المراد إضافته.
!>
