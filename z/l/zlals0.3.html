وثائق الوظيفة/الروتين الفرعي

subroutine zlals0 (integer icompq, integer nl, integer nr, integer sqre, integer nrhs, complex*16, dimension( ldb, * ) b, integer ldb, complex*16, dimension( ldbx, * ) bx, integer ldbx, integer, dimension( * ) perm, integer givptr, integer, dimension( ldgcol, * ) givcol, integer ldgcol, double precision, dimension( ldgnum, * ) givnum, integer ldgnum, double precision, dimension( ldgnum, * ) poles, double precision, dimension( * ) difl, double precision, dimension( ldgnum, * ) difr, double precision, dimension( * ) z, integer k, double precision c, double precision s, double precision, dimension( * ) rwork, integer info)

ZLALS0 تطبق عوامل الضرب الخلفية matrices الفردية اليمنى أو اليسرى لمصفوفة قطرية مضافة إلى صف إلى مصفوفة الطرف الأيمن B في حل مشكلة المربعات الصغرى باستخدام نهج SVD التقسيمي.

الغرض:

!>
!> ZLALS0 تطبق عوامل الضرب الخلفية matrices المتجهات الفردية اليمنى أو اليسرى لمصفوفة قطرية مضافة إلى صف إلى مصفوفة الطرف الأيمن B في حل مشكلة المربعات الصغرى باستخدام نهج SVD التقسيمي.
!>
!> بالنسبة لمصفوفة المتجهات الفردية اليسرى، تتضمن ثلاثة أنواع من matrices المتعامدة:
!>
!> (1L) دورانات جيفنز: عدد هذه الدورانات هو GIVPTR؛ الأزواج من الأعمدة/الصفوف التي تم تطبيقها عليها مخزنة في GIVCOL؛
!>      وقيم C و S لهذه الدورانات مخزنة في GIVNUM.
!>
!> (2L) التبديل. الصف (NL+1) من B سيتم نقله إلى الصف الأول، ولـ J=2:N، سيتم نقل الصف PERM(J) من B
!>      إلى الصف J.
!>
!> (3L) مصفوفة المتجهات الفردية اليسرى للمصفوفة المتبقية.
!>
!> بالنسبة لمصفوفة المتجهات الفردية اليمنى، تتضمن أربعة أنواع من matrices المتعامدة:
!>
!> (1R) مصفوفة المتجهات الفردية اليمنى للمصفوفة المتبقية.
!>
!> (2R) إذا كان SQRE = 1، دوران جيفنز إضافي لتوليد الفضاء الصفري الأيمن.
!>
!> (3R) التحويل العكسي لـ (2L).
!>
!> (4R) التحويل العكسي لـ (1L).
!>

المعلمات

ICOMPQ 
!>          ICOMPQ هو INTEGER
!>         يحدد ما إذا كان سيتم حساب المتجهات الفردية في شكل مُشَكَّل:
!>         = 0: مصفوفة المتجهات الفردية اليسرى.
!>         = 1: مصفوفة المتجهات الفردية اليمنى.
!>

NL

!>          NL هو INTEGER
!>         البعد الصفّي للكتلة العلوية. NL >= 1.
!>

NR

!>          NR هو INTEGER
!>         البعد الصفّي للكتلة السفلية. NR >= 1.
!>

SQRE

!>          SQRE هو INTEGER
!>         = 0: الكتلة السفلية هي مصفوفة مربعة بحجم NR × NR.
!>         = 1: الكتلة السفلية هي مصفوفة مستطيلة بحجم NR × (NR+1).
!>
!>         المصفوفة ثنائية القطر لها بعد صفّي N = NL + NR + 1،
!>         وبعد عمودي M = N + SQRE.
!>

NRHS

!>          NRHS هو INTEGER
!>         عدد أعمدة B و BX. يجب أن يكون NRHS على الأقل 1.
!>

B

!>          B هي مصفوفة COMPLEX*16، حجمها ( LDB, NRHS )
!>         عند الإدخال، تحتوي B على الطرف الأيمن لمشكلة المربعات الصغرى
!>         في الصفوف من 1 إلى M. عند الإخراج، تحتوي B على
!>         الحل X في الصفوف من 1 إلى N.
!>

LDB

!>          LDB هو INTEGER
!>         البعد الرئيسي لـ B. يجب أن يكون LDB على الأقل
!>         max(1,MAX( M, N ) ).
!>

BX

!>          BX هي مصفوفة COMPLEX*16، حجمها ( LDBX, NRHS )
!>

LDBX

!>          LDBX هو INTEGER
!>         البعد الرئيسي لـ BX.
!>

PERM

!>          PERM هي مصفوفة INTEGER، حجمها ( N )
!>         التبديلات (من التقسيم والفرز) التي تم تطبيقها
!>         على الكتلين.
!>

GIVPTR

!>          GIVPTR هو INTEGER
!>         عدد دورانات جيفنز التي حدثت في هذه
!>         المشكلة الفرعية.
!>

GIVCOL

!>          GIVCOL هي مصفوفة INTEGER، حجمها ( LDGCOL, 2 )
!>         يشير كل زوج من الأرقام إلى زوج من الصفوف/الأعمدة
!>         المشتركة في دوران جيفنز.
!>

LDGCOL

!>          LDGCOL هو INTEGER
!>         البعد الرئيسي لـ GIVCOL، يجب أن يكون على الأقل N.
!>

GIVNUM

!>          GIVNUM هي مصفوفة DOUBLE PRECISION، حجمها ( LDGNUM, 2 )
!>         يشير كل رقم إلى قيمة C أو S المستخدمة في
!>         دوران جيفنز المقابل.
!>

LDGNUM

!>          LDGNUM هو INTEGER
!>         البعد الرئيسي للمصفوفات DIFR و POLES و
!>         GIVNUM، يجب أن يكون على الأقل K.
!>

POLES

!>          POLES هي مصفوفة DOUBLE PRECISION، حجمها ( LDGNUM, 2 )
!>         عند الإدخال، تحتوي POLES(1:K, 1) على المتجهات الفردية الجديدة
!>         الناتجة عن حل المعادلة العلمانية،
!>         و POLES(1:K, 2) هي مصفوفة تحتوي على الأقطاب في
!>         المعادلة العلمانية.
!>

DIFL

!>          DIFL هي مصفوفة DOUBLE PRECISION، حجمها ( K ).
!>         عند الإدخال، DIFL(I) هي المسافة بين قيمة المتجه الفردي المحدثة (غير المفككة)
!>         I-th والقديمة (غير المفككة)
!>         I-th.
!>

DIFR

!>          DIFR هي مصفوفة DOUBLE PRECISION، حجمها ( LDGNUM, 2 ).
!>         عند الإدخال، تحتوي DIFR(I, 1) على المسافات بين قيمة المتجه الفردي المحدثة (غير المفككة)
!>         I-th وقيمة المتجه الفردي (غير المفككة) القديمة
!>         I+1-th. و DIFR(I, 2) هي
!>         عامل التطبيع للمتجه الفردي الأيمن I-th.
!>

Z

!>          Z هي مصفوفة DOUBLE PRECISION، حجمها ( K )
!>         تحتوي على مكونات المتجه الصفّي المحدث المُعدّل بالتقسيم.
!>

K

!>          K هو INTEGER
!>         يحتوي على بعد المصفوفة غير المفككة،
!>         هذا هو ترتيب المعادلة العلمانية ذات الصلة. 1 <= K <=N.
!>

C

!>          C هي DOUBLE PRECISION
!>         تحتوي C على قيم غير مفيدة إذا كان SQRE =0 وقيمة C لدوران
!>         جيفنز المتعلق بالفضاء الصفري الأيمن إذا كان SQRE = 1.
!>

S

!>          S هي DOUBLE PRECISION
!>         تحتوي S على قيم غير مفيدة إذا كان SQRE =0 وقيمة S لدوران
!>         جيفنز المتعلق بالفضاء الصفري الأيمن إذا كان SQRE = 1.
!>

RWORK

!>          RWORK هي مصفوفة DOUBLE PRECISION، حجمها
!>         ( K*(1+NRHS) + 2*NRHS )
!>

INFO

!>          INFO هو INTEGER
!>          = 0:  تم الخروج بنجاح.
!>          < 0:  إذا كان INFO = -i، فإن المعلمة i كانت لها قيمة غير قانونية.
!>
