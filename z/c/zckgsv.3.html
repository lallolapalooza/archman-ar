subroutine zckgsv (integer nm, integer, dimension( * ) mval, integer, dimension( * ) pval, integer, dimension( * ) nval, integer nmats, integer, dimension( 4 ) iseed, double precision thresh, integer nmax, complex*16, dimension( * ) a, complex*16, dimension( * ) af, complex*16, dimension( * ) b, complex*16, dimension( * ) bf, complex*16, dimension( * ) u, complex*16, dimension( * ) v, complex*16, dimension( * ) q, double precision, dimension( * ) alpha, double precision, dimension( * ) beta, complex*16, dimension( * ) r, integer, dimension( * ) iwork, complex*16, dimension( * ) work, double precision, dimension( * ) rwork, integer nin, integer nout, integer info)

ZCKGSV

الغرض:

!>
!> ZCKGSV تختبر ZGGSVD:
!>        الـ GSVD للمصفوفة A ذات الأبعاد M-by-N والمصفوفة B ذات الأبعاد P-by-N.
!>

المعلمات

NM 
!>          NM هو عدد صحيح
!>          عدد القيم لـ M الموجودة في المتجه MVAL.
!>

MVAL

!>          MVAL هو مصفوفة أعداد صحيحة، بأبعاد (NM)
!>          القيم لبعد الصفوف M للمصفوفة.
!>

PVAL

!>          PVAL هي مصفوفة أعداد صحيحة، بأبعاد (NP)
!>          القيم لبعد الصفوف P للمصفوفة.
!>

NVAL

!>          NVAL هي مصفوفة أعداد صحيحة، بأبعاد (NN)
!>          القيم لبعد الأعمدة N للمصفوفة.
!>

NMATS

!>          NMATS هو عدد صحيح
!>          عدد أنواع المصفوفات التي سيتم اختبارها لكل مجموعة
!>          من أبعاد المصفوفة. إذا كان NMATS >= NTYPES (الحد الأقصى
!>          لعدد أنواع المصفوفات)، فسيتم إنشاء جميع الأنواع المختلفة
!>          للاختبار. إذا كان NMATS < NTYPES، فسيتم قراءة سطر إدخال آخر
!>          للحصول على أرقام أنواع المصفوفات المراد استخدامها.
!>

ISEED

!>          ISEED هي مصفوفة أعداد صحيحة، بأبعاد (4)
!>          عند الدخول، بذرة مولد الأرقام العشوائية. يجب أن تكون عناصر
!>          المصفوفة بين 0 و 4095، وإلا فسيتم تقليلها بالباقي 4096،
!>          ويجب أن يكون ISEED(4) فرديًا.
!>          عند الخروج، البذرة التالية في تسلسل الأرقام العشوائية بعد
!>          إنشاء جميع مصفوفات الاختبار.
!>

THRESH

!>          THRESH هو رقم ذو دقة مزدوجة
!>          القيمة الحدية لنسب الاختبار. يتم تضمين النتيجة
!>          في ملف الإخراج إذا كانت RESULT >= THRESH. لطباعة كل نسبة اختبار،
!>          استخدم THRESH = 0.
!>

NMAX

!>          NMAX هو عدد صحيح
!>          القيمة القصوى المسموح بها لـ M أو N، تستخدم في تحديد
!>          أبعاد مصفوفات العمل.
!>

A

!>          A هي مصفوفة COMPLEX*16، بأبعاد (NMAX*NMAX)
!>

AF

!>          AF هي مصفوفة COMPLEX*16، بأبعاد (NMAX*NMAX)
!>

B

!>          B هي مصفوفة COMPLEX*16، بأبعاد (NMAX*NMAX)
!>

BF

!>          BF هي مصفوفة COMPLEX*16، بأبعاد (NMAX*NMAX)
!>

U

!>          U هي مصفوفة COMPLEX*16، بأبعاد (NMAX*NMAX)
!>

V

!>          V هي مصفوفة COMPLEX*16، بأبعاد (NMAX*NMAX)
!>

Q

!>          Q هي مصفوفة COMPLEX*16، بأبعاد (NMAX*NMAX)
!>

ALPHA

!>          ALPHA هي مصفوفة رقم بدقة مزدوجة، بأبعاد (NMAX)
!>

BETA

!>          BETA هي مصفوفة رقم بدقة مزدوجة، بأبعاد (NMAX)
!>

R

!>          R هي مصفوفة COMPLEX*16، بأبعاد (NMAX*NMAX)
!>

IWORK

!>          IWORK هي مصفوفة أعداد صحيحة، بأبعاد (NMAX)
!>

WORK

!>          WORK هي مصفوفة COMPLEX*16، بأبعاد (NMAX*NMAX)
!>

RWORK

!>          RWORK هي مصفوفة رقم بدقة مزدوجة، بأبعاد (NMAX)
!>

NIN

!>          NIN هو عدد صحيح
!>          رقم الوحدة للإدخال.
!>

NOUT

!>          NOUT هو عدد صحيح
!>          رقم الوحدة للإخراج.
!>

INFO

!>          INFO هو عدد صحيح
!>          = 0 :  خروج ناجح
!>          > 0 :  إذا أعاد ZLATMS رمز خطأ، يتم إرجاع القيمة المطلقة
!>                 له.
!>
