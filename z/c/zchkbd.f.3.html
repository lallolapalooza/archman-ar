!> (PARAMETER MAXTYP)، ثم !> سيتم تجاهل الأنواع NTYPES+1 حتى MAXTYP. إذا !> كان NTYPES أكبر من MAXTYP، فإن DOTYPE(MAXTYP+1) حتى !> DOTYPE(NTYPES) سيتم تجاهلها. !>

NRHS

!>          NRHS هو عدد صحيح
!>          عدد الأعمدة في المصفوفات X و Y و
!>          Z، المستخدمة في اختبار ZBDSQR. إذا كان NRHS = 0، فسيتم
!>          تجاهل العمليات على الجانب الأيمن.
!>          يجب أن يكون NRHS على الأقل 0.
!>

ISEED

!>          ISEED هو مصفوفة أعداد صحيحة، بأبعاد (4)
!>          عند الإدخال، تحدد ISEED بذرة مولد الأرقام العشوائية.
!>          يجب أن تكون عناصر المصفوفة بين 0 و 4095؛
!>          إذا لم تكن كذلك، فسيتم تقليلها حسب 4096. أيضًا، يجب أن يكون ISEED(4)
!>          فرديًا. تتغير قيم ISEED عند الخروج، ويمكن
!>          استخدامها في الاستدعاء التالي لـ ZCHKBD لمواصلة نفس تسلسل
!>          الأرقام العشوائية.
!>

THRESH

!>          THRESH هو دقة مزدوجة
!>          القيمة الحدية لنسب الاختبار. يتم تضمين النتيجة
!>          في ملف الإخراج إذا كانت RESULT >= THRESH. لطباعة
!>          كل نسبة اختبار، استخدم THRESH = 0. لاحظ أن
!>          القيمة المتوقعة لنسب الاختبار هي O(1)، لذا يجب أن يكون THRESH
!>          مضاعفًا صغيرًا بشكل معقول لـ 1، على سبيل المثال، 10 أو 100.
!>

A

!>          A هي مصفوفة COMPLEX*16، بأبعاد (LDA,NMAX)
!>          حيث NMAX هي القيمة القصوى لـ N في NVAL.
!>

LDA

!>          LDA هو عدد صحيح
!>          البعد الرائد للمصفوفة A. LDA >= max(1,MMAX)،
!>          حيث MMAX هي القيمة القصوى لـ M في MVAL.
!>

BD

!>          BD هي مصفوفة دقة مزدوجة، بأبعاد
!>                      (max(min(MVAL(j),NVAL(j))))
!>

BE

!>          BE هي مصفوفة دقة مزدوجة، بأبعاد
!>                      (max(min(MVAL(j),NVAL(j))))
!>

S1

!>          S1 هي مصفوفة دقة مزدوجة، بأبعاد
!>                      (max(min(MVAL(j),NVAL(j))))
!>

S2

!>          S2 هي مصفوفة دقة مزدوجة، بأبعاد
!>                      (max(min(MVAL(j),NVAL(j))))
!>

X

!>          X هي مصفوفة COMPLEX*16، بأبعاد (LDX,NRHS)
!>

LDX

!>          LDX هو عدد صحيح
!>          البعد الرائد للمصفوفات X و Y و Z.
!>          LDX >= max(1,MMAX).
!>

Y

!>          Y هي مصفوفة COMPLEX*16، بأبعاد (LDX,NRHS)
!>

Z

!>          Z هي مصفوفة COMPLEX*16، بأبعاد (LDX,NRHS)
!>

Q

!>          Q هي مصفوفة COMPLEX*16، بأبعاد (LDQ,MMAX)
!>

LDQ

!>          LDQ هو عدد صحيح
!>          البعد الرائد للمصفوفة Q. LDQ >= max(1,MMAX).
!>

PT

!>          PT هي مصفوفة COMPLEX*16، بأبعاد (LDPT,NMAX)
!>

LDPT

!>          LDPT هو عدد صحيح
!>          البعد الرائد للمصفوفات PT و U و V.
!>          LDPT >= max(1, max(min(MVAL(j),NVAL(j)))).
!>

U

!>          U هي مصفوفة COMPLEX*16، بأبعاد
!>                      (LDPT,max(min(MVAL(j),NVAL(j))))
!>

VT

!>          VT هي مصفوفة COMPLEX*16، بأبعاد
!>                      (LDPT,max(min(MVAL(j),NVAL(j))))
!>

WORK

!>          WORK هي مصفوفة COMPLEX*16، بأبعاد (LWORK)
!>

LWORK

!>          LWORK هو عدد صحيح
!>          عدد عناصر WORK. يجب أن يكون على الأقل
!>          3(M+N) و M(M + max(M,N,k) + 1) + N*min(M,N) لجميع
!>          الأزواج (M,N)=(MM(j),NN(j))
!>

RWORK

!>          RWORK هي مصفوفة دقة مزدوجة، بأبعاد
!>                      (5*max(min(M,N)))
!>

NOUT

!>          NOUT هو عدد صحيح
!>          رقم وحدة FORTRAN لطباعة رسائل الخطأ
!>          (على سبيل المثال، إذا أعادت روتين IINFO قيمة غير صفرية).
!>

INFO

!>          INFO هو عدد صحيح
!>          إذا كان 0، فقد تم تشغيل كل شيء بنجاح.
!>           -1: NSIZES < 0
!>           -2: بعض MM(j) < 0
!>           -3: بعض NN(j) < 0
!>           -4: NTYPES < 0
!>           -6: NRHS < 0
!>           -8: THRESH < 0
!>          -11: LDA < 1 أو LDA < MMAX، حيث MMAX هو max( MM(j) ).
!>          -17: LDB < 1 أو LDB < MMAX.
!>          -21: LDQ < 1 أو LDQ < MMAX.
!>          -23: LDP < 1 أو LDP < MNMAX.
!>          -27: LWORK صغير جدًا.
!>          إذا أعادت ZLATMR أو CLATMS أو ZGEBRD أو ZUNGBR أو ZBDSQR
!>              رمز خطأ، فإن
!>              القيمة المطلقة له يتم إرجاعها.
!>
!>-----------------------------------------------------------------------
!>
!>     بعض المتغيرات والمعاملات المحلية:
!>     ---- ----- --------- --- ----------
!>
!>     ZERO, ONE       0 و 1 حقيقيان.
!>     MAXTYP          عدد الأنواع المعرفة.
!>     NTEST           عدد الاختبارات التي تم إجراؤها، أو التي يمكن
!>                     إجراؤها حتى الآن، للمصفوفة الحالية.
!>     MMAX            أكبر قيمة في NN.
!>     NMAX            أكبر قيمة في NN.
!>     MNMIN           min(MM(j), NN(j)) (بعد المصفوفة ثنائية القطر
!>                     .)
!>     MNMAX           أكبر قيمة لـ MNMIN لـ j=1,...,NSIZES.
!>     NFAIL           عدد الاختبارات التي تجاوزت THRESH
!>     COND, IMODE     قيم سيتم تمريرها إلى مولدات المصفوفات.
!>     ANORM           معيار A؛ يتم تمريره إلى مولدات المصفوفات.
!>
!>     OVFL, UNFL      حدود الفيضان والتدفق السفلي.
!>     RTOVFL, RTUNFL  الجذور التربيعية للقيمتين السابقتين.
!>     ULP, ULPINV     أدق دقة نسبية وعكسها.
!>
!>             تقوم المصفوفات الأربعة التالية بفك تشفير JTYPE:
!>     KTYPE(j)        النوع العام (1-10) للنوع.
!>     KMODE(j)        قيمة MODE التي سيتم تمريرها إلى مولد المصفوفات
!>                     للنوع.
!>     KMAGN(j)        ترتيب الحجم ( O(1)،
!>                     O(overflow^(1/2) )، O(underflow^(1/2) )
!>
