المعاملات

DOTYPE 
!>          DOTYPE هو مصفوفة LOGICAL، البعد (NTYPES)
!>          أنواع المصفوفات التي سيتم استخدامها للاختبار. المصفوفات من النوع j
!>          (لـ 1 <= j <= NTYPES) تستخدم للاختبار إذا كان DOTYPE(j) =
!>          .TRUE.؛ إذا كان DOTYPE(j) = .FALSE.، فلن يتم استخدام النوع j.
!>

NN

!>          NN هو INTEGER
!>          عدد القيم لـ N الموجودة في المتجه NVAL.
!>

NVAL

!>          NVAL هو مصفوفة INTEGER، البعد (NN)
!>          قيم البعد العمودي للمصفوفة N.
!>

NNB

!>          NNB هو INTEGER
!>          عدد القيم لـ NB الموجودة في المتجه NBVAL.
!>

NBVAL

!>          NBVAL هو مصفوفة INTEGER، البعد (NNB)
!>          قيم حجم الكتلة NB.
!>

NNS

!>          NNS هو INTEGER
!>          عدد القيم لـ NRHS الموجودة في المتجه NSVAL.
!>

NSVAL

!>          NSVAL هو مصفوفة INTEGER، البعد (NNS)
!>          قيم عدد الأطراف اليمنى NRHS.
!>

THRESH

!>          THRESH هو DOUBLE PRECISION
!>          قيمة الحد لنسب الاختبار. يتم تضمين نتيجة
!>          في ملف الإخراج إذا كانت RESULT >= THRESH. لطباعة كل نسبة اختبار، استخدم THRESH = 0.
!>

TSTERR

!>          TSTERR هو LOGICAL
!>          العلامة التي تشير إلى ما إذا كان سيتم اختبار مخارج الخطأ.
!>

NMAX

!>          NMAX هو INTEGER
!>          البعد الرائد لمصفوفات العمل.
!>          NMAX >= القيمة القصوى لـ N في NVAL.
!>

A

!>          A هي مصفوفة COMPLEX*16، البعد (NMAX*NMAX)
!>

AINV

!>          AINV هي مصفوفة COMPLEX*16، البعد (NMAX*NMAX)
!>

B

!>          B هي مصفوفة COMPLEX*16، البعد (NMAX*NSMAX)
!>          حيث NSMAX هو أكبر إدخال في NSVAL.
!>

X

!>          X هي مصفوفة COMPLEX*16، البعد (NMAX*NSMAX)
!>

XACT

!>          XACT هي مصفوفة COMPLEX*16، البعد (NMAX*NSMAX)
!>

WORK

!>          WORK هي مصفوفة COMPLEX*16، البعد
!>                      (NMAX*max(3,NSMAX))
!>

RWORK

!>          RWORK هي مصفوفة DOUBLE PRECISION، البعد
!>                      (max(NMAX,2*NSMAX))
!>

NOUT

!>          NOUT هو INTEGER
!>          رقم الوحدة للإخراج.
!>
