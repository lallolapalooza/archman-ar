subroutine zcsdts (integer m, integer p, integer q, complex*16, dimension( ldx, * ) x, complex*16, dimension( ldx, * ) xf, integer ldx, complex*16, dimension( ldu1, * ) u1, integer ldu1, complex*16, dimension( ldu2, * ) u2, integer ldu2, complex*16, dimension( ldv1t, * ) v1t, integer ldv1t, complex*16, dimension( ldv2t, * ) v2t, integer ldv2t, double precision, dimension( * ) theta, integer, dimension( * ) iwork, complex*16, dimension( lwork ) work, integer lwork, double precision, dimension( * ) rwork, double precision, dimension( 15 ) result)

ZCSDTS

الغرض:

!>
!> ZCSDTS تختبر ZUNCSD، والتي، بالنظر إلى مصفوفة X موحدة مجزأة بحجم M-by-M
!>             Q  M-Q
!>      X = [ X11 X12 ] P   ,
!>          [ X21 X22 ] M-P
!>
!> تحسب CSD
!>
!>      [ U1    ]**T * [ X11 X12 ] * [ V1    ]
!>      [    U2 ]      [ X21 X22 ]   [    V2 ]
!>
!>                            [  I  0  0 |  0  0  0 ]
!>                            [  0  C  0 |  0 -S  0 ]
!>                            [  0  0  0 |  0  0 -I ]
!>                          = [---------------------] = [ D11 D12 ] .
!>                            [  0  0  0 |  I  0  0 ]   [ D21 D22 ]
!>                            [  0  S  0 |  0  C  0 ]
!>                            [  0  0  I |  0  0  0 ]
!>
!> وأيضًا SORCSD2BY1، والتي، بالنظر إلى
!>         Q
!>      [ X11 ] P   ,
!>      [ X21 ] M-P
!>
!> تحسب CSD ثنائية الأبعاد 1
!>
!>                                    [  I  0  0 ]
!>                                    [  0  C  0 ]
!>                                    [  0  0  0 ]
!>      [ U1    ]**T * [ X11 ] * V1 = [----------] = [ D11 ] ,
!>      [    U2 ]      [ X21 ]        [  0  0  0 ]   [ D21 ]
!>                                    [  0  S  0 ]
!>                                    [  0  0  I ]
!>

المعلمات

M 
!>          M هو عدد صحيح
!>          عدد الصفوف للمصفوفة X.  M >= 0.
!>

P

!>          P هو عدد صحيح
!>          عدد الصفوف للمصفوفة X11.  P >= 0.
!>

Q

!>          Q هو عدد صحيح
!>          عدد الأعمدة للمصفوفة X11.  Q >= 0.
!>

X

!>          X هي مصفوفة COMPLEX*16، ذات بعد (LDX,M)
!>          المصفوفة X بحجم M-by-M.
!>

XF

!>          XF هي مصفوفة COMPLEX*16، ذات بعد (LDX,M)
!>          تفاصيل CSD لـ X، كما تم إرجاعها بواسطة ZUNCSD؛
!>          راجع ZUNCSD لمزيد من التفاصيل.
!>

LDX

!>          LDX هو عدد صحيح
!>          البعد الرائد للمصفوفتين X و XF.
!>          LDX >= max( 1,M ).
!>

U1

!>          U1 هي مصفوفة COMPLEX*16، ذات بعد (LDU1,P)
!>          المصفوفة الموحدة U1 بحجم P-by-P.
!>

LDU1

!>          LDU1 هو عدد صحيح
!>          البعد الرائد للمصفوفة U1. LDU >= max(1,P).
!>

U2

!>          U2 هي مصفوفة COMPLEX*16، ذات بعد (LDU2,M-P)
!>          المصفوفة الموحدة U2 بحجم (M-P)-by-(M-P).
!>

LDU2

!>          LDU2 هو عدد صحيح
!>          البعد الرائد للمصفوفة U2. LDU >= max(1,M-P).
!>

V1T

!>          V1T هي مصفوفة COMPLEX*16، ذات بعد (LDV1T,Q)
!>          المصفوفة الموحدة V1T بحجم Q-by-Q.
!>

LDV1T

!>          LDV1T هو عدد صحيح
!>          البعد الرائد للمصفوفة V1T. LDV1T >=
!>          max(1,Q).
!>

V2T

!>          V2T هي مصفوفة COMPLEX*16، ذات بعد (LDV2T,M-Q)
!>          المصفوفة الموحدة V2T بحجم (M-Q)-by-(M-Q).
!>

LDV2T

!>          LDV2T هو عدد صحيح
!>          البعد الرائد للمصفوفة V2T. LDV2T >=
!>          max(1,M-Q).
!>

THETA

!>          THETA هي مصفوفة DOUBLE PRECISION، ذات بعد MIN(P,M-P,Q,M-Q)
!>          قيم CS لـ X؛ يتم بناء المصفوفات القطرية بشكل أساسي C و
!>          S من THETA؛ راجع الروتين الفرعي ZUNCSD للحصول على
!>          التفاصيل.
!>

IWORK

!>          IWORK هي مصفوفة INTEGER، ذات بعد (M)
!>

WORK

!>          WORK هي مصفوفة COMPLEX*16، ذات بعد (LWORK)
!>

LWORK

!>          LWORK هو عدد صحيح
!>          بعد المصفوفة WORK
!>

RWORK

!>          RWORK هي مصفوفة DOUBLE PRECISION
!>

RESULT

!>          RESULT هي مصفوفة DOUBLE PRECISION، ذات بعد (15)
!>          نسب الاختبار:
!>          أولاً، CSD ثنائية الأبعاد 2:
!>          RESULT(1) = norm( U1'*X11*V1 - D11 ) / ( MAX(1,P,Q)*EPS2 )
!>          RESULT(2) = norm( U1'*X12*V2 - D12 ) / ( MAX(1,P,M-Q)*EPS2 )
!>          RESULT(3) = norm( U2'*X21*V1 - D21 ) / ( MAX(1,M-P,Q)*EPS2 )
!>          RESULT(4) = norm( U2'*X22*V2 - D22 ) / ( MAX(1,M-P,M-Q)*EPS2 )
!>          RESULT(5) = norm( I - U1'*U1 ) / ( MAX(1,P)*ULP )
!>          RESULT(6) = norm( I - U2'*U2 ) / ( MAX(1,M-P)*ULP )
!>          RESULT(7) = norm( I - V1T'*V1T ) / ( MAX(1,Q)*ULP )
!>          RESULT(8) = norm( I - V2T'*V2T ) / ( MAX(1,M-Q)*ULP )
!>          RESULT(9) = 0        إذا كانت THETA بترتيب متزايد و
!>                               جميع الزوايا في [0,pi/2]
!>                    = ULPINV   بخلاف ذلك.
!>          ثم، CSD ثنائية الأبعاد 1:
!>          RESULT(10) = norm( U1'*X11*V1 - D11 ) / ( MAX(1,P,Q)*EPS2 )
!>          RESULT(11) = norm( U2'*X21*V1 - D21 ) / ( MAX(1,M-P,Q)*EPS2 )
!>          RESULT(12) = norm( I - U1'*U1 ) / ( MAX(1,P)*ULP )
!>          RESULT(13) = norm( I - U2'*U2 ) / ( MAX(1,M-P)*ULP )
!>          RESULT(14) = norm( I - V1T'*V1T ) / ( MAX(1,Q)*ULP )
!>          RESULT(15) = 0        إذا كانت THETA بترتيب متزايد و
!>                               جميع الزوايا في [0,pi/2]
!>                     = ULPINV   بخلاف ذلك.
!>          ( EPS2 = MAX( norm( I - X'*X ) / M, ULP ). )
!>
