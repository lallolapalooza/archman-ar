وثائق الوظيفة/الإجراء الفرعي

subroutine zchkbb (integer nsizes, integer, dimension( * ) mval, integer, dimension( * ) nval, integer nwdths, integer, dimension( * ) kk, integer ntypes, logical, dimension( * ) dotype, integer nrhs, integer, dimension( 4 ) iseed, double precision thresh, integer nounit, complex*16, dimension( lda, * ) a, integer lda, complex*16, dimension( ldab, * ) ab, integer ldab, double precision, dimension( * ) bd, double precision, dimension( * ) be, complex*16, dimension( ldq, * ) q, integer ldq, complex*16, dimension( ldp, * ) p, integer ldp, complex*16, dimension( ldc, * ) c, integer ldc, complex*16, dimension( ldc, * ) cc, complex*16, dimension( * ) work, integer lwork, double precision, dimension( * ) rwork, double precision, dimension( * ) result, integer info)

ZCHKBB

Purpose:

!>
!> ZCHKBB تختبر اختزال مصفوفة نطاق مستطيلة عامة معقدة إلى شكل ثنائي قطري حقيقي.
!>
!> ZGBBRD يقوم بتحليل مصفوفة نطاق عامة A إلى Q B P* ، حيث * تعني المترافقة المستعرضة،
!> B ثنائية القطر علوية، و Q و P وحدويتان؛ يمكن لـ ZGBBRD أيضًا الكتابة فوق مصفوفة
!> معطاة C بـ Q* C .
!>
!> لكل زوج من أبعاد المصفوفة (M,N) ولكل نوع مصفوفة محدد، يتم إنشاء مصفوفة A بحجم M x N
!> ومصفوفة C بحجم M x NRHS. أبعاد المشكلة هي كما يلي
!>    A:          M x N
!>    Q:          M x M
!>    P:          N x N
!>    B:          min(M,N) x min(M,N)
!>    C:          M x NRHS
!>
!> لكل مصفوفة تم إنشاؤها، يتم إجراء 4 اختبارات:
!>
!> (1)   | A - Q B PT | / ( |A| max(M,N) ulp )، PT = P'
!>
!> (2)   | I - Q' Q | / ( M ulp )
!>
!> (3)   | I - PT PT' | / ( N ulp )
!>
!> (4)   | Y - Q' C | / ( |Y| max(M,NRHS) ulp )، حيث Y = Q' C.
!>
!> يتم تحديد أنواع المصفوفات بواسطة مصفوفة منطقية DOTYPE( 1:NTYPES )؛
!> إذا كان DOTYPE(j) هو .TRUE.، فسيتم إنشاء مصفوفة من النوع j.
!> حاليًا، قائمة الأنواع الممكنة هي:
!>
!> أنواع المصفوفات الممكنة هي
!>
!> (1)  المصفوفة الصفرية.
!> (2)  المصفوفة الوحدوية.
!>
!> (3)  مصفوفة قطرية ذات مدخلات متباعدة بالتساوي
!>      1, ..., ULP  وعلامات عشوائية.
!>      (ULP = (أول رقم أكبر من 1) - 1)
!> (4)  مصفوفة قطرية ذات مدخلات متباعدة هندسيًا
!>      1, ..., ULP  وعلامات عشوائية.
!> (5)  مصفوفة قطرية ذات مدخلات 1، ULP، ..., ULP
!>      وعلامات عشوائية.
!>
!> (6)  نفس (3)، ولكن مضروبة في SQRT( عتبة التدفق الزائد )
!> (7)  نفس (3)، ولكن مضروبة في SQRT( عتبة التدفق الناقص )
!>
!> (8)  مصفوفة من الشكل U D V، حيث U و V متعامدتان و
!>      D يحتوي على مدخلات متباعدة بالتساوي 1، ..., ULP مع علامات عشوائية
!>      على القطر.
!>
!> (9)  مصفوفة من الشكل U D V، حيث U و V متعامدتان و
!>      D يحتوي على مدخلات متباعدة هندسيًا 1، ..., ULP مع علامات عشوائية
!>      على القطر.
!>
!> (10) مصفوفة من الشكل U D V، حيث U و V متعامدتان و
!>      D يحتوي على مدخلات 1، ULP، ..., ULP مع علامات عشوائية
!>      على القطر.
!>
!> (11) نفس (8)، ولكن مضروبة في SQRT( عتبة التدفق الزائد )
!> (12) نفس (8)، ولكن مضروبة في SQRT( عتبة التدفق الناقص )
!>
!> (13) مصفوفة مستطيلة بمدخلات عشوائية مختارة من (-1,1).
!> (14) نفس (13)، ولكن مضروبة في SQRT( عتبة التدفق الزائد )
!> (15) نفس (13)، ولكن مضروبة في SQRT( عتبة التدفق الناقص )
!>

Parameters

NSIZES 
!>          NSIZES هو عدد صحيح (INTEGER)
!>          عدد القيم لـ M و N الموجودة في المتجهين
!>          MVAL و NVAL. تستخدم أحجام المصفوفة في أزواج (M,N).
!>          إذا كان NSIZES صفرًا، فإن ZCHKBB لا تفعل شيئًا. يجب أن يكون NSIZES
!>          على الأقل صفرًا.
!>

MVAL

!>          MVAL هو مصفوفة أعداد صحيحة (INTEGER array)، بحجم (NSIZES)
!>          قيم البعد الصف للمصفوفة M.
!>

NVAL

!>          NVAL هو مصفوفة أعداد صحيحة (INTEGER array)، بحجم (NSIZES)
!>          قيم البعد العمودي للمصفوفة N.
!>

NWDTHS

!>          NWDTHS هو عدد صحيح (INTEGER)
!>          عدد عرض النطاقات المراد استخدامه. إذا كان صفرًا،
!>          فإن ZCHKBB لا تفعل شيئًا. يجب أن يكون على الأقل صفرًا.
!>

KK

!>          KK هو مصفوفة أعداد صحيحة (INTEGER array)، بحجم (NWDTHS)
!>          مصفوفة تحتوي على عرض النطاقات المراد استخدامه للمصفوفات
!>          النطاقية. يجب أن تكون القيم على الأقل صفرًا.
!>

NTYPES

!>          NTYPES هو عدد صحيح (INTEGER)
!>          عدد العناصر في DOTYPE. إذا كان صفرًا، فإن ZCHKBB
!>          لا تفعل شيئًا. يجب أن يكون على الأقل صفرًا. إذا كان MAXTYP+1
!>          و NSIZES يساوي 1، فسيتم تعريف نوع إضافي، MAXTYP+1،
!>          وهو استخدام أي مصفوفة موجودة في A. هذا
!>          مفيد فقط إذا كان DOTYPE(1:MAXTYP) يساوي .FALSE. و
!>          DOTYPE(MAXTYP+1) يساوي .TRUE. .
!>

DOTYPE

!>          DOTYPE هو مصفوفة منطقية (LOGICAL array)، بحجم (NTYPES)
!>          إذا كان DOTYPE(j) يساوي .TRUE. ، فلكل حجم في NN
!>          سيتم إنشاء مصفوفة بهذا الحجم ومن النوع j.
!>          إذا كان NTYPES أصغر من الحد الأقصى لعدد الأنواع
!>          المعرفة (PARAMETER MAXTYP)، فلن يتم إنشاء الأنواع NTYPES+1 إلى
!>          MAXTYP. إذا كان NTYPES أكبر
!>          من MAXTYP، فسيتم تجاهل DOTYPE(MAXTYP+1) إلى DOTYPE(NTYPES).
!>

NRHS

!>          NRHS هو عدد صحيح (INTEGER)
!>          عدد الأعمدة في المصفوفة اليمنى C.
!>          إذا كان NRHS = 0، فلن يتم اختبار العمليات على الجانب الأيمن.
!>          يجب أن يكون NRHS على الأقل 0.
!>

ISEED

!>          ISEED هو مصفوفة أعداد صحيحة (INTEGER array)، بحجم (4)
!>          عند الإدخال، يحدد ISEED بذرة مولد الأرقام العشوائية.
!>          يجب أن تكون عناصر المصفوفة بين 0 و 4095؛
!>          إذا لم تكن كذلك، فسيتم تقليلها بمعامل 4096. أيضًا، يجب أن يكون ISEED(4)
!>          فرديًا. يستخدم مولد الأرقام العشوائية تسلسل توافقي خطي
!>          محدود بأعداد صحيحة صغيرة، وبالتالي
!>          ينبغي أن ينتج أرقامًا عشوائية مستقلة عن الآلة.
!>          تتغير قيم ISEED عند الخروج، ويمكن استخدامها في
!>          الاستدعاء التالي لـ ZCHKBB لمواصلة نفس تسلسل الأرقام العشوائية.
!>

THRESH

!>          THRESH هو دقة مزدوجة (DOUBLE PRECISION)
!>          سيتم احتساب الاختبار كـ "خطأ" إذا كانت نسبة الخطأ،
!>          المحسوبة كما هو موضح أعلاه، تتجاوز THRESH. لاحظ أن الخطأ
!>          يتم قياسه ليكون O(1)، لذا يجب أن يكون THRESH مضاعفًا صغيرًا بشكل معقول
!>          من 1، على سبيل المثال، 10 أو 100. على وجه الخصوص،
!>          يجب ألا يعتمد على الدقة (فردية مقابل مزدوجة)
!>          أو حجم المصفوفة. يجب أن يكون على الأقل صفرًا.
!>

NOUNIT

!>          NOUNIT هو عدد صحيح (INTEGER)
!>          رقم وحدة FORTRAN لطباعة رسائل الخطأ
!>          (على سبيل المثال، إذا أعادت روتينية IINFO لا تساوي 0.)
!>
