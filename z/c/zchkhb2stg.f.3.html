subroutine zchkhb2stg (integer nsizes, integer, dimension( * ) nn, integer nwdths, integer, dimension( * ) kk, integer ntypes, logical, dimension( * ) dotype, integer, dimension( 4 ) iseed, double precision thresh, integer nounit, complex*16, dimension( lda, * ) a, integer lda, double precision, dimension( * ) sd, double precision, dimension( * ) se, double precision, dimension( * ) d1, double precision, dimension( * ) d2, double precision, dimension( * ) d3, complex*16, dimension( ldu, * ) u, integer ldu, complex*16, dimension( * ) work, integer lwork, double precision, dimension( * ) rwork, double precision, dimension( * ) result, integer info)

ZCHKHB2STG

الغرض:

!>
!> ZCHKHB2STG يختبر اختزال مصفوفة نطاق هيرميتية إلى شكل ثلاثي الأقطار،
!> المستخدم مع مشكلة القيم الذاتية الهيرميتية.
!>
!> ZHBTRD يحلل مصفوفة نطاق هيرميتية A إلى  U S U* ، حيث * تعني
!> منقولة مترافقة، S هي ثلاثية الأقطار متناظرة، و U هي وحدوية.
!> يمكن لـ ZHBTRD استخدام إما المثلث السفلي أو العلوي فقط
!> لـ A؛ ZCHKHB2STG يتحقق من كلا الحالتين.
!>
!> ZHETRD_HB2ST يحلل مصفوفة نطاق هيرميتية A إلى  U S U* ،
!> حيث * تعني منقولة مترافقة، S هي ثلاثية الأقطار متناظرة، و U هي
!> وحدوية. يمكن لـ ZHETRD_HB2ST استخدام إما المثلث السفلي أو العلوي فقط
!> لـ A؛ ZCHKHB2STG يتحقق من كلا الحالتين.
!>
!> DSTEQR يحلل S إلى  Z D1 Z'.
!> D1 هي مصفوفة القيم الذاتية المحسوبة عندما لا يتم حساب Z
!> ومن S الناتجة من DSBTRD  (تستخدم كمرجع لـ DSYTRD_SB2ST)
!> D2 هي مصفوفة القيم الذاتية المحسوبة عندما لا يتم حساب Z
!> ومن S الناتجة من DSYTRD_SB2ST .
!> D3 هي مصفوفة القيم الذاتية المحسوبة عندما لا يتم حساب Z
!> ومن S الناتجة من DSYTRD_SB2ST .
!>
!> عندما يتم استدعاء ZCHKHB2STG، يتم تحديد عدد من أحجام المصفوفات  (NN)،
!> عدد من عرض النطاقات (KK) ، وعدد من أنواع المصفوفات (DOTYPE).
!> لكل حجم (NN)، كل عرض نطاق (KK) أقل من أو يساوي ، وكل نوع من المصفوفات،
!> سيتم إنشاء مصفوفة واحدة واستخدامها لاختبار روتين الاختزال الهيرميتي ذي النطاق.
!> لكل مصفوفة، سيتم إجراء عدد من الاختبارات:
!>
!> (1)     | A - V S V* | / ( |A| n ulp )  محسوبة بواسطة ZHBTRD مع
!>                                         UPLO='U'
!>
!> (2)     | I - UU* | / ( n ulp )
!>
!> (3)     | A - V S V* | / ( |A| n ulp )  محسوبة بواسطة ZHBTRD مع
!>                                         UPLO='L'
!>
!> (4)     | I - UU* | / ( n ulp )
!>
!> (5)     | D1 - D2 | / ( |D1| ulp )      حيث يتم حساب D1 بواسطة
!>                                         DSBTRD مع UPLO='U' و
!>                                         D2 يتم حسابها بواسطة
!>                                         ZHETRD_HB2ST مع UPLO='U'
!>
!> (6)     | D1 - D3 | / ( |D1| ulp )      حيث يتم حساب D1 بواسطة
!>                                         DSBTRD مع UPLO='U' و
!>                                         D3 يتم حسابها بواسطة
!>                                         ZHETRD_HB2ST مع UPLO='L'
!>
!> تحدد أحجام المصفوفات بواسطة مصفوفة NN(1:NSIZES)؛ يحدد عنصر NN(j) حجمًا واحدًا.
!> تحدد أنواع المصفوفات بواسطة مصفوفة منطقية DOTYPE( 1:NTYPES )؛
!> إذا كانت DOTYPE(j) هي .TRUE.، فسيتم إنشاء مصفوفة من النوع j.
!> حاليًا، قائمة الأنواع الممكنة هي:
!>
!> (1)  المصفوفة الصفرية.
!> (2)  المصفوفة الوحدوية.
!>
!> (3)  مصفوفة قطرية بمدخلات متباعدة بالتساوي
!>      1, ..., ULP  وعلامات عشوائية.
!>      (ULP = (أول عدد أكبر من 1) - 1 )
!> (4)  مصفوفة قطرية بمدخلات متباعدة هندسيًا
!>      1, ..., ULP  وعلامات عشوائية.
!> (5)  مصفوفة قطرية بمدخلات  1, ULP, ..., ULP
!>      وعلامات عشوائية.
!>
!> (6)  نفس (4)، ولكن مضروبة في الجذر التربيعي ( عتبة الفيضان )
!> (7)  نفس (4)، ولكن مضروبة في الجذر التربيعي ( عتبة التدفق السفلي )
!>
!> (8)  مصفوفة من الشكل  U* D U، حيث U وحدوية و
!>      D له مدخلات متباعدة بالتساوي 1, ..., ULP مع علامات عشوائية
!>      على القطر.
!>
!> (9)  مصفوفة من الشكل  U* D U، حيث U وحدوية و
!>      D له مدخلات متباعدة هندسيًا 1, ..., ULP مع علامات عشوائية
!>      على القطر.
!>
!> (10) مصفوفة من الشكل  U* D U، حيث U وحدوية و
!>      D له مدخلات  1, ULP,..., ULP مع علامات عشوائية
!>      على القطر.
!>
!> (11) نفس (8)، ولكن مضروبة في الجذر التربيعي ( عتبة الفيضان )
!> (12) نفس (8)، ولكن مضروبة في الجذر التربيعي ( عتبة التدفق السفلي )
!>
!> (13) مصفوفة هيرميتية بمدخلات عشوائية مختارة من (-1,1).
!> (14) نفس (13)، ولكن مضروبة في الجذر التربيعي ( عتبة الفيضان )
!> (15) نفس (13)، ولكن مضروبة في الجذر التربيعي ( عتبة التدفق السفلي )
!>

المعلمات

NSIZES 
!>          NSIZES هو عدد صحيح
!>          عدد أحجام المصفوفات المراد استخدامها. إذا كان صفرًا،
!>          فإن ZCHKHB2STG لا يفعل شيئًا. يجب أن يكون أكبر من أو يساوي صفرًا.
!>

NN

!>          NN هو مصفوفة أعداد صحيحة، البعد (NSIZES)
!>          مصفوفة تحتوي على الأحجام المراد استخدامها للمصفوفات.
!>          سيتم تخطي القيم الصفرية. يجب أن تكون القيم أكبر من أو تساوي
!>          صفرًا.
!>

NWDTHS

!>          NWDTHS هو عدد صحيح
!>          عدد عرض النطاقات المراد استخدامها. إذا كان صفرًا،
!>          فإن ZCHKHB2STG لا يفعل شيئًا. يجب أن يكون أكبر من أو يساوي صفرًا.
!>

KK

!>          KK هو مصفوفة أعداد صحيحة، البعد (NWDTHS)
!>          مصفوفة تحتوي على عرض النطاقات المراد استخدامها لمصفوفات النطاق.
!>          يجب أن تكون القيم أكبر من أو تساوي صفرًا.
!>

NTYPES

!>          NTYPES هو عدد صحيح
!>          عدد العناصر في DOTYPE. إذا كان صفرًا، فإن ZCHKHB2STG
!>          لا يفعل شيئًا. يجب أن يكون أكبر من أو يساوي صفرًا. إذا كان MAXTYP+1
!>          وكان NSIZES هو 1، فسيتم تعريف نوع إضافي، MAXTYP+1،
!>          وهو استخدام أي مصفوفة موجودة في A. هذا
!>          مفيد فقط إذا كان DOTYPE(1:MAXTYP) هو .FALSE. و
!>          DOTYPE(MAXTYP+1) هو .TRUE. .
!>

DOTYPE

!>          DOTYPE هي مصفوفة منطقية، البعد (NTYPES)
!>          إذا كانت DOTYPE(j) هي .TRUE.، فسيتم إنشاء مصفوفة من النوع j
!>          لكل حجم في NN.
!>          إذا كانت NTYPES أصغر من العدد الأقصى للأنواع
!>          المعرفة (PARAMETER MAXTYP)، فلن يتم إنشاء الأنواع NTYPES+1 حتى
!>          MAXTYP. إذا كانت NTYPES أكبر
!>          من MAXTYP، فسيتم تجاهل DOTYPE(MAXTYP+1) حتى DOTYPE(NTYPES).
!>

ISEED

!>          ISEED هي مصفوفة أعداد صحيحة، البعد (4)
!>          عند الدخول، يحدد ISEED البذرة لمولد الأرقام العشوائية. يجب أن تكون
!>          عناصر المصفوفة بين 0 و 4095؛
!>          إذا لم تكن كذلك، فسيتم تقليلها حسب 4096. أيضًا، يجب أن يكون ISEED(4)
!>          فرديًا. يستخدم مولد الأرقام العشوائية تسلسلًا خطيًا توافقيًا
!>          محدودًا بأعداد صحيحة صغيرة، وبالتالي
!>          يجب أن ينتج أرقامًا عشوائية مستقلة عن الجهاز. تتغير
!>          قيم ISEED عند الخروج، ويمكن استخدامها في
!>          الاستدعاء التالي لـ ZCHKHB2STG لمواصلة نفس تسلسل الأرقام العشوائية.
!>

THRESH

!>          THRESH هو DOUBLE PRECISION
!>          سيتم احتساب الاختبار إذا كانت القيمة المحسوبة
!>          كما هو موضح أعلاه، تتجاوز THRESH. لاحظ أن الخطأ
!>          يتم قياسه ليكون O(1)، لذا يجب أن يكون THRESH مضاعفاً
!>          معقولاً وصغيراً للرقم 1، على سبيل المثال، 10 أو 100.
!>          بشكل خاص، لا يجب أن يعتمد على الدقة (أحادية مقابل مزدوجة)
!>          أو حجم المصفوفة. يجب أن يكون على الأقل صفرًا.
!>

NOUNIT

!>          NOUNIT هو INTEGER
!>          رقم وحدة FORTRAN لطباعة رسائل الخطأ
!>          (على سبيل المثال، إذا عادت روتينية IINFO بقيمة غير صفرية.)
!>

A

!>          A هي مصفوفة COMPLEX*16، ذات أبعاد
!>                            (LDA, max(NN))
!>          تستخدم لاحتواء المصفوفة التي سيتم حساب قيمها الذاتية.
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرئيسي للمصفوفة A. يجب أن يكون على الأقل 2 (ليس 1!)
!>          وعلى الأقل max( KK )+1.
!>

SD

!>          SD هي مصفوفة DOUBLE PRECISION، ذات أبعاد (max(NN))
!>          تستخدم لاحتواء القطر الرئيسي للمصفوفة الثلاثية الأقطار
!>          التي تم حسابها بواسطة ZHBTRD.
!>

SE

!>          SE هي مصفوفة DOUBLE PRECISION، ذات أبعاد (max(NN))
!>          تستخدم لاحتواء الأقطار الفرعية للمصفوفة الثلاثية الأقطار
!>          التي تم حسابها بواسطة ZHBTRD.
!>

D1

!>          D1 هي مصفوفة DOUBLE PRECISION، ذات أبعاد (max(NN))
!>

D2

!>          D2 هي مصفوفة DOUBLE PRECISION، ذات أبعاد (max(NN))
!>


*>
D3

!>          D3 هي مصفوفة DOUBLE PRECISION، ذات أبعاد (max(NN))
!>

U

!>          U هي مصفوفة COMPLEX*16، ذات أبعاد (LDU, max(NN))
!>          تستخدم لاحتواء المصفوفة الوحدوية التي تم حسابها بواسطة ZHBTRD.
!>

LDU

!>          LDU هو INTEGER
!>          البعد الرئيسي للمصفوفة U. يجب أن يكون على الأقل 1
!>          وعلى الأقل max( NN ).
!>

WORK

!>          WORK هي مصفوفة COMPLEX*16، ذات أبعاد (LWORK)
!>

LWORK

!>          LWORK هو INTEGER
!>          عدد المدخلات في WORK. يجب أن يكون على الأقل
!>          max( LDA+1, max(NN)+1 )*max(NN).
!>

RWORK

!>          RWORK هي مصفوفة DOUBLE PRECISION
!>

RESULT

!>          RESULT هي مصفوفة DOUBLE PRECISION، ذات أبعاد (4)
!>          القيم المحسوبة بواسطة الاختبارات الموضحة أعلاه.
!>          القيم محدودة حاليًا بـ 1/ulp، لتجنب
!>          الفيضان (overflow).
!>

INFO

!>          INFO هو INTEGER
!>          إذا كانت القيمة 0، فقد تم تشغيل كل شيء بنجاح.
!>
!>-----------------------------------------------------------------------
!>
!>       بعض المتغيرات والمعلمات المحلية:
!>       ---- ----- --------- --- ----------
!>       ZERO, ONE       قيم حقيقية 0 و 1.
!>       MAXTYP          عدد الأنواع المعرفة.
!>       NTEST           عدد الاختبارات التي تم إجراؤها، أو التي يمكن
!>                       إجراؤها حتى الآن، للمصفوفة الحالية.
!>       NTESTT          إجمالي عدد الاختبارات التي تم إجراؤها حتى الآن.
!>       NMAX            أكبر قيمة في NN.
!>       NMATS           عدد المصفوفات التي تم إنشاؤها حتى الآن.
!>       NERRS           عدد الاختبارات التي تجاوزت THRESH
!>                       حتى الآن.
!>       COND, IMODE     قيم سيتم تمريرها إلى مولدات المصفوفات.
!>       ANORM           معيار المصفوفة A؛ سيتم تمريره إلى مولدات المصفوفات.
!>
!>       OVFL, UNFL      عتبات الفيضان (overflow) والجريان السفلي (underflow).
!>       ULP, ULPINV     أدق دقة نسبية ومقلوبها.
!>       RTOVFL, RTUNFL  الجذور التربيعية للقيمتين السابقتين.
!>               المصفوفات الأربع التالية تفك تشفير JTYPE:
!>       KTYPE(j)        النوع العام (1-10) للنوع j.
!>       KMODE(j)        قيمة MODE التي سيتم تمريرها إلى مولد المصفوفة
!>                       للنوع j.
!>       KMAGN(j)        الرتبة الحجمية ( O(1),
!>                       O(overflow^(1/2) ), O(underflow^(1/2) )
!>
