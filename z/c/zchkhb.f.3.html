subroutine zchkhb (integer nsizes, integer, dimension( * ) nn, integer nwdths, integer, dimension( * ) kk, integer ntypes, logical, dimension( * ) dotype, integer, dimension( 4 ) iseed, double precision thresh, integer nounit, complex*16, dimension( lda, * ) a, integer lda, double precision, dimension( * ) sd, double precision, dimension( * ) se, complex*16, dimension( ldu, * ) u, integer ldu, complex*16, dimension( * ) work, integer lwork, double precision, dimension( * ) rwork, double precision, dimension( * ) result, integer info)

ZCHKHB

الغرض:

!>
!> ZCHKHB يختبر اختزال مصفوفة هيرميتية نطاقية إلى شكل ثلاثي الأقطار،
!> يستخدم لمشكلة القيم الذاتية الهيرميتية.
!>
!> ZHBTRD يحلل مصفوفة نطاقية هيرميتية A إلى U S U* ، حيث * تعني
!> منقول مترافق، S ثلاثي الأقطار متناظر، و U وحدوي.
!> يمكن لـ ZHBTRD استخدام إما المثلث السفلي فقط أو المثلث العلوي فقط
!> لـ A؛ ZCHKHB يتحقق من كلا الحالتين.
!>
!> عند استدعاء ZCHKHB، يتم تحديد عدد من أحجام المصفوفات (NN)، وعدد
!> من عرض النطاقات (KK)، وعدد من أنواع المصفوفات (DOTYPE). لكل حجم ()،
!> كل عرض نطاق () أقل من أو يساوي الحجم، وكل نوع مصفوفة، سيتم توليد
!> مصفوفة واحدة واستخدامها لاختبار إجراء الاختزال الهيرميتي النطاقي. لكل
!> مصفوفة، سيتم إجراء عدد من الاختبارات:
!>
!> (1)     | A - V S V* | / ( |A| n ulp )  محسوب بواسطة ZHBTRD مع
!>                                         UPLO='U'
!>
!> (2)     | I - UU* | / ( n ulp )
!>
!> (3)     | A - V S V* | / ( |A| n ulp )  محسوب بواسطة ZHBTRD مع
!>                                         UPLO='L'
!>
!> (4)     | I - UU* | / ( n ulp )
!>
!> يتم تحديد الأحجام بواسطة المصفوفة NN(1:NSIZES)؛ قيمة
!> كل عنصر NN(j) تحدد حجمًا واحدًا.
!> يتم تحديد أنواع المصفوفات بواسطة مصفوفة منطقية DOTYPE( 1:NTYPES )؛
!> إذا كانت DOTYPE(j) هي .TRUE.، فسيتم توليد مصفوفة من النوع j.
!> حاليًا، قائمة الأنواع الممكنة هي:
!>
!> (1)  المصفوفة الصفرية.
!> (2)  مصفوفة الوحدة.
!>
!> (3)  مصفوفة قطرية ذات مدخلات متباعدة بالتساوي
!>      1, ..., ULP  وإشارات عشوائية.
!>      (ULP = (أول رقم أكبر من 1) - 1 )
!> (4)  مصفوفة قطرية ذات مدخلات متباعدة هندسيًا
!>      1, ..., ULP  وإشارات عشوائية.
!> (5)  مصفوفة قطرية ذات مدخلات  1, ULP, ..., ULP
!>      وإشارات عشوائية.
!>
!> (6)  نفس (4)، ولكن مضروبة في SQRT( عتبة تجاوز السعة )
!> (7)  نفس (4)، ولكن مضروبة في SQRT( عتبة نقص السعة )
!>
!> (8)  مصفوفة من الشكل  U* D U، حيث U وحدوي و
!>      D له مدخلات متباعدة بالتساوي 1, ..., ULP مع إشارات عشوائية
!>      على القطر.
!>
!> (9)  مصفوفة من الشكل  U* D U، حيث U وحدوي و
!>      D له مدخلات متباعدة هندسيًا 1, ..., ULP مع إشارات عشوائية
!>      على القطر.
!>
!> (10) مصفوفة من الشكل  U* D U، حيث U وحدوي و
!>      D له مدخلات  1, ULP,..., ULP مع إشارات
!>      عشوائية على القطر.
!>
!> (11) نفس (8)، ولكن مضروبة في SQRT( عتبة تجاوز السعة )
!> (12) نفس (8)، ولكن مضروبة في SQRT( عتبة نقص السعة )
!>
!> (13) مصفوفة هيرميتية بمدخلات عشوائية مختارة من (-1,1).
!> (14) نفس (13)، ولكن مضروبة في SQRT( عتبة تجاوز السعة )
!> (15) نفس (13)، ولكن مضروبة في SQRT( عتبة نقص السعة )
!>

المعلمات

NSIZES 
!>          NSIZES هو عدد صحيح (INTEGER)
!>          عدد أحجام المصفوفات المراد استخدامها. إذا كان صفرًا،
!>          فإن ZCHKHB لا تفعل شيئًا. يجب أن يكون على الأقل صفرًا.
!>

NN

!>          NN هي مصفوفة أعداد صحيحة (INTEGER array)، ذات بعد (NSIZES)
!>          مصفوفة تحتوي على الأحجام المراد استخدامها للمصفوفات.
!>          سيتم تجاهل القيم الصفرية. يجب أن تكون القيم على الأقل
!>          صفرًا.
!>

NWDTHS

!>          NWDTHS هو عدد صحيح (INTEGER)
!>          عدد عرض النطاقات المراد استخدامها. إذا كان صفرًا،
!>          فإن ZCHKHB لا تفعل شيئًا. يجب أن يكون على الأقل صفرًا.
!>

KK

!>          KK هي مصفوفة أعداد صحيحة (INTEGER array)، ذات بعد (NWDTHS)
!>          مصفوفة تحتوي على عرض النطاقات المراد استخدامها للمصفوفات
!>          النطاقية. يجب أن تكون القيم على الأقل صفرًا.
!>

NTYPES

!>          NTYPES هو عدد صحيح (INTEGER)
!>          عدد العناصر في DOTYPE. إذا كان صفرًا، فإن ZCHKHB
!>          لا تفعل شيئًا. يجب أن يكون على الأقل صفرًا. إذا كان MAXTYP+1
!>          و NSIZES هو 1، فسيتم تعريف نوع إضافي، MAXTYP+1، وهو استخدام
!>          أي مصفوفة موجودة في A. هذا مفيد فقط إذا كانت DOTYPE(1:MAXTYP)
!>          هي .FALSE. و DOTYPE(MAXTYP+1) هي .TRUE. .
!>

DOTYPE

!>          DOTYPE هي مصفوفة منطقية (LOGICAL array)، ذات بعد (NTYPES)
!>          إذا كانت DOTYPE(j) هي .TRUE.، فلكل حجم في NN سيتم توليد
!>          مصفوفة بهذا الحجم ومن النوع j.
!>          إذا كانت NTYPES أصغر من الحد الأقصى لعدد الأنواع
!>          المعرفة (PARAMETER MAXTYP)، فلن يتم توليد الأنواع NTYPES+1
!>          حتى MAXTYP. إذا كانت NTYPES أكبر
!>          من MAXTYP، سيتم تجاهل DOTYPE(MAXTYP+1) حتى DOTYPE(NTYPES).
!>

ISEED

!>          ISEED هي مصفوفة أعداد صحيحة (INTEGER array)، ذات بعد (4)
!>          عند الدخول، تحدد ISEED بذرة مولد الأرقام العشوائية. يجب أن تكون
!>          عناصر المصفوفة بين 0 و 4095؛
!>          إذا لم تكن كذلك، فسيتم تقليصها إلى المعامل 4096. أيضًا، يجب أن يكون
!>          ISEED(4) فرديًا. يستخدم مولد الأرقام العشوائية تسلسلًا خطيًا
!>          متطابقًا ومحصورًا بأعداد صحيحة صغيرة، ولذلك
!>          يجب أن ينتج أرقامًا عشوائية مستقلة عن الآلة.
!>          سيتم تغيير قيم ISEED عند الخروج، ويمكن استخدامها في
!>          الاستدعاء التالي لـ ZCHKHB لمواصلة نفس تسلسل الأرقام
!>          العشوائية.
!>

THRESH

!>          THRESH هو دقة مزدوجة (DOUBLE PRECISION)
!>          سيتم اعتبار الاختبار خاطئًا إذا كان الخطأ المحسوب،
!>          كما هو موضح أعلاه، يتجاوز THRESH. لاحظ أن الخطأ
!>          يتم قياسه ليكون O(1)، لذلك يجب أن يكون THRESH مضاعفًا صغيرًا بشكل معقول
!>          لـ 1، على سبيل المثال، 10 أو 100. على وجه الخصوص،
!>          لا يجب أن يعتمد على الدقة (أحادية مقابل مزدوجة)
!>          أو حجم المصفوفة. يجب أن يكون على الأقل صفرًا.
!>

NOUNIT

!>          NOUNIT هو عدد صحيح (INTEGER)
!>          رقم وحدة FORTRAN لطباعة رسائل الخطأ
!>          (على سبيل المثال، إذا أعادت روتينية IINFO غير صفرية).
!>

A

!>          A هي مصفوفة COMPLEX*16، ذات بعد
!>                            (LDA, max(NN))
!>          تستخدم لاحتواء المصفوفة التي سيتم حساب قيمها الذاتية.
!>

LDA

!>          LDA هو عدد صحيح (INTEGER)
!>          البعد الرائد لـ A. يجب أن يكون على الأقل 2 (ليس 1!)
!>          وعلى الأقل max( KK )+1.
!>

SD

!>          SD هي مصفوفة دقة مزدوجة (DOUBLE PRECISION array)، ذات بعد (max(NN))
!>          تستخدم لاحتواء قطر المصفوفة ثلاثية الأقطار المحسوبة
!>          بواسطة ZHBTRD.
!>

SE

!>          SE هي مصفوفة دقة مزدوجة (DOUBLE PRECISION array)، ذات بعد (max(NN))
!>          تستخدم لاحتواء الأقطار الفرعية للمصفوفة ثلاثية الأقطار
!>          المحسوبة بواسطة ZHBTRD.
!>

U

!>          U هي مصفوفة COMPLEX*16، ذات بعد (LDU, max(NN))
!&g
