المعلمات

DOTYPE 
!>          DOTYPE هو مصفوفة منطقية، البعد (NTYPES)
!>          أنواع المصفوفات التي سيتم استخدامها للاختبار. يتم استخدام المصفوفات
!>          من النوع j (لـ 1 <= j <= NTYPES) للاختبار إذا كان DOTYPE(j) =
!>          .TRUE.؛ إذا كان DOTYPE(j) = .FALSE.، فلن يتم استخدام النوع j.
!>

NN

!>          NN هو عدد صحيح
!>          عدد القيم لـ N الموجودة في المتجه NVAL.
!>

NVAL

!>          NVAL هي مصفوفة أعداد صحيحة، البعد (NN)
!>          قيم بُعد المصفوفة N.
!>

NNS

!>          NNS هو عدد صحيح
!>          عدد القيم لـ NRHS الموجودة في المتجه NSVAL.
!>

NSVAL

!>          NSVAL هي مصفوفة أعداد صحيحة، البعد (NNS)
!>          قيم عدد الأطراف اليمنى NRHS.
!>

THRESH

!>          THRESH هو دقة مزدوجة
!>          القيمة الحدية لنسب النتائج. يتم تضمين نتيجة
!>          في ملف الإخراج إذا كانت RESULT >= THRESH. لطباعة كل نسبة اختبار،
!>          استخدم THRESH = 0.
!>

TSTERR

!>          TSTERR هو منطقي
!>          علامة تشير إلى ما إذا كان سيتم اختبار مخرجات الخطأ.
!>

NMAX

!>          NMAX هو عدد صحيح
!>          الحد الأقصى للقيمة المسموح بها لـ N، المستخدمة في تحديد أبعاد
!>          مصفوفات العمل.
!>

A

!>          A هي مصفوفة COMPLEX*16، البعد
!>                      (NMAX*(NMAX+1)/2)
!>

AFAC

!>          AFAC هي مصفوفة COMPLEX*16، البعد
!>                      (NMAX*(NMAX+1)/2)
!>

AINV

!>          AINV هي مصفوفة COMPLEX*16، البعد
!>                      (NMAX*(NMAX+1)/2)
!>

B

!>          B هي مصفوفة COMPLEX*16، البعد (NMAX*NSMAX)
!>          حيث NSMAX هو أكبر إدخال في NSVAL.
!>

X

!>          X هي مصفوفة COMPLEX*16، البعد (NMAX*NSMAX)
!>

XACT

!>          XACT هي مصفوفة COMPLEX*16، البعد (NMAX*NSMAX)
!>

WORK

!>          WORK هي مصفوفة COMPLEX*16، البعد
!>                      (NMAX*max(3,NSMAX))
!>

RWORK

!>          RWORK هي مصفوفة DOUBLE PRECISION، البعد
!>                      (max(NMAX,2*NSMAX))
!>

NOUT

!>          NOUT هو عدد صحيح
!>          رقم الوحدة للإخراج.
!>
