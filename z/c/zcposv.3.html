subroutine zcposv (character uplo, integer n, integer nrhs, complex*16, dimension( lda, * ) a, integer lda, complex*16, dimension( ldb, * ) b, integer ldb, complex*16, dimension( ldx, * ) x, integer ldx, complex*16, dimension( n, * ) work, complex, dimension( * ) swork, double precision, dimension( * ) rwork, integer iter, integer info)

ZCPOSV تحسب حل نظام المعادلات الخطية A * X = B لمصفوفات PO

الغرض:

!>
!> ZCPOSV تحسب حل نظام خطي معقد
!>    A * X = B,
!> حيث A مصفوفة N-by-N هرميتية موجبة محددة و X و B
!> مصفوفتان N-by-NRHS.
!>
!> تحاول ZCPOSV أولاً تفكيك المصفوفة في COMPLEX واستخدام هذا
!> التفكيك ضمن إجراء تنقيح تكراري لإنتاج
!> حل بجودة خطأ رجعي معياري من الدرجة COMPLEX*16 (انظر أدناه).
!> إذا فشل النهج، ينتقل إلى تفكيك من الدرجة COMPLEX*16
!> والحل.
!>
!> لن يكون إجراء التنقيح التكراري استراتيجية رابحة إذا
!> كانت نسبة أداء COMPLEX إلى أداء COMPLEX*16 صغيرة جدًا.
!> يجب أن تأخذ استراتيجية معقولة في الاعتبار عدد الأطراف اليمنى
!> وحجم المصفوفة. يمكن القيام بذلك
!> باستدعاء ILAENV في المستقبل. حتى الآن، نحاول دائمًا
!> التنقيح التكراري.
!>
!> يتم إيقاف عملية التنقيح التكراري إذا
!>     ITER > ITERMAX
!> أو لجميع الأطراف اليمنى لدينا:
!>     RNRM < SQRT(N)*XNRM*ANRM*EPS*BWDMAX
!> حيث
!>     o ITER هو رقم التكرار الحالي في عملية التنقيح
!>       التكراري
!>     o RNRM هو معيار القيمة المطلقة للباقي
!>     o XNRM هو معيار القيمة المطلقة للحل
!>     o ANRM هو معيار عامل المصفوفة A
!>     o EPS هو خطأ الآلة الذي أعادته DLAMCH('Epsilon')
!> يتم تعيين ITERMAX و BWDMAX إلى 30 و 1.0D+00
!> على التوالي.
!>

المعلمات

UPLO 
!>          UPLO is CHARACTER*1
!>          = 'U':  يتم تخزين الجزء العلوي من A؛
!>          = 'L':  يتم تخزين الجزء السفلي من A.
!>

N

!>          N is INTEGER
!>          عدد المعادلات الخطية، أي رتبة
!>          المصفوفة A. N >= 0.
!>

NRHS

!>          NRHS is INTEGER
!>          عدد الأطراف اليمنى، أي عدد الأعمدة
!>          للمصفوفة B. NRHS >= 0.
!>

A

!>          A is COMPLEX*16 array,
!>          dimension (LDA,N)
!>          عند الإدخال، المصفوفة الهرميتية A. إذا كان UPLO = 'U'، فإن الجزء
!>          العلوي المثلثي N-by-N من A يحتوي على الجزء
!>          المثلثي العلوي للمصفوفة A، ولا يتم الرجوع إلى الجزء المثلثي
!>          السفلي الصارم لـ A. إذا كان UPLO = 'L'، فإن الجزء
!>          السفلي المثلثي N-by-N من A يحتوي على الجزء
!>          المثلثي السفلي للمصفوفة A، ولا يتم الرجوع إلى الجزء المثلثي
!>          العلوي الصارم لـ A.
!>
!>          لاحظ أنه لا يلزم تعيين الأجزاء التخيلية للعناصر
!>          القطرية، ويفترض أنها صفر.
!>
!>          عند الخروج، إذا تم استخدام التنقيح التكراري بنجاح
!>          (INFO = 0 و ITER >= 0، انظر الوصف أدناه)، فإن A دون تغيير،
!>          إذا تم استخدام تفكيك الدقة المزدوجة
!>          (INFO = 0 و ITER < 0، انظر الوصف أدناه)، فإن
!>          المصفوفة A تحتوي على العامل U أو L من تفكيك Cholesky
!>          A = U**H*U أو A = L*L**H.
!>

LDA

!>          LDA is INTEGER
!>          البعد الرائد للمصفوفة A. LDA >= max(1,N).
!>

B

!>          B is COMPLEX*16 array, dimension (LDB,NRHS)
!>          المصفوفة B ذات الطرف الأيمن N-by-NRHS.
!>

LDB

!>          LDB is INTEGER
!>          البعد الرائد للمصفوفة B. LDB >= max(1,N).
!>

X

!>          X is COMPLEX*16 array, dimension (LDX,NRHS)
!>          إذا كان INFO = 0، فإن مصفوفة الحل X ذات الأبعاد N-by-NRHS.
!>

LDX

!>          LDX is INTEGER
!>          البعد الرائد للمصفوفة X. LDX >= max(1,N).
!>

WORK

!>          WORK is COMPLEX*16 array, dimension (N,NRHS)
!>          تستخدم هذه المصفوفة للاحتفاظ بمتجهات الباقي.
!>

SWORK

!>          SWORK is COMPLEX array, dimension (N*(N+NRHS))
!>          تستخدم هذه المصفوفة لاستخدام مصفوفة الدقة الفردية
!>          والأطراف اليمنى أو الحلول بالدقة الفردية.
!>

RWORK

!>          RWORK is DOUBLE PRECISION array, dimension (N)
!>

ITER

!>          ITER is INTEGER
!>          < 0: فشل التنقيح التكراري، تم إجراء تفكيك
!>               COMPLEX*16
!>               -1 : تراجعت الروتين إلى الدقة الكاملة لـ
!>                    أسباب متعلقة بالتنفيذ أو الجهاز
!>               -2 : تضييق الدقة تسبب في فيضان،
!>                    تراجعت الروتين إلى الدقة الكاملة
!>               -3 : فشل CPOTRF
!>               -31: توقف التنقيح التكراري بعد 30 تكرارًا
!>          > 0: تم استخدام التنقيح التكراري بنجاح.
!>               يعيد عدد التكرارات
!>

INFO

!>          INFO is INTEGER
!>          = 0:  خروج ناجح
!>          < 0:  إذا كان INFO = -i، فقد كان الوسيط i غير قانوني
!>          > 0:  إذا كان INFO = i، فإن المحدد الرئيسي الرائد من الرتبة i
!>                لـ (COMPLEX*16) A ليس موجبًا، وبالتالي لم يكن من الممكن
!>                إكمال التفكيك، ولم يتم حساب الحل.
!>
