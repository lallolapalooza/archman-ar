subroutine zunbdb2 (integer m, integer p, integer q, complex*16, dimension(ldx11,*) x11, integer ldx11, complex*16, dimension(ldx21,*) x21, integer ldx21, double precision, dimension(*) theta, double precision, dimension(*) phi, complex*16, dimension(*) taup1, complex*16, dimension(*) taup2, complex*16, dimension(*) tauq1, complex*16, dimension(*) work, integer lwork, integer info)

ZUNBDB2

الغرض:

!>
!> ZUNBDB2 تقوم بتحديد كتل مصفوفة طويلة وضيقة X ذات أعمدة متعامدة بشكل متزامن:
!>
!>                            [ B11 ]
!>      [ X11 ]   [ P1 |    ] [  0  ]
!>      [-----] = [---------] [-----] Q1**T .
!>      [ X21 ]   [    | P2 ] [ B21 ]
!>                            [  0  ]
!>
!> X11 هي P-by-Q، و X21 هي (M-P)-by-Q. يجب أن لا يكون P أكبر من M-P,
!> Q, أو M-Q. الروتينات ZUNBDB1, ZUNBDB3, و ZUNBDB4 تعالج الحالات التي
!> يكون فيها P ليس الحد الأدنى للبعد.
!>
!> المصفوفات الوحدوية P1, P2, و Q1 هي P-by-P, (M-P)-by-(M-P),
!> و (M-Q)-by-(M-Q)، على التوالي. يتم تمثيلها ضمنياً بواسطة
!> متجهات هاوسهولدر.
!>
!> B11 و B21 هي مصفوفات ثنائية القطر B11 و B21 ممثلة ضمنياً بواسطة
!> الزوايا THETA, PHI.
!>
!>

المعاملات

M 
!>          M هو عدد صحيح (INTEGER)
!>           عدد الصفوف في X11 بالإضافة إلى عدد الصفوف في X21.
!>

P

!>          P هو عدد صحيح (INTEGER)
!>           عدد الصفوف في X11. 0 <= P <= min(M-P,Q,M-Q).
!>

Q

!>          Q هو عدد صحيح (INTEGER)
!>           عدد الأعمدة في X11 و X21. 0 <= Q <= M.
!>

X11

!>          X11 هي مصفوفة COMPLEX*16، البعد (LDX11,Q)
!>           عند الإدخال، الكتلة العلوية للمصفوفة X المراد اختزالها. عند
!>           الخروج، تحدد أعمدة tril(X11) معاملات الانعكاس لـ P1
!>           وتحدد صفوف triu(X11,1) معاملات الانعكاس لـ Q1.
!>

LDX11

!>          LDX11 هو عدد صحيح (INTEGER)
!>           البعد الرائد لـ X11. LDX11 >= P.
!>

X21

!>          X21 هي مصفوفة COMPLEX*16، البعد (LDX21,Q)
!>           عند الإدخال، الكتلة السفلية للمصفوفة X المراد اختزالها. عند
!>           الخروج، تحدد أعمدة tril(X21) معاملات الانعكاس لـ P2.
!>

LDX21

!>          LDX21 هو عدد صحيح (INTEGER)
!>           البعد الرائد لـ X21. LDX21 >= M-P.
!>

THETA

!>          THETA هي مصفوفة DOUBLE PRECISION، البعد (Q)
!>           تُعرّف عناصر الكتل ثنائية القطر B11, B21 بواسطة
!>           THETA و PHI. انظر التفاصيل الإضافية.
!>

PHI

!>          PHI هي مصفوفة DOUBLE PRECISION، البعد (Q-1)
!>           تُعرّف عناصر الكتل ثنائية القطر B11, B21 بواسطة
!>           THETA و PHI. انظر التفاصيل الإضافية.
!>

TAUP1

!>          TAUP1 هي مصفوفة COMPLEX*16، البعد (P-1)
!>           العوامل القياسية لمعاملات الانعكاس الأولية التي تحدد
!>           P1.
!>

TAUP2

!>          TAUP2 هي مصفوفة COMPLEX*16، البعد (Q)
!>           العوامل القياسية لمعاملات الانعكاس الأولية التي تحدد
!>           P2.
!>

TAUQ1

!>          TAUQ1 هي مصفوفة COMPLEX*16، البعد (Q)
!>           العوامل القياسية لمعاملات الانعكاس الأولية التي تحدد
!>           Q1.
!>

WORK

!>          WORK هي مصفوفة COMPLEX*16، البعد (LWORK)
!>

LWORK

!>          LWORK هو عدد صحيح (INTEGER)
!>           بعد المصفوفة WORK. LWORK >= M-Q.
!>
!>           إذا كان LWORK = -1، يفترض استعلام مساحة العمل؛ تقوم الوحدة الفرعية
!>           فقط بحساب الحجم الأمثل للمصفوفة WORK، وتعيد
!>           هذه القيمة كالإدخال الأول للمصفوفة WORK، ولا تصدر رسالة خطأ
!>           متعلقة بـ LWORK بواسطة XERBLA.
!>

INFO

!>          INFO هو عدد صحيح (INTEGER)
!>           = 0:  خروج ناجح.
!>           < 0:  إذا كان INFO = -i، كان الوسيط i غير قانوني.
!>

تفاصيل إضافية:

!>
!>  الكتل ثنائية القطر العلوية B11, B21 ممثلة ضمنياً بواسطة
!>  الزوايا THETA(1), ..., THETA(Q) و PHI(1), ..., PHI(Q-1). كل عنصر
!>  في كل نطاق ثنائي القطر هو حاصل ضرب جيب التمام أو الجيب لـ THETA
!>  مع جيب التمام أو الجيب لـ PHI. انظر [1] أو ZUNCSD للتفاصيل.
!>
!>  P1, P2, و Q1 ممثلة كحاصل ضرب لمعاملات انعكاس أولية.
!>  انظر ZUNCSD2BY1 للحصول على تفاصيل حول إنشاء P1, P2, و Q1 باستخدام ZUNGQR
!>  و ZUNGLQ.
!>
