subroutine zunglq (integer m, integer n, integer k, complex*16, dimension( lda, * ) a, integer lda, complex*16, dimension( * ) tau, complex*16, dimension( * ) work, integer lwork, integer info)

ZUNGLQ

الغرض:

!>
!> ZUNGLQ تنشئ مصفوفة Q معقدة بحجم M*N ذات صفوف متعامدة،
!> والتي تُعرّف بأنها أول M صفوف من حاصل ضرب K من المعكوسات الأولية
!> ذات الترتيب N
!>
!>       Q  =  H(k)**H . . . H(2)**H H(1)**H
!>
!> كما تم إرجاعها بواسطة ZGELQF.
!>

المعلمات

M 
!>          M هو عدد صحيح (INTEGER)
!>          عدد صفوف المصفوفة Q. M >= 0.
!>

N

!>          N هو عدد صحيح (INTEGER)
!>          عدد أعمدة المصفوفة Q. N >= M.
!>

K

!>          K هو عدد صحيح (INTEGER)
!>          عدد المعكوسات الأولية التي يحدد حاصل ضربها
!>          المصفوفة Q. M >= K >= 0.
!>

A

!>          A هي مصفوفة معقدة (COMPLEX*16) بأبعاد (LDA,N)
!>          عند الإدخال، يجب أن يحتوي الصف i على المتجه الذي يحدد
!>          المعكوس الأولي H(i)، لـ i = 1,2,...,k، كما تم إرجاعه
!>          بواسطة ZGELQF في أول K صفوف من وسيط مصفوفتها A.
!>          عند الخروج، المصفوفة Q بحجم M*N.
!>

LDA

!>          LDA هو عدد صحيح (INTEGER)
!>          البعد الأول للمصفوفة A. LDA >= max(1,M).
!>

TAU

!>          TAU هي مصفوفة معقدة (COMPLEX*16) بأبعاد (K)
!>          يجب أن تحتوي TAU(i) على العامل القياسي للمعكوس
!>          الأولي H(i)، كما تم إرجاعه بواسطة ZGELQF.
!>

WORK

!>          WORK هي مصفوفة معقدة (COMPLEX*16) بأبعاد (MAX(1,LWORK))
!>          عند الخروج، إذا كانت INFO = 0، فإن WORK(1) تُرجع حجم LWORK الأمثل.
!>

LWORK

!>          LWORK هو عدد صحيح (INTEGER)
!>          بعد المصفوفة WORK. LWORK >= max(1,M).
!>          للأداء الأمثل LWORK >= M*NB، حيث NB هو
!>          حجم الكتلة الأمثل.
!>
!>          إذا كانت LWORK = -1، فسيتم افتراض استعلام مساحة العمل؛ يقوم الإجراء الفرعي
!>          بحساب حجم LWORK الأمثل فقط، ويرجع
!>          هذه القيمة كإدخال أول في مصفوفة WORK، ولا يصدر رسالة خطأ
!>          متعلقة بـ LWORK بواسطة XERBLA.
!>

INFO

!>          INFO هو عدد صحيح (INTEGER)
!>          = 0:  انتهاء ناجح؛
!>          < 0:  إذا كانت INFO = -i، فإن الوسيط i لديه قيمة غير قانونية
!>
