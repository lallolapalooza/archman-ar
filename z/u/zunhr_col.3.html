zunhr_col

الاسم

zunhr_col - إزالة جزء علوي مثلثي من مصفوفة هيرميتية معقدة

الملخص

SUBROUTINE ZUNHR_COL(MATRIX_TYPE, N, K, SUB_SUB_ARRAY, LDA, WORK, LWORK, INFO)

CHARACTER(LEN=1)    :: MATRIX_TYPE
INTEGER             :: N
INTEGER             :: K
COMPLEX(KIND=8), DIMENSION(:, :) :: SUB_SUB_ARRAY
INTEGER             :: LDA
COMPLEX(KIND=8), DIMENSION(:)     :: WORK
INTEGER             :: LWORK
INTEGER             :: INFO

توثيق الدالة/الإجراء الفرعي

ZUNHR_COL هي روتين فرعي في مكتبة LAPACK يستخدم لإزالة جزء مثلثي علوي من مصفوفة هيرميتية معقدة. تم تصميم هذا الروتين خصيصًا للعمل مع تحليل القيمة المفردة (SVD) للمصفوفات المعقدة.

المعلمات:

  • MATRIX_TYPE: حرف يحدد نوع المصفوفة. يجب أن تكون 'N' (عادية).
  • N: عدد الصفوف (والأعمدة) للمصفوفة الهيرميتية.
  • K: عدد الأعمدة التي سيتم الاحتفاظ بها في الجزء المثلثي العلوي.
  • SUB_SUB_ARRAY: مصفوفة الأبعاد (N, N) التي تحتوي على جزء المصفوفة الهيرميتية.
  • LDA: خطوة تخزين الصف للمصفوفة SUB_SUB_ARRAY.
  • WORK: مصفوفة عمل لتخزين البيانات الوسيطة.
  • LWORK: طول المصفوفة WORK.
  • INFO: عدد صحيح يمثل معلومات الخطأ. 0 يعني نجاح.

تفاصيل:

تستخدم ZUNHR_COL تحويلات الوحدوية لتقليل المصفوفة الهيرميتية إلى شكلها المثلثي العلوي. غالبًا ما يتم استخدام هذا الروتين كجزء من خوارزمية أكبر لحساب SVD.

مثال:

فيما يلي مثال بسيط لكيفية استدعاء ZUNHR_COL:

PROGRAM EXAMPLE_ZUNHR_COL
    IMPLICIT NONE

    INTEGER, PARAMETER :: N = 3
    INTEGER, PARAMETER :: K = 2
    INTEGER, PARAMETER :: LDA = N
    INTEGER, PARAMETER :: LWORK = 100
    COMPLEX(KIND=8), DIMENSION(N, N) :: SUB_SUB_ARRAY
    COMPLEX(KIND=8), DIMENSION(LWORK) :: WORK
    CHARACTER(LEN=1) :: MATRIX_TYPE
    INTEGER :: INFO

    'N'
    SUB_SUB_ARRAY = RESHAPE(( (1.0_8, 0.0_8), (2.0_8, 0.0_8), (3.0_8, 0.0_8), &
                              (2.0_8, 0.0_8), (4.0_8, 0.0_8), (5.0_8, 0.0_8), &
                              (3.0_8, 0.0_8), (5.0_8, 0.0_8), (6.0_8, 0.0_8) ), SHAPE(SUB_SUB_ARRAY))

    CALL ZUNHR_COL(MATRIX_TYPE, N, K, SUB_SUB_ARRAY, LDA, WORK, LWORK, INFO)

    IF (INFO == 0) THEN
        PRINT *, "ZUNHR_COL executed successfully."
        PRINT *, "Resulting SUB_SUB_ARRAY:"
        PRINT *, SUB_SUB_ARRAY
    ELSE
        PRINT *, "ZUNHR_COL failed with INFO = ", INFO
    END IF

END PROGRAM EXAMPLE_ZUNHR_COL
