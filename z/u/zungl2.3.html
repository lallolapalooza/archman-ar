subroutine zungl2 (m, n, k, a, lda, tau, work, info)
ZUNGL2 ينشئ مصفوفة موحدة (unitary) Q بالكامل أو جزئيًا من تحليل LQ الذي تم تحديده بواسطة cgelqf (خوارزمية غير محظورة).

توثيق الوظيفة/الروتين الفرعي

subroutine zungl2 (integer m, integer n, integer k, complex*16, dimension( lda, * ) a, integer lda, complex*16, dimension( * ) tau, complex*16, dimension( * ) work, integer info)

ZUNGL2 ينشئ مصفوفة موحدة (unitary) Q بالكامل أو جزئيًا من تحليل LQ الذي تم تحديده بواسطة cgelqf (خوارزمية غير محظورة).

الغرض:

!>
!> ZUNGL2 ينشئ مصفوفة معقدة Q بأبعاد m × n مع صفوف متعامدة
!>، والتي تُعرّف على أنها أول m صفوف من حاصل ضرب k عاكسات أولية
!> من الرتبة n
!>
!>       Q  =  H(k)**H . . . H(2)**H H(1)**H
!>
!> كما تم إرجاعها بواسطة ZGELQF.
!>

المعاملات

M 
!>          M هو عدد صحيح (INTEGER)
!>          عدد صفوف المصفوفة Q. M >= 0.
!>

N

!>          N هو عدد صحيح (INTEGER)
!>          عدد أعمدة المصفوفة Q. N >= M.
!>

K

!>          K هو عدد صحيح (INTEGER)
!>          عدد العاكسات الأولية التي يحدد حاصل ضربها المصفوفة Q.
!>          M >= K >= 0.
!>

A

!>          A هي مصفوفة معقدة (COMPLEX*16)، بأبعاد (LDA,N)
!>          عند الإدخال، يجب أن يحتوي الصف i على المتجه الذي يحدد
!>          العاكس الأولي H(i)، لـ i = 1,2,...,k، كما تم إرجاعه
!>          بواسطة ZGELQF في أول k صفوف من وسيطتها المصفوفية A.
!>          عند الخروج، المصفوفة Q بأبعاد m × n.
!>

LDA

!>          LDA هو عدد صحيح (INTEGER)
!>          البعد الأول للمصفوفة A. LDA >= max(1,M).
!>

TAU

!>          TAU هي مصفوفة معقدة (COMPLEX*16)، بأبعاد (K)
!>          يجب أن تحتوي TAU(i) على العامل القياسي للعكس الأولي
!>          H(i)، كما تم إرجاعه بواسطة ZGELQF.
!>

WORK

!>          WORK هي مصفوفة معقدة (COMPLEX*16)، بأبعاد (M)
!>

INFO

!>          INFO هو عدد صحيح (INTEGER)
!>          = 0: خروج ناجح
!>          < 0: إذا كان INFO = -i، فإن الوسيط i لديه قيمة غير قانونية
!>
