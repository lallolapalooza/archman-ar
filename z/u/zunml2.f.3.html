subroutine zunml2 (character side, character trans, integer m, integer n, integer k, complex*16, dimension( lda, * ) a, integer lda, complex*16, dimension( * ) tau, complex*16, dimension( ldc, * ) c, integer ldc, complex*16, dimension( * ) work, integer info)

ZUNML2 تضرب مصفوفة عامة بمصفوفة الوحدة من تحليل LQ المحدد بواسطة cgelqf (خوارزمية غير محظورة).

الغرض:

!>
!> ZUNML2 تستبدل المصفوفة C المعقدة العامة m-by-n بـ
!>
!>       Q * C  إذا كان SIDE = 'L' و TRANS = 'N', أو
!>
!>       Q**H* C  إذا كان SIDE = 'L' و TRANS = 'C', أو
!>
!>       C * Q  إذا كان SIDE = 'R' و TRANS = 'N', أو
!>
!>       C * Q**H إذا كان SIDE = 'R' و TRANS = 'C',
!>
!> حيث Q هي مصفوفة الوحدة المعقدة المعرفة كناتج لمُعكِسات عنصرية k
!>
!>       Q = H(k)**H . . . H(2)**H H(1)**H
!>
!> كما تم إرجاعها بواسطة ZGELQF. Q من الرتبة m إذا كان SIDE = 'L' ومن الرتبة n
!> إذا كان SIDE = 'R'.
!>

المعلمات

SIDE 
!>          SIDE هو CHARACTER*1
!>          = 'L': تطبيق Q أو Q**H من اليسار
!>          = 'R': تطبيق Q أو Q**H من اليمين
!>

TRANS

!>          TRANS هو CHARACTER*1
!>          = 'N': تطبيق Q  (بدون تبديل)
!>          = 'C': تطبيق Q**H (منقول مترافق)
!>

M

!>          M هو INTEGER
!>          عدد الصفوف في المصفوفة C. M >= 0.
!>

N

!>          N هو INTEGER
!>          عدد الأعمدة في المصفوفة C. N >= 0.
!>

K

!>          K هو INTEGER
!>          عدد المُعكِسات العنصرية التي يُعرّف حاصل ضربها
!>          المصفوفة Q.
!>          إذا كان SIDE = 'L', M >= K >= 0;
!>          إذا كان SIDE = 'R', N >= K >= 0.
!>

A

!>          A هو مصفوفة COMPLEX*16, البعد
!>                               (LDA,M) إذا كان SIDE = 'L',
!>                               (LDA,N) إذا كان SIDE = 'R'
!>          يجب أن يحتوي الصف i على المتجه الذي يحدد
!>          المُعكِس العنصري H(i), لـ i = 1,2,...,k, كما تم إرجاعه بواسطة
!>          ZGELQF في الصفوف k الأولى من مصفوفة الوسيط A الخاصة به.
!>          يتم تعديل A بواسطة الإجراء ولكن يتم استعادته عند الخروج.
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرائد للمصفوفة A. LDA >= max(1,K).
!>

TAU

!>          TAU هو مصفوفة COMPLEX*16, البعد (K)
!>          يجب أن يحتوي TAU(i) على العامل القياسي لـ
!>          المُعكِس العنصري H(i), كما تم إرجاعه بواسطة ZGELQF.
!>

C

!>          C هو مصفوفة COMPLEX*16, البعد (LDC,N)
!>          عند الدخول، المصفوفة C بحجم m-by-n.
!>          عند الخروج، يتم استبدال C بـ Q*C أو Q**H*C أو C*Q**H أو C*Q.
!>

LDC

!>          LDC هو INTEGER
!>          البعد الرائد للمصفوفة C. LDC >= max(1,M).
!>

WORK

!>          WORK هو مصفوفة COMPLEX*16, البعد
!>                                   (N) إذا كان SIDE = 'L',
!>                                   (M) إذا كان SIDE = 'R'
!>

INFO

!>          INFO هو INTEGER
!>          = 0: خروج ناجح
!>          < 0: إذا كان INFO = -i، فإن الوسيط i كان له قيمة غير قانونية
!>
