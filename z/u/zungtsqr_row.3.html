توثيق الوظيفة/الروتين الفرعي

subroutine zungtsqr_row (integer m, integer n, integer mb, integer nb, complex*16, dimension( lda, * ) a, integer lda, complex*16, dimension( ldt, * ) t, integer ldt, complex*16, dimension( * ) work, integer lwork, integer info)

ZUNGTSQR_ROW

الغرض:

!>
!> ZUNGTSQR_ROW يولد مصفوفة مركبة Q_out ذات أبعاد M × N بأعمدة متعامدة
!> من ناتج ZLATSQR. هذه الأعمدة المتعامدة N هي الأعمدة الأولى لحاصل ضرب
!> مصفوفات وحدوية مركبة Q(k)_in من الرتبة M، والتي يتم إرجاعها بواسطة ZLATSQR
!> في صيغة خاصة.
!>
!>      Q_out = أول_N_أعمدة_من( Q(1)_in * Q(2)_in * ... * Q(k)_in ).
!>
!> مصفوفات الإدخال Q(k)_in مخزنة في كتل صفوف وأعمدة في A.
!> راجع توثيق ZLATSQR لمزيد من التفاصيل حول صيغة
!> Q(k)_in، حيث يتم تمثيل كل Q(k)_in بتحويلات هاوسهولدر مجمعة.
!> تستدعي هذه الروتين روتينًا مساعدًا ZLARFB_GETT، حيث يتم إجراء الحساب
!> على كل كتلة فردية. المسح الخوارزمي أولاً لكتل الأعمدة بحجم NB
!> من اليمين إلى اليسار بدءًا من كتلة الصف السفلي ويستمر إلى كتلة الصف العلوي
!> (وبالتالي _ROW في اسم الروتين). هذا المسح بترتيب عكسي
!> لترتيب قيام ZLATSQR بإنشاء كتل الإخراج.
!>

المعلمات:

M 
!>          M هو INTEGER
!>          عدد صفوف المصفوفة A.  M >= 0.
!>

N

!>          N هو INTEGER
!>          عدد أعمدة المصفوفة A. M >= N >= 0.
!>

MB

!>          MB هو INTEGER
!>          حجم كتلة الصف المستخدم بواسطة ZLATSQR لإرجاع
!>          المصفوفات A و T. MB > N.
!>          (ملاحظة: إذا كان MB > M، فسيتم استخدام M بدلاً من MB
!>          كحجم كتلة الصف).
!>

NB

!>          NB هو INTEGER
!>          حجم كتلة العمود المستخدم بواسطة ZLATSQR لإرجاع
!>          المصفوفات A و T. NB >= 1.
!>          (ملاحظة: إذا كان NB > N، فسيتم استخدام N بدلاً من NB
!>          كحجم كتلة العمود).
!>

A

!>          A هي مصفوفة COMPLEX*16، أبعادها (LDA,N)
!>
!>          عند الإدخال:
!>
!>             العناصر الموجودة على القطر وفوقه غير مستخدمة كـ
!>             مدخلات. العناصر الموجودة تحت القطر تمثل المصفوفة
!>             المجمعة شبه القطرية السفلية V التي حسبها ZLATSQR
!>             والتي تحدد مصفوفات الإدخال Q_in(k) (الواحدات على
!>             القطر غير مخزنة). انظر ZLATSQR لمزيد من التفاصيل.
!>
!>          عند الخروج:
!>
!>             تحتوي المصفوفة A على مصفوفة متعامدة Q_out بأبعاد M × N،
!>             أي أن أعمدة A هي متجهات وحدات متعامدة.
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرئيسي للمصفوفة A.  LDA >= max(1,M).
!>

T

!>          T هي مصفوفة COMPLEX*16،
!>          أبعادها (LDT, N * NIRB)
!>          حيث NIRB = عدد كتل الصفوف المدخلة
!>                     = MAX( 1, CEIL((M-N)/(MB-N)) )
!>          لنفترض NICB = عدد كتل الأعمدة المدخلة
!>                   = CEIL(N/NB)
!>
!>          محددات الكتلة المثلثية العلوية المستخدمة لتحديد
!>          مصفوفات الإدخال Q_in(k)، k=(1:NIRB*NICB).
!>          محددات الكتلة مخزنة بصيغة مدمجة في N * NIRB كتلة عمود
!>          من T وتتكون من NICB كتل عمود أصغر بحجم NB × NB
!>          مثلثة علوية. انظر ZLATSQR لمزيد من التفاصيل حول صيغة
!>          T.
!>

LDT

!>          LDT هو INTEGER
!>          البعد الرئيسي للمصفوفة T.
!>          LDT >= max(1,min(NB,N)).
!>

WORK

!>          (مساحة العمل) مصفوفة COMPLEX*16، أبعادها (MAX(1,LWORK))
!>          عند الخروج، إذا كان INFO = 0، فإن WORK(1) يرجع LWORK الأمثل.
!>

LWORK

!>          LWORK هو INTEGER
!>          بعد المصفوفة WORK.
!>          LWORK >= NBLOCAL * MAX(NBLOCAL,(N-NBLOCAL)),
!>          حيث NBLOCAL=MIN(NB,N).
!>          إذا كان LWORK = -1، فسيتم افتراض استعلام عن مساحة العمل.
!>          تحسب الروتين فقط الحجم الأمثل لمصفوفة WORK،
!>          وترجع هذا الحجم كالإدخال الأول لمصفوفة WORK،
!>          ولا تصدر رسالة خطأ متعلقة بـ LWORK بواسطة XERBLA.
!>

INFO

!>          INFO هو INTEGER
!>          = 0:  انتهاء ناجح
!>          < 0:  إذا كان INFO = -i، فإن الوسيط i كان له قيمة غير قانونية
!>
