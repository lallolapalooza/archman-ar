subroutine zuncsd2by1 (character jobu1, character jobu2, character jobv1t, integer m, integer p, integer q, complex*16, dimension(ldx11,*) x11, integer ldx11, complex*16, dimension(ldx21,*) x21, integer ldx21, double precision, dimension(*) theta, complex*16, dimension(ldu1,*) u1, integer ldu1, complex*16, dimension(ldu2,*) u2, integer ldu2, complex*16, dimension(ldv1t,*) v1t, integer ldv1t, complex*16, dimension(*) work, integer lwork, double precision, dimension(*) rwork, integer lrwork, integer, dimension(*) iwork, integer info)

ZUNCSD2BY1

الغرض:

!>
!> ZUNCSD2BY1 يحسب تفكيك CS لمصفوفة X بحجم M × Q ذات أعمدة متعامدة تم تقسيمها
!> إلى هيكل كتلة 2 × 1:
!>
!>                                [  I1 0  0 ]
!>                                [  0  C  0 ]
!>          [ X11 ]   [ U1 |    ] [  0  0  0 ]
!>      X = [-----] = [---------] [----------] V1**T .
!>          [ X21 ]   [    | U2 ] [  0  0  0 ]
!>                                [  0  S  0 ]
!>                                [  0  0  I2]
!>
!> X11 بحجم P × Q. المصفوفات الوحدوية U1، U2، و V1 هي P × P،
!> (M-P) × (M-P)، و Q × Q على التوالي. C و S هي مصفوفات قطرية غير سالبة بحجم R × R تحقق C^2 + S^2 = I، حيث
!> R = MIN(P,M-P,Q,M-Q). I1 هي مصفوفة وحدة بحجم K1 × K1 و I2 هي مصفوفة وحدة بحجم
!> K2 × K2، حيث K1 = MAX(Q+P-M,0)، K2 = MAX(Q-P,0).
!>

المعلمات

JOBU1 
!>          JOBU1 هي CHARACTER
!>          = 'Y':      يتم حساب U1؛
!>          غير ذلك:  لا يتم حساب U1.
!>

JOBU2

!>          JOBU2 هي CHARACTER
!>          = 'Y':      يتم حساب U2؛
!>          غير ذلك:  لا يتم حساب U2.
!>

JOBV1T

!>          JOBV1T هي CHARACTER
!>          = 'Y':      يتم حساب V1T؛
!>          غير ذلك:  لا يتم حساب V1T.
!>

M

!>          M هي INTEGER
!>          عدد الصفوف في X.
!>

P

!>          P هي INTEGER
!>          عدد الصفوف في X11. 0 <= P <= M.
!>

Q

!>          Q هي INTEGER
!>          عدد الأعمدة في X11 و X21. 0 <= Q <= M.
!>

X11

!>          X11 هي مصفوفة COMPLEX*16، البعد (LDX11,Q)
!>          عند الدخول، جزء من المصفوفة الوحدوية التي يراد تفكيك CS الخاص بها.
!>

LDX11

!>          LDX11 هي INTEGER
!>          البعد الرئيسي لـ X11. LDX11 >= MAX(1,P).
!>

X21

!>          X21 هي مصفوفة COMPLEX*16، البعد (LDX21,Q)
!>          عند الدخول، جزء من المصفوفة الوحدوية التي يراد تفكيك CS الخاص بها.
!>

LDX21

!>          LDX21 هي INTEGER
!>          البعد الرئيسي لـ X21. LDX21 >= MAX(1,M-P).
!>

THETA

!>          THETA هي مصفوفة DOUBLE PRECISION، البعد (R)، حيث R =
!>          MIN(P,M-P,Q,M-Q).
!>          C = DIAG( COS(THETA(1)), ... , COS(THETA(R)) ) و
!>          S = DIAG( SIN(THETA(1)), ... , SIN(THETA(R)) ).
!>

U1

!>          U1 هي مصفوفة COMPLEX*16، البعد (P)
!>          إذا كان JOBU1 = 'Y'، فإن U1 تحتوي على المصفوفة الوحدوية U1 بحجم P × P.
!>

LDU1

!>          LDU1 هي INTEGER
!>          البعد الرئيسي لـ U1. إذا كان JOBU1 = 'Y'، فإن LDU1 >=
!>          MAX(1,P).
!>

U2

!>          U2 هي مصفوفة COMPLEX*16، البعد (M-P)
!>          إذا كان JOBU2 = 'Y'، فإن U2 تحتوي على المصفوفة الوحدوية U2 بحجم (M-P) × (M-P).
!>

LDU2

!>          LDU2 هي INTEGER
!>          البعد الرئيسي لـ U2. إذا كان JOBU2 = 'Y'، فإن LDU2 >=
!>          MAX(1,M-P).
!>

V1T

!>          V1T هي مصفوفة COMPLEX*16، البعد (Q)
!>          إذا كان JOBV1T = 'Y'، فإن V1T تحتوي على المصفوفة الوحدوية V1**T بحجم Q × Q.
!>

LDV1T

!>          LDV1T هي INTEGER
!>          البعد الرئيسي لـ V1T. إذا كان JOBV1T = 'Y'، فإن LDV1T >=
!>          MAX(1,Q).
!>

WORK

!>          WORK هي مصفوفة COMPLEX*16، البعد (MAX(1,LWORK))
!>          عند الخروج، إذا كان INFO = 0، فإن WORK(1) تعيد LWORK المثلى.
!>

LWORK

!>          LWORK هي INTEGER
!>          بعد مصفوفة WORK.
!>
!>          إذا كان LWORK = -1، فيفترض استعلام مساحة العمل؛ تقوم الروتين فقط
!>          بحساب الحجم الأمثل لمصفوفتي WORK و RWORK، وتعيد هذا الحجم كمدخل أول
!>          لمصفوفة WORK و RWORK على التوالي، ولا يتم إصدار أي رسالة خطأ متعلقة
!>          بـ LWORK أو LRWORK بواسطة XERBLA.
!>

RWORK

!>          RWORK هي مصفوفة DOUBLE PRECISION، البعد (MAX(1,LRWORK))
!>          عند الخروج، إذا كان INFO = 0، فإن RWORK(1) تعيد LRWORK المثلى.
!>          إذا كان INFO > 0 عند الخروج، فإن RWORK(2:R) تحتوي على قيم PHI(1)،
!>          ...، PHI(R-1) التي، مع THETA(1)، ...، THETA(R)،
!>          تحدد المصفوفة في الشكل الثنائي القطري الوسيط
!>          المتبقي بعد عدم التقارب. INFO يحدد عدد
!>          قيم PHI غير الصفرية.
!>

LRWORK

!>          LRWORK هي INTEGER
!>          بعد مصفوفة RWORK.
!>
!>          إذا كان LRWORK = -1، فيفترض استعلام مساحة العمل؛ تقوم الروتين فقط
!>          بحساب الحجم الأمثل لمصفوفتي WORK و RWORK، وتعيد هذا الحجم كمدخل أول
!>          لمصفوفة WORK و RWORK على التوالي، ولا يتم إصدار أي رسالة خطأ متعلقة
!>          بـ LWORK أو LRWORK بواسطة XERBLA.
!>

IWORK

!>          IWORK هي مصفوفة INTEGER، البعد (M-MIN(P,M-P,Q,M-Q))
!>

INFO

!>          INFO هي INTEGER
!>          = 0:  خروج ناجح.
!>          < 0:  إذا كان INFO = -i، فإن المعامل i كان له قيمة غير قانونية.
!>          > 0:  لم يتقارب ZBBCSD. انظر وصف WORK
!>                أعلاه للحصول على التفاصيل.
!>
