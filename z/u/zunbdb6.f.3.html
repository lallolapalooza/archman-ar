subroutine zunbdb6 (integer m1, integer m2, integer n, complex*16, dimension(*) x1, integer incx1, complex*16, dimension(*) x2, integer incx2, complex*16, dimension(ldq1,*) q1, integer ldq1, complex*16, dimension(ldq2,*) q2, integer ldq2, complex*16, dimension(*) work, integer lwork, integer info)

ZUNBDB6

الغرض:

!>
!> ZUNBDB6 يعامد المتجه العمودي
!>      X = [ X1 ]
!>          [ X2 ]
!> بالنسبة للأعمدة
!>      Q = [ Q1 ] .
!>          [ Q2 ]
!> يجب أن تكون أعمدة Q متعامدة. سيكون المتجه المتعامد صفراً إذا وفقط إذا كان يقع بالكامل في نطاق Q.
!>
!> يتم حساب الإسقاط بما يصل إلى تكرارين لخوارزمية Gram-Schmidt الكلاسيكية، انظر
!> * L. Giraud, J. Langou, M. Rozložník. 
!>   2002. CERFACS Technical Report No. TR/PA/02/33. URL:
!>   https://www.cerfacs.fr/algor/reports/2002/TR_PA_02_33.pdf
!>
!>

المعلمات

M1 
!>          M1 is INTEGER
!>           بعد X1 وعدد الصفوف في Q1. 0 <= M1.
!>

M2

!>          M2 is INTEGER
!>           بعد X2 وعدد الصفوف في Q2. 0 <= M2.
!>

N

!>          N is INTEGER
!>           عدد الأعمدة في Q1 و Q2. 0 <= N.
!>

X1

!>          X1 is COMPLEX*16 array, dimension (M1)
!>           عند الدخول، الجزء العلوي من المتجه المراد تعامده.
!>           عند الخروج، الجزء العلوي من المتجه المسقط.
!>

INCX1

!>          INCX1 is INTEGER
!>           زيادة لإدخالات X1.
!>

X2

!>          X2 is COMPLEX*16 array, dimension (M2)
!>           عند الدخول، الجزء السفلي من المتجه المراد
!>           تعامده. عند الخروج، الجزء السفلي من المتجه المسقط.
!>

INCX2

!>          INCX2 is INTEGER
!>           زيادة لإدخالات X2.
!>

Q1

!>          Q1 is COMPLEX*16 array, dimension (LDQ1, N)
!>           الجزء العلوي من مصفوفة الأساس المتعامد.
!>

LDQ1

!>          LDQ1 is INTEGER
!>           البعد الرائد لـ Q1. LDQ1 >= M1.
!>

Q2

!>          Q2 is COMPLEX*16 array, dimension (LDQ2, N)
!>           الجزء السفلي من مصفوفة الأساس المتعامد.
!>

LDQ2

!>          LDQ2 is INTEGER
!>           البعد الرائد لـ Q2. LDQ2 >= M2.
!>

WORK

!>          WORK is COMPLEX*16 array, dimension (LWORK)
!>

LWORK

!>          LWORK is INTEGER
!>           بعد المصفوفة WORK. LWORK >= N.
!>

INFO

!>          INFO is INTEGER
!>           = 0:  نجاح الخروج.
!>           < 0:  إذا كان INFO = -i، فإن الوسيط i كان له قيمة غير قانونية.
!>
