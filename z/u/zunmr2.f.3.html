توثيق الوظيفة/الروتين الفرعي

subroutine zunmr2 (character side, character trans, integer m, integer n, integer k, complex*16, dimension( lda, * ) a, integer lda, complex*16, dimension( * ) tau, complex*16, dimension( ldc, * ) c, integer ldc, complex*16, dimension( * ) work, integer info)

ZUNMR2 تضرب مصفوفة عامة بالمصفوفة الوحدوية الناتجة من تحليل RQ المحدد بواسطة cgerqf (خوارزمية غير محظورة).

الغرض:

!>
!> ZUNMR2 تستبدل المصفوفة المعقدة C التي أبعادها m × n بـ
!>
!>       Q * C  إذا كانت SIDE = 'L' و TRANS = 'N'، أو
!>
!>       Q**H* C  إذا كانت SIDE = 'L' و TRANS = 'C'، أو
!>
!>       C * Q  إذا كانت SIDE = 'R' و TRANS = 'N'، أو
!>
!>       C * Q**H إذا كانت SIDE = 'R' و TRANS = 'C'،
!>
!> حيث Q هي مصفوفة وحدوية معقدة محددة كحاصل ضرب k
!> عواكس عنصرية
!>
!>       Q = H(1)**H H(2)**H . . . H(k)**H
!>
!> كما تم إرجاعها بواسطة ZGERQF. Q تكون من الرتبة m إذا كانت SIDE = 'L' ومن الرتبة n
!> إذا كانت SIDE = 'R'.
!>

المعلمات

SIDE 
!>          SIDE هو CHARACTER*1
!>          = 'L': تطبيق Q أو Q**H من اليسار
!>          = 'R': تطبيق Q أو Q**H من اليمين
!>

TRANS

!>          TRANS هو CHARACTER*1
!>          = 'N': تطبيق Q (بدون تبديل)
!>          = 'C': تطبيق Q**H (تبديل مقترن)
!>

M

!>          M هو INTEGER
!>          عدد صفوف المصفوفة C. M >= 0.
!>

N

!>          N هو INTEGER
!>          عدد أعمدة المصفوفة C. N >= 0.
!>

K

!>          K هو INTEGER
!>          عدد العواكس العنصرية التي يحدد حاصل ضربها
!>          المصفوفة Q.
!>          إذا كانت SIDE = 'L'، فإن M >= K >= 0؛
!>          إذا كانت SIDE = 'R'، فإن N >= K >= 0.
!>

A

!>          A هي مصفوفة COMPLEX*16، أبعادها
!>                               (LDA,M) إذا كانت SIDE = 'L'،
!>                               (LDA,N) إذا كانت SIDE = 'R'
!>          يجب أن يحتوي الصف i على المتجه الذي يحدد
!>          العاكس العنصري H(i)، لـ i = 1، 2، ...، k، كما تم إرجاعه بواسطة
!>          ZGERQF في الصفوف k الأخيرة من وسيط المصفوفة A الخاص به.
!>          يتم تعديل A بواسطة الروتين الفرعي ولكن يتم استعادتها عند الخروج.
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرئيسي للمصفوفة A. LDA >= max(1,K).
!>

TAU

!>          TAU هي مصفوفة COMPLEX*16، أبعادها (K)
!>          يجب أن يحتوي TAU(i) على العامل العددي للعامل
!>          العنصري H(i)، كما تم إرجاعه بواسطة ZGERQF.
!>

C

!>          C هي مصفوفة COMPLEX*16، أبعادها (LDC,N)
!>          عند الإدخال، المصفوفة C ذات الأبعاد m × n.
!>          عند الخروج، يتم استبدال C بـ Q*C أو Q**H*C أو C*Q**H أو C*Q.
!>

LDC

!>          LDC هو INTEGER
!>          البعد الرئيسي للمصفوفة C. LDC >= max(1,M).
!>

WORK

!>          WORK هي مصفوفة COMPLEX*16، أبعادها
!>                                   (N) إذا كانت SIDE = 'L'،
!>                                   (M) إذا كانت SIDE = 'R'
!>

INFO

!>          INFO هو INTEGER
!>          = 0: خروج ناجح
!>          < 0: إذا كان INFO = -i، فإن الوسيط i كان له قيمة غير قانونية
!>
