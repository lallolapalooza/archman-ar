توثيق الوظيفة/الوحدة الفرعية

subroutine zunmbr (character vect, character side, character trans, integer m, integer n, integer k, complex*16, dimension( lda, * ) a, integer lda, complex*16, dimension( * ) tau, complex*16, dimension( ldc, * ) c, integer ldc, complex*16, dimension( * ) work, integer lwork, integer info)

ZUNMBR

الغرض:

!>
!> إذا كانت VECT = 'Q'، فإن ZUNMBR تستبدل المصفوفة المعقدة العامة M-by-N المسماة C بـ
!>                 SIDE = 'L'     SIDE = 'R'
!> TRANS = 'N':      Q * C          C * Q
!> TRANS = 'C':      Q**H * C       C * Q**H
!>
!> إذا كانت VECT = 'P'، فإن ZUNMBR تستبدل المصفوفة المعقدة العامة M-by-N المسماة C بـ
!>                 SIDE = 'L'     SIDE = 'R'
!> TRANS = 'N':      P * C          C * P
!> TRANS = 'C':      P**H * C       C * P**H
!>
!> هنا Q و P**H هما المصفوفتان الوحدويتان اللتان تحددهما ZGEBRD عند
!> تقليل مصفوفة معقدة A إلى شكل ثنائي القطر: A = Q * B * P**H. تُعرّف Q
!> و P**H كناتج لمُنشِئات انعكاس أولية H(i) و G(i) على التوالي.
!>
!> ليكن nq = m إذا كانت SIDE = 'L' و nq = n إذا كانت SIDE = 'R'. وبالتالي nq هو
!> ترتيب المصفوفة الوحدوية Q أو P**H التي يتم تطبيقها.
!>
!> إذا كانت VECT = 'Q'، يُفترض أن A كانت مصفوفة NQ-by-K:
!> إذا كان nq >= k، فإن Q = H(1) H(2) . . . H(k)؛
!> إذا كان nq < k، فإن Q = H(1) H(2) . . . H(nq-1).
!>
!> إذا كانت VECT = 'P'، يُفترض أن A كانت مصفوفة K-by-NQ:
!> إذا كان k < nq، فإن P = G(1) G(2) . . . G(k)؛
!> إذا كان k >= nq، فإن P = G(1) G(2) . . . G(nq-1).
!>

المعلمات

VECT 
!>          VECT هو CHARACTER*1
!>          = 'Q': تطبيق Q أو Q**H؛
!>          = 'P': تطبيق P أو P**H.
!>

SIDE

!>          SIDE هو CHARACTER*1
!>          = 'L': تطبيق Q أو Q**H أو P أو P**H من اليسار؛
!>          = 'R': تطبيق Q أو Q**H أو P أو P**H من اليمين.
!>

TRANS

!>          TRANS هو CHARACTER*1
!>          = 'N':  بدون تبديل، تطبيق Q أو P؛
!>          = 'C':  تبديل مرافق، تطبيق Q**H أو P**H.
!>

M

!>          M هو INTEGER
!>          عدد صفوف المصفوفة C. M >= 0.
!>

N

!>          N هو INTEGER
!>          عدد أعمدة المصفوفة C. N >= 0.
!>

K

!>          K هو INTEGER
!>          إذا كانت VECT = 'Q'، عدد الأعمدة في المصفوفة الأصلية
!>          التي تم تقليلها بواسطة ZGEBRD.
!>          إذا كانت VECT = 'P'، عدد الصفوف في المصفوفة الأصلية
!>          التي تم تقليلها بواسطة ZGEBRD.
!>          K >= 0.
!>

A

!>          A هي مصفوفة COMPLEX*16، الأبعاد
!>                                (LDA,min(nq,K)) إذا كانت VECT = 'Q'
!>                                (LDA,nq)        إذا كانت VECT = 'P'
!>          المتجهات التي تحدد مُنشِئات الانعكاس الأولية H(i) و
!>          G(i)، والتي تحدد نواتجها المصفوفتين Q و P، كما
!>          تم إرجاعها بواسطة ZGEBRD.
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرائد للمصفوفة A.
!>          إذا كانت VECT = 'Q'، فإن LDA >= max(1,nq)؛
!>          إذا كانت VECT = 'P'، فإن LDA >= max(1,min(nq,K)).
!>

TAU

!>          TAU هي مصفوفة COMPLEX*16، الأبعاد (min(nq,K))
!>          يجب أن تحتوي TAU(i) على العامل القياسي لمُنشِئ الانعكاس الأولي
!>          H(i) أو G(i) الذي يحدد Q أو P، كما تم إرجاعه
!>          بواسطة ZGEBRD في وسيط المصفوفة TAUQ أو TAUP.
!>

C

!>          C هي مصفوفة COMPLEX*16، الأبعاد (LDC,N)
!>          عند الإدخال، المصفوفة C بحجم M-by-N.
!>          عند الخروج، يتم استبدال C بـ Q*C أو Q**H*C أو C*Q**H أو C*Q
!>          أو P*C أو P**H*C أو C*P أو C*P**H.
!>

LDC

!>          LDC هو INTEGER
!>          البعد الرائد للمصفوفة C. LDC >= max(1,M).
!>

WORK

!>          WORK هي مصفوفة COMPLEX*16، الأبعاد (MAX(1,LWORK))
!>          عند الخروج، إذا كانت INFO = 0، فإن WORK(1) ترجع LWORK الأمثل.
!>

LWORK

!>          LWORK هو INTEGER
!>          بعد المصفوفة WORK.
!>          إذا كانت SIDE = 'L'، فإن LWORK >= max(1,N)؛
!>          إذا كانت SIDE = 'R'، فإن LWORK >= max(1,M)؛
!>          إذا كان N = 0 أو M = 0، فإن LWORK >= 1.
!>          للحصول على الأداء الأمثل، LWORK >= max(1,N*NB) إذا كانت SIDE = 'L'،
!>          و LWORK >= max(1,M*NB) إذا كانت SIDE = 'R'، حيث NB هو
!>          حجم الكتلة الأمثل. (NB = 0 إذا كان M = 0 أو N = 0.)
!>
!>          إذا كانت LWORK = -1، فسيتم افتراض استعلام عن مساحة العمل؛ تقوم الوحدة الفرعية
!>          بحساب حجم مساحة العمل الأمثل فقط، وترجع
!>          هذه القيمة كالإدخال الأول لمساحة العمل، ولن يصدر XERBLA أي رسالة خطأ
!>          متعلقة بـ LWORK.
!>

INFO

!>          INFO هو INTEGER
!>          = 0:  انتهاء ناجح
!>          < 0:  إذا كانت INFO = -i، فإن الوسيط i كان له قيمة غير قانونية
!>
