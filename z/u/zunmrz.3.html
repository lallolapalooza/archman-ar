توثيق الوظيفة/الروتين الفرعي

subroutine zunmrz (character side, character trans, integer m, integer n, integer k, integer l, complex*16, dimension( lda, * ) a, integer lda, complex*16, dimension( * ) tau, complex*16, dimension( ldc, * ) c, integer ldc, complex*16, dimension( * ) work, integer lwork, integer info)

ZUNMRZ

الغرض:

!>
!> ZUNMRZ يستبدل مصفوفة C المركبة العامة M-by-N بـ
!>
!>                 SIDE = 'L'     SIDE = 'R'
!> TRANS = 'N':      Q * C          C * Q
!> TRANS = 'C':      Q**H * C       C * Q**H
!>
!> حيث Q هي مصفوفة مركبة وحدوية محددة كناتج لـ k
!> عاكسات أولية
!>
!>       Q = H(1) H(2) . . . H(k)
!>
!> كما تم إرجاعها بواسطة ZTZRZF. Q لها رتبة M إذا كان SIDE = 'L' ورتبة N
!> إذا كان SIDE = 'R'.
!>

المعلمات

SIDE 
!>          SIDE هو CHARACTER*1
!>          = 'L': تطبيق Q أو Q**H من اليسار؛
!>          = 'R': تطبيق Q أو Q**H من اليمين.
!>

TRANS

!>          TRANS هو CHARACTER*1
!>          = 'N':  لا يوجد تبديل، تطبيق Q؛
!>          = 'C':  تبديل مترافق، تطبيق Q**H.
!>

M

!>          M هو INTEGER
!>          عدد صفوف المصفوفة C. M >= 0.
!>

N

!>          N هو INTEGER
!>          عدد أعمدة المصفوفة C. N >= 0.
!>

K

!>          K هو INTEGER
!>          عدد العاكسات الأولية التي يحدد حاصل ضربها
!>          المصفوفة Q.
!>          إذا كان SIDE = 'L'، فإن M >= K >= 0؛
!>          إذا كان SIDE = 'R'، فإن N >= K >= 0.
!>

L

!>          L هو INTEGER
!>          عدد أعمدة المصفوفة A التي تحتوي على
!>          الجزء الهام من عاكسات هاوسهولدر.
!>          إذا كان SIDE = 'L'، فإن M >= L >= 0، إذا كان SIDE = 'R'، فإن N >= L >= 0.
!>

A

!>          A هي مصفوفة COMPLEX*16، ذات بعد
!>                               (LDA,M) إذا كان SIDE = 'L'،
!>                               (LDA,N) إذا كان SIDE = 'R'
!>          يجب أن يحتوي الصف i على المتجه الذي يحدد
!>          العاكس الأولي H(i)، لـ i = 1,2,...,k، كما تم إرجاعه بواسطة
!>          ZTZRZF في الصفوف k الأخيرة من وسيطة المصفوفة A الخاصة به.
!>          يتم تعديل A بواسطة الروتين ولكن يتم استعادته عند الخروج.
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرائد للمصفوفة A. LDA >= max(1,K).
!>

TAU

!>          TAU هي مصفوفة COMPLEX*16، ذات بعد (K)
!>          يجب أن تحتوي TAU(i) على العامل القياسي
!>          للعاكس الأولي H(i)، كما تم إرجاعه بواسطة ZTZRZF.
!>

C

!>          C هي مصفوفة COMPLEX*16، ذات بعد (LDC,N)
!>          عند الإدخال، المصفوفة C ذات الأبعاد M-by-N.
!>          عند الخروج، يتم استبدال C بـ Q*C أو Q**H*C أو C*Q**H أو C*Q.
!>

LDC

!>          LDC هو INTEGER
!>          البعد الرائد للمصفوفة C. LDC >= max(1,M).
!>

WORK

!>          WORK هي مصفوفة COMPLEX*16، ذات بعد (MAX(1,LWORK))
!>          عند الخروج، إذا كان INFO = 0، فإن WORK(1) يرجع LWORK المثلى.
!>

LWORK

!>          LWORK هو INTEGER
!>          بعد المصفوفة WORK.
!>          إذا كان SIDE = 'L'، فإن LWORK >= max(1,N)؛
!>          إذا كان SIDE = 'R'، فإن LWORK >= max(1,M).
!>          لأداء جيد، يجب أن يكون LWORK بشكل عام أكبر.
!>
!>          إذا كان LWORK = -1، فسيتم افتراض استعلام مساحة عمل؛ الروتين
!>          يحسب فقط الحجم الأمثل لمصفوفة WORK، ويعيد
!>          هذه القيمة كالإدخال الأول لمصفوفة WORK، ولا يتم إصدار أي رسالة خطأ
!>          متعلقة بـ LWORK بواسطة XERBLA.
!>

INFO

!>          INFO هو INTEGER
!>          = 0:  خروج ناجح
!>          < 0:  إذا كان INFO = -i، فإن الوسيط i كان له قيمة غير قانونية
!>
