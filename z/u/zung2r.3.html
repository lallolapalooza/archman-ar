توثيق الدالة/الدُرَجية

subroutine zung2r (integer m, integer n, integer k, complex*16, dimension( lda, * ) a, integer lda, complex*16, dimension( * ) tau, complex*16, dimension( * ) work, integer info)

ZUNG2R

الغرض:

!>
!> ZUNG2R تولد مصفوفة معقدة Q بحجم m × n بأعمدة متعامدة،
!> والتي تُعرّف على أنها أول n أعمدة من حاصل ضرب k عواكس أولية من الرتبة m
!>
!>       Q  =  H(1) H(2) . . . H(k)
!>
!> كما تم إرجاعها بواسطة ZGEQRF.
!>

المعلمات

M 
!>          M هو عدد صحيح (INTEGER)
!>          عدد صفوف المصفوفة Q. M >= 0.
!>

N

!>          N هو عدد صحيح (INTEGER)
!>          عدد أعمدة المصفوفة Q. M >= N >= 0.
!>

K

!>          K هو عدد صحيح (INTEGER)
!>          عدد العواكس الأولية التي يُعرّف حاصل ضربها المصفوفة Q. N >= K >= 0.
!>

A

!>          A هي مصفوفة من النوع COMPLEX*16، أبعادها (LDA,N)
!>          عند الإدخال، يجب أن يحتوي العمود i على المتجه الذي
!>          يُعرّف العاكس الأولي H(i)، من أجل i = 1,2,...,k، كما
!>          تم إرجاعها بواسطة ZGEQRF في أول n أعمدة من مصفوفة الوسائط
!>          A الخاصة بها.
!>          عند الخروج، المصفوفة Q بحجم m × n.
!>

LDA

!>          LDA هو عدد صحيح (INTEGER)
!>          البعد الأول للمصفوفة A. LDA >= max(1,M).
!>

TAU

!>          TAU هي مصفوفة من النوع COMPLEX*16، أبعادها (K)
!>          يجب أن تحتوي TAU(i) على المعامل العددي للعاكس الأولي
!>          H(i)، كما تم إرجاعها بواسطة ZGEQRF.
!>

WORK

!>          WORK هي مصفوفة من النوع COMPLEX*16، أبعادها (N)
!>

INFO

!>          INFO هو عدد صحيح (INTEGER)
!>          = 0: خروج ناجح
!>          < 0: إذا كان INFO = -i، فإن الوسيط i لديه قيمة غير صالحة
!>
