subroutine zunbdb (character trans, character signs, integer m, integer p, integer q, complex*16, dimension( ldx11, * ) x11, integer ldx11, complex*16, dimension( ldx12, * ) x12, integer ldx12, complex*16, dimension( ldx21, * ) x21, integer ldx21, complex*16, dimension( ldx22, * ) x22, integer ldx22, double precision, dimension( * ) theta, double precision, dimension( * ) phi, complex*16, dimension( * ) taup1, complex*16, dimension( * ) taup2, complex*16, dimension( * ) tauq1, complex*16, dimension( * ) tauq2, complex*16, dimension( * ) work, integer lwork, integer info)

ZUNBDB

الغرض:

!>
!> تقوم ZUNBDB بإجراء تحليل ثنائي متزامن لكتل المصفوفة الوحدوية X بحجم M-by-M:
!>
!>                                 [ B11 | B12 0  0 ]
!>     [ X11 | X12 ]   [ P1 |    ] [  0  |  0 -I  0 ] [ Q1 |    ]**H
!> X = [-----------] = [---------] [----------------] [---------]   .
!>     [ X21 | X22 ]   [    | P2 ] [ B21 | B22 0  0 ] [    | Q2 ]
!>                                 [  0  |  0  0  I ]
!>
!> X11 هي بحجم P-by-Q. يجب أن لا تكون Q أكبر من P أو M-P أو M-Q. (إذا لم يكن
!> الأمر كذلك، فيجب تبديل X و/أو إعادة ترتيبها. يمكن القيام بذلك في وقت ثابت
!> باستخدام خيارات TRANS و SIGNS. انظر ZUNCSD للتفاصيل.)
!>
!> المصفوفات الوحدوية P1، P2، Q1، و Q2 هي بحجم P-by-P، (M-P)-by-(M-P)،
!> Q-by-Q، و (M-Q)-by-(M-Q) على التوالي. يتم تمثيلها ضمنيًا بواسطة متجهات هاوسهولدر.
!>
!> B11، B12، B21، و B22 هي مصفوفات ثنائية القطر بحجم Q-by-Q ممثلة
!> ضمنيًا بالزاويتين THETA و PHI.
!>

المعلمات

TRANS 
!>          TRANS هي CHARACTER
!>          = 'T':      يتم تخزين X، U1، U2، V1T، و V2T بترتيب الصفوف؛
!>          غير ذلك:   يتم تخزين X، U1، U2، V1T، و V2T بترتيب الأعمدة.
!>

SIGNS

!>          SIGNS هي CHARACTER
!>          = 'O':      تصبح الكتلة السفلية اليسرى غير موجبة (المعيار)؛
!>          غير ذلك:   تصبح الكتلة العلوية اليمنى غير موجبة (المعيار).
!>

M

!>          M هي INTEGER
!>          عدد الصفوف والأعمدة في X.
!>

P

!>          P هي INTEGER
!>          عدد الصفوف في X11 و X12. 0 <= P <= M.
!>

Q

!>          Q هي INTEGER
!>          عدد الأعمدة في X11 و X21. 0 <= Q <=
!>          MIN(P,M-P,M-Q).
!>

X11

!>          X11 هي مصفوفة COMPLEX*16، البعد (LDX11,Q)
!>          عند الدخول، الكتلة العلوية اليسرى للمصفوفة الوحدوية المراد
!>          تقليلها. عند الخروج، يعتمد الشكل على TRANS:
!>          إذا كان TRANS = 'N'، فإن
!>             أعمدة tril(X11) تحدد المعاملات لـ P1،
!>             صفوف triu(X11,1) تحدد المعاملات لـ Q1؛
!>          إذا كان TRANS = 'T'، و
!>             صفوف triu(X11) تحدد المعاملات لـ P1،
!>             أعمدة tril(X11,-1) تحدد المعاملات لـ Q1.
!>

LDX11

!>          LDX11 هي INTEGER
!>          البعد الرائد لـ X11. إذا كان TRANS = 'N'، فإن LDX11 >=
!>          P؛ إذا كان TRANS = 'T'، فإن LDX11 >= Q.
!>

X12

!>          X12 هي مصفوفة COMPLEX*16، البعد (LDX12,M-Q)
!>          عند الدخول، الكتلة العلوية اليمنى للمصفوفة الوحدوية المراد
!>          تقليلها. عند الخروج، يعتمد الشكل على TRANS:
!>          إذا كان TRANS = 'N'، فإن
!>             صفوف triu(X12) تحدد المعاملات الأولى لـ Q2؛
!>          إذا كان TRANS = 'T'، و
!>             أعمدة tril(X12) تحدد المعاملات الأولى لـ Q2.
!>

LDX12

!>          LDX12 هي INTEGER
!>          البعد الرائد لـ X12. إذا كان TRANS = 'N'، فإن LDX12 >=
!>          P؛ إذا كان TRANS = 'T'، فإن LDX11 >= M-Q.
!>

X21

!>          X21 هي مصفوفة COMPLEX*16، البعد (LDX21,Q)
!>          عند الدخول، الكتلة السفلية اليسرى للمصفوفة الوحدوية المراد
!>          تقليلها. عند الخروج، يعتمد الشكل على TRANS:
!>          إذا كان TRANS = 'N'، فإن
!>             أعمدة tril(X21) تحدد المعاملات لـ P2؛
!>          إذا كان TRANS = 'T'، و
!>             صفوف triu(X21) تحدد المعاملات لـ P2.
!>

LDX21

!>          LDX21 هي INTEGER
!>          البعد الرائد لـ X21. إذا كان TRANS = 'N'، فإن LDX21 >=
!>          M-P؛ إذا كان TRANS = 'T'، فإن LDX21 >= Q.
!>

X22

!>          X22 هي مصفوفة COMPLEX*16، البعد (LDX22,M-Q)
!>          عند الدخول، الكتلة السفلية اليمنى للمصفوفة الوحدوية المراد
!>          تقليلها. عند الخروج، يعتمد الشكل على TRANS:
!>          إذا كان TRANS = 'N'، فإن
!>             صفوف triu(X22(Q+1:M-P,P+1:M-Q)) تحدد المعاملات الأخيرة
!>             لـ Q2،
!>          إذا كان TRANS = 'T'، و
!>             أعمدة tril(X22(P+1:M-Q,Q+1:M-P)) تحدد المعاملات الأخيرة
!>             لـ P2.
!>

LDX22

!>          LDX22 هي INTEGER
!>          البعد الرائد لـ X22. إذا كان TRANS = 'N'، فإن LDX22 >=
!>          M-P؛ إذا كان TRANS = 'T'، فإن LDX22 >= M-Q.
!>

THETA

!>          THETA هي مصفوفة DOUBLE PRECISION، البعد (Q)
!>          يمكن حساب عناصر الكتل الثنائية القطر B11، B12، B21، B22
!!>          من الزوايا THETA و PHI. انظر التفاصيل الإضافية.
!>

PHI

!>          PHI هي مصفوفة DOUBLE PRECISION، البعد (Q-1)
!>          يمكن حساب عناصر الكتل الثنائية القطر B11، B12، B21، B22
!>          من الزوايا THETA و PHI. انظر التفاصيل الإضافية.
!>

TAUP1

!>          TAUP1 هي مصفوفة COMPLEX*16، البعد (P)
!>          عوامل القياس للمعاملات الأولية التي تحدد P1.
!>

TAUP2

!>          TAUP2 هي مصفوفة COMPLEX*16، البعد (M-P)
!>          عوامل القياس للمعاملات الأولية التي تحدد P2.
!>

TAUQ1

!>          TAUQ1 هي مصفوفة COMPLEX*16، البعد (Q)
!>          عوامل القياس للمعاملات الأولية التي تحدد Q1.
!>

TAUQ2

!>          TAUQ2 هي مصفوفة COMPLEX*16، البعد (M-Q)
!>          عوامل القياس للمعاملات الأولية التي تحدد Q2.
!>

WORK

!>          WORK هي مصفوفة COMPLEX*16، البعد (LWORK)
!>

LWORK

!>          LWORK هي INTEGER
!>          بعد المصفوفة WORK. LWORK >= M-Q.
!>
!>          إذا كان LWORK = -1، فيفترض استعلام مساحة العمل؛ تقوم الوحدة الفرعية
!>          فقط بحساب الحجم الأمثل للمصفوفة WORK، وتعيد هذا الحجم كأول إدخال
!>          في المصفوفة WORK، ولا يتم إصدار أي رسالة خطأ تتعلق بـ LWORK بواسطة XERBLA.
!>

INFO

!>          INFO هي INTEGER
!>          = 0:  انتهاء ناجح.
!>          < 0:  إذا كان INFO = -i، فإن الوسيط i كان له قيمة غير قانونية.
!>

تفاصيل إضافية:

!>
!>  يتم تمثيل الكتل ثنائية القطر B11، B12، B21، و B22
!>  بشكل ضمني بواسطة الزوايا THETA(1)، ..., THETA(Q) و PHI(1)، ...,
!>  PHI(Q-1). B11 و B21 هما ثنائيا القطر علويان، بينما B21 و B22 هما
!>  ثنائيا القطر سفليان. كل مدخل في كل نطاق ثنائي القطر هو ناتج
!>  ضرب جيب أو جيب تمام زاوية THETA بجيب أو جيب تمام زاوية PHI. انظر
!>  [1] أو ZUNCSD للحصول على التفاصيل.
!>
!>  يتم تمثيل P1، P2، Q1، و Q2 كناتج ضرب لمُبدِلات أولية. انظر ZUNCSD
!>  للحصول على تفاصيل حول إنشاء P1، P2، Q1، و Q2
!>  باستخدام ZUNGQR و ZUNGLQ.
!>
