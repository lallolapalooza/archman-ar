subroutine zunhr_col01 (integer m, integer n, integer mb1, integer nb1, integer nb2, double precision, dimension(6) result)

ZUNHR_COL01

الغرض:

!>
!> ZUNHR_COL01 تختبر ZUNGTSQR و ZUNHR_COL باستخدام ZLATSQR، ZGEMQRT.
!> لذلك، يجب اختبار ZLATSQR (جزء من ZGEQR) و ZGEMQRT (جزء من ZGEMQR)
!> قبل هذا الاختبار.
!>
!>

المعلمات

M 
!>          M هو عدد صحيح (INTEGER)
!>          عدد الصفوف في مصفوفة الاختبار.
!>

N

!>          N هو عدد صحيح (INTEGER)
!>          عدد الأعمدة في مصفوفة الاختبار.
!>

MB1

!>          MB1 هو عدد صحيح (INTEGER)
!>          عدد الصفوف في كتلة الصف في مصفوفة الاختبار المدخلة.
!>

NB1

!>          NB1 هو عدد صحيح (INTEGER)
!>          عدد الأعمدة في كتلة العمود في مصفوفة الاختبار المدخلة.
!>

NB2

!>          NB2 هو عدد صحيح (INTEGER)
!>          عدد الأعمدة في كتلة العمود في مصفوفة الاختبار المخرجة.
!>

RESULT

!>          RESULT هو مصفوفة ذات دقة مزدوجة (DOUBLE PRECISION)، بُعدها (6)
!>          نتائج كل من الاختبارات الستة أدناه.
!>
!>            A هي مصفوفة اختبار مدخلة بحجم m × n ليتم تحليلها.
!>            بحيث A = Q_gr * ( R )
!>                               ( 0 )،
!>
!>            Q_qr هي مصفوفة وحدوية ضمنية بحجم m × m، نتيجة
!>            التحليل في تمثيل WY المحظور،
!>            مخزنة بتنسيق إخراج ZGEQRT.
!>
!>            R هي مصفوفة علوية مثلثية بحجم n × n،
!>
!>            0 هي مصفوفة صفرية بحجم (m-n) × n،
!>
!>            Q هي مصفوفة وحدوية صريحة بحجم m × m، Q = Q_gr * I
!>
!>            C هي مصفوفة عشوائية بحجم m × n،
!>
!>            D هي مصفوفة عشوائية بحجم n × m.
!>
!>            الاختبارات الستة هي:
!>
!>            RESULT(1) = |R - (Q**H) * A| / ( eps * m * |A| )
!>              مكافئ للاختبار لـ | A - Q * R | / (eps * m * |A|)،
!>
!>            RESULT(2) = |I - (Q**H) * Q| / ( eps * m )،
!>
!>            RESULT(3) = | Q_qr * C - Q * C | / (eps * m * |C|)،
!>
!>            RESULT(4) = | (Q_gr**H) * C - (Q**H) * C | / (eps * m * |C|)
!>
!>            RESULT(5) = | D * Q_qr - D * Q | / (eps * m * |D|)
!>
!>            RESULT(6) = | D * (Q_qr**H) - D * (Q**H) | / (eps * m * |D|)،
!>
!>            حيث:
!>              Q_qr * C، (Q_gr**H) * C، D * Q_qr، D * (Q_qr**H) يتم
!>              حسابها باستخدام ZGEMQRT،
!>
!>              Q * C، (Q**H) * C، D * Q، D * (Q**H) يتم
!>              حسابها باستخدام ZGEMM.
!>
