subroutine zunm22 (character side, character trans, integer m, integer n, integer n1, integer n2, complex*16, dimension( ldq, * ) q, integer ldq, complex*16, dimension( ldc, * ) c, integer ldc, complex*16, dimension( * ) work, integer lwork, integer info)

ZUNM22 تضرب مصفوفة عامة بمصفوفة وحدة شريطية.

الغرض

!>
!>  ZUNM22 تستبدل المصفوفة C العامة المعقدة M-by-N بـ
!>
!>                  SIDE = 'L'     SIDE = 'R'
!>  TRANS = 'N':      Q * C          C * Q
!>  TRANS = 'C':      Q**H * C       C * Q**H
!>
!>  حيث Q هي مصفوفة وحدة معقدة من الرتبة NQ، مع NQ = M إذا كان
!>  SIDE = 'L' و NQ = N إذا كان SIDE = 'R'.
!>  تعالج مصفوفة الوحدة Q هيكل كتلة 2x2
!>
!>         [  Q11  Q12  ]
!>     Q = [            ]
!>         [  Q21  Q22  ],
!>
!>  حيث Q12 هي مصفوفة مثلثة سفلية بحجم N1xN1 و Q21 هي مصفوفة مثلثة علوية بحجم N2xN2.
!>

المعلمات

SIDE 
!>          SIDE is CHARACTER*1
!>          = 'L': تطبيق Q أو Q**H من اليسار؛
!>          = 'R': تطبيق Q أو Q**H من اليمين.
!>

TRANS

!>          TRANS is CHARACTER*1
!>          = 'N':  تطبيق Q (بدون تبديل)؛
!>          = 'C':  تطبيق Q**H (منقول مترافق).
!>

M

!>          M is INTEGER
!>          عدد صفوف المصفوفة C. M >= 0.
!>

N

!>          N is INTEGER
!>          عدد أعمدة المصفوفة C. N >= 0.
!>

N1
N2

!>          N1 is INTEGER
!>          N2 is INTEGER
!>          أبعاد Q12 و Q21 على التوالي. N1, N2 >= 0.
!>          يجب استيفاء المتطلب التالي:
!>          N1 + N2 = M إذا كان SIDE = 'L' و N1 + N2 = N إذا كان SIDE = 'R'.
!>

Q

!>          Q is COMPLEX*16 array, dimension
!>                              (LDQ,M) if SIDE = 'L'
!>                              (LDQ,N) if SIDE = 'R'
!>

LDQ

!>          LDQ is INTEGER
!>          البعد الرئيسي للمصفوفة Q.
!>          LDQ >= max(1,M) if SIDE = 'L'; LDQ >= max(1,N) if SIDE = 'R'.
!>

C

!>          C is COMPLEX*16 array, dimension (LDC,N)
!>          عند الدخول، المصفوفة C بحجم M-by-N.
!>          عند الخروج، يتم استبدال C بـ Q*C أو Q**H*C أو C*Q**H أو C*Q.
!>

LDC

!>          LDC is INTEGER
!>          البعد الرئيسي للمصفوفة C. LDC >= max(1,M).
!>

WORK

!>          WORK is COMPLEX*16 array, dimension (MAX(1,LWORK))
!>          عند الخروج، إذا كان INFO = 0، فإن WORK(1) ترجع LWORK الأمثل.
!>

LWORK

!>          LWORK is INTEGER
!>          بعدد المصفوفة WORK.
!>          إذا كان SIDE = 'L'، فإن LWORK >= max(1,N)؛
!>          إذا كان SIDE = 'R'، فإن LWORK >= max(1,M).
!>          للحصول على الأداء الأمثل، LWORK >= M*N.
!>
!>          إذا كان LWORK = -1، فإنه يُفترض استعلام عن مساحة العمل؛ تقوم الوحدة
!>          فقط بحساب الحجم الأمثل للمصفوفة WORK، وإرجاع
!>          هذه القيمة كالإدخال الأول للمصفوفة WORK، ولا يصدر XERBLA أي رسالة خطأ
!>          متعلقة بـ LWORK.
!>

INFO

!>          INFO is INTEGER
!>          = 0:  انتهاء ناجح
!>          < 0:  إذا كان INFO = -i، فإن الوسيط i كان له قيمة غير قانونية
!>
