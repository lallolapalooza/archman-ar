subroutine zunt03 (character*( * ) rc, integer mu, integer mv, integer n, integer k, complex*16, dimension( ldu, * ) u, integer ldu, complex*16, dimension( ldv, * ) v, integer ldv, complex*16, dimension( * ) work, integer lwork, double precision, dimension( * ) rwork, double precision result, integer info)

ZUNT03

الغرض:

!>
!> ZUNT03 يقارن مصفوفتين وحدويتين U و V لمعرفة ما إذا كانت الصفوف أو الأعمدة المقابلة لهما تمتد المساحات نفسها. يتم فحص الصفوف إذا كان RC = 'R'، ويتم فحص الأعمدة إذا كان RC = 'C'.
!>
!> RESULT هو الحد الأقصى لـ
!>
!>    | V*V' - I | / ( MV ulp )، إذا كان RC = 'R'، أو
!>
!>    | V'*V - I | / ( MV ulp )، إذا كان RC = 'C'،
!>
!> والحد الأقصى عبر الصفوف (أو الأعمدة) من 1 إلى K من
!>
!>    | U(i) - S*V(i) |/ ( N ulp )
!>
!> حيث abs(S) = 1 (مختارة لتقليل التعبير)، U(i) هو الصف (العمود) i من U، و V(i) هو الصف (العمود) i من V.
!>

المعلمات

RC 
!>          RC هو CHARACTER*1
!>          إذا كان RC = 'R' فسيتم مقارنة صفوف U و V.
!>          إذا كان RC = 'C' فسيتم مقارنة أعمدة U و V.
!>

MU

!>          MU هو INTEGER
!>          عدد صفوف U إذا كان RC = 'R'، وعدد الأعمدة إذا كان RC = 'C'. إذا كان MU = 0، فإن ZUNT03 لا تفعل شيئًا.
!>          يجب أن يكون MU صفرًا أو أكبر.
!>

MV

!>          MV هو INTEGER
!>          عدد صفوف V إذا كان RC = 'R'، وعدد الأعمدة إذا كان RC = 'C'. إذا كان MV = 0، فإن ZUNT03 لا تفعل شيئًا.
!>          يجب أن يكون MV صفرًا أو أكبر.
!>

N

!>          N هو INTEGER
!>          إذا كان RC = 'R'، عدد أعمدة المصفوفتين U و V، وإذا كان RC = 'C'، عدد صفوف U و V. إذا كان N = 0، فإن ZUNT03 لا تفعل شيئًا.
!>          يجب أن يكون N صفرًا أو أكبر.
!>

K

!>          K هو INTEGER
!>          عدد الصفوف أو الأعمدة من U و V للمقارنة.
!>          0 <= K <= max(MU,MV).
!>

U

!>          U هو مصفوفة COMPLEX*16، بأبعاد (LDU,N)
!>          المصفوفة الأولى للمقارنة. إذا كان RC = 'R'، فإن U هي MU × N، وإذا كان RC = 'C'، فإن U هي N × MU.
!>

LDU

!>          LDU هو INTEGER
!>          البعد الرائد لـ U. إذا كان RC = 'R'، فإن LDU >= max(1,MU)، وإذا كان RC = 'C'، فإن LDU >= max(1,N).
!>

V

!>          V هو مصفوفة COMPLEX*16، بأبعاد (LDV,N)
!>          المصفوفة الثانية للمقارنة. إذا كان RC = 'R'، فإن V هي MV × N، وإذا كان RC = 'C'، فإن V هي N × MV.
!>

LDV

!>          LDV هو INTEGER
!>          البعد الرائد لـ V. إذا كان RC = 'R'، فإن LDV >= max(1,MV)، وإذا كان RC = 'C'، فإن LDV >= max(1,N).
!>

WORK

!>          WORK هو مصفوفة COMPLEX*16، بأبعاد (LWORK)
!>

LWORK

!>          LWORK هو INTEGER
!>          طول مصفوفة WORK. للحصول على أفضل أداء، يجب أن يكون LWORK
!>          على الأقل N*N إذا كان RC = 'C' أو M*M إذا كان RC = 'R'، ولكن
!>          سيتم إجراء الاختبارات حتى لو كان LWORK = 0.
!>

RWORK

!>          RWORK هو مصفوفة DOUBLE PRECISION، بأبعاد (max(MV,N))
!>

RESULT

!>          RESULT هو DOUBLE PRECISION
!>          القيمة المحسوبة بواسطة الاختبار الموصوف أعلاه. يتم تحديد RESULT
!>          بـ 1/ulp لتجنب تجاوز السعة.
!>

INFO

!>          INFO هو INTEGER
!>          0 يشير إلى خروج ناجح
!>          -k يشير إلى أن المعلمة k كانت ذات قيمة غير قانونية
!>
