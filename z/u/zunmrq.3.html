subroutine zunmrq (character side, character trans, integer m, integer n, integer k, complex*16, dimension( lda, * ) a, integer lda, complex*16, dimension( * ) tau, complex*16, dimension( ldc, * ) c, integer ldc, complex*16, dimension( * ) work, integer lwork, integer info)

ZUNMRQ

الغرض:

!>
!> ZUNMRQ تستبدل المصفوفة العامة المعقدة M-by-N المسماة C بـ
!>
!>                 SIDE = 'L'     SIDE = 'R'
!> TRANS = 'N':      Q * C          C * Q
!> TRANS = 'C':      Q**H * C       C * Q**H
!>
!> حيث Q هي مصفوفة وحدوية معقدة محددة كناتج لـ k
!> عاكسات أولية
!>
!>       Q = H(1)**H H(2)**H . . . H(k)**H
!>
!> كما تم إرجاعها بواسطة ZGERQF. Q من الرتبة M إذا كانت SIDE = 'L' ومن الرتبة N
!> إذا كانت SIDE = 'R'.
!>

المعلمات

SIDE 
!>          SIDE is CHARACTER*1
!>          = 'L': تطبيق Q أو Q**H من اليسار؛
!>          = 'R': تطبيق Q أو Q**H من اليمين.
!>

TRANS

!>          TRANS is CHARACTER*1
!>          = 'N':  بدون تبديل، تطبيق Q؛
!>          = 'C':  تبديل مقترن، تطبيق Q**H.
!>

M

!>          M is INTEGER
!>          عدد صفوف المصفوفة C. M >= 0.
!>

N

!>          N is INTEGER
!>          عدد أعمدة المصفوفة C. N >= 0.
!>

K

!>          K is INTEGER
!>          عدد العاكسات الأولية التي تحدد ناتجها
!>          المصفوفة Q.
!>          إذا كانت SIDE = 'L', M >= K >= 0؛
!>          إذا كانت SIDE = 'R', N >= K >= 0.
!>

A

!>          A is COMPLEX*16 array, dimension
!>                               (LDA,M) if SIDE = 'L',
!>                               (LDA,N) if SIDE = 'R'
!>          يجب أن يحتوي الصف i على المتجه الذي يحدد
!>          العاكس الأولي H(i), من أجل i = 1,2,...,k, كما تم إرجاعه بواسطة
!>          ZGERQF في آخر k صفوف من وسيط المصفوفة A الخاص به.
!>

LDA

!>          LDA is INTEGER
!>          البعد الرئيسي للمصفوفة A. LDA >= max(1,K).
!>

TAU

!>          TAU is COMPLEX*16 array, dimension (K)
!>          يجب أن يحتوي TAU(i) على العامل القياسي للعنصر
!>          العاكس H(i), كما تم إرجاعه بواسطة ZGERQF.
!>

C

!>          C is COMPLEX*16 array, dimension (LDC,N)
!>          عند الدخول، المصفوفة M-by-N C.
!>          عند الخروج، يتم استبدال C بـ Q*C أو Q**H*C أو C*Q**H أو C*Q.
!>

LDC

!>          LDC is INTEGER
!>          البعد الرئيسي للمصفوفة C. LDC >= max(1,M).
!>

WORK

!>          WORK is COMPLEX*16 array, dimension (MAX(1,LWORK))
!>          عند الخروج، إذا كان INFO = 0، فإن WORK(1) يعيد LWORK المثلى.
!>

LWORK

!>          LWORK is INTEGER
!>          بعد المصفوفة WORK.
!>          إذا كانت SIDE = 'L', LWORK >= max(1,N)؛
!>          إذا كانت SIDE = 'R', LWORK >= max(1,M).
!>          لأداء جيد، يجب أن يكون LWORK أكبر عمومًا.
!>
!>          إذا كان LWORK = -1، فسيتم افتراض استعلام مساحة العمل؛ تقوم الوحدة الفرعية
!>          بحساب حجم مساحة العمل المثلى فقط، وإرجاع
!>          هذه القيمة كالإدخال الأول لمساحة العمل، ولا يصدر أي رسالة خطأ
!>          متعلقة بـ LWORK بواسطة XERBLA.
!>

INFO

!>          INFO is INTEGER
!>          = 0:  الانتهاء بنجاح
!>          < 0:  إذا كان INFO = -i، فإن الوسيط i كان له قيمة غير قانونية
!>
