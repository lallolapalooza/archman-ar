subroutine zunm2r (character side, character trans, integer m, integer n, integer k, complex*16, dimension( lda, * ) a, integer lda, complex*16, dimension( * ) tau, complex*16, dimension( ldc, * ) c, integer ldc, complex*16, dimension( * ) work, integer info)

ZUNM2R تضرب مصفوفة عامة بمصفوفة وحدوية من تحليل QR محدد بواسطة cgeqrf (خوارزمية غير محظورة).

الغرض:

!>
!> ZUNM2R تستبدل المصفوفة C العامة المعقدة m-by-n بـ
!>
!>       Q * C  إذا كان SIDE = 'L' و TRANS = 'N', أو
!>
!>       Q**H* C  إذا كان SIDE = 'L' و TRANS = 'C', أو
!>
!>       C * Q  إذا كان SIDE = 'R' و TRANS = 'N', أو
!>
!>       C * Q**H إذا كان SIDE = 'R' و TRANS = 'C',
!>
!> حيث Q مصفوفة وحدوية معقدة معرفة كناتج لـ k
!> عاكسات أولية
!>
!>       Q = H(1) H(2) . . . H(k)
!>
!> كما تم إرجاعها بواسطة ZGEQRF. Q من الرتبة m إذا كان SIDE = 'L' ومن الرتبة n
!> إذا كان SIDE = 'R'.
!>

المعلمات

SIDE 
!>          SIDE هو CHARACTER*1
!>          = 'L': تطبيق Q أو Q**H من اليسار
!>          = 'R': تطبيق Q أو Q**H من اليمين
!>

TRANS

!>          TRANS هو CHARACTER*1
!>          = 'N': تطبيق Q  (بدون تبديل)
!>          = 'C': تطبيق Q**H (تبديل مترافق)
!>

M

!>          M هو INTEGER
!>          عدد صفوف المصفوفة C. M >= 0.
!>

N

!>          N هو INTEGER
!>          عدد أعمدة المصفوفة C. N >= 0.
!>

K

!>          K هو INTEGER
!>          عدد العاكسات الأولية التي يحدد ناتجها
!>          المصفوفة Q.
!>          إذا كان SIDE = 'L', M >= K >= 0;
!>          إذا كان SIDE = 'R', N >= K >= 0.
!>

A

!>          A هو مصفوفة COMPLEX*16، البعد (LDA,K)
!>          يجب أن يحتوي العمود i على المتجه الذي يحدد
!>          العاكس الأولي H(i), لـ i = 1,2,...,k, كما تم إرجاعه بواسطة
!>          ZGEQRF في الأعمدة k الأولى من وسيط مصفوفته A.
!>          يتم تعديل A بواسطة الروتين ولكن يتم استعادتها عند الخروج.
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرائد للمصفوفة A.
!>          إذا كان SIDE = 'L', LDA >= max(1,M);
!>          إذا كان SIDE = 'R', LDA >= max(1,N).
!>

TAU

!>          TAU هي مصفوفة COMPLEX*16، البعد (K)
!>          يجب أن تحتوي TAU(i) على المعامل القياسي للعامل الأولي
!>          العاكس H(i), كما تم إرجاعه بواسطة ZGEQRF.
!>

C

!>          C هي مصفوفة COMPLEX*16، البعد (LDC,N)
!>          عند الإدخال، المصفوفة C ذات الأبعاد m-by-n.
!>          عند الخروج، يتم استبدال C بـ Q*C أو Q**H*C أو C*Q**H أو C*Q.
!>

LDC

!>          LDC هو INTEGER
!>          البعد الرائد للمصفوفة C. LDC >= max(1,M).
!>

WORK

!>          WORK هي مصفوفة COMPLEX*16، البعد
!>                                   (N) إذا كان SIDE = 'L',
!>                                   (M) إذا كان SIDE = 'R'
!>

INFO

!>          INFO هو INTEGER
!>          = 0: خروج ناجح
!>          < 0: إذا كان INFO = -i، كان الوسيط i لديه قيمة غير قانونية
!>
