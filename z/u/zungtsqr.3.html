توثيق الوظيفة/الإجراء الفرعي

subroutine zungtsqr (integer m, integer n, integer mb, integer nb, complex*16, dimension( lda, * ) a, integer lda, complex*16, dimension( ldt, * ) t, integer ldt, complex*16, dimension( * ) work, integer lwork, integer info)

ZUNGTSQR

الغرض:

!>
!> ZUNGTSQR تولد مصفوفة معقدة M × N ذات أعمدة متعامدة، وهي الأعمدة N الأولى من ناتج ضرب مصفوفات وحدوية معقدة من الرتبة M التي يتم إرجاعها بواسطة ZLATSQR
!>
!>      Q_out = الأعمدة_N_الأولى_من( Q(1)_in * Q(2)_in * ... * Q(k)_in ).
!>
!> راجع التوثيق الخاص بـ ZLATSQR.
!>

المعلمات

M 
!>          M هو عدد صحيح
!>          عدد الصفوف في المصفوفة A.  M >= 0.
!>

N

!>          N هو عدد صحيح
!>          عدد الأعمدة في المصفوفة A. M >= N >= 0.
!>

MB

!>          MB هو عدد صحيح
!>          حجم كتلة الصف المستخدم بواسطة ZLATSQR لإرجاع
!>          المصفوفتين A و T. MB > N.
!>          (لاحظ أنه إذا كان MB > M، فسيتم استخدام M بدلاً من MB
!>          كحجم كتلة الصف).
!>

NB

!>          NB هو عدد صحيح
!>          حجم كتلة العمود المستخدم بواسطة ZLATSQR لإرجاع
!>          المصفوفتين A و T. NB >= 1.
!>          (لاحظ أنه إذا كان NB > N، فسيتم استخدام N بدلاً من NB
!>          كحجم كتلة العمود).
!>

A

!>          A هي مصفوفة COMPLEX*16، البعد (LDA,N)
!>
!>          عند الدخول:
!>
!>             لا يتم الوصول إلى العناصر على القطر و فوقه.
!>             العناصر أسفل القطر تمثل المصفوفة المعينة ذات الشكل شبه المنحرف السفلي المكعب V المحسوبة بواسطة ZLATSQR
!>             التي تحدد المصفوفات المدخلة Q_in(k) (الواحدات على
!>             القطر غير مخزنة) (نفس تنسيق الإخراج A
!>             أسفل القطر في ZLATSQR).
!>
!>          عند الخروج:
!>
!>             تحتوي المصفوفة A على مصفوفة متعامدة M × N هي Q_out،
!>             أي أن أعمدة A هي متجهات وحدوية متعامدة.
!>

LDA

!>          LDA هو عدد صحيح
!>          البعد الرائد للمصفوفة A.  LDA >= max(1,M).
!>

T

!>          T هي مصفوفة COMPLEX*16،
!>          البعد (LDT, N * NIRB)
!>          حيث NIRB = عدد كتل الصفوف المدخلة
!>                     = MAX( 1, CEIL((M-N)/(MB-N)) )
!>          Let NICB = عدد كتل الأعمدة المدخلة
!>                   = CEIL(N/NB)
!>
!>          تعكس المصفوفات المثلثية العليا المستخدمة لتحديد
!>          المصفوفات المدخلة Q_in(k)، k=(1:NIRB*NICB). يتم تخزين
!>          العواكس القطاعية في شكل مضغوط في كتل الأعمدة N × NIRB
!>          الأكبر، وتتكون من NICB كتل أعمدة أصغر مثلثية عليا NB × NB.
!>          (نفس تنسيق الإخراج T في ZLATSQR).
!>

LDT

!>          LDT هو عدد صحيح
!>          البعد الرائد للمصفوفة T.
!>          LDT >= max(1,min(NB1,N)).
!>

WORK

!>          (مساحة عمل) مصفوفة COMPLEX*16، البعد (MAX(2,LWORK))
!>          عند الخروج، إذا كانت INFO = 0، فإن WORK(1) ترجع LWORK المثلى.
!>

LWORK

!>          LWORK هو عدد صحيح
!>          بعد المصفوفة WORK.  LWORK >= (M+NB)*N.
!>          إذا كان LWORK = -1، فيفترض استعلام عن مساحة العمل.
!>          تقوم الإجراءات فقط بحساب الحجم الأمثل لمصفوفة WORK،
!>          وترجع هذا الحجم كإدخال أول في مصفوفة WORK، ولا يصدر
!>          XERBLA أي رسالة خطأ متعلقة بـ LWORK.
!>

INFO

!>          INFO هو عدد صحيح
!>          = 0:  انتهاء ناجح
!>          < 0:  إذا كانت INFO = -i، فإن الوسيط i كان له قيمة غير قانونية
!>
