subroutine zunbdb5 (integer m1, integer m2, integer n, complex*16, dimension(*) x1, integer incx1, complex*16, dimension(*) x2, integer incx2, complex*16, dimension(ldq1,*) q1, integer ldq1, complex*16, dimension(ldq2,*) q2, integer ldq2, complex*16, dimension(*) work, integer lwork, integer info)

ZUNBDB5

الغرض:

!>
!> ZUNBDB5 يقوم بتطبيع المتجه العمودي
!>      X = [ X1 ]
!>          [ X2 ]
!> بالنسبة إلى أعمدة
!>      Q = [ Q1 ] .
!>          [ Q2 ]
!> يجب أن تكون أعمدة Q متعامدة.
!>
!> إذا كان الإسقاط صفرًا وفقًا لمعيار كاهان،
!> فيتم إرجاع متجه آخر من المكمل المتعامد.
!> يتم اختيار هذا المتجه بطريقة اعتباطية ولكن حتمية.
!>
!>

المعلمات

M1 
!>          M1 هو عدد صحيح (INTEGER)
!>           بعد X1 وعدد صفوف Q1. 0 <= M1.
!>

M2

!>          M2 هو عدد صحيح (INTEGER)
!>           بعد X2 وعدد صفوف Q2. 0 <= M2.
!>

N

!>          N هو عدد صحيح (INTEGER)
!>           عدد أعمدة Q1 و Q2. 0 <= N.
!>

X1

!>          X1 هو مصفوفة مركبة (COMPLEX*16)، البعد (M1)
!>           عند الإدخال، الجزء العلوي من المتجه المراد تطبيعه.
!>           عند الخروج، الجزء العلوي من المتجه المسقط.
!>

INCX1

!>          INCX1 هو عدد صحيح (INTEGER)
!>           زيادة لإدخالات X1.
!>

X2

!>          X2 هو مصفوفة مركبة (COMPLEX*16)، البعد (M2)
!>           عند الإدخال، الجزء السفلي من المتجه المراد
!>           تطبيعه. عند الخروج، الجزء السفلي من المتجه المسقط.
!>

INCX2

!>          INCX2 هو عدد صحيح (INTEGER)
!>           زيادة لإدخالات X2.
!>

Q1

!>          Q1 هو مصفوفة مركبة (COMPLEX*16)، البعد (LDQ1, N)
!>           الجزء العلوي من مصفوفة الأساس المتعامدة.
!>

LDQ1

!>          LDQ1 هو عدد صحيح (INTEGER)
!>           البعد الرئيسي لـ Q1. LDQ1 >= M1.
!>

Q2

!>          Q2 هو مصفوفة مركبة (COMPLEX*16)، البعد (LDQ2, N)
!>           الجزء السفلي من مصفوفة الأساس المتعامدة.
!>

LDQ2

!>          LDQ2 هو عدد صحيح (INTEGER)
!>           البعد الرئيسي لـ Q2. LDQ2 >= M2.
!>

WORK

!>          WORK هو مصفوفة مركبة (COMPLEX*16)، البعد (LWORK)
!>

LWORK

!>          LWORK هو عدد صحيح (INTEGER)
!>           بعد المصفوفة WORK. LWORK >= N.
!>

INFO

!>          INFO هو عدد صحيح (INTEGER)
!>           = 0: خروج ناجح.
!>           < 0: إذا كان INFO = -i، فإن المعامل i كان له قيمة غير قانونية.
!>
