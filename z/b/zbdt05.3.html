توثيق الدالة/الروتين الفرعي

subroutine zbdt05 (integer m, integer n, complex*16, dimension( lda, * ) a, integer lda, double precision, dimension( * ) s, integer ns, complex*16, dimension( * ) u, integer ldu, complex*16, dimension( ldvt, * ) vt, integer ldvt, complex*16, dimension( * ) work, double precision resid)

ZBDT05

الغرض:

!>
!> ZBDT05 يعيد بناء مصفوفة ثنائية القطر B من تحليلها التفردي (الجزئي) SVD:
!>    S = U' * B * V
!> حيث U و V مصفوفتان متعامدتان و S مصفوفة قطرية.
!>
!> نسبة الاختبار لاختبار تحليل القيم التفردية هي
!>    RESID = norm( S - U' * B * V ) / ( n * norm(B) * EPS )
!> حيث VT = V' و EPS هو دقة الآلة.
!>

المعلمات

M 
!>          M هو عدد صحيح (INTEGER)
!>          عدد الصفوف في المصفوفتين A و U.
!>

N

!>          N هو عدد صحيح (INTEGER)
!>          عدد الأعمدة في المصفوفتين A و VT.
!>

A

!>          A هي مصفوفة مركبة*16 (COMPLEX*16)، أبعادها (LDA,N)
!>          المصفوفة A ذات الأبعاد m × n.
!>

LDA

!>          LDA هو عدد صحيح (INTEGER)
!>          البعد الرئيسي للمصفوفة A. LDA >= max(1,M).
!>

S

!>          S هي مصفوفة ذات دقة مزدوجة (DOUBLE PRECISION)، أبعادها (NS)
!>          القيم التفردية من تحليل القيم التفردية (الجزئي) للمصفوفة B، مرتبة
!>          بترتيب تنازلي.
!>

NS

!>          NS هو عدد صحيح (INTEGER)
!>          عدد القيم/المتجهات التفردية من تحليل القيم التفردية (الجزئي)
!>          للمصفوفة B.
!>

U

!>          U هي مصفوفة مركبة*16 (COMPLEX*16)، أبعادها (LDU,NS)
!>          المصفوفة المتعامدة U ذات الأبعاد n × ns في S = U' * B * V.
!>

LDU

!>          LDU هو عدد صحيح (INTEGER)
!>          البعد الرئيسي للمصفوفة U. LDU >= max(1,N)
!>

VT

!>          VT هي مصفوفة مركبة*16 (COMPLEX*16)، أبعادها (LDVT,N)
!>          المصفوفة المتعامدة V ذات الأبعاد n × ns في S = U' * B * V.
!>

LDVT

!>          LDVT هو عدد صحيح (INTEGER)
!>          البعد الرئيسي للمصفوفة VT.
!>

WORK

!>          WORK هي مصفوفة مركبة*16 (COMPLEX*16)، أبعادها (M,N)
!>

RESID

!>          RESID هو ذو دقة مزدوجة (DOUBLE PRECISION)
!>          نسبة الاختبار:  norm(S - U' * A * V) / ( n * norm(A) * EPS )
!>
