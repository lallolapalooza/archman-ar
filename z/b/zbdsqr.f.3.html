subroutine zbdsqr (character uplo, integer n, integer ncvt, integer nru, integer ncc, double precision, dimension( * ) d, double precision, dimension( * ) e, complex*16, dimension( ldvt, * ) vt, integer ldvt, complex*16, dimension( ldu, * ) u, integer ldu, complex*16, dimension( ldc, * ) c, integer ldc, double precision, dimension( * ) rwork, integer info)

ZBDSQR

الغرض:

!>
!> ZBDSQR يحسب القيم المفردة، اختيارياً، المتجهات المفردة اليمنى و/أو اليسرى
!> من تحليل القيم المفردة (SVD) لمصفوفة حقيقية N-by-N (علوية أو سفلية) ثنائية القطر B باستخدام خوارزمية QR الضمنية بدون إزاحة.
!> تحليل القيم المفردة لـ B له الشكل
!>
!>    B = Q * S * P**H
!>
!> حيث S هي المصفوفة القطرية للقيم المفردة، Q هي مصفوفة متعامدة للمتجهات المفردة اليسرى،
!> و P هي مصفوفة متعامدة للمتجهات المفردة اليمنى. إذا طُلبت المتجهات المفردة اليسرى،
!> فإن هذا الإجراء الفرعي يعيد في الواقع U*Q بدلاً من Q، وإذا طُلبت المتجهات المفردة اليمنى،
!> فإن هذا الإجراء الفرعي يعيد P**H*VT بدلاً من P**H، لمصفوفات الإدخال المركبة المعطاة U و VT.
!> عندما تكون U و VT هما المصفوفتان الوحدويتان اللتان تقللان مصفوفة عامة A إلى شكل ثنائي القطر: A = U*B*VT،
!> كما تم حسابه بواسطة ZGEBRD، فإن
!>
!>    A = (U*Q) * S * (P**H*VT)
!>
!> هو تحليل القيم المفردة لـ A. اختياريًا، قد يحسب الإجراء الفرعي أيضًا Q**H*C لمصفوفة مركبة C معطاة.
!>
!> انظر  بواسطة J. Demmel و W. Kahan،
!> مذكرة عمل LAPACK #3 (أو SIAM J. Sci. Statist. Comput. vol. 11،
!> no. 5, pp. 873-912, Sept 1990) و
!>  بواسطة
!> B. Parlett و V. Fernando، تقرير فني CPAM-554، قسم الرياضيات،
!> جامعة كاليفورنيا في بيركلي، يوليو 1992
!> لوصف مفصل للخوارزمية.
!>

المعاملات:

UPLO 
!>          UPLO هو CHARACTER*1
!>          = 'U':  B هي ثنائية القطر العلوية؛
!>          = 'L':  B هي ثنائية القطر السفلية.
!>

N

!>          N هو INTEGER
!>          رتبة المصفوفة B.  N >= 0.
!>

NCVT

!>          NCVT هو INTEGER
!>          عدد أعمدة المصفوفة VT. NCVT >= 0.
!>

NRU

!>          NRU هو INTEGER
!>          عدد صفوف المصفوفة U. NRU >= 0.
!>

NCC

!>          NCC هو INTEGER
!>          عدد أعمدة المصفوفة C. NCC >= 0.
!>

D

!>          D هي مصفوفة DOUBLE PRECISION، حجمها (N)
!>          عند الدخول، العناصر القطرية n للمصفوفة ثنائية القطر B.
!>          عند الخروج، إذا كان INFO=0، القيم المفردة لـ B بترتيب تنازلي.
!>

E

!>          E هي مصفوفة DOUBLE PRECISION، حجمها (N-1)
!>          عند الدخول، العناصر خارج القطر n-1 للمصفوفة ثنائية القطر B.
!>          عند الخروج، إذا كان INFO = 0، يتم تدمير E؛ إذا كان INFO > 0، فإن D و E
!>          سيحتويان على العناصر القطرية وشبه القطرية لمصفوفة ثنائية القطر المكافئة متعامداً للمدخلة
!>          المعطاة.
!>

VT

!>          VT هي مصفوفة COMPLEX*16، حجمها (LDVT, NCVT)
!>          عند الدخول، مصفوفة N-by-NCVT VT.
!>          عند الخروج، يتم استبدال VT بـ P**H * VT.
!>          لا يتم الإشارة إليها إذا كان NCVT = 0.
!>

LDVT

!>          LDVT هو INTEGER
!>          البعد الرئيسي للمصفوفة VT.
!>          LDVT >= max(1,N) إذا كان NCVT > 0؛ LDVT >= 1 إذا كان NCVT = 0.
!>

U

!>          U هي مصفوفة COMPLEX*16، حجمها (LDU, N)
!>          عند الدخول، مصفوفة NRU-by-N U.
!>          عند الخروج، يتم استبدال U بـ U * Q.
!>          لا يتم الإشارة إليها إذا كان NRU = 0.
!>

LDU

!>          LDU هو INTEGER
!>          البعد الرئيسي للمصفوفة U.  LDU >= max(1,NRU).
!>

C

!>          C هي مصفوفة COMPLEX*16، حجمها (LDC, NCC)
!>          عند الدخول، مصفوفة N-by-NCC C.
!>          عند الخروج، يتم استبدال C بـ Q**H * C.
!>          لا يتم الإشارة إليها إذا كان NCC = 0.
!>

LDC

!>          LDC هو INTEGER
!>          البعد الرئيسي للمصفوفة C.
!>          LDC >= max(1,N) إذا كان NCC > 0؛ LDC >=1 إذا كان NCC = 0.
!>

RWORK

!>          RWORK هي مصفوفة DOUBLE PRECISION، حجمها (LRWORK)
!>          LRWORK = 4*N، إذا كان NCVT = NRU = NCC = 0، و
!>          LRWORK = 4*(N-1)، بخلاف ذلك
!>

INFO

!>          INFO هو INTEGER
!>          = 0:  الخروج بنجاح
!>          < 0:  إذا كان INFO = -i، فإن المعامل i كان له قيمة غير قانونية
!>          > 0:  لم تتقارب الخوارزمية؛ تحتوي D و E على عناصر مصفوفة ثنائية القطر المكافئة متعامداً للمصفوفة المدخلة B؛
!>                إذا كان INFO = i، فإن i عنصرًا من E لم تتقارب إلى الصفر.
!>

المعاملات الداخلية:

!>  TOLMUL  DOUBLE PRECISION، الافتراضي = max(10,min(100,EPS**(-1/8)))
!>          TOLMUL يتحكم في معيار التقارب لحلقة QR.
!>          إذا كانت موجبة، فإن TOLMUL*EPS هي الدقة النسبية المطلوبة
!>             في القيم المفردة المحسوبة.
!>          إذا كانت سالبة، فإن abs(TOLMUL*EPS*sigma_max) هي الدقة المطلقة المطلوبة
!>             في القيم المفردة المحسوبة (تتوافق مع الدقة النسبية
!>             abs(TOLMUL*EPS) في أكبر قيمة مفردة.
!>          يجب أن تكون قيمة abs(TOLMUL) بين 1 و 1/EPS، ويفضل
!>             أن تكون بين 10 (للتقارب السريع) و .1/EPS
!>             (ليكون هناك بعض الدقة في النتائج).
!>          الافتراضي هو فقدان ثمن رقم أو رقمين عشريين من الأرقام المتاحة
!>             في كل قيمة مفردة محسوبة (أيهما أقل).
!>
!>  MAXITR  INTEGER، الافتراضي = 6
!>          MAXITR يتحكم في الحد الأقصى لعدد تمريرات
!>          الخوارزمية عبر حلقتها الداخلية. تتوقف الخوارزمية
!>          (وبالتالي تفشل في التقارب) إذا تجاوز عدد التمريرات
!>          عبر الحلقة الداخلية MAXITR*N**2.
!>
!>
