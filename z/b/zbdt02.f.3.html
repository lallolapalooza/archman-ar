توثيق الوظيفة/الروتين الفرعي

subroutine zbdt02 (integer m, integer n, complex*16, dimension( ldb, * ) b, integer ldb, complex*16, dimension( ldc, * ) c, integer ldc, complex*16, dimension( ldu, * ) u, integer ldu, complex*16, dimension( * ) work, double precision, dimension( * ) rwork, double precision resid)

ZBDT02

الغرض:

!>
!> ZBDT02 يختبر تغيير الأساس C = U**H * B عن طريق حساب الباقي
!>
!>    RESID = norm(B - U * C) / ( max(m,n) * norm(B) * EPS ),
!>
!> حيث B و C مصفوفتان M × N، و U مصفوفة متعامدة M × M، و EPS هي الدقة الآلية.
!>

المعاملات

M 
!>          M هو عدد صحيح
!>          عدد الصفوف في المصفوفتين B و C وترتيب المصفوفة Q.
!>

N

!>          N هو عدد صحيح
!>          عدد الأعمدة في المصفوفتين B و C.
!>

B

!>          B هي مصفوفة COMPLEX*16، البعد (LDB,N)
!>          المصفوفة B بحجم m × n.
!>

LDB

!>          LDB هو عدد صحيح
!>          البعد الرئيسي للمصفوفة B. LDB >= max(1,M).
!>

C

!>          C هي مصفوفة COMPLEX*16، البعد (LDC,N)
!>          المصفوفة C بحجم m × n، يُفترض أنها تحتوي على U**H * B.
!>

LDC

!>          LDC هو عدد صحيح
!>          البعد الرئيسي للمصفوفة C. LDC >= max(1,M).
!>

U

!>          U هي مصفوفة COMPLEX*16، البعد (LDU,M)
!>          المصفوفة المتعامدة U بحجم m × m.
!>

LDU

!>          LDU هو عدد صحيح
!>          البعد الرئيسي للمصفوفة U. LDU >= max(1,M).
!>

WORK

!>          WORK هي مصفوفة COMPLEX*16، البعد (M)
!>

RWORK

!>          RWORK هي مصفوفة DOUBLE PRECISION، البعد (M)
!>

RESID

!>          RESID هي DOUBLE PRECISION
!>          RESID = norm(B - U * C) / ( max(m,n) * norm(B) * EPS ),
!>
