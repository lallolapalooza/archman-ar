zmq_atomic_counter_new(3)

الحزمة: zeromq

التراخيص: MPL2

ZMQ_ATOMIC_COUNTER_N(3) دليل 0MQ ZMQ_ATOMIC_COUNTER_N(3)

الاسم

zmq_atomic_counter_new - إنشاء عداد ذري جديد

الملخص

void *zmq_atomic_counter_new (void);

الوصف

تقوم الدالة zmq_atomic_counter_new بإنشاء عداد ذري جديد. يمكنك استخدام هذا في التطبيقات متعددة الخيوط للقيام، على سبيل المثال، بحساب المراجع للكائنات المشتركة. حجم العداد الذري هو 32 بت على الأقل. تستخدم هذه الدالة عمليات ذرية خاصة بالمنصة.

قيمة الإرجاع

تقوم الدالة zmq_atomic_counter_new() بإرجاع العداد الذري الجديد في حالة النجاح. بخلاف ذلك، فإنها ترجع NULL.

مثال

رمز اختبار للعدادات الذرية.

void *counter = zmq_atomic_counter_new ();
assert (zmq_atomic_counter_value (counter) == 0);
assert (zmq_atomic_counter_inc (counter) == 0);
assert (zmq_atomic_counter_inc (counter) == 1);
assert (zmq_atomic_counter_inc (counter) == 2);
assert (zmq_atomic_counter_value (counter) == 3);
assert (zmq_atomic_counter_dec (counter) == 1);
assert (zmq_atomic_counter_dec (counter) == 1);
assert (zmq_atomic_counter_dec (counter) == 0);
zmq_atomic_counter_set (counter, 2);
assert (zmq_atomic_counter_dec (counter) == 1);
assert (zmq_atomic_counter_dec (counter) == 0);
zmq_atomic_counter_destroy (&counter);
return 0;
