الحزمة: zeromq

التراخيص: MPL2

king.

ZMQ_POLLERR

بالنسبة لمقابس 0MQ، لا يؤثر هذا العلامة على الدوال zmq_poller_add و zmq_poller_modify، ولا يتم تعيينها أبدًا في عضو events في بنية zmq_poller_event_t. بالنسبة للمقابس القياسية، يتم تمرير هذه العلامة عبر zmq_poller_wait_all إلى استدعاء النظام الأساسي poll() وعمومًا يعني وجود نوع من حالة الخطأ في المقبس المحدد بواسطة fd.

ZMQ_POLLPRI

بالنسبة لمقابس 0MQ، لا يؤثر هذا العلامة على الدوال zmq_poller_add و zmq_poller_modify، ولا يتم تعيينها أبدًا في عضو events في بنية zmq_poller_event_t. بالنسبة للمقابس القياسية، يعني هذا وجود بيانات عاجلة للقراءة. ارجع إلى العلامة POLLPRI لمزيد من المعلومات. بالنسبة لمعرّف الملف، ارجع إلى وثائق نظام التشغيل الخاص بك: على سبيل المثال، يتم الإشارة إلى مقاطعات GPIO من خلال حدث POLLPRI. ليس لهذه العلامة أي تأثير على نظام Windows.

ملاحظة

قد يتم تنفيذ دوال zmq_poller*_ أو محاكاتها باستخدام واجهات نظام تشغيل بخلاف poll()، وبالتالي قد تخضع لحدود تلك الواجهات بطرق غير محددة في هذه الوثائق.

سلامة الخيط

مثل معظم كائنات 0MQ الأخرى، فإن الـ poller ليس آمنًا للخيوط. يجب استدعاء جميع العمليات من نفس الخيط. بخلاف ذلك، يكون السلوك غير محدد.

بالإضافة إلى ذلك، إذا كنت ترغب في إضافة مقبس إلى مثيلات poller متعددة موجودة، فيجب أن يكون المقبس نفسه آمنًا للخيوط (خادم، عميل، ...). بخلاف ذلك، يكون السلوك غير محدد.

قيمة الإرجاع

zmq_poller_new تُرجع مؤشرًا صالحًا إلى poller، أو NULL في حالة الفشل.

جميع الدوال التي تُرجع عددًا صحيحًا، تُرجع -1 في حالة الفشل. في هذه الحالة، يمكن استخدام zmq_errno() للاستعلام عن نوع الخطأ كما هو موضح أدناه.

zmq_poller_wait_all تُرجع عدد الأحداث التي تم الإشارة إليها وإرجاعها في مصفوفة الأحداث. لا تُرجع أبدًا 0.

جميع الدوال الأخرى تُرجع 0 في حالة التنفيذ الناجح.

الأخطاء

على zmq_poller_new:

ENOMEM

لم يتم تخصيص poller جديد بنجاح.

على zmq_poller_destroy:

EFAULT

لم يكن poller_p يشير إلى poller صالح. لاحظ أن تمرير مؤشر غير صالح (مثل مؤشر إلى ذاكرة تم تحريرها) قد يتسبب في سلوك غير محدد (مثل انتهاك الوصول).

على zmq_poller_size:

EFAULT

لم يكن poller يشير إلى poller صالح. لاحظ أن تمرير مؤشر غير صالح (مثل مؤشر إلى ذاكرة تم تحريرها) قد يتسبب في سلوك غير محدد (مثل انتهاك الوصول).

على zmq_poller_add، zmq_poller_modify و zmq_poller_remove:

EFAULT

لم يكن poller يشير إلى poller صالح. لاحظ أن تمرير مؤشر غير صالح (مثل مؤشر إلى ذاكرة تم تحريرها) قد يتسبب في سلوك غير محدد (مثل انتهاك الوصول).

ENOTSOCK

لم يكن socket يشير إلى مقبس صالح. لاحظ أن تمرير مؤشر غير صالح (مثل مؤشر إلى ذاكرة تم تحريرها) قد يتسبب في سلوك غير محدد (مثل انتهاك الوصول).

على zmq_poller_add أو zmq_poller_add_fd:

ENOMEM

لم يتم تخصيص الموارد اللازمة.

EINVAL

تم تسجيل socket أو fd بالفعل مع الـ poller.

على zmq_poller_modify، zmq_poller_modify_fd، zmq_poller_remove أو zmq_poller_remove_fd:

EINVAL

لم يتم تسجيل socket أو fd مع الـ poller.

على zmq_poller_add_fd، zmq_poller_modify_fd و zmq_poller_remove_fd:

EBADF

كان fd المحدد هو الـ fd المتقاعد.

على zmq_poller_wait و zmq_poller_wait_all:

ENOMEM

لم يتم تخصيص الموارد اللازمة.

ETERM

واحد على الأقل من الكائنات المسجلة هو socket تم إنهاء السياق 0MQ المرتبط به.

EFAULT

كان events المقدم NULL، أو لم يكن poller يشير إلى poller صالح، أو لا توجد كائنات مسجلة أو تم تعطيل جميع اشتراكات الأحداث وكان timeout سالبًا.

EINTR

تم مقاطعة العملية بواسطة تسليم إشارة قبل توفر أي أحداث.

EAGAIN

لم يتم الإشارة إلى أي حدث مسجل قبل انتهاء المهلة.

على zmq_poller_fd:

EINVAL

ليس لدى الـ poller معرّف ملف مرتبط.

EFAULT

لم يكن poller المقدم يشير إلى poller صالح.

مثال

الانتظار بلا نهاية لأحداث الإدخال على كل من مقبس 0MQ ومقبس قياسي..

void *poller = zmq_poller_new ();
/* العنصر الأول يشير إلى مقبس 0MQ 'socket' */
zmq_poller_add (poller, socket, NULL, ZMQ_POLLIN);
/* العنصر الثاني يشير إلى المقبس القياسي 'fd' */
zmq_poller_add_fd (poller, fd, NULL, ZMQ_POLLIN);
zmq_poller_event_t events [2];
/* انتظر الأحداث بلا نهاية */
int rc = zmq_poller_wait_all (poller, events, 2, -1);
assert (rc >= 0);
/* سيتم تخزين الأحداث المرتجعة في 'events' */
for (int i = 0; i < 2; ++i) {
    if (events[i].socket == socket && events[i].events & ZMQ_POLLIN) {
        // ...
    } else if (events[i].fd == fd && events[i].events & ZMQ_POLLIN)) {
        // ...
    }
}
zmq_poller_destroy (&poller);
