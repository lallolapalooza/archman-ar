subroutine zgelqt (integer m, integer n, integer mb, complex*16, dimension( lda, * ) a, integer lda, complex*16, dimension( ldt, * ) t, integer ldt, complex*16, dimension( * ) work, integer info)

ZGELQT

الغرض:

!>
!> ZGELQT تحسب تحليل LQ مسدود لمصفوفة معقدة M-by-N A
!> باستخدام التمثيل المدمج WY لـ Q.
!>

المعاملات

M 
!>          M هو عدد صحيح
!>          عدد صفوف المصفوفة A. M >= 0.
!>

N

!>          N هو عدد صحيح
!>          عدد أعمدة المصفوفة A. N >= 0.
!>

MB

!>          MB هو عدد صحيح
!>          حجم الكتلة المستخدم في QR المسدود. MIN(M,N) >= MB >= 1.
!>

A

!>          A هو مصفوفة COMPLEX*16، البعد (LDA,N)
!>          عند الإدخال، مصفوفة M-by-N A.
!>          عند الخروج، تحتوي العناصر على القطر وعلى أسفله على المصفوفة شبه المنحرفة السفلية L (L
!>          مثلثة سفلية إذا كان M <= N)؛ العناصر فوق القطر هي صفوف V.
!>

LDA

!>          LDA هو عدد صحيح
!>          البعد الأمامي للمصفوفة A. LDA >= max(1,M).
!>

T

!>          T هو مصفوفة COMPLEX*16، البعد (LDT,MIN(M,N))
!>          عوامل الانعكاس الكتلية المثلثية العلوية المخزنة بالشكل المدمج
!>          كسلسلة من الكتل المثلثية العلوية. انظر أدناه
!>          لمزيد من التفاصيل.
!>

LDT

!>          LDT هو عدد صحيح
!>          البعد الأمامي للمصفوفة T. LDT >= MB.
!>

WORK

!>          WORK هي مصفوفة COMPLEX*16، البعد (MB*N)
!>

INFO

!>          INFO هو عدد صحيح
!>          = 0: خروج ناجح
!>          < 0: إذا كان INFO = -i، فإن المعامل i كان له قيمة غير قانونية
!>

تفاصيل إضافية:

!>
!>  تخزن المصفوفة V عوامل الانعكاس الأولية H(i) في الصف i
!>  فوق القطر. على سبيل المثال، إذا كان M=5 و N=3، فإن المصفوفة V هي
!>
!>               V = (  1  v1 v1 v1 v1 )
!>                   (     1  v2 v2 v2 )
!>                   (         1 v3 v3 )
!>
!>
!>  حيث تمثل الـ vi المتجهات التي تحدد H(i)، والتي يتم إرجاعها
!>  في المصفوفة A. الـ 1s على طول قطر V لا يتم تخزينها في A.
!>  لتكن K = MIN(M,N). عدد الكتل هو B = سقف(K/MB)، حيث كل
!>  كتلة من الرتبة MB باستثناء الكتلة الأخيرة، والتي تكون من الرتبة
!>  IB = K - (B-1)*MB. لكل كتلة من الكتل B، يتم حساب عامل
!>  انعكاس كتلة مثلثية علوية: T1, T2, ..., TB. يتم تخزين T's
!>  ذات الحجم MB-by-MB (و IB-by-IB للكتلة الأخيرة) في المصفوفة T
!>  ذات الحجم MB-by-K كـ
!>
!>               T = (T1 T2 ... TB).
!>
