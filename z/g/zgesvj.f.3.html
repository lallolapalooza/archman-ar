توثيق الوظيفة/الإجراء الفرعي

subroutine zgesvj (character*1 joba, character*1 jobu, character*1 jobv, integer m, integer n, complex*16, dimension( lda, * ) a, integer lda, double precision, dimension( n ) sva, integer mv, complex*16, dimension( ldv, * ) v, integer ldv, complex*16, dimension( lwork ) cwork, integer lwork, double precision, dimension( lrwork ) rwork, integer lrwork, integer info)

ZGESVJ

الغرض:

!>
!> ZGESVJ يحسب تحليل القيم المفردة (SVD) للمصفوفة المعقدة
!> A ذات الأبعاد M × N، حيث M >= N. يكتب تحليل القيم المفردة لـ A بالشكل
!>                                    [++]   [xx]   [x0]   [xx]
!>              A = U * SIGMA * V^*,  [++] = [xx] * [ox] * [xx]
!>                                    [++]   [xx]
!> حيث SIGMA مصفوفة قطرية بحجم N × N، U مصفوفة متعامدة بأبعاد M × N،
!> و V مصفوفة وحدوية بحجم N × N. العناصر القطرية لـ SIGMA هي القيم المفردة لـ A.
!> الأعمدة لـ U و V هي المتجهات المفردة اليسرى واليمنى لـ A على التوالي.
!>

المعلمات:

JOBA 
!>          JOBA هو CHARACTER*1
!>          يحدد بنية المصفوفة A.
!>          = 'L': المصفوفة المدخلة A هي مصفوفة مثلثية سفلية؛
!>          = 'U': المصفوفة المدخلة A هي مصفوفة مثلثية علوية؛
!>          = 'G': المصفوفة المدخلة A هي مصفوفة عامة بأبعاد M × N، M >= N.
!>

JOBU

!>          JOBU هو CHARACTER*1
!>          يحدد ما إذا كان سيتم حساب المتجهات المفردة اليسرى
!>          (أعمدة U):
!>          = 'U' أو 'F': يتم حساب المتجهات المفردة اليسرى المقابلة للقيم
!>                 المفردة غير الصفرية وإرجاعها في الأعمدة الرائدة لـ A.
!>                 انظر مزيدًا من التفاصيل في وصف A.
!>                 يتم تعيين حد التعامد العددي الافتراضي إلى
!>                 تقريبًا TOL=CTOL*EPS، CTOL=SQRT(M)، EPS=DLAMCH('E').
!>          = 'C': على غرار JOBU='U'، باستثناء أن المستخدم يمكنه التحكم في
!>                 مستوى التعامد العددي للمتجهات المفردة اليسرى المحسوبة.
!>                 يمكن تعيين TOL إلى TOL = CTOL*EPS، حيث
!>                 يتم إعطاء CTOL عند الإدخال في الصفيف WORK.
!>                 لا يُسمح بأي CTOL أصغر من ONE. CTOL الأكبر
!>                 من 1 / EPS لا معنى له. يمكن استخدام الخيار 'C'
!>                 إذا كانت M*EPS كافية للتعامد
!>                 للمتجهات المفردة اليسرى المحسوبة، بحيث يمكن لـ CTOL=M
!>                 توفير بضع دورات من دورانات يعقوبي.
!>                 انظر وصف A و WORK(1).
!>          = 'N': المصفوفة U غير محسوبة. ومع ذلك، انظر
!>                 وصف A.
!>

JOBV

!>          JOBV هو CHARACTER*1
!>          يحدد ما إذا كان سيتم حساب المتجهات المفردة اليمنى، أي
!>          المصفوفة V:
!>          = 'V' أو 'J': يتم حساب المصفوفة V وإرجاعها في الصفيف V
!>          = 'A': يتم تطبيق دورانات يعقوبي على الصفيف MV × N
!>                 الموجود في V. بعبارة أخرى، المصفوفة المفردة اليمنى
!>                 V غير محسوبة بشكل صريح؛ بدلاً من ذلك يتم
!>                 تطبيقها على مصفوفة MV × N مخزنة مبدئيًا في
!>                 الأول MV صفوف V.
!>          = 'N': المصفوفة V غير محسوبة ولا يتم الرجوع إلى الصفيف V
!>

M

!>          M هو INTEGER
!>          عدد صفوف المصفوفة المدخلة A. 1/DLAMCH('E') > M >= 0.
!>

N

!>          N هو INTEGER
!>          عدد أعمدة المصفوفة المدخلة A.
!>          M >= N >= 0.
!>

A

!>          A هو صفيف COMPLEX*16، البعد (LDA,N)
!>          عند الإدخال، المصفوفة A بأبعاد M × N.
!>          عند الخروج،
!>          إذا كان JOBU = 'U' .OR. JOBU = 'C':
!>                 إذا كان INFO = 0:
!>                 يتم إرجاع أعمدة U المتعامدة للرتبة RANKA
!>                 في الأعمدة الرائدة للرتبة RANKA للمصفوفة A. هنا RANKA <= N
!>                 هي عدد القيم المفردة المحسوبة لـ A التي
!>                 تزيد عن عتبة التدفق السفلي DLAMCH('S'). المتجهات المفردة
!>                 المقابلة للقيم المفردة التي بها تدفق سفلي أو صفر
!>                 غير محسوبة. يتم إرجاع قيمة RANKA
!>                 في الصفيف RWORK مثل RANKA=NINT(RWORK(2)). انظر أيضًا
!>                 أوصاف SVA و RWORK. أعمدة U المحسوبة
!>                 متعامدة رقميًا بشكل متبادل تقريبًا
!>                 حتى TOL=SQRT(M)*EPS (افتراضي)؛ أو TOL=CTOL*EPS (JOBU = 'C')،
!>                 انظر وصف JOBU.
!>                 إذا كان INFO > 0،
!>                 لم تتقارب الإجراء ZGESVJ في العدد المحدد
!>                 من التكرارات (الدورات). في هذه الحالة، قد لا تكون
!>                 الأعمدة المحسوبة لـ U متعامدة حتى TOL. الإخراج
!>                 U (المخزن في A)، SIGMA (المعطاة بواسطة القيم المفردة المحسوبة
!>                 في SVA(1:N)) و V لا تزال تحليلًا للمصفوفة
!>                 المدخلة A بالمعنى الذي يكون فيه الباقي
!>                 || A - SCALE * U * SIGMA * V^* ||_2 / ||A||_2 صغيرًا.
!>          إذا كان JOBU = 'N':
!>                 إذا كان INFO = 0:
!>                 لاحظ أن المتجهات المفردة اليسرى "مجانية" في
!>                 خوارزمية SVD يعقوبي أحادية الجانب. ومع ذلك، إذا كانت
!>                 القيم المفردة فقط مطلوبة، فإن مستوى التعامد العددي
!>                 لـ U ليس مشكلة وتتوقف التكرارات
!>                 عندما تكون أعمدة المصفوفة المتكررة
!>                 متعامدة رقميًا تقريبًا حتى M*EPS. وبالتالي،
!>                 عند الخروج، تحتوي A على أعمدة U مضروبة في
!>                 القيم المفردة المقابلة.
!>                 إذا كان INFO > 0:
!>                 لم تتقارب الإجراء ZGESVJ في العدد المحدد
!>                 من التكرارات (الدورات).
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرئيسي للصفيف A. LDA >= max(1,M).
!>

SVA

!>          SVA هو صفيف DOUBLE PRECISION، البعد (N)
!>          عند الخروج،
!>          إذا كان INFO = 0:
!>          اعتمادًا على قيمة SCALE = RWORK(1)، لدينا:
!>                 إذا كان SCALE = ONE:
!>                 تحتوي SVA(1:N) على القيم المفردة المحسوبة لـ A.
!>                 خلال الحساب، تحتوي SVA على معايير الأعمدة الإقليدية
!>                 للمصفوفات المتكررة في المصفوفة A.
!>                 إذا كان SCALE != ONE:
!>                 القيم المفردة لـ A هي SCALE*SVA(1:N)، وهذا
!>                 التمثيل المفكك يرجع إلى حقيقة أن بعض
!>                 القيم المفردة لـ A قد تحدث فيها تدفق سفلي أو فائض.
!>
!>          إذا كان INFO > 0:
!>          لم تتقارب الإجراء ZGESVJ في العدد المحدد
!>          من التكرارات (الدورات) وقد لا تكون SCALE*SVA(1:N) دقيقة.
!>

MV

!>          MV هو INTEGER
!>          إذا كان JOBV = 'A'، فإن حاصل ضرب دورانات يعقوبي في ZGESVJ
!>          يتم تطبيقه على أول MV صفوف من V. انظر وصف JOBV.
!>

V

!>          V هو صفيف COMPLEX*16، البعد (LDV,N)
!>          إذا كان JOBV = 'V'، فإن V تحتوي عند الخروج على المصفوفة N × N لـ
!>                         المتجهات المفردة اليمنى؛
!>          إذا كان JOBV = 'A'، فإن V تحتوي على حاصل ضرب المصفوفة المفردة اليمنى
!>                         المحسوبة والمصفوفة الأولية في
!>                         الصفيف V.
!>          إذا كان JOBV = 'N'، فلا يتم الرجوع إلى V.
!>

LDV

!>          LDV هو INTEGER
!>          البعد الرئيسي للصفيف V. LDA >= max(1,N).
!>

WORK

!>          WORK هو صفيف COMPLEX*16، البعد (LWORK)
!>          عند الإدخال، إذا كان JOBU = 'C'، فإن WORK(1) يحتوي على CTOL.
!>          عند الخروج، يتم استخدام WORK لتخزين أي معلومات إضافية
!>          (غير مستخدمة في هذا التنفيذ).
!>

LWORK

!>          LWORK هو INTEGER
!>          طول مساحة العمل. LWORK >= N.
!>

RWORK

!>          RWORK هو صفيف DOUBLE PRECISION، البعد (LRWORK)
!>          عند الخروج،
!>          RWORK(1) يحتوي على SCALE، وهو عامل قياس يستخدم لإدارة
!>          احتمالية تدفق القيم المفردة السفلي أو الفائض.
!>          RWORK(2) يحتوي على RANKA، وهو عدد القيم المفردة المحسوبة
!>          التي تزيد عن عتبة التدفق السفلي.
!>

LRWORK

!>          LRWORK هو INTEGER
!>          طول مساحة العمل. LRWORK >= 2.
!>

INFO

!>          INFO هو INTEGER
!>          = 0: تم التنفيذ بنجاح.
!>          > 0: لم يتقارب الإجراء ZGESVJ في العدد المحدد من التكرارات (الدورات).
!>

ملاحظات:

يستخدم الإجراء الفرعي ZGESVJ خوارزمية يعقوبي أحادية الجانب لحساب تحليل القيم المفردة (SVD) لمصفوفة معقدة.

يتم استخدام دورانات يعقوبي بشكل متكرر لتصفير العناصر خارج القطر للمصفوفة، مما يؤدي إلى تقارب أعمدة المصفوفة لتصبح متعامدة، بينما تقترب القيم القطرية من القيم المفردة.

التعقيد الزمني تقريبًا O(N*M*min(M,N)*log(1/EPS)) دورات.
