وثائق الوظيفة/الروتين الفرعي

subroutine zgeqls (integer m, integer n, integer nrhs, complex*16, dimension( lda, * ) a, integer lda, complex*16, dimension( * ) tau, complex*16, dimension( ldb, * ) b, integer ldb, complex*16, dimension( lwork ) work, integer lwork, integer info)

ZGEQLS

الغرض:

!>
!> حل مشكلة المربعات الصغرى
!>     min || A*X - B ||
!> باستخدام التحلل QL
!>     A = Q*L
!> الذي تم حسابه بواسطة ZGEQLF.
!>

المعلمات

M 
!>          M هو عدد صحيح
!>          عدد الصفوف في المصفوفة A.  M >= 0.
!>

N

!>          N هو عدد صحيح
!>          عدد الأعمدة في المصفوفة A.  M >= N >= 0.
!>

NRHS

!>          NRHS هو عدد صحيح
!>          عدد الأعمدة في B.  NRHS >= 0.
!>

A

!>          A هي مصفوفة COMPLEX*16، البعد (LDA,N)
!>          تفاصيل تحلل QL للمصفوفة الأصلية A كما
!>          تم إرجاعها بواسطة ZGEQLF.
!>

LDA

!>          LDA هو عدد صحيح
!>          البعد الرائد للمصفوفة A.  LDA >= M.
!>

TAU

!>          TAU هي مصفوفة COMPLEX*16، البعد (N)
!>          تفاصيل المصفوفة المتعامدة Q.
!>

B

!>          B هي مصفوفة COMPLEX*16، البعد (LDB,NRHS)
!>          عند الإدخال، مصفوفة الطرف الأيمن B بحجم m-by-nrhs.
!>          عند الخروج، مصفوفة الحل X بحجم n-by-nrhs، مخزنة في الصفوف
!>          m-n+1:m.
!>

LDB

!>          LDB هو عدد صحيح
!>          البعد الرائد للمصفوفة B. LDB >= M.
!>

WORK

!>          WORK هي مصفوفة COMPLEX*16، البعد (LWORK)
!>

LWORK

!>          LWORK هو عدد صحيح
!>          طول المصفوفة WORK.  يجب أن تكون LWORK على الأقل NRHS،
!>          ويجب أن تكون على الأقل NRHS*NB، حيث NB هو حجم الكتلة
!>          لهذه البيئة.
!>

INFO

!>          INFO هو عدد صحيح
!>          = 0: خروج ناجح
!>          < 0: إذا كان INFO = -i، فإن الوسيط i كان له قيمة غير قانونية
!>
