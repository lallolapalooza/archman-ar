subroutine zgejsv (character*1 joba, character*1 jobu, character*1 jobv, character*1 jobr, character*1 jobt, character*1 jobp, integer m, integer n, complex*16, dimension( lda, * ) a, integer lda, double precision, dimension( n ) sva, complex*16, dimension( ldu, * ) u, integer ldu, complex*16, dimension( ldv, * ) v, integer ldv, complex*16, dimension( lwork ) cwork, integer lwork, double precision, dimension( lrwork ) rwork, integer lrwork, integer, dimension( * ) iwork, integer info)

ZGEJSV

الغرض:

!>
!> ZGEJSV تحسب تفكيك القيم المفردة (SVD) لمصفوفة معقدة M-by-N [A]، حيث M >= N. يُكتب تفكيك القيم المفردة لـ [A] كـ
!>
!>              [A] = [U] * [SIGMA] * [V]^*,
!>
!> حيث [SIGMA] هي مصفوفة N-by-N (M-by-N) وهي صفرية باستثناء عناصرها القطرية N، و[U] هي مصفوفة وحدوية M-by-N (أو M-by-M)، و
!> [V] هي مصفوفة وحدوية N-by-N. العناصر القطرية لـ [SIGMA] هي
!> القيم المفردة لـ [A]. أعمدة [U] و[V] هي المتجهات المفردة اليسرى واليمنى
!> لـ [A] على التوالي. يتم حساب المصفوفتين [U] و[V] وتخزينهما في المصفوفات U وV على التوالي. يتم حساب القطر لـ [SIGMA] وتخزينه في المصفوفة SVA.
!>

المعلمات

JOBA 
!>          JOBA هو CHARACTER*1
!>         يحدد مستوى الدقة:
!>       = 'C': يعمل هذا الخيار جيدًا (دقة نسبية عالية) إذا كان A = B * D،
!>              مع B جيدة الحالة ومصفوفة قطرية D عشوائية.
!>              لا يمكن أن تتلف الدقة عن طريق التحجيم العمودي (COLUMN scaling).
!>              تعتمد دقة المخرجات المحسوبة على حالة
!>              B، والإجراء يهدف إلى أفضل دقة نظرية.
!>              الخطأ النسبي max_{i=1:N}|d sigma_i| / sigma_i
!>              محدود بـ f(M,N)*epsilon* cond(B)، مستقل عن D.
!>              تتم معالجة المصفوفة المدخلة مسبقًا باستخدام QRF مع
!>              تجميع الأعمدة. هذه المعالجة المسبقة الأولية والتهيئة
!>              بواسطة تحليل QR كاشف الرتبة أمر شائع لجميع قيم
!>              JOBA. يتم تحديد الإجراءات الإضافية كما يلي:
!>       = 'E': حساب كما هو الحال مع 'C' مع تقدير إضافي لـ
!>              عدد حالات B. يوفر حدود خطأ واقعية.
!>       = 'F': إذا كان A = D1 * C * D2 مع حالات تحجيم قطرية غير جيدة
!>              D1، D2، ومصفوفة C جيدة الحالة، فإن هذا الخيار يعطي
!>              دقة أعلى من الخيار 'C'. إذا كان هيكل
!>              المصفوفة المدخلة غير معروف، والدقة النسبية
!>              مرغوبة، فإن هذا الخيار موصى به. تتم معالجة المصفوفة المدخلة A
!>              مسبقًا باستخدام تحليل QR مع تجميع كامل (الصف والعمود)
!>              .
!>       = 'G': حساب كما هو الحال مع 'F' مع تقدير إضافي لـ
!>              عدد حالات B، حيث A=B*D. إذا كانت A تحتوي على صفوف ذات أوزان ثقيلة،
!>              فإن استخدام عدد الحالات هذا يعطي حدود خطأ متفائلة للغاية.
!>       = 'A': القيم المفردة الصغيرة لا يتم تحديدها جيدًا بواسطة البيانات
!>              وتعتبر ضوضاء؛ تُعامل المصفوفة على أنها
!>              ذات رتبة ناقصة رقميًا. الخطأ في
!>              القيم المفردة المحسوبة محدود بـ f(m,n)*epsilon*||A||.
!>              يقوم تفكيك SVD المحسوب A = U * S * V^* باستعادة A حتى
!>              f(m,n)*epsilon*||A||.
!>              هذا يمنح الإجراء الترخيص للتخلص (تعيين إلى صفر)
!>              جميع القيم المفردة الأقل من N*epsilon*||A||.
!>       = 'R': مشابه لـ 'A'. يتم استخدام خاصية الكشف عن الرتبة لتحليل
!>              QR الأولي للكشف (باستخدام العامل المثلثي)
!>              عن فجوة sigma_{r+1} < epsilon * sigma_r في هذه الحالة يتم
!>              إعلان الرتبة الرقمية بأنها r. يتم حساب SVD مع
!>              حدود خطأ مطلقة، ولكن بدقة أكبر من 'A'.
!>

JOBU

!>          JOBU هو CHARACTER*1
!>         يحدد ما إذا كان سيتم حساب أعمدة U:
!>       = 'U': يتم إرجاع N أعمدة من U في المصفوفة U.
!>       = 'F': يتم إرجاع المجموعة الكاملة من M المتجهات المفردة اليسرى في المصفوفة U.
!>       = 'W': يمكن استخدام U كمساحة عمل بطول M*N. انظر وصف
!>              U.
!>       = 'N': لم يتم حساب U.
!>

JOBV

!>          JOBV هو CHARACTER*1
!>         يحدد ما إذا كان سيتم حساب المصفوفة V:
!>       = 'V': يتم إرجاع N أعمدة من V في المصفوفة V؛ لا يتم تجميع دوران جاكوبي صراحةً.
!>       = 'J': يتم إرجاع N أعمدة من V في المصفوفة V، ولكن يتم
!>              حسابها كناتج لدورانات جاكوبي، إذا كان JOBT = 'N'.
!>       = 'W': يمكن استخدام V كمساحة عمل بطول N*N. انظر وصف
!>              V.
!>       = 'N': لم يتم حساب V.
!>

JOBR

!>          JOBR هو CHARACTER*1
!>         يحدد النطاق (RANGE) للقيم المفردة. يمنح الترخيص لـ
!>         تعيين القيم المفردة الموجبة الصغيرة إلى صفر إذا كانت خارج
!>         النطاق المحدد. إذا تم تحجيم A .NE. 0 بحيث تكون أكبر قيمة مفردة
!>         لـ c*A حول SQRT(BIG)، BIG=DLAMCH('O')، فإن JOBR يمنح
!>         الترخيص لإلغاء أعمدة A التي يكون معيارها في c*A أقل من
!>         SQRT(SFMIN) (لـ JOBR = 'R')، أو أقل من SMALL=SFMIN/EPSLN،
!>         حيث SFMIN=DLAMCH('S')، EPSLN=DLAMCH('E').
!>       = 'N': لا تلغِ الأعمدة الصغيرة لـ c*A. يفترض هذا الخيار أن
!>              BLAS وتحليلات QR وحلول المثلثية
!>              مُنفذة للعمل في هذا النطاق. إذا كان شرط A
!>              أكبر من BIG، استخدم ZGESVJ.
!>       = 'R': نطاق مقيد (RESTRICTED) لـ sigma(c*A) هو [SQRT(SFMIN)، SQRT(BIG)]
!>              (تقريبًا، كما هو موضح أعلاه). هذا الخيار موصى به.
!>                                             ===========================
!>         لحساب القيم المفردة في النطاق الكامل [SFMIN، BIG]
!>         استخدم ZGESVJ.
!>

JOBT

!>          JOBT هو CHARACTER*1
!>         إذا كانت المصفوفة مربعة، فقد يقرر الإجراء استخدام
!>         منقولها A^* إذا بدا أن A^* أفضل من حيث التقارب.
!>         إذا لم تكن المصفوفة مربعة، يتم تجاهل JOBT.
!>         يستند القرار إلى قيمتين للانتروبيا على مدار المدار المرافق لـ
!>         A^* * A. انظر وصف RWORK(6) و RWORK(7).
!>       = 'T': استخدم المنقول إذا أشارت اختبارات الانتروبيا إلى احتمال تقارب أسرع
!>         لعملية جاكوبي إذا تم أخذ A^* كمدخل. إذا تم
!>         استبدال A بـ A^*، فسيتم تضمين تجميع الصفوف تلقائيًا.
!>       = 'N': لا تخمن.
!>         يمكن استخدام الخيار 'T' لحساب القيم المفردة فقط، أو
!>         SVD الكامل (U، SIGMA، و V). لمجموعة واحدة فقط من المتجهات المفردة
!>         (U أو V)، يجب على المتصل توفير كل من U و V، حيث يتم استخدام إحدى
!>         المصفوفتين كمساحة عمل إذا تم تبديل المصفوفة A.
!>         يمكن للمنفذ إزالة هذا القيد بسهولة وجعل
!>         الكود أكثر تعقيدًا. انظر وصف U و V.
!>         بشكل عام، يعتبر هذا الخيار تجريبيًا، ويجب
!>         تفضيل 'N'. هذا عرضة للتغيير في المستقبل.
!>

JOBP

!>          JOBP هو CHARACTER*1
!>         يمنح الترخيص لإدخال اضطرابات منظمة لإغراق
!>         الأرقام غير المعيارية (denormalized numbers). يجب أن يكون هذا الترخيص نشطًا إذا كانت
!>         الأرقام غير المعيارية
