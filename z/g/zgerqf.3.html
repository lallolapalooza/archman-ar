subroutine zgerqf (integer m, integer n, complex*16, dimension( lda, * ) a, integer lda, complex*16, dimension( * ) tau, complex*16, dimension( * ) work, integer lwork, integer info)

ZGERQF

الغرض:

!>
!> ZGERQF يحسب تحليل RQ لمصفوفة A المعقدة M-by-N:
!> A = R * Q.
!>

المعلمات

M 
!>          M هو عدد صحيح (INTEGER)
!>          عدد صفوف المصفوفة A.  M >= 0.
!>

N

!>          N هو عدد صحيح (INTEGER)
!>          عدد أعمدة المصفوفة A.  N >= 0.
!>

A

!>          A هي مصفوفة COMPLEX*16، البعد (LDA,N)
!>          عند الإدخال، المصفوفة A بحجم M-by-N.
!>          عند الخروج:
!>          إذا كان m <= n، فإن المثلث العلوي للمصفوفة الفرعية
!>          A(1:m,n-m+1:n) يحتوي على المصفوفة المثلثية العليا R بحجم M-by-M.
!>          إذا كان m >= n، فإن العناصر على وفوق القطر الفرعي (m-n)
!>          تحتوي على المصفوفة شبه المنحرفة العليا R بحجم M-by-N.
!>          العناصر المتبقية، مع المصفوفة TAU، تمثل
!>          المصفوفة الوحدوية Q كناتج لـ min(m,n) عواكس أولية
!>          (انظر مزيد من التفاصيل).
!>

LDA

!>          LDA هو عدد صحيح (INTEGER)
!>          البعد الرائد للمصفوفة A.  LDA >= max(1,M).
!>

TAU

!>          TAU هي مصفوفة COMPLEX*16، البعد (min(M,N))
!>          العوامل القياسية للعواكس الأولية (انظر مزيد من التفاصيل).
!>

WORK

!>          WORK هي مصفوفة COMPLEX*16، البعد (MAX(1,LWORK))
!>          عند الخروج، إذا كان INFO = 0، فإن WORK(1) يعيد LWORK المثلى.
!>

LWORK

!>          LWORK هو عدد صحيح (INTEGER)
!>          بعد مصفوفة WORK.
!>          LWORK >= 1، إذا كان MIN(M,N) = 0، و LWORK >= M، وإلا.
!>          للحصول على أفضل أداء، LWORK >= M*NB، حيث NB
!>          هو حجم الكتلة الأمثل.
!>
!>          إذا كان LWORK = -1، فيفترض أنه استعلام مساحة العمل؛ الوحدة الفرعية
!>          تحسب فقط الحجم الأمثل للمصفوفة WORK، وتعيد
!>          هذه القيمة كإدخال أول للمصفوفة WORK، ولا يتم إصدار أي رسالة خطأ
!>          متعلقة بـ LWORK بواسطة XERBLA.
!>

INFO

!>          INFO هو عدد صحيح (INTEGER)
!>          = 0:  انتهاء ناجح
!>          < 0:  إذا كان INFO = -i، فإن المعامل i كان له قيمة غير قانونية
!>

مزيد من التفاصيل:

!>
!>  يتم تمثيل المصفوفة Q كناتج للعواكس الأولية
!>
!>     Q = H(1)**H H(2)**H . . . H(k)**H, حيث k = min(m,n).
!>
!>  كل H(i) له الشكل
!>
!>     H(i) = I - tau * v * v**H
!>
!>  حيث tau هو عامل قياسي معقد، و v هو متجه معقد مع
!>  v(n-k+i+1:n) = 0 و v(n-k+i) = 1؛ conjg(v(1:n-k+i-1)) مخزن
!>  عند الخروج في A(m-k+i,1:n-k+i-1)، و tau في TAU(i).
!>
