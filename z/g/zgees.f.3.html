توثيق الوظيفة/الروتين الفرعي

subroutine zgees (character jobvs, character sort, external select, integer n, complex*16, dimension( lda, * ) a, integer lda, integer sdim, complex*16, dimension( * ) w, complex*16, dimension( ldvs, * ) vs, integer ldvs, complex*16, dimension( * ) work, integer lwork, double precision, dimension( * ) rwork, logical, dimension( * ) bwork, integer info)

ZGEES يحسب القيم الذاتية، الشكل العودي، وبشكل اختياري، مصفوفة المتجهات العودية لمصفوفات GE

الغرض:

!>
!> ZGEES يحسب لمصفوفة معقدة غير متناظرة بحجم N*N وهي A، القيم الذاتية، الشكل العودي T، وبشكل اختياري، مصفوفة المتجهات العودية Z. هذا يعطي التحليل العودي A = Z*T*(Z**H).
!>
!> بشكل اختياري، فإنه يرتب القيم الذاتية على القطر للشكل العودي بحيث تكون القيم الذاتية المختارة في أعلى اليسار.
!> الأعمدة الرائدة لـ Z تشكل أساسًا متعامدًا للمساحة الثابتة المقابلة للقيم الذاتية المختارة.
!>
!> المصفوفة المعقدة تكون في شكل عودي إذا كانت مثلثة عليا.
!>

المعاملات

JOBVS 
!>          JOBVS هو CHARACTER*1
!>          = 'N': لا يتم حساب المتجهات العودية؛
!>          = 'V': يتم حساب المتجهات العودية.
!>

SORT

!>          SORT هو CHARACTER*1
!>          يحدد ما إذا كان سيتم ترتيب القيم الذاتية على قطر الشكل العودي أم لا.
!>          = 'N': لا يتم ترتيب القيم الذاتية:
!>          = 'S': يتم ترتيب القيم الذاتية (انظر SELECT).
!>

SELECT

!>          SELECT هي وظيفة LOGICAL لمعامل COMPLEX*16 واحد
!>          يجب تعريف SELECT كـ EXTERNAL في الروتين الفرعي المستدعي.
!>          إذا كان SORT = 'S'، يتم استخدام SELECT لاختيار القيم الذاتية لترتيبها
!>          في أعلى يسار الشكل العودي.
!>          إذا كان SORT = 'N'، لا يتم استخدام SELECT.
!>          يتم اختيار القيمة الذاتية W(j) إذا كان SELECT(W(j)) صحيحًا.
!>

N

!>          N هو INTEGER
!>          رتبة المصفوفة A. N >= 0.
!>

A

!>          A هو مصفوفة COMPLEX*16، البعد (LDA,N)
!>          عند الدخول، المصفوفة N*N A.
!>          عند الخروج، تم استبدال A بالشكل العودي T الخاص بها.
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرئيسي للمصفوفة A. LDA >= max(1,N).
!>

SDIM

!>          SDIM هو INTEGER
!>          إذا كان SORT = 'N'، فإن SDIM = 0.
!>          إذا كان SORT = 'S'، فإن SDIM = عدد القيم الذاتية التي
!>                         يكون SELECT لها صحيحًا.
!>

W

!>          W هي مصفوفة COMPLEX*16، البعد (N)
!>          تحتوي W على القيم الذاتية المحسوبة، بنفس الترتيب الذي
!>          تظهر به على القطر للشكل العودي الناتج T.
!>

VS

!>          VS هي مصفوفة COMPLEX*16، البعد (LDVS,N)
!>          إذا كان JOBVS = 'V'، فإن VS تحتوي على المصفوفة الوحدوية Z للمتجهات العودية.
!>          إذا كان JOBVS = 'N'، لا يتم استخدام VS.
!>

LDVS

!>          LDVS هو INTEGER
!>          البعد الرئيسي للمصفوفة VS. LDVS >= 1؛ إذا
!>          كان JOBVS = 'V'، فإن LDVS >= N.
!>

WORK

!>          WORK هي مصفوفة COMPLEX*16، البعد (MAX(1,LWORK))
!>          عند الخروج، إذا كان INFO = 0، فإن WORK(1) يعيد LWORK المثالي.
!>

LWORK

!>          LWORK هو INTEGER
!>          بعد المصفوفة WORK. LWORK >= max(1,2*N).
!>          للحصول على أداء جيد، يجب أن يكون LWORK أكبر بشكل عام.
!>
!>          إذا كان LWORK = -1، فمن المفترض وجود استعلام مساحة عمل؛ الروتين
!>          يحسب فقط الحجم الأمثل لمصفوفة WORK، ويعيد
!>          هذه القيمة كمدخل أول لمصفوفة WORK، ولا يصدر أي رسالة خطأ
!>          متعلقة بـ LWORK بواسطة XERBLA.
!>

RWORK

!>          RWORK هي مصفوفة DOUBLE PRECISION، البعد (N)
!>

BWORK

!>          BWORK هي مصفوفة LOGICAL، البعد (N)
!>          لا يتم استخدامها إذا كان SORT = 'N'.
!>

INFO

!>          INFO هو INTEGER
!>          = 0: خروج ناجح
!>          < 0: إذا كان INFO = -i، فإن المعامل i كان له قيمة غير قانونية.
!>          > 0: إذا كان INFO = i، و i هو
!>               <= N: فشل خوارزمية QR في حساب جميع
!>                      القيم الذاتية؛ العناصر 1:ILO-1 و i+1:N من W
!>                      تحتوي على تلك القيم الذاتية التي تقاربت؛
!>                      إذا كان JOBVS = 'V'، فإن VS تحتوي على المصفوفة التي
!>                      تختزل A إلى شكلها العودي المتقارب جزئيًا.
!>               = N+1: لم يكن من الممكن إعادة ترتيب القيم الذاتية لأن
!>                      بعض القيم الذاتية كانت قريبة جدًا من الانفصال (المشكلة
!>                      غير مستقرة للغاية)؛
!>               = N+2: بعد إعادة الترتيب، غيرت الدقة قيم
!>                      بعض القيم الذاتية المعقدة بحيث لم تعد القيم الذاتية الرائدة
!>                      في الشكل العودي تحقق
!>                      SELECT = .TRUE.. قد يكون هذا ناتجًا أيضًا عن
!>                      التقصير بسبب القياس.
!>
