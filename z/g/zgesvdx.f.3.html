subroutine zgesvdx (character jobu, character jobvt, character range, integer m, integer n, complex*16, dimension( lda, * ) a, integer lda, double precision vl, double precision vu, integer il, integer iu, integer ns, double precision, dimension( * ) s, complex*16, dimension( ldu, * ) u, integer ldu, complex*16, dimension( ldvt, * ) vt, integer ldvt, complex*16, dimension( * ) work, integer lwork, double precision, dimension( * ) rwork, integer, dimension( * ) iwork, integer info)

ZGESVDX تحسب تحليل القيم المفردة (SVD) لمصفوفات GE

الغرض:

!>
!>  ZGESVDX تحسب تحليل القيم المفردة (SVD) لمصفوفة A المعقدة ذات الأبعاد M×N،
!>  مع حساب اختياري للمتجهات المفردة اليسرى و/أو اليمنى. يتم كتابة SVD بالشكل
!>
!>      A = U * SIGMA * transpose(V)
!>
!>  حيث SIGMA هي مصفوفة بأبعاد M×N وهي صفرية باستثناء عناصرها القطرية
!>  التي يبلغ عددها min(m,n)، و U هي مصفوفة وحدوية بأبعاد M×M، و V هي مصفوفة
!>  وحدوية بأبعاد N×N. العناصر القطرية لـ SIGMA هي القيم المفردة لـ A؛ وهي
!>  حقيقية وغير سالبة، ويتم إرجاعها بترتيب تنازلي. الأعمدة min(m,n) الأولى
!>  من U و V هي المتجهات المفردة اليسرى واليمنى لـ A.
!>
!>  تستخدم ZGESVDX مشكلة قيم ذاتية للحصول على SVD، مما يسمح بحساب مجموعة فرعية
!>  من القيم والمتجهات المفردة. انظر DBDSVDX لمزيد من التفاصيل.
!>
!>  لاحظ أن الإجراء الفرعي يرجع V**T، وليس V.
!>

المعلمات

JOBU 
!>          JOBU هو CHARACTER*1
!>          يحدد خيارات لحساب جزء أو كل المصفوفة U:
!>          = 'V':  يتم إرجاع الأعمدة min(m,n) الأولى من U (المتجهات المفردة اليسرى)
!>                  أو حسب ما يحدده RANGE في المصفوفة U؛
!>          = 'N':  لا يتم حساب أي أعمدة من U (لا توجد متجهات مفردة يسرى).
!>

JOBVT

!>          JOBVT هو CHARACTER*1
!>           يحدد خيارات لحساب جزء أو كل المصفوفة V**T:
!>           = 'V':  يتم إرجاع الصفوف min(m,n) الأولى من V**T (المتجهات المفردة اليمنى)
!>                   أو حسب ما يحدده RANGE في المصفوفة VT؛
!>           = 'N':  لا يتم حساب أي صفوف من V**T (لا توجد متجهات مفردة يمينية).
!>

RANGE

!>          RANGE هو CHARACTER*1
!>          = 'A': سيتم العثور على جميع القيم المفردة.
!>          = 'V': سيتم العثور على جميع القيم المفردة في النطاق النصفي المفتوح (VL,VU].
!>          = 'I': سيتم العثور على القيم المفردة من IL إلى IU.
!>

M

!>          M هو INTEGER
!>          عدد صفوف المصفوفة المدخلة A.  M >= 0.
!>

N

!>          N هو INTEGER
!>          عدد أعمدة المصفوفة المدخلة A.  N >= 0.
!>

A

!>          A هو مصفوفة COMPLEX*16، بأبعاد (LDA,N)
!>          عند الدخول، مصفوفة A ذات الأبعاد M×N.
!>          عند الخروج، يتم تدمير محتويات A.
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرائد للمصفوفة A.  LDA >= max(1,M).
!>

VL

!>          VL هو DOUBLE PRECISION
!>          إذا كان RANGE='V'، الحد الأدنى للنطاق الذي سيتم البحث فيه عن
!>          القيم المفردة. VU > VL.
!>          لا يتم الرجوع إليه إذا كان RANGE = 'A' أو 'I'.
!>

VU

!>          VU هو DOUBLE PRECISION
!>          إذا كان RANGE='V'، الحد الأعلى للنطاق الذي سيتم البحث فيه عن
!>          القيم المفردة. VU > VL.
!>          لا يتم الرجوع إليه إذا كان RANGE = 'A' أو 'I'.
!>

IL

!>          IL هو INTEGER
!>          إذا كان RANGE='I'، فهرس أصغر قيمة مفردة سيتم إرجاعها.
!>          1 <= IL <= IU <= min(M,N)، إذا كان min(M,N) > 0.
!>          لا يتم الرجوع إليه إذا كان RANGE = 'A' أو 'V'.
!>

IU

!>          IU هو INTEGER
!>          إذا كان RANGE='I'، فهرس أكبر قيمة مفردة سيتم إرجاعها.
!>          1 <= IL <= IU <= min(M,N)، إذا كان min(M,N) > 0.
!>          لا يتم الرجوع إليه إذا كان RANGE = 'A' أو 'V'.
!>

NS

!>          NS هو INTEGER
!>          العدد الإجمالي للقيم المفردة التي تم العثور عليها،
!>          0 <= NS <= min(M,N).
!>          إذا كان RANGE = 'A'، فإن NS = min(M,N)؛ إذا كان RANGE = 'I'، فإن NS = IU-IL+1.
!>

S

!>          S هي مصفوفة DOUBLE PRECISION، بأبعاد (min(M,N))
!>          القيم المفردة لـ A، مرتبة بحيث S(i) >= S(i+1).
!>

U

!>          U هي مصفوفة COMPLEX*16، بأبعاد (LDU,UCOL)
!>          إذا كان JOBU = 'V'، فإن U تحتوي على أعمدة U (المتجهات المفردة اليسرى،
!>          مخزنة عمودياً) حسب ما يحدده RANGE؛ إذا كان JOBU = 'N'،
!>          فلا يتم الرجوع إلى U.
!>          ملاحظة: يجب على المستخدم التأكد من أن UCOL >= NS؛ إذا كان RANGE = 'V'،
!>          فإن القيمة الدقيقة لـ NS غير معروفة مسبقًا ويجب استخدام حد أعلى.
!>

LDU

!>          LDU هو INTEGER
!>          البعد الرائد للمصفوفة U.  LDU >= 1؛ إذا كان
!>          JOBU = 'V'، فإن LDU >= M.
!>

VT

!>          VT هي مصفوفة COMPLEX*16، بأبعاد (LDVT,N)
!>          إذا كان JOBVT = 'V'، فإن VT تحتوي على صفوف V**T (المتجهات المفردة اليمنى،
!>          مخزنة صفياً) حسب ما يحدده RANGE؛ إذا كان JOBVT = 'N'،
!>          فلا يتم الرجوع إلى VT.
!>          ملاحظة: يجب على المستخدم التأكد من أن LDVT >= NS؛ إذا كان RANGE = 'V'،
!>          فإن القيمة الدقيقة لـ NS غير معروفة مسبقًا ويجب استخدام حد أعلى.
!>

LDVT

!>          LDVT هو INTEGER
!>          البعد الرائد للمصفوفة VT.  LDVT >= 1؛ إذا كان
!>          JOBVT = 'V'، فإن LDVT >= NS (انظر أعلاه).
!>

WORK

!>          WORK هي مصفوفة COMPLEX*16، بأبعاد (MAX(1,LWORK))
!>          عند الخروج، إذا كان INFO = 0، فإن WORK(1) يرجع LWORK المثلى؛
!>

LWORK

!>          LWORK هو INTEGER
!>          بعد المصفوفة WORK.
!>          LWORK >= MAX(1,MIN(M,N)*(MIN(M,N)+4)) للمسارات (انظر
!>          التعليقات داخل الكود):
!>             - PATH 1  (M أكبر بكثير من N)
!>             - PATH 1t (N أكبر بكثير من M)
!>          LWORK >= MAX(1,MIN(M,N)*2+MAX(M,N)) للمسارات الأخرى.
!>          لأداء جيد، يجب أن يكون LWORK أكبر بشكل عام.
!>
!>          إذا كان LWORK = -1، فسيتم افتراض استعلام مساحة العمل؛ يقوم الإجراء الفرعي
!>          بحساب حجم مصفوفة WORK الأمثل فقط، ويرجع هذا الحجم كإدخال أول
!>          لمصفوفة WORK، ولا يتم إصدار أي رسالة خطأ متعلقة بـ LWORK بواسطة XERBLA.
!>

RWORK

!>          RWORK هي مصفوفة DOUBLE PRECISION، بأبعاد (MAX(1,LRWORK))
!>          LRWORK >= MIN(M,N)*(MIN(M,N)*2+15*MIN(M,N)).
!>

IWORK

!>          IWORK هي مصفوفة INTEGER، بأبعاد (12*MIN(M,N))
!>          إذا كان INFO = 0، فإن العناصر الأولى NS من IWORK هي صفر. إذا كان INFO > 0،
!>          فإن IWORK تحتوي على فهارس المتجهات الذاتية التي فشلت
!>          في التقارب في DBDSVDX/DSTEVX.
!>

INFO

!>          INFO هو INTEGER
!>          = 0: التنفيذ الناجح.
!>          < 0: إذا كان المعلمة -i تشير إلى المعلمة i-th غير صالحة.
!>          > 0: فشل الإجراء الفرعي DBDSVDX في التقارب، و INFO هو عدد العناصر
!>               التي فشلت في التقارب.
!>
