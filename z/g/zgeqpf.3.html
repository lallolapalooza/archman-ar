subroutine zgeqpf (integer m, integer n, complex*16, dimension( lda, * ) a, integer lda, integer, dimension( * ) jpvt, complex*16, dimension( * ) tau, complex*16, dimension( * ) work, double precision, dimension( * ) rwork, integer info)

ZGEQPF

الغرض:

!>
!> تم إهمال هذا الإجراء الفرعي واستبداله بالإجراء الفرعي ZGEQP3.
!>
!> ZGEQPF يحسب تحليل QR مع تثبيت الأعمدة لمصفوفة
!> معقدة M-by-N وهي A: A*P = Q*R.
!>

المعلمات

M 
!>          M هو عدد صحيح (INTEGER)
!>          عدد صفوف المصفوفة A. M >= 0.
!>

N

!>          N هو عدد صحيح (INTEGER)
!>          عدد أعمدة المصفوفة A. N >= 0
!>

A

!>          A هي مصفوفة معقدة (COMPLEX*16) ، ذات أبعاد (LDA,N)
!>          عند الإدخال، المصفوفة A ذات الأبعاد M-by-N.
!>          عند الخروج، يحتوي المثلث العلوي للمصفوفة على المصفوفة
!>          المثلثية العليا R ذات الأبعاد min(M,N)-by-N؛ العناصر
!>          أسفل القطر، جنبًا إلى جنب مع المصفوفة TAU،
!>          تمثل المصفوفة الموحدة Q كناتج لـ
!>          min(m,n) مجمعات أولية.
!>

LDA

!>          LDA هو عدد صحيح (INTEGER)
!>          البعد القيادي للمصفوفة A. LDA >= max(1,M).
!>

JPVT

!>          JPVT هي مصفوفة أعداد صحيحة (INTEGER)، ذات أبعاد (N)
!>          عند الإدخال، إذا كان JPVT(i) .ne. 0، فإن العمود i من A يتم تبديله
!>          إلى مقدمة A*P (عمود قيادي)؛ إذا كان JPVT(i) = 0،
!>          فإن العمود i من A هو عمود حر.
!>          عند الخروج، إذا كان JPVT(i) = k ، فإن العمود i من A*P
!>          كان العمود k من A.
!>

TAU

!>          TAU هي مصفوفة معقدة (COMPLEX*16)، ذات أبعاد (min(M,N))
!>          العوامل القياسية للمجمعات الأولية.
!>

WORK

!>          WORK هي مصفوفة معقدة (COMPLEX*16)، ذات أبعاد (N)
!>

RWORK

!>          RWORK هي مصفوفة ذات دقة مزدوجة (DOUBLE PRECISION)، ذات أبعاد (2*N)
!>

INFO

!>          INFO هو عدد صحيح (INTEGER)
!>          = 0:  خروج ناجح
!>          < 0:  إذا كان INFO = -i ، فإن الوسيط i كان له قيمة غير قانونية
!>

تفاصيل إضافية:

!>
!>  يتم تمثيل المصفوفة Q كناتج للمجمعات الأولية
!>
!>     Q = H(1) H(2) . . . H(n)
!>
!>  كل H(i) له الشكل
!>
!>     H = I - tau * v * v**H
!>
!>  حيث tau هو عامل قياسي معقد، و v هو متجه معقد مع
!>  v(1:i-1) = 0 و v(i) = 1؛ يتم تخزين v(i+1:m) عند الخروج في A(i+1:m,i).
!>
!>  يتم تمثيل المصفوفة P في jpvt على النحو التالي: إذا
!>     jpvt(j) = i
!>  فإن العمود j من P هو المتجه القاموس i.
!>
!>  تم تعديل استراتيجية تحديث معايير الأعمدة الجزئية بواسطة
!>    Z. Drmac and Z. Bujanovic, Dept. of Mathematics,
!>    University of Zagreb, Croatia.
