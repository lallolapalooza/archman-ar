subroutine zgerfs (character trans, integer n, integer nrhs, complex*16, dimension( lda, * ) a, integer lda, complex*16, dimension( ldaf, * ) af, integer ldaf, integer, dimension( * ) ipiv, complex*16, dimension( ldb, * ) b, integer ldb, complex*16, dimension( ldx, * ) x, integer ldx, double precision, dimension( * ) ferr, double precision, dimension( * ) berr, complex*16, dimension( * ) work, double precision, dimension( * ) rwork, integer info)

ZGERFS

الغرض:

!>
!> ZGERFS يحسن الحل المحسوب لنظام المعادلات الخطية
!> ويوفر حدود الخطأ وتقديرات الخطأ الخلفي للحل.
!>

المعلمات

TRANS 
!>          TRANS is CHARACTER*1
!>          يحدد شكل نظام المعادلات:
!>          = 'N':  A * X = B     (بدون تبديل)
!>          = 'T':  A**T * X = B  (تبديل)
!>          = 'C':  A**H * X = B  (تبديل مرافق)
!>

N

!>          N is INTEGER
!>          رتبة المصفوفة A.  N >= 0.
!>

NRHS

!>          NRHS is INTEGER
!>          عدد الأيادي اليمنى، أي عدد أعمدة
!>          المصفوفتين B و X.  NRHS >= 0.
!>

A

!>          A is COMPLEX*16 array, dimension (LDA,N)
!>          المصفوفة الأصلية A بحجم N*N.
!>

LDA

!>          LDA is INTEGER
!>          البعد الرئيسي للمصفوفة A.  LDA >= max(1,N).
!>

AF

!>          AF is COMPLEX*16 array, dimension (LDAF,N)
!>          العوامل L و U من تحليل A = P*L*U
!>          كما تم حسابه بواسطة ZGETRF.
!>

LDAF

!>          LDAF is INTEGER
!>          البعد الرئيسي للمصفوفة AF.  LDAF >= max(1,N).
!>

IPIV

!>          IPIV is INTEGER array, dimension (N)
!>          فهارس المحور من ZGETRF؛ لـ 1<=i<=N، تم تبديل الصف i من
!>          المصفوفة مع الصف IPIV(i).
!>

B

!>          B is COMPLEX*16 array, dimension (LDB,NRHS)
!>          مصفوفة الطرف الأيمن B.
!>

LDB

!>          LDB is INTEGER
!>          البعد الرئيسي للمصفوفة B.  LDB >= max(1,N).
!>

X

!>          X is COMPLEX*16 array, dimension (LDX,NRHS)
!>          عند الإدخال، مصفوفة الحل X، كما تم حسابها بواسطة ZGETRS.
!>          عند الخروج، مصفوفة الحل المحسنة X.
!>

LDX

!>          LDX is INTEGER
!>          البعد الرئيسي للمصفوفة X.  LDX >= max(1,N).
!>

FERR

!>          FERR is DOUBLE PRECISION array, dimension (NRHS)
!>          تقدير حد الخطأ الأمامي لكل متجه حل
!>          X(j) (العمود j من مصفوفة الحل X).
!>          إذا كان XTRUE هو الحل الحقيقي المقابل لـ X(j)، فإن FERR(j)
!>          هو تقدير للحد الأعلى لقيمة أكبر
!>          عنصر في (X(j) - XTRUE) مقسومًا على قيمة أكبر
!>          عنصر في X(j).  التقدير موثوق به مثل
!>          تقدير RCOND، وهو دائمًا تقريبًا أقل من
!>          القيمة الحقيقية للخطأ.
!>

BERR

!>          BERR is DOUBLE PRECISION array, dimension (NRHS)
!>          خطأ التقريب النسبي المكون من كل متجه حل
!>          X(j) (أي أصغر تغيير نسبي في
!>          أي عنصر في A أو B يجعل X(j) حلاً دقيقًا).
!>

WORK

!>          WORK is COMPLEX*16 array, dimension (2*N)
!>

RWORK

!>          RWORK is DOUBLE PRECISION array, dimension (N)
!>

INFO

!>          INFO is INTEGER
!>          = 0:  خروج ناجح
!>          < 0:  إذا كان INFO = -i، فإن المعلمة i كانت قيمة غير قانونية
!>
