subroutine zgeql2 (integer m, integer n, complex*16, dimension( lda, * ) a, integer lda, complex*16, dimension( * ) tau, complex*16, dimension( * ) work, integer info)

ZGEQL2 تحسب التحلل QL لمصفوفة مستطيلة عامة باستخدام خوارزمية غير مجمعة.

الغرض:

!>
!> ZGEQL2 تحسب تحلل QL لمصفوفة A المعقدة m × n:
!> A = Q * L.
!>

المعلمات

M 
!>          M هو عدد صحيح (INTEGER)
!>          عدد صفوف المصفوفة A.  M >= 0.
!>

N

!>          N هو عدد صحيح (INTEGER)
!>          عدد أعمدة المصفوفة A.  N >= 0.
!>

A

!>          A هي مصفوفة معقدة (COMPLEX*16) ، البعد (LDA,N)
!>          عند الإدخال، المصفوفة A بحجم m × n.
!>          عند الخروج، إذا كان m >= n، فإن المثلث السفلي للمصفوفة الفرعية
!>          A(m-n+1:m,1:n) يحتوي على المصفوفة المثلثية السفلية L بحجم n × n؛
!>          إذا كان m <= n، فإن العناصر على وتحت القطر العلوي رقم (n-m)
!>          تحتوي على المصفوفة شبه المنحرفة السفلية L بحجم m × n؛
!>          العناصر المتبقية، مع المصفوفة TAU، تمثل المصفوفة الوحدوية
!>          Q كناتج للمنعكسات الابتدائية (انظر التفاصيل الإضافية).
!>

LDA

!>          LDA هو عدد صحيح (INTEGER)
!>          البعد الرائد للمصفوفة A.  LDA >= max(1,M).
!>

TAU

!>          TAU هي مصفوفة معقدة (COMPLEX*16) ، البعد (min(M,N))
!>          العوامل القياسية للمنعكسات الابتدائية (انظر التفاصيل الإضافية).
!>

WORK

!>          WORK هي مصفوفة معقدة (COMPLEX*16) ، البعد (N)
!>

INFO

!>          INFO هو عدد صحيح (INTEGER)
!>          = 0: خروج ناجح
!>          < 0: إذا كان INFO = -i، فإن الوسيط i كان له قيمة غير قانونية
!>

التفاصيل الإضافية:

!>
!>  يتم تمثيل المصفوفة Q كناتج للمنعكسات الابتدائية
!>
!>     Q = H(k) . . . H(2) H(1)، حيث k = min(m,n).
!>
!>  كل H(i) لها الشكل
!>
!>     H(i) = I - tau * v * v**H
!>
!>  حيث tau هو عدد قياسي معقد، و v هو متجه معقد مع
!>  v(m-k+i+1:m) = 0 و v(m-k+i) = 1؛ يتم تخزين v(1:m-k+i-1) عند الخروج في
!>  A(1:m-k+i-1,n-k+i)، و tau في TAU(i).
!>
