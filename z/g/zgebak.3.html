توثيق الوظيفة/الإجراء الفرعي

subroutine zgebak (character job, character side, integer n, integer ilo, integer ihi, double precision, dimension( * ) scale, integer m, complex*16, dimension( ldv, * ) v, integer ldv, integer info)

ZGEBAK

الغرض:

!>
!> ZGEBAK تشكل المتجهات الذاتية اليمنى أو اليسرى لمصفوفة عامة معقدة
!> عن طريق التحويل العكسي على المتجهات الذاتية المحسوبة للمصفوفة المتوازنة
!> الناتجة عن ZGEBAL.
!>

المعاملات

JOB 
!>          JOB هو CHARACTER*1
!>          يحدد نوع التحويل العكسي المطلوب:
!>          = 'N': لا تفعل شيئا، عد فوراً؛
!>          = 'P': قم بالتحويل العكسي لعملية التبديل فقط؛
!>          = 'S': قم بالتحويل العكسي للتحجيم فقط؛
!>          = 'B': قم بالتحويلات العكسية للتبديل والتحجيم معاً.
!>          يجب أن يكون JOB هو نفسه المعامل JOB المقدم إلى ZGEBAL.
!>

SIDE

!>          SIDE هو CHARACTER*1
!>          = 'R':  تحتوي V على المتجهات الذاتية اليمنى؛
!>          = 'L':  تحتوي V على المتجهات الذاتية اليسرى.
!>

N

!>          N هو INTEGER
!>          عدد صفوف المصفوفة V.  N >= 0.
!>

ILO

!>          ILO هو INTEGER
!>

IHI

!>          IHI هو INTEGER
!>          الأعداد الصحيحة ILO و IHI التي تم تحديدها بواسطة ZGEBAL.
!>          1 <= ILO <= IHI <= N، إذا كان N > 0؛ ILO=1 و IHI=0، إذا كان N=0.
!>

SCALE

!>          SCALE هو مصفوفة DOUBLE PRECISION، حجمها (N)
!>          تفاصيل عوامل التبديل والتحجيم، كما تم إرجاعها
!>          بواسطة ZGEBAL.
!>

M

!>          M هو INTEGER
!>          عدد أعمدة المصفوفة V.  M >= 0.
!>

V

!>          V هي مصفوفة COMPLEX*16، حجمها (LDV,M)
!>          عند الإدخال، مصفوفة المتجهات الذاتية اليمنى أو اليسرى التي سيتم
!>          تحويلها، كما تم إرجاعها بواسطة ZHSEIN أو ZTREVC.
!>          عند الخروج، يتم استبدال V بالمتجهات الذاتية المحولة.
!>

LDV

!>          LDV هو INTEGER
!>          البعد الرئيسي للمصفوفة V. LDV >= max(1,N).
!>

INFO

!>          INFO هو INTEGER
!>          = 0:  انتهاء ناجح
!>          < 0:  إذا كان INFO = -i، فإن المعامل i كان له قيمة غير قانونية.
!>
