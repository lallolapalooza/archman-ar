توثيق الوظيفة/الروتين الفرعي

subroutine zgglse (integer m, integer n, integer p, complex*16, dimension( lda, * ) a, integer lda, complex*16, dimension( ldb, * ) b, integer ldb, complex*16, dimension( * ) c, complex*16, dimension( * ) d, complex*16, dimension( * ) x, complex*16, dimension( * ) work, integer lwork, integer info)

ZGGLSE يحل الأنظمة المفرطة في المحدد أو الناقصة في المحدد للمصفوفات الأخرى

الغرض:

!>
!> ZGGLSE يحل مشكلة المربعات الصغرى المقيدة خطيًا (LSE)
!> المشكلة:
!>
!>         تصغير || c - A*x ||_2   بشرط   B*x = d
!>
!> حيث A مصفوفة M-by-N، و B مصفوفة P-by-N، و c هو متجه معطى
!> M، و d هو متجه معطى P. يفترض أن
!> P <= N <= M+P، و
!>
!>          rank(B) = P و  rank( (A) ) = N.
!>                               ( (B) )
!>
!> تضمن هذه الشروط أن مشكلة LSE لها حل فريد،
!> والذي يتم الحصول عليه باستخدام عامل الاستبدال العام (generalized RQ factorization) للمصفوفتين
!> (B, A) المعطاة بالصيغة
!>
!>    B = (0 R)*Q,   A = Z*T*Q.
!>
!> يجب على المتصلين بهذا الروتين الفرعي ملاحظة أن فحوصات التفرد/نقص الرتبة
!> المطبقة في هذا الروتين الفرعي بدائية. الروتين الفرعي ZTRTRS الذي يستدعيه هذا
!> الروتين الفرعي يشير فقط إلى فشل بسبب التفرد إذا كانت المشكلة فريدة تمامًا.
!>
!> من الممكن أن يكون أحد العوامل (أو أكثر) المشاركة في عامل الاستبدال العام (generalized RQ factorization)
!> للمزدوج (B, A) قريبًا بشكل غير طبيعي من التفرد دون أن يشير هذا
!> الروتين الفرعي إلى خطأ. قد تكون الحلول المحسوبة لمثل هذه المشاكل شبه المفرطة في المحدد
!> أقل دقة بسبب فقدان الدقة العددية.
!> 
!>

المعلمات

M 
!>          M هو عدد صحيح (INTEGER)
!>          عدد صفوف المصفوفة A.  M >= 0.
!>

N

!>          N هو عدد صحيح (INTEGER)
!>          عدد أعمدة المصفوفتين A و B. N >= 0.
!>

P

!>          P هو عدد صحيح (INTEGER)
!>          عدد صفوف المصفوفة B. 0 <= P <= N <= M+P.
!>

A

!>          A هو مصفوفة من نوع COMPLEX*16، بأبعاد (LDA,N)
!>          عند الإدخال، مصفوفة A بأبعاد M-by-N.
!>          عند الخروج، تحتوي العناصر على القطر وفوقه في المصفوفة
!>          على المصفوفة شبه المنحرفة العليا T بأبعاد min(M,N)-by-N.
!>

LDA

!>          LDA هو عدد صحيح (INTEGER)
!>          البعد الرئيسي للمصفوفة A. LDA >= max(1,M).
!>

B

!>          B هي مصفوفة من نوع COMPLEX*16، بأبعاد (LDB,N)
!>          عند الإدخال، مصفوفة B بأبعاد P-by-N.
!>          عند الخروج، يحتوي المثلث العلوي من المصفوفة الفرعية B(1:P,N-P+1:N)
!>          على المصفوفة المثلثية العليا R بأبعاد P-by-P.
!>

LDB

!>          LDB هو عدد صحيح (INTEGER)
!>          البعد الرئيسي للمصفوفة B. LDB >= max(1,P).
!>

C

!>          C هي مصفوفة من نوع COMPLEX*16، بأبعاد (M)
!>          عند الإدخال، تحتوي C على متجه الطرف الأيمن لمشكلة
!>          المربعات الصغرى.
!>          عند الخروج، يكون مجموع مربعات الخطأ للحل
!>          مُعطى بمجموع مربعات العناصر من N-P+1 إلى M
!>          للمتجه C.
!>

D

!>          D هي مصفوفة من نوع COMPLEX*16، بأبعاد (P)
!>          عند الإدخال، يحتوي D على متجه الطرف الأيمن للمعادلة
!>          المقيدة.
!>          عند الخروج، يتم تدمير D.
!>

X

!>          X هي مصفوفة من نوع COMPLEX*16، بأبعاد (N)
!>          عند الخروج، X هو حل مشكلة LSE.
!>

WORK

!>          WORK هي مصفوفة من نوع COMPLEX*16، بأبعاد (MAX(1,LWORK))
!>          عند الخروج، إذا كان INFO = 0، فإن WORK(1) يعيد قيمة LWORK المثلى.
!>

LWORK

!>          LWORK هو عدد صحيح (INTEGER)
!>          بعد المصفوفة WORK. LWORK >= max(1,M+N+P).
!>          لأداء مثالي LWORK >= P+min(M,N)+max(M,N)*NB,
!>          حيث NB هو حد علوي لأحجام الكتل المثلى لـ
!>          ZGEQRF, CGERQF, ZUNMQR و CUNMRQ.
!>
!>          إذا كان LWORK = -1، فسيتم افتراض استعلام مساحة العمل؛ الروتين
!>          يحسب فقط الحجم الأمثل للمصفوفة WORK، ويعيد
!>          هذه القيمة كالإدخال الأول للمصفوفة WORK، ولا يصدر رسالة خطأ
!>          متعلقة بـ LWORK بواسطة XERBLA.
!>

INFO

!>          INFO هو عدد صحيح (INTEGER)
!>          = 0:  انتهاء ناجح.
!>          < 0:  إذا كان INFO = -i، فإن الوسيط i كان له قيمة غير قانونية.
!>          = 1:  العامل المثلثي العلوي R المرتبط بـ B في
!>                عامل الاستبدال العام (generalized RQ factorization) للمزدوج (B, A) هو بالضبط
!>                متفرد، بحيث rank(B) < P؛ لم يكن من الممكن حساب
!>                حل المربعات الصغرى.
!>          = 2:  الجزء (N-P) by (N-P) من العامل شبه المنحرف العلوي
!>                T المرتبط بـ A في عامل الاستبدال العام (generalized RQ factorization)
!>                للمزدوج (B, A) هو بالضبط متفرد، بحيث
!>                rank( (A) ) < N؛ لم يكن من الممكن حساب
!>                    ( (B) )
!>                حل المربعات الصغرى.
!>
