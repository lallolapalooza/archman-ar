subroutine zget08 (character trans, integer m, integer n, integer nrhs, complex*16, dimension( lda, * ) a, integer lda, complex*16, dimension( ldx, * ) x, integer ldx, complex*16, dimension( ldb, * ) b, integer ldb, double precision, dimension( * ) rwork, double precision resid)

ZGET08

الغرض:

!>
!> ZGET08 يحسب الباقي لحل نظام المعادلات الخطية  A*x = b  أو  A'*x = b:
!>    RESID = norm(B - A*X) / ( norm(A) * norm(X) * EPS ),
!> حيث EPS هو خطأ الآلة.
!>

المعلمات

TRANS 
!>          TRANS is CHARACTER*1
!>          يحدد شكل نظام المعادلات:
!>          = 'N':  A *x = b
!>          = 'T':  A^T*x = b, حيث A^T هي منقولة A
!>          = 'C':  A^H*x = b, حيث A^H هي المرافقة المنقولة لـ A
!>

M

!>          M is INTEGER
!>          عدد صفوف المصفوفة A.  M >= 0.
!>

N

!>          N is INTEGER
!>          عدد أعمدة المصفوفة A.  N >= 0.
!>

NRHS

!>          NRHS is INTEGER
!>          عدد أعمدة B، مصفوفة الأيادي اليمنى.
!>          NRHS >= 0.
!>

A

!>          A is COMPLEX*16 array, dimension (LDA,N)
!>          المصفوفة الأصلية A بحجم M x N.
!>

LDA

!>          LDA is INTEGER
!>          البعد الرائد للمصفوفة A.  LDA >= max(1,M).
!>

X

!>          X is COMPLEX*16 array, dimension (LDX,NRHS)
!>          متجهات الحل المحسوبة لأنظمة المعادلات الخطية.
!>

LDX

!>          LDX is INTEGER
!>          البعد الرائد للمصفوفة X.  إذا كان TRANS = 'N'،
!>          LDX >= max(1,N)؛ إذا كان TRANS = 'T' أو 'C'، LDX >= max(1,M).
!>

B

!>          B is COMPLEX*16 array, dimension (LDB,NRHS)
!>          عند الإدخال، متجهات اليد اليمنى لأنظمة المعادلات الخطية.
!>          عند الخروج، يتم استبدال B بـ B - A*X.
!>

LDB

!>          LDB is INTEGER
!>          البعد الرائد للمصفوفة B.  إذا كان TRANS = 'N'،
!>          LDB >= max(1,M)؛ إذا كان TRANS = 'T' أو 'C'، LDB >= max(1,N).
!>

RWORK

!>          RWORK is DOUBLE PRECISION array, dimension (M)
!>

RESID

!>          RESID is DOUBLE PRECISION
!>          الحد الأقصى لعدد الأيادي اليمنى لـ
!>          norm(B - A*X) / ( norm(A) * norm(X) * EPS ).
!>
