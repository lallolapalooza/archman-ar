subroutine zgtsv (integer n, integer nrhs, complex*16, dimension( * ) dl, complex*16, dimension( * ) d, complex*16, dimension( * ) du, complex*16, dimension( ldb, * ) b, integer ldb, integer info)

ZGTSV يحسب حل نظام المعادلات الخطية A * X = B لمصفوفات GT

الغرض:

!>
!> ZGTSV  يحل المعادلة
!>
!>    A*X = B,
!>
!> حيث A مصفوفة ثلاثية الأقطار N × N، عن طريق الحذف الغاوسي مع التقسيم الجزئي.
!>
!> لاحظ أنه يمكن حل المعادلة  A**T *X = B  عن طريق تبديل ترتيب
!> الوسائط DU و DL.
!>

المعاملات

N 
!>          N هو عدد صحيح (INTEGER)
!>          مرتبة المصفوفة A.  N >= 0.
!>

NRHS

!>          NRHS هو عدد صحيح (INTEGER)
!>          عدد الأطراف اليمنى، أي عدد أعمدة
!>          المصفوفة B.  NRHS >= 0.
!>

DL

!>          DL هو مصفوفة COMPLEX*16، البعد (N-1)
!>          عند الإدخال، يجب أن تحتوي DL على العناصر الفرعية تحت القطر (n-1)
!>          للمصفوفة A.
!>          عند الخروج، يتم الكتابة فوق DL بالعناصر (n-2) من
!>          القطر الفائق الثاني للمصفوفة المثلثية العلوية U من
!>          تحليل LU للمصفوفة A، في DL(1)، ..., DL(n-2).
!>

D

!>          D هي مصفوفة COMPLEX*16، البعد (N)
!>          عند الإدخال، يجب أن تحتوي D على العناصر القطرية لـ A.
!>          عند الخروج، يتم الكتابة فوق D بالعناصر القطرية n لـ U.
!>

DU

!>          DU هي مصفوفة COMPLEX*16، البعد (N-1)
!>          عند الإدخال، يجب أن تحتوي DU على العناصر فوق القطر (n-1)
!>          للمصفوفة A.
!>          عند الخروج، يتم الكتابة فوق DU بالعناصر (n-1) من
!>          القطر الفائق الأول لـ U.
!>

B

!>          B هي مصفوفة COMPLEX*16، البعد (LDB,NRHS)
!>          عند الإدخال، مصفوفة الأطراف اليمنى B ذات الأبعاد N × NRHS.
!>          عند الخروج، إذا كان INFO = 0، فإن مصفوفة الحل X ذات الأبعاد N × NRHS.
!>

LDB

!>          LDB هو عدد صحيح (INTEGER)
!>          البعد الرئيسي للمصفوفة B.  LDB >= max(1,N).
!>

INFO

!>          INFO هو عدد صحيح (INTEGER)
!>          = 0:  اكتمال ناجح
!>          < 0:  إذا كان INFO = -i، فإن المعامل i كان له قيمة غير قانونية
!>          > 0:  إذا كان INFO = i، فإن U(i,i) تساوي صفرًا بالضبط، ولم يتم حساب الحل.
!>                لم يكتمل التحليل إلا إذا كان i = N.
!>
