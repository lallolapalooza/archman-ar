subroutine zgsvj1 (character*1 jobv, integer m, integer n, integer n1, complex*16, dimension( lda, * ) a, integer lda, complex*16, dimension( n ) d, double precision, dimension( n ) sva, integer mv, complex*16, dimension( ldv, * ) v, integer ldv, double precision eps, double precision sfmin, double precision tol, integer nsweep, complex*16, dimension( lwork ) work, integer lwork, integer info)

ZGSVJ1 معالج مسبق للروتين zgesvj، يطبق دورانات جاكوبي تستهدف محاور محددة فقط.

الغرض:

!>
!> يتم استدعاء ZGSVJ1 من ZGESVJ كمعالج مسبق وهذا هو غرضه الرئيسي.
!> يطبق دورانات جاكوبي بنفس الطريقة التي يعمل بها ZGESVJ، ولكنه يستهدف
!> محاور محددة فقط ولا يتحقق من التقارب (معيار الإيقاف). تتوفر بعض معلمات
!> الضبط (المميزة بـ [TP]) للمنفذ.
!>
!> تفاصيل إضافية
!> ~~~~~~~~~~~~~~~
!> يطبق ZGSVJ1 بضع دورات من دورانات جاكوبي في مساحة الأعمدة للمصفوفة A
!> المدخلة M × N. تؤخذ أزواج المحاور من الكتلة خارج القطر (1،2) في مصفوفة
!> جرام المقابلة A^T * A. يتم تمييز مدخلات الكتلة (البلاطات) للكتلة خارج
!> القطر (1،2) بواسطة [x] في المخطط التالي:
!>
!>    | *  *  * [x] [x] [x]|
!>    | *  *  * [x] [x] [x]|    دورة الصفوف في الكتل [x] nblr × nblc.
!>    | *  *  * [x] [x] [x]|    محاور دورانية داخل كل كتلة [x].
!>    |[x] [x] [x] *  *  * |
!>    |[x] [x] [x] *  *  * |
!>    |[x] [x] [x] *  *  * |
!>
!> فيما يتعلق بأعمدة A، يتم تدوير الأعمدة N1 الأولى 'مقابل' الأعمدة المتبقية
!> N-N1 من A، في محاولة لزيادة الزاوية بين المساحات الفرعية المقابلة. الكتلة
!> خارج القطر هي N1 × (N-N1) ويتم تبليطها باستخدام بلاطات مربعة من
!> الجانب KBL. هنا، KBL هي معلمة ضبط. عدد الدورات معطى في NSWEEP وعتبة
!> التعامد معطاة في TOL.
!>

المعلمات

JOBV 
!>          JOBV هو CHARACTER*1
!>          يحدد ما إذا كان سيتم استخدام المخرجات من هذا الإجراء
!>          لحساب المصفوفة V:
!>          = 'V': يتم تجميع ناتج دورانات جاكوبي
!>                 عن طريق الضرب اللاحق بالمصفوفة N×N V.
!>                (انظر وصف V.)
!>          = 'A': يتم تجميع ناتج دورانات جاكوبي
!>                 عن طريق الضرب اللاحق بالمصفوفة MV×N V.
!>                (انظر وصف MV و V.)
!>          = 'N': لا يتم تجميع دورانات جاكوبي.
!>

M

!>          M هو INTEGER
!>          عدد الصفوف في المصفوفة المدخلة A.  M >= 0.
!>

N

!>          N هو INTEGER
!>          عدد الأعمدة في المصفوفة المدخلة A.
!>          M >= N >= 0.
!>

N1

!>          N1 هو INTEGER
!>          يحدد N1 تقسيم الكتلة 2 × 2، حيث يتم تدوير الأعمدة N1 الأولى
!>          'مقابل' الأعمدة المتبقية N-N1 من A.
!>

A

!>          A هو مصفوفة COMPLEX*16، البعد (LDA,N)
!>          عند الإدخال، المصفوفة M × N A، بحيث تمثل A * diag(D)
!>          المصفوفة المدخلة.
!>          عند الخروج،
!>          A_onexit * D_onexit تمثل المصفوفة المدخلة A * diag(D)
!>          مضروبة لاحقًا بسلسلة من دورانات جاكوبي، حيث
!>          عتبة الدوران والعدد الإجمالي للدورات معطاة في
!>          TOL و NSWEEP، على التوالي.
!>          (انظر وصف N1، D، TOL و NSWEEP.)
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرئيسي للمصفوفة A.  LDA >= max(1,M).
!>

D

!>          D هو مصفوفة COMPLEX*16، البعد (N)
!>          تجمع المصفوفة D عوامل القياس من دورانات جاكوبي المقيّسة بسرعة.
!>          عند الإدخال، تمثل A * diag(D) المصفوفة المدخلة.
!>          عند الخروج، تمثل A_onexit * diag(D_onexit) المصفوفة المدخلة
!>          مضروبة لاحقًا بسلسلة من دورانات جاكوبي، حيث
!>          عتبة الدوران والعدد الإجمالي للدورات معطاة في
!>          TOL و NSWEEP، على التوالي.
!>          (انظر وصف N1، A، TOL و NSWEEP.)
!>

SVA

!>          SVA هي مصفوفة DOUBLE PRECISION، البعد (N)
!>          عند الإدخال، تحتوي SVA على المعايير الإقليدية لأعمدة
!>          المصفوفة A * diag(D).
!>          عند الخروج، تحتوي SVA على المعايير الإقليدية لأعمدة
!>          المصفوفة onexit * diag(D_onexit).
!>

MV

!>          MV هو INTEGER
!>          إذا كان JOBV = 'A'، فسيتم ضرب MV صفًا من V لاحقًا بسلسلة
!>          من دورانات جاكوبي.
!>          إذا كان JOBV = 'N'، فلن يتم الرجوع إلى MV.
!>

V

!>          V هي مصفوفة COMPLEX*16، البعد (LDV,N)
!>          إذا كان JOBV = 'V'، فسيتم ضرب N صفًا من V لاحقًا بسلسلة
!>          من دورانات جاكوبي.
!>          إذا كان JOBV = 'A'، فسيتم ضرب MV صفًا من V لاحقًا بسلسلة
!>          من دورانات جاكوبي.
!>          إذا كان JOBV = 'N'، فلن يتم الرجوع إلى V.
!>

LDV

!>          LDV هو INTEGER
!>          البعد الرئيسي للمصفوفة V، LDV >= 1.
!>          إذا كان JOBV = 'V'، فإن LDV >= N.
!>          إذا كان JOBV = 'A'، فإن LDV >= MV.
!>

EPS

!>          EPS هو DOUBLE PRECISION
!>          EPS = DLAMCH('Epsilon')
!>

SFMIN

!>          SFMIN هو DOUBLE PRECISION
!>          SFMIN = DLAMCH('Safe Minimum')
!>

TOL

!>          TOL هو DOUBLE PRECISION
!>          TOL هي عتبة دورانات جاكوبي. لزوج
!>          A(:,p)، A(:,q) من محاور الدوران، يتم تطبيق دوران جاكوبي فقط
!>          إذا كان ABS(COS(angle(A(:,p),A(:,q)))) > TOL.
!>

NSWEEP

!>          NSWEEP هو INTEGER
!>          NSWEEP هو عدد دورات جاكوبي التي سيتم
!>          تنفيذها.
!>

WORK

!>          WORK هو مصفوفة COMPLEX*16، البعد (LWORK)
!>

LWORK

!>          LWORK هو INTEGER
!>          LWORK هو بعد WORK. LWORK >= M.
!>

INFO

!>          INFO هو INTEGER
!>          = 0:  انتهاء ناجح.
!>          < 0:  إذا كان INFO = -i، فإن المعامل i كان له قيمة غير قانونية
!>
