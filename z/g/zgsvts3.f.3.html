subroutine zgsvts3 (integer m, integer p, integer n, complex*16, dimension( lda, * ) a, complex*16, dimension( lda, * ) af, integer lda, complex*16, dimension( ldb, * ) b, complex*16, dimension( ldb, * ) bf, integer ldb, complex*16, dimension( ldu, * ) u, integer ldu, complex*16, dimension( ldv, * ) v, integer ldv, complex*16, dimension( ldq, * ) q, integer ldq, double precision, dimension( * ) alpha, double precision, dimension( * ) beta, complex*16, dimension( ldr, * ) r, integer ldr, integer, dimension( * ) iwork, complex*16, dimension( lwork ) work, integer lwork, double precision, dimension( * ) rwork, double precision, dimension( 6 ) result)

ZGSVTS3

الغرض:

!>
!> ZGSVTS3 تختبر ZGGSVD3، التي تحسب GSVD للمصفوفة M-by-N A
!> والمصفوفة P-by-N B:
!>              U'*A*Q = D1*R و V'*B*Q = D2*R.
!>

المعلمات

M 
!>          M هو عدد صحيح (INTEGER)
!>          عدد صفوف المصفوفة A.  M >= 0.
!>

P

!>          P هو عدد صحيح (INTEGER)
!>          عدد صفوف المصفوفة B.  P >= 0.
!>

N

!>          N هو عدد صحيح (INTEGER)
!>          عدد أعمدة المصفوفتين A و B.  N >= 0.
!>

A

!>          A هو مصفوفة COMPLEX*16، البعد (LDA,M)
!>          المصفوفة A بحجم M-by-N.
!>

AF

!>          AF هو مصفوفة COMPLEX*16، البعد (LDA,N)
!>          تفاصيل GSVD للمصفوفتين A و B، كما تم إرجاعها بواسطة ZGGSVD3،
!>          راجع ZGGSVD3 لمزيد من التفاصيل.
!>

LDA

!>          LDA هو عدد صحيح (INTEGER)
!>          البعد الرئيسي للمصفوفتين A و AF.
!>          LDA >= max( 1,M ).
!>

B

!>          B هو مصفوفة COMPLEX*16، البعد (LDB,P)
!>          عند الدخول، المصفوفة B بحجم P-by-N.
!>

BF

!>          BF هو مصفوفة COMPLEX*16، البعد (LDB,N)
!>          تفاصيل GSVD للمصفوفتين A و B، كما تم إرجاعها بواسطة ZGGSVD3،
!>          راجع ZGGSVD3 لمزيد من التفاصيل.
!>

LDB

!>          LDB هو عدد صحيح (INTEGER)
!>          البعد الرئيسي للمصفوفتين B و BF.
!>          LDB >= max(1,P).
!>

U

!>          U هي مصفوفة COMPLEX*16، البعد (LDU,M)
!>          المصفوفة الوحدوية U بحجم M by M.
!>

LDU

!>          LDU هو عدد صحيح (INTEGER)
!>          البعد الرئيسي للمصفوفة U. LDU >= max(1,M).
!>

V

!>          V هي مصفوفة COMPLEX*16، البعد (LDV,M)
!>          المصفوفة الوحدوية V بحجم P by P.
!>

LDV

!>          LDV هو عدد صحيح (INTEGER)
!>          البعد الرئيسي للمصفوفة V. LDV >= max(1,P).
!>

Q

!>          Q هي مصفوفة COMPLEX*16، البعد (LDQ,N)
!>          المصفوفة الوحدوية Q بحجم N by N.
!>

LDQ

!>          LDQ هو عدد صحيح (INTEGER)
!>          البعد الرئيسي للمصفوفة Q. LDQ >= max(1,N).
!>

ALPHA

!>          ALPHA هي مصفوفة DOUBLE PRECISION، البعد (N)
!>

BETA

!>          BETA هي مصفوفة DOUBLE PRECISION، البعد (N)
!>
!>          أزواج القيم المفردة المعممة لـ A و B،
!>          يتم بناء المصفوفتين القطريتين "D1" و "D2" من
!>          ALPHA و BETA، راجع الروتين الفرعي ZGGSVD3 للتفاصيل.
!>

R

!>          R هي مصفوفة COMPLEX*16، البعد (LDQ,N)
!>          المصفوفة المثلثية العليا R.
!>

LDR

!>          LDR هو عدد صحيح (INTEGER)
!>          البعد الرئيسي للمصفوفة R. LDR >= max(1,N).
!>

IWORK

!>          IWORK هي مصفوفة INTEGER، البعد (N)
!>

WORK

!>          WORK هي مصفوفة COMPLEX*16، البعد (LWORK)
!>

LWORK

!>          LWORK هو عدد صحيح (INTEGER)
!>          بعد المصفوفة WORK،
!>          LWORK >= max(M,P,N)*max(M,P,N).
!>

RWORK

!>          RWORK هي مصفوفة DOUBLE PRECISION، البعد (max(M,P,N))
!>

RESULT

!>          RESULT هي مصفوفة DOUBLE PRECISION، البعد (6)
!>          نسب الاختبار:
!>          RESULT(1) = norm( U'*A*Q - D1*R ) / ( MAX(M,N)*norm(A)*ULP)
!>          RESULT(2) = norm( V'*B*Q - D2*R ) / ( MAX(P,N)*norm(B)*ULP)
!>          RESULT(3) = norm( I - U'*U ) / ( M*ULP )
!>          RESULT(4) = norm( I - V'*V ) / ( P*ULP )
!>          RESULT(5) = norm( I - Q'*Q ) / ( N*ULP )
!>          RESULT(6) = 0        إذا كان ALPHA بترتيب تنازلي؛
!>                    = ULPINV   بخلاف ذلك.
!>
