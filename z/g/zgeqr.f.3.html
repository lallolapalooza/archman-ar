توثيق الوظيفة/الروتين الفرعي

subroutine zgeqr (integer m, integer n, complex*16, dimension( lda, * ) a, integer lda, complex*16, dimension( * ) t, integer tsize, complex*16, dimension( * ) work, integer lwork, integer info)

ZGEQR

الغرض:

!>
!> ZGEQR يحسب تحليل QR لمصفوفة مركبة M-by-N A:
!>
!>    A = Q * ( R ),
!>            ( 0 )
!>
!> حيث:
!>
!>    Q هي مصفوفة متعامدة M-by-M؛
!>    R هي مصفوفة مثلثة عليا N-by-N؛
!>    0 هي مصفوفة صفرية (M-N)-by-N، إذا كان M > N.
!>
!>

المعلمات

M 
!>          M هو عدد صحيح (INTEGER)
!>          عدد صفوف المصفوفة A.  M >= 0.
!>

N

!>          N هو عدد صحيح (INTEGER)
!>          عدد أعمدة المصفوفة A.  N >= 0.
!>

A

!>          A هي مصفوفة معقدة*16 (COMPLEX*16)، أبعادها (LDA,N)
!>          عند الإدخال، المصفوفة A بأبعاد M-by-N.
!>          عند الخروج، العناصر الموجودة على القطر وفي أعلاه تحتوي على المصفوفة شبه المنحرفة العليا R بأبعاد min(M,N)-by-N
!>          (R هي مصفوفة مثلثة عليا إذا كان M >= N)؛
!>          العناصر الموجودة أسفل القطر تستخدم لتخزين جزء من بنية البيانات لتمثيل Q.
!>

LDA

!>          LDA هو عدد صحيح (INTEGER)
!>          البعد الرائد للمصفوفة A.  LDA >= max(1,M).
!>

T

!>          T هي مصفوفة معقدة*16 (COMPLEX*16)، أبعادها (MAX(5,TSIZE))
!>          عند الخروج، إذا كان INFO = 0، فإن T(1) ترجع حجم TSIZE الأمثل (أو الأدنى أو الأمثل، إذا كان الاستعلام مفترضًا). انظر TSIZE للتفاصيل.
!>          المتبقي من T يحتوي على جزء من بنية البيانات المستخدمة لتمثيل Q.
!>          إذا أراد المرء تطبيق أو بناء Q، فإنه يحتاج إلى الاحتفاظ بـ T
!>          (بالإضافة إلى A) وتمريرها إلى روتينات فرعية أخرى.
!>

TSIZE

!>          TSIZE هو عدد صحيح (INTEGER)
!>          إذا كان TSIZE >= 5، فإن بعد المصفوفة T.
!>          إذا كان TSIZE = -1 أو -2، فسيتم افتراض استعلام مساحة العمل. يقوم الروتين
!>          فقط بحساب أحجام مصفوفتي T و WORK، ويعيد هذه
!>          القيم كأول إدخالات في مصفوفتي T و WORK، ولا يتم إصدار أي رسالة خطأ
!>          متعلقة بـ T أو WORK بواسطة XERBLA.
!>          إذا كان TSIZE = -1، يحسب الروتين الحجم الأمثل لـ T لـ
!>          الأداء الأمثل ويعيد هذه القيمة في T(1).
!>          إذا كان TSIZE = -2، يحسب الروتين الحجم الأدنى لـ T و
!>          يعيد هذه القيمة في T(1).
!>

WORK

!>          (مساحة عمل) مصفوفة معقدة*16 (COMPLEX*16)، أبعادها (MAX(1,LWORK))
!>          عند الخروج، إذا كان INFO = 0، فإن WORK(1) تحتوي على الحجم الأمثل (أو الأدنى أو الأمثل، إذا كان الاستعلام مفترضًا) LWORK.
!>          انظر LWORK للتفاصيل.
!>

LWORK

!>          LWORK هو عدد صحيح (INTEGER)
!>          بعد المصفوفة WORK. LWORK >= 1.
!>          إذا كان LWORK = -1 أو -2، فسيتم افتراض استعلام مساحة العمل. يقوم الروتين
!>          فقط بحساب أحجام مصفوفتي T و WORK، ويعيد هذه
!>          القيم كأول إدخالات في مصفوفتي T و WORK، ولا يتم إصدار أي رسالة خطأ
!>          متعلقة بـ T أو WORK بواسطة XERBLA.
!>          إذا كان LWORK = -1، يحسب الروتين الحجم الأمثل لـ WORK لـ
!>          الأداء الأمثل ويعيد هذه القيمة في WORK(1).
!>          إذا كان LWORK = -2، يحسب الروتين الحجم الأدنى لـ WORK و
!>          يعيد هذه القيمة في WORK(1).
!>

INFO

!>          INFO هو عدد صحيح (INTEGER)
!>          = 0:  انتهاء ناجح
!>          < 0:  إذا كان INFO = -i، فإن الوسيط i كان له قيمة غير قانونية
!>

تفاصيل إضافية

!>
!> الهدف من الواجهة هو منح أقصى قدر من الحرية للمطورين لـ
!> إنشاء أي خوارزمية تحليل QR يرغبون فيها. المصفوفة المثلثية
!> (شبه المنحرفة) R يجب تخزينها في الجزء العلوي من A. الجزء السفلي من A
!> والمصفوفة T يمكن استخدامها لتخزين أي معلومات ذات صلة لتطبيق أو
!> بناء العامل Q. يمكن التخلص من مصفوفة WORK بأمان بعد الخروج.
!>
!> تحذير: لا ينبغي للمرء أن يتوقع أن تكون أحجام T و WORK متساوية من تنفيذ
!> LAPACK إلى آخر، أو حتى من تنفيذ إلى آخر.
!> هناك حاجة لاستعلام مساحة العمل (لـ T و WORK) في كل تنفيذ. ومع ذلك،
!> بالنسبة لتنفيذ معين، يكون حجم T و WORK ثابتين ولن يتغير
!> من استعلام إلى آخر.
!>
!>

تفاصيل إضافية خاصة بتطبيق LAPACK هذا:

!>
!> هذه التفاصيل خاصة بتطبيق LAPACK هذا. يجب على المستخدمين عدم
!> اعتبارها أمرًا مسلمًا به. قد تتغير هذه التفاصيل في المستقبل، ومن غير المحتمل
!> أن تكون صحيحة لتطبيق LAPACK آخر. هذه التفاصيل ذات صلة إذا أراد المرء
!> محاولة فهم الكود. وهي ليست جزءًا من الواجهة.
!>
!> في هذا الإصدار،
!>
!>          T(2): حجم كتلة الصف (MB)
!>          T(3): حجم كتلة العمود (NB)
!>          T(6:TSIZE): بنية البيانات المطلوبة لـ Q، التي تم حسابها بواسطة
!>                           ZLATSQR أو ZGEQRT
!>
!>  اعتمادًا على أبعاد المصفوفة M و N، وأحجام الكتل للصفوف والأعمدة
!>  MB و NB التي تعيدها ILAENV، سيستخدم ZGEQR إما
!>  ZLATSQR (إذا كانت المصفوفة طويلة ورفيعة) أو ZGEQRT لحساب
!>  تحليل QR.
!>
!>
