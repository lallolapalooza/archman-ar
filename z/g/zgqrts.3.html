subroutine zgqrts (integer n, integer m, integer p, complex*16, dimension( lda, * ) a, complex*16, dimension( lda, * ) af, complex*16, dimension( lda, * ) q, complex*16, dimension( lda, * ) r, integer lda, complex*16, dimension( * ) taua, complex*16, dimension( ldb, * ) b, complex*16, dimension( ldb, * ) bf, complex*16, dimension( ldb, * ) z, complex*16, dimension( ldb, * ) t, complex*16, dimension( ldb, * )bwk, integer ldb, complex*16, dimension( * ) taub, complex*16, dimension( lwork ) work, integer lwork, double precision, dimension( * ) rwork, double precision, dimension( 4 ) result)

ZGQRTS

الغرض:

!>
!> ZGQRTS تختبر ZGGQRF، التي تحسب تحليل GQR لمصفوفة N-by-M A ومصفوفة N-by-P B:
!> A = Q*R و B = Q*T*Z.
!>

المعلمات

N 
!>          N هو عدد صحيح
!>          عدد صفوف المصفوفات A و B. N >= 0.
!>

M

!>          M هو عدد صحيح
!>          عدد أعمدة المصفوفة A. M >= 0.
!>

P

!>          P هو عدد صحيح
!>          عدد أعمدة المصفوفة B. P >= 0.
!>

A

!>          A هي مصفوفة COMPLEX*16، البعد (LDA,M)
!>          المصفوفة A ذات الأبعاد N-by-M.
!>

AF

!>          AF هي مصفوفة COMPLEX*16، البعد (LDA,N)
!>          تفاصيل تحليل GQR للمصفوفتين A و B، كما تم إرجاعها
!>          بواسطة ZGGQRF، راجع CGGQRF لمزيد من التفاصيل.
!>

Q

!>          Q هي مصفوفة COMPLEX*16، البعد (LDA,N)
!>          المصفوفة الوحدوية Q ذات الأبعاد M-by-M.
!>

R

!>          R هي مصفوفة COMPLEX*16، البعد (LDA,MAX(M,N))
!>

LDA

!>          LDA هو عدد صحيح
!>          البعد الرئيسي للمصفوفات A، AF، R و Q.
!>          LDA >= max(M,N).
!>

TAUA

!>          TAUA هي مصفوفة COMPLEX*16، البعد (min(M,N))
!>          العوامل القياسية للمنعكسات الأولية، كما تم إرجاعها
!>          بواسطة ZGGQRF.
!>

B

!>          B هي مصفوفة COMPLEX*16، البعد (LDB,P)
!>          عند الإدخال، المصفوفة B ذات الأبعاد N-by-P.
!>

BF

!>          BF هي مصفوفة COMPLEX*16، البعد (LDB,N)
!>          تفاصيل تحليل GQR للمصفوفتين A و B، كما تم إرجاعها
!>          بواسطة ZGGQRF، راجع CGGQRF لمزيد من التفاصيل.
!>

Z

!>          Z هي مصفوفة COMPLEX*16، البعد (LDB,P)
!>          المصفوفة الوحدوية Z ذات الأبعاد P-by-P.
!>

T

!>          T هي مصفوفة COMPLEX*16، البعد (LDB,max(P,N))
!>

BWK

!>          BWK هي مصفوفة COMPLEX*16، البعد (LDB,N)
!>

LDB

!>          LDB هو عدد صحيح
!>          البعد الرئيسي للمصفوفات B، BF، Z و T.
!>          LDB >= max(P,N).
!>

TAUB

!>          TAUB هي مصفوفة COMPLEX*16، البعد (min(P,N))
!>          العوامل القياسية للمنعكسات الأولية، كما تم إرجاعها
!>          بواسطة DGGRQF.
!>

WORK

!>          WORK هي مصفوفة COMPLEX*16، البعد (LWORK)
!>

LWORK

!>          LWORK هو عدد صحيح
!>          بعد المصفوفة WORK، LWORK >= max(N,M,P)**2.
!>

RWORK

!>          RWORK هي مصفوفة DOUBLE PRECISION، البعد (max(N,M,P))
!>

RESULT

!>          RESULT هي مصفوفة DOUBLE PRECISION، البعد (4)
!>          نسب الاختبار:
!>            RESULT(1) = norm( R - Q'*A ) / ( MAX(M,N)*norm(A)*ULP)
!>            RESULT(2) = norm( T*Z - Q'*B ) / (MAX(P,N)*norm(B)*ULP)
!>            RESULT(3) = norm( I - Q'*Q ) / ( M*ULP )
!>            RESULT(4) = norm( I - Z'*Z ) / ( P*ULP )
!>
