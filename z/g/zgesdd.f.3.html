وثائق الوظيفة/الروتين الفرعي

subroutine zgesdd (character jobz, integer m, integer n, complex*16, dimension( lda, * ) a, integer lda, double precision, dimension( * ) s, complex*16, dimension( ldu, * ) u, integer ldu, complex*16, dimension( ldvt, * ) vt, integer ldvt, complex*16, dimension( * ) work, integer lwork, double precision, dimension( * ) rwork, integer, dimension( * ) iwork, integer info)

ZGESDD

الغرض:

!>
!> ZGESDD تحسب التحلل القيمي المفرد (SVD) لمصفوفة معقدة A بحجم M*N،
!> محسوبة اختيارياً المتجهات المفردة اليسرى و/أو اليمنى، باستخدام طريقة
!> فرق-واجابه. التحلل القيمي المفرد مكتوب بالشكل
!>
!>      A = U * SIGMA * conjugate-transpose(V)
!>
!> حيث SIGMA مصفوفة بحجم M*N وهي صفر باستثناء العناصر القطرية
!> min(m,n)، U مصفوفة وحدوية بحجم M*M، و V مصفوفة وحدوية بحجم N*N.
!> العناصر القطرية في SIGMA هي القيم المفردة للمصفوفة A؛ وهي حقيقية
!> وغير سالبة، ويتم إرجاعها بترتيب تنازلي. الأعمدة min(m,n) الأولى في
!> U و V هي المتجهات المفردة اليسرى واليمنى للمصفوفة A.
!>
!> لاحظ أن الروتين يرجع VT = V**H، وليس V.
!>
!>

المعلمات

JOBZ 
!>          JOBZ is CHARACTER*1
!>          يحدد خيارات حساب كل أو جزء من المصفوفة U:
!>          = 'A':  تُرجع جميع أعمدة U البالغ عددها M وجميع صفوف V**H البالغ عددها N
!>                  في المصفوفات U و VT؛
!>          = 'S':  تُرجع الأعمدة min(M,N) الأولى من U والصفوف min(M,N) الأولى من V**H
!>                  في المصفوفات U و VT؛
!>          = 'O':  إذا كان M >= N، تُستبدل الأعمدة N الأولى من U في المصفوفة A وتُرجع
!>                  جميع صفوف V**H في المصفوفة VT؛
!>                  بخلاف ذلك، تُرجع جميع أعمدة U في المصفوفة U وتُستبدل الصفوف M الأولى من V**H
!>                  في المصفوفة A؛
!>          = 'N':  لا يتم حساب أي أعمدة من U أو صفوف من V**H.
!>

M

!>          M is INTEGER
!>          عدد صفوف المصفوفة المدخلة A.  M >= 0.
!>

N

!>          N is INTEGER
!>          عدد أعمدة المصفوفة المدخلة A.  N >= 0.
!>

A

!>          A is COMPLEX*16 array, dimension (LDA,N)
!>          عند الإدخال، المصفوفة A بحجم M*N.
!>          عند الخروج،
!>          إذا كان JOBZ = 'O'، تُستبدل A بالأعمدة N الأولى من U (المتجهات المفردة
!>                          اليسرى، مخزنة كأعمدة) إذا كان M >= N؛
!>                          تُستبدل A بالصفوف M الأولى من V**H (المتجهات المفردة
!>                          اليمنى، مخزنة كصفوف) بخلاف ذلك.
!>          إذا كان JOBZ .ne. 'O'، يتم تدمير محتويات A.
!>

LDA

!>          LDA is INTEGER
!>          البعد الرائد للمصفوفة A.  LDA >= max(1,M).
!>

S

!>          S is DOUBLE PRECISION array, dimension (min(M,N))
!>          القيم المفردة للمصفوفة A، مرتبة بحيث S(i) >= S(i+1).
!>

U

!>          U is COMPLEX*16 array, dimension (LDU,UCOL)
!>          UCOL = M إذا كان JOBZ = 'A' أو JOBZ = 'O' و M < N؛
!>          UCOL = min(M,N) إذا كان JOBZ = 'S'.
!>          إذا كان JOBZ = 'A' أو JOBZ = 'O' و M < N، تحتوي U على المصفوفة
!>                          الوحدوية U بحجم M*M؛
!>          إذا كان JOBZ = 'S'، تحتوي U على الأعمدة min(M,N) الأولى من U
!>                          (المتجهات المفردة اليسرى، مخزنة كأعمدة)؛
!>          إذا كان JOBZ = 'O' و M >= N، أو JOBZ = 'N'، لا يتم استخدام U.
!>

LDU

!>          LDU is INTEGER
!>          البعد الرائد للمصفوفة U.  LDU >= 1؛
!>          إذا كان JOBZ = 'S' أو 'A' أو JOBZ = 'O' و M < N، فإن LDU >= M.
!>

VT

!>          VT is COMPLEX*16 array, dimension (LDVT,N)
!>          إذا كان JOBZ = 'A' أو JOBZ = 'O' و M >= N، تحتوي VT على المصفوفة
!>                          الوحدوية V**H بحجم N*N؛
!>          إذا كان JOBZ = 'S'، تحتوي VT على الصفوف min(M,N) الأولى من
!>                          V**H (المتجهات المفردة اليمنى، مخزنة كصفوف)؛
!>          إذا كان JOBZ = 'O' و M < N، أو JOBZ = 'N'، لا يتم استخدام VT.
!>

LDVT

!>          LDVT is INTEGER
!>          البعد الرائد للمصفوفة VT.  LDVT >= 1؛
!>          إذا كان JOBZ = 'A' أو JOBZ = 'O' و M >= N، فإن LDVT >= N؛
!>          إذا كان JOBZ = 'S'، فإن LDVT >= min(M,N).
!>

WORK

!>          WORK is COMPLEX*16 array, dimension (MAX(1,LWORK))
!>          عند الخروج، إذا كان INFO = 0، تُرجع WORK(1) قيمة LWORK المثلى.
!>

LWORK

!>          LWORK is INTEGER
!>          بعد المصفوفة WORK. LWORK >= 1.
!>          إذا كان LWORK = -1، يُفترض استعلام عن مساحة العمل. يتم حساب
!>          الحجم الأمثل لمصفوفة WORK وتخزينه في WORK(1)، ولا يتم إجراء
!>          أي عمل آخر بخلاف التحقق من الوسائط.
!>
!>          لنفترض mx = max(M,N) و mn = min(M,N).
!>          إذا كان JOBZ = 'N'، فإن LWORK >= 2*mn + mx.
!>          إذا كان JOBZ = 'O'، فإن LWORK >= 2*mn*mn + 2*mn + mx.
!>          إذا كان JOBZ = 'S'، فإن LWORK >= mn*mn + 3*mn.
!>          إذا كان JOBZ = 'A'، فإن LWORK >= mn*mn + 2*mn + mx.
!>          هذه ليست حدودًا دنيا دقيقة في جميع الحالات؛ انظر التعليقات داخل الكود.
!>          للحصول على أداء جيد، يجب أن يكون LWORK بشكل عام أكبر؛
!>          يُنصح بالاستعلام.
!>

RWORK

!>          RWORK is DOUBLE PRECISION array, dimension (MAX(1,LRWORK))
!>          لنفترض mx = max(M,N) و mn = min(M,N).
!>          إذا كان JOBZ = 'N'، فإن LRWORK >= 5*mn (LAPACK <= 3.6 يحتاج 7*mn)؛
!>          else if mx >> mn, LRWORK >= 5*mn*mn + 5*mn؛
!>          else              LRWORK >= max( 5*mn*mn + 5*mn,
!>                                           2*mx*mn + 2*mn*mn + mn ).
!>

IWORK

!>          IWORK is INTEGER array, dimension (8*min(M,N))
!>

INFO

!>          INFO is INTEGER
!>          <  0:  إذا كان INFO = -i، فإن الوسيط i كان لديه قيمة غير قانونية.
!>          = -4:  إذا كانت A تحتوي على مدخل NAN.
!>          >  0:  لم تتقارب عملية تحديث DBDSDC.
!>          =  0:  انتهاء ناجح.
!>
