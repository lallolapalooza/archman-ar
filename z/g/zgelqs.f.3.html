توثيق الوظيفة/الروتين الفرعي

subroutine zgelqs (integer m, integer n, integer nrhs, complex*16, dimension( lda, * ) a, integer lda, complex*16, dimension( * ) tau, complex*16, dimension( ldb, * ) b, integer ldb, complex*16, dimension( lwork ) work, integer lwork, integer info)

ZGELQS

الغرض:

!>
!> حساب حل الحد الأدنى للمعيار
!>     min || A*X - B ||
!> باستخدام تحليل LQ
!>     A = L*Q
!> الذي تم حسابه بواسطة ZGELQF.
!>

المعلمات

M 
!>          M هو عدد صحيح
!>          عدد الصفوف للمصفوفة A.  M >= 0.
!>

N

!>          N هو عدد صحيح
!>          عدد الأعمدة للمصفوفة A.  N >= M >= 0.
!>

NRHS

!>          NRHS هو عدد صحيح
!>          عدد أعمدة B.  NRHS >= 0.
!>

A

!>          A هو مصفوفة COMPLEX*16، البعد (LDA,N)
!>          تفاصيل تحليل LQ للمصفوفة الأصلية A كما
!>          تم إرجاعها بواسطة ZGELQF.
!>

LDA

!>          LDA هو عدد صحيح
!>          البعد الرئيسي للمصفوفة A.  LDA >= M.
!>

TAU

!>          TAU هو مصفوفة COMPLEX*16، البعد (M)
!>          تفاصيل المصفوفة المتعامدة Q.
!>

B

!>          B هو مصفوفة COMPLEX*16، البعد (LDB,NRHS)
!>          عند الإدخال، مصفوفة الجانب الأيمن B ذات الأبعاد m-by-nrhs.
!>          عند الخروج، مصفوفة الحل X ذات الأبعاد n-by-nrhs.
!>

LDB

!>          LDB هو عدد صحيح
!>          البعد الرئيسي للمصفوفة B. LDB >= N.
!>

WORK

!>          WORK هو مصفوفة COMPLEX*16، البعد (LWORK)
!>

LWORK

!>          LWORK هو عدد صحيح
!>          طول المصفوفة WORK.  يجب أن يكون LWORK على الأقل NRHS،
!>          ويجب أن يكون على الأقل NRHS*NB، حيث NB هو حجم الكتلة
!>          لهذه البيئة.
!>

INFO

!>          INFO هو عدد صحيح
!>          = 0: خروج ناجح
!>          < 0: إذا كان INFO = -i، فإن المعامل i كان له قيمة غير قانونية
!>
