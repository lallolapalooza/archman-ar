subroutine zggesx (character jobvsl, character jobvsr, character sort, external selctg, character sense, integer n, complex*16, dimension( lda, * ) a, integer lda, complex*16, dimension( ldb, * ) b, integer ldb, integer sdim, complex*16, dimension( * ) alpha, complex*16, dimension( * ) beta, complex*16, dimension( ldvsl, * ) vsl, integer ldvsl, complex*16, dimension( ldvsr, * ) vsr, integer ldvsr, double precision, dimension( 2 ) rconde, double precision, dimension( 2 ) rcondv, complex*16, dimension( * ) work, integer lwork, double precision, dimension( * ) rwork, integer, dimension( * ) iwork, integer liwork, logical, dimension( * ) bwork, integer info)

ZGGESX تحسب القيم الذاتية، شكل شور، وبشكل اختياري، مصفوفة متجهات شور لمصفوفات GE

الغرض:

!>
!> ZGGESX تحسب، لزوج من المصفوفات المعقدة غير المتماثلة N-by-N (A,B)، القيم الذاتية المعممة، شكل شور المعقد (S,T)،
!> وبشكل اختياري، مصفوفات متجهات شور اليسرى و/أو اليمنى (VSL و VSR). هذا يعطي التحلل المعمم لشور
!>
!>      (A,B) = ( (VSL) S (VSR)**H, (VSL) T (VSR)**H )
!>
!> حيث (VSR)**H هو المرافق المتعامد لـ VSR.
!>
!> وبشكل اختياري، فإنها تقوم أيضًا بترتيب القيم الذاتية بحيث تظهر مجموعة مختارة من القيم الذاتية في القطاعات القطرية الأمامية
!> للمصفوفة العلوية المثلثية S والمصفوفة العلوية المثلثية T؛ وتحسب
!> رقم شرط عكسي لمتوسط القيم الذاتية المختارة (RCONDE)؛ وتحسب رقم شرط عكسي
!> للفضاءات المتلاشية اليمنى واليسرى المقابلة للقيم الذاتية المختارة
!> (RCONDV). تشكل الأعمدة الأمامية لـ VSL و VSR بعد ذلك
!> أساسًا متعامدًا للفضاءات الذاتية اليسرى واليمنى المقابلة
!> (فضاءات التلاشي).
!>
!> القيمة الذاتية المعممة لزوج من المصفوفات (A,B) هي قيمة عددية w
!> أو نسبة alpha/beta = w، بحيث تكون A - w*B شاذة. وغالباً ما يتم تمثيلها
!> كزوج (alpha,beta)، حيث يوجد تفسير معقول لـ beta=0 أو لكليهما صفر.
!>
!> الزوج من المصفوفات (S,T) في شكل شور المعمم المعقد إذا كانت T
!> علوية مثلثية مع أقطار غير سالبة و S علوية مثلثية.
!>

المعلمات

JOBVSL 
!>          JOBVSL هو CHARACTER*1
!>          = 'N':  لا تحسب متجهات شور اليسرى؛
!>          = 'V':  احسب متجهات شور اليسرى.
!>

JOBVSR

!>          JOBVSR هو CHARACTER*1
!>          = 'N':  لا تحسب متجهات شور اليمنى؛
!>          = 'V':  احسب متجهات شور اليمنى.
!>

SORT

!>          SORT هو CHARACTER*1
!>          يحدد ما إذا كان سيتم ترتيب القيم الذاتية على القطر
!>          لشكل شور المعمم أم لا.
!>          = 'N':  القيم الذاتية ليست مرتبة؛
!>          = 'S':  القيم الذاتية مرتبة (انظر SELCTG).
!>

SELCTG

!>          SELCTG هي وظيفة LOGICAL لمعاملين COMPLEX*16
!>          يجب تعريف SELCTG كـ EXTERNAL في الروتين الفرعي المستدعي.
!>          إذا كان SORT = 'N'، فلن يتم الرجوع إلى SELCTG.
!>          إذا كان SORT = 'S'، يتم استخدام SELCTG لتحديد القيم الذاتية لفرزها
!>          إلى أعلى اليسار من شكل شور.
!>          لاحظ أن القيمة الذاتية المعقدة المختارة قد لا تفي بعد ذلك
!>          بـ SELCTG(ALPHA(j),BETA(j)) = .TRUE. بعد الترتيب، حيث أن
!>          الترتيب قد يغير قيمة القيم الذاتية المعقدة
!>          (خاصة إذا كانت القيمة الذاتية غير مستقرة)، في هذه الحالة
!>          يتم تعيين INFO إلى N+3 انظر INFO أدناه).
!>

SENSE

!>          SENSE هو CHARACTER*1
!>          يحدد أرقام الشروط العكسية التي يتم حسابها.
!>          = 'N': لا شيء محسوب؛
!>          = 'E': محسوب لمتوسط القيم الذاتية المختارة فقط؛
!>          = 'V': محسوب لفضاءات التلاشي المختارة فقط؛
!>          = 'B': محسوب لكليهما.
!>          إذا كان SENSE = 'E'، 'V'، أو 'B'، فيجب أن يكون SORT = 'S'.
!>

N

!>          N هو INTEGER
!>          ترتيب المصفوفات A، B، VSL، و VSR. N >= 0.
!>

A

!>          A هي مصفوفة COMPLEX*16، البعد (LDA, N)
!>          عند الدخول، الجزء الأول من زوج المصفوفات.
!>          عند الخروج، تم استبدال A بشكل شور المعمم الخاص بها
!>          S.
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرئيسي لـ A. LDA >= max(1,N).
!>

B

!>          B هي مصفوفة COMPLEX*16، البعد (LDB, N)
!>          عند الدخول، الجزء الثاني من زوج المصفوفات.
!>          عند الخروج، تم استبدال B بشكل شور المعمم الخاص بها
!>          T.
!>

LDB

!>          LDB هو INTEGER
!>          البعد الرئيسي لـ B. LDB >= max(1,N).
!>

SDIM

!>          SDIM هو INTEGER
!>          إذا كان SORT = 'N'، فإن SDIM = 0.
!>          إذا كان SORT = 'S'، فإن SDIM = عدد القيم الذاتية (بعد الترتيب)
!>          التي يكون فيها SELCTG صحيحًا.
!>

ALPHA

!>          ALPHA هي مصفوفة COMPLEX*16، البعد (N)
!>

BETA

!>          BETA هي مصفوفة COMPLEX*16، البعد (N)
!>          عند الخروج، ALPHA(j)/BETA(j)، j=1,...,N، ستكون
!>          القيم الذاتية المعممة. ALPHA(j) و BETA(j)، j=1,...,N هي
!>          الأقطار لـ شكل شور المعقد (S,T). سيكون BETA(j)
!>          غير سالب حقيقي.
!>
!>          ملاحظة: قد تتجاوز نسب ALPHA(j)/BETA(j) أو تقل بسهولة،
!>          وقد يكون BETA(j) صفرًا. لذلك، يجب على المستخدم
!>          تجنب حساب نسبة alpha/beta بشكل سطحي.
!>          ومع ذلك، سيكون ALPHA دائمًا أقل من، وعادة ما يكون قابلاً للمقارنة
!>          مع norm(A) في الحجم، و BETA دائمًا أقل من، وعادة ما يكون قابلاً للمقارنة
!>          مع norm(B).
!>

VSL

!>          VSL هي مصفوفة COMPLEX*16، البعد (LDVSL,N)
!>          إذا كان JOBVSL = 'V'، فسوف تحتوي VSL على متجهات شور اليسرى.
!>          لا يتم الرجوع إليها إذا كان JOBVSL = 'N'.
!>

LDVSL

!>          LDVSL هو INTEGER
!>          البعد الرئيسي للمصفوفة VSL. LDVSL >=1، و
!>          إذا كان JOBVSL = 'V'، فإن LDVSL >= N.
!>

VSR

!>          VSR هي مصفوفة COMPLEX*16، البعد (LDVSR,N)
!>          إذا كان JOBVSR = 'V'، فسوف تحتوي VSR على متجهات شور اليمنى.
!>          لا يتم الرجوع إليها إذا كان JOBVSR = 'N'.
!>

LDVSR

!>          LDVSR هو INTEGER
!>          البعد الرئيسي للمصفوفة VSR. LDVSR >= 1، و
!>          إذا كان JOBVSR = 'V'، فإن LDVSR >= N.
!>

RCONDE

!>          RCONDE هي مصفوفة DOUBLE PRECISION، البعد ( 2 )
!>          إذا كان SENSE = 'E' أو 'B'، تحتوي RCONDE(1) و RCONDE(2) على
!>          أرقام الشروط العكسية لمتوسط القيم الذاتية المختارة
es.
!>          غير مستخدمة إذا كان SENSE = 'N' أو 'V'.
!>

RCONDV

!>          RCONDV هي مصفوفة DOUBLE PRECISION، البعد ( 2 )
!>          إذا كان SENSE = 'V' أو 'B'، فإن RCONDV(1) و RCONDV(2) يحتويان على
!>          الشرط العكسي للفضاءات الفرعية المختارة.
!>          غير مستخدمة إذا كان SENSE = 'N' أو 'E'.
!>

WORK

!>          WORK هي مصفوفة COMPLEX*16، البعد (MAX(1,LWORK))
!>          عند الخروج، إذا كان INFO = 0، فإن WORK(1) يعيد LWORK الأمثل.
!>

LWORK

!>          LWORK هو INTEGER
!>          بعد المصفوفة WORK.
!>          إذا كان N = 0، LWORK >= 1، وإلا إذا كان SENSE = 'E'، 'V'، أو 'B'،
!>          LWORK >= MAX(1,2*N,2*SDIM*(N-SDIM))، وإلا
!>          LWORK >= MAX(1,2*N). لاحظ أن 2*SDIM*(N-SDIM) <= N*N/2.
!>          لاحظ أيضًا أن الخطأ يتم إرجاعه فقط إذا كان
!>          LWORK < MAX(1,2*N)، ولكن إذا كان SENSE = 'E' أو 'V' أو 'B' فقد لا يكون
!>          هذا كبيرًا بما يكفي.
!>
!>          إذا كان LWORK = -1، فسيتم افتراض استعلام مساحة عمل؛ تقوم الروتين
!>          فقط بحساب الحد الأقصى للحجم الأمثل لمصفوفة WORK
!>          والحد الأدنى لحجم مصفوفة IWORK، وتعيد هذه
!>          القيم كأول إدخالات في مصفوفتي WORK و IWORK،
!>          ولا يصدر XERBLA أي رسالة خطأ تتعلق بـ LWORK أو LIWORK.
!>

RWORK

!>          RWORK هي مصفوفة DOUBLE PRECISION، البعد ( 8*N )
!>          مساحة عمل حقيقية.
!>

IWORK

!>          IWORK هي مصفوفة INTEGER، البعد (MAX(1,LIWORK))
!>          عند الخروج، إذا كان INFO = 0، فإن IWORK(1) يعيد LIWORK الأدنى.
!>

LIWORK

!>          LIWORK هو INTEGER
!>          بعد المصفوفة IWORK.
!>          إذا كان SENSE = 'N' أو N = 0، LIWORK >= 1، وإلا
!>          LIWORK >= N+2.
!>
!>          إذا كان LIWORK = -1، فسيتم افتراض استعلام مساحة عمل؛ تقوم الروتين
!>          فقط بحساب الحد الأقصى للحجم الأمثل لمصفوفة
!>          WORK والحد الأدنى لحجم مصفوفة IWORK، وتعيد هذه
!>          القيم كأول إدخالات في مصفوفتي WORK و IWORK،
!>          ولا يصدر XERBLA أي رسالة خطأ تتعلق بـ LWORK أو LIWORK.
!>

BWORK

!>          BWORK هي مصفوفة LOGICAL، البعد (N)
!>          غير مستخدمة إذا كان SORT = 'N'.
!>

INFO

!>          INFO هو INTEGER
!>          = 0:  انتهاء ناجح
!>          < 0:  إذا كان INFO = -i، فإن الوسيط i كان لديه قيمة غير قانونية.
!>          = 1,...,N:
!>                فشلت تكرارات QZ. (A,B) ليست في شكل Schur
!>                ولكن يجب أن تكون ALPHA(j) و BETA(j) صحيحة لـ
!>                j=INFO+1,...,N.
!>          > N:  =N+1: فشلت تكرارات QZ بخلاف ذلك في ZHGEQZ
!>                =N+2: بعد إعادة الترتيب، قامت تقريبات الأخطاء بتغيير قيم
!>                      بعض القيم الذاتية المعقدة بحيث لم تعد
!>                      القيم الذاتية الرائدة في شكل Schur العام
!>                      تفي بـ SELCTG=.TRUE. قد يكون هذا ناتجًا أيضًا
!>                      عن التحجيم.
!>                =N+3: فشلت إعادة الترتيب في ZTGSEN.
!>
