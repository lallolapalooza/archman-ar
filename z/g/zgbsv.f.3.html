subroutine zgbsv (integer n, integer kl, integer ku, integer nrhs, complex*16, dimension( ldab, * ) ab, integer ldab, integer, dimension( * ) ipiv, complex*16, dimension( ldb, * ) b, integer ldb, integer info)

ZGBSV تحسب الحل لنظام المعادلات الخطية A * X = B لمصفوفات GB (برنامج تشغيل بسيط)

الغرض:

!>
!> ZGBSV تحسب الحل لنظام معادلات خطية معقدة
!> A * X = B، حيث A مصفوفة نطاق من الرتبة N مع KL تحت الأقطار
!> و KU فوق الأقطار، و X و B مصفوفتان N-by-NRHS.
!>
!> يتم استخدام تحلل LU مع التقسيم الجزئي وتبديلات الصفوف
!> لتحليل A كـ A = L * U، حيث L هو حاصل ضرب مصفوفات التبديل
!> والمثلثية السفلية الوحيدة مع KL تحت الأقطار، و U
!> مثلثية عليا مع KL+KU فوق الأقطار.  ثم يتم استخدام الشكل المحلل لـ A
!> لحل نظام المعادلات A * X = B.
!>

المعلمات

N 
!>          N هو عدد صحيح
!>          عدد المعادلات الخطية، أي رتبة
!>          المصفوفة A.  N >= 0.
!>

KL

!>          KL هو عدد صحيح
!>          عدد تحت الأقطار داخل نطاق A.  KL >= 0.
!>

KU

!>          KU هو عدد صحيح
!>          عدد فوق الأقطار داخل نطاق A.  KU >= 0.
!>

NRHS

!>          NRHS هو عدد صحيح
!>          عدد الأطراف اليمنى، أي عدد الأعمدة
!>          للمصفوفة B.  NRHS >= 0.
!>

AB

!>          AB هي مصفوفة COMPLEX*16، ذات بعد (LDAB,N)
!>          عند الدخول، المصفوفة A بتخزين النطاق، في الصفوف KL+1 إلى
!>          2*KL+KU+1؛ الصفوف 1 إلى KL من المصفوفة لا تحتاج إلى تعيين.
!>          يتم تخزين العمود j من A في العمود j من
!>          المصفوفة AB كما يلي:
!>          AB(KL+KU+1+i-j,j) = A(i,j) for max(1,j-KU)<=i<=min(N,j+KL)
!>          عند الخروج، تفاصيل التحليل: يتم تخزين U كـ
!>          مصفوفة نطاق عليا مثلثة مع KL+KU فوق الأقطار في
!>          الصفوف 1 إلى KL+KU+1، ويتم تخزين المضاعفات المستخدمة أثناء
!>          التحليل في الصفوف KL+KU+2 إلى 2*KL+KU+1.
!>          انظر أدناه لمزيد من التفاصيل.
!>

LDAB

!>          LDAB هو عدد صحيح
!>          البعد الرائد للمصفوفة AB.  LDAB >= 2*KL+KU+1.
!>

IPIV

!>          IPIV هي مصفوفة عدد صحيح، ذات بعد (N)
!>          فهارس التقسيم التي تحدد مصفوفة التبديل P؛
!>          تم تبديل الصف i من المصفوفة مع الصف IPIV(i).
!>

B

!>          B هي مصفوفة COMPLEX*16، ذات بعد (LDB,NRHS)
!>          عند الدخول، مصفوفة الطرف الأيمن B ذات البعد N-by-NRHS.
!>          عند الخروج، إذا كان INFO = 0، فإن مصفوفة الحل X ذات البعد N-by-NRHS.
!>

LDB

!>          LDB هو عدد صحيح
!>          البعد الرائد للمصفوفة B.  LDB >= max(1,N).
!>

INFO

!>          INFO هو عدد صحيح
!>          = 0:  خروج ناجح
!>          < 0:  إذا كان INFO = -i، فإن الوسيط i كان له قيمة غير قانونية
!>          > 0:  إذا كان INFO = i، فإن U(i,i) يساوي صفرًا تمامًا. تم إكمال التحليل،
!>                ولكن العامل U منفرد تمامًا، ولم يتم حساب الحل.
!>
