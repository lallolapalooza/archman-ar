subroutine zgbequb (integer m, integer n, integer kl, integer ku, complex*16, dimension( ldab, * ) ab, integer ldab, double precision, dimension( * ) r, double precision, dimension( * ) c, double precision rowcnd, double precision colcnd, double precision amax, integer info)

ZGBEQUB

الغرض:

!>
!> ZGBEQUB تحسب قياسات الصفوف والأعمدة المقصودة لمعادلة مصفوفة A حجمها M × N
!> وتقليل عدد شرطها. تُرجع R عوامل قياس الصفوف و C عوامل قياس الأعمدة،
!> مختارة لمحاولة جعل أكبر عنصر في كل صف وعمود من المصفوفة B بعناصر
!> B(i,j)=R(i)*A(i,j)*C(j) لها قيمة مطلقة لا تتجاوز الأساس.
!>
!> R(i) و C(j) مقيدة لتكون قوة للأساس بين
!> SMLNUM = أصغر عدد آمن و BIGNUM = أكبر عدد آمن. استخدام
!> عوامل القياس هذه لا يضمن تقليل عدد شرط A ولكنه يعمل بشكل جيد في الممارسة العملية.
!>
!> هذا الروتين يختلف عن ZGEEQU في تقييد عوامل القياس
!> لتكون قوة للأساس. باستثناء الحمل الزائد والنقصان، القياس بواسطة
!> هذه العوامل لا يدخل أي أخطاء تقريب إضافية. ومع ذلك، فإن
!> مقادير المدخلات المقاسة لم تعد تقريبًا 1 ولكنها تقع
!> بين sqrt(radix) و 1/sqrt(radix).
!>

المعلمات

M 
!>          M هو عدد صحيح (INTEGER)
!>          عدد صفوف المصفوفة A. M >= 0.
!>

N

!>          N هو عدد صحيح (INTEGER)
!>          عدد أعمدة المصفوفة A. N >= 0.
!>

KL

!>          KL هو عدد صحيح (INTEGER)
!>          عدد الأقطار الفرعية داخل نطاق A. KL >= 0.
!>

KU

!>          KU هو عدد صحيح (INTEGER)
!>          عدد الأقطار الفرعية الفائقة داخل نطاق A. KU >= 0.
!>

AB

!>          AB هو مصفوفة ذات أرقام مركبة*16 (COMPLEX*16)، بُعدها (LDAB,N)
!>          عند الدخول، المصفوفة A بتخزين نطاق، في الصفوف من 1 إلى KL+KU+1.
!>          يتم تخزين العمود j من A في العمود j من المصفوفة
!>          AB كالتالي:
!>          AB(KU+1+i-j,j) = A(i,j) لـ max(1,j-KU)<=i<=min(N,j+kl)
!>

LDAB

!>          LDAB هو عدد صحيح (INTEGER)
!>          البعد الرائد للمصفوفة A. LDAB >= max(1,M).
!>

R

!>          R هي مصفوفة ذات أرقام حقيقية مزدوجة الدقة (DOUBLE PRECISION)، بُعدها (M)
!>          إذا كانت INFO = 0 أو INFO > M، فإن R تحتوي على عوامل قياس الصفوف
!>          لـ A.
!>

C

!>          C هي مصفوفة ذات أرقام حقيقية مزدوجة الدقة (DOUBLE PRECISION)، بُعدها (N)
!>          إذا كانت INFO = 0، فإن C تحتوي على عوامل قياس الأعمدة لـ A.
!>

ROWCND

!>          ROWCND هو رقم حقيقي مزدوج الدقة (DOUBLE PRECISION)
!>          إذا كانت INFO = 0 أو INFO > M، فإن ROWCND يحتوي على نسبة
!>          أصغر R(i) إلى أكبر R(i). إذا كان ROWCND >= 0.1 وكان
!>          AMAX ليس كبيرًا جدًا أو صغيرًا جدًا، فلا يستحق الأمر
!>          القياس بواسطة R.
!>

COLCND

!>          COLCND هو رقم حقيقي مزدوج الدقة (DOUBLE PRECISION)
!>          إذا كانت INFO = 0، فإن COLCND يحتوي على نسبة أصغر
!>          C(i) إلى أكبر C(i). إذا كان COLCND >= 0.1، فلا يستحق الأمر
!>          القياس بواسطة C.
!>

AMAX

!>          AMAX هو رقم حقيقي مزدوج الدقة (DOUBLE PRECISION)
!>          القيمة المطلقة لأكبر عنصر مصفوفة. إذا كان AMAX قريبًا جدًا
!>          من الحمل الزائد أو قريبًا جدًا من النقصان، فيجب قياس المصفوفة.
!>

INFO

!>          INFO هو عدد صحيح (INTEGER)
!>          = 0:  الخروج بنجاح
!>          < 0:  إذا كانت INFO = -i، فقد كانت المعلمة i غير صالحة
!>          > 0:  إذا كانت INFO = i، و i هو
!>                <= M:  الصف i من A هو صفر تمامًا
!>                >  M:  العمود (i-M) من A هو صفر تمامًا
!>
