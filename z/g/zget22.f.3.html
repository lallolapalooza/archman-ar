توثيق الوظيفة/الروتين الفرعي

subroutine zget22 (character transa, character transe, character transw, integer n, complex*16, dimension( lda, * ) a, integer lda, complex*16, dimension( lde, * ) e, integer lde, complex*16, dimension( * ) w, complex*16, dimension( * ) work, double precision, dimension( * ) rwork, double precision, dimension( 2 ) result)

ZGET22

الغرض:

!>
!> ZGET22 يقوم بفحص المتجهات الذاتية.
!>
!> الاختبار الأساسي هو:
!>
!>    RESULT(1) = | A E  -  E W | / ( |A| |E| ulp )
!>
!> باستخدام المعيار 1. كما أنه يختبر تطبيع E:
!>
!>    RESULT(2) = max | m-norm(E(j)) - 1 | / ( n ulp )
!>                 j
!>
!> حيث E(j) هو المتجه الذاتي j، و m-norm هو المعيار الأقصى لمتجه.
!> المعيار الأقصى لمتجه معقد n هو الحد الأقصى لـ |re(x(i)| + |im(x(i)|
!> عبر i = 1، ..., n.
!>

المعلمات

TRANSA 
!>          TRANSA is CHARACTER*1
!>          يحدد ما إذا كان A قد تم نقله أم لا.
!>          = 'N':  لا يوجد نقل
!>          = 'T':  نقل
!>          = 'C':  نقل مقترن
!>

TRANSE

!>          TRANSE is CHARACTER*1
!>          يحدد ما إذا كان E قد تم نقله أم لا.
!>          = 'N':  لا يوجد نقل، المتجهات الذاتية في أعمدة E
!>          = 'T':  نقل، المتجهات الذاتية في صفوف E
!>          = 'C':  نقل مقترن، المتجهات الذاتية في صفوف E
!>

TRANS

!>          TRANS is CHARACTER*1
!>          يحدد ما إذا كان W قد تم نقله أم لا.
!>          = 'N':  لا يوجد نقل
!>          = 'T':  نقل، نفس TRANS = 'N'
!>          = 'C':  نقل مقترن، استخدم -WI(j) بدلاً من WI(j)
!>

N

!>          N is INTEGER
!>          رتبة المصفوفة A. N >= 0.
!>

A

!>          A is COMPLEX*16 array, dimension (LDA,N)
!>          المصفوفة التي تكون متجهاتها الذاتية في E.
!>

LDA

!>          LDA is INTEGER
!>          البعد الرائد للمصفوفة A. LDA >= max(1,N).
!>

E

!>          E is COMPLEX*16 array, dimension (LDE,N)
!>          مصفوفة المتجهات الذاتية. إذا كان TRANSE = 'N'، يتم تخزين
!>          المتجهات الذاتية في أعمدة E، إذا كان TRANSE = 'T' أو 'C'،
!>          يتم تخزين المتجهات الذاتية في صفوف E.
!>

LDE

!>          LDE is INTEGER
!>          البعد الرائد للمصفوفة E. LDE >= max(1,N).
!>

W

!>          W is COMPLEX*16 array, dimension (N)
!>          القيم الذاتية للمصفوفة A.
!>

WORK

!>          WORK is COMPLEX*16 array, dimension (N*N)
!>

RWORK

!>          RWORK is DOUBLE PRECISION array, dimension (N)
!>

RESULT

!>          RESULT is DOUBLE PRECISION array, dimension (2)
!>          RESULT(1) = | A E  -  E W | / ( |A| |E| ulp )
!>          RESULT(2) = max | m-norm(E(j)) - 1 | / ( n ulp )
!>                       j
!>
