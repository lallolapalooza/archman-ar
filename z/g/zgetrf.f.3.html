توثيق الوظيفة/الروتين الفرعي

subroutine zgetrf (integer m, integer n, complex*16, dimension( lda, * ) a, integer lda, integer, dimension( * ) ipiv, integer info)

ZGETRF

الغرض:

!>
!> ZGETRF تحسب تحليل LU لمصفوفة عامة M × N المسماة A
!> باستخدام التقسيم الجزئي مع تبديل الصفوف.
!>
!> التحليل يأخذ الشكل
!>    A = P * L * U
!> حيث P مصفوفة تبديل، L مصفوفة مثلثة سفلى بعناصر قطرية واحدة
!> (شبه منحرف سفلي إذا كان m > n)، و U مصفوفة عليا مثلثة
!> (شبه منحرف علوي إذا كان m < n).
!>
!> هذا هو الإصدار الصحيح من المستوى 3 BLAS للخوارزمية.
!>

المعاملات

M 
!>          M هو عدد صحيح (INTEGER)
!>          عدد صفوف المصفوفة A.  M >= 0.
!>

N

!>          N هو عدد صحيح (INTEGER)
!>          عدد أعمدة المصفوفة A.  N >= 0.
!>

A

!>          A هي مصفوفة مركبة (COMPLEX*16)، ذات البعد (LDA,N)
!>          عند الإدخال، المصفوفة M × N التي سيتم تحليلها.
!>          عند الخروج، العوامل L و U من التحليل
!>          A = P*L*U؛ لا يتم تخزين عناصر القطرية الواحدية لـ L.
!>

LDA

!>          LDA هو عدد صحيح (INTEGER)
!>          البعد الرئيسي للمصفوفة A.  LDA >= max(1,M).
!>

IPIV

!>          IPIV هي مصفوفة أعداد صحيحة (INTEGER)، ذات البعد (min(M,N))
!>          فهارس التقسيم؛ بالنسبة لـ 1 <= i <= min(M,N)، تم تبديل الصف i
!>          مع الصف IPIV(i).
!>

INFO

!>          INFO هو عدد صحيح (INTEGER)
!>          = 0:  انتهاء ناجح
!>          < 0:  إذا كان INFO = -i، كان المعامل i ذو قيمة غير قانونية
!>          > 0:  إذا كان INFO = i، فإن U(i,i) تساوي صفرًا تمامًا. تم إكمال التحليل،
!>                ولكن العامل U منفرد تمامًا، وسيحدث قسمة على صفر إذا تم استخدامه
!>                لحل نظام معادلات.
!>
