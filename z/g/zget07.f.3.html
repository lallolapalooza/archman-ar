توثيق الوظيفة/الوحدة الفرعية

subroutine zget07 (character trans, integer n, integer nrhs, complex*16, dimension( lda, * ) a, integer lda, complex*16, dimension( ldb, * ) b, integer ldb, complex*16, dimension( ldx, * ) x, integer ldx, complex*16, dimension( ldxact, * ) xact, integer ldxact, double precision, dimension( * ) ferr, logical chkferr, double precision, dimension( * ) berr, double precision, dimension( * ) reslts)

ZGET07

الغرض:

!>
!> ZGET07 تختبر حدود الخطأ من التحسين التكراري للحل المحسوب لنظام
!> المعادلات op(A)*X = B، حيث A هي مصفوفة n×n عامة و op(A) = A أو A**T،
!> اعتمادًا على TRANS.
!>
!> RESLTS(1) = اختبار حد الخطأ
!>           = norm(X - XACT) / ( norm(X) * FERR )
!>
!> يتم إرجاع قيمة كبيرة إذا لم تكن هذه النسبة أقل من واحد.
!>
!> RESLTS(2) = الخطأ المتبقي من روتين التحسين التكراري
!>           = الحد الأقصى لـ BERR / ( (n+1)*EPS + (*) )، حيث
!>             (*) = (n+1)*UNFL / (min_i (abs(op(A))*abs(X) +abs(b))_i )
!>

المعلمات

TRANS 
!>          TRANS هي CHARACTER*1
!>          تحدد شكل نظام المعادلات.
!>          = 'N':  A * X = B     (بدون تبديل)
!>          = 'T':  A**T * X = B  (تبديل)
!>          = 'C':  A**H * X = B  (تبديل مترافق = تبديل)
!>

N

!>          N هي INTEGER
!>          عدد صفوف المصفوفات X و XACT.  N >= 0.
!>

NRHS

!>          NRHS هي INTEGER
!>          عدد أعمدة المصفوفات X و XACT.  NRHS >= 0.
!>

A

!>          A هي مصفوفة COMPLEX*16، البعد (LDA,N)
!>          المصفوفة الأصلية A مقاس n×n.
!>

LDA

!>          LDA هي INTEGER
!>          البعد الرائد للمصفوفة A.  LDA >= max(1,N).
!>

B

!>          B هي مصفوفة COMPLEX*16، البعد (LDB,NRHS)
!>          متجهات الطرف الأيمن لنظام المعادلات الخطية.
!>

LDB

!>          LDB هي INTEGER
!>          البعد الرائد للمصفوفة B.  LDB >= max(1,N).
!>

X

!>          X هي مصفوفة COMPLEX*16، البعد (LDX,NRHS)
!>          متجهات الحل المحسوبة. يتم تخزين كل متجه كعمود في المصفوفة X.
!>

LDX

!>          LDX هي INTEGER
!>          البعد الرائد للمصفوفة X.  LDX >= max(1,N).
!>

XACT

!>          XACT هي مصفوفة COMPLEX*16، البعد (LDXACT,NRHS)
!>          متجهات الحل الدقيقة. يتم تخزين كل متجه كعمود في المصفوفة XACT.
!>

LDXACT

!>          LDXACT هي INTEGER
!>          البعد الرائد للمصفوفة XACT.  LDXACT >= max(1,N).
!>

FERR

!>          FERR هي مصفوفة DOUBLE PRECISION، البعد (NRHS)
!>          حدود الخطأ الأمامي المقدرة لكل متجه حل X. إذا كان XTRUE هو الحل
!>          الحقيقي، فإن FERR يحد من حجم أكبر مدخل في (X - XTRUE) مقسومًا
!>          على حجم أكبر مدخل في X.
!>

CHKFERR

!>          CHKFERR هي LOGICAL
!>          تم تعيينها على .TRUE. للتحقق من FERR، و .FALSE. لعدم التحقق من FERR.
!>          عندما يكون نظام الاختبار غير مشروط، قد يكون الحل في XACT غير صحيح.
!>

BERR

!>          BERR هي مصفوفة DOUBLE PRECISION، البعد (NRHS)
!>          الخطأ النسبي المكوني لكل متجه حل (أي، أصغر تغيير نسبي في أي مدخل
!>          في A أو B يجعل X حلاً دقيقًا).
!>

RESLTS

!>          RESLTS هي مصفوفة DOUBLE PRECISION، البعد (2)
!>          الحد الأقصى لمتجهات الحل NRHS للنسب التالية:
!>          RESLTS(1) = norm(X - XACT) / ( norm(X) * FERR )
!>          RESLTS(2) = BERR / ( (n+1)*EPS + (*) )
!>
