توثيق الوظيفة/الروتين الفرعي

subroutine zgeev (character jobvl, character jobvr, integer n, complex*16, dimension( lda, * ) a, integer lda, complex*16, dimension( * ) w, complex*16, dimension( ldvl, * ) vl, integer ldvl, complex*16, dimension( ldvr, * ) vr, integer ldvr, complex*16, dimension( * ) work, integer lwork, double precision, dimension( * ) rwork, integer info)

ZGEEV تحسب القيم الذاتية، اختياريًا، المتجهات الذاتية اليسرى و/أو اليمنى لمصفوفات GE

الغرض:

!>
!> ZGEEV تحسب لمصفوفة مركبة غير متناظرة N-by-N A،
!> القيم الذاتية، واختياريًا، المتجهات الذاتية اليسرى و/أو اليمنى.
!>
!> المتجه الذاتي الأيمن v(j) لـ A يحقق
!>                  A * v(j) = lambda(j) * v(j)
!> حيث lambda(j) هي قيمته الذاتية.
!> المتجه الذاتي الأيسر u(j) لـ A يحقق
!>               u(j)**H * A = lambda(j) * u(j)**H
!> حيث u(j)**H يدل على تبديل مرافق u(j).
!>
!> المتجهات الذاتية المحسوبة تم تطبيعها لتكون لها معيار إقليدي
!> يساوي 1 وأكبر مركبة حقيقية.
!>

المعلمات

JOBVL 
!>          JOBVL هو CHARACTER*1
!>          = 'N': لا يتم حساب المتجهات الذاتية اليسرى لـ A؛
!>          = 'V': يتم حساب المتجهات الذاتية اليسرى لـ A.
!>

JOBVR

!>          JOBVR هو CHARACTER*1
!>          = 'N': لا يتم حساب المتجهات الذاتية اليمنى لـ A؛
!>          = 'V': يتم حساب المتجهات الذاتية اليمنى لـ A.
!>

N

!>          N هو INTEGER
!>          رتبة المصفوفة A. N >= 0.
!>

A

!>          A هو مصفوفة COMPLEX*16، البعد (LDA,N)
!>          عند الإدخال، المصفوفة A بحجم N-by-N.
!>          عند الخروج، تم استبدال A.
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرئيسي للمصفوفة A. LDA >= max(1,N).
!>

W

!>          W هي مصفوفة COMPLEX*16، البعد (N)
!>          تحتوي W على القيم الذاتية المحسوبة.
!>

VL

!>          VL هي مصفوفة COMPLEX*16، البعد (LDVL,N)
!>          إذا كان JOBVL = 'V'، يتم تخزين المتجهات الذاتية اليسرى u(j)
!>          واحدة تلو الأخرى في أعمدة VL، بنفس ترتيب
!>          قيمها الذاتية.
!>          إذا كان JOBVL = 'N'، لا يتم الرجوع إلى VL.
!>          u(j) = VL(:,j)، العمود j من VL.
!>

LDVL

!>          LDVL هو INTEGER
!>          البعد الرئيسي للمصفوفة VL. LDVL >= 1؛ إذا
!>          كان JOBVL = 'V'، فإن LDVL >= N.
!>

VR

!>          VR هي مصفوفة COMPLEX*16، البعد (LDVR,N)
!>          إذا كان JOBVR = 'V'، يتم تخزين المتجهات الذاتية اليمنى v(j)
!>          واحدة تلو الأخرى في أعمدة VR، بنفس ترتيب
!>          قيمها الذاتية.
!>          إذا كان JOBVR = 'N'، لا يتم الرجوع إلى VR.
!>          v(j) = VR(:,j)، العمود j من VR.
!>

LDVR

!>          LDVR هو INTEGER
!>          البعد الرئيسي للمصفوفة VR. LDVR >= 1؛ إذا
!>          كان JOBVR = 'V'، فإن LDVR >= N.
!>

WORK

!>          WORK هي مصفوفة COMPLEX*16، البعد (MAX(1,LWORK))
!>          عند الخروج، إذا كان INFO = 0، فإن WORK(1) ترجع LWORK المثلى.
!>

LWORK

!>          LWORK هو INTEGER
!>          بعد المصفوفة WORK. LWORK >= max(1,2*N).
!>          لأداء جيد، يجب أن يكون LWORK بشكل عام أكبر.
!>
!>          إذا كان LWORK = -1، فسيتم افتراض استعلام مساحة العمل؛ تقوم الروتين
!>          فقط بحساب الحجم الأمثل لمصفوفة WORK، وإرجاع
!>          هذه القيمة كالإدخال الأول لمصفوفة WORK، ولا يتم إصدار أي رسالة خطأ
!>          متعلقة بـ LWORK بواسطة XERBLA.
!>

RWORK

!>          RWORK هي مصفوفة DOUBLE PRECISION، البعد (2*N)
!>

INFO

!>          INFO هو INTEGER
!>          = 0:  تم الخروج بنجاح
!>          < 0:  إذا كان INFO = -i، فإن الوسيط i كان له قيمة غير قانونية.
!>          > 0:  إذا كان INFO = i، فشلت خوارزمية QR في حساب جميع
!>                القيم الذاتية، ولم يتم حساب أي متجهات ذاتية؛
!>                العناصر i+1:N من W تحتوي على القيم الذاتية التي
!>                تقاربت.
!>
