توثيق الوظيفة/الإجراء الفرعي

subroutine zgesv (integer n, integer nrhs, complex*16, dimension( lda, * ) a, integer lda, integer, dimension( * ) ipiv, complex*16, dimension( ldb, * ) b, integer ldb, integer info)

الغرض:

!>
!> ZGESV تحسب حل نظام المعادلات الخطية المعقدة
!>    A * X = B,
!> حيث A هي مصفوفة N*N و X و B هي مصفوفات N*NRHS.
!>
!> يتم استخدام تحليل LU مع التنبيذ الجزئي وتبادلات الصفوف لتحليل A إلى
!>    A = P * L * U,
!> حيث P مصفوفة تبديل، L هي مصفوفة مثلثية سفلية وحيدة، و U هي مصفوفة مثلثية عليا.  ثم يتم استخدام الشكل المحلل لـ A لحل نظام المعادلات A * X = B.
!>

المعاملات

N 
!>          N هو عدد صحيح (INTEGER)
!>          عدد المعادلات الخطية، أي ترتيب المصفوفة A.  N >= 0.
!>

NRHS

!>          NRHS هو عدد صحيح (INTEGER)
!>          عدد الأطراف اليمنى، أي عدد أعمدة المصفوفة B.  NRHS >= 0.
!>

A

!>          A هي مصفوفة معقدة * 16 (COMPLEX*16) ، أبعادها (LDA,N)
!>          عند الإدخال، مصفوفة المعاملات A ذات الأبعاد N*N.
!>          عند الخروج، العوامل L و U من التحليل
!>          A = P*L*U؛ عناصر القطر الوحيدة لـ L غير مخزنة.
!>

LDA

!>          LDA هو عدد صحيح (INTEGER)
!>          البعد الرائد للمصفوفة A.  LDA >= max(1,N).
!>

IPIV

!>          IPIV هي مصفوفة أعداد صحيحة (INTEGER) ، أبعادها (N)
!>          مؤشرات التنبيذ التي تحدد مصفوفة التبديل P؛
!>          تم تبديل الصف i للمصفوفة بالصف IPIV(i).
!>

B

!>          B هي مصفوفة معقدة * 16 (COMPLEX*16) ، أبعادها (LDB,NRHS)
!>          عند الإدخال، مصفوفة الأطراف اليمنى B ذات الأبعاد N*NRHS.
!>          عند الخروج، إذا كانت INFO = 0، مصفوفة الحل X ذات الأبعاد N*NRHS.
!>

LDB

!>          LDB هو عدد صحيح (INTEGER)
!>          البعد الرائد للمصفوفة B.  LDB >= max(1,N).
!>

INFO

!>          INFO هو عدد صحيح (INTEGER)
!>          = 0:  انتهاء ناجح
!>          < 0:  إذا كانت INFO = -i، فإن المعامل i كان له قيمة غير قانونية
!>          > 0:  إذا كانت INFO = i، فإن U(i,i) يساوي صفرًا تمامًا.  تم الانتهاء من التحليل،
!>                ولكن العامل U كان مفردًا تمامًا، لذلك لم يكن من الممكن حساب الحل.
!>
