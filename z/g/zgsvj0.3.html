subroutine zgsvj0 (character*1 jobv, integer m, integer n, complex*16, dimension( lda, * ) a, integer lda, complex*16, dimension( n ) d, double precision, dimension( n ) sva, integer mv, complex*16, dimension( ldv, * ) v, integer ldv, double precision eps, double precision sfmin, double precision tol, integer nsweep, complex*16, dimension( lwork ) work, integer lwork, integer info)

ZGSVJ0 معالج أولي للإجراء الفرعي zgesvj.

الغرض:

!>
!> ZGSVJ0 يتم استدعاؤه من ZGESVJ كمعالج أولي وهذا هو غرضه الرئيسي.
!> إنه يطبق دورانات جاكوبي بنفس الطريقة التي يفعلها ZGESVJ، ولكنه لا
!> يتحقق من التقارب (معيار التوقف). تتوفر بعض معلمات الضبط (المميزة بـ [TP])
!> للمنفذ.
!>

المعلمات

JOBV 
!>          JOBV هو CHARACTER*1
!>          يحدد ما إذا كان الإخراج من هذا الإجراء سيتم استخدامه
!>          لحساب المصفوفة V:
!>          = 'V': يتم تجميع حاصل ضرب دورانات جاكوبي
!>                 عن طريق الضرب اللاحق للمصفوفة N-by-N V.
!>                (انظر وصف V.)
!>          = 'A': يتم تجميع حاصل ضرب دورانات جاكوبي
!>                 عن طريق الضرب اللاحق للمصفوفة MV-by-N V.
!>                (انظر أوصاف MV و V.)
!>          = 'N': لا يتم تجميع دورانات جاكوبي.
!>

M

!>          M هو INTEGER
!>          عدد صفوف المصفوفة المدخلة A.  M >= 0.
!>

N

!>          N هو INTEGER
!>          عدد أعمدة المصفوفة المدخلة A.
!>          M >= N >= 0.
!>

A

!>          A هي مصفوفة COMPLEX*16، أبعادها (LDA,N)
!>          عند الدخول، المصفوفة M-by-N A، بحيث A*diag(D) تمثل
!>          المصفوفة المدخلة.
!>          عند الخروج،
!>          A_onexit * diag(D_onexit) تمثل المصفوفة المدخلة A*diag(D)
!>          بعد ضربها بسلسلة من دورانات جاكوبي، حيث
!>          عتبة الدوران وإجمالي عدد الدورات معطاة في
!>          TOL و NSWEEP، على التوالي.
!>          (انظر أوصاف D، TOL و NSWEEP.)
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرئيسي للمصفوفة A.  LDA >= max(1,M).
!>

D

!>          D هي مصفوفة COMPLEX*16، أبعادها (N)
!>          تجمع المصفوفة D عوامل القياس من دورانات جاكوبي المعقدة المقيّسة.
!>          عند الدخول، A*diag(D) تمثل المصفوفة المدخلة.
!>          عند الخروج، A_onexit*diag(D_onexit) تمثل المصفوفة المدخلة
!>          بعد ضربها بسلسلة من دورانات جاكوبي، حيث
!>          عتبة الدوران وإجمالي عدد الدورات معطاة في
!>          TOL و NSWEEP، على التوالي.
!>          (انظر أوصاف A، TOL و NSWEEP.)
!>

SVA

!>          SVA هي مصفوفة DOUBLE PRECISION، أبعادها (N)
!>          عند الدخول، تحتوي SVA على المعايير الإقليدية لأعمدة
!>          المصفوفة A*diag(D).
!>          عند الخروج، تحتوي SVA على المعايير الإقليدية لأعمدة
!>          المصفوفة A_onexit*diag(D_onexit).
!>

MV

!>          MV هو INTEGER
!>          إذا كان JOBV = 'A'، فسيتم ضرب MV صفًا من V
!>                           بشكل لاحق بسلسلة من دورانات جاكوبي.
!>          إذا كان JOBV = 'N'،   فلا يتم الرجوع إلى MV.
!>

V

!>          V هي مصفوفة COMPLEX*16، أبعادها (LDV,N)
!>          إذا كان JOBV = 'V' فسيتم ضرب N صفًا من V
!>                           بشكل لاحق بسلسلة من دورانات جاكوبي.
!>          إذا كان JOBV = 'A' فسيتم ضرب MV صفًا من V
!>                           بشكل لاحق بسلسلة من دورانات جاكوبي.
!>          إذا كان JOBV = 'N'،   فلا يتم الرجوع إلى V.
!>

LDV

!>          LDV هو INTEGER
!>          البعد الرئيسي للمصفوفة V،  LDV >= 1.
!>          إذا كان JOBV = 'V'، LDV >= N.
!>          إذا كان JOBV = 'A'، LDV >= MV.
!>

EPS

!>          EPS هو DOUBLE PRECISION
!>          EPS = DLAMCH('Epsilon')
!>

SFMIN

!>          SFMIN هو DOUBLE PRECISION
!>          SFMIN = DLAMCH('Safe Minimum')
!>

TOL

!>          TOL هو DOUBLE PRECISION
!>          TOL هي العتبة لدورانات جاكوبي. لزوج
!>          A(:,p) و A(:,q) من الأعمدة المحورية، يتم تطبيق دوران جاكوبي
!>          فقط إذا كان ABS(COS(angle(A(:,p),A(:,q)))) > TOL.
!>

NSWEEP

!>          NSWEEP هو INTEGER
!>          NSWEEP هو عدد الدورات من دورانات جاكوبي التي يجب
!>          تنفيذها.
!>

WORK

!>          WORK هي مصفوفة COMPLEX*16، أبعادها (LWORK)
!>

LWORK

!>          LWORK هو INTEGER
!>          LWORK هو بعد WORK. LWORK >= M.
!>

INFO

!>          INFO هو INTEGER
!>          = 0:  خروج ناجح.
!>          < 0:  إذا كان INFO = -i، فإن الوسيط i كان له قيمة غير قانونية.
!>
