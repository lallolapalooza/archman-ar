subroutine zgeev (character jobvl, character jobvr, integer n, complex*16, dimension( lda, * ) a, integer lda, complex*16, dimension( * ) w, complex*16, dimension( ldvl, * ) vl, integer ldvl, complex*16, dimension( ldvr, * ) vr, integer ldvr, complex*16, dimension( * ) work, integer lwork, double precision, dimension( * ) rwork, integer info)

ZGEEV تحسب القيم الذاتية، و اختيارياً، المتجهات الذاتية اليسرى و/أو اليمنى لمصفوفات GE

الغرض:

!>
!> ZGEEV تحسب لمصفوفة معقدة غير متناظرة N-by-N A،
!> القيم الذاتية، واختيارياً، المتجهات الذاتية اليسرى و/أو اليمنى.
!>
!> المتجه الذاتي الأيمن v(j) للمصفوفة A يحقق
!>                  A * v(j) = lambda(j) * v(j)
!> حيث lambda(j) هي قيمتها الذاتية.
!> المتجه الذاتي الأيسر u(j) للمصفوفة A يحقق
!>               u(j)**H * A = lambda(j) * u(j)**H
!> حيث u(j)**H يمثل منقول الصفوف المترافقة لـ u(j).
!>
!> المتجهات الذاتية المحسوبة يتم تسويتها ليكون لها معيار إقليدي
!> يساوي 1 وأكبر مركبة حقيقية.
!>

المعلمات

JOBVL 
!>          JOBVL هو CHARACTER*1
!>          = 'N': المتجهات الذاتية اليسرى لـ A لا يتم حسابها؛
!>          = 'V': المتجهات الذاتية اليسرى لـ A يتم حسابها.
!>

JOBVR

!>          JOBVR هو CHARACTER*1
!>          = 'N': المتجهات الذاتية اليمنى لـ A لا يتم حسابها؛
!>          = 'V': المتجهات الذاتية اليمنى لـ A يتم حسابها.
!>

N

!>          N هو INTEGER
!>          رتبة المصفوفة A. N >= 0.
!>

A

!>          A هي مصفوفة COMPLEX*16، أبعادها (LDA,N)
!>          عند الدخول، مصفوفة A ذات الأبعاد N-by-N.
!>          عند الخروج، تم استبدال A.
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرائد للمصفوفة A. LDA >= max(1,N).
!>

W

!>          W هي مصفوفة COMPLEX*16، أبعادها (N)
!>          تحتوي W على القيم الذاتية المحسوبة.
!>

VL

!>          VL هي مصفوفة COMPLEX*16، أبعادها (LDVL,N)
!>          إذا كان JOBVL = 'V'، يتم تخزين المتجهات الذاتية اليسرى u(j)
!>          واحدة تلو الأخرى في أعمدة VL، بنفس ترتيب
!>          قيمها الذاتية.
!>          إذا كان JOBVL = 'N'، لا يتم استخدام VL.
!>          u(j) = VL(:,j)، العمود j من VL.
!>

LDVL

!>          LDVL هو INTEGER
!>          البعد الرائد للمصفوفة VL. LDVL >= 1؛ إذا كان
!>          JOBVL = 'V'، فإن LDVL >= N.
!>

VR

!>          VR هي مصفوفة COMPLEX*16، أبعادها (LDVR,N)
!>          إذا كان JOBVR = 'V'، يتم تخزين المتجهات الذاتية اليمنى v(j)
!>          واحدة تلو الأخرى في أعمدة VR، بنفس ترتيب
!>          قيمها الذاتية.
!>          إذا كان JOBVR = 'N'، لا يتم استخدام VR.
!>          v(j) = VR(:,j)، العمود j من VR.
!>

LDVR

!>          LDVR هو INTEGER
!>          البعد الرائد للمصفوفة VR. LDVR >= 1؛ إذا كان
!>          JOBVR = 'V'، فإن LDVR >= N.
!>

WORK

!>          WORK هي مصفوفة COMPLEX*16، أبعادها (MAX(1,LWORK))
!>          عند الخروج، إذا كان INFO = 0، تعيد WORK(1) قيمة LWORK المثلى.
!>

LWORK

!>          LWORK هو INTEGER
!>          بعد المصفوفة WORK. LWORK >= max(1,2*N).
!>          لأداء جيد، يجب أن يكون LWORK أكبر بشكل عام.
!>
!>          إذا كان LWORK = -1، يفترض الاستعلام عن مساحة العمل؛ تقوم الروتين
!>          بحساب حجم مساحة العمل المثلى فقط، وإعادة
!>          هذه القيمة كالإدخال الأول في مساحة العمل، ولا تصدر رسالة خطأ
!>          متعلقة بـ LWORK بواسطة XERBLA.
!>

RWORK

!>          RWORK هي مصفوفة DOUBLE PRECISION، أبعادها (2*N)
!>

INFO

!>          INFO هو INTEGER
!>          = 0:  خروج ناجح
!>          < 0:  إذا كان INFO = -i، فإن المعلمة i كانت لها قيمة غير قانونية.
!>          > 0:  إذا كان INFO = i، فشلت خوارزمية QR في حساب جميع
!>                القيم الذاتية، ولم يتم حساب أي متجهات ذاتية؛
!>                العناصر i+1:N من W تحتوي على القيم الذاتية التي
!>                تقاربت.
!>
