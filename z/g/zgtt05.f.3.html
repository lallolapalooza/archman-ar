subroutine zgtt05 (character trans, integer n, integer nrhs, complex*16, dimension( * ) dl, complex*16, dimension( * ) d, complex*16, dimension( * ) du, complex*16, dimension( ldb, * ) b, integer ldb, complex*16, dimension( ldx, * ) x, integer ldx, complex*16, dimension( ldxact, * ) xact, integer ldxact, double precision, dimension( * ) ferr, double precision, dimension( * ) berr, double precision, dimension( * ) reslts)

ZGTT05

الغرض:

!>
!> ZGTT05 تختبر حدود الخطأ للتنقيح التكراري لحل النظام
!> A*X = B، حيث A مصفوفة ثلاثية قطرية عامة من الرتبة n و op(A) = A أو A**T،
!> اعتمادًا على TRANS.
!>
!> RESLTS(1) = اختبار حد الخطأ
!>           = norm(X - XACT) / ( norm(X) * FERR )
!>
!> تُرجع قيمة كبيرة إذا لم تكن هذه النسبة أقل من واحد.
!>
!> RESLTS(2) = الباقي من روتين التنقيح التكراري
!>           = الحد الأقصى لـ BERR / ( NZ*EPS + (*) )، حيث
!>             (*) = NZ*UNFL / (min_i (abs(op(A))*abs(X) +abs(b))_i )
!>             و NZ = أقصى عدد من العناصر غير الصفرية في أي صف من A، بالإضافة إلى 1
!>

المعاملات

TRANS 
!>          TRANS هو CHARACTER*1
!>          يحدد شكل نظام المعادلات.
!>          = 'N':  A * X = B     (بدون تبديل)
!>          = 'T':  A**T * X = B  (تبديل)
!>          = 'C':  A**H * X = B  (تبديل مترافق = تبديل)
!>

N

!>          N هو INTEGER
!>          عدد صفوف المصفوفات X و XACT.  N >= 0.
!>

NRHS

!>          NRHS هو INTEGER
!>          عدد أعمدة المصفوفات X و XACT.  NRHS >= 0.
!>

DL

!>          DL هو COMPLEX*16 array, dimension (N-1)
!>          العناصر الفرعية القطرية (n-1) للمصفوفة A.
!>

D

!>          D هو COMPLEX*16 array, dimension (N)
!>          العناصر القطرية للمصفوفة A.
!>

DU

!>          DU هو COMPLEX*16 array, dimension (N-1)
!>          العناصر فوق القطرية (n-1) للمصفوفة A.
!>

B

!>          B هو COMPLEX*16 array, dimension (LDB,NRHS)
!>          متجهات الطرف الأيمن لنظام المعادلات الخطية.
!>

LDB

!>          LDB هو INTEGER
!>          البعد الرائد للمصفوفة B.  LDB >= max(1,N).
!>

X

!>          X هو COMPLEX*16 array, dimension (LDX,NRHS)
!>          متجهات الحل المحسوبة.  كل متجه مخزن كعمود في المصفوفة X.
!>

LDX

!>          LDX هو INTEGER
!>          البعد الرائد للمصفوفة X.  LDX >= max(1,N).
!>

XACT

!>          XACT هو COMPLEX*16 array, dimension (LDX,NRHS)
!>          متجهات الحل الدقيقة.  كل متجه مخزن كعمود في المصفوفة XACT.
!>

LDXACT

!>          LDXACT هو INTEGER
!>          البعد الرائد للمصفوفة XACT.  LDXACT >= max(1,N).
!>

FERR

!>          FERR هو DOUBLE PRECISION array, dimension (NRHS)
!>          حدود الخطأ الأمامي المقدرة لكل متجه حل X.
!>          إذا كان XTRUE هو الحل الحقيقي، فإن FERR يحدد حجم أكبر إدخال في
!>          (X - XTRUE) مقسومًا على حجم أكبر إدخال في X.
!>

BERR

!>          BERR هو DOUBLE PRECISION array, dimension (NRHS)
!>          الخطأ النسبي المكون لكل متجه حل (أي، أصغر تغيير نسبي في أي إدخال
!>          لـ A أو B يجعل X حلاً دقيقًا).
!>

RESLTS

!>          RESLTS هو DOUBLE PRECISION array, dimension (2)
!>          الحد الأقصى عبر متجهات الحل NRHS للنسب التالية:
!>          RESLTS(1) = norm(X - XACT) / ( norm(X) * FERR )
!>          RESLTS(2) = BERR / ( NZ*EPS + (*) )
!>
