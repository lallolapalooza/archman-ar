subroutine zget37 (double precision, dimension( 3 ) rmax, integer, dimension( 3 ) lmax, integer, dimension( 3 ) ninfo, integer knt, integer nin)

ZGET37

الغرض:

!>
!> ZGET37 تختبر ZTRSNA، وهي روتين لتقدير أرقام شروط القيم الذاتية و/أو المتجهات الذاتية اليمنى للمصفوفة.
!>
!> يتم قراءة مصفوفات الاختبار من ملف برقم وحدة منطقي NIN.
!>

المعلمات

RMAX 
!>          RMAX هي مصفوفة DOUBLE PRECISION، أبعادها (3)
!>          قيمة أكبر نسبة اختبار.
!>          RMAX(1) = أكبر نسبة تقارن بين استدعاءات مختلفة لـ ZTRSNA
!>          RMAX(2) = أكبر خطأ في الأرقام المعكوسة للشروط
!>                    مع مراعاة شروطها
!>          RMAX(3) = أكبر خطأ في الأرقام المعكوسة للشروط
!>                    دون مراعاة شروطها (قد تكون أكبر من RMAX(2))
!>

LMAX

!>          LMAX هي مصفوفة INTEGER، أبعادها (3)
!>          LMAX(i) هو رقم المثال حيث يتحقق أكبر نسبة اختبار
!>          RMAX(i). أيضًا:
!>          إذا أعادت ZGEHRD قيمة INFO غير صفرية في المثال i، فإن LMAX(1)=i
!>          إذا أعادت ZHSEQR قيمة INFO غير صفرية في المثال i، فإن LMAX(2)=i
!>          إذا أعادت ZTRSNA قيمة INFO غير صفرية في المثال i، فإن LMAX(3)=i
!>

NINFO

!>          NINFO هي مصفوفة INTEGER، أبعادها (3)
!>          NINFO(1) = عدد المرات التي أعادت فيها ZGEHRD قيمة INFO غير صفرية
!>          NINFO(2) = عدد المرات التي أعادت فيها ZHSEQR قيمة INFO غير صفرية
!>          NINFO(3) = عدد المرات التي أعادت فيها ZTRSNA قيمة INFO غير صفرية
!>

KNT

!>          KNT هو INTEGER
!>          العدد الإجمالي للأمثلة التي تم اختبارها.
!>

NIN

!>          NIN هو INTEGER
!>          رقم الوحدة المنطقية للإدخال
!>
