subroutine zgemqrt (character side, character trans, integer m, integer n, integer k, integer nb, complex*16, dimension( ldv, * ) v, integer ldv, complex*16, dimension( ldt, * ) t, integer ldt, complex*16, dimension( ldc, * ) c, integer ldc, complex*16, dimension( * ) work, integer info)

ZGEMQRT

الغرض:

!>
!> ZGEMQRT يستبدل مصفوفة C العامة ذات الحجم M-by-N بالمصفوفة
!>
!>                 SIDE = 'L'     SIDE = 'R'
!> TRANS = 'N':      Q C            C Q
!> TRANS = 'C':    Q**H C            C Q**H
!>
!> حيث Q هي مصفوفة متعامدة مركبة تُعرّف كناتج لـ K
!> عواكس أولية:
!>
!>       Q = H(1) H(2) . . . H(K) = I - V T V**H
!>
!> تم إنشاؤها باستخدام تمثيل WY المدمج كما تم إرجاعه بواسطة ZGEQRT.
!>
!> Q هي من الرتبة M إذا كانت SIDE = 'L' ومن الرتبة N إذا كانت SIDE = 'R'.
!>

المعلمات

SIDE 
!>          SIDE هي CHARACTER*1
!>          = 'L': تطبيق Q أو Q**H من اليسار؛
!>          = 'R': تطبيق Q أو Q**H من اليمين.
!>

TRANS

!>          TRANS هي CHARACTER*1
!>          = 'N':  لا يوجد تبديل، تطبيق Q؛
!>          = 'C':  تبديل مرافق، تطبيق Q**H.
!>

M

!>          M هو INTEGER
!>          عدد صفوف المصفوفة C. M >= 0.
!>

N

!>          N هو INTEGER
!>          عدد أعمدة المصفوفة C. N >= 0.
!>

K

!>          K هو INTEGER
!>          عدد العواكس الأولية التي يحدد ناتجها
!>          المصفوفة Q.
!>          إذا كانت SIDE = 'L'، فإن M >= K >= 0؛
!>          إذا كانت SIDE = 'R'، فإن N >= K >= 0.
!>

NB

!>          NB هو INTEGER
!>          حجم الكتلة المستخدم لتخزين T. K >= NB >= 1.
!>          يجب أن تكون هذه هي نفس قيمة NB المستخدمة لإنشاء T
!>          في ZGEQRT.
!>

V

!>          V هي مصفوفة COMPLEX*16، أبعادها (LDV,K)
!>          العمود i يجب أن يحتوي على المتجه الذي يحدد
!>          العاكس الأولي H(i)، لـ i = 1,2,...,k، كما تم إرجاعه بواسطة
!>          ZGEQRT في الأعمدة K الأولى من وسيطته A.
!>

LDV

!>          LDV هو INTEGER
!>          البعد الرائد للمصفوفة V.
!>          إذا كانت SIDE = 'L'، فإن LDA >= max(1,M)؛
!>          إذا كانت SIDE = 'R'، فإن LDA >= max(1,N).
!>

T

!>          T هي مصفوفة COMPLEX*16، أبعادها (LDT,K)
!>          عوامل المثلث العلوي للعواكس الكتلة
!>          كما تم إرجاعها بواسطة ZGEQRT، مخزنة كمصفوفة NB-by-N.
!>

LDT

!>          LDT هو INTEGER
!>          البعد الرائد للمصفوفة T. LDT >= NB.
!>

C

!>          C هي مصفوفة COMPLEX*16، أبعادها (LDC,N)
!>          عند الإدخال، مصفوفة C بحجم M-by-N.
!>          عند الخروج، يتم استبدال C بـ Q C، Q**H C، C Q**H أو C Q.
!>

LDC

!>          LDC هو INTEGER
!>          البعد الرائد للمصفوفة C. LDC >= max(1,M).
!>

WORK

!>          WORK هي مصفوفة COMPLEX*16. أبعاد WORK هي
!>           N*NB إذا كانت SIDE = 'L'، أو M*NB إذا كانت SIDE = 'R'.
!>

INFO

!>          INFO هو INTEGER
!>          = 0:  انتهاء ناجح
!>          < 0:  إذا كان INFO = -i، فإن الوسيط i كان له قيمة غير قانونية
!>
