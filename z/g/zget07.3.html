توثيق الدالة/الروتين الفرعي

subroutine zget07 (character trans, integer n, integer nrhs, complex*16, dimension( lda, * ) a, integer lda, complex*16, dimension( ldb, * ) b, integer ldb, complex*16, dimension( ldx, * ) x, integer ldx, complex*16, dimension( ldxact, * ) xact, integer ldxact, double precision, dimension( * ) ferr, logical chkferr, double precision, dimension( * ) berr, double precision, dimension( * ) reslts)

ZGET07

الغرض:

!>
!> ZGET07 تختبر حدود الخطأ من التنقيح التكراري للحل المحسوب لنظام المعادلات
!> op(A)*X = B، حيث A هي مصفوفة عامة n × n و op(A) = A أو A**T، اعتمادًا على TRANS.
!>
!> RESLTS(1) = اختبار حد الخطأ
!>           = norm(X - XACT) / ( norm(X) * FERR )
!>
!> يتم إرجاع قيمة كبيرة إذا لم تكن هذه النسبة أقل من واحد.
!>
!> RESLTS(2) = الخطأ المتبقي من روتين التنقيح التكراري
!>           = الحد الأقصى لـ BERR / ( (n+1)*EPS + (*) )، حيث
!>             (*) = (n+1)*UNFL / (min_i (abs(op(A))*abs(X) +abs(b))_i )
!>

المعلمات

TRANS 
!>          TRANS is CHARACTER*1
!>          يحدد شكل نظام المعادلات.
!>          = 'N':  A * X = B     (بدون تدوير)
!>          = 'T':  A**T * X = B  (تدوير)
!>          = 'C':  A**H * X = B  (تدوير مترافق = تدوير)
!>

N

!>          N is INTEGER
!>          عدد صفوف المصفوفات X و XACT. N >= 0.
!>

NRHS

!>          NRHS is INTEGER
!>          عدد أعمدة المصفوفات X و XACT. NRHS >= 0.
!>

A

!>          A is COMPLEX*16 array, dimension (LDA,N)
!>          المصفوفة الأصلية A مقاس n × n.
!>

LDA

!>          LDA is INTEGER
!>          البعد الرائد للمصفوفة A. LDA >= max(1,N).
!>

B

!>          B is COMPLEX*16 array, dimension (LDB,NRHS)
!>          متجهات الطرف الأيمن لأنظمة المعادلات الخطية.
!>

LDB

!>          LDB is INTEGER
!>          البعد الرائد للمصفوفة B. LDB >= max(1,N).
!>

X

!>          X is COMPLEX*16 array, dimension (LDX,NRHS)
!>          متجهات الحل المحسوبة. يتم تخزين كل متجه كعمود في المصفوفة X.
!>

LDX

!>          LDX is INTEGER
!>          البعد الرائد للمصفوفة X. LDX >= max(1,N).
!>

XACT

!>          XACT is COMPLEX*16 array, dimension (LDX,NRHS)
!>          متجهات الحل الدقيقة. يتم تخزين كل متجه كعمود في المصفوفة XACT.
!>

LDXACT

!>          LDXACT is INTEGER
!>          البعد الرائد للمصفوفة XACT. LDXACT >= max(1,N).
!>

FERR

!>          FERR is DOUBLE PRECISION array, dimension (NRHS)
!>          حدود الخطأ الأمامي المقدرة لكل متجه حل X. إذا كان XTRUE هو الحل
!>          الحقيقي، فإن FERR يحد من حجم أكبر مدخل في (X - XTRUE) مقسومًا على
!>          حجم أكبر مدخل في X.
!>

CHKFERR

!>          CHKFERR is LOGICAL
!>          تم التعيين إلى .TRUE. للتحقق من FERR، و .FALSE. لعدم التحقق من FERR.
!>          عندما يكون نظام الاختبار سيء التكييف، قد يكون الحل في XACT غير صحيح.
!>

BERR

!>          BERR is DOUBLE PRECISION array, dimension (NRHS)
!>          الخطأ العكسي النسبي المكون من عنصر لكل متجه حل
!>          (أي، أصغر تغيير نسبي في أي مدخل في A أو B يجعل X حلاً دقيقًا).
!>

RESLTS

!>          RESLTS is DOUBLE PRECISION array, dimension (2)
!>          الحد الأقصى لمتجهات الحل NRHS للنسب التالية:
!>          RESLTS(1) = norm(X - XACT) / ( norm(X) * FERR )
!>          RESLTS(2) = BERR / ( (n+1)*EPS + (*) )
!>
