توثيق الوظيفة/الروتين الفرعي

subroutine zgebal (character job, integer n, complex*16, dimension( lda, * ) a, integer lda, integer ilo, integer ihi, double precision, dimension( * ) scale, integer info)

ZGEBAL

الغرض:

!>
!> ZGEBAL توازن مصفوفة عامة معقدة A. وهذا يشمل، أولاً،
!> تبديل A بتحويل تشابه لعزل القيم الذاتية في العناصر من 1 إلى ILO-1 الأولى والأخيرة IHI+1 إلى N على
!> القطر؛ وثانياً، تطبيق تحويل تشابه قطري
!> للصفوف والأعمدة ILO إلى IHI لجعل الصفوف والأعمدة قريبة من المعيار قدر الإمكان. كلا الخطوتين اختياريتان.
!>
!> قد يقلل التوازن من معيار 1 للمصفوفة، ويحسن
!> دقة القيم الذاتية و/أو المتجهات الذاتية المحسوبة.
!>

المعلمات

JOB 
!>          JOB هو CHARACTER*1
!>          يحدد العمليات التي سيتم إجراؤها على A:
!>          = 'N':  لا شيء:  ببساطة قم بتعيين ILO = 1، IHI = N، SCALE(I) = 1.0
!>                  لـ i = 1,...,N؛
!>          = 'P':  تبديل فقط؛
!>          = 'S':  تحجيم فقط؛
!>          = 'B':  تبديل وتحجيم معاً.
!>

N

!>          N هو INTEGER
!>          ترتيب المصفوفة A.  N >= 0.
!>

A

!>          A هو مصفوفة COMPLEX*16، حجمها (LDA,N)
!>          عند الدخول، المصفوفة المدخلة A.
!>          عند الخروج، يتم الكتابة فوق A بالمصفوفة المتوازنة.
!>          إذا كان JOB = 'N'، فلن يتم الرجوع إلى A.
!>          انظر التفاصيل الإضافية.
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرائد للمصفوفة A.  LDA >= max(1,N).
!>

ILO

!>          ILO هو INTEGER
!>

IHI

!>          IHI هو INTEGER
!>          يتم تعيين ILO و IHI إلى عددين صحيحين بحيث عند الخروج
!>          A(i,j) = 0 إذا كان i > j و j = 1,...,ILO-1 أو I = IHI+1,...,N.
!>          إذا كان JOB = 'N' أو 'S'، فإن ILO = 1 و IHI = N.
!>

SCALE

!>          SCALE هي مصفوفة DOUBLE PRECISION، حجمها (N)
!>          تفاصيل التبديلات وعوامل التحجيم المطبقة على
!>          A. إذا كان P(j) هو فهرس الصف والعمود المتبادل
!>          مع الصف والعمود j وكان D(j) هو عامل التحجيم
!>          المطبق على الصف والعمود j، فإن
!>          SCALE(j) = P(j)    لـ j = 1,...,ILO-1
!>                   = D(j)    لـ j = ILO,...,IHI
!>                   = P(j)    لـ j = IHI+1,...,N.
!>          الترتيب الذي تتم به المبادلات هو N إلى IHI+1،
!>          ثم 1 إلى ILO-1.
!>

INFO

!>          INFO هو INTEGER
!>          = 0:  الخروج بنجاح.
!>          < 0:  إذا كان INFO = -i، فإن الوسيط i كان له قيمة غير قانونية.
!>

تفاصيل إضافية:

!>
!>  تتكون التبديلات من تبادلات الصفوف والأعمدة التي تضع
!>  المصفوفة في الشكل
!>
!>             ( T1   X   Y  )
!>     P A P = (  0   B   Z  )
!>             (  0   0   T2 )
!>
!>  حيث T1 و T2 هي مصفوفات مثلثية عليا تقع قيمها الذاتية
!>  على القطر. تحدد فهارس الأعمدة ILO و IHI بداية
!>  ونهاية المصفوفة الفرعية B. يتكون التوازن من تطبيق
!>  تحويل تشابه قطري inv(D) * B * D لجعل
!>  معايير 1 لكل صف من B وعموده المقابل متساوية تقريباً.
!>  المصفوفة الناتجة هي
!>
!>     ( T1     X*D          Y    )
!>     (  0  inv(D)*B*D  inv(D)*Z ).
!>     (  0      0           T2   )
!>
!>  تتم إرجاع معلومات حول التبديلات P والمصفوفة القطرية D
!>  في المتجه SCALE.
!>
!>  هذا الروتين الفرعي يعتمد على روتين EISPACK CBAL.
!>
!>  تم تعديله بواسطة Tzu-Yi Chen، قسم علوم الكمبيوتر، جامعة
!>    كاليفورنيا في بيركلي، الولايات المتحدة الأمريكية
!>
!>  تمت إعادة هيكلته بواسطة Evert Provoost، قسم علوم الكمبيوتر،
!>    KU Leuven، بلجيكا
!>
