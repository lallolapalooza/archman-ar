subroutine zgbequ (integer m, integer n, integer kl, integer ku, complex*16, dimension( ldab, * ) ab, integer ldab, double precision, dimension( * ) r, double precision, dimension( * ) c, double precision rowcnd, double precision colcnd, double precision amax, integer info)

ZGBEQU

الغرض:

!>
!> ZGBEQU يحسب مقاييس الصفوف والأعمدة المعدة لمعايرة مصفوفة شريطية M-by-N A
!> وتقليل رقم حالتها.  R يعيد عوامل مقياس الصفوف و C عوامل الأعمدة، المختارة لمحاولة
!> جعل أكبر عنصر في كل صف وعمود من المصفوفة B بعناصر B(i,j)=R(i)*A(i,j)*C(j)
!> لديها قيمة مطلقة 1.
!>
!> R(i) و C(j) مقيدان بأن يكونا بين SMLNUM = أصغر رقم آمن
!> و BIGNUM = أكبر رقم آمن.  استخدام هذه العوامل القياسية ليس مضمونًا
!> لتقليل رقم حالة A ولكنه يعمل بشكل جيد في الممارسة العملية.
!>

المعاملات

M 
!>          M هو عدد صحيح (INTEGER)
!>          عدد صفوف المصفوفة A.  M >= 0.
!>

N

!>          N هو عدد صحيح (INTEGER)
!>          عدد أعمدة المصفوفة A.  N >= 0.
!>

KL

!>          KL هو عدد صحيح (INTEGER)
!>          عدد الأقطار الفرعية داخل نطاق A.  KL >= 0.
!>

KU

!>          KU هو عدد صحيح (INTEGER)
!>          عدد الأقطار الفائقة داخل نطاق A.  KU >= 0.
!>

AB

!>          AB هي مصفوفة COMPLEX*16، الأبعاد (LDAB,N)
!>          مصفوفة الشريط A، مخزنة في الصفوف 1 إلى KL+KU+1.  العمود j
!>          من A مخزن في العمود j من المصفوفة AB كالتالي:
!>          AB(ku+1+i-j,j) = A(i,j) لـ max(1,j-ku)<=i<=min(m,j+kl).
!>

LDAB

!>          LDAB هو عدد صحيح (INTEGER)
!>          البعد الرائد للمصفوفة AB.  LDAB >= KL+KU+1.
!>

R

!>          R هي مصفوفة DOUBLE PRECISION، الأبعاد (M)
!>          إذا كان INFO = 0، أو INFO > M، تحتوي R على عوامل مقياس الصفوف
!>          لـ A.
!>

C

!>          C هي مصفوفة DOUBLE PRECISION، الأبعاد (N)
!>          إذا كان INFO = 0، تحتوي C على عوامل مقياس الأعمدة لـ A.
!>

ROWCND

!>          ROWCND هو DOUBLE PRECISION
!>          إذا كان INFO = 0 أو INFO > M، يحتوي ROWCND على نسبة
!>          أصغر R(i) إلى أكبر R(i).  إذا كان ROWCND >= 0.1 و
!>          AMAX ليس كبيرًا جدًا أو صغيرًا جدًا، فلا يستحق الأمر
!>          المعايرة بواسطة R.
!>

COLCND

!>          COLCND هو DOUBLE PRECISION
!>          إذا كان INFO = 0، يحتوي COLCND على نسبة
!>          أصغر C(i) إلى أكبر C(i).  إذا كان COLCND >= 0.1، فلا
!>          يستحق الأمر المعايرة بواسطة C.
!>

AMAX

!>          AMAX هو DOUBLE PRECISION
!>          القيمة المطلقة لأكبر عنصر في المصفوفة.  إذا كان AMAX قريبًا جدًا
!>          من الفيضان أو قريبًا جدًا من النقصان، فيجب معايرة المصفوفة.
!>

INFO

!>          INFO هو عدد صحيح (INTEGER)
!>          = 0:  خروج ناجح
!>          < 0:  إذا كان INFO = -i، فقد كان المعامل i ذو قيمة غير قانونية
!>          > 0:  إذا كان INFO = i، وكان i
!>                <= M:  الصف i من A هو صفر تمامًا
!>                >  M:  العمود (i-M) من A هو صفر تمامًا
!>
