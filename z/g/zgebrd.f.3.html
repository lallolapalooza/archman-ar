وثائق الوظيفة/الوحدة الفرعية

subroutine zgebrd (integer m, integer n, complex*16, dimension( lda, * ) a, integer lda, double precision, dimension( * ) d, double precision, dimension( * ) e, complex*16, dimension( * ) tauq, complex*16, dimension( * ) taup, complex*16, dimension( * ) work, integer lwork, integer info)

ZGEBRD

الغرض:

!>
!> ZGEBRD تختزل مصفوفة A العامة المعقدة M-by-N إلى شكل B ثنائي القطر علوي أو سفلي
!> بواسطة تحويل وحدوي: Q**H * A * P = B.
!>
!> إذا كان m >= n، فإن B يكون ثنائي القطر علوي؛ إذا كان m < n، فإن B يكون ثنائي القطر سفلي.
!>

المعاملات

M 
!>          M هو عدد صحيح (INTEGER)
!>          عدد الصفوف في المصفوفة A.  M >= 0.
!>

N

!>          N هو عدد صحيح (INTEGER)
!>          عدد الأعمدة في المصفوفة A.  N >= 0.
!>

A

!>          A هي مصفوفة COMPLEX*16، البعد (LDA,N)
!>          عند الدخول، المصفوفة العامة M-by-N المطلوب اختزالها.
!>          عند الخروج،
!>          إذا كان m >= n، يتم الكتابة فوق القطر الرئيسي والقطر العلوي الأول بالمصفوفة ثنائية القطر B؛
!>            العناصر الموجودة تحت القطر الرئيسي، مع المصفوفة TAUQ، تمثل المصفوفة الوحدوية Q كناتج
!>            عواكس أولية، والعناصر فوق القطر العلوي الأول، مع المصفوفة TAUP، تمثل المصفوفة الوحدوية P كناتج
!>            عواكس أولية؛
!>          إذا كان m < n، يتم الكتابة فوق القطر الرئيسي والقطر السفلي الأول بالمصفوفة ثنائية القطر B؛
!>            العناصر الموجودة تحت القطر السفلي الأول، مع المصفوفة TAUQ، تمثل المصفوفة الوحدوية Q كناتج
!>            عواكس أولية، والعناصر الموجودة فوق القطر الرئيسي، مع المصفوفة TAUP، تمثل المصفوفة الوحدوية P كناتج
!>            عواكس أولية.
!>          انظر التفاصيل الإضافية.
!>

LDA

!>          LDA هو عدد صحيح (INTEGER)
!>          البعد الرائد للمصفوفة A.  LDA >= max(1,M).
!>

D

!>          D هي مصفوفة DOUBLE PRECISION، البعد (min(M,N))
!>          عناصر القطر الرئيسي للمصفوفة ثنائية القطر B:
!>          D(i) = A(i,i).
!>

E

!>          E هي مصفوفة DOUBLE PRECISION، البعد (min(M,N)-1)
!>          عناصر القطر غير الرئيسي للمصفوفة ثنائية القطر B:
!>          إذا كان m >= n، فإن E(i) = A(i,i+1) لـ i = 1,2,...,n-1؛
!>          إذا كان m < n، فإن E(i) = A(i+1,i) لـ i = 1,2,...,m-1.
!>

TAUQ

!>          TAUQ هي مصفوفة COMPLEX*16، البعد (min(M,N))
!>          العوامل القياسية للعواكس الأولية التي تمثل المصفوفة الوحدوية Q. انظر التفاصيل الإضافية.
!>

TAUP

!>          TAUP هي مصفوفة COMPLEX*16، البعد (min(M,N))
!>          العوامل القياسية للعواكس الأولية التي تمثل المصفوفة الوحدوية P. انظر التفاصيل الإضافية.
!>

WORK

!>          WORK هي مصفوفة COMPLEX*16، البعد (MAX(1,LWORK))
!>          عند الخروج، إذا كان INFO = 0، فإن WORK(1) يعيد LWORK الأمثل.
!>

LWORK

!>          LWORK هو عدد صحيح (INTEGER)
!>          طول المصفوفة WORK.
!>          LWORK >= 1، إذا كان MIN(M,N) = 0، و LWORK >= MAX(M,N)، بخلاف ذلك.
!>          لأداء أمثل LWORK >= (M+N)*NB، حيث NB هو حجم الكتلة الأمثل.
!>
!>          إذا كان LWORK = -1، فسيتم افتراض استعلام عن مساحة العمل؛ الوحدة الفرعية
!>          تحسب فقط الحجم الأمثل للمصفوفة WORK، وتعيد
!>          هذه القيمة كإدخال أول للمصفوفة WORK، ولا يصدر XERBLA أي رسالة خطأ
!>          متعلقة بـ LWORK.
!>

INFO

!>          INFO هو عدد صحيح (INTEGER)
!>          = 0:  انتهاء ناجح.
!>          < 0:  إذا كان INFO = -i، فإن المعامل i كان له قيمة غير قانونية.
!>

تفاصيل إضافية:

!>
!>  يتم تمثيل المصفوفتين Q و P كناتج عواكس أولية:
!>
!>  إذا كان m >= n،
!>
!>     Q = H(1) H(2) . . . H(n)  و  P = G(1) G(2) . . . G(n-1)
!>
!>  كل H(i) و G(i) له الشكل:
!>
!>     H(i) = I - tauq * v * v**H  و G(i) = I - taup * u * u**H
!>
!>  حيث tauq و taup هما عاملان قياسيان معقدان، و v و u متجهان معقدان؛ v(1:i-1) = 0، v(i) = 1،
!>  ويتم تخزين v(i+1:m) عند الخروج في A(i+1:m,i)؛ u(1:i) = 0، u(i+1) = 1،
!>  ويتم تخزين u(i+2:n) عند الخروج في A(i,i+2:n)؛ يتم تخزين tauq في TAUQ(i) و taup في TAUP(i).
!>
!>  إذا كان m < n،
!>
!>     Q = H(1) H(2) . . . H(m-1)  و  P = G(1) G(2) . . . G(m)
!>
!>  كل H(i) و G(i) له الشكل:
!>
!>     H(i) = I - tauq * v * v**H  و G(i) = I - taup * u * u**H
!>
!>  حيث tauq و taup هما عاملان قياسيان معقدان، و v و u متجهان معقدان؛ v(1:i) = 0، v(i+1) = 1،
!>  ويتم تخزين v(i+2:m) عند الخروج في A(i+2:m,i)؛ u(1:i-1) = 0، u(i) = 1،
!>  ويتم تخزين u(i+1:n) عند الخروج في A(i,i+1:n)؛ يتم تخزين tauq في TAUQ(i) و taup في TAUP(i).
!>
!>  محتويات A عند الخروج موضحة بالأمثلة التالية:
!>
!>  m = 6 و n = 5 (m > n):          m = 5 و n = 6 (m < n):
!>
!>    (  d   e   u1  u1  u1 )           (  d   u1  u1  u1  u1  u1 )
!>    (  v1  d   e   u2  u2 )           (  e   d   u2  u2  u2  u2 )
!>    (  v1  v2  d   e   u3 )           (  v1  e   d   u3  u3  u3 )
!>    (  v1  v2  v3  d   e  )           (  v1  v2  e   d   u4  u4 )
!>    (  v1  v2  v3  v4  d  )           (  v1  v2  v3  e   d   u5 )
!>    (  v1  v2  v3  v4  v5 )
!>
!>  حيث d و e يمثلان عناصر القطر الرئيسي وغير الرئيسي لـ B، vi يمثل عنصراً من المتجه الذي يحدد H(i)،
!>  و ui عنصراً من المتجه الذي يحدد G(i).
!>
