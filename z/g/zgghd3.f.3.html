subroutine zgghd3 (character compq, character compz, integer n, integer ilo, integer ihi, complex*16, dimension( lda, * ) a, integer lda, complex*16, dimension( ldb, * ) b, integer ldb, complex*16, dimension( ldq, * ) q, integer ldq, complex*16, dimension( ldz, * ) z, integer ldz, complex*16, dimension( * ) work, integer lwork, integer info)

ZGGHD3

الغرض:

!>
!> ZGGHD3 يقلل زوجًا من المصفوفات المعقدة (A,B) إلى الشكل الهندسي العلوي
!> باستخدام تحويلات وحدوية، حيث A مصفوفة عامة و B مصفوفة مثلثية عليا.
!> شكل مشكلة القيمة الذاتية المعممة هو
!>    A*x = lambda*B*x,
!> ويتم عادةً جعل B مثلثية عليا عن طريق حساب تحليل QR الخاص بها
!> ونقل المصفوفة الوحدوية Q إلى الجانب الأيسر من المعادلة.
!>
!> هذا الروتين الفرعي يقلل A بشكل متزامن إلى مصفوفة هندسية H:
!>    Q**H*A*Z = H
!> ويحول B إلى مصفوفة مثلثية عليا أخرى T:
!>    Q**H*B*Z = T
!> لتقليل المشكلة إلى شكلها القياسي
!>    H*y = lambda*T*y
!> حيث y = Z**H*x.
!>
!> يتم تحديد المصفوفات الوحدوية Q و Z كحاصل ضرب لدورانات جيفنز.
!> يمكن تشكيلهما بشكل صريح، أو يمكن ضربهما لاحقًا في مصفوفات الإدخال Q1 و Z1، بحيث
!>      Q1 * A * Z1**H = (Q1*Q) * H * (Z1*Z)**H
!>      Q1 * B * Z1**H = (Q1*Q) * T * (Z1*Z)**H
!> إذا كانت Q1 هي المصفوفة الوحدوية من تحليل QR لـ B في المعادلة الأصلية
!> A*x = lambda*B*x، فإن ZGGHD3 يقلل المشكلة الأصلية إلى الشكل الهندسي المعمم.
!>
!> هذا هو متغير محظور من CGGHRD، يستخدم ضرب المصفوفات بالمصفوفات
!> لأجزاء من الحساب لتعزيز الأداء.
!>

المعلمات

COMPQ 
!>          COMPQ هو CHARACTER*1
!>          = 'N': لا تحسب Q؛
!>          = 'I': يتم تهيئة Q إلى المصفوفة الوحدوية، ويتم إرجاع
!>                 المصفوفة الوحدوية Q؛
!>          = 'V': يجب أن تحتوي Q على مصفوفة وحدوية Q1 عند الدخول،
!>                 ويتم إرجاع حاصل الضرب Q1*Q.
!>

COMPZ

!>          COMPZ هو CHARACTER*1
!>          = 'N': لا تحسب Z؛
!>          = 'I': يتم تهيئة Z إلى المصفوفة الوحدوية، ويتم إرجاع
!>                 المصفوفة الوحدوية Z؛
!>          = 'V': يجب أن تحتوي Z على مصفوفة وحدوية Z1 عند الدخول،
!>                 ويتم إرجاع حاصل الضرب Z1*Z.
!>

N

!>          N هو INTEGER
!>          رتبة المصفوفات A و B. N >= 0.
!>

ILO

!>          ILO هو INTEGER
!>

IHI

!>          IHI هو INTEGER
!>
!>          ILO و IHI يحددان الصفوف والأعمدة من A التي سيتم تقليلها.
!>          من المفترض أن A هي بالفعل مثلثية عليا
!>          في الصفوف والأعمدة 1:ILO-1 و IHI+1:N. عادةً ما يتم تعيين ILO و IHI
!>          بواسطة استدعاء سابق لـ ZGGBAL؛ وإلا فيجب تعيينهما إلى 1 و N على التوالي.
!>          1 <= ILO <= IHI <= N، إذا كان N > 0؛ ILO=1 و IHI=0، إذا كان N=0.
!>

A

!>          A هي مصفوفة COMPLEX*16، الأبعاد (LDA, N)
!>          عند الدخول، المصفوفة العامة N×N المراد تقليلها.
!>          عند الخروج، المثلث العلوي والقطري الفرعي الأول من A
!>          يتم استبدالهما بالمصفوفة الهندسية العليا H، والباقي يتم تعيينه إلى صفر.
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرائد للمصفوفة A. LDA >= max(1,N).
!>

B

!>          B هي مصفوفة COMPLEX*16، الأبعاد (LDB, N)
!>          عند الدخول، المصفوفة المثلثية العليا B بحجم N×N.
!>          عند الخروج، المصفوفة المثلثية العليا T = Q**H B Z.
!>          العناصر تحت القطر يتم تعيينها إلى صفر.
!>

LDB

!>          LDB هو INTEGER
!>          البعد الرائد للمصفوفة B. LDB >= max(1,N).
!>

Q

!>          Q هي مصفوفة COMPLEX*16، الأبعاد (LDQ, N)
!>          عند الدخول، إذا كان COMPQ = 'V'، المصفوفة الوحدوية Q1، عادةً
!>          من تحليل QR لـ B.
!>          عند الخروج، إذا كان COMPQ='I'، المصفوفة الوحدوية Q، وإذا كان
!>          COMPQ = 'V'، حاصل الضرب Q1*Q.
!>          لا يتم الاستشهاد به إذا كان COMPQ='N'.
!>

LDQ

!>          LDQ هو INTEGER
!>          البعد الرائد للمصفوفة Q.
!>          LDQ >= N إذا كان COMPQ='V' أو 'I'؛ LDQ >= 1 بخلاف ذلك.
!>

Z

!>          Z هي مصفوفة COMPLEX*16، الأبعاد (LDZ, N)
!>          عند الدخول، إذا كان COMPZ = 'V'، المصفوفة الوحدوية Z1.
!>          عند الخروج، إذا كان COMPZ='I'، المصفوفة الوحدوية Z، وإذا كان
!>          COMPZ = 'V'، حاصل الضرب Z1*Z.
!>          لا يتم الاستشهاد به إذا كان COMPZ='N'.
!>

LDZ

!>          LDZ هو INTEGER
!>          البعد الرائد للمصفوفة Z.
!>          LDZ >= N إذا كان COMPZ='V' أو 'I'؛ LDZ >= 1 بخلاف ذلك.
!>

WORK

!>          WORK هي مصفوفة COMPLEX*16، الأبعاد (MAX(1,LWORK))
!>          عند الخروج، إذا كان INFO = 0، فإن WORK(1) تُرجع LWORK المثلى.
!>

LWORK

!>          LWORK هو INTEGER
!>          طول المصفوفة WORK. LWORK >= 1.
!>          للحصول على الأداء الأمثل، LWORK >= 6*N*NB، حيث NB هو
!>          حجم الكتلة الأمثل.
!>
!>          إذا كان LWORK = -1، فسيتم افتراض استعلام مساحة العمل؛ الروتين
!>          يحسب فقط الحجم الأمثل لمصفوفة WORK، ويعيد
!>          هذه القيمة كإدخال أول في مصفوفة WORK، ولا يصدر أي رسالة خطأ
!>          متعلقة بـ LWORK بواسطة XERBLA.
!>

INFO

!>          INFO هو INTEGER
!>          = 0:  انتهاء ناجح.
!>          < 0:  إذا كان INFO = -i، فإن المعامل i كان له قيمة غير قانونية.
!>
