subroutine zgrqts (integer m, integer p, integer n, complex*16, dimension( lda, * ) a, complex*16, dimension( lda, * ) af, complex*16, dimension( lda, * ) q, complex*16, dimension( lda, * ) r, integer lda, complex*16, dimension( * ) taua, complex*16, dimension( ldb, * ) b, complex*16, dimension( ldb, * ) bf, complex*16, dimension( ldb, * ) z, complex*16, dimension( ldb, * ) t, complex*16, dimension( ldb, * ) bwk, integer ldb, complex*16, dimension( * ) taub, complex*16, dimension( lwork ) work, integer lwork, double precision, dimension( * ) rwork, double precision, dimension( 4 ) result)

ZGRQTS

الغرض:

!>
!> ZGRQTS تختبر ZGGRQF، التي تقوم بحساب تحليل GR لمصفوفة A ذات الأبعاد M-by-N ومصفوفة B ذات الأبعاد P-by-N: A = R*Q و B = Z*T*Q.
!>

المعلمات

M 
!>          M هو عدد صحيح (INTEGER)
!>          عدد صفوف المصفوفة A.  M >= 0.
!>

P

!>          P هو عدد صحيح (INTEGER)
!>          عدد صفوف المصفوفة B.  P >= 0.
!>

N

!>          N هو عدد صحيح (INTEGER)
!>          عدد أعمدة المصفوفتين A و B.  N >= 0.
!>

A

!>          A هي مصفوفة COMPLEX*16، أبعادها (LDA,N)
!>          المصفوفة A ذات الأبعاد M-by-N.
!>

AF

!>          AF هي مصفوفة COMPLEX*16، أبعادها (LDA,N)
!>          تفاصيل تحليل GR للمصفوفتين A و B، كما تم إرجاعها
!>          بواسطة ZGGRQF، راجع CGGRQF لمزيد من التفاصيل.
!>

Q

!>          Q هي مصفوفة COMPLEX*16، أبعادها (LDA,N)
!>          المصفوفة الوحدوية Q ذات الأبعاد N-by-N.
!>

R

!>          R هي مصفوفة COMPLEX*16، أبعادها (LDA,MAX(M,N))
!>

LDA

!>          LDA هو عدد صحيح (INTEGER)
!>          البعد الرائد للمصفوفات A, AF, R و Q.
!>          LDA >= max(M,N).
!>

TAUA

!>          TAUA هي مصفوفة COMPLEX*16، أبعادها (min(M,N))
!>          العوامل القياسية للانعكاسات الأولية، كما تم إرجاعها
!>          بواسطة DGGQRC.
!>

B

!>          B هي مصفوفة COMPLEX*16، أبعادها (LDB,N)
!>          عند الإدخال، المصفوفة A ذات الأبعاد P-by-N.
!>

BF

!>          BF هي مصفوفة COMPLEX*16، أبعادها (LDB,N)
!>          تفاصيل تحليل GQR للمصفوفتين A و B، كما تم إرجاعها
!>          بواسطة ZGGRQF، راجع CGGRQF لمزيد من التفاصيل.
!>

Z

!>          Z هي مصفوفة DOUBLE PRECISION، أبعادها (LDB,P)
!>          المصفوفة الوحدوية Z ذات الأبعاد P-by-P.
!>

T

!>          T هي مصفوفة COMPLEX*16، أبعادها (LDB,max(P,N))
!>

BWK

!>          BWK هي مصفوفة COMPLEX*16، أبعادها (LDB,N)
!>

LDB

!>          LDB هو عدد صحيح (INTEGER)
!>          البعد الرائد للمصفوفات B, BF, Z و T.
!>          LDB >= max(P,N).
!>

TAUB

!>          TAUB هي مصفوفة COMPLEX*16، أبعادها (min(P,N))
!>          العوامل القياسية للانعكاسات الأولية، كما تم إرجاعها
!>          بواسطة DGGRQF.
!>

WORK

!>          WORK هي مصفوفة COMPLEX*16، أبعادها (LWORK)
!>

LWORK

!>          LWORK هو عدد صحيح (INTEGER)
!>          بعد المصفوفة WORK، LWORK >= max(M,P,N)**2.
!>

RWORK

!>          RWORK هي مصفوفة DOUBLE PRECISION، أبعادها (M)
!>

RESULT

!>          RESULT هي مصفوفة DOUBLE PRECISION، أبعادها (4)
!>          نسب الاختبار:
!>            RESULT(1) = norm( R - A*Q' ) / ( MAX(M,N)*norm(A)*ULP)
!>            RESULT(2) = norm( T*Q - Z'*B ) / (MAX(P,N)*norm(B)*ULP)
!>            RESULT(3) = norm( I - Q'*Q ) / ( N*ULP )
!>            RESULT(4) = norm( I - Z'*Z ) / ( P*ULP )
!>
