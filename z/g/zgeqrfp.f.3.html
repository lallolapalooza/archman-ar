subroutine zgeqrfp (integer m, integer n, complex*16, dimension( lda, * ) a, integer lda, complex*16, dimension( * ) tau, complex*16, dimension( * ) work, integer lwork, integer info)

ZGEQRFP

الغرض:

!>
!> ZGEQR2P يحسب تفكيك QR لمصفوفة معقدة M-by-N A:
!>
!>    A = Q * ( R ),
!>            ( 0 )
!>
!> حيث:
!>
!>    Q هي مصفوفة متعامدة M-by-M؛
!>    R هي مصفوفة مثلثية عليا N-by-N ذات مدخلات قطرية غير سالبة؛
!>    0 هي مصفوفة صفرية (M-N)-by-N، إذا كان M > N.
!>
!>

المعاملات

M 
!>          M هو عدد صحيح
!>          عدد صفوف المصفوفة A. M >= 0.
!>

N

!>          N هو عدد صحيح
!>          عدد أعمدة المصفوفة A. N >= 0.
!>

A

!>          A هو مصفوفة COMPLEX*16، البعد (LDA,N)
!>          عند الإدخال، المصفوفة A بحجم M-by-N.
!>          عند الخروج، تحتوي العناصر على القطر وما فوقه من المصفوفة
!>          على المصفوفة شبه المنحرفة العلوية R بحجم min(M,N)-by-N (R
!>          هي مثلثية عليا إذا كان m >= n). مدخلات R القطرية
!>          حقيقية وغير سالبة؛ العناصر تحت القطر،
!>          مع المصفوفة TAU، تمثل المصفوفة الوحدية Q كـ
!>          ناتج لـ min(m,n) عاكسات أولية (انظر التفاصيل الإضافية).
!>

LDA

!>          LDA هو عدد صحيح
!>          البعد الرائد للمصفوفة A. LDA >= max(1,M).
!>

TAU

!>          TAU هي مصفوفة COMPLEX*16، البعد (min(M,N))
!>          العوامل القياسية للعواكس الأولية (انظر التفاصيل الإضافية).
!>

WORK

!>          WORK هي مصفوفة COMPLEX*16، البعد (MAX(1,LWORK))
!>          عند الخروج، إذا كان INFO = 0، يعيد WORK(1) القيمة LWORK المثلى.
!>

LWORK

!>          LWORK هو عدد صحيح
!>          بعد المصفوفة WORK.
!>          LWORK >= 1، إذا كان MIN(M,N) = 0، و LWORK >= N، بخلاف ذلك.
!>          لأداء مثالي LWORK >= N*NB، حيث NB
!>          هي حجم الكتلة الأمثل.
!>
!>          إذا كان LWORK = -1، فسيتم افتراض استعلام عن مساحة العمل؛ يقوم الإجراء
!>          فقط بحساب الحجم الأمثل للمصفوفة WORK، ويعيد
!>          هذه القيمة كإدخال أول في المصفوفة WORK، ولا يتم إصدار أي رسالة خطأ
!>          متعلقة بـ LWORK بواسطة XERBLA.
!>

INFO

!>          INFO هو عدد صحيح
!>          = 0:  انتهاء ناجح
!>          < 0:  إذا كان INFO = -i، كان وسيط الإدخال i ذو قيمة غير قانونية
!>

تفاصيل إضافية:

!>
!>  يتم تمثيل المصفوفة Q كناتج لعواكس أولية
!>
!>     Q = H(1) H(2) . . . H(k)، حيث k = min(m,n).
!>
!>  كل H(i) له الشكل
!>
!>     H(i) = I - tau * v * v**H
!>
!>  حيث tau هو عامل قياسي معقد، و v هو متجه معقد ذو
!>  v(1:i-1) = 0 و v(i) = 1؛ يتم تخزين v(i+1:m) عند الخروج في A(i+1:m,i)،
!>  و tau في TAU(i).
!>
!> انظر مذكرة عمل Lapack 203 للتفاصيل
!>
