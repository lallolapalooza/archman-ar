subroutine zgbtrf (integer m, integer n, integer kl, integer ku, complex*16, dimension( ldab, * ) ab, integer ldab, integer, dimension( * ) ipiv, integer info)

ZGBTRF

الغرض:

!>
!> ZGBTRF يحسب تحليل LU لمصفوفة نطاق معقدة m-by-n A
!> باستخدام تحويل الصفوف الجزئي مع تبديلات الصفوف.
!>
!> هذا هو الإصدار المجمع من الخوارزمية، استدعاء BLAS المستوى 3.
!>

المعلمات

M 
!>          M هو عدد صحيح (INTEGER)
!>          عدد صفوف المصفوفة A.  M >= 0.
!>

N

!>          N هو عدد صحيح (INTEGER)
!>          عدد أعمدة المصفوفة A.  N >= 0.
!>

KL

!>          KL هو عدد صحيح (INTEGER)
!>          عدد الأقطار الفرعية داخل نطاق A.  KL >= 0.
!>

KU

!>          KU هو عدد صحيح (INTEGER)
!>          عدد الأقطار الفائقة داخل نطاق A.  KU >= 0.
!>

AB

!>          AB هو مصفوفة مركبة * 16 (COMPLEX*16) ، البعد (LDAB,N)
!>          عند الدخول، المصفوفة A بتخزين النطاق، في الصفوف KL+1 إلى
!>          2*KL+KU+1؛ لا تحتاج الصفوف 1 إلى KL من المصفوفة إلى التعيين.
!>          يتم تخزين العمود j من A في العمود j من
!>          المصفوفة AB كما يلي:
!>          AB(kl+ku+1+i-j,j) = A(i,j) لـ max(1,j-ku) <= i <= min(m,j+kl)
!>
!>          عند الخروج، تفاصيل التحليل: يتم تخزين U كـ
!>          مصفوفة مثلثة عليا ذات نطاق KL+KU فائق، في
!>          الصفوف 1 إلى KL+KU+1، والعوامل المستخدمة أثناء
!>          التحليل يتم تخزينها في الصفوف KL+KU+2 إلى 2*KL+KU+1.
!>          انظر أدناه لمزيد من التفاصيل.
!>

LDAB

!>          LDAB هو عدد صحيح (INTEGER)
!>          البعد الرائد للمصفوفة AB.  LDAB >= 2*KL+KU+1.
!>

IPIV

!>          IPIV هي مصفوفة أعداد صحيحة (INTEGER)، البعد (min(M,N))
!>          فهارس المحور؛ لـ 1 <= i <= min(M,N)، تم تبديل الصف i من
!>          المصفوفة مع الصف IPIV(i).
!>

INFO

!>          INFO هو عدد صحيح (INTEGER)
!>          = 0: خروج ناجح
!>          < 0: إذا كان INFO = -i، فإن الوسيط i لديه قيمة غير قانونية
!>          > 0: إذا كان INFO = +i، فإن U(i,i) يساوي صفرًا تمامًا. التحليل
!>               تم إكماله، لكن العامل U هو بالضبط
!>               فردي، وسيحدث قسمة على صفر إذا تم استخدامه
!>               لحل نظام من المعادلات.
!>

!>
!>  عند الدخول:                       عند الخروج:
!>
!>      *    *    *    +    +    +       *    *    *   u14  u25  u36

!>      *   a12  a23  a34  a45  a56      *   u12  u23  u34  u45  u56
!>     a11  a22  a33  a44  a55  a66     u11  u22  u33  u44  u55  u66
!>     a21  a32  a43  a54  a65   *      m21  m32  m43  m54  m65   *
!>     a31  a42  a53  a64   *    *      m31  m42  m53  m64   *    *
!>
!>  العناصر (*) في المصفوفة لا تستخدمها الروتين؛ العناصر
