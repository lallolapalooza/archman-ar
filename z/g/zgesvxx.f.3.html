subroutine zgesvxx (character fact, character trans, integer n, integer nrhs, complex*16, dimension( lda, * ) a, integer lda, complex*16, dimension( ldaf, * ) af, integer ldaf, integer, dimension( * ) ipiv, character equed, double precision, dimension( * ) r, double precision, dimension( * ) c, complex*16, dimension( ldb, * ) b, integer ldb, complex*16, dimension( ldx , * ) x, integer ldx, double precision rcond, double precision rpvgrw, double precision, dimension( * ) berr, integer n_err_bnds, double precision, dimension( nrhs, * ) err_bnds_norm, double precision, dimension( nrhs, * ) err_bnds_comp, integer nparams, double precision, dimension( * ) params, complex*16, dimension( * ) work, double precision, dimension( * ) rwork, integer info)

ZGESVXX يحسب الحل لنظام المعادلات الخطية A * X = B لمصفوفات GE

الغرض:

!>
!>    ZGESVXX يستخدم تحليل LU لحساب الحل لنظام المعادلات الخطية المركبة*16
!>    A * X = B، حيث A مصفوفة N-بـ N و X و B مصفوفتان N-بـ NRHS.
!>
!>    إذا طُلب، سيتم إرجاع حدود الخطأ القياسية والحد الأقصى المكونية.
!>    ستُرجع ZGESVXX حلاً بخطأ صغير مضمون (O(eps) حيث eps هي دقة الآلة العاملة)
!>    ما لم تكن المصفوفة سيئة التكييف للغاية، وفي هذه الحالة يتم إرجاع تحذير.
!>    يتم أيضًا حساب أرقام التكييف ذات الصلة وإرجاعها.
!>
!>    تقبل ZGESVXX التحليلات وعوامل المعايرة المقدمة من المستخدم؛
!>    راجع تعريفات خيارات FACT و EQUED.
!>    سيؤدي الحل مع التنقيح واستخدام تحليل من استدعاء ZGESVXX سابق أيضًا
!>    إلى حل بخطأ O(eps) أو تحذيرات، ولكن لا يمكننا الادعاء بذلك
!>    للتحليلات وعوامل المعايرة العامة المقدمة من المستخدم إذا كانت
!>    تختلف عن ما تنتجه ZGESVXX بنفسها.
!>

الوصف:

!>
!>    يتم تنفيذ الخطوات التالية:
!>
!>    1. إذا كان FACT = 'E'، يتم حساب عوامل القياس مزدوجة الدقة لمعايرة
!!>    النظام:
!>
!>      TRANS = 'N':  diag(R)*A*diag(C)     *inv(diag(C))*X = diag(R)*B
!>      TRANS = 'T': (diag(R)*A*diag(C))**T *inv(diag(R))*X = diag(C)*B
!>      TRANS = 'C': (diag(R)*A*diag(C))**H *inv(diag(R))*X = diag(C)*B
!>
!>    يعتمد ما إذا كان سيتم معايرة النظام أم لا على
!>    قياس المصفوفة A، ولكن إذا تم استخدام المعايرة، يتم
!!>    الكتابة فوق A بواسطة diag(R)*A*diag(C) و B بواسطة diag(R)*B (إذا كان TRANS='N')
!>    أو diag(C)*B (إذا كان TRANS = 'T' أو 'C').
!>
!>    2. إذا كان FACT = 'N' أو 'E'، يتم استخدام تحليل LU لتحليل
!>    المصفوفة A (بعد المعايرة إذا كان FACT = 'E') كما يلي:
!>
!>      A = P * L * U,
!>
!>    حيث P مصفوفة تبديل، L مصفوفة ثلاثية سفلية وحدة، و U مصفوفة مثلثة عليا.
!>
!>    3. إذا كان بعض U(i,i)=0، بحيث تكون U مفردة تمامًا، فإن
!>    الإجراء الفرعي يرجع مع INFO = i. بخلاف ذلك، يتم استخدام الشكل المحلل لـ A
!>    لتقدير رقم تكييف المصفوفة A (انظر الوسيط RCOND). إذا كان
!>    معكوس رقم التكييف أقل من دقة الآلة، يستمر الإجراء الفرعي
!>    في الحل لـ X وحساب حدود الخطأ كما هو موضح أدناه.
!>
!>    4. يتم حل نظام المعادلات لـ X باستخدام الشكل المحلل لـ A.
!>
!>    5. افتراضيًا (ما لم يتم تعيين PARAMS(LA_LINRX_ITREF_I) إلى صفر)،
!>    سيستخدم الإجراء الفرعي التنقيح التكراري لمحاولة الحصول على خطأ صغير
!>    وحدود خطأ. يحسب التنقيح الباقي بدقة لا تقل عن ضعف دقة العمل.
!>
!>    6. إذا تم استخدام المعايرة، يتم ضرب المصفوفة X مسبقًا بواسطة
!>    diag(C) (إذا كان TRANS = 'N') أو diag(R) (إذا كان TRANS = 'T' أو 'C') بحيث
!>    يحل النظام الأصلي قبل المعايرة.
!> 
!>     يتم تجميع بعض المعاملات الاختيارية في مصفوفة PARAMS. تحدد هذه
!>     الإعدادات كيفية أداء التنقيح، ولكن غالبًا ما تكون القيم الافتراضية مقبولة.
!>     إذا كانت القيم الافتراضية مقبولة، يمكن للمستخدمين
!>     تمرير NPARAMS = 0 مما يمنع الكود المصدري من الوصول إلى
!>     الوسيط PARAMS.
!>

المعاملات

FACT 
!>          FACT is CHARACTER*1
!>     يحدد ما إذا كان الشكل المحلل للمصفوفة A
!>     مقدمًا عند الإدخال، وإذا لم يكن كذلك، ما إذا كان يجب معايرة المصفوفة A
!>     قبل تحليلها.
!>       = 'F':  عند الإدخال، تحتوي AF و IPIV على الشكل المحلل لـ A.
!>               إذا كان EQUED ليس 'N'، فقد تمت معايرة المصفوفة A
!>               باستخدام عوامل القياس في R و/أو C.
!>               لا يتم تعديل A أو AF أو IPIV.
!>       = 'N':  سيتم نسخ المصفوفة A إلى AF وتحليلها.
!>       = 'E':  سيتم معايرة المصفوفة A إذا لزم الأمر، ثم
!>               نسخها إلى AF وتحليلها.
!>

TRANS

!>          TRANS is CHARACTER*1
!>     يحدد شكل نظام المعادلات:
!>       = 'N':  A * X = B     (بدون تبديل)
!>       = 'T':  A**T * X = B  (تبديل)
!>       = 'C':  A**H * X = B  (تبديل مقترن)
!>

N

!>          N is INTEGER
!>     عدد المعادلات الخطية، أي ترتيب
!>     المصفوفة A.  N >= 0.
!>

NRHS

!>          NRHS is INTEGER
!>     عدد الأطراف اليمنى، أي عدد الأعمدة
!>     للمصفوفتين B و X.  NRHS >= 0.
!>

A

!>          A is COMPLEX*16 array, dimension (LDA,N)
!>     عند الإدخال، المصفوفة A بحجم N × N. إذا كان FACT = 'F' وكان EQUED
!>     ليس 'N'، فيجب أن تكون A قد تمت معايرتها بواسطة عوامل القياس
!>     في R و/أو C.
!>     لا يتم تعديل A إذا كان FACT = 'F' أو
!>     'N'، أو إذا كان FACT = 'E' وكان EQUED = 'N' عند الخروج.
!>
!>     عند الخروج، إذا كان EQUED .ne. 'N'، تتم معايرة المصفوفة A كما يلي:
!>     EQUED = 'R':  A := diag(R) * A
!>     EQUED = 'C':  A := A * diag(C)
!>     EQUED = 'B':  A := diag(R) * A * diag(C).
!>

LDA

!>          LDA is INTEGER
!>     البعد الرئيسي للمصفوفة A.  LDA >= max(1,N).
!>

AF

!>          AF is COMPLEX*16 array, dimension (LDAF,N)
!>     إذا كان FACT = 'F'، فإن AF هو وسيط إدخال وعند الإدخال
!>     يحتوي على عاملي L و U من تحليل
!>     A = P*L*U كما تم حسابه بواسطة ZGETRF. إذا كان EQUED .ne. 'N'، فإن
!>     AF هو الشكل المحلل للمصفوفة A المعايرة.
!>
!>     إذا كان FACT = 'N'، فإن AF هو وسيط إخراج وعند الخروج
!>     يرجع عاملي L و U من تحليل A = P*L*U
!>     للمصفوفة الأصلية A.
!>
!>     إذا كان FACT = 'E'، فإن AF هو وسيط إخراج وعند الخروج
!>     يرجع عاملي L و U من تحليل A = P*L*U
