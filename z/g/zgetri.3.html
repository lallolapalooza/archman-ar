توثيق الوظيفة/الروتين الفرعي

subroutine zgetri (integer n, complex*16, dimension( lda, * ) a, integer lda, integer, dimension( * ) ipiv, complex*16, dimension( * ) work, integer lwork, integer info)

ZGETRI

الغرض:

!>
!> ZGETRI تحسب معكوس مصفوفة باستخدام تحليل LU
!> الذي تم حسابه بواسطة ZGETRF.
!>
!> تقوم هذه الطريقة بحساب معكوس U ثم تحسب inv(A) عن طريق حل النظام
!> inv(A)*L = inv(U) بالنسبة لـ inv(A).
!>

المعلمات

N 
!>          N هو عدد صحيح
!>          رتبة المصفوفة A.  N >= 0.
!>

A

!>          A هي مصفوفة COMPLEX*16، حجمها (LDA,N)
!>          عند الدخول، عوامل L و U من التحليل
!>          A = P*L*U كما تم حسابه بواسطة ZGETRF.
!>          عند الخروج، إذا كان INFO = 0، فإن معكوس المصفوفة الأصلية A.
!>

LDA

!>          LDA هو عدد صحيح
!>          البعد الرائد للمصفوفة A.  LDA >= max(1,N).
!>

IPIV

!>          IPIV هي مصفوفة عدد صحيح، حجمها (N)
!>          مؤشرات المحور من ZGETRF؛ لـ 1<=i<=N، تم تبديل الصف i من
!>          المصفوفة مع الصف IPIV(i).
!>

WORK

!>          WORK هي مصفوفة COMPLEX*16، حجمها (MAX(1,LWORK))
!>          عند الخروج، إذا كان INFO=0، فإن WORK(1) تعيد LWORK الأمثل.
!>

LWORK

!>          LWORK هو عدد صحيح
!>          بعد المصفوفة WORK.  LWORK >= max(1,N).
!>          للأداء الأمثل LWORK >= N*NB، حيث NB هو
!>          حجم الكتلة الأمثل الذي أعادته ILAENV.
!>
!>          إذا كان LWORK = -1، فيفترض استعلام مساحة العمل؛ الروتين
!>          يحسب فقط الحجم الأمثل للمصفوفة WORK، ويعيد
!>          هذه القيمة كالإدخال الأول للمصفوفة WORK، ولا يتم إصدار رسالة خطأ
!>          متعلقة بـ LWORK بواسطة XERBLA.
!>

INFO

!>          INFO هو عدد صحيح
!>          = 0:  إنهاء ناجح
!>          < 0:  إذا كان INFO = -i، كان الوسيط i له قيمة غير قانونية
!>          > 0:  إذا كان INFO = i، فإن U(i,i) تساوي صفرًا تمامًا؛ المصفوفة
!>                منفردة ولا يمكن حساب معكوسها.
!>
