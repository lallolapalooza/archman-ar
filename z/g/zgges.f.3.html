توثيق الوظيفة/الإجراء الفرعي

subroutine zgges (character jobvsl, character jobvsr, character sort, external selctg, integer n, complex*16, dimension( lda, * ) a, integer lda, complex*16, dimension( ldb, * ) b, integer ldb, integer sdim, complex*16, dimension( * ) alpha, complex*16, dimension( * ) beta, complex*16, dimension( ldvsl, * ) vsl, integer ldvsl, complex*16, dimension( ldvsr, * ) vsr, integer ldvsr, complex*16, dimension( * ) work, integer lwork, double precision, dimension( * ) rwork, logical, dimension( * ) bwork, integer info)

ZGGES تحسب القيم الذاتية، شكل شور المعمم، واختياريًا، متجهات شور اليسرى و/أو اليمنى لمصفوفتين N-by-N المعقدتين غير المتماثلتين (A,B)

الغرض:

!>
!> ZGGES تحسب لمجموعة من المصفوفات المعقدة غير المتماثلة N-by-N
!> (A,B)، القيم الذاتية المعممة، شكل شور المعمم (S, T)، واختياريًا، متجهات شور اليسرى و/أو اليمنى (VSL
!> و VSR). هذا يعطي التحلل المعمم لشكل شور
!>
!>         (A,B) = ( (VSL)*S*(VSR)**H, (VSL)*T*(VSR)**H )
!>
!> حيث (VSR)**H هو المرافق المتعامد لـ VSR.
!>
!> اختياريًا، تقوم أيضًا بترتيب القيم الذاتية بحيث تظهر مجموعة مختارة
!> من القيم الذاتية في الكتل القطرية الرائدة للمصفوفة شبه المنتظمة العلوية
!> S والمصفوفة شبه المنتظمة العلوية T. الأعمدة الرائدة لـ VSL و VSR
!> تشكل بعد ذلك أساسًا وحدويًا لمتجهات الفضاء الذاتي الأيسر والأيمن المقابلة (فضاءات التفاضل).
!>
!> (إذا كانت القيم الذاتية المعممة فقط مطلوبة، استخدم الدالة الموجهة
!> ZGGEV بدلاً من ذلك، وهي أسرع.)
!>
!> القيمة الذاتية المعممة لمجموعة مصفوفات (A,B) هي قيمة عددية w
!> أو نسبة alpha/beta = w، بحيث يكون A - w*B شاذًا. وعادة ما يتم تمثيلها
!> بالزوج (alpha,beta)، حيث يوجد تفسير معقول لـ beta=0، وحتى عندما يكون كلاهما صفرًا.
!>
!> مجموعة المصفوفات (S,T) تكون في شكل شور المعمم إذا كانت S
!> و T شبه منتظمة علوية، وبشكل إضافي، تكون العناصر القطرية
!> لـ T أرقامًا حقيقية غير سالبة.
!>

المعلمات

JOBVSL 
!>          JOBVSL هو CHARACTER*1
!>          = 'N':  لا تحسب متجهات شور اليسرى؛
!>          = 'V':  احسب متجهات شور اليسرى.
!>

JOBVSR

!>          JOBVSR هو CHARACTER*1
!>          = 'N':  لا تحسب متجهات شور اليمنى؛
!>          = 'V':  احسب متجهات شور اليمنى.
!>

SORT

!>          SORT هو CHARACTER*1
!>          تحدد ما إذا كان سيتم ترتيب القيم الذاتية على
!>          القطر لشكل شور المعمم أم لا.
!>          = 'N':  القيم الذاتية غير مرتبة؛
!>          = 'S':  القيم الذاتية مرتبة (انظر SELCTG).
!>

SELCTG

!>          SELCTG هي وظيفة منطقية ذات وسيطين COMPLEX*16
!>          يجب تعريف SELCTG كـ EXTERNAL في الإجراء الفرعي المستدعي.
!>          إذا كان SORT = 'N'، فلن يتم الرجوع إلى SELCTG.
!>          إذا كان SORT = 'S'، يتم استخدام SELCTG لاختيار القيم الذاتية
!>          لترتيبها في الجزء العلوي الأيسر من شكل شور.
!>          يتم اختيار القيمة الذاتية ALPHA(j)/BETA(j) إذا كان
!>          SELCTG(ALPHA(j),BETA(j)) صحيحًا.
!>
!>          لاحظ أن القيمة الذاتية المعقدة المختارة قد لا تفي بعد
!>          بـ SELCTG(ALPHA(j),BETA(j)) = .TRUE. بعد الترتيب، حيث أن
!>          الترتيب قد يغير قيمة القيم الذاتية المعقدة
!>          (خاصة إذا كانت القيمة الذاتية غير مستقرة)، في هذه الحالة
!>          يتم تعيين INFO إلى N+2 (انظر INFO أدناه).
!>

N

!>          N هو INTEGER
!>          رتبة المصفوفات A، B، VSL، و VSR.  N >= 0.
!>

A

!>          A هي مصفوفة COMPLEX*16، ذات الأبعاد (LDA, N)
!>          عند الدخول، الجزء الأول من زوج المصفوفات.
!>          عند الخروج، تم استبدال A بشكل شور المعمم الخاص بها
!>          S.
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرئيسي لـ A.  LDA >= max(1,N).
!>

B

!>          B هي مصفوفة COMPLEX*16، ذات الأبعاد (LDB, N)
!>          عند الدخول، الجزء الثاني من زوج المصفوفات.
!>          عند الخروج، تم استبدال B بشكل شور المعمم الخاص بها
!>          T.
!>

LDB

!>          LDB هو INTEGER
!>          البعد الرئيسي لـ B.  LDB >= max(1,N).
!>

SDIM

!>          SDIM هو INTEGER
!>          إذا كان SORT = 'N'، فإن SDIM = 0.
!>          إذا كان SORT = 'S'، فإن SDIM = عدد القيم الذاتية (بعد الترتيب)
!>          التي تكون SELCTG لها صحيحة.
!>

ALPHA

!>          ALPHA هي مصفوفة COMPLEX*16، ذات الأبعاد (N)
!>

BETA

!>          BETA هي مصفوفة COMPLEX*16، ذات الأبعاد (N)
!>          عند الخروج، ستكون ALPHA(j)/BETA(j)، j=1,...,N، هي
!>          القيم الذاتية المعممة.  ALPHA(j)، j=1,...,N  و  BETA(j)،
!>          j=1,...,N  هي الأقطار لشكل شور المعمم (A,B)
!>          الناتج عن ZGGES. ستكون BETA(j) أرقامًا حقيقية غير سالبة.
!>
!>          ملاحظة: قد تتجاوز نسب ALPHA(j)/BETA(j) أو تقل بسهولة،
!>          وقد تكون BETA(j) صفرًا حتى.  لذلك، يجب على المستخدم
!>          تجنب حساب النسبة alpha/beta بشكل سطحي.
!>          ومع ذلك، فإن ALPHA ستكون دائمًا أقل من ومعادلة عادةً
!>          لمعيار (norm) A في الحجم، و BETA دائمًا أقل من ومعادلة عادةً
!>          لمعيار (norm) B.
!>

VSL

!>          VSL هي مصفوفة COMPLEX*16، ذات الأبعاد (LDVSL,N)
!>          إذا كان JOBVSL = 'V'، فستحتوي VSL على متجهات شور اليسرى.
!>          لا يتم الرجوع إليها إذا كان JOBVSL = 'N'.
!>

LDVSL

!>          LDVSL هو INTEGER
!>          البعد الرئيسي للمصفوفة VSL. LDVSL >= 1، و
!>          إذا كان JOBVSL = 'V'، فإن LDVSL >= N.
!>

VSR

!>          VSR هي مصفوفة COMPLEX*16، ذات الأبعاد (LDVSR,N)
!>          إذا كان JOBVSR = 'V'، فستحتوي VSR على متجهات شور اليمنى.
!>          لا يتم الرجوع إليها إذا كان JOBVSR = 'N'.
!>

LDVSR

!>          LDVSR هو INTEGER
!>          البعد الرئيسي للمصفوفة VSR. LDVSR >= 1، و
!>          إذا كان JOBVSR = 'V'، فإن LDVSR >= N.
!>

WORK

!>          WORK هي مصفوفة COMPLEX*16، ذات الأبعاد (MAX(1,LWORK))
!>          عند الخروج، إذا كان INFO = 0، فإن WORK(1) ستعيد حجم LWORK الأمثل.
!>

LWORK

!>          LWORK هو INTEGER
!>          بعد المصفوفة WORK.  LWORK >= max(1,2*N).
!>          للحصول على أداء جيد، يجب أن يكون LWORK أكبر بشكل عام.
!>
!>          إذا كان LWORK = -1، فسيتم افتراض استعلام عن مساحة العمل؛
!>          تحسب الدالة فقط الحجم الأمثل لمصفوفة WORK، وتعيد
!>          هذه القيمة كأول مدخل في مصفوفة WORK، ولا يتم إصدار أي رسالة خطأ
!>          متعلقة بـ LWORK بواسطة XERBLA.
!>

RWORK

!>          RWORK هي مصفوفة DOUBLE PRECISION، ذات الأبعاد (8*N)
!>

BWORK

!>          BWORK هي مصفوفة LOGICAL، ذات الأبعاد (N)
!>          لا يتم الرجوع إليها إذا كان SORT = 'N'.
!>

INFO

!>          INFO هو INTEGER
!>          = 0:  تم الخروج بنجاح
!>          < 0:  إذا كان INFO = -i، فإن المعلمة i كانت لها قيمة غير قانونية.
!>
