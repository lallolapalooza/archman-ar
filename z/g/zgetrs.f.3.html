وثائق الوظيفة/الروتين الفرعي

subroutine zgetrs (character trans, integer n, integer nrhs, complex*16, dimension( lda, * ) a, integer lda, integer, dimension( * ) ipiv, complex*16, dimension( ldb, * ) b, integer ldb, integer info)

ZGETRS

الغرض:

!>
!> ZGETRS يحل نظام المعادلات الخطية
!>    A * X = B,  A**T * X = B,  أو  A**H * X = B
!> لمصفوفة عامة N-by-N A باستخدام تفكيك LU الذي تم حسابه
!> بواسطة ZGETRF.
!>

المعاملات

TRANS 
!>          TRANS هو CHARACTER*1
!>          يحدد شكل نظام المعادلات:
!>          = 'N':  A * X = B     (بدون تبديل)
!>          = 'T':  A**T * X = B  (تبديل)
!>          = 'C':  A**H * X = B  (تبديل مرافق)
!>

N

!>          N هو INTEGER
!>          رتبة المصفوفة A.  N >= 0.
!>

NRHS

!>          NRHS هو INTEGER
!>          عدد الأطراف اليمنى، أي عدد أعمدة
!>          المصفوفة B.  NRHS >= 0.
!>

A

!>          A هي مصفوفة COMPLEX*16، ذات البعد (LDA,N)
!>          العوامل L و U من تفكيك A = P*L*U
!>          كما تم حسابها بواسطة ZGETRF.
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرائد للمصفوفة A.  LDA >= max(1,N).
!>

IPIV

!>          IPIV هي مصفوفة INTEGER، ذات البعد (N)
!>          فهارس المحور من ZGETRF؛ لـ 1<=i<=N، تم تبديل الصف i من
!>          المصفوفة مع الصف IPIV(i).
!>

B

!>          B هي مصفوفة COMPLEX*16، ذات البعد (LDB,NRHS)
!>          عند الإدخال، مصفوفة الطرف الأيمن B.
!>          عند الخروج، مصفوفة الحل X.
!>

LDB

!>          LDB هو INTEGER
!>          البعد الرائد للمصفوفة B.  LDB >= max(1,N).
!>

INFO

!>          INFO هو INTEGER
!>          = 0:  خروج ناجح
!>          < 0:  إذا كان INFO = -i، فإن المعامل i كان لديه قيمة غير قانونية
!>
