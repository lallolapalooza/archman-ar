subroutine zgebak (character job, character side, integer n, integer ilo, integer ihi, double precision, dimension( * ) scale, integer m, complex*16, dimension( ldv, * ) v, integer ldv, integer info)

ZGEBAK

الغرض:

!>
!> ZGEBAK تشكل المتجهات الذاتية اليمنى أو اليسرى لمصفوفة عامة معقدة
!> عن طريق التحويل العكسي على المتجهات الذاتية المحسوبة للمصفوفة المتوازنة
!> الناتجة عن ZGEBAL.
!>

المعلمات

JOB 
!>          JOB is CHARACTER*1
!>          تحدد نوع التحويل العكسي المطلوب:
!>          = 'N': لا تفعل شيئًا، ارجع فورًا؛
!>          = 'P': قم بالتحويل العكسي للتبديل فقط؛
!>          = 'S': قم بالتحويل العكسي للتحجيم فقط؛
!>          = 'B': قم بإجراء تحويلات عكسية لكل من التبديل و
!>                 التحجيم.
!>          يجب أن يكون JOB هو نفس الوسيط JOB المقدم إلى ZGEBAL.
!>

SIDE

!>          SIDE is CHARACTER*1
!>          = 'R':  تحتوي V على المتجهات الذاتية اليمنى؛
!>          = 'L':  تحتوي V على المتجهات الذاتية اليسرى.
!>

N

!>          N is INTEGER
!>          عدد صفوف المصفوفة V. N >= 0.
!>

ILO

!>          ILO is INTEGER
!>

IHI

!>          IHI is INTEGER
!>          الأعداد الصحيحة ILO و IHI التي حددتها ZGEBAL.
!>          1 <= ILO <= IHI <= N، إذا كان N > 0؛ ILO=1 و IHI=0، إذا كان N=0.
!>

SCALE

!>          SCALE is DOUBLE PRECISION array, dimension (N)
!>          تفاصيل عوامل التبديل والتحجيم، كما تم إرجاعها
!>          من ZGEBAL.
!>

M

!>          M is INTEGER
!>          عدد أعمدة المصفوفة V. M >= 0.
!>

V

!>          V is COMPLEX*16 array, dimension (LDV,M)
!>          عند الإدخال، مصفوفة المتجهات الذاتية اليمنى أو اليسرى التي يجب
!>          تحويلها، كما تم إرجاعها بواسطة ZHSEIN أو ZTREVC.
!>          عند الخروج، يتم استبدال V بالمتجهات الذاتية المحولة.
!>

LDV

!>          LDV is INTEGER
!>          البعد الرائد للمصفوفة V. LDV >= max(1,N).
!>

INFO

!>          INFO is INTEGER
!>          = 0:  انتهاء ناجح
!>          < 0:  إذا كان INFO = -i، فإن الوسيط i كان له قيمة غير قانونية.
!>
