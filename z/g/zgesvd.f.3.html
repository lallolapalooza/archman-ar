توثيق الوظيفة/الروتين الفرعي

subroutine zgesvd (character jobu, character jobvt, integer m, integer n, complex*16, dimension( lda, * ) a, integer lda, double precision, dimension( * ) s, complex*16, dimension( ldu, * ) u, integer ldu, complex*16, dimension( ldvt, * ) vt, integer ldvt, complex*16, dimension( * ) work, integer lwork, double precision, dimension( * ) rwork, integer info)

ZGESVD تحسب التحلل القيمي المفرد (SVD) لمصفوفة معقدة M-by-N A، مع حساب اختياري للمتجهات المفردة اليسرى و/أو اليمنى. التحلل القيمي المفرد مكتوب

الغرض:

!>
!> ZGESVD تحسب التحلل القيمي المفرد (SVD) لمصفوفة معقدة
!> M-by-N A، مع حساب اختياري للمتجهات المفردة اليسرى و/أو اليمنى. التحلل القيمي المفرد مكتوب
!>
!>      A = U * SIGMA * conjugate-transpose(V)
!>
!> حيث SIGMA هي مصفوفة M-by-N وهي صفر باستثناء عناصرها القطرية
!> min(m,n)، U هي مصفوفة موحدة M-by-M، و
!> V هي مصفوفة موحدة N-by-N. العناصر القطرية لـ SIGMA
!> هي القيم المفردة لـ A؛ وهي حقيقية وغير سالبة،
!> ويتم إرجاعها بترتيب تنازلي. الأعمدة min(m,n) الأولى من
!> U و V هي المتجهات المفردة اليسرى واليمنى لـ A.
!>
!> لاحظ أن الروتين يعيد V**H، وليس V.
!>

المعلمات

JOBU 
!>          JOBU هو CHARACTER*1
!>          يحدد خيارات حساب كل أو جزء من المصفوفة U:
!>          = 'A':  يتم إرجاع جميع أعمدة M من U في المصفوفة U:
!>          = 'S':  يتم إرجاع الأعمدة min(m,n) الأولى من U (المتجهات المفردة اليسرى)
!>                  في المصفوفة U؛
!>          = 'O':  يتم الكتابة فوق الأعمدة min(m,n) الأولى من U (المتجهات المفردة اليسرى)
!>                  على المصفوفة A؛
!>          = 'N':  لا يتم حساب أي أعمدة لـ U (لا توجد متجهات مفردة يسارية).
!>

JOBVT

!>          JOBVT هو CHARACTER*1
!>          يحدد خيارات حساب كل أو جزء من المصفوفة
!>          V**H:
!>          = 'A':  يتم إرجاع جميع صفوف N من V**H في المصفوفة VT؛
!>          = 'S':  يتم إرجاع الصفوف min(m,n) الأولى من V**H (المتجهات المفردة اليمنى)
!>                  في المصفوفة VT؛
!>          = 'O':  يتم الكتابة فوق الصفوف min(m,n) الأولى من V**H (المتجهات المفردة اليمنى)
!>                  على المصفوفة A؛
!>          = 'N':  لا يتم حساب أي صفوف لـ V**H (لا توجد متجهات مفردة يمينية).
!>
!>          لا يمكن أن يكون JOBVT و JOBU كلاهما 'O'.
!>

M

!>          M هو INTEGER
!>          عدد صفوف المصفوفة المدخلة A.  M >= 0.
!>

N

!>          N هو INTEGER
!>          عدد أعمدة المصفوفة المدخلة A.  N >= 0.
!>

A

!>          A هي مصفوفة COMPLEX*16، البعد (LDA,N)
!>          عند الدخول، المصفوفة M-by-N A.
!>          عند الخروج،
!>          إذا كان JOBU = 'O'، يتم الكتابة فوق A بالأعمدة min(m,n) الأولى من U (المتجهات المفردة اليسرى،
!>                          مخزنة عمودياً)؛
!>          إذا كان JOBVT = 'O'، يتم الكتابة فوق A بالصفوف min(m,n) الأولى من V**H (المتجهات المفردة اليمنى،
!>                          مخزنة صفياً)؛
!>          إذا كان JOBU .ne. 'O' و JOBVT .ne. 'O'، يتم تدمير محتويات A.
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرائد للمصفوفة A.  LDA >= max(1,M).
!>

S

!>          S هي مصفوفة DOUBLE PRECISION، البعد (min(M,N))
!>          القيم المفردة لـ A، مرتبة بحيث S(i) >= S(i+1).
!>

U

!>          U هي مصفوفة COMPLEX*16، البعد (LDU,UCOL)
!>          (LDU,M) إذا كان JOBU = 'A' أو (LDU,min(M,N)) إذا كان JOBU = 'S'.
!>          إذا كان JOBU = 'A'، تحتوي U على المصفوفة الموحدة M-by-M U؛
!>          إذا كان JOBU = 'S'، تحتوي U على الأعمدة min(m,n) الأولى من U
!>          (المتجهات المفردة اليسرى، مخزنة عمودياً)؛
!>          إذا كان JOBU = 'N' أو 'O'، لا يتم الرجوع إلى U.
!>

LDU

!>          LDU هو INTEGER
!>          البعد الرائد للمصفوفة U.  LDU >= 1؛ إذا
!>          كان JOBU = 'S' أو 'A'، فإن LDU >= M.
!>

VT

!>          VT هي مصفوفة COMPLEX*16، البعد (LDVT,N)
!>          إذا كان JOBVT = 'A'، تحتوي VT على المصفوفة الموحدة N-by-N
!>          V**H؛
!>          إذا كان JOBVT = 'S'، تحتوي VT على الصفوف min(m,n) الأولى من
!>          V**H (المتجهات المفردة اليمنى، مخزنة صفياً)؛
!>          إذا كان JOBVT = 'N' أو 'O'، لا يتم الرجوع إلى VT.
!>

LDVT

!>          LDVT هو INTEGER
!>          البعد الرائد للمصفوفة VT.  LDVT >= 1؛ إذا
!>          كان JOBVT = 'A'، فإن LDVT >= N؛ إذا كان JOBVT = 'S'، فإن LDVT >= min(M,N).
!>

WORK

!>          WORK هي مصفوفة COMPLEX*16، البعد (MAX(1,LWORK))
!>          عند الخروج، إذا كان INFO = 0، فإن WORK(1) تعيد LWORK المثلى.
!>

LWORK

!>          LWORK هو INTEGER
!>          بعد المصفوفة WORK.
!>          LWORK >=  MAX(1,2*MIN(M,N)+MAX(M,N)).
!>          لأداء جيد، يجب أن يكون LWORK أكبر بشكل عام.
!>
!>          إذا كان LWORK = -1، فيفترض استعلام مساحة العمل؛ تقوم الروتين فقط
!>          بحساب الحجم الأمثل للمصفوفة WORK، وتعيد
!>          هذه القيمة كالدخول الأول للمصفوفة WORK، ولا يتم إصدار أي رسالة خطأ
!>          متعلقة بـ LWORK بواسطة XERBLA.
!>

RWORK

!>          RWORK هي مصفوفة DOUBLE PRECISION، البعد (5*min(M,N))
!>          عند الخروج، إذا كان INFO > 0، فإن RWORK(1:MIN(M,N)-1) تحتوي على
!>          العناصر القطرية الفائقة غير المتقاربة لمصفوفة ثنائية قطرية وسيطة B
!>          التي قطرها الرئيسي في S (ليس بالضرورة مرتبة).
!>          B تحقق A = U * B * VT، لذا لها نفس القيم المفردة
!>          مثل A، والمتجهات المفردة مرتبطة بواسطة U و VT.
!>

INFO

!>          INFO هو INTEGER
!>          = 0:  انتهاء ناجح.
!>          < 0:  إذا كان INFO = -i، فإن المعلمة i كانت ذات قيمة غير قانونية.
!>          > 0:  إذا لم يتقارب ZBDSQR، يحدد INFO عدد
!>                الأقطار الفائقة للشكل الثنائي القطري الوسيط B
!>                التي لم تتقارب إلى الصفر. انظر وصف RWORK
!>                أعلاه للحصول على التفاصيل.
!>
