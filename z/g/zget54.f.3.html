subroutine zget54 (integer n, complex*16, dimension( lda, * ) a, integer lda, complex*16, dimension( ldb, * ) b, integer ldb, complex*16, dimension( lds, * ) s, integer lds, complex*16, dimension( ldt, * ) t, integer ldt, complex*16, dimension( ldu, * ) u, integer ldu, complex*16, dimension( ldv, * ) v, integer ldv, complex*16, dimension( * ) work, double precision result)

ZGET54

الغرض:

!>
!> ZGET54 يتحقق من تحليل تعميمي من الشكل
!>
!>          A = U*S*V'  و B = U*T* V'
!>
!> حيث ' تعني مرافق تبديل (conjugate transpose) و U و V وحدويتان (unitary).
!>
!> على وجه التحديد،
!>
!>   RESULT = ||( A - U*S*V', B - U*T*V' )|| / (||( A, B )||*n*ulp )
!>

المعلمات

N 
!>          N هو عدد صحيح (INTEGER)
!>          حجم المصفوفة. إذا كان صفرًا، فإن DGET54 لا يفعل شيئًا.
!>          يجب أن يكون على الأقل صفرًا.
!>

A

!>          A هي مصفوفة من النوع COMPLEX*16، أبعادها (LDA, N)
!>          المصفوفة الأصلية (غير المفككة) A.
!>

LDA

!>          LDA هو عدد صحيح (INTEGER)
!>          البعد الرائد للمصفوفة A. يجب أن يكون على الأقل 1
!>          وعلى الأقل N.
!>

B

!>          B هي مصفوفة من النوع COMPLEX*16، أبعادها (LDB, N)
!>          المصفوفة الأصلية (غير المفككة) B.
!>

LDB

!>          LDB هو عدد صحيح (INTEGER)
!>          البعد الرائد للمصفوفة B. يجب أن يكون على الأقل 1
!>          وعلى الأقل N.
!>

S

!>          S هي مصفوفة من النوع COMPLEX*16، أبعادها (LDS, N)
!>          المصفوفة المفككة S.
!>

LDS

!>          LDS هو عدد صحيح (INTEGER)
!>          البعد الرائد للمصفوفة S. يجب أن يكون على الأقل 1
!>          وعلى الأقل N.
!>

T

!>          T هي مصفوفة من النوع COMPLEX*16، أبعادها (LDT, N)
!>          المصفوفة المفككة T.
!>

LDT

!>          LDT هو عدد صحيح (INTEGER)
!>          البعد الرائد للمصفوفة T. يجب أن يكون على الأقل 1
!>          وعلى الأقل N.
!>

U

!>          U هي مصفوفة من النوع COMPLEX*16، أبعادها (LDU, N)
!>          المصفوفة الوحدوية على الجانب الأيسر في
!>          التحليل.
!>

LDU

!>          LDU هو عدد صحيح (INTEGER)
!>          البعد الرائد للمصفوفة U. يجب أن يكون LDU على الأقل N
!>          وعلى الأقل 1.
!>

V

!>          V هي مصفوفة من النوع COMPLEX*16، أبعادها (LDV, N)
!>          المصفوفة الوحدوية على الجانب الأيسر في
!>          التحليل.
!>

LDV

!>          LDV هو عدد صحيح (INTEGER)
!>          البعد الرائد للمصفوفة V. يجب أن يكون LDV على الأقل N
!>          وعلى الأقل 1.
!>

WORK

!>          WORK هي مصفوفة من النوع COMPLEX*16، أبعادها (3*N**2)
!>

RESULT

!>          RESULT هو رقم ذو دقة مزدوجة (DOUBLE PRECISION)
!>          القيمة RESULT، وهي محدودة حاليًا بـ 1/ulp، لتجنب
!>          الفيضان. يتم الإبلاغ عن الأخطاء بواسطة RESULT=10/ulp.
!>
