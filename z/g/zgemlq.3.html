وثائق الوظيفة/الإجراء الفرعي

subroutine zgemlq (character side, character trans, integer m, integer n, integer k, complex*16, dimension( lda, * ) a, integer lda, complex*16, dimension( * ) t, integer tsize, complex*16, dimension( ldc, * ) c, integer ldc, complex*16, dimension( * ) work, integer lwork, integer info)

ZGEMLQ

الغرض:

!>
!>     ZGEMLQ تعيد كتابة المصفوفة الحقيقية العامة M-by-N المسماة C مع
!>
!>                      SIDE = 'L'     SIDE = 'R'
!>      TRANS = 'N':      Q * C          C * Q
!>      TRANS = 'C':      Q**H * C       C * Q**H
!>      حيث Q هي مصفوفة وحدوية معقدة معرفة كناتج
!>      لمُبدِّلات أساسية محظورة تم حسابها بواسطة تحليل LQ قصير وعريض
!>      (ZGELQ)
!>
!>

المعلمات:

SIDE 
!>          SIDE هو CHARACTER*1
!>          = 'L': تطبيق Q أو Q**H من اليسار؛
!>          = 'R': تطبيق Q أو Q**H من اليمين.
!>

TRANS

!>          TRANS هو CHARACTER*1
!>          = 'N':  بدون تحويل، تطبيق Q؛
!>          = 'C':  تحويل مرافق، تطبيق Q**H.
!>

M

!>          M هو INTEGER
!>          عدد صفوف المصفوفة A. M >=0.
!>

N

!>          N هو INTEGER
!>          عدد أعمدة المصفوفة C. N >= 0.
!>

K

!>          K هو INTEGER
!>          عدد المُبدِّلات الأساسية التي يُعرّف حاصلها
!>          المصفوفة Q.
!>          إذا كان SIDE = 'L'، فإن M >= K >= 0؛
!>          إذا كان SIDE = 'R'، فإن N >= K >= 0.
!>
!>

A

!>          A هي مصفوفة COMPLEX*16، حجمها
!>                               (LDA,M) إذا كان SIDE = 'L'،
!>                               (LDA,N) إذا كان SIDE = 'R'
!>          جزء من هيكل البيانات لتمثيل Q كما تم إرجاعه بواسطة ZGELQ.
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرائد للمصفوفة A. LDA >= max(1,K).
!>

T

!>          T هي مصفوفة COMPLEX*16، حجمها (MAX(5,TSIZE)).
!>          جزء من هيكل البيانات لتمثيل Q كما تم إرجاعه بواسطة ZGELQ.
!>

TSIZE

!>          TSIZE هو INTEGER
!>          حجم المصفوفة T. TSIZE >= 5.
!>

C

!>          C هي مصفوفة COMPLEX*16، حجمها (LDC,N)
!>          عند الإدخال، المصفوفة C بالحجم M-by-N.
!>          عند الخروج، يتم إعادة كتابة C بواسطة Q*C أو Q**H*C أو C*Q**H أو C*Q.
!>

LDC

!>          LDC هو INTEGER
!>          البعد الرائد للمصفوفة C. LDC >= max(1,M).
!>

WORK

!>          (مساحة عمل) مصفوفة COMPLEX*16، حجمها (MAX(1,LWORK))
!>          عند الخروج، إذا كان INFO = 0، فإن WORK(1) ترجع الحد الأدنى LWORK.
!>

LWORK

!>          LWORK هو INTEGER
!>          حجم المصفوفة WORK. LWORK >= 1.
!>          إذا كان LWORK = -1، فافترض أنه استعلام مساحة العمل. الوظيفة
!>          تحسب فقط حجم مصفوفة WORK، وترجع هذه القيمة كـ WORK(1)، ولا يتم
!>          إصدار رسالة خطأ تتعلق بـ WORK بواسطة XERBLA.
!>

INFO

!>          INFO هو INTEGER
!>          = 0:  انتهاء ناجح
!>          < 0:  إذا كان INFO = -i، فقد كانت المعلمة i ذات قيمة غير صالحة
!>
