توثيق الوظيفة/الروتين الفرعي

subroutine zgelst (character trans, integer m, integer n, integer nrhs, complex*16, dimension( lda, * ) a, integer lda, complex*16, dimension( ldb, * ) b, integer ldb, complex*16, dimension( * ) work, integer lwork, integer info)

ZGELST يحل الأنظمة الزائدة أو الناقصة للوصول للحل الأمثل لمصفوفات GE باستخدام QR أو LQ factorization مع تمثيل WY مضغوط لـ Q.

الغرض:

!>
!> ZGELST يحل الأنظمة الخطية الحقيقية الزائدة أو الناقصة
!> التي تتضمن مصفوفة A بحجم M-by-N، أو مرافقتها المتبادلة، باستخدام QR
!> أو LQ factorization لـ A مع تمثيل WY مضغوط لـ Q.
!>
!> يُفترض أن A لديها رتبة كاملة، ويتم توفير حماية أساسية فقط
!> ضد المصفوفات الناقصة الرتبة. هذا الروتين الفرعي يكتشف فقط
!> النقصان الدقيق في الرتبة، حيث يكون عنصر قطري في العامل المثلثي
!> لـ A يساوي صفرًا تمامًا.
!>
!> من الممكن أن يكون أحد (أو أكثر) العناصر القطرية للعامل المثلثي
!> لـ A أرقامًا صغيرة جدًا دون أن يصدر هذا الروتين الفرعي
!> خطأ. الحلول المحسوبة لمثل هذه المصفوفات شبه الناقصة الرتبة قد
!> تكون أقل دقة بسبب فقدان الدقة الرقمية.
!>
!> الخيارات التالية متاحة:
!>
!> 1. إذا كانت TRANS = 'N' و m >= n:  ابحث عن حل المربعات الصغرى
!>    لنظام زائد، أي، حل مشكلة المربعات الصغرى
!>                 تصغير || B - A*X ||.
!>
!> 2. إذا كانت TRANS = 'N' و m < n:  ابحث عن الحل الأقل معيارًا
!>    لنظام ناقص A * X = B.
!>
!> 3. إذا كانت TRANS = 'C' و m >= n:  ابحث عن الحل الأقل معيارًا
!>    لنظام ناقص A**T * X = B.
!>
!> 4. إذا كانت TRANS = 'C' و m < n:  ابحث عن حل المربعات الصغرى
!>    لنظام زائد، أي، حل مشكلة المربعات الصغرى
!>                 تصغير || B - A**T * X ||.
!>
!> يمكن التعامل مع العديد من متجهات الطرف الأيمن b ومتجهات الحل x
!> في استدعاء واحد؛ يتم تخزينها كأعمدة في
!> مصفوفة الطرف الأيمن B بحجم M-by-NRHS ومصفوفة الحل
!> X بحجم N-by-NRHS.
!>

المعلمات

TRANS 
!>          TRANS هو CHARACTER*1
!>          = 'N': النظام الخطي يتضمن A؛
!>          = 'C': النظام الخطي يتضمن A**H.
!>

M

!>          M هو INTEGER
!>          عدد صفوف المصفوفة A.  M >= 0.
!>

N

!>          N هو INTEGER
!>          عدد أعمدة المصفوفة A.  N >= 0.
!>

NRHS

!>          NRHS هو INTEGER
!>          عدد الطرف الأيمن، أي، عدد
!>          أعمدة المصفوفات B و X. NRHS >=0.
!>

A

!>          A هو مصفوفة COMPLEX*16، ذات البعد (LDA,N)
!>          عند الدخول، المصفوفة A بحجم M-by-N.
!>          عند الخروج،
!>            إذا كانت M >= N، يتم استبدال A بتفاصيل QR
!>                       factorization كما تم إرجاعها بواسطة ZGEQRT؛
!>            إذا كانت M <  N، يتم استبدال A بتفاصيل LQ
!>                       factorization كما تم إرجاعها بواسطة ZGELQT.
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرئيسي للمصفوفة A.  LDA >= max(1,M).
!>

B

!>          B هو مصفوفة COMPLEX*16، ذات البعد (LDB,NRHS)
!>          عند الدخول، المصفوفة B للمتجهات الطرفية اليمنى، مخزنة
!>          عموديًا؛ B بحجم M-by-NRHS إذا كانت TRANS = 'N'، أو N-by-NRHS
!>          إذا كانت TRANS = 'C'.
!>          عند الخروج، إذا كان INFO = 0، يتم استبدال B بمتجهات
!>          الحل، مخزنة عموديًا:
!>          إذا كانت TRANS = 'N' و m >= n، فإن الصفوف من 1 إلى n في B تحتوي على متجهات
!>          حل المربعات الصغرى؛ مجموع مربعات البواقي لل
!>          الحل في كل عمود يُعطى بمجموع مربعات
!>          المقياس للعناصر من N+1 إلى M في هذا العمود؛
!>          إذا كانت TRANS = 'N' و m < n، فإن الصفوف من 1 إلى N في B تحتوي على
!>          متجهات الحل الأقل معيارًا؛
!>          إذا كانت TRANS = 'C' و m >= n، فإن الصفوف من 1 إلى M في B تحتوي على
!>          متجهات الحل الأقل معيارًا؛
!>          إذا كانت TRANS = 'C' و m < n، فإن الصفوف من 1 إلى M في B تحتوي على
!>          متجهات حل المربعات الصغرى؛ مجموع مربعات البواقي
!>          للحل في كل عمود يُعطى بمجموع
!>          مربعات المقياس للعناصر من M+1 إلى N في هذا العمود.
!>

LDB

!>          LDB هو INTEGER
!>          البعد الرئيسي للمصفوفة B. LDB >= MAX(1,M,N).
!>

WORK

!>          WORK هي مصفوفة COMPLEX*16، ذات البعد (MAX(1,LWORK))
!>          عند الخروج، إذا كان INFO = 0، فإن WORK(1) تُرجع LWORK المثلى.
!>

LWORK

!>          LWORK هو INTEGER
!>          بعد المصفوفة WORK.
!>          LWORK >= max( 1, MN + max( MN, NRHS ) ).
!>          لأداء مثالي،
!>          LWORK >= max( 1, (MN + max( MN, NRHS ))*NB ).
!>          حيث MN = min(M,N) و NB هو حجم الكتلة الأمثل.
!>
!>          إذا كان LWORK = -1، فسيتم افتراض استعلام عن مساحة العمل؛ يقوم الروتين
!>          بحساب حجم مساحة العمل المثلى فقط، ويرجع
!>          هذه القيمة كالإدخال الأول لمصفوفة WORK، ولا يصدر رسالة خطأ
!>          متعلقة بـ LWORK بواسطة XERBLA.
!>

INFO

!>          INFO هو INTEGER
!>          = 0:  انتهاء ناجح
!>          < 0:  إذا كان INFO = -i، فإن المعامل i كان له قيمة غير قانونية
!>          > 0:  إذا كان INFO =  i، فإن العنصر القطري i في
!>                العامل المثلثي لـ A هو صفر تمامًا، بحيث لا تمتلك A
!>                رتبة كاملة؛ لم يتمكن من حساب حل المربعات الصغرى.
!>
