subroutine zgemqrt (character side, character trans, integer m, integer n, integer k, integer nb, complex*16, dimension( ldv, * ) v, integer ldv, complex*16, dimension( ldt, * ) t, integer ldt, complex*16, dimension( ldc, * ) c, integer ldc, complex*16, dimension( * ) work, integer info)

ZGEMQRT

الغرض:

!>
!> ZGEMQRT يقوم بالكتابة فوق المصفوفة المعقدة العامة M-by-N المسماة C بالقيم التالية:
!>
!>                 SIDE = 'L'     SIDE = 'R'
!> TRANS = 'N':      Q C            C Q
!> TRANS = 'C':    Q**H C            C Q**H
!>
!> حيث Q هي مصفوفة متعامدة معقدة معرفة كناتج K من الانعكاسات الابتدائية:
!>
!>       Q = H(1) H(2) . . . H(K) = I - V T V**H
!>
!> تم إنشاؤها باستخدام تمثيل WY المدمج كما تم إرجاعه بواسطة ZGEQRT.
!>
!> Q هي من الرتبة M إذا كانت SIDE = 'L' ومن الرتبة N إذا كانت SIDE = 'R'.
!>

المعلمات

SIDE 
!>          SIDE is CHARACTER*1
!>          = 'L': تطبيق Q أو Q**H من اليسار؛
!>          = 'R': تطبيق Q أو Q**H من اليمين.
!>

TRANS

!>          TRANS is CHARACTER*1
!>          = 'N':  عدم النقل، تطبيق Q؛
!>          = 'C':  النقل المترافق، تطبيق Q**H.
!>

M

!>          M is INTEGER
!>          عدد صفوف المصفوفة C. M >= 0.
!>

N

!>          N is INTEGER
!>          عدد أعمدة المصفوفة C. N >= 0.
!>

K

!>          K is INTEGER
!>          عدد الانعكاسات الابتدائية التي يحدد ناتجها
!>          المصفوفة Q.
!>          إذا كانت SIDE = 'L'، فإن M >= K >= 0؛
!>          إذا كانت SIDE = 'R'، فإن N >= K >= 0.
!>

NB

!>          NB is INTEGER
!>          حجم الكتلة المستخدم لتخزين T.  K >= NB >= 1.
!>          يجب أن تكون هذه نفس قيمة NB المستخدمة لإنشاء T
!>          في ZGEQRT.
!>

V

!>          V is COMPLEX*16 array, dimension (LDV,K)
!>          يجب أن يحتوي العمود i على المتجه الذي يحدد الانعكاس
!>          الابتدائي H(i)، لكل i = 1,2,...,k، كما تم إرجاعه بواسطة
!>          ZGEQRT في الأعمدة K الأولى من مصفوفة الوسيط A الخاصة به.
!>

LDV

!>          LDV is INTEGER
!>          البعد الرائد للمصفوفة V.
!>          إذا كانت SIDE = 'L'، فإن LDA >= max(1,M)؛
!>          إذا كانت SIDE = 'R'، فإن LDA >= max(1,N).
!>

T

!>          T is COMPLEX*16 array, dimension (LDT,K)
!>          العوامل المثلثية العليا للانعكاسات الكتلية
!>          كما تم إرجاعها بواسطة ZGEQRT، مخزنة كمصفوفة NB-by-N.
!>

LDT

!>          LDT is INTEGER
!>          البعد الرائد للمصفوفة T.  LDT >= NB.
!>

C

!>          C is COMPLEX*16 array, dimension (LDC,N)
!>          عند الإدخال، المصفوفة C بحجم M-by-N.
!>          عند الخروج، يتم الكتابة فوق C بواسطة Q C، Q**H C، C Q**H أو C Q.
!>

LDC

!>          LDC is INTEGER
!>          البعد الرائد للمصفوفة C. LDC >= max(1,M).
!>

WORK

!>          WORK is COMPLEX*16 array. The dimension of WORK is
!>           N*NB if SIDE = 'L', or  M*NB if SIDE = 'R'.
!>          (مصفوفة COMPLEX*16. بُعد WORK هو N*NB إذا كانت SIDE = 'L'، أو M*NB إذا كانت SIDE = 'R'.)
!>

INFO

!>          INFO is INTEGER
!>          = 0:  انتهاء ناجح
!>          < 0:  إذا كان INFO = -i، فإن الوسيط i لديه قيمة غير قانونية
!>
