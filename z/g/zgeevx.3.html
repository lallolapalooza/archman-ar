subroutine zgeevx (character balanc, character jobvl, character jobvr, character sense, integer n, complex*16, dimension( lda, * ) a, integer lda, complex*16, dimension( * ) w, complex*16, dimension( ldvl, * ) vl, integer ldvl, complex*16, dimension( ldvr, * ) vr, integer ldvr, integer ilo, integer ihi, double precision, dimension( * ) scale, double precision abnrm, double precision, dimension( * ) rconde, double precision, dimension( * ) rcondv, complex*16, dimension( * ) work, integer lwork, double precision, dimension( * ) rwork, integer info)

ZGEEVX تحسب القيم الذاتية، واختيارياً، المتجهات الذاتية اليسرى و/أو اليمنى لمصفوفات GE

الغرض:

!>
!> ZGEEVX تحسب للمصفوفة المعقدة غير المتناظرة N-by-N A،
!> القيم الذاتية، واختيارياً، المتجهات الذاتية اليسرى و/أو اليمنى.
!>
!> اختيارياً أيضاً، تحسب تحويل موازنة لتحسين
!> ظروف القيم الذاتية والمتجهات الذاتية (ILO، IHI،
!> SCALE، و ABNRM)، الأعداد الشرطية المقلوبة للقيم الذاتية
!> (RCONDE)، والأعداد الشرطية المقلوبة للمتجهات الذاتية اليمنى
!> (RCONDV).
!>
!> المتجه الذاتي الأيمن v(j) لـ A يحقق
!>                  A * v(j) = lambda(j) * v(j)
!> حيث lambda(j) هي القيمة الذاتية له.
!> المتجه الذاتي الأيسر u(j) لـ A يحقق
!>               u(j)**H * A = lambda(j) * u(j)**H
!> حيث u(j)**H تمثل المحول المترافق لـ u(j).
!>
!> المتجهات الذاتية المحسوبة يتم تطبيعها لتحقيق معيار إقليدي
!> يساوي 1 والمكون الأكبر حقيقي.
!>
!> موازنة المصفوفة تعني تبديل الصفوف والأعمدة لجعلها
!> أقرب إلى شبه مثلثية، وتطبيق تحويل تشابه قطري
!> D * A * D**(-1)، حيث D مصفوفة قطرية
!> لجعل صفوف وأعمدة المصفوفة متقاربة في المعيار والأعداد الشرطية
!> للقيم الذاتية والمتجهات الذاتية أصغر. المحسوبة
!> الأعداد الشرطية المقلوبة تتوافق مع المصفوفة الموزونة.
!> تبديل الصفوف والأعمدة لن يغير الأعداد الشرطية
!> (في الحساب الدقيق) ولكن التحجيم القطري سيفعل. لمزيد من
!> الشرح حول الموازنة، انظر القسم 4.10.2 من دليل مستخدمي LAPACK.
!>

المعلمات

BALANC 
!>          BALANC هو CHARACTER*1
!>          يشير إلى كيفية تحجيم المصفوفة المدخلة قطرياً
!>          و/أو تبديلها لتحسين ظروف القيم الذاتية.
!>          = 'N': لا تقم بالتحجيم القطري أو التبديل؛
!>          = 'P': قم بإجراء تبديلات لجعل المصفوفة أقرب إلى
!>                 شبه مثلثية. لا تقم بالتحجيم القطري؛
!>          = 'S': قم بتحجيم المصفوفة قطرياً، أي استبدال A بـ
!>                 D*A*D**(-1)، حيث D مصفوفة قطرية تم اختيارها
!>                 لجعل صفوف وأعمدة A متساوية في المعيار.
!>                 لا تقم بالتبديل؛
!>          = 'B': قم بكلا التحجيم القطري وتبديل A.
!>
!>          الأعداد الشرطية المقلوبة المحسوبة ستكون للمصفوفة
!>          بعد الموازنة و/أو التبديل. التبديل لا يغير
!>          الأعداد الشرطية (في الحساب الدقيق)، ولكن الموازنة تفعل.
!>

JOBVL

!>          JOBVL هو CHARACTER*1
!>          = 'N': لا يتم حساب المتجهات الذاتية اليسرى لـ A؛
!>          = 'V': يتم حساب المتجهات الذاتية اليسرى لـ A.
!>          إذا كان SENSE = 'E' أو 'B'، يجب أن يكون JOBVL = 'V'.
!>

JOBVR

!>          JOBVR هو CHARACTER*1
!>          = 'N': لا يتم حساب المتجهات الذاتية اليمنى لـ A؛
!>          = 'V': يتم حساب المتجهات الذاتية اليمنى لـ A.
!>          إذا كان SENSE = 'E' أو 'B'، يجب أن يكون JOBVR = 'V'.
!>

SENSE

!>          SENSE هو CHARACTER*1
!>          يحدد أي الأعداد الشرطية المقلوبة يتم حسابها.
!>          = 'N': لا يتم حساب أي شيء؛
!>          = 'E': محسوبة للقيم الذاتية فقط؛
!>          = 'V': محسوبة للمتجهات الذاتية اليمنى فقط؛
!>          = 'B': محسوبة للقيم الذاتية والمتجهات الذاتية اليمنى.
!>
!>          إذا كان SENSE = 'E' أو 'B'، يجب أن يتم حساب كل من المتجهات
!>          الذاتية اليسرى واليمنى أيضاً (JOBVL = 'V' و JOBVR = 'V').
!>

N

!>          N هو INTEGER
!>          ترتيب المصفوفة A. N >= 0.
!>

A

!>          A هو COMPLEX*16 array, dimension (LDA,N)
!>          عند الدخول، مصفوفة A بحجم N-by-N.
!>          عند الخروج، تم استبدال A. إذا كان JOBVL = 'V' أو
!>          JOBVR = 'V'، تحتوي A على شكل شور (Schur form)
!>          للنسخة الموزونة من المصفوفة A.
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرئيسي للمصفوفة A.  LDA >= max(1,N).
!>

W

!>          W هو COMPLEX*16 array, dimension (N)
!>          تحتوي W على القيم الذاتية المحسوبة.
!>

VL

!>          VL هو COMPLEX*16 array, dimension (LDVL,N)
!>          إذا كان JOBVL = 'V'، يتم تخزين المتجهات الذاتية اليسرى u(j)
!>          واحدة تلو الأخرى في أعمدة VL، بنفس ترتيب
!>          قيمها الذاتية.
!>          إذا كان JOBVL = 'N'، لا يتم استخدام VL.
!>          u(j) = VL(:,j)، العمود j من VL.
!>

LDVL

!>          LDVL هو INTEGER
!>          البعد الرئيسي للمصفوفة VL.  LDVL >= 1؛ إذا
!>          كان JOBVL = 'V'، فإن LDVL >= N.
!>

VR

!>          VR هو COMPLEX*16 array, dimension (LDVR,N)
!>          إذا كان JOBVR = 'V'، يتم تخزين المتجهات الذاتية اليمنى v(j)
!>          واحدة تلو الأخرى في أعمدة VR، بنفس ترتيب
!>          قيمها الذاتية.
!>          إذا كان JOBVR = 'N'، لا يتم استخدام VR.
!>          v(j) = VR(:,j)، العمود j من VR.
!>

LDVR

!>          LDVR هو INTEGER
!>          البعد الرئيسي للمصفوفة VR.  LDVR >= 1؛ إذا
!>          كان JOBVR = 'V'، فإن LDVR >= N.
!>

ILO

!>          ILO هو INTEGER
!>

IHI

!>          IHI هو INTEGER
!>          ILO و IHI هي قيم صحيحة تم تحديدها عند موازنة A.
!>          المصفوفة الموزونة A(i,j) = 0 إذا كان I > J و
!>          J = 1,...,ILO-1 أو I = IHI+1,...,N.
!>

SCALE

!>          SCALE هو DOUBLE PRECISION array, dimension (N)
!>          تفاصيل التبديلات وعوامل التحجيم المطبقة
!>          عند موازنة A. إذا كان P(j) هو فهرس الصف والعمود
!>          المتبادلين مع الصف والعمود j، وعامل التحجيم
!>          D(j) المطبق على الصف والعمود j، فإن
!>          SCALE(J) = P(J)،    للـ J = 1,...,ILO-1
!>                   = D(J)،    للـ J = ILO,...,IHI
!>                   = P(J)     للـ J = IHI+1,...,N.
!>          الترتيب الذي تتم به التبادلات هو N إلى IHI+1،
!>          ثم 1 إلى ILO-1.
!>

ABNRM

!>          ABNRM هو DOUBLE PRECISION
!>          المعيار الواحد للمصفوفة الموزونة (القيمة القصوى
!>          لمجموع القيم المطلقة لعناصر أي عمود).
!>

RCONDE

!>          RCONDE هو DOUBLE PRECISION array, dimension (N)
!>          RCONDE(j) هو العدد الشرطي المقلوب للقيمة الذاتية j.
!>

NAME

zgeevx - computes the eigenvalues and left and right eigenvectors of a general square matrix.

SYNOPSIS

SUBROUTINE ZGEEVX(
   JOBVL, JOBVR, SENSE, N, A, LDA, VL, LDVL, VR, LDVR, ILO, IHI,
   SCALE, ABNORM, RCONDV, WORK, LWORK, RWORK, INFO
)
INTEGER ILO, IHI, LDA, LDVL, LDVR, LWORK, N
INTEGER INFO
COMPLEX*16 A( LDA, * ), VL( LDVL, * ), VR( LDVR, * )
DOUBLE PRECISION SCALE, ABNORM
DOUBLE PRECISION RCONDV( * )
COMPLEX*16 WORK( * )
DOUBLE PRECISION RWORK( * )
CHARACTER JOBVL, JOBVR, SENSE

Function/Subroutine Documentation

This subroutine computes the eigenvalues and, optionally, the left and right eigenvectors of a general complex matrix A. The condition numbers of the eigenvectors are also computed, and optionally, the condition numbers of the eigenvectors.

Arguments

[in] JOBVL
CHARACTER*1. Must be 'V' or 'N'. If JOBVL = 'V', compute the left eigenvectors of A. If JOBVL = 'N', do not compute the left eigenvectors.
[in] JOBVR
CHARACTER*1. Must be 'V' or 'N'. If JOBVR = 'V', compute the right eigenvectors of A. If JOBVR = 'N', do not compute the right eigenvectors.
[in] SENSE
CHARACTER*1. Determines which condition numbers are computed.
[in] N
INTEGER. The order of the matrix A. N >= 0.
[in,out] A
COMPLEX*16 array, dimension (LDA, N). On entry, the n-by-n matrix A. On exit, the upper triangle of A is overwritten by information about the Schur factorization and diagonal elements of the unitary matrix Q, and the lower triangle is overwritten by information about the Schur factorization.
[in] LDA
INTEGER. The leading dimension of the array A. LDA >= max(1, N).
[out] VL
COMPLEX*16 array, dimension (LDVL, N). If JOBVL = 'V', VL contains the left eigenvectors of A. If JOBVL = 'N', VL is not used.
[in] LDVL
INTEGER. The leading dimension of the array VL. If JOBVL = 'V', LDVL >= max(1, N). Otherwise, LDVL is not used.
[out] VR
COMPLEX*16 array, dimension (LDVR, N). If JOBVR = 'V', VR contains the right eigenvectors of A. If JOBVR = 'N', VR is not used.
[in] LDVR
INTEGER. The leading dimension of the array VR. If JOBVR = 'V', LDVR >= max(1, N). Otherwise, LDVR is not used.
[out] ILO
INTEGER.
[out] IHI
INTEGER.
[out] SCALE
DOUBLE PRECISION array, dimension (N).
[out] ABNORM
DOUBLE PRECISION.
[out] RCONDV
DOUBLE PRECISION array, dimension (N).
[out] WORK
COMPLEX*16 array, dimension (MAX(1,LWORK)).
[out] LWORK
INTEGER.
[out] RWORK
DOUBLE PRECISION array, dimension (2*N).
[out] INFO
INTEGER.

Option

SENSE

!>          SENSE is CHARACTER*1
!>          SENSE specifies which of the error bounds are to be computed.
!>          = 'N': none.
!>          = 'E': only the reciprocal condition number of the eigenvector
!>                 submatrix.
!>          = 'V': only the reciprocal condition numbers of the right
!>                 eigenvectors.
!>          = 'B': both the reciprocal condition number of the eigenvector
!>                 submatrix and the right eigenvectors.
!

Return values

RCONDV

!>          RCONDV is DOUBLE PRECISION array, dimension (N)
!>          RCONDV(j) is the reciprocal condition number of the j-th
!>          right eigenvector.
!>

WORK

!>          WORK is COMPLEX*16 array, dimension (MAX(1,LWORK))
!>          On exit, if INFO = 0, WORK(1) returns the optimal LWORK.
!>

LWORK

!>          LWORK is INTEGER
!>          The dimension of the array WORK.  If SENSE = 'N' or 'E',
!>          LWORK >= max(1,2*N), and if SENSE = 'V' or 'B',
!>          LWORK >= N*N+2*N.
!>          For good performance, LWORK must generally be larger.
!>
!>          If LWORK = -1, then a workspace query is assumed; the routine
!>          only calculates the optimal size of the WORK array, returns
!>          this value as the first entry of the WORK array, and no error
!>          message related to LWORK is issued by XERBLA.
!>

RWORK

!>          RWORK is DOUBLE PRECISION array, dimension (2*N)
!>

INFO

!>          INFO is INTEGER
!>          = 0:  successful exit
!>          < 0:  if INFO = -i, the i-th argument had an illegal value.
!>          > 0:  if INFO = i, the QR algorithm failed to compute all the
!>                eigenvalues, and no eigenvectors or condition numbers
!>                have been computed; elements 1:ILO-1 and i+1:N of W
!>                contain eigenvalues which have converged.
!>
