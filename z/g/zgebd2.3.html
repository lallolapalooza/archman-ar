وثائق الوظيفة/الروتين الفرعي

subroutine zgebd2 (integer m, integer n, complex*16, dimension( lda, * ) a, integer lda, double precision, dimension( * ) d, double precision, dimension( * ) e, complex*16, dimension( * ) tauq, complex*16, dimension( * ) taup, complex*16, dimension( * ) work, integer info)

ZGEBD2 تقلل مصفوفة عامة إلى شكل ثنائي قطري باستخدام خوارزمية غير محظورة.

الغرض:

!>
!> ZGEBD2 تقلل مصفوفة عامة من النوع المعقد m × n، A، إلى شكل B ثنائي قطري حقيقي علوي أو سفلي
!> باستخدام تحويل وحدوي: Q**H * A * P = B.
!>
!> إذا كانت m >= n، فإن B تكون ثنائية قطرية علوية؛ إذا كانت m < n، فإن B تكون ثنائية قطرية سفلية.
!>

المعلمات

M 
!>          M هو عدد صحيح (INTEGER)
!>          عدد الصفوف في المصفوفة A. M >= 0.
!>

N

!>          N هو عدد صحيح (INTEGER)
!>          عدد الأعمدة في المصفوفة A. N >= 0.
!>

A

!>          A هي مصفوفة من النوع COMPLEX*16، ذات أبعاد (LDA,N)
!>          عند الإدخال، المصفوفة العامة m × n المراد تقليلها.
!>          عند الخروج:
!>          إذا كانت m >= n، يتم استبدال عناصر القطر الرئيسي والقطر الفوقي الأول بالمصفوفة الثنائية القطرية B؛
!>          العناصر الموجودة أسفل القطر الرئيسي، مع مصفوفة TAUQ، تمثل المصفوفة الوحدوية Q كناتج انعكاسات أولية،
!>          والعناصر الموجودة فوق القطر الفوقي الأول، مع مصفوفة TAUP، تمثل المصفوفة الوحدوية P كناتج انعكاسات أولية.
!>          إذا كانت m < n، يتم استبدال عناصر القطر الرئيسي والقطر السفلي الأول بالمصفوفة الثنائية القطرية B؛
!>          العناصر الموجودة أسفل القطر السفلي الأول، مع مصفوفة TAUQ، تمثل المصفوفة الوحدوية Q كناتج انعكاسات أولية،
!>          والعناصر الموجودة فوق القطر الرئيسي، مع مصفوفة TAUP، تمثل المصفوفة الوحدوية P كناتج انعكاسات أولية.
!>          انظر التفاصيل الإضافية.
!>

LDA

!>          LDA هو عدد صحيح (INTEGER)
!>          البعد الرائد للمصفوفة A. LDA >= max(1,M).
!>

D

!>          D هي مصفوفة من النوع DOUBLE PRECISION، ذات أبعاد (min(M,N))
!>          عناصر القطر الرئيسي للمصفوفة الثنائية القطرية B:
!>          D(i) = A(i,i).
!>

E

!>          E هي مصفوفة من النوع DOUBLE PRECISION، ذات أبعاد (min(M,N)-1)
!>          عناصر خارج القطر الرئيسي للمصفوفة الثنائية القطرية B:
!>          إذا كانت m >= n، فإن E(i) = A(i,i+1) لـ i = 1,2,...,n-1؛
!>          إذا كانت m < n، فإن E(i) = A(i+1,i) لـ i = 1,2,...,m-1.
!>

TAUQ

!>          TAUQ هي مصفوفة من النوع COMPLEX*16، ذات أبعاد (min(M,N))
!>          العوامل العددية للانعكاسات الأولية التي تمثل المصفوفة الوحدوية Q. انظر التفاصيل الإضافية.
!>

TAUP

!>          TAUP هي مصفوفة من النوع COMPLEX*16، ذات أبعاد (min(M,N))
!>          العوامل العددية للانعكاسات الأولية التي تمثل المصفوفة الوحدوية P. انظر التفاصيل الإضافية.
!>

WORK

!>          WORK هي مصفوفة من النوع COMPLEX*16، ذات أبعاد (max(M,N))
!>

INFO

!>          INFO هو عدد صحيح (INTEGER)
!>          = 0: اكتمال ناجح
!>          < 0: إذا كان INFO = -i، فإن المعامل i كان لديه قيمة غير قانونية.
!>

تفاصيل إضافية:

!>
!>  المصفوفتان Q و P ممثلتان كناتج لانعكاسات أولية:
!>
!>  إذا كانت m >= n،
!>
!>     Q = H(1) H(2) . . . H(n)  و  P = G(1) G(2) . . . G(n-1)
!>
!>  كل من H(i) و G(i) له الشكل التالي:
!>
!>     H(i) = I - tauq * v * v**H  و G(i) = I - taup * u * u**H
!>
!>  حيث tauq و taup عوامل عددية معقدة، v و u متجهات معقدة؛ v(1:i-1) = 0، v(i) = 1،
!>  ويتم تخزين v(i+1:m) عند الخروج في A(i+1:m,i)؛ u(1:i) = 0، u(i+1) = 1،
!>  ويتم تخزين u(i+2:n) عند الخروج في A(i,i+2:n)؛ tauq مخزن في TAUQ(i) و taup في TAUP(i).
!>
!>  إذا كانت m < n،
!>
!>     Q = H(1) H(2) . . . H(m-1)  و  P = G(1) G(2) . . . G(m)
!>
!>  كل من H(i) و G(i) له الشكل التالي:
!>
!>     H(i) = I - tauq * v * v**H  و G(i) = I - taup * u * u**H
!>
!>  حيث tauq و taup عوامل عددية معقدة، v و u متجهات معقدة؛
!>  v(1:i) = 0، v(i+1) = 1، ويتم تخزين v(i+2:m) عند الخروج في A(i+2:m,i)؛
!>  u(1:i-1) = 0، u(i) = 1، ويتم تخزين u(i+1:n) عند الخروج في A(i,i+1:n)؛
!>  tauq مخزن في TAUQ(i) و taup في TAUP(i).
!>
!>  محتويات A عند الخروج موضحة بالأمثلة التالية:
!>
!>  m = 6 و n = 5 (m >= n):          m = 5 و n = 6 (m < n):
!>
!>    (  d   e   u1  u1  u1 )           (  d   u1  u1  u1  u1  u1 )
!>    (  v1  d   e   u2  u2 )           (  e   d   u2  u2  u2  u2 )
!>    (  v1  v2  d   e   u3 )           (  v1  e   d   u3  u3  u3 )
!>    (  v1  v2  v3  d   e  )           (  v1  v2  e   d   u4  u4 )
!>    (  v1  v2  v3  v4  d  )           (  v1  v2  v3  e   d   u5 )
!>    (  v1  v2  v3  v4  v5 )
!>
!>  حيث d و e تمثل عناصر القطر الرئيسي والعناصر خارج القطر الرئيسي لـ B،
!>  vi يمثل عنصرًا من المتجه الذي يحدد H(i)، و ui عنصرًا من المتجه الذي يحدد G(i).
!>
