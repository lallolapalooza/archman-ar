subroutine zgelss (integer m, integer n, integer nrhs, complex*16, dimension( lda, * ) a, integer lda, complex*16, dimension( ldb, * ) b, integer ldb, double precision, dimension( * ) s, double precision rcond, integer rank, complex*16, dimension( * ) work, integer lwork, double precision, dimension( * ) rwork, integer info)

ZGELSS تحل أنظمة مفرطة التحديد أو ناقصة التحديد للمصفوفات GE

الغرض:

!>
!> ZGELSS تحسب الحد الأدنى لـ norm solution لمشكلة المربعات الصغرى الخطية المعقدة:
!>
!> تقليل 2-norm(| b - A*x |).
!>
!> باستخدام التحلل القيمي المفرد (SVD) لـ A. A هي مصفوفة M-by-N
!> قد تكون ناقصة الرتبة.
!>
!> يمكن معالجة عدة متجهات طرف أيمن b ومتجهات حل x في استدعاء واحد؛
!> يتم تخزينها كأعمدة للمصفوفة B ذات الطرف الأيمن NRHS-by-M والمصفوفة X
!> ذات الحل N-by-NRHS.
!>
!> يتم تحديد الرتبة الفعالة لـ A عن طريق معاملة القيم المفردة
!> التي تقل عن RCOND مضروبة في أكبر قيمة مفردة على أنها صفر.
!>

المعاملات

M 
!>          M هو عدد صحيح (INTEGER)
!>          عدد صفوف المصفوفة A. M >= 0.
!>

N

!>          N هو عدد صحيح (INTEGER)
!>          عدد أعمدة المصفوفة A. N >= 0.
!>

NRHS

!>          NRHS هو عدد صحيح (INTEGER)
!>          عدد الأطراف اليمنى، أي عدد أعمدة المصفوفتين B و X. NRHS >= 0.
!>

A

!>          A هي مصفوفة COMPLEX*16، البعد (LDA,N)
!>          عند الإدخال، مصفوفة A ذات البعد M-by-N.
!>          عند الخروج، يتم الكتابة فوق أول min(m,n) صفوف من A
!>          متجهات القيم المفردة اليمنى الخاصة بها، مخزنة صفا بصف.
!>

LDA

!>          LDA هو عدد صحيح (INTEGER)
!>          البعد الرائد للمصفوفة A. LDA >= max(1,M).
!>

B

!>          B هي مصفوفة COMPLEX*16، البعد (LDB,NRHS)
!>          عند الإدخال، مصفوفة الطرف الأيمن B ذات البعد M-by-NRHS.
!>          عند الخروج، يتم استبدال B بمصفوفة الحل X ذات البعد N-by-NRHS.
!>          إذا كانت m >= n وكانت RANK = n، فإن مجموع مربعات البواقي
!>          للحل في العمود i معطى بمجموع مربعات قيم العناصر n+1:m في ذلك العمود.
!>

LDB

!>          LDB هو عدد صحيح (INTEGER)
!>          البعد الرائد للمصفوفة B. LDB >= max(1,M,N).
!>

S

!>          S هي مصفوفة DOUBLE PRECISION، البعد (min(M,N))
!>          القيم المفردة لـ A بترتيب تنازلي.
!>          نسبة الأرقام الشرطية لـ A في 2-norm = S(1)/S(min(m,n)).
!>

RCOND

!>          RCOND هي DOUBLE PRECISION
!>          يستخدم RCOND لتحديد الرتبة الفعالة لـ A.
!>          تُعامل القيم المفردة S(i) <= RCOND*S(1) على أنها صفر.
!>          إذا كان RCOND < 0، يتم استخدام دقة الآلة بدلاً من ذلك.
!>

RANK

!>          RANK هو عدد صحيح (INTEGER)
!>          الرتبة الفعالة لـ A، أي عدد القيم المفردة
!>          التي تكون أكبر من RCOND*S(1).
!>

WORK

!>          WORK هي مصفوفة COMPLEX*16، البعد (MAX(1,LWORK))
!>          عند الخروج، إذا كان INFO = 0، فإن WORK(1) يعيد LWORK المثلى.
!>

LWORK

!>          LWORK هو عدد صحيح (INTEGER)
!>          بعد المصفوفة WORK. LWORK >= 1، وأيضًا:
!>          LWORK >=  2*min(M,N) + max(M,N,NRHS)
!>          للحصول على أداء جيد، يجب أن تكون LWORK أكبر بشكل عام.
!>
!>          إذا كان LWORK = -1، فيُفترض استعلام عن مساحة العمل؛ تقوم الروتين
!>          فقط بحساب الحجم الأمثل لمصفوفة WORK، وتعيد
!>          هذه القيمة كمدخل أول لمصفوفة WORK، ولا يصدر XERBLA أي رسالة خطأ
!>          متعلقة بـ LWORK.
!>

RWORK

!>          RWORK هي مصفوفة DOUBLE PRECISION، البعد (5*min(M,N))
!>

INFO

!>          INFO هو عدد صحيح (INTEGER)
!>          = 0:  الخروج بنجاح
!>          < 0:  إذا كان INFO = -i، فإن المعامل i كان له قيمة غير قانونية.
!>          > 0:  فشلت خوارزمية حساب SVD في التقارب؛
!>                إذا كان INFO = i، فإن i عنصرًا غير قطري في شكل ثنائي قطري وسيط
!>                لم يتقارب إلى الصفر.
!>
