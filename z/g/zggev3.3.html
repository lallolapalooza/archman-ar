subroutine zggev3 (character jobvl, character jobvr, integer n, complex*16, dimension( lda, * ) a, integer lda, complex*16, dimension( ldb, * ) b, integer ldb, complex*16, dimension( * ) alpha, complex*16, dimension( * ) beta, complex*16, dimension( ldvl, * ) vl, integer ldvl, complex*16, dimension( ldvr, * ) vr, integer ldvr, complex*16, dimension( * ) work, integer lwork, double precision, dimension( * ) rwork, integer info)

ZGGEV3 تحسب القيم الذاتية، واختياريًا، المتجهات الذاتية اليسرى و/أو اليمنى لمصفوفات GE (خوارزمية مجمعة)

الغرض:

!>
!> ZGGEV3 تحسب، لزوج من المصفوفات المعقدة غير المتماثلة N-by-N
!> (A,B)، القيم الذاتية المعممة، واختياريًا، المتجهات الذاتية المعممة اليسرى و/أو اليمنى.
!>
!> القيمة الذاتية المعممة لزوج من المصفوفات (A,B) هي عدد قياسي
!> lambda أو نسبة alpha/beta = lambda، بحيث A - lambda*B تكون
!> منفردة. عادة ما يتم تمثيلها بالزوج (alpha,beta)، حيث
!> يوجد تفسير معقول لـ beta=0، وحتى لكلاهما.
!>
!> المتجه الذاتي المعمم الأيمن v(j) المقابل للقيمة الذاتية المعممة
!> lambda(j) لـ (A,B) يحقق
!>
!>              A * v(j) = lambda(j) * B * v(j).
!>
!> المتجه الذاتي المعمم الأيسر u(j) المقابل للقيمة الذاتية المعممة
!> lambda(j) لـ (A,B) يحقق
!>
!>              u(j)**H * A = lambda(j) * u(j)**H * B
!>
!> حيث u(j)**H هو المرافق المتحول لـ u(j).
!>

المعاملات

JOBVL 
!>          JOBVL هو CHARACTER*1
!>          = 'N':  لا تحسب المتجهات الذاتية اليسرى المعممة؛
!>          = 'V':  احسب المتجهات الذاتية اليسرى المعممة.
!>

JOBVR

!>          JOBVR هو CHARACTER*1
!>          = 'N':  لا تحسب المتجهات الذاتية اليمنى المعممة؛
!>          = 'V':  احسب المتجهات الذاتية اليمنى المعممة.
!>

N

!>          N هو INTEGER
!>          رتبة المصفوفات A, B, VL, و VR.  N >= 0.
!>

A

!>          A هو مصفوفة COMPLEX*16، ذات بعد (LDA, N)
!>          عند الإدخال، المصفوفة A في الزوج (A,B).
!>          عند الخروج، تم استبدال A.
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرئيسي لـ A.  LDA >= max(1,N).
!>

B

!>          B هو مصفوفة COMPLEX*16، ذات بعد (LDB, N)
!>          عند الإدخال، المصفوفة B في الزوج (A,B).
!>          عند الخروج، تم استبدال B.
!>

LDB

!>          LDB هو INTEGER
!>          البعد الرئيسي لـ B.  LDB >= max(1,N).
!>

ALPHA

!>          ALPHA هي مصفوفة COMPLEX*16، ذات بعد (N)
!>

BETA

!>          BETA هي مصفوفة COMPLEX*16، ذات بعد (N)
!>          عند الخروج، ستكون ALPHA(j)/BETA(j)، j=1,...,N، هي
!>          القيم الذاتية المعممة.
!>
!>          ملاحظة: قد تتجاوز أو تنقص نسب ALPHA(j)/BETA(j) بسهولة.
!>          قد تكون BETA(j) حتى صفرًا.  لذلك، يجب على المستخدم
!>          تجنب حساب النسبة alpha/beta بشكل ساذج.
!>          ومع ذلك، سيكون ALPHA دائمًا أقل وغالبًا
!>          قابل للمقارنة مع norm(A) في المقدار، و BETA دائمًا أقل
!>          وغالبًا قابل للمقارنة مع norm(B).
!>

VL

!>          VL هي مصفوفة COMPLEX*16، ذات بعد (LDVL,N)
!>          إذا كان JOBVL = 'V'، يتم تخزين المتجهات الذاتية اليسرى المعممة u(j)
!>          واحدًا تلو الآخر في أعمدة VL، بنفس
!>          الترتيب كما هو الحال في قيمها الذاتية.
!>          يتم قياس كل متجه ذاتي بحيث يكون المكون الأكبر
!>          abs(الجزء الحقيقي) + abs(الجزء التخيلي) = 1.
!>          غير مستخدم إذا كان JOBVL = 'N'.
!>

LDVL

!>          LDVL هو INTEGER
!>          البعد الرئيسي للمصفوفة VL. LDVL >= 1، وإذا
!>          كان JOBVL = 'V'، فإن LDVL >= N.
!>

VR

!>          VR هي مصفوفة COMPLEX*16، ذات بعد (LDVR,N)
!>          إذا كان JOBVR = 'V'، يتم تخزين المتجهات الذاتية اليمنى المعممة v(j)
!>          واحدًا تلو الآخر في أعمدة VR، بنفس
!>          الترتيب كما هو الحال في قيمها الذاتية.
!>          يتم قياس كل متجه ذاتي بحيث يكون المكون الأكبر
!>          abs(الجزء الحقيقي) + abs(الجزء التخيلي) = 1.
!>          غير مستخدم إذا كان JOBVR = 'N'.
!>

LDVR

!>          LDVR هو INTEGER
!>          البعد الرئيسي للمصفوفة VR. LDVR >= 1، وإذا
!>          كان JOBVR = 'V'، فإن LDVR >= N.
!>

WORK

!>          WORK هي مصفوفة COMPLEX*16، ذات بعد (MAX(1,LWORK))
!>          عند الخروج، إذا كان INFO = 0، فإن WORK(1) يعيد LWORK الأمثل.
!>

LWORK

!>          LWORK هو INTEGER
!>          بعد المصفوفة WORK. LWORK >= MAX(1,2*N).
!>          لأداء جيد، يجب أن يكون LWORK أكبر بشكل عام.
!>
!>          إذا كان LWORK = -1، فسيتم افتراض استعلام عن مساحة العمل؛ تقوم الروتين
!>          بحساب حجم مساحة العمل الأمثل فقط، وإعادة
!>          هذه القيمة كإدخال أول في مصفوفة WORK، ولا يتم إصدار أي رسالة خطأ
!>          متعلقة بـ LWORK بواسطة XERBLA.
!>

RWORK

!>          RWORK هي مصفوفة DOUBLE PRECISION، ذات بعد (8*N)
!>

INFO

!>          INFO هو INTEGER
!>          = 0:  انتهاء ناجح
!>          < 0:  إذا كان INFO = -i، فإن المعامل i كان لديه قيمة غير قانونية.
!>          =1,...,N:
!>                فشلت تكرارات QZ. لم يتم حساب أي متجهات ذاتية،
!>                ولكن يجب أن تكون ALPHA(j) و BETA(j)
!>                صحيحة لـ j=INFO+1,...,N.
!>          > N:  =N+1: فشلت تكرارات QZ في ZHGEQZ،
!>                =N+2: عودة خطأ من ZTGEVC.
!>
