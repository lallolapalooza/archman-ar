توثيق الوظيفة/الروتين الفرعي

subroutine zgeesx (character jobvs, character sort, external select, character sense, integer n, complex*16, dimension( lda, * ) a, integer lda, integer sdim, complex*16, dimension( * ) w, complex*16, dimension( ldvs, * ) vs, integer ldvs, double precision rconde, double precision rcondv, complex*16, dimension( * ) work, integer lwork, double precision, dimension( * ) rwork, logical, dimension( * ) bwork, integer info)

ZGEESX تحسب القيم الذاتية، وشكل شتور، واختياريًا، مصفوفة متجهات شتور لمصفوفات GE

الغرض:

!>
!> ZGEESX تحسب مصفوفة معقدة غير متناظرة A بحجم N×N، والقيم الذاتية، وشكل شتور T، واختياريًا، مصفوفة متجهات شتور Z. هذا يعطي تحليل شتور A = Z*T*(Z**H).
!>
!> اختياريًا، تقوم أيضًا بترتيب القيم الذاتية على القطر الرئيسي لشكل شتور بحيث تكون القيم الذاتية المحددة في الجزء العلوي الأيسر؛
!> تحسب مقلوب رقم شرطي لمتوسط القيم الذاتية المحددة (RCONDE)؛ وتحسب مقلوب رقم شرطي للمتجه الفرعي الأيمن الموافق للقيم الذاتية المحددة (RCONDV).
!> تشكل الأعمدة الرائدة لـ Z أساسًا متعامدًا لهذا المتجه الفرعي.
!>
!> لمزيد من الشرح لمقلوب الأرقام الشرطية RCONDE و RCONDV، انظر القسم 4.10 من دليل مستخدمي LAPACK (حيث تسمى هذه الكميات s و sep على التوالي).
!>
!> المصفوفة المعقدة تكون في شكل شتور إذا كانت علوية مثلثة.
!>

المعلمات

JOBVS 
!>          JOBVS هو CHARACTER*1
!>          = 'N': متجهات شتور غير محسوبة؛
!>          = 'V': متجهات شتور محسوبة.
!>

SORT

!>          SORT هو CHARACTER*1
!>          يحدد ما إذا كان سيتم ترتيب القيم الذاتية على القطر الرئيسي لشكل شتور أم لا.
!>          = 'N': القيم الذاتية غير مرتبة؛
!>          = 'S': القيم الذاتية مرتبة (انظر SELECT).
!>

SELECT

!>          SELECT هي وظيفة LOGICAL تأخذ وسيطًا واحدًا من نوع COMPLEX*16
!>          يجب تعريف SELECT كـ EXTERNAL في الروتين الفرعي المستدعي.
!>          إذا كان SORT = 'S'، يتم استخدام SELECT لتحديد القيم الذاتية لترتيبها
!>          في الجزء العلوي الأيسر من شكل شتور.
!>          إذا كان SORT = 'N'، لا يتم الرجوع إلى SELECT.
!>          يتم اختيار القيمة الذاتية W(j) إذا كان SELECT(W(j)) صحيحًا.
!>

SENSE

!>          SENSE هو CHARACTER*1
!>          يحدد مقلوب الأرقام الشرطية التي سيتم حسابها.
!>          = 'N': لا شيء محسوب؛
!>          = 'E': محسوب لمتوسط القيم الذاتية المحددة فقط؛
!>          = 'V': محسوب للمتجه الفرعي الأيمن المحدد فقط؛
!>          = 'B': محسوب لكليهما.
!>          إذا كان SENSE = 'E'، 'V' أو 'B'، يجب أن يكون SORT = 'S'.
!>

N

!>          N هو INTEGER
!>          رتبة المصفوفة A. N >= 0.
!>

A

!>          A هي مصفوفة COMPLEX*16، ذات بعد (LDA, N)
!>          عند الدخول، المصفوفة A بحجم N×N.
!>          عند الخروج، يتم استبدال A بشكل شتور T الخاص بها.
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرئيسي للمصفوفة A. LDA >= max(1,N).
!>

SDIM

!>          SDIM هو INTEGER
!>          إذا كان SORT = 'N'، فإن SDIM = 0.
!>          إذا كان SORT = 'S'، فإن SDIM = عدد القيم الذاتية التي
!>                         يكون SELECT لها صحيحًا.
!>

W

!>          W هي مصفوفة COMPLEX*16، ذات بعد (N)
!>          تحتوي W على القيم الذاتية المحسوبة، بنفس الترتيب
!>          الذي تظهر به على القطر الرئيسي لشكل شتور الناتج T.
!>

VS

!>          VS هي مصفوفة COMPLEX*16، ذات بعد (LDVS,N)
!>          إذا كان JOBVS = 'V'، تحتوي VS على المصفوفة المتعامدة Z لمتجهات شتور.
!>          إذا كان JOBVS = 'N'، لا يتم الرجوع إلى VS.
!>

LDVS

!>          LDVS هو INTEGER
!>          البعد الرئيسي للمصفوفة VS. LDVS >= 1، وإذا
!>          كان JOBVS = 'V'، فإن LDVS >= N.
!>

RCONDE

!>          RCONDE هو DOUBLE PRECISION
!>          إذا كان SENSE = 'E' أو 'B'، فإن RCONDE يحتوي على مقلوب
!>          الرقم الشرطي لمتوسط القيم الذاتية المحددة.
!>          لا يتم الرجوع إليه إذا كان SENSE = 'N' أو 'V'.
!>

RCONDV

!>          RCONDV هو DOUBLE PRECISION
!>          إذا كان SENSE = 'V' أو 'B'، فإن RCONDV يحتوي على مقلوب
!>          الرقم الشرطي للمتجه الفرعي الأيمن المحدد.
!>          لا يتم الرجوع إليه إذا كان SENSE = 'N' أو 'E'.
!>

WORK

!>          WORK هي مصفوفة COMPLEX*16، ذات بعد (MAX(1,LWORK))
!>          عند الخروج، إذا كان INFO = 0، فإن WORK(1) يعيد LWORK الأمثل.
!>

LWORK

!>          LWORK هو INTEGER
!>          بعد المصفوفة WORK. LWORK >= max(1,2*N).
!>          أيضًا، إذا كان SENSE = 'E' أو 'V' أو 'B'، فإن LWORK >= 2*SDIM*(N-SDIM)،
!>          حيث SDIM هو عدد القيم الذاتية المحددة التي حسبها
!>          هذا الروتين. لاحظ أن 2*SDIM*(N-SDIM) <= N*N/2. لاحظ أيضًا
!>          أن الخطأ يتم إرجاعه فقط إذا كان LWORK < max(1,2*N)، ولكن إذا
!>          كان SENSE = 'E' أو 'V' أو 'B'، فقد لا يكون هذا كافيًا.
!>          لأداء جيد، يجب أن يكون LWORK أكبر بشكل عام.
!>
!>          إذا كان LWORK = -1، فسيتم افتراض استعلام مساحة العمل؛ الروتين
!>          يحسب فقط الحد الأعلى للحجم الأمثل للمصفوفة
!>          WORK، ويعيد هذه القيمة كالإدخال الأول للمصفوفة
!>          WORK، ولا يصدر خطأ متعلق بـ LWORK بواسطة
!>          XERBLA.
!>

RWORK

!>          RWORK هي مصفوفة DOUBLE PRECISION، ذات بعد (N)
!>

BWORK

!>          BWORK هي مصفوفة LOGICAL، ذات بعد (N)
!>          لا يتم الرجوع إليها إذا كان SORT = 'N'.
!>

INFO

!>          INFO هو INTEGER
!>          = 0: خروج ناجح
!>          < 0: إذا كان INFO = -i، فإن الوسيط i كان له قيمة غير قانونية.
!>          > 0: إذا كان INFO = i، و i هو
!>             <= N: فشل خوارزمية QR في حساب جميع
!>                   القيم الذاتية؛ العناصر 1:ILO-1 و i+1:N من W
!>                   تحتوي على تلك القيم الذاتية التي تقاربت؛ إذا
!>                   كان JOBVS = 'V'، تحتوي VS على التحويل الذي
!>                   يختزل A إلى شكل شتور المتقارب جزئيًا.
!>             = N+1: لم يكن من الممكن إعادة ترتيب القيم الذاتية لأن بعض
!>                   القيم الذاتية كانت قريبة جدًا من الفصل (المشكلة
!>                   غير مشروطة للغاية)؛
!>             = N+2: بعد إعادة الترتيب، غير التقريب قيم بعض
!>                   القيم الذاتية المعقدة بحيث لم تعد القيم الذاتية الرائدة
!>                   في شكل شتور تحقق SELECT=.TRUE.  قد يكون هذا ناتجًا أيضًا
!>                   عن التدفق السفلي بسبب القياس.
!>
