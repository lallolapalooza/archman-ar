subroutine zgtts2 (integer itrans, integer n, integer nrhs, complex*16, dimension( * ) dl, complex*16, dimension( * ) d, complex*16, dimension( * ) du, complex*16, dimension( * ) du2, integer, dimension( * ) ipiv, complex*16, dimension( ldb, * ) b, integer ldb)

ZGTTS2 تحل نظامًا من المعادلات الخطية مع مصفوفة ثلاثية الأقطار باستخدام التحليل LU الذي تم حسابه بواسطة sgttrf.

الغرض:

!>
!> ZGTTS2 تحل أحد أنظمة المعادلات
!>    A * X = B,  A**T * X = B,  أو  A**H * X = B,
!> مع مصفوفة ثلاثية الأقطار A باستخدام التحليل LU المحسوب
!> بواسطة ZGTTRF.
!>

المعلمات

ITRANS 
!>          ITRANS هو عدد صحيح
!>          يحدد شكل نظام المعادلات.
!>          = 0:  A * X = B     (بدون تبديل)
!>          = 1:  A**T * X = B  (تبديل)
!>          = 2:  A**H * X = B  (تبديل مرافق)
!>

N

!>          N هو عدد صحيح
!>          رتبة المصفوفة A.
!>

NRHS

!>          NRHS هو عدد صحيح
!>          عدد الأطراف اليمنى، أي عدد أعمدة
!>          المصفوفة B.  NRHS >= 0.
!>

DL

!>          DL هو مصفوفة COMPLEX*16، البعد (N-1)
!>          المضاعفات (n-1) التي تحدد المصفوفة L من
!>          التحليل LU للمصفوفة A.
!>

D

!>          D هو مصفوفة COMPLEX*16، البعد (N)
!>          العناصر القطرية n للمصفوفة المثلثية العلوية U من
!>          التحليل LU للمصفوفة A.
!>

DU

!>          DU هو مصفوفة COMPLEX*16، البعد (N-1)
!>          العناصر (n-1) في القطر العلوي الأول لـ U.
!>

DU2

!>          DU2 هو مصفوفة COMPLEX*16، البعد (N-2)
!>          العناصر (n-2) في القطر العلوي الثاني لـ U.
!>

IPIV

!>          IPIV هي مصفوفة عدد صحيح، البعد (N)
!>          مؤشرات المحور؛ لـ 1 <= i <= n، تم تبديل الصف i من المصفوفة
!>          مع الصف IPIV(i).  سيكون IPIV(i) دائمًا إما
!>          i أو i+1؛ يشير IPIV(i) = i إلى أنه لم يكن هناك حاجة لتبديل الصف.
!>

B

!>          B هي مصفوفة COMPLEX*16، البعد (LDB,NRHS)
!>          عند الدخول، المصفوفة التي تحتوي على متجهات الطرف الأيمن B.
!>          عند الخروج، يتم الكتابة فوق B بمتجهات الحل X.
!>

LDB

!>          LDB هو عدد صحيح
!>          البعد الرئيسي للمصفوفة B.  LDB >= max(1,N).
!>
