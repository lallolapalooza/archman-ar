الغرض:

!>
!>    ZGERFSX يحسن الحل المحسوب لنظام المعادلات الخطية ويوفر حدود الخطأ وتقديرات الخطأ العكسي
!>    للحل. بالإضافة إلى حد الخطأ الإجمالي، يوفر الكود حد الخطأ المكون الأقصى قدر الإمكان. راجع
!>    التعليقات لـ ERR_BNDS_NORM و ERR_BNDS_COMP لتفاصيل حدود الخطأ.
!>
!>    قد يكون نظام المعادلات الخطية الأصلي قد تمت موازنته
!>    قبل استدعاء هذا الروتين، كما هو موضح بواسطة الوسيطات EQUED و R
!>    و C أدناه. في هذه الحالة، يتم إرجاع الحل وحدود الخطأ
!>    للنظام الأصلي غير المتوازن.
!> 
!>     يتم تجميع بعض المعلمات الاختيارية في مصفوفة PARAMS. هذه
!>     تحدد الإعدادات كيفية إجراء التنقيح، ولكن غالبًا ما تكون
!>     الإعدادات الافتراضية مقبولة. إذا كانت الإعدادات الافتراضية مقبولة، يمكن للمستخدمين
!>     تمرير NPARAMS = 0 الذي يمنع الكود المصدري من الوصول
!>     إلى وسيطة PARAMS.
!>

المعلمات

TRANS 
!>          TRANS هو CHARACTER*1
!>     يحدد شكل نظام المعادلات:
!>       = 'N':  A * X = B     (بدون تبديل)
!>       = 'T':  A**T * X = B  (تبديل)
!>       = 'C':  A**H * X = B  (تبديل مترافق)
!>

EQUED

!>          EQUED هو CHARACTER*1
!>     يحدد شكل الموازنة الذي تم إجراؤه على A
!>     قبل استدعاء هذا الروتين. هذا ضروري لحساب
!>     الحل وحدود الخطأ بشكل صحيح.
!>       = 'N':  لا يوجد موازنة
!>       = 'R':  موازنة الصفوف، أي تم ضرب A مسبقًا بواسطة
!>               diag(R).
!>       = 'C':  موازنة الأعمدة، أي تم ضرب A لاحقًا
!>               بواسطة diag(C).
!>       = 'B':  موازنة الصفوف والأعمدة، أي تم استبدال A بـ
!>               diag(R) * A * diag(C).
!>               تم تغيير الجانب الأيمن B وفقًا لذلك.
!>

N

!>          N هو INTEGER
!>     رتبة المصفوفة A.  N >= 0.
!>

NRHS

!>          NRHS هو INTEGER
!>     عدد الجوانب اليمنى، أي عدد أعمدة
!>     المصفوفتين B و X.  NRHS >= 0.
!>

A

!>          A هو COMPLEX*16 array, dimension (LDA,N)
!>     المصفوفة الأصلية A بحجم N×N.
!>

LDA

!>          LDA هو INTEGER
!>     البعد الرائد للمصفوفة A.  LDA >= max(1,N).
!>

AF

!>          AF هو COMPLEX*16 array, dimension (LDAF,N)
!>     العوامل L و U من تحلل A = P*L*U
!>     كما تم حسابه بواسطة ZGETRF.
!>

LDAF

!>          LDAF هو INTEGER
!>     البعد الرائد للمصفوفة AF.  LDAF >= max(1,N).
!>

IPIV

!>          IPIV هو INTEGER array, dimension (N)
!>     فهارس المحور من ZGETRF؛ لـ 1<=i<=N، تم تبديل الصف i من
!>     المصفوفة مع الصف IPIV(i).
!>

R

!>          R هو DOUBLE PRECISION array, dimension (N)
!>     عوامل قياس الصف لـ A.  إذا كان EQUED = 'R' أو 'B'، يتم
!>     ضرب A من اليسار بواسطة diag(R)؛ إذا كان EQUED = 'N' أو 'C'،
!>     فإن R لا يتم الوصول إليه.
!>     إذا تم الوصول إلى R، يجب أن يكون كل عنصر في R قوة من الأساس
!>     لضمان حل موثوق به وتقديرات خطأ. لا يسبب القياس
!>     بقوى الأساس أخطاء تقريب ما لم تكن النتيجة تحت التدفق أو
!>     تجاوز التدفق. تؤدي أخطاء التقريب أثناء القياس
!>     إلى التنقيح بمصفوفة غير مكافئة للمصفوفة المدخلة، مما ينتج عنه
!>     تقديرات خطأ قد لا تكون موثوقة.
!>

C

!>          C هو DOUBLE PRECISION array, dimension (N)
!>     عوامل قياس العمود لـ A.  إذا كان EQUED = 'C' أو 'B'، يتم
!>     ضرب A من اليمين بواسطة diag(C)؛ إذا كان EQUED = 'N' أو 'R'،
!>     فإن C لا يتم الوصول إليه.
!>     إذا تم الوصول إلى C، يجب أن يكون كل عنصر في C قوة من الأساس
!>     لضمان حل موثوق به وتقديرات خطأ. لا يسبب القياس
!>     بقوى الأساس أخطاء تقريب ما لم تكن النتيجة تحت التدفق أو
!>     تجاوز التدفق. تؤدي أخطاء التقريب أثناء القياس
!>     إلى التنقيح بمصفوفة غير مكافئة للمصفوفة المدخلة، مما ينتج عنه
!>     تقديرات خطأ قد لا تكون موثوقة.
!>

B

!>          B هو COMPLEX*16 array, dimension (LDB,NRHS)
!>     مصفوفة الجانب الأيمن B.
!>

LDB

!>          LDB هو INTEGER
!>     البعد الرائد للمصفوفة B.  LDB >= max(1,N).
!>

X

!>          X هو COMPLEX*16 array, dimension (LDX,NRHS)
!>     عند الدخول، مصفوفة الحل X، كما تم حسابها بواسطة ZGETRS.
!>     عند الخروج، مصفوفة الحل المحسنة X.
!>

LDX

!>          LDX هو INTEGER
!>     البعد الرائد للمصفوفة X.  LDX >= max(1,N).
!>

RCOND

!>          RCOND هو DOUBLE PRECISION
!>     مقلوب رقم الشرط المقاس. هذا تقدير لـ
!>     مقلوب رقم شرط Skeel للمصفوفة A بعد
!>     الموازنة (إذا تم إجراؤها). إذا كان هذا أقل من دقة الآلة
!>     (على وجه الخصوص، إذا كان صفرًا)، فإن المصفوفة منفردة
!>     للدقة العاملة. لاحظ أن الخطأ قد لا يزال صغيرًا حتى
!>     إذا كان هذا الرقم صغيرًا جدًا وتبدو المصفوفة غير مشروطة.
!>

BERR

!>          BERR هو DOUBLE PRECISION array, dimension (NRHS)
!>     خطأ عكسي نسبي مكون بمكون. هذا هو
!>     الخطأ العكسي النسبي مكون بمكون لكل متجه حل X(j)
!>     (أي، أصغر تغيير نسبي في أي عنصر من A أو B يجعل
!>     X(j) حلاً دقيقًا).
!>

N_ERR_BNDS

!>          N_ERR_BNDS هو INTEGER
!>     عدد حدود الخطأ المطلوب إرجاعها لكل جانب أيمن
!>     وكل نوع (إجمالي أو مكون بمكون). راجع ERR_BNDS_NORM و
!>     ERR_BNDS_COMP أدناه.
!>

ERR_BNDS_NORM

!>          ERR_BNDS_NORM هو DOUBLE PRECISION array, dimension (NRHS, N_ERR_BNDS)
!>     لكل جانب أيمن، ث
