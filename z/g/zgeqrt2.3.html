subroutine zgeqrt2 (integer m, integer n, complex*16, dimension( lda, * ) a, integer lda, complex*16, dimension( ldt, * ) t, integer ldt, integer info)

ZGEQRT2 يحسب تحليلاً QR لمصفوفة عامة حقيقية أو معقدة باستخدام تمثيل WY المدمج لـ Q.

الغرض:

!>
!> ZGEQRT2 يحسب تحليلاً QR لمصفوفة A المعقدة M-by-N،
!> باستخدام تمثيل WY المدمج لـ Q.
!>

المعلمات

M 
!>          M هو عدد صحيح (INTEGER)
!>          عدد صفوف المصفوفة A. M >= N.
!>

N

!>          N هو عدد صحيح (INTEGER)
!>          عدد أعمدة المصفوفة A. N >= 0.
!>

A

!>          A هو مصفوفة من النوع COMPLEX*16، البعد (LDA,N)
!>          عند الدخول، المصفوفة المعقدة M-by-N A. عند الخروج، تحتوي العناصر الموجودة على القطر وفوقه على المصفوفة المثلثية العلوية R بحجم N-by-N؛
!>          العناصر الموجودة تحت القطر تحتوي على أعمدة V. راجع ما يلي لمزيد من التفاصيل.
!>

LDA

!>          LDA هو عدد صحيح (INTEGER)
!>          البعد الرائد للمصفوفة A. LDA >= max(1,M).
!>

T

!>          T هو مصفوفة من النوع COMPLEX*16، البعد (LDT,N)
!>          المصفوفة المثلثية العلوية بحجم N-by-N للمنعكس القطاعي.
!>          العناصر الموجودة على القطر وفوقه تحتوي على المنعكس القطاعي T؛ العناصر الموجودة تحت القطر لا تُستخدم.
!>          راجع ما يلي لمزيد من التفاصيل.
!>

LDT

!>          LDT هو عدد صحيح (INTEGER)
!>          البعد الرائد للمصفوفة T. LDT >= max(1,N).
!>

INFO

!>          INFO هو عدد صحيح (INTEGER)
!>          = 0: انتهاء ناجح
!>          < 0: إذا كان INFO = -i، فإن المعامل i كان له قيمة غير قانونية
!>

تفاصيل إضافية:

!>
!>  المصفوفة V تخزن المنعكسات الأولية H(i) في العمود i
!>  تحت القطر. على سبيل المثال، إذا كانت M=5 و N=3، فإن المصفوفة V هي
!>
!>               V = (  1       )
!>                   ( v1  1    )
!>                   ( v1 v2  1 )
!>                   ( v1 v2 v3 )
!>                   ( v1 v2 v3 )
!>
!>  حيث تمثل vi المتجهات التي تحدد H(i)، والتي يتم إرجاعها
!>  في المصفوفة A.  الواحدات على طول قطر V لا يتم تخزينها في A.  المنعكس القطاعي H يعطى بعد ذلك بواسطة
!>
!>               H = I - V * T * V**H
!>
!>  حيث V**H هو المرافق المتنقل لـ V.
!>
