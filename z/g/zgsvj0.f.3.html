subroutine zgsvj0 (character*1 jobv, integer m, integer n, complex*16, dimension( lda, * ) a, integer lda, complex*16, dimension( n ) d, double precision, dimension( n ) sva, integer mv, complex*16, dimension( ldv, * ) v, integer ldv, double precision eps, double precision sfmin, double precision tol, integer nsweep, complex*16, dimension( lwork ) work, integer lwork, integer info)

ZGSVJ0 معالج مسبق للروتين zgesvj.

الغرض:

!>
!> يتم استدعاء ZGSVJ0 من ZGESVJ كمعالج مسبق وهذا هو غرضه الرئيسي. يقوم بتطبيق دورانات يعقوبي بنفس الطريقة التي يقوم بها ZGESVJ، ولكنه لا يتحقق من التقارب (معيار التوقف). تتوفر بعض معلمات الضبط (المشار إليها بـ [TP]) للمنفذ.
!>

المعلمات

JOBV 
!>          JOBV هو CHARACTER*1
!>          يحدد ما إذا كان سيتم استخدام المخرجات من هذا الإجراء
!>          لحساب المصفوفة V:
!>          = 'V': يتم تجميع حاصل ضرب دورانات يعقوبي
!>                 عن طريق الضرب اللاحق للمصفوفة V ذات الأبعاد N×N.
!>                (انظر وصف V.)
!>          = 'A': يتم تجميع حاصل ضرب دورانات يعقوبي
!>                 عن طريق الضرب اللاحق للمصفوفة V ذات الأبعاد MV×N.
!>                (انظر وصف MV و V.)
!>          = 'N': لا يتم تجميع دورانات يعقوبي.
!>

M

!>          M هو INTEGER
!>          عدد صفوف المصفوفة المدخلة A.  M >= 0.
!>

N

!>          N هو INTEGER
!>          عدد أعمدة المصفوفة المدخلة A.
!>          M >= N >= 0.
!>

A

!>          A هي مصفوفة COMPLEX*16، الأبعاد (LDA,N)
!>          عند الدخول، مصفوفة A ذات الأبعاد M×N، بحيث تمثل A*diag(D)
!>          المصفوفة المدخلة.
!>          عند الخروج،
!>          A_onexit * diag(D_onexit) تمثل المصفوفة المدخلة A*diag(D)
!>          مضروبة لاحقًا في سلسلة من دورانات يعقوبي، حيث
!>          يتم تحديد عتبة الدوران وعدد الدورات الكلي في
!>          TOL و NSWEEP، على التوالي.
!>          (انظر أوصاف D و TOL و NSWEEP.)
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرائد للمصفوفة A.  LDA >= max(1,M).
!>

D

!>          D هي مصفوفة COMPLEX*16، الأبعاد (N)
!>          تجمع المصفوفة D عوامل القياس من دورانات يعقوبي المعقدة والمقاسة.
!>          عند الدخول، تمثل A*diag(D) المصفوفة المدخلة.
!>          عند الخروج، تمثل A_onexit*diag(D_onexit) المصفوفة المدخلة
!>          مضروبة لاحقًا في سلسلة من دورانات يعقوبي، حيث
!>          يتم تحديد عتبة الدوران وعدد الدورات الكلي في
!>          TOL و NSWEEP، على التوالي.
!>          (انظر أوصاف A و TOL و NSWEEP.)
!>

SVA

!>          SVA هي مصفوفة DOUBLE PRECISION، الأبعاد (N)
!>          عند الدخول، تحتوي SVA على المعايير الإقليدية لأعمدة
!>          المصفوفة A*diag(D).
!>          عند الخروج، تحتوي SVA على المعايير الإقليدية لأعمدة
!>          المصفوفة A_onexit*diag(D_onexit).
!>

MV

!>          MV هو INTEGER
!>          إذا كان JOBV = 'A'، فسيتم ضرب MV صفًا من V لاحقًا في
!>                           سلسلة من دورانات يعقوبي.
!>          إذا كان JOBV = 'N'،   فلن يتم الرجوع إلى MV.
!>

V

!>          V هي مصفوفة COMPLEX*16، الأبعاد (LDV,N)
!>          إذا كان JOBV = 'V' فسيتم ضرب N صفًا من V لاحقًا في
!>                           سلسلة من دورانات يعقوبي.
!>          إذا كان JOBV = 'A' فسيتم ضرب MV صفًا من V لاحقًا في
!>                           سلسلة من دورانات يعقوبي.
!>          إذا كان JOBV = 'N'،   فلن يتم الرجوع إلى V.
!>

LDV

!>          LDV هو INTEGER
!>          البعد الرائد للمصفوفة V،  LDV >= 1.
!>          إذا كان JOBV = 'V'، LDV >= N.
!>          إذا كان JOBV = 'A'، LDV >= MV.
!>

EPS

!>          EPS هو DOUBLE PRECISION
!>          EPS = DLAMCH('Epsilon')
!>

SFMIN

!>          SFMIN هو DOUBLE PRECISION
!>          SFMIN = DLAMCH('Safe Minimum')
!>

TOL

!>          TOL هو DOUBLE PRECISION
!>          TOL هي العتبة لدورانات يعقوبي. لزوج
!>          A(:,p) و A(:,q) من أعمدة المحور، يتم تطبيق دوران يعقوبي
!>          فقط إذا كانت ABS(COS(angle(A(:,p),A(:,q)))) > TOL.
!>

NSWEEP

!>          NSWEEP هو INTEGER
!>          NSWEEP هو عدد دورات يعقوبي التي سيتم إجراؤها.
!>

WORK

!>          WORK هي مصفوفة COMPLEX*16، الأبعاد (LWORK)
!>

LWORK

!>          LWORK هو INTEGER
!>          LWORK هو بعد WORK. LWORK >= M.
!>

INFO

!>          INFO هو INTEGER
!>          = 0:  الخروج بنجاح.
!>          < 0:  إذا كان INFO = -i، فإن الوسيط i كان له قيمة غير قانونية
!>
