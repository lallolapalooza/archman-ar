subroutine zgttrf (integer n, complex*16, dimension( * ) dl, complex*16, dimension( * ) d, complex*16, dimension( * ) du, complex*16, dimension( * ) du2, integer, dimension( * ) ipiv, integer info)

ZGTTRF

الغرض:

!>
!> ZGTTRF يحسب تحليل LU لمصفوفة A ثلاثية الأقطار معقدة
!> باستخدام الاستبعاد مع المحوري الجزئي وتبادلات الصفوف.
!>
!> التحليل له الشكل
!>    A = L * U
!> حيث L هو حاصل ضرب مصفوفات التبديل الثنائية القطرية الوحدوية
!> و U مصفوفة مثلثة علوية ذات عناصر غير صفرية فقط في القطر الرئيسي
!> والقطرين الفائقين الأولين.
!>

المعلمات

N 
!>          N هو عدد صحيح (INTEGER)
!>          رتبة المصفوفة A.
!>

DL

!>          DL هو مصفوفة من نوع COMPLEX*16، حجمها (N-1)
!>          عند الإدخال، يجب أن تحتوي DL على عناصر دون القطر الفرعي (n-1) للمصفوفة
!>          A.
!>
!>          عند الخروج، يتم الكتابة فوق DL بواسطة المضاعفات (n-1) التي
!>          تحدد المصفوفة L من تحليل LU للمصفوفة A.
!>

D

!>          D هو مصفوفة من نوع COMPLEX*16، حجمها (N)
!>          عند الإدخال، يجب أن تحتوي D على عناصر القطر الرئيسي للمصفوفة A.
!>
!>          عند الخروج، يتم الكتابة فوق D بواسطة عناصر القطر الرئيسي (n) للمصفوفة
!>          المثلثة العلوية U من تحليل LU للمصفوفة A.
!>

DU

!>          DU هو مصفوفة من نوع COMPLEX*16، حجمها (N-1)
!>          عند الإدخال، يجب أن تحتوي DU على عناصر القطر الفائق (n-1)
!>          للمصفوفة A.
!>
!>          عند الخروج، يتم الكتابة فوق DU بواسطة عناصر القطر الفائق الأول (n-1) للمصفوفة U.
!>

DU2

!>          DU2 هو مصفوفة من نوع COMPLEX*16، حجمها (N-2)
!>          عند الخروج، يتم الكتابة فوق DU2 بواسطة عناصر القطر الفائق الثاني (n-2) للمصفوفة U.
!>

IPIV

!>          IPIV هو مصفوفة من نوع INTEGER، حجمها (N)
!>          مؤشرات المحوري؛ لكل 1 <= i <= n، تم تبديل الصف i للمصفوفة مع الصف IPIV(i).
!>          سيكون IPIV(i) دائمًا إما i أو i+1؛ يشير IPIV(i) = i إلى أنه لم يكن هناك حاجة لتبديل صف.
!>

INFO

!>          INFO هو عدد صحيح (INTEGER)
!>          = 0:  خروج ناجح
!>          < 0:  إذا كان INFO = -k، فإن الوسيط k كان له قيمة غير قانونية
!>          > 0:  إذا كان INFO = k، فإن U(k,k) يساوي صفرًا تمامًا. تم
!>                إكمال التحليل، ولكن العامل U هو بشكل كامل
!>                فردي، وسيحدث قسمة على صفر إذا تم استخدامه
!>                لحل نظام من المعادلات.
!>
