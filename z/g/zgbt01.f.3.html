توثيق الوظيفة/الإجراء الفرعي

subroutine zgbt01 (integer m, integer n, integer kl, integer ku, complex*16, dimension( lda, * ) a, integer lda, complex*16, dimension( ldafac, * ) afac, integer ldafac, integer, dimension( * ) ipiv, complex*16, dimension( * ) work, double precision resid)

ZGBT01

الغرض:

!>
!> ZGBT01 يعيد بناء مصفوفة النطاق A من عامل L*U الخاص بها ويحسب الباقي:
!>    norm(L*U - A) / ( N * norm(A) * EPS )
!> حيث EPS هو خطأ الآلة.
!>
!> يتم حساب التعبير L*U - A عمودًا تلو الآخر، لذا لا يتم تعديل A و
!> AFAC.
!>

المعاملات

M 
!>          M هو عدد صحيح (INTEGER)
!>          عدد صفوف المصفوفة A.  M >= 0.
!>

N

!>          N هو عدد صحيح (INTEGER)
!>          عدد أعمدة المصفوفة A.  N >= 0.
!>

KL

!>          KL هو عدد صحيح (INTEGER)
!>          عدد الأقطار الفرعية داخل نطاق A.  KL >= 0.
!>

KU

!>          KU هو عدد صحيح (INTEGER)
!>          عدد الأقطار العلوية داخل نطاق A.  KU >= 0.
!>

A

!>          A هي مصفوفة من نوع COMPLEX*16، ذات أبعاد (LDA,N)
!>          المصفوفة الأصلية A بتخزين النطاق، مخزنة في الصفوف من 1 إلى
!>          KL+KU+1.
!>

LDA

!>          LDA هو عدد صحيح (INTEGER).
!>          البعد الرائد للمصفوفة A.  LDA >= max(1,KL+KU+1).
!>

AFAC

!>          AFAC هي مصفوفة من نوع COMPLEX*16، ذات أبعاد (LDAFAC,N)
!>          الشكل المفكوك للمصفوفة A.  AFAC تحتوي على عوامل النطاق L و U
!>          من عامل L*U، كما تم حسابه بواسطة ZGBTRF.  U مخزنة كمصفوفة
!>          نطاق علوي مثلثي بـ KL+KU من الأقطار العلوية في الصفوف من 1 إلى
!>          KL+KU+1، والعوامل المستخدمة أثناء عملية التحليل مخزنة في الصفوف
!>          من KL+KU+2 إلى 2*KL+KU+1.  انظر ZGBTRF لمزيد من التفاصيل.
!>

LDAFAC

!>          LDAFAC هو عدد صحيح (INTEGER)
!>          البعد الرائد للمصفوفة AFAC.
!>          LDAFAC >= max(1,2*KL*KU+1).
!>

IPIV

!>          IPIV هي مصفوفة من نوع INTEGER، ذات أبعاد (min(M,N))
!>          مؤشرات المحور من ZGBTRF.
!>

WORK

!>          WORK هي مصفوفة من نوع COMPLEX*16، ذات أبعاد (2*KL+KU+1)
!>

RESID

!>          RESID هو DOUBLE PRECISION
!>          norm(L*U - A) / ( N * norm(A) * EPS )
!>
