subroutine zgeqrf (integer m, integer n, complex*16, dimension( lda, * ) a, integer lda, complex*16, dimension( * ) tau, complex*16, dimension( * ) work, integer lwork, integer info)

ZGEQRF

الغرض:

!>
!> ZGEQRF تحسب تحليل QR لمصفوفة معقدة M-by-N A:
!>
!>    A = Q * ( R ),
!>            ( 0 )
!>
!> حيث:
!>
!>    Q هي مصفوفة متعامدة M-by-M؛
!>    R هي مصفوفة علوية مثلثية N-by-N؛
!>    0 هي مصفوفة صفرية (M-N)-by-N، إذا كانت M > N.
!>
!>

المعلمات

M 
!>          M هو عدد صحيح (INTEGER)
!>          عدد صفوف المصفوفة A.  M >= 0.
!>

N

!>          N هو عدد صحيح (INTEGER)
!>          عدد أعمدة المصفوفة A.  N >= 0.
!>

A

!>          A هو مصفوفة أعداد مركبة *16 (COMPLEX*16) ، حجم (LDA,N)
!>          عند الدخول، مصفوفة A ذات الأبعاد M-by-N.
!>          عند الخروج، العناصر على القطر الرئيسي وما فوقه في المصفوفة
!>          تحتوي على المصفوفة الشبه منحرفة العلوية R ذات الأبعاد min(M,N)-by-N (R
!>          مثلثة علوية إذا كانت m >= n)؛ العناصر تحت القطر الرئيسي،
!>          مع المصفوفة TAU، تمثل المصفوفة الأحادية Q كـ
!>          ناتج ضرب min(m,n) عاكسات ابتدائية (انظر التفاصيل الإضافية).
!>

LDA

!>          LDA هو عدد صحيح (INTEGER)
!>          البعد الرائد للمصفوفة A.  LDA >= max(1,M).
!>

TAU

!>          TAU هو مصفوفة أعداد مركبة *16 (COMPLEX*16) ، حجم (min(M,N))
!>          العوامل العددية للعواكس الابتدائية (انظر التفاصيل الإضافية).
!>

WORK

!>          WORK هي مصفوفة أعداد مركبة *16 (COMPLEX*16) ، حجم (MAX(1,LWORK))
!>          عند الخروج، إذا كان INFO = 0، فإن WORK(1) تعيد قيمة LWORK المثلى.
!>

LWORK

!>          LWORK هو عدد صحيح (INTEGER)
!>          بعد مصفوفة WORK.
!>          LWORK >= 1، إذا كان MIN(M,N) = 0، و LWORK >= N، وإلا.
!>          للأداء الأمثل LWORK >= N*NB، حيث NB هو
!>          حجم الكتلة الأمثل.
!>
!>          إذا كان LWORK = -1، فسيتم افتراض استعلام عن مساحة العمل؛ الروتين
!>          يحسب فقط الحجم الأمثل للمصفوفة WORK، ويعيد
!>          هذه القيمة كالإدخال الأول للمصفوفة WORK، ولا يتم إصدار أي رسالة خطأ
!>          متعلقة بـ LWORK بواسطة XERBLA.
!>

INFO

!>          INFO هو عدد صحيح (INTEGER)
!>          = 0:  الخروج بنجاح
!>          < 0:  إذا كان INFO = -i، فإن الوسيط i كان له قيمة غير قانونية
!>

تفاصيل إضافية:

!>
!>  يتم تمثيل المصفوفة Q كناتج ضرب عواكس ابتدائية
!>
!>     Q = H(1) H(2) . . . H(k)، حيث k = min(m,n).
!>
!>  كل H(i) لها الشكل
!>
!>     H(i) = I - tau * v * v**H
!>
!>  حيث tau هو عامل قياسي معقد، و v هو متجه معقد مع
!>  v(1:i-1) = 0 و v(i) = 1؛ يتم تخزين v(i+1:m) عند الخروج في A(i+1:m,i)،
!>  و tau في TAU(i).
!>
