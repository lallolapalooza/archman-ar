subroutine zget38 (double precision, dimension( 3 ) rmax, integer, dimension( 3 ) lmax, integer, dimension( 3 ) ninfo, integer knt, integer nin)

ZGET38

الغرض:

!>
!> ZGET38 يختبر ZTRSEN، وهو إجراء لتقدير أرقام الشروط لمجموعة من القيم الذاتية
!> و/أو الفضاء المتجهي الأيمن المرتبط بها.
!>
!> يتم قراءة مصفوفات الاختبار من ملف برقم الوحدة المنطقي NIN.
!>

المعلمات

RMAX 
!>          RMAX هو مصفوفة DOUBLE PRECISION، بُعدها (3)
!>          قيم أكبر نسب الاختبار.
!>          RMAX(1) = أكبر البواقي من ZHST01 أو مقارنة
!>                    استدعاءات مختلفة لـ ZTRSEN
!>          RMAX(2) = أكبر خطأ في أرقام الشروط العكسية
!>                    مع الأخذ في الاعتبار ظروفها
!>          RMAX(3) = أكبر خطأ في أرقام الشروط العكسية
!>                    بدون أخذ ظروفها في الاعتبار (قد تكون أكبر من RMAX(2))
!>

LMAX

!>          LMAX هي مصفوفة INTEGER، بُعدها (3)
!>          LMAX(i) هو رقم المثال حيث يتم تحقيق أكبر نسبة اختبار
!>          RMAX(i). أيضًا:
!>          إذا كانت ZGEHRD تُرجع INFO غير صفرية في المثال i، فإن LMAX(1)=i
!>          إذا كانت ZHSEQR تُرجع INFO غير صفرية في المثال i، فإن LMAX(2)=i
!>          إذا كانت ZTRSEN تُرجع INFO غير صفرية في المثال i، فإن LMAX(3)=i
!>

NINFO

!>          NINFO هي مصفوفة INTEGER، بُعدها (3)
!>          NINFO(1) = عدد مرات إرجاع ZGEHRD لـ INFO غير صفرية
!>          NINFO(2) = عدد مرات إرجاع ZHSEQR لـ INFO غير صفرية
!>          NINFO(3) = عدد مرات إرجاع ZTRSEN لـ INFO غير صفرية
!>

KNT

!>          KNT هو INTEGER
!>          العدد الإجمالي للأمثلة التي تم اختبارها.
!>

NIN

!>          NIN هو INTEGER
!>          رقم الوحدة المنطقي للإدخال.
!>
