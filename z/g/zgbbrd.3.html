وثائق الوظيفة/الروتين الفرعي

subroutine zgbbrd (character vect, integer m, integer n, integer ncc, integer kl, integer ku, complex*16, dimension( ldab, * ) ab, integer ldab, double precision, dimension( * ) d, double precision, dimension( * ) e, complex*16, dimension( ldq, * ) q, integer ldq, complex*16, dimension( ldpt, * ) pt, integer ldpt, complex*16, dimension( ldc, * ) c, integer ldc, complex*16, dimension( * ) work, double precision, dimension( * ) rwork, integer info)

ZGBBRD

الغرض:

!>
!> ZGBBRD يقلل مصفوفة النطاق العامة المعقدة A من الحجم m*n إلى شكل ثنائي قطري حقيقي B بواسطة تحويل وحيد: Q**H * A * P = B.
!>
!> تقوم الروتين بحساب B، وتشكل اختياريًا Q أو P**H، أو تحسب Q**H*C لمصفوفة C معطاة.
!>

المعلمات

VECT 
!>          VECT هو CHARACTER*1
!>          تحدد ما إذا كان يجب تشكيل المصفوفتين Q و P**H أم لا.
!>          = 'N': لا تشكل Q أو P**H؛
!>          = 'Q': تشكل Q فقط؛
!>          = 'P': تشكل P**H فقط؛
!>          = 'B': تشكل كلاهما.
!>

M

!>          M هو INTEGER
!>          عدد صفوف المصفوفة A. M >= 0.
!>

N

!>          N هو INTEGER
!>          عدد أعمدة المصفوفة A. N >= 0.
!>

NCC

!>          NCC هو INTEGER
!>          عدد أعمدة المصفوفة C. NCC >= 0.
!>

KL

!>          KL هو INTEGER
!>          عدد الأقطار الفرعية للمصفوفة A. KL >= 0.
!>

KU

!>          KU هو INTEGER
!>          عدد الأقطار الفوقية للمصفوفة A. KU >= 0.
!>

AB

!>          AB هو COMPLEX*16 array، البعد (LDAB,N)
!>          عند الإدخال، مصفوفة النطاق A بحجم m*n، مخزنة في الصفوف من 1 إلى
!>          KL+KU+1. يتم تخزين العمود j من A في العمود j من
!>          المصفوفة AB كالتالي:
!>          AB(ku+1+i-j,j) = A(i,j) حيث max(1,j-ku)<=i<=min(m,j+kl).
!>          عند الخروج، يتم استبدال A بقيم تم إنشاؤها أثناء
!>          الاختزال.
!>

LDAB

!>          LDAB هو INTEGER
!>          البعد الرئيسي للمصفوفة A. LDAB >= KL+KU+1.
!>

D

!>          D هو DOUBLE PRECISION array، البعد (min(M,N))
!>          العناصر القطرية للمصفوفة ثنائية القطر B.
!>

E

!>          E هو DOUBLE PRECISION array، البعد (min(M,N)-1)
!>          العناصر القطرية الفوقية للمصفوفة ثنائية القطر B.
!>

Q

!>          Q هو COMPLEX*16 array، البعد (LDQ,M)
!>          إذا كان VECT = 'Q' أو 'B'، فإن المصفوفة الوحدة Q بحجم m*m.
!>          إذا كان VECT = 'N' أو 'P'، فإن المصفوفة Q غير مستخدمة.
!>

LDQ

!>          LDQ هو INTEGER
!>          البعد الرئيسي للمصفوفة Q.
!>          LDQ >= max(1,M) إذا كان VECT = 'Q' أو 'B'؛ LDQ >= 1 بخلاف ذلك.
!>

PT

!>          PT هو COMPLEX*16 array، البعد (LDPT,N)
!>          إذا كان VECT = 'P' أو 'B'، فإن المصفوفة الوحدة P' بحجم n*n.
!>          إذا كان VECT = 'N' أو 'Q'، فإن المصفوفة PT غير مستخدمة.
!>

LDPT

!>          LDPT هو INTEGER
!>          البعد الرئيسي للمصفوفة PT.
!>          LDPT >= max(1,N) إذا كان VECT = 'P' أو 'B'؛ LDPT >= 1 بخلاف ذلك.
!>

C

!>          C هو COMPLEX*16 array، البعد (LDC,NCC)
!>          عند الإدخال، مصفوفة C بحجم m*ncc.
!>          عند الخروج، يتم استبدال C بـ Q**H*C.
!>          لا يتم استخدام C إذا كان NCC = 0.
!>

LDC

!>          LDC هو INTEGER
!>          البعد الرئيسي للمصفوفة C.
!>          LDC >= max(1,M) إذا كان NCC > 0؛ LDC >= 1 إذا كان NCC = 0.
!>

WORK

!>          WORK هو COMPLEX*16 array، البعد (max(M,N))
!>

RWORK

!>          RWORK هو DOUBLE PRECISION array، البعد (max(M,N))
!>

INFO

!>          INFO هو INTEGER
!>          = 0:  الخروج بنجاح.
!>          < 0:  إذا كان INFO = -i، فإن الوسيط i كان له قيمة غير قانونية.
!>
