توثيق الوظيفة/الروتين الفرعي

subroutine zget03 (integer n, complex*16, dimension( lda, * ) a, integer lda, complex*16, dimension( ldainv, * ) ainv, integer ldainv, complex*16, dimension( ldwork, * ) work, integer ldwork, double precision, dimension( * ) rwork, double precision rcond, double precision resid)

ZGET03

الغرض:

!>
!> ZGET03 يحسب الباقي لمصفوفة عامة مضروبة في معكوسها:
!>    norm( I - AINV*A ) / ( N * norm(A) * norm(AINV) * EPS ),
!> حيث EPS هي العدد الأولي الآلي.
!>

المعلمات

N 
!>          N هو عدد صحيح (INTEGER).
!>          عدد الصفوف والأعمدة للمصفوفة A.  N >= 0.
!>

A

!>          A هو مصفوفة COMPLEX*16، البعد (LDA,N).
!>          المصفوفة الأصلية N x N A.
!>

LDA

!>          LDA هو عدد صحيح (INTEGER).
!>          البعد الرئيسي للمصفوفة A.  LDA >= max(1,N).
!>

AINV

!>          AINV هو مصفوفة COMPLEX*16، البعد (LDAINV,N).
!>          معكوس المصفوفة A.
!>

LDAINV

!>          LDAINV هو عدد صحيح (INTEGER).
!>          البعد الرئيسي للمصفوفة AINV.  LDAINV >= max(1,N).
!>

WORK

!>          WORK هي مصفوفة COMPLEX*16، البعد (LDWORK,N).
!>

LDWORK

!>          LDWORK هو عدد صحيح (INTEGER).
!>          البعد الرئيسي للمصفوفة WORK.  LDWORK >= max(1,N).
!>

RWORK

!>          RWORK هي مصفوفة DOUBLE PRECISION، البعد (N).
!>

RCOND

!>          RCOND هو DOUBLE PRECISION.
!>          مقلوب رقم الشرط للمصفوفة A، محسوب كـ
!>          ( 1/norm(A) ) / norm(AINV).
!>

RESID

!>          RESID هو DOUBLE PRECISION.
!>          norm(I - AINV*A) / ( N * norm(A) * norm(AINV) * EPS )
!>
