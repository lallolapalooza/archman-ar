subroutine zgeesx (character jobvs, character sort, external select, character sense, integer n, complex*16, dimension( lda, * ) a, integer lda, integer sdim, complex*16, dimension( * ) w, complex*16, dimension( ldvs, * ) vs, integer ldvs, double precision rconde, double precision rcondv, complex*16, dimension( * ) work, integer lwork, double precision, dimension( * ) rwork, logical, dimension( * ) bwork, integer info)

ZGEESX يحسب القيم الذاتية، شكل شور، وبشكل اختياري، مصفوفة متجهات شور لمصفوفات GE

الغرض:

!>
!> ZGEESX يحسب لمصفوفة معقدة غير متناظرة N × N، القيم الذاتية، شكل شور T، وبشكل اختياري، مصفوفة متجهات شور Z. هذا يعطي تحليل شور A = Z*T*(Z**H).
!>
!> بشكل اختياري، يقوم أيضًا بترتيب القيم الذاتية على القطر لشكل شور بحيث تكون القيم الذاتية المحددة في أعلى اليسار؛
!> يحسب رقم شرط معكوس لمتوسط القيم الذاتية المحددة (RCONDE)؛ ويحسب رقم شرط معكوس للفضاء المتجهي الأيمن المقابل للقيم الذاتية المحددة (RCONDV). الأعمدة الرائدة لـ Z تشكل أساسًا متعامدًا لهذا الفضاء المتجهي.
!>
!> لمزيد من التوضيح حول الأرقام الشرطية المعكوسة RCONDE و RCONDV، راجع القسم 4.10 من دليل مستخدمي LAPACK (حيث تسمى هذه الكميات s و sep على التوالي).
!>
!> المصفوفة المعقدة في شكل شور إذا كانت مثلثة علوية.
!>

المعلمات

JOBVS 
!>          JOBVS هو CHARACTER*1
!>          = 'N': لا يتم حساب متجهات شور؛
!>          = 'V': يتم حساب متجهات شور.
!>

SORT

!>          SORT هو CHARACTER*1
!>          يحدد ما إذا كان سيتم ترتيب القيم الذاتية على قطر شكل شور أم لا.
!>          = 'N': القيم الذاتية غير مرتبة؛
!>          = 'S': القيم الذاتية مرتبة (انظر SELECT).
!>

SELECT

!>          SELECT هي وظيفة LOGICAL (منطقية) لحجة COMPLEX*16 واحدة
!>          يجب تعريف SELECT على أنها EXTERNAL في الروتين الفرعي المستدعي.
!>          إذا كان SORT = 'S'، يتم استخدام SELECT لتحديد القيم الذاتية لترتيبها
!>          إلى أعلى اليسار لشكل شور.
!>          إذا كان SORT = 'N'، لا يتم الرجوع إلى SELECT.
!>          يتم تحديد القيمة الذاتية W(j) إذا كانت SELECT(W(j)) صحيحة.
!>

SENSE

!>          SENSE هو CHARACTER*1
!>          يحدد أي أرقام شرطية معكوسة يتم حسابها.
!>          = 'N': لا يتم حساب أي منها؛
!>          = 'E': يتم حسابها لمتوسط القيم الذاتية المحددة فقط؛
!>          = 'V': يتم حسابها للفضاء المتجهي الأيمن المختار فقط؛
!>          = 'B': يتم حسابها لكليهما.
!>          إذا كان SENSE = 'E'، 'V' أو 'B'، يجب أن يكون SORT = 'S'.
!>

N

!>          N هو INTEGER
!>          ترتيب المصفوفة A. N >= 0.
!>

A

!>          A هي مصفوفة COMPLEX*16، أبعادها (LDA, N)
!>          عند الدخول، المصفوفة N × N A.
!>          عند الخروج، يتم استبدال A بشكل شور T الخاص بها.
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرئيسي للمصفوفة A. LDA >= max(1,N).
!>

SDIM

!>          SDIM هو INTEGER
!>          إذا كان SORT = 'N'، فإن SDIM = 0.
!>          إذا كان SORT = 'S'، فإن SDIM = عدد القيم الذاتية التي
!>                         SELECT هي صحيحة بالنسبة لها.
!>

W

!>          W هي مصفوفة COMPLEX*16، أبعادها (N)
!>          تحتوي W على القيم الذاتية المحسوبة، بنفس الترتيب
!>          الذي تظهر به على قطر شكل شور الناتج T.
!>

VS

!>          VS هي مصفوفة COMPLEX*16، أبعادها (LDVS,N)
!>          إذا كان JOBVS = 'V'، فإن VS تحتوي على المصفوفة الوحدة Z لمتجهات شور.
!>          إذا كان JOBVS = 'N'، لا يتم الرجوع إلى VS.
!>

LDVS

!>          LDVS هو INTEGER
!>          البعد الرئيسي للمصفوفة VS. LDVS >= 1، وإذا كان
!>          JOBVS = 'V'، فإن LDVS >= N.
!>

RCONDE

!>          RCONDE هو DOUBLE PRECISION
!>          إذا كان SENSE = 'E' أو 'B'، فإن RCONDE تحتوي على الرقم الشرطي المعكوس
!>          لمتوسط القيم الذاتية المحددة.
!>          لا يتم الرجوع إليه إذا كان SENSE = 'N' أو 'V'.
!>

RCONDV

!>          RCONDV هو DOUBLE PRECISION
!>          إذا كان SENSE = 'V' أو 'B'، فإن RCONDV تحتوي على الرقم الشرطي المعكوس
!>          للفضاء المتجهي الأيمن المختار.
!>          لا يتم الرجوع إليه إذا كان SENSE = 'N' أو 'E'.
!>

WORK

!>          WORK هي مصفوفة COMPLEX*16، أبعادها (MAX(1,LWORK))
!>          عند الخروج، إذا كان INFO = 0، فإن WORK(1) يعيد LWORK الأمثل.
!>

LWORK

!>          LWORK هو INTEGER
!>          بعد المصفوفة WORK. LWORK >= max(1,2*N).
!>          أيضًا، إذا كان SENSE = 'E' أو 'V' أو 'B'، فإن LWORK >= 2*SDIM*(N-SDIM)،
!>          حيث SDIM هو عدد القيم الذاتية المحددة التي تم حسابها بواسطة
!>          هذا الروتين الفرعي. لاحظ أن 2*SDIM*(N-SDIM) <= N*N/2. لاحظ أيضًا
!>          أنه لا يتم إرجاع خطأ إلا إذا كان LWORK < max(1,2*N)، ولكن إذا كان
!>          SENSE = 'E' أو 'V' أو 'B' فقد لا يكون هذا كافياً.
!>          للحصول على أداء جيد، يجب أن يكون LWORK بشكل عام أكبر.
!>
!>          إذا كان LWORK = -1، فيفترض استعلام مساحة العمل؛ الروتين الفرعي
!>          يحسب فقط الحد الأعلى للحجم الأمثل للمصفوفة
!>          WORK، ويعيد هذه القيمة كالإدخال الأول للمصفوفة
!>          WORK، ولا يصدر XERBLA أي رسالة خطأ تتعلق بـ LWORK.
!>

RWORK

!>          RWORK هي مصفوفة DOUBLE PRECISION، أبعادها (N)
!>

BWORK

!>          BWORK هي مصفوفة LOGICAL، أبعادها (N)
!>          لا يتم الرجوع إليها إذا كان SORT = 'N'.
!>

INFO

!>          INFO هو INTEGER
!>          = 0: خروج ناجح
!>          < 0: إذا كان INFO = -i، فإن الوسيط i كان له قيمة غير قانونية.
!>          > 0: إذا كان INFO = i، و i هو
!>             <= N: فشل خوارزمية QR في حساب جميع
!>                   القيم الذاتية؛ العناصر 1:ILO-1 و i+1:N من W
!>                   تحتوي على تلك القيم الذاتية التي تقاربت؛ إذا كان
!>                   JOBVS = 'V'، فإن VS تحتوي على التحويل الذي
!>                   يقلل A إلى شكل شور الجزئي المتقارب الخاص بها.
!>             = N+1: لا يمكن إعادة ترتيب القيم الذاتية لأن بعض
!>                   القيم الذاتية كانت قريبة جدًا من الانفصال (المشكلة
!>                   غير مستقرة للغاية)؛
!>             = N+2: بعد إعادة الترتيب، قامت الدقة العشرية بتغيير قيم بعض
!>                   القيم الذاتية المعقدة بحيث لم تعد القيم الذاتية الرائدة في
!>                   شكل شور تحقق SELECT=.TRUE. يمكن أن يكون هذا ناجمًا أيضًا
!>                   عن نقص التدفق بسبب التحجيم.
!>
