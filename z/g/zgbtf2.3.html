توثيق الوظيفة/الروتين الفرعي

subroutine zgbtf2 (integer m, integer n, integer kl, integer ku, complex*16, dimension( ldab, * ) ab, integer ldab, integer, dimension( * ) ipiv, integer info)

ZGBTF2 يحسب تفكيك LU لمصفوفة نطاق عام باستخدام الإصدار غير المحظور من الخوارزمية.

الغرض:

!>
!> ZGBTF2 يحسب تفكيك LU لمصفوفة نطاق معقدة m-by-n
!> A باستخدام التجزئة الجزئية مع تبديلات الصفوف.
!>
!> هذا هو الإصدار غير المحظور من الخوارزمية، الذي يستدعي BLAS المستوى 2.
!>

المعلمات

M 
!>          M هو عدد صحيح (INTEGER)
!>          عدد صفوف المصفوفة A. M >= 0.
!>

N

!>          N هو عدد صحيح (INTEGER)
!>          عدد أعمدة المصفوفة A. N >= 0.
!>

KL

!>          KL هو عدد صحيح (INTEGER)
!>          عدد الأقطار الفرعية داخل نطاق A. KL >= 0.
!>

KU

!>          KU هو عدد صحيح (INTEGER)
!>          عدد الأقطار الفائقة داخل نطاق A. KU >= 0.
!>

AB

!>          AB هو مصفوفة معقدة *16 (COMPLEX*16)، ذات أبعاد (LDAB,N)
!>          عند الإدخال، المصفوفة A بتخزين النطاق، في الصفوف KL+1 إلى
!>          2*KL+KU+1؛ الصفوف 1 إلى KL من المصفوفة لا تحتاج إلى تعيين.
!>          يتم تخزين العمود j من A في العمود j من المصفوفة
!>          AB كما يلي:
!>          AB(kl+ku+1+i-j,j) = A(i,j) لـ max(1,j-ku)<=i<=min(m,j+kl)
!>
!>          عند الخروج، تفاصيل التفكيك: يتم تخزين U كمصفوفة نطاق مثلثية عليا
!>          مع KL+KU أقطار فائقة في الصفوف 1 إلى KL+KU+1، ويتم تخزين
!>          العوامل المستخدمة أثناء التفكيك في الصفوف KL+KU+2 إلى 2*KL+KU+1.
!>          انظر أدناه لمزيد من التفاصيل.
!>

LDAB

!>          LDAB هو عدد صحيح (INTEGER)
!>          البعد الرائد للمصفوفة AB. LDAB >= 2*KL+KU+1.
!>

IPIV

!>          IPIV هو مصفوفة أعداد صحيحة (INTEGER array)، ذات أبعاد (min(M,N))
!>          مؤشرات التقسيم؛ لـ 1 <= i <= min(M,N)، تم تبديل الصف i من
!>          المصفوفة مع الصف IPIV(i).
!>

INFO

!>          INFO هو عدد صحيح (INTEGER)
!>          = 0: خروج ناجح
!>          < 0: إذا كان INFO = -i، فإن الوسيط i كان له قيمة غير قانونية
!>          > 0: إذا كان INFO = +i، فإن U(i,i) يساوي صفرًا تمامًا. تم
!>               إكمال التفكيك، ولكن العامل U هو مفرد تمامًا، وسوف يحدث
!>               القسمة على صفر إذا تم استخدامه لحل نظام معادلات.
!>

تفاصيل إضافية:

!>
!>  يتم توضيح مخطط تخزين النطاق بالمثال التالي، عندما
!>  M = N = 6، KL = 2، KU = 1:
!>
!>  عند الإدخال:                       عند الخروج:
!>
!>      *    *    *    +    +    +       *    *    *   u14  u25  u36
!>      *    *    +    +    +    +       *    *   u13  u24  u35  u46
!>      *   a12  a23  a34  a45  a56      *   u12  u23  u34  u45  u56
!>     a11  a22  a33  a44  a55  a66     u11  u22  u33  u44  u55  u66
!>     a21  a32  a43  a54  a65   *      m21  m32  m43  m54  m65   *
!>     a31  a42  a53  a64   *    *      m31  m42  m53  m64   *    *
!>
!>  العناصر المحددة بالعلامة * لا تستخدمها الروتين؛ العناصر المحددة
!>  بالعلامة + لا تحتاج إلى تعيين عند الإدخال، ولكنها مطلوبة بواسطة الروتين لتخزين
!>  عناصر U، بسبب الملء الناتج عن تبديلات الصفوف.
!>
