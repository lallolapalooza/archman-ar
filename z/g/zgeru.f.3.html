subroutine zgeru (integer m, integer n, complex*16 alpha, complex*16, dimension(*) x, integer incx, complex*16, dimension(*) y, integer incy, complex*16, dimension(lda,*) a, integer lda)

ZGERU

الغرض:

!>
!> ZGERU  تقوم بعمل الرتبة 1
!>
!>    A := alpha*x*y**T + A,
!>
!> حيث alpha عدد قياسي، x متجه من m عنصر، y متجه من n عنصر
!> و A مصفوفة m في n.
!>

المعلمات

M 
!>          M هو عدد صحيح (INTEGER)
!>           عند الإدخال، يحدد M عدد الصفوف في المصفوفة A.
!>           يجب أن يكون M على الأقل صفرًا.
!>

N

!>          N هو عدد صحيح (INTEGER)
!>           عند الإدخال، يحدد N عدد الأعمدة في المصفوفة A.
!>           يجب أن يكون N على الأقل صفرًا.
!>

ALPHA

!>          ALPHA هو عدد مركب من 16 بت (COMPLEX*16)
!>           عند الإدخال، يحدد ALPHA العدد القياسي alpha.
!>

X

!>          X هو مصفوفة من أعداد مركبة من 16 بت (COMPLEX*16)، حجمها على الأقل
!>           ( 1 + ( m - 1 )*abs( INCX ) ).
!>           قبل الإدخال، يجب أن تحتوي المصفوفة X المزادة على متجه x ذي m عنصر.
!>

INCX

!>          INCX هو عدد صحيح (INTEGER)
!>           عند الإدخال، يحدد INCX الزيادة للعناصر في X.
!>           يجب ألا يكون INCX صفرًا.
!>

Y

!>          Y هو مصفوفة من أعداد مركبة من 16 بت (COMPLEX*16)، حجمها على الأقل
!>           ( 1 + ( n - 1 )*abs( INCY ) ).
!>           قبل الإدخال، يجب أن تحتوي المصفوفة Y المزادة على متجه y ذي n عنصر.
!>

INCY

!>          INCY هو عدد صحيح (INTEGER)
!>           عند الإدخال، يحدد INCY الزيادة للعناصر في Y.
!>           يجب ألا يكون INCY صفرًا.
!>

A

!>          A هي مصفوفة من أعداد مركبة من 16 بت (COMPLEX*16)، حجمها ( LDA, N )
!>           قبل الإدخال، يجب أن يحتوي الجزء الأمامي من المصفوفة A بحجم m في n على مصفوفة المعاملات.
!>           عند الخروج، يتم استبدال A بالمصفوفة المحدثة.
!>

LDA

!>          LDA هو عدد صحيح (INTEGER)
!>           عند الإدخال، يحدد LDA البعد الأول لـ A كما هو معلن في البرنامج (الفرعي) المستدعي.
!>           يجب أن يكون LDA على الأقل max( 1, m ).
!>
