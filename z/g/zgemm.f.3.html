subroutine zgemm (character transa, character transb, integer m, integer n, integer k, complex*16 alpha, complex*16, dimension(lda,*) a, integer lda, complex*16, dimension(ldb,*) b, integer ldb, complex*16 beta, complex*16, dimension(ldc,*) c, integer ldc)

ZGEMM

الغرض:

!>
!> ZGEMM  تقوم بإحدى عمليات ضرب المصفوفات-المصفوفات
!>
!>    C := alpha*op( A )*op( B ) + beta*C,
!>
!> حيث  op( X ) هي إحدى هذه العمليات:
!>
!>    op( X ) = X   أو   op( X ) = X**T   أو   op( X ) = X**H,
!>
!> alpha و beta هما عددان قياسيان، و A، B، و C هي مصفوفات، حيث op( A )
!> مصفوفة من حجم m × k، و op( B ) مصفوفة من حجم k × n، و C مصفوفة من حجم m × n.
!>

المعلمات

TRANSA 
!>          TRANSA هو CHARACTER*1
!>           عند الإدخال، تحدد TRANSA شكل op( A ) المستخدمة في
!>           ضرب المصفوفات كما يلي:
!>
!>              TRANSA = 'N' أو 'n'،  op( A ) = A.
!>
!>              TRANSA = 'T' أو 't'،  op( A ) = A**T.
!>
!>              TRANSA = 'C' أو 'c'،  op( A ) = A**H.
!>

TRANSB

!>          TRANSB هو CHARACTER*1
!>           عند الإدخال، تحدد TRANSB شكل op( B ) المستخدمة في
!>           ضرب المصفوفات كما يلي:
!>
!>              TRANSB = 'N' أو 'n'،  op( B ) = B.
!>
!>              TRANSB = 'T' أو 't'،  op( B ) = B**T.
!>
!>              TRANSB = 'C' أو 'c'،  op( B ) = B**H.
!>

M

!>          M هو INTEGER
!>           عند الإدخال، يحدد  M  عدد صفوف المصفوفة
!>           op( A ) والمصفوفة  C.  يجب أن يكون  M  على الأقل صفر.
!>

N

!>          N هو INTEGER
!>           عند الإدخال، يحدد  N  عدد أعمدة المصفوفة
!>           op( B ) وعدد أعمدة المصفوفة C. يجب أن يكون
!>           N على الأقل صفر.
!>

K

!>          K هو INTEGER
!>           عند الإدخال، يحدد  K  عدد أعمدة المصفوفة
!>           op( A ) وعدد صفوف المصفوفة op( B ). يجب أن يكون
!>           K على الأقل صفر.
!>

ALPHA

!>          ALPHA هو COMPLEX*16
!>           عند الإدخال، يحدد ALPHA العدد القياسي alpha.
!>

A

!>          A هو مصفوفة COMPLEX*16، حجم ( LDA, ka )، حيث ka هي
!>           k  عندما تكون  TRANSA = 'N' أو 'n'،  و هي  m  في الحالات الأخرى.
!>           قبل الدخول مع  TRANSA = 'N' أو 'n'،  يجب أن يحتوي الجزء الرئيسي m × k
!>           من المصفوفة  A  على المصفوفة  A،  وخلاف ذلك
!>           يجب أن يحتوي الجزء الرئيسي k × m  من المصفوفة  A  على
!>           المصفوفة A.
!>

LDA

!>          LDA هو INTEGER
!>           عند الإدخال، يحدد LDA البعد الأول للمصفوفة A كما هو معلن
!>           في البرنامج (الفرعي) المستدعي. عندما تكون  TRANSA = 'N' أو 'n'
!>           يجب أن يكون LDA على الأقل  max( 1, m )، وإلا يجب أن يكون LDA
!>           على الأقل  max( 1, k ).
!>

B

!>          B هو مصفوفة COMPLEX*16، حجم ( LDB, kb )، حيث kb هي
!>           n  عندما تكون  TRANSB = 'N' أو 'n'،  و هي  k  في الحالات الأخرى.
!>           قبل الدخول مع  TRANSB = 'N' أو 'n'،  يجب أن يحتوي الجزء الرئيسي k × n
!>           من المصفوفة  B  على المصفوفة  B،  وخلاف ذلك
!>           يجب أن يحتوي الجزء الرئيسي n × k  من المصفوفة  B  على
!>           المصفوفة B.
!>

LDB

!>          LDB هو INTEGER
!>           عند الإدخال، يحدد LDB البعد الأول للمصفوفة B كما هو معلن
!>           في البرنامج (الفرعي) المستدعي. عندما تكون  TRANSB = 'N' أو 'n'
!>           يجب أن يكون LDB على الأقل  max( 1, k )، وإلا يجب أن يكون LDB
!>           على الأقل  max( 1, n ).
!>

BETA

!>          BETA هو COMPLEX*16
!>           عند الإدخال، يحدد  BETA  العدد القياسي  beta.  عندما يتم
!>           تزويد  BETA  بالقيمة صفر، فلا حاجة لتعيين C عند الإدخال.
!>

C

!>          C هو مصفوفة COMPLEX*16، حجم ( LDC, N )
!>           قبل الدخول، يجب أن يحتوي الجزء الرئيسي m × n
!>           من المصفوفة  C  على المصفوفة  C،  باستثناء عندما يكون  beta  صفرًا،
!>           في هذه الحالة لا يلزم تعيين C عند الإدخال.
!>           عند الخروج، يتم استبدال المصفوفة  C  بالمصفوفة  m × n
!>           ( alpha*op( A )*op( B ) + beta*C ).
!>

LDC

!>          LDC هو INTEGER
!>           عند الإدخال، يحدد LDC البعد الأول للمصفوفة C كما هو معلن
!>           في البرنامج (الفرعي) المستدعي.  يجب أن يكون  LDC  على الأقل
!>           max( 1, m ).
!>
