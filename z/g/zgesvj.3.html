توثيق الوظيفة/الإجراء الفرعي

subroutine zgesvj (character*1 joba, character*1 jobu, character*1 jobv, integer m, integer n, complex*16, dimension( lda, * ) a, integer lda, double precision, dimension( n ) sva, integer mv, complex*16, dimension( ldv, * ) v, integer ldv, complex*16, dimension( lwork ) cwork, integer lwork, double precision, dimension( lrwork ) rwork, integer lrwork, integer info)

ZGESVJ

الغرض:

!>
!> ZGESVJ تحسب التحلل القيمي المنفرد (SVD) للمصفوفة المعقدة A بحجم M-by-N، حيث M >= N.
!> التحلل القيمي المنفرد لـ A يُكتب كـ
!>                                    [++]   [xx]   [x0]   [xx]
!>              A = U * SIGMA * V^*,  [++] = [xx] * [ox] * [xx]
!>                                    [++]   [xx]
!> حيث SIGMA هي مصفوفة قطرية بحجم N-by-N، U هي مصفوفة متعامدة بحجم M-by-N،
!> و V هي مصفوفة وحدوية بحجم N-by-N. العناصر القطرية لـ SIGMA هي القيم المنفردة لـ A.
!> الأعمدة في U و V هي المتجهات المنفردة اليسرى واليمنى لـ A على التوالي.
!>

المعلمات:

JOBA 
!>          JOBA هو CHARACTER*1
!>          يحدد بنية المصفوفة A.
!>          = 'L': المصفوفة المدخلة A هي مصفوفة مثلثية سفلية؛
!>          = 'U': المصفوفة المدخلة A هي مصفوفة مثلثية علوية؛
!>          = 'G': المصفوفة المدخلة A هي مصفوفة عامة بحجم M-by-N، M >= N.
!>

JOBU

!>          JOBU هو CHARACTER*1
!>          يحدد ما إذا كان سيتم حساب المتجهات المنفردة اليسرى
!>          (أعمدة U):
!>          = 'U' أو 'F': سيتم حساب المتجهات المنفردة اليسرى المقابلة للقيم
!>                 المنفردة غير الصفرية وإرجاعها في الأعمدة الرائدة للمصفوفة A.
!>                 انظر المزيد من التفاصيل في وصف A.
!>                 يتم تعيين الحد الافتراضي للتعامد الرقمي إلى
!>                 حوالي TOL=CTOL*EPS، CTOL=SQRT(M)، EPS=DLAMCH('E').
!>          = 'C': مماثل لـ JOBU='U'، باستثناء أنه يمكن للمستخدم التحكم في
!>                 مستوى التعامد الرقمي للمتجهات المنفردة اليسرى المحسوبة.
!>                 يمكن تعيين TOL إلى TOL = CTOL*EPS، حيث CTOL مُدخلة في المصفوفة WORK.
!>                 لا يُسمح بقيمة CTOL أصغر من ONE. قيمة CTOL أكبر
!>                 من 1 / EPS لا معنى لها. يمكن استخدام الخيار 'C'
!>                 إذا كان M*EPS متعامدًا بما فيه الكفاية للمتجهات المنفردة اليسرى المحسوبة،
!>                 لذا يمكن لـ CTOL=M توفير بضع دورات من دورانات يعقوبي.
!>                 راجع وصف A و WORK(1).
!>          = 'N': المصفوفة U غير محسوبة. ومع ذلك، انظر وصف A.
!>

JOBV

!>          JOBV هو CHARACTER*1
!>          يحدد ما إذا كان سيتم حساب المتجهات المنفردة اليمنى، أي
!>          المصفوفة V:
!>          = 'V' أو 'J': سيتم حساب المصفوفة V وإرجاعها في المصفوفة V
!>          = 'A': سيتم تطبيق دورانات يعقوبي على مصفوفة MV-by-N
!>                 المخزنة مبدئيًا في الصفوف MV الأولى من V.
!>          = 'N': المصفوفة V غير محسوبة ولا يتم الرجوع إلى المصفوفة V.
!>

M

!>          M هو INTEGER
!>          عدد الصفوف في المصفوفة المدخلة A. 1/DLAMCH('E') > M >= 0.
!>

N

!>          N هو INTEGER
!>          عدد الأعمدة في المصفوفة المدخلة A.
!>          M >= N >= 0.
!>

A

!>          A هو مصفوفة COMPLEX*16، بأبعاد (LDA,N)
!>          عند الدخول، المصفوفة A بحجم M-by-N.
!>          عند الخروج،
!>          إذا كان JOBU = 'U' أو JOBU = 'C':
!>                 إذا كان INFO = 0:
!>                 سيتم إرجاع الأعمدة المتعامدة RANKA من U في
!>                 الأعمدة الرائدة RANKA للمصفوفة A. هنا RANKA <= N
!>                 هو عدد القيم المنفردة المحسوبة لـ A التي
!>                 تتجاوز عتبة الانهيار السفلي DLAMCH('S'). المتجهات المنفردة
!>                 المقابلة للقيم المنفردة المنهارة أو الصفرية
!>                 غير محسوبة. سيتم إرجاع قيمة RANKA
!>                 في المصفوفة RWORK كـ RANKA=NINT(RWORK(2)). انظر أيضًا
!>                 أوصاف SVA و RWORK. الأعمدة المحسوبة من U
!>                 متعامدة رقميًا فيما بينها حتى ما يقرب من
!>                 TOL=SQRT(M)*EPS (افتراضي)؛ أو TOL=CTOL*EPS (JOBU = 'C')،
!>                 راجع وصف JOBU.
!>                 إذا كان INFO > 0،
!>                 فإن الإجراء ZGESVJ لم يتقارب في العدد المحدد
!>                 من التكرارات (الدورات). في هذه الحالة، قد لا تكون
!>                 الأعمدة المحسوبة لـ U متعامدة حتى TOL.
!>                 المخرجات U (مخزنة في A)، SIGMA (المعطاة بالقيم المنفردة
!>                 المحسوبة في SVA(1:N)) و V لا تزال تمثل تحللًا للمدخلات
!>                 A بالمعنى أن الباقي
!>                 || A - SCALE * U * SIGMA * V^* ||_2 / ||A||_2 صغير.
!>          إذا كان JOBU = 'N':
!>                 إذا كان INFO = 0:
!>                 لاحظ أن المتجهات المنفردة اليسرى مجانية في
!>                 خوارزمية SVD ذات الجانب الواحد لـ Jacobi. ومع ذلك، إذا
!>                 كانت القيم المنفردة فقط هي المطلوبة، فإن مستوى التعامد
!>                 الرقمي لـ U ليس قضية ويتم إيقاف التكرارات
!>                 عندما تكون أعمدة المصفوفة المكررة
!>                 متعامدة رقميًا حتى ما يقرب من M*EPS. وبالتالي،
!>                 عند الخروج، تحتوي A على أعمدة U مضروبة في
!>                 القيم المنفردة المقابلة.
!>                 إذا كان INFO > 0:
!>                 فإن الإجراء ZGESVJ لم يتقارب في العدد المحدد
!>                 من التكرارات (الدورات).
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرائد للمصفوفة A. LDA >= max(1,M).
!>

SVA

!>          SVA هي مصفوفة DOUBLE PRECISION، بأبعاد (N)
!>          عند الخروج،
!>          إذا كان INFO = 0:
!>          اعتمادًا على قيمة SCALE = RWORK(1)، لدينا:
!>                 إذا كان SCALE = ONE:
!>                 تحتوي SVA(1:N) على القيم المنفردة المحسوبة لـ A.
!>                 أثناء الحساب، تحتوي SVA على معايير الأعمدة الإقليدية
!>                 للمصفوفات المكررة في المصفوفة A.
!>                 إذا كان SCALE .NE. ONE:
!>                 القيم المنفردة لـ A هي SCALE*SVA(1:N)، وهذه
!>                 التمثيل المأخوذ هو بسبب حقيقة أن بعض
!>                 القيم المنفردة لـ A قد تنهار سفليًا أو تفيض.
!>
!>          إذا كان INFO > 0:
!>          فإن الإجراء ZGESVJ لم يتقارب في العدد المحدد
!>          من التكرارات (الدورات) و SCALE*SVA(1:N) قد لا تكون دقيقة.
!>

MV

!>          MV هو INTEGER
!>          إذا كان JOBV = 'A'، فسيتم تطبيق حاصل ضرب دورانات يعقوبي في ZGESVJ
!>          على الصفوف MV الأولى من V. انظر وصف JOBV.
!>

V

!>          V هي مصفوفة COMPLEX*16، بأبعاد (LDV,N)
!>          إذا كان JOBV = 'V'، فإن V تحتوي عند الخروج على المصفوفة N-by-N
!>                         للمتجهات المنفردة اليمنى؛
!>          إذا كان JOBV = 'A'، فإن V تحتوي على حاصل ضرب مصفوفة المتجهات المنفردة
!>                         المحسوبة والمصفوفة الأولية
!>                         في المصفوفة V.
!>          إذا كان JOBV = 'N'، فلا يتم الرجوع إلى V.
!>

LDV

!>          LDV هو INTEGER
!>          البعد الرائد
