subroutine recursive zgetrf2 (integer m, integer n, complex*16, dimension( lda, * ) a, integer lda, integer, dimension( * ) ipiv, integer info)

ZGETRF2

الغرض:

!>
!> ZGETRF2 يحسب تحليل LU لمصفوفة عامة M-by-N تسمى A
!> باستخدام تقنية التحريك الجزئي مع تبديلات الصفوف.
!>
!> التحليل له الشكل
!>    A = P * L * U
!> حيث P هي مصفوفة تبديل، L هي مصفوفة مثلثة سفلى ذات عناصر قطرية موحدة
!> (شبه منحرف سفلي إذا كان m > n)، و U هي مصفوفة مثلثة عليا
!> (شبه منحرف علوي إذا كان m < n).
!>
!> هذه هي النسخة التكرارية للخوارزمية. تقوم بتقسيم
!> المصفوفة إلى أربع مصفوفات فرعية:
!>
!>        [  A11 | A12  ]  حيث A11 بحجم n1 × n1 و A22 بحجم n2 × n2
!>    A = [ -----|----- ]  مع n1 = min(m,n)/2
!>        [  A21 | A22  ]       n2 = n-n1
!>
!>                                       [ A11 ]
!> يستدعي الروتين الفرعي نفسه لتحليل [ --- ]،
!>                                       [ A12 ]
!>                 [ A12 ]
!> يقوم بالتبديلات على [ --- ]، يحل A12، يحدث A22،
!>                 [ A22 ]
!>
!> ثم يستدعي نفسه لتحليل A22 والقيام بالتبديلات على A21.
!>
!>

المعلمات

M 
!>          M هو عدد صحيح (INTEGER)
!>          عدد صفوف المصفوفة A. M >= 0.
!>

N

!>          N هو عدد صحيح (INTEGER)
!>          عدد أعمدة المصفوفة A. N >= 0.
!>

A

!>          A هي مصفوفة من نوع COMPLEX*16، أبعادها (LDA,N)
!>          عند الإدخال، المصفوفة M-by-N المراد تحليلها.
!>          عند الخروج، عوامل L و U من التحليل
!>          A = P*L*U؛ لا يتم تخزين عناصر القطر الموحدة لـ L.
!>

LDA

!>          LDA هو عدد صحيح (INTEGER)
!>          البعد الرئيسي للمصفوفة A. LDA >= max(1,M).
!>

IPIV

!>          IPIV هي مصفوفة أعداد صحيحة (INTEGER)، أبعادها (min(M,N))
!>          مؤشرات المحاور؛ بالنسبة لـ 1 <= i <= min(M,N)، تم تبديل الصف i من
!>          المصفوفة مع الصف IPIV(i).
!>

INFO

!>          INFO هو عدد صحيح (INTEGER)
!>          = 0:  خروج ناجح
!>          < 0:  إذا كان INFO = -i، فإن المعامل i كان له قيمة غير قانونية
!>          > 0:  إذا كان INFO = i، فإن U(i,i) يساوي صفر بالضبط. تم
!>                إكمال التحليل، ولكن العامل U فريد تمامًا،
!>                وسينتج قسمة على صفر إذا تم استخدامه
!>                لحل نظام معادلات.
!>
