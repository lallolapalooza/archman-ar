subroutine zget51 (integer itype, integer n, complex*16, dimension( lda, * ) a, integer lda, complex*16, dimension( ldb, * ) b, integer ldb, complex*16, dimension( ldu, * ) u, integer ldu, complex*16, dimension( ldv, * ) v, integer ldv, complex*16, dimension( * ) work, double precision, dimension( * ) rwork, double precision result)

ZGET51

الغرض:

!>
!>      ZGET51  بشكل عام تتحقق من تحليل للشكل
!>
!>              A = U B V**H
!>
!>      حيث **H تعني المرافق المتحول و U و V وحدويتان.
!>
!>      على وجه التحديد، إذا كانت ITYPE=1
!>
!>              RESULT = | A - U B V**H | / ( |A| n ulp )
!>
!>      إذا كانت ITYPE=2، فإن:
!>
!>              RESULT = | A - B | / ( |A| n ulp )
!>
!>      إذا كانت ITYPE=3، فإن:
!>
!>              RESULT = | I - U U**H | / ( n ulp )
!>

المعلمات

ITYPE 
!>          ITYPE هو عدد صحيح (INTEGER)
!>          يحدد نوع الاختبارات التي سيتم إجراؤها.
!>          =1: RESULT = | A - U B V**H | / ( |A| n ulp )
!>          =2: RESULT = | A - B | / ( |A| n ulp )
!>          =3: RESULT = | I - U U**H | / ( n ulp )
!>

N

!>          N هو عدد صحيح (INTEGER)
!>          حجم المصفوفة. إذا كان صفرًا، فإن ZGET51 لا تفعل شيئًا.
!>          يجب أن يكون على الأقل صفرًا.
!>

A

!>          A هي مصفوفة معقدة*16 (COMPLEX*16)، ذات بعد (LDA, N)
!>          المصفوفة الأصلية (غير المفككة).
!>

LDA

!>          LDA هو عدد صحيح (INTEGER)
!>          البعد الرائد للمصفوفة A. يجب أن يكون على الأقل 1
!>          وعلى الأقل N.
!>

B

!>          B هي مصفوفة معقدة*16 (COMPLEX*16)، ذات بعد (LDB, N)
!>          المصفوفة المفككة.
!>

LDB

!>          LDB هو عدد صحيح (INTEGER)
!>          البعد الرائد للمصفوفة B. يجب أن يكون على الأقل 1
!>          وعلى الأقل N.
!>

U

!>          U هي مصفوفة معقدة*16 (COMPLEX*16)، ذات بعد (LDU, N)
!>          المصفوفة الوحدوية على الجانب الأيسر في
!>          التحليل.
!>          غير مستخدمة إذا كانت ITYPE=2
!>

LDU

!>          LDU هو عدد صحيح (INTEGER)
!>          البعد الرائد للمصفوفة U. يجب أن يكون LDU على الأقل N و
!>          على الأقل 1.
!>

V

!>          V هي مصفوفة معقدة*16 (COMPLEX*16)، ذات بعد (LDV, N)
!>          المصفوفة الوحدوية على الجانب الأيسر في
!>          التحليل.
!>          غير مستخدمة إذا كانت ITYPE=2
!>

LDV

!>          LDV هو عدد صحيح (INTEGER)
!>          البعد الرائد للمصفوفة V. يجب أن يكون LDV على الأقل N و
!>          على الأقل 1.
!>

WORK

!>          WORK هي مصفوفة معقدة*16 (COMPLEX*16)، ذات بعد (2*N**2)
!>

RWORK

!>          RWORK هي مصفوفة ذات دقة مزدوجة (DOUBLE PRECISION)، ذات بعد (N)
!>

RESULT

!>          RESULT هي دقة مزدوجة (DOUBLE PRECISION)
!>          القيم التي تم حسابها بواسطة الاختبار المحدد بواسطة ITYPE.
!>          القيمة محدودة حاليًا بـ 1/ulp، لتجنب تجاوز السعة.
!>          يتم الإبلاغ عن الأخطاء بواسطة RESULT=10/ulp.
!>
