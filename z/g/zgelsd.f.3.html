subroutine zgelsd (integer m, integer n, integer nrhs, complex*16, dimension( lda, * ) a, integer lda, complex*16, dimension( ldb, * ) b, integer ldb, double precision, dimension( * ) s, double precision rcond, integer rank, complex*16, dimension( * ) work, integer lwork, double precision, dimension( * ) rwork, integer, dimension( * ) iwork, integer info)

ZGELSD تحسب الحل بأقل معيار لمشكلة المربعات الصغرى لـ GE Matrices

الغرض:

!>
!> ZGELSD تحسب الحل بأقل معيار لمشكلة المربعات الصغرى الحقيقية:
!>     تصغير المعيار 2 لـ | b - A*x |
!> باستخدام تفكيك القيم المفردة (SVD) لـ A. A هي مصفوفة M-by-N
!> قد تكون ناقصة الرتبة.
!>
!> يمكن معالجة عدة متجهات للجانب الأيمن b ومتجهات الحل x
!> في استدعاء واحد؛ يتم تخزينها كأعمدة للمصفوفة اليمنى B
!> ذات الجانب الأيمن M-by-NRHS ومصفوفة الحل X
!> ذات الجانب الأيمن N-by-NRHS.
!>
!> يتم حل المشكلة في ثلاث خطوات:
!> (1) تقليل المصفوفة المعاملة A إلى شكل ثنائي القطر باستخدام
!>     تحويلات هاوسهولدر، مما يقلل المشكلة الأصلية
!>     إلى (BLS)
!> (2) حل BLS باستخدام نهج التقسيم والغزو.
!> (3) تطبيق جميع تحويلات هاوسهولدر إلى الخلف لحل
!>     مشكلة المربعات الصغرى الأصلية.
!>
!> يتم تحديد الرتبة الفعالة لـ A بمعاملة القيم المفردة
!> الأقل من RCOND مضروبة في أكبر قيمة مفردة على أنها صفر.
!>
!>

المعلمات

M 
!>          M هو عدد صحيح (INTEGER)
!>          عدد الصفوف في المصفوفة A. M >= 0.
!>

N

!>          N هو عدد صحيح (INTEGER)
!>          عدد الأعمدة في المصفوفة A. N >= 0.
!>

NRHS

!>          NRHS هو عدد صحيح (INTEGER)
!>          عدد الأجناب اليمنى، أي عدد أعمدة
!>          المصفوفتين B و X. NRHS >= 0.
!>

A

!>          A هي مصفوفة COMPLEX*16، البعد (LDA,N)
!>          عند الإدخال، المصفوفة A ذات الأبعاد M-by-N.
!>          عند الخروج، تم تدمير A.
!>

LDA

!>          LDA هو عدد صحيح (INTEGER)
!>          البعد الرئيسي للمصفوفة A. LDA >= max(1,M).
!>

B

!>          B هي مصفوفة COMPLEX*16، البعد (LDB,NRHS)
!>          عند الإدخال، مصفوفة الجانب الأيمن B ذات الأبعاد M-by-NRHS.
!>          عند الخروج، يتم استبدال B بمصفوفة الحل X ذات الأبعاد N-by-NRHS.
!>          إذا كانت m >= n و RANK = n، فإن مجموع مربعات البواقي
!>          للحل في العمود i يُعطى بمجموع
!>          مربعات قيم المطلقة للعناصر n+1:m في ذلك العمود.
!>

LDB

!>          LDB هو عدد صحيح (INTEGER)
!>          البعد الرئيسي للمصفوفة B. LDB >= max(1,M,N).
!>

S

!>          S هي مصفوفة DOUBLE PRECISION، البعد (min(M,N))
!>          القيم المفردة لـ A بترتيب تنازلي.
!>          عدد الشرط لـ A بالنسبة للمعيار 2 = S(1)/S(min(m,n)).
!>

RCOND

!>          RCOND هي DOUBLE PRECISION
!>          يُستخدم RCOND لتحديد الرتبة الفعالة لـ A.
!>          تُعامل القيم المفردة S(i) <= RCOND*S(1) على أنها صفر.
!>          إذا كان RCOND < 0، يتم استخدام الدقة الآلية بدلاً من ذلك.
!>

RANK

!>          RANK هو عدد صحيح (INTEGER)
!>          الرتبة الفعالة لـ A، أي عدد القيم المفردة
!>          الأكبر من RCOND*S(1).
!>

WORK

!>          WORK هي مصفوفة COMPLEX*16، البعد (MAX(1,LWORK))
!>          عند الخروج، إذا كان INFO = 0، فإن WORK(1) يعيد LWORK المثلى.
!>

LWORK

!>          LWORK هو عدد صحيح (INTEGER)
!>          بعد المصفوفة WORK. يجب أن يكون LWORK على الأقل 1.
!>          يعتمد الحد الأدنى الدقيق لمقدار مساحة العمل المطلوبة على M،
!>          N و NRHS. طالما أن LWORK على الأقل
!>              2*N + N*NRHS
!>          إذا كانت M أكبر من أو تساوي N أو
!>              2*M + M*NRHS
!>          إذا كانت M أقل من N، ستعمل الوحدة بشكل صحيح.
!>          لأداء جيد، يجب أن يكون LWORK بشكل عام أكبر.
!>
!>          إذا كان LWORK = -1، فإنه يُفترض استعلام عن مساحة العمل؛ الوحدة
!>          تحسب فقط الحجم الأمثل للمصفوفة WORK
!>          والأحجام الدنيا للمصفوفتين RWORK و IWORK، وتُعيد
!>          هذه القيم كأول إدخالات في مصفوفات WORK و RWORK و
!>          IWORK، ولا يتم إصدار أي رسالة خطأ متعلقة بـ LWORK
!>          بواسطة XERBLA.
!>

RWORK

!>          RWORK هي مصفوفة DOUBLE PRECISION، البعد (MAX(1,LRWORK))
!>          LRWORK >=
!>             10*N + 2*N*SMLSIZ + 8*N*NLVL + 3*SMLSIZ*NRHS +
!>             MAX( (SMLSIZ+1)**2, N*(1+NRHS) + 2*NRHS )
!>          إذا كانت M أكبر من أو تساوي N أو
!>             10*M + 2*M*SMLSIZ + 8*M*NLVL + 3*SMLSIZ*NRHS +
!>             MAX( (SMLSIZ+1)**2, N*(1+NRHS) + 2*NRHS )
!>          إذا كانت M أقل من N، ستعمل الوحدة بشكل صحيح.
!>          SMLSIZ تُعاد بواسطة ILAENV وهي تساوي الحد الأقصى
!>          لحجم المشاكل الفرعية في أسفل شجرة الحساب
!>          (عادة حوالي 25)، و
!>             NLVL = MAX( 0, INT( LOG_2( MIN( M,N )/(SMLSIZ+1) ) ) + 1 )
!>          عند الخروج، إذا كان INFO = 0، فإن RWORK(1) يعيد LRWORK الأدنى.
!>

IWORK

!>          IWORK هي مصفوفة INTEGER، البعد (MAX(1,LIWORK))
!>          LIWORK >= max(1, 3*MINMN*NLVL + 11*MINMN)،
!>          حيث MINMN = MIN( M,N ).
!>          عند الخروج، إذا كان INFO = 0، فإن IWORK(1) يعيد LIWORK الأدنى.
!>

INFO

!>          INFO هو عدد صحيح (INTEGER)
!>          = 0: انتهاء ناجح
!>          < 0: إذا كان INFO = -i، فإن المعلمة i كانت ذات قيمة غير قانونية.
!>          > 0: فشل خوارزمية حساب SVD في التقارب؛
!>                إذا كان INFO = i، فإن i عنصرًا خارج القطر لمصفوفة ثنائية القطر وسيطة
!>                لم تتقارب إلى الصفر.
!>
