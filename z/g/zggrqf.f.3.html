subroutine zggrqf (integer m, integer p, integer n, complex*16, dimension( lda, * ) a, integer lda, complex*16, dimension( * ) taua, complex*16, dimension( ldb, * ) b, integer ldb, complex*16, dimension( * ) taub, complex*16, dimension( * ) work, integer lwork, integer info)

ZGGRQF

الغرض:

!>
!> ZGGRQF يحسب تحليل RQ معمماً للمصفوفة M-by-N A
!> والمصفوفة P-by-N B:
!>
!>             A = R*Q,        B = Z*T*Q,
!>
!> حيث Q مصفوفة وحدة N-by-N، Z مصفوفة وحدة P-by-P، و R و T تفترض أحد الأشكال:
!>
!> إذا كان M <= N،  R = ( 0  R12 ) M،   أو إذا كان M > N،  R = ( R11 ) M-N،
!>                  N-M  M                           ( R21 ) N
!>                                                      N
!>
!> حيث R12 أو R21 مثلثة عليا، و
!>
!> إذا كان P >= N،  T = ( T11 ) N  ،   أو إذا كان P < N،  T = ( T11  T12 ) P،
!>                 (  0  ) P-N                         P   N-P
!>                    N
!>
!> حيث T11 مثلثة عليا.
!>
!> بشكل خاص، إذا كانت B مربعة وغير منفردة، فإن تحليل GRQ
!> للمصفوفة A و B يعطي ضمنياً تحليل RQ للمصفوفة A*inv(B):
!>
!>              A*inv(B) = (R*inv(T))*Z**H
!>
!> حيث inv(B) تعني معكوس المصفوفة B، و Z**H تعني منقول المصفوفة Z المترافقة.
!>

المعاملات:

M 
!>          M هو عدد صحيح (INTEGER)
!>          عدد صفوف المصفوفة A. M >= 0.
!>

P

!>          P هو عدد صحيح (INTEGER)
!>          عدد صفوف المصفوفة B. P >= 0.
!>

N

!>          N هو عدد صحيح (INTEGER)
!>          عدد أعمدة المصفوفات A و B. N >= 0.
!>

A

!>          A هو مصفوفة معقدة*16 (COMPLEX*16)، أبعادها (LDA,N)
!>          عند الدخول، المصفوفة A بالمقاس M-by-N.
!>          عند الخروج، إذا كان M <= N، فإن الجزء المثلثي العلوي من المصفوفة الفرعية
!>          A(1:M,N-M+1:N) يحتوي على المصفوفة المثلثية العليا R بالمقاس M-by-M؛
!>          إذا كان M > N، فإن العناصر على وتحت القطر الفرعي (M-N)
!>          تحتوي على المصفوفة شبه المنحرفة العليا R بالمقاس M-by-N؛ العناصر المتبقية،
!>          مع المصفوفة TAUA، تمثل المصفوفة الوحدة Q كناتج لمُنعكسات أولية (انظر التفاصيل الإضافية).
!>

LDA

!>          LDA هو عدد صحيح (INTEGER)
!>          البعد الرائد للمصفوفة A. LDA >= max(1,M).
!>

TAUA

!>          TAUA هي مصفوفة معقدة*16 (COMPLEX*16)، أبعادها (min(M,N))
!>          العوامل القياسية للمنعكسات الأولية التي
!>          تمثل المصفوفة الوحدة Q (انظر التفاصيل الإضافية).
!>

B

!>          B هي مصفوفة معقدة*16 (COMPLEX*16)، أبعادها (LDB,N)
!>          عند الدخول، المصفوفة B بالمقاس P-by-N.
!>          عند الخروج، العناصر على و فوق القطر الرئيسي في المصفوفة
!>          تحتوي على المصفوفة شبه المنحرفة العليا T بالمقاس min(P,N)-by-N (T مثلثة عليا
!>          إذا كان P >= N)؛ العناصر تحت القطر الرئيسي،
!>          مع المصفوفة TAUB، تمثل المصفوفة الوحدة Z كناتج لمُنعكسات أولية (انظر التفاصيل الإضافية).
!>

LDB

!>          LDB هو عدد صحيح (INTEGER)
!>          البعد الرائد للمصفوفة B. LDB >= max(1,P).
!>

TAUB

!>          TAUB هي مصفوفة معقدة*16 (COMPLEX*16)، أبعادها (min(P,N))
!>          العوامل القياسية للمنعكسات الأولية التي
!>          تمثل المصفوفة الوحدة Z (انظر التفاصيل الإضافية).
!>

WORK

!>          WORK هي مصفوفة معقدة*16 (COMPLEX*16)، أبعادها (MAX(1,LWORK))
!>          عند الخروج، إذا كان INFO = 0، فإن WORK(1) تحتوي على LWORK المثلى.
!>

LWORK

!>          LWORK هو عدد صحيح (INTEGER)
!>          بعد المصفوفة WORK. LWORK >= max(1,N,M,P).
!>          لأداء مثالي LWORK >= max(N,M,P)*max(NB1,NB2,NB3)،
!>          حيث NB1 هو الحجم المثالي للمجموعة لتحليل RQ
!>          لمصفوفة M-by-N، NB2 هو الحجم المثالي للمجموعة لتحليل QR
!>          لمصفوفة P-by-N، و NB3 هو الحجم المثالي للمجموعة لاستدعاء ZUNMRQ.
!>
!>          إذا كان LWORK = -1، فسيتم افتراض استعلام عن مساحة العمل؛ الروتين
!>          يحسب فقط الحجم الأمثل لمصفوفة WORK، ويعيد
!>          هذه القيمة كالإدخال الأول لمصفوفة WORK، ولا يصدر رسالة خطأ
!>          بخصوص LWORK بواسطة XERBLA.
!>

INFO

!>          INFO هو عدد صحيح (INTEGER)
!>          = 0:  انتهاء ناجح
!>          < 0:  إذا كان INFO=-i، فإن المعامل i كان له قيمة غير قانونية.
!>

تفاصيل إضافية:

!>
!>  تمثل المصفوفة Q كناتج لمُنعكسات أولية
!>
!>     Q = H(1) H(2) . . . H(k)، حيث k = min(m,n).
!>
!>  كل H(i) لها الشكل
!>
!>     H(i) = I - taua * v * v**H
!>
!>  حيث taua هو عامل قياسي معقد، و v هو متجه معقد مع
!>  v(n-k+i+1:n) = 0 و v(n-k+i) = 1؛ يتم تخزين v(1:n-k+i-1) عند الخروج في
!>  A(m-k+i,1:n-k+i-1)، و taua في TAUA(i).
!>  لتكوين Q بشكل صريح، استخدم الروتين الفرعي ZUNGRQ في LAPACK.
!>  لاستخدام Q لتحديث مصفوفة أخرى، استخدم الروتين الفرعي ZUNMRQ في LAPACK.
!>
!>  تمثل المصفوفة Z كناتج لمُنعكسات أولية
!>
!>     Z = H(1) H(2) . . . H(k)، حيث k = min(p,n).
!>
!>  كل H(i) لها الشكل
!>
!>     H(i) = I - taub * v * v**H
!>
!>  حيث taub هو عامل قياسي معقد، و v هو متجه معقد مع
!>  v(1:i-1) = 0 و v(i) = 1؛ يتم تخزين v(i+1:p) عند الخروج في B(i+1:p,i)،
!>  و taub في TAUB(i).
!>  لتكوين Z بشكل صريح، استخدم الروتين الفرعي ZUNGQR في LAPACK.
!>  لاستخدام Z لتحديث مصفوفة أخرى، استخدم الروتين الفرعي ZUNMQR في LAPACK.
!>
