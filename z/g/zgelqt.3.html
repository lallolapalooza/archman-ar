توثيق الوظيفة/الروتين الفرعي

subroutine zgelqt (integer m, integer n, integer mb, complex*16, dimension( lda, * ) a, integer lda, complex*16, dimension( ldt, * ) t, integer ldt, complex*16, dimension( * ) work, integer info)

ZGELQT

الغرض:

!>
!> ZGELQT يحسب عامل التحلل LQ لمصفوفة A المعقدة M-by-N
!> باستخدام تمثيل WY المضغوط لـ Q.
!>

المعلمات

M 
!>          M هو عدد صحيح (INTEGER)
!>          عدد صفوف المصفوفة A.  M >= 0.
!>

N

!>          N هو عدد صحيح (INTEGER)
!>          عدد أعمدة المصفوفة A.  N >= 0.
!>

MB

!>          MB هو عدد صحيح (INTEGER)
!>          حجم الكتلة (block size) الذي سيتم استخدامه في التحليل QR المكتل (blocked QR).  MIN(M,N) >= MB >= 1.
!>

A

!>          A هي مصفوفة (complex*16 array)، البعد (LDA,N)
!>          عند الدخول، مصفوفة A ذات الأبعاد M-by-N.
!>          عند الخروج، تحتوي العناصر على القطر وفي الأسفل من المصفوفة على المصفوفة شبه المنحرفة السفلية L ذات الأبعاد M-by-MIN(M,N) (L تكون مثلثية سفلية إذا كانت M <= N)؛ العناصر فوق القطر هي صفوف V.
!>

LDA

!>          LDA هو عدد صحيح (INTEGER)
!>          البعد الرائد لمصفوفة A.  LDA >= max(1,M).
!>

T

!>          T هي مصفوفة (complex*16 array)، البعد (LDT,MIN(M,N))
!>          عوامل التحلل المثلثية العلوية المخزنة بشكل مضغوط
!>          كسلسلة من الكتل المثلثية العلوية.  انظر أدناه
!>          لمزيد من التفاصيل.
!>

LDT

!>          LDT هو عدد صحيح (INTEGER)
!>          البعد الرائد للمصفوفة T.  LDT >= MB.
!>

WORK

!>          WORK هي مصفوفة (complex*16 array)، البعد (MB*N)
!>

INFO

!>          INFO هو عدد صحيح (INTEGER)
!>          = 0:  تم الخروج بنجاح
!>          < 0:  إذا كان INFO = -i، فقد كان المعامل i غير قانوني
!>

تفاصيل إضافية:

!>
!>  تخزن المصفوفة V عوامل التحلل الأولية H(i) في الصف i
!>  فوق القطر. على سبيل المثال، إذا كانت M=5 و N=3، فإن المصفوفة V هي
!>
!>               V = (  1  v1 v1 v1 v1 )
!>                   (     1  v2 v2 v2 )
!>                   (         1 v3 v3 )
!>
!>
!>  حيث تمثل الـ vi المتجهات التي تحدد H(i)، والتي يتم إرجاعها
!>  في المصفوفة A.  الأرقام 1 على طول قطر V لا يتم تخزينها في A.
!>  لتكن K=MIN(M,N).  عدد الكتل هو B = سقف(K/MB)، حيث كل
!>  كتلة من الرتبة MB باستثناء الكتلة الأخيرة، والتي تكون من الرتبة
!>  IB = K - (B-1)*MB.  لكل كتلة من الكتل B، يتم حساب عامل
!>  التحلل الكتلي المثلثي العلوي: T1, T2, ..., TB.  يتم تخزين T's
!>  من الرتبة MB-by-MB (و IB-by-IB للكتلة الأخيرة) في المصفوفة T
!>  ذات الأبعاد MB-by-K كـ
!>
!>               T = (T1 T2 ... TB).
!>
