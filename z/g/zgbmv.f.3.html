توثيق الوظيفة/الروتين الفرعي

subroutine zgbmv (character trans, integer m, integer n, integer kl, integer ku, complex*16 alpha, complex*16, dimension(lda,*) a, integer lda, complex*16, dimension(*) x, integer incx, complex*16 beta, complex*16, dimension(*) y, integer incy)

ZGBMV

الغرض:

!>
!> ZGBMV  تنفذ إحدى عمليات المصفوفة المتجهة التالية:
!>
!>    y := alpha*A*x + beta*y,   أو   y := alpha*A**T*x + beta*y,   أو
!>
!>    y := alpha*A**H*x + beta*y,
!>
!> حيث alpha و beta هي أعداد قياسية، x و y هي متجهات و A هي مصفوفة
!> ذات نطاق (m × n)، مع kl من الأقطار الفرعية و ku من الأقطار الفوقية.
!>

المعلمات

TRANS 
!>          TRANS هو CHARACTER*1
!>           عند الدخول، يحدد TRANS العملية التي سيتم تنفيذها كما يلي:
!>
!>              TRANS = 'N' أو 'n'   y := alpha*A*x + beta*y.
!>
!>              TRANS = 'T' أو 't'   y := alpha*A**T*x + beta*y.
!>
!>              TRANS = 'C' أو 'c'   y := alpha*A**H*x + beta*y.
!>

M

!>          M هو INTEGER
!>           عند الدخول، يحدد M عدد صفوف المصفوفة A.
!>           يجب أن تكون M أكبر من أو تساوي صفر.
!>

N

!>          N هو INTEGER
!>           عند الدخول، يحدد N عدد أعمدة المصفوفة A.
!>           يجب أن تكون N أكبر من أو تساوي صفر.
!>

KL

!>          KL هو INTEGER
!>           عند الدخول، يحدد KL عدد الأقطار الفرعية للمصفوفة A.
!>           يجب أن يحقق KL: 0 .le. KL.
!>

KU

!>          KU هو INTEGER
!>           عند الدخول، يحدد KU عدد الأقطار الفوقية للمصفوفة A.
!>           يجب أن يحقق KU: 0 .le. KU.
!>

ALPHA

!>          ALPHA هو COMPLEX*16
!>           عند الدخول، يحدد ALPHA العدد القياسي alpha.
!>

A

!>          A هو مصفوفة COMPLEX*16، ذات أبعاد ( LDA, N )
!>           قبل الدخول، يجب أن يحتوي الجزء الرائد ( kl + ku + 1 ) × N من
!>           المصفوفة A على مصفوفة المعاملات، يتم توفيرها
!>           عمودًا بعمود، مع القطر الرئيسي للمصفوفة في
!>           الصف ( ku + 1 ) من المصفوفة، ويبدأ القطر الفوقي الأول
!>           عند الموضع 2 في الصف ku، ويبدأ القطر الفرعي الأول
!>           عند الموضع 1 في الصف ( ku + 2 )، وهكذا.
!>           العناصر في المصفوفة A التي لا تتوافق مع عناصر
!>           في مصفوفة النطاق (مثل الجزء العلوي الأيسر من مثلث ku × ku)
!>           لا يتم الرجوع إليها.
!>           سيقوم مقطع البرنامج التالي بنقل مصفوفة نطاق
!>           من التخزين الكامل التقليدي إلى تخزين النطاق:
!>
!>                 DO 20, J = 1, N
!>                    K = KU + 1 - J
!>                    DO 10, I = MAX( 1, J - KU ), MIN( M, J + KL )
!>                       A( K + I, J ) = matrix( I, J )
!>              10    CONTINUE
!>              20 CONTINUE
!>

LDA

!>          LDA هو INTEGER
!>           عند الدخول، يحدد LDA البعد الأول للمصفوفة A كما تم تعريفه
!>           في البرنامج (الفرعي) المستدعي. يجب أن يكون LDA على الأقل
!>           ( kl + ku + 1 ).
!>

X

!>          X هو مصفوفة COMPLEX*16، ذات أبعاد على الأقل
!>           ( 1 + ( n - 1 )*abs( INCX ) ) عندما يكون TRANS = 'N' أو 'n'
!>           وعلى الأقل
!>           ( 1 + ( m - 1 )*abs( INCX ) ) بخلاف ذلك.
!>           قبل الدخول، يجب أن يحتوي المتجه X المحدد بزيادات على
!>           المتجه x.
!>

INCX

!>          INCX هو INTEGER
!>           عند الدخول، يحدد INCX الزيادة لعناصر
!>           X. يجب أن لا يكون INCX صفرًا.
!>

BETA

!>          BETA هو COMPLEX*16
!>           عند الدخول، يحدد BETA العدد القياسي beta. عندما يتم
!>           توفير BETA كصفر، فلا يلزم تعيين Y عند الإدخال.
!>

Y

!>          Y هو مصفوفة COMPLEX*16، ذات أبعاد على الأقل
!>           ( 1 + ( m - 1 )*abs( INCY ) ) عندما يكون TRANS = 'N' أو 'n'
!>           وعلى الأقل
!>           ( 1 + ( n - 1 )*abs( INCY ) ) بخلاف ذلك.
!>           قبل الدخول، يجب أن يحتوي المتجه Y المحدد بزيادات على
!>           المتجه y. عند الخروج، يتم الكتابة فوق Y بالمتجه y المحدث.
!>           إذا كان m أو n صفرًا، فلا يتم الرجوع إلى Y وتقوم الوظيفة
!>           بالعودة بسرعة.
!>

INCY

!>          INCY هو INTEGER
!>           عند الدخول، يحدد INCY الزيادة لعناصر
!>           Y. يجب أن لا يكون INCY صفرًا.
!>
