توثيق الوظيفة/الروتين الفرعي

subroutine zgbt02 (character trans, integer m, integer n, integer kl, integer ku, integer nrhs, complex*16, dimension( lda, * ) a, integer lda, complex*16, dimension( ldx, * ) x, integer ldx, complex*16, dimension( ldb, * ) b, integer ldb, double precision, dimension( * ) rwork, double precision resid)

ZGBT02

الغرض:

!>
!> ZGBT02 يحسب الباقي لحل نظام معادلات مصفوفة ذات نطاق
!> op(A)*X = B:
!>    RESID = norm(B - op(A)*X) / ( norm(op(A)) * norm(X) * EPS ),
!> حيث op(A) = A, A**T, أو A**H، اعتمادًا على TRANS، و EPS هو
!> رقم الآلة العائم.
!>

المعلمات

TRANS 
!>          TRANS هو CHARACTER*1
!>          يحدد شكل نظام المعادلات:
!>          = 'N':  A    * X = B  (بدون تبديل)
!>          = 'T':  A**T * X = B  (تبديل)
!>          = 'C':  A**H * X = B  (تبديل مترافق)
!>

M

!>          M هو INTEGER
!>          عدد صفوف المصفوفة A.  M >= 0.
!>

N

!>          N هو INTEGER
!>          عدد أعمدة المصفوفة A.  N >= 0.
!>

KL

!>          KL هو INTEGER
!>          عدد الأقطار الفرعية داخل نطاق A.  KL >= 0.
!>

KU

!>          KU هو INTEGER
!>          عدد الأقطار الفائقة داخل نطاق A.  KU >= 0.
!>

NRHS

!>          NRHS هو INTEGER
!>          عدد أعمدة B.  NRHS >= 0.
!>

A

!>          A هو مصفوفة COMPLEX*16، الأبعاد (LDA,N)
!>          المصفوفة الأصلية A بتخزين النطاق، مخزنة في الصفوف من 1 إلى
!>          KL+KU+1.
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرئيسي للمصفوفة A.  LDA >= max(1,KL+KU+1).
!>

X

!>          X هو مصفوفة COMPLEX*16، الأبعاد (LDX,NRHS)
!>          حلول متجهات الحل المحسوبة لنظام المعادلات الخطية.
!>

LDX

!>          LDX هو INTEGER
!>          البعد الرئيسي للمصفوفة X.  إذا كان TRANS = 'N'،
!>          LDX >= max(1,N)؛ إذا كان TRANS = 'T' أو 'C'، LDX >= max(1,M).
!>

B

!>          B هو مصفوفة COMPLEX*16، الأبعاد (LDB,NRHS)
!>          عند الإدخال، متجهات الطرف الأيمن لنظام المعادلات الخطية.
!>          عند الخروج، سيتم استبدال B بالفرق B - A*X.
!>

LDB

!>          LDB هو INTEGER
!>          البعد الرئيسي للمصفوفة B.  إذا كان TRANS = 'N'،
!>          LDB >= max(1,M)؛ إذا كان TRANS = 'T' أو 'C'، LDB >= max(1,N).
!>

RWORK

!>          RWORK هي مصفوفة DOUBLE PRECISION، الأبعاد (MAX(1,LRWORK))،
!>          حيث LRWORK >= M عندما يكون TRANS = 'T' أو 'C'؛ بخلاف ذلك،
!>          لا يتم الرجوع إلى RWORK.
!>

RESID

!>          RESID هو DOUBLE PRECISION
!>          القيمة القصوى على عدد الأطراف اليمنى لـ
!>          norm(B - op(A)*X) / ( norm(op(A)) * norm(X) * EPS ).
!>
