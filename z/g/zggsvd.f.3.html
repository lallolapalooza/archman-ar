توثيق الوظيفة/الإجراء الفرعي

subroutine zggsvd (character jobu, character jobv, character jobq, integer m, integer n, integer p, integer k, integer l, complex*16, dimension( lda, * ) a, integer lda, complex*16, dimension( ldb, * ) b, integer ldb, double precision, dimension( * ) alpha, double precision, dimension( * ) beta, complex*16, dimension( ldu, * ) u, integer ldu, complex*16, dimension( ldv, * ) v, integer ldv, complex*16, dimension( ldq, * ) q, integer ldq, complex*16, dimension( * ) work, double precision, dimension( * ) rwork, integer, dimension( * ) iwork, integer info)

ZGGSVD يحسب التحلل القيم المفردة المعمم (GSVD) للمصفوفة المعقدة A بحجم M × N والمصفوفة المعقدة B بحجم P × N:

الغرض:

!>
!> هذا الإجراء الفرعي تم إيقافه واستبداله بالإجراء الفرعي ZGGSVD3.
!>
!> ZGGSVD يحسب التحلل القيم المفردة المعمم (GSVD)
!> للمصفوفة المعقدة A بحجم M × N والمصفوفة المعقدة B بحجم P × N:
!>
!>       U**H*A*Q = D1*( 0 R )،    V**H*B*Q = D2*( 0 R )
!>
!> حيث U، V و Q هي مصفوفات وحدوية.
!> ليكن K+L = الرتبة العددية الفعالة للمصفوفة
!> (A**H,B**H)**H، فإن R هي مصفوفة مثلثية علوية غير منفردة بحجم (K+L)×(K+L)، D1 و D2 هما
!> مصفوفتان بحجم M×(K+L) و P×(K+L) على التوالي، ولها الهياكل التالية:
!>
!> إذا كان M-K-L >= 0،
!>
!>                     K  L
!>        D1 =     K ( I  0 )
!>                 L ( 0  C )
!>             M-K-L ( 0  0 )
!>
!>                   K  L
!>        D2 =   L ( 0  S )
!>             P-L ( 0  0 )
!>
!>                 N-K-L  K    L
!>   ( 0 R ) = K (  0   R11  R12 )
!>             L (  0    0   R22 )
!> حيث
!>
!>   C = diag( ALPHA(K+1)، ... ، ALPHA(K+L) )،
!>   S = diag( BETA(K+1)،  ... ، BETA(K+L) )،
!>   C**2 + S**2 = I.
!>
!>   يتم تخزين R في A(1:K+L,N-K-L+1:N) عند الخروج.
!>
!> إذا كان M-K-L < 0،
!>
!>                   K M-K K+L-M
!>        D1 =   K ( I  0    0   )
!>             M-K ( 0  C    0   )
!>
!>                     K M-K K+L-M
!>        D2 =   M-K ( 0  S    0  )
!>             K+L-M ( 0  0    I  )
!>               P-L ( 0  0    0  )
!>
!>                    N-K-L  K   M-K  K+L-M
!>   ( 0 R ) =     K ( 0    R11  R12  R13  )
!>               M-K ( 0     0   R22  R23  )
!>             K+L-M ( 0     0    0   R33  )
!>
!> حيث
!>
!>   C = diag( ALPHA(K+1)، ... ، ALPHA(M) )،
!>   S = diag( BETA(K+1)،  ... ، BETA(M) )،
!>   C**2 + S**2 = I.
!>
!>   (R11 R12 R13 ) يتم تخزينها في A(1:M, N-K-L+1:N)، و R33 يتم تخزينها
!>   ( 0  R22 R23 )
!>   في B(M-K+1:L,N+M-K-L+1:N) عند الخروج.
!>
!> يقوم الإجراء الفرعي بحساب C، S، R، وبشكل اختياري مصفوفات التحويل الوحدوية
!> U، V و Q.
!>
!> بشكل خاص، إذا كانت B مصفوفة غير منفردة بحجم N×N، فإن GSVD لـ
!> A و B تعطي ضمنيًا SVD لـ A*inv(B):
!>                      A*inv(B) = U*(D1*inv(D2))*V**H.
!> إذا كان ( A**H,B**H)**H له أعمدة متعامدة، فإن GSVD لـ A و B يعادل أيضًا
!> التحلل CS لـ A و B. علاوة على ذلك، يمكن استخدام GSVD
!> لاشتقاق حل مشكلة القيمة الذاتية:
!>                      A**H*A x = lambda* B**H*B x.
!> في بعض الأدبيات، يتم تقديم GSVD لـ A و B بالشكل
!>                  U**H*A*X = ( 0 D1 )،   V**H*B*X = ( 0 D2 )
!> حيث U و V متعامدتان و X غير منفردة، و D1 و D2 "قطريتان".
!> يمكن تحويل صيغة GSVD السابقة إلى الصيغة الأخيرة
!> بأخذ المصفوفة غير المنفردة X على النحو التالي:
!>
!>                       X = Q*(  I   0    )
!>                             (  0 inv(R) )
!>

المعلمات

JOBU 
!>          JOBU is CHARACTER*1
!>          = 'U':  يتم حساب المصفوفة الوحدوية U؛
!>          = 'N':  لا يتم حساب U.
!>

JOBV

!>          JOBV is CHARACTER*1
!>          = 'V':  يتم حساب المصفوفة الوحدوية V؛
!>          = 'N':  لا يتم حساب V.
!>

JOBQ

!>          JOBQ is CHARACTER*1
!>          = 'Q':  يتم حساب المصفوفة الوحدوية Q؛
!>          = 'N':  لا يتم حساب Q.
!>

M

!>          M is INTEGER
!>          عدد صفوف المصفوفة A. M >= 0.
!>

N

!>          N is INTEGER
!>          عدد أعمدة المصفوفتين A و B. N >= 0.
!>

P

!>          P is INTEGER
!>          عدد صفوف المصفوفة B. P >= 0.
!>

K

!>          K is INTEGER
!>

L

!>          L is INTEGER
!>
!>          عند الخروج، تحدد K و L أبعاد المربعات الفرعية
!>          الموصوفة في الغرض.
!>          K + L = الرتبة العددية الفعالة لـ (A**H,B**H)**H.
!>

A

!>          A is COMPLEX*16 array, dimension (LDA,N)
!>          عند الإدخال، المصفوفة A بحجم M×N.
!>          عند الخروج، تحتوي A على المصفوفة المثلثية R، أو جزء من R.
!>          انظر الغرض للتفاصيل.
!>

LDA

!>          LDA is INTEGER
!>          البعد الرئيسي للمصفوفة A. LDA >= max(1,M).
!>

B

!>          B is COMPLEX*16 array, dimension (LDB,N)
!>          عند الإدخال، المصفوفة B بحجم P×N.
!>          عند الخروج، تحتوي B على جزء من المصفوفة المثلثية R إذا
!>          كان M-K-L < 0. انظر الغرض للتفاصيل.
!>

LDB

!>          LDB is INTEGER
!>          البعد الرئيسي للمصفوفة B. LDB >= max(1,P).
!>

ALPHA

!>          ALPHA is DOUBLE PRECISION array, dimension (N)
!>

BETA

!>          BETA is DOUBLE PRECISION array, dimension (N)
!>
!>          عند الخروج، تحتوي ALPHA و BETA على أزواج القيم المفردة المعممة
!>          لـ A و B؛
!>            ALPHA(1:K) = 1،
!>            BETA(1:K)  = 0،
!>          وإذا كان M-K-L >= 0،
!>            ALPHA(K+1:K+L) = C،
!>            BETA(K+1:K+L)  = S،
!>          أو إذا كان M-K-L < 0،
!>            ALPHA(K+1:M)=C, ALPHA(M+1:K+L)=0
!>            BETA(K+1:M) =S, BETA(M+1:K+L) =1
!>          و
!>            ALPHA(K+L+1:N) = 0
!>            BETA(K+L+1:N)  = 0
!>

U

!>          U is COMPLEX*16 array, dimension (LDU,M)
!>          إذا كان JOBU = 'U'، تحتوي U على المصفوفة الوحدوية U بحجم M×M.
!>          إذا كان JOBU = 'N'، لا يتم الرجوع إلى U.
!>

LDU

!>          LDU is INTEGER
!>          البعد الرئيسي للمصفوفة U. LDU >= max(1,M) إذا
!>          كان JOBU = 'U'. LDU >= 1 بخلاف ذلك.
!>

V

!>          V is COMPLEX*16 array, dimension (LDV,P)
!>          إذا كان JOBV = 'V'، تحتوي V على المصفوفة الوحدوية V بحجم P×P.
!>          إذا كان JOBV = 'N'، لا يتم الرجوع إلى V.
!>

LDV

!>          LDV is INTEGER
!>          البعد الرئيسي للمصفوفة V. LDV >= max(1,P) إذا
!>          كان JOBV = 'V'. LDV >= 1 بخلاف ذلك.
!>

Q

!>          Q is COMPLEX*16 array, dimension (LDQ,N)
!>          إذا كان JOBQ = 'Q'، تحتوي Q على المصفوفة الوحدوية Q بحجم N×N.
!>          إذا كان JOBQ = 'N'، لا يتم الرجوع إلى Q.
!>

LDQ

!>          LDQ is INTEGER
!>          البعد الرئيسي للمصفوفة Q. LDQ >= max(1,N) إذا
!>          كان JOBQ = 'Q'. LDQ >= 1 بخلاف ذلك.
!>

WORK

!>          WORK is COMPLEX*16 array, dimension (max(1, LWORK))
!>          متغير مساحة عمل.
!>

RWORK

!>          RWORK is DOUBLE PRECISION array, dimension (LRWORK)
!>          متغير مساحة عمل.
!>

IWORK

!>          IWORK is INTEGER array, dimension (LIWORK)
!>          متغير مساحة عمل.
!>

INFO

!>          INFO is INTEGER
!>          = 0:  نجاح.
!>          < 0:  إذا كان المعلمة -ibias هي قيمة غير صالحة، حيث ibias هو
!>                رقم المعلمة التي تم تحديد قيمتها غير الصالحة.
!>          > 0:  فشل حساب القيم المفردة.
!>                إذا كان INFO = i > 0، فأن الإجراء الفرعي CSQSVD(I)
!>                قد فشل في حساب i القيم المفردة.
!>
