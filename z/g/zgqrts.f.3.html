وثائق الوظيفة/الدالة الفرعية

subroutine zgqrts (integer n, integer m, integer p, complex*16, dimension( lda, * ) a, complex*16, dimension( lda, * ) af, complex*16, dimension( lda, * ) q, complex*16, dimension( lda, * ) r, integer lda, complex*16, dimension( * ) taua, complex*16, dimension( ldb, * ) b, complex*16, dimension( ldb, * ) bf, complex*16, dimension( ldb, * ) z, complex*16, dimension( ldb, * ) t, complex*16, dimension( ldb, * ) bwk, integer ldb, complex*16, dimension( * ) taub, complex*16, dimension( lwork ) work, integer lwork, double precision, dimension( * ) rwork, double precision, dimension( 4 ) result)

ZGQRTS

الغرض:

!>
!> ZGQRTS تختبر ZGGQRF، التي تحسب عامل GQR لمصفوفة A بحجم N×M ومصفوفة B بحجم N×P: A = Q*R و B = Q*T*Z.
!>

المعلمات

N 
!>          N هو عدد صحيح (INTEGER)
!>          عدد صفوف المصفوفتين A و B. N >= 0.
!>

M

!>          M هو عدد صحيح (INTEGER)
!>          عدد أعمدة المصفوفة A. M >= 0.
!>

P

!>          P هو عدد صحيح (INTEGER)
!>          عدد أعمدة المصفوفة B. P >= 0.
!>

A

!>          A هي مصفوفة مركبة (COMPLEX*16) بالأبعاد (LDA,M)
!>          المصفوفة A بحجم N×M.
!>

AF

!>          AF هي مصفوفة مركبة (COMPLEX*16) بالأبعاد (LDA,N)
!>          تفاصيل عامل GQR للمصفوفتين A و B، كما تم إرجاعها بواسطة ZGGQRF، انظر CGGQRF لمزيد من التفاصيل.
!>

Q

!>          Q هي مصفوفة مركبة (COMPLEX*16) بالأبعاد (LDA,N)
!>          المصفوفة الوحدوية Q بحجم M×M.
!>

R

!>          R هي مصفوفة مركبة (COMPLEX*16) بالأبعاد (LDA,MAX(M,N))
!>

LDA

!>          LDA هو عدد صحيح (INTEGER)
!>          البعد الرئيسي للمصفوفات A، AF، R و Q.
!>          LDA >= max(M,N).
!>

TAUA

!>          TAUA هي مصفوفة مركبة (COMPLEX*16) بالأبعاد (min(M,N))
!>          العوامل القياسية للمنعكسات الأولية، كما تم إرجاعها بواسطة ZGGQRF.
!>

B

!>          B هي مصفوفة مركبة (COMPLEX*16) بالأبعاد (LDB,P)
!>          عند الدخول، المصفوفة B بحجم N×P.
!>

BF

!>          BF هي مصفوفة مركبة (COMPLEX*16) بالأبعاد (LDB,N)
!>          تفاصيل عامل GQR للمصفوفتين A و B، كما تم إرجاعها بواسطة ZGGQRF، انظر CGGQRF لمزيد من التفاصيل.
!>

Z

!>          Z هي مصفوفة مركبة (COMPLEX*16) بالأبعاد (LDB,P)
!>          المصفوفة الوحدوية Z بحجم P×P.
!>

T

!>          T هي مصفوفة مركبة (COMPLEX*16) بالأبعاد (LDB,max(P,N))
!>

BWK

!>          BWK هي مصفوفة مركبة (COMPLEX*16) بالأبعاد (LDB,N)
!>

LDB

!>          LDB هو عدد صحيح (INTEGER)
!>          البعد الرئيسي للمصفوفات B، BF، Z و T.
!>          LDB >= max(P,N).
!>

TAUB

!>          TAUB هي مصفوفة مركبة (COMPLEX*16) بالأبعاد (min(P,N))
!>          العوامل القياسية للمنعكسات الأولية، كما تم إرجاعها بواسطة DGGRQF.
!>

WORK

!>          WORK هي مصفوفة مركبة (COMPLEX*16) بالأبعاد (LWORK)
!>

LWORK

!>          LWORK هو عدد صحيح (INTEGER)
!>          بعد المصفوفة WORK، LWORK >= max(N,M,P)**2.
!>

RWORK

!>          RWORK هي مصفوفة ذات دقة مزدوجة (DOUBLE PRECISION) بالأبعاد (max(N,M,P))
!>

RESULT

!>          RESULT هي مصفوفة ذات دقة مزدوجة (DOUBLE PRECISION) بالأبعاد (4)
!>          نسب الاختبار:
!>            RESULT(1) = norm( R - Q'*A ) / ( MAX(M,N)*norm(A)*ULP)
!>            RESULT(2) = norm( T*Z - Q'*B ) / (MAX(P,N)*norm(B)*ULP)
!>            RESULT(3) = norm( I - Q'*Q ) / ( M*ULP )
!>            RESULT(4) = norm( I - Z'*Z ) / ( P*ULP )
!>
