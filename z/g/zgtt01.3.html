subroutine zgtt01 (integer n, complex*16, dimension( * ) dl, complex*16, dimension( * ) d, complex*16, dimension( * ) du, complex*16, dimension( * ) dlf, complex*16, dimension( * ) df, complex*16, dimension( * ) duf, complex*16, dimension( * ) du2, integer, dimension( * ) ipiv, complex*16, dimension( ldwork, * ) work, integer ldwork, double precision, dimension( * ) rwork, double precision resid)

ZGTT01

الغرض:

!>
!> ZGTT01 يعيد بناء مصفوفة ثلاثية القطر A من تحليلها LU ويحسب المتبقي
!>    norm(L*U - A) / ( norm(A) * EPS ),
!> حيث EPS هي ثابت الآلة (machine epsilon).
!>

المعلمات

N 
!>          N هو عدد صحيح (INTEGER)
!>          رتبة المصفوفة A. N >= 0.
!>

DL

!>          DL هي مصفوفة أعداد مركبة *16 (COMPLEX*16)، حجمها (N-1)
!>          عناصر القطر السفلي (sub-diagonal) للمصفوفة A.
!>

D

!>          D هي مصفوفة أعداد مركبة *16 (COMPLEX*16)، حجمها (N)
!>          عناصر القطر الرئيسي للمصفوفة A.
!>

DU

!>          DU هي مصفوفة أعداد مركبة *16 (COMPLEX*16)، حجمها (N-1)
!>          عناصر القطر العلوي (super-diagonal) للمصفوفة A.
!>

DLF

!>          DLF هي مصفوفة أعداد مركبة *16 (COMPLEX*16)، حجمها (N-1)
!>          العوامل (multipliers) التي تحدد المصفوفة L من
!>          تحليل LU للمصفوفة A.
!>

DF

!>          DF هي مصفوفة أعداد مركبة *16 (COMPLEX*16)، حجمها (N)
!>          عناصر القطر الرئيسي للمصفوفة العلوية المثلثية U من
!>          تحليل LU للمصفوفة A.
!>

DUF

!>          DUF هي مصفوفة أعداد مركبة *16 (COMPLEX*16)، حجمها (N-1)
!>          عناصر القطر العلوي الأول للمصفوفة U.
!>

DU2

!>          DU2 هي مصفوفة أعداد مركبة *16 (COMPLEX*16)، حجمها (N-2)
!>          عناصر القطر العلوي الثاني للمصفوفة U.
!>

IPIV

!>          IPIV هي مصفوفة أعداد صحيحة (INTEGER)، حجمها (N)
!>          فهارس المحاور (pivot indices)؛ من 1 <= i <= n، تم تبديل الصف i للمصفوفة
!>          مع الصف IPIV(i). IPIV(i) سيكون دائما إما
!>          i أو i+1؛ IPIV(i) = i يشير إلى عدم الحاجة لتبديل صف.
!>

WORK

!>          WORK هي مصفوفة أعداد مركبة *16 (COMPLEX*16)، حجمها (LDWORK,N)
!>

LDWORK

!>          LDWORK هو عدد صحيح (INTEGER)
!>          البعد الرائد للمصفوفة WORK. LDWORK >= max(1,N).
!>

RWORK

!>          RWORK هي مصفوفة أعداد حقيقية مضاعفة الدقة (DOUBLE PRECISION)، حجمها (N)
!>

RESID

!>          RESID هو عدد حقيقي مضاعف الدقة (DOUBLE PRECISION)
!>          المتبقي المُقاس: norm(L*U - A) / (norm(A) * EPS)
!>
