وثائق الوظيفة/الروتين الفرعي

subroutine zggglm (integer n, integer m, integer p, complex*16, dimension( lda, * ) a, integer lda, complex*16, dimension( ldb, * ) b, integer ldb, complex*16, dimension( * ) d, complex*16, dimension( * ) x, complex*16, dimension( * ) y, complex*16, dimension( * ) work, integer lwork, integer info)

ZGGGLM

الغرض:

!>
!> ZGGGLM يحل مشكلة نموذج غاوس-ماركوف الخطي العام (GLM):
!>
!>         Minimize || y ||_2   subject to   d = A*x + B*y
!>             x
!>
!> حيث A مصفوفة N-by-M، و B مصفوفة N-by-P، و d متجه N معطى. يُفترض أن M <= N <= M+P، و
!>
!>            rank(A) = M    and    rank( A B ) = N.
!>
!> في ظل هذه الافتراضات، تكون المعادلة المقيدة متسقة دائمًا، وهناك حل فريد x وحل ذو الحد الأدنى للمعيار 2 y، والذي يتم الحصول عليه باستخدام تحليل QR المعمم للمصفوفات (A, B) المعطاة بواسطة
!>
!>    A = Q*(R),   B = Q*T*Z.
!>          (0)
!>
!> على وجه الخصوص، إذا كانت المصفوفة B مربعة وغير منفردة، فإن المشكلة
!> GLM تعادل مشكلة المربعات الصغرى الخطية الموزونة التالية
!>
!>              minimize || inv(B)*(d-A*x) ||_2
!>                  x
!>
!> حيث تشير inv(B) إلى معكوس B.
!>
!> يجب على المتصلين بهذا الروتين الفرعي ملاحظة أن عمليات التحقق من التفرد/عدم الرتبة
!> المنفذة في هذا الروتين الفرعي بدائية. الروتين الفرعي ZTRTRS الذي يستدعيه هذا
!> الروتين الفرعي يشير فقط إلى فشل بسبب التفرد إذا كانت المشكلة منفردة تمامًا.
!>
!> من الممكن أن تكون واحدة (أو أكثر) من العوامل المعنية في تحليل QR المعمم
!> للزوج (A, B) قريبة بشكل غير طبيعي من التفرد دون أن يشير هذا
!> الروتين الفرعي إلى خطأ. قد تكون الحلول المحسوبة لمثل هذه المشاكل شبه الفريدة
!> أقل دقة بسبب فقدان الدقة العددية.
!> 
!>

المعاملات

N 
!>          N is INTEGER
!>          عدد صفوف المصفوفات A و B. N >= 0.
!>

M

!>          M is INTEGER
!>          عدد أعمدة المصفوفة A. 0 <= M <= N.
!>

P

!>          P is INTEGER
!>          عدد أعمدة المصفوفة B. P >= N-M.
!>

A

!>          A is COMPLEX*16 array, dimension (LDA,M)
!>          عند الإدخال، المصفوفة A بحجم N-by-M.
!>          عند الخروج، الجزء المثلثي العلوي من المصفوفة A يحتوي على
!>          المصفوفة المثلثية العلوية R بحجم M-by-M.
!>

LDA

!>          LDA is INTEGER
!>          البعد الرائد للمصفوفة A. LDA >= max(1,N).
!>

B

!>          B is COMPLEX*16 array, dimension (LDB,P)
!>          عند الإدخال، المصفوفة B بحجم N-by-P.
!>          عند الخروج، إذا كانت N <= P، فإن الجزء المثلثي العلوي من الجزء الفرعي
!>          B(1:N,P-N+1:P) يحتوي على المصفوفة المثلثية العلوية R بحجم N-by-N؛
!>          إذا كانت N > P، فإن العناصر على وفوق القطر الفرعي (N-M)
!>          تحتوي على شبه المصفوفة شبه المثلثية العلوية T بحجم N-by-P.
!>

LDB

!>          LDB is INTEGER
!>          البعد الرائد للمصفوفة B. LDB >= max(1,N).
!>

D

!>          D is COMPLEX*16 array, dimension (N)
!>          عند الإدخال، D هو الطرف الأيسر لمعادلة GLM.
!>          عند الخروج، يتم تدمير D.
!>

X

!>          X is COMPLEX*16 array, dimension (M)
!>

Y

!>          Y is COMPLEX*16 array, dimension (P)
!>
!>          عند الخروج، X و Y هما حلا مشكلة GLM.
!>

WORK

!>          WORK is COMPLEX*16 array, dimension (MAX(1,LWORK))
!>          عند الخروج، إذا كان INFO = 0، فإن WORK(1) يعيد قيمة LWORK المثلى.
!>

LWORK

!>          LWORK is INTEGER
!>          بعد المصفوفة WORK. LWORK >= max(1,N+M+P).
!>          لأداء مثالي، LWORK >= M+min(N,P)+max(N,P)*NB،
!>          حيث NB هو حد أعلى لأحجام القطع المثلى لـ
!>          ZGEQRF, ZGERQF, ZUNMQR و ZUNMRQ.
!>
!>          إذا كانت LWORK = -1، فسيتم افتراض استعلام مساحة العمل؛ الروتين
!>          يحسب فقط الحجم الأمثل للمصفوفة WORK، ويعيد
!>          هذه القيمة كالإدخال الأول للمصفوفة WORK، ولا يتم إصدار أي رسالة خطأ
!>          متعلقة بـ LWORK بواسطة XERBLA.
!>

INFO

!>          INFO is INTEGER
!>          = 0:  خروج ناجح.
!>          < 0:  إذا كان INFO = -i، فإن الوسيط i كان له قيمة غير قانونية.
!>          = 1:  العامل المثلثي العلوي R المرتبط بـ A في
!>                تحليل QR المعمم للزوج (A, B) هو منفرد تمامًا، بحيث rank(A) < M؛ لم يكن بالإمكان حساب الحل المربعات الصغرى.
!>          = 2:  الجزء السفلي الذي حجمه (N-M) بـ (N-M) من العامل شبه المثلثي العلوي
!>                T المرتبط بـ B في تحليل QR المعمم
!>                للزوج (A, B) هو منفرد تمامًا، بحيث
!>                rank( A B ) < N؛ لم يكن بالإمكان حساب الحل المربعات الصغرى.
!>
