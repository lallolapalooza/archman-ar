وثائق الوظيفة/الإجراء الفرعي

subroutine zgesc2 (integer n, complex*16, dimension( lda, * ) a, integer lda, complex*16, dimension( * ) rhs, integer, dimension( * ) ipiv, integer, dimension( * ) jpiv, double precision scale)

ZGESC2 يحل نظام معادلات خطية باستخدام تحليل LU مع التنقيح الكامل الذي تم حسابه بواسطة sgetc2.

الغرض:

!>
!> ZGESC2 يحل نظام معادلات خطية
!>
!>           A * X = scale* RHS
!>
!> مع مصفوفة عامة N-by-N A باستخدام تحليل LU مع
!> التنقيح الكامل الذي تم حسابه بواسطة ZGETC2.
!>
!>

المعلمات

N 
!>          N هو عدد صحيح
!>          عدد أعمدة المصفوفة A.
!>

A

!>          A هو مصفوفة COMPLEX*16، ذات أبعاد (LDA, N)
!>          عند الدخول، جزء LU لتحليل المصفوفة N-by-N
!>          A التي تم حسابها بواسطة ZGETC2: A = P * L * U * Q
!>

LDA

!>          LDA هو عدد صحيح
!>          البعد الرائد للمصفوفة A. LDA >= max(1, N).
!>

RHS

!>          RHS هو مصفوفة COMPLEX*16، ذات أبعاد N.
!>          عند الدخول، متجه الطرف الأيمن b.
!>          عند الخروج، متجه الحل X.
!>

IPIV

!>          IPIV هو مصفوفة أعداد صحيحة، ذات أبعاد (N).
!>          مؤشرات التنقيح؛ لـ 1 <= i <= N، تم تبديل الصف i للمصفوفة
!>          مع الصف IPIV(i).
!>

JPIV

!>          JPIV هو مصفوفة أعداد صحيحة، ذات أبعاد (N).
!>          مؤشرات التنقيح؛ لـ 1 <= j <= N، تم تبديل العمود j للمصفوفة
!>          مع العمود JPIV(j).
!>

SCALE

!>          SCALE هو DOUBLE PRECISION
!>           عند الخروج، يحتوي SCALE على عامل القياس. تم اختيار SCALE
!>           0 <= SCALE <= 1 لمنع تجاوز سعة الحل.
!>
