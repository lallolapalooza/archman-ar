subroutine zggev (character jobvl, character jobvr, integer n, complex*16, dimension( lda, * ) a, integer lda, complex*16, dimension( ldb, * ) b, integer ldb, complex*16, dimension( * ) alpha, complex*16, dimension( * ) beta, complex*16, dimension( ldvl, * ) vl, integer ldvl, complex*16, dimension( ldvr, * ) vr, integer ldvr, complex*16, dimension( * ) work, integer lwork, double precision, dimension( * ) rwork, integer info)

ZGGEV يحسب القيم الذاتية، واختياريًا، المتجهات الذاتية اليسرى و/أو اليمنى لمصفوفات GE

الغرض:

!>
!> ZGGEV يحسب لزوج من مصفوفات N-by-N المركبة غير المتناظرة (A,B)، القيم الذاتية المعممة، واختياريًا، المتجهات الذاتية المعممة اليسرى و/أو اليمنى.
!>
!> القيمة الذاتية المعممة لزوج من المصفوفات (A,B) هي عدد سالب lambda أو نسبة alpha/beta = lambda، بحيث A - lambda*B تكون مفردة (singular). عادة ما يتم تمثيلها كزوج (alpha,beta)، حيث يوجد تفسير معقول لـ beta=0، وحتى لكليهما يكونان صفرًا.
!>
!> المتجه الذاتي المعممة اليمنى v(j) المقابل للقيمة الذاتية المعممة lambda(j) لـ (A,B) يحقق
!>
!>              A * v(j) = lambda(j) * B * v(j).
!>
!> المتجه الذاتي المعممة اليسرى u(j) المقابل للقيمة الذاتية المعممة lambda(j) لـ (A,B) يحقق
!>
!>              u(j)**H * A = lambda(j) * u(j)**H * B
!>
!> حيث u(j)**H هي المرافق المتبادل (conjugate-transpose) لـ u(j).
!>

المعلمات

JOBVL 
!>          JOBVL هو CHARACTER*1
!>          = 'N':  لا تحسب المتجهات الذاتية المعممة اليسرى؛
!>          = 'V':  احسب المتجهات الذاتية المعممة اليسرى.
!>

JOBVR

!>          JOBVR هو CHARACTER*1
!>          = 'N':  لا تحسب المتجهات الذاتية المعممة اليمنى؛
!>          = 'V':  احسب المتجهات الذاتية المعممة اليمنى.
!>

N

!>          N هو INTEGER
!>          ترتيب المصفوفات A, B, VL, و VR.  N >= 0.
!>

A

!>          A هو مصفوفة COMPLEX*16، ذات أبعاد (LDA, N)
!>          عند الإدخال، المصفوفة A في الزوج (A,B).
!>          عند الخروج، تم استبدال A.
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرئيسي لـ A.  LDA >= max(1,N).
!>

B

!>          B هو مصفوفة COMPLEX*16، ذات أبعاد (LDB, N)
!>          عند الإدخال، المصفوفة B في الزوج (A,B).
!>          عند الخروج، تم استبدال B.
!>

LDB

!>          LDB هو INTEGER
!>          البعد الرئيسي لـ B.  LDB >= max(1,N).
!>

ALPHA

!>          ALPHA هي مصفوفة COMPLEX*16، ذات أبعاد (N)
!>

BETA

!>          BETA هي مصفوفة COMPLEX*16، ذات أبعاد (N)
!>          عند الخروج، ALPHA(j)/BETA(j)، j=1,...,N، ستكون القيم الذاتية المعممة.
!>
!>          ملاحظة: قد تتجاوز النسب ALPHA(j)/BETA(j) أو تقل بشكل كبير، وقد تكون BETA(j) صفرًا. لذا، يجب على المستخدم تجنب حساب النسبة alpha/beta بشكل ساذج.
!>          ومع ذلك، سيكون ALPHA دائمًا أقل من، وعادة ما يكون قابلاً للمقارنة مع، norm(A) في الحجم، وBETA دائمًا أقل من، وعادة ما يكون قابلاً للمقارنة مع، norm(B).
!>

VL

!>          VL هي مصفوفة COMPLEX*16، ذات أبعاد (LDVL,N)
!>          إذا كان JOBVL = 'V'، يتم تخزين المتجهات الذاتية المعممة اليسرى u(j) متتالية في أعمدة VL، بنفس ترتيب قيمها الذاتية.
!>          يتم قياس كل متجه ذاتي بحيث يكون المكون الأكبر لديه abs(الجزء الحقيقي) + abs(الجزء التخيلي) = 1.
!>          لا يتم الرجوع إليه إذا كان JOBVL = 'N'.
!>

LDVL

!>          LDVL هو INTEGER
!>          البعد الرئيسي للمصفوفة VL. LDVL >= 1، وإذا كان JOBVL = 'V'، فإن LDVL >= N.
!>

VR

!>          VR هي مصفوفة COMPLEX*16، ذات أبعاد (LDVR,N)
!>          إذا كان JOBVR = 'V'، يتم تخزين المتجهات الذاتية المعممة اليمنى v(j) متتالية في أعمدة VR، بنفس ترتيب قيمها الذاتية.
!>          يتم قياس كل متجه ذاتي بحيث يكون المكون الأكبر لديه abs(الجزء الحقيقي) + abs(الجزء التخيلي) = 1.
!>          لا يتم الرجوع إليه إذا كان JOBVR = 'N'.
!>

LDVR

!>          LDVR هو INTEGER
!>          البعد الرئيسي للمصفوفة VR. LDVR >= 1، وإذا كان JOBVR = 'V'، فإن LDVR >= N.
!>

WORK

!>          WORK هي مصفوفة COMPLEX*16، ذات أبعاد (MAX(1,LWORK))
!>          عند الخروج، إذا كان INFO = 0، فإن WORK(1) يعيد LWORK المثلى.
!>

LWORK

!>          LWORK هو INTEGER
!>          بعد المصفوفة WORK.  LWORK >= max(1,2*N).
!>          للحصول على أداء جيد، يجب أن يكون LWORK أكبر بشكل عام.
!>
!>          إذا كان LWORK = -1، فيفترض استعلام عن مساحة العمل؛ يقوم الروتين فقط بحساب الحجم الأمثل لمصفوفة WORK، ويعيد هذا الحجم كإدخال أول في مصفوفة WORK، ولا يتم إصدار أي رسالة خطأ متعلقة بـ LWORK بواسطة XERBLA.
!>

RWORK

!>          RWORK هي مصفوفة DOUBLE PRECISION، ذات أبعاد (8*N)
!>

INFO

!>          INFO هو INTEGER
!>          = 0:  خروج ناجح
!>          < 0:  إذا كان INFO = -i، كان المعامل i غير قانوني.
!>          =1,...,N:
!>                فشلت تكرارات QZ. لم يتم حساب أي متجهات ذاتية، ولكن يجب أن تكون ALPHA(j) و BETA(j) صحيحة لـ j=INFO+1,...,N.
!>          > N:  =N+1: فشلت تكرارات QZ بخلاف ذلك في ZHGEQZ،
!>                =N+2: عودة خطأ من ZTGEVC.
!>
