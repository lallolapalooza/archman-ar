subroutine zgsvj1 (character*1 jobv, integer m, integer n, integer n1, complex*16, dimension( lda, * ) a, integer lda, complex*16, dimension( n ) d, double precision, dimension( n ) sva, integer mv, complex*16, dimension( ldv, * ) v, integer ldv, double precision eps, double precision sfmin, double precision tol, integer nsweep, complex*16, dimension( lwork ) work, integer lwork, integer info)

ZGSVJ1 معالج مسبق للروتين zgesvj، يطبق دورانات جاكوبي تستهدف فقط محاور محددة.

الغرض:

!>
!> يتم استدعاء ZGSVJ1 من ZGESVJ كمعالج مسبق وهذا هو غرضه الرئيسي.
!> يطبق دورانات جاكوبي بنفس الطريقة التي يطبقها ZGESVJ، ولكنه يستهدف
!> محاور محددة فقط ولا يتحقق من التقارب (معيار التوقف). تتوفر بعض
!> معلمات الضبط (المميزة بـ [TP]) للمنفذ.
!>
!> تفاصيل إضافية
!> ~~~~~~~~~~~~~~~
!> يطبق ZGSVJ1 بعض دورات دورانات جاكوبي في الفضاء العمودي للمصفوفة
!> المدخلة A ذات الأبعاد M × N. يتم أخذ أزواج المحاور من الكتلة خارج القطر (1،2)
!> في مصفوفة جرام المقابلة A^T * A ذات الأبعاد N × N.
!> يتم تمييز مداخل الكتل (البلاطات) للكتل خارج القطر (1،2) بـ
!> [x] في المخطط التالي:
!>
!>    | *  *  * [x] [x] [x]|
!>    | *  *  * [x] [x] [x]|    دورات صفوف في كتل [x] ذات الأبعاد nblr × nblc.
!>    | *  *  * [x] [x] [x]|    محاور دورانية داخل كل كتلة [x].
!>    |[x] [x] [x] *  *  * |
!>    |[x] [x] [x] *  *  * |
!>    |[x] [x] [x] *  *  * |
!>
!> من حيث أعمدة A، يتم تدوير الأعمدة N1 الأولى 'مقابل' الأعمدة المتبقية N-N1
!> من A، في محاولة لزيادة الزاوية بين الفضاءات الفرعية المقابلة.
!> الكتلة خارج القطر هي N1 × (N-N1) ويتم تبليطها باستخدام بلاطات مربعة
!> ذات طول جانبي KBL. هنا، KBL هي معلمة ضبط.
!> عدد الدورات معطى في NSWEEP وعتبة التعامد معطاة في TOL.
!>

المعلمات

JOBV 
!>          JOBV هو CHARACTER*1
!>          يحدد ما إذا كان الناتج من هذا الإجراء سيستخدم
!>          لحساب المصفوفة V:
!>          = 'V': يتم تجميع حاصل ضرب دورانات جاكوبي
!>                 عن طريق ضرب المصفوفة V ذات الأبعاد N × N من اليمين.
!>                (انظر وصف V.)
!>          = 'A': يتم تجميع حاصل ضرب دورانات جاكوبي
!>                 عن طريق ضرب المصفوفة MV × N المسماة V من اليمين.
!>                (انظر وصف MV و V.)
!>          = 'N': لا يتم تجميع دورانات جاكوبي.
!>

M

!>          M هو INTEGER
!>          عدد صفوف المصفوفة المدخلة A.  M >= 0.
!>

N

!>          N هو INTEGER
!>          عدد أعمدة المصفوفة المدخلة A.
!>          M >= N >= 0.
!>

N1

!>          N1 هو INTEGER
!>          يحدد N1 تقسيم الكتلة 2 × 2، يتم تدوير الأعمدة N1 الأولى 'مقابل'
!>          الأعمدة المتبقية N-N1 من A.
!>

A

!>          A هو مصفوفة COMPLEX*16، البعد (LDA,N)
!>          عند الإدخال، مصفوفة A ذات الأبعاد M × N، بحيث تمثل A*diag(D)
!>          المصفوفة المدخلة.
!>          عند الخروج،
!>          A_onexit * D_onexit تمثل المصفوفة المدخلة A*diag(D)
!>          مضروبة من اليمين بسلسلة من دورانات جاكوبي، حيث
!>          عتبة الدوران والعدد الإجمالي للدورات معطاة في
!>          TOL و NSWEEP، على التوالي.
!>          (انظر وصف N1، D، TOL و NSWEEP.)
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرئيسي للمصفوفة A.  LDA >= max(1,M).
!>

D

!>          D هي مصفوفة COMPLEX*16، البعد (N)
!>          تجمع المصفوفة D عوامل القياس من دورانات جاكوبي السريعة والمقاسة.
!>          عند الإدخال، تمثل A*diag(D) المصفوفة المدخلة.
!>          عند الخروج، تمثل A_onexit*diag(D_onexit) المصفوفة المدخلة
!>          مضروبة من اليمين بسلسلة من دورانات جاكوبي، حيث
!>          عتبة الدوران والعدد الإجمالي للدورات معطاة في
!>          TOL و NSWEEP، على التوالي.
!>          (انظر وصف N1، A، TOL و NSWEEP.)
!>

SVA

!>          SVA هي مصفوفة DOUBLE PRECISION، البعد (N)
!>          عند الإدخال، تحتوي SVA على المعايير الإقليدية لأعمدة
!>          المصفوفة A*diag(D).
!>          عند الخروج، تحتوي SVA على المعايير الإقليدية لأعمدة
!>          المصفوفة onexit*diag(D_onexit).
!>

MV

!>          MV هو INTEGER
!>          إذا كان JOBV = 'A'، فإن MV صفًا من V يتم ضربها من اليمين
!>                           بسلسلة من دورانات جاكوبي.
!>          إذا كان JOBV = 'N'،   فلن يتم استخدام MV.
!>

V

!>          V هي مصفوفة COMPLEX*16، البعد (LDV,N)
!>          إذا كان JOBV = 'V'، فإن N صفًا من V يتم ضربها من اليمين
!>                           بسلسلة من دورانات جاكوبي.
!>          إذا كان JOBV = 'A'، فإن MV صفًا من V يتم ضربها من اليمين
!>                           بسلسلة من دورانات جاكوبي.
!>          إذا كان JOBV = 'N'،   فلن يتم استخدام V.
!>

LDV

!>          LDV هو INTEGER
!>          البعد الرئيسي للمصفوفة V،  LDV >= 1.
!>          إذا كان JOBV = 'V'، فإن LDV >= N.
!>          إذا كان JOBV = 'A'، فإن LDV >= MV.
!>

EPS

!>          EPS هو DOUBLE PRECISION
!>          EPS = DLAMCH('Epsilon')
!>

SFMIN

!>          SFMIN هو DOUBLE PRECISION
!>          SFMIN = DLAMCH('Safe Minimum')
!>

TOL

!>          TOL هو DOUBLE PRECISION
!>          TOL هي عتبة دورانات جاكوبي. بالنسبة لزوج
!>          من أعمدة المحاور A(:,p)، A(:,q)، يتم تطبيق دوران جاكوبي
!>          فقط إذا كان ABS(COS(angle(A(:,p),A(:,q)))) > TOL.
!>

NSWEEP

!>          NSWEEP هو INTEGER
!>          NSWEEP هو عدد دورات جاكوبي التي سيتم إجراؤها.
!>

WORK

!>          WORK هي مصفوفة COMPLEX*16، البعد (LWORK)
!>

LWORK

!>          LWORK هو INTEGER
!>          LWORK هو بعد WORK. LWORK >= M.
!>

INFO

!>          INFO هو INTEGER
!>          = 0:  الخروج بنجاح.
!>          < 0:  إذا كان INFO = -i، فإن الوسيط i كان له قيمة غير قانونية
!>
