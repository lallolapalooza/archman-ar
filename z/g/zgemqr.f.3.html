subroutine zgemqr (character side, character trans, integer m, integer n, integer k, complex*16, dimension( lda, * ) a, integer lda, complex*16, dimension( * ) t, integer tsize, complex*16, dimension( ldc, * ) c, integer ldc, complex*16, dimension( * ) work, integer lwork, integer info)

ZGEMQR

الغرض:

!>
!> ZGEMQR يستبدل المصفوفة الحقيقية العامة M-by-N المسماة C بـ
!>
!>                      SIDE = 'L'     SIDE = 'R'
!>      TRANS = 'N':      Q * C          C * Q
!>      TRANS = 'T':      Q**H * C       C * Q**H
!>
!> حيث Q هي مصفوفة وحدوية مركبة معرفة كناتج
!> لمُعكِسات أولية مجمعة تم حسابها بواسطة تحليل QR النحيف (ZGEQR)
!>
!>

المعاملات

SIDE 
!>          SIDE is CHARACTER*1
!>          = 'L': تطبيق Q أو Q**H من اليسار؛
!>          = 'R': تطبيق Q أو Q**H من اليمين.
!>

TRANS

!>          TRANS is CHARACTER*1
!>          = 'N':  لا يوجد تبديل، تطبيق Q؛
!>          = 'C':  تبديل مرافق، تطبيق Q**H.
!>

M

!>          M is INTEGER
!>          عدد صفوف المصفوفة A.  M >=0.
!>

N

!>          N is INTEGER
!>          عدد أعمدة المصفوفة C. N >= 0.
!>

K

!>          K is INTEGER
!>          عدد المُعكِسات الأولية التي يُعرّف حاصل ضربها
!>          المصفوفة Q.
!>          إذا كان SIDE = 'L'، فإن M >= K >= 0؛
!>          إذا كان SIDE = 'R'، فإن N >= K >= 0.
!>

A

!>          A is COMPLEX*16 array, dimension (LDA,K)
!>          جزء من بنية البيانات لتمثيل Q كما تم إرجاعه بواسطة ZGEQR.
!>

LDA

!>          LDA is INTEGER
!>          البعد الرئيسي للمصفوفة A.
!>          إذا كان SIDE = 'L'، فإن LDA >= max(1,M)؛
!>          إذا كان SIDE = 'R'، فإن LDA >= max(1,N).
!>

T

!>          T is COMPLEX*16 array, dimension (MAX(5,TSIZE)).
!>          جزء من بنية البيانات لتمثيل Q كما تم إرجاعه بواسطة ZGEQR.
!>

TSIZE

!>          TSIZE is INTEGER
!>          بعد المصفوفة T. TSIZE >= 5.
!>

C

!>          C is COMPLEX*16 array, dimension (LDC,N)
!>          عند الإدخال، مصفوفة C بحجم M-by-N.
!>          عند الخروج، يتم استبدال C بـ Q*C أو Q**H*C أو C*Q**H أو C*Q.
!>

LDC

!>          LDC is INTEGER
!>          البعد الرئيسي للمصفوفة C. LDC >= max(1,M).
!>

WORK

!>          (مساحة عمل) COMPLEX*16 array, dimension (MAX(1,LWORK))
!>          عند الخروج، إذا كان INFO = 0، فإن WORK(1) يُرجع الحد الأدنى LWORK.
!>

LWORK

!>          LWORK is INTEGER
!>          بعد المصفوفة WORK. LWORK >= 1.
!>          إذا كان LWORK = -1، فسيتم افتراض استعلام مساحة العمل. يقوم الروتين
!>          بحساب حجم مصفوفة WORK فقط، ويرجع هذه القيمة كـ WORK(1)، ولا يتم إصدار
!>          أي رسالة خطأ متعلقة بـ WORK بواسطة XERBLA.
!>

INFO

!>          INFO is INTEGER
!>          = 0:  انتهاء ناجح
!>          < 0:  إذا كان INFO = -i، فإن المعامل i كان له قيمة غير قانونية
!>

!>
!> هذه التفاصيل خاصة بتطبيق LAPACK هذا. لا ينبغي على المستخدمين
!> اعتبارها مسلمات. قد تتغير هذه التفاصيل في المستقبل، ومن غير المحتمل
!> أن تكون صحيحة لتطبيق LAPACK آخر. هذه التفاصيل ذات صلة إذا أراد
!> المرء محاولة فهم الكود. وهي ليست جزءًا من الواجهة.
!>
!> في هذا الإصدار،
!>
!>          T(2): حجم كتلة الصف (MB)
!>          T(3): حجم كتلة العمود (NB)
!>          T(6:TSIZE): بنية البيانات اللازمة لـ Q، تم حسابها بواسطة
!>                           ZLATSQR أو ZGEQRT
!>
!>  اعتمادًا على أبعاد المصفوفة M و N، وأحجام كتل الصفوف والأعمدة MB و NB
!>  التي تم إرجاعها بواسطة ILAENV، سيستخدم ZGEQR إما
!>  ZLATSQR (إذا كانت المصفوفة نحيفة وطويلة) أو ZGEQRT لحساب
!>  تحليل QR.
!>  سيستخدم هذا الإصدار من ZGEMQR إما ZLAMTSQR أو ZGEMQRT لـ
!>  ضرب المصفوفة Q بمصفوفة أخرى.
!>  مزيد من التفاصيل في ZLAMTSQR أو ZGEMQRT.
!>
!>
