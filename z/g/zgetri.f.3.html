subroutine zgetri (integer n, complex*16, dimension( lda, * ) a, integer lda, integer, dimension( * ) ipiv, complex*16, dimension( * ) work, integer lwork, integer info)

ZGETRI

الغرض:

!>
!> ZGETRI يحسب معكوس مصفوفة باستخدام تحليل LU
!> الذي تم حسابه بواسطة ZGETRF.
!>
!> تستخدم هذه الطريقة معكوس U ثم تحسب inv(A) عن طريق حل النظام
!> inv(A)*L = inv(U) للحصول على inv(A).
!>

المعلمات

N 
!>          N هو عدد صحيح (INTEGER)
!>          رتبة المصفوفة A. N >= 0.
!>

A

!>          A هي مصفوفة مركبة*16 (COMPLEX*16)، ذات البعد (LDA,N)
!>          عند الدخول، عوامل L و U من التحليل
!>          A = P*L*U كما تم حسابه بواسطة ZGETRF.
!>          عند الخروج، إذا كان INFO = 0، فإنها معكوس المصفوفة الأصلية A.
!>

LDA

!>          LDA هو عدد صحيح (INTEGER)
!>          البعد الرائد للمصفوفة A. LDA >= max(1,N).
!>

IPIV

!>          IPIV هي مصفوفة أعداد صحيحة (INTEGER array)، ذات البعد (N)
!>          مؤشرات المحور من ZGETRF؛ لـ 1<=i<=N، تم تبديل الصف i للمصفوفة
!>          مع الصف IPIV(i).
!>

WORK

!>          WORK هي مصفوفة مركبة*16 (COMPLEX*16 array)، ذات البعد (MAX(1,LWORK))
!>          عند الخروج، إذا كان INFO=0، فإن WORK(1) ترجع LWORK المثلى.
!>

LWORK

!>          LWORK هو عدد صحيح (INTEGER)
!>          بعد المصفوفة WORK. LWORK >= max(1,N).
!>          لأداء أمثل LWORK >= N*NB، حيث NB هي
!>          حجم الكتلة الأمثل الذي تم إرجاعه بواسطة ILAENV.
!>
!>          إذا كان LWORK = -1، فسيتم افتراض استعلام مساحة العمل؛ الروتين
!>          يحسب فقط الحجم الأمثل لمصفوفة WORK، ويعيد
!>          هذه القيمة كالإدخال الأول لمصفوفة WORK، ولا يتم إصدار رسالة خطأ
!>          متعلقة بـ LWORK بواسطة XERBLA.
!>

INFO

!>          INFO هو عدد صحيح (INTEGER)
!>          = 0:  انتهاء ناجح
!>          < 0:  إذا كان INFO = -i، فإن الوسيط i كان له قيمة غير قانونية
!>          > 0:  إذا كان INFO = i، فإن U(i,i) يساوي صفرًا تمامًا؛ المصفوفة
!>                فردية ولا يمكن حساب معكوسها.
!>
