subroutine zgbsvxx (character fact, character trans, integer n, integer kl, integer ku, integer nrhs, complex*16, dimension( ldab, * ) ab, integer ldab, complex*16, dimension( ldafb, * ) afb, integer ldafb, integer, dimension( * ) ipiv, character equed, double precision, dimension( * ) r, double precision, dimension( * ) c, complex*16, dimension( ldb, * ) b, integer ldb, complex*16, dimension( ldx , * ) x, integer ldx, double precision rcond, double precision rpvgrw, double precision, dimension( * ) berr, integer n_err_bnds, double precision, dimension( nrhs, * ) err_bnds_norm, double precision, dimension( nrhs, * ) err_bnds_comp, integer nparams, double precision, dimension( * ) params, complex*16, dimension( * ) work, double precision, dimension( * ) rwork, integer info)

ZGBSVXX تحسب حل نظام المعادلات الخطية A * X = B لمصفوفات GB

الغرض:

!>
!>    يستخدم ZGBSVXX تحليل LU لحساب الحل لنظام المعادلات الخطية المعقدة*16 A * X = B،
!>    حيث A مصفوفة N × N والمصفوفتان X و B هما N × NRHS.
!>
!>    إذا طُلب ذلك، يتم إرجاع حدود خطأ معيارية (normwise) وأقصى خطأ مكوني (componentwise).
!>    سيعيد ZGBSVXX حلاً بخطأ صغير مضمون (O(eps) حيث eps هو دقة الآلة العاملة)
!>    إلا إذا كانت المصفوفة غير مشروطة للغاية، وفي هذه الحالة يتم إرجاع تحذير.
!>    يتم أيضًا حساب أرقام الشرط ذات الصلة وإعادتها.
!>
!>    يقبل ZGBSVXX تحليلات وعوامل موازنة مقدمة من المستخدم؛ انظر تعريفات خيارات FACT و EQUED.
!>    سيؤدي الحل مع التنقيح واستخدام تحليل من استدعاء ZGBSVXX سابق أيضًا إلى حل
!>    بأخطاء O(eps) أو تحذيرات، ولكن لا يمكننا تقديم هذا الادعاء للتحليلات وعوامل
!>    الموازنة العامة المقدمة من المستخدم إذا اختلفت عما سينتجه ZGBSVXX بنفسه.
!>

الوصف:

!>
!>    يتم تنفيذ الخطوات التالية:
!>
!>    1. إذا كان FACT = 'E'، يتم حساب عوامل قياس ذات دقة مزدوجة لموازنة
!>    النظام:
!>
!>      TRANS = 'N':  diag(R)*A*diag(C)     *inv(diag(C))*X = diag(R)*B
!>      TRANS = 'T': (diag(R)*A*diag(C))**T *inv(diag(R))*X = diag(C)*B
!>      TRANS = 'C': (diag(R)*A*diag(C))**H *inv(diag(R))*X = diag(C)*B
!>
!>    سواء تم موازنة النظام أم لا يعتمد على قياس المصفوفة A، ولكن إذا تم استخدام
!>    الموازنة، يتم استبدال A بـ diag(R)*A*diag(C) و B بـ diag(R)*B (إذا كان TRANS='N')
!>    أو diag(C)*B (إذا كان TRANS = 'T' أو 'C').
!>
!>    2. إذا كان FACT = 'N' أو 'E'، يتم استخدام تحليل LU لتحليل
!>    المصفوفة A (بعد الموازنة إذا كان FACT = 'E') كـ
!>
!>      A = P * L * U,
!>
!>    حيث P هي مصفوفة تبديل، L هي مصفوفة ثلاثية سفلية وحدوية، و U هي مصفوفة مثلثة علوية.
!>
!>    3. إذا كان U(i,i)=0 لبعض i، مما يجعل U مفردة تمامًا، فإن
!>    الروتين يعود بـ INFO = i. وإلا، يتم استخدام الشكل المحلل لـ A لتقدير
!>    رقم شرط المصفوفة A (انظر الوسيط RCOND). إذا كان معكوس رقم الشرط أقل
!>    من دقة الآلة، فإن الروتين يستمر في الحل لـ X وحساب حدود الخطأ كما هو موضح أدناه.
!>
!>    4. يتم حل نظام المعادلات لـ X باستخدام الشكل المحلل لـ A.
!>
!>    5. افتراضيًا (ما لم يتم تعيين PARAMS(LA_LINRX_ITREF_I) إلى صفر)،
!>    سيستخدم الروتين التنقيح التكراري لمحاولة الحصول على خطأ صغير وحدود خطأ.
!>    يحسب التنقيح الباقي بدقة مضاعفة على الأقل للدقة العاملة.
!>
!>    6. إذا تم استخدام الموازنة، يتم ضرب المصفوفة X بالمصفوفة القطرية
!>    diag(C) (إذا كان TRANS = 'N') أو diag(R) (إذا كان TRANS = 'T' أو 'C')
!>    بحيث تحل النظام الأصلي قبل الموازنة.
!> 
!>     يتم تجميع بعض المعلمات الاختيارية في مصفوفة PARAMS. تحدد هذه
!>     الإعدادات كيفية إجراء التنقيح، ولكن غالبًا ما تكون القيم الافتراضية مقبولة.
!>     إذا كانت القيم الافتراضية مقبولة، يمكن للمستخدمين تمرير NPARAMS = 0
!>     مما يمنع الكود المصدري من الوصول إلى الوسيط PARAMS.
!>

المعلمات

FACT 
!>          FACT هو CHARACTER*1
!>     يحدد ما إذا كان الشكل المحلل للمصفوفة A مزودًا عند الدخول، وإذا لم يكن كذلك،
!>     ما إذا كان يجب موازنة المصفوفة A قبل تحليلها.
!>       = 'F':  عند الدخول، تحتوي AF و IPIV على الشكل المحلل لـ A.
!>               إذا لم يكن EQUED = 'N'، فقد تم موازنة المصفوفة A
!>               بعوامل القياس المعطاة بواسطة R و C.
!>               لا يتم تعديل A و AF و IPIV.
!>       = 'N':  سيتم نسخ المصفوفة A إلى AF وتحليلها.
!>       = 'E':  سيتم موازنة المصفوفة A إذا لزم الأمر، ثم
!>               نسخها إلى AF وتحليلها.
!>

TRANS

!>          TRANS هو CHARACTER*1
!>     يحدد شكل نظام المعادلات:
!>       = 'N':  A * X = B     (بدون تبديل)
!>       = 'T':  A**T * X = B  (تبديل)
!>       = 'C':  A**H * X = B  (تبديل مترافق = تبديل)
!>

N

!>          N هو INTEGER
!>     عدد المعادلات الخطية، أي رتبة
!>     المصفوفة A.  N >= 0.
!>

KL

!>          KL هو INTEGER
!>     عدد الأقطار الفرعية داخل نطاق A.  KL >= 0.
!>

KU

!>          KU هو INTEGER
!>     عدد الأقطار الفائقة داخل نطاق A.  KU >= 0.
!>

NRHS

!>          NRHS هو INTEGER
!>     عدد الأطراف اليمنى، أي عدد الأعمدة
!>     للمصفوفتين B و X.  NRHS >= 0.
!>

AB

!>          AB هو مصفوفة COMPLEX*16، البعد (LDAB,N)
!>     عند الدخول، المصفوفة A بتخزين النطاق، في الصفوف 1 إلى KL+KU+1.
!>     يتم تخزين العمود j من A في العمود j من
!>     المصفوفة AB كما يلي:
!>     AB(KU+1+i-j,j) = A(i,j) لـ max(1,j-KU)<=i<=min(N,j+kl)
!>
!>     إذا كان FACT = 'F' و EQUED غير 'N'، فيجب أن تكون AB
!>     قد تمت موازنتها بواسطة عوامل القياس في R و/أو C.
!>     لا يتم تعديل AB إذا كان FACT = 'F' أو 'N'، أو إذا كان FACT = 'E' و
!>     EQUED = 'N' عند الخروج.
!>
!>     عند الخروج، إذا كان EQUED .ne. 'N'، يتم قياس A كما يلي:
!>     EQUED = 'R':  A := diag(R) * A
!>     EQUED = 'C':  A := A * diag(C)
!>     EQUED = 'B':  A := diag(R) * A * diag(C).
!>

LDAB

!>          LDAB هو INTEGER
!>     البعد الرائد للمصفوفة AB.  LDAB >= KL+KU+1.
!>

AFB

!>          AFB هو مصفوفة COMPLEX*16، البعد (LDAFB,N)
!>     إذا كان FACT =
'F'، ثم AFB هي وسيطة إدخال وفي الإدخال !> تحتوي على تفاصيل تحليل LU للمصفوفة النطاقية !> A، كما تم حسابها بواسطة ZGBTRF. يتم تخزين U كمصفوفة مثلثية عليا !> نطاقية مع KL+KU أقطار فائقة في الصفوف من 1 إلى KL+KU+1، !> وتُخزّن المضاعفات المستخدمة أثناء التحليل !> في الصفوف من KL+KU+2 إلى 2*KL+KU+1. إذا كان EQUED .ne. 'N'، فإن AFB !> هي الشكل المحلل للمصفوفة المتوازنة A. !> !> إذا كان FACT = 'N'، فإن AF هي وسيطة إخراج وفي الإخراج !> تُعيد العوامل L و U من تحليل A = P*L*U !> للمصفوفة الأصلية A. !> !> إذا كان FACT = 'E'، فإن AF هي وسيطة إخراج وفي الإخراج !> تُعيد العوامل L و U من تحليل A = P*L*U !> للمصفوفة المتوازنة A (انظر وصف A لشكل !> المصفوفة المتوازنة). !>

LDAFB

!>          LDAFB هو عدد صحيح (INTEGER)
!>     البعد الرئيسي للمصفوفة AFB.  LDAFB >= 2*KL+KU+1.

IPIV


!>          IPIV هي مصفوفة أعداد صحيحة (INTEGER array)، ذات بعد (N)
!>     إذا كان FACT = 'F'، فإن IPIV هي وسيطة إدخال وفي الإدخال
!>     تحتوي على مؤشرات الارتكاز من تحليل A = P*L*U
!>     كما تم حسابها بواسطة ZGETRF؛ تم تبديل الصف i من المصفوفة
!>     مع الصف IPIV(i).
!>
!>     إذا كان FACT = 'N'، فإن IPIV هي وسيطة إخراج وفي الإخراج
!>     تُعيد مؤشرات الارتكاز من تحليل A = P*L*U
!>     للمصفوفة الأصلية A.
!>
!>     إذا كان FACT = 'E'، فإن IPIV هي وسيطة إخراج وفي الإخراج
!>     تُعيد مؤشرات الارتكاز من تحليل A = P*L*U
!>     للمصفوفة المتوازنة A.
!>

EQUED


!>          EQUED هو حرف مفرد (CHARACTER*1)
!>     يحدد شكل الموازنة التي تم إجراؤها.
!>       = 'N':  لا توجد موازنة (دائماً صحيح إذا كان FACT = 'N').
!>       = 'R':  موازنة الصفوف، أي تم ضرب A مسبقاً بـ
!>               diag(R).
!>       = 'C':  موازنة الأعمدة، أي تم ضرب A لاحقاً بـ
!>               diag(C).
!>       = 'B':  موازنة الصفوف والأعمدة، أي تم استبدال A بـ
!>               diag(R) * A * diag(C).
!>     EQUED هي وسيطة إدخال إذا كان FACT = 'F'؛ بخلاف ذلك، فهي وسيطة
!>     إخراج.

R


!>          R هي مصفوفة ذات دقة مضاعفة (DOUBLE PRECISION array)، ذات بعد (N)
!>     عوامل مقياس الصفوف لـ A. إذا كان EQUED = 'R' أو 'B'، يتم ضرب A
!>     من اليسار بـ diag(R)؛ إذا كان EQUED = 'N' أو 'C'، فلا يتم الوصول إلى R.
!>     R هي وسيطة إدخال إذا كان FACT = 'F'؛ بخلاف ذلك، R هي وسيطة إخراج.
!>     إذا كان FACT = 'F' و EQUED = 'R' أو 'B'، يجب أن يكون كل عنصر في R موجباً.
!>     إذا كانت R مخرجات، فإن كل عنصر في R هو قوة للجذر.
!>     إذا كانت R مدخلات، يجب أن يكون كل عنصر في R قوة للجذر
!>     لضمان حل موثوق وتقديرات للأخطاء. القياس باستخدام
!>     قوى الجذر لا يسبب أخطاء تقريب إلا إذا كانت النتيجة
!>     تتجاوز الحدود الدنيا أو القصوى. تؤدي أخطاء التقريب أثناء القياس
!>     إلى تحسين مصفوفة ليست مكافئة للمصفوفة المدخلة، مما ينتج عنه تقديرات
!>     للأخطاء قد لا تكون موثوقة.

C


!>          C هي مصفوفة ذات دقة مضاعفة (DOUBLE PRECISION array)، ذات بعد (N)
!>     عوامل مقياس الأعمدة لـ A. إذا كان EQUED = 'C' أو 'B'، يتم ضرب A
!>     من اليمين بـ diag(C)؛ إذا كان EQUED = 'N' أو 'R'، فلا يتم الوصول إلى C.
!>     C هي وسيطة إدخال إذا كان FACT = 'F'؛ بخلاف ذلك، C هي وسيطة إخراج.
!>     إذا كان FACT = 'F' و EQUED = 'C' أو 'B'، يجب أن يكون كل عنصر في C موجباً.
!>     إذا كانت C مخرجات، فإن كل عنصر في C هو قوة للجذر.
!>     إذا كانت C مدخلات، يجب أن يكون كل عنصر في C قوة للجذر
!>     لضمان حل موثوق وتقديرات للأخطاء. القياس باستخدام
!>     قوى الجذر لا يسبب أخطاء تقريب إلا إذا كانت النتيجة
!>     تتجاوز الحدود الدنيا أو القصوى. تؤدي أخطاء التقريب أثناء القياس
!>     إلى تحسين مصفوفة ليست مكافئة للمصفوفة المدخلة، مما ينتج عنه تقديرات
!>     للأخطاء قد لا تكون موثوقة.

B


!>          B هي مصفوفة مركبة مزدوجة (COMPLEX*16 array)، ذات بعد (LDB,NRHS)
!>     عند الإدخال، المصفوفة B ذات الـ NRHS عمود على اليمين.
!>     عند الإخراج،
!>     إذا كان EQUED = 'N'، فلا يتم تعديل B؛
!>     إذا كان TRANS = 'N' و EQUED = 'R' أو 'B'، يتم استبدال B بـ
!>        diag(R)*B؛
!>     إذا كان TRANS = 'T' أو 'C' و EQUED = 'C' أو 'B'، يتم استبدال B بـ
!>        diag(C)*B.

LDB


!>          LDB هو عدد صحيح (INTEGER)
!>     البعد الرئيسي للمصفوفة B.  LDB >= max(1,N).

X


!>          X هي مصفوفة مركبة مزدوجة (COMPLEX*16 array)، ذات بعد (LDX,NRHS)
!>     إذا كان INFO = 0، فإن مصفوفة الحل X ذات الـ NRHS عمود للنظام الأصلي
!>     للمعادلات. لاحظ أن A و B يتم تعديلهما عند الإخراج
!>     إذا كان EQUED .ne. 'N'، وأن حل النظام المتوازن هو
!>     inv(diag(C))*X إذا كان TRANS = 'N' و EQUED = 'C' أو 'B'، أو
!>     inv(diag(R))*X إذا كان TRANS = 'T' أو 'C' و EQUED = 'R' أو 'B'.

LDX


!>          LDX هو عدد صحيح (INTEGER)
!>     البعد الرئيسي للمصفوفة X.  LDX >= max(1,N).

RCOND


!>          RCOND هو دقة مضاعفة (DOUBLE PRECISION)
!>     المقلوب النسبي لرقم الحالة. هذا تقدير لـ
!>     رقم الحالة Skeel النسبي للمصفوفة A بعد
!>     الموازنة (إذا تم إجراؤها). إذا كان هذا أقل من دقة الجهاز
!>     (على وجه الخصوص، إذا كان صفرًا)، فإن المصفوفة منفردة
!>     بدقة العمل. لاحظ أن الخطأ قد لا يزال صغيرًا حتى
!>     إذا كان هذا الرقم صغيرًا جدًا وتبدو المصفوفة متعثرة
!>     الحالة.

RPVGRW


!>          RPVGRW هو دقة مضاعفة (DOUBLE PRECISION)
!>     مقلوب نمو الارتكاز. عند الإخراج، يحتوي هذا على مقلوب
!>     عامل نمو الارتكاز norm(A)/norm(U). يتم استخدام
!>     المعيار. إذا كان هذا أقل بكثير من 1، فإن استقرار
!>     تحليل LU للمصفوفة (المتوازنة) A قد يكون ضعيفًا.
!>     هذا يعني أيضًا أن الحل X، وأرقام الحالة المقدرة،
!>     وحدود الخطأ قد لا تكون موثوقة. إذا فشل التحليل مع
!>     0<INFO<=N، فإن هذا يحتوي على مقلوب عامل نمو الارتكاز
!>     للأعمدة INFO الرائدة من A. في ZGESVX، هذه الكمية
!>     تُعاد في RWORK(1).

BERR


!>          BERR هي مصفوفة دقة مضاعفة (DOUBLE PRECISION array)، ذات بعد (NRHS)
!>     الخطأ النسبي الخلفي المكوني. هذا هو
!>     الخطأ النسبي المكوني لكل متجه حل X(j)
!>     (أي، أصغر تغيير نسبي في أي عنصر من A أو B يجعل X(j) حلاً دقيقًا).

N_ERR_BNDS


!>          N_ERR_BNDS هو عدد صحيح (INTEGER)
!>     عدد حدود الخطأ المطلوب إرجاعها لكل جانب أيمن
!>     ولكل نوع (إجمالي المعيار أو المكوني). انظر ERR_BNDS_NORM و
!>     ERR_BNDS_COMP أدناه.

ERR_BNDS_NORM


!>          ERR_BNDS_NORM هي مصفوفة دقة مضاعفة (DOUBLE PRECISION array)، ذات بعد (NRHS, N_ERR_BNDS)
!>     لكل جانب أيمن، تحتوي هذه المصفوفة على معلومات حول
!>     حدود خطأ وأرقام حالة مختلفة تتعلق بـ
!>     الخطأ النسبي الإجمالي (normwise)، والذي يتم تعريفه على النحو التالي:
!>
!>     الخطأ النسبي الإجمالي في متجه الحل i:
!>             max_j (abs(XTRUE(j,i) - X(j,i)))
!>            ------------------------------
!>                  max_j abs(X(j,i))
!>
!>     يتم فهرسة المصفوفة حسب نوع معلومات الخطأ كما هو موضح
!>     أدناه. يوجد حاليًا ما يصل إلى ثلاث قطع من المعلومات
!>     يتم إرجاعها.
!>
!>     الفهرس الأول في ERR_BNDS_NORM(i,:) يتوافق مع الجانب الأيمن i.
!>
!>     الفهرس الثاني في ERR_BNDS_NORM(:,err) يحتوي على الحقول الثلاثة التالية:
!>     err = 1  قيمة منطقية. ثق بالإجابة إذا كان
!>              المقلوب النسبي لرقم الحالة أقل من الحد
!>              sqrt(n) * dlamch('Epsilon').
!>
!>     err = 2  حد الخطأ: الخطأ الأمامي المقدر،
!>              من المؤكد تقريبًا أنه ضمن عامل 10 من الخطأ الحقيقي
!>              طالما أن الإدخال التالي أكبر من الحد
!>              sqrt(n) * dlamch('Epsilon'). يجب أن يكون حد الخطأ هذا فقط
!>              موثوقًا به إذا كانت القيمة المنطقية السابقة صحيحة.
!>
!>     err = 3  المقلوب النسبي لرقم الحالة: مقلوب رقم الحالة
!>              الإجمالي المقدر. تتم مقارنته بالحد
!>              sqrt(n) * dlamch('Epsilon') لتحديد ما إذا كان تقدير الخطأ
!>              . هذه الأرقام الحالة النسبية هي 1 / (norm(Z^{-1},inf) * norm(Z,inf)) لبعض
!>              المصفوفات Z المتوازنة بشكل مناسب.
!>              لتكن Z = S*A، حيث تقوم S بقياس كل صف بقوة للجذر
!>              بحيث تكون جميع مجاميع الصفوف المطلقة لـ Z حوالي 1.
!>
!>     انظر Lapack Working Note 165 لمزيد من التفاصيل والتحذيرات الإضافية.

ERR_BNDS_COMP


!>          ERR_BNDS_COMP هي مصفوفة دقة مضاعفة (DOUBLE PRECISION array)، ذات بعد (NRHS, N_ERR_BNDS)
!>     لكل جانب أيمن، تحتوي هذه المصفوفة على معلومات حول
!>     حدود خطأ وأرقام حالة مختلفة تتعلق بـ
!>     الخطأ النسبي المكوني، والذي يتم تعريفه على النحو التالي:
!>
!>     الخطأ النسبي المكوني في متجه الحل i:
!>                    abs(XTRUE(j,i) - X(j,i))
!>             max_j ----------------------
!>                         abs(X(j,i))
!>
!>     يتم فهرسة المصفوفة حسب الجانب الأيمن i (الذي يعتمد عليه
!>     الخطأ النسبي المكوني)، ونوع معلومات الخطأ كما هو موضح
!>     أدناه. يوجد حاليًا ما يصل إلى ثلاثة
!>     قطع من المعلومات يتم إرجاعها لكل جانب أيمن. إذا
!>     لم يتم طلب الدقة المكونية (PARAMS(3) = 0.0)، فلا يتم الوصول إلى
!>     ERR_BNDS_COMP. إذا كان N_ERR_BNDS < 3، فلا يتم إرجاع سوى أول (:,N_ERR_BNDS)
!>     مدخل كحد أقصى.
!>
!>     الفهرس الأول في ERR_BNDS_COMP(i,:) يتوافق مع الجانب الأيمن i.
!>
!>     الفهرس الثاني في ERR_BNDS_COMP(:,err) يحتوي على الحقول الثلاثة التالية:
!>     err = 1  قيمة منطقية. ثق بالإجابة إذا كان
!>              المقلوب النسبي لرقم الحالة أقل من الحد
!>              sqrt(n) * dlamch('Epsilon').
!>
!>     err = 2  حد الخطأ: الخطأ الأمامي المقدر،
!>              من المؤكد تقريبًا أنه ضمن عامل 10 من الخطأ الحقيقي
!>              طالما أن الإدخال التالي أكبر من الحد
!>              sqrt(n) * dlamch('Epsilon'). يجب أن يكون حد الخطأ هذا فقط
!>              موثوقًا به إذا كانت القيمة المنطقية السابقة صحيحة.
!>
!>     err = 3  المقلوب النسبي لرقم الحالة: المقلوب النسبي المكوني
!>              المقدر لرقم الحالة. تتم مقارنته بالحد
!>              sqrt(n) * dlamch('Epsilon') لتحديد ما إذا كان تقدير الخطأ

NAME


zgbsvxx —  zgbsvxx computes the solution to a complex general banded
   linear system of equations A * X = B, such as


zgbsvxx


— computes the solution to a complex general banded linear system of equations A * X = B, such as


 A * X = B,



and computing the condition number and the error bounds for each
   right hand side.


A is an N by N banded matrix, by default of bandwidth (ML, MU)
   where ML is the number of subdiagonals and MU is the number of superdiagonals.


Input


  N         (input) INTEGER
            The order of the matrix A. N >= 0.

  KL        (input) INTEGER
            The number of subdiagonals, KL. KL must satisfy KL >= 0 and
            KL < N.

  KU        (input) INTEGER
            The number of superdiagonals, KU. KU must satisfy KU >= 0 and
            KU < N.

  AF        (input) COMPLEX*16 array, dimension (LDDA,N)
            On entry, the unit factor L and the upper triangular band matrix U
            of the block LU factorization of A, as returned by ZGBTF2.

  IAF       (input) INTEGER
            The offset of the first row of A in AF.

  LAF       (input) INTEGER
            The offset of the first column of A in AF.

  LL        (input) INTEGER
            The number of rows of the matrix L in AF. LL = N - KL.

  LU        (input) INTEGER
            The number of columns of the matrix U in AF. LU = N - KU.

  D         (input) COMPLEX*16 array, dimension (min(LL,LU))
            The block diagonal matrix D stored column-by-column.

  ID        (input) INTEGER
            The offset of the first row of D in AF.

  LD        (input) INTEGER
            The offset of the first column of D in AF.

  B         (input) COMPLEX*16 array, dimension (LDB,NRHS)
            The right hand side matrix B.

  IB        (input) INTEGER
            The offset of the first row of B.

  LB        (input) INTEGER
            The offset of the first column of B.

  RHS       (output) COMPLEX*16 array, dimension (LRHS,NRHS)
            The solution matrix X to the system of equations A * X = B.

  IRHS      (output) INTEGER
            The offset of the first row of RHS.

  LRHS      (output) INTEGER
            The offset of the first column of RHS.

  RCOND     (output) DOUBLE PRECISION
            The estimate of the reciprocal condition number of A.
            For F07HAF, RCOND is computed using the pair
            ( LU(1:N,1:N), U(1:N,1:N) ).

  BERR      (output) DOUBLE PRECISION array, dimension (NRHS)
            Approximate error for each right-hand side.
            For each matrix X and the corresponding right hand side vector b,
            BERR(j) is an estimate of the relative error
            ||X - X_computed|| / ||X||, for the solution of the system
            A * X = B.

  ERR_BNDS_NORM (output) DOUBLE PRECISION array, dimension (NRHS, 2)
            Error bounds for each right-hand side.
            For each matrix X and the corresponding right hand side vector b,
            ERR_BNDS_NORM(j,1) is the normwise relative error bound
            ||X - X_computed|| / ||X||, and ERR_BNDS_NORM(j,2) is the
            normwise backward error bound || b - A*X_computed || / || b ||.

  ERR_BNDS_COMP (output) DOUBLE PRECISION array, dimension (NRHS, 2)
            Error bounds for each right-hand side.
            For each matrix X and the corresponding right hand side vector b,
            ERR_BNDS_COMP(j,1) is the componentwise relative error bound
            max | X(i,j) - X_computed(i,j) | / | X(i,j) |, and
            ERR_BNDS_COMP(j,2) is the componentwise backward error bound
            max | b(i) - A*X_computed(i,j) | / | b(i) |.

  PARAMS    (input) DOUBLE PRECISION array, dimension NPARAMS
            Parameters for the algorithm.

  WORK      (workspace) COMPLEX*16 array, dimension (2*N)

  RWORK     (workspace) DOUBLE PRECISION array, dimension (2*N)

  INFO      (output) INTEGER
            = 0: successful exit. The solution to every right hand side is
                 guaranteed.
            < 0: If INFO = -i, the i-th argument had an illegal value.
            > 0 and <= N: U(INFO,INFO) is exactly zero. The factorization has been
                 completed, but the factor U is exactly singular, so the
                 solution and error bounds could not be computed. RCOND = 0
                 is returned.
            = N+J: The solution corresponding to the Jth right hand side is
                 not guaranteed. The solutions corresponding to other right-
                 hand sides K with K > J may not be guaranteed as well, but
                 only the first such right hand side is reported. If a small
                 componentwise error is not requested (PARAMS(3) = 0.0) then
                 the Jth right hand side is the first with a normwise error
                 bound that is not guaranteed (the smallest J such
                 that ERR_BNDS_NORM(J,1) = 0.0). By default (PARAMS(3) = 1.0)
                 the Jth right hand side is the first with either a normwise or
                 componentwise error bound that is not guaranteed (the smallest
                 J such that either ERR_BNDS_NORM(J,1) = 0.0 or
                 ERR_BNDS_COMP(J,1) = 0.0). See the definition of
                 ERR_BNDS_NORM(:,1) and ERR_BNDS_COMP(:,1). To get information
                 about all of the right hand sides check ERR_BNDS_NORM or
                 ERR_BNDS_COMP.
