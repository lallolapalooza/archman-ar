توثيق الوظيفة/الروتين الفرعي

subroutine zgghrd (character compq, character compz, integer n, integer ilo, integer ihi, complex*16, dimension( lda, * ) a, integer lda, complex*16, dimension( ldb, * ) b, integer ldb, complex*16, dimension( ldq, * ) q, integer ldq, complex*16, dimension( ldz, * ) z, integer ldz, integer info)

ZGGHRD

الغرض:

!>
!> ZGGHRD يقلل زوجًا من المصفوفات المعقدة (A,B) إلى الشكل العام لمنحنى حاد (Hessenberg) باستخدام تحويلات وحدوية، حيث A هي
!> مصفوفة عامة و B هي مصفوفة مثلثية عليا. شكل مشكلة القيمة الذاتية المعممة هو
!>    A*x = lambda*B*x,
!> ويتم عادةً جعل B مثلثية عليا عن طريق حساب تحليل QR الخاص بها ونقل المصفوفة الوحدوية Q إلى الجانب الأيسر
!> من المعادلة.
!>
!> هذا الروتين الفرعي يقلل A بشكل متزامن إلى مصفوفة منحنية حادة H:
!>    Q**H*A*Z = H
!> ويحول B إلى مصفوفة مثلثية عليا أخرى T:
!>    Q**H*B*Z = T
!> وذلك لتقليل المشكلة إلى شكلها القياسي
!>    H*y = lambda*T*y
!> حيث y = Z**H*x.
!>
!> يتم تحديد المصفوفات الوحدوية Q و Z كمنتجات لدورانات جيفنز. قد يتم تكوينها بشكل صريح، أو قد يتم
!> ضربها بعديًا في مصفوفات الإدخال Q1 و Z1، بحيث
!>      Q1 * A * Z1**H = (Q1*Q) * H * (Z1*Z)**H
!>      Q1 * B * Z1**H = (Q1*Q) * T * (Z1*Z)**H
!> إذا كانت Q1 هي المصفوفة الوحدوية من تحليل QR الخاص بـ B في المعادلة الأصلية A*x = lambda*B*x، فإن ZGGHRD يقلل المشكلة الأصلية إلى شكل منحنى حاد عام.
!>

المعلمات

COMPQ 
!>          COMPQ is CHARACTER*1
!>          = 'N': لا تحسب Q؛
!>          = 'I': يتم تهيئة Q إلى مصفوفة الوحدة، ويتم إرجاع المصفوفة الوحدوية Q؛
!>          = 'V': يجب أن تحتوي Q على مصفوفة وحدوية Q1 عند الدخول،
!>                 ويتم إرجاع حاصل الضرب Q1*Q.
!>

COMPZ

!>          COMPZ is CHARACTER*1
!>          = 'N': لا تحسب Z؛
!>          = 'I': يتم تهيئة Z إلى مصفوفة الوحدة، ويتم إرجاع المصفوفة الوحدوية Z؛
!>          = 'V': يجب أن تحتوي Z على مصفوفة وحدوية Z1 عند الدخول،
!>                 ويتم إرجاع حاصل الضرب Z1*Z.
!>

N

!>          N is INTEGER
!>          رتبة المصفوفتين A و B. N >= 0.
!>

ILO

!>          ILO is INTEGER
!>

IHI

!>          IHI is INTEGER
!>
!>          ILO و IHI يحددان الصفوف والأعمدة من A التي يجب تقليلها. يُفترض أن A هي بالفعل مصفوفة مثلثية عليا
!>          في الصفوف والأعمدة 1:ILO-1 و IHI+1:N. عادةً ما يتم تعيين ILO و IHI بواسطة استدعاء سابق لـ ZGGBAL؛ وإلا يجب
!>          تعيينهما إلى 1 و N على التوالي.
!>          1 <= ILO <= IHI <= N، إذا كان N > 0؛ ILO=1 و IHI=0، إذا كان N=0.
!>

A

!>          A is COMPLEX*16 array, dimension (LDA, N)
!>          عند الدخول، المصفوفة العامة N*N المراد تقليلها.
!>          عند الخروج، يتم استبدال المثلث العلوي والقطر الفرعي الأول من A
!>          بالمصفوفة المنحنية الحادة H، ويتم تعيين الباقي إلى الصفر.
!>

LDA

!>          LDA is INTEGER
!>          البعد الرائد للمصفوفة A. LDA >= max(1,N).
!>

B

!>          B is COMPLEX*16 array, dimension (LDB, N)
!>          عند الدخول، المصفوفة B المثلثية العليا N*N.
!>          عند الخروج، المصفوفة المثلثية العليا T = Q**H B Z. يتم
!>          تعيين العناصر الموجودة أسفل القطر إلى الصفر.
!>

LDB

!>          LDB is INTEGER
!>          البعد الرائد للمصفوفة B. LDB >= max(1,N).
!>

Q

!>          Q is COMPLEX*16 array, dimension (LDQ, N)
!>          عند الدخول، إذا كان COMPQ = 'V'، المصفوفة الوحدوية Q1، عادةً
!>          من تحليل QR لـ B.
!>          عند الخروج، إذا كان COMPQ='I'، المصفوفة الوحدوية Q، وإذا كان
!>          COMPQ = 'V'، حاصل الضرب Q1*Q.
!>          لا تتم الإشارة إليها إذا كان COMPQ='N'.
!>

LDQ

!>          LDQ is INTEGER
!>          البعد الرائد للمصفوفة Q.
!>          LDQ >= N إذا كان COMPQ='V' أو 'I'؛ LDQ >= 1 بخلاف ذلك.
!>

Z

!>          Z is COMPLEX*16 array, dimension (LDZ, N)
!>          عند الدخول، إذا كان COMPZ = 'V'، المصفوفة الوحدوية Z1.
!>          عند الخروج، إذا كان COMPZ='I'، المصفوفة الوحدوية Z، وإذا كان
!>          COMPZ = 'V'، حاصل الضرب Z1*Z.
!>          لا تتم الإشارة إليها إذا كان COMPZ='N'.
!>

LDZ

!>          LDZ is INTEGER
!>          البعد الرائد للمصفوفة Z.
!>          LDZ >= N إذا كان COMPZ='V' أو 'I'؛ LDZ >= 1 بخلاف ذلك.
!>

INFO

!>          INFO is INTEGER
!>          = 0:  انتهاء ناجح.
!>          < 0:  إذا كان INFO = -i، كان الوسيط i يحتوي على قيمة غير قانونية.
!>
