subroutine zggsvd3 (character jobu, character jobv, character jobq, integer m, integer n, integer p, integer k, integer l, complex*16, dimension( lda, * ) a, integer lda, complex*16, dimension( ldb, * ) b, integer ldb, double precision, dimension( * ) alpha, double precision, dimension( * ) beta, complex*16, dimension( ldu, * ) u, integer ldu, complex*16, dimension( ldv, * ) v, integer ldv, complex*16, dimension( ldq, * ) q, integer ldq, complex*16, dimension( * ) work, integer lwork, double precision, dimension( * ) rwork, integer, dimension( * ) iwork, integer info)

ZGGSVD3 تحسب تفاضل القيمة المفردة المعمم (GSVD) للمصفوفة A المعقدة من نوع M-by-N والمصفوفة B المعقدة من نوع P-by-N:

الغرض:

!>
!> ZGGSVD3 تحسب تفاضل القيمة المفردة المعمم (GSVD)
!> للمصفوفة A المعقدة من نوع M-by-N والمصفوفة B المعقدة من نوع P-by-N:
!>
!>       U**H*A*Q = D1*( 0 R ),    V**H*B*Q = D2*( 0 R )
!>
!> حيث U، V، و Q هي مصفوفات وحدوية.
!> لنفترض K+L = الرتبة العددية الفعالة للمصفوفة
!> (A**H,B**H)**H، ثم R هي مصفوفة غير شاذة مثلثية عليا بحجم (K+L)-by-(K+L)،
!> و D1 و D2 هي مصفوفات بحجم M-by-(K+L) و P-by-(K+L)
!> ذات الهياكل التالية، على التوالي:
!>
!> إذا كان M-K-L >= 0،
!>
!>                     K  L
!>        D1 =     K ( I  0 )
!>                 L ( 0  C )
!>             M-K-L ( 0  0 )
!>
!>                   K  L
!>        D2 =   L ( 0  S )
!>             P-L ( 0  0 )
!>
!>                 N-K-L  K    L
!>   ( 0 R ) =     K (  0   R11  R12 )
!>             L (  0    0   R22 )
!> حيث
!>
!>   C = diag( ALPHA(K+1), ... , ALPHA(K+L) ),
!>   S = diag( BETA(K+1),  ... , BETA(K+L) ),
!>   C**2 + S**2 = I.
!>
!>   يتم تخزين R في A(1:K+L,N-K-L+1:N) عند الخروج.
!>
!> إذا كان M-K-L < 0،
!>
!>                   K M-K K+L-M
!>        D1 =   K ( I  0    0   )
!>             M-K ( 0  C    0   )
!>
!>                     K M-K K+L-M
!>        D2 =   M-K ( 0  S    0  )
!>             K+L-M ( 0  0    I  )
!>               P-L ( 0  0    0  )
!>
!>                    N-K-L  K   M-K  K+L-M
!>   ( 0 R ) =     K ( 0    R11  R12  R13  )
!>               M-K ( 0     0   R22  R23  )
!>             K+L-M ( 0     0    0   R33  )
!>
!> حيث
!>
!>   C = diag( ALPHA(K+1), ... , ALPHA(M) ),
!>   S = diag( BETA(K+1),  ... , BETA(M) ),
!>   C**2 + S**2 = I.
!>
!>   يتم تخزين (R11 R12 R13 ) في A(1:M, N-K-L+1:N)، ويتم تخزين R33
!!>   ( 0  R22 R23 )
!>   في B(M-K+1:L,N+M-K-L+1:N) عند الخروج.
!>
!> تقوم الإجراءات الفرعية بحساب C، S، R، وإجراءات التحويل الوحدوية الاختيارية
!> U، V، و Q.
!>
!> على وجه الخصوص، إذا كانت B مصفوفة غير شاذة بحجم N-by-N، فإن GSVD
!> لـ A و B تعطي ضمنيًا SVD لـ A*inv(B):
!>                      A*inv(B) = U*(D1*inv(D2))*V**H.
!> إذا كان ( A**H,B**H)**H له أعمدة متعامدة، فإن GSVD لـ A و B يتطابق أيضًا
!> مع التحلل المعياري CS لـ A و B. علاوة على ذلك، يمكن استخدام GSVD
!> لاشتقاق حل مشكلة القيمة الذاتية:
!>                      A**H*A x = lambda* B**H*B x.
!> في بعض الأدبيات، يتم تقديم GSVD لـ A و B في الشكل
!>                  U**H*A*X = ( 0 D1 ),   V**H*B*X = ( 0 D2 )
!> حيث U و V متعامدة و X غير شاذة، و D1 و D2 "قطرية".
!> يمكن تحويل صيغة GSVD السابقة إلى الصيغة اللاحقة
!> عن طريق أخذ المصفوفة غير الشاذة X كـ
!>
!>                       X = Q*(  I   0    )
!>                             (  0 inv(R) )
!>

المعلمات

JOBU 
!>          JOBU هو CHARACTER*1
!>          = 'U':  يتم حساب المصفوفة الوحدوية U؛
!>          = 'N':  لا يتم حساب U.
!>

JOBV

!>          JOBV هو CHARACTER*1
!>          = 'V':  يتم حساب المصفوفة الوحدوية V؛
!>          = 'N':  لا يتم حساب V.
!>

JOBQ

!>          JOBQ هو CHARACTER*1
!>          = 'Q':  يتم حساب المصفوفة الوحدوية Q؛
!>          = 'N':  لا يتم حساب Q.
!>

M

!>          M هو INTEGER
!>          عدد صفوف المصفوفة A.  M >= 0.
!>

N

!>          N هو INTEGER
!>          عدد أعمدة المصفوفتين A و B.  N >= 0.
!>

P

!>          P هو INTEGER
!>          عدد صفوف المصفوفة B.  P >= 0.
!>

K

!>          K هو INTEGER
!>

L

!>          L هو INTEGER
!>
!>          عند الخروج، تحدد K و L أبعاد الأجزاء الفرعية
!>          الموصوفة في الغرض.
!>          K + L = الرتبة العددية الفعالة لـ (A**H,B**H)**H.
!>

A

!>          A هو مصفوفة COMPLEX*16، البعد (LDA,N)
!>          عند الدخول، تحتوي A على المصفوفة المثلثية R، أو جزء من R.
!>          انظر الغرض للتفاصيل.
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرائد للمصفوفة A. LDA >= max(1,M).
!>

B

!>          B هو مصفوفة COMPLEX*16، البعد (LDB,N)
!>          عند الدخول، تحتوي B على المصفوفة P-by-N.
!>          عند الخروج، تحتوي B على جزء من المصفوفة المثلثية R إذا
!>          كان M-K-L < 0. انظر الغرض للتفاصيل.
!>

LDB

!>          LDB هو INTEGER
!>          البعد الرائد للمصفوفة B. LDB >= max(1,P).
!>

ALPHA

!>          ALPHA هي مصفوفة DOUBLE PRECISION، البعد (N)
!>

BETA

!>          BETA هي مصفوفة DOUBLE PRECISION، البعد (N)
!>
!>          عند الخروج، تحتوي ALPHA و BETA على أزواج القيم المفردة المعممة
!>          لـ A و B؛
!>            ALPHA(1:K) = 1,
!>            BETA(1:K)  = 0,
!>          وإذا كان M-K-L >= 0،
!>            ALPHA(K+1:K+L) = C,
!>            BETA(K+1:K+L)  = S,
!>          أو إذا كان M-K-L < 0،
!>            ALPHA(K+1:M)=C, ALPHA(M+1:K+L)=0
!>            BETA(K+1:M) =S, BETA(M+1:K+L) =1
!>          و
!>            ALPHA(K+L+1:N) = 0
!>            BETA(K+L+1:N)  = 0
!>

U

!>          U هي مصفوفة COMPLEX*16، البعد (LDU,M)
!>          إذا كان JOBU = 'U'، تحتوي U على المصفوفة الوحدوية U بحجم M-by-M.
!>          إذا كان JOBU = 'N'، لا يتم استخدام U.
!>

LDU

!>          LDU هو INTEGER
!>          البعد الرائد للمصفوفة U. LDU >= max(1,M) إذا
!>          كان JOBU = 'U'. LDU >= 1 خلاف ذلك.
!>

V

!>          V هي مصفوفة COMPLEX*16، البعد (LDV,P)
!>          إذا كان JOBV = 'V'، تحتوي V على المصفوفة الوحدوية V بحجم P-by-P.
!>          إذا كان JOBV = 'N'، لا يتم استخدام V.
!>

LDV

!>          LDV هو INTEGER
!>          البعد الرائد للمصفوفة V. LDV >= max(1,P) إذا
!>          كان JOBV = 'V'. LDV >= 1 خلاف ذلك.
!>

Q

!>          Q هي مصفوفة COMPLEX*16، البعد (LDQ,N)
!>          إذا كان JOBQ = 'Q'، تحتوي Q على المصفوفة الوحدوية Q بحجم N-by-N.
!>          إذا كان JOBQ = 'N'، لا يتم استخدام Q.
!>

LDQ

!>          LDQ هو INTEGER
!>          البعد الرائد للمصفوفة Q. LDQ >= max(1,N) إذا
!>          كان JOBQ = 'Q'. LDQ >= 1 خلاف ذلك.
!>

WORK

!>          WORK هو COMPLEX*16، البعد (LWORK)
!>          عند الخروج، يستخدم WORK لتخزين مخرجات إضافية.
!>

LWORK

!>          LWORK هو INTEGER
!>          حجم مصفوفة العمل WORK.
!>          LWORK >= max(1, M + N + P).
!>          إذا كان LWORK = -1، يتم حساب الأبعاد المثلى لـ WORK.
!>

RWORK

!>          RWORK هو DOUBLE PRECISION، البعد (LRWORK)
!>          عند الخروج، يستخدم RWORK لتخزين مخرجات إضافية.
!>

LRWORK

!>          LRWORK هو INTEGER
!>          حجم مصفوفة العمل RWORK.
!>          LRWORK >= max(1, 2*N).
!>          إذا كان LRWORK = -1، يتم حساب الأبعاد المثلى لـ RWORK.
!>

IWORK

!>          IWORK هو INTEGER، البعد (LIWORK)
!>          عند الخروج، يستخدم IWORK لتخزين مخرجات إضافية.
!>

LIWORK

!>          LIWORK هو INTEGER
!>          حجم مصفوفة العمل IWORK.
!>          LIWORK >= max(1, N).
!>          إذا كان LIWORK = -1، يتم حساب الأبعاد المثلى لـ IWORK.
!>

INFO

!>          INFO هو INTEGER
!>          = 0:  النجاح؛
!>          < 0:  إذا كان المعلمة i-th غير صالحة؛
!>          = 1:  حدث خطأ في حساب SVD؛ قد تكون المصفوفة (A**H, B**H)**H
!>                ذات رتبة أقل من N.
!>
