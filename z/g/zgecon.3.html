توثيق الوظيفة/الإجراء الفرعي

subroutine zgecon (character norm, integer n, complex*16, dimension( lda, * ) a, integer lda, double precision anorm, double precision rcond, complex*16, dimension( * ) work, double precision, dimension( * ) rwork, integer info)

ZGECON

الغرض:

!>
!> ZGECON تقدر معكوس رقم الشرط لمصفوفة عامة معقدة A، إما في المعيار 1-norm أو المعيار اللانهائي infinity-norm، باستخدام
!> التحلل LU الذي تم حسابه بواسطة ZGETRF.
!>
!> يتم الحصول على تقدير لـ norm(inv(A))، ويتم حساب معكوس رقم الشرط كـ
!>    RCOND = 1 / ( norm(A) * norm(inv(A)) ).
!>

المعاملات

NORM 
!>          NORM هو CHARACTER*1
!>          يحدد ما إذا كان رقم الشرط 1-norm أو infinity-norm مطلوبًا:
!>          = '1' أو 'O':  1-norm؛
!>          = 'I':         Infinity-norm.
!>

N

!>          N هو INTEGER
!>          ترتيب المصفوفة A.  N >= 0.
!>

A

!>          A هو مصفوفة COMPLEX*16، البعد (LDA,N)
!>          العوامل L و U من التحلل A = P*L*U
!>          كما تم حسابه بواسطة ZGETRF.
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرائد للمصفوفة A.  LDA >= max(1,N).
!>

ANORM

!>          ANORM هو DOUBLE PRECISION
!>          إذا كان NORM = '1' أو 'O'، فإن المعيار 1-norm للمصفوفة الأصلية A.
!>          إذا كان NORM = 'I'، فإن المعيار اللانهائي infinity-norm للمصفوفة الأصلية A.
!>

RCOND

!>          RCOND هو DOUBLE PRECISION
!>          معكوس رقم شرط المصفوفة A،
!>          محسوب كـ RCOND = 1/(norm(A) * norm(inv(A))).
!>

WORK

!>          WORK هي مصفوفة COMPLEX*16، البعد (2*N)
!>

RWORK

!>          RWORK هي مصفوفة DOUBLE PRECISION، البعد (2*N)
!>

INFO

!>          INFO هو INTEGER
!>          = 0:  خروج ناجح
!>          < 0:  إذا كان INFO = -i، فإن المعامل i كان له قيمة غير قانونية.
!>                قيم NaN غير قانونية لـ ANORM، وتنتشر إلى
!>                المعامل الناتج RCOND.
!>                اللانهاية غير قانونية لـ ANORM، وتنتشر إلى الناتج
!>                المعامل RCOND كـ 0.
!>          = 1:  إذا كان RCOND = NaN، أو
!>                   RCOND = Inf، أو
!>                   المعيار المحسوب لمقلوب A هو 0.
!>                في الحالة الأخيرة، يتم إرجاع RCOND = 0.
!>
