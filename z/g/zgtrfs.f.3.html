subroutine zgtrfs (character trans, integer n, integer nrhs, complex*16, dimension( * ) dl, complex*16, dimension( * ) d, complex*16, dimension( * ) du, complex*16, dimension( * ) dlf, complex*16, dimension( * ) df, complex*16, dimension( * ) duf, complex*16, dimension( * ) du2, integer, dimension( * ) ipiv, complex*16, dimension( ldb, * ) b, integer ldb, complex*16, dimension( ldx, * ) x, integer ldx, double precision, dimension( * ) ferr, double precision, dimension( * ) berr, complex*16, dimension( * ) work, double precision, dimension( * ) rwork, integer info)

ZGTRFS

الغرض:

!>
!> ZGTRFS يحسن الحل المحسوب لنظام المعادلات الخطية
!> عندما تكون مصفوفة المعاملات ثلاثية الأقطار، ويوفر
!> حدود الخطأ وتقديرات الخطأ الخلفي للحل.
!>

المعلمات

TRANS 
!>          TRANS هي CHARACTER*1
!>          تحدد شكل نظام المعادلات:
!>          = 'N':  A * X = B     (بدون تبديل)
!>          = 'T':  A**T * X = B  (تبديل)
!>          = 'C':  A**H * X = B  (تبديل مقترن)
!>

N

!>          N هي INTEGER
!>          رتبة المصفوفة A.  N >= 0.
!>

NRHS

!>          NRHS هي INTEGER
!>          عدد الأطراف اليمنى، أي عدد أعمدة
!>          المصفوفة B.  NRHS >= 0.
!>

DL

!>          DL هي مصفوفة COMPLEX*16، البعد (N-1)
!>          عناصر تحت القطر الفرعي (N-1) للمصفوفة A.
!>

D

!>          D هي مصفوفة COMPLEX*16، البعد (N)
!>          عناصر القطر للمصفوفة A.
!>

DU

!>          DU هي مصفوفة COMPLEX*16، البعد (N-1)
!>          عناصر فوق القطر الفرعي (N-1) للمصفوفة A.
!>

DLF

!>          DLF هي مصفوفة COMPLEX*16، البعد (N-1)
!>          المضاعفات (N-1) التي تحدد المصفوفة L من
!>          تحليل LU للمصفوفة A كما تم حسابه بواسطة ZGTTRF.
!>

DF

!>          DF هي مصفوفة COMPLEX*16، البعد (N)
!>          عناصر القطر N للمصفوفة المثلثية العليا U من
!>          تحليل LU للمصفوفة A.
!>

DUF

!>          DUF هي مصفوفة COMPLEX*16، البعد (N-1)
!>          عناصر القطر الفوقي الثاني (N-1) للمصفوفة U.
!>

DU2

!>          DU2 هي مصفوفة COMPLEX*16، البعد (N-2)
!>          عناصر القطر الفوقي الثالث (N-2) للمصفوفة U.
!>

IPIV

!>          IPIV هي مصفوفة INTEGER، البعد (N)
!>          فهارس المحور؛ من 1 <= i <= n، تم تبديل الصف i من المصفوفة
!>          مع الصف IPIV(i).  سيكون IPIV(i) دائمًا إما
!>          i أو i+1؛ IPIV(i) = i يشير إلى عدم الحاجة إلى تبديل صف.
!>

B

!>          B هي مصفوفة COMPLEX*16، البعد (LDB,NRHS)
!>          مصفوفة الطرف الأيمن B.
!>

LDB

!>          LDB هي INTEGER
!>          البعد الرائد للمصفوفة B.  LDB >= max(1,N).
!>

X

!>          X هي مصفوفة COMPLEX*16، البعد (LDX,NRHS)
!>          عند الإدخال، مصفوفة الحل X، كما تم حسابها بواسطة ZGTTRS.
!>          عند الخروج، مصفوفة الحل المحسنة X.
!>

LDX

!>          LDX هي INTEGER
!>          البعد الرائد للمصفوفة X.  LDX >= max(1,N).
!>

FERR

!>          FERR هي مصفوفة DOUBLE PRECISION، البعد (NRHS)
!>          التقدير لحد الخطأ الأمامي لكل متجه حل
!>          X(j) (العمود j من مصفوفة الحل X).
!>          إذا كانت XTRUE هي الحل الحقيقي المقابل لـ X(j)، فإن FERR(j)
!>          هو حد علوي تقديري لقيمة الجزء الأكبر
!>          من (X(j) - XTRUE) مقسومًا على قيمة الجزء الأكبر
!>          من X(j).  التقدير موثوق به مثل
!>          التقدير لـ RCOND، وهو تقريبًا دائمًا
!>          مبالغة طفيفة في الخطأ الحقيقي.
!>

BERR

!>          BERR هي مصفوفة DOUBLE PRECISION، البعد (NRHS)
!>          الخطأ الخلفي النسبي المكون لكل متجه حل
!>          X(j) (أي أصغر تغيير نسبي في
!>          أي عنصر في A أو B يجعل X(j) حلاً دقيقًا).
!>

WORK

!>          WORK هي مصفوفة COMPLEX*16، البعد (2*N)
!>

RWORK

!>          RWORK هي مصفوفة DOUBLE PRECISION، البعد (N)
!>

INFO

!>          INFO هي INTEGER
!>          = 0:  خروج ناجح
!>          < 0:  إذا كانت INFO = -i، كان المعامل i له قيمة غير قانونية
!>
