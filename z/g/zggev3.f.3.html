subroutine zggev3 (character jobvl, character jobvr, integer n, complex*16, dimension( lda, * ) a, integer lda, complex*16, dimension( ldb, * ) b, integer ldb, complex*16, dimension( * ) alpha, complex*16, dimension( * ) beta, complex*16, dimension( ldvl, * ) vl, integer ldvl, complex*16, dimension( ldvr, * ) vr, integer ldvr, complex*16, dimension( * ) work, integer lwork, double precision, dimension( * ) rwork, integer info)

ZGGEV3 يحسب القيم الذاتية، واختياريا، المتجهات الذاتية اليسرى و/أو اليمنى لمصفوفات GE (خوارزمية محظورة)

الغرض:

!>
!> ZGGEV3 يحسب لأزواج من مصفوفات N-by-N المعقدة غير المتماثلة
!> (A,B)، القيم الذاتية المعممة، واختياريا، المتجهات الذاتية المعممة اليسرى و/أو اليمنى.
!>
!> القيمة الذاتية المعممة لزوج من المصفوفات (A,B) هي عدد قياسي
!> lambda أو نسبة alpha/beta = lambda، بحيث A - lambda*B هو
!> شاذ. عادة ما يتم تمثيله كزوج (alpha,beta)، كما
!> هناك تفسير معقول لـ beta=0، وحتى لكليهما
!> يكونان صفرًا.
!>
!> المتجه الذاتي المعمم الأيمن v(j) المقابل للقيمة الذاتية المعممة
!> lambda(j) لـ (A,B) يحقق
!>
!>              A * v(j) = lambda(j) * B * v(j).
!>
!> المتجه الذاتي المعمم الأيسر u(j) المقابل للقيمة الذاتية المعممة
!> lambda(j) لـ (A,B) يحقق
!>
!>              u(j)**H * A = lambda(j) * u(j)**H * B
!>
!> حيث u(j)**H هو المرافق المتحول لـ u(j).
!>

المعلمات

JOBVL 
!>          JOBVL هو CHARACTER*1
!>          = 'N':  لا تحسب المتجهات الذاتية المعممة اليسرى؛
!>          = 'V':  احسب المتجهات الذاتية المعممة اليسرى.
!>

JOBVR

!>          JOBVR هو CHARACTER*1
!>          = 'N':  لا تحسب المتجهات الذاتية المعممة اليمنى؛
!>          = 'V':  احسب المتجهات الذاتية المعممة اليمنى.
!>

!>          رتبة المصفوفات A، B، VL، و VR.  N >= 0.

!>          A هو مصفوفة COMPLEX*16، البعد (LDA, N)
!>          عند الدخول، المصفوفة A في الزوج (A,B).
!>          عند الخروج، تم استبدال A.
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرئيسي لـ A.  LDA >= max(1,N).
!>

B

!>          B هو مصفوفة COMPLEX*16، البعد (LDB, N)
!>          عند الدخول، المصفوفة B في الزوج (A,B).
!>          عند الخروج، تم استبدال B.
!>

LDB

!>          LDB هو INTEGER
!>          البعد الرئيسي لـ B.  LDB >= max(1,N).
!>

ALPHA

!>          ALPHA هو مصفوفة COMPLEX*16، البعد (N)
!>

BETA

!>          BETA هو مصفوفة COMPLEX*16، البعد (N)
!>          عند الخروج، ALPHA(j)/BETA(j)، j=1,...,N، ستكون
!>          القيم الذاتية المعممة.
!>
!>          ملاحظة: قد تتجاوز الكسور ALPHA(j)/BETA(j) بسهولة أو
!>          تتضخم. وقد تكون BETA(j) حتى صفرًا. وبالتالي، يجب على المستخدم
!>          تجنب حساب النسبة alpha/beta بشكل سطحي.
!>          ومع ذلك، ستكون ALPHA دائمًا أقل من، وعادة ما تكون
!>          مقارنة بـ norm(A) في المقدار، و BETA دائمًا أقل من، وعادة ما تكون
!>          مقارنة بـ norm(B).
!>

VL

!>          VL هو مصفوفة COMPLEX*16، البعد (LDVL,N)
!>          إذا كان JOBVL = 'V'، يتم تخزين المتجهات الذاتية المعممة اليسرى u(j)
!>          واحدة تلو الأخرى في أعمدة VL، بنفس
!>          الترتيب مثل قيمها الذاتية.
!>          يتم قياس كل متجه ذاتي بحيث يكون أكبر مكون لديه
!>          abs(real part) + abs(imag. part) = 1.
!>          غير مرجعي إذا كان JOBVL = 'N'.
!>

LDVL

!>          LDVL هو INTEGER
!>          البعد الرئيسي للمصفوفة VL. LDVL >= 1، و
!>          إذا كان JOBVL = 'V'، فإن LDVL >= N.
!>

VR

!>          VR هو مصفوفة COMPLEX*16، البعد (LDVR,N)
!>          إذا كان JOBVR = 'V'، يتم تخزين المتجهات الذاتية المعممة اليمنى v(j)
!>          واحدة تلو الأخرى في أعمدة VR، بنفس
!>          الترتيب مثل قيمها الذاتية.
!>          يتم قياس كل متجه ذاتي بحيث يكون أكبر مكون لديه
!>          abs(real part) + abs(imag. part) = 1.
!>          غير مرجعي إذا كان JOBVR = 'N'.
!>

LDVR

!>          LDVR هو INTEGER
!>          البعد الرئيسي للمصفوفة VR. LDVR >= 1، و
!>          إذا كان JOBVR = 'V'، فإن LDVR >= N.
!>

WORK

!>          WORK هو مصفوفة COMPLEX*16، البعد (MAX(1,LWORK))
!>          عند الخروج، إذا كان INFO = 0، يعيد WORK(1) قيمة LWORK المثلى.
!>

LWORK

!>          LWORK هو INTEGER
!>          بعد المصفوفة WORK. LWORK >= MAX(1,2*N).
!>          لأداء جيد، يجب أن يكون LWORK بشكل عام أكبر.
!>
!>          إذا كان LWORK = -1، فسيتم افتراض استعلام عن مساحة العمل؛ الروتين
!>          يحسب فقط الحجم الأمثل لمصفوفة WORK، ويعيد
!>          هذه القيمة كالإدخال الأول لمصفوفة WORK، ولا يتم إصدار رسالة خطأ
!>          متعلقة بـ LWORK بواسطة XERBLA.
!>

RWORK

!>          RWORK هو مصفوفة DOUBLE PRECISION، البعد (8*N)
!>

INFO

!>          INFO هو INTEGER
!>          = 0:  خروج ناجح
!>          < 0:  إذا كان INFO = -i، فإن المعلمة i كانت ذات قيمة غير قانونية.
!>          =1,...,N:
!>                فشلت تكرارات QZ. لم يتم حساب المتجهات الذاتية.
!>                ولكن يجب أن تكون ALPHA(j) و BETA(j)
!>                صحيحة لـ j=INFO+1,...,N.
!>          > N:  =N+1: فشل تكرار QZ بخلاف ذلك في ZHGEQZ،
!>                =N+2: خطأ في العودة من ZTGEVC.
!>

    NAME

    zggev3 - computes the generalized eigenvalues for a pair of complex general matrices

    SYNOPSIS

    SUBROUTINE ZGGEV3(JOBVL, JOBVR, N, A, LDA, B, LDB, ALPHAR, ALPHAI, BETA, VL, LDVL, VR, LDVR, WORK, LWORK, INFO)

  CHARACTER JOBVL, JOBVR
  INTEGER N, LDA, LDB, LDVL, LDVR, LWORK, INFO
  DOUBLE PRECISION ALPHAR, ALPHAI
  COMPLEX*16 A, B, VL, VR, WORK
  DOUBLE PRECISION BETA

    Function/Subroutine Documentation

    ZGGEV3 computes for a pair of complex matrices (A, B), the generalized eigenvalues and, optionally, the left and right generalized eigenvectors. The generalized eigenvalues are the solutions to the equation A*x = lambda*B*x, which is equivalent to det(A - lambda*B) = 0.

    Parameters:

    JOBVL (CHARACTER):

    Specifies whether to compute the left generalized eigenvectors. 'N' = No left eigenvectors are computed. 'V' = Left eigenvectors are computed.

    JOBVR (CHARACTER):

    Specifies whether to compute the right generalized eigenvectors. 'N' = No right eigenvectors are computed. 'V' = Right eigenvectors are computed.

    N (INTEGER):

    The order of the matrices A and B. N >= 0.

    A (COMPLEX*16), dimension (LDA, N):

    On entry, the first matrix A. On exit, A is overwritten by its upper triangular representation if JOBVL = 'V' or JOBVR = 'V'.

    LDA (INTEGER):

    The leading dimension of A. LDA >= max(1, N).

    B (COMPLEX*16), dimension (LDB, N):

    On entry, the second matrix B. On exit, B is overwritten by its upper triangular representation if JOBVL = 'V' or JOBVR = 'V'.

    LDB (INTEGER):

    The leading dimension of B. LDB >= max(1, N).

    ALPHAR (DOUBLE PRECISION), dimension (N):

    On exit, the real parts of the complex generalized eigenvalues.

    ALPHAI (DOUBLE PRECISION), dimension (N):

    On exit, the imaginary parts of the complex generalized eigenvalues.

    BETA (DOUBLE PRECISION), dimension (N):

    On exit, the values of beta for the generalized eigenvalues.

    VL (COMPLEX*16), dimension (LDVL, N):

    On exit, if JOBVL = 'V', the left generalized eigenvectors.

    LDVL (INTEGER):

    The leading dimension of VL. LDVL >= N.

    VR (COMPLEX*16), dimension (LDVR, N):

    On exit, if JOBVR = 'V', the right generalized eigenvectors.

    LDVR (INTEGER):

    The leading dimension of VR. LDVR >= N.

    WORK (COMPLEX*16), dimension (LWORK):

    Workspace array. On exit, WORK(1) contains the minimum required dimension of WORK.

    LWORK (INTEGER):

    The dimension of the array WORK. LWORK >= 1.

    INFO (INTEGER):

    On exit, INFO = 0 if the execution is successful. INFO < 0 if INFO = -i, then the i-th argument had an illegal value. INFO > 0 if the algorithm failed to converge.
