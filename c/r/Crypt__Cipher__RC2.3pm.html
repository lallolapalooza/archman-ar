Crypt__Cipher__RC2(3pm)

الحزمة: perl-cryptx

التراخيص: PerlArtistic, GPL

Crypt::Cipher::RC2(3) وثائق مقدمة من المستخدم لـ Perl Crypt::Cipher::RC2(3)

الاسم

Crypt::Cipher::RC2 - تشفير متماثل RC2، حجم المفتاح: 40-1024 بت

الملخص

### مثال 1
use Crypt::Mode::CBC;
my $key = '...'; # يجب أن يكون طول المفتاح حجم مفتاح صالح لهذا التشفير
my $iv = '...';  # 16 بايت
my $cbc = Crypt::Mode::CBC->new('RC2');
my $ciphertext = $cbc->encrypt("بيانات سرية", $key, $iv);
### مثال 2 (أبطأ)
use Crypt::CBC;
use Crypt::Cipher::RC2;
my $key = '...'; # يجب أن يكون طول المفتاح حجم مفتاح صالح لهذا التشفير
my $iv = '...';  # 16 بايت
my $cbc = Crypt::CBC->new( -cipher=>'Cipher::RC2', -key=>$key, -iv=>$iv );
my $ciphertext = $cbc->encrypt("بيانات سرية");

الوصف

تنفذ هذه الوحدة تشفير RC2. الواجهة المقدمة متوافقة مع وحدة Crypt::CBC.

تحذير: تنفذ هذه الوحدة فقط عملية "تشفير/فك تشفير" أساسية لـ "كتلة واحدة" - إذا كنت ترغب في تشفير/فك تشفير بيانات عامة، يجب عليك استخدام بعض أوضاع كتل التشفير - تحقق على سبيل المثال من Crypt::Mode::CBC، Crypt::Mode::CTR أو Crypt::CBC (والذي سيكون أبطأ).

الطرق

جديد

$c = Crypt::Cipher::RC2->new($key);
# أو
$c = Crypt::Cipher::RC2->new($key, $rounds);

تشفير

$ciphertext = $c->encrypt($plaintext);

فك تشفير

$plaintext = $c->decrypt($ciphertext);

حجم المفتاح

$c->keysize;
# أو
Crypt::Cipher::RC2->keysize;
# أو
Crypt::Cipher::RC2::keysize;

حجم الكتلة

$c->blocksize;
# أو
Crypt::Cipher::RC2->blocksize;
# أو
Crypt::Cipher::RC2::blocksize;

أقصى حجم للمفتاح

$c->max_keysize;
# أو
Crypt::Cipher::RC2->max_keysize;
# أو
Crypt::Cipher::RC2::max_keysize;

أدنى حجم للمفتاح

$c->min_keysize;
# أو
Crypt::Cipher::RC2->min_keysize;
# أو
Crypt::Cipher::RC2::min_keysize;

الجولات الافتراضية

$c->default_rounds;
# أو
Crypt::Cipher::RC2->default_rounds;
# أو
Crypt::Cipher::RC2::default_rounds;
