الحزمة: perl-cryptx

التراخيص: PerlArtistic, GPL

Crypt::Stream::Sosemanuk(3) وثائق Perl المقدمة من المستخدم Crypt::Stream::Sosemanuk(3)

الاسم

Crypt::Stream::Sosemanuk - تشفير الدفق Sosemanuk

الملخص

use Crypt::Stream::Sosemanuk;
# تشفير
$key = "1234567890123456";
$iv  = "123456789012";
$stream = Crypt::Stream::Sosemanuk->new($key, $iv);
$ct = $stream->crypt("plain message");
# فك تشفير
$key = "1234567890123456";
$iv  = "123456789012";
$stream = Crypt::Stream::Sosemanuk->new($key, $iv);
$pt = $stream->crypt($ct);

الوصف

يوفر واجهة لتشفير الدفق Sosemanuk.

الأساليب

جديد

$stream = Crypt::Stream::Sosemanuk->new($key, $iv);
# $key .. يجب أن يكون طول المفتاح مضاعفاً للبايتات 4
# $iv  .. يجب أن يكون طول المتجه المجهول مضاعفاً للبايتات 4 (اختياري)

تشفير

$ciphertext = $stream->crypt($plaintext);
# أو
$plaintext = $stream->crypt($ciphertext);

دفق المفاتيح

$random_key = $stream->keystream($length);

استنساخ

$stream->clone();
