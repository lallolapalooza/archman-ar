Crypt__AuthEnc__CCM(3pm)

الحزمة: perl-cryptx

التراخيص: PerlArtistic, GPL

Crypt::AuthEnc::CCM(3) وثائق Perl المقدمة من المستخدم Crypt::AuthEnc::CCM(3)

الاسم

Crypt::AuthEnc::CCM - التشفير الموثق في وضع CCM

الملخص

### واجهة الكائنات
use Crypt::AuthEnc::CCM;
# تشفير وتوثيق
my $ae = Crypt::AuthEnc::CCM->new("AES", $key, $iv, $adata, $tag_len, $pt_len);
my $ct = $ae->encrypt_add('data1');
$ct .= $ae->encrypt_add('data2');
$ct .= $ae->encrypt_add('data3');
my $tag = $ae->encrypt_done();
# فك تشفير والتحقق
my $ae = Crypt::AuthEnc::CCM->new("AES", $key, $iv, $adata, $tag_len, $pt_len);
my $pt = $ae->decrypt_add('ciphertext1');
$pt .= $ae->decrypt_add('ciphertext2');
$pt .= $ae->decrypt_add('ciphertext3');
my $tag = $ae->decrypt_done();
die "decrypt failed" unless $tag eq $expected_tag;
# أو
my $result = $ae->decrypt_done($expected_tag); # 0 أو 1
### واجهة الدوال
use Crypt::AuthEnc::CCM qw(ccm_encrypt_authenticate ccm_decrypt_verify);
($ciphertext, $tag) = ccm_encrypt_authenticate('AES', $key, $nonce, $adata, $tag_len, $plaintext);
$plaintext = ccm_decrypt_verify('AES', $key, $nonce, $adata, $ciphertext, $tag);

الوصف

CCM هو وضع تشفير + توثيق يركز على استخدام AES (أو أي تشفير بحجم 16 بايت) كعنصر أساسي. على عكس وضعي EAX و OCB، فهو مخصص فقط لوضع الحزم حيث يتم معرفة طول المدخلات مسبقًا.

تصدير

لا يتم تصدير أي شيء افتراضيًا.

يمكنك تصدير دوال محددة:

use Crypt::AuthEnc::CCM qw(ccm_encrypt_authenticate ccm_decrypt_verify);

الدوال

ccm_encrypt_authenticate

my ($ciphertext, $tag) = ccm_encrypt_authenticate($cipher, $key, $nonce, $adata, $tag_len, $plaintext);
# $cipher .. 'AES' أو اسم أي تشفير آخر بطول كتلة 16 بايت
# $key ..... مفتاح بطول مناسب (مثل 128/192/256 بت لـ AES)
# $nonce ... معرف فريد/ملح (لا حاجة لإبقائه سريًا)
# $adata ... بيانات مصادق عليها إضافية
# $tag_len . الطول المطلوب لعلامة الإخراج

يجب أن تتبع معلمات CCM http://nvlpubs.nist.gov/nistpubs/Legacy/SP/nistspecialpublication800-38c.pdf

# طول العلامة:   4، 6، 8، 10، 12، 14، 16 (الحد الأدنى المعقول هو 8)
# طول المعرّف: 7، 8، 9، 10، 11، 12، 13 (إذا لم تكن متأكدًا، استخدم 11)
# انتبه: طول المعرّف يحدد الحد الأقصى لحجم البيانات المشفرة/المفكوكة: max_data_size = 2^(8*(15-nonce_len))

ccm_decrypt_verify

my $plaintext = ccm_decrypt_verify($cipher, $key, $nonce, $adata, $ciphertext, $tag);
# في حالة الخطأ، يعيد undef

الأساليب

new

my $ae = Crypt::AuthEnc::CCM->new($cipher, $key, $nonce, $adata, $tag_len, $pt_len);
# $cipher .. 'AES' أو اسم أي تشفير آخر بطول كتلة 16 بايت
# $key ..... مفتاح بطول مناسب (مثل 128/192/256 بت لـ AES)
# $nonce ... معرف فريد/ملح (لا حاجة لإبقائه سريًا)
# $adata ... بيانات مصادق عليها إضافية
# $tag_len . الطول المطلوب لعلامة الإخراج
# $pt_len .. الطول المتوقع للنص العادي/النص المشفر للتشفير/فك التشفير

encrypt_add

$ciphertext = $ae->encrypt_add($data);        # يمكن استدعاؤها عدة مرات

encrypt_done

my $tag = $ae->encrypt_done;                  # يعيد قيمة $tag

decrypt_add

$plaintext = $ae->decrypt_add($ciphertext);   # يمكن استدعاؤها عدة مرات

decrypt_done

my $tag = $ae->decrypt_done;           # يعيد قيمة $tag
# أو
my $result = $ae->decrypt_done($tag);  # يعيد 1 (نجاح) أو 0 (فشل)

clone

my $ae_new = $ae->clone;
