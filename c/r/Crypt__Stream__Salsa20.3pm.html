Crypt__Stream__Salsa20(3pm)

الحزمة: perl-cryptx

التراخيص: PerlArtistic, GPL

Crypt::Stream::Salsa20(3) وثائق Perl المساهمة من المستخدم Crypt::Stream::Salsa20(3)

الاسم

Crypt::Stream::Salsa20 - تشفير التدفق Salsa20

الملخص

use Crypt::Stream::Salsa20;
# تشفير
$key = "1234567890123456";
$iv  = "12345678";
$stream = Crypt::Stream::Salsa20->new($key, $iv);
$ct = $stream->crypt("رسالة واضحة");
# فك التشفير
$key = "1234567890123456";
$iv  = "12345678";
$stream = Crypt::Stream::Salsa20->new($key, $iv);
$pt = $stream->crypt($ct);

الوصف

يوفر واجهة لتشفير التدفق Salsa20.

الطرق

جديد

$stream = Crypt::Stream::Salsa20->new($key, $iv);
# أو
$stream = Crypt::Stream::Salsa20->new($key, $iv, $counter, $rounds);
# $key     .. 32 أو 16 بايت
# $iv      .. 8 بايت
# $counter .. قيمة العداد الأولية (الافتراضي: 0)
# $rounds  .. عدد الجولات (الافتراضي: 20)

تشفير/فك تشفير

$ciphertext = $stream->crypt($plaintext);
# أو
$plaintext = $stream->crypt($ciphertext);

تدفق المفتاح

$random_key = $stream->keystream($length);

استنساخ

$stream->clone();
