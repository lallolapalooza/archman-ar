وثائق الوظيفة/الروتين الفرعي

subroutine crqt03 (integer m, integer n, integer k, complex, dimension( lda, * ) af, complex, dimension( lda, * ) c, complex, dimension( lda, * ) cc, complex, dimension( lda, * ) q, integer lda, complex, dimension( * ) tau, complex, dimension( lwork ) work, integer lwork, real, dimension( * ) rwork, real, dimension( * ) result)

CRQT03

الغرض:

!>
!> CRQT03 تختبر CUNMRQ، التي تحسب Q*C، Q'*C، C*Q أو C*Q'.
!>
!> تقارن CRQT03 نتائج استدعاء CUNMRQ مع نتائج
!> تشكيل Q صراحةً باستدعاء CUNGRQ ثم أداء ضرب المصفوفات
!> باستدعاء CGEMM.
!>

المعلمات

M 
!>          M هو عدد صحيح (INTEGER)
!>          عدد الصفوف أو الأعمدة للمصفوفة C؛ C هي n-by-m إذا
!>          تم تطبيق Q من اليسار، أو m-by-n إذا
!>          تم تطبيق Q من اليمين. M >= 0.
!>

N

!>          N هو عدد صحيح (INTEGER)
!>          رتبة المصفوفة المتعامدة Q. N >= 0.
!>

K

!>          K هو عدد صحيح (INTEGER)
!>          عدد المقذوفات الأولية التي يُعرّف حاصل ضربها
!>          المصفوفة المتعامدة Q. N >= K >= 0.
!>

AF

!>          AF هي مصفوفة مركبة (COMPLEX)، الأبعاد (LDA,N)
!>          تفاصيل تحليل RQ لمصفوفة m-by-n، كما
!>          تم إرجاعها بواسطة CGERQF. انظر CGERQF لمزيد من التفاصيل.
!>

C

!>          C هي مصفوفة مركبة (COMPLEX)، الأبعاد (LDA,N)
!>

CC

!>          CC هي مصفوفة مركبة (COMPLEX)، الأبعاد (LDA,N)
!>

Q

!>          Q هي مصفوفة مركبة (COMPLEX)، الأبعاد (LDA,N)
!>

LDA

!>          LDA هو عدد صحيح (INTEGER)
!>          البعد الرائد للمصفوفات AF، C، CC، و Q.
!>

TAU

!>          TAU هي مصفوفة مركبة (COMPLEX)، الأبعاد (min(M,N))
!>          العوامل القياسية للمقذوفات الأولية المقابلة
!>          لتحليل RQ في AF.
!>

WORK

!>          WORK هي مصفوفة مركبة (COMPLEX)، الأبعاد (LWORK)
!>

LWORK

!>          LWORK هو عدد صحيح (INTEGER)
!>          طول WORK. يجب أن يكون LWORK على الأقل M، ويجب أن يكون
!>          M*NB، حيث NB هو حجم الكتلة لهذا البيئة.
!>

RWORK

!>          RWORK هي مصفوفة حقيقية (REAL)، الأبعاد (M)
!>

RESULT

!>          RESULT هي مصفوفة حقيقية (REAL)، الأبعاد (4)
!>          تقارن نسب الاختبار بين تقنيتين لضرب مصفوفة عشوائية C
!>          بمصفوفة متعامدة n-by-n Q.
!>          RESULT(1) = norm( Q*C - Q*C )  / ( N * norm(C) * EPS )
!>          RESULT(2) = norm( C*Q - C*Q )  / ( N * norm(C) * EPS )
!>          RESULT(3) = norm( Q'*C - Q'*C )/ ( N * norm(C) * EPS )
!>          RESULT(4) = norm( C*Q' - C*Q' )/ ( N * norm(C) * EPS )
!>
