وثائق الوظيفة/الروتين الفرعي

subroutine crqt01 (integer m, integer n, complex, dimension( lda, * ) a, complex, dimension( lda, * ) af, complex, dimension( lda, * ) q, complex, dimension( lda, * ) r, integer lda, complex, dimension( * ) tau, complex, dimension( lwork ) work, integer lwork, real, dimension( * ) rwork, real, dimension( * ) result)

CRQT01

الغرض:

!>
!> CRQT01 تختبر CGERQF، الذي يحسب تحلل RQ لمصفوفة m-by-n
!> A، وتختبر جزئيًا CUNGRQ الذي يشكل مصفوفة Q العمودية n-by-n.
!>
!> CRQT01 تقارن R مع A*Q'، وتتحقق من أن Q متعامدة.
!>

المعاملات

M 
!>          M هو عدد صحيح
!>          عدد صفوف المصفوفة A.  M >= 0.
!>

N

!>          N هو عدد صحيح
!>          عدد أعمدة المصفوفة A.  N >= 0.
!>

A

!>          A هي مصفوفة معقدة، البعد (LDA,N)
!>          المصفوفة m-by-n A.
!>

AF

!>          AF هي مصفوفة معقدة، البعد (LDA,N)
!>          تفاصيل تحلل RQ للمصفوفة A، كما تم إرجاعها بواسطة CGERQF.
!>          انظر CGERQF لمزيد من التفاصيل.
!>

Q

!>          Q هي مصفوفة معقدة، البعد (LDA,N)
!>          المصفوفة المتعامدة n-by-n Q.
!>

R

!>          R هي مصفوفة معقدة، البعد (LDA,max(M,N))
!>

LDA

!>          LDA هو عدد صحيح
!>          البعد الرائد للمصفوفات A، AF، Q و L.
!>          LDA >= max(M,N).
!>

TAU

!>          TAU هي مصفوفة معقدة، البعد (min(M,N))
!>          العوامل القياسية للمُنشئات الأولية، كما تم إرجاعها
!>          بواسطة CGERQF.
!>

WORK

!>          WORK هي مصفوفة معقدة، البعد (LWORK)
!>

LWORK

!>          LWORK هو عدد صحيح
!>          بعد المصفوفة WORK.
!>

RWORK

!>          RWORK هي مصفوفة حقيقية، البعد (max(M,N))
!>

RESULT

!>          RESULT هي مصفوفة حقيقية، البعد (2)
!>          نسب الاختبار:
!>          RESULT(1) = norm( R - A*Q' ) / ( N * norm(A) * EPS )
!>          RESULT(2) = norm( I - Q*Q' ) / ( N * EPS )
!>
