CREATE_EVENT_TRIGGER(7)

الحزمة: postgresql

التراخيص: PostgreSQL

CREATE EVENT TRIGGER(7) توثيق PostgreSQL 18.1 CREATE EVENT TRIGGER(7)

الاسم

CREATE_EVENT_TRIGGER - تعريف مشغل حدث جديد

الملخص

CREATE EVENT TRIGGER name
    ON event
    [ WHEN filter_variable IN (filter_value [, ... ]) [ AND ... ] ]
    EXECUTE { FUNCTION | PROCEDURE } function_name()

الوصف

CREATE EVENT TRIGGER ينشئ مشغل حدث جديد. في كل مرة يحدث فيها الحدث المحدد ويتم استيفاء شرط WHEN المرتبط بالمشغل، إن وجد، سيتم تنفيذ دالة المشغل. للحصول على مقدمة عامة لمشغلات الأحداث، انظر الفصل 38. يصبح المستخدم الذي ينشئ مشغل حدث هو مالكه.

المعاملات

name

الاسم المراد إعطاؤه للمشغل الجديد. يجب أن يكون هذا الاسم فريدًا داخل قاعدة البيانات.

event

اسم الحدث الذي يؤدي إلى استدعاء الدالة المعطاة. انظر القسم 38.1 لمزيد من المعلومات حول أسماء الأحداث.

filter_variable

اسم متغير يستخدم لتصفية الأحداث. هذا يجعل من الممكن تقييد تفعيل المشغل إلى مجموعة فرعية من الحالات التي يدعم فيها. حاليًا، المتغير الوحيد المدعوم هو TAG.

filter_value

قائمة بالقيم للمتغير filter_variable المرتبط به الذي يجب أن يتفعل المشغل من أجله. بالنسبة لـ TAG، يعني هذا قائمة بعلامات الأوامر (على سبيل المثال، 'DROP FUNCTION').

function_name

دالة يوفرها المستخدم تم الإعلان عنها بأنها لا تأخذ أي وسيط وتعيد النوع event_trigger.

في صيغة CREATE EVENT TRIGGER، الكلمتان FUNCTION و PROCEDURE متكافئتان، ولكن الدالة المشار إليها يجب أن تكون دالة في أي حال، وليس إجراء. استخدام الكلمة الإجرائية هنا تاريخي وقد تم إهماله.

ملاحظات

يمكن فقط للمستخدمين الخارقين إنشاء مشغلات أحداث.

يتم تعطيل مشغلات الأحداث في الوضع أحادي المستخدم (انظر postgres(1)) وكذلك عندما يتم تعيين event_triggers إلى false. إذا قامت مشغلات الأحداث الخاطئة بتعطيل قاعدة البيانات بشكل كبير لدرجة أنه لا يمكنك حتى إسقاط المشغل، أعد التشغيل مع تعيين event_triggers إلى false لتعطيل مشغلات الأحداث مؤقتًا، أو في الوضع أحادي المستخدم، وستتمكن من القيام بذلك.

أمثلة

منع تنفيذ أي أمر DDL:

CREATE OR REPLACE FUNCTION abort_any_command()
  RETURNS event_trigger
 LANGUAGE plpgsql
  AS $$
BEGIN
  RAISE EXCEPTION 'command % is disabled', tg_tag;
END;
$$;
CREATE EVENT TRIGGER abort_ddl ON ddl_command_start
   EXECUTE FUNCTION abort_any_command();

التوافق

لا توجد عبارة CREATE EVENT TRIGGER في معيار SQL.

انظر أيضًا

ALTER EVENT TRIGGER (ALTER_EVENT_TRIGGER(7))، DROP EVENT TRIGGER (DROP_EVENT_TRIGGER(7))، CREATE FUNCTION (CREATE_FUNCTION(7))
