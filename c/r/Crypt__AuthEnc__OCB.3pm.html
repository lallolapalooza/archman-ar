Crypt__AuthEnc__OCB(3pm)

الحزمة: perl-cryptx

التراخيص: PerlArtistic, GPL

Crypt::AuthEnc::OCB(3) وثائق مقدمة من المستخدم لـ Perl Crypt::AuthEnc::OCB(3)

الاسم

Crypt::AuthEnc::OCB - التشفير الموثق بوضع OCBv3

الملخص

### واجهة الكائنات
use Crypt::AuthEnc::OCB;
# تشفير وتوثيق
my $ae = Crypt::AuthEnc::OCB->new("AES", $key, $nonce, $tag_len);
$ae->adata_add('additional_authenticated_data1');
$ae->adata_add('additional_authenticated_data2');
my $ct = $ae->encrypt_add('data1');
$ct .= $ae->encrypt_add('data2');
$ct .= $ae->encrypt_add('data3');
$ct .= $ae->encrypt_last('rest of data');
my $tag = $ae->encrypt_done();
# فك التشفير والتحقق
my $ae = Crypt::AuthEnc::OCB->new("AES", $key, $nonce, $tag_len);
$ae->adata_add('additional_authenticated_data1');
$ae->adata_add('additional_authenticated_data2');
my $pt = $ae->decrypt_add('ciphertext1');
$pt .= $ae->decrypt_add('ciphertext2');
$pt .= $ae->decrypt_add('ciphertext3');
$pt .= $ae->decrypt_last('rest of data');
my $tag = $ae->decrypt_done();
die "decrypt failed" unless $tag eq $expected_tag;
# أو
my $result = $ae->decrypt_done($expected_tag); # 0 أو 1
### واجهة الوظائف
use Crypt::AuthEnc::OCB qw(ocb_encrypt_authenticate ocb_decrypt_verify);
my ($ciphertext, $tag) = ocb_encrypt_authenticate('AES', $key, $nonce, $adata, $tag_len, $plaintext);
my $plaintext = ocb_decrypt_verify('AES', $key, $nonce, $adata, $ciphertext, $tag);

الوصف

تنفذ هذه الوحدة OCB v3 وفقًا لـ https://tools.ietf.org/html/rfc7253

تصدير

لا يتم تصدير أي شيء افتراضيًا.

يمكنك تصدير وظائف محددة:

use Crypt::AuthEnc::OCB qw(ocb_encrypt_authenticate ocb_decrypt_verify);

الوظائف

ocb_encrypt_authenticate

my ($ciphertext, $tag) = ocb_encrypt_authenticate($cipher, $key, $nonce, $adata, $tag_len, $plaintext);
# $cipher .. 'AES' أو اسم أي خوارزمية تشفير أخرى ذات طول كتلة 16 بايت
# $key ..... مفتاح AES بالطول المناسب (128/192/256 بت)
# $nonce ... قيمة فريدة (لا حاجة لإبقائها سرية)
# $adata ... بيانات مصادق عليها إضافية
# $tag_len . طول العلامة المطلوب للمخرجات

ocb_decrypt_verify

my $plaintext = ocb_decrypt_verify($cipher, $key, $nonce, $adata, $ciphertext, $tag);
# عند الخطأ، تعيد undef

الأساليب

new

my $ae = Crypt::AuthEnc::OCB->new($cipher, $key, $nonce, $tag_len);
# $cipher .. 'AES' أو اسم أي خوارزمية تشفير أخرى ذات طول كتلة 16 بايت
# $key ..... مفتاح AES بالطول المناسب (128/192/256 بت)
# $nonce ... قيمة فريدة (لا حاجة لإبقائها سرية)
# $tag_len . طول العلامة المطلوب للمخرجات

adata_add

$ae->adata_add($adata);                        # يمكن استدعاؤها عدة مرات

encrypt_add

$ciphertext = $ae->encrypt_add($data);         # يمكن استدعاؤها عدة مرات
# تنبيه: يجب أن يكون حجم $data مضاعفًا لطول الكتلة (16 لـ AES)

encrypt_last

$ciphertext = $ae->encrypt_last($data);

encrypt_done

$tag = $ae->encrypt_done();                    # تعيد قيمة $tag

decrypt_add

$plaintext = $ae->decrypt_add($ciphertext);    # يمكن استدعاؤها عدة مرات
# تنبيه: يجب أن يكون حجم $ciphertext مضاعفًا لطول الكتلة (16 لـ AES)

decrypt_last

$plaintext = $ae->decrypt_last($data);

decrypt_done

my $tag = $ae->decrypt_done;           # تعيد قيمة $tag
# أو
my $result = $ae->decrypt_done($tag);  # تعيد 1 (نجاح) أو 0 (فشل)

clone

my $ae_new = $ae->clone;
