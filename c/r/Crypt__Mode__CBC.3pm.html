Crypt__Mode__CBC(3pm)

الحزمة: perl-cryptx

التراخيص: PerlArtistic, GPL

Crypt::Mode::CBC(3) وثائق بيرل المساهمة من المستخدم Crypt::Mode::CBC(3)

الاسم

Crypt::Mode::CBC - وضع تشفير الكتلة CBC [سلسلة كتل التشفير]

الملخص

use Crypt::Mode::CBC;
my $m = Crypt::Mode::CBC->new('AES');
#(تشفير|فك تشفير) دفعة واحدة
my $ciphertext = $m->encrypt($plaintext, $key, $iv);
my $plaintext = $m->decrypt($ciphertext, $key, $iv);
#تشفير المزيد من الأجزاء
$m->start_encrypt($key, $iv);
my $ciphertext = $m->add('some data');
$ciphertext .= $m->add('more data');
$ciphertext .= $m->finish;
#فك تشفير المزيد من الأجزاء
$m->start_decrypt($key, $iv);
my $plaintext = $m->add($some_ciphertext);
$plaintext .= $m->add($more_ciphertext);
$plaintext .= $m->finish;

الوصف

تطبق هذه الوحدة وضع تشفير CBC. **ملاحظة:** تعمل فقط مع التشفيرات من CryptX (Crypt::Cipher::NNNN).

الوظائف

جديد

my $m = Crypt::Mode::CBC->new($name);
#أو
my $m = Crypt::Mode::CBC->new($name, $padding);
#أو
my $m = Crypt::Mode::CBC->new($name, $padding, $cipher_rounds);
# $name ....... واحد من 'AES', 'Anubis', 'Blowfish', 'CAST5', 'Camellia', 'DES', 'DES_EDE',
#               'KASUMI', 'Khazad', 'MULTI2', 'Noekeon', 'RC2', 'RC5', 'RC6',
#               'SAFERP', 'SAFER_K128', 'SAFER_K64', 'SAFER_SK128', 'SAFER_SK64',
#               'SEED', 'Skipjack', 'Twofish', 'XTEA', 'IDEA', 'Serpent'
#               ببساطة أي <اسم> يوجد له Crypt::Cipher::<اسم>
# $padding .... 0 بدون حشو (يجب أن يكون حجم النص العادي مضاعفًا لطول الكتلة)
#               1 حشو PKCS5، "قياسي" لـ Crypt::CBC - افتراضي
#               2 "واحد وأصفار" لـ Crypt::CBC
#               3 حشو ANSI X.923
#               4 حشو الأصفار
#               5 حشو الأصفار (+كتلة أصفار إذا كان طول الإخراج قابلاً للقسمة على حجم الكتلة)
# $cipher_rounds ... عدد دورات اختياري للتشفير المحدد

تشفير

my $ciphertext = $m->encrypt($plaintext, $key, $iv);

فك تشفير

my $plaintext = $m->decrypt($ciphertext, $key, $iv);

بدء التشفير

$m->start_encrypt($key, $iv);

بدء فك التشفير

$m->start_decrypt($key, $iv);

إضافة

# في وضع التشفير
my $plaintext = $m->add($ciphertext);
# في وضع فك التشفير
my $ciphertext = $m->add($plaintext);

إنهاء

#تشفير المزيد من الأجزاء
$m->start_encrypt($key, $iv);
my $ciphertext = '';
$ciphertext .= $m->add('some data');
$ciphertext .= $m->add('more data');
$ciphertext .= $m->finish;
#فك تشفير المزيد من الأجزاء
$m->start_decrypt($key, $iv);
my $plaintext = '';
$plaintext .= $m->add($some_ciphertext);
$plaintext .= $m->add($more_ciphertext);
$plaintext .= $m->finish;
