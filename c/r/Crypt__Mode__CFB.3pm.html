Crypt__Mode__CFB(3pm)

الحزمة: perl-cryptx

التراخيص: PerlArtistic, GPL

Crypt::Mode::CFB(3) وثائق مقدمة من المستخدم لـ Perl Crypt::Mode::CFB(3)

الاسم

Crypt::Mode::CFB - وضع التشفير CFB [تغذية التشفير الراجعة]

الملخص

use Crypt::Mode::CFB;
my $m = Crypt::Mode::CFB->new('AES');
#(تشفير|فك تشفير) دفعة واحدة
my $ciphertext = $m->encrypt($plaintext, $key, $iv);
my $plaintext = $m->decrypt($ciphertext, $key, $iv);
#تشفير قطع أخرى
$m->start_encrypt($key, $iv);
my $ciphertext = $m->add('some data');
$ciphertext .= $m->add('more data');
#فك تشفير قطع أخرى
$m->start_decrypt($key, $iv);
my $plaintext = $m->add($some_ciphertext);
$plaintext .= $m->add($more_ciphertext);

الوصف

تطبق هذه الوحدة وضع التشفير CFB. ملاحظة: تعمل فقط مع خوارزميات التشفير من CryptX (Crypt::Cipher::NNNN).

الطرق

جديد

my $m = Crypt::Mode::CFB->new($name);
#أو
my $m = Crypt::Mode::CFB->new($name, $cipher_rounds);
# $name ............ أحد خيارات 'AES', 'Anubis', 'Blowfish', 'CAST5', 'Camellia', 'DES', 'DES_EDE',
#                    'KASUMI', 'Khazad', 'MULTI2', 'Noekeon', 'RC2', 'RC5', 'RC6',
#                    'SAFERP', 'SAFER_K128', 'SAFER_K64', 'SAFER_SK128', 'SAFER_SK64',
#                    'SEED', 'Skipjack', 'Twofish', 'XTEA', 'IDEA', 'Serpent'
#                    ببساطة أي <اسم> يوجد له Crypt::Cipher::<اسم>
# $cipher_rounds ... عدد اختياري للجولات للخوارزمية المعطاة

تشفير

my $ciphertext = $m->encrypt($plaintext, $key, $iv);

فك تشفير

my $plaintext = $m->decrypt($ciphertext, $key, $iv);

بدء_التشفير

$m->start_encrypt($key, $iv);

بدء_فك_التشفير

$m->start_decrypt($key, $iv);

إضافة

# في وضع التشفير
my $plaintext = $m->add($ciphertext);
# في وضع فك التشفير
my $ciphertext = $m->add($plaintext);
