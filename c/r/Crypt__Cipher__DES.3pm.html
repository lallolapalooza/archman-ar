Crypt__Cipher__DES(3pm)

الحزمة: perl-cryptx

التراخيص: PerlArtistic, GPL

Crypt::Cipher::DES(3) توثيق مساهمة المستخدم في Perl Crypt::Cipher::DES(3)

الاسم

Crypt::Cipher::DES - تشفير متماثل DES، حجم المفتاح: 64[56] بت

الموجز

### مثال 1
use Crypt::Mode::CBC;
my $key = '...'; # يجب أن يكون الطول حجم مفتاح صالح لهذا التشفير
my $iv = '...';  # 16 بايت
my $cbc = Crypt::Mode::CBC->new('DES');
my $ciphertext = $cbc->encrypt("بيانات سرية", $key, $iv);
### مثال 2 (أبطأ)
use Crypt::CBC;
use Crypt::Cipher::DES;
my $key = '...'; # يجب أن يكون الطول حجم مفتاح صالح لهذا التشفير
my $iv = '...';  # 16 بايت
my $cbc = Crypt::CBC->new( -cipher=>'Cipher::DES', -key=>$key, -iv=>$iv );
my $ciphertext = $cbc->encrypt("بيانات سرية");

الوصف

هذه الوحدة تنفذ تشفير DES. الواجهة المقدمة متوافقة مع وحدة Crypt::CBC.

تحذير: هذه الوحدة تنفذ فقط عملية "تشفير/فك تشفير" أساسية لـ "كتلة واحدة" - إذا كنت ترغب في تشفير/فك تشفير بيانات عامة، يجب عليك استخدام أحد أوضاع كتل التشفير - تحقق على سبيل المثال من Crypt::Mode::CBC، أو Crypt::Mode::CTR، أو Crypt::CBC (والتي ستكون أبطأ).

الطرق

جديد

$c = Crypt::Cipher::DES->new($key);
# أو
$c = Crypt::Cipher::DES->new($key, $rounds);

تشفير

$ciphertext = $c->encrypt($plaintext);

فك تشفير

$plaintext = $c->decrypt($ciphertext);

حجم المفتاح

$c->keysize;
# أو
Crypt::Cipher::DES->keysize;
# أو
Crypt::Cipher::DES::keysize;

حجم الكتلة

$c->blocksize;
# أو
Crypt::Cipher::DES->blocksize;
# أو
Crypt::Cipher::DES::blocksize;

أقصى حجم للمفتاح

$c->max_keysize;
# أو
Crypt::Cipher::DES->max_keysize;
# أو
Crypt::Cipher::DES::max_keysize;

أدنى حجم للمفتاح

$c->min_keysize;
# أو
Crypt::Cipher::DES->min_keysize;
# أو
Crypt::Cipher::DES::min_keysize;

الجولات الافتراضية

$c->default_rounds;
# أو
Crypt::Cipher::DES->default_rounds;
# أو
Crypt::Cipher::DES::default_rounds;
