CREATE_MATERIALIZED_VIEW(7)

الحزمة: postgresql

التراخيص: PostgreSQL

CREATE MATERIALIZED VIEW(7) وثائق PostgreSQL 18.1 CREATE MATERIALIZED VIEW(7)

الاسم

CREATE_MATERIALIZED_VIEW - تعريف عرض مادي جديد

الموجز

CREATE MATERIALIZED VIEW [ IF NOT EXISTS ] table_name
    [ (column_name [, ...] ) ]
    [ USING method ]
    [ WITH ( storage_parameter [= value] [, ... ] ) ]
    [ TABLESPACE tablespace_name ]
    AS query
    [ WITH [ NO ] DATA ]

الوصف

CREATE MATERIALIZED VIEW يحدد عرضًا ماديًا لاستعلام. يتم تنفيذ الاستعلام واستخدامه لملء العرض عند إصدار الأمر (ما لم يتم استخدام WITH NO DATA) وقد يتم تحديثه لاحقًا باستخدام REFRESH MATERIALIZED VIEW.

CREATE MATERIALIZED VIEW مشابه لـ CREATE TABLE AS، باستثناء أنه يتذكر أيضًا الاستعلام المستخدم لتهيئة العرض، بحيث يمكن تحديثه لاحقًا عند الطلب. للعرض المادي العديد من الخصائص المشابهة للجدول، ولكن لا يوجد دعم للعروض المادية المؤقتة.

CREATE MATERIALIZED VIEW يتطلب صلاحية CREATE على المخطط المستخدم للعرض المادي.

المعلمات

IF NOT EXISTS

لا يتم طرح خطأ إذا كان عرض مادي بنفس الاسم موجودًا بالفعل. يتم إصدار إشعار في هذه الحالة. لاحظ أنه لا يوجد ضمان بأن العرض المادي الموجود يشبه أي شيء يمكن إنشاؤه.

table_name

الاسم (اختياريًا مؤهل بالمخطط) للعرض المادي المراد إنشاؤه. يجب أن يكون الاسم مختلفًا عن اسم أي علاقة أخرى (جدول، تسلسل، فهرس، عرض، عرض مادي، أو جدول خارجي) في نفس المخطط.

column_name

اسم عمود في العرض المادي الجديد. إذا لم يتم توفير أسماء الأعمدة، يتم أخذها من أسماء أعمدة الإخراج للاستعلام.

USING method

يحدد هذا الشرط الاختياري طريقة الوصول إلى الجدول المراد استخدامها لتخزين محتويات العرض المادي الجديد؛ يجب أن تكون الطريقة طريقة وصول من نوع TABLE. انظر الفصل 62 لمزيد من المعلومات. إذا لم يتم تحديد هذا الخيار، يتم اختيار طريقة الوصول الافتراضية للجدول للعرض المادي الجديد. انظر default_table_access_method لمزيد من المعلومات.

WITH ( storage_parameter [= value] [, ... ] )

يحدد هذا الشرط معلمات تخزين اختيارية للعرض المادي الجديد؛ انظر معلمات التخزين في CREATE TABLE (وثائق CREATE_TABLE(7)) لمزيد من المعلومات. يتم دعم جميع المعلمات المدعومة لـ CREATE TABLE أيضًا لـ CREATE MATERIALIZED VIEW. انظر CREATE TABLE (وثائق CREATE_TABLE(7)) لمزيد من المعلومات.

TABLESPACE tablespace_name

tablespace_name هو اسم مساحة الجدول التي سيتم إنشاء العرض المادي الجديد فيها. إذا لم يتم تحديده، يتم الرجوع إلى default_tablespace.

query

أمر SELECT أو TABLE أو VALUES. سيتم تشغيل هذا الاستعلام ضمن عملية مقيدة الأمان؛ على وجه الخصوص، ستفشل استدعاءات الوظائف التي تنشئ جداول مؤقتة بنفسها. أيضًا، أثناء تشغيل الاستعلام، يتم تغيير search_path مؤقتًا إلى pg_catalog، pg_temp.

WITH [ NO ] DATA

يحدد هذا الشرط ما إذا كان يجب ملء العرض المادي عند الإنشاء أم لا. إذا لم يكن الأمر كذلك، فسيتم تمييز العرض المادي على أنه غير قابل للمسح ولن يمكن الاستعلام عنه حتى يتم استخدام REFRESH MATERIALIZED VIEW.

التوافق

CREATE MATERIALIZED VIEW هو امتداد PostgreSQL.

انظر أيضًا
