Crypt__Cipher__Camellia(3pm)

الحزمة: perl-cryptx

التراخيص: PerlArtistic, GPL

Crypt::Cipher::Camellia(3) وثائق Perl المساهمة من المستخدم Crypt::Cipher::Camellia(3)

الاسم

Crypt::Cipher::Camellia - خوارزمية التشفير المتماثل Camellia، حجم المفتاح: 128/192/256 بت

الملخص

### مثال 1
use Crypt::Mode::CBC;
my $key = '...'; # يجب أن يكون طول المفتاح صالحًا لهذه الخوارزمية
my $iv = '...';  # 16 بايت
my $cbc = Crypt::Mode::CBC->new('Camellia');
my $ciphertext = $cbc->encrypt("بيانات سرية", $key, $iv);
### مثال 2 (أبطأ)
use Crypt::CBC;
use Crypt::Cipher::Camellia;
my $key = '...'; # يجب أن يكون طول المفتاح صالحًا لهذه الخوارزمية
my $iv = '...';  # 16 بايت
my $cbc = Crypt::CBC->new( -cipher=>'Cipher::Camellia', -key=>$key, -iv=>$iv );
my $ciphertext = $cbc->encrypt("بيانات سرية");

الوصف

هذه الوحدة تطبق خوارزمية Camellia. الواجهة المقدمة متوافقة مع وحدة Crypt::CBC.

تحذير: هذه الوحدة تطبق فقط عملية "تشفير/فك تشفير" أساسية لكتلة واحدة - إذا كنت ترغب في تشفير/فك تشفير بيانات عامة، يجب عليك استخدام أحد أوضاع الكتل المشفرة - تحقق على سبيل المثال من Crypt::Mode::CBC، Crypt::Mode::CTR أو Crypt::CBC (والتي ستكون أبطأ).

الطرق

جديد

$c = Crypt::Cipher::Camellia->new($key);
# أو
$c = Crypt::Cipher::Camellia->new($key, $rounds);

تشفير

$ciphertext = $c->encrypt($plaintext);

فك التشفير

$plaintext = $c->decrypt($ciphertext);

حجم المفتاح

$c->keysize;
# أو
Crypt::Cipher::Camellia->keysize;
# أو
Crypt::Cipher::Camellia::keysize;

حجم الكتلة

$c->blocksize;
# أو
Crypt::Cipher::Camellia->blocksize;
# أو
Crypt::Cipher::Camellia::blocksize;

أقصى حجم للمفتاح

$c->max_keysize;
# أو
Crypt::Cipher::Camellia->max_keysize;
# أو
Crypt::Cipher::Camellia::max_keysize;

أدنى حجم للمفتاح

$c->min_keysize;
# أو
Crypt::Cipher::Camellia->min_keysize;
# أو
Crypt::Cipher::Camellia::min_keysize;

جولات افتراضية

$c->default_rounds;
# أو
Crypt::Cipher::Camellia->default_rounds;
# أو
Crypt::Cipher::Camellia::default_rounds;
