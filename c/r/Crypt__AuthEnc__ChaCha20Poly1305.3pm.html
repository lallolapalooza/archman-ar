Crypt__AuthEnc__ChaCha20Poly1305(3pm)

الحزمة: perl-cryptx

التراخيص: PerlArtistic, GPL

Crypt::AuthEnc::ChaCha20Poly1305(3) توثيق برمجي مساهم من بيرل Crypt::AuthEnc::ChaCha20Poly1305(3)

الاسم

Crypt::AuthEnc::ChaCha20Poly1305 - التشفير الموثق في وضع ChaCha20-Poly1305

الملخص

### واجهة الكائن الموجه
use Crypt::AuthEnc::ChaCha20Poly1305;
# تشفير وتوثيق
my $ae = Crypt::AuthEnc::ChaCha20Poly1305->new($key, $iv);
$ae->adata_add('additional_authenticated_data1');
$ae->adata_add('additional_authenticated_data2');
my $ct = $ae->encrypt_add('data1');
$ct .= $ae->encrypt_add('data2');
$ct .= $ae->encrypt_add('data3');
my $tag = $ae->encrypt_done();
# فك التشفير والتحقق
my $ae = Crypt::AuthEnc::ChaCha20Poly1305->new($key, $iv);
$ae->adata_add('additional_authenticated_data1');
$ae->adata_add('additional_authenticated_data2');
my $pt = $ae->decrypt_add('ciphertext1');
$pt .= $ae->decrypt_add('ciphertext2');
$pt .= $ae->decrypt_add('ciphertext3');
my $tag = $ae->decrypt_done();
die "decrypt failed" unless $tag eq $expected_tag;
# أو
my $result = $ae->decrypt_done($expected_tag); # 0 أو 1
### واجهة وظيفية
use Crypt::AuthEnc::ChaCha20Poly1305 qw(chacha20poly1305_encrypt_authenticate chacha20poly1305_decrypt_verify);
my ($ciphertext, $tag) = chacha20poly1305_encrypt_authenticate($key, $iv, $adata, $plaintext);
my $plaintext = chacha20poly1305_decrypt_verify($key, $iv, $adata, $ciphertext, $tag);

الوصف

يوفر التشفير والتوثيق بناءً على ChaCha20 + Poly1305 كما هو محدد في RFC 7539 - https://tools.ietf.org/html/rfc7539

التصدير

لا يتم تصدير شيء افتراضيًا.

يمكنك تصدير دوال مختارة:

use Crypt::AuthEnc::ChaCha20Poly1305 qw(chacha20poly1305_encrypt_authenticate chacha20poly1305_decrypt_verify);

الدوال

chacha20poly1305_encrypt_authenticate

my ($ciphertext, $tag) = chacha20poly1305_encrypt_authenticate($key, $iv, $adata, $plaintext);
# $key ..... المفتاح بطول مناسب (128 أو 256 بت / 16 أو 32 بايت)
# $iv ...... متجه التهيئة (64 أو 96 بت / 8 أو 12 بايت)
# $adata ... بيانات مصادق عليها إضافية (اختياري)

chacha20poly1305_decrypt_verify

my $plaintext = chacha20poly1305_decrypt_verify($key, $iv, $adata, $ciphertext, $tag);
# عند الخطأ يعيد undef

الأساليب

new

my $ae = Crypt::AuthEnc::ChaCha20Poly1305->new($key, $iv);
# $key ..... مفتاح التشفير بطول مناسب (128 أو 256 بت / 16 أو 32 بايت)
# $iv ...... متجه التهيئة (64 أو 96 بت / 8 أو 12 بايت)

adata_add

إضافة بيانات مصادق عليها إضافية. يمكن استدعاؤها قبل أول "encrypt_add" أو "decrypt_add"؛

$ae->adata_add($aad_data);                     # يمكن استدعاؤها عدة مرات

encrypt_add

$ciphertext = $ae->encrypt_add($data);         # يمكن استدعاؤها عدة مرات

encrypt_done

$tag = $ae->encrypt_done();                    # يعيد قيمة $tag

decrypt_add

$plaintext = $ae->decrypt_add($ciphertext);    # يمكن استدعاؤها عدة مرات

decrypt_done

my $tag = $ae->decrypt_done;           # يعيد قيمة $tag
# أو
my $result = $ae->decrypt_done($tag);  # يعيد 1 (نجاح) أو 0 (فشل)

set_iv

my $ae = Crypt::AuthEnc::ChaCha20Poly1305->new($key)->set_iv($iv);
# $iv ...... متجه التهيئة (64 أو 96 بت / 8 أو 12 بايت)

set_iv_rfc7905

انظر https://tools.ietf.org/html/rfc7905

my $ae = Crypt::AuthEnc::ChaCha20Poly1305->new($key)->set_iv_rfc7905($iv, $seqnum);
# $iv ...... متجه التهيئة (96 بت / 12 بايت)
# $seqnum .. عدد صحيح 64 بت (رقم التسلسل)

clone

my $ae_new = $ae->clone;
