Crypt__Cipher__Anubis(3pm)

الحزمة: perl-cryptx

التراخيص: PerlArtistic, GPL

Crypt::Cipher::Anubis(3) وثائق Perl مقدمة من المستخدم Crypt::Cipher::Anubis(3)

الاسم

Crypt::Cipher::Anubis - خوارزمية التشفير المتناظر Anubis، حجم المفتاح: 128-320 بت

الملخص

### المثال 1
use Crypt::Mode::CBC;
my $key = '...'; # يجب أن يكون طول المفتاح حجمًا صالحًا لهذه الخوارزمية
my $iv = '...';  # 16 بايت
my $cbc = Crypt::Mode::CBC->new('Anubis');
my $ciphertext = $cbc->encrypt("بيانات سرية", $key, $iv);
### المثال 2 (أبطأ)
use Crypt::CBC;
use Crypt::Cipher::Anubis;
my $key = '...'; # يجب أن يكون طول المفتاح حجمًا صالحًا لهذه الخوارزمية
my $iv = '...';  # 16 بايت
my $cbc = Crypt::CBC->new( -cipher=>'Cipher::Anubis', -key=>$key, -iv=>$iv );
my $ciphertext = $cbc->encrypt("بيانات سرية");

الوصف

تنفذ هذه الوحدة خوارزمية Anubis. الواجهة المقدمة متوافقة مع وحدة Crypt::CBC.

تحذير: هذه الوحدة تنفذ فقط عملية "تشفير/فك تشفير" أساسية لـ "كتلة واحدة" - إذا كنت تريد تشفير/فك تشفير بيانات عامة، فيجب عليك استخدام أحد أوضاع كتل التشفير - تحقق على سبيل المثال من Crypt::Mode::CBC، أو Crypt::Mode::CTR أو Crypt::CBC (والذي سيكون أبطأ).

الطرق

جديد

$c = Crypt::Cipher::Anubis->new($key);
# أو
$c = Crypt::Cipher::Anubis->new($key, $rounds);

تشفير

$ciphertext = $c->encrypt($plaintext);

فك تشفير

$plaintext = $c->decrypt($ciphertext);

# أو

# أو

أدنى حجم مفتاح

$c->min_keysize;
# أو

# أو
Crypt::Cipher::Anubis::min_keysize;
