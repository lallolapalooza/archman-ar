Crypt__Stream__RC4(3pm)

الحزمة: perl-cryptx

التراخيص: PerlArtistic, GPL

Crypt::Stream::RC4(3) وثائق Perl المساهمة من المستخدم Crypt::Stream::RC4(3)

الاسم

Crypt::Stream::RC4 - تشفير التدفق RC4

ملخص

use Crypt::Stream::RC4;
# تشفير
$key = "1234567890123456";
$stream = Crypt::Stream::RC4->new($key);
$ct = $stream->crypt("رسالة واضحة");
# فك التشفير
$key = "1234567890123456";
$stream = Crypt::Stream::RC4->new($key);
$pt = $stream->crypt($ct);

الوصف

يوفر واجهة لتشفير التدفق RC4.

الأساليب

جديد

$stream = Crypt::Stream::RC4->new($key);
# $key .. طول 5-256 بايت (40 - 2048 بت)

تشفير

$ciphertext = $stream->crypt($plaintext);
# أو
$plaintext = $stream->crypt($ciphertext);

مفتاح التدفق

$random_key = $stream->keystream($length);

استنساخ

$stream->clone();
