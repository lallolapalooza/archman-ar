subroutine crot (integer n, complex, dimension( * ) cx, integer incx, complex, dimension( * ) cy, integer incy, real c, complex s)

CROT تطبق دوران مستوي بجيب تمام حقيقي وجيب مركب على زوج من المتجهات المركبة.

الغرض:

!>
!> CROT تطبق دوران مستوي، حيث يكون الجيب تمام (C) حقيقي والجيب (S) مركب، والمتجهات CX و CY مركبة.
!>

المعاملات

N 
!>          N هو عدد صحيح (INTEGER)
!>          عدد العناصر في المتجهات CX و CY.
!>

CX

!>          CX هو مصفوفة مركبة (COMPLEX array)، ذات بعد (N)
!>          عند الإدخال، المتجه X.
!>          عند الإخراج، يتم استبدال CX بـ C*X + S*Y.
!>

INCX

!>          INCX هو عدد صحيح (INTEGER)
!>          الزيادة بين القيم المتتالية لـ CX. INCX <> 0.
!>

CY

!>          CY هي مصفوفة مركبة (COMPLEX array)، ذات بعد (N)
!>          عند الإدخال، المتجه Y.
!>          عند الإخراج، يتم استبدال CY بـ -CONJG(S)*X + C*Y.
!>

INCY

!>          INCY هو عدد صحيح (INTEGER)
!>          الزيادة بين القيم المتتالية لـ CY. INCX <> 0.
!>

C

!>          C هو عدد حقيقي (REAL)
!>

S

!>          S هو عدد مركب (COMPLEX)
!>          C و S يحددان دورانًا
!>             [  C          S  ]
!>             [ -conjg(S)   C  ]
!>          حيث C*C + S*CONJG(S) = 1.0.
!>
