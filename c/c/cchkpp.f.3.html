المعلمات

DOTYPE 
!>          DOTYPE هي مصفوفة LOGICAL، البعد (NTYPES)
!>          أنواع المصفوفات التي سيتم استخدامها للاختبار. المصفوفات من النوع j
!>          (من أجل 1 <= j <= NTYPES) تستخدم للاختبار إذا كان DOTYPE(j) =
!>          .TRUE.؛ إذا كان DOTYPE(j) = .FALSE.، فلن يتم استخدام النوع j.
!>

NN

!>          NN هو INTEGER
!>          عدد القيم N الموجودة في المتجه NVAL.
!>

NVAL

!>          NVAL هي مصفوفة INTEGER، البعد (NN)
!>          قيم بُعد المصفوفة N.
!>

NNS

!>          NNS هو INTEGER
!>          عدد القيم NRHS الموجودة في المتجه NSVAL.
!>

NSVAL

!>          NSVAL هي مصفوفة INTEGER، البعد (NNS)
!>          قيم عدد الأطراف اليمنى NRHS.
!>

THRESH

!>          THRESH هو REAL
!>          القيمة الحدية لنسب الاختبار. يتم تضمين النتيجة
!>          في ملف الإخراج إذا كانت RESULT >= THRESH. لطباعة
!>          كل نسبة اختبار، استخدم THRESH = 0.
!>

TSTERR

!>          TSTERR هو LOGICAL
!>          علم يشير إلى ما إذا كان سيتم اختبار مخارج الخطأ.
!>

NMAX

!>          NMAX هو INTEGER
!>          الحد الأقصى المسموح به للقيمة N، يستخدم في تحديد أبعاد
!>          مصفوفات العمل.
!>

A

!>          A هي مصفوفة COMPLEX، البعد
!>                      (NMAX*(NMAX+1)/2)
!>

AFAC

!>          AFAC هي مصفوفة COMPLEX، البعد
!>                      (NMAX*(NMAX+1)/2)
!>

AINV

!>          AINV هي مصفوفة COMPLEX، البعد
!>                      (NMAX*(NMAX+1)/2)
!>

B

!>          B هي مصفوفة COMPLEX، البعد (NMAX*NSMAX)
!>          حيث NSMAX هو أكبر مدخل في NSVAL.
!>

X

!>          X هي مصفوفة COMPLEX، البعد (NMAX*NSMAX)
!>

XACT

!>          XACT هي مصفوفة COMPLEX، البعد (NMAX*NSMAX)
!>

WORK

!>          WORK هي مصفوفة COMPLEX، البعد
!>                      (NMAX*max(3,NSMAX))
!>

RWORK

!>          RWORK هي مصفوفة REAL، البعد
!>                      (max(NMAX,2*NSMAX))
!>

NOUT

!>          NOUT هو INTEGER
!>          رقم الوحدة للإخراج.
!>
