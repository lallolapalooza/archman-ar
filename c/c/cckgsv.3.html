subroutine cckgsv (integer nm, integer, dimension( * ) mval, integer, dimension( * ) pval, integer, dimension( * ) nval, integer nmats, integer, dimension( 4 ) iseed, real thresh, integer nmax, complex, dimension( * ) a, complex, dimension( * ) af, complex, dimension( * ) b, complex, dimension( * ) bf, complex, dimension( * ) u, complex, dimension( * ) v, complex, dimension( * ) q, real, dimension( * ) alpha, real, dimension( * ) beta, complex, dimension( * ) r, integer, dimension( * ) iwork, complex, dimension( * ) work, real, dimension( * ) rwork, integer nin, integer nout, integer info)

CCKGSV

الغرض:

!>
!> CCKGSV تختبر CGGSVD:
!>        GSVD للمصفوفة A ذات الأبعاد M-by-N والمصفوفة B ذات الأبعاد P-by-N.
!>

المعلمات

NM 
!>          NM هو عدد صحيح (INTEGER)
!>          عدد قيم M الموجودة في المتجه MVAL.
!>

MVAL

!>          MVAL هو مصفوفة أعداد صحيحة (INTEGER array)، أبعادها (NM)
!>          قيم البعد M لصفوف المصفوفة.
!>

PVAL

!>          PVAL هو مصفوفة أعداد صحيحة (INTEGER array)، أبعادها (NP)
!>          قيم البعد P لصفوف المصفوفة.
!>

NVAL

!>          NVAL هو مصفوفة أعداد صحيحة (INTEGER array)، أبعادها (NN)
!>          قيم البعد N لأعمدة المصفوفة.
!>

NMATS

!>          NMATS هو عدد صحيح (INTEGER)
!>          عدد أنواع المصفوفات التي سيتم اختبارها لكل تركيبة
!>          من أبعاد المصفوفة. إذا كان NMATS >= NTYPES (الحد الأقصى
!>          لعدد أنواع المصفوفات)، فسيتم إنشاء جميع الأنواع المختلفة
!>          للاختبار. إذا كان NMATS < NTYPES، يتم قراءة سطر إدخال آخر
!>          للحصول على أرقام أنواع المصفوفات المراد استخدامها.
!>

ISEED

!>          ISEED هو مصفوفة أعداد صحيحة (INTEGER array)، أبعادها (4)
!>          عند الإدخال، بذرة مولد الأرقام العشوائية. يجب أن تكون عناصر
!>          المصفوفة بين 0 و 4095، وإلا فسيتم تقليلها حسب المعامل 4096،
!>          ويجب أن يكون ISEED(4) فرديًا.
!>          عند الخروج، البذرة التالية في تسلسل الأرقام العشوائية بعد
!>          إنشاء جميع مصفوفات الاختبار.
!>

THRESH

!>          THRESH هو قيمة حقيقية (REAL)
!>          القيمة الحدية لنسب الاختبار. يتم تضمين نتيجة
!>          في ملف الإخراج إذا كانت RESULT >= THRESH. لطباعة
!>          جميع نسب الاختبار، استخدم THRESH = 0.
!>

NMAX

!>          NMAX هو عدد صحيح (INTEGER)
!>          الحد الأقصى المسموح به لـ M أو N، ويستخدم في تحديد
!>          أبعاد مصفوفات العمل.
!>

A

!>          A هي مصفوفة مركبة (COMPLEX array)، أبعادها (NMAX*NMAX)
!>

AF

!>          AF هي مصفوفة مركبة (COMPLEX array)، أبعادها (NMAX*NMAX)
!>

B

!>          B هي مصفوفة مركبة (COMPLEX array)، أبعادها (NMAX*NMAX)
!>

BF

!>          BF هي مصفوفة مركبة (COMPLEX array)، أبعادها (NMAX*NMAX)
!>

U

!>          U هي مصفوفة مركبة (COMPLEX array)، أبعادها (NMAX*NMAX)
!>

V

!>          V هي مصفوفة مركبة (COMPLEX array)، أبعادها (NMAX*NMAX)
!>

Q

!>          Q هي مصفوفة مركبة (COMPLEX array)، أبعادها (NMAX*NMAX)
!>

ALPHA

!>          ALPHA هي مصفوفة حقيقية (REAL array)، أبعادها (NMAX)
!>

BETA

R

IWORK

!>

WORK

RWORK

NOUT
