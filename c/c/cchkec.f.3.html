subroutine cchkec (real thresh, logical tsterr, integer nin, integer nout)

CCHKEC

الغرض:

!>
!> CCHKEC تختبر روتينات تقدير شرط القيمة الذاتية
!>        CTRSYL, CTRSYL3, CTREXC, CTRSNA, CTRSEN
!>
!> في جميع الحالات، يمر الروتين عبر مجموعة ثابتة من الأمثلة الرقمية،
!> يخضعها لاختبارات مختلفة، ويقارن نتائج الاختبار بعتبة THRESH. بالإضافة إلى ذلك،
!> يتم اختبار CTRSNA و CTRSEN عن طريق قراءة أمثلة محسوبة مسبقًا من ملف (على وحدة الإدخال NIN).
!> يتم كتابة الإخراج إلى وحدة الإخراج NOUT.
!>

المعلمات

THRESH 
!>          THRESH هو REAL
!>          عتبة لاختبارات البواقي. اجتياز نسبة اختبار محسوبة
!>          العتبة إذا كانت أقل من THRESH.
!>

TSTERR

!>          TSTERR هو LOGICAL
!>          علامة تشير إلى ما إذا كان سيتم اختبار مخارج الخطأ.
!>

NIN

!>          NIN هو INTEGER
!>          رقم وحدة الإدخال المنطقية.
!>

NOUT

!>          NOUT هو INTEGER
!>          رقم وحدة الإخراج المنطقية.
!>
