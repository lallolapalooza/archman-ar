subroutine cchktp (logical, dimension( * ) dotype, integer nn, integer, dimension( * ) nval, integer nns, integer, dimension( * ) nsval, real thresh, logical tsterr, integer nmax, complex, dimension( * ) ap, complex, dimension( * ) ainvp, complex, dimension( * ) b, complex, dimension( * ) x, complex, dimension( * ) xact, complex, dimension( * ) work, real, dimension( * ) rwork, integer nout)

!>
!> CCHKTP تختبر CTPTRI، -TRS، -RFS، و -CON، و CLATPS
!>

المعلمات

DOTYPE 
!>          DOTYPE هو مصفوفة منطقية، البعد (NTYPES)
!>          أنواع المصفوفات التي سيتم استخدامها للاختبار. المصفوفات من النوع j
!>          (لـ 1 <= j <= NTYPES) تستخدم للاختبار إذا كان DOTYPE(j) =
!>          .TRUE.؛ إذا كان DOTYPE(j) = .FALSE.، فإن النوع j لا يستخدم.
!>

NN

!>          NN هو عدد صحيح
!>          عدد القيم لـ N الواردة في المتجه NVAL.
!>

NVAL

!>          NVAL هو مصفوفة أعداد صحيحة، البعد (NN)
!>          قيم البعد العمودي للمصفوفة N.
!>

NNS

!>          NNS هو عدد صحيح
!>          عدد القيم لـ NRHS الواردة في المتجه NSVAL.
!>

NSVAL

!>          NSVAL هو مصفوفة أعداد صحيحة، البعد (NNS)
!>          قيم عدد الأطراف اليمنى NRHS.
!>

THRESH

!>          THRESH هو قيمة حقيقية
!>          القيمة الحدية لنسب الاختبار. يتم تضمين النتيجة
!>          في ملف الإخراج إذا كانت RESULT >= THRESH. لطباعة
!>          كل نسبة اختبار، استخدم THRESH = 0.
!>

TSTERR

!>          TSTERR هو منطقي
!>          علامة تشير إلى ما إذا كان سيتم اختبار مخارج الأخطاء.
!>

NMAX

!>          NMAX هو عدد صحيح
!>          البعد الرائد لمصفوفات العمل. NMAX >= القيمة
!>          القصوى لـ N في NVAL.
!>

AP

!>          AP هي مصفوفة مركبة، البعد (NMAX*(NMAX+1)/2)
!>

AINVP

!>          AINVP هي مصفوفة مركبة، البعد (NMAX*(NMAX+1)/2)
!>

B

!>          B هي مصفوفة مركبة، البعد (NMAX*NSMAX)
!>          حيث NSMAX هو أكبر قيمة في NSVAL.
!>

X

!>          X هي مصفوفة مركبة، البعد (NMAX*NSMAX)
!>

XACT

!>          XACT هي مصفوفة مركبة، البعد (NMAX*NSMAX)
!>

WORK

!>          WORK هي مصفوفة مركبة، البعد
!>                      (NMAX*max(3,NSMAX))
!>

RWORK

!>          RWORK هي مصفوفة حقيقية، البعد
!>                      (max(NMAX,2*NSMAX))
!>

NOUT

!>          NOUT هو عدد صحيح
!>          رقم الوحدة للإخراج.
!>
