توثيق الوظيفة/الإجراء الفرعي

subroutine cqlt01 (integer m, integer n, complex, dimension( lda, * ) a, complex, dimension( lda, * ) af, complex, dimension( lda, * ) q, complex, dimension( lda, * ) l, integer lda, complex, dimension( * ) tau, complex, dimension( lwork ) work, integer lwork, real, dimension( * ) rwork, real, dimension( * ) result)

CQLT01

الغرض:

!>
!> CQLT01 تختبر CGEQLF، التي تحسب تحليل QL لمصفوفة m-by-n A، وتختبر جزئيًا CUNGQL
!> التي تشكل المصفوفة المتعامدة m-by-m Q.
!>
!> CQLT01 تقارن L مع Q'*A، وتتحقق من أن Q متعامدة.
!>

المعلمات

M 
!>          M هو عدد صحيح (INTEGER)
!>          عدد صفوف المصفوفة A.  M >= 0.
!>

N

!>          N هو عدد صحيح (INTEGER)
!>          عدد أعمدة المصفوفة A.  N >= 0.
!>

A

!>          A هي مصفوفة مركبة (COMPLEX)، البعد (LDA,N)
!>          المصفوفة A بحجم m-by-n.
!>

AF

!>          AF هي مصفوفة مركبة (COMPLEX)، البعد (LDA,N)
!>          تفاصيل تحليل QL للمصفوفة A، كما تم إرجاعها بواسطة CGEQLF.
!>          انظر CGEQLF لمزيد من التفاصيل.
!>

Q

!>          Q هي مصفوفة مركبة (COMPLEX)، البعد (LDA,M)
!>          المصفوفة المتعامدة Q بحجم m-by-m.
!>

L

!>          L هي مصفوفة مركبة (COMPLEX)، البعد (LDA,max(M,N))
!>

LDA

!>          LDA هو عدد صحيح (INTEGER)
!>          البعد الرئيسي للمصفوفات A، AF، Q و R.
!>          LDA >= max(M,N).
!>

TAU

!>          TAU هي مصفوفة مركبة (COMPLEX)، البعد (min(M,N))
!>          العوامل العددية للمنعكسات الأولية، كما تم إرجاعها
!>          بواسطة CGEQLF.
!>

WORK

!>          WORK هي مصفوفة مركبة (COMPLEX)، البعد (LWORK)
!>

LWORK

!>          LWORK هو عدد صحيح (INTEGER)
!>          بعد مصفوفة WORK.
!>

RWORK

!>          RWORK هي مصفوفة حقيقية (REAL)، البعد (M)
!>

RESULT

!>          RESULT هي مصفوفة حقيقية (REAL)، البعد (2)
!>          نسب الاختبار:
!>          RESULT(1) = norm( L - Q'*A ) / ( M * norm(A) * EPS )
!>          RESULT(2) = norm( I - Q'*Q ) / ( M * EPS )
!>
