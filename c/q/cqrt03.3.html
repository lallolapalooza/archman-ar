subroutine cqrt03 (integer m, integer n, integer k, complex, dimension( lda, * ) af, complex, dimension( lda, * ) c, complex, dimension( lda, * ) cc, complex, dimension( lda, * ) q, integer lda, complex, dimension( * ) tau, complex, dimension( lwork ) work, integer lwork, real, dimension( * ) rwork, real, dimension( * ) result)

CQRT03

الغرض:

!>
!> CQRT03 تختبر CUNMQR، التي تحسب Q*C، Q'*C، C*Q أو C*Q'.
!>
!> تقارن CQRT03 نتائج استدعاء لـ CUNMQR مع نتائج
!> تشكيل Q بشكل صريح عن طريق استدعاء CUNGQR ثم إجراء
!> ضرب المصفوفات عن طريق استدعاء CGEMM.
!>

المعلمات

M 
!>          M هو عدد صحيح
!>          رتبة المصفوفة المتعامدة Q.  M >= 0.
!>

N

!>          N هو عدد صحيح
!>          عدد الصفوف أو الأعمدة في المصفوفة C؛ C هي m-by-n إذا
!>          تم تطبيق Q من اليسار، أو n-by-m إذا تم تطبيق Q من
!>          اليمين.  N >= 0.
!>

K

!>          K هو عدد صحيح
!>          عدد الموجهات الأولية التي يشكل حاصل ضربها المصفوفة المتعامدة
!>          Q.  M >= K >= 0.
!>

AF

!>          AF هي مصفوفة COMPLEX، بأبعاد (LDA,N)
!>          تفاصيل تحليل QR لمصفوفة m-by-n، كما
!>          تم إرجاعها بواسطة CGEQRF. راجع CGEQRF لمزيد من التفاصيل.
!>

C

!>          C هي مصفوفة COMPLEX، بأبعاد (LDA,N)
!>

CC

!>          CC هي مصفوفة COMPLEX، بأبعاد (LDA,N)
!>

Q

!>          Q هي مصفوفة COMPLEX، بأبعاد (LDA,M)
!>

LDA

!>          LDA هو عدد صحيح
!>          البعد الرائد للمصفوفات AF، C، CC، و Q.
!>

TAU

!>          TAU هي مصفوفة COMPLEX، بأبعاد (min(M,N))
!>          العوامل القياسية للموجهات الأولية المقابلة
!>          لتحليل QR في AF.
!>

WORK

!>          WORK هي مصفوفة COMPLEX، بأبعاد (LWORK)
!>

LWORK

!>          LWORK هو عدد صحيح
!>          طول WORK. يجب أن يكون LWORK على الأقل M، ويجب أن يكون
!>          M*NB، حيث NB هو حجم الكتلة لهذه البيئة.
!>

RWORK

!>          RWORK هي مصفوفة REAL، بأبعاد (M)
!>

RESULT

!>          RESULT هي مصفوفة REAL، بأبعاد (4)
!>          نسب الاختبار تقارن تقنيتين لضرب مصفوفة عشوائية C
!>          بمصفوفة متعامدة m-by-m Q.
!>          RESULT(1) = norm( Q*C - Q*C )  / ( M * norm(C) * EPS )
!>          RESULT(2) = norm( C*Q - C*Q )  / ( M * norm(C) * EPS )
!>          RESULT(3) = norm( Q'*C - Q'*C )/ ( M * norm(C) * EPS )
!>          RESULT(4) = norm( C*Q' - C*Q' )/ ( M * norm(C) * EPS )
!>
