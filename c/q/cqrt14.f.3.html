توثيق الوظيفة/الروتين الفرعي

real function cqrt14 (character trans, integer m, integer n, integer nrhs, complex, dimension( lda, * ) a, integer lda, complex, dimension( ldx, * ) x, integer ldx, complex, dimension( lwork ) work, integer lwork)

CQRT14

الغرض:

!>
!> CQRT14 يتحقق مما إذا كان X موجودًا في مساحة الصفوف لـ A أو A'. يتم ذلك
!> عن طريق قياس كل من X و A بحيث تكون مقاييسهما في النطاق
!> [sqrt(eps), 1/sqrt(eps)]، ثم حساب تحليل QR لـ [A,X]
!> (إذا كان TRANS = 'C') أو تحليل LQ لـ [A',X]' (إذا كان TRANS = 'N')،
!> وإرجاع مقياس المثلث اللاحق، مقاسًا بـ
!> MAX(M,N,NRHS)*eps.
!>

المعلمات

TRANS 
!>          TRANS is CHARACTER*1
!>          = 'N':  بدون تبديل، تحقق من X في مساحة الصفوف لـ A
!>          = 'C':  تبديل مترافق، تحقق من X في مساحة الصفوف لـ A'.
!>

M

!>          M is INTEGER
!>          عدد صفوف المصفوفة A.
!>

N

!>          N is INTEGER
!>          عدد أعمدة المصفوفة A.
!>

NRHS

!>          NRHS is INTEGER
!>          عدد الأطراف اليمنى، أي عدد أعمدة
!>          X.
!>

A

!>          A is COMPLEX array, dimension (LDA,N)
!>          المصفوفة A ذات الأبعاد M × N.
!>

LDA

!>          LDA is INTEGER
!>          البعد الرائد للمصفوفة A.
!>

X

!>          X is COMPLEX array, dimension (LDX,NRHS)
!>          إذا كان TRANS = 'N'، المصفوفة X ذات الأبعاد N × NRHS.
!>          إذا كان TRANS = 'C'، المصفوفة X ذات الأبعاد M × NRHS.
!>

LDX

!>          LDX is INTEGER
!>          البعد الرائد للمصفوفة X.
!>

WORK

!>          WORK is COMPLEX array dimension (LWORK)
!>

LWORK

!>          LWORK is INTEGER
!>          طول مساحة العمل المطلوبة
!>          إذا كان TRANS = 'N'، LWORK >= (M+NRHS)*(N+2)؛
!>          إذا كان TRANS = 'C'، LWORK >= (N+NRHS)*(M+2).
!>
