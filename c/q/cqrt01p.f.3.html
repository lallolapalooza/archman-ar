توثيق الوظيفة/الإجراء الفرعي

subroutine cqrt01p (integer m, integer n, complex, dimension( lda, * ) a, complex, dimension( lda, * ) af, complex, dimension( lda, * ) q, complex, dimension( lda, * ) r, integer lda, complex, dimension( * ) tau, complex, dimension( lwork ) work, integer lwork, real, dimension( * ) rwork, real, dimension( * ) result)

CQRT01P

الغرض:

!>
!> CQRT01P يختبر CGEQRFP، الذي يحسب تحلل QR للمصفوفة A ذات الأبعاد m × n، ويختبر جزئيًا CUNGQR الذي يشكل المصفوفة المتعامدة Q ذات الأبعاد m × m.
!>
!> CQRT01P يقارن R مع Q'*A، ويتحقق من أن Q متعامدة.
!>

المعلمات

M 
!>          M هو عدد صحيح (INTEGER)
!>          عدد صفوف المصفوفة A.  M >= 0.
!>

N

!>          N هو عدد صحيح (INTEGER)
!>          عدد أعمدة المصفوفة A.  N >= 0.
!>

A

!>          A هو مصفوفة مركبة (COMPLEX)، البعد (LDA,N)
!>          المصفوفة A ذات الأبعاد m × n.
!>

AF

!>          AF هي مصفوفة مركبة (COMPLEX)، البعد (LDA,N)
!>          تفاصيل تحلل QR للمصفوفة A، كما تم إرجاعها بواسطة CGEQRFP.
!>          انظر CGEQRFP لمزيد من التفاصيل.
!>

Q

!>          Q هي مصفوفة مركبة (COMPLEX)، البعد (LDA,M)
!>          المصفوفة المتعامدة Q ذات الأبعاد m × m.
!>

R

!>          R هي مصفوفة مركبة (COMPLEX)، البعد (LDA,max(M,N))
!>

LDA

!>          LDA هو عدد صحيح (INTEGER)
!>          البعد الرئيسي للمصفوفات A، AF، Q و R.
!>          LDA >= max(M,N).
!>

TAU

!>          TAU هي مصفوفة مركبة (COMPLEX)، البعد (min(M,N))
!>          العوامل العددية للمنعكسات الأولية، كما تم إرجاعها
!>          بواسطة CGEQRFP.
!>

WORK

!>          WORK هي مصفوفة مركبة (COMPLEX)، البعد (LWORK)
!>

LWORK

!>          LWORK هو عدد صحيح (INTEGER)
!>          بعد المصفوفة WORK.
!>

RWORK

!>          RWORK هي مصفوفة حقيقية (REAL)، البعد (M)
!>

RESULT

!>          RESULT هي مصفوفة حقيقية (REAL)، البعد (2)
!>          نسب الاختبار:
!>          RESULT(1) = norm( R - Q'*A ) / ( M * norm(A) * EPS )
!>          RESULT(2) = norm( I - Q'*Q ) / ( M * EPS )
!>
