subroutine clatsqr (integer m, integer n, integer mb, integer nb, complex, dimension( lda, * ) a, integer lda, complex, dimension( ldt, * ) t, integer ldt, complex, dimension( * ) work, integer lwork, integer info)

CLATSQR

الغرض:

!>
!> CLATSQR يحسب عامل QR الضيق المجمع (blocked Tall-Skinny QR factorization)
!> للمصفوفة المعقدة A من النوع M-by-N حيث M >= N:
!>
!>    A = Q * ( R ),
!>            ( 0 )
!>
!> حيث:
!>
!>    Q هي مصفوفة متعامدة من النوع M-by-M، مخزنة عند الخروج في شكل ضمني
!>    في العناصر الموجودة أسفل القطر الرئيسي للمصفوفة A وفي
!>    عناصر المصفوفة T؛
!>
!>    R هي مصفوفة مثلثية عليا من النوع N-by-N، مخزنة عند الخروج في
!>    العناصر الموجودة على القطر الرئيسي وما فوقه في المصفوفة A.
!>
!>    0 هي مصفوفة صفرية من النوع (M-N)-by-N، وغير مخزنة.
!>
!>

المعاملات

M 
!>          M هو عدد صحيح (INTEGER)
!>          عدد صفوف المصفوفة A.  M >= 0.
!>

N

!>          N هو عدد صحيح (INTEGER)
!>          عدد أعمدة المصفوفة A. M >= N >= 0.
!>

MB

!>          MB هو عدد صحيح (INTEGER)
!>          حجم الكتلة الصفية المستخدمة في QR المجمع.
!>          MB > N.
!>

NB

!>          NB هو عدد صحيح (INTEGER)
!>          حجم الكتلة العمودية المستخدمة في QR المجمع.
!>          N >= NB >= 1.
!>

A

!>          A هي مصفوفة معقدة (COMPLEX array)، ذات بعد (LDA,N)
!>          عند الدخول، المصفوفة A من النوع M-by-N.
!>          عند الخروج، العناصر الموجودة على القطر الرئيسي وما فوقه
!>          في المصفوفة تحتوي على المصفوفة المثلثية العليا R من النوع N-by-N؛
!>          العناصر الموجودة أسفل القطر تمثل Q بواسطة أعمدة
!>          V المجمعة (انظر التفاصيل الإضافية).
!>

LDA

!>          LDA هو عدد صحيح (INTEGER)
!>          البعد الرائد للمصفوفة A.  LDA >= max(1,M).
!>

T

!>          T هي مصفوفة معقدة (COMPLEX array)،
!>          ذات بعد (LDT, N * Number_of_row_blocks)
!>          حيث Number_of_row_blocks = CEIL((M-N)/(MB-N))
!>          تمثل عواكس الكتلة المثلثية العليا المجمعة مخزنة في شكل مضغوط
!>          كسلسلة من الكتل المثلثية العليا.
!>          انظر التفاصيل الإضافية أدناه.
!>

LDT

!>          LDT هو عدد صحيح (INTEGER)
!>          البعد الرائد للمصفوفة T.  LDT >= NB.
!>

WORK

!>          (مساحة عمل) مصفوفة معقدة (COMPLEX array)، ذات بعد (MAX(1,LWORK))
!>          عند الخروج، إذا كانت INFO = 0، فإن WORK(1) تعيد الحد الأدنى لـ LWORK.
!>

LWORK

!>          LWORK هو عدد صحيح (INTEGER)
!>          بعد المصفوفة WORK.
!>          LWORK >= 1، إذا كان MIN(M,N) = 0، و LWORK >= NB*N، بخلاف ذلك.
!>
!>          إذا كانت LWORK = -1، فسيتم افتراض استعلام عن مساحة العمل؛ يقوم الإجراء الفرعي
!>          فقط بحساب الحد الأدنى لحجم مصفوفة WORK، ويعيد
!>          هذه القيمة كالإدخال الأول لمصفوفة WORK، ولا يتم إصدار أي رسالة خطأ
!>          متعلقة بـ LWORK بواسطة XERBLA.
!>

INFO

!>          INFO هو عدد صحيح (INTEGER)
!>          = 0:  انتهاء ناجح
!>          < 0:  إذا كانت INFO = -i، فإن المعامل i كان له قيمة غير قانونية
!>

تفاصيل إضافية:

!> عامل QR الضيق (TSQR) يقوم بعمل QR عن طريق سلسلة من التحويلات المتعامدة،
!> يمثل Q كحاصل ضرب لمصفوفات متعامدة أخرى
!>   Q = Q(1) * Q(2) * . . . * Q(k)
!> حيث كل Q(i) يصفر الإدخالات تحت القطرية لكتلة من MB صفوف من A:
!>   Q(1) يصفر الإدخالات تحت القطرية للصفوف 1:MB من A
!>   Q(2) يصفر الصفوف MB-N السفلية من الصفوف [1:N,MB+1:2*MB-N] من A
!>   Q(3) يصفر الصفوف MB-N السفلية من الصفوف [1:N,2*MB-N+1:3*MB-2*N] من A
!>   . . .
!>
!> يتم حساب Q(1) بواسطة GEQRT، والذي يمثل Q(1) بواسطة متجهات هاوسهولدر
!> مخزنة أسفل القطر الرئيسي لصفوف 1:MB من A، وبواسطة عواكس كتل
!> مثلثية عليا، مخزنة في المصفوفة T(1:LDT,1:N).
!> لمزيد من المعلومات انظر التفاصيل الإضافية في GEQRT.
!>
!> يتم حساب Q(i) لـ i>1 بواسطة TPQRT، والذي يمثل Q(i) بواسطة متجهات هاوسهولدر
!> مخزنة في الصفوف [(i-1)*(MB-N)+N+1:i*(MB-N)+N] من A، وبواسطة عواكس كتل
!> مثلثية عليا، مخزنة في المصفوفة T(1:LDT,(i-1)*N+1:i*N).
!> قد تستخدم Q(k) الأخيرة عددًا أقل من الصفوف.
!> لمزيد من المعلومات انظر التفاصيل الإضافية في TPQRT.
!>
!> لمزيد من التفاصيل حول الخوارزمية الإجمالية، انظر وصف
!> TSQR المتسلسل في القسم 2.2 من [1].
!>
!> [1] “Communication-Optimal Parallel and Sequential QR and LU Factorizations,”
!>     J. Demmel, L. Grigori, M. Hoemmen, J. Langou,
!>     SIAM J. Sci. Comput, vol. 34, no. 1, 2012
!>
