subroutine clascl (type, kl, ku, cfrom, cto, m, n, a, lda, info)
CLASCL تضرب مصفوفة مستطيلة عامة في عدد قياسي حقيقي معرف كـ cto/cfrom.

توثيق الوظيفة/الروتين الفرعي

subroutine clascl (character type, integer kl, integer ku, real cfrom, real cto, integer m, integer n, complex, dimension( lda, * ) a, integer lda, integer info)

CLASCL تضرب مصفوفة مركبة M × N A في العدد القياسي الحقيقي CTO/CFROM. يتم ذلك دون تجاوز السعة أو نقص السعة طالما أن النتيجة النهائية CTO*A(I,J)/CFROM لا تتجاوز السعة أو تنقص عنها.

الغرض:

!>
!> CLASCL تضرب المصفوفة المركبة M × N A في العدد القياسي الحقيقي
!> CTO/CFROM. يتم ذلك دون تجاوز السعة / نقص السعة طالما أن النتيجة النهائية
!> CTO*A(I,J)/CFROM لا تتجاوز السعة / تنقص عنها. TYPE يحدد أن
!> A قد تكون مصفوفة كاملة، مثلثة علوية، مثلثة سفلية، هسنبرغ علوية،
!> أو مصفوفة حزمية.
!>

المعاملات

TYPE 
!>          TYPE هو CHARACTER*1
!>          TYPE يحدد نوع تخزين المصفوفة المدخلة.
!>          = 'G':  A مصفوفة كاملة.
!>          = 'L':  A مصفوفة مثلثة سفلية.
!>          = 'U':  A مصفوفة مثلثة علوية.
!>          = 'H':  A مصفوفة هسنبرغ علوية.
!>          = 'B':  A مصفوفة حزمية متناظرة ذات عرض نطاق سفلي KL
!>                  وعرض نطاق علوي KU ويتم تخزين النصف السفلي فقط.
!>          = 'Q':  A مصفوفة حزمية متناظرة ذات عرض نطاق سفلي KL
!>                  وعرض نطاق علوي KU ويتم تخزين النصف العلوي فقط.
!>          = 'Z':  A مصفوفة حزمية ذات عرض نطاق سفلي KL وعرض نطاق علوي
!>                  KU. راجع CGBTRF لتفاصيل التخزين.
!>

KL

!>          KL هو INTEGER
!>          عرض النطاق السفلي لـ A. يتم الإشارة إليه فقط إذا كان TYPE = 'B'،
!>          'Q'، أو 'Z'.
!>

KU

!>          KU هو INTEGER
!>          عرض النطاق العلوي لـ A. يتم الإشارة إليه فقط إذا كان TYPE = 'B'،
!>          'Q'، أو 'Z'.
!>

CFROM

!>          CFROM هو REAL
!>

CTO

!>          CTO هو REAL
!>
!>          يتم ضرب المصفوفة A في CTO/CFROM. يتم حساب A(I,J)
!>          دون تجاوز السعة/نقص السعة إذا كانت النتيجة النهائية CTO*A(I,J)/CFROM
!>          يمكن تمثيلها دون تجاوز السعة/نقص السعة. يجب أن يكون CFROM
!>          غير صفري.
!>

M

!>          M هو INTEGER
!>          عدد صفوف المصفوفة A.  M >= 0.
!>

N

!>          N هو INTEGER
!>          عدد أعمدة المصفوفة A.  N >= 0.
!>

A

!>          A هو مصفوفة COMPLEX، بأبعاد (LDA,N)
!>          المصفوفة التي سيتم ضربها في CTO/CFROM.  راجع TYPE لنوع
!>          التخزين.
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرائد للمصفوفة A.
!>          إذا كان TYPE = 'G'، 'L'، 'U'، 'H'، فإن LDA >= max(1,M)؛
!>             TYPE = 'B'، فإن LDA >= KL+1؛
!>             TYPE = 'Q'، فإن LDA >= KU+1؛
!>             TYPE = 'Z'، فإن LDA >= 2*KL+KU+1.
!>

INFO

!>          INFO هو INTEGER
!>          0  - خروج ناجح
!>          <0 - إذا كان INFO = -i، فإن المعامل i له قيمة غير قانونية.
!>
