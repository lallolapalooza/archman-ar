توثيق الوظيفة/الروتين الفرعي

subroutine clatms (integer m, integer n, character dist, integer, dimension( 4 ) iseed, character sym, real, dimension( * ) d, integer mode, real cond, real dmax, integer kl, integer ku, character pack, complex, dimension( lda, * ) a, integer lda, complex, dimension( * ) work, integer info)

CLATMS

الغرض:

!>
!>    CLATMS تولد مصفوفات عشوائية بقيم مفردة محددة
!>    (أو هيرميتية بقيم ذاتية محددة)
!>    لاختبار برامج LAPACK.
!>
!>    تعمل CLATMS عن طريق تطبيق التسلسل التالي من
!!>    العمليات:
!>
!>      عيّن القطر إلى D، حيث يمكن أن يكون D مدخلًا أو
!>         محسوبًا وفقًا لـ MODE و COND و DMAX و SYM
!>         كما هو موضح أدناه.
!>
!>      توليد مصفوفة بالهيكل النطاقي المناسب، بواسطة واحدة
!>         من طريقتين:
!>
!>      الطريقة أ:
!>          توليد مصفوفة كثيفة M × N عن طريق ضرب D على اليسار
!>              وعلى اليمين بمصفوفات وحدوية عشوائية، ثم:
!>
!>          تقليل عرض النطاق وفقًا لـ KL و KU، باستخدام
!>              تحويلات هاوسهولدر.
!>
!>      الطريقة ب:
!>          تحويل مصفوفة عرض النطاق الصفري (أي قطرية) إلى
!>              مصفوفة عرض نطاق 1 باستخدام دوران جيفنز،
!>              عناصر خارج النطاق للخلف، كثيرًا مثل QR؛ ثم تحويل
!>              عرض النطاق 1 إلى مصفوفة عرض النطاق 2، إلخ. لاحظ
!>              أنه بالنسبة لعروض النطاقات الصغيرة بشكل معقول (مقارنة بـ M و
!>              N) هذا يتطلب تخزينًا أقل، حيث لا يتم تخزين مصفوفة كثيفة.
!>              أيضًا، بالنسبة للمصفوفات الهيرميتية أو المتماثلة،
!>              يتم توليد مثلث واحد فقط.
!>
!>      يتم اختيار الطريقة أ إذا كان عرض النطاق يمثل جزءًا كبيرًا من
!>          رتبة المصفوفة، وكان LDA على الأقل M (بحيث يمكن تخزين
!>          مصفوفة كثيفة.) يتم اختيار الطريقة ب إذا كان عرض النطاق
!>          صغيرًا (< 1/2 N للمصفوفات الهيرميتية أو المتماثلة، < .3 N+M لغير
!>          المتماثلة)، أو LDA أقل من M وليس أقل من
!>          عرض النطاق.
!>
!>      قم بتعبئة المصفوفة إذا رغبت في ذلك. الخيارات المحددة بواسطة PACK هي:
!>         لا يوجد تعبئة
!>         صفر النصف العلوي (إذا كانت هيرميتية)
!>         صفر النصف السفلي (إذا كانت هيرميتية)
!>         تخزين النصف العلوي عموديًا (إذا كانت هيرميتية أو
!>               مثلثة علوية)
!>         تخزين النصف السفلي عموديًا (إذا كانت هيرميتية أو
!>               مثلثة سفلية)
!>         تخزين المثلث السفلي بتنسيق نطاقي (إذا كانت هيرميتية أو
!>               مثلثة سفلية)
!>         تخزين المثلث العلوي بتنسيق نطاقي (إذا كانت هيرميتية أو
!>               مثلثة علوية)
!>         تخزين المصفوفة بأكملها بتنسيق نطاقي
!>      إذا تم اختيار الطريقة ب، وتم تحديد تنسيق النطاق، فسيتم
!>         توليد المصفوفة بتنسيق النطاق، لذلك لن تكون هناك حاجة لإعادة التعبئة.
!>

المعلمات

M 
!>          M هو عدد صحيح (INTEGER)
!>           عدد صفوف A. لم يتم تعديله.
!>

N

!>          N هو عدد صحيح (INTEGER)
!>           عدد أعمدة A. يجب أن يساوي N، M إذا كانت المصفوفة
!>           متماثلة أو هيرميتية (أي، إذا كان SYM ليس 'N')
!>           لم يتم تعديله.
!>

DIST

!>          DIST هو حرف (CHARACTER*1)
!>           عند الإدخال، يحدد DIST نوع التوزيع الذي سيتم استخدامه
!>           لتوليد القيم المفردة/الذاتية العشوائية.
!>           'U' => منتظم (0، 1) ('U' للمنتظم)
!>           'S' => منتظم (-1، 1) ('S' للمتماثل)
!>           'N' => طبيعي (0، 1) ('N' للطبيعي)
!>           لم يتم تعديله.
!>

ISEED

!>          ISEED هو مصفوفة أعداد صحيحة (INTEGER array)، حجمها (4)
!>           عند الإدخال، يحدد ISEED بذرة مولد الأرقام
!>           العشوائية. يجب أن تتراوح بين 0 و 4095 شاملة،
!>           ويجب أن يكون ISEED(4) فرديًا. يستخدم مولد الأرقام
!>           العشوائية تسلسلًا خطيًا تكراريًا محدودًا بأعداد
!>           صغيرة، وبالتالي يجب أن ينتج أرقامًا عشوائية مستقلة عن الآلة.
!>           تتغير قيم ISEED عند الخروج، ويمكن استخدامها في الاستدعاء التالي لـ CLATMS
!>           لمواصلة نفس تسلسل الأرقام العشوائية.
!>           تم تعديله عند الخروج.
!>

SYM

!>          SYM هو حرف (CHARACTER*1)
!>           إذا كان SYM='H'، فإن المصفوفة المولدة هيرميتية، مع
!>             قيم ذاتية محددة بواسطة D و COND و MODE و DMAX؛ يمكن أن
!>             تكون موجبة أو سالبة أو صفرية.
!>           إذا كان SYM='P'، فإن المصفوفة المولدة هيرميتية، مع
!>             قيم مفردة (قيم ذاتية) محددة بواسطة D و COND و
!>             MODE و DMAX؛ لن تكون سالبة.
!>           إذا كان SYM='N'، فإن المصفوفة المولدة غير متماثلة، مع
!>             قيم مفردة محددة بواسطة D و COND و MODE و DMAX؛
!>             لن تكون سالبة.
!>           إذا كان SYM='S'، فإن المصفوفة المولدة متماثلة (مركبة)، مع
!>             قيم مفردة محددة بواسطة D و COND و MODE و DMAX؛
!>             لن تكون سالبة.
!>           لم يتم تعديله.
!>

D

!>          D هو مصفوفة حقيقية (REAL array)، حجمها (MIN( M, N ))
!>           تستخدم هذه المصفوفة لتحديد القيم المفردة أو
!>           القيم الذاتية لـ A (انظر SYM، أعلاه.) إذا كان MODE=0، فسيتم
!>           افتراض أن D تحتوي على القيم المفردة/الذاتية، وإلا
!>           سيتم حسابها وفقًا لـ MODE و COND و DMAX،
!>           وستوضع في D.
!>           تم تعديله إذا كان MODE غير صفري.
!>

MODE

!>          MODE هو عدد صحيح (INTEGER)
!>           عند الإدخال، يصف هذا كيفية تحديد القيم المفردة/الذاتية:
!>           MODE = 0 يعني استخدام D كمدخل
!>           MODE = 1 يعيّن D(1)=1 و D(2:N)=1.0/COND
!>           MODE = 2 يعيّن D(1:N-1)=1 و D(N)=1.0/COND
!>           MODE = 3 يعيّن D(I)=COND**(-(I-1)/(N-1))
!>           MODE = 4 يعيّن D(i)=1 - (i-1)/(N-1)*(1 - 1/COND)
!>           MODE = 5 يعيّن D إلى أرقام عشوائية في النطاق
!>                    ( 1/COND ، 1 ) بحيث تكون لوغاريتماتها
!>                    موزعة بشكل منتظم.
!>           MODE = 6 يعيّن D إلى أرقام عشوائية من نفس التوزيع
!>                    مثل بقية المصفوفة.
!>           MODE < 0 له نفس المعنى مثل ABS(MODE)، باستثناء أن
!>              ترتيب عناصر D معكوس.
!>           لذلك، إذا كان MODE موجبًا، فإن D تحتوي على عناصر تتراوح من
!>              1 إلى 1/COND، وإذا كان سالبًا، من 1/COND إلى 1،
!>           إذا كان SYM='H'، وكان MODE ليس 0 أو 6 أو -6، فسيتم
!>              ضرب عناصر D أيضًا بإشارة عشوائية (أي، +1 أو -1.)
!>           لم يتم تعديله.
!>

COND

!>          COND هو عدد حقيقي (REAL)
!>           عند الإدخال، يتم استخدامه كما هو موضح تحت MODE أعلاه.
!>           إذا تم استخدامه، فيجب أن يكون >= 1. لم يتم تعديله.
!>

DMAX

!>          DMAX هو عدد حقيقي (REAL)
!>           إذا كان MODE ليس -6 أو 0 أو 6، فسيتم تحجيم محتويات D، كما
!>           تم حسابها وفقًا لـ MODE و COND، بواسطة
!>           DMAX / max(abs(D(i))); وبالتالي، فإن أكبر قيمة مطلقة مفردة أو
!>           ذاتية (وهي المعيار) ستكون abs(DMAX).
!>           لاحظ أن DMAX لا يجب أن يكون موجبًا: إذا كان DMAX سالبًا
!>           (أو صفرًا)، فسيتم تحجيم D بعدد سالب (أو صفر).
!>           لم يتم تعديله.
!>

KL

!>          KL هو عدد صحيح (INTEGER)
!>           يحدد هذا عرض النطاق السفلي للمصفوفة. على سبيل المثال،
!>           KL=0 يعني مثلث علوي، KL=1 يعني هسنبرغ علوي، و KL أكبر من أو يساوي M-1
!>           يعني أن المصفوفة لديها عرض نطاق سفلي كامل. يجب أن يساوي KL، KU إذا كانت المصفوفة
!>           متماثلة أو هيرميتية.
!>           لم يتم تعديله.
!>
