subroutine cla_gbamv (trans, m, n, kl, ku, alpha, ab, ldab, x, incx, beta, y, incy)
CLA_GBAMV تجري عملية ضرب مصفوفة-متجه لحساب حدود الخطأ.

توثيق الوظيفة/الروتين الفرعي

subroutine cla_gbamv (integer trans, integer m, integer n, integer kl, integer ku, real alpha, complex, dimension( ldab, * ) ab, integer ldab, complex, dimension( * ) x, integer incx, real beta, real, dimension( * ) y, integer incy)

CLA_GBAMV تجري عملية ضرب مصفوفة-متجه لحساب حدود الخطأ.

الغرض:

!>
!> CLA_GBAMV  تجري إحدى عمليات المصفوفة-المتجه:
!>
!>         y := alpha*abs(A)*abs(x) + beta*abs(y),
!>    أو   y := alpha*abs(A)**T*abs(x) + beta*abs(y),
!>
!> حيث alpha و beta أعداد قياسية، x و y متجهات، و A مصفوفة
!> بحجم m × n.
!>
!> هذه الوظيفة تستخدم بشكل أساسي في حساب حدود الخطأ.
!> للحماية من النقصان تحت الحد الأدنى أثناء التقييم، يتم اضطراب
!> المكونات في المتجه الناتج بعيداً عن الصفر بـ (N+1)
!> أضعاف حد النقصان تحت الحد الأدنى. لمنع الأخطاء الكبيرة بشكل غير ضروري
!> للهياكل المضمنة في المصفوفات العامة،
!> لا يتم اضطراب المكونات الصفرية. يتم اعتبار إدخال صفري
!> إذا كانت جميع عمليات الضرب المتضمنة
!> في حساب هذا الإدخال لديها مضاعف صفري واحد على الأقل.
!>

المعلمات

TRANS 
!>          TRANS هو عدد صحيح (INTEGER)
!>           عند الإدخال، يحدد TRANS العملية التي ستجرى كما يلي:
!>
!>             BLAS_NO_TRANS      y := alpha*abs(A)*abs(x) + beta*abs(y)
!>             BLAS_TRANS         y := alpha*abs(A**T)*abs(x) + beta*abs(y)
!>             BLAS_CONJ_TRANS    y := alpha*abs(A**T)*abs(x) + beta*abs(y)
!>
!>           لا تتغير عند الخروج.
!>

M

!>          M هو عدد صحيح (INTEGER)
!>           عند الإدخال، يحدد M عدد صفوف المصفوفة A.
!>           يجب أن يكون M على الأقل صفر.
!>           لا يتغير عند الخروج.
!>

N

!>          N هو عدد صحيح (INTEGER)
!>           عند الإدخال، يحدد N عدد أعمدة المصفوفة A.
!>           يجب أن يكون N على الأقل صفر.
!>           لا يتغير عند الخروج.
!>

KL

!>          KL هو عدد صحيح (INTEGER)
!>           عدد الأقطار الفرعية ضمن نطاق A.  KL >= 0.
!>

KU

!>          KU هو عدد صحيح (INTEGER)
!>           عدد الأقطار الفوقية ضمن نطاق A.  KU >= 0.
!>

ALPHA

!>          ALPHA هو عدد حقيقي (REAL)
!>           عند الإدخال، يحدد ALPHA العدد القياسي alpha.
!>           لا يتغير عند الخروج.
!>

AB

!>          AB هو مصفوفة مركبة (COMPLEX)، حجمها (LDAB,n)
!>           قبل الدخول، يجب أن يحتوي الجزء الرئيسي بحجم m × n من المصفوفة AB
!>           على مصفوفة المعاملات.
!>           لا تتغير عند الخروج.
!>

LDAB

!>          LDAB هو عدد صحيح (INTEGER)
!>           عند الإدخال، يحدد LDAB البعد الأول للمصفوفة AB كما هو معلن
!>           في البرنامج المستدعي (الفرعي). يجب أن يكون LDAB على الأقل
!>           max( 1, m ).
!>           لا يتغير عند الخروج.
!>

X

!>          X هو مصفوفة مركبة (COMPLEX)، حجمها
!>           ( 1 + ( n - 1 )*abs( INCX ) ) عندما TRANS = 'N' أو 'n'
!>           وعلى الأقل
!>           ( 1 + ( m - 1 )*abs( INCX ) ) بخلاف ذلك.
!>           قبل الدخول، يجب أن تحتوي المصفوفة X المعدلة على
!>           المتجه x.
!>           لا تتغير عند الخروج.
!>

INCX

!>          INCX هو عدد صحيح (INTEGER)
!>           عند الإدخال، يحدد INCX الفاصل بين عناصر
!>           X. يجب أن لا يكون INCX صفر.
!>           لا يتغير عند الخروج.
!>

BETA

!>          BETA هو عدد حقيقي (REAL)
!>           عند الإدخال، يحدد BETA العدد القياسي beta. عندما يكون BETA
!>           مقدماً كصفر، فلن تحتاج Y إلى تعيين عند الإدخال.
!>           لا يتغير عند الخروج.
!>

Y

!>          Y هي مصفوفة حقيقية (REAL)، حجمها
!>           ( 1 + ( m - 1 )*abs( INCY ) ) عندما TRANS = 'N' أو 'n'
!>           وعلى الأقل
!>           ( 1 + ( n - 1 )*abs( INCY ) ) بخلاف ذلك.
!>           قبل الدخول مع قيمة BETA غير صفرية، يجب أن تحتوي المصفوفة Y
!>           المعدلة على المتجه y. عند الخروج، يتم استبدال Y بالمتجه
!>           y المحدث.
!>           إذا كان m أو n صفرًا، فلن يتم الرجوع إلى Y وسيتم إنهاء الوظيفة بسرعة.
!>

INCY

!>          INCY هو عدد صحيح (INTEGER)
!>           عند الإدخال، يحدد INCY الفاصل بين عناصر
!>           Y. يجب أن لا يكون INCY صفر.
!>           لا يتغير عند الخروج.
!>
!>  روتين BLAS من المستوى 2.
!>
