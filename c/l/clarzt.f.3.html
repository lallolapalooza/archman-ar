subroutine clarzt (character direct, character storev, integer n, integer k, complex, dimension( ldv, * ) v, integer ldv, complex, dimension( * ) tau, complex, dimension( ldt, * ) t, integer ldt)

CLARZT تشكل العامل المثلثي T للمعاكس القطاعي H = I - vtvH.

الغرض:

!>
!> CLARZT تشكل العامل المثلثي T لمعاكس قطاعي معقد H ذو رتبة > n، والذي يتم تعريفه كحاصل ضرب k معاكسات أولية.
!>
!> إذا كان DIRECT = 'F', فإن H = H(1) H(2) . . . H(k) و T مثلثي علوي؛
!>
!> إذا كان DIRECT = 'B', فإن H = H(k) . . . H(2) H(1) و T مثلثي سفلي.
!>
!> إذا كان STOREV = 'C', فإن المتجه الذي يحدد المعاكس الأولي H(i) مخزن في العمود i للمصفوفة V، و
!>
!>    H  =  I - V * T * V**H
!>
!> إذا كان STOREV = 'R', فإن المتجه الذي يحدد المعاكس الأولي H(i) مخزن في الصف i للمصفوفة V، و
!>
!>    H  =  I - V**H * T * V
!>
!> حاليًا، يتم دعم STOREV = 'R' و DIRECT = 'B' فقط.
!>

المعلمات

DIRECT 
!>          DIRECT هو CHARACTER*1
!>          يحدد ترتيب ضرب المعاكسات الأولية لتشكيل المعاكس القطاعي:
!>          = 'F': H = H(1) H(2) . . . H(k) (أمامي، غير مدعوم بعد)
!>          = 'B': H = H(k) . . . H(2) H(1) (خلفي)
!>

STOREV

!>          STOREV هو CHARACTER*1
!>          يحدد كيفية تخزين المتجهات التي تحدد المعاكسات الأولية (انظر أيضًا المزيد من التفاصيل):
!>          = 'C': عمودياً (غير مدعوم بعد)
!>          = 'R': أفقياً
!>

N

!>          N هو INTEGER
!>          رتبة المعاكس القطاعي H. N >= 0.
!>

K

!>          K هو INTEGER
!>          رتبة العامل المثلثي T (= عدد المعاكسات الأولية). K >= 1.
!>

V

!>          V هو مصفوفة COMPLEX، ذات أبعاد
!>                               (LDV,K) إذا كان STOREV = 'C'
!>                               (LDV,N) إذا كان STOREV = 'R'
!>          المصفوفة V. انظر المزيد من التفاصيل.
!>

LDV

!>          LDV هو INTEGER
!>          البعد الرائد للمصفوفة V.
!>          إذا كان STOREV = 'C', فإن LDV >= max(1,N)؛ إذا كان STOREV = 'R', فإن LDV >= K.
!>

TAU

!>          TAU هو مصفوفة COMPLEX، ذات أبعاد (K)
!>          يجب أن تحتوي TAU(i) على المعامل العددي للمعاكس الأولي H(i).
!>

T

!>          T هو مصفوفة COMPLEX، ذات أبعاد (LDT,K)
!>          العامل المثلثي T ذو الأبعاد k × k للمعكوس القطاعي.
!>          إذا كان DIRECT = 'F', فإن T مثلثي علوي؛ إذا كان DIRECT = 'B', فإن T مثلثي
!>          سفلي. لا يتم استخدام باقي المصفوفة.
!>

LDT

!>          LDT هو INTEGER
!>          البعد الرائد للمصفوفة T. LDT >= K.
!>

المزيد من التفاصيل:

!>
!>  يتم توضيح شكل المصفوفة V وتخزين المتجهات التي تحدد H(i) بشكل أفضل من خلال المثال التالي مع n = 5 و
!>  k = 3. العناصر التي تساوي 1 ليست مخزنة؛ يتم تعديل عناصر المصفوفة المقابلة واستعادتها عند الخروج.
!>  باقي المصفوفة غير مستخدم.
!>
!>  DIRECT = 'F' و STOREV = 'C':         DIRECT = 'F' و STOREV = 'R':
!>
!>                                              ______V_____
!>         ( v1 v2 v3 )                        /            \
!>         ( v1 v2 v3 )                      ( v1 v1 v1 v1 v1 . . . . 1 )
!>     V = ( v1 v2 v3 )                      ( v2 v2 v2 v2 v2 . . . 1   )
!>         ( v1 v2 v3 )                      ( v3 v3 v3 v3 v3 . . 1     )
!>         ( v1 v2 v3 )
!>            .  .  .
!>            .  .  .
!>            1  .  .
!>               1  .
!>                  1
!>
!>  DIRECT = 'B' و STOREV = 'C':         DIRECT = 'B' و STOREV = 'R':
!>
!>                                                        ______V_____
!!>            1                                          /            \
!>            .  1                           ( 1 . . . . v1 v1 v1 v1 v1 )
!>            .  .  1                        ( . 1 . . . v2 v2 v2 v2 v2 )
!>            .  .  .                        ( . . 1 . . v3 v3 v3 v3 v3 )
!>            .  .  .
!>         ( v1 v2 v3 )
!>         ( v1 v2 v3 )
!>     V = ( v1 v2 v3 )
!>         ( v1 v2 v3 )
!>         ( v1 v2 v3 )
!>
