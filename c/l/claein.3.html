CLAEIN تحسب متجهًا ذاتيًا يمينًا أو يسارًا معينًا لمصفوفة هسنبرغ علوية باستخدام التكرار العكسي.

توثيق الوظيفة/الإجراء الفرعي

subroutine claein (logical rightv, logical noinit, integer n, complex, dimension( ldh, * ) h, integer ldh, complex w, complex, dimension( * ) v, complex, dimension( ldb, * ) b, integer ldb, real, dimension( * ) rwork, real eps3, real smlnum, integer info)

CLAEIN تحسب متجهًا ذاتيًا يمينًا أو يسارًا معينًا لمصفوفة هسنبرغ علوية باستخدام التكرار العكسي.

الغرض:

!>
!> CLAEIN تستخدم التكرار العكسي للعثور على متجه ذاتي أيمن أو أيسر
!> مقابل القيمة الذاتية W لمصفوفة هسنبرغ العلوية H.
!>

المعلمات

RIGHTV 
!>          RIGHTV هو LOGICAL (منطقي)
!>          = .TRUE. : حساب المتجه الذاتي الأيمن؛
!>          = .FALSE.: حساب المتجه الذاتي الأيسر.
!>

NOINIT

!>          NOINIT هو LOGICAL (منطقي)
!>          = .TRUE. : لم يتم توفير متجه أولي في V
!>          = .FALSE.: تم توفير متجه أولي في V.
!>

N

!>          N هو INTEGER (عدد صحيح)
!>          رتبة المصفوفة H. N >= 0.
!>

H

!>          H هي مصفوفة COMPLEX (مركبة)، البعد (LDH,N)
!>          مصفوفة هسنبرغ العلوية H.
!>

LDH

!>          LDH هو INTEGER (عدد صحيح)
!>          البعد الرئيسي للمصفوفة H. LDH >= max(1,N).
!>

W

!>          W هو COMPLEX (مركب)
!>          القيمة الذاتية لـ H التي يجب حساب متجهها الذاتي الأيمن أو
!>          الأيسر المقابل لها.
!>

V

!>          V هي مصفوفة COMPLEX (مركبة)، البعد (N)
!>          عند الإدخال، إذا كان NOINIT = .FALSE.، فيجب أن تحتوي V على متجه
!>          بدء للتكرار العكسي؛ بخلاف ذلك، لا يلزم تعيين V.
!>          عند الخروج، تحتوي V على المتجه الذاتي المحسوب، ويتم تطبيعه
!>          بحيث يكون للمكون ذي القيمة المطلقة الأكبر قيمة مطلقة 1؛ هنا
!>          تُؤخذ القيمة المطلقة لعدد مركب (x,y) على أنها
!>          |x| + |y|.
!>

B

!>          B هي مصفوفة COMPLEX (مركبة)، البعد (LDB,N)
!>

LDB

!>          LDB هو INTEGER (عدد صحيح)
!>          البعد الرئيسي للمصفوفة B. LDB >= max(1,N).
!>

RWORK

!>          RWORK هي مصفوفة REAL (حقيقية)، البعد (N)
!>

EPS3

!>          EPS3 هو REAL (حقيقي)
!>          قيمة صغيرة تعتمد على الآلة تُستخدم لزيادة القيم الذاتية
!>          القريبة، ولتعديل المحاور الصفرية.
!>

SMLNUM

!>          SMLNUM هو REAL (حقيقي)
!>          قيمة قريبة من عتبة التدفق السفلي تعتمد على الآلة.
!>

INFO

!>          INFO هو INTEGER (عدد صحيح)
!>          = 0:  انتهاء ناجح
!>          = 1:  لم يتقارب التكرار العكسي؛ تم تعيين V إلى التكرار
!>                الأخير.
!>
