subroutine clapmr (logical forwrd, integer m, integer n, complex, dimension( ldx, * ) x, integer ldx, integer, dimension( * ) k)

CLAPMR يعيد ترتيب صفوف مصفوفة حسب ما يحدده متجه التبديل.

الغرض:

!>
!> CLAPMR يعيد ترتيب صفوف المصفوفة M x N المسماة X كما هو محدد
!> بواسطة التبديل K(1),K(2),...,K(M) للأعداد الصحيحة 1,...,M.
!> إذا كان FORWRD = .TRUE. ، تبديل أمامي:
!>
!>      يتم نقل X(K(I),*) إلى X(I,*) من أجل I = 1,2,...,M.
!>
!> إذا كان FORWRD = .FALSE. ، تبديل خلفي:
!>
!>      يتم نقل X(I,*) إلى X(K(I),*) من أجل I = 1,2,...,M.
!>

المعلمات

FORWRD 
!>          FORWRD هو LOGICAL
!>          = .TRUE. ، تبديل أمامي
!>          = .FALSE. ، تبديل خلفي
!>

M

!>          M هو INTEGER
!>          عدد صفوف المصفوفة X. M >= 0.
!>

N

!>          N هو INTEGER
!>          عدد أعمدة المصفوفة X. N >= 0.
!>

X

!>          X هي مصفوفة COMPLEX ، أبعادها (LDX,N)
!>          عند الإدخال، المصفوفة M x N المسماة X.
!>          عند الخروج، تحتوي X على المصفوفة X المبدلة.
!>

LDX

!>          LDX هو INTEGER
!>          البعد الرائد للمصفوفة X، LDX >= MAX(1,M).
!>

K

!>          K هي مصفوفة INTEGER ، أبعادها (M)
!>          عند الإدخال، تحتوي K على متجه التبديل. يتم استخدام K
!>          كمساحة عمل داخلية، ولكن يتم إعادة تعيينها إلى قيمتها الأصلية عند
!>          الخروج.
!>
