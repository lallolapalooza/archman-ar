subroutine cla_geamv (integer trans, integer m, integer n, real alpha, complex, dimension( lda, * ) a, integer lda, complex, dimension( * ) x, integer incx, real beta, real, dimension( * ) y, integer incy)

CLA_GEAMV يحسب ناتج مصفوفة-متجه باستخدام مصفوفة عامة لحساب حدود الخطأ.

الغرض:

!>
!> CLA_GEAMV  يقوم بإحدى عمليات المصفوفة-المتجه التالية:
!>
!>         y := alpha*abs(A)*abs(x) + beta*abs(y),
!>    أو   y := alpha*abs(A)**T*abs(x) + beta*abs(y),
!>
!> حيث alpha و beta هما قيمتان عدديتان، x و y هما متجهان، و A هي مصفوفة
!> بحجم m × n.
!>
!> هذه الوظيفة تستخدم بشكل أساسي في حساب حدود الخطأ.
!> للحماية من التدفق السفلي أثناء التقييم، يتم إزاحة المكونات في
!> المتجه الناتج بعيدًا عن الصفر بمقدار (N+1)
!> مضروبًا في عتبة التدفق السفلي. لمنع الأخطاء غير الضرورية للمصفوفات
!> الفرعية المضمنة في المصفوفات العامة،
!> لا يتم إزاحة المكونات الصفرية. تعتبر القيمة الصفرية
!> إذا كانت جميع عمليات الضرب المتضمنة
!> في حساب هذه القيمة تحتوي على مضاعف صفري واحد على الأقل.
!>

المعلمات

TRANS 
!>          TRANS هو عدد صحيح (INTEGER)
!>           عند الدخول، يحدد TRANS العملية التي سيتم إجراؤها كما يلي:
!>
!>             BLAS_NO_TRANS      y := alpha*abs(A)*abs(x) + beta*abs(y)
!>             BLAS_TRANS         y := alpha*abs(A**T)*abs(x) + beta*abs(y)
!>             BLAS_CONJ_TRANS    y := alpha*abs(A**T)*abs(x) + beta*abs(y)
!>
!>           لا يتغير عند الخروج.
!>

M

!>          M هو عدد صحيح (INTEGER)
!>           عند الدخول، يحدد M عدد صفوف المصفوفة A.
!>           يجب أن يكون M أكبر من أو يساوي صفر.
!>           لا يتغير عند الخروج.
!>

N

!>          N هو عدد صحيح (INTEGER)
!>           عند الدخول، يحدد N عدد أعمدة المصفوفة A.
!>           يجب أن يكون N أكبر من أو يساوي صفر.
!>           لا يتغير عند الخروج.
!>

ALPHA

!>          ALPHA هو قيمة حقيقية (REAL)
!>           عند الدخول، تحدد ALPHA القيمة العددية alpha.
!>           لا يتغير عند الخروج.
!>

A

!>          A هي مصفوفة مركبة (COMPLEX)، ذات بعد (LDA,n)
!>           قبل الدخول، يجب أن يحتوي الجزء الرئيسي بحجم m × n من المصفوفة A
!>           على معاملات المصفوفة.
!>           لا يتغير عند الخروج.
!>

LDA

!>          LDA هو عدد صحيح (INTEGER)
!>           عند الدخول، يحدد LDA البعد الأول للمصفوفة A كما تم تعريفه
!>           في البرنامج المستدعي (الإجراء الفرعي). يجب أن يكون LDA أكبر من أو يساوي
!>           max( 1, m ).
!>           لا يتغير عند الخروج.
!>

X

!>          X هي مصفوفة مركبة (COMPLEX)، ذات بعد
!>           ( 1 + ( n - 1 )*abs( INCX ) ) عندما تكون TRANS = 'N' أو 'n'
!>           و على الأقل
!>           ( 1 + ( m - 1 )*abs( INCX ) ) خلاف ذلك.
!>           قبل الدخول، يجب أن تحتوي المصفوفة X ذات الأبعاد المزادة على
!>           المتجه x.
!>           لا يتغير عند الخروج.
!>

INCX

!>          INCX هو عدد صحيح (INTEGER)
!>           عند الدخول، يحدد INCX الزيادة لعناصر X. يجب ألا يكون INCX صفراً.
!>           لا يتغير عند الخروج.
!>

BETA

!>          BETA هو قيمة حقيقية (REAL)
!>           عند الدخول، تحدد BETA القيمة العددية beta. عندما تكون BETA
!>           مُقدمة كصفر، فإن Y لا تحتاج إلى تعيين عند الدخول.
!>           لا يتغير عند الخروج.
!>

Y

!>          Y هي مصفوفة حقيقية (REAL)، ذات بعد
!>           ( 1 + ( m - 1 )*abs( INCY ) ) عندما تكون TRANS = 'N' أو 'n'
!>           و على الأقل
!>           ( 1 + ( n - 1 )*abs( INCY ) ) خلاف ذلك.
!>           قبل الدخول عندما تكون BETA غير صفرية، يجب أن تحتوي المصفوفة Y ذات الأبعاد المزادة على
!>           المتجه y. عند الخروج، يتم استبدال Y بالمتجه y المحدث.
!>           إذا كان m أو n يساوي صفرًا، فإن Y لا تتم الإشارة إليه وتقوم الوظيفة
!>           بإرجاع سريع.
!>

INCY

!>          INCY هو عدد صحيح (INTEGER)
!>           عند الدخول، يحدد INCY الزيادة لعناصر Y. يجب ألا يكون INCY صفراً.
!>           لا يتغير عند الخروج.
!>
!>  إجراء من المستوى 2 من BLAS.
!>
