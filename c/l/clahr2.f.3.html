subroutine clahr2 (n, k, nb, a, lda, tau, t, ldt, y, ldy)
CLAHR2 يقلل عدد الأعمدة الأولى المحددة للمصفوفة المستطيلة العامة A بحيث تصبح العناصر الموجودة تحت القطر الفرعي المحدد صفرًا، ويعيد المصفوفات المساعدة اللازمة لتطبيق التحويل على الجزء غير المخفض من A.

توثيق الوظيفة/الروتين الفرعي

subroutine clahr2 (integer n, integer k, integer nb, complex, dimension( lda, * ) a, integer lda, complex, dimension( nb ) tau, complex, dimension( ldt, nb ) t, integer ldt, complex, dimension( ldy, nb ) y, integer ldy)

CLAHR2 يقلل عدد الأعمدة الأولى المحددة للمصفوفة العامة المعقدة A بحيث تصبح العناصر الموجودة تحت القطر الفرعي المحدد أصفارًا، ويعيد المصفوفات المساعدة اللازمة لتطبيق التحويل على الجزء غير المخفض من A.

الغرض:

!>
!> CLAHR2 يقلل الأعمدة NB الأولى لمصفوفة A المعقدة العامة n-BY-(n-k+1)
!> بحيث تصبح العناصر الموجودة تحت القطر الفرعي k صفرًا. يتم إجراء التخفيض
!> بواسطة تحويل تشابه وحدوي Q**H * A * Q. يعيد الروتين المصفوفات V و T
!> التي تحدد Q كمُنعكس كتلة I - V*T*v**H، وكذلك المصفوفة Y = A * V * T.
!>
!> هذا روتين مساعد يتم استدعاؤه بواسطة CGEHRD.
!>

المعلمات

N 
!>          N هو عدد صحيح (INTEGER)
!>          ترتيب المصفوفة A.
!>

K

!>          K هو عدد صحيح (INTEGER)
!>          الإزاحة للتخفيض. يتم تخفيض العناصر الموجودة تحت القطر الفرعي
!>          k في الأعمدة NB الأولى إلى الصفر. K < N.
!>

NB

!>          NB هو عدد صحيح (INTEGER)
!>          عدد الأعمدة المطلوب تخفيضها.
!>

A

!>          A هي مصفوفة معقدة (COMPLEX array)، البعد (LDA,N-K+1)
!>          عند الإدخال، المصفوفة العامة A ذات الأبعاد n × (n-k+1).
!>          عند الخروج، يتم الكتابة فوق العناصر الموجودة على القطر الفرعي
!>          k وفوقه في الأعمدة NB الأولى بالعناصر المقابلة للمصفوفة المخفضة؛
!>          العناصر الموجودة تحت القطر الفرعي k، مع مصفوفة TAU، تمثل المصفوفة Q
!>          كمنتج من المنعكسات الأولية. الأعمدة الأخرى من A لا تتغير. انظر التفاصيل الإضافية.
!>

LDA

!>          LDA هو عدد صحيح (INTEGER)
!>          البعد الرئيسي للمصفوفة A. LDA >= max(1,N).
!>

TAU

!>          TAU هي مصفوفة معقدة (COMPLEX array)، البعد (NB)
!>          العوامل القياسية للمنعكسات الأولية. انظر التفاصيل الإضافية.
!>

T

!>          T هي مصفوفة معقدة (COMPLEX array)، البعد (LDT,NB)
!>          المصفوفة المثلثية العليا T.
!>

LDT

!>          LDT هو عدد صحيح (INTEGER)
!>          البعد الرئيسي للمصفوفة T. LDT >= NB.
!>

Y

!>          Y هي مصفوفة معقدة (COMPLEX array)، البعد (LDY,NB)
!>          المصفوفة Y ذات الأبعاد n × nb.
!>

LDY

!>          LDY هو عدد صحيح (INTEGER)
!>          البعد الرئيسي للمصفوفة Y. LDY >= N.
!>

تفاصيل إضافية:

!>
!>  يتم تمثيل المصفوفة Q كمنتج من nb منعكس أولي
!>
!>     Q = H(1) H(2) . . . H(nb).
!>
!>  كل H(i) له الشكل
!>
!>     H(i) = I - tau * v * v**H
!>
!>  حيث tau عامل قياسي معقد، و v متجه معقد مع
!>  v(1:i+k-1) = 0، v(i+k) = 1؛ v(i+k+1:n) مخزن عند الخروج في
!>  A(i+k+1:n,i)، و tau في TAU(i).
!>
!>  تشكل عناصر المتجهات v معًا المصفوفة V ذات الأبعاد (n-k+1)-by-nb
!>  والتي تُستخدم، مع T و Y، لتطبيق التحويل على الجزء غير المخفض من المصفوفة،
!>  باستخدام تحديث بالشكل:
!>  A := (I - V*T*V**H) * (A - Y*V**H).
!>
!>  محتويات A عند الخروج موضحة بالمثال التالي
!>  مع n = 7، k = 3 و nb = 2:
!>
!>     ( a   a   a   a   a )
!>     ( a   a   a   a   a )
!>     ( a   a   a   a   a )
!>     ( h   h   a   a   a )
!>     ( v1  h   a   a   a )
!>     ( v1  v2  a   a   a )
!>     ( v1  v2  a   a   a )
!>
!>  حيث تمثل a عنصرًا من المصفوفة الأصلية A، ويمثل h عنصرًا معدلاً
!>  من المصفوفة هسنبرغ العليا H، ويمثل vi عنصرًا من المتجه الذي يحدد H(i).
!>
!>  هذا الروتين الفرعي هو تعديل طفيف لـ CLAHRD في LAPACK-3.0
!>  ويتضمن تحسينات مقترحة من قبل Quintana-Orti و Van de Gejin.
!>  لاحظ أن مدخلات A(1:K,2:NB) تختلف عن تلك
!>  التي يعيدها الروتين الأصلي CLAHRD في LAPACK-3.0. (هذا
!>  الروتين الفرعي غير متوافق مع الإصدار السابق مع CLAHRD في LAPACK-3.0.)
!>

المراجع:

Gregorio Quintana-Orti and Robert van de Geijn, "Improving the performance of reduction to Hessenberg form," ACM Transactions on Mathematical Software, 32(2):180-194, June 2006.
