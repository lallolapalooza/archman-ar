الوظائف/الإجراءات الفرعية


real function cla_gercond_c (trans, n, a, lda, af, ldaf, ipiv, c, capply, info, work, rwork)
CLA_GERCOND_C تحسب عدد شرط op(A)*inv(diag(c)) للمصفوفات العامة.

توثيق الوظيفة/الإجراء الفرعي

real function cla_gercond_c (character trans, integer n, complex, dimension( lda, * ) a, integer lda, complex, dimension( ldaf, * ) af, integer ldaf, integer, dimension( * ) ipiv, real, dimension( * ) c, logical capply, integer info, complex, dimension( * ) work, real, dimension( * ) rwork)

CLA_GERCOND_C تحسب عدد شرط ذروة اللانهائي (infinity norm condition number) لـ op(A)*inv(diag(c)) للمصفوفات العامة.

الغرض:

!>
!>
!>    CLA_GERCOND_C يحسب عدد شرط ذروة اللانهائي لـ
!>    op(A) * inv(diag(C)) حيث C متجه حقيقي.
!>

المعلمات

TRANS 
!>          TRANS is CHARACTER*1
!>     يحدد شكل نظام المعادلات:
!>       = 'N':  A * X = B     (بدون تبديل)
!>       = 'T':  A**T * X = B  (تبديل)
!>       = 'C':  A**H * X = B  (تبديل مترافق = تبديل)
!>

N

!>          N is INTEGER
!>     عدد المعادلات الخطية، أي رتبة
!>     المصفوفة A.  N >= 0.
!>

A

!>          A is COMPLEX array, dimension (LDA,N)
!>     عند الدخول، المصفوفة N-by-N A
!>

LDA

!>          LDA is INTEGER
!>     البعد الرائد للمصفوفة A.  LDA >= max(1,N).
!>

AF

!>          AF is COMPLEX array, dimension (LDAF,N)
!>     العوامل L و U من التحليل
!>     A = P*L*U كما تم حسابه بواسطة CGETRF.
!>

LDAF

!>          LDAF is INTEGER
!>     البعد الرائد للمصفوفة AF.  LDAF >= max(1,N).
!>

IPIV

!>          IPIV is INTEGER array, dimension (N)
!>     فهارس المحور من تحليل A = P*L*U
!>     كما تم حسابه بواسطة CGETRF؛ تم تبديل الصف i من المصفوفة
!>     مع الصف IPIV(i).
!>

C

!>          C is REAL array, dimension (N)
!>     المتجه C في الصيغة op(A) * inv(diag(C)).
!>

CAPPLY

!>          CAPPLY is LOGICAL
!>     إذا كان .TRUE. فيتم الوصول إلى المتجه C في الصيغة أعلاه.
!>

INFO

!>          INFO is INTEGER
!>       = 0:  انتهاء ناجح.
!>     i > 0:  المعلمة i غير صالحة.
!>

WORK

!>          WORK is COMPLEX array, dimension (2*N).
!>     مساحة العمل.
!>

RWORK

!>          RWORK is REAL array, dimension (N).
!>     مساحة العمل.
!>
