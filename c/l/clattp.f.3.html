توثيق الوظيفة/الوحدة الفرعية

subroutine clattp (integer imat, character uplo, character trans, character diag, integer, dimension( 4 ) iseed, integer n, complex, dimension( * ) ap, complex, dimension( * ) b, complex, dimension( * ) work, real, dimension( * ) rwork, integer info)

CLATTP

الغرض:

!>
!> CLATTP تولد مصفوفة اختبار مثلثية في تخزين مكدس.
!> IMAT و UPLO يحددان بشكل فريد خصائص مصفوفة الاختبار،
!> والتي يتم إرجاعها في المصفوفة AP.
!>

المعلمات

IMAT 
!>          IMAT هو عدد صحيح
!>          مفتاح عدد صحيح يصف أي مصفوفة يجب توليدها لهذا
!>          المسار.
!>

UPLO

!>          UPLO هو حرف واحد
!>          يحدد ما إذا كانت المصفوفة A ستكون مثلثية علوية أم سفلية.
!>          = 'U':  مثلثية علوية
!>          = 'L':  مثلثية سفلية
!>

TRANS

!>          TRANS هو حرف واحد
!>          يحدد ما إذا كانت المصفوفة أو منقولتها ستستخدم.
!>          = 'N':  بدون منقولة
!>          = 'T':  منقولة
!>          = 'C':  منقولة مقترنة
!>

DIAG

!>          DIAG هو حرف واحد
!>          يحدد ما إذا كانت المصفوفة A مثلثية وحدوية أم لا.
!>          = 'N':  مثلثية غير وحدوية
!>          = 'U':  مثلثية وحدوية
!>

ISEED

!>          ISEED هو مصفوفة أعداد صحيحة، حجمها (4)
!>          متجه البذرة لمولد الأرقام العشوائية (يستخدم في
!>          CLATMS). يتم تعديله عند الخروج.
!>

N

!>          N هو عدد صحيح
!>          رتبة المصفوفة التي سيتم توليدها.
!>

AP

!>          AP هي مصفوفة مركبة، حجمها (N*(N+1)/2)
!>          المصفوفة A المثلثية العلوية أو السفلية، مكدسة عمودياً في
!>          مصفوفة خطية. يتم تخزين العمود j من A في المصفوفة
!>          AP على النحو التالي:
!>          إذا كان UPLO = 'U'، فإن AP((j-1)*j/2 + i) = A(i,j) لـ 1<=i<=j؛
!>          إذا كان UPLO = 'L'،
!>             AP((j-1)*(n-j) + j*(j+1)/2 + i-j) = A(i,j) لـ j<=i<=n.
!>

B

!>          B هي مصفوفة مركبة، حجمها (N)
!>          المتجه للجانب الأيمن، إذا كان IMAT > 10.
!>

WORK

!>          WORK هي مصفوفة مركبة، حجمها (2*N)
!>

RWORK

!>          RWORK هي مصفوفة حقيقية، حجمها (N)
!>

INFO

!>          INFO هو عدد صحيح
!>          = 0:  انتهاء ناجح
!>          < 0:  إذا كان INFO = -i، فإن المعلمة i كانت لها قيمة غير قانونية
!>
