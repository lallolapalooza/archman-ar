subroutine clacrm (integer m, integer n, complex, dimension( lda, * ) a, integer lda, real, dimension( ldb, * ) b, integer ldb, complex, dimension( ldc, * ) c, integer ldc, real, dimension( * ) rwork)

CLACRM تضرب مصفوفة معقدة بمصفوفة مربعة حقيقية.

الغرض:

!>
!> CLACRM تقوم بعملية ضرب مصفوفة-مصفوفة بسيطة جدًا:
!>          C := A * B,
!> حيث A هي M × N ومعقدة؛ B هي N × N وحقيقية؛
!> C هي M × N ومعقدة.
!>

المعلمات

M 
!>          M هو عدد صحيح
!>          عدد صفوف المصفوفة A والمصفوفة C.
!>          M >= 0.
!>

N

!>          N هو عدد صحيح
!>          عدد الأعمدة والصفوف للمصفوفة B و
!>          عدد الأعمدة للمصفوفة C.
!>          N >= 0.
!>

A

!>          A هي مصفوفة معقدة، البعد (LDA, N)
!>          عند الدخول، تحتوي A على المصفوفة A بحجم M × N.
!>

LDA

!>          LDA هو عدد صحيح
!>          البعد الرائد للمصفوفة A. LDA >=max(1,M).
!>

B

!>          B هي مصفوفة حقيقية، البعد (LDB, N)
!>          عند الدخول، تحتوي B على المصفوفة B بحجم N × N.
!>

LDB

!>          LDB هو عدد صحيح
!>          البعد الرائد للمصفوفة B. LDB >=max(1,N).
!>

C

!>          C هي مصفوفة معقدة، البعد (LDC, N)
!>          عند الخروج، تحتوي C على المصفوفة C بحجم M × N.
!>

LDC

!>          LDC هو عدد صحيح
!>          البعد الرائد للمصفوفة C. LDC >=max(1,N).
!>

RWORK

!>          RWORK هي مصفوفة حقيقية، البعد (2*M*N)
!>
