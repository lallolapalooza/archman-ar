subroutine clarz (character side, integer m, integer n, integer l, complex, dimension( * ) v, integer incv, complex tau, complex, dimension( ldc, * ) c, integer ldc, complex, dimension( * ) work)

CLARZ يطبق عاكسًا عنصريًا (كما تم إرجاعه بواسطة stzrzf) على مصفوفة عامة.

الغرض:

!>
!> CLARZ يطبق عاكسًا عنصريًا مركبًا H على مصفوفة مركبة عامة C
!> بحجم M × N، من اليسار أو اليمين. يتم تمثيل H بالشكل
!>
!>       H = I - tau * v * v**H
!>
!> حيث tau هو عدد مركب و v هو متجه مركب.
!>
!> إذا كان tau = 0، يتم اعتبار H مصفوفة الوحدة.
!>
!> لتطبيق H**H (المرافق المنقول لـ H)، قم بتمرير conjg(tau) بدلاً من
!> tau.
!>
!> H هو ناتج ضرب k عواكس عنصرية كما تم إرجاعه بواسطة CTZRZF.
!>

المعلمات

SIDE 
!>          SIDE هو CHARACTER*1
!>          = 'L': تكوين  H * C
!>          = 'R': تكوين  C * H
!>

M

!>          M هو INTEGER
!>          عدد صفوف المصفوفة C.
!>

N

!>          N هو INTEGER
!>          عدد أعمدة المصفوفة C.
!>

L

!>          L هو INTEGER
!>          عدد عناصر المتجه V التي تحتوي على
!>          الجزء الهادف من متجهات هاوسهولدر.
!>          إذا كان SIDE = 'L'، فإن M >= L >= 0، إذا كان SIDE = 'R'، فإن N >= L >= 0.
!>

V

!>          V هو مصفوفة COMPLEX، البعد (1+(L-1)*abs(INCV))
!>          المتجه v في تمثيل H كما تم إرجاعه بواسطة
!>          CTZRZF. لا يتم استخدام V إذا كان TAU = 0.
!>

INCV

!>          INCV هو INTEGER
!>          الزيادة بين عناصر v. INCV <> 0.
!>

TAU

!>          TAU هو COMPLEX
!>          القيمة tau في تمثيل H.
!>

C

!>          C هي مصفوفة COMPLEX، البعد (LDC,N)
!>          عند الإدخال، المصفوفة C بحجم M × N.
!>          عند الخروج، يتم الكتابة فوق C بالمصفوفة H * C إذا كان SIDE = 'L'،
!>          أو C * H إذا كان SIDE = 'R'.
!>

LDC

!>          LDC هو INTEGER
!>          البعد الرئيسي للمصفوفة C. LDC >= max(1,M).
!>

WORK

!>          WORK هي مصفوفة COMPLEX، البعد
!>                         (N) إذا كان SIDE = 'L'
!>                      أو (M) إذا كان SIDE = 'R'
!>
