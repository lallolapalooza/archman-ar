توثيق الوظيفة/الإجراء الفرعي

subroutine claqge (integer m, integer n, complex, dimension( lda, * ) a, integer lda, real, dimension( * ) r, real, dimension( * ) c, real rowcnd, real colcnd, real amax, character equed)

CLAQGE تقوم بقياس مصفوفة مستطيلة عامة، باستخدام عوامل القياس الصفية والعمودية المحسوبة بواسطة sgeequ.

الغرض:

!>
!> CLAQGE توازن مصفوفة عامة M في N باستخدام عوامل القياس الصفية والعمودية في المتجهات R و C.
!>

المعلمات

M 
!>          M هو عدد صحيح
!>          عدد صفوف المصفوفة A.  M >= 0.
!>

N

!>          N هو عدد صحيح
!>          عدد أعمدة المصفوفة A.  N >= 0.
!>

A

!>          A هو مصفوفة مركبة، أبعادها (LDA,N)
!>          عند الإدخال، المصفوفة A بحجم M في N.
!>          عند الخروج، المصفوفة المتوازنة. انظر EQUED لمعرفة شكل
!>          المصفوفة المتوازنة.
!>

LDA

!>          LDA هو عدد صحيح
!>          البعد الرائد للمصفوفة A.  LDA >= max(M,1).
!>

R

!>          R هو مصفوفة حقيقية، أبعادها (M)
!>          عوامل القياس الصفية للمصفوفة A.
!>

C

!>          C هو مصفوفة حقيقية، أبعادها (N)
!>          عوامل القياس العمودية للمصفوفة A.
!>

ROWCND

!>          ROWCND هو قيمة حقيقية
!>          نسبة أصغر R(i) إلى أكبر R(i).
!>

COLCND

!>          COLCND هو قيمة حقيقية
!>          نسبة أصغر C(i) إلى أكبر C(i).
!>

AMAX

!>          AMAX هو قيمة حقيقية
!>          القيمة المطلقة لأكبر عنصر في المصفوفة.
!>

EQUED

!>          EQUED هو حرف واحد * 1
!>          يحدد شكل التوازن الذي تم إجراؤه.
!>          = 'N':  لا يوجد توازن
!>          = 'R':  توازن صفي، أي أن A تم ضربها مسبقًا بـ
!>                  diag(R).
!>          = 'C':  توازن عمودي، أي أن A تم ضربها لاحقًا بـ
!>                  diag(C).
!>          = 'B':  توازن صفي وعمودي، أي أن A تم استبدالها بـ
!>                  diag(R) * A * diag(C).
!>

المعلمات الداخلية:

!>  THRESH هي قيمة حدية تستخدم لتحديد ما إذا كان يجب إجراء قياس صفي أو عمودي
!>  بناءً على نسبة عوامل القياس الصفية أو العمودية. إذا كان ROWCND < THRESH، يتم إجراء القياس الصفي، وإذا كان
!>  COLCND < THRESH، يتم إجراء القياس العمودي.
!>
!>  LARGE و SMALL هي قيم حدية تستخدم لتحديد ما إذا كان يجب إجراء قياس صفي
!>  بناءً على الحجم المطلق لأكبر عنصر في المصفوفة. إذا كان AMAX > LARGE أو AMAX < SMALL، يتم إجراء القياس الصفي.
!>
