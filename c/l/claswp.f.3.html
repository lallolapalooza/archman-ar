subroutine claswp (n, a, lda, k1, k2, ipiv, incx)
CLASWP تجري سلسلة من تبادلات الصفوف على مصفوفة مستطيلة عامة.

توثيق الوظيفة/الإجراء الفرعي

subroutine claswp (integer n, complex, dimension( lda, * ) a, integer lda, integer k1, integer k2, integer, dimension( * ) ipiv, integer incx)

CLASWP تجري سلسلة من تبادلات الصفوف على مصفوفة مستطيلة عامة.

الغرض:

!>
!> CLASWP تجري سلسلة من تبادلات الصفوف على المصفوفة A.
!> يتم إجراء تبادل صف واحد لكل من الصفوف K1 إلى K2 من A.
!>

المعلمات

N 
!>          N هو عدد صحيح (INTEGER)
!>          عدد أعمدة المصفوفة A.
!>

A

!>          A هو مصفوفة مركبة (COMPLEX)، بأبعاد (LDA,N)
!>          عند الإدخال، مصفوفة البعد N التي سيتم تطبيق تبادلات الصفوف عليها.
!>          عند الخروج، المصفوفة المتبادلة.
!>

LDA

!>          LDA هو عدد صحيح (INTEGER)
!>          البعد الرائد للمصفوفة A.
!>

K1

!>          K1 هو عدد صحيح (INTEGER)
!>          العنصر الأول في IPIV الذي سيتم إجراء تبادل صف له.
!>

K2

!>          K2 هو عدد صحيح (INTEGER)
!>          (K2-K1+1) هو عدد عناصر IPIV التي سيتم إجراء تبادل صف لها.
!>

IPIV

!>          IPIV هو مصفوفة أعداد صحيحة (INTEGER)، بأبعاد (K1+(K2-K1)*abs(INCX))
!>          المتجه بفهارس المحاور. يتم الوصول فقط إلى العناصر في المواضع
!>          من K1 إلى K1+(K2-K1)*abs(INCX) في IPIV.
!>          IPIV(K1+(K-K1)*abs(INCX)) = L يعني أنه سيتم تبديل الصفوف K و L.
!>

INCX

!>          INCX هو عدد صحيح (INTEGER)
!>          الزيادة بين القيم المتتالية لـ IPIV. إذا كان INCX
!>          سلبيًا، يتم تطبيق المحاور بترتيب عكسي.
!>
