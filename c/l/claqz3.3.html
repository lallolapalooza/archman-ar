ILSCHUR

!>              يحدد ما إذا كان سيتم تحديث الشكل Schur الكامل أم لا
!>

ILQ

!>          ILQ هو LOGICAL (منطقي)
!>              يحدد ما إذا كان سيتم تحديث المصفوفة Q أم لا
!>

ILZ

!>          ILZ هو LOGICAL (منطقي)
!>              يحدد ما إذا كان سيتم تحديث المصفوفة Z أم لا
!>

N

!>          N هو INTEGER (عدد صحيح)
!>          رتبة المصفوفات A, B, Q, و Z. N >= 0.
!>

ILO

!>          ILO هو INTEGER (عدد صحيح)
!>

IHI

!>          IHI هو INTEGER (عدد صحيح)
!>

NSHIFTS

!>          NSHIFTS هو INTEGER (عدد صحيح)
!>          العدد المطلوب من التحولات لاستخدامها
!>

NBLOCK_DESIRED

!>          NBLOCK_DESIRED هو INTEGER (عدد صحيح)
!>          حجم نوافذ الحساب المطلوبة
!>

ALPHA

!>          ALPHA هو مصفوفة COMPLEX (مركبة). يحتوي SR على
!>          الأجزاء ألفا للتحولات المراد استخدامها.
!>

BETA

!>          BETA هو مصفوفة COMPLEX (مركبة). يحتوي SS على
!>          مقياس التحولات المراد استخدامها.
!>

A

!>          A هي مصفوفة COMPLEX (مركبة)، البعد (LDA, N)
!>

LDA

!>          LDA هو INTEGER (عدد صحيح)
!>          البعد الرائد للمصفوفة A. LDA >= max( 1, N ).
!>

B

!>          B هي مصفوفة COMPLEX (مركبة)، البعد (LDB, N)
!>

LDB

!>          LDB هو INTEGER (عدد صحيح)
!>          البعد الرائد للمصفوفة B. LDB >= max( 1, N ).
!>

Q

!>          Q هي مصفوفة COMPLEX (مركبة)، البعد (LDQ, N)
!>

LDQ

!>          LDQ هو INTEGER (عدد صحيح)
!>

Z

!>          Z هي مصفوفة COMPLEX (مركبة)، البعد (LDZ, N)
!>

LDZ

!>          LDZ هو INTEGER (عدد صحيح)
!>

QC

!>          QC هي مصفوفة COMPLEX (مركبة)، البعد (LDQC, NBLOCK_DESIRED)
!>

LDQC

!>          LDQC هو INTEGER (عدد صحيح)
!>

ZC

!>          ZC هي مصفوفة COMPLEX (مركبة)، البعد (LDZC, NBLOCK_DESIRED)
!>

LDZC

!>          LDZ هو INTEGER (عدد صحيح)
!>

WORK

!>          WORK هي مصفوفة COMPLEX (مركبة)، البعد (MAX(1,LWORK))
!>          عند الخروج، إذا كان INFO >= 0، فإن WORK(1) يحتوي على LWORK الأمثل.
!>

LWORK

!>          LWORK هو INTEGER (عدد صحيح)
!>          بعد المصفوفة WORK. LWORK >= max(1,N).
!>
!>          إذا كان LWORK = -1، فسيتم افتراض استعلام مساحة العمل؛ تقوم الروتين الفرعي
!>          بحساب حجم مصفوفة WORK الأمثل فقط، وإرجاع
!>          هذه القيمة كالإدخال الأول للمصفوفة WORK، ولا يتم إصدار أي رسالة خطأ
!>          متعلقة بـ LWORK بواسطة XERBLA.
!>

INFO

!>          INFO هو INTEGER (عدد صحيح)
!>          = 0: خروج ناجح
!>          < 0: إذا كان INFO = -i، فإن الوسيط i كان لديه قيمة غير قانونية
!>
