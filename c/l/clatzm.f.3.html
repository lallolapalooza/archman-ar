subroutine clatzm (character side, integer m, integer n, complex, dimension( * ) v, integer incv, complex tau, complex, dimension( ldc, * ) c1, complex, dimension( ldc, * ) c2, integer ldc, complex, dimension( * ) work)

CLATZM

الغرض:

!>
!> تم إهمال هذا الإجراء الفرعي واستبداله بالإجراء الفرعي CUNMRZ.
!>
!> يطبق CLATZM مصفوفة هاوسهولدر تم إنشاؤها بواسطة CTZRQF على مصفوفة.
!>
!> لتكن P = I - tau*u*u**H,   u = ( 1 ),
!>                               ( v )
!> حيث v هو متجه (m-1) إذا كان SIDE = 'L'، أو متجه (n-1) إذا كان
!> SIDE = 'R'.
!>
!> إذا كان SIDE يساوي 'L'، لتكن
!>        C = [ C1 ] 1
!>            [ C2 ] m-1
!>              n
!> عندها يتم استبدال C بـ P*C.
!>
!> إذا كان SIDE يساوي 'R'، لتكن
!>        C = [ C1, C2 ] m
!>               1  n-1
!> عندها يتم استبدال C بـ C*P.
!>

المعاملات

SIDE 
!>          SIDE هو CHARACTER*1
!>          = 'L': شكل P * C
!>          = 'R': شكل C * P
!>

M

!>          M هو INTEGER
!>          عدد صفوف المصفوفة C.
!>

N

!>          N هو INTEGER
!>          عدد أعمدة المصفوفة C.
!>

V

!>          V هي مصفوفة COMPLEX، البعد
!>                  (1 + (M-1)*abs(INCV)) إذا كان SIDE = 'L'
!>                  (1 + (N-1)*abs(INCV)) إذا كان SIDE = 'R'
!>          المتجه v في تمثيل P. لا يتم استخدام V
!>          إذا كان TAU = 0.
!>

INCV

!>          INCV هو INTEGER
!>          الزيادة بين عناصر v. INCV <> 0
!>

TAU

!>          TAU هو COMPLEX
!>          القيمة tau في تمثيل P.
!>

C1

!>          C1 هي مصفوفة COMPLEX، البعد
!>                         (LDC,N) إذا كان SIDE = 'L'
!>                         (M,1)   إذا كان SIDE = 'R'
!>          عند الدخول، المتجه n C1 إذا كان SIDE = 'L'، أو المتجه m C1
!>          إذا كان SIDE = 'R'.
!>
!>          عند الخروج، الصف الأول من P*C إذا كان SIDE = 'L'، أو العمود الأول
!>          من C*P إذا كان SIDE = 'R'.
!>

C2

!>          C2 هي مصفوفة COMPLEX، البعد
!>                         (LDC, N)   إذا كان SIDE = 'L'
!>                         (LDC, N-1) إذا كان SIDE = 'R'
!>          عند الدخول، المصفوفة (m - 1) x n C2 إذا كان SIDE = 'L'، أو
!>          المصفوفة m x (n - 1) C2 إذا كان SIDE = 'R'.
!>
!>          عند الخروج، الصفوف 2:m من P*C إذا كان SIDE = 'L'، أو الأعمدة 2:m من C*P
!>          إذا كان SIDE = 'R'.
!>

LDC

!>          LDC هو INTEGER
!>          البعد الرئيسي للمصفوفات C1 و C2.
!>          LDC >= max(1,M).
!>

WORK

!>          WORK هي مصفوفة COMPLEX، البعد
!>                      (N) إذا كان SIDE = 'L'
!>                      (M) إذا كان SIDE = 'R'
!>
