توثيق الوظيفة/الروتين الفرعي

recursive subroutine clarft (character direct, character storev, integer n, integer k, complex, dimension( ldv, * ) v, integer ldv, complex, dimension( * ) tau, complex, dimension( ldt, * ) t, integer ldt)

CLARFT تشكل العامل المثلثي T للمنعكس الكتلي H = I - vtvH

الغرض:

!>
!> CLARFT تشكل العامل المثلثي T لمنعكس كتلي معقد H من الدرجة n، والذي يتم تعريفه كحاصل ضرب k منعكسات ابتدائية.
!>
!> إذا كان DIRECT = 'F'، فإن H = H(1) H(2) . . . H(k) و T مثلثي علوي؛
!>
!> إذا كان DIRECT = 'B'، فإن H = H(k) . . . H(2) H(1) و T مثلثي سفلي.
!>
!> إذا كان STOREV = 'C'، فإن المتجه الذي يعرف المنعكس الابتدائي H(i) مخزن في العمود i من المصفوفة V، و
!>
!>    H  =  I - V * T * V**H
!>
!> إذا كان STOREV = 'R'، فإن المتجه الذي يعرف المنعكس الابتدائي H(i) مخزن في الصف i من المصفوفة V، و
!>
!>    H  =  I - V**H * T * V
!>

المعلمات

DIRECT 
!>          DIRECT هو CHARACTER*1
!>          يحدد ترتيب ضرب المنعكسات الابتدائية لتشكيل المنعكس الكتلي:
!>          = 'F': H = H(1) H(2) . . . H(k) (أمامي)
!>          = 'B': H = H(k) . . . H(2) H(1) (خلفي)
!>

STOREV

!>          STOREV هو CHARACTER*1
!>          يحدد كيفية تخزين المتجهات التي تعرف المنعكسات الابتدائية (انظر أيضًا التفاصيل الإضافية):
!>          = 'C': عمودي
!>          = 'R': صفي
!>

N

!>          N هو INTEGER
!>          درجة المنعكس الكتلي H. N >= 0.
!>

K

!>          K هو INTEGER
!>          درجة العامل المثلثي T (= عدد المنعكسات الابتدائية). K >= 1.
!>

V

!>          V هي مصفوفة COMPLEX، ذات بعد
!>                               (LDV,K) إذا كان STOREV = 'C'
!>                               (LDV,N) إذا كان STOREV = 'R'
!>          المصفوفة V. انظر التفاصيل الإضافية.
!>

LDV

!>          LDV هو INTEGER
!>          البعد الرئيسي للمصفوفة V.
!>          إذا كان STOREV = 'C'، فإن LDV >= max(1,N)؛ إذا كان STOREV = 'R'، فإن LDV >= K.
!>

TAU

!>          TAU هي مصفوفة COMPLEX، ذات بعد (K)
!>          يجب أن تحتوي TAU(i) على العامل القياسي للمنعكس الابتدائي H(i).
!>

T

!>          T هي مصفوفة COMPLEX، ذات بعد (LDT,K)
!>          العامل المثلثي T بحجم k × k للمنعكس الكتلي.
!>          إذا كان DIRECT = 'F'، فإن T مثلثي علوي؛ إذا كان DIRECT = 'B'، فإن T مثلثي سفلي.
!>          بقية المصفوفة لا تستخدم.
!>

LDT

!>          LDT هو INTEGER
!>          البعد الرئيسي للمصفوفة T. LDT >= K.
!>

تفاصيل إضافية:

!>
!>  يتم توضيح شكل المصفوفة V وتخزين المتجهات التي تعرف H(i) بشكل أفضل من خلال المثال التالي مع n = 5 و k = 3.
!>  العناصر التي تساوي 1 غير مخزنة.
!>
!>  DIRECT = 'F' و STOREV = 'C':         DIRECT = 'F' و STOREV = 'R':
!>
!>               V = (  1       )                 V = (  1 v1 v1 v1 v1 )
!>                   ( v1  1    )                     (     1 v2 v2 v2 )
!>                   ( v1 v2  1 )                     (        1 v3 v3 )
!>                   ( v1 v2 v3 )
!>                   ( v1 v2 v3 )
!>
!>  DIRECT = 'B' و STOREV = 'C':         DIRECT = 'B' و STOREV = 'R':
!>
!>               V = ( v1 v2 v3 )                 V = ( v1 v1  1       )
!>                   ( v1 v2 v3 )                     ( v2 v2 v2  1    )
!>                   (  1 v2 v3 )                     ( v3 v3 v3 v3  1 )
!>                   (     1 v3 )
!>                   (        1 )
!>
