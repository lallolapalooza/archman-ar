subroutine cla_gbrfsx_extended (integer prec_type, integer trans_type, integer n, integer kl, integer ku, integer nrhs, complex, dimension( ldab, * ) ab, integer ldab, complex, dimension( ldafb, * ) afb, integer ldafb, integer, dimension( * ) ipiv, logical colequ, real, dimension( * ) c, complex, dimension( ldb, * ) b, integer ldb, complex, dimension( ldy, * ) y, integer ldy, real, dimension( * ) berr_out, integer n_norms, real, dimension( nrhs, * ) err_bnds_norm, real, dimension( nrhs, * ) err_bnds_comp, complex, dimension( * ) res, real, dimension(*) ayb, complex, dimension( * ) dy, complex, dimension( * ) y_tail, real rcond, integer ithresh, real rthresh, real dz_ub, logical ignore_cwise, integer info)

CLA_GBRFSX_EXTENDED تحسن الحل المحسوب لنظام معادلات خطية لمصفوفات متعددة النطاقات عامة عن طريق إجراء تكرار تحسين بدقة إضافية وتوفر حدود خطأ وتقديرات خطأ خلفي للحل.

الغرض:

!>
!> CLA_GBRFSX_EXTENDED تحسن الحل المحسوب لنظام معادلات خطية عن طريق إجراء
!> تكرار تحسين بدقة إضافية وتوفر حدود خطأ وتقديرات خطأ خلفي للحل.
!> يتم استدعاء هذا الروتين الفرعي بواسطة CGBRFSX لإجراء التحسين التكراري.
!> بالإضافة إلى حد الخطأ المعياري، يقدم الكود حد خطأ عنصري نسبي أقصى
!> إذا كان ذلك ممكنًا. انظر التعليقات لـ ERR_BNDS_NORM و ERR_BNDS_COMP
!> للحصول على تفاصيل حدود الخطأ. لاحظ أن هذا
!> الروتين الفرعي مسؤول فقط عن تعيين الحقول الثانية لـ
!> ERR_BNDS_NORM و ERR_BNDS_COMP.
!>

المعلمات

PREC_TYPE 
!>          PREC_TYPE هو INTEGER
!>     يحدد الدقة الوسيطة المراد استخدامها في التحسين.
!>     تُعرّف القيمة بواسطة ILAPREC(P) حيث P هو CHARACTER و P
!>          = 'S':  Single (أحادي)
!>          = 'D':  Double (مزدوج)
!>          = 'I':  Indigenous (أصلي)
!>          = 'X' or 'E':  Extra (إضافي)
!>

TRANS_TYPE

!>          TRANS_TYPE هو INTEGER
!>     يحدد عملية النقل على A.
!>     تُعرّف القيمة بواسطة ILATRANS(T) حيث T هو CHARACTER و T
!>          = 'N':  No transpose (بدون نقل)
!>          = 'T':  Transpose (نقل)
!>          = 'C':  Conjugate transpose (نقل مترافق)
!>

N

!>          N هو INTEGER
!>     عدد المعادلات الخطية، أي ترتيب
!>     المصفوفة A. N >= 0.
!>

KL

!>          KL هو INTEGER
!>     عدد النطاقات الفرعية داخل نطاق A. KL >= 0.
!>

KU

!>          KU هو INTEGER
!>     عدد النطاقات الفائقة داخل نطاق A. KU >= 0
!>

NRHS

!>          NRHS هو INTEGER
!>     عدد الأطراف اليمنى، أي عدد أعمدة
!>     المصفوفة B.
!>

AB

!>          AB هو مصفوفة COMPLEX، ذات الأبعاد (LDAB,N)
!>     عند الدخول، المصفوفة N-by-N AB.
!>

LDAB

!>          LDAB هو INTEGER
!>     البعد الرائد للمصفوفة AB. LDAB >= max(1,N).
!>

AFB

!>          AFB هو مصفوفة COMPLEX، ذات الأبعاد (LDAF,N)
!>     العوامل L و U من التحليل
!>     A = P*L*U كما تم حسابه بواسطة CGBTRF.
!>

LDAFB

!>          LDAFB هو INTEGER
!>     البعد الرائد للمصفوفة AF. LDAF >= max(1,N).
!>

IPIV

!>          IPIV هو مصفوفة INTEGER، ذات الأبعاد (N)
!>     فهارس المحاور من تحليل A = P*L*U
!>     كما تم حسابه بواسطة CGBTRF؛ تم تبديل الصف i للمصفوفة
!>     مع الصف IPIV(i).
!>

COLEQU

!>          COLEQU هو LOGICAL
!>     إذا كان .TRUE. فقد تم إجراء موازنة الأعمدة لـ A قبل استدعاء
!>     هذا الروتين. هذا ضروري لحساب الحل وحدود الخطأ
!>     بشكل صحيح.
!>

C

!>          C هو مصفوفة REAL، ذات الأبعاد (N)
!>     عوامل قياس الأعمدة لـ A. إذا كان COLEQU = .FALSE.، لا يتم
!>     الوصول إلى C. إذا تم إدخال C، يجب أن يكون كل عنصر من C قوة
!>     للأساس لضمان حل موثوق وتقديرات خطأ.
!>     لا تسبب المقاييس باستخدام قوى الأساس أخطاء تقريب إلا إذا
!>     كانت النتيجة تحت التدفق أو تجاوز التدفق. تؤدي أخطاء التقريب أثناء
!>     المقاييس إلى تحسين مصفوفة ليست مكافئة للمصفوفة
!>     المدخلة، مما ينتج تقديرات خطأ قد لا تكون
!>     موثوقة.
!>

B

!>          B هو مصفوفة COMPLEX، ذات الأبعاد (LDB,NRHS)
!>     مصفوفة الطرف الأيمن B.
!>

LDB

!>          LDB هو INTEGER
!>     البعد الرائد للمصفوفة B. LDB >= max(1,N).
!>

Y

!>          Y هو مصفوفة COMPLEX، ذات الأبعاد (LDY,NRHS)
!>     عند الدخول، مصفوفة الحل X، كما تم حسابها بواسطة CGBTRS.
!>     عند الخروج، مصفوفة الحل المحسنة Y.
!>

LDY

!>          LDY هو INTEGER
!>     البعد الرائد للمصفوفة Y. LDY >= max(1,N).
!>

BERR_OUT

!>          BERR_OUT هو مصفوفة REAL، ذات الأبعاد (NRHS)
!>     عند الخروج، يحتوي BERR_OUT(j) على الخطأ الخلفي النسبي العنصري
!>     للطرف الأيمن j من الصيغة
!>         max(i) ( abs(RES(i)) / ( abs(op(A_s))*abs(Y) + abs(B_s) )(i) )
!>     حيث abs(Z) هي القيمة المطلقة العنصرية للمصفوفة
!>     أو المتجه Z. يتم حساب هذا بواسطة CLA_LIN_BERR.
!>

N_NORMS

!>          N_NORMS هو INTEGER
!>     يحدد حدود الخطأ التي سيتم إرجاعها (انظر ERR_BNDS_NORM
!>     و ERR_BNDS_COMP).
!>     إذا كان N_NORMS >= 1، قم بإرجاع حدود الخطأ المعيارية.
!>     إذا كان N_NORMS >= 2، قم بإرجاع حدود الخطأ العنصرية.
!>

ERR_BNDS_NORM

!>          ERR_BNDS_NORM هو مصفوفة REAL، ذات الأبعاد (NRHS, N_ERR_BNDS)
!>     لكل طرف أيمن، تحتوي هذه المصفوفة على معلومات حول
!>     حدود الخطأ المختلفة وأرقام الشروط المقابلة
!>     للخطأ النسبي المعياري، والذي يُعرّف على النحو التالي:
!>
!>     الخطأ النسبي المعياري في متجه الحل i:
!>             max_j (abs(XTRUE(j,i) - X(j,i)))
!>            ------------------------------
!>                  max_j abs(X(j,i))
!>
!>     يتم فهرسة المصفوفة حسب نوع معلومات الخطأ كما هو موضح
!>     أدناه. يوجد حاليًا ما يصل إلى ثلاث قطع من المعلومات
!>     يتم إرجاعها.
!>
!>     الفهرس الأول في ERR_BNDS_NORM(i,:) يتوافق مع الطرف الأيمن
!>     i.
!>
!>     الفهرس الثاني في ERR_BNDS_NORM(:,err) يحتوي على الـ f
