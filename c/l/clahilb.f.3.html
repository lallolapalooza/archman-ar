subroutine clahilb (integer n, integer nrhs, complex, dimension(lda,n) a, integer lda, complex, dimension(ldx, nrhs) x, integer ldx, complex, dimension(ldb, nrhs) b, integer ldb, real, dimension(n) work, integer info, character*3 path)

CLAHILB

الغرض:

!>
!> CLAHILB تقوم بإنشاء مصفوفة هيلبرت بحجم N × N مُقاسة مع
!> NRHS أطراف يمنى في B وحلول في X بحيث A*X=B.
!>
!> يتم قياس مصفوفة هيلبرت بواسطة M = LCM(1, 2, ..., 2*N-1) بحيث تكون جميع
!> المدخلات أعدادًا صحيحة. الأطراف اليمنى هي الأعمدة NRHS الأولى
!> من مصفوفة الوحدة M. والحلول هي الأعمدة NRHS الأولى
!> للمصفوفة الهيلبرتية المعكوسة.
!>
!> شرط مصفوفة هيلبرت ينمو بشكل أسي مع
!> حجمها، تقريبًا O(e ** (3.5*N)). بالإضافة إلى ذلك، المصفوفات
!> الهيلبرتية المعكوسة بعد بعد معين لا يمكن
!> إنشاؤها بدقة بدون دقة إضافية. يتم استنفاد الدقة
!> عندما يكون أكبر مدخل في المصفوفة الهيلبرتية المعكوسة أكبر من
!> 2 مرفوعة إلى قوة عدد البتات في الجزء العشري لنوع البيانات
!> المستخدم بالإضافة إلى واحد، وهو 24 للدقة الأحادية.
!>
!> بالدقة الأحادية، يكون الحل المُنشأ دقيقًا لـ N <= 6 وله
!> خطأ صغير مكوني لـ 7 <= N <= 11.
!>

المعلمات

N 
!>          N هو عدد صحيح (INTEGER)
!>          بعد المصفوفة A.
!>

NRHS

!>          NRHS هو عدد صحيح (INTEGER)
!>          العدد المطلوب للأطراف اليمنى.
!>

A

!>          A هي مصفوفة مركبة (COMPLEX)، البعد (LDA, N)
!>          المصفوفة الهيلبرتية المُقاسة المُنشأة.
!>

LDA

!>          LDA هو عدد صحيح (INTEGER)
!>          البعد الرائد للمصفوفة A. LDA >= N.
!>

X

!>          X هي مصفوفة مركبة (COMPLEX)، البعد (LDX, NRHS)
!>          الحلول الدقيقة المُنشأة. حاليًا، الأعمدة NRHS الأولى
!>          للمصفوفة الهيلبرتية المعكوسة.
!>

LDX

!>          LDX هو عدد صحيح (INTEGER)
!>          البعد الرائد للمصفوفة X. LDX >= N.
!>

B

!>          B هي مصفوفة حقيقية (REAL)، البعد (LDB, NRHS)
!>          الأطراف اليمنى المُنشأة. حاليًا، الأعمدة NRHS الأولى
!>          لـ LCM(1, 2, ..., 2*N-1) * مصفوفة الوحدة.
!>

LDB

!>          LDB هو عدد صحيح (INTEGER)
!>          البعد الرائد للمصفوفة B. LDB >= N.
!>

WORK

!>          WORK هي مصفوفة حقيقية (REAL)، البعد (N)
!>

INFO

!>          INFO هو عدد صحيح (INTEGER)
!>          = 0: انتهاء ناجح
!>          = 1: N كبير جدًا؛ لا يزال يتم إنشاء البيانات ولكن قد لا
!>               تكون دقيقة.
!>          < 0: إذا كان INFO = -i، فإن المعلمة i كانت قيمة غير قانونية
!>

PATH

!>          PATH هو حرف (CHARACTER*3)
!>          اسم مسار LAPACK.
!>
