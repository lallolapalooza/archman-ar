subroutine تكرارية clarft (character direct, character storev, integer n, integer k, complex, dimension( ldv, * ) v, integer ldv, complex, dimension( * ) tau, complex, dimension( ldt, * ) t, integer ldt)

CLARFT تشكل العامل المثلثي T للمنعكس الجزئي H = I - vtvH

الغرض:

!>
!> CLARFT تشكل العامل المثلثي T للمنعكس الجزئي المعقد H من الدرجة n،
!> والذي يتم تعريفه كناتج لـ k منعكس جزئي.
!>
!> إذا كان DIRECT = 'F'، فإن H = H(1) H(2) . . . H(k) و T مثلثية علوية؛
!>
!> إذا كان DIRECT = 'B'، فإن H = H(k) . . . H(2) H(1) و T مثلثية سفلية.
!>
!> إذا كان STOREV = 'C'، فإن المتجه الذي يحدد المنعكس الجزئي H(i)
!> مخزن في العمود i للمصفوفة V، و
!>
!>    H  =  I - V * T * V**H
!>
!> إذا كان STOREV = 'R'، فإن المتجه الذي يحدد المنعكس الجزئي H(i)
!> مخزن في الصف i للمصفوفة V، و
!>
!>    H  =  I - V**H * T * V
!>

المعلمات

DIRECT 
!>          DIRECT هو CHARACTER*1
!>          يحدد ترتيب ضرب المنعكسات الجزئية لتشكيل المنعكس الجزئي:
!>          = 'F': H = H(1) H(2) . . . H(k) (أمامي)
!>          = 'B': H = H(k) . . . H(2) H(1) (خلفي)
!>

STOREV

!>          STOREV هو CHARACTER*1
!>          يحدد كيفية تخزين المتجهات التي تحدد المنعكسات الجزئية
!>          (انظر التفاصيل الإضافية):
!>          = 'C': حسب العمود
!>          = 'R': حسب الصف
!>

N

!>          N هو INTEGER
!>          درجة المنعكس الجزئي H. N >= 0.
!>

K

!>          K هو INTEGER
!>          درجة العامل المثلثي T (= عدد المنعكسات الجزئية). K >= 1.
!>

V

!>          V هي مصفوفة COMPLEX، أبعادها
!>                               (LDV,K) إذا كان STOREV = 'C'
!>                               (LDV,N) إذا كان STOREV = 'R'
!>          المصفوفة V. انظر التفاصيل الإضافية.
!>

LDV

!>          LDV هو INTEGER
!>          البعد الرائد للمصفوفة V.
!>          إذا كان STOREV = 'C'، فإن LDV >= max(1,N)؛ إذا كان STOREV = 'R'، فإن LDV >= K.
!>

TAU

!>          TAU هي مصفوفة COMPLEX، أبعادها (K)
!>          يجب أن تحتوي TAU(i) على العامل القياسي للمنعكس الجزئي H(i).
!>

T

!>          T هي مصفوفة COMPLEX، أبعادها (LDT,K)
!>          العامل المثلثي k × k T للمنعكس الجزئي.
!>          إذا كان DIRECT = 'F'، فإن T مثلثية علوية؛ إذا كان DIRECT = 'B'، فإن T
!>          مثلثية سفلية. باقي المصفوفة لا يستخدم.
!>

LDT

!>          LDT هو INTEGER
!>          البعد الرائد للمصفوفة T. LDT >= K.
!>

تفاصيل إضافية:

!>
!>  يتم توضيح شكل المصفوفة V وتخزين المتجهات التي تحدد H(i) بشكل أفضل
!>  من خلال المثال التالي مع n = 5 و k = 3. العناصر التي تساوي 1 غير مخزنة.
!>
!>  DIRECT = 'F' و STOREV = 'C':         DIRECT = 'F' و STOREV = 'R':
!>
!>               V = (  1       )                 V = (  1 v1 v1 v1 v1 )
!>                   ( v1  1    )                     (     1 v2 v2 v2 )
!>                   ( v1 v2  1 )                     (        1 v3 v3 )
!>                   ( v1 v2 v3 )
!>                   ( v1 v2 v3 )
!>
!>  DIRECT = 'B' و STOREV = 'C':         DIRECT = 'B' و STOREV = 'R':
!>
!>               V = ( v1 v2 v3 )                 V = ( v1 v1  1       )
!>                   ( v1 v2 v3 )                     ( v2 v2 v2  1    )
!>                   (  1 v2 v3 )                     ( v3 v3 v3 v3  1 )
!>                   (     1 v3 )
!>                   (        1 )
!>
