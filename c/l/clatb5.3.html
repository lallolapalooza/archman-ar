subroutine clatb5 (character*3 path, integer imat, integer n, character type, integer kl, integer ku, real anorm, integer mode, real cndnum, character dist)

CLATB5

الغرض:

!>
!> CLATB5 يحدد المعلمات لمولد المصفوفة بناءً على نوع المصفوفة المراد إنشاؤها.
!>

المعلمات

PATH 
!>          PATH هو CHARACTER*3
!>          اسم مسار LAPACK.
!>

IMAT

!>          IMAT هو INTEGER
!>          مفتاح رقمي يصف أي مصفوفة سيتم إنشاؤها لهذا المسار.
!>

N

!>          N هو INTEGER
!>          عدد الصفوف والأعمدة في المصفوفة المراد إنشاؤها.
!>

TYPE

!>          TYPE هو CHARACTER*1
!>          نوع المصفوفة المراد إنشاؤها:
!>          = 'S':  مصفوفة متماثلة
!>          = 'P':  مصفوفة موجبة (شبه) محددة
!>          = 'N':  مصفوفة غير متماثلة
!>

KL

!>          KL هو INTEGER
!>          عرض النطاق السفلي للمصفوفة المراد إنشاؤها.
!>

KU

!>          KU هو INTEGER
!>          عرض النطاق العلوي للمصفوفة المراد إنشاؤها.
!>

ANORM

!>          ANORM هو REAL
!>          القيمة المطلوبة للمصفوفة المراد إنشاؤها. يتم قياس مصفوفة
!>          القيم المفردة أو القيم الذاتية القطرية بواسطة هذه القيمة.
!>

MODE

!>          MODE هو INTEGER
!>          مفتاح يشير إلى كيفية اختيار متجه القيم الذاتية.
!>

CNDNUM

!>          CNDNUM هو REAL
!>          العدد الشرطي المطلوب.
!>

DIST

!>          DIST هو CHARACTER*1
!>          نوع التوزيع الذي سيتم استخدامه بواسطة مولد الأرقام العشوائية.
!>
