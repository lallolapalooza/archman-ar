توثيق الوظيفة/الإجراء الفرعي

subroutine clalsd (character uplo, integer smlsiz, integer n, integer nrhs, real, dimension( * ) d, real, dimension( * ) e, complex, dimension( ldb, * ) b, integer ldb, real rcond, integer rank, complex, dimension( * ) work, real, dimension( * ) rwork, integer, dimension( * ) iwork, integer info)

CLALSD يستخدم تحليل القيمة المفردة للمصفوفة A لحل مشكلة المربعات الصغرى.

الغرض:

!>
!> CLALSD يستخدم تحليل القيمة المفردة للمصفوفة A لحل مشكلة
!> المربعات الصغرى لإيجاد X لتقليل المعيار الإقليدي لكل
!> عمود من A*X-B، حيث A هي مصفوفة علوية ثنائية الأقطار بحجم N × N،
!> و X و B هما مصفوفتان بحجم N × NRHS. الحل X يكتب فوق B.
!>
!> القيم المفردة للمصفوفة A الأصغر من RCOND مضروبة في أكبر
!> قيمة مفردة تُعامل على أنها صفر في حل مشكلة المربعات الصغرى؛
!> في هذه الحالة يتم إرجاع حل بالمعيار الأدنى.
!> القيم المفردة الفعلية يتم إرجاعها في D بترتيب تصاعدي.
!>
!>

المعاملات

UPLO 
!>          UPLO is CHARACTER*1
!>         = 'U': D و E يحددان مصفوفة ثنائية الأقطار علوية.
!>         = 'L': D و E يحددان مصفوفة ثنائية الأقطار سفلية.
!>

SMLSIZ

!>          SMLSIZ is INTEGER
!>         الحد الأقصى لحجم المشاكل الفرعية في أسفل شجرة الحساب.
!>

N

!>          N is INTEGER
!>         بعد المصفوفة ثنائية الأقطار. N >= 0.
!>

NRHS

!>          NRHS is INTEGER
!>         عدد أعمدة B. يجب أن يكون NRHS على الأقل 1.
!>

D

!>          D is REAL array, dimension (N)
!>         عند الإدخال يحتوي D على القطر الرئيسي للمصفوفة ثنائية
!>         الأقطار. عند الخروج، إذا كان INFO = 0، يحتوي D على
!>         قيمه المفردة.
!>

E

!>          E is REAL array, dimension (N-1)
!>         يحتوي على إدخالات فوق القطر الرئيسي للمصفوفة ثنائية
!>         الأقطار. عند الخروج، تم تدمير E.
!>

B

!>          B is COMPLEX array, dimension (LDB,NRHS)
!>         عند الإدخال، يحتوي B على الجوانب اليمنى لمشكلة المربعات
!>         الصغرى. عند الخروج، يحتوي B على الحل X.
!>

LDB

!>          LDB is INTEGER
!>         البعد الرائد لـ B في البرنامج الفرعي المستدعي.
!>         يجب أن يكون LDB على الأقل max(1,N).
!>

RCOND

!>          RCOND is REAL
!>         القيم المفردة للمصفوفة A الأصغر من أو تساوي RCOND مضروبة
!>         في أكبر قيمة مفردة تُعامل على أنها صفر في حل مشكلة
!>         المربعات الصغرى. إذا كان RCOND سالبًا،
!>         يتم استخدام دقة الآلة بدلاً من ذلك.
!>         على سبيل المثال، إذا كان diag(S)*X=B هي مشكلة المربعات الصغرى،
!>         حيث diag(S) هي مصفوفة قطرية للقيم المفردة، فإن
!>         الحل سيكون X(i) = B(i) / S(i) إذا كانت S(i) أكبر من
!>         RCOND*max(S)، و X(i) = 0 إذا كانت S(i) أصغر من أو تساوي
!>         RCOND*max(S).
!>

RANK

!>          RANK is INTEGER
!>         عدد القيم المفردة للمصفوفة A الأكبر من RCOND مضروبة
!>         في أكبر قيمة مفردة.
!>

WORK

!>          WORK is COMPLEX array, dimension (N * NRHS).
!>

RWORK

!>          RWORK is REAL array, dimension at least
!>         (9*N + 2*N*SMLSIZ + 8*N*NLVL + 3*SMLSIZ*NRHS +
!>         MAX( (SMLSIZ+1)**2, N*(1+NRHS) + 2*NRHS ),
!>         حيث
!>         NLVL = MAX( 0, INT( LOG_2( MIN( M,N )/(SMLSIZ+1) ) ) + 1 )
!>

IWORK

!>          IWORK is INTEGER array, dimension (3*N*NLVL + 11*N).
!>

INFO

!>          INFO is INTEGER
!>         = 0:  انتهاء ناجح.
!>         < 0:  إذا كان INFO = -i، فإن المعامل i كان له قيمة غير قانونية.
!>         > 0:  فشل الخوارزمية في حساب قيمة مفردة أثناء العمل على
!>               المصفوفة الفرعية الواقعة في الصفوف والأعمدة
!>               INFO/(N+1) حتى MOD(INFO,N+1).
!>
