توثيق الوظيفة/الإجراء الفرعي

subroutine clatmt (integer m, integer n, character dist, integer, dimension( 4 ) iseed, character sym, real, dimension( * ) d, integer mode, real cond, real dmax, integer rank, integer kl, integer ku, character pack, complex, dimension( lda, * ) a, integer lda, complex, dimension( * ) work, integer info)

CLATMT

الغرض:

!>
!>    CLATMT يولد مصفوفات عشوائية بقيم مفردة محددة
!>    (أو هرميتية بقيم ذاتية محددة)
!>    لاختبار برامج LAPACK.
!>
!>    تعمل CLATMT عن طريق تطبيق التسلسل التالي من
!>    العمليات:
!>
!>      تعيين القطر إلى D، حيث يمكن أن تكون D مدخلة أو
!!>         محسوبة وفقًا لـ MODE، COND، DMAX، و SYM
!>         كما هو موضح أدناه.
!>
!>      توليد مصفوفة ذات بنية نطاق مناسبة، بواسطة إحدى
!>         الطريقتين:
!>
!>      الطريقة أ:
!>          توليد مصفوفة M x N كثيفة عن طريق ضرب D من اليسار
!>              ومن اليمين بمصفوفات وحدوية عشوائية، ثم:
!>
!>          تقليل عرض النطاق وفقًا لـ KL و KU، باستخدام
!>              تحويلات هاوسهولدر.
!>
!>      الطريقة ب:
!>          تحويل مصفوفة عرض النطاق الصفري (أي، قطرية) إلى
!>              مصفوفة عرض نطاق-1 باستخدام دورانات جيفنز،
!>              عناصر خارج النطاق للخلف، بنفس طريقة QR؛ ثم تحويل
!>              عرض النطاق-1 إلى مصفوفة عرض نطاق-2، وهكذا. لاحظ
!>              أنه لعروض النطاقات الصغيرة بشكل معقول (مقارنة بـ M و
!>              N) يتطلب هذا تخزينًا أقل، حيث لا يتم توليد مصفوفة
!>              كثيفة. أيضًا، للمصفوفات الهرميتية أو المتماثلة،
!>              يتم توليد مثلث واحد فقط.
!>
!>      يتم اختيار الطريقة أ إذا كان عرض النطاق جزءًا كبيرًا من
!>          رتبة المصفوفة، و LDA أكبر من أو يساوي M (بحيث يمكن تخزين
!>          مصفوفة كثيفة). يتم اختيار الطريقة ب إذا كان عرض النطاق
!>          صغيرًا (< 1/2 N للمصفوفات الهرميتية أو المتماثلة، < .3 N+M
!>          للمصفوفات غير المتماثلة)، أو LDA أقل من M وليس أقل من
!>          عرض النطاق.
!>
!>      تعبئة المصفوفة إذا رغبت في ذلك. الخيارات المحددة بواسطة PACK هي:
!>         لا يوجد تعبئة
!>         تصفير النصف العلوي (إذا كانت هرميتية)
!>         تصفير النصف السفلي (إذا كانت هرميتية)
!>         تخزين النصف العلوي في عمود (إذا كانت هرميتية أو
!>               مثلثة عليا)
!>         تخزين النصف السفلي في عمود (إذا كانت هرميتية أو
!>               مثلثة سفلى)
!>         تخزين المثلث السفلي بتنسيق نطاق (إذا كانت هرميتية أو
!>               مثلثة سفلى)
!>         تخزين المثلث العلوي بتنسيق نطاق (إذا كانت هرميتية أو
!>               مثلثة عليا)
!>         تخزين المصفوفة بأكملها بتنسيق نطاق
!>      إذا تم اختيار الطريقة ب، وتم تحديد تنسيق النطاق، فسيتم
!>         توليد المصفوفة بتنسيق النطاق، لذلك لن تكون هناك حاجة لإعادة
!>         التعبئة.
!>

المعلمات

M 
!>          M هو عدد صحيح (INTEGER)
!>           عدد صفوف A. لم يتم تعديله.
!>

N

!>          N هو عدد صحيح (INTEGER)
!>           عدد أعمدة A. يجب أن يساوي N قيمة M إذا كانت المصفوفة
!>           متماثلة أو هرميتية (أي، إذا لم تكن SYM 'N')
!>           لم يتم تعديله.
!>

DIST

!>          DIST هو حرف مفرد (CHARACTER*1)
!>           عند الإدخال، تحدد DIST نوع التوزيع الذي سيتم استخدامه
!>           لتوليد القيم الذاتية/المفردة العشوائية.
!>           'U' => موحد ( 0, 1 )  ( 'U' لـ uniform )
!>           'S' => موحد ( -1, 1 ) ( 'S' لـ symmetric )
!>           'N' => طبيعي ( 0, 1 )   ( 'N' لـ normal )
!>           لم يتم تعديله.
!>

ISEED

!>          ISEED هو مصفوفة أعداد صحيحة (INTEGER array)، البعد ( 4 )
!>           عند الإدخال، تحدد ISEED بذرة مولد الأرقام العشوائية.
!>           يجب أن تكون القيم بين 0 و 4095 شاملة، ويجب أن يكون ISEED(4)
!>           فرديًا. يستخدم مولد الأرقام العشوائية تسلسلًا خطيًا مطابقًا
!>           محدودًا بأعداد صحيحة صغيرة، وبالتالي يجب أن ينتج أرقامًا
!>           عشوائية مستقلة عن الجهاز. تتغير قيم ISEED عند الخروج،
!>           ويمكن استخدامها في الاستدعاء التالي لـ CLATMT
!>           لمواصلة نفس تسلسل الأرقام العشوائية.
!>           تم تعديله عند الخروج.
!>

SYM

!>          SYM هو حرف مفرد (CHARACTER*1)
!>           إذا كانت SYM='H'، فإن المصفوفة المولدة تكون هرميتية، مع
!>             قيم ذاتية محددة بواسطة D، COND، MODE، و DMAX؛ يمكن أن
!>             تكون موجبة، سالبة، أو صفرية.
!>           إذا كانت SYM='P'، فإن المصفوفة المولدة تكون هرميتية، مع
!>             قيم ذاتية (= قيم مفردة) محددة بواسطة D، COND،
!>             MODE، و DMAX؛ لن تكون سالبة.
!>           إذا كانت SYM='N'، فإن المصفوفة المولدة تكون غير متماثلة، مع
!>             قيم مفردة محددة بواسطة D، COND، MODE، و DMAX؛
!>             لن تكون سالبة.
!>           إذا كانت SYM='S'، فإن المصفوفة المولدة تكون متماثلة (مركبة)،
!>             مع قيم مفردة محددة بواسطة D، COND، MODE، و
!>             DMAX؛ لن تكون سالبة.
!>           لم يتم تعديله.
!>

D

!>          D هي مصفوفة حقيقية (REAL array)، البعد ( MIN( M, N ) )
!>           تستخدم هذه المصفوفة لتحديد القيم المفردة أو
!>           الذاتية لـ A (انظر SYM، أعلاه.) إذا كان MODE=0، فسيتم
!>           افتراض أن D تحتوي على القيم الذاتية/المفردة، وإلا
!>           سيتم حسابها وفقًا لـ MODE، COND، و DMAX،
!>           ووضعها في D.
!>           تم تعديله إذا كان MODE غير صفري.
!>

MODE

!>          MODE هو عدد صحيح (INTEGER)
!>           عند الإدخال، يصف هذا كيف سيتم تحديد القيم الذاتية/المفردة:
!>           MODE = 0 يعني استخدام D كمدخل
!>           MODE = 1 يعين D(1)=1 و D(2:RANK)=1.0/COND
!>           MODE = 2 يعين D(1:RANK-1)=1 و D(RANK)=1.0/COND
!>           MODE = 3 يعين D(I)=COND**(-(I-1)/(RANK-1))
!>           MODE = 4 يعين D(i)=1 - (i-1)/(N-1)*(1 - 1/COND)
!>           MODE = 5 يعين D لأرقام عشوائية في النطاق
!>                    ( 1/COND , 1 ) بحيث تكون لوغاريتماتها
!>                    موزعة بشكل موحد.
!>           MODE = 6 يعين D لأرقام عشوائية من نفس التوزيع
!>                    مثل بقية المصفوفة.
!>           MODE < 0 له نفس المعنى مثل ABS(MODE)، باستثناء أن
!>              ترتيب عناصر D يتم عكسه.
!>           وبالتالي، إذا كان MODE موجبًا، فإن D تحتوي على عناصر تتراوح من
!>              1 إلى 1/COND، وإذا كان سالبًا، من 1/COND إلى 1.
!>           إذا كانت SYM='H'، وMODE ليس 0، 6، أو -6، فسيتم
!>              ضرب عناصر D أيضًا بإشارة عشوائية (أي، +1 أو -1).
!>           لم يتم تعديله.
!>

COND

!>          COND هو حقيقي (REAL)
!>           عند الإدخال، يستخدم كما هو موضح تحت MODE أعلاه.
!>           إذا تم استخدامه، يجب أن يكون >= 1. لم يتم تعديله.
!>

DMAX

!>          DMAX هو حقيقي (REAL)
!>           إذا لم يكن MODE هو -6، 0، أو 6، فسيتم ضرب محتويات D، كما
!>           تم حسابها وفقًا لـ MODE و COND، بواسطة
!>           DMAX / max(abs(D(i))); وبالتالي، فإن أقصى قيمة ذاتية أو
!>           مفردة مطلقة (وهي المعيار) ستكون abs(DMAX).
!>           لاحظ أنه لا يلزم أن يكون DMAX موجبًا: إذا كان DMAX سالبًا
!>           (أو صفرًا)، فسيتم ضرب D بعدد سالب (أو صفر).
!>           لم يتم تعديله.
!>

RANK

!>          RANK هو عدد صحيح (INTEGER)
!>           رتبة المصفوفة التي سيتم إنشاؤها للوضعيات 1، 2، 3 فقط.
!>           D( RANK+1:N ) = 0.
!>           لم يتم تعديله.
!>

KL

!>          KL هو عدد صحيح (INTEGER)
!>           يحدد هذا عرض النطاق السفلي للمصفوفة. على سبيل المثال، KL=0
!>           يحدد
!>
