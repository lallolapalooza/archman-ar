subroutine clatdf (ijob, n, z, ldz, rhs, rdsum, rdscal, ipiv, jpiv)
CLATDF يستخدم تحليل LU للمصفوفة n-by-n المحسوبة بواسطة sgetc2 ويحسب مساهمة في تقدير Dif العكسي.

توثيق الوظيفة/الروتين الفرعي

subroutine clatdf (integer ijob, integer n, complex, dimension( ldz, * ) z, integer ldz, complex, dimension( * ) rhs, real rdsum, real rdscal, integer, dimension( * ) ipiv, integer, dimension( * ) jpiv)

CLATDF يستخدم تحليل LU للمصفوفة n-by-n المحسوبة بواسطة sgetc2 ويحسب مساهمة في تقدير Dif العكسي.

الغرض:

!>
!> CLATDF يحسب المساهمة في تقدير Dif العكسي
!> عن طريق الحل لـ x في Z * x = b، حيث يتم اختيار b بحيث يكون معيار
!> x أكبر ما يمكن. يُفترض أن تحليل LU
!> لـ Z قد تم حسابه بواسطة CGETC2. عند الدخول RHS = f يحمل
!> المساهمة من الأنظمة الفرعية التي تم حلها مسبقًا، وعند العودة RHS = x.
!>
!> تحليل Z الذي تم إرجاعه بواسطة CGETC2 له الشكل
!> Z = P * L * U * Q، حيث P و Q مصفوفتان تبديل. L هو مثلثي سفلي
!> مع عناصر قطرية وحدة و U مثلثي علوي.
!>

المعلمات

IJOB 
!>          IJOB هو عدد صحيح
!>          IJOB = 2: أولاً، احسب متجهًا فارغًا تقريبيًا e
!>              لـ Z باستخدام CGECON، يتم تطبيع e والحل لـ
!>              Zx = +-e - f مع الإشارة التي تعطي أكبر قيمة لـ
!>              2-norm(x). أغلى بحوالي 5 مرات من الافتراضي.
!>          IJOB .ne. 2: استراتيجية التطلع المحلي حيث
!>              يتم اختيار جميع مدخلات الجانب الأيمن b إما كـ +1 أو
!>              -1. افتراضي.
!>

N

!>          N هو عدد صحيح
!>          عدد أعمدة المصفوفة Z.
!>

Z

!>          Z هي مصفوفة معقدة، البعد (LDZ, N)
!>          عند الدخول، جزء LU من تحليل المصفوفة n-by-n
!>          Z المحسوبة بواسطة CGETC2: Z = P * L * U * Q
!>

LDZ

!>          LDZ هو عدد صحيح
!>          البعد الرئيسي للمصفوفة Z. LDA >= max(1, N).
!>

RHS

!>          RHS هي مصفوفة معقدة، البعد (N).
!>          عند الدخول، يحتوي RHS على مساهمات من أنظمة فرعية أخرى.
!>          عند الخروج، يحتوي RHS على حل النظام الفرعي مع
!>          مدخلات وفقًا لقيمة IJOB (انظر أعلاه).
!>

RDSUM

!>          RDSUM هو رقم حقيقي
!>          عند الدخول، مجموع مربعات المساهمات المحسوبة لـ
!>          تقدير Dif قيد الحساب بواسطة CTGSYL، حيث تم استخراج
!>          معامل القياس RDSCAL (انظر أدناه).
!>          عند الخروج، مجموع المربعات المقابل المحدث بـ
!>          المساهمات من النظام الفرعي الحالي.
!>          إذا كان TRANS = 'T' فلن يتم لمس RDSUM.
!>          ملاحظة: RDSUM له معنى فقط عندما يتم استدعاء CTGSY2 بواسطة CTGSYL.
!>

RDSCAL

!>          RDSCAL هو رقم حقيقي
!>          عند الدخول، معامل القياس المستخدم لمنع تجاوز التدفق في RDSUM.
!>          عند الخروج، يتم تحديث RDSCAL فيما يتعلق بالمساهمات الحالية
!>          في RDSUM.
!>          إذا كان TRANS = 'T'، فلن يتم لمس RDSCAL.
!>          ملاحظة: RDSCAL له معنى فقط عندما يتم استدعاء CTGSY2 بواسطة
!>          CTGSYL.
!>

IPIV

!>          IPIV هي مصفوفة أعداد صحيحة، البعد (N).
!>          فهارس المحور؛ لـ 1 <= i <= N، تم تبديل الصف i للمصفوفة
!>          مع الصف IPIV(i).
!>

JPIV

!>          JPIV هي مصفوفة أعداد صحيحة، البعد (N).
!>          فهارس المحور؛ لـ 1 <= j <= N، تم تبديل العمود j للمصفوفة
!>          مع العمود JPIV(j).
!>
