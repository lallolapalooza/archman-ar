subroutine clsets (integer m, integer p, integer n, complex, dimension( lda, * ) a, complex, dimension( lda, * ) af, integer lda, complex, dimension( ldb, * ) b, complex, dimension( ldb, * ) bf, integer ldb, complex, dimension( * ) c, complex, dimension( * ) cf, complex, dimension( * ) d, complex, dimension( * ) df, complex, dimension( * ) x, complex, dimension( lwork ) work, integer lwork, real, dimension( * ) rwork, real, dimension( 2 ) result)

CLSETS

الغرض:

!>
!> CLSETS تختبر CGGLSE - وهي إجراء فرعي لحل مشكلة المربعات الصغرى المقيدة بالمعادلات الخطية (LSE).
!>

المعلمات

M 
!>          M هو عدد صحيح (INTEGER)
!>          عدد صفوف المصفوفة A.  M >= 0.
!>

P

!>          P هو عدد صحيح (INTEGER)
!>          عدد صفوف المصفوفة B.  P >= 0.
!>

N

!>          N هو عدد صحيح (INTEGER)
!>          عدد أعمدة المصفوفتين A و B.  N >= 0.
!>

A

!>          A هي مصفوفة مركبة (COMPLEX) ، البعد (LDA,N)
!>          المصفوفة M-by-N A.
!>

AF

!>          AF هي مصفوفة مركبة (COMPLEX) ، البعد (LDA,N)
!>

LDA

!>          LDA هو عدد صحيح (INTEGER)
!>          البعد الرئيسي للمصفوفات A، AF، Q و R.
!>          LDA >= max(M,N).
!>

B

!>          B هي مصفوفة مركبة (COMPLEX) ، البعد (LDB,N)
!>          المصفوفة P-by-N A.
!>

BF

!>          BF هي مصفوفة مركبة (COMPLEX) ، البعد (LDB,N)
!>

LDB

!>          LDB هو عدد صحيح (INTEGER)
!>          البعد الرئيسي للمصفوفات B، BF، V و S.
!>          LDB >= max(P,N).
!>

C

!>          C هي مصفوفة مركبة (COMPLEX) ، البعد ( M )
!>          المتجه C في مشكلة LSE.
!>

CF

!>          CF هي مصفوفة مركبة (COMPLEX) ، البعد ( M )
!>

D

!>          D هي مصفوفة مركبة (COMPLEX) ، البعد ( P )
!>          المتجه D في مشكلة LSE.
!>

DF

!>          DF هي مصفوفة مركبة (COMPLEX) ، البعد ( P )
!>

X

!>          X هي مصفوفة مركبة (COMPLEX) ، البعد ( N )
!>          متجه الحل X في مشكلة LSE.
!>

WORK

!>          WORK هي مصفوفة مركبة (COMPLEX) ، البعد (LWORK)
!>

LWORK

!>          LWORK هو عدد صحيح (INTEGER)
!>          بعد المصفوفة WORK.
!>

RWORK

!>          RWORK هي مصفوفة حقيقية (REAL) ، البعد (M)
!>

RESULT

!>          RESULT هي مصفوفة حقيقية (REAL) ، البعد (2)
!>          نسب الاختبار:
!>            RESULT(1) = norm( A*x - c )/ norm(A)*norm(X)*EPS
!>            RESULT(2) = norm( B*x - d )/ norm(B)*norm(X)*EPS
!>
