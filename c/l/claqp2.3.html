subroutine claqp2 (integer m, integer n, integer offset, complex, dimension( lda, * ) a, integer lda, integer, dimension( * ) jpvt, complex, dimension( * ) tau, real, dimension( * ) vn1, real, dimension( * ) vn2, complex, dimension( * ) work)

CLAQP2 تحسب تحليل QR مع تبديل الأعمدة للكتلة المصفوفة.

الغرض:

!>
!> CLAQP2 تحسب تحليل QR مع تبديل الأعمدة للكتلة A(OFFSET+1:M,1:N).
!> الكتلة A(1:OFFSET,1:N) يتم تبديل أعمدتها بشكل مناسب، ولكن لا يتم تحليلها.
!>

المعلمات

M 
!>          M هو عدد صحيح (INTEGER)
!>          عدد صفوف المصفوفة A. M >= 0.
!>

N

!>          N هو عدد صحيح (INTEGER)
!>          عدد أعمدة المصفوفة A. N >= 0.
!>

OFFSET

!>          OFFSET هو عدد صحيح (INTEGER)
!>          عدد صفوف المصفوفة A التي يجب تبديل أعمدتها
!>          ولكن لا يتم تحليلها. OFFSET >= 0.
!>

A

!>          A هي مصفوفة مركبة (COMPLEX)، البعد (LDA,N)
!>          عند الإدخال، المصفوفة A بحجم M-by-N.
!>          عند الخروج، الجزء العلوي المثلثي من الكتلة A(OFFSET+1:M,1:N) هو
!>          العامل المثلثي الناتج؛ العناصر في الكتلة
!>          A(OFFSET+1:M,1:N) أسفل القطر، جنبًا إلى جنب مع
!>          المصفوفة TAU، تمثل المصفوفة المتعامدة Q كناتج
!>          للانعكاسات الأولية. تم تبديل أعمدة الكتلة A(1:OFFSET,1:N)
!>          بشكل مناسب، ولكن لم يتم تحليلها.
!>

LDA

!>          LDA هو عدد صحيح (INTEGER)
!>          البعد الرائد للمصفوفة A. LDA >= max(1,M).
!>

JPVT

!>          JPVT هي مصفوفة أعداد صحيحة (INTEGER)، البعد (N)
!>          عند الإدخال، إذا كان JPVT(i) .ne. 0، فإن العمود i من A يتم تبديله
!>          إلى مقدمة A*P (عمود رائد)؛ إذا كان JPVT(i) = 0،
!>          فإن العمود i من A هو عمود حر.
!>          عند الخروج، إذا كان JPVT(i) = k، فإن العمود i من A*P
!>          كان العمود k من A.
!>

TAU

!>          TAU هي مصفوفة مركبة (COMPLEX)، البعد (min(M,N))
!>          العوامل القياسية للانعكاسات الأولية.
!>

VN1

!>          VN1 هي مصفوفة حقيقية (REAL)، البعد (N)
!>          المتجه الذي يحتوي على المعايير الجزئية للأعمدة.
!>

VN2

!>          VN2 هي مصفوفة حقيقية (REAL)، البعد (N)
!>          المتجه الذي يحتوي على المعايير الدقيقة للأعمدة.
!>

WORK

!>          WORK هي مصفوفة مركبة (COMPLEX)، البعد (N)
!>
