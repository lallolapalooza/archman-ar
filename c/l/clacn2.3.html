وثائق الوظيفة/الإجراء الفرعي

subroutine clacn2 (integer n, complex, dimension( * ) v, complex, dimension( * ) x, real est, integer kase, integer, dimension( 3 ) isave)

CLACN2 تقدر المعيار 1 لمصفوفة مربعة معقدة، باستخدام الاتصال العكسي لتقييم ضرب المصفوفة في المتجه.

الغرض:

!>
!> CLACN2 تقدر المعيار 1 لمصفوفة مربعة معقدة A.
!> يستخدم الاتصال العكسي لتقييم ضرب المصفوفة في المتجه.
!>

المعلمات

N 
!>          N هو عدد صحيح
!>         ترتيب المصفوفة. N >= 1.
!>

V

!>          V هو مصفوفة معقدة، البعد (N)
!>         عند العودة النهائية، V = A*W، حيث EST = norm(V)/norm(W)
!>         (W لا يتم إرجاعها).
!>

X

!>          X هو مصفوفة معقدة، البعد (N)
!>         عند العودة الوسطى، يجب استبدال X بـ
!>               A * X،   إذا كان KASE=1،
!>               A**H * X،  إذا كان KASE=2،
!>         حيث A**H هو تبديل A المصاحب، ويجب استدعاء CLACN2 مرة أخرى
!>         مع بقاء جميع المعلمات الأخرى دون تغيير.
!>

EST

!>          EST هو عدد حقيقي
!>         عند الإدخال مع KASE = 1 أو 2 و ISAVE(1) = 3، يجب أن يكون EST
!>         غير متغير عن الاستدعاء السابق لـ CLACN2.
!>         عند الخروج، EST هو تقدير (حد أدنى) للمعيار (A).
!>

KASE

!>          KASE هو عدد صحيح
!>         عند الاستدعاء الأولي لـ CLACN2، يجب أن يكون KASE = 0.
!>         عند العودة الوسطى، سيكون KASE = 1 أو 2، مشيرًا
!>         إلى ما إذا كان يجب استبدال X بـ A * X  أو A**H * X.
!>         عند العودة النهائية من CLACN2، سيكون KASE = 0 مرة أخرى.
!>

ISAVE

!>          ISAVE هي مصفوفة أعداد صحيحة، البعد (3)
!>         تستخدم ISAVE لحفظ المتغيرات بين استدعاءات SLACN2
!>
