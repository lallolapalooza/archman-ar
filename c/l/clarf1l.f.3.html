وثائق الوظيفة/الإجراء الفرعي

subroutine clarf1l (character side, integer m, integer n, complex, dimension( * ) v, integer incv, complex tau, complex, dimension( ldc, * ) c, integer ldc, complex, dimension( * ) work)

CLARF1L يطبق عاكس أولي معقد H على مصفوفة عامة مستطيلة معقدة C، من اليسار أو اليمين. يتم تمثيل H بالشكل

الغرض:

!>
!> CLARF1L يطبق عاكس أولي معقد H على مصفوفة C معقدة m × n،
!> من اليسار أو اليمين. يتم تمثيل H بالشكل
!>
!>       H = I - tau * v * v**H
!>
!> حيث tau هو عدد صحيح حقيقي و v هو متجه حقيقي يفترض أن v(lastv) = 1،
!> حيث lastv هو آخر عنصر غير صفري.
!>
!> إذا كان tau = 0، فإن H تؤخذ لتكون مصفوفة الوحدة.
!>
!> لتطبيق H**H (المتبادل المرافق لـ H)، قم بتمرير conjg(tau) بدلاً من
!> tau.
!>

المعلمات

SIDE 
!>          SIDE هو CHARACTER*1
!>          = 'L': تشكيل H * C
!>          = 'R': تشكيل C * H
!>

M

!>          M هو INTEGER
!>          عدد صفوف المصفوفة C.
!>

N

!>          N هو INTEGER
!>          عدد أعمدة المصفوفة C.
!>

V

!>          V هو مصفوفة COMPLEX، أبعادها
!>                     (1 + (M-1)*abs(INCV)) إذا كان SIDE = 'L'
!>                  أو (1 + (N-1)*abs(INCV)) إذا كان SIDE = 'R'
!>          المتجه v في تمثيل H. لا يتم استخدام V إذا كان
!>          TAU = 0.
!>

INCV

!>          INCV هو INTEGER
!>          الزيادة بين عناصر v. INCV > 0.
!>

TAU

!>          TAU هو COMPLEX
!>          القيمة tau في تمثيل H.
!>

C

!>          C هي مصفوفة COMPLEX، أبعادها (LDC,N)
!>          عند الدخول، المصفوفة C بأبعاد m × n.
!>          عند الخروج، يتم استبدال C بالمصفوفة H * C إذا كان SIDE = 'L'،
!>          أو C * H إذا كان SIDE = 'R'.
!>

LDC

!>          LDC هو INTEGER
!>          البعد الرائد للمصفوفة C. LDC >= max(1,M).
!>

WORK

!>          WORK هي مصفوفة COMPLEX، أبعادها
!>                         (N) إذا كان SIDE = 'L'
!>                      أو (M) إذا كان SIDE = 'R'
!>
