Class__Data__Inheritable(3pm)

الحزمة: perl-class-data-inheritable

التراخيص: GPL, PerlArtistic

Class::Data::Inheritable(3) وثائق Perl من مساهمات المستخدم Class::Data::Inheritable(3)

الاسم

Class::Data::Inheritable - بيانات فئة قابلة للتوريث، قابلة للتجاوز

ملخص

package Stuff;
use base qw(Class::Data::Inheritable);
# إعداد DataFile كبيانات فئة قابلة للتوريث.
Stuff->mk_classdata('DataFile');
# تحديد موقع ملف البيانات لهذه الفئة.
Stuff->DataFile('/etc/stuff/data');
# أو، كل ذلك في خطوة واحدة:
Stuff->mk_classdata(DataFile => '/etc/stuff/data');

الوصف

Class::Data::Inheritable مخصص لإنشاء مساعدين/معدلين لبيانات الفئة. أي، إذا كنت ترغب في تخزين شيء ما حول فئتك ككل (بدلاً من كائن فردي). يتم بعد ذلك توريث هذه البيانات بواسطة فئاتك الفرعية ويمكن تجاوزها.

على سبيل المثال:

Pere::Ubu->mk_classdata('Suitcase');

سيؤدي هذا إلى إنشاء التابع Suitcase() في الفئة Pere::Ubu.

يمكن استخدام هذا التابع الجديد للحصول على قطعة بيانات الفئة وتعيينها.

Pere::Ubu->Suitcase('Red');
$suitcase = Pere::Ubu->Suitcase;

يحدث الجزء المثير للاهتمام عندما ترث فئة من Pere::Ubu:

package Raygun;
use base qw(Pere::Ubu);

# حقيبة Raygun هي حمراء.
$suitcase = Raygun->Suitcase;

ترث Raygun بيانات فئتها Suitcase من Pere::Ubu.

يعمل توريث بيانات الفئة بشكل مشابه لتوريث التوابع. طالما أن Raygun لا "تتجاوز" بيانات فئتها الموروثة (باستخدام Suitcase() لتعيين قيمة جديدة) ستستمر في استخدام ما تم تعيينه في Pere::Ubu وتوريث التغييرات المستقبلية:

# حقائب كل من Raygun و Pere::Ubu الآن زرقاء
Pere::Ubu->Suitcase('Blue');

ومع ذلك، إذا قررت Raygun تعيين Suitcase() الخاصة بها، فقد "تجاوزت" الآن Pere::Ubu وهي بمفردها، تمامًا كما لو كانت قد تجاوزت تابعًا:

# لدى Raygun حقيبة برتقالية، بينما لدى Pere::Ubu لا تزال زرقاء.
Raygun->Suitcase('Orange');

الآن بعد أن تجاوزت Raygun Pere::Ubu، فإن التغييرات المستقبلية من قبل Pere::Ubu لم تعد تؤثر على Raygun.

# لا تزال Raygun لديها حقيبة برتقالية، لكن Pere::Ubu تستخدم Samsonite.
Pere::Ubu->Suitcase('Samsonite');

التوابع

mk_classdata

Class->mk_classdata($data_accessor_name);
Class->mk_classdata($data_accessor_name => $value);

هذا تابع فئة يستخدم للإعلان عن مساعدي بيانات فئة جدد. سيتم إنشاء مساعد جديد في الفئة باستخدام الاسم من $data_accessor_name، وربما تعيينه اختياريًا إلى القيمة المعطاة.

لتسهيل التجاوز، ينشئ mk_classdata اسمًا مستعارًا للتابع، _field_accessor(). لذلك، سيكون لـ Suitcase() اسم مستعار _Suitcase_accessor() الذي يقوم بنفس الشيء بالضبط مثل Suitcase(). هذا مفيد إذا كنت ترغب في تغيير سلوك تابع واحد مع الاستمرار في الحصول على فوائد بيانات الفئة القابلة للتوريث. على سبيل المثال.

sub Suitcase {
    my($self) = shift;
    warn "Fashion tragedy" if @_ and $_[0] eq 'Plaid';
    $self->_Suitcase_accessor(@_);
}
