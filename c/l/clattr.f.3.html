subroutine clattr (integer imat, character uplo, character trans, character diag, integer, dimension( 4 ) iseed, integer n, complex, dimension( lda, * ) a, integer lda, complex, dimension( * ) b, complex, dimension( * ) work, real, dimension( * ) rwork, integer info)

CLATTR

الغرض:

!>
!> CLATTR يولد مصفوفة اختبار مثلثية في تخزين ثنائي الأبعاد.
!> IMAT و UPLO يحددان خصائص مصفوفة الاختبار بشكل فريد،
!> والتي يتم إرجاعها في المصفوفة A.
!>

المعاملات

IMAT 
!>          IMAT هو عدد صحيح
!>          مفتاح عدد صحيح يصف أي مصفوفة سيتم إنشاؤها لهذا
!>          المسار.
!>

UPLO

!>          UPLO هو حرف*1
!>          يحدد ما إذا كانت المصفوفة A ستكون علوية أو سفلية
!>          مثلثية.
!>          = 'U':  مثلثية علوية
!>          = 'L':  مثلثية سفلية
!>

TRANS

!>          TRANS هو حرف*1
!>          يحدد ما إذا كانت المصفوفة أو منقولتها سيتم استخدامها.
!>          = 'N':  بدون منقول
!>          = 'T':  منقول
!>          = 'C':  منقول مرافق
!>

DIAG

!>          DIAG هو حرف*1
!>          يحدد ما إذا كانت المصفوفة A مثلثية وحدية أم لا.
!>          = 'N':  مثلثية غير وحدية
!>          = 'U':  مثلثية وحدية
!>

ISEED

!>          ISEED هي مصفوفة عدد صحيح، أبعادها (4)
!>          متجه البذرة لمولد الأرقام العشوائية (مستخدم في
!>          CLATMS).  يتم تعديلها عند الخروج.
!>

N

!>          N هو عدد صحيح
!>          رتبة المصفوفة التي سيتم إنشاؤها.
!>

A

!>          A هي مصفوفة معقدة، أبعادها (LDA,N)
!>          المصفوفة المثلثية A.  إذا كان UPLO = 'U'، فإن الجزء العلوي
!>          المثلثي N x N الرائد للمصفوفة A يحتوي على المصفوفة
!>          المثلثية العلوية، والجزء السفلي المثلثي الصارم لـ A
!>          لا يتم الإشارة إليه.  إذا كان UPLO = 'L'، فإن الجزء السفلي
!>          المثلثي N x N الرائد للمصفوفة A يحتوي على المصفوفة
!>          المثلثية السفلية والجزء العلوي المثلثي الصارم لـ A
!>          لا يتم الإشارة إليه.
!>

LDA

!>          LDA هو عدد صحيح
!>          البعد الرائد للمصفوفة A.  LDA >= max(1,N).
!>

B

!>          B هي مصفوفة معقدة، أبعادها (N)
!>          متجه الجانب الأيمن، إذا كان IMAT > 10.
!>

WORK

!>          WORK هي مصفوفة معقدة، أبعادها (2*N)
!>

RWORK

!>          RWORK هي مصفوفة حقيقية، أبعادها (N)
!>

INFO

!>          INFO هو عدد صحيح
!>          = 0:  اكتمال ناجح
!>          < 0:  إذا كان INFO = -i، فإن المعامل i كان له قيمة غير قانونية
!>
