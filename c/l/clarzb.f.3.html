subroutine clarzb (character side, character trans, character direct, character storev, integer m, integer n, integer k, integer l, complex, dimension( ldv, * ) v, integer ldv, complex, dimension( ldt, * ) t, integer ldt, complex, dimension( ldc, * ) c, integer ldc, complex, dimension( ldwork, * ) work, integer ldwork)

CLARZB يطبق عاكس كتلة أو محوّله المترافق على مصفوفة عامة.

الغرض:

!>
!> CLARZB يطبق عاكس كتلة عقدي H أو محوّله المترافق H**H
!> على مصفوفة C عقدية موزعة بحجم M-by-N من اليسار أو اليمين.
!>
!> حالياً، يتم دعم STOREV = 'R' و DIRECT = 'B' فقط.
!>

المعاملات

SIDE 
!>          SIDE هو CHARACTER*1
!>          = 'L': تطبيق H أو H**H من اليسار
!>          = 'R': تطبيق H أو H**H من اليمين
!>

TRANS

!>          TRANS هو CHARACTER*1
!>          = 'N': تطبيق H (بدون تحويل)
!>          = 'C': تطبيق H**H (تحويل مترافق)
!>

DIRECT

!>          DIRECT هو CHARACTER*1
!>          يحدد كيفية تكوين H من حاصل ضرب عواكس أولية
!>          = 'F': H = H(1) H(2) . . . H(k) (أمامي، غير مدعوم بعد)
!>          = 'B': H = H(k) . . . H(2) H(1) (خلفي)
!>

STOREV

!>          STOREV هو CHARACTER*1
!>          يحدد كيفية تخزين المتجهات التي تعرف العواكس الأولية:
!>          = 'C': عمودي                        (غير مدعوم بعد)
!>          = 'R': صفي
!>

M

!>          M هو INTEGER
!>          عدد صفوف المصفوفة C.
!>

N

!>          N هو INTEGER
!>          عدد أعمدة المصفوفة C.
!>

K

!>          K هو INTEGER
!>          ترتيب المصفوفة T (= عدد العواكس الأولية التي يشكل حاصل ضربها
!>          العاكس الكتلي).
!>

L

!>          L هو INTEGER
!>          عدد أعمدة المصفوفة V التي تحتوي على الجزء الهادف
!>          من عواكس هاوسهولدر.
!>          إذا كانت SIDE = 'L'، فإن M >= L >= 0، إذا كانت SIDE = 'R'، فإن N >= L >= 0.
!>

V

!>          V هي مصفوفة COMPLEX، حجمها (LDV,NV).
!>          إذا كانت STOREV = 'C'، فإن NV = K؛ إذا كانت STOREV = 'R'، فإن NV = L.
!>

LDV

!>          LDV هو INTEGER
!>          البعد الرئيسي للمصفوفة V.
!>          إذا كانت STOREV = 'C'، فإن LDV >= L؛ إذا كانت STOREV = 'R'، فإن LDV >= K.
!>

T

!>          T هي مصفوفة COMPLEX، حجمها (LDT,K)
!>          المصفوفة المربعة T في تمثيل العاكس الكتلي.
!>

LDT

!>          LDT هو INTEGER
!>          البعد الرئيسي للمصفوفة T. LDT >= K.
!>

C

!>          C هي مصفوفة COMPLEX، حجمها (LDC,N)
!>          عند الإدخال، المصفوفة C بحجم M-by-N.
!>          عند الخروج، يتم استبدال C بـ H*C أو H**H*C أو C*H أو C*H**H.
!>

LDC

!>          LDC هو INTEGER
!>          البعد الرئيسي للمصفوفة C. LDC >= max(1,M).
!>

WORK

!>          WORK هي مصفوفة COMPLEX، حجمها (LDWORK,K)
!>

LDWORK

!>          LDWORK هو INTEGER
!>          البعد الرئيسي للمصفوفة WORK.
!>          إذا كانت SIDE = 'L'، فإن LDWORK >= max(1,N)؛
!>          إذا كانت SIDE = 'R'، فإن LDWORK >= max(1,M).
!>
