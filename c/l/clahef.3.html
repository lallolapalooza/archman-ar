subroutine clahef (uplo, n, nb, kb, a, lda, ipiv, w, ldw, info)
CLAHEF تحسب تحليل جزئي لمصفوفة هيرميتية غير محددة مركبة باستخدام طريقة التقليب القطري لبنش-كوفمان (خوارزمية محظورة، تستدعي BLAS المستوى 3).

توثيق الوظيفة/الإجراء الفرعي

subroutine clahef (character uplo, integer n, integer nb, integer kb, complex, dimension( lda, * ) a, integer lda, integer, dimension( * ) ipiv, complex, dimension( ldw, * ) w, integer ldw, integer info)

CLAHEF تحسب تحليل جزئي لمصفوفة هيرميتية غير محددة مركبة باستخدام طريقة التقليب القطري لبنش-كوفمان (خوارزمية محظورة، تستدعي BLAS المستوى 3).

الغرض:

!>
!> CLAHEF تحسب تحليل جزئي لمصفوفة هيرميتية مركبة A باستخدام طريقة التقليب القطري لبنش-كوفمان.
!> التحليل الجزئي له الشكل:
!>
!> A  =  ( I  U12 ) ( A11  0  ) (  I      0     )  إذا كان UPLO = 'U'، أو:
!>       ( 0  U22 ) (  0   D  ) ( U12**H U22**H )
!>
!> A  =  ( L11  0 ) (  D   0  ) ( L11**H L21**H )  إذا كان UPLO = 'L'
!>       ( L21  I ) (  0  A22 ) (  0      I     )
!>
!> حيث رتبة D لا تتجاوز NB. الرتبة الفعلية تُرجع في المتغير KB، وهي إما NB أو NB-1، أو N إذا كان N <= NB.
!> لاحظ أن U**H يمثل التبديل المترافق لـ U.
!>
!> CLAHEF هي روتين مساعد يتم استدعاؤه بواسطة CHETRF. تستخدم رموزًا محظورة
!> (تستدعي BLAS المستوى 3) لتحديث المصفوفة الفرعية A11 (إذا كان UPLO = 'U') أو
!> A22 (إذا كان UPLO = 'L').
!>

المعلمات

UPLO 
!>          UPLO هو CHARACTER*1
!>          يحدد ما إذا كان الجزء العلوي أو السفلي المثلثي للمصفوفة الهيرميتية A مخزنًا:
!>          = 'U':  مثلثي علوي
!>          = 'L':  مثلثي سفلي
!>

N

!>          N هو INTEGER
!>          رتبة المصفوفة A.  N >= 0.
!>

NB

!>          NB هو INTEGER
!>          الحد الأقصى لعدد الأعمدة من المصفوفة A التي يجب تحليلها.
!>          يجب أن يكون NB على الأقل 2 للسماح بكتل التقليب 2x2.
!>

KB

!>          KB هو INTEGER
!>          عدد الأعمدة من A التي تم تحليلها فعليًا.
!>          KB هو إما NB-1 أو NB، أو N إذا كان N <= NB.
!>

A

!>          A هو مصفوفة COMPLEX، البعد (LDA,N)
!>          عند الإدخال، المصفوفة الهيرميتية A.  إذا كان UPLO = 'U'، فإن الجزء العلوي
!>          المثلثي العلوي من A بحجم n*n يحتوي على الجزء المثلثي العلوي
!>          للمصفوفة A، ولا يتم الرجوع إلى الجزء المثلثي السفلي الصارم لـ A.  إذا كان UPLO = 'L'، فإن الجزء السفلي
!>          المثلثي السفلي من A بحجم n*n يحتوي على الجزء المثلثي السفلي
!>          للمصفوفة A، ولا يتم الرجوع إلى الجزء المثلثي العلوي الصارم لـ A.
!>          عند الخروج، تحتوي A على تفاصيل التحليل الجزئي.
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرائد للمصفوفة A.  LDA >= max(1,N).
!>

IPIV

!>          IPIV هو مصفوفة INTEGER، البعد (N)
!>          تفاصيل التبادلات وهيكل الكتلة لـ D.
!>
!>          إذا كان UPLO = 'U':
!>             يتم تعيين آخر KB عنصرًا فقط من IPIV.
!>
!>             إذا كان IPIV(k) > 0، فسيتم تبديل الصفوف والأعمدة k و IPIV(k)
!>             وكان D(k,k) كتلة قطرية 1x1.
!>
!>             إذا كان IPIV(k) = IPIV(k-1) < 0، فسيتم تبديل الصفوف والأعمدة
!>             k-1 و -IPIV(k) وكانت D(k-1:k,k-1:k)
!>             كتلة قطرية 2x2.
!>
!>          إذا كان UPLO = 'L':
!>             يتم تعيين أول KB عنصرًا فقط من IPIV.
!>
!>             إذا كان IPIV(k) > 0، فسيتم تبديل الصفوف والأعمدة k و IPIV(k)
!>             وكان D(k,k) كتلة قطرية 1x1.
!>
!>             إذا كان IPIV(k) = IPIV(k+1) < 0، فسيتم تبديل الصفوف والأعمدة
!>             k+1 و -IPIV(k) وكانت D(k:k+1,k:k+1)
!>             كتلة قطرية 2x2.
!>

W

!>          W هي مصفوفة COMPLEX، البعد (LDW,NB)
!>

LDW

!>          LDW هو INTEGER
!>          البعد الرائد للمصفوفة W.  LDW >= max(1,N).
!>

INFO

!>          INFO هو INTEGER
!>          = 0: خروج ناجح
!>          > 0: إذا كان INFO = k، فإن D(k,k) يساوي صفرًا تمامًا.  تم
!>               إكمال التحليل، ولكن المصفوفة القطرية D
!>               هي بشكل خاص مفردة.
!>
