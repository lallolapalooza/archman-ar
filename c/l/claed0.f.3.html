subroutine claed0 (qsiz, n, d, e, q, ldq, qstore, ldqs, rwork, iwork, info)
CLAED0 يستخدم بواسطة CSTEDC. يحسب جميع القيم الذاتية والمتجهات الذاتية المقابلة لمصفوفة ثلاثية قطرية متناظرة غير مختزلة باستخدام طريقة التفرع والالتقاط.

توثيق الوظيفة/الروتين الفرعي

subroutine claed0 (integer qsiz, integer n, real, dimension( * ) d, real, dimension( * ) e, complex, dimension( ldq, * ) q, integer ldq, complex, dimension( ldqs, * ) qstore, integer ldqs, real, dimension( * ) rwork, integer, dimension( * ) iwork, integer info)

CLAED0 يستخدم بواسطة CSTEDC. يحسب جميع القيم الذاتية والمتجهات الذاتية المقابلة لمصفوفة ثلاثية قطرية متناظرة غير مختزلة باستخدام طريقة التفرع والالتقاط.

الغرض:

!>
!> باستخدام طريقة التفرع والالتقاط، يحسب CLAED0 جميع القيم الذاتية
!> لمصفوفة ثلاثية قطرية متناظرة وهي عبارة عن كتلة قطرية واحدة
!> من تلك الناتجة عن اختزال مصفوفة هرميتية كثيفة أو نطاقية
!> والمتجهات الذاتية المقابلة للمصفوفة الكثيفة أو النطاقية.
!>

المعلمات

QSIZ 
!>          QSIZ هو عدد صحيح
!>         بعد المصفوفة الوحدوية المستخدمة لاختزال
!>         المصفوفة الكاملة إلى مصفوفة ثلاثية قطرية.  QSIZ >= N إذا كان ICOMPQ = 1.
!>

N

!>          N هو عدد صحيح
!>         بعد المصفوفة الثلاثية القطرية المتناظرة.  N >= 0.
!>

D

!>          D هي مصفوفة REAL، بعد (N)
!>         عند الدخول، العناصر القطرية للمصفوفة الثلاثية القطرية.
!>         عند الخروج، القيم الذاتية بترتيب تصاعدي.
!>

E

!>          E هي مصفوفة REAL، بعد (N-1)
!>         عند الدخول، العناصر خارج القطر للمصفوفة الثلاثية القطرية.
!>         عند الخروج، تم تدمير E.
!>

Q

!>          Q هي مصفوفة COMPLEX، بعد (LDQ,N)
!>         عند الدخول، يجب أن تحتوي Q على مصفوفة QSIZ × N
!>         التي أعمدتها متعامدة وحدوية. إنها جزء من المصفوفة الوحدوية
!>         التي تختزل المصفوفة الهرميتية الكثيفة الكاملة إلى مصفوفة
!>         ثلاثية قطرية متناظرة (قابلة للاختزال).
!>

LDQ

!>          LDQ هو عدد صحيح
!>         البعد الرائد للمصفوفة Q.  LDQ >= max(1,N).
!>

IWORK

!>          IWORK هي مصفوفة INTEGER،
!>         يجب أن يكون بعد IWORK على الأقل
!>                      6 + 6*N + 5*N*lg N
!>                      ( lg( N ) = أصغر عدد صحيح k
!>                                  بحيث 2^k >= N )
!>

RWORK

!>          RWORK هي مصفوفة REAL،
!>                               بعد (1 + 3*N + 2*N*lg N + 3*N**2)
!>                        ( lg( N ) = أصغر عدد صحيح k
!>                                    بحيث 2^k >= N )
!>

QSTORE

!>          QSTORE هي مصفوفة COMPLEX، بعد (LDQS, N)
!>         تستخدم لتخزين أجزاء من
!>         مصفوفة المتجهات الذاتية عندما تحدث عمليات ضرب المصفوفة
!>         التي يتم تحديثها.
!>

LDQS

!>          LDQS هو عدد صحيح
!>         البعد الرائد للمصفوفة QSTORE.
!>         LDQS >= max(1,N).
!>

INFO

!>          INFO هو عدد صحيح
!>          = 0:  خروج ناجح.
!>          < 0:  إذا كان INFO = -i، فإن المعلمة i كانت لها قيمة غير قانونية.
!>          > 0:  فشل الخوارزمية في حساب قيمة ذاتية أثناء
!>                العمل على المصفوفة الفرعية الواقعة في الصفوف والأعمدة
!>                INFO/(N+1) حتى mod(INFO,N+1).
!>
