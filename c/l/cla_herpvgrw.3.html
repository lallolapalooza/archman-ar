real function cla_herpvgrw (character*1 uplo, integer n, integer info, complex, dimension( lda, * ) a, integer lda, complex, dimension( ldaf, * ) af, integer ldaf, integer, dimension( * ) ipiv, real, dimension( * ) work)

CLA_HERPVGRW

الغرض:

!>
!>
!> CLA_HERPVGRW يحسب عامل نمو المحور المقلوب
!> norm(A)/norm(U). يتم استخدام المعيار . إذا كان هذا
!> أقل بكثير من 1، فقد يكون استقرار تحليل LU للمصفوفة (الموزونة)
!> A ضعيفًا. هذا يعني أيضًا أن الحل X، والأعداد الشرطية المقدرة، وحدود الخطأ قد تكون
!> غير موثوقة.
!>

المعلمات

UPLO 
!>          UPLO is CHARACTER*1
!>       = 'U':  يتم تخزين المثلث العلوي لـ A؛
!>       = 'L':  يتم تخزين المثلث السفلي لـ A.
!>

N

!>          N is INTEGER
!>     عدد المعادلات الخطية، أي ترتيب
!>     المصفوفة A. N >= 0.
!>

INFO

!>          INFO is INTEGER
!>     القيمة INFO التي تم إرجاعها من SSYTRF، أي المحور في
!>     العمود INFO هو 0 تمامًا.
!>

A

!>          A is COMPLEX array, dimension (LDA,N)
!>     عند الإدخال، المصفوفة A بحجم N*N.
!>

LDA

!>          LDA is INTEGER
!>     البعد الرائد للمصفوفة A. LDA >= max(1,N).
!>

AF

!>          AF is COMPLEX array, dimension (LDAF,N)
!>     المصفوفة القطرية الفرعية D والمعاملات المستخدمة
!>     للحصول على العامل U أو L كما تم حسابه بواسطة CHETRF.
!>

LDAF

!>          LDAF is INTEGER
!>     البعد الرائد للمصفوفة AF. LDAF >= max(1,N).
!>

IPIV

!>          IPIV is INTEGER array, dimension (N)
!>     تفاصيل التبادلات والبنية الفرعية لـ D
!>     كما تم تحديده بواسطة CHETRF.
!>

WORK

!>          WORK is REAL array, dimension (2*N)
!>
