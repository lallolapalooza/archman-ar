subroutine claunhr_col_getrfnp (integer m, integer n, complex, dimension( lda, * ) a, integer lda, complex, dimension( * ) d, integer info)

CLAUNHR_COL_GETRFNP

الغرض:

!>
!> يقوم CLAUNHR_COL_GETRFNP بحساب تحليل LU المعدل بدون
!> تبديل صفوف لمصفوفة معقدة عامة M-by-N وهي A. التحليل له الشكل:
!>
!>     A - S = L * U,
!>
!> حيث:
!>    S هي مصفوفة إشارة قطرية M-by-N مع القطر D، بحيث
!>    D(i) = S(i,i)، 1 <= i <= min(M,N). يتم بناء القطر D
!>    كـ D(i)=-SIGN(A(i,i))، حيث A(i,i) هي القيمة بعد إجراء
!>    خطوات الاستبعاد الغاوسي i-1. هذا يعني أن العنصر القطري في
!>    كل خطوة من خطوات الاستبعاد الغاوسي
!>    يكون على الأقل واحدًا في القيمة المطلقة (بحيث لا يكون القسمة على
!>    صفر ممكنة أثناء القسمة على العنصر القطري)؛
!>
!>    L هي مصفوفة مثلثة سفلية M-by-N مع عناصر قطرية وحدوية
!>    (شبه منحرفة سفلية إذا كان M > N)؛
!>
!>    و U هي مصفوفة مثلثة علوية M-by-N
!>    (شبه منحرفة علوية إذا كان M < N).
!>
!> هذا الروتين هو روتين مساعد يستخدم في روتين إعادة بناء هاوسهولدر
!> CUNHR_COL. في CUNHR_COL، يتم تطبيق هذا الروتين على مصفوفة
!> A ذات أعمدة متعامدة M-by-N، حيث يتم تحديد كل عنصر
!> بأنه أقل من أو يساوي واحدًا في القيمة المطلقة. مع اختيار
!> المصفوفة S المذكورة أعلاه، يمكن إظهار أن العنصر القطري في كل
!> خطوة من خطوات الاستبعاد الغاوسي هو الأكبر (في القيمة المطلقة) في
!> العمود على أو أسفل القطر، بحيث لا يلزم تبديل الصفوف
!> لتحقيق استقرار عددي [1].
!>
!> لمزيد من التفاصيل حول خوارزمية إعادة بناء هاوسهولدر،
!> بما في ذلك تحليل LU المعدل، انظر [1].
!>
!> هذه هي النسخة اليمينية المجمعة من الخوارزمية،
!> التي تستدعي BLAS المستوى 3 لتحديث المصفوفة الفرعية. لتحليل مجموعة،
!> يستدعي هذا الروتين الروتين المتكرر CLAUNHR_COL_GETRFNP2.
!>
!> [1] ,
!>     G. Ballard, J. Demmel, L. Grigori, M. Jacquelin, H.D. Nguyen,
!>     E. Solomonik, J. Parallel Distrib. Comput.,
!>     vol. 85, pp. 3-31, 2015.
!>

المعلمات

M 
!>          M هو عدد صحيح (INTEGER)
!>          عدد الصفوف للمصفوفة A.  M >= 0.
!>

N

!>          N هو عدد صحيح (INTEGER)
!>          عدد الأعمدة للمصفوفة A.  N >= 0.
!>

A

!>          A هي مصفوفة معقدة (COMPLEX array)، أبعادها (LDA,N)
!>          عند الإدخال، المصفوفة M-by-N المطلوب تحليلها.
!>          عند الخروج، عوامل L و U من التحليل
!>          A-S=L*U؛ عناصر الوحدة القطرية لـ L غير مخزنة.
!>

LDA

!>          LDA هو عدد صحيح (INTEGER)
!>          البعد الرئيسي للمصفوفة A.  LDA >= max(1,M).
!>

D

!>          D هي مصفوفة معقدة (COMPLEX array)، أبعادها min(M,N)
!>          العناصر القطرية لمصفوفة الإشارة القطرية M-by-N S،
!>          D(i) = S(i,i)، حيث 1 <= i <= min(M,N). يمكن أن تكون العناصر
!>          فقط ( +1.0, 0.0 ) أو (-1.0, 0.0 ).
!>

INFO

!>          INFO هو عدد صحيح (INTEGER)
!>          = 0:  خروج ناجح
!>          < 0:  إذا كان INFO = -i، فإن الوسيط i كان له قيمة غير قانونية
!>
