الوظائف/الروتينات الفرعية


real function cla_hercond_c (uplo, n, a, lda, af, ldaf, ipiv, c, capply, info, work, rwork)
CLA_HERCOND_C تحسب عدد شرط المعيار اللانهائي للمصفوفة op(A)*inv(diag(c)) للمصفوفات الهرمية غير المحددة.

توثيق الوظيفة/الروتين الفرعي

real function cla_hercond_c (character uplo, integer n, complex, dimension( lda, * ) a, integer lda, complex, dimension( ldaf, * ) af, integer ldaf, integer, dimension( * ) ipiv, real, dimension ( * ) c, logical capply, integer info, complex, dimension( * ) work, real, dimension( * ) rwork)

CLA_HERCOND_C تحسب عدد شرط المعيار اللانهائي للمصفوفة op(A)*inv(diag(c)) للمصفوفات الهرمية غير المحددة.

الغرض:

!>
!>    CLA_HERCOND_C تحسب عدد شرط المعيار اللانهائي لـ
!>    op(A) * inv(diag(C)) حيث C هو متجه حقيقي.
!>

المعاملات

UPLO 
!>          UPLO هو CHARACTER*1
!>       = 'U':  المثلث العلوي لـ A مخزن؛
!>       = 'L':  المثلث السفلي لـ A مخزن.
!>

N

!>          N هو INTEGER
!>     عدد المعادلات الخطية، أي ترتيب
!>     المصفوفة A.  N >= 0.
!>

A

!>          A هو مصفوفة COMPLEX، البعد (LDA,N)
!>     عند الدخول، المصفوفة A ذات البعد N × N
!>

LDA

!>          LDA هو INTEGER
!>     البعد الرئيسي للمصفوفة A.  LDA >= max(1,N).
!>

AF

!>          AF هو مصفوفة COMPLEX، البعد (LDAF,N)
!>     المصفوفة القطرية الكتلية D والمعاملات المستخدمة
!>     للحصول على العامل U أو L كما تم حسابه بواسطة CHETRF.
!>

LDAF

!>          LDAF هو INTEGER
!>     البعد الرئيسي للمصفوفة AF.  LDAF >= max(1,N).
!>

IPIV

!>          IPIV هو مصفوفة INTEGER، البعد (N)
!>     تفاصيل التبادلات والهيكل الكتلي لـ D
!>     كما تم تحديده بواسطة CHETRF.
!>

C

!>          C هو مصفوفة REAL، البعد (N)
!>     المتجه C في الصيغة op(A) * inv(diag(C)).
!>

CAPPLY

!>          CAPPLY هو LOGICAL
!>     إذا كان .TRUE.، فقم بالوصول إلى المتجه C في الصيغة أعلاه.
!>

INFO

!>          INFO هو INTEGER
!>       = 0:  خروج ناجح.
!>     i > 0:  المعامل i غير صالح.
!>

WORK

!>          WORK هو مصفوفة COMPLEX، البعد (2*N).
!>     مساحة عمل.
!>

RWORK

!>          RWORK هو مصفوفة REAL، البعد (N).
!>     مساحة عمل.
!>
