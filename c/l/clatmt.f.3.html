subroutine clatmt (integer m, integer n, character dist, integer, dimension( 4 ) iseed, character sym, real, dimension( * ) d, integer mode, real cond, real dmax, integer rank, integer kl, integer ku, character pack, complex, dimension( lda, * ) a, integer lda, complex, dimension( * ) work, integer info)

CLATMT

الغرض:

!>
!>    CLATMT يولد مصفوفات عشوائية بقيم مفردة محددة
!>    (أو هيرميتية بقيم ذاتية محددة)
!>    لاختبار برامج LAPACK.
!>
!>    تعمل CLATMT عن طريق تطبيق التسلسل التالي من
!>    العمليات:
!>
!>      تعيين القطر إلى D، حيث يمكن أن تكون D مدخلة أو
!!>         محسوبة وفقًا لـ MODE، COND، DMAX، و SYM
!>         كما هو موضح أدناه.
!>
!>      توليد مصفوفة بالبنية النطاقية المناسبة، عن طريق إحدى
!>         الطريقتين:
!>
!>      الطريقة أ:
!>          توليد مصفوفة كثيفة M × N عن طريق ضرب D على اليسار
!>              وعلى اليمين بمصفوفات وحدوية عشوائية، ثم:
!>
!>          تقليل النطاق الترددي وفقًا لـ KL و KU، باستخدام
!>              تحويلات هاوسهولدر.
!>
!>      الطريقة ب:
!>          تحويل المصفوفة ذات النطاق الصفري (أي القطرية) إلى
!>              مصفوفة نطاق واحد باستخدام دورانات جيفنز،
!>              العناصر خارج النطاق مرة أخرى، إلى حد كبير كما في QR؛ ثم تحويل
!>              النطاق الواحد إلى نطاق اثنين، وهكذا. لاحظ
!>              أنه بالنسبة للنطاقات الترددية الصغيرة بشكل معقول (مقارنة بـ M و
!>              N) يتطلب هذا مساحة تخزين أقل، حيث لا يتم توليد مصفوفة كثيفة.
!>              أيضًا، بالنسبة للمصفوفات الهيرميتية أو المتماثلة،
!>              يتم إنشاء مثلث واحد فقط.
!>
!>      يتم اختيار الطريقة أ إذا كان النطاق الترددي جزءًا كبيرًا من
!>          رتبة المصفوفة، و LDA على الأقل M (بحيث يمكن تخزين مصفوفة كثيفة).
!>          يتم اختيار الطريقة ب إذا كان النطاق الترددي
!>          صغيرًا (< 1/2 N للمصفوفات الهيرميتية أو المتماثلة، < .3 N+M لغير
!>          المتماثلة)، أو LDA أقل من M وليس أقل من
!>          النطاق الترددي.
!>
!>      حزم المصفوفة إذا رغبت في ذلك. الخيارات المحددة بواسطة PACK هي:
!>         لا حزم
!>         إلغاء النصف العلوي (إذا كانت هيرميتية)
!>         إلغاء النصف السفلي (إذا كانت هيرميتية)
!>         تخزين النصف العلوي بشكل عمودي (إذا كانت هيرميتية أو
!>               مثلثة عليا)
!>         تخزين النصف السفلي بشكل عمودي (إذا كانت هيرميتية أو
!>               مثلثة سفلى)
!>         تخزين المثلث السفلي بتنسيق نطاقي (إذا كانت هيرميتية أو
!>               مثلثة سفلى)
!>         تخزين المثلث العلوي بتنسيق نطاقي (إذا كانت هيرميتية أو
!>               مثلثة عليا)
!>         تخزين المصفوفة بأكملها بتنسيق نطاقي
!>      إذا تم اختيار الطريقة ب، وتم تحديد تنسيق النطاق، فسيتم
!>         إنشاء المصفوفة بتنسيق النطاق، لذلك لن تكون هناك حاجة لإعادة التعبئة.
!>

المعلمات

M 
!>          M هو عدد صحيح
!>           عدد صفوف A. لم يتم تعديله.
!>

N

!>          N هو عدد صحيح
!>           عدد أعمدة A. يجب أن يساوي N القيمة M إذا كانت المصفوفة
!>           متماثلة أو هيرميتية (أي، إذا لم تكن SYM 'N')
!>           لم يتم تعديله.
!>

DIST

!>          DIST هو حرف واحد (CHARACTER*1)
!>           عند الإدخال، يحدد DIST نوع التوزيع الذي سيتم استخدامه
!>           لتوليد القيم الذاتية/المفردة العشوائية.
!>           'U' => منتظم (0, 1) ('U' لمنتظم)
!>           'S' => منتظم (-1, 1) ('S' لمتماثل)
!>           'N' => طبيعي (0, 1) ('N' لطبيعي)
!>           لم يتم تعديله.
!>

ISEED

!>          ISEED هو مصفوفة أعداد صحيحة، بأبعاد (4)
!>           عند الإدخال، يحدد ISEED بذرة مولد الأرقام
!>           العشوائية. يجب أن تكون بين 0 و 4095 شاملة،
!>           ويجب أن يكون ISEED(4) فرديًا. يستخدم مولد الأرقام
!>           العشوائية تسلسلًا خطيًا تكراريًا محدودًا بأعداد
!>           صحيحة صغيرة، وبالتالي يجب أن ينتج أرقامًا عشوائية مستقلة عن الجهاز.
!>           يتم تغيير قيم ISEED عند الخروج، ويمكن استخدامها في
!>           الاستدعاء التالي لـ CLATMT
!>           للمتابعة في نفس تسلسل الأرقام العشوائية.
!>           تم التغيير عند الخروج.
!>

SYM

!>          SYM هو حرف واحد (CHARACTER*1)
!>           إذا كانت SYM='H'، فإن المصفوفة المولدة هيرميتية، مع
!>             قيم ذاتية محددة بواسطة D، COND، MODE، و DMAX؛ يمكن أن تكون
!>             موجبة أو سالبة أو صفرية.
!>           إذا كانت SYM='P'، فإن المصفوفة المولدة هيرميتية، مع
!>             قيم ذاتية (= قيم مفردة) محددة بواسطة D، COND،
!>             MODE، و DMAX؛ لن تكون سالبة.
!>           إذا كانت SYM='N'، فإن المصفوفة المولدة غير متماثلة، مع
!>             قيم مفردة محددة بواسطة D، COND، MODE، و DMAX؛
!>             لن تكون سالبة.
!>           إذا كانت SYM='S'، فإن المصفوفة المولدة متماثلة (مركبة)،
!>             مع قيم مفردة محددة بواسطة D، COND، MODE، و
!>             DMAX؛ لن تكون سالبة.
!>           لم يتم تعديله.
!>

D

!>          D هي مصفوفة حقيقية، بأبعاد (MIN(M, N))
!>           تستخدم هذه المصفوفة لتحديد القيم المفردة أو
!>           القيم الذاتية لـ A (انظر SYM، أعلاه). إذا كان MODE=0،
!>           فإن D مفترضة أنها تحتوي على القيم المفردة/الذاتية، وإلا
!>           سيتم حسابها وفقًا لـ MODE، COND، و DMAX،
!>           وسيتم وضعها في D.
!>           تم التعديل إذا كان MODE غير صفري.
!>

MODE

!>          MODE هو عدد صحيح
!>           عند الإدخال، يصف هذا كيفية تحديد القيم المفردة/الذاتية:
!>           MODE = 0 يعني استخدام D كمدخل
!>           MODE = 1 يعين D(1)=1 و D(2:RANK)=1.0/COND
!>           MODE = 2 يعين D(1:RANK-1)=1 و D(RANK)=1.0/COND
!>           MODE = 3 يعين D(I)=COND**(-(I-1)/(RANK-1))
!>           MODE = 4 يعين D(i)=1 - (i-1)/(N-1)*(1 - 1/COND)
!>           MODE = 5 يعين D إلى أرقام عشوائية في النطاق
!>                    ( 1/COND , 1 ) بحيث تكون لوغاريتماتها
!>                    موزعة بشكل منتظم.
!>           MODE = 6 يعين D إلى أرقام عشوائية من نفس التوزيع
!>                    مثل بقية المصفوفة.
!>           MODE < 0 له نفس المعنى مثل ABS(MODE)، باستثناء أن
!>              ترتيب عناصر D يتم عكسه.
!>           وبالتالي، إذا كان MODE موجبًا، فإن D تحتوي على عناصر تتراوح من
!>              1 إلى 1/COND، وإذا كان سالبًا، من 1/COND إلى 1،
!!>           إذا كانت SYM='H'، وكان MODE ليس 0 أو 6 أو -6، فإن
!>              عناصر D سيتم ضربها أيضًا بإشارة عشوائية (أي، +1 أو -1).
!>           لم يتم تعديله.
!>

COND

!>          COND هو عدد حقيقي
!>           عند الإدخال، يستخدم كما هو موضح تحت MODE أعلاه.
!>           إذا تم استخدامه، يجب أن يكون >= 1. لم يتم تعديله.
!>

DMAX

!>          DMAX هو عدد حقيقي
!>           إذا كان MODE ليس -6 أو 0 أو 6، فإن محتويات D، كما
!>           تم حسابها وفقًا لـ MODE و COND، سيتم قياسها بواسطة
!>           DMAX / max(abs(D(i)))؛ وبالتالي، فإن أقصى قيمة ذاتية أو
!>           قيمة مفردة مطلقة (وهي المعيار) ستكون abs(DMAX).
!>           لاحظ أن DMAX لا يجب أن يكون موجبًا: إذا كان DMAX سالبًا
!>           (أو صفرًا)، فسيتم قياس D برقم سالب (أو صفر).
!>           لم يتم تعديله.
!>

RANK

!>          RANK هو عدد صحيح
!>           رتبة المصفوفة المراد إنشاؤها للوضع 1، 2، 3 فقط.
!>           D(RANK+1:N) = 0.
!>           لم يتم تعديله.
!>

KL

!>          KL هو عدد صحيح
!>           هذا يحدد النطاق الترددي السفلي للمصفوفة. على سبيل المثال، KL=0
