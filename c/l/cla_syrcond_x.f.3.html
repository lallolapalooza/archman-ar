real function cla_syrcond_x (uplo, n, a, lda, af, ldaf, ipiv, x, info, work, rwork)
CLA_SYRCOND_X تحسب رقم الشرط للمعيار اللانهائي للمصفوفة المتماثلة غير المحددة op(A)*diag(x).

وثائق الوظائف/الروتينات الفرعية

real function cla_syrcond_x (character uplo, integer n, complex, dimension( lda, * ) a, integer lda, complex, dimension( ldaf, * ) af, integer ldaf, integer, dimension( * ) ipiv, complex, dimension( * ) x, integer info, complex, dimension( * ) work, real, dimension( * ) rwork)

CLA_SYRCOND_X تحسب رقم الشرط للمعيار اللانهائي للمصفوفة المتماثلة غير المحددة op(A)*diag(x).

الغرض:

!>
!>    CLA_SYRCOND_X تحسب رقم الشرط للمعيار اللانهائي لـ
!>    op(A) * diag(X) حيث X متجه COMPLEX.
!>

المعلمات

UPLO 
!>          UPLO is CHARACTER*1
!>       = 'U':  يتم تخزين المثلث العلوي لـ A؛
!>       = 'L':  يتم تخزين المثلث السفلي لـ A.
!>

N

!>          N is INTEGER
!>     عدد المعادلات الخطية، أي، ترتيب المصفوفة A.  N >= 0.
!>

A

!>          A is COMPLEX array, dimension (LDA,N)
!>     عند الإدخال، المصفوفة N-by-N A.
!>

LDA

!>          LDA is INTEGER
!>     البعد الرائد للمصفوفة A.  LDA >= max(1,N).
!>

AF

!>          AF is COMPLEX array, dimension (LDAF,N)
!>     المصفوفة القطرية الجزئية D والمضاعفات المستخدمة للحصول على
!>     العامل U أو L كما تم حسابه بواسطة CSYTRF.
!>

LDAF

!>          LDAF is INTEGER
!>     البعد الرائد للمصفوفة AF.  LDAF >= max(1,N).
!>

IPIV

!>          IPIV is INTEGER array, dimension (N)
!>     تفاصيل التبادلات والبنية الجزئية لـ D
!>     كما تم تحديده بواسطة CSYTRF.
!>

X

!>          X is COMPLEX array, dimension (N)
!>     المتجه X في الصيغة op(A) * diag(X).
!>

INFO

!>          INFO is INTEGER
!>       = 0:  خروج ناجح.
!>     i > 0:  المعلمة i غير صالحة.
!>

WORK

!>          WORK is COMPLEX array, dimension (2*N).
!>     مساحة عمل.
!>

RWORK

!>          RWORK is REAL array, dimension (N).
!>     مساحة عمل.
!>
