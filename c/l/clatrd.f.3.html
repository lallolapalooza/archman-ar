subroutine clatrd (uplo, n, nb, a, lda, e, tau, w, ldw)
CLATRD يقلل أول nb صفوف وأعمدة من مصفوفة متناظرة/هرميتية A إلى شكل قطري حقيقي عن طريق تحويل تشابه موحد.

توثيق الوظيفة/الإجراء الفرعي

subroutine clatrd (character uplo, integer n, integer nb, complex, dimension( lda, * ) a, integer lda, real, dimension( * ) e, complex, dimension( * ) tau, complex, dimension( ldw, * ) w, integer ldw)

CLATRD يقلل أول nb صفوف وأعمدة من مصفوفة هرميتية A إلى شكل قطري هرميتي عن طريق تحويل تشابه موحد.

الغرض:

!>
!> CLATRD يقلل NB صفوف وأعمدة من مصفوفة هرميتية معقدة A إلى
!> شكل قطري هرميتي عن طريق تحويل تشابه موحد Q**H * A * Q، ويعيد المصفوفات V و W اللازمة لتطبيق التحويل على الجزء غير المختزل من A.
!>
!> إذا كان UPLO = 'U'، فإن CLATRD يقلل آخر NB صفوف وأعمدة من
!> مصفوفة، والتي يتم توفير مثلثها العلوي؛
!> إذا كان UPLO = 'L'، فإن CLATRD يقلل أول NB صفوف وأعمدة من
!> مصفوفة، والتي يتم توفير مثلثها السفلي.
!>
!> هذا إجراء فرعي مساعد يتم استدعاؤه بواسطة CHETRD.
!>

المعلمات

UPLO 
!>          UPLO هو CHARACTER*1
!>          يحدد ما إذا كان الجزء العلوي أو السفلي من المصفوفة الهرميتية A مخزنًا:
!>          = 'U': مثلث علوي
!>          = 'L': مثلث سفلي
!>

N

!>          N هو INTEGER
!>          رتبة المصفوفة A.
!>

NB

!>          NB هو INTEGER
!>          عدد الصفوف والأعمدة التي سيتم تقليلها.
!>

A

!>          A هو مصفوفة COMPLEX، البعد (LDA,N)
!>          عند الدخول، المصفوفة الهرميتية A. إذا كان UPLO = 'U'، فإن الجزء العلوي
!>          المثلثي العلوي n-by-n من A يحتوي على الجزء العلوي
!>          المثلثي للمصفوفة A، والجزء السفلي
!>          المثلثي من A لا يتم الإشارة إليه. إذا كان UPLO = 'L'، فإن
!>          الجزء السفلي المثلثي العلوي n-by-n من A يحتوي على الجزء السفلي
!>          المثلثي للمصفوفة A، والجزء العلوي
!>          المثلثي من A لا يتم الإشارة إليه.
!>          عند الخروج:
!>          إذا كان UPLO = 'U'، فقد تم تقليل آخر NB عمود إلى
!>            شكل قطري، مع تجاوز العناصر القطرية
!>            العناصر القطرية لـ A؛ العناصر فوق القطر
!>            مع المصفوفة TAU، تمثل المصفوفة الموحدة Q كـ
!>            ضرب للانعكاسات الأولية؛
!>          إذا كان UPLO = 'L'، فقد تم تقليل أول NB عمود إلى
!>            شكل قطري، مع تجاوز العناصر القطرية
!>            العناصر القطرية لـ A؛ العناصر تحت القطر
!>            مع المصفوفة TAU، تمثل المصفوفة الموحدة Q كـ
!>            ضرب للانعكاسات الأولية.
!>          انظر التفاصيل الإضافية.
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرائد للمصفوفة A. LDA >= max(1,N).
!>

E

!>          E هي مصفوفة REAL، البعد (N-1)
!>          إذا كان UPLO = 'U'، فإن E(n-nb:n-1) تحتوي على العناصر فوق القطرية
!>          لآخر NB عمود من المصفوفة المختزلة؛
!>          إذا كان UPLO = 'L'، فإن E(1:nb) تحتوي على العناصر تحت القطرية
!>          لأول NB عمود من المصفوفة المختزلة.
!>

TAU

!>          TAU هي مصفوفة COMPLEX، البعد (N-1)
!>          عوامل القياس للانعكاسات الأولية، مخزنة في
!>          TAU(n-nb:n-1) إذا كان UPLO = 'U'، وفي TAU(1:nb) إذا كان UPLO = 'L'.
!>          انظر التفاصيل الإضافية.
!>

W

!>          W هي مصفوفة COMPLEX، البعد (LDW,NB)
!>          المصفوفة n-by-nb W المطلوبة لتحديث الجزء غير المختزل
!>          من A.
!>

LDW

!>          LDW هو INTEGER
!>          البعد الرائد للمصفوفة W. LDW >= max(1,N).
!>

تفاصيل إضافية:

!>
!>  إذا كان UPLO = 'U'، فإن المصفوفة Q ممثلة كضرب للانعكاسات الأولية
!>
!>     Q = H(n) H(n-1) . . . H(n-nb+1).
!>
!>  كل H(i) له الشكل
!>
!>     H(i) = I - tau * v * v**H
!>
!>  حيث tau هو عامل قياس معقد، و v هو متجه معقد مع
!>  v(i:n) = 0 و v(i-1) = 1؛ v(1:i-1) مخزن عند الخروج في A(1:i-1,i)،
!>  و tau في TAU(i-1).
!>
!>  إذا كان UPLO = 'L'، فإن المصفوفة Q ممثلة كضرب للانعكاسات الأولية
!>
!>     Q = H(1) H(2) . . . H(nb).
!>
!>  كل H(i) له الشكل
!>
!>     H(i) = I - tau * v * v**H
!>
!>  حيث tau هو عامل قياس معقد، و v هو متجه معقد مع
!>  v(1:i) = 0 و v(i+1) = 1؛ v(i+1:n) مخزن عند الخروج في A(i+1:n,i)،
!>  و tau في TAU(i).
!>
!>  العناصر المتجهة v تشكل معًا المصفوفة V ذات البعد n-by-nb
!>  المطلوبة، مع W، لتطبيق التحويل على الجزء غير المختزل
!>  من المصفوفة، باستخدام تحديث rank-2k هرميتي من الشكل:
!>  A := A - V*W**H - W*V**H.
!>
!>  محتويات A عند الخروج موضحة بالأمثلة التالية
!>  مع n = 5 و nb = 2:
!>
!>  إذا كان UPLO = 'U':                       إذا كان UPLO = 'L':
!>
!>    (  a   a   a   v4  v5 )              (  d                  )
!>    (      a   a   v4  v5 )              (  1   d              )
!>    (          a   1   v5 )              (  v1  1   a          )
!>    (              d   1  )              (  v1  v2  a   a      )
!>    (                  d  )              (  v1  v2  a   a   a  )
!>
!>  حيث d يمثل عنصرًا قطريًا من المصفوفة المختزلة، a يمثل
!>  عنصرًا من المصفوفة الأصلية لم يتغير، و vi يمثل
!>  عنصرًا من المتجه الذي يحدد H(i).
!>
