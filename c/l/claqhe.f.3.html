subroutine claqhe (character uplo, integer n, complex, dimension( lda, * ) a, integer lda, real, dimension( * ) s, real scond, real amax, character equed)

CLAQHE يقوم بقياس مصفوفة هرميتية.

الغرض:

!>
!> CLAQHE يوازن مصفوفة هرميتية A باستخدام عوامل القياس
!> في المتجه S.
!>

المعلمات

UPLO 
!>          UPLO هو CHARACTER*1
!>          يحدد ما إذا كان الجزء المثلثي العلوي أو السفلي من
!>          المصفوفة الهرميتية A مخزنًا.
!>          = 'U':  مثلثي علوي
!>          = 'L':  مثلثي سفلي
!>

N

!>          N هو INTEGER
!>          رتبة المصفوفة A.  N >= 0.
!>

A

!>          A هو مصفوفة COMPLEX، البعد (LDA,N)
!>          عند الإدخال، المصفوفة الهرميتية A.  إذا كان UPLO = 'U'، الجزء
!>          المثلثي العلوي بحجم n × n للمصفوفة A يحتوي على الجزء
!>          المثلثي العلوي للمصفوفة A، ولا يتم الرجوع إلى الجزء المثلثي
!>          السفلي الصارم لـ A.  إذا كان UPLO = 'L'، الجزء المثلثي
!>          السفلي بحجم n × n للمصفوفة A يحتوي على الجزء المثلثي
!>          السفلي للمصفوفة A، ولا يتم الرجوع إلى الجزء المثلثي
!>          العلوي الصارم لـ A.
!>
!>          عند الخروج، إذا كان EQUED = 'Y'، المصفوفة المتوازنة:
!>          diag(S) * A * diag(S).
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرئيسي للمصفوفة A.  LDA >= max(N,1).
!>

S

!>          S هو مصفوفة REAL، البعد (N)
!>          عوامل القياس لـ A.
!>

SCOND

!>          SCOND هو REAL
!>          نسبة أصغر S(i) إلى أكبر S(i).
!>

AMAX

!>          AMAX هو REAL
!>          القيمة المطلقة لأكبر عنصر في المصفوفة.
!>

EQUED

!>          EQUED هو CHARACTER*1
!>          يحدد ما إذا تم إجراء القياس أم لا.
!>          = 'N':  لم يتم إجراء القياس.
!>          = 'Y':  تم إجراء القياس، أي تم استبدال A بـ
!>                  diag(S) * A * diag(S).
!>

المعلمات الداخلية:

!>  THRESH هو قيمة حدية تستخدم لتحديد ما إذا كان يجب إجراء القياس
!>  بناءً على نسبة عوامل القياس.  إذا كان SCOND < THRESH،
!>  يتم إجراء القياس.
!>
!>  LARGE و SMALL هي قيم حدية تستخدم لتحديد ما إذا كان يجب إجراء القياس
!>  بناءً على الحجم المطلق لأكبر عنصر في المصفوفة.
!>  إذا كان AMAX > LARGE أو AMAX < SMALL، يتم إجراء القياس.
!>
