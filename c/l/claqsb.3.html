subroutine claqsb (character uplo, integer n, integer kd, complex, dimension( ldab, * ) ab, integer ldab, real, dimension( * ) s, real scond, real amax, character equed)

CLAQSB تقوم بقياس مصفوفة متناظرة/هيرميتية نطاقية، باستخدام عوامل القياس في المتجه S.

الغرض:

!>
!> CLAQSB تقوم بمعادلة مصفوفة نطاقية متناظرة A باستخدام عوامل القياس
!> في المتجه S.
!>

المعاملات

UPLO 
!>          UPLO هو CHARACTER*1
!>          يحدد ما إذا كان الجزء العلوي أو السفلي المثلثي للمصفوفة المتناظرة
!>          A مخزنًا.
!>          = 'U':  مثلثي علوي
!>          = 'L':  مثلثي سفلي
!>

N

!>          N هو INTEGER
!>          ترتيب المصفوفة A.  N >= 0.
!>

KD

!>          KD هو INTEGER
!>          عدد الأقطار الفائقة للمصفوفة A إذا كان UPLO = 'U',
!>          أو عدد الأقطار الفرعية إذا كان UPLO = 'L'.  KD >= 0.
!>

AB

!>          AB هو مصفوفة COMPLEX، أبعادها (LDAB,N)
!>          عند الإدخال، الجزء العلوي أو السفلي المثلثي للمصفوفة المتناظرة
!>          النطاقية A، مخزن في الصفوف KD+1 الأولى للمصفوفة.  العمود
!>          j للمصفوفة A مخزن في العمود j للمصفوفة AB
!>          كما يلي:
!>          إذا كان UPLO = 'U', AB(kd+1+i-j,j) = A(i,j) لـ max(1,j-kd)<=i<=j;
!>          إذا كان UPLO = 'L', AB(1+i-j,j)    = A(i,j) لـ j<=i<=min(n,j+kd).
!>
!>          عند الخروج، إذا كان INFO = 0، فإن العامل المثلثي U أو L من
!>          التحليل Cholesky A = U**H *U أو A = L*L**H للمصفوفة النطاقية
!>          A، بنفس تنسيق التخزين مثل A.
!>

LDAB

!>          LDAB هو INTEGER
!>          البعد الرائد للمصفوفة AB.  LDAB >= KD+1.
!>

S

!>          S هو مصفوفة REAL، أبعادها (N)
!>          عوامل القياس لـ A.
!>

SCOND

!>          SCOND هو REAL
!>          نسبة أصغر S(i) إلى أكبر S(i).
!>

AMAX

!>          AMAX هو REAL
!>          القيمة المطلقة لأكبر عنصر في المصفوفة.
!>

EQUED

!>          EQUED هو CHARACTER*1
!>          يحدد ما إذا تمت المعادلة أم لا.
!>          = 'N':  لم تتم المعادلة.
!>          = 'Y':  تمت المعادلة، أي تم استبدال A بـ
!>                  diag(S) * A * diag(S).
!>

المعاملات الداخلية:

!>  THRESH هو قيمة حدية تستخدم لتحديد ما إذا كان يجب إجراء القياس
!>  بناءً على نسبة عوامل القياس.  إذا كان SCOND < THRESH,
!>  يتم إجراء القياس.
!>
!>  LARGE و SMALL هي قيم حدية تستخدم لتحديد ما إذا كان يجب إجراء القياس
!>  بناءً على الحجم المطلق لأكبر عنصر في المصفوفة.
!>  إذا كان AMAX > LARGE أو AMAX < SMALL, يتم إجراء القياس.
!>
