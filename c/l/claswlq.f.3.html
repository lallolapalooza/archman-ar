توثيق الوظيفة/الروتين الفرعي

subroutine claswlq (integer m, integer n, integer mb, integer nb, complex, dimension( lda, * ) a, integer lda, complex, dimension( ldt, * ) t, integer ldt, complex, dimension( * ) work, integer lwork, integer info)

CLASWLQ

الغرض:

!>
!> CLASWLQ يحسب تحليلاً طولياً نحيفاً مجمعاً لمصفوفة A المعقدة M-by-N للمصفوفة M <= N:
!>
!>    A = ( L 0 ) *  Q,
!>
!> حيث:
!>
!>    Q هي مصفوفة متعامدة N-by-N، مخزنة عند الخروج في شكل ضمني
!>    فوق القطر للمصفوفة A وعناصر المصفوفة T؛
!>    L هي مصفوفة مثلثة سفلية M-by-M مخزنة عند الخروج في
!>    عناصر على القطر وتحته في المصفوفة A.
!>    0 هي مصفوفة صفرية M-by-(N-M)، إذا كانت M < N، ولا يتم تخزينها.
!>
!>

المعلمات:

M 
!>          M هو عدد صحيح (INTEGER)
!>          عدد صفوف المصفوفة A.  M >= 0.
!>

N

!>          N هو عدد صحيح (INTEGER)
!>          عدد أعمدة المصفوفة A.  N >= M >= 0.
!>

MB

!>          MB هو عدد صحيح (INTEGER)
!>          حجم الكتلة الصفية المستخدم في QR المجمع.
!>          M >= MB >= 1
!>

NB

!>          NB هو عدد صحيح (INTEGER)
!>          حجم الكتلة العمودية المستخدم في QR المجمع.
!>          NB > 0.
!>

A

!>          A هي مصفوفة معقدة (COMPLEX)، البعد (LDA,N)
!>          عند الدخول، مصفوفة A بحجم M-by-N.
!>          عند الخروج، تحتوي العناصر على القطر وتحته
!>          للمصفوفة على المصفوفة المثلثية السفلية L بحجم N-by-N؛
!>          تمثل العناصر فوق القطر Q بواسطة صفوف
!>          V المجمعة (انظر التفاصيل الإضافية).
!>
!>

LDA

!>          LDA هو عدد صحيح (INTEGER)
!>          البعد الرئيسي للمصفوفة A.  LDA >= max(1,M).
!>

T

!>          T هي مصفوفة معقدة (COMPLEX)،
!>          البعد (LDT, N * Number_of_row_blocks)
!>          حيث Number_of_row_blocks = CEIL((N-M)/(NB-M))
!>          تخزن عاكسات الكتل المثلثية العلوية المجمعة في شكل مدمج
!>          كسلسلة من الكتل المثلثية العلوية.
!>          انظر التفاصيل الإضافية أدناه.
!>

LDT

!>          LDT هو عدد صحيح (INTEGER)
!>          البعد الرئيسي للمصفوفة T.  LDT >= MB.
!>

WORK

!>          (مساحة عمل) مصفوفة معقدة (COMPLEX)، البعد (MAX(1,LWORK))
!>          عند الخروج، إذا كان INFO = 0، فإن WORK(1) يعيد الحد الأدنى LWORK.
!>

LWORK

!>          LWORK هو عدد صحيح (INTEGER)
!>          بعد المصفوفة WORK.
!>          LWORK >= 1، إذا كان MIN(M,N) = 0، و LWORK >= MB*M، وإلا.
!>
!>          إذا كان LWORK = -1، فيفترض استعلام عن مساحة العمل؛ تقوم الروتينات الفرعية
!>          بحساب الحد الأدنى لحجم مصفوفة WORK فقط، وتعيد
!>          هذه القيمة كمدخل أول لمصفوفة WORK، ولا يتم إصدار أي رسالة خطأ
!>          متعلقة بـ LWORK بواسطة XERBLA.
!>

INFO

!>          INFO هو عدد صحيح (INTEGER)
!>          = 0:  انتهاء ناجح
!>          < 0:  إذا كان INFO = -i، فإن الوسيط i كان له قيمة غير قانونية
!>

التفاصيل الإضافية:

!> تحلل LQ الطويل والضيق (SWLQ) يقوم بالتحليل LQ من خلال سلسلة من التحويلات المتعامدة،
!> يمثل Q كمنتج لمصفوفات متعامدة أخرى
!>   Q = Q(1) * Q(2) * . . . * Q(k)
!> حيث يقوم كل Q(i) بتصفير المدخلات القطرية العلوية لكتلة من صفوف NB من A:
!>   Q(1) يصفر المدخلات القطرية العلوية للصفوف 1:NB من A
!>   Q(2) يصفر الصفوف السفلية MB-N من الصفوف [1:M,NB+1:2*NB-M] من A
!>   Q(3) يصفر الصفوف السفلية MB-N من الصفوف [1:M,2*NB-M+1:3*NB-2*M] من A
!>   . . .
!>
!> يتم حساب Q(1) بواسطة GELQT، والتي تمثل Q(1) بواسطة متجهات هاوسهولدر
!> مخزنة تحت القطر للصفوف 1:MB من A، وبواسطة كتل عاكسات مثلثة علوية،
!> مخزنة في المصفوفة T(1:LDT,1:N).
!> لمزيد من المعلومات انظر التفاصيل الإضافية في GELQT.
!>
!> Q(i) لـ i>1 يتم حسابها بواسطة TPLQT، والتي تمثل Q(i) بواسطة متجهات هاوسهولدر
!> مخزنة في الأعمدة [(i-1)*(NB-M)+M+1:i*(NB-M)+M] من A، وبواسطة كتل عاكسات مثلثية علوية،
!> مخزنة في المصفوفة T(1:LDT,(i-1)*M+1:i*M).
!> قد تستخدم Q(k) الأخيرة عددًا أقل من الصفوف.
!> لمزيد من المعلومات انظر التفاصيل الإضافية في TPQRT.
!>
!> لمزيد من التفاصيل حول الخوارزمية العامة، انظر وصف
!> TSQR المتسلسل في القسم 2.2 من [1].
!>
!> [1] "تحليلات QR و LU المتوازية والمتسلسلة المثلى للتواصل،"
!>     J. Demmel، L. Grigori، M. Hoemmen، J. Langou،
!>     SIAM J. Sci. Comput، المجلد 34، رقم 1، 2012
!>
