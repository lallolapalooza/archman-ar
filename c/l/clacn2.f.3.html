subroutine clacn2 (n, v, x, est, kase, isave)

CLACN2 تقدّر القيمة المطلقة 1 لمصفوفة مربعة، باستخدام الاتصال العكسي لتقييم نواتج الضرب بين المصفوفات والمتجهات.

توثيق الدالة/الروتين الفرعي

subroutine clacn2 (integer n, complex, dimension( * ) v, complex, dimension( * ) x, real est, integer kase, integer, dimension( 3 ) isave)

CLACN2 تقدّر القيمة المطلقة 1 لمصفوفة مربعة معقدة.

الغرض:

!>
!> CLACN2 تقدّر القيمة المطلقة 1 لمصفوفة مربعة معقدة A.
!> يتم استخدام الاتصال العكسي لتقييم نواتج الضرب بين المصفوفات والمتجهات.
!>

المعاملات

N 
!>          N هو عدد صحيح (INTEGER)
!>         رتبة المصفوفة. N >= 1.
!>

V

!>          V هي مصفوفة معقدة (COMPLEX array)، ذات بعد (N)
!>         عند العودة النهائية، V = A*W، حيث EST = norm(V)/norm(W)
!>         (W لا يتم إرجاعها).
!>

X

!>          X هي مصفوفة معقدة (COMPLEX array)، ذات بعد (N)
!>         عند العودة المرحلية، يجب استبدال X بـ
!>               A * X،   إذا كان KASE=1،
!>               A**H * X،  إذا كان KASE=2،
!>         حيث A**H هو منقول A المعقد، ويجب استدعاء CLACN2
!>         مرة أخرى مع بقاء جميع المعاملات الأخرى دون تغيير.
!>

EST

!>          EST هو قيمة حقيقية (REAL)
!>         عند الإدخال مع KASE = 1 أو 2 و ISAVE(1) = 3، يجب أن يبقى EST
!>         دون تغيير عن الاستدعاء السابق لـ CLACN2.
!>         عند الخروج، EST هو تقدير (حد أدنى) للقيمة المطلقة لـ norm(A).
!>

KASE

!>          KASE هو عدد صحيح (INTEGER)
!>         عند الاستدعاء الأولي لـ CLACN2، يجب أن يكون KASE = 0.
!>         عند العودة المرحلية، سيكون KASE = 1 أو 2، مشيرًا
!>         إلى ما إذا كان يجب استبدال X بـ A * X  أو A**H * X.
!>         عند العودة النهائية من CLACN2، سيكون KASE = 0 مرة أخرى.
!>

ISAVE

!>          ISAVE هي مصفوفة أعداد صحيحة (INTEGER array)، ذات بعد (3)
!>         تُستخدم ISAVE لحفظ المتغيرات بين استدعاءات SLACN2
!>
