subroutine claqhb (character uplo, integer n, integer kd, complex, dimension( ldab, * ) ab, integer ldab, real, dimension( * ) s, real scond, real amax, character equed)

CLAQHB يقوم بتوسيع مصفوفة هيرميتية ذات نطاق، باستخدام عوامل توسيع تم حسابها بواسطة cpbequ.

الغرض:

!>
!> CLAQHB يعمل على موازنة مصفوفة هيرميتية A باستخدام عوامل التوسيع
!> الموجودة في المتجه S.
!>

المعلمات

UPLO 
!>          UPLO هو CHARACTER*1
!>          يحدد ما إذا كان الجزء المثلثي العلوي أو السفلي من المصفوفة
!>          المتناظرة A مخزنًا.
!>          = 'U':  مثلثي علوي
!>          = 'L':  مثلثي سفلي
!>

N

!>          N هو INTEGER
!>          رتبة المصفوفة A.  N >= 0.
!>

KD

!>          KD هو INTEGER
!>          عدد الأقطار الفائقة للمصفوفة A إذا كان UPLO = 'U'،
!>          أو عدد الأقطار الفرعية إذا كان UPLO = 'L'.  KD >= 0.
!>

AB

!>          AB هو مصفوفة COMPLEX، البعد (LDAB,N)
!>          عند الدخول، الجزء العلوي أو السفلي من المصفوفة ذات النطاق
!>          المتناظر A، مخزنة في أول KD+1 صفوف من المصفوفة.  العمود
!>          j من A مخزن في العمود j من المصفوفة AB
!>          كما يلي:
!>          إذا كان UPLO = 'U'، فإن AB(kd+1+i-j,j) = A(i,j) لـ max(1,j-kd)<=i<=j؛
!>          إذا كان UPLO = 'L'، فإن AB(1+i-j,j)    = A(i,j) لـ j<=i<=min(n,j+kd).
!>
!>          عند الخروج، إذا كان INFO = 0، فإن العامل المثلثي U أو L من
!>          تحليل Cholesky A = U**H *U أو A = L*L**H للمصفوفة ذات النطاق
!>          A، بنفس تنسيق التخزين مثل A.
!>

LDAB

!>          LDAB هو INTEGER
!>          البعد الرائد للمصفوفة AB.  LDAB >= KD+1.
!>

S

!>          S هو مصفوفة REAL، البعد (N)
!>          عوامل التوسيع للمصفوفة A.
!>

SCOND

!>          SCOND هو REAL
!>          نسبة أصغر S(i) إلى أكبر S(i).
!>

AMAX

!>          AMAX هو REAL
!>          القيمة المطلقة لأكبر عنصر في المصفوفة.
!>

EQUED

!>          EQUED هو CHARACTER*1
!>          يحدد ما إذا كان قد تم إجراء الموازنة أم لا.
!>          = 'N':  لم يتم إجراء الموازنة.
!>          = 'Y':  تم إجراء الموازنة، أي تم استبدال A بـ
!>                  diag(S) * A * diag(S).
!>

المعلمات الداخلية:

!>  THRESH هو قيمة حد يستخدم لتحديد ما إذا كان يجب إجراء التوسيع
!>  بناءً على نسبة عوامل التوسيع.  إذا كان SCOND < THRESH،
!>  يتم إجراء التوسيع.
!>
!>  LARGE و SMALL هي قيم حدود تستخدم لتحديد ما إذا كان يجب إجراء التوسيع
!>  بناءً على الحجم المطلق لأكبر عنصر في المصفوفة.
!>  إذا كان AMAX > LARGE أو AMAX < SMALL، يتم إجراء التوسيع.
!>
