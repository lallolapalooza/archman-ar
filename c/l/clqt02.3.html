وثائق الوظائف/الروتينات الفرعية

subroutine clqt02 (integer m, integer n, integer k, complex, dimension( lda, * ) a, complex, dimension( lda, * ) af, complex, dimension( lda, * ) q, complex, dimension( lda, * ) l, integer lda, complex, dimension( * ) tau, complex, dimension( lwork ) work, integer lwork, real, dimension( * ) rwork, real, dimension( * ) result)

CLQT02

الغرض:

!>
!> CLQT02 تختبر CUNGLQ، التي تولد مصفوفة Q بحجم m x n مع
!> صفوف متعامدة تعرف كناتج لمُنعكسات أولية k.
!>
!> بالنظر إلى تحليل LQ للمصفوفة A بحجم m x n، تقوم CLQT02 بتوليد
!> المصفوفة المتعامدة Q المعرفة بتحليل الصفوف k الأولى للمصفوفة A؛
!> وتقارن L(1:k,1:m) مع A(1:k,1:n)*Q(1:m,1:n)'،
!> وتتحقق من أن صفوف Q متعامدة.
!>

المعاملات

M 
!>          M هو عدد صحيح (INTEGER)
!>          عدد صفوف المصفوفة Q التي سيتم توليدها. M >= 0.
!>

N

!>          N هو عدد صحيح (INTEGER)
!>          عدد أعمدة المصفوفة Q التي سيتم توليدها.
!>          N >= M >= 0.
!>

K

!>          K هو عدد صحيح (INTEGER)
!>          عدد المُنعكسات الأولية التي يُعرّف ناتجها المصفوفة Q.
!>          M >= K >= 0.
!>

A

!>          A هي مصفوفة معقدة (COMPLEX)، البعد (LDA,N)
!>          المصفوفة A بحجم m x n التي تم تحليلها بواسطة CLQT01.
!>

AF

!>          AF هي مصفوفة معقدة (COMPLEX)، البعد (LDA,N)
!>          تفاصيل تحليل LQ للمصفوفة A، كما تم إرجاعها بواسطة CGELQF.
!>          انظر CGELQF لمزيد من التفاصيل.
!>

Q

!>          Q هي مصفوفة معقدة (COMPLEX)، البعد (LDA,N)
!>

L

!>          L هي مصفوفة معقدة (COMPLEX)، البعد (LDA,M)
!>

LDA

!>          LDA هو عدد صحيح (INTEGER)
!>          البعد الرئيسي للمصفوفات A, AF, Q, و L. LDA >= N.
!>

TAU

!>          TAU هي مصفوفة معقدة (COMPLEX)، البعد (M)
!>          العوامل العددية للمُنعكسات الأولية المقابلة لتحليل LQ في AF.
!>

WORK

!>          WORK هي مصفوفة معقدة (COMPLEX)، البعد (LWORK)
!>

LWORK

!>          LWORK هو عدد صحيح (INTEGER)
!>          بعد المصفوفة WORK.
!>

RWORK

!>          RWORK هي مصفوفة حقيقية (REAL)، البعد (M)
!>

RESULT

!>          RESULT هي مصفوفة حقيقية (REAL)، البعد (2)
!>          نسب الاختبار:
!>          RESULT(1) = norm( L - A*Q' ) / ( N * norm(A) * EPS )
!>          RESULT(2) = norm( I - Q*Q' ) / ( N * EPS )
!>
