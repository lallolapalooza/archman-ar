subroutine clatm5 (integer prtype, integer m, integer n, complex, dimension( lda, * ) a, integer lda, complex, dimension( ldb, * ) b, integer ldb, complex, dimension( ldc, * ) c, integer ldc, complex, dimension( ldd, * ) d, integer ldd, complex, dimension( lde, * ) e, integer lde, complex, dimension( ldf, * ) f, integer ldf, complex, dimension( ldr, * ) r, integer ldr, complex, dimension( ldl, * ) l, integer ldl, real alpha, integer qblcka, integer qblckb)

CLATM5

الغرض:

!>
!> CLATM5 تقوم بإنشاء المصفوفات المشاركة في معادلة سيلفستر المعممة:
!>
!>     A * R - L * B = C
!>     D * R - L * E = F
!>
!> وهي تحقق أيضًا (شرط التطبيع)
!>
!>  [ I -L ] ( [ A  -C ], [ D -F ] ) [ I  R ] = ( [ A    ], [ D    ] )
!>  [    I ] ( [     B ]  [    E ] ) [    I ]   ( [    B ]  [    E ] )
!>
!>

المعلمات

PRTYPE 
!>          PRTYPE هو عدد صحيح (INTEGER)
!>           إلى نوع معين من المصفوفات المراد إنشاؤها
!>          (انظر المزيد من التفاصيل).
!>

M

!>          M هو عدد صحيح (INTEGER)
!>          يحدد رتبة A و D وعدد الصفوف في
!>          C و F و R و L.
!>

N

!>          N هو عدد صحيح (INTEGER)
!>          يحدد رتبة B و E وعدد الأعمدة في
!>          C و F و R و L.
!>

A

!>          A هو مصفوفة مركبة (COMPLEX)، أبعادها (LDA, M).
!>          عند الخروج، A بحجم M-by-M يتم تهيئتها وفقًا لـ PRTYPE.
!>

LDA

!>          LDA هو عدد صحيح (INTEGER)
!>          البعد القيادي لـ A.
!>

B

!>          B هي مصفوفة مركبة (COMPLEX)، أبعادها (LDB, N).
!>          عند الخروج، B بحجم N-by-N يتم تهيئتها وفقًا لـ PRTYPE.
!>

LDB

!>          LDB هو عدد صحيح (INTEGER)
!>          البعد القيادي لـ B.
!>

C

!>          C هي مصفوفة مركبة (COMPLEX)، أبعادها (LDC, N).
!>          عند الخروج، C بحجم M-by-N يتم تهيئتها وفقًا لـ PRTYPE.
!>

LDC

!>          LDC هو عدد صحيح (INTEGER)
!>          البعد القيادي لـ C.

D

!>          D هي مصفوفة مركبة (COMPLEX)، أبعادها (LDD, M).

LDD

!>          البعد القيادي لـ D.
!>

E

L

QBLCKB

!>          QBLCKB هو عدد صحيح (INTEGER)

تفاصيل إضافية:

!>
!>  PRTYPE = 1: A و B هما كتل جوردان، D و E هما مصفوفات الوحدة
!>
!>             A : إذا كان (i == j) فإن A(i, j) = 1.0
!>                 إذا كان (j == i + 1) فإن A(i, j) = -1.0
!>                 غير ذلك A(i, j) = 0.0,            i, j = 1...M
!>
!>             B : إذا كان (i == j) فإن B(i, j) = 1.0 - ALPHA
!>                 إذا كان (j == i + 1) فإن B(i, j) = 1.0
!>                 غير ذلك B(i, j) = 0.0,            i, j = 1...N
!>
!>             D : إذا كان (i == j) فإن D(i, j) = 1.0
!>                 غير ذلك D(i, j) = 0.0,            i, j = 1...M
!>
!>             E : إذا كان (i == j) فإن E(i, j) = 1.0
!>                 غير ذلك E(i, j) = 0.0,            i, j = 1...N
!>
!>             L = R يتم اختيارها من [-10...10]،
!>                  مما يحدد الجوانب اليمنى (C, F).
!>
!>  PRTYPE = 2 أو 3: مثلثية و/أو شبه مثلثية.
!>
!>             A : إذا كان (i <= j) فإن A(i, j) = [-1...1]
!>                 غير ذلك A(i, j) = 0.0,             i, j = 1...M
!>
!>                 إذا كان (PRTYPE = 3)
!>                    A(k + 1, k + 1) = A(k, k)
!>                    A(k + 1, k) = [-1...1]
!>                    sign(A(k, k + 1) = -(sin(A(k + 1, k))
!>                        k = 1, M - 1, QBLCKA
!>
!>             B : إذا كان (i <= j) فإن B(i, j) = [-1...1]
!>                 غير ذلك B(i, j) = 0.0,            i, j = 1...N
!>
!>                 إذا كان (PRTYPE = 3)
!>                    B(k + 1, k + 1) = B(k, k)
!>                    B(k + 1, k) = [-1...1]
!>                    sign(B(k, k + 1) = -(sign(B(k + 1, k))
!>                        k = 1, N - 1, QBLCKB
!>
!>             D : إذا كان (i <= j) فإن D(i, j) = [-1...1].
!>                 غير ذلك D(i, j) = 0.0,            i, j = 1...M
!>
!>
!>             E : إذا كان (i <= j) فإن D(i, j) = [-1...1]
!>                 غير ذلك E(i, j) = 0.0,            i, j = 1...N
!>
!>                 L, R يتم اختيارها من [-10...10]،
!>                 مما يحدد الجوانب اليمنى (C, F).
!>
!>  PRTYPE = 4 كامل
!>             A(i, j) = [-10...10]
!>             D(i, j) = [-1...1]    i,j = 1...M
!>             B(i, j) = [-10...10]
!>             E(i, j) = [-1...1]    i,j = 1...N
!>             R(i, j) = [-10...10]
!>             L(i, j) = [-1...1]    i = 1..M ,j = 1...N
!>
!>             L, R يحددان الجوانب اليمنى (C, F).
!>
!>  PRTYPE = 5 حالة خاصة شائعة و/أو قيم ذاتية قريبة.
!>
