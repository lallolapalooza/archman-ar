void clnt_pcreateerror(const char *s);
طباعة رسالة إلى الخطأ القياسي تشير إلى سبب عدم إنشاء مقبض عميل RPC. تسبق الرسالة السلسلة s ونقطتين. تستخدم عند فشل استدعاء clnt_create() أو clntraw_create() أو clnttcp_create() أو clntudp_create().
void clnt_perrno(enum clnt_stat stat);

clnt_perror(CLIENT *clnt, const char *s);
طباعة رسالة إلى الخطأ القياسي تشير إلى سبب فشل استدعاء RPC؛ clnt هو المقبض المستخدم لإجراء الاستدعاء. تسبق الرسالة السلسلة s ونقطتين. تستخدم بعد clnt_call().
char *clnt_spcreateerror(const char *s);
مثل clnt_pcreateerror()، باستثناء أنها تُرجع سلسلة بدلاً من الطباعة إلى الخطأ القياسي.
عيوب: تُرجع مؤشرًا إلى بيانات ثابتة يتم الكتابة فوقها في كل استدعاء.
char *clnt_sperrno(enum clnt_stat stat);
تأخذ نفس وسائط clnt_perrno()، ولكن بدلاً من إرسال رسالة إلى الخطأ القياسي تشير إلى سبب فشل استدعاء RPC، تُرجع مؤشرًا إلى سلسلة تحتوي على الرسالة. تنتهي السلسلة بسطر جديد.
يتم استخدام clnt_sperrno() بدلاً من clnt_perrno() إذا لم يكن لدى البرنامج خطأ قياسي (كما هو الحال في برنامج يعمل كخادم)، أو إذا كان المبرمج لا يريد أن يتم إخراج الرسالة باستخدام printf(3)، أو إذا كان سيتم استخدام تنسيق رسالة مختلف عن ذلك الذي تدعمه clnt_perrno(). ملاحظة: على عكس clnt_sperror() و clnt_spcreateerror()، تُرجع clnt_sperrno() مؤشرًا إلى بيانات ثابتة، ولكن النتيجة لن يتم الكتابة فوقها في كل استدعاء.
char *clnt_sperror(CLIENT *rpch, const char *s);

عيوب: تُرجع مؤشرًا إلى بيانات ثابتة يتم الكتابة فوقها في كل استدعاء.
CLIENT *clntraw_create(unsigned long prognum, unsigned long versnum);
تقوم هذه الروتين بإنشاء عميل RPC وهمي للبرنامج البعيد prognum، الإصدار versnum. النقل المستخدم لتمرير الرسائل إلى الخدمة هو في الواقع مخزن مؤقت داخل مساحة عنوان العملية، لذا يجب أن يكون خادم RPC المقابل موجودًا في نفس مساحة العنوان؛ انظر svcraw_create(). هذا يسمح بمحاكاة RPC واكتساب نفقات RPC، مثل أوقات الرحلة ذهابًا وإيابًا، دون أي تدخل من النواة. تُرجع هذه الروتين NULL إذا فشلت.
CLIENT *clnttcp_create(struct sockaddr_in *addr,
                unsigned long prognum, unsigned long versnum,
                int *sockp, unsigned int sendsz, unsigned int recvsz);
تقوم هذه الروتين بإنشاء عميل RPC للبرنامج البعيد prognum، الإصدار versnum؛ يستخدم العميل TCP/IP كوسيلة نقل. يقع البرنامج البعيد في عنوان الإنترنت *addr. إذا كان addr->sin_port صفرًا، فسيتم تعيينه على المنفذ الفعلي الذي يستمع إليه البرنامج البعيد (يتم استشارة خدمة portmap البعيدة للحصول على هذه المعلومات). الوسيط sockp هو مقبس؛ إذا كان RPC_ANYSOCK، فتقوم هذه الروتين بفتح مقبس جديد وتعيين sockp. نظرًا لأن RPC المستند إلى TCP يستخدم I/O المخزن مؤقتًا، يمكن للمستخدم تحديد حجم المخازن المؤقتة للإرسال والاستلام باستخدام الوسائط sendsz و recvsz؛ القيم الصفرية تختار قيمًا افتراضية مناسبة. تُرجع هذه الروتين NULL إذا فشلت.
CLIENT *clntudp_create(struct sockaddr_in *addr,
                unsigned long prognum, unsigned long versnum,
                struct timeval wait, int *sockp);
تقوم هذه الروتين بإنشاء عميل RPC للبرنامج البعيد prognum، الإصدار versnum؛ يستخدم العميل UDP/IP كوسيلة نقل. يقع البرنامج البعيد في عنوان الإنترنت addr. إذا كان addr->sin_port صفرًا، فسيتم تعيينه على المنفذ الفعلي الذي يستمع إليه البرنامج البعيد (يتم استشارة خدمة portmap البعيدة للحصول على هذه المعلومات). الوسيط sockp هو مقبس؛ إذا كان RPC_ANYSOCK، فتقوم هذه الروتين بفتح مقبس جديد وتعيين sockp. يعيد نقل UDP إرسال رسالة الاستدعاء على فترات زمنية قدرها wait حتى يتم استلام استجابة أو حتى ينتهي وقت الاستدعاء. يتم تحديد الوقت الإجمالي لانتهاء صلاحية الاستدعاء بواسطة clnt_call().
تحذير: نظرًا لأن رسائل RPC المستندة إلى UDP يمكنها حمل ما يصل إلى 8 كيلوبايتات من البيانات المشفرة فقط، لا يمكن استخدام هذا النقل للإجراءات التي تأخذ وسائط كبيرة أو تُرجع نتائج ضخمة.
CLIENT *clntudp_bufcreate(struct sockaddr_in *addr,

void get_myaddress(struct sockaddr_in *addr);
يضع عنوان IP الخاص بالجهاز في *addr، دون استشارة روتينيات المكتبة التي تتعامل مع /etc/hosts. يتم دائمًا تعيين رقم المنفذ على htons(PMAPPORT).
struct pmaplist *pmap_getmaps(struct sockaddr_in *addr);
واجهة مستخدم لخدمة portmap، والتي تُرجع قائمة بخرائط البرامج RPC الحالية إلى المنافذ على المضيف الموجود في عنوان IP *addr. يمكن أن تُرجع هذه الروتين NULL. الأمر rpcinfo -p يستخدم هذه الروتين.
unsigned short pmap_getport(struct sockaddr_in *addr,
                    unsigned long prognum, unsigned long versnum,
                    unsigned int protocol);
واجهة مستخدم لخدمة portmap، والتي تُرجع رقم المنفذ الذي تنتظر عليه خدمة تدعم رقم البرنامج prognum، والإصدار versnum، وتتحدث بروتوكول النقل المرتبط بـ protocol. القيمة protocol على الأرجح هي IPPROTO_UDP أو IPPROTO_TCP. القيمة المرجعة صفر تعني أن الخريطة غير موجودة أو أن نظام RPC فشل في الاتصال بخدمة portmap البعيدة. في الحالة الأخيرة، تحتوي المتغير العام rpc_createerr على حالة RPC.
enum clnt_stat pmap_rmtcall(struct sockaddr_in *addr,
                    unsigned long prognum, unsigned long versnum,
                    unsigned long procnum,
                    xdrproc_t inproc, char *in,
                    xdrproc_t outproc, char *out,
                    struct timeval tout, unsigned long *portp);
واجهة مستخدم لخدمة portmap، والتي توجه portmap على المضيف في عنوان IP *addr لإجراء استدعاء RPC نيابة عنك إلى إجراء على هذا المضيف. سيتم تعديل الوسيط *portp ليصبح رقم منفذ البرنامج إذا نجح الإجراء. تم مناقشة تعريفات الوسائط الأخرى في callrpc() و clnt_call(). يجب استخدام هذا الإجراء لـ "ping" ولا شيء آخر. انظر أيضًا clnt_broadcast().
bool_t pmap_set(unsigned long prognum, unsigned long versnum,
                int protocol, unsigned short port);
واجهة مستخدم لخدمة portmap، والتي تنشئ تعيينًا بين الثلاثي [prognum،versnum،protocol] و port على خدمة portmap الخاصة بالجهاز. القيمة protocol على الأرجح هي IPPROTO_UDP أو IPPROTO_TCP. تُرجع هذه الروتين واحدًا إذا نجحت، وصفرًا بخلاف ذلك. يتم ذلك تلقائيًا بواسطة svc_register().
bool_t pmap_unset(unsigned long prognum, unsigned long versnum);
واجهة مستخدم لخدمة portmap، والتي تدمر جميع التعيينات بين الثلاثي [prognum،versnum،*] و ports على خدمة portmap الخاصة بالجهاز. تُرجع هذه الروتين واحدًا إذا نجحت، وصفرًا بخلاف ذلك.
int registerrpc(unsigned long prognum, unsigned long versnum,
                unsigned long procnum, typeof(char *(char *)) *procname,
                xdrproc_t inproc, xdrproc_t outproc);
تسجيل الإجراء procname مع حزمة خدمات RPC. إذا وصل طلب للبرنامج prognum، والإصدار versnum، والإجراء procnum، يتم استدعاء procname بمؤشر إلى وسيطه (وسائطه)؛ يجب أن تُرجع procname مؤشرًا إلى نتيجته (نتائجه) الثابتة؛ يتم استخدام inproc لفك تشفير الوسائط بينما يتم استخدام outproc لتشفير النتائج. تُرجع هذه الروتين صفرًا إذا نجح التسجيل، -1 بخلاف ذلك.
