subroutine claqhb (character uplo, integer n, integer kd, complex, dimension( ldab, * ) ab, integer ldab, real, dimension( * ) s, real scond, real amax, character equed)

CLAQHB يقوم بتوسيع مصفوفة هيرميتية ذات نطاق، باستخدام عوامل التوسيع المحسوبة بواسطة cpbequ.

الغرض:

!>
!> CLAQHB تقوم بمعادلة مصفوفة هيرميتية ذات نطاق A باستخدام عوامل التوسيع
!> في المتجه S.
!>

المعلمات

UPLO 
!>          UPLO is CHARACTER*1
!>          يحدد ما إذا كان الجزء العلوي أو السفلي المثلثي للمصفوفة المتماثلة A
!>          مخزنًا.
!>          = 'U':  مثلثي علوي
!>          = 'L':  مثلثي سفلي
!>

N

!>          N is INTEGER
!>          ترتيب المصفوفة A.  N >= 0.
!>

KD

!>          KD is INTEGER
!>          عدد الأقطار الفائقة للمصفوفة A إذا كان UPLO = 'U'،
!>          أو عدد الأقطار الفرعية إذا كان UPLO = 'L'.  KD >= 0.
!>

AB

!>          AB is COMPLEX array, dimension (LDAB,N)
!>          عند الإدخال، الجزء العلوي أو السفلي المثلثي للمصفوفة المتماثلة ذات النطاق
!>          A، مخزن في أول KD+1 صفوف من المصفوفة.  العمود j من A مخزن
!>          في العمود j من المصفوفة AB كما يلي:
!>          إذا كان UPLO = 'U'، AB(kd+1+i-j,j) = A(i,j) لـ max(1,j-kd)<=i<=j؛
!>          إذا كان UPLO = 'L'، AB(1+i-j,j)    = A(i,j) لـ j<=i<=min(n,j+kd).
!>
!>          عند الخروج، إذا كان INFO = 0، فإن العامل المثلثي U أو L من
!>          تحليل تشوليسكي A = U**H *U أو A = L*L**H للمصفوفة ذات النطاق
!>          A، بنفس تنسيق التخزين مثل A.
!>

LDAB

!>          LDAB is INTEGER
!>          البعد الرائد للمصفوفة AB.  LDAB >= KD+1.
!>

S

!>          S is REAL array, dimension (N)
!>          عوامل التوسيع للمصفوفة A.
!>

SCOND

!>          SCOND is REAL
!>          نسبة أصغر S(i) إلى أكبر S(i).
!>

AMAX

!>          AMAX is REAL
!>          القيمة المطلقة لأكبر عنصر في المصفوفة.
!>

EQUED

!>          EQUED is CHARACTER*1
!>          يحدد ما إذا كان قد تم إجراء المعادلة أم لا.
!>          = 'N':  لم يتم إجراء المعادلة.
!>          = 'Y':  تم إجراء المعادلة، أي تم استبدال A بـ
!>                  diag(S) * A * diag(S).
!>

المعلمات الداخلية:

!>  THRESH هو قيمة حدية تستخدم لتحديد ما إذا كان يجب إجراء التوسيع
!>  بناءً على نسبة عوامل التوسيع.  إذا كان SCOND < THRESH،
!>  يتم إجراء التوسيع.
!>
!>  LARGE و SMALL هما قيم حدية تستخدم لتحديد ما إذا كان يجب إجراء
!>  التوسيع بناءً على الحجم المطلق لأكبر عنصر في المصفوفة.
!>  إذا كان AMAX > LARGE أو AMAX < SMALL، يتم إجراء التوسيع.
!>
