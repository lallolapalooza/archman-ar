subroutine clarfb (side, trans, direct, storev, m, n, k, v, ldv, t, ldt, c, ldc, work, ldwork)
CLARFB يطبق عاكس كتلة أو معكوسه المرافق على مصفوفة مستطيلة عامة.

توثيق الوظيفة/الإجراء الفرعي

subroutine clarfb (character side, character trans, character direct, character storev, integer m, integer n, integer k, complex, dimension( ldv, * ) v, integer ldv, complex, dimension( ldt, * ) t, integer ldt, complex, dimension( ldc, * ) c, integer ldc, complex, dimension( ldwork, * ) work, integer ldwork)

CLARFB يطبق عاكس كتلة معقد H أو معكوسه المرافق H**H على مصفوفة مستطيلة عامة C، من اليسار أو اليمين.

الغرض:

!>
!> CLARFB يطبق عاكس كتلة معقد H أو معكوسه المرافق H**H على مصفوفة
!> مستطيلة عامة M-by-N وهي C، من اليسار أو اليمين.
!>

المعاملات

SIDE 
!>          SIDE is CHARACTER*1
!>          = 'L': تطبيق H أو H**H من اليسار
!>          = 'R': تطبيق H أو H**H من اليمين
!>

TRANS

!>          TRANS is CHARACTER*1
!>          = 'N': تطبيق H (بدون تبديل)
!>          = 'C': تطبيق H**H (تبديل مرافق)
!>

DIRECT

!>          DIRECT is CHARACTER*1
!>          يشير إلى كيفية تشكيل H من حاصل ضرب عواكس أولية
!>          = 'F': H = H(1) H(2) . . . H(k) (أمامي)
!>          = 'B': H = H(k) . . . H(2) H(1) (خلفي)
!>

STOREV

!>          STOREV is CHARACTER*1
!>          يشير إلى كيفية تخزين المتجهات التي تحدد العواكس الأولية:
!>          = 'C': عموديا
!>          = 'R': صفيا
!>

M

!>          M is INTEGER
!>          عدد صفوف المصفوفة C.
!>

N

!>          N is INTEGER
!>          عدد أعمدة المصفوفة C.
!>

K

!>          K is INTEGER
!>          رتبة المصفوفة T (= عدد العواكس الأولية التي يشكل حاصل ضربها
!>          العاكس الكتلي).
!>          إذا كانت SIDE = 'L', M >= K >= 0؛
!>          إذا كانت SIDE = 'R', N >= K >= 0.
!>

V

!>          V is COMPLEX array, dimension
!>                                (LDV,K) if STOREV = 'C'
!>                                (LDV,M) if STOREV = 'R' and SIDE = 'L'
!>                                (LDV,N) if STOREV = 'R' and SIDE = 'R'
!>          المصفوفة V. انظر التفاصيل الإضافية.
!>

LDV

!>          LDV is INTEGER
!>          البعد الرائد للمصفوفة V.
!>          إذا كانت STOREV = 'C' و SIDE = 'L', LDV >= max(1,M)؛
!>          إذا كانت STOREV = 'C' و SIDE = 'R', LDV >= max(1,N)؛
!>          إذا كانت STOREV = 'R', LDV >= K.
!>

T

!>          T is COMPLEX array, dimension (LDT,K)
!>          المصفوفة المثلثية K-by-K T في تمثيل العاكس الكتلي.
!>

LDT

!>          LDT is INTEGER
!>          البعد الرائد للمصفوفة T. LDT >= K.
!>

C

!>          C is COMPLEX array, dimension (LDC,N)
!>          عند الإدخال، المصفوفة C ذات الأبعاد M-by-N.
!>          عند الخروج، يتم استبدال C بـ H*C أو H**H*C أو C*H أو C*H**H.
!>

LDC

!>          LDC is INTEGER
!>          البعد الرائد للمصفوفة C. LDC >= max(1,M).
!>

WORK

!>          WORK is COMPLEX array, dimension (LDWORK,K)
!>

LDWORK

!>          LDWORK is INTEGER
!>          البعد الرائد للمصفوفة WORK.
!>          إذا كانت SIDE = 'L', LDWORK >= max(1,N)؛
!>          إذا كانت SIDE = 'R', LDWORK >= max(1,M).
!>

تفاصيل إضافية:

!>
!>  يتم توضيح شكل المصفوفة V وتخزين المتجهات التي تحدد H(i) بشكل أفضل
!>  من خلال المثال التالي مع n = 5 و k = 3. الجزء المثلثي من V (بما في ذلك
!>  قطره) غير مرجعي.
!>
!>  DIRECT = 'F' و STOREV = 'C':         DIRECT = 'F' و STOREV = 'R':
!>
!>               V = (  1       )                 V = (  1 v1 v1 v1 v1 )
!>                   ( v1  1    )                     (     1 v2 v2 v2 )
!>                   ( v1 v2  1 )                     (        1 v3 v3 )
!>                   ( v1 v2 v3 )
!>                   ( v1 v2 v3 )
!>
!>  DIRECT = 'B' و STOREV = 'C':         DIRECT = 'B' و STOREV = 'R':
!>
!>               V = ( v1 v2 v3 )                 V = ( v1 v1  1       )
!>                   ( v1 v2 v3 )                     ( v2 v2 v2  1    )
!>                   (  1 v2 v3 )                     ( v3 v3 v3 v3  1 )
!>                   (     1 v3 )
!>                   (        1 )
!>
