توثيق الوظيفة/الروتين الفرعي

subroutine clqt01 (integer m, integer n, complex, dimension( lda, * ) a, complex, dimension( lda, * ) af, complex, dimension( lda, * ) q, complex, dimension( lda, * ) l, integer lda, complex, dimension( * ) tau, complex, dimension( lwork ) work, integer lwork, real, dimension( * ) rwork, real, dimension( * ) result)

CLQT01

الغرض:

!>
!> CLQT01 يختبر CGELQF، الذي يحسب تحليل LQ لمصفوفة m × n
!> A، ويختبر جزئيًا CUNGLQ الذي يشكل مصفوفة Q المتعامدة n × n.
!>
!> CLQT01 يقارن L مع A*Q'، ويتحقق من أن Q متعامدة.
!>

المعلمات

M 
!>          M هو عدد صحيح (INTEGER)
!>          عدد صفوف المصفوفة A.  M >= 0.
!>

N

!>          N هو عدد صحيح (INTEGER)
!>          عدد أعمدة المصفوفة A.  N >= 0.
!>

A

!>          A هو مصفوفة مركبة (COMPLEX)، أبعادها (LDA,N)
!>          المصفوفة A ذات الأبعاد m × n.
!>

AF

!>          AF هو مصفوفة مركبة (COMPLEX)، أبعادها (LDA,N)
!>          تفاصيل تحليل LQ للمصفوفة A، كما تم إرجاعها بواسطة CGELQF.
!>          انظر CGELQF لمزيد من التفاصيل.
!>

Q

!>          Q هو مصفوفة مركبة (COMPLEX)، أبعادها (LDA,N)
!>          المصفوفة المتعامدة Q ذات الأبعاد n × n.
!>

L

!>          L هو مصفوفة مركبة (COMPLEX)، أبعادها (LDA,max(M,N))
!>

LDA

!>          LDA هو عدد صحيح (INTEGER)
!>          البعد الرئيسي للمصفوفات A، AF، Q و L.
!>          LDA >= max(M,N).
!>

TAU

!>          TAU هو مصفوفة مركبة (COMPLEX)، أبعادها (min(M,N))
!>          عوامل القياس للمنعكسات الأولية، كما تم إرجاعها
!>          بواسطة CGELQF.
!>

WORK

!>          WORK هو مصفوفة مركبة (COMPLEX)، أبعادها (LWORK)
!>

LWORK

!>          LWORK هو عدد صحيح (INTEGER)
!>          بعد المصفوفة WORK.
!>

RWORK

!>          RWORK هو مصفوفة حقيقية (REAL)، أبعادها (max(M,N))
!>

RESULT

!>          RESULT هو مصفوفة حقيقية (REAL)، أبعادها (2)
!>          نسب الاختبار:
!>          RESULT(1) = norm( L - A*Q' ) / ( N * norm(A) * EPS )
!>          RESULT(2) = norm( I - Q*Q' ) / ( N * EPS )
!>
