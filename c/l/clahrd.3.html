subroutine clahrd (n, k, nb, a, lda, tau, t, ldt, y, ldy)
CLAHRD تقلل أول NB عمود من مصفوفة عامة مستطيلة A بحيث تصبح العناصر الموجودة أسفل القطر الفرعي k صفرًا، وتعيد مصفوفات مساعدة مطلوبة لتطبيق التحويل على الجزء غير المخفض من A.

توثيق الوظيفة/الإجراء الفرعي

subroutine clahrd (integer n, integer k, integer nb, complex, dimension( lda, * ) a, integer lda, complex, dimension( nb ) tau, complex, dimension( ldt, nb ) t, integer ldt, complex, dimension( ldy, nb ) y, integer ldy)

CLAHRD تقلل أول NB عمود من مصفوفة عامة مستطيلة A بحيث تصبح العناصر الموجودة أسفل القطر الفرعي k صفرًا، وتعيد مصفوفات مساعدة مطلوبة لتطبيق التحويل على الجزء غير المخفض من A.

الغرض:

!>
!> هذا الإجراء الفرعي مهمل وتم استبداله بالإجراء الفرعي CLAHR2.
!>
!> CLAHRD تقلل أول NB عمود من مصفوفة A العامة المعقدة n×(n-k+1)
!> بحيث تصبح العناصر الموجودة أسفل القطر الفرعي k صفرًا. يتم
!> إجراء التخفيض بواسطة تحويل تشابه موحد
!> Q**H * A * Q. يعيد الإجراء الفرعي المصفوفات V و T التي تحدد
!> Q كمُسْكِت كتلة I - V*T*V**H، وكذلك المصفوفة Y = A * V * T.
!>

المعاملات

N 
!>          N هو عدد صحيح (INTEGER)
!>          رتبة المصفوفة A.
!>

K

!>          K هو عدد صحيح (INTEGER)
!>          الإزاحة للتخفيض. يتم تخفيض العناصر الموجودة أسفل القطر الفرعي k
!>          في أول NB عمود إلى الصفر.
!>

NB

!>          NB هو عدد صحيح (INTEGER)
!>          عدد الأعمدة التي سيتم تخفيضها.
!>

A

!>          A هي مصفوفة مركبة (COMPLEX)، البعد (LDA,N-K+1)
!>          عند الدخول، المصفوفة العامة A بحجم n×(n-k+1).
!>          عند الخروج، يتم الكتابة فوق العناصر الموجودة على القطر الفرعي k وما فوقه في
!>          أول NB عمود بالعناصر المقابلة للمصفوفة المخفضة؛ العناصر الموجودة أسفل القطر الفرعي k،
!>          مع المصفوفة TAU، تمثل المصفوفة Q كـ ناتج لمُسْكِتات أولية. الأعمدة الأخرى من A
!>          تبقى دون تغيير. انظر التفاصيل الإضافية.
!>

LDA

!>          LDA هو عدد صحيح (INTEGER)
!>          البعد الرئيسي للمصفوفة A. LDA >= max(1,N).
!>

TAU

!>          TAU هي مصفوفة مركبة (COMPLEX)، البعد (NB)
!>          العوامل القياسية للمُسْكِتات الأولية. انظر التفاصيل الإضافية.
!>

T

!>          T هي مصفوفة مركبة (COMPLEX)، البعد (LDT,NB)
!>          المصفوفة المثلثية العليا T.
!>

LDT

!>          LDT هو عدد صحيح (INTEGER)
!>          البعد الرئيسي للمصفوفة T. LDT >= NB.
!>

Y

!>          Y هي مصفوفة مركبة (COMPLEX)، البعد (LDY,NB)
!>          المصفوفة Y بحجم n×nb.
!>

LDY

!>          LDY هو عدد صحيح (INTEGER)
!>          البعد الرئيسي للمصفوفة Y. LDY >= max(1,N).
!>

التفاصيل الإضافية:

!>
!>  يتم تمثيل المصفوفة Q كـ ناتج لـ nb مُسْكِت أولي
!>
!>     Q = H(1) H(2) . . . H(nb).
!>
!>  كل H(i) لها الشكل
!>
!>     H(i) = I - tau * v * v**H
!>
!>  حيث tau هو عامل قياسي مركب، و v هو متجه مركب مع
!>  v(1:i+k-1) = 0، v(i+k) = 1؛ يتم تخزين v(i+k+1:n) عند الخروج في
!>  A(i+k+1:n,i)، و tau في TAU(i).
!>
!>  عناصر المتجهات v تشكل معًا المصفوفة (n-k+1)×nb
!>  V المطلوبة، مع T و Y، لتطبيق التحويل على
!>  الجزء غير المخفض من المصفوفة، باستخدام تحديث بالشكل:
!>  A := (I - V*T*V**H) * (A - Y*V**H).
!>
!>  محتويات A عند الخروج موضحة بالمثال التالي
!>  مع n = 7، k = 3 و nb = 2:
!>
!>     ( a   h   a   a   a )
!>     ( a   h   a   a   a )
!>     ( a   h   a   a   a )
!>     ( h   h   a   a   a )
!>     ( v1  h   a   a   a )
!>     ( v1  v2  a   a   a )
!>     ( v1  v2  a   a   a )
!>
!>  حيث تشير 'a' إلى عنصر من المصفوفة الأصلية A، و 'h' تشير إلى عنصر معدل
!>  من المصفوفة هسنبرغ العليا H، و 'vi' تشير إلى عنصر
!>  من المتجه الذي يحدد H(i).
!>
