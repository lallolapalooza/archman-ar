وثائق الوظيفة/الروتين الفرعي

subroutine clacp2 (character uplo, integer m, integer n, real, dimension( lda, * ) a, integer lda, complex, dimension( ldb, * ) b, integer ldb)

CLACP2 تنسخ جزءًا أو كل مصفوفة ثنائية الأبعاد حقيقية إلى مصفوفة معقدة.

الغرض:

!>
!> CLACP2 تنسخ مصفوفة حقيقية ثنائية الأبعاد A، كلياً أو جزئياً، إلى مصفوفة
!> معقدة B.
!>

المعاملات

UPLO 
!>          UPLO هو CHARACTER*1
!>          يحدد جزء المصفوفة A الذي سيتم نسخه إلى B.
!>          = 'U':      الجزء المثلثي العلوي
!>          = 'L':      الجزء المثلثي السفلي
!>          غير ذلك:  كل مصفوفة A
!>

M

!>          M هو INTEGER
!>          عدد صفوف المصفوفة A.  M >= 0.
!>

N

!>          N هو INTEGER
!>          عدد أعمدة المصفوفة A.  N >= 0.
!>

A

!>          A هي مصفوفة REAL، ذات بُعد (LDA,N)
!>          المصفوفة A التي أبعادها m × n.  إذا كانت UPLO = 'U'، يتم الوصول
!>          فقط إلى شبه المنحرف العلوي؛ إذا كانت UPLO = 'L'، يتم الوصول فقط
!>          إلى شبه المنحرف السفلي.
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرائد للمصفوفة A.  LDA >= max(1,M).
!>

B

!>          B هي مصفوفة COMPLEX، ذات بُعد (LDB,N)
!>          عند الخروج، B = A في المواقع المحددة بواسطة UPLO.
!>

LDB

!>          LDB هو INTEGER
!>          البعد الرائد للمصفوفة B.  LDB >= max(1,M).
!>
