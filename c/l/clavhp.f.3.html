subroutine clavhp (character uplo, character trans, character diag, integer n, integer nrhs, complex, dimension( * ) a, integer, dimension( * ) ipiv, complex, dimension( ldb, * ) b, integer ldb, integer info)

CLAVHP

الغرض:

!>
!>    CLAVHP  تقوم بواحدة من عمليات المصفوفة والمتجه
!>       x := A*x  أو  x := A^H*x,
!>    حيث x هو متجه من N عنصر و A هي إحدى العوامل
!>    من التحلل المتماثل الذي تم حسابه بواسطة CHPTRF.
!>    ينتج CHPTRF تحللاً بالشكل
!>         U * D * U^H     أو     L * D * L^H,
!>    حيث U (أو L) هو حاصل ضرب مصفوفات تبديل ومصفوفات مثلثة علوية (سفلية)
!>    وحدوية، U^H (أو L^H) هو المرافق النقال لـ U (أو L)، و D هي مصفوفة متماثلة
!>    قطرية كتلية ذات كتل قطرية 1x1 و 2x2. يتم تخزين المضاعفات للتحويلات
!>    والأجزاء المثلثية العلوية أو السفلية للكتل القطرية كأعمدة في تنسيق مدمج
!>    في المصفوفة الخطية A.
!>
!>    إذا كان TRANS = 'N' أو 'n'، تضرب CLAVHP إما بـ U أو U * D
!>    (أو L أو L * D).
!>    إذا كان TRANS = 'C' أو 'c'، تضرب CLAVHP إما بـ U^H أو D * U^H
!>    (أو L^H أو D * L^H).
!> 
!>  UPLO   - CHARACTER*1
!>           عند الإدخال، تحدد UPLO ما إذا كانت المصفوفة المثلثية
!>           المخزنة في A علوية أو سفلية.
!>              UPLO = 'U' أو 'u'   المصفوفة علوية.
!>              UPLO = 'L' أو 'l'   المصفوفة سفلية.
!>           لا تتغير عند الخروج.
!>
!>  TRANS  - CHARACTER*1
!>           عند الإدخال، تحدد TRANS العملية التي سيتم إجراؤها كما يلي:
!>              TRANS = 'N' أو 'n'   x := A*x.
!>              TRANS = 'C' أو 'c'   x := A^H*x.
!>           لا تتغير عند الخروج.
!>
!>  DIAG   - CHARACTER*1
!>           عند الإدخال، تحدد DIAG ما إذا كانت الكتل القطرية
!>           تُفترض أنها مصفوفات وحدوية، كما يلي:
!>              DIAG = 'U' أو 'u'   الكتل القطرية مصفوفات وحدوية.
!>              DIAG = 'N' أو 'n'   الكتل القطرية ليست وحدوية.
!>           لا تتغير عند الخروج.
!>
!>  N      - INTEGER
!>           عند الإدخال، يحدد N رتبة المصفوفة A.
!>           يجب أن يكون N على الأقل صفرًا.
!>           لا يتغير عند الخروج.
!>
!>  NRHS   - INTEGER
!>           عند الإدخال، يحدد NRHS عدد الأيادي اليمنى،
!>           أي عدد المتجهات x التي سيتم ضربها بـ A.
!>           يجب أن يكون NRHS على الأقل صفرًا.
!>           لا يتغير عند الخروج.
!>
!>  A      - COMPLEX array, dimension( N*(N+1)/2 )
!>           عند الإدخال، تحتوي A على مصفوفة قطرية كتلية
!>           والمضاعفات للتحويلات المستخدمة للحصول عليها،
!>           مخزنة كمصفوفة مثلثة مدمجة.
!>           لا تتغير عند الخروج.
!>
!>  IPIV   - INTEGER array, dimension( N )
!>           عند الإدخال، تحتوي IPIV على متجه مؤشرات الارتكاز كما
!>           حددته CSPTRF أو CHPTRF.
!>           إذا كان IPIV( K ) = K، لم يتم إجراء أي تبديل.
!>           إذا كان IPIV( K ) <> K ولكن IPIV( K ) > 0، فقد تم تبديل الصف K
!>           مع الصف IPIV( K ) وتم استخدام كتلة 1x1.
!>           إذا كان IPIV( K ) < 0 و UPLO = 'U'، فقد تم تبديل الصف K-1
!>           مع الصف | IPIV( K ) | وتم استخدام كتلة 2x2.
!>           إذا كان IPIV( K ) < 0 و UPLO = 'L'، فقد تم تبديل الصف K+1
!>           مع الصف | IPIV( K ) | وتم استخدام كتلة 2x2.
!>
!>  B      - COMPLEX array, dimension( LDB, NRHS )
!>           عند الإدخال، تحتوي B على NRHS متجه بطول N.
!>           عند الخروج، يتم استبدال B بـ A * B.
!>
!>  LDB    - INTEGER
!>           عند الإدخال، يحتوي LDB على البعد الرائد لـ B كما
!>           أعلن عنه في البرنامج المستدعي. يجب أن يكون LDB على الأقل
!>           max( 1, N ).
!>           لا يتغير عند الخروج.
!>
!>  INFO   - INTEGER
!>           INFO هو مؤشر الخطأ.
!>           عند الخروج، تشير القيمة 0 إلى خروج ناجح.
!>           تشير قيمة سالبة، مثل -K، إلى أن الوسيط K
!>           له قيمة غير قانونية.
!>
