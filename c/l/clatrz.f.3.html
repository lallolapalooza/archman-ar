subroutine clatrz (m, n, l, a, lda, tau, work)
CLATRZ تقوم بتحليل مصفوفة شبه منحرفة عليا بواسطة تحويلات وحدوية.

توثيق الوظيفة/الوحدة الفرعية

subroutine clatrz (integer m, integer n, integer l, complex, dimension( lda, * ) a, integer lda, complex, dimension( * ) tau, complex, dimension( * ) work)

CLATRZ تقوم بتحليل مصفوفة شبه منحرفة عليا بواسطة تحويلات وحدوية.

الغرض:

!>
!> CLATRZ تقوم بتحليل المصفوفة شبه المنحرفة العليا المعقدة M-by-(M+L)
!> [ A1 A2 ] = [ A(1:M,1:M) A(1:M,N-L+1:N) ] إلى ( R  0 ) * Z بواسطة
!> تحويلات وحدوية، حيث Z هي مصفوفة وحدوية (M+L)-by-(M+L) و R و A1 هي
!> مصفوفات مثلثية عليا M-by-M.
!>

المعلمات

M 
!>          M هو عدد صحيح INTEGER
!>          عدد صفوف المصفوفة A.  M >= 0.
!>

N

!>          N هو عدد صحيح INTEGER
!>          عدد أعمدة المصفوفة A.  N >= 0.
!>

L

!>          L هو عدد صحيح INTEGER
!>          عدد أعمدة المصفوفة A التي تحتوي على الجزء الهادف
!>          من متجهات هاوسهولدر. N-M >= L >= 0.
!>

A

!>          A هي مصفوفة معقدة COMPLEX array، ذات بُعد (LDA,N)
!>          عند الدخول، يجب أن يحتوي الجزء العلوي شبه المنحرف من A
!>          بحجم M-by-N على المصفوفة المراد تحليلها.
!>          عند الخروج، يحتوي الجزء العلوي المثلثي من A بحجم M-by-M
!>          على المصفوفة المثلثية العليا R، والعناصر من N-L+1 إلى N
!>          من الصفوف M الأولى من A، مع المصفوفة TAU، على المصفوفة
!>          الوحدوية Z كناتج لـ M من الانعكاسات الأولية.
!>

LDA

!>          LDA هو عدد صحيح INTEGER
!>          البعد الرائد للمصفوفة A.  LDA >= max(1,M).
!>

TAU

!>          TAU هي مصفوفة معقدة COMPLEX array، ذات بُعد (M)
!>          العوامل القياسية للانعكاسات الأولية.
!>

WORK

!>          WORK هي مصفوفة معقدة COMPLEX array، ذات بُعد (M)
!>

تفاصيل إضافية:

!>
!> يتم الحصول على التحليل باستخدام طريقة هاوسهولدر. مصفوفة التحويل
!> k-th، Z( k )، التي تستخدم لإدخال أصفار في الصف ( m - k + 1 ) من A،
!> تُعطى في الشكل
!>
!>     Z( k ) = ( I     0   ),
!>              ( 0  T( k ) )
!>
!> حيث
!>
!>     T( k ) = I - tau*u( k )*u( k )**H,   u( k ) = (   1    ),
!>                                                 (   0    )
!>                                                 ( z( k ) )
!>
!>  tau هو عامل قياسي و z( k ) هو متجه ذو l عنصر. يتم اختيار tau و z( k )
!>  لإبطال عناصر الصف k من A2.
!>
!>  يتم إرجاع العامل القياسي tau في العنصر k من TAU والمتجه
!>  u( k ) في الصف k من A2، بحيث تكون عناصر z( k ) موجودة في
!>  a( k, l + 1 )، ...، a( k, n ). يتم إرجاع عناصر R في
!>  الجزء المثلثي العلوي من A1.
!>
!>  Z مُعطاة بواسطة
!>
!>     Z =  Z( 1 ) * Z( 2 ) * ... * Z( m ).
!>
