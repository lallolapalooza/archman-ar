subroutine clauu2 (uplo, n, a, lda, info)
CLAUU2 تحسب حاصل الضرب UUH أو LHL، حيث U و L هي مصفوفات مثلثية عليا أو سفلى (خوارزمية غير مجمعة).

توثيق الوظيفة/الإجراء الفرعي

subroutine clauu2 (character uplo, integer n, complex, dimension( lda, * ) a, integer lda, integer info)

CLAUU2 تحسب حاصل الضرب UUH أو LHL، حيث U و L هي مصفوفات مثلثية عليا أو سفلى (خوارزمية غير مجمعة).

الغرض:

!>
!> CLAUU2 تحسب حاصل الضرب U * U**H أو L**H * L، حيث العامل المثلثي
!> U أو L مخزن في الجزء العلوي أو السفلي من المصفوفة
!> A.
!>
!> إذا كان UPLO = 'U' أو 'u' فإن المثلث العلوي للنتيجة يتم تخزينه،
!> مستبدلاً العامل U في A.
!> إذا كان UPLO = 'L' أو 'l' فإن المثلث السفلي للنتيجة يتم تخزينه،
!> مستبدلاً العامل L في A.
!>
!> هذا هو الشكل غير المجمع للخوارزمية، والذي يستدعي BLAS من المستوى 2.
!>

المعلمات

UPLO 
!>          UPLO هو CHARACTER*1
!>          يحدد ما إذا كان العامل المثلثي المخزن في المصفوفة A
!>          هو مثلثي علوي أو سفلي:
!>          = 'U':  مثلثي علوي
!>          = 'L':  مثلثي سفلي
!>

N

!>          N هو INTEGER
!>          رتبة العامل المثلثي U أو L.  N >= 0.
!>

A

!>          A هو مصفوفة COMPLEX، البعد (LDA,N)
!>          عند الدخول، العامل المثلثي U أو L.
!>          عند الخروج، إذا كان UPLO = 'U'، فإن المثلث العلوي لـ A
!>          يتم استبداله بالمثلث العلوي لحاصل الضرب U * U**H؛
!>          إذا كان UPLO = 'L'، فإن المثلث السفلي لـ A يتم استبداله
!>          بالمثلث السفلي لحاصل الضرب L**H * L.
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرائد للمصفوفة A.  LDA >= max(1,N).
!>

INFO

!>          INFO هو INTEGER
!>          = 0: انتهاء ناجح
!>          < 0: إذا كان INFO = -k، فإن المعامل k كان لديه قيمة غير قانونية
!>
