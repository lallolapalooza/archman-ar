subroutine clatdf (ijob, n, z, ldz, rhs, rdsum, rdscal, ipiv, jpiv)
CLATDF يستخدم تحليل LU للمصفوفة n×n المحسوبة بواسطة sgetc2 ويحسب مساهمة في تقدير Dif العكسي.

توثيق الوظيفة/الروتين الفرعي

subroutine clatdf (integer ijob, integer n, complex, dimension( ldz, * ) z, integer ldz, complex, dimension( * ) rhs, real rdsum, real rdscal, integer, dimension( * ) ipiv, integer, dimension( * ) jpiv)

CLATDF يستخدم تحليل LU للمصفوفة n×n المحسوبة بواسطة sgetc2 ويحسب مساهمة في تقدير Dif العكسي.

الغرض:

!>
!> CLATDF يحسب المساهمة في تقدير Dif العكسي
!> عن طريق الحل للمعادلة x في Z * x = b ، حيث يتم اختيار b بحيث يكون
!> معيار x أكبر ما يمكن. يُفترض أن تحليل LU للمصفوفة Z قد تم حسابه بواسطة CGETC2.
!> عند الدخول، RHS = f يحتوي على المساهمة من الأنظمة الفرعية التي تم حلها سابقًا،
!> وعند الخروج، RHS = x.
!>
!> تحليل Z الذي تم إرجاعه بواسطة CGETC2 له الشكل
!> Z = P * L * U * Q ، حيث P و Q مصفوفتان تبديل. L هي مصفوفة ثلاثية سفلية
!> ذات عناصر قطرية وحدة و U مصفوفة ثلاثية علوية.
!>

المعلمات:

IJOB 
!>          IJOB هو عدد صحيح (INTEGER)
!>          IJOB = 2: أولاً، قم بحساب متجه صفري تقريبي e
!>              للمصفوفة Z باستخدام CGECON، ويتم تطبيع e وحل المعادلة
!>              Zx = +-e - f مع الإشارة التي تعطي أكبر قيمة لـ
!>              2-norm(x). تكلفة أغلى بحوالي 5 مرات من الافتراضي.
!>          IJOB .ne. 2: استراتيجية النظر المستقبلي المحلية حيث
!>              يتم اختيار جميع مدخلات الطرف الأيمن b لتكون إما +1 أو
!>              -1. افتراضي.
!>

N

!>          N هو عدد صحيح (INTEGER)
!>          عدد أعمدة المصفوفة Z.
!>

Z

!>          Z هي مصفوفة مركبة (COMPLEX array)، البعد (LDZ, N)
!>          عند الدخول، الجزء LU لتحليل المصفوفة n×n
!>          Z المحسوبة بواسطة CGETC2: Z = P * L * U * Q
!>

LDZ

!>          LDZ هو عدد صحيح (INTEGER)
!>          البعد الرائد للمصفوفة Z. LDA >= max(1, N).
!>

RHS

!>          RHS هي مصفوفة مركبة (COMPLEX array)، البعد (N).
!>          عند الدخول، تحتوي RHS على مساهمات من أنظمة فرعية أخرى.
!>          عند الخروج، تحتوي RHS على حل النظام الفرعي مع
!>          المدخلات وفقًا لقيمة IJOB (انظر أعلاه).
!>

RDSUM

!>          RDSUM هو عدد حقيقي (REAL)
!>          عند الدخول، مجموع مربعات المساهمات المحسوبة لـ
!>          تقدير Dif قيد الحساب بواسطة CTGSYL، حيث
!>          تم فصل عامل القياس RDSCAL (انظر أدناه).
!>          عند الخروج، مجموع المربعات المقابل الذي تم تحديثه بـ
!>          المساهمات من النظام الفرعي الحالي.
!>          إذا كان TRANS = 'T'، فلا يتم المساس بـ RDSUM.
!>          ملاحظة: RDSUM له معنى فقط عندما يتم استدعاء CTGSY2 بواسطة CTGSYL.
!>

RDSCAL

!>          RDSCAL هو عدد حقيقي (REAL)
!>          عند الدخول، عامل القياس المستخدم لمنع تجاوز السعة في RDSUM.
!>          عند الخروج، يتم تحديث RDSCAL بالنسبة للمساهمات الحالية
!>          في RDSUM.
!>          إذا كان TRANS = 'T'، فلا يتم المساس بـ RDSCAL.
!>          ملاحظة: RDSCAL له معنى فقط عندما يتم استدعاء CTGSY2 بواسطة
!>          CTGSYL.
!>

IPIV

!>          IPIV هي مصفوفة أعداد صحيحة (INTEGER array)، البعد (N).
!>          مؤشرات المحور؛ لـ 1 <= i <= N، تم تبديل الصف i للمصفوفة
!>          مع الصف IPIV(i).
!>

JPIV

!>          JPIV هي مصفوفة أعداد صحيحة (INTEGER array)، البعد (N).
!>          مؤشرات المحور؛ لـ 1 <= j <= N، تم تبديل العمود j للمصفوفة
!>          مع العمود JPIV(j).
!>
