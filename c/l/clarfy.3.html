subroutine clarfy (character uplo, integer n, complex, dimension( * ) v, integer incv, complex tau, complex, dimension( ldc, * ) c, integer ldc, complex, dimension( * ) work)

CLARFY

الغرض:

!>
!> CLARFY تُطبّق مُنْعَكِسًا عنصريًا، أو مصفوفة هاوسهولدر، H،
!> على مصفوفة متقارنة n x n وهي C، من اليسار واليمين.
!>
!> H مُمَثَّلة بالشكل
!>
!>    H = I - tau * v * v'
!>
!> حيث tau قيمة عددية و v متجهة.
!>
!> إذا كانت tau = 0، فتُعتبر H مصفوفة الوحدة.
!>

المُعامِلات

UPLO 
!>          UPLO هو CHARACTER*1
!>          يُحدِّد ما إذا كان الجزء العلوي أو السفلي من المصفوفة المتقارنة C
!>          مُخزَّنًا.
!>          = 'U':  الجزء العلوي
!>          = 'L':  الجزء السفلي
!>

N

!>          N هو INTEGER
!>          عدد الصفوف والأعمدة للمصفوفة C. N >= 0.
!>

V

!>          V هو مصفوفة COMPLEX، ذات بُعد
!>                  (1 + (N-1)*abs(INCV))
!>          المتجهة v كما هو موضح أعلاه.
!>

INCV

!>          INCV هو INTEGER
!>          الزيادة بين العناصر المتتالية لـ v. يجب ألا يكون INCV
!>          يساوي صفرًا.
!>

TAU

!>          TAU هو COMPLEX
!>          القيمة tau كما هو موضح أعلاه.
!>

C

!>          C هي مصفوفة COMPLEX، ذات بُعد (LDC, N)
!>          عند الدخول، المصفوفة C.
!>          عند الخروج، يتم استبدال C بـ H * C * H'.
!>

LDC

!>          LDC هو INTEGER
!>          البُعد الرائد للمصفوفة C. LDC >= max( 1, N ).
!>

WORK

!>          WORK هي مصفوفة COMPLEX، ذات بُعد (N)
!>
