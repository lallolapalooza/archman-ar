الوظائف/الروتينات الفرعية


real function cla_syrcond_c (uplo, n, a, lda, af, ldaf, ipiv, c, capply, info, work, rwork)
CLA_SYRCOND_C يحسب رقم شرط المعيار اللانهائي لـ op(A)*inv(diag(c)) للمصفوفات المتناظرة غير المحددة.

وثائق الوظيفة/الروتين الفرعي

real function cla_syrcond_c (character uplo, integer n, complex, dimension( lda, * ) a, integer lda, complex, dimension( ldaf, * ) af, integer ldaf, integer, dimension( * ) ipiv, real, dimension( * ) c, logical capply, integer info, complex, dimension( * ) work, real, dimension( * ) rwork)

CLA_SYRCOND_C يحسب رقم شرط المعيار اللانهائي لـ op(A)*inv(diag(c)) للمصفوفات المتناظرة غير المحددة.

الغرض:

!>
!>    CLA_SYRCOND_C يحسب رقم شرط المعيار اللانهائي لـ
!>    op(A) * inv(diag(C)) حيث C هو متجه حقيقي.
!>

المعلمات

UPLO 
!>          UPLO هو CHARACTER*1
!>       = 'U':  يتم تخزين المثلث العلوي لـ A؛
!>       = 'L':  يتم تخزين المثلث السفلي لـ A.
!>

N

!>          N هو INTEGER
!>     عدد المعادلات الخطية، أي ترتيب
!>     المصفوفة A.  N >= 0.
!>

A

!>          A هو COMPLEX array, dimension (LDA,N)
!>     عند الإدخال، المصفوفة A بحجم N*N
!>

LDA

!>          LDA هو INTEGER
!>     البعد الرائد للمصفوفة A.  LDA >= max(1,N).
!>

AF

!>          AF هو COMPLEX array, dimension (LDAF,N)
!>     المصفوفة القطرية الجزئية D والمعاملات المستخدمة
!>     للحصول على العامل U أو L كما تم حسابه بواسطة CSYTRF.
!>

LDAF

!>          LDAF هو INTEGER
!>     البعد الرائد للمصفوفة AF.  LDAF >= max(1,N).
!>

IPIV

!>          IPIV هو INTEGER array, dimension (N)
!>     تفاصيل التبادلات والبنية الجزئية لـ D
!>     كما تم تحديدها بواسطة CSYTRF.
!>

C

!>          C هو REAL array, dimension (N)
!>     المتجه C في الصيغة op(A) * inv(diag(C)).
!>

CAPPLY

!>          CAPPLY هو LOGICAL
!>     إذا كان .TRUE. ثم الوصول إلى المتجه C في الصيغة أعلاه.
!>

INFO

!>          INFO هو INTEGER
!>       = 0:  خروج ناجح.
!>     i > 0:  المعلمة i غير صالحة.
!>

WORK

!>          WORK هو COMPLEX array, dimension (2*N).
!>     مساحة العمل.
!>

RWORK

!>          RWORK هو REAL array, dimension (N).
!>     مساحة العمل.
!>
