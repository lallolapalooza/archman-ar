real function clansy (norm, uplo, n, a, lda, work)

  CLANSY تُرجع قيمة القاعدة 1، أو قاعدة فروبنيوس، أو القاعدة اللانهائية، أو
    العنصر ذي القيمة المطلقة الأكبر لمصفوفة مركبة متماثلة.

توثيق الوظيفة/الروتين الفرعي

real function clansy (character norm, character uplo, integer n, complex, dimension( lda, * ) a, integer lda, real, dimension( * ) work)

CLANSY تُرجع قيمة القاعدة 1، أو قاعدة فروبنيوس، أو القاعدة اللانهاائية، أو العنصر ذي القيمة المطلقة الأكبر لمصفوفة مركبة متماثلة.

الغرض:

!>
!> CLANSY  تُرجع قيمة القاعدة واحد،  أو قاعدة فروبنيوس،  أو
!> القاعدة اللانهائية،  أو  العنصر ذي القيمة المطلقة الأكبر لمصفوفة
!> مركبة متماثلة A.
!>

النتائج

CLANSY 
!>
!>    CLANSY = ( max(abs(A(i,j))), NORM = 'M' or 'm'
!>             (
!>             ( norm1(A),         NORM = '1', 'O' or 'o'
!>             (
!>             ( normI(A),         NORM = 'I' or 'i'
!>             (
!>             ( normF(A),         NORM = 'F', 'f', 'E' or 'e'
!>
!> حيث  norm1  تشير إلى  القاعدة واحد لمصفوفة (أكبر مجموع عمود)،
!> normI  تشير إلى  القاعدة اللانهائية  لمصفوفة  (أكبر مجموع صف) و
!> normF  تشير إلى  قاعدة فروبنيوس  لمصفوفة  (الجذر التربيعي لمجموع
!> المربعات).  لاحظ أن  max(abs(A(i,j)))  ليست قاعدة مصفوفة متسقة.
!>

المعلمات

NORM 
!>          NORM is CHARACTER*1
!>          تحدد القيمة المراد إرجاعها في CLANSY كما هو موضح
!>          أعلاه.
!>

UPLO

!>          UPLO is CHARACTER*1
!>          تحدد ما إذا كان سيتم الإشارة إلى الجزء العلوي أو السفلي
!>          المثلثي للمصفوفة المتماثلة A.
!>          = 'U':  يتم الإشارة إلى الجزء العلوي المثلثي من A
!>          = 'L':  يتم الإشارة إلى الجزء السفلي المثلثي من A
!>

N

!>          N is INTEGER
!>          رتبة المصفوفة A.  N >= 0.  عندما N = 0، يتم تعيين CLANSY
!>          إلى صفر.
!>

A

!>          A is COMPLEX array, dimension (LDA,N)
!>          المصفوفة المتماثلة A.  إذا كان UPLO = 'U'، فإن الجزء
!>          العلوي المثلثي من A الذي حجمه n x n يحتوي على الجزء
!>          العلوي المثلثي للمصفوفة A، ولا تتم الإشارة إلى الجزء
!>          المثلثي السفلي الصارم من A.  إذا كان UPLO = 'L'، فإن الجزء
!>          السفلي المثلثي من A الذي حجمه n x n يحتوي على الجزء
!>          السفلي المثلثي للمصفوفة A، ولا تتم الإشارة إلى الجزء
!>          المثلثي العلوي الصارم من A.
!>

LDA

!>          LDA is INTEGER
!>          البعد الرائد للمصفوفة A.  LDA >= max(N,1).
!>

WORK

!>          WORK is REAL array, dimension (MAX(1,LWORK)),
!>          حيث LWORK >= N عندما NORM = 'I' أو '1' أو 'O'. بخلاف ذلك،
!>          لا تتم الإشارة إلى WORK.
!>
