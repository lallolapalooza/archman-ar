real function cla_gbrcond_c (trans, n, kl, ku, ab, ldab, afb, ldafb, ipiv, c, capply, info, work, rwork)
CLA_GBRCOND_C تحسب رقم الشرط للمعيار اللانهائي لـ op(A)*inv(diag(c)) للمصفوفات النطاقية العامة.

توثيق الوظيفة/الروتين الفرعي

real function cla_gbrcond_c (character trans, integer n, integer kl, integer ku, complex, dimension( ldab, * ) ab, integer ldab, complex, dimension( ldafb, * ) afb, integer ldafb, integer, dimension( * ) ipiv, real, dimension( * ) c, logical capply, integer info, complex, dimension( * ) work, real, dimension( * ) rwork)

CLA_GBRCOND_C تحسب رقم الشرط للمعيار اللانهائي لـ op(A)*inv(diag(c)) للمصفوفات النطاقية العامة.

الغرض:

!>
!>    CLA_GBRCOND_C تحسب رقم الشرط للمعيار اللانهائي لـ
!>    op(A) * inv(diag(C)) حيث C هو متجه حقيقي.
!>

المعاملات

TRANS 
!>          TRANS is CHARACTER*1
!>     يحدد شكل نظام المعادلات:
!>       = 'N':  A * X = B     (بدون تبديل)
!>       = 'T':  A**T * X = B  (تبديل)
!>       = 'C':  A**H * X = B  (تبديل مترافق = تبديل)
!>

N

!>          N is INTEGER
!>     عدد المعادلات الخطية، أي رتبة
!>     المصفوفة A.  N >= 0.
!>

KL

!>          KL is INTEGER
!>     عدد الأقطار الفرعية داخل نطاق A.  KL >= 0.
!>

KU

!>          KU is INTEGER
!>     عدد الأقطار الفائقة داخل نطاق A.  KU >= 0.
!>

AB

!>          AB is COMPLEX array, dimension (LDAB,N)
!>     عند الدخول، المصفوفة A بتخزين نطاقي، في الصفوف من 1 إلى KL+KU+1.
!>     يتم تخزين العمود j من A في العمود j من
!>     المصفوفة AB كما يلي:
!>     AB(KU+1+i-j,j) = A(i,j) لـ max(1,j-KU)<=i<=min(N,j+kl)
!>

LDAB

!>          LDAB is INTEGER
!>     البعد الرائد للمصفوفة AB.  LDAB >= KL+KU+1.
!>

AFB

!>          AFB is COMPLEX array, dimension (LDAFB,N)
!>     تفاصيل تحلل LU للمصفوفة النطاقية A، كما
!>     تم حسابها بواسطة CGBTRF.  U مخزنة كمصفوفة نطاقية مثلثة عليا
!>     مع KL+KU أقطار فائقة في الصفوف من 1 إلى KL+KU+1،
!>     ويتم تخزين المضاعفات المستخدمة أثناء التحلل
!>     في الصفوف من KL+KU+2 إلى 2*KL+KU+1.
!>

LDAFB

!>          LDAFB is INTEGER
!>     البعد الرائد للمصفوفة AFB.  LDAFB >= 2*KL+KU+1.
!>

IPIV

!>          IPIV is INTEGER array, dimension (N)
!>     مؤشرات المحور من تحلل A = P*L*U
!>     كما تم حسابها بواسطة CGBTRF؛ تم تبديل الصف i من المصفوفة
!>     مع الصف IPIV(i).
!>

C

!>          C is REAL array, dimension (N)
!>     المتجه C في الصيغة op(A) * inv(diag(C)).
!>

CAPPLY

!>          CAPPLY is LOGICAL
!>     إذا كانت .TRUE.، قم بالوصول إلى المتجه C في الصيغة أعلاه.
!>

INFO

!>          INFO is INTEGER
!>       = 0:  خروج ناجح.
!>     i > 0:  المعامل i غير صالح.
!>

WORK

!>          WORK is COMPLEX array, dimension (2*N).
!>     مساحة عمل.
!>

RWORK

!>          RWORK is REAL array, dimension (N).
!>     مساحة عمل.
!>
