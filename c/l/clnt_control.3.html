rpc(3) دليل وظائف المكتبة rpc(3)

الاسم

rpc - روتينات مكتبة لاستدعاءات الإجراءات عن بعد

المكتبة

مكتبة C القياسية (libc، -lc)

الملخص والوصف

تسمح هذه الروتينات لبرامج C بإجراء استدعاءات إجراءات على أجهزة أخرى عبر الشبكة. أولاً، يستدعي العميل إجراءً لإرسال حزمة بيانات إلى الخادم. عند استلام الحزمة، يستدعي الخادم روتين توزيع لأداء الخدمة المطلوبة، ثم يرسل ردًا. أخيرًا، يعود استدعاء الإجراء إلى العميل.

للاستفادة من هذه الروتينات، قم بتضمين ملف الرأس <rpc/rpc.h>.

تعريفات البروتوتايب أدناه تستخدم الأنواع التالية:

typedef int bool_t;
typedef typeof(bool_t (XDR *, void *, ...))  *xdrproc_t;
typedef typeof(bool_t (caddr_t resp, struct sockaddr_in *raddr)
    *resultproc_t;

راجع ملفات الرأس لتعريفات أنواع AUTH، CLIENT، SVCXPRT، و XDR.

void auth_destroy(AUTH *auth);
ماكرو يقوم بتدمير معلومات المصادقة المرتبطة بـ auth. يتضمن التدمير عادةً تخصيص هياكل البيانات الخاصة. استخدام auth غير محدد بعد استدعاء auth_destroy().
AUTH *authnone_create(void);
إنشاء وإرجاع مقبض مصادقة RPC يمرر معلومات مصادقة غير قابلة للاستخدام مع كل استدعاء إجراء عن بعد. هذه هي المصادقة الافتراضية التي يستخدمها RPC.
AUTH *authunix_create(char *host, uid_t uid, gid_t gid,
                      int n, gid_t aup_gids[n]);
إنشاء وإرجاع مقبض مصادقة RPC يحتوي على معلومات المصادقة. المعلمة host هي اسم الجهاز الذي تم إنشاء المعلومات عليه؛ uid هو معرّف المستخدم؛ gid هو معرّف المجموعة الحالي للمستخدم؛ n و aup_gids تشير إلى مصفوفة محسوبة للمجموعات التي ينتمي إليها المستخدم. من السهل انتحال شخصية مستخدم.
AUTH *authunix_create_default(void);
يستدعي authunix_create() بالمعلمات المناسبة.
int callrpc(char *host, unsigned long prognum,
            unsigned long versnum, unsigned long procnum,
            xdrproc_t inproc, const char *in,
            xdrproc_t outproc, char *out);
استدعاء الإجراء البعيد المرتبط بـ prognum و versnum و procnum على الجهاز host. المعلمة in هي عنوان الوسيط (الوسائط) للإجراء، و out هو العنوان الذي سيتم فيه وضع النتيجة (النتائج)؛ يتم استخدام inproc لتشفير معاملات الإجراء، ويتم استخدام outproc لفك تشفير نتائج الإجراء؛ تعيد هذه الروتينات الصفر إذا نجحت، أو قيمة enum clnt_stat محوّلة إلى عدد صحيح إذا فشلت. الروتين clnt_perrno() مفيد لترجمة حالات الفشل إلى رسائل.
تحذير: استدعاء الإجراءات البعيدة بهذا الروتين يستخدم UDP/IP كوسيلة نقل؛ انظر clntudp_create() للقيود. ليس لديك تحكم في المهلات أو المصادقة باستخدام هذا الروتين.
enum clnt_stat clnt_broadcast(unsigned long prognum,
                     unsigned long versnum, unsigned long procnum,
                     xdrproc_t inproc, char *in,
                     xdrproc_t outproc, char *out,
                     resultproc_t eachresult);
مثل callrpc()، باستثناء أن رسالة الاستدعاء يتم بثها إلى جميع شبكات البث المتصلة محليًا. في كل مرة تتلقى فيها ردًا، يستدعي هذا الروتين eachresult()، والذي يكون شكله:

eachresult(char *out, struct sockaddr_in *addr);

حيث out هو نفس out الذي تم تمريره إلى clnt_broadcast()، باستثناء أنه تم فك تشفير مخرجات الإجراء البعيد هناك؛ addr يشير إلى عنوان الجهاز الذي أرسل النتائج. إذا أعاد eachresult() صفرًا، فإن clnt_broadcast() ينتظر المزيد من الردود؛ بخلاف ذلك، يعود بالحالة المناسبة.
تحذير: مقابس البث محدودة في الحجم بأقصى وحدة نقل للارتباط البيانات. بالنسبة للإيثرنت، تبلغ هذه القيمة 1500 بايت.
enum clnt_stat clnt_call(CLIENT *clnt, unsigned long procnum,
                    xdrproc_t inproc, char *in,
                    xdrproc_t outproc, char *out,
                    struct timeval tout);
ماكرو يستدعي الإجراء البعيد procnum المرتبط بمقبض العميل clnt، والذي تم الحصول عليه باستخدام روتين إنشاء عميل RPC مثل clnt_create(). المعلمة in هي عنوان الوسيط (الوسائط) للإجراء، و out هو العنوان الذي سيتم فيه وضع النتيجة (النتائج)؛ يتم استخدام inproc لتشفير معاملات الإجراء، ويتم استخدام outproc لفك تشفير نتائج الإجراء؛ tout هو الوقت المسموح به لعودة النتائج.
clnt_destroy(CLIENT *clnt);
ماكرو يقوم بتدمير مقبض RPC الخاص بالعميل. يتضمن التدمير عادةً تخصيص هياكل البيانات الخاصة، بما في ذلك clnt نفسه. استخدام clnt غير محدد بعد استدعاء clnt_destroy(). إذا قامت مكتبة RPC بفتح المقبس المرتبط، فسيتم إغلاقه أيضًا. بخلاف ذلك، يظل المقبس مفتوحًا.
CLIENT *clnt_create(const char *host, unsigned long prog,
                    unsigned long vers, const char *proto);
روتين إنشاء عميل عام. host يحدد اسم المضيف البعيد حيث يوجد الخادم. proto يشير إلى نوع بروتوكول النقل المراد استخدامه. القيم المدعومة حاليًا لهذا الحقل هي "udp" و "tcp". يتم تعيين مهلات افتراضية، ولكن يمكن تعديلها باستخدام clnt_control().
تحذير: استخدام UDP له عيوبه. نظرًا لأن رسائل RPC المستندة إلى UDP يمكن أن تحتوي على 8 كيلوبايت فقط من البيانات المشفرة، فلا يمكن استخدام وسيلة النقل هذه للإجراءات التي تتلقى وسائط كبيرة أو تعيد نتائج ضخمة.
bool_t clnt_control(CLIENT *cl, int req, char *info);
ماكرو يستخدم لتغيير أو استرداد معلومات مختلفة حول كائن العميل. req يشير إلى نوع العملية، و info هو مؤشر إلى المعلومات. لكل من UDP و TCP، القيم المدعومة لـ req وأنواع معلماتها وما تفعله هي:

CLSET_TIMEOUT  struct timeval // تعيين المهلة الإجمالية
CLGET_TIMEOUT  struct timeval // الحصول على المهلة الإجمالية

ملاحظة: إذا قمت بتعيين المهلة باستخدام clnt_control()، فسيتم تجاهل معلمة المهلة التي تم تمريرها إلى clnt_call() في جميع الاستدعاءات المستقبلية.

CLGET_SERVER_ADDR  struct sockaddr_in
                // الحصول على عنوان الخادم

العمليات التالية صالحة لـ UDP فقط:

CLSET_RETRY_TIMEOUT  struct timeval // تعيين مهلة إعادة المحاولة
CLGET_RETRY_TIMEOUT  struct timeval // الحصول على مهلة إعادة المحاولة

مهلة إعادة المحاولة هي الوقت الذي ينتظره "UDP RPC" رد الخادم قبل إعادة إرسال الطلب.
clnt_freeres(CLIENT * clnt, xdrproc_t outproc, char *out);
ماكرو يقوم بتحرير أي بيانات تم تخصيصها بواسطة نظام RPC/XDR عند فك تشفير نتائج استدعاء RPC. المعلمة out هي عنوان النتائج، و outproc هو روتين XDR الذي يصف النتائج. تعيد هذه الروتينات واحدًا إذا تم تحرير النتائج بنجاح، وصفرًا بخلاف ذلك.
void clnt_geterr(CLIENT *clnt, struct rpc_err
