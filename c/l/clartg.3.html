subroutine clartg (complex(wp) f, complex(wp) g, real(wp) c, complex(wp) s, complex(wp) r)

CLARTG تولد دوران مستوي مع جيب تمام حقيقي وجيب مركب.

الغرض:

!>
!> CLARTG تولد دوران مستوي بحيث
!>
!>    [  C         S  ] . [ F ]  =  [ R ]
!>    [ -conjg(S)  C  ]   [ G ]     [ 0 ]
!>
!> حيث C حقيقي و C**2 + |S|**2 = 1.
!>
!> الصيغ الرياضية المستخدمة لـ C و S هي
!>
!>    sgn(x) = {  x / |x|,   x != 0
!>             {  1,         x  = 0
!>
!>    R = sgn(F) * sqrt(|F|**2 + |G|**2)
!>
!>    C = |F| / sqrt(|F|**2 + |G|**2)
!>
!>    S = sgn(F) * conjg(G) / sqrt(|F|**2 + |G|**2)
!>
!> حالات خاصة:
!>    إذا كان G=0، فإن C=1 و S=0.
!>    إذا كان F=0، فإن C=0 ويتم اختيار S بحيث يكون R حقيقيًا.
!>
!> عندما تكون F و G حقيقية، تبسط الصيغ إلى C = F/R و
!> S = G/R، ويجب أن تكون القيم المعادة لـ C و S و R متطابقة
!> مع تلك التي تعيدها SLARTG.
!>
!> تتضمن الخوارزمية المستخدمة لحساب هذه الكميات التحجيم
!> لتجنب الفيضان أو التدفق الناقص في حساب الجذر التربيعي لمجموع المربعات.
!>
!> هذه هي نفس روتين CROTG من BLAS1، باستثناء أن
!> F و G لا تتغيران عند العودة.
!>
!> أدناه، wp=>sp تعني الدقة الأحادية من وحدة LA_CONSTANTS.
!>

المعلمات

F 
!>          F هو COMPLEX(wp)
!>          المكون الأول للمتجه المراد تدويره.
!>

G

!>          G هو COMPLEX(wp)
!>          المكون الثاني للمتجه المراد تدويره.
!>

C

!>          C هو REAL(wp)
!>          جيب تمام الدوران.
!>

S

!>          S هو COMPLEX(wp)
!>          جيب الدوران.
!>

R

!>          R هو COMPLEX(wp)
!>          المكون غير الصفري للمتجه المدور.
!>
