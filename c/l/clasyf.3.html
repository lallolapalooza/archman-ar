subroutine clasyf (uplo, n, nb, kb, a, lda, ipiv, w, ldw, info)
CLASYF تحسب تجزئة جزئية لمصفوفة متناظرة مركبة باستخدام طريقة التنقيب القطري لـ Bunch-Kaufman.

توثيق الوظيفة/الإجراء الفرعي

subroutine clasyf (character uplo, integer n, integer nb, integer kb, complex, dimension( lda, * ) a, integer lda, integer, dimension( * ) ipiv, complex, dimension( ldw, * ) w, integer ldw, integer info)

CLASYF تحسب تجزئة جزئية لمصفوفة متناظرة مركبة باستخدام طريقة التنقيب القطري لـ Bunch-Kaufman.

الغرض:

!>
!> CLASYF تحسب تجزئة جزئية لمصفوفة متناظرة مركبة
!> A باستخدام طريقة التنقيب القطري لـ Bunch-Kaufman. التجزئة الجزئية
!> تأخذ الشكل:
!>
!> A  =  ( I  U12 ) ( A11  0  ) (  I       0    )  إذا كان UPLO = 'U', أو:
!>       ( 0  U22 ) (  0   D  ) ( U12**T U22**T )
!>
!> A  =  ( L11  0 ) ( D    0  ) ( L11**T L21**T )  إذا كان UPLO = 'L'
!>       ( L21  I ) ( 0   A22 ) (  0       I    )
!>
!> حيث أن ترتيب D لا يتجاوز NB. يتم إرجاع الترتيب الفعلي في
!> المتغير KB، وهو إما NB أو NB-1، أو N إذا كان N <= NB.
!> لاحظ أن U**T تمثل منقول U.
!>
!> CLASYF هي روتين مساعد يتم استدعاؤه بواسطة CSYTRF. تستخدم رمزًا محظورًا
!> (استدعاء BLAS المستوى 3) لتحديث المصفوفة الفرعية A11 (إذا كان UPLO = 'U') أو
!> A22 (إذا كان UPLO = 'L').
!>

المعلمات

UPLO 
!>          UPLO هو CHARACTER*1
!>          يحدد ما إذا كان الجزء العلوي أو السفلي الثلاثي من المصفوفة المتناظرة A
!>          مخزنًا:
!>          = 'U':  مثلثي علوي
!>          = 'L':  مثلثي سفلي
!>

N

!>          N هو INTEGER
!>          ترتيب المصفوفة A.  N >= 0.
!>

NB

!>          NB هو INTEGER
!>          الحد الأقصى لعدد أعمدة المصفوفة A التي يجب
!>          تجزئتها.  يجب أن يكون NB على الأقل 2 للسماح بكتل محورية 2x2.
!>

KB

!>          KB هو INTEGER
!>          عدد أعمدة A التي تم تجزئتها فعليًا.
!>          KB هو إما NB-1 أو NB، أو N إذا كان N <= NB.
!>

A

!>          A هي مصفوفة COMPLEX، أبعادها (LDA,N)
!>          عند الدخول، المصفوفة المتناظرة A.  إذا كان UPLO = 'U', فإن الجزء
!>          العلوي الثلاثي nxn الرائد من A يحتوي على الجزء
!>          المثلثي العلوي للمصفوفة A، ولا يتم الرجوع إلى الجزء المثلثي
!>          السفلي الصارم من A.  إذا كان UPLO = 'L', فإن الجزء
!>          السفلي الثلاثي nxn الرائد من A يحتوي على الجزء
!>          المثلثي السفلي للمصفوفة A، ولا يتم الرجوع إلى الجزء المثلثي
!>          العلوي الصارم من A.
!>          عند الخروج، تحتوي A على تفاصيل التجزئة الجزئية.
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرائد للمصفوفة A.  LDA >= max(1,N).
!>

IPIV

!>          IPIV هي مصفوفة INTEGER، أبعادها (N)
!>          تفاصيل التبادلات وهيكل الكتلة لـ D.
!>
!>          إذا كان UPLO = 'U':
!>             يتم تعيين آخر KB عنصر فقط من IPIV.
!>
!>             إذا كان IPIV(k) > 0, فقد تم تبديل الصفوف والأعمدة k و IPIV(k)
!>             و D(k,k) هي كتلة قطرية 1x1.
!>
!>             إذا كان IPIV(k) = IPIV(k-1) < 0, فقد تم تبديل الصفوف والأعمدة
!>             k-1 و -IPIV(k) و D(k-1:k,k-1:k)
!>             هي كتلة قطرية 2x2.
!>
!>          إذا كان UPLO = 'L':
!>             يتم تعيين أول KB عنصر فقط من IPIV.
!>
!>             إذا كان IPIV(k) > 0, فقد تم تبديل الصفوف والأعمدة k و IPIV(k)
!>             و D(k,k) هي كتلة قطرية 1x1.
!>
!>             إذا كان IPIV(k) = IPIV(k+1) < 0, فقد تم تبديل الصفوف والأعمدة
!>             k+1 و -IPIV(k) و D(k:k+1,k:k+1)
!>             هي كتلة قطرية 2x2.
!>

W

!>          W هي مصفوفة COMPLEX، أبعادها (LDW,NB)
!>

LDW

!>          LDW هو INTEGER
!>          البعد الرائد للمصفوفة W.  LDW >= max(1,N).
!>

INFO

!>          INFO هو INTEGER
!>          = 0: خروج ناجح
!>          > 0: إذا كان INFO = k, فإن D(k,k) يساوي صفرًا تمامًا.  تم
!>               إكمال التجزئة، ولكن المصفوفة القطرية D
!>               متفردة تمامًا.
!>
