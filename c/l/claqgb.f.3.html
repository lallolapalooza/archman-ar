توثيق الوظيفة/الإجراء الفرعي

subroutine claqgb (integer m, integer n, integer kl, integer ku, complex, dimension( ldab, * ) ab, integer ldab, real, dimension( * ) r, real, dimension( * ) c, real rowcnd, real colcnd, real amax, character equed)

CLAQGB تقوم بقياس مصفوفة نطاق عامة، باستخدام عوامل قياس الصف والعمود التي تم حسابها بواسطة sgbequ.

الغرض:

!>
!> CLAQGB توازن مصفوفة نطاق عامة M × N ذات KL قطر سفلي و KU قطر علوي باستخدام عوامل القياس للصفوف والأعمدة في المتجهات R و C.
!>

المعلمات

M 
!>          M هو عدد صحيح (INTEGER)
!>          عدد صفوف المصفوفة A.  M >= 0.
!>

N

!>          N هو عدد صحيح (INTEGER)
!>          عدد أعمدة المصفوفة A.  N >= 0.
!>

KL

!>          KL هو عدد صحيح (INTEGER)
!>          عدد الأقطار الفرعية ضمن نطاق A.  KL >= 0.
!>

KU

!>          KU هو عدد صحيح (INTEGER)
!>          عدد الأقطار الفرعية العلوية ضمن نطاق A.  KU >= 0.
!>

AB

!>          AB هو مصفوفة مركبة (COMPLEX)، ذات بعد (LDAB,N)
!>          عند الإدخال، المصفوفة A بتخزين النطاق، في الصفوف من 1 إلى KL+KU+1.
!>          يتم تخزين العمود j من A في العمود j من المصفوفة
!>          AB كالتالي:
!>          AB(ku+1+i-j,j) = A(i,j) لـ max(1,j-ku)<=i<=min(m,j+kl)
!>
!>          عند الخروج، المصفوفة المتوازنة، بنفس تنسيق التخزين
!>          مثل A. انظر EQUED لشكل المصفوفة المتوازنة.
!>

LDAB

!>          LDAB هو عدد صحيح (INTEGER)
!>          البعد الرائد للمصفوفة AB.  LDA >= KL+KU+1.
!>

R

!>          R هي مصفوفة حقيقية (REAL)، ذات بعد (M)
!>          عوامل قياس الصفوف للمصفوفة A.
!>

C

!>          C هي مصفوفة حقيقية (REAL)، ذات بعد (N)
!>          عوامل قياس الأعمدة للمصفوفة A.
!>

ROWCND

!>          ROWCND هو حقيقي (REAL)
!>          نسبة أصغر R(i) إلى أكبر R(i).
!>

COLCND

!>          COLCND هو حقيقي (REAL)
!>          نسبة أصغر C(i) إلى أكبر C(i).
!>

AMAX

!>          AMAX هو حقيقي (REAL)
!>          القيمة المطلقة لأكبر عنصر في المصفوفة.
!>

EQUED

!>          EQUED هو حرف واحد (CHARACTER*1)
!>          يحدد شكل الموازنة التي تم إجراؤها.
!>          = 'N':  لا توجد موازنة
!>          = 'R':  موازنة الصفوف، أي تم ضرب A مسبقًا بـ
!>                  diag(R).
!>          = 'C':  موازنة الأعمدة، أي تم ضرب A لاحقًا بـ
!>                  diag(C).
!>          = 'B':  موازنة الصفوف والأعمدة، أي تم استبدال A بـ
!>                  diag(R) * A * diag(C).
!>

المعلمات الداخلية:

!>  THRESH هو قيمة حدية تستخدم لتحديد ما إذا كان يجب إجراء قياس الصف أو العمود بناءً على نسبة عوامل قياس الصف أو العمود.
!>  إذا كان ROWCND < THRESH، يتم إجراء قياس الصف، وإذا كان
!>  COLCND < THRESH، يتم إجراء قياس العمود.
!>
!>  LARGE و SMALL هي قيم حدية تستخدم لتحديد ما إذا كان يجب إجراء قياس الصف بناءً على الحجم المطلق لأكبر عنصر في المصفوفة.
!>  إذا كان AMAX > LARGE أو AMAX < SMALL، يتم إجراء قياس الصف.
!>
