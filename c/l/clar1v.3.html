subroutine clar1v (integer n, integer b1, integer bn, real lambda, real, dimension( * ) d, real, dimension( * ) l, real, dimension( * ) ld, real, dimension( * ) lld, real pivmin, real gaptol, complex, dimension( * ) z, logical wantnc, integer negcnt, real ztz, real mingma, integer r, integer, dimension( * ) isuppz, real nrminv, real resid, real rqcorr, real, dimension( * ) work)

CLAR1V يحسب العمود r (المُقاس) للمصفوفة العكسية للمصفوفة الفرعية في الصفوف b1 حتى bn للمصفوفة الثلاثية LDLT - λI.

الغرض:

!>
!> CLAR1V يحسب العمود r (المُقاس) للمصفوفة العكسية للمصفوفة الفرعية
!> في الصفوف B1 حتى BN للمصفوفة الثلاثية L D L**T - sigma I. عندما تكون sigma قريبة من قيمة ذاتية،
!> يكون المتجه المحسوب متجهًا ذاتيًا دقيقًا. عادةً، يتوافق r
!> مع الفهرس حيث يكون المتجه الذاتي هو الأكبر من حيث القيمة المطلقة.
!> الخطوات التالية تحقق هذه الحسابات:
!> (أ) تحويل qd الثابت، L D L**T - sigma I = L(+) D(+) L(+)**T،
!> (ب) تحويل qd التقدمي، L D L**T - sigma I = U(-) D(-) U(-)**T،
!> (ج) حساب العناصر القطرية للمصفوفة العكسية لـ
!>     L D L**T - sigma I من خلال دمج التحويلات المذكورة أعلاه، واختيار
!>     r كفهرس حيث يكون قطر المصفوفة العكسية (واحد من)
!>     الأكبر من حيث القيمة المطلقة.
!> (د) حساب العمود r (المُقاس) للمصفوفة العكسية باستخدام
!>     التحليل الملتوي الناتج عن دمج الجزء العلوي من
!>     التحويل الثابت والجزء السفلي من التحويل التقدمي.
!>

المعلمات

N 
!>          N هو عدد صحيح
!>           ترتيب المصفوفة L D L**T.
!>

B1

!>          B1 هو عدد صحيح
!>           الفهرس الأول للمصفوفة الفرعية لـ L D L**T.
!>

BN

!>          BN هو عدد صحيح
!>           الفهرس الأخير للمصفوفة الفرعية لـ L D L**T.
!>

LAMBDA

!>          LAMBDA هو عدد حقيقي
!>           الانحراف. لحساب متجه ذاتي دقيق،
!>           يجب أن يكون LAMBDA تقريبًا جيدًا لقيمة ذاتية
!>           لـ L D L**T.
!>

L

!>          L هو مصفوفة حقيقية، بأبعاد (N-1)
!>           العناصر الفرعية القطرية (n-1) للمصفوفة الأحادية القطرية
!>           L، في العناصر 1 إلى N-1.
!>

D

!>          D هو مصفوفة حقيقية، بأبعاد (N)
!>           العناصر القطرية n للمصفوفة القطرية D.
!>

LD

!>          LD هو مصفوفة حقيقية، بأبعاد (N-1)
!>           العناصر L(i)*D(i) بمقدار n-1.
!>

LLD

!>          LLD هو مصفوفة حقيقية، بأبعاد (N-1)
!>           العناصر L(i)*L(i)*D(i) بمقدار n-1.
!>

PIVMIN

!>          PIVMIN هو عدد حقيقي
!>           الارتكاز الأدنى في تسلسل ستورم.
!>

GAPTOL

!>          GAPTOL هو عدد حقيقي
!>           التفاوت الذي يشير إلى متى تكون مدخلات المتجه الذاتي ضئيلة
!>           فيما يتعلق بمساهمتها في الباقي.
!>

Z

!>          Z هي مصفوفة مركبة، بأبعاد (N)
!>           عند الإدخال، يجب تعيين جميع إدخالات Z إلى 0.
!>           عند الإخراج، تحتوي Z على العمود r (المُقاس) للمصفوفة العكسية.
!>           القياس هو أن Z(R) تساوي 1.
!>

WANTNC

!>          WANTNC هو منطقي
!>           يحدد ما إذا كان يجب حساب NEGCNT.
!>

NEGCNT

!>          NEGCNT هو عدد صحيح
!>           إذا كان WANTNC هو .TRUE. فإن NEGCNT = عدد الارتكازات < pivmin
!>           في تحليل المصفوفة L D L**T، و NEGCNT = -1 بخلاف ذلك.
!>

ZTZ

!>          ZTZ هو عدد حقيقي
!>           مربع المعيار الثنائي لـ Z.
!>

MINGMA

!>          MINGMA هو عدد حقيقي
!>           مقلوب أكبر عنصر قطري (من حيث القيمة المطلقة)
!>           للمصفوفة العكسية لـ L D L**T - sigma I.
!>

R

!>          R هو عدد صحيح
!>           فهرس الالتواء للتحليل الملتوي المستخدم لحساب
!>           Z.
!>           عند الإدخال، 0 <= R <= N. إذا تم إدخال R كـ 0، يتم تعيين R
!>           إلى الفهرس حيث تكون (L D L**T - sigma I)^{-1} أكبر
!>           من حيث القيمة المطلقة. إذا كان 1 <= R <= N، يبقى R دون تغيير.
!>           عند الإخراج، يحتوي R على فهرس الالتواء المستخدم لحساب Z.
!>           من الناحية المثالية، يشير R إلى موضع الحد الأقصى للمدخل في
!>           المتجه الذاتي.
!>

ISUPPZ

!>          ISUPPZ هي مصفوفة أعداد صحيحة، بأبعاد (2)
!>           دعم المتجه في Z، أي أن المتجه Z
!>           غير صفري فقط في العناصر ISUPPZ(1) حتى ISUPPZ( 2 ).
!>

NRMINV

!>          NRMINV هو عدد حقيقي
!>           NRMINV = 1/SQRT( ZTZ )
!>

RESID

!>          RESID هو عدد حقيقي
!>           باقي المتجه FP.
!>           RESID = ABS( MINGMA )/SQRT( ZTZ )
!>

RQCORR

!>          RQCORR هو عدد حقيقي
!>           تصحيح النسبة الريليه لـ LAMBDA.
!>           RQCORR = MINGMA*TMP
!>

WORK

!>          WORK هي مصفوفة حقيقية، بأبعاد (4*N)
!>
