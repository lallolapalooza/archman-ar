الوظيفة الحقيقية cla_gbrcond_x (trans, n, kl, ku, ab, ldab, afb, ldafb, ipiv, x, info, work, rwork)
CLA_GBRCOND_X تحسب رقم شرط المعيار اللانهائي لـ op(A)*diag(x) للمصفوفات الشريطية العامة.

توثيق الوظيفة/الروتين الفرعي

الوظيفة الحقيقية cla_gbrcond_x (trans حرف، n عدد صحيح، kl عدد صحيح، ku عدد صحيح، complex، dimension(ldab, *) ab، ldab عدد صحيح، complex، dimension(ldafb, *) afb، ldafb عدد صحيح، integer، dimension( *) ipiv، complex، dimension( *) x، info عدد صحيح، complex، dimension( *) work، real، dimension( *) rwork)

CLA_GBRCOND_X تحسب رقم شرط المعيار اللانهائي لـ op(A)*diag(x) للمصفوفات الشريطية العامة.

الغرض:

!>
!>    CLA_GBRCOND_X تحسب رقم شرط المعيار اللانهائي لـ
!>    op(A) * diag(X) حيث X هو متجه COMPLEX.
!>

المعلمات

TRANS 
!>          TRANS هو CHARACTER*1
!>     يحدد شكل نظام المعادلات:
!>       = 'N':  A * X = B     (بدون تبديل)
!>       = 'T':  A**T * X = B  (تبديل)
!>       = 'C':  A**H * X = B  (تبديل مرافق = تبديل)
!>

N

!>          N هو INTEGER
!>     عدد معادلات الخطية، أي، ترتيب
!>     المصفوفة A.  N >= 0.
!>

KL

!>          KL هو INTEGER
!>     عدد الأقطار الفرعية داخل نطاق A.  KL >= 0.
!>

KU

!>          KU هو INTEGER
!>     عدد الأقطار الفوقية داخل نطاق A.  KU >= 0.
!>

AB

!>          AB هو مصفوفة COMPLEX، dimension (LDAB,N)
!>     عند الإدخال، المصفوفة A بتخزين شريطي، في الصفوف من 1 إلى KL+KU+1.
!>     العمود j للمصفوفة A مخزن في العمود j للمصفوفة
!>     AB كما يلي:
!>     AB(KU+1+i-j,j) = A(i,j) لأجل max(1,j-KU)<=i<=min(N,j+kl)
!>

LDAB

!>          LDAB هو INTEGER
!>     البعد الرائد للمصفوفة AB.  LDAB >= KL+KU+1.
!>

AFB

!>          AFB هو مصفوفة COMPLEX، dimension (LDAFB,N)
!>     تفاصيل تحليل LU للمصفوفة الشريطية A، كما
!>     تم حسابها بواسطة CGBTRF.  U مخزن كمصفوفة
!>     مثلثة عليا شريطية مع KL+KU أقطار فوقية في الصفوف من 1 إلى KL+KU+1،
!>     والعوامل المضروبة المستخدمة أثناء التحليل مخزنة
!>     في الصفوف من KL+KU+2 إلى 2*KL+KU+1.
!>

LDAFB

!>          LDAFB هو INTEGER
!>     البعد الرائد للمصفوفة AFB.  LDAFB >= 2*KL+KU+1.
!>

IPIV

!>          IPIV هو مصفوفة INTEGER، dimension (N)
!>     مؤشرات المحور من تحليل A = P*L*U
!>     كما تم حسابها بواسطة CGBTRF؛ تم تبديل الصف i للمصفوفة
!>     مع الصف IPIV(i).
!>

X

!>          X هو مصفوفة COMPLEX، dimension (N)
!>     المتجه X في الصيغة op(A) * diag(X).
!>

INFO

!>          INFO هو INTEGER
!>       = 0:  انتهاء ناجح.
!>     i > 0:  المعلمة i غير صالحة.
!>

WORK

!>          WORK هي مصفوفة COMPLEX، dimension (2*N).
!>     مساحة عمل.
!>

RWORK

!>          RWORK هي مصفوفة REAL، dimension (N).
!>     مساحة عمل.
!>
