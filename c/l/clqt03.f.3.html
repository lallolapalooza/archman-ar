توثيق الدالة/الروتين الفرعي

subroutine clqt03 (integer m, integer n, integer k, complex, dimension( lda, * ) af, complex, dimension( lda, * ) c, complex, dimension( lda, * ) cc, complex, dimension( lda, * ) q, integer lda, complex, dimension( * ) tau, complex, dimension( lwork ) work, integer lwork, real, dimension( * ) rwork, real, dimension( * ) result)

CLQT03

الغرض:

!>
!> CLQT03 تختبر CUNMLQ، والتي تحسب Q*C، Q'*C، C*Q أو C*Q'.
!>
!> CLQT03 تقارن نتائج استدعاء CUNMLQ مع نتائج
!> تكوين Q بشكل صريح عن طريق استدعاء CUNGLQ ثم إجراء ضرب المصفوفات
!> عن طريق استدعاء CGEMM.
!>

المعلمات

M 
!>          M هو عدد صحيح (INTEGER)
!>          عدد الصفوف أو الأعمدة للمصفوفة C؛ C هي n×m إذا
!>          تم تطبيق Q من اليسار، أو m×n إذا تم تطبيق Q من
!>          اليمين.  M >= 0.
!>

N

!>          N هو عدد صحيح (INTEGER)
!>          رتبة المصفوفة المتعامدة Q.  N >= 0.
!>

K

!>          K هو عدد صحيح (INTEGER)
!>          عدد الانعكاسات الأولية التي يشكل حاصل ضربها المصفوفة المتعامدة
!>          Q.  N >= K >= 0.
!>

AF

!>          AF هو مصفوفة مركبة (COMPLEX)، البعد (LDA,N)
!>          تفاصيل تحليل LQ لمصفوفة m×n، كما
!>          تم إرجاعها بواسطة CGELQF. انظر CGELQF لمزيد من التفاصيل.
!>

C

!>          C هي مصفوفة مركبة (COMPLEX)، البعد (LDA,N)
!>

CC

!>          CC هي مصفوفة مركبة (COMPLEX)، البعد (LDA,N)
!>

Q

!>          Q هي مصفوفة مركبة (COMPLEX)، البعد (LDA,N)
!>

LDA

!>          LDA هو عدد صحيح (INTEGER)
!>          البعد الرائد للمصفوفات AF، C، CC، و Q.
!>

TAU

!>          TAU هي مصفوفة مركبة (COMPLEX)، البعد (min(M,N))
!>          العوامل القياسية للانعكاسات الأولية المقابلة
!>          لتحليل LQ في AF.
!>

WORK

!>          WORK هي مصفوفة مركبة (COMPLEX)، البعد (LWORK)
!>

LWORK

!>          LWORK هو عدد صحيح (INTEGER)
!>          طول WORK.  يجب أن يكون LWORK على الأقل M، ويجب أن يكون
!>          M*NB، حيث NB هو حجم الكتلة لهذه البيئة.
!>

RWORK

!>          RWORK هي مصفوفة حقيقية (REAL)، البعد (M)
!>

RESULT

!>          RESULT هي مصفوفة حقيقية (REAL)، البعد (4)
!>          تقارن نسب الاختبار تقنيتين لضرب مصفوفة عشوائية C
!>          بمصفوفة متعامدة n×n، Q.
!>          RESULT(1) = norm( Q*C - Q*C )  / ( N * norm(C) * EPS )
!>          RESULT(2) = norm( C*Q - C*Q )  / ( N * norm(C) * EPS )
!>          RESULT(3) = norm( Q'*C - Q'*C )/ ( N * norm(C) * EPS )
!>          RESULT(4) = norm( C*Q' - C*Q' )/ ( N * norm(C) * EPS )
!>
