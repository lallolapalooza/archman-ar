subroutine clatsy (character uplo, integer n, complex, dimension( ldx, * ) x, integer ldx, integer, dimension( * ) iseed)

CLATSY

الغرض:

!>
!> CLATSY تولد مصفوفة اختبار خاصة للتحليل المتماثل المعقد (غير المحدد).
!> ستكون كتل المحور للمصفوفة المولدة بالترتيب التالي:
!> كتلة محور 2x2، غير قابلة للقطر.
!> كتلة محور 1x1
!> كتلة محور 2x2، قابلة للقطر.
!> (تتكرر الدورة)
!> يلزم تبديل الصف لكل كتلة 2x2 غير قابلة للقطر.
!>

المعلمات

UPLO 
!>          UPLO هو CHARACTER
!>          يحدد ما إذا كانت المصفوفة المولدة ستكون علوية أم سفلية
!>          المثلث.
!>          = 'U':  مثلث علوي
!>          = 'L':  مثلث سفلي
!>

N

!>          N هو INTEGER
!>          بعد المصفوفة التي سيتم توليدها.
!>

X

!>          X هو COMPLEX array, dimension (LDX,N)
!>          المصفوفة المولدة، تتكون من كتل قطرية 3x3 و 2x2
!>          التي تؤدي إلى تسلسل المحور المذكور أعلاه.
!>          المصفوفة خارج هذه الكتل القطرية صفرية.
!>

LDX

!>          LDX هو INTEGER
!>          البعد الرئيسي للمصفوفة X.
!>

ISEED

!>          ISEED هو INTEGER array, dimension (4)
!>          عند الإدخال، بذرة مولد الأرقام العشوائية. الأخير
!>          من الأعداد الصحيحة الأربعة يجب أن يكون فرديًا. (تم تعديله عند الخروج)
!>
