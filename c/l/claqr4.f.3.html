توثيق الوظيفة/الوحدة الفرعية

subroutine claqr4 (logical wantt, logical wantz, integer n, integer ilo, integer ihi, complex, dimension( ldh, * ) h, integer ldh, complex, dimension( * ) w, integer iloz, integer ihiz, complex, dimension( ldz, * ) z, integer ldz, complex, dimension( * ) work, integer lwork, integer info)

CLAQR4 تحسب القيم الذاتية لمصفوفة هسنبرج، واختياريًا مصفوفات التحلل المشوري.

الغرض:

!>
!>    CLAQR4 تطبق مستوى واحد من العودية لـ CLAQR0.
!>    إنها تنفيذ كامل لخوارزمية QR متعددة التحولات الملتفة الصغيرة.  يمكن استدعاؤها بواسطة CLAQR0، ولحجم نافذة انحدار كبيرة بما يكفي، يمكن استدعاؤها بواسطة CLAQR3.  هذه الوحدة الفرعية مطابقة لـ CLAQR0 باستثناء أنها تستدعي CLAQR2 بدلاً من CLAQR3.
!>
!>    CLAQR4 تحسب القيم الذاتية لمصفوفة هسنبرج H، واختياريًا، المصفوفتين T و Z من التحلل المشوري H = Z T Z**H، حيث T هي مصفوفة مثلثة علوية (الشكل المشوري)، و Z هي مصفوفة وحدوية من المتجهات المشورية.
!>
!>    اختياريًا، يمكن ضرب Z من اليمين في مصفوفة وحدوية مدخلة Q حتى تتمكن هذه الوحدة الفرعية من إعطاء التحلل المشوري لمصفوفة A التي تم اختزالها إلى شكل هسنبرج H بواسطة المصفوفة الوحدوية Q:  A = Q*H*Q**H = (QZ)*H*(QZ)**H.
!>

المعاملات

WANTT 
!>          WANTT هو LOGICAL (منطقي)
!>          = .TRUE. : الشكل المشوري الكامل T مطلوب؛
!>          = .FALSE.: القيم الذاتية فقط مطلوبة.
!>

WANTZ

!>          WANTZ هو LOGICAL (منطقي)
!>          = .TRUE. : مصفوفة المتجهات المشورية Z مطلوبة؛
!>          = .FALSE.: المتجهات المشورية غير مطلوبة.
!>

N

!>          N هو INTEGER (عدد صحيح)
!>           ترتيب المصفوفة H.  N >= 0.
!>

ILO

!>          ILO هو INTEGER (عدد صحيح)
!>

IHI

!>          IHI هو INTEGER (عدد صحيح)
!>           من المفترض أن H بالفعل مثلثة علوية في الصفوف والأعمدة 1:ILO-1 و IHI+1:N، وإذا كان ILO > 1، فإن H(ILO,ILO-1) يساوي صفرًا. يتم تعيين ILO و IHI عادةً بواسطة استدعاء سابق لـ CGEBAL، ثم يتم تمريرهما إلى CGEHRD عند اختزال مخرجات CGEBAL إلى شكل هسنبرج.
!>           بخلاف ذلك، يجب تعيين ILO و IHI إلى 1 و N على التوالي.  إذا كان N > 0، فإن 1 <= ILO <= IHI <= N.
!>           إذا كان N = 0، فإن ILO = 1 و IHI = 0.
!>

H

!>          H هي مصفوفة COMPLEX (مركبة)، البعد (LDH,N)
!>           عند الدخول، مصفوفة هسنبرج العلوية H.
!>           عند الخروج، إذا كان INFO = 0 وكان WANTT هو .TRUE.، فإن H
!>           تحتوي على المصفوفة المثلثة العلوية T من التحلل المشوري
!>           (الشكل المشوري). إذا كان INFO = 0 وكان WANT هو
!>           .FALSE.، فإن محتويات H غير محددة عند الخروج.
!>           (قيمة H عند الخروج عندما يكون INFO > 0 معطاة ضمن وصف INFO أدناه.)
!>
!>           قد تحدد هذه الوحدة الفرعية صراحة H(i,j) = 0 عندما يكون i > j و
!>           j = 1, 2, ... ILO-1 أو j = IHI+1, IHI+2, ... N.
!>

LDH

!>          LDH هو INTEGER (عدد صحيح)
!>           البعد الرائد للمصفوفة H. LDH >= max(1,N).
!>

W

!>          W هي مصفوفة COMPLEX (مركبة)، البعد (N)
!>           القيم الذاتية المحسوبة لـ H(ILO:IHI,ILO:IHI) مخزنة
!>           في W(ILO:IHI). إذا كان WANTT هو .TRUE.، فإن القيم الذاتية
!>           مخزنة بنفس ترتيب القطري للشكل المشوري
!>           المعاد في H، مع W(i) = H(i,i).
!>

ILOZ

!>          ILOZ هو INTEGER (عدد صحيح)
!>

IHIZ

!>          IHIZ هو INTEGER (عدد صحيح)
!>           تحديد الصفوف من Z التي يجب تطبيق التحويلات عليها
!>           إذا كان WANTZ هو .TRUE..
!>           1 <= ILOZ <= ILO; IHI <= IHIZ <= N.
!>

Z

!>          Z هي مصفوفة COMPLEX (مركبة)، البعد (LDZ,IHI)
!>           إذا كان WANTZ هو .FALSE.، فلا يتم الرجوع إلى Z.
!>           إذا كان WANTZ هو .TRUE.، فإن Z(ILO:IHI,ILOZ:IHIZ)
!>           يتم استبدالها بـ Z(ILO:IHI,ILOZ:IHIZ)*U حيث U هي
!>           عامل التحلل المشوري الوحدوي لـ H(ILO:IHI,ILO:IHI).
!>           (قيمة Z عند الخروج عندما يكون INFO > 0 معطاة ضمن
!>           وصف INFO أدناه.)
!>

LDZ

!>          LDZ هو INTEGER (عدد صحيح)
!>           البعد الرائد للمصفوفة Z.  إذا كان WANTZ هو .TRUE.
!>           فإن LDZ >= MAX(1,IHIZ).  بخلاف ذلك، LDZ >= 1.
!>

WORK

!>          WORK هي مصفوفة COMPLEX (مركبة)، البعد LWORK
!>           عند الخروج، إذا كان LWORK = -1، فإن WORK(1) يعيد تقديرًا
!>           للقيمة المثلى لـ LWORK.
!>

LWORK

!>          LWORK هو INTEGER (عدد صحيح)
!>           بعد المصفوفة WORK.  LWORK >= max(1,N)
!>           كافٍ، ولكن LWORK عادةً ما يصل إلى 6*N قد يكون
!>           مطلوبًا للأداء الأمثل.  يوصى باستعلام مساحة العمل
!>           لتحديد حجم مساحة العمل المثلى.
!>
!>           إذا كان LWORK = -1، فإن CLAQR4 يقوم باستعلام مساحة العمل.
!>           في هذه الحالة، تتحقق CLAQR4 من معاملات الإدخال وتقدر
!>           حجم مساحة العمل الأمثل للقيم N، ILO، و IHI المعطاة.
!>           يتم إرجاع التقدير في WORK(1).  لا يتم إصدار رسالة خطأ تتعلق بـ LWORK
!>           بواسطة XERBLA.  لا يتم الوصول إلى H ولا Z.
!>

INFO

!>          INFO هو INTEGER (عدد صحيح)
!>             = 0:  الخروج ناجح
!>             > 0:  إذا كان INFO = i، فشلت CLAQR4 في حساب جميع
!>                القيم الذاتية.  العناصر 1:ilo-1 و i+1:n من WR
!>                و WI تحتوي على تلك القيم الذاتية التي تم
!>                حسابها بنجاح.  (الفشل نادر.)
!>
!>                إذا كان INFO > 0 وكان WANT هو .FALSE.، فعند الخروج،
!>                القيم الذاتية المتبقية غير المتقاربة هي القيم الذاتية
!>                لمصفوفة هسنبرج العلوية للصفوف والأعمدة ILO إلى INFO
!>                من القيمة النهائية لـ H.
!>
!>                إذا كان INFO > 0 وكان WANTT هو .TRUE.، فعند الخروج
!>
!>           (*)  (القيمة الأولية لـ H)*U  = U*(القيمة النهائية لـ H)
!>
!>                حيث U هي مصفوفة وحدوية.  القيمة النهائية لـ H
!>                هي هسنبرج علوية ومثلثة في الصفوف والأعمدة INFO+1 إلى IHI.
!>
!>                إذا كان INFO > 0 وكان WANTZ هو .TRUE.، فعند الخروج
!>
!>                  (القيمة النهائية لـ Z(ILO:IHI,ILOZ:IHIZ)
!>                   =  (القيمة الأولية لـ Z(ILO:IHI,ILOZ:IHIZ)*U
!>
!>                حيث U هي المصفوفة الوحدوية في (*) (بغض النظر عن قيمة WANTT).
!>
!>                إذا كان INFO > 0 وكان WANTZ هو .FALSE.، فلا يتم الوصول إلى Z.
!>
