subroutine clarf1f (character side, integer m, integer n, complex, dimension( * ) v, integer incv, complex tau, complex, dimension( ldc, * ) c, integer ldc, complex, dimension( * ) work)

CLARF1F يطبق عاكس أولي معقد H على مصفوفة مستطيلة عامة

الغرض:

!>
!> CLARF1F يطبق عاكس أولي مركب H على مصفوفة عامة مركبة m × n
!> C، من اليسار أو من اليمين. يتم تمثيل H بالشكل
!>
!>       H = I - tau * v * v**H
!>
!> حيث tau هو عدد مركب و v هو متجه مركب بافتراض v(1) = 1.
!>
!> إذا كان tau = 0، يتم اعتبار H مصفوفة الوحدة.
!>
!> لتطبيق H**H (المرافق المتحول لـ H)، قم بتمرير conjg(tau) بدلاً من
!> tau.
!>

المعاملات

SIDE 
!>          SIDE هو CHARACTER*1
!>          = 'L': تشكيل  H * C
!>          = 'R': تشكيل  C * H
!>

M

!>          M هو INTEGER
!>          عدد صفوف المصفوفة C.
!>

N

!>          N هو INTEGER
!>          عدد أعمدة المصفوفة C.
!>

V

!>          V هو مصفوفة COMPLEX، الأبعاد
!>                     (1 + (M-1)*abs(INCV)) إذا كان SIDE = 'L'
!>                  أو (1 + (N-1)*abs(INCV)) إذا كان SIDE = 'R'
!>          المتجه v في تمثيل H. لا يتم استخدام V إذا كان
!>          TAU = 0.
!>

INCV

!>          INCV هو INTEGER
!>          الفارق بين عناصر v. INCV <> 0.
!>

TAU

!>          TAU هو COMPLEX
!>          القيمة tau في تمثيل H.
!>

C

!>          C هو مصفوفة COMPLEX، الأبعاد (LDC,N)
!>          عند الدخول، المصفوفة C ذات الأبعاد m × n.
!>          عند الخروج، يتم استبدال C بالمصفوفة H * C إذا كان SIDE = 'L'،
!>          أو C * H إذا كان SIDE = 'R'.
!>

LDC

!>          LDC هو INTEGER
!>          البعد الرئيسي للمصفوفة C. LDC >= max(1,M).
!>

WORK

!>          WORK هو مصفوفة COMPLEX، الأبعاد
!>                         (N) إذا كان SIDE = 'L'
!>                      أو (M) إذا كان SIDE = 'R'
!>
