subroutine clatm1 (integer mode, real cond, integer irsign, integer idist, integer, dimension( 4 ) iseed, complex, dimension( * ) d, integer n, integer info)

CLATM1

الغرض:

!>
!>    CLATM1 تحسب مدخلات D(1..N) كما هو محدد بواسطة
!>    MODE، COND و IRSIGN. IDIST و ISEED تحدد توليد
!>    الأرقام العشوائية. يتم استدعاء CLATM1 بواسطة CLATMR لتوليد
!>    مصفوفات اختبار عشوائية لبرامج LAPACK.
!>

المعلمات

MODE 
!>          MODE هو عدد صحيح
!>           عند الإدخال يصف كيف يجب حساب D:
!>           MODE = 0 يعني عدم تغيير D.
!>           MODE = 1 يضبط D(1)=1 و D(2:N)=1.0/COND
!>           MODE = 2 يضبط D(1:N-1)=1 و D(N)=1.0/COND
!>           MODE = 3 يضبط D(I)=COND**(-(I-1)/(N-1))
!>           MODE = 4 يضبط D(i)=1 - (i-1)/(N-1)*(1 - 1/COND)
!>           MODE = 5 يضبط D لأرقام عشوائية في النطاق
!>                    ( 1/COND , 1 ) بحيث لوغاريتماتها
!>                    موزعة بشكل موحد.
!>           MODE = 6 يضبط D لأرقام عشوائية من نفس التوزيع
!>                    مثل باقي المصفوفة.
!>           MODE < 0 له نفس المعنى مثل ABS(MODE)، باستثناء أن
!>              ترتيب عناصر D معكوس.
!>           لذلك إذا كان MODE موجبًا، فإن D تحتوي على مدخلات تتراوح من
!>              1 إلى 1/COND، إذا كان سالبًا، من 1/COND إلى 1،
!>           غير معدل.
!>

COND

!>          COND هو عدد حقيقي
!>           عند الإدخال، يستخدم كما هو موضح تحت MODE أعلاه.
!>           إذا تم استخدامه، يجب أن يكون >= 1. غير معدل.
!>

IRSIGN

!>          IRSIGN هو عدد صحيح
!>           عند الإدخال، إذا لم يكن MODE -6 أو 0 أو 6، يحدد إشارة
!>           مدخلات D
!>           0 => ترك مدخلات D دون تغيير
!>           1 => ضرب كل مدخل في D برقم مركب عشوائي
!>                موزع بشكل موحد بقيمة مطلقة 1
!>

IDIST

!>          IDIST هو عدد صحيح
!>           عند الإدخال، يحدد IDIST نوع التوزيع الذي سيتم
!>           استخدامه لتوليد مصفوفة عشوائية.
!>           1 => الأجزاء الحقيقية والتخيلية كل منها UNIFORM( 0, 1 )
!>           2 => الأجزاء الحقيقية والتخيلية كل منها UNIFORM( -1, 1 )
!>           3 => الأجزاء الحقيقية والتخيلية كل منها NORMAL( 0, 1 )
!>           4 => رقم مركب موحد في DISK( 0, 1 )
!>           غير معدل.
!>

ISEED

!>          ISEED هو مصفوفة أعداد صحيحة، بأبعاد ( 4 )
!>           عند الإدخال، يحدد ISEED بذرة مولد الأرقام
!>           العشوائية. يستخدم مولد الأرقام
!>           العشوائية تسلسل خطي متطابق محدود بأعداد
!>           صحيحة صغيرة، ولذلك يجب أن ينتج أرقامًا عشوائية
!>           مستقلة عن الآلة. يتم تغيير قيم ISEED عند
!>           الخروج، ويمكن استخدامها في الاستدعاء التالي لـ CLATM1
!>           لمواصلة نفس تسلسل الأرقام العشوائية.
!>           تم التغيير عند الخروج.
!>

D

!>          D هي مصفوفة مركبة، بأبعاد ( N )
!>           المصفوفة التي سيتم حسابها وفقًا لـ MODE و COND و IRSIGN.
!>           قد يتم تغييرها عند الخروج إذا كان MODE غير صفري.
!>

N

!>          N هو عدد صحيح
!>           عدد مدخلات D. غير معدل.
!>

INFO

!>          INFO هو عدد صحيح
!>            0  => إنهاء طبيعي
!>           -1  => إذا كان MODE خارج النطاق -6 إلى 6
!>           -2  => إذا لم يكن MODE -6 أو 0 أو 6، و
!>                  IRSIGN ليس 0 أو 1
!>           -3  => إذا لم يكن MODE -6 أو 0 أو 6 و COND أقل من 1
!>           -4  => إذا كان MODE يساوي 6 أو -6 و IDIST خارج النطاق 1 إلى 4
!>           -7  => إذا كان N سالبًا
!>
