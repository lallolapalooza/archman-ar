ck_sequence(3)

الحزمة: libck

التراخيص: BSD, Apache, Custom

ck_sequence(3) دليل الدوال للمكتبة ck_sequence(3)

الاسم

ck_sequence_init، ck_sequence_read_begin، ck_sequence_read_retry، ck_sequence_write_begin، ck_sequence_write_end — أقفال التسلسل

المكتبة

مجموعة التزامن (libck, -lck)

الملخص

#include <ck_sequence.h>

ck_sequence_t seqlock = CK_SEQUENCE_INITIALIZER;


void
ck_sequence_init(ck_sequence_t *sq);

unsigned int
ck_sequence_read_begin(const ck_sequence_t *sq);

bool
ck_sequence_read_retry(const ck_sequence_t *sq, unsigned int version);

void
ck_sequence_write_begin(ck_sequence_t *sq);

void
ck_sequence_write_end(ck_sequence_t *sq);

الوصف

يُوصى باستخدام ck_sequence عندما يلزم مزامنة كمية صغيرة من البيانات التي لا يمكن الوصول إليها بشكل ذري مع القراء بطريقة لا تحظر أي كاتب. يمكن للقراء تنفيذ أقسامهم الحرجة من جانب القارئ دون أي عمليات ذرية. يجب تهيئة ck_sequence_t قبل الاستخدام. يمكن تهيئتها باستخدام مُهيئ ثابت (CK_SEQUENCE_INITIALIZER) أو باستخدام ck_sequence_init(). قبل أن يحاول القراء قراءة البيانات التي قد يتم تعديلها بشكل متزامن، يجب عليهم أولاً حفظ القيمة التي تم إرجاعها من ck_sequence_read_begin(). أثناء أو بعد الانتهاء من نسخ القارئ للبيانات المرتبطة بـ ck_sequence_t، يجب عليه تمرير القيمة التي تم إرجاعها سابقًا من ck_sequence_read_begin() إلى ck_sequence_read_retry(). إذا أعادت ck_sequence_read_retry() القيمة true، فقد تكون نسخة البيانات غير متسقة ويجب إعادة محاولة عملية القراءة. يجب على الكتاب الاعتماد على آليات المزامنة الخاصة بهم. بمجرد دخول الكاتب قسمه الحرج، يجب عليه استدعاء ck_sequence_write_begin() للإشارة إلى نية تحديث البيانات التي تحميها ck_sequence_t. قبل مغادرة الكاتب لقسمه الحرج، يجب عليه تنفيذ ck_sequence_write_end() للإشارة إلى أن التحديثات قد تركت الكائنات المعنية في حالة متسقة.

مثال

#include <ck_sequence.h>
#include <stdlib.h>

static struct example {
	int a;
	int b;
	int c;
} global;

static ck_sequence_t seqlock = CK_SEQUENCE_INITIALIZER;

void
reader(void)
{
	struct example copy;
	unsigned int version;

	/*
	 * محاولة قراءة هيكل البيانات. إذا تم تعديل الهيكل
	 * بين ck_sequence_read_begin و
	 * ck_sequence_read_retry، فحاول قراءة أخرى لأن
	 * البيانات قد تكون في حالة غير متسقة.
	 */
	do {
		version = ck_sequence_read_begin(&seqlock);
		copy = global;
	} while (ck_sequence_read_retry(&seqlock, version));

	/*
	 * يمكن أيضًا التعبير عما سبق باستخدام CK_SEQUENCE_READ.
	 * بشكل عام، يوصى باستخدام ck_sequence_read_retry فقط
	 * إذا كنت ترغب في اكتشاف كتابة متعارضة على مستوى
	 * أعلى.
	 */
	CK_SEQUENCE_READ(&seqlock, &version) {
		copy = global;
	}

	return;
}

void
writer(void)
{

	for (;;) {
		ck_sequence_write_begin(&seqlock);
		global.a = rand();
		global.b = global.a + global.b;
		global.c = global.b + global.c;
		ck_sequence_write_end(&seqlock);
	}

	return;
}
