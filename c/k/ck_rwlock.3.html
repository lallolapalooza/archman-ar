وصف

هذا قفل قراءة-كتابة مركزي منحاز للكتابة. يتطلب مساحة إضافية قليلة جدًا ويحتوي على مسار سريع بزمن وصول منخفض. الكتابة الجانبية للتكرار تتطلب استخدام ck_rwlock_recursive. تكرار القراءة غير مسموح به. تقوم الدالة ck_rwlock_write_downgrade() بتقليل اكتساب جانب الكتابة للمتصل إلى اكتساب جانب القراءة دون التنازل عن القسم الحرج الحالي.

مثال

#include <ck_rwlock.h>

static ck_rwlock_t lock = CK_RWLOCK_INITIALIZER;

static void
reader(void)
{

	for (;;) {
		ck_rwlock_read_lock(&lock);
		/* قسم القراءة الحرج. */
		ck_rwlock_read_unlock(&lock);

		if (ck_rwlock_read_trylock(&lock) == true) {
			/* قسم القراءة الحرج. */
			ck_rwlock_read_unlock(&lock);
		}
	}

	return;
}

static void
writer(void)
{

	for (;;) {
		ck_rwlock_write_lock(&lock);
		/* قسم الكتابة الحرج. */
		ck_rwlock_write_unlock(&lock);

		if (ck_rwlock_write_trylock(&lock, 1) == true) {
			/* قسم الكتابة الحرج. */
			ck_rwlock_write_unlock(&lock);
		}
	}

	return;
}
