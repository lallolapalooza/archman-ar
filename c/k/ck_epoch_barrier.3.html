ck_epoch_barrier(3)

الحزمة: libck

التراخيص: BSD, Apache, Custom

CK_EPOCH_BARRIER(3) دليل وظائف المكتبة CK_EPOCH_BARRIER(3)

الاسم

ck_epoch_barrier — يمنع حتى فترة سماح ويتم إرسال جميع الاستدعاءات

المكتبة

Concurrency Kit (libck, -lck)

الملخص

#include <ck_epoch.h>

void
ck_epoch_barrier(ck_epoch_record_t *record);

الوصف

وظيفة ck_epoch_barrier(3) ستمنع المستدعي حتى يتم اكتشاف فترة سماح، وفقًا لمعاني استعادة العصر. أي كائنات تتطلب استعادة ذاكرة آمنة تم حذفها منطقيًا تكون آمنة للحذف الفعلي بعد استدعاء ck_epoch_barrier(3). ستقوم هذه الوظيفة أيضًا بإرسال جميع الاستدعاءات المرتبطة بـ epoch والتي تم جدولتها سابقًا عبر ck_epoch_call(3).

مثال

#include <ck_epoch.h>
#include <ck_stack.h>
#include <stdlib.h>

/*
 * تم تهيئة العصر مسبقًا باستخدام ck_epoch_init.
 * تم تهيئة المكدس مسبقًا باستخدام ck_stack_init.
 */
ck_epoch_t *epoch;
ck_stack_t *stack;

void
function(void)
{
	ck_epoch_record_t *record;
	ck_stack_entry_t *s;

	record = malloc(sizeof *record);
	ck_epoch_register(&epoch, record, NULL);

	/*
	 * نحن نستخدم قسم العصر هنا لضمان عدم حذف
	 * العقد في المكدس أثناء إحالتنا إليها. هذا مطلوب هنا
	 * نظرًا لوجود عدة كتاب. إذا كان هناك خيط واحد فقط
	 * يقوم بسحب هذا المكدس، فلن تكون هناك حاجة إلى ck_epoch_begin/ck_epoch_end.
	 */
	ck_epoch_begin(record);

	/* احذف كائنًا منطقيًا. */
	s = ck_stack_pop_upmc(stack);

	ck_epoch_end(record);

	/*
	 * انتظر حتى لا تتمكن الخيوط من امتلاك مرجع للكائن
	 * الذي قمنا بسحبه للتو (افترض أن جميع الخيوط تنفذ ببساطة
	 * ck_stack_pop_upmc).
	 */
	ck_epoch_barrier(record);

	/* الآن أصبح من الآمن حذف الكائن فعليًا. */
	free(s);
	return;
}

قيم الإرجاع

لا تحتوي هذه الوظيفة على قيمة إرجاع.

الأخطاء

السلوك غير محدد إذا كان الكائن المشار إليه بواسطة epoch ليس كائن عصر صالحًا. يجب أن يكون الكائن المشار إليه بواسطة record قد تم تسجيله مسبقًا عبر ck_epoch_register(3).
