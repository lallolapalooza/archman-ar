ck_epoch_call(3)

الحزمة: libck

التراخيص: BSD، Apache، مخصص

CK_EPOCH_CALL(3) دليل وظائف المكتبة CK_EPOCH_CALL(3)

الاسم

ck_epoch_call — تأجيل تنفيذ الدالة حتى فترة سماح

المكتبة

Concurrency Kit (libck, -lck)

ملخص

#include <ck_epoch.h> typedef struct ck_epoch_entry ck_epoch_entry_t;
typedef void ck_epoch_cb_t(ck_epoch_entry_t *);
void
ck_epoch_call(ck_epoch_record_t *record, ck_epoch_entry_t *entry, ck_epoch_cb_t *function);

الوصف

ستقوم الدالة ck_epoch_call(3) بتأجيل تنفيذ الدالة التي يشير إليها function حتى يتم اكتشاف فترة سماح في epoch. سيتم تزويد الدالة بالمؤشر المحدد بواسطة entry. سيتم تنفيذ الدالة في وقت ما في المستقبل عبر استدعاءات لـ ck_epoch_reclaim(3) أو ck_epoch_barrier(3) أو ck_epoch_poll(3).

مثال

#include <ck_epoch.h>
#include <ck_stack.h>
#include <stdlib.h>

/*
 * تم تهيئة epoch مسبقًا باستخدام ck_epoch_init.
 */
ck_epoch_t *epoch;

struct object {
	int value;
	ck_epoch_entry_t epoch_entry;
};
static struct object *global;

CK_EPOCH_CONTAINER(struct object, epoch_entry, object_container)

void
destroy_object(ck_epoch_entry_t *e)
{
	struct object *o = object_container(e);

	free(o);
	return;
}

void
function(void)
{
	ck_epoch_record_t *record;
	struct object *n;

	record = malloc(sizeof *record);
	ck_epoch_register(&epoch, record, NULL);

	n = malloc(sizeof *n);
	if (n == NULL)
		return;

	n->value = 1;

	/*
	 * نستخدم قسم epoch هنا لأن هناك العديد من الكتاب.
	 * كما أنه خيار لاستخدام أشكال أخرى من حظر
	 * مزامنة جانب الكتابة مثل mutexes.
	 */
	ck_epoch_begin(record);
	n = ck_pr_fas_ptr(&global, n);
	ck_epoch_end(record);

	/* تأجيل تدمير الكائن السابق. */
	ck_epoch_call(record, &n->epoch_entry, destroy_object);

	/* استطلاع نظام epoch الفرعي بطريقة غير مانعة. */
	ck_epoch_poll(record);
	return;
}

قيم الإرجاع

لا تحتوي هذه الدالة على قيمة إرجاع.

الأخطاء

يجب أن يكون الكائن الذي يشير إليه record قد تم تسجيله مسبقًا عبر ck_epoch_register(3).
