ck_tflock(3)

الحزمة: libck

التراخيص: BSD, Apache, Custom

ck_tflock(3) دليل وظائف المكتبة ck_tflock(3)

الاسم

ck_tflock_ticket_init، ck_tflock_ticket_write_lock، ck_tflock_ticket_write_unlock، ck_tflock_ticket_read_lock، ck_tflock_ticket_read_unlock، — أقفال مركزية للقراءة/الكتابة عادلة للمهام

المكتبة

مجموعة التزامن (libck، -lck)

موجز

#include <ck_tflock.h>

ck_tflock_ticket_t lock = CK_TFLOCK_TICKET_INITIALIZER;


void
ck_tflock_ticket_init(ck_tflock_ticket_t *lock);

void
ck_tflock_ticket_write_lock(ck_tflock_ticket_t *lock);

void
ck_tflock_ticket_write_unlock(ck_tflock_ticket_t *lock);

void
ck_tflock_ticket_read_lock(ck_tflock_ticket_t *lock);

void
ck_tflock_ticket_read_unlock(ck_tflock_ticket_t *lock);

الوصف

هذا قفل مركزي للقراءة/الكتابة عادل للمهام. يتطلب الحد الأدنى من استهلاك المساحة ولديه مسار سريع بزمن استجابة منخفض.

مثال

#include <ck_tflock.h>

static ck_tflock_ticket_t lock = CK_TFLOCK_INITIALIZER;

static void
reader(void)
{

	for (;;) {
		ck_tflock_ticket_read_lock(&lock);
		/* قسم القراءة النقدي. */
		ck_tflock_ticket_read_unlock(&lock);
	}

	return;
}

static void
writer(void)
{

	for (;;) {
		ck_tflock_ticket_write_lock(&lock);
		/* قسم الكتابة النقدي. */
		ck_tflock_ticket_write_unlock(&lock);
	}

	return;
}
