ck_pr_store(3)

الحزمة: libck

التراخيص: BSD, Apache, Custom

ck_pr_store(3) دليل وظائف المكتبة ck_pr_store(3)

الاسم

ck_pr_store_ptr، ck_pr_store_double، ck_pr_store_uint، ck_pr_store_int، ck_pr_store_char، ck_pr_store_64، ck_pr_store_32، ck_pr_store_16، ck_pr_store_8 — عمليات تخزين متزامنة للقيم المتطايرة

المكتبة

Concurrency Kit (libck, -lck)

الملخص

#include <ck_pr.h>

void
ck_pr_store_ptr(void *target, void *value);

void
ck_pr_store_double(double *target, double value);

void
ck_pr_store_uint(unsigned int *target, unsigned int value);

void
ck_pr_store_int(int *target, int value);

void
ck_pr_store_char(char *target, char value);

void
ck_pr_store_64(uint64_t *target, uint64_t value);

void
ck_pr_store_32(uint32_t *target, uint32_t value);

void
ck_pr_store_16(uint16_t *target, uint16_t value);

void
ck_pr_store_8(uint8_t *target, uint8_t value);

الوصف

تقوم عائلة الوظائف ck_pr_store(3) بتخزين القيمة المحددة بواسطة value بشكل متزامن في الموقع الذي يشير إليه target. تخدم عائلة الوظائف هذه دائمًا كحاجز ضمني للمترجم وهي غير عرضة لإعادة ترتيب المترجم.

قيم الإرجاع

ليس لهذه العائلة من الوظائف قيمة إرجاع.
