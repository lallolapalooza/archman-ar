ck_ring_enqueue_spsc_size(3)

الحزمة: libck

التراخيص: BSD, Apache, Custom

CK_RING_ENQUEUE_SPSC_SIZE(3) دليل وظائف المكتبة CK_RING_ENQUEUE_SPSC_SIZE(3)

الاسم

ck_ring_enqueue_spsc_size — إضافة مؤشر إلى طابور FIFO مقيد وإرجاع حجم المخزن المؤقت

المكتبة

مجموعة التزامن (libck, -lck)

الملخص

#include <ck_ring.h>

bool
ck_ring_enqueue_spsc_size(ck_ring_t *ring, ck_ring_buffer_t *buffer, void *entry, unsigned int *size);

الوصف

تقوم الوظيفة ck_ring_enqueue_spsc_size(3) بإضافة المؤشر entry إلى المخزن المؤقت المقيد الذي يشير إليه ring بترتيب FIFO. يجب أن يكون المخزن المؤقت الذي يشير إليه buffer فريدًا لـ ring ويشير إلى مصفوفة من ck_ring_buffer_t بطول كافٍ (وفقًا لعناصر قوة العدد 2 في المخزن المؤقت). يخدم فصل الخاتم عن المخزن المؤقت معالجة حالات الاستخدام التي تتضمن مساحات عناوين متعددة و DMA، من بين أمور أخرى. إذا كنت على أنظمة غير POSIX أو ترغب في الامتثال الصارم للغة C، فمن المستحسن تمرير مؤشر من النوع void ** لـ entry. هذه الوظيفة آمنة للاستدعاء بدون قفل لتزامن واحد على الأقل لاستدعاء ck_ring_dequeue_spsc(3). توفر هذه الوظيفة ضمانات تقدم بدون انتظار.

مثال

#include <ck_ring.h>

/* تم تهيئة هذا الخاتم مسبقًا باستخدام ck_ring_init. */
ck_ring_t ring;

/* تم تهيئة الخاتم لـ 1023 عنصرًا. */
ck_ring_buffer_t buffer[1024];

void
enqueue(void)
{
	void *entry = some_object;
	unsigned int length;

	/* محاولة إضافة مؤشر إلى some_object إلى المخزن المؤقت. */
	if (ck_ring_enqueue_spsc(&ring, &buffer, &entry, &length) == false) {
		/*
		 * كان المخزن المؤقت ممتلئًا وفشلت عملية الإضافة.
		 */
		return;
	}

	/*
	 * إذا كان طول المخزن المؤقت 100 عنصر أو أكثر في وقت إضافة العنصر،
	 * قم بشيء ما.
	 */
	if (length > 100) {
		do_something;
	}

	return;
}

قيم الإرجاع

تُرجع الوظيفة true إذا تم إضافة قيمة entry بنجاح إلى ring. تُرجع هذه الوظيفة عدد العناصر في ring فيما يتعلق بنقطة التسلسل (النقطة في العنصر التي تم فيها إضافة entry). تُرجع الوظيفة false إذا لم يمكن إضافة قيمة entry، وهو ما يحدث فقط إذا كان ring ممتلئًا.
