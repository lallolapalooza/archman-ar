subroutine csyr (character uplo, integer n, complex alpha, complex, dimension( * ) x, integer incx, complex, dimension( lda, * ) a, integer lda)

CSYR تجري تحديث الرتبة-1 المتماثل لمصفوفة متماثلة معقدة.

الغرض:

!>
!> CSYR   تجري عملية الرتبة 1 المتماثلة
!>
!>    A := alpha*x*x**H + A,
!>
!> حيث alpha هو قياسي معقد، x هو متجه من n عنصر، و A هي مصفوفة
!> متماثلة بحجم n × n.
!>

المعلمات

UPLO 
!>          UPLO is CHARACTER*1
!>           عند الدخول، تحدد UPLO ما إذا كان الجزء المثلثي العلوي أو السفلي
!>           من المصفوفة A سيتم الرجوع إليه كما يلي:
!>
!>              UPLO = 'U' or 'u'   يتم الرجوع فقط إلى الجزء المثلثي العلوي من A
!>                                  .
!>
!>              UPLO = 'L' or 'l'   يتم الرجوع فقط إلى الجزء المثلثي السفلي من A
!>                                  .
!>
!>           غير متغيرة عند الخروج.
!>

N

!>          N is INTEGER
!>           عند الدخول، يحدد N رتبة المصفوفة A.
!>           يجب أن يكون N أكبر من أو يساوي صفر.
!>           غير متغيرة عند الخروج.
!>

ALPHA

!>          ALPHA is COMPLEX
!>           عند الدخول، يحدد ALPHA القياسي alpha.
!>           غير متغيرة عند الخروج.
!>

X

!>          X is COMPLEX array, dimension at least
!>           ( 1 + ( N - 1 )*abs( INCX ) ).
!>           قبل الدخول، يجب أن يحتوي مصفوفة X ذات الفهرس المتزايد على المتجه x
!>           ذو الـ N عنصر.
!>           غير متغيرة عند الخروج.
!>

INCX

!>          INCX is INTEGER
!>           عند الدخول، يحدد INCX الزيادة لعناصر X. يجب ألا يكون INCX صفرًا.
!>           غير متغيرة عند الخروج.
!>

A

!>          A is COMPLEX array, dimension ( LDA, N )
!>           قبل الدخول، مع UPLO = 'U' أو 'u'، يجب أن يحتوي الجزء العلوي الرئيسي
!>           بحجم n × n من المصفوفة A على الجزء المثلثي العلوي للمصفوفة المتماثلة،
!>           ولا يتم الرجوع إلى الجزء السفلي الصارم من A. عند الخروج، يتم
!>           الكتابة فوق الجزء المثلثي العلوي من المصفوفة A بالجزء المثلثي العلوي
!>           من المصفوفة المحدثة.
!>           قبل الدخول، مع UPLO = 'L' أو 'l'، يجب أن يحتوي الجزء السفلي الرئيسي
!>           بحجم n × n من المصفوفة A على الجزء المثلثي السفلي للمصفوفة المتماثلة،
!>           ولا يتم الرجوع إلى الجزء العلوي الصارم من A. عند الخروج، يتم
!>           الكتابة فوق الجزء المثلثي السفلي من المصفوفة A بالجزء المثلثي السفلي
!>           من المصفوفة المحدثة.
!>

LDA

!>          LDA is INTEGER
!>           عند الدخول، يحدد LDA البعد الأول لـ A كما هو معلن
!>           في البرنامج (الفرعي) المستدعي. يجب أن يكون LDA على الأقل
!>           max( 1, N ).
!>           غير متغيرة عند الخروج.
!>
