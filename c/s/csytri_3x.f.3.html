subroutine csytri_3x (character uplo, integer n, complex, dimension( lda, * ) a, integer lda, complex, dimension( * ) e, integer, dimension( * ) ipiv, complex, dimension( n+nb+1, * ) work, integer nb, integer info)

CSYTRI_3X

الغرض:

!> CSYTRI_3X يحسب معكوس مصفوفة معقدة متماثلة غير محددة
!> A باستخدام التحليل الذي تم حسابه بواسطة CSYTRF_RK أو CSYTRF_BK:
!>
!>     A = P*U*D*(U**T)*(P**T) أو A = P*L*D*(L**T)*(P**T),
!>
!> حيث U (أو L) مصفوفة مثلثة عليا (أو سفلى) وحدة،
!> U**T (أو L**T) هي منقولة U (أو L)، P مصفوفة تبديل، P**T هي منقولة P،
!> و D مصفوفة متماثلة وقطرية كتلة مع كتل قطرية 1x1 و 2x2.
!>
!> هذه هي النسخة المجمعة من الخوارزمية، تستدعي BLAS المستوى 3.
!>

المعاملات

UPLO 
!>          UPLO هو CHARACTER*1
!>          يحدد ما إذا كانت تفاصيل التحليل مخزنة كمصفوفة مثلثة عليا
!>          أو سفلى.
!>          = 'U':  يتم تخزين المثلث العلوي من A؛
!>          = 'L':  يتم تخزين المثلث السفلي من A.
!>

N

!>          N هو INTEGER
!>          رتبة المصفوفة A.  N >= 0.
!>

A

!>          A هو مصفوفة COMPLEX، ذات البعد (LDA,N)
!>          عند الدخول، قطر المصفوفة القطرية الكتلية المتماثلة D
!>          والعوامل U أو L كما تم حسابها بواسطة CSYTRF_RK و CSYTRF_BK:
!>            أ) عناصر قطرية فقط من المصفوفة القطرية الكتلية المتماثلة
!>               D على قطر A، أي D(k,k) = A(k,k)؛
!>               (يجب توفير العناصر فوق القطرية (أو تحت القطرية)
!>                لـ D في المصفوفة E عند الدخول)، و
!>            ب) إذا كان UPLO = 'U': العامل U في جزء فوق القطر من A.
!>               إذا كان UPLO = 'L': العامل L في جزء تحت القطر من A.
!>
!>          عند الخروج، إذا كان INFO = 0، فإن معكوس المصفوفة الأصلية
!>          المتماثلة.
!>             إذا كان UPLO = 'U': يتم تشكيل الجزء العلوي المثلثي من المعكوس
!>             ولا يتم الرجوع إلى الجزء السفلي من A؛
!>             إذا كان UPLO = 'L': يتم تشكيل الجزء السفلي المثلثي من المعكوس
!>             ولا يتم الرجوع إلى الجزء العلوي من A.
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرائد للمصفوفة A.  LDA >= max(1,N).
!>

E

!>          E هو مصفوفة COMPLEX، ذات البعد (N)
!>          عند الدخول، تحتوي على العناصر فوق القطرية (أو تحت القطرية)
!>          للمصفوفة القطرية الكتلية المتماثلة D
!>          مع كتل قطرية 1x1 أو 2x2، حيث
!>          إذا كان UPLO = 'U': E(i) = D(i-1,i), i=2:N, E(1) لا يتم الرجوع إليها؛
!>          إذا كان UPLO = 'L': E(i) = D(i+1,i), i=1:N-1, E(N) لا يتم الرجوع إليها.
!>
!>          ملاحظة: لكتلة قطرية 1x1 D(k)، حيث
!>          1 <= k <= N، لا يتم الرجوع إلى العنصر E(k) في كلتا
!>          حالات UPLO = 'U' أو UPLO = 'L'.
!>

IPIV

!>          IPIV هو مصفوفة INTEGER، ذات البعد (N)
!>          تفاصيل التبادلات وهيكل الكتلة لـ D
!>          كما تم تحديده بواسطة CSYTRF_RK أو CSYTRF_BK.
!>

WORK

!>          WORK هي مصفوفة COMPLEX، ذات البعد (N+NB+1,NB+3).
!>

NB

!>          NB هو INTEGER
!>          حجم الكتلة.
!>

INFO

!>          INFO هو INTEGER
!>          = 0: خروج ناجح
!>          < 0: إذا كان INFO = -i، فإن المعامل i كان له قيمة غير قانونية
!>          > 0: إذا كان INFO = i، فإن D(i,i) = 0؛ المصفوفة شاذة ولا يمكن
!>               حساب معكوسها.
!>
