subroutine csytrf_aa (character uplo, integer n, complex, dimension( lda, * ) a, integer lda, integer, dimension( * ) ipiv, complex, dimension( * ) work, integer lwork, integer info)

CSYTRF_AA

الغرض:

!>
!> CSYTRF_AA يحسب تحليل مصفوفة متناظرة معقدة A
!> باستخدام خوارزمية Aasen. شكل التحليل هو
!>
!>    A = U**T*T*U  أو  A = L*T*L**T
!>
!> حيث U (أو L) هي ناتج تبديلات ومصفوفات مثلثية عليا (سفلى) وحدوية،
!> و T هي مصفوفة ثلاثية القطر متناظرة معقدة.
!>
!> هذا هو الإصدار المحصور من الخوارزمية، والذي يستدعي BLAS المستوى 3.
!>

المعلمات

UPLO 
!>          UPLO هو CHARACTER*1
!>          = 'U':  المثلث العلوي لـ A مخزن؛
!>          = 'L':  المثلث السفلي لـ A مخزن.
!>

N

!>          N هو INTEGER
!>          رتبة المصفوفة A.  N >= 0.
!>

A

!>          A هي مصفوفة COMPLEX، ذات بعد (LDA,N)
!>          عند الإدخال، المصفوفة المتناظرة A.  إذا كان UPLO = 'U'، فإن
!>          الجزء المثلثي العلوي N-by-N من A يحتوي على الجزء المثلثي
!>          العلوي للمصفوفة A، والجزء المثلثي السفلي بدقة
!>          ليس مرجعًا.  إذا كان UPLO = 'L'، فإن
!>          الجزء المثلثي السفلي N-by-N من A يحتوي على الجزء المثلثي
!>          السفلي للمصفوفة A، والجزء المثلثي العلوي بدقة
!>          ليس مرجعًا.
!>
!>          عند الخروج، تكون المصفوفة ثلاثية القطر مخزنة في الأقطار
!>          والأقطار الفرعية أسفل (أو فوق) الأقطار،
!>          و L مخزنة أسفل (أو فوق) الأقطار الفرعية، عندما يكون UPLO
!>          هو 'L' (أو 'U').
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرائد للمصفوفة A.  LDA >= max(1,N).
!>

IPIV

!>          IPIV هي مصفوفة INTEGER، ذات بعد (N)
!>          عند الخروج، تحتوي على تفاصيل التبادلات، أي
!>          تم تبديل الصف والعمود k من A مع
!>          الصف والعمود IPIV(k).
!>

WORK

!>          WORK هي مصفوفة COMPLEX، ذات بعد (MAX(1,LWORK))
!>          عند الخروج، إذا كان INFO = 0، فإن WORK(1) تعيد LWORK الأمثل.
!>

LWORK

!>          LWORK هو INTEGER
!>          طول WORK. LWORK >= MAX(1,2*N). لتحقيق الأداء الأمثل
!>          LWORK >= N*(1+NB)، حيث NB هو حجم الكتلة الأمثل.
!>
!>          إذا كان LWORK = -1، فسيتم افتراض استعلام مساحة العمل؛ يقوم الإجراء
!>          بحساب حجم مساحة العمل الأمثل فقط، ويعيد
!>          هذه القيمة كالإدخال الأول لمصفوفة WORK، ولا يصدر رسالة خطأ
!>          متعلقة بـ LWORK بواسطة XERBLA.
!>

INFO

!>          INFO هو INTEGER
!>          = 0:  انتهاء ناجح
!>          < 0:  إذا كان INFO = -i، فإن المعلمة i كانت ذات قيمة غير قانونية.
!>
