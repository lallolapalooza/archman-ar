توثيق الوظيفة/الروتين الفرعي

subroutine cspt02 (character uplo, integer n, integer nrhs, complex, dimension( * ) a, complex, dimension( ldx, * ) x, integer ldx, complex, dimension( ldb, * ) b, integer ldb, real, dimension( * ) rwork, real resid)

CSPT02

الغرض:

!>
!> CSPT02 يحسب الباقي في حل نظام المعادلات الخطية المتماثل المعقد  A*x = b
!> عند استخدام تخزين مضغوط للمصفوفة المعاملة. النسبة المحسوبة هي
!>
!>    RESID = norm( B - A*X ) / ( norm(A) * norm(X) * EPS).
!>
!> حيث EPS هي دقة الآلة.
!>

المعلمات

UPLO 
!>          UPLO is CHARACTER*1
!>          يحدد ما إذا كان الجزء العلوي أو السفلي المثلثي من المصفوفة
!>          المتماثلة المعقدة A مخزنًا:
!>          = 'U':  مثلثي علوي
!>          = 'L':  مثلثي سفلي
!>

N

!>          N is INTEGER
!>          عدد الصفوف والأعمدة للمصفوفة A.  N >= 0.
!>

NRHS

!>          NRHS is INTEGER
!>          عدد الأعمدة لـ B، مصفوفة الطرف الأيمن.
!>          NRHS >= 0.
!>

A

!>          A is COMPLEX array, dimension (N*(N+1)/2)
!>          المصفوفة المتماثلة المعقدة الأصلية A، مخزنة كمصفوفة
!>          مثلثية مضغوطة.
!>

X

!>          X is COMPLEX array, dimension (LDX,NRHS)
!>          متجهات الحلول المحسوبة لنظام المعادلات الخطية.
!>

LDX

!>          LDX is INTEGER
!>          البعد الرائد للمصفوفة X.   LDX >= max(1,N).
!>

B

!>          B is COMPLEX array, dimension (LDB,NRHS)
!>          عند الدخول، متجهات الطرف الأيمن لنظام المعادلات الخطية.
!>          عند الخروج، يتم استبدال B بـ B - A*X.
!>

LDB

!>          LDB is INTEGER
!>          البعد الرائد للمصفوفة B.  LDB >= max(1,N).
!>

RWORK

!>          RWORK is REAL array, dimension (N)
!>

RESID

!>          RESID is REAL
!>          القيمة القصوى لعدد الأطراف اليمنى لـ
!>          norm(B - A*X) / ( norm(A) * norm(X) * EPS ).
!>
