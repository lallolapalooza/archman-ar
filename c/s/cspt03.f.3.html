توثيق الوظائف/الروتينات الفرعية

subroutine cspt03 (character uplo, integer n, complex, dimension( * ) a, complex, dimension( * ) ainv, complex, dimension( ldw, * ) work, integer ldw, real, dimension( * ) rwork, real rcond, real resid)

CSPT03

الغرض:

!>
!> CSPT03 يحسب الباقي لمصفوفة متماثلة مركبة مضغوطة مضروبة في معكوسها:
!>    norm( I - A*AINV ) / ( N * norm(A) * norm(AINV) * EPS ),
!> حيث EPS هو الخطأ الرياضي.
!>

المعاملات

UPLO 
!>          UPLO هو CHARACTER*1
!>          يحدد ما إذا كان الجزء العلوي أو السفلي المثلثي للمصفوفة A
!>          المتماثلة المركبة مخزنًا:
!>          = 'U':  مثلثي علوي
!>          = 'L':  مثلثي سفلي
!>

N

!>          N هو INTEGER
!>          عدد الصفوف والأعمدة للمصفوفة A.  N >= 0.
!>

A

!>          A هو مصفوفة COMPLEX، أبعادها (N*(N+1)/2)
!>          المصفوفة الأصلية المتماثلة المركبة A، مخزنة كمصفوفة
!>          مثلثية مضغوطة.
!>

AINV

!>          AINV هو مصفوفة COMPLEX، أبعادها (N*(N+1)/2)
!>          معكوس المصفوفة A (المتماثل)، مخزن كمصفوفة
!>          مثلثية مضغوطة.
!>

WORK

!>          WORK هو مصفوفة COMPLEX، أبعادها (LDW,N)
!>

LDW

!>          LDW هو INTEGER
!>          البعد الرئيسي للمصفوفة WORK.  LDW >= max(1,N).
!>

RWORK

!>          RWORK هو مصفوفة REAL، أبعادها (N)
!>

RCOND

!>          RCOND هو REAL
!>          مقلوب رقم شرط المصفوفة A، محسوب كـ
!>          ( 1/norm(A) ) / norm(AINV).
!>

RESID

!>          RESID هو REAL
!>          norm(I - A*AINV) / ( N * norm(A) * norm(AINV) * EPS )
!>
