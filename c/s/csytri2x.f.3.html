subroutine csytri2x (character uplo, integer n, complex, dimension( lda, * ) a, integer lda, integer, dimension( * ) ipiv, complex, dimension( n+nb+1,* ) work, integer nb, integer info)

CSYTRI2X

الغرض:

!>
!> CSYTRI2X يحسب معكوس مصفوفة حقيقية متناظرة غير محددة
!> A باستخدام التحلل A = U*D*U**T أو A = L*D*L**T الذي تم حسابه بواسطة
!> CSYTRF.
!>

المعاملات

UPLO 
!>          UPLO هو CHARACTER*1
!>          يحدد ما إذا كانت تفاصيل التحلل مخزنة
!>          كمصفوفة مثلثة علوية أو سفلية.
!>          = 'U':  مثلثة علوية، الشكل هو A = U*D*U**T؛
!>          = 'L':  مثلثة سفلية، الشكل هو A = L*D*L**T.
!>

N

!>          N هو INTEGER
!>          رتبة المصفوفة A.  N >= 0.
!>

A

!>          A هي مصفوفة COMPLEX، البعد (LDA,N)
!>          عند الإدخال، مصفوفة D القطرية NNB والعوامل
!>          المستخدمة للحصول على العامل U أو L كما تم حسابه بواسطة CSYTRF.
!>
!>          عند الخروج، إذا كان INFO = 0، فإن معكوس المصفوفة الأصلية
!>          (المتناظرة).  إذا كان UPLO = 'U'، يتم تشكيل الجزء المثلث العلوي
!>          من المعكوس ولا تتم الإشارة إلى الجزء الموجود أسفل القطر في A؛
!>          إذا كان UPLO = 'L' يتم تشكيل الجزء المثلث السفلي
!>          من المعكوس ولا تتم الإشارة إلى الجزء الموجود فوق القطر في A.
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرائد للمصفوفة A.  LDA >= max(1,N).
!>

IPIV

!>          IPIV هي مصفوفة INTEGER، البعد (N)
!>          تفاصيل التبادلات وهيكل D المكون من NNB
!>          كما تم تحديده بواسطة CSYTRF.
!>

WORK

!>          WORK هي مصفوفة COMPLEX، البعد (N+NB+1,NB+3)
!>

NB

!>          NB هو INTEGER
!>          حجم الكتلة
!>

INFO

!>          INFO هو INTEGER
!>          = 0: خروج ناجح
!>          < 0: إذا كان INFO = -i، فإن المعامل i كان له قيمة غير قانونية
!>          > 0: إذا كان INFO = i، فإن D(i,i) = 0؛ المصفوفة مفردة ولا يمكن
!>               حساب معكوسها.
!>
