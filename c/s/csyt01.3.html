subroutine csyt01 (character uplo, integer n, complex, dimension( lda, * ) a, integer lda, complex, dimension( ldafac, * ) afac, integer ldafac, integer, dimension( * ) ipiv, complex, dimension( ldc, * ) c, integer ldc, real, dimension( * ) rwork, real resid)

CSYT01

الغرض:

!>
!> CSYT01 يعيد بناء مصفوفة A المعقدة المتماثلة غير المحددة من تحليلها
!> L*D*L' أو U*D*U' وتحسب الباقي
!>    norm( C - A ) / ( N * norm(A) * EPS ),
!> حيث C هي المصفوفة المعاد بناؤها، EPS هو الخطأ المطبعي،
!> L' هو منقول L، و U' هو منقول U.
!>

المعلمات

UPLO 
!>          UPLO هو CHARACTER*1
!>          يحدد ما إذا كان الجزء العلوي أو السفلي المثلثي من
!>          المصفوفة المعقدة المتماثلة A مخزنًا:
!>          = 'U':  علوي مثلثي
!>          = 'L':  سفلي مثلثي
!>

N

!>          N هو INTEGER
!>          عدد الصفوف والأعمدة للمصفوفة A.  N >= 0.
!>

A

!>          A هي مصفوفة COMPLEX، أبعادها (LDA,N)
!>          المصفوفة المعقدة المتماثلة الأصلية A.
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرئيسي للمصفوفة A.  LDA >= max(1,N)
!>

AFAC

!>          AFAC هي مصفوفة COMPLEX، أبعادها (LDAFAC,N)
!>          الشكل المفكك للمصفوفة A.  AFAC تحتوي على مصفوفة القطرية
!>          D والمضاعفات المستخدمة للحصول على العامل L أو U من تحليل
!>          L*D*L' أو U*D*U' كما تم حسابه بواسطة CSYTRF.
!>

LDAFAC

!>          LDAFAC هو INTEGER
!>          البعد الرئيسي للمصفوفة AFAC.  LDAFAC >= max(1,N).
!>

IPIV

!>          IPIV هي مصفوفة INTEGER، أبعادها (N)
!>          فهارس المحور من CSYTRF.
!>

C

!>          C هي مصفوفة COMPLEX، أبعادها (LDC,N)
!>

LDC

!>          LDC هو INTEGER
!>          البعد الرئيسي للمصفوفة C.  LDC >= max(1,N).
!>

RWORK

!>          RWORK هي مصفوفة REAL، أبعادها (N)
!>

RESID

!>          RESID هو REAL
!>          إذا كان UPLO = 'L'، norm(L*D*L' - A) / ( N * norm(A) * EPS )
!>          إذا كان UPLO = 'U'، norm(U*D*U' - A) / ( N * norm(A) * EPS )
!>
