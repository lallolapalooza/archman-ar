subroutine cspr (uplo, n, alpha, x, incx, ap)


CSPR تقوم بتحديث المصفوفة المتماثلة المعبأة برتبة-1.

توثيق الوظيفة/الروتين الفرعي

subroutine cspr (uplo حرف, n عدد صحيح, alpha مركب, x مركب, dimension( * ) x, incx عدد صحيح, ap مركب, dimension( * ) ap)

CSPR تقوم بتحديث المصفوفة المتماثلة المعبأة برتبة-1.

الغرض:

!>
!> CSPR    تقوم بعملية الرتبة 1 المتماثلة
!>
!>    A := alpha*x*x**H + A,
!>
!> حيث alpha هي قيمة قياسية مركبة، x هو متجه من n عنصر، و A هي
!> مصفوفة متماثلة بحجم n×n، مقدمة في شكل مضغوط.
!>

المعاملات

UPLO 
!>          UPLO هو CHARACTER*1
!>           عند الدخول، يحدد UPLO ما إذا كان الجزء المثلثي العلوي أو السفلي
!>           من المصفوفة A مزودًا في المصفوفة AP كما يلي:
!>
!>              UPLO = 'U' أو 'u'   يتم توفير الجزء المثلثي العلوي من A
!>                                  في AP.
!>
!>              UPLO = 'L' أو 'l'   يتم توفير الجزء المثلثي السفلي من A
!>                                  في AP.
!>
!>           لم يتغير عند الخروج.
!>

N

!>          N هو INTEGER
!>           عند الدخول، يحدد N رتبة المصفوفة A.
!>           يجب أن يكون N على الأقل صفر.
!>           لم يتغير عند الخروج.
!>

ALPHA

!>          ALPHA هو COMPLEX
!>           عند الدخول، يحدد ALPHA القيمة القياسية alpha.
!>           لم يتغير عند الخروج.
!>

X

!>          X هي مصفوفة COMPLEX، ذات بعد على الأقل
!>           ( 1 + ( N - 1 )*abs( INCX ) ).
!>           قبل الدخول، يجب أن تحتوي المصفوفة X ذات الزيادة على المتجه x
!>           ذي الـ N عنصر.
!>           لم يتغير عند الخروج.
!>

INCX

!>          INCX هو INTEGER
!>           عند الدخول، يحدد INCX الزيادة لعناصر X. يجب ألا يكون INCX صفرًا.
!>           لم يتغير عند الخروج.
!>

AP

!>          AP هي مصفوفة COMPLEX، ذات بعد على الأقل
!>           ( ( N*( N + 1 ) )/2 ).
!>           قبل الدخول، مع UPLO = 'U' أو 'u'، يجب أن تحتوي المصفوفة AP
!>           على الجزء المثلثي العلوي من المصفوفة المتماثلة
!>           معبأ بشكل متسلسل، عمودًا بعمود، بحيث تحتوي AP( 1 )
!>           على a( 1, 1 )، وتحتوي AP( 2 ) و AP( 3 ) على a( 1, 2 )
!>           و a( 2, 2 ) على التوالي، وهكذا. عند الخروج، يتم استبدال محتويات المصفوفة
!>           AP بالجزء المثلثي العلوي من المصفوفة المحدثة.
!>           قبل الدخول، مع UPLO = 'L' أو 'l'، يجب أن تحتوي المصفوفة AP
!>           على الجزء المثلثي السفلي من المصفوفة المتماثلة
!>           معبأ بشكل متسلسل، عمودًا بعمود، بحيث تحتوي AP( 1 )
!>           على a( 1, 1 )، وتحتوي AP( 2 ) و AP( 3 ) على a( 2, 1 )
!>           و a( 3, 1 ) على التوالي، وهكذا. عند الخروج، يتم استبدال محتويات المصفوفة
!>           AP بالجزء المثلثي السفلي من المصفوفة المحدثة.
!>           لاحظ أن الأجزاء التخيلية للعناصر القطرية لا تحتاج إلى تعيينها،
!>           ويفترض أنها صفر، وعند الخروج يتم تعيينها إلى صفر.
!>
