subroutine csytrs_3 (character uplo, integer n, integer nrhs, complex, dimension( lda, * ) a, integer lda, complex, dimension( * ) e, integer, dimension( * ) ipiv, complex, dimension( ldb, * ) b, integer ldb, integer info)

CSYTRS_3

الغرض:

!> CSYTRS_3 تحل نظام المعادلات الخطية A * X = B مع مصفوفة متناظرة مركبة A
!> باستخدام التحليل الذي تم حسابه بواسطة CSYTRF_RK أو CSYTRF_BK:
!>
!>    A = P*U*D*(U**T)*(P**T) أو A = P*L*D*(L**T)*(P**T),
!>
!> حيث U (أو L) هي مصفوفة وحدوية علوية (أو سفلية)،
!> U**T (أو L**T) هي منقولة U (أو L)، P هي مصفوفة تبديل، P**T هي منقولة P،
!> و D هي مصفوفة متناظرة وقطرية مجمعة مع كتل قطرية 1x1 و 2x2.
!>
!> هذه الخوارزمية تستخدم BLAS المستوى 3.
!>

المعاملات

UPLO 
!>          UPLO هو CHARACTER*1
!>          يحدد ما إذا كانت تفاصيل التحليل مخزنة كمصفوفة مثلثية علوية
!>          أو سفلية:
!>          = 'U':  مثلثية علوية، الشكل هو A = P*U*D*(U**T)*(P**T);
!>          = 'L':  مثلثية سفلية، الشكل هو A = P*L*D*(L**T)*(P**T).
!>

N

!>          N هو INTEGER
!>          رتبة المصفوفة A.  N >= 0.
!>

NRHS

!>          NRHS هو INTEGER
!>          عدد الأطراف اليمنى، أي عدد أعمدة المصفوفة B.  NRHS >= 0.
!>

A

!>          A هي مصفوفة COMPLEX، الأبعاد (LDA,N)
!>          قطر المصفوفة القطرية المجمعة D وعوامل U أو L
!>          كما تم حسابها بواسطة CSYTRF_RK و CSYTRF_BK:
!>            أ) فقط العناصر القطرية للمصفوفة القطرية المجمعة المتناظرة
!>               D على قطر A، أي D(k,k) = A(k,k);
!>               (يجب توفير عناصر فوق القطر (أو تحت القطر) للمصفوفة D
!>                في مدخلات المصفوفة E)، و
!>            ب) إذا كان UPLO = 'U': العامل U في الجزء فوق القطري من A.
!>               إذا كان UPLO = 'L': العامل L في الجزء تحت القطري من A.
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرئيسي للمصفوفة A.  LDA >= max(1,N).
!>

E

!>          E هي مصفوفة COMPLEX، الأبعاد (N)
!>          عند المدخل، تحتوي على عناصر فوق القطر (أو تحت القطر)
!>          للمصفوفة القطرية المجمعة المتناظرة D
!>          مع كتل قطرية 1x1 أو 2x2، حيث
!>          إذا كان UPLO = 'U': E(i) = D(i-1,i),i=2:N, E(1) غير مشار إليها؛
!>          إذا كان UPLO = 'L': E(i) = D(i+1,i),i=1:N-1, E(N) غير مشار إليها.
!>
!>          ملاحظة: لكتلة قطرية 1x1 D(k)، حيث
!>          1 <= k <= N، لا يتم الإشارة إلى العنصر E(k) في كل من
!>          حالات UPLO = 'U' أو UPLO = 'L'.
!>

IPIV

!>          IPIV هي مصفوفة INTEGER، الأبعاد (N)
!>          تفاصيل التبادلات وهيكل الكتل لـ D
!>          كما تم تحديده بواسطة CSYTRF_RK أو CSYTRF_BK.
!>

B

!>          B هي مصفوفة COMPLEX، الأبعاد (LDB,NRHS)
!>          عند المدخل، مصفوفة الطرف الأيمن B.
!>          عند الخروج، مصفوفة الحل X.
!>

LDB

!>          LDB هو INTEGER
!>          البعد الرئيسي للمصفوفة B.  LDB >= max(1,N).
!>

INFO

!>          INFO هو INTEGER
!>          = 0:  خروج ناجح
!>          < 0:  إذا كان INFO = -i، فإن المعامل i كان لديه قيمة غير قانونية
!>
