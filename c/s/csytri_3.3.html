subroutine csytri_3 (character uplo, integer n, complex, dimension( lda, * ) a, integer lda, complex, dimension( * ) e, integer, dimension( * ) ipiv, complex, dimension( * ) work, integer lwork, integer info)

CSYTRI_3

الغرض:

!> CSYTRI_3 يحسب معكوس مصفوفة غير محددة متماثلة مركبة A باستخدام التحليل الذي تم حسابه بواسطة CSYTRF_RK أو CSYTRF_BK:
!>
!>     A = P*U*D*(U**T)*(P**T) أو A = P*L*D*(L**T)*(P**T),
!>
!> حيث U (أو L) هي مصفوفة مثلثة علوية (أو سفلية) وحدة،
!> U**T (أو L**T) هي منقولة U (أو L)، P هي مصفوفة تبديل، P**T هي منقولة P، و D هي مصفوفة متماثلة وقطرية كتلة مع كتل قطرية 1x1 و 2x2.
!>
!> CSYTRI_3 يحدد البعد الرئيسي لمساحة العمل قبل استدعاء
!> CSYTRI_3X الذي يحسب المعكوس فعليًا. هذه هي النسخة المجمعة
!> من الخوارزمية، تستدعي BLAS من المستوى 3.
!>

المعاملات

UPLO 
!>          UPLO هو CHARACTER*1
!>          يحدد ما إذا كانت تفاصيل التحليل
!>          مخزنة كمصفوفة مثلثة علوية أو سفلية.
!>          = 'U':  يتم تخزين المثلث العلوي من A؛
!>          = 'L':  يتم تخزين المثلث السفلي من A.
!>

N

!>          N هو INTEGER
!>          رتبة المصفوفة A. N >= 0.
!>

A

!>          A هي مصفوفة COMPLEX، حجمها (LDA,N)
!>          عند الدخول، القطر للمصفوفة القطرية الكتلة D والعوامل U أو L كما تم حسابها بواسطة CSYTRF_RK و CSYTRF_BK:
!>            أ) فقط العناصر القطرية للمصفوفة القطرية الكتلة المتماثلة
!>               D على قطر A، أي D(k,k) = A(k,k)؛
!>               (يجب توفير عناصر فوق القطر (أو تحت القطر) لـ D
!>                عند الدخول في المصفوفة E)، و
!>            ب) إذا كان UPLO = 'U': العامل U في الجزء فوق القطري لـ A.
!>               إذا كان UPLO = 'L': العامل L في الجزء تحت القطري لـ A.
!>
!>          عند الخروج، إذا كان INFO = 0، فإن معكوس المصفوفة الأصلية المتماثلة.
!>             إذا كان UPLO = 'U': يتم تشكيل الجزء العلوي المثلثي من المعكوس
!>             ولا يتم الرجوع إلى الجزء السفلي من A؛
!>             إذا كان UPLO = 'L': يتم تشكيل الجزء السفلي المثلثي من المعكوس
!>             ولا يتم الرجوع إلى الجزء العلوي من A.
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرئيسي للمصفوفة A. LDA >= max(1,N).
!>

E

!>          E هي مصفوفة COMPLEX، حجمها (N)
!>          عند الدخول، تحتوي على العناصر فوق القطرية (أو تحت القطرية)
!>          للمصفوفة القطرية الكتلة المتماثلة D
!>          مع كتل قطرية 1x1 أو 2x2، حيث
!>          إذا كان UPLO = 'U': E(i) = D(i-1,i), i=2:N, E(1) لا يتم الرجوع إليها؛
!>          إذا كان UPLO = 'L': E(i) = D(i+1,i), i=1:N-1, E(N) لا يتم الرجوع إليها.
!>
!>          ملاحظة: لكتلة قطرية 1x1 D(k)، حيث
!>          1 <= k <= N، لا يتم الرجوع إلى العنصر E(k) في كلتا
!>          حالات UPLO = 'U' أو UPLO = 'L'.
!>

IPIV

!>          IPIV هي مصفوفة INTEGER، حجمها (N)
!>          تفاصيل التبادلات وهيكل الكتلة لـ D
!>          كما تم تحديده بواسطة CSYTRF_RK أو CSYTRF_BK.
!>

WORK

!>          WORK هي مصفوفة COMPLEX، حجمها (N+NB+1)*(NB+3).
!>          عند الخروج، إذا كان INFO = 0، فإن WORK(1) يعيد LWORK الأمثل.
!>

LWORK

!>          LWORK هو INTEGER
!>          طول WORK. LWORK >= (N+NB+1)*(NB+3).
!>
!>          إذا كان LDWORK = -1، فمن المفترض استعلام مساحة العمل؛
!>          يحدد الروتين فقط الحجم الأمثل لمساحة العمل
!>          المثلى للمصفوفة WORK، ويعيد هذا القيمة كمدخل أول
!>          للمصفوفة WORK، ولا يصدر XERBLA أي رسالة خطأ متعلقة بـ
!>          LWORK.
!>

INFO

!>          INFO هو INTEGER
!>          = 0: خروج ناجح
!>          < 0: إذا كان INFO = -i، فإن المعامل i كان له قيمة غير قانونية
!>          > 0: إذا كان INFO = i، فإن D(i,i) = 0؛ المصفوفة مفردة ولا يمكن حساب معكوسها.
!>
