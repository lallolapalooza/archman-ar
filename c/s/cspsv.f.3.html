subroutine cspsv (character uplo, integer n, integer nrhs, complex, dimension( * ) ap, integer, dimension( * ) ipiv, complex, dimension( ldb, * ) b, integer ldb, integer info)

CSPSV يحسب حل نظام المعادلات الخطية A * X = B للمصفوفات الأخرى

الغرض:

!>
!> CSPSV يحسب حل نظام المعادلات الخطية المعقدة
!>    A * X = B,
!> حيث A هي مصفوفة متماثلة N-by-N مخزنة بتنسيق مدمج، و X و B هي مصفوفات N-by-NRHS.
!>
!> تستخدم طريقة التنقيط القطري لتحليل A إلى
!>    A = U * D * U**T,  إذا كان UPLO = 'U', أو
!>    A = L * D * L**T,  إذا كان UPLO = 'L',
!> حيث U (أو L) هو ناتج ضرب مصفوفات التبديل والمصفوفات المثلثية العلوية (السفلية) الوحدوية،
!> و D متماثلة وكتلة قطرية ذات كتل قطرية 1-by-1 و 2-by-2. ثم يتم استخدام الشكل المحلل لـ A
!> لحل نظام المعادلات A * X = B.
!>

المعلمات

UPLO 
!>          UPLO هو CHARACTER*1
!>          = 'U':  يتم تخزين المثلث العلوي لـ A؛
!>          = 'L':  يتم تخزين المثلث السفلي لـ A.
!>

N

!>          N هو INTEGER
!>          عدد المعادلات الخطية، أي ترتيب المصفوفة A. N >= 0.
!>

NRHS

!>          NRHS هو INTEGER
!>          عدد الأطراف اليمنى، أي عدد أعمدة المصفوفة B. NRHS >= 0.
!>

AP

!>          AP هو مصفوفة COMPLEX، ذات بعد (N*(N+1)/2)
!>          عند الإدخال، المثلث العلوي أو السفلي للمصفوفة المتماثلة
!>          A، مدمج عموديًا في مصفوفة خطية. يتم تخزين العمود j من A
!>          في المصفوفة AP بالشكل التالي:
!>          إذا كان UPLO = 'U', AP(i + (j-1)*j/2) = A(i,j) لـ 1<=i<=j؛
!>          إذا كان UPLO = 'L', AP(i + (j-1)*(2n-j)/2) = A(i,j) لـ j<=i<=n.
!>          انظر أدناه لمزيد من التفاصيل.
!>
!>          عند الخروج، يتم تخزين المصفوفة القطرية الكتلة D والمضاعفات المستخدمة
!>          للحصول على العامل U أو L من التحليل
!>          A = U*D*U**T أو A = L*D*L**T  كما تم حسابه بواسطة CSPTRF،
!>          كمصفوفة مثلثية مدمجة بنفس تنسيق التخزين مثل A.
!>

IPIV

!>          IPIV هو مصفوفة INTEGER، ذات بعد (N)
!>          تفاصيل التبديلات وهيكل الكتلة لـ D، كما
!>          تم تحديده بواسطة CSPTRF. إذا كان IPIV(k) > 0، فقد تم تبديل
!>          الصفوف والأعمدة k و IPIV(k)، و D(k,k) هي كتلة قطرية 1-by-1.
!>          إذا كان UPLO = 'U' وكان IPIV(k) = IPIV(k-1) < 0، فقد تم تبديل
!>          الصفوف والأعمدة k-1 و -IPIV(k) و D(k-1:k,k-1:k) هي كتلة قطرية 2-by-2.
!>          إذا كان UPLO = 'L' وكان IPIV(k) = IPIV(k+1) < 0، فقد تم تبديل
!>          الصفوف والأعمدة k+1 و -IPIV(k) و D(k:k+1,k:k+1) هي كتلة قطرية 2-by-2.
!>

B

!>          B هي مصفوفة COMPLEX، ذات بعد (LDB,NRHS)
!>          عند الإدخال، مصفوفة الطرف الأيمن B ذات البعد N-by-NRHS.
!>          عند الخروج، إذا كان INFO = 0، فإن مصفوفة الحل X ذات البعد N-by-NRHS.
!>

LDB

!>          LDB هو INTEGER
!>          البعد الرئيسي للمصفوفة B. LDB >= max(1,N).
!>

INFO

!>          INFO هو INTEGER
!>          = 0:  خروج ناجح
!>          < 0:  إذا كان INFO = -i، فقد كان المعامل i ذو قيمة غير قانونية
!>          > 0:  إذا كان INFO = i، فإن D(i,i) يساوي صفر تمامًا.
!>                تم إكمال التحليل، ولكن المصفوفة القطرية الكتلة D
!>                منفردة تمامًا، لذلك لم يكن من الممكن حساب الحل.
!>
