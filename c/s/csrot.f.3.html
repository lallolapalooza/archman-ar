subroutine csrot (integer n, complex, dimension( * ) cx, integer incx, complex, dimension( * ) cy, integer incy, real c, real s)

CSROT

الغرض:

!>
!> CSROT تطبق دوران المستوى، حيث يكون cos و sin (c و s) أعدادًا حقيقية
!> والمتجهات cx و cy أعدادًا مركبة.
!> جاك دونجارا، لينباك، 3/11/78.
!>

المعاملات

N 
!>          N هو عدد صحيح (INTEGER)
!>           عند الإدخال، تحدد N رتبة المتجهات cx و cy.
!>           يجب أن تكون N على الأقل صفرًا.
!>

CX

!>          CX هو مصفوفة مركبة (COMPLEX)، حجمها على الأقل
!>           ( 1 + ( N - 1 )*abs( INCX ) ).
!>           قبل الإدخال، يجب أن تحتوي المصفوفة المُنظمة CX على متجه cx ذي العناصر n.
!>           عند الخروج، يتم استبدال CX بمتجه cx المحدّث.
!>

INCX

!>          INCX هو عدد صحيح (INTEGER)
!>           عند الإدخال، تحدد INCX الزيادة لعناصر CX.
!>           يجب ألا تكون INCX صفرًا.
!>

CY

!>          CY هو مصفوفة مركبة (COMPLEX)، حجمها على الأقل
!>           ( 1 + ( N - 1 )*abs( INCY ) ).
!>           قبل الإدخال، يجب أن تحتوي المصفوفة المُنظمة CY على متجه cy ذي العناصر n.
!>           عند الخروج، يتم استبدال CY بمتجه cy المحدّث.
!>

INCY

!>          INCY هو عدد صحيح (INTEGER)
!>           عند الإدخال، تحدد INCY الزيادة لعناصر CY.
!>           يجب ألا تكون INCY صفرًا.
!>

C

!>          C هو عدد حقيقي (REAL)
!>           عند الإدخال، يحدد C جيب التمام (cos).
!>

S

!>          S هو عدد حقيقي (REAL)
!>           عند الإدخال، يحدد S جيب الزاوية (sin).
!>
