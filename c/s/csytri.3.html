توثيق الوظيفة/الروتين الفرعي

subroutine csytri (character uplo, integer n, complex, dimension( lda, * ) a, integer lda, integer, dimension( * ) ipiv, complex, dimension( * ) work, integer info)

CSYTRI

الغرض:

!>
!> CSYTRI تحسب معكوس مصفوفة معقدة متناظرة غير محددة A باستخدام التحليل A = U*D*U**T أو A = L*D*L**T المحسوب بواسطة
!> CSYTRF.
!>

المعلمات

UPLO 
!>          UPLO هو CHARACTER*1
!>          يحدد ما إذا كانت تفاصيل التحليل مخزنة
!>          كمصفوفة مثلثية عليا أو سفلى.
!>          = 'U':  مثلثية عليا، الشكل هو A = U*D*U**T؛
!>          = 'L':  مثلثية سفلى، الشكل هو A = L*D*L**T.
!>

N

!>          N هو INTEGER
!>          رتبة المصفوفة A.  N >= 0.
!>

A

!>          A هي مصفوفة COMPLEX، البعد (LDA,N)
!>          عند الإدخال، المصفوفة القطرية الجزئية D والعوامل
!>          المستخدمة للحصول على العامل U أو L كما حسبها CSYTRF.
!>
!>          عند الخروج، إذا كان INFO = 0، معكوس المصفوفة الأصلية
!>          (المتناظرة). إذا كان UPLO = 'U'، فإن الجزء المثلثي العلوي
!>          للمعكوس يتم تكوينه والجزء من A أسفل القطر لا يتم الرجوع إليه؛
!>          إذا كان UPLO = 'L'، فإن الجزء المثلثي السفلي
!>          للمعكوس يتم تكوينه والجزء من A فوق القطر
!>          لا يتم الرجوع إليه.
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرئيسي للمصفوفة A.  LDA >= max(1,N).
!>

IPIV

!>          IPIV هي مصفوفة INTEGER، البعد (N)
!>          تفاصيل التبادلات وهيكل D الجزئي
!>          كما تم تحديده بواسطة CSYTRF.
!>

WORK

!>          WORK هي مصفوفة COMPLEX، البعد (2*N)
!>

INFO

!>          INFO هو INTEGER
!>          = 0: خروج ناجح
!>          < 0: إذا كان INFO = -i، فإن المعامل i كان له قيمة غير قانونية
!>          > 0: إذا كان INFO = i، فإن D(i,i) = 0؛ المصفوفة شاذة ولا يمكن حساب معكوسها.
!>
