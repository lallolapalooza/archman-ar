subroutine cstegr (character jobz, character range, integer n, real, dimension( * ) d, real, dimension( * ) e, real vl, real vu, integer il, integer iu, real abstol, integer m, real, dimension( * ) w, complex, dimension( ldz, * ) z, integer ldz, integer, dimension( * ) isuppz, real, dimension( * ) work, integer lwork, integer, dimension( * ) iwork, integer liwork, integer info)

CSTEGR

الغرض:

!>
!> CSTEGR تحسب قيم المتجهات الذاتية المختارة، واختيارياً، المتجهات الذاتية
!> لمصفوفة متناظرة ثلاثية القطر الحقيقية T. أي مصفوفة غير مختزلة كهذه لها
!> مجموعة محددة جيداً من القيم الذاتية الحقيقية المختلفة زوجياً، والمتجهات الذاتية
!> الحقيقية المقابلة تكون متعامدة زوجياً.
!>
!> يمكن حساب الطيف بالكامل أو جزئيًا عن طريق تحديد
!> إما فترة (VL,VU] أو نطاق من المؤشرات IL:IU للقيم الذاتية المطلوبة.
!>
!> CSTEGR هو غلاف توافق حول الروتين المحسن CSTEMR.
!> انظر SSTEMR لمزيد من التفاصيل.
!>
!> أحد التغييرات المهمة هو أن المعامل ABSTOL لم يعد يقدم أي فائدة وبالتالي
!> لم يعد مستخدماً.
!>
!> ملاحظة: يعمل CSTEGR و CSTEMR فقط على الأجهزة التي تتبع
!> معيار الفاصلة العائمة IEEE-754 في معالجتها للمالانهايات والقيم غير الرقمية (NaNs).
!> قد ينشئ التنفيذ العادي هذه القيم الاستثنائية وبالتالي
!> قد يتعطل بسبب استثناء الفاصلة العائمة في البيئات التي
!> لا تتوافق مع معيار IEEE-754.
!>

المعاملات

JOBZ 
!>          JOBZ is CHARACTER*1
!>          = 'N':  حساب القيم الذاتية فقط؛
!>          = 'V':  حساب القيم الذاتية والمتجهات الذاتية.
!>

RANGE

!>          RANGE is CHARACTER*1
!>          = 'A': سيتم العثور على جميع القيم الذاتية.
!>          = 'V': سيتم العثور على جميع القيم الذاتية في الفترة نصف المفتوحة (VL,VU].
!>          = 'I': سيتم العثور على القيم الذاتية من IL-th إلى IU-th.
!>

N

!>          N is INTEGER
!>          رتبة المصفوفة. N >= 0.
!>

D

!>          D is REAL array, dimension (N)
!>          عند الإدخال، العناصر القطرية N للمصفوفة الثلاثية القطر
!>          T. عند الخروج، يتم الكتابة فوق D.
!>

E

!>          E is REAL array, dimension (N)
!>          عند الإدخال، عناصر القطر الفرعي (N-1) للمصفوفة الثلاثية القطر
!>          T في العناصر من 1 إلى N-1 من E. لا يلزم تعيين E(N) عند
!>          الإدخال، ولكن يتم استخدامه داخليًا كمساحة عمل.
!>          عند الخروج، يتم الكتابة فوق E.
!>

VL

!>          VL is REAL
!>
!>          إذا كانت RANGE='V'، الحد الأدنى للفترة المراد
!>          البحث فيها عن القيم الذاتية. VL < VU.
!>          لا يتم الرجوع إليها إذا كانت RANGE = 'A' أو 'I'.
!>

VU

!>          VU is REAL
!>
!>          إذا كانت RANGE='V'، الحد الأعلى للفترة المراد
!>          البحث فيها عن القيم الذاتية. VL < VU.
!>          لا يتم الرجوع إليها إذا كانت RANGE = 'A' أو 'I'.
!>

IL

!>          IL is INTEGER
!>
!>          إذا كانت RANGE='I'، مؤشر
!>          أصغر قيمة ذاتية سيتم إرجاعها.
!>          1 <= IL <= IU <= N، إذا كان N > 0.
!>          لا يتم الرجوع إليها إذا كانت RANGE = 'A' أو 'V'.
!>

IU

!>          IU is INTEGER
!>
!>          إذا كانت RANGE='I'، مؤشر
!>          أكبر قيمة ذاتية سيتم إرجاعها.
!>          1 <= IL <= IU <= N، إذا كان N > 0.
!>          لا يتم الرجوع إليها إذا كانت RANGE = 'A' أو 'V'.
!>

ABSTOL

!>          ABSTOL is REAL
!>          غير مستخدم. كان الخطأ المطلق لتحمل
!>          القيم الذاتية/المتجهات الذاتية في الإصدارات السابقة.
!>

M

!>          M is INTEGER
!>          العدد الإجمالي للقيم الذاتية التي تم العثور عليها. 0 <= M <= N.
!>          إذا كانت RANGE = 'A'، فإن M = N، وإذا كانت RANGE = 'I'، فإن M = IU-IL+1.
!>

W

!>          W is REAL array, dimension (N)
!>          تحتوي العناصر الأولى M على القيم الذاتية المختارة في
!>          ترتيب تصاعدي.
!>

Z

!>          Z is COMPLEX array, dimension (LDZ, max(1,M) )
!>          إذا كانت JOBZ = 'V'، وإذا كان INFO = 0، فإن الأعمدة M الأولى من Z
!>          تحتوي على المتجهات الذاتية المتعامدة للمصفوفة T
!>          المقابلة للقيم الذاتية المختارة، مع عمود Z i-th
!>          الذي يحمل المتجه الذاتي المرتبط بـ W(i).
!>          إذا كانت JOBZ = 'N'، فلا يتم الرجوع إلى Z.
!>          ملاحظة: يجب على المستخدم التأكد من توفير ما لا يقل عن max(1,M) عمودًا
!>          في المصفوفة Z؛ إذا كانت RANGE = 'V'، فإن القيمة الدقيقة لـ M
!>          ليست معروفة مسبقًا ويجب استخدام حد أعلى.
!>          توفير N أعمدة آمن دائمًا.
!>

LDZ

!>          LDZ is INTEGER
!>          البعد الرائد للمصفوفة Z. LDZ >= 1، وإذا
!>          كان JOBZ = 'V'، فإن LDZ >= max(1,N).
!>

ISUPPZ

!>          ISUPPZ is INTEGER array, dimension ( 2*max(1,M) )
!>          دعم المتجهات الذاتية في Z، أي المؤشرات
!>          التي تشير إلى العناصر غير الصفرية في Z. المتجه الذاتي المحسوب i-th
!>          يكون غير صفري فقط في العناصر ISUPPZ( 2*i-1 ) إلى
!>          ISUPPZ( 2*i ). هذا ذو صلة في حالة
!>          تقسيم المصفوفة. لا يتم الوصول إلى ISUPPZ إلا عند كون JOBZ 'V' و N > 0.
!>

WORK

!>          WORK is REAL array, dimension (LWORK)
!>          عند الخروج، إذا كان INFO = 0، فإن WORK(1) يرجع
!>          LWORK الأمثل (والحد الأدنى).
!>

LWORK

!>          LWORK is INTEGER
!>          بعد المصفوفة WORK. LWORK >= max(1,18*N)
!>          إذا كان JOBZ = 'V'، و LWORK >= max(1,12*N) إذا كان JOBZ = 'N'.
!>          إذا كان LWORK = -1، فسيتم افتراض استعلام مساحة العمل؛ الروتين
!>          يحسب فقط الحجم الأمثل لمصفوفة WORK، ويرجع
!>          هذه القيمة كإدخال أول لمصفوفة WORK، ولا يتم إصدار أي رسالة خطأ
!>          متعلقة بـ LWORK بواسطة XERBLA.
!>

IWORK

!>          IWORK is INTEGER array, dimension (LIWORK)
!>          عند الخروج، إذا كان INFO = 0، فإن IWORK(1) يرجع
!>          LIWORK الأمثل.
!>

LIWORK

!>          LIWORK is INTEGER
!>          بعد المصفوفة IWORK. LIWORK >= max(1,10*N)
!>          إذا كانت المتجهات الذاتية مطلوبة، و LIWORK >= max(1,8*N)
!>          إذا كان سيتم حساب القيم الذاتية فقط.
!>          إذا كان LIWORK = -1، فسيتم افتراض استعلام مساحة العمل؛ الـ
!>          روتين يحسب فقط الحجم الأمثل لمصفوفة IWORK،
!>          ويرجع هذه القيمة كإدخال أول لمصفوفة IWORK، ولا
!>          يتم إصدار أي رسالة خطأ متعلقة بـ LIWORK بواسطة XERBLA.
!>

INFO

!>          INFO is INTEGER
!>          عند الخروج، INFO
!>          = 0:  خروج ناجح
!>          < 0:  إذا كان INFO = -i، فإن المعامل i كان له قيمة غير قانونية
!>          > 0:  إذا كان INFO = 1X، خطأ داخلي في SLARRE،
!>                إذا كان INFO = 2X، خطأ داخلي في CLARRV.
!>                هنا، الرقم X = ABS( IINFO ) < 10، حيث IINFO هو
!>                رمز الخطأ غير الصفري الذي تم إرجاعه بواسطة SLARRE أو
!>                CLARRV، على التوالي.
!>
