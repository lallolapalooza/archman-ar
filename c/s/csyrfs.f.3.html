subroutine csyrfs (character uplo, integer n, integer nrhs, complex, dimension( lda, * ) a, integer lda, complex, dimension( ldaf, * ) af, integer ldaf, integer, dimension( * ) ipiv, complex, dimension( ldb, * ) b, integer ldb, complex, dimension( ldx, * ) x, integer ldx, real, dimension( * ) ferr, real, dimension( * ) berr, complex, dimension( * ) work, real, dimension( * ) rwork, integer info)

CSYRFS

الغرض:

!>
!> CSYRFS يحسن الحل المحسوب لنظام المعادلات الخطية
!> عندما تكون مصفوفة المعاملات متناظرة غير محددة، و
!> يوفر حدود الخطأ وتقديرات الخطأ الخلفي للحل.
!>

المعلمات

UPLO 
!>          UPLO is CHARACTER*1
!>          = 'U':  يتم تخزين المثلث العلوي لـ A؛
!>          = 'L':  يتم تخزين المثلث السفلي لـ A.
!>

N

!>          N is INTEGER
!>          رتبة المصفوفة A. N >= 0.
!>

NRHS

!>          NRHS is INTEGER
!>          عدد الأيادي اليمنى، أي عدد الأعمدة
!>          للمصفوفتين B و X. NRHS >= 0.
!>

A

!>          A is COMPLEX array, dimension (LDA,N)
!>          المصفوفة المتناظرة A. إذا كان UPLO = 'U'، فإن الجزء العلوي
!>          المثلثي بحجم N-by-N الرائد في A يحتوي على الجزء العلوي
!>          المثلثي للمصفوفة A، ولا تتم الإشارة إلى الجزء السفلي
!>          المثلثي الصارم لـ A. إذا كان UPLO = 'L'، فإن الجزء السفلي
!>          المثلثي بحجم N-by-N الرائد في A يحتوي على الجزء السفلي
!>          المثلثي للمصفوفة A، ولا تتم الإشارة إلى الجزء العلوي
!>          المثلثي الصارم لـ A.
!>

LDA

!>          LDA is INTEGER
!>          البعد الرائد للمصفوفة A. LDA >= max(1,N).
!>

AF

!>          AF is COMPLEX array, dimension (LDAF,N)
!>          الشكل المحلل للمصفوفة A. تحتوي AF على المصفوفة القطرية الكتلية D
!>          والمضاعفات المستخدمة للحصول على U أو L من تحليل
!>          A = U*D*U**T أو A = L*D*L**T كما تم حسابه بواسطة CSYTRF.
!>

LDAF

!>          LDAF is INTEGER
!>          البعد الرائد للمصفوفة AF. LDAF >= max(1,N).
!>

IPIV

!>          IPIV is INTEGER array, dimension (N)
!>          تفاصيل التبادلات وهيكل الكتل لـ D
!>          كما تم تحديده بواسطة CSYTRF.
!>

B

!>          B is COMPLEX array, dimension (LDB,NRHS)
!>          مصفوفة اليد اليمنى B.
!>

LDB

!>          LDB is INTEGER
!>          البعد الرائد للمصفوفة B. LDB >= max(1,N).
!>

X

!>          X is COMPLEX array, dimension (LDX,NRHS)
!>          عند الدخول، مصفوفة الحل X، كما تم حسابها بواسطة CSYTRS.
!>          عند الخروج، مصفوفة الحل المحسنة X.
!>

LDX

!>          LDX is INTEGER
!>          البعد الرائد للمصفوفة X. LDX >= max(1,N).
!>

FERR

!>          FERR is REAL array, dimension (NRHS)
!>          تقدير حد الخطأ الأمامي لكل متجه حل
!>          X(j) (العمود j من مصفوفة الحل X).
!>          إذا كان XTRUE هو الحل الحقيقي المقابل لـ X(j)، فإن FERR(j)
!>          هو حد علوي مقدر لقيمة أكبر
!>          عنصر في (X(j) - XTRUE) مقسومًا على قيمة أكبر
!>          عنصر في X(j). التقدير موثوق به بقدر
!>          موثوقية تقدير RCOND، وهو دائمًا تقريبًا
!>          زيادة تقديرية للخطأ الحقيقي.
!>

BERR

!>          BERR is REAL array, dimension (NRHS)
!>          الخطأ الخلفي النسبي المكوني لكل متجه حل
!>          X(j) (أي، أصغر تغيير نسبي في
!>          أي عنصر في A أو B يجعل X(j) حلاً دقيقًا).
!>

WORK

!>          WORK is COMPLEX array, dimension (2*N)
!>

RWORK

!>          RWORK is REAL array, dimension (N)
!>

INFO

!>          INFO is INTEGER
!>          = 0:  الخروج الناجح
!>          < 0:  إذا كان INFO = -i، فإن المعلمة i كانت لها قيمة غير قانونية
!>
