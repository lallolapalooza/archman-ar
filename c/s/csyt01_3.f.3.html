subroutine csyt01_3 (character uplo, integer n, complex, dimension( lda, * ) a, integer lda, complex, dimension( ldafac, * ) afac, integer ldafac, complex, dimension( * ) e, integer, dimension( * ) ipiv, complex, dimension( ldc, * ) c, integer ldc, real, dimension( * ) rwork, real resid)

CSYT01_3

الغرض:

!>
!> CSYT01_3 يعيد بناء مصفوفة متناظرة غير محددة A من عواملها L*D*L' أو U*D*U'
!>  المحسوبة بواسطة CSYTRF_RK (أو CSYTRF_BK) ويحسب الباقي
!>    norm( C - A ) / ( N * norm(A) * EPS ),
!> حيث C هي المصفوفة المعاد بناؤها و EPS هو الخطأ المطبعي الآلي.
!>

المعلمات

UPLO 
!>          UPLO هو CHARACTER*1
!>          يحدد ما إذا كان الجزء العلوي أو السفلي المثلثي للمصفوفة
!>          المتناظرة A مخزنًا:
!>          = 'U':  مثلثي علوي
!>          = 'L':  مثلثي سفلي
!>

N

!>          N هو INTEGER
!>          عدد الصفوف والأعمدة للمصفوفة A.  N >= 0.
!>

A

!>          A هي مصفوفة COMPLEX، الأبعاد (LDA,N)
!>          المصفوفة المتناظرة الأصلية A.
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرائد للمصفوفة A.  LDA >= max(1,N)
!>

AFAC

!>          AFAC هي مصفوفة COMPLEX، الأبعاد (LDAFAC,N)
!>          قطر المصفوفة القطرية المتصلة D وعوامل U أو L
!>          كما تم حسابها بواسطة CSYTRF_RK و CSYTRF_BK:
!>            أ) فقط العناصر القطرية للمصفوفة القطرية المتصلة المتناظرة
!>               D على قطر A، أي D(k,k) = A(k,k);
!>               (يجب توفير العناصر القطرية الفوقية (أو تحت القطرية) لـ D
!>                في المصفوفة E عند الإدخال)، و
!>            ب) إذا كان UPLO = 'U': العامل U في الجزء القطري الفوقي لـ A.
!>               إذا كان UPLO = 'L': العامل L في الجزء القطري السفلي لـ A.
!>

LDAFAC

!>          LDAFAC هو INTEGER
!>          البعد الرائد للمصفوفة AFAC.
!>          LDAFAC >= max(1,N).
!>

E

!>          E هي مصفوفة COMPLEX، الأبعاد (N)
!>          عند الإدخال، تحتوي على العناصر القطرية الفوقية (أو تحت القطرية)
!>          للمصفوفة القطرية المتصلة المتناظرة D
!>          مع كتل قطرية 1x1 أو 2x2، حيث
!>          إذا كان UPLO = 'U': E(i) = D(i-1,i),i=2:N, E(1) غير مستخدمة؛
!>          إذا كان UPLO = 'L': E(i) = D(i+1,i),i=1:N-1, E(N) غير مستخدمة.
!>

IPIV

!>          IPIV هي مصفوفة INTEGER، الأبعاد (N)
!>          مؤشرات المحور من CSYTRF_RK (أو CSYTRF_BK).
!>

C

!>          C هي مصفوفة COMPLEX، الأبعاد (LDC,N)
!>

LDC

!>          LDC هو INTEGER
!>          البعد الرائد للمصفوفة C.  LDC >= max(1,N).
!>

RWORK

!>          RWORK هي مصفوفة REAL، الأبعاد (N)
!>

RESID

!>          RESID هو REAL
!>          إذا كان UPLO = 'L'، norm(L*D*L' - A) / ( N * norm(A) * EPS )
!>          إذا كان UPLO = 'U'، norm(U*D*U' - A) / ( N * norm(A) * EPS )
!>
