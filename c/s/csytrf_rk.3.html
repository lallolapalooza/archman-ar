subroutine csytrf_rk (character uplo, integer n, complex, dimension( lda, * ) a, integer lda, complex, dimension( * ) e, integer, dimension( * ) ipiv, complex, dimension( * ) work, integer lwork, integer info)

CSYTRF_RK يحسب تحليل مصفوفة متناظرة غير محددة مركبة باستخدام طريقة محورية قطرية Bunch-Kaufman (rook) المحدودة (خوارزمية BLAS3 المكتلة).

الغرض:

!> CSYTRF_RK يحسب تحليل مصفوفة متناظرة مركبة A
!> باستخدام طريقة المحورية القطرية Bunch-Kaufman (rook) المحدودة:
!>
!>    A = P*U*D*(U**T)*(P**T) أو A = P*L*D*(L**T)*(P**T)،
!>
!> حيث U (أو L) هي مصفوفة علوية (أو سفلية) وحدوية،
!> U**T (أو L**T) هي منقولة U (أو L)، P هي مصفوفة تبديل،
!> P**T هي منقولة P، و D هي مصفوفة متناظرة وكتلية قطرية ذات
!> كتل قطرية بحجم 1x1 و 2x2.
!>
!> هذه هي النسخة المكتلة من الخوارزمية، تستدعي BLAS المستوى 3.
!> لمزيد من المعلومات انظر قسم التفاصيل الإضافية.
!>

المعلمات

UPLO 
!>          UPLO هو CHARACTER*1
!>          يحدد ما إذا كان الجزء العلوي أو السفلي المثلثي للمصفوفة
!>          المتناظرة A مخزنًا:
!>          = 'U':  مثلثي علوي
!>          = 'L':  مثلثي سفلي
!>

N

!>          N هو INTEGER
!>          رتبة المصفوفة A. N >= 0.
!>

A

!>          A هو مصفوفة COMPLEX، ذات البعد (LDA,N)
!>          عند الدخول، المصفوفة المتناظرة A.
!>            إذا كان UPLO = 'U': الجزء العلوي المثلثي بحجم N*N
!>            من A يحتوي على الجزء العلوي المثلثي للمصفوفة A،
!>            والجزء السفلي المثلثي الصارم لـ A لا تتم الإشارة إليه.
!>
!>            إذا كان UPLO = 'L': الجزء السفلي المثلثي بحجم N*N
!>            من A يحتوي على الجزء السفلي المثلثي للمصفوفة A،
!>            والجزء العلوي المثلثي الصارم لـ A لا تتم الإشارة إليه.
!>
!>          عند الخروج، تحتوي على:
!>            أ) عناصر قطرية فقط للمصفوفة الكتلية القطرية المتناظرة
!>               D على القطر الرئيسي للمصفوفة A، أي D(k,k) = A(k,k)؛
!>               (عناصر فوق القطر الرئيسي (أو تحته) لـ D
!>                مخزنة عند الخروج في المصفوفة E)، و
!>            ب) إذا كان UPLO = 'U': العامل U في الجزء فوق القطري لـ A.
!>               إذا كان UPLO = 'L': العامل L في الجزء تحت القطري لـ A.
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرئيسي للمصفوفة A. LDA >= max(1,N).
!>

E

!>          E هي مصفوفة COMPLEX، ذات البعد (N)
!>          عند الخروج، تحتوي على العناصر فوق القطر الرئيسي (أو تحته)
!>          للمصفوفة الكتلية القطرية المتناظرة D
!>          ذات الكتل القطرية بحجم 1x1 أو 2x2، حيث
!>          إذا كان UPLO = 'U': E(i) = D(i-1,i), i=2:N, E(1) مضبوط على 0؛
!>          إذا كان UPLO = 'L': E(i) = D(i+1,i), i=1:N-1, E(N) مضبوط على 0.
!>
!>          ملاحظة: لكتلة قطرية بحجم 1x1 D(k)، حيث
!>          1 <= k <= N، يتم ضبط العنصر E(k) على 0 في كل من
!>          حالات UPLO = 'U' أو UPLO = 'L'.
!>

IPIV

!>          IPIV هي مصفوفة INTEGER، ذات البعد (N)
!>          تصف IPIV مصفوفة التبديل P في تحليل
!>          المصفوفة A كالتالي. القيمة المطلقة لـ IPIV(k)
!>          تمثل فهرس الصف والعمود اللذين تم تبديلهما
!>          مع الصف والعمود k-th. قيمة UPLO
!>          تصف ترتيب تطبيق التبادلات.
!>          أيضًا، إشارة IPIV تمثل البنية الكتلية
!>          للمصفوفة الكتلية القطرية المتناظرة D ذات الكتل
!>          القطرية 1x1 أو 2x2 والتي تتوافق مع تبديل واحد أو اثنين
!>          في كل خطوة تحليل. لمزيد من المعلومات انظر قسم
!>          التفاصيل الإضافية.
!>
!>          إذا كان UPLO = 'U'،
!>          (في ترتيب التحليل، k يتناقص من N إلى 1):
!>            أ) إدخال موجب واحد IPIV(k) > 0 يعني:
!>               D(k,k) هي كتلة قطرية بحجم 1x1.
!>               إذا كان IPIV(k) != k، تم تبديل الصفوف والأعمدة k و IPIV(k)
!>               في المصفوفة A(1:N,1:N)؛
!>               إذا كان IPIV(k) = k، لم يحدث تبديل.
!>
!>            ب) زوج من الإدخالات المتتالية السالبة
!>               IPIV(k) < 0 و IPIV(k-1) < 0 يعني:
!>               D(k-1:k,k-1:k) هي كتلة قطرية بحجم 2x2.
!>               (ملاحظة: الإدخالات السالبة في IPIV تظهر فقط في أزواج).
!>               1) إذا كان -IPIV(k) != k، تم تبديل الصفوف والأعمدة
!>                  k و -IPIV(k)
!>                  في المصفوفة A(1:N,1:N).
!>                  إذا كان -IPIV(k) = k، لم يحدث تبديل.
!>               2) إذا كان -IPIV(k-1) != k-1، تم تبديل الصفوف والأعمدة
!>                  k-1 و -IPIV(k-1)
!>                  في المصفوفة A(1:N,1:N).
!>                  إذا كان -IPIV(k-1) = k-1، لم يحدث تبديل.
!>
!>            ج) في كلتا الحالتين أ) و ب)، دائمًا ABS( IPIV(k) ) <= k.
!>
!>            د) ملاحظة: أي إدخال IPIV(k) غير صفري دائمًا عند الخروج.
!>
!>          إذا كان UPLO = 'L'،
!>          (في ترتيب التحليل، k يزداد من 1 إلى N):
!>            أ) إدخال موجب واحد IPIV(k) > 0 يعني:
!>               D(k,k) هي كتلة قطرية بحجم 1x1.
!>               إذا كان IPIV(k) != k، تم تبديل الصفوف والأعمدة k و IPIV(k)
!>               في المصفوفة A(1:N,1:N).
!>               إذا كان IPIV(k) = k، لم يحدث تبديل.
!>
!>            ب) زوج من الإدخالات المتتالية السالبة
!>               IPIV(k) < 0 و IPIV(k+1) < 0 يعني:
!>               D(k:k+1,k:k+1) هي كتلة قطرية بحجم 2x2.
!>               (ملاحظة: الإدخالات السالبة في IPIV تظهر فقط في أزواج).
!>               1) إذا كان -IPIV(k) != k، تم تبديل الصفوف والأعمدة
!>                  k و -IPIV(k)
!>                  في المصفوفة A(1:N,1:N).
!>                  إذا كان -IPIV(k) = k، لم يحدث تبديل.
!>               2) إذا كان -IPIV(k+1) != k+1، تم تبديل الصفوف والأعمدة
!>                  k+1 و -IPIV(k+1)
!>                  في المصفوفة A(1:N,1:N).
!>                  إذا كان -IPIV(k+1) = k+1، لم يحدث تبديل.
!>
!>            ج) في كلتا الحالتين أ) و ب)، دائمًا ABS( IPIV(k) ) >= k.
!>
!>            د) ملاحظة: أي إدخال IPIV(k) غير صفري دائمًا عند الخروج.
!>

WORK

!>          WORK هي مصفوفة COMPLEX، ذات البعد ( MAX(1,LWORK) ).
!>          عند الخروج، إذا كان INFO = 0، فإن WORK(1) ترجع LWORK المثلى.
!>

LWORK

!>          LWORK هو INTEGER
!>          طول WORK. LWORK >=1. للأداء الأمثل
!>          LWORK >= N*NB، حيث NB هو حجم الكتلة الذي ترجعه
!>          الدالة ILAENV.
!>
!>          إذا كان LWORK = -1، فإنه يُفترض استعلام مساحة العمل؛
!>          الروتين الفرعي يحسب فقط الحجم الأمثل لمصفوفة WORK،
!>          ويرجع هذا القيمة كالإدخال الأول لمصفوفة WORK،
!>          ولا يتم إصدار أي رسالة خطأ تتعلق بـ LWORK بواسطة XERBLA.
!>

INFO

!>          INFO هو INTEGER
!>          = 0: انتهاء ناجح
!>
!>          < 0: إذا كان INFO = -k، فإن المعلمة k-th كانت لها قيمة غير قانونية
!>
!>          > 0: إذا كان INFO = k، فإن المصفوفة A منفردة، وذلك بسبب:
!>                 إذا كان UPLO = 'U': العمود k في الجزء

!>                 إذا كان UPLO = 'L': العمود k في الجزء
!>                 المثلثي السفلي من A يحتوي على كل الأصفار.
!>
!>               لذلك فإن D(k,k) يساوي صفرًا تمامًا، والعناصر فوق القطر
