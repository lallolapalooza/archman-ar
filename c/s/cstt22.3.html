subroutine cstt22 (integer n, integer m, integer kband, real, dimension( * ) ad, real, dimension( * ) ae, real, dimension( * ) sd, real, dimension( * ) se, complex, dimension( ldu, * ) u, integer ldu, complex, dimension( ldwork, * ) work, integer ldwork, real, dimension( * ) rwork, real, dimension( 2 ) result)

CSTT22

الغرض:

!>
!> CSTT22  يتحقق من مجموعة من M قيم ذاتية ومتجهات ذاتية،
!>
!>     A U = U S
!>
!> حيث A مصفوفة هيرميتية ثلاثية الأقطار، وأعمدة U وحدوية،
!> و S قطرية (إذا كان KBAND=0) أو هيرميتية ثلاثية الأقطار (إذا كان KBAND=1).
!> يتم إجراء اختبارين:
!>
!>    RESULT(1) = | U* A U - S | / ( |A| m ulp )
!>
!>    RESULT(2) = | I - U*U | / ( m ulp )
!>

المعلمات:

N 
!>          N هو عدد صحيح (INTEGER)
!>          حجم المصفوفة. إذا كان صفرًا، فلا يقوم CSTT22 بأي شيء.
!>          يجب أن يكون على الأقل صفرًا.
!>

M

!>          M هو عدد صحيح (INTEGER)
!>          عدد الأزواج الذاتية المراد التحقق منها. إذا كان صفرًا، فلا يقوم CSTT22
!>          بأي شيء. يجب أن يكون على الأقل صفرًا.
!>

KBAND

!>          KBAND هو عدد صحيح (INTEGER)
!>          عرض المصفوفة S. يمكن أن يكون صفرًا أو واحدًا فقط.
!>          إذا كان صفرًا، فإن S قطرية، و SE غير مستخدمة. إذا
!>          كان واحدًا، فإن S هيرميتية ثلاثية الأقطار.
!>

AD

!>          AD هو مصفوفة حقيقية (REAL array)، البعد (N)
!>          القطر الرئيسي للمصفوفة الأصلية (غير المفككة) A. A
!>          يفترض أنها مصفوفة هيرميتية ثلاثية الأقطار.
!>

AE

!>          AE هو مصفوفة حقيقية (REAL array)، البعد (N)
!>          العناصر خارج القطر الرئيسي للمصفوفة الأصلية (غير المفككة) A.
!>          يفترض أن A مصفوفة هيرميتية ثلاثية الأقطار. AE(1) يتم تجاهلها،
!>          AE(2) هو العنصر (1,2) و (2,1)، وهكذا.
!>

SD

!>          SD هو مصفوفة حقيقية (REAL array)، البعد (N)
!>          القطر الرئيسي للمصفوفة (الهيرميتية ثلاثية الأقطار) S.
!>

SE

!>          SE هو مصفوفة حقيقية (REAL array)، البعد (N)
!>          العناصر خارج القطر الرئيسي للمصفوفة (الهيرميتية ثلاثية الأقطار) S.
!>          غير مستخدمة إذا كانت KBSND=0. إذا كان KBAND=1، فإن AE(1)
!>          يتم تجاهلها، SE(2) هو العنصر (1,2) و (2,1)، وهكذا.
!>

U

!>          U هي مصفوفة حقيقية (REAL array)، البعد (LDU, N)
!>          المصفوفة الوحدوية في التحلل.
!>

LDU

!>          LDU هو عدد صحيح (INTEGER)
!>          البعد الرائد لـ U. يجب أن يكون LDU على الأقل N.
!>

WORK

!>          WORK هي مصفوفة مركبة (COMPLEX array)، البعد (LDWORK, M+1)
!>

LDWORK

!>          LDWORK هو عدد صحيح (INTEGER)
!>          البعد الرائد لـ WORK. يجب أن يكون LDWORK على الأقل
!>          max(1,M).
!>

RWORK

!>          RWORK هي مصفوفة حقيقية (REAL array)، البعد (N)
!>

RESULT

!>          RESULT هي مصفوفة حقيقية (REAL array)، البعد (2)
!>          القيم المحسوبة بواسطة الاختبارين الموصوفين أعلاه.
!>          القيم محدودة حاليًا بـ 1/ulp، لتجنب تجاوز السعة.
!>
