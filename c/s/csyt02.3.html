subroutine csyt02 (character uplo, integer n, integer nrhs, complex, dimension( lda, * ) a, integer lda, complex, dimension( ldx, * ) x, integer ldx, complex, dimension( ldb, * ) b, integer ldb, real, dimension( * ) rwork, real resid)

CSYT02

الغرض:

!>
!> CSYT02 يحسب الباقي لحل نظام خطي متماثل مركب A*x = b:
!>
!>    RESID = norm(B - A*X) / ( norm(A) * norm(X) * EPS ),
!>
!> حيث EPS هو تقريب الآلة.
!>

المعاملات

UPLO 
!>          UPLO is CHARACTER*1
!>          يحدد ما إذا كان الجزء العلوي أو السفلي المثلثي من المصفوفة المتماثلة A مخزناً:
!>          = 'U':  علوي مثلثي
!>          = 'L':  سفلي مثلثي
!>

N

!>          N is INTEGER
!>          عدد الصفوف والأعمدة للمصفوفة A. N >= 0.
!>

NRHS

!>          NRHS is INTEGER
!>          عدد الأعمدة لـ B، مصفوفة الأطراف اليمنى.
!>          NRHS >= 0.
!>

A

!>          A is COMPLEX array, dimension (LDA,N)
!>          المصفوفة المتماثلة المركبة الأصلية A.
!>

LDA

!>          LDA is INTEGER
!>          البعد الرائد للمصفوفة A. LDA >= max(1,N)
!>

X

!>          X is COMPLEX array, dimension (LDX,NRHS)
!>          متجهات الحل المحسوبة للنظام الخطي.
!>

LDX

!>          LDX is INTEGER
!>          البعد الرائد للمصفوفة X. LDX >= max(1,N).
!>

B

!>          B is COMPLEX array, dimension (LDB,NRHS)
!>          عند الإدخال، متجهات الطرف الأيمن للنظام الخطي.
!>          عند الخروج، يتم استبدال B بـ B - A*X.
!>

LDB

!>          LDB is INTEGER
!>          البعد الرائد للمصفوفة B. LDB >= max(1,N).
!>

RWORK

!>          RWORK is REAL array, dimension (N)
!>

RESID

!>          RESID is REAL
!>          الحد الأقصى لعدد الأطراف اليمنى لـ
!>          norm(B - A*X) / ( norm(A) * norm(X) * EPS ).
!>
