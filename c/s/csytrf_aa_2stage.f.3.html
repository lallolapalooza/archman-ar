subroutine csytrf_aa_2stage (character uplo, integer n, complex, dimension( lda, * ) a, integer lda, complex, dimension( * ) tb, integer ltb, integer, dimension( * ) ipiv, integer, dimension( * ) ipiv2, complex, dimension( * ) work, integer lwork, integer info)

CSYTRF_AA_2STAGE

الغرض:

!>
!> CSYTRF_AA_2STAGE يحسب تحليل مصفوفة متماثلة معقدة A
!> باستخدام خوارزمية Aasen. شكل التحليل هو
!>
!>    A = U**T*T*U  أو  A = L*T*L**T
!>
!> حيث U (أو L) هي حاصل ضرب مصفوفات التبديل والمصفوفة المثلثية العلوية (السفلية)
!> ذات الوحدة، و T هي مصفوفة متماثلة معقدة ذات نطاق ترددي NB (يتم اختيار NB داخليًا وتخزينه في TB( 1 )، ويتم
!> تحليل T إلى LU مع تجنب جزئي).
!>
!> هذا هو الإصدار المجمع للخوارزمية، والذي يستدعي BLAS المستوى 3.
!>

المعلمات

UPLO 
!>          UPLO هو CHARACTER*1
!>          = 'U':  يتم تخزين المثلث العلوي من A؛
!>          = 'L':  يتم تخزين المثلث السفلي من A.
!>

N

!>          N هو INTEGER
!>          رتبة المصفوفة A.  N >= 0.
!>

A

!>          A هو مصفوفة COMPLEX، أبعادها (LDA,N)
!>          عند الدخول، المصفوفة المترافقة A. إذا كان UPLO = 'U'، فإن الجزء المثلثي العلوي
!>          من A بحجم N × N يحتوي على الجزء المثلثي العلوي من المصفوفة A، والجزء السفلي
!>          المثلثي بدقة من A لا يتم الرجوع إليه. إذا كان UPLO = 'L'، فإن الجزء السفلي
!>          المثلثي من A بحجم N × N يحتوي على الجزء المثلثي السفلي من المصفوفة A، والجزء العلوي
!>          المثلثي بدقة من A لا يتم الرجوع إليه.
!>
!>          عند الخروج، يتم تخزين L أسفل (أو فوق) كتل الأقطار الفرعية،
!>          عندما يكون UPLO = 'L' (أو 'U').
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرئيسي للمصفوفة A.  LDA >= max(1,N).
!>

TB

!>          TB هي مصفوفة COMPLEX، أبعادها (LTB)
!>          عند الخروج، تفاصيل تحليل LU للمصفوفة ذات النطاق.
!>

LTB

!>          LTB هو INTEGER
!>          حجم المصفوفة TB. LTB >= 4*N، يتم استخدامه داخليًا
!>          لاختيار NB بحيث يكون LTB >= (3*NB+1)*N.
!>
!>          إذا كان LTB = -1، فمن المفترض وجود استعلام عن مساحة العمل؛
!>          تقوم الإجراءات الفرعية فقط بحساب الحجم الأمثل لـ LTB،
!>          وتعيد هذا الحجم كإدخال أول في TB،
!>          ولا يتم إصدار رسالة خطأ تتعلق بـ LTB بواسطة XERBLA.
!>

IPIV

!>          IPIV هي مصفوفة INTEGER، أبعادها (N)
!>          عند الخروج، تحتوي على تفاصيل التبادلات، أي،
!>          تم تبادل الصف والعمود k من A مع
!>          الصف والعمود IPIV(k).
!>

IPIV2

!>          IPIV2 هي مصفوفة INTEGER، أبعادها (N)
!>          عند الخروج، تحتوي على تفاصيل التبادلات، أي،
!>          تم تبادل الصف والعمود k من T مع
!>          الصف والعمود IPIV(k).
!>

WORK

!>          WORK هي مساحة عمل COMPLEX بحجم LWORK
!>

LWORK

!>          LWORK هو INTEGER
!>          حجم WORK. LWORK >= N، يتم استخدامه داخليًا لاختيار NB
!>          بحيث يكون LWORK >= N*NB.
!>
!>          إذا كان LWORK = -1، فمن المفترض وجود استعلام عن مساحة العمل؛
!>          تقوم الإجراءات الفرعية فقط بحساب الحجم الأمثل لمصفوفة WORK،
!>          وتعيد هذا الحجم كإدخال أول في مصفوفة WORK،
!>          ولا يتم إصدار رسالة خطأ تتعلق بـ LWORK بواسطة XERBLA.
!>

INFO

!>          INFO هو INTEGER
!>          = 0:  انتهاء ناجح
!>          < 0:  إذا كان INFO = -i، فإن الوسيط i كان لديه قيمة غير قانونية.
!>          > 0:  إذا كان INFO = i، فشل تحليل LU للنطاق في العمود i
!>
