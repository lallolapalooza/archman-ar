subroutine csyswapr (character uplo, integer n, complex, dimension( lda, n ) a, integer lda, integer i1, integer i2)

CSYSWAPR

الغرض:

!>
!> CSYSWAPR تطبق تبديلاً أولياً على صفوف وأعمدة مصفوفة متناظرة.
!>

المعلمات

UPLO 
!>          UPLO is CHARACTER*1
!>          يحدد ما إذا كانت تفاصيل التحليل مخزنة
!>          كمصفوفة مثلثية عليا أو سفلى.
!>          = 'U':  مثلثية عليا، الشكل هو A = U*D*U**T؛
!>          = 'L':  مثلثية سفلى، الشكل هو A = L*D*L**T.
!>

N

!>          N is INTEGER
!>          رتبة المصفوفة A.  N >= 0.
!>

A

!>          A is COMPLEX array, dimension (LDA,N)
!>          عند الدخول، المصفوفة A بحجم N*N. عند الخروج، المصفوفة المبَدَّلة
!>          حيث يتم تبديل الصفوف I1 و I2 والأعمدة I1 و I2.
!>          إذا كان UPLO = 'U'، يتم تطبيق التبادلات على الجزء المثلثي العلوي
!>          ولا تتم الإشارة إلى الجزء المثلثي السفلي تمامًا من A؛ إذا كان UPLO = 'L'، يتم تطبيق التبادلات على
!>          الجزء المثلثي السفلي ولا تتم الإشارة إلى الجزء من A فوق القطر.
!>

LDA

!>          LDA is INTEGER
!>          البعد الرائد للمصفوفة A.  LDA >= max(1,N).
!>

I1

!>          I1 is INTEGER
!>          فهرس الصف الأول المراد تبديله
!>

I2

!>          I2 is INTEGER
!>          فهرس الصف الثاني المراد تبديله
!>
