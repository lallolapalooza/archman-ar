subroutine cspr (character uplo, integer n, complex alpha, complex, dimension( * ) x, integer incx, complex, dimension( * ) ap)

CSPR تقوم بتحديث مصفوفة معقدة متماثلة مجمعة برتبة واحدة.

الغرض:

!>
!> CSPR    تقوم بتنفيذ عملية الرتبة 1 المتماثلة
!>
!>    A := alpha*x*x**H + A,
!>
!> حيث alpha هو عدد قياسي معقد، x هو متجه n عنصراً و A هي مصفوفة
!> متماثلة بحجم n × n، مقدمة في شكل مجمع.
!>

المعلمات

UPLO 
!>          UPLO هو CHARACTER*1
!>           عند الدخول، تحدد UPLO ما إذا كان الجزء العلوي أو السفلي
!>           من المثلث للمصفوفة A مقدم في مصفوفة AP كما يلي:
!>
!>              UPLO = 'U' أو 'u'   يتم تقديم الجزء العلوي من المثلث للمصفوفة A
!>                                  في AP.
!>
!>              UPLO = 'L' أو 'l'   يتم تقديم الجزء السفلي من المثلث للمصفوفة A
!>                                  في AP.
!>
!>           دون تغيير عند الخروج.
!>

N

!>          N هو INTEGER
!>           عند الدخول، يحدد N رتبة المصفوفة A.
!>           يجب أن يكون N على الأقل صفر.
!>           دون تغيير عند الخروج.
!>

ALPHA

!>          ALPHA هو COMPLEX
!>           عند الدخول، يحدد ALPHA العدد القياسي alpha.
!>           دون تغيير عند الخروج.
!>

X

!>          X هي مصفوفة COMPLEX، البعد على الأقل
!>           ( 1 + ( N - 1 )*abs( INCX ) ).
!>           قبل الدخول، يجب أن تحتوي المصفوفة X المزايدة على المتجه x
!>           ذو N عنصر.
!>           دون تغيير عند الخروج.
!>

INCX

!>          INCX هو INTEGER
!>           عند الدخول، يحدد INCX الزيادة لعناصر X. يجب ألا يكون INCX صفراً.
!>           دون تغيير عند الخروج.
!>

AP

!>          AP هي مصفوفة COMPLEX، البعد على الأقل
!>           ( ( N*( N + 1 ) )/2 ).
!>           قبل الدخول، مع  UPLO = 'U' أو 'u'، يجب أن تحتوي المصفوفة AP
!>           على الجزء العلوي من المثلث للمصفوفة المتماثلة
!>           مجمعة بالتتابع، عموداً بعمود، بحيث تحتوي AP( 1 )
!>           على a( 1, 1 )، وتحتوي AP( 2 ) و AP( 3 ) على a( 1, 2 )
!>           و a( 2, 2 ) على التوالي، وهكذا. عند الخروج، يتم الكتابة فوق المصفوفة
!>           AP بالجزء العلوي من المثلث للمصفوفة المحدثة.
!>           قبل الدخول، مع UPLO = 'L' أو 'l'، يجب أن تحتوي المصفوفة AP
!>           على الجزء السفلي من المثلث للمصفوفة المتماثلة
!>           مجمعة بالتتابع، عموداً بعمود، بحيث تحتوي AP( 1 )
!>           على a( 1, 1 )، وتحتوي AP( 2 ) و AP( 3 ) على a( 2, 1 )
!>           و a( 3, 1 ) على التوالي، وهكذا. عند الخروج، يتم الكتابة فوق المصفوفة
!>           AP بالجزء السفلي من المثلث للمصفوفة المحدثة.
!>           لاحظ أن الأجزاء التخيلية للعناصر القطرية لا تحتاج
!>           إلى تعيينها، فهي مفترضة أن تكون صفر، وعند الخروج يتم
!>           تعيينها إلى صفر.
!>
