subroutine csytrs (character uplo, integer n, integer nrhs, complex, dimension( lda, * ) a, integer lda, integer, dimension( * ) ipiv, complex, dimension( ldb, * ) b, integer ldb, integer info)

CSYTRS

الغرض:

!>
!> CSYTRS يحل نظام المعادلات الخطية A*X = B بمصفوفة متماثلة معقدة A باستخدام
!> تحليل A = U*D*U**T أو A = L*D*L**T المحسوب بواسطة CSYTRF.
!>

المعاملات

UPLO 
!>          UPLO هو CHARACTER*1
!>          يحدد ما إذا كانت تفاصيل التحليل مخزنة
!>          كمصفوفة مثلثية علوية أو سفلية.
!>          = 'U':  مثلثية علوية، الشكل هو A = U*D*U**T;
!>          = 'L':  مثلثية سفلية، الشكل هو A = L*D*L**T.
!>

N

!>          N هو INTEGER
!>          رتبة المصفوفة A.  N >= 0.
!>

NRHS

!>          NRHS هو INTEGER
!>          عدد الأطراف اليمنى، أي عدد أعمدة
!>          المصفوفة B.  NRHS >= 0.
!>

A

!>          A هي مصفوفة COMPLEX، ذات أبعاد (LDA,N)
!>          المصفوفة القطرية الجزئية D والمضاعفات المستخدمة
!>          للحصول على العامل U أو L كما تم حسابه بواسطة CSYTRF.
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرئيسي للمصفوفة A.  LDA >= max(1,N).
!>

IPIV

!>          IPIV هي مصفوفة INTEGER، ذات أبعاد (N)
!>          تفاصيل التبادلات والبنية الجزئية لـ D
!>          كما تم تحديدها بواسطة CSYTRF.
!>

B

!>          B هي مصفوفة COMPLEX، ذات أبعاد (LDB,NRHS)
!>          عند الدخول، مصفوفة الطرف الأيمن B.
!>          عند الخروج، مصفوفة الحل X.
!>

LDB

!>          LDB هو INTEGER
!>          البعد الرئيسي للمصفوفة B.  LDB >= max(1,N).
!>

INFO

!>          INFO هو INTEGER
!>          = 0:  خروج ناجح
!>          < 0:  إذا كان INFO = -i، فإن المعامل i كان له قيمة غير قانونية
!>
