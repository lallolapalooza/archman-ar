subroutine csytrf_rook (character uplo, integer n, complex, dimension( lda, * ) a, integer lda, integer, dimension( * ) ipiv, complex, dimension( * ) work, integer lwork, integer info)

CSYTRF_ROOK

الغرض:

!>
!> CSYTRF_ROOK تحسب تحليل مصفوفة متماثلة معقدة A
!> باستخدام طريقة تحويل النقاط المحورية القطرية المحدودة Bunch-Kaufman ().
!> شكل التحليل هو
!>
!>    A = U*D*U**T  أو  A = L*D*L**T
!>
!> حيث U (أو L) هي ناتج عن تبديل ومصفوفات مثلثة عليا (سفلى) أحادية،
!> و D متماثلة وقطرية كتلية ذات كتل قطرية 1×1 و 2×2.
!>
!> هذه هي النسخة المحظورة من الخوارزمية، وتستدعي BLAS المستوى 3.
!>

المعاملات

UPLO 
!>          UPLO هو CHARACTER*1
!>          = 'U':  يتم تخزين المثلث العلوي لـ A;
!>          = 'L':  يتم تخزين المثلث السفلي لـ A.
!>

N

!>          N هو INTEGER
!>          رتبة المصفوفة A.  N >= 0.
!>

A

!>          A هي مصفوفة COMPLEX، بعدها (LDA,N)
!>          عند الإدخال، المصفوفة المتماثلة A.  إذا كان UPLO = 'U'، فإن الجزء
!>          المثلثي العلوي N×N الرائد لـ A يحتوي على الجزء المثلثي العلوي
!>          للمصفوفة A، ولا يتم الإشارة إلى الجزء المثلثي السفلي الصارم لـ A.
!>          إذا كان UPLO = 'L'، فإن الجزء المثلثي السفلي الرائد N×N لـ A يحتوي
!>          على الجزء المثلثي السفلي للمصفوفة A، ولا يتم الإشارة إلى الجزء
!>          المثلثي العلوي الصارم لـ A.
!>
!>          عند الخروج، المصفوفة القطرية الكتلية D والمضاعفات المستخدمة
!>          للحصول على العامل U أو L (انظر أدناه لمزيد من التفاصيل).
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرائد للمصفوفة A.  LDA >= max(1,N).
!>

IPIV

!>          IPIV هي مصفوفة INTEGER، بعدها (N)
!>          تفاصيل التبادلات وهيكل الكتلة لـ D.
!>
!>          إذا كان UPLO = 'U':
!>               إذا كان IPIV(k) > 0، فقد تم تبديل الصفوف والأعمدة k و IPIV(k)
!>               وكانت D(k,k) كتلة قطرية 1×1.
!>
!>               إذا كان IPIV(k) < 0 و IPIV(k-1) < 0، فقد تم تبديل الصفوف
!>               والأعمدة k و -IPIV(k) وتم تبديل الصفوف
!>               والأعمدة k-1 و -IPIV(k-1)،
!>               وكانت D(k-1:k,k-1:k) كتلة قطرية 2×2.
!>
!>          إذا كان UPLO = 'L':
!>               إذا كان IPIV(k) > 0، فقد تم تبديل الصفوف والأعمدة k و IPIV(k)
!>               وكانت D(k,k) كتلة قطرية 1×1.
!>
!>               إذا كان IPIV(k) < 0 و IPIV(k+1) < 0، فقد تم تبديل الصفوف
!>               والأعمدة k و -IPIV(k) وتم تبديل الصفوف
!>               والأعمدة k+1 و -IPIV(k+1)،
!>               وكانت D(k:k+1,k:k+1) كتلة قطرية 2×2.
!>

WORK

!>          WORK هي مصفوفة COMPLEX، بعدها (MAX(1,LWORK)).
!>          عند الخروج، إذا كان INFO = 0، فإن WORK(1) يرجع LWORK المثلى.
!>

LWORK

!>          LWORK هو INTEGER
!>          طول WORK.  LWORK >=1.  لأفضل أداء
!>          LWORK >= N*NB، حيث NB هو حجم الكتلة الذي يتم إرجاعه بواسطة ILAENV.
!>
!>          إذا كان LWORK = -1، فسيتم افتراض استعلام مساحة العمل؛ تقوم الروتين
!>          فقط بحساب الحجم الأمثل لمصفوفة WORK، وتعيد
!>          هذه القيمة كالإدخال الأول لمصفوفة WORK، ولا يصدر XERBLA أي رسالة خطأ
!>          متعلقة بـ LWORK.
!>

INFO

!>          INFO هو INTEGER
!>          = 0:  انتهاء ناجح
!>          < 0:  إذا كان INFO = -i، فإن الوسيط i كان له قيمة غير قانونية
!>          > 0:  إذا كان INFO = i، فإن D(i,i) تساوي صفر تمامًا.  اكتمل التحليل،
!>                ولكن المصفوفة القطرية الكتلية D منفردة تمامًا، وسيحدث
!>                قسمة على صفر إذا تم استخدامها لحل نظام من المعادلات.
!>

تفاصيل إضافية:

!>
!>  إذا كان UPLO = 'U'، فإن A = U*D*U**T، حيث
!>     U = P(n)*U(n)* ... *P(k)U(k)* ...,
!>  أي، U هو ناتج المصطلحات P(k)*U(k)، حيث k تقل من n إلى
!>  1 بخطوات 1 أو 2، و D هي مصفوفة قطرية كتلية ذات كتل قطرية 1×1
!>  و 2×2 D(k). P(k) هي مصفوفة تبديل كما
!>  هو محدد بواسطة IPIV(k)، و U(k) هي مصفوفة مثلثة عليا أحادية، بحيث
!>  إذا كانت الكتلة القطرية D(k) من الرتبة s (s = 1 أو 2)، فإن
!>
!>             (   I    v    0   )   k-s
!>     U(k) =  (   0    I    0   )   s
!>             (   0    0    I   )   n-k
!>                k-s   s   n-k
!>
!>  إذا كان s = 1، فإن D(k) تستبدل A(k,k)، و v تستبدل A(1:k-1,k).
!>  إذا كان s = 2، فإن المثلث العلوي لـ D(k) يستبدل A(k-1,k-1)، A(k-1,k)،
!>  و A(k,k)، و v تستبدل A(1:k-2,k-1:k).
!>
!>  إذا كان UPLO = 'L'، فإن A = L*D*L**T، حيث
!>     L = P(1)*L(1)* ... *P(k)*L(k)* ...,
!>  أي، L هو ناتج المصطلحات P(k)*L(k)، حيث k تزداد من 1 إلى
!>  n بخطوات 1 أو 2، و D هي مصفوفة قطرية كتلية ذات كتل قطرية 1×1
!>  و 2×2 D(k). P(k) هي مصفوفة تبديل كما
!>  هو محدد بواسطة IPIV(k)، و L(k) هي مصفوفة مثلثة سفلى أحادية، بحيث
!>  إذا كانت الكتلة القطرية D(k) من الرتبة s (s = 1 أو 2)، فإن
!>
!>             (   I    0     0   )  k-1
!>     L(k) =  (   0    I     0   )  s
!>             (   0    v     I   )  n-k-s+1
!>                k-1   s  n-k-s+1
!>
!>  إذا كان s = 1، فإن D(k) تستبدل A(k,k)، و v تستبدل A(k+1:n,k).
!>  إذا كان s = 2، فإن المثلث السفلي لـ D(k) يستبدل A(k,k)، A(k+1,k)،
!>  و A(k+1,k+1)، و v تستبدل A(k+2:n,k:k+1).
!>
