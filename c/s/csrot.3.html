subroutine csrot (integer n, complex, dimension( * ) cx, integer incx, complex, dimension( * ) cy, integer incy, real c, real s)

CSROT

الغرض:

!>
!> CSROT يطبق دوران مستوى، حيث أن cos و sin (c و s) حقيقيان
!> والمتجهات cx و cy مركبة.
!> جاك دونجارا، linpack، 3/11/78.
!>

المعلمات

N 
!>          N هو عدد صحيح
!>           عند الدخول، تحدد N ترتيب المتجهات cx و cy.
!>           يجب أن يكون N على الأقل صفر.
!>

CX

!>          CX هو مصفوفة مركبة، بأبعاد على الأقل
!>           ( 1 + ( N - 1 )*abs( INCX ) ).
!>           قبل الدخول، يجب أن تحتوي المصفوفة CX ذات الفواصل المحددة على عناصر
!>           المتجه cx البالغ عددها n. عند الخروج، سيتم استبدال CX بالمتجه
!>           cx المحدث.
!>

INCX

!>          INCX هو عدد صحيح
!>           عند الدخول، تحدد INCX الفاصل بين عناصر
!>           CX. يجب ألا تكون INCX صفرًا.
!>

CY

!>          CY هي مصفوفة مركبة، بأبعاد على الأقل
!>           ( 1 + ( N - 1 )*abs( INCY ) ).
!>           قبل الدخول، يجب أن تحتوي المصفوفة CY ذات الفواصل المحددة على عناصر
!>           المتجه cy البالغ عددها n. عند الخروج، سيتم استبدال CY بالمتجه
!>           cy المحدث.
!>

INCY

!>          INCY هو عدد صحيح
!>           عند الدخول، تحدد INCY الفاصل بين عناصر
!>           CY. يجب ألا تكون INCY صفرًا.
!>

C

!>          C هو قيمة حقيقية
!>           عند الدخول، تحدد C جيب التمام، cos.
!>

S

!>          S هو قيمة حقيقية
!>           عند الدخول، تحدد S جيب الزاوية، sin.
!>
