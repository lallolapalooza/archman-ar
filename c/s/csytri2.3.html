subroutine csytri2 (character uplo, integer n, complex, dimension( lda, * ) a, integer lda, integer, dimension( * ) ipiv, complex, dimension( * ) work, integer lwork, integer info)

CSYTRI2

الغرض:

!>
!> CSYTRI2 يحسب معكوس المصفوفة المعقدة المتماثلة غير المحددة
!> A باستخدام التحليل A = U*D*U**T أو A = L*D*L**T الذي تم حسابه بواسطة
!> CSYTRF. CSYTRI2 يضبط البعد الرئيسي لمساحة العمل
!> قبل استدعاء CSYTRI2X الذي يحسب المعكوس بالفعل.
!>

المعاملات

UPLO 
!>          UPLO هو CHARACTER*1
!>          يحدد ما إذا كانت تفاصيل التحليل مخزنة
!>          كمصفوفة علوية أو سفلية مثلثة.
!>          = 'U':  مثلثية علوية، الشكل هو A = U*D*U**T؛
!>          = 'L':  مثلثية سفلية، الشكل هو A = L*D*L**T.
!>

N

!>          N هو INTEGER
!>          ترتيب المصفوفة A.  N >= 0.
!>

A

!>          A هي مصفوفة COMPLEX، البعد (LDA,N)
!>          عند الدخول، المصفوفة القطرية الفرعية D والمعاملات
!>          المستخدمة للحصول على العامل U أو L كما تم حسابه بواسطة CSYTRF.
!>
!>          عند الخروج، إذا كان INFO = 0، معكوس (المتماثل) للمصفوفة الأصلية
!>          . إذا كان UPLO = 'U'، يتم تشكيل الجزء المثلثي العلوي من المعكوس
!>          ولا يتم الرجوع إلى الجزء السفلي من A؛ إذا كان UPLO = 'L' يتم تشكيل
!>          الجزء المثلثي السفلي من المعكوس ولا يتم الرجوع إلى الجزء العلوي من A.
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرئيسي للمصفوفة A.  LDA >= max(1,N).
!>

IPIV

!>          IPIV هي مصفوفة INTEGER، البعد (N)
!>          تفاصيل التبادلات والهيكل القطري لـ D
!>          كما تم تحديده بواسطة CSYTRF.
!>

WORK

!>          WORK هي مصفوفة COMPLEX، البعد (N+NB+1)*(NB+3)
!>

LWORK

!>          LWORK هو INTEGER
!>          بعد مصفوفة WORK.
!>          WORK هو حجم >= (N+NB+1)*(NB+3)
!>          إذا كان LWORK = -1، فيفترض استعلام مساحة العمل؛ الروتين
!>           يحسب:
!>              - الحجم الأمثل لمصفوفة WORK، يعيد
!>          هذه القيمة كمدخل أول للمصفوفة WORK،
!>              - ولا تصدر رسالة خطأ متعلقة بـ LWORK بواسطة XERBLA.
!>

INFO

!>          INFO هو INTEGER
!>          = 0: خروج ناجح
!>          < 0: إذا كان INFO = -i، كان المعامل i غير قانوني
!>          > 0: إذا كان INFO = i، D(i,i) = 0؛ المصفوفة مفردة ولم يتمكن من حساب معكوسها.
!>
