subroutine csyrfs (character uplo, integer n, integer nrhs, complex, dimension( lda, * ) a, integer lda, complex, dimension( ldaf, * ) af, integer ldaf, integer, dimension( * ) ipiv, complex, dimension( ldb, * ) b, integer ldb, complex, dimension( ldx, * ) x, integer ldx, real, dimension( * ) ferr, real, dimension( * ) berr, complex, dimension( * ) work, real, dimension( * ) rwork, integer info)

CSYRFS

الغرض:

!>
!> CSYRFS يحسن الحل المحسوب لنظام المعادلات الخطية
!> عندما تكون مصفوفة المعاملات متماثلة وغير محددة، و
!> يوفر حدود خطأ وتقديرات للخطأ الخلفي للحل.
!>

المعلمات

UPLO 
!>          UPLO is CHARACTER*1
!>          = 'U':  المثلث العلوي من A مخزن؛
!>          = 'L':  المثلث السفلي من A مخزن.
!>

N

!>          N is INTEGER
!>          رتبة المصفوفة A.  N >= 0.
!>

NRHS

!>          NRHS is INTEGER
!>          عدد الأطراف اليمنى، أي عدد الأعمدة
!>          للمصفوفتين B و X.  NRHS >= 0.
!>

A

!>          A is COMPLEX array, dimension (LDA,N)
!>          المصفوفة المتماثلة A.  إذا كان UPLO = 'U'، الجزء العلوي
!>          المثلثي الرئيسي N-by-N من A يحتوي على الجزء العلوي
!>          المثلثي للمصفوفة A، والجزء السفلي المثلثي الصارم من A
!>          غير مستخدم.  إذا كان UPLO = 'L'، الجزء السفلي
!>          المثلثي الرئيسي N-by-N من A يحتوي على الجزء السفلي
!>          المثلثي للمصفوفة A، والجزء العلوي المثلثي الصارم من A
!>          غير مستخدم.
!>

LDA

!>          LDA is INTEGER
!>          البعد الرائد للمصفوفة A.  LDA >= max(1,N).
!>

AF

!>          AF is COMPLEX array, dimension (LDAF,N)
!>          الشكل المفكك للمصفوفة A.  AF تحتوي على مصفوفة القطر
!>          الرئيسي D والمضاعفات المستخدمة للحصول على
!>          العامل U أو L من التحليل A = U*D*U**T أو
!>          A = L*D*L**T كما تم حسابه بواسطة CSYTRF.
!>

LDAF

!>          LDAF is INTEGER
!>          البعد الرائد للمصفوفة AF.  LDAF >= max(1,N).
!>

IPIV

!>          IPIV is INTEGER array, dimension (N)
!>          تفاصيل التبادلات وهيكل الكتلة D
!>          كما تم تحديدها بواسطة CSYTRF.
!>

B

!>          B is COMPLEX array, dimension (LDB,NRHS)
!>          مصفوفة الطرف الأيمن B.
!>

LDB

!>          LDB is INTEGER
!>          البعد الرائد للمصفوفة B.  LDB >= max(1,N).
!>

X

!>          X is COMPLEX array, dimension (LDX,NRHS)
!>          عند الإدخال، مصفوفة الحل X، كما تم حسابها بواسطة CSYTRS.
!>          عند الخروج، مصفوفة الحل المحسنة X.
!>

LDX

!>          LDX is INTEGER
!>          البعد الرائد للمصفوفة X.  LDX >= max(1,N).
!>

FERR

!>          FERR is REAL array, dimension (NRHS)
!>          التقدير لحد الخطأ الأمامي لكل متجه حل
!>          X(j) (العمود j للمصفوفة الحل X).
!>          إذا كان XTRUE هو الحل الحقيقي المقابل لـ X(j)، فإن FERR(j)
!>          هو حد علوي مقدر لحجم أكبر
!>          عنصر في (X(j) - XTRUE) مقسومًا على حجم أكبر
!>          عنصر في X(j).  التقدير موثوق به مثل
!>          التقدير لـ RCOND، وهو دائمًا تقريبًا
!>          مبالغة في تقدير الخطأ الحقيقي.
!>

BERR

!>          BERR is REAL array, dimension (NRHS)
!>          الخطأ الخلفي النسبي المكوني لكل متجه حل
!>          X(j) (أي، أصغر تغيير نسبي في
!>          أي عنصر في A أو B يجعل X(j) حلاً دقيقًا).
!>

WORK

!>          WORK is COMPLEX array, dimension (2*N)
!>

RWORK

!>          RWORK is REAL array, dimension (N)
!>

INFO

!>          INFO is INTEGER
!>          = 0:  خروج ناجح
!>          < 0:  إذا كان INFO = -i، فإن المعلمة i كانت لها قيمة غير قانونية
!>
