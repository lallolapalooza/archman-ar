وثائق الوظيفة/الروتين الفرعي

subroutine cstein (integer n, real, dimension( * ) d, real, dimension( * ) e, integer m, real, dimension( * ) w, integer, dimension( * ) iblock, integer, dimension( * ) isplit, complex, dimension( ldz, * ) z, integer ldz, real, dimension( * ) work, integer, dimension( * ) iwork, integer, dimension( * ) ifail, integer info)

CSTEIN

الغرض:

!>
!> CSTEIN تحسب المتجهات الذاتية لمصفوفة متماثلة ثلاثية الأقطار حقيقية
!> T المقابلة للقيم الذاتية المحددة، باستخدام التكرار العكسي.
!>
!> الحد الأقصى لعدد التكرارات المسموح بها لكل متجه ذاتي يتم تحديده
!> بواسطة معلمة داخلية MAXITS (حالياً تم تعيينها إلى 5).
!>
!> على الرغم من أن المتجهات الذاتية حقيقية، إلا أنها مخزنة في مصفوفة
!> مركبة، والتي يمكن تمريرها إلى CUNMTR أو CUPMTR للتحويل العكسي
!> إلى المتجهات الذاتية لمصفوفة هرمية مركبة تم اختزالها إلى شكل ثلاثي
!> الأقطار.
!>
!>

المعلمات

N 
!>          N هو عدد صحيح (INTEGER)
!>          ترتيب المصفوفة. N >= 0.
!>

D

!>          D هي مصفوفة حقيقية (REAL array)، البعد (N)
!>          العناصر القطرية n للمصفوفة ثلاثية الأقطار T.
!>

E

!>          E هي مصفوفة حقيقية (REAL array)، البعد (N-1)
!>          العناصر تحت القطرية (n-1) للمصفوفة ثلاثية الأقطار
!>          T، مخزنة في العناصر من 1 إلى N-1.
!>

M

!>          M هو عدد صحيح (INTEGER)
!>          عدد المتجهات الذاتية المطلوب حسابها. 0 <= M <= N.
!>

W

!>          W هي مصفوفة حقيقية (REAL array)، البعد (N)
!>          العناصر M الأولى في W تحتوي على القيم الذاتية التي
!>          يتم حساب المتجهات الذاتية لها. يجب تجميع القيم الذاتية
!>          حسب الكتلة المفصولة وترتيبها من الأصغر إلى الأكبر
!>          داخل الكتلة. (مصفوفة الإخراج W من SSTEBZ مع ORDER = 'B' متوقعة هنا.)
!>

IBLOCK

!>          IBLOCK هي مصفوفة أعداد صحيحة (INTEGER array)، البعد (N)
!>          فهارس المصفوفات الفرعية المرتبطة بالقيم الذاتية المقابلة
!>          في W؛ IBLOCK(i)=1 إذا كانت القيمة الذاتية W(i) تنتمي إلى
!>          المصفوفة الفرعية الأولى من الأعلى، =2 إذا كانت W(i) تنتمي إلى
!>          المصفوفة الفرعية الثانية، إلخ. (مصفوفة الإخراج IBLOCK
!>          من SSTEBZ متوقعة هنا.)
!>

ISPLIT

!>          ISPLIT هي مصفوفة أعداد صحيحة (INTEGER array)، البعد (N)
!>          نقاط الفصل، التي تنقسم عندها T إلى مصفوفات فرعية.
!>          المصفوفة الفرعية الأولى تتكون من الصفوف/الأعمدة من 1 إلى
!>          ISPLIT( 1 )، والثانية من الصفوف/الأعمدة ISPLIT( 1 )+1
!>          حتى ISPLIT( 2 )، إلخ.
!>          (مصفوفة الإخراج ISPLIT من SSTEBZ متوقعة هنا.)
!>

Z

!>          Z هي مصفوفة مركبة (COMPLEX array)، البعد (LDZ, M)
!>          المتجهات الذاتية المحسوبة. المتجه الذاتي المرتبط
!>          بالقيمة الذاتية W(i) مخزن في العمود i من
!>          Z. أي متجه يفشل في التقارب يتم تعيينه إلى التكرار الحالي
!>          بعد MAXITS تكرار.
!>          الأجزاء التخيلية للمتجهات الذاتية يتم تعيينها إلى صفر.
!>

LDZ

!>          LDZ هو عدد صحيح (INTEGER)
!>          البعد الرئيسي للمصفوفة Z. LDZ >= max(1,N).
!>

WORK

!>          WORK هي مصفوفة حقيقية (REAL array)، البعد (5*N)
!>

IWORK

!>          IWORK هي مصفوفة أعداد صحيحة (INTEGER array)، البعد (N)
!>

IFAIL

!>          IFAIL هي مصفوفة أعداد صحيحة (INTEGER array)، البعد (M)
!>          عند الخروج الطبيعي، جميع عناصر IFAIL تكون صفرًا.
!>          إذا فشل متجه ذاتي واحد أو أكثر في التقارب بعد
!>          MAXITS تكرار، يتم تخزين فهارسها في
!>          المصفوفة IFAIL.
!>

INFO

!>          INFO هو عدد صحيح (INTEGER)
!>          = 0: خروج ناجح
!>          < 0: إذا كان INFO = -i، فإن المعلمة i كان لها قيمة غير قانونية
!>          > 0: إذا كان INFO = i، فإن i متجهات ذاتية فشلت في التقارب
!>               في MAXITS تكرار. يتم تخزين فهارسها في
!>               المصفوفة IFAIL.
!>

المعلمات الداخلية:

!>  MAXITS  عدد صحيح (INTEGER)، الافتراضي = 5
!>          الحد الأقصى لعدد التكرارات التي تم إجراؤها.
!>
!>  EXTRA   عدد صحيح (INTEGER)، الافتراضي = 2
!>          عدد التكرارات التي تم إجراؤها بعد تلبية معيار نمو المعيار،
!>          يجب أن يكون على الأقل 1.
!>
