subroutine cspsv (character uplo, integer n, integer nrhs, complex, dimension( * ) ap, integer, dimension( * ) ipiv, complex, dimension( ldb, * ) b, integer ldb, integer info)

CSPSV تحسب الحل لنظام المعادلات الخطية A * X = B للمصفوفات الأخرى

الغرض:

!>
!> CSPSV تحسب الحل لنظام خطي معقد للمعادلات
!>    A * X = B,
!> حيث A مصفوفة متناظرة N × N مخزنة في تنسيق مكدس و X و B مصفوفتان N × NRHS.
!>
!> يتم استخدام طريقة المحورية القطرية لتحليل A إلى
!>    A = U * D * U**T,  إذا كان UPLO = 'U', أو
!>    A = L * D * L**T,  إذا كان UPLO = 'L',
!> حيث U (أو L) هو ناتج ضرب مصفوفات تبديل ومصفوفات مثلثة عليا (سفلى) أحادية،
!> D متناظرة ومصفوفة قطرية كتلة مع كتل قطرية 1 × 1 و 2 × 2. ثم يتم استخدام
!> الشكل المحلل لـ A لحل نظام المعادلات A * X = B.
!>

المعلمات

UPLO 
!>          UPLO هو CHARACTER*1
!>          = 'U':  يتم تخزين المثلث العلوي لـ A;
!>          = 'L':  يتم تخزين المثلث السفلي لـ A.
!>

N

!>          N هو INTEGER
!>          عدد المعادلات الخطية، أي ترتيب المصفوفة A.  N >= 0.
!>

NRHS

!>          NRHS هو INTEGER
!>          عدد الأطراف اليمنى، أي عدد أعمدة المصفوفة B.  NRHS >= 0.
!>

AP

!>          AP هو مصفوفة COMPLEX، بعد (N*(N+1)/2)
!>          عند الدخول، المثلث العلوي أو السفلي للمصفوفة المتناظرة
!>          A، مكدسة حسب العمود في مصفوفة خطية.  يتم تخزين العمود j من A
!>          في المصفوفة AP كما يلي:
!>          إذا كان UPLO = 'U', AP(i + (j-1)*j/2) = A(i,j) لـ 1<=i<=j;
!>          إذا كان UPLO = 'L', AP(i + (j-1)*(2n-j)/2) = A(i,j) لـ j<=i<=n.
!>          انظر أدناه لمزيد من التفاصيل.
!>
!>          عند الخروج، المصفوفة القطرية الكتلة D والمضاعفات المستخدمة
!>          للحصول على العامل U أو L من التحليل
!>          A = U*D*U**T أو A = L*D*L**T كما تم حسابه بواسطة CSPTRF، مخزنة كـ
!>          مصفوفة مثلثة مكدسة بنفس تنسيق التخزين مثل A.
!>

IPIV

!>          IPIV هو مصفوفة INTEGER، بعد (N)
!>          تفاصيل التبادلات وهيكل الكتلة لـ D، كما
!>          تم تحديدها بواسطة CSPTRF.  إذا كان IPIV(k) > 0، فسيتم تبديل
!>          الصفوف والأعمدة k و IPIV(k)، و D(k,k) هي كتلة قطرية 1 × 1.
!>          إذا كان UPLO = 'U' وكان IPIV(k) = IPIV(k-1) < 0،
!>          فقد تم تبديل الصفوف والأعمدة k-1 و -IPIV(k) و
!>          D(k-1:k,k-1:k) هي كتلة قطرية 2 × 2.  إذا كان UPLO = 'L' وكان
!>          IPIV(k) = IPIV(k+1) < 0، فقد تم تبديل الصفوف والأعمدة k+1 و
!>          -IPIV(k) و D(k:k+1,k:k+1) هي كتلة قطرية 2 × 2.
!>

B

!>          B هي مصفوفة COMPLEX، بعد (LDB,NRHS)
!>          عند الدخول، مصفوفة الطرف الأيمن B ذات الأبعاد N × NRHS.
!>          عند الخروج، إذا كان INFO = 0، مصفوفة الحل X ذات الأبعاد N × NRHS.
!>

LDB

!>          LDB هو INTEGER
!>          البعد الرئيسي للمصفوفة B.  LDB >= max(1,N).
!>

INFO

!>          INFO هو INTEGER
!>          = 0:  خروج ناجح
!>          < 0:  إذا كان INFO = -i، فإن المعامل i كان له قيمة غير قانونية
!>          > 0:  إذا كان INFO = i، فإن D(i,i) يساوي صفرًا تمامًا.  تم إكمال التحليل،
!>                ولكن المصفوفة القطرية الكتلة D مفردة تمامًا، لذا لم يكن من الممكن
!>                حساب الحل.
!>
