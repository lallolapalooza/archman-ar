subroutine csytrs_rook (character uplo, integer n, integer nrhs, complex, dimension( lda, * ) a, integer lda, integer, dimension( * ) ipiv, complex, dimension( ldb, * ) b, integer ldb, integer info)

CSYTRS_ROOK

الغرض:

!>
!> CSYTRS_ROOK يحل نظام المعادلات الخطية A*X = B
!> مع مصفوفة متماثلة مركبة A باستخدام التحلل A = U*D*U**T أو
!> A = L*D*L**T الذي تم حسابه بواسطة CSYTRF_ROOK.
!>

المعلمات

UPLO 
!>          UPLO هو CHARACTER*1
!>          يحدد ما إذا كانت تفاصيل التحلل مخزنة
!>          كمصفوفة علوية أو سفلية مثلثة.
!>          = 'U':  مثلثية علوية، الشكل هو A = U*D*U**T؛
!>          = 'L':  مثلثية سفلية، الشكل هو A = L*D*L**T.
!>

N

!>          N هو INTEGER
!>          رتبة المصفوفة A.  N >= 0.
!>

NRHS

!>          NRHS هو INTEGER
!>          عدد الأطراف اليمنى، أي عدد الأعمدة
!>          للمصفوفة B.  NRHS >= 0.
!>

A

!>          A هي مصفوفة COMPLEX، الأبعاد (LDA,N)
!>          المصفوفة القطرية الكتلة D والمضاعفات المستخدمة
!>          للحصول على العامل U أو L كما تم حسابه بواسطة CSYTRF_ROOK.
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرائد للمصفوفة A.  LDA >= max(1,N).
!>

IPIV

!>          IPIV هي مصفوفة INTEGER، الأبعاد (N)
!>          تفاصيل التبادلات وهيكل الكتلة لـ D
!>          كما تم تحديده بواسطة CSYTRF_ROOK.
!>

B

!>          B هي مصفوفة COMPLEX، الأبعاد (LDB,NRHS)
!>          عند الدخول، مصفوفة الطرف الأيمن B.
!>          عند الخروج، مصفوفة الحل X.
!>

LDB

!>          LDB هو INTEGER
!>          البعد الرائد للمصفوفة B.  LDB >= max(1,N).
!>

INFO

!>          INFO هو INTEGER
!>          = 0:  خروج ناجح
!>          < 0:  إذا كان INFO = -i، فإن المعلمة i-th كانت لها قيمة غير صالحة
!>
