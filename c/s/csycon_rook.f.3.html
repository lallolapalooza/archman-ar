subroutine csycon_rook (character uplo, integer n, complex, dimension( lda, * ) a, integer lda, integer, dimension( * ) ipiv, real anorm, real rcond, complex, dimension( * ) work, integer info)

CSYCON_ROOK

الغرض:

!>
!> CSYCON_ROOK تقدّر مقلوب رقم الشرط (بالـ 1-norm) لمصفوفة متماثلة مركبة A باستخدام التحليل
!> A = U*D*U**T أو A = L*D*L**T الذي تم حسابه بواسطة CSYTRF_ROOK.
!>
!> يتم الحصول على تقدير لـ norm(inv(A))، ويتم حساب مقلوب رقم الشرط كـ RCOND = 1 / (ANORM * norm(inv(A))).
!>

المعاملات

UPLO 
!>          UPLO هو CHARACTER*1
!>          يحدد ما إذا كانت تفاصيل التحليل مخزنة
!>          كمصفوفة مثلثة عليا أو سفلى.
!>          = 'U':  مثلثة عليا، الشكل هو A = U*D*U**T;
!>          = 'L':  مثلثة سفلى، الشكل هو A = L*D*L**T.
!>

N

!>          N هو INTEGER
!>          رتبة المصفوفة A.  N >= 0.
!>

A

!>          A هي مصفوفة COMPLEX، البعد (LDA,N)
!>          المصفوفة القطرية الجزئية D والمضاعفات المستخدمة
!>          للحصول على العامل U أو L كما تم حسابه بواسطة CSYTRF_ROOK.
!>

LDA

!>          LDA هو INTEGER
!>          البعد الأمامي للمصفوفة A.  LDA >= max(1,N).
!>

IPIV

!>          IPIV هي مصفوفة INTEGER، البعد (N)
!>          تفاصيل التبادلات والبنية الجزئية لـ D
!>          كما تم تحديده بواسطة CSYTRF_ROOK.
!>

ANORM

!>          ANORM هو REAL
!>          الـ 1-norm للمصفوفة الأصلية A.
!>

RCOND

!>          RCOND هو REAL
!>          مقلوب رقم الشرط للمصفوفة A،
!>          محسوب كـ RCOND = 1/(ANORM * AINVNM)، حيث AINVNM هو تقدير
!>          لـ 1-norm لـ inv(A) محسوب في هذا الإجراء.
!>

WORK

!>          WORK هي مصفوفة COMPLEX، البعد (2*N)
!>

INFO

!>          INFO هو INTEGER
!>          = 0:  انتهاء ناجح
!>          < 0:  إذا كان INFO = -i، فإن المعامل i كان له قيمة غير قانونية
!>
