subroutine cspt03 (character uplo, integer n, complex, dimension( * ) a, complex, dimension( * ) ainv, complex, dimension( ldw, * ) work, integer ldw, real, dimension( * ) rwork, real rcond, real resid)

CSPT03

الغرض:

!>
!> CSPT03 تحسب الباقي لضرب مصفوفة متماثلة معقدة معبأة في معكوسها:
!>    norm( I - A*AINV ) / ( N * norm(A) * norm(AINV) * EPS ),
!> حيث EPS هو الرقم المعياري للآلة.
!>

المعاملات

UPLO 
!>          UPLO هو CHARACTER*1
!>          تحدد ما إذا كان الجزء العلوي أو السفلي من المصفوفة المتماثلة
!>          المعقدة A مخزنًا:
!>          = 'U':  مثلثي علوي
!>          = 'L':  مثلثي سفلي
!>

N

!>          N هو INTEGER
!>          عدد الصفوف والأعمدة في المصفوفة A.  N >= 0.
!>

A

!>          A هو مصفوفة COMPLEX، ذات أبعاد (N*(N+1)/2)
!>          المصفوفة المتماثلة المعقدة الأصلية A، مخزنة كمصفوفة
!>          مثلثية معبأة.
!>

AINV

!>          AINV هو مصفوفة COMPLEX، ذات أبعاد (N*(N+1)/2)
!>          المعكوس (المتماثل) للمصفوفة A، مخزن كمصفوفة
!>          مثلثية معبأة.
!>

WORK

!>          WORK هو مصفوفة COMPLEX، ذات أبعاد (LDW,N)
!>

LDW

!>          LDW هو INTEGER
!>          البعد الرئيسي للمصفوفة WORK.  LDW >= max(1,N).
!>

RWORK

!>          RWORK هو مصفوفة REAL، ذات أبعاد (N)
!>

RCOND

!>          RCOND هو REAL
!>          مقلوب رقم الشرط للمصفوفة A، محسوب على أنه
!>          ( 1/norm(A) ) / norm(AINV).
!>

RESID

!>          RESID هو REAL
!>          norm(I - A*AINV) / ( N * norm(A) * norm(AINV) * EPS )
!>
