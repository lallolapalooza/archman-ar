subroutine cspmv (character uplo, integer n, complex alpha, complex, dimension( * ) ap, complex, dimension( * ) x, integer incx, complex beta, complex, dimension( * ) y, integer incy)

CSPMV تحسب حاصل ضرب متجه مصفوفة باستخدام مصفوفة متناظرة مجمعة معقدة

الغرض:

!>
!> CSPMV  تجري عملية ضرب المتجه بالمصفوفة
!>
!>    y := alpha*A*x + beta*y,
!>
!> حيث alpha و beta هي مقادير قياسية، x و y هي متجهات ذات n عنصر،
!> و A هي مصفوفة متناظرة بحجم n × n، مقدمة في شكل مجمّع.
!>

المعلمات

UPLO 
!>          UPLO is CHARACTER*1
!>           عند الإدخال، يحدد UPLO ما إذا كان الجزء المثلثي العلوي أو السفلي
!>           من المصفوفة A مقدمًا في مصفوفة AP على النحو التالي:
!>
!>              UPLO = 'U' or 'u'   يتم تقديم الجزء المثلثي العلوي من A
!>                                  في AP.
!>
!>              UPLO = 'L' or 'l'   يتم تقديم الجزء المثلثي السفلي من A
!>                                  في AP.
!>
!>           غير متغير عند الخروج.
!>

N

!>          N is INTEGER
!>           عند الإدخال، يحدد N رتبة المصفوفة A.
!>           يجب أن يكون N على الأقل صفرًا.
!>           غير متغير عند الخروج.
!>

ALPHA

!>          ALPHA is COMPLEX
!>           عند الإدخال، يحدد ALPHA المقدار القياسي alpha.
!>           غير متغير عند الخروج.
!>

AP

!>          AP is COMPLEX array, dimension at least
!>           ( ( N*( N + 1 ) )/2 ).
!>           قبل الإدخال، مع UPLO = 'U' أو 'u'، يجب أن تحتوي مصفوفة AP
!>           على الجزء المثلثي العلوي من المصفوفة المتناظرة
!>           مجمّعًا بالتتابع، عمودًا بعمود، بحيث تحتوي AP( 1 )
!>           على a( 1, 1 )، وتحتوي AP( 2 ) و AP( 3 ) على a( 1, 2 )
!>           و a( 2, 2 ) على التوالي، وهكذا.
!>           قبل الإدخال، مع UPLO = 'L' أو 'l'، يجب أن تحتوي مصفوفة AP
!>           على الجزء المثلثي السفلي من المصفوفة المتناظرة
!>           مجمّعًا بالتتابع، عمودًا بعمود، بحيث تحتوي AP( 1 )
!>           على a( 1, 1 )، وتحتوي AP( 2 ) و AP( 3 ) على a( 2, 1 )
!>           و a( 3, 1 ) على التوالي، وهكذا.
!>           غير متغير عند الخروج.
!>

X

!>          X is COMPLEX array, dimension at least
!>           ( 1 + ( N - 1 )*abs( INCX ) ).
!>           قبل الإدخال، يجب أن يحتوي المتجه X المعزز بالفواصل على المتجه x
!>           ذي العناصر N.
!>           غير متغير عند الخروج.
!>

INCX

!>          INCX is INTEGER
!>           عند الإدخال، يحدد INCX الفاصل بين عناصر X. يجب ألا يكون INCX صفرًا.
!>           غير متغير عند الخروج.
!>

BETA

!>          BETA is COMPLEX
!>           عند الإدخال، يحدد BETA المقدار القياسي beta. عندما يتم تقديم BETA
!>           كصفر، فإن Y لا تحتاج إلى التعيين عند الإدخال.
!>           غير متغير عند الخروج.
!>

Y

!>          Y is COMPLEX array, dimension at least
!>           ( 1 + ( N - 1 )*abs( INCY ) ).
!>           قبل الإدخال، يجب أن يحتوي المتجه Y المعزز بالفواصل على المتجه y
!>           ذي العناصر n. عند الخروج، يتم استبدال Y بالمتجه y المحدث.
!>

INCY

!>          INCY is INTEGER
!>           عند الإدخال، يحدد INCY الفاصل بين عناصر Y. يجب ألا يكون INCY صفرًا.
!>           غير متغير عند الخروج.
!>
