subroutine csysv_rk (character uplo, integer n, integer nrhs, complex, dimension( lda, * ) a, integer lda, complex, dimension( * ) e, integer, dimension( * ) ipiv, complex, dimension( ldb, * ) b, integer ldb, complex, dimension( * ) work, integer lwork, integer info)

CSYSV_RK تحسب حل نظام المعادلات الخطية A * X = B لمصفوفات SY

الغرض:

!> CSYSV_RK تحسب حل نظام المعادلات الخطية المركبة A * X = B، حيث A مصفوفة متناظرة N × N و X و B مصفوفتان N × NRHS.
!>
!> تُستخدم طريقة الارتكاز القطري التراكمي بونش-كوفمان (rook) لتفكيك A إلى:
!>    A = P*U*D*(U**T)*(P**T)، إذا كان UPLO = 'U'، أو
!>    A = P*L*D*(L**T)*(P**T)، إذا كان UPLO = 'L'،
!> حيث U (أو L) مصفوفة مثلثية علوية (أو سفلية) وحدوية،
!> U**T (أو L**T) هي منقولة U (أو L)، P مصفوفة تبديل، P**T هي منقولة P، و D مصفوفة متناظرة وقطرية جزئية ذات كتل قطرية 1×1 و 2×2.
!>
!> يتم استدعاء CSYTRF_RK لحساب تفكيك مصفوفة متناظرة مركبة. ثم يتم استخدام الشكل المفكك لـ A لحل
!> نظام المعادلات A * X = B عن طريق استدعاء روتين BLAS3 CSYTRS_3.
!>

المعلمات

UPLO 
!>          UPLO هي CHARACTER*1
!>          تحدد ما إذا كان الجزء المثلثي العلوي أو السفلي للمصفوفة المتناظرة A مخزنًا:
!>          = 'U':  يتم تخزين المثلث العلوي لـ A؛
!>          = 'L':  يتم تخزين المثلث السفلي لـ A.
!>

N

!>          N هي INTEGER
!>          عدد المعادلات الخطية، أي، رتبة المصفوفة A. N >= 0.
!>

NRHS

!>          NRHS هي INTEGER
!>          عدد الأطراف اليمنى، أي، عدد أعمدة المصفوفة B. NRHS >= 0.
!>

A

!>          A هي مصفوفة COMPLEX، حجمها (LDA,N)
!>          عند الإدخال، المصفوفة المتناظرة A.
!>            إذا كان UPLO = 'U': الجزء المثلثي العلوي N × N من A يحتوي على الجزء المثلثي العلوي للمصفوفة A،
!>            والجزء السفلي الصارم للمصفوفة A لا يتم الرجوع إليه.
!>
!>            إذا كان UPLO = 'L': الجزء المثلثي السفلي N × N من A يحتوي على الجزء المثلثي السفلي للمصفوفة A،
!>            والجزء العلوي الصارم للمصفوفة A لا يتم الرجوع إليه.
!>
!>          عند الخروج، إذا كان INFO = 0، القطر للمصفوفة القطرية الجزئية
!>          D والعوامل U أو L كما تم حسابها بواسطة CSYTRF_RK:
!>            أ) العناصر القطرية فقط للمصفوفة المتناظرة القطرية الجزئية
!>               D على قطر A، أي D(k,k) = A(k,k)؛
!>               (العناصر فوق القطرية (أو تحت القطرية) لـ D
!>                مخزنة عند الخروج في المصفوفة E)، و
!>            ب) إذا كان UPLO = 'U': العامل U في الجزء فوق القطري لـ A.
!>               إذا كان UPLO = 'L': العامل L في الجزء تحت القطري لـ A.
!>
!>          لمزيد من المعلومات، راجع وصف الروتين الفرعي CSYTRF_RK.
!>

LDA

!>          LDA هي INTEGER
!>          البعد الرائد للمصفوفة A. LDA >= max(1,N).
!>

E

!>          E هي مصفوفة COMPLEX، حجمها (N)
!>          عند الخروج، تحتوي على الإخراج الذي تم حسابه بواسطة روتين التفكيك
!>          CSYTRF_RK، أي، العناصر فوق القطرية (أو تحت القطرية) للمصفوفة المتناظرة القطرية الجزئية
!>          مع الكتل القطرية 1×1 أو 2×2، حيث:
!>          إذا كان UPLO = 'U': E(i) = D(i-1,i)، i=2:N، E(1) تم تعيينه إلى 0؛
!>          إذا كان UPLO = 'L': E(i) = D(i+1,i)، i=1:N-1، E(N) تم تعيينه إلى 0.
!>
!>          ملاحظة: بالنسبة لكتلة قطرية 1×1 D(k)، حيث
!>          1 <= k <= N، يتم تعيين العنصر E(k) إلى 0 في كل من
!>          حالات UPLO = 'U' أو UPLO = 'L'.
!>
!>          لمزيد من المعلومات، راجع وصف الروتين الفرعي CSYTRF_RK.
!>

IPIV

!>          IPIV هي مصفوفة INTEGER، حجمها (N)
!>          تفاصيل التبادلات وهيكل الكتلة لـ D،
!>          كما تم تحديده بواسطة CSYTRF_RK.
!>
!>          لمزيد من المعلومات، راجع وصف الروتين الفرعي CSYTRF_RK.
!>

B

!>          B هي مصفوفة COMPLEX، حجمها (LDB,NRHS)
!>          عند الإدخال، المصفوفة الطرفية اليمنى B ذات الأبعاد N × NRHS.
!>          عند الخروج، إذا كان INFO = 0، المصفوفة الحل X ذات الأبعاد N × NRHS.
!>

LDB

!>          LDB هي INTEGER
!>          البعد الرائد للمصفوفة B. LDB >= max(1,N).
!>

WORK

!>          WORK هي مصفوفة COMPLEX، حجمها ( MAX(1,LWORK) ).
!>          مصفوفة العمل المستخدمة في مرحلة التفكيك.
!>          عند الخروج، إذا كان INFO = 0، فإن WORK(1) تعيد LWORK المثلى.
!>

LWORK

!>          LWORK هي INTEGER
!>          طول WORK. LWORK >= 1. للحصول على أفضل أداء
!>          لمرحلة التفكيك LWORK >= max(1,N*NB)، حيث NB هو
!>          حجم الكتلة الأمثل لـ CSYTRF_RK.
!>
!>          إذا كان LWORK = -1، فسيتم افتراض استعلام عن مساحة العمل؛
!>          يقوم الروتين بحساب حجم مصفوفة العمل الأمثل فقط
!>          لمرحلة التفكيك، ويعيد هذه القيمة في
!>          المدخل الأول لمصفوفة العمل، ولا يتم إصدار رسالة خطأ
!>          متعلقة بـ LWORK بواسطة XERBLA.
!>

INFO

!>          INFO هي INTEGER
!>          = 0: خروج ناجح
!>
!>          < 0: إذا كان INFO = -k، فإن المعامل k كان له قيمة غير قانونية
!>
!>          > 0: إذا كان INFO = k، فإن المصفوفة A منفردة، بسبب:
!>                 إذا كان UPLO = 'U': العمود k في الجزء
!>                 المثلثي العلوي من A يحتوي على كل الأصفار.
!>                 إذا كان UPLO = 'L': العمود k في الجزء
!>                 المثلثي السفلي من A يحتوي على كل الأصفار.
!>
!>               وبالتالي فإن D(k,k) تساوي صفر تمامًا، والعناصر فوق القطرية
!>               للعمود k من U (أو العناصر تحت القطرية لـ
!>               العمود k من L) كلها أصفار. تم إكمال التفكيك،
!>               ولكن المصفوفة القطرية الجزئية D منفردة تمامًا،
!>               وسيحدث قسمة على صفر إذا تم استخدامها لحل نظام معادلات.
!>
!>               ملاحظة: يخزن INFO حدوث التفرد الأول فقط،
!>               أي حدوث تفرد لاحق لا يتم تخزينه في INFO على الرغم من
!>               أن التفكيك يكتمل دائمًا.
!>
