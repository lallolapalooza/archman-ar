subroutine csycon (character uplo, integer n, complex, dimension( lda, * ) a, integer lda, integer, dimension( * ) ipiv, real anorm, real rcond, complex, dimension( * ) work, integer info)

CSYCON

الغرض:

!>
!> CSYCON تقدر المقلوب لرقم الشرط (بال-1-norm) لمصفوفة متناظرة مركبة A باستخدام التحليل
!> A = U*D*U**T أو A = L*D*L**T المحسوب بواسطة CSYTRF.
!>
!> يتم الحصول على تقدير لـ norm(inv(A))، ويتم حساب مقلوب رقم الشرط على أنه RCOND = 1 / (ANORM * norm(inv(A))).
!>

المعلمات

UPLO 
!>          UPLO هو CHARACTER*1
!>          يحدد ما إذا كانت تفاصيل التحليل مخزنة
!>          كمصفوفة مثلثية علوية أو سفلية.
!>          = 'U':  مثلثية علوية، الشكل هو A = U*D*U**T;
!>          = 'L':  مثلثية سفلية، الشكل هو A = L*D*L**T.
!>

N

!>          N هو INTEGER
!>          ترتيب المصفوفة A.  N >= 0.
!>

A

!>          A هو مصفوفة COMPLEX، حجمها (LDA,N)
!>          المصفوفة القطرية الكتلية D والمضاعفات المستخدمة
!>          للحصول على العامل U أو L كما تم حسابه بواسطة CSYTRF.
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرائد للمصفوفة A.  LDA >= max(1,N).
!>

IPIV

!>          IPIV هو مصفوفة INTEGER، حجمها (N)
!>          تفاصيل التبادلات والبنية الكتلية لـ D
!>          كما تم تحديدها بواسطة CSYTRF.
!>

ANORM

!>          ANORM هو REAL
!>          القيمة المطلقة 1 للمصفوفة الأصلية A.
!>

RCOND

!>          RCOND هو REAL
!>          مقلوب رقم الشرط للمصفوفة A،
!>          محسوب على أنه RCOND = 1/(ANORM * AINVNM)، حيث AINVNM هو
!>          تقدير للقيمة المطلقة 1 لـ inv(A) محسوب في هذا الروتين.
!>

WORK

!>          WORK هو مصفوفة COMPLEX، حجمها (2*N)
!>

INFO

!>          INFO هو INTEGER
!>          = 0:  خروج ناجح
!>          < 0:  إذا كان INFO = -i، فإن الوسيط i كان له قيمة غير قانونية
!>
