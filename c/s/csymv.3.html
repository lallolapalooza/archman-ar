subroutine csymv (character uplo, integer n, complex alpha, complex, dimension( lda, * ) a, integer lda, complex, dimension( * ) x, integer incx, complex beta, complex, dimension( * ) y, integer incy)

CSYMV تحسب ناتج ضرب مصفوفة-متجه لمصفوفة متناظرة مركبة.

الغرض:

!>
!> CSYMV  ينفذ عملية ضرب المصفوفة بالمتجه
!>
!>    y := alpha*A*x + beta*y,
!>
!> حيث alpha و beta قيمتان قياسيتان، x و y متجهان من n عنصر،
!!> و A مصفوفة متناظرة بحجم n × n.
!>

المعلمات

UPLO 
!>          UPLO هو CHARACTER*1
!>           عند الدخول، تحدد UPLO ما إذا كان الجزء العلوي أو السفلي
!>           من المصفوفة A سيتم الرجوع إليه على النحو التالي:
!>
!>              UPLO = 'U' أو 'u'   سيتم الرجوع فقط إلى الجزء العلوي من A
!>                                  .
!>
!>              UPLO = 'L' أو 'l'   سيتم الرجوع فقط إلى الجزء السفلي من A
!>                                  .
!>
!>           لا تتغير عند الخروج.
!>

N

!>          N هو INTEGER
!>           عند الدخول، تحدد N رتبة المصفوفة A.
!>           يجب أن تكون N أكبر من أو تساوي الصفر.
!>           لا تتغير عند الخروج.
!>

ALPHA

!>          ALPHA هو COMPLEX
!>           عند الدخول، تحدد ALPHA القيمة القياسية alpha.
!>           لا تتغير عند الخروج.
!>

A

!>          A هي مصفوفة COMPLEX، البعد ( LDA, N )
!>           قبل الدخول، مع UPLO = 'U' أو 'u'، يجب أن يحتوي الجزء العلوي
!>           من المصفوفة A بحجم n × n على الجزء العلوي
!>           من المصفوفة المتناظرة، والجزء السفلي تمامًا من A
!>           لا يتم الرجوع إليه.
!>           قبل الدخول، مع UPLO = 'L' أو 'l'، يجب أن يحتوي الجزء السفلي
!>           من المصفوفة A بحجم n × n على الجزء السفلي
!>           من المصفوفة المتناظرة، والجزء العلوي تمامًا من A
!>           لا يتم الرجوع إليه.
!>           لا تتغير عند الخروج.
!>

LDA

!>          LDA هو INTEGER
!>           عند الدخول، تحدد LDA البعد الأول لـ A كما هو معلن
!>           في البرنامج (الفرعي) المستدعي. يجب أن تكون LDA على الأقل
!>           max( 1, N ).
!>           لا تتغير عند الخروج.
!>

X

!>          X هي مصفوفة COMPLEX، البعد على الأقل
!>           ( 1 + ( N - 1 )*abs( INCX ) ).
!>           قبل الدخول، يجب أن تحتوي المصفوفة X ذات الفهرسة المتزايدة على المتجه
!>           x المكون من N عنصر.
!>           لا يتغير عند الخروج.
!>

INCX

!>          INCX هو INTEGER
!>           عند الدخول، يحدد INCX الزيادة لعناصر X.
!>           يجب ألا يكون INCX صفرًا.
!>           لا يتغير عند الخروج.
!>

BETA

!>          BETA هو COMPLEX
!>           عند الدخول، تحدد BETA القيمة القياسية beta. عندما يتم
!>           توفير BETA كصفر، فلن تحتاج Y إلى الإعداد عند الإدخال.
!>           لا تتغير عند الخروج.
!>

Y

!>          Y هي مصفوفة COMPLEX، البعد على الأقل
!>           ( 1 + ( N - 1 )*abs( INCY ) ).
!>           قبل الدخول، يجب أن تحتوي المصفوفة Y ذات الفهرسة المتزايدة على المتجه
!>           y المكون من n عنصر. عند الخروج، سيتم استبدال Y بالمتجه
!>           y المحدث.
!>

INCY

!>          INCY هو INTEGER
!>           عند الدخول، يحدد INCY الزيادة لعناصر Y.
!>           يجب ألا يكون INCY صفرًا.
!>           لا يتغير عند الخروج.
!>
