subroutine csysv_aa (character uplo, integer n, integer nrhs, complex, dimension( lda, * ) a, integer lda, integer, dimension( * ) ipiv, complex, dimension( ldb, * ) b, integer ldb, complex, dimension( * ) work, integer lwork, integer info)

CSYSV_AA تحسب حل نظام المعادلات الخطية A * X = B للمصفوفات SY

الغرض:

!>
!> CSYSV تحسب حل نظام المعادلات الخطية المعقدة
!>    A * X = B,
!> حيث A مصفوفة متناظرة N×N و X و B مصفوفتان N×NRHS.
!>
!> يتم استخدام خوارزمية Aasen لتحليل A كـ
!>    A = U**T * T * U,  إذا كان UPLO = 'U', أو
!>    A = L * T * L**T,  إذا كان UPLO = 'L',
!> حيث U (أو L) هو ناتج ضرب مصفوفة تبديل ومصفوفة مثلثية علوية (سفلية) وحدوية،
!> و T هي مصفوفة ثلاثية قطرية متناظرة. ثم يتم استخدام الشكل المحلل لـ A
!> لحل نظام المعادلات A * X = B.
!>

المعاملات

UPLO 
!>          UPLO هو CHARACTER*1
!>          = 'U':  المثلث العلوي من A مخزن؛
!>          = 'L':  المثلث السفلي من A مخزن.
!>

N

!>          N هو INTEGER
!>          عدد المعادلات الخطية، أي رتبة المصفوفة A.  N >= 0.
!>

NRHS

!>          NRHS هو INTEGER
!>          عدد الأطراف اليمنى، أي عدد أعمدة المصفوفة B.  NRHS >= 0.
!>

A

!>          A هي مصفوفة COMPLEX، البعد (LDA,N)
!>          عند الدخول، المصفوفة المتناظرة A.  إذا كان UPLO = 'U'، فإن الجزء
!>          المثلثي العلوي N×N الرائد من A يحتوي على الجزء المثلثي العلوي
!>          من المصفوفة A، ولا يتم الرجوع إلى الجزء المثلثي السفلي الصارم من A.
!>          إذا كان UPLO = 'L'، فإن الجزء المثلثي السفلي N×N الرائد من A يحتوي
!>          على الجزء المثلثي السفلي من المصفوفة A، ولا يتم الرجوع إلى الجزء
!>          المثلثي العلوي الصارم من A.
!>
!>          عند الخروج، إذا كان INFO = 0، فإن المصفوفة الثلاثية القطرية T
!>          والمضروبات المستخدمة للحصول على العامل U أو L من تحليل
!>          A = U**T*T*U أو A = L*T*L**T كما تم حسابه بواسطة CSYTRF.
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرائد للمصفوفة A.  LDA >= max(1,N).
!>

IPIV

!>          IPIV هي مصفوفة INTEGER، البعد (N)
!>          عند الخروج، تحتوي على تفاصيل التبادلات، أي أنه تم تبديل الصف والعمود k
!>          من A مع الصف والعمود IPIV(k).
!>

B

!>          B هي مصفوفة COMPLEX، البعد (LDB,NRHS)
!>          عند الدخول، مصفوفة الطرف الأيمن N×NRHS B.
!>          عند الخروج، إذا كان INFO = 0، فإن مصفوفة الحل N×NRHS X.
!>

LDB

!>          LDB هو INTEGER
!>          البعد الرائد للمصفوفة B.  LDB >= max(1,N).
!>

WORK

!>          WORK هي مصفوفة COMPLEX، البعد (MAX(1,LWORK))
!>          عند الخروج، إذا كان INFO = 0، فإن WORK(1) يعيد LWORK المثالي.
!>

LWORK

!>          LWORK هو INTEGER
!>          طول WORK.  LWORK >= MAX(2*N, 3*N-2)، ولأفضل أداء،
!>          LWORK >= max(1,N*NB)، حيث NB هو حجم الكتلة الأمثل لـ CSYTRF_AA.
!>
!>          إذا كان LWORK = -1، فسيتم افتراض استعلام مساحة العمل؛ تقوم الروتين
!>          بحساب حجم مساحة العمل المثلى فقط، وإعادة هذا الحجم كمدخل أول
!>          لمساحة العمل، ولا يصدر XERBLA أي رسالة خطأ تتعلق بـ LWORK.
!>

INFO

!>          INFO هو INTEGER
!>          = 0: تم الانتهاء بنجاح
!>          < 0: إذا كان INFO = -i، فإن المعامل i كان له قيمة غير قانونية
!>          > 0: إذا كان INFO = i، فإن D(i,i) يساوي صفرًا تمامًا. لقد اكتمل التحليل،
!>               ولكن المصفوفة القطرية المربعة D فريدة تمامًا، لذلك لم يكن من الممكن
!>               حساب الحل.
!>
