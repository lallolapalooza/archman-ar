subroutine cgemmtr (character uplo, character transa, character transb, integer n, integer k, complex alpha, complex, dimension(lda,*) a, integer lda, complex, dimension(ldb,*) b, integer ldb, complex beta, complex, dimension(ldc,*) c, integer ldc)

CGEMMTR

الغرض:

!>
!> CGEMMTR  تقوم بإحدى عمليات المصفوفة-المصفوفة
!>
!>    C := alpha*op( A )*op( B ) + beta*C,
!>
!> حيث  op( X )  هي واحدة مما يلي:
!>
!>    op( X ) = X   أو   op( X ) = X**T,
!>
!> alpha و beta هي قيم قياسية، و A، B، C هي مصفوفات، حيث op( A )
!> هي مصفوفة n × k،  op( B )  هي مصفوفة  k × n، و C هي مصفوفة n × n.
!> بناءً على ذلك، يقوم الإجراء الفرعي فقط بالوصول إلى الجزء المثلثي العلوي أو السفلي
!> من مصفوفة النتيجة C وتحديثه. يمكن استخدام هذا السلوك إذا
!> كان معروفًا أن المصفوفة الناتجة C هي هرميتية أو متماثلة.
!>

المعلمات

UPLO 
!>          UPLO هي CHARACTER*1
!>           عند الدخول، تحدد UPLO ما إذا كان الجزء المثلثي السفلي أو العلوي
!>           من C يتم الوصول إليه وتحديثه.
!>
!>              UPLO = 'L' أو 'l'، يتم استخدام الجزء المثلثي السفلي من C.
!>
!>              UPLO = 'U' أو 'u'، يتم استخدام الجزء المثلثي العلوي من C.
!>

TRANSA

!>          TRANSA هي CHARACTER*1
!>           عند الدخول، تحدد TRANSA شكل op( A ) لاستخدامها في
!>           ضرب المصفوفات على النحو التالي:
!>
!>              TRANSA = 'N' أو 'n'،  op( A ) = A.
!>
!>              TRANSA = 'T' أو 't'،  op( A ) = A**T.
!>
!>              TRANSA = 'C' أو 'c'،  op( A ) = A**H.
!>

TRANSB

!>          TRANSB هي CHARACTER*1
!>           عند الدخول، تحدد TRANSB شكل op( B ) لاستخدامها في
!>           ضرب المصفوفات على النحو التالي:
!>
!>              TRANSB = 'N' أو 'n'،  op( B ) = B.
!>
!>              TRANSB = 'T' أو 't'،  op( B ) = B**T.
!>
!>              TRANSB = 'C' أو 'c'،  op( B ) = B**H.
!>

N

!>          N هو INTEGER
!>           عند الدخول، تحدد N عدد الصفوف والأعمدة
!>           للمصفوفة C، وعدد الأعمدة في op(B)، وعدد
!>           الصفوف في op(A). يجب أن تكون N أكبر من أو تساوي صفر.
!>

K

!>          K هو INTEGER
!>           عند الدخول، تحدد K عدد الأعمدة في المصفوفة
!>           op( A ) وعدد الصفوف في المصفوفة op( B ). يجب أن يكون
!>           K أكبر من أو يساوي صفر.
!>

ALPHA

!>          ALPHA هو COMPLEX.
!>           عند الدخول، تحدد ALPHA القيمة القياسية alpha.
!>

A

!>          A هي مصفوفة COMPLEX، أبعادها ( LDA, ka )، حيث ka هي
!>           k  عندما  TRANSA = 'N' أو 'n'،  و n  بخلاف ذلك.
!>           قبل الدخول مع  TRANSA = 'N' أو 'n'،  يجب أن يحتوي الجزء الرائد
!>           n × k  من المصفوفة  A  على المصفوفة  A،  وإلا
!>           يجب أن يحتوي الجزء الرائد  k × m  من المصفوفة  A  على
!>           المصفوفة A.
!>

LDA

!>          LDA هو INTEGER
!>           عند الدخول، تحدد LDA البعد الأول لـ A كما هو معلن
!>           في البرنامج المتصل (الفرعي). عندما  TRANSA = 'N' أو 'n'،
!>           يجب أن تكون LDA أكبر من أو تساوي  max( 1, n )، وإلا
!>           يجب أن تكون LDA أكبر من أو تساوي  max( 1, k ).
!>

B

!>          B هي مصفوفة COMPLEX، أبعادها ( LDB, kb )، حيث kb هي
!>           n  عندما  TRANSB = 'N' أو 'n'،  و k  بخلاف ذلك.
!>           قبل الدخول مع  TRANSB = 'N' أو 'n'،  يجب أن يحتوي الجزء الرائد
!>           k × n  من المصفوفة  B  على المصفوفة  B،  وإلا
!>           يجب أن يحتوي الجزء الرائد  n × k  من المصفوفة  B  على
!>           المصفوفة B.
!>

LDB

!>          LDB هو INTEGER
!>           عند الدخول، تحدد LDB البعد الأول لـ B كما هو معلن
!>           في البرنامج المتصل (الفرعي). عندما  TRANSB = 'N' أو 'n'،
!>           يجب أن تكون LDB أكبر من أو تساوي  max( 1, k )، وإلا
!>           يجب أن تكون LDB أكبر من أو تساوي  max( 1, n ).
!>

BETA

!>          BETA هو COMPLEX.
!>           عند الدخول، تحدد BETA القيمة القياسية beta.  عندما تكون
!>           BETA  صفرًا، لا يلزم تعيين C عند الإدخال.
!>

C

!>          C هي مصفوفة COMPLEX، أبعادها ( LDC, N )
!>           قبل الدخول، يجب أن يحتوي الجزء الرائد
!>           n × n  من المصفوفة  C  على المصفوفة  C،  باستثناء عندما يكون
!>           beta صفرًا، وفي هذه الحالة لا يلزم تعيين C عند الإدخال.
!>           عند الخروج، يتم استبدال الجزء العلوي أو السفلي المثلثي من المصفوفة
!>           C  بالمصفوفة n × n
!>           ( alpha*op( A )*op( B ) + beta*C ).
!>

LDC

!>          LDC هو INTEGER
!>           عند الدخول، تحدد LDC البعد الأول لـ C كما هو معلن
!>           في  البرنامج  المتصل  (الفرعي).   يجب  أن  تكون
!>           LDC  أكبر من أو تساوي  max( 1, n ).
!>
