subroutine cgeqrt (integer m, integer n, integer nb, complex, dimension( lda, * ) a, integer lda, complex, dimension( ldt, * ) t, integer ldt, complex, dimension( * ) work, integer info)

CGEQRT

الغرض:

!>
!> CGEQRT يحسب تحليل QR المكدس لمصفوفة مركبة M-by-N A
!> باستخدام تمثيل WY المدمج لـ Q.
!>

المعلمات

M 
!>          M هو عدد صحيح
!>          عدد صفوف المصفوفة A.  M >= 0.
!>

N

!>          N هو عدد صحيح
!>          عدد أعمدة المصفوفة A.  N >= 0.
!>

NB

!>          NB هو عدد صحيح
!>          حجم الكتلة المراد استخدامه في QR المكدس.  MIN(M,N) >= NB >= 1.
!>

A

!>          A هي مصفوفة مركبة، البعد (LDA,N)
!>          عند الإدخال، المصفوفة M-by-N A.
!>          عند الخروج، تحتوي العناصر على القطر وفوقه في المصفوفة
!>          على المصفوفة شبه المنحرفة العلوية R (R هي
!>          مثلثية علوية إذا كانت M >= N)؛ العناصر أسفل القطر
!>          هي أعمدة V.
!>

LDA

!>          LDA هو عدد صحيح
!>          البعد الرئيسي للمصفوفة A.  LDA >= max(1,M).
!>

T

!>          T هي مصفوفة مركبة، البعد (LDT,MIN(M,N))
!>          المتعكسات الكتلية المثلثية العلوية المخزنة بالشكل المدمج
!>          كسلسلة من الكتل المثلثية العلوية.  انظر أدناه
!>          لمزيد من التفاصيل.
!>

LDT

!>          LDT هو عدد صحيح
!>          البعد الرئيسي للمصفوفة T.  LDT >= NB.
!>

WORK

!>          WORK هي مصفوفة مركبة، البعد (NB*N)
!>

INFO

!>          INFO هو عدد صحيح
!>          = 0:  انتهاء ناجح
!>          < 0:  إذا كان INFO = -i، كان الوسيط i له قيمة غير قانونية
!>

تفاصيل إضافية:

!>
!>  المصفوفة V تخزن المتعكسات الأولية H(i) في العمود i
!>  أسفل القطر. على سبيل المثال، إذا كان M=5 و N=3، فإن المصفوفة V هي
!>
!>               V = (  1       )
!>                   ( v1  1    )
!>                   ( v1 v2  1 )
!>                   ( v1 v2 v3 )
!>                   ( v1 v2 v3 )
!>
!>  حيث تمثل vi's المتجهات التي تحدد H(i)، والتي يتم إرجاعها
!>  في المصفوفة A.  لا يتم تخزين الرقم 1 على طول قطر V في A.
!>
!>  ليكن K=MIN(M,N).  عدد الكتل هو B = سقف(K/NB)، حيث كل
!>  كتلة من الدرجة NB ما عدا الكتلة الأخيرة، والتي هي من الدرجة
!>  IB = K - (B-1)*NB.  لكل من الكتل B، يتم حساب عامل متعكس كتلي مثلثي علوي:
!>  T1، T2، ...، TB.  تخزن T's بحجم NB-by-NB (و IB-by-IB
!>  للأخيرة) في المصفوفة T بحجم NB-by-K بالشكل
!>
!>               T = (T1 T2 ... TB).
!>
