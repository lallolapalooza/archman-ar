subroutine cgbequb (integer m, integer n, integer kl, integer ku, complex, dimension( ldab, * ) ab, integer ldab, real, dimension( * ) r, real, dimension( * ) c, real rowcnd, real colcnd, real amax, integer info)

CGBEQUB

الغرض:

!>
!> CGBEQUB يحسب عوامل القياس للصف والأعمدة المقصود بها موازنة مصفوفة A بحجم
!> M×N وتقليل رقم شرطها. تعيد R عوامل قياس الصفوف و C عوامل قياس الأعمدة،
!> المختارة لمحاولة جعل أكبر عنصر في كل صف وعمود من المصفوفة B، ذات العناصر
!> B(i,j)=R(i)*A(i,j)*C(j)، قيمة مطلقة لا تزيد عن الأساس.
!>
!> R(i) و C(j) مقيدة بأن تكون قوة للأساس بين SMLNUM = أصغر عدد آمن
!> و BIGNUM = أكبر عدد آمن. استخدام عوامل القياس هذه لا يضمن تقليل رقم شرط A
!> ولكنه يعمل بشكل جيد في الممارسة العملية.
!>
!> هذا الإجراء يختلف عن CGEEQU بتقييد عوامل القياس لتكون قوة للأساس.
!> مع استبعاد حالات الحمل الزائد والتحميل الناقص، القياس بهذه العوامل لا يسبب
!> أخطاء تقريب إضافية. ومع ذلك، فإن قيم العناصر المقاسة لم تعد تقريبًا 1 ولكنها
!> تقع بين sqrt(radix) و 1/sqrt(radix).
!>

المعلمات

M 
!>          M هو عدد صحيح (INTEGER)
!>          عدد صفوف المصفوفة A.  M >= 0.
!>

N

!>          N هو عدد صحيح (INTEGER)
!>          عدد أعمدة المصفوفة A.  N >= 0.
!>

KL

!>          KL هو عدد صحيح (INTEGER)
!>          عدد الأقطار الفرعية داخل نطاق A.  KL >= 0.
!>

KU

!>          KU هو عدد صحيح (INTEGER)
!>          عدد الأقطار الفائقة داخل نطاق A.  KU >= 0.
!>

AB

!>          AB هو مصفوفة مركبة (COMPLEX array)، البعد (LDAB,N)
!>          عند الدخول، المصفوفة A بتخزين النطاق، في الصفوف 1 إلى KL+KU+1.
!>          يتم تخزين العمود j من A في العمود j من المصفوفة
!>          AB كما يلي:
!>          AB(KU+1+i-j,j) = A(i,j) لـ max(1,j-KU)<=i<=min(N,j+kl)
!>

LDAB

!>          LDAB هو عدد صحيح (INTEGER)
!>          البعد الرئيسي للمصفوفة A.  LDAB >= max(1,M).
!>

R

!>          R هي مصفوفة حقيقية (REAL array)، البعد (M)
!>          إذا كان INFO = 0 أو INFO > M، فإن R تحتوي على عوامل قياس الصفوف
!>          لـ A.
!>

C

!>          C هي مصفوفة حقيقية (REAL array)، البعد (N)
!>          إذا كان INFO = 0، فإن C تحتوي على عوامل قياس الأعمدة لـ A.
!>

ROWCND

!>          ROWCND هو عدد حقيقي (REAL)
!>          إذا كان INFO = 0 أو INFO > M، فإن ROWCND يحتوي على نسبة
!>          أصغر R(i) إلى أكبر R(i).  إذا كان ROWCND >= 0.1 وكان
!>          AMAX ليس كبيرًا جدًا أو صغيرًا جدًا، فلا يستحق القياس بواسطة R.
!>

COLCND

!>          COLCND هو عدد حقيقي (REAL)
!>          إذا كان INFO = 0، فإن COLCND يحتوي على نسبة أصغر
!>          C(i) إلى أكبر C(i).  إذا كان COLCND >= 0.1، فلا يستحق
!>          القياس بواسطة C.
!>

AMAX

!>          AMAX هو عدد حقيقي (REAL)
!>          القيمة المطلقة لأكبر عنصر في المصفوفة.  إذا كان AMAX قريبًا جدًا
!>          من الحمل الزائد أو قريبًا جدًا من التحميل الناقص، يجب قياس
!>          المصفوفة.
!>

INFO

!>          INFO هو عدد صحيح (INTEGER)
!>          = 0:  خروج ناجح
!>          < 0:  إذا كان INFO = -i، فإن المعلمة i كانت لها قيمة غير قانونية
!>          > 0:  إذا كان INFO = i، وكان i
!>                <= M:  الصف i من A هو صفر تمامًا
!>                >  M:  العمود (i-M) من A هو صفر تمامًا
!>
