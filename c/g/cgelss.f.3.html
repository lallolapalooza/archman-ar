توثيق الوظيفة/الروتين الفرعي

subroutine cgelss (integer m, integer n, integer nrhs, complex, dimension( lda, * ) a, integer lda, complex, dimension( ldb, * ) b, integer ldb, real, dimension( * ) s, real rcond, integer rank, complex, dimension( * ) work, integer lwork, real, dimension( * ) rwork, integer info)

CGELSS يحل أنظمة مفرطة التحديد أو ناقصة التحديد لمصفوفات GE

الغرض:

!>
!> CGELSS يحسب الحل بأقل معيار لمشكلة المربعات الصغرى الخطية المعقدة:
!>
!> تقليل معيار 2 لـ | b - A*x |.
!>
!> باستخدام التحلل القيمة المفردة (SVD) لـ A. A هي مصفوفة M-by-N
!> قد تكون ذات رتبة منخفضة.
!>
!> يمكن التعامل مع العديد من المتجهات الجانبية اليمنى b ومتجهات الحل x
!> في استدعاء واحد؛ يتم تخزينها كأعمدة
!> المصفوفة B ذات الجانب الأيمن M-by-NRHS والمصفوفة X ذات الحل N-by-NRHS.
!>
!> يتم تحديد الرتبة الفعالة لـ A عن طريق معاملة القيم المفردة
!> التي تقل عن RCOND مضروبة في أكبر قيمة مفردة كصفر.
!>

المعلمات

M 
!>          M هو عدد صحيح (INTEGER)
!>          عدد الصفوف في المصفوفة A. M >= 0.
!>

N

!>          N هو عدد صحيح (INTEGER)
!>          عدد الأعمدة في المصفوفة A. N >= 0.
!>

NRHS

!>          NRHS هو عدد صحيح (INTEGER)
!>          عدد الأجناب اليمنى، أي عدد الأعمدة
!>          للمصفوفتين B و X. NRHS >= 0.
!>

A

!>          A هي مصفوفة مركبة (COMPLEX array)، ذات بُعد (LDA,N)
!>          عند الدخول، مصفوفة A بحجم M-by-N.
!>          عند الخروج، يتم استبدال أول min(m,n) صفوف من A
!>          بمتجهاتها المفردة اليمنى، مخزنة صفًا بصف.
!>

LDA

!>          LDA هو عدد صحيح (INTEGER)
!>          البعد الرئيسي للمصفوفة A. LDA >= max(1,M).
!>

B

!>          B هي مصفوفة مركبة (COMPLEX array)، ذات بُعد (LDB,NRHS)
!>          عند الدخول، مصفوفة الجانب الأيمن B بحجم M-by-NRHS.
!>          عند الخروج، يتم استبدال B بمصفوفة الحل X بحجم N-by-NRHS.
!>          إذا كانت m >= n و RANK = n، فإن مجموع مربعات البواقي
!>          للحل في العمود i يُعطى بمجموع
!>          مربعات قيم العناصر المطلقة n+1:m في ذلك العمود.
!>

LDB

!>          LDB هو عدد صحيح (INTEGER)
!>          البعد الرئيسي للمصفوفة B. LDB >= max(1,M,N).
!>

S

!>          S هي مصفوفة حقيقية (REAL array)، ذات بُعد (min(M,N))
!>          القيم المفردة لـ A بترتيب تنازلي.
!>          رقم شرط A بمعيار 2 = S(1)/S(min(m,n)).
!>

RCOND

!>          RCOND هو عدد حقيقي (REAL)
!>          يُستخدم RCOND لتحديد الرتبة الفعالة لـ A.
!>          القيم المفردة S(i) <= RCOND*S(1) تُعامل كصفر.
!>          إذا كان RCOND < 0، يتم استخدام دقة الجهاز بدلاً من ذلك.
!>

RANK

!>          RANK هو عدد صحيح (INTEGER)
!>          الرتبة الفعالة لـ A، أي عدد القيم المفردة
!>          التي تزيد عن RCOND*S(1).
!>

WORK

!>          WORK هي مصفوفة مركبة (COMPLEX array)، ذات بُعد (MAX(1,LWORK))
!>          عند الخروج، إذا كان INFO = 0، فإن WORK(1) يُرجع LWORK الأمثل.
!>

LWORK

!>          LWORK هو عدد صحيح (INTEGER)
!>          بُعد المصفوفة WORK. LWORK >= 1، وأيضًا:
!>          LWORK >=  2*min(M,N) + max(M,N,NRHS)
!>          لأداء جيد، يجب أن يكون LWORK أكبر بشكل عام.
!>
!>          إذا كان LWORK = -1، فإنه يفترض استعلام عن مساحة العمل؛ الروتين
!>          يحسب فقط الحجم الأمثل للمصفوفة WORK، ويعيد
!>          هذه القيمة كإدخال أول للمصفوفة WORK، ولا يصدر رسالة خطأ
!>          متعلقة بـ LWORK بواسطة XERBLA.
!>

RWORK

!>          RWORK هي مصفوفة حقيقية (REAL array)، ذات بُعد (5*min(M,N))
!>

INFO

!>          INFO هو عدد صحيح (INTEGER)
!>          = 0:  انتهاء ناجح
!>          < 0:  إذا كان INFO = -i، فإن المعلمة i كان لها قيمة غير قانونية.
!>          > 0:  فشل خوارزمية حساب SVD في التقارب؛
!>                إذا كان INFO = i، فإن i عنصرًا غير قطري لشكل ثنائي قطري وسيط
!>                لم يتقارب إلى الصفر.
!>
