subroutine cgemqr (character side, character trans, integer m, integer n, integer k, complex, dimension( lda, * ) a, integer lda, complex, dimension( * ) t, integer tsize, complex, dimension( ldc, * ) c, integer ldc, complex, dimension( * ) work, integer lwork, integer info)

CGEMQR

الغرض:

!>
!> CGEMQR تستبدل المصفوفة C الحقيقية العامة M-by-N بـ
!>
!>                      SIDE = 'L'     SIDE = 'R'
!>      TRANS = 'N':      Q * C          C * Q
!>      TRANS = 'T':      Q**H * C       C * Q**H
!>
!> حيث Q هي مصفوفة وحدوية مركبة معرفة كناتج
!> لمُنشِئات تعكس بسيطة محسوبة بواسطة تحليل QR الرفيع جدًا (CGEQR)
!>
!>

المعلمات

SIDE 
!>          SIDE is CHARACTER*1
!>          = 'L': تطبيق Q أو Q**H من اليسار؛
!>          = 'R': تطبيق Q أو Q**H من اليمين.
!>

TRANS

!>          TRANS is CHARACTER*1
!>          = 'N':  لا يوجد تبديل، تطبيق Q؛
!>          = 'C':  تبديل مرافق، تطبيق Q**H.
!>

M

!>          M is INTEGER
!>          عدد الصفوف للمصفوفة A. M >=0.
!>

N

!>          N is INTEGER
!>          عدد الأعمدة للمصفوفة C. N >= 0.
!>

K

!>          K is INTEGER
!>          عدد المُنشِئات التعاكسية البسيطة التي يُعرّفها حاصل الضرب
!>          للمصفوفة Q.
!>          إذا كانت SIDE = 'L'، فإن M >= K >= 0؛
!>          إذا كانت SIDE = 'R'، فإن N >= K >= 0.
!>

A

!>          A is COMPLEX array, dimension (LDA,K)
!>          جزء من هيكل البيانات لتمثيل Q كما تم إرجاعه بواسطة CGEQR.
!>

LDA

!>          LDA is INTEGER
!>          البعد الرئيسي للمصفوفة A.
!>          إذا كانت SIDE = 'L'، فإن LDA >= max(1,M)؛
!>          إذا كانت SIDE = 'R'، فإن LDA >= max(1,N).
!>

T

!>          T is COMPLEX array, dimension (MAX(5,TSIZE)).
!>          جزء من هيكل البيانات لتمثيل Q كما تم إرجاعه بواسطة CGEQR.
!>

TSIZE

!>          TSIZE is INTEGER
!>          بعد المصفوفة T. TSIZE >= 5.
!>

C

!>          C is COMPLEX array, dimension (LDC,N)
!>          عند الإدخال، المصفوفة C ذات الأبعاد M-by-N.
!>          عند الخروج، يتم استبدال C بـ Q*C أو Q**H*C أو C*Q**H أو C*Q.
!>

LDC

!>          LDC is INTEGER
!>          البعد الرئيسي للمصفوفة C. LDC >= max(1,M).
!>

WORK

!>          (مساحة عمل) COMPLEX array, dimension (MAX(1,LWORK))
!>          عند الخروج، إذا كان INFO = 0، فإن WORK(1) تُرجع الحد الأدنى LWORK.
!>

LWORK

!>          LWORK is INTEGER
!>          بعد المصفوفة WORK. LWORK >= 1.
!>          إذا كانت LWORK = -1، فيُفترض استعلام مساحة عمل. تقوم الدالة
!>          بحساب حجم مصفوفة WORK فقط، وتُرجع هذه القيمة كـ WORK(1)، ولا يصدر
!>          XERBLA أي رسالة خطأ متعلقة بـ WORK.
!>

INFO

!>          INFO is INTEGER
!>          = 0:  خروج ناجح
!>          < 0:  إذا كانت INFO = -i، فإن المعلمة i كانت لها قيمة غير قانونية
!>

!>
!>          T(2): حجم كتلة الصف (MB)
!>          T(3): حجم كتلة العمود (NB)

!>                           CLATSQR أو CGEQRT
!>
!>  اعتمادًا على أبعاد المصفوفة M و N، وأحجام كتل الصفوف والأعمدة MB و NB التي تم إرجاعها بواسطة ILAENV، ستستخدم CGEQR إما
!>  CLATSQR (إذا كانت المصفوفة رفيعة وطويلة) أو CGEQRT لحساب
!>  تحليل QR.
!>  سيستخدم هذا الإصدار من CGEMQR إما CLAMTSQR أو CGEMQRT لـ
!>  ضرب المصفوفة Q بمصفوفة أخرى.
!>  تفاصيل إضافية في CLAMTSQR أو CGEMQRT.

!>
