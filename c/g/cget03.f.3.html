توثيق الوظيفة/الروتين الفرعي

subroutine cget03 (integer n, complex, dimension( lda, * ) a, integer lda, complex, dimension( ldainv, * ) ainv, integer ldainv, complex, dimension( ldwork, * ) work, integer ldwork, real, dimension( * ) rwork, real rcond, real resid)

CGET03

الغرض:

!>
!> CGET03 تحسب الباقي لمصفوفة عامة مضروبة في معكوسها:
!>    norm( I - AINV*A ) / ( N * norm(A) * norm(AINV) * EPS ),
!> حيث EPS هو خطأ الآلة.
!>

المعلمات

N 
!>          N هو عدد صحيح (INTEGER)
!>          عدد الصفوف والأعمدة للمصفوفة A. N >= 0.
!>

A

!>          A هو مصفوفة مركبة (COMPLEX) ، البعد (LDA,N)
!>          المصفوفة الأصلية A بحجم N x N.
!>

LDA

!>          LDA هو عدد صحيح (INTEGER)
!>          البعد الرائد للمصفوفة A. LDA >= max(1,N).
!>

AINV

!>          AINV هي مصفوفة مركبة (COMPLEX) ، البعد (LDAINV,N)
!>          معكوس المصفوفة A.
!>

LDAINV

!>          LDAINV هو عدد صحيح (INTEGER)
!>          البعد الرائد للمصفوفة AINV. LDAINV >= max(1,N).
!>

WORK

!>          WORK هي مصفوفة مركبة (COMPLEX) ، البعد (LDWORK,N)
!>

LDWORK

!>          LDWORK هو عدد صحيح (INTEGER)
!>          البعد الرائد للمصفوفة WORK. LDWORK >= max(1,N).
!>

RWORK

!>          RWORK هي مصفوفة حقيقية (REAL) ، البعد (N)
!>

RCOND

!>          RCOND هو عدد حقيقي (REAL)
!>          مقلوب رقم شرط المصفوفة A ، محسوب على النحو التالي:
!>          ( 1/norm(A) ) / norm(AINV).
!>

RESID

!>          RESID هو عدد حقيقي (REAL)
!>          norm(I - AINV*A) / ( N * norm(A) * norm(AINV) * EPS )
!>
