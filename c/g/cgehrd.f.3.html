subroutine cgehrd (integer n, integer ilo, integer ihi, complex, dimension( lda, * ) a, integer lda, complex, dimension( * ) tau, complex, dimension( * ) work, integer lwork, integer info)

CGEHRD

الغرض:

!>
!> CGEHRD تختزل مصفوفة عامة مركبة A إلى شكل هسنبرغ العلوي H عن طريق
!> تحويل تشابه وحدوي:  Q**H * A * Q = H .
!>

المعلمات

N 
!>          N هو عدد صحيح (INTEGER)
!>          مرتبة المصفوفة A.  N >= 0.
!>

ILO

!>          ILO هو عدد صحيح (INTEGER)
!>

IHI

!>          IHI هو عدد صحيح (INTEGER)
!>
!>          يفترض أن A بالفعل مثلثية علوية في الصفوف
!>          والأعمدة 1:ILO-1 و IHI+1:N. يتم عادةً
!>          تعيين ILO و IHI بواسطة استدعاء سابق لـ CGEBAL؛ وإلا فيجب
!>          تعيينهما إلى 1 و N على التوالي. انظر التفاصيل الإضافية.
!>          1 <= ILO <= IHI <= N، إذا كانت N > 0؛ ILO=1 و IHI=0، إذا كانت N=0.
!>

A

!>          A هي مصفوفة مركبة (COMPLEX)، البعد (LDA,N)
!>          عند الدخول، المصفوفة العامة N×N المراد اختزالها.
!>          عند الخروج، يتم استبدال المثلث العلوي والقطر الفرعي الأول في A
!>          بمصفوفة هسنبرغ العلوية H، والعناصر أسفل القطر الفرعي الأول، مع
!>          المصفوفة TAU، تمثل المصفوفة الوحدوية Q كحاصل ضرب لمُعكِسات
!>          عنصريه. انظر التفاصيل الإضافية.
!>

LDA

!>          LDA هو عدد صحيح (INTEGER)
!>          البعد الرئيسي للمصفوفة A.  LDA >= max(1,N).
!>

TAU

!>          TAU هي مصفوفة مركبة (COMPLEX)، البعد (N-1)
!>          العوامل القياسية للمُعكِسات العنصرية (انظر التفاصيل الإضافية).
!>          العناصر 1:ILO-1 و IHI:N-1 من TAU يتم تعيينها إلى
!>          صفر.
!>

WORK

!>          WORK هي مصفوفة مركبة (COMPLEX)، البعد (MAX(1,LWORK))
!>          عند الخروج، إذا كان INFO = 0، فإن WORK(1) يعيد LWORK المثلى.
!>

LWORK

!>          LWORK هو عدد صحيح (INTEGER)
!>          طول المصفوفة WORK.  LWORK >= max(1,N).
!>          لأداء جيد، يجب أن يكون LWORK أكبر عمومًا.
!>
!>          إذا كان LWORK = -1، فيفترض استعلام عن مساحة العمل؛ الروتين
!>          يحسب فقط الحجم الأمثل للمصفوفة WORK، ويعيد
!>          هذه القيمة كالإدخال الأول للمصفوفة WORK، ولا يصدر أي رسالة خطأ
!>          متعلقة بـ LWORK بواسطة XERBLA.
!>

INFO

!>          INFO هو عدد صحيح (INTEGER)
!>          = 0:  تم الخروج بنجاح
!>          < 0:  إذا كان INFO = -i، فإن الوسيط i كان لديه قيمة غير قانونية.
!>

تفاصيل إضافية:

!>
!>  يتم تمثيل المصفوفة Q كحاصل ضرب لـ (ihi-ilo) مُعكِس
!>  عنصري
!>
!>     Q = H(ilo) H(ilo+1) . . . H(ihi-1).
!>
!>  كل H(i) له الشكل
!>
!>     H(i) = I - tau * v * v**H
!>
!>  حيث tau هو عامل قياسي مركب، و v هو متجه مركب مع
!>  v(1:i) = 0، v(i+1) = 1 و v(ihi+1:n) = 0؛ يتم تخزين v(i+2:ihi) عند
!>  الخروج في A(i+2:ihi,i)، و tau في TAU(i).
!>
!>  يتم توضيح محتويات A بالمثال التالي، مع
!>  n = 7، ilo = 2 و ihi = 6:
!>
!>  عند الدخول،                        عند الخروج،
!>
!>  ( a   a   a   a   a   a   a )    (  a   a   h   h   h   h   a )
!>  (     a   a   a   a   a   a )    (      a   h   h   h   h   a )
!>  (     a   a   a   a   a   a )    (      h   h   h   h   h   h )
!>  (     a   a   a   a   a   a )    (      v2  h   h   h   h   h )
!>  (     a   a   a   a   a   a )    (      v2  v3  h   h   h   h )
!>  (     a   a   a   a   a   a )    (      v2  v3  v4  h   h   h )
!>  (                         a )    (                          a )
!>
!>  حيث a يمثل عنصراً من المصفوفة الأصلية A، h يمثل عنصراً معدلاً
!>  من مصفوفة هسنبرغ العلوية H، و vi يمثل عنصراً من المتجه الذي يحدد H(i).
!>
!>  هذا الملف هو تعديل طفيف لـ LAPACK-3.0's CGEHRD
!>  إجراء فرعي يدمج التحسينات المقترحة بواسطة Quintana-Orti و
!>  Van de Geijn (2006). (انظر CLAHR2.)
!>
