وثائق الوظيفة/الروتين الفرعي

subroutine cgesv (integer n, integer nrhs, complex, dimension( lda, * ) a, integer lda, integer, dimension( * ) ipiv, complex, dimension( ldb, * ) b, integer ldb, integer info)

الغرض:

!>
!> CGESV يحسب حل نظام المعادلات الخطية المعقدة
!>    A * X = B,
!> حيث A مصفوفة N × N والمصفوفتان X و B هما N × NRHS.
!>
!> يتم استخدام تحليل LU مع التقسيم الجزئي وتبديلات الصفوف لتحليل A بالشكل
!>    A = P * L * U,
!> حيث P مصفوفة تبديل، L مصفوفة ثلاثية سفلية، و U مصفوفة علوية.  ثم يستخدم الشكل المحلل لـ A لحل
!> نظام المعادلات A * X = B.
!>

المعلمات

N 
!>          N هو عدد صحيح (INTEGER)
!>          عدد المعادلات الخطية، أي رتبة المصفوفة A.  N >= 0.
!>

NRHS

!>          NRHS هو عدد صحيح (INTEGER)
!>          عدد الجوانب اليمنى، أي عدد أعمدة المصفوفة B.  NRHS >= 0.
!>

A

!>          A هي مصفوفة مركبة (COMPLEX)، البعد (LDA,N)
!>          عند الإدخال، مصفوفة المعاملات A ذات البعد N × N.
!>          عند الخروج، العوامل L و U من التحليل
!>          A = P*L*U؛ لا يتم تخزين عناصر القطرية الواحدة لـ L.
!>

LDA

!>          LDA هو عدد صحيح (INTEGER)
!>          البعد الرئيسي للمصفوفة A.  LDA >= max(1,N).
!>

IPIV

!>          IPIV هي مصفوفة أعداد صحيحة (INTEGER)، البعد (N)
!>          فهارس المحور التي تحدد مصفوفة التبديل P؛
!>          تم تبديل الصف i من المصفوفة مع الصف IPIV(i).
!>

B

!>          B هي مصفوفة مركبة (COMPLEX)، البعد (LDB,NRHS)
!>          عند الإدخال، المصفوفة B ذات الجوانب اليمنى ذات البعد N × NRHS.
!>          عند الخروج، إذا كان INFO = 0، فإن مصفوفة الحل X ذات البعد N × NRHS.
!>

LDB

!>          LDB هو عدد صحيح (INTEGER)
!>          البعد الرئيسي للمصفوفة B.  LDB >= max(1,N).
!>

INFO

!>          INFO هو عدد صحيح (INTEGER)
!>          = 0:  انتهاء ناجح
!>          < 0:  إذا كان INFO = -i، فإن المعامل i له قيمة غير قانونية
!>          > 0:  إذا كان INFO = i، فإن U(i,i) يساوي صفرًا تمامًا.  لقد اكتمل التحليل،
!>                لكن العامل U مفرد تمامًا، لذلك لم يكن من الممكن حساب الحل.
!>
