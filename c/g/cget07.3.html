وثائق الوظيفة/الروتين الفرعي

subroutine cget07 (character trans, integer n, integer nrhs, complex, dimension( lda, * ) a, integer lda, complex, dimension( ldb, * ) b, integer ldb, complex, dimension( ldx, * ) x, integer ldx, complex, dimension( ldxact, * ) xact, integer ldxact, real, dimension( * ) ferr, logical chkferr, real, dimension( * ) berr, real, dimension( * ) reslts)

CGET07

الغرض:

!>
!> CGET07 تختبر حدود الخطأ للتحسين التكراري للحل المحسوب لنظام المعادلات op(A)*X = B، حيث A مصفوفة عامة n × n و op(A) = A أو A**T، اعتمادًا على TRANS.
!>
!> RESLTS(1) = اختبار حد الخطأ
!>           = norm(X - XACT) / ( norm(X) * FERR )
!>
!> يتم إرجاع قيمة كبيرة إذا لم تكن هذه النسبة أقل من واحد.
!>
!> RESLTS(2) = الخطأ المتبقي من روتين التحسين التكراري
!>           = الحد الأقصى لـ BERR / ( (n+1)*EPS + (*) )، حيث
!>             (*) = (n+1)*UNFL / (min_i (abs(op(A))*abs(X) +abs(b))_i )
!>

المعلمات

TRANS 
!>          TRANS هو CHARACTER*1
!>          يحدد شكل نظام المعادلات.
!>          = 'N':  A * X = B     (بدون تبديل)
!>          = 'T':  A**T * X = B  (تبديل)
!>          = 'C':  A**H * X = B  (تبديل مرافق = تبديل)
!>

N

!>          N هو INTEGER
!>          عدد صفوف المصفوفات X و XACT.  N >= 0.
!>

NRHS

!>          NRHS هو INTEGER
!>          عدد أعمدة المصفوفات X و XACT.  NRHS >= 0.
!>

A

!>          A هو مصفوفة COMPLEX، أبعادها (LDA,N)
!>          المصفوفة الأصلية A بحجم n × n.
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرائد للمصفوفة A.  LDA >= max(1,N).
!>

B

!>          B هو مصفوفة COMPLEX، أبعادها (LDB,NRHS)
!>          متجهات الطرف الأيمن لنظام المعادلات الخطية.
!>

LDB

!>          LDB هو INTEGER
!>          البعد الرائد للمصفوفة B.  LDB >= max(1,N).
!>

X

!>          X هو مصفوفة COMPLEX، أبعادها (LDX,NRHS)
!>          متجهات الحل المحسوبة. يتم تخزين كل متجه كعمود في المصفوفة X.
!>

LDX

!>          LDX هو INTEGER
!>          البعد الرائد للمصفوفة X.  LDX >= max(1,N).
!>

XACT

!>          XACT هو مصفوفة COMPLEX، أبعادها (LDXACT,NRHS)
!>          متجهات الحل الدقيقة. يتم تخزين كل متجه كعمود في المصفوفة XACT.
!>

LDXACT

!>          LDXACT هو INTEGER
!>          البعد الرائد للمصفوفة XACT.  LDXACT >= max(1,N).
!>

FERR

!>          FERR هو مصفوفة REAL، أبعادها (NRHS)
!>          حدود الخطأ الأمامية المقدرة لكل متجه حل X. إذا كان XTRUE هو الحل الحقيقي، فإن FERR يحدد حجم أكبر مدخل في (X - XTRUE) مقسومًا على حجم أكبر مدخل في X.
!>

CHKFERR

!>          CHKFERR هو LOGICAL
!>          تم تعيينه إلى .TRUE. للتحقق من FERR، و .FALSE. لعدم التحقق من FERR.
!>          عندما يكون نظام الاختبار غير منظم، قد يكون الحل في XACT غير صحيح.
!>

BERR

!>          BERR هو مصفوفة REAL، أبعادها (NRHS)
!>          الخطأ الخلفي النسبي المكون من كل متجه حل (أي، أصغر تغيير نسبي في أي مدخل من A أو B يجعل X حلاً دقيقًا).
!>

RESLTS

!>          RESLTS هو مصفوفة REAL، أبعادها (2)
!>          الحد الأقصى لمتجهات الحل NRHS للنسب:
!>          RESLTS(1) = norm(X - XACT) / ( norm(X) * FERR )
!>          RESLTS(2) = BERR / ( (n+1)*EPS + (*) )
!>
