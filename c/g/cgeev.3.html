subroutine cgeev (character jobvl, character jobvr, integer n, complex, dimension( lda, * ) a, integer lda, complex, dimension( * ) w, complex, dimension( ldvl, * ) vl, integer ldvl, complex, dimension( ldvr, * ) vr, integer ldvr, complex, dimension( * ) work, integer lwork, real, dimension( * ) rwork, integer info)

CGEEV يحسب القيم الذاتية، وبشكل اختياري، المتجهات الذاتية اليسرى و/أو اليمنى للمصفوفات GE

الغرض:

!>
!> CGEEV يحسب للمصفوفة المعقدة غير المتماثلة N-by-N A،
!> القيم الذاتية، وبشكل اختياري، المتجهات الذاتية اليسرى و/أو اليمنى.
!>
!> المتجه الذاتي الأيمن v(j) للمصفوفة A يحقق:
!>                  A * v(j) = lambda(j) * v(j)
!> حيث lambda(j) هي القيمة الذاتية الخاصة به.
!> المتجه الذاتي الأيسر u(j) للمصفوفة A يحقق:
!>               u(j)**H * A = lambda(j) * u(j)**H
!> حيث u(j)**H يمثل المرافق المتعاقب للمتجه u(j).
!>
!> يتم تطبيع المتجهات الذاتية المحسوبة لتكون لها معيار إقليدي
!> يساوي 1 وأكبر مكون حقيقي.
!>

المعلمات

JOBVL 
!>          JOBVL هو CHARACTER*1
!>          = 'N': المتجهات الذاتية اليسرى للمصفوفة A غير محسوبة؛
!>          = 'V': المتجهات الذاتية اليسرى للمصفوفة A محسوبة.
!>

JOBVR

!>          JOBVR هو CHARACTER*1
!>          = 'N': المتجهات الذاتية اليمنى للمصفوفة A غير محسوبة؛
!>          = 'V': المتجهات الذاتية اليمنى للمصفوفة A محسوبة.
!>

N

!>          N هو INTEGER
!>          رتبة المصفوفة A. N >= 0.
!>

A

!>          A هي مصفوفة COMPLEX، بأبعاد (LDA,N)
!>          عند الإدخال، المصفوفة N-by-N A.
!>          عند الخروج، تم استبدال A.
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرائد للمصفوفة A. LDA >= max(1,N).
!>

W

!>          W هي مصفوفة COMPLEX، بأبعاد (N)
!>          تحتوي W على القيم الذاتية المحسوبة.
!>

VL

!>          VL هي مصفوفة COMPLEX، بأبعاد (LDVL,N)
!>          إذا كان JOBVL = 'V'، يتم تخزين المتجهات الذاتية اليسرى u(j)
!>          واحدة تلو الأخرى في أعمدة VL، بنفس ترتيب
!>          قيمها الذاتية.
!>          إذا كان JOBVL = 'N'، لا يتم استخدام VL.
!>          u(j) = VL(:,j)، العمود j من VL.
!>

LDVL

!>          LDVL هو INTEGER
!>          البعد الرائد للمصفوفة VL. LDVL >= 1؛ إذا كان
!>          JOBVL = 'V'، فإن LDVL >= N.
!>

VR

!>          VR هي مصفوفة COMPLEX، بأبعاد (LDVR,N)
!>          إذا كان JOBVR = 'V'، يتم تخزين المتجهات الذاتية اليمنى v(j)
!>          واحدة تلو الأخرى في أعمدة VR، بنفس ترتيب
!>          قيمها الذاتية.
!>          إذا كان JOBVR = 'N'، لا يتم استخدام VR.
!>          v(j) = VR(:,j)، العمود j من VR.
!>

LDVR

!>          LDVR هو INTEGER
!>          البعد الرائد للمصفوفة VR. LDVR >= 1؛ إذا كان
!>          JOBVR = 'V'، فإن LDVR >= N.
!>

WORK

!>          WORK هي مصفوفة COMPLEX، بأبعاد (MAX(1,LWORK))
!>          عند الخروج، إذا كان INFO = 0، فإن WORK(1) ترجع LWORK الأمثل.
!>

LWORK

!>          LWORK هو INTEGER
!>          بعد المصفوفة WORK. LWORK >= max(1,2*N).
!>          لأداء جيد، يجب أن يكون LWORK أكبر بشكل عام.
!>
!>          إذا كان LWORK = -1، فسيتم افتراض استعلام مساحة العمل؛ يقوم الروتين
!>          بحساب حجم مصفوفة WORK الأمثل فقط، ويرجع
!>          هذه القيمة كالإدخال الأول لمصفوفة WORK، ولا يتم إصدار أي رسالة خطأ
!>          متعلقة بـ LWORK بواسطة XERBLA.
!>

RWORK

!>          RWORK هي مصفوفة REAL، بأبعاد (2*N)
!>

INFO

!>          INFO هو INTEGER
!>          = 0:  الخروج ناجح
!>          < 0:  إذا كان INFO = -i، فإن المعلمة i كانت قيمتها غير قانونية.
!>          > 0:  إذا كان INFO = i، فقد فشل خوارزمية QR في حساب جميع
!>                القيم الذاتية، ولم يتم حساب أي متجهات ذاتية؛
!>                العناصر i+1:N من W تحتوي على قيم ذاتية تم
!>                تقاربها.
!>
