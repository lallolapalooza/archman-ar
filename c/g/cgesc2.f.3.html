subroutine cgesc2 (integer n, complex, dimension( lda, * ) a, integer lda, complex, dimension( * ) rhs, integer, dimension( * ) ipiv, integer, dimension( * ) jpiv, real scale)

CGESC2 يحل نظام المعادلات الخطية باستخدام تحليل LU مع التقسيم الكامل الذي تم حسابه بواسطة sgetc2.

الغرض:

!>
!> CGESC2 يحل نظام المعادلات الخطية
!>
!>           A * X = scale* RHS
!>
!> مع مصفوفة عامة N×N هي A باستخدام تحليل LU مع
!> التقسيم الكامل الذي تم حسابه بواسطة CGETC2.
!>
!>

المعاملات

N 
!>          N هو عدد صحيح (INTEGER)
!>          عدد أعمدة المصفوفة A.
!>

A

!>          A هي مصفوفة مركبة (COMPLEX)، أبعادها (LDA, N)
!>          عند الإدخال، جزء LU من تحليل المصفوفة N×N
!>          A المحسوب بواسطة CGETC2: A = P * L * U * Q
!>

LDA

!>          LDA هو عدد صحيح (INTEGER)
!>          البعد الرائد للمصفوفة A. LDA >= max(1, N).
!>

RHS

!>          RHS هي مصفوفة مركبة (COMPLEX)، أبعادها N.
!>          عند الإدخال، متجه الطرف الأيمن b.
!>          عند الخروج، متجه الحل X.
!>

IPIV

!>          IPIV هي مصفوفة أعداد صحيحة (INTEGER)، أبعادها (N).
!>          فهارس التقسيم؛ لـ 1 <= i <= N، تم تبديل الصف i من
!>          المصفوفة مع الصف IPIV(i).
!>

JPIV

!>          JPIV هي مصفوفة أعداد صحيحة (INTEGER)، أبعادها (N).
!>          فهارس التقسيم؛ لـ 1 <= j <= N، تم تبديل العمود j من
!>          المصفوفة مع العمود JPIV(j).
!>

SCALE

!>          SCALE هو عدد حقيقي (REAL)
!>           عند الخروج، يحتوي SCALE على معامل القياس. تم اختيار SCALE
!>           0 <= SCALE <= 1 لمنع تجاوز سعة القيم في الحل.
!>
