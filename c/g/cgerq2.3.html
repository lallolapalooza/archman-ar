subroutine cgerq2 (integer m, integer n, complex, dimension( lda, * ) a, integer lda, complex, dimension( * ) tau, complex, dimension( * ) work, integer info)

CGERQ2 يحسب تحليل RQ لمصفوفة مستطيلة عامة باستخدام خوارزمية غير محظورة.

الغرض:

!>
!> CGERQ2 يحسب تحليل RQ لمصفوفة مركبة m × n A:
!> A = R * Q.
!>

المعلمات

M 
!>          M هو عدد صحيح (INTEGER)
!>          عدد الصفوف في المصفوفة A.  M >= 0.
!>

N

!>          N هو عدد صحيح (INTEGER)
!>          عدد الأعمدة في المصفوفة A.  N >= 0.
!>

A

!>          A هي مصفوفة مركبة (COMPLEX)، البعد (LDA,N)
!>          عند الإدخال، المصفوفة A بحجم m × n.
!>          عند الخروج، إذا كانت m <= n، فإن الجزء العلوي المثلث من المصفوفة الفرعية
!>          A(1:m,n-m+1:n) يحتوي على المصفوفة المثلثية العليا R بحجم m × m؛
!>          إذا كانت m >= n، فإن العناصر على وتحت القطر الفرعي (m-n)
!>          تحتوي على المصفوفة شبه المنحرفة العليا R بحجم m × n؛ العناصر المتبقية،
!>          مع المصفوفة TAU، تمثل المصفوفة الوحدوية
!>          Q كناتج لمُنعكسات ابتدائية (انظر التفاصيل الإضافية).
!>

LDA

!>          LDA هو عدد صحيح (INTEGER)
!>          البعد الرائد للمصفوفة A.  LDA >= max(1,M).
!>

TAU

!>          TAU هي مصفوفة مركبة (COMPLEX)، البعد (*)
!>          العوامل القياسية للمنعكسات الابتدائية (انظر التفاصيل الإضافية).
!>

WORK

!>          WORK هي مصفوفة مركبة (COMPLEX)، البعد (M)
!>

INFO

!>          INFO هو عدد صحيح (INTEGER)
!>          = 0: خروج ناجح
!>          < 0: إذا كان INFO = -i، فإن المعامل i كان لديه قيمة غير قانونية
!>

التفاصيل الإضافية:

!>
!>  يتم تمثيل المصفوفة Q كناتج لمُنعكسات ابتدائية
!>
!>     Q = H(1)**H H(2)**H . . . H(k)**H, حيث k = min(m,n).
!>
!>  كل H(i) لها الشكل
!>
!>     H(i) = I - tau * v * v**H
!>
!>  حيث tau هو عامل قياسي مركب، و v هو متجه مركب مع
!>  v(n-k+i+1:n) = 0 و v(n-k+i) = 1؛ conjg(v(1:n-k+i-1)) مخزن عند
!>  الخروج في A(m-k+i,1:n-k+i-1)، و tau في TAU(i).
!>
