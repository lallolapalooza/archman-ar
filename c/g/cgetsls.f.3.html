توثيق الوظيفة/الإجراء الفرعي

subroutine cgetsls (character trans, integer m, integer n, integer nrhs, complex, dimension( lda, * ) a, integer lda, complex, dimension( ldb, * ) b, integer ldb, complex, dimension( * ) work, integer lwork, integer info)

CGETSLS

الغرض:

!>
!> CGETSLS يحل أنظمة الخطية المعقدة، إما زائدة أو ناقصة التحديد،
!> التي تتضمن مصفوفة A بحجم M-by-N، باستخدام تحليل QR للنظام الطويل أو تحليل LQ للنظام القصير العريض لـ A.
!>
!> يفترض أن A لها رتبة كاملة، ويتم توفير حماية بدائية فقط ضد المصفوفات ذات الرتبة الناقصة.
!> هذه الوظيفة الفرعية تكتشف فقط النقص التام في الرتبة، حيث يكون عنصر قطري من العامل المثلثي لـ A صفرًا تمامًا.
!>
!> من الممكن أن يكون أحد (أو أكثر) من العناصر القطرية للعامل المثلثي لـ A أرقامًا صغيرة جدًا دون أن تشير هذه الوظيفة الفرعية إلى خطأ.
!> قد تكون الحلول المحسوبة لهذه المصفوفات شبه عديمة الرتبة أقل دقة بسبب فقدان الدقة العددية.
!>
!>
!> الخيارات التالية متاحة:
!>
!> 1. إذا كان TRANS = 'N' و m >= n: البحث عن الحل بأقل مربع لنظام زائد، أي حل مشكلة أقل مربع
!>    minimize || B - A*X ||.
!>
!> 2. إذا كان TRANS = 'N' و m < n: البحث عن الحل بأقل معيار لنظام ناقص التحديد A * X = B.
!>
!> 3. إذا كان TRANS = 'C' و m >= n: البحث عن الحل بأقل معيار لنظام غير محدد A**T * X = B.
!>
!> 4. إذا كان TRANS = 'C' و m < n: البحث عن الحل بأقل مربع لنظام زائد، أي حل مشكلة أقل مربع
!>    minimize || B - A**T * X ||.
!>
!> يمكن معالجة عدة متجهات للجانب الأيمن b ومتجهات حل x في استدعاء واحد؛ يتم تخزينها كأعمدة
!> للمصفوفة B للجانب الأيمن و X لمصفوفة الحل.
!>

المعلمات

TRANS 
!>          TRANS is CHARACTER*1
!>          = 'N': النظام الخطي يتضمن A;
!>          = 'C': النظام الخطي يتضمن A**H.
!>

M

!>          M is INTEGER
!>          عدد الصفوف في المصفوفة A.  M >= 0.
!>

N

!>          N is INTEGER
!>          عدد الأعمدة في المصفوفة A.  N >= 0.
!>

NRHS

!>          NRHS is INTEGER
!>          عدد الجوانب اليمنى، أي عدد أعمدة المصفوفات B و X. NRHS >=0.
!>

A

!>          A is COMPLEX array, dimension (LDA,N)
!>          عند الإدخال، المصفوفة A بحجم M-by-N.
!>          عند الخروج،
!>          يتم استبدال A بتفاصيل تحليل QR أو LQ الخاص بها
!>          كما تم إرجاعه بواسطة CGEQR أو CGELQ.
!>

LDA

!>          LDA is INTEGER
!>          البعد الرئيسي للمصفوفة A.  LDA >= max(1,M).
!>

B

!>          B is COMPLEX array, dimension (LDB,NRHS)
!>          عند الإدخال، مصفوفة متجهات الجانب الأيمن، مخزنة
!>          بالأعمدة؛ B بحجم M-by-NRHS إذا كان TRANS = 'N'، أو N-by-NRHS
!>          إذا كان TRANS = 'C'.
!>          عند الخروج، إذا كان INFO = 0، يتم استبدال B بمتجهات الحل،
!>          مخزنة بالأعمدة:
!>          إذا كان TRANS = 'N' و m >= n، الصفوف من 1 إلى n في B تحتوي على متجهات حل أقل مربع.
!>          إذا كان TRANS = 'N' و m < n، الصفوف من 1 إلى N في B تحتوي على
!>          متجهات الحل بأقل معيار؛
!>          إذا كان TRANS = 'C' و m >= n، الصفوف من 1 إلى M في B تحتوي على
!>          متجهات الحل بأقل معيار؛
!>          إذا كان TRANS = 'C' و m < n، الصفوف من 1 إلى M في B تحتوي على
!>          متجهات حل أقل مربع.
!>

LDB

!>          LDB is INTEGER
!>          البعد الرئيسي للمصفوفة B. LDB >= MAX(1,M,N).
!>

WORK

!>          (مساحة عمل) COMPLEX array, dimension (MAX(1,LWORK))
!>          عند الخروج، إذا كان INFO = 0، فإن WORK(1) يحتوي على حجم LWORK الأمثل (أو الحد الأدنى أو الأمثل، إذا افترضنا استعلامًا).
!>          راجع LWORK للتفاصيل.
!>

LWORK

!>          LWORK is INTEGER
!>          بعد المصفوفة WORK. LWORK >= 1.
!>          إذا كان LWORK = -1 أو -2، فسيتم افتراض استعلام مساحة عمل.
!>          إذا كان LWORK = -1، تقوم الوظيفة الفرعية بحساب الحجم الأمثل لـ WORK
!>          للأداء الأمثل وترجع هذا الحجم في WORK(1).
!>          إذا كان LWORK = -2، تقوم الوظيفة الفرعية بحساب الحجم الأدنى لـ WORK
!>          وترجع هذا الحجم في WORK(1).
!>

INFO

!>          INFO is INTEGER
!>          = 0:  انتهاء ناجح
!>          < 0:  إذا كان INFO = -i، فإن الوسيط i كان له قيمة غير قانونية
!>          > 0:  إذا كان INFO =  i، فإن العنصر القطري i من
!>                العامل المثلثي لـ A هو صفر تمامًا، بحيث لا تمتلك A
!>                رتبة كاملة؛ لم يكن من الممكن حساب حل أقل مربع.
!>
