توثيق الوظيفة/الإجراء الفرعي

subroutine cggglm (integer n, integer m, integer p, complex, dimension( lda, * ) a, integer lda, complex, dimension( ldb, * ) b, integer ldb, complex, dimension( * ) d, complex, dimension( * ) x, complex, dimension( * ) y, complex, dimension( * ) work, integer lwork, integer info)

CGGGLM

الغرض:

!>
!> CGGGLM يحل مشكلة نموذج غاوس-ماركوف الخطي العام (GLM):
!>
!>         minimize || y ||_2   subject to   d = A*x + B*y
!>             x
!>
!> حيث A مصفوفة N×M، و B مصفوفة N×P، و d متجه N معطى. يفترض أن M <= N <= M+P، و
!>
!>            rank(A) = M    and    rank( A B ) = N.
!>
!> تحت هذه الافتراضات، تكون المعادلة المقيدة متسقة دائمًا، ويوجد حل فريد x وحل بأقل معيار 2 y، والذي يتم الحصول عليه باستخدام عامل QR المعمم للمصفوفات (A, B) المعطاة بواسطة
!>
!>    A = Q*(R),   B = Q*T*Z.
!>          (0)
!>
!> على وجه الخصوص، إذا كانت المصفوفة B مربعة وغير مفردة، فإن مشكلة
!> GLM تعادل مشكلة المربعات الصغرى الخطية الموزونة التالية
!>
!>              minimize || inv(B)*(d-A*x) ||_2
!>                  x
!>
!> حيث تشير inv(B) إلى معكوس B.
!>
!> يجب أن يلاحظ المتصلون بهذا الإجراء الفرعي أن فحوصات التفرد/النقص الرتبة
!> المطبقة في هذا الإجراء الفرعي هي بدائية. الإجراء الفرعي CTRTRS الذي يستدعيه هذا
!> الإجراء الفرعي يشير فقط إلى فشل بسبب التفرد إذا كانت المشكلة مفردة تمامًا.
!>
!> من الممكن أن يكون أحد العوامل (أو أكثر) المشاركة في عامل QR المعمم
!> للمزدوج (A, B) قريبًا بشكل غير طبيعي من التفرد دون أن يشير هذا
!> الإجراء الفرعي إلى خطأ. قد تكون الحلول المحسوبة لمثل هذه المشاكل غير المكتملة الرتبة
!> أقل دقة بسبب فقدان الدقة الرقمية.
!> 
!>

المعلمات

N 
!>          N is INTEGER
!>          عدد الصفوف للمصفوفتين A و B.  N >= 0.
!>

M

!>          M is INTEGER
!>          عدد الأعمدة للمصفوفة A.  0 <= M <= N.
!>

P

!>          P is INTEGER
!>          عدد الأعمدة للمصفوفة B.  P >= N-M.
!>

A

!>          A is COMPLEX array, dimension (LDA,M)
!>          عند الدخول، المصفوفة A بحجم N×M.
!>          عند الخروج، الجزء العلوي المثلثي من المصفوفة A يحتوي على
!>          المصفوفة المثلثية العلوية R بحجم M×M.
!>

LDA

!>          LDA is INTEGER
!>          البعد الرئيسي للمصفوفة A. LDA >= max(1,N).
!>

B

!>          B is COMPLEX array, dimension (LDB,P)
!>          عند الدخول، المصفوفة B بحجم N×P.
!>          عند الخروج، إذا كان N <= P، فإن الجزء العلوي المثلثي من المصفوفة الفرعية
!>          B(1:N,P-N+1:P) يحتوي على المصفوفة المثلثية العلوية T بحجم N×N؛
!>          إذا كان N > P، فإن العناصر على وفوق القطر الفرعي (N-M)
!>          تحتوي على المصفوفة الشبه منحرفة العلوية T بحجم N×P.
!>

LDB

!>          LDB is INTEGER
!>          البعد الرئيسي للمصفوفة B. LDB >= max(1,N).
!>

D

!>          D is COMPLEX array, dimension (N)
!>          عند الدخول، D هو الطرف الأيسر لمعادلة GLM.
!>          عند الخروج، يتم تدمير D.
!>

X

!>          X is COMPLEX array, dimension (M)
!>

Y

!>          Y is COMPLEX array, dimension (P)
!>
!>          عند الخروج، X و Y هما حلا مشكلة GLM.
!>

WORK

!>          WORK is COMPLEX array, dimension (MAX(1,LWORK))
!>          عند الخروج، إذا كان INFO = 0، فإن WORK(1) يعيد LWORK الأمثل.
!>

LWORK

!>          LWORK is INTEGER
!>          بعد المصفوفة WORK. LWORK >= max(1,N+M+P).
!>          للحصول على أفضل أداء، LWORK >= M+min(N,P)+max(N,P)*NB,
!>          حيث NB هو حد أعلى لأحجام الكتل المثلى لـ
!>          CGEQRF, CGERQF, CUNMQR و CUNMRQ.
!>
!>          إذا كان LWORK = -1، فيفترض وجود استعلام مساحة عمل؛ الإجراء الفرعي
!>          يحسب فقط الحجم الأمثل للمصفوفة WORK، ويعيد
!>          هذه القيمة كأول مدخل للمصفوفة WORK، ولا يصدر الإجراء XERBLA أي رسالة خطأ
!>          متعلقة بـ LWORK.
!>

INFO

!>          INFO is INTEGER
!>          = 0:  انتهاء ناجح.
!>          < 0:  إذا كان INFO = -i، فقد كان الوسيط i غير قانوني.
!>          = 1:  العامل المثلثي العلوي R المرتبط بـ A في
!>                عامل QR المعمم للمزدوج (A, B) مفرد تمامًا، بحيث rank(A) < M؛ لم يتمكن من حساب
!>                حل المربعات الصغرى.
!>          = 2:  الجزء السفلي بحجم (N-M)×(N-M) من العامل شبه المنحرف العلوي
!>                T المرتبط بـ B في عامل QR المعمم
!>                للمزدوج (A, B) مفرد تمامًا، بحيث rank( A B ) < N؛ لم يتمكن من حساب
!>                حل المربعات الصغرى.
!>
