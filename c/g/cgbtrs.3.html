توثيق الوظيفة/الإجراء الفرعي

subroutine cgbtrs (character trans, integer n, integer kl, integer ku, integer nrhs, complex, dimension( ldab, * ) ab, integer ldab, integer, dimension( * ) ipiv, complex, dimension( ldb, * ) b, integer ldb, integer info)

CGBTRS

الغرض:

!>
!> CGBTRS تحل نظام المعادلات الخطية
!>    A * X = B,  A**T * X = B,  أو  A**H * X = B
!> مع مصفوفة شريطية عامة A باستخدام التحليل LU الذي تم حسابه
!> بواسطة CGBTRF.
!>

المعلمات

TRANS 
!>          TRANS هي CHARACTER*1
!>          تحدد شكل نظام المعادلات.
!>          = 'N':  A * X = B     (بدون تبديل)
!>          = 'T':  A**T * X = B  (تبديل)
!>          = 'C':  A**H * X = B  (تبديل مترافق)
!>

N

!>          N هي INTEGER
!>          رتبة المصفوفة A.  N >= 0.
!>

KL

!>          KL هي INTEGER
!>          عدد الأقطار الفرعية داخل الشريط A.  KL >= 0.
!>

KU

!>          KU هي INTEGER
!>          عدد الأقطار الفائقة داخل الشريط A.  KU >= 0.
!>

NRHS

!>          NRHS هي INTEGER
!>          عدد الأطراف اليمنى، أي عدد أعمدة
!>          المصفوفة B.  NRHS >= 0.
!>

AB

!>          AB هي COMPLEX array, dimension (LDAB,N)
!>          تفاصيل التحليل LU للمصفوفة الشريطية A، كما
!>          تم حسابها بواسطة CGBTRF.  U مخزنة كمصفوفة شريطية عليا
!>          مثلثة مع KL+KU أقطار فائقة في الصفوف 1 إلى KL+KU+1، و
!>          المضاعفات المستخدمة أثناء التحليل مخزنة في
!>          الصفوف KL+KU+2 إلى 2*KL+KU+1.
!>

LDAB

!>          LDAB هي INTEGER
!>          البعد الرئيسي للمصفوفة AB.  LDAB >= 2*KL+KU+1.
!>

IPIV

!>          IPIV هي INTEGER array, dimension (N)
!>          مؤشرات المحاور؛ لـ 1 <= i <= N، تم تبديل الصف i للمصفوفة
!>          مع الصف IPIV(i).
!>

B

!>          B هي COMPLEX array, dimension (LDB,NRHS)
!>          عند الدخول، مصفوفة الطرف الأيمن B.
!>          عند الخروج، مصفوفة الحل X.
!>

LDB

!>          LDB هي INTEGER
!>          البعد الرئيسي للمصفوفة B.  LDB >= max(1,N).
!>

INFO

!>          INFO هي INTEGER
!>          = 0:  انتهاء ناجح
!>          < 0:  إذا كان INFO = -i، فإن المعامل i له قيمة غير قانونية
!>
