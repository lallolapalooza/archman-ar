subroutine cgqrts (integer n, integer m, integer p, complex, dimension( lda, * ) a, complex, dimension( lda, * ) af, complex, dimension( lda, * ) q, complex, dimension( lda, * ) r, integer lda, complex, dimension( * ) taua, complex, dimension( ldb, * ) b, complex, dimension( ldb, * ) bf, complex, dimension( ldb, * ) z, complex, dimension( ldb, * ) t, complex, dimension( ldb, * )bwk, integer ldb, complex, dimension( * ) taub, complex, dimension( lwork ) work, integer lwork, real, dimension( * ) rwork, real, dimension( 4 ) result)

CGQRTS

الغرض:

!>
!> CGQRTS تختبر CGGQRF، التي تحسب تحليل GQR للمصفوفة N-by-M A والمصفوفة N-by-P B: A = Q*R و B = Q*T*Z.
!>

المعلمات

N 
!>          N هو عدد صحيح
!>          عدد صفوف المصفوفات A و B. N >= 0.
!>

M

!>          M هو عدد صحيح
!>          عدد أعمدة المصفوفة A. M >= 0.
!>

P

!>          P هو عدد صحيح
!>          عدد أعمدة المصفوفة B. P >= 0.
!>

A

!>          A هي مصفوفة COMPLEX، البعد (LDA,M)
!>          المصفوفة A ذات الأبعاد N-by-M.
!>

AF

!>          AF هي مصفوفة COMPLEX، البعد (LDA,N)
!>          تفاصيل تحليل GQR للمصفوفتين A و B، كما تم إرجاعها
!>          بواسطة CGGQRF، انظر CGGQRF لمزيد من التفاصيل.
!>

Q

!>          Q هي مصفوفة COMPLEX، البعد (LDA,N)
!>          المصفوفة الوحدة Q ذات الأبعاد M-by-M.
!>

R

!>          R هي مصفوفة COMPLEX، البعد (LDA,MAX(M,N))
!>

LDA

!>          LDA هو عدد صحيح
!>          البعد الرئيسي للمصفوفات A، AF، R و Q.
!>          LDA >= max(M,N).
!>

TAUA

!>          TAUA هي مصفوفة COMPLEX، البعد (min(M,N))
!>          العوامل العددية للمنعكسات الأولية، كما تم إرجاعها
!>          بواسطة CGGQRF.
!>

B

!>          B هي مصفوفة COMPLEX، البعد (LDB,P)
!>          عند الإدخال، المصفوفة A ذات الأبعاد N-by-P.
!>

BF

!>          BF هي مصفوفة COMPLEX، البعد (LDB,N)
!>          تفاصيل تحليل GQR للمصفوفتين A و B، كما تم إرجاعها
!>          بواسطة CGGQRF، انظر CGGQRF لمزيد من التفاصيل.
!>

Z

!>          Z هي مصفوفة COMPLEX، البعد (LDB,P)
!>          المصفوفة الوحدة Z ذات الأبعاد P-by-P.
!>

T

!>          T هي مصفوفة COMPLEX، البعد (LDB,max(P,N))
!>

BWK

!>          BWK هي مصفوفة COMPLEX، البعد (LDB,N)
!>

LDB

!>          LDB هو عدد صحيح
!>          البعد الرئيسي للمصفوفات B، BF، Z و T.
!>          LDB >= max(P,N).
!>

TAUB

!>          TAUB هي مصفوفة COMPLEX، البعد (min(P,N))
!>          العوامل العددية للمنعكسات الأولية، كما تم إرجاعها
!>          بواسطة SGGRQF.
!>

WORK

!>          WORK هي مصفوفة COMPLEX، البعد (LWORK)
!>

LWORK

!>          LWORK هو عدد صحيح
!>          بعد المصفوفة WORK، LWORK >= max(N,M,P)**2.
!>

RWORK

!>          RWORK هي مصفوفة REAL، البعد (max(N,M,P))
!>

RESULT

!>          RESULT هي مصفوفة REAL، البعد (4)
!>          نسب الاختبار:
!>            RESULT(1) = norm( R - Q'*A ) / ( MAX(M,N)*norm(A)*ULP)
!>            RESULT(2) = norm( T*Z - Q'*B ) / (MAX(P,N)*norm(B)*ULP)
!>            RESULT(3) = norm( I - Q'*Q ) / ( M*ULP )
!>            RESULT(4) = norm( I - Z'*Z ) / ( P*ULP )
!>
