subroutine cgetsqrhrt (integer m, integer n, integer mb1, integer nb1, integer nb2, complex, dimension( lda, * ) a, integer lda, complex, dimension( ldt, * ) t, integer ldt, complex, dimension( * ) work, integer lwork, integer info)

CGETSQRHRT

الغرض:

!>
!> CGETSQRHRT يحسب تحليل QR بحجم NB2 من الأعمدة لمصفوفة A المعقدة M-by-N حيث M >= N،
!>
!>    A = Q * R.
!>
!> تستخدم الروتين داخليًا تحليل TSQR محظور بحجم NB1 من الأعمدة ومحظور بحجم MB1 من الصفوف
!> وتقوم بإعادة بناء متجهات هاوسهولدر من مخرجات TSQR. يقوم الروتين أيضًا
!> بتحويل عامل R_tsqr من مخرجات تحليل TSQR إلى
!> العامل R الذي يتوافق مع تحليل QR بواسطة هاوسهولدر،
!>
!>    A = Q_tsqr * R_tsqr = Q * R.
!>
!> يتم تخزين عاملي Q و R الناتجين بنفس تنسيق CGEQRT
!> (Q في تمثيل WY المدمج المحظور). انظر توثيق
!> CGEQRT لمزيد من التفاصيل حول التنسيق.
!>

المعلمات

M 
!>          M هو عدد صحيح (INTEGER)
!>          عدد صفوف المصفوفة A. M >= 0.
!>

N

!>          N هو عدد صحيح (INTEGER)
!>          عدد أعمدة المصفوفة A. M >= N >= 0.
!>

MB1

!>          MB1 هو عدد صحيح (INTEGER)
!>          حجم كتلة الصفوف المراد استخدامها في TSQR المحظور.
!>          MB1 > N.
!>

NB1

!>          NB1 هو عدد صحيح (INTEGER)
!>          حجم كتلة الأعمدة المراد استخدامها في TSQR المحظور.
!>          N >= NB1 >= 1.
!>

NB2

!>          NB2 هو عدد صحيح (INTEGER)
!>          الحجم الذي سيتم استخدامه في QR المحظور الذي
!>          هو المخرج. NB2 >= 1.
!>

A

!>          A هي مصفوفة COMPLEX*16، الأبعاد (LDA,N)
!>
!>          عند الإدخال: مصفوفة A بحجم M-by-N.
!>
!>          عند الخروج:
!>           أ) العناصر الموجودة على القطر وفي أعلاه
!>              للمصفوفة تحتوي على المصفوفة شبه المثلثية العليا N-by-N
!>              R المقابلة لتحليل QR بواسطة هاوسهولدر؛
!>           ب) العناصر الموجودة أسفل القطر تمثل Q بواسطة
!>              أعمدة V المحظورة (تمثيل WY المدمج).
!>

LDA

!>          LDA هو عدد صحيح (INTEGER)
!>          البعد الرائد للمصفوفة A. LDA >= max(1,M).
!>

T

!>          T هي مصفوفة COMPLEX، الأبعاد (LDT,N))
!>          المحددات المثلثية العليا المحظورة المخزنة بشكل مدمج
!>          كسلسلة من الكتل المثلثية العليا.
!>

LDT

!>          LDT هو عدد صحيح (INTEGER)
!>          البعد الرائد للمصفوفة T. LDT >= NB2.
!>

WORK

!>          (مساحة عمل) مصفوفة COMPLEX، الأبعاد (MAX(1,LWORK))
!>          عند الخروج، إذا كان INFO = 0، فإن WORK(1) يعيد LWORK المثلى.
!>

LWORK

!>          LWORK هو عدد صحيح (INTEGER)
!>          بعد المصفوفة WORK.
!>          إذا كان MIN(M,N) = 0، فإن LWORK >= 1، وإلا
!>          LWORK >= MAX( 1, LWT + LW1, MAX( LWT+N*N+LW2, LWT+N*N+N ) )،
!>          حيث
!>             NUM_ALL_ROW_BLOCKS = CEIL((M-N)/(MB1-N))،
!>             NB1LOCAL = MIN(NB1,N).
!>             LWT = NUM_ALL_ROW_BLOCKS * N * NB1LOCAL،
!>             LW1 = NB1LOCAL * N،
!>             LW2 = NB1LOCAL * MAX( NB1LOCAL, ( N - NB1LOCAL ) ).
!>
!>          إذا كان LWORK = -1، فمن المفترض وجود استعلام مساحة عمل.
!>          يقوم الروتين بحساب حجم مساحة العمل WORK المثلى فقط،
!>          ويعيد هذه القيمة كالإدخال الأول لمصفوفة WORK،
!>          ولا يتم إصدار أي رسالة خطأ متعلقة بـ LWORK
!>          بواسطة XERBLA.
!>

INFO

!>          INFO هو عدد صحيح (INTEGER)
!>          = 0:  خروج ناجح
!>          < 0:  إذا كان INFO = -i، فإن الوسيط i كان له قيمة غير قانونية
!>
