subroutine cgeqrt2 (integer m, integer n, complex, dimension( lda, * ) a, integer lda, complex, dimension( ldt, * ) t, integer ldt, integer info)

CGEQRT2 تحسب تحليل QR لمصفوفة عامة حقيقية أو معقدة باستخدام تمثيل WY المدمج لـ Q.

الغرض:

!>
!> CGEQRT2 تحسب تحليل QR لمصفوفة معقدة M-by-N A،
!> باستخدام تمثيل WY المدمج لـ Q.
!>

المعلمات

M 
!>          M هو عدد صحيح (INTEGER)
!>          عدد صفوف المصفوفة A.  M >= N.
!>

N

!>          N هو عدد صحيح (INTEGER)
!>          عدد أعمدة المصفوفة A.  N >= 0.
!>

A

!>          A هي مصفوفة معقدة (COMPLEX)، البعد (LDA,N)
!>          عند الإدخال، المصفوفة المعقدة M-by-N A. عند الخروج، العناصر على
!>          القطر وفوقه تحتوي على المصفوفة المثلثية العليا N-by-N R؛
!>          العناصر تحت القطر هي أعمدة V. انظر أدناه لمزيد من التفاصيل.
!>

LDA

!>          LDA هو عدد صحيح (INTEGER)
!>          البعد الرئيسي للمصفوفة A.  LDA >= max(1,M).
!>

T

!>          T هي مصفوفة معقدة (COMPLEX)، البعد (LDT,N)
!>          المصفوفة المثلثية العليا N-by-N لعامل الانعكاس الكتلي.
!>          العناصر على القطر وفوقه تحتوي على عامل الانعكاس الكتلي
!>          T؛ العناصر تحت القطر لا تستخدم.
!>          انظر أدناه لمزيد من التفاصيل.
!>

LDT

!>          LDT هو عدد صحيح (INTEGER)
!>          البعد الرئيسي للمصفوفة T.  LDT >= max(1,N).
!>

INFO

!>          INFO هو عدد صحيح (INTEGER)
!>          = 0: خروج ناجح
!>          < 0: إذا كان INFO = -i، فقد كان المعامل i غير قانوني
!>

تفاصيل إضافية:

!>
!>  المصفوفة V تخزن الانعكاسات الأولية H(i) في العمود i
!>  تحت القطر. على سبيل المثال، إذا كان M=5 و N=3، فإن المصفوفة V هي
!>
!>               V = (  1       )
!>                   ( v1  1    )
!>                   ( v1 v2  1 )
!>                   ( v1 v2 v3 )
!>                   ( v1 v2 v3 )
!>
!>  حيث تمثل الـ vi's المتجهات التي تحدد H(i)، والتي يتم إرجاعها
!>  في المصفوفة A.  الـ 1s على طول قطر V لا يتم تخزينها في A.  عامل الانعكاس الكتلي H
!>  يعطى بعد ذلك بواسطة
!>
!>               H = I - V * T * V**H
!>
!>  حيث V**H هو المرافق المنقول لـ V.
!>
