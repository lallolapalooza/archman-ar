subroutine cgeqrf (integer m, integer n, complex, dimension( lda, * ) a, integer lda, complex, dimension( * ) tau, complex, dimension( * ) work, integer lwork, integer info)

CGEQRF

الغرض:

!>
!> CGEQRF يحسب تحليل QR لمصفوفة معقدة M-by-N وهي A:
!>
!>    A = Q * ( R ),
!>            ( 0 )
!>
!> حيث:
!>
!>    Q هي مصفوفة متعامدة M-by-M؛
!>    R هي مصفوفة مثلثية عليا N-by-N؛
!>    0 هي مصفوفة صفرية (M-N)-by-N، إذا كان M > N.
!>
!>

المعاملات

M 
!>          M هو عدد صحيح (INTEGER)
!>          عدد الصفوف للمصفوفة A.  M >= 0.
!>

N

!>          N هو عدد صحيح (INTEGER)
!>          عدد الأعمدة للمصفوفة A.  N >= 0.
!>

A

!>          A هي مصفوفة معقدة (COMPLEX array)، حجمها (LDA,N)
!>          عند الإدخال، المصفوفة A بحجم M-by-N.
!>          عند الخروج، تحتوي العناصر الموجودة على القطر العلوي وفيما فوقه على
!>          المصفوفة شبه المنحرفة العلوية R ذات الحجم min(M,N)-by-N (R
!>          مثلثية عليا إذا كان m >= n)؛ العناصر الموجودة أسفل القطر،
!>          مع المصفوفة TAU، تمثل المصفوفة الوحدوية Q كحاصل ضرب
!>          min(m,n) من الانعكاسات الأولية (انظر التفاصيل الإضافية).
!>

LDA

!>          LDA هو عدد صحيح (INTEGER)
!>          البعد الرائد للمصفوفة A.  LDA >= max(1,M).
!>

TAU

!>          TAU هي مصفوفة معقدة (COMPLEX array)، حجمها (min(M,N))
!>          العوامل القياسية للانعكاسات الأولية (انظر التفاصيل الإضافية).
!>

WORK

!>          WORK هي مصفوفة معقدة (COMPLEX array)، حجمها (MAX(1,LWORK))
!>          عند الخروج، إذا كان INFO = 0، فإن WORK(1) يعيد الحجم الأمثل لـ LWORK.
!>

LWORK

!>          LWORK هو عدد صحيح (INTEGER)
!>          حجم مصفوفة WORK.
!>          LWORK >= 1، إذا كان MIN(M,N) = 0، و LWORK >= N، بخلاف ذلك.
!>          للحصول على أداء أمثل LWORK >= N*NB، حيث NB هو
!>          حجم الكتلة الأمثل.
!>
!>          إذا كان LWORK = -1، فسيتم افتراض استعلام مساحة العمل؛ تقوم الروتين
!>          فقط بحساب الحجم الأمثل لمصفوفة WORK، وتعيد
!>          هذه القيمة كالإدخال الأول لمصفوفة WORK، ولا يصدر XERBLA أي رسالة خطأ
!>          متعلقة بـ LWORK.
!>

INFO

!>          INFO هو عدد صحيح (INTEGER)
!>          = 0:  انتهاء ناجح
!>          < 0:  إذا كان INFO = -i، فإن المعامل i كان له قيمة غير قانونية
!>

التفاصيل الإضافية:

!>
!>  يتم تمثيل المصفوفة Q كحاصل ضرب للانعكاسات الأولية
!>
!>     Q = H(1) H(2) . . . H(k)، حيث k = min(m,n).
!>
!>  كل H(i) لها الشكل
!>
!>     H(i) = I - tau * v * v**H
!>
!>  حيث tau هو عامل قياسي معقد، و v هو متجه معقد مع
!>  v(1:i-1) = 0 و v(i) = 1؛ يتم تخزين v(i+1:m) عند الخروج في A(i+1:m,i)،
!>  و tau في TAU(i).
!>
