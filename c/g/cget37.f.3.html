subroutine cget37 (real, dimension( 3 ) rmax, integer, dimension( 3 ) lmax, integer, dimension( 3 ) ninfo, integer knt, integer nin)

CGET37

الغرض:

!>
!> CGET37 تختبر CTRSNA، إجراء فرعي لتقدير أرقام الشروط للقيم الذاتية و/أو المتجهات الذاتية اليمنى لمصفوفة.
!>
!> يتم قراءة مصفوفات الاختبار من ملف برقم وحدة منطقي NIN.
!>

المعلمات

RMAX 
!>          RMAX هي مصفوفة حقيقية، بأبعاد (3)
!>          قيمة أكبر نسبة اختبار.
!>          RMAX(1) = أكبر نسبة تقارن استدعاءات مختلفة لـ CTRSNA
!>          RMAX(2) = أكبر خطأ في مقلوب أرقام الشروط
!>                    مع الأخذ في الاعتبار شروطها
!>          RMAX(3) = أكبر خطأ في مقلوب أرقام الشروط
!>                    دون الأخذ في الاعتبار شروطها (قد تكون أكبر من RMAX(2))
!>

LMAX

!>          LMAX هي مصفوفة صحيحة، بأبعاد (3)
!>          LMAX(i) هو رقم المثال حيث يتم تحقيق أكبر نسبة اختبار
!>          RMAX(i). أيضًا:
!>          إذا أعادت CGEHRD قيمة INFO غير صفرية في المثال i، فإن LMAX(1)=i
!>          إذا أعادت CHSEQR قيمة INFO غير صفرية في المثال i، فإن LMAX(2)=i
!>          إذا أعادت CTRSNA قيمة INFO غير صفرية في المثال i، فإن LMAX(3)=i
!>

NINFO

!>          NINFO هي مصفوفة صحيحة، بأبعاد (3)
!>          NINFO(1) = عدد المرات التي أعادت فيها CGEHRD قيمة INFO غير صفرية
!>          NINFO(2) = عدد المرات التي أعادت فيها CHSEQR قيمة INFO غير صفرية
!>          NINFO(3) = عدد المرات التي أعادت فيها CTRSNA قيمة INFO غير صفرية
!>

KNT

!>          KNT هو عدد صحيح
!>          إجمالي عدد الأمثلة التي تم اختبارها.
!>

NIN

!>          NIN هو عدد صحيح
!>          رقم الوحدة المنطقية للإدخال
!>
