subroutine cgetf2 (m, n, a, lda, ipiv, info)
CGETF2 يحسب عامل LU لمصفوفة عامة m-by-n باستخدام التقسيم الجزئي مع تبديلات الصفوف (خوارزمية غير مجمعة).

توثيق الوظيفة/الروتين الفرعي

subroutine cgetf2 (integer m, integer n, complex, dimension( lda, * ) a, integer lda, integer, dimension( * ) ipiv, integer info)

CGETF2 يحسب عامل LU لمصفوفة عامة m-by-n باستخدام التقسيم الجزئي مع تبديلات الصفوف (خوارزمية غير مجمعة).

الغرض:

!>
!> CGETF2 يحسب عامل LU لمصفوفة عامة m-by-n A
!> باستخدام التقسيم الجزئي مع تبديلات الصفوف.
!>
!> العامل له الشكل
!>    A = P * L * U
!> حيث P مصفوفة تبديل، L مصفوفة مثلثة سفلية بعناصر قطرية موحدة (شبه منحرفة سفلية إذا كان m > n)، و U مصفوفة مثلثة علوية (شبه منحرفة علوية إذا كان m < n).
!>
!> هذه هي النسخة اليمنى المستوى 2 BLAS من الخوارزمية.
!>

المعلمات

M 
!>          M هو عدد صحيح (INTEGER)
!>          عدد صفوف المصفوفة A. M >= 0.
!>

N

!>          N هو عدد صحيح (INTEGER)
!>          عدد أعمدة المصفوفة A. N >= 0.
!>

A

!>          A هي مصفوفة مركبة (COMPLEX array)، ذات بعد (LDA,N)
!>          عند الدخول، المصفوفة m × n المراد تحليلها.
!>          عند الخروج، عوامل L و U من التحليل
!>          A = P*L*U؛ لا يتم تخزين عناصر القطر الموحدة لـ L.
!>

LDA

!>          LDA هو عدد صحيح (INTEGER)
!>          البعد الرائد للمصفوفة A. LDA >= max(1,M).
!>

IPIV

!>          IPIV هي مصفوفة أعداد صحيحة (INTEGER array)، ذات بعد (*)
!>          مؤشرات التقسيم؛ من أجل 1 <= i <= min(M,N)، تم تبديل الصف i من
!>          المصفوفة مع الصف IPIV(i).
!>

INFO

!>          INFO هو عدد صحيح (INTEGER)
!>          = 0: خروج ناجح
!>          < 0: إذا كان INFO = -k، كان الوسيط k غير قانوني
!>          > 0: إذا كان INFO = k، فإن U(k,k) يساوي صفرًا تمامًا. تم
!>               إكمال التحليل، ولكن العامل U هو بالضبط
!>               مصفوفة مفردة، وسيحدث قسمة على صفر إذا تم استخدامها
!>               لحل نظام معادلات.
!>
