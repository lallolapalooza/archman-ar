subroutine cget01 (integer m, integer n, complex, dimension( lda, * ) a, integer lda, complex, dimension( ldafac, * ) afac, integer ldafac, integer, dimension( * ) ipiv, real, dimension( * ) rwork, real resid)

CGET01

الغرض:

!>
!> CGET01 يعيد بناء مصفوفة A من تحليلها L*U ويحسب الباقي
!>    norm(L*U - A) / ( N * norm(A) * EPS )،
!> حيث EPS هو معامل آلة الحوسبة.
!>

المعاملات

M 
!>          M هو عدد صحيح
!>          عدد صفوف المصفوفة A.  M >= 0.
!>

N

!>          N هو عدد صحيح
!>          عدد أعمدة المصفوفة A.  N >= 0.
!>

A

!>          A هي مصفوفة مركبة، البعد (LDA,N)
!>          المصفوفة الأصلية M x N، A.
!>

LDA

!>          LDA هو عدد صحيح
!>          البعد الرئيسي للمصفوفة A.  LDA >= max(1,M).
!>

AFAC

!>          AFAC هي مصفوفة مركبة، البعد (LDAFAC,N)
!>          الشكل المفكك للمصفوفة A.  AFAC يحتوي على العوامل
!>          L و U من تحليل L*U كما تم حسابه بواسطة CGETRF.
!>          يتم الكتابة فوقها بالمصفوفة المعاد بناؤها، ثم بفرق
!>          L*U - A.
!>

LDAFAC

!>          LDAFAC هو عدد صحيح
!>          البعد الرئيسي للمصفوفة AFAC.  LDAFAC >= max(1,M).
!>

IPIV

!>          IPIV هي مصفوفة أعداد صحيحة، البعد (N)
!>          فهارس المحاور من CGETRF.
!>

RWORK

!>          RWORK هي مصفوفة حقيقية، البعد (M)
!>

RESID

!>          RESID هو عدد حقيقي
!>          norm(L*U - A) / ( N * norm(A) * EPS )
!>
