subroutine cgtsv (integer n, integer nrhs, complex, dimension( * ) dl, complex, dimension( * ) d, complex, dimension( * ) du, complex, dimension( ldb, * ) b, integer ldb, integer info)

CGTSV تحسب حل نظام المعادلات الخطية A * X = B لمصفوفات GT

الغرض:

!>
!> CGTSV  يحل المعادلة
!>
!>    A*X = B,
!>
!> حيث A هي مصفوفة ثلاثية قطرية N-by-N، عن طريق الحذف الغاوسي مع التقسيم الجزئي.
!>
!> لاحظ أنه يمكن حل المعادلة  A**T *X = B  عن طريق تبديل ترتيب
!> الوسيطين DU و DL.
!>

المعلمات

N 
!>          N هو عدد صحيح
!>          رتبة المصفوفة A.  N >= 0.
!>

NRHS

!>          NRHS هو عدد صحيح
!>          عدد الأطراف اليمنى، أي عدد أعمدة المصفوفة B.  NRHS >= 0.
!>

DL

!>          DL هو مصفوفة مركبة، البعد (N-1)
!>          عند الإدخال، يجب أن يحتوي DL على العناصر تحت القطرية (n-1)
!>          للمصفوفة A.
!>          عند الخروج، يتم الكتابة فوق DL بالعناصر (n-2) من
!>          القطر العلوي الثاني للمصفوفة المثلثية العليا U من
!>          تحليل LU للمصفوفة A، في DL(1)، ..., DL(n-2).
!>

D

!>          D هو مصفوفة مركبة، البعد (N)
!>          عند الإدخال، يجب أن يحتوي D على العناصر القطرية للمصفوفة A.
!>          عند الخروج، يتم الكتابة فوق D بالعناصر القطرية n للمصفوفة U.
!>

DU

!>          DU هو مصفوفة مركبة، البعد (N-1)
!>          عند الإدخال، يجب أن يحتوي DU على العناصر فوق القطرية (n-1)
!>          للمصفوفة A.
!>          عند الخروج، يتم الكتابة فوق DU بالعناصر (n-1) من
!>          القطر العلوي الأول للمصفوفة U.
!>

B

!>          B هي مصفوفة مركبة، البعد (LDB,NRHS)
!>          عند الإدخال، مصفوفة الطرف الأيمن B ذات الأبعاد N-by-NRHS.
!>          عند الخروج، إذا كان INFO = 0، فإن مصفوفة الحل X ذات الأبعاد N-by-NRHS.
!>

LDB

!>          LDB هو عدد صحيح
!>          البعد الرائد للمصفوفة B.  LDB >= max(1,N).
!>

INFO

!>          INFO هو عدد صحيح
!>          = 0:  انتهاء ناجح
!>          < 0:  إذا كان INFO = -i، فإن الوسيط i كان له قيمة غير قانونية
!>          > 0:  إذا كان INFO = i، فإن U(i,i) يساوي صفر تمامًا، ولم يتم حساب الحل.
!>                لم يكتمل التحليل إلا إذا كان i = N.
!>
