recursive subroutine cgelqt3 (m, n, a, lda, t, ldt, info)
CGELQT3

توثيق الدالة/الدُرجاء

recursive subroutine cgelqt3 (integer m, integer n, complex, dimension( lda, * ) a, integer lda, complex, dimension( ldt, * ) t, integer ldt, integer info)

CGELQT3

الغرض:

!>
!> CGELQT3 تحسب بشكل تكراري تفكيك LQ لمصفوفة معقدة M-by-N
!> A، باستخدام تمثيل WY المدمج لـ Q.
!>
!> بناءً على خوارزمية Elmroth و Gustavson،
!> IBM J. Res. Develop. Vol 44 No. 4 July 2000.
!>

المعلمات

M 
!>          M هو عدد صحيح (INTEGER)
!>          عدد صفوف المصفوفة A.  M =< N.
!>

N

!>          N هو عدد صحيح (INTEGER)
!>          عدد أعمدة المصفوفة A.  N >= 0.
!>

A

!>          A هي مصفوفة مركبة (COMPLEX)، الأبعاد (LDA,N)
!>          عند الدخول، تحتوي العناصر على القطر والقطر السفلي على المصفوفة المثلثية السفلية L المكونة من N-by-N؛
!>          العناصر فوق القطر هي صفوف V. انظر أدناه لمزيد من التفاصيل.
!>

LDA

!>          LDA هو عدد صحيح (INTEGER)
!>          البعد الرائد للمصفوفة A.  LDA >= max(1,M).
!>

T

!>          T هي مصفوفة مركبة (COMPLEX)، الأبعاد (LDT,N)
!>          المصفوفة المثلثية العلوية المكونة من N-by-N للعامل المنعكس الكتلي.
!>          العناصر على القطر وفوقه تحتوي على المنعكس الكتلي T؛
!>          العناصر أسفل القطر غير مستخدمة. انظر أدناه لمزيد من التفاصيل.
!>

LDT

!>          LDT هو عدد صحيح (INTEGER)
!>          البعد الرائد للمصفوفة T.  LDT >= max(1,N).
!>

INFO

!>          INFO هو عدد صحيح (INTEGER)
!>          = 0: خروج ناجح
!>          < 0: إذا كان INFO = -i، فإن المعلمة i كان لها قيمة غير قانونية
!>

تفاصيل إضافية:

!>
!>  تخزن المصفوفة V المنعكسات الأولية H(i) في الصف i
!>  فوق القطر. على سبيل المثال، إذا كانت M=5 و N=3، فإن المصفوفة V هي
!>
!>               V = (  1  v1 v1 v1 v1 )
!>                   (     1  v2 v2 v2 )
!>                   (     1  v3 v3 v3 )
!>
!>
!>  حيث تمثل vi's المتجهات التي تحدد H(i)، والتي يتم إرجاعها
!>  في المصفوفة A. لا يتم تخزين الـ 1s على طول قطر V في A.
!>  المنعكس الكتلي H يُعطى بعد ذلك بواسطة
!>
!>               H = I - V * T * V**T
!>
!>  حيث V**T هو منقول V.
!>
!>  للحصول على تفاصيل الخوارزمية، انظر Elmroth و Gustavson (المذكور أعلاه).
!>
