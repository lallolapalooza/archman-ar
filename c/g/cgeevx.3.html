subroutine cgeevx (character balanc, character jobvl, character jobvr, character sense, integer n, complex, dimension( lda, * ) a, integer lda, complex, dimension( * ) w, complex, dimension( ldvl, * ) vl, integer ldvl, complex, dimension( ldvr, * ) vr, integer ldvr, integer ilo, integer ihi, real, dimension( * ) scale, real abnrm, real, dimension( * ) rconde, real, dimension( * ) rcondv, complex, dimension( * ) work, integer lwork, real, dimension( * ) rwork, integer info)

CGEEVX تحسب القيم الذاتية، اختياريًا، المتجهات الذاتية اليسرى و/أو اليمنى لمصفوفات GE

الغرض:

!>
!> CGEEVX تحسب للمصفوفة المعقدة غير المتماثلة N-by-N A،
!> القيم الذاتية، اختياريًا، المتجهات الذاتية اليسرى و/أو اليمنى.
!>
!> اختياريًا أيضًا، تحسب تحويل موازنة لتحسين
!> ظروف القيم الذاتية والمتجهات الذاتية (ILO، IHI،
!> SCALE، و ABNRM)، معكوسات أرقام الشروط للقيم الذاتية
!> (RCONDE)، ومعكوسات أرقام الشروط للمتجهات الذاتية اليمنى
!> (RCONDV).
!>
!> المتجه الذاتي الأيمن v(j) للمصفوفة A يحقق
!>                  A * v(j) = lambda(j) * v(j)
!> حيث lambda(j) هي القيمة الذاتية الخاصة به.
!> المتجه الذاتي الأيسر u(j) للمصفوفة A يحقق
!>               u(j)**H * A = lambda(j) * u(j)**H
!> حيث u(j)**H يرمز إلى المترافق المستعرض لـ u(j).
!>
!> المتجهات الذاتية المحسوبة يتم تطبيعها ليكون لها معيار إقليدي
!> يساوي 1 وأكبر مركبة حقيقية.
!>
!> موازنة مصفوفة تعني تبديل الصفوف والأعمدة لجعلها
!> أقرب إلى مصفوفة مثلثية عليا، وتطبيق تحويل تشابه قطري
!> D * A * D**(-1)، حيث D مصفوفة قطرية، ل
!> جعل معايير صفوفها وأعمدتها أقرب وأرقام الشروط
!> لقيمها الذاتية والمتجهات الذاتية أصغر. تحسب
!> معكوسات أرقام الشروط للمصفوفة المتوازنة.
!> تبديل الصفوف والأعمدة لن يغير أرقام الشروط
!> (في الحساب الدقيق) ولكن التحجيم القطري سيفعل. لمزيد من
!> الشرح للموازنة، انظر القسم 4.10.2 من دليل مستخدمي LAPACK.
!>

المعلمات:

BALANC 
!>          BALANC هو CHARACTER*1
!>          يشير إلى كيفية تحجيم المصفوفة المدخلة قطريًا
!>          و/أو تبديلها لتحسين شروط قيمها الذاتية.
!>          = 'N': لا تقم بالتحجيم القطري أو التبديل؛
!>          = 'P': قم بإجراء تبديلات لجعل المصفوفة أقرب إلى
!>                 المثلثية العليا. لا تقم بالتحجيم القطري؛
!>          = 'S': قم بتحجيم المصفوفة قطريًا، أي استبدل A بـ
!>                 D*A*D**(-1)، حيث D مصفوفة قطرية مختارة
!>                 لجعل صفوف وأعمدة A متساوية في المعيار.
!>                 لا تقم بالتبديل؛
!>          = 'B': قم بتحجيم وتبديل A قطريًا.
!>
!>          ستكون معكوسات أرقام الشروط المحسوبة للمصفوفة
!>          بعد الموازنة و/أو التبديل. التبديل لا يغير
!>          أرقام الشروط (في الحساب الدقيق)، ولكن الموازنة تفعل.
!>

JOBVL

!>          JOBVL هو CHARACTER*1
!>          = 'N': المتجهات الذاتية اليسرى لـ A غير محسوبة؛
!>          = 'V': المتجهات الذاتية اليسرى لـ A محسوبة.
!>          إذا كان SENSE = 'E' أو 'B'، يجب أن يكون JOBVL = 'V'.
!>

JOBVR

!>          JOBVR هو CHARACTER*1
!>          = 'N': المتجهات الذاتية اليمنى لـ A غير محسوبة؛
!>          = 'V': المتجهات الذاتية اليمنى لـ A محسوبة.
!>          إذا كان SENSE = 'E' أو 'B'، يجب أن يكون JOBVR = 'V'.
!>

SENSE

!>          SENSE هو CHARACTER*1
!>          يحدد معكوسات أرقام الشروط التي يتم حسابها.
!>          = 'N': لا يتم حساب أي منها؛
!>          = 'E': محسوبة للقيم الذاتية فقط؛
!>          = 'V': محسوبة للمتجهات الذاتية اليمنى فقط؛
!>          = 'B': محسوبة للقيم الذاتية والمتجهات الذاتية اليمنى.
!>
!>          إذا كان SENSE = 'E' أو 'B'، يجب أيضًا حساب كل من المتجهات الذاتية
!>          اليسرى واليمنى (JOBVL = 'V' و JOBVR = 'V').
!>

N

!>          N هو INTEGER
!>          رتبة المصفوفة A. N >= 0.
!>

A

!>          A هو مصفوفة COMPLEX، البعد (LDA,N)
!>          عند الدخول، المصفوفة N-by-N A.
!>          عند الخروج، تم استبدال A. إذا كان JOBVL = 'V' أو
!>          JOBVR = 'V'، فإن A تحتوي على شكل شور (Schur form)
!>          للنسخة المتوازنة من المصفوفة A.
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرئيسي للمصفوفة A. LDA >= max(1,N).
!>

W

!>          W هو مصفوفة COMPLEX، البعد (N)
!>          تحتوي W على القيم الذاتية المحسوبة.
!>

VL

!>          VL هو مصفوفة COMPLEX، البعد (LDVL,N)
!>          إذا كان JOBVL = 'V'، فإن المتجهات الذاتية اليسرى u(j) مخزنة
!>          واحدة تلو الأخرى في أعمدة VL، بنفس الترتيب
!>          مثل قيمها الذاتية.
!>          إذا كان JOBVL = 'N'، لا يتم استخدام VL.
!>          u(j) = VL(:,j)، العمود j من VL.
!>

LDVL

!>          LDVL هو INTEGER
!>          البعد الرئيسي للمصفوفة VL. LDVL >= 1؛ إذا كان
!>          JOBVL = 'V'، فإن LDVL >= N.
!>

VR

!>          VR هو مصفوفة COMPLEX، البعد (LDVR,N)
!>          إذا كان JOBVR = 'V'، فإن المتجهات الذاتية اليمنى v(j) مخزنة
!>          واحدة تلو الأخرى في أعمدة VR، بنفس الترتيب
!>          مثل قيمها الذاتية.
!>          إذا كان JOBVR = 'N'، لا يتم استخدام VR.
!>          v(j) = VR(:,j)، العمود j من VR.
!>

LDVR

!>          LDVR هو INTEGER
!>          البعد الرئيسي للمصفوفة VR. LDVR >= 1؛ إذا كان
!>          JOBVR = 'V'، فإن LDVR >= N.
!>

ILO

!>          ILO هو INTEGER
!>

IHI

!>          IHI هو INTEGER
!>          ILO و IHI هي قيم صحيحة محددة عند موازنة A.
!>          المصفوفة المتوازنة A(i,j) = 0 إذا كان I > J و
!>          J = 1,...,ILO-1 أو I = IHI+1,...,N.
!>

SCALE

!>          SCALE هو مصفوفة REAL، البعد (N)
!>          تفاصيل التبادلات وعوامل التحجيم المطبقة
!>          عند موازنة A. إذا كان P(j) هو فهرس الصف والعمود
!>          المستبدل بالصف والعمود j، و D(j) هو عامل التحجيم
!>          المطبق على الصف والعمود j، فإن
!>          SCALE(J) = P(J)،    للـ J = 1,...,ILO-1
!>                   = D(J)،    للـ J = ILO,...,IHI
!>                   = P(J)     للـ J = IHI+1,...,N.
!>          ترتيب التبادلات هو N إلى IHI+1،
!>          ثم 1 إلى ILO-1.
!>

ABNRM

!>          ABNRM هو REAL
!>          المعيار الواحد للمصفوفة المتوازنة (الحد الأقصى
!>          لمجموع القيم المطلقة لعناصر أي عمود).
!>

RCONDE

!>          RCONDE هو مصفوفة REAL، البعد (N)
!>          RCONDE(j) هو معكوس رقم الشرط للقيمة الذاتية j.
!>

RCONDV

!>          RCONDV هو مصفوفة REAL، البعد (N)
!>          RCONDV(j) هو معكوس رقم الشرط للمتجه الذاتي الأيمن j.
!>

WORK

!>          WORK هو COMPLEX ar
