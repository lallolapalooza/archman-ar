subroutine cgeru (integer m, integer n, complex alpha, complex, dimension(*) x, integer incx, complex, dimension(*) y, integer incy, complex, dimension(lda,*) a, integer lda)

CGERU

الغرض:

!>
!> CGERU  يقوم بعملية الرتبة 1
!>
!>    A := alpha*x*y**T + A,
!>
!> حيث alpha عدد قياسي، x متجه من m عنصر، y متجه من n عنصر
!> و A مصفوفة m × n.
!>

المعلمات

M 
!>          M هو عدد صحيح (INTEGER)
!>           عند الإدخال، يحدد M عدد صفوف المصفوفة A.
!>           يجب أن يكون M على الأقل صفرًا.
!>

N

!>          N هو عدد صحيح (INTEGER)
!>           عند الإدخال، يحدد N عدد أعمدة المصفوفة A.
!>           يجب أن يكون N على الأقل صفرًا.
!>

ALPHA

!>          ALPHA هو عدد مركب (COMPLEX)
!>           عند الإدخال، يحدد ALPHA العدد القياسي alpha.
!>

X

!>          X هو مصفوفة مركبة (COMPLEX array)، أبعادها على الأقل
!>           ( 1 + ( m - 1 )*abs( INCX ) ).
!>           قبل الإدخال، يجب أن تحتوي المصفوفة X المتزايدة على متجه x
!>           المكون من m عنصر.
!>

INCX

!>          INCX هو عدد صحيح (INTEGER)
!>           عند الإدخال، يحدد INCX الزيادة لعناصر X.
!>           يجب ألا يكون INCX صفرًا.
!>

Y

!>          Y هي مصفوفة مركبة (COMPLEX array)، أبعادها على الأقل
!>           ( 1 + ( n - 1 )*abs( INCY ) ).
!>           قبل الإدخال، يجب أن تحتوي المصفوفة Y المتزايدة على متجه y
!>           المكون من n عنصر.
!>

INCY

!>          INCY هو عدد صحيح (INTEGER)
!>           عند الإدخال، يحدد INCY الزيادة لعناصر Y.
!>           يجب ألا يكون INCY صفرًا.
!>

A

!>          A هي مصفوفة مركبة (COMPLEX array)، أبعادها ( LDA, N )
!>           قبل الإدخال، يجب أن يحتوي الجزء الرئيسي من المصفوفة A
!>           المكون من m صف و n عمود على مصفوفة المعاملات. عند الخروج،
!>           تُستبدل المصفوفة A بالمصفوفة المحدثة.
!>

LDA

!>          LDA هو عدد صحيح (INTEGER)
!>           عند الإدخال، يحدد LDA البعد الأول للمصفوفة A كما هو معلن
!>           في البرنامج (الفرعي) المستدعي.
!>           يجب أن يكون LDA على الأقل max( 1, m ).
!>
