subroutine cgtts2 (integer itrans, integer n, integer nrhs, complex, dimension( * ) dl, complex, dimension( * ) d, complex, dimension( * ) du, complex, dimension( * ) du2, integer, dimension( * ) ipiv, complex, dimension( ldb, * ) b, integer ldb)

CGTTS2 يحل نظامًا من المعادلات الخطية بمصفوفة ثلاثية الأقطار باستخدام تحليل LU الذي تم حسابه بواسطة sgttrf.

الغرض:

!>
!> CGTTS2 يحل أحد أنظمة المعادلات
!>    A * X = B,  A**T * X = B,  أو  A**H * X = B,
!> بمصفوفة ثلاثية الأقطار A باستخدام تحليل LU الذي تم حسابه
!> بواسطة CGTTRF.
!>

المعلمات

ITRANS 
!>          ITRANS هو عدد صحيح (INTEGER)
!>          يحدد شكل نظام المعادلات.
!>          = 0:  A * X = B     (بدون تبديل)
!>          = 1:  A**T * X = B  (تبديل)
!>          = 2:  A**H * X = B  (تبديل مصاحب)
!>

N

!>          N هو عدد صحيح (INTEGER)
!>          رتبة المصفوفة A.
!>

NRHS

!>          NRHS هو عدد صحيح (INTEGER)
!>          عدد الأطراف اليمنى، أي عدد أعمدة المصفوفة B.  NRHS >= 0.
!>

DL

!>          DL هو مصفوفة مركبة (COMPLEX)، البعد (N-1)
!>          المضاعفات (n-1) التي تحدد المصفوفة L من
!>          تحليل LU للمصفوفة A.
!>

D

!>          D هو مصفوفة مركبة (COMPLEX)، البعد (N)
!>          العناصر القطرية n للمصفوفة المثلثية العليا U من
!>          تحليل LU للمصفوفة A.
!>

DU

!>          DU هو مصفوفة مركبة (COMPLEX)، البعد (N-1)
!>          العناصر (n-1) من القطر العلوي الأول لـ U.
!>

DU2

!>          DU2 هو مصفوفة مركبة (COMPLEX)، البعد (N-2)
!>          العناصر (n-2) من القطر العلوي الثاني لـ U.
!>

IPIV

!>          IPIV هو مصفوفة أعداد صحيحة (INTEGER)، البعد (N)
!>          مؤشرات المحور؛ لـ 1 <= i <= n، تم تبديل الصف i للمصفوفة
!>          مع الصف IPIV(i).  سيكون IPIV(i) دائمًا إما
!>          i أو i+1؛ IPIV(i) = i يشير إلى عدم الحاجة إلى تبديل الصف.
!>

B

!>          B هي مصفوفة مركبة (COMPLEX)، البعد (LDB,NRHS)
!>          عند الإدخال، المصفوفة ذات المتجهات الطرفية اليمنى B.
!>          عند الخروج، يتم استبدال B بمتجهات الحل X.
!>

LDB

!>          LDB هو عدد صحيح (INTEGER)
!>          البعد الرئيسي للمصفوفة B.  LDB >= max(1,N).
!>
