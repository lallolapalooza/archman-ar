subroutine cgebd2 (m, n, a, lda, d, e, tauq, taup, work, info)
CGEBD2 تقلل مصفوفة عامة إلى شكل ثنائي قطري باستخدام خوارزمية غير مجمعة.

توثيق الدالة/الإجراء الفرعي

subroutine cgebd2 (integer m, integer n, complex, dimension( lda, * ) a, integer lda, real, dimension( * ) d, real, dimension( * ) e, complex, dimension( * ) tauq, complex, dimension( * ) taup, complex, dimension( * ) work, integer info)

CGEBD2 تقلل مصفوفة عامة معقدة m × n إلى شكل ثنائي قطري حقيقي علوي أو سفلي B بواسطة تحويل وحدوي: Q**H * A * P = B.

الغرض:

!>
!> CGEBD2 تقلل مصفوفة عامة معقدة m × n A إلى شكل ثنائي قطري حقيقي علوي أو سفلي B بواسطة تحويل وحدوي: Q**H * A * P = B.
!>
!> إذا كان m >= n، فإن B يكون ثنائي قطري علوي؛ إذا كان m < n، فإن B يكون ثنائي قطري سفلي.
!>

المعلمات

M 
!>          M هو عدد صحيح (INTEGER)
!>          عدد الصفوف في المصفوفة A.  M >= 0.
!>

N

!>          N هو عدد صحيح (INTEGER)
!>          عدد الأعمدة في المصفوفة A.  N >= 0.
!>

A

!>          A هي مصفوفة معقدة (COMPLEX)، البعد (LDA,N)
!>          عند الدخول، المصفوفة العامة m × n المراد تقليلها.
!>          عند الخروج،
!>          إذا كان m >= n، يتم استبدال القطر الرئيسي والقطر فوق الرئيسي الأول بالمصفوفة الثنائية القطرية العلوية B؛ العناصر الموجودة أسفل القطر، مع المصفوفة TAUQ، تمثل المصفوفة الوحدوية Q كناتج عواكس أولية، والعناصر الموجودة فوق القطر فوق الرئيسي الأول، مع المصفوفة TAUP، تمثل المصفوفة الوحدوية P كناتج عواكس أولية؛
!>          إذا كان m < n، يتم استبدال القطر الرئيسي والقطر تحت الرئيسي الأول بالمصفوفة الثنائية القطرية السفلية B؛ العناصر الموجودة أسفل القطر تحت الرئيسي الأول، مع المصفوفة TAUQ، تمثل المصفوفة الوحدوية Q كناتج عواكس أولية، والعناصر الموجودة فوق القطر، مع المصفوفة TAUP، تمثل المصفوفة الوحدوية P كناتج عواكس أولية.
!>          راجع التفاصيل الإضافية.
!>

LDA

!>          LDA هو عدد صحيح (INTEGER)
!>          البعد الرائد للمصفوفة A.  LDA >= max(1,M).
!>

D

!>          D هي مصفوفة حقيقية (REAL)، البعد (min(M,N))
!>          العناصر القطرية للمصفوفة الثنائية القطرية B:
!>          D(i) = A(i,i).
!>

E

!>          E هي مصفوفة حقيقية (REAL)، البعد (min(M,N)-1)
!>          العناصر غير القطرية للمصفوفة الثنائية القطرية B:
!>          إذا كان m >= n، فإن E(i) = A(i,i+1) لـ i = 1,2,...,n-1؛
!>          إذا كان m < n، فإن E(i) = A(i+1,i) لـ i = 1,2,...,m-1.
!>

TAUQ

!>          TAUQ هي مصفوفة معقدة (COMPLEX)، البعد (min(M,N))
!>          العوامل القياسية للعواكس الأولية التي تمثل المصفوفة الوحدوية Q. راجع التفاصيل الإضافية.
!>

TAUP

!>          TAUP هي مصفوفة معقدة (COMPLEX)، البعد (min(M,N))
!>          العوامل القياسية للعواكس الأولية التي تمثل المصفوفة الوحدوية P. راجع التفاصيل الإضافية.
!>

WORK

!>          WORK هي مصفوفة معقدة (COMPLEX)، البعد (max(M,N))
!>

INFO

!>          INFO هو عدد صحيح (INTEGER)
!>          = 0: الخروج بنجاح
!>          < 0: إذا كان INFO = -i، فإن المعلمة i كانت ذات قيمة غير قانونية.
!>

التفاصيل الإضافية:

!>
!>  المصفوفتان Q و P ممثلتان كناتج عواكس أولية:
!>
!>  إذا كان m >= n،
!>
!>     Q = H(1) H(2) . . . H(n)  و  P = G(1) G(2) . . . G(n-1)
!>
!>  كل H(i) و G(i) لها الشكل التالي:
!>
!>     H(i) = I - tauq * v * v**H  و G(i) = I - taup * u * u**H
!>
!>  حيث tauq و taup هما عاملان قياسيان معقدان، و v و u هما متجهان معقدان؛ v(1:i-1) = 0، v(i) = 1، ويتم تخزين v(i+1:m) عند الخروج في A(i+1:m,i)؛ u(1:i) = 0، u(i+1) = 1، ويتم تخزين u(i+2:n) عند الخروج في A(i,i+2:n)؛ tauq مخزن في TAUQ(i) و taup في TAUP(i).
!>
!>  إذا كان m < n،
!>
!>     Q = H(1) H(2) . . . H(m-1)  و  P = G(1) G(2) . . . G(m)
!>
!>  كل H(i) و G(i) لها الشكل التالي:
!>
!>     H(i) = I - tauq * v * v**H  و G(i) = I - taup * u * u**H
!>
!>  حيث tauq و taup هما عاملان قياسيان معقدان، v و u هما متجهان معقدان؛ v(1:i) = 0، v(i+1) = 1، ويتم تخزين v(i+2:m) عند الخروج في A(i+2:m,i)؛ u(1:i-1) = 0، u(i) = 1، ويتم تخزين u(i+1:n) عند الخروج في A(i,i+1:n)؛ tauq مخزن في TAUQ(i) و taup في TAUP(i).
!>
!>  يتم توضيح محتويات A عند الخروج بالأمثلة التالية:
!>
!>  m = 6 و n = 5 (m > n):          m = 5 و n = 6 (m < n):
!>
!>    (  d   e   u1  u1  u1 )           (  d   u1  u1  u1  u1  u1 )
!>    (  v1  d   e   u2  u2 )           (  e   d   u2  u2  u2  u2 )
!>    (  v1  v2  d   e   u3 )           (  v1  e   d   u3  u3  u3 )
!>    (  v1  v2  v3  d   e  )           (  v1  v2  e   d   u4  u4 )
!>    (  v1  v2  v3  v4  d  )           (  v1  v2  v3  e   d   u5 )
!>    (  v1  v2  v3  v4  v5 )
!>
!>  حيث d و e يمثلان عناصر قطرية وغير قطرية من B، vi يمثل عنصرًا من المتجه الذي يحدد H(i)، و ui عنصرًا من المتجه الذي يحدد G(i).
!>
