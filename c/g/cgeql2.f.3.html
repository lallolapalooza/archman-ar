subroutine cgeql2 (integer m, integer n, complex, dimension( lda, * ) a, integer lda, complex, dimension( * ) tau, complex, dimension( * ) work, integer info)

CGEQL2 تحسب التحلل QL لمصفوفة مستطيلة عامة باستخدام خوارزمية غير محظورة.

الغرض:

!>
!> CGEQL2 تحسب تحلل QL لمصفوفة A المعقدة m × n:
!> A = Q * L.
!>

المعلمات

M 
!>          M هو عدد صحيح (INTEGER)
!>          عدد الصفوف في المصفوفة A.  M >= 0.
!>

N

!>          N هو عدد صحيح (INTEGER)
!>          عدد الأعمدة في المصفوفة A.  N >= 0.
!>

A

!>          A هو مصفوفة معقدة (COMPLEX)، أبعادها (LDA,N)
!>          عند الدخول، المصفوفة A بالمقاس m × n.
!>          عند الخروج، إذا كانت m >= n، فإن المثلث السفلي من المصفوفة الفرعية
!>          A(m-n+1:m,1:n) يحتوي على المصفوفة المثلثية السفلية L بالمقاس n × n؛
!>          إذا كانت m <= n، فإن العناصر على وتحت القطر فوق القطري رقم (n-m)
!>          تحتوي على المصفوفة الشبه منحرفة السفلية L بالمقاس m × n؛
!>          العناصر المتبقية، مع المصفوفة TAU، تمثل المصفوفة الوحدوية Q كناتج
!>          لمنعكسات ابتدائية (انظر التفاصيل الإضافية).
!>

LDA

!>          LDA هو عدد صحيح (INTEGER)
!>          البعد الرائد للمصفوفة A.  LDA >= max(1,M).
!>

TAU

!>          TAU هي مصفوفة معقدة (COMPLEX)، أبعادها (min(M,N))
!>          العوامل القياسية للمنعكسات الابتدائية (انظر التفاصيل الإضافية).
!>

WORK

!>          WORK هي مصفوفة معقدة (COMPLEX)، أبعادها (N)
!>

INFO

!>          INFO هو عدد صحيح (INTEGER)
!>          = 0: خروج ناجح
!>          < 0: إذا كان INFO = -i، فإن المعامل i كان له قيمة غير قانونية
!>

تفاصيل إضافية:

!>
!>  يتم تمثيل المصفوفة Q كناتج لمنعكسات ابتدائية
!>
!>     Q = H(k) . . . H(2) H(1)، حيث k = min(m,n).
!>
!>  كل H(i) لها الشكل
!>
!>     H(i) = I - tau * v * v**H
!>
!>  حيث tau هو معامل معقد، و v هو متجه معقد مع
!>  v(m-k+i+1:m) = 0 و v(m-k+i) = 1؛ v(1:m-k+i-1) مخزن عند الخروج في
!>  A(1:m-k+i-1,n-k+i)، و tau في TAU(i).
!>

تعريف في السطر 122 من الملف cgeql2.f.
