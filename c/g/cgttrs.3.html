وثائق الوظيفة/الإجراء الفرعي

subroutine cgttrs (character trans, integer n, integer nrhs, complex, dimension( * ) dl, complex, dimension( * ) d, complex, dimension( * ) du, complex, dimension( * ) du2, integer, dimension( * ) ipiv, complex, dimension( ldb, * ) b, integer ldb, integer info)

CGTTRS

الغرض:

!>
!> CGTTRS يحل أحد أنظمة المعادلات
!>    A * X = B,  A**T * X = B,  أو  A**H * X = B,
!> مع مصفوفة ثلاثية قطرية A باستخدام تحليل LU الذي تم حسابه
!> بواسطة CGTTRF.
!>

المعلمات

TRANS 
!>          TRANS هو CHARACTER*1
!>          يحدد شكل نظام المعادلات.
!>          = 'N':  A * X = B     (بدون تبديل)
!>          = 'T':  A**T * X = B  (تبديل)
!>          = 'C':  A**H * X = B  (تبديل مترافق)
!>

N

!>          N هو INTEGER
!>          رتبة المصفوفة A.
!>

NRHS

!>          NRHS هو INTEGER
!>          عدد الأطراف اليمنى، أي عدد أعمدة
!>          المصفوفة B.  NRHS >= 0.
!>

DL

!>          DL هو COMPLEX array, dimension (N-1)
!>          المضاعفات (n-1) التي تحدد المصفوفة L من
!>          تحليل LU للمصفوفة A.
!>

D

!>          D هو COMPLEX array, dimension (N)
!>          العناصر القطرية N للمصفوفة المثلثية العلوية U من
!>          تحليل LU للمصفوفة A.
!>

DU

!>          DU هو COMPLEX array, dimension (N-1)
!>          العناصر (n-1) في الأقطار فوق القطرية الأولى لـ U.
!>

DU2

!>          DU2 هو COMPLEX array, dimension (N-2)
!>          العناصر (n-2) في الأقطار فوق القطرية الثانية لـ U.
!>

IPIV

!>          IPIV هو INTEGER array, dimension (N)
!>          مؤشرات المحور؛ لـ 1 <= i <= n، تم تبديل الصف i للمصفوفة
!>          مع الصف IPIV(i).  سيكون IPIV(i) دائمًا إما
!>          i أو i+1؛ يشير IPIV(i) = i إلى عدم الحاجة لتبديل صفوف.
!>

B

!>          B هو COMPLEX array, dimension (LDB,NRHS)
!>          عند الدخول، المصفوفة ذات الأطراف اليمنى B.
!>          عند الخروج، يتم استبدال B بمتجهات الحل X.
!>

LDB

!>          LDB هو INTEGER
!>          البعد الرئيسي للمصفوفة B.  LDB >= max(1,N).
!>

INFO

!>          INFO هو INTEGER
!>          = 0:  انتهاء ناجح
!>          < 0:  إذا كان INFO = -k، فإن الوسيط k كان له قيمة غير قانونية
!>
