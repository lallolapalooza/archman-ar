subroutine cgeqrfp (integer m, integer n, complex, dimension( lda, * ) a, integer lda, complex, dimension( * ) tau, complex, dimension( * ) work, integer lwork, integer info)

CGEQRFP

الغرض:

!>
!> CGEQR2P تحسب عامل QR لمصفوفة معقدة M-by-N A:
!>
!>    A = Q * ( R ),
!>            ( 0 )
!>
!> حيث:
!>
!>    Q هي مصفوفة متعامدة M-by-M؛
!>    R هي مصفوفة شبه مصفوفة علوية N-by-N ذات مدخلات قطرية غير سالبة؛
!>    0 هي مصفوفة صفرية (M-N)-by-N، إذا كانت M > N.
!>
!>

المعلمات

M 
!>          M هو عدد صحيح (INTEGER)
!>          عدد الصفوف في المصفوفة A.  M >= 0.
!>

N

!>          N هو عدد صحيح (INTEGER)
!>          عدد الأعمدة في المصفوفة A.  N >= 0.
!>

A

!>          A هي مصفوفة مركبة (COMPLEX array)، بأبعاد (LDA,N)
!>          عند الإدخال، مصفوفة M-by-N A.
!>          عند الخروج، تحتوي العناصر على أو فوق القطر للمصفوفة على
!>          المصفوفة شبه المنحرفة العلوية R ذات الأبعاد min(M,N)-by-N (R
!>          هي شبه مصفوفة علوية إذا كانت m >= n). المدخلات القطرية لـ R
!>          حقيقية وغير سالبة؛ العناصر تحت القطر،
!>          بالاشتراك مع المصفوفة TAU، تمثل المصفوفة الوحدية Q كـ
!>          ناتج ضرب min(m,n) من انعكاسات أولية (انظر التفاصيل الإضافية).
!>

LDA

!>          LDA هو عدد صحيح (INTEGER)
!>          البعد الرائد للمصفوفة A.  LDA >= max(1,M).
!>

TAU

!>          TAU هي مصفوفة مركبة (COMPLEX array)، بأبعاد (min(M,N))
!>          العوامل القياسية للانعكاسات الأولية (انظر التفاصيل الإضافية).
!>

WORK

!>          WORK هي مصفوفة مركبة (COMPLEX array)، بأبعاد (MAX(1,LWORK))
!>          عند الخروج، إذا كانت INFO = 0، فإن WORK(1) تعيد LWORK المثلى.
!>

LWORK

!>          LWORK هو عدد صحيح (INTEGER)
!>          بعد المصفوفة WORK.
!>          LWORK >= 1، إذا كان MIN(M,N) = 0، و LWORK >= N، بخلاف ذلك.
!>          للحصول على الأداء الأمثل LWORK >= N*NB، حيث NB هو
!>          حجم الكتلة الأمثل.
!>
!>          إذا كانت LWORK = -1، فإنه يُفترض استعلام عن مساحة العمل؛ الروتين
!>          يحسب فقط الحجم الأمثل للمصفوفة WORK، ويعيد
!>          هذه القيمة كمدخل أول للمصفوفة WORK، ولا يتم إصدار رسالة خطأ
!>          متعلقة بـ LWORK بواسطة XERBLA.
!>

INFO

!>          INFO هو عدد صحيح (INTEGER)
!>          = 0:  انتهاء ناجح
!>          < 0:  إذا كانت INFO = -i، فإن المعلمة i كان لها قيمة غير قانونية
!>

تفاصيل إضافية:

!>
!>  يتم تمثيل المصفوفة Q كناتج ضرب انعكاسات أولية
!>
!>     Q = H(1) H(2) . . . H(k)، حيث k = min(m,n).
!>
!>  كل H(i) له الشكل
!>
!>     H(i) = I - tau * v * v**H
!>
!>  حيث tau هو عامل قياسي مركب، و v هو متجه مركب مع
!>  v(1:i-1) = 0 و v(i) = 1؛ يتم تخزين v(i+1:m) عند الخروج في A(i+1:m,i)،
!>  و tau في TAU(i).
!>
!> انظر مذكرة عمل Lapack 203 لمزيد من التفاصيل
!>
