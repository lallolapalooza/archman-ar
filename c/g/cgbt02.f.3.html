وثائق الوظيفة/الروتين الفرعي

subroutine cgbt02 (character trans, integer m, integer n, integer kl, integer ku, integer nrhs, complex, dimension( lda, * ) a, integer lda, complex, dimension( ldx, * ) x, integer ldx, complex, dimension( ldb, * ) b, integer ldb, real, dimension( * ) rwork, real resid)

CGBT02

الغرض:

!>
!> CGBT02 يحسب الباقي لحل نظام من المعادلات النطاقية op(A)*X = B:
!>    RESID = norm(B - op(A)*X) / ( norm(op(A)) * norm(X) * EPS ),
!> حيث op(A) = A, A**T, أو A**H, اعتمادًا على TRANS, و EPS هو
!> ثابت آلة الخطأ.
!>

المعلمات

TRANS 
!>          TRANS is CHARACTER*1
!>          يحدد شكل نظام المعادلات:
!>          = 'N':  A    * X = B  (بدون تبديل)
!>          = 'T':  A**T * X = B  (تبديل)
!>          = 'C':  A**H * X = B  (تبديل مقترن)
!>

M

!>          M is INTEGER
!>          عدد صفوف المصفوفة A.  M >= 0.
!>

N

!>          N is INTEGER
!>          عدد أعمدة المصفوفة A.  N >= 0.
!>

KL

!>          KL is INTEGER
!>          عدد الأقطار الفرعية داخل نطاق A.  KL >= 0.
!>

KU

!>          KU is INTEGER
!>          عدد الأقطار الفوقية داخل نطاق A.  KU >= 0.
!>

NRHS

!>          NRHS is INTEGER
!>          عدد أعمدة B.  NRHS >= 0.
!>

A

!>          A is COMPLEX array, dimension (LDA,N)
!>          المصفوفة الأصلية A بتخزين نطاقي، مخزنة في الصفوف من 1 إلى
!>          KL+KU+1.
!>

LDA

!>          LDA is INTEGER
!>          البعد الرائد للمصفوفة A.  LDA >= max(1,KL+KU+1).
!>

X

!>          X is COMPLEX array, dimension (LDX,NRHS)
!>          متجهات الحلول المحسوبة لنظام المعادلات الخطية.
!>

LDX

!>          LDX is INTEGER
!>          البعد الرائد للمصفوفة X.  إذا كان TRANS = 'N',
!>          LDX >= max(1,N)؛ إذا كان TRANS = 'T' أو 'C', LDX >= max(1,M).
!>

B

!>          B is COMPLEX array, dimension (LDB,NRHS)
!>          عند الإدخال، متجهات الطرف الأيمن لنظام المعادلات الخطية.
!>          عند الخروج، يتم استبدال B بالفرق B - A*X.
!>

LDB

!>          LDB is INTEGER
!>          البعد الرائد للمصفوفة B.  إذا كان TRANS = 'N',
!>          LDB >= max(1,M)؛ إذا كان TRANS = 'T' أو 'C', LDB >= max(1,N).
!>

RWORK

!>          RWORK is REAL array, dimension (MAX(1,LRWORK)),
!>          حيث LRWORK >= M عندما يكون TRANS = 'T' أو 'C'؛ بخلاف ذلك، لا تتم
!>          الإشارة إلى RWORK.
!>

RESID

!>          RESID is REAL
!>          القيمة القصوى لعدد الأطراف اليمنى من
!>          norm(B - op(A)*X) / ( norm(op(A)) * norm(X) * EPS ).
!>
