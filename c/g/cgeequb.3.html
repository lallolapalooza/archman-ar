subroutine cgeequb (integer m, integer n, complex, dimension( lda, * ) a, integer lda, real, dimension( * ) r, real, dimension( * ) c, real rowcnd, real colcnd, real amax, integer info)

CGEEQUB

الغرض:

!>
!> CGEEQUB تحسب عوامل القياس للصف والأعمدة بهدف موازنة المصفوفة M-by-N A
!> وتقليل رقم شرطها. R ترجع عوامل قياس الصفوف و C عوامل قياس الأعمدة،
!> التي يتم اختيارها لمحاولة جعل أكبر قيمة مطلقة للعنصر في كل صف وعمود
!> للمصفوفة B ذات العناصر B(i,j)=R(i)*A(i,j)*C(j) لا تتجاوز
!> أساس النظام (radix).
!>
!> R(i) و C(j) مقيدة لتكون قوة من أساس النظام بين
!> SMLNUM = أصغر رقم آمن و BIGNUM = أكبر رقم آمن. استخدام
!> عوامل القياس هذه لا يضمن تقليل رقم شرط A ولكنه يعمل بشكل جيد عمليًا.
!>
!> هذا الإجراء يختلف عن CGEEQU بتقييد عوامل القياس
!> لتكون قوة من أساس النظام. باستثناء حالات تجاوز السعة أو السعة الدنيا،
!> القياس بهذه العوامل لا يسبب أخطاء تقريب إضافية. ومع ذلك،
!> فإن قيم عناصر المصفوفة المقاسة لم تعد تقريبًا 1 ولكنها تقع
!> بين sqrt(radix) و 1/sqrt(radix).
!>

المعلمات

M 
!>          M هو عدد صحيح (INTEGER)
!>          عدد صفوف المصفوفة A.  M >= 0.
!>

N

!>          N هو عدد صحيح (INTEGER)
!>          عدد أعمدة المصفوفة A.  N >= 0.
!>

A

!>          A هي مصفوفة مركبة (COMPLEX array)، أبعادها (LDA,N)
!>          المصفوفة M-by-N التي سيتم حساب عوامل موازنتها.
!>

LDA

!>          LDA هو عدد صحيح (INTEGER)
!>          البعد الرئيسي للمصفوفة A.  LDA >= max(1,M).
!>

R

!>          R هي مصفوفة حقيقية (REAL array)، أبعادها (M)
!>          إذا كان INFO = 0 أو INFO > M، تحتوي R على عوامل قياس الصفوف
!>          للمصفوفة A.
!>

C

!>          C هي مصفوفة حقيقية (REAL array)، أبعادها (N)
!>          إذا كان INFO = 0، تحتوي C على عوامل قياس الأعمدة للمصفوفة A.
!>

ROWCND

!>          ROWCND هو عدد حقيقي (REAL)
!>          إذا كان INFO = 0 أو INFO > M، يحتوي ROWCND على نسبة أصغر R(i)
!>          إلى أكبر R(i). إذا كان ROWCND >= 0.1 و AMAX ليس كبيرًا جدًا
!>          أو صغيرًا جدًا، فلا فائدة من القياس باستخدام R.
!>

COLCND

!>          COLCND هو عدد حقيقي (REAL)
!>          إذا كان INFO = 0، يحتوي COLCND على نسبة أصغر C(i) إلى أكبر C(i).
!>          إذا كان COLCND >= 0.1، فلا فائدة من القياس باستخدام C.
!>

AMAX

!>          AMAX هو عدد حقيقي (REAL)
!>          القيمة المطلقة لأكبر عنصر في المصفوفة. إذا كان AMAX قريبًا جدًا
!>          من تجاوز السعة أو قريبًا جدًا من السعة الدنيا، فيجب قياس المصفوفة.
!>

INFO

!>          INFO هو عدد صحيح (INTEGER)
!>          = 0:  انتهاء ناجح
!>          < 0:  إذا كان INFO = -i، فإن المعلمة i كانت ذات قيمة غير قانونية
!>          > 0:  إذا كان INFO = i، و i هو
!>                <= M:  الصف i من A هو صفر تمامًا
!>                >  M:  العمود (i-M) من A هو صفر تمامًا
!>
