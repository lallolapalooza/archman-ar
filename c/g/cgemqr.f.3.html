توثيق الوظيفة/الروتين الفرعي

subroutine cgemqr (character side, character trans, integer m, integer n, integer k, complex, dimension( lda, * ) a, integer lda, complex, dimension( * ) t, integer tsize, complex, dimension( ldc, * ) c, integer ldc, complex, dimension( * ) work, integer lwork, integer info)

CGEMQR

الغرض:

!>
!> CGEMQR تستبدل مصفوفة C الحقيقية العامة M-by-N بـ
!>
!>                      SIDE = 'L'     SIDE = 'R'
!>      TRANS = 'N':      Q * C          C * Q
!>      TRANS = 'T':      Q**H * C       C * Q**H
!>
!> حيث Q هي مصفوفة وحدوية معقدة معرفة كناتج
!> لمصفوفات الانعكاس الأولية المجمعة المحسوبة بواسطة تحليل QR الضيق (CGEQR)
!>
!>

المعلمات

SIDE 
!>          SIDE is CHARACTER*1
!>          = 'L': تطبيق Q أو Q**H من اليسار؛
!>          = 'R': تطبيق Q أو Q**H من اليمين.
!>

TRANS

!>          TRANS is CHARACTER*1
!>          = 'N':  عدم النقل، تطبيق Q؛
!>          = 'C':  النقل المرافق، تطبيق Q**H.
!>

M

!>          M is INTEGER
!>          عدد الصفوف في المصفوفة A.  M >=0.
!>

N

!>          N is INTEGER
!>          عدد الأعمدة في المصفوفة C. N >= 0.
!>

K

!>          K is INTEGER
!>          عدد مصفوفات الانعكاس الأولية التي يعرف حاصل ضربها
!>          المصفوفة Q.
!>          إذا كان SIDE = 'L'، فإن M >= K >= 0؛
!>          إذا كان SIDE = 'R'، فإن N >= K >= 0.
!>

A

!>          A is COMPLEX array, dimension (LDA,K)
!>          جزء من هيكل البيانات لتمثيل Q كما تم إرجاعه بواسطة CGEQR.
!>

LDA

!>          LDA is INTEGER
!>          البعد الرائد للمصفوفة A.
!>          إذا كان SIDE = 'L'، فإن LDA >= max(1,M)؛
!>          إذا كان SIDE = 'R'، فإن LDA >= max(1,N).
!>

T

!>          T is COMPLEX array, dimension (MAX(5,TSIZE)).
!>          جزء من هيكل البيانات لتمثيل Q كما تم إرجاعه بواسطة CGEQR.
!>

TSIZE

!>          TSIZE is INTEGER
!>          بعد المصفوفة T. TSIZE >= 5.
!>

C

!>          C is COMPLEX array, dimension (LDC,N)
!>          عند الدخول، مصفوفة C بالمقاس M-by-N.
!>          عند الخروج، يتم استبدال C بـ Q*C أو Q**H*C أو C*Q**H أو C*Q.
!>

LDC

!>          LDC is INTEGER
!>          البعد الرائد للمصفوفة C. LDC >= max(1,M).
!>

WORK

!>          (مساحة عمل) COMPLEX array, dimension (MAX(1,LWORK))
!>          عند الخروج، إذا كان INFO = 0، فإن WORK(1) يعيد الحد الأدنى LWORK.
!>

LWORK

!>          LWORK is INTEGER
!>          بعد المصفوفة WORK. LWORK >= 1.
!>          إذا كان LWORK = -1، فسيتم افتراض استعلام عن مساحة العمل. تقوم الروتين
!>          بحساب حجم مصفوفة WORK فقط، وإرجاع هذه القيمة كـ WORK(1)، ولا يتم
!>          إصدار أي رسالة خطأ متعلقة بـ WORK بواسطة XERBLA.
!>

INFO

!>          INFO is INTEGER
!>          = 0:  اكتمال ناجح
!>          < 0:  إذا كان INFO = -i، فإن المعامل i كان له قيمة غير قانونية
!>

!>
!> هذه التفاصيل خاصة بتطبيق LAPACK هذا. لا ينبغي للمستخدمين
!> الاعتماد عليها. قد تتغير هذه التفاصيل في المستقبل، ومن غير المرجح
!> أن تكون صحيحة لتطبيق LAPACK آخر. هذه التفاصيل مهمة إذا أراد
!> المرء محاولة فهم الكود. وهي ليست جزءًا من الواجهة.
!>
!> في هذا الإصدار،
!>
!>          T(2): حجم كتلة الصف (MB)
!>          T(3): حجم كتلة العمود (NB)
!>          T(6:TSIZE): هيكل البيانات اللازم لـ Q، تم حسابه بواسطة
!>                           CLATSQR أو CGEQRT
!>
!>  اعتمادًا على أبعاد المصفوفة M و N، وأحجام كتل الصفوف والأعمدة MB و NB
!>  التي تم إرجاعها بواسطة ILAENV، سيستخدم CGEQR إما
!>  CLATSQR (إذا كانت المصفوفة ضيقة وطويلة) أو CGEQRT لحساب
!>  تحليل QR.
!>  سيستخدم هذا الإصدار من CGEMQR إما CLAMTSQR أو CGEMQRT لـ
!>  ضرب المصفوفة Q بمصفوفة أخرى.
!>  مزيد من التفاصيل في CLAMTSQR أو CGEMQRT.
!>
!>
