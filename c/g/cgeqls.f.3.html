توثيق الوظيفة/الروتين الفرعي

subroutine cgeqls (integer m, integer n, integer nrhs, complex, dimension( lda, * ) a, integer lda, complex, dimension( * ) tau, complex, dimension( ldb, * ) b, integer ldb, complex, dimension( lwork ) work, integer lwork, integer info)

CGEQLS

الغرض:

!>
!> حل مشكلة أقل المربعات
!>     min || A*X - B ||
!> باستخدام تحليل QL
!>     A = Q*L
!> الذي تم حسابه بواسطة CGEQLF.
!>

المعلمات

M 
!>          M هو عدد صحيح
!>          عدد صفوف المصفوفة A.  M >= 0.
!>

N

!>          N هو عدد صحيح
!>          عدد أعمدة المصفوفة A.  M >= N >= 0.
!>

NRHS

!>          NRHS هو عدد صحيح
!>          عدد أعمدة B.  NRHS >= 0.
!>

A

!>          A هو مصفوفة مركبة (COMPLEX)، البعد (LDA,N)
!>          تفاصيل تحليل QL للمصفوفة الأصلية A كما
!>          تم إرجاعها بواسطة CGEQLF.
!>

LDA

!>          LDA هو عدد صحيح
!>          البعد الرائد للمصفوفة A.  LDA >= M.
!>

TAU

!>          TAU هو مصفوفة مركبة (COMPLEX)، البعد (N)
!>          تفاصيل المصفوفة المتعامدة Q.
!>

B

!>          B هو مصفوفة مركبة (COMPLEX)، البعد (LDB,NRHS)
!>          عند الإدخال، المصفوفة B ذات الجانب الأيمن التي أبعادها m-by-nrhs.
!>          عند الخروج، المصفوفة الحل X التي أبعادها n-by-nrhs، مخزنة في الصفوف
!>          m-n+1:m.
!>

LDB

!>          LDB هو عدد صحيح
!>          البعد الرائد للمصفوفة B. LDB >= M.
!>

WORK

!>          WORK هي مصفوفة مركبة (COMPLEX)، البعد (LWORK)
!>

LWORK

!>          LWORK هو عدد صحيح
!>          طول المصفوفة WORK.  يجب أن يكون LWORK على الأقل NRHS،
!>          ويجب أن يكون على الأقل NRHS*NB، حيث NB هو حجم الكتلة
!>          لهذه البيئة.
!>

INFO

!>          INFO هو عدد صحيح
!>          = 0: الخروج ناجح
!>          < 0: إذا كان INFO = -i، فإن المعلمة i كانت لها قيمة غير قانونية
!>
