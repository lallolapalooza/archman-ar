توثيق الوظيفة/الإجراء الفرعي

subroutine cgemlq (character side, character trans, integer m, integer n, integer k, complex, dimension( lda, * ) a, integer lda, complex, dimension( * ) t, integer tsize, complex, dimension( ldc, * ) c, integer ldc, complex, dimension( * ) work, integer lwork, integer info)

CGEMLQ

الغرض:

!>
!>     CGEMLQ تستبدل المصفوفة الحقيقية العامة M-by-N المسماة C بـ
!>
!>                      SIDE = 'L'     SIDE = 'R'
!>      TRANS = 'N':      Q * C          C * Q
!>      TRANS = 'C':      Q**H * C       C * Q**H
!>      حيث Q هي مصفوفة وحدة معقدة معرفة كناتج
!>      لمنكسرات أولية مجمعة تم حسابها بواسطة عامل LQ ضيق واسع
!>      (CGELQ)
!>
!>

المعلمات

SIDE 
!>          SIDE is CHARACTER*1
!>          = 'L': طبق Q أو Q**H من اليسار؛
!>          = 'R': طبق Q أو Q**H من اليمين.
!>

TRANS

!>          TRANS is CHARACTER*1
!>          = 'N':  بدون تبديل، طبق Q؛
!>          = 'C':  تبديل منقول مصاحب، طبق Q**H.
!>

M

!>          M is INTEGER
!>          عدد صفوف المصفوفة A.  M >=0.
!>

N

!>          N is INTEGER
!>          عدد أعمدة المصفوفة C. N >= 0.
!>

K

!>          K is INTEGER
!>          عدد المنكسرات الأولية التي يحدد حاصل ضربها
!>          المصفوفة Q.
!>          إذا كان SIDE = 'L'، فإن M >= K >= 0؛
!>          إذا كان SIDE = 'R'، فإن N >= K >= 0.
!>

A

!>          A is COMPLEX array, dimension
!>                               (LDA,M) if SIDE = 'L',
!>                               (LDA,N) if SIDE = 'R'
!>          جزء من هيكل البيانات لتمثيل Q كما تم إرجاعه بواسطة CGELQ.
!>

LDA

!>          LDA is INTEGER
!>          البعد الرائد للمصفوفة A. LDA >= max(1,K).
!>

T

!>          T is COMPLEX array, dimension (MAX(5,TSIZE)).
!>          جزء من هيكل البيانات لتمثيل Q كما تم إرجاعه بواسطة CGELQ.
!>

TSIZE

!>          TSIZE is INTEGER
!>          بعد المصفوفة T. TSIZE >= 5.
!>

C

!>          C is COMPLEX array, dimension (LDC,N)
!>          عند الإدخال، المصفوفة M-by-N المسماة C.
!>          عند الخروج، يتم استبدال C بـ Q*C أو Q**H*C أو C*Q**H أو C*Q.
!>

LDC

!>          LDC is INTEGER
!>          البعد الرائد للمصفوفة C. LDC >= max(1,M).
!>

WORK

!>          (مساحة عمل) COMPLEX array, dimension (MAX(1,LWORK))
!>          عند الخروج، إذا كان INFO = 0، فإن WORK(1) يعيد الحد الأدنى LWORK.
!>

LWORK

!>          LWORK is INTEGER
!>          بعد المصفوفة WORK. LWORK >= 1.
!>          إذا كان LWORK = -1، فيفترض استعلام عن مساحة العمل. يقوم الروتين
!>          فقط بحساب حجم المصفوفة WORK، ويعيد هذا
!>          القيمة كـ WORK(1)، ولا يصدر رسالة خطأ متعلقة بـ WORK
!>          بواسطة XERBLA.
!>

INFO

!>          INFO is INTEGER
!>          = 0:  خروج ناجح
!>          < 0:  إذا كان INFO = -i، فإن الوسيط i كان له قيمة غير قانونية
!>

!>
!> هذه التفاصيل خاصة بهذا التنفيذ لـ LAPACK. لا ينبغي للمستخدمين
!> اعتبارها أمرًا مفروغًا منه. قد تتغير هذه التفاصيل في المستقبل، وهي ليست
!> بالضرورة صحيحة لتنفيذ LAPACK آخر. هذه التفاصيل ذات صلة إذا أراد
!> المرء فهم الكود. وهي ليست جزءًا من الواجهة.
!>
!> في هذا الإصدار،
!>
!>          T(2): حجم الكتلة الصفية (MB)
!>          T(3): حجم الكتلة العمودية (NB)
!>          T(6:TSIZE): هيكل البيانات المطلوب لـ Q، محسوب بواسطة
!>                           CLASWQR أو CGELQT
!>
!>  اعتمادًا على أبعاد المصفوفة M و N، وأحجام الكتل الصفية والعمودية
!>  MB و NB التي تم إرجاعها بواسطة ILAENV، سيستخدم CGELQ إما
!>  CLASWLQ (إذا كانت المصفوفة عريضة وقصيرة) أو CGELQT لحساب
!>  عامل LQ.
!>  سيستخدم هذا الإصدار من CGEMLQ إما CLAMSWLQ أو CGEMLQT لـ
!>  ضرب المصفوفة Q بمصفوفة أخرى.
!>  تفاصيل إضافية في CLAMSWLQ أو CGEMLQT.
!>
