subroutine cgbmv (character trans, integer m, integer n, integer kl, integer ku, complex alpha, complex, dimension(lda,*) a, integer lda, complex, dimension(*) x, integer incx, complex beta, complex, dimension(*) y, integer incy)

CGBMV

الغرض:

!>
!> CGBMV  تنفذ إحدى عمليات المصفوفة والمتجه التالية:
!>
!>    y := alpha*A*x + beta*y,   أو   y := alpha*A**T*x + beta*y,   أو
!>
!>    y := alpha*A**H*x + beta*y,
!>
!> حيث alpha و beta عددان قياسيان، x و y متجهان، و A مصفوفة نطاقية (band matrix)
!> بحجم m × n، مع kl تحت الأقطار (sub-diagonals) و ku فوق الأقطار (super-diagonals).
!>

المعلمات

TRANS 
!>          TRANS هي CHARACTER*1
!>           عند الدخول، تحدد TRANS العملية المراد تنفيذها كالتالي:
!>
!>              TRANS = 'N' أو 'n'   y := alpha*A*x + beta*y.
!>
!>              TRANS = 'T' أو 't'   y := alpha*A**T*x + beta*y.
!>
!>              TRANS = 'C' أو 'c'   y := alpha*A**H*x + beta*y.
!>

M

!>          M هو INTEGER
!>           عند الدخول، تحدد M عدد صفوف المصفوفة A.
!>           يجب أن تكون M على الأقل صفر.
!>

N

!>          N هو INTEGER
!>           عند الدخول، تحدد N عدد أعمدة المصفوفة A.
!>           يجب أن تكون N على الأقل صفر.
!>

KL

!>          KL هو INTEGER
!>           عند الدخول، تحدد KL عدد تحت الأقطار للمصفوفة A.
!>           يجب أن يحقق KL  0 .le. KL.
!>

KU

!>          KU هو INTEGER
!>           عند الدخول، تحدد KU عدد فوق الأقطار للمصفوفة A.
!>           يجب أن يحقق KU  0 .le. KU.
!>

ALPHA

!>          ALPHA هو COMPLEX
!>           عند الدخول، تحدد ALPHA العدد القياسي alpha.
!>

A

!>          A هو مصفوفة COMPLEX، أبعادها ( LDA, N )
!>           قبل الدخول، يجب أن يحتوي الجزء الرائد ( kl + ku + 1 ) × N من المصفوفة
!>           على معاملات المصفوفة، مع توفيرها عمودًا بعمود، بحيث يكون القطر الرئيسي
!>           للمصفوفة في الصف ( ku + 1 ) من المصفوفة، ويبدأ فوق القطر الأول
!>           عند الموضع 2 في الصف ku، ويبدأ تحت القطر الأول عند الموضع 1 في الصف
!>           ( ku + 2 )، وهكذا.
!>           لا يتم الرجوع إلى العناصر في المصفوفة A التي لا تتوافق مع عناصر
!>           في مصفوفة النطاق (مثل المثلث العلوي الأيسر بحجم ku × ku).
!>           يمكن لمقطع البرنامج التالي نقل مصفوفة نطاق
!>           من التخزين الكامل التقليدي إلى تخزين النطاق:
!>
!>                 DO 20, J = 1, N
!>                    K = KU + 1 - J
!>                    DO 10, I = MAX( 1, J - KU ), MIN( M, J + KL )
!>                       A( K + I, J ) = matrix( I, J )
!>              10    CONTINUE
!>              20 CONTINUE
!>

LDA

!>          LDA هو INTEGER
!>           عند الدخول، تحدد LDA البعد الأول لـ A كما هو معلن
!>           في البرنامج المستدعي. يجب أن يكون LDA على الأقل
!>           ( kl + ku + 1 ).
!>

X

!>          X هي مصفوفة COMPLEX، أبعادها على الأقل
!>           ( 1 + ( n - 1 )*abs( INCX ) ) عندما TRANS = 'N' أو 'n'
!>           وعلى الأقل
!>           ( 1 + ( m - 1 )*abs( INCX ) ) بخلاف ذلك.
!>           قبل الدخول، يجب أن تحتوي المصفوفة X المزودة بزيادات على المتجه
!>           x.
!>

INCX

!>          INCX هو INTEGER
!>           عند الدخول، تحدد INCX الزيادة للعناصر في
!>           X. يجب أن تكون INCX غير صفر.
!>

BETA

!>          BETA هو COMPLEX
!>           عند الدخول، تحدد BETA العدد القياسي beta. عندما يتم
!>           توفير BETA بقيمة صفر، لا يلزم تعيين Y عند الإدخال.
!>

Y

!>          Y هي مصفوفة COMPLEX، أبعادها على الأقل
!>           ( 1 + ( m - 1 )*abs( INCY ) ) عندما TRANS = 'N' أو 'n'
!>           وعلى الأقل
!>           ( 1 + ( n - 1 )*abs( INCY ) ) بخلاف ذلك.
!>           قبل الدخول، يجب أن تحتوي المصفوفة Y المزودة بزيادات على المتجه
!>           y. عند الخروج، يتم الكتابة فوق Y بالمتجه y المحدث.
!>           إذا كان m أو n يساوي صفرًا، فلا يتم الرجوع إلى Y وتنفذ الدالة
!>           عودة سريعة.
!>

INCY

!>          INCY هو INTEGER
!>           عند الدخول، تحدد INCY الزيادة للعناصر في
!>           Y. يجب أن تكون INCY غير صفر.
!>
