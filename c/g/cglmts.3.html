subroutine cglmts (integer n, integer m, integer p, complex, dimension( lda, * ) a, complex, dimension( lda, * ) af, integer lda, complex, dimension( ldb, * ) b, complex, dimension( ldb, * ) bf, integer ldb, complex, dimension( * ) d, complex, dimension( * ) df, complex, dimension( * ) x, complex, dimension( * ) u, complex, dimension( lwork ) work, integer lwork, real, dimension( * ) rwork, real result)

CGLMTS

الغرض:

!>
!> CGLMTS تختبر CGGGLM - روتين فرعي لحل مشكلة النموذج الخطي المعمم.
!>

المعلمات

N 
!>          N هو عدد صحيح (INTEGER)
!>          عدد صفوف المصفوفات A و B. N >= 0.
!>

M

!>          M هو عدد صحيح (INTEGER)
!>          عدد أعمدة المصفوفة A. M >= 0.
!>

P

!>          P هو عدد صحيح (INTEGER)
!>          عدد أعمدة المصفوفة B. P >= 0.
!>

A

!>          A هي مصفوفة مركبة (COMPLEX) ، البعد (LDA,M)
!>          المصفوفة N×M A.
!>

AF

!>          AF هي مصفوفة مركبة (COMPLEX) ، البعد (LDA,M)
!>

LDA

!>          LDA هو عدد صحيح (INTEGER)
!>          البعد الرائد للمصفوفات A, AF. LDA >= max(M,N).
!>

B

!>          B هي مصفوفة مركبة (COMPLEX) ، البعد (LDB,P)
!>          المصفوفة N×P A.
!>

BF

!>          BF هي مصفوفة مركبة (COMPLEX) ، البعد (LDB,P)
!>

LDB

!>          LDB هو عدد صحيح (INTEGER)
!>          البعد الرائد للمصفوفات B, BF. LDB >= max(P,N).
!>

D

!>          D هي مصفوفة مركبة (COMPLEX) ، البعد ( N )
!>          عند الإدخال، الجانب الأيسر من GLM.
!>

DF

!>          DF هي مصفوفة مركبة (COMPLEX) ، البعد ( N )
!>

X

!>          X هي مصفوفة مركبة (COMPLEX) ، البعد ( M )
!>          المتجه الحل X في مشكلة GLM.
!>

U

!>          U هي مصفوفة مركبة (COMPLEX) ، البعد ( P )
!>          المتجه الحل U في مشكلة GLM.
!>

WORK

!>          WORK هي مصفوفة مركبة (COMPLEX) ، البعد (LWORK)
!>

LWORK

!>          LWORK هو عدد صحيح (INTEGER)
!>          بعد المصفوفة WORK.
!>

RWORK

!>          RWORK هي مصفوفة حقيقية (REAL) ، البعد (M)
!>

RESULT

!>          RESULT هو عدد حقيقي (REAL)
!>          نسبة الاختبار:
!>                           norm( d - A*x - B*u )
!>            RESULT = -----------------------------------------
!>                     (norm(A)+norm(B))*(norm(x)+norm(u))*EPS
!>
