subroutine cgelqt (integer m, integer n, integer mb, complex, dimension( lda, * ) a, integer lda, complex, dimension( ldt, * ) t, integer ldt, complex, dimension( * ) work, integer info)

CGELQT

الغرض:

!>
!> CGELQT يحسب تحليل LQ لمصفوفة معقدة M-by-N تسمى A
!> باستخدام تمثيل WY المدمج لـ Q.
!>

المعلمات

M 
!>          M هو عدد صحيح (INTEGER)
!>          عدد صفوف المصفوفة A.  M >= 0.
!>

N

!>          N هو عدد صحيح (INTEGER)
!>          عدد أعمدة المصفوفة A.  N >= 0.
!>

MB

!>          MB هو عدد صحيح (INTEGER)
!>          حجم الكتلة (block size) المستخدم في QR المكتلي (blocked QR).  MIN(M,N) >= MB >= 1.
!>

A

!>          A هي مصفوفة مركبة (COMPLEX)، ذات بعد (LDA,N)
!>          عند الإدخال، المصفوفة A ذات الأبعاد M-by-N.
!>          عند الخروج، تحتوي العناصر الموجودة على القطر أو أسفله في المصفوفة
!>          على المصفوفة شبه المنحرفة السفلية L ذات الأبعاد M-by-MIN(M,N) (L
!>          هي مصفوفة مثلثة سفلية إذا كانت M <= N)؛ العناصر فوق القطر
!>          هي صفوف V.
!>

LDA

!>          LDA هو عدد صحيح (INTEGER)
!>          البعد الرائد (leading dimension) للمصفوفة A.  LDA >= max(1,M).
!>

T

!>          T هي مصفوفة مركبة (COMPLEX)، ذات بعد (LDT,MIN(M,N))
!>          عوامل الانعكاس القطاعية المثلثية العلوية المخزنة في شكل مدمج
!>          كسلسلة من القطاعات المثلثية العلوية.  راجع أدناه
!>          لمزيد من التفاصيل.
!>

LDT

!>          LDT هو عدد صحيح (INTEGER)
!>          البعد الرائد (leading dimension) للمصفوفة T.  LDT >= MB.
!>

WORK

!>          WORK هي مصفوفة مركبة (COMPLEX)، ذات بعد (MB*N)
!>

INFO

!>          INFO هو عدد صحيح (INTEGER)
!>          = 0:  اكتمال ناجح
!>          < 0:  إذا كان INFO = -i، فإن المعامل i كان له قيمة غير قانونية
!>

تفاصيل إضافية:

!>
!>  المصفوفة V تخزن عوامل الانعكاس الأولية H(i) في الصف i
!>  فوق القطر. على سبيل المثال، إذا كانت M=5 و N=3، فإن المصفوفة V هي
!>
!>               V = (  1  v1 v1 v1 v1 )
!>                   (     1  v2 v2 v2 )
!>                   (         1 v3 v3 )
!>
!>
!>  حيث تمثل الـ vi's المتجهات التي تحدد H(i)، والتي يتم إرجاعها
!>  في المصفوفة A.  الـ 1's على طول قطر V غير مخزنة في A.
!>  لتكن K=MIN(M,N).  عدد القطاعات هو B = ceiling(K/MB)، حيث كل
!>  قطاع له حجم MB باستثناء القطاع الأخير، الذي حجمه
!>  IB = K - (B-1)*MB.  لكل من القطاعات B، يتم حساب عامل
!>  انعكاس قطاعي مثلثي علوي: T1, T2, ..., TB.  يتم تخزين T's بحجم
!>  MB-by-MB (و IB-by-IB للقطاع الأخير) في المصفوفة T ذات الأبعاد
!>  MB-by-K على النحو التالي
!>
!>               T = (T1 T2 ... TB).
!>
