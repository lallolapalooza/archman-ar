recursive subroutine cgelqt3 (m, n, a, lda, t, ldt, info)

توثيق الوظيفة/الروتين الفرعي

recursive subroutine cgelqt3 (integer m, integer n, complex, dimension( lda, * ) a, integer lda, complex, dimension( ldt, * ) t, integer ldt, integer info)

CGELQT3

الغرض:

!>
!> CGELQT3 تحسب بشكل متكرر تحلل LQ لمصفوفة معقدة M-by-N
!> A، باستخدام تمثيل WY المدمج لـ Q.
!>
!> يستند إلى خوارزمية Elmroth و Gustavson،
!> IBM J. Res. Develop. المجلد 44 الرقم 4 يوليو 2000.
!>

المعلمات

M 
!>          M هو عدد صحيح (INTEGER)
!>          عدد صفوف المصفوفة A. M =< N.
!>

N

!>          N هو عدد صحيح (INTEGER)
!>          عدد أعمدة المصفوفة A. N >= 0.
!>

A

!>          A هي مصفوفة معقدة (COMPLEX)، البعد (LDA,N)
!>          عند الإدخال، العناصر على القطر و أسفله تحتوي على المصفوفة المثلثية
!>          السفلية L بحجم N-by-N؛ العناصر فوق القطر تحتوي على صفوف V.
!>          انظر أدناه لمزيد من التفاصيل.
!>

LDA

!>          LDA هو عدد صحيح (INTEGER)
!>          البعد الرائد للمصفوفة A. LDA >= max(1,M).
!>

T

!>          T هي مصفوفة معقدة (COMPLEX)، البعد (LDT,N)
!>          المصفوفة المثلثية العليا بحجم N-by-N للعامل المصفف الكتلي.
!>          العناصر على القطر و فوقه تحتوي على المصفف الكتلي T؛
!>          العناصر أسفل القطر غير مستخدمة.
!>          انظر أدناه لمزيد من التفاصيل.
!>

LDT

!>          LDT هو عدد صحيح (INTEGER)
!>          البعد الرائد للمصفوفة T. LDT >= max(1,N).
!>

INFO

!>          INFO هو عدد صحيح (INTEGER)
!>          = 0: خروج ناجح
!>          < 0: إذا كانت INFO = -i، فإن المعامل i كان له قيمة غير قانونية
!>

تفاصيل إضافية:

!>
!>  المصفوفة V تخزن المصففات الأولية H(i) في الصف i
!>  فوق القطر. على سبيل المثال، إذا كان M=5 و N=3، فإن المصفوفة V هي
!>
!>               V = (  1  v1 v1 v1 v1 )
!>                   (     1  v2 v2 v2 )
!>                   (     1  v3 v3 v3 )
!>
!>
!>  حيث تمثل الـ vi's المتجهات التي تعرف H(i)، والتي يتم إرجاعها
!>  في المصفوفة A.  الأرقام 1 على طول قطر V لا يتم تخزينها في A.  المصفف
!>  الكتلي H يُعطى بعد ذلك بواسطة
!>
!>               H = I - V * T * V**T
!>
!>  حيث V**T هو منقول V.
!>
!>  للحصول على تفاصيل الخوارزمية، انظر Elmroth و Gustavson (المذكور أعلاه).
!>
