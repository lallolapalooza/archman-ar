subroutine cgemqrt (character side, character trans, integer m, integer n, integer k, integer nb, complex, dimension( ldv, * ) v, integer ldv, complex, dimension( ldt, * ) t, integer ldt, complex, dimension( ldc, * ) c, integer ldc, complex, dimension( * ) work, integer info)

CGEMQRT

الغرض:

!>
!> CGEMQRT يقوم بالكتابة فوق المصفوفة المعقدة العامة M-by-N C بـ
!>
!>                 SIDE = 'L'     SIDE = 'R'
!> TRANS = 'N':      Q C            C Q
!> TRANS = 'C':    Q**H C            C Q**H
!>
!> حيث Q هي مصفوفة متعامدة معقدة معرفة كحاصل ضرب K
!> عاكسات أولية:
!>
!>       Q = H(1) H(2) . . . H(K) = I - V T V**H
!>
!> تم إنشاؤها باستخدام تمثيل WY المدمج كما تم إرجاعه بواسطة CGEQRT.
!>
!> Q من الرتبة M إذا كانت SIDE = 'L' ومن الرتبة N إذا كانت SIDE = 'R'.
!>

المعلمات:

SIDE 
!>          SIDE is CHARACTER*1
!>          = 'L': تطبيق Q أو Q**H من اليسار؛
!>          = 'R': تطبيق Q أو Q**H من اليمين.
!>

TRANS

!>          TRANS is CHARACTER*1
!>          = 'N':  لا يوجد تبديل، تطبيق Q؛
!>          = 'C':  تبديل مترافق، تطبيق Q**H.
!>

M

!>          M is INTEGER
!>          عدد صفوف المصفوفة C. M >= 0.
!>

N

!>          N is INTEGER
!>          عدد أعمدة المصفوفة C. N >= 0.
!>

K

!>          K is INTEGER
!>          عدد العاكسات الأولية التي يُعرّف حاصل ضربها
!>          المصفوفة Q.
!>          إذا كانت SIDE = 'L', M >= K >= 0؛
!>          إذا كانت SIDE = 'R', N >= K >= 0.
!>

NB

!>          NB is INTEGER
!>          حجم الكتلة المستخدم لتخزين T.  K >= NB >= 1.
!>          يجب أن تكون هذه نفس قيمة NB المستخدمة لإنشاء T
!>          في CGEQRT.
!>

V

!>          V is COMPLEX array, dimension (LDV,K)
!>          يجب أن يحتوي العمود i على المتجه الذي يعرّف
!>          العاكس الأولي H(i)، لـ i = 1,2,...,k، كما تم إرجاعه بواسطة
!>          CGEQRT في أول K أعمدة من وسيطة المصفوفة A الخاصة به.
!>

LDV

!>          LDV is INTEGER
!>          البعد الرائد للمصفوفة V.
!>          إذا كانت SIDE = 'L', LDA >= max(1,M)؛
!>          إذا كانت SIDE = 'R', LDA >= max(1,N).
!>

T

!>          T is COMPLEX array, dimension (LDT,K)
!>          عوامل المثلث العلوي للعواكس المجمعة
!>          كما تم إرجاعها بواسطة CGEQRT، مخزنة كمصفوفة NB-by-N.
!>

LDT

!>          LDT is INTEGER
!>          البعد الرائد للمصفوفة T.  LDT >= NB.
!>

C

!>          C is COMPLEX array, dimension (LDC,N)
!>          عند الدخول، المصفوفة C بحجم M-by-N.
!>          عند الخروج، يتم الكتابة فوق C بـ Q C، Q**H C، C Q**H أو C Q.
!>

LDC

!>          LDC is INTEGER
!>          البعد الرائد للمصفوفة C. LDC >= max(1,M).
!>

WORK

!>          WORK is COMPLEX array. The dimension of WORK is
!>           N*NB if SIDE = 'L', or  M*NB if SIDE = 'R'.
!>          (مصفوفة معقدة. بُعد WORK هو N*NB إذا كانت SIDE = 'L'، أو M*NB إذا كانت SIDE = 'R'.)
!>

INFO

!>          INFO is INTEGER
!>          = 0:  خروج ناجح
!>          < 0:  إذا كان INFO = -i، فإن الوسيط i كان له قيمة غير قانونية
!>
