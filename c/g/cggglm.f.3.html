توثيق الوظيفة/الإجراء الفرعي

subroutine cggglm (integer n, integer m, integer p, complex, dimension( lda, * ) a, integer lda, complex, dimension( ldb, * ) b, integer ldb, complex, dimension( * ) d, complex, dimension( * ) x, complex, dimension( * ) y, complex, dimension( * ) work, integer lwork, integer info)

CGGGLM

الغرض:

!>
!> CGGGLM يحل مشكلة نموذج غاوس-ماركوف الخطي العام (GLM):
!>
!>         تصغير || y ||_2   بشرط   d = A*x + B*y
!>             x
!>
!> حيث A مصفوفة N × M، و B مصفوفة N × P، و d متجه N معطى. يُفترض أن M <= N <= M+P، وأن
!>
!>            rank(A) = M    و    rank( A B ) = N.
!>
!> تحت هذه الافتراضات، تكون المعادلة المقيدة متسقة دائمًا، ويوجد حل فريد x وحل ذو معيار 2 أدنى y،
!> والذي يتم الحصول عليه باستخدام تحلل QR العام للمصفوفات (A, B) المعطاة بواسطة
!>
!>    A = Q*(R)،   B = Q*T*Z.
!>          (0)
!>
!> على وجه الخصوص، إذا كانت المصفوفة B مربعة وغير مفردة، فإن مشكلة
!> GLM تكافئ مشكلة المربعات الصغرى الخطية الموزونة التالية
!>
!>              تصغير || inv(B)*(d-A*x) ||_2
!>                  x
!>
!> حيث يشير inv(B) إلى معكوس B.
!>
!> يجب على المتصلين بهذا الإجراء الفرعي ملاحظة أن فحوصات التفرد/الركود التي يتم تنفيذها
!> في هذا الإجراء الفرعي بدائية. الإجراء الفرعي CTRTRS الذي يستدعيه هذا الإجراء الفرعي
!> يشير فقط إلى فشل بسبب التفرد إذا كانت المشكلة مفردة تمامًا.
!>
!> من الممكن أن يكون أحد (أو أكثر) العوامل المعنية بتحليل QR العام
!> للزوج (A، B) قريبًا بشكل دون المستوى من التفرد دون أن يشير هذا
!> الإجراء الفرعي إلى خطأ. قد تكون الحلول المحسوبة لمثل هذه المشاكل التي تكاد تكون راكدة
!> أقل دقة بسبب فقدان الدقة العددية.
!>
!>

المعلمات

N 
!>          N هو عدد صحيح (INTEGER)
!>          عدد صفوف المصفوفات A و B. N >= 0.
!>

M

!>          M هو عدد صحيح (INTEGER)
!>          عدد أعمدة المصفوفة A. 0 <= M <= N.
!>

P

!>          P هو عدد صحيح (INTEGER)
!>          عدد أعمدة المصفوفة B. P >= N-M.
!>

A

!>          A هو مصفوفة مركبة (COMPLEX array)، البعد (LDA,M)
!>          عند الدخول، مصفوفة A بحجم N×M.
!>          عند الخروج، الجزء العلوي المثلثي للمصفوفة A يحتوي على
!>          المصفوفة المثلثية العلوية R بحجم M×M.
!>

LDA

!>          LDA هو عدد صحيح (INTEGER)
!>          البعد الرائد للمصفوفة A. LDA >= max(1,N).
!>

B

!>          B هو مصفوفة مركبة (COMPLEX array)، البعد (LDB,P)
!>          عند الدخول، مصفوفة B بحجم N×P.
!>          عند الخروج، إذا كان N <= P، فإن الجزء العلوي المثلثي من الجزء الفرعي
!>          B(1:N,P-N+1:P) يحتوي على المصفوفة المثلثية العلوية T بحجم N×N؛
!>          إذا كان N > P، فإن العناصر على وفوق القطر الفرعي (N-P)
!>          تحتوي على المصفوفة شبه المنحرفة العلوية T بحجم N×P.
!>

LDB

!>          LDB هو عدد صحيح (INTEGER)
!>          البعد الرائد للمصفوفة B. LDB >= max(1,N).
!>

D

!>          D هي مصفوفة مركبة (COMPLEX array)، البعد (N)
!>          عند الدخول، D هو الجانب الأيسر لمعادلة GLM.
!>          عند الخروج، يتم تدمير D.
!>

X

!>          X هي مصفوفة مركبة (COMPLEX array)، البعد (M)
!>

Y

!>          Y هي مصفوفة مركبة (COMPLEX array)، البعد (P)
!>
!>          عند الخروج، X و Y هما حلا مشكلة GLM.
!>

WORK

!>          WORK هي مصفوفة مركبة (COMPLEX array)، البعد (MAX(1,LWORK))
!>          عند الخروج، إذا كان INFO = 0، فإن WORK(1) يعيد قيمة LWORK المثلى.
!>

LWORK

!>          LWORK هو عدد صحيح (INTEGER)
!>          بعد المصفوفة WORK. LWORK >= max(1,N+M+P).
!>          للحصول على أداء أمثل، LWORK >= M+min(N,P)+max(N,P)*NB،
!>          حيث NB هو حد أعلى لأحجام القطع المثلى لـ
!>          CGEQRF، CGERQF، CUNMQR و CUNMRQ.
!>
!>          إذا كان LWORK = -1، فيُفترض استعلام عن مساحة العمل؛ يقوم الإجراء الفرعي
!>          بحساب حجم مساحة العمل المثلى فقط، ويعيد
!>          هذه القيمة كالإدخال الأول لمصفوفة WORK، ولا يتم إصدار أي رسالة خطأ
!>          متعلقة بـ LWORK بواسطة XERBLA.
!>

INFO

!>          INFO هو عدد صحيح (INTEGER)
!>          = 0:  خروج ناجح.
!>          < 0:  إذا كان INFO = -i، فإن الوسيط i كان له قيمة غير قانونية.
!>          = 1:  العامل المثلثي العلوي R المرتبط بـ A في
!>                تحلل QR العام للزوج (A، B) مفرد تمامًا، بحيث rank(A) < M؛ لم يكن بالإمكان حساب
!>                حل المربعات الصغرى.
!>          = 2:  الجزء السفلي بحجم (N-M) × (N-M) من العامل شبه المنحرف العلوي
!>                T المرتبط بـ B في تحلل QR العام
!>                للزوج (A، B) مفرد تمامًا، بحيث rank( A B ) < N؛ لم يكن بالإمكان حساب
!>                حل المربعات الصغرى.
!>
