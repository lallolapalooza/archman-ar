subroutine cgesvx (character fact, character trans, integer n, integer nrhs, complex, dimension( lda, * ) a, integer lda, complex, dimension( ldaf, * ) af, integer ldaf, integer, dimension( * ) ipiv, character equed, real, dimension( * ) r, real, dimension( * ) c, complex, dimension( ldb, * ) b, integer ldb, complex, dimension( ldx, * ) x, integer ldx, real rcond, real, dimension( * ) ferr, real, dimension( * ) berr, complex, dimension( * ) work, real, dimension( * ) rwork, integer info)

CGESVX تحسب حل نظام المعادلات الخطية A * X = B لمصفوفات GE

الغرض:

!>
!> CGESVX تستخدم تفكيك LU لحساب حل نظام المعادلات الخطية المعقدة
!>    A * X = B,
!> حيث A مصفوفة N × N و X و B مصفوفتان N × NRHS.
!>
!> يتم أيضًا تقديم حدود الخطأ على الحل وتقدير الشرط.
!>

الوصف:

!>
!> يتم تنفيذ الخطوات التالية:
!>
!> 1. إذا كان FACT = 'E'، يتم حساب عوامل قياس حقيقية لمعادلة
!>    النظام:
!>       TRANS = 'N':  diag(R)*A*diag(C)     *inv(diag(C))*X = diag(R)*B
!>       TRANS = 'T': (diag(R)*A*diag(C))**T *inv(diag(R))*X = diag(C)*B
!>       TRANS = 'C': (diag(R)*A*diag(C))**H *inv(diag(R))*X = diag(C)*B
!>    سواء تمت معادلة النظام أم لا يعتمد على
!>    قياس المصفوفة A، ولكن إذا تم استخدام المعادلة، يتم
!>    الكتابة فوق A بواسطة diag(R)*A*diag(C) و B بواسطة diag(R)*B (إذا كان TRANS='N')
!>    أو diag(C)*B (إذا كان TRANS = 'T' أو 'C').
!>
!> 2. إذا كان FACT = 'N' أو 'E'، يتم استخدام تفكيك LU لتفكيك
!>    المصفوفة A (بعد المعادلة إذا كان FACT = 'E') على النحو التالي
!>       A = P * L * U,
!>    حيث P مصفوفة تبديل، L مصفوفة مثلثية سفلية وحدة، و U مصفوفة مثلثة علوية.
!>
!> 3. إذا كان بعض U(i,i)=0، بحيث تكون U منفردة تمامًا، فإن الروتين
!>    يعود مع INFO = i. وإلا، يتم استخدام الشكل المفكك لـ A لـ
!>    تقدير عدد شروط المصفوفة A. إذا كان
!>    مقلوب عدد الشروط أقل من الدقة الآلية،
!>    يتم إرجاع INFO = N+1 كتحذير، ولكن الروتين لا يزال يستمر لـ
!>    حل X وحساب حدود الخطأ كما هو موضح أدناه.
!>
!> 4. يتم حل نظام المعادلات لـ X باستخدام الشكل المفكك لـ
!>    A.
!>
!> 5. يتم تطبيق التحسين التكراري لتحسين الحل المحسوب
!>    المصفوفة وحساب حدود الخطأ وتقديرات الخطأ الخلفي لها.
!>
!> 6. إذا تم استخدام المعادلة، يتم ضرب المصفوفة X مسبقًا بـ
!>    diag(C) (إذا كان TRANS = 'N') أو diag(R) (إذا كان TRANS = 'T' أو 'C') بحيث
!>    تحل النظام الأصلي قبل المعادلة.
!>

المعلمات

FACT 
!>          FACT is CHARACTER*1
!>          يحدد ما إذا كان الشكل المفكك للمصفوفة A مقدمًا أم لا، وإذا لم يكن كذلك،
!>          ما إذا كان يجب معادلة المصفوفة A قبل تفكيكها.
!>          = 'F':  عند الإدخال، تحتوي AF و IPIV على الشكل المفكك لـ A.
!>                  إذا كان EQUED ليس 'N'، فقد تمت معادلة المصفوفة A
!>                  بعوامل القياس المعطاة بواسطة R و C.
!>                  لا يتم تعديل A و AF و IPIV.
!>          = 'N':  سيتم نسخ المصفوفة A إلى AF وتفكيكها.
!>          = 'E':  سيتم معادلة المصفوفة A إذا لزم الأمر، ثم
!>                  نسخها إلى AF وتفكيكها.
!>

TRANS

!>          TRANS is CHARACTER*1
!>          يحدد شكل نظام المعادلات:
!>          = 'N':  A * X = B     (بدون تبديل)
!>          = 'T':  A**T * X = B  (تبديل)
!>          = 'C':  A**H * X = B  (تبديل مترافق)
!>

N

!>          N is INTEGER
!>          عدد المعادلات الخطية، أي رتبة
!>          المصفوفة A.  N >= 0.
!>

NRHS

!>          NRHS is INTEGER
!>          عدد الأطراف اليمنى، أي عدد الأعمدة
!>          للمصفوفتين B و X.  NRHS >= 0.
!>

A

!>          A is COMPLEX array, dimension (LDA,N)
!>          عند الإدخال، المصفوفة A بحجم N×N.  إذا كان FACT = 'F' و EQUED
!>          ليس 'N'، فيجب أن تكون A قد تمت معادلتها بواسطة عوامل القياس
!>          في R و/أو C.  لا يتم تعديل A إذا كان FACT = 'F' أو
!>          'N'، أو إذا كان FACT = 'E' و EQUED = 'N' عند الخروج.
!>
!>          عند الخروج، إذا كان EQUED .ne. 'N'، يتم قياس A على النحو التالي:
!>          EQUED = 'R':  A := diag(R) * A
!>          EQUED = 'C':  A := A * diag(C)
!>          EQUED = 'B':  A := diag(R) * A * diag(C).
!>

LDA

!>          LDA is INTEGER
!>          البعد الرائد للمصفوفة A.  LDA >= max(1,N).
!>

AF

!>          AF is COMPLEX array, dimension (LDAF,N)
!>          إذا كان FACT = 'F'، فإن AF هي معامل إدخال وعند الإدخال
!>          تحتوي على العوامل L و U من تفكيك
!>          A = P*L*U كما تم حسابه بواسطة CGETRF.  إذا كان EQUED .ne. 'N'، فإن
!>          AF هو الشكل المفكك للمصفوفة A المعادلة.
!>
!>          إذا كان FACT = 'N'، فإن AF هي معامل إخراج وعند الخروج
!>          تعيد العوامل L و U من تفكيك A = P*L*U
!>          للمصفوفة الأصلية A.
!>
!>          إذا كان FACT = 'E'، فإن AF هي معامل إخراج وعند الخروج
!>          تعيد العوامل L و U من تفكيك A = P*L*U
!>          للمصفوفة المعادلة A (انظر وصف A لـ
!>          شكل المصفوفة المعادلة).
!>

LDAF

!>          LDAF is INTEGER
!>          البعد الرائد للمصفوفة AF.  LDAF >= max(1,N).
!>

IPIV

!>          IPIV is INTEGER array, dimension (N)
!>          إذا كان FACT = 'F'، فإن IPIV هي معامل إدخال وعند الإدخال
!>          تحتوي على مؤشرات المحور من تفكيك A = P*L*U
!>          كما تم حسابه بواسطة CGETRF؛ الصف i تم تبديله
!>          مع الصف IPIV(i).
!>
!>          إذا كان FACT = 'N'، فإن IPIV هي معامل إخراج وعند الخروج
!>          تحتوي على مؤشرات المحور من تفكيك A = P*L*U
!>          للمصفوفة الأصلية A.
!>
!>          إذا كان FACT = 'E'، فإن IPIV هي معامل إخراج وعند الخروج
!>          تحتوي على مؤشرات المحور من تفكيك A = P*L*U
!>          للمصفوفة المعادلة A.
!>

EQUED

!>          EQUED is CHARACTER*1
!>          يحدد شكل المعادلة التي تم إجراؤها.
!>          = 'N':  لا توجد معادلة (صحيح دائمًا إذا كان FACT = 'N').
!>          = 'R':  معادلة الصف، أي تم ضرب A مسبقًا بـ
!>                  diag(R).
!>          = 'C':  معادلة العمود، أي تم ضرب A لاحقًا
!>                  بـ diag(C).
!>          = 'B':  معادلة الصف والعمود، أي تم استبدال A بـ
!>                  diag(R) * A * diag(C).
!>          EQUED هو معامل إدخال إذا كان FACT = 'F'؛ بخلاف ذلك، فهو معامل
!>          إخراج.
!>

R

!>          R is REAL array, dimension (N)
!>          عوامل قياس الصف لـ A.  إذا كان EQUED = 'R' أو 'B'، يتم
!>          ضرب A على اليسار بـ diag(R)؛ إذا كان EQUED = 'N' أو 'C'، لا يتم
!>          الوصول إلى R.  R هو معامل إدخال إذا كان FACT = 'F'؛
!>          أخرى
