subroutine cgeequ (integer m, integer n, complex, dimension( lda, * ) a, integer lda, real, dimension( * ) r, real, dimension( * ) c, real rowcnd, real colcnd, real amax, integer info)

CGEEQU

الغرض:

!>
!> CGEEQU يحسب عوامل قياس الصفوف والأعمدة المقصود بها موازنة مصفوفة
!> M-by-N A وتقليل عدد شروطها.  R تعيد عوامل قياس الصفوف و C عوامل قياس
!> الأعمدة، مختارة لمحاولة جعل أكبر عنصر في كل صف وعمود في المصفوفة B
!> بالعناصر B(i,j)=R(i)*A(i,j)*C(j) له قيمة مطلقة 1.
!>
!> R(i) و C(j) مقيدة لتكون بين SMLNUM = أصغر رقم آمن و BIGNUM = أكبر رقم آمن.
!> استخدام عوامل القياس هذه لا يضمن تقليل عدد شروط A ولكنه يعمل بشكل جيد
!> في الممارسة العملية.
!>

المعلمات

M 
!>          M هو عدد صحيح
!>          عدد صفوف المصفوفة A.  M >= 0.
!>

N

!>          N هو عدد صحيح
!>          عدد أعمدة المصفوفة A.  N >= 0.
!>

A

!>          A هي مصفوفة مركبة، البعد (LDA,N)
!>          المصفوفة M-by-N التي سيتم حساب عوامل موازنتها.
!>

LDA

!>          LDA هو عدد صحيح
!>          البعد الرائد للمصفوفة A.  LDA >= max(1,M).
!>

R

!>          R هي مصفوفة حقيقية، البعد (M)
!>          إذا كان INFO = 0 أو INFO > M، تحتوي R على عوامل قياس الصفوف
!>          للمصفوفة A.
!>

C

!>          C هي مصفوفة حقيقية، البعد (N)
!>          إذا كان INFO = 0، تحتوي C على عوامل قياس الأعمدة للمصفوفة A.
!>

ROWCND

!>          ROWCND هو عدد حقيقي
!>          إذا كان INFO = 0 أو INFO > M، يحتوي ROWCND على نسبة أصغر R(i)
!>          إلى أكبر R(i).  إذا كان ROWCND >= 0.1 ولم يكن AMAX كبيرًا جدًا
!>          أو صغيرًا جدًا، فلا داعي للقياس باستخدام R.
!>

COLCND

!>          COLCND هو عدد حقيقي
!>          إذا كان INFO = 0، يحتوي COLCND على نسبة أصغر C(i) إلى أكبر C(i).
!>          إذا كان COLCND >= 0.1، فلا داعي للقياس باستخدام C.
!>

AMAX

!>          AMAX هو عدد حقيقي
!>          القيمة المطلقة لأكبر عنصر في المصفوفة.  إذا كان AMAX قريبًا جدًا
!>          من الفائض أو قريبًا جدًا من النقصان، فيجب قياس المصفوفة.
!>

INFO

!>          INFO هو عدد صحيح
!>          = 0:  انتهاء ناجح
!>          < 0:  إذا كان INFO = -i، فإن المعلمة i كانت لها قيمة غير قانونية
!>          > 0:  إذا كان INFO = i، وكان i
!>                <= M:  الصف i من A هو صفر تمامًا
!>                >  M:  العمود (i-M) من A هو صفر تمامًا
!>
