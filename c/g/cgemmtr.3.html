subroutine cgemmtr (character uplo, character transa, character transb, integer n, integer k, complex alpha, complex, dimension(lda,*) a, integer lda, complex, dimension(ldb,*) b, integer ldb, complex beta, complex, dimension(ldc,*) c, integer ldc)

CGEMMTR

الغرض:

!>
!> CGEMMTR  يقوم بإحدى عمليات ضرب المصفوفات:
!>
!>    C := alpha*op( A )*op( B ) + beta*C,
!>
!> حيث  op( X )  هي إحدى العمليات التالية:
!>
!>    op( X ) = X   أو   op( X ) = X**T,
!>
!> alpha و beta هما عددان صحيحان، و A, B, C هي مصفوفات، حيث op( A )
!> مصفوفة n × k، و op( B ) مصفوفة k × n، و C مصفوفة n × n.
!> وبذلك، فإن الروتين يصل فقط إلى الجزء العلوي أو السفلي من المصفوفة الناتجة C ويقوم بتحديثه. يمكن استخدام هذا السلوك إذا كان من المعروف أن المصفوفة الناتجة C هرمية أو متماثلة.
!>

المعاملات

UPLO 
!>          UPLO هو CHARACTER*1
!>           عند الإدخال، يحدد UPLO ما إذا كان سيتم الوصول إلى الجزء السفلي أو العلوي المثلثي من C وتحديثه.
!>
!>              UPLO = 'L' أو 'l'، يتم استخدام الجزء السفلي المثلثي من C.
!>
!>              UPLO = 'U' أو 'u'، يتم استخدام الجزء العلوي المثلثي من C.
!>

TRANSA

!>          TRANSA هو CHARACTER*1
!>           عند الإدخال، يحدد TRANSA شكل op( A ) المراد استخدامه في
!>           ضرب المصفوفات على النحو التالي:
!>
!>              TRANSA = 'N' أو 'n'،  op( A ) = A.
!>
!>              TRANSA = 'T' أو 't'،  op( A ) = A**T.
!>
!>              TRANSA = 'C' أو 'c'،  op( A ) = A**H.
!>

TRANSB

!>          TRANSB هو CHARACTER*1
!>           عند الإدخال، يحدد TRANSB شكل op( B ) المراد استخدامه في
!>           ضرب المصفوفات على النحو التالي:
!>
!>              TRANSB = 'N' أو 'n'،  op( B ) = B.
!>
!>              TRANSB = 'T' أو 't'،  op( B ) = B**T.
!>
!>              TRANSB = 'C' أو 'c'،  op( B ) = B**H.
!>

N

!>          N هو INTEGER
!>           عند الإدخال، يحدد N عدد الصفوف والأعمدة للمصفوفة
!>           C، وعدد أعمدة op(B)، وعدد صفوف op(A). يجب أن يكون N على الأقل صفرًا.
!>

K

!>          K هو INTEGER
!>           عند الإدخال، يحدد K عدد أعمدة المصفوفة
!>           op( A ) وعدد صفوف المصفوفة op( B ). يجب أن يكون K على الأقل صفرًا.
!>

ALPHA

!>          ALPHA هو COMPLEX.
!>           عند الإدخال، يحدد ALPHA العدد الصحيح alpha.
!>

A

!>          A هي مصفوفة COMPLEX، بأبعاد ( LDA, ka )، حيث ka
!>           هو k عندما TRANSA = 'N' أو 'n'، وهو n خلاف ذلك.
!>           قبل الإدخال مع TRANSA = 'N' أو 'n'، يجب أن يحتوي الجزء الرائد n × k
!>           من المصفوفة A على المصفوفة A، وإلا
!>           يجب أن يحتوي الجزء الرائد k × m من المصفوفة A على
!>           المصفوفة A.
!>

LDA

!>          LDA هو INTEGER
!>           عند الإدخال، يحدد LDA البعد الأول للمصفوفة A كما تم تعريفه
!>           في البرنامج (الروتيني الفرعي) المستدعي. عندما يكون TRANSA = 'N' أو 'n'،
!>           يجب أن يكون LDA على الأقل max( 1, n )، وإلا
!>           يجب أن يكون LDA على الأقل max( 1, k ).
!>

B

!>          B هي مصفوفة COMPLEX، بأبعاد ( LDB, kb )، حيث kb
!>           هو n عندما TRANSB = 'N' أو 'n'، وهو k خلاف ذلك.
!>           قبل الإدخال مع TRANSB = 'N' أو 'n'، يجب أن يحتوي الجزء الرائد k × n
!>           من المصفوفة B على المصفوفة B، وإلا
!>           يجب أن يحتوي الجزء الرائد n × k من المصفوفة B على
!>           المصفوفة B.
!>

LDB

!>          LDB هو INTEGER
!>           عند الإدخال، يحدد LDB البعد الأول للمصفوفة B كما تم تعريفه
!>           في البرنامج (الروتيني الفرعي) المستدعي. عندما يكون TRANSB = 'N' أو 'n'،
!>           يجب أن يكون LDB على الأقل max( 1, k )، وإلا
!>           يجب أن يكون LDB على الأقل max( 1, n ).
!>

BETA

!>          BETA هو COMPLEX.
!>           عند الإدخال، يحدد BETA العدد الصحيح beta. عندما يكون BETA
!>           مُدخلًا بصفر، لا يلزم تعيين C عند الإدخال.
!>

C

!>          C هي مصفوفة COMPLEX، بأبعاد ( LDC, N )
!>           قبل الإدخال، يجب أن يحتوي الجزء الرائد n × n
!>           من المصفوفة C على المصفوفة C، باستثناء عندما يكون beta صفرًا،
!>           وفي هذه الحالة لا يلزم تعيين C عند الإدخال.
!>           عند الخروج، يتم استبدال الجزء العلوي أو السفلي المثلثي للمصفوفة
!>           C بالمصفوفة n × n
!>           ( alpha*op( A )*op( B ) + beta*C ).
!>

LDC

!>          LDC هو INTEGER
!>           عند الإدخال، يحدد LDC البعد الأول للمصفوفة C كما تم تعريفه
!>           في البرنامج (الروتيني الفرعي) المستدعي. يجب أن يكون LDC على الأقل
!>           max( 1, n ).
!>
