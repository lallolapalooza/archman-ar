CGBRFSX

الغرض:

!>
!>    يحسّن CGBRFSX الحل المحسوب لنظام من المعادلات الخطية
!>    ويوفر حدود الخطأ وتقديرات الخطأ العكسي
!>    للحل.  بالإضافة إلى حد الخطأ المعياري، يوفر الكود
!>    حد الخطأ الأقصى للمكونات إذا كان ذلك ممكنًا.  انظر
!>    تعليقات ERR_BNDS_NORM و ERR_BNDS_COMP للحصول على تفاصيل
!>    حدود الخطأ.
!>
!>    قد يكون النظام الأصلي للمعادلات الخطية قد تمت موازنته
!>    قبل استدعاء هذا الروتين، كما هو موضح بالوسائط EQUED و R
!>    و C أدناه. في هذه الحالة، يتم إرجاع الحل وحدود الخطأ
!>    للنظام الأصلي غير الموازن.
!> 
!>     بعض الوسائط الاختيارية مجمعة في مصفوفة PARAMS.  تحدد هذه
!>     الإعدادات كيفية إجراء التحسين، ولكن غالبًا ما تكون
!>     القيم الافتراضية مقبولة.  إذا كانت القيم الافتراضية مقبولة، يمكن للمستخدمين
!>     تمرير NPARAMS = 0 مما يمنع الكود المصدري من الوصول
!>     إلى وسيطة PARAMS.
!>

الوسائط

TRANS 
!>          TRANS هو CHARACTER*1
!>     يحدد شكل نظام المعادلات:
!>       = 'N':  A * X = B     (بدون تبديل)
!>       = 'T':  A**T * X = B  (تبديل)
!>       = 'C':  A**H * X = B  (تبديل مرافق)
!>

EQUED

!>          EQUED هو CHARACTER*1
!>     يحدد شكل الموازنة التي تم إجراؤها على A
!>     قبل استدعاء هذا الروتين. هذا مطلوب لحساب
!>     الحل وحدود الخطأ بشكل صحيح.
!>       = 'N':  بدون موازنة
!>       = 'R':  موازنة الصفوف، أي أن A قد تم ضربها من اليسار بـ
!>               diag(R).
!>       = 'C':  موازنة الأعمدة، أي أن A قد تم ضربها من اليمين
!>               بـ diag(C).
!>       = 'B':  موازنة الصفوف والأعمدة معًا، أي أن A قد تم
!>               استبدالها بـ diag(R) * A * diag(C).
!>               تم تغيير الطرف الأيمن B وفقًا لذلك.
!>

N

!>          N هو INTEGER
!>     رتبة المصفوفة A.  N >= 0.
!>

KL

!>          KL هو INTEGER
!>     عدد الأقطار الفرعية ضمن نطاق A.  KL >= 0.
!>

KU

!>          KU هو INTEGER
!>     عدد الأقطار الفوقية ضمن نطاق A.  KU >= 0.
!>

NRHS

!>          NRHS هو INTEGER
!>     عدد الأطراف اليمنى، أي عدد أعمدة
!>     المصفوفتين B و X.  NRHS >= 0.
!>

AB

!>          AB هو مصفوفة COMPLEX، البعد (LDAB,N)
!>     المصفوفة الشريطية الأصلية A، مخزنة في الصفوف من 1 إلى KL+KU+1.
!>     يتم تخزين العمود j من A في العمود j من
!>     المصفوفة AB كما يلي:
!>     AB(ku+1+i-j,j) = A(i,j) لـ max(1,j-ku)<=i<=min(n,j+kl).
!>

LDAB

!>          LDAB هو INTEGER
!>     البعد الرئيسي للمصفوفة AB.  LDAB >= KL+KU+1.
!>

AFB

!>          AFB هو مصفوفة COMPLEX، البعد (LDAFB,N)
!>     تفاصيل تحليل LU للمصفوفة الشريطية A، كما
!>     تم حسابه بواسطة CGBTRF.  يتم تخزين U كمصفوفة شريطية
!>     مثلثية عليا مع KL+KU أقطار فوقية في الصفوف من 1 إلى KL+KU+1، و
!>     يتم تخزين المضاعفات المستخدمة أثناء التحليل في
!>     الصفوف من KL+KU+2 إلى 2*KL+KU+1.
!>

LDAFB

!>          LDAFB هو INTEGER
!>     البعد الرئيسي للمصفوفة AFB.  LDAFB >= 2*KL*KU+1.
!>

IPIV

!>          IPIV هو مصفوفة INTEGER، البعد (N)
!>     مؤشرات المحور من CGETRF؛ لـ 1<=i<=N، تم تبديل الصف i من
!>     المصفوفة مع الصف IPIV(i).
!>

R

!>          R هو مصفوفة REAL، البعد (N)
!>     عوامل مقياس الصفوف لـ A.  إذا كان EQUED = 'R' أو 'B'، يتم
!>     ضرب A من اليسار بـ diag(R)؛ إذا كان EQUED = 'N' أو 'C'، لا يتم
!>     الوصول إلى R.  R هي وسيطة إدخال إذا كان FACT = 'F'؛
!>     وإلا، فإن R هي وسيطة إخراج.  إذا كان FACT = 'F' و
!>     EQUED = 'R' أو 'B'، يجب أن يكون كل عنصر من R موجبًا.
!>     إذا كانت R إخراجًا، فإن كل عنصر من R هو قوة للأساس.
!>     إذا كانت R إدخالًا، يجب أن يكون كل عنصر من R قوة للأساس
!>     لضمان حل وتقديرات خطأ موثوقة. القياس بقوى
!>     الأساس لا يسبب أخطاء تقريب إلا إذا كانت
!>     النتيجة تحت التدفق أو فوق التدفق. أخطاء التقريب أثناء القياس
!>     تؤدي إلى التحسين بمصفوفة غير مكافئة للمصفوفة
!>     المدخلة، مما ينتج تقديرات خطأ قد لا تكون
!>     موثوقة.
!>

C

!>          C هو مصفوفة REAL، البعد (N)
!>     عوامل مقياس الأعمدة لـ A.  إذا كان EQUED = 'C' أو 'B'، يتم
!>     ضرب A من اليمين بـ diag(C)؛ إذا كان EQUED = 'N' أو 'R'، لا يتم
!>     الوصول إلى C.  C هي وسيطة إدخال إذا كان FACT = 'F'؛
!>     وإلا، فإن C هي وسيطة إخراج.  إذا كان FACT = 'F' و
!>     EQUED = 'C' أو 'B'، يجب أن يكون كل عنصر من C موجبًا.
!>     إذا كانت C إخراجًا، فإن كل عنصر من C هو قوة للأساس.
!>     إذا كانت C إدخالًا، يجب أن يكون كل عنصر من C قوة للأساس
!>     لضمان حل وتقديرات خطأ موثوقة. القياس بقوى
!>     الأساس لا يسبب أخطاء تقريب إلا إذا كانت
!>     النتيجة تحت التدفق أو فوق التدفق. أخطاء التقريب أثناء القياس
!>     تؤدي إلى التحسين بمصفوفة غير مكافئة للمصفوفة
!>     المدخلة، مما ينتج تقديرات خطأ قد لا تكون
!>     موثوقة.
!>

B

!>          B هو مصفوفة COMPLEX، البعد (LDB,NRHS)
!>     مصفوفة الطرف الأيمن B.
!>

LDB

!>          LDB هو INTEGER
!>     البعد الرئيسي للمصفوفة B.  LDB >= max(1,N).
!>

X

!>          X هو مصفوفة COMPLEX، البعد (LDX,NRHS)
!>     عند الإدخال، مصفوفة الحل X، كما تم حسابها بواسطة CGETRS.
!>     عند الإخراج، مصفوفة الحل المحسّنة X.
!>

LDX

!>          LDX هو INTEGER
!>     البعد الرئيسي للمصفوفة X.  LDX >= max(1,N).
!>

RCOND

!>          RCOND هو REAL
!> مقلوب رقم الشرط المقيس.  هذا تقدير لمقلوب
!> رقم شرط Skeel للمصفوفة A بعد
!> الموازنة (إذا تمت).  إذا كان هذا أقل من دقة
!> الآلة (على وجه الخصوص، إذا كان صفرًا)، فإن المصفوفة مفردة
!> بالنسبة لدقة العمل.  لاحظ أن الخطأ قد يظلصغير جدًا حتى
!>     إذا كان هذا الرقم صغيرًا جدًا وظهرت المصفوفة سيئة
!>     الشرط.
!>

BERR

!>          BERR هو مصفوفة REAL، البُعد (NRHS)
!>     خطأ خلفي نسبي حسب المكوّن. هذا هو
!>     الخطأ الخلفي النسبي حسب المكوّن لكل متجه حل X(j)
!>     (أي أصغر تغيير نسبي في أي عنصر من A أو B يجعل
!>     X(j) حلًا دقيقًا).
!>

N_ERR_BNDS

!>          N_ERR_BNDS هو INTEGER
!>     عدد حدود الخطأ التي تُعاد لكل طرف أيمن
!>     ولكل نوع (حسب المعيار أو حسب المكوّن). انظر ERR_BNDS_NORM و
!>     ERR_BNDS_COMP أدناه.
!>

ERR_BNDS_NORM

!>          ERR_BNDS_NORM هو مصفوفة REAL، البُعد (NRHS, N_ERR_BNDS)
!>     لكل طرف أيمن، تحتوي هذه المصفوفة على معلومات حول
!>     حدود خطأ مختلفة وأرقام شرط مقابلة لـ
!>     الخطأ النسبي حسب المعيار، والذي يُعرَّف كما يلي:
!>
!>     الخطأ النسبي حسب المعيار في متجه الحل رقم i:
!>             max_j (abs(XTRUE(j,i) - X(j,i)))
!>            ------------------------------
!>                  max_j abs(X(j,i))
!>
!>     تُفهرَس المصفوفة حسب نوع معلومات الخطأ كما هو موصوف
!>     أدناه. يوجد حاليًا ما يصل إلى ثلاث قطع من المعلومات
!>     تُعاد.
!>
!>     الفهرس الأول في ERR_BNDS_NORM(i,:) يقابل الطرف الأيمن
!>     رقم i.
!>
!>     الفهرس الثاني في ERR_BNDS_NORM(:,err) يحتوي على الحقول
!>     الثلاثة التالية:
!>     err = 1  منطقي. ثق بالإجابة إذا كان
!>              رقم الشرط المتبادل أقل من العتبة
!>              sqrt(n) * slamch('Epsilon').
!>
!>     err = 2  حد الخطأ: الخطأ الأمامي المقدَّر،
!>              على الأرجح ضمن معامل 10 من الخطأ الحقيقي
!>              طالما أن الإدخال التالي أكبر من العتبة
!>              sqrt(n) * slamch('Epsilon'). يجب الوثوق بحد الخطأ هذا فقط
!>              إذا كان المنطقي السابق صحيحًا.
!>
!>     err = 3  رقم الشرط المتبادل: رقم الشرط المتبادل
!>              المقدَّر حسب المعيار. يُقارَن مع العتبة
!>              sqrt(n) * slamch('Epsilon') لتحديد ما إذا كان تقدير
!>              الخطأ هو . أرقام الشرط المتبادل
!>              هذه هي 1 / (norm(Z^{-1},inf) * norm(Z,inf)) لبعض
!>              المصفوفات Z المقيَّسة بشكل مناسب.
!>              لنفترض Z = S*A، حيث S يقيس كل صف بقوة من
!>              الأساس بحيث تكون جميع مجاميع الصفوف المطلقة لـ Z تقريبًا 1.
!>
!>     انظر Lapack Working Note 165 لمزيد من التفاصيل والتحذيرات
!>     الإضافية.
!>

ERR_BNDS_COMP

!>          ERR_BNDS_COMP هو مصفوفة REAL، البُعد (NRHS, N_ERR_BNDS)
!>     لكل طرف أيمن، تحتوي هذه المصفوفة على معلومات حول
!>     حدود خطأ مختلفة وأرقام شرط مقابلة لـ
!>     الخطأ النسبي حسب المكوّن، والذي يُعرَّف كما يلي:
!>
!>     الخطأ النسبي حسب المكوّن في متجه الحل رقم i:
!>                    abs(XTRUE(j,i) - X(j,i))
!>             max_j ----------------------
!>                         abs(X(j,i))
!>
!>     تُفهرَس المصفوفة حسب الطرف الأيمن i (الذي يعتمد عليه
!>     الخطأ النسبي حسب المكوّن)، ونوع معلومات الخطأ
!>     كما هو موصوف أدناه. يوجد حاليًا ما يصل إلى ثلاث
!>     قطع من المعلومات تُعاد لكل طرف أيمن. إذا
!>     لم تُطلب الدقة حسب المكوّن (PARAMS(3) = 0.0)، فإن
!>     ERR_BNDS_COMP لا يُ accessed. إذا كان N_ERR_BNDS < 3، فإنه على الأكثر
!>     تُعاد الإدخالات الأولى (:,N_ERR_BNDS).
!>
!>     الفهرس الأول في ERR_BNDS_COMP(i,:) يقابل الطرف الأيمن
!>     رقم i.
!>
!>     الفهرس الثاني في ERR_BNDS_COMP(:,err) يحتوي على الحقول
!>     الثلاثة التالية:
!>     err = 1  منطقي. ثق بالإجابة إذا كان
!>              رقم الشرط المتبادل أقل من العتبة
!>              sqrt(n) * slamch('Epsilon').
!>
!>     err = 2  حد الخطأ: الخطأ الأمامي المقدَّر،
!>              على الأرجح ضمن معامل 10 من الخطأ الحقيقي
!>              طالما أن الإدخال التالي أكبر من العتبة
!>              sqrt(n) * slamch('Epsilon'). يجب الوثوق بحد الخطأ هذا فقط
!>              إذا كان المنطقي السابق صحيحًا.
!>
!>     err = 3  رقم الشرط المتبادل: رقم الشرط المتبادل
!>              المقدَّر حسب المكوّن. يُقارَن مع العتبة
!>              sqrt(n) * slamch('Epsilon') لتحديد ما إذا كان تقدير
!>              الخطأ هو . أرقام الشرط المتبادل
!>              هذه هي 1 / (norm(Z^{-1},inf) * norm(Z,inf)) لبعض
!>              المصفوفات Z المقيَّسة بشكل مناسب.
!>              لنفترض Z = S*(A*diag(x))، حيث x هو الحل للطرف الأيمن
!>              الحالي و S يقيس كل صف من
!>              A*diag(x) بقوة من الأساس بحيث تكون جميع مجاميع الصفوف المطلقة
!>              لـ Z تقريبًا 1.
!>
!>     انظر Lapack Working Note 165 لمزيد من التفاصيل والتحذيرات
!>     الإضافية.
!>

NPARAMS

!>          NPARAMS هو INTEGER
!>     يحدد عدد المعاملات المضبوطة في PARAMS. إذا كان <= 0، فإن
!>     مصفوفة PARAMS لا يُشار إليها أبدًا وتُستخدم القيم الافتراضية.
!>

PARAMS

!>          PARAMS هو مصفوفة REAL، البُعد NPARAMS
!>     يحدد معاملات الخوارزمية. إذا كان إدخال < 0.0، فإن
!>     ذلك الإدخال سيُملأ بالقيمة الافتراضية المستخدمة لذلك
!>     المعامل. تُ accessed المواضع حتى NPARAMS فقط؛ تُستخدم
!>     القيم الافتراضية للمعاملات ذات الأرقام الأعلى.
!>
!>       PARAMS(LA_LINRX_ITREF_I = 1) : ما إذا كان سيتم إجراء تحسين
!>            تكراري أم لا.
!>         الافتراضي: 1.0
!>            = 0.0:  لا يُجرى تحسين، ولا تُحسب حدود
!>                    الخطأ.
!>            = 1.0:  استخدم خوارزمية التحسين ذات الدقة المزدوجة،
!>                    ربما مع حسابات أحادية مزدوجة إذا كانت
!>                    بيئة الترجمة لا تدعم DOUBLE
!>                    PRECISION.
!>              (القيم الأخرى محجوزة للاستخدام المستقبلي)
!>
!>       PARAMS(LA_LINRX_ITHRESH_I = 2) : الحد الأقصى لعدد حسابات
!>            البواقي المسموح بها للتحسين.
!>         الافتراضي: 10
!>         عدواني: اضبطه على 100 للسماح بالتقارب باستخدام
!>                     تحليلات تقريبية أو تحليلات غير LU. إذا
!>                     استخدم التحليل تقنية غير
!>                     الحذف الغاوسي، فإن الضمانات في
!>                     err_bnds_norm و err_bnds_comp قد لا تعود
!>                     موثوقة.
!>
!>       PARAMS(LA_LINRX_CWISE_I = 3) : علم يحدد ما إذا كان الكود
!>            سيحاول إيجاد حل بخطأ نسبي صغير
!>            حسب المكوّن في خوارزمية الدقة المزدوجة. الموجب
!>            صحيح، 0.0 خطأ.
!>         الافتراضي: 1.0 (محاولة التقارب حسب المكوّن)
!>

WORK

!>          WORK هو مصفوفة COMPLEX، البُعد (2*N)
!>

RWORK

!>          RWORK هو مصفوفة REAL، البُعد (2*N)
!>

INFO

!>          INFO هو INTEGER
!>       = 0:  خروج ناجح. الحل لكل طرف أيمن
!>         مضمون.
!>       < 0:  إذا كان INFO = -i، فإن الوسيطة رقم i لها قيمة غير قانونية
!>       > 0 و <= N:  U(INFO,INFO) صفر تمامًا. اكتمل
!>         التحليل، لكن العامل U مفرد تمامًا، لذا
!>         لم يمكن حساب الحل وحدود الخطأ. يُعاد RCOND = 0
!>         .
!>       = N+J: الحل المقابل للطرف الأيمن رقم J
!>         غير مضمون. الحلول المقابلة لأطراف أيمن
!>         أخرى K حيث K > J قد لا تكون مضمونة أيضًا، لكن
!>         يُبلَّغ فقط عن أول طرف أيمن من هذا القبيل. إذا لم يُطلب
!>         خطأ صغير حسب المكوّن (PARAMS(3) = 0.0) فإن
!>         الطرف الأيمن رقم J هو الأول بحد خطأ
!>         حسب المعيار غير مضمون (أصغر J بحيث
!>         ERR_BNDS_NORM(J,1) = 0.0). افتراضيًا (PARAMS(3) = 1.0)
!>         الطرف الأيمن رقم J هو الأول بحد خطأ إما حسب المعيار أو
!>         حسب المكوّن غير مضمون (أصغر
!>         J بحيث إما ERR_BNDS_NORM(J,1) = 0.0 أو
!>         ERR_BNDS_COMP(J,1) = 0.0). انظر تعريف
!>         ERR_BNDS_NORM(:,1) و ERR_BNDS_COMP(:,1). للحصول على معلومات
