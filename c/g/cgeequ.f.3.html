subroutine cgeequ (integer m, integer n, complex, dimension( lda, * ) a, integer lda, real, dimension( * ) r, real, dimension( * ) c, real rowcnd, real colcnd, real amax, integer info)

CGEEQU

الغرض:

!>
!> CGEEQU يحسب عوامل القياس للصف والأعمدة المخصصة لمعادلة المصفوفة M-by-N A
!> وتقليل عدد حالتها. تعيد R عوامل القياس للصف و C عوامل القياس للأعمدة،
!> مختارة لمحاولة جعل أكبر عنصر في كل صف وعمود من المصفوفة B ذات العناصر
!> B(i,j)=R(i)*A(i,j)*C(j) لها قيمة مطلقة 1.
!>
!> يتم تقييد R(i) و C(j) لتكون بين SMLNUM = أصغر رقم آمن و BIGNUM = أكبر رقم آمن.
!> استخدام عوامل القياس هذه لا يضمن تقليل عدد حالة A ولكنه يعمل بشكل جيد
!> في الممارسة العملية.
!>

المعلمات

M 
!>          M هو عدد صحيح (INTEGER)
!>          عدد صفوف المصفوفة A.  M >= 0.
!>

N

!>          N هو عدد صحيح (INTEGER)
!>          عدد أعمدة المصفوفة A.  N >= 0.
!>

A

!>          A هي مصفوفة معقدة (COMPLEX array)، ذات البعد (LDA,N)
!>          المصفوفة M-by-N التي يجب حساب عوامل معادلتها.
!>

LDA

!>          LDA هو عدد صحيح (INTEGER)
!>          البعد الرئيسي للمصفوفة A.  LDA >= max(1,M).
!>

R

!>          R هي مصفوفة حقيقية (REAL array)، ذات البعد (M)
!>          إذا كان INFO = 0 أو INFO > M، تحتوي R على عوامل قياس الصفوف لـ A.
!>

C

!>          C هي مصفوفة حقيقية (REAL array)، ذات البعد (N)
!>          إذا كان INFO = 0، تحتوي C على عوامل قياس الأعمدة لـ A.
!>

ROWCND

!>          ROWCND هي قيمة حقيقية (REAL)
!>          إذا كان INFO = 0 أو INFO > M، تحتوي ROWCND على نسبة أصغر R(i)
!>          إلى أكبر R(i).  إذا كان ROWCND >= 0.1 وكان AMAX ليس كبيرًا جدًا
!>          ولا صغيرًا جدًا، فلا فائدة من القياس بواسطة R.
!>

COLCND

!>          COLCND هي قيمة حقيقية (REAL)
!>          إذا كان INFO = 0، تحتوي COLCND على نسبة أصغر C(i) إلى أكبر C(i).
!>          إذا كان COLCND >= 0.1، فلا فائدة من القياس بواسطة C.
!>

AMAX

!>          AMAX هي قيمة حقيقية (REAL)
!>          القيمة المطلقة لأكبر عنصر في المصفوفة.  إذا كان AMAX قريبًا جدًا
!>          من تجاوز الحد الأقصى أو قريبًا جدًا من تجاوز الحد الأدنى،
!>          يجب قياس المصفوفة.
!>

INFO

!>          INFO هو عدد صحيح (INTEGER)
!>          = 0:  انتهاء ناجح
!>          < 0:  إذا كان INFO = -i، فإن المعلمة i كانت لها قيمة غير قانونية
!>          > 0:  إذا كان INFO = i، و i هو
!>                <= M:  الصف i من A هو صفر تمامًا
!>                >  M:  العمود (i-M) من A هو صفر تمامًا
!>
