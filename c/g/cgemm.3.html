subroutine cgemm (character transa, character transb, integer m, integer n, integer k, complex alpha, complex, dimension(lda,*) a, integer lda, complex, dimension(ldb,*) b, integer ldb, complex beta, complex, dimension(ldc,*) c, integer ldc)

CGEMM

الغرض:

!>
!> CGEMM  تقوم بإحدى عمليات المصفوفة-المصفوفة
!>
!>    C := alpha*op( A )*op( B ) + beta*C,
!>
!> حيث  op( X ) هي واحدة من
!>
!>    op( X ) = X   أو   op( X ) = X**T   أو   op( X ) = X**H,
!>
!> alpha و beta عددان سكالر، والمصفوفات A و B و C، مع كون op( A )
!> مصفوفة m × k، و op( B ) مصفوفة k × n، و C مصفوفة m × n.
!>

المعلمات

TRANSA 
!>          TRANSA هو CHARACTER*1
!>           عند الإدخال، تحدد TRANSA شكل op( A ) المراد استخدامه في
!>           ضرب المصفوفات على النحو التالي:
!>
!>              TRANSA = 'N' أو 'n'،  op( A ) = A.
!>
!>              TRANSA = 'T' أو 't'،  op( A ) = A**T.
!>
!>              TRANSA = 'C' أو 'c'،  op( A ) = A**H.
!>

TRANSB

!>          TRANSB هو CHARACTER*1
!>           عند الإدخال، تحدد TRANSB شكل op( B ) المراد استخدامه في
!>           ضرب المصفوفات على النحو التالي:
!>
!>              TRANSB = 'N' أو 'n'،  op( B ) = B.
!>
!>              TRANSB = 'T' أو 't'،  op( B ) = B**T.
!>
!>              TRANSB = 'C' أو 'c'،  op( B ) = B**H.
!>

M

!>          M هو INTEGER
!>           عند الإدخال، تحدد M  عدد الصفوف في المصفوفة
!>           op( A ) والمصفوفة C. يجب أن تكون M على الأقل صفر.
!>

N

!>          N هو INTEGER
!>           عند الإدخال، تحدد N  عدد الأعمدة في المصفوفة
!>           op( B ) وعدد الأعمدة في المصفوفة C. يجب أن تكون N
!>           على الأقل صفر.
!>

K

!>          K هو INTEGER
!>           عند الإدخال، تحدد K  عدد الأعمدة في المصفوفة
!>           op( A ) وعدد الصفوف في المصفوفة op( B ). يجب أن تكون K
!>           على الأقل صفر.
!>

ALPHA

!>          ALPHA هو COMPLEX
!>           عند الإدخال، يحدد ALPHA العدد السكالر alpha.
!>

A

!>          A هي مصفوفة COMPLEX، الأبعاد ( LDA, ka )، حيث ka
!>           هو k  عندما تكون  TRANSA = 'N' أو 'n'،  و m  بخلاف ذلك.
!>           قبل الإدخال مع  TRANSA = 'N' أو 'n'،  يجب أن يحتوي الجزء الرئيسي
!>           m × k  من المصفوفة A  على المصفوفة A، بخلاف ذلك
!>           يجب أن يحتوي الجزء الرئيسي k × m  من المصفوفة A  على
!>           المصفوفة A.
!>

LDA

!>          LDA هو INTEGER
!>           عند الإدخال، تحدد LDA البعد الأول لـ A كما تم تعريفه
!>           في البرنامج (الفرعي) المستدعي. عندما تكون  TRANSA = 'N' أو 'n'،
!>           يجب أن يكون LDA على الأقل  max( 1, m )، بخلاف ذلك
!>           يجب أن يكون LDA على الأقل  max( 1, k ).
!>

B

!>          B هي مصفوفة COMPLEX، الأبعاد ( LDB, kb )، حيث kb
!>           هو n  عندما تكون  TRANSB = 'N' أو 'n'،  و k  بخلاف ذلك.
!>           قبل الإدخال مع  TRANSB = 'N' أو 'n'،  يجب أن يحتوي الجزء الرئيسي
!>           k × n  من المصفوفة B  على المصفوفة B، بخلاف ذلك
!>           يجب أن يحتوي الجزء الرئيسي n × k  من المصفوفة B  على
!>           المصفوفة B.
!>

LDB

!>          LDB هو INTEGER
!>           عند الإدخال، تحدد LDB البعد الأول لـ B كما تم تعريفه
!>           في البرنامج (الفرعي) المستدعي. عندما تكون  TRANSB = 'N' أو 'n'،
!>           يجب أن يكون LDB على الأقل  max( 1, k )، بخلاف ذلك
!>           يجب أن يكون LDB على الأقل  max( 1, n ).
!>

BETA

!>          BETA هو COMPLEX
!>           عند الإدخال، يحدد  BETA  العدد السكالر beta.  عندما يتم
!>           توفير BETA  كصفر، فلا يلزم تعيين C عند الإدخال.
!>

C

!>          C هي مصفوفة COMPLEX، الأبعاد ( LDC, N )
!>           قبل الإدخال، يجب أن يحتوي الجزء الرئيسي m × n  من المصفوفة C  على
!>           المصفوفة C،  باستثناء عندما يكون beta صفرًا، في هذه الحالة
!>           لا يلزم تعيين C عند الإدخال.
!>           عند الخروج، يتم استبدال المصفوفة C  بالمصفوفة m × n
!>           ( alpha*op( A )*op( B ) + beta*C ).
!>

LDC

!>          LDC هو INTEGER
!>           عند الإدخال، تحدد LDC البعد الأول لـ C كما تم تعريفه
!>           في البرنامج (الفرعي) المستدعي. يجب أن يكون LDC  على الأقل
!>           max( 1, m ).
!>
