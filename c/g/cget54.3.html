subroutine cget54 (integer n, complex, dimension( lda, * ) a, integer lda, complex, dimension( ldb, * ) b, integer ldb, complex, dimension( lds, * ) s, integer lds, complex, dimension( ldt, * ) t, integer ldt, complex, dimension( ldu, * ) u, integer ldu, complex, dimension( ldv, * ) v, integer ldv, complex, dimension( * ) work, real result)

CGET54

الغرض:

!>
!> CGET54 يتحقق من التحلل المعمم للشكل
!>
!>          A = U*S*V'  و B = U*T* V'
!>
!> حيث ' تعني منقول مترافق و U و V وحدويتان.
!>
!> على وجه التحديد،
!>
!>   RESULT = ||( A - U*S*V', B - U*T*V' )|| / (||( A, B )||*n*ulp )
!>

المعلمات

N 
!>          N هو عدد صحيح (INTEGER)
!>          حجم المصفوفة. إذا كان صفرًا، فإن SGET54 لا تفعل شيئًا.
!>          يجب أن يكون أكبر من أو يساوي الصفر.
!>

A

!>          A هي مصفوفة مركبة (COMPLEX)، البعد (LDA, N)
!>          المصفوفة الأصلية (غير المفككة) A.
!>

LDA

!>          LDA هو عدد صحيح (INTEGER)
!>          البعد الرئيسي للمصفوفة A. يجب أن يكون على الأقل 1
!>          وعلى الأقل N.
!>

B

!>          B هي مصفوفة مركبة (COMPLEX)، البعد (LDB, N)
!>          المصفوفة الأصلية (غير المفككة) B.
!>

LDB

!>          LDB هو عدد صحيح (INTEGER)
!>          البعد الرئيسي للمصفوفة B. يجب أن يكون على الأقل 1
!>          وعلى الأقل N.
!>

S

!>          S هي مصفوفة مركبة (COMPLEX)، البعد (LDS, N)
!>          المصفوفة المفككة S.
!>

LDS

!>          LDS هو عدد صحيح (INTEGER)
!>          البعد الرئيسي للمصفوفة S. يجب أن يكون على الأقل 1
!>          وعلى الأقل N.
!>

T

!>          T هي مصفوفة مركبة (COMPLEX)، البعد (LDT, N)
!>          المصفوفة المفككة T.
!>

LDT

!>          LDT هو عدد صحيح (INTEGER)
!>          البعد الرئيسي للمصفوفة T. يجب أن يكون على الأقل 1
!>          وعلى الأقل N.
!>

U

!>          U هي مصفوفة مركبة (COMPLEX)، البعد (LDU, N)
!>          المصفوفة المتعامدة على الجانب الأيسر في
!>          التحلل.
!>

LDU

!>          LDU هو عدد صحيح (INTEGER)
!>          البعد الرئيسي للمصفوفة U. يجب أن يكون LDU على الأقل N
!>          وعلى الأقل 1.
!>

V

!>          V هي مصفوفة مركبة (COMPLEX)، البعد (LDV, N)
!>          المصفوفة المتعامدة على الجانب الأيسر في
!>          التحلل.
!>

LDV

!>          LDV هو عدد صحيح (INTEGER)
!>          البعد الرئيسي للمصفوفة V. يجب أن يكون LDV على الأقل N
!>          وعلى الأقل 1.
!>

WORK

!>          WORK هي مصفوفة مركبة (COMPLEX)، البعد (3*N**2)
!>

RESULT

!>          RESULT هو عدد حقيقي (REAL)
!>          القيمة RESULT، وهي حاليًا محدودة بـ 1/ulp، لتجنب
!>          الفيضان. يتم تحديد الأخطاء بواسطة RESULT=10/ulp.
!>
