subroutine cgesvx (character fact, character trans, integer n, integer nrhs, complex, dimension( lda, * ) a, integer lda, complex, dimension( ldaf, * ) af, integer ldaf, integer, dimension( * ) ipiv, character equed, real, dimension( * ) r, real, dimension( * ) c, complex, dimension( ldb, * ) b, integer ldb, complex, dimension( ldx, * ) x, integer ldx, real rcond, real, dimension( * ) ferr, real, dimension( * ) berr, complex, dimension( * ) work, real, dimension( * ) rwork, integer info)

CGESVX يحسب الحل لنظام المعادلات الخطية A * X = B لمصفوفات GE

الغرض:

!>
!> يستخدم CGESVX تحليل LU لحساب الحل لنظام خطي معقد
!>    A * X = B،
!> حيث A هي مصفوفة N×N و X و B هي مصفوفات N×NRHS.
!>
!> يتم أيضًا توفير حدود الخطأ للحل وتقدير للحالة.
!>

الوصف:

!>
!> يتم تنفيذ الخطوات التالية:
!>
!> 1. إذا كان FACT = 'E'، يتم حساب عوامل قياس حقيقية لموازنة
!>    النظام:
!>       TRANS = 'N':  diag(R)*A*diag(C)     *inv(diag(C))*X = diag(R)*B
!>       TRANS = 'T': (diag(R)*A*diag(C))**T *inv(diag(R))*X = diag(C)*B
!>       TRANS = 'C': (diag(R)*A*diag(C))**H *inv(diag(R))*X = diag(C)*B
!>    سيعتمد ما إذا كان النظام سيعادل أم لا على
!>    قياس المصفوفة A، ولكن إذا تم استخدام الموازنة، فسيتم
!>    استبدال A بـ diag(R)*A*diag(C) و B بـ diag(R)*B (إذا كان TRANS='N')
!>    أو diag(C)*B (إذا كان TRANS = 'T' أو 'C').
!>
!> 2. إذا كان FACT = 'N' أو 'E'، يتم استخدام تحليل LU لتحليل
!>    المصفوفة A (بعد الموازنة إذا كان FACT = 'E') كـ
!>       A = P * L * U،
!>    حيث P هي مصفوفة تبديل، L هي مصفوفة ثلاثية سفلية وحدوية، و U هي مصفوفة ثلاثية علوية.
!>
!> 3. إذا كان بعض U(i,i)=0، بحيث تكون U مفردة تمامًا، فإن الإجراء
!>    يعود بـ INFO = i. بخلاف ذلك، يتم استخدام الشكل المحلل لـ A
!>    لتقدير رقم حالة المصفوفة A.  إذا كان
!>    مقلوب رقم الحالة أقل من دقة الآلة،
!>    يتم إرجاع INFO = N+1 كتحذير، ولكن الإجراء لا يزال يستمر
!>    في حل X وحساب حدود الخطأ كما هو موضح أدناه.
!>
!> 4. يتم حل نظام المعادلات لـ X باستخدام الشكل المحلل لـ A.
!>
!> 5. يتم تطبيق التنقيح التكراري لتحسين الحل المحسوب
!>    وحساب حدود الخطأ وتقديرات الخطأ الخلفي له.
!>
!> 6. إذا تم استخدام الموازنة، يتم ضرب المصفوفة X مسبقًا بـ
!>    diag(C) (إذا كان TRANS = 'N') أو diag(R) (إذا كان TRANS = 'T' أو 'C') بحيث
!>    تحل النظام الأصلي قبل الموازنة.
!>

المعلمات

FACT 
!>          FACT is CHARACTER*1
!>          يحدد ما إذا كان الشكل المحلل للمصفوفة A موجودًا عند الإدخال، وإذا لم يكن كذلك، فما إذا كان يجب موازنة المصفوفة A قبل تحليلها.
!>          = 'F':  عند الإدخال، تحتوي AF و IPIV على الشكل المحلل لـ A.
!>                  إذا كان EQUED لا يساوي 'N'، فقد تم موازنة المصفوفة A
!>                  بعوامل قياس معطاة بواسطة R و C.
!>                  لا يتم تعديل A و AF و IPIV.
!>          = 'N':  سيتم نسخ المصفوفة A إلى AF وتحليلها.
!>          = 'E':  سيتم موازنة المصفوفة A إذا لزم الأمر، ثم
!>                  نسخها إلى AF وتحليلها.
!>

TRANS

!>          TRANS is CHARACTER*1
!>          يحدد شكل نظام المعادلات:
!>          = 'N':  A * X = B     (بدون تبديل)
!>          = 'T':  A**T * X = B  (تبديل)
!>          = 'C':  A**H * X = B  (تبديل مترافق)
!>

N

!>          N is INTEGER
!>          عدد المعادلات الخطية، أي ترتيب
!>          المصفوفة A.  N >= 0.
!>

NRHS

!>          NRHS is INTEGER
!>          عدد الأطراف اليمنى، أي عدد الأعمدة
!>          للمصفوفتين B و X.  NRHS >= 0.
!>

A

!>          A is COMPLEX array, dimension (LDA,N)
!>          عند الإدخال، المصفوفة N×N A.  إذا كان FACT = 'F' و EQUED لا يساوي 'N'،
!>          فيجب أن تكون A قد تمت موازنتها بعوامل القياس
!>          في R و/أو C.  لا يتم تعديل A إذا كان FACT = 'F' أو
!>          'N'، أو إذا كان FACT = 'E' و EQUED = 'N' عند الخروج.
!>
!>          عند الخروج، إذا كان EQUED .ne. 'N'، تتم موازنة A كالتالي:
!>          EQUED = 'R':  A := diag(R) * A
!>          EQUED = 'C':  A := A * diag(C)
!>          EQUED = 'B':  A := diag(R) * A * diag(C).
!>

LDA

!>          LDA is INTEGER
!>          البعد الرئيسي للمصفوفة A.  LDA >= max(1,N).
!>

AF

!>          AF is COMPLEX array, dimension (LDAF,N)
!>          إذا كان FACT = 'F'، فإن AF هو معامل إدخال وعند الإدخال
!>          يحتوي على العوامل L و U من تحليل
!>          A = P*L*U كما تم حسابه بواسطة CGETRF.  إذا كان EQUED .ne. 'N'، فإن
!>          AF هو الشكل المحلل للمصفوفة A المعاد موازنتها.
!>
!>          إذا كان FACT = 'N'، فإن AF هو معامل إخراج وعند الخروج
!>          يعيد العوامل L و U من تحليل A = P*L*U
!>          للمصفوفة الأصلية A.
!>
!>          إذا كان FACT = 'E'، فإن AF هو معامل إخراج وعند الخروج
!>          يعيد العوامل L و U من تحليل A = P*L*U
!>          للمصفوفة A المعاد موازنتها (انظر وصف A لـ
!>          شكل المصفوفة المعاد موازنتها).
!>

LDAF

!>          LDAF is INTEGER
!>          البعد الرئيسي للمصفوفة AF.  LDAF >= max(1,N).
!>

IPIV

!>          IPIV is INTEGER array, dimension (N)
!>          إذا كان FACT = 'F'، فإن IPIV هو معامل إدخال وعند الإدخال
!>          يحتوي على مؤشرات المحور من تحليل A = P*L*U
!>          كما تم حسابه بواسطة CGETRF؛ تم تبديل الصف i من المصفوفة
!>          مع الصف IPIV(i).
!>
!>          إذا كان FACT = 'N'، فإن IPIV هو معامل إخراج وعند الخروج
!>          يحتوي على مؤشرات المحور من تحليل A = P*L*U
!>          للمصفوفة الأصلية A.
!>
!>          إذا كان FACT = 'E'، فإن IPIV هو معامل إخراج وعند الخروج
!>          يحتوي على مؤشرات المحور من تحليل A = P*L*U
!>          للمصفوفة A المعاد موازنتها.
!>

EQUED

!>          EQUED is CHARACTER*1
!>          يحدد شكل الموازنة التي تم إجراؤها.
!>          = 'N':  لا توجد موازنة (صحيحة دائمًا إذا كان FACT = 'N').
!>          = 'R':  موازنة الصفوف، أي تم ضرب A مسبقًا بـ
!>                  diag(R).
!>          = 'C':  موازنة الأعمدة، أي تم ضرب A لاحقًا بـ
!>                  diag(C).
!>          = 'B':  موازنة الصفوف والأعمدة، أي تم استبدال A بـ
!>                  diag(R) * A * diag(C).
!>          EQUED هو معامل إدخال إذا كان FACT = 'F'. بخلاف ذلك، فهو معامل
!>          إخراج.
!>

R

!>          R is REAL array, dimension (N)
!>          عوامل قياس الصفوف لـ A.  إذا كان EQUED = 'R' أو 'B'، يتم ضرب A
!>          من اليسار بـ diag(R)؛ إذا كان EQUED = 'N' أو 'C'، لا يتم الوصول إلى R.
!>          R هو معامل إدخال إذا كان FACT = 'F'.  بخلاف ذلك،
