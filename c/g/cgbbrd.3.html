subroutine cgbbrd (character vect, integer m, integer n, integer ncc, integer kl, integer ku, complex, dimension( ldab, * ) ab, integer ldab, real, dimension( * ) d, real, dimension( * ) e, complex, dimension( ldq, * ) q, integer ldq, complex, dimension( ldpt, * ) pt, integer ldpt, complex, dimension( ldc, * ) c, integer ldc, complex, dimension( * ) work, real, dimension( * ) rwork, integer info)

CGBBRD

الغرض:

!>
!> CGBBRD تقلل مصفوفة A المعقدة ذات النطاق العام m-by-n إلى شكل bidiagonal حقيقي علوي B بواسطة تحويل وحدوي: Q**H * A * P = B.
!>
!> تقوم الإجراءية بحساب B، وتشكل اختياريًا Q أو P**H، أو تحسب
!> Q**H*C لمصفوفة معينة C.
!>

المعاملات

VECT 
!>          VECT is CHARACTER*1
!>          تحدد ما إذا كان يجب تشكيل المصفوفتين Q و P**H أم لا.
!>          = 'N': لا تشكل Q أو P**H؛
!>          = 'Q': تشكل Q فقط؛
!>          = 'P': تشكل P**H فقط؛
!>          = 'B': تشكل كليهما.
!>

M

!>          M is INTEGER
!>          عدد الصفوف للمصفوفة A. M >= 0.
!>

N

!>          N is INTEGER
!>          عدد الأعمدة للمصفوفة A. N >= 0.
!>

NCC

!>          NCC is INTEGER
!>          عدد الأعمدة للمصفوفة C. NCC >= 0.
!>

KL

!>          KL is INTEGER
!>          عدد الأقطار الفرعية للمصفوفة A. KL >= 0.
!>

KU

!>          KU is INTEGER
!>          عدد الأقطار الفوقية للمصفوفة A. KU >= 0.
!>

AB

!>          AB is COMPLEX array, dimension (LDAB,N)
!>          عند الدخول، مصفوفة النطاق A ذات الأبعاد m-by-n، مخزنة في الصفوف من 1 إلى
!>          KL+KU+1. يتم تخزين العمود j للمصفوفة A في العمود j للمصفوفة
!>          AB كالتالي:
!>          AB(ku+1+i-j,j) = A(i,j) لـ max(1,j-ku) <=i<=min(m,j+kl).
!>          عند الخروج، يتم استبدال A بقيم تم إنشاؤها أثناء
!>          الاختزال.
!>

LDAB

!>          LDAB is INTEGER
!>          البعد الرئيسي للمصفوفة A. LDAB >= KL+KU+1.
!>

D

!>          D is REAL array, dimension (min(M,N))
!>          العناصر القطرية للمصفوفة bidiagonal B.
!>

E

!>          E is REAL array, dimension (min(M,N)-1)
!>          العناصر القطرية الفوقية للمصفوفة bidiagonal B.
!>

Q

!>          Q is COMPLEX array, dimension (LDQ,M)
!>          إذا كانت VECT = 'Q' أو 'B'، فإن المصفوفة الوحدوية Q ذات الأبعاد m-by-m.
!>          إذا كانت VECT = 'N' أو 'P'، فإن المصفوفة Q لا يتم الرجوع إليها.
!>

LDQ

!>          LDQ is INTEGER
!>          البعد الرئيسي للمصفوفة Q.
!>          LDQ >= max(1,M) إذا كانت VECT = 'Q' أو 'B'؛ LDQ >= 1 خلاف ذلك.
!>

PT

!>          PT is COMPLEX array, dimension (LDPT,N)
!>          إذا كانت VECT = 'P' أو 'B'، فإن المصفوفة الوحدوية P' ذات الأبعاد n-by-n.
!>          إذا كانت VECT = 'N' أو 'Q'، فإن المصفوفة PT لا يتم الرجوع إليها.
!>

LDPT

!>          LDPT is INTEGER
!>          البعد الرئيسي للمصفوفة PT.
!>          LDPT >= max(1,N) إذا كانت VECT = 'P' أو 'B'؛ LDPT >= 1 خلاف ذلك.
!>

C

!>          C is COMPLEX array, dimension (LDC,NCC)
!>          عند الدخول، مصفوفة C ذات الأبعاد m-by-ncc.
!>          عند الخروج، يتم استبدال C بـ Q**H*C.
!>          لا يتم الرجوع إلى C إذا كانت NCC = 0.
!>

LDC

!>          LDC is INTEGER
!>          البعد الرئيسي للمصفوفة C.
!>          LDC >= max(1,M) إذا كانت NCC > 0؛ LDC >= 1 إذا كانت NCC = 0.
!>

WORK

!>          WORK is COMPLEX array, dimension (max(M,N))
!>

RWORK

!>          RWORK is REAL array, dimension (max(M,N))
!>

INFO

!>          INFO is INTEGER
!>          = 0:  الخروج بنجاح.
!>          < 0:  إذا كان INFO = -i، فإن المعامل i كان له قيمة غير قانونية.
!>
