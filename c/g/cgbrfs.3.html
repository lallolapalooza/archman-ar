subroutine cgbrfs (character trans, integer n, integer kl, integer ku, integer nrhs, complex, dimension( ldab, * ) ab, integer ldab, complex, dimension( ldafb, * ) afb, integer ldafb, integer, dimension( * ) ipiv, complex, dimension( ldb, * ) b, integer ldb, complex, dimension( ldx, * ) x, integer ldx, real, dimension( * ) ferr, real, dimension( * ) berr, complex, dimension( * ) work, real, dimension( * ) rwork, integer info)

CGBRFS

الغرض:

!>
!> CGBRFS يحسن الحل المحسوب لنظام المعادلات الخطية
!> عندما تكون المصفوفة المعاملة مصفوفة نطاقية، ويوفر
!> حدود الخطأ وتقديرات الخطأ الخلفي للحل.
!>

المعلمات

TRANS 
!>          TRANS is CHARACTER*1
!>          يحدد شكل نظام المعادلات:
!>          = 'N':  A * X = B     (بدون تبديل)
!>          = 'T':  A**T * X = B  (تبديل)
!>          = 'C':  A**H * X = B  (تبديل مرافق)
!>

N

!>          N is INTEGER
!>          رتبة المصفوفة A.  N >= 0.
!>

KL

!>          KL is INTEGER
!>          عدد الأقطار الفرعية داخل نطاق A.  KL >= 0.
!>

KU

!>          KU is INTEGER
!>          عدد الأقطار الفائقة داخل نطاق A.  KU >= 0.
!>

NRHS

!>          NRHS is INTEGER
!>          عدد الأطراف اليمنى، أي عدد أعمدة
!>          المصفوفتين B و X.  NRHS >= 0.
!>

AB

!>          AB is COMPLEX array, dimension (LDAB,N)
!>          مصفوفة النطاق الأصلية A، مخزنة في الصفوف من 1 إلى KL+KU+1.
!>          يتم تخزين العمود j من A في العمود j من
!>          المصفوفة AB كما يلي:
!>          AB(ku+1+i-j,j) = A(i,j) لـ max(1,j-ku)<=i<=min(n,j+kl).
!>

LDAB

!>          LDAB is INTEGER
!>          البعد الرائد للمصفوفة AB.  LDAB >= KL+KU+1.
!>

AFB

!>          AFB is COMPLEX array, dimension (LDAFB,N)
!>          تفاصيل تحلل LU لمصفوفة النطاق A، كما
!>          تم حسابها بواسطة CGBTRF.  U مخزنة كمصفوفة نطاق علوية مثلثية
!>          مع KL+KU أقطار فائقة في الصفوف من 1 إلى KL+KU+1، و
!>          تُخزن المضاعفات المستخدمة أثناء التحلل
!>          في الصفوف KL+KU+2 إلى 2*KL+KU+1.
!>

LDAFB

!>          LDAFB is INTEGER
!>          البعد الرائد للمصفوفة AFB.  LDAFB >= 2*KL*KU+1.
!>

IPIV

!>          IPIV is INTEGER array, dimension (N)
!>          مؤشرات المحاور من CGBTRF؛ لـ 1<=i<=N، تم تبديل الصف i من
!>          المصفوفة مع الصف IPIV(i).
!>

B

!>          B is COMPLEX array, dimension (LDB,NRHS)
!>          مصفوفة الطرف الأيمن B.
!>

LDB

!>          LDB is INTEGER
!>          البعد الرائد للمصفوفة B.  LDB >= max(1,N).
!>

X

!>          X is COMPLEX array, dimension (LDX,NRHS)
!>          عند الإدخال، مصفوفة الحل X، كما تم حسابها بواسطة CGBTRS.
!>          عند الخروج، مصفوفة الحل المحسنة X.
!>

LDX

!>          LDX is INTEGER
!>          البعد الرائد للمصفوفة X.  LDX >= max(1,N).
!>

FERR

!>          FERR is REAL array, dimension (NRHS)
!>          حد الخطأ الأمامي المقدر لكل متجه حل
!>          X(j) (العمود j من مصفوفة الحل X).
!>          إذا كان XTRUE هو الحل الحقيقي المقابل لـ X(j)، فإن FERR(j)
!>          هو حد علوي مقدر لقيمة أكبر
!>          عنصر في (X(j) - XTRUE) مقسومًا على قيمة أكبر
!>          عنصر في X(j). التقدير موثوق به قدر
!>          موثوقية تقدير RCOND، وهو دائمًا تقريبًا زيادة طفيفة
!>          في الخطأ الحقيقي.
!>

BERR

!>          BERR is REAL array, dimension (NRHS)
!>          الخطأ النسبي المكون من عنصر لعنصر لكل متجه حل
!>          X(j) (أي، أصغر تغيير نسبي في
!>          أي عنصر من A أو B يجعل X(j) حلاً دقيقًا).
!>

WORK

!>          WORK is COMPLEX array, dimension (2*N)
!>

RWORK

!>          RWORK is REAL array, dimension (N)
!>

INFO

!>          INFO is INTEGER
!>          = 0:  خروج ناجح
!>          < 0:  إذا كان INFO = -i، فإن المعلمة i كانت ذات قيمة غير قانونية
!>

المعلمات الداخلية:

!>
