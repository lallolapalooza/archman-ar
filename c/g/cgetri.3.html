توثيق الوظيفة/الروتين الفرعي

subroutine cgetri (integer n, complex, dimension( lda, * ) a, integer lda, integer, dimension( * ) ipiv, complex, dimension( * ) work, integer lwork, integer info)

CGETRI

الغرض:

!>
!> CGETRI يحسب معكوس مصفوفة باستخدام تحليل LU
!> الذي تم حسابه بواسطة CGETRF.
!>
!> تستخدم هذه الطريقة معكوس U ثم تحسب معكوس(A) عن طريق حل النظام
!> معكوس(A)*L = معكوس(U) لـ معكوس(A).
!>

المعاملات

N 
!>          N هو عدد صحيح (INTEGER)
!>          رتبة المصفوفة A. N >= 0.
!>

A

!>          A هي مصفوفة مركبة (COMPLEX) ، ذات بعد (LDA,N)
!>          عند الإدخال، عوامل L و U من تحليل
!>          A = P*L*U كما تم حسابه بواسطة CGETRF.
!>          عند الخروج، إذا كان INFO = 0، فإن معكوس المصفوفة الأصلية A.
!>

LDA

!>          LDA هو عدد صحيح (INTEGER)
!>          البعد الرائد للمصفوفة A. LDA >= max(1,N).
!>

IPIV

!>          IPIV هي مصفوفة أعداد صحيحة (INTEGER) ، ذات بعد (N)
!>          مؤشرات المحور من CGETRF؛ لـ 1<=i<=N، تم تبديل الصف i للمصفوفة
!>          مع الصف IPIV(i).
!>

WORK

!>          WORK هي مصفوفة مركبة (COMPLEX) ، ذات بعد (MAX(1,LWORK))
!>          عند الخروج، إذا كان INFO=0، فإن WORK(1) يعيد قيمة LWORK المثلى.
!>

LWORK

!>          LWORK هو عدد صحيح (INTEGER)
!>          بعد المصفوفة WORK. LWORK >= max(1,N).
!>          للحصول على أداء أمثل LWORK >= N*NB، حيث NB هو
!>          حجم الكتلة الأمثل الذي تعيده ILAENV.
!>
!>          إذا كان LWORK = -1، فسيتم افتراض استعلام عن مساحة العمل؛ الروتين
!>          يحسب فقط الحجم الأمثل لمصفوفة WORK، ويعيد
!>          هذه القيمة كإدخال أول لمصفوفة WORK، ولا يتم إصدار رسالة خطأ
!>          متعلقة بـ LWORK بواسطة XERBLA.
!>

INFO

!>          INFO هو عدد صحيح (INTEGER)
!>          = 0: الخروج بنجاح
!>          < 0: إذا كان INFO = -i، فإن المعامل i كان له قيمة غير قانونية
!>          > 0: إذا كان INFO = i، فإن U(i,i) هو صفر تمامًا؛ المصفوفة
!>                منفردة ولا يمكن حساب معكوسها.
!>
