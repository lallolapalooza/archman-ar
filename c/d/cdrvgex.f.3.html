الغرض:

!>
!> CDRVGE تختبر الروتينات الدليلية CGESV, -SVX, and -SVXX.
!>
!> لاحظ أن هذا الملف يستخدم فقط عندما تكون XBLAS متاحة،

!>

المعلمات

DOTYPE 
!>          DOTYPE هو مصفوفة LOGICAL، أبعادها (NTYPES)
!>          أنواع المصفوفات التي سيتم استخدامها للاختبار. المصفوفات من النوع j
!>          (لـ 1 <= j <= NTYPES) تستخدم للاختبار إذا كان DOTYPE(j) =
!>          .TRUE.؛ إذا كان DOTYPE(j) = .FALSE.، فإن النوع j لا يتم استخدامه.
!>

NN

!>          NN هو INTEGER
!>          عدد القيم لـ N الموجودة في المتجه NVAL.
!>

NVAL

!>          NVAL هي مصفوفة INTEGER، أبعادها (NN)
!>          قيم البعد العمودي للمصفوفة N.
!>

NRHS

!>          NRHS هو INTEGER
!>          عدد المتجهات الطرفية اليمنى التي سيتم إنشاؤها لكل نظام خطي.
!>

THRESH

!>          THRESH هو REAL
!>          القيمة الحدية لنسب الاختبار. يتم تضمين نتيجة
!>          في ملف الإخراج إذا كان RESULT >= THRESH. لطباعة
!>          كل نسبة اختبار، استخدم THRESH = 0.
!>

TSTERR

!>          TSTERR هو LOGICAL
!>          علم يشير إلى ما إذا كان سيتم اختبار مخارج الخطأ.
!>

NMAX

!>          NMAX هو INTEGER
!>          الحد الأقصى المسموح به لـ N، يستخدم في تحديد أبعاد
!>          مصفوفات العمل.
!>

A

!>          A هي مصفوفة COMPLEX، أبعادها (NMAX*NMAX)
!>

AFAC

!>          AFAC هي مصفوفة COMPLEX، أبعادها (NMAX*NMAX)
!>

ASAV

!>          ASAV هي مصفوفة COMPLEX، أبعادها (NMAX*NMAX)
!>

B

!>          B هي مصفوفة COMPLEX، أبعادها (NMAX*NRHS)
!>

BSAV

!>          BSAV هي مصفوفة COMPLEX، أبعادها (NMAX*NRHS)
!>

X

!>          X هي مصفوفة COMPLEX، أبعادها (NMAX*NRHS)
!>

XACT

!>          XACT هي مصفوفة COMPLEX، أبعادها (NMAX*NRHS)
!>

S

!>          S هي مصفوفة REAL، أبعادها (2*NMAX)
!>

WORK

!>          WORK هي مصفوفة COMPLEX، أبعادها
!>                      (NMAX*max(3,NRHS))
!>

RWORK

!>          RWORK هي مصفوفة REAL، أبعادها (2*NRHS+NMAX)
!>

IWORK

!>          IWORK هي مصفوفة INTEGER، أبعادها (NMAX)

NOUT
