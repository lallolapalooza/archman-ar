cmake-buildsystem(7)

الحزمة: cmake

التراخيص: custom

able(exe2 exe2.cpp) target_link_libraries(exe2 lib1Version2 lib1Version3) # تشخيص

تحدد خاصية الهدف COMPATIBLE_INTERFACE_NUMBER_MAX <#prop_tgt:COMPATIBLE_INTERFACE_NUMBER_MAX> أن المحتوى سيتم تقييمه عدديًا وسيتم حساب الحد الأقصى للعدد بين جميع الأعداد المحددة:

add_library(lib1Version2 SHARED lib1_v2.cpp)
set_property(TARGET lib1Version2 PROPERTY INTERFACE_CONTAINER_SIZE_REQUIRED 200)
set_property(TARGET lib1Version2 APPEND PROPERTY
  COMPATIBLE_INTERFACE_NUMBER_MAX CONTAINER_SIZE_REQUIRED
)
add_library(lib1Version3 SHARED lib1_v3.cpp)
set_property(TARGET lib1Version3 PROPERTY INTERFACE_CONTAINER_SIZE_REQUIRED 1000)
add_executable(exe1 exe1.cpp)
# CONTAINER_SIZE_REQUIRED سيكون "200"
target_link_libraries(exe1 lib1Version2)
add_executable(exe2 exe2.cpp)
# CONTAINER_SIZE_REQUIRED سيكون "1000"
target_link_libraries(exe2 lib1Version2 lib1Version3)

وبالمثل، يمكن استخدام COMPATIBLE_INTERFACE_NUMBER_MIN <#prop_tgt:COMPATIBLE_INTERFACE_NUMBER_MIN> لحساب الحد الأدنى العددي لقيمة خاصية من التبعيات.

يمكن قراءة كل قيمة خاصية "متوافقة" محسوبة في المستهلك في وقت الإنشاء باستخدام تعبيرات المولد.

لاحظ أنه لكل تبعية، يجب ألا تتقاطع مجموعة الخصائص المحددة في كل خاصية واجهة متوافقة مع المجموعة المحددة في أي من الخصائص الأخرى.

تصحيح أصل الخاصية

نظرًا لأنه يمكن تحديد مواصفات الإنشاء بواسطة التبعيات، فإن عدم وجود محليّة للكود الذي ينشئ هدفًا والكود المسؤول عن تحديد مواصفات الإنشاء قد يجعل الكود أكثر صعوبة في فهمه. يوفر cmake(1) <#manual:cmake(1)> مرفق تصحيح أخطاء لطباعة أصل محتويات الخصائص التي يمكن تحديدها بواسطة التبعيات. الخصائص التي يمكن تصحيحها مدرجة في وثائق المتغير CMAKE_DEBUG_TARGET_PROPERTIES <#variable:CMAKE_DEBUG_TARGET_PROPERTIES>:

set(CMAKE_DEBUG_TARGET_PROPERTIES
  INCLUDE_DIRECTORIES
  COMPILE_DEFINITIONS
  POSITION_INDEPENDENT_CODE
  CONTAINER_SIZE_REQUIRED
  LIB_VERSION
)
add_executable(exe1 exe1.cpp)

في حالة الخصائص المدرجة في COMPATIBLE_INTERFACE_BOOL <#prop_tgt:COMPATIBLE_INTERFACE_BOOL> أو COMPATIBLE_INTERFACE_STRING <#prop_tgt:COMPATIBLE_INTERFACE_STRING>، يوضح خرج التصحيح الهدف الذي كان مسؤولاً عن تعيين الخاصية، وأي التبعيات الأخرى التي عرفت الخاصية أيضًا. في حالة COMPATIBLE_INTERFACE_NUMBER_MAX <#prop_tgt:COMPATIBLE_INTERFACE_NUMBER_MAX> و COMPATIBLE_INTERFACE_NUMBER_MIN <#prop_tgt:COMPATIBLE_INTERFACE_NUMBER_MIN>، يوضح خرج التصحيح قيمة الخاصية من كل تبعية، وما إذا كانت القيمة تحدد الحد الأقصى الجديد.

مواصفات الإنشاء باستخدام تعبيرات المولد

قد تستخدم مواصفات الإنشاء تعبيرات المولد <#manual:cmake-generator-expressions(7)> التي تحتوي على محتوى قد يكون شرطيًا أو معروفًا فقط في وقت الإنشاء. على سبيل المثال، يمكن قراءة القيمة "المتوافقة" المحسوبة لخاصية باستخدام تعبير TARGET_PROPERTY:

add_library(lib1Version2 SHARED lib1_v2.cpp)
set_property(TARGET lib1Version2 PROPERTY
  INTERFACE_CONTAINER_SIZE_REQUIRED 200)
set_property(TARGET lib1Version2 APPEND PROPERTY
  COMPATIBLE_INTERFACE_NUMBER_MAX CONTAINER_SIZE_REQUIRED
)
add_executable(exe1 exe1.cpp)
target_link_libraries(exe1 lib1Version2)
target_compile_definitions(exe1 PRIVATE
    CONTAINER_SIZE=$<TARGET_PROPERTY:CONTAINER_SIZE_REQUIRED>
)

في هذه الحالة، سيتم تجميع ملفات المصدر exe1 باستخدام -DCONTAINER_SIZE=200.

يتم تقييم تعبير المولد الأحادي TARGET_PROPERTY وتعبير المولد TARGET_POLICY بسياق الهدف المستهلك. هذا يعني أنه يمكن تقييم مواصفات متطلبات الاستخدام بشكل مختلف بناءً على المستهلك:

add_library(lib1 lib1.cpp)
target_compile_definitions(lib1 INTERFACE
  $<$<STREQUAL:$<TARGET_PROPERTY:TYPE>,EXECUTABLE>:LIB1_WITH_EXE>
  $<$<STREQUAL:$<TARGET_PROPERTY:TYPE>,SHARED_LIBRARY>:LIB1_WITH_SHARED_LIB>
  $<$<TARGET_POLICY:CMP0182>:CONSUMER_CMP0182_NEW>
)
add_executable(exe1 exe1.cpp)
target_link_libraries(exe1 lib1)
cmake_policy(SET CMP0182 NEW)
add_library(shared_lib shared_lib.cpp)
target_link_libraries(shared_lib lib1)

سيتم تجميع الملف التنفيذي exe1 باستخدام -DLIB1_WITH_EXE، بينما سيتم تجميع المكتبة المشتركة shared_lib باستخدام -DLIB1_WITH_SHARED_LIB و -DCONSUMER_CMP0182_NEW، لأن السياسة CMP0182 <#policy:CMP0182> هي NEW في النقطة التي تم فيها إنشاء الهدف shared_lib.

يقوم تعبير BUILD_INTERFACE بتغليف المتطلبات التي تستخدم فقط عند استهلاكها من هدف في نفس نظام الإنشاء، أو عند استهلاكها من هدف تم تصديره إلى دليل الإنشاء باستخدام الأمر export() <#command:export>. يقوم تعبير INSTALL_INTERFACE بتغليف المتطلبات التي تستخدم فقط عند استهلاكها من هدف تم تثبيته وتصديره باستخدام الأمر install(EXPORT) <#export>:

add_library(ClimbingStats climbingstats.cpp)
target_compile_definitions(ClimbingStats INTERFACE
  $<BUILD_INTERFACE:ClimbingStats_FROM_BUILD_LOCATION>
  $<INSTALL_INTERFACE:ClimbingStats_FROM_INSTALLED_LOCATION>
)
install(TARGETS ClimbingStats EXPORT libExport ${InstallArgs})
install(EXPORT libExport NAMESPACE Upstream::
        DESTINATION lib/cmake/ClimbingStats)
export(EXPORT libExport NAMESPACE Upstream::)
add_executable(exe1 exe1.cpp)
target_link_libraries(exe1 ClimbingStats)

في هذه الحالة، سيتم تجميع الملف التنفيذي exe1 باستخدام -DClimbingStats_FROM_BUILD_LOCATION. تقوم أوامر التصدير بإنشاء أهداف IMPORTED <#prop_tgt:IMPORTED> إما مع إغفال INSTALL_INTERFACE أو BUILD_INTERFACE، وإزالة علامة *_INTERFACE. مشروع منفصل يستهلك حزمة ClimbingStats سيحتوي على:

find_package(ClimbingStats REQUIRED)
add_executable(Downstream main.cpp)
target_link_libraries(Downstream Upstream::ClimbingStats)

اعتمادًا على ما إذا كانت حزمة ClimbingStats قد تم استخدامها من موقع الإنشاء أو موقع التثبيت، سيتم تجميع الهدف Downstream إما باستخدام -DClimbingStats_FROM_BUILD_LOCATION أو -DClimbingStats_FROM_INSTALL_LOCATION. لمزيد من المعلومات حول الحزم والتصدير، راجع دليل cmake-packages(7) <#manual:cmake-packages(7)>.

مسارات التضمين ومتطلبات الاستخدام

تتطلب مسارات التضمين بعض الاعتبارات الخاصة عند تحديدها كمتطلبات استخدام وعند استخدامها مع تعبيرات المولد. يقبل الأمر target_include_directories() <#command:target_include_directories> مسارات نسبية ومطلقة:

add_library(lib1 lib1.cpp)
target_include_directories(lib1 PRIVATE
  /absolute/path
  relative/path
)

يتم تفسير المسارات النسبية بالنسبة لدليل المصدر حيث يظهر الأمر. المسارات النسبية غير مسموح بها في INTERFACE_INCLUDE_DIRECTORIES <#prop_tgt:INTERFACE_INCLUDE_DIRECTORIES> للأهداف IMPORTED <#prop_tgt:IMPORTED>.

في الحالات التي يتم فيها استخدام تعبير مولد غير تافه، يمكن استخدام تعبير INSTALL_PREFIX داخل وسيطة تعبير INSTALL_INTERFACE. إنه علامة استبدال تتوسع إلى بادئة التثبيت عند استهلاكها بواسطة مشروع مستهلك.

غالبًا ما تختلف متطلبات استخدام مسارات التضمين بين شجرة الإنشاء وشجرة التثبيت. يمكن استخدام تعبيرات المولد BUILD_INTERFACE و INSTALL_INTERFACE لوصف متطلبات استخدام منفصلة بناءً على موقع الاستخدام. المسارات النسبية مسموح بها داخل تعبير INSTALL_INTERFACE ويتم تفسيرها بالنسبة لبادئة التثبيت. على سبيل المثال:

add_library(ClimbingStats climbingstats.cpp)
target_include_directories(ClimbingStats INTERFACE
  $<BUILD_INTERFACE:${CMAKE_CURRENT_BINARY_DIR}/generated>
  $<INSTALL_INTERFACE:/absolute/path>
  $<INSTALL_INTERFACE:relative/path>
  $<INSTALL_INTERFACE:$<INSTALL_PREFIX>/$<CONFIG>/generated>
)

يتم توفير واجهتين ملائمتين تتعلقان بمتطلبات استخدام مسارات التضمين. يمكن تمكين المتغير CMAKE_INCLUDE_CURRENT_DIR_IN_INTERFACE <#variable:CMAKE_INCLUDE_CURRENT_DIR_IN_INTERFACE>، مع تأثير مكافئ لـ:

set_property(TARGET tgt APPEND PROPERTY INTERFACE_INCLUDE_DIRECTORIES
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR};${CMAKE_CURRENT_BINARY_DIR}>
)

لكل هدف متأثر. الراحة للأهداف المثبتة هي مكون INCLUDES DESTINATION مع الأمر install(TARGETS) <#targets>:

install(TARGETS foo bar bat EXPORT tgts ${dest_args}
  INCLUDES DESTINATION include
)
install(EXPORT tgts ${other_args})
install(FILES ${headers} DESTINATION include)

هذا يعادل إضافة ${CMAKE_INSTALL_PREFIX}/include إلى INTERFACE_INCLUDE_DIRECTORIES <#prop_tgt:INTERFACE_INCLUDE_DIRECTORIES> لكل هدف IMPORTED <#prop_tgt:IMPORTED> المثبت عند إنشائه بواسطة install(EXPORT) <#export>.

عند استهلاك INTERFACE_INCLUDE_DIRECTORIES <#prop_tgt:INTERFACE_INCLUDE_DIRECTORIES> لهدف مستورد، قد يتم التعامل مع الإدخالات في الخاصية كمسارات تضمين نظام. آثار ذلك تعتمد على سلسلة الأدوات، ولكن أحد الآثار الشائعة هو تخطي تحذيرات المترجم لرؤوس موجودة في تلك الأدلة. تحدد الخاصية SYSTEM <#prop_tgt:SYSTEM> لهذا السلوك (انظر خاصية EXPORT_NO_SYSTEM <#prop_tgt:EXPORT_NO_SYSTEM> لكيفية تعديل القيمة المثبتة لهدف). من الممكن أيضًا تغيير كيفية تفسير المستهلكين للنظام
