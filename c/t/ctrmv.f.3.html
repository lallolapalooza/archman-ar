subroutine ctrmv (character uplo, character trans, character diag, integer n, complex, dimension(lda,*) a, integer lda, complex, dimension(*) x, integer incx)

CTRMV

الغرض:

!>
!> CTRMV  تقوم بإحدى عمليات ضرب المصفوفة في المتجه:
!>
!>    x := A*x,   أو   x := A**T*x,   أو   x := A**H*x,
!>
!> حيث x متجه بعناصر n، و A مصفوفة مثلثة عليا أو سفلى، واحدة أو غير واحدة، بحجم n × n.
!>

المعاملات

UPLO 
!>          UPLO هو CHARACTER*1
!>           عند الدخول، تحدد UPLO ما إذا كانت المصفوفة مثلثة عليا أو سفلى كما يلي:
!>
!>              UPLO = 'U' أو 'u'   A مصفوفة مثلثة عليا.
!>
!>              UPLO = 'L' أو 'l'   A مصفوفة مثلثة سفلى.
!>

TRANS

!>          TRANS هو CHARACTER*1
!>           عند الدخول، تحدد TRANS العملية المراد تنفيذها كما يلي:
!>
!>              TRANS = 'N' أو 'n'   x := A*x.
!>
!>              TRANS = 'T' أو 't'   x := A**T*x.
!>
!>              TRANS = 'C' أو 'c'   x := A**H*x.
!>

DIAG

!>          DIAG هو CHARACTER*1
!>           عند الدخول، تحدد DIAG ما إذا كانت A واحدة أو غير واحدة
!>           مثلثة كما يلي:
!>
!>              DIAG = 'U' أو 'u'   يفترض أن A مثلثة واحدة.
!>
!>              DIAG = 'N' أو 'n'   لا يفترض أن A مثلثة واحدة.
!>

N

!>          N هو INTEGER
!>           عند الدخول، تحدد N رتبة المصفوفة A.
!>           يجب أن تكون N أكبر من أو تساوي صفر.
!>

A

!>          A هو مصفوفة COMPLEX، بحجم ( LDA, N ).
!>           قبل الدخول مع UPLO = 'U' أو 'u'، يجب أن يحتوي الجزء العلوي الأيمن الذي حجمه n × n
!>           من المصفوفة A على المصفوفة المثلثة العليا، ولا يتم الرجوع إلى الجزء السفلي الأيمن
!>           المعروف بدقته من A.
!>           قبل الدخول مع UPLO = 'L' أو 'l'، يجب أن يحتوي الجزء السفلي الأيمن الذي حجمه n × n
!>           من المصفوفة A على المصفوفة المثلثة السفلى، ولا يتم الرجوع إلى الجزء العلوي الأيمن
!>           المعروف بدقته من A.
!>           لاحظ أنه عندما يكون DIAG = 'U' أو 'u'، لا يتم الرجوع إلى العناصر القطرية لـ
!>           A أيضًا، ولكن يفترض أنها الوحدة.
!>

LDA

!>          LDA هو INTEGER
!>           عند الدخول، تحدد LDA البعد الأول لـ A كما هو معلن
!>           في البرنامج (الفرعي) المستدعي. يجب أن يكون LDA أكبر من أو يساوي
!>           max( 1, n ).
!>

X

!>          X هي مصفوفة COMPLEX، بحجم لا يقل عن
!>           ( 1 + ( n - 1 )*abs( INCX ) ).
!>           قبل الدخول، يجب أن تحتوي المصفوفة X المتزايدة على متجه x ذي n عنصر.
!>           عند الخروج، يتم استبدال X بالمتجه x المحول.
!>

INCX

!>          INCX هو INTEGER
!>           عند الدخول، تحدد INCX الزيادة للعناصر في
!>           X. يجب أن لا تكون INCX صفرًا.
!>
