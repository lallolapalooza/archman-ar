subroutine ctrti2 (uplo, diag, n, a, lda, info)
CTRTI2 تحسب معكوس مصفوفة مثلثة (خوارزمية غير مكدسة).

توثيق الوظيفة/الوحدة الفرعية

subroutine ctrti2 (character uplo, character diag, integer n, complex, dimension( lda, * ) a, integer lda, integer info)

CTRTI2 تحسب معكوس مصفوفة مثلثة (خوارزمية غير مكدسة).

الغرض:

!>
!> CTRTI2 تحسب معكوس مصفوفة مثلثة علوية أو سفلية معقدة
!>
!> هذه هي نسخة المستوى 2 BLAS من الخوارزمية.
!>

المعاملات

UPLO 
!>          UPLO هو CHARACTER*1
!>          يحدد ما إذا كانت المصفوفة A علوية أو سفلية.
!>          = 'U':  علوية
!>          = 'L':  سفلية
!>

DIAG

!>          DIAG هو CHARACTER*1
!>          يحدد ما إذا كانت المصفوفة A وحدة مثلثة أم لا.
!>          = 'N':  غير وحدة مثلثة
!>          = 'U':  وحدة مثلثة
!>

N

!>          N هو INTEGER
!>          رتبة المصفوفة A.  N >= 0.
!>

A

!>          A هو مصفوفة COMPLEX، ذات بعد (LDA,N)
!>          عند الإدخال، المصفوفة المثلثة A.  إذا كان UPLO = 'U'، فإن
!>          الجزء العلوي المثلثي من المصفوفة A بأبعاد n × n يحتوي على
!>          المصفوفة المثلثة العلوية، والجزء السفلي المثلثي من A لا تتم الإشارة إليه.  إذا كان UPLO = 'L'، فإن
!>          الجزء السفلي المثلثي من المصفوفة A بأبعاد n × n يحتوي على
!>          المصفوفة المثلثة السفلية، والجزء العلوي المثلثي من A لا تتم الإشارة إليه.  إذا كان DIAG = 'U'، فإن
!>          العناصر القطرية لـ A لا تتم الإشارة إليها أيضًا ويفترض أنها 1.
!>
!>          عند الخروج، معكوس (المثلث) للمصفوفة الأصلية، بنفس تنسيق التخزين.
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرئيسي للمصفوفة A.  LDA >= max(1,N).
!>

INFO

!>          INFO هو INTEGER
!>          = 0: خروج ناجح
!>          < 0: إذا كان INFO = -k، فإن المعامل k كان له قيمة غير قانونية
!>
