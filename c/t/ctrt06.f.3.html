subroutine ctrt06 (real rcond, real rcondc, character uplo, character diag, integer n, complex, dimension( lda, * ) a, integer lda, real, dimension( * ) rwork, real rat)

CTRT06

الغرض:

!>
!> CTRT06 يحسب نسبة اختبار تقارن RCOND (الرقم الشرطي المعكوس لمصفوفة مثلثية A) و RCONDC، التقدير
!> الذي تم حسابه بواسطة CTRCON. يتم استخدام المعلومات حول المصفوفة المثلثية A
!> إذا كان أحد التقديرين صفرًا والآخر غير صفر لتحديد ما إذا كان
!> نقص التدفق في التقدير مبررًا.
!>

المعلمات

RCOND 
!>          RCOND هو REAL
!>          تقدير الرقم الشرطي المعكوس الذي تم الحصول عليه عن طريق
!>          تشكيل معكوس صريح للمصفوفة A وحساب
!>          RCOND = 1/( norm(A) * norm(inv(A)) ).
!>

RCONDC

!>          RCONDC هو REAL
!>          تقدير الرقم الشرطي المعكوس الذي تم حسابه بواسطة
!>          CTRCON.
!>

UPLO

!>          UPLO هو CHARACTER
!>          يحدد ما إذا كانت المصفوفة A علوية أو سفلية مثلثية.
!>          = 'U':  مثلثية علوية
!>          = 'L':  مثلثية سفلية
!>

DIAG

!>          DIAG هو CHARACTER
!>          يحدد ما إذا كانت المصفوفة A وحدوية مثلثية أم لا.
!>          = 'N':  غير وحدوية مثلثية
!>          = 'U':  وحدوية مثلثية
!>

N

!>          N هو INTEGER
!>          ترتيب المصفوفة A.  N >= 0.
!>

A

!>          A هي مصفوفة COMPLEX، البعد (LDA,N)
!>          المصفوفة المثلثية A.  إذا كان UPLO = 'U'، الجزء العلوي المثلثي العلوي n في n
!>          من المصفوفة A يحتوي على المصفوفة المثلثية العلوية، والجزء السفلي المثلثي الصارم
!>          من A غير مستخدم.  إذا كان UPLO = 'L'، الجزء السفلي المثلثي السفلي n في n
!>          من المصفوفة A يحتوي على المصفوفة المثلثية السفلية، والجزء العلوي المثلثي الصارم
!>          من A غير مستخدم.  إذا كان DIAG = 'U'، عناصر القطر في A
!>          غير مستخدمة أيضًا ويفترض أنها 1.
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرائد للمصفوفة A.  LDA >= max(1,N).
!>

RWORK

!>          RWORK هي مصفوفة REAL، البعد (N)
!>

RAT

!>          RAT هو REAL
!>          نسبة الاختبار.  إذا كان كلا من RCOND و RCONDC غير صفريين،
!>             RAT = MAX( RCOND, RCONDC )/MIN( RCOND, RCONDC ) - 1.
!>          إذا كان RAT = 0، فإن التقديرين متطابقان تمامًا.
!>
