subroutine ctrsm (character side, character uplo, character transa, character diag, integer m, integer n, complex alpha, complex, dimension(lda,*) a, integer lda, complex, dimension(ldb,*) b, integer ldb)

CTRSM

الغرض:

!>
!> CTRSM  يحل أحد معادلات المصفوفات التالية:
!>
!>    op( A )*X = alpha*B,   أو   X*op( A ) = alpha*B,
!>
!> حيث alpha هو عدد قياسي، X و B مصفوفتان m × n، A مصفوفة مثلثة عليا أو سفلى،
!> موحدة أو غير موحدة، و op( A ) هي واحدة مما يلي:
!>
!>    op( A ) = A   أو   op( A ) = A**T   أو   op( A ) = A**H.
!>
!> يتم الكتابة فوق المصفوفة X في B.
!>

المعاملات

SIDE 
!>          SIDE هو CHARACTER*1
!>           عند الإدخال، تحدد SIDE ما إذا كانت op( A ) تظهر على اليسار
!>           أو اليمين بالنسبة لـ X كما يلي:
!>
!>              SIDE = 'L' أو 'l'   op( A )*X = alpha*B.
!>
!>              SIDE = 'R' أو 'r'   X*op( A ) = alpha*B.
!>

UPLO

!>          UPLO هو CHARACTER*1
!>           عند الإدخال، تحدد UPLO ما إذا كانت المصفوفة A مثلثة عليا أو
!>           سفلى كما يلي:
!>
!>              UPLO = 'U' أو 'u'   A مصفوفة مثلثة عليا.
!>
!>              UPLO = 'L' أو 'l'   A مصفوفة مثلثة سفلى.
!>

TRANSA

!>          TRANSA هو CHARACTER*1
!>           عند الإدخال، تحدد TRANSA شكل op( A ) المطلوب استخدامه في
!>           ضرب المصفوفات كما يلي:
!>
!>              TRANSA = 'N' أو 'n'   op( A ) = A.
!>
!>              TRANSA = 'T' أو 't'   op( A ) = A**T.
!>
!>              TRANSA = 'C' أو 'c'   op( A ) = A**H.
!>

DIAG

!>          DIAG هو CHARACTER*1
!>           عند الإدخال، تحدد DIAG ما إذا كانت A موحدة مثلثة أو لا
!>           كما يلي:
!>
!>              DIAG = 'U' أو 'u'   A مفترضة أنها موحدة مثلثة.
!>
!>              DIAG = 'N' أو 'n'   A ليست مفترضة أنها موحدة مثلثة.
!>

M

!>          M هو INTEGER
!>           عند الإدخال، يحدد M عدد صفوف B. يجب أن يكون M على الأقل صفر.
!>

N

!>          N هو INTEGER
!>           عند الإدخال، يحدد N عدد أعمدة B. يجب أن يكون N على الأقل صفر.
!>

ALPHA

!>          ALPHA هو COMPLEX
!>           عند الإدخال، يحدد ALPHA العدد القياسي alpha. عندما يكون alpha
!>           صفرًا، لا يتم الرجوع إلى A ولا يلزم تعيين B قبل الدخول.
!>

A

!>          A هو مصفوفة COMPLEX، أبعادها ( LDA, k )،
!>           حيث k يساوي m عندما يكون SIDE = 'L' أو 'l'
!>             و k يساوي n عندما يكون SIDE = 'R' أو 'r'.
!>           قبل الإدخال مع UPLO = 'U' أو 'u'، يجب أن يحتوي الجزء العلوي
!>           المقدم من المصفوفة A بحجم k × k على المصفوفة المثلثة العليا
!>           ولا يتم الرجوع إلى الجزء السفلي الصارم من A.
!>           قبل الإدخال مع UPLO = 'L' أو 'l'، يجب أن يحتوي الجزء السفلي
!>           المقدم من المصفوفة A بحجم k × k على المصفوفة المثلثة السفلى
!>           ولا يتم الرجوع إلى الجزء العلوي الصارم من A.
!>           لاحظ أنه عندما يكون DIAG = 'U' أو 'u'، لا يتم الرجوع إلى العناصر
!>           القطرية لـ A أيضًا، ولكنها مفترضة أنها الوحدة.
!>

LDA

!>          LDA هو INTEGER
!>           عند الإدخال، يحدد LDA البعد الأول لـ A كما هو معلن
!>           في البرنامج الأصلي (الفرعي). عندما يكون SIDE = 'L' أو 'l'،
!>           يجب أن يكون LDA على الأقل max( 1, m )، عندما يكون SIDE = 'R' أو 'r'،
!>           يجب أن يكون LDA على الأقل max( 1, n ).
!>

B

!>          B هي مصفوفة COMPLEX، أبعادها ( LDB, N )
!>           قبل الإدخال، يجب أن يحتوي الجزء المقدم من المصفوفة B بحجم m × n
!>           على المصفوفة B للجانب الأيمن، وعند الخروج يتم الكتابة فوقها
!>           بالمصفوفة X للحل.
!>

LDB

!>          LDB هو INTEGER
!>           عند الإدخال، يحدد LDB البعد الأول لـ B كما هو معلن
!>           في البرنامج الأصلي (الفرعي). يجب أن يكون LDB على الأقل
!>           max( 1, m ).
!>
