subroutine ctrttf (transr, uplo, n, a, lda, arf, info)
CTRTTF ينسخ مصفوفة مثلثية من التنسيق الكامل القياسي (TR) إلى التنسيق المعبأ الكامل المستطيل (TF).

توثيق الوظيفة/الروتين الفرعي

subroutine ctrttf (character transr, character uplo, integer n, complex, dimension( 0: lda-1, 0: * ) a, integer lda, complex, dimension( 0: * ) arf, integer info)

CTRTTF تنسخ مصفوفة مثلثية A من التنسيق الكامل القياسي (TR) إلى التنسيق المعبأ الكامل المستطيل (TF).

الغرض:

!>
!> CTRTTF تنسخ مصفوفة مثلثية A من التنسيق الكامل القياسي (TR)
!> إلى التنسيق المعبأ الكامل المستطيل (TF) .
!>

المعلمات

TRANSR 
!>          TRANSR هو CHARACTER*1
!>          = 'N':  ARF في وضع Normal مطلوب؛
!>          = 'C':  ARF في وضع Conjugate Transpose مطلوب؛
!>

UPLO

!>          UPLO هو CHARACTER*1
!>          = 'U':  A مثلثية عليا؛
!>          = 'L':  A مثلثية سفلى.
!>

N

!>          N هو INTEGER
!>          رتبة المصفوفة A.  N >= 0.
!>

A

!>          A هي مصفوفة COMPLEX، بأبعاد ( LDA, N )
!>          عند الإدخال، تحتوي المصفوفة المثلثية A.  إذا كان UPLO = 'U'،
!>          فإن الجزء العلوي المثلثي من A بأبعاد N-by-N يحتوي
!>          على المصفوفة المثلثية العليا، ولا يتم الرجوع إلى الجزء
!>          المثلثي السفلي الصارم من A.  إذا كان UPLO = 'L'، فإن الجزء
!>          المثلثي السفلي من A بأبعاد N-by-N يحتوي
!>          على المصفوفة المثلثية السفلى، ولا يتم الرجوع إلى الجزء
!>          المثلثي العلوي الصارم من A.
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرائد للمصفوفة A.  LDA >= max(1,N).
!>

ARF

!>          ARF هي مصفوفة COMPLEX، بأبعاد ( N*(N+1)/2 )،
!>          عند الخروج، تحتوي المصفوفة المثلثية العليا أو السفلى A المخزنة
!>          بتنسيق RFP. لمناقشة أخرى، راجع الملاحظات أدناه.
!>

INFO

!>          INFO هو INTEGER
!>          = 0:  اكتمال ناجح
!>          < 0:  إذا كان INFO = -i، فإن الوسيط i كان له قيمة غير قانونية
!>

تفاصيل إضافية:

!>
!>  ننظر أولاً إلى التنسيق المعبأ القياسي عندما يكون N زوجيًا.
!>  نعطي مثالًا حيث N = 6.
!>
!>      AP هي عليا             AP هي سفلى
!>
!>   00 01 02 03 04 05       00
!>      11 12 13 14 15       10 11
!>         22 23 24 25       20 21 22
!>            33 34 35       30 31 32 33
!>               44 45       40 41 42 43 44
!>                  55       50 51 52 53 54 55
!>
!>
!>  إذا كان TRANSR = 'N'. تحتوي RFP على AP كما يلي:
!>  بالنسبة لـ UPLO = 'U'، فإن شبه المنحرف العلوي A(0:5,0:2) يتكون من
!>  الأعمدة الثلاثة الأخيرة من AP العليا. المثلث السفلي A(4:6,0:2) يتكون من
!>  مرافق منقول للأعمدة الثلاثة الأولى من AP العليا.
!>  بالنسبة لـ UPLO = 'L'، فإن شبه المنحرف السفلي A(1:6,0:2) يتكون من
!>  الأعمدة الثلاثة الأولى من AP السفلى. المثلث العلوي A(0:2,0:2) يتكون من
!>  مرافق منقول للأعمدة الثلاثة الأخيرة من AP السفلى.
!>  للإشارة إلى المرافق، نضع -- فوق العنصر. هذا يغطي
!>  الحالة N زوجي و TRANSR = 'N'.
!>
!>         RFP A                   RFP A
!>
!>                                -- -- --
!>        03 04 05                33 43 53
!>                                   -- --
!>        13 14 15                00 44 54
!>                                      --
!>        23 24 25                10 11 55
!>
!>        33 34 35                20 21 22
!>        --
!>        00 44 45                30 31 32
!>        -- --
!>        01 11 55                40 41 42
!>        -- -- --
!>        02 12 22                50 51 52
!>
!>  الآن إذا كان TRANSR = 'C'. RFP A في كلتا حالتي UPLO هي مجرد
!>  مرافق منقول لـ RFP A أعلاه. وبالتالي نحصل على:
!>
!>
!>           RFP A                   RFP A
!>
!>     -- -- -- --                -- -- -- -- -- --
!>     03 13 23 33 00 01 02    33 00 10 20 30 40 50
!>     -- -- -- -- --                -- -- -- -- --
!>     04 14 24 34 44 11 12    43 44 11 21 31 41 51
!>     -- -- -- -- -- --                -- -- -- --
!>     05 15 25 35 45 55 22    53 54 55 22 32 42 52
!>
!>
!>  بعد ذلك ننظر إلى التنسيق المعبأ القياسي عندما يكون N فرديًا.
!>  نعطي مثالًا حيث N = 5.
!>
!>     AP هي عليا                 AP هي سفلى
!>
!>   00 01 02 03 04              00
!>      11 12 13 14              10 11
!>         22 23 24              20 21 22
!>            33 34              30 31 32 33
!>               44              40 41 42 43 44
!>
!>
!>  إذا كان TRANSR = 'N'. تحتوي RFP على AP كما يلي:
!>  بالنسبة لـ UPLO = 'U'، فإن شبه المنحرف العلوي A(0:4,0:2) يتكون من
!>  الأعمدة الثلاثة الأخيرة من AP العليا. المثلث السفلي A(3:4,0:1) يتكون من
!>  مرافق منقول للأعمدة الاثنين الأخيرتين من AP العليا.
!>  بالنسبة لـ UPLO = 'L'، فإن شبه المنحرف السفلي A(0:4,0:2) يتكون من
!>  الأعمدة الثلاثة الأولى من AP السفلى. المثلث العلوي A(0:1,1:2) يتكون من
!>  مرافق منقول للأعمدة الاثنين الأخيرتين من AP السفلى.
!>  للإشارة إلى المرافق، نضع -- فوق العنصر. هذا يغطي
!>  الحالة N فردي و TRANSR = 'N'.
!>
!>         RFP A                   RFP A
!>
!>                                   -- --
!>        02 03 04                00 33 43
!>                                      --
!>        12 13 14                10 11 44
!>
!>        22 23 24                20 21 22
!>        --
!>        00 33 34                30 31 32
!>        -- --
!>        01 11 44                40 41 42
