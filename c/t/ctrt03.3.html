subroutine ctrt03 (character uplo, character trans, character diag, integer n, integer nrhs, complex, dimension( lda, * ) a, integer lda, real scale, real, dimension( * ) cnorm, real tscal, complex, dimension( ldx, * ) x, integer ldx, complex, dimension( ldb, * ) b, integer ldb, complex, dimension( * ) work, real resid)

CTRT03

الغرض:

!>
!> CTRT03 يحسب الباقي لحل نظام مثلثي معادقي القياس
!> A*x = s*b,  A**T *x = s*b,  أو  A**H *x = s*b.
!> هنا A هي مصفوفة مثلثية، A**T تمثل منقول A، A**H
!> تمثل المرافق المنقول لـ A، s هو قياسي، و x و b هما
!> مصفوفتان N × NRHS. نسبة الاختبار هي الحد الأقصى لعدد
!> الأطراف اليمنى لـ
!>    norm(s*b - op(A)*x) / ( norm(op(A)) * norm(x) * EPS ),
!> حيث op(A) يمثل A، A**T، أو A**H، و EPS هو خطأ الآلة.
!>

المعلمات

UPLO 
!>          UPLO هو CHARACTER*1
!>          يحدد ما إذا كانت المصفوفة A علوية أم سفلية مثلثية.
!>          = 'U':  علوية مثلثية
!>          = 'L':  سفلية مثلثية
!>

TRANS

!>          TRANS هو CHARACTER*1
!>          يحدد العملية المطبقة على A.
!>          = 'N':  A *x = s*b     (بدون نقل)
!>          = 'T':  A**T *x = s*b  (منقول)
!>          = 'C':  A**H *x = s*b  (مرافق منقول)
!>

DIAG

!>          DIAG هو CHARACTER*1
!>          يحدد ما إذا كانت المصفوفة A مثلثية الوحدة أم لا.
!>          = 'N':  غير الوحدة المثلثية
!>          = 'U':  الوحدة المثلثية
!>

N

!>          N هو INTEGER
!>          رتبة المصفوفة A.  N >= 0.
!>

NRHS

!>          NRHS هو INTEGER
!>          عدد الأطراف اليمنى، أي عدد أعمدة
!>          المصفوفتين X و B.  NRHS >= 0.
!>

A

!>          A هو مصفوفة COMPLEX، أبعاد (LDA,N)
!>          المصفوفة المثلثية A.  إذا كان UPLO = 'U'، فإن الجزء العلوي
!>          المثلثي العلوي بحجم n × n في المصفوفة A يحتوي على
!>          المصفوفة المثلثية العلوية، والجزء السفلي المثلثي الصارم
!>          من A غير مستخدم.  إذا كان UPLO = 'L'، فإن الجزء السفلي
!>          المثلثي السفلي بحجم n × n في المصفوفة A يحتوي على
!>          المصفوفة المثلثية السفلية، والجزء العلوي المثلثي الصارم
!>          من A غير مستخدم.  إذا كان DIAG = 'U'، فإن عناصر القطر
!>          لـ A غير مستخدمة أيضًا ويفترض أنها 1.
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرائد للمصفوفة A.  LDA >= max(1,N).
!>

SCALE

!>          SCALE هو REAL
!>          عامل القياس s المستخدم في حل النظام المثلثي.
!>

CNORM

!>          CNORM هو مصفوفة REAL، أبعاد (N)
!>          المعايير 1 لأعمدة A، لا تشمل القطر.
!>

TSCAL

!>          TSCAL هو REAL
!>          عامل القياس المستخدم في حساب معايير 1 في CNORM.
!>          CNORM في الواقع يحتوي على معايير أعمدة TSCAL*A.
!>

X

!>          X هي مصفوفة COMPLEX، أبعاد (LDX,NRHS)
!>          متجهات الحل المحسوبة لأنظمة المعادلات الخطية.
!>

LDX

!>          LDX هو INTEGER
!>          البعد الرائد للمصفوفة X.  LDX >= max(1,N).
!>

B

!>          B هي مصفوفة COMPLEX، أبعاد (LDB,NRHS)
!>          متجهات الطرف الأيمن لأنظمة المعادلات الخطية.
!>

LDB

!>          LDB هو INTEGER
!>          البعد الرائد للمصفوفة B.  LDB >= max(1,N).
!>

WORK

!>          WORK هي مصفوفة COMPLEX، أبعاد (N)
!>

RESID

!>          RESID هو REAL
!>          الحد الأقصى لعدد الأطراف اليمنى لـ
!>          norm(op(A)*x - s*b) / ( norm(op(A)) * norm(x) * EPS ).
!>
