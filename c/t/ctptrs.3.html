subroutine ctptrs (character uplo, character trans, character diag, integer n, integer nrhs, complex, dimension( * ) ap, complex, dimension( ldb, * ) b, integer ldb, integer info)

CTPTRS

الغرض:

!>
!> CTPTRS تحل نظامًا مثلثيًا بالشكل:
!>
!>    A * X = B,  A**T * X = B,  أو  A**H * X = B,
!>
!> حيث A هي مصفوفة مثلثية من الرتبة N مخزنة بتنسيق مضغوط، و B هي مصفوفة N-by-NRHS.
!>
!> هذا الروتين الفرعي يتحقق من أن A غير منفردة، ولكن يجب على المتصلين ملاحظة أن التفرد المطلق فقط هو ما يتم اكتشافه. من الممكن أن تكون عناصر قطرية واحدة أو أكثر في A أرقامًا صغيرة جدًا دون أن يشير هذا الروتين الفرعي إلى خطأ.
!>
!> إذا كان فقدان الدقة العددية بسبب المصفوفات شبه المنفردة يمثل مصدر قلق، فيجب على المتصل التحقق من أن A غير منفردة ضمن بعض التفاوت قبل استدعاء هذا الروتين الفرعي.
!>

المعلمات

UPLO 
!>          UPLO هو CHARACTER*1
!>          = 'U':  A هي مثلثية عليا؛
!>          = 'L':  A هي مثلثية سفلى.
!>

TRANS

!>          TRANS هو CHARACTER*1
!>          يحدد شكل نظام المعادلات:
!>          = 'N':  A * X = B     (بدون تبديل)
!>          = 'T':  A**T * X = B  (تبديل)
!>          = 'C':  A**H * X = B  (تبديل مترافق)
!>

DIAG

!>          DIAG هو CHARACTER*1
!>          = 'N':  A ليست مثلثية وحدوية؛
!>          = 'U':  A هي مثلثية وحدوية.
!>

N

!>          N هو INTEGER
!>          ترتيب المصفوفة A.  N >= 0.
!>

NRHS

!>          NRHS هو INTEGER
!>          عدد الأطراف اليمنى، أي عدد الأعمدة
!>          للمصفوفة B.  NRHS >= 0.
!>

AP

!>          AP هو مصفوفة COMPLEX، البعد (N*(N+1)/2)
!>          المصفوفة المثلثية العليا أو السفلى A، مضغوطة حسب الأعمدة في
!>          مصفوفة خطية.  يتم تخزين العمود j من A في
!>          AP بالشكل التالي:
!>          إذا كان UPLO = 'U'، فإن AP(i + (j-1)*j/2) = A(i,j) لـ 1<=i<=j؛
!>          إذا كان UPLO = 'L'، فإن AP(i + (j-1)*(2*n-j)/2) = A(i,j) لـ j<=i<=n.
!>

B

!>          B هي مصفوفة COMPLEX، البعد (LDB,NRHS)
!>          عند الإدخال، مصفوفة الطرف الأيمن B.
!>          عند الخروج، إذا كان INFO = 0، فإن مصفوفة الحل X.
!>

LDB

!>          LDB هو INTEGER
!>          البعد الرائد للمصفوفة B.  LDB >= max(1,N).
!>

INFO

!>          INFO هو INTEGER
!>          = 0:  خروج ناجح
!>          < 0:  إذا كان INFO = -i، فإن المعلمة i كانت لها قيمة غير قانونية
!>          > 0:  إذا كان INFO = i، فإن العنصر القطري i من A هو بالضبط صفر،
!>                مما يشير إلى أن المصفوفة منفردة وأن
!>                الحلول X لم يتم حسابها.
!>
