subroutine ctrsyl (character trana, character tranb, integer isgn, integer m, integer n, complex, dimension( lda, * ) a, integer lda, complex, dimension( ldb, * ) b, integer ldb, complex, dimension( ldc, * ) c, integer ldc, real scale, integer info)

CTRSYL

الغرض:

!>
!> CTRSYL يحل معادلة مصفوفة سيلفستر المعقدة:
!>
!>    op(A)*X + X*op(B) = scale*C أو
!>    op(A)*X - X*op(B) = scale*C,
!>
!> حيث op(A) = A أو A**H، و A و B كلاهما مثلثيان علويان. A
!> بحجم M*M و B بحجم N*N؛ والجانب الأيمن C والحل X هما
!> بحجم M*N؛ و scale هو عامل قياس للمخرجات، يتم تعيينه <= 1 لتجنب
!> الحمل الزائد في X.
!>

المعلمات

TRANA 
!>          TRANA هو CHARACTER*1
!>          يحدد الخيار op(A):
!>          = 'N': op(A) = A    (بدون تبديل)
!>          = 'C': op(A) = A**H (تبديل مقترن)
!>

TRANB

!>          TRANB هو CHARACTER*1
!>          يحدد الخيار op(B):
!>          = 'N': op(B) = B    (بدون تبديل)
!>          = 'C': op(B) = B**H (تبديل مقترن)
!>

ISGN

!>          ISGN هو INTEGER
!>          يحدد الإشارة في المعادلة:
!>          = +1: حل op(A)*X + X*op(B) = scale*C
!>          = -1: حل op(A)*X - X*op(B) = scale*C
!>

M

!>          M هو INTEGER
!>          رتبة المصفوفة A، وعدد الصفوف في
!>          المصفوفتين X و C. M >= 0.
!>

N

!>          N هو INTEGER
!>          رتبة المصفوفة B، وعدد الأعمدة في
!>          المصفوفتين X و C. N >= 0.
!>

A

!>          A هو مصفوفة COMPLEX، البعد (LDA,M)
!>          المصفوفة المثلثية العليا A.
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرائد للمصفوفة A. LDA >= max(1,M).
!>

B

!>          B هو مصفوفة COMPLEX، البعد (LDB,N)
!>          المصفوفة المثلثية العليا B.
!>

LDB

!>          LDB هو INTEGER
!>          البعد الرائد للمصفوفة B. LDB >= max(1,N).
!>

C

!>          C هو مصفوفة COMPLEX، البعد (LDC,N)
!>          عند الإدخال، المصفوفة C ذات الجانب الأيمن بحجم M*N.
!>          عند الخروج، يتم استبدال C بالمصفوفة الحل X.
!>

LDC

!>          LDC هو INTEGER
!>          البعد الرائد للمصفوفة C. LDC >= max(1,M)
!>

SCALE

!>          SCALE هو REAL
!>          عامل القياس، scale، يتم تعيينه <= 1 لتجنب الحمل الزائد في X.
!>

INFO

!>          INFO هو INTEGER
!>          = 0: خروج ناجح
!>          < 0: إذا كان INFO = -i، فإن الوسيط i كان له قيمة غير قانونية
!>          = 1: A و B لهما قيم ذاتية مشتركة أو قريبة جدًا؛ تم استخدام قيم مضطربة
!>               لحل المعادلة (ولكن المصفوفتين A و B لم تتغير).
!>
