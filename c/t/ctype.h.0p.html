ctype.h(0P) دليل مبرمج POSIX ctype.h(0P)

مقدمة

هذه الصفحة الدليلية هي جزء من دليل مبرمج POSIX. قد يختلف تنفيذ هذا الواجهة في لينكس (راجع صفحة دليل لينكس المقابلة لتفاصيل سلوك لينكس)، أو قد لا يتم تنفيذ الواجهة على لينكس.

الاسم

ctype.h — أنواع الأحرف

الملخص

#include <ctype.h>

الوصف

بعض الوظائف الموضحة في صفحة المرجع هذه توسع معيار ISO C. يجب على التطبيقات تحديد ماكرو اختبار الميزة المناسب (راجع مجلد واجهات النظام من POSIX.1-2017، القسم 2.2، بيئة الترجمة) لتمكين ظهور هذه الرموز في هذا الرأس.

يجب على الرأس <ctype.h> تعريف نوع locale_t كما هو موضح في <locale.h>، والذي يمثل كائن موقع.

يجب الإعلان عن ما يلي كدوال وقد يتم تعريفها أيضًا كـ macros. سيتم توفير نماذج الدوال لاستخدامها مع مترجمات معيار ISO C.

int   isalnum(int);
int   isalnum_l(int, locale_t);
int   isalpha(int);
int   isalpha_l(int, locale_t);
int   isascii(int);
int   isblank(int);
int   isblank_l(int, locale_t);
int   iscntrl(int);
int   iscntrl_l(int, locale_t);
int   isdigit(int);
int   isdigit_l(int, locale_t);
int   isgraph(int);
int   isgraph_l(int, locale_t);
int   islower(int);
int   islower_l(int, locale_t);
int   isprint(int);
int   isprint_l(int, locale_t);
int   ispunct(int);
int   ispunct_l(int, locale_t);
int   isspace(int);
int   isspace_l(int, locale_t);
int   isupper(int);
int   isupper_l(int, locale_t);
int   isxdigit(int);
int   isxdigit_l(int, locale_t);
int   toascii(int);
int   tolower(int);
int   tolower_l(int, locale_t);
int   toupper(int);
int   toupper_l(int, locale_t);

يجب على الرأس <ctype.h> تعريف ما يلي كـ macros:

int   _toupper(int);
int   _tolower(int);
