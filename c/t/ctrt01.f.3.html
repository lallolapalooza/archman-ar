توثيق الوظيفة/الإجراء الفرعي

subroutine ctrt01 (character uplo, character diag, integer n, complex, dimension( lda, * ) a, integer lda, complex, dimension( ldainv, * ) ainv, integer ldainv, real rcond, real, dimension( * ) rwork, real resid)

CTRT01

الغرض:

!>
!> CTRT01 يحسب الباقي لمصفوفة مثلثة A مضروبة في معكوسها:
!>    RESID = norm( A*AINV - I ) / ( N * norm(A) * norm(AINV) * EPS ),
!> حيث EPS هو خطأ الآلة.
!>

المعاملات

UPLO 
!>          UPLO هو CHARACTER*1
!>          يحدد ما إذا كانت المصفوفة A علوية أو سفلية.
!>          = 'U':  علوية
!>          = 'L':  سفلية
!>

DIAG

!>          DIAG هو CHARACTER*1
!>          يحدد ما إذا كانت المصفوفة A موحدة أم لا.
!>          = 'N':  غير موحدة
!>          = 'U':  موحدة
!>

N

!>          N هو INTEGER
!>          ترتيب المصفوفة A.  N >= 0.
!>

A

!>          A هو مصفوفة COMPLEX، البعد (LDA,N)
!>          المصفوفة المثلثة A.  إذا كان UPLO = 'U'، فإن الجزء العلوي المثلثي
!>          من المصفوفة A بحجم N × N يحتوي على المصفوفة المثلثة العلوية،
!>          ولا تتم الإشارة إلى الجزء السفلي المثلثي الصارم من A.
!>          إذا كان UPLO = 'L'، فإن الجزء السفلي المثلثي من المصفوفة A بحجم N × N
!>          يحتوي على المصفوفة المثلثة السفلية، ولا تتم الإشارة إلى الجزء العلوي
!>          المثلثي الصارم من A.  إذا كان DIAG = 'U'، فإن العناصر القطرية لـ A
!>          لا تتم الإشارة إليها أيضًا ويفترض أنها 1.
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرئيسي للمصفوفة A.  LDA >= max(1,N).
!>

AINV

!>          AINV هو مصفوفة COMPLEX، البعد (LDAINV,N)
!>          عند الدخول، معكوس المصفوفة A (المثلثة) ، بنفس تنسيق التخزين
!>          مثل A.
!>          عند الخروج، يتم تدمير محتويات AINV.
!>

LDAINV

!>          LDAINV هو INTEGER
!>          البعد الرئيسي للمصفوفة AINV.  LDAINV >= max(1,N).
!>

RCOND

!>          RCOND هو REAL
!>          مقلوب رقم الشرط لـ A، محسوب كـ
!>          1/(norm(A) * norm(AINV)).
!>

RWORK

!>          RWORK هو مصفوفة REAL، البعد (N)
!>

RESID

!>          RESID هو REAL
!>          norm(A*AINV - I) / ( N * norm(A) * norm(AINV) * EPS )
!>
