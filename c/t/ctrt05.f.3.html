subroutine ctrt05 (character uplo, character trans, character diag, integer n, integer nrhs, complex, dimension( lda, * ) a, integer lda, complex, dimension( ldb, * ) b, integer ldb, complex, dimension( ldx, * ) x, integer ldx, complex, dimension( ldxact, * ) xact, integer ldxact, real, dimension( * ) ferr, real, dimension( * ) berr, real, dimension( * ) reslts)

CTRT05

الغرض:

!>
!> CTRT05 تختبر حدود الخطأ من التنقيح التكراري للحل
!> المحسوب لنظام المعادلات A*X = B، حيث A هي مصفوفة مثلثية n × n.
!>
!> RESLTS(1) = اختبار حد الخطأ
!>           = norm(X - XACT) / ( norm(X) * FERR )
!>
!> يتم إرجاع قيمة كبيرة إذا لم تكن هذه النسبة أقل من واحد.
!>
!> RESLTS(2) = الباقي من روتين التنقيح التكراري
!>           = الحد الأقصى لـ BERR / ( (n+1)*EPS + (*) )، حيث
!>             (*) = (n+1)*UNFL / (min_i (abs(A)*abs(X) +abs(b))_i )
!>

المعلمات:

UPLO 
!>          UPLO هو CHARACTER*1
!>          يحدد ما إذا كانت المصفوفة A علوية أم سفلية مثلثية.
!>          = 'U':  مثلثية علوية
!>          = 'L':  مثلثية سفلية
!>

TRANS

!>          TRANS هو CHARACTER*1
!>          يحدد شكل نظام المعادلات.
!>          = 'N':  A * X = B  (بدون تبديل)
!>          = 'T':  A'* X = B  (تبديل)
!>          = 'C':  A'* X = B  (تبديل مرافق = تبديل)
!>

DIAG

!>          DIAG هو CHARACTER*1
!>          يحدد ما إذا كانت المصفوفة A مثلثية موحدة أم لا.
!>          = 'N':  غير موحدة مثلثية
!>          = 'U':  موحدة مثلثية
!>

N

!>          N هو INTEGER
!>          عدد الصفوف في المصفوفات X، B، و XACT، و
!>          ترتيب المصفوفة A.  N >= 0.
!>

NRHS

!>          NRHS هو INTEGER
!>          عدد الأعمدة في المصفوفات X، B، و XACT.
!>          NRHS >= 0.
!>

A

!>          A هي مصفوفة COMPLEX، حجم (LDA,N)
!>          المصفوفة المثلثية A.  إذا كان UPLO = 'U'، فإن الجزء العلوي
!>          المثلثي الذي حجمه n × n من المصفوفة A يحتوي على المصفوفة
!>          المثلثية العلوية، ولا يتم الرجوع إلى الجزء المثلثي السفلي
!>          الصارم من A.  إذا كان UPLO = 'L'، فإن الجزء السفلي
!>          المثلثي الذي حجمه n × n من المصفوفة A يحتوي على المصفوفة
!>          المثلثية السفلية، ولا يتم الرجوع إلى الجزء المثلثي العلوي
!>          الصارم من A.  إذا كان DIAG = 'U'، فإن عناصر القطر للمصفوفة A
!>          لا يتم الرجوع إليها أيضًا وتُفترض أنها 1.
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرائد للمصفوفة A.  LDA >= max(1,N).
!>

B

!>          B هي مصفوفة COMPLEX، حجم (LDB,NRHS)
!>          متجهات الطرف الأيمن لنظام المعادلات الخطية.
!>

LDB

!>          LDB هو INTEGER
!>          البعد الرائد للمصفوفة B.  LDB >= max(1,N).
!>

X

!>          X هي مصفوفة COMPLEX، حجم (LDX,NRHS)
!>          متجهات الحل المحسوبة.  يتم تخزين كل متجه كعمود في المصفوفة X.
!>

LDX

!>          LDX هو INTEGER
!>          البعد الرائد للمصفوفة X.  LDX >= max(1,N).
!>

XACT

!>          XACT هي مصفوفة COMPLEX، حجم (LDX,NRHS)
!>          متجهات الحل الدقيقة.  يتم تخزين كل متجه كعمود في المصفوفة XACT.
!>

LDXACT

!>          LDXACT هو INTEGER
!>          البعد الرائد للمصفوفة XACT.  LDXACT >= max(1,N).
!>

FERR

!>          FERR هي مصفوفة REAL، حجم (NRHS)
!>          تقديرات حدود الخطأ الأمامي لكل متجه حل
!>          X.  إذا كان XTRUE هو الحل الحقيقي، فإن FERR يضع حدًا للقيمة المطلقة
!>          لأكبر مدخل في (X - XTRUE) مقسومًا على القيمة المطلقة
!>          لأكبر مدخل في X.
!>

BERR

!>          BERR هي مصفوفة REAL، حجم (NRHS)
!>          الخطأ الخلفي النسبي المكون لكل متجه حل
!>          (أي، أصغر تغيير نسبي في أي مدخل في A أو B يجعل X حلاً دقيقًا).
!>

RESLTS

!>          RESLTS هي مصفوفة REAL، حجم (2)
!>          الحد الأقصى لمتجهات الحل NRHS للنسب التالية:
!>          RESLTS(1) = norm(X - XACT) / ( norm(X) * FERR )
!>          RESLTS(2) = BERR / ( (n+1)*EPS + (*) )
!>
