subroutine ctrmv (character uplo, character trans, character diag, integer n, complex, dimension(lda,*) a, integer lda, complex, dimension(*) x, integer incx)

CTRMV

الغرض:

!>
!> CTRMV  تقوم بإحدى عمليات المتجه المصفوفي التالية:
!>
!>    x := A*x,   أو   x := A**T*x,   أو   x := A**H*x,
!>
!> حيث x هو متجه ذو n عنصر و A هي مصفوفة مثلثة علوية أو سفلية، وحدة أو غير وحدة، ذات الأبعاد n × n.
!>

المعلمات

UPLO 
!>          UPLO هو CHARACTER*1
!>           عند الإدخال، يحدد UPLO ما إذا كانت المصفوفة مصفوفة علوية أو
!>           سفلية على النحو التالي:
!>
!>              UPLO = 'U' أو 'u'   A هي مصفوفة مثلثة علوية.
!>
!>              UPLO = 'L' أو 'l'   A هي مصفوفة مثلثة سفلية.
!>

TRANS

!>          TRANS هو CHARACTER*1
!>           عند الإدخال، يحدد TRANS العملية التي سيتم تنفيذها على النحو التالي:
!>
!>              TRANS = 'N' أو 'n'   x := A*x.
!>
!>              TRANS = 'T' أو 't'   x := A**T*x.
!>
!>              TRANS = 'C' أو 'c'   x := A**H*x.
!>

DIAG

!>          DIAG هو CHARACTER*1
!>           عند الإدخال، يحدد DIAG ما إذا كانت A وحدة مثلثة أم لا،
!>           على النحو التالي:
!>
!>              DIAG = 'U' أو 'u'   يفترض أن A هي وحدة مثلثة.
!>
!>              DIAG = 'N' أو 'n'   لا يُفترض أن A هي وحدة مثلثة.
!>

N

!>          N هو INTEGER
!>           عند الإدخال، يحدد N ترتيب المصفوفة A.
!>           يجب أن تكون N على الأقل صفر.
!>

A

!>          A هو مصفوفة COMPLEX، أبعادها ( LDA, N ).
!>           قبل الدخول مع UPLO = 'U' أو 'u'، يجب أن يحتوي الجزء العلوي الأيسر ذو الأبعاد n × n
!>           من المصفوفة A على المصفوفة المثلثة العلوية، والجزء السفلي الصارم
!>           من A غير مرجعي.
!>           قبل الدخول مع UPLO = 'L' أو 'l'، يجب أن يحتوي الجزء السفلي الأيسر ذو الأبعاد n × n
!>           من المصفوفة A على المصفوفة المثلثة السفلية، والجزء العلوي الصارم
!>           من A غير مرجعي.
!>           لاحظ أنه عندما يكون DIAG = 'U' أو 'u'، فإن عناصر القطر للمصفوفة
!>           A غير مرجعية أيضًا، ولكن يُفترض أنها الوحدة.
!>

LDA

!>          LDA هو INTEGER
!>           عند الإدخال، يحدد LDA البعد الأول للمصفوفة A كما تم تعريفه
!>           في البرنامج (الفرعي) المستدعي. يجب أن تكون LDA على الأقل
!>           max( 1, n ).
!>

X

!>          X هو مصفوفة COMPLEX، بأبعاد على الأقل
!>           ( 1 + ( n - 1 )*abs( INCX ) ).
!>           قبل الدخول، يجب أن تحتوي المصفوفة المزايدة X على المتجه x ذي n عنصر. عند الخروج،
!>           يتم الكتابة فوق X بالمتجه المحول x.
!>

INCX

!>          INCX هو INTEGER
!>           عند الإدخال، يحدد INCX الزيادة لعناصر X. يجب ألا يكون INCX صفرًا.
!>
