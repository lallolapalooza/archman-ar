subroutine ctrt02 (character uplo, character trans, character diag, integer n, integer nrhs, complex, dimension( lda, * ) a, integer lda, complex, dimension( ldx, * ) x, integer ldx, complex, dimension( ldb, * ) b, integer ldb, complex, dimension( * ) work, real, dimension( * ) rwork, real resid)

CTRT02

الغرض:

!>
!> CTRT02 يحسب المتبقي للحل المحسوب لنظام خطي مثلثي op(A)*X = B, حيث A مصفوفة مثلثية. نسبة الاختبار هي الحد الأقصى على
!>    norm(b - op(A)*x) / ( ||op(A)||_1 * norm(x) * EPS ),
!> حيث op(A) = A, A**T, أو A**H, b هو عمود B, x هو متجه الحل، و EPS هو الخطأ المطبعي للجهاز.
!>

المعلمات

UPLO 
!>          UPLO is CHARACTER*1
!>          يحدد ما إذا كانت المصفوفة A مثلثة علوية أم سفلية.
!>          = 'U':  مثلثة علوية
!>          = 'L':  مثلثة سفلية
!>

TRANS

!>          TRANS is CHARACTER*1
!>          يحدد العملية المطبقة على A.
!>          = 'N':  A    * X = B  (بدون تبديل)
!>          = 'T':  A**T * X = B  (تبديل)
!>          = 'C':  A**H * X = B  (تبديل مرافق)
!>

DIAG

!>          DIAG is CHARACTER*1
!>          يحدد ما إذا كانت المصفوفة A مثلثة وحدوية أم لا.
!>          = 'N':  مثلثة غير وحدوية
!>          = 'U':  مثلثة وحدوية
!>

N

!>          N is INTEGER
!>          رتبة المصفوفة A.  N >= 0.
!>

NRHS

!>          NRHS is INTEGER
!>          عدد الأطراف اليمنى، أي عدد أعمدة المصفوفات X و B.  NRHS >= 0.
!>

A

!>          A is COMPLEX array, dimension (LDA,N)
!>          المصفوفة المثلثية A.  إذا كان UPLO = 'U'، فإن الجزء العلوي المثلثي العلوي الذي يبلغ حجمه n × n في المصفوفة A يحتوي على المصفوفة المثلثية العلوية، ولا يتم الرجوع إلى الجزء السفلي المثلثي الصارم لـ A. إذا كان UPLO = 'L'، فإن الجزء السفلي المثلثي السفلي الذي يبلغ حجمه n × n في المصفوفة A يحتوي على المصفوفة المثلثية السفلية، ولا يتم الرجوع إلى الجزء العلوي المثلثي الصارم لـ A. إذا كان DIAG = 'U'، فلا يتم الرجوع إلى عناصر القطر لـ A ويُفترض أنها 1.
!>

LDA

!>          LDA is INTEGER
!>          البعد الرئيسي للمصفوفة A.  LDA >= max(1,N).
!>

X

!>          X is COMPLEX array, dimension (LDX,NRHS)
!>          متجهات الحل المحسوبة لنظام المعادلات الخطية.
!>

LDX

!>          LDX is INTEGER
!>          البعد الرئيسي للمصفوفة X.  LDX >= max(1,N).
!>

B

!>          B is COMPLEX array, dimension (LDB,NRHS)
!>          متجهات الطرف الأيمن لنظام المعادلات الخطية.
!>

LDB

!>          LDB is INTEGER
!>          البعد الرئيسي للمصفوفة B.  LDB >= max(1,N).
!>

WORK

!>          WORK is COMPLEX array, dimension (N)
!>

RWORK

!>          RWORK is REAL array, dimension (N)
!>

RESID

!>          RESID is REAL
!>          الحد الأقصى على عدد الأطراف اليمنى لـ
!>          norm(op(A)*X - B) / ( norm(op(A)) * norm(X) * EPS ).
!>
