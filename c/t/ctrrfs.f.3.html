subroutine ctrrfs (character uplo, character trans, character diag, integer n, integer nrhs, complex, dimension( lda, * ) a, integer lda, complex, dimension( ldb, * ) b, integer ldb, complex, dimension( ldx, * ) x, integer ldx, real, dimension( * ) ferr, real, dimension( * ) berr, complex, dimension( * ) work, real, dimension( * ) rwork, integer info)

CTRRFS

الغرض:

!>
!> CTRRFS توفر حدود الخطأ وتقديرات الخطأ الخلفي لحل نظام المعادلات الخطية
!> مع مصفوفة معاملات مثلثية.
!>
!> يجب حساب مصفوفة الحل X بواسطة CTRTRS أو بأي وسيلة أخرى
!> قبل الدخول إلى هذا الروتين. CTRRFS لا تقوم بالتحسين التكراري
!> لأنه لا يمكن أن يؤدي إلى تحسين الخطأ الخلفي.
!>

المعلمات

UPLO 
!>          UPLO هو CHARACTER*1
!>          = 'U':  A هي مثلثية علوية؛
!>          = 'L':  A هي مثلثية سفلية.
!>

TRANS

!>          TRANS هو CHARACTER*1
!>          يحدد شكل نظام المعادلات:
!>          = 'N':  A * X = B     (بدون تبديل)
!>          = 'T':  A**T * X = B  (تبديل)
!>          = 'C':  A**H * X = B  (تبديل مترافق)
!>

DIAG

!>          DIAG هو CHARACTER*1
!>          = 'N':  A ليست مثلثية موحدة؛
!>          = 'U':  A هي مثلثية موحدة.
!>

N

!>          N هو INTEGER
!>          ترتيب المصفوفة A.  N >= 0.
!>

NRHS

!>          NRHS هو INTEGER
!>          عدد الأيادي اليمنى، أي عدد الأعمدة
!>          للمصفوفتين B و X.  NRHS >= 0.
!>

A

!>          A هي مصفوفة COMPLEX، البعد (LDA,N)
!>          المصفوفة المثلثية A.  إذا كانت UPLO = 'U'، فإن الجزء العلوي
!>          المثلثي العلوي N-by-N من المصفوفة A يحتوي على الجزء العلوي
!>          المثلثي، والجزء السفلي المثلثي الصارم
!>          من A غير مستخدم.  إذا كانت UPLO = 'L'، فإن الجزء السفلي N-by-N
!>          المثلثي السفلي من المصفوفة A يحتوي على الجزء السفلي المثلثي،
!>          والجزء العلوي المثلثي الصارم من A غير مستخدم.
!>          إذا كانت DIAG = 'U'، فإن عناصر القطر في A
!>          غير مستخدمة أيضًا وتفترض أنها 1.
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرائد للمصفوفة A.  LDA >= max(1,N).
!>

B

!>          B هي مصفوفة COMPLEX، البعد (LDB,NRHS)
!>          مصفوفة اليد اليمنى B.
!>

LDB

!>          LDB هو INTEGER
!>          البعد الرائد للمصفوفة B.  LDB >= max(1,N).
!>

X

!>          X هي مصفوفة COMPLEX، البعد (LDX,NRHS)
!>          مصفوفة الحل X.
!>

LDX

!>          LDX هو INTEGER
!>          البعد الرائد للمصفوفة X.  LDX >= max(1,N).
!>

FERR

!>          FERR هي مصفوفة REAL، البعد (NRHS)
!>          حد الخطأ الأمامي المقدر لكل متجه حل
!>          X(j) (العمود j من مصفوفة الحل X).
!>          إذا كان XTRUE هو الحل الحقيقي المقابل لـ X(j)، فإن FERR(j)
!>          هو حد علوي مقدر لقيمة أكبر
!>          عنصر في (X(j) - XTRUE) مقسومًا على قيمة أكبر
!>          عنصر في X(j).  التقدير موثوق به مثل
!>          التقدير لـ RCOND، وهو دائمًا تقريبًا
!>          تقدير مبالغ فيه للخطأ الحقيقي.
!>

BERR

!>          BERR هي مصفوفة REAL، البعد (NRHS)
!>          الخطأ الخلفي النسبي للمكون الواحد لكل متجه حل
!>          X(j) (أي، أصغر تغيير نسبي في
!>          أي عنصر من A أو B يجعل X(j) حلًا دقيقًا).
!>

WORK

!>          WORK هي مصفوفة COMPLEX، البعد (2*N)
!>

RWORK

!>          RWORK هي مصفوفة REAL، البعد (N)
!>

INFO

!>          INFO هو INTEGER
!>          = 0:  انتهاء ناجح
!>          < 0:  إذا كان INFO = -i، فإن المعامل i كان له قيمة غير قانونية
!>
