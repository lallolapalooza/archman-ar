subroutine ctplqt2 (integer m, integer n, integer l, complex, dimension( lda, * ) a, integer lda, complex, dimension( ldb, * ) b, integer ldb, complex, dimension( ldt, * ) t, integer ldt, integer info)

CTPLQT2

الغرض:

!>
!> CTPLQT2 يحسب تحليل LQ لمصفوفة معقدة C، والتي تتكون من مصفوفة مثلثية كتلية A ومصفوفة خماسية B،
!> باستخدام التمثيل المدمج WY لـ Q.
!>

المعاملات

M 
!>          M هو INTEGER
!>          العدد الإجمالي للصفوف في المصفوفة B.
!>          M >= 0.
!>

N

!>          N هو INTEGER
!>          عدد الأعمدة في المصفوفة B، وترتيب
!>          المصفوفة المثلثية A.
!>          N >= 0.
!>

L

!>          L هو INTEGER
!>          عدد الصفوف في الجزء شبه المنحرف السفلي من B.
!>          MIN(M,N) >= L >= 0. انظر التفاصيل الإضافية.
!>

A

!>          A هي مصفوفة COMPLEX، البعد (LDA,M)
!>          عند الإدخال، المصفوفة المثلثية السفلى M-by-M A.
!>          عند الخروج، تحتوي العناصر على القطر الرئيسي وتحته في المصفوفة
!>          على المصفوفة المثلثية السفلى L.
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرائد للمصفوفة A. LDA >= max(1,M).
!>

B

!>          B هي مصفوفة COMPLEX، البعد (LDB,N)
!>          عند الإدخال، المصفوفة الخماسية M-by-N B. الأعمدة N-L الأولى
!>          هي مستطيلة، والأعمدة L الأخيرة هي شبه منحرفة سفلى.
!>          عند الخروج، تحتوي B على المصفوفة الخماسية V. انظر التفاصيل الإضافية.
!>

LDB

!>          LDB هو INTEGER
!>          البعد الرائد للمصفوفة B. LDB >= max(1,M).
!>

T

!>          T هي مصفوفة COMPLEX، البعد (LDT,M)
!>          المصفوفة المثلثية العليا T ذات البعد N-by-N لمصفوفة الانعكاس الكتلية.
!>          انظر التفاصيل الإضافية.
!>

LDT

!>          LDT هو INTEGER
!>          البعد الرائد للمصفوفة T. LDT >= max(1,M)
!>

INFO

!>          INFO هو INTEGER
!>          = 0: الخروج بنجاح
!>          < 0: إذا كان INFO = -i، فإن المعامل i كان له قيمة غير قانونية
!>

التفاصيل الإضافية:

!>
!>  المصفوفة المدخلة C هي مصفوفة M-by-(M+N)
!>
!>               C = [ A ][ B ]
!>
!>
!>  حيث A هي مصفوفة مثلثية سفلى M-by-M، و B هي مصفوفة خماسية M-by-N
!>  تتكون من مصفوفة مستطيلة B1 مقاس M-by-(N-L) على يسار مصفوفة
!>  شبه منحرفة سفلى B2 مقاس M-by-L:
!>
!>               B = [ B1 ][ B2 ]
!>                   [ B1 ]  <-     مستطيلة M-by-(N-L)
!>                   [ B2 ]  <-     شبه منحرفة سفلى M-by-L.
!>
!>  تتكون المصفوفة شبه المنحرفة السفلى B2 من الأعمدة L الأولى لمصفوفة
!>  مثلثية سفلى N-by-N، حيث 0 <= L <= MIN(M,N). إذا كانت L=0،
!>  فإن B مستطيلة M-by-N؛ إذا كانت M=L=N، فإن B مثلثية سفلى.
!>
!>  المصفوفة W تخزن الانعكاسات الأولية H(i) في الصف i
!>  فوق القطر الرئيسي (لـ A) في المصفوفة المدخلة C ذات البعد (M+N)-by-(M+N)
!>
!>               C = [ A ][ B ]
!>                   [ A ]  <- مثلثية سفلى M-by-M
!>                   [ B ]  <- خماسية M-by-N
!>
!>  بحيث يمكن تمثيل W كـ
!>
!>               W = [ I ][ V ]
!>                   [ I ]  <- هوية، M-by-M
!>                   [ V ]  <- M-by-N، بنفس شكل B.
!>
!>  وبالتالي، فإن جميع المعلومات المطلوبة لـ W موجودة عند الخروج في B،
!>  والتي نسميها V أعلاه. لاحظ أن V لها نفس شكل B؛ أي،
!>
!>               W = [ V1 ][ V2 ]
!>                   [ V1 ] <-     مستطيلة M-by-(N-L)
!>                   [ V2 ] <-     شبه منحرفة سفلى M-by-L.
!>
!>  تمثل صفوف V المتجهات التي تحدد H(i).
!>  ثم يتم إعطاء مصفوفة الانعكاس الكتلية (M+N)-by-(M+N) H بواسطة
!>
!>               H = I - W**T * T * W
!>
!>  حيث W**T هو المرافق التبادلي لـ W و T هي العامل المثلثي العلوي
!>  لمصفوفة الانعكاس.
!>
