subroutine ctgsy2 (character trans, integer ijob, integer m, integer n, complex, dimension( lda, * ) a, integer lda, complex, dimension( ldb, * ) b, integer ldb, complex, dimension( ldc, * ) c, integer ldc, complex, dimension( ldd, * ) d, integer ldd, complex, dimension( lde, * ) e, integer lde, complex, dimension( ldf, * ) f, integer ldf, real scale, real rdsum, real rdscal, integer info)

CTGSY2 تحل معادلة سيلفستر المعممة (خوارزمية غير محظورة).

الغرض:

!>
!> CTGSY2 تحل معادلة سيلفستر المعممة
!>
!>             A * R - L * B = scale *  C               (1)
!>             D * R - L * E = scale * F
!>
!> باستخدام BLAS المستوى 1 و 2، حيث R و L مصفوفتان غير معروفتين بحجم M-by-N،
!> (A, D)، (B, E) و (C, F) هي أزواج مصفوفات معطاة بأبعاد M-by-M،
!> N-by-N و M-by-N، على التوالي. A، B، D و E مثلثة عليا
!> (أي، (A,D) و (B,E) في الشكل Schur المعمم).
!>
!> الحل (R, L) يستبدل (C, F). 0 <= SCALE <= 1 هو عامل قياس
!> خارجي تم اختياره لتجنب الفائض.
!>
!> في ترميز المصفوفة، حل المعادلة (1) يعادل حل
!> Zx = scale * b، حيث Z معرفة كـ
!>
!>        Z = [ kron(In, A)  -kron(B**H, Im) ]             (2)
!>            [ kron(In, D)  -kron(E**H, Im) ] ،
!>
!> Ik هي المصفوفة الهوية بحجم k و X**H هي منقول X.
!> kron(X, Y) هو ناتج كرونيكر بين المصفوفتين X و Y.
!>
!> إذا كان TRANS = 'C'، يتم حل y في النظام المترافق الزائدي Z**H*y = scale*b
!>، وهو ما يعادل الحل لـ R و L في
!>
!>             A**H * R  + D**H * L   = scale * C           (3)
!>             R  * B**H + L  * E**H  = scale * -F
!>
!> يتم استخدام هذه الحالة لحساب تقدير لـ Dif[(A, D), (B, E)] =
!> = sigma_min(Z) باستخدام الاتصال العكسي مع CLACON.
!>
!> CTGSY2 أيضاً (IJOB >= 1) تساهم في حساب CTGSYL
!> لحد أقصى تقديري للانفصال بين أزواج المصفوفات. ثم
!> المدخلات (A, D)، (B, E) هي أجزاء فرعية من أزواج مصفوفات في
!> CTGSYL.
!>

المعاملات

TRANS 
!>          TRANS is CHARACTER*1
!>          = 'N': حل معادلة سيلفستر المعممة (1).
!>          = 'T': حل النظام 'المتحول' (3).
!>

IJOB

!>          IJOB is INTEGER
!>          يحدد نوع الوظيفة التي سيتم تنفيذها.
!>          = 0: حل (1) فقط.
!>          = 1: يتم حساب مساهمة من هذا النظام الفرعي لتقدير يعتمد على معيار Frobenius
!>               لانفصال بين زوجين من المصفوفات. (يتم استخدام استراتيجية النظر إلى الأمام).
!>          = 2: يتم حساب مساهمة من هذا النظام الفرعي لتقدير يعتمد على معيار Frobenius
!>               لانفصال بين زوجين من المصفوفات. (يتم استخدام SGECON على الأنظمة الفرعية).
!>          لا يتم الرجوع إليه إذا كان TRANS = 'T'.
!>

M

!>          M is INTEGER
!>          عند الدخول، يحدد M رتبة A و D، والبعد الصف للمصفوفات C، F، R و L.
!>

N

!>          N is INTEGER
!>          عند الدخول، يحدد N رتبة B و E، والبعد العمود للمصفوفات C، F، R و L.
!>

A

!>          A is COMPLEX array, dimension (LDA, M)
!>          عند الدخول، تحتوي A على مصفوفة مثلثة عليا.
!>

LDA

!>          LDA is INTEGER
!>          البعد الرئيسي للمصفوفة A. LDA >= max(1, M).
!>

B

!>          B is COMPLEX array, dimension (LDB, N)
!>          عند الدخول، تحتوي B على مصفوفة مثلثة عليا.
!>

LDB

!>          LDB is INTEGER
!>          البعد الرئيسي للمصفوفة B. LDB >= max(1, N).
!>

C

!>          C is COMPLEX array, dimension (LDC, N)
!>          عند الدخول، تحتوي C على الطرف الأيمن للمعادلة الأولى في (1).
!>          عند الخروج، إذا كان IJOB = 0، تم استبدال C بالحل
!>          R.
!>

LDC

!>          LDC is INTEGER
!>          البعد الرئيسي للمصفوفة C. LDC >= max(1, M).
!>

D

!>          D is COMPLEX array, dimension (LDD, M)
!>          عند الدخول، تحتوي D على مصفوفة مثلثة عليا.
!>

LDD

!>          LDD is INTEGER
!>          البعد الرئيسي للمصفوفة D. LDD >= max(1, M).
!>

E

!>          E is COMPLEX array, dimension (LDE, N)
!>          عند الدخول، تحتوي E على مصفوفة مثلثة عليا.
!>

LDE

!>          LDE is INTEGER
!>          البعد الرئيسي للمصفوفة E. LDE >= max(1, N).
!>

F

!>          F is COMPLEX array, dimension (LDF, N)
!>          عند الدخول، تحتوي F على الطرف الأيمن للمعادلة الثانية في (1).
!>          عند الخروج، إذا كان IJOB = 0، تم استبدال F بالحل
!>          L.
!>

LDF

!>          LDF is INTEGER
!>          البعد الرئيسي للمصفوفة F. LDF >= max(1, M).
!>

SCALE

!>          SCALE is REAL
!>          عند الخروج، 0 <= SCALE <= 1. إذا كان 0 < SCALE < 1، فإن الحلول
!>          R و L (C و F عند الدخول) ستحتوي على حلول لنظام
!>          مضطرب قليلاً ولكن لم تتغير المصفوفات المدخلة A، B، D و
!>          E. إذا كان SCALE = 0، ستحتوي R و L على
!>          حلول للنظام المتجانس مع C = F = 0.
!>          عادة، SCALE = 1.
!>

RDSUM

!>          RDSUM is REAL
!>          عند الدخول، مجموع مربعات المساهمات المحسوبة في
!>          تقدير Dif قيد الحساب بواسطة CTGSYL، حيث
!>          تم استبعاد عامل القياس RDSCAL (انظر أدناه).
!>          عند الخروج، مجموع المربعات المقابل المحدث بـ
!>          المساهمات من النظام الفرعي الحالي.
!>          إذا كان TRANS = 'T'، لا يتم لمس RDSUM.
!>          ملاحظة: RDSUM له معنى فقط عندما يتم استدعاء CTGSY2 بواسطة
!>          CTGSYL.
!>

RDSCAL

!>          RDSCAL is REAL
!>          عند الدخول، عامل القياس المستخدم لمنع الفائض في RDSUM.
!>          عند الخروج، يتم تحديث RDSCAL فيما يتعلق بالمساهمات الحالية
!>          في RDSUM.
!>          إذا كان TRANS = 'T'، لا يتم لمس RDSCAL.
!>          ملاحظة: RDSCAL له معنى فقط عندما يتم استدعاء CTGSY2 بواسطة
!>          CTGSYL.
!>

INFO

!>          INFO is INTEGER
!>          عند الخروج، إذا تم تعيين INFO إلى
!>            =0: خروج ناجح
!>            <0: إذا كان INFO = -i، فإن رقم معامل الإدخال i غير قانوني.
!>            >0: أزواج المصفوفات (A, D) و (B, E) لها قيم ذاتية مشتركة أو قريبة جدًا.
!>
