subroutine ctpt03 (character uplo, character trans, character diag, integer n, integer nrhs, complex, dimension( * ) ap, real scale, real, dimension( * ) cnorm, real tscal, complex, dimension( ldx, * ) x, integer ldx, complex, dimension( ldb, * ) b, integer ldb, complex, dimension( * ) work, real resid)

CTPT03

الغرض:

!>
!> CTPT03 يحسب الباقي لحل نظام مثلثي معاد قياسه من المعادلات A*x = s*b,  A**T *x = s*b,  أو  A**H *x = s*b,
!> عندما يتم تخزين المصفوفة المثلثية A بتنسيق مضغوط. هنا A**T
!> يمثل منقول A، A**H يمثل المترافق المنقول لـ A، s هو ثابت قياس، و x و b هما مصفوفتان N × NRHS.
!> نسبة الاختبار هي الحد الأقصى على عدد الأيادي اليمنى
!>    norm(s*b - op(A)*x) / ( norm(op(A)) * norm(x) * EPS ),
!> حيث op(A) يمثل A، A**T، أو A**H، و EPS هو الجزء العشري من الآلة.
!>

المعلمات

UPLO 
!>          UPLO هو CHARACTER*1
!>          يحدد ما إذا كانت المصفوفة A مثلثية عليا أم سفلية.
!>          = 'U':  مثلثية عليا
!>          = 'L':  مثلثية سفلية
!>

TRANS

!>          TRANS هو CHARACTER*1
!>          يحدد العملية المطبقة على A.
!>          = 'N':  A *x = s*b     (بدون منقول)
!>          = 'T':  A**T *x = s*b  (منقول)
!>          = 'C':  A**H *x = s*b  (منقول مترافق)
!>

DIAG

!>          DIAG هو CHARACTER*1
!>          يحدد ما إذا كانت المصفوفة A مثلثية واحدية أم لا.
!>          = 'N':  غير واحدية
!>          = 'U':  واحدية
!>

N

!>          N هو INTEGER
!>          رتبة المصفوفة A.  N >= 0.
!>

NRHS

!>          NRHS هو INTEGER
!>          عدد الأيادي اليمنى، أي عدد أعمدة المصفوفتين X و B.  NRHS >= 0.
!>

AP

!>          AP هو مصفوفة COMPLEX، بأبعاد (N*(N+1)/2)
!>          المصفوفة المثلثية العليا أو السفلية A، مضغوطة حسب الأعمدة في
!>          مصفوفة خطية. يتم تخزين العمود j من A في
!>          المصفوفة AP على النحو التالي:
!>          إذا كان UPLO = 'U'، فإن AP((j-1)*j/2 + i) = A(i,j) لـ 1<=i<=j؛
!>          إذا كان UPLO = 'L'،
!>             AP((j-1)*(n-j) + j*(j+1)/2 + i-j) = A(i,j) لـ j<=i<=n.
!>

SCALE

!>          SCALE هو REAL
!>          عامل القياس s المستخدم في حل النظام المثلثي.
!>

CNORM

!>          CNORM هو مصفوفة REAL، بأبعاد (N)
!>          الـ 1-norms لأعمدة A، باستثناء القطر.
!>

TSCAL

!>          TSCAL هو REAL
!>          عامل القياس المستخدم في حساب الـ 1-norms في CNORM.
!>          CNORM يحتوي فعليًا على معايير أعمدة TSCAL*A.
!>

X

!>          X هي مصفوفة COMPLEX، بأبعاد (LDX,NRHS)
!>          متجهات الحل المحسوبة لنظام المعادلات الخطية.
!>

LDX

!>          LDX هو INTEGER
!>          البعد الرئيسي للمصفوفة X.  LDX >= max(1,N).
!>

B

!>          B هي مصفوفة COMPLEX، بأبعاد (LDB,NRHS)
!>          متجهات الطرف الأيمن لنظام المعادلات الخطية.
!>

LDB

!>          LDB هو INTEGER
!>          البعد الرئيسي للمصفوفة B.  LDB >= max(1,N).
!>

WORK

!>          WORK هي مصفوفة COMPLEX، بأبعاد (N)
!>

RESID

!>          RESID هو REAL
!>          الحد الأقصى على عدد الأيادي اليمنى
!>          norm(op(A)*x - s*b) / ( norm(op(A)) * norm(x) * EPS ).
!>
