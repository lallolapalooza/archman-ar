subroutine ctbsv (character uplo, character trans, character diag, integer n, integer k, complex, dimension(lda,*) a, integer lda, complex, dimension(*) x, integer incx)

CTBSV

الغرض:

!>
!> CTBSV  حل إحدى أنظمة المعادلات
!>
!>    A*x = b,   أو   A**T*x = b,   أو   A**H*x = b,
!>
!> حيث b و x هما متجهان بعنصر n ومصفوفة A هي مصفوفة مثلثة ذات نطاق n × n،
!> وحدة أو غير وحدة، علوية أو سفلية، ذات ( k + 1 ) قطري.
!>
!> لا يتم تضمين اختبار التفرد أو القرب من التفرد في هذا الإجراء. يجب
!> إجراء مثل هذه الاختبارات قبل استدعاء هذا الإجراء.
!>

المعلمات

UPLO 
!>          UPLO هو CHARACTER*1
!>           عند الإدخال، يحدد UPLO ما إذا كانت المصفوفة مصفوفة مثلثة علوية أو سفلية
!>           على النحو التالي:
!>
!>              UPLO = 'U' أو 'u'   A هي مصفوفة مثلثة علوية.
!>
!>              UPLO = 'L' أو 'l'   A هي مصفوفة مثلثة سفلية.
!>

TRANS

!>          TRANS هو CHARACTER*1
!>           عند الإدخال، يحدد TRANS المعادلات التي سيتم حلها على النحو التالي:
!>
!>              TRANS = 'N' أو 'n'   A*x = b.
!>
!>              TRANS = 'T' أو 't'   A**T*x = b.
!>
!>              TRANS = 'C' أو 'c'   A**H*x = b.
!>

DIAG

!>          DIAG هو CHARACTER*1
!>           عند الإدخال، يحدد DIAG ما إذا كانت A مثلثة وحدة أم لا على النحو التالي:
!>
!>              DIAG = 'U' أو 'u'   يفترض أن A مثلثة وحدة.
!>
!>              DIAG = 'N' أو 'n'   لا يُفترض أن A مثلثة وحدة.
!>

N

!>          N هو INTEGER
!>           عند الإدخال، يحدد N رتبة المصفوفة A.
!>           يجب أن يكون N على الأقل صفرًا.
!>

K

!>          K هو INTEGER
!>           عند الإدخال مع UPLO = 'U' أو 'u'، يحدد K عدد الأقطار فوق الرئيسية
!>           للمصفوفة A.
!>           عند الإدخال مع UPLO = 'L' أو 'l'، يحدد K عدد الأقطار الفرعية
!>           للمصفوفة A.
!>           يجب أن يحقق K  0 .le. K.
!>

A

!>          A هو مصفوفة COMPLEX، بأبعاد ( LDA, N )
!>           قبل الدخول مع UPLO = 'U' أو 'u'، يجب أن يحتوي الجزء ( k + 1 ) × N
!>           الرائد من المصفوفة A على الجزء العلوي المثلثي ذي النطاق للمصفوفة
!>           المعاملة، مقدمة عمودًا بعمود، مع القطر الرئيسي للمصفوفة في الصف
!>           ( k + 1 ) من المصفوفة، القطر العلوي الأول يبدأ من الموضع 2 في الصف k،
!>           وهكذا. لا يتم الرجوع إلى الجزء العلوي الأيسر للمصفوفة A بأبعاد k × k.
!>           سيقوم مقطع البرنامج التالي بنقل مصفوفة مثلثة ذات نطاق من تخزين
!>           المصفوفة الكاملة التقليدية إلى تخزين النطاق:
!>
!>                 DO 20, J = 1, N
!>                    M = K + 1 - J
!>                    DO 10, I = MAX( 1, J - K ), J
!>                       A( M + I, J ) = matrix( I, J )
!>              10    CONTINUE
!>              20 CONTINUE
!>
!>           قبل الدخول مع UPLO = 'L' أو 'l'، يجب أن يحتوي الجزء ( k + 1 ) × N
!>           الرائد من المصفوفة A على الجزء السفلي المثلثي ذي النطاق للمصفوفة
!>           المعاملة، مقدمة عمودًا بعمود، مع القطر الرئيسي للمصفوفة في الصف
!>           1 من المصفوفة، القطر الفرعي الأول يبدأ من الموضع 1 في الصف 2،
!>           وهكذا. لا يتم الرجوع إلى الجزء السفلي الأيمن للمصفوفة A بأبعاد k × k.
!>           سيقوم مقطع البرنامج التالي بنقل مصفوفة مثلثة ذات نطاق من تخزين
!>           المصفوفة الكاملة التقليدية إلى تخزين النطاق:
!>
!>                 DO 20, J = 1, N
!>                    M = 1 - J
!>                    DO 10, I = J, MIN( N, J + K )
!>                       A( M + I, J ) = matrix( I, J )
!>              10    CONTINUE
!>              20 CONTINUE
!>
!>           لاحظ أنه عندما يكون DIAG = 'U' أو 'u'، فإن عناصر المصفوفة A
!>           المقابلة لعناصر القطر الرئيسي للمصفوفة لا يتم الرجوع إليها، ولكن
!>           يفترض أنها تساوي واحدًا.
!>

LDA

!>          LDA هو INTEGER
!>           عند الإدخال، يحدد LDA البعد الأول لـ A كما هو معلن
!>           في البرنامج المستدعي (الفرعي). يجب أن يكون LDA على الأقل
!>           ( k + 1 ).
!>

X

!>          X هي مصفوفة COMPLEX، بأبعاد لا تقل عن
!>           ( 1 + ( n - 1 )*abs( INCX ) ).
!>           قبل الإدخال، يجب أن تحتوي المصفوفة X المتزايدة على المتجه الأيمن b
!>           ذي العناصر n. عند الخروج، يتم الكتابة فوق X بالمتجه الحل x.
!>

INCX

!>          INCX هو INTEGER
!>           عند الإدخال، يحدد INCX الزيادة لعناصر X. يجب ألا يكون INCX صفرًا.
!>
