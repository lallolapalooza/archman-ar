توثيق الوظيفة/الروتين الفرعي

subroutine ctbcon (character norm, character uplo, character diag, integer n, integer kd, complex, dimension( ldab, * ) ab, integer ldab, real rcond, complex, dimension( * ) work, real, dimension( * ) rwork, integer info)

CTBCON

الغرض:

!>
!> CTBCON يقدر مقلوب رقم الشرط لمصفوفة شريطية مثلثية A، سواء في معيار 1-norm أو
!> معيار اللانهاية.
!>
!> يتم حساب معيار A ويتم الحصول على تقدير لـ
!> norm(inv(A))، ثم يتم حساب مقلوب رقم الشرط كـ
!>    RCOND = 1 / ( norm(A) * norm(inv(A)) ).
!>

المعاملات

NORM 
!>          NORM هو CHARACTER*1
!>          يحدد ما إذا كان رقم الشرط 1-norm أو رقم الشرط لللانهاية
!>          مطلوبًا:
!>          = '1' أو 'O':  1-norm؛
!>          = 'I':         Infinity-norm.
!>

UPLO

!>          UPLO هو CHARACTER*1
!>          = 'U':  A مثلثية علوية؛
!>          = 'L':  A مثلثية سفلية.
!>

DIAG

!>          DIAG هو CHARACTER*1
!>          = 'N':  A غير وحدوية مثلثية؛
!>          = 'U':  A وحدوية مثلثية.
!>

N

!>          N هو INTEGER
!>          رتبة المصفوفة A.  N >= 0.
!>

KD

!>          KD هو INTEGER
!>          عدد الأقطار الفوقية أو الأقطار الفرعية لمصفوفة الشريط
!>          المثلثية A.  KD >= 0.
!>

AB

!>          AB هو مصفوفة COMPLEX، البعد (LDAB,N)
!>          المصفوفة الشريطية المثلثية العلوية أو السفلية A، مخزنة في
!>          الصفوف الـ kd+1 الأولى للمصفوفة. يتم تخزين العمود j من A
!>          في العمود j من المصفوفة AB كما يلي:
!>          إذا كان UPLO = 'U'، فإن AB(kd+1+i-j,j) = A(i,j) لـ max(1,j-kd)<=i<=j؛
!>          إذا كان UPLO = 'L'، فإن AB(1+i-j,j)    = A(i,j) لـ j<=i<=min(n,j+kd).
!>          إذا كان DIAG = 'U'، فإن العناصر القطرية لـ A لا تتم الإشارة إليها
!>          ويفترض أنها 1.
!>

LDAB

!>          LDAB هو INTEGER
!>          البعد الرئيسي للمصفوفة AB.  LDAB >= KD+1.
!>

RCOND

!>          RCOND هو REAL
!>          مقلوب رقم شرط المصفوفة A،
!>          محسوب كـ RCOND = 1/(norm(A) * norm(inv(A))).
!>

WORK

!>          WORK هي مصفوفة COMPLEX، البعد (2*N)
!>

RWORK

!>          RWORK هي مصفوفة REAL، البعد (N)
!>

INFO

!>          INFO هو INTEGER
!>          = 0:  انتهاء ناجح
!>          < 0:  إذا كان INFO = -i، فإن المعامل i كان له قيمة غير قانونية
!>
