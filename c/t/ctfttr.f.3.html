توثيق الوظيفة/الإجراء الفرعي

subroutine ctfttr (character transr, character uplo, integer n, complex, dimension( 0: * ) arf, complex, dimension( 0: lda-1, 0: * ) a, integer lda, integer info)

CTFTTR تنسخ مصفوفة مثلثة من التنسيق الكامل المعبأ المستطيل (TF) إلى التنسيق الكامل القياسي (TR).

الغرض:

!>
!> CTFTTR تنسخ مصفوفة مثلثة A من التنسيق الكامل المعبأ المستطيل
!> (TF) إلى التنسيق الكامل القياسي (TR).
!>

المعلمات

TRANSR 
!>          TRANSR هو CHARACTER*1
!>          = 'N':  ARF بالتنسيق العادي؛
!>          = 'C':  ARF بتنسيق المرافق-المبدل؛
!>

UPLO

!>          UPLO هو CHARACTER*1
!>          = 'U':  A مثلثة علوية؛
!>          = 'L':  A مثلثة سفلية.
!>

N

!>          N هو INTEGER
!>          رتبة المصفوفة A.  N >= 0.
!>

ARF

!>          ARF هو مصفوفة COMPLEX، البعد ( N*(N+1)/2 )،
!>          عند الإدخال، المصفوفة المثلثية A العلوية أو السفلية المخزنة
!>          بتنسيق RFP. لمناقشة إضافية انظر الملاحظات أدناه.
!>

A

!>          A هي مصفوفة COMPLEX، البعد ( LDA, N )
!>          عند الخروج، المصفوفة المثلثية A.  إذا كان UPLO = 'U'، فإن
!>          الجزء العلوي المثلثي الرائد N-by-N في المصفوفة A يحتوي
!>          على المصفوفة المثلثية العلوية، والجزء السفلي المثلثي
!>          الصرف لـ A لا تتم الإشارة إليه.  إذا كان UPLO = 'L'، فإن
!>          الجزء السفلي المثلثي الرائد N-by-N في المصفوفة A يحتوي
!>          على المصفوفة المثلثية السفلية، والجزء العلوي المثلثي
!>          الصرف لـ A لا تتم الإشارة إليه.
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرائد للمصفوفة A.  LDA >= max(1,N).
!>

INFO

!>          INFO هو INTEGER
!>          = 0:  اكتمال ناجح
!>          < 0:  إذا كان INFO = -i، فإن الوسيط i كان له قيمة غير قانونية
!>

!>
!>      AP علوي             AP سفلي
!>
!>   00 01 02 03 04 05       00

!>               44 45       40 41 42 43 44

!>
!>  دع TRANSR = 'N'. RFP يحتوي على AP كما يلي:
!>  بالنسبة لـ UPLO = 'U'، الشبه منحرف العلوي A(0:5,0:2) يتكون من الأعمدة
!>  الثلاثة الأخيرة من AP العلوي. المثلث السفلي A(4:6,0:2) يتكون من
!>  المرافق-المبدل للأعمدة الثلاثة الأولى من AP العلوي.
!>  بالنسبة لـ UPLO = 'L'، شبه المنحرف السفلي A(1:6,0:2) يتكون من الأعمدة
!>  الثلاثة الأولى من AP السفلي. المثلث العلوي A(0:2,0:2) يتكون من
!>  المرافق-المبدل للأعمدة الثلاثة الأخيرة من AP السفلي.
!>  للإشارة إلى المرافق، نضع -- فوق العنصر. وهذا يغطي
!>  حالة N زوجي و TRANSR = 'N'.
!>
!>         RFP A                   RFP A
!>

!>
!>     AP علوي                 AP سفلي

!>   00 01 02 03 04              00

!>  الثلاثة الأخيرة من AP العلوي. المثلث السفلي A(3:4,0:1) يتكون من
