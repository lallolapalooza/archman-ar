subroutine ctrt03 (character uplo, character trans, character diag, integer n, integer nrhs, complex, dimension( lda, * ) a, integer lda, real scale, real, dimension( * ) cnorm, real tscal, complex, dimension( ldx, * ) x, integer ldx, complex, dimension( ldb, * ) b, integer ldb, complex, dimension( * ) work, real resid)

CTRT03

الغرض:

!>
!> CTRT03 يحسب الباقي لحل نظام مثلثي معادلة القياس A*x = s*b,  A**T *x = s*b,  أو  A**H *x = s*b.
!> هنا A مصفوفة مثلثية، A**T تمثل منقول A، A**H تمثل منقول A المقترن، s هو ثابت عددي، و x و b هما مصفوفتان N في NRHS.
!> نسبة الاختبار هي الحد الأقصى على عدد الأطراف اليمنى من
!>    norm(s*b - op(A)*x) / ( norm(op(A)) * norm(x) * EPS ),
!> حيث op(A) تمثل A، A**T، أو A**H، و EPS هو خطأ الآلة.
!>

المعاملات

UPLO 
!>          UPLO هو CHARACTER*1
!>          تحدد ما إذا كانت المصفوفة A مثلثية عليا أم سفلى.
!>          = 'U':  مثلثية عليا
!>          = 'L':  مثلثية سفلى
!>

TRANS

!>          TRANS هو CHARACTER*1
!>          تحدد العملية المطبقة على A.
!>          = 'N':  A *x = s*b     (لا يوجد منقول)
!>          = 'T':  A**T *x = s*b  (منقول)
!>          = 'C':  A**H *x = s*b  (منقول مقترن)
!>

DIAG

!>          DIAG هو CHARACTER*1
!>          تحدد ما إذا كانت المصفوفة A مثلثية وحيدة أم لا.
!>          = 'N':  غير وحيدة
!>          = 'U':  وحيدة
!>

N

!>          N هو INTEGER
!>          رتبة المصفوفة A.  N >= 0.
!>

NRHS

!>          NRHS هو INTEGER
!>          عدد الأطراف اليمنى، أي عدد الأعمدة
!>          في المصفوفتين X و B.  NRHS >= 0.
!>

A

!>          A هو مصفوفة COMPLEX، ذات بعد (LDA,N)
!>          المصفوفة المثلثية A.  إذا كانت UPLO = 'U'، فإن الجزء العلوي الأيسر الذي
!>          هو n في n من المصفوفة A يحتوي على المصفوفة المثلثية العليا، والجزء السفلي
!>          المثلثي الصارم من A لا تتم الإشارة إليه.  إذا كانت UPLO = 'L'، فإن الجزء السفلي الأيسر
!>          الذي هو n في n من المصفوفة A يحتوي على المصفوفة المثلثية السفلى، والجزء العلوي
!>          المثلثي الصارم من A لا تتم الإشارة إليه.  إذا كانت DIAG = 'U'، فإن العناصر القطرية
!>          من A لا تتم الإشارة إليها ويُفترض أنها 1.
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرائد للمصفوفة A.  LDA >= max(1,N).
!>

SCALE

!>          SCALE هو REAL
!>          عامل القياس s المستخدم في حل النظام المثلثي.
!>

CNORM

!>          CNORM هو مصفوفة REAL، ذات بعد (N)
!>          المقاييس 1 لأعمدة A، بدون حساب القطر.
!>

TSCAL

!>          TSCAL هو REAL
!>          عامل القياس المستخدم في حساب المقاييس 1 في CNORM.
!>          CNORM يحتوي في الواقع على مقاييس الأعمدة لـ TSCAL*A.
!>

X

!>          X هو مصفوفة COMPLEX، ذات بعد (LDX,NRHS)
!>          متجهات الحل المحسوبة لأنظمة المعادلات الخطية.
!>

LDX

!>          LDX هو INTEGER
!>          البعد الرائد للمصفوفة X.  LDX >= max(1,N).
!>

B

!>          B هو مصفوفة COMPLEX، ذات بعد (LDB,NRHS)
!>          متجهات الطرف الأيمن لأنظمة المعادلات الخطية.
!>

LDB

!>          LDB هو INTEGER
!>          البعد الرائد للمصفوفة B.  LDB >= max(1,N).
!>

WORK

!>          WORK هو مصفوفة COMPLEX، ذات بعد (N)
!>

RESID

!>          RESID هو REAL
!>          الحد الأقصى على عدد الأطراف اليمنى من
!>          norm(op(A)*x - s*b) / ( norm(op(A)) * norm(x) * EPS ).
!>
