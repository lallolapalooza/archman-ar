subroutine ctzrqf (integer m, integer n, complex, dimension( lda, * ) a, integer lda, complex, dimension( * ) tau, integer info)

CTZRQF

الغرض:

!>
!> هذا الروتين الفرعي تم إهماله وتم استبداله بالروتين الفرعي CTZRZF.
!>
!> يقوم CTZRQF بتقليل مصفوفة A شبه المنحرفة المعقدة ذات الأبعاد M-by-N ( M<=N )
!> إلى شكل مثلثي علوي بواسطة تحويلات وحدوية.
!>
!> يتم تحليل المصفوفة شبه المنحرفة A على النحو التالي:
!>
!>    A = ( R  0 ) * Z,
!>
!> حيث Z هي مصفوفة وحدوية N-by-N و R هي مصفوفة مثلثية علوية M-by-M.
!>

المعلمات

M 
!>          M هو عدد صحيح (INTEGER).
!>          عدد صفوف المصفوفة A.  M >= 0.
!>

N

!>          N هو عدد صحيح (INTEGER).
!>          عدد أعمدة المصفوفة A.  N >= M.
!>

A

!>          A هي مصفوفة معقدة (COMPLEX)، ذات أبعاد (LDA,N).
!>          عند الإدخال، يجب أن يحتوي الجزء العلوي شبه المنحرف من المصفوفة A،
!>          بأبعاد M-by-N، على المصفوفة المراد تحليلها.
!>          عند الخروج، يحتوي الجزء العلوي المثلثي من A، بأبعاد M-by-M،
!>          على المصفوفة المثلثية العلوية R، والعناصر من M+1 إلى N
!>          في الصفوف M الأولى من A، مع المصفوفة TAU، تمثل المصفوفة
!>          الوحدوية Z كناتج لـ M انعكاسات أولية.
!>

LDA

!>          LDA هو عدد صحيح (INTEGER).
!>          البعد الرائد للمصفوفة A.  LDA >= max(1,M).
!>

TAU

!>          TAU هي مصفوفة معقدة (COMPLEX)، ذات أبعاد (M).
!>          العوامل القياسية للانعكاسات الأولية.
!>

INFO

!>          INFO هو عدد صحيح (INTEGER).
!>          = 0: تم الخروج بنجاح.
!>          < 0: إذا كان INFO = -i، فقد كان الوسيط i غير قانوني.
!>

تفاصيل إضافية:

!>
!>  يتم الحصول على التحليل بطريقة هاوسهولدر. مصفوفة التحويل k ، Z( k )،
!>  والتي يتم استخدام منقولها المترافق لإدخال أصفار في الصف (m - k + 1) من A،
!>  تُعطى على الصورة:
!>
!>     Z( k ) = ( I     0   ),
!>              ( 0  T( k ) )
!>
!>  حيث:
!>
!>     T( k ) = I - tau*u( k )*u( k )**H,   u( k ) = (   1    ),
!>                                                   (   0    )
!>                                                   ( z( k ) )
!>
!>  يتم اختيار tau و z( k ) لإلغاء عناصر الصف k من X.
!>
!>  يتم إرجاع العامل tau في العنصر k من TAU والمتجه u( k ) في الصف k من A،
!>  بحيث توجد عناصر z( k ) في a( k, m + 1 )، ...، a( k, n ).
!>  يتم إرجاع عناصر R في الجزء المثلثي العلوي من A.
!>
!>  Z تُعطى بواسطة:
!>
!>     Z =  Z( 1 ) * Z( 2 ) * ... * Z( m ).
!>
