توثيق الوظيفة/الروتين الفرعي

subroutine ctrti2 (character uplo, character diag, integer n, complex, dimension( lda, * ) a, integer lda, integer info)

CTRTI2 يحسب معكوس مصفوفة مثلثة (خوارزمية غير محظورة).

الغرض:

!>
!> CTRTI2 يحسب معكوس مصفوفة مثلثة علوية أو سفلية مركبة.
!>
!> هذا هو الإصدار من المستوى 2 من الخوارزمية BLAS.
!>

المعاملات

UPLO 
!>          UPLO is CHARACTER*1
!>          يحدد ما إذا كانت المصفوفة A علوية أم سفلية.
!>          = 'U':  علوية
!>          = 'L':  سفلية
!>

DIAG

!>          DIAG is CHARACTER*1
!>          يحدد ما إذا كانت المصفوفة A مثلثة وحدوية أم لا.
!>          = 'N':  غير وحدوية
!>          = 'U':  وحدوية
!>

N

!>          N is INTEGER
!>          رتبة المصفوفة A.  N >= 0.
!>

A

!>          A is COMPLEX array, dimension (LDA,N)
!>          عند الإدخال، الجزء العلوي الأيسر من المصفوفة A الذي حجمه n × n
!>          يحتوي على المصفوفة المثلثة، والجزء السفلي المثلثي الصارم
!>          للمصفوفة A لا يتم الرجوع إليه. إذا كانت UPLO = 'L'، فإن
!>          الجزء السفلي الأيسر من المصفوفة A الذي حجمه n × n يحتوي
!>          على المصفوفة المثلثة السفلية، والجزء العلوي المثلثي الصارم
!>          للمصفوفة A لا يتم الرجوع إليه. إذا كانت DIAG = 'U'، فإن
!>          عناصر القطر الرئيسية للمصفوفة A لا يتم الرجوع إليها وتُفترض
!>          أنها 1.
!>
!>          عند الخروج، معكوس المصفوفة الأصلية (المثلثة) مخزن بنفس
!>          تنسيق التخزين.
!>

LDA

!>          LDA is INTEGER
!>          البعد الرئيسي للمصفوفة A.  LDA >= max(1,N).
!>

INFO

!>          INFO is INTEGER
!>          = 0: خروج ناجح
!>          < 0: إذا كان INFO = -k، فإن المعامل k له قيمة غير قانونية
!>
