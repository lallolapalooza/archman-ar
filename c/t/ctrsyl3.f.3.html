subroutine ctrsyl3 (character trana, character tranb, integer isgn, integer m, integer n, complex, dimension( lda, * ) a, integer lda, complex, dimension( ldb, * ) b, integer ldb, complex, dimension( ldc, * ) c, integer ldc, real scale, real, dimension( ldswork, * ) swork, integer ldswork, integer info)

CTRSYL3

الغرض:

!>
!>  CTRSYL3 تحل معادلة مصفوفة سيلفستر المعقدة:
!>
!>     op(A)*X + X*op(B) = scale*C أو
!>     op(A)*X - X*op(B) = scale*C,
!>
!>  حيث op(A) = A أو A**H، و A و B كلاهما مثلثيتان علويتان. A
!>  بحجم M-by-M و B بحجم N-by-N؛ الطرف الأيمن C والحل X هما
!>  بحجم M-by-N؛ و scale هو عامل قياس خرج، يتم تعيينه <= 1 لتجنب
!>  الفيضان في X.
!>
!>  هذا هو الإصدار الكتلي للخوارزمية.
!>

المعلمات

TRANA 
!>          TRANA هو CHARACTER*1
!>          يحدد خيار op(A):
!>          = 'N': op(A) = A    (بدون تبديل)
!>          = 'C': op(A) = A**H (تبديل مترافق)
!>

TRANB

!>          TRANB هو CHARACTER*1
!>          يحدد خيار op(B):
!>          = 'N': op(B) = B    (بدون تبديل)
!>          = 'C': op(B) = B**H (تبديل مترافق)
!>

ISGN

!>          ISGN هو INTEGER
!>          يحدد الإشارة في المعادلة:
!>          = +1: حل op(A)*X + X*op(B) = scale*C
!>          = -1: حل op(A)*X - X*op(B) = scale*C
!>

M

!>          M هو INTEGER
!>          رتبة المصفوفة A، وعدد الصفوف في
!>          المصفوفتين X و C. M >= 0.
!>

N

!>          N هو INTEGER
!>          رتبة المصفوفة B، وعدد الأعمدة في
!>          المصفوفتين X و C. N >= 0.
!>

A

!>          A هي مصفوفة COMPLEX، البعد (LDA,M)
!>          المصفوفة المثلثية العلوية A.
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرائد للمصفوفة A. LDA >= max(1,M).
!>

B

!>          B هي مصفوفة COMPLEX، البعد (LDB,N)
!>          المصفوفة المثلثية العلوية B.
!>

LDB

!>          LDB هو INTEGER
!>          البعد الرائد للمصفوفة B. LDB >= max(1,N).
!>

C

!>          C هي مصفوفة COMPLEX، البعد (LDC,N)
!>          عند الإدخال، مصفوفة الطرف الأيمن C بحجم M-by-N.
!>          عند الخروج، يتم استبدال C بمصفوفة الحل X.
!>

LDC

!>          LDC هو INTEGER
!>          البعد الرائد للمصفوفة C. LDC >= max(1,M)
!>

SCALE

!>          SCALE هو REAL
!>          عامل القياس، scale، يتم تعيينه <= 1 لتجنب الفيضان في X.
!>

SWORK

!>          SWORK هي مصفوفة REAL، البعد (MAX(2, ROWS), MAX(1,COLS)).
!>          عند الخروج، إذا كان INFO = 0، تعيد SWORK(1) القيمة المثلى ROWS
!>          وتعيد SWORK(2) القيمة المثلى COLS.
!>

LDSWORK

!>          LDSWORK هو INTEGER
!>          LDSWORK >= MAX(2,ROWS)، حيث ROWS = ((M + NB - 1) / NB + 1)
!>          و NB هو حجم الكتلة الأمثل.
!>
!>          إذا كان LDSWORK = -1، فسيتم افتراض استعلام مساحة العمل؛ الروتين
!!          يحسب فقط الأبعاد المثلى لمصفوفة SWORK،
!>          يعيد هذه القيم كالإدخال الأول والثاني لمصفوفة SWORK،
!>          ولا يتم إصدار أي رسالة خطأ متعلقة بـ LWORK بواسطة XERBLA.
!>

INFO

!>          INFO هو INTEGER
!>          = 0: خروج ناجح
!>          < 0: إذا كان INFO = -i، فإن المعامل i كان لديه قيمة غير قانونية
!>          = 1: A و B لديهما قيم ذاتية مشتركة أو قريبة جداً؛ تم استخدام
!>               قيم معدلة لحل المعادلة (لكن المصفوفتين A و B لم تتغيرا).
!>
