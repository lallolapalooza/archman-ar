subroutine ctprfb (character side, character trans, character direct, character storev, integer m, integer n, integer k, integer l, complex, dimension( ldv, * ) v, integer ldv, complex, dimension( ldt, * ) t, integer ldt, complex, dimension( lda, * ) a, integer lda, complex, dimension( ldb, * ) b, integer ldb, complex, dimension( ldwork, * ) work, integer ldwork)

CTPRFB تطبق "عاكس كتلة مثلثي-خماسي" معقد على مصفوفة معقدة, والتي تتكون من كتلتين.

الغرض:

!>
!> CTPRFB تطبق عاكس كتلة معقد H أو منقولته المرافقة H**H
!> على مصفوفة معقدة C، والتي تتكون من كتلتين A و B، إما من اليسار أو اليمين.
!>
!>

المعاملات

SIDE 
!>          SIDE هو CHARACTER*1
!>          = 'L': تطبيق H أو H**H من اليسار
!>          = 'R': تطبيق H أو H**H من اليمين
!>

TRANS

!>          TRANS هو CHARACTER*1
!>          = 'N': تطبيق H (بدون منقولة)
!>          = 'C': تطبيق H**H (منقولة مرافقة)
!>

DIRECT

!>          DIRECT هو CHARACTER*1
!>          يشير إلى كيفية تشكيل H من حاصل ضرب عواكس أولية
!>          = 'F': H = H(1) H(2) . . . H(k) (أمامي)
!>          = 'B': H = H(k) . . . H(2) H(1) (خلفي)
!>

STOREV

!>          STOREV هو CHARACTER*1
!>          يشير إلى كيفية تخزين المتجهات التي تحدد العواكس الأولية:
!>          = 'C': أعمدة
!>          = 'R': صفوف
!>

M

!>          M هو INTEGER
!>          عدد صفوف المصفوفة B.
!>          M >= 0.
!>

N

!>          N هو INTEGER
!>          عدد أعمدة المصفوفة B.
!>          N >= 0.
!>

K

!>          K هو INTEGER
!>          ترتيب المصفوفة T، أي عدد العواكس الأولية التي يشكل حاصل ضربها
!>          عاكس الكتلة.
!>          K >= 0.
!>

L

!>          L هو INTEGER
!>          ترتيب الجزء شبه المنحرف من V.
!>          K >= L >= 0. انظر التفاصيل الإضافية.
!>

V

!>          V هو مصفوفة COMPLEX، البعد
!>                                (LDV,K) إذا كان STOREV = 'C'
!>                                (LDV,M) إذا كان STOREV = 'R' و SIDE = 'L'
!>                                (LDV,N) إذا كان STOREV = 'R' و SIDE = 'R'
!>          المصفوفة الخماسية V، التي تحتوي على العواكس الأولية
!>          H(1), H(2), ..., H(K). انظر التفاصيل الإضافية.
!>

LDV

!>          LDV هو INTEGER
!>          البعد الرئيسي للمصفوفة V.
!>          إذا كان STOREV = 'C' و SIDE = 'L'، فإن LDV >= max(1,M)؛
!>          إذا كان STOREV = 'C' و SIDE = 'R'، فإن LDV >= max(1,N)؛
!>          إذا كان STOREV = 'R'، فإن LDV >= K.
!>

T

!>          T هو مصفوفة COMPLEX، البعد (LDT,K)
!>          المصفوفة المثلثية T بحجم K*K في تمثيل عاكس الكتلة.
!>

LDT

!>          LDT هو INTEGER
!>          البعد الرئيسي للمصفوفة T.
!>          LDT >= K.
!>

A

!>          A هو مصفوفة COMPLEX، البعد
!>          (LDA,N) إذا كان SIDE = 'L' أو (LDA,K) إذا كان SIDE = 'R'
!>          عند الإدخال، المصفوفة A بحجم K*N أو M*K.
!>          عند الخروج، يتم الكتابة فوق A بالكتلة المقابلة من
!>          H*C أو H**H*C أو C*H أو C*H**H. انظر التفاصيل الإضافية.
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرئيسي للمصفوفة A.
!>          إذا كان SIDE = 'L'، فإن LDA >= max(1,K)؛
!>          إذا كان SIDE = 'R'، فإن LDA >= max(1,M).
!>

B

!>          B هو مصفوفة COMPLEX، البعد (LDB,N)
!>          عند الإدخال، المصفوفة B بحجم M*N.
!>          عند الخروج، يتم الكتابة فوق B بالكتلة المقابلة من
!>          H*C أو H**H*C أو C*H أو C*H**H. انظر التفاصيل الإضافية.
!>

LDB

!>          LDB هو INTEGER
!>          البعد الرئيسي للمصفوفة B.
!>          LDB >= max(1,M).
!>

WORK

!>          WORK هو مصفوفة COMPLEX، البعد
!>          (LDWORK,N) إذا كان SIDE = 'L'،
!>          (LDWORK,K) إذا كان SIDE = 'R'.
!>

LDWORK

!>          LDWORK هو INTEGER
!>          البعد الرئيسي للمصفوفة WORK.
!>          إذا كان SIDE = 'L'، فإن LDWORK >= K؛
!>          إذا كان SIDE = 'R'، فإن LDWORK >= M.
!>

التفاصيل الإضافية:

!>
!>  المصفوفة C هي مصفوفة مركبة مكونة من الكتلتين A و B.
!>  الكتلة B بحجم M*N؛ إذا كان SIDE = 'R'، فإن A بحجم M*K،
!>  وإذا كان SIDE = 'L'، فإن A بحجم K*N.
!>
!>  إذا كان SIDE = 'R' و DIRECT = 'F'، فإن C = [A B].
!>
!>  إذا كان SIDE = 'L' و DIRECT = 'F'، فإن C = [A]
!>                                      [B].
!>
!>  إذا كان SIDE = 'R' و DIRECT = 'B'، فإن C = [B A].
!>
!>  إذا كان SIDE = 'L' و DIRECT = 'B'، فإن C = [B]
!>                                      [A].
!>
!>  المصفوفة الخماسية V مكونة من كتلة مستطيلة V1 وكتلة
!>  شبه منحرفة V2. يتم تحديد حجم الكتلة شبه المنحرفة بواسطة
!>  المعامل L، حيث 0<=L<=K. إذا كان L=K، فإن كتلة V2 في V مثلثة؛
!>  إذا كان L=0، لا توجد كتلة شبه منحرفة، وبالتالي V = V1 مستطيلة.
!>
!>  إذا كان DIRECT = 'F' و STOREV = 'C': V = [V1]
!>                                         [V2]
!>     - V2 شبه منحرفة علوية (أول L صفوف من K*K مثلثية علوية)
!>
!>  إذا كان DIRECT = 'F' و STOREV = 'R': V = [V1 V2]
!>
!>     - V2 شبه منحرفة سفلية (أول L أعمدة من K*K مثلثية سفلية)
!>
!>  إذا كان DIRECT = 'B' و STOREV = 'C': V = [V2]
!>                                         [V1]
!>     - V2 شبه منحرفة سفلية (آخر L صفوف من K*K مثلثية سفلية)
!>
!>  إذا كان DIRECT = 'B' و STOREV = 'R': V = [V2 V1]
!>
!>     - V2 شبه منحرفة علوية (آخر L أعمدة من K*K مثلثية علوية)
!>
!>  إذا كان STOREV = 'C' و SIDE = 'L'، فإن V هي M*K مع V2 L*K.
!>
!>  إذا كان STOREV = 'C' و SIDE = 'R'، فإن V هي N*K مع V2 L*K.
!>
!>  إذا كان STOREV = 'R' و SIDE = 'L'، فإن V هي K*M مع V2 K*L.
!>
!>  إذا كان STOREV = 'R' و SIDE = 'R'، فإن V هي K*N مع V2 K*L.
!>
