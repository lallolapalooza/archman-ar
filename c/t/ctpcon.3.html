subroutine ctpcon (character norm, character uplo, character diag, integer n, complex, dimension( * ) ap, real rcond, complex, dimension( * ) work, real, dimension( * ) rwork, integer info)

CTPCON

الغرض:

!>
!> CTPCON يقدر مقلوب رقم شرط مصفوفة مثلثية مضغوطة A، إما بمعيار 1-norm أو
!> بمعيار infinity-norm.
!>
!> يتم حساب معيار A ويتم الحصول على تقدير لـ
!> norm(inv(A))، ثم يتم حساب مقلوب رقم الشرط على النحو التالي:
!>    RCOND = 1 / ( norm(A) * norm(inv(A)) ).
!>

المعلمات

NORM 
!>          NORM هو CHARACTER*1
!>          يحدد ما إذا كان رقم الشرط 1-norm أو infinity-norm مطلوبًا:
!>          = '1' أو 'O':  1-norm؛
!>          = 'I':         Infinity-norm.
!>

UPLO

!>          UPLO هو CHARACTER*1
!>          = 'U':  A هي مصفوفة مثلثية عليا؛
!>          = 'L':  A هي مصفوفة مثلثية سفلى.
!>

DIAG

!>          DIAG هو CHARACTER*1
!>          = 'N':  A مصفوفة غير وحيدة مثلثية؛
!>          = 'U':  A مصفوفة وحيدة مثلثية.
!>

N

!>          N هو INTEGER
!>          ترتيب المصفوفة A.  N >= 0.
!>

AP

!>          AP هو مصفوفة COMPLEX، أبعادها (N*(N+1)/2)
!>          المصفوفة المثلثية العليا أو السفلى A، مضغوطة حسب الأعمدة في
!>          مصفوفة خطية. يتم تخزين العمود j من A في المصفوفة
!>          AP كما يلي:
!>          إذا كان UPLO = 'U'، فإن AP(i + (j-1)*j/2) = A(i,j) لـ 1<=i<=j؛
!>          إذا كان UPLO = 'L'، فإن AP(i + (j-1)*(2n-j)/2) = A(i,j) لـ j<=i<=n.
!>          إذا كان DIAG = 'U'، فإن عناصر القطر الرئيسي لـ A لا تتم الإشارة إليها
!>          ويفترض أنها 1.
!>

RCOND

!>          RCOND هو REAL
!>          مقلوب رقم شرط المصفوفة A،
!>          محسوب كـ RCOND = 1/(norm(A) * norm(inv(A))).
!>

WORK

!>          WORK هي مصفوفة COMPLEX، أبعادها (2*N)
!>

RWORK

!>          RWORK هي مصفوفة REAL، أبعادها (N)
!>

INFO

!>          INFO هو INTEGER
!>          = 0:  خروج ناجح
!>          < 0:  إذا كان INFO = -i، فإن الوسيط i كان لديه قيمة غير قانونية
!>
