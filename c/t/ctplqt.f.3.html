subroutine ctplqt (integer m, integer n, integer l, integer mb, complex, dimension( lda, * ) a, integer lda, complex, dimension( ldb, * ) b, integer ldb, complex, dimension( ldt, * ) t, integer ldt, complex, dimension( * ) work, integer info)

CTPLQT

الغرض:

!>
!> CTPLQT تحسب تحليل LQ مسدود لمصفوفة مركبة C، والتي تتكون من
!>  مصفوفة مثلثية A ومصفوفة خماسية B، باستخدام التمثيل المدمج
!> WY لـ Q.
!>

المعلمات

M 
!>          M هو عدد صحيح (INTEGER).
!>          عدد صفوف المصفوفة B، ورتبة
!>          المصفوفة المثلثية A.
!>          M >= 0.
!>

N

!>          N هو عدد صحيح (INTEGER).
!>          عدد أعمدة المصفوفة B.
!>          N >= 0.
!>

L

!>          L هو عدد صحيح (INTEGER).
!>          عدد صفوف الجزء شبه المنحرف السفلي من B.
!>          MIN(M,N) >= L >= 0. انظر التفاصيل الإضافية.
!>

MB

!>          MB هو عدد صحيح (INTEGER).
!>          حجم الكتلة المستخدم في QR المسدود. M >= MB >= 1.
!>

A

!>          A هو مصفوفة مركبة (COMPLEX)، البعد (LDA,M).
!>          عند الإدخال، المصفوفة المثلثية السفلية A ذات الأبعاد M*M.
!>          عند الخروج، تحتوي العناصر على القطر وما تحته في المصفوفة
!>          على المصفوفة المثلثية السفلية L.
!>

LDA

!>          LDA هو عدد صحيح (INTEGER).
!>          البعد الرئيسي للمصفوفة A. LDA >= max(1,M).
!>

B

!>          B هي مصفوفة مركبة (COMPLEX)، البعد (LDB,N).
!>          عند الإدخال، المصفوفة الخماسية B ذات الأبعاد M*N. الأعمدة N-L
!>          الأولى مستطيلة، والأعمدة L الأخيرة شبه منحرفة سفلية.
!>          عند الخروج، تحتوي B على المصفوفة الخماسية V. انظر التفاصيل الإضافية.
!>

LDB

!>          LDB هو عدد صحيح (INTEGER).
!>          البعد الرئيسي للمصفوفة B. LDB >= max(1,M).
!>

T

!>          T هي مصفوفة مركبة (COMPLEX)، البعد (LDT,N).
!>          عواكس الكتل المثلثية السفلية مخزنة في شكل مدمج
!>          كسلسلة من الكتل المثلثية العليا. انظر التفاصيل الإضافية.
!>

LDT

!>          LDT هو عدد صحيح (INTEGER).
!>          البعد الرئيسي للمصفوفة T. LDT >= MB.
!>

WORK

!>          WORK هي مصفوفة مركبة (COMPLEX)، البعد (MB*M).
!>

INFO

!>          INFO هو عدد صحيح (INTEGER).
!>          = 0:  انتهاء ناجح
!>          < 0:  إذا كان INFO = -i، فإن المعلمة i كانت ذات قيمة غير قانونية

تفاصيل إضافية:

!>
!>  المصفوفة المدخلة C هي مصفوفة من الأبعاد M*(M+N)
!>
!>               C = [ A ] [ B ]
!>
!>
!>  حيث A هي مصفوفة مثلثية سفلية بأبعاد M*M، و B هي مصفوفة خماسية بأبعاد M*N
!>  تتكون من مصفوفة مستطيلة B1 بأبعاد M*(N-L) على يسار مصفوفة شبه منحرفة سفلية B2
!>  بأبعاد M*L:
!>          [ B ] = [ B1 ] [ B2 ]
!>                   [ B1 ]  <- M*(N-L) مستطيلة
!>                   [ B2 ]  <-     M*L شبه منحرفة سفلية.
!>
!>  تتكون المصفوفة شبه المنحرفة السفلية B2 من الأعمدة L الأولى لمصفوفة
!>  مثلثية سفلية بأبعاد M*M، حيث 0 <= L <= MIN(M,N). إذا كان L=0،
!>  فإن B مستطيلة بأبعاد M*N؛ إذا كان M=L=N، فإن B مثلثية سفلية.
!>
!>  المصفوفة W تخزن الانعكاسات الأولية H(i) في الصف i
!>  أعلى القطر (في A) في المصفوفة المدخلة C ذات الأبعاد M*(M+N)
!>            [ C ] = [ A ] [ B ]
!>                   [ A ]  <- مثلثية سفلية M*M
!>                   [ B ]  <- خماسية M*N
!>
!>  بحيث يمكن تمثيل W كـ
!>            [ W ] = [ I ] [ V ]
!>                   [ I ]  <- هوية، M*M
!>                   [ V ]  <- M*N، نفس شكل B.
!>
!>  بالتالي، فإن جميع المعلومات المطلوبة لـ W موجودة عند الخروج في B،
!>  والتي نسميها V أعلاه. لاحظ أن V لها نفس شكل B؛ أي،
!>            [ V ] = [ V1 ] [ V2 ]
!>                   [ V1 ] <- M*(N-L) مستطيلة
!>                   [ V2 ] <-     M*L شبه منحرفة سفلية.
!>
!>  تمثل صفوف V المتجهات التي تحدد H(i).
!>
!>  عدد الكتل هو B = ceiling(M/MB)، حيث كل
!>  كتلة من الرتبة MB باستثناء الكتلة الأخيرة، والتي تكون من الرتبة
!>  IB = M - (M-1)*MB. لكل من الكتل B، يتم حساب عامل كتلة مثلثية عليا
!>  انعكاسية: T1, T2, ..., TB. يتم تخزين T's ذات الأبعاد MB*MB (و IB*IB
!>  لل كتلة الأخيرة) في المصفوفة T ذات الأبعاد MB*N كـ
!>
!>               T = [T1 T2 ... TB].
!>

تعريف في السطر 172 من الملف ctplqt.f.
