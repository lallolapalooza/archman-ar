subroutine ctpttf (transr, uplo, n, ap, arf, info)
CTPTTF تنسخ مصفوفة مثلثية من الصيغة المعبأة القياسية (TP) إلى الصيغة المعبأة الكاملة المستطيلة (TF).

توثيق الوظيفة/الإجراء الفرعي

subroutine ctpttf (character transr, character uplo, integer n, complex, dimension( 0: * ) ap, complex, dimension( 0: * ) arf, integer info)

CTPTTF تنسخ مصفوفة مثلثية من الصيغة المعبأة القياسية (TP) إلى الصيغة المعبأة الكاملة المستطيلة (TF).

الغرض:

!>
!> CTPTTF تنسخ مصفوفة مثلثية A من الصيغة المعبأة القياسية (TP)
!> إلى الصيغة المعبأة الكاملة المستطيلة (TF).
!>

المعلمات

TRANSR 
!>          TRANSR هو CHARACTER*1
!>          = 'N':  يُطلب ARF بالصيغة العادية؛
!>          = 'C':  يُطلب ARF بصيغة النسخ المرافق (conjugate-transpose).
!>

UPLO

!>          UPLO هو CHARACTER*1
!>          = 'U':  A مثلثية علوية؛
!>          = 'L':  A مثلثية سفلية.
!>

N

!>          N هو INTEGER
!>          رتبة المصفوفة A.  N >= 0.
!>

AP

!>          AP هو مصفوفة COMPLEX، البعد ( N*(N+1)/2 )،
!>          عند الإدخال، المصفوفة المثلثية العلوية أو السفلية A، معبأة
!>          حسب العمود في مصفوفة خطية. يتم تخزين العمود j من A
!>          في المصفوفة AP كما يلي:
!>          إذا كان UPLO = 'U'، فإن AP(i + (j-1)*j/2) = A(i,j) لـ 1<=i<=j؛
!>          إذا كان UPLO = 'L'، فإن AP(i + (j-1)*(2n-j)/2) = A(i,j) لـ j<=i<=n.
!>

ARF

!>          ARF هو مصفوفة COMPLEX، البعد ( N*(N+1)/2 )،
!>          عند الخروج، المصفوفة المثلثية A المخزنة بصيغة
!>          RFP (Rectangular Full Packed). لمزيد من المناقشة، انظر الملاحظات أدناه.
!>

INFO

!>          INFO هو INTEGER
!>          = 0:  انتهاء ناجح
!>          < 0:  إذا كان INFO = -i، فإن الوسيط i كان له قيمة غير قانونية
!>

!>
!>      AP هي علوية             AP هي سفلية
!>
!>   00 01 02 03 04 05       00
!>      11 12 13 14 15       10 11

!>            33 34 35       30 31 32 33
!>               44 45       40 41 42 43 44
!>                  55       50 51 52 53 54 55
!>
!>
!>  دع TRANSR = 'N'. تحتفظ RFP بـ AP كما يلي:
!>  بالنسبة لـ UPLO = 'U'، يتكون شبه المنحرف العلوي A(0:5,0:2) من الأعمدة الثلاثة الأخيرة
!>  من AP العلوية. يتكون المثلث السفلي A(4:6,0:2) من
!>  النسخ المرافق للأعمدة الثلاثة الأولى من AP العلوية.
!>  بالنسبة لـ UPLO = 'L'، يتكون شبه المنحرف السفلي A(1:6,0:2) من الأعمدة الثلاثة الأولى
!>  من AP السفلية. يتكون المثلث العلوي A(0:2,0:2) من
!>  النسخ المرافق للأعمدة الثلاثة الأخيرة من AP السفلية.
!>  للدلالة على المرافق، نضع -- فوق العنصر. هذا يغطي
!>  الحالة N زوجي و TRANSR = 'N'.
!>
!>         RFP A                   RFP A
!>
!>                                -- -- --
!>        03 04 05                33 43 53
!>                                   -- --
!>        13 14 15                00 44 54
!>                                      --
!>        23 24 25                10 11 55
!>

!>        --
!>        00 44 45                30 31 32
!>        -- --
!>        01 11 55                40 41 42
!>        -- -- --
!>        02 12 22                50 51 52
!>

!>  لـ RFP A أعلاه. وبالتالي نحصل على:
!>
!>
!>           RFP A                   RFP A
!>

!>     -- -- -- -- --                -- -- -- -- --
!>     04 14 24 34 44 11 12    43 44 11 21 31 41 51

!>     05 15 25 35 45 55 22    53 54 55 22 32 42 52
!>

!>  نعطي مثالاً حيث N = 5.
!>
!>     AP هي علوية                 AP هي سفلية
!>
!>   00 01 02 03 04              00
!>      11 12 13 14              10 11
!>         22 23 24              20 21 22
!>            33 34              30 31 32 33
!>               44              40 41 42 43 44
!>
!>
!>  دع TRANSR = 'N'. تحتفظ RFP بـ AP كما يلي:
!>  بالنسبة لـ UPLO = 'U'، يتكون شبه المنحرف العلوي A(0:4,0:2) من الأعمدة الثلاثة الأخيرة
!>  من AP العلوية. يتكون المثلث السفلي A(3:4,0:1) من
!>  النسخ المرافق للعمودين الأولين من AP العلوية.
!>  بالنسبة لـ UPLO = 'L'، يتكون شبه المنحرف السفلي A(0:4,0:2) من الأعمدة الثلاثة الأولى

!>  للدلالة على المرافق، نضع -- فوق العنصر. هذا يغطي
