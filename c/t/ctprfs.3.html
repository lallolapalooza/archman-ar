subroutine ctprfs (character uplo, character trans, character diag, integer n, integer nrhs, complex, dimension( * ) ap, complex, dimension( ldb, * ) b, integer ldb, complex, dimension( ldx, * ) x, integer ldx, real, dimension( * ) ferr, real, dimension( * ) berr, complex, dimension( * ) work, real, dimension( * ) rwork, integer info)

CTPRFS

الغرض:

!>
!> يوفر CTPRFS حدود الخطأ وتقديرات الخطأ الخلفي لحل نظام المعادلات الخطية
!> مع مصفوفة معامل مثلثة معبأة.
!>
!> يجب أن تكون مصفوفة الحل X محسوبة بواسطة CTPTRS أو بوسيلة أخرى
!> قبل الدخول إلى هذا الروتين. لا يقوم CTPRFS بتنفيذ التنقيح التكراري
!> لأن القيام بذلك لا يمكن أن يحسن الخطأ الخلفي.
!>

المعلمات

UPLO 
!>          UPLO هو CHARACTER*1
!>          = 'U':  A هي مصفوفة مثلثة عليا؛
!>          = 'L':  A هي مصفوفة مثلثة سفلى.
!>

TRANS

!>          TRANS هو CHARACTER*1
!>          يحدد شكل نظام المعادلات:
!>          = 'N':  A * X = B     (بدون تبديل)
!>          = 'T':  A**T * X = B  (تبديل)
!>          = 'C':  A**H * X = B  (تبديل مترافق)
!>

DIAG

!>          DIAG هو CHARACTER*1
!>          = 'N':  A مصفوفة مثلثة غير وحدة؛
!>          = 'U':  A مصفوفة مثلثة وحدة.
!>

N

!>          N هو INTEGER
!>          رتبة المصفوفة A.  N >= 0.
!>

NRHS

!>          NRHS هو INTEGER
!>          عدد الأطراف اليمنى، أي عدد الأعمدة
!>          للمصفوفتين B و X.  NRHS >= 0.
!>

AP

!>          AP هو مصفوفة COMPLEX، البعد (N*(N+1)/2)
!>          المصفوفة المثلثة العليا أو السفلى A، معبأة عمودياً في
!>          مصفوفة خطية.  يتم تخزين العمود j من A في المصفوفة
!>          AP بالشكل التالي:
!>          إذا كان UPLO = 'U'، فإن AP(i + (j-1)*j/2) = A(i,j) لـ 1<=i<=j؛
!>          إذا كان UPLO = 'L'، فإن AP(i + (j-1)*(2n-j)/2) = A(i,j) لـ j<=i<=n.
!>          إذا كان DIAG = 'U'، لا يتم الرجوع إلى العناصر القطرية لـ A
!>          ويفترض أنها 1.
!>

B

!>          B هي مصفوفة COMPLEX، البعد (LDB,NRHS)
!>          مصفوفة الطرف الأيمن B.
!>

LDB

!>          LDB هو INTEGER
!>          البعد الرئيسي للمصفوفة B.  LDB >= max(1,N).
!>

X

!>          X هي مصفوفة COMPLEX، البعد (LDX,NRHS)
!>          مصفوفة الحل X.
!>

LDX

!>          LDX هو INTEGER
!>          البعد الرئيسي للمصفوفة X.  LDX >= max(1,N).
!>

FERR

!>          FERR هي مصفوفة REAL، البعد (NRHS)
!>          تقدير حد الخطأ الأمامي لكل متجه حل
!>          X(j) (العمود j من مصفوفة الحل X).
!>          إذا كان XTRUE هو الحل الحقيقي المقابل لـ X(j)، فإن FERR(j)
!>          هو حد أعلى مقدر لقيمة أكبر
!>          عنصر في (X(j) - XTRUE) مقسومًا على قيمة أكبر
!>          عنصر في X(j).  التقدير موثوق به مثل
!>          التقدير الخاص بـ RCOND، وهو في الغالب زيادة طفيفة
!>          للخطأ الحقيقي.
!>

BERR

!>          BERR هي مصفوفة REAL، البعد (NRHS)
!>          خطأ الارتداد النسبي لكل مكون لمتجه الحل X(j)
!>          (أي، أصغر تغيير نسبي في
!>          أي عنصر في A أو B يجعل X(j) حلاً دقيقًا).
!>

WORK

!>          WORK هي مصفوفة COMPLEX، البعد (2*N)
!>

RWORK

!>          RWORK هي مصفوفة REAL، البعد (N)
!>

INFO

!>          INFO هو INTEGER
!>          = 0:  اكتمال ناجح
!>          < 0:  إذا كان INFO = -i، فإن المعلمة i كانت لها قيمة غير قانونية
!>
