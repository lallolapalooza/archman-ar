subroutine ctbt06 (real rcond, real rcondc, character uplo, character diag, integer n, integer kd, complex, dimension( ldab, * ) ab, integer ldab, real, dimension( * ) rwork, real rat)

CTBT06

الغرض:

!>
!> CTBT06 يحسب نسبة اختبار تقارن RCOND (الرقم الشرطي العكسي للمصفوفة المثلثية A) و RCONDC، التقدير
!> الذي تم حسابه بواسطة CTBCON. يتم استخدام معلومات حول المصفوفة المثلثية A
!> إذا كان أحد التقديرين صفرًا والآخر غير صفر لتحديد ما إذا كان
!> التدفق السفلي في التقدير مبررًا.
!>

المعلمات

RCOND 
!>          RCOND هو REAL
!>          تقدير الرقم الشرطي العكسي الذي تم الحصول عليه بواسطة
!>          تشكيل معكوس المصفوفة A صراحة وحساب
!>          RCOND = 1/( norm(A) * norm(inv(A)) ).
!>

RCONDC

!>          RCONDC هو REAL
!>          تقدير الرقم الشرطي العكسي الذي تم حسابه بواسطة
!>          CTBCON.
!>

UPLO

!>          UPLO هو CHARACTER
!>          يحدد ما إذا كانت المصفوفة A مثلثية علوية أم سفلية.
!>          = 'U':  مثلثية علوية
!>          = 'L':  مثلثية سفلية
!>

DIAG

!>          DIAG هو CHARACTER
!>          يحدد ما إذا كانت المصفوفة A مثلثية واحدة أم لا.
!>          = 'N':  مثلثية غير واحدة
!>          = 'U':  مثلثية واحدة
!>

N

!>          N هو INTEGER
!>          ترتيب المصفوفة A.  N >= 0.
!>

KD

!>          KD هو INTEGER
!>          عدد الأقطار الفائقة أو الأقطار الفرعية للمصفوفة النطاقية المثلثية A.  KD >= 0.
!>

AB

!>          AB هو مصفوفة COMPLEX، البعد (LDAB,N)
!>          المصفوفة المثلثية العلوية أو السفلية النطاقية A، مخزنة في
!>          الصفوف الـ KD+1 الأولى من المصفوفة. يتم تخزين العمود j من A
!>          في العمود j من المصفوفة AB على النحو التالي:
!>          إذا كان UPLO = 'U'، فإن AB(kd+1+i-j,j) = A(i,j) لـ max(1,j-kd)<=i<=j؛
!>          إذا كان UPLO = 'L'، فإن AB(1+i-j,j)    = A(i,j) لـ j<=i<=min(n,j+kd).
!>

LDAB

!>          LDAB هو INTEGER
!>          البعد الرائد للمصفوفة AB.  LDAB >= KD+1.
!>

RWORK

!>          RWORK هي مصفوفة REAL، البعد (N)
!>

RAT

!>          RAT هو REAL
!>          نسبة الاختبار.  إذا كان RCOND و RCONDC كلاهما غير صفري،
!>             RAT = MAX( RCOND, RCONDC )/MIN( RCOND, RCONDC ) - 1.
!>          إذا كان RAT = 0، فإن التقديرين متطابقان تمامًا.
!>
