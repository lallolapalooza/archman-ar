subroutine ctbmv (character uplo, character trans, character diag, integer n, integer k, complex, dimension(lda,*) a, integer lda, complex, dimension(*) x, integer incx)

CTBMV

الغرض:

!>
!> CTBMV  يقوم بإحدى عمليات ضرب المصفوفة بالمتجه
!>
!>    x := A*x,   أو   x := A**T*x,   أو   x := A**H*x,
!>
!> حيث x متجه يحتوي على n عنصر و A مصفوفة مثلثية علوية أو سفلية، موحدة أو غير موحدة،
!> ذات نطاق ( k + 1 ) قطر.
!>

المعاملات

UPLO 
!>          UPLO هو CHARACTER*1
!>           عند الإدخال، يحدد UPLO ما إذا كانت المصفوفة مصفوفة مثلثية علوية أو سفلية
!>           كما يلي:
!>
!>              UPLO = 'U' أو 'u'   A مصفوفة مثلثية علوية.
!>
!>              UPLO = 'L' أو 'l'   A مصفوفة مثلثية سفلية.
!>

TRANS

!>          TRANS هو CHARACTER*1
!>           عند الإدخال، يحدد TRANS العملية التي سيتم تنفيذها كما يلي:
!>
!>              TRANS = 'N' أو 'n'   x := A*x.
!>
!>              TRANS = 'T' أو 't'   x := A**T*x.
!>
!>              TRANS = 'C' أو 'c'   x := A**H*x.
!>

DIAG

!>          DIAG هو CHARACTER*1
!>           عند الإدخال، يحدد DIAG ما إذا كانت A موحدة أم لا
!>           مثلثية كما يلي:
!>
!>              DIAG = 'U' أو 'u'   يفترض أن A مصفوفة موحدة.
!>
!>              DIAG = 'N' أو 'n'   لا يفترض أن A مصفوفة موحدة.
!>                                  مثلثية.
!>

N

!>          N هو INTEGER
!>           عند الإدخال، يحدد N رتبة المصفوفة A.
!>           يجب أن يكون N صفرًا على الأقل.
!>

K

!>          K هو INTEGER
!>           عند الإدخال مع UPLO = 'U' أو 'u'، يحدد K عدد
!>           الأقطار فوق الرئيسية للمصفوفة A.
!>           عند الإدخال مع UPLO = 'L' أو 'l'، يحدد K عدد
!>           الأقطار تحت الرئيسية للمصفوفة A.
!>           يجب أن يحقق K:  0 .le. K.
!>

A

!>          A هو مصفوفة COMPLEX، أبعادها ( LDA, N ).
!>           قبل الإدخال مع UPLO = 'U' أو 'u'، الجزء العلوي ( k + 1 )
!>           في N من المصفوفة A يجب أن يحتوي على الجزء العلوي المثلثي
!>           النطاقي للمصفوفة ذات المعاملات، يتم توفيرها عمودًا بعمود، مع
!>           القطر الرئيسي للمصفوفة في الصف
!>           ( k + 1 ) من المصفوفة، القطر الفوقي الأول يبدأ عند
!>           الموضع 2 في الصف k، وهكذا. الجزء العلوي الأيسر k في k مثلث
!>           من المصفوفة A غير مشار إليه.
!>           سيقوم مقطع البرنامج التالي بنقل مصفوفة نطاق مثلثية علوية
!>           من تخزين المصفوفة الكامل التقليدي
!>           إلى تخزين النطاق:
!>
!>                 DO 20, J = 1, N
!>                    M = K + 1 - J
!>                    DO 10, I = MAX( 1, J - K ), J
!>                       A( M + I, J ) = matrix( I, J )
!>              10    CONTINUE
!>              20 CONTINUE
!>
!>           قبل الإدخال مع UPLO = 'L' أو 'l'، الجزء العلوي ( k + 1 )
!>           في N من المصفوفة A يجب أن يحتوي على الجزء السفلي المثلثي
!>           النطاقي للمصفوفة ذات المعاملات، يتم توفيرها عمودًا بعمود، مع
!>           القطر الرئيسي للمصفوفة في الصف 1 من
!>           المصفوفة، القطر الفرعي الأول يبدأ عند الموضع 1 في
!>           الصف 2، وهكذا. الجزء السفلي الأيمن k في k مثلث
!>           من المصفوفة A غير مشار إليه.
!>           سيقوم مقطع البرنامج التالي بنقل مصفوفة نطاق مثلثية سفلية
!>           من تخزين المصفوفة الكامل التقليدي
!>           إلى تخزين النطاق:
!>
!>                 DO 20, J = 1, N
!>                    M = 1 - J
!>                    DO 10, I = J, MIN( N, J + K )
!>                       A( M + I, J ) = matrix( I, J )
!>              10    CONTINUE
!>              20 CONTINUE
!>
!>           لاحظ أنه عندما يكون DIAG = 'U' أو 'u'، فإن عناصر المصفوفة A
!>           المقابلة لعناصر القطر الرئيسي للمصفوفة غير مشار إليها،
!>           ولكن يفترض أنها واحد.
!>

LDA

!>          LDA هو INTEGER
!>           عند الإدخال، يحدد LDA البعد الأول لـ A كما تم تعريفه
!>           في البرنامج (الفرعي) المستدعي. يجب أن يكون LDA على الأقل
!>           ( k + 1 ).
!>

X

!>          X هو مصفوفة COMPLEX، أبعادها على الأقل
!>           ( 1 + ( n - 1 )*abs( INCX ) ).
!>           قبل الإدخال، يجب أن يحتوي المتجه X المعزز على n
!>           عناصر المتجه x. عند الخروج، يتم استبدال X بـ
!>           المتجه المحول x.
!>

INCX

!>          INCX هو INTEGER
!>           عند الإدخال، يحدد INCX الزيادة لعناصر
!>           X. يجب أن لا يكون INCX صفرًا.
!>
