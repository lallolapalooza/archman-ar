subroutine ctptrs (character uplo, character trans, character diag, integer n, integer nrhs, complex, dimension( * ) ap, complex, dimension( ldb, * ) b, integer ldb, integer info)

CTPTRS

الغرض:

!>
!> CTPTRS يحل نظامًا مثلثيًا بالشكل:
!>
!>    A * X = B,  A**T * X = B,  أو  A**H * X = B,
!>
!> حيث A هي مصفوفة مثلثية من الرتبة N مخزنة بتنسيق مدمج، و B هي مصفوفة N-by-NRHS.
!>
!> يتحقق هذا الإجراء الفرعي من أن A غير مفردة، ولكن يجب ملاحظة أن التفرد المطلق فقط هو الذي يتم اكتشافه.
!> من الممكن أن تكون هناك عناصر قطرية واحدة أو أكثر في A أرقامًا صغيرة جدًا دون أن يشير هذا الإجراء الفرعي إلى خطأ.
!>
!> إذا كان فقدان الدقة العددية المحتمل بسبب المصفوفات القريبة من التفرد مصدر قلق،
!> يجب على المتصل التحقق من أن A غير مفردة ضمن هامش معين قبل استدعاء هذا الإجراء الفرعي.
!>

المعلمات

UPLO 
!>          UPLO is CHARACTER*1
!>          = 'U':  A is upper triangular; (A مثلثية عليا)
!>          = 'L':  A is lower triangular. (A مثلثية سفلى)
!>

TRANS

!>          TRANS is CHARACTER*1
!>          Specifies the form of the system of equations: (يحدد شكل نظام المعادلات)
!>          = 'N':  A * X = B     (No transpose) (بدون تبديل)
!>          = 'T':  A**T * X = B  (Transpose) (تبديل)
!>          = 'C':  A**H * X = B  (Conjugate transpose) (تبديل مرافق)
!>

DIAG

!>          DIAG is CHARACTER*1
!>          = 'N':  A is non-unit triangular; (A مثلثية غير وحدوية)
!>          = 'U':  A is unit triangular. (A مثلثية وحدوية)
!>

N

!>          N is INTEGER
!>          The order of the matrix A.  N >= 0. (رتبة المصفوفة A. N >= 0)
!>

NRHS

!>          NRHS is INTEGER
!>          The number of right hand sides, i.e., the number of columns
!>          of the matrix B.  NRHS >= 0. (عدد الجوانب اليمنى، أي عدد أعمدة المصفوفة B. NRHS >= 0)
!>

AP

!>          AP is COMPLEX array, dimension (N*(N+1)/2)
!>          The upper or lower triangular matrix A, packed columnwise in
!>          a linear array.  The j-th column of A is stored in the array
!>          AP as follows:
!>          if UPLO = 'U', AP(i + (j-1)*j/2) = A(i,j) for 1<=i<=j;
!>          if UPLO = 'L', AP(i + (j-1)*(2*n-j)/2) = A(i,j) for j<=i<=n.
!> (المصفوفة A المثلثية العليا أو السفلى، مدمجة عموديًا في مصفوفة خطية. يتم تخزين العمود j من A في المصفوفة AP كالتالي:
!> إذا كان UPLO = 'U'، فإن AP(i + (j-1)*j/2) = A(i,j) لـ 1<=i<=j؛
!> إذا كان UPLO = 'L'، فإن AP(i + (j-1)*(2*n-j)/2) = A(i,j) لـ j<=i<=n.)
!>

B

!>          B is COMPLEX array, dimension (LDB,NRHS)
!>          On entry, the right hand side matrix B.
!>          On exit, if INFO = 0, the solution matrix X.
!> (عند الإدخال، مصفوفة الجانب الأيمن B.
!> عند الخروج، إذا كان INFO = 0، فإن مصفوفة الحل X.)
!>

LDB

!>          LDB is INTEGER
!>          The leading dimension of the array B.  LDB >= max(1,N).
!> (البعد الرئيسي للمصفوفة B. LDB >= max(1,N))
!>

INFO

!>          INFO is INTEGER
!>          = 0:  successful exit (خروج ناجح)
!>          < 0:  if INFO = -i, the i-th argument had an illegal value (إذا كان INFO = -i، فإن الوسيط i كان له قيمة غير قانونية)
!>          > 0:  if INFO = i, the i-th diagonal element of A is exactly zero,
!>                indicating that the matrix is singular and the
!>                solutions X have not been computed.
!>                (إذا كان INFO = i، فإن العنصر القطري i من A هو صفر بالضبط،
!>                مما يشير إلى أن المصفوفة مفردة وأن الحلول X لم يتم حسابها.)
!>
