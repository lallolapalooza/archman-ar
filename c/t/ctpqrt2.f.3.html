subroutine ctpqrt2 (integer m, integer n, integer l, complex, dimension( lda, * ) a, integer lda, complex, dimension( ldb, * ) b, integer ldb, complex, dimension( ldt, * ) t, integer ldt, integer info)

CTPQRT2 يحسب تحليل QR لمصفوفة مركبة "مثلثة-خماسية"، والتي تتكون من كتلة مثلثة وكتلة خماسية، باستخدام تمثيل WY المدمج لـ Q.

الغرض:

!>
!> CTPQRT2 يحسب تحليل QR لمصفوفة مركبة C، والتي تتكون من كتلة مثلثة A وكتلة خماسية B،
!> باستخدام تمثيل WY المدمج لـ Q.
!>

التفاصيل الإضافية:

!>
!>  المصفوفة المدخلة C هي مصفوفة بحجم (N+M) * N
!>
!>               C = [ A ]
!>                   [ B ]
!>
!>  حيث A هي مصفوفة مثلثة علوية بحجم N*N، و B هي مصفوفة خماسية بحجم M*N
!>  تتكون من مصفوفة مستطيلة (M-L)*N تسمى B1 في الأعلى ومصفوفة شبه منحرفة علوية L*N تسمى B2 في الأسفل:
!>
!>               B = [ B1 ]  <- مستطيلة (M-L)*N
!>                   [ B2 ]  <- شبه منحرفة علوية L*N.
!>
!>  المصفوفة شبه المنحرفة العلوية B2 تتكون من الصفوف L الأولى لمصفوفة مثلثة علوية بحجم N*N، حيث 0 <= L <= MIN(M,N). إذا كان L=0،
!>  فإن B تكون مستطيلة M*N؛ إذا كان M=L=N، فإن B تكون مثلثة علوية.
!>
!>  تخزن المصفوفة W المنعكسات الأولية H(i) في العمود i
!>  تحت القطر (لـ A) في المصفوفة المدخلة C بحجم (N+M)*N
!>
!>               C = [ A ]  <- مثلثة علوية N*N
!>                   [ B ]  <- خماسية M*N
!>
!>  بحيث يمكن تمثيل W كـ
!>
!>               W = [ I ]  <- مصفوفة الوحدة، N*N
!>                   [ V ]  <- M*N، بنفس شكل B.
!>
!>  وبالتالي، فإن جميع المعلومات المطلوبة لـ W موجودة عند الخروج في B،
!>  والتي نسميها V أعلاه. لاحظ أن V لها نفس شكل B؛ أي،
!>
!>               V = [ V1 ] <- مستطيلة (M-L)*N
!>                   [ V2 ] <- شبه منحرفة علوية L*N.
!>
!>  أعمدة V تمثل المتجهات التي تحدد H(i).
!>  ثم يُعطى المنعكس الكتلي H بحجم (M+N)*(M+N) بواسطة
!>
!>               H = I - W * T * W**H
!>
!>  حيث W**H هو المرافق المتعامد لـ W و T هو العامل المثلثي العلوي
!>  للمنعكس الكتلي.
!>
