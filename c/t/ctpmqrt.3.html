subroutine ctpmqrt (character side, character trans, integer m, integer n, integer k, integer l, integer nb, complex, dimension( ldv, * ) v, integer ldv, complex, dimension( ldt, * ) t, integer ldt, complex, dimension( lda, * ) a, integer lda, complex, dimension( ldb, * ) b, integer ldb, complex, dimension( * ) work, integer info)

CTPMQRT

الغرض:

!>
!> CTPMQRT يطبق مصفوفة متعامدة معقدة Q تم الحصول عليها من
!>  عاكس كتلة معقد H على مصفوفة عامة معقدة C، تتكون من كتلتين A و B.
!>

المعلمات

SIDE 
!>          SIDE is CHARACTER*1
!>          = 'L': تطبيق Q أو Q**H من اليسار؛
!>          = 'R': تطبيق Q أو Q**H من اليمين.
!>

TRANS

!>          TRANS is CHARACTER*1
!>          = 'N':  بدون تبديل، تطبيق Q؛
!>          = 'C':  تبديل مرافق، تطبيق Q**H.
!>

M

!>          M is INTEGER
!>          عدد صفوف المصفوفة B. M >= 0.
!>

N

!>          N is INTEGER
!>          عدد أعمدة المصفوفة B. N >= 0.
!>

K

!>          K is INTEGER
!>          عدد العواكس الأولية التي يشكل حاصل ضربها
!>          المصفوفة Q.
!>

L

!>          L is INTEGER
!>          ترتيب الجزء شبه المنحرف من V.
!>          K >= L >= 0.  انظر التفاصيل الإضافية.
!>

NB

!>          NB is INTEGER
!>          حجم الكتلة المستخدم لتخزين T.  K >= NB >= 1.
!>          يجب أن تكون هذه هي نفس قيمة NB المستخدمة لإنشاء T
!>          في CTPQRT.
!>

V

!>          V is COMPLEX array, dimension (LDV,K)
!>          يجب أن تحتوي العمود i على المتجه الذي يحدد
!>          العاكس الأولي H(i), لـ i = 1,2,...,k, كما تم إرجاعه بواسطة
!>          CTPQRT في B.  انظر التفاصيل الإضافية.
!>

LDV

!>          LDV is INTEGER
!>          البعد الرئيسي للمصفوفة V.
!>          إذا كان SIDE = 'L'، فإن LDV >= max(1,M)؛
!>          إذا كان SIDE = 'R'، فإن LDV >= max(1,N).
!>

T

!>          T is COMPLEX array, dimension (LDT,K)
!>          العوامل المثلثية العلوية للعواكس الكتلة
!>          كما تم إرجاعه بواسطة CTPQRT, مخزنة كمصفوفة NB × K.
!>

LDT

!>          LDT is INTEGER
!>          البعد الرئيسي للمصفوفة T.  LDT >= NB.
!>

A

!>          A is COMPLEX array, dimension
!>          (LDA,N) إذا كان SIDE = 'L' أو
!>          (LDA,K) إذا كان SIDE = 'R'
!>          عند الدخول، المصفوفة A بالحجم K×N أو M×K.
!>          عند الخروج، يتم استبدال A بالكتلة المقابلة من
!>          Q*C أو Q**H*C أو C*Q أو C*Q**H.  انظر التفاصيل الإضافية.
!>

LDA

!>          LDA is INTEGER
!>          البعد الرئيسي للمصفوفة A.
!>          إذا كان SIDE = 'L'، فإن LDC >= max(1,K)؛
!>          إذا كان SIDE = 'R'، فإن LDC >= max(1,M).
!>

B

!>          B is COMPLEX array, dimension (LDB,N)
!>          عند الدخول، المصفوفة B بالحجم M×N.
!>          عند الخروج، يتم استبدال B بالكتلة المقابلة من
!>          Q*C أو Q**H*C أو C*Q أو C*Q**H.  انظر التفاصيل الإضافية.
!>

LDB

!>          LDB is INTEGER
!>          البعد الرئيسي للمصفوفة B.
!>          LDB >= max(1,M).
!>

WORK

!>          WORK is COMPLEX array. The dimension of WORK is
!>           N*NB إذا كان SIDE = 'L'، أو M*NB إذا كان SIDE = 'R'.
!>

INFO

!>          INFO is INTEGER
!>          = 0:  الخروج ناجح
!>          < 0:  إذا كان INFO = -i، كان الوسيط i له قيمة غير قانونية
!>

تفاصيل إضافية:

!>
!>  الأعمدة الخاصة بالمصفوفة الخماسية V تحتوي على العواكس الأولية
!>  H(1), H(2), ..., H(K); تتكون V من كتلة مستطيلة V1 وكتلة
!>  شبه منحرفة V2:
!>
!>        V = [V1]
!>            [V2].
!>
!>  يتم تحديد حجم الكتلة شبه المنحرفة V2 بواسطة المعلمة L،
!>  حيث 0 <= L <= K؛ V2 شبه منحرفة علوية، تتكون من أول L
!>  صفوف من مصفوفة علوية مثلثة بحجم K×K.  إذا كان L=K، فإن V2 تكون مثلثة علوية؛
!>  إذا كان L=0، فلا توجد كتلة شبه منحرفة، وبالتالي V = V1 مستطيلة.
!>
!>  إذا كان SIDE = 'L':  C = [A]  حيث A بحجم K×N،  B بحجم M×N و V بحجم M×K.
!>                      [B]
!>
!>  إذا كان SIDE = 'R':  C = [A B]  حيث A بحجم M×K، B بحجم M×N و V بحجم N×K.
!>
!>  يتم تكوين المصفوفة المتعامدة المعقدة Q من V و T.
!>
!>  إذا كان TRANS='N' و SIDE='L'، يتم استبدال C عند الخروج بـ Q * C.
!>
!>  إذا كان TRANS='C' و SIDE='L'، يتم استبدال C عند الخروج بـ Q**H * C.
!>
!>  إذا كان TRANS='N' و SIDE='R'، يتم استبدال C عند الخروج بـ C * Q.
!>
!>  إذا كان TRANS='C' و SIDE='R'، يتم استبدال C عند الخروج بـ C * Q**H.
!>
