subroutine ctpt01 (character uplo, character diag, integer n, complex, dimension( * ) ap, complex, dimension( * ) ainvp, real rcond, real, dimension( * ) rwork, real resid)

CTPT01

الغرض:

!>
!> CTPT01 تحسب الباقي لمصفوفة مثلثة A مضروبة في معكوسها عندما يتم تخزين A
!> بصيغة مضغوطة:
!>    RESID = norm(A*AINV - I) / ( N * norm(A) * norm(AINV) * EPS ),
!> حيث EPS هي ثابت الآلة.
!>

المعاملات

UPLO 
!>          UPLO هو CHARACTER*1
!>          يحدد ما إذا كانت المصفوفة A مثلثة عليا أو سفلى.
!>          = 'U':  مثلثة عليا
!>          = 'L':  مثلثة سفلى
!>

DIAG

!>          DIAG هو CHARACTER*1
!>          يحدد ما إذا كانت المصفوفة A مثلثة وحدة أم لا.
!>          = 'N':  غير مثلثة وحدة
!>          = 'U':  مثلثة وحدة
!>

N

!>          N هو INTEGER
!>          رتبة المصفوفة A.  N >= 0.
!>

AP

!>          AP هو مصفوفة COMPLEX, dimension (N*(N+1)/2)
!>          المصفوفة الأصلية A المثلثة العلوية أو السفلية، مضغوطة
!>          عموديًا في مصفوفة خطية. يتم تخزين العمود j من A
!>          في المصفوفة AP كالتالي:
!>          إذا كان UPLO = 'U'، فإن AP((j-1)*j/2 + i) = A(i,j) لـ 1<=i<=j؛
!>          إذا كان UPLO = 'L'،
!>             AP((j-1)*(n-j) + j*(j+1)/2 + i-j) = A(i,j) لـ j<=i<=n.
!>

AINVP

!>          AINVP هي مصفوفة COMPLEX, dimension (N*(N+1)/2)
!>          عند الإدخال، معكوس المصفوفة A (المثلثية)، مضغوطة
!>          عموديًا في مصفوفة خطية كما في AP.
!>          عند الخروج، يتم تدمير محتويات AINVP.
!>

RCOND

!>          RCOND هو REAL
!>          مقلوب رقم الشرط للمصفوفة A، محسوب كـ
!>          1/(norm(A) * norm(AINV)).
!>

RWORK

!>          RWORK هي مصفوفة REAL, dimension (N)
!>

RESID

!>          RESID هو REAL
!>          norm(A*AINV - I) / ( N * norm(A) * norm(AINV) * EPS )
!>
