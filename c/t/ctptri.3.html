subroutine ctptri (character uplo, character diag, integer n, complex, dimension( * ) ap, integer info)

CTPTRI

الغرض:

!>
!> CTPTRI يحسب معكوس مصفوفة مثلثة علوية أو سفلية مركبة مخزنة بتنسيق مضغوط.
!>

المعاملات

UPLO 
!>          UPLO هو CHARACTER*1
!>          = 'U':  A مصفوفة مثلثة علوية؛
!>          = 'L':  A مصفوفة مثلثة سفلية.
!>

DIAG

!>          DIAG هو CHARACTER*1
!>          = 'N':  A مصفوفة غير موحدة مثلثة؛
!>          = 'U':  A مصفوفة موحدة مثلثة.
!>

N

!>          N هو INTEGER
!>          رتبة المصفوفة A.  N >= 0.
!>

AP

!>          AP هو مصفوفة COMPLEX، البعد (N*(N+1)/2)
!>          عند الدخول، المصفوفة مثلثة علوية أو سفلية A، مخزنة
!>          بعمود في مصفوفة خطية.  يتم تخزين العمود j من A
!>          في المصفوفة AP كالتالي:
!>          إذا كان UPLO = 'U'، فإن AP(i + (j-1)*j/2) = A(i,j) لـ 1<=i<=j؛
!>          إذا كان UPLO = 'L'، فإن AP(i + (j-1)*((2*n-j)/2) = A(i,j) لـ j<=i<=n.
!>          انظر أدناه لمزيد من التفاصيل.
!>          عند الخروج، معكوس (المثلث) للمصفوفة الأصلية، بنفس
!>          تنسيق التخزين المضغوط.
!>

INFO

!>          INFO هو INTEGER
!>          = 0:  الانتهاء بنجاح
!>          < 0:  إذا كان INFO = -i، فإن المعامل i كان له قيمة غير قانونية
!>          > 0:  إذا كان INFO = i، فإن A(i,i) تساوي صفر تمامًا.  المصفوفة
!>                المثلثة مفردة ولا يمكن حساب معكوسها.
!>

تفاصيل إضافية:

!>
!>  يمكن نقل المصفوفة المثلثية A إلى تخزين مضغوط باستخدام أحد
!>  مقاطع البرنامج التالية:
!>
!>  UPLO = 'U':                      UPLO = 'L':
!>
!>        JC = 1                           JC = 1
!>        DO 2 J = 1, N                    DO 2 J = 1, N
!>           DO 1 I = 1, J                    DO 1 I = J, N
!>              AP(JC+I-1) = A(I,J)              AP(JC+I-J) = A(I,J)
!>      1    CONTINUE                    1    CONTINUE
!>           JC = JC + J                      JC = JC + N - J + 1
!>      2 CONTINUE                       2 CONTINUE
!>
