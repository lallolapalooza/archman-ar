subroutine ctrsm (character side, character uplo, character transa, character diag, integer m, integer n, complex alpha, complex, dimension(lda,*) a, integer lda, complex, dimension(ldb,*) b, integer ldb)

CTRSM

الغرض:

!>
!> CTRSM  يحل إحدى معادلات المصفوفة
!>
!>    op( A )*X = alpha*B,   أو   X*op( A ) = alpha*B,
!>
!> حيث alpha هو عدد قياسي، X و B مصفوفتان m × n، A مصفوفة مثلثة عليا أو سفلى،
!> وحدوية أو غير وحدوية، و op( A ) هي إحدى العمليات التالية:
!>
!>    op( A ) = A   أو   op( A ) = A**T   أو   op( A ) = A**H.
!>
!> يتم استبدال المصفوفة X بالمصفوفة B.
!>

المعلمات

SIDE 
!>          SIDE هو CHARACTER*1
!>           عند الدخول، تحدد SIDE ما إذا كانت op( A ) تظهر على اليسار
!>           أو اليمين لـ X كما يلي:
!>
!>              SIDE = 'L' أو 'l'   op( A )*X = alpha*B.
!>
!>              SIDE = 'R' أو 'r'   X*op( A ) = alpha*B.
!>

UPLO

!>          UPLO هو CHARACTER*1
!>           عند الدخول، تحدد UPLO ما إذا كانت المصفوفة A مثلثة عليا أو
!>           مثلثة سفلى كما يلي:
!>
!>              UPLO = 'U' أو 'u'   A مصفوفة مثلثة عليا.
!>
!>              UPLO = 'L' أو 'l'   A مصفوفة مثلثة سفلى.
!>

TRANSA

!>          TRANSA هو CHARACTER*1
!>           عند الدخول، تحدد TRANSA شكل op( A ) المستخدم في
!>           ضرب المصفوفات كما يلي:
!>
!>              TRANSA = 'N' أو 'n'   op( A ) = A.
!>
!>              TRANSA = 'T' أو 't'   op( A ) = A**T.
!>
!>              TRANSA = 'C' أو 'c'   op( A ) = A**H.
!>

DIAG

!>          DIAG هو CHARACTER*1
!>           عند الدخول، تحدد DIAG ما إذا كانت A وحدوية أم لا
!>           كما يلي:
!>
!>              DIAG = 'U' أو 'u'   يفترض أن A وحدوية.
!>
!>              DIAG = 'N' أو 'n'   لا يفترض أن A وحدوية.
!>

M

!>          M هو INTEGER
!>           عند الدخول، تحدد M عدد صفوف B. يجب أن يكون M على الأقل صفر.
!>

N

!>          N هو INTEGER
!>           عند الدخول، تحدد N عدد أعمدة B. يجب أن يكون N على الأقل صفر.
!>

ALPHA

!>          ALPHA هو COMPLEX
!>           عند الدخول، يحدد ALPHA العدد القياسي alpha. عندما يكون alpha
!>           صفرًا، لا يتم الرجوع إلى A ولا يلزم تعيين B قبل
!>           الدخول.
!>

A

!>          A هو مصفوفة COMPLEX، ذات البعد ( LDA, k )،
!>           حيث k هو m عندما تكون SIDE = 'L' أو 'l'
!>             و k هو n عندما تكون SIDE = 'R' أو 'r'.
!>           قبل الدخول مع UPLO = 'U' أو 'u'، يجب أن يحتوي الجزء العلوي
!>           الأيسر من المصفوفة k × k في المصفوفة A على المصفوفة
!>           المثلثة العليا، ولا يتم الرجوع إلى الجزء السفلي الأيسر تمامًا
!>           من A.
!>           قبل الدخول مع UPLO = 'L' أو 'l'، يجب أن يحتوي الجزء السفلي
!>           الأيسر من المصفوفة k × k في المصفوفة A على المصفوفة
!>           المثلثة السفلى، ولا يتم الرجوع إلى الجزء العلوي الأيسر تمامًا
!>           من A.
!>           لاحظ أنه عندما يكون DIAG = 'U' أو 'u'، لا يتم الرجوع إلى
!>           العناصر القطرية لـ A أيضًا، ولكن يفترض أنها الوحدة.
!>

LDA

!>          LDA هو INTEGER
!>           عند الدخول، تحدد LDA البعد الأول لـ A كما تم تعريفه
!>           في البرنامج المستدعي (الفرعي). عندما تكون SIDE = 'L' أو 'l'،
!>           يجب أن يكون LDA على الأقل max( 1, m )، عندما تكون SIDE = 'R' أو 'r'
!>           فيجب أن يكون LDA على الأقل max( 1, n ).
!>

B

!>          B هي مصفوفة COMPLEX، ذات البعد ( LDB, N )
!>           قبل الدخول، يجب أن يحتوي الجزء الأيسر من المصفوفة m × n في المصفوفة B على
!>           مصفوفة الجانب الأيمن B، وعند الخروج يتم استبدالها
!>           بالمصفوفة الحل X.
!>

LDB

!>          LDB هو INTEGER
!>           عند الدخول، تحدد LDB البعد الأول لـ B كما تم تعريفه
!>           في البرنامج المستدعي (الفرعي). يجب أن يكون LDB على الأقل
!>           max( 1, m ).
!>
