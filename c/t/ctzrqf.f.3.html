subroutine ctzrqf (integer m, integer n, complex, dimension( lda, * ) a, integer lda, complex, dimension( * ) tau, integer info)

CTZRQF

الغرض:

!>
!> هذا الروتين مهمل وتم استبداله بالروتين CTZRZF.
!>
!> يقوم CTZRQF بتحويل المصفوفة المعينة العليا ذات البعد M×N ( M<=N ) المعقدة A
!> إلى شكل مثلثي علوي عن طريق تحويلات وحدوية.
!>
!> يتم تحليل المصفوفة المعينة العليا A على النحو التالي:
!>
!>    A = ( R  0 ) * Z,
!>
!> حيث Z مصفوفة وحدوية N×N و R مصفوفة مثلثية عليا M×M.
!>

المعلمات:

M 
!>          M هو عدد صحيح (INTEGER)
!>          عدد صفوف المصفوفة A. M >= 0.
!>

N

!>          N هو عدد صحيح (INTEGER)
!>          عدد أعمدة المصفوفة A. N >= M.
!>

A

!>          A هو مصفوفة معقدة (COMPLEX)، البعد (LDA,N)
!>          عند الدخول، يجب أن يحتوي الجزء العلوي المعين من المصفوفة A ذي البعد M×N على المصفوفة المراد تحليلها.
!>          عند الخروج، يحتوي الجزء العلوي المثلثي من A ذي البعد M×M على المصفوفة المثلثية العليا R، والعناصر من M+1 إلى N في الصفوف M الأولى من A، مع المصفوفة TAU، تمثل المصفوفة الوحدوية Z كناتج لـ M من المحددات الأولية.
!>

LDA

!>          LDA هو عدد صحيح (INTEGER)
!>          البعد الرائد للمصفوفة A. LDA >= max(1,M).
!>

TAU

!>          TAU هي مصفوفة معقدة (COMPLEX)، البعد (M)
!>          العوامل القياسية للمحددات الأولية.
!>

INFO

!>          INFO هو عدد صحيح (INTEGER)
!>          = 0: خروج ناجح
!>          < 0: إذا كان INFO = -i، فإن الوسيط i لديه قيمة غير قانونية
!>

تفاصيل إضافية:

!>
!>  يتم الحصول على التحليل باستخدام طريقة هاوسهولدر. مصفوفة التحويل k،
!>  Z( k )، التي يتم استخدام المتبادل المترافق لها لإدخال أصفار في الصف (m - k + 1) من A،
!>  يتم إعطاؤها في الشكل
!>
!>     Z( k ) = ( I     0   )،
!>              ( 0  T( k ) )
!>
!>  حيث
!>
!>     T( k ) = I - tau*u( k )*u( k )**H،   u( k ) = (   1    )،
!>                                                    (   0    )
!>                                                    ( z( k ) )
!>
!>  tau هو عامل قياسي و z( k ) هو متجه بعناصر ( n - m ).
!>  يتم اختيار tau و z( k ) لإلغاء عناصر الصف k من X.
!>
!>  يتم إرجاع العامل tau في العنصر k من TAU والمتجه
!>  u( k ) في الصف k من A، بحيث تكون عناصر z( k )
!>  في a( k, m + 1 )، ...، a( k, n ). يتم إرجاع عناصر R في
!>  الجزء المثلثي العلوي من A.
!>
!>  Z معطاة بواسطة
!>
!>     Z =  Z( 1 ) * Z( 2 ) * ... * Z( m ).
!>
