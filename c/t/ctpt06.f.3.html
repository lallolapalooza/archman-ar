subroutine ctpt06 (real rcond, real rcondc, character uplo, character diag, integer n, complex, dimension( * ) ap, real, dimension( * ) rwork, real rat)

CTPT06

الغرض:

!>
!> CTPT06 تحسب نسبة اختبار تقارن RCOND (مقلوب رقم الشرط للمصفوفة المثلثية A) و RCONDC، التقدير
!> المحسوب بواسطة CTPCON. يتم استخدام معلومات عن المصفوفة المثلثية
!> إذا كان أحد التقديرين صفرًا والآخر غير صفري لتحديد ما إذا كان
!> التدفق السفلي في التقدير مبررًا.
!>

المعلمات

RCOND 
!>          RCOND هو REAL
!>          تقدير مقلوب رقم الشرط الذي تم الحصول عليه عن طريق
!>          تشكيل معكوس المصفوفة A بشكل صريح وحساب
!>          RCOND = 1/( norm(A) * norm(inv(A)) ).
!>

RCONDC

!>          RCONDC هو REAL
!>          تقدير مقلوب رقم الشرط المحسوب بواسطة
!>          CTPCON.
!>

UPLO

!>          UPLO هو CHARACTER
!>          تحدد ما إذا كانت المصفوفة A مثلثية عليا أم سفلى.
!>          = 'U':  مثلثية عليا
!>          = 'L':  مثلثية سفلى
!>

DIAG

!>          DIAG هو CHARACTER
!>          تحدد ما إذا كانت المصفوفة A مثلثية موحدة أم لا.
!>          = 'N':  مثلثية غير موحدة
!>          = 'U':  مثلثية موحدة
!>

N

!>          N هو INTEGER
!>          رتبة المصفوفة A.  N >= 0.
!>

AP

!>          AP هو مصفوفة COMPLEX، البعد (N*(N+1)/2)
!>          المصفوفة المثلثية العليا أو السفلى A، مجمعة عمودياً في
!>          مصفوفة خطية. يتم تخزين العمود j من A في المصفوفة
!>          AP كما يلي:
!>          إذا كان UPLO = 'U'، فإن AP((j-1)*j/2 + i) = A(i,j) لـ 1<=i<=j؛
!>          إذا كان UPLO = 'L'،
!>             AP((j-1)*(n-j) + j*(j+1)/2 + i-j) = A(i,j) لـ j<=i<=n.
!>

RWORK

!>          RWORK هو مصفوفة REAL، البعد (N)
!>

RAT

!>          RAT هو REAL
!>          نسبة الاختبار. إذا كان RCOND و RCONDC كلاهما غير صفري،
!>             RAT = MAX( RCOND, RCONDC )/MIN( RCOND, RCONDC ) - 1.
!>          إذا كان RAT = 0، فإن التقديرين متطابقان تمامًا.
!>
