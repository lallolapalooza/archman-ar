توثيق الوظيفة/الروتين الفرعي

subroutine ctgex2 (logical wantq, logical wantz, integer n, complex, dimension( lda, * ) a, integer lda, complex, dimension( ldb, * ) b, integer ldb, complex, dimension( ldq, * ) q, integer ldq, complex, dimension( ldz, * ) z, integer ldz, integer j1, integer info)

CTGEX2 تبادل كتل قطرية متجاورة 1x1 (A11,B11) و (A22,B22) في زوج مصفوفات مثلثية علوية (A, B) من خلال تحويل تكافؤ وحدوي.

الغرض:

!>
!> CTGEX2 تبادل كتل قطرية متجاورة 1x1 (A11,B11) و (A22,B22)
!> في زوج مصفوفات مثلثية علوية (A, B) من خلال تحويل تكافؤ وحدوي.
!>
!> يجب أن تكون (A, B) في الشكل المعياري العام، أي أن A و
!> B كلاهما مثلثيان علويان.
!>
!> اختياريًا، يتم تحديث مصفوفات المتجهات المعيارية العامة Q و Z.
!>
!>        Q(in) * A(in) * Z(in)**H = Q(out) * A(out) * Z(out)**H
!>        Q(in) * B(in) * Z(in)**H = Q(out) * B(out) * Z(out)**H
!>
!>

المعلمات

WANTQ 
!>          WANTQ هو LOGICAL
!>          .TRUE. : تحديث مصفوفة التحويل اليسرى Q؛
!>          .FALSE.: لا تحديث Q.
!>

WANTZ

!>          WANTZ هو LOGICAL
!>          .TRUE. : تحديث مصفوفة التحويل اليمنى Z؛
!>          .FALSE.: لا تحديث Z.
!>

N

!>          N هو INTEGER
!>          ترتيب المصفوفتين A و B. N >= 0.
!>

A

!>          A هو مصفوفة COMPLEX، أبعادها (LDA,N)
!>          عند الإدخال، المصفوفة A في الزوج (A, B).
!>          عند الخروج، المصفوفة A المحدثة.
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرئيسي للمصفوفة A. LDA >= max(1,N).
!>

B

!>          B هو مصفوفة COMPLEX، أبعادها (LDB,N)
!>          عند الإدخال، المصفوفة B في الزوج (A, B).
!>          عند الخروج، المصفوفة B المحدثة.
!>

LDB

!>          LDB هو INTEGER
!>          البعد الرئيسي للمصفوفة B. LDB >= max(1,N).
!>

Q

!>          Q هو مصفوفة COMPLEX، أبعادها (LDQ,N)
!>          إذا كان WANTQ = .TRUE، عند الإدخال، المصفوفة الوحدوية Q. عند الخروج،
!>          المصفوفة Q المحدثة.
!>          لا يتم الرجوع إليها إذا كان WANTQ = .FALSE..
!>

LDQ

!>          LDQ هو INTEGER
!>          البعد الرئيسي للمصفوفة Q. LDQ >= 1؛
!>          إذا كان WANTQ = .TRUE.، LDQ >= N.
!>

Z

!>          Z هو مصفوفة COMPLEX، أبعادها (LDZ,N)
!>          إذا كان WANTZ = .TRUE، عند الإدخال، المصفوفة الوحدوية Z. عند الخروج،
!>          المصفوفة Z المحدثة.
!>          لا يتم الرجوع إليها إذا كان WANTZ = .FALSE..
!>

LDZ

!>          LDZ هو INTEGER
!>          البعد الرئيسي للمصفوفة Z. LDZ >= 1؛
!>          إذا كان WANTZ = .TRUE.، LDZ >= N.
!>

J1

!>          J1 هو INTEGER
!>          المؤشر إلى الكتلة الأولى (A11, B11).
!>

INFO

!>          INFO هو INTEGER
!>           =0:  الخروج بنجاح.
!>           =1:  ستكون زوج المصفوفات المحول (A, B) بعيدًا جدًا
!>                عن الشكل المعياري العام؛ المشكلة غير مستقرة
!>                شرطيًا.
!>
