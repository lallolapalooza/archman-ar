subroutine ctrt02 (character uplo, character trans, character diag, integer n, integer nrhs, complex, dimension( lda, * ) a, integer lda, complex, dimension( ldx, * ) x, integer ldx, complex, dimension( ldb, * ) b, integer ldb, complex, dimension( * ) work, real, dimension( * ) rwork, real resid)

CTRT02

الغرض:

!>
!> CTRT02 يحسب الباقي للحل المحسوب لنظام المعادلات الخطية المثلثي op(A)*X = B، حيث A مصفوفة مثلثية. نسبة الاختبار هي الحد الأقصى لـ
!>    norm(b - op(A)*x) / ( ||op(A)||_1 * norm(x) * EPS )،
!> حيث op(A) = A, A**T, أو A**H، b هو عمود B، x هو متجه الحل، و EPS هو ثابت خطأ الآلة.
!>

المعلمات

UPLO 
!>          UPLO هي CHARACTER*1
!>          تحدد ما إذا كانت المصفوفة A علوية أم سفلية مثلثية.
!>          = 'U':  علوية مثلثية
!>          = 'L':  سفلية مثلثية
!>

TRANS

!>          TRANS هي CHARACTER*1
!>          تحدد العملية المطبقة على A.
!>          = 'N':  A    * X = B  (بدون تبديل)
!>          = 'T':  A**T * X = B  (تبديل)
!>          = 'C':  A**H * X = B  (تبديل مترافق)
!>

DIAG

!>          DIAG هي CHARACTER*1
!>          تحدد ما إذا كانت المصفوفة A مثلثية واحدية أم لا.
!>          = 'N':  غير واحدية مثلثية
!>          = 'U':  واحدية مثلثية
!>

N

!>          N هو INTEGER
!>          رتبة المصفوفة A.  N >= 0.
!>

NRHS

!>          NRHS هو INTEGER
!>          عدد الأطراف اليمنى، أي عدد أعمدة المصفوفات X و B.  NRHS >= 0.
!>

A

!>          A هي مصفوفة COMPLEX، ذات البعد (LDA,N)
!>          المصفوفة المثلثية A.  إذا كان UPLO = 'U'، فإن الجزء العلوي المثلثي الذي طوله n في n
!>          من المصفوفة A يحتوي على المصفوفة المثلثية العلوية، والجزء المثلثي السفلي الصارم
!>          لـ A لا تتم الإشارة إليه.  إذا كان UPLO = 'L'، فإن الجزء السفلي المثلثي الذي طوله n في n
!>          من المصفوفة A يحتوي على المصفوفة المثلثية السفلية، والجزء المثلثي العلوي الصارم
!>          لـ A لا تتم الإشارة إليه.  إذا كان DIAG = 'U'، فإن العناصر القطرية لـ A
!>          لا تتم الإشارة إليها وتُفترض أنها 1.
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرئيسي للمصفوفة A.  LDA >= max(1,N).
!>

X

!>          X هي مصفوفة COMPLEX، ذات البعد (LDX,NRHS)
!>          متجهات الحل المحسوبة لنظام المعادلات الخطية.
!>

LDX

!>          LDX هو INTEGER
!>          البعد الرئيسي للمصفوفة X.  LDX >= max(1,N).
!>

B

!>          B هي مصفوفة COMPLEX، ذات البعد (LDB,NRHS)
!>          متجهات الطرف الأيمن لنظام المعادلات الخطية.
!>

LDB

!>          LDB هو INTEGER
!>          البعد الرئيسي للمصفوفة B.  LDB >= max(1,N).
!>

WORK

!>          WORK هي مصفوفة COMPLEX، ذات البعد (N)
!>

RWORK

!>          RWORK هي مصفوفة REAL، ذات البعد (N)
!>

RESID

!>          RESID هو REAL
!>          الحد الأقصى لعدد الأطراف اليمنى لـ
!>          norm(op(A)*X - B) / ( norm(op(A)) * norm(X) * EPS ).
!>
