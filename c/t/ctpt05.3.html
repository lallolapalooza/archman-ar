توثيق الوظيفة/الإجراء الفرعي

subroutine ctpt05 (character uplo, character trans, character diag, integer n, integer nrhs, complex, dimension( * ) ap, complex, dimension( ldb, * ) b, integer ldb, complex, dimension( ldx, * ) x, integer ldx, complex, dimension( ldxact, * ) xact, integer ldxact, real, dimension( * ) ferr, real, dimension( * ) berr, real, dimension( * ) reslts)

CTPT05

الغرض:

!>
!> CTPT05 تختبر حدود الخطأ للتحسين التكراري للحل المحسوب لنظام المعادلات A*X = B، حيث A مصفوفة مثلثة في شكل تخزين مضغوط.
!>
!> RESLTS(1) = اختبار حد الخطأ
!>           = norm(X - XACT) / ( norm(X) * FERR )
!>
!> يتم إرجاع قيمة كبيرة إذا لم تكن هذه النسبة أقل من واحد.
!>
!> RESLTS(2) = الباقي من إجراء التحسين التكراري
!>           = الحد الأقصى لـ BERR / ( (n+1)*EPS + (*) )، حيث
!>             (*) = (n+1)*UNFL / (min_i (abs(A)*abs(X) +abs(b))_i )
!>

المعلمات

UPLO 
!>          UPLO هو CHARACTER*1
!>          يحدد ما إذا كانت المصفوفة A علوية أو سفلية مثلثة.
!>          = 'U':  مثلثة علوية
!>          = 'L':  مثلثة سفلية
!>

TRANS

!>          TRANS هو CHARACTER*1
!>          يحدد شكل نظام المعادلات.
!>          = 'N':  A * X = B  (بدون تبديل)
!>          = 'T':  A'* X = B  (تبديل)
!>          = 'C':  A'* X = B  (تبديل مترافق = تبديل)
!>

DIAG

!>          DIAG هو CHARACTER*1
!>          يحدد ما إذا كانت المصفوفة A موحدة مثلثة أم لا.
!>          = 'N':  غير موحدة مثلثة
!>          = 'U':  موحدة مثلثة
!>

N

!>          N هو INTEGER
!>          عدد الصفوف في المصفوفات X، B، و XACT، وترتيب المصفوفة A. N >= 0.
!>

NRHS

!>          NRHS هو INTEGER
!>          عدد الأعمدة في المصفوفات X، B، و XACT. NRHS >= 0.
!>

AP

!>          AP هو مصفوفة COMPLEX، البعد (N*(N+1)/2)
!>          المصفوفة المثلثة العلوية أو السفلية A، مضغوطة عموديًا في مصفوفة خطية.
!>          يتم تخزين العمود j من A في المصفوفة AP كما يلي:
!>          إذا كانت UPLO = 'U'، فإن AP(i + (j-1)*j/2) = A(i,j) لـ 1<=i<=j؛
!>          إذا كانت UPLO = 'L'، فإن AP(i + (j-1)*(2n-j)/2) = A(i,j) لـ j<=i<=n.
!>          إذا كانت DIAG = 'U'، لا يتم الإشارة إلى العناصر القطرية لـ A
!>          ويفترض أنها 1.
!>

B

!>          B هي مصفوفة COMPLEX، البعد (LDB,NRHS)
!>          متجهات الطرف الأيمن لأنظمة المعادلات الخطية.
!>

LDB

!>          LDB هو INTEGER
!>          البعد الرئيسي للمصفوفة B. LDB >= max(1,N).
!>

X

!>          X هي مصفوفة COMPLEX، البعد (LDX,NRHS)
!>          متجهات الحل المحسوبة. يتم تخزين كل متجه كعمود في المصفوفة X.
!>

LDX

!>          LDX هو INTEGER
!>          البعد الرئيسي للمصفوفة X. LDX >= max(1,N).
!>

XACT

!>          XACT هي مصفوفة COMPLEX، البعد (LDX,NRHS)
!>          متجهات الحل الدقيقة. يتم تخزين كل متجه كعمود في المصفوفة XACT.
!>

LDXACT

!>          LDXACT هو INTEGER
!>          البعد الرئيسي للمصفوفة XACT. LDXACT >= max(1,N).
!>

FERR

!>          FERR هي مصفوفة REAL، البعد (NRHS)
!>          حدود الخطأ الأمامي المقدرة لكل متجه حل X.
!>          إذا كانت XTRUE هي الحل الحقيقي، فإن FERR تحد من حجم أكبر إدخال في (X - XTRUE) مقسومًا على حجم أكبر إدخال في X.
!>

BERR

!>          BERR هي مصفوفة REAL، البعد (NRHS)
!>          الخطأ الخلفي النسبي المكوني لكل متجه حل
!>          (أي، أصغر تغيير نسبي في أي إدخال في A أو B يجعل X حلاً دقيقًا).
!>

RESLTS

!>          RESLTS هي مصفوفة REAL، البعد (2)
!>          الحد الأقصى لمتجهات الحل NRHS للنسب التالية:
!>          RESLTS(1) = norm(X - XACT) / ( norm(X) * FERR )
!>          RESLTS(2) = BERR / ( (n+1)*EPS + (*) )
!>
