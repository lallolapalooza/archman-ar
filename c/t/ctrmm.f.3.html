subroutine ctrmm (character side, character uplo, character transa, character diag, integer m, integer n, complex alpha, complex, dimension(lda,*) a, integer lda, complex, dimension(ldb,*) b, integer ldb)

CTRMM

الغرض:

!>
!> CTRMM  يؤدي إحدى عمليات المصفوفة-المصفوفة التالية:
!>
!>    B := alpha*op( A )*B,   أو   B := alpha*B*op( A )
!>
!> حيث  alpha  هو ثابت،  B  هي مصفوفة  m × n ،  A  هي مصفوفة مثلثية
!> وحيدة، أو غير وحيدة، علوية أو سفلية، و op( A )  هي إحدى العمليات التالية:
!>
!>    op( A ) = A   أو   op( A ) = A**T   أو   op( A ) = A**H.
!>

المعلمات

SIDE 
!>          SIDE هو CHARACTER*1
!>           عند الإدخال، يحدد  SIDE  ما إذا كانت  op( A )  تضرب  B  من
!>           اليسار أو اليمين على النحو التالي:
!>
!>              SIDE = 'L' أو 'l'   B := alpha*op( A )*B.
!>
!>              SIDE = 'R' أو 'r'   B := alpha*B*op( A ).
!>

UPLO

!>          UPLO هو CHARACTER*1
!>           عند الإدخال، يحدد  UPLO  ما إذا كانت المصفوفة  A  هي مصفوفة
!>           مثلثية علوية أو سفلية على النحو التالي:
!>
!>              UPLO = 'U' أو 'u'   A  هي مصفوفة مثلثية علوية.
!>
!>              UPLO = 'L' أو 'l'   A  هي مصفوفة مثلثية سفلية.
!>

TRANSA

!>          TRANSA هو CHARACTER*1
!>           عند الإدخال، يحدد  TRANSA  شكل  op( A )  الذي سيتم استخدامه في
!>           ضرب المصفوفات على النحو التالي:
!>
!>              TRANSA = 'N' أو 'n'   op( A ) = A.
!>
!>              TRANSA = 'T' أو 't'   op( A ) = A**T.
!>
!>              TRANSA = 'C' أو 'c'   op( A ) = A**H.
!>

DIAG

!>          DIAG هو CHARACTER*1
!>           عند الإدخال، يحدد  DIAG  ما إذا كانت  A  مثلثية وحيدة أم لا
!>           على النحو التالي:
!>
!>              DIAG = 'U' أو 'u'   يُفترض أن  A  مثلثية وحيدة.
!>
!>              DIAG = 'N' أو 'n'   لا يُفترض أن  A  مثلثية وحيدة.
!>

M

!>          M هو INTEGER
!>           عند الإدخال، يحدد  M  عدد صفوف  B. يجب أن يكون  M  على
!>           الأقل صفرًا.
!>

N

!>          N هو INTEGER
!>           عند الإدخال، يحدد  N  عدد أعمدة  B. يجب أن يكون  N  على
!>           الأقل صفرًا.
!>

ALPHA

!>          ALPHA هو COMPLEX
!>           عند الإدخال، يحدد  ALPHA  الثابت  alpha. عندما يكون  alpha
!>           صفرًا، لا يتم الإشارة إلى  A  ولا يلزم تعيين  B  قبل
!>           الدخول.
!>

A

!>          A هو مصفوفة COMPLEX، بُعدها ( LDA, k ) ، حيث  k  هو m
!>           عندما يكون  SIDE = 'L' أو 'l'  و n  عندما يكون  SIDE = 'R' أو 'r'.
!>           قبل الدخول مع  UPLO = 'U' أو 'u'، يجب أن تحتوي الجزء العلوي
!>           المثلثي الأمامي  k × k  من المصفوفة  A  على المصفوفة
!>           المثلثية العلوية، ولا يتم الإشارة إلى الجزء السفلي
!>           المثلثي الصارم من  A.
!>           قبل الدخول مع  UPLO = 'L' أو 'l'، يجب أن تحتوي الجزء السفلي
!>           المثلثي الأمامي  k × k  من المصفوفة  A  على المصفوفة
!>           المثلثية السفلية، ولا يتم الإشارة إلى الجزء العلوي
!>           المثلثي الصارم من  A.
!>           لاحظ أنه عندما يكون  DIAG = 'U' أو 'u'، لا يتم الإشارة إلى
!>           العناصر القطرية لـ  A  أيضًا، ولكن يُفترض أنها الوحدة.
!>

LDA

!>          LDA هو INTEGER
!>           عند الإدخال، يحدد  LDA  البعد الأول لـ  A  كما تم تعريفه
!>           في البرنامج المتصل (الفرعي). عندما يكون  SIDE = 'L' أو 'l' ،
!>           يجب أن يكون  LDA  على الأقل  max( 1, m ) ، وعندما يكون
!>           SIDE = 'R' أو 'r' ، يجب أن يكون  LDA  على الأقل max( 1, n ).
!>

B

!>          B هو مصفوفة COMPLEX، بُعدها ( LDB, N ).
!>           قبل الدخول، يجب أن تحتوي الجزء الأمامي  m × n  من المصفوفة  B
!>           على المصفوفة  B، وعند الخروج يتم استبدالها بالمصفوفة المحولة.
!>

LDB

!>          LDB هو INTEGER
!>           عند الإدخال، يحدد  LDB  البعد الأول لـ  B  كما تم تعريفه
!>           في البرنامج المتصل (الفرعي). يجب أن يكون  LDB  على الأقل
!>           max( 1, m ).
!>
