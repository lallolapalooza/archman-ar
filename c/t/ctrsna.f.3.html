subroutine ctrsna (job, howmny, select, n, t, ldt, vl, ldvl, vr, ldvr, s, sep, mm, m, work, ldwork, rwork, info)

CTRSNA

الغرض:

!>
!> CTRSNA يقدر الأرقام الشرطية المتبادلة للقيم الذاتية المحددة
!> و/أو المتجهات الذاتية اليمنى لمصفوفة مثلثية علوية معقدة T (أو لأي مصفوفة Q*T*Q**H مع Q وحدوية).
!>

المعلمات

JOB 
!>          JOB هو CHARACTER*1
!>          يحدد ما إذا كانت الأرقام الشرطية مطلوبة للقيم الذاتية (S) أو المتجهات الذاتية (SEP):
!>          = 'E': للقيم الذاتية فقط (S)؛
!>          = 'V': للمتجهات الذاتية فقط (SEP)؛
!>          = 'B': لكل من القيم الذاتية والمتجهات الذاتية (S و SEP).
!>

HOWMNY

!>          HOWMNY هو CHARACTER*1
!>          = 'A': حساب الأرقام الشرطية لجميع الأزواج الذاتية؛
!>          = 'S': حساب الأرقام الشرطية للأزواج الذاتية المحددة
!>                 المحددة بواسطة المصفوفة SELECT.
!>

SELECT

!>          SELECT هي مصفوفة LOGICAL، البعد (N)
!>          إذا كان HOWMNY = 'S'، تحدد SELECT الأزواج الذاتية التي
!>          تتطلب الأرقام الشرطية لها. لتحديد الأرقام الشرطية
!>          للزوج الذاتي j-th، يجب تعيين SELECT(j) إلى .TRUE..
!>          إذا كان HOWMNY = 'A'، فلن يتم الرجوع إلى SELECT.
!>

N

!>          N هو INTEGER
!>          ترتيب المصفوفة T. N >= 0.
!>

T

!>          T هي مصفوفة COMPLEX، البعد (LDT,N)
!>          المصفوفة المثلثية العليا T.
!>

LDT

!>          LDT هو INTEGER
!>          البعد الرائد للمصفوفة T. LDT >= max(1,N).
!>

VL

!>          VL هي مصفوفة COMPLEX، البعد (LDVL,M)
!>          إذا كان JOB = 'E' أو 'B'، يجب أن تحتوي VL على المتجهات الذاتية اليسرى لـ T
!>          (أو لأي Q*T*Q**H مع Q وحدوية)، المقابلة للأزواج الذاتية
!>          المحددة بواسطة HOWMNY و SELECT. يجب تخزين المتجهات الذاتية
!>          في أعمدة متتالية من VL، كما تم إرجاعه بواسطة
!>          CHSEIN أو CTREVC.
!>          إذا كان JOB = 'V'، فلن يتم الرجوع إلى VL.
!>

LDVL

!>          LDVL هو INTEGER
!>          البعد الرائد للمصفوفة VL.
!>          LDVL >= 1؛ وإذا كان JOB = 'E' أو 'B'، فإن LDVL >= N.
!>

VR

!>          VR هي مصفوفة COMPLEX، البعد (LDVR,M)
!>          إذا كان JOB = 'E' أو 'B'، يجب أن تحتوي VR على المتجهات الذاتية اليمنى لـ T
!>          (أو لأي Q*T*Q**H مع Q وحدوية)، المقابلة للأزواج الذاتية
!>          المحددة بواسطة HOWMNY و SELECT. يجب تخزين المتجهات الذاتية
!>          في أعمدة متتالية من VR، كما تم إرجاعه بواسطة
!>          CHSEIN أو CTREVC.
!>          إذا كان JOB = 'V'، فلن يتم الرجوع إلى VR.
!>

LDVR

!>          LDVR هو INTEGER
!>          البعد الرائد للمصفوفة VR.
!>          LDVR >= 1؛ وإذا كان JOB = 'E' أو 'B'، فإن LDVR >= N.
!>

S

!>          S هي مصفوفة REAL، البعد (MM)
!>          إذا كان JOB = 'E' أو 'B'، فإن الأرقام الشرطية المتبادلة للقيم الذاتية
!>          المحددة، مخزنة في عناصر متتالية من
!>          المصفوفة. وبالتالي فإن S(j) و SEP(j) والأعمدة j-th من VL و VR
!>          تتوافق جميعها مع نفس الزوج الذاتي (ولكن ليس بالضرورة الزوج الذاتي j-th،
!>          إلا إذا تم تحديد جميع الأزواج الذاتية).
!>          إذا كان JOB = 'V'، فلن يتم الرجوع إلى S.
!>

SEP

!>          SEP هي مصفوفة REAL، البعد (MM)
!>          إذا كان JOB = 'V' أو 'B'، فإن الأرقام الشرطية المتبادلة المقدرة للمتجهات الذاتية
!>          المحددة، مخزنة في عناصر متتالية من
!>          المصفوفة.
!>          إذا كان JOB = 'E'، فلن يتم الرجوع إلى SEP.
!>

MM

!>          MM هو INTEGER
!>          عدد العناصر في المصفوفات S (إذا كان JOB = 'E' أو 'B')
!>           و/أو SEP (إذا كان JOB = 'V' أو 'B'). MM >= M.
!>

M

!>          M هو INTEGER
!>          عدد العناصر في المصفوفات S و/أو SEP المستخدمة فعليًا
!>          لتخزين الأرقام الشرطية المقدرة.
!>          إذا كان HOWMNY = 'A'، يتم تعيين M إلى N.
!>

WORK

!>          WORK هي مصفوفة COMPLEX، البعد (LDWORK,N+6)
!>          إذا كان JOB = 'E'، فلن يتم الرجوع إلى WORK.
!>

LDWORK

!>          LDWORK هو INTEGER
!>          البعد الرائد للمصفوفة WORK.
!>          LDWORK >= 1؛ وإذا كان JOB = 'V' أو 'B'، فإن LDWORK >= N.
!>

RWORK

!>          RWORK هي مصفوفة REAL، البعد (N)
!>          إذا كان JOB = 'E'، فلن يتم الرجوع إلى RWORK.
!>

INFO

!>          INFO هو INTEGER
!>          = 0: خروج ناجح
!>          < 0: إذا كان INFO = -i، فإن الوسيط i كان له قيمة غير قانونية
!>

تفاصيل إضافية:

!>
!>  الرقم المتبادل للشرط لرقم ذاتي lambda يُعرف بـ
!>
!>          S(lambda) = |v**H*u| / (norm(u)*norm(v))
!>
!>  حيث u و v هما المتجهان الذاتيان الأيمن والأيسر لـ T المقابلان
!>  لـ lambda؛ v**H يمثل منقول v المترافق، و norm(u)
!>  يمثل المعيار الإقليدي. هذه الأرقام الشرطية المتبادلة تقع دائمًا
!>  بين صفر (شرط سيء جدًا) وواحد (شرط جيد جدًا). إذا كان n = 1، يتم تعريف S(lambda) إلى 1.
!>
!>  يُعطى حد خطأ تقريبي لقيمة ذاتية محسوبة W(i) بواسطة
!>
!>                      EPS * norm(T) / S(i)
!>
!>  حيث EPS هي دقة الآلة.
!>
!>  الرقم المتبادل للشرط للمتجه الذاتي الأيمن u
!>  المقابل لـ lambda يُعرف كالتالي. نفترض
!>
!>              T = ( lambda  c  )
!>                  (   0    T22 )
!>
!>  ثم الرقم المتبادل للشرط هو
!>
!>          SEP( lambda, T22 ) = sigma-min( T22 - lambda*I )
!>
!>  حيث sigma-min يمثل أصغر قيمة مفردة. نقوم بتقدير
!>  أصغر قيمة مفردة بمقلوب تقدير
!>  للمعيار واحد لمقلوب T22 - lambda*I. إذا كان n = 1، يتم تعريف SEP(1) إلى abs(T(1,1)).
!>
!>  يُعطى حد خطأ تقريبي للمتجه الذاتي الأيمن المحسوب VR(i) بواسطة
!>
!>                      EPS * norm(T) / SEP(i)
!>
