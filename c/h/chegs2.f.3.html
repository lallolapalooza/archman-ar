توثيق الوظيفة/الروتين الفرعي

subroutine chegs2 (integer itype, character uplo, integer n, complex, dimension( lda, * ) a, integer lda, complex, dimension( ldb, * ) b, integer ldb, integer info)

CHEGS2 تختزل مشكلة قيمة ذاتية معتمدة هرميتية معتمدة إلى شكل قياسي، باستخدام نتائج التحليل التي تم الحصول عليها من cpotrf (خوارزمية غير محظورة).

الغرض:

!>
!> CHEGS2 تختزل مشكلة قيمة ذاتية معتمدة هرميتية معتمدة معقدة إلى شكل قياسي.
!>
!> إذا كان ITYPE = 1، فإن المشكلة هي A*x = lambda*B*x،
!> ويتم الكتابة فوق A بـ inv(U**H)*A*inv(U) أو inv(L)*A*inv(L**H)
!>
!> إذا كان ITYPE = 2 أو 3، فإن المشكلة هي A*B*x = lambda*x أو
!> B*A*x = lambda*x، ويتم الكتابة فوق A بـ U*A*U**H أو L**H *A*L.
!>
!> يجب أن يكون B قد تم تحليله مسبقًا على أنه U**H *U أو L*L**H بواسطة ZPOTRF.
!>

المعلمات

ITYPE 
!>          ITYPE هو عدد صحيح
!>          = 1: حساب inv(U**H)*A*inv(U) أو inv(L)*A*inv(L**H)؛
!>          = 2 أو 3: حساب U*A*U**H أو L**H *A*L.
!>

UPLO

!>          UPLO هو حرف واحد
!>          يحدد ما إذا كان الجزء العلوي أو السفلي المثلثي للمصفوفة الهرميتية A مخزنًا، وكيف تم تحليل B.
!>          = 'U':  مثلثي علوي
!>          = 'L':  مثلثي سفلي
!>

N

!>          N هو عدد صحيح
!>          ترتيب المصفوفات A و B.  N >= 0.
!>

A

!>          A هي مصفوفة معقدة، البعد (LDA,N)
!>          عند الدخول، المصفوفة الهرميتية A.  إذا كان UPLO = 'U'، فإن الجزء العلوي المثلثي الرئيسي ذو الأبعاد n × n من A يحتوي على الجزء العلوي
!>          المثلثي من المصفوفة A، والجزء المثلثي السفلي الصارم من A لا يتم الرجوع إليه.  إذا كان UPLO = 'L'، فإن الجزء السفلي المثلثي الرئيسي ذو الأبعاد n × n من A يحتوي على الجزء السفلي
!>          المثلثي من المصفوفة A، والجزء المثلثي العلوي الصارم من A لا يتم الرجوع إليه.
!>
!>          عند الخروج، إذا كان INFO = 0، فإن المصفوفة المحولة، المخزنة بنفس تنسيق A.
!>

LDA

!>          LDA هو عدد صحيح
!>          البعد الرئيسي للمصفوفة A.  LDA >= max(1,N).
!>

B

!>          B هي مصفوفة معقدة، البعد (LDB,N)
!>          العامل المثلثي من التحليل الكوليسكي لـ B،
!>          كما تم إرجاعه بواسطة CPOTRF.
!>          يتم تعديل B بواسطة الروتين ولكن يتم استعادته عند الخروج.
!>

LDB

!>          LDB هو عدد صحيح
!>          البعد الرئيسي للمصفوفة B.  LDB >= max(1,N).
!>

INFO

!>          INFO هو عدد صحيح
!>          = 0:  خروج ناجح.
!>          < 0:  إذا كان INFO = -i، فإن الوسيط i كان له قيمة غير قانونية.
!>
