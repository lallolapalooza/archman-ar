subroutine chet22 (integer itype, character uplo, integer n, integer m, integer kband, complex, dimension( lda, * ) a, integer lda, real, dimension( * ) d, real, dimension( * ) e, complex, dimension( ldu, * ) u, integer ldu, complex, dimension( ldv, * ) v, integer ldv, complex, dimension( * ) tau, complex, dimension( * ) work, real, dimension( * ) rwork, real, dimension( 2 ) result)

الغرض:

!>
!>      CHET22  بشكل عام يتحقق من تفكيك للشكل
!>
!>              A U = U S
!>
!>      حيث A هو هيرميتي معقد، وأعمدة U متعامدة،
!>      و S قطري (إذا كان KBAND=0) أو ثلاثي القطر متماثل (إذا كان
!>      KBAND=1). إذا كان ITYPE=1، يتم تمثيل U كمصفوفة كثيفة،
!>      وإلا يتم التعبير عن U كناتج تحويلات هاوسهولدر،
!>      والتي يتم تخزين متجهات في المصفوفة  
!>      الثوابت القياسية في  
!>       سنستخدم الحرف
!>       للإشارة إلى ناتج تحويلات هاوسهولدر
!>      (والذي يجب أن يساوي U).
!>
!>      على وجه التحديد، إذا كان ITYPE=1، فإن:
!>
!>              RESULT(1) = | U**H A U - S | / ( |A| m ulp ) and
!>              RESULT(2) = | I - U**H U | / ( m ulp )
!> 
!>  ITYPE   INTEGER
!>          يحدد نوع الاختبارات التي سيتم إجراؤها.
!>          1: U ممثلة كمصفوفة متعامدة كثيفة:
!>             RESULT(1) = | A - U S U**H | / ( |A| n ulp )  and
!>             RESULT(2) = | I - U U**H | / ( n ulp )
!>
!>  UPLO    CHARACTER
!>          إذا كان UPLO='U'، فسيتم استخدام المثلث العلوي لـ A ولن
!>          يتم الرجوع إلى المثلث السفلي (الصارم). إذا كان
!>          UPLO='L'، فسيتم استخدام المثلث السفلي لـ A ولن
!>          يتم الرجوع إلى المثلث العلوي (الصارم).
!>          لم يتم تعديله.
!>
!>  N       INTEGER
!>          حجم المصفوفة. إذا كانت صفرًا، فلن تفعل CHET22 شيئًا.
!>          يجب أن تكون على الأقل صفرًا.
!>          لم يتم تعديله.
!>
!>  M       INTEGER
!>          عدد أعمدة U. إذا كانت صفرًا، فلن تفعل CHET22 شيئًا.
!>          يجب أن تكون على الأقل صفرًا.
!>          لم يتم تعديله.
!>
!>  KBAND   INTEGER
!>          عرض النطاق الترددي للمصفوفة. قد يكون صفرًا أو واحدًا فقط.
!>          إذا كان صفرًا، فإن S قطري، ولا يتم الرجوع إلى E. إذا كان
!>          واحدًا، فإن S ثلاثي القطر متماثل.
!>          لم يتم تعديله.
!>
!>  A       COMPLEX array, dimension (LDA , N)
!>          المصفوفة الأصلية (غير المفككة). يُفترض أنها
!>          متماثلة، وسيتم الرجوع فقط إلى الجزء العلوي (UPLO='U') أو
!>          الجزء السفلي (UPLO='L') فقط.
!>          لم يتم تعديله.
!>
!>  LDA     INTEGER
!>          البعد الرائد لـ A. يجب أن يكون على الأقل 1
!>          وعلى الأقل N.
!>          لم يتم تعديله.
!>
!>  D       REAL array, dimension (N)
!>          القطر للمصفوفة القطرية (ثلاثية الأقطار المتماثلة).
!>          لم يتم تعديله.
!>
!>  E       REAL array, dimension (N)
!>          القُطر غير القطري للمصفوفة القطرية (ثلاثية الأقطار المتماثلة).
!>          يتم تجاهل E(1)، E(2) هو العنصر (1,2) و (2,1)، إلخ.
!>          لا يتم الرجوع إليه إذا كان KBAND=0.
!>          لم يتم تعديله.
!>
!>  U       COMPLEX array, dimension (LDU, N)
!>          إذا كان ITYPE=1، فهذه تحتوي على المصفوفة المتعامدة
!>          في التفكيك، ممثلة كمصفوفة كثيفة.
!>          لم يتم تعديله.
!>
!>  LDU     INTEGER
!>          البعد الرائد لـ U. يجب أن يكون LDU على الأقل N و
!>          على الأقل 1.
!>          لم يتم تعديله.
!>
!>  V       COMPLEX array, dimension (LDV, N)
!>          إذا كان ITYPE=2 أو 3، فإن المثلث السفلي لهذه المصفوفة يحتوي
!>          على متجهات هاوسهولدر التي تصف المصفوفة المتعامدة
!>          في التفكيك. إذا كان ITYPE=1، فلن يتم
!>          الرجوع إليها.
!>          لم يتم تعديله.
!>
!>  LDV     INTEGER
!>          البعد الرائد لـ V. يجب أن يكون LDV على الأقل N و
!>          على الأقل 1.
!>          لم يتم تعديله.
!>
!>  TAU     COMPLEX array, dimension (N)
!>          إذا كان ITYPE >= 2، فإن TAU(j) هو المعامل القياسي لـ
!>          v(j) v(j)**H في تحويل هاوسهولدر H(j) لـ
!!>          الناتج  U = H(1)...H(n-2)
!>          إذا كان ITYPE < 2، فلن يتم الرجوع إلى TAU.
!>          لم يتم تعديله.
!>
!>  WORK    COMPLEX array, dimension (2*N**2)
!>          مساحة العمل.
!>          تم التعديل.
!>
!>  RWORK   REAL array, dimension (N)
!>          مساحة العمل.
!>          تم التعديل.
!>
!>  RESULT  REAL array, dimension (2)
!>          القيم التي تم حسابها بواسطة الاختبارين الموصوفين أعلاه.
!>          القيم محدودة حاليًا بـ 1/ulp، لتجنب تجاوز السعة.
!>          يتم تعديل RESULT(1) دائمًا. يتم تعديل RESULT(2) فقط
!>          إذا كان LDU على الأقل N.
!>          تم التعديل.
!>
