وثائق الوظيفة/الروتين الفرعي

subroutine chbtrd (character vect, character uplo, integer n, integer kd, complex, dimension( ldab, * ) ab, integer ldab, real, dimension( * ) d, real, dimension( * ) e, complex, dimension( ldq, * ) q, integer ldq, complex, dimension( * ) work, integer info)

CHBTRD

الغرض:

!>
!> CHBTRD يختزل مصفوفة هرميتية A ذات النطاق المركب إلى شكل T المتماثل الحقيقي
!> ثلاثي الأقطار عن طريق تحويل تشابه وحدوي:
!> Q**H * A * Q = T.
!>

المعلمات

VECT 
!>          VECT هو CHARACTER*1
!>          = 'N':  لا تشكل Q؛
!>          = 'V':  شكل Q؛
!>          = 'U':  قم بتحديث مصفوفة X، عن طريق تشكيل X*Q.
!>

UPLO

!>          UPLO هو CHARACTER*1
!>          = 'U':  يتم تخزين المثلث العلوي من A؛
!>          = 'L':  يتم تخزين المثلث السفلي من A.
!>

N

!>          N هو INTEGER
!>          ترتيب المصفوفة A.  N >= 0.
!>

KD

!>          KD هو INTEGER
!>          عدد الأقطار الفائقة للمصفوفة A إذا كان UPLO = 'U'،
!>          أو عدد الأقطار الفرعية إذا كان UPLO = 'L'.  KD >= 0.
!>

AB

!>          AB هو مصفوفة COMPLEX، البعد (LDAB,N)
!>          عند الدخول، المثلث العلوي أو السفلي للمصفوفة الهرميتية A،
!>          مخزنة في الصفوف KD+1 الأولى من المصفوفة.  يتم تخزين العمود j
!>          من A في العمود j من المصفوفة AB على النحو التالي:
!>          إذا كان UPLO = 'U'، AB(kd+1+i-j,j) = A(i,j) لـ max(1,j-kd)<=i<=j؛
!>          إذا كان UPLO = 'L'، AB(1+i-j,j)    = A(i,j) لـ j<=i<=min(n,j+kd).
!>          عند الخروج، يتم الكتابة فوق عناصر القطر للمصفوفة T بواسطة
!>          عناصر القطر للمصفوفة ثلاثية الأقطار T؛ إذا كان KD > 0، فإن العناصر
!>          على القطر الفائق الأول (إذا كان UPLO = 'U') أو القطر الفرعي
!>          الأول (إذا كان UPLO = 'L') يتم الكتابة فوقها بواسطة العناصر
!>          خارج القطر لـ T؛ يتم الكتابة فوق باقي AB بالقيم التي تم إنشاؤها
!>          أثناء الاختزال.
!>

LDAB

!>          LDAB هو INTEGER
!>          البعد الرائد للمصفوفة AB.  LDAB >= KD+1.
!>

D

!>          D هي مصفوفة REAL، البعد (N)
!>          عناصر القطر للمصفوفة ثلاثية الأقطار T.
!>

E

!>          E هي مصفوفة REAL، البعد (N-1)
!>          العناصر خارج القطر للمصفوفة ثلاثية الأقطار T:
!>          E(i) = T(i,i+1) إذا كان UPLO = 'U'؛ E(i) = T(i+1,i) إذا كان UPLO = 'L'.
!>

Q

!>          Q هي مصفوفة COMPLEX، البعد (LDQ,N)
!>          عند الدخول، إذا كان VECT = 'U'، فيجب أن تحتوي Q على مصفوفة
!>          وحدوية X بحجم N*N؛ إذا كان VECT = 'N' أو 'V'، فلا يلزم تعيين Q.
!>
!>          عند الخروج:
!>          إذا كان VECT = 'V'، فإن Q تحتوي على المصفوفة الوحدوية Q بحجم N*N؛
!>          إذا كان VECT = 'U'، فإن Q تحتوي على حاصل ضرب X*Q؛
!>          إذا كان VECT = 'N'، فلا يتم الاستشهاد بالمصفوفة Q.
!>

LDQ

!>          LDQ هو INTEGER
!>          البعد الرائد للمصفوفة Q.
!>          LDQ >= 1، و LDQ >= N إذا كان VECT = 'V' أو 'U'.
!>

WORK

!>          WORK هي مصفوفة COMPLEX، البعد (N)
!>

INFO

!>          INFO هو INTEGER
!>          = 0:  انتهاء ناجح
!>          < 0:  إذا كان INFO = -i، فإن الوسيط i كان له قيمة غير قانونية
!>
