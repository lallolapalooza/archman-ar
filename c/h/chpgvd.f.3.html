subroutine chpgvd (integer itype, character jobz, character uplo, integer n, complex, dimension( * ) ap, complex, dimension( * ) bp, real, dimension( * ) w, complex, dimension( ldz, * ) z, integer ldz, complex, dimension( * ) work, integer lwork, real, dimension( * ) rwork, integer lrwork, integer, dimension( * ) iwork, integer liwork, integer info)

CHPGVD

الغرض:

!>
!> CHPGVD تحسب جميع القيم الذاتية، اختياريًا، المتجهات الذاتية
!> لمشكلة الهرْمتية المعممة المحددة، بالشكل
!> A*x=(lambda)*B*x,  A*Bx=(lambda)*x,  أو B*A*x=(lambda)*x. هنا A و
!> B يفترض أنهما هرْمتيان، مخزنتان بتنسيق مضغوط، و B أيضاً
!> موجبة محددة.
!> إذا كانت المتجهات الذاتية مطلوبة، فإنها تستخدم خوارزمية تقسيم الغزو.
!>
!>

المعاملات

ITYPE 
!>          ITYPE هو عدد صحيح (INTEGER)
!>          يحدد نوع المشكلة المراد حلها:
!>          = 1:  A*x = (lambda)*B*x
!>          = 2:  A*B*x = (lambda)*x
!>          = 3:  B*A*x = (lambda)*x
!>

JOBZ

!>          JOBZ هو حرف واحد (CHARACTER*1)
!>          = 'N':  حساب القيم الذاتية فقط؛
!>          = 'V':  حساب القيم الذاتية والمتجهات الذاتية.
!>

UPLO

!>          UPLO هو حرف واحد (CHARACTER*1)
!>          = 'U':  المثلثات العلوية لـ A و B مخزنة؛
!>          = 'L':  المثلثات السفلية لـ A و B مخزنة.
!>

N

!>          N هو عدد صحيح (INTEGER)
!>          ترتيب المصفوفات A و B.  N >= 0.
!>

AP

!>          AP هي مصفوفة مركبة (COMPLEX)، ذات أبعاد (N*(N+1)/2)
!>          عند الدخول، المثلث العلوي أو السفلي للمصفوفة الهرمتية
!>          A، مضغوطة بعمودية في مصفوفة خطية.  العمود j من A
!>          مخزن في المصفوفة AP كما يلي:
!>          إذا كان UPLO = 'U'، فإن AP(i + (j-1)*j/2) = A(i,j) لـ 1<=i<=j؛
!>          إذا كان UPLO = 'L'، فإن AP(i + (j-1)*(2*n-j)/2) = A(i,j) لـ j<=i<=n.
!>
!>          عند الخروج، يتم تدمير محتويات AP.
!>

BP

!>          BP هي مصفوفة مركبة (COMPLEX)، ذات أبعاد (N*(N+1)/2)
!>          عند الدخول، المثلث العلوي أو السفلي للمصفوفة الهرمتية
!>          B، مضغوطة بعمودية في مصفوفة خطية.  العمود j من B
!>          مخزن في المصفوفة BP كما يلي:
!>          إذا كان UPLO = 'U'، فإن BP(i + (j-1)*j/2) = B(i,j) لـ 1<=i<=j؛
!>          إذا كان UPLO = 'L'، فإن BP(i + (j-1)*(2*n-j)/2) = B(i,j) لـ j<=i<=n.
!>
!>          عند الخروج، العامل المثلثي U أو L من تفكيك Cholesky
!>          B = U**H*U أو B = L*L**H، بنفس تنسيق التخزين
!>          مثل B.
!>

W

!>          W هي مصفوفة حقيقية (REAL)، ذات أبعاد (N)
!>          إذا كان INFO = 0، فإن القيم الذاتية مرتبة تصاعديًا.
!>

Z

!>          Z هي مصفوفة مركبة (COMPLEX)، ذات أبعاد (LDZ, N)
!>          إذا كان JOBZ = 'V'، فعندما يكون INFO = 0، فإن Z تحتوي على المصفوفة Z للمتجهات الذاتية.
!>          المتجهات الذاتية معيارية كما يلي:
!>          إذا كان ITYPE = 1 أو 2، فإن Z**H*B*Z = I؛
!>          إذا كان ITYPE = 3، فإن Z**H*inv(B)*Z = I.
!>          إذا كان JOBZ = 'N'، فلن يتم الرجوع إلى Z.
!>

LDZ

!>          LDZ هو عدد صحيح (INTEGER)
!>          البعد الرائد للمصفوفة Z.  LDZ >= 1، وإذا
!>          كان JOBZ = 'V'، فإن LDZ >= max(1,N).
!>

WORK

!>          WORK هي مصفوفة مركبة (COMPLEX)، ذات أبعاد (MAX(1,LWORK))
!>          عند الخروج، إذا كان INFO = 0، فإن WORK(1) يعيد LWORK المطلوب.
!>

LWORK

!>          LWORK هو عدد صحيح (INTEGER)
!>          بعد المصفوفة WORK.
!>          إذا كان N <= 1،               LWORK >= 1.
!>          إذا كان JOBZ = 'N' و N > 1، LWORK >= N.
!>          إذا كان JOBZ = 'V' و N > 1، LWORK >= 2*N.
!>
!>          إذا كان LWORK = -1، فسيتم افتراض استعلام عن مساحة العمل؛
!>          تحسب الوحدة الفرعية فقط الأحجام المطلوبة للمصفوفات WORK و RWORK و
!>          IWORK، وتعيد هذه القيم كإدخالات أولى لـ
!>          المصفوفات WORK و RWORK و IWORK، ولا يصدر أي رسالة خطأ
!>          متعلقة بـ LWORK أو LRWORK أو LIWORK بواسطة XERBLA.
!>

RWORK

!>          RWORK هي مصفوفة حقيقية (REAL)، ذات أبعاد (MAX(1,LRWORK))
!>          عند الخروج، إذا كان INFO = 0، فإن RWORK(1) يعيد LRWORK المطلوب.
!>

LRWORK

!>          LRWORK هو عدد صحيح (INTEGER)
!>          بعد المصفوفة RWORK.
!>          إذا كان N <= 1،               LRWORK >= 1.
!>          إذا كان JOBZ = 'N' و N > 1، LRWORK >= N.
!>          إذا كان JOBZ = 'V' و N > 1، LRWORK >= 1 + 5*N + 2*N**2.
!>
!>          إذا كان LRWORK = -1، فسيتم افتراض استعلام عن مساحة العمل؛
!>          تحسب الوحدة الفرعية فقط الأحجام المطلوبة للمصفوفات WORK و RWORK و
!>          IWORK، وتعيد هذه القيم كإدخالات أولى لـ
!>          المصفوفات WORK و RWORK و IWORK، ولا يصدر أي رسالة خطأ
!>          متعلقة بـ LWORK أو LRWORK أو LIWORK بواسطة XERBLA.
!>

IWORK

!>          IWORK هي مصفوفة أعداد صحيحة (INTEGER)، ذات أبعاد (MAX(1,LIWORK))
!>          عند الخروج، إذا كان INFO = 0، فإن IWORK(1) يعيد LIWORK المطلوب.
!>

LIWORK

!>          LIWORK هو عدد صحيح (INTEGER)
!>          بعد المصفوفة IWORK.
!>          إذا كان JOBZ  = 'N' أو N <= 1، LIWORK >= 1.
!>          إذا كان JOBZ  = 'V' و N > 1، LIWORK >= 3 + 5*N.
!>
!>          إذا كان LIWORK = -1، فسيتم افتراض استعلام عن مساحة العمل؛
!>          تحسب الوحدة الفرعية فقط الأحجام المطلوبة للمصفوفات WORK و RWORK و
!>          IWORK، وتعيد هذه القيم كإدخالات أولى لـ
!>          المصفوفات WORK و RWORK و IWORK، ولا يصدر أي رسالة خطأ
!>          متعلقة بـ LWORK أو LRWORK أو LIWORK بواسطة XERBLA.
!>

INFO

!>          INFO هو عدد صحيح (INTEGER)
!>          = 0:  انتهاء ناجح
!>          < 0:  إذا كان INFO = -i، فإن المعامل i كان له قيمة غير قانونية
!>          > 0:  CPPTRF أو CHPEVD أعادت رمز خطأ:
!>             <= N:  إذا كان INFO = i، فإن CHPEVD فشلت في التقارب؛
!>                    i عناصر خارج القطر لمصفوفة ثلاثية وسيطة
!>                    لم تتقارب إلى الصفر؛
!>             > N:   إذا كان INFO = N + i، لـ 1 <= i <= n، فإن
!>                    المُضاعِف الرئيسي الرائد من الرتبة i للمصفوفة B ليس موجبًا.
!>                    لم يكن من الممكن إكمال تفكيك B ولم يتم حساب أي قيم ذاتية أو متجهات ذاتية.
!>
