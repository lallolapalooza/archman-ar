توثيق الوظيفة/الروتين الفرعي

subroutine chpt01 (character uplo, integer n, complex, dimension( * ) a, complex, dimension( * ) afac, integer, dimension( * ) ipiv, complex, dimension( ldc, * ) c, integer ldc, real, dimension( * ) rwork, real resid)

CHPT01

الغرض:

!>
!> CHPT01 يعيد بناء مصفوفة هرميتية غير محددة A من عامل L*D*L' أو U*D*U'
!> الخاص بها ويحسب الباقي
!>    norm( C - A ) / ( N * norm(A) * EPS ),
!> حيث C هي المصفوفة المعاد بناؤها، EPS هو خطأ الآلة،
!> L' هو المرافق التبادلي لـ L، و U' هو المرافق التبادلي لـ U.
!>

المعلمات

UPLO 
!>          UPLO is CHARACTER*1
!>          يحدد ما إذا كان الجزء المثلثي العلوي أو السفلي من
!>          المصفوفة الهرميتية A مخزنًا:
!>          = 'U':  مثلثي علوي
!>          = 'L':  مثلثي سفلي
!>

N

!>          N is INTEGER
!>          عدد الصفوف والأعمدة للمصفوفة A.  N >= 0.
!>

A

!>          A is COMPLEX array, dimension (N*(N+1)/2)
!>          المصفوفة الهرميتية الأصلية A، مخزنة كمصفوفة
!>          مثلثية مكدسة.
!>

AFAC

!>          AFAC is COMPLEX array, dimension (N*(N+1)/2)
!>          الشكل المفكوك للمصفوفة A، مخزن كمصفوفة
!>          مثلثية مكدسة.  تحتوي AFAC على المصفوفة القطرية الكتلة D
!>          والمضاعفات المستخدمة للحصول على العامل L أو U من
!>          عامل L*D*L' أو U*D*U' كما تم حسابه بواسطة CHPTRF.
!>

IPIV

!>          IPIV is INTEGER array, dimension (N)
!>          مؤشرات المحور من CHPTRF.
!>

C

!>          C is COMPLEX array, dimension (LDC,N)
!>

LDC

!>          LDC is INTEGER
!>          البعد الرئيسي للمصفوفة C.  LDC >= max(1,N).
!>

RWORK

!>          RWORK is REAL array, dimension (N)
!>

RESID

!>          RESID is REAL
!>          If UPLO = 'L', norm(L*D*L' - A) / ( N * norm(A) * EPS )
!>          If UPLO = 'U', norm(U*D*U' - A) / ( N * norm(A) * EPS )
!>
