subroutine chetrs_rook (uplo, n, nrhs, a, lda, ipiv, b, ldb, info)
CHETRS_ROOK تحسب الحل لنظام المعادلات الخطية A * X = B لمصفوفات HE باستخدام التحلل الناتج عن إحدى طرق المحورية القطرية المحدودة (تبديلات بحد أقصى 2)

توثيق الوظيفة/الإجراء الفرعي

subroutine chetrs_rook (character uplo, integer n, integer nrhs, complex, dimension( lda, * ) a, integer lda, integer, dimension( * ) ipiv, complex, dimension( ldb, * ) b, integer ldb, integer info)

CHETRS_ROOK تحسب الحل لنظام المعادلات الخطية A * X = B لمصفوفات HE باستخدام التحلل الناتج عن إحدى طرق المحورية القطرية المحدودة (تبديلات بحد أقصى 2)

الغرض:

!>
!> CHETRS_ROOK يحل نظام المعادلات الخطية A*X = B لمصفوفة هرميتية معقدة A باستخدام التحلل A = U*D*U**H أو
!> A = L*D*L**H الذي تم حسابه بواسطة CHETRF_ROOK.
!>

المعلمات

UPLO 
!>          UPLO هو CHARACTER*1
!>          يحدد ما إذا كانت تفاصيل التحلل مخزنة
!>          كمصفوفة مثلثية علوية أو سفلية.
!>          = 'U':  مثلثية علوية، الشكل هو A = U*D*U**H;
!>          = 'L':  مثلثية سفلية، الشكل هو A = L*D*L**H.
!>

N

!>          N هو INTEGER
!>          رتبة المصفوفة A.  N >= 0.
!>

NRHS

!>          NRHS هو INTEGER
!>          عدد الأطراف اليمنى، أي عدد أعمدة
!>          المصفوفة B.  NRHS >= 0.
!>

A

!>          A هو مصفوفة COMPLEX، الأبعاد (LDA,N)
!>          المصفوفة القطرية المتصلة D والمعاملات المستخدمة
!>          للحصول على العامل U أو L كما تم حسابه بواسطة CHETRF_ROOK.
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرئيسي للمصفوفة A.  LDA >= max(1,N).
!>

IPIV

!>          IPIV هو مصفوفة INTEGER، الأبعاد (N)
!>          تفاصيل التبادلات وهيكل D
!>          كما تم تحديده بواسطة CHETRF_ROOK.
!>

B

!>          B هو مصفوفة COMPLEX، الأبعاد (LDB,NRHS)
!>          عند الدخول، مصفوفة الطرف الأيمن B.
!>          عند الخروج، مصفوفة الحل X.
!>

LDB

!>          LDB هو INTEGER
!>          البعد الرئيسي للمصفوفة B.  LDB >= max(1,N).
!>

INFO

!>          INFO هو INTEGER
!>          = 0:  انتهاء ناجح
!>          < 0:  إذا كان INFO = -i، فإن المعلمة i كان لها قيمة غير قانونية
!>
