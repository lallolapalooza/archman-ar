subroutine chet22 (integer itype, character uplo, integer n, integer m, integer kband, complex, dimension( lda, * ) a, integer lda, real, dimension( * ) d, real, dimension( * ) e, complex, dimension( ldu, * ) u, integer ldu, complex, dimension( ldv, * ) v, integer ldv, complex, dimension( * ) tau, complex, dimension( * ) work, real, dimension( * ) rwork, real, dimension( 2 ) result)

CHET22

الغرض:

!>
!>      CHET22  بشكل عام يتحقق من تفكيك للشكل
!>
!>              A U = U S
!>
!>      حيث A هرمية مركبة، أعمدة U متعامدة،
!>      و S قطرية (إذا كان KBAND=0) أو ثلاثية متناظرة (إذا كان
!>      KBAND=1). إذا كان ITYPE=1، فإن U ممثلة كمصفوفة كثيفة،
!>      وإلا فإن U معبر عنها كناتج تحويلات هاوسهولدر، والمتجهات
!>      مخزنة في المصفوفة  ، والثوابت القياسية فيها.
!>      سنستخدم الحرف  للإشارة إلى ناتج تحويلات هاوسهولدر
!>      (والتي يجب أن تكون مساوية لـ U).
!>
!>      على وجه التحديد، إذا كان ITYPE=1، فإن:
!>
!>              RESULT(1) = | U**H A U - S | / ( |A| m ulp ) و
!>              RESULT(2) = | I - U**H U | / ( m ulp )
!> 
!>  ITYPE   INTEGER
!>          يحدد نوع الاختبارات التي سيتم إجراؤها.
!>          1: U ممثلة كمصفوفة متعامدة كثيفة:
!>             RESULT(1) = | A - U S U**H | / ( |A| n ulp ) و
!>             RESULT(2) = | I - U U**H | / ( n ulp )
!>
!>  UPLO    CHARACTER
!>          إذا كان UPLO='U'، سيتم استخدام المثلث العلوي من A ولن
!>          يتم الإشارة إلى المثلث السفلي (تمامًا). إذا كان
!>          UPLO='L'، سيتم استخدام المثلث السفلي من A ولن
!>          يتم الإشارة إلى المثلث العلوي (تمامًا).
!>          لا يتم تعديله.
!>
!>  N       INTEGER
!>          حجم المصفوفة. إذا كان صفرًا، فإن CHET22 لا تفعل شيئًا.
!>          يجب أن يكون أكبر من أو يساوي صفرًا.
!>          لا يتم تعديله.
!>
!>  M       INTEGER
!>          عدد أعمدة U. إذا كان صفرًا، فإن CHET22 لا تفعل شيئًا.
!>          يجب أن يكون أكبر من أو يساوي صفرًا.
!>          لا يتم تعديله.
!>
!>  KBAND   INTEGER
!>          عرض النطاق للمصفوفة. قد يكون صفرًا أو واحدًا فقط.
!>          إذا كان صفرًا، فإن S قطرية، ولا يتم الإشارة إلى E. إذا كان
!>          واحدًا، فإن S ثلاثية متناظرة.
!>          لا يتم تعديله.
!>
!>  A       مصفوفة COMPLEX، ذات بعد (LDA , N)
!>          المصفوفة الأصلية (غير المفككة). يُفترض أنها
!>          متناظرة، وسيتم الإشارة فقط إلى الجزء العلوي (UPLO='U') أو
!>          الجزء السفلي (UPLO='L') فقط.
!>          لا يتم تعديله.
!>
!>  LDA     INTEGER
!>          البعد الرائد لـ A. يجب أن يكون على الأقل 1
!>          وعلى الأقل N.
!>          لا يتم تعديله.
!>
!>  D       مصفوفة REAL، ذات بعد (N)
!>          القطر للمصفوفة (ثلاثية متناظرة) القطرية.
!>          لا يتم تعديله.
!>
!>  E       مصفوفة REAL، ذات بعد (N)
!>          العناصر خارج القطر للمصفوفة (ثلاثية متناظرة) القطرية.
!>          يتم تجاهل E(1)، E(2) هي العنصر (1,2) و (2,1)، إلخ.
!>          لا يتم الإشارة إليها إذا كان KBAND=0.
!>          لا يتم تعديله.
!>
!>  U       مصفوفة COMPLEX، ذات بعد (LDU, N)
!>          إذا كان ITYPE=1، فإن هذه تحتوي على المصفوفة المتعامدة في
!>          التفكيك، معبرًا عنها كمصفوفة كثيفة.
!>          لا يتم تعديله.
!>
!>  LDU     INTEGER
!>          البعد الرائد لـ U. يجب أن يكون LDU على الأقل N و
!>          على الأقل 1.
!>          لا يتم تعديله.
!>
!>  V       مصفوفة COMPLEX، ذات بعد (LDV, N)
!>          إذا كان ITYPE=2 أو 3، فإن المثلث السفلي لهذه المصفوفة يحتوي
!>          على متجهات هاوسهولدر التي تصف المصفوفة المتعامدة في
!>          التفكيك. إذا كان ITYPE=1، فلن يتم الإشارة إليها.
!>          لا يتم تعديله.
!>
!>  LDV     INTEGER
!>          البعد الرائد لـ V. يجب أن يكون LDV على الأقل N و
!>          على الأقل 1.
!>          لا يتم تعديله.
!>
!>  TAU     مصفوفة COMPLEX، ذات بعد (N)
!>          إذا كان ITYPE >= 2، فإن TAU(j) هو العامل القياسي لـ
!>          v(j) v(j)**H في تحويل هاوسهولدر H(j) لـ
!>          الناتج  U = H(1)...H(n-2)
!>          إذا كان ITYPE < 2، فلن يتم الإشارة إلى TAU.
!>          لا يتم تعديله.
!>
!>  WORK    مصفوفة COMPLEX، ذات بعد (2*N**2)
!>          مساحة العمل.
!>          تم التعديل.
!>
!>  RWORK   مصفوفة REAL، ذات بعد (N)
!>          مساحة العمل.
!>          تم التعديل.
!>
!>  RESULT  مصفوفة REAL، ذات بعد (2)
!>          القيم المحسوبة من خلال الاختبارين الموصوفين أعلاه.
!>          القيم محدودة حاليًا بـ 1/ulp، لتجنب تجاوز السعة.
!>          يتم تعديل RESULT(1) دائمًا. يتم تعديل RESULT(2) فقط
!>          إذا كان LDU على الأقل N.
!>          تم التعديل.
!>
