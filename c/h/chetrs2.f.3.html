توثيق الوظيفة/الإجراء الفرعي

subroutine chetrs2 (character uplo, integer n, integer nrhs, complex, dimension( lda, * ) a, integer lda, integer, dimension( * ) ipiv, complex, dimension( ldb, * ) b, integer ldb, complex, dimension( * ) work, integer info)

CHETRS2

الغرض:

!>
!> CHETRS2 تحل نظام المعادلات الخطية A*X = B بمصفوفة هرميتية معقدة A
!> باستخدام التحليل A = U*D*U**H أو A = L*D*L**H المحسوب بواسطة CHETRF
!> والمحول بواسطة CSYCONV.
!>

المعلمات

UPLO 
!>          UPLO هو CHARACTER*1
!>          يحدد ما إذا كانت تفاصيل التحليل مخزنة
!>          كمصفوفة مثلثة عليا أو سفلى.
!>          = 'U':  مثلثة عليا، الشكل هو A = U*D*U**H؛
!>          = 'L':  مثلثة سفلى، الشكل هو A = L*D*L**H.
!>

N

!>          N هو INTEGER
!>          رتبة المصفوفة A.  N >= 0.
!>

NRHS

!>          NRHS هو INTEGER
!>          عدد الأطراف اليمنى، أي عدد أعمدة
!>          المصفوفة B.  NRHS >= 0.
!>

A

!>          A هي مصفوفة COMPLEX، الأبعاد (LDA,N)
!>          المصفوفة القطرية الكتلية D والعوامل المستخدمة
!>          للحصول على العامل U أو L كما هو محسوب بواسطة CHETRF.
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرائد للمصفوفة A.  LDA >= max(1,N).
!>

IPIV

!>          IPIV هي مصفوفة INTEGER، الأبعاد (N)
!>          تفاصيل التبادلات والتركيب الكتلية لـ D
!>          كما تم تحديدها بواسطة CHETRF.
!>

B

!>          B هي مصفوفة COMPLEX، الأبعاد (LDB,NRHS)
!>          عند الإدخال، مصفوفة الطرف الأيمن B.
!>          عند الخروج، مصفوفة الحل X.
!>

LDB

!>          LDB هو INTEGER
!>          البعد الرائد للمصفوفة B.  LDB >= max(1,N).
!>

WORK

!>          WORK هي مصفوفة COMPLEX، الأبعاد (N)
!>

INFO

!>          INFO هو INTEGER
!>          = 0:  الخروج بنجاح
!>          < 0:  إذا كان INFO = -i، فقد كان الوسيط i له قيمة غير قانونية
!>
