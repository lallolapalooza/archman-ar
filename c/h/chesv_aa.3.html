subroutine chesv_aa (character uplo, integer n, integer nrhs, complex, dimension( lda, * ) a, integer lda, integer, dimension( * ) ipiv, complex, dimension( ldb, * ) b, integer ldb, complex, dimension( * ) work, integer lwork, integer info)

CHESV_AA يحسب حل نظام المعادلات الخطية A * X = B لمصفوفات HE

الغرض:

!>
!> CHESV_AA يحسب حل نظام المعادلات الخطية المعقدة
!>    A * X = B,
!> حيث A مصفوفة هيرميتية N × N و X و B مصفوفتان N × NRHS.
!>
!> يتم استخدام خوارزمية Aasen لتفكيك A إلى
!>    A = U**H * T * U,  إذا كان UPLO = 'U', أو
!>    A = L * T * L**H,  إذا كان UPLO = 'L',
!> حيث U (أو L) هو حاصل ضرب مصفوفات تبادلية ومثلثية علوية (سفلية) وحدوية،
!> و T مصفوفة هيرميتية ثلاثية الأقطار. ثم يتم استخدام الشكل المفكك لـ A
!> لحل نظام المعادلات A * X = B.
!>

المعلمات

UPLO 
!>          UPLO هو CHARACTER*1
!>          = 'U':  يتم تخزين الجزء العلوي من A؛
!>          = 'L':  يتم تخزين الجزء السفلي من A.
!>

N

!>          N هو INTEGER
!>          عدد المعادلات الخطية، أي رتبة المصفوفة A. N >= 0.
!>

NRHS

!>          NRHS هو INTEGER
!>          عدد الأطراف اليمنى، أي عدد أعمدة المصفوفة B. NRHS >= 0.
!>

A

!>          A هي مصفوفة COMPLEX، البعد (LDA,N)
!>          عند الدخول، المصفوفة الهيرميتية A. إذا كان UPLO = 'U'، فإن الجزء
!>          المثلثي العلوي N × N الرائد في A يحتوي على الجزء المثلثي العلوي
!>          للمصفوفة A، ولا يتم الرجوع إلى الجزء المثلثي السفلي الصارم لـ A.
!>          إذا كان UPLO = 'L'، فإن الجزء المثلثي السفلي N × N الرائد في A يحتوي
!>          على الجزء المثلثي السفلي للمصفوفة A، ولا يتم الرجوع إلى الجزء المثلثي
!>          العلوي الصارم لـ A.
!>
!>          عند الخروج، إذا كان INFO = 0، فإن المصفوفة ثلاثية الأقطار T
!>          والمضاعفات المستخدمة للحصول على العامل U أو L من التفكيك
!>          A = U**H*T*U أو A = L*T*L**H كما تم حسابه بواسطة CHETRF_AA.
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرائد للمصفوفة A. LDA >= max(1,N).
!>

IPIV

!>          IPIV هي مصفوفة INTEGER، البعد (N)
!>          عند الخروج، تحتوي على تفاصيل التبادلات، أي أنه تم تبادل الصف والعمود k
!>          من A مع الصف والعمود IPIV(k).
!>

B

!>          B هي مصفوفة COMPLEX، البعد (LDB,NRHS)
!>          عند الدخول، المصفوفة الطرفية اليمنى B بحجم N × NRHS.
!>          عند الخروج، إذا كان INFO = 0، فإن مصفوفة الحل X بحجم N × NRHS.
!>

LDB

!>          LDB هو INTEGER
!>          البعد الرائد للمصفوفة B. LDB >= max(1,N).
!>

WORK

!>          WORK هي مصفوفة COMPLEX، البعد (MAX(1,LWORK))
!>          عند الخروج، إذا كان INFO = 0، فإن WORK(1) يعيد LWORK الأمثل.
!>

LWORK

!>          LWORK هو INTEGER
!>          طول WORK. LWORK >= MAX(1,2*N,3*N-2)، وللحصول على أفضل أداء
!>          LWORK >= MAX(1,N*NB)، حيث NB هو حجم الكتلة الأمثل لـ CHETRF.
!>
!>          إذا كان LWORK = -1، فسيتم افتراض استعلام مساحة العمل؛ يقوم الروتين
!>          بحساب حجم مصفوفة WORK الأمثل فقط، ويعيد هذا الحجم كالإدخال الأول
!>          لمصفوفة WORK، ولا يصدر XERBLA أي رسالة خطأ متعلقة بـ LWORK.
!>

INFO

!>          INFO هو INTEGER
!>          = 0: خروج ناجح
!>          < 0: إذا كان INFO = -i، فإن الوسيط i كان له قيمة غير قانونية
!>          > 0: إذا كان INFO = i، فإن D(i,i) تساوي صفرًا بالضبط. تم إكمال
!>               التفكيك، ولكن المصفوفة القطرية الدقيقة D فردية تمامًا،
!>               لذا لم يكن من الممكن حساب الحل.
!>
