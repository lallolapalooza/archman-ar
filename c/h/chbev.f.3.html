subroutine chbev (jobz, uplo, n, kd, ab, ldab, w, z, ldz, work, rwork, info)
CHBEV تحسب القيم الذاتية، اختياريًا، المتجهات الذاتية اليسرى و/أو اليمنى لمصفوفات أخرى

توثيق الوظيفة/الروتين الفرعي

subroutine chbev (character jobz, character uplo, integer n, integer kd, complex, dimension( ldab, * ) ab, integer ldab, real, dimension( * ) w, complex, dimension( ldz, * ) z, integer ldz, complex, dimension( * ) work, real, dimension( * ) rwork, integer info)

CHBEV تحسب جميع القيم الذاتية، اختياريًا، والمتجهات الذاتية لمصفوفة نطاق هرميتية معقدة A.

الغرض:

!>
!> CHBEV تحسب جميع القيم الذاتية، اختياريًا، والمتجهات الذاتية لمصفوفة نطاق هرميتية معقدة A.
!>

المعاملات

JOBZ 
!>          JOBZ هو CHARACTER*1
!>          = 'N':  حساب القيم الذاتية فقط؛
!>          = 'V':  حساب القيم الذاتية والمتجهات الذاتية.
!>

UPLO

!>          UPLO هو CHARACTER*1
!>          = 'U':  يتم تخزين المثلث العلوي لـ A؛
!>          = 'L':  يتم تخزين المثلث السفلي لـ A.
!>

N

!>          N هو INTEGER
!>          ترتيب المصفوفة A.  N >= 0.
!>

KD

!>          KD هو INTEGER
!>          عدد الأقطار الفائقة للمصفوفة A إذا كان UPLO = 'U'،
!>          أو عدد الأقطار الفرعية إذا كان UPLO = 'L'.  KD >= 0.
!>

AB

!>          AB هو مصفوفة COMPLEX، البعد (LDAB, N)
!>          عند الإدخال، المثلث العلوي أو السفلي للمصفوفة النطاقية الهرميتية
!>          A، مخزنة في أول KD+1 صفوف من المصفوفة.  العمود j من A
!>          مخزن في العمود j من المصفوفة AB
!>          كما يلي:
!>          إذا كان UPLO = 'U'، فإن AB(kd+1+i-j,j) = A(i,j) لـ max(1,j-kd)<=i<=j؛
!>          إذا كان UPLO = 'L'، فإن AB(1+i-j,j)    = A(i,j) لـ j<=i<=min(n,j+kd).
!>
!>          عند الخروج، يتم الكتابة فوق AB بالقيم التي تم إنشاؤها أثناء
!>          الاختزال إلى مصفوفة ثلاثية الأقطار.  إذا كان UPLO = 'U'، فإن القطر الفائق الأول
!>          والقطر الرئيسي للمصفوفة ثلاثية الأقطار T
!>          يتم إرجاعهما في الصفين KD و KD+1 من AB، وإذا كان UPLO = 'L'،
!>          يتم إرجاع القطر الرئيسي والقطر الفرعي الأول لـ T في
!>          الصفين الأولين من AB.
!>

LDAB

!>          LDAB هو INTEGER
!>          البعد الرئيسي للمصفوفة AB.  LDAB >= KD + 1.
!>

W

!>          W هي مصفوفة REAL، البعد (N)
!>          إذا كان INFO = 0، فإن القيم الذاتية مرتبة تصاعديًا.
!>

Z

!>          Z هي مصفوفة COMPLEX، البعد (LDZ, N)
!>          إذا كان JOBZ = 'V'، فعندما يكون INFO = 0، تحتوي Z على المتجهات الذاتية
!>          المتعامدة للمصفوفة A، حيث يحمل العمود i من Z
!>          المتجه الذاتي المرتبط بـ W(i).
!>          إذا كان JOBZ = 'N'، فلا يتم الرجوع إلى Z.
!>

LDZ

!>          LDZ هو INTEGER
!>          البعد الرئيسي للمصفوفة Z.  LDZ >= 1، وإذا كان
!>          JOBZ = 'V'، فإن LDZ >= max(1,N).
!>

WORK

!>          WORK هي مصفوفة COMPLEX، البعد (N)
!>

RWORK

!>          RWORK هي مصفوفة REAL، البعد (max(1,3*N-2))
!>

INFO

!>          INFO هو INTEGER
!>          = 0:  انتهاء ناجح.
!>          < 0:  إذا كان INFO = -i، فإن المعامل i كان له قيمة غير قانونية.
!>          > 0:  إذا كان INFO = i، فشل الخوارزمية في التقارب؛ i
!>                العناصر غير القطرية لمصفوفة ثلاثية الأقطار وسيطة
!>                لم تتقارب إلى الصفر.
!>
