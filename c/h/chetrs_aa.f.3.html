توثيق الوظيفة/الإجراء الفرعي

subroutine chetrs_aa (character uplo, integer n, integer nrhs, complex, dimension( lda, * ) a, integer lda, integer, dimension( * ) ipiv, complex, dimension( ldb, * ) b, integer ldb, complex, dimension( * ) work, integer lwork, integer info)

CHETRS_AA

الغرض:

!>
!> CHETRS_AA يحل نظام المعادلات الخطية A*X = B مع مصفوفة هيرميتية معقدة A باستخدام التحليل A = U**H*T*U أو
!> A = L*T*L**H المحسوب بواسطة CHETRF_AA.
!>

المعلمات

UPLO 
!>          UPLO is CHARACTER*1
!>          يحدد ما إذا كانت تفاصيل التحليل مخزنة
!>          كمصفوفة مثلثية عليا أو سفلية.
!>          = 'U':  مثلثية عليا، الشكل هو A = U**H*T*U؛
!>          = 'L':  مثلثية سفلية، الشكل هو A = L*T*L**H.
!>

N

!>          N is INTEGER
!>          رتبة المصفوفة A.  N >= 0.
!>

NRHS

!>          NRHS is INTEGER
!>          عدد الأطراف اليمنى، أي عدد أعمدة
!>          المصفوفة B.  NRHS >= 0.
!>

A

!>          A is COMPLEX array, dimension (LDA,N)
!>          تفاصيل العوامل المحسوبة بواسطة CHETRF_AA.
!>

LDA

!>          LDA is INTEGER
!>          البعد الرئيسي للمصفوفة A.  LDA >= max(1,N).
!>

IPIV

!>          IPIV is INTEGER array, dimension (N)
!>          تفاصيل التبادلات المحسوبة بواسطة CHETRF_AA.
!>

B

!>          B is COMPLEX array, dimension (LDB,NRHS)
!>          عند الإدخال، مصفوفة الطرف الأيمن B.
!>          عند الخروج، مصفوفة الحل X.
!>

LDB

!>          LDB is INTEGER
!>          البعد الرئيسي للمصفوفة B.  LDB >= max(1,N).
!>

WORK

!>          WORK is COMPLEX array, dimension (MAX(1,LWORK))
!>

LWORK

!>          LWORK is INTEGER
!>          بعد المصفوفة WORK.
!>          إذا كان MIN(N,NRHS) = 0، فإن LWORK >= 1، وإلا فإن LWORK >= 3*N-2.
!>
!>          إذا كان LWORK = -1، فيفترض استعلام عن مساحة عمل؛ تقوم الإجراءات الفرعية
!>          فقط بحساب الحد الأدنى لحجم مصفوفة WORK، وتعيد
!>          هذه القيمة كأول إدخال لمصفوفة WORK، ولا يصدر أي رسالة خطأ
!>          متعلقة بـ LWORK بواسطة XERBLA.
!>

INFO

!>          INFO is INTEGER
!>          = 0:  الخروج الناجح
!>          < 0:  إذا كان INFO = -i، فإن وسيطة i كانت قيمة غير قانونية
!>
