توثيق الوظيفة/الروتين الفرعي

subroutine chetrf_aa (character uplo, integer n, complex, dimension( lda, * ) a, integer lda, integer, dimension( * ) ipiv, complex, dimension( * ) work, integer lwork, integer info)

CHETRF_AA

الغرض:

!>
!> CHETRF_AA يحسب تحليل مصفوفة هيرميتية مركبة A
!> باستخدام خوارزمية Aasen. شكل التحليل هو
!>
!>    A = U**H*T*U  أو  A = L*T*L**H
!>
!> حيث U (أو L) هو ناتج تباديل ومصفوفات ثلاثية علوية (سفلية) وحدية.
!> و T هي مصفوفة ثلاثية هيرميتية.
!>
!> هذه هي النسخة المجمعة من الخوارزمية، تستدعي المستوى 3 BLAS.
!>

المعلمات

UPLO 
!>          UPLO هو CHARACTER*1
!>          = 'U':  يتم تخزين المثلث العلوي لـ A؛
!>          = 'L':  يتم تخزين المثلث السفلي لـ A.
!>

N

!>          N هو INTEGER
!>          رتبة المصفوفة A.  N >= 0.
!>

A

!>          A هي مصفوفة COMPLEX، البعد (LDA,N)
!>          عند الدخول، المصفوفة الهيرميتية A.  إذا كان UPLO = 'U'، فإن
!>          الجزء المثلثي العلوي N × N الرائد من A يحتوي على الجزء
!>          المثلثي العلوي للمصفوفة A، والجزء المثلثي السفلي الصارم
!>          لـ A لا يتم الرجوع إليه.  إذا كان UPLO = 'L'، فإن
!>          الجزء المثلثي السفلي N × N الرائد من A يحتوي على الجزء
!>          المثلثي السفلي للمصفوفة A، والجزء المثلثي العلوي الصارم
!>          لـ A لا يتم الرجوع إليه.
!>
!>          عند الخروج، يتم تخزين المصفوفة الثلاثية في الأقطار
!>          والأقطار الفرعية لـ A أسفل (أو أعلى) الأقطار مباشرةً،
!>          ويتم تخزين L أسفل (أو أعلى) الأقطار الفرعية، عندما يكون UPLO
!>          هو 'L' (أو 'U').
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرائد للمصفوفة A.  LDA >= max(1,N).
!>

IPIV

!>          IPIV هي مصفوفة INTEGER، البعد (N)
!>          عند الخروج، تحتوي على تفاصيل التبادلات، أي
!>          تم تبديل الصف والعمود k من A مع
!>          الصف والعمود IPIV(k).
!>

WORK

!>          WORK هي مصفوفة COMPLEX، البعد (MAX(1,LWORK))
!>          عند الخروج، إذا كان INFO = 0، فإن WORK(1) تعيد LWORK الأمثل.
!>

LWORK

!>          LWORK هو INTEGER
!>          طول WORK.
!>          LWORK >= 1، إذا كان N <= 1، و LWORK >= 2*N، وإلا.
!>          للحصول على الأداء الأمثل LWORK >= N*(1+NB)، حيث NB هو
!>          حجم الكتلة الأمثل، الذي تم إرجاعه بواسطة ILAENV.
!>
!>          إذا كان LWORK = -1، فمن المفترض وجود استعلام مساحة عمل؛ الروتين
!>          يحسب فقط الحجم الأمثل لمصفوفة WORK، ويعيد
!>          هذه القيمة كالإدخال الأول لمصفوفة WORK، ولا يتم إصدار أي رسالة خطأ
!>          متعلقة بـ LWORK بواسطة XERBLA.
!>

INFO

!>          INFO هو INTEGER
!>          = 0:  خروج ناجح
!>          < 0:  إذا كان INFO = -i، فإن وسيطة i كانت قيمة غير قانونية.
!>
