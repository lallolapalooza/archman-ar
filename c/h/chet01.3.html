subroutine chet01 (character uplo, integer n, complex, dimension( lda, * ) a, integer lda, complex, dimension( ldafac, * ) afac, integer ldafac, integer, dimension( * ) ipiv, complex, dimension( ldc, * ) c, integer ldc, real, dimension( * ) rwork, real resid)

CHET01

الغرض:

!>
!> CHET01 يعيد بناء مصفوفة هرميتية غير محددة A من تحليلها
!> L*D*L' أو U*D*U' الكتلوي ويحسب معيار الخطأ
!>    norm( C - A ) / ( N * norm(A) * EPS ),
!> حيث C هي المصفوفة المعاد بناؤها، EPS هو خطأ الآلة،
!> L' هو المنقول المقترن لـ L، و U' هو المنقول المقترن لـ U.
!>

المعاملات

UPLO 
!>          UPLO هو CHARACTER*1
!>          يحدد ما إذا كان الجزء المثلثي العلوي أو السفلي للمصفوفة
!>          الهرميتية A مخزنًا:
!>          = 'U':  مثلثي علوي
!>          = 'L':  مثلثي سفلي
!>

N

!>          N هو INTEGER
!>          عدد الصفوف والأعمدة للمصفوفة A. N >= 0.
!>

A

!>          A هي مصفوفة COMPLEX، حجم (LDA,N)
!>          المصفوفة الهرميتية الأصلية A.
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرائد للمصفوفة A. LDA >= max(1,N)
!>

AFAC

!>          AFAC هي مصفوفة COMPLEX، حجم (LDAFAC,N)
!>          الشكل المحلل للمصفوفة A. AFAC يحتوي على المصفوفة القطرية
!>          الكتلوية D والمعاملات المستخدمة للحصول على العامل L أو U
!>          من تحليل L*D*L' أو U*D*U' الكتلوية كما تم حسابه بواسطة
!>          CHETRF.
!>

LDAFAC

!>          LDAFAC هو INTEGER
!>          البعد الرائد للمصفوفة AFAC. LDAFAC >= max(1,N).
!>

IPIV

!>          IPIV هي مصفوفة INTEGER، حجم (N)
!>          فهارس المحور من CHETRF.
!>

C

!>          C هي مصفوفة COMPLEX، حجم (LDC,N)
!>

LDC

!>          LDC هو INTEGER
!>          البعد الرائد للمصفوفة C. LDC >= max(1,N).
!>

RWORK

!>          RWORK هي مصفوفة REAL، حجم (N)
!>

RESID

!>          RESID هو REAL
!>          إذا كان UPLO = 'L'، norm(L*D*L' - A) / ( N * norm(A) * EPS )
!>          إذا كان UPLO = 'U'، norm(U*D*U' - A) / ( N * norm(A) * EPS )
!>
