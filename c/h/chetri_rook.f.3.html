subroutine chetri_rook (uplo, n, a, lda, ipiv, work, info)
CHETRI_ROOK يحسب معكوس المصفوفة HE باستخدام التفكيك الذي تم الحصول عليه بطريقة ركائز التماثل Bunch-Kaufman المحدودة (rook").

توثيق الوظيفة/الإجراء الفرعي

subroutine chetri_rook (character uplo, integer n, complex, dimension( lda, * ) a, integer lda, integer, dimension( * ) ipiv, complex, dimension( * ) work, integer info)

CHETRI_ROOK يحسب معكوس المصفوفة الهرميتية غير المحددة HE باستخدام التفكيك الذي تم الحصول عليه بطريقة ركائز التماثل Bunch-Kaufman المحدودة ("rook").

الغرض:

!>
!> CHETRI_ROOK يحسب معكوس مصفوفة هرميتية مركبة غير محدودة
!> A باستخدام التفكيك A = U*D*U**H أو A = L*D*L**H الذي تم حسابه بواسطة
!> CHETRF_ROOK.
!>

المعاملات

UPLO 
!>          UPLO is CHARACTER*1
!>          يحدد ما إذا كانت تفاصيل التفكيك مخزنة
!>          كمصفوفة مثلثة علوية أو سفلية.
!>          = 'U':  مثلثة علوية، الشكل هو A = U*D*U**H؛
!>          = 'L':  مثلثة سفلية، الشكل هو A = L*D*L**H.
!>

N

!>          N is INTEGER
!>          رتبة المصفوفة A. N >= 0.
!>

A

!>          A is COMPLEX array, dimension (LDA,N)
!>          عند الإدخال، المصفوفة القطرية الفرعية D والمضاعفات
!>          المستخدمة للحصول على العامل U أو L كما تم حسابه بواسطة CHETRF_ROOK.
!>
!>          عند الخروج، إذا كان INFO = 0، معكوس المصفوفة الأصلية
!>          (الهرميتية). إذا كان UPLO = 'U'، يتم تشكيل الجزء المثلثي العلوي
!>          من المعكوس ولا يتم الرجوع إلى الجزء السفلي من A؛ إذا كان UPLO = 'L'
!>          يتم تشكيل الجزء المثلثي السفلي من المعكوس ولا يتم الرجوع إلى الجزء
!>          العلوي من A.
!>

LDA

!>          LDA is INTEGER
!>          البعد الرائد للمصفوفة A. LDA >= max(1,N).
!>

IPIV

!>          IPIV is INTEGER array, dimension (N)
!>          تفاصيل التبادلات وهيكل R الخاص بـ D
!>          كما تم تحديده بواسطة CHETRF_ROOK.
!>

WORK

!>          WORK is COMPLEX array, dimension (N)
!>

INFO

!>          INFO is INTEGER
!>          = 0: خروج ناجح
!>          < 0: إذا كان INFO = -i، فإن المعامل i كان له قيمة غير قانونية
!>          > 0: إذا كان INFO = i، فإن D(i,i) = 0؛ المصفوفة شاذة ولا يمكن
!>               حساب معكوسها.
!>
