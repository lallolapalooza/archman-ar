subroutine chprfs (character uplo, integer n, integer nrhs, complex, dimension( * ) ap, complex, dimension( * ) afp, integer, dimension( * ) ipiv, complex, dimension( ldb, * ) b, integer ldb, complex, dimension( ldx, * ) x, integer ldx, real, dimension( * ) ferr, real, dimension( * ) berr, complex, dimension( * ) work, real, dimension( * ) rwork, integer info)

CHPRFS

الغرض:

!>
!> CHPRFS يحسّن الحل المحسوب لنظام المعادلات الخطية
!> عندما تكون مصفوفة المعاملات هرميتية غير محددة
!> ومعبأة، ويوفر حدود الأخطاء وتقديرات الخطأ الخلفي
!> للحل.
!>

المعاملات

UPLO 
!>          UPLO is CHARACTER*1
!>          = 'U':  المثلث العلوي من A مخزن؛
!>          = 'L':  المثلث السفلي من A مخزن.
!>

N

!>          N is INTEGER
!>          رتبة المصفوفة A.  N >= 0.
!>

NRHS

!>          NRHS is INTEGER
!>          عدد الأطراف اليمنى، أي عدد أعمدة
!>          المصفوفتين B و X.  NRHS >= 0.
!>

AP

!>          AP is COMPLEX array, dimension (N*(N+1)/2)
!>          المثلث العلوي أو السفلي من المصفوفة الهرميتية A، معبأ
!>          عمودياً في مصفوفة خطية.  يتم تخزين العمود j من A
!>          في المصفوفة AP كما يلي:
!>          إذا كان UPLO = 'U'، فإن AP(i + (j-1)*j/2) = A(i,j) حيث 1<=i<=j؛
!>          إذا كان UPLO = 'L'، فإن AP(i + (j-1)*(2*n-j)/2) = A(i,j) حيث j<=i<=n.
!>

AFP

!>          AFP is COMPLEX array, dimension (N*(N+1)/2)
!>          الشكل المتبع للمصفوفة A.  يحتوي AFP على القطرية
!>          المتعددة D والمضاعفات المستخدمة للحصول على
!>          U أو L من التفكيك A = U*D*U**H أو
!>          A = L*D*L**H كما تم حسابه بواسطة CHPTRF، مخزنة كـ
!>          مصفوفة مثلثية معبأة.
!>

IPIV

!>          IPIV is INTEGER array, dimension (N)
!>          تفاصيل التبادلات والهيكل القطري لـ D
!>          كما تم تحديده بواسطة CHPTRF.
!>

B

!>          B is COMPLEX array, dimension (LDB,NRHS)
!>          مصفوفة الطرف الأيمن B.
!>

LDB

!>          LDB is INTEGER
!>          البعد الرائد للمصفوفة B.  LDB >= max(1,N).
!>

X

!>          X is COMPLEX array, dimension (LDX,NRHS)
!>          عند الإدخال، مصفوفة الحل X، كما تم حسابها بواسطة CHPTRS.
!>          عند الخروج، مصفوفة الحل المحسنة X.
!>

LDX

!>          LDX is INTEGER
!>          البعد الرائد للمصفوفة X.  LDX >= max(1,N).
!>

FERR

!>          FERR is REAL array, dimension (NRHS)
!>          حد الخطأ الأمامي المقدر لكل متجه حل
!>          X(j) (العمود j من مصفوفة الحل X).
!>          إذا كانت XTRUE هي الحل الحقيقي المقابل لـ X(j)، فإن FERR(j)
!>          هو حد علوي مقدر لقيمة الجزء الأكبر
!>          من (X(j) - XTRUE) مقسومًا على قيمة الجزء الأكبر
!>          من X(j).  التقدير موثوق به بقدر
!>          موثوقية تقدير RCOND، وهو دائمًا تقريبًا
!>          مبالغة طفيفة في الخطأ الحقيقي.
!>

BERR

!>          BERR is REAL array, dimension (NRHS)
!>          الخطأ الخلفي النسبي المكون لكل متجه حل
!>          X(j) (أي، أصغر تغيير نسبي في
!>          أي عنصر في A أو B يجعل X(j) حلاً دقيقًا).
!>

WORK

!>          WORK is COMPLEX array, dimension (2*N)
!>

RWORK

!>          RWORK is REAL array, dimension (N)
!>

INFO

!>          INFO is INTEGER
!>          = 0:  خروج ناجح
!>          < 0:  إذا كان INFO = -i، فإن المعامل i كان له قيمة غير قانونية
!>

المعاملات الداخلية:

!>
