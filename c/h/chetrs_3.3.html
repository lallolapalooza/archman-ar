وثائق الوظيفة/الروتين الفرعي

subroutine chetrs_3 (character uplo, integer n, integer nrhs, complex, dimension( lda, * ) a, integer lda, complex, dimension( * ) e, integer, dimension( * ) ipiv, complex, dimension( ldb, * ) b, integer ldb, integer info)

CHETRS_3

الغرض:

!> CHETRS_3 يحل نظام المعادلات الخطية A * X = B بمصفوفة هرميتية معقدة A باستخدام التحليل الذي تم حسابه
!> بواسطة CHETRF_RK أو CHETRF_BK:
!>
!>    A = P*U*D*(U**H)*(P**T) أو A = P*L*D*(L**H)*(P**T),
!>
!> حيث U (أو L) هي مصفوفة مثلثة علوية (أو سفلية) موحدة،
!!> U**H (أو L**H) هي المرافق لـ U (أو L)، P هي مصفوفة تبديل، P**T هي منقول P، و D هي هرميتية وشبه قطرية مع كتل قطرية 1x1 و 2x2.
!>
!> هذه الخوارزمية تستخدم BLAS المستوى 3.
!>

المعلمات

UPLO 
!>          UPLO هي CHARACTER*1
!>          تحدد ما إذا كانت تفاصيل التحليل مخزنة كمصفوفة مثلثة علوية أو سفلية:
!>          = 'U':  مثلثية علوية، الشكل هو A = P*U*D*(U**H)*(P**T);
!>          = 'L':  مثلثية سفلية، الشكل هو A = P*L*D*(L**H)*(P**T).
!>

N

!>          N هي INTEGER
!>          رتبة المصفوفة A. N >= 0.
!>

NRHS

!>          NRHS هي INTEGER
!>          عدد الأطراف اليمنى، أي عدد أعمدة المصفوفة B. NRHS >= 0.
!>

A

!>          A هي مصفوفة COMPLEX، بأبعاد (LDA,N)
!>          القطر للمصفوفة شبه القطرية المتصلة D والعوامل U أو L
!>          كما تم حسابه بواسطة CHETRF_RK و CHETRF_BK:
!>            أ) عناصر القطر فقط للمصفوفة شبه القطرية المتصلة الهرميتية
!>               D على قطر A، أي D(k,k) = A(k,k);
!>               (يجب توفير عناصر فوق القطر (أو تحت القطر) للمصفوفة D
!>                عند الدخول في المصفوفة E)، و
!>            ب) إذا كان UPLO = 'U': العامل U في الجزء فوق القطري من A.
!>               إذا كان UPLO = 'L': العامل L في الجزء تحت القطري من A.
!>

LDA

!>          LDA هي INTEGER
!>          البعد الرائد للمصفوفة A. LDA >= max(1,N).
!>

E

!>          E هي مصفوفة COMPLEX، بأبعاد (N)
!>          عند الدخول، تحتوي على عناصر فوق القطر (أو تحت القطر)
!>          للمصفوفة شبه القطرية المتصلة الهرميتية D
!>          مع كتل قطرية 1x1 أو 2x2، حيث
!>          إذا كان UPLO = 'U': E(i) = D(i-1,i),i=2:N, E(1) غير مرجع؛
!>          إذا كان UPLO = 'L': E(i) = D(i+1,i),i=1:N-1, E(N) غير مرجع.
!>
!>          ملاحظة: لكتلة قطرية 1x1 D(k)، حيث
!>          1 <= k <= N، العنصر E(k) غير مرجع في كلتا
!>          حالات UPLO = 'U' أو UPLO = 'L'.
!>

IPIV

!>          IPIV هي مصفوفة INTEGER، بأبعاد (N)
!>          تفاصيل التبادلات وهيكل الكتلة لـ D
!>          كما تم تحديده بواسطة CHETRF_RK أو CHETRF_BK.
!>

B

!>          B هي مصفوفة COMPLEX، بأبعاد (LDB,NRHS)
!>          عند الدخول، مصفوفة الطرف الأيمن B.
!>          عند الخروج، مصفوفة الحل X.
!>

LDB

!>          LDB هي INTEGER
!>          البعد الرائد للمصفوفة B. LDB >= max(1,N).
!>

INFO

!>          INFO هي INTEGER
!>          = 0:  انتهاء ناجح
!>          < 0:  إذا كان INFO = -i، فإن المعامل i كان له قيمة غير قانونية
!>
