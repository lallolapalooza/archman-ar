subroutine cheequb (character uplo, integer n, complex, dimension( lda, * ) a, integer lda, real, dimension( * ) s, real scond, real amax, complex, dimension( * ) work, integer info)

CHEEQUB

الغرض:

!>
!> CHEEQUB تحسب عوامل قياس الصف والعمود المقصود بها موازنة مصفوفة هرميتية A (بالنسبة للمعيار الإقليدي) وتقليل
!> رقم حالتها. يتم حساب عوامل القياس S بواسطة خوارزمية BIN
!> (راجع المراجع) بحيث تكون المصفوفة المقاسة B ذات العناصر
!> B(i,j) = S(i)*A(i,j)*S(j) لها رقم حالة ضمن عامل N من
!> أصغر رقم حالة ممكن عبر جميع القياسات القطرية الممكنة.
!>

المعلمات

UPLO 
!>          UPLO هو CHARACTER*1
!>          = 'U':  المثلث العلوي لـ A مخزن؛
!>          = 'L':  المثلث السفلي لـ A مخزن.
!>

N

!>          N هو INTEGER
!>          رتبة المصفوفة A. N >= 0.
!>

A

!>          A هو مصفوفة COMPLEX، البعد (LDA,N)
!>          المصفوفة الهرميتية N×N التي سيتم حساب عوامل قياسها.
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرئيسي للمصفوفة A. LDA >= max(1,N).
!>

S

!>          S هو مصفوفة REAL، البعد (N)
!>          إذا كان INFO = 0، تحتوي S على عوامل القياس لـ A.
!>

SCOND

!>          SCOND هو REAL
!>          إذا كان INFO = 0، تحتوي S على نسبة أصغر S(i) إلى
!>          أكبر S(i). إذا كان SCOND >= 0.1 ولم يكن AMAX كبيرًا جدًا
!>          أو صغيرًا جدًا، فلا يستحق القياس بواسطة S.
!>

AMAX

!>          AMAX هو REAL
!>          أكبر قيمة مطلقة لأي عنصر في المصفوفة. إذا كان AMAX قريبًا جدًا
!>          من الفائض أو قريبًا جدًا من النقص، فيجب قياس المصفوفة.
!>

WORK

!>          WORK هي مصفوفة COMPLEX، البعد (2*N)
!>

INFO

!>          INFO هو INTEGER
!>          = 0:  انتهاء ناجح
!>          < 0:  إذا كان INFO = -i، فإن المعلمة i كان لها قيمة غير قانونية
!>          > 0:  إذا كان INFO = i، فإن العنصر القطري i غير موجب.
!>
