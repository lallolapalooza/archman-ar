subroutine chbev (jobz, uplo, n, kd, ab, ldab, w, z, ldz, work, rwork, info)
CHBEV تحسب القيم الذاتية، واختيارياً، المتجهات الذاتية اليسرى و/أو اليمنى لمصفوفات أخرى

توثيق الوظيفة/الإجراء الفرعي

subroutine chbev (character jobz, character uplo, integer n, integer kd, complex, dimension( ldab, * ) ab, integer ldab, real, dimension( * ) w, complex, dimension( ldz, * ) z, integer ldz, complex, dimension( * ) work, real, dimension( * ) rwork, integer info)

CHBEV تحسب القيم الذاتية، واختيارياً، المتجهات الذاتية لمصفوفة هرميتية معقدة A ذات النطاق A.

الغرض:

!>
!> CHBEV تحسب جميع القيم الذاتية، واختيارياً، المتجهات الذاتية لمصفوفة هرميتية معقدة ذات نطاق A.
!>

المعلمات

JOBZ 
!>          JOBZ is CHARACTER*1
!>          = 'N':  احسب القيم الذاتية فقط؛
!>          = 'V':  احسب القيم الذاتية والمتجهات الذاتية.
!>

UPLO

!>          UPLO is CHARACTER*1
!>          = 'U':  يتم تخزين المثلث العلوي لـ A؛
!>          = 'L':  يتم تخزين المثلث السفلي لـ A.
!>

N

!>          N is INTEGER
!>          رتبة المصفوفة A.  N >= 0.
!>

KD

!>          KD is INTEGER
!>          عدد الأقطار العلوية للمصفوفة A إذا كان UPLO = 'U'،
!>          أو عدد الأقطار السفلية إذا كان UPLO = 'L'.  KD >= 0.
!>

AB

!>          AB is COMPLEX array, dimension (LDAB, N)
!>          عند الإدخال، المثلث العلوي أو السفلي للمصفوفة الهرميتية ذات النطاق
!>          A، مخزنة في أول KD+1 صفوف من المصفوفة.
!>          يتم تخزين العمود j من A في العمود j من المصفوفة AB
!>          على النحو التالي:
!>          if UPLO = 'U', AB(kd+1+i-j,j) = A(i,j) for max(1,j-kd)<=i<=j;
!>          if UPLO = 'L', AB(1+i-j,j)    = A(i,j) for j<=i<=min(n,j+kd).
!>
!>          عند الخروج، يتم استبدال AB بالقيم التي تم إنشاؤها أثناء
!>          الاختزال إلى شكل ثلاثي القطر. إذا كان UPLO = 'U'، فإن القطر العلوي
!>          الأول والقطر الرئيسي للمصفوفة ثلاثية القطر T
!>          يتم إرجاعهما في الصفين KD و KD+1 من AB، وإذا كان UPLO = 'L'،
!>          يتم إرجاع القطر الرئيسي والقطر الفرعي الأول لـ T في
!>          الصفين الأولين من AB.
!>

LDAB

!>          LDAB is INTEGER
!>          البعد الرئيسي للمصفوفة AB.  LDAB >= KD + 1.
!>

W

!>          W is REAL array, dimension (N)
!>          إذا كان INFO = 0، فإن القيم الذاتية بترتيب تصاعدي.
!>

Z

!>          Z is COMPLEX array, dimension (LDZ, N)
!>          إذا كان JOBZ = 'V'، فعندما يكون INFO = 0، تحتوي Z على المتجهات الذاتية
!>          المتعامدة للمصفوفة A، حيث يحمل العمود i من Z
!>          المتجه الذاتي المرتبط بـ W(i).
!>          إذا كان JOBZ = 'N'، فلا يتم الرجوع إلى Z.
!>

LDZ

!>          LDZ is INTEGER
!>          البعد الرئيسي للمصفوفة Z.  LDZ >= 1، وإذا
!>          كان JOBZ = 'V'، فإن LDZ >= max(1,N).
!>

WORK

!>          WORK is COMPLEX array, dimension (N)
!>

RWORK

!>          RWORK is REAL array, dimension (max(1,3*N-2))
!>

INFO

!>          INFO is INTEGER
!>          = 0:  انتهاء ناجح.
!>          < 0:  إذا كان INFO = -i، فإن المعلمة i كانت لها قيمة غير قانونية.
!>          > 0:  إذا كان INFO = i، فشل الخوارزمية في التقارب؛ i
!>                لم تتقارب العناصر غير القطرية للشكل الثلاثي المتوسط
!>                إلى الصفر.
!>
