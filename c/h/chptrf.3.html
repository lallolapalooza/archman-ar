subroutine chptrf (character uplo, integer n, complex, dimension( * ) ap, integer, dimension( * ) ipiv, integer info)

CHPTRF

الغرض:

!>
!> CHPTRF يحسب تحليل مصفوفة هيرميتية معبأة معقدة A باستخدام طريقة بيفتينغ المحورية القطرية لبنش-كوفمان:
!>
!>    A = U*D*U**H  أو  A = L*D*L**H
!>
!> حيث U (أو L) هو حاصل ضرب مصفوفة تبديل ومصفوفة مثلثية عليا (سفلى) وحدة،
!> و D هي مصفوفة هيرميتية ومصفوفة قطرية كتلية ذات كتل قطرية 1×1 و 2×2.
!>

المعلمات

UPLO 
!>          UPLO هو CHARACTER*1
!>          = 'U':  المثلث العلوي لـ A مخزن;
!>          = 'L':  المثلث السفلي لـ A مخزن.
!>

N

!>          N هو INTEGER
!>          ترتيب المصفوفة A.  N >= 0.
!>

AP

!>          AP هو مصفوفة COMPLEX، ذات بعد (N*(N+1)/2)
!>          عند الدخول، المثلث العلوي أو السفلي للمصفوفة الهيرميتية
!>          A، معبأة عمودياً في مصفوفة خطية.  يتم تخزين العمود j للمصفوفة A
!>          في المصفوفة AP كما يلي:
!>          إذا كان UPLO = 'U'، فإن AP(i + (j-1)*j/2) = A(i,j) لـ 1<=i<=j;
!>          إذا كان UPLO = 'L'، فإن AP(i + (j-1)*(2n-j)/2) = A(i,j) لـ j<=i<=n.
!>
!>          عند الخروج، مصفوفة القطر الكتلية D والعوامل المستخدمة
!>          للحصول على العامل U أو L، مخزنة كمصفوفة مثلثية معبأة
!>          تحل محل A (انظر أدناه لمزيد من التفاصيل).
!>

IPIV

!>          IPIV هو مصفوفة INTEGER، ذات بعد (N)
!>          تفاصيل التبادلات وبنية القطر D.
!>          إذا كان IPIV(k) > 0، فقد تم تبديل الصفوف والأعمدة k و IPIV(k)
!>          وكان D(k,k) كتلة قطرية 1×1.
!>          إذا كان UPLO = 'U' و IPIV(k) = IPIV(k-1) < 0، فقد تم تبديل الصفوف
!>          والأعمدة k-1 و -IPIV(k) وكانت D(k-1:k,k-1:k)
!>          كتلة قطرية 2×2.  إذا كان UPLO = 'L' و IPIV(k) =
!>          IPIV(k+1) < 0، فقد تم تبديل الصفوف والأعمدة k+1 و -IPIV(k)
!>          وكان D(k:k+1,k:k+1) كتلة قطرية 2×2.
!>

INFO

!>          INFO هو INTEGER
!>          = 0: خروج ناجح
!>          < 0: إذا كان INFO = -i، فإن المعلمة i كانت قيمة غير قانونية
!>          > 0: إذا كان INFO = i، فإن D(i,i) تساوي صفرًا تمامًا.  تم إكمال التحليل،
!>               ولكن المصفوفة القطرية الكتلية D منفردة تمامًا، وسوف يحدث
!>               قسمة على صفر إذا تم استخدامها لحل نظام معادلات.
!>

تفاصيل إضافية:

!>
!>  إذا كان UPLO = 'U'، فإن A = U*D*U**H، حيث
!>     U = P(n)*U(n)* ... *P(k)U(k)* ...,
!>  أي أن U هو حاصل ضرب الحدود P(k)*U(k)، حيث ينقص k من n إلى
!>  1 بخطوات 1 أو 2، و D هي مصفوفة قطرية كتلية ذات كتل قطرية 1×1
!>  و 2×2، D(k).  P(k) هي مصفوفة تبديل كما هو محدد بواسطة IPIV(k)،
!>  و U(k) هي مصفوفة مثلثة عليا وحدة، بحيث إذا كانت الكتلة القطرية D(k)
!>  من الرتبة s (s = 1 أو 2)، فإن
!>
!>             (   I    v    0   )   k-s
!>     U(k) =  (   0    I    0   )   s
!>             (   0    0    I   )   n-k
!>                k-s   s   n-k
!>
!>  إذا كان s = 1، فإن D(k) تستبدل A(k,k)، و v تستبدل A(1:k-1,k).
!>  إذا كان s = 2، فإن المثلث العلوي لـ D(k) يستبدل A(k-1,k-1)، A(k-1,k)،
!>  و A(k,k)، و v تستبدل A(1:k-2,k-1:k).
!>
!>  إذا كان UPLO = 'L'، فإن A = L*D*L**H، حيث
!>     L = P(1)*L(1)* ... *P(k)*L(k)* ...,
!>  أي أن L هو حاصل ضرب الحدود P(k)*L(k)، حيث يزيد k من 1 إلى
!>  n بخطوات 1 أو 2، و D هي مصفوفة قطرية كتلية ذات كتل قطرية 1×1
!>  و 2×2، D(k).  P(k) هي مصفوفة تبديل كما هو محدد بواسطة IPIV(k)،
!>  و L(k) هي مصفوفة مثلثة سفلية وحدة، بحيث إذا كانت الكتلة القطرية D(k)
!>  من الرتبة s (s = 1 أو 2)، فإن
!>
!>             (   I    0     0   )  k-1
!>     L(k) =  (   0    I     0   )  s
!>             (   0    v     I   )  n-k-s+1
!>                k-1   s  n-k-s+1
!>
!>  إذا كان s = 1، فإن D(k) تستبدل A(k,k)، و v تستبدل A(k+1:n,k).
!>  إذا كان s = 2، فإن المثلث السفلي لـ D(k) يستبدل A(k,k)، A(k+1,k)،
!>  و A(k+1,k+1)، و v تستبدل A(k+2:n,k:k+1).
!>
