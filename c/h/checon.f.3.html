subroutine checon (character uplo, integer n, complex, dimension( lda, * ) a, integer lda, integer, dimension( * ) ipiv, real anorm, real rcond, complex, dimension( * ) work, integer info)

CHECON

الغرض:

!>
!> CHECON تقدر مقلوب العدد المشروط لمصفوفة معقدة متوحدة A باستخدام التحليل A = U*D*U**H أو
!> A = L*D*L**H محسوب بواسطة CHETRF.
!>
!> يتم الحصول على تقدير لـ norm(inv(A))، ويتم حساب مقلوب العدد المشروط على أنه RCOND = 1 / (ANORM * norm(inv(A))).
!>

المعلمات

UPLO 
!>          UPLO هو CHARACTER*1
!>          يحدد ما إذا كانت تفاصيل التحليل مخزنة
!>          كمصفوفة مثلثية علوية أو سفلية.
!>          = 'U':  مثلثية علوية، الشكل هو A = U*D*U**H؛
!>          = 'L':  مثلثية سفلية، الشكل هو A = L*D*L**H.
!>

N

!>          N هو INTEGER
!>          ترتيب المصفوفة A.  N >= 0.
!>

A

!>          A هو مصفوفة COMPLEX، البعد (LDA,N)
!>          المصفوفة القطرية الجزئية D والمضاعفات المستخدمة
!>          للحصول على العامل U أو L كما تم حسابه بواسطة CHETRF.
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرائد للمصفوفة A.  LDA >= max(1,N).
!>

IPIV

!>          IPIV هو مصفوفة INTEGER، البعد (N)
!>          تفاصيل التبادلات وهيكل D الجزئي
!>          كما تم تحديده بواسطة CHETRF.
!>

ANORM

!>          ANORM هو REAL
!>          المعيار 1 للمصفوفة الأصلية A.
!>

RCOND

!>          RCOND هو REAL
!>          مقلوب العدد المشروط للمصفوفة A،
!>          محسوب كـ RCOND = 1/(ANORM * AINVNM)، حيث AINVNM هو تقدير
!>          لمعيار 1 للمصفوفة inv(A) المحسوب في هذا الروتين.
!>

WORK

!>          WORK هي مصفوفة COMPLEX، البعد (2*N)
!>

INFO

!>          INFO هو INTEGER
!>          = 0:  خروج ناجح
!>          < 0:  إذا كان INFO = -i، كان المعامل i له قيمة غير قانونية
!>
