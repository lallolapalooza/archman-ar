توثيق الوظيفة/الروتين الفرعي

subroutine cherk (character uplo, character trans, integer n, integer k, real alpha, complex, dimension(lda,*) a, integer lda, real beta, complex, dimension(ldc,*) c, integer ldc)

CHERK

الغرض:

!>
!> CHERK  تنفيذ إحدى عمليات الرتبة k الهرمية
!>
!>    C := alpha*A*A**H + beta*C,
!>
!> أو
!>
!>    C := alpha*A**H*A + beta*C,
!>
!> حيث  alpha  و beta  هما قيمتان قياسيتان حقيقيتان، C هي مصفوفة هرمية
!> بحجم n × n  و A هي مصفوفة بحجم n × k  في الحالة الأولى و مصفوفة بحجم
!> k × n في الحالة الثانية.
!>

المعلمات

UPLO 
!>          UPLO هو CHARACTER*1
!>           عند الدخول،  UPLO  يحدد ما إذا كان الجزء المثلثي العلوي أو السفلي
!>           من المصفوفة C يجب الإشارة إليه على النحو التالي:
!>
!>              UPLO = 'U' أو 'u'   يجب الإشارة فقط إلى الجزء المثلثي العلوي من C.
!>
!>              UPLO = 'L' أو 'l'   يجب الإشارة فقط إلى الجزء المثلثي السفلي من C.
!>

TRANS

!>          TRANS هو CHARACTER*1
!>           عند الدخول،  TRANS  يحدد العملية المراد تنفيذها على النحو التالي:
!>
!>              TRANS = 'N' أو 'n'   C := alpha*A*A**H + beta*C.
!>
!>              TRANS = 'C' أو 'c'   C := alpha*A**H*A + beta*C.
!>

N

!>          N هو INTEGER
!>           عند الدخول،  N  يحدد رتبة المصفوفة C. يجب أن تكون N
!>           أكبر من أو تساوي الصفر.
!>

K

!>          K هو INTEGER
!>           عند الدخول مع  TRANS = 'N' أو 'n'،  K  يحدد عدد
!>           الأعمدة  في المصفوفة   A، وعند الدخول مع
!>           TRANS = 'C' أو 'c'،  K  يحدد عدد الصفوف في
!>           المصفوفة A. يجب أن تكون K أكبر من أو تساوي الصفر.
!>

ALPHA

!>          ALPHA هو REAL
!>           عند الدخول،  ALPHA  يحدد القيمة القياسية alpha.
!>

A

!>          A هي مصفوفة COMPLEX، بأبعاد ( LDA, ka )، حيث ka هي
!>           k  عندما  TRANS = 'N' أو 'n'،  و n  في الحالات الأخرى.
!>           قبل الدخول مع  TRANS = 'N' أو 'n'،  يجب أن يحتوي الجزء الرئيسي بحجم n × k
!>           من المصفوفة A على المصفوفة A، وفي الحالات الأخرى
!>           يجب أن يحتوي الجزء الرئيسي بحجم k × n  من المصفوفة A على
!>           المصفوفة A.
!>

LDA

!>          LDA هو INTEGER
!>           عند الدخول،  LDA  يحدد البعد الأول للمصفوفة A كما تم تعريفه
!>           في البرنامج (الفرعي) المستدعي.   عندما  TRANS = 'N' أو 'n'
!>           يجب أن تكون LDA أكبر من أو تساوي  max( 1, n )، وفي الحالات الأخرى
!>           يجب أن تكون LDA أكبر من أو تساوي  max( 1, k ).
!>

BETA

!>          BETA هو REAL
!>           عند الدخول،  BETA  يحدد القيمة القياسية beta.
!>

C

!>          C هي مصفوفة COMPLEX، بأبعاد ( LDC, N )
!>           قبل الدخول مع  UPLO = 'U' أو 'u'،  يجب أن يحتوي الجزء المثلثي العلوي الرئيسي بحجم n × n
!>           من المصفوفة C على الجزء المثلثي العلوي
!>           من المصفوفة الهرمية  والجزء المثلثي السفلي الصارم لـ C غير مشار إليه.   عند الخروج،
!>           يتم الكتابة فوق الجزء المثلثي العلوي من المصفوفة C بالجزء
!>           المثلثي العلوي للمصفوفة المحدثة.
!>           قبل الدخول مع  UPLO = 'L' أو 'l'،  يجب أن يحتوي الجزء المثلثي السفلي الرئيسي بحجم n × n
!>           من المصفوفة C على الجزء المثلثي السفلي
!>           من المصفوفة الهرمية  والجزء المثلثي العلوي الصارم لـ C غير مشار إليه.   عند الخروج،
!>           يتم الكتابة فوق الجزء المثلثي السفلي من المصفوفة C بالجزء
!>           المثلثي السفلي للمصفوفة المحدثة.
!>           لاحظ أن الأجزاء التخيلية للعناصر القطرية لا تحتاج
!>           إلى تعيينها،  ويفترض أنها صفر، وعند الخروج
!>           يتم تعيينها إلى صفر.
!>

LDC

!>          LDC هو INTEGER
!>           عند الدخول،  LDC  يحدد البعد الأول للمصفوفة C كما تم تعريفه
!>           في البرنامج (الفرعي) المستدعي.   يجب أن تكون LDC
!>           أكبر من أو تساوي  max( 1, n ).
!>
