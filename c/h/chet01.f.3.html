توثيق الوظيفة/الروتين الفرعي

subroutine chet01 (character uplo, integer n, complex, dimension( lda, * ) a, integer lda, complex, dimension( ldafac, * ) afac, integer ldafac, integer, dimension( * ) ipiv, complex, dimension( ldc, * ) c, integer ldc, real, dimension( * ) rwork, real resid)

CHET01

الغرض:

!>
!> CHET01 يعيد بناء مصفوفة هرميتية غير محددة A من تحليلها
!> L*D*L' أو U*D*U' ويحسب مقدار الباقي
!>    norm( C - A ) / ( N * norm(A) * EPS )،
!> حيث C هي المصفوفة المعاد بناؤها، EPS هو الخطأ المطبعي للآلة،
!> L' هو منقول معكوس L، و U' هو منقول معكوس U.
!>

المعاملات

UPLO 
!>          UPLO هو CHARACTER*1
!>          يحدد ما إذا كان الجزء العلوي أو السفلي المثلث من المصفوفة
!>          الهرميتية A مخزنًا:
!>          = 'U':  مثلثي علوي
!>          = 'L':  مثلثي سفلي
!>

N

!>          N هو INTEGER
!>          عدد الصفوف والأعمدة للمصفوفة A.  N >= 0.
!>

A

!>          A هو مصفوفة COMPLEX، البعد (LDA,N)
!>          المصفوفة الهرميتية الأصلية A.
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرئيسي للمصفوفة A.  LDA >= max(1,N)
!>

AFAC

!>          AFAC هو مصفوفة COMPLEX، البعد (LDAFAC,N)
!>          الشكل المفكك للمصفوفة A.  AFAC يحتوي على المصفوفة القطرية
!>          الكتلة D والمعاملات المستخدمة للحصول على العامل
!>          L أو U من تحليل الكتلة L*D*L' أو U*D*U'
!>          كما تم حسابه بواسطة CHETRF.
!>

LDAFAC

!>          LDAFAC هو INTEGER
!>          البعد الرئيسي للمصفوفة AFAC.  LDAFAC >= max(1,N).
!>

IPIV

!>          IPIV هو مصفوفة INTEGER، البعد (N)
!>          مؤشرات المحورية من CHETRF.
!>

C

!>          C هو مصفوفة COMPLEX، البعد (LDC,N)
!>

LDC

!>          LDC هو INTEGER
!>          البعد الرئيسي للمصفوفة C.  LDC >= max(1,N).
!>

RWORK

!>          RWORK هو مصفوفة REAL، البعد (N)
!>

RESID

!>          RESID هو REAL
!>          إذا كان UPLO = 'L'، norm(L*D*L' - A) / ( N * norm(A) * EPS )
!>          إذا كان UPLO = 'U'، norm(U*D*U' - A) / ( N * norm(A) * EPS )
!>
