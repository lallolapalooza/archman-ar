subroutine chptrs (character uplo, integer n, integer nrhs, complex, dimension( * ) ap, integer, dimension( * ) ipiv, complex, dimension( ldb, * ) b, integer ldb, integer info)

CHPTRS

الغرض:

!>
!> CHPTRS يحل نظام المعادلات الخطية A*X = B مع مصفوفة هرميتية معقدة A مخزنة بصيغة مضغوطة باستخدام التحليل
!> A = U*D*U**H أو A = L*D*L**H محسوبة بواسطة CHPTRF.
!>

المعاملات

UPLO 
!>          UPLO هو CHARACTER*1
!>          يحدد ما إذا كانت تفاصيل التحليل مخزنة كمصفوفة مثلثة علوية أو سفلية.
!>          = 'U':  مثلثة علوية، الشكل هو A = U*D*U**H;
!>          = 'L':  مثلثة سفلية، الشكل هو A = L*D*L**H.
!>

N

!>          N هو INTEGER
!>          ترتيب المصفوفة A.  N >= 0.
!>

NRHS

!>          NRHS هو INTEGER
!>          عدد الأطراف اليمنى، أي عدد أعمدة المصفوفة B.  NRHS >= 0.
!>

AP

!>          AP هو مصفوفة COMPLEX، الأبعاد (N*(N+1)/2)
!>          المصفوفة القطرية المدمجة D والمعاملات المستخدمة للحصول على العامل U أو L كما تم حسابها بواسطة CHPTRF، مخزنة كمصفوفة مثلثية مضغوطة.
!>

IPIV

!>          IPIV هو مصفوفة INTEGER، الأبعاد (N)
!>          تفاصيل التبادلات والهيكل المدمج لـ D كما تم تحديده بواسطة CHPTRF.
!>

B

!>          B هي مصفوفة COMPLEX، الأبعاد (LDB,NRHS)
!>          عند الإدخال، مصفوفة الطرف الأيمن B.
!>          عند الخروج، مصفوفة الحل X.
!>

LDB

!>          LDB هو INTEGER
!>          البعد الرائد للمصفوفة B.  LDB >= max(1,N).
!>

INFO

!>          INFO هو INTEGER
!>          = 0:  الخروج الناجح
!>          < 0: إذا كان INFO = -i، فإن المعامل i كان له قيمة غير قانونية
!>
