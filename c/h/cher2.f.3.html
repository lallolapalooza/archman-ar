توثيق الدالة/الروتين الفرعي

subroutine cher2 (character uplo, integer n, complex alpha, complex, dimension(*) x, integer incx, complex, dimension(*) y, integer incy, complex, dimension(lda,*) a, integer lda)

CHER2

الغرض:

!>
!> CHER2  يقوم بعملية الرتبة 2 الهرمية
!>
!>    A := alpha*x*y**H + conjg( alpha )*y*x**H + A,
!>
!> حيث alpha هو عدد قياسي، x و y هما متجهان يحتويان على n عنصر، و A مصفوفة هرمية بحجم n × n.
!>

المعلمات

UPLO 
!>          UPLO هو CHARACTER*1
!>           عند الإدخال، يحدد UPLO ما إذا كان سيتم الرجوع إلى الجزء العلوي أو السفلي
!>           من المصفوفة A كما يلي:
!>
!>              UPLO = 'U' أو 'u'   يتم الرجوع فقط إلى الجزء العلوي المثلثي من A
!>                                  .
!>
!>              UPLO = 'L' أو 'l'   يتم الرجوع فقط إلى الجزء السفلي المثلثي من A
!>                                  .
!>

N

!>          N هو INTEGER
!>           عند الإدخال، يحدد N رتبة المصفوفة A.
!>           يجب أن يكون N أكبر من أو يساوي صفر.
!>

ALPHA

!>          ALPHA هو COMPLEX
!>           عند الإدخال، يحدد ALPHA العدد القياسي alpha.
!>

X

!>          X هو مصفوفة COMPLEX، بحجم لا يقل عن
!>           ( 1 + ( n - 1 )*abs( INCX ) ).
!>           قبل الإدخال، يجب أن تحتوي المصفوفة X المعدلة على المتجه x المكون من n عنصر.
!>

INCX

!>          INCX هو INTEGER
!>           عند الإدخال، يحدد INCX الزيادة لعناصر X.
!>           يجب ألا يكون INCX صفرًا.
!>

Y

!>          Y هو مصفوفة COMPLEX، بحجم لا يقل عن
!>           ( 1 + ( n - 1 )*abs( INCY ) ).
!>           قبل الإدخال، يجب أن تحتوي المصفوفة Y المعدلة على المتجه y المكون من n عنصر.
!>

INCY

!>          INCY هو INTEGER
!>           عند الإدخال، يحدد INCY الزيادة لعناصر Y.
!>           يجب ألا يكون INCY صفرًا.
!>

A

!>          A هي مصفوفة COMPLEX، بحجم ( LDA, N )
!>           عند الإدخال مع UPLO = 'U' أو 'u'، يجب أن يحتوي الجزء العلوي المثلثي الرئيسي
!>           بمقاس n × n من المصفوفة A على الجزء العلوي المثلثي للمصفوفة الهرمية،
!>           ولا يتم الرجوع إلى الجزء السفلي المثلثي الصارم لـ A. عند الخروج،
!>           يتم الكتابة فوق الجزء العلوي المثلثي من المصفوفة A بالجزء العلوي المثلثي
!>           للمصفوفة المحدثة.
!>           عند الإدخال مع UPLO = 'L' أو 'l'، يجب أن يحتوي الجزء السفلي المثلثي الرئيسي
!>           بمقاس n × n من المصفوفة A على الجزء السفلي المثلثي للمصفوفة الهرمية،
!>           ولا يتم الرجوع إلى الجزء العلوي المثلثي الصارم لـ A. عند الخروج،
!>           يتم الكتابة فوق الجزء السفلي المثلثي من المصفوفة A بالجزء السفلي المثلثي
!>           للمصفوفة المحدثة.
!>           لاحظ أن الأجزاء التخيلية للعناصر القطرية لا تحتاج إلى تعيين،
!>           يفترض أنها صفر، وعند الخروج يتم تعيينها إلى صفر.
!>

LDA

!>          LDA هو INTEGER
!>           عند الإدخال، يحدد LDA البعد الأول للمصفوفة A كما تم تعريفه
!>           في البرنامج (الفرعي) المستدعي.
!>           يجب أن يكون LDA على الأقل max( 1, n ).
!>
