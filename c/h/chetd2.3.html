subroutine chetd2 (uplo, n, a, lda, d, e, tau, info)
CHETD2 تقلل مصفوفة هيرميتية إلى شكل ثلاثي الأقطار متماثل حقيقي عن طريق تحويل تشابه وحدوي (خوارزمية غير مجمعة).

توثيق الوظيفة/الروتين الفرعي

subroutine chetd2 (character uplo, integer n, complex, dimension( lda, * ) a, integer lda, real, dimension( * ) d, real, dimension( * ) e, complex, dimension( * ) tau, integer info)

CHETD2 تقلل مصفوفة هيرميتية معقدة إلى شكل ثلاثي الأقطار متماثل حقيقي عن طريق تحويل تشابه وحدوي (خوارزمية غير مجمعة).

الغرض:

!>
!> CHETD2 تقلل مصفوفة هيرميتية معقدة A إلى شكل ثلاثي الأقطار متماثل حقيقي T
!> عن طريق تحويل تشابه وحدوي:
!> Q**H * A * Q = T.
!>

المعلمات

UPLO 
!>          UPLO هو CHARACTER*1
!>          يحدد ما إذا كان الجزء العلوي أو السفلي المثلثي للمصفوفة الهيرميتية A
!>          مخزنًا:
!>          = 'U':  مثلثي علوي
!>          = 'L':  مثلثي سفلي
!>

N

!>          N هو INTEGER
!>          رتبة المصفوفة A.  N >= 0.
!>

A

!>          A هو مصفوفة COMPLEX، البعد (LDA,N)
!>          عند الدخول، المصفوفة الهيرميتية A.  إذا كان UPLO = 'U'، فإن الجزء
!>          المثلثي العلوي n-by-n من A يحتوي على الجزء المثلثي العلوي
!>          للمصفوفة A، ولا يتم الرجوع إلى الجزء المثلثي السفلي الصارم لـ A.  إذا كان UPLO = 'L'، فإن الجزء
!>          المثلثي السفلي n-by-n من A يحتوي على الجزء المثلثي السفلي
!>          للمصفوفة A، ولا يتم الرجوع إلى الجزء المثلثي العلوي الصارم لـ A.
!>          عند الخروج، إذا كان UPLO = 'U'، يتم استبدال عناصر القطر والعناصر فوق القطر الأول
!>          لـ A بالعناصر المقابلة للمصفوفة ثلاثية الأقطار T، والعناصر فوق القطر الأول،
!>          مع المصفوفة TAU، تمثل المصفوفة الوحدوية Q كحاصل ضرب عاكسات أولية؛ إذا كان UPLO
!>          = 'L'، يتم استبدال عناصر القطر والعناصر تحت القطر الأول
!>          لـ A بالعناصر المقابلة للمصفوفة ثلاثية الأقطار T، والعناصر تحت القطر الأول،
!>          مع المصفوفة TAU، تمثل المصفوفة الوحدوية Q كحاصل ضرب عاكسات أولية. انظر التفاصيل الإضافية.
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرئيسي للمصفوفة A.  LDA >= max(1,N).
!>

D

!>          D هي مصفوفة REAL، البعد (N)
!>          عناصر القطر للمصفوفة ثلاثية الأقطار T:
!>          D(i) = A(i,i).
!>

E

!>          E هي مصفوفة REAL، البعد (N-1)
!>          العناصر خارج القطر للمصفوفة ثلاثية الأقطار T:
!>          E(i) = A(i,i+1) إذا كان UPLO = 'U'، E(i) = A(i+1,i) إذا كان UPLO = 'L'.
!>

TAU

!>          TAU هي مصفوفة COMPLEX، البعد (N-1)
!>          العوامل العددية للعواكس الأولية (انظر التفاصيل الإضافية).
!>

INFO

!>          INFO هو INTEGER
!>          = 0:  انتهاء ناجح
!>          < 0:  إذا كان INFO = -i، فإن الوسيط i كان له قيمة غير قانونية.
!>

تفاصيل إضافية:

!>
!>  إذا كان UPLO = 'U'، يتم تمثيل المصفوفة Q كحاصل ضرب عاكسات أولية
!>  Q = H(n-1) . . . H(2) H(1).
!>
!>  كل H(i) لها الشكل
!>
!>     H(i) = I - tau * v * v**H
!>
!>  حيث tau هو عدد مركب، و v هو متجه مركب مع
!>  v(i+1:n) = 0 و v(i) = 1؛ v(1:i-1) مخزن عند الخروج في
!>  A(1:i-1,i+1)، و tau في TAU(i).
!>
!>  إذا كان UPLO = 'L'، يتم تمثيل المصفوفة Q كحاصل ضرب عاكسات أولية
!>  Q = H(1) H(2) . . . H(n-1).
!>
!>  كل H(i) لها الشكل
!>
!>     H(i) = I - tau * v * v**H
!>
!>  حيث tau هو عدد مركب، و v هو متجه مركب مع
!>  v(1:i) = 0 و v(i+1) = 1؛ v(i+2:n) مخزن عند الخروج في A(i+2:n,i)،
!>  و tau في TAU(i).
!>
!>  محتويات A عند الخروج موضحة بالأمثلة التالية
!>  مع n = 5:
!>
!>  إذا كان UPLO = 'U':                       إذا كان UPLO = 'L':
!>
!>    (  d   e   v2  v3  v4 )              (  d                  )
!>    (      d   e   v3  v4 )              (  e   d              )
!>    (          d   e   v4 )              (  v1  e   d          )
!>    (              d   e  )              (  v1  v2  e   d      )
!>    (                  d  )              (  v1  v2  v3  e   d  )
!>
!>  حيث d و e تمثل عناصر القطر والعناصر خارج القطر لـ T، و vi
!>  يمثل عنصرًا من المتجه الذي يحدد H(i).
!>
