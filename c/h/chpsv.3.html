وثائق الوظيفة/الإجراء الفرعي

subroutine chpsv (character uplo, integer n, integer nrhs, complex, dimension( * ) ap, integer, dimension( * ) ipiv, complex, dimension( ldb, * ) b, integer ldb, integer info)

CHPSV يحسب حل نظام المعادلات الخطية المعقدة A * X = B،

الغرض:

!>
!> CHPSV يحسب حل نظام المعادلات الخطية المعقدة
!>    A * X = B,
!> حيث A مصفوفة هيرميتية N × N مخزنة بتنسيق مضغوط و X و B مصفوفتان N × NRHS.
!>
!> يتم استخدام طريقة التقسيم القطري لتحليل A إلى:
!>    A = U * D * U**H،  إذا كان UPLO = 'U'، أو
!>    A = L * D * L**H،  إذا كان UPLO = 'L'،
!> حيث U (أو L) هو ناتج ضرب مصفوفة تبديل ومصفوفة عليا (سفلى) وحدوية.
!> D هي مصفوفة هيرميتية وقطرية جزئية ذات كتل قطرية 1 × 1 و 2 × 2.
!> ثم يتم استخدام الشكل المحلل لـ A لحل نظام المعادلات A * X = B.
!>

المعلمات

UPLO 
!>          UPLO هو CHARACTER*1
!>          = 'U':  المثلث العلوي لـ A مخزن؛
!>          = 'L':  المثلث السفلي لـ A مخزن.
!>

N

!>          N هو INTEGER
!>          عدد المعادلات الخطية، أي رتبة المصفوفة A.  N >= 0.
!>

NRHS

!>          NRHS هو INTEGER
!>          عدد الأطراف اليمنى، أي عدد أعمدة المصفوفة B.  NRHS >= 0.
!>

AP

!>          AP هي مصفوفة COMPLEX، الأبعاد (N*(N+1)/2)
!>          عند الدخول، المثلث العلوي أو السفلي للمصفوفة الهيرميتية A،
!>          مضغوطة عموديًا في مصفوفة خطية. يتم تخزين العمود j من A
!>          في المصفوفة AP كما يلي:
!>          إذا كان UPLO = 'U'، AP(i + (j-1)*j/2) = A(i,j) لـ 1<=i<=j؛
!>          إذا كان UPLO = 'L'، AP(i + (j-1)*(2n-j)/2) = A(i,j) لـ j<=i<=n.
!>          انظر أدناه لمزيد من التفاصيل.
!>
!>          عند الخروج، المصفوفة القطرية الجزئية D والعوامل المستخدمة
!>          للحصول على العامل U أو L من التحليل
!>          A = U*D*U**H أو A = L*D*L**H كما تم حسابها بواسطة CHPTRF،
!>          مخزنة كمصفوفة مثلثة مضغوطة في نفس تنسيق التخزين لـ A.
!>

IPIV

!>          IPIV هي مصفوفة INTEGER، الأبعاد (N)
!>          تفاصيل التبادلات وهيكل الكتلة لـ D، كما
!>          تم تحديدها بواسطة CHPTRF.  إذا كان IPIV(k) > 0، فقد تم تبديل
!>          الصفوف والأعمدة k و IPIV(k)، و D(k,k) هي كتلة قطرية 1 × 1.
!>          إذا كان UPLO = 'U' وكان IPIV(k) = IPIV(k-1) < 0،
!>          فقد تم تبديل الصفوف والأعمدة k-1 و -IPIV(k) و D(k-1:k,k-1:k)
!>          هي كتلة قطرية 2 × 2.  إذا كان UPLO = 'L' وكان
!>          IPIV(k) = IPIV(k+1) < 0، فقد تم تبديل الصفوف والأعمدة k+1 و
!>          -IPIV(k) و D(k:k+1,k:k+1) هي كتلة قطرية 2 × 2.
!>

B

!>          B هي مصفوفة COMPLEX، الأبعاد (LDB,NRHS)
!>          عند الدخول، مصفوفة الطرف الأيمن B ذات الأبعاد N × NRHS.
!>          عند الخروج، إذا كان INFO = 0، مصفوفة الحل X ذات الأبعاد N × NRHS.
!>

LDB

!>          LDB هو INTEGER
!>          البعد الرئيسي للمصفوفة B.  LDB >= max(1,N).
!>

INFO

!>          INFO هو INTEGER
!>          = 0:  انتهاء ناجح
!>          < 0:  إذا كان INFO = -i، فقد كان الوسيط i غير صالح
!>          > 0:  إذا كان INFO = i، فإن D(i,i) يساوي صفرًا تمامًا.  اكتمل
!>                التحليل، ولكن المصفوفة القطرية الجزئية D مفردة تمامًا،
!>                لذلك لم يكن من الممكن حساب الحل.
!>
