subroutine chetri2x (character uplo, integer n, complex, dimension( lda, * ) a, integer lda, integer, dimension( * ) ipiv, complex, dimension( n+nb+1,* ) work, integer nb, integer info)

CHETRI2X

الغرض:

!>
!> CHETRI2X يحسب معكوس مصفوفة هيرميتية غير محددة معقدة A باستخدام التحليل A = U*D*U**H أو A = L*D*L**H الذي تم حسابه بواسطة CHETRF.
!>

المعلمات

UPLO 
!>          UPLO is CHARACTER*1
!>          يحدد ما إذا كانت تفاصيل التحليل مخزنة كمصفوفة مثلثة علوية أو سفلية.
!>          = 'U':  مثلثة علوية، الشكل هو A = U*D*U**H؛
!>          = 'L':  مثلثة سفلية، الشكل هو A = L*D*L**H.
!>

N

!>          N is INTEGER
!>          رتبة المصفوفة A. N >= 0.
!>

A

!>          A is COMPLEX array, dimension (LDA,N)
!>          عند الإدخال، مصفوفة القطر NNB والمعاملات المستخدمة للحصول على العامل U أو L كما تم حسابه بواسطة CHETRF.
!>
!>          عند الخروج، إذا كان INFO = 0، فإن معكوس (متناظر) للمصفوفة الأصلية. إذا كان UPLO = 'U'، يتم تشكيل الجزء المثلثي العلوي من المعكوس ولا يتم الرجوع إلى جزء A تحت القطر؛ إذا كان UPLO = 'L'، يتم تشكيل الجزء المثلثي السفلي من المعكوس ولا يتم الرجوع إلى جزء A فوق القطر.
!>

LDA

!>          LDA is INTEGER
!>          البعد الرائد للمصفوفة A. LDA >= max(1,N).
!>

IPIV

!>          IPIV is INTEGER array, dimension (N)
!>          تفاصيل التبادلات وهيكل NNB للمصفوفة D كما تم تحديدها بواسطة CHETRF.
!>

WORK

!>          WORK is COMPLEX array, dimension (N+NB+1,NB+3)
!>

NB

!>          NB is INTEGER
!>          حجم الكتلة
!>

INFO

!>          INFO is INTEGER
!>          = 0: خروج ناجح
!>          < 0: إذا كان INFO = -i، فإن المعامل i كان له قيمة غير قانونية
!>          > 0: إذا كان INFO = i، فإن D(i,i) = 0؛ المصفوفة منفردة ولا يمكن حساب معكوسها.
!>
