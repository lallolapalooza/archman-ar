subroutine chegv_2stage (integer itype, character jobz, character uplo, integer n, complex, dimension( lda, * ) a, integer lda, complex, dimension( ldb, * ) b, integer ldb, real, dimension( * ) w, complex, dimension( * ) work, integer lwork, real, dimension( * ) rwork, integer info)

CHEGV_2STAGE

الغرض:

!>
!> CHEGV_2STAGE يحسب جميع القيم الذاتية، اختياريًا، المتجهات الذاتية
!> لمشكلة القيم الذاتية الهرميتية المعرفة عمومًا، من الشكل
!> A*x=(lambda)*B*x,  A*Bx=(lambda)*x,  أو B*A*x=(lambda)*x.
!> هنا يُفترض أن A و B هرميتيان وأن B أيضًا
!> معرفة موجبة.
!> تستخدم هذه الروتينات تقنية المرحلتين (2stage) للاختزال إلى الشكل ثلاثي الأقطار
!> والتي أظهرت أداءً أعلى على المعماريات الحديثة وللأحجام الكبيرة N>2000.
!>

المعلمات

ITYPE 
!>          ITYPE هو عدد صحيح (INTEGER)
!>          يحدد نوع المشكلة المراد حلها:
!>          = 1:  A*x = (lambda)*B*x
!>          = 2:  A*B*x = (lambda)*x
!>          = 3:  B*A*x = (lambda)*x
!>

JOBZ

!>          JOBZ هو حرف (CHARACTER*1)
!>          = 'N':  حساب القيم الذاتية فقط؛
!>          = 'V':  حساب القيم الذاتية والمتجهات الذاتية.
!>                  غير متاح في هذا الإصدار.
!>

UPLO

!>          UPLO هو حرف (CHARACTER*1)
!>          = 'U':  المثلثات العليا لـ A و B مخزنة؛
!>          = 'L':  المثلثات السفلى لـ A و B مخزنة.
!>

N

!>          N هو عدد صحيح (INTEGER)
!>          ترتيب المصفوفتين A و B.  N >= 0.
!>

A

!>          A هي مصفوفة مركبة (COMPLEX)، البعد (LDA, N)
!>          عند الدخول، المصفوفة الهرميتية A.  إذا كانت UPLO = 'U'، فإن
!>          الجزء العلوي من المصفوفة A من الحجم N×N يحتوي على
!>          الجزء العلوي من المصفوفة A.  إذا كانت UPLO = 'L'، فإن
!>          الجزء السفلي من المصفوفة A من الحجم N×N يحتوي على
!>          الجزء السفلي من المصفوفة A.
!>
!>          عند الخروج، إذا كانت JOBZ = 'V'، فإن إذا كانت INFO = 0، تحتوي A على
!>          المصفوفة Z للمتجهات الذاتية.  تُطبع المتجهات الذاتية
!>          على النحو التالي:
!>          إذا كان ITYPE = 1 أو 2، Z**H*B*Z = I؛
!>          إذا كان ITYPE = 3، Z**H*inv(B)*Z = I.
!>          إذا كانت JOBZ = 'N'، فإن عند الخروج يتم تدمير المثلث العلوي (إذا كان UPLO='U')
!>          أو المثلث السفلي (إذا كان UPLO='L') من A، بما في ذلك
!>          القطر الرئيسي.
!>

LDA

!>          LDA هو عدد صحيح (INTEGER)
!>          البعد الرائد للمصفوفة A.  LDA >= max(1,N).
!>

B

!>          B هي مصفوفة مركبة (COMPLEX)، البعد (LDB, N)
!>          عند الدخول، المصفوفة الهرميتية الموجبة المعرفة B.
!>          إذا كانت UPLO = 'U'، فإن الجزء العلوي من المصفوفة B من الحجم N×N يحتوي على
!>          الجزء العلوي من المصفوفة B.
!>          إذا كانت UPLO = 'L'، فإن الجزء السفلي من المصفوفة B من الحجم N×N يحتوي على
!>          الجزء السفلي من المصفوفة B.
!>
!>          عند الخروج، إذا كانت INFO <= N، فإن الجزء من B الذي يحتوي على المصفوفة
!>          يتجاوز بتجاوز العامل المثلثي U أو L من تحليل Cholesky
!>          B = U**H*U أو B = L*L**H.
!>

LDB

!>          LDB هو عدد صحيح (INTEGER)
!>          البعد الرائد للمصفوفة B.  LDB >= max(1,N).
!>

W

!>          W هي مصفوفة حقيقية (REAL)، البعد (N)
!>          إذا كانت INFO = 0، فإن القيم الذاتية بترتيب تصاعدي.
!>

WORK

!>          WORK هي مصفوفة مركبة (COMPLEX)، البعد (MAX(1,LWORK))
!>          عند الخروج، إذا كانت INFO = 0، فإن WORK(1) يعيد حجم LWORK الأمثل.
!>

LWORK

!>          LWORK هو عدد صحيح (INTEGER)
!>          طول المصفوفة WORK. LWORK >= 1، عندما N <= 1؛
!>          بخلاف ذلك
!>          إذا كانت JOBZ = 'N' و N > 1، يجب الاستعلام عن LWORK.
!>                                   LWORK = MAX(1, dimension) حيث
!>                                   dimension = max(stage1,stage2) + (KD+1)*N + N
!>                                             = N*KD + N*max(KD+1,FACTOPTNB)
!>                                               + max(2*KD*KD, KD*NTHREADS)
!>                                               + (KD+1)*N + N
!>                                   حيث KD هو حجم الكتلة للاختزال،
!>                                   FACTOPTNB هو الحجم المستخدم بواسطة خوارزمية QR أو LQ
!>                                   عادةً ما يكون FACTOPTNB=128 خيارًا جيدًا
!>                                   NTHREADS هو عدد الخيوط المستخدمة عند
!>                                   تم تفعيل تجميع OpenMP، وإلا =1.
!>          إذا كانت JOBZ = 'V' و N > 1، يجب الاستعلام عن LWORK. لم يتوفر بعد.
!>
!>          إذا كانت LWORK = -1، فإن استعلام مساحة العمل مفترض؛ الروتين
!>          يحسب فقط الحجم الأمثل للمصفوفة WORK، ويعيد
!>          هذه القيمة كإدخال أول للمصفوفة WORK، ولا يتم إصدار أي رسالة خطأ
!>          متعلقة بـ LWORK بواسطة XERBLA.
!>

RWORK

!>          RWORK هي مصفوفة حقيقية (REAL)، البعد (max(1, 3*N-2))
!>

INFO

!>          INFO هو عدد صحيح (INTEGER)
!>          = 0:  نهاية ناجحة
!>          < 0:  إذا كان INFO = -i، فإن الوسيط i كان له قيمة غير قانونية
!>          > 0:  CPOTRF أو CHEEV أعاد رمز خطأ:
!>             <= N:  إذا كان INFO = i، فإن CHEEV فشل في التقارب؛
!>                    i من العناصر خارج القطر لمصفوفة وسيطة
!>                    ثلاثية الأقطار لم تتقارب إلى الصفر؛
!>             > N:   إذا كان INFO = N + i، لـ 1 <= i <= N، فإن
!>                    المُحدِّد الرئيسي الرائد من الدرجة i لـ B ليس موجبًا.
!>                    لم يكن من الممكن إكمال تحليل B
!>                    ولم يتم حساب أي قيم ذاتية أو متجهات ذاتية.
!>
