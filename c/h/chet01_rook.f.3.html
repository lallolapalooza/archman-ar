توثيق الوظيفة/الإجراء الفرعي

subroutine chet01_rook (character uplo, integer n, complex, dimension( lda, * ) a, integer lda, complex, dimension( ldafac, * ) afac, integer ldafac, integer, dimension( * ) ipiv, complex, dimension( ldc, * ) c, integer ldc, real, dimension( * ) rwork, real resid)

CHET01_ROOK

الغرض:

!>
!> CHET01_ROOK يعيد بناء مصفوفة هرميتية معقدة A من تحليلها
!> L*D*L' أو U*D*U' ويحسب الباقي
!>    norm( C - A ) / ( N * norm(A) * EPS ),
!> حيث C هي المصفوفة المعاد بناؤها، EPS هي وحدة الآلة،
!> L' هي منقولة L، و U' هي منقولة U.
!>

المعلمات

UPLO 
!>          UPLO is CHARACTER*1
!>          يحدد ما إذا كان الجزء العلوي أو السفلي المثلثي للمصفوفة
!>          الهرميتية المعقدة A مخزنًا:
!>          = 'U':  مثلثي علوي
!>          = 'L':  مثلثي سفلي
!>

N

!>          N is INTEGER
!>          عدد الصفوف والأعمدة للمصفوفة A.  N >= 0.
!>

A

!>          A is COMPLEX array, dimension (LDA,N)
!>          المصفوفة الهرميتية المعقدة الأصلية A.
!>

LDA

!>          LDA is INTEGER
!>          البعد الرائد للمصفوفة A.  LDA >= max(1,N)
!>

AFAC

!>          AFAC is COMPLEX array, dimension (LDAFAC,N)
!>          الشكل المحلل للمصفوفة A.  AFAC يحتوي على المصفوفة القطرية
!>          الكتلية D والمضاعفات المستخدمة للحصول على العامل L أو U
!>          من تحليل L*D*L' أو U*D*U' الكتلي
!>          كما تم حسابه بواسطة CSYTRF_ROOK.
!>

LDAFAC

!>          LDAFAC is INTEGER
!>          البعد الرائد للمصفوفة AFAC.  LDAFAC >= max(1,N).
!>

IPIV

!>          IPIV is INTEGER array, dimension (N)
!>          فهارس المحور من CSYTRF_ROOK.
!>

C

!>          C is COMPLEX array, dimension (LDC,N)
!>

LDC

!>          LDC is INTEGER
!>          البعد الرائد للمصفوفة C.  LDC >= max(1,N).
!>

RWORK

!>          RWORK is REAL array, dimension (N)
!>

RESID

!>          RESID is REAL
!>          إذا كان UPLO = 'L'، norm(L*D*L' - A) / ( N * norm(A) * EPS )
!>          إذا كان UPLO = 'U'، norm(U*D*U' - A) / ( N * norm(A) * EPS )
!>
