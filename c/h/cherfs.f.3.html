subroutine cherfs (character uplo, integer n, integer nrhs, complex, dimension( lda, * ) a, integer lda, complex, dimension( ldaf, * ) af, integer ldaf, integer, dimension( * ) ipiv, complex, dimension( ldb, * ) b, integer ldb, complex, dimension( ldx, * ) x, integer ldx, real, dimension( * ) ferr, real, dimension( * ) berr, complex, dimension( * ) work, real, dimension( * ) rwork, integer info)

CHERFS

الغرض:

!>
!> CHERFS يحسن الحل المحسوب لنظام المعادلات الخطية
!> عندما تكون مصفوفة المعاملات هرميتية غير محددة، و
!> يوفر حدود الخطأ وتقديرات الخطأ الخلفي للحل.
!>

المعلمات

UPLO 
!>          UPLO is CHARACTER*1
!>          = 'U':  يتم تخزين الجزء العلوي من A؛
!>          = 'L':  يتم تخزين الجزء السفلي من A.
!>

N

!>          N is INTEGER
!>          ترتيب المصفوفة A.  N >= 0.
!>

NRHS

!>          NRHS is INTEGER
!>          عدد الأيادي اليمنى، أي عدد أعمدة
!>          المصفوفتين B و X.  NRHS >= 0.
!>

A

!>          A is COMPLEX array, dimension (LDA,N)
!>          المصفوفة الهرميتية A.  إذا كان UPLO = 'U'، فإن الجزء العلوي
!>          من A الذي حجمه N × N يحتوي على الجزء العلوي
!>          من المصفوفة A، والجزء السفلي القطري تمامًا من A
!>          لا يتم الرجوع إليه.  إذا كان UPLO = 'L'، فإن الجزء السفلي
!>          من A الذي حجمه N × N يحتوي على الجزء السفلي
!>          من المصفوفة A، والجزء العلوي القطري تمامًا من A
!>          لا يتم الرجوع إليه.
!>

LDA

!>          LDA is INTEGER
!>          البعد الرائد للمصفوفة A.  LDA >= max(1,N).
!>

AF

!>          AF is COMPLEX array, dimension (LDAF,N)
!>          الشكل المفكك للمصفوفة A.  تحتوي AF على مصفوفة قطرية
!>          كتلة D والمضاعفات المستخدمة للحصول على
!>          U أو L من التحلل A = U*D*U**H أو
!>          A = L*D*L**H كما تم حسابه بواسطة CHETRF.
!>

LDAF

!>          LDAF is INTEGER
!>          البعد الرائد للمصفوفة AF.  LDAF >= max(1,N).
!>

IPIV

!>          IPIV is INTEGER array, dimension (N)
!>          تفاصيل التبادلات وهيكل الكتلة لـ D
!>          كما تم تحديده بواسطة CHETRF.
!>

B

!>          B is COMPLEX array, dimension (LDB,NRHS)
!>          مصفوفة الجانب الأيمن B.
!>

LDB

!>          LDB is INTEGER
!>          البعد الرائد للمصفوفة B.  LDB >= max(1,N).
!>

X

!>          X is COMPLEX array, dimension (LDX,NRHS)
!>          عند الدخول، مصفوفة الحل X، كما تم حسابها بواسطة CHETRS.
!>          عند الخروج، مصفوفة الحل المحسنة X.
!>

LDX

!>          LDX is INTEGER
!>          البعد الرائد للمصفوفة X.  LDX >= max(1,N).
!>

FERR

!>          FERR is REAL array, dimension (NRHS)
!>          حد الخطأ الأمامي المقدر لكل متجه حل
!>          X(j) (العمود j من مصفوفة الحل X).
!>          إذا كان XTRUE هو الحل الحقيقي المقابل لـ X(j)، فإن FERR(j)
!>          هو حد أعلى مقدر لحجم أكبر
!>          عنصر في (X(j) - XTRUE) مقسومًا على حجم أكبر
!>          عنصر في X(j).  التقدير موثوق به مثل
!>          التقدير لـ RCOND، وهو دائمًا تقريبًا
!>          مبالغة في تقدير الخطأ الحقيقي.
!>

BERR

!>          BERR is REAL array, dimension (NRHS)
!>          الخطأ النسبي المكون لكل متجه حل
!>          X(j) (أي، أصغر تغيير نسبي في
!>          أي عنصر في A أو B يجعل X(j) حلاً دقيقًا).
!>

WORK

!>          WORK is COMPLEX array, dimension (2*N)
!>

RWORK

!>          RWORK is REAL array, dimension (N)
!>

INFO

!>          INFO is INTEGER
!>          = 0:  خروج ناجح
!>          < 0:  إذا كان INFO = -i، فإن المعامل i لديه قيمة غير قانونية
!>
