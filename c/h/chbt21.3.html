subroutine chbt21 (character uplo, integer n, integer ka, integer ks, complex, dimension( lda, * ) a, integer lda, real, dimension( * ) d, real, dimension( * ) e, complex, dimension( ldu, * ) u, integer ldu, complex, dimension( * ) work, real, dimension( * ) rwork, real, dimension( 2 ) result)

CHBT21

الغرض:

!>
!> CHBT21  يقوم بفحص تحليل عام للشكل
!>
!>         A = U S U**H
!>
!> حيث **H تعني منقول مترافق، A مصفوفة هرميتية مع حزم، U مصفوفة وحدوية،
!> و S مصفوفة قطرية (إذا كان KS=0) أو مصفوفة متناظرة ثلاثية الأقطار (إذا كان KS=1).
!>
!> بالتحديد:
!>
!>         RESULT(1) = | A - U S U**H | / ( |A| n ulp ) و
!>         RESULT(2) = | I - U U**H | / ( n ulp )
!>

المعلمات

UPLO 
!>          UPLO هو CHARACTER
!>          إذا كانت UPLO='U'، فسيتم استخدام المثلث العلوي للمصفوفتين A و V،
!>          ولن تتم الإشارة إلى المثلث السفلي (بشكل صارم).
!>          إذا كانت UPLO='L'، فسيتم استخدام المثلث السفلي للمصفوفتين A و V،
!>          ولن تتم الإشارة إلى المثلث العلوي (بشكل صارم).
!>

N

!>          N هو INTEGER
!>          حجم المصفوفة. إذا كانت صفرًا، فإن CHBT21 لا تفعل شيئًا.
!>          يجب أن يكون على الأقل صفرًا.
!>

KA

!>          KA هو INTEGER
!>          عرض الحزمة للمصفوفة A. يجب أن يكون على الأقل صفرًا. إذا
!>          كان أكبر من N-1، فسيتم استخدام max( 0, N-1 ).
!>

KS

!>          KS هو INTEGER
!>          عرض الحزمة للمصفوفة S. يمكن أن يكون صفرًا أو واحدًا فقط.
!>          إذا كان صفرًا، فإن S تكون قطرية، ولا تتم الإشارة إلى E. إذا
!>          كان واحدًا، فإن S تكون متناظرة ثلاثية الأقطار.
!>

A

!>          A هي مصفوفة COMPLEX، البعد (LDA, N)
!>          المصفوفة الأصلية (غير المحللة). يُفترض أنها
!>          هرميتية، وسيتم الإشارة فقط إلى الجزء العلوي (UPLO='U') أو الجزء السفلي فقط
!>          (UPLO='L').
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرئيسي للمصفوفة A. يجب أن يكون على الأقل 1
!>          وعلى الأقل min( KA, N-1 ).
!>

D

!>          D هي مصفوفة REAL، البعد (N)
!>          القطر للمصفوفة S (المتناظرة الثلاثية الأقطار).
!>

E

!>          E هي مصفوفة REAL، البعد (N-1)
!>          العناصر خارج القطر للمصفوفة S (المتناظرة الثلاثية الأقطار).
!>          E(1) هو العنصر (1,2) و (2,1)، E(2) هو العنصر (2,3) و
!>          (3,2)، وهكذا.
!>          لا تتم الإشارة إليها إذا كان KS=0.
!>

U

!>          U هي مصفوفة COMPLEX، البعد (LDU, N)
!>          المصفوفة الوحدوية في التحليل، معبرًا عنها كمصفوفة
!>          كثيفة (أي، ليست كناتج تحويلات هاوسهولدر،
!>          تحويلات جيفنز، إلخ.)
!>

LDU

!>          LDU هو INTEGER
!>          البعد الرئيسي للمصفوفة U. يجب أن يكون LDU على الأقل N
!>          وعلى الأقل 1.
!>

WORK

!>          WORK هي مصفوفة COMPLEX، البعد (N**2)
!>

RWORK

!>          RWORK هي مصفوفة REAL، البعد (N)
!>

RESULT

!>          RESULT هي مصفوفة REAL، البعد (2)
!>          القيم التي تم حسابها بواسطة الاختبارين الموصوفين أعلاه.
!>          القيم حاليًا محدودة بـ 1/ulp، لتجنب تجاوز السعة.
!>
