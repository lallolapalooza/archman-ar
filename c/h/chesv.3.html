توثيق الوظيفة/الروتين الفرعي

subroutine chesv (character uplo, integer n, integer nrhs, complex, dimension( lda, * ) a, integer lda, integer, dimension( * ) ipiv, complex, dimension( ldb, * ) b, integer ldb, complex, dimension( * ) work, integer lwork, integer info)

CHESV تحسب حل نظام المعادلات الخطية A * X = B لمصفوفات HE

الغرض:

!>
!> CHESV تحسب حل نظام المعادلات الخطية المعقدة
!>    A * X = B,
!> حيث A مصفوفة هيرميتية N-by-N و X و B مصفوفتان N-by-NRHS.
!>
!> يتم استخدام طريقة التنقيب القطري لتحليل A إلى
!>    A = U * D * U**H,  إذا كان UPLO = 'U', أو
!>    A = L * D * L**H,  إذا كان UPLO = 'L',
!> حيث U (أو L) هو حاصل ضرب مصفوفة تبديلية ومصفوفات مثلثة علوية (سفلية)
!> موحدة، و D هي مصفوفة هيرميتية وقطرية كتلية مع كتل قطرية 1-by-1 و 2-by-2.
!> ثم يتم استخدام الشكل المحلل لـ A لحل نظام المعادلات A * X = B.
!>

المعلمات

UPLO 
!>          UPLO هو CHARACTER*1
!>          = 'U':  يتم تخزين الجزء العلوي من A؛
!>          = 'L':  يتم تخزين الجزء السفلي من A؛
!>

N

!>          N هو INTEGER
!>          عدد المعادلات الخطية، أي ترتيب
!>          المصفوفة A.  N >= 0.
!>

NRHS

!>          NRHS هو INTEGER
!>          عدد الأطراف اليمنى، أي عدد الأعمدة
!>          للمصفوفة B.  NRHS >= 0.
!>

A

!>          A هي مصفوفة COMPLEX، الأبعاد (LDA,N)
!>          عند الإدخال، المصفوفة الهيرميتية A.  إذا كان UPLO = 'U'، فإن الجزء
!>          العلوي المثلثي الرئيسي N-by-N في A يحتوي على الجزء
!>          المثلثي العلوي للمصفوفة A، والجزء السفلي المثلثي الصارم
!>          من A غير مشار إليه.  إذا كان UPLO = 'L'، فإن الجزء
!>          السفلي المثلثي الرئيسي N-by-N في A يحتوي على الجزء
!>          المثلثي السفلي للمصفوفة A، والجزء العلوي المثلثي الصارم
!>          من A غير مشار إليه.
!>
!>          عند الخروج، إذا كان INFO = 0، فإن المصفوفة القطرية الكتلية D
!>          والمضاعفات المستخدمة للحصول على العامل U أو L من
!>          التحليل A = U*D*U**H أو A = L*D*L**H كما تم حسابها بواسطة
!>          CHETRF.
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرئيسي للمصفوفة A.  LDA >= max(1,N).
!>

IPIV

!>          IPIV هي مصفوفة INTEGER، الأبعاد (N)
!>          تفاصيل التبادلات وهيكل الكتلة لـ D، كما
!>          تم تحديده بواسطة CHETRF.  إذا كان IPIV(k) > 0، فسيتم تبديل
!>          الصفوف والأعمدة k و IPIV(k)، و D(k,k) هي كتلة قطرية 1-by-1.
!>          إذا كان UPLO = 'U' و IPIV(k) = IPIV(k-1) < 0،
!>          فقد تم تبديل الصفوف والأعمدة k-1 و -IPIV(k) و
!>          D(k-1:k,k-1:k) هي كتلة قطرية 2-by-2.  إذا كان UPLO = 'L' و
!>          IPIV(k) = IPIV(k+1) < 0، فسيتم تبديل الصفوف والأعمدة k+1 و
!>          -IPIV(k) و D(k:k+1,k:k+1) هي كتلة قطرية 2-by-2.
!>

B

!>          B هي مصفوفة COMPLEX، الأبعاد (LDB,NRHS)
!>          عند الإدخال، مصفوفة الطرف الأيمن B ذات الأبعاد N-by-NRHS.
!>          عند الخروج، إذا كان INFO = 0، فإن مصفوفة الحل X ذات الأبعاد N-by-NRHS.
!>

LDB

!>          LDB هو INTEGER
!>          البعد الرئيسي للمصفوفة B.  LDB >= max(1,N).
!>

WORK

!>          WORK هي مصفوفة COMPLEX، الأبعاد (MAX(1,LWORK))
!>          عند الخروج، إذا كان INFO = 0، فإن WORK(1) تعيد قيمة LWORK المثلى.
!>

LWORK

!>          LWORK هو INTEGER
!>          طول WORK.  LWORK >= 1، ولأفضل أداء
!>          LWORK >= max(1,N*NB)، حيث NB هو الحجم الأمثل للكتلة لـ
!>          CHETRF.
!>          إذا كان LWORK < N، فسيتم إجراء TRS باستخدام المستوى BLAS 2
!>          إذا كان LWORK >= N، فسيتم إجراء TRS باستخدام المستوى BLAS 3
!>
!>          إذا كان LWORK = -1، فسيتم افتراض استعلام مساحة العمل؛ تقوم الروتين
!>          فقط بحساب الحجم الأمثل للمصفوفة WORK، وتعيد
!>          هذه القيمة كالإدخال الأول للمصفوفة WORK، ولا يتم إصدار أي رسالة خطأ
!>          متعلقة بـ LWORK بواسطة XERBLA.
!>

INFO

!>          INFO هو INTEGER
!>          = 0: خروج ناجح
!>          < 0: إذا كان INFO = -i، فقد كانت المعلمة i غير صالحة
!>          > 0: إذا كان INFO = i، فإن D(i,i) يساوي صفرًا تمامًا.  لقد تم إكمال التحليل،
!>               ولكن المصفوفة القطرية الكتلية D
!>               فردية تمامًا، لذلك لم يكن من الممكن حساب الحل.
!>
