subroutine cheequb (character uplo, integer n, complex, dimension( lda, * ) a, integer lda, real, dimension( * ) s, real scond, real amax, complex, dimension( * ) work, integer info)

CHEEQUB

الغرض:

!>
!> CHEEQUB يحسب عوامل قياس الصف والعمود المعدة لموازنة مصفوفة هرميتية A (بالنسبة للمعامل الإقليدي) وتقليل
!> رقم شرطها. يتم حساب عوامل القياس S بواسطة خوارزمية BIN (انظر المراجع) بحيث تكون المصفوفة المقاسة B بعناصر
!> B(i,j) = S(i)*A(i,j)*S(j) لها رقم شرط ضمن عامل N من
!> أصغر رقم شرط ممكن عبر جميع القياسات القطرية الممكنة.
!>

المعلمات

UPLO 
!>          UPLO هو CHARACTER*1
!>          = 'U':  المثلث العلوي من A مخزن؛
!>          = 'L':  المثلث السفلي من A مخزن.
!>

N

!>          N هو INTEGER
!>          رتبة المصفوفة A. N >= 0.
!>

A

!>          A هو مصفوفة COMPLEX، البعد (LDA,N)
!>          المصفوفة الهرميتية N-by-N التي سيتم حساب عوامل القياس الخاصة بها.
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرائد للمصفوفة A. LDA >= max(1,N).
!>

S

!>          S هي مصفوفة REAL، البعد (N)
!>          إذا كان INFO = 0، فإن S تحتوي على عوامل القياس لـ A.
!>

SCOND

!>          SCOND هو REAL
!>          إذا كان INFO = 0، فإن S تحتوي على نسبة أصغر S(i) إلى
!>          أكبر S(i). إذا كان SCOND >= 0.1 وكان AMAX ليس كبيرًا جدًا أو صغيرًا جدًا،
!>          فلا فائدة من القياس بـ S.
!>

AMAX

!>          AMAX هو REAL
!>          أكبر قيمة مطلقة لأي عنصر في المصفوفة. إذا كان AMAX قريبًا جدًا من
!>          الفيضان أو قريبًا جدًا من التدفق السفلي، فيجب قياس المصفوفة.
!>

WORK

!>          WORK هي مصفوفة COMPLEX، البعد (2*N)
!>

INFO

!>          INFO هو INTEGER
!>          = 0:  انتهاء ناجح
!>          < 0:  إذا كان INFO = -i، فإن الوسيط i كان له قيمة غير قانونية
!>          > 0:  إذا كان INFO = i، فإن العنصر القطري i غير موجب.
!>
