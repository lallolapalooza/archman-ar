توثيق الوظيفة/الروتين الفرعي

subroutine cher (character uplo, integer n, real alpha, complex, dimension(*) x, integer incx, complex, dimension(lda,*) a, integer lda)

CHER

الغرض:

!>
!> CHER   تقوم بعملية هيرميتية من الرتبة 1
!>
!>    A := alpha*x*x**H + A,
!>
!> حيث alpha هو ثابت حقيقي، x هو متجه مكون من n عنصر، و A هي
!> مصفوفة هيرميتية بحجم n × n.
!>

المعلمات

UPLO 
!>          UPLO is CHARACTER*1
!>           عند الإدخال، يحدد UPLO ما إذا كان الجزء المثلثي العلوي أو السفلي
!>           من المصفوفة A سيتم الإشارة إليه كما يلي:
!>
!>              UPLO = 'U' or 'u'   سيتم الإشارة فقط إلى الجزء المثلثي العلوي من A
!>                                  .
!>
!>              UPLO = 'L' or 'l'   سيتم الإشارة فقط إلى الجزء المثلثي السفلي من A
!>                                  .
!>

N

!>          N is INTEGER
!>           عند الإدخال، يحدد N رتبة المصفوفة A.
!>           يجب أن تكون N أكبر من أو تساوي الصفر.
!>

ALPHA

!>          ALPHA is REAL
!>           عند الإدخال، يحدد ALPHA الثابت alpha.
!>

X

!>          X is COMPLEX array, dimension at least
!>           ( 1 + ( n - 1 )*abs( INCX ) ).
!>           قبل الإدخال، يجب أن يحتوي مصفوفة X المتزايدة على المتجه x
!>           المكون من n عنصر.
!>

INCX

!>          INCX is INTEGER
!>           عند الإدخال، يحدد INCX الزيادة لعناصر X.
!>           يجب ألا تكون INCX صفرًا.
!>

A

!>          A is COMPLEX array, dimension ( LDA, N )
!>           قبل الإدخال مع  UPLO = 'U' or 'u', يجب أن يحتوي الجزء العلوي
!>           الأمامي بحجم n × n من المصفوفة A على الجزء المثلثي العلوي
!>           للمصفوفة الهيرميتية، ولن تتم الإشارة إلى الجزء المثلثي السفلي
!>           الصارم لـ A. عند الخروج، سيتم الكتابة فوق الجزء المثلثي العلوي
!>           من المصفوفة A بالجزء المثلثي العلوي للمصفوفة المحدثة.
!>           قبل الإدخال مع UPLO = 'L' or 'l', يجب أن يحتوي الجزء السفلي
!>           الأمامي بحجم n × n من المصفوفة A على الجزء المثلثي السفلي
!>           للمصفوفة الهيرميتية، ولن تتم الإشارة إلى الجزء المثلثي العلوي
!>           الصارم لـ A. عند الخروج، سيتم الكتابة فوق الجزء المثلثي السفلي
!>           من المصفوفة A بالجزء المثلثي السفلي للمصفوفة المحدثة.
!>           لاحظ أن الأجزاء التخيلية للعناصر القطرية لا تحتاج إلى تعيين،
!>           يفترض أنها صفر، وعند الخروج يتم تعيينها إلى صفر.
!>

LDA

!>          LDA is INTEGER
!>           عند الإدخال، يحدد LDA البعد الأول لـ A كما تم تعريفه
!>           في البرنامج (الروتين الفرعي) المستدعي. يجب أن يكون LDA
!>           على الأقل  max( 1, n ).
!>
