subroutine chpr2 (character uplo, integer n, complex alpha, complex, dimension(*) x, integer incx, complex, dimension(*) y, integer incy, complex, dimension(*) ap)

CHPR2

الغرض:

!>
!> CHPR2  يقوم بعملية الرتبة 2 الهيرميتية
!>
!>    A := alpha*x*y**H + conjg( alpha )*y*x**H + A,
!>
!> حيث alpha هو عدد قياسي، x و y هما متجهات من n عنصر، و A هي مصفوفة
!> هيرميتية بحجم n × n، مقدمة في شكل مضغوط.
!>

المعلمات

UPLO 
!>          UPLO هو CHARACTER*1
!>           عند الإدخال، يحدد UPLO ما إذا كان الجزء المثلثي العلوي أو السفلي
!>           للمصفوفة A مقدمًا في مصفوفة AP كما يلي:
!>
!>              UPLO = 'U' أو 'u'   يتم تقديم الجزء المثلثي العلوي من A
!>                                  في AP.
!>
!>              UPLO = 'L' أو 'l'   يتم تقديم الجزء المثلثي السفلي من A
!>                                  في AP.
!>

N

!>          N هو INTEGER
!>           عند الإدخال، يحدد N رتبة المصفوفة A.
!>           يجب أن يكون N أكبر من أو يساوي صفر.
!>

ALPHA

!>          ALPHA هو COMPLEX
!>           عند الإدخال، يحدد ALPHA العدد القياسي alpha.
!>

X

!>          X هي مصفوفة COMPLEX، بحجم لا يقل عن
!>           ( 1 + ( n - 1 )*abs( INCX ) ).
!>           قبل الإدخال، يجب أن تحتوي المصفوفة X المعززة على متجه x
!>           المكون من n عنصر.
!>

INCX

!>          INCX هو INTEGER
!>           عند الإدخال، يحدد INCX الزيادة لعناصر X.
!>           يجب ألا يكون INCX صفرًا.
!>

Y

!>          Y هي مصفوفة COMPLEX، بحجم لا يقل عن
!>           ( 1 + ( n - 1 )*abs( INCY ) ).
!>           قبل الإدخال، يجب أن تحتوي المصفوفة Y المعززة على متجه y
!>           المكون من n عنصر.
!>

INCY

!>          INCY هو INTEGER
!>           عند الإدخال، يحدد INCY الزيادة لعناصر Y.
!>           يجب ألا يكون INCY صفرًا.
!>

AP

!>          AP هي مصفوفة COMPLEX، بحجم لا يقل عن
!>           ( ( n*( n + 1 ) )/2 ).
!>           قبل الإدخال مع UPLO = 'U' أو 'u'، يجب أن تحتوي المصفوفة AP
!>           على الجزء المثلثي العلوي للمصفوفة الهيرميتية مضغوطًا بشكل متسلسل،
!>           عمودًا بعد عمود، بحيث تحتوي AP( 1 ) على a( 1, 1 )، وتحتوي AP( 2 )
!>           و AP( 3 ) على a( 1, 2 ) و a( 2, 2 ) على التوالي، وهكذا. عند الخروج،
!>           يتم الكتابة فوق المصفوفة AP بالجزء المثلثي العلوي للمصفوفة المحدثة.
!>           قبل الإدخال مع UPLO = 'L' أو 'l'، يجب أن تحتوي المصفوفة AP
!>           على الجزء المثلثي السفلي للمصفوفة الهيرميتية مضغوطًا بشكل متسلسل،
!>           عمودًا بعد عمود، بحيث تحتوي AP( 1 ) على a( 1, 1 )، وتحتوي AP( 2 )
!>           و AP( 3 ) على a( 2, 1 ) و a( 3, 1 ) على التوالي، وهكذا. عند الخروج،
!>           يتم الكتابة فوق المصفوفة AP بالجزء المثلثي السفلي للمصفوفة المحدثة.
!>           لاحظ أن الأجزاء التخيلية للعناصر القطرية لا تحتاج إلى تعيينها،
!>           ويفترض أنها صفر، وعند الخروج يتم تعيينها إلى صفر.
!>
