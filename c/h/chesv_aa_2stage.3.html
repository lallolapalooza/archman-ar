subroutine chesv_aa_2stage (character uplo, integer n, integer nrhs, complex, dimension( lda, * ) a, integer lda, complex, dimension( * ) tb, integer ltb, integer, dimension( * ) ipiv, integer, dimension( * ) ipiv2, complex, dimension( ldb, * ) b, integer ldb, complex, dimension( * ) work, integer lwork, integer info)

CHESV_AA_2STAGE تحسب حل نظام المعادلات الخطية A * X = B للمصفوفات HE

الغرض:

!>
!> CHESV_AA_2STAGE تحسب حل نظام خطي معقد
!>    A * X = B,
!> حيث A مصفوفة هرميتية N-by-N و X و B مصفوفتان N-by-NRHS.
!>
!> يتم استخدام خوارزمية Aasen's ذات المرحلتين لتحليل A إلى
!>    A = U**H * T * U,  إذا كان UPLO = 'U', أو
!>    A = L * T * L**H,  إذا كان UPLO = 'L',
!> حيث U (أو L) هو ناتج تبديلات ومصفوفات مثلثية علوية (سفلية) وحدوية، و T
!> هرميتية ونطاقية. ثم يتم تحليل T باستخدام LU مع مبادلة جزئية. ثم يتم
!> استخدام الشكل المحلل لـ A لحل نظام المعادلات A * X = B.
!>
!> هذا هو الإصدار المجمع من الخوارزمية، والذي يستدعي BLAS المستوى 3.
!>

المعلمات

UPLO 
!>          UPLO is CHARACTER*1
!>          = 'U':  يتم تخزين المثلث العلوي لـ A؛
!>          = 'L':  يتم تخزين المثلث السفلي لـ A؛
!>

N

!>          N is INTEGER
!>          رتبة المصفوفة A.  N >= 0.
!>

NRHS

!>          NRHS is INTEGER
!>          عدد الأطراف اليمنى، أي عدد أعمدة المصفوفة B.  NRHS >= 0.
!>

A

!>          A is COMPLEX array, dimension (LDA,N)
!>          عند الإدخال، المصفوفة الهرميتية A.  إذا كان UPLO = 'U'، فإن الجزء
!>          المثلثي العلوي N-by-N الرائد للمصفوفة A يحتوي على الجزء
!>          المثلثي العلوي للمصفوفة A، ولا يتم الرجوع إلى الجزء المثلثي
!>          السفلي الصارم للمصفوفة A.  إذا كان UPLO = 'L'، فإن الجزء
!>          المثلثي السفلي N-by-N الرائد للمصفوفة A يحتوي على الجزء
!>          المثلثي السفلي للمصفوفة A، ولا يتم الرجوع إلى الجزء المثلثي
!>          العلوي الصارم للمصفوفة A.
!>
!>          عند الخروج، يتم تخزين L أسفل (أو أعلى) كتل تحت القطر،
!>          عندما يكون UPLO  هو 'L' (أو 'U').
!>

LDA

!>          LDA is INTEGER
!>          البعد الرائد للمصفوفة A.  LDA >= max(1,N).
!>

TB

!>          TB is COMPLEX array, dimension (MAX(1,LTB)).
!>          عند الخروج، تفاصيل تحليل LU للمصفوفة النطاقية.
!>

LTB

!>          LTB is INTEGER
!>          حجم المصفوفة TB. LTB >= MAX(1,4*N)، يستخدم داخليًا
!>          لتحديد NB بحيث يكون LTB >= (3*NB+1)*N.
!>
!>          إذا كان LTB = -1، فسيتم افتراض استعلام مساحة عمل؛
!>          يقوم الإجراء فقط بحساب الحجم الأمثل لـ LTB،
!>          ويعيد هذا الحجم كإدخال أول لـ TB،
!>          ولا يتم إصدار أي رسالة خطأ متعلقة بـ LTB بواسطة XERBLA.
!>

IPIV

!>          IPIV is INTEGER array, dimension (N)
!>          عند الخروج، يحتوي على تفاصيل التبادلات، أي
!>          تم تبديل الصف والعمود k من A مع
!>          الصف والعمود IPIV(k).
!>

IPIV2

!>          IPIV2 is INTEGER array, dimension (N)
!>          عند الخروج، يحتوي على تفاصيل التبادلات، أي
!>          تم تبديل الصف والعمود k من T مع
!>          الصف والعمود IPIV(k).
!>

B

!>          B is COMPLEX array, dimension (LDB,NRHS)
!>          عند الإدخال، مصفوفة الطرف الأيمن B.
!>          عند الخروج، مصفوفة الحل X.
!>

LDB

!>          LDB is INTEGER
!>          البعد الرائد للمصفوفة B.  LDB >= max(1,N).
!>

WORK

!>          WORK is COMPLEX workspace of size (MAX(1,LWORK))
!>          عند الخروج، إذا كان INFO = 0، فإن WORK(1) يعيد LWORK الأمثل.
!>

LWORK

!>          LWORK is INTEGER
!>          حجم WORK. LWORK >= MAX(1,N)، يستخدم داخليًا
!>          لتحديد NB بحيث يكون LWORK >= N*NB.
!>
!>          إذا كان LWORK = -1، فسيتم افتراض استعلام مساحة عمل؛
!>          يقوم الإجراء فقط بحساب الحجم الأمثل لمصفوفة WORK،
!>          ويعيد هذا الحجم كإدخال أول لمصفوفة WORK،
!>          ولا يتم إصدار أي رسالة خطأ متعلقة بـ LWORK بواسطة XERBLA.
!>

INFO

!>          INFO is INTEGER
!>          = 0:  خروج ناجح
!>          < 0:  إذا كان INFO = -i، فإن الوسيط i كان له قيمة غير قانونية.
!>          > 0:  إذا كان INFO = i، فشل تحليل LU النطاقي في العمود i.
!>
