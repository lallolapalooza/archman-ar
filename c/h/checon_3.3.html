subroutine checon_3 (character uplo, integer n, complex, dimension( lda, * ) a, integer lda, complex, dimension( * ) e, integer, dimension( * ) ipiv, real anorm, real rcond, complex, dimension( * ) work, integer info)

CHECON_3

الغرض:

!> CHECON_3 يقدر مقلوب رقم الشرط (في المعيار 1) لمصفوفة هرمية معقدة A باستخدام التحليل
!> الذي تم حسابه بواسطة CHETRF_RK أو CHETRF_BK:
!>
!>    A = P*U*D*(U**H)*(P**T) أو A = P*L*D*(L**H)*(P**T),
!>
!> حيث U (أو L) هي مصفوفة قطرية علوية (أو سفلية) موحدة،
!> U**H (أو L**H) هي المرافق لـ U (أو L)، P هي مصفوفة تبديل،
!> P**T هي منقول P، و D هي هرمية وكتلة قطرية ذات كتل قطرية 1x1 و 2x2.
!>
!> يتم الحصول على تقدير لـ norm(inv(A))، ويتم حساب مقلوب رقم الشرط كـ
!> RCOND = 1 / (ANORM * norm(inv(A))).
!> تستخدم هذه الإجراءات الفرعية محلل BLAS3 CHETRS_3.
!>

المعلمات

UPLO 
!>          UPLO هو CHARACTER*1
!>          يحدد ما إذا كانت تفاصيل التحليل مخزنة كمصفوفة مثلثة علوية أو سفلية:
!>          = 'U':  مثلثة علوية، الشكل هو A = P*U*D*(U**H)*(P**T);
!>          = 'L':  مثلثة سفلية، الشكل هو A = P*L*D*(L**H)*(P**T).
!>

N

!>          N هو INTEGER
!>          رتبة المصفوفة A.  N >= 0.
!>

A

!>          A هي مصفوفة COMPLEX، ذات أبعاد (LDA,N)
!>          القطر الرئيسي للمصفوفة القطرية الكتلة D والعوامل U أو L
!>          كما تم حسابها بواسطة CHETRF_RK و CHETRF_BK:
!>            أ) فقط العناصر القطرية الرئيسية للمصفوفة القطرية الكتلة الهرمية
!>               D على القطر الرئيسي لـ A، أي D(k,k) = A(k,k);
!>               (يجب توفير العناصر فوق القطر الرئيسي (أو تحته) لـ D
!>                عند الدخول في المصفوفة E)، و
!>            ب) إذا كان UPLO = 'U': العامل U في الجزء فوق القطر الرئيسي لـ A.
!>               إذا كان UPLO = 'L': العامل L في الجزء تحت القطر الرئيسي لـ A.
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرئيسي للمصفوفة A.  LDA >= max(1,N).
!>

E

!>          E هي مصفوفة COMPLEX، ذات أبعاد (N)
!>          عند الدخول، تحتوي على العناصر فوق القطر الرئيسي (أو تحته)
!>          للمصفوفة القطرية الكتلة الهرمية D
!>          ذات الكتل القطرية 1x1 أو 2x2، حيث
!>          إذا كان UPLO = 'U': E(i) = D(i-1,i)، i=2:N، E(1) غير مستدعاة؛
!>          إذا كان UPLO = 'L': E(i) = D(i+1,i)، i=1:N-1، E(N) غير مستدعاة.
!>
!>          ملاحظة: بالنسبة لكتلة قطرية 1x1 D(k)، حيث
!>          1 <= k <= N، لا يتم استدعاء العنصر E(k) في كلتا
!>          الحالتين UPLO = 'U' أو UPLO = 'L'.
!>

IPIV

!>          IPIV هي مصفوفة INTEGER، ذات أبعاد (N)
!>          تفاصيل التبادلات وهيكل الكتلة لـ D
!>          كما تم تحديده بواسطة CHETRF_RK أو CHETRF_BK.
!>

ANORM

!>          ANORM هو REAL
!>          المعيار 1 للمصفوفة الأصلية A.
!>

RCOND

!>          RCOND هو REAL
!>          مقلوب رقم شرط المصفوفة A،
!>          محسوب كـ RCOND = 1/(ANORM * AINVNM)، حيث AINVNM هو
!>          تقدير للمعيار 1 لـ inv(A) الذي تم حسابه في هذه الإجراءات الفرعية.
!>

WORK

!>          WORK هي مصفوفة COMPLEX، ذات أبعاد (2*N)
!>

INFO

!>          INFO هو INTEGER
!>          = 0:  خروج ناجح
!>          < 0:  إذا كان INFO = -i، فإن المعلمة i كانت ذات قيمة غير قانونية
!>
