__extern int caca_put_char (caca_canvas_t * cv, int x, int y, uint32_t ch)

اطبع حرف ASCII أو Unicode عند الإحداثيات المحددة، باستخدام قيم الألوان الأمامية والخلفية الافتراضية.

إذا كانت الإحداثيات خارج حدود اللوحة، فلن يتم طباعة شيء. إذا تم الكتابة فوق حرف Unicode بعرض كامل، فسيتم استبدال الأجزاء المرئية المتبقية بمسافات. إذا كانت حدود اللوحة ستقسم الحرف ذو العرض الكامل إلى قسمين، فسيتم طباعة مسافة بدلاً منه.

السلوك عند طباعة أحرف غير قابلة للطباعة أو أحرف UTF-32 غير صالحة غير محدد. لطباعة تسلسل بايتات يشكل حرف UTF-8 بدلاً من حرف UTF-32، استخدم الدالة caca_put_str().

ترجع هذه الدالة عرض الحرف المطبوع. إذا كان حرفًا بعرض كامل، فسيتم إرجاع 2. بخلاف ذلك، سيتم إرجاع 1.

هذه الدالة لا تفشل أبداً.

المعاملات

القيمة المرجعة

عرض الحرف المطبوع: 2 لحرف بعرض كامل، 1 بخلاف ذلك.

تشير إلى caca_add_dirty_rect()، CACA_MAGIC_FULLWIDTH، و caca_utf32_is_fullwidth().

مشار إليها بواسطة caca_conio_cgets()، caca_conio_cputs()، caca_conio_putch()، caca_dither_bitmap()، caca_fill_box()، caca_fill_triangle()، caca_flush_figlet()، caca_put_figchar()، و caca_put_str().

__extern uint32_t caca_get_char (caca_canvas_t const * cv, int x, int y)

الحصول على قيمة ASCII أو Unicode للحرف عند الإحداثيات المحددة. إذا كانت القيمة أقل من أو تساوي 127 (0x7f)، يمكن طباعة الحرف كـ ASCII. خلاف ذلك، يجب التعامل معه كقيمة UTF-32.

إذا كانت الإحداثيات خارج حدود اللوحة، فسيتم إرجاع مسافة (0x20).

استثناء خاص هو عند إرجاع CACA_MAGIC_FULLWIDTH. هذه القيمة مضمونة ألا تكون حرف Unicode صالحًا، وتشير إلى أن الحرف الموجود على يسار الحرف المطلوب هو حرف ذو عرض كامل.

هذه الدالة لا تفشل أبداً.

المعاملات

القيمة المرجعة

حرف Unicode عند الإحداثيات المحددة.

مشار إليها بواسطة caca_flush_figlet()، و caca_put_figchar().

__extern int caca_put_str (caca_canvas_t * cv, int x, int y, char const * str)

اطبع سلسلة نصية UTF-8 عند الإحداثيات المحددة، باستخدام افتراضي الألوان الأمامية والخلفية.

إذا كانت الإحداثيات خارج حدود اللوحة، فلن يتم طباعة شيء. إذا تم الكتابة فوق حرف بعرض كامل، فسيتم استبدال الأجزاء المرئية المتبقية بمسافات. إذا كانت حدود اللوحة ستقسم الحرف ذو العرض الكامل إلى قسمين، فسيتم طباعة مسافة بدلاً منه.

يتم دفع المؤشر إلى الأمام بمقدار عرض السلسلة النصية المطبوعة. يمكن أن يكون هذا العرض أكبر من عدد أحرف السلسلة النصية إذا كانت تحتوي على أحرف بعرض كامل.

هذه الدالة لا تفشل أبداً.

x إحداثي X.

القيمة المرجعة

عرض السلسلة النصية المطبوعة.

تشير إلى caca_add_dirty_rect()، CACA_MAGIC_FULLWIDTH، caca_utf32_to_utf8()، caca_utf8_to_utf32()، و caca_utf8_width().

مشار إليها بواسطة caca_conio_cputs()، caca_conio_printf()، caca_display_menu()، caca_dither_bitmap()، caca_fill_box()، caca_fill_triangle()، caca_flush_figlet()، caca_put_figstr()، و caca_printf().
