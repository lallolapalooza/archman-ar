cap_to_text(3)

الحزمة: libcap

التراخيص: BSD-3-Clause OR GPL-2.0-only

CAP_FROM_TEXT(3) دليل مبرمج Linux CAP_FROM_TEXT(3)

الاسم

cap_from_text, cap_to_text, cap_to_name, cap_from_name - تمثيل الحالة النصية للقدرة ترجمة

الملخص

#include <sys/capability.h>
cap_t cap_from_text(const char *buf_p);
char *cap_to_text(cap_t caps, ssize_t *len_p);
int cap_from_name(const char *name, cap_value_t *cap_p);
char *cap_to_name(cap_value_t cap);

اربط بـ -lcap.

الوصف

تقوم هذه الدوال بترجمة حالة القدرة بين تمثيل داخلي وتمثيل نصي. التمثيل الداخلي تتم إدارته بواسطة دوال القدرة في مساحة العمل. التمثيل النصي هو سلسلة نصية منظمة، قابلة للقراءة من قبل الإنسان، ومناسبة للعرض.

cap_from_text() تقوم بتخصيص وتهيئة حالة قدرة في مساحة العمل. ثم تقوم بتعيين محتويات حالة القدرة المنشأة حديثًا هذه إلى الحالة الممثلة بسلسلة نصية حرفية قابلة للقراءة من قبل الإنسان، ومنتهية بـ nul، يشير إليها buf_p. تعيد مؤشرًا إلى حالة القدرة المنشأة حديثًا. عندما لا تعود حالة القدرة في مساحة العمل مطلوبة، يجب على المستدعي تحرير أي ذاكرة قابلة للتحرير عن طريق استدعاء cap_free() مع cap_t كوسيط. تعيد الدالة خطأ إذا لم تتمكن من تحليل محتويات السلسلة المشار إليها بواسطة buf_p أو لم تتعرف على أي capability_name أو حرف علامة كصالح. تعيد الدالة أيضًا خطأ إذا تم تعيين أي علامة وإلغاؤها في جملة واحدة.

cap_to_text() تقوم بتحويل حالة القدرة في مساحة العمل المحددة بواسطة caps إلى سلسلة نصية حرفية قابلة للقراءة من قبل الإنسان ومنتهية بـ nul. تقوم هذه الدالة بتخصيص أي ذاكرة ضرورية لاحتواء السلسلة، وتعيد مؤشرًا إلى السلسلة. إذا لم يكن المؤشر len_p هو NULL، فيجب على الدالة أيضًا إعادة الطول الكامل للسلسلة (دون تضمين الـ nul المُنهي) في الموقع المشار إليه بواسطة len_p. حالة القدرة في مساحة العمل، المحددة بواسطة caps، ممثلة بالكامل في السلسلة الحرفية. عندما لا تعود حالة القدرة في مساحة العمل مطلوبة، يجب على المستدعي تحرير أي ذاكرة قابلة للتحرير عن طريق استدعاء cap_free() مع مؤشر السلسلة المُعاد كوسيط.

cap_from_name() تقوم بتحويل تمثيل نصي للقدرة، مثل "cap_chown"، إلى تمثيلها الرقمي (CAP_CHOWN=0)، وكتابة القيمة المُفككة في *cap_p. إذا كان cap_p هو NULL، فلا يتم كتابة نتيجة، ولكن رمز الإعادة للدالة يشير إلى ما إذا كانت القدرة المحددة يمكن تمثيلها بواسطة المكتبة أم لا.

cap_to_name() تقوم بتحويل قيمة فهرس القدرة، cap، إلى سلسلة نصية مخصصة بواسطة libcap. يجب إلغاء تخصيص هذه السلسلة باستخدام cap_free().

التمثيل النصي

تم وصف التنسيق النصي في صفحة الدليل cap_text_formats(7).

قيمة الإعادة

cap_from_text() و cap_to_text() و cap_to_name() تعيد قيمة غير NULL عند النجاح، و NULL عند الفشل. cap_from_name() تعيد 0 للنجاح، و -1 عند الفشل (قدرة غير معروفة).

عند الفشل، يتم تعيين errno إلى EINVAL، أو ENOMEM.

التوافق

cap_from_text() و cap_to_text() محددة بواسطة مسودة مواصفات POSIX.1e المسحوبة. cap_from_name() و cap_to_name() هي امتدادات Linux.

مثال

يوضح برنامج المثال أدناه استخدام cap_from_text() و cap_to_text(). تظهر جلسة shell التالية بعض عمليات التشغيل النموذجية:

$ ./a.out "cap_chown=p cap_chown+e"
caps_to_text() returned "cap_chown=ep"
$ ./a.out "all=pe cap_chown-e cap_kill-pe"
caps_to_text() returned "=ep cap_chown-e cap_kill-ep"

رمز البرنامج المصدر هو كما يلي:

#include <stdlib.h>
#include <stdio.h>
#include <sys/capability.h>
#define handle_error(msg) \
    do { perror(msg); exit(EXIT_FAILURE); } while (0)
int
main(int argc, char *argv[])
{
    cap_t caps;
    char *txt_caps;
    if (argc != 2) {
        fprintf(stderr, "%s <textual-cap-set>\n", argv[0]);
        exit(EXIT_FAILURE);
    }
    caps = cap_from_text(argv[1]);
    if (caps == NULL)
        handle_error("cap_from_text");
    txt_caps = cap_to_text(caps, NULL);
    if (txt_caps == NULL)
        handle_error("cap_to_text");
    printf("caps_to_text() returned \"%s\"\n", txt_caps);
    if (cap_free(txt_caps) != 0 || cap_free(caps) != 0)
        handle_error("cap_free");
    exit(EXIT_SUCCESS);
}
