NAME

caca_vprintf - print formatted output to a canvas

SYNOPSIS


#include <cacard.h>

int caca_vprintf(caca_canvas_t *canvas, int x, int y, const char *format, va_list ap);

DESCRIPTION

The caca_vprintf() function is similar to the standard library's vprintf function. It takes a format string, a list of arguments, and prints the resulting string to the specified canvas, at the given coordinates (x, y).

The format string supports the standard printf format specifiers, as well as a few extensions specific to libcaca:

  • %b: Prints a block of characters. The argument should be a caca_ruby_t.
  • %c: Prints a single character.
  • %d, %i: Prints a signed decimal integer.
  • %f: Prints a floating-point number.
  • %o: Prints an unsigned octal integer.
  • %s: Prints a string.
  • %u: Prints an unsigned decimal integer.
  • %x, %X: Prints an unsigned hexadecimal integer.
  • %%: Prints a literal '%'.

The coordinates (x, y) specify the top-left corner where the text will be rendered. The text is drawn using the current graphic style of the canvas.

RETURN VALUE

On success, caca_vprintf() returns the number of characters written. On error, it returns a negative value.

EXAMPLE


#include <cacard.h>
#include <stdarg.h>

int main(void) {
    caca_canvas_t *canvas;
    canvas = caca_create_canvas(80, 24);
    if (canvas == NULL) {
        return 1;
    }

    caca_string_t str = caca_create_string("Hello, world!");
    caca_vprintf(canvas, 10, 5, "Message: %s\nLine 2: %d", 123);
    caca_free_string(str);

    caca_display_t *display = caca_create_display(canvas);
    caca_refresh_display(display);
    // ... wait for user input ...
    caca_free_display(display);
    caca_free_canvas(canvas);
    return 0;
}

الاسم

caca_vprintf - طباعة مخرجات منسقة إلى لوحة رسم

الملخص


#include <cacard.h>

int caca_vprintf(caca_canvas_t *canvas, int x, int y, const char *format, va_list ap);

الوصف

تشبه الدالة caca_vprintf() الدالة vprintf القياسية في المكتبة. إنها تأخذ سلسلة تنسيق، وقائمة من الوسائط، وتطبع السلسلة الناتجة إلى اللوحة المحددة، عند الإحداثيات المعطاة (x, y).

تدعم سلسلة التنسيق مواصفات التنسيق القياسية لـ printf، بالإضافة إلى بعض الإضافات الخاصة بـ libcaca:

  • %b: يطبع كتلة من الأحرف. يجب أن تكون الوسيطة caca_ruby_t.
  • %c: يطبع حرفًا واحدًا.
  • %d, %i: يطبع عددًا صحيحًا عشريًا موقعًا.
  • %f: يطبع رقمًا عشريًا.
  • %o: يطبع عددًا صحيحًا ثمانيًا غير موقع.
  • %s: يطبع سلسلة نصية.
  • %u: يطبع عددًا صحيحًا عشريًا غير موقع.
  • %x, %X: يطبع عددًا صحيحًا ست عشريًا غير موقع.
  • %%: يطبع حرف '%' حرفيًا.

تحدد الإحداثيات (x, y) الزاوية العلوية اليسرى حيث سيتم عرض النص. يتم رسم النص باستخدام النمط الرسومي الحالي للوحة.

قيمة الإرجاع

عند النجاح، تعيد الدالة caca_vprintf() عدد الأحرف المكتوبة. عند حدوث خطأ، تعيد قيمة سالبة.

مثال


#include <cacard.h>
#include <stdarg.h>

int main(void) {
    caca_canvas_t *canvas;
    canvas = caca_create_canvas(80, 24);
    if (canvas == NULL) {
        return 1;
    }

    caca_string_t str = caca_create_string("Hello, world!");
    caca_vprintf(canvas, 10, 5, "Message: %s\nLine 2: %d", 123);
    caca_free_string(str);

    caca_display_t *display = caca_create_display(canvas);
    caca_refresh_display(display);
    // ... انتظر إدخال المستخدم ...
    caca_free_display(display);
    caca_free_canvas(canvas);
    return 0;
}
