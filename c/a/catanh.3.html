catanh(3) دليل وظائف المكتبة catanh(3)

الاسم

catanh, catanhf, catanhl - ظل الزاوية القطعي المعقد

المكتبة

مكتبة الرياضيات (libm, -lm)

الملخص

#include <complex.h>
double complex catanh(double complex z);
float complex catanhf(float complex z);
long double complex catanhl(long double complex z);

الوصف

تحسب هذه الدوال ظل الزاوية القطعي المعقد للمتغير z. إذا كان y = catanh(z)، فإن z = ctanh(y). الجزء التخيلي من y مختار في الفترة [-pi/2,pi/2].

يكون لدينا:


catanh(z) = 0.5 * (clog(1 + z) - clog(1 - z))

الخصائص

لشرح المصطلحات المستخدمة في هذا القسم، انظر attributes(7).

الواجهة الخاصية القيمة
catanh (), catanhf (), catanhl () سلامة الخيوط (Thread safety) آمنة للخيوط (MT-Safe)

المعايير

C11، POSIX.1-2008.

التاريخ

glibc 2.1. C99، POSIX.1-2001.

أمثلة

/* اربط بـ "-lm" */
#include <complex.h>
#include <stdio.h>
#include <stdlib.h>
#include <unistd.h>
int
main(int argc, char *argv[])
{
    double complex z, c, f;
    if (argc != 3) {
        fprintf(stderr, "Usage: %s <real> <imag>\n", argv[0]);
        exit(EXIT_FAILURE);
    }
    z = atof(argv[1]) + atof(argv[2]) * I;
    c = catanh(z);
    printf("catanh() = %6.3f %6.3f*i\n", creal(c), cimag(c));
    f = 0.5 * (clog(1 + z) - clog(1 - z));
    printf("formula  = %6.3f %6.3f*i\n", creal(f), cimag(f));
    exit(EXIT_SUCCESS);
}
