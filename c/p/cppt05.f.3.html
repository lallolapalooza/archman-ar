subroutine cppt05 (character uplo, integer n, integer nrhs, complex, dimension( * ) ap, complex, dimension( ldb, * ) b, integer ldb, complex, dimension( ldx, * ) x, integer ldx, complex, dimension( ldxact, * ) xact, integer ldxact, real, dimension( * ) ferr, real, dimension( * ) berr, real, dimension( * ) reslts)

CPPT05

الغرض:

!>
!> CPPT05 يختبر حدود الخطأ من التحسين التكراري للحل المحسوب لنظام المعادلات A*X = B،
!> حيث A مصفوفة هرميتية بصيغة تخزين مكدسة.
!>
!> RESLTS(1) = اختبار حد الخطأ
!>           = norm(X - XACT) / ( norm(X) * FERR )
!>
!> يتم إرجاع قيمة كبيرة إذا كانت هذه النسبة ليست أقل من واحد.
!>
!> RESLTS(2) = الباقي من روتين التحسين التكراري
!>           = الحد الأقصى لـ BERR / ( (n+1)*EPS + (*) )، حيث
!>             (*) = (n+1)*UNFL / (min_i (abs(A)*abs(X) +abs(b))_i )
!>

المعلمات

UPLO 
!>          UPLO هو CHARACTER*1
!>          يحدد ما إذا كان الجزء العلوي أو السفلي المثلثي للمصفوفة الهرميتية A مخزنًا.
!>          = 'U':  مثلثي علوي
!>          = 'L':  مثلثي سفلي
!>

N

!>          N هو INTEGER
!>          عدد صفوف المصفوفات X، B، و XACT، ورتبة المصفوفة A.  N >= 0.
!>

NRHS

!>          NRHS هو INTEGER
!>          عدد أعمدة المصفوفات X، B، و XACT.
!>          NRHS >= 0.
!>

AP

!>          AP هي مصفوفة COMPLEX، ذات أبعاد (N*(N+1)/2)
!>          الجزء العلوي أو السفلي المثلثي للمصفوفة الهرميتية A، مكدس
!>          عموديًا في مصفوفة خطية.  يتم تخزين العمود j من A
!>          في المصفوفة AP كالتالي:
!>          إذا كان UPLO = 'U'، AP(i + (j-1)*j/2) = A(i,j) لـ 1<=i<=j؛
!>          إذا كان UPLO = 'L'، AP(i + (j-1)*(2n-j)/2) = A(i,j) لـ j<=i<=n.
!>

B

!>          B هي مصفوفة COMPLEX، ذات أبعاد (LDB,NRHS)
!>          متجهات الطرف الأيمن لنظام المعادلات الخطية.
!>

LDB

!>          LDB هو INTEGER
!>          البعد الرئيسي للمصفوفة B.  LDB >= max(1,N).
!>

X

!>          X هي مصفوفة COMPLEX، ذات أبعاد (LDX,NRHS)
!>          متجهات الحل المحسوبة.  يتم تخزين كل متجه كعمود في المصفوفة X.
!>

LDX

!>          LDX هو INTEGER
!>          البعد الرئيسي للمصفوفة X.  LDX >= max(1,N).
!>

XACT

!>          XACT هي مصفوفة COMPLEX، ذات أبعاد (LDX,NRHS)
!>          متجهات الحل الدقيقة.  يتم تخزين كل متجه كعمود في المصفوفة XACT.
!>

LDXACT

!>          LDXACT هو INTEGER
!>          البعد الرئيسي للمصفوفة XACT.  LDXACT >= max(1,N).
!>

FERR

!>          FERR هي مصفوفة REAL، ذات أبعاد (NRHS)
!>          حدود الخطأ الأمامي المقدرة لكل متجه حل X.  إذا كان XTRUE هو الحل الحقيقي،
!>          فإن FERR يحدد حجم أكبر مدخل في (X - XTRUE) مقسومًا على حجم أكبر مدخل في X.
!>

BERR

!>          BERR هي مصفوفة REAL، ذات أبعاد (NRHS)
!>          الخطأ الخلفي النسبي المكون لكل متجه حل (أي، أصغر تغيير نسبي في أي مدخل
!>          في A أو B يجعل X حلاً دقيقًا).
!>

RESLTS

!>          RESLTS هي مصفوفة REAL، ذات أبعاد (2)
!>          الحد الأقصى عبر متجهات الحل NRHS للنسب التالية:
!>          RESLTS(1) = norm(X - XACT) / ( norm(X) * FERR )
!>          RESLTS(2) = BERR / ( (n+1)*EPS + (*) )
!>
