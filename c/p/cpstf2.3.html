subroutine cpstf2 (character uplo, integer n, complex, dimension( lda, * ) a, integer lda, integer, dimension( n ) piv, integer rank, real tol, real, dimension( 2*n ) work, integer info)

CPSTF2 تحسب تحليل تشولسكي مع محورية كاملة لمصفوفة هيرميتية موجبة شبه محددة معقدة.

الغرض:

!>
!> CPSTF2 تحسب تحليل تشولسكي مع محورية كاملة
!> لمصفوفة هيرميتية موجبة شبه محددة A.
!>
!> التحليل له الشكل
!>    P**T * A * P = U**H * U ,  إذا كان UPLO = 'U',
!>    P**T * A * P = L  * L**H,  إذا كان UPLO = 'L',
!> حيث U مصفوفة مثلثية عليا و L مصفوفة مثلثية سفلى، و
!> P مخزنة كمتجه PIV.
!>
!> هذه الخوارزمية لا تحاول التحقق من أن A موجبة
!> شبه محددة. هذا الإصدار من الخوارزمية يستدعي BLAS المستوى 2.
!>

المعاملات

UPLO 
!>          UPLO هو CHARACTER*1
!>          يحدد ما إذا كان الجزء المثلثي العلوي أو السفلي من
!>          المصفوفة المتماثلة A مخزنًا.
!>          = 'U':  مثلثي علوي
!>          = 'L':  مثلثي سفلي
!>

N

!>          N هو INTEGER
!>          رتبة المصفوفة A.  N >= 0.
!>

A

!>          A هو مصفوفة COMPLEX، الأبعاد (LDA,N)
!>          عند الدخول، المصفوفة المتماثلة A.  إذا كان UPLO = 'U'، فإن الجزء
!>          المثلثي العلوي بحجم n × n في A يحتوي على الجزء
!>          المثلثي العلوي للمصفوفة A، ولا يتم الإشارة إلى الجزء السفلي
!>          المثلثي الصارم لـ A.  إذا كان UPLO = 'L'، فإن الجزء
!>          المثلثي السفلي بحجم n × n في A يحتوي على الجزء
!>          المثلثي السفلي للمصفوفة A، ولا يتم الإشارة إلى الجزء العلوي
!>          المثلثي الصارم لـ A.
!>
!>          عند الخروج، إذا كان INFO = 0، فإن العامل U أو L من تحليل تشولسكي
!>          كما هو مذكور أعلاه.
!>

PIV

!>          PIV هو مصفوفة INTEGER، الأبعاد (N)
!>          PIV بحيث تكون المدخلات غير الصفرية هي P( PIV(K), K ) = 1.
!>

RANK

!>          RANK هو INTEGER
!>          رتبة A معطاة بعدد الخطوات التي أكملتها الخوارزمية.
!>

TOL

!>          TOL هو REAL
!>          تفاوت معرف من قبل المستخدم. إذا كان TOL < 0، فسيتم استخدام N*U*MAX( A( K,K ) ).
!>          تتوقف الخوارزمية عند الخطوة (K-1) إذا كانت المحورية <= TOL.
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرائد للمصفوفة A.  LDA >= max(1,N).
!>

WORK

!>          WORK هو مصفوفة REAL، الأبعاد (2*N)
!>          مساحة عمل.
!>

INFO

!>          INFO هو INTEGER
!>          < 0: إذا كان INFO = -K، فإن المعامل K كان له قيمة غير صالحة،
!>          = 0: اكتملت الخوارزمية بنجاح، و
!>          > 0: إما أن المصفوفة A ذات رتبة ناقصة مع رتبة محسوبة
!>               كما تم إرجاعها في RANK، أو أنها ليست موجبة شبه محددة. انظر
!>               القسم 7 من مذكرة عمل LAPACK رقم 161 لمزيد من
!>               المعلومات.
!>
