subroutine cpoequb (integer n, complex, dimension( lda, * ) a, integer lda, real, dimension( * ) s, real scond, real amax, integer info)

CPOEQUB

الغرض:

!>
!> CPOEQUB يحسب قياسات الصفوف والأعمدة المخصصة لمعادلة مصفوفة A الهرمية الموجبة القطعية وتقليل عدد حالاتها (بالنسبة للمعيار الثنائي). S تحتوي على عوامل القياس،
!> S(i) = 1/sqrt(A(i,i))، المختارة بحيث تحتوي المصفوفة المقيسة B ذات العناصر B(i,j) = S(i)*A(i,j)*S(j) على وحدات على القطر. هذا
!> الاختيار لـ S يضع عدد حالات B ضمن عامل N من
!> أصغر عدد حالات ممكن عبر جميع قياسات قطرية ممكنة.
!>
!> هذا الإجراء يختلف عن CPOEQU بتقييد عوامل القياس
!> لقوة الأساس. بغض النظر عن تجاوز التدفق السفلي والعلوي، فإن القياس بواسطة
!> هذه العوامل لا يسبب أخطاء تقريب إضافية. ومع ذلك، فإن
!> العناصر القطرية المقيسة لم تعد تقريبًا 1 ولكنها تقع
!> بين sqrt(radix) و 1/sqrt(radix).
!>

المعاملات

N 
!>          N هو عدد صحيح (INTEGER)
!>          رتبة المصفوفة A. N >= 0.
!>

A

!>          A هي مصفوفة مركبة (COMPLEX)، ذات بعد (LDA,N)
!>          المصفوفة الهرمية الموجبة القطعية N×N التي سيتم حساب عوامل القياس الخاصة بها. يتم الرجوع فقط إلى العناصر القطرية لـ A.
!>

LDA

!>          LDA هو عدد صحيح (INTEGER)
!>          البعد الرائد للمصفوفة A. LDA >= max(1,N).
!>

S

!>          S هي مصفوفة حقيقية (REAL)، ذات بعد (N)
!>          إذا كان INFO = 0، فإن S تحتوي على عوامل القياس لـ A.
!>

SCOND

!>          SCOND هو عدد حقيقي (REAL)
!>          إذا كان INFO = 0، فإن S تحتوي على نسبة أصغر S(i) إلى
!>          أكبر S(i). إذا كان SCOND >= 0.1 ولم يكن AMAX كبيرًا جدًا أو صغيرًا جدًا،
!>          فلا فائدة من القياس بـ S.
!>

AMAX

!>          AMAX هو عدد حقيقي (REAL)
!>          القيمة المطلقة لأكبر عنصر في المصفوفة. إذا كان AMAX قريبًا جدًا
!>          من تجاوز التدفق أو قريبًا جدًا من التدفق السفلي، فيجب قياس المصفوفة.
!>

INFO

!>          INFO هو عدد صحيح (INTEGER)
!>          = 0: إنهاء ناجح
!>          < 0: إذا كان INFO = -i، فإن المعامل i كان له قيمة غير قانونية
!>          > 0: إذا كان INFO = i، فإن العنصر القطري i غير موجب.
!>
