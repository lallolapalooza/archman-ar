subroutine cpbequ (character uplo, integer n, integer kd, complex, dimension( ldab, * ) ab, integer ldab, real, dimension( * ) s, real scond, real amax, integer info)

CPBEQU

الغرض:

!>
!> CPBEQU يحسب عوامل قياس للصف والأعمدة المخصصة لمعايرة مصفوفة هيرميتية موجبة محددة ذات نطاق A وتقليل عدد حالتها
!> (بالنسبة للمعيار الثنائي). تحتوي S على عوامل القياس،
!> S(i) = 1/sqrt(A(i,i))، مختارة بحيث تكون المصفوفة الموسعة B ذات
!> العناصر B(i,j) = S(i)*A(i,j)*S(j) بها وحدات على القطر.
!> هذا الاختيار لـ S يضع عدد حالة B ضمن عامل N من
!> أقل عدد حالة ممكن عبر جميع قياسات الأقطار الممكنة.
!>

المعلمات

UPLO 
!>          UPLO هو CHARACTER*1
!>          = 'U':  يتم تخزين المثلث العلوي لـ A؛
!>          = 'L':  يتم تخزين المثلث السفلي لـ A.
!>

N

!>          N هو INTEGER
!>          ترتيب المصفوفة A. N >= 0.
!>

KD

!>          KD هو INTEGER
!>          عدد الأقطار الفائقة للمصفوفة A إذا كان UPLO = 'U'،
!>          أو عدد الأقطار الفرعية إذا كان UPLO = 'L'. KD >= 0.
!>

AB

!>          AB هو مصفوفة COMPLEX، البعد (LDAB,N)
!>          المثلث العلوي أو السفلي لمصفوفة النطاق الهيرميتية،
!>          مخزنة في أول KD+1 صفوف من المصفوفة. يتم تخزين العمود j
!>          للمصفوفة A في العمود j للمصفوفة AB كما يلي:
!>          إذا كان UPLO = 'U'، AB(kd+1+i-j,j) = A(i,j) لـ max(1,j-kd)<=i<=j؛
!>          إذا كان UPLO = 'L'، AB(1+i-j,j) = A(i,j) لـ j<=i<=min(n,j+kd).
!>

LDAB

!>          LDAB هو INTEGER
!>          البعد الرائد للمصفوفة A. LDAB >= KD+1.
!>

S

!>          S هو مصفوفة REAL، البعد (N)
!>          إذا كان INFO = 0، تحتوي S على عوامل القياس لـ A.
!>

SCOND

!>          SCOND هو REAL
!>          إذا كان INFO = 0، تحتوي S على نسبة أصغر S(i) إلى
!>          أكبر S(i). إذا كان SCOND >= 0.1 وكان AMAX ليس كبيرًا جدًا أو صغيرًا جدًا،
!>          فلا فائدة من القياس بواسطة S.
!>

AMAX

!>          AMAX هو REAL
!>          القيمة المطلقة لأكبر عنصر في المصفوفة. إذا كان AMAX قريبًا جدًا
!>          من الفيضان أو قريبًا جدًا من التدفق السفلي، يجب قياس المصفوفة.
!>

INFO

!>          INFO هو INTEGER
!>          = 0:  انتهاء ناجح
!>          < 0:  إذا كان INFO = -i، فقد كان الوسيط i له قيمة غير قانونية.
!>          > 0:  إذا كان INFO = i، فإن العنصر القطري i غير موجب.
!>
