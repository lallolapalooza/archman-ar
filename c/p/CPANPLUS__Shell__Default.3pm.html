CPANPLUS__Shell__Default(3pm)

الحزمة: perl-cpanplus

التراخيص: PerlArtistic, GPL

CPANPLUS::Shell::Default(3) وثائق بيرل مساهمة من المستخدم CPANPLUS::Shell::Default(3)

الاسم

CPANPLUS::Shell::Default - الواجهة الافتراضية لـ CPANPLUS

الملخص

### تحميل الواجهة:
$ cpanp                     # تشغيل 'cpanp' من سطر الأوامر
$ perl -MCPANPLUS -eshell   # تحميل الواجهة من سطر الأوامر
use CPANPLUS::Shell qw[Default];        # تحميل هذه الواجهة عبر API
                                        # يتم ذلك دائمًا عبر CPANPLUS::Shell
my $ui = CPANPLUS::Shell->new;
$ui->shell;                             # تشغيل الواجهة
$ui->dispatch_on_input( input => 'x');  # تحديث المصدر باستخدام
                                        # طريقة dispatch
### عند التواجد في الواجهة:
### لاحظ أن جميع الأوامر يمكن أن تأخذ خيارات أيضًا.
### انظر إلى طرق CPANPLUS::Backend الأساسية الخاصة بها لمعرفة
### الخيارات التي تستخدمها.
cpanp> h                 # عرض رسائل المساعدة
cpanp> ?                 # عرض رسائل المساعدة
cpanp> m Acme            # البحث عن وحدات acme، يسمح بالتعبيرات القياسية
cpanp> a KANE            # البحث عن وحدات بواسطة kane، يسمح بالتعبيرات القياسية
cpanp> f Acme::Foo       # الحصول على قائمة بجميع إصدارات Acme::Foo
cpanp> i Acme::Foo       # تثبيت Acme::Foo
cpanp> i Acme-Foo-1.3    # تثبيت الإصدار 1.3 من Acme::Foo
cpanp> i <URI>           # التثبيت من URI، مثل ftp://foo.com/X.tgz
cpanp> i <DIR>           # التثبيت من دليل مطلق أو نسبي
cpanp> i 1 3..5          # تثبيت نتائج البحث 1، 3، 4 و 5
cpanp> i *               # تثبيت جميع نتائج البحث
cpanp> a KANE; i *;      # البحث عن وحدات بواسطة kane، تثبيت جميع النتائج
cpanp> t Acme::Foo       # اختبار Acme::Foo، دون تثبيتها
cpanp> u Acme::Foo       # إلغاء تثبيت Acme::Foo
cpanp> d Acme::Foo       # تحميل Acme::Foo
cpanp> z Acme::Foo       # تحميل واستخراج Acme::Foo، ثم فتح
                         # واجهة في دليل الاستخراج
cpanp> c Acme::Foo       # الحصول على قائمة بنتائج الاختبار لـ Acme::Foo
cpanp> l Acme::Foo       # عرض تفاصيل حول حزمة Acme::Foo
cpanp> r Acme::Foo       # عرض ملف README الخاص بـ Acme::Foo
cpanp> o                 # الحصول على قائمة بجميع الوحدات المثبتة
                         # التي انتهت صلاحيتها
cpanp> o 1..3            # سرد حالة التحديث من بحث سابق
cpanp> s conf            # عرض إعدادات التكوين
cpanp> s conf md5 1      # تمكين فحوصات md5
cpanp> s program         # عرض إعدادات البرنامج
cpanp> s edit            # تحرير ملف التكوين
cpanp> s reconfigure     # المرور عبر التكوين الأولي مرة أخرى
cpanp> s selfupdate      # تحديث تثبيت CPANPLUS الخاص بك
cpanp> s save            # حفظ التكوين على القرص
cpanp> s mirrors         # عرض المرايا المحددة حاليًا
cpanp> ! [PERL CODE]     # تنفيذ كود بيرل التالي
cpanp> b                 # إنشاء حزمة تلقائية لتثبيت بيرل
                         # الخاص بهذا الكمبيوتر
cpanp> x                 # إعادة تحميل ملفات الفهرس (مسح ذاكرة التخزين المؤقت)
cpanp> x --update_source # إعادة تحميل ملفات الفهرس، الحصول على ملفات مصدر جديدة
cpanp> p [FILE]          # طباعة مكدس الأخطاء (إلى ملف)
cpanp> v                 # عرض الشعار
cpanp> w                 # عرض نتائج البحث الأخيرة مرة أخرى
cpanp> q                 # الخروج من الواجهة
cpanp> e                 # الخروج من الواجهة وإعادة التحميل
cpanp> /plugins          # سرد المكونات الإضافية المتاحة
cpanp> /? PLUGIN         # سرد المساعدة الخاصة بـ <PLUGIN>
### خيارات شائعة:
cpanp> i ... --skiptest # تخطي الاختبارات
cpanp> i ... --force    # فرض جميع العمليات
cpanp> i ... --verbose  # التشغيل في وضع الإسهاب

الوصف

توفر هذه الوحدة الواجهة الافتراضية للمستخدم لـ "CPANPLUS". يمكنك تشغيلها عبر البرنامج "cpanp"، أو كما هو مفصل في الملخص.
