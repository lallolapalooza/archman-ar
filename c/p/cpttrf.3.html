subroutine cpttrf (integer n, real, dimension( * ) d, complex, dimension( * ) e, integer info)

CPTTRF

الغرض:

!>
!> CPTTRF يحسب تفكيك L*D*L**H لمصفوفة ثلاثية قطرية مترافقة موجبة محددة A. قد يُنظر إلى التفكيك أيضًا
!> على أنه يأخذ الشكل A = U**H *D*U.
!>

المعلمات

N 
!>          N هو عدد صحيح
!>          ترتيب المصفوفة A. N >= 0.
!>

D

!>          D هو مصفوفة حقيقية، أبعادها (N)
!>          عند الدخول، العناصر القطرية n للمصفوفة الثلاثية القطرية
!>          A. عند الخروج، العناصر القطرية n للمصفوفة القطرية
!>          D من تفكيك L*D*L**H للمصفوفة A.
!>

E

!>          E هي مصفوفة مركبة، أبعادها (N-1)
!>          عند الدخول، العناصر تحت القطرية (n-1) للمصفوفة الثلاثية القطرية
!>          A. عند الخروج، العناصر تحت القطرية (n-1) للعامل
!>          الوحدوي ثنائي القطر L من تفكيك L*D*L**H للمصفوفة A.
!>          يمكن اعتبار E أيضًا القطر العلوي للعامل
!>          الوحدوي ثنائي القطر U من تفكيك U**H *D*U للمصفوفة A.
!>

INFO

!>          INFO هو عدد صحيح
!>          = 0: خروج ناجح
!>          < 0: إذا كان INFO = -k، فإن الوسيط k كان له قيمة غير قانونية
!>          > 0: إذا كان INFO = k، فإن العامل الرئيسي الأولي من الرتبة k
!>               ليس موجبًا؛ إذا كان k < N، لم يكن من الممكن إكمال التفكيك، بينما إذا كان k = N، تم
!>               إكمال التفكيك، ولكن D(N) <= 0.
!>
