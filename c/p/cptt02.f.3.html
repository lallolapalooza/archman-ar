subroutine cptt02 (character uplo, integer n, integer nrhs, real, dimension( * ) d, complex, dimension( * ) e, complex, dimension( ldx, * ) x, integer ldx, complex, dimension( ldb, * ) b, integer ldb, real resid)

CPTT02

الغرض:

!>
!> CPTT02 يحسب الباقي لحل نظام المعادلات المتماثل ثلاثي القطر:
!>    RESID = norm(B - A*X) / (norm(A) * norm(X) * EPS),
!> حيث EPS هو اقل وحدة للآلة.
!>

المعلمات

UPLO 
!>          UPLO هو CHARACTER*1
!>          يحدد ما إذا كان القطر العلوي أو السفلي للمصفوفة ثلاثية القطر A
!>          مخزناً.
!>          = 'U':  E هو القطر العلوي لـ A
!>          = 'L':  E هو القطر السفلي لـ A
!>

N

!>          N هو INTEGER
!>          رتبة المصفوفة A.
!>

NRHS

!>          NRHS هو INTEGER
!>          عدد الأطراف اليمنى، أي عدد الأعمدة
!>          للمصفوفتين B و X. NRHS >= 0.
!>

D

!>          D هو مصفوفة REAL، ذات الأبعاد (N)
!>          العناصر القطرية n للمصفوفة ثلاثية القطر A.
!>

E

!>          E هو مصفوفة COMPLEX، ذات الأبعاد (N-1)
!>          العناصر القطرية الفرعية (n-1) للمصفوفة ثلاثية القطر A.
!>

X

!>          X هي مصفوفة COMPLEX، ذات الأبعاد (LDX,NRHS)
!>          المصفوفة n في nrhs لمتجهات الحل X.
!>

LDX

!>          LDX هو INTEGER
!>          البعد الرائد للمصفوفة X. LDX >= max(1,N).
!>

B

!>          B هي مصفوفة COMPLEX، ذات الأبعاد (LDB,NRHS)
!>          عند الإدخال، المصفوفة n في nrhs لمتجهات الطرف الأيمن B.
!>          عند الخروج، يتم استبدال B بـ B - A*X.
!>

LDB

!>          LDB هو INTEGER
!>          البعد الرائد للمصفوفة B. LDB >= max(1,N).
!>

RESID

!>          RESID هو REAL
!>          norm(B - A*X) / (norm(A) * norm(X) * EPS)
!>
