وثائق الوظيفة/الروتين الفرعي

subroutine cpot05 (character uplo, integer n, integer nrhs, complex, dimension( lda, * ) a, integer lda, complex, dimension( ldb, * ) b, integer ldb, complex, dimension( ldx, * ) x, integer ldx, complex, dimension( ldxact, * ) xact, integer ldxact, real, dimension( * ) ferr, real, dimension( * ) berr, real, dimension( * ) reslts)

CPOT05

الغرض:

!>
!> CPOT05 تختبر حدود الخطأ من التنقيح التكراري للحل المحسوب لنظام المعادلات A*X = B، حيث A مصفوفة هرميتية n × n.
!>
!> RESLTS(1) = اختبار حد الخطأ
!>           = norm(X - XACT) / ( norm(X) * FERR )
!>
!> يتم إرجاع قيمة كبيرة إذا لم تكن هذه النسبة أقل من واحد.
!>
!> RESLTS(2) = الباقي من روتين التنقيح التكراري
!>           = الحد الأقصى لـ BERR / ( (n+1)*EPS + (*) )، حيث
!>             (*) = (n+1)*UNFL / (min_i (abs(A)*abs(X) +abs(b))_i )
!>

المعلمات

UPLO 
!>          UPLO هو CHARACTER*1
!>          يحدد ما إذا كان الجزء العلوي أو السفلي المثلثي للمصفوفة الهرميتية A مخزنًا.
!>          = 'U':  مثلثي علوي
!>          = 'L':  مثلثي سفلي
!>

N

!>          N هو INTEGER
!>          عدد صفوف المصفوفات X، B، و XACT، وترتيب المصفوفة A. N >= 0.
!>

NRHS

!>          NRHS هو INTEGER
!>          عدد أعمدة المصفوفات X، B، و XACT. NRHS >= 0.
!>

A

!>          A هو مصفوفة COMPLEX، البعد (LDA,N)
!>          المصفوفة الهرميتية A. إذا كان UPLO = 'U'، فإن الجزء العلوي المثلثي n × n الرائد للمصفوفة A يحتوي على الجزء العلوي المثلثي للمصفوفة A، ولا تتم الإشارة إلى الجزء السفلي المثلثي الصارم للمصفوفة A. إذا كان UPLO = 'L'، فإن الجزء السفلي المثلثي n × n الرائد للمصفوفة A يحتوي على الجزء السفلي المثلثي للمصفوفة A، ولا تتم الإشارة إلى الجزء العلوي المثلثي الصارم للمصفوفة A.
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرائد للمصفوفة A. LDA >= max(1,N).
!>

B

!>          B هي مصفوفة COMPLEX، البعد (LDB,NRHS)
!>          متجهات الطرف الأيمن لنظام المعادلات الخطية.
!>

LDB

!>          LDB هو INTEGER
!>          البعد الرائد للمصفوفة B. LDB >= max(1,N).
!>

X

!>          X هي مصفوفة COMPLEX، البعد (LDX,NRHS)
!>          متجهات الحلول المحسوبة. يتم تخزين كل متجه كعمود في المصفوفة X.
!>

LDX

!>          LDX هو INTEGER
!>          البعد الرائد للمصفوفة X. LDX >= max(1,N).
!>

XACT

!>          XACT هي مصفوفة COMPLEX، البعد (LDX,NRHS)
!>          متجهات الحلول الدقيقة. يتم تخزين كل متجه كعمود في المصفوفة XACT.
!>

LDXACT

!>          LDXACT هو INTEGER
!>          البعد الرائد للمصفوفة XACT. LDXACT >= max(1,N).
!>

FERR

!>          FERR هي مصفوفة REAL، البعد (NRHS)
!>          تقديرات حدود الخطأ الأمامي لكل متجه حل X. إذا كان XTRUE هو الحل الحقيقي، فإن FERR يحد من حجم أكبر مدخل في (X - XTRUE) مقسومًا على حجم أكبر مدخل في X.
!>

BERR

!>          BERR هي مصفوفة REAL، البعد (NRHS)
!>          الخطأ الخلفي النسبي المكون لكل متجه حل (أي، أصغر تغيير نسبي في أي مدخل في A أو B يجعل X حلاً دقيقًا).
!>

RESLTS

!>          RESLTS هي مصفوفة REAL، البعد (2)
!>          الحد الأقصى على متجهات الحلول NRHS للنسب:
!>          RESLTS(1) = norm(X - XACT) / ( norm(X) * FERR )
!>          RESLTS(2) = BERR / ( (n+1)*EPS + (*) )
!>
