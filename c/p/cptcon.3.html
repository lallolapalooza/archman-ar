subroutine cptcon (integer n, real, dimension( * ) d, complex, dimension( * ) e, real anorm, real rcond, real, dimension( * ) rwork, integer info)

CPTCON

الغرض:

!>
!> CPTCON يحسب مقلوب رقم الشرط (في المعيار 1) لمصفوفة معقدة هرميتية موجبة محددة ثلاثية القطر
!> باستخدام التحليل A = L*D*L**H أو A = U**H*D*U المحسوب بواسطة
!> CPTTRF.
!>
!> يتم حساب Norm(inv(A)) بطريقة مباشرة، ويتم حساب مقلوب
!> رقم الشرط على النحو التالي:
!>                  RCOND = 1 / (ANORM * norm(inv(A))).
!>

المعلمات

N 
!>          N هو عدد صحيح (INTEGER)
!>          رتبة المصفوفة A.  N >= 0.
!>

D

!>          D هو مصفوفة حقيقية (REAL array)، ذات بعد (N)
!>          العناصر القطرية الـ n للمصفوفة القطرية D من
!>          تحليل A، كما هو محسوب بواسطة CPTTRF.
!>

E

!>          E هي مصفوفة معقدة (COMPLEX array)، ذات بعد (N-1)
!>          العناصر غير القطرية الـ (N-1) للعامل الثنائي الوحدة
!>          U أو L من تحليل A، كما هو محسوب بواسطة CPTTRF.
!>

ANORM

!>          ANORM هو عدد حقيقي (REAL)
!>          المعيار 1 للمصفوفة الأصلية A.
!>

RCOND

!>          RCOND هو عدد حقيقي (REAL)
!>          مقلوب رقم شرط المصفوفة A،
!>          المحسوب على النحو RCOND = 1/(ANORM * AINVNM)، حيث AINVNM هو
!>          المعيار 1 للمصفوفة المعكوسة inv(A) المحسوب في هذا الروتين.
!>

RWORK

!>          RWORK هي مصفوفة حقيقية (REAL array)، ذات بعد (N)
!>

INFO

!>          INFO هو عدد صحيح (INTEGER)
!>          = 0:  خروج ناجح
!>          < 0:  إذا كان INFO = -i، فإن المعلمة i كانت لها قيمة غير قانونية
!>
