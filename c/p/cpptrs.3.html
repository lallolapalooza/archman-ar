subroutine cpptrs (character uplo, integer n, integer nrhs, complex, dimension( * ) ap, complex, dimension( ldb, * ) b, integer ldb, integer info)

CPPTRS

الغرض:

!>
!> CPPTRS تحل نظام المعادلات الخطية A*X = B بمصفوفة هرميتية موجبة محددة A
!> في تخزين مجمع باستخدام تحليل Cholesky A = U**H*U أو A = L*L**H الذي تم حسابه بواسطة CPPTRF.
!>

المعلمات

UPLO 
!>          UPLO هو CHARACTER*1
!>          = 'U':  يتم تخزين المثلث العلوي من A؛
!>          = 'L':  يتم تخزين المثلث السفلي من A.
!>

N

!>          N هو INTEGER
!>          رتبة المصفوفة A.  N >= 0.
!>

NRHS

!>          NRHS هو INTEGER
!>          عدد الأطراف اليمنى، أي عدد الأعمدة
!>          من المصفوفة B.  NRHS >= 0.
!>

AP

!>          AP هو مصفوفة COMPLEX، البعد (N*(N+1)/2)
!>          العامل المثلثي U أو L من تحليل Cholesky
!>          A = U**H*U أو A = L*L**H، مجمع عموديًا في مصفوفة خطية.
!>          يتم تخزين العمود j من U أو L في المصفوفة AP
!>          كما يلي:
!>          إذا كان UPLO = 'U'، فإن AP(i + (j-1)*j/2) = U(i,j) لـ 1<=i<=j؛
!>          إذا كان UPLO = 'L'، فإن AP(i + (j-1)*(2n-j)/2) = L(i,j) لـ j<=i<=n.
!>

B

!>          B هي مصفوفة COMPLEX، البعد (LDB,NRHS)
!>          عند الدخول، مصفوفة الطرف الأيمن B.
!>          عند الخروج، مصفوفة الحل X.
!>

LDB

!>          LDB هو INTEGER
!>          البعد الرائد للمصفوفة B.  LDB >= max(1,N).
!>

INFO

!>          INFO هو INTEGER
!>          = 0:  انتهاء ناجح
!>          < 0:  إذا كان INFO = -i، فإن المعلمة i كان لها قيمة غير قانونية
!>
