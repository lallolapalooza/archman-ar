subroutine cptt02 (character uplo, integer n, integer nrhs, real, dimension( * ) d, complex, dimension( * ) e, complex, dimension( ldx, * ) x, integer ldx, complex, dimension( ldb, * ) b, integer ldb, real resid)

CPTT02

الغرض:

!>
!> CPTT02 يحسب الباقي لحل نظام معادلات ثلاثي قطري متناظر:
!>    RESID = norm(B - A*X) / (norm(A) * norm(X) * EPS),
!> حيث EPS هو الخطأ المعياري للآلة.
!>

المعلمات

UPLO 
!>          UPLO هو CHARACTER*1
!>          يحدد ما إذا كان القطر العلوي أو القطر السفلي للمصفوفة
!>          ثلاثية الأقطار A مخزنة.
!>          = 'U':  E هو القطر العلوي لـ A
!>          = 'L':  E هو القطر السفلي لـ A
!>

N

!>          N هو INTEGER
!>          ترتيب المصفوفة A.
!>

NRHS

!>          NRHS هو INTEGER
!>          عدد الأطراف اليمنى، أي عدد أعمدة
!>          المصفوفتين B و X. NRHS >= 0.
!>

D

!>          D هو مصفوفة REAL، البعد (N)
!>          عناصر القطر للمصفوفة ثلاثية الأقطار A.
!>

E

!>          E هو مصفوفة COMPLEX، البعد (N-1)
!>          عناصر الأقطار الفرعية للمصفوفة ثلاثية الأقطار A.
!>

X

!>          X هي مصفوفة COMPLEX، البعد (LDX,NRHS)
!>          مصفوفة الحلول X التي حجمها N × NRHS.
!>

LDX

!>          LDX هو INTEGER
!>          البعد الرئيسي للمصفوفة X. LDX >= max(1,N).
!>

B

!>          B هي مصفوفة COMPLEX، البعد (LDB,NRHS)
!>          عند الدخول، مصفوفة الأطراف اليمنى B التي حجمها N × NRHS.
!>          عند الخروج، يتم استبدال B بالفرق B - A*X.
!>

LDB

!>          LDB هو INTEGER
!>          البعد الرئيسي للمصفوفة B. LDB >= max(1,N).
!>

RESID

!>          RESID هو REAL
!>          norm(B - A*X) / (norm(A) * norm(X) * EPS)
!>
