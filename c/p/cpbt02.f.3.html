subroutine cpbt02 (character uplo, integer n, integer kd, integer nrhs, complex, dimension( lda, * ) a, integer lda, complex, dimension( ldx, * ) x, integer ldx, complex, dimension( ldb, * ) b, integer ldb, real, dimension( * ) rwork, real resid)

CPBT02

الغرض:

!>
!> CPBT02 يحسب الباقي لحل نظام المعادلات المرتفعة الهرمية A*x = b:
!>    RESID = norm( B - A*X ) / ( norm(A) * norm(X) * EPS)
!> حيث EPS هي دقة الآلة.
!>

المعلمات

UPLO 
!>          UPLO هو CHARACTER*1
!>          يحدد ما إذا كان الجزء العلوي أو السفلي من المصفوفة الهرمية A مخزنًا:
!>          = 'U':  مثلث علوي
!>          = 'L':  مثلث سفلي
!>

N

!>          N هو INTEGER
!>          عدد الصفوف والأعمدة للمصفوفة A.  N >= 0.
!>

KD

!>          KD هو INTEGER
!>          عدد الأقطار العلوية للمصفوفة A إذا كان UPLO = 'U'،
!>          أو عدد الأقطار السفلية إذا كان UPLO = 'L'.  KD >= 0.
!>

NRHS

!>          NRHS هو INTEGER
!>          عدد الأطراف اليمنى. NRHS >= 0.
!>

A

!>          A هو مصفوفة COMPLEX، الأبعاد (LDA,N)
!>          المصفوفة المرتفعة الهرمية الأصلية A.  إذا كان UPLO = 'U'،
!>          يتم تخزين الجزء المثلثي العلوي من A كمصفوفة شريطية؛ إذا
!>          كان UPLO = 'L'، يتم تخزين الجزء المثلثي السفلي من A.  يتم
!>          تخزين أعمدة المثلث المناسب في أعمدة A ويتم تخزين أقطار
!>          المثلث في صفوف A.  راجع CPBTRF لمزيد من التفاصيل.
!>

LDA

!>          LDA هو INTEGER.
!>          البعد الرائد للمصفوفة A.  LDA >= max(1,KD+1).
!>

X

!>          X هي مصفوفة COMPLEX، الأبعاد (LDX,NRHS)
!>          متجهات الحلول المحسوبة لنظام المعادلات الخطية.
!>

LDX

!>          LDX هو INTEGER
!>          البعد الرائد للمصفوفة X.  LDX >= max(1,N).
!>

B

!>          B هي مصفوفة COMPLEX، الأبعاد (LDB,NRHS)
!>          عند الدخول، متجهات الطرف الأيمن لنظام المعادلات الخطية.
!>          عند الخروج، يتم استبدال B بالفرق B - A*X.
!>

LDB

!>          LDB هو INTEGER
!>          البعد الرائد للمصفوفة B.  LDB >= max(1,N).
!>

RWORK

!>          RWORK هي مصفوفة REAL، الأبعاد (N)
!>

RESID

!>          RESID هو REAL
!>          الحد الأقصى على عدد الأطراف اليمنى لـ
!>          norm(B - A*X) / ( norm(A) * norm(X) * EPS ).
!>
