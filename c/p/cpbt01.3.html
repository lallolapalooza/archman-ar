subroutine cpbt01 (character uplo, integer n, integer kd, complex, dimension( lda, * ) a, integer lda, complex, dimension( ldafac, * ) afac, integer ldafac, real, dimension( * ) rwork, real resid)

CPBT01

الغرض:

!>
!> CPBT01 يعيد بناء مصفوفة هيرميتية موجبة محددة ذات نطاق A من
!> تحليلها L*L' أو U'*U ويحسب الباقي
!>    norm( L*L' - A ) / ( N * norm(A) * EPS ) أو
!>    norm( U'*U - A ) / ( N * norm(A) * EPS ),
!> حيث EPS هو الخطأ النسبي للآلة، L' هو مرافق متحول L، و U' هو مرافق متحول U.
!>

المعاملات

UPLO 
!>          UPLO is CHARACTER*1
!>          يحدد ما إذا كان الجزء العلوي أو السفلي المثلثي من المصفوفة الهيرميتية A مخزنًا:
!>          = 'U':  مثلثي علوي
!>          = 'L':  مثلثي سفلي
!>

N

!>          N is INTEGER
!>          عدد الصفوف والأعمدة في المصفوفة A. N >= 0.
!>

KD

!>          KD is INTEGER
!>          عدد الأقطار الفرعية العلوية للمصفوفة A إذا كان UPLO = 'U'،
!>          أو عدد الأقطار الفرعية إذا كان UPLO = 'L'. KD >= 0.
!>

A

!>          A is COMPLEX array, dimension (LDA,N)
!>          المصفوفة الهيرميتية الأصلية ذات النطاق A. إذا كان UPLO = 'U'،
!>          يتم تخزين الجزء المثلثي العلوي من A كمصفوفة ذات نطاق؛ إذا
!>          كان UPLO = 'L'، يتم تخزين الجزء المثلثي السفلي من A.
!>          يتم تخزين أعمدة المثلث المناسب في أعمدة
!>          A ويتم تخزين أقطار المثلث في صفوف
!>          A. انظر CPBTRF لمزيد من التفاصيل.
!>

LDA

!>          LDA is INTEGER.
!>          البعد الرائد للمصفوفة A. LDA >= max(1,KD+1).
!>

AFAC

!>          AFAC is COMPLEX array, dimension (LDAFAC,N)
!>          الشكل المحلل للمصفوفة A. AFAC يحتوي على العامل
!>          L أو U من تحليل L*L' أو U'*U بتنسيق تخزين النطاق
!>          ، كما تم حسابه بواسطة CPBTRF.
!>

LDAFAC

!>          LDAFAC is INTEGER
!>          البعد الرائد للمصفوفة AFAC.
!>          LDAFAC >= max(1,KD+1).
!>

RWORK

!>          RWORK is REAL array, dimension (N)
!>

RESID

!>          RESID is REAL
!>          إذا كان UPLO = 'L'، norm(L*L' - A) / ( N * norm(A) * EPS )
!>          إذا كان UPLO = 'U'، norm(U'*U - A) / ( N * norm(A) * EPS )
!>
