subroutine cppcon (character uplo, integer n, complex, dimension( * ) ap, real anorm, real rcond, complex, dimension( * ) work, real, dimension( * ) rwork, integer info)

CPPCON

الغرض:

!>
!> CPPCON يقدر مقلوب رقم الشرط (في المعيار 1) لمصفوفة هرميتية معقدة موجبة محددة مجمعة باستخدام
!> تحويل Cholesky A = U**H*U أو A = L*L**H المحسوب بواسطة
!> CPPTRF.
!>
!> يتم الحصول على تقدير لـ norm(inv(A))، ويتم حساب مقلوب
!> رقم الشرط كـ RCOND = 1 / (ANORM * norm(inv(A))).
!>

المعاملات

UPLO 
!>          UPLO is CHARACTER*1
!>          = 'U':  يتم تخزين المثلث العلوي لـ A؛
!>          = 'L':  يتم تخزين المثلث السفلي لـ A.
!>

N

!>          N is INTEGER
!>          ترتيب المصفوفة A.  N >= 0.
!>

AP

!>          AP is COMPLEX array, dimension (N*(N+1)/2)
!>          العامل المثلثي U أو L من تحويل Cholesky
!>          A = U**H*U أو A = L*L**H، مجمعة حسب العمود في مصفوفة
!>          خطية. يتم تخزين العمود j من U أو L في المصفوفة AP
!>          على النحو التالي:
!>          إذا كان UPLO = 'U'، فإن AP(i + (j-1)*j/2) = U(i,j) لـ 1<=i<=j؛
!>          إذا كان UPLO = 'L'، فإن AP(i + (j-1)*(2n-j)/2) = L(i,j) لـ j<=i<=n.
!>

ANORM

!>          ANORM is REAL
!>          المعيار 1 (أو المعيار اللانهائي) للمصفوفة الهرميتية A.
!>

RCOND

!>          RCOND is REAL
!>          مقلوب رقم الشرط للمصفوفة A،
!>          المحسوب كـ RCOND = 1/(ANORM * AINVNM)، حيث AINVNM هو
!>          تقدير للمعيار 1 لـ inv(A) المحسوب في هذا الروتين.
!>

WORK

!>          WORK is COMPLEX array, dimension (2*N)
!>

RWORK

!>          RWORK is REAL array, dimension (N)
!>

INFO

!>          INFO is INTEGER
!>          = 0:  خروج ناجح
!>          < 0:  إذا كان INFO = -i، فإن المعامل i لديه قيمة غير قانونية
!>
