subroutine cpoequ (integer n, complex, dimension( lda, * ) a, integer lda, real, dimension( * ) s, real scond, real amax, integer info)

CPOEQU

الغرض:

!>
!> CPOEQU يحسب معاملات الصف والعمود المقصود بها موازنة مصفوفة هرميتية محددة موجبة A
!> وتقليل رقم حالتها (بالنسبة لمعيار الاثنين). S يحتوي على معاملات القياس،
!> S(i) = 1/sqrt(A(i,i))، المختارة بحيث تكون المصفوفة المقاسة B بعناصر
!> B(i,j) = S(i)*A(i,j)*S(j) لها واحدات على القطر. هذا الاختيار لـ S
!> يضع رقم حالة B ضمن عامل N من أصغر رقم حالة ممكن عبر جميع المقاييس القطرية الممكنة.
!>

المعلمات

N 
!>          N هو عدد صحيح (INTEGER)
!>          رتبة المصفوفة A. N >= 0.
!>

A

!>          A هي مصفوفة مركبة (COMPLEX)، الأبعاد (LDA,N)
!>          المصفوفة الهرميتية المحددة الموجبة N×N التي يتم حساب معاملات القياس لها.
!>          يتم الرجوع فقط إلى العناصر القطرية لـ A.
!>

LDA

!>          LDA هو عدد صحيح (INTEGER)
!>          البعد الرائد للمصفوفة A. LDA >= max(1,N).
!>

S

!>          S هي مصفوفة حقيقية (REAL)، الأبعاد (N)
!>          إذا كان INFO = 0، فإن S تحتوي على معاملات القياس لـ A.
!>

SCOND

!>          SCOND هو عدد حقيقي (REAL)
!>          إذا كان INFO = 0، فإن S تحتوي على نسبة أصغر S(i) إلى أكبر S(i).
!>          إذا كان SCOND >= 0.1 ولم يكن AMAX كبيرًا جدًا أو صغيرًا جدًا،
!>          فليس من المفيد القياس بواسطة S.
!>

AMAX

!>          AMAX هو عدد حقيقي (REAL)
!>          القيمة المطلقة لأكبر عنصر مصفوفة. إذا كان AMAX قريبًا جدًا من الفائض
!>          أو قريبًا جدًا من النقصان، فيجب قياس المصفوفة.
!>

INFO

!>          INFO هو عدد صحيح (INTEGER)
!>          = 0: نجاح الخروج
!>          < 0: إذا كان INFO = -i، فإن المعلمة i كانت ذات قيمة غير قانونية
!>          > 0: إذا كان INFO = i، فإن العنصر القطري i غير موجب.
!>
