subroutine cpbcon (character uplo, integer n, integer kd, complex, dimension( ldab, * ) ab, integer ldab, real anorm, real rcond, complex, dimension( * ) work, real, dimension( * ) rwork, integer info)

CPBCON

الغرض:

!>
!> CPBCON تقدر معكوس رقم الشرط (بمعيار 1-norm) لمصفوفة هيرميتية معقدة موجبة محددة شبه منحرفة
!> باستخدام تحليل Cholesky A = U**H*U أو A = L*L**H الذي تم حسابه بواسطة
!> CPBTRF.
!>
!> يتم الحصول على تقدير لـ norm(inv(A))، ويتم حساب معكوس رقم الشرط على أنه RCOND = 1 / (ANORM * norm(inv(A))).
!>

المعلمات

UPLO 
!>          UPLO is CHARACTER*1
!>          = 'U':  العامل المثلثي العلوي مخزن في AB؛
!>          = 'L':  العامل المثلثي السفلي مخزن في AB.
!>

N

!>          N is INTEGER
!>          ترتيب المصفوفة A.  N >= 0.
!>

KD

!>          KD is INTEGER
!>          عدد الأقطار العلوية للمصفوفة A إذا كان UPLO = 'U'،
!>          أو عدد الأقطار السفلية إذا كان UPLO = 'L'.  KD >= 0.
!>

AB

!>          AB is COMPLEX array, dimension (LDAB,N)
!>          العامل المثلثي U أو L من تحليل Cholesky
!>          A = U**H*U أو A = L*L**H للمصفوفة شبه المنحرفة A، مخزن في
!>          أول KD+1 صفوف من المصفوفة.  يتم تخزين العمود j من U أو L
!>          في العمود j من المصفوفة AB كما يلي:
!>          إذا كان UPLO ='U'، فإن AB(kd+1+i-j,j) = U(i,j) لـ max(1,j-kd)<=i<=j؛
!>          إذا كان UPLO ='L'، فإن AB(1+i-j,j)    = L(i,j) لـ j<=i<=min(n,j+kd).
!>

LDAB

!>          LDAB is INTEGER
!>          البعد الرائد للمصفوفة AB.  LDAB >= KD+1.
!>

ANORM

!>          ANORM is REAL
!>          المعيار 1 (أو المعيار اللانهائي) للمصفوفة الهيرميتية شبه المنحرفة A.
!>

RCOND

!>          RCOND is REAL
!>          معكوس رقم شرط المصفوفة A،
!>          محسوب على أنه RCOND = 1/(ANORM * AINVNM)، حيث AINVNM هو
!>          تقدير للمعيار 1 لـ inv(A) المحسوب في هذه الوحدة.
!>

WORK

!>          WORK is COMPLEX array, dimension (2*N)
!>

RWORK

!>          RWORK is REAL array, dimension (N)
!>

INFO

!>          INFO is INTEGER
!>          = 0:  خروج ناجح
!>          < 0:  إذا كان INFO = -i، فإن المعلمة i كانت لها قيمة غير قانونية
!>
