توثيق الوظيفة/الإجراء الفرعي

subroutine cpbsv (character uplo, integer n, integer kd, integer nrhs, complex, dimension( ldab, * ) ab, integer ldab, complex, dimension( ldb, * ) b, integer ldb, integer info)

CPBSV يحسب حل نظام المعادلات الخطية A * X = B لمصفوفات أخرى

الغرض:

!>
!> CPBSV يحسب حل نظام المعادلات الخطية المعقدة
!>    A * X = B,
!> حيث A هي مصفوفة نطاق هيرميتية موجبة محددة من N*N و X و B هي مصفوفات N*NRHS.
!>
!> يتم استخدام تحليل تشولسكي لتحليل A كـ
!>    A = U**H * U,  إذا كان UPLO = 'U', أو
!>    A = L * L**H,  إذا كان UPLO = 'L',
!> حيث U هي مصفوفة نطاق مثلثية عليا، و L هي مصفوفة نطاق مثلثية سفلى، بنفس عدد النطاقات العلوية أو السفلية مثل A. ثم يتم استخدام الشكل المحلل لـ A لحل نظام المعادلات A * X = B.
!>

المعلمات

UPLO 
!>          UPLO هو CHARACTER*1
!>          = 'U':  يتم تخزين المثلث العلوي لـ A;
!>          = 'L':  يتم تخزين المثلث السفلي لـ A.
!>

N

!>          N هو INTEGER
!>          عدد المعادلات الخطية، أي رتبة المصفوفة A. N >= 0.
!>

KD

!>          KD هو INTEGER
!>          عدد النطاقات العلوية للمصفوفة A إذا كان UPLO = 'U',
!>          أو عدد النطاقات السفلية إذا كان UPLO = 'L'. KD >= 0.
!>

NRHS

!>          NRHS هو INTEGER
!>          عدد الأطراف اليمنى، أي عدد أعمدة المصفوفة B. NRHS >= 0.
!>

AB

!>          AB هو مصفوفة COMPLEX، البعد (LDAB,N)
!>          عند الدخول، المثلث العلوي أو السفلي للمصفوفة النطاقية الهيرميتية
!>          A، مخزنة في أول KD+1 صف من المصفوفة.
!>          يتم تخزين العمود j من A في العمود j من المصفوفة AB
!>          كالتالي:
!>          إذا كان UPLO = 'U', AB(KD+1+i-j,j) = A(i,j) لـ max(1,j-KD)<=i<=j;
!>          إذا كان UPLO = 'L', AB(1+i-j,j)    = A(i,j) لـ j<=i<=min(N,j+KD).
!>          انظر أدناه لمزيد من التفاصيل.
!>
!>          عند الخروج، إذا كان INFO = 0، العامل المثلثي U أو L من
!>          تحليل تشولسكي A = U**H*U أو A = L*L**H للمصفوفة النطاقية
!>          A، بنفس تنسيق التخزين مثل A.
!>

LDAB

!>          LDAB هو INTEGER
!>          البعد الرئيسي للمصفوفة AB. LDAB >= KD+1.
!>

B

!>          B هي مصفوفة COMPLEX، البعد (LDB,NRHS)
!>          عند الدخول، مصفوفة الطرف الأيمن B ذات الأبعاد N*NRHS.
!>          عند الخروج، إذا كان INFO = 0، المصفوفة الحل X ذات الأبعاد N*NRHS.
!>

LDB

!>          LDB هو INTEGER
!>          البعد الرئيسي للمصفوفة B. LDB >= max(1,N).
!>

INFO

!>          INFO هو INTEGER
!>          = 0:  الخروج بنجاح
!>          < 0:  إذا كان INFO = -i، فإن المعلمة i كانت لها قيمة غير قانونية
!>          > 0:  إذا كان INFO = i، فإن المحدد الرئيسي للمصفوفة A من الرتبة i
!>                ليس موجباً، لذا لم يكن من الممكن إكمال التحليل، ولم يتم حساب الحل.
!>

!>
!>  يوضح مخطط تخزين النطاق بالمثال التالي، عندما
!>  N = 6، KD = 2، و UPLO = 'U':
!>
!>  عند الدخول:                       عند الخروج:
!>
!>      *    *   a13  a24  a35  a46      *    *   u13  u24  u35  u46
!>      *   a12  a23  a34  a45  a56      *   u12  u23  u34  u45  u56
!>     a11  a22  a33  a44  a55  a66     u11  u22  u33  u44  u55  u66
!>
!>  وبالمثل، إذا كان UPLO = 'L' فإن تنسيق A كالتالي:
!>
!>  عند الدخول:                       عند الخروج:
!>
!>     a11  a22  a33  a44  a55  a66     l11  l22  l33  l44  l55  l66
!>     a21  a32  a43  a54  a65   *      l21  l32  l43  l54  l65   *
!>     a31  a42  a53  a64   *    *      l31  l42  l53  l64   *    *
!>
!>  العناصر المحددة بـ * لا يتم استخدامها بواسطة الإجراء الفرعي.
!>
