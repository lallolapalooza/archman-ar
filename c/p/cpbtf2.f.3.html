subroutine cpbtf2 (uplo, n, kd, ab, ldab, info)


CPBTF2 تحسب تحليل Cholesky لمصفوفة شريطية متناظرة/هرمية موجبة محددة (خوارزمية غير محظورة).

توثيق الوظيفة/الروتين الفرعي

subroutine cpbtf2 (character uplo, integer n, integer kd, complex, dimension( ldab, * ) ab, integer ldab, integer info)

CPBTF2 تحسب تحليل Cholesky لمصفوفة شريطية هرمية معقدة موجبة محددة (خوارزمية غير محظورة).

الغرض:

!>
!> CPBTF2 تحسب تحليل Cholesky لمصفوفة هرمية معقدة موجبة محددة A.
!>
!> التحليل له الشكل
!>    A = U**H * U ,  إذا كان UPLO = 'U', أو
!>    A = L  * L**H,  إذا كان UPLO = 'L',
!> حيث U مصفوفة مثلثية عليا، U**H هي النسخة المترافقة المستبدلة لـ U، و L مصفوفة مثلثة سفلى.
!>
!> هذا هو الإصدار غير المحظور من الخوارزمية، الذي يستدعي BLAS من المستوى 2.
!>

المعلمات

UPLO 
!>          UPLO هو CHARACTER*1
!>          يحدد ما إذا كان الجزء المثلثي العلوي أو السفلي من المصفوفة الهرمية A مخزنًا:
!>          = 'U':  مثلثية عليا
!>          = 'L':  مثلثية سفلى
!>

N

!>          N هو INTEGER
!>          رتبة المصفوفة A.  N >= 0.
!>

KD

!>          KD هو INTEGER
!>          عدد الأقطار الفوقية للمصفوفة A إذا كان UPLO = 'U'،
!>          أو عدد الأقطار التحتية إذا كان UPLO = 'L'.  KD >= 0.
!>

AB

!>          AB هو مصفوفة COMPLEX، البعد (LDAB,N)
!>          عند الدخول، الجزء المثلثي العلوي أو السفلي من مصفوفة الشريط الهرمية A، مخزن في أول KD+1 صفوف من المصفوفة.  يتم تخزين العمود j من A في العمود j من المصفوفة AB
!>          كما يلي:
!>          إذا كان UPLO = 'U', AB(kd+1+i-j,j) = A(i,j) لـ max(1,j-kd)<=i<=j؛
!>          إذا كان UPLO = 'L', AB(1+i-j,j)    = A(i,j) لـ j<=i<=min(n,j+kd).
!>
!>          عند الخروج، إذا كان INFO = 0، فإن العامل المثلثي U أو L من تحليل Cholesky A = U**H *U أو A = L*L**H لمصفوفة الشريط A، بنفس تنسيق التخزين مثل A.
!>

LDAB

!>          LDAB هو INTEGER
!>          البعد الرائد للمصفوفة AB.  LDAB >= KD+1.
!>

INFO

!>          INFO هو INTEGER
!>          = 0: خروج ناجح
!>          < 0: إذا كان INFO = -k، فإن المعلمة k كان لها قيمة غير قانونية
!>          > 0: إذا كان INFO = k، فإن الأصغر الرئيسي الرائد من الرتبة k
!>               ليس موجبًا، ولم يكن من الممكن إكمال التحليل.
!>

تفاصيل إضافية:

!>
!>  يتم توضيح مخطط تخزين الشريط بالمثال التالي، عندما
!>  N = 6, KD = 2, و UPLO = 'U':
!>
!>  عند الدخول:                       عند الخروج:
!>
!>      *    *   a13  a24  a35  a46      *    *   u13  u24  u35  u46
!>      *   a12  a23  a34  a45  a56      *   u12  u23  u34  u45  u56
!>     a11  a22  a33  a44  a55  a66     u11  u22  u33  u44  u55  u66
!>
!>  وبالمثل، إذا كان UPLO = 'L' فإن تنسيق A هو كالتالي:
!>
!>  عند الدخول:                       عند الخروج:
!>
!>     a11  a22  a33  a44  a55  a66     l11  l22  l33  l44  l55  l66
!>     a21  a32  a43  a54  a65   *      l21  l32  l43  l54  l65   *
!>     a31  a42  a53  a64   *    *      l31  l42  l53  l64   *    *
!>
!>  العناصر في المصفوفة المشار إليها بـ * لا تستخدمها الروتين.
!>
