subroutine cpot01 (character uplo, integer n, complex, dimension( lda, * ) a, integer lda, complex, dimension( ldafac, * ) afac, integer ldafac, real, dimension( * ) rwork, real resid)

CPOT01

الغرض:

!>
!> CPOT01 تعيد بناء مصفوفة هيرميتية موجبة قطعية A من
!> تحليلها L*L' أو U'*U وتحسب الباقي
!>    norm( L*L' - A ) / ( N * norm(A) * EPS ) أو
!>    norm( U'*U - A ) / ( N * norm(A) * EPS )،
!> حيث EPS هو مكافئ الآلة، L' هو المرافق الظهري لـ L،
!> و U' هو المرافق الظهري لـ U.
!>

المعلمات

UPLO 
!>          UPLO is CHARACTER*1
!>          يحدد الجزء المثلثي العلوي أو السفلي من المصفوفة الهيرميتية A
!>          الذي تم تخزينه:
!>          = 'U':  مثلثي علوي
!>          = 'L':  مثلثي سفلي
!>

N

!>          N is INTEGER
!>          عدد الصفوف والأعمدة في المصفوفة A.  N >= 0.
!>

A

!>          A is COMPLEX array, dimension (LDA,N)
!>          المصفوفة الهيرميتية الأصلية A.
!>

LDA

!>          LDA is INTEGER
!>          البعد الرئيسي للمصفوفة A.  LDA >= max(1,N)
!>

AFAC

!>          AFAC is COMPLEX array, dimension (LDAFAC,N)
!>          عند الإدخال، العامل L أو U من تحليل L * L**H أو U**H * U
!>          لـ A.
!>          يتم الكتابة فوقه بالمصفوفة المعاد بناؤها، ثم بالفرق
!>          L * L**H - A (أو U**H * U - A).
!>

LDAFAC

!>          LDAFAC is INTEGER
!>          البعد الرئيسي للمصفوفة AFAC.  LDAFAC >= max(1,N).
!>

RWORK

!>          RWORK is REAL array, dimension (N)
!>

RESID

!>          RESID is REAL
!>          إذا كان UPLO = 'L'، norm(L * L**H - A) / ( N * norm(A) * EPS )
!>          إذا كان UPLO = 'U'، norm(U**H * U - A) / ( N * norm(A) * EPS )
!>
