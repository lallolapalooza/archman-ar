subroutine cpbt01 (character uplo, integer n, integer kd, complex, dimension( lda, * ) a, integer lda, complex, dimension( ldafac, * ) afac, integer ldafac, real, dimension( * ) rwork, real resid)

CPBT01

الغرض:

!>
!> CPBT01 يعيد بناء مصفوفة A موجبة محددة هرميتية من النوع الشريطي من
!> تحليلها L*L' أو U'*U ويحسب الباقي
!>    norm( L*L' - A ) / ( N * norm(A) * EPS ) أو
!>    norm( U'*U - A ) / ( N * norm(A) * EPS ),
!> حيث EPS هو الخطأ الآلي، L' هو منقول المعقد لـ L، و U' هو منقول المعقد لـ U.
!>

المعلمات

UPLO 
!>          UPLO هو CHARACTER*1
!>          يحدد ما إذا كان الجزء العلوي أو السفلي المثلثي للمصفوفة A
!>          الهرميتية مخزنًا:
!>          = 'U':  علوي مثلثي
!>          = 'L':  سفلي مثلثي
!>

N

!>          N هو INTEGER
!>          عدد الصفوف والأعمدة للمصفوفة A.  N >= 0.
!>

KD

!>          KD هو INTEGER
!>          عدد الأقطار الفائقة للمصفوفة A إذا كان UPLO = 'U'،
!>          أو عدد الأقطار الفرعية إذا كان UPLO = 'L'.  KD >= 0.
!>

A

!>          A هي مصفوفة COMPLEX، حجمها (LDA,N)
!>          المصفوفة الهرميتية الشريطية الأصلية A.  إذا كان UPLO = 'U'،
!>          يتم تخزين الجزء العلوي المثلثي لـ A كمصفوفة شريطية؛ إذا كان
!>          UPLO = 'L'، يتم تخزين الجزء السفلي المثلثي لـ A.  يتم تخزين
!>          الأعمدة من المثلث المناسب في أعمدة A ويتم تخزين الأقطار
!>          للمثلث في صفوف A.  انظر CPBTRF لمزيد من التفاصيل.
!>

LDA

!>          LDA هو INTEGER.
!>          البعد الرائد للمصفوفة A.  LDA >= max(1,KD+1).
!>

AFAC

!>          AFAC هي مصفوفة COMPLEX، حجمها (LDAFAC,N)
!>          الشكل المحلل للمصفوفة A.  يحتوي AFAC على العامل
!>          L أو U من تحليل L*L' أو U'*U بتنسيق تخزين شريطي،
!>          كما تم حسابه بواسطة CPBTRF.
!>

LDAFAC

!>          LDAFAC هو INTEGER
!>          البعد الرائد للمصفوفة AFAC.
!>          LDAFAC >= max(1,KD+1).
!>

RWORK

!>          RWORK هي مصفوفة REAL، حجمها (N)
!>

RESID

!>          RESID هو REAL
!>          إذا كان UPLO = 'L'، norm(L*L' - A) / ( N * norm(A) * EPS )
!>          إذا كان UPLO = 'U'، norm(U'*U - A) / ( N * norm(A) * EPS )
!>
