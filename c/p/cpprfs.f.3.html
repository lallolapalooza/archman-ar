subroutine cpprfs (character uplo, integer n, integer nrhs, complex, dimension( * ) ap, complex, dimension( * ) afp, complex, dimension( ldb, * ) b, integer ldb, complex, dimension( ldx, * ) x, integer ldx, real, dimension( * ) ferr, real, dimension( * ) berr, complex, dimension( * ) work, real, dimension( * ) rwork, integer info)

CPPRFS

الغرض:

!>
!> CPPRFS يحسن الحل المحسوب لنظام المعادلات الخطية
!> عندما تكون مصفوفة المعاملات هرميتية موجبة محددة
!> ومعبأة، ويوفر حدود خطأ وتقديرات خطأ خلفي
!> للحل.
!>

المعلمات

UPLO 
!>          UPLO is CHARACTER*1
!>          = 'U':  المثلث العلوي لـ A مخزن؛
!>          = 'L':  المثلث السفلي لـ A مخزن.
!>

N

!>          N is INTEGER
!>          رتبة المصفوفة A.  N >= 0.
!>

NRHS

!>          NRHS is INTEGER
!>          عدد الأيادي اليمنى، أي عدد الأعمدة
!>          للمصفوفتين B و X.  NRHS >= 0.
!>

AP

!>          AP is COMPLEX array, dimension (N*(N+1)/2)
!>          المثلث العلوي أو السفلي للمصفوفة الهرميتية A، معبأ
!>          عمودياً في مصفوفة خطية.  العمود j من A مخزن
!>          في المصفوفة AP كما يلي:
!>          إذا كان UPLO = 'U'، AP(i + (j-1)*j/2) = A(i,j) لـ 1<=i<=j؛
!>          إذا كان UPLO = 'L'، AP(i + (j-1)*(2n-j)/2) = A(i,j) لـ j<=i<=n.
!>

AFP

!>          AFP is COMPLEX array, dimension (N*(N+1)/2)
!>          العامل المثلثي U أو L من تحليل Cholesky
!>          A = U**H*U أو A = L*L**H، كما تم حسابه بواسطة SPPTRF/CPPTRF،
!>          معبأ عمودياً في مصفوفة خطية بنفس تنسيق A
!>          (انظر AP).
!>

B

!>          B is COMPLEX array, dimension (LDB,NRHS)
!>          مصفوفة الجهة اليمنى B.
!>

LDB

!>          LDB is INTEGER
!>          البعد الرائد للمصفوفة B.  LDB >= max(1,N).
!>

X

!>          X is COMPLEX array, dimension (LDX,NRHS)
!>          عند الإدخال، مصفوفة الحل X، كما تم حسابها بواسطة CPPTRS.
!>          عند الخروج، مصفوفة الحل المحسنة X.
!>

LDX

!>          LDX is INTEGER
!>          البعد الرائد للمصفوفة X.  LDX >= max(1,N).
!>

FERR

!>          FERR is REAL array, dimension (NRHS)
!>          حد الخطأ الأمامي المقدر لكل متجه حل
!>          X(j) (العمود j من مصفوفة الحل X).
!>          إذا كان XTRUE هو الحل الحقيقي المقابل لـ X(j)، فإن FERR(j)
!>          هو حد علوي مقدر لقيمة القيمة المطلقة لأكبر
!>          عنصر في (X(j) - XTRUE) مقسومة على القيمة المطلقة لأكبر
!>          عنصر في X(j).  التقدير موثوق به بقدر
!>          موثوقية تقدير RCOND، وهو تقريبًا دائمًا
!>          زيادة طفيفة في الخطأ الحقيقي.
!>

BERR

!>          BERR is REAL array, dimension (NRHS)
!>          خطأ الخلفية النسبي المكون من عنصر واحد لكل متجه حل
!>          X(j) (أي، أصغر تغيير نسبي في
!>          أي عنصر في A أو B يجعل X(j) حلاً دقيقًا).
!>

WORK

!>          WORK is COMPLEX array, dimension (2*N)
!>

RWORK

!>          RWORK is REAL array, dimension (N)
!>

INFO

!>          INFO is INTEGER
!>          = 0:  إنهاء ناجح
!>          < 0:  إذا كان INFO = -i، فإن المعلمة i كانت ذات قيمة غير قانونية
!>
