subroutine cppt01 (character uplo, integer n, complex, dimension( * ) a, complex, dimension( * ) afac, real, dimension( * ) rwork, real resid)

CPPT01

الغرض:

!>
!> CPPT01 تعيد بناء مصفوفة هيرميتية موجبة محددة A
!> من تحليلها L*L' أو U'*U وتحسب المتبقي
!>    norm( L*L' - A ) / ( N * norm(A) * EPS ) أو
!>    norm( U'*U - A ) / ( N * norm(A) * EPS ),
!> حيث EPS هو الجزء العشري لآلة الحاسبة، L' هو المرافق المقابل لـ L،
!> و U' هو المرافق المقابل لـ U.
!>

المعاملات

UPLO 
!>          UPLO is CHARACTER*1
!>          يحدد ما إذا كان الجزء العلوي أو السفلي المثلثي من المصفوفة
!>          الهيرميتية A مخزنًا:
!>          = 'U':  مثلثي علوي
!>          = 'L':  مثلثي سفلي
!>

N

!>          N is INTEGER
!>          عدد الصفوف والأعمدة للمصفوفة A.  N >= 0.
!>

A

!>          A is COMPLEX array, dimension (N*(N+1)/2)
!>          المصفوفة الهيرميتية الأصلية A، مخزنة كمصفوفة
!>          مثلثية مضغوطة.
!>

AFAC

!>          AFAC is COMPLEX array, dimension (N*(N+1)/2)
!>          عند الإدخال، المعامل L أو U من تحليل L*L' أو U'*U
!>          للمصفوفة A، مخزن كمصفوفة مثلثية مضغوطة.
!>          يتم استبداله بالمصفوفة المعاد بناؤها، ثم بالفرق
!>          L*L' - A (أو U'*U - A).
!>

RWORK

!>          RWORK is REAL array, dimension (N)
!>

RESID

!>          RESID is REAL
!>          إذا كان UPLO = 'L'، norm(L*L' - A) / ( N * norm(A) * EPS )
!>          إذا كان UPLO = 'U'، norm(U'*U - A) / ( N * norm(A) * EPS )
!>
