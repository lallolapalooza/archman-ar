subroutine cpoequ (integer n, complex, dimension( lda, * ) a, integer lda, real, dimension( * ) s, real scond, real amax, integer info)

CPOEQU

الغرض:

!>
!> CPOEQU يحسب عوامل قياس الصف والعمود المقصودة لمعادلة مصفوفة A الهرمية الموجبة المحددة
!> وتقليل عدد حالتها (بالنسبة للمعيار الثنائي). S تحتوي على عوامل القياس،
!> S(i) = 1/sqrt(A(i,i))، المختارة بحيث أن المصفوفة المقيّسة B ذات العناصر
!> B(i,j) = S(i)*A(i,j)*S(j) لها واحدات على القطر. هذا الاختيار لـ S
!> يضع عدد حالة B ضمن عامل N من أصغر عدد حالة ممكن عبر جميع القياسات القطرية الممكنة.
!>

المعلمات

N 
!>          N هو عدد صحيح (INTEGER)
!>          رتبة المصفوفة A.  N >= 0.
!>

A

!>          A هي مصفوفة معقدة (COMPLEX)، البعد (LDA,N)
!>          المصفوفة الهرمية الموجبة المحددة N×N التي سيتم حساب عوامل القياس لها.
!>          يتم الرجوع فقط إلى العناصر القطرية لـ A.
!>

LDA

!>          LDA هو عدد صحيح (INTEGER)
!>          البعد الرئيسي للمصفوفة A.  LDA >= max(1,N).
!>

S

!>          S هي مصفوفة حقيقية (REAL)، البعد (N)
!>          إذا كانت INFO = 0، تحتوي S على عوامل القياس لـ A.
!>

SCOND

!>          SCOND هي قيمة حقيقية (REAL)
!>          إذا كانت INFO = 0، تحتوي S على نسبة أصغر S(i) إلى أكبر S(i).
!>          إذا كان SCOND >= 0.1 و AMAX ليس كبيرًا جدًا أو صغيرًا جدًا،
!>          فليس من المفيد القياس باستخدام S.
!>

AMAX

!>          AMAX هي قيمة حقيقية (REAL)
!>          القيمة المطلقة لأكبر عنصر في المصفوفة.
!>          إذا كان AMAX قريبًا جدًا من الفائض أو قريبًا جدًا من النقص،
!>          فيجب قياس المصفوفة.
!>

INFO

!>          INFO هو عدد صحيح (INTEGER)
!>          = 0:  انتهاء ناجح
!>          < 0:  إذا كانت INFO = -i، فإن المعلمة i كان لها قيمة غير قانونية
!>          > 0:  إذا كانت INFO = i، فإن العنصر القطري i غير موجب.
!>
