subroutine cptrfs (character uplo, integer n, integer nrhs, real, dimension( * ) d, complex, dimension( * ) e, real, dimension( * ) df, complex, dimension( * ) ef, complex, dimension( ldb, * ) b, integer ldb, complex, dimension( ldx, * ) x, integer ldx, real, dimension( * ) ferr, real, dimension( * ) berr, complex, dimension( * ) work, real, dimension( * ) rwork, integer info)

CPTRFS

Purpose:

!>
!> CPTRFS يحسن الحل المحسوب لنظام المعادلات الخطية
!> عندما تكون مصفوفة المعاملات هرميتية موجبة محددة
!> وثلاثية قطرية، ويوفر حدود الخطأ وتقديرات الخطأ الخلفي
!> للحل.
!>

Parameters

UPLO 
!>          UPLO is CHARACTER*1
!>          يحدد ما إذا كان القطر العلوي أو السفلي للمصفوفة الثلاثية
!>          A مخزنًا وشكل التحليل:
!>          = 'U':  E هو القطر العلوي لـ A، و A = U**H*D*U؛
!>          = 'L':  E هو القطر السفلي لـ A، و A = L*D*L**H.
!>          (الشكلان متكافئان إذا كانت A حقيقية.)
!>

N

!>          N is INTEGER
!>          رتبة المصفوفة A.  N >= 0.
!>

NRHS

!>          NRHS is INTEGER
!>          عدد الجوانب اليمنى، أي عدد أعمدة
!>          المصفوفة B.  NRHS >= 0.
!>

D

!>          D is REAL array, dimension (N)
!>          عناصر القطرية الحقيقية n للمصفوفة الثلاثية القطرية A.
!>

E

!>          E is COMPLEX array, dimension (N-1)
!>          العناصر القطرية غير القطرية (n-1) للمصفوفة الثلاثية القطرية A
!>          (انظر UPLO).
!>

DF

!>          DF is REAL array, dimension (N)
!>          عناصر القطرية الحقيقية n للمصفوفة القطرية D من
!>          التحليل الذي تم حسابه بواسطة CPTTRF.
!>

EF

!>          EF is COMPLEX array, dimension (N-1)
!>          العناصر القطرية غير القطرية (n-1) للمصفوفة ثنائية القطرية
!>          U أو L من التحليل الذي تم حسابه بواسطة CPTTRF
!>          (انظر UPLO).
!>

B

!>          B is COMPLEX array, dimension (LDB,NRHS)
!>          مصفوفة الجانب الأيمن B.
!>

LDB

!>          LDB is INTEGER
!>          البعد الرئيسي للمصفوفة B.  LDB >= max(1,N).
!>

X

!>          X is COMPLEX array, dimension (LDX,NRHS)
!>          عند الإدخال، مصفوفة الحل X، كما تم حسابها بواسطة CPTTRS.
!>          عند الخروج، مصفوفة الحل المحسنة X.
!>

LDX

!>          LDX is INTEGER
!>          البعد الرئيسي للمصفوفة X.  LDX >= max(1,N).
!>

FERR

!>          FERR is REAL array, dimension (NRHS)
!>          حد الخطأ الأمامي لكل متجه حل
!>          X(j) (العمود j للمصفوفة الحل X).
!>          إذا كان XTRUE هو الحل الحقيقي المقابل لـ X(j)، فإن FERR(j)
!>          هو حد أعلى تقديري لقيمة أكبر
!>          عنصر في (X(j) - XTRUE) مقسومًا على قيمة أكبر
!>          عنصر في X(j).
!>

BERR

!>          BERR is REAL array, dimension (NRHS)
!>          الخطأ النسبي المكوني الخلفي لكل متجه حل
!>          X(j) (أي، أصغر تغيير نسبي في
!>          أي عنصر في A أو B يجعل X(j) حلاً دقيقًا).
!>

WORK

!>          WORK is COMPLEX array, dimension (N)
!>

RWORK

!>          RWORK is REAL array, dimension (N)
!>

INFO

!>          INFO is INTEGER
!>          = 0:  خروج ناجح
!>          < 0:  إذا كان INFO = -i، فإن المعامل i كان له قيمة غير قانونية
!>

Internal Parameters:

!>  ITMAX هو الحد الأقصى لعدد خطوات التنقيح التكراري.
!>
