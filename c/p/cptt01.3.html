subroutine cptt01 (integer n, real, dimension( * ) d, complex, dimension( * ) e, real, dimension( * ) df, complex, dimension( * ) ef, complex, dimension( * ) work, real resid)

CPTT01

الغرض:

!>
!> CPTT01 يعيد بناء مصفوفة ثلاثية قطرية A من تحليلها L*D*L'
!> ويحسب الباقي
!>    norm(L*D*L' - A) / ( n * norm(A) * EPS ),
!> حيث EPS هي وحدة الآلة.
!>

المعاملات

N 
!>          N هو صحيح (INTEGER)
!>          ترتيب المصفوفة A.
!>

D

!>          D هي مصفوفة حقيقية (REAL)، البعد (N)
!>          العناصر القطرية N للمصفوفة الثلاثية القطرية A.
!>

E

!>          E هي مصفوفة مركبة (COMPLEX)، البعد (N-1)
!>          العناصر تحت القطرية (N-1) للمصفوفة الثلاثية القطرية A.
!>

DF

!>          DF هي مصفوفة حقيقية (REAL)، البعد (N)
!>          العناصر القطرية N للعامل L من تحليل L*D*L'
!>          للمصفوفة A.
!>

EF

!>          EF هي مصفوفة مركبة (COMPLEX)، البعد (N-1)
!>          العناصر تحت القطرية (N-1) للعامل L من
!>          تحليل L*D*L' للمصفوفة A.
!>

WORK

!>          WORK هي مصفوفة مركبة (COMPLEX)، البعد (2*N)
!>

RESID

!>          RESID هو حقيقي (REAL)
!>          norm(L*D*L' - A) / (n * norm(A) * EPS)
!>
