توثيق الوظيفة/الوحدة الفرعية

subroutine cbdt03 (character uplo, integer n, integer kd, real, dimension( * ) d, real, dimension( * ) e, complex, dimension( ldu, * ) u, integer ldu, real, dimension( * ) s, complex, dimension( ldvt, * ) vt, integer ldvt, complex, dimension( * ) work, real resid)

CBDT03

الغرض:

!>
!> CBDT03 يعيد بناء مصفوفة ثنائية القطر B من SVD الخاصة بها:
!>    S = U' * B * V
!> حيث U و V مصفوفتان متعامدتان و S مصفوفة قطرية.
!>
!> نسبة الاختبار لاختبار التحلل القيمي الفردي هي
!>    RESID = norm( B - U * S * VT ) / ( n * norm(B) * EPS )
!> حيث VT = V' و EPS هي دقة الآلة.
!>

المعلمات

UPLO 
!>          UPLO هو CHARACTER*1
!>          يحدد ما إذا كانت المصفوفة B ثنائية القطر علوية أم سفلية.
!>          = 'U':  ثنائية القطر علوية
!>          = 'L':  ثنائية القطر سفلية
!>

N

!>          N هو INTEGER
!>          رتبة المصفوفة B.
!>

KD

!>          KD هو INTEGER
!>          عرض النطاق للمصفوفة ثنائية القطر B. إذا كان KD = 1، فإن
!>          المصفوفة B ثنائية القطر، وإذا كان KD = 0، فإن B قطرية و E
!>          لا يتم الرجوع إليها. إذا كان KD أكبر من 1، فمن المفترض أنه
!>          1، وإذا كان KD أقل من 0، فمن المفترض أنه 0.
!>

D

!>          D هو مصفوفة REAL، حجمها (N)
!>          عناصر القطر n للمصفوفة ثنائية القطر B.
!>

E

!>          E هو مصفوفة REAL، حجمها (N-1)
!>          عناصر القطر العلوي (n-1) للمصفوفة ثنائية القطر B
!>          إذا كان UPLO = 'U'، أو عناصر القطر السفلي (n-1) لـ B إذا
!>          كان UPLO = 'L'.
!>

U

!>          U هو مصفوفة COMPLEX، حجمها (LDU,N)
!>          المصفوفة المتعامدة U بحجم n × n في التحويل B = U'*A*P.
!>

LDU

!>          LDU هو INTEGER
!>          البعد الرئيسي للمصفوفة U. LDU >= max(1,N)
!>

S

!>          S هو مصفوفة REAL، حجمها (N)
!>          القيم الفردية من SVD لـ B، مرتبة بترتيب تنازلي.
!>

VT

!>          VT هو مصفوفة COMPLEX، حجمها (LDVT,N)
!>          المصفوفة المتعامدة V' بحجم n × n في التحويل
!>          B = U * S * V'.
!>

LDVT

!>          LDVT هو INTEGER
!>          البعد الرئيسي للمصفوفة VT.
!>

WORK

!>          WORK هو مصفوفة COMPLEX، حجمها (2*N)
!>

RESID

!>          RESID هو REAL
!>          نسبة الاختبار:  norm(B - U * S * V') / ( n * norm(A) * EPS )
!>
