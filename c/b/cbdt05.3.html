توثيق الوظيفة/الروتين الفرعي

subroutine cbdt05 (integer m, integer n, complex, dimension( lda, * ) a, integer lda, real, dimension( * ) s, integer ns, complex, dimension( * ) u, integer ldu, complex, dimension( ldvt, * ) vt, integer ldvt, complex, dimension( * ) work, real resid)

CBDT05

الغرض:

!>
!> CBDT05 يعيد بناء مصفوفة ثنائية القطر B من تحليل القيمة المفردة الجزئي لها:
!>    S = U' * B * V
!> حيث U و V مصفوفتان متعامدتان و S مصفوفة قطرية.
!>
!> نسبة الاختبار لاختبار تحليل القيمة المفردة هي
!>    RESID = norm( S - U' * B * V ) / ( n * norm(B) * EPS )
!> حيث VT = V' و EPS هي دقة الآلة.
!>

المعلمات

M 
!>          M هو عدد صحيح (INTEGER)
!>          عدد صفوف المصفوفتين A و U.
!>

N

!>          N هو عدد صحيح (INTEGER)
!>          عدد أعمدة المصفوفتين A و VT.
!>

A

!>          A مصفوفة مركبة (COMPLEX array)، البعد (LDA,N)
!>          المصفوفة A بحجم m x n.
!>

LDA

!>          LDA هو عدد صحيح (INTEGER)
!>          البعد الرائد للمصفوفة A. LDA >= max(1,M).
!>

S

!>          S مصفوفة حقيقية (REAL array)، البعد (NS)
!>          القيم المفردة من تحليل القيمة المفردة الجزئي لـ B، مرتبة
!>          بترتيب تنازلي.
!>

NS

!>          NS هو عدد صحيح (INTEGER)
!>          عدد القيم/المتجهات المفردة من تحليل القيمة المفردة الجزئي
!>          لـ B.
!>

U

!>          U مصفوفة مركبة (COMPLEX array)، البعد (LDU,NS)
!>          المصفوفة المتعامدة U بحجم n x ns في S = U' * B * V.
!>

LDU

!>          LDU هو عدد صحيح (INTEGER)
!>          البعد الرائد للمصفوفة U. LDU >= max(1,N)
!>

VT

!>          VT مصفوفة مركبة (COMPLEX array)، البعد (LDVT,N)
!>          المصفوفة المتعامدة V بحجم n x ns في S = U' * B * V.
!>

LDVT

!>          LDVT هو عدد صحيح (INTEGER)
!>          البعد الرائد للمصفوفة VT.
!>

WORK

!>          WORK مصفوفة مركبة (COMPLEX array)، البعد (M,N)
!>

RESID

!>          RESID قيمة حقيقية (REAL)
!>          نسبة الاختبار:  norm(S - U' * A * V) / ( n * norm(A) * EPS )
!>
