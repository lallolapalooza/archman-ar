توثيق الوظيفة/الروتين الفرعي

subroutine cbdt01 (integer m, integer n, integer kd, complex, dimension( lda, * ) a, integer lda, complex, dimension( ldq, * ) q, integer ldq, real, dimension( * ) d, real, dimension( * ) e, complex, dimension( ldpt, * ) pt, integer ldpt, complex, dimension( * ) work, real, dimension( * ) rwork, real resid)

CBDT01

الغرض:

!>
!> CBDT01 تعيد بناء مصفوفة عامة A من شكلها ثنائي القطر
!>    A = Q * B * P**H
!> حيث Q (m by min(m,n)) و P**H (min(m,n) by n) هي مصفوفات وحدوية
!> و B ثنائية القطر.
!>
!> نسبة الاختبار لاختبار الاختزال هي
!>    RESID = norm(A - Q * B * P**H) / ( n * norm(A) * EPS )
!> حيث EPS هي دقة الآلة.
!>

المعلمات

M 
!>          M هو عدد صحيح (INTEGER)
!>          عدد صفوف المصفوفات A و Q.
!>

N

!>          N هو عدد صحيح (INTEGER)
!>          عدد أعمدة المصفوفات A و P**H.
!>

KD

!>          KD هو عدد صحيح (INTEGER)
!>          إذا كان KD = 0، فإن B قطرية والمصفوفة E غير مستدعاة.
!>          إذا كان KD = 1، تم إجراء الاختزال بواسطة xGEBRD؛ B قطرية علوية
!>          إذا كان M >= N، وقطرية سفلية إذا كان M < N.
!>          إذا كان KD = -1، تم إجراء الاختزال بواسطة xGBBRD؛ B دائمًا قطرية علوية.
!>

A

!>          A هي مصفوفة مركبة (COMPLEX) ، البعد (LDA,N)
!>          المصفوفة A ذات الأبعاد m × n.
!>

LDA

!>          LDA هو عدد صحيح (INTEGER)
!>          البعد الرئيسي للمصفوفة A. LDA >= max(1,M).
!>

Q

!>          Q هي مصفوفة مركبة (COMPLEX) ، البعد (LDQ,N)
!>          المصفوفة وحدوية Q ذات الأبعاد m × min(m,n) في الاختزال
!>          A = Q * B * P**H.
!>

LDQ

!>          LDQ هو عدد صحيح (INTEGER)
!>          البعد الرئيسي للمصفوفة Q. LDQ >= max(1,M).
!>

D

!>          D هي مصفوفة حقيقية (REAL) ، البعد (min(M,N))
!>          العناصر القطرية للمصفوفة ثنائية القطر B.
!>

E

!>          E هي مصفوفة حقيقية (REAL) ، البعد (min(M,N)-1)
!>          العناصر فوق القطرية للمصفوفة ثنائية القطر B إذا كان
!>          m >= n، أو العناصر تحت القطرية لـ B إذا كان m < n.
!>

PT

!>          PT هي مصفوفة مركبة (COMPLEX) ، البعد (LDPT,N)
!>          المصفوفة وحدوية P**H ذات الأبعاد min(m,n) × n في الاختزال
!>          A = Q * B * P**H.
!>

LDPT

!>          LDPT هو عدد صحيح (INTEGER)
!>          البعد الرئيسي للمصفوفة PT.
!>          LDPT >= max(1,min(M,N)).
!>

WORK

!>          WORK هي مصفوفة مركبة (COMPLEX) ، البعد (M+N)
!>

RWORK

!>          RWORK هي مصفوفة حقيقية (REAL) ، البعد (M)
!>

RESID

!>          RESID هو حقيقي (REAL)
!>          نسبة الاختبار:
!>          norm(A - Q * B * P**H) / ( n * norm(A) * EPS )
!>
