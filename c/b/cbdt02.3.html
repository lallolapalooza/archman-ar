subroutine cbdt02 (integer m, integer n, complex, dimension( ldb, * ) b, integer ldb, complex, dimension( ldc, * ) c, integer ldc, complex, dimension( ldu, * ) u, integer ldu, complex, dimension( * ) work, real, dimension( * ) rwork, real resid)

CBDT02

الغرض:

!>
!> يقوم CBDT02 باختبار تغيير الأساس C = U**H * B عن طريق حساب الباقي
!>
!>    RESID = norm(B - U * C) / ( max(m,n) * norm(B) * EPS ),
!>
!> حيث B و C هما مصفوفتان M × N، و U مصفوفة متعامدة M × M، و EPS هي دقة الآلة.
!>

المعلمات

M 
!>          M هو عدد صحيح (INTEGER)
!>          عدد الصفوف للمصفوفتين B و C ورتبة المصفوفة Q.
!>

N

!>          N هو عدد صحيح (INTEGER)
!>          عدد الأعمدة للمصفوفتين B و C.
!>

B

!>          B هي مصفوفة مركبة (COMPLEX)، ذات بعد (LDB,N)
!>          المصفوفة B ذات البعد M × N.
!>

LDB

!>          LDB هو عدد صحيح (INTEGER)
!>          البعد الرئيسي للمصفوفة B. LDB >= max(1,M).
!>

C

!>          C هي مصفوفة مركبة (COMPLEX)، ذات بعد (LDC,N)
!>          المصفوفة C ذات البعد M × N، يُفترض أنها تحتوي على U**H * B.
!>

LDC

!>          LDC هو عدد صحيح (INTEGER)
!>          البعد الرئيسي للمصفوفة C. LDC >= max(1,M).
!>

U

!>          U هي مصفوفة مركبة (COMPLEX)، ذات بعد (LDU,M)
!>          المصفوفة المتعامدة U ذات البعد M × M.
!>

LDU

!>          LDU هو عدد صحيح (INTEGER)
!>          البعد الرئيسي للمصفوفة U. LDU >= max(1,M).
!>

WORK

!>          WORK هي مصفوفة مركبة (COMPLEX)، ذات بعد (M)
!>

RWORK

!>          RWORK هي مصفوفة حقيقية (REAL)، ذات بعد (M)
!>

RESID

!>          RESID هو عدد حقيقي (REAL)
