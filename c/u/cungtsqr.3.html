subroutine cungtsqr (integer m, integer n, integer mb, integer nb, complex, dimension( lda, * ) a, integer lda, complex, dimension( ldt, * ) t, integer ldt, complex, dimension( * ) work, integer lwork, integer info)

CUNGTSQR

الغرض:

!>
!> CUNGTSQR ينشئ مصفوفة معقدة M-by-N Q_out بأعمدة متعامدة، وهي أول N أعمدة
!> من حاصل ضرب مصفوفتين وحدويتين معقدتين من الرتبة M التي تم إرجاعها بواسطة CLATSQR
!>
!>      Q_out = first_N_columns_of( Q(1)_in * Q(2)_in * ... * Q(k)_in ).
!>
!> انظر التوثيق الخاص بـ CLATSQR.
!>

المعلمات

M 
!>          M هو عدد صحيح (INTEGER)
!>          عدد الصفوف في المصفوفة A.  M >= 0.
!>

N

!>          N هو عدد صحيح (INTEGER)
!>          عدد الأعمدة في المصفوفة A. M >= N >= 0.
!>

MB

!>          MB هو عدد صحيح (INTEGER)
!>          حجم كتلة الصف المستخدمة بواسطة CLATSQR لإرجاع
!>          المصفوفتين A و T. MB > N.
!>          (لاحظ أنه إذا كان MB > M، فسيتم استخدام M بدلاً من MB
!>          كحجم كتلة الصف).
!>

NB

!>          NB هو عدد صحيح (INTEGER)
!>          حجم كتلة العمود المستخدمة بواسطة CLATSQR لإرجاع
!>          المصفوفتين A و T. NB >= 1.
!>          (لاحظ أنه إذا كان NB > N، فسيتم استخدام N بدلاً من NB
!>          كحجم كتلة العمود).
!>

A

!>          A هي مصفوفة معقدة (COMPLEX)، البعد (LDA,N)
!>
!>          عند الإدخال:
!>
!>             العناصر الموجودة على القطر وفوقه لا يتم الوصول إليها.
!>             تمثل العناصر الموجودة أسفل القطر مصفوفة V ذات شبه منحرف
!>             أدنى قطري تم حسابها بواسطة CLATSQR
!>             والتي تحدد المصفوفات المدخلة Q_in(k) (الوحدات على
!>             القطر لا يتم تخزينها) (نفس تنسيق الإخراج A
!>             أسفل القطر في CLATSQR).
!>
!>          عند الإخراج:
!>
!>             تحتوي المصفوفة A على مصفوفة متعامدة M-by-N Q_out،
!>             أي أن أعمدة A هي متجهات وحدة متعامدة.
!>

LDA

!>          LDA هو عدد صحيح (INTEGER)
!>          البعد الرئيسي للمصفوفة A.  LDA >= max(1,M).
!>

T

!>          T هي مصفوفة معقدة (COMPLEX)،
!>          البعد (LDT, N * NIRB)
!>          حيث NIRB = Number_of_input_row_blocks
!>                     = MAX( 1, CEIL((M-N)/(MB-N)) )
!>          Let NICB = Number_of_input_col_blocks
!>                   = CEIL(N/NB)
!>
!>          المرايا الكتلية العليا المثلثية التي تحدد المصفوفات المدخلة
!>          Q_in(k)، k=(1:NIRB*NICB). يتم تخزين المرايا الكتلية
!>          بشكل مضغوط في NIRB تسلسلات مرايا كتلية. كل من تسلسلات المرايا
!>          الكتلية NIRB يتم تخزينها في كتلة عمود أكبر NB-by-N في T وتتكون
!>          من NICB كتل أعمدة أصغر NB-by-NB علوية مثلثة.
!>          (نفس تنسيق الإخراج T في CLATSQR).
!>

LDT

!>          LDT هو عدد صحيح (INTEGER)
!>          البعد الرئيسي للمصفوفة T.
!>          LDT >= max(1,min(NB1,N)).
!>

WORK

!>          (مساحة عمل) مصفوفة معقدة (COMPLEX)، البعد (MAX(2,LWORK))
!>          عند الإخراج، إذا كان INFO = 0، فإن WORK(1) ترجع LWORK الأمثل.
!>

LWORK

!>          LWORK هو عدد صحيح (INTEGER)
!>          بعد المصفوفة WORK.  LWORK >= (M+NB)*N.
!>          إذا كان LWORK = -1، فسيتم افتراض استعلام عن مساحة العمل.
!>          تقوم الروتين فقط بحساب الحجم الأمثل لمصفوفة WORK،
!>          وتعيد هذا الحجم كمدخل أول لمصفوفة WORK، ولا يتم إصدار أي رسالة خطأ
!>          متعلقة بـ LWORK بواسطة XERBLA.
!>

INFO

!>          INFO هو عدد صحيح (INTEGER)
!>          = 0:  انتهاء ناجح
!>          < 0:  إذا كان INFO = -i، فإن الوسيط i كان له قيمة غير قانونية
!>
