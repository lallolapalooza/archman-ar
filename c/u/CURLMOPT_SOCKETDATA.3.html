CURLMOPT_SOCKETDATA(3)

الحزمة: curl

التراخيص: MIT

CURLMOPT_SOCKETDATA(3) دليل الوظائف المكتبية CURLMOPT_SOCKETDATA(3)

الاسم

CURLMOPT_SOCKETDATA - مؤشر مخصص يتم تمريره إلى استدعاء المقبس

الملخص

#include <curl/curl.h>
CURLMcode curl_multi_setopt(CURLM *handle, CURLMOPT_SOCKETDATA, void *pointer);

الوصف

مؤشر بيانات pointer لتمريره إلى استدعاء المقبس المحدد باستخدام الخيار CURLMOPT_SOCKETFUNCTION(3).

libcurl لا تتلاعب بهذا المؤشر ولكنه يتم تمريره فقط كوسيط clientp لاستدعاء المقبس.

الافتراضي

NULL

البروتوكولات

تؤثر هذه الوظيفة على جميع البروتوكولات المدعومة

مثال

struct priv {
  void *ours;
};
static int sock_cb(CURL *e, curl_socket_t s, int what, void *cbp, void *sockp)
{
  struct priv *p = sockp;
  printf("my ptr: %p\n", p->ours);
  if(what == CURL_POLL_REMOVE) {
    /* إزالة المقبس من مجموعتنا */
  }
  if(what & CURL_POLL_IN) {
    /* انتظار القراءة على هذا المقبس */
  }
  if(what & CURL_POLL_OUT) {
    /* انتظار الكتابة على هذا المقبس */
  }
  return 0;
}
int main(void)
{
  struct priv setup;
  CURLM *multi = curl_multi_init();
  /* ... استخدام استدعاء المقبس والمؤشر المخصص */
  curl_multi_setopt(multi, CURLMOPT_SOCKETFUNCTION, sock_cb);
  curl_multi_setopt(multi, CURLMOPT_SOCKETDATA, &setup);
}
