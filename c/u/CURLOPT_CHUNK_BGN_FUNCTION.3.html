CURLOPT_CHUNK_BGN_FUNCTION(3)

الحزمة: curl

التراخيص: MIT

CURLOPT_CHUNK_BGN_FUNCTION(3) دليل الوظائف للمكتبة CURLOPT_CHUNK_BGN_FUNCTION(3)

اسم

CURLOPT_CHUNK_BGN_FUNCTION - دالة استدعاء قبل عملية نقل مع مطابقة أحرف البدل FTP

ملخص

#include <curl/curl.h>
struct curl_fileinfo {
  char *filename;
  curlfiletype filetype;
  time_t time;   /* دائما صفر */
  unsigned int perm;
  int uid;
  int gid;
  curl_off_t size;
  long int hardlinks;
  struct {
    /* إذا لم يكن بعض هذه الحقول NULL، فهو مؤشر إلى b_data. */
    char *time;
    char *perm;
    char *user;
    char *group;
    char *target; /* مؤشر إلى اسم الملف المستهدف للرابط الرمزي */
  } strings;
  unsigned int flags;
  /* يستخدم داخليا */
  char *b_data;
  size_t b_size;
  size_t b_used;
};
long chunk_bgn_callback(const void *transfer_info, void *ptr,
                        int remains);
CURLcode curl_easy_setopt(CURL *handle, CURLOPT_CHUNK_BGN_FUNCTION,
                          chunk_bgn_callback);

وصف

مرر مؤشرًا إلى دالة الاستدعاء الخاصة بك، والتي يجب أن تتطابق مع النموذج الأولي المعروض أعلاه.

سيتم استدعاء دالة الاستدعاء هذه بواسطة libcurl قبل أن يتم نقل جزء من الدفق (إذا كانت عملية النقل تدعم الأجزاء).

المؤشر transfer_info يشير إلى بنية curl_fileinfo تحتوي على تفاصيل حول الملف الذي سيتم نقله.

تجعل دالة الاستدعاء هذه منطقية فقط عند استخدام خيار CURLOPT_WILDCARDMATCH(3) في الوقت الحالي.

الهدف من المعامل transfer_info هو بنية "تعتمد على الميزة". بالنسبة لتنزيل أحرف البدل FTP، فإن الهدف هو بنية curl_fileinfo (انظر curl/curl.h). المعامل ptr هو مؤشر تم تقديمه بواسطة CURLOPT_CHUNK_DATA(3). المعامل remains يحتوي على عدد الأجزاء المتبقية لكل عملية نقل. إذا كانت الميزة غير متوفرة، فإن المعامل له قيمة صفر.

أرجع CURL_CHUNK_BGN_FUNC_OK إذا كان كل شيء على ما يرام، CURL_CHUNK_BGN_FUNC_SKIP إذا كنت تريد تخطي الجزء المحدد، أو CURL_CHUNK_BGN_FUNC_FAIL لإخبار libcurl بالتوقف إذا حدث خطأ ما.

الافتراضي

NULL

البروتوكولات

تؤثر هذه الوظيفة على ftp فقط

مثال

#include <stdio.h>
struct callback_data {
   FILE *output;
};
static long file_is_coming(struct curl_fileinfo *finfo,
                           void *ptr,
                           int remains)
{
  struct callback_data *data = ptr;
  printf("%3d %40s %10luB ", remains, finfo->filename,
         (unsigned long)finfo->size);
  switch(finfo->filetype) {
  case CURLFILETYPE_DIRECTORY:
    printf(" DIR\n");
    break;
  case CURLFILETYPE_FILE:
    printf("FILE ");
    break;
  default:
    printf("OTHER\n");
    break;
  }
  if(finfo->filetype == CURLFILETYPE_FILE) {
    /* لا تنقل الملفات >= 50B */
    if(finfo->size > 50) {
      printf("SKIPPED\n");
      return CURL_CHUNK_BGN_FUNC_SKIP;
    }
    data->output = fopen(finfo->filename, "wb");
    if(!data->output) {
      return CURL_CHUNK_BGN_FUNC_FAIL;
    }
  }
  return CURL_CHUNK_BGN_FUNC_OK;
}
int main()
{
  /* بيانات لدالة الاستدعاء */
  struct callback_data callback_info;
  CURL *curl = curl_easy_init();
  /* يتم استدعاء دالة الاستدعاء قبل بدء تنزيل ملف معين */
  curl_easy_setopt(curl, CURLOPT_CHUNK_BGN_FUNCTION, file_is_coming);
  curl_easy_setopt(curl, CURLOPT_CHUNK_DATA, &callback_info);
}

قيمة الإرجاع

curl_easy_setopt(3) ترجع CURLcode يشير إلى النجاح أو الخطأ.

CURLE_OK (0) يعني أن كل شيء كان على ما يرام، والقيمة غير الصفرية تعني حدوث خطأ، انظر libcurl-errors(3).
