CURLOPT_OPENSOCKETFUNCTION(3)

الحزمة: curl

التراخيص: MIT

CURLOPT_OPENSOCKETFUNCTION(3) دليل وظائف المكتبة CURLOPT_OPENSOCKETFUNCTION(3)

الاسم

CURLOPT_OPENSOCKETFUNCTION - دالة استدعاء لفتح المقبس

الملخص

#include <curl/curl.h>
typedef enum  {
  CURLSOCKTYPE_IPCXN,  /* مقبس تم إنشاؤه لاتصال IP محدد */
} curlsocktype;
struct curl_sockaddr {
  int family;
  int socktype;
  int protocol;
  unsigned int addrlen;
  struct sockaddr addr;
};
curl_socket_t opensocket_callback(void *clientp,
                                  curlsocktype purpose,
                                  struct curl_sockaddr *address);
CURLcode curl_easy_setopt(CURL *handle, CURLOPT_OPENSOCKETFUNCTION, opensocket_callback);

الوصف

مرر مؤشرًا إلى دالة الاستدعاء الخاصة بك، والتي يجب أن تتطابق مع النموذج الأولي المعروض أعلاه.

يتم استدعاء دالة الاستدعاء هذه بواسطة libcurl بدلاً من استدعاء socket(2). يحدد وسيط purpose لدالة الاستدعاء الغرض الدقيق لهذا المقبس المحدد. CURLSOCKTYPE_IPCXN مخصص للاتصالات المستندة إلى IP وهو الغرض الوحيد المستخدم حاليًا في libcurl. قد تدعم الإصدارات المستقبلية من libcurl أغراضًا إضافية.

يحتوي مؤشر clientp على أي قيمة محددة من قبل المستخدم تم تعيينها باستخدام وظيفة CURLOPT_OPENSOCKETDATA(3).

تتلقى دالة الاستدعاء عنوان الطرف الآخر المحلل كوسيط address ويُسمح لها بتعديل العنوان أو رفض الاتصال تمامًا. يجب أن تُرجع دالة الاستدعاء المقبس المنشأ حديثًا أو CURL_SOCKET_BAD في حالة عدم التمكن من إنشاء اتصال أو اكتشاف خطأ آخر. يمكن بالطبع إجراء أي استدعاءات إضافية لـ setsockopt(2) على المقبس حسب تقدير المستخدم.

إذا تم إرجاع CURL_SOCKET_BAD بواسطة دالة الاستدعاء، فإن libcurl تعاملها كفشل في الاتصال وتحاول فتح مقبس للاتصال بعنوان IP مختلف مرتبط بالتحويل. إذا لم تكن هناك عناوين أخرى للمحاولة، تفشل libcurl التحويل مع رمز الخطأ CURLE_COULDNT_CONNECT.

يمكنك الحصول على عنوان IP الذي تفتح libcurl المقبس من أجله عن طريق تحويل address->addr إلى sockaddr_in إذا كان address->family هو AF_INET، أو إلى sockaddr_in6 إذا كان address->family هو AF_INET6. للحصول على مثال لكيفية مقارنة هذه البيانات، ارجع إلى docs/examples/block_ip.c.

إذا كنت ترغب في تمرير مقبس باتصال تم إنشاؤه بالفعل، فقم بإرجاع المقبس باستخدام دالة الاستدعاء هذه ثم استخدم CURLOPT_SOCKOPTFUNCTION(3) للإشارة إلى أنه متصل بالفعل.

الافتراضي

ما يعادل هذا:

   return socket(addr->family, addr->socktype, addr->protocol);

البروتوكولات

تؤثر هذه الوظيفة على جميع البروتوكولات المدعومة

مثال

/* اجعل libcurl تستخدم المقبس الذي تم إنشاؤه بالفعل 'sockfd' */
static curl_socket_t opensocket(void *clientp,
                                curlsocktype purpose,
                                struct curl_sockaddr *address)
{
  curl_socket_t sockfd;
  sockfd = *(curl_socket_t *)clientp;
  /* يتم تمرير المقبس المحدد خارجيًا عبر خيار OPENSOCKETDATA */
  return sockfd;
}
static int sockopt_callback(void *clientp, curl_socket_t curlfd,
                            curlsocktype purpose)
{
  return CURL_SOCKOPT_ALREADY_CONNECTED;
}
int main(void)
{
  CURL *curl = curl_easy_init();
  if(curl) {
    CURLcode res;
    extern int sockfd; /* المتصل بالفعل */
    /* تعتقد libcurl أنك تتصل بالمضيف */
    /* والمنفذ الذي تحدده في خيار URL. */
    curl_easy_setopt(curl, CURLOPT_URL, "http://99.99.99.99:9999");
    /* استدعاء هذه الوظيفة للحصول على مقبس */
    curl_easy_setopt(curl, CURLOPT_OPENSOCKETFUNCTION, opensocket);
    curl_easy_setopt(curl, CURLOPT_OPENSOCKETDATA, &sockfd);
    /* استدعاء هذه الوظيفة لتعيين خيارات للمقبس */
    curl_easy_setopt(curl, CURLOPT_SOCKOPTFUNCTION, sockopt_callback);
    res = curl_easy_perform(curl);
    curl_easy_cleanup(curl);
  }
}

التوفر

تمت الإضافة في curl 7.17.1

قيمة الإرجاع

curl_easy_setopt(3) تُرجع CURLcode يشير إلى النجاح أو الخطأ.

CURLE_OK (0) يعني أن كل شيء على ما يرام، وغير الصفر يعني حدوث خطأ، انظر libcurl-errors(3).
