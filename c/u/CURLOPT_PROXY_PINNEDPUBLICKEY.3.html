CURLOPT_PROXY_PINNEDPUBLICKEY(3)

الحزمة: curl

التراخيص: MIT

CURLOPT_PROXY_PINNEDPUBLICKEY(3) دليل وظائف المكتبة CURLOPT_PROXY_PINNEDPUBLICKEY(3)

الاسم

CURLOPT_PROXY_PINNEDPUBLICKEY - المفتاح العام المثبت لـ https proxy

الملخص

#include <curl/curl.h>
CURLcode curl_easy_setopt(CURL *handle, CURLOPT_PROXY_PINNEDPUBLICKEY,
                          char *pinnedpubkey);

الوصف

مرر مؤشرًا إلى سلسلة نصية تنتهي بصفر كمعامل. يمكن أن تكون السلسلة اسم ملف المفتاح العام المثبت الخاص بك. تنسيق الملف المتوقع هو "PEM" أو "DER". يمكن أن تكون السلسلة أيضًا أي عدد من تجزئات sha256 المشفرة بـ base64 مسبوقة بـ "sha256//" ومفصولة بـ ";"

عند التفاوض على اتصال TLS أو SSL، يرسل خادم الوكيل https شهادة تشير إلى هويته. يتم استخراج مفتاح عام من هذه الشهادة، وإذا لم يتطابق تمامًا مع المفتاح العام المقدم لهذا الخيار، فإن libcurl يتخلى عن الاتصال قبل إرسال أو استقبال أي بيانات.

عند عدم التطابق، يتم إرجاع CURLE_SSL_PINNEDPUBKEYNOTMATCH.

لا يحتاج التطبيق إلى الاحتفاظ بالسلسلة النصية بعد تعيين هذا الخيار.

يؤدي استخدام هذا الخيار عدة مرات إلى قيام السلسلة الأخيرة المعينة بتجاوز السلاسل السابقة. قم بتعيينها إلى NULL لتعطيل استخدامها مرة أخرى.

الافتراضي

NULL

البروتوكولات

تؤثر هذه الوظيفة على جميع البروتوكولات المستندة إلى TLS: HTTPS، FTPS، IMAPS، POP3S، SMTPS، إلخ.

هذا الخيار يعمل فقط مع الواجهات الخلفية التالية لـ TLS: GnuTLS، OpenSSL، mbedTLS، و wolfSSL

مثال

int main(void)
{
  CURL *curl = curl_easy_init();
  if(curl) {
    curl_easy_setopt(curl, CURLOPT_URL, "https://example.com");
    curl_easy_setopt(curl, CURLOPT_PROXY, "https://proxy:443");
    curl_easy_setopt(curl, CURLOPT_PROXY_PINNEDPUBLICKEY,
                     "sha256//YhKJKSzoTt2b5FP18fvpHo7fJYqQCjA"
                     "a3HWY3tvRMwE=;sha256//t62CeU2tQiqkexU74"
                     "Gxa2eg7fRbEgoChTociMee9wno=");
    /* تنفيذ الطلب */
    curl_easy_perform(curl);
  }
}

استخراج المفتاح العام

إذا لم يكن لديك ملف المفتاح العام لخادم الوكيل https، يمكنك استخراجه من شهادة خادم الوكيل https.

# استرجاع شهادة الخادم إذا لم تكن لديك بالفعل
#
# تأكد من فحص الشهادة لمعرفة ما إذا كانت كما توقعت
#
# خاص بنظام Windows:
# - استخدم NUL بدلاً من /dev/null.
# - قد ينتظر OpenSSL الإدخال بدلاً من قطع الاتصال. اضغط Enter.
# - إذا لم يكن لديك sed، فما عليك سوى نسخ الشهادة إلى ملف:
#   الأسطر من -----BEGIN CERTIFICATE----- إلى -----END CERTIFICATE-----.
#
openssl s_client -servername www.example.com -connect www.example.com:443 \
  < /dev/null | sed -n "/-----BEGIN/,/-----END/p" > www.example.com.pem
# استخراج المفتاح العام بتنسيق pem من الشهادة
openssl x509 -in www.example.com.pem -pubkey -noout > www.example.com.pubkey.pem
# تحويل المفتاح العام من pem إلى der
openssl asn1parse -noout -inform pem -in www.example.com.pubkey.pem \
  -out www.example.com.pubkey.der
# تجزئة sha256 وتشفير der بـ base64 إلى سلسلة للاستخدام
openssl dgst -sha256 -binary www.example.com.pubkey.der | openssl base64

يحتوي المفتاح العام بتنسيق PEM على رأس، وبيانات base64، وذيل:

-----BEGIN PUBLIC KEY-----
[بيانات BASE 64]
-----END PUBLIC KEY-----
