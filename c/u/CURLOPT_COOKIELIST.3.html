CURLOPT_COOKIELIST(3)

الحزمة: curl

التراخيص: MIT

CURLOPT_COOKIELIST(3) دليل وظائف المكتبة CURLOPT_COOKIELIST(3)

الاسم

CURLOPT_COOKIELIST - إضافة أو معالجة ملفات تعريف الارتباط المخزنة في الذاكرة

الملخص

#include <curl/curl.h>
CURLcode curl_easy_setopt(CURL *handle, CURLOPT_COOKIELIST,
                          char *cookie);

الوصف

مرر مؤشر حرف إلى سلسلة cookie.

يمكن أن يكون ملف تعريف الارتباط هذا عبارة عن سطر واحد بتنسيق Netscape / Mozilla أو تنسيق رأس HTTP عادي (Set-Cookie:). يقوم هذا الخيار أيضًا بتمكين محرك ملفات تعريف الارتباط. هذا يضيف ملف تعريف الارتباط الفردي هذا إلى مخزن ملفات تعريف الارتباط الداخلي.

نحن ننصح بشدة بعدم تحميل ملفات تعريف الارتباط من ملف رأس HTTP، لأن هذا تنسيق تبادل بيانات أقل شأناً.

توخ الحذر إذا كنت تستخدم هذا الخيار وقد تحدث عمليات نقل متعددة. إذا استخدمت تنسيق Set-Cookie ولم تحدد السلسلة مجالًا، فسيتم إرسال ملف تعريف الارتباط لأي مجال (حتى بعد متابعة عمليات إعادة التوجيه) ولا يمكن تعديله بواسطة ملف تعريف الارتباط الذي تم تعيينه بواسطة الخادم. إذا قام خادم بتعيين ملف تعريف ارتباط بنفس الاسم (أو ربما قمت باستيراد واحد)، فسيتم إرسال كليهما في عمليات النقل المستقبلية إلى هذا الخادم، وهو على الأرجح ليس ما قصدته. لمعالجة هذه المشكلات، قم بتعيين مجال في Set-Cookie (يتضمن ذلك النطاقات الفرعية) أو الأفضل من ذلك: استخدم تنسيق ملف Netscape.

بالإضافة إلى ذلك، هناك أوامر متاحة تقوم بإجراءات إذا قمت بتمرير هذه السلاسل بالضبط:

ALL
يمسح جميع ملفات تعريف الارتباط المخزنة في الذاكرة
SESS
يمسح جميع ملفات تعريف الارتباط الخاصة بالجلسة المخزنة في الذاكرة
FLUSH
يكتب جميع ملفات تعريف الارتباط المعروفة إلى الملف المحدد بواسطة CURLOPT_COOKIEJAR(3)
RELOAD
يقوم بتحميل جميع ملفات تعريف الارتباط من الملفات المحددة بواسطة CURLOPT_COOKIEFILE(3)

الافتراضي

NULL

البروتوكولات

تؤثر هذه الوظيفة على http فقط

مثال

/* استيراد ملف تعريف ارتباط مضمن بتنسيق Netscape. */
#define SEP  "\t"  /* علامة جدولة تفصل الحقول */
int main(void)
{
  const char *my_cookie =
    "example.com"    /* اسم المضيف */
    SEP "FALSE"      /* تضمين النطاقات الفرعية */
    SEP "/"          /* المسار */
    SEP "FALSE"      /* آمن */
    SEP "0"          /* انتهاء الصلاحية بتنسيق وقت العصر. 0 == جلسة */
    SEP "foo"        /* الاسم */
    SEP "bar";       /* القيمة */
  CURL *curl = curl_easy_init();
  if(curl) {
    /* يتم استيراد my_cookie على الفور عبر CURLOPT_COOKIELIST. */
    curl_easy_setopt(curl, CURLOPT_COOKIELIST, my_cookie);
    /* لن يتم استيراد قائمة ملفات تعريف الارتباط في cookies.txt إلا قبل
       إجراء عملية نقل مباشرة. سيتم تخطي ملفات تعريف الارتباط في القائمة التي لها نفس
       اسم المضيف والمسار والاسم مثل my_cookie. وذلك لأن
       libcurl قد استورد بالفعل my_cookie ويعتبر ملف تعريف ارتباط "حي".
       ملف تعريف الارتباط الحي لا يتم استبداله بواحد مقروء من ملف.
    */
    curl_easy_setopt(curl, CURLOPT_COOKIEFILE, "cookies.txt");  /* استيراد */
    /* يتم تصدير ملفات تعريف الارتباط بعد استدعاء curl_easy_cleanup. قد يكون الخادم
       أضاف أو حذف أو عدّل ملفات تعريف الارتباط بحلول ذلك الوقت. ملفات تعريف الارتباط التي
       تم تخطيها عند الاستيراد لا يتم تصديرها.
    */
    curl_easy_setopt(curl, CURLOPT_COOKIEJAR, "cookies.txt");  /* تصدير */
    curl_easy_perform(curl);  /* تم استيراد ملفات تعريف الارتباط من cookies.txt */
    curl_easy_cleanup(curl);  /* تم تصدير ملفات تعريف الارتباط إلى cookies.txt */
  }
}
