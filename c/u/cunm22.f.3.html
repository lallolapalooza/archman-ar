subroutine cunm22 (character side, character trans, integer m, integer n, integer n1, integer n2, complex, dimension( ldq, * ) q, integer ldq, complex, dimension( ldc, * ) c, integer ldc, complex, dimension( * ) work, integer lwork, integer info)

CUNM22 يضرب مصفوفة عامة بمصفوفة وحدوية شريطية.

الغرض

!>
!>  CUNM22 يكتب فوق المصفوفة العامة المعقدة M-by-N المسماة C بـ
!>
!>                  SIDE = 'L'     SIDE = 'R'
!>  TRANS = 'N':      Q * C          C * Q
!>  TRANS = 'C':      Q**H * C       C * Q**H
!>
!>  حيث Q هي مصفوفة وحدوية معقدة من الرتبة NQ، حيث NQ = M إذا كان
!>  SIDE = 'L' و NQ = N إذا كان SIDE = 'R'.
!>  المصفوفة الوحدوية Q تعالج بنية كتلة 2x2
!>
!>         [  Q11  Q12  ]
!>     Q = [            ]
!>         [  Q21  Q22  ],
!>
!>  حيث Q12 هي مصفوفة مثلثية سفلية بحجم N1xN1 و Q21 هي مصفوفة مثلثية علوية
!>  بحجم N2xN2.
!>

المعلمات

SIDE 
!>          SIDE is CHARACTER*1
!>          = 'L': طبق Q أو Q**H من اليسار؛
!>          = 'R': طبق Q أو Q**H من اليمين.
!>

TRANS

!>          TRANS is CHARACTER*1
!>          = 'N':  طبق Q (بدون تبديل)؛
!>          = 'C':  طبق Q**H (مبدل مترافق).
!>

M

!>          M is INTEGER
!>          عدد صفوف المصفوفة C. M >= 0.
!>

N

!>          N is INTEGER
!>          عدد أعمدة المصفوفة C. N >= 0.
!>

N1
N2

!>          N1 is INTEGER
!>          N2 is INTEGER
!>          أبعاد Q12 و Q21 على التوالي. N1, N2 >= 0.
!>          يجب استيفاء المتطلب التالي:
!>          N1 + N2 = M إذا كان SIDE = 'L' و N1 + N2 = N إذا كان SIDE = 'R'.
!>

Q

!>          Q is COMPLEX array, dimension
!>                              (LDQ,M) if SIDE = 'L'
!>                              (LDQ,N) if SIDE = 'R'
!>

LDQ

!>          LDQ is INTEGER
!>          البعد الرئيسي للمصفوفة Q.
!>          LDQ >= max(1,M) if SIDE = 'L'; LDQ >= max(1,N) if SIDE = 'R'.
!>

C

!>          C is COMPLEX array, dimension (LDC,N)
!>          عند الإدخال، المصفوفة C بحجم M-by-N.
!>          عند الخروج، يتم الكتابة فوق C بـ Q*C أو Q**H*C أو C*Q**H أو C*Q.
!>

LDC

!>          LDC is INTEGER
!>          البعد الرئيسي للمصفوفة C. LDC >= max(1,M).
!>

WORK

!>          WORK is COMPLEX array, dimension (MAX(1,LWORK))
!>          عند الخروج، إذا كان INFO = 0، فإن WORK(1) يعيد LWORK الأمثل.
!>

LWORK

!>          LWORK is INTEGER
!>          بعد المصفوفة WORK.
!>          إذا كان SIDE = 'L'، فإن LWORK >= max(1,N)؛
!>          إذا كان SIDE = 'R'، فإن LWORK >= max(1,M).
!>          للحصول على الأداء الأمثل، LWORK >= M*N.
!>
!>          إذا كان LWORK = -1، فيفترض استعلام مساحة العمل؛ يقوم الروتين
!>          بحساب فقط الحجم الأمثل للمصفوفة WORK، ويعيد
!>          هذه القيمة كالإدخال الأول في المصفوفة WORK، ولا يتم إصدار أي رسالة خطأ
!>          متعلقة بـ LWORK بواسطة XERBLA.
!>

INFO

!>          INFO is INTEGER
!>          = 0:  انتهاء ناجح
!>          < 0:  إذا كان INFO = -i، فإن المعامل i كان له قيمة غير قانونية
!>
