توثيق الوظيفة/الروتين الفرعي

subroutine cungqr (integer m, integer n, integer k, complex, dimension( lda, * ) a, integer lda, complex, dimension( * ) tau, complex, dimension( * ) work, integer lwork, integer info)

CUNGQR

الغرض:

!>
!> CUNGQR تنشئ مصفوفة معقدة M × N ذات أعمدة متعامدة،
!> والتي يتم تعريفها كأول N أعمدة من حاصل ضرب K من العواكس الأولية
!> من الرتبة M
!>
!>       Q  =  H(1) H(2) . . . H(k)
!>
!> كما تم إرجاعه بواسطة CGEQRF.
!>

المعلمات

M 
!>          M هو عدد صحيح (INTEGER)
!>          عدد صفوف المصفوفة Q. M >= 0.
!>

N

!>          N هو عدد صحيح (INTEGER)
!>          عدد أعمدة المصفوفة Q. M >= N >= 0.
!>

K

!>          K هو عدد صحيح (INTEGER)
!>          عدد العواكس الأولية التي ينتج عنها المصفوفة Q.
!>          N >= K >= 0.
!>

A

!>          A هي مصفوفة معقدة (COMPLEX)، البعد (LDA,N)
!>          عند الإدخال، يجب أن يحتوي العمود i على المتجه الذي
!>          يعرف العاكس الأولي H(i)، لـ i = 1,2,...,k، كما
!>          تم إرجاعه بواسطة CGEQRF في أول K أعمدة من مصفوفة الوسيط A.
!>          عند الخروج، تكون المصفوفة M × N هي Q.
!>

LDA

!>          LDA هو عدد صحيح (INTEGER)
!>          البعد الأول للمصفوفة A. LDA >= max(1,M).
!>

TAU

!>          TAU هي مصفوفة معقدة (COMPLEX)، البعد (K)
!>          يجب أن يحتوي TAU(i) على العامل القياسي للعاكس
!>          الأولي H(i)، كما تم إرجاعه بواسطة CGEQRF.
!>

WORK

!>          WORK هي مصفوفة معقدة (COMPLEX)، البعد (MAX(1,LWORK))
!>          عند الخروج، إذا كان INFO = 0، فإن WORK(1) يعيد قيمة LWORK المثلى.
!>

LWORK

!>          LWORK هو عدد صحيح (INTEGER)
!>          بعد المصفوفة WORK. LWORK >= max(1,N).
!>          لأداء مثالي LWORK >= N*NB، حيث NB هو حجم الكتلة الأمثل.
!>
!>          إذا كان LWORK = -1، يتم افتراض استعلام عن مساحة العمل؛
!>          يقوم الروتين فقط بحساب الحجم الأمثل للمصفوفة WORK،
!>          ويعيد هذه القيمة كمدخل أول للمصفوفة WORK، ولا يصدر
!>          أي رسالة خطأ متعلقة بـ LWORK بواسطة XERBLA.
!>

INFO

!>          INFO هو عدد صحيح (INTEGER)
!>          = 0: إنهاء ناجح
!>          < 0: إذا كان INFO = -i، فإن الوسيط i لديه قيمة غير قانونية
!>
