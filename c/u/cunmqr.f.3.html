توثيق الوظيفة/الإجراء الفرعي

subroutine cunmqr (character side, character trans, integer m, integer n, integer k, complex, dimension( lda, * ) a, integer lda, complex, dimension( * ) tau, complex, dimension( ldc, * ) c, integer ldc, complex, dimension( * ) work, integer lwork, integer info)

CUNMQR

الغرض:

!>
!> CUNMQR تستبدل المصفوفة العامة المعقدة M-by-N المسماة C بـ
!>
!>                 SIDE = 'L'     SIDE = 'R'
!> TRANS = 'N':      Q * C          C * Q
!> TRANS = 'C':      Q**H * C       C * Q**H
!>
!> حيث Q هي مصفوفة وحدوية معقدة معرفة كناتج لـ k
!> عواكس أولية
!>
!>       Q = H(1) H(2) . . . H(k)
!>
!> كما تم إرجاعها بواسطة CGEQRF. Q من الرتبة M إذا كانت SIDE = 'L' ومن الرتبة N
!> إذا كانت SIDE = 'R'.
!>

المعلمات

SIDE 
!>          SIDE هو CHARACTER*1
!>          = 'L': تطبيق Q أو Q**H من اليسار؛
!>          = 'R': تطبيق Q أو Q**H من اليمين.
!>

TRANS

!>          TRANS هو CHARACTER*1
!>          = 'N':  لا يوجد تبديل، تطبيق Q؛
!>          = 'C':  تبديل منسجم، تطبيق Q**H.
!>

M

!>          M هو INTEGER
!>          عدد صفوف المصفوفة C. M >= 0.
!>

N

!>          N هو INTEGER
!>          عدد أعمدة المصفوفة C. N >= 0.
!>

K

!>          K هو INTEGER
!>          عدد العواكس الأولية التي يحدد ناتجها
!>          المصفوفة Q.
!>          إذا كانت SIDE = 'L', M >= K >= 0؛
!>          إذا كانت SIDE = 'R', N >= K >= 0.
!>

A

!>          A هي مصفوفة COMPLEX، ذات بعد (LDA,K)
!>          يجب أن يحتوي العمود i على المتجه الذي يحدد
!>          العاكس الأولي H(i)، لـ i = 1,2,...,k، كما تم إرجاعه بواسطة
!>          CGEQRF في الأعمدة k الأولى من مصفوفة الوسيط A.
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرائد للمصفوفة A.
!>          إذا كانت SIDE = 'L', LDA >= max(1,M)؛
!>          إذا كانت SIDE = 'R', LDA >= max(1,N).
!>

TAU

!>          TAU هي مصفوفة COMPLEX، ذات بعد (K)
!>          يجب أن تحتوي TAU(i) على العامل العددي للعواكس
!>          الأولية H(i)، كما تم إرجاعه بواسطة CGEQRF.
!>

C

!>          C هي مصفوفة COMPLEX، ذات بعد (LDC,N)
!>          عند الإدخال، المصفوفة C بحجم M-by-N.
!>          عند الخروج، يتم استبدال C بـ Q*C أو Q**H*C أو C*Q**H أو C*Q.
!>

LDC

!>          LDC هو INTEGER
!>          البعد الرائد للمصفوفة C. LDC >= max(1,M).
!>

WORK

!>          WORK هي مصفوفة COMPLEX، ذات بعد (MAX(1,LWORK))
!>          عند الخروج، إذا كان INFO = 0، فإن WORK(1) ترجع LWORK الأمثل.
!>

LWORK

!>          LWORK هو INTEGER
!>          بعد المصفوفة WORK.
!>          إذا كانت SIDE = 'L', LWORK >= max(1,N)؛
!>          إذا كانت SIDE = 'R', LWORK >= max(1,M).
!>          لأداء جيد، يجب أن يكون LWORK أكبر بشكل عام.
!>
!>          إذا كان LWORK = -1، فيفترض استعلام عن مساحة العمل؛ يقوم الإجراء الفرعي
!>          بحساب حجم مساحة العمل الأمثل فقط، ويعيد
!>          هذه القيمة كالإدخال الأول لمصفوفة WORK، ولا يتم إصدار رسالة خطأ
!>          متعلقة بـ LWORK بواسطة XERBLA.
!>

INFO

!>          INFO هو INTEGER
!>          = 0:  انتهاء ناجح
!>          < 0:  إذا كان INFO = -i، فإن المعلمة i كانت لها قيمة غير قانونية
!>
