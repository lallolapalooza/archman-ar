curl_easy_ssls_import(3)

الحزمة: curl

التراخيص: MIT

curl_easy_ssls_import(3) دليل الدوال المكتبية curl_easy_ssls_import(3)

الاسم

curl_easy_ssls_import - استيراد جلسات SSL

الملخص

#include <curl/curl.h>
CURLcode curl_easy_ssls_import(CURL *handle,
                               const char *session_key,
                               const unsigned char *shmac, size_t shmac_len,
                               const unsigned char *sdata, size_t sdata_len);

الوصف

تقوم هذه الدالة باستيراد تذكرة جلسة SSL تم تصديرها مسبقًا. يجب دائمًا توفير sdata و sdata_len. إذا كان session_key هو NULL، فيجب توفير shmac و shmac_len كما تم استلامها أثناء التصدير. انظر curl_easy_ssls_export(3) لوصفها.

قد يفشل استيراد تذاكر الجلسات من إصدارات curl الأخرى بسبب تغييرات في معالجة shmac أو sdata. يتم تجاهل تذكرة الجلسة التي انتهت صلاحيتها بالفعل بصمت.

البروتوكولات

تؤثر هذه الوظيفة على جميع البروتوكولات المستندة إلى TLS: HTTPS، FTPS، IMAPS، POP3S، SMTPS إلخ.

تعمل هذه الخيار فقط مع خلفيات TLS التالية: GnuTLS، OpenSSL، mbedTLS و wolfSSL

مثال

int main(void)
{
  CURLSHcode sh;
  CURLSH *share = curl_share_init();
  CURLcode rc;
  CURL *curl;
  sh = curl_share_setopt(share, CURLSHOPT_SHARE, CURL_LOCK_DATA_SSL_SESSION);
  if(sh)
    printf("Error: %s\n", curl_share_strerror(sh));
  curl = curl_easy_init();
  if(curl) {
    extern unsigned char *shmac, *sdata;
    size_t hlen = 4, slen = 5;
    curl_easy_setopt(curl, CURLOPT_SHARE, share);
    /* قراءة shmac و sdata من التخزين */
    rc = curl_easy_ssls_import(curl, NULL, shmac, hlen, sdata, slen);
    /* التنظيف دائمًا */
    curl_easy_cleanup(curl);
  }
  curl_share_cleanup(share);
}

التوفر

تمت الإضافة في curl 8.12.0

قيمة الإرجاع

ترجع هذه الدالة قيمة CURLcode تشير إلى النجاح أو الخطأ.

CURLE_OK (0) تعني أن كل شيء كان على ما يرام، وتشير القيمة غير الصفرية إلى حدوث خطأ، انظر libcurl-errors(3). إذا تم تعيين CURLOPT_ERRORBUFFER(3) باستخدام curl_easy_setopt(3)، فقد تكون هناك رسالة خطأ مخزنة في مخزن الأخطاء عند إرجاع قيمة غير صفرية.
