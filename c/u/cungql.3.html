وثائق الوظيفة/الإجراء الفرعي

subroutine cungql (integer m, integer n, integer k, complex, dimension( lda, * ) a, integer lda, complex, dimension( * ) tau, complex, dimension( * ) work, integer lwork, integer info)

CUNGQL

الغرض:

!>
!> CUNGQL ينشئ مصفوفة معقدة Q بحجم M × N بأعمدة متعامدة،
!> والتي تُعرّف بأنها آخر N أعمدة من ناتج K من الانعكاسات الأولية
!> ذات الرتبة M
!>
!>       Q  =  H(k) . . . H(2) H(1)
!>
!> كما تم إرجاعها بواسطة CGEQLF.
!>

المعلمات

M 
!>          M هو عدد صحيح (INTEGER)
!>          عدد الصفوف في المصفوفة Q. M >= 0.
!>

N

!>          N هو عدد صحيح (INTEGER)
!>          عدد الأعمدة في المصفوفة Q. M >= N >= 0.
!>

K

!>          K هو عدد صحيح (INTEGER)
!>          عدد الانعكاسات الأولية التي يحدد ناتجها المصفوفة Q.
!>          N >= K >= 0.
!>

A

!>          A هي مصفوفة معقدة (COMPLEX array)، بحجم (LDA,N)
!>          عند الإدخال، يجب أن يحتوي العمود (n-k+i) على المتجه الذي
!>          يحدد الانعكاس الأولي H(i)، من أجل i = 1,2,...,k، كما
!>          تم إرجاعه بواسطة CGEQLF في آخر k أعمدة من مصفوفة الوسيط A الخاصة به.
!>          عند الخروج، تكون المصفوفة Q بحجم M × N.
!>

LDA

!>          LDA هو عدد صحيح (INTEGER)
!>          البعد الأول للمصفوفة A. LDA >= max(1,M).
!>

TAU

!>          TAU هي مصفوفة معقدة (COMPLEX array)، بحجم (K)
!>          يجب أن تحتوي TAU(i) على العامل العددي للانعكاس الأولي H(i)،
!>          كما تم إرجاعه بواسطة CGEQLF.
!>

WORK

!>          WORK هي مصفوفة معقدة (COMPLEX array)، بحجم (MAX(1,LWORK))
!>          عند الخروج، إذا كان INFO = 0، فإن WORK(1) يرجع LWORK الأمثل.
!>

LWORK

!>          LWORK هو عدد صحيح (INTEGER)
!>          بعد المصفوفة WORK. LWORK >= max(1,N).
!>          للحصول على أداء أمثل LWORK >= N*NB، حيث NB هو حجم الكتلة الأمثل.
!>
!>          إذا كان LWORK = -1، فيُفترض استعلام عن مساحة العمل؛ تقوم الروتين
!>          فقط بحساب الحجم الأمثل للمصفوفة WORK، وترجع هذا الحجم كأول مدخل
!>          في المصفوفة WORK، ولا يصدر XERBLA أي رسالة خطأ متعلقة بـ LWORK.
!>

INFO

!>          INFO هو عدد صحيح (INTEGER)
!>          = 0:  خروج ناجح
!>          < 0:  إذا كان INFO = -i، فإن الوسيط i له قيمة غير قانونية
!>
