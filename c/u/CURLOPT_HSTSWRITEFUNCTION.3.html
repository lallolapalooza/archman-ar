CURLOPT_HSTSWRITEFUNCTION(3)

الحزمة: curl

التراخيص: MIT

CURLOPT_HSTSWRITEFUNCTION(3) دليل وظائف المكتبة CURLOPT_HSTSWRITEFUNCTION(3)

اسم

CURLOPT_HSTSWRITEFUNCTION - دالة رد اتصال لكتابة مضيفي HSTS

ملخص

#include <curl/curl.h>
struct curl_hstsentry {
  char *name;
  size_t namelen;
  unsigned int includeSubDomains:1;
  char expire[18]; /* YYYYMMDD HH:MM:SS [null-terminated] */
};
struct curl_index {
  size_t index; /* the provided entry's "index" or count */
  size_t total; /* total number of entries to save */
};
CURLSTScode hstswrite(CURL *easy, struct curl_hstsentry *sts,
                      struct curl_index *count, void *clientp);
CURLcode curl_easy_setopt(CURL *handle, CURLOPT_HSTSWRITEFUNCTION, hstswrite);

الوصف

مرر مؤشرًا إلى دالة رد الاتصال الخاصة بك، وفقًا للنموذج الأولي الموضح أعلاه.

يتم استدعاء دالة رد الاتصال هذه بشكل متكرر بواسطة libcurl للسماح للتطبيق بتخزين ذاكرة التخزين المؤقت لـ HSTS في الذاكرة عندما يكون libcurl على وشك تجاهلها.

اضبط الوسيط clientp باستخدام الخيار CURLOPT_HSTSWRITEDATA(3) أو سيكون NULL. عند استدعاء دالة رد الاتصال، يشير المؤشر sts إلى بنية مملوءة: اقرأ اسم المضيف إلى 'name' (وهو بطول namelen بايت وينتهي بصفر. الحقل includeSubDomains غير صفري إذا كانت الإدخالة تطابق النطاقات الفرعية. السلسلة expire هي ختم تاريخ سلسلة تنتهي بصفر باستخدام الصيغة YYYYMMDD HH:MM:SS.

يجب أن تُرجع دالة رد الاتصال CURLSTS_OK إذا نجحت وكانت مستعدة للاستدعاء مرة أخرى (لمضيف آخر) أو CURLSTS_DONE إذا لم يكن هناك المزيد للقيام به. يمكنها أيضًا إرجاع CURLSTS_FAIL للإشارة إلى وجود خطأ.

هذا الخيار لا يمكّن HSTS، تحتاج إلى استخدام CURLOPT_HSTS_CTRL(3) للقيام بذلك.

افتراضي

NULL - لا توجد دالة رد اتصال.

البروتوكولات

تؤثر هذه الوظيفة على http فقط

مثال

struct priv {
  void *custom;
};
static CURLSTScode hswr_cb(CURL *easy, struct curl_hstsentry *sts,
                           struct curl_index *count, void *clientp)
{
  /* save the passed in HSTS data somewhere */
  return CURLSTS_OK;
}
int main(void)
{
  CURL *curl = curl_easy_init();
  if(curl) {
    struct priv my_stuff;
    CURLcode res;
    /* set HSTS read callback */
    curl_easy_setopt(curl, CURLOPT_HSTSWRITEFUNCTION, hswr_cb);
    /* pass in suitable argument to the callback */
    curl_easy_setopt(curl, CURLOPT_HSTSWRITEDATA, &my_stuff);
    res = curl_easy_perform(curl);
  }
}
