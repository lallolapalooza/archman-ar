CURLOPT_SSH_KEYFUNCTION(3)

الحزمة: curl

التراخيص: MIT

CURLOPT_SSH_KEYFUNCTION(3) دليل وظائف المكتبة CURLOPT_SSH_KEYFUNCTION(3)

الاسم

CURLOPT_SSH_KEYFUNCTION - دالة رد اتصال لمنطق مطابقة المضيف المعروف

الملخص

#include <curl/curl.h>
enum curl_khstat {
  CURLKHSTAT_FINE_ADD_TO_FILE,
  CURLKHSTAT_FINE,
  CURLKHSTAT_REJECT, /* رفض الاتصال، وإعادة خطأ */
  CURLKHSTAT_DEFER,  /* لا تقبلها، ولكن لا يمكننا الإجابة في الوقت الحالي. تسبب خطأ CURLE_PEER_FAILED_VERIFICATION ولكن الاتصال يبقى سليماً */
  CURLKHSTAT_FINE_REPLACE
};
enum curl_khmatch {
  CURLKHMATCH_OK,       /* تطابق */
  CURLKHMATCH_MISMATCH, /* تم العثور على المضيف، عدم تطابق المفتاح */
  CURLKHMATCH_MISSING,  /* لم يتم العثور على مضيف/مفتاح مطابق */
};
struct curl_khkey {
  const char *key; /* يشير إلى سلسلة نصية منتهية بـ null مشفرة بـ
                      base64 إذا كان الطول صفرًا، وإلا فإنه يشير إلى البيانات "الخام" */
  size_t len;
  enum curl_khtype keytype;
};
int ssh_keycallback(CURL *easy,
                    const struct curl_khkey *knownkey,
                    const struct curl_khkey *foundkey,
                    enum curl_khmatch match,
                    void *clientp);
CURLcode curl_easy_setopt(CURL *handle, CURLOPT_SSH_KEYFUNCTION,
                          ssh_keycallback);

الوصف

مرر مؤشرًا إلى دالة رد الاتصال الخاصة بك، والتي يجب أن تتطابق مع النموذج الأولي الموضح أعلاه.

يتم استدعاؤها عند اكتمال مطابقة المضيف المعروف، للسماح للتطبيق بالتصرف وتحديد كيفية المضي قدمًا لـ libcurl. لا يتم استدعاء دالة رد الاتصال إلا إذا تم تعيين CURLOPT_SSH_KNOWNHOSTS(3) أيضًا.

يتم تمرير دالة رد الاتصال هذه بمقبض curl، والمفتاح من ملف known_hosts knownkey، والمفتاح من الموقع البعيد foundkey، ومعلومات من libcurl حول حالة المطابقة، ومؤشر مخصص (تم تعيينه باستخدام CURLOPT_SSH_KEYDATA(3)). يجب أن تعيد أحد رموز الإرجاع التالية لإخبار libcurl بكيفية التصرف:

CURLKHSTAT_FINE_REPLACE
المضيف + المفتاح الجديد مقبولان وتقوم libcurl باستبدال المضيف + المفتاح القديم في ملف known_hosts قبل المتابعة بالاتصال. هذا يضيف أيضًا المضيف + المفتاح الجديد إلى مجموعة known_host المحفوظة في الذاكرة إذا لم يكن موجودًا بالفعل هناك. تتم إضافة البيانات إلى الملف عن طريق استبدال الملف بالكامل بنسخة جديدة، لذلك يجب أن تسمح أذونات الملف بذلك. (تمت إضافته في 7.73.0)
CURLKHSTAT_FINE_ADD_TO_FILE
المضيف + المفتاح مقبولان وتقوم libcurl بإلحاقه بملف known_hosts قبل المتابعة بالاتصال. هذا يضيف أيضًا المضيف + المفتاح إلى مجموعة known_host المحفوظة في الذاكرة إذا لم يكن موجودًا بالفعل هناك. تتم إضافة البيانات إلى الملف عن طريق استبدال الملف بالكامل بنسخة جديدة، لذلك يجب أن تسمح أذونات الملف بذلك.
CURLKHSTAT_FINE
المضيف + المفتاح مقبولان وتتابع libcurl بالاتصال. هذا يضيف أيضًا المضيف + المفتاح إلى مجموعة known_host المحفوظة في الذاكرة إذا لم يكن موجودًا بالفعل هناك.
CURLKHSTAT_REJECT
المضيف + المفتاح مرفوضان. ترفض libcurl استمرار الاتصال ويتم إغلاقه.
CURLKHSTAT_DEFER
المضيف + المفتاح مرفوضان، ولكن يطلب من اتصال SSH البقاء على قيد الحياة. يمكن استخدام هذه الميزة عندما تريد التطبيق العودة والتصرف بشأن وضع المضيف + المفتاح ثم إعادة المحاولة دون الحاجة إلى عبء إعداده من البداية مرة أخرى.

البروتوكولات

تؤثر هذه الوظيفة على scp و sftp

مثال

struct mine {
  void *custom;
};
static int keycb(CURL *easy,
                 const struct curl_khkey *knownkey,
                 const struct curl_khkey *foundkey,
                 enum curl_khmatch match,
                 void *clientp)
{
  /* 'clientp' يشير إلى هيكل callback_data */
  /* التحقيق في الوضع وإعادة القيمة الصحيحة */
  return CURLKHSTAT_FINE_ADD_TO_FILE;
}
int main(void)
{
  CURL *curl = curl_easy_init();
  if(curl) {
    struct mine callback_data;
    curl_easy_setopt(curl, CURLOPT_URL, "sftp://example.com/thisfile.txt");
    curl_easy_setopt(curl, CURLOPT_SSH_KEYFUNCTION, keycb);
    curl_easy_setopt(curl, CURLOPT_SSH_KEYDATA, &callback_data);
    curl_easy_setopt(curl, CURLOPT_SSH_KNOWNHOSTS, "/home/user/known_hosts");
    curl_easy_perform(curl);
}
}

قيمة الإرجاع

curl_easy_setopt(3) تعيد رمز CURLcode يشير إلى النجاح أو الخطأ.

CURLE_OK (0) يعني أن كل شيء على ما يرام، والقيمة غير الصفرية تعني حدوث خطأ، انظر libcurl-errors(3).
