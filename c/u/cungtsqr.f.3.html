توثيق الوظيفة/الإجراء الفرعي

subroutine cungtsqr (integer m, integer n, integer mb, integer nb, complex, dimension( lda, * ) a, integer lda, complex, dimension( ldt, * ) t, integer ldt, complex, dimension( * ) work, integer lwork, integer info)

CUNGTSQR

الغرض:

!>
!> CUNGTSQR يولد مصفوفة Q_out معقدة بحجم M × N بأعمدة متعامدة،
!> وهي الأعمدة N الأولى من حاصل ضرب مصفوفات وحدوية معقدة
!> حجمها M والتي تم إرجاعها بواسطة CLATSQR
!>
!>      Q_out = الأعمدة_N_الأولى( Q(1)_in * Q(2)_in * ... * Q(k)_in ).
!>
!> انظر توثيق CLATSQR.
!>

المعلمات

M 
!>          M هو عدد صحيح (INTEGER)
!>          عدد صفوف المصفوفة A. M >= 0.
!>

N

!>          N هو عدد صحيح (INTEGER)
!>          عدد أعمدة المصفوفة A. M >= N >= 0.
!>

MB

!>          MB هو عدد صحيح (INTEGER)
!>          حجم كتلة الصف المستخدمة بواسطة CLATSQR لإرجاع
!>          المصفوفتين A و T. MB > N.
!>          (ملاحظة: إذا كان MB > M، فسيتم استخدام M بدلاً من MB
!>          كحجم كتلة الصف).
!>

NB

!>          NB هو عدد صحيح (INTEGER)
!>          حجم كتلة العمود المستخدمة بواسطة CLATSQR لإرجاع
!>          المصفوفتين A و T. NB >= 1.
!>          (ملاحظة: إذا كان NB > N، فسيتم استخدام N بدلاً من NB
!>          كحجم كتلة العمود).
!>

A

!>          A هي مصفوفة معقدة (COMPLEX)، البعد (LDA,N)
!>
!>          عند الإدخال:
!>
!>             العناصر على القطر وفوقه لا يتم الوصول إليها.
!>             العناصر الموجودة تحت القطر تمثل المصفوفة شبه المنحرفة
!>             السفلية الموحدة V المحسوبة بواسطة CLATSQR
!>             التي تحدد مصفوفات الإدخال Q_in(k) (الواحدات على
!>             القطر لا يتم تخزينها) (نفس تنسيق الإخراج A
!>             تحت القطر في CLATSQR).
!>
!>          عند الإخراج:
!>
!>             تحتوي المصفوفة A على مصفوفة متعامدة Q_out بحجم M × N،
!>             أي أن أعمدة A هي متجهات وحدة متعامدة.
!>

LDA

!>          LDA هو عدد صحيح (INTEGER)
!>          البعد الرائد للمصفوفة A. LDA >= max(1,M).
!>

T

!>          T هي مصفوفة معقدة (COMPLEX)،
!>          البعد (LDT, N * NIRB)
!>          حيث NIRB = عدد كتل الصف المدخلة
!>                     = MAX( 1, CEIL((M-N)/(MB-N)) )
!>          لتكن NICB = عدد كتل العمود المدخلة
!>                   = CEIL(N/NB)
!>
!>          المعكسات القطاعية المثلثية العلوية المستخدمة لتحديد
!>          مصفوفات الإدخال Q_in(k), k=(1:NIRB*NICB). يتم تخزين
!>          المعكسات القطاعية في شكل مضغوط في NIRB تسلسلات
!>          معكوسات القطاعات. يتم تخزين كل من تسلسلات NIRB المعكوسات
!>          القطاعية في كتلة عمود أكبر بحجم NB × N من T وتتكون
!>          من NICB كتل عمود أصغر بحجم NB × NB مثلثية علوية.
!>          (نفس تنسيق الإخراج T في CLATSQR).
!>

LDT

!>          LDT هو عدد صحيح (INTEGER)
!>          البعد الرائد للمصفوفة T.
!>          LDT >= max(1,min(NB1,N)).
!>

WORK

!>          (مساحة العمل) مصفوفة معقدة (COMPLEX)، البعد (MAX(2,LWORK))
!>          عند الإخراج، إذا كان INFO = 0، فإن WORK(1) يعيد LWORK المثلى.
!>

LWORK

!>          LWORK هو عدد صحيح (INTEGER)
!>          بعد المصفوفة WORK. LWORK >= (M+NB)*N.
!>          إذا كان LWORK = -1، فسيتم افتراض استعلام مساحة العمل.
!>          تقوم الروتين فقط بحساب الحجم الأمثل لمصفوفة WORK،
!>          وتعيد هذا الحجم كالإدخال الأول لمصفوفة WORK،
!>          ولا يتم إصدار أي رسالة خطأ متعلقة بـ LWORK بواسطة XERBLA.
!>

INFO

!>          INFO هو عدد صحيح (INTEGER)
!>          = 0:  نجاح التنفيذ
!>          < 0:  إذا كان INFO = -i، فإن المعلمة i كانت لها قيمة غير قانونية
!>
