subroutine cunmrq (character side, character trans, integer m, integer n, integer k, complex, dimension( lda, * ) a, integer lda, complex, dimension( * ) tau, complex, dimension( ldc, * ) c, integer ldc, complex, dimension( * ) work, integer lwork, integer info)

CUNMRQ

الغرض:

!>
!> CUNMRQ يستبدل المصفوفة المعقدة العامة M-by-N ذات الـ C بـ
!>
!>                 SIDE = 'L'     SIDE = 'R'
!> TRANS = 'N':      Q * C          C * Q
!> TRANS = 'C':      Q**H * C       C * Q**H
!>
!> حيث Q هي مصفوفة معقدة وحدوية معرفة كناتج لـ k
!> عواكس أولية
!>
!>       Q = H(1)**H H(2)**H . . . H(k)**H
!>
!> كما تم إرجاعها بواسطة CGERQF. Q من الدرجة M إذا كان SIDE = 'L' ومن الدرجة N
!> إذا كان SIDE = 'R'.
!>

المعلمات

SIDE 
!>          SIDE هو CHARACTER*1
!>          = 'L': تطبيق Q أو Q**H من اليسار؛
!>          = 'R': تطبيق Q أو Q**H من اليمين.
!>

TRANS

!>          TRANS هو CHARACTER*1
!>          = 'N':  لا يوجد تبديل، تطبيق Q؛
!>          = 'C':  تبديل مرافق، تطبيق Q**H.
!>

M

!>          M هو INTEGER
!>          عدد صفوف المصفوفة C. M >= 0.
!>

N

!>          N هو INTEGER
!>          عدد أعمدة المصفوفة C. N >= 0.
!>

K

!>          K هو INTEGER
!>          عدد العواكس الأولية التي يحدد ناتجها
!>          المصفوفة Q.
!>          إذا كان SIDE = 'L'، فإن M >= K >= 0؛
!>          إذا كان SIDE = 'R'، فإن N >= K >= 0.
!>

A

!>          A هو مصفوفة COMPLEX، أبعادها
!>                               (LDA,M) إذا كان SIDE = 'L'،
!>                               (LDA,N) إذا كان SIDE = 'R'
!>          يجب أن يحتوي الصف i على المتجه الذي يحدد
!>          العاكس الأولي H(i)، لكل i = 1,2,...,k، كما تم إرجاعه بواسطة
!>          CGERQF في الصفوف k الأخيرة من وسيط المصفوفة A الخاص بها.
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرائد للمصفوفة A. LDA >= max(1,K).
!>

TAU

!>          TAU هو مصفوفة COMPLEX، أبعادها (K)
!>          يجب أن يحتوي TAU(i) على العامل العددي
!>          للعاكس الأولي H(i)، كما تم إرجاعه بواسطة CGERQF.
!>

C

!>          C هو مصفوفة COMPLEX، أبعادها (LDC,N)
!>          عند الإدخال، المصفوفة C ذات الأبعاد M-by-N.
!>          عند الخروج، يتم استبدال C بـ Q*C أو Q**H*C أو C*Q**H أو C*Q.
!>

LDC

!>          LDC هو INTEGER
!>          البعد الرائد للمصفوفة C. LDC >= max(1,M).
!>

WORK

!>          WORK هو مصفوفة COMPLEX، أبعادها (MAX(1,LWORK))
!>          عند الخروج، إذا كان INFO = 0، يقوم WORK(1) بإرجاع LWORK الأمثل.
!>

LWORK

!>          LWORK هو INTEGER
!>          بعد المصفوفة WORK.
!>          إذا كان SIDE = 'L'، فإن LWORK >= max(1,N)؛
!>          إذا كان SIDE = 'R'، فإن LWORK >= max(1,M).
!>          لأداء جيد، يجب أن يكون LWORK أكبر بشكل عام.
!>
!>          إذا كان LWORK = -1، فإنه يُفترض استعلام مساحة العمل؛ تقوم الروتين
!>          بحساب حجم مساحة العمل الأمثل فقط، وإرجاع
!>          هذه القيمة كالإدخال الأول للمصفوفة WORK، ولا يصدر XERBLA أي رسالة خطأ
!>          متعلقة بـ LWORK.
!>

INFO

!>          INFO هو INTEGER
!>          = 0:  خروج ناجح
!>          < 0:  إذا كان INFO = -i، فقد كان الوسيط i ذو قيمة غير قانونية
!>
