CURLOPT_UPLOAD_FLAGS(3)

الحزمة: curl

التراخيص: MIT

CURLOPT_UPLOAD_FLAGS(3) دليل وظائف المكتبة CURLOPT_UPLOAD_FLAGS(3)

اسم

CURLOPT_UPLOAD_FLAGS - أعلام التحميل لـ IMAP

ملخص

#include <curl/curl.h>
CURLcode curl_easy_setopt(CURL *handle, CURLOPT_UPLOAD_FLAGS, long bitmask);

الوصف

مرر قيمة من نوع long كوسيط، والتي يتم تعيينها إلى قناع بت، لإخبار مكتبة libcurl بالأعلام التي يجب إرسالها إلى الخادم فيما يتعلق بالملفات التي تم تحميلها. تشمل العلامات المدعومة حاليًا CURLULFLAG_ANSWERED، والتي تعين علامة Answered لعمليات تحميل IMAP، و CURLULFLAG_DELETED، التي تعين علامة Deleted لعمليات تحميل IMAP، و CURLULFLAG_DRAFT، التي تعين علامة Draft لعمليات تحميل IMAP، و CURLULFLAG_FLAGGED، التي تعين علامة Flagged لعمليات تحميل IMAP، و CURLULFLAG_SEEN، التي تعين علامة Seen لعمليات تحميل IMAP.

افتراضي

قناع بت يحتوي فقط على علامة CURLULFLAG_SEEN.

البروتوكولات

تؤثر هذه الوظيفة على imap و imaps

مثال

static size_t read_cb(char *ptr, size_t size, size_t nmemb, void *userdata)
{
  FILE *src = userdata;
  /* انسخ أكبر قدر ممكن من البيانات إلى المخزن المؤقت 'ptr'، ولكن لا تزيد عن
     'size' * 'nmemb' بايت */
  size_t retcode = fread(ptr, size, nmemb, src);
  return retcode;
}
int main(void)
{
  CURL *curl = curl_easy_init();
  if(curl) {
    FILE *src = fopen("local-file", "r");
    curl_off_t fsize = 9876; /* اضبط هذا على حجم الملف المدخل */
    /* نريد استخدام وظيفة القراءة الخاصة بنا */
    curl_easy_setopt(curl, CURLOPT_READFUNCTION, read_cb);
    /* تمكين التحميل */
    curl_easy_setopt(curl, CURLOPT_UPLOAD, 1L);
    /* تحديد الهدف */
    curl_easy_setopt(curl, CURLOPT_URL, "imap://example.com:993/mailbox");
    /* توفير اسم المستخدم */
    curl_easy_setopt(curl, CURLOPT_USERNAME, "user@example.com");
    /* توفير كلمة المرور */
    curl_easy_setopt(curl, CURLOPT_PASSWORD, "password");
    /* تحديد أن البريد الذي تم تحميله يجب اعتباره مميزًا */
    curl_easy_setopt(curl, CURLOPT_UPLOAD_FLAGS, CURLULFLAG_FLAGGED);
    /* الآن حدد أي مؤشر سيتم تمريره إلى وظيفتنا الراجعة */
    curl_easy_setopt(curl, CURLOPT_READDATA, src);
    /* تعيين حجم الملف المراد تحميله */
    curl_easy_setopt(curl, CURLOPT_INFILESIZE_LARGE, (curl_off_t)fsize);
    /* تنفيذ التحميل */
    curl_easy_perform(curl);
  }
}

التوفر

تمت إضافته في curl 8.13.0

قيمة الإرجاع

curl_easy_setopt(3) ترجع CURLcode يشير إلى النجاح أو الخطأ.

CURLE_OK (0) يعني أن كل شيء على ما يرام، وأي قيمة غير صفرية تعني حدوث خطأ، انظر libcurl-errors(3).
