CURLOPT_SSH_HOSTKEYDATA(3)

الحزمة: curl

التراخيص: MIT

CURLOPT_SSH_HOSTKEYDATA(3) دليل وظائف المكتبة CURLOPT_SSH_HOSTKEYDATA(3)

الاسم

CURLOPT_SSH_HOSTKEYDATA - مؤشر لتمريره إلى دالة رد اتصال مفتاح مضيف SSH

الاختصار

#include <curl/curl.h>
CURLcode curl_easy_setopt(CURL *handle, CURLOPT_SSH_HOSTKEYDATA, void *pointer);

الوصف

قم بتمرير مؤشر void كمعلمة. يتم تمرير pointer هذا دون تغيير إلى دالة رد الاتصال المعينة باستخدام CURLOPT_SSH_HOSTKEYFUNCTION(3).

الافتراضي

NULL

البروتوكولات

تؤثر هذه الوظيفة على scp و sftp

مثال

struct mine {
  void *custom;
};
static int hostkeycb(void *clientp,   /* CURLOPT_SSH_HOSTKEYDATA */
                     int keytype,     /* CURLKHTYPE */
                     const char *key, /* مفتاح المضيف للتحقق */
                     size_t keylen)   /* طول المفتاح */
{
  /* 'clientp' يشير إلى بنية callback_data */
  /* تحقق من الوضع وأرجع القيمة الصحيحة */
  return CURLKHMATCH_OK;
}
int main(void)
{
  CURL *curl = curl_easy_init();
  if(curl) {
    struct mine callback_data;
    curl_easy_setopt(curl, CURLOPT_URL, "sftp://example.com/thisfile.txt");
    curl_easy_setopt(curl, CURLOPT_SSH_HOSTKEYFUNCTION, hostkeycb);
    curl_easy_setopt(curl, CURLOPT_SSH_HOSTKEYDATA, &callback_data);
    curl_easy_perform(curl);
  }
}

ملاحظات

يعمل فقط مع الواجهة الخلفية لـ libssh2.

التوفر

تمت إضافته في curl 7.84.0

قيمة الإرجاع

curl_easy_setopt(3) ترجع قيمة CURLcode تشير إلى النجاح أو الخطأ.

CURLE_OK (0) يعني أن كل شيء كان على ما يرام، القيمة غير الصفرية تعني حدوث خطأ، انظر libcurl-errors(3).
