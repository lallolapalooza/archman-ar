subroutine cunt01 (character rowcol, integer m, integer n, complex, dimension( ldu, * ) u, integer ldu, complex, dimension( * ) work, integer lwork, real, dimension( * ) rwork, real resid)

CUNT01

الغرض:

!>
!> CUNT01 تتحقق من أن المصفوفة U وحدوية عن طريق حساب النسبة
!>
!>    RESID = norm( I - U*U' ) / ( n * EPS )، إذا كان ROWCOL = 'R'،
!> أو
!>    RESID = norm( I - U'*U ) / ( m * EPS )، إذا كان ROWCOL = 'C'.
!>
!> بدلاً من ذلك، إذا لم تكن هناك مساحة عمل كافية لتشكيل
!> I - U*U' أو I - U'*U، يتم حساب النسبة كـ
!>
!>    RESID = abs( I - U*U' ) / ( n * EPS )، إذا كان ROWCOL = 'R'،
!> أو
!>    RESID = abs( I - U'*U ) / ( m * EPS )، إذا كان ROWCOL = 'C'.
!>
!> حيث EPS هي دقة الآلة. يتم استخدام ROWCOL فقط إذا كان m = n؛
!> إذا كان m > n، يُفترض أن ROWCOL هو 'C'، وإذا كان m < n، يُفترض أن ROWCOL هو
!> 'R'.
!>

المعلمات

ROWCOL 
!>          ROWCOL هو CHARACTER
!>          يحدد ما إذا كان يجب التحقق من صفوف أو أعمدة U
!>          للتعامد.  يستخدم فقط إذا كان M = N.
!>          = 'R':  التحقق من الصفوف المتعامدة لـ U
!>          = 'C':  التحقق من الأعمدة المتعامدة لـ U
!>

M

!>          M هو INTEGER
!>          عدد صفوف المصفوفة U.
!>

N

!>          N هو INTEGER
!>          عدد أعمدة المصفوفة U.
!>

U

!>          U هو مصفوفة COMPLEX، أبعادها (LDU,N)
!>          المصفوفة الوحدوية U. يتم التحقق من U للأعمدة المتعامدة
!>          إذا كان m > n أو إذا كان m = n و ROWCOL = 'C'. يتم التحقق من U
!>          للصفوف المتعامدة إذا كان m < n أو إذا كان m = n و ROWCOL = 'R'.
!>

LDU

!>          LDU هو INTEGER
!>          البعد الرئيسي للمصفوفة U.  LDU >= max(1,M).
!>

WORK

!>          WORK هي مصفوفة COMPLEX، أبعادها (LWORK)
!>

LWORK

!>          LWORK هو INTEGER
!>          طول المصفوفة WORK.  للحصول على أفضل أداء، يجب أن يكون LWORK
!>          على الأقل N*N إذا كان ROWCOL = 'C' أو M*M إذا كان
!>          ROWCOL = 'R'، ولكن سيتم إجراء الاختبار حتى لو كان LWORK = 0.
!>

RWORK

!>          RWORK هي مصفوفة REAL، أبعادها (min(M,N))
!>          يستخدم فقط إذا كان LWORK كبيرًا بما يكفي لاستخدام كود BLAS من المستوى 3.
!>

RESID

!>          RESID هو REAL
!>          RESID = norm( I - U * U' ) / ( n * EPS )، إذا كان ROWCOL = 'R'، أو
!>          RESID = norm( I - U' * U ) / ( m * EPS )، إذا كان ROWCOL = 'C'.
!>
