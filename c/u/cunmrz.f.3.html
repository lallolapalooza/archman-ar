توثيق الوظيفة/الإجراء الفرعي

subroutine cunmrz (character side, character trans, integer m, integer n, integer k, integer l, complex, dimension( lda, * ) a, integer lda, complex, dimension( * ) tau, complex, dimension( ldc, * ) c, integer ldc, complex, dimension( * ) work, integer lwork, integer info)

CUNMRZ

الغرض:

!>
!> CUNMRZ يكتب فوق المصفوفة المعقدة العامة M-by-N المسماة C بـ
!>
!>                 SIDE = 'L'     SIDE = 'R'
!> TRANS = 'N':      Q * C          C * Q
!> TRANS = 'C':      Q**H * C       C * Q**H
!>
!> حيث Q هي مصفوفة وحدوية معقدة معرفة كناتج لـ k
!> انعكاسات ابتدائية
!>
!>       Q = H(1) H(2) . . . H(k)
!>
!> كما تم إرجاعها بواسطة CTZRZF. Q من الدرجة M إذا كان SIDE = 'L' ومن الدرجة N
!> إذا كان SIDE = 'R'.
!>

المعلمات

SIDE 
!>          SIDE هو CHARACTER*1
!>          = 'L': تطبيق Q أو Q**H من اليسار؛
!>          = 'R': تطبيق Q أو Q**H من اليمين.
!>

TRANS

!>          TRANS هو CHARACTER*1
!>          = 'N':  لا يوجد تبديل، تطبيق Q؛
!>          = 'C':  تبديل معكوس، تطبيق Q**H.
!>

M

!>          M هو INTEGER
!>          عدد الصفوف في المصفوفة C. M >= 0.
!>

N

!>          N هو INTEGER
!>          عدد الأعمدة في المصفوفة C. N >= 0.
!>

K

!>          K هو INTEGER
!>          عدد الانعكاسات الابتدائية التي يحدد حاصل ضربها
!>          المصفوفة Q.
!>          إذا كان SIDE = 'L', M >= K >= 0؛
!>          إذا كان SIDE = 'R', N >= K >= 0.
!>

L

!>          L هو INTEGER
!>          عدد الأعمدة في المصفوفة A التي تحتوي على
!>          الجزء الهام من انعكاسات هاوسهولدر.
!>          إذا كان SIDE = 'L', M >= L >= 0، إذا كان SIDE = 'R', N >= L >= 0.
!>

A

!>          A هو مصفوفة COMPLEX، البعد
!>                               (LDA,M) إذا كان SIDE = 'L'،
!>                               (LDA,N) إذا كان SIDE = 'R'
!>          يجب أن يحتوي الصف i على المتجه الذي يحدد
!>          الانعكاس الابتدائي H(i)، لكل i = 1،2،...،k، كما تم إرجاعه بواسطة
!>          CTZRZF في آخر k صفوف من مصفوفة الوسيط A.
!>          يتم تعديل A بواسطة الروتين ولكن يتم استعادتها عند الخروج.
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرائد للمصفوفة A. LDA >= max(1,K).
!>

TAU

!>          TAU هي مصفوفة COMPLEX، البعد (K)
!>          يجب أن يحتوي TAU(i) على العامل العددي للانعكاس
!>          الابتدائي H(i)، كما تم إرجاعه بواسطة CTZRZF.
!>

C

!>          C هي مصفوفة COMPLEX، البعد (LDC,N)
!>          عند الإدخال، مصفوفة C بحجم M-by-N.
!>          عند الخروج، يتم استبدال C بـ Q*C أو Q**H*C أو C*Q**H أو C*Q.
!>

LDC

!>          LDC هو INTEGER
!>          البعد الرائد للمصفوفة C. LDC >= max(1,M).
!>

WORK

!>          WORK هي مصفوفة COMPLEX، البعد (MAX(1,LWORK))
!>          عند الخروج، إذا كان INFO = 0، فإن WORK(1) يعيد LWORK الأمثل.
!>

LWORK

!>          LWORK هو INTEGER
!>          بعد المصفوفة WORK.
!>          إذا كان SIDE = 'L', LWORK >= max(1,N)؛
!>          إذا كان SIDE = 'R', LWORK >= max(1,M).
!>          لأداء جيد، يجب أن يكون LWORK أكبر بشكل عام.
!>
!>          إذا كان LWORK = -1، يتم افتراض استعلام مساحة العمل؛ الروتين
!>          يحسب فقط الحجم الأمثل للمصفوفة WORK، يعيد
!>          هذه القيمة كالإدخال الأول للمصفوفة WORK، ولا يتم إصدار أي رسالة خطأ
!>          متعلقة بـ LWORK بواسطة XERBLA.
!>

INFO

!>          INFO هو INTEGER
!>          = 0:  خروج ناجح
!>          < 0:  إذا كان INFO = -i، كان الوسيط i ذو قيمة غير قانونية
!>
