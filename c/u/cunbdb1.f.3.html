subroutine cunbdb1 (integer m, integer p, integer q, complex, dimension(ldx11,*) x11, integer ldx11, complex, dimension(ldx21,*) x21, integer ldx21, real, dimension(*) theta, real, dimension(*) phi, complex, dimension(*) taup1, complex, dimension(*) taup2, complex, dimension(*) tauq1, complex, dimension(*) work, integer lwork, integer info)

CUNBDB1

الغرض:

!>
!> CUNBDB1 يقوم بتقطيع كتل المصفوفة الطويلة والرفيعة X ذات الأعمدة المتعامدة بشكل متزامن:
!>
!>                            [ B11 ]
!>      [ X11 ]   [ P1 |    ] [  0  ]
!>      [-----] = [---------] [-----] Q1**T .
!>      [ X21 ]   [    | P2 ] [ B21 ]
!>                            [  0  ]
!>
!> X11 هي P-by-Q، و X21 هي (M-P)-by-Q. يجب أن تكون Q لا تزيد عن P،
!> M-P، أو M-Q. الروتينات CUNBDB2، CUNBDB3، و CUNBDB4 تتعامل مع الحالات
!> التي لا تكون فيها Q هي الحد الأدنى للبُعد.
!>
!> المصفوفات الوحدوية P1، P2، و Q1 هي P-by-P، (M-P)-by-(M-P)،
!> و (M-Q)-by-(M-Q)، على التوالي. يتم تمثيلها ضمنيًا بواسطة
!> متجهات هاوسهولدر.
!>
!> B11 و B21 هي مصفوفات ثنائية القطر Q-by-Q ممثلة ضمنيًا بواسطة
!> الزوايا THETA، PHI.
!>
!>

المعلمات

M 
!>          M هو عدد صحيح (INTEGER)
!>           عدد الصفوف في X11 بالإضافة إلى عدد الصفوف في X21.
!>

P

!>          P هو عدد صحيح (INTEGER)
!>           عدد الصفوف في X11. 0 <= P <= M.
!>

Q

!>          Q هو عدد صحيح (INTEGER)
!>           عدد الأعمدة في X11 و X21. 0 <= Q <=
!>           MIN(P,M-P,M-Q).
!>

X11

!>          X11 هي مصفوفة مركبة (COMPLEX array)، البُعد (LDX11,Q)
!>           عند الدخول، الكتلة العلوية للمصفوفة X المراد تقليلها. عند
!>           الخروج، تحدد أعمدة tril(X11) الموجهات لـ P1
!>           وتحدد صفوف triu(X11,1) الموجهات لـ Q1.
!>

LDX11

!>          LDX11 هو عدد صحيح (INTEGER)
!>           البُعد الرائد لـ X11. LDX11 >= P.
!>

X21

!>          X21 هي مصفوفة مركبة (COMPLEX array)، البُعد (LDX21,Q)
!>           عند الدخول، الكتلة السفلية للمصفوفة X المراد تقليلها. عند
!>           الخروج، تحدد أعمدة tril(X21) الموجهات لـ P2.
!>

LDX21

!>          LDX21 هو عدد صحيح (INTEGER)
!>           البُعد الرائد لـ X21. LDX21 >= M-P.
!>

THETA

!>          THETA هي مصفوفة حقيقية (REAL array)، البُعد (Q)
!>           يتم تعريف مدخلات الكتل ثنائية القطر B11، B21 بواسطة
!>           THETA و PHI. انظر التفاصيل الإضافية.
!>

PHI

!>          PHI هي مصفوفة حقيقية (REAL array)، البُعد (Q-1)
!>           يتم تعريف مدخلات الكتل ثنائية القطر B11، B21 بواسطة
!>           THETA و PHI. انظر التفاصيل الإضافية.
!>

TAUP1

!>          TAUP1 هي مصفوفة مركبة (COMPLEX array)، البُعد (P)
!>           العوامل العددية للموجهات الأولية التي تحدد
!>           P1.
!>

TAUP2

!>          TAUP2 هي مصفوفة مركبة (COMPLEX array)، البُعد (M-P)
!>           العوامل العددية للموجهات الأولية التي تحدد
!>           P2.
!>

TAUQ1

!>          TAUQ1 هي مصفوفة مركبة (COMPLEX array)، البُعد (Q)
!>           العوامل العددية للموجهات الأولية التي تحدد
!>           Q1.
!>

WORK

!>          WORK هي مصفوفة مركبة (COMPLEX array)، البُعد (LWORK)
!>

LWORK

!>          LWORK هو عدد صحيح (INTEGER)
!>           بُعد المصفوفة WORK. LWORK >= M-Q.
!>
!>           إذا كان LWORK = -1، فسيتم افتراض استعلام مساحة العمل؛ يقوم الروتين
!>           بحساب حجم مساحة العمل المثلى فقط، ويعيد
!>           هذه القيمة كالإدخال الأول لمصفوفة WORK، ولا يصدر رسالة خطأ
!>           متعلقة بـ LWORK بواسطة XERBLA.
!>

INFO

!>          INFO هو عدد صحيح (INTEGER)
!>           = 0:  انتهاء ناجح.
!>           < 0:  إذا كان INFO = -i، فإن الوسيط i كان له قيمة غير قانونية.
!>

تفاصيل إضافية:

!>
!>  يتم تمثيل الكتل ثنائية القطر العلوية B11، B21 ضمنيًا بواسطة
!>  الزوايا THETA(1)، ..., THETA(Q) و PHI(1)، ..., PHI(Q-1). كل إدخال
!>  في كل نطاق ثنائي القطر هو حاصل ضرب جيب أو جيب تمام لـ THETA
!>  مع جيب أو جيب تمام لـ PHI. انظر [1] أو CUNCSD للحصول على التفاصيل.
!>
!>  يتم تمثيل P1، P2، و Q1 كحاصل ضرب للموجهات الأولية.
!>  انظر CUNCSD2BY1 للحصول على تفاصيل حول توليد P1، P2، و Q1 باستخدام CUNGQR
!>  و CUNGLQ.
!>
