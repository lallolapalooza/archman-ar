CURLOPT_CERTINFO(3)

الحزمة: curl

التراخيص: MIT

CURLOPT_CERTINFO(3) دليل الوظائف المكتبية CURLOPT_CERTINFO(3)

الاسم

CURLOPT_CERTINFO - طلب معلومات شهادة SSL

الملخص

#include <curl/curl.h>
CURLcode curl_easy_setopt(CURL *handle, CURLOPT_CERTINFO, long certinfo);

الوصف

مرر قيمة طويلة (long) مضبوطة على 1 لتمكين مجمع معلومات سلسلة شهادات libcurl. مع تمكين هذا الخيار، يستخرج libcurl الكثير من المعلومات والبيانات حول الشهادات في سلسلة الشهادات المستخدمة في اتصال SSL. يمكن استرداد هذه البيانات بعد النقل باستخدام curl_easy_getinfo(3) وخياره CURLINFO_CERTINFO(3).

الافتراضي

0

البروتوكولات

تؤثر هذه الوظيفة على جميع البروتوكولات القائمة على TLS: HTTPS، FTPS، IMAPS، POP3S، SMTPS إلخ.

يعمل هذا الخيار فقط مع الواجهات الخلفية TLS التالية: GnuTLS، OpenSSL، Schannel و rustls

مثال

int main(void)
{
  CURL *curl = curl_easy_init();
  if(curl) {
    CURLcode res;
    curl_easy_setopt(curl, CURLOPT_URL, "https://www.example.com/");
    /* الاتصال بأي موقع HTTPS، موثوق به أو غير موثوق به */
    curl_easy_setopt(curl, CURLOPT_SSL_VERIFYPEER, 0L);
    curl_easy_setopt(curl, CURLOPT_SSL_VERIFYHOST, 0L);
    curl_easy_setopt(curl, CURLOPT_CERTINFO, 1L);
    res = curl_easy_perform(curl);
    if(!res) {
      struct curl_certinfo *ci;
      res = curl_easy_getinfo(curl, CURLINFO_CERTINFO, &ci);
      if(!res) {
        int i;
        printf("%d شهادات!\n", ci->num_of_certs);
        for(i = 0; i < ci->num_of_certs; i++) {
          struct curl_slist *slist;
          for(slist = ci->certinfo[i]; slist; slist = slist->next)
            printf("%s\n", slist->data);
        }
      }
    }
    curl_easy_cleanup(curl);
  }
}

قيمة الإرجاع

curl_easy_setopt(3) يعيد رمز CURLcode يشير إلى النجاح أو الخطأ.

CURLE_OK (0) يعني أن كل شيء كان على ما يرام، والقيمة غير الصفرية تعني حدوث خطأ، انظر libcurl-errors(3).
