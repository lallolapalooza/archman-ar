CURLMOPT_PUSHDATA(3)

الحزمة: curl

التراخيص: MIT

CURLMOPT_PUSHDATA(3) دليل وظائف المكتبة CURLMOPT_PUSHDATA(3)

الاسم

CURLMOPT_PUSHDATA - مؤشر لتمريره إلى دالة رد الاتصال للدفع

الملخص

#include <curl/curl.h>
CURLMcode curl_multi_setopt(CURLM *handle, CURLMOPT_PUSHDATA, void *pointer);

الوصف

قم بتعيين مؤشر لتمريره كوسيط أخير إلى دالة رد الاتصال CURLMOPT_PUSHFUNCTION(3). لا يتم المساس بالمؤشر أو استخدامه بواسطة libcurl نفسها، بل يتم تمريره فقط إلى دالة رد الاتصال.

الافتراضي

NULL

البروتوكولات

تؤثر هذه الوظيفة على http فقط

مثال

#include <string.h>
/* السماح بالدفع فقط للملفات التي تبدأ بـ "push-" */
int push_callback(CURL *parent,
                  CURL *easy,
                  size_t num_headers,
                  struct curl_pushheaders *headers,
                  void *clientp)
{
  char *headp;
  int *transfers = (int *)clientp;
  FILE *out;
  headp = curl_pushheader_byname(headers, ":path");
  if(headp && !strncmp(headp, "/push-", 6)) {
    fprintf(stderr, "المسار هو %s\n", headp);
    /* حفظ الدفعة هنا */
    out = fopen("pushed-stream", "wb");
    /* الكتابة إلى هذا الملف */
    curl_easy_setopt(easy, CURLOPT_WRITEDATA, out);
    (*transfers)++; /* دفعة أخرى */
    return CURL_PUSH_OK;
  }
  return CURL_PUSH_DENY;
}
int main(void)
{
  int counter;
  CURLM *multi = curl_multi_init();
  curl_multi_setopt(multi, CURLMOPT_PUSHFUNCTION, push_callback);
  curl_multi_setopt(multi, CURLMOPT_PUSHDATA, &counter);
}

التوفر

تمت الإضافة في curl 7.44.0

قيمة الإرجاع

curl_multi_setopt(3) تعيد قيمة CURLMcode تشير إلى النجاح أو الخطأ.

CURLM_OK (0) يعني أن كل شيء كان على ما يرام، غير الصفر يعني حدوث خطأ، انظر libcurl-errors(3).
