CURLOPT_DEBUGDATA(3)

الحزمة: curl

التراخيص: MIT

CURLOPT_DEBUGDATA(3) دليل وظائف المكتبة CURLOPT_DEBUGDATA(3)

الاسم

CURLOPT_DEBUGDATA - مؤشر يتم تمريره إلى استدعاء التصحيح

الملخص

#include <curl/curl.h>
CURLcode curl_easy_setopt(CURL *handle, CURLOPT_DEBUGDATA, void *pointer);

الوصف

مرر مؤشر إلى ما تريد تمريره إلى CURLOPT_DEBUGFUNCTION(3) في الوسيطة void * الأخيرة. لا تستخدم libcurl هذا المؤشر، بل يتم تمريره فقط إلى الاستدعاء.

الافتراضي

NULL

البروتوكولات

تؤثر هذه الوظيفة على جميع البروتوكولات المدعومة

مثال

struct data {
  void *custom;
};
static int my_trace(CURL *handle, curl_infotype type,
                    char *data, size_t size,
                    void *clientp)
{
  struct data *mine = clientp;
  printf("our ptr: %p\n", mine->custom);
  /* output debug info */
  return 0;
}
int main(void)
{
  CURL *curl;
  CURLcode res;
  struct data my_tracedata;
  curl = curl_easy_init();
  if(curl) {
    curl_easy_setopt(curl, CURLOPT_DEBUGFUNCTION, my_trace);
    curl_easy_setopt(curl, CURLOPT_DEBUGDATA, &my_tracedata);
    /* the DEBUGFUNCTION has no effect until we enable VERBOSE */
    curl_easy_setopt(curl, CURLOPT_VERBOSE, 1L);
    curl_easy_setopt(curl, CURLOPT_URL, "https://example.com/");
    res = curl_easy_perform(curl);
    /* always cleanup */
    curl_easy_cleanup(curl);
  }
  return 0;
}

التوفر

تمت إضافته في curl 7.9.6

قيمة الإرجاع

curl_easy_setopt(3) ترجع CURLcode يشير إلى النجاح أو الخطأ.

CURLE_OK (0) يعني أن كل شيء كان على ما يرام، أي قيمة غير صفرية تعني حدوث خطأ، انظر libcurl-errors(3).
