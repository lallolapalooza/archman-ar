توثيق الوظيفة/الروتين الفرعي

subroutine cungql (integer m, integer n, integer k, complex, dimension( lda, * ) a, integer lda, complex, dimension( * ) tau, complex, dimension( * ) work, integer lwork, integer info)

CUNGQL

الغرض:

!>
!> CUNGQL تنشئ مصفوفة Q مركبة بحجم M × N بأعمدة متعامدة،
!> معرفة كآخر N أعمدة لحاصل ضرب K من الانعكاسات الأولية من الرتبة M
!>
!>       Q  =  H(k) . . . H(2) H(1)
!>
!> كما تم إرجاعها بواسطة CGEQLF.
!>

المعلمات

M 
!>          M هو عدد صحيح (INTEGER)
!>          عدد الصفوف في المصفوفة Q. M >= 0.
!>

N

!>          N هو عدد صحيح (INTEGER)
!>          عدد الأعمدة في المصفوفة Q. M >= N >= 0.
!>

K

!>          K هو عدد صحيح (INTEGER)
!>          عدد الانعكاسات الأولية التي تحدد حاصل ضربها المصفوفة Q.
!>          N >= K >= 0.
!>

A

!>          A هي مصفوفة مركبة (COMPLEX array)، ذات بعد (LDA,N)
!>          عند الدخول، يجب أن يحتوي العمود (n-k+i) على المتجه الذي
!>          يحدد الانعكاس الأولي H(i)، لكل i = 1,2,...,k، كما
!>          تم إرجاعه بواسطة CGEQLF في آخر K أعمدة من وسيطة المصفوفة A الخاصة به.
!>          عند الخروج، المصفوفة Q بحجم M × N.
!>

LDA

!>          LDA هو عدد صحيح (INTEGER)
!>          البعد الأول للمصفوفة A. LDA >= max(1,M).
!>

TAU

!>          TAU هي مصفوفة مركبة (COMPLEX array)، ذات بعد (K)
!>          يجب أن تحتوي TAU(i) على العامل القياسي للانعكاس الأولي H(i)،
!>          كما تم إرجاعه بواسطة CGEQLF.
!>

WORK

!>          WORK هي مصفوفة مركبة (COMPLEX array)، ذات بعد (MAX(1,LWORK))
!>          عند الخروج، إذا كانت INFO = 0، فإن WORK(1) ترجع LWORK المثلى.
!>

LWORK

!>          LWORK هو عدد صحيح (INTEGER)
!>          بعد المصفوفة WORK. LWORK >= max(1,N).
!>          لأداء مثالي LWORK >= N*NB، حيث NB هو حجم الكتلة الأمثل.
!>
!>          إذا كانت LWORK = -1، فيفترض أن يتم استعلام مساحة العمل؛
!>          تقوم الروتين فقط بحساب الحجم الأمثل لمصفوفة WORK،
!>          وتعيد هذا الحجم كالإدخال الأول لمصفوفة WORK، ولا يصدر
!>          رسالة خطأ متعلقة بـ LWORK بواسطة XERBLA.
!>

INFO

!>          INFO هو عدد صحيح (INTEGER)
!>          = 0: خروج ناجح
!>          < 0: إذا كانت INFO = -i، فإن الوسيط i لديه قيمة غير قانونية
!>
