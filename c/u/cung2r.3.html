وثائق الوظيفة/الوحدة الفرعية

subroutine cung2r (integer m, integer n, integer k, complex, dimension( lda, * ) a, integer lda, complex, dimension( * ) tau, complex, dimension( * ) work, integer info)

CUNG2R

الغرض:

!>
!> CUNG2R تولد مصفوفة معقدة m × n مع أعمدة متعامدة،
!> والتي يتم تعريفها على أنها أول n عمود من حاصل ضرب k عاكس أولي
!> من الرتبة m
!>
!>       Q  =  H(1) H(2) . . . H(k)
!>
!> كما تم إرجاعها بواسطة CGEQRF.
!>

المعلمات

M 
!>          M هو عدد صحيح (INTEGER)
!>          عدد الصفوف في المصفوفة Q. M >= 0.
!>

N

!>          N هو عدد صحيح (INTEGER)
!>          عدد الأعمدة في المصفوفة Q. M >= N >= 0.
!>

K

!>          K هو عدد صحيح (INTEGER)
!>          عدد العواكس الأولية التي يحدد حاصل ضربها المصفوفة Q. N >= K >= 0.
!>

A

!>          A هو مصفوفة أعداد مركبة (COMPLEX array)، الأبعاد (LDA,N)
!>          عند الإدخال، يجب أن يحتوي العمود i على المتجه الذي
!>          يحدد العاكس الأولي H(i)، لـ i = 1,2,...,k، كما
!>          تم إرجاعه بواسطة CGEQRF في أول k عمود من مصفوفة الحجة A الخاصة به.
!>          عند الخروج، المصفوفة Q ذات الأبعاد m × n.
!>

LDA

!>          LDA هو عدد صحيح (INTEGER)
!>          البعد الأول للمصفوفة A. LDA >= max(1,M).
!>

TAU

!>          TAU هي مصفوفة أعداد مركبة (COMPLEX array)، الأبعاد (K)
!>          يجب أن يحتوي TAU(i) على المعامل القياسي للعكس الأولي
!>          H(i)، كما تم إرجاعه بواسطة CGEQRF.
!>

WORK

!>          WORK هي مصفوفة أعداد مركبة (COMPLEX array)، الأبعاد (N)
!>

INFO

!>          INFO هو عدد صحيح (INTEGER)
!>          = 0: خروج ناجح
!>          < 0: إذا كان INFO = -i، فإن الوسيط i لديه قيمة غير قانونية
!>
