CURLOPT_SEEKFUNCTION(3)

الحزمة: curl

التراخيص: MIT

CURLOPT_SEEKFUNCTION(3) دليل وظائف المكتبة CURLOPT_SEEKFUNCTION(3)

الاسم

CURLOPT_SEEKFUNCTION - دالة استدعاء مستخدم للبحث في مجرى الإدخال

الملخص

#include <curl/curl.h>
/* هذه هي رموز الإرجاع لدوال استدعاء البحث */
#define CURL_SEEKFUNC_OK       0
#define CURL_SEEKFUNC_FAIL     1 /* فشل عملية النقل بالكامل */
#define CURL_SEEKFUNC_CANTSEEK 2 /* إخبار libcurl بأنه لا يمكن إجراء البحث، لذا
                                    قد تحاول libcurl وسائل أخرى بدلاً من ذلك */
int seek_callback(void *clientp, curl_off_t offset, int origin);
CURLcode curl_easy_setopt(CURL *handle, CURLOPT_SEEKFUNCTION, seek_callback);

الوصف

مرر مؤشرًا إلى دالة الاستدعاء الخاصة بك، والتي يجب أن تتطابق مع النموذج الأولي الموضح أعلاه.

يتم استدعاء هذه الدالة بواسطة libcurl للبحث في موضع معين في مجرى الإدخال ويمكن استخدامها لتسريع ملف في عملية تحميل مستأنفة (بدلاً من قراءة جميع البايتات التي تم تحميلها باستخدام دالة/استدعاء القراءة العادي). يتم استدعاؤها أيضًا لإعادة تعيين مجرى عندما تم إرسال البيانات بالفعل إلى الخادم وتحتاج إلى إرسالها مرة أخرى. قد يحدث هذا عند إجراء HTTP PUT أو POST باستخدام طريقة مصادقة متعددة المرور، أو عندما يتم إعادة استخدام اتصال HTTP موجود في وقت متأخر جدًا ويغلق الخادم الاتصال. يجب أن تعمل الدالة مثل fseek(3) أو lseek(3) وتحصل على SEEK_SET، SEEK_CUR أو SEEK_END كوسيط لـ origin، على الرغم من أن libcurl حاليًا تمرر SEEK_SET فقط.

clientp هو المؤشر الذي قمت بتعيينه باستخدام CURLOPT_SEEKDATA(3).

يجب أن تُرجع دالة الاستدعاء CURL_SEEKFUNC_OK عند النجاح، CURL_SEEKFUNC_FAIL للتسبب في فشل عملية التحميل، أو CURL_SEEKFUNC_CANTSEEK للإشارة إلى أنه بينما فشل البحث، فإن libcurl حرة في العمل حول المشكلة إذا كان ذلك ممكنًا. يمكن القيام بالأخير في بعض الأحيان عن طريق قراءة المدخلات أو ما شابه ذلك.

إذا قمت بتمرير وسيطات الإدخال مباشرة إلى fseek(3) أو lseek(3)، لاحظ أن نوع البيانات لـ offset ليس هو نفسه كما هو محدد لـ curl_off_t في العديد من الأنظمة.

الافتراضي

NULL

البروتوكولات

تؤثر هذه الوظيفة على ftp، http و sftp

مثال

#include <unistd.h> /* لـ lseek */
struct data {
  int our_fd;
};
static int seek_cb(void *clientp, curl_off_t offset, int origin)
{
  struct data *d = (struct data *)clientp;
  lseek(d->our_fd, offset, origin);
  return CURL_SEEKFUNC_OK;
}
int main(void)
{
  struct data seek_data;
  CURL *curl = curl_easy_init();
  if(curl) {
    curl_easy_setopt(curl, CURLOPT_SEEKFUNCTION, seek_cb);
    curl_easy_setopt(curl, CURLOPT_SEEKDATA, &seek_data);
  }
}

التوفر

تمت إضافته في curl 7.18.0

قيمة الإرجاع

curl_easy_setopt(3) تُرجع قيمة CURLcode تشير إلى النجاح أو الخطأ.

CURLE_OK (0) تعني أن كل شيء كان على ما يرام، والقيمة غير الصفرية تعني حدوث خطأ، انظر libcurl-errors(3).
