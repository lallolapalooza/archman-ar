subroutine cungtsqr_row (integer m, integer n, integer mb, integer nb, complex, dimension( lda, * ) a, integer lda, complex, dimension( ldt, * ) t, integer ldt, complex, dimension( * ) work, integer lwork, integer info)

CUNGTSQR_ROW

الغرض:

!>
!> CUNGTSQR_ROW تنشئ مصفوفة مركبة Q_out ذات أبعاد M × N بأعمدة متعامدة
!> من خرج CLATSQR. هذه الأعمدة المتعامدة N هي الأعمدة N الأولى من حاصل ضرب
!> مصفوفات الوحدة المركبة Q(k)_in ذات الرتبة M، والتي يتم إرجاعها بواسطة CLATSQR
!> بتنسيق خاص.
!>
!>      Q_out = first_N_columns_of( Q(1)_in * Q(2)_in * ... * Q(k)_in ).
!>
!> مصفوفات الإدخال Q(k)_in مخزنة في كتل صفوف وأعمدة في A.
!> انظر توثيق CLATSQR لمزيد من التفاصيل حول تنسيق
!> Q(k)_in، حيث يتم تمثيل كل Q(k)_in بواسطة تحويلات هاوسهولدر المجمعة.
!> تقوم هذه الروتينية باستدعاء روتين مساعد CLARFB_GETT، حيث يتم إجراء الحساب
!> على كل كتلة فردية. يقوم الخوارزمية أولاً بمسح كتل الأعمدة ذات الحجم NB
!> من اليمين إلى اليسار بدءًا من كتلة الصف السفلية وتستمر إلى كتلة الصف العلوية
!> (لذلك _ROW في اسم الروتينية). هذا المسح بترتيب عكسي
!> للترتيب الذي تولد به CLATSQR كتل الإخراج.
!>

المعلمات

M 
!>          M هو عدد صحيح (INTEGER)
!>          عدد صفوف المصفوفة A.  M >= 0.
!>

N

!>          N هو عدد صحيح (INTEGER)
!>          عدد أعمدة المصفوفة A. M >= N >= 0.
!>

MB

!>          MB هو عدد صحيح (INTEGER)
!>          حجم كتلة الصف المستخدم بواسطة CLATSQR لإرجاع
!>          المصفوفات A و T. MB > N.
!>          (لاحظ أنه إذا كان MB > M، فسيتم استخدام M بدلاً من MB
!>          كحجم كتلة الصف).
!>

NB

!>          NB هو عدد صحيح (INTEGER)
!>          حجم كتلة العمود المستخدم بواسطة CLATSQR لإرجاع
!>          المصفوفات A و T. NB >= 1.
!>          (لاحظ أنه إذا كان NB > N، فسيتم استخدام N بدلاً من NB
!>          كحجم كتلة العمود).
!>

A

!>          A هي مصفوفة مركبة (COMPLEX)، أبعادها (LDA,N)
!>
!>          عند الدخول:
!>
!>             العناصر على القطر وفوقه لا تستخدم كمدخلات.
!>             العناصر تحت القطر تمثل مصفوفة V المجمعة شبه القطري
!>             السفلي السفلي التي تم حسابها بواسطة CLATSQR والتي تحدد
!>             مصفوفات الإدخال Q_in(k) (الواحدات على القطر غير مخزنة).
!>             راجع CLATSQR لمزيد من التفاصيل.
!>
!>          عند الخروج:
!>
!>             المصفوفة A تحتوي على مصفوفة الوحدة المركبة Q_out ذات
!>             الأبعاد M × N، أي أن أعمدة A هي متجهات وحدوية متعامدة.
!>

LDA

!>          LDA هو عدد صحيح (INTEGER)
!>          البعد الرائد للمصفوفة A.  LDA >= max(1,M).
!>

T

!>          T هي مصفوفة مركبة (COMPLEX)،
!>          أبعادها (LDT, N * NIRB)
!>          حيث NIRB = Number_of_input_row_blocks
!>                     = MAX( 1, CEIL((M-N)/(MB-N)) )
!>          لنفرض NICB = Number_of_input_col_blocks
!>                   = CEIL(N/NB)
!>
!>          المعكوسات المجمعة القطري العلوي التي تحدد مصفوفات الإدخال
!>          Q_in(k)، k=(1:NIRB*NICB). يتم تخزين المعكوسات المجمعة
!>          في شكل مدمج في كتل الأعمدة N × N من T وتتكون من
!>          NICB كتل أعمدة أصغر ذات أبعاد NB × NB قطري علوي.
!>          راجع CLATSQR لمزيد من التفاصيل حول تنسيق T.
!>

LDT

!>          LDT هو عدد صحيح (INTEGER)
!>          البعد الرائد للمصفوفة T.
!>          LDT >= max(1,min(NB,N)).
!>

WORK

!>          (مساحة عمل) مصفوفة مركبة (COMPLEX)، أبعادها (MAX(1,LWORK))
!>          عند الخروج، إذا كان INFO = 0، فإن WORK(1) يرجع LWORK الأمثل.
!>

LWORK

!>          LWORK هو عدد صحيح (INTEGER)
!>          بعد المصفوفة WORK.
!>          LWORK >= NBLOCAL * MAX(NBLOCAL,(N-NBLOCAL)),
!>          حيث NBLOCAL=MIN(NB,N).
!>          إذا كان LWORK = -1، فمن المفترض وجود استعلام مساحة عمل.
!>          تقوم الروتينية بحساب الحجم الأمثل لمصفوفة WORK فقط،
!>          وتعيد هذا القيمة كالإدخال الأول لمصفوفة WORK، ولا يتم إصدار
!>          أي رسالة خطأ متعلقة بـ LWORK بواسطة XERBLA.
!>

INFO

!>          INFO هو عدد صحيح (INTEGER)
!>          = 0:  تم الخروج بنجاح
!>          < 0:  إذا كان INFO = -i، فإن الوسيط i كان له قيمة غير قانونية
!>
