subroutine cungr2 (m, n, k, a, lda, tau, work, info)
CUNGR2 يقوم بإنشاء جزء أو كل المصفوفة الوحدوية Q من تحليل RQ الذي تم تحديده بواسطة cgerqf (خوارزمية غير محظورة).

توثيق الوظيفة/الروتين الفرعي

subroutine cungr2 (integer m, integer n, integer k, complex, dimension( lda, * ) a, integer lda, complex, dimension( * ) tau, complex, dimension( * ) work, integer info)

CUNGR2 يقوم بإنشاء جزء أو كل المصفوفة الوحدوية Q من تحليل RQ الذي تم تحديده بواسطة cgerqf (خوارزمية غير محظورة).

الغرض:

!>
!> CUNGR2 يقوم بإنشاء مصفوفة معقدة Q بحجم m × n ذات صفوف متعامدة،
!> والتي تُعرّف بأنها الصفوف m الأخيرة من ناتج k من الانعكاسات الابتدائية
!> ذات الرتبة n
!>
!>       Q  =  H(1)**H H(2)**H . . . H(k)**H
!>
!> كما تم إرجاعه بواسطة CGERQF.
!>

المعلمات

M 
!>          M هو عدد صحيح (INTEGER)
!>          عدد صفوف المصفوفة Q. M >= 0.
!>

N

!>          N هو عدد صحيح (INTEGER)
!>          عدد أعمدة المصفوفة Q. N >= M.
!>

K

!>          K هو عدد صحيح (INTEGER)
!>          عدد الانعكاسات الابتدائية التي يُعرّف ناتجها المصفوفة Q. M >= K >= 0.
!>

A

!>          A هي مصفوفة معقدة (COMPLEX)، ذات بُعد (LDA,N)
!>          عند الدخول، يجب أن يحتوي الصف (m-k+i) على المتجه الذي
!>          يُعرّف الانعكاس الابتدائي H(i)، من أجل i = 1,2,...,k، كما
!>          تم إرجاعه بواسطة CGERQF في الصفوف k الأخيرة من وسيطة المصفوفة الخاصة به
!>          A.
!>          عند الخروج، المصفوفة Q بحجم m × n.
!>

LDA

!>          LDA هو عدد صحيح (INTEGER)
!>          البعد الأول للمصفوفة A. LDA >= max(1,M).
!>

TAU

!>          TAU هي مصفوفة معقدة (COMPLEX)، ذات بُعد (K)
!>          يجب أن يحتوي TAU(i) على العامل القياسي للانعكاس الابتدائي
!>          H(i)، كما تم إرجاعه بواسطة CGERQF.
!>

WORK

!>          WORK هي مصفوفة معقدة (COMPLEX)، ذات بُعد (M)
!>

INFO

!>          INFO هو عدد صحيح (INTEGER)
!>          = 0: انتهاء ناجح
!>          < 0: إذا كان INFO = -i، فإن الوسيطة i تحتوي على قيمة غير قانونية
!>
