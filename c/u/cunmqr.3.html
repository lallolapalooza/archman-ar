subroutine cunmqr (character side, character trans, integer m, integer n, integer k, complex, dimension( lda, * ) a, integer lda, complex, dimension( * ) tau, complex, dimension( ldc, * ) c, integer ldc, complex, dimension( * ) work, integer lwork, integer info)

CUNMQR

Purpose:

!>
!> CUNMQR يقوم بالكتابة فوق مصفوفة C العامة المعقدة ذات البعد M-by-N بـ
!>
!>                 SIDE = 'L'     SIDE = 'R'
!> TRANS = 'N':      Q * C          C * Q
!> TRANS = 'C':      Q**H * C       C * Q**H
!>
!> حيث Q هي مصفوفة وحدوية معقدة معرفة كناتج لـ k
!> عواكس أولية
!>
!>       Q = H(1) H(2) . . . H(k)
!>
!> كما تم إرجاعها بواسطة CGEQRF. Q لها رتبة M إذا كان SIDE = 'L' ورتبة N
!> إذا كان SIDE = 'R'.
!>

Parameters

SIDE 
!>          SIDE is CHARACTER*1
!>          = 'L': تطبيق Q أو Q**H من اليسار؛
!>          = 'R': تطبيق Q أو Q**H من اليمين.
!>

TRANS

!>          TRANS is CHARACTER*1
!>          = 'N':  بدون تبديل، تطبيق Q؛
!>          = 'C':  تبديل مترافق، تطبيق Q**H.
!>

M

!>          M is INTEGER
!>          عدد الصفوف في المصفوفة C. M >= 0.
!>

N

!>          N is INTEGER
!>          عدد الأعمدة في المصفوفة C. N >= 0.
!>

K

!>          K is INTEGER
!>          عدد العواكس الأولية التي يحدد ناتجها
!>          المصفوفة Q.
!>          إذا كان SIDE = 'L'، M >= K >= 0؛
!>          إذا كان SIDE = 'R'، N >= K >= 0.
!>

A

!>          A is COMPLEX array, dimension (LDA,K)
!>          يجب أن يحتوي العمود i على المتجه الذي يحدد
!>          العاكس الأولي H(i)، لكل i = 1,2,...,k، كما تم إرجاعه بواسطة
!>          CGEQRF في الأعمدة k الأولى من وسيط المصفوفة A الخاص به.
!>

LDA

!>          LDA is INTEGER
!>          البعد الرئيسي للمصفوفة A.
!>          إذا كان SIDE = 'L'، LDA >= max(1,M)؛
!>          إذا كان SIDE = 'R'، LDA >= max(1,N).
!>

TAU

!>          TAU is COMPLEX array, dimension (K)
!>          يجب أن يحتوي TAU(i) على العامل القياسي للعامل
!>          العاكس الأولي H(i)، كما تم إرجاعه بواسطة CGEQRF.
!>

C

!>          C is COMPLEX array, dimension (LDC,N)
!>          عند الإدخال، المصفوفة C ذات البعد M-by-N.
!>          عند الخروج، يتم الكتابة فوق C بواسطة Q*C أو Q**H*C أو C*Q**H أو C*Q.
!>

LDC

!>          LDC is INTEGER
!>          البعد الرئيسي للمصفوفة C. LDC >= max(1,M).
!>

WORK

!>          WORK is COMPLEX array, dimension (MAX(1,LWORK))
!>          عند الخروج، إذا كان INFO = 0، فإن WORK(1) ترجع LWORK الأمثل.
!>

LWORK

!>          LWORK is INTEGER
!>          بعد المصفوفة WORK.
!>          إذا كان SIDE = 'L'، LWORK >= max(1,N)؛
!>          إذا كان SIDE = 'R'، LWORK >= max(1,M).
!>          لأداء جيد، يجب أن يكون LWORK بشكل عام أكبر.
!>
!>          إذا كان LWORK = -1، فسيتم افتراض استعلام عن مساحة العمل؛ تقوم الروتين
!>          فقط بحساب الحجم الأمثل للمصفوفة WORK، وتعيد
!>          هذه القيمة كالإدخال الأول للمصفوفة WORK، ولا يتم إصدار أي رسالة خطأ
!>          متعلقة بـ LWORK بواسطة XERBLA.
!>

INFO

!>          INFO is INTEGER
!>          = 0:  الخروج الناجح
!>          < 0:  إذا كان INFO = -i، فإن الوسيط i كان له قيمة غير قانونية
!>
