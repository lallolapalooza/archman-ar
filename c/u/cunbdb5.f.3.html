subroutine cunbdb5 (integer m1, integer m2, integer n, complex, dimension(*) x1, integer incx1, complex, dimension(*) x2, integer incx2, complex, dimension(ldq1,*) q1, integer ldq1, complex, dimension(ldq2,*) q2, integer ldq2, complex, dimension(*) work, integer lwork, integer info)

CUNBDB5

الغرض:

!>
!> CUNBDB5 يقوم بتطبيع المتجه العمودي
!>      X = [ X1 ]
!>          [ X2 ]
!> بالنسبة للأعمدة الخاصة بـ
!>      Q = [ Q1 ] .
!>          [ Q2 ]
!> يجب أن تكون أعمدة Q متعامدة.
!>
!> إذا كان الإسقاط صفرًا وفقًا لمعيار كاهان،
!> يتم إرجاع متجه آخر من المكمل المتعامد.
!> يتم اختيار هذا المتجه بطريقة اعتباطية ولكن محددة.
!>
!>

المعلمات

M1 
!>          M1 is INTEGER
!>           بعد X1 وعدد صفوف Q1. 0 <= M1.
!>

M2

!>          M2 is INTEGER
!>           بعد X2 وعدد صفوف Q2. 0 <= M2.
!>

N

!>          N is INTEGER
!>           عدد أعمدة Q1 و Q2. 0 <= N.
!>

X1

!>          X1 is COMPLEX array, dimension (M1)
!>           عند الإدخال، الجزء العلوي من المتجه المراد تطبيعه.
!>           عند الخروج، الجزء العلوي من المتجه المسقط.
!>

INCX1

!>          INCX1 is INTEGER
!>           زيادة لإدخالات X1.
!>

X2

!>          X2 is COMPLEX array, dimension (M2)
!>           عند الإدخال، الجزء السفلي من المتجه المراد
!>           تطبيعه. عند الخروج، الجزء السفلي من المتجه المسقط.
!>

INCX2

!>          INCX2 is INTEGER
!>           زيادة لإدخالات X2.
!>

Q1

!>          Q1 is COMPLEX array, dimension (LDQ1, N)
!>           الجزء العلوي من مصفوفة الأساس المتعامدة.
!>

LDQ1

!>          LDQ1 is INTEGER
!>           البعد الرائد لـ Q1. LDQ1 >= M1.
!>

Q2

!>          Q2 is COMPLEX array, dimension (LDQ2, N)
!>           الجزء السفلي من مصفوفة الأساس المتعامدة.
!>

LDQ2

!>          LDQ2 is INTEGER
!>           البعد الرائد لـ Q2. LDQ2 >= M2.
!>

WORK

!>          WORK is COMPLEX array, dimension (LWORK)
!>

LWORK

!>          LWORK is INTEGER
!>           بعد المصفوفة WORK. LWORK >= N.
!>

INFO

!>          INFO is INTEGER
!>           = 0:  نجاح الخروج.
!>           < 0:  إذا كان INFO = -i، فإن المعامل i كان له قيمة غير قانونية.
!>
