توثيق الوظيفة/الروتين الفرعي

subroutine cunmtr (character side, character uplo, character trans, integer m, integer n, complex, dimension( lda, * ) a, integer lda, complex, dimension( * ) tau, complex, dimension( ldc, * ) c, integer ldc, complex, dimension( * ) work, integer lwork, integer info)

CUNMTR

الغرض:

!>
!> CUNMTR يستبدل مصفوفة C المعقدة العامة M-by-N بـ
!>
!>                 SIDE = 'L'     SIDE = 'R'
!> TRANS = 'N':      Q * C          C * Q
!> TRANS = 'C':      Q**H * C       C * Q**H
!>
!> حيث Q هي مصفوفة وحدة معقدة من الرتبة nq، مع nq = m إذا كان
!> SIDE = 'L' و nq = n إذا كان SIDE = 'R'. يتم تعريف Q كناتج
!> لـ nq-1 عاكس أولي، كما هو مسترجع بواسطة CHETRD:
!>
!> إذا كان UPLO = 'U'، فإن Q = H(nq-1) . . . H(2) H(1);
!>
!> إذا كان UPLO = 'L'، فإن Q = H(1) H(2) . . . H(nq-1).
!>

المعلمات

SIDE 
!>          SIDE هو CHARACTER*1
!>          = 'L': طبق Q أو Q**H من اليسار؛
!>          = 'R': طبق Q أو Q**H من اليمين.
!>

UPLO

!>          UPLO هو CHARACTER*1
!>          = 'U': المثلث العلوي لـ A يحتوي على عواكس أولية
!>                 من CHETRD؛
!>          = 'L': المثلث السفلي لـ A يحتوي على عواكس أولية
!>                 من CHETRD.
!>

TRANS

!>          TRANS هو CHARACTER*1
!>          = 'N':  بدون تبديل، طبق Q؛
!>          = 'C':  تبديل مترافق، طبق Q**H.
!>

M

!>          M هو INTEGER
!>          عدد صفوف المصفوفة C. M >= 0.
!>

N

!>          N هو INTEGER
!>          عدد أعمدة المصفوفة C. N >= 0.
!>

A

!>          A هو مصفوفة COMPLEX، الأبعاد
!>                               (LDA,M) إذا كان SIDE = 'L'
!>                               (LDA,N) إذا كان SIDE = 'R'
!>          المتجهات التي تعرف العواكس الأولية، كما
!>          تم استرجاعها بواسطة CHETRD.
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرئيسي للمصفوفة A.
!>          LDA >= max(1,M) إذا كان SIDE = 'L'؛ LDA >= max(1,N) إذا كان SIDE = 'R'.
!>

TAU

!>          TAU هي مصفوفة COMPLEX، الأبعاد
!>                               (M-1) إذا كان SIDE = 'L'
!>                               (N-1) إذا كان SIDE = 'R'
!>          يجب أن يحتوي TAU(i) على العامل القياسي للعاكس الأولي
!>          H(i)، كما تم استرجاعه بواسطة CHETRD.
!>

C

!>          C هي مصفوفة COMPLEX، الأبعاد (LDC,N)
!>          عند الإدخال، مصفوفة C ذات الأبعاد M-by-N.
!>          عند الخروج، يتم استبدال C بـ Q*C أو Q**H*C أو C*Q**H أو C*Q.
!>

LDC

!>          LDC هو INTEGER
!>          البعد الرئيسي للمصفوفة C. LDC >= max(1,M).
!>

WORK

!>          WORK هي مصفوفة COMPLEX، الأبعاد (MAX(1,LWORK))
!>          عند الخروج، إذا كان INFO = 0، فإن WORK(1) يسترجع LWORK الأمثل.
!>

LWORK

!>          LWORK هو INTEGER
!>          بعد مصفوفة WORK.
!>          إذا كان SIDE = 'L'، فإن LWORK >= max(1,N)؛
!>          إذا كان SIDE = 'R'، فإن LWORK >= max(1,M).
!>          لأداء مثالي LWORK >= N*NB إذا كان SIDE = 'L'، و
!>          LWORK >=M*NB إذا كان SIDE = 'R'، حيث NB هو حجم الكتلة الأمثل.
!>
!>          إذا كان LWORK = -1، فسيتم افتراض استعلام عن مساحة العمل؛ يقوم الروتين
!>          فقط بحساب الحجم الأمثل لمصفوفة WORK، ويسترجع
!>          هذه القيمة كالإدخال الأول لمصفوفة WORK، ولا يصدر رسالة خطأ
!>          متعلقة بـ LWORK بواسطة XERBLA.
!>

INFO

!>          INFO هو INTEGER
!>          = 0:  خروج ناجح
!>          < 0:  إذا كان INFO = -i، فإن المعلمة i كانت ذات قيمة غير قانونية
!>
