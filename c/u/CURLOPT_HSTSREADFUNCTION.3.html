CURLOPT_HSTSREADFUNCTION(3)

الحزمة: curl

التراخيص: MIT

CURLOPT_HSTSREADFUNCTION(3) دليل وظائف المكتبة CURLOPT_HSTSREADFUNCTION(3)

الاسم

CURLOPT_HSTSREADFUNCTION - دالة الاستدعاء لقراءة المضيفين HSTS

ملخص

#include <curl/curl.h>
struct curl_hstsentry {
  char *name;
  size_t namelen;
  unsigned int includeSubDomains:1;
  char expire[18]; /* YYYYMMDD HH:MM:SS [null-terminated] */
};
CURLSTScode hstsread(CURL *easy, struct curl_hstsentry *sts, void *clientp);
CURLcode curl_easy_setopt(CURL *handle, CURLOPT_HSTSREADFUNCTION, hstsread);

الوصف

مرر مؤشرًا إلى دالة الاستدعاء الخاصة بك، وفقًا للنموذج أعلاه.

يتم استدعاء دالة الاستدعاء هذه بواسطة libcurl بشكل متكرر عند ملء ذاكرة التخزين المؤقت HSTS في الذاكرة.

قم بتعيين الوسيط clientp باستخدام الخيار CURLOPT_HSTSREADDATA(3) أو إذا كان NULL.

عند استدعاء دالة الاستدعاء هذه، يشير المؤشر sts إلى بنية مملوءة: انسخ اسم المضيف إلى name (لا يزيد طوله عن namelen بايت). اجعله ينتهي بصفر. اضبط includeSubDomains على TRUE أو FALSE. اضبط expire على طابع زمني أو سلسلة ذات طول صفر لـ الأبد (قد يتسبب تنسيق الطابع الزمني غير الصحيح في عدم قبول الاسم)

يجب أن تُرجع دالة الاستدعاء CURLSTS_OK إذا أعادت اسمًا وكانت مستعدة ليتم استدعاؤها مرة أخرى (لمضيف آخر) أو CURLSTS_DONE إذا لم يكن لديها إدخال لإعادته. يمكنها أيضًا إرجاع CURLSTS_FAIL للإشارة إلى خطأ. يؤدي إرجاع CURLSTS_FAIL إلى إيقاف النقل من التنفيذ وجعل CURLE_ABORTED_BY_CALLBACK يتم إرجاعه.

هذا الخيار لا يمكّن HSTS، تحتاج إلى استخدام CURLOPT_HSTS_CTRL(3) للقيام بذلك.

الافتراضي

NULL - لا توجد دالة استدعاء.

البروتوكولات

تؤثر هذه الوظيفة على http فقط

مثال

struct priv {
  void *custom;
};
static CURLSTScode hsts_cb(CURL *easy, struct curl_hstsentry *sts,
                           void *clientp)
{
  /* populate the struct as documented */
  return CURLSTS_OK;
}
int main(void)
{
  CURL *curl = curl_easy_init();
  if(curl) {
    struct priv my_stuff;
    CURLcode res;
    /* set HSTS read callback */
    curl_easy_setopt(curl, CURLOPT_HSTSREADFUNCTION, hsts_cb);
    /* pass in suitable argument to the callback */
    curl_easy_setopt(curl, CURLOPT_HSTSREADDATA, &my_stuff);
    res = curl_easy_perform(curl);
  }
}
