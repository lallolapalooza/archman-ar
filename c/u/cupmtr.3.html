subroutine cupmtr (character side, character uplo, character trans, integer m, integer n, complex, dimension( * ) ap, complex, dimension( * ) tau, complex, dimension( ldc, * ) c, integer ldc, complex, dimension( * ) work, integer info)

CUPMTR

الغرض:

!>
!> CUPMTR يستبدل المصفوفة C المعقدة العامة ذات الأبعاد M*N بـ
!>
!>                 SIDE = 'L'     SIDE = 'R'
!> TRANS = 'N':      Q * C          C * Q
!> TRANS = 'C':      Q**H * C       C * Q**H
!>
!> حيث Q هي مصفوفة وحدوية معقدة من الرتبة nq، مع nq = m إذا كان
!> SIDE = 'L' و nq = n إذا كان SIDE = 'R'. Q معرفة كحاصل ضرب
!!> nq-1 عواكس أولية، كما تم إرجاعها بواسطة CHPTRD باستخدام تخزين
!> مجمع:
!>
!> إذا كان UPLO = 'U'، فإن Q = H(nq-1) . . . H(2) H(1)؛
!>
!> إذا كان UPLO = 'L'، فإن Q = H(1) H(2) . . . H(nq-1).
!>

المعلمات

SIDE 
!>          SIDE is CHARACTER*1
!>          = 'L': تطبيق Q أو Q**H من اليسار؛
!>          = 'R': تطبيق Q أو Q**H من اليمين.
!>

UPLO

!>          UPLO is CHARACTER*1
!>          = 'U': تخزين مثلثي علوي مجمع مستخدم في استدعاء سابق
!>                 إلى CHPTRD؛
!>          = 'L': تخزين مثلثي سفلي مجمع مستخدم في استدعاء سابق
!>                 إلى CHPTRD.
!>

TRANS

!>          TRANS is CHARACTER*1
!>          = 'N':  بدون تبديل، تطبيق Q؛
!>          = 'C':  تبديل مرافق، تطبيق Q**H.
!>

M

!>          M is INTEGER
!>          عدد صفوف المصفوفة C. M >= 0.
!>

N

!>          N is INTEGER
!>          عدد أعمدة المصفوفة C. N >= 0.
!>

AP

!>          AP is COMPLEX array, dimension
!>                               (M*(M+1)/2) if SIDE = 'L'
!>                               (N*(N+1)/2) if SIDE = 'R'
!>          المتجهات التي تحدد العواكس الأولية، كما
!>          تم إرجاعها بواسطة CHPTRD.  يتم تعديل AP بواسطة الروتين الفرعي
!>          ولكن يتم استعادته عند الخروج.
!>

TAU

!>          TAU is COMPLEX array, dimension (M-1) if SIDE = 'L'
!>                                     or (N-1) if SIDE = 'R'
!>          يجب أن يحتوي TAU(i) على العامل القياسي للعكس الأولي
!>          H(i)، كما تم إرجاعه بواسطة CHPTRD.
!>

C

!>          C is COMPLEX array, dimension (LDC,N)
!>          عند الدخول، المصفوفة C ذات الأبعاد M*N.
!>          عند الخروج، يتم استبدال C بـ Q*C أو Q**H*C أو C*Q**H أو C*Q.
!>

LDC

!>          LDC is INTEGER
!>          البعد الرائد للمصفوفة C. LDC >= max(1,M).
!>

WORK

!>          WORK is COMPLEX array, dimension
!>                                   (N) if SIDE = 'L'
!!>                                   (M) if SIDE = 'R'
!>

INFO

!>          INFO is INTEGER
!>          = 0:  خروج ناجح
!>          < 0:  إذا كان INFO = -i، فإن الوسيط i له قيمة غير قانونية
!>
