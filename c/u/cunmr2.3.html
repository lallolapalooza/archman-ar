توثيق الدالة/الإجراء الفرعي

subroutine cunmr2 (character side, character trans, integer m, integer n, integer k, complex, dimension( lda, * ) a, integer lda, complex, dimension( * ) tau, complex, dimension( ldc, * ) c, integer ldc, complex, dimension( * ) work, integer info)

CUNMR2 تضرب مصفوفة عامة في المصفوفة الوحدوية من تحليل RQ المحدد بواسطة cgerqf (خوارزمية غير محظورة).

الغرض:

!>
!> CUNMR2 تستبدل المصفوفة المعقدة m-by-n العامة C بـ
!>
!>       Q * C  إذا كان SIDE = 'L' و TRANS = 'N'، أو
!>
!>       Q**H* C  إذا كان SIDE = 'L' و TRANS = 'C'، أو
!>
!>       C * Q  إذا كان SIDE = 'R' و TRANS = 'N'، أو
!>
!>       C * Q**H إذا كان SIDE = 'R' و TRANS = 'C'،
!>
!> حيث Q هي مصفوفة وحدوية معقدة معرفة كناتج لـ k
!> عواكس ابتدائية
!>
!>       Q = H(1)**H H(2)**H . . . H(k)**H
!>
!> كما تم إرجاعها بواسطة CGERQF. Q من الرتبة m إذا كان SIDE = 'L' ومن الرتبة n
!> إذا كان SIDE = 'R'.
!>

المعلمات

SIDE 
!>          SIDE هو CHARACTER*1
!>          = 'L': تطبيق Q أو Q**H من اليسار
!>          = 'R': تطبيق Q أو Q**H من اليمين
!>

TRANS

!>          TRANS هو CHARACTER*1
!>          = 'N': تطبيق Q  (بدون تبديل)
!>          = 'C': تطبيق Q**H (تبديل مرافق)
!>

M

!>          M هو INTEGER
!>          عدد صفوف المصفوفة C. M >= 0.
!>

N

!>          N هو INTEGER
!>          عدد أعمدة المصفوفة C. N >= 0.
!>

K

!>          K هو INTEGER
!>          عدد العواكس الابتدائية التي يحدد ناتجها
!>          المصفوفة Q.
!>          إذا كان SIDE = 'L'، فإن M >= K >= 0؛
!>          إذا كان SIDE = 'R'، فإن N >= K >= 0.
!>

A

!>          A هي مصفوفة COMPLEX، الأبعاد
!>                               (LDA,M) إذا كان SIDE = 'L'،
!>                               (LDA,N) إذا كان SIDE = 'R'
!>          الصف i يجب أن يحتوي على المتجه الذي يحدد
!>          العاكس الابتدائي H(i)، لـ i = 1,2,...,k، كما تم إرجاعها بواسطة
!>          CGERQF في الصفوف k الأخيرة من مصفوفة الوسيط A.
!>          يتم تعديل A بواسطة الإجراء الفرعي ولكن يتم استعادتها عند الخروج.
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرائد للمصفوفة A. LDA >= max(1,K).
!>

TAU

!>          TAU هي مصفوفة COMPLEX، الأبعاد (K)
!>          يجب أن يحتوي TAU(i) على العامل القياسي للعكس
!>          الابتدائي H(i)، كما تم إرجاعه بواسطة CGERQF.
!>

C

!>          C هي مصفوفة COMPLEX، الأبعاد (LDC,N)
!>          عند الإدخال، المصفوفة المعقدة m-by-n C.
!>          عند الخروج، يتم استبدال C بـ Q*C أو Q**H*C أو C*Q**H أو C*Q.
!>

LDC

!>          LDC هو INTEGER
!>          البعد الرائد للمصفوفة C. LDC >= max(1,M).
!>

WORK

!>          WORK هي مصفوفة COMPLEX، الأبعاد
!>                                   (N) إذا كان SIDE = 'L'،
!>                                   (M) إذا كان SIDE = 'R'
!>

INFO

!>          INFO هو INTEGER
!>          = 0: خروج ناجح
!>          < 0: إذا كان INFO = -i، فإن المعلمة i كانت لها قيمة غير قانونية
!>
