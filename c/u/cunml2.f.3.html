subroutine cunml2 (character side, character trans, integer m, integer n, integer k, complex, dimension( lda, * ) a, integer lda, complex, dimension( * ) tau, complex, dimension( ldc, * ) c, integer ldc, complex, dimension( * ) work, integer info)

CUNML2 يضرب مصفوفة عامة بالمصفوفة الوحدوية من تفكيك LQ المحدد بواسطة cgelqf (خوارزمية غير محظورة).

الغرض:

!>
!> CUNML2 تستبدل المصفوفة العامة المعقدة m-by-n C بـ
!>
!>       Q * C  إذا كان SIDE = 'L' و TRANS = 'N', أو
!>
!>       Q**H* C  إذا كان SIDE = 'L' و TRANS = 'C', أو
!>
!>       C * Q  إذا كان SIDE = 'R' و TRANS = 'N', أو
!>
!>       C * Q**H إذا كان SIDE = 'R' و TRANS = 'C',
!>
!> حيث Q هي مصفوفة وحدوية معقدة معرفة كحاصل ضرب k
!> عواكس ابتدائية
!>
!>       Q = H(k)**H . . . H(2)**H H(1)**H
!>
!> كما تم إرجاعها بواسطة CGELQF. Q ذات رتبة m إذا كان SIDE = 'L' وذات رتبة n
!> إذا كان SIDE = 'R'.
!>

المعلمات

SIDE 
!>          SIDE هو CHARACTER*1
!>          = 'L': تطبيق Q أو Q**H من اليسار
!>          = 'R': تطبيق Q أو Q**H من اليمين
!>

TRANS

!>          TRANS هو CHARACTER*1
!>          = 'N': تطبيق Q  (بدون تبديل)
!>          = 'C': تطبيق Q**H (تبديل منسق)
!>

M

!>          M هو INTEGER
!>          عدد صفوف المصفوفة C. M >= 0.
!>

N

!>          N هو INTEGER
!>          عدد أعمدة المصفوفة C. N >= 0.
!>

K

!>          K هو INTEGER
!>          عدد العواكس الابتدائية التي يحدد حاصل ضربها
!>          المصفوفة Q.
!>          إذا كان SIDE = 'L', M >= K >= 0;
!>          إذا كان SIDE = 'R', N >= K >= 0.
!>

A

!>          A هو مصفوفة COMPLEX, أبعادها
!>                               (LDA,M) إذا كان SIDE = 'L',
!>                               (LDA,N) إذا كان SIDE = 'R'
!>          يجب أن يحتوي الصف i على المتجه الذي يحدد
!>          العاكس الابتدائي H(i), لـ i = 1,2,...,k, كما تم إرجاعه بواسطة
!>          CGELQF في أول k صفوف لمصفوفته الوسيطة A.
!>          يتم تعديل A بواسطة الروتين ولكن يتم استعادتها عند الخروج.
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرئيسي للمصفوفة A. LDA >= max(1,K).
!>

TAU

!>          TAU هي مصفوفة COMPLEX, أبعادها (K)
!>          يجب أن يحتوي TAU(i) على العامل القياسي
!>          للعاكس الابتدائي H(i), كما تم إرجاعه بواسطة CGELQF.
!>

C

!>          C هي مصفوفة COMPLEX, أبعادها (LDC,N)
!>          عند الإدخال, المصفوفة m-by-n C.
!>          عند الخروج, يتم استبدال C بـ Q*C أو Q**H*C أو C*Q**H أو C*Q.
!>

LDC

!>          LDC هو INTEGER
!>          البعد الرئيسي للمصفوفة C. LDC >= max(1,M).
!>

WORK

!>          WORK هي مصفوفة COMPLEX, أبعادها
!>                                   (N) إذا كان SIDE = 'L',
!!>                                   (M) إذا كان SIDE = 'R'
!>

INFO

!>          INFO هو INTEGER
!>          = 0: خروج ناجح
!>          < 0: إذا كان INFO = -i, فإن المعامل i كان لديه قيمة غير قانونية
!>
