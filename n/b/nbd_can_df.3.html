nbd_can_df(3)

الحزمة: libnbd

التراخيص: LGPL-2.1-or-later

nbd_can_df(3) LIBNBD nbd_can_df(3)

الاسم

nbd_can_df - هل الخادم يدعم علامة عدم التجزئة للقراءة؟

الملخص

#include <libnbd.h>
int nbd_can_df (
      struct nbd_handle *h
    );

الوصف

يعيد قيمة صحيحة إذا كان الخادم يدعم القراءات المنظمة مع القدرة على طلب قراءة غير مجزأة (انظر nbd_pread_structured(3), nbd_aio_pread_structured(3)). يعيد قيمة خاطئة إذا كان الخادم يفتقر إلى القراءات المنظمة أو إذا كان لا يدعم طلب قراءة غير مجزأة.

هذه المكالمة لا تحظر، لأنها تعيد البيانات المحفوظة في المقبض من مصافحة بروتوكول NBD.

قيمة الإرجاع

تعيد هذه المكالمة قيمة منطقية.

الأخطاء

عند حدوث خطأ، يتم إرجاع -1.

ارجع إلى "معالجة الأخطاء" في libnbd(3) للحصول على تفاصيل إضافية حول الخطأ.

يجب ألا تكون المعلمات التالية فارغة: "h". لمزيد من المعلومات، انظر "المعلمات غير الفارغة" في libnbd(3).

حالة المقبض

يمكن استدعاء nbd_can_df عندما يكون المقبض في الحالات التالية:

┌─────────────────────────────────────┬─────────────────────────┐
│ تم إنشاء المقبض، قبل الاتصال │ ❌ خطأ │
│ الاتصال │ ❌ خطأ │
│ الاتصال والتعامل (opt_mode) │ ✅ مسموح به │
│ متصل بالخادم │ ✅ مسموح به │
│ تم إغلاق الاتصال │ ✅ مسموح به │
│ المقبض ميت │ ❌ خطأ │
└─────────────────────────────────────┴─────────────────────────┘

الإصدار

ظهرت هذه الدالة لأول مرة في libnbd 1.0.

إذا كنت بحاجة إلى التحقق مما إذا كانت هذه الدالة متاحة في وقت الترجمة، تحقق مما إذا كان الماكرو التالي معرفًا:

#define LIBNBD_HAVE_NBD_CAN_DF 1

مثال

هذا المثال متاح أيضًا باسم examples/server-flags.c في الكود المصدري لـ libnbd.

/* هذا المثال يوضح كيفية الاتصال بخادم NBD
 * وطباعة علامات التصدير.
 *
 * يمكنك اختباره باستخدام nbdkit على هذا النحو:
 *
 * nbdkit -U - memory 1M \
 *   --run './server-flags $unixsocket'
 */
#include <stdio.h>
#include <stdlib.h>
#include <stdbool.h>
#include <libnbd.h>
int
main (int argc, char *argv[])
{
  struct nbd_handle *nbd;
  char *str;
  int flag;
  if (argc != 2) {
    fprintf (stderr, "%s socket\n", argv[0]);
    exit (EXIT_FAILURE);
  }
  /* إنشاء مقبض libnbd. */
  nbd = nbd_create ();
  if (nbd == NULL) {
    fprintf (stderr, "%s\n", nbd_get_error ());
    exit (EXIT_FAILURE);
  }
  /* طلب معلومات كاملة. */
#if LIBNBD_HAVE_NBD_SET_FULL_INFO /* تمت إضافته في 1.4 */
  if (nbd_set_full_info (nbd, true) == -1) {
    fprintf (stderr, "%s\n", nbd_get_error ());
    exit (EXIT_FAILURE);
  }
#endif
  /* الاتصال بخادم NBD عبر
   * مقبس نطاق Unix.
   */
  if (nbd_connect_unix (nbd, argv[1]) == -1) {
    fprintf (stderr, "%s\n", nbd_get_error ());
    exit (EXIT_FAILURE);
  }
  /* معرفة ما إذا كان الخادم قد قدم تفاصيل إضافية،
   * باستخدام الدوال التي تمت إضافتها في 1.4
   */
#if LIBNBD_HAVE_NBD_GET_EXPORT_DESCRIPTION
  str = nbd_get_canonical_export_name (nbd);
  if (str)
    printf ("canonical_name = %s\n", str);
  free (str);
  str = nbd_get_export_description (nbd);
  if (str)
    printf ("description = %s\n", str);
  free (str);
#endif
  /* قراءة وطباعة العلامات. */
#define PRINT_FLAG(flag_fn)                     \
  flag = flag_fn (nbd);                         \
  if (flag == -1) {                             \
    fprintf (stderr, "%s\n", nbd_get_error ()); \
    exit (EXIT_FAILURE);                        \
  }                                             \
  printf (#flag_fn " = %s\n",                   \
          flag ? "true" : "false");
  PRINT_FLAG (nbd_can_cache);
  PRINT_FLAG (nbd_can_df);
  PRINT_FLAG (nbd_can_flush);
  PRINT_FLAG (nbd_can_fua);
  PRINT_FLAG (nbd_can_multi_conn);
  PRINT_FLAG (nbd_can_trim);
  PRINT_FLAG (nbd_can_zero);
#if LIBNBD_HAVE_NBD_CAN_FAST_ZERO
  /* تمت إضافته في 1.2 */
  PRINT_FLAG (nbd_can_fast_zero);
#endif
#if LIBNBD_HAVE_NBD_CAN_BLOCK_STATUS_PAYLOAD
  /* تمت إضافته في 1.18 */
  PRINT_FLAG (nbd_can_block_status_payload);
#endif
  PRINT_FLAG (nbd_is_read_only);
  PRINT_FLAG (nbd_is_rotational);
  /* إغلاق مقبض libnbd. */
  nbd_close (nbd);
  exit (EXIT_SUCCESS);
}
