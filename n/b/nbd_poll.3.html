nbd_poll(3)

الحزمة: libnbd

التراخيص: LGPL-2.1-or-later

nbd_poll(3) LIBNBD nbd_poll(3)

الاسم

nbd_poll - فحص المقبض مرة واحدة

الملخص

#include <libnbd.h>
int nbd_poll (
      struct nbd_handle *h, int timeout
    );

الوصف

هذا تطبيق بسيط للدالة poll(2) والتي تستخدم داخليًا بواسطة واجهات برمجة التطبيقات المتزامنة. عند النجاح، تُرجع 0 إذا حدث "timeout" (بالمللي ثانية)، أو 1 إذا اكتمل الفحص وتقدمت آلة الحالة. قم بتعيين "timeout" إلى -1 للحظر إلى أجل غير مسمى (ولكن كن حذرًا من أن الإجراء النهائي متوقع بالفعل - على سبيل المثال، إذا تم إنشاء الاتصال ولكن لا توجد أوامر قيد التنفيذ، فإن استخدام مهلة لا نهائية سيؤدي إلى حظر دائم).

هذه الدالة مفيدة بشكل أساسي كمثال لكيفية دمج libnbd مع حلقة الأحداث الخاصة بك، بدلاً من أن تكون شيئًا ستستخدمه.

قيمة الإرجاع

تُرجع هذه المكالمة عددًا صحيحًا ≥ 0.

الأخطاء

عند حدوث خطأ، تُرجع -1.

ارجع إلى "معالجة الأخطاء" في libnbd(3) للحصول على مزيد من التفاصيل حول الخطأ.

يجب ألا تكون المعلمات التالية فارغة (NULL): "h". لمزيد من المعلومات، انظر "المعلمات غير الفارغة" في libnbd(3).

الإصدار

ظهرت هذه الدالة لأول مرة في libnbd 1.0.

إذا كنت بحاجة إلى اختبار ما إذا كانت هذه الدالة متاحة وقت الترجمة، فتحقق مما إذا كان هذا الماكرو محددًا:

#define LIBNBD_HAVE_NBD_POLL 1

مثال

هذا المثال متاح أيضًا باسم examples/aio-connect-read.c في شفرة مصدر libnbd.

/* هذا المثال يوضح كيفية استخدام واجهة برمجة التطبيقات المنخفضة المستوى (غير المتزامنة) AIO
 * للاتصال بخادم وقراءة القرص.
 *
 * إليك بعض الطرق لتجربة هذا المثال:
 *
 * nbdkit -U - linuxdisk . \
 *   --run './aio-connect-read $unixsocket'
 *
 * nbdkit -U - floppy . \
 *   --run './aio-connect-read $unixsocket'
 *
 * nbdkit -U - pattern size=1M \
 *   --run './aio-connect-read $unixsocket'
 */
#include <stdio.h>
#include <stdlib.h>
#include <stdint.h>
#include <inttypes.h>
#include <errno.h>
#include <assert.h>
#include <libnbd.h>
#define NR_SECTORS 32
#define SECTOR_SIZE 512
struct data {
  uint64_t offset;
  char sector[SECTOR_SIZE];
};
static int
hexdump (void *user_data, int *error)
{
  struct data *data = user_data;
  FILE *pp;
  if (*error) {
    errno = *error;
    perror ("فشل القراءة");
    exit (EXIT_FAILURE);
  }
  printf ("القطاع عند الإزاحة 0x%" PRIx64 ":\n",
          data->offset);
  pp = popen ("hexdump -C", "w");
  if (pp == NULL) {
    perror ("popen: hexdump");
    exit (EXIT_FAILURE);
  }
  fwrite (data->sector, SECTOR_SIZE, 1, pp);
  pclose (pp);
  printf ("\n");
  /* الإرجاع 1 من رد الاتصال يتقاعد
   * الأمر تلقائيًا.
   */
  return 1;
}
static struct data data[NR_SECTORS];
int
main (int argc, char *argv[])
{
  struct nbd_handle *nbd;
  size_t i;
  if (argc != 2) {
    fprintf (stderr, "%s socket\n", argv[0]);
    exit (EXIT_FAILURE);
  }
  /* إنشاء مقبض libnbd. */
  nbd = nbd_create ();
  if (nbd == NULL) {
    fprintf (stderr, "%s\n", nbd_get_error ());
    exit (EXIT_FAILURE);
  }
  /* الاتصال بخادم NBD عبر مقبس مجال Unix.
   * هذا يبدأ الاتصال فقط.
   */
  if (nbd_aio_connect_unix (nbd, argv[1]) == -1) {
    fprintf (stderr, "%s\n", nbd_get_error ());
    exit (EXIT_FAILURE);
  }
  /* انتظر حتى يكتمل الاتصال.  استخدام
   * nbd_poll هنا هو فقط كمثال.  يمكنك أيضًا
   * دمج هذا مع poll(2) أو glib أو حلقة
   * أحداث أخرى.  اقرأ libnbd(3) وملف المصدر lib/poll.c.
   */
  while (!nbd_aio_is_ready (nbd)) {
    if (nbd_poll (nbd, -1) == -1) {
      fprintf (stderr, "%s\n", nbd_get_error ());
      exit (EXIT_FAILURE);
    }
  }
  assert (nbd_get_size (nbd) >= NR_SECTORS * SECTOR_SIZE);
  /* إصدار أوامر قراءة لأول NR قطاعات. */
  for (i = 0; i < NR_SECTORS; ++i) {
    data[i].offset = i * SECTOR_SIZE;
    /* يتم استدعاء رد الاتصال (hexdump) عند
     * اكتمال الأمر.  يجب أن تستمر الذاكرة المؤقتة في الوجود بينما
     * يكون الأمر قيد التشغيل.
     */
    if (nbd_aio_pread (nbd, data[i].sector, SECTOR_SIZE,
                       data[i].offset,
                       (nbd_completion_callback) {
                         .callback = hexdump,
                         .user_data = &data[i],
                       }, 0) == -1) {
      fprintf (stderr, "%s\n", nbd_get_error ());
      exit (EXIT_FAILURE);
    }
  }
  /* تشغيل حلقة الأحداث الرئيسية حتى تكتمل جميع الأوامر
   * وتتقاعد.  مرة أخرى، استخدام nbd_poll
   * هنا هو فقط كمثال.
   */
  while (nbd_aio_in_flight (nbd) > 0) {
    if (nbd_poll (nbd, -1) == -1) {
      fprintf (stderr, "%s\n", nbd_get_error ());
      exit (EXIT_FAILURE);
    }
  }
  /* إغلاق مقبض libnbd. */
  nbd_close (nbd);
  exit (EXIT_SUCCESS);
}
