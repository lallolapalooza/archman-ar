nbd_can_fua(3)

الحزمة: libnbd

التراخيص: LGPL-2.1-or-later

nbd_can_fua(3) LIBNBD nbd_can_fua(3)

الاسم

nbd_can_fua - هل يدعم الخادم علامة FUA؟

الملخص

#include <libnbd.h>
int nbd_can_fua (
      struct nbd_handle *h
    );

الوصف

تُرجع قيمة صحيحة إذا كان الخادم يدعم علامة FUA في أوامر معينة (راجع nbd_pwrite(3)).

لا يؤدي هذا الاستدعاء إلى حظر، لأنه يُرجع بيانات محفوظة في المقبض من المصافحة لبروتوكول NBD.

قيمة الإرجاع

تُرجع هذه الاستدعاء قيمة منطقية.

الأخطاء

عند حدوث خطأ، يتم إرجاع -1.

ارجع إلى "معالجة الأخطاء" في libnbd(3) لمعرفة كيفية الحصول على تفاصيل إضافية حول الخطأ.

يجب ألا تكون المعلمات التالية فارغة: "h". لمزيد من المعلومات، راجع "المعلمات غير الفارغة" في libnbd(3).

حالة المقبض

يمكن استدعاء nbd_can_fua عندما يكون المقبض في الحالات التالية:

┌─────────────────────────────────────┬─────────────────────────┐
│ تم إنشاء المقبض، قبل الاتصال   │ ❌ خطأ                │
│ جارٍ الاتصال                      │ ❌ خطأ                │
│ الاتصال والمصافحة (opt_mode) │ ✅ مسموح به              │
│ متصل بالخادم             │ ✅ مسموح به              │
│ تم قطع الاتصال                │ ✅ مسموح به              │
│ المقبض غير صالح                 │ ❌ خطأ                │
└─────────────────────────────────────┴─────────────────────────┘

الإصدار

ظهرت هذه الوظيفة لأول مرة في libnbd 1.0.

إذا كنت بحاجة إلى اختبار ما إذا كانت هذه الوظيفة متاحة وقت الترجمة، تحقق مما إذا كان ما يلي معرفًا:

#define LIBNBD_HAVE_NBD_CAN_FUA 1

مثال

هذا المثال متاح أيضًا كـ examples/server-flags.c في الكود المصدري لـ libnbd.

/* هذا المثال يوضح كيفية الاتصال بخادم NBD
 * وطباعة علامات التصدير.
 *
 * يمكنك اختباره باستخدام nbdkit كالتالي:
 *
 * nbdkit -U - memory 1M \
 *   --run './server-flags $unixsocket'
 */
#include <stdio.h>
#include <stdlib.h>
#include <stdbool.h>
#include <libnbd.h>
int
main (int argc, char *argv[])
{
  struct nbd_handle *nbd;
  char *str;
  int flag;
  if (argc != 2) {
    fprintf (stderr, "%s socket\n", argv[0]);
    exit (EXIT_FAILURE);
  }
  /* إنشاء مقبض libnbd. */
  nbd = nbd_create ();
  if (nbd == NULL) {
    fprintf (stderr, "%s\n", nbd_get_error ());
    exit (EXIT_FAILURE);
  }
  /* طلب معلومات كاملة. */
#if LIBNBD_HAVE_NBD_SET_FULL_INFO /* تمت إضافته في 1.4 */
  if (nbd_set_full_info (nbd, true) == -1) {
    fprintf (stderr, "%s\n", nbd_get_error ());
    exit (EXIT_FAILURE);
  }
#endif
  /* الاتصال بخادم NBD عبر
   * مقبس مجال Unix.
   */
  if (nbd_connect_unix (nbd, argv[1]) == -1) {
    fprintf (stderr, "%s\n", nbd_get_error ());
    exit (EXIT_FAILURE);
  }
  /* معرفة ما إذا كان الخادم قد قدم تفاصيل إضافية،
   * باستخدام الوظائف التي تمت إضافتها في 1.4
   */
#if LIBNBD_HAVE_NBD_GET_EXPORT_DESCRIPTION
  str = nbd_get_canonical_export_name (nbd);
  if (str)
    printf ("canonical_name = %s\n", str);
  free (str);
  str = nbd_get_export_description (nbd);
  if (str)
    printf ("description = %s\n", str);
  free (str);
#endif
  /* قراءة وطباعة العلامات. */
#define PRINT_FLAG(flag_fn)                     \
  flag = flag_fn (nbd);                         \
  if (flag == -1) {                             \
    fprintf (stderr, "%s\n", nbd_get_error ()); \
    exit (EXIT_FAILURE);                        \
  }                                             \
  printf (#flag_fn " = %s\n",                   \
          flag ? "true" : "false");
  PRINT_FLAG (nbd_can_cache);
  PRINT_FLAG (nbd_can_df);
  PRINT_FLAG (nbd_can_flush);
  PRINT_FLAG (nbd_can_fua);
  PRINT_FLAG (nbd_can_multi_conn);
  PRINT_FLAG (nbd_can_trim);
  PRINT_FLAG (nbd_can_zero);
#if LIBNBD_HAVE_NBD_CAN_FAST_ZERO
  /* تمت إضافته في 1.2 */
  PRINT_FLAG (nbd_can_fast_zero);
#endif
#if LIBNBD_HAVE_NBD_CAN_BLOCK_STATUS_PAYLOAD
  /* تمت إضافته في 1.18 */
  PRINT_FLAG (nbd_can_block_status_payload);
#endif
  PRINT_FLAG (nbd_is_read_only);
  PRINT_FLAG (nbd_is_rotational);
  /* إغلاق مقبض libnbd. */
  nbd_close (nbd);
  exit (EXIT_SUCCESS);
}
