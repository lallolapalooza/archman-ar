nbd_get_error(3)

الحزمة: libnbd

التراخيص: LGPL-2.1-or-later

nbd_create(3) LIBNBD nbd_create(3)

الاسم

nbd_create, nbd_close, nbd_get_error, nbd_get_errno - إنشاء مؤشرات libnbd وجلب الأخطاء

الملخص

#include <libnbd.h>
struct nbd_handle *nbd;
struct nbd_handle *nbd_create (void);
void nbd_close (struct nbd_handle *nbd);
const char *nbd_get_error (void);
int nbd_get_errno (void);

الوصف

struct nbd_handle هو هيكل غير شفاف يصف مؤشر عميل NBD والاتصال بخادم NBD.

إنشاء مؤشر libnbd

nbd_create ينشئ مؤشرًا جديدًا. يعيد مؤشرًا إلى هيكل المؤشر غير الشفاف.

عند حدوث خطأ، يعيد "NULL". راجع "معالجة الأخطاء" في libnbd(3) لمعرفة كيفية الحصول على تفاصيل إضافية للخطأ.

إغلاق مؤشر

nbd_close يغلق المؤشر، ويغلق ويحرر أي موارد مرتبطة به.

الحالة النهائية لأي أمر لم يتم إكماله (سواء بواسطة nbd_aio_command_completed(3) أو بواسطة رد اتصال منخفض المستوى يعيد 1) تُفقد.

هذه الدالة ليست آمنة للاستدعاء بينما أي خيط آخر لا يزال يستخدم أي واجهة "nbd_*" على نفس المؤشر. يمكن أن تحظر هذه الدالة في الحالة التي ننتظر فيها عملية فرعية (على سبيل المثال، تم إنشاؤها باستخدام nbd_connect_command(3)).

الحصول على آخر رسالة خطأ في الخيط

nbd_get_error يعيد آخر رسالة خطأ في الخيط الحالي. رسالة الخطأ صالحة فقط إذا تم استدعاؤها فورًا بعد الاستدعاء الفاشل، من نفس الخيط. سلسلة الخطأ المعاد تحريرها عند استدعاء أي واجهة libnbd أخرى من نفس الخيط، لذلك إذا كنت بحاجة إلى الاحتفاظ بها، يجب عليك إنشاء نسخة.

لا ينبغي أن يعيد هذا "NULL" شريطة أن يكون هناك خطأ قد أعيد من استدعاء libnbd السابق مباشرة في الخيط الحالي.

nbd_get_errno يعيد آخر "errno" في الخيط الحالي. ليست كل الأخطاء لها أرقام أخطاء مقابلة، لذلك حتى لو كان هناك خطأ، فقد يعيد هذا 0. رموز الأخطاء هي الأكواد القياسية من "<errno.h>".

راجع "معالجة الأخطاء" في libnbd(3) لمزيد من المناقشة حول كيفية عمل معالجة الأخطاء في libnbd.

مثال

#include <libnbd.h>
main ()
{
  struct nbd_handle *nbd;
  nbd = nbd_create ();
  if (nbd == NULL) {
    fprintf (stderr, "%s\n", nbd_get_error ());
    nbd_close (nbd);
    exit (EXIT_FAILURE);
  }
  nbd_close (nbd);
  exit (EXIT_SUCCESS);
}
