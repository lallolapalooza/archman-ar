nbd_aio_block_status_64(3)

الحزمة: libnbd

التراخيص: LGPL-2.1-or-later

nbd_aio_block_status_64(3) LIBNBD nbd_aio_block_status_64(3)

الاسم

nbd_aio_block_status_64 - إرسال أمر حالة الكتلة، مع رد اتصال 64 بت

ملخص

#include <libnbd.h>
typedef struct {
  int (*callback) (void *user_data,
                   const char *metacontext,
                   uint64_t offset, nbd_extent *entries,
                   size_t nr_entries, int *error);
  void *user_data;
  void (*free) (void *user_data);
} nbd_extent64_callback;
typedef struct {
  int (*callback) (void *user_data, int *error);
  void *user_data;
  void (*free) (void *user_data);
} nbd_completion_callback;
int64_t nbd_aio_block_status_64 (
          struct nbd_handle *h, uint64_t count,
          uint64_t offset,
          nbd_extent64_callback extent64_callback,
          nbd_completion_callback completion_callback,
          uint32_t flags
        );

الوصف

إرسال أمر حالة الكتلة إلى خادم NBD.

للتحقق مما إذا كان الأمر قد اكتمل، استدعِ nbd_aio_command_completed(3). أو قم بتوفير "completion_callback" الاختياري الذي سيتم استدعاؤه كما هو موضح في "ردود الاتصال الكاملة" في libnbd(3).

تتصرف المعلمات الأخرى كما هو موثق في nbd_block_status_64(3).

افتراضيًا، سترفض libnbd المحاولات لاستخدام هذه الدالة بمعلمات من المحتمل أن تؤدي إلى فشل الخادم، مثل طلب علامة أمر غير معروفة. يمكن استخدام الدالة nbd_set_strict_mode(3) لتغيير السيناريوهات التي تنتظر استجابة الخادم بدلاً من الفشل السريع.

قيمة الإرجاع

ترجع هذه المكالمة المعرف 64 بت للأمر. المعرف ≥ 1. المعرفات فريدة (لكل مقبض libnbd، وليس عالميًا).

الأخطاء

عند حدوث خطأ، يتم إرجاع -1.

ارجع إلى "معالجة الأخطاء" في libnbd(3) للحصول على تفاصيل إضافية عن الخطأ.

يجب ألا تكون المعلمات التالية NULL: "h". لمزيد من المعلومات، انظر "المعلمات غير الخالية" في libnbd(3).

حالة المقبض

يمكن استدعاء nbd_aio_block_status_64 عندما يكون المقبض في الحالة التالية:

┌─────────────────────────────────────┬─────────────────────────┐
│ تم إنشاء المقبض، قبل الاتصال │ ❌ خطأ ❌                │
│ جارٍ الاتصال │ ❌ خطأ ❌                │
│ جارٍ الاتصال & المصافحة (opt_mode) │ ❌ خطأ ❌                │
│ متصل بالخادم │ ✅ مسموح به ✅              │
│ تم إنهاء الاتصال │ ❌ خطأ ❌                │
│ المقبض ميت │ ❌ خطأ ❌                │
└─────────────────────────────────────┴─────────────────────────┘

الإصدار

ظهرت هذه الدالة لأول مرة في libnbd 1.18.

إذا كنت بحاجة إلى اختبار ما إذا كانت هذه الدالة متاحة وقت الترجمة، فتحقق مما إذا كان الماكرو التالي معرفًا:

#define LIBNBD_HAVE_NBD_AIO_BLOCK_STATUS_64 1

انظر أيضاً

nbd_aio_block_status(3), nbd_aio_command_completed(3), nbd_block_status_64(3), nbd_can_meta_context(3), nbd_create(3), nbd_set_strict_mode(3), "إصدار الأوامر غير المتزامنة" في libnbd(3), libnbd(3).
