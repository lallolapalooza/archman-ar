nbd_close(3)

الحزمة: libnbd

التراخيص: LGPL-2.1-or-later

nbd_create(3) LIBNBD nbd_create(3)

الاسم

nbd_create, nbd_close, nbd_get_error, nbd_get_errno - إنشاء معالجات libnbd وجلب الأخطاء

الملخص

#include <libnbd.h>
struct nbd_handle *nbd;
struct nbd_handle *nbd_create (void);
void nbd_close (struct nbd_handle *nbd);
const char *nbd_get_error (void);
int nbd_get_errno (void);

الوصف

struct nbd_handle هو هيكل مبهم يصف مقبض عميل NBD والاتصال بخادم NBD.

إنشاء مقبض libnbd

nbd_create ينشئ مقبضًا جديدًا. يعيد مؤشرًا إلى هيكل المقبض المبهم.

عند حدوث خطأ، يعيد "NULL". انظر "معالجة الأخطاء" في libnbd(3) للحصول على تفاصيل إضافية حول الخطأ.

إغلاق مقبض

nbd_close يغلق المقبض، ويغلق ويحرر أي موارد مرتبطة به.

يتم فقدان الحالة النهائية لأي أمر لم يتم الانتهاء منه (سواء بواسطة nbd_aio_command_completed(3) أو بواسطة رد اتصال منخفض المستوى يعيد 1).

هذه الدالة ليست آمنة للاستدعاء بينما لا يزال أي خيط آخر يستخدم أي واجهة برمجة تطبيقات "nbd_*" على نفس المقبض. يمكن لهذه الدالة أن تحجب في حالة انتظار عملية فرعية (مثل تلك التي تم إنشاؤها باستخدام nbd_connect_command(3)).

الحصول على أحدث رسالة خطأ في الخيط

nbd_get_error يعيد أحدث رسالة خطأ في الخيط الحالي. تكون رسالة الخطأ صالحة فقط إذا تم استدعاؤها مباشرة بعد الاستدعاء الفاشل، من نفس الخيط. سيتم تحرير سلسلة الخطأ المعاد توفيرها في المرة التالية التي يتم فيها استدعاء أي واجهة برمجة تطبيقات libnbd من نفس الخيط، لذلك إذا كنت بحاجة إلى الاحتفاظ بها، فيجب عليك إنشاء نسخة.

لا يجب أن يعيد هذا "NULL" بشرط وجود خطأ تم إرجاعه من استدعاء libnbd السابق مباشرة في الخيط الحالي.

nbd_get_errno يعيد أحدث "errno" في الخيط الحالي. لا تحتوي جميع الأخطاء على أرقام أخطاء مقابلة، لذلك حتى لو كان هناك خطأ، فقد يعيد هذا 0. رموز الأخطاء هي الرموز القياسية من "<errno.h>".

انظر "معالجة الأخطاء" في libnbd(3) لمزيد من المناقشة حول كيفية عمل معالجة الأخطاء في libnbd.

مثال

#include <libnbd.h>
main ()
{
  struct nbd_handle *nbd;
  nbd = nbd_create ();
  if (nbd == NULL) {
    fprintf (stderr, "%s\n", nbd_get_error ());
    nbd_close (nbd);
    exit (EXIT_FAILURE);
  }
  nbd_close (nbd);
  exit (EXIT_SUCCESS);
}
