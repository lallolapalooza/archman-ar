nbd_shutdown(3)

الحزمة: libnbd

التراخيص: LGPL-2.1-or-later

nbd_shutdown(3) LIBNBD nbd_shutdown(3)

الاسم

nbd_shutdown - قطع الاتصال بخادم NBD

الملخص

#include <libnbd.h>
int nbd_shutdown (
      struct nbd_handle *h, uint32_t flags
    );

الوصف

إصدار أمر قطع الاتصال لخادم NBD. هذه طريقة لطيفة لإخبار الخادم بأننا سنغادر، ولكن من وجهة نظر العميل ليس لها ميزة على قطع الاتصال بشكل مفاجئ (انظر nbd_close(3)).

تعمل هذه الدالة سواء كان المقبض جاهزًا لإرسال الأوامر أم لا. إذا كانت هناك حاجة إلى تحكم أكثر دقة، انظر nbd_aio_opt_abort(3) و nbd_aio_disconnect(3).

الوسيط "flags" هو قناع بت، بما في ذلك صفر أو أكثر من علامات إيقاف التشغيل التالية:

"LIBNBD_SHUTDOWN_ABANDON_PENDING" = 0x10000
إذا كانت هناك أي طلبات معلقة لم يتم إرسالها بعد إلى الخادم (انظر nbd_aio_in_flight(3))، فتخلّ عنها دون إرسالها إلى الخادم، بدلاً من الممارسة المعتادة المتمثلة في إصدار تلك الأوامر قبل إبلاغ الخادم بنية قطع الاتصال.

للتسهيل، الثابت "LIBNBD_SHUTDOWN_MASK" متاح لوصف جميع علامات إيقاف التشغيل التي تم التعرف عليها من خلال هذا الإصدار من libnbd. قد يضيف إصدار مستقبلي من المكتبة علامات جديدة.

قيمة الإرجاع

في حالة نجاح الاستدعاء، ترجع الدالة 0.

الأخطاء

عند حدوث خطأ، ترجع -1.

ارجع إلى "معالجة الأخطاء" في libnbd(3) للحصول على مزيد من التفاصيل حول الخطأ.

يجب ألا تكون المعلمات التالية فارغة (NULL): "h". لمزيد من المعلومات، انظر "المعلمات غير الفارغة" في libnbd(3).

حالة المقبض

يمكن استدعاء nbd_shutdown عندما يكون المقبض في الحالات التالية:

┌─────────────────────────────────────┬─────────────────────────┐
│ تم إنشاء المقبض، قبل الاتصال   │ ❌ خطأ                │
│ جاري الاتصال                          │ ❌ خطأ                │
│ جاري الاتصال & المصافحة (opt_mode) │ ✅ مسموح به              │
│ متصل بالخادم             │ ✅ مسموح به              │
│ تم قطع اتصال الاتصال                │ ❌ خطأ                │
│ المقبض معطل                         │ ❌ خطأ                │
└─────────────────────────────────────┴─────────────────────────┘

الإصدار

ظهرت هذه الدالة لأول مرة في libnbd 1.0.

إذا كنت بحاجة إلى اختبار ما إذا كانت هذه الدالة متاحة في وقت الترجمة، فتحقق مما إذا كان ما يلي معرفًا:

#define LIBNBD_HAVE_NBD_SHUTDOWN 1

مثال

هذا المثال متاح أيضًا كـ examples/reads-and-writes.c في كود مصدر libnbd.

/* هذا المثال يوضح كيفية إجراء عمليات قراءة
 * وكتابة متزامنة عشوائياً عبر أول ميغابايت من
 * خادم NBD. ملاحظة: سيؤدي هذا إلى تدمير أي محتوى موجود
 * على قرص NBD.
 *
 * لاختباره مع nbdkit وقرص ذاكرة:
 *
 * nbdkit -U - memory 1M \
 *     --run './simple-reads-and-writes $unixsocket'
 */
#include <stdio.h>
#include <stdlib.h>
#include <stdint.h>
#include <inttypes.h>
#include <assert.h>
#include <time.h>
#include <libnbd.h>
int
main (int argc, char *argv[])
{
  struct nbd_handle *nbd;
  char buf[512];
  size_t i;
  int64_t exportsize;
  uint64_t offset;
  srand (time (NULL));
  if (argc != 2) {
    fprintf (stderr, "%s socket\n", argv[0]);
    exit (EXIT_FAILURE);
  }
  /* إنشاء مقبض libnbd. */
  nbd = nbd_create ();
  if (nbd == NULL) {
    fprintf (stderr, "%s\n", nbd_get_error ());
    exit (EXIT_FAILURE);
  }
  /* الاتصال بخادم NBD عبر
   * مقبس مجال Unix.
   */
  if (nbd_connect_unix (nbd, argv[1]) == -1) {
    fprintf (stderr, "%s\n", nbd_get_error ());
    exit (EXIT_FAILURE);
  }
  /* الحصول على حجم القرص والتحقق
   * من أنه كبير بما فيه الكفاية.
   */
  exportsize = nbd_get_size (nbd);
  if (exportsize == -1) {
    fprintf (stderr, "%s\n", nbd_get_error ());
    exit (EXIT_FAILURE);
  }
  assert (exportsize >= sizeof buf);
  /* التحقق من أن الخادم قابل للكتابة. */
  if (nbd_is_read_only (nbd) == 1) {
    fprintf (stderr, "%s: "
             "error: this NBD export is read-only\n",
             argv[0]);
    exit (EXIT_FAILURE);
  }
  for (i = 0; i < sizeof buf; ++i)
    buf[i] = rand ();
  /* 1000 عملية كتابة. */
  for (i = 0; i < 1000; ++i) {
    offset = rand () % (exportsize - sizeof buf);
    if (nbd_pwrite (nbd, buf, sizeof buf,
                    offset, 0) == -1) {
      fprintf (stderr, "%s\n", nbd_get_error ());
      exit (EXIT_FAILURE);
    }
  }
  /* 1000 عملية قراءة وكتابة. */
  for (i = 0; i < 1000; ++i) {
    offset = rand () % (exportsize - sizeof buf);
    if (nbd_pread (nbd, buf, sizeof buf,
                   offset, 0) == -1) {
      fprintf (stderr, "%s\n", nbd_get_error ());
      exit (EXIT_FAILURE);
    }
    offset = rand () % (exportsize - sizeof buf);
    if (nbd_pwrite (nbd, buf, sizeof buf,
                    offset, 0) == -1) {
      fprintf (stderr, "%s\n", nbd_get_error ());
      exit (EXIT_FAILURE);
    }
  }
  /* يرسل إيقاف تشغيل سلس للخادم. */
  if (nbd_shutdown (nbd, 0) == -1) {
    fprintf (stderr, "%s\n", nbd_get_error ());
    exit (EXIT_FAILURE);
  }
  nbd_close (nbd);
  exit (EXIT_SUCCESS);
}

nbd_shutdown(3)

nbd_shutdown — shuts down the NBD connection.

NAME

nbd_shutdown — shuts down the NBD connection.

SYNOPSIS

#include <libnbd.h>

void nbd_shutdown(NBDClient *client);

DESCRIPTION

The nbd_shutdown function shuts down the NBD connection associated with the given client. Any outstanding requests will be cancelled, and the connection will be closed.

This function is non-blocking. It initiates the shutdown process, but the actual closing of the connection may happen later, when the event loop is run.

RETURN VALUE

None.

ERRORS

There are no specific error return values. Errors during shutdown are handled internally.

HANDLE STATE

After calling nbd_shutdown, the client handle is in an invalid state. It should not be used for any further operations. The connection will be closed asynchronously. You can check if the connection is closed using nbd_is_connected.

VERSION

This function was introduced in libnbd version 1.2.

EXAMPLE

#include <stdio.h>
#include <stdlib.h>
#include <libnbd.h>

int main() {
    NBDClient *client;
    int ret;

    client = nbd_connect("unix:/path/to/socket");
    if (client == NULL) {
        fprintf(stderr, "failed to connect to NBD server: %s\n", libnbd_get_last_error());
        return 1;
    }

    /* ... perform some NBD operations ... */

    nbd_shutdown(client);
    printf("Shutdown initiated.\n");

    /* The connection might still be open at this point. */
    /* You might want to poll or wait for the event loop to process the shutdown. */

    /* You can close the client handle once the connection is fully closed. */
    nbd_close(client);

    return 0;
}
