nbd_can_multi_conn(3)

الحزمة: libnbd

التراخيص: LGPL-2.1-or-later

nbd_can_multi_conn(3) LIBNBD nbd_can_multi_conn(3)

الاسم

nbd_can_multi_conn - هل يدعم الخادم اتصالات متعددة؟

الملخص

#include <libnbd.h>
int nbd_can_multi_conn (
      struct nbd_handle *h
    );

الوصف

يعيد قيمة صحيحة إذا كان الخادم يدعم اتصالات متعددة. يعيد قيمة خاطئة إذا لم يكن الخادم يدعمها.

ليس من الآمن فتح مقبضين متصلين بنفس الخادم إذا كنت ستكتب إلى الخادم ولم يعلن الخادم عن دعم الاتصالات المتعددة. الطريقة الآمنة للتحقق من ذلك هي فتح اتصال واحد، والتحقق من أن هذا العلم صحيح، ثم فتح اتصالات إضافية حسب الحاجة.

هذه المكالمة لا تحظر، لأنها تعيد بيانات تم حفظها في المقبض من مصافحة بروتوكول NBD.

قيمة الإرجاع

تعيد هذه المكالمة قيمة منطقية.

الأخطاء

عند حدوث خطأ، يتم إرجاع -1.

ارجع إلى "معالجة الأخطاء" في libnbd(3) لمعرفة كيفية الحصول على مزيد من التفاصيل حول الخطأ.

يجب ألا تكون المعلمات التالية فارغة: "h". لمزيد من المعلومات، راجع "المعلمات غير الفارغة" في libnbd(3).

حالة المقبض

يمكن استدعاء nbd_can_multi_conn عندما يكون المقبض في الحالات التالية:

┌─────────────────────────────────────┬─────────────────────────┐
│ تم إنشاء المقبض، قبل الاتصال │ ❌ خطأ │
│ يتم الاتصال │ ❌ خطأ │
│ يتم الاتصال والمصافحة (opt_mode) │ ✅ مسموح به │
│ متصل بالخادم │ ✅ مسموح به │
│ تم إغلاق الاتصال │ ✅ مسموح به │
│ المقبض ميت │ ❌ خطأ │
└─────────────────────────────────────┴─────────────────────────┘

الإصدار

ظهرت هذه الوظيفة لأول مرة في libnbd 1.0.

إذا كنت بحاجة إلى اختبار ما إذا كانت هذه الوظيفة متاحة وقت الترجمة، فتحقق مما إذا كان ما يلي معرفًا:

#define LIBNBD_HAVE_NBD_CAN_MULTI_CONN 1

مثال

هذا المثال متاح أيضًا باسم examples/server-flags.c في كود مصدر libnbd.

/* هذا المثال يوضح كيفية الاتصال بخادم NBD
 * وطباعة علامات التصدير.
 *
 * يمكنك اختباره باستخدام nbdkit كما يلي:
 *
 * nbdkit -U - memory 1M \
 *   --run './server-flags $unixsocket'
 */
#include <stdio.h>
#include <stdlib.h>
#include <stdbool.h>
#include <libnbd.h>
int
main (int argc, char *argv[])
{
  struct nbd_handle *nbd;
  char *str;
  int flag;
  if (argc != 2) {
    fprintf (stderr, "%s socket\n", argv[0]);
    exit (EXIT_FAILURE);
  }
  /* إنشاء مقبض libnbd. */
  nbd = nbd_create ();
  if (nbd == NULL) {
    fprintf (stderr, "%s\n", nbd_get_error ());
    exit (EXIT_FAILURE);
  }
  /* طلب معلومات كاملة. */
#if LIBNBD_HAVE_NBD_SET_FULL_INFO /* تم الإضافة في 1.4 */
  if (nbd_set_full_info (nbd, true) == -1) {
    fprintf (stderr, "%s\n", nbd_get_error ());
    exit (EXIT_FAILURE);
  }
#endif
  /* الاتصال بخادم NBD عبر
   * مقبس نطاق Unix.
   */
  if (nbd_connect_unix (nbd, argv[1]) == -1) {
    fprintf (stderr, "%s\n", nbd_get_error ());
    exit (EXIT_FAILURE);
  }
  /* انظر ما إذا كان الخادم قد قدم تفاصيل إضافية،
   * باستخدام الوظائف التي تمت إضافتها في 1.4
   */
#if LIBNBD_HAVE_NBD_GET_EXPORT_DESCRIPTION
  str = nbd_get_canonical_export_name (nbd);
  if (str)
    printf ("canonical_name = %s\n", str);
  free (str);
  str = nbd_get_export_description (nbd);
  if (str)
    printf ("description = %s\n", str);
  free (str);
#endif
  /* قراءة وطباعة العلامات. */
#define PRINT_FLAG(flag_fn)                     \
  flag = flag_fn (nbd);                         \
  if (flag == -1) {                             \
    fprintf (stderr, "%s\n", nbd_get_error ()); \
    exit (EXIT_FAILURE);                        \
  }                                             \
  printf (#flag_fn " = %s\n",                   \
          flag ? "true" : "false");
  PRINT_FLAG (nbd_can_cache);
  PRINT_FLAG (nbd_can_df);
  PRINT_FLAG (nbd_can_flush);
  PRINT_FLAG (nbd_can_fua);
  PRINT_FLAG (nbd_can_multi_conn);
  PRINT_FLAG (nbd_can_trim);
  PRINT_FLAG (nbd_can_zero);
#if LIBNBD_HAVE_NBD_CAN_FAST_ZERO
  /* تم الإضافة في 1.2 */
  PRINT_FLAG (nbd_can_fast_zero);
#endif
#if LIBNBD_HAVE_NBD_CAN_BLOCK_STATUS_PAYLOAD
  /* تم الإضافة في 1.18 */
  PRINT_FLAG (nbd_can_block_status_payload);
#endif
  PRINT_FLAG (nbd_is_read_only);
  PRINT_FLAG (nbd_is_rotational);
  /* إغلاق مقبض libnbd. */
  nbd_close (nbd);
  exit (EXIT_SUCCESS);
}

NAME

nbd_can_multi_conn - check if the server supports multiple connections

SYNOPSIS

#include <libnbd.h>

int nbd_can_multi_conn(NBD_CONTEXT *ctx);

DESCRIPTION

The nbd_can_multi_conn function checks if the NBD server connected to via the context ctx supports multiple simultaneous connections.

This function can only be called after a connection to the server has been established (e.g., using nbd_connect_unix, nbd_connect_socket, nbd_connect_uri or nbd_connect).

If the server supports multiple connections, the function returns 1. Otherwise, it returns 0.

PARAMETERS

ctx
A pointer to the NBD_CONTEXT.

RETURN

Returns 1 if the server supports multiple connections, 0 otherwise.

ERRORS

On error, it returns -1 and sets errno.

The following error codes can be set:

errno EFAULT
ctx is NULL.
errno ENOTCONN
The NBD context is not connected to a server.
errno EPROTO
The server sent a response that could not be parsed.
errno EIO
An I/O error occurred while communicating with the server.
errno EACCES
Permission denied.

HANDLE STATE

The handle state remains unchanged.

VERSION

This function was introduced in libnbd version 1.10.

EXAMPLE

#include <libnbd.h>
#include <stdio.h>

int main(void) {
    NBD_CONTEXT *ctx;
    int can_multi_conn;

    ctx = nbd_context_new();
    if (!ctx) {
        perror("nbd_context_new");
        return 1;
    }

    /* Connect to a server (replace with your server details) */
    if (nbd_connect_uri(ctx, "nbd:my-server.example.com:10809") == -1) {
        fprintf(stderr, "nbd_connect_uri failed: %s\n", nbd_get_error(ctx));
        nbd_context_destroy(ctx);
        return 1;
    }

    can_multi_conn = nbd_can_multi_conn(ctx);
    if (can_multi_conn == -1) {
        fprintf(stderr, "nbd_can_multi_conn failed: %s\n", nbd_get_error(ctx));
    } else if (can_multi_conn) {
        printf("Server supports multiple connections.\n");
    } else {
        printf("Server does not support multiple connections.\n");
    }

    nbd_shutdown(ctx);
    nbd_context_destroy(ctx);
    return 0;
}

الاسم

nbd_can_multi_conn - التحقق مما إذا كان الخادم يدعم اتصالات متعددة

الملخص

#include <libnbd.h>

int nbd_can_multi_conn(NBD_CONTEXT *ctx);

الوصف

تقوم الدالة nbd_can_multi_conn بالتحقق مما إذا كان خادم NBD المتصل عبر السياق ctx يدعم اتصالات متزامنة متعددة.

لا يمكن استدعاء هذه الدالة إلا بعد إنشاء اتصال بالخادم (على سبيل المثال، باستخدام nbd_connect_unix أو nbd_connect_socket أو nbd_connect_uri أو nbd_connect).

إذا كان الخادم يدعم اتصالات متعددة، تُرجع الدالة 1. بخلاف ذلك، تُرجع 0.

المعاملات

ctx
مؤشر إلى NBD_CONTEXT.

الإرجاع

تُرجع 1 إذا كان الخادم يدعم اتصالات متعددة، 0 بخلاف ذلك.

الأخطاء

عند حدوث خطأ، تُرجع -1 وتضبط errno.

يمكن تعيين رموز الخطأ التالية:

errno EFAULT
ctx هو NULL.
errno ENOTCONN
سياق NBD غير متصل بخادم.
errno EPROTO
أرسل الخادم استجابة لم يمكن تحليلها.
errno EIO
حدث خطأ إدخال/إخراج أثناء الاتصال بالخادم.
errno EACCES
تم رفض الإذن.

حالة المقبض

تبقى حالة المقبض دون تغيير.

الإصدار

تم تقديم هذه الدالة في إصدار libnbd 1.10.

مثال

#include <libnbd.h>
#include <stdio.h>

int main(void) {

    int can_multi_conn;

        perror("nbd_context_new");
        return 1;
    }

    /* الاتصال بخادم (استبدل بتفاصيل الخادم الخاص بك) */
    if (nbd_connect_uri(ctx, "nbd:my-server.example.com:10809") == -1) {
        fprintf(stderr, "nbd_connect_uri failed: %s\n", nbd_get_error(ctx));
        nbd_context_destroy(ctx);
        return 1;
    }

    can_multi_conn = nbd_can_multi_conn(ctx);
    if (can_multi_conn == -1) {
        fprintf(stderr, "nbd_can_multi_conn failed: %s\n", nbd_get_error(ctx));
    } else if (can_multi_conn) {
        printf("الخادم يدعم اتصالات متعددة.\n");
    } else {
        printf("الخادم لا يدعم اتصالات متعددة.\n");
    }

    nbd_shutdown(ctx);
    nbd_context_destroy(ctx);
    return 0;
}
