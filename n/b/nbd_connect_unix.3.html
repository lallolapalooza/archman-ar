nbd_connect_unix(3)

الحزمة: libnbd

التراخيص: LGPL-2.1-or-later

nbd_connect_unix(3) LIBNBD nbd_connect_unix(3)

الاسم

nbd_connect_unix - الاتصال بخادم NBD عبر مقبس مجال Unix

الملخص

#include <libnbd.h>
int nbd_connect_unix (
      struct nbd_handle *h, const char *unixsocket
    );

الوصف

الاتصال (بشكل متزامن) عبر مقبس مجال Unix المسمى ("unixsocket") بخادم NBD يعمل على نفس الجهاز.

يعود هذا الاستدعاء عند إنشاء الاتصال. بشكل افتراضي، يمضي هذا حتى مرحلة الإرسال، ولكن يمكن استخدام nbd_set_opt_mode(3) للتحكم اليدوي في مفاوضات الخيارات التي تتم قبل مرحلة الإرسال.

قيمة الإرجاع

إذا نجح الاستدعاء، تعيد الدالة 0.

الأخطاء

عند حدوث خطأ، يتم إرجاع -1.

راجع "معالجة الأخطاء" في libnbd(3) للحصول على مزيد من التفاصيل حول الخطأ.

يجب ألا تكون المعلمات التالية NULL: "h"، "unixsocket". لمزيد من المعلومات، راجع "المعلمات غير الفارغة" في libnbd(3).

حالة المقبض

يمكن استدعاء nbd_connect_unix عندما يكون المقبض في الحالة التالية:

┌─────────────────────────────────────┬─────────────────────────┐
│ تم إنشاء المقبض، قبل الاتصال │ ✅ مسموح به │
│ جاري الاتصال │ ❌ خطأ │
│ جاري الاتصال والمصافحة (opt_mode) │ ❌ خطأ │
│ تم الاتصال بالخادم │ ❌ خطأ │
│ تم إغلاق الاتصال │ ❌ خطأ │
│ المقبض معطل │ ❌ خطأ │
└─────────────────────────────────────┴─────────────────────────┘

الإصدار

ظهرت هذه الوظيفة لأول مرة في libnbd 1.0.

إذا كنت بحاجة إلى اختبار ما إذا كانت هذه الوظيفة متاحة في وقت الترجمة، فتحقق مما إذا كان الماكرو التالي معرفًا:

#define LIBNBD_HAVE_NBD_CONNECT_UNIX 1

مثال

هذا المثال متاح أيضًا باسم examples/fetch-first-sector.c في كود مصدر libnbd.

/* يوضح هذا المثال كيفية الاتصال بخادم NBD
 * وجلب وطباعة القطاع الأول (عادةً ما يكون قطاع التمهيد
 * أو جدول الأقسام أو الموصل الفائق لنظام الملفات).
 *
 * يمكنك اختباره باستخدام nbdkit كما يلي:
 *
 * nbdkit -U - floppy . \
 *   --run './fetch-first-sector $unixsocket'
 *
 * يقوم مكون nbdkit المكون من الأقراص المرنة بإنشاء قرص MBR، لذا فإن
 * القطاع الأول هو جدول الأقسام.
 */
#include <stdio.h>
#include <stdlib.h>
#include <libnbd.h>
int
main (int argc, char *argv[])
{
  struct nbd_handle *nbd;
  char buf[512];
  FILE *pp;
  if (argc != 2) {
    fprintf (stderr, "%s socket\n", argv[0]);
    exit (EXIT_FAILURE);
  }
  /* إنشاء مقبض libnbd. */
  nbd = nbd_create ();
  if (nbd == NULL) {
    fprintf (stderr, "%s\n", nbd_get_error ());
    exit (EXIT_FAILURE);
  }
  /* الاتصال بخادم NBD عبر
   * مقبس مجال Unix.
   */
  if (nbd_connect_unix (nbd, argv[1]) == -1) {
    fprintf (stderr, "%s\n", nbd_get_error ());
    exit (EXIT_FAILURE);
  }
  /* قراءة القطاع الأول بشكل متزامن. */
  if (nbd_pread (nbd, buf, sizeof buf, 0, 0) == -1) {
    fprintf (stderr, "%s\n", nbd_get_error ());
    exit (EXIT_FAILURE);
  }
  /* إغلاق مقبض libnbd. */
  nbd_close (nbd);
  /* طباعة القطاع الأول. */
  pp = popen ("hexdump -C", "w");
  if (pp == NULL) {
    perror ("popen: hexdump");
    exit (EXIT_FAILURE);
  }
  fwrite (buf, sizeof buf, 1, pp);
  pclose (pp);
  exit (EXIT_SUCCESS);
}
