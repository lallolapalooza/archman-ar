nbd_can_block_status_payload(3)

الحزمة: libnbd

التراخيص: LGPL-2.1-or-later

nbd_can_block_status_payload(3) LIBNBD nbd_can_block_status_payload(3)

الاسم

nbd_can_block_status_payload - هل يدعم الخادم علامة حمولة حالة الكتلة؟

ملخص

#include <libnbd.h>
int nbd_can_block_status_payload (
      struct nbd_handle *h
    );

الوصف

تُرجع قيمة صحيحة إذا كان الخادم يدعم استخدام العلامة "LIBNBD_CMD_FLAG_PAYLOAD_LEN" للسماح بتصفية أمر حالة الكتلة (انظر nbd_block_status_filter(3)). تُرجع قيمة خاطئة إذا لم يكن الخادم يدعمها. لاحظ أن هذا لن يُرجع قيمة صحيحة أبدًا إذا كانت nbd_get_extended_headers_negotiated(3) خاطئة.

هذه المكالمة لا تمنع، لأنها تُرجع بيانات تم حفظها في المقبض من مصافحة بروتوكول NBD.

قيمة الإرجاع

تُرجع هذه المكالمة قيمة منطقية.

الأخطاء

عند حدوث خطأ، يتم إرجاع -1.

ارجع إلى "معالجة الأخطاء" في libnbd(3) للحصول على مزيد من التفاصيل حول الخطأ.

يجب ألا تكون المعلمات التالية NULL: "h". لمزيد من المعلومات، انظر "المعلمات غير الفارغة" في libnbd(3).

حالة المقبض

يمكن استدعاء nbd_can_block_status_payload عندما يكون المقبض في الحالات التالية:

┌─────────────────────────────────────┬─────────────────────────┐
│ تم إنشاء المقبض، قبل الاتصال   │ ❌ خطأ                │
│ جارٍ الاتصال                          │ ❌ خطأ                │
│ جارٍ الاتصال & المصافحة (opt_mode) │ ✅ مسموح به              │
│ تم الاتصال بالخادم             │ ✅ مسموح به              │
│ تم إنهاء الاتصال                │ ✅ مسموح به              │
│ المقبض معطل                         │ ❌ خطأ                │
└─────────────────────────────────────┴─────────────────────────┘

الإصدار

ظهرت هذه الوظيفة لأول مرة في libnbd 1.18.

إذا كنت بحاجة إلى التحقق مما إذا كانت هذه الوظيفة متاحة وقت التصريف، تحقق مما إذا كان ما يلي معرفًا:

#define LIBNBD_HAVE_NBD_CAN_BLOCK_STATUS_PAYLOAD 1

مثال

هذا المثال متاح أيضًا باسم examples/server-flags.c في شفرة المصدر libnbd.

/* هذا المثال يوضح كيفية الاتصال بخادم NBD
 * وطباعة علامات التصدير.
 *
 * يمكنك اختباره باستخدام nbdkit مثل هذا:
 *
 * nbdkit -U - memory 1M \
 *   --run './server-flags $unixsocket'
 */
#include <stdio.h>
#include <stdlib.h>
#include <stdbool.h>
#include <libnbd.h>
int
main (int argc, char *argv[])
{
  struct nbd_handle *nbd;
  char *str;
  int flag;
  if (argc != 2) {
    fprintf (stderr, "%s socket\n", argv[0]);
    exit (EXIT_FAILURE);
  }
  /* إنشاء مقبض libnbd. */
  nbd = nbd_create ();
  if (nbd == NULL) {
    fprintf (stderr, "%s\n", nbd_get_error ());
    exit (EXIT_FAILURE);
  }
  /* طلب معلومات كاملة. */
#if LIBNBD_HAVE_NBD_SET_FULL_INFO /* تم الإضافة في 1.4 */
  if (nbd_set_full_info (nbd, true) == -1) {
    fprintf (stderr, "%s\n", nbd_get_error ());
    exit (EXIT_FAILURE);
  }
#endif
  /* الاتصال بخادم NBD عبر
   * مقبس مجال Unix.
   */
  if (nbd_connect_unix (nbd, argv[1]) == -1) {
    fprintf (stderr, "%s\n", nbd_get_error ());
    exit (EXIT_FAILURE);
  }
  /* التحقق مما إذا كان الخادم قد قدم تفاصيل إضافية،
   * باستخدام الوظائف التي تم إضافتها في 1.4
   */
#if LIBNBD_HAVE_NBD_GET_EXPORT_DESCRIPTION
  str = nbd_get_canonical_export_name (nbd);
  if (str)
    printf ("canonical_name = %s\n", str);
  free (str);
  str = nbd_get_export_description (nbd);
  if (str)
    printf ("description = %s\n", str);
  free (str);
#endif
  /* قراءة وطباعة العلامات. */
#define PRINT_FLAG(flag_fn)                     \
  flag = flag_fn (nbd);                         \
  if (flag == -1) {                             \
    fprintf (stderr, "%s\n", nbd_get_error ()); \
    exit (EXIT_FAILURE);                        \
  }                                             \
  printf (#flag_fn " = %s\n",                   \
          flag ? "true" : "false");
  PRINT_FLAG (nbd_can_cache);
  PRINT_FLAG (nbd_can_df);
  PRINT_FLAG (nbd_can_flush);
  PRINT_FLAG (nbd_can_fua);
  PRINT_FLAG (nbd_can_multi_conn);
  PRINT_FLAG (nbd_can_trim);
  PRINT_FLAG (nbd_can_zero);
#if LIBNBD_HAVE_NBD_CAN_FAST_ZERO
  /* تم الإضافة في 1.2 */
  PRINT_FLAG (nbd_can_fast_zero);
#endif
#if LIBNBD_HAVE_NBD_CAN_BLOCK_STATUS_PAYLOAD
  /* تم الإضافة في 1.18 */
  PRINT_FLAG (nbd_can_block_status_payload);
#endif
  PRINT_FLAG (nbd_is_read_only);
  PRINT_FLAG (nbd_is_rotational);
  /* إغلاق مقبض libnbd. */
  nbd_close (nbd);
  exit (EXIT_SUCCESS);
}

NAME

nbd_can_block_status_payload - check if NBD server can send block status payload

SYNOPSIS

#include <libnbd.h>

int
nbd_can_block_status_payload (NbdHandle *h);

DESCRIPTION

The nbd_can_block_status_payload() function checks if the connected NBD server supports sending the block status payload as defined in the NBD_OPT_INFO option of the NBD protocol. This feature allows the client to receive information about the status of blocks (e.g., trim/discarded, hole, etc.) from the server.

This function is used to determine if the client can rely on receiving block status information for operations like `nbd_trim`, `nbd_discard`, or `nbd_zero`.

RETURN VALUE

Returns 1 if the server supports block status payload, 0 otherwise.

ERRORS

This function does not set errno. However, if the connection is not established or has errors, the behavior is undefined. Use `nbd_get_error` after calling other `libnbd` functions to check for errors.

HANDLE STATE

The `NbdHandle` must be connected to an NBD server.

VERSION

This function is available since libnbd 1.14.

EXAMPLE

#include <stdio.h>
#include <libnbd.h>

int main() {
    NbdHandle *h = nbd_connect_simple("unix:/path/to/nbd.sock");
    if (!h) {
        fprintf(stderr, "Failed to connect to NBD server: %s\n", nbd_get_error(NULL));
        return 1;
    }

    if (nbd_can_block_status_payload(h)) {
        printf("Server supports block status payload.\n");
    } else {
        printf("Server does not support block status payload.\n");
    }

    nbd_shutdown(h);
    nbd_close(h);
    return 0;
}

الاسم

nbd_can_block_status_payload - التحقق مما إذا كان خادم NBD يمكنه إرسال حمولة حالة الكتلة

الملخص

#include <libnbd.h>

int
nbd_can_block_status_payload (NbdHandle *h);

الوصف

تقوم الدالة nbd_can_block_status_payload() بالتحقق مما إذا كان خادم NBD المتصل يدعم إرسال حمولة حالة الكتلة كما هو محدد في الخيار NBD_OPT_INFO من بروتوكول NBD. تسمح هذه الميزة للعميل بتلقي معلومات حول حالة الكتل (مثل، تم اقتطاعها/تجاهلها، ثقب، إلخ) من الخادم.

تُستخدم هذه الدالة لتحديد ما إذا كان بإمكان العميل الاعتماد على تلقي معلومات حالة الكتلة لعمليات مثل `nbd_trim`، `nbd_discard`، أو `nbd_zero`.

قيمة الإرجاع

تُرجع 1 إذا كان الخادم يدعم حمولة حالة الكتلة، و 0 بخلاف ذلك.

الأخطاء

لا تقوم هذه الدالة بتعيين errno. ومع ذلك، إذا لم يكن الاتصال قائمًا أو كانت هناك أخطاء، فإن السلوك غير محدد. استخدم `nbd_get_error` بعد استدعاء وظائف `libnbd` الأخرى للتحقق من الأخطاء.

حالة المقبض

يجب أن يكون `NbdHandle` متصلاً بخادم NBD.

الإصدار

هذه الدالة متاحة منذ libnbd 1.14.

مثال

#include <stdio.h>
#include <libnbd.h>

int main() {
    NbdHandle *h = nbd_connect_simple("unix:/path/to/nbd.sock");
    if (!h) {
        fprintf(stderr, "فشل الاتصال بخادم NBD: %s\n", nbd_get_error(NULL));
        return 1;
    }

    if (nbd_can_block_status_payload(h)) {
        printf("الخادم يدعم حمولة حالة الكتلة.\n");
    } else {
        printf("الخادم لا يدعم حمولة حالة الكتلة.\n");
    }

    nbd_shutdown(h);
    nbd_close(h);
    return 0;
}
