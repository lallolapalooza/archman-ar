nbd_can_trim(3)

الحزمة: libnbd

التراخيص: LGPL-2.1-or-later

nbd_can_trim(3) LIBNBD nbd_can_trim(3)

الاسم

nbd_can_trim - هل يدعم الخادم أمر القص (trim)؟

الملخص

#include <libnbd.h>
int nbd_can_trim (
      struct nbd_handle *h
    );

الوصف

يعيد قيمة صحيحة إذا كان الخادم يدعم أمر القص (trim) (انظر nbd_trim(3)، nbd_aio_trim(3)). يعيد قيمة خاطئة إذا لم يكن الخادم يدعمه.

لا يقوم هذا الاستدعاء بحظر، لأنه يعيد بيانات تم حفظها في المقبض من المصافحة بروتوكول NBD.

قيمة الإعادة

يقوم هذا الاستدعاء بإعادة قيمة منطقية.

الأخطاء

عند حدوث خطأ، يتم إرجاع -1.

راجع "معالجة الأخطاء" في libnbd(3) لمعرفة كيفية الحصول على تفاصيل إضافية حول الخطأ.

يجب ألا تكون المعلمات التالية فارغة: "h". لمزيد من المعلومات، انظر "المعلمات غير الفارغة" في libnbd(3).

حالة المقبض

يمكن استدعاء nbd_can_trim عندما يكون المقبض في الحالات التالية:

┌─────────────────────────────────────┬─────────────────────────┐
│ تم إنشاء المقبض، قبل الاتصال   │ ❌ خطأ                │
│ جاري الاتصال                      │ ❌ خطأ                │
│ جاري الاتصال والمصافحة (opt_mode) │ ✅ مسموح به            │
│ تم الاتصال بالخادم             │ ✅ مسموح به            │
│ تم إغلاق الاتصال                │ ✅ مسموح به            │
│ المقبض ميت                       │ ❌ خطأ                │
└─────────────────────────────────────┴─────────────────────────┘

الإصدار

ظهرت هذه الوظيفة لأول مرة في libnbd 1.0.

إذا كنت بحاجة إلى اختبار ما إذا كانت هذه الوظيفة متاحة في وقت الترجمة تحقق مما إذا كان ما يلي معرفًا:

#define LIBNBD_HAVE_NBD_CAN_TRIM 1

مثال

هذا المثال متاح أيضًا باسم examples/server-flags.c في شفرة مصدر libnbd.

/* هذا المثال يوضح كيفية الاتصال بخادم NBD
 * وطباعة علامات التصدير.
 *
 * يمكنك اختباره باستخدام nbdkit مثل هذا:
 *
 * nbdkit -U - memory 1M \
 *   --run './server-flags $unixsocket'
 */
#include <stdio.h>
#include <stdlib.h>
#include <stdbool.h>
#include <libnbd.h>
int
main (int argc, char *argv[])
{
  struct nbd_handle *nbd;
  char *str;
  int flag;
  if (argc != 2) {
    fprintf (stderr, "%s socket\n", argv[0]);
    exit (EXIT_FAILURE);
  }
  /* إنشاء مقبض libnbd. */
  nbd = nbd_create ();
  if (nbd == NULL) {
    fprintf (stderr, "%s\n", nbd_get_error ());
    exit (EXIT_FAILURE);
  }
  /* طلب معلومات كاملة. */
#if LIBNBD_HAVE_NBD_SET_FULL_INFO /* تمت الإضافة في 1.4 */
  if (nbd_set_full_info (nbd, true) == -1) {
    fprintf (stderr, "%s\n", nbd_get_error ());
    exit (EXIT_FAILURE);
  }
#endif
  /* الاتصال بخادم NBD عبر
   * مقبس نطاق Unix.
   */
  if (nbd_connect_unix (nbd, argv[1]) == -1) {
    fprintf (stderr, "%s\n", nbd_get_error ());
    exit (EXIT_FAILURE);
  }
  /* انظر ما إذا كان الخادم قد قدم تفاصيل إضافية،
   * باستخدام وظائف تمت إضافتها في 1.4
   */
#if LIBNBD_HAVE_NBD_GET_EXPORT_DESCRIPTION
  str = nbd_get_canonical_export_name (nbd);
  if (str)
    printf ("canonical_name = %s\n", str);
  free (str);
  str = nbd_get_export_description (nbd);
  if (str)
    printf ("description = %s\n", str);
  free (str);
#endif
  /* قراءة وطباعة العلامات. */
#define PRINT_FLAG(flag_fn)                     \
  flag = flag_fn (nbd);                         \
  if (flag == -1) {                             \
    fprintf (stderr, "%s\n", nbd_get_error ()); \
    exit (EXIT_FAILURE);                        \
  }                                             \
  printf (#flag_fn " = %s\n",                   \
          flag ? "true" : "false");
  PRINT_FLAG (nbd_can_cache);
  PRINT_FLAG (nbd_can_df);
  PRINT_FLAG (nbd_can_flush);
  PRINT_FLAG (nbd_can_fua);
  PRINT_FLAG (nbd_can_multi_conn);
  PRINT_FLAG (nbd_can_trim);
  PRINT_FLAG (nbd_can_zero);
#if LIBNBD_HAVE_NBD_CAN_FAST_ZERO
  /* تمت الإضافة في 1.2 */
  PRINT_FLAG (nbd_can_fast_zero);
#endif
#if LIBNBD_HAVE_NBD_CAN_BLOCK_STATUS_PAYLOAD
  /* تمت الإضافة في 1.18 */
  PRINT_FLAG (nbd_can_block_status_payload);
#endif
  PRINT_FLAG (nbd_is_read_only);
  PRINT_FLAG (nbd_is_rotational);
  /* إغلاق مقبض libnbd. */
  nbd_close (nbd);
  exit (EXIT_SUCCESS);
}

nbd_can_trim(3)

NAME

nbd_can_trim - Check if the NBD server can trim blocks

SYNOPSIS

int nbd_can_trim(int fd);

DESCRIPTION

The nbd_can_trim function checks if the NBD server connected to file descriptor fd supports the NBD_OPT_CAN_TRIM option. This option indicates that the server can handle trim (discard) operations. It is used to inform the client that it's safe to send trim commands to the server.

PARAMETERS

  • fd: The file descriptor of the connected NBD socket.

RETURN VALUE

Returns 1 if the server supports trimming, 0 otherwise. Returns -1 if there was an error communicating with the server.

EXAMPLE

#include <libnbd.h>

int main() {
    int fd = /* ... connect to NBD server ... */;
    if (nbd_can_trim(fd)) {
        printf("Server supports trimming.\n");
    } else {
        printf("Server does not support trimming.\n");
    }
    // ... close NBD connection ...
    return 0;
}

EXPORTS

This function is exported by the libnbd library.
