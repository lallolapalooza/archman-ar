nbd_aio_pread(3)

الحزمة: libnbd

التراخيص: LGPL-2.1-or-later

nbd_aio_pread(3) LIBNBD nbd_aio_pread(3)

NAME

nbd_aio_pread - قراءة من خادم NBD

SYNOPSIS

#include <libnbd.h>
typedef struct {
  int (*callback) (void *user_data, int *error);
  void *user_data;
  void (*free) (void *user_data);
} nbd_completion_callback;
int64_t nbd_aio_pread (
          struct nbd_handle *h, void *buf, size_t count,
          uint64_t offset,
          nbd_completion_callback completion_callback,
          uint32_t flags
        );

DESCRIPTION

إصدار أمر قراءة إلى خادم NBD.

للتحقق مما إذا كان الأمر قد اكتمل، استدعِ nbd_aio_command_completed(3). أو قم بتوفير "completion_callback" الاختياري الذي سيتم استدعاؤه كما هو موضح في "Completion callbacks" في libnbd(3).

لاحظ أنه يجب عليك التأكد من أن "buf" صالح حتى يكتمل الأمر. علاوة على ذلك، إذا تم تعيين المعلمة "error" إلى "completion_callback" أو إذا أبلغ nbd_aio_command_completed(3) عن فشل، وإذا أعادت nbd_get_pread_initialize(3) قيمة true، فقد قامت libnbd بتطهير "buf"، ولكن لم يتم تحديد ما إذا كانت محتويات "buf" ستقرأ كصفر أو كنتائج جزئية من الخادم. إذا أعادت nbd_get_pread_initialize(3) قيمة false، فإن libnbd لم تطهر "buf"، والمحتويات غير محددة عند الفشل.

تتصرف المعلمات الأخرى كما هو موثق في nbd_pread(3).

بشكل افتراضي، سترفض libnbd المحاولات لاستخدام هذه الدالة مع معلمات من المحتمل أن تؤدي إلى فشل الخادم، مثل طلب علامة أمر غير معروفة. يمكن استخدام الدالة nbd_set_strict_mode(3) لتغيير السيناريوهات التي يجب أن تنتظر فيها رد الخادم بدلاً من الفشل السريع.

RETURN VALUE

تعيد هذه المكالمة معرف الأمر ذو الـ 64 بت. المعرف هو ≥ 1. المعرفات فريدة (لكل مقبض libnbd، وليس عالميًا).

ERRORS

عند حدوث خطأ، يتم إرجاع -1.

ارجع إلى "ERROR HANDLING" في libnbd(3) للحصول على مزيد من التفاصيل حول الخطأ.

يجب ألا تكون المعلمات التالية NULL: "h"، "buf". لمزيد من المعلومات، انظر "Non-NULL parameters" في libnbd(3).

HANDLE STATE

يمكن استدعاء nbd_aio_pread عندما يكون المقبض في الحالة التالية:

┌─────────────────────────────────────┬─────────────────────────┐
│ تم إنشاء المقبض، قبل الاتصال   │ ❌ خطأ                │
│ جارٍ الاتصال                     │ ❌ خطأ                │
│ جارٍ الاتصال وإتمام المصافحة (opt_mode) │ ❌ خطأ                │
│ متصل بالخادم                  │ ✅ مسموح به              │
│ تم إغلاق الاتصال                 │ ❌ خطأ                │
│ المقبض ميت                       │ ❌ خطأ                │
└─────────────────────────────────────┴─────────────────────────┘

VERSION

ظهرت هذه الدالة لأول مرة في libnbd 1.0.

إذا كنت بحاجة إلى اختبار ما إذا كانت هذه الدالة متاحة وقت الترجمة، تحقق مما إذا كان الماكرو التالي معرفًا:

#define LIBNBD_HAVE_NBD_AIO_PREAD 1

EXAMPLE

هذا المثال متاح أيضًا باسم examples/aio-connect-read.c في الكود المصدري لـ libnbd.

/* يوضح هذا المثال كيفية استخدام واجهة برمجة التطبيقات المنخفضة المستوى (غير متزامنة)
 * للاتصال بخادم وقراءة القرص.
 *
 * إليك بعض الطرق لتجربة هذا المثال:
 *
 * nbdkit -U - linuxdisk . \
 *   --run './aio-connect-read $unixsocket'
 *
 * nbdkit -U - floppy . \
 *   --run './aio-connect-read $unixsocket'
 *
 * nbdkit -U - pattern size=1M \
 *   --run './aio-connect-read $unixsocket'
 */
#include <stdio.h>
#include <stdlib.h>
#include <stdint.h>
#include <inttypes.h>
#include <errno.h>
#include <assert.h>
#include <libnbd.h>
#define NR_SECTORS 32
#define SECTOR_SIZE 512
struct data {
  uint64_t offset;
  char sector[SECTOR_SIZE];
};
static int
hexdump (void *user_data, int *error)
{
  struct data *data = user_data;
  FILE *pp;
  if (*error) {
    errno = *error;
    perror ("failed to read");
    exit (EXIT_FAILURE);
  }
  printf ("sector at offset 0x%" PRIx64 ":\n",
          data->offset);
  pp = popen ("hexdump -C", "w");
  if (pp == NULL) {
    perror ("popen: hexdump");
    exit (EXIT_FAILURE);
  }
  fwrite (data->sector, SECTOR_SIZE, 1, pp);
  pclose (pp);
  printf ("\n");
  /* العودة بـ 1 من رد الاتصال ينهي
   * الأمر تلقائيًا.
   */
  return 1;
}
static struct data data[NR_SECTORS];
int
main (int argc, char *argv[])
{
  struct nbd_handle *nbd;
  size_t i;
  if (argc != 2) {
    fprintf (stderr, "%s socket\n", argv[0]);
    exit (EXIT_FAILURE);
  }
  /* إنشاء مقبض libnbd. */
  nbd = nbd_create ();
  if (nbd == NULL) {
    fprintf (stderr, "%s\n", nbd_get_error ());
    exit (EXIT_FAILURE);
  }
  /* الاتصال بخادم NBD عبر مقبس مجال Unix.
   * يبدأ هذا الاتصال فقط.
   */
  if (nbd_aio_connect_unix (nbd, argv[1]) == -1) {
    fprintf (stderr, "%s\n", nbd_get_error ());
    exit (EXIT_FAILURE);
  }
  /* الانتظار حتى يكتمل الاتصال. استخدام
   * nbd_poll هنا هو كمثال فقط. يمكنك أيضًا
   * دمج هذا مع poll(2) أو glib أو حلقة رئيسية أخرى.
   * اقرأ libnbd(3) وملف المصدر lib/poll.c.
   */
  while (!nbd_aio_is_ready (nbd)) {
    if (nbd_poll (nbd, -1) == -1) {
      fprintf (stderr, "%s\n", nbd_get_error ());
      exit (EXIT_FAILURE);
    }
  }
  assert (nbd_get_size (nbd) >= NR_SECTORS * SECTOR_SIZE);
  /* إصدار أوامر القراءة لأول NR قطاعات. */
  for (i = 0; i < NR_SECTORS; ++i) {
    data[i].offset = i * SECTOR_SIZE;
    /* يتم استدعاء رد الاتصال (hexdump) عند اكتمال
     * الأمر. يجب أن تستمر الذاكرة المؤقتة في الوجود
     * أثناء تشغيل الأمر.
     */
    if (nbd_aio_pread (nbd, data[i].sector, SECTOR_SIZE,
                       data[i].offset,
                       (nbd_completion_callback) {
                         .callback = hexdump,
                         .user_data = &data[i],
                       }, 0) == -1) {
      fprintf (stderr, "%s\n", nbd_get_error ());
      exit (EXIT_FAILURE);
    }
  }
  /* تشغيل الحلقة الرئيسية حتى تكتمل جميع الأوامر
   * وتتقاعد. مرة أخرى، استخدام nbd_poll
   * هنا هو كمثال فقط.
   */
  while (nbd_aio_in_flight (nbd) > 0) {
    if (nbd_poll (nbd, -1) == -1) {
      fprintf (stderr, "%s\n", nbd_get_error ());
      exit (EXIT_FAILURE);
    }
  }
  /* إغلاق مقبض libnbd. */
  nbd_close (nbd);
  exit (EXIT_SUCCESS);
}
