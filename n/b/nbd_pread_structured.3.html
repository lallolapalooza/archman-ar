nbd_pread_structured(3)

الحزمة: libnbd

التراخيص: LGPL-2.1-or-later

nbd_pread_structured(3) LIBNBD nbd_pread_structured(3)

الاسم

nbd_pread_structured - القراءة من خادم NBD

ملخص

#include <libnbd.h>
typedef struct {
  int (*callback) (void *user_data, const void *subbuf,
                   size_t count, uint64_t offset,
                   unsigned status, int *error);
  void *user_data;
  void (*free) (void *user_data);
} nbd_chunk_callback;
int nbd_pread_structured (
      struct nbd_handle *h, void *buf, size_t count,
      uint64_t offset, nbd_chunk_callback chunk_callback,
      uint32_t flags
    );

الوصف

إصدار أمر قراءة إلى خادم NBD للنطاق الذي يبدأ عند "offset" وينتهي عند "offset" + "count" - 1. قد يتم تقسيم استجابة الخادم إلى أجزاء قد تصل بترتيب غير صحيح قبل إعادة تجميعها في المخزن المؤقت الأصلي؛ يتم استخدام الدالة الإجرائية "chunk" للإشعار بعد وصول كل جزء، وقد تقوم بإجراء فحوصات إضافية على استجابة الخادم. لا يمكن للدالة الإجرائية استدعاء واجهات برمجة التطبيقات "nbd_*" على نفس المقبض لأنها تحتفظ بقفل المقبض وستؤدي إلى حالة تعليق. إذا أعادت الدالة الإجرائية القيمة -1، ولم يتم اكتشاف أي خطأ سابق، فسيفشل أمر القراءة الإجمالي بأي قيمة غير صفرية مخزنة في معامل "error" الخاص بالدالة الإجرائية (مع القيمة الافتراضية "EPROTO")؛ ولكن سيتم استدعاء الدالة الإجرائية لأي أجزاء أخرى.

يتم استدعاء الدالة "chunk" مرة واحدة لكل جزء من البيانات المستلمة، مع تمرير "user_data" إلى هذه الدالة. تمثل المعاملات "subbuf" و "count" الجزء من المخزن المؤقت الأصلي الذي تم ملؤه للتو بنتائج من الخادم (في C، يشير "subbuf" دائمًا إلى المخزن المؤقت الأصلي "buf"؛ ولكن هذا الضمان قد لا يمتد إلى ربط اللغات الأخرى). يمثل المعامل "offset" الإزاحة المطلقة التي يبدأ عندها "subbuf" داخل الصورة (لاحظ أن هذه ليست الإزاحة النسبية لـ "subbuf" داخل المخزن المؤقت الأصلي "buf"). يتم تجاهل التغييرات على "error" عند الإخراج ما لم تفشل الدالة الإجرائية. يتم التحكم في معنى الإدخال للمعامل "error" بواسطة المعامل "status"، وهو أحد الخيارات التالية:

"LIBNBD_READ_DATA" = 1
تم ملء "subbuf" بـ "count" بايت من البيانات. عند الإدخال، يحتوي "error" على قيمة errno لأي خطأ تم اكتشافه سابقًا، أو صفر.
"LIBNBD_READ_HOLE" = 2
يمثل "subbuf" فجوة، ويحتوي على "count" بايت NUL. عند الإدخال، يحتوي "error" على قيمة errno لأي خطأ تم اكتشافه سابقًا، أو صفر.
"LIBNBD_READ_ERROR" = 3
قيمة "count" هي 0، لذلك "subbuf" غير قابل للاستخدام. عند الإدخال، يحتوي "error" على قيمة errno التي أبلغ عنها الخادم أثناء قراءة "offset"، بغض النظر عما إذا تم اكتشاف أي خطأ سابق.

قد تسمح امتدادات NBD المستقبلية بقيم أخرى لـ "status"، ولكن لن يتم إرجاعها إلى عميل لم يختر طلب مثل هذه الامتدادات. إذا كان الخادم غير متوافق، فمن الممكن استدعاء الدالة "chunk" أكثر من مرة مما تتوقعه أو بقيمة "count" تساوي 0 لـ "LIBNBD_READ_DATA" أو "LIBNBD_READ_HOLE". من الممكن أيضًا ألا يتم استدعاء الدالة "chunk" على الإطلاق (على وجه الخصوص، يتم استخدام "LIBNBD_READ_ERROR" فقط عند ربط خطأ بإزاحة معينة، وليس عند إبلاغ الخادم عن خطأ عام)، ولكن مضمون لك أن الدالة الإجرائية قد تم استدعاؤها مرة واحدة على الأقل إذا نجحت القراءة الإجمالية. لا يتحقق libnbd مما إذا كان الخادم قد امتثل لمتطلب أن أمر القراءة يجب ألا يحتوي على أجزاء متداخلة ويجب ألا ينجح دون أجزاء كافية لتغطية الطلب بأكمله.

لاحظ أن libnbd يفرض حاليًا حدًا أقصى لمخزن القراءة يبلغ 64 ميبي، حتى لو سمح الخادم بمخزن أكبر في معاملة واحدة؛ ستؤدي محاولات تجاوز هذا الحد إلى خطأ "ERANGE". قد يفرض الخادم حدًا أصغر، والذي يمكن معرفته باستخدام nbd_get_block_size(3).

قد يحتوي المعامل "flags" على 0 لعدم وجود علامات، أو قد يحتوي على "LIBNBD_CMD_FLAG_DF" مما يعني أنه يجب ألا يرد الخادم بأكثر من جزء واحد (إذا كان ذلك مدعومًا - بعض الخوادم لا تستطيع فعل ذلك، انظر nbd_can_df(3)). لا يتحقق libnbd من أن الخادم يلتزم فعليًا بالعلامة.

لاحظ أنه إذا فشل هذا الأمر، وأرجعت nbd_get_pread_initialize(3) القيمة true، فقد قام libnbd بتطهير "buf"، ولكن لم يتم تحديد ما إذا كانت محتويات "buf" ستقرأ كصفر أو كنتائج جزئية من الخادم. إذا أعادت nbd_get_pread_initialize(3) القيمة false، فلم يقم libnbd بتطهير "buf"، والمحتويات غير محددة عند الفشل.

افتراضيًا، سيرفض libnbd المحاولات لاستخدام هذه الدالة بمعاملات من المحتمل أن تؤدي إلى فشل الخادم، مثل طلب علامة أمر غير معروفة. يمكن استخدام الدالة nbd_set_strict_mode(3) لتغيير السيناريوهات التي يجب أن تنتظر رد الخادم بدلاً من الفشل بسرعة.

القيمة المرجعة

في حالة نجاح الاستدعاء، تعيد الدالة القيمة 0.

الأخطاء

عند حدوث خطأ، يتم إرجاع القيمة -1.

راجع "معالجة الأخطاء" في libnbd(3) للحصول على مزيد من تفاصيل الخطأ.

يجب ألا تكون المعاملات التالية NULL: "h"، "buf". لمزيد من المعلومات، راجع "المعاملات غير الفارغة" في libnbd(3).
