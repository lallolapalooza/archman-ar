nbd_pread(3)

الحزمة: libnbd

التراخيص: LGPL-2.1-or-later

nbd_pread(3) LIBNBD nbd_pread(3)

الاسم

nbd_pread - القراءة من خادم NBD

الملخص

#include <libnbd.h>
int nbd_pread (
      struct nbd_handle *h, void *buf, size_t count,
      uint64_t offset, uint32_t flags
    );

الوصف

إصدار أمر قراءة إلى خادم NBD للنطاق الذي يبدأ من "offset" وينتهي عند "offset" + "count" - 1. يمكن لـ NBD فقط القراءة بالكامل أو لا شيء باستخدام هذا الاستدعاء. يعود الاستدعاء عندما تتم قراءة البيانات بالكامل في "buf" أو عند حدوث خطأ. انظر أيضاً nbd_pread_structured(3)، إذا كانت هناك حاجة لرؤية أدق في ردود الخادم، أو إذا كنت ترغب في استخدام "LIBNBD_CMD_FLAG_DF".

لاحظ أن libnbd تفرض حاليًا حدًا أقصى لمخزن القراءة يبلغ 64MiB، حتى لو سمح الخادم بمخزن أكبر في معاملة واحدة؛ المحاولات لتجاوز هذا ستؤدي إلى خطأ "ERANGE". قد يفرض الخادم حدًا أصغر، والذي يمكن معرفته باستخدام nbd_get_block_size(3).

يجب أن تكون معلمة "flags" 0 في الوقت الحالي (فهي موجودة لتوسيع بروتوكول NBD المستقبلي).

لاحظ أنه إذا فشل هذا الأمر، وإذا أعادت nbd_get_pread_initialize(3) قيمة صحيحة، فقد قامت libnbd بتنظيف "buf"، ولكن لا يُحدد ما إذا كانت محتويات "buf" ستقرأ كصفر أو كنتائج جزئية من الخادم. إذا أعادت nbd_get_pread_initialize(3) قيمة خاطئة، فقد لم تقم libnbd بتنظيف "buf"، وتكون المحتويات غير معرفة عند الفشل.

افتراضيًا، سترفض libnbd المحاولات لاستخدام هذه الدالة بمعلمات من المحتمل أن تؤدي إلى فشل الخادم، مثل طلب علامة أمر غير معروفة. يمكن استخدام الدالة nbd_set_strict_mode(3) لتغيير السيناريوهات التي يجب أن تنتظر رد الخادم بدلاً من الفشل بسرعة.

قيمة الإرجاع

إذا نجح الاستدعاء، تعيد الدالة 0.

الأخطاء

عند حدوث خطأ، يتم إرجاع -1.

راجع "معالجة الأخطاء" في libnbd(3) للحصول على مزيد من التفاصيل حول الخطأ.

يجب ألا تكون المعلمات التالية NULL: "h"، "buf". لمزيد من المعلومات، انظر "المعلمات غير NULL" في libnbd(3).

حالة المقبض

يمكن استدعاء nbd_pread عندما يكون المقبض في الحالة التالية:

┌─────────────────────────────────────┬─────────────────────────┐
│ تم إنشاء المقبض، قبل الاتصال   │ ❌ خطأ                │
│ جاري الاتصال                          │ ❌ خطأ                │
│ جاري الاتصال والمصافحة (opt_mode) │ ❌ خطأ                │
│ متصل بالخادم             │ ✅ مسموح به              │
│ تم إنهاء الاتصال                │ ❌ خطأ                │
│ المقبض معطل                         │ ❌ خطأ                │
└─────────────────────────────────────┴─────────────────────────┘

الإصدار

ظهرت هذه الدالة لأول مرة في libnbd 1.0.

إذا كنت بحاجة إلى اختبار ما إذا كانت هذه الدالة متاحة وقت الترجمة، فتحقق مما إذا كان الماكرو التالي معرفًا:

#define LIBNBD_HAVE_NBD_PREAD 1

مثال

هذا المثال متاح أيضًا باسم examples/fetch-first-sector.c في كود مصدر libnbd.

/* هذا المثال يوضح كيفية الاتصال بخادم NBD
 * وجلب وطباعة القطاع الأول (عادةً
 * قطاع الإقلاع أو جدول الأقسام أو المجلد الفائق لنظام الملفات).
 *
 * يمكنك اختباره باستخدام nbdkit بهذه الطريقة:
 *
 * nbdkit -U - floppy . \
 *   --run './fetch-first-sector $unixsocket'
 *
 * يحدد مكون floppy الخاص بـ nbdkit قرص MBR، لذا فإن القطاع الأول هو جدول الأقسام.
 */
#include <stdio.h>
#include <stdlib.h>
#include <libnbd.h>
int
main (int argc, char *argv[])
{
  struct nbd_handle *nbd;
  char buf[512];
  FILE *pp;
  if (argc != 2) {
    fprintf (stderr, "%s socket\n", argv[0]);
    exit (EXIT_FAILURE);
  }
  /* إنشاء مقبض libnbd. */
  nbd = nbd_create ();
  if (nbd == NULL) {
    fprintf (stderr, "%s\n", nbd_get_error ());
    exit (EXIT_FAILURE);
  }
  /* الاتصال بخادم NBD عبر
   * مقبس نطاق Unix.
   */
  if (nbd_connect_unix (nbd, argv[1]) == -1) {
    fprintf (stderr, "%s\n", nbd_get_error ());
    exit (EXIT_FAILURE);
  }
  /* قراءة القطاع الأول بشكل متزامن. */
  if (nbd_pread (nbd, buf, sizeof buf, 0, 0) == -1) {
    fprintf (stderr, "%s\n", nbd_get_error ());
    exit (EXIT_FAILURE);
  }
  /* إغلاق مقبض libnbd. */
  nbd_close (nbd);
  /* طباعة القطاع الأول. */
  pp = popen ("hexdump -C", "w");
  if (pp == NULL) {
    perror ("popen: hexdump");
    exit (EXIT_FAILURE);
  }
  fwrite (buf, sizeof buf, 1, pp);
  pclose (pp);
  exit (EXIT_SUCCESS);
}
