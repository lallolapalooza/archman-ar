nbd_opt_go(3)

الحزمة: libnbd

التراخيص: LGPL-2.1-or-later

nbd_opt_go(3) LIBNBD nbd_opt_go(3)

الاسم

nbd_opt_go - إنهاء التفاوض والمضي قدمًا في استخدام تصدير

الملخص

#include <libnbd.h>
int nbd_opt_go (
      struct nbd_handle *h
    );

الوصف

اطلب من الخادم إنهاء التفاوض والمضي قدمًا في تقديم التصدير الذي تم تحديده مسبقًا بواسطة أحدث استدعاء لـ nbd_set_export_name(3) أو nbd_connect_uri(3). لا يمكن استخدام هذا إلا إذا قامت nbd_set_opt_mode(3) بتمكين وضع الخيارات.

بشكل افتراضي، سيطلب libnbd تلقائيًا جميع سياقات البيانات الوصفية المسجلة بواسطة nbd_add_meta_context(3) كجزء من هذا الاستدعاء؛ ولكن يمكن قمع ذلك باستخدام nbd_set_request_meta_context(3)، خاصة إذا تم استخدام nbd_opt_set_meta_context(3) في وقت سابق في تسلسل التفاوض.

إذا فشل هذا، قد لا يزال الخادم في وضع التفاوض، حيث من الممكن محاولة خيار آخر مثل اسم تصدير مختلف؛ على الرغم من أن الخوادم الأقدم ستقوم بدلاً من ذلك بإنهاء الاتصال.

قيمة الإرجاع

إذا كان الاستدعاء ناجحًا، فإن الدالة ترجع 0.

الأخطاء

عند حدوث خطأ، يتم إرجاع -1.

راجع "معالجة الأخطاء" في libnbd(3) للحصول على تفاصيل إضافية حول الخطأ.

يجب ألا تكون المعلمات التالية خالية (NULL): "h". لمزيد من المعلومات، انظر "المعلمات غير الخالية" في libnbd(3).

حالة المقبض

يمكن استدعاء nbd_opt_go عندما يكون المقبض في الحالة التالية:

┌─────────────────────────────────────┬─────────────────────────┐
│ تم إنشاء المقبض، قبل الاتصال │ ❌ خطأ ❌                │
│ جارٍ الاتصال                         │ ❌ خطأ ❌                │
│ جارٍ الاتصال والمصافحة (opt_mode) │ ✅ مسموح به ✅              │
│ تم الاتصال بالخادم             │ ❌ خطأ ❌                │
│ تم إغلاق الاتصال                │ ❌ خطأ ❌                │
│ المقبض ميت                         │ ❌ خطأ ❌                │
└─────────────────────────────────────┴─────────────────────────┘

الإصدار

ظهرت هذه الدالة لأول مرة في libnbd 1.4.

إذا كنت بحاجة إلى اختبار ما إذا كانت هذه الدالة متاحة وقت الترجمة، فتحقق مما إذا كان هذا الماكرو معرفًا:

#define LIBNBD_HAVE_NBD_OPT_GO 1

مثال

هذا المثال متاح أيضًا باسم examples/list-exports.c في شفرة مصدر libnbd.

/* هذا المثال يوضح كيفية سرد تصديرات NBD.
 *
 * لاختبار هذا مع qemu-nbd:
 *   $ qemu-nbd -x "hello" -t -k /tmp/sock disk.img
 *   $ ./run examples/list-exports /tmp/sock
 *   [0] hello
 *   أي تصدير للاتصال به ( -1 للخروج)؟ 0
 *   جارٍ الاتصال بـ hello ...
 *   /tmp/sock: hello: الحجم = 2048 بايت
 *
 * لاختبار هذا مع nbdkit (يتطلب 1.22):
 *   $ nbdkit -U /tmp/sock sh - <<\EOF
 *   case $1 in
 *     list_exports) echo NAMES; echo foo; echo foobar ;;
 *     open) echo "$3" ;;
 *     get_size) echo "$2" | wc -c ;;
 *     pread) echo "$2" | dd bs=1 skip=$4 count=$3 ;;
 *     *) exit 2 ;;
 *   esac
 *   EOF
 *   $ ./run examples/list-exports /tmp/sock
 *   [0] foo
 *   [1] foobar
 *   أي تصدير للاتصال به ( -1 للخروج)؟ 1
 *   جارٍ الاتصال بـ foobar ...
 *   /tmp/sock: foobar: الحجم = 7 بايت
 */
#include <stdio.h>
#include <stdlib.h>
#include <stdbool.h>
#include <stdint.h>
#include <string.h>
#include <inttypes.h>
#include <errno.h>
#include <libnbd.h>
struct export_list {
  int i;
  char **names;
};
/* دالة رد الاتصال لـ nbd_opt_list */
static int
list_one (void *opaque, const char *name,
          const char *description)
{
  struct export_list *l = opaque;
  char **names;
  printf ("[%d] %s\n", l->i, name);
  if (*description)
    printf ("  (%s)\n", description);
  names = realloc (l->names,
                   (l->i + 1) * sizeof *names);
  if (!names) {
    perror ("realloc");
    exit (EXIT_FAILURE);
  }
  names[l->i] = strdup (name);
  if (!names[l->i]) {
    perror ("strdup");
    exit (EXIT_FAILURE);
  }
  l->names = names;
  l->i++;
  return 0;
}
int
main (int argc, char *argv[])
{
  struct nbd_handle *nbd;
  int i;
  const char *name;
  int64_t size;
  struct export_list list = { 0 };
  if (argc != 2) {
    fprintf (stderr, "%s socket\n", argv[0]);
    exit (EXIT_FAILURE);
  }
  /* إنشاء مقبض libnbd. */
  nbd = nbd_create ();
  if (nbd == NULL) {
    fprintf (stderr, "%s\n", nbd_get_error ());
    exit (EXIT_FAILURE);
  }
  /* تعيين وضع الخيارات. */
  nbd_set_opt_mode (nbd, true);
  /* الاتصال بخادم NBD عبر
   * مقبس مجال Unix. إذا لم
   * ندخل في وضع الخيارات، فلن يكون
   * سرد البيانات ممكنًا.
   */
  if (nbd_connect_unix (nbd, argv[1]) == -1) {
    fprintf (stderr, "%s\n", nbd_get_error ());
    exit (EXIT_FAILURE);
  }
  if (!nbd_aio_is_negotiating (nbd)) {
    fprintf (stderr, "الخادم لا يدعم "
             "سرد التصديرات.\n");
    exit (EXIT_FAILURE);
  }
  /* طباعة قائمة التصديرات. */
  if (nbd_opt_list (nbd,
                    (nbd_list_callback) {
                      .callback = list_one,
                      .user_data = &list, }) == -1) {
    fprintf (stderr, "%s\n", nbd_get_error ());
    exit (EXIT_FAILURE);
  }
  /* عرض قائمة التصديرات. */
  printf ("أي تصدير للاتصال به؟ ");
  if (scanf ("%d", &i) != 1) exit (EXIT_FAILURE);
  if (i == -1) {
    if (nbd_opt_abort (nbd) == -1) {
      fprintf (stderr, "%s\n", nbd_get_error ());
      exit (EXIT_FAILURE);
    }
    nbd_close (nbd);
    exit (EXIT_SUCCESS);
  }
  if (i < 0 || i >= list.i) {
    fprintf (stderr, "الفهرس %d خارج النطاق", i);
    exit (EXIT_FAILURE);
  }
  name = list.names[i];
  printf ("جارٍ الاتصال بـ %s ...\n", name);
  /* استئناف الاتصال بالتصدير المختار. */
  if (nbd_set_export_name (nbd, name) == -1 ||
      nbd_opt_go (nbd) == -1) {
    fprintf (stderr, "%s\n", nbd_get_error ());
    exit (EXIT_FAILURE);
  }
  if (!nbd_aio_is_ready (nbd)) {
    fprintf (stderr, "أغلق الخادم مبكرًا\n");
    exit (EXIT_FAILURE);
  }
  /* قراءة الحجم بالبايت وطباعته. */
  size = nbd_get_size (nbd);
  if (size == -1) {
    fprintf (stderr, "%s\n", nbd_get_error ());
    exit (EXIT_FAILURE);
  }
  printf ("%s: %s: الحجم = %" PRIi64 " بايت\n",
          argv[1], name, size);
  /* إغلاق مقبض libnbd. */
  nbd_close (nbd);
  for (i = 0; i < list.i; i++)
    free (list.names[i]);
  free (list.names);
  exit (EXIT_SUCCESS);
}
