nbd_set_tls(3)

الحزمة: libnbd

التراخيص: LGPL-2.1-or-later

nbd_set_tls(3) LIBNBD nbd_set_tls(3)

الاسم

nbd_set_tls - تمكين أو طلب TLS (مصادقة وتشفير)

ملخص

#include <libnbd.h>
int nbd_set_tls (
      struct nbd_handle *h, int tls
    );

الوصف

تمكين أو طلب TLS (اتصالات مصادق عليها ومشفرة) إلى خادم NBD. الإعدادات الممكنة هي:

"LIBNBD_TLS_DISABLE"
تعطيل TLS. (الإعداد الافتراضي، ما لم يتم استخدام nbd_connect_uri(3) مع URI يتطلب TLS).

هذا الإعداد ضروري أيضًا إذا استخدمت nbd_set_opt_mode(3) وتريد التفاعل بنص عادي مع خادم يطبق وضع "SELECTIVETLS" لبروتوكول NBD، قبل تمكين TLS باستخدام nbd_opt_starttls(3). معظم خوادم NBD التي تدعم TLS تفضل وضع "FORCEDTLS" الخاص ببروتوكول NBD، لذلك يميل هذا النوع من التفاعل اليدوي إلى أن يكون مفيدًا بشكل أساسي أثناء اختبار التكامل.

"LIBNBD_TLS_ALLOW"
تمكين TLS إذا كان ذلك ممكنًا.

هذا الخيار غير آمن (أو أفضل جهد) لأنه في بعض الحالات سيعود إلى اتصال غير مشفر و/أو غير مصادق عليه إذا لم يتمكن من إنشاء اتصال TLS. استخدم "LIBNBD_TLS_REQUIRE" أدناه إذا كان الاتصال يجب أن يكون مشفرًا.

بعض الخوادم ستقطع الاتصال إذا فشل TLS، لذلك قد لا يكون الرجوع ممكنًا.

"LIBNBD_TLS_REQUIRE"
طلب اتصال TLS مشفر ومصادق عليه. فشل دائمًا في الاتصال إذا لم يكن الاتصال مشفرًا ومصادقًا عليه.

بالإضافة إلى استدعاء هذه الوظيفة، قد تحتاج أيضًا إلى توفير المسار إلى دليل الشهادات (nbd_set_tls_certificates(3))، واسم المستخدم (nbd_set_tls_username(3)) و/أو ملف المفاتيح المشتركة مسبقًا (PSK) (nbd_set_tls_psk_file(3)). في الوقت الحالي، عند استخدام nbd_connect_uri(3)، لا يتم التعامل مع أي معلمات استعلام URI متعلقة بـ TLS تلقائيًا. سيؤدي تعيين المستوى أعلى من الصفر إلى فشل إذا لم يتم تجميع libnbd مقابل gnutls؛ يمكنك اختبار ما إذا كان هذا هو الحال باستخدام nbd_supports_tls(3).

قيمة الإرجاع

إذا كان الاستدعاء ناجحًا، تعيد الوظيفة 0.

الأخطاء

عند حدوث خطأ، يتم إرجاع -1.

ارجع إلى "معالجة الأخطاء" في libnbd(3) للحصول على مزيد من التفاصيل حول الخطأ.

يجب ألا تكون المعلمات التالية NULL: "h". لمزيد من المعلومات، راجع "المعلمات غير الصفرية" في libnbd(3).

حالة المقبض

يمكن استدعاء nbd_set_tls عندما يكون المقبض في الحالة التالية:

┌─────────────────────────────────────┬─────────────────────────┐
│ تم إنشاء المقبض، قبل الاتصال   │ ✅ مسموح به              │
│ جاري الاتصال                          │ ❌ خطأ                │
│ جاري الاتصال & المصافحة (opt_mode) │ ❌ خطأ                │
│ متصل بالخادم             │ ❌ خطأ                │
│ تم إيقاف الاتصال                │ ❌ خطأ                │
│ المقبض معطل                         │ ❌ خطأ                │
└─────────────────────────────────────┴─────────────────────────┘

الإصدار

ظهرت هذه الوظيفة لأول مرة في libnbd 1.0.

إذا كنت بحاجة إلى اختبار ما إذا كانت هذه الوظيفة متاحة في وقت الترجمة تحقق مما إذا كان ماكرو المعرّف التالي معرفًا:

#define LIBNBD_HAVE_NBD_SET_TLS 1

مثال

هذا المثال متاح أيضًا باسم examples/encryption.c في شفرة المصدر لـ libnbd.

/* مثال يوضح كيفية الاتصال بخادم يستخدم
 * تشفير TLS.
 *
 * هذا يتطلب nbdkit، و psktool من gnutls.
 *
 * يدعم كل من libnbd و nbdkit تشفير TLS-PSK، وهو
 * شكل أسهل للنشر للتشفير. (بالطبع
 * يتم دعم التشفير المستند إلى الشهادات أيضًا، ولكنه
 * أصعب في إنشاء مثال مكتفٍ ذاتيًا).
 */
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <unistd.h>
#include <libnbd.h>
#define TMPDIR "/tmp/XXXXXX"
#define KEYS "keys.psk"
#define USERNAME "alice"
static char dir[] = TMPDIR;
static char keys[] = TMPDIR "/" KEYS;
static char cmd[] =
  "psktool -u " USERNAME " -p " TMPDIR "/" KEYS;
/* إزالة ملف المفاتيح المؤقت عند خروج البرنامج
 * .
 */
static void
cleanup_keys (void)
{
  unlink (keys);
  rmdir (dir);
}
/* إنشاء ملف المفاتيح المؤقت لمشاركته مع
 * الخادم.
 */
static void
create_keys (void)
{
  size_t i;
  if (mkdtemp (dir) == NULL) {
    perror ("mkdtemp");
    exit (EXIT_FAILURE);
  }
  i = strlen (cmd) - strlen (TMPDIR) - strlen (KEYS) - 1;
  memcpy (&cmd[i], dir, strlen (TMPDIR));
  memcpy (keys, dir, strlen (TMPDIR));
  if (system (cmd) != 0) {
    fprintf (stderr, "فشل أمر psktool\n");
    exit (EXIT_FAILURE);
  }
  atexit (cleanup_keys);
}
int
main (int argc, char *argv[])
{
  struct nbd_handle *nbd;
  char buf[512];
  create_keys ();
  /* إنشاء مقبض libnbd. */
  nbd = nbd_create ();
  if (nbd == NULL) {
    fprintf (stderr, "%s\n", nbd_get_error ());
    exit (EXIT_FAILURE);
  }
  /* تمكين TLS في العميل. */
  if (nbd_set_tls (nbd, LIBNBD_TLS_REQUIRE) == -1) {
    fprintf (stderr, "%s\n", nbd_get_error ());
    exit (EXIT_FAILURE);
  }
  /* تمكين TLS-PSK وتمرير اسم ملف المفاتيح. */
  if (nbd_set_tls_psk_file (nbd, keys) == -1) {
    fprintf (stderr, "%s\n", nbd_get_error ());
    exit (EXIT_FAILURE);
  }
  /* تعيين اسم المستخدم المحلي للمصادقة. */
  if (nbd_set_tls_username (nbd, USERNAME) == -1) {
    fprintf (stderr, "%s\n", nbd_get_error ());
    exit (EXIT_FAILURE);
  }
  /* تشغيل nbdkit كعملية فرعية، وتمكين وتطلب
   * تشفير TLS-PSK.
   */
  char *args[] = {
    "nbdkit", "-s", "--exit-with-parent",
    "--tls", "require", "--tls-psk", keys,
    "pattern", "size=1M", NULL
  };
  if (nbd_connect_command (nbd, args) == -1) {
    fprintf (stderr, "%s\n", nbd_get_error ());
    exit (EXIT_FAILURE);
  }
  /* قراءة القطاع الأول. */
  if (nbd_pread (nbd, buf, sizeof buf, 0, 0) == -1) {
    fprintf (stderr, "%s\n", nbd_get_error ());
    exit (EXIT_FAILURE);
  }
  /* يجب إيقاف اتصالات TLS. */
  if (nbd_shutdown (nbd, 0) == -1) {
    fprintf (stderr, "%s\n", nbd_get_error ());
    exit (EXIT_FAILURE);
  }
  /* إغلاق مقبض libnbd. */
  nbd_close (nbd);
  exit (EXIT_SUCCESS);
}
