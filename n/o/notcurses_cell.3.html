الاسم

notcurses_cell - عمليات على كائنات nccell

الملخص

#include <notcurses/notcurses.h>

// انظر الوصف أدناه للحصول على معلومات حول ترميز EGC
typedef struct nccell {
  uint32_t gcluster;          // 4 بايت ← 4 بايت
  uint8_t gcluster_backstop;  // 1 بايت ← 5 بايت (8 بتات صفر)
  uint8_t width;              // 1 بايت ← 6 بايت (8 بتات عرض العمود)
  uint16_t stylemask;         // 2 بايت ← 8 بايت (16 بت سمات NCSTYLE_*)
  uint64_t channels;
} nccell;
#define NCCELL_TRIVIAL_INITIALIZER \
 { .gcluster = '\0', .stylemask = 0, .channels = 0, }
#define NCCELL_CHAR_INITIALIZER(c) \
 { .gcluster = (c), .stylemask = 0, .channels = 0, }
#define NCCELL_INITIALIZER(c, s, chan) \
 { .gcluster = (c), .stylemask = (s), .channels = (chan), }
#define NCALPHA_HIGHCONTRAST  0x30000000ull
#define NCALPHA_TRANSPARENT   0x20000000ull
#define NCALPHA_BLEND         0x10000000ull

الوصف

الخلايا تشكل إطار العرض المرتبط بكل مستوى، مع خلية واحدة لكل إحداثي يمكن الوصول إليه. لا تحتاج عادةً إلى التفاعل مباشرة مع nccells.

كل nccell يحتوي على مجموعة رسومية ممتدة واحدة بالضبط. إذا تم ترميز EGC في UTF-8 بأربعة بايتات أو أقل (جميع نقاط ترميز Unicode اعتبارًا من Unicode 13 يمكن ترميزها في أربعة بايتات UTF-8 أو أقل)، يتم ترميزها مباشرة في حقل gcluster الخاص بـ nccell، ولا يلزم تخزين إضافي. بخلاف ذلك، يتم تخزين EGC كسلسلة UTF-8 منتهية بصفر في بعض مخازن egcpool الخلفية. ترتبط egcpools بـ ncplanes، لذلك يجب اعتبار nccells مرتبطة بـ ncplanes. في الواقع، ncplane_erase يدمر التخزين الخلفي لجميع خلايا المستوى، مما يجعلها غير صالحة. يتم تكوين هذه العلاقة وقت nccell_load أو nccell_prime أو nccell_duplicate. تستدعي جميع هذه الوظائف أولاً nccell_release، وكذلك nccell_load_egc32 و nccell_load_char و nccell_load_ucs32. عند الانتهاء من استخدام nccell بالكامل، استدعِ nccell_release. سيقوم ncplane_destroy بتحرير الذاكرة المستخدمة بواسطة nccell، ولكن الحد الأقصى لحجم egcpool الخلفي هو 16MiB، وقد يؤدي عدم تحرير nccells إلى حظر الإخراج الجديد في النهاية. الكتابة فوق خلايا ncplane تحررها تلقائيًا.

nccell_extended_gcluster يوفر مقبضًا منتهيًا بصفر لـ EGC. يجب اعتباره غير صالح بسبب التغييرات في nccell أو egcpool. المقبض غير مخصص ديناميكيًا؛ لا تحاول free(3). يمكن الحصول على نسخة مخصصة ديناميكيًا باستخدام nccell_strdup.

ncstrwidth_valid يعيد عدد الأعمدة التي تشغلها سلسلة UTF-8 صالحة، أو -1 في حالة حدوث خطأ. في كلتا الحالتين، يتم استهلاك عدد البايتات والأعمدة الصالحة، على التوالي، قبل الخطأ في validbytes و validwidth (بافتراض أنها ليست NULL).

nccell_fg_palindex يستخرج فهرس اللوحة من قناة مقدمة الخلية. يجب أن تكون القناة لوحة

قيم الإرجاع

nccell_load والدوال المشابهة ترجع عدد البايتات التي تم تحميلها من EGC، أو -1 عند الفشل. يمكن أن تفشل هذه الدوال بسبب مدخلات UTF-8 غير صالحة، أو وصول egcpool الأساسي إلى حده الأقصى.

nccell_set_fg_rgb8 والدوال المشابهة سترجع -1 إذا تم تقديم مدخلات غير صالحة، و 0 بخلاف ذلك.

nccellcmp يرجع عددًا صحيحًا سالبًا، أو 0، أو عددًا صحيحًا موجبًا إذا كان c1 أقل من، أو يساوي، أو أكثر من c2، على التوالي.

nccell_extended_gcluster يرجع NULL إذا تم استدعاؤه على sprixel (انظر notcurses_visual(3).

nccell_cols يرجع عدد الأعمدة التي يشغلها c، وفقًا لـ wcwidth(3)*. ncstrwidth هو مكافئ للسلاسل النصية. ncstrwidth_valid يرجع نفس القيمة مثل ncstrwidth.

ملاحظات

تمت إعادة تسمية cell إلى nccell في Notcurses 2.2.0، حتى لا تتسرب هذه المصطلحات الشائعة إلى مساحة الأسماء (الوحيدة والعالمية) في C. سيتم الاحتفاظ بـ cell كاسم بديل حتى Notcurses 3.0. وبالمثل، تمت إعادة تسمية العديد من الدوال التي تبدأ بـ cell لتبدأ بـ nccell.
