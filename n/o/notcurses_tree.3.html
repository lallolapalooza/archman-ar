notcurses_tree(3)

الحزمة: notcurses

التراخيص: Apache

notcurses_tree(3) دليل الوظائف المكتبية notcurses_tree(3)

الاسم

notcurses_tree - بيانات عالية المستوى تعتمد على خطوط هرمية

الملخص

#include <notcurses/notcurses.h>

struct nctree;
struct ncplane;
typedef struct nctree_item {
  void* curry;
  struct nctree_item* subs;
  unsigned subcount;
} nctree_item;
typedef struct nctree_options {
  const nctree_item* items; // مصفوفة nctree_item ذات المستوى الأعلى
  unsigned count;           // حجم |items|
  int (*nctreecb)(struct ncplane*, void*, int); // دالة رد اتصال العنصر
  int indentcols;           // أعمدة للإزاحة لكل تسلسل هرمي
  uint64_t flags;           // حقل بت لـ NCTREE_OPTION_*
} nctree_options;

struct nctree* nctree_create(struct ncplane* n, const nctree_options* opts);

struct ncplane* nctree_plane(struct nctree* n);

int nctree_redraw(struct nctree* n);

bool nctree_offer_input(struct nctree* n, const ncinput* ni);

void* nctree_focused(struct nctree* n);

void* nctree_next(struct nctree* n);

void* nctree_prev(struct nctree* n);

void* nctree_goto(struct nctree* n, const int* path, int* failpath);

int nctree_add(struct nctree* n, const unsigned* path, const nctree_item* add);

int nctree_del(struct nctree* n, const unsigned* path);

void nctree_destroy(struct nctree* n);

الوصف

nctrees تنظم العناصر الهرمية، وتسمح بتصفحها. يمكن لكل عنصر أن يكون له عناصر فرعية اعتباطية. يمكن طي وتوسيع العناصر. يدعم العرض التمرير والبحث.

لا يمكن أن يكون nctree فارغًا. يجب أن يكون count غير صفر، ويجب ألا يكون items يساوي NULL. يجب أيضًا أن تكون دالة رد الاتصال nctreecb غير NULL.

تُستدعى دالة رد الاتصال nctreecb على عناصر الشجرة عند إعادة رسم الشجرة. سيتم استدعاؤها على كل عنصر مرئي، وأي عنصر أصبح مخفيًا. إذا كان العنصر مخفيًا حديثًا، فسيكون الوسيط ncplane هو NULL. إذا كان العنصر مرئيًا حديثًا، فسيكون الوسيط ncplane مستويًا فارغًا. إذا كان العنصر مرئيًا بالفعل، فسيكون الوسيط ncplane هو نفس المستوي الذي تم تمريره سابقًا. إذا لم يتغير العنصر منذ آخر مرة تم فيها استدعاء رد الاتصال، فلا داعي لتغيير المستوي، ويمكن لدالة رد الاتصال أن تعود فورًا. بخلاف ذلك، يجب رسم المستوي بواسطة رد الاتصال. يجب تقليم الصفوف غير المستخدمة باستخدام ncplane_resize. إذا تم توسيع المستوي في رد الاتصال، فسيعود ليتقلص بواسطة الأداة. يشير المعامل int إلى المسافة من العنصر المركز. إذا كان المعامل سالبًا، فإن العنصر يسبق العنصر المركز؛ يشير المعامل الموجب إلى أن العنصر يتبع العنصر المركز؛ العنصر المركز نفسه يتم تمرير صفر له.

nctree_goto و nctree_add و nctree_del كلها تستخدم مفهوم path. المسار هو مصفوفة من unsigned القيم، تنتهي بـ UINT_MAX، حيث يفهرس كل قيمة متتالية في التسلسل الهرمي حتى تلك اللحظة. وبالتالي، فإن العقدة الجذرية لـ nctree لها دائمًا مسار من عنصرين: [0، UINT_MAX].

قيم الإرجاع

nctree_create ترجع NULL للخيارات غير الصالحة. يشمل ذلك حقل items أو nctreecb يساوي NULL، أو حقل count يساوي صفر.

nctree_next و nctree_prev ترجعان مؤشر curry من العنصر الذي تم التركيز عليه حديثًا. nctree_focused ترجع مؤشر curry من العنصر الذي تم التركيز عليه بالفعل.

nctree_goto ترجع مؤشر curry من العنصر الذي تم التركيز عليه حديثًا. إذا كان path غير صالح، يتم كتابة مستوى التسلسل الهرمي غير الصالح الأول في failpath، ويتم إرجاع NULL. إذا كان path صالحًا، يتم كتابة القيمة -1 في failpath. نظرًا لأنه من الممكن أن يكون مؤشر curry يساوي NULL، يجب عليك التحقق من failpath قبل افتراض فشل العملية.

nctree_add و nctree_del ترجعان -1 لمسار path غير صالح، و 0 خلاف ذلك.

ملاحظات

nctree تشترك في العديد من الخصائص مع notcurses_reel. على عكس الأخير، تدعم nctree مستويات هرمية اعتباطية.

كان nctree يتعامل مع البيانات الثابتة فقط في السابق، ولكن تمت إضافة nctree_add و nctree_del في Notcurses 3.0.1.
