NAME

notcurses_stats - إحصائيات Notcurses

SYNOPSIS

#include <notcurses/notcurses.h>

struct ncstats* notcurses_stats_alloc(const struct notcurses* nc);

void notcurses_stats_free(struct ncstats* stats);

int notcurses_stats_nbytes(const struct ncstats* stats);

int notcurses_stats_ncycles(const struct ncstats* stats);

int notcurses_stats_nbyteswritten(const struct ncstats* stats);

int notcurses_stats_nbytesread(const struct ncstats* stats);

int notcurses_stats_nopen(const struct ncstats* stats);

int notcurses_stats_nbytesallocated(const struct ncstats* stats);

int notcurses_stats_nbytesevicted(const struct ncstats* stats);

int notcurses_stats_nfilesopened(const struct ncstats* stats);

int notcurses_stats_nfilesclosed(const struct ncstats* stats);

int notcurses_stats_ninputevents(const struct ncstats* stats);

int notcurses_stats_nrenderloops(const struct ncstats* stats);

int notcurses_stats_nbytesfreed(const struct ncstats* stats);

int notcurses_stats_nbytesrendered(const struct ncstats* stats);

DESCRIPTION

يتم استخدام notcurses_stats_alloc لتخصيص بنية ncstats. تقوم هذه البنية بتخزين إحصائيات التشغيل الداخلية لـ Notcurses. يتم تمرير مؤشر إلى struct notcurses النشط. يجب تحرير الذاكرة المخصصة باستخدام notcurses_stats_free.

وظائف notcurses_stats_* ترجع قيمًا مفردة من بنية ncstats.

  • notcurses_stats_nbytes: عدد البايتات في ذاكرة التخزين المؤقت النهائية.
  • notcurses_stats_ncycles: عدد دورات العرض.
  • notcurses_stats_nbyteswritten: عدد البايتات التي تم كتابتها إلى الواجهة الخلفية.
  • notcurses_stats_nbytesread: عدد البايتات التي تم قراءتها من الواجهة الخلفية.
  • notcurses_stats_nopen: عدد العمليات التي لا تزال مفتوحة.
  • notcurses_stats_nbytesallocated: عدد البايتات المخصصة بواسطة Notcurses.
  • notcurses_stats_nbytesevicted: عدد البايتات التي تم إخلاءها من ذاكرة التخزين المؤقت.
  • notcurses_stats_nfilesopened: عدد الملفات التي تم فتحها.
  • notcurses_stats_nfilesclosed: عدد الملفات التي تم إغلاقها.
  • notcurses_stats_ninputevents: عدد أحداث الإدخال التي تمت معالجتها.
  • notcurses_stats_nrenderloops: عدد حلقات العرض.
  • notcurses_stats_nbytesfreed: عدد البايتات التي تم تحريرها بواسطة Notcurses.
  • notcurses_stats_nbytesrendered: عدد البايتات التي تم عرضها.

NOTES

هذه الوظائف مفيدة بشكل أساسي لتصحيح الأخطاء وفهم سلوك Notcurses.

RETURN VALUES

تُرجع وظائف notcurses_stats_* القيم العددية للإحصائيات المقابلة. إذا كان nc الذي تم تمريره إلى notcurses_stats_alloc هو NULL، فستُرجع NULL.
