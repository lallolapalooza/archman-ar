الاسم

notcurses_init - تهيئة مثيل Notcurses

الملخص

#include <notcurses/notcurses.h>

#define NCOPTION_INHIBIT_SETLOCALE   0x0001ull
#define NCOPTION_NO_CLEAR_BITMAPS    0x0002ull
#define NCOPTION_NO_WINCH_SIGHANDLER 0x0004ull
#define NCOPTION_NO_QUIT_SIGHANDLERS 0x0008ull
#define NCOPTION_PRESERVE_CURSOR     0x0010ull
#define NCOPTION_SUPPRESS_BANNERS    0x0020ull
#define NCOPTION_NO_ALTERNATE_SCREEN 0x0040ull
#define NCOPTION_NO_FONT_CHANGES     0x0080ull
#define NCOPTION_DRAIN_INPUT         0x0100ull
#define NCOPTION_SCROLLING           0x0200ull
#define NCOPTION_CLI_MODE (NCOPTION_NO_ALTERNATE_SCREEN \
                           |NCOPTION_NO_CLEAR_BITMAPS \
                           |NCOPTION_PRESERVE_CURSOR \
                           |NCOPTION_SCROLLING)
typedef enum {
  NCLOGLEVEL_SILENT,  // لا تطبع أي شيء بمجرد بدء خدمة وضع ملء الشاشة
  NCLOGLEVEL_PANIC,   // الافتراضي. اطبع التشخيصات قبل أن نتعطل/نخرج
  NCLOGLEVEL_FATAL,   // سنستمر، لكننا واجهنا خطأ فادحًا
  NCLOGLEVEL_ERROR,   // لا يمكننا الاستمرار في هذا، ولكن يمكننا القيام بأشياء أخرى
  NCLOGLEVEL_WARNING, // ربما لا تريد ما يحدث
  NCLOGLEVEL_INFO,    // "معلومات قياسية"
  NCLOGLEVEL_VERBOSE, // "معلومات مفصلة"
  NCLOGLEVEL_DEBUG,   // هذا كثير جدًا بصراحة
  NCLOGLEVEL_TRACE,   // ربما هناك طريقة أفضل للقيام بما تريد
} ncloglevel_e;
typedef struct notcurses_options {
  const char* termtype;
  ncloglevel_e loglevel;
  unsigned margin_t, margin_r, margin_b, margin_l;
  uint64_t flags; // من بتات NCOPTION_*

الوصف

notcurses_init يجهز الطرفية لوضع قابل للعنونة بالمؤشر (متعدد الأسطر)، ويكتب إلى ملف FILE ما. إذا كان الملف المقدم كـ fp هو NULL، فسيتم استخدام stdout. يجب أن يكون أي ملف FILE مستخدم قابل للكتابة، ويفضل أن يكون مخزنًا مؤقتًا بالكامل، ويجب أن يكون موجهًا بالبايت (انظر fwide(3)). إذا لم يكن الملف FILE متصلاً بطرفية (على سبيل المثال عند إعادة توجيهه إلى ملف)، فسيتم فتح /dev/tty (إذا أمكن) للتواصل مع الطرفية المتحكمة. تتم كتابة معظم المخرجات (بما في ذلك جميع الأنماط والألوان) إلى الملف FILE؛ فقط الاستعلامات تحتاج إلى إرسالها إلى طرفية حقيقية. إذا لم تكن هناك طرفية متاحة (على سبيل المثال عند التشغيل كخدمة)، فسيتم افتراض قيم افتراضية فيما يتعلق بأشياء مثل حجم الشاشة ولوحة الألوان.

التركيب struct notcurses_option الذي يتم تمريره كـ opts يتحكم في السلوك. تمرير NULL opts يكافئ تمرير opts مكون من أصفار (افتراضي). يمكن أن يكون لدى العملية سياق Notcurses واحد فقط نشط في كل مرة؛ سيؤدي استدعاء notcurses_init مرة أخرى قبل استدعاء notcurses_stop إلى إرجاع NULL (هذا موثوق حتى لو تم استدعاؤه بشكل متزامن من خيطين).

عند النجاح، يتم إرجاع مؤشر إلى struct notcurses صالح. يتم إرجاع NULL عند الفشل. قبل خروج العملية، يجب استدعاء notcurses_stop(3) لإعادة تعيين الطرفية وتحرير الموارد.

يجب أن يكون هناك إدخال terminfo(5) مناسب للطرفية. يتم تحديد هذا الإدخال عادةً باستخدام قيمة متغير البيئة TERM (انظر getenv(3))، ولكن قيمة غير NULL لـ termtype ستبطل هذا (لا يتم استخدام terminfo على أنظمة Microsoft Windows، حيث لا يكون ذا معنى ولا ضروريًا لتحديد TERM). يمكن أن يؤدي مواصفات terminfo غير صالحة إلى تقليل الأداء، وتقليل إمكانيات العرض، و/أو أخطاء العرض. تستهدف Notcurses أصلاً الألوان RGB بدقة 24 بت/8 بت لكل مكون، ولذلك من المستحسن استخدام طرفية مع القدرة rgb (مثل xterm-direct الخاصة بـ xterm). سيتم في خلاف ذلك تقليل الألوان إلى ما تدعمه الطرفية.

إذا أعلنت الطرفية عن دعمها لـ "شاشة بديلة" عبر قدرة terminfo smcup، فستستخدمها Notcurses افتراضيًا. يمكن منع ذلك عن طريق تعيين NCOPTION_NO_ALTERNATE_SCREEN في flags. يميل المستخدمون إلى أن يكون لديهم آراء قوية بشأن الشاشة البديلة، لذا غالبًا ما يكون من المفيد كشف ذلك عبر خيار سطر الأوامر. عند عدم استخدام الشاشة البديلة، تظل محتويات الطرفية عند بدء التشغيل مرئية حتى يتم محوها، خلية بخلية (انظر notcurses_plane(3) لمزيد من التفاصيل حول مسح الشاشة عند بدء التشغيل دون استخدام الشاشة البديلة). إذا لم تكن الشاشة البديلة متاحة، فسيتم مسح الشاشة حتى بدون NCOPTION_NO_ALTERNATE_SCREEN.

تخفي Notcurses المؤشر افتراضيًا. يمكن تمكينه ديناميكيًا، أو تحريكه، أو تعطيله أثناء التنفيذ عبر notcurses_cursor_enable و notcurses_cursor_disable. سيتم إخفاؤه أثناء تحديث الشاشة. يمكن الحصول على الموقع الحالي لمؤشر الطرفية باستخدام notcurses_cursor_yx، سواء كان مرئيًا أم لا.

notcurses_init يصدر عادةً بعض التشخيصات عند بدء التشغيل، بما في ذلك معلومات الإصدار وبعض تفاصيل الطرفية المهيأة. يمكن تثبيط ذلك باستخدام NCOPTION_SUPPRESS_BANNERS. سيؤدي هذا أيضًا إلى تثبيط ملخص الأداء الذي يطبعه عادةً notcurses_stop(3).

يمكن لـ Notcurses التصيير إلى منطقة فرعية من الطرفية عن طريق تحديد الهوامش المطلوبة على الجوانب الأربعة. افتراضيًا، جميع الهوامش صفر، وبالتالي سيتم إجراء التصيير على كامل منطقة العرض. هذا متعامد مع استخدام الشاشة البديلة؛ استخدام الشاشة البديلة مع الهوامش سيؤدي إلى مسح الشاشة بالكامل، متبوعًا بالتصيير إلى منطقة فرعية. تثبيط الشاشة البديلة مع الهوامش سيؤدي إلى التصيير إلى منطقة فرعية، مع عدم مسح المنطقة خارج هذه المنطقة. الهوامش هي أفضل مجهود. توفر notcurses_lex_margins تحليل تعبير وسيط للهامش في أحد الشكلين التاليين:

•
رقم واحد، والذي سيتم تطبيقه على جميع الجوانب، أو
•
أربعة أرقام مفصولة بفواصل، مطبقة على الأعلى، اليمين، الأسفل، واليسار.

للسماح بخيارات مستقبلية دون الحاجة إلى إعادة تعريف التركيب، حقل flags هو فقط حقل بت محدد جزئيًا. يجب أن تكون البتات غير المحددة صفرًا. الأعلام التالية محددة:

•
NCOPTION_INHIBIT_SETLOCALE: ما لم يتم تعيين هذا العلم، ستستدعي notcurses_init setlocale(LC_ALL, NULL). إذا كانت النتيجة إما "C" أو "POSIX"، فستقوم بطباعة تشخيص إلى stderr، ثم تستدعي setlocale(LC_ALL, ""). سيؤدي هذا إلى محاولة تعيين الإعدادات المحلية بناءً على متغير البيئة LANG. يجب على برنامجك استدعاء setlocale(3) بنفسه، عادةً كأحد الأسطر الأولى.
•
NCOPTION_NO_CLEAR_BITMAPS: عند الدخول، لا تبذل أي محاولة خاصة لمسح أي خرائط بت موجودة مسبقًا. لاحظ أنه قد يتم مسحها حتى إذا تم تعيين هذا، وقد لا يتم مسحها حتى إذا لم يتم تعيين هذا.
•
NCOPTION_NO_WINCH_SIGHANDLER: سيتم عادةً تثبيت معالج إشارة لـ SIGWINCH و SIGCONT، مما يؤدي إلى إنشاء أحداث NCKEY_RESIZE عند الإدخال. مع هذا العلم، لن يتم تثبيت المعالج.
•
NCOPTION_NO_QUIT_SIGHANDLERS: سيتم عادةً تثبيت معالج إشارة لـ SIGABRT، SIGBUS، SIGFPE، SIGILL، SIGINT، SIGQUIT
