notcurses_palette(3)

الحزمة: notcurses

التراخيص: Apache

notcurses_palette(3) دليل الوظائف المكتبية notcurses_palette(3)

الاسم

notcurses_palette - عمليات على لوحات ألوان notcurses

الملخص

#include <notcurses/notcurses.h>

typedef struct ncpalette {
  // نقوم بتخزين قيم RGB كقناة عادية
  uint32_t chans[NCPALETTESIZE];
} ncpalette;

bool notcurses_cantruecolor(const struct notcurses* nc);

ncpalette* ncpalette_new(struct notcurses* nc);

int ncpalette_use(struct notcurses* nc, const ncpalette* p);

int ncpalette_set_rgb8(ncpalette* p, int idx, int r, int g, int b);

int ncpalette_set(ncpalette* p, int idx, unsigned rgb);

int ncpalette_get(const ncpalette* p, int idx, uint32_t* palent);

int ncpalette_get_rgb8(const ncpalette* p, int idx, int* restrict r, int* restrict g, int* restrict b);

void ncpalette_free(ncpalette* p);

bool notcurses_canchangecolors(const struct notcurses* nc);

الوصف

تدعم بعض المحطات الطرفية 256 لونًا فقط، لكنها تسمح بتحديد اللوحة الكاملة بألوان RGB اعتباطية. في جميع الحالات، يكون استخدام الألوان المفهرسة أكثر كفاءة، حيث إنها بيانات أقل بكثير للكتابة إلى الطرفية. إذا كان بإمكانك قصر نفسك على 256 لونًا، فمن المحتمل أن يكون ذلك أفضل.

بالإضافة إلى ذلك، يسمح التلوين المستند إلى اللوحة بتأثيرات دورة ألوان سريعة جدًا، حيث يمكن لأمر واحد أن يؤثر على العديد من الخلايا على الشاشة.

قيم الإرجاع

الوظائف التي تُرجع int تُرجع -1 عند الفشل، أو 0 عند النجاح. الفشل يكون دائمًا بسبب مدخلات غير صالحة. الوظائف التي تُرجع bool هي عبارات شرطية، وتُرجع القيمة المطلوبة. الوظائف التي تُرجع unsigned تُرجع المدخل، بعد التعديل حسب المطلوب.
