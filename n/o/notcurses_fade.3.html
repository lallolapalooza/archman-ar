notcurses_fade(3)

الحزمة: notcurses

التراخيص: Apache

notcurses_fade(3) دليل وظائف المكتبة notcurses_fade(3)

الاسم

notcurses_fade - تلاشي/ظهور ncplanes

ملخص

#include <notcurses/notcurses.h>

struct ncfadectx;
// يتم استدعاؤها لكل دلتا تم إجراؤها في تلاشي على ncp. إذا تم إرجاع أي شيء بخلاف 0،
// تتوقف عملية التلاشي فورًا، ويتم تمرير هذه القيمة. إذا تم توفيرها ولم تكن NULL،
// فلن تقوم وظائف التلاشي نفسها باستدعاء notcurses_render().
typedef int (*fadecb)(struct notcurses* nc, struct ncplane* ncp,
                      const struct timespec*, void* curry);

bool notcurses_canfade(const struct notcurses* nc);

int ncplane_fadeout(struct ncplane* n, const struct timespec* ts, fadecb fader, void* curry);

int ncplane_fadein(struct ncplane* n, const struct timespec* ts, fadecb fader, void* curry);

int ncplane_pulse(struct ncplane* n, const struct timespec* ts, fadecb fader, void* curry);

struct ncfadectx* ncfadectx_setup(struct ncplane* n);

int ncfadectx_iterations(const struct ncfadectx* nctx);

int ncplane_fadeout_iteration(struct ncplane* n, struct ncfadectx* nctx, int iter, fadecb fader, void* curry);

int ncplane_fadein_iteration(struct ncplane* n, struct ncfadectx* nctx, int iter, fadecb fader, void* curry);

void ncfadectx_free(struct ncfadectx* nctx);

وصف

ncplane_fadeout و ncplane_fadein و ncplane_pulse هي واجهات برمجية بسيطة لتلاشي/ظهور المستويات. تتطلب عمليات التلاشي دعم RGB أو دعم إعادة برمجة لوحة الألوان من الطرفية (تُفضل طريقة RGB، وستُستخدم كلما أمكن). تحدد المعلمة ts إجمالي الوقت اللازم لعملية التلاشي. العملية نفسها قابلة للتكيف مع الوقت (أي إذا وجدت نفسها متأخرة، فستتخطى التكرارات؛ وإذا كانت تتقدم بسرعة كبيرة، فستنام).

هذه التفافات لواجهة ncfadectx الأكثر مرونة. قم بإنشاء ncfadectx باستخدام ncfadectx_setup. يمكن الوصول إلى عدد التغييرات الممكنة في الحالة (التكرارات) باستخدام ncfadectx_iterations. يمكن الوصول إلى حالة باستخدام ncplane_fadeout_iteration أو ncplane_fadein_iteration. أخيرًا، قم بتدمير ncfadectx باستخدام ncfadectx_free.

قيم الإرجاع

ncplane_fadeout_iteration و ncplane_fadein_iteration ستعيد أي قيمة غير صفرية من الاستدعاء fader.

أخطاء

يمكن أن تؤثر إعادة برمجة لوحة الألوان على المحتويات الأخرى للطرفية بطرق معقدة. هذه ليست مشكلة عند استخدام طريقة RGB.
