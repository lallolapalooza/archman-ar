notcurses_menu(3)

الحزمة: notcurses

التراخيص: Apache

notcurses_menu(3) دليل وظائف المكتبة notcurses_menu(3)

الاسم

notcurses_menu - عمليات على كائنات ncmenu

ملخص

#include <notcurses/notcurses.h>

struct ncmenu;
struct ncplane;
struct notcurses;
struct ncmenu_section {
  const char* name;   // سلسلة نصيةutf-8
  struct ncmenu_item {
    const char* desc; // عنصر قائمةutf-8، NULL للفصل الأفقي
    ncinput shortcut; // اختصار، يجب أن تكون جميعها مميزة
  }* items;
  int itemcount;
};
#define NCMENU_OPTION_BOTTOM 0x0001 // الصف السفلي (مقابل الصف العلوي)
#define NCMENU_OPTION_HIDING 0x0002 // إخفاء القائمة عندما لا تكون مفتوحة
typedef struct ncmenu_options {
  struct ncmenu_section* sections; // 'sectioncount' menu_sections
  int sectioncount;         // يجب أن تكون موجبة
  uint64_t headerchannels;  // تنسيق الرأس
  uint64_t sectionchannels; // تنسيق الأقسام
  uint64_t flags;           // حقل بت على NCMENU_OPTION_*
} ncmenu_options;

struct ncmenu* ncmenu_create(struct notcurses* nc, const menu_options* opts);

int ncmenu_unroll(struct ncmenu* n, int sectionidx);

int ncmenu_rollup(struct ncmenu* n);

int ncmenu_nextsection(struct ncmenu* n);

int ncmenu_prevsection(struct ncmenu* n);

int ncmenu_nextitem(struct ncmenu* n);

int ncmenu_previtem(struct ncmenu* n);

int ncmenu_item_set_status(struct ncmenu* n, const char* section, const char* item, bool enabled);

const char* ncmenu_selected(const struct ncmenu* n, ncinput* ni);

const char* ncmenu_mouse_selected(const struct ncmenu* n, const ncinput* click, ncinput* ni);

struct ncplane* ncmenu_plane(struct ncmenu* n);

bool ncmenu_offer_input(struct ncmenu* n, const ncinput* nc);

void ncmenu_destroy(struct ncmenu* n);

الوصف

يدعم مثيل notcurses أشرطة القوائم في الصف العلوي أو السفلي من المستوى. تتكون القائمة من أقسام، والتي تتكون بدورها من عناصر. إما لا توجد أقسام مرئية، والقائمة "مطوية"، أو يتم "فتح" قسم واحد بالضبط. يضع ncmenu_rollup قائمة ncmenu في الحالة المطوية. يقوم ncmenu_unroll بطي أي قسم مفتوح، وفتح القسم المحدد. يقوم ncmenu_destroy بإزالة شريط القائمة، وتحرير جميع الموارد المرتبطة.

يعيد ncmenu_selected وصف العنصر المحدد، أو NULL إذا لم يكن هناك قسم مفتوح، أو لم يتم تحديد عنصر صالح. يعيد ncmenu_mouse_selected العنصر المحدد بنقرة الماوس (الموصوفة في click)، والتي يجب أن تكون على القسم المفتوح حاليًا. في كلتا الحالتين، إذا كان هناك اختصار للعنصر و ni ليس NULL، فسيتم ملء ni بالاختصار.

يمكن قيادة القائمة إما بالكامل بواسطة التطبيق، عبر استدعاءات مباشرة لـ ncmenu_previtem و ncmenu_prevsection وما شابه، أو يمكن تمرير كائنات ncinput إلى ncmenu_offer_input. في الحالة الأخيرة، ستدير القائمة نفسها إلى حد كبير. يجب على التطبيق معالجة تحديد العنصر (عادةً عبر مفتاح Enter و / أو نقرة الماوس) بنفسه، حيث لا يمكن للقائمة الاستدعاء بشكل تعسفي إلى التطبيق. ستتعامل ncmenu_offer_input مع النقرات لفتح أقسام القائمة، والنقرات لإغلاق القائمة، و Escape لإغلاق القائمة، واليسار واليمين لتغيير أقسام القائمة، والأعلى والأسفل لتغيير عناصر القائمة (هذه الأخيرة تستهلك فقط إذا كان هناك قسم مفتوح حاليًا).

يمكن تعطيل العناصر باستخدام ncmenu_item_set_status. مرر false لتعطيل العنصر، أو true لتمكينه. جميع العناصر ممكّنة افتراضيًا. لن يتم إرجاع عنصر معطل بواسطة ncmenu_mouse_selected، ولن يتم الوصول إليه بواسطة ncmenu_nextsection ولا ncmenu_prevsection.

قيم الإرجاع

يعيد ncmenu_create NULL عند حدوث خطأ، أو مؤشرًا إلى ncmenu جديد صالح. تعيد الوظائف الأخرى قيمة غير صفرية عند حدوث خطأ، أو صفر عند النجاح. جميع الأخطاء تقريبًا ناتجة عن معلمات غير صالحة.

يعيد ncmenu_offer_input true إذا "استهلكت" القائمة الإدخال، أي وجدت أنه ذو صلة واتخذت إجراءً. بخلاف ذلك، يتم إرجاع false، ويجب اعتبار ncinput غير ذي صلة بالقائمة.
