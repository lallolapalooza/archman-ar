الاسم

notcurses_input - الإدخال عبر notcurses

الملخص

#include <notcurses/notcurses.h>

struct timespec;
struct notcurses;
typedef enum {
  NCTYPE_UNKNOWN,
  NCTYPE_PRESS,
  NCTYPE_REPEAT,
  NCTYPE_RELEASE,
} ncintype_e;
// حدث إدخال. إحداثيات الخلية معرّفة حالياً فقط لأحداث
// الفأرة. ليس مضموناً أننا نستطيع تعيين المعدّلات لحدث
// ncinput معين. نحن نشمل نقاط ترميز يونيكود المفردة، وليس EGCs كاملة.
// FIXME لـ abi4، ادمج القيم المنطقية في |modifiers|
typedef struct ncinput {
  uint32_t id;       // نقطة ترميز يونيكود أو حدث NCKEY مُصنّع
  int y, x;          // إحداثي الخلية y/x للحدث، -1 إذا كان غير معرّف
  char utf8[5];      // تمثيل utf8، إن وُجد
  // DEPRECATED لا تستخدمه! سيُزال في 4.0
  bool alt;          // هل كان alt مضغوطاً؟
  bool shift;        // هل كان shift مضغوطاً؟
  bool ctrl;         // هل كان ctrl مضغوطاً؟
  // END DEPRECATION
  ncintype_e evtype;
  unsigned modifiers;// قناع بتات على NCKEY_MOD_*
  int ypx, xpx;      // إزاحات البكسل داخل الخلية، -1 إذا كانت غير معرّفة
  uint32_t eff_text[5];  // تمثيل utf32 الفعّال، مع مراعاة 
                     // مفاتيح المعدّلات. يمكن أن يكون عدة
                     // نقاط ترميز. المصفوفة منتهية بالصفر.
} ncinput;
#define NCMICE_NO_EVENTS     0
#define NCMICE_MOVE_EVENT    0x1
#define NCMICE_BUTTON_EVENT  0x2
#define NCMICE_DRAG_EVENT    0x4

int notcurses_linesigs_disable(struct notcurses* n);

الوصف

يدعم notcurses الإدخال من لوحات المفاتيح والفئران، وأي جهاز يشبهها. يتطلب دعم الفأرة وسيطاً مثل GPM أو Wayland أو Xorg، ويجب تمكينه صراحةً عبر notcurses_mice_enable. النطاق الكامل 32-بت من يونيكود مدعوم (انظر unicode(7))، مع أحداث مُصنّعة تُعيَّن فوق نقاط الترميز 1,114,112 الخاصة بمستويات يونيكود 14.0 السبعة عشر. تُعاد أحرف يونيكود مباشرةً كـ UCS-32، نقطة ترميز واحدة في كل مرة. عندما يكون للإدخال تمثيل UTF8، يُكتب إلى utf8؛ هذا الحقل منتهٍ دائماً بـ NUL.

يأخذ notcurses إدخال لوحة المفاتيح من stdin، والذي سيُوضع في الوضع غير الحاجب طوال مدة التشغيل. يُوضع الطرفية في الوضع غير القانوني (انظر termios(3))، وبالتالي تُستقبل المفاتيح دون تخزين مؤقت للأسطر. يحتفظ notcurses بمخزنه المؤقت الخاص لأحرف الإدخال، والذي سيحاول ملأه كلما قرأ.

يسمح notcurses_get بتحديد struct timespec كمهلة زمنية. إذا كان ts هو NULL، فسيحجب notcurses_get حتى يقرأ الإدخال، أو يُقاطع بواسطة إشارة. إذا كانت قيمه أصفاراً، فلن يكون هناك حجب. وإلا، يحدد ts موعداً نهائياً مطلقاً (مأخوذاً مقابل CLOCK_MONOTONIC؛ انظر clock_gettime(2)). عند انتهاء المهلة، يُعاد 0. ستُبلَّغ تفاصيل الحدث في ni، ما لم يكن ni هو NULL.

يوفر notcurses_inputready_fd واصف ملف مناسباً للاستخدام مع مضاعفات الإدخال/الإخراج مثل poll(2). قد يكون هذا الواصف أو لا يكون واصف ملف الإدخال الفعلي. إذا كان قابلاً للقراءة، فيمكن استدعاء notcurses_get دون احتمال الحجب.

يقارن ncinput_equal_p بين بنيتي ncinput من حيث تساوي البيانات (أي دون اعتبار الحشو)، ويعيد true إذا كانتا تمثلان نفس الإدخال (وإن لم يكن بالضرورة نفس حدث الإدخال). لاحظ أن NCTYPE_UNKNOWN وNCTYPE_PRESS يُعتبران متكافئين لأغراض ncinput_equal_p.

يعطّل notcurses_linesigs_disable تحويل المدخلات INTR وQUIT وSUSP وDSUSP إلى SIGINT وSIGQUIT وSIGTSTP. هذه التحويلات ممكّنة افتراضياً. يلغي notcurses_linesigs_enable هذا الإجراء، لكن الإشارات في الفترة المؤقتة تُفقد نهائياً.

تختبر ncinput_shift_p وncinput_ctrl_p وncinput_alt_p وncinput_meta_p n لمعرفة ما إذا كان المعدّل المعني مضبوطاً. هذا مفضّل على الوصول المباشر إلى أعضاء البنية.

الفئران

بالنسبة لأحداث الفأرة، تُضبط الحقول الإضافية y وx وypx وxpx. هذه الحقول ليست ذات معنى لأحداث ضغط المفاتيح. يمكن تمييز أحداث الفأرة باستخدام المسند nckey_mouse_p. يطلب NCMICE_MOVE_EVENT الأحداث كلما تحركت الفأرة عندما لا تكون أي أزرار مضغوطة. يطلب NCMICE_DRAG_EVENT الأحداث عندما تتحرك الفأرة مع ضغط الأزرار. يطلب NCMICE_BUTTON_EVENT الأحداث عند تغيّر حالة الزر. يُوفَّر NCMICE_ALL_EVENTS للراحة وللتحصين المستقبلي ضد تغييرات API (وإن لم يكن ABI).

ضغطات المفاتيح المُصنّعة

العديد من المفاتيح ليس لها تمثيل يونيكود، فضلاً عن ASCII. تشمل الأمثلة مفاتيح المعدّلات (Alt، Meta، إلخ)، ومفاتيح "الوظائف"، ومفاتيح الأسهم على لوحة المفاتيح الرقمية. المفاتيح الخاصة المتاحة للطرفية معرّفة في إدخال terminfo(5)، الذي يحمّله notcurses عند بدء التشغيل. عند استقبال رمز هروب يطابق قدرة إدخال terminfo، يُصنّع notcurses قيمة خاصة. يجب أن يصل تسلسل الهروب بالكامل إلى notcurses؛ نفاد الإدخال في منتصف تسلسل هروب سيؤدي إلى رفضه. وبالمثل، أي خطأ أثناء معالجة تسلسل هروب سيؤدي إلى إلغاء التحليل، وإعادة تشغيل التسلسل حتى الآن كضغطات مفاتيح حرفية مستقلة.

القائمة الكاملة للمفاتيح المُصنّعة (هناك أكثر من مئة بكثير) يمكن العثور عليها في <notcurses/notcurses.h>. لمزيد من التفاصيل، راجع terminfo(5).

NCKEY_RESIZE

ما لم يتم تثبيط معالج SIGWINCH (انظر notcurses_init)، سيلتقط Notcurses تلقائياً تغييرات حجم الشاشة، ويصنّع حدث NCKEY_RESIZE. عند استقبال هذا الحدث، يمكن للمستخدم استدعاء notcurses_refresh لفرض إعادة تدفق فورية، أو فقط الانتظار حتى الاستدعاء التالي لـ notcurses_render، عندما سيلتقط notcurses تغيير الحجم بنفسه. إذا تم تثبيط معالج SIGWINCH، فلن يُولَّد NCKEY_RESIZE أبداً.

NCKEY_SIGNAL

ما لم يتم تثبيط معالج SIGWINCH (انظر notcurses_init)، سيلتقط Notcurses SIGCONT

، وتوليد حدث NCKEY_SIGNAL. يشير هذا عادةً إلى أن البرنامج قد أُعيد تشغيله بعد إيقافه مؤقتًا أو وضعه في الخلفية. ستكون عملية التنقيط التالية إعادة بناء كاملة للشاشة، كما لو كانت notcurses_refresh قد استُدعيت؛ وقد يرغب المستخدم في استدعاء notcurses_refresh بنفسه فورًا.

NCKEY_EOF

عند الوصول إلى نهاية الإدخال، سيُعاد NCKEY_EOF. عند هذه النقطة، ستعيد أي استدعاءات لاحقة NCKEY_EOF فورًا. لاحظ أن هذا لا ينتج بالضرورة عن الضغط على مثل Ctrl+D.

قيم الإرجاع

عند حدوث خطأ، تُعيد عائلة دوال get القيمة (uint32_t)-1. يمكن تحديد سبب الخطأ باستخدام errno(3). ما لم يكن الخطأ مؤقتًا (خاصةً مثل EINTR)، فمن المحتمل ألا يكون استدعاء notcurses_get مفيدًا في الحال. عند انتهاء المهلة، تُعاد القيمة 0. وإلا، تُعاد قيمة UCS-32 لنقطة ترميز يونيكود، أو حدث مُولَّد.

إذا وُوجه خطأ قبل أن يقرأ notcurses_getvec أي إدخال، فسيُعيد -1. وإذا انتهت المهلة قبل قراءة أي إدخال، فسيُعيد 0. وإلا، فسيُعيد عدد كائنات ncinput المكتوبة.

تُعيد notcurses_mice_enable القيمة 0 عند النجاح، وقيمة غير صفرية عند الفشل، وكذلك تفعل notcurses_mice_disable. لا يعني النجاح بالضرورة أن الفأرة متاحة ولا أن جميع الأحداث المطلوبة ستُولَّد.

تُعيد ncinput_equal_p القيمة true إذا كان بنيتا ncinput تمثلان الإدخال نفسه (وإن لم يكن بالضرورة حدث الإدخال نفسه)، وfalse خلاف ذلك.

ملاحظات

مثل أي دالة notcurses أخرى، من الخطأ استدعاء notcurses_get أثناء أو بعد استدعاء notcurses_stop. إذا كان خيط دائمًا ينتظر على إدخال حاجز، فقد يكون من الصعب ضمان عدم حدوث ذلك.

يمكن لخيط واحد فقط استدعاء مكدس الإدخال في وقت واحد، ولكن على عكس كل دالة أخرى تقريبًا في notcurses، يمكن استدعاء notcurses_get وما شابهها بالتزامن مع notcurses_render.

لا تقم ببساطة بـ poll على واصف الملف المرتبط بـ stdin لاختبار جاهزية الإدخال. بدلًا من ذلك، استخدم واصف الملف الذي تُعيده notcurses_inputready_fd لضمان التوافق مع الإصدارات المستقبلية من Notcurses (من الممكن أن تعالج الإصدارات المستقبلية الإدخال في سياقاتها الخاصة).

القائمة الكاملة للأحداث المُولَّدة متاحة في <notcurses/nckeys.h>.

في API4، ستختفي حقول المُعدِّلات المتنوعة من نوع bool، وستُدمج هذه الحالات في قناع البتات modifiers. يُشجَّع على استخدام ncinput_shift_p وما شابهها لجعل شفرتك محصنة ضد التغييرات المستقبلية.

الأخطاء

تحاول Notcurses استخدام بروتوكولي XTMODKEYS وKitty لإزالة الغموض عن لوحة المفاتيح. عند دعمهما، فإنهما يزيلان معظم هذه المشكلات.

لا يُشار تقليديًا إلى مفتاح Shift بالاقتران مع نص يونيكود النموذجي. فإذا استُخدم مثل Shift لتوليد حرف كبير 'A'، فستكون id مساوية لـ 'A'، وستكون shift مساوية لـ false. وبالمثل، عندما يُضغط Ctrl مع حرف، فسيُبلَّغ عن الحرف حاليًا دائمًا بصيغته الكبيرة. فمثلًا، إذا ضُغط Shift وCtrl و'a' جميعًا، فهذا لا يمكن تمييزه عن Ctrl و'A'.

يُسجَّل ضغط Ctrl مع 'J' أو 'M'، سواء ضُغط Shift أم لا، حاليًا على أنه NCKEY_ENTER. من المحتمل أن يتغير هذا في المستقبل.

عند استخدام بروتوكول Kitty لإزالة الغموض عن لوحة المفاتيح، تُحل معظم هذه المشكلات. يمكنك تحديد ما إذا كان البروتوكول قيد الاستخدام بفحص مخرجات notcurses-info(1). إذا أُشير إلى خاصية kbd، فأنت تستخدم بروتوكول Kitty.

لاحظ أن eff_text سيعيد دائمًا قيمة النص الفعلية للمفتاح، مع مراعاة أي مفاتيح ميتا مضغوطة. (لذا سيعيد shift-a دائمًا 'A' في هذا الحقل، بغض النظر عما إذا كان بروتوكول Kitty لإزالة الغموض عن لوحة المفاتيح مستخدمًا.) ومع ذلك، لا يمكن الوثوق بهذا الحقل عمومًا إلا في أحداث NCTYPE_PRESS وNCTYPE_REPEAT. لاحظ أيضًا أن هذا مصفوفة (منتهية بصفر) لأنه يمكن أن يعيد نقاط ترميز متعددة.

لن تُسلَّم أحداث الفأرة في الهوامش اليسرى أبدًا إلى التطبيق (كما هو مقصود)، لكن أحداث الفأرة في الهوامش السفلية واليمنى يمكن أن تُسلَّم أحيانًا إذا وقع الحدث قبل تغيير حجم النافذة.

الحقلان ypx و xpx غير صالحين حاليًا أبدًا (أي أنهما دائمًا -1). ينبغي إصلاح هذا في المستقبل باستخدام بروتوكول فأرة SGR PixelMode.

في بعض أنظمة التشغيل، يُستخدم CLOCK_REALTIME كأساس للمهلات بدلًا من CLOCK_MONOTONIC. ينبغي إصلاح هذا.
