nlopt_minimize_constrained(3)

الحزمة: nlopt

التراخيص: LGPL-2.1-or-later

NLOPT_MINIMIZE_CONSTRAINED(3) NLopt programming manual NLOPT_MINIMIZE_CONSTRAINED(3)

الاسم

nlopt_minimize_constrained - تقليل دالة غير خطية متعددة المتغيرات مقيدة بقيود غير خطية

الملخص

#include <nlopt.h>

nlopt_result nlopt_minimize_constrained(nlopt_algorithm algorithm,
                            int n,
                            nlopt_func f,
                            void* f_data,
                            int m,
                            nlopt_func fc,
                            void* fc_data,
                            ptrdiff_t fc_datum_size,
                            const double* lb,
                            const double* ub,
                            double* x,
                            double* minf,
                            double minf_max,
                            double ftol_rel,
                            double ftol_abs,
                            

                            double xtol_rel,
                            const double* xtol_abs,
                            int maxeval,
                            double maxtime);

يجب ربط البرنامج الناتج بعلامات الرابط
-lnlopt -lm على يونكس.

الوصف

تحاول nlopt_minimize_constrained() تقليل دالة غير خطية f لـ n متغيرات تصميم، مقيدة بـ m قيود غير خطية موصوفة بواسطة الدالة fc (انظر أدناه)، باستخدام الخوارزمية المحددة algorithm. يتم إرجاع الحد الأدنى لقيمة الدالة في minf، مع إرجاع قيم متغيرات التصميم المقابلة في المصفوفة x بطول n. يجب أن تكون القيم المدخلة في x تخمينًا أوليًا للقيمة المثلى. المدخلات lb و ub هي مصفوفات بطول n تحتوي على الحدود الدنيا والعليا، على التوالي، لمتغيرات التصميم x. تحدد المعلمات الأخرى معايير التوقف (التفاوتات، الحد الأقصى لعدد تقييمات الدالة، إلخ) ومعلومات أخرى كما هو موضح بمزيد من التفصيل أدناه. قيمة الإرجاع هي رمز صحيح يشير إلى النجاح (موجب) أو الفشل (سالب)، كما هو موضح أدناه.

عن طريق تغيير المعلمة algorithm بين عدة ثوابت محددة مسبقًا موصوفة أدناه، يمكن التبديل بسهولة بين مجموعة متنوعة من خوارزميات التقليل. تتطلب بعض هذه الخوارزميات توفير التدرج (المشتقات) للدالة عبر f، ولا تتطلب خوارزميات أخرى مشتقات. تحاول بعض الخوارزميات العثور على حد أدنى عالمي ضمن الحدود المعطاة، ويجد البعض الآخر حدًا أدنى محليًا فقط. تتعامل معظم الخوارزميات فقط مع الحالة التي تكون فيها m صفرًا (لا توجد قيود غير خطية صريحة)؛ الخوارزميات الوحيدة التي تدعم حاليًا m موجب هي NLOPT_LD_MMA و NLOPT_LN_COBYLA.

nlopt_minimize_constrained هي دالة تغليف حول العديد من حزم التقليل المجانية/مفتوحة المصدر، بالإضافة إلى بعض التطبيقات الجديدة للخوارزميات المنشورة. يمكنك بالطبع تجميع هذه الحزم واستدعائها بشكل منفصل، وفي بعض الحالات سيوفر هذا مرونة أكبر مما هو متاح عبر واجهة nlopt_minimize_constrained. ومع ذلك، اعتمادًا على الدالة المحددة التي يتم تقليلها، ستختلف فعالية الخوارزميات المختلفة. الغرض من nlopt_minimize_constrained هو السماح لك بالتبديل بسرعة بين الخوارزميات لتجربتها لمشكلتك، عن طريق توفير واجهة موحدة بسيطة لهذه الروتينات الفرعية.

دالة الهدف

تقوم nlopt_minimize_constrained() بتقليل دالة هدف f بالشكل:

double f(int n,
const double* x,
double* grad,
void* f_data);

يجب أن تكون القيمة المرجعة هي قيمة الدالة عند النقطة x، حيث تشير x إلى مصفوفة بطول n لمتغيرات التصميم. البعد n مطابق للبعد الذي تم تمريره إلى nlopt_minimize_constrained().

بالإضافة إلى ذلك، إذا كان الوسيط grad غير NULL، فإن grad يشير إلى مصفوفة بطول n والتي يجب (عند الإرجاع) تعيينها إلى تدرج الدالة بالنسبة لمتغيرات التصميم عند x. أي، يجب أن تحتوي grad[i] عند الإرجاع على المشتقة الجزئية df/dx[i]، لـ 0 <= i < n، إذا كان grad غير NULL. لا تستخدم جميع خوارزميات التحسين (أدناه) معلومات التدرج: بالنسبة للخوارزميات المدرجة كـ "خالية من المشتقات"، سيكون الوسيط grad دائمًا NULL ولا يحتاج أبدًا إلى حسابه. (بالنسبة للخوارزميات التي تستخدم معلومات التدرج، ومع ذلك، قد يظل grad NULL لبعض الاستدعاءات.)

الوسيط f_data هو نفس الوسيط الذي تم تمريره إلى nlopt_minimize_constrained()، وقد يستخدم لتمرير أي بيانات إضافية إلى الدالة. (أي، قد يكون مؤشرًا إلى بنية بيانات/نوع معرف من قبل المستدعي يحتوي على معلومات تحتاجها دالتك، والتي تقوم بتحويلها من void* عن طريق تحويل النوع.)

قيود الحدود

تم تصميم معظم الخوارزميات في NLopt لتقليل الدوال ذات القيود الحدودية البسيطة على المدخلات. أي، يتم تقييد المتجهات المدخلة x[i] لتكون في مستطيل فائق lb[i] <= x[i] <= ub[i] لـ 0 <= i < n، حيث lb و ub هما المصفوفتان اللتان تم تمريرهما إلى nlopt_minimize_constrained().

ومع ذلك، تدعم بعض الخوارزميات التحسين الجزئي أو الكلي غير المقيد، كما هو موضح أدناه، حيث يتم الإشارة إلى متغير تصميم غير مقيد (بالكامل أو جزئيًا) عن طريق حد أدنى يساوي -Inf و/أو حد أعلى يساوي +Inf. هنا، Inf هو لانهائية الفاصلة العائمة IEEE-754، والتي (في ANSI C99) يتم تمثيلها بالماكرو INFINITY في math.h. بدلاً من ذلك، للإصدارات الأقدم من C، يمكنك أيضًا استخدام الماكرو HUGE_VAL (أيضًا في math.h).

مع بعض الخوارزميات، خاصة تلك التي لا تتطلب معلومات المشتقات، تتمثل طريقة بسيطة (ولكنها ليست فعالة بشكل خاص) لتطبيق قيود غير خطية اعتباطية في إرجاع Inf (انظر أعلاه) كلما تم انتهاك القيود بواسطة مدخل x معين. بشكل عام، هناك طرق مختلفة لتطبيق القيود عن طريق إضافة "مصطلحات عقوبة" إلى دالة الهدف الخاصة بك، والتي توصف في أدبيات التحسين. طريقة أكثر كفاءة لتحديد القيود غير الخطية موصوفة أدناه، ولكنها مدعومة فقط من قبل مجموعة صغيرة من الخوارزميات.

قيود غير خطية

تسمح لك الدالة nlopt_minimize_constrained() أيضًا بتحديد m قيود غير خطية عبر الدالة fc، حيث m هو أي عدد صحيح غير سالب. ومع ذلك، فإن m غير الصفر مدعوم حاليًا فقط من قبل الخوارزميات NLOPT_LD_MMA و NLOPT_LN_COBYLA أدناه.

على وجه الخصوص، قيود غير خطية هي بالشكل fc(x) <= 0، حيث الدالة fc لها نفس شكل دالة الهدف الموصوفة أعلاه:

double fc(int n,
const double* x,
double* grad,
void* fc_datum);

يجب أن تكون القيمة المرجعة هي قيمة القيد عند النقطة x، حيث البعد n مطابق للبعد الذي تم تمريره إلى nlopt_minimize_constrained(). كما هو الحال بالنسبة لدالة الهدف، إذا كان الوسيط grad غير NULL، فإن grad يشير إلى مصفوفة بطول n والتي يجب (عند الإرجاع) تعيينها إلى تدرج الدالة بالنسبة إلى x. (لأي خوارزمية مدرجة كـ "خالية من المشتقات" أدناه، سيكون الوسيط grad دائمًا NULL ولا يحتاج أبدًا إلى حسابه.)

الوسيط fc_datum يعتمد على الوسيط fc_data الذي تم تمريره إلى nlopt_minimize_constrained()، وقد يستخدم لتمرير أي بيانات إضافية إلى الدالة، ويستخدم للتمييز بين القيود المختلفة.

الخوارزميات

يحدد المعامل algorithm خوارزمية التحسين (لمزيد من التفاصيل حول هذه الخوارزميات، انظر ملفات README في الدلائل الفرعية لشفرة المصدر)، ويمكن أن يأخذ أيًا من القيم الثابتة التالية. الثوابت التي تحتوي على _G{N,D}_ في أسمائها تشير إلى طرق التحسين العالمي، بينما تشير _L{N,D}_ إلى طرق التحسين المحلي (التي تحاول إيجاد حد أدنى محلي بدءًا من التخمين الابتدائي x). الثوابت التي تحتوي على _{G,L}N_ تشير إلى خوارزميات غير مشتقة (خالية من المشتقات) لا تتطلب من دالة الهدف توفير مشتق، بينما تشير _{G,L}D_ إلى خوارزميات تعتمد على المشتقات وتتطلب من دالة الهدف توفير مشتق. (خاصة بالنسبة للتحسين المحلي، تكون الخوارزميات المعتمدة على المشتقات أفضل بشكل عام من الخوارزميات الخالية من المشتقات: المشتق مفيد *إذا* كان يمكنك حسابه بتكلفة منخفضة، على سبيل المثال عبر طريقة التبعية.)

NLOPT_GN_DIRECT_L
إجراء تحسين عالمي (G) خالٍ من المشتقات (N) باستخدام خوارزمية البحث DIRECT-L بواسطة Jones et al. كما عدلها Gablonsky et al. لتكون أكثر تركيزًا على البحث المحلي. لا تدعم التحسين غير المقيد. هناك أيضًا العديد من المتغيرات الأخرى لخوارزمية DIRECT التي يتم دعمها: NLOPT_GN_DIRECT، وهي خوارزمية DIRECT الأصلية؛ NLOPT_GN_DIRECT_L_RAND، وهي نسخة عشوائية قليلاً من DIRECT-L قد تكون أفضل في مساحات البحث عالية الأبعاد؛ NLOPT_GN_DIRECT_NOSCAL، NLOPT_GN_DIRECT_L_NOSCAL، و NLOPT_GN_DIRECT_L_RAND_NOSCAL، وهي إصدارات من DIRECT حيث لا يتم تغيير مقياس الأبعاد إلى مكعب الوحدة (مما يعني أن الأبعاد ذات الحدود الأكبر تحصل على وزن أكبر).
NLOPT_GN_ORIG_DIRECT_L
تحسين عالمي (G) خالٍ من المشتقات باستخدام خوارزمية DIRECT-L كما هو موضح أعلاه، جنبًا إلى جنب مع NLOPT_GN_ORIG_DIRECT وهي خوارزمية DIRECT الأصلية. بخلاف NLOPT_GN_DIRECT_L أعلاه، تشير هاتان الخوارزميتان إلى شفرة تستند إلى شفرة Fortran الأصلية لـ Gablonsky et al.، والتي تحتوي على بعض القيود المضمنة على عدد التقسيمات وما إلى ذلك ولا تدعم جميع معايير التوقف لـ NLopt، ولكنها تدعم من ناحية أخرى القيود غير الخطية العشوائية كما هو موضح أعلاه.
NLOPT_GD_STOGO
تحسين عالمي (G) باستخدام خوارزمية StoGO بواسطة Madsen et al. تستغل StoGO معلومات المشتق (D) (التي يجب توفيرها بواسطة الهدف) لبحوثها المحلية، وتقوم بالبحث العالمي عن طريق تقنية الفرع والحد. يتم دعم التحسين المقيد بالحدود فقط. هناك أيضًا متغير آخر لهذه الخوارزمية، NLOPT_GD_STOGO_RAND، وهي نسخة عشوائية من مخطط بحث StoGO. خوارزميات StoGO متاحة فقط إذا تم تجميع NLopt مع تمكين C++، ويجب ربطها عبر -lnlopt_cxx (عبر مترجم C++، لربط مكتبات C++ القياسية).
NLOPT_LN_NELDERMEAD
إجراء تحسين محلي (L) خالٍ من المشتقات (N)، بدءًا من x، باستخدام خوارزمية Nelder-Mead simplex، المعدلة لدعم قيود الحدود. Nelder-Mead، على الرغم من شعبيتها، معروفة أحيانًا بالفشل في التقارب لبعض دوال الهدف، لذا يجب استخدامها بحذر. الأدلة القصصية، من ناحية أخرى، تشير إلى أنها تعمل بشكل جيد إلى حد ما للأهداف المتقطعة. انظر أيضًا NLOPT_LN_SBPLX أدناه.
NLOPT_LN_SBPLX
إجراء تحسين محلي (L) خالٍ من المشتقات (N)، بدءًا من x، باستخدام خوارزمية تستند إلى خوارزمية Subplex بواسطة Rowan et al.، وهي نسخة محسنة من Nelder-Mead (أعلاه). تنفيذنا لا يستخدم شفرة Rowan الأصلية، ولها بعض التعديلات الطفيفة مثل الدعم الصريح لقيود الحدود. (مثل Nelder-Mead، غالبًا ما تعمل Subplex بشكل جيد في الممارسة العملية، حتى للأهداف المتقطعة، ولكن لا يوجد ضمان صارم بأنها ستتقارب.) يمكن دعم القيود غير الخطية بشكل تقريبي عن طريق إرجاع +Inf عند انتهاك القيود، كما هو موضح أعلاه.
NLOPT_LN_PRAXIS
تحسين محلي (L) خالٍ من المشتقات (N) باستخدام طريقة المحور الرئيسي، بناءً على شفرة Richard Brent. مصممة للتحسين غير المقيد، على الرغم من أن قيود الحدود مدعومة أيضًا (عبر الطريقة غير الفعالة لإرجاع +Inf عند انتهاك القيود).
NLOPT_LD_LBFGS
تحسين محلي (L) يعتمد على المشتقات (D) باستخدام خوارزمية BFGS محدودة الذاكرة (L-BFGS). (يجب أن توفر دالة الهدف المشتق.) يتم دعم التحسين غير المقيد بالإضافة إلى قيود الحدود البسيطة (انظر أعلاه). بناءً على تنفيذ بواسطة Luksan et al.
NLOPT_LD_VAR2
تحسين محلي (L) يعتمد على المشتقات (D) باستخدام طريقة متغيرة محدودة الذاكرة مزاحة بناءً على شفرة Luksan et al.، تدعم التحسين غير المقيد والمقيد بالحدود. NLOPT_LD_VAR2 يستخدم طريقة من الرتبة الثانية، بينما .B NLOPT_LD_VAR1 هو متغير آخر يستخدم طريقة من الرتبة الأولى.
NLOPT_LD_TNEWTON_PRECOND_RESTART
تحسين محلي (L) يعتمد على المشتقات (D) باستخدام طريقة نيوتن المقطوعة المسبقة بـ LBFGS مع إعادة تشغيل الانحدار الأشد، بناءً على شفرة Luksan et al.، تدعم التحسين غير المقيد والمقيد بالحدود. هناك العديد من المتغيرات الأخرى لهذه الخوارزمية: NLOPT_LD_TNEWTON_PRECOND (نفس الشيء بدون إعادة التشغيل)، NLOPT_LD_TNEWTON_RESTART (نفس الشيء بدون التكييف المسبق)، و NLOPT_LD_TNEWTON (نفس الشيء بدون إعادة التشغيل أو التكييف المسبق).
NLOPT_GN_CRS2_LM
تحسين عالمي (G) خالٍ من المشتقات (N) باستخدام خوارزمية البحث العشوائي المتحكم فيه (CRS2) بواسطة Price، مع تعديل "الطفرة المحلية" (LM) المقترح بواسطة Kaelo و Ali.
NLOPT_GD_MLSL_LDS, NLOPT_GN_MLSL_LDS
تحسين عالمي (G) يعتمد على المشتقات (D) أو خالٍ من المشتقات (N) باستخدام خوارزمية الارتباط المتعدد المستويات (MLSL) مع تسلسل ذي تباين منخفض (LDS). تنفذ هذه الخوارزمية تسلسلًا شبه عشوائي (LDS) من عمليات البحث المحلي، مع طريقة تجميع لتجنب عمليات البحث المحلي المتعددة لنفس الحد الأدنى المحلي. يستخدم البحث المحلي خوارزمية المشتق/غير المشتق المحددة بواسطة nlopt_set_local_search_algorithm (افتراضي حاليًا إلى NLOPT_LD_MMA و NLOPT_LN_COBYLA للبحث بالمشتق/بدون مشتق، على التوالي). هناك أيضًا متغيران آخران، NLOPT_GD_MLSL و NLOPT_GN_MLSL، والتي تستخدم أرقامًا شبه عشوائية (بدلاً من LDS) كما في خوارزمية MLSL الأصلية.
NLOPT_LD_MMA
تحسين محلي (L) يعتمد على المشتقات (D) باستخدام طريقة المتوازيات المتحركة (MMA)، أو بالأحرى نسخة محسنة من الخوارزمية كما نشرها Svanberg (2002). (يستخدم NLopt تنفيذًا مستقلاً ومفتوح المصدر لخوارزمية Svanberg.) تدعم خوارزمية NLOPT_LD_MMA كلاً من التحسين المقيد بالحدود وغير المقيد، كما تدعم عددًا عشوائيًا (m) من القيود غير الخطية كما هو موضح أعلاه.
NLOPT_LN_COBYLA
تحسين محلي (L) خالٍ من المشتقات (N) باستخدام خوارزمية COBYLA لبowel (التحسين المقيد بواسطة التقريبات الخطية). تدعم خوارزمية NLOPT_LN_COBYLA كلاً من التحسين المقيد بالحدود وغير المقيد، كما تدعم عددًا عشوائيًا (m) من القيود غير الخطية كما هو موضح أعلاه.
NLOPT_LN_NEWUOA
تحسين محلي (L) خالٍ من المشتقات (N) باستخدام متغير من خوارزمية NEWUOA لبowel، بناءً على التقريبات التربيعية المتتابعة لدالة الهدف. قمنا بتعديل الخوارزمية لدعم قيود الحدود. الخوارزمية الأصلية NEWUOA متاحة أيضًا، باسم NLOPT_LN_NEWUOA، ولكن هذه الخوارزمية تتجاهل قيود الحدود lb و ub، ولذا يجب استخدامها فقط للمشاكل غير المقيدة.

معايير التوقف

يتم دعم معايير توقف متعددة للتحسين، كما هو محدد بالمعاملات التالية لـ nlopt_minimize_constrained(). تتوقف عملية التحسين عندما يتم استيفاء أي من هذه المعايير. في بعض الحالات، يعتمد التفسير الدقيق لمعيار التوقف على خوارزمية التحسين المذكورة أعلاه (على الرغم من أننا حاولنا جعلها متسقة قدر الإمكان)، وبعض الخوارزميات لا تدعم جميع معايير التوقف.

هام: لست بحاجة إلى استخدام جميع معايير التوقف! في معظم الحالات، تحتاج فقط إلى واحد أو اثنين، ويمكنك تعيين الباقي لقيم لا تفعل شيئًا (كما هو موضح أدناه).

minf_max
توقف عندما يتم العثور على قيمة دالة أقل من أو تساوي minf_max. اضبطها على -Inf أو NaN (انظر قسم القيود أعلاه) لتعطيلها.
ftol_rel
التسامح النسبي على قيمة الدالة: توقف عندما يغير خطوة التحسين (أو تقدير الحد الأدنى) قيمة الدالة بأقل من ftol_rel مضروبة في القيمة المطلقة لقيمة الدالة. (إذا كان هناك أي احتمال أن تكون قيمة الدالة الدنيا قريبة من الصفر، فقد ترغب في تعيين تسامح مطلق مع ftol_abs أيضًا.) معطلة إذا كانت غير موجبة.
ftol_abs
التسامح المطلق على قيمة الدالة: توقف عندما يغير خطوة التحسين (أو تقدير الحد الأدنى) قيمة الدالة بأقل

معايير الإيقاف

ftol_rel
التسامح النسبي في قيمة الدالة: توقف عندما تتغير قيمة الدالة المحسوبة للتكرار الحالي بمقدار أقل من ftol_rel مضروبًا في القيمة المطلقة للدالة في التكرار السابق. معطل إذا كان غير موجب.
ftol_abs
التسامح المطلق في قيمة الدالة: توقف عندما تتغير قيمة الدالة المحسوبة للتكرار الحالي بمقدار أقل من ftol_abs. معطل إذا كان غير موجب (أو إذا كان أقل من ftol_rel).
xtol_rel
التسامح النسبي في متغيرات التصميم: توقف عندما يغير كل متغير تصميم (أو تقدير للحد الأدنى) بمقدار أقل من xtol_rel مضروبًا في القيمة المطلقة لمتغير التصميم. (إذا كان هناك أي احتمال لأن يكون متغير التصميم الأمثل قريبًا من الصفر، فقد ترغب في تعيين تسامح مطلق مع xtol_abs أيضًا.) معطل إذا كان غير موجب.
xtol_abs
مؤشر إلى مصفوفة بطول n تعطي التسامحات المطلقة في متغيرات التصميم: توقف عندما يغير كل متغير تصميم x[i] بمقدار أقل من xtol_abs[i]. معطل إذا كان غير موجب، أو إذا كان xtol_abs هو NULL.
maxeval
توقف عندما يتجاوز عدد تقييمات الدالة maxeval. (هذا ليس حدًا أقصى صارمًا: قد يتجاوز عدد تقييمات الدالة maxeval قليلاً، اعتمادًا على الخوارزمية.) معطل إذا كان غير موجب.
maxtime
توقف عندما يتجاوز وقت التحسين (بالثواني) maxtime. (هذا ليس حدًا أقصى صارمًا: قد يتجاوز الوقت maxtime قليلاً، اعتمادًا على الخوارزمية وعلى مدى بطء تقييم الدالة لديك.) معطل إذا كان غير موجب.

قيمة الإرجاع

القيمة المرجعة هي واحدة من الثوابت التعدادية التالية.

انتهاء ناجح (قيم إرجاع موجبة):

NLOPT_SUCCESS
قيمة إرجاع نجاح عامة.
NLOPT_MINF_MAX_REACHED
توقف التحسين لأن minf_max (أعلاه) قد تم الوصول إليه.
NLOPT_FTOL_REACHED
توقف التحسين لأن ftol_rel أو ftol_abs (أعلاه) قد تم الوصول إليه.
NLOPT_XTOL_REACHED
توقف التحسين لأن xtol_rel أو xtol_abs (أعلاه) قد تم الوصول إليه.
NLOPT_MAXEVAL_REACHED
توقف التحسين لأن maxeval (أعلاه) قد تم الوصول إليه.
NLOPT_MAXTIME_REACHED
توقف التحسين لأن maxtime (أعلاه) قد تم الوصول إليه.

رموز الخطأ (قيم إرجاع سالبة):

NLOPT_FAILURE
رمز فشل عام.
NLOPT_INVALID_ARGS
وسائط غير صالحة (مثل، الحدود الدنيا أكبر من الحدود العليا، تم تحديد خوارزمية غير معروفة، إلخ).
NLOPT_OUT_OF_MEMORY
نفدت الذاكرة.

الأرقام شبه العشوائية

بالنسبة لخوارزميات التحسين الاحتمالية، نستخدم أرقامًا شبه عشوائية تم إنشاؤها بواسطة خوارزمية Mersenne Twister، استنادًا إلى كود من Makoto Matsumoto. افتراضيًا، يتم إنشاء بذرة الأرقام العشوائية من وقت النظام، بحيث تكون مختلفة في كل مرة تقوم فيها بتشغيل البرنامج. إذا كنت تريد استخدام أرقام عشوائية حتمية، يمكنك تعيين البذرة عن طريق استدعاء:

void nlopt_srand(unsigned long seed);

تدعم بعض الخوارزميات أيضًا استخدام تسلسلات التباعد المنخفض (LDS)، المعروفة أحيانًا باسم الأرقام شبه العشوائية. يستخدم NLopt تسلسل Sobol LDS، والذي تم تنفيذه لما يصل إلى 1111 بُعدًا.
