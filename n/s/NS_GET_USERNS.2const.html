NS_GET_USERNS(2const) NS_GET_USERNS(2const)

الاسم

NS_GET_USERNS, NS_GET_PARENT - اكتشاف علاقات مساحات الأسماء

الملخص

#include <linux/nsfs.h>  /* تعريف الثوابت NS_GET_* */
                                #include <sys/ioctl.h>
int ioctl(int fd, unsigned long op);

الوصف

توفر عمليات ioctl(2) التالية للسماح باكتشاف علاقات مساحات الأسماء (انظر user_namespaces(7) و pid_namespaces(7)).

في كل حالة، يشير fd إلى ملف /proc/pid/ns/*. تعيد كلتا العمليتين واصف ملف جديد عند النجاح.

NS_GET_USERNS
تعيد واصف ملف يشير إلى مساحة أسماء المستخدم التي تشير إليها مساحة الأسماء التي يشير إليها fd.
NS_GET_PARENT
تعيد واصف ملف يشير إلى مساحة الأسماء الأصل لمساحة الأسماء التي يشير إليها fd. هذه العملية صالحة فقط لمساحات الأسماء الهرمية (أي مساحات أسماء PID والمستخدم). بالنسبة لمساحات أسماء المستخدم، فإن NS_GET_PARENT يعادل NS_GET_USERNS.

واصف الملف الجديد الذي تعيده هذه العمليات يتم فتحه باستخدام العلامات O_RDONLY و O_CLOEXEC (إغلاق عند التنفيذ؛ انظر fcntl(2)).

بتطبيق fstat(2) على واصف الملف المعاده، تحصل على بنية stat التي تستخدم st_dev (الجهاز المقيم) و st_ino (رقم inode) معًا لتحديد مساحة أسماء المستخدم/الأصل. قد يتطابق رقم inode هذا مع رقم inode لملف /proc/pid/ns/{pid,user} آخر لتحديد ما إذا كان مساحة أسماء المستخدم/الأصل.

قيمة الإرجاع

يتم إرجاع واصف ملف عند النجاح. أو خطأ، يتم إرجاع -1 ويتم تعيين errno للإشارة إلى الخطأ.

الأخطاء

EPERM
مساحة الأسماء المطلوبة خارج نطاق مساحة أسماء المتصل. قد يحدث هذا، على سبيل المثال، إذا كانت مساحة أسماء المستخدم المشار إليها هي سلف لمساحة أسماء المتصل الحالية. قد يحدث أيضًا أثناء محاولة الحصول على أصل مساحة أسماء المستخدم الأولية أو مساحة أسماء PID الأولية.
ENOTTY
العملية غير مدعومة في هذا النواة.

بالإضافة إلى ذلك، قد تفشل العملية NS_GET_PARENT مع الخطأ التالي:

EINVAL
fd يشير إلى مساحة أسماء غير هرمية.

أمثلة

يستخدم المثال المعروض عمليات ioctl(2) الموضحة أعلاه لإجراء اكتشاف بسيط لعلاقات مساحات الأسماء. تظهر جلسات shell التالية أمثلة مختلفة لاستخدام هذا البرنامج.

محاولة الحصول على الأصل لمساحة أسماء المستخدم الأولية تفشل، حيث لا يوجد لها أصل:

$ ./ns_show /proc/self/ns/user p

إنشاء عملية موجودة في مساحة أسماء مستخدم جديدة ومساحة أسماء UTS، وإظهار أن مساحة أسماء UTS الجديدة مرتبطة بمساحة أسماء المستخدم الجديدة:

$ unshare -Uu sleep 1000 &

$ ./ns_show /proc/23235/ns/uts u

بعد ذلك، أظهر أن الأصل لمساحة أسماء المستخدم الجديدة في المثال السابق هو مساحة أسماء المستخدم الأولية:

$ readlink /proc/self/ns/user

ابدأ shell في مساحة أسماء مستخدم جديدة وأظهر أنه من داخل هذا الـ shell لا يمكن اكتشاف مساحة أسماء المستخدم الأصل. بالمثل، لا يمكن اكتشاف مساحة أسماء UTS (المرتبطة بمساحة أسماء المستخدم الأولية).

$ PS1="sh2$ " unshare -U bash
$ ./ns_show /proc/self/ns/user p
$ ./ns_show /proc/self/ns/uts u

مصدر البرنامج

/* ns_show.c
           مرخص بموجب رخصة جنو العمومية العامة الإصدار 2 أو أحدث.
       */
#include <errno.h>
#include <fcntl.h>
#include <linux/nsfs.h>
#include <stdint.h>
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <sys/ioctl.h>
#include <sys/stat.h>
#include <sys/sysmacros.h>
#include <unistd.h>
int
main(int argc, char *argv[])
{
    int          fd, userns_fd, parent_fd;
    struct stat  sb;
    if (argc < 2) {
        fprintf(stderr, "الاستخدام: %s /proc/[pid]/ns/[file] [p|u]\n", argv[0]);
        fprintf(stderr, "\nيعرض نتيجة عملية واحدة أو كليهما NS_GET_USERNS (u) أو NS_GET_PARENT (p) للملف المحدد في /proc/[pid]/ns/[file]. إذا لم يتم تحديد 'p' أو 'u'، فإن الافتراضي هو NS_GET_USERNS.\n");
        exit(EXIT_FAILURE);
    }
    /* الحصول على واصف ملف للملف 'ns' المحدد في argv[1]. */
    fd = open(argv[1], O_RDONLY);
    if (fd == -1) {
        perror("open");
        exit(EXIT_FAILURE);
    }
    /* الحصول على واصف ملف لمساحة أسماء المستخدم ثم الحصول على رقم inode لهذه المساحة وعرضه. */
    if (argc < 3 || strchr(argv[2], 'u')) {
        userns_fd = ioctl(fd, NS_GET_USERNS);
        if (userns_fd == -1) {
            if (errno == EPERM)
                printf("مساحة أسماء المستخدم المشار إليها خارج نطاق مساحة أسماءك\n");
            else
                perror("ioctl-NS_GET_USERNS");
            exit(EXIT_FAILURE);
        }
        if (fstat(userns_fd, &sb) == -1) {
            perror("fstat-userns");
            exit(EXIT_FAILURE);
        }
        printf("رقم جهاز/inode مساحة أسماء المستخدم هو: [%x,%x] / %ju\n", major(sb.st_dev), minor(sb.st_dev), (uintmax_t) sb.st_ino);
        close(userns_fd);
    }
    /* الحصول على واصف ملف لمساحة الأسماء الأصل ثم الحصول على رقم inode لهذه المساحة وعرضه. */
    if (argc > 2 && strchr(argv[2], 'p')) {
        parent_fd = ioctl(fd, NS_GET_PARENT);
        if (parent_fd == -1) {
            if (errno == EINVAL)
                printf("لا يمكن الحصول على مساحة الأسماء الأصل من مساحة أسماء غير هرمية\n");
            else if (errno == EPERM)
                printf("مساحة الأسماء الأصل خارج نطاق مساحة أسماءك\n");
            else
                perror("ioctl-NS_GET_PARENT");
            exit(EXIT_FAILURE);
        }
        if (fstat(parent_fd, &sb) == -1) {
            perror("fstat-parentns");
            exit(EXIT_FAILURE);
        }
        printf("رقم جهاز/inode مساحة الأسماء الأصل هو: [%x,%x] / %ju\n", major(sb.st_dev), minor(sb.st_dev), (uintmax_t) sb.st_ino);
        close(parent_fd);
    }
    exit(EXIT_SUCCESS);
}
