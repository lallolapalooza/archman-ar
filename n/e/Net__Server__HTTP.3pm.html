Net__Server__HTTP(3pm)

الحزمة: perl-net-server

التراخيص: PerlArtistic, GPL

Net::Server::HTTP(3) وثائق حزم Perl المساهمة Net::Server::HTTP(3)

الاسم

Net::Server::HTTP - فئة أساسية لخادم HTTP يعتمد على Net::Server

اختبار سطر واحد

perl -e 'use base qw(Net::Server::HTTP); main->run(port => 8080)'
# سيقوم بتشغيل خادم echo

الملخص

use base qw(Net::Server::HTTP);
__PACKAGE__->run;
sub process_http_request {
    my $self = shift;
    print "Content-type: text/html\n\n";
    print "<form method=post action=/bam><input type=text name=foo><input type=submit></form>";
    require Data::Dumper;
    local $Data::Dumper::Sortkeys = 1;
    require CGI;
    my $form = {};
    my $q = CGI->new; $form->{$_} = $q->param($_) for $q->param;
    print "<pre>".Data::Dumper->Dump([\%ENV, $form], ['*ENV', 'form'])."</pre>";
}

الوصف

على الرغم من أن Net::Server::HTTP غير موجود في الموازاة الطبيعية لنكهات Net::Server الأخرى، فإن معالجة طلبات HTTP هي ميزة مطلوبة بشكل شائع وهي بروتوكول قياسي وبسيط.

تبدأ Net::Server::HTTP بنكهة MultiType الأساسية افتراضيًا والتي تتوافق مع Net::Server::PreFork. من السهل تغييرها إلى أي نكهة Net::Server أخرى عن طريق تمرير server_type => $other_flavor في تكوين الخادم. تم تعيين المنفذ أيضًا افتراضيًا إلى 80 - ولكن يمكن تغييره بسهولة إلى منفذ آخر عبر تكوين الخادم. يمكنك أيضًا إضافة SSL بسهولة عن طريق تضمين proto => "ssl" وتوفير SSL_cert_file و SSL_key_file.

على سبيل المثال، إليك خادم أساسي سيقوم بالاتصال بجميع الواجهات، والتحدث بكل من HTTP على المنفذ 8080 و HTTPS على 8443، والتحدث بكل من IPv4 و IPv6 إذا كان متاحًا.

use base qw(Net::Server::HTTP);
__PACKAGE__->run(
    port  => [8080, "8443/ssl"],
    ipv   => '*', # IPv6 إذا كان متاحًا
    SSL_key_file  => '/my/key',
    SSL_cert_file => '/my/cert',
);

الأساليب

"_init_access_log"
يستخدم لفتح وتهيئة أي سجل وصول (انظر access_log_file و access_log_format).
"_tie_client_stdout"
يستخدم لتهيئة تحليل رؤوس الاستجابة الآلي.
"process_http_request"
يتم تمرير مقبض العميل إليه ويتم ربط STDIO به. خلال هذا الأسلوب، سيتم تكوين %ENV بالفعل ببيئة CGI قياسية. يجب عليك التأكد من طباعة رأس Content-type. هذا اختلاف جوهري مقارنة بنكهات Net::Server القياسية الأخرى.
"process_request"
تم تجاوز هذا الأسلوب في Net::Server::HTTP - لا يجب عليك استخدامه أثناء استخدام Net::Server::HTTP. يقوم هذا الأسلوب المتجاوز بتحليل البيئة وتكوين تنبيهات الطلب ومعالجة الفشل. يستدعي process_http_request بمجرد أن يصبح الطلب جاهزًا وتم تحليل الرؤوس.
"request_denied_hook"
تم تجاوز هذا الأسلوب لاستدعاء send_400. إنه جديد. للحصول على السلوك السابق (حيث يتم إغلاق العميل بدون مؤشر)، ما عليك سوى توفير
"process_headers"
يستخدم لقراءة الرؤوس الواردة وتكوين %ENV.
"_init_http_request_info"
يتم استدعاؤه في نهاية process_headers. يقوم بتهيئة محتويات http_request_info.
"http_request_info"
يعيد hashref من المعلومات الخاصة بالطلب الحالي. سيتم استخدام هذه المعلومات للتسجيل لاحقًا.
"send_status"
يتلقى حالة HTTP، ورسالة اختيارية، وجسم اختياري، وعلامة اختيارية generate_body. يرسل الرؤوس الصحيحة. 
$self->send_status(500);
$self->send_status(500, 'Internal Server Error');
$self->send_status(500, 'Internal Server Error', "<h1>Internal Server Error</h1><p>Msg</p>");
$self->send_status(500, undef, undef, ['Msg']);
"send_400"
يستدعي send_status مع 400 والحجج المارة كـ generate_body.
"send_500"
يستدعي send_status مع 500 والحجة المارة إلى send_500.
c<log_http_request>
يتم استدعاؤه في نهاية post_process_request. يبحث الأسلوب الافتراضي عن access_log_format الافتراضي ويتحقق مما إذا كان السجل قد تم تهيئته أثناء _init_access_log. إذا كان كلاهما موجودًا، فسيتم تشكيل http_request_info باستخدام http_log_format وسيتم تسجيل النتيجة.
"http_log_format"
يتلقى سلسلة تنسيق، و request_info، ويعيد سلسلة منسقة. يجب أن يتبع التنسيق مواصفات mod_log_config الخاصة بـ Apache. كما في مواصفات mod_log_config، يجب الهروب من الشرطات المائلة للخلف وعلامات الاقتباس بشرطات مائلة للخلف، ويمكنك أيضًا تضمين أحرف \n و \t.

فيما يلي قائمة بالمعلمات المتاحة بالإضافة إلى مثال للمخرجات بناءً على خادم HTTP بسيط جدًا.

%%                %                 # نسبة مئوية
%a                ::1               # IP عن بعد
%A                ::1               # IP محلي
%b                83                # حجم الاستجابة (- إذا كان 0) تنسيق السجل الشائع
%B                83                # حجم الاستجابة
%{bar}C           baz               # قيمة ملف تعريف الارتباط بهذا الاسم
%D                916               # الوقت المنقضي بالميكروثانية
%{HTTP_COOKIE}e   bar=baz           # قيمة %ENV بهذا الاسم
%f                -                 # اسم الملف - غير مستخدم
%h                ::1               # المضيف البعيد إذا كانت الاستعلامات نشطة، IP البعيد بخلاف ذلك
%H                http              # طريقة الطلب
%{Host}i          localhost:8080    # قيمة الطلب بهذا الاسم
%I                336               # البايتات المستلمة بما في ذلك الرؤوس
%l                -                 # اسم تسجيل الدخول البعيد - غير مستخدم
%m                GET               # طريقة الطلب
%n                Solo un noticia   # ملاحظة %ENV بهذا الاسم
%{Content-type}o  text/html         # رأس الإخراج بهذا الاسم
%O                189               # حجم الاستجابة بما في ذلك الرؤوس
%p                8080              # منفذ الخادم
%P                22999             # PID - لا يدعم %{tid}P
q                 ?hello=there      # سلسلة الاستعلام بما في ذلك ؟ (- بخلاف ذلك)
r                 GET /bam?hello=there HTTP/1.1      # السطر الأول للطلب
%s                200               # رمز حالة الاستجابة
%u                -                 # المستخدم البعيد - غير مستخدم
%U                /bam              # مسار الطلب (بدون سلسلة استعلام)
%t                [06/Jun/2012:12:14:21 -0600]       # تنسيق سجل الوقت القياسي http_log_time
%t{%F %T %z}t     [2012-06-06 12:14:21 -0600]        # تنسيق الوقت المنسق
%T                0                 # الوقت المنقضي بالثواني
%v                localhost:8080    # vhost http_log_vhost - تطبيق جزئي
%V                localhost:8080    # vhost http_log_vhost - تطبيق جزئي
%X                -                 # تم إنهاء الاتصال وهو 'close' (-)

بالإضافة إلى ذلك، يسمح تحليل السجلات بالتنسيق التالي:

%>s               200               # حالة آخر طلب
%<s               200               # حالة الطلب الأصلي
%400a             -                 # IP عن بعد إذا كانت الحالة 400
%!400a            ::1               # IP عن بعد إذا لم تكن الحالة 400
%!200a            -                 # IP عن بعد إذا لم تكن الحالة 200

هناك عدد قليل من الأجزاء غير المطبقة بالكامل:

> y <    # لا يوجد إعادة توجيه داخلية
%I         # الاستجابة تستند إلى حجم الرأس و Content-length، وليس عدد البايتات المقروءة الصحيح على الرغم من أنه في الحالات الشائعة ستكون هذه هي نفسها.
%X         # لا يوجد keepalive في الخادم افتراضيًا.
%v y %V  # لا توجد vhosts في خادم HTTP افتراضيًا.
%{tid}P    # الخوادم الافتراضية ليست متعددة الخيوط.

انظر خيار "access_log_format" لمعرفة كيفية تكوين تنسيق مختلف وكذلك لرؤية السلسلة الافتراضية.
