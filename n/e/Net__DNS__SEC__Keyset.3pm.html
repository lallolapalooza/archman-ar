Net__DNS__SEC__Keyset(3pm)

الحزمة: perl-net-dns-sec

التراخيص: PerlArtistic, GPL

Net::DNS::SEC::Keyset(3) توثيق بييرل مقدم من المستخدم Net::DNS::SEC::Keyset(3)

الاسم

Net::DNS::SEC::Keyset - فئة كائن مجموعة المفاتيح DNSSEC

الملخص

use Net::DNS::SEC::Keyset;

الوصف

مجموعة المفاتيح هي وحدة "إدارية" تستخدم لصيانة DNSSEC.

توفر هذه الفئة واجهات لإنشاء وقراءة وكتابة مجموعات المفاتيح.

يتم توفير أساليب الكائن لاستخراج سجلات DNSKEY و RRSIG و DS.

لاحظ أن هذه الفئة لا تزال قيد التطوير. السمات والأساليب عرضة للتغيير.

جديد (من ملف)

$keyset = Net::DNS::SEC::Keyset->new( $filename );
$keyset = Net::DNS::SEC::Keyset->new( $filename, $directory );
die Net::DNS::SEC::Keyset->keyset_err unless $keyset;

أسلوب المنشئ الذي يقرأ ملف مجموعة المفاتيح المحدد ويعيد كائن مجموعة مفاتيح.

يحدد الوسيط الاختياري الثاني دليل قاعدة اسم الملف.

يضبط keyset_err ويعيد قيمة فارغة عند الفشل.

جديد (بواسطة مفاتيح التوقيع)

$keyset = Net::DNS::SEC::Keyset->new( [@keyrr], $privatekeypath );
die Net::DNS::SEC::Keyset->keyset_err unless $keyset;

ينشئ كائن مجموعة مفاتيح من المفاتيح المقدمة عبر المرجع إلى مصفوفة من كائنات Net::DNS::RR::DNSKEY.

سيقوم الأسلوب بإنشاء مجموعة المفاتيح بأكملها وتوقيعها ذاتيًا. يُفترض أن تكون المفاتيح الخاصة كما تم إنشاؤها بواسطة أداة BIND dnssec-keygen في الدليل الحالي أو، إذا تم تحديدها، الدليل المشار إليه بواسطة $privatekeypath.

يضبط keyset_err ويعيد قيمة فارغة عند الفشل.

جديد (من مجموعات سجلات المفاتيح والتوقيع)

$keyset = Net::DNS::Keyset->new( [@keyrr], [@sigrr] );
die Net::DNS::SEC::Keyset->keyset_err unless $keyset;

ينشئ كائن مجموعة مفاتيح من المفاتيح المقدمة عبر المراجع إلى مصفوفات من كائنات Net::DNS::RR::DNSKEY و Net::DNS::RR::RRSIG.

يضبط keyset_err ويعيد قيمة فارغة عند الفشل.

جديد (من الحزمة)

$resolver = Net::DNS::Resolver->new;
$resolver->dnssec(1);

$reply = $res->send ( "example.com", "DNSKEY" );
$keyset = Net::DNS::SEC::Keyset->new( $reply );
die Net::DNS::SEC::Keyset->keyset_err unless $keyset;

ينشئ كائن مجموعة مفاتيح من حزمة Net::DNS::Packet تحتوي على إجابة لاستعلام عن سجلات المفاتيح في قمة المنطقة.

هذا هو الأسلوب الذي يجب عليك استخدامه لجلب المفاتيح تلقائيًا.

يضبط keyset_err ويعيد قيمة فارغة عند الفشل.

المفاتيح

@keyrr = $keyset->keys;

يعيد مصفوفة من كائنات Net::DNS::RR::DNSKEY.

التوقيعات

@sigrr = $keyset->sigs;

يعيد مصفوفة من كائنات Net::DNS::RR::RRSIG.

استخراج DS

@ds = $keyset->extract_ds();        # افتراضي SHA-1
@ds = $keyset->extract_ds( digtype => 'SHA-256' );
die Net::DNS::SEC::Keyset->keyset_err unless @ds;

يستخرج سجلات DS من مجموعة المفاتيح. لاحظ أنه سيتم التحقق من صحة مجموعة المفاتيح أثناء الاستخراج. ستكون جميع المفاتيح بحاجة إلى توقيع ذاتي صالح.

يضبط الأسلوب keyset_err إذا فشل التحقق.

التحقق

@keytags = $keyset->verify();
die Net::DNS::SEC::Keyset->keyset_err unless @keytags;
$keyset->verify( $keytag ) || die $keyset->keyset_err;

إذا لم يتم تقديم وسيطات:

  • يتحقق مما إذا كانت جميع التوقيعات الموجودة تحقق مجموعة المفاتيح.
  • يتحقق مما إذا كانت هناك مفاتيح DNSKEY مع تعيين علم SEP، فهناك على الأقل توقيع RRSIG واحد تم إنشاؤه باستخدام هذا المفتاح.
  • يتحقق من أنه إذا لم تكن هناك مفاتيح DNSKEY مع تعيين علم SEP، فهناك على الأقل توقيع RRSIG واحد تم إنشاؤه بأحد المفاتيح من مجموعة المفاتيح.

إذا تم تقديم وسيطة، فيجب أن تكون علامة المفتاح الرقمية للمفتاح في مجموعة المفاتيح التي سيتم التحقق منها باستخدام RRSIG المقابل.

يعيد الأسلوب قائمة بعلامات المفاتيح للمفاتيح التي تم التحقق منها في مجموعة المفاتيح.

يضبط الأسلوب keyset_err ويعيد قائمة فارغة إذا فشل التحقق.

مفتاح_خطأ

$keyset_err = Net::DNS::SEC::Keyset->keyset_err;

يعيد سلسلة خطأ مجموعة المفاتيح.

سلسلة

$string = $keyset->string;

يعيد تمثيل سلسلة لمجموعة المفاتيح.

اطبع

$keyset->print;             # يشبه print( $keyset->string )

يطبع مجموعة المفاتيح.

كتابة_مجموعة_المفاتيح

$keyset->writekeyset;
$keyset->writekeyset( $path );
$keyset->writekeyset( $prefix );
$keyset->writekeyset( $prefix, $path );

يكتب مجموعة المفاتيح إلى ملف يسمى "keyset-<domain>." في دليل العمل الحالي أو الدليل المحدد بواسطة الوسيط الاختياري $path.

يحدد الوسيط الاختياري $prefix البادئة التي سيتم إلحاقها باسم المجال لتشكيل اسم ملف مجموعة المفاتيح.
