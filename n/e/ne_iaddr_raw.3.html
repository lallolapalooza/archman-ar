ne_iaddr_raw(3)

الحزمة: neon

التراخيص: LGPL-2.0-or-later

NE_IADDR_MAKE(3) مرجع واجهة برمجة تطبيقات neon NE_IADDR_MAKE(3)

الاسم

ne_iaddr_make, ne_iaddr_cmp, ne_iaddr_print, ne_iaddr_typeof, ne_iaddr_parse, ne_iaddr_raw, ne_iaddr_reverse, ne_iaddr_free - دوال لمعالجة عناوين الشبكة

الملخص

#include <ne_socket.h>
typedef enum {
    ne_iaddr_ipv4 = 0,
    ne_iaddr_ipv6
} ne_iaddr_type;

ne_inet_addr *ne_iaddr_make(ne_iaddr_type type, const unsigned char *raw);

int ne_iaddr_cmp(const ne_inet_addr *ia1, const ne_inet_addr *ia2);

char *ne_iaddr_print(const ne_inet_addr *ia, char *buffer, size_t bufsiz);

ne_iaddr_type ne_iaddr_typeof(const ne_inet_addr *ia);

ne_inet_addr *ne_iaddr_parse(const char *address, ne_iaddr_type type);

unsigned char *ne_iaddr_raw(const ne_inet_addr *ia, unsigned char *buffer);

int ne_iaddr_reverse(const ne_inet_addr *ia, char *buffer, size_t buflen);

void ne_iaddr_free(const ne_inet_addr *ia);

الوصف

ne_iaddr_make تنشئ كائن ne_inet_addr من عنوان شبكة ثنائي خام؛ على سبيل المثال، تمثل البايتات الأربعة 0x7f 0x00 0x00 0x01 عنوان IPv4 127.0.0.1. الكائن المُعاد مناسب لتمريره إلى ne_sock_connect. يحتوي عنوان IPv4 الثنائي على أربعة بايتات؛ وعنوان IPv6 الثنائي يحتوي على ستة عشر بايتًا؛ يجب أن تكون العناوين المُمررة بترتيب البايتات الشبكي.

ne_iaddr_cmp تقارن كائنين من عناوين الشبكة؛ تعيد صفرًا فقط إذا كانا متطابقين. لا يلزم أن يكون للكائنات نفس نوع العنوان؛ إذا لم تكن العناوين من نفس النوع، فإن القيمة المُعادة مضمونة أن تكون غير صفرية.

ne_iaddr_print تطبع تمثيلًا نصيًا قابلًا للقراءة البشرية لعنوان شبكة في مخزن مؤقت، على سبيل المثال النص "127.0.0.1".

ne_iaddr_typeof تعيد نوع كائن عنوان الشبكة المُعطى.

ne_iaddr_parse تحلل تمثيلًا نصيًا لعنوان شبكة (مثل "127.0.0.1") وتنشئ كائن عنوان شبكة لتمثيل العنوان المُحلل.

ne_iaddr_raw تكتب التمثيل الثنائي الخام لعنوان شبكة إلى المخزن المؤقت المُقدم. تُكتب البايتات بترتيب البايتات الشبكي؛ يجب أن يكون المخزن المؤقت بطول مناسب لنوع العنوان (4 بايتات لعنوان IPv4، 16 بايتًا لعنوان IPv6).

ne_iaddr_reverse تقوم ببحث اسم عكسي على كائن العنوان، وتكتب اسم المضيف (الأول) المرتبط بعنوان IP إلى المخزن المؤقت المُقدم. إذا كان اسم المضيف أطول من المخزن المؤقت، فسيتم اقتطاعه بصمت؛ عند النجاح، يكون النص المكتوب في المخزن المؤقت دائمًا مُنتهياً بـ NULL.

ne_iaddr_free تحرر الذاكرة المرتبطة بكائن عنوان الشبكة.

قيمة الإرجاع

ne_iaddr_make تعيد NULL إذا كان نوع العنوان المُمرر غير مدعوم (على سبيل المثال، على منصة لا تدعم IPv6).

ne_iaddr_print تعيد مؤشر buffer، ولا تعيد NULL أبدًا.

ne_iaddr_parse تعيد كائن عنوان شبكة عند النجاح، أو NULL عند الفشل في تحليل المعامل address.

ne_iaddr_reverse تعيد صفرًا عند النجاح أو قيمة غير صفرية إذا لم يكن هناك اسم مضيف مرتبط بالعنوان.

ne_iaddr_raw تعيد المعامل buffer، ولا تعيد NULL أبدًا.

أمثلة

يوصل المثال التالي مقبسًا بالمنفذ 80 على العنوان 127.0.0.1.

unsigned char addr[] = "\0x7f\0x00\0x00\0x01";
ne_inet_addr *ia;
ia = ne_iaddr_make(ne_iaddr_ipv4, addr);
if (ia != NULL) {
    ne_socket *sock = ne_sock_connect(ia, 80);
    ne_iaddr_free(ia);
    /* ... */
} else {
    /* ... */
}
