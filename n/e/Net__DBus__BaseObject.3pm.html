Net__DBus__BaseObject(3pm)

الحزمة: perl-net-dbus

التراخيص: GPL, PerlArtistic

Net::DBus::BaseObject(3) توثيق الحزم المساهمة في Perl Net::DBus::BaseObject(3)

الاسم

Net::DBus::BaseObject - فئة أساسية لتصدير الكائنات إلى الناقل

الملخص

# لنكن كائن DBus
use base qw(Net::DBus::BaseObject);
# تصدير إشارة 'Greeting' مع معامل من نوع string
dbus_signal("Greeting", ["string"]);
# تصدير 'Hello' كدالة تقبل معاملًا من نوع string
# وتعيد قيمة من نوع string
dbus_method("Hello", ["string"], ["string"]);
sub new {
    my $class = shift;
    my $service = shift;
    my $self = $class->SUPER::new($service, "/org/demo/HelloWorld");
    bless $self, $class;
    return $self;
}
sub _dispatch_object {
    my $self = shift;
    my $connection = shift;
    my $message = shift;
    if (....$message refers to an object method...) {
       ... process this object's interfaces/methods...
       return $reply;
    }
}

الوصف

هذه هي قاعدة لجميع الكائنات التي يتم تصديرها إلى ناقل الرسائل. يوفر الدعم الأساسي لفحص الأنواع الضروري للكائنات المصدرة إلى الناقل. عند الوراثة من هذه الفئة، يجب تنفيذ طريقة "_dispatch" لمعالجة الرسائل الواردة. يتم استخدام الوحدة Net::DBus::Exporter للإعلان عن الطرق (والإشارات) التي يتم تصديرها إلى الناقل.

ستقوم جميع الحزم التي ترث من هذه الفئة بتسجيل الواجهة "org.freedesktop.DBus.Introspectable" تلقائيًا مع Net::DBus::Exporter، وسيتم تصدير الطريقة "Introspect" الموجودة بداخلها.

نادرًا ما يستخدم مطورو التطبيقات هذه الفئة مباشرة، وبدلاً من ذلك، تعتبر Net::DBus::Object أو "Net::DBus::ProxyObject" خيارات شائعة. سيتم استخدام هذه الفئة فقط إذا كنت ترغب في كتابة نهج جديد لتوزيع استدعاءات الطرق الواردة.

الطرق

my $object = Net::DBus::BaseObject->new($service, $path)
ينشئ هذا كائن DBus جديدًا بمسار $path مسجلاً ضمن الخدمة $service. يجب أن يكون المعامل $path سلسلة تفي بمتطلبات مسارات الكائنات العادية في DBus، بينما يجب أن يكون المعامل $service كائنًا من نوع Net::DBus::Service. يتم الحصول على الأخير عادةً عن طريق استدعاء الطريقة "export_service" على كائن Net::DBus.
my $object = Net::DBus::BaseObject->new($parentobj, $subpath)
ينشئ هذا كائن ابن جديدًا بمسار $subpath بالنسبة للكائن الأب $parentobj. يجب أن يكون المعامل $subpath سلسلة تفي بمتطلبات مسارات الكائنات العادية في DBus، بينما يجب أن يكون المعامل $parentobj كائنًا من نوع Net::DBus::BaseObject.
$object->disconnect()
تقوم هذه الطريقة بفصل الكائن عن الناقل، لذلك لن يتلقى أي رسائل أخرى مرسلة من عملاء آخرين. سيتم فصل أي كائنات ابن بشكل متكرر أيضًا. بعد فصل الكائن، قد يتمكن Perl من جمع الكائن كقمامة. كما أنها ستسمح بربط كائن جديد بنفس المسار.
my $bool = $object->is_connected
يعيد قيمة صحيحة إذا كان الكائن متصلاً بالناقل وبالتالي يمكن الوصول إليه من قبل العملاء البعيدين. يعيد قيمة خاطئة إذا كان الكائن مفصولًا وبالتالي جاهزًا لجمع القمامة. تبدأ جميع الكائنات في حالة متصلة ولن تتحول إلى حالة مفصولة إلا إذا تم استدعاء الطريقة "disconnect".
my $service = $object->get_service
الحصول على كائن Net::DBus::Service الذي تم تصدير هذا الكائن عليه.
my $path = $object->get_object_path
الحصول على المسار الذي تم تصدير هذا الكائن تحته.
$object->emit_signal_in($name, $interface, $recipient, @args);
يرسل إشارة من الكائن بالاسم $name. إذا كان المعامل $interface محددًا، فسيتم تربيع الإشارة ضمن هذا الواجهة. إذا كان المعامل $recipient محددًا، فسيتم إرسال الإشارة كـ unicast إلى هذا العميل على الناقل. يجب تسجيل أنواع الإشارة وأنواع المعاملات @args مع Net::DBus::Exporter عن طريق استدعاء الدالة "dbus_signal".
$self->emit_signal_to($name, $recipient, @args);
يرسل إشارة من الكائن بالاسم $name. يجب تسجيل أنواع الإشارة وأنواع المعاملات @args مع Net::DBus::Exporter عن طريق استدعاء الدالة "dbus_signal". سيتم إرسال الإشارة فقط إلى العميل المحدد بالمعامل $recipient.
$self->emit_signal($name, @args);
يرسل إشارة من الكائن بالاسم $name. يجب تسجيل أنواع الإشارة وأنواع المعاملات @args مع Net::DBus::Exporter عن طريق استدعاء الدالة "dbus_signal". سيتم إرسال الإشارة إلى جميع عملاء الناقل.
$object->connect_to_signal_in($name, $interface, $coderef);
يتصل بـ callback لإشارة مرسلة من الكائن. المعامل $name هو اسم الإشارة داخل الكائن، و $coderef هو مرجع إلى روتين فرعي مجهول. عند إرسال الإشارة $name من الكائن البعيد، سيتم استدعاء الروتين الفرعي $coderef وسيتم تمرير معاملات الإشارة. يستخدم المعامل $interface لتحديد الواجهة الصريحة المتصلة بالإشارة.
$object->connect_to_signal($name, $coderef);
يتصل بـ callback لإشارة مرسلة من الكائن. المعامل $name هو اسم الإشارة داخل الكائن، و $coderef هو مرجع إلى روتين فرعي مجهول. عند إرسال الإشارة $name من الكائن البعيد، سيتم استدعاء الروتين الفرعي $coderef وسيتم تمرير معاملات الإشارة.
$reply = $object->_dispatch_object($connection, $message);
تستخدم الطريقة "_dispatch_object" لمعالجة توزيع الطرق التي تنفذها الكائن. التطبيق الافتراضي هو NOP ويجب أن يتم تجاوزه من قبل الفئات المشتقة. إذا كان يمكن معالجة $message، فسيتم إرجاع مثيل "Net::DBus::Binding::Message" آخر للرد. إذا تم إرجاع "undef"، فسيتم إرجاع خطأ عام إلى المتصل.
$currvalue = $object->_dispatch_property($name); =item $object->_dispatch_property($name, $newvalue);
تستخدم الطريقة "_dispatch_property" لمعالجة قراءة وكتابة الخصائص. المعامل $name هو اسم الخاصية التي يتم الوصول إليها. إذا تم توفير $newvalue، فسيتم تحديث الخاصية، وإلا فسيتم إرجاع القيمة الحالية. التطبيق الافتراضي سيطلق خطأ، لذلك يجب تجاوزه في الفئات المشتقة.
