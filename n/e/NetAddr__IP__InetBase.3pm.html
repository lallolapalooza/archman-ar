الاسم

NetAddr::IP::InetBase -- أدوات مساعدة لـ IPv4 و IPV6

الملخص

use NetAddr::IP::Base qw(
      :upper
      inet_aton
      inet_ntoa
      ipv6_aton
      ipv6_ntoa
      ipv6_n2x
      ipv6_n2d
      inet_any2n
      inet_n2dx
      inet_n2ad
      inet_pton
      inet_ntop
      packzeros
      isIPv4
      isNewIPv4
      isAnyIPv4
      AF_INET
      AF_INET6
      fake_AF_INET6
      fillIPv4
);
use NetAddr::IP::Util qw(:all :inet :ipv4 :ipv6 :math)
:ipv4   =>    inet_aton, inet_ntoa, fillIPv4
:ipv6   =>    ipv6_aton, ipv6_ntoa,ipv6_n2x, ipv6_n2d,
              inet_any2n, inet_n2dx, inet_n2ad
              inet_pton, inet_ntop, packzeros
$dotquad = inet_ntoa($netaddr);
$netaddr = inet_aton($dotquad);
$ipv6naddr = ipv6_aton($ipv6_text);
$ipv6_text = ipv6_ntoa($ipv6naddr);
$hex_text = ipv6_n2x($ipv6naddr);
$dec_text = ipv6_n2d($ipv6naddr);
$ipv6naddr = inet_any2n($dotquad or $ipv6_text);
$dotquad or $hex_text = inet_n2dx($ipv6naddr);
$dotquad or $dec_text = inet_n2ad($ipv6naddr);
$netaddr = inet_pton($AF_family,$text_addr);
$text_addr = inet_ntop($AF_family,$netaddr);
$hex_text = packzeros($hex_text);
$rv = isIPv4($bits128);
$rv = isNewIPv4($bits128);
$rv = isAnyIPv4($bits128);
$constant = AF_INET();
$constant = AF_INET6();
$trueif   = fake_AF_INET6();
$ip_filled = fillIPv4($shortIP);
NetAddr::IP::InetBase::lower();
NetAddr::IP::InetBase::upper();

الوصف

NetAddr::IP::InetBase يوفر مجموعة من دوال الشبكة للتحويل مكتوبة بالكامل بلغة Perl لتحويل كل من عناوين IPv4 و IPv6 من وإلى تنسيق عنوان الشبكة وتنسيق النص.

تدعم دوال IPv6 جميع تنسيقات rfc1884.

على سبيل المثال:  x:x:x:x:x:x:x:x:x
      x:x:x:x:x:x:x:d.d.d.d
      ::x:x:x
      ::x:d.d.d.d
وما إلى ذلك...
  • $dotquad = inet_ntoa($netaddr);

    تحويل عنوان شبكة IPv4 المعبأ إلى عنوان IP بنقطة رباعية.

    الإدخال:        عنوان شبكة معبأ
يعيد:      عنوان IP، على سبيل المثال 10.4.12.123
  • $netaddr = inet_aton($dotquad);

    تحويل عنوان IP بنقطة رباعية إلى عنوان شبكة IPv4 معبأ.

    الإدخال:        عنوان IP، على سبيل المثال 192.5.16.32
يعيد:      عنوان شبكة معبأ
  • $ipv6addr = ipv6_aton($ipv6_text);

    تأخذ عنوان IPv6 بالشكل الموضح في rfc1884 وتعيد سلسلة بيانات ثنائية 128 بت.

    الإدخال:        نص IPv6
يعيد:      سلسلة بيانات ثنائية 128 بت
  • $ipv6text = ipv6_ntoa($ipv6naddr);

    تحويل عنوان IPv6 الثنائي 128 بت إلى تمثيل نصي مضغوط rfc 1884.

    الإدخال:        سلسلة بيانات ثنائية 128 بت
يعيد:      نص IPv6
  • $hex_text = ipv6_n2x($ipv6addr);

    تأخذ سلسلة بيانات IPv6 وتعيد عنوان IPv6 سداسي عشري من 8 أجزاء

    الإدخال:        سلسلة بيانات ثنائية 128 بت
يعيد:      x:x:x:x:x:x:x:x
ملاحظة: هذه الدالة لا تقوم بضغط السلاسل المتجاورة
من 0:0:0:0 إلى تنسيق ::
  • $dec_text = ipv6_n2d($ipv6addr);

    تأخذ سلسلة بيانات IPv6 وتعيد عنوان IPv6 سداسي عشري مختلط - عشري مع الـ 6 أجزاء العليا بالسداسي عشري والـ 32 بت السفلى بتنسيق النقطة الرباعية.

    الإدخال:        سلسلة بيانات ثنائية 128 بت
يعيد:      x:x:x:x:x:x:d.d.d.d
ملاحظة: هذه الدالة لا تقوم بضغط السلاسل المتجاورة
من 0:0:0:0 إلى تنسيق ::
  • $ipv6naddr = inet_any2n($dotquad or $ipv6_text);

    هذه الدالة تحول عنوان IPv4 أو IPv6 نصي بأي تنسيق قياسي إلى سلسلة عنوان IPv6 ثنائية 128 بت. تقوم بإضافة البادئة '::' لأي عنوان نقطة رباعية (إذا تم العثور عليه) وتمرره إلى ipv6_aton.

    الإدخال:        نقطة رباعية أو عنوان rfc1844
يعيد:      سلسلة IPv6 ثنائية 128 بت
  • $dotquad or $hex_text = inet_n2dx($ipv6naddr);

    هذه الدالة **تقوم بالشيء الصحيح** وتعيد النص إما لعنوان IPv4 بنقطة رباعية أو لعنوان IPv6 بتنسيق سداسي عشري.

    الإدخال:        سلسلة IPv6 ثنائية 128 بت
يعيد:      ddd.ddd.ddd.ddd
          أو  x:x:x:x:x:x:x:x
ملاحظة: هذه الدالة لا تقوم بضغط السلاسل المتجاورة
من 0:0:0:0 إلى تنسيق ::
  • $dotquad or $dec_text = inet_n2ad($ipv6naddr);

    هذه الدالة **تقوم بالشيء الصحيح** وتعيد النص إما لعنوان IPv4 بنقطة رباعية أو لعنوان IPv6 بتنسيق سداسي عشري::عشري.

    الإدخال:        سلسلة IPv6 ثنائية 128 بت
يعيد:      ddd.ddd.ddd.ddd
          أو  x:x:x:x:x:x:ddd.ddd.ddd.dd
ملاحظة: هذه الدالة لا تقوم بضغط السلاسل المتجاورة
من 0:0:0:0 إلى تنسيق ::
  • $netaddr = inet_pton($AF_family,$text_addr);

    هذه الدالة تأخذ عنوان IP بتنسيق نص IPv4 أو IPv6 وتحوله إلى تنسيق ثنائي. يتم التحكم في نوع تحويل عنوان IP بواسطة الوسيط FAMILY.

    ملاحظة: inet_pton، inet_ntop و AF_INET6 تأتي من مكتبة Socket6 إذا كانت موجودة على هذا المضيف.

  • $text_addr = inet_ntop($AF_family,$netaddr);

    هذه الدالة تأخذ عنوان IP بتنسيق ثنائي وتحوله إلى تنسيق نصي. يتم التحكم في نوع تحويل عنوان IP بواسطة الوسيط FAMILY.

    ملاحظة: inet_ntop تعيد دائمًا أحرفًا صغيرة.

    ملاحظة: inet_pton، inet_ntop و AF_INET6 تأتي من مكتبة Socket6 إذا كانت موجودة على هذا المضيف.

  • $hex_text = packzeros($hex_text);

    هذه الدالة تقوم بتحسين عنوان IPv6 السداسي عشري rfc 1884 لتقليل عدد السلاسل الطويلة من البتات الصفرية كما هو محدد في rfc 1884، 2.2 (2) عن طريق استبدال **::** بأول ظهور لأطول سلسلة من الأصفار في العنوان.

  • $ipv6naddr = ipv4to6($netaddr);

    تحويل عنوان شبكة ipv4 إلى عنوان شبكة ipv6.

    الإدخال:        عنوان شبكة 32 بت
يعيد:      عنوان شبكة 128 بت
  • $rv = isIPv4($bits128);

    تعيد هذه الدالة صحيحًا إذا لم تكن هناك بتات مضاءة موجودة في جزء IPv6 من سلسلة الـ 128 بت، وخطأ بخلاف ذلك.

    أي:  يجب أن يكون العنوان بالشكل - ::d.d.d.d

    ملاحظة: هذا هو عنوان IPv6 قديم ومهمل متوافق مع IPv4

  • $rv = isNewIPv4($bits128);

    تعيد هذه الدالة صحيحًا إذا كانت سلسلة IPv6 128 بت بالشكل

    ::ffff:d.d.d.d
  • $rv = isAnyIPv4($bits128);

    تعيد هذه الدالة صحيحًا إذا كانت سلسلة بتات IPv6 بالشكل

    ::d.d.d.d       أو      ::ffff:d.d.d.d
  • NetAddr::IP::InetBase::lower();

    إعادة سلاسل IPv6 بأحرف صغيرة. هذا هو الافتراضي.

  • NetAddr::IP::InetBase::upper();

    إعادة سلاسل IPv6 بأحرف كبيرة.

    يمكن ضبط الافتراضي على الأحرف الكبيرة عند تحميل الوحدة عن طريق استدعاء العلامة :upper. أي:

    use NetAddr::IP::InetBase qw( :upper );
  • $constant = AF_INET;

    تعيد هذه الدالة القيمة النظامية لـ AF_INET.

  • $constant = AF_INET6;

    AF_INET6 موجود أحيانًا في مكتبة Socket ودائمًا موجود في مكتبة Socket6. عندما لا تحتوي مكتبة Socket على AF_INET6 وعندما لا تكون Socket6 موجودة، يتم "تخمين" قيمة عنصر نائب بناءً على نظام التشغيل المضيف الأساسي. انظر fake_AF_INET6 أدناه.

    ملاحظة: inet_pton، inet_ntop و AF_INET6 تأتي من مكتبة Socket6 إذا كانت موجودة على هذا المضيف.

  • $trueif = fake_AF_INET6;

    تعيد هذه الدالة FALSE إذا تم توفير AF_INET6 بواسطة Socket أو Socket6. بخلاف ذلك، فإنها تعيد أفضل تخمين بناءً على اسم نظام التشغيل المضيف.

  • $ip_filled = fillIPv4($shortIP);

    هذه الدالة تحول عناوين IPv4 بالشكل 127.1 إلى الشكل الطويل 127.0.0.1

    إذا تم تمرير وسيطة لا تتطابق مع شكل عنوان IP إلى الدالة، يتم إعادة الوسيطة الأصلية. أي: قم بتمرير اسم مضيف أو عنوان IP قصير وستعود

هذا الموقع غير مرتبط بـ Arch Linux ولا تابع له بأي شكل من الأشكال. المحتوى هو ترجمة فقط لصفحات دليل Arch المنشورة على man.archlinux.org.

NAME

NetAddr::IP::InetBase ‐ Base class for IPv4/IPv6 address parsing.

SYNOPSIS

 use NetAddr::IP::InetBase;
 my $ip = new NetAddr::IP::InetBase;
 # $ip->inet_aton(...);
 # $ip->inet_ntoa(...);
 # ...

INSTALLATION

This module is part of the NetAddr::IP module. See the NetAddr::IP(3) man page for installation instructions.

DESCRIPTION

This is the base class for the IPv4 and IPv6 address conversion routines. Most of the methods are exported by NetAddr::IP and its subclasses. They are included here for completeness.

The methods will convert between the text representation (dotted decimal for IPv4, colon-separated for IPv6) and the packed binary representation.

The module also contains routines to determine if an IP address string is valid for the specified address family.

If a method is called with a hostname it will attempt to resolve it. The address family is determined by the call. If inet_aton is called, it will try to resolve to an IPv4 address. If ipv6_aton is called, it will try to resolve to an IPv6 address. If the resolution fails, or if the input is not a valid IP address format, it will return undef.

If a method is called with an IP address it will return the appropriate binary representation.

If a method is called with an empty string or undef it will return undef.

  • inet_aton

    Convert an IPv4 address string (dotted decimal or hostname) into a packed binary string.

  • inet_ntoa

    Convert a packed IPv4 binary string into a dotted decimal string.

  • ipv6_aton

    Convert an IPv6 address string (colon-separated or hostname) into a packed binary string.

  • ipv6_ntoa

    Convert a packed IPv6 binary string into a colon-separated string.

  • ipv6_n2x

    Convert a packed IPv6 binary string into a compressed colon-separated string.

  • ipv6_n2d

    Convert a packed IPv6 binary string into a dotted decimal string (used for IPv4-mapped IPv6 addresses).

  • inet_any2n

    Convert an IPv4 or IPv6 address string (or hostname) into a packed binary string. The address family is determined by the input.

  • inet_n2dx

    Convert a packed IPv4 or IPv6 binary string into a compressed colon-separated string.

  • inet_n2ad

    Convert a packed IPv4 or IPv6 binary string into a dotted decimal string.

  • inet_pton

    Convert an IP address string (IPv4 or IPv6) into a packed binary string. The address family is determined by the input. This is the standard POSIX function.

  • inet_ntop

    Convert a packed binary string (IPv4 or IPv6) into an IP address string. The address family is determined by the length of the packed string. This is the standard POSIX function.

  • packzeros

    Fill a packed binary string with zeros to the specified length. Used internally.

  • isIPv4

    Returns true if the string is a valid IPv4 address.

  • isNewIPv4

    Returns true if the string is a valid IPv4 address according to newer standards (e.g., allows leading zeros).

  • isAnyIPv4

    Returns true if the string is a valid IPv4 address or can be converted to one.

  • AF_INET

    Constant for the IPv4 address family.

  • AF_INET6

    Constant for the IPv6 address family.

  • fake_AF_INET6

    A constant that can be used to force IPv6 interpretation when the input might be ambiguous (e.g., a hostname that resolves to both IPv4 and IPv6).

  • fillIPv4

    Fill a packed binary string with an IPv4 address. If the input is a hostname, it will return a filled IP.
