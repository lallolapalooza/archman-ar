Net__DBus__Test__MockMessage(3pm)

الحزمة: perl-net-dbus

التراخيص: GPL, PerlArtistic

Net::DBus::Test::MockMessage(3) توثيق Perl من مساهمة المستخدم Net::DBus::Test::MockMessage(3)

الاسم

Net::DBus::Test::MockMessage - محاكاة كائن رسالة عند اختبار الوحدة

الملخص

إرسال رسالة

my $msg = new Net::DBus::Test::MockMessage;
my $iterator = $msg->iterator;
$iterator->append_byte(132);
$iterator->append_int32(14241);
$connection->send($msg);

الوصف

توفر هذه الوحدة نظيرًا "مزيفًا" لفئة Net::DBus::Binding::Message. إنها بشكل أساسي كائن رسالة وهمي بالكامل بلغة Perl يوفر نفس العقد الخاص بكائن الرسالة الحقيقي. وهي مخصصة للاستخدام داخليًا بواسطة واجهات برمجة التطبيقات للاختبار.

الطرق

my $call = Net::DBus::Test::MockMessage->new_method_call( service_name => $service, object_path => $object, interface => $interface, method_name => $name);
إنشاء رسالة تمثل استدعاء للكائن الموجود في المسار "object_path" داخل العميل الذي يمتلك الاسم المعروف المحدد بواسطة "service_name". الاسم الذي سيتم استدعاؤه هو "method_name" ضمن الواجهة المحددة بواسطة المعلمة "interface".
my $msg = Net::DBus::Test::MockMessage->new_method_return( replyto => $method_call);
إنشاء رسالة تمثل ردًا على استدعاء الطريقة المقدم في المعلمة "replyto".
my $signal = Net::DBus::Test::MockMessage->new_signal( object_path => $path, interface => $interface, signal_name => $name);
ينشئ رسالة جديدة، تمثل إشارة [سيتم] بثها بواسطة الكائن الموجود تحت المسار المحدد بواسطة المعلمة "object_path". اسم الإشارة هو "signal_name"، وهو ضمن نطاق الواجهة المحددة بواسطة المعلمة "interface".
my $msg = Net::DBus::Test::MockMessage->new_error( replyto => $method_call, name => $name, description => $description);
ينشئ رسالة جديدة، تمثل خطأ حدث أثناء معالجة كائن استدعاء الطريقة المقدم كـ "replyto". المعلمة "name" هي الاسم الرسمي لحالة الخطأ، بينما "description" هي عبارة نصية قصيرة تقدم معلومات أكثر تحديدًا عن الخطأ.
my $type = $msg->get_type
استرداد رمز نوع هذه الرسالة. القيمة المرجعة تتوافق مع أحد الثوابت الأربعة "Net::DBus::Test::MockMessage::MESSAGE_TYPE_*".
my $name = $msg->get_error_name
إرجاع الاسم الرسمي للخطأ، كما تم تمريره سابقًا عبر المعلمة "name" في المنشئ.
my $interface = $msg->get_interface
استرداد اسم الواجهة المستهدفة بواسطة هذه الرسالة، وقد تكون سلسلة فارغة إذا لم تكن هناك واجهة قابلة للتطبيق لهذه الرسالة.
my $path = $msg->get_path
استرداد مسار الكائن المرتبط بالرسالة، وقد يكون سلسلة فارغة إذا لم يكن هناك كائن قابل للتطبيق لهذه الرسالة.
my $name = $msg->get_destination
استرداد اسم الناقل الفريد أو المعروف للعميل المقصود أن يكون المستلم للرسالة. قد ترجع سلسلة فارغة إذا كانت الرسالة تبث لجميع العملاء.
my $name = $msg->get_sender
استرداد الاسم الفريد للعميل الذي يرسل الرسالة.
my $serial = $msg->get_serial
استرداد الرقم التسلسلي الفريد لهذه الرسالة. يضمن أن يكون الرقم فريدًا طالما أن الاتصال الذي تم إرسال الرسالة من خلاله لا يزال مفتوحًا. قد ترجع صفرًا، إذا لم يتم إرسال الرسالة بعد.
my $name = $msg->get_member
بالنسبة لاستدعاءات الطرق، استرداد اسم الطريقة التي سيتم استدعاؤها، بينما بالنسبة للإشارات، استرداد اسم الإشارة.
$msg->set_sender($name)
تعيين اسم العميل الذي يرسل الرسالة. يجب أن يكون الاسم هو الاسم الفريد للعميل.
$msg->set_destination($name)
تعيين اسم المستلم المقصود للرسالة. يستخدم هذا عادةً للإشارات لتحويلها من بث إلى بث أحادي.
my $iterator = $msg->iterator;
استرداد مكرر يمكن استخدامه لقراءة أو كتابة حقول الرسالة. الكائن المرجع هو مثيل للفئة "Net::DBus::Binding::Iterator".
$boolean = $msg->get_no_reply()
الحصول على العلامة التي تشير إلى ما إذا كانت الرسالة تتوقع ردًا ليتم إرساله.
$msg->set_no_reply($boolean)
تبديل العلامة التي تشير إلى ما إذا كانت الرسالة تتوقع ردًا ليتم إرساله. تتوقع جميع رسائل استدعاءات الطرق ردًا افتراضيًا. عن طريق تبديل هذه العلامة، يتم تقليل زمن انتقال الاتصال عن طريق إزالة الحاجة إلى انتظار العميل.
my @values = $msg->get_args_list
إلغاء تجميع جميع القيم في جسم الرسالة، باستخدام توقيع الرسالة لتحديد أنواع البيانات. يتم إرجاع القيم كقائمة.
$msg->append_args_list(@values)
إلحاق مجموعة من القيم بجسم الرسالة. سيتم تشفير القيم كسلسلة أو قائمة أو قاموس حسب الاقتضاء لنوع بيانات Perl الخاص بها. لتلبية احتياجات أنواع بيانات أكثر تحديدًا، يجب استخدام كائن Net::DBus::Binding::Iterator بدلاً من ذلك.
my $sig = $msg->get_signature
استرداد سلسلة تمثل توقيع النوع للقيم المعبأة في جسم الرسالة.
