Net__Server__SIG(3pm)

الحزمة: perl-net-server

التراخيص: PerlArtistic, GPL

Net::Server::SIG(3) وثائق Perl مقدمة من المستخدم Net::Server::SIG(3)

الاسم

Net::Server::SIG - adpf - معالجة الإشارات بأمان أكبر

الملخص

use Net::Server::SIG qw(register_sig check_sigs);
use IO::Select ();
use POSIX qw(WNOHANG);
my $select = IO::Select->new();
register_sig(PIPE => 'IGNORE',
             HUP  => 'DEFAULT',
             USR1 => sub { print "I got a SIG $_[0]\n"; },
             USR2 => sub { print "I got a SIG $_[0]\n"; },
             CHLD => sub { 1 while waitpid(-1, WNOHANG) > 0; },
             );
# إضافة بعض المقابض إلى select
$select->add(\*STDIN);
# حلقة لا نهائية تحاول البقاء على قيد الحياة
while (1) {
    # استخدم مهلة زمنية لمعرفة ما إذا كانت أي إشارات قد تم تمريرها إلينا
    # بينما كنا نعالج إشارة أخرى
    my @fh = $select->can_read(10);
    my $key;
    my $val;
    # هذا هو معالج الإشارات الآمن (جيد تحت الظروف غير الآمنة أيضًا)
    if (check_sigs()) {
      # أو my @sigs = check_sigs();
      next unless @fh;
    }
    my $handle = $fh[@fh];
    # افعل شيئًا مع المقبض
}

الوصف

الإشارات قبل الإصدار 5.7 من Perl كانت غير آمنة. منذ ذلك الحين، تم تنفيذ الإشارات في خوارزمية أكثر أمانًا. يوفر Net::Server::SIG توافقًا رجعيًا، مع العمل بشكل موثوق مع الإصدارات الأحدث.

باستخدام خاصية لوظيفة select()، يحاول Net::Server::SIG إصلاح مشكلة عدم الأمان. إذا كانت العملية معلقة على select()، فإن أي إشارة ستؤدي إلى تقصير select. باستخدام هذا المفهوم، يقوم Net::Server::SIG بأقل قدر ممكن من العمل (تغيير بت واحد من 0 إلى 1). ويعتمد على المعالجة الفعلية للإشارات لتحدث مباشرة بعد استدعاء select عبر وظيفة "check_sigs". انظر المثال المعروض أعلاه، وانظر أيضًا إلى البرنامج النصي sigtest.pl الموجود في دليل الأمثلة لهذا التوزيع.

الوظائف

"register_sig($SIG => \&code_ref)"
يأخذ أزواج مفتاح/قيمة حيث المفتاح هو اسم الإشارة، والوسيط هو إما مرجع للرمز، أو الكلمتين 'DEFAULT' أو 'IGNORE'. يجب استخدام وظيفة register_sig بالاقتران مع check_sigs، ومع استدعاء وظيفة select() معلق -- وإلا، ستلاحظ اختفاء الإشارة المسجلة بشكل غامض.
unregister_sig($SIG)
يأخذ اسم إشارة كوسيط. يستدعي register_sig مع اسم الإشارة هذا و 'DEFAULT' كوسائط (نفس الشيء مثل register_sig(SIG,'DEFAULT')).
check_sigs()
يتحقق مما إذا كانت أي إشارات مسجلة قد حدثت. إذا كان الأمر كذلك، فسيقوم بتشغيل مرجع الرمز المسجل لتلك الإشارة. القيمة المرجعة هي مصفوفة تحتوي على أي أسماء إشارات قد حدثت.
sig_is_registered($SIG)
يأخذ اسم إشارة ويعيد أي مرجع رمز مسجل لتلك الإشارة.
