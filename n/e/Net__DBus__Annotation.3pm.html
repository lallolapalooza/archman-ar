Net__DBus__Annotation(3pm)

الحزمة: perl-net-dbus

التراخيص: GPL, PerlArtistic

Net::DBus::Annotation(3) وثائق Perl مقدمة من المستخدم Net::DBus::Annotation(3)

الاسم

Net::DBus::Annotation - التعليقات التوضيحية لتغيير سلوك واجهات برمجة التطبيقات

الملخص

use Net::DBus::Annotation qw(:call);
my $object = $service->get_object("/org/example/systemMonitor");
# انتظر حتى يتم سرد العمليات
my $processes = $object->list_processes("someuser");
# تجاهل قائمة العمليات ببساطة، وهو أمر عديم الفائدة تقريبًا
# في هذا المثال، ولكنه مفيد إذا لم يكن للطريقة
# قيمة إرجاع
$object->list_processes(dbus_call_noreply, "someuser");
# سرد العمليات والحصول على العمل أثناء العمل
# حتى يتم إرجاع القائمة.
my $asyncreply = $object->list_processes(dbus_call_async, "someuser");
... بعد بعض الوقت ...
my $processes = $asyncreply->get_data;
# سرد العمليات، مع مهلة أقصر مدتها 10 ثوانٍ، بدلاً من
# المهلة الافتراضية 60 ثانية
my $object->list_processes(dbus_call_timeout, 10 * 1000, "someuser");

الوصف

توفر هذه الوحدة عددًا من التعليقات التوضيحية التي ستكون مفيدة عند التعامل مع واجهات برمجة تطبيقات DBus. هناك تعليقات توضيحية للتبديل بين استدعاءات الواجهة عن بعد في وضع المزامنة، وغير المتزامنة، وعدم الرد. قد تتم إضافة المزيد من التعليقات التوضيحية بمرور الوقت.

الأساليب

dbus_call_sync
يطلب إجراء استدعاء الطريقة بشكل متزامن، والانتظار حتى يتم استلام الرد أو رسالة الخطأ قبل المتابعة.
dbus_call_async
يطلب إجراء استدعاء الطريقة بشكل غير متزامن، وإرجاع كائن استدعاء معلق، والذي سيجمع الرد عند وصوله في النهاية.
dbus_call_noreply
يطلب إجراء استدعاء الطريقة بشكل غير متزامن، مع تجاهل أي رسالة رد أو خطأ ممكنة.
dbus_call_timeout
يشير إلى أن المعلمة التالية لاستدعاء الطريقة ستحدد الوقت المطلوب للانتظار للحصول على رد بالمللي ثانية. إذا تم حذفه، فسيتم استخدام المهلة الافتراضية للائن.
