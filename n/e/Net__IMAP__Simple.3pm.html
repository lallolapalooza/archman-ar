Net__IMAP__Simple(3pm)

Net::IMAP::Simple(3) وثائق Perl مقدمة من المستخدم Net::IMAP::Simple(3)

الاسم

Net::IMAP::Simple - امتداد Perl للتعامل البسيط مع حسابات IMAP.

الملخص

use strict;
use warnings;
use Net::IMAP::Simple;
use Email::Simple;
# إنشاء الكائن
my $imap = Net::IMAP::Simple->new('imap.example.com') ||
   die "Unable to connect to IMAP: $Net::IMAP::Simple::errstr\n";
# تسجيل الدخول
if(!$imap->login('user','pass')){
    print STDERR "Login failed: " . $imap->errstr . "\n";
    exit(64);
}
# طباعة مواضيع جميع الرسائل في صندوق الوارد (INBOX)
my $nm = $imap->select('INBOX');
for(my $i = 1; $i <= $nm; $i++){
    if($imap->seen($i)){
        print "*";
    } else {
        print " ";
    }
    my $es = Email::Simple->new(join '', @{ $imap->top($i) } );
    printf("[%03d] %s\n", $i, $es->header('Subject'));
}
$imap->quit;

الوصف

هذه الوحدة هي طريقة بسيطة للوصول إلى حسابات IMAP.

طريقة إنشاء الكائن

my $imap = Net::IMAP::Simple->new( $server [ :port ]);
# أو
my $imap = Net::IMAP::Simple->new( $server [, option_name => option_value ] );

جديد (new)

هذه الطريقة الكلاسية تقوم بإنشاء كائن Net::IMAP::Simple جديد. تأخذ معلمة واحدة مطلوبة وهي الخادم للاتصال به، ومعلمات اختيارية إضافية.

قد تحدد معلمة الخادم الخادم فقط، أو الخادم ورقم المنفذ. لتحديد منفذ بديل، افصله عن الخادم بنقطتين رأسيتين (:":") ، "example.com:5143".

عند النجاح، يتم إرجاع كائن. عند الفشل، لا يتم إرجاع شيء ويتم تعيين رسالة خطأ إلى $Net::IMAP::Simple.

انظر "PREAUTH" أدناه للحصول على استدعاء اسم مضيف خاص لا يستخدم المقابس (داخليًا).

يتم توفير الخيارات كـ hash إلى new():

port => int
تعيين رقم المنفذ (الافتراضي: 143)
timeout => int (الافتراضي: 90)
مهلة الاتصال بالثواني.
retry => int (الافتراضي: 1)
محاولة إعادة محاولة الاتصال (x) مرات قبل الاستسلام
retry_delay => int (الافتراضي: 5)
الانتظار (x) ثوانٍ قبل إعادة محاولة الاتصال
use_v6 => BOOL
إذا تم تعيينه على صحيح، حاول استخدام مقابس IPv6 بدلاً من مقابس IPv4.

يتطلب هذا الخيار وحدة IO::Socket::INET6

use_ssl => BOOL
إذا تم تعيينه على صحيح، حاول استخدام مقابس IO::Socket::SSL بدلاً من المقابس العادية.

لاحظ أنه لا يتم بذل أي محاولة للتحقق من صحة الشهادة افتراضيًا. هذا أمر سيء للغاية لأمانك الشخصي ولكنه يطابق السلوك السابق لهذه الوحدة. يرجى النظر في استخدام "find_ssl_defaults" أدناه.

يتطلب هذا الخيار وحدة IO::Socket::SSL

ssl_version => version
يجب أن يكون هذا واحدًا أو أكثر مما يلي (مفصولة بمسافة): SSLv3 SSLv2 TLSv1. إذا حددت، على سبيل المثال، "SSLv3 SSLv2"، فسيحاول IO::Socket::SSL التفاوض التلقائي. في وقت كتابة هذه السطور، كانت السلسلة الافتراضية هي التفاوض التلقائي v3/v2 - قد يكون قد تغير بحلول الوقت الذي تقرأ فيه هذا.

تحذير: سيؤدي تعيين هذا إلى تعيين "use_ssl" أيضًا.

find_ssl_defaults => []
يبحث في بعض الأماكن القياسية لمكتبات شهادات CA وإذا تم العثور عليها، يقوم بتعيين قيم افتراضية معقولة على غرار ما يلي. 
ssl_options => [ SSL_ca_path => "/etc/ssl/certs/",
    SSL_verify_mode => IO::Socket::SSL::SSL_VERIFY_PEER() ]

تحذير: سيؤدي تعيين هذا إلى تعيين "use_ssl" أيضًا.

ssl_options => []
يمكنك توفير خيارات IO::Socket::SSL الخاصة بك إذا كنت ترغب في ذلك. يتم تجاوزها تمامًا بواسطة "find_ssl_defaults" أعلاه.
bindaddr => str
تعيين عنوان محلي للربط
use_select_cache => BOOL
تمكين ذاكرة التخزين المؤقت select() داخليًا
select_cache_ttl => int
عدد الثواني المسموح بها لاستمرارية نتيجة ذاكرة التخزين المؤقت لـ select قبل تشغيل "$imap->select()" مرة أخرى.
debug => BOOL | \*HANDLE | warn | file:name
تمكين إخراج التصحيح. إذا كان "\*HANDLE" مقبض ملف صالحًا، فسيتم كتابة التصحيح إليه. إذا كانت سلسلة "warn"، فسيتم كتابة التصحيح باستخدام الأمر warn. إذا كانت سلسلة بالشكل "file:name"، فسيتم فتح الملف المسمى للكتابة الإضافية وسيتم كتابة التصحيحات إليه. بخلاف ذلك، سيتم كتابة التصحيح إلى "STDOUT"
readline_callback => CODE
يمكنك اختيار تمرير دالة استدعاء لأغراض المعالجة المسبقة للأسطر قبل تسليمها لبقية الأجزاء الداخلية لـ Net::IMAP::Simple. يمكن أن يكون هذا مفيدًا لتحريك مؤشر أو تعديل سلوك IMAP.

المصادقة المسبقة (PREAUTH)

بدلاً من تمرير رقم منفذ وإصدار تسجيل دخول، قد يكون من الملائم في بعض الحالات المصادقة باستخدام ssh (على سبيل المثال) واستدعاء dovecot يدويًا (على سبيل المثال).

إذا بدأ اسم الخادم بـ "cmd:"، فستصدر Net::IMAP::Simple الأمر بدلاً من بناء المقابس. هذا إعداد نموذجي:

my $cmd = "ssh -C mailhost dovecot --exec-mail imap";
my $imap = Net::IMAP::Simple->new("cmd:$cmd");
 # $imap->login(); ... لا تحتاج إلى هذا
my $number_of_messages = $imap->select("INBOX");

الأساليب

starttls
$imap->starttls;

إذا بدأت جلسة IMAP وترغب في الترقية إلى SSL لاحقًا، يمكنك استخدام هذه الدالة لبدء TLS. ستحاول هذه الدالة "require" IO::Socket::SSL و Net::SSLeay في وقت التشغيل.

تسجيل الدخول (login)
my $inbox_msgs = $imap->login($user, $passwd);

تأخذ هذه الطريقة معلمين مطلوبين، اسم مستخدم وكلمة مرور. يتم المصادقة على هذه الأزواج مقابل الخادم. إذا نجحت المصادقة، فسيتم إرجاع TRUE (1).

لا يتم إرجاع شيء عند الفشل ويتم تعيين معالج الخطأ errstr() مع رسالة الخطأ.

الحالة (status)
my $num_messages                     = $imap->status($folder);
my ($unseen, $recent, $num_messages) = $imap->status($folder);

إصدار أمر "STATUS". يقوم أمر "STATUS" بعد الرسائل دون تغيير حالة صندوق البريد المحدد (اختياريًا). يقوم بإرجاع إما عدد الرسائل، أو عدد الرسائل غير المرئية، والحديثة، وإجمالي عدد الرسائل.

$folder معلمة اختيارية. ستستخدم status() صندوق البريد الحالي أو "INBOX" إذا لم يتم توفير معلمة $folder.

هذه الطريقة لا تستخدم التخزين المؤقت.

يمكن لهذه الطريقة أيضًا الاستعلام عن قيم حالة مخصصة. يُفترض أن المعلمة الأولى للدالة (إن وجدت) هي اسم المجلد، لذا فإن

إذا تم عرض الرسالة كسلسلة نصية أكثر من مرة ، فقد يكون من المنطقي فرض تطبيع السلسلة النصية.

my $message = $imap->get(1);
   $message = "$message"; # forzar la normalización a cadena

لا يلزم عادةً القيام بذلك.

put
$imap->put( $mailbox_name, $message, @flags ) or warn $imap->errstr;

يحفظ رسالة على الخادم تحت المجلد المسمى $mailbox_name. يمكنك اختياريًا تحديد علامات البريد (على سبيل المثال ، "\Seen"، "\Answered")، ولكن يجب أن تبدأ بشرطة مائلة.

إذا كان $message عبارة عن مرجع لمصفوفة، فسيتم طباعة الأسطر بشكل صحيح.

put_with_date
$imap->put_with_date( $mailbox_name, $message, $date, @flags ) or warn $imap->errstr;

يحفظ رسالة على الخادم تحت المجلد المسمى $mailbox_name بشكل مشابه للطريقة put، ولكن توفير تاريخ سيضبط التاريخ الداخلي لخادم IMAP للرسالة إذا كان مدعومًا بواسطة RFC 3501 القسم 6.3.11.

ملاحظة عند استخدامه مع Gmail يتوقع التاريخ بالتنسيق: DD-Mon-YYYY hh:mm:ss tz على سبيل المثال 31-Dec-2016 12:59:59 -0500

if( defined( my $flags = $imap->msg_flags($num) ) ) {
    # tiene $flags!
} else {
    warn "problema al listar los marcadores para el mensaje #$num: "
       . $imap->errstr;
}

في سياق القائمة، يجب عليك استدعاء waserr() لاختبار النجاح.

warn "problema al listar los marcadores para el mensaje #$num: "
   . $imap->errstr if $imap->waserr;

quit
$imap->quit;
OR
$imap->quit(BOOL);

تغلق هذه الطريقة الاتصال بخادم IMAP، وتحذف المجلد المحدد، وتغلق المقبس. لن تُرجع هذه الطريقة رسالة خطأ أبدًا.

اختياريًا، إذا كانت BOOL صحيحة (1)، فسيتم إجراء قطع صارم يغلق المقبس. سيظل هذا القطع الصارم يصدر كلاً من أوامر EXPUNGE و LOGOUT، ولكن يتم تجاهل الاستجابة ويتم إغلاق المقبس بعد إصدار الأوامر.

logout
$imap->logout;

هذه الطريقة مشابهة لطريقة quit، ولكن ليس لديها خيار قطع صارم ولا تحذف المجلد قبل قطع الاتصال وإغلاق المقبس.

last
my $message_number = $imap->last;

تُرجع هذه الطريقة رقم أحدث رسالة في المجلد المحدد، منذ آخر مرة تم فيها تحديد المجلد. في حالة حدوث خطأ، تُرجع لا شيء ويتم تعيين errstr() برسالة الخطأ.

print "Borrado!" if $imap->delete( $message_number );

تضبط هذه الطريقة علامة "\Deleted" على الرسالة (أو الرسائل) المحددة. في حالة النجاح، تُرجع صحيحًا، أو خطأ في حالة حدوث خطأ، ويتم تعيين errstr() برسالة الخطأ. إذا كانت العلامة موجودة بالفعل، فلن يتم إنشاء رسالة خطأ. يتم قبول إما رقم رسالة أو "مجموعة رسائل" كمعامل وحيد. لاحظ أنه لا يتم حذف الرسائل فعليًا حتى يتم التخلص منها (انظر expunge_mailbox).

METHODS

copy
print "copied" if $imap->copy( $message_number, $mailbox );

هذه الطريقة تنسخ رقم الرسالة (أو "مجموعة التسلسل") في علبة البريد المحددة حاليًا إلى المجلد المحدد في الوسيط الثاني. كلا الوسيطين مطلوبان. عند النجاح، تعيد هذه الطريقة القيمة true. تعيد القيمة false عند الفشل ويتم تعيين معالج الأخطاء errstr() مع رسالة الخطأ.

uidcopy
print "copied" if $imap->uidcopy( $message_uid, $mailbox );

هذه الطريقة مطابقة لـ copy() أعلاه، باستثناء أنها تستخدم أرقام UID بدلاً من أرقام التسلسل.

noop
$imap->noop;

تنفذ عملية غير تشغيلية. قد يكون هذا ضروريًا للحصول على تحديثات حول علبة بريد، أو للتأكد من أن الخادم لا يغلق الاتصال كـ idle. تنص RFC 3501 على أن مهلات الخمول للخوادم يجب ألا تقل عن 30 دقيقة.

errstr
print "Login ERROR: " . $imap->errstr . "\n" if !$imap->login($user, $pass);

إرجاع سلسلة آخر خطأ تم التقاطه لآخر عملية فشلت.

waserr
my @flags = $imap->msg_flags(14);
die $imap->errstr if $imap->waserr;

نظرًا لأن msg_flags() يمكنها اختياريًا إرجاع قائمة، فليس من الممكن حقًا اكتشاف الفشل في سياق القائمة. لذلك، يجب عليك استدعاء waserr() إذا كنت ترغب في اكتشاف الأخطاء.

عدد قليل من طرق Net::IMAP::Simple تستخدم waserr(). تلك التي تفعل ذلك ستذكر ذلك.

list2range
في بعض الأحيان يكون لديك قائمة طويلة من أرقام التسلسل المتتالية وتريد حقًا أن تكون نطاقًا بأسلوب IMAP. 
my @list  = (5..9, 13..38, 55,56,57);
my $short = $imap->list2range(@list);
# $short يقول الآن: 5:9,13:38,55:57
range2list
عكس "list2range" تقريبًا. 
my @list = $imap->range2list("1,3,5:9");
# @list هي (1,3,5,6,7,8,9);

البحث

search
تعيد هذه الدالة مصفوفة من أرقام الرسائل (في سياق القائمة) أو عدد الرسائل المتطابقة (في سياق القياسي). تأخذ وسيطًا واحدًا: البحث.

يمكن أن يكون البحث في IMAP مربكًا بعض الشيء وهذه الدالة لا تحاول تحليل عمليات البحث الخاصة بك. إذا كنت ترغب في إجراء عمليات بحث يدويًا، فيرجى الرجوع إلى RFC 3501.

يتم دعم الفرز في IMAP (انظر RFC 5256) عبر وسيط ثانٍ اختياري. تتطلب RFC تحديد ترميز الأحرف، والذي يمكن توفيره عبر وسيط ثالث اختياري (افتراضيًا UTF-8).

فيما يلي بعض الأمثلة:

my @ids = $imap->search("UNSEEN");
my @ids = $imap->search('SUBJECT "blarg is \"blarg\""');
my @ids = $imap->search('FROM "joe@aol.com"');
my @ids = $imap->search("DELETED");
# مثال من RFC 3501، يتم ربط شروط البحث معًا بـ AND
my @ids = $imap->search('FLAGGED SINCE 1-Feb-1994 NOT FROM "Smith"');
# مثال من RFC 3501، يتم ربط شروط البحث معًا بـ OR
my @ids = $imap->search('OR BODY "blard" SUBJECT "blarg"');
# flagged and ( since x or !from y ):
my @ids = $imap->search('FLAGGED OR SINCE x NOT FROM "y"');
  # لا يوجد خطأ مطبعي أعلاه، راجع RFC
# مثال من RFC 5256، مرتبة حسب الموضوع والتاريخ العكسي
my @ids = $imap->search('BODY "zaphod"', 'SUBJECT REVERSE DATE');

نظرًا لأن هذه الوحدة مخصصة للبساطة، فإن Net::IMAP::Simple لديها بعض أدوات المساعدة للبحث. إذا كنت بحاجة إلى عمليات بحث منطقية معقدة وأشياء أخرى، فستحتاج إلى تعلم البحث. إذا كنت بحاجة إلى بحث سريع للرسائل غير المرئية، فانظر أدناه.

كل هذه تعيد مصفوفة من الرسائل أو عدد الرسائل تمامًا كما تفعل دالة البحث. يأخذ بعضها وسائط، والبعض الآخر لا يأخذ. إنها تحاول فهم وسائطك قليلاً، وسيتم ذكر آليات ذلك (إن وجدت) أدناه.

search_seen
يعيد أرقام الرسائل التي تحتوي على علامة \Seen.
search_recent
يعيد أرقام الرسائل التي تحتوي على علامة \Recent.
search_answered
يعيد أرقام الرسائل التي تحتوي على علامة \Answered.
search_deleted
يعيد أرقام الرسائل التي تحتوي على علامة \Deleted.
search_flagged
يعيد أرقام الرسائل التي تحتوي على علامة \Flagged.
search_draft
يعيد أرقام الرسائل التي تحتوي على علامة \Draft.
search_unseen
يعيد أرقام الرسائل التي لا تحتوي على علامة \Seen.
search_old
يعيد أرقام الرسائل التي لا تحتوي على علامة \Recent.
search_unanswered
يعيد أرقام الرسائل التي لا تحتوي على علامة \Answered.
search_undeleted
يعيد أرقام الرسائل التي لا تحتوي على علامة \Deleted.
search_unflagged
يعيد أرقام الرسائل التي لا تحتوي على علامة \Flagged.
search_smaller
تأخذ هذه الدالة وسيطًا واحدًا سنسميه "<x>" وتعيد أرقام الرسائل التي تكون أصغر من "<x>" بايت. ستحاول هذه الدالة فرض وسيطك ليكون رقمًا قبل تمريره إلى خادم IMAP.
search_larger
تأخذ هذه الدالة وسيطًا واحدًا سنسميه "<x>" وتعيد أرقام الرسائل التي تكون أكبر من "<x>" بايت. ستحاول هذه الدالة فرض وسيطك ليكون رقمًا قبل تمريره إلى خادم IMAP.
search_from
تأخذ هذه الدالة وسيطًا واحدًا سنسميه "<x>" وتعيد أرقام الرسائل التي تحتوي على "<x>" في رأس المرسل. ستحاول هذه الدالة فرض السلسلة النصية الخاصة بك في تنسيق السلسلة المقتبسة RFC3501.
search_to
تأخذ هذه الدالة وسيطًا واحدًا سنسميه "<x>" وتعيد أرقام الرسائل التي تحتوي على "<x>" في رأس المستلم. ستحاول هذه الدالة فرض السلسلة النصية الخاصة بك في تنسيق السلسلة المقتبسة RFC3501.
search_cc
تأخذ هذه الدالة وسيطًا واحدًا سنسميه "<x>" وتعيد أرقام الرسائل التي تحتوي على "<x>" في رأس النسخة الكربونية. ستحاول هذه الدالة فرض السلسلة النصية الخاصة بك في تنسيق السلسلة المقتبسة RFC3501.
search_bcc
تأخذ هذه الدالة وسيطًا واحدًا سنسميه "<x>" وتعيد أرقام الرسائل التي تحتوي على "<x>" في رأس النسخة المخفية. ستحاول هذه الدالة فرض السلسلة النصية الخاصة بك في تنسيق السلسلة المقتبسة RFC3501.
search_subject
تأخذ هذه الدالة وسيطًا واحدًا سنسميه "<x>" وتعيد أرقام الرسائل التي تحتوي على "<x>" في رأس الموضوع. ستحاول هذه الدالة فرض السلسلة النصية الخاصة بك في تنسيق السلسلة المقتبسة RFC3501.
search_body
تأخذ هذه الدالة وسيطًا واحدًا سنسميه "<x>" وتعيد أرقام الرسائل التي تحتوي على "<x>" في نص الرسالة. ستحاول هذه الدالة فرض السلسلة النصية الخاصة بك في تنسيق السلسلة المقتبسة RFC3501.
search_before
تأخذ هذه الدالة وسيطًا واحدًا سنسميه "<x>" وتعيد أرقام الرسائل التي تم استلامها قبل "<x>". إذا كان لديك Date::Manip مثبتًا (اختياري)، فستحاول هذه الدالة فرض التاريخ في تنسيق "%d-%b-%Y" (اليوم-اسم الشهر-السنة) كما تتطلبه RFC3501. إذا لم يكن لديك هذه الوحدة، فلن يتم بذل أي محاولة لإجبار تاريخك على التنسيق الصحيح.
search_since
تأخذ هذه الدالة وسيطًا واحدًا سنسميه "<x>" وتعيد أرقام الرسائل التي تم استلامها بعد "<x>". إذا كان لديك Date::Manip مثبتًا (اختياري)، فستحاول هذه الدالة فرض التاريخ في تنسيق "%d-%m-%Y" (اليوم-الشهر-السنة) كما تتطلبه RFC3501. إذا لم يكن لديك هذه الوحدة، فلن يتم بذل أي محاولة لإجبار تاريخك على التنسيق الصحيح.
