ne_iaddr_parse(3)

الحزمة: neon

التراخيص: LGPL-2.0-or-later

NE_IADDR_MAKE(3) مرجع واجهة برمجة تطبيقات neon NE_IADDR_MAKE(3)

الاسم

ne_iaddr_make, ne_iaddr_cmp, ne_iaddr_print, ne_iaddr_typeof, ne_iaddr_parse, ne_iaddr_raw, ne_iaddr_reverse, ne_iaddr_free - وظائف لمعالجة عناوين الشبكة

الملخص

#include <ne_socket.h>
typedef enum {
    ne_iaddr_ipv4 = 0,
    ne_iaddr_ipv6
} ne_iaddr_type;

ne_inet_addr *ne_iaddr_make(ne_iaddr_type type, const unsigned char *raw);

int ne_iaddr_cmp(const ne_inet_addr *ia1, const ne_inet_addr *ia2);

char *ne_iaddr_print(const ne_inet_addr *ia, char *buffer, size_t bufsiz);

ne_iaddr_type ne_iaddr_typeof(const ne_inet_addr *ia);

ne_inet_addr *ne_iaddr_parse(const char *address, ne_iaddr_type type);

unsigned char *ne_iaddr_raw(const ne_inet_addr *ia, unsigned char *buffer);

int ne_iaddr_reverse(const ne_inet_addr *ia, char *buffer, size_t buflen);

void ne_iaddr_free(const ne_inet_addr *ia);

الوصف

ne_iaddr_make ينشئ كائن ne_inet_addr من عنوان شبكة ثنائية خام؛ على سبيل المثال، تمثل البايتات الأربعة 0x7f 0x00 0x00 0x01 عنوان IPv4 127.0.0.1. الكائن المُعاد مناسب للتمرير إلى ne_sock_connect. يحتوي عنوان IPv4 الثنائي على أربعة بايتات؛ يحتوي عنوان IPv6 الثنائي على ستة عشر بايتًا؛ يجب أن تكون العناوين التي يتم تمريرها بترتيب بايت الشبكة.

ne_iaddr_cmp يقارن بين كائنين لعنوان شبكة؛ ويعيد صفرًا فقط إذا كانا متطابقين. لا تحتاج الكائنات إلى أن يكون لها نفس نوع العنوان؛ إذا لم تكن العناوين من نفس النوع، فإن القيمة المُعادة مضمونة أن تكون غير صفرية.

ne_iaddr_print يطبع تمثيلًا نصيًا سهل القراءة لعنوان شبكة في مخزن مؤقت، على سبيل المثال النص "127.0.0.1".

ne_iaddr_typeof يعيد نوع كائن عنوان الشبكة المحدد.

ne_iaddr_parse يحلل تمثيلًا نصيًا لعنوان شبكة (مثل "127.0.0.1") وينشئ كائن عنوان شبكة لتمثيل العنوان المُحلل.

ne_iaddr_raw يكتب التمثيل الثنائي الخام لعنوان شبكة إلى المخزن المؤقت المقدم. تتم كتابة البايتات بترتيب بايت الشبكة؛ يجب أن يكون المخزن المؤقت بطول مناسب لنوع العنوان (4 بايتات لعنوان IPv4، 16 بايتًا لعنوان IPv6).

ne_iaddr_reverse يقوم بالبحث العكسي عن اسم لعنوان الشبكة المحدد، ويكتب (أول) اسم المضيف المرتبط بعنوان IP إلى المخزن المؤقت المقدم. إذا كان اسم المضيف أطول من المخزن المؤقت، فسيتم اقتطاعه بصمت؛ عند النجاح، يكون النص المكتوب في المخزن المؤقت دائمًا منتهيًا بحرف NULL.

ne_iaddr_free يحرر الذاكرة المرتبطة بكائن عنوان الشبكة.

قيمة الإرجاع

ne_iaddr_make يعيد NULL إذا كان نوع العنوان المُمرر غير مدعوم (على سبيل المثال، على منصة لا تدعم IPv6).

ne_iaddr_print يعيد مؤشر buffer، ولا يعيد NULL أبدًا.

ne_iaddr_parse يعيد كائن عنوان شبكة عند النجاح، أو NULL عند الفشل في تحليل المعامل address.

ne_iaddr_reverse يعيد صفرًا عند النجاح أو غير صفر إذا لم يكن هناك اسم مضيف مرتبط بالعنوان.

ne_iaddr_raw يعيد المعامل buffer، ولا يعيد NULL أبدًا.

أمثلة

يقوم المثال التالي بتوصيل مقبس بالمنفذ 80 على العنوان 127.0.0.1.

unsigned char addr[] = "\0x7f\0x00\0x00\0x01";
ne_inet_addr *ia;
ia = ne_iaddr_make(ne_iaddr_ipv4, addr);
if (ia != NULL) {
    ne_socket *sock = ne_sock_connect(ia, 80);
    ne_iaddr_free(ia);
    /* ... */
} else {
    /* ... */
}
