Net__DNS__RR__SIG(3pm)

الحزمة: perl-net-dns

التراخيص: PerlArtistic

Net::DNS::RR::SIG(3) وثائق Perl المساهمة من المستخدم Net::DNS::RR::SIG(3)

الاسم

Net::DNS::RR::SIG - سجل مورد DNS SIG

الملخص

use Net::DNS;
$rr = Net::DNS::RR->new('name SIG typecovered algorithm labels
                        orgttl sigexpiration siginception
                        keytag signame signature');
use Net::DNS::SEC;
$sigrr = Net::DNS::RR::SIG->create(
                        $string, $keypath,
                        sigval => 10    # دقائق
                        );
$sigrr->verify( $string, $keyrr ) || die $sigrr->vrfyerrstr;
$sigrr->verify( $packet, $keyrr ) || die $sigrr->vrfyerrstr;

الوصف

فئة لسجلات موارد توقيع DNS الرقمي (SIG).

بالإضافة إلى الطرق القياسية الموروثة من Net::DNS::RR، تحتوي الفئة على طريقة لتوقيع الحزم وسلاسل البيانات القياسية باستخدام المفاتيح الخاصة (create) وطريقة للتحقق من التواقيع.

سجل SIG هو تطبيق لـ RFC2931. انظر Net::DNS::RR::RRSIG لتطبيق RFC4034.

الطرق

الطرق المتاحة هي تلك الموروثة من الفئة الأساسية مضافًا إليها الطرق الخاصة بالنوع المعرفة في هذه الحزمة.

لا يُنصح باستخدام ميزات الحزمة غير الموثقة أو الوصول المباشر إلى هياكل البيانات الداخلية، وقد يؤدي ذلك إلى إنهاء البرنامج أو سلوك غير متوقع آخر.

الخوارزمية

$algorithm = $rr->algorithm;

يشير حقل رقم الخوارزمية إلى الخوارزمية التشفيرية المستخدمة لإنشاء التوقيع.

يمكن أيضًا استدعاء algorithm() كطريقة فئة أو دالة بسيطة لإجراء ترجمة الأكواد الرقمية والاسمية.

أوقات انتهاء التوقيع وبدء التوقيع

sigex sigin sigval

$expiration = $rr->sigexpiration;
$expiration = $rr->sigexpiration( $value );
$inception = $rr->siginception;
$inception = $rr->siginception( $value );

يحدد حقلا انتهاء صلاحية التوقيع وبدء التوقيع فترة صلاحية زمنية للتوقيع.

يمكن تحديد القيمة بواسطة سلسلة نصية بالتنسيق 'yyyymmddhhmmss' أو قيمة time() الخاصة بـ Perl.

قيم الإرجاع ذات قيم مزدوجة، وتوفر قيمة نصية أو قيمة رقمية لـ Perl time().

keytag

$keytag = $rr->keytag;
$rr->keytag( $keytag );

يحتوي حقل keytag على قيمة علامة المفتاح لسجل KEY RR الذي يتحقق من صحة هذا التوقيع.

signame

$signame = $rr->signame;
$rr->signame( $signame );

تحدد قيمة حقل اسم الموقّع اسم المالك لسجل KEY RR الذي من المفترض أن يستخدمه المدقق للتحقق من صحة هذا التوقيع.

signature

sig

$sig = $rr->sig;
$rr->sig( $sig );

يحتوي حقل التوقيع على التوقيع التشفيري الذي يغطي بيانات SIG RDATA (باستثناء حقل التوقيع) والبيانات الموضوعة.

sigbin

$sigbin = $rr->sigbin;
$rr->sigbin( $sigbin );

تمثيل ثنائي للتوقيع التشفيري.

create

إنشاء توقيع على البيانات القياسية.

use Net::DNS::SEC;
$keypath = '/home/olaf/keys/Kbla.foo.+001+60114.private';
$sigrr = Net::DNS::RR::SIG->create( $data, $keypath );
$sigrr = Net::DNS::RR::SIG->create(
                        $data, $keypath,
                        sigval => 10
                        );
$sigrr->print;
# بدلاً من ذلك استخدم Net::DNS::SEC::Private
$private = Net::DNS::SEC::Private->new($keypath);
$sigrr= Net::DNS::RR::SIG->create( $data, $private );

create() هو مُنشئ بديل لكائن SIG.

تُرجع هذه الطريقة سجل SIG مع التوقيع على البيانات التي تم إنشاؤها بالمفتاح الخاص المخزن في ملف المفتاح.

الوسيط الأول هو قيمة قياسية تحتوي على البيانات المراد توقيعها.

الوسيط الثاني هو سلسلة نصية تحدد المسار إلى ملف يحتوي على المفتاح الخاص كما تم إنشاؤه باستخدام dnssec-keygen، وهو برنامج يأتي مع توزيعة ISC BIND.

الوسائط الاختيارية المتبقية تتكون من أزواج (name => value) كما يلي:

sigin  => 20241201010101,       # بدء التوقيع
sigex  => 20241201011101,       # انتهاء التوقيع
sigval => 10,                   # نافذة الصلاحية (دقائق)

يمكن تحديد قيم sigin و sigex كقيم وقت Perl أو كسلسلة نصية بالتنسيق 'yyyymmddhhmmss'. القيمة الافتراضية لـ sigin هي وقت التوقيع.

يحدد الوسيط sigval نافذة صلاحية التوقيع بالدقائق ( sigex = sigin + sigval ).

افتراضيًا، يكون التوقيع صالحًا لمدة 10 دقائق.

verify

$verify = $sigrr->verify( $data, $keyrr );
$verify = $sigrr->verify( $data, [$keyrr, $keyrr2, $keyrr3] );

تقوم الطريقة verify() بالتحقق من SIG0 للبيانات المحددة مقابل التوقيع الموجود في الكائن $sigrr نفسه باستخدام المفتاح العام في $keyrr.

إذا تم توفير مرجع إلى Net::DNS::Packet، تقوم الطريقة بإجراء تحقق من SIG0 على بيانات الحزمة.

يمكن أن يكون الوسيط الثاني إما كائن Net::DNS::RR::KEYRR أو مرجع إلى مصفوفة من هذه الكائنات. سيعود التحقق بالنجاح بمجرد أن يؤدي أحد المفاتيح في المصفوفة إلى التحقق الإيجابي.

تُرجع خطأ (false) عند حدوث خطأ وتعيّن $sig->vrfyerrstr

vrfyerrstr

$sig0 = $packet->sigrr || die 'not signed';
print $sig0->vrfyerrstr unless $sig0->verify( $packet, $keyrr );
$sigrr->verify( $packet, $keyrr ) || die $sigrr->vrfyerrstr;

إنشاء المفاتيح

يتم إنشاء ملفات المفاتيح الخاصة وسجلات DNSKEY المقابلة لها بشكل مريح باستخدام dnssec-keygen، وهو برنامج يأتي مع توزيعة ISC BIND.

dnssec-keygen -a 10 -b 2048 rsa.example.
dnssec-keygen -a 13 -f ksk  ecdsa.example.
dnssec-keygen -a 13         ecdsa.example.

لا تقم بتغيير اسم ملف المفتاح الخاص. يستخدم الأمر create اسم الملف كما تم إنشاؤه بواسطة dnssec-keygen لتحديد مالك المفتاح والخوارزمية ومعرف المفتاح (keytag).

ملاحظات

الكود غير مُحسَّن للسرعة.

إذا ظل هذا الكود موجودًا في عام 2100 (وهي ليست سنة كبيسة) فستحتاج إلى التحقق من المعالجة الصحيحة للأوقات بعد 28 فبراير.

شكر وتقدير

على الرغم من أن كودهم الأصلي قد اختفى بعد إعادة تصميم Net::DNS، Net::DNS::SEC وواجهة برمجة تطبيقات OpenSSL، إلا أن المساهمين الأفراد التاليين يستحقون الاعتراف بتأثيرهم الكبير على تطوير حزمة SIG.

قدم آندي فاسكيس (Network Associates Laboratories) كودًا لـ RSA.

قدم T.J. Mather دعمًا لخوارزمية DSA.
