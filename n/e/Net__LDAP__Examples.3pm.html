التراخيص: GPL

الاسم

Net::LDAP::Examples - PERL LDAP بالمثال

الوصف

الأمثلة التالية هي بالطبع كود PERL، تم العثور عليها لتعمل مع وحدات Net::LDAP.

الهدف من هذا المستند هو منح القارئ نقطة انطلاق للنسخ واللصق لبدء تشغيل تطبيق LDAP.

ستجد أدناه مقتطفات من الكود التي يجب أن تعمل كما هي مع القليل من العمل لتصحيح أي تعيينات للمتغيرات وتفاصيل LDAP الخاصة، مثل صيغة اسم مميز، المتعلقة بتنفيذ المستخدم الخاص.

الإجراء التشغيلي القياسي الذي يتم اتباعه هنا هو:

1 الحزمة - use Net::LDAP
2 التهيئة - new
3 الربط - bind
4 العملية - add modify moddn search
4.1 المعالجة - عرض البيانات من البحث
5 الخطأ - عرض معلومات الخطأ
6 فك الربط - unbind

ابحث في كل من هذه للحصول على مقتطف من الكود لتلبية احتياجاتك.

ما لم يتم تغطيته في هذه الأمثلة حاليًا:

طرق abandon و compare
روتينات callback الفرعية

العملية - توليد بحث

sub LDAPsearch
{
  my ($ldap,$searchString,$attrs,$base) = @_;
  # إذا لم يمرروا قاعدة... قم بتعيينها لهم
  if (!$base ) { $base = "o=mycompany, c=mycountry"; }
  # إذا لم يمرروا مصفوفة من السمات...
  # قم بإعداد شيء لهم
  if (!$attrs ) { $attrs = [ 'cn','mail' ]; }
  my $result = $ldap->search ( base    => "$base",
                               scope   => "sub",
                               filter  => "$searchString",
                               attrs   =>  $attrs
                             );

}

my @Attrs = ( );               # طلب جميع السمات المتاحة
                               # ليتم إرجاعها.
my $result = LDAPsearch ( $ldap, "sn=*", \@Attrs );

المعالجة - عرض نتائج البحث

#------------
#
# الوصول إلى البيانات كما لو كانت في هيكل
# أي باستخدام الطريقة "as_struct"
#
my $href = $result->as_struct;
# الحصول على مصفوفة من أسماء DN
my @arrayOfDNs  = keys %$href;        # استخدام تجزئات DN
# معالجة كل DN باستخدامها كمفتاح
foreach ( @arrayOfDNs ) {
  print $_, "\n";
  my $valref = $$href{$_};
  # الحصول على مصفوفة من أسماء السمات
  # تم تمريرها لهذا DN.
  my @arrayOfAttrs = sort keys %$valref; # استخدام تجزئات السمات
  my $attrName;
  foreach $attrName (@arrayOfAttrs) {
    # تخطي أي بيانات ثنائية: مقرف!
    next if ( $attrName =~ /;binary$/ );
    # الحصول على قيمة السمة (مؤشر) باستخدام
    # اسم السمة كتجزئة
    my $attrVal =  @$valref{$attrName};
    print "\t $attrName: @$attrVal \n";
  }
  print "#-------------------------------\n";
  # نهاية ذلك الـ DN
}
#
# نهاية طريقة as_struct
#
#--------
#------------
#
# معالجة كل نتيجة بشكل مستقل
# أي باستخدام طريقة المرور
#
my @entries = $result->entries;
my $entr;
foreach $entr ( @entries ) {
  print "DN: ", $entr->dn, "\n";
  my $attr;
  foreach $attr ( sort $entr->attributes ) {
    # تخطي البيانات الثنائية التي لا يمكننا التعامل معها
    next if ( $attr =~ /;binary$/ );
    print "  $attr : ", $entr->get_value ( $attr ) ,"\n";
  }
  print "#-------------------------------\n";
}
#
# نهاية طريقة المرور
#------------

#
#   تعديل
#
#   لكل تعديل من التعديلات أدناه ستحتاج إلى توفير
#   اسم DN كامل (اسم مميز) للمتغير $dn.
#   مثال:
#    cn=Jo User,ou=person,o=mycompany,c=mycountry
#
#   أوصي بإجراء بحث (مدرج أعلاه)
#   ثم استخدام اسم الـ DN الذي تم إرجاعه لملء المتغير $dn.
#
#   هل لدينا نتيجة واحدة فقط تم إرجاعها من البحث؟
if ( $result->count != 1 ) { exit; }  # لا.. إنهاء
my $dn = $entries[0]->dn;         # نعم.. الحصول على اسم الـ DN
#######################################
#
#   تعديل باستخدام تجزئة (HASH)
#
my %ReplaceHash = ( keyword => "x", proxy => "x" );
my $result = LDAPmodifyUsingHash ( $ldap, $dn, \%ReplaceHash );
sub LDAPmodifyUsingHash
{
  my ($ldap, $dn, $whatToChange ) = @_;
  my $result = $ldap->modify ( $dn,
                               replace => { %$whatToChange }
                             );
  return $result;
}
#######################################
#
#   تعديل باستخدام قائمة مصفوفة
#
my @ReplaceArrayList = [ 'keyword', "xxxxxxxxxx",
                         'proxy' , "yyyyyyyyyy"   ];
my $result = LDAPmodifyUsingArrayList ( $ldap, $dn, \@ReplaceArrayList );
sub LDAPmodifyUsingArrayList
{
  my ($ldap, $dn, $whatToChange ) = @_;
  my $result = $ldap->modify ( $dn,
                               changes => [
                                 replace => @$whatToChange
                               ]
                             );
  return $result;
}
#######################################
#
#   تعديل باستخدام مصفوفة
#
my @ReplaceArray = ( 'keyword', "xxxxxxxxxx" ,
                     'proxy' , "yyyyyyyyyy"   );
my $result = LDAPmodifyUsingArray ( $ldap, $dn, \@ReplaceArray );
sub LDAPmodifyUsingArray
{
  my ($ldap, $dn, $whatToChange ) = @_;
  my $result = $ldap->modify ( $dn,
                               changes => [
                                 replace => [ @$whatToChange ]
                               ]
                             );
  return $result;
}
#######################################
#
#   تعديل سجل موجود باستخدام 'Changes'
#    (أو مزيج من الإضافة/الحذف/الاستبدال)
#
my @whatToChange;
my @ReplaceArray;
my @DeleteArray;
my @AddArray;
push @AddArray, 'cn', "me myself";
push @ReplaceArray, 'sn', '!@#$%^&*()__+Hello There';
push @ReplaceArray, 'cn', "me myself I";
push @DeleteArray, 'cn', "me myself";
if ( $#ReplaceArray > 0 ) {
  push @whatToChange, 'replace';
  push @whatToChange, \@ReplaceArray;
}
if ( $#DeleteArray > 0 ) {
  push @whatToChange, 'delete';
  push @whatToChange, \@DeleteArray;
}
if ( $#AddArray > 0 ) {
  push @whatToChange, 'add';
  push @whatToChange, \@AddArray;
}
$result = LDAPmodify ( $ldap, $dn, \@whatToChange );
sub LDAPmodify
{
  my ($ldap, $dn, $whatToChange) = @_;
  my $result = $ldap->modify ( $dn,
                               changes => [
                                 @$whatToChange
                               ]
                             );
  return $result;
}

العملية - تغيير الـ RDN

my $newRDN = "cn=Joseph User";
my $result = LDAPrdnChange ( $ldap, $dn, $newRDN, "archive" );
sub LDAPrdnChange
{
  my ($ldap,$dn,$whatToChange,$action) = @_;
  my $branch;
  #
  # إذا تم تحديد إجراء الأرشفة، قم بنقل هذا
  # الإدخال إلى مكان آخر في الدليل.
  #
  if ( $action =~ /archive/i )  {
    $branch = "ou=newbranch, o=mycompany, c=mycountry";
  }
  #
  # استخدم 'deleteoldrdn' لتجنب الحصول على
  # قيم متعددة في سمة التسمية.
  # في معظم الحالات ستكون سمة 'CN'
  #
  my $result = $ldap->moddn ( $dn,
                              newrdn => $whatToChange,
                              deleteoldrdn => '1',
                              newsuperior => $branch
                            );
  return $result;
}
