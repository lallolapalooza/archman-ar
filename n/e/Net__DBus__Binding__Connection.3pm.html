Net__DBus__Binding__Connection(3pm)

الحزمة: perl-net-dbus

التراخيص: GPL, PerlArtistic

اسم

Net::DBus::Binding::Connection - اتصال بـ D-Bus

الملخص

use Net::DBus::Binding::Connection;

# إنشاء اتصال جديد
my $con = Net::DBus::Binding::Connection->new();

# إرسال واستقبال رسالة
my $msg = $con->make_method_call_message( ... );
$con->send($msg);
my $reply = $con->wait_for_reply($msg);

# التعامل مع الردود
if ($reply->is_error) {
    # التعامل مع الخطأ
} else {
    # التعامل مع النجاح
}

الوصف

تمثل هذه الفئة اتصالاً بمثيل D-Bus.

الطرق

new()
ينشئ كائن اتصال Net::DBus::Binding::Connection جديد. بشكل افتراضي، يقوم بالاتصال بـ bus الجلسة.
new( bus_type => $bus_type )
ينشئ كائن اتصال Net::DBus::Binding::Connection جديد، يتصل بـ $bus_type المحدد. $bus_type يمكن أن تكون إحدى القيم التالية: "session"، "system"، أو "none".
send( $msg )
يرسل الرسالة $msg عبر الاتصال. يجب أن تكون الرسالة قد تم إنشاؤها باستخدام إحدى طرق make_*_message.
my $msg = $con->make_raw_message($rawmsg)
ينشئ رسالة جديدة، مع تهيئتها من كائن الرسالة ذي المستوى المنخفض C الموفر بواسطة المعلمة $rawmsg. سيتم تحويل الكائن الناتج إلى الفئة الفرعية المناسبة من Net::DBus::Binding::Message.
my $msg = $con->make_error_message( replyto => $method_call, name => $name, description => $description);
ينشئ رسالة جديدة، تمثل خطأ حدث أثناء التعامل مع كائن استدعاء الطريقة الممرر كالمعلمة "replyto". المعلمة "name" هي الاسم الرسمي لحالة الخطأ، بينما "description" هي قطعة نصية قصيرة تقدم معلومات أكثر تحديدًا حول الخطأ.
my $call = $con->make_method_call_message( $service_name, $object_path, $interface, $method_name);
ينشئ رسالة تمثل استدعاء للطريقة على الكائن الموجود في المسار $object_path داخل العميل الذي يمتلك الاسم المعروف المعطى بواسطة $service_name. الطريقة المراد استدعاؤها لها الاسم $method_name داخل الواجهة المحددة بواسطة المعلمة $interface.
my $msg = $con->make_method_return_message( replyto => $method_call);
ينشئ رسالة تمثل ردًا على استدعاء الطريقة الممرر في المعلمة "replyto".
my $signal = $con->make_signal_message( object_path => $path, interface => $interface, signal_name => $name);
ينشئ رسالة جديدة، تمثل إشارة [سيتم] بثها بواسطة الكائن الموجود تحت المسار المعطى بواسطة المعلمة "object_path". اسم الإشارة معطى بواسطة المعلمة "signal_name"، ويتم تحديد نطاقه إلى الواجهة المعطاة بواسطة المعلمة "interface".
