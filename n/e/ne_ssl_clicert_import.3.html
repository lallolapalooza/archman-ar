ne_ssl_clicert_import(3)

الحزمة: neon

التراخيص: LGPL-2.0-or-later

NE_SSL_CLIENT_CERT(3) مرجع واجهة برمجة تطبيقات neon NE_SSL_CLIENT_CERT(3)

الاسم

ne_ssl_clicert_read, ne_ssl_clicert_import, ne_ssl_clicert_fromuri, ne_ssl_clicert_name, ne_ssl_clicert_encrypted, ne_ssl_clicert_decrypt, ne_ssl_clicert_owner, ne_ssl_clicert_free - معالجة شهادة عميل SSL

الملخص

#include <ne_ssl.h>

ne_ssl_client_cert *ne_ssl_clicert_read(const char *filename);

ne_ssl_client_cert *ne_ssl_clicert_import(const unsigned char *filename, size_t buflen);

ne_ssl_client_cert *ne_ssl_clicert_fromuri(const char *uri, unsigned flags);

const char *ne_ssl_clicert_name(const ne_ssl_client_cert *ccert);

int ne_ssl_clicert_encrypted(const ne_ssl_client_cert *ccert);

int ne_ssl_clicert_decrypt(ne_ssl_client_cert *ccert, const char *password);

const ne_ssl_certificate *ne_ssl_clicert_owner(const ne_ssl_client_cert *ccert);

void ne_ssl_clicert_free(ne_ssl_client_cert *ccert);

الوصف

تقوم الدالة ne_ssl_clicert_read بقراءة شهادة العميل من ملف بتنسيق PKCS#12، وتعيد كائن ne_ssl_client_cert. تقوم الدالة ne_ssl_clicert_import بقراءة شهادة العميل من مخزن مؤقت للذاكرة باستخدام تنسيق PKCS#12. تقوم الدالة ne_ssl_clicert_fromuri بتحميل شهادة العميل من URI. مخططات URI المدعومة محددة بواسطة مجموعة أدوات SSL/TLS، وقد تشمل URIs PKCS#11[1].

يمثل كائن ne_ssl_client_cert كلاً من شهادة العميل والمفتاح الخاص المرتبط بها؛ يُستخدم مصطلح "شهادة العميل" هنا للإشارة إلى هذه المجموعة. يمكن أن يكون كائن شهادة العميل الذي تم تحميله بواسطة أي من الدوال أعلاه في إحدى حالتين: *مشفرة* أو *غير مشفرة*. ستعيد الدالة ne_ssl_clicert_encrypted قيمة غير صفرية إذا كانت شهادة العميل في الحالة *المشفرة*.

يجب أن تكون شهادة العميل في الحالة غير المشفرة قبل ربطها بجلسة. يمكن استخدام ne_ssl_clicert_decrypt لفك تشفير شهادة العميل باستخدام كلمة المرور المناسبة. يجب استدعاء هذه الدالة فقط إذا كان الكائن في الحالة *المشفرة*؛ إذا فشل فك التشفير، لا تتغير حالة الشهادة، لذا يمكن محاولة فك التشفير أكثر من مرة باستخدام كلمات مرور مختلفة.

يمكن إعطاء شهادة العميل "اسمًا ودودًا" عند إنشائها؛ ستعيد ne_ssl_clicert_name هذا الاسم (أو NULL إذا لم يتم تحديد اسم ودود). يمكن استخدام ne_ssl_clicert_name عندما تكون شهادة العميل في الحالة المشفرة أو غير المشفرة، وستعيد نفس السلسلة النصية طوال عمر الكائن.

تعيد الدالة ne_ssl_clicert_owner جزء الشهادة من شهادة العميل؛ يجب استدعاؤها فقط إذا كانت شهادة العميل في الحالة *غير المشفرة*.

عندما لا تكون هناك حاجة لكائن شهادة العميل، يجب استخدام الدالة ne_ssl_clicert_free لتدمير الكائن.

قيمة الإرجاع

تعيد كل من ne_ssl_clicert_read و ne_ssl_clicert_import و ne_ssl_clicert_fromuri كائن شهادة عميل، أو NULL إذا لم يكن من الممكن قراءة/استيراد الشهادة. تعيد ne_ssl_clicert_encrypted صفرًا إذا كان الكائن في الحالة غير المشفرة، أو قيمة غير صفرية إذا كان في الحالة المشفرة. تعيد ne_ssl_clicert_name سلسلة نصية مضمنة بـ NULL، أو NULL. تعيد ne_ssl_clicert_owner كائن شهادة.

أمثلة

يقرأ الكود التالي شهادة عميل ويفك تشفيرها إذا لزم الأمر، ثم يقوم بتحميلها في جلسة HTTP.

ne_ssl_client_cert *ccert;
ccert = ne_ssl_clicert_read("/path/to/client.p12");
if (ccert == NULL) {
   /* التعامل مع الخطأ... */
} else if (ne_ssl_clicert_encrypted(ccert)) {
   char *password = prompt_for_password();
   if (ne_ssl_clicert_decrypt(ccert, password)) {
      /* لم يكن بالإمكان فك التشفير! التعامل مع الخطأ... */
   }
}
ne_ssl_set_clicert(sess, ccert);
