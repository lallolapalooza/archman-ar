NetAddr__IP(3pm)

الحزمة: perl-netaddr-ip

التراخيص: PerlArtistic, GPL

الاسم وثائق المستخدمين المساهمين في Perl الاسم

الاسم

NetAddr::IP - يدير عناوين IPv4 و IPv6 والشبكات الفرعية

الصيغة

use NetAddr::IP qw(
      Compact
      Coalesce
      Zeros
      Ones
      V4mask
      V4net
      netlimit
      :aton           DEPRECATED
      :lower
      :upper
      :old_storable
      :old_nth
      :rfc3021
      :nofqdn
);
ملاحظة: يحتوي NetAddr::IP::Util على مجموعة كاملة من أدوات العناوين للتحويل بين الثنائي والنص.
    inet_aton, inet_ntoa, ipv6_aton, ipv6_ntoa 
    ipv6_n2x, ipv6_n2d inet_any2d, inet_n2dx, 
    inet_n2ad, inetanyto6, ipv6to4

انظر NetAddr::IP::Util

my $ip = new NetAddr::IP '127.0.0.1';
       أو إذا كنت تفضل
my $ip = NetAddr::IP->new('127.0.0.1);
      أو من عنوان IPv4 معبأ
my $ip = new_from_aton NetAddr::IP (inet_aton('127.0.0.1'));
      أو من عنوان IPv4 مصفى بصيغة ثمانية
my $ip = new_no NetAddr::IP '127.012.0.0';
print "العنوان هو ", $ip->addr, " مع قناع ", $ip->mask, "\n" ;
if ($ip->within(new NetAddr::IP "127.0.0.0", "255.0.0.0")) {
    print "إنه عنوان loopback\n";
}
                              # هذا يطبع 127.0.0.1/32
print "يمكنك أيضاً القول $ip...\n";

* الدوال الأربع التالية تُرجع تمثيلات IPv6 لـ:

::                                       = Zeros();
FFFF:FFFF:FFFF:FFFF:FFFF:FFFF:FFFF:FFFF  = Ones();
FFFF:FFFF:FFFF:FFFF:FFFF:FFFF::          = V4mask();
::FFFF:FFFF                              = V4net();
ستُرجع أيضاً تمثيل IPv4 أو IPv6 لعنوان قابل للحل
    من اسم نطاق مؤهل بالكامل (FQDN).

###### DEPRECATED، ستُزال في الإصدار 5 ############

* لقبول العناوين بالتنسيق الذي تُرجعه inet_aton، استدعِ الوحدة كالتالي:
use NetAddr::IP qw(:aton);

###### استخدم new_from_aton بدلاً من ذلك ##########################

* لتمكين استخدام ملفات بيانات قديمة تحتوي على كائنات NetAddr::IP مخزنة باستخدام وحدة Storable.

use NetAddr::IP qw(:old_storable);

* لضغط العديد من الشبكات الفرعية الصغيرة (انظر: "$me->compact($addr1,$addr2,...)"

@compacted_object_list = Compact(@object_list)

* يُرجع مرجعاً إلى قائمة من شبكات "NetAddr::IP" الفرعية بطول قناع $masklen، عندما تكون $number أو أكثر من عناوين @list_of_subnets محتواة في تلك الشبكة الفرعية.

$arrayref = Coalesce($masklen, $number, @list_of_subnets)

* افتراضياً، تُرجع دوال وطرق "NetAddr::IP" عناوين IPv6 بأحرف كبيرة. لتغيير ذلك إلى أحرف صغيرة:

ملاحظة: ينص RFC5952 من أغسطس 2010 على:

4.3. الأحرف الصغيرة
يجب تمثيل الأحرف "a"، "b"، "c"، "d"، "e"، و "f" في عنوان IPv6
    بأحرف صغيرة.

يُوصى بأن تُستدعى التطبيقات الجديدة التي تستخدم NetAddr::IP كالتالي:

use NetAddr::IP qw(:lower);

* لضمان عدم تغير السلوك الحالي لسلسلة IPv6 حتى لو تغير الافتراضي:

use NetAddr::IP qw(:upper);

* لتعيين حد على عدد nets المعالجة أو المُرجعة بواسطة NetAddr::IP.

يعيّن الحد الأقصى لعدد الشبكات التي ستُرجَع عندما يُرجع NetAddr::IP خطأً، كقوة للعدد 2 (القيمة الافتراضية 16 أو 65536 شبكة). كل 2**16 يستهلك حوالي 4 ميغابايت من الذاكرة. 2**20 يستهلك 64 ميغابايت من الذاكرة، 2**24 يستهلك 1 غيغابايت من الذاكرة.

use NetAddr::IP qw(netlimit);
netlimit 20;

الحد الأقصى المسموح به لـ netlimit هو 2**24. تُتجاهل محاولات تعيين حدود أقل من القيمة الافتراضية 16 أو أعلى من الحد الأقصى 24.

يُرجع صحيحاً في حالة النجاح، وإلا "undef".

فك ضغط التوزيعة في دليل مناسب ونفّذ:

perl Makefile.PL
make
make test
make install

NetAddr::IP يعتمد على NetAddr::IP::Util، الذي يُثبَّت افتراضياً مع دواله الأساسية المترجمة باستخدام امتدادات XS الخاصة بـ Perl لبناء مكتبة C. إذا لم يكن لديك مترجم C متاح أو تفضل استخدام لغة Perl الأبطأ لأسباب أخرى، فنفّذ:

perl Makefile.PL -noxs
make
make test
make install

الوصف

توفر هذه الوحدة تجريداً موجهاً للكائنات على عناوين IPv4 أو IPv6، مما يسمح بالتلاعب بها بسهولة. الإصدار 4.xx من NetAddr::IP يعمل مع إصدارات Perl القديمة ومتوافق مع Math::BigInt.

التمثيل الداخلي لجميع كائنات IP هو بترميز IPv6 ذي 128 بت. يمكن خلط كائنات IPv4 و IPv6 بحرية.

المعاملات الموجهة للكائنات

تم تجاوز العديد من المعاملات، كما هو موضح أدناه:

الإسناد ("=")
تم تحسينه لنسخ كائن NetAddr::IP إلى آخر بسرعة.
"->copy()"
معامل الإسناد ("=") يُنفَّذ فقط عندما يُعدَّل الكائن المنسوخ بواسطة معامل متجاوز آخر. انظر الرموز لاستخدام overload لمزيد من التفاصيل.

"->copy()" ينشئ فعلياً كائناً جديداً عند استدعائه.

تطبيع السلسلة
يمكن استخدام الكائن ببساطة كسلسلة. على سبيل المثال، الكود التالي: 
my $ip = new NetAddr::IP '192.168.1.123';
print "$ip\n";

يطبع السلسلة 192.168.1.123/32.

التكافؤ
يمكنك التحقق من التكافؤ باستخدام "eq" أو "==". "eq" يسمح بالمقارنة مع سلاسل عشوائية وكذلك مع كائنات NetAddr::IP. المثال التالي: 
if (NetAddr::IP->new('127.0.0.1','255.0.0.0') eq '127.0.0.1/8')
   { print "Yes\n"; }

يطبع "Yes" على الشاشة.

المقارنة باستخدام "==" تتطلب أن يكون كلا المعاملين كائنات NetAddr::IP.

إذا كانت قيمة CIDR الممثلة بالمعاملين متساوية، تُرجع قيمة صحيحة.

المقارنة باستخدام >، <، >=، <=، <=> و "cmp"
داخلياً، جميع كائنات الشبكة ممثلة بصيغة 128 بت. يُقارَن التمثيل الرقمي للشبكة بواسطة العملية المقابلة. تُجرى المقارنات أولاً على جزء العنوان من الكائن، وإذا كان متساوياً، تُقارَن الأجزاء الرقمية للأقنعة. هذا يؤدي إلى نتيجة متناقضة، مثل: 
/24 > /16

لا يجب مقارنة كائنات الشبكة ذات أقنعة CIDR مختلفة مباشرة، لأن ذلك قد ينتج نتائج غير محددة أو غير متوقعة. بدلاً من ذلك، لتحديد أي كتلة شبكة أكبر أو أصغر، يجب مقارنة:

$ip1->masklen <=> $ip2->masklen
جمع قيمة ثابتة ("+")
يضيف قيمة عددية صحيحة 32 بت إلى جزء العنوان من كائن NetAddr. تغيّر هذه العملية جزء العنوان ليشير إلى عدد المضيفين فوق بداية العنوان الحالي. على سبيل المثال، هذا الكود: 
print NetAddr::IP->new('127.0.0.1/8') + 5;

يطبع 127.0.0.6/8. سيلتف العنوان إلى بث الشبكة. هذا الكود:

print NetAddr::IP->new('10.0.0.1/24') + 255;
outputs 10.0.0.0/24.

يُرجع الكائن دون تغيير إذا كانت القيمة الثابتة غائبة أو خارج النطاق.

2147483647 <= constant >= -2147483648
طرح قيمة ثابتة ("-")
مكمّل جمع قيمة ثابتة.
الفرق ("-")
يُرجع الفرق بين أجزاء العنوان لكائنين NetAddr::IP. يُرجع رقماً صحيحاً موقّعاً 32 بت.

يُرجع undef إذا كان الفرق خارج النطاق.

الزيادة التلقائية
تؤدي الزيادة التلقائية لكائن NetAddr::IP إلى تعديل جزء العنوان إلى عنوان المضيف التالي ضمن الشبكة الفرعية. وسيعود إلى البداية عند عنوان البث ويبدأ مجددًا من عنوان الشبكة.
الإنقاص التلقائي
يؤدي الإنقاص التلقائي لكائن NetAddr::IP إلى عكس ما تفعله الزيادة التلقائية تمامًا، كما هو متوقع.

التسلسل وإلغاء التسلسل

تعريف هذا الموديول خطافات للتعاون مع Storable لتسلسل كائنات "NetAddr::IP"، عبر سلاسل نصية مدمجة وقابلة للقراءة البشرية. يمكنك العودة إلى التنسيق القديم باستدعاء هذا الموديول كالتالي

use NetAddr::IP ':old_storable';

يجب عليك فعل ذلك إذا كانت لديك ملفات بيانات قديمة تحتوي على كائنات NetAddr::IP مخزنة باستخدام موديول Storable.

الطرق

"->new([$addr, [ $mask|IPv6 ]])"
"->new6([$addr, [ $mask]])"
"->new_no([$addr, [ $mask]])"
"->new_from_aton($netaddr)"
new_cis و new_cis6 مهجورتان (DEPRECATED)
"->new_cis("$addr $mask)"
"->new_cis6("$addr $mask)"
تنشئ الطريقتان الأوليان عنوانًا جديدًا بالعنوان المقدم في $addr وقناع شبكة اختياري $mask، والذي يمكن حذفه للحصول على قناع /32 أو /128 لعناوين IPv4 / IPv6 على التوالي.

الطريقة الثالثة "new_no" مخصصة حصريًا لعناوين IPv4 وتُرشّح سلاسل النقاط الرباعية سيئة التنسيق بحثًا عن الأصفار البادئة التي كان NetAddr سيفسرها عادةً كتنسيق ثماني وفقًا لمواصفات inet_aton.

new_from_aton تأخذ عنوان IPv4 محزومًا وتفترض قناع /32. تحل هذه الدالة محل وظيفة :aton المهجورة التي أصبحت معطلة جوهريًا.

تختلف الطريقتان الأخيرتان new_cis و new_cis6 عن new و new6 فقط في أنهما تقبلان تدوين Cisco الشائع لأزواج العنوان/القناع مع مسافة كفاصل بدلاً من الشرطة المائلة (/)

هذه الطرق مهجورة لأن الوظيفة أصبحت الآن مضمنة في طرق "new" الأخرى

أي  ->new_cis('1.2.3.0 24')
      أو
      ->new_cis6('::1.2.3.0 120')

تُعلّم "->new6" و "->new_cis6" العنوان على أنه في فضاء عناوين ipV6 حتى لو كان التنسيق يوحي بغير ذلك.

أي  ->new6('1.2.3.4') ستؤدي إلى ::102:304
العناوين المقدمة إلى ->new بتدوين ipV6 ستبقى
في ذلك التدوين بشكل دائم. أي
      ->new('::1.2.3.4') ستؤدي إلى ::102:304
بينما new('1.2.3.4') ستُطبع كـ 1.2.3.4
انظر "STRINGIFICATION" أدناه.

يمكن أن يكون $addr تقريبًا أي شيء يمكن تحويله إلى عنوان IP بجميع التدوينات التي رأيتها على مر الزمن. ويمكن أن يحتوي اختياريًا على القناع بتدوين CIDR.

تدوين prefix مفهوم، مع القيد أن النطاق المحدد بواسطة البادئة يجب أن يتطابق مع شبكة فرعية صالحة.

يمكن أيضًا فهم العناوين بنفس التنسيق الذي تعيده "inet_aton" أو "gethostbyname"، رغم أنه لا يمكن تحديد قناع لها. الافتراضي هو عدم محاولة التعرف على هذا التنسيق، لأنه يبدو نادر الاستخدام.

use NetAddr::IP ':aton'

إذا استُدعيت بسلسلة فارغة كوسيط، تُعيد 'undef'

اسم نطاق مؤهل بالكامل (Fully Qualified Domain Name) يعيد عنوان ipV4 أو عنوان ipV6، مجسدًا بهذا الترتيب. يمكن تعطيل هذه الميزة غير الموثقة سابقًا باستخدام:

use NetAddr::IP::Lite ':nofqdn';

إذا استُدعيت بدون وسائط، يُفترض 'default'.

إذا استُدعيت بسلسلة فارغة كوسيط، تُعيد 'undef'

"->broadcast()"
تُعيد كائنًا جديدًا يشير إلى عنوان البث لشبكة فرعية معينة. يحتوي عنوان البث على آحاد في جميع مواضع البتات حيث يحتوي قناع الشبكة على بتات صفرية. يُستخدم هذا عادةً لعنونة جميع المضيفين في شبكة فرعية معينة.
"->network()"
تُعيد كائنًا جديدًا يشير إلى عنوان الشبكة لشبكة فرعية معينة. يحتوي عنوان الشبكة على أصفار في جميع البتات حيث تكون بتات قناع الشبكة صفرية. يُستخدم هذا عادةً للإشارة إلى شبكة فرعية.
"->addr()"
تُعيد قيمة عددية تحتوي على جزء العنوان من الكائن كسلسلة نصية IPv4 أو IPv6 حسب الاقتضاء. هذا مفيد للطباعة أو لتمرير جزء العنوان من كائن NetAddr::IP إلى مكونات أخرى تتوقع عنوان IP. إذا كان الكائن عنوان ipV6 أو أُنشئ باستخدام ->new6($ip) فسيُبلَّغ عنه بتنسيق ipV6 السداسي عشري وإلا فسيُبلَّغ عنه بتنسيق النقاط الرباعية فقط إذا كان يقيم في فضاء عناوين ipV4.
"->mask()"
تُعيد قيمة عددية تحتوي على القناع كسلسلة نصية IPv4 أو IPv6 كما هو موضح أعلاه.
"->masklen()"
تُعيد قيمة عددية تمثل عدد البتات الآحادية في القناع.
"->bits()"
تُعيد عرض العنوان بالبتات. عادةً 32 لـ v4 و128 لـ v6.
"->version()"
تُعيد إصدار العنوان أو الشبكة الفرعية. حاليًا يمكن أن يكون إما 4 أو 6.
"->cidr()"
تُعيد قيمة عددية تحتوي على العنوان والقناع بتدوين CIDR. كائن NetAddr::IP يُحوَّل إلى سلسلة إلى نتيجة هذه الدالة. (انظر التعليقات حول ->new6() و ->addr() لتنسيقات الإخراج)
"->aton()"
تُعيد جزء العنوان من كائن NetAddr::IP بنفس تنسيق دالة inet_aton() أو "ipv6_aton" على التوالي. إذا أُنشئ الكائن باستخدام ->new6($ip)، فسيكون العنوان المُعاد دائمًا بتنسيق ipV6، حتى للعناوين في فضاء عناوين ipV4.
"->range()"
تُعيد قيمة عددية تحتوي على العنوان الأساسي وعنوان البث مفصولين بشرطة ومسافات. يُسمى هذا تدوين النطاق.
"->prefix()"
تُعيد قيمة عددية تحتوي على العنوان والقناع بتمثيل بادئة ipV4. هذا مفيد لبعض البرامج التي تتوقع أن يكون إدخالها بهذا التنسيق. ستتضمن هذه الطريقة عنوان البث في الترميز.
"->nprefix()"
تمامًا مثل "->prefix()"، لكنها لا تتضمن عنوان البث.
"->numeric()"
عند استدعائها في سياق عددي، ستُعيد تمثيلاً رقميًا لجزء العنوان من عنوان IP. عند استدعائها في سياق مصفوفة، تُعيد قائمة من عنصرين. العنصر الأول كما هو موضح، والعنصر الثاني هو التمثيل الرقمي لقناع الشبكة.

هذه الطريقة ضرورية لتسلسل تمثيل شبكة فرعية.

"->bigint()"
عند استدعائها في سياق عددي، ستُعيد تمثيل Math::BigInt لجزء العنوان من عنوان IP. عند استدعائها في سياق مصفوفة، تُعيد قائمة من عنصرين، العنصر الأول كما هو موضح، والعنصر الثاني هو تمثيل Math::BigInt لقناع الشبكة.
"->wildcard()"
عند استدعائها في سياق عددي، تُعيد بتات البدل المقابلة للقناع، بتنسيق النقاط الرباعية أو ipV6 حسب الاقتضاء.

عند استدعائها في سياق مصفوفة، تُعيد مصفوفة من عنصرين. العنصر الأول هو جزء العنوان. العنصر الثاني هو ترجمة البدل للقناع.

"->short()"
تُعيد جزء العنوان بتدوين قصير أو مدمج. 
(أي، 127.0.0.1 تصبح 127.1).

تعمل مع كل من V4 و V6.

"->canon()"
تُعيد جزء العنوان بتدوين قانوني كسلسلة نصية. بالنسبة لـ ipV4، هذا هو النقاط الرباعية، وهو نفس القيمة المُعادة من "->addr()". بالنسبة لـ ipV6 فهو وفق RFC5952، وهو نفس القيمة بالأحرف الصغيرة التي تُعيدها "->short()".
"->full()"
تُعيد جزء العنوان بتدوين كامل (FULL) لـ ipV4 و ipV6 على التوالي. 
أي لـ ipV4
  0000:0000:0000:0000:0000:0000:127.0.0.1
     لـ ipV6
  0000:0000:0000:0000:0000:0000:0000:0000

لإجبار عناوين ipV4 على التنسيق الكامل لـ ipV6 استخدم:

"->full6()"
تُعيد جزء العنوان بتدوين ipV6 الكامل
"->full6m()"
تُعيد جزء القناع بتدوين ipV6 الكامل
"$me->contains($other)"
تُعيد صحيحًا عندما يحتوي $me بالكامل على $other. وتُعيد خطأ خلاف ذلك، وتُعيد "undef" إذا لم يكن $me و $other كلاهما كائنات "NetAddr::IP".
"$me->within($other)"
مكمّل "->contains()". تُعيد صحيحًا عندما يكون $me محتوى بالكامل ضمن $other.

لاحظ أن $me و $other يجب أن يكونا كائنات "NetAddr::IP".

C->is_rfc1918()>
تُعيد صحيحًا عندما يكون $me عنوان RFC 1918.
10.0.0.0      -   10.255.255.255  (بادئة 10/8)
172.16.0.0    -   172.31.255.255  (بادئة 172.16/12)
192.168.0.0   -   192.168.255.255 (بادئة 192.168/16)
"->is_local()"
يُرجع true عندما يكون $me عنوان شبكة محلية. 
      أي    ipV4    127.0.0.0 - 127.255.255.255
أو            ipV6    === ::1
"->splitref($bits,[optional $bits1,$bits2,...])"
يُرجع مرجعًا إلى قائمة من الكائنات، تمثل شبكات فرعية لقناع "bits" الناتجة عن تقسيم الكائن الأصلي، الذي يبقى دون تغيير. لاحظ أن $bits يجب أن يكون أطول من القناع الأصلي حتى يكون قابلًا للتقسيم.

شروط الخطأ:

->splitref سيتوقف (DIE) برسالة 'netlimit exceeded'
  إذا تجاوز عدد الكائنات المُرجعة 'netlimit'.
  انظر الدالة 'netlimit' أعلاه (الافتراضي 2**16 أو 65536 شبكة).
->splitref يُرجع undef عندما لا تتناسب C<bits> أو (قائمة bits)
  ضمن الكائن الأصلي.
->splitref يُرجع undef إذا كان قناع ipV4 أو ipV6 أو NetAddr
  المقدم بتنسيق غير مناسب،

bits قد يكون قناع CIDR، أو dot quad أو سلسلة ipV6 أو كائن NetAddr::IP. إذا كان "bits" مفقودًا، يتم تقسيم الكائن إلى جميع العناوين المتاحة ضمن كائن ipV4 أو ipV6 (قناع تلقائي CIDR 32، 128 على التوالي).

مع قائمة "bits" الإضافية الاختيارية، يتم تقسيم الكائن الأصلي إلى أجزاء محددة الحجم بناءً على القائمة. ملاحظة: القائمة القصيرة ستكرر العنصر الأخير. إذا كان العنصر الأخير كبيرًا جدًا بالنسبة لما يتبقى من الكائن بعد فصل الأجزاء الأولى من القائمة، فسيتم إرجاع قائمة "best fits" من الكائنات المتبقية بناءً على ترتيب تصاعدي لقيم CIDR في قائمة "bits".

أي  my $ip = new NetAddr::IP('192.168.0.0/24');
      my $objptr = $ip->split(28, 29, 28, 29, 26);
 لديه خطة تقسيم 28 29 28 29 26 26 26 28
 ويُرجع هذه القائمة من الكائنات
      192.168.0.0/28
      192.168.0.16/29
      192.168.0.24/28
      192.168.0.40/29
      192.168.0.48/26
      192.168.0.112/26
      192.168.0.176/26
      192.168.0.240/28

ملاحظة: أن /26 يتكرر مرتين بعد الطلب الأصلي و /28 يملأ متطلب الكائن المتبقي.

"->rsplitref($bits,[optional $bits1,$bits2,...])"
"->rsplitref" هو نفس "->splitref" أعلاه باستثناء أن خطة التقسيم تُطبق على الكائن الأصلي بترتيب عكسي. 
أي  my $ip = new NetAddr::IP('192.168.0.0/24');
      my @objects = $ip->split(28, 29, 28, 29, 26);
 لديه خطة تقسيم 28 26 26 26 29 28 29 28
 ويُرجع هذه القائمة من الكائنات
      192.168.0.0/28
      192.168.0.16/26
      192.168.0.80/26
      192.168.0.144/26
      192.168.0.208/29
      192.168.0.216/28
      192.168.0.232/29
      192.168.0.240/28
"->split($bits,[optional $bits1,$bits2,...])"
مشابه لـ "->splitref" أعلاه لكنه يُرجع القائمة بدلًا من مرجع قائمة. قد لا ترغب في استخدام هذا إذا كان يُتوقع عدد كبير من الكائنات.
"->rsplit($bits,[optional $bits1,$bits2,...])"
مشابه لـ "->rsplitref" أعلاه لكنه يُرجع القائمة بدلًا من مرجع قائمة. قد لا ترغب في استخدام هذا إذا كان يُتوقع عدد كبير من الكائنات.
"->hostenum()"
يُرجع قائمة المضيفين ضمن شبكة فرعية.

شروط الخطأ:

->hostenum سيتوقف (DIE) برسالة 'netlimit exceeded'
  إذا تجاوز عدد الكائنات المُرجعة 'netlimit'.
  انظر الدالة 'netlimit' أعلاه (الافتراضي 2**16 أو 65536 شبكة).
"->hostenumref()"
نسخة أسرع من "->hostenum()"، تُرجع مرجعًا إلى قائمة.

ملاحظة: hostenum و hostenumref يُبلغان عن صفر (0) مضيفين قابلين للاستخدام في شبكة /31. هذا هو السلوك المتوقع قبل RFC 3021. للإبلاغ عن مضيفين قابلين للاستخدام (2) للاستخدام في شبكات نقطة إلى نقطة، استخدم وسم :rfc3021.

use NetAddr::IP qw(:rfc3021);

سيؤدي هذا إلى جعل hostenum و hostenumref يُرجعان مضيفين قابلين للاستخدام (2) في شبكة /31.

"$me->compact($addr1, $addr2, ...)"
"@compacted_object_list = Compact(@object_list)"
بالنظر إلى قائمة من الكائنات (بما في ذلك $me)، ستقوم هذه الطريقة بضغط جميع العناوين والشبكات الفرعية إلى أكبر (أي، الأقل تحديدًا) شبكات فرعية ممكنة تحتوي بالضبط على جميع الكائنات المعطاة.

لاحظ أنه في الإصدارات السابقة لـ 3.02، إذا تم تغذيتها بنفس الشبكات الفرعية IP عدة مرات، فسيتم إرجاع هذه الشبكات الفرعية. من 3.02 فصاعدًا، تم اعتماد نهج أكثر "صحة" وسيتم إرجاع عنوان واحد فقط.

لاحظ أن $me وجميع $addr يجب أن تكون كائنات "NetAddr::IP".

"$me->compactref(\@list)"
"$compacted_object_list = Compact(\@list)"
كالمعتاد، نسخة أسرع من "->compact()" تُرجع مرجعًا إلى قائمة. لاحظ أن هذه الطريقة تأخذ مرجعًا إلى قائمة بدلًا من ذلك.

لاحظ أن $me يجب أن يكون كائن "NetAddr::IP".

"$me->coalesce($masklen, $number, @list_of_subnets)"
"$arrayref = Coalesce($masklen,$number,@list_of_subnets)"
سيُرجع مرجعًا إلى قائمة من شبكات "NetAddr::IP" الفرعية بطول قناع $masklen، عندما يتم العثور على $number أو أكثر من العناوين من @list_of_subnets محتواة في تلك الشبكة الفرعية.

الشبكات الفرعية من @list_of_subnets بقناع أقصر من $masklen تُمرر "كما هي" إلى قائمة الإرجاع.

الشبكات الفرعية من @list_of_subnets بقناع أطول من $masklen سيتم احتسابها (في الواقع، يتم احتساب عدد عناوين IP) نحو $number.

عند استدعائها كطريقة، ستتضمن المصفوفة $me.

تحذير: يجب أن تكون قائمة الشبكات الفرعية من نفس النوع. أي ipV4 أو ipV6

"->first()"
يُرجع كائنًا جديدًا يمثل أول عنوان IP قابل للاستخدام ضمن الشبكة الفرعية (أي، عنوان المضيف الأول).
"->last()"
يُرجع كائنًا جديدًا يمثل آخر عنوان IP قابل للاستخدام ضمن الشبكة الفرعية (أي، واحد أقل من عنوان البث).
"->nth($index)"
يُرجع كائنًا جديدًا يمثل عنوان IP القابل للاستخدام رقم n ضمن الشبكة الفرعية (أي، عنوان المضيف رقم n). إذا لم يكن هناك عنوان متاح (على سبيل المثال، عندما تكون الشبكة صغيرة جدًا لـ $index من المضيفين)، يتم إرجاع "undef".

الإصدار 4.00 من NetAddr::IP والإصدار 1.00 من NetAddr::IP::Lite ينفذان "->nth($index)" و "->num()" تمامًا كما تنص الوثائق. الإصدارات السابقة تصرفت بشكل مختلف قليلًا وبطريقة غير متسقة. انظر ملف README للتفاصيل.

لاستخدام السلوك القديم لـ "->nth($index)" و "->num()":

use NetAddr::IP::Lite qw(:old_nth);
السلوك القديم:
NetAddr::IP->new('10/32')->nth(0) == undef
NetAddr::IP->new('10/32')->nth(1) == undef
NetAddr::IP->new('10/31')->nth(0) == undef
NetAddr::IP->new('10/31')->nth(1) == 10.0.0.1/31
NetAddr::IP->new('10/30')->nth(0) == undef
NetAddr::IP->new('10/30')->nth(1) == 10.0.0.1/30
NetAddr::IP->new('10/30')->nth(2) == 10.0.0.2/30
NetAddr::IP->new('10/30')->nth(3) == 10.0.0.3/30

لاحظ أنه في كل حالة، يتم تمثيل عنوان البث في مجموعة الإخراج وأن الفهرس 'صفر' دائمًا undef باستثناء شبكة نقطة إلى نقطة /31 أو /127 حيث يوجد بالضبط عنوانان في الشبكة.

السلوك الجديد:
NetAddr::IP->new('10/32')->nth(0)  == 10.0.0.0/32
NetAddr::IP->new('10.1/32'->nth(0) == 10.0.0.1/32
NetAddr::IP->new('10/31')->nth(0)  == 10.0.0.0/31
NetAddr::IP->new('10/31')->nth(1)  == 10.0.0.1/31
NetAddr::IP->new('10/30')->nth(0) == 10.0.0.1/30 
NetAddr::IP->new('10/30')->nth(1) == 10.0.0.2/30 
NetAddr::IP->new('10/30')->nth(2) == undef

لاحظ أن شبكة /32 دائمًا لديها عنوان واحد قابل للاستخدام بينما /31 لديها بالضبط عنوانان قابلان للاستخدام لعنونة نقطة إلى نقطة. الفهرس الأول (0) يُرجع العنوان الذي يلي عنوان الشبكة مباشرة باستثناء /31 أو /127 حيث يُرجع عنوان الشبكة.

"->num()"
اعتبارًا من الإصدار 4.42 من NetAddr::IP والإصدار 1.27 من NetAddr::IP::Lite، /31 و /127 سيُرجعان قيمة num صافية قدرها 2 بدلًا من 0 (صفر) لشبكات نقطة إلى نقطة.

الإصدار 4.00 من NetAddr::IP والإصدار 1.00 من NetAddr::IP::Lite يُرجعان عدد عناوين IP القابلة للاستخدام ضمن الشبكة الفرعية، دون احتساب عنوان البث أو عنوان الشبكة.

الإصدارات السابقة عملت فقط لعناوين ipV4، وأرجعت أقصى مدى 2**32 وأرجعت عدد عناوين IP دون احتساب عنوان البث.
(واحد أكبر من السلوك الجديد)

use NetAddr::IP::Lite qw(:old_nth);

تحذير:

سيحسب NetAddr::IP ويُرجع سلسلة رقمية لـ نطاقات الشبكة بحجم يصل إلى 2**128. هذه القيم هي سلاسل نصية ويمكن لـ perl التعامل معها كأعداد صحيحة للحسابات الرقمية.

Perl على منصات 32 بت يتعامل فقط مع الأعداد الصحيحة حتى 2**32 وعلى منصات 64 بت حتى 2**64.

إذا كنت ترغب في معالجة السلاسل الرقمية التي يُرجعها NetAddr::IP والتي تكون أكبر من 2**32 أو 2**64، على التوالي، يجب عليك تحميل وحدات إضافية مثل Math::BigInt أو bignum أو بعض الحزم المشابهة لإجراء الحسابات الصحيحة.

"->re()"
يُرجع تعبيرًا نمطيًا Perl الذي سيطابق عنوان IP ضمن الشبكة الفرعية المعطاة. الافتراضي هو ترميز ipV4. سيُرجع تعبيرًا نمطيًا ipV6 إذا كان العنوان ليس في نطاق ipV4.
"->re6()"
يُرجع تعبيرًا نمطيًا Perl الذي سيطابق عنوان IP ضمن الشبكة الفرعية المعطاة. يُرجع دائمًا تعبيرًا نمطيًا ipV6.
