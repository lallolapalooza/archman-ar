Net__OAuth(3pm)

الحزمة: perl-net-oauth

التراخيص: PerlArtistic, GPL

Net::OAuth(3) وثائق مساهمة من المستخدم في Perl Net::OAuth(3)

الاسم

Net::OAuth - OAuth 1.0 لـ Perl

الملخص

# مثال خادم الويب (Dancer)
# هذا المثال مبسط لأغراض توضيحية، انظر الكود الكامل في /demo
# لاحظ أن client_id هو مفتاح المستهلك (Consumer Key) و client_secret هو سر المستهلك (Consumer Secret)
use Dancer;
use Net::OAuth::Client;
sub client {
      Net::OAuth::Client->new(
              config->{client_id},
              config->{client_secret},
              site => 'https://www.google.com/',
              request_token_path => '/accounts/OAuthGetRequestToken?scope=https%3A%2F%2Fwww.google.com%2Fm8%2Ffeeds%2F',
              authorize_path => '/accounts/OAuthAuthorizeToken',
              access_token_path => '/accounts/OAuthGetAccessToken',
              callback => uri_for("/auth/google/callback"),
              session => \&session,
      );
}
# إرسال المستخدم لتفويض الخدمة مع مزود الخدمة
get '/auth/google' => sub {
      redirect client->authorize_url;
};
# عاد المستخدم مع الرمز والمحقق المرفقين بعنوان URL.
get '/auth/google/callback' => sub {
      # استخدم رمز التفويض لجلب رمز الوصول
      my $access_token =  client->get_access_token(params->{oauth_token}, params->{oauth_verifier});
      # استخدم رمز الوصول لجلب مورد محمي
      my $response = $access_token->get('/m8/feeds/contacts/default/full');
      # افعل شيئًا بهذا المورد...
      if ($response->is_success) {
        return "رائع، لقد نجح الأمر: " . $response->decoded_content;
      }
      else {
        return "خطأ: " . $response->status_line;
      }
};
dance;

هام

يوفر Net::OAuth واجهة برمجة تطبيقات منخفضة المستوى لقراءة وكتابة رسائل OAuth.

من المحتمل أن تبدأ بـ Net::OAuth::Client.

ملخص

OAuth هو

"بروتوكول مفتوح للسماح بالمصادقة الآمنة لواجهات برمجة التطبيقات بطريقة بسيطة وقياسية من تطبيقات سطح المكتب والويب."

من الناحية العملية، OAuth هي آلية للمستهلك (Consumer) لطلب موارد محمية من مزود الخدمة (Service Provider) نيابة عن المستخدم.

يرجى الرجوع إلى مواصفات OAuth: http://oauth.net/documentation/spec

يوفر Net::OAuth:

  • فئات تغلف رسائل OAuth (طلبات واستجابات).
  • توقيع الرسائل
  • تسلسل الرسائل وتحليلها.
  • طلبات ثنائية الأرجل (المعروفة أيضًا بطلبات بدون رمز، والمعروفة أيضًا بطلبات المستهلك)، انظر "طلبات المستهلك"

لا يوفر Net::OAuth:

  • تغليف المستهلك أو مزود الخدمة
  • تخزين/إدارة الرموز/الأرقام العشوائية/المفاتيح

الوصف

رسائل OAuth

رسالة OAuth هي مجموعة من أزواج المفتاح والقيمة. أنواع الرسائل التالية مدعومة:

الطلبات

  • طلب رمز (Net::OAuth::RequestTokenRequest)
  • طلب رمز وصول (Net::OAuth::AccessTokenRequest)
  • طلب مصادقة المستخدم (Net::OAuth::UserAuthRequest)
  • طلب مورد محمي (Net::OAuth::ProtectedResourceRequest)
  • طلب مستهلك (Net::OAuth::ConsumerRequest) (طلب ثنائي الأرجل / بدون رمز)

الاستجابات

  • استجابة رمز طلب (Net::OAuth::RequestTokenResponse)
  • استجابة رمز وصول (Net::OAuth:AccessTokenResponse)
  • استجابة مصادقة المستخدم (Net::OAuth::UserAuthResponse)

كل نوع من أنواع رسائل OAuth لديه معلمة واحدة أو أكثر مطلوبة، وصفر أو أكثر من المعلمات الاختيارية، ومعظمها يسمح بمعلمات عشوائية.

يجب على المستهلك توقيع جميع طلبات OAuth. ومع ذلك، لا يتم توقيع الاستجابات من مزود الخدمة.

لإنشاء رسالة، أسهل طريقة هي استخدام طرق المصنع (Net::OAuth->request، Net::OAuth->response، Net::OAuth->message). استدعاءات الطرق التالية كلها مكافئة:

$request = Net::OAuth->request('user authentication')->new(%params);
$request = Net::OAuth->request('user_auth')->new(%params);
$request = Net::OAuth->request('UserAuth')->new(%params);
$request = Net::OAuth->message('UserAuthRequest')->new(%params);

الطريقة الأكثر تفصيلاً هي استخدام الفئة مباشرة:

use Net::OAuth::UserAuthRequest;
$request = Net::OAuth::UserAuthRequest->new(%params);

يمكنك أيضًا إنشاء رسالة عن طريق إلغاء تسلسلها من عنوان Authorization، أو URL، أو جدول البحث، أو جسم POST

$request = Net::OAuth->request('protected resource')->from_authorization_header($ENV{HTTP_AUTHORIZATION}, %api_params);
$request = Net::OAuth->request('protected resource')->from_url($url, %api_params);
$request = Net::OAuth->request('protected resource')->from_hash({$q->Vars}, %api_params); # CGI
$request = Net::OAuth->request('protected resource')->from_hash($c->request->params, %api_params); # Catalyst
$response = Net::OAuth->response('request token')->from_post_body($response_content, %api_params);

لاحظ أن طرق إلغاء التسلسل (على عكس new()) تتوقع أن تكون معلمات بروتوكول OAuth مسبوقة بـ 'oauth_'، كما تتوقع في رسالة OAuth صالحة.

قبل إرسال طلب، يجب على المستهلك أولاً توقيعه:

$request->sign;

عند تلقي طلب، يجب على مزود الخدمة أولاً التحقق من التوقيع:

die "فشل التحقق من التوقيع" unless $request->verify;

عند إرسال رسالة، تكون الخطوة الأخيرة هي تسلسلها وإرسالها إلى المكان الذي تحتاج إليه. طرق التسلسل التالية متاحة:

$response->to_post_body # جسم POST بتنسيق application/x-www-form-urlencoded
$request->to_url # سلسلة الاستعلام لـ URL
$request->to_authorization_header # قيمة رأس HTTP Authorization
$request->to_hash # جدول يمكن استخدامه لبعض التسلسلات الأخرى

معلمات API مقابل معلمات الرسالة

يحدد Net::OAuth 'معلمات الرسالة' كمعلمات جزء من رسالة OAuth المرسلة. وتشمل هذه أي معلمة بروتوكول (مسبوقة بـ 'oauth_' في الرسالة)، وأي معلمات رسالة إضافية (جدول extra_params).

'معلمات API' هي المعلمات المطلوبة لبناء كائن رسالة لا يتم إرسالها مع الرسالة، على سبيل المثال، consumer_secret، token_secret، request_url، request_method.

هناك طرق مختلفة لفحص فئة الرسالة لمعرفة المعلمات المحددة:

$request->required_message_params;
$request->optional_message_params;
$request->all_message_params;
$request->required_api_params;
$request->optional_api_params;
$request->all_api_params;
$request->all_params;

على سبيل المثال:

use Net::OAuth;
use Data::Dumper;
print Dumper(Net::OAuth->request("protected resource")->required_message_params);
$VAR1 = [
         'consumer_key',
         'signature_method',
         'timestamp',
         'nonce',
         'token'
       ];

الوصول إلى المعلمات

يمكن الحصول على جميع المعلمات وتعيينها باستخدام طرق الوصول. على سبيل المثال:

my $consumer_key = $request->consumer_key;
$request->request_method('POST');

معلمة عنوان URL للطلب (REQUEST_URL)

تتم إزالة أي معلمات استعلام في request_url وإضافتها إلى جدول extra_params عند إنشاء التوقيع.

على سبيل المثال، الطلبات التالية متكافئة تقريبًا:

 my $request = Net::OAuth->request('Request Token')->new(
  %params,
  request_url => 'https://photos.example.net/request_token',
  extra_params => {
   foo => 'bar'
  },
);
 my $request = Net::OAuth->request('Request Token')->new(
  %params,
  request_url => 'https://photos.example.net/request_token?foo=bar',
 );

سيظل استدعاء $request->request_url يعيد ما قمت بتعيينه له في الأصل. إذا كنت ترغب في الحصول على عنوان URL للطلب مع إزالة معلمات الاستعلام، يمكنك القيام بذلك:

my $url = $request->normalized_request_url;

  • HMAC-SHA1
