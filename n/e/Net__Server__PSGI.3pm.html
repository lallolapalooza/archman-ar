Net__Server__PSGI(3pm)

الحزمة: perl-net-server

التراخيص: PerlArtistic, GPL

Net::Server::PSGI(3) وثائق Perl المساهمة من المستخدم Net::Server::PSGI(3)

الاسم

Net::Server::PSGI - فئة خادم PSGI HTTP أساسية تعتمد على Net::Server

اختبار سطر واحد

perl -e 'use base qw(Net::Server::PSGI); main->run(port => 8080, ipv => "*")'
# يقوم بتشغيل خادم صدى افتراضي

الملخص

use base qw(Net::Server::PSGI);
__PACKAGE__->run(app => \&my_echo_handler); # سيربط المنفذ IPv4 80
sub my_echo_handler {
    my $env = shift;
    my $txt = qq{<form method="post" action="/bam"><input type="text" name="foo"><input type="submit"></form>\n};
    require Data::Dumper;
    local $Data::Dumper::Sortkeys = 1;
    require CGI::PSGI;
    my $form = {};
    my $q = CGI::PSGI->new($env);
    $form->{$_} = $q->param($_) for $q->param;
    $txt .= "<pre>".Data::Dumper->Dump([$env, $form], ['env', 'form'])."</pre>";
    return [200, ['Content-type', 'text/html'], [$txt]];
}

الوصف

إذا كنت تريد تجربة PSGI أكثر اكتمالاً، سيكون من الحكمة النظر إلى مجموعات وحدات Plack و Starman. الغرض من Net::Server::PSGI هو أن يكون بوابة سهلة إلى PSGI. ولكن للحصول على أقصى استفادة من كل ما تقدمه PSGI، يجب عليك مراجعة Plack و Plack::Middleware. إذا كنت تحتاج فقط إلى شيء أكثر بدائية، فقد يكون Net::Server::PSGI مناسبًا لك.

تأخذ Net::Server::PSGI مستوى Net::Server::HTTP خطوة أبعد. تبدأ بنوع أساسي MultiType افتراضيًا إلى Net::Server::Fork. من السهل تغييره إلى أي من نكهات Net::Server الأخرى عن طريق تمرير server_type => $other_flavor في تكوين الخادم. تم افتراض المنفذ أيضًا إلى المنفذ 80 - ولكن يمكن تغييره بسهولة إلى منفذ آخر من خلال تكوين الخادم. يمكنك أيضًا إضافة SSL بسهولة بالغة عن طريق تضمين، proto=>"ssl" وتوفير SSL_cert_file و SSL_key_file.

على سبيل المثال، إليك خادم أساسي سيربط بجميع الواجهات، وسيتحدث HTTP على المنفذ 8080 بالإضافة إلى HTTPS على 8443، وسيتحدث IPv4، بالإضافة إلى IPv6 إذا كان متاحًا.

use base qw(Net::Server::PSGI);
__PACKAGE__->run(
    port  => [8080, "8443/ssl"],
    ipv   => '*', # IPv6 إذا كان متاحًا
    SSL_key_file  => '/my/key',
    SSL_cert_file => '/my/cert',
);

الأساليب

"process_request"
تم تجاوز هذا الأسلوب في Net::Server::PSGI - يجب ألا تستخدمه أثناء استخدام Net::Server::PSGI. يقوم هذا الأسلوب المتجاوز بتحليل البيئة وإعداد إنذارات الطلب والتعامل مع فشل الموت. يستدعي process_psgi_request بمجرد أن يكون الطلب جاهزًا وتم تحليل الرؤوس.
"process_psgi_request"
يستخدم عندما يكون psgi_enabled صحيحًا. خلال هذا الأسلوب، سيتم استدعاء find_psgi_handler لإرجاع معالج استجابة PSGI المناسب. بمجرد الانتهاء، يتم استخدام print_psgi_headers و print_psgi_body لطباعة الاستجابة. انظر PSGI.

عادةً لا ينبغي تجاوز هذا الأسلوب. بدلاً من ذلك، يجب توفير أسلوب مناسب للعثور على التطبيق لـ find_psgi_handler أو app.

"find_psgi_handler"
يستخدم للبحث عن معالج PSGI المناسب. يتم تمرير مرجع إلى hashref البيئة المحللة بالفعل $env. سيتم تهيئة PATH_INFO إلى جزء المسار الكامل لـ URI. سيتم تهيئة SCRIPT_NAME إلى سلسلة فارغة. يجب أن يقوم هذا المعالج بتعيين القيم المناسبة لـ SCRIPT_NAME و PATH_INFO اعتمادًا على المسار المطابق. يجب إرجاع مرجع إلى رمز للمعالج. سيقوم find_psgi_handler الافتراضي باستدعاء الأسلوب "app". إذا فشل ذلك، يتم إرجاع مرجع إلى psgi_echo_handler كتطبيق افتراضي. 
sub find_psgi_handler {
    my ($self, $env) = @_;
    if ($env->{'PATH_INFO'} && $env->{'PATH_INFO'} =~ s{^ (/foo) (?= $ | /) }{}x) {
        $env->{'SCRIPT_NAME'} = $1;
        return \&foo_app;
    }
    return $self->SUPER::find_psgi_handler($env);
}
"app"
إرجاع مرجع إلى التطبيق الذي يتم تقديمه. يجب أن يكون هذا تطبيق PSGI صالحًا. انظر PSGI. افتراضيًا، سيبحث في قيمة خيار التكوين "app". يمكن أيضًا استخدام الأسلوب "app" لتعيين خيار التكوين "app". 
package MyApp;
use base qw(Net::Server::PSGI);
sub default_server_type { 'Prefork' }
sub my_app {
    my $env = shift;
    return [200, ['Content-type', 'text/html'], ["Hello world"]];
}
MyApp->run(app => \&my_app);
# أو
sub app { \&my_app }
MyApp->run;
# أو
my $server = MyApp->new;
$server->app(\&my_app);
$server->run;

الخيارات

بالإضافة إلى وسائط سطر الأوامر لفئات Net::Server::HTTP الأساسية، يمكنك أيضًا تعيين الخيارات التالية.

app
يجب أن يعيد مرجع رمز لتطبيق PSGI. يتم إرجاعه بواسطة الأسلوب app.
