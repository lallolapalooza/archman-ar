Net__DNS__RR__OPT(3pm)

الحزمة: perl-net-dns

التراخيص: PerlArtistic

Net::DNS::RR::OPT(3) وثائق Perl المساهمة من المستخدم Net::DNS::RR::OPT(3)

الاسم

Net::DNS::RR::OPT - سجل مورد DNS OPT

الملخص

use Net::DNS;
my $packet = Net::DNS::Packet->new( ... );
$packet->header->do(1);         # علم الرأس الموسع
$packet->edns->UDPsize(1232);   # حجم حمولة UDP
$packet->edns->option( 'NSID' => {'OPTION-DATA' => 'rawbytes'} );
$packet->edns->option( 'DAU'  => [8, 10, 13, 14, 15, 16] );
$packet->edns->option( 'TCP-KEEPALIVE' => 200 );
$packet->edns->option( 'EXTENDED-ERROR' => {'INFO-CODE' => 123} );
$packet->edns->option( '65023' => {'BASE16' => '076578616d706c6500'} );
$packet->edns->print;
;; {    "EDNS-VERSION": 0,
;;      "FLAGS":        32768,
;;      "RCODE":        0,
;;      "UDPSIZE":      1232,
;;      "OPTIONS":      [
;;              {"NSID": "7261776279746573"},
;;              {"DAU": [ 8, 10, 13, 14, 15, 16 ]},
;;              {"TCP-KEEPALIVE": {"TIMEOUT": 200}},
;;              {"EXTENDED-ERROR": {"INFO-CODE": 123, "EXTRA-TEXT": ""}},
;;              {"65023": {"BASE16": "076578616d706c6500"}} ]
;;      }

الوصف

سجل مورد EDNS OPT الزائف.

يدعم سجل OPT امتدادات بروتوكول EDNS ولا يُقصد به إنشاء تطبيقات مباشرة أو الوصول إليها أو تعديلها بواسطة تطبيقات المستخدم.

تتم جميع ميزات EDNS بشكل غير مباشر من خلال عمليات على الكائنات التي تم إرجاعها بواسطة طرق الإنشاء $packet->header و$packet->edns. يجب أن تكون الآليات الأساسية مخفية تمامًا عن المستخدم.

الطرق

الطرق المتاحة هي تلك الموروثة من الفئة الأساسية المعززة بالطرق الخاصة بالنوع المعرفة في هذه الحزمة.

يتم تثبيط استخدام ميزات الحزمة غير الموثقة أو الوصول المباشر إلى هياكل البيانات الداخلية وقد يؤدي ذلك إلى إنهاء البرنامج أو سلوك آخر غير متوقع.

الإصدار

$version = $packet->edns->version;

إصدار EDNS الذي يدعمه سجل OPT هذا.

UDPsize

$size = $packet->edns->UDPsize;
$packet->edns->UDPsize($size);

UDPsize() يعلن عن الحد الأقصى لحجم حزمة UDP (أوكتيه) التي يمكن إعادة تجميعها في المكدس الشبكي للمضيف الأصلي.

rcode

$extended_rcode = $packet->header->rcode;

رمز الاستجابة الممتد المكون من 12 بت. يتم الحصول على أهم 8 بتات من سجل OPT. تقع أقل 4 بتات في رأس الحزمة.

flags

$do = $packet->header->do;
$packet->header->do(1);
$edns_flags = $packet->edns->flags;

حقل 16 بت يحتوي على أعلام رأس EDNS الموسعة.

options

my @options = $packet->edns->options;

عند استدعائها في سياق قائمة، ترجع options() قائمة برموز الخيارات الموجودة في سجل OPT.

option

my $octets = $packet->edns->option('COOKIE');
my $base16 = unpack 'H*', $octets;

عند استدعائها في سياق قياسي مع وسيط واحد، ترجع option() قيمة الخيار المحدد كسلسلة أوكتيه غير مفسرة. ترجع الطريقة undef إذا كان الخيار غائبًا.

$packet->edns->option( 'COOKIE' => {'OPTION-DATA' => $octets} );
$packet->edns->option( '10'     => {'BASE16' => $base16} );

يمكن إضافة خيار أو استبداله عن طريق توفير زوج (الاسم، القيمة). يتم حذف الخيار إذا كانت القيمة غير معرفة.

my ($structure) = $packet->edns->option("DAU");
my $array       = $$structure{"DAU"};
my @algorithms  = @$array;
my ($structure) = $packet->edns->option(15);
my $table       = $$structure{"EXTENDED-ERROR"};
my $info_code   = $$table{'INFO-CODE'};
my $extra_text  = $$table{'EXTRA-TEXT'};

عند استدعائها في سياق قائمة مع وسيط واحد، ترجع option() تمثيلًا منظمًا للخيار المحدد.

يمكن استخدام أشكال مماثلة من صيغة المصفوفة أو الجدول لإنشاء قيمة الخيار:

$packet->edns->option( 'DAU' => [8, 10, 13, 14, 15, 16] );
$packet->edns->option( 'EXTENDED-ERROR' => {
                                'INFO-CODE'  => 123,
                                'EXTRA-TEXT' => ""
                                } );
