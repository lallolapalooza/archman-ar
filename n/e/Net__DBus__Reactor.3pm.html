Net__DBus__Reactor(3pm)

الحزمة: perl-net-dbus

التراخيص: GPL, PerlArtistic

Net::DBus::Reactor(3) وثائق Perl المساهمة من المستخدم Net::DBus::Reactor(3)

الاسم

Net::DBus::Reactor - حلقة أحداث التطبيق

الملخص

إنشاء وتشغيل حلقة أحداث:

use Net::DBus::Reactor;
my $reactor = Net::DBus::Reactor->main();
$reactor->run();

إدارة بعض معالجات الملفات

$reactor->add_read($fd,
                   Net::DBus::Callback->new(method => sub {
                      my $fd = shift;
                      ...قراءة بعض البيانات...
                   }, args => [$fd]));
$reactor->add_write($fd,
                    Net::DBus::Callback->new(method => sub {
                       my $fd = shift;
                       ...كتابة بعض البيانات...
                    }, args => [$fd]));

تمكين/تعطيل معالج بشكل مؤقت

# تعطيل
$reactor->toggle_read($fd, 0);
# تمكين
$reactor->toggle_read($fd, 1);

إزالة معالج بشكل دائم

$reactor->remove_read($fd);

إدارة مهلة منتظمة كل 100 مللي ثانية

my $timer = $reactor->add_timeout(100,
                                  Net::DBus::Callback->new(
           method => sub {
              ...معالجة الإنذار...
           }));

تمكين/تعطيل مؤقت بشكل مؤقت

# تعطيل
$reactor->toggle_timeout($timer, 0);
# تمكين
$reactor->toggle_timeout($timer, 1);

إزالة مؤقت بشكل دائم

$reactor->remove_timeout($timer);

إضافة خطاف (hook) بعد الإرسال

my $hook = $reactor->add_hook(Net::DBus::Callback->new(
      method => sub {
         ... القيام ببعض العمل...
      }));

إزالة خطاف

$reactor->remove_hook($hook);

الوصف

توفر هذه الفئة حلقة أحداث للأغراض العامة لمضاعفة أحداث الإدخال/الإخراج والمهلات الزمنية في عملية واحدة. يتم تنفيذ التنفيذ الأساسي باستخدام استدعاء النظام select. يمكن تسجيل مقابض الملفات لمراقبة أحداث القراءة والكتابة والاستثناء (بيانات خارج النطاق). يمكن تسجيل المهلات الزمنية لتنتهي بتردد دوري. يتم تنفيذ هذه باستخدام معلمة المهلة الزمنية لاستدعاء النظام select. نظرًا لأن هذه المعلمة تمثل فقط حدًا أقصى للمدة التي يُسمح لاستدعاء النظام select بالنوم فيها، فقد يختلف التردد الفعلي للمهل الزمنية. تحت الحمل العادي، يتراوح هذا الاختلاف عادةً 10 مللي ثانية. أخيرًا، يمكن تسجيل الخطافات التي سيتم استدعاؤها في كل تكرار لحلقة الأحداث (أي بعد معالجة أحداث الملف، أو المهلات الزمنية التي يشير إليها عودة استدعاء النظام select).

الطرق

my $reactor = Net::DBus::Reactor->new();
ينشئ حلقة أحداث جديدة جاهزة لمراقبة مقابض الملفات، أو إنشاء مهلات زمنية. باستثناء الظروف غير العادية جدًا (لا يمكنني التفكير في أمثلة لها) ليس من الضروري أو المرغوب فيه إنشاء مثيلات مفاعل جديدة بشكل صريح. بدلاً من ذلك، قم باستدعاء الطريقة الرئيسية للحصول على مقبض للمثيل الوحيد.
$reactor = Net::DBus::Reactor->main;
إرجاع مقبض للمثيل الوحيد للمفاعل. هذه هي الطريقة الموصى بها للحصول على مفاعل، حيث إنها تزيل الحاجة إلى قيام الوحدات بتمرير مقابض إلى مفاعلاتها التي تم إنشاؤها بشكل خاص.
$reactor->manage($connection);
$reactor->manage($server);
يسجل كائن "Net::DBus::Binding::Connection" أو "Net::DBus::Binding::Server" للإدارة بواسطة حلقة الأحداث. يتضمن هذا بشكل أساسي ربط ردود الاتصال للمراقبة والمهلة الزمنية بحلقة الأحداث. بالنسبة للاتصالات، سيقوم أيضًا بتسجيل خطاف لاستدعاء طريقة "dispatch" بشكل دوري.
$reactor->run();
يبدأ مراقبة حلقة الأحداث لأي مقابض ملفات ومهلات زمنية مسجلة. يجب أن يكون هناك مقبض ملف واحد على الأقل، أو مؤقت مسجل قبل تشغيل المفاعل، وإلا فإنه سيخرج فورًا. سيتم تشغيل المفاعل حتى يتم إزالة جميع مقابض الملفات المسجلة، أو المهلات الزمنية، أو تعطيلها. يمكن إيقاف المفاعل بشكل صريح عن طريق استدعاء الطريقة "shutdown".
$reactor->shutdown();
إغلاق المفاعل بشكل صريح بعد معالجة الأحداث المعلقة.
$reactor->step();
أداء تكرار واحد لحلقة الأحداث، والغرق حتى يحدث حدث على مقبض ملف مسجل، أو تحدث مهلة زمنية. هذه الطريقة ليست مطلوبة بشكل عام في الاستخدام اليومي.
$reactor->add_read($fd, $callback[, $status]);
يسجل مقبض ملف لمراقبة أحداث القراءة. تحدد معلمة $callback إما مرجع رمز لدالة فرعية، أو مثيل لكائن "Net::DBus::Callback" لاستدعائه في كل مرة يحدث فيها حدث. تحدد المعلمة الاختيارية $status قيمة منطقية لتحديد ما إذا كانت المراقبة مفعلة مبدئيًا.
$reactor->add_write($fd, $callback[, $status]);
يسجل مقبض ملف لمراقبة أحداث الكتابة. تحدد معلمة $callback إما مرجع رمز لدالة فرعية، أو مثيل لكائن "Net::DBus::Callback" لاستدعائه في كل مرة يحدث فيها حدث. تحدد المعلمة الاختيارية $status قيمة منطقية لتحديد ما إذا كانت المراقبة مفعلة مبدئيًا.
$reactor->add_exception($fd, $callback[, $status]);
يسجل مقبض ملف لمراقبة أحداث الاستثناء. تحدد معلمة $callback إما مرجع رمز لدالة فرعية، أو مثيل لكائن "Net::DBus::Callback" لاستدعائه في كل مرة يحدث فيها حدث. تحدد المعلمة الاختيارية $status قيمة منطقية لتحديد ما إذا كانت المراقبة مفعلة مبدئيًا.
my $id = $reactor->add_timeout($interval, $callback, $status);
يسجل مهلة زمنية جديدة لتنتهي كل $interval مللي ثانية. تحدد معلمة $callback إما مرجع رمز لدالة فرعية، أو مثيل لكائن "Net::DBus::Callback" لاستدعائه في كل مرة تنتهي فيها المهلة الزمنية. تحدد المعلمة الاختيارية $status قيمة منطقية لتحديد ما إذا كانت المهلة الزمنية مفعلة مبدئيًا. المعلمة المرجعة هي معرف فريد يمكن استخدامه لاحقًا لإزالة المهلة الزمنية أو تعطيلها.
$reactor->remove_timeout($id);
يزيل مهلة زمنية مسجلة مسبقًا محددة بواسطة المعلمة $id.
$reactor->toggle_timeout($id, $status[, $interval]);
يحدث حالة مهلة زمنية مسجلة مسبقًا محددة بواسطة المعلمة $id. تحدد المعلمة $status ما إذا كانت المهلة الزمنية ستكون ممكّنة أو معطلة، بينما يمكن استخدام المعلمة الاختيارية $interval لتغيير فترة المهلة الزمنية.
my $id = $reactor->add_hook($callback[, $status]);
يسجل خطافًا جديدًا ليتم تشغيله في كل تكرار لحلقة الأحداث. تحدد معلمة $callback إما مرجع رمز لدالة فرعية، أو مثيل لفئة "Net::DBus::Callback" لاستدعائه. تحدد المعلمة $status ما إذا كان الخطاف مفعلًا مبدئيًا، أو معطلًا. المعلمة المرجعة هي معرف فريد يجب استخدامه لاحقًا لإزالة الخطاف أو تعطيله.
$reactor->remove_hook($id)
يزيل الخطاف المسجل مسبقًا والمعرّف بالمعلمة $id.
$reactor->toggle_hook($id, $status)
يحدث حالة الخطاف المسجل مسبقًا والمعرّف بالمعلمة $id. تحدد المعلمة $status ما إذا كان الخطاف سيتم تمكينه أو تعطيله.
$reactor->remove_read($fd);
$reactor->remove_write($fd);
$reactor->remove_exception($fd);
يزيل مراقبة على مقبض الملف $fd.
$reactor->toggle_read($fd, $status);
$reactor->toggle_write($fd, $status);
$reactor->toggle_exception($fd, $status);
يحدث حالة مراقبة على مقبض الملف $fd. تحدد المعلمة $status ما إذا كانت المراقبة ستكون ممكّنة أو معطلة.
