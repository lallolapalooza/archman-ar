Net__DNS__RR__NSEC3(3pm)

الحزمة: perl-net-dns

التراخيص: PerlArtistic

Net::DNS::RR::NSEC3(3) وثائق مساهمة من مستخدمي Perl Net::DNS::RR::NSEC3(3)

الاسم

Net::DNS::RR::NSEC3 - سجل مورد DNS NSEC3

الملخص

use Net::DNS;
$rr = Net::DNS::RR->new('name NSEC3 algorithm flags iterations salt hnxtname');

الوصف

فئة لسجل مورد DNS NSEC3.

يوفر سجل مورد DNS NSEC3 (RR) إنكارًا للوجود موثقًا لمجموعات موارد DNS.

يسرد سجل مورد NSEC3 (RR) أنواع الموارد الموجودة في الاسم الأصلي لسجل NSEC3. يتضمن اسم المالك المجزأ التالي بترتيب تجزئة المنطقة. يشير مجموع سجلات مورد NSEC3 (RR) الكامل في منطقة إلى أي مجموعات موارد (RRsets) موجودة للاسم الأصلي للسجل وتشكل سلسلة من الأسماء المجزأة في المنطقة.

الأساليب

الأساليب المتاحة هي تلك الموروثة من الفئة الأساسية، بغض النظر عن الأساليب المحددة المعرفة في هذه الحزمة.

يتم تثبيط استخدام ميزات الحزمة غير الموثقة أو الوصول المباشر إلى الهياكل الداخلية وقد يؤدي إلى إنهاء البرنامج أو سلوك غير متوقع.

الخوارزمية

$algorithm = $rr->algorithm;
$rr->algorithm( $algorithm );

يتم تمثيل حقل الخوارزمية المكون من 8 بت كرقم عشري غير موقع، ولكن يمكن تحديده باستخدام اسم الخوارزمية فقط.

يمكن أيضًا استدعاء algorithm() كأسلوب فئة أو دالة بسيطة لإجراء ترجمة الرمز التذكيري والرقمي.

الأعلام

$flags = $rr->flags;
$rr->flags( $flags );

يتم تمثيل حقل الأعلام كرقم عشري غير موقع يتم تفسيره على أنه ثماني قيم منطقية متسلسلة.

optout
$rr->optout(1);
if ( $rr->optout ) {
        ...
}

علم إلغاء الاشتراك المنطقي.

التكرارات

$iterations = $rr->iterations;
$rr->iterations( $iterations );

يتم تمثيل حقل التكرارات كرقم عشري غير موقع. القيمة تتراوح بين 0 و 65535، شاملة.

الملح

$salt = $rr->salt;
$rr->salt( $salt );

يتم تمثيل حقل الملح كسلسلة مستمرة من الأرقام السداسية العشرية. يتم استخدام علامة (-) (بدون علامات اقتباس) في شكل سلسلة للإشارة إلى أن حقل الملح غير موجود.

saltbin

$saltbin = $rr->saltbin;
$rr->saltbin( $saltbin );

حقل الملح كسلسلة من البايتات.

hnxtname

$hnxtname = $rr->hnxtname;
$rr->hnxtname( $hnxtname );

يشير اسم المالك المجزأ التالي إلى العقدة التالية التي تحتوي على بيانات سلطوية أو تحتوي على نقطة تفويض لسجل NS.

typelist

@typelist = $rr->typelist;
$typelist = $rr->typelist;
$rr->typelist( @typelist );

typelist() يحدد أنواع مجموعات الموارد (RRsets) الموجودة في اسم النطاق المقابل لسجل NSEC3. عند استدعائه في سياق قياسي، يتم تحويل القائمة إلى سلسلة.

typemap

$exists = $rr->typemap($rrtype);

typemap() يعيد قيمة منطقية صحيحة إذا كان نوع مورد (RR) المحدد يحدث في خريطة أنواع سجل NSEC3.

match

$matched = $rr->match( 'example.foo' );

match() يعيد قيمة منطقية صحيحة إذا كان تجزئة اسم النطاق الوسيط يتطابق مع اسم المالك المجزأ لسجل NSEC3.

covers

$covered = $rr->covers( 'example.foo' );

covers() يعيد قيمة منطقية صحيحة إذا كان تجزئة اسم النطاق الوسيط، أو سلف هذا الاسم، يقع بين اسم المالك واسم المالك المجزأ التالي لسجل NSEC3.

encloser, nextcloser, wildcard

$encloser = $rr->encloser( 'example.foo' );
print "encloser: $encloser\n" if $encloser;

encloser() يعيد اسم ضامن يمكن إثباته لاسم الاستعلام الوسيط المستمد من سجل NSEC3.

nextcloser() يعيد اسم القريب التالي، وهو اسم أطول من القريب الأقرب. هذا صالح فقط بعد أن يعيد encloser() اسم نطاق صالحًا.

wildcard() يعيد اسم البرية غير الموسع الذي قد يكون قد تم تصنيعه من اسم القريب التالي. هذا صالح فقط بعد أن يعيد encloser() اسم نطاق صالحًا.
