Net__DBus__Test__MockConnection(3pm)

الحزمة: perl-net-dbus

التراخيص: GPL, PerlArtistic

Net::DBus::Test::MockConnection(3) وثائق Perl المقدمة من المستخدم Net::DBus::Test::MockConnection(3)

الاسم

Net::DBus::Test::MockConnection - محاكاة اتصال بحافلة لغرض الاختبارات الوحدوية

الملخص

use Net::DBus;
my $bus = Net::DBus->test
# تسجيل خدمة، والكائن المراد اختباره
use MyObject
my $service = $bus->export_service("org.example.MyService");
my $object = MyObject->new($service);
# الحصول على الخدمة وإجراء الاختبارات
my $remote_service = $bus->get_service('org.example.MyService');
my $remote_object = $service->get_object("/org/example/MyObjct");
# هذا يجتاز الاتصال الوهمي، ويستدعي في النهاية 'testSomething' على $object أعلاه.
$remote_object->testSomething()

الوصف

يوفر هذا الكائن تطبيقًا وهميًا لـ Net::DBus::Binding::Connection، مما يتيح محاكاة حافلة رسائل "في الذاكرة" فقط. يهدف هذا إلى تسهيل إنشاء اختبارات وحدوية للخدمات التي قد تحتاج بخلاف ذلك إلى استدعاء كائنات أخرى على حافلة رسائل حية. يتم استخدامه كرفيق لوحدة Net::DBus::Test::MockObject، وهي الطريقة التي يتم بها توفير كائنات وهمية على الحافلة الوهمية.

الطرق

my $con = Net::DBus::Test::MockConnection->new()
إنشاء مثيل جديد لكائن اتصال وهمي. لا يلزم عادةً إنشاء مثيلات لهذا الكائن مباشرة، بدلاً من ذلك، يمكن استخدام الطريقة "test" على كائن Net::DBus للحصول على مقبض لحافلة اختبار.
$con->send($message)
إرسال رسالة عبر الاتصال الوهمي. إذا كانت الرسالة عبارة عن استدعاء طريقة، فسيتم توجيهها مباشرة إلى أي كائن وهمي مطابق مسجل. إذا كانت الرسالة عبارة عن خطأ أو إرجاع طريقة، فسيتم إتاحتها كقيمة إرجاع للطريقة "send_with_reply_and_block". إذا كانت الرسالة عبارة عن إشارة، فسيتم وضعها في قائمة انتظار للمعالجة بواسطة الطريقة "dispatch".
$bus->request_name($service_name)
محاكاة إرسال طلب إلى الحافلة لتسجيل الاسم المعروف المحدد في المعلمة $service_name. في الواقع، هذه مجرد عملية لا تفعل شيئًا، مما يعطي انطباعًا بأن الاسم تم تسجيله بنجاح.
my $reply = $con->send_with_reply_and_block($msg)
إرسال رسالة عبر الاتصال الوهمي والانتظار للرد. يجب أن تكون $msg مثيلاً لـ "Net::DBus::Binding::Message::MethodCall" وسيكون الرد $reply مثيلاً لـ "Net::DBus::Binding::Message::MethodReturn". من الممكن أيضًا حدوث خطأ، حيث يتم إلقاء الخطأ المسجل في فئة "Net::DBus::Error".
$con->dispatch;
يقوم بتوزيع أي رسائل معلقة في قائمة الانتظار الواردة إلى معالجات رسائلها. يجب على مجموعات الاختبار استدعاء هذه الطريقة كلما توقعت وجود إشارات معلقة للتعامل معها.
$con->add_filter($coderef);
يضيف مرشحًا إلى الاتصال سيتم استدعاؤه كلما تم استلام رسالة. يجب أن يكون $coderef مرجعًا إلى روتين فرعي، والذي يعيد قيمة صحيحة إذا كان يجب تصفية الرسالة، أو قيمة خاطئة إذا كان يجب إجراء توزيع الرسالة العادي.
$bus->add_match($rule)
تسجيل قاعدة مطابقة للإشارة مع متحكم الحافلة، مما يسمح بتوجيه الإشارات المبثوثة المطابقة إلى هذا العميل. في الواقع، هذه مجرد عملية لا تفعل شيئًا، مما يعطي انطباعًا بأن المطابقة تم تسجيلها بنجاح.
$bus->remove_match($rule)
إلغاء تسجيل قاعدة مطابقة للإشارة مع متحكم الحافلة، مما يمنع توجيه المزيد من الإشارات المبثوثة إلى هذا العميل. في الواقع، هذه مجرد عملية لا تفعل شيئًا، مما يعطي انطباعًا بأن المطابقة تم إلغاء تسجيلها بنجاح.
$con->register_object_path($path, \&handler)
يسجل معالجًا للرسائل التي يتطابق مسارها مع المسار المحدد في المعلمة $path. سيتم استدعاء مرجع الكود المقدم مع معلمتين، كائن الاتصال الذي تم استلام الرسالة عليه، والرسالة المراد معالجتها (مثيل لفئة "Net::DBus::Binding::Message").
$con->register_fallback($path, \&handler)
يسجل معالجًا للرسائل التي يبدأ مسارها بالبادئة المحددة في المعلمة $path. سيتم استدعاء مرجع الكود المقدم مع معلمتين، كائن الاتصال الذي تم استلام الرسالة عليه، والرسالة المراد معالجتها (مثيل لفئة "Net::DBus::Binding::Message").
$con->unregister_object_path($path)
يلغي تسجيل المعالج المرتبط بمسار الكائن $path. كان سيتم تسجيل المعالج سابقًا باستخدام الطريقتين "register_object_path" أو "register_fallback".
my $msg = $con->make_error_message($replyto, $name, $description)
ينشئ رسالة جديدة، تمثل خطأ حدث أثناء معالجة كائن استدعاء الطريقة المقدم كالمعلمة $replyto. المعلمة $name هي الاسم الرسمي لحالة الخطأ، بينما $description هي قطعة نصية قصيرة تقدم معلومات أكثر تحديدًا حول الخطأ.
my $call = $con->make_method_call_message( $service_name, $object_path, $interface, $method_name);
ينشئ رسالة تمثل استدعاء على الكائن الموجود في المسار $object_path ضمن العميل الذي يمتلك الاسم المعروف المعطى بواسطة $service_name. الطريقة المراد استدعاؤها لها الاسم $method_name ضمن الواجهة المحددة بالمعلمة $interface.
my $msg = $con->make_method_return_message($replyto)
ينشئ رسالة تمثل ردًا على رسالة استدعاء الطريقة المقدمة في المعلمة $replyto.
my $msg = $con->make_signal_message($object_path, $interface, $signal_name);
ينشئ رسالة جديدة، تمثل إشارة [سيتم] بثها بواسطة الكائن الموجود تحت المسار المعطى بالمعلمة $object_path. اسم الإشارة معطى بالمعلمة $signal_name، ويتم تحديده في الواجهة المعطاة بالمعلمة $interface.

الأخطاء

لا تقوم بإعادة إنشاء API الخاصة بـ Net::DBus::Binding::Connection بالكامل، بل تكفي لجعل الروابط عالية المستوى تعمل في سيناريو اختبار.
