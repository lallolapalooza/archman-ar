Net::DBus::Tutorial::ExportingObjects(3) وثائق مساهمات المستخدم في Perl Net::DBus::Tutorial::ExportingObjects(3)

اسم

Net::DBus::Tutorial::ExportingObjects - دروس حول تقديم خدمة DBus

الوصف

يقدم هذا المستند درسًا حول تقديم خدمة DBus باستخدام روابط تطبيق Perl Net::DBus. ستستند الأمثلة في هذا المستند إلى الكود الموجود في توزيع Music::Player، وهي خدمة DBus بسيطة توفر مشغل مسارات موسيقية.

إنشاء كائن

الخطوة الأولى في إنشاء كائن هي إنشاء حزمة جديدة ترث من Net::DBus::Object. يوفر كائن Music::Player::Manager واجهة برمجة تطبيقات لإدارة مجموعة من مشغلات الموسيقى الخلفية لأنواع المسارات المختلفة. للبدء، لننشئ هيكل الحزمة & وبانيها. يتوقع باني النوع الفائق، Net::DBus::Object، أن يتم تمرير معلمتين له: مقبض لخدمة Net::DBus التي تمتلك الكائن، ومسار يجب تصدير الكائن تحته. نظرًا لأن فئة المدير مخصصة لتكون كائنًا فريدًا، يمكننا ترميز المسار الخاص بها بشكل ثابت داخل الباني:

package Music::Player::Manager;
use base qw(Net::DBus::Object);
sub new {
    my $class = shift;
    my $service = shift;
    my $self = $class->SUPER::new($service, "/music/player/manager");
    bless $self, $class;
    return $self;
}
1;

الآن، كما ذكرنا، سيتعامل المدير مع عدد من مشغلات الخلفية المختلفة. لذلك نحتاج إلى توفير طرق لتسجيل مشغلات خلفية جديدة، والاستعلام عن مشغلات خلفية قادرة على تشغيل نوع ملف معين. لذا، بتعديل الكود أعلاه، نضيف جدول تجزئة في الباني، لتخزين المشغلات الخلفية:

sub new {
    my $class = shift;
    my $service = shift;
    my $self = $class->SUPER::new($service, "/music/player/manager");
    $self->{backends} = {};
    bless $self, $class;
    return $self;
}

والآن طريقة لتسجيل مشغل خلفي جديد. تأخذ هذه اسم وحدة Perl وتستخدمه لإنشاء مشغل خلفي. نظرًا لأن المشغلات الخلفية ستكون أيضًا كائنات DBus، نحتاج إلى تمرير مرجع للخدمة التي نرفق بها، جنبًا إلى جنب مع مسار لتسجيل المشغل الخلفي تحته. نستخدم طريقة "get_service" لاسترداد مرجع للخدمة التي يرتبط بها المدير، ونرفق مشغل الموسيقى الخلفي بنفس الخدمة: عندما يتم استدعاء طريقة على كائن DBus، فإن المعلمة الأولى هي مرجع الكائن ($self)، والباقي هي المعلمات المقدمة لاستدعاء الطريقة. وبالتالي، فإن كتابة تنفيذ طريقة على DBus لا تختلف حقًا عن Perl الموجه للكائنات العادية (راجع perltoot):

sub register_backend {
    my $self = shift;
    my $name = shift;
    my $module = shift;
    eval "use $module";
    if ($@) {
        die "cannot load backend $module: $@" ;
    }
    $self->{backends}->{$name} = $module->new($self->get_service,
                                              "/music/player/backend/$name");
}

بالنظر إلى هذا، قد يتساءل المرء عما يحدث إذا تم تشغيل طريقة "die". في مثل هذا السيناريو، بدلاً من إنهاء عملية الخدمة، سيتم التقاط الخطأ وسيتم تمريره إلى المتصل البعيد للتعامل معه.

توفر مشغلات الموسيقى الخلفية طريقة "get_track_types" التي تُرجع مرجع مصفوفة لأنواع المسارات الموسيقية التي تدعمها. يمكننا استخدام هذه الطريقة لتوفير واجهة برمجة تطبيقات للسماح بالاسترداد السهل لمشغل خلفي لنوع مسار معين. سترجع هذه الطريقة مسارًا يمكن الوصول إليه من خلاله إلى كائن المشغل الخلفي

sub find_backend {
    my $self = shift;
    my $extension = shift;
    foreach my $name (keys %{$self->{backends}}) {
       my $backend = $self->{backends}->{$name};
       foreach my $type (@{$backend->get_track_types}) {
          if ($type eq $extension) {
              return $backend->get_object_path;
          }
       }
    }
    die "no backend for type $extension";
}

دعنا نأخذ لحظة سريعة للتفكير في كيفية استخدام هذه الطريقة لتشغيل مسار موسيقي. إذا لم تكن قد فعلت ذلك بالفعل، قم بتحديث ذاكرتك من Net::DBus::Tutorial::UsingObjects. الآن، لدينا ملف MP3 نريد تشغيله، لذلك نبحث عن المسار إلى مشغل خلفي، ثم نسترد الكائن الخاص به، ونشغل المسار:

...get the music player service...
# Ask for a path to a player for mp3 files
my $path = $service->find_backend("mp3");
# $path now contains '/music/player/backend/mpg123'
# and we can get the backend object
my $backend = $service->get_object($path);
# and finally play the track
$backend->play("/vol/music/beck/guero/09-scarecrow.mp3");

توفير بيانات الفحص

الكود أعلاه هو كائن كامل يعمل، جاهز للتسجيل مع خدمة، وبما أن المعلمات والقيم المعادة للطريقتين هما مجرد سلاسل بسيطة، يمكننا التوقف عند هذا الحد. في بعض الحالات، قد يرغب المرء في أن يكون أكثر تحديدًا بشأن أنواع البيانات المتوقعة للمعلمات، على سبيل المثال أعداد صحيحة موقعة مقابل غير موقعة. إضافة تصنيف بيانات صريح يجعل التفاعل مع لغات البرمجة الأخرى أكثر موثوقية. يُعرف توفير تعريفات أنواع بيانات صريحة للطرق المصدرة في عالم DBus باسم "الفحص"، ويجعل الأمور أكثر موثوقية لمستخدمي خدمتك الذين قد يستخدمون لغة قوية الكتابة مثل C.

الخطوة الأولى في توفير بيانات الفحص لكائن DBus في Perl هي تحديد اسم الواجهة التي يوفرها الكائن. هذا عادة ما يكون سلسلة مفصولة بنقاط، وبموجب الاصطلاح يحتوي على اسم المجال للتطبيق كمكون أول. نظرًا لأن معظم وحدات Perl تنتهي بالعيش على CPAN، قد يستخدم المرء "org.cpan" كمكون أول، متبوعًا باسم الحزمة للوحدة (مع استبدال :: بـ .)، على سبيل المثال "org.cpan.music.player.manager". إذا لم يكن هناك تخطيط لاستضافة الوحدة على CPAN، يمكن استخدام مجال شخصي/مشروع، على سبيل المثال "com.berrange.music.player.manager". يتم تعريف الواجهة لكائن عن طريق تحميل الوحدة Net::DBus::Exporter، وتوفير الواجهة كمعلمتها الأولى. لذا، سيتم تعديل مثال الكود السابق ليبدو كالتالي:

package Music::Player::Manager;
use base qw(Net::DBus);
use Net::DBus::Exporter qw(com.berrange.music.player.manager)

بعد ذلك، من الضروري توفير أنواع البيانات للمعلمات والقيم المعادة للطرق. توفر الوحدة Net::DBus::Exporter طريقة "dbus_method" لهذا الغرض، والتي تأخذ ثلاث معلمات: اسم الطريقة التي يتم تصديرها، مرجع مصفوفة لأنواع المعلمات، ومرجع مصفوفة لأنواع القيم المعادة (يمكن حذف الأخيرة إذا لم تكن هناك قيم معادة). يمكن استدعاء هذا في أي نقطة في كود الوحدة، ولكن بموجب الاصطلاح، يفضل ربط استدعاءات "dbus_method" بتنفيذ الطريقة الفعلية، وبالتالي:

dbus_method("register_backend", ["string", "string"]);
sub register_backend {
    my $self = shift;
    my $name = shift;
    my $module = shift;
    .. snipped rest of method body ...
}

وبالتالي:

dbus_method("find_backend", ["string"], ["string"])
sub find_backend {
    my $self = shift;
    my $extension = shift;
    ... snip method body...
}

تعريف خدمة

الآن بعد أن تمت كتابة الكائنات، حان الوقت لتعريف خدمة. الخدمة ليست أكثر من اسم معروف جيدًا لعقد واجهة برمجة تطبيقات معين. يمكن التفكير في العقد كتحديد لقائمة من مسارات الكائنات، والواجهات المقابلة التي توفرها. لذلك، يمكن لشخص آخر أن يأتي ويقدم تطبيق مشغل موسيقى بديلاً باستخدام روابط Python أو QT لـ DBus، وإذا قدم نفس مجموعة مسارات الكائنات & والواجهات، فيمكنه بحق تسجيل نفس الخدمة على الناقل.

توفر وحدة Net::DBus::Service الوسيلة لتسجيل خدمة. يتوقع بانيها مرجعًا لكائن الناقل (مثال من Net::DBus)، جنبًا إلى جنب مع اسم الخدمة. كما هو الحال مع أسماء الواجهات، عادة ما يتم اشتقاق المكون الأول لاسم الخدمة من اسم مجال، ثم يتم إضافة اسم التطبيق إليه، في مثالنا يشكل "org.cpan.Music.Player". بينما سيتم إنشاء بعض الكائنات أثناء التنفيذ أثناء تشغيل التطبيق، سيتم إنشاء البعض الآخر عند بدء التشغيل الأولي. كائن مدير مشغل الموسيقى

هذا الموقع غير مرتبط بـ Arch Linux ولا تابع له بأي شكل من الأشكال. المحتوى هو ترجمة فقط لصفحات دليل Arch المنشورة على man.archlinux.org.

إنشاء كائن

تم إنشاء مثال سابق في هذا البرنامج التعليمي للمثال الأخير. من المعتاد إنشاء وتسجيل هذه الكائنات في المُنشئ للخدمة. وبالتالي، سيبدو كائن الخدمة لتطبيق مشغل الموسيقى كالتالي:

package Music::Player;
use base qw(Net::DBus::Service);
sub new {
    my $class = shift;
    my $bus = shift;
    my $self = $class->SUPER::new($bus, "org.cpan.music.player");
    bless $self, $class;
    $self->{manager} = Music::Player::Manager->new($self);
    return $self;
}

يوفر Net::DBus::Service تلقائيًا كائنًا خاصًا واحدًا لجميع الخدمات، تحت المسار "/org/freedesktop/DBus/Exporter". يطبق هذا الكائن الواجهة "org.freedesktop.DBus.Exporter" والتي تحتوي على طريقة "ListObject". يتيح ذلك للعملاء تحديد قائمة بجميع الكائنات المصدرة داخل خدمة. في حين أنه ليس ضروريًا وظيفيًا لمعظم التطبيقات، إلا أنه أداة مفيدة للمطورين الذين يقومون بتصحيح الأخطاء في التطبيقات، أو يتساءلون عما توفره الخدمة.

الاتصال بالحافلة

الخطوة النهائية في تشغيل خدمتنا هي توصيلها بالحافلة. هذا يثير معضلة مثيرة للاهتمام، هل يتم تصدير الخدمة على حافلة النظام (مشتركة بين جميع المستخدمين والعمليات على الجهاز)، أم حافلة الجلسة (واحدة لكل مستخدم مسجل الدخول إلى الجهاز). في بعض الحالات، الإجابة، مع وجود واحدة فقط من الحافلتين المفهومتين منطقية. في حالات أخرى، ومع ذلك، فإن كل من حافلة الجلسة وحافلة النظام صالحتان. في الحالة الأولى، سيتم استخدام طرق "session" أو <system> على Net::DBus للحصول على مقبض للحافلة المطلوبة، بينما في الحالة الأخيرة، سيتم استخدام طريقة "find". يطبق هذا نهجًا استدلاليًا لتحديد الحافلة الصحيحة بناءً على بيئة التنفيذ. في حالة مشغل الموسيقى، تكون أي من الحافلتين ذات صلة، لذا سيبدو الكود لتوصيل الخدمة بالحافلة كالتالي:

use Net::DBus;
my $bus = Net::DBus->find;
my $player = Music::Player->new($bus);

مع توصيل الخدمة بالحافلة، يكفي تشغيل حلقة معالجة الأحداث الرئيسية للاستماع إلى رسائل DBus الواردة والتعامل معها. لذلك، يتم تعديل الكود أعلاه لبدء مفاعل بسيط:

use Net::DBus;
use Net::DBus::Reactor;
my $bus = Net::DBus->find;
my $player = Music::Player->new($bus);
Net::DBus::Reactor->main->run;
exit 0;

حفظ هذا الكود في برنامج نصي "/usr/bin/music-player.pl"، ويكتمل الترميز وتصبح الخدمة جاهزة للاستخدام من قبل العملاء على الحافلة.

تنشيط الخدمة

قد يتساءل المرء الآن عن أفضل طريقة لبدء الخدمة، خاصة إذا كانت خدمة قادرة على العمل على كل من حافلات النظام والجلسة. لدى DBus الإجابة في مفهوم "التنشيط". ما يحدث هو أنه عندما يحاول عميل على الحافلة استدعاء طريقة، أو تسجيل معالج إشارات ضد خدمة غير قيد التشغيل حاليًا، فإنه سيحاول أولاً بدء الخدمة. تحتاج الخدمات التي ترغب في المشاركة في هذه العملية ببساطة إلى وضع ملف تعريف خدمة بسيط في الدليل "/usr/share/dbus-1/services". يجب تسمية الملف لمطابقة اسم الخدمة، مع إضافة امتداد الملف ".service". مثال، "/usr/share/dbus-1/services/org.cpan.music.player.service". يحتوي الملف على مفتاحين، أولاً اسم الخدمة، وثانياً اسم الملف التنفيذي المستخدم لتشغيل الخدمة، أو في هذه الحالة البرنامج النصي Perl. لذلك، لخدمتنا البسيطة، سيحتوي ملف البيانات على:

[D-BUS Service]
Name=org.cpan.music.player
Exec=/usr/bin/music-player.pl
