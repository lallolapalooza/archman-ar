Net__DNS__SEC__Tools__conf(3pm)

الحزمة: dnssec-tools

التراخيص: BSD

conf(3) وثائق Perl المساهمة من المستخدم conf(3)

الاسم

Net::DNS::SEC::Tools::conf - إجراءات تكوين DNSSEC-Tools.

الملخص

use Net::DNS::SEC::Tools::conf;
%dtconf = parseconfig();
%dtconf = parseconfig("localzone.keyrec");
cmdcheck(\%options_hashref);
$conferrs = dt_confcheck();
$prefixdir = getprefixdir();
$confdir = getconfdir();
$conffile = getconffile();
setconffile("dt-local.conf");
$statedir = getlocalstatedir();
$statedir = makelocalstatedir();
$statesub = makelocalstatedir("logs/zones");
$packed = runpacked();
erraction(ERR_MSG);
err("لا يمكن فتح ملف keyrec",1);

الوصف

تقوم الإجراءات في هذه الوحدة بإجراء عمليات التكوين. تصل بعض الإجراءات إلى ملف تكوين DNSSEC-Tools، بينما تتحقق إجراءات أخرى من بيئة التنفيذ.

تمتلك أدوات DNSSEC ملف تكوين للقيم شائعة الاستخدام. هذه القيم هي الافتراضيات لمجموعة متنوعة من الأشياء، مثل خوارزمية التشفير وطول مفتاح التشفير. توفر الوحدة Net::DNS::SEC::Tools::conf طرقًا للوصول إلى بيانات التكوين في هذا الملف.

dnssec-tools.conf هو اسم ملف تكوين أدوات DNSSEC. يعتمد المسار الكامل على كيفية تكوين DNSSEC-Tools؛ انظر قسم الدلائل للمسار الكامل. يتم تعيين المسارات التي تتطلبها conf.pm وقت تكوين DNSSEC-Tools.

يتكون ملف تكوين أدوات DNSSEC من مجموعة من إدخالات قيم التكوين، مع إدخال واحد فقط في كل سطر. كل إدخال له تنسيق "كلمة مفتاحية قيمة". أثناء التحليل، يتم تقسيم السطر إلى رموز، مع فصل الرموز بالمسافات وعلامات الجدولة. يُؤخذ الرمز الأول في السطر كالكلمة المفتاحية. يتم دمج جميع الرموز الأخرى في هذا السطر في سلسلة واحدة، مع مسافة تفصل كل رمز. تتم إضافة السلسلة غير الرمزية إلى جدول هاش، مع الكلمة المفتاحية كمفتاح للقيمة.

يمكن تضمين التعليقات عن طريق وضع علامة '#' أو ';' قبلها. يمكن أن تشمل هذه التعليقات سطرًا كاملاً أو قد تتبع إدخال تكوين. إذا شارك تعليق سطرًا مع إدخال، يتوقف ترميز القيمة قبل حرف التعليق مباشرة.

مثال على ملف تكوين:

# إدخالات تكوين نموذجية.
algorithm       rsasha1     # خوارزمية التشفير.
ksk_length      2048        ; طول مفتاح KSK.

جانب آخر من تكوين DNSSEC-Tools هو إجراء الخطأ المستخدم في وحدات Perl الخاصة بـ DNSSEC-Tools. يحدد الإجراء ما إذا كانت حالة الخطأ ستعطي فقط قيمة إرجاع للخطأ، أو تطبع رسالة خطأ إلى STDERR، أو تطبع رسالة خطأ وتخرج. تُستخدم الواجهات erraction() و err() لهذه العمليات.

الواجهات

dt_confcheck(errflag)
تقوم هذه العملية بإجراء عدد من فحوصات التكوين لضمان أن البيئة كافية لدعم تكوين DNSSEC-Tools. إذا كانت errflag تساوي 0، فسيتم إجراء الفحوصات بهدوء؛ بخلاف ذلك، سيتم طباعة رسائل الخطأ.

الفحوصات هي:

* وجود دليل sysconf الخاص بـ dnssec-tools.
* دليل sysconf الخاص بـ dnssec-tools هو دليل.
* وجود دليل dnssec-tools.
* دليل dnssec-tools هو دليل.
* وجود ملف تكوين dnssec-tools.
* ملف تكوين dnssec-tools هو ملف عادي.
* ملف تكوين dnssec-tools ليس فارغًا.
* اسم دليل الحالة المحلي ليس أطول من 75
  حرفًا (للسماح بمقبس الأوامر rollmgr.)
* دليل الحالة المحلي هو دليل.
* يمكن إنشاء دليل الحالة المحلي إذا لزم الأمر.
* يمكن إنشاء الدليل الفرعي dnssec-tools الخاص بدليل الحالة المحلي
  إذا لزم الأمر، أو يمكن الكتابة فيه إذا كان موجودًا بالفعل.
* يمكن إنشاء دليل التشغيل الخاص بدليل الحالة المحلي
  إذا لزم الأمر، أو يمكن الكتابة فيه إذا كان موجودًا بالفعل.

قيم الإرجاع:

0       لم يتم العثور على أخطاء
>0      فشلت بعض فحوصات التكوين
parseconfig()
تقوم هذه العملية بقراءة وتحليل ملف تكوين أدوات DNSSEC للنظام. يتم وضع المحتويات المحللة في جدول هاش، والذي يتم إرجاعه إلى المتصل.
parseconfig(conffile)
تقوم هذه العملية بقراءة وتحليل ملف تكوين أدوات DNSSEC محدد من قبل المتصل. يتم وضع المحتويات المحللة في جدول هاش، والذي يتم إرجاعه إلى المتصل. تعود العملية بصمت إذا لم يكن ملف التكوين موجودًا.
cmdcheck(\%options_hashref)
تضمن هذه العملية توفر الأوامر اللازمة وقابليتها للتنفيذ. إذا كانت أي من الأوامر غير موجودة أو غير قابلة للتنفيذ، فسيتم إصدار رسالة خطأ وسيتم إنهاء العملية. إذا كان كل شيء على ما يرام، فسيستمر كل شيء بهدوء.

مفاتيح الأوامر التي يتم فحصها حاليًا هي zonecheck و keygen و zonesign. يتم العثور على مسارات هذه الأوامر في خيارات التجزئة المحددة المشار إليها بواسطة %options_hashref. إذا لم يكن التجزئة يحتوي على إدخال لأحد هذه الأوامر، فلن يتم فحصه.

إذا تم استدعاء هذه العملية من نص برمجي مجمع بواسطة PAR، فستبحث في دليل الحزمة عن الأوامر. كما ستضبط أوضاع ملفاتها على 0755، حيث يبدو أن PAR يتجاهل أوضاع الملفات عند تجميع البرامج.

getconfdir()
تقوم هذه العملية بإرجاع اسم دليل تكوين DNSSEC-Tools.
getconffile()
تقوم هذه العملية بإرجاع اسم ملف تكوين DNSSEC-Tools.
setconffile()
تقوم هذه العملية بتعيين اسم ملف تكوين DNSSEC-Tools.

قيم الإرجاع:
1 تم الإرجاع عند النجاح
0 تم الإرجاع إذا كان ملف التكوين المحدد لا
يوجد أو ليس ملفًا عاديًا

getprefixdir()
تقوم هذه العملية بإرجاع اسم دليل بادئة DNSSEC-Tools.
getlocalstatedir()
تقوم هذه العملية بإرجاع اسم دليل الحالة المحلي.
runpacked()
تقوم هذه العملية بإرجاع قيمة منطقية تشير إلى ما إذا كان الأمر التنفيذي يعمل من نص برمجي مجمع بواسطة PAR.
makelocalstatedir(subdir)
تقوم هذه العملية بإنشاء دليل الحالة المحلي وإرجاع اسمه. يتم إنشاء الدليل فقط إذا لم يكن موجودًا بالفعل.

إذا تم تحديد الدليل الفرعي الاختياري subdir، فسيتم إنشاء هذا الدليل داخل دليل الحالة المحلي. في هذه الحالة، يتم إرجاع مسار subdir. قد يتكون subdir من عدة أدلة وسيطة، بالإضافة إلى الدليل النهائي. على سبيل المثال، makelocalstatedir("logs/zones/errors") سينشئ التسلسل الهرمي logs/zones/errors داخل دليل الحالة المحلي.

تستخدم makelocalstatedir(subdir) الوحدة File::Path، وهي متاحة في جميع إصدارات Perl الحديثة.

يتم إرجاع سلسلة فارغة في حالة حدوث أي أخطاء. قد تتم مواجهة الأخطاء التالية:

* فشل تحميل I<File::Path>
* عدم القدرة على إنشاء دليل الحالة المحلي
* عدم القدرة على إنشاء مكون من I<subdir>
* المسار الكامل (دليل الحالة المحلي و I<subdir>) موجود بالفعل
  وليس دليلًا
boolconvert(config-value)
تقوم هذه العملية بتحويل قيم التكوين إلى قيم منطقية مناسبة. التحويلات النصية التالية تتم: 
1 - 'true', 't', 'yes', 'y'
0 - 'false', 'f', 'no', 'n'

جميع القيم النصية الأخرى يتم تحويلها إلى 0.

يتم تحويل القيم الموجبة إلى 1. يتم تحويل القيم السالبة إلى 0.

erraction(error_action)
تقوم هذه الواجهة بتعيين إجراء الخطأ لوحدات Perl الخاصة بـ DNSSEC-Tools. الإجراءات الصالحة هي
e: 
ERR_SILENT          لا تطبع رسالة خطأ، لا تخرج.
ERR_MSG             اطبع رسالة خطأ، لا تخرج.
ERR_EXIT            اطبع رسالة خطأ، اخرج.

ERR_SILENT هو الإجراء الافتراضي.

يتم إرجاع إجراء الخطأ الذي تم تعيينه مسبقًا.

err("رسالة الخطأ",رمز_الخروج)
يتم استخدام الواجهة err() بواسطة وحدات DNSSEC-Tools Perl للإبلاغ عن خطأ والخروج، اعتمادًا على إجراء الخطأ.

الوسيط الأول هو رسالة خطأ للطباعة -- إذا سمح إجراء الخطأ بطباعة رسائل الخطأ.

الوسيط الثاني هو رمز خروج -- إذا تطلب إجراء الخطأ أن تخرج العملية.

المجلدات

المجلدات الافتراضية لهذا التثبيت هي:

prefix                         : /usr
sysconf                        : /etc
localstatedir                  : /usr/var
DNSSEC-Tools configuration file: /etc/dnssec-tools

يمكن تجاوز هذه المتغيرات باستخدام متغيرات البيئة التالية:

prefix                         : DT_PREFIX
sysconf                        : DT_SYSCONFDIR
localstatedir                  : DT_STATEDIR
