Net__OAuth__Client(3pm)

الحزمة: perl-net-oauth

التراخيص: PerlArtistic, GPL

Net::OAuth::Client(3) وثائق Perl مقدمة من المستخدم Net::OAuth::Client(3)

الاسم

Net::OAuth::Client - عميل OAuth 1.0A

الملخص

# مثال خادم الويب (Dancer)
# هذا المثال مبسط لأغراض توضيحية، انظر الكود الكامل في /demo
# لاحظ أن client_id هو مفتاح المستهلك و client_secret هو سر المستهلك
use Dancer;
use Net::OAuth::Client;
sub client {
      Net::OAuth::Client->new(
              config->{client_id},
              config->{client_secret},
              site => 'https://www.google.com/',
              request_token_path => '/accounts/OAuthGetRequestToken?scope=https%3A%2F%2Fwww.google.com%2Fm8%2Ffeeds%2F',
              authorize_path => '/accounts/OAuthAuthorizeToken',
              access_token_path => '/accounts/OAuthGetAccessToken',
              callback => uri_for("/auth/google/callback"),
              session => \&session,
      );
}
# إرسال المستخدم للموافقة مع مزود الخدمة
get '/auth/google' => sub {
      redirect client->authorize_url;
};
# عاد المستخدم مع الرمز والمُحقِّق الملحقين بالرابط.
get '/auth/google/callback' => sub {
      # استخدام رمز المصادقة لجلب رمز الوصول
      my $access_token =  client->get_access_token(params->{oauth_token}, params->{oauth_verifier});
      # استخدام رمز الوصول لجلب مورد محمي
      my $response = $access_token->get('/m8/feeds/contacts/default/full');
      # فعل شيئًا ما بهذه الموارد...
      if ($response->is_success) {
        return "نعم، لقد نجح الأمر: " . $response->decoded_content;
      }
      else {
        return "خطأ: " . $response->status_line;
      }
};
dance;

الوصف

يمثل Net::OAuth::Client عميل OAuth أو مستهلك.

تحذير: Net::OAuth::Client هو كود تجريبي (alpha). بقية Net::OAuth مستقرة تمامًا ولكن هذه الوحدة بالذات جديدة، وهي غير موثقة بشكل كافٍ وغير مختبرة بشكل كافٍ.

الأساليب

new($client_id, $client_secret, %params)
إنشاء عميل جديد
  • $client_id

    المعروف أيضًا باسم مفتاح المستهلك - تحصل عليه من مزود الخدمة عند تسجيل تطبيقك.

  • $client_secret

    المعروف أيضًا باسم سر المستهلك - تحصل عليه من مزود الخدمة عند تسجيل تطبيقك.

  • $params{site}
  • $params{request_token_path}
  • $params{authorize_path}
  • $params{access_token_path}
  • $params{callback}
  • $params{session}
