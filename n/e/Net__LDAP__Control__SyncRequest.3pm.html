Net__LDAP__Control__SyncRequest(3pm)

الحزمة: perl-ldap

التراخيص: GPL

Net::LDAP::Control::SyncRequest(3) وثائق Perl المساهمة من المستخدم Net::LDAP::Control::SyncRequest(3)

الاسم

Net::LDAP::Control::SyncRequest - كائن طلب مزامنة LDAPv3

ملخص

use Net::LDAP;
use Net::LDAP::Control::SyncRequest;
use Net::LDAP::Constant qw(
 LDAP_SYNC_REFRESH_ONLY
 LDAP_SYNC_REFRESH_AND_PERSIST
 LDAP_SUCCESS );
$ldap = Net::LDAP->new( "ldap.mydomain.eg" );
$req = Net::LDAP::Control::SyncRequest->new( mode => LDAP_SYNC_REFRESH_ONLY );
my $mesg = $ldap->search(base=> 'dc=mydomain,dc='eg',
                         scope    => 'sub',
                         control  => [ $req ],
                         callback => \&searchCallback, # استدعاء لكل إدخال
                         filter   => "(objectClass=*)",
                         attrs    => [ '*']);
sub searchCallback {
  my $message = shift;
  my $entry = shift;
  my @controls = $message->control;
  if ($controls[0]->isa('Net::LDAP::Control::SyncState')) {
    print "تم استلام التحكم في حالة المزامنة\n";
    print $entry->dn()."\n";
    print 'الحالة: '.$controls[0]->state."\n".', معرف_الإدخال: '.$controls[0]->entryUUID.', ملف_التعريف: '.$controls[0]->cookie;
  } elsif ($controls[0]->isa('Net::LDAP::Control::SyncDone')) {
    print "تم استلام التحكم في انتهاء المزامنة\n";
    print 'ملف_التعريف: '.$controls[0]->cookie.', حذف_التحديث: '.$controls[0]->refreshDeletes;
  }
}

الوصف

"Net::LDAP::Control::SyncRequest" يوفر واجهة لإنشاء ومعالجة الكائنات التي تمثل "التحكم في طلب المزامنة" كما هو موضح بواسطة RFC 4533.

وسيطات المنشئ

بالإضافة إلى وسيطات المنشئ الموضحة في Net::LDAP::Control، يتم توفير ما يلي.

الطرق

كما هو الحال مع Net::LDAP::Control، فإن كل وسيطة منشئ موصوفة أعلاه متاحة أيضًا كطريقة على الكائن والتي ستعيد القيمة الحالية للسمة إذا تم استدعاؤها بدون وسيطة، وتقوم بتعيين قيمة جديدة للسمة إذا تم استدعاؤها بوسيطة.
