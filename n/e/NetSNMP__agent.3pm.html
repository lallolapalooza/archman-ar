NetSNMP__agent(3pm)

الحزمة: net-snmp

التراخيص: BSD

agent(3) مساهمات مستخدم وثائق Perl agent(3)

الاسم

NetSNMP::agent - امتداد Perl لوكيل net-snmp.

الملخص

use NetSNMP::agent;
my $agent = new NetSNMP::agent('Name' => 'my_agent_name');

الوصف

تنفذ هذه الوحدة مجموعة من واجهات برمجة التطبيقات (API) لجعل وكيل SNMP يعمل كوكيل snmp، وكيل فرعي snmp (باستخدام بروتوكول الوكيل الفرعي AgentX) و/أو واجهات perl مضمنة مباشرة داخل خدمة وكيل net-snmp التقليدية.

انظر أيضًا البرنامج التعليمي حول واجهة برمجة تطبيقات C العامة لـ Net-SNMP، والتي تنفذها هذه الوحدة بطريقة Perl، وبرنامج تعليمي خاص بـ Perl على:

http://www.net-snmp.org/tutorial-5/toolkit/

أمثلة

مثال على وكيل فرعي

use NetSNMP::agent (':all');
use NetSNMP::ASN qw(ASN_OCTET_STR);
my $value = "hello world";
sub myhandler {
    my ($handler, $registration_info, $request_info, $requests) = @_;
    my $request;
    for($request = $requests; $request; $request = $request->next()) {
        my $oid = $request->getOID();
        if ($request_info->getMode() == MODE_GET) {
            # ... بشكل عام، ستحسب القيمة من oid
            if ($oid == new NetSNMP::OID(".1.3.6.1.4.1.8072.9999.9999.7375.1.0")) {
                $request->setValue(ASN_OCTET_STR, $value);
            }
        } elsif ($request_info->getMode() == MODE_GETNEXT) {
            # ... بشكل عام، ستحسب القيمة من oid
            if ($oid < new NetSNMP::OID(".1.3.6.1.4.1.8072.9999.9999.7375.1.0")) {
                $request->setOID(".1.3.6.1.4.1.8072.9999.9999.7375.1.0");
                $request->setValue(ASN_OCTET_STR, $value);
            }
        } elsif ($request_info->getMode() == MODE_SET_RESERVE1) {
            if ($oid != new NetSNMP::OID(".1.3.6.1.4.1.8072.9999.9999.7375.1.0")) {  # قم بإجراء التحقق من الأخطاء هنا
                $request->setError($request_info, SNMP_ERR_NOSUCHNAME);
            }
        } elsif ($request_info->getMode() == MODE_SET_ACTION) {
            # ... (أو استخدم القيمة)
            $value = $request->getValue();
        }
    }
}
my $agent = new NetSNMP::agent(
                        # يجعل الوكيل يقرأ ملف my_agent_name.conf
                        'Name' => "my_agent_name",
                        'AgentX' => 1
                        );
$agent->register("my_agent_name", ".1.3.6.1.4.1.8072.9999.9999.7375",
                 \&myhandler);
my $running = 1;
while($running) {
        $agent->agent_check_and_process(1);
}
$agent->shutdown();

مثال على وكيل مضمن

# ضع هذا في ملف .pl، ثم في ملف snmpd.conf الخاص بك ضع:
#    perl do '/path/to/file.pl';
use NetSNMP::agent;
my $agent;
sub myhandler {
    my ($handler, $registration_info, $request_info, $requests) = @_;
    # ...
}
$agent = new NetSNMP::agent(
                        'Name' => 'my_agent_name'
                        );
$agent->register("my_agent_name", ".1.3.6.1.4.1.8072.9999.9999.7375",
                 \&myhandler);
$agent->main_loop();

المنشئ

new ( OPTIONS )
    هذا هو منشئ كائن NetSNMP::agent جديد.
    الخيارات الممكنة هي:
    Name    - اسم الوكيل (اختياري، الافتراضي هو "perl")
              (سيقرأ مكتب SNMP ملف تكوين snmp باسم NAME.conf
              بناءً على هذه الوسيطة.)
    AgentX  - اجعلنا وكيلاً فرعيًا (0 = خطأ، 1 = صحيح)
              (يجب أن يكون وكيل Master AgentX من NetSNMP قيد التشغيل أولاً)
    Ports   - المنافذ التي سيستمع عليها هذا الوكيل (مثال: "udp:161,tcp:161")
    مثال:
    $agent = new NetSNMP::agent(
                             'Name' => 'my_agent_name',
                             'AgentX' => 1
                             );

الأساليب

register (NAME, OID, \&handler_routine )
    يسجل معالج الاستدعاء مع OID معين.
    $agent->register();
    رمز الإرجاع 0 يشير إلى عدم وجود خطأ.
agent_check_and_process ( BLOCKING )
    قم بتشغيل تكرار واحد للحلقة الرئيسية.
    BLOCKING - استدعاء حظر أو عدم حظر. 1 = صحيح، 0 = خطأ.
    $agent->agent_check_and_process(1);
main_loop ()
    يدير الوكيل في حلقة. لا يعود.
shutdown ()
    يقوم بإغلاق الوكيل أو الوكيل الفرعي بشكل أنيق.
    $agent->shutdown();

استدعاءات المعالج

handler ( HANDLER, REGISTRATION_INFO, REQUEST_INFO, REQUESTS )
    يتم استدعاء المعالج بالمعلمات التالية:
    HANDLER                 - تم الإصلاح
    REGISTRATION_INFO       - ما هي المعاني الصحيحة لهذه؟
    REQUEST_INFO            -
    REQUESTS                -
    مثال للمعالج:
    sub myhandler {
        my ($handler, $reg_info, $request_info, $requests) = @_;
        # ...
    }

سيتم استدعاء الإجراء الفرعي للمعالج عند استلام طلب SNMP بواسطة الوكيل لأي شيء تحت OID المسجل. يتم تمرير 4 وسيطات إلى المعالج: $handler، $registration_info، $request_info، $requests. تتوافق هذه مع الوسيطات التي تم تمريرها إلى إصدار C من نفس واجهة برمجة التطبيقات. لاحظ أنها ليست كائنات كاملة تمامًا ولكنها "كافية" وظيفيًا في هذا الوقت.

وظائف كائن $request_info

getMode ()
    يعيد وضع الطلب. انظر قسم الأوضاع (MODES) للحصول على
    قائمة الأوضاع الصالحة.
    $mode = $request->getMode();

وظائف كائن $registration_info

getRootOID ()
    يعيد كائن NetSNMP::OID يصف نقطة التسجيل
    التي يتم استدعاء المعالج لها (في حال
    قمت بتسجيل وظيفة معالج واحدة لـ OIDs متعددة، وهو ما ينبغي أن يكون
    نادرًا على أي حال)
    $root_oid = $request->getRootOID();

وظائف كائن $request

next ()
    يعيد الطلب التالي في القائمة أو undef إذا لم يكن هناك
    طلب تالي.
    $request = $request->next();
getOID ()
    يعيد OID الطلب (فئة NetSNMP::OID).
    $oid = $request->getOID();
setOID (new NetSNMP::OID("someoid"))
    يضبط OID الطلب إلى قيمة oid تم تمريرها. يجب
    عادةً ما يتم القيام بذلك فقط أثناء معالجة طلبات GETNEXT.
    $request->setOID(new NetSNMP::OID("someoid"));
getValue ()
    يعيد قيمة الطلب. تستخدم على سبيل المثال عند
    تعيين القيم.
    $value = $request->getValue();
    تم الإصلاح: كيف تحصل على نوع القيمة؟ هل هو متاح حتى؟
           [Wes: لا، ليس بعد.]
setValue ( TYPE, DATA )
    يضبط البيانات التي سيتم إرجاعها إلى الوكيل.
    يعيد 1 عند النجاح، 0 عند الخطأ.
    TYPE - نوع البيانات. انظر NetSNMP::ASN للأنواع الصالحة.
    DATA - البيانات المراد إرجاعها.
    $ret = $request->setValue(ASN_OCTET_STR, "test");
setError ( REQUEST_INFO, ERROR_CODE )
    يضبط رمز الخطأ المحدد للطلب. انظر قسم رموز الأخطاء (ERROR CODES)
    للحصول على قائمة الرموز الصالحة.
    $request->setError($request_info, SNMP_ERR_NOTWRITABLE);
getProcessed ()
    يشير علامة المعالجة إلى أن الطلب لا يحتاج إلى
    معالجة لأنه تم التعامل معه بالفعل بواسطة شخص آخر (معالج أعلى).
    $processed = $request->getProcessed();
setProcessed ( PROCESSED )
    يضبط علامة المعالجة في الطلب. لا تحتاج عمومًا
    إلى تعيينها بنفسك.
    PROCESSED - 0 = خطأ، 1 = صحيح
    $request->setProcessed(1);
getDelegated ()
    إذا كان بإمكانك التعامل مع طلب في الخلفية أو في وقت لاحق
    (مثال، أنت تنتظر مقبض ملف، أو حركة مرور شبكة، أو ...)، يمكن تعيين علامة التفويض
    في الطلب. عندما
    تتم معالجة الطلب في المستقبل، يجب إعادة تعيين العلامة
    إلى 0 حتى يعرف الوكيل أنه يمكنه إنهاء
    الطلب الأصلي وإعادته إلى المدير. لم يتم اختبار هذا
    داخل Perl، ولكن من المفترض أن يعمل.
    $delegated = $request->getDelegated();
setDelegated ( DELEGATED )
    يضبط علامة التفويض.
    DELEGATED - 0 = خطأ، 1 = صحيح
    $request->setDelegated(1);
getRepeat ()
    تشير علامة التكرار إلى أنه يتم التعامل مع عملية getbulk وهذا
    يشير إلى عدد الإجابات التي يجب إرجاعها. بشكل عام، إذا لم تقم بالتسجيل
    لمعالجة دعم getbulk مباشرة بنفسك، فلن تحتاج
