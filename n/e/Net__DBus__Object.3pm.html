Net__DBus__Object(3pm)

الحزمة: perl-net-dbus

التراخيص: GPL, PerlArtistic

Net::DBus::Object(3) وثائق Perl مقدمة من المستخدم Net::DBus::Object(3)

الاسم

Net::DBus::Object - تنفيذ الكائنات للتصدير إلى الناقل

الملخص

# توصيل كائن بالناقل، تحت خدمة
package main;
use Net::DBus;
# الاتصال بالناقل
my $bus = Net::DBus->find;
# الحصول على خدمة 'org.demo.Hello'
my $service = $bus->export_service("org.demo.Hello");
# تصدير كائننا ضمن الخدمة
my $object = Demo::HelloWorld->new($service);
....بقية البرنامج...
# تعريف حزمة جديدة للكائن الذي سنقوم
# بتصديره
package Demo::HelloWorld;
# تحديد الواجهة الرئيسية التي يوفرها كائننا
use Net::DBus::Exporter qw(org.example.demo.Greeter);
# سنكون كائن DBus
use base qw(Net::DBus::Object);
# تصدير إشارة 'Greeting' تقبل معلمة سلسلة نصية
dbus_signal("Greeting", ["string"]);
# تصدير 'Hello' كطريقة تقبل معلمة سلسلة نصية واحدة،
# وتعيد قيمة سلسلة نصية واحدة
dbus_method("Hello", ["string"], ["string"]);
sub new {
    my $class = shift;
    my $service = shift;
    my $self = $class->SUPER::new($service, "/org/demo/HelloWorld");
    bless $self, $class;
    return $self;
}
sub Hello {
  my $self = shift;
  my $name = shift;
  $self->emit_signal("Greeting", "Hello $name");
  return "Said hello to $name";
}
# تصدير 'Goodbye' كطريقة تقبل معلمة سلسلة نصية واحدة،
# وتعيد سلسلة نصية واحدة، ولكن ضعها
# في الواجهة 'org.exaple.demo.Farewell'
dbus_method("Goodbye", ["string"], ["string"], "org.example.demo.Farewell");
sub Goodbye {
  my $self = shift;
  my $name = shift;
  $self->emit_signal("Greeting", "Goodbye $name");
  return "Said goodbye to $name";
}

الوصف

هذا هو الأساس لتنفيذ الكائنات التي يتم تصديرها مباشرة إلى الناقل. يتم تعيين الطرق المنفذة في فئة فرعية إلى طرق على الناقل. باستخدام هذه الفئة، يقوم التطبيق بربط وظائف RPC مباشرة بنموذج الكائنات الخاص به. قد تفضل التطبيقات بذلك استخدام فئة "Net::DBus::ProxyObject" التي تسمح بالحفاظ على وظائف RPC بشكل منفصل عن نموذج الكائنات الأساسي، عن طريق تمرير استدعاءات طرق RPC.

الطرق

my $object = Net::DBus::Object->new($service, $path)
ينشئ هذا كائن DBus جديد بمسار $path مسجل ضمن الخدمة $service. يجب أن تكون معلمة $path سلسلة نصية تتوافق مع متطلبات DBus المعتادة لمسارات الكائنات، بينما يجب أن تكون معلمة $service مثيلاً لـ Net::DBus::Service. يتم الحصول على الأخير عادةً عن طريق استدعاء طريقة "export_service" على كائن Net::DBus.
my $object = Net::DBus::Object->new($parentobj, $subpath)
ينشئ هذا كائن ابن DBus جديد بمسار $subpath نسبيًا إلى الوالد $parentobj. يجب أن تكون معلمة $subpath سلسلة نصية تتوافق مع متطلبات DBus المعتادة لمسارات الكائنات، بينما يجب أن تكون معلمة $parentobj مثيلاً لـ Net::DBus::BaseObject أو فئة فرعية.
