NetAddr::IP::UtilPP -- وظائف بيرل نقية لـ NetAddr::IP::Util

الملخص

use NetAddr::IP::UtilPP qw(
      hasbits
      shiftleft
      addconst
      add128
      sub128
      notcontiguous
      ipv4to6
      mask4to6
      ipanyto6
      maskanyto6
      ipv6to4
      bin2bcd
      bcd2bin
);
use NetAddr::IP::UtilPP qw(:all)
$rv = hasbits($bits128);
$bitsX2 = shiftleft($bits128,$n);
$carry = addconst($ipv6naddr,$signed_32con);
($carry,$ipv6naddr)=addconst($ipv6naddr,$signed_32con);
$carry = add128($ipv6naddr1,$ipv6naddr2);
($carry,$ipv6naddr)=add128($ipv6naddr1,$ipv6naddr2);
$carry = sub128($ipv6naddr1,$ipv6naddr2);
($spurious,$cidr) = notcontiguous($mask128);
($carry,$ipv6naddr)=sub128($ipv6naddr1,$ipv6naddr2);

الوصف

NetAddr::IP::UtilPP توفر وظائف بيرل نقية لـ NetAddr::IP::Util

  • $rv = hasbits($bits128);

    هذه الوظيفة ترجع صحيحًا إذا كانت هناك بتات واحد موجودة في السلسلة المكونة من 128 بت، وترجع خطأ إذا كانت جميع البتات صفرًا.

    أي:  if (hasbits($bits128)) {
        &do_something;
      }
أو    if (hasbits($bits128 & $mask128) {
        &do_something;
      }

    هذا يسمح بتنفيذ وظائف منطقية بالشكل التالي:

          if ($bits128 & $mask128) {
          ...
المدخل:        سلسلة IPv6 مكونة من 128 بت
الناتج:       صحيح إذا كانت هناك أي بتات موجودة
  • $bitsXn = shiftleft($bits128,$n); 
    المدخل:        متغير سلسلة مكون من 128 بت،
              عدد الإزاحات [اختياري]
الناتج:       بتات X n إزاحات
ملاحظة: يتم إرجاع بتات المدخل
      إذا لم يتم تحديد $n
  • addconst($ipv6naddr,$signed_32con);

    أضف ثابتًا موقعًا إلى متغير سلسلة مكون من 128 بت.

    المدخل:        سلسلة IPv6 مكونة من 128 بت،
              عدد صحيح 32 بت موقع
الناتج:  قيمة واحدة      الحمل (carry)
          مصفوفة       (الحمل, النتيجة)
  • add128($ipv6naddr1,$ipv6naddr2);

    أضف متغيري سلسلة مكونين من 128 بت.

    المدخل:        متغير سلسلة IPv6 مكونة من 128 بت 1،
              متغير سلسلة IPv6 مكونة من 128 بت 2
الناتج:  قيمة واحدة      الحمل
          مصفوفة       (الحمل, النتيجة)
  • sub128($ipv6naddr1,$ipv6naddr2);

    اطرح متغيري سلسلة مكونين من 128 بت.

    المدخل:        متغير سلسلة IPv6 مكونة من 128 بت 1،
              متغير سلسلة IPv6 مكونة من 128 بت 2
الناتج:  قيمة واحدة      الحمل
          مصفوفة       (الحمل, النتيجة)

    ملاحظة: الحمل من هذه العملية هو نتيجة إضافة المكمل الأحادي للمعامل الثاني +1 إلى المعامل الأول. إنه منطقيًا ليس استعارة (borrow).

    أي:    إذا كان المعامل الأول >= المعامل الثاني فإن الحمل = 1
أو      إذا كان المعامل الأول  < المعامل الثاني فإن الحمل = 0
  • ($spurious,$cidr) = notcontiguous($mask128);

    تقوم هذه الوظيفة بعد المواضع المتبقية للبتات في القناع عند إزالة الأصفار الموجودة في أقصى اليمين.

    المدخل:  قناع شبكة مكون من 128 بت
الناتج: ترجع صحيحًا إذا كانت هناك أصفار متفرقة
            متبقية في القناع، وترجع خطأ إذا كان القناع
            عبارة عن وحدات متجاورة،
        قناع CIDR مكون من 128 بت
  • $ipv6naddr = ipv4to6($netaddr);

    قم بتحويل عنوان شبكة IPv4 إلى عنوان شبكة IPv6.

    المدخل:        عنوان شبكة 32 بت
الناتج:      عنوان شبكة 128 بت
  • $ipv6naddr = mask4to6($netaddr);

    قم بتحويل عنوان شبكة IPv4 إلى قناع شبكة IPv6.

    المدخل:        عنوان شبكة/قناع 32 بت
الناتج:      عنوان شبكة/قناع 128 بت

    ملاحظة: ترجع أعلى 96 بت كـ وحدات.

  • $ipv6naddr = ipanyto6($netaddr);

    مشابه لـ ipv4to6 باستثناء أن هذه الوظيفة تأخذ مدخلاً IPv4 أو IPv6 وتعيد دائمًا عنوان شبكة IPv6 مكونًا من 128 بت.

    المدخل:        عنوان شبكة 32 أو 128 بت
الناتج:      عنوان شبكة 128 بت
  • $ipv6naddr = maskanyto6($netaddr);

    مشابه لـ mask4to6 باستثناء أن هذه الوظيفة تأخذ قناع شبكة IPv4 أو IPv6 وتعيد دائمًا قناع شبكة IPv6 مكونًا من 128 بت.

    المدخل:        قناع شبكة 32 أو 128 بت
الناتج:      قناع شبكة 128 بت
  • $netaddr = ipv6to4($pv6naddr);

    اقطع أعلى 96 بت من عنوان 128 بت وأعد البتات الـ 32 السفلية. يعيد عنوان IPv4 كما يعيده inet_aton.

    المدخل:        عنوان شبكة 128 بت
الناتج:      عنوان شبكة inet_aton 32 بت
  • $bcdtext = bin2bcd($bits128);

    قم بتحويل سلسلة ثنائية مكونة من 128 بت إلى أرقام عشرية مشفرة ثنائيًا (BCD).

    المدخل:        متغير سلسلة مكون من 128 بت
الناتج:      سلسلة من أرقام BCD النصية
  • $bits128 = bcd2bin($bcdtxt);

    قم بتحويل سلسلة نصية BCD إلى متغير سلسلة مكون من 128 بت.

    المدخل:        سلسلة من أرقام BCD النصية
الناتج:      متغير سلسلة مكون من 128 بت

NetAddr::IP::UtilPP(3pm)

NAME

NetAddr::IP::UtilPP - Pure Perl utility functions for NetAddr::IP

SYNOPSIS

        use NetAddr::IP::UtilPP qw(parse_ipv4 parse_ipv6);

        my $ipv4 = parse_ipv4('127.0.0.1');
        my $ipv6 = parse_ipv6('::1');

DESCRIPTION

This module provides a set of utility functions that are used by the NetAddr::IP module. These functions are implemented in pure Perl and are intended to be a fallback for systems where the C-based NetAddr::IP::Util module cannot be compiled or installed.

The primary functions provided by this module are:

  • parse_ipv4: Parses an IPv4 address string and returns it in network byte order.
  • parse_ipv6: Parses an IPv6 address string and returns it in network byte order.

These functions are generally slower than their C-based counterparts and should only be used when the C-based module is unavailable.

الوصف

توفر هذه الوحدة مجموعة من وظائف الأدوات المساعدة التي يستخدمها وحدة NetAddr::IP. تم تنفيذ هذه الوظائف بلغة Perl النقية وهي مخصصة كحل بديل للأنظمة التي لا يمكن فيها تجميع أو تثبيت وحدة NetAddr::IP::Util المستندة إلى C.

الوظائف الأساسية التي توفرها هذه الوحدة هي:

  • parse_ipv4: تحلل سلسلة عنوان IPv4 وتعيدها بترتيب البايت الشبكي.
  • parse_ipv6: تحلل سلسلة عنوان IPv6 وتعيدها بترتيب البايت الشبكي.

هذه الوظائف أبطأ بشكل عام من نظيراتها المستندة إلى C ويجب استخدامها فقط عندما تكون الوحدة المستندة إلى C غير متوفرة.
