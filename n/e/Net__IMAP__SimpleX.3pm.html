Net__IMAP__SimpleX(3pm)

الحزمة: perl-net-imap-simple

التراخيص: GPL, PerlArtistic

Net::IMAP::SimpleX(3) وثائق الحزم المساهمة في Perl Net::IMAP::SimpleX(3)

الاسم

Net::IMAP::SimpleX - ملحقات لـ Net::IMAP::Simple

الموجز

use strict;
use warnings;
use Net::IMAP::SimpleX;

يستخدم Net::IMAP::SimpleX Net::IMAP::Simple كأساس، لذا فإن إنشاء الكائنات هو نفس إنشاء الكائنات للكائن الموروث:

my $imap = Net::IMAP::SimpleX->new('imap.example.com') ||
   die "لا يمكن الاتصال بـ IMAP: $Net::IMAP::Simple::errstr\n";
$imap->select("INBOX");

Net::IMAP::SimpleX هي مجموعة من الطرق المفيدة التي ليست بسيطة، وتتطلب Parse::RecDescent، أو تجريبية.

الوصف

تضيف هذه الوحدة بعض الوظائف المفيدة، ولكنها ليست بسيطة، فوق Net::IMAP::Simple.

الطرق

جديد
تفاصيل حول الاستدعاء، اقرأ Net::IMAP::Simple.
ملخص_الجسم
تأخذ الاستدعاءات النموذجية هذا الشكل العام. 
# الحصول على تمثيل لجسم الرسالة
my $summary = $imap->body_summary($message_number);
# رسالة متعددة الأجزاء
if ($summary->has_parts) {
    for my $subpart ($summary->parts) {
        if ($subpart->has_parts) { ... }
        # فحص جزء الرسالة
        my @attr = map { $subpart->$_ } qw/content_type encoding encoded_size/;
        # الحصول على جزء الرسالة
        my $subpart_body = $imap->get($message_number, $subpart->part_number);
    }
} else {
    my $body = $summary->body;
    my @attr = map { $body->$_ } qw/content_type encoding encoded_size/;
}

ترجع هذه الطريقة كائنًا بسيطًا يحتوي على تمثيل لجسم الرسالة. يتم بناء الكائن باستخدام محلل Parse::RecDescent باستخدام ناتج أمر IMAP fetch body. يستخدم المحلل الصيغة الرسمية المحددة بواسطة RFC3501 http://tools.ietf.org/html/rfc3501#section-9.

my $body = $summary->body;
my @attr = map { $body->$_ } qw/
    content_description
    encoded_size
    charset
    content_type
    part_number
    format
    id
    encoding
/;

بالنسبة للرسائل متعددة الأجزاء، يحتوي الكائن على كائنات فرعية لكل جزء من أجزاء الرسالة، يمكن الوصول إليها عن طريق الطريقة parts() وفحصها باستخدام الطريقة has_parts(). تصف الطريقة type نوع الجزء المتعدد (مثل mixed أو alternative). ترجع الطريقة parts قائمة بالأجزاء الفرعية، والتي يمكن أن يكون لها أجزاء فرعية خاصة بها، وهكذا.

مثال على رسالة متعددة الأجزاء، بديلة مع نص جسم وإصدار HTML للجسم سيكون شيئًا مثل:

if ($summary->has_parts) {
    if ($summary->type eq 'alternative') {
        my ($html) = grep { $_->content_type eq 'text/html' } $summary->parts;
    }
}

قد تبدو رسالة متعددة الأجزاء معقدة حقًا مثل:

if ($summary->has_parts && $summary->type eq 'mixed') {
    for my $part ($summary->parts) {
        if ($part->has_parts && $part->type eq 'mixed') { ... }
        ...
    }
}
جلب
يرجع أمر fetch أجزاء الرسائل التي يطلبها المستخدمون. إنه معقد للغاية (يتبع RFC3501 باستخدام قواعد/محلل)، ولكن هناك بعض الأنماط الأساسية التي يتبعها. 
my $res  =$imap->fetch('30:32' => 'UID BODY.PEEK[HEADER.FIELDS (DATE)] FLAGS')
# $res = {
#   30 => {
#           "BODY[HEADER.FIELDS (DATE)]" => "Date: Sun, 18 Jul 2010 20:54:48 -0400\r\n\r\n",
#           "FLAGS" => ["\\Flagged", "\\Seen"],
#           "UID" => 58890,
#         },
#   31 => {
#           "BODY[HEADER.FIELDS (DATE)]" => "Date: Wed, 21 Jul 2010 09:09:04 -0400\r\n\r\n",
#           "FLAGS" => ["\\Seen"],
#           "UID" => 58891,
#         },
#   32 => {
#           "BODY[HEADER.FIELDS (DATE)]" => "Date: Sat, 24 Jul 2010 05:12:06 -0700\r\n\r\n",
#           "FLAGS" => ["\\Seen"],
#           "UID" => 58892,
#         },
# }

سيتم إرجاع ما يسمى "الأقواس" كمصفوفة (انظر "FLAGS") ولكن سيتم إرجاع كل شيء آخر تقريبًا كسلاسل. وهذا يشمل الاستعلامات ذات الأقواس. على سبيل المثال، "BODY.PEAK[HEADER.FIELDS (DATE FROM SUBJECT)]"، ستكون النتيجة هي أسطر رأس RFC822 (مثل المثال أعلاه "Date: Sun, ...").

لمزيد من المعلومات حول أنواع الاستعلامات المختلفة، انظر RFC3501. هناك عدد مذهل من الأشياء التي يمكن الاستعلام عنها.

uidfetch
هذا هو نفس الطريقة fetch() السابقة تقريبًا، ولكن الاستعلام يتم إجراؤه على UID بدلاً من التسلسل. مفاتيح $res لا تزال التسلسلات.

my $res  =$imap->fetch('58890' => 'UID BODY.PEEK[HEADER.FIELDS (DATE)] FLAGS')
# $res = {
#   30 => {
#           "BODY[HEADER.FIELDS (DATE)]" => "Date: Sun, 18 Jul 2010 20:54:48 -0400\r\n\r\n",
#           "FLAGS" => ["\\Flagged", "\\Seen"],
#           "UID" => 58890,
#         },
#   ...
