Number__Misc(3pm)

الحزمة: perl-number-misc

التراخيص: PerlArtistic, GPL

Number::Misc(3) توثيق Perl مساهم من المستخدم Number::Misc(3)

الاسم

Number::Misc - أدوات مفيدة للأرقام

ملخص

use Number::Misc ':all';
is_numeric('x');        # خاطئ
to_number('3,000');     # 3000
commafie('3000');       # 3,000
zero_pad(2, 10);        # 0000000002
rand_in_range(3, 10);   # رقم عشوائي من 3 إلى 10، شامل
is_even(3)              # صحيح
is_odd(4);              # صحيح

الوصف

يوفر Number::Misc بعض الأدوات المتنوعة المفيدة لمعالجة الأرقام. تتعامل هذه الأدوات مع معالجة الأرقام كسلاسل نصية، أو تحديد الخصائص الأساسية للأرقام، أو اختيار رقم عشوائي من نطاق.

التثبيت

يمكن تثبيت Number::Misc بالروتين المعتاد:

perl Makefile.PL
make
make test
make install

الوظائف

is_numeric

يعيد صحيحًا إذا كان المقياس المعطى هو رقم. قيمة غير محددة تعيد خاطئًا. يُعرّف "الرقم" بأنه يتكون حصريًا من أرقام (أي الأحرف 0-9)، مع فاصلة عشرية واحدة على الأكثر، وإشارة ناقص أو زائد واحدة في البداية على الأكثر.

is_numeric('3');       # صحيح
is_numeric('-3');      # صحيح
is_numeric('+3');      # صحيح
is_numeric('0003');    # صحيح
is_numeric('0.003');   # صحيح
is_numeric('0.00.3');  # خاطئ
is_numeric('3,003');   # خاطئ
is_numeric('  3');     # خاطئ
is_numeric(undef);     # خاطئ
الخيار: convertible
إذا كنت ترغب في اختبار ما إذا كانت السلسلة النصية يمكن أن تكون رقمًا إذا تم تشغيلها من خلال to_number()، فاستخدم الخيار convertible. 
is_numeric('3,003',  convertible=>1);  # صحيح
is_numeric('  3',    convertible=>1);  # صحيح
is_numeric('0.00.3', convertible=>1);  # خاطئ

to_number

يحول سلسلة نصية إلى رقم عن طريق إزالة الفواصل والمسافات. إذا لم يكن بالإمكان تحويل السلسلة، فإنه يعيد undefined. بعض الأمثلة:

to_number(' 3 ');       # يعيد 3
to_number(' 3,000 ');   # يعيد 3000
to_number('whatever');  # يعيد undefined
الخيار: always_number
إذا لم يكن بالإمكان تحويل السلسلة النصية إلى رقم، فقم بإعادة 0 بدلاً من undefined. على سبيل المثال، هذا الاستدعاء: 
to_number('whatever', always_number=>1)

يعيد 0.

commafie

يحول رقمًا إلى سلسلة نصية تمثل نفس الرقم ولكن مع فواصل

commafie(2000);     #  2,000
commafie(-2000);    # -1,000
commafie(2000.33);  #  2,000.33
commafie(100);      #    100

الخيار: sep

يسمح لك الخيار "sep" بتعيين ما تستخدمه كفاصل بدلاً من الفاصلة. على سبيل المثال، إذا كنت ترغب في استخدام ":" بدلاً من ","، فستفعل ذلك كما يلي:

commafie('2000', sep=>':');

والذي سيعطيك هذا:

2:000

zero_pad

يضيف أصفارًا في بداية الرقم لجعله بطول محدد. المعامل الأول هو الرقم، والثاني هو الطول المستهدف. إذا كان طول الرقم مساويًا للطول المحدد أو أطول منه، فلن يتغير شيء.

zero_pad(2, 3);   # 002
zero_pad(2, 10);  # 0000000002
zero_pad(444, 2); # 444

rand_in_range

بالنظر إلى الحدود الدنيا والعليا، يعيد رقمًا عشوائيًا أكبر من أو يساوي الحد الأدنى وأقل من أو يساوي الحد الأعلى. يعمل فقط على الأعداد الصحيحة.

rand_in_range(3, 10);   # رقم عشوائي من 3 إلى 10، شامل
rand_in_range(-1, 10);  # رقم عشوائي من -1 إلى 10، شامل

is_even / is_odd

"is_even" يعيد صحيحًا إذا كان الرقم زوجيًا. "is_odd" يعيد صحيحًا إذا كان الرقم فرديًا. الأرقام غير الصحيحة والأرقام العشرية تعيد undefined.
