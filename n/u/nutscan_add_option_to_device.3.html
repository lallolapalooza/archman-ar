nutscan_add_option_to_device(3)

الحزمة: nut

التراخيص: GPL-2.0-or-later AND GPL-3.0-or-later

NUTSCAN_ADD_OPTION_T(3) دليل NUT NUTSCAN_ADD_OPTION_T(3)

الاسم

nutscan_add_option_to_device, nutscan_add_commented_option_to_device - إضافة بيانات خيار إلى الجهاز المحدد.

ملخص

        #include <nut-scan.h>
        /* إضافة بيانات خيار ممكّن إلى الجهاز المحدد. */
        void nutscan_add_option_to_device(
                nutscan_device_t * device,
                char * option_name,
                char * value);
        /* منذ الإصدار 2.5.0 من libnutscan:
         * إضافة بيانات خيار إلى الجهاز المحدد مع علامة تعليق اختيارية
         * للخيارات المقترحة، ولكنها غير ممكّنة حاليًا للاستخدام الفعلي. */
        void nutscan_add_commented_option_to_device(
                nutscan_device_t * device,
                char * option_name,
                char * value,
                char * comment_tag);

الوصف

يحتوي nutscan_device_t على المتغيرات التالية:

nutscan_device_type_t   type;
char *          driver;
char *          alt_driver_names;
char *          port;
nutscan_options_t       opt;
struct nutscan_device * prev;
struct nutscan_device * next;

هذه قائمة مزدوجة للروابط للأجهزة. يتم وصف كل device بنوعه، واسم مشغله، ومنفذه، وأي عدد من البيانات الاختيارية.

تقوم nutscan_add_option_to_device() بإضافة بيانات اختيارية في device المحدد. تتكون البيانات الاختيارية من option_name وقيمة value مرتبطة بها، و comment_tag اختياري. يتم تخزين نسخ من option_name و value و comment_tag في device، لذلك يمكن للمتصل استدعاء free(3) بأمان لجميع السلاسل النصية الأصلية المستخدمة كوسائط.


ملاحظة

القيمة غير الفارغة لـ comment_tag تجعل الخيار غير ممكّن للاستخدام الحالي. اعتمادًا على تنسيق الإخراج، قد يتم تجاهله تمامًا، أو عرضه كتعليق مع هذه القيمة كبادئة (يمكن تمرير سلسلة نصية فارغة "\0" لعدم وجود بادئة).

يمكن فحص هذه الخيارات وقيمها بشكل إضافي والتأكد من صحتها والإبلاغ عنها كتحذيرات بواسطة الموزع nutscan_display_sanity_check() وطرقها ذات الصلة التي تنفذ منطق الفحوصات المحددة. يتم استخدام هذا على سبيل المثال عند إنشاء اقتراحات لمحتوى ملف ups.conf باستخدام الطريقة nutscan_display_ups_conf_with_sanity_check().

ملاحظات

تقنيًا، يتم تعريف الدالة حاليًا في الملف nutscan-device.h.
