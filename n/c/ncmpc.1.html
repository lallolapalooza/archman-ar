NAME

ncmpc - وثائق ncmpc

DESCRIPTION

افتراضيًا، يتصل ncmpc بنسخة MPD المحلية. يمكن تحديد نسخة MPD مختلفة باستخدام خيارات سطر الأوامر --host و --port، أو عن طريق تعيين متغيرات البيئة MPD_HOST و MPD_PORT:

ncmpc --host=musicserver --port=44000

يمكنك الاتصال بـ "مقبس" محلي عن طريق تعيين المضيف إلى مسار ملف (مثل "/run/mpd/socket"). يمكن استخدام المقابس المجردة مع بادئة "@" (مثل "@mpd").

لاستخدام كلمة مرور مع MPD، قم بتعيين MPD_HOST إلى password@host أو استخدم خيار سطر الأوامر --password. القيم من سطر الأوامر تتجاوز القيم من البيئة.

SYNOPSIS

ncmpc [options]

OPTIONS

-?, --help
عرض المساعدة.
-V, --version
عرض معلومات الإصدار وتكوين وقت البناء.
-c, --colors
تمكين الألوان.
-C, --no-colors
تعطيل الألوان.
-m, --mouse
تمكين الماوس.
--host=HOST
مضيف MPD للاتصال به.
--port=PORT, -p PORT
المنفذ للاتصال به.
-P, --password=PASSWORD
استخدام كلمة مرور عند الاتصال.
-f, --config=FILE
قراءة التكوين من الملف.
-k, --key-file=FILE
قراءة ربطات المفاتيح من الملف.

CONFIGURATION

عند بدء تشغيل ncmpc، يحاول قراءة ملف تكوين المستخدم، $XDG_CONFIG_HOME/ncmpc/config (عادةً ~/.config/ncmpc/config). إذا لم يتم العثور على ملف تكوين مستخدم، يحاول ncmpc تحميل الإعدادات العامة من $SYSCONFDIR/ncmpc/config (المسار الفعلي معروض في خرج خيار --version). تم شحن مثال لملف تكوين (config.sample) مع ncmpc.

الاتصال

host = HOST - مضيف MPD للاتصال به.

port = PORT - المنفذ للاتصال به.

password = PASSWORD - استخدم كلمة مرور عند الاتصال.

timeout = TIMEOUT - محاولة إعادة الاتصال بـ mpd إذا لم يتم استلام استجابة لأمر خلال TIMEOUT ثانية. تحديد قيمة في ملف التكوين يتجاوز متغير البيئة "MPD_TIMEOUT". إذا لم يتم تحديد مهلة في ملف التكوين أو في البيئة، فإن الافتراضي هو 5 ثوانٍ.

الواجهة

show-title-bar = yes|no - "no" يسمح بإخفاء شريط العنوان.

show-server-address = yes|no - "yes" يعرض عنوان الخادم في شريط العنوان. هذا معطل افتراضيًا لأن معظم الناس يستخدمون نسخة MPD واحدة فقط.

enable-mouse = yes|no - تمكين دعم الماوس (إذا تم تمكينه وقت التجميع).

screen-list = SCREEN1 SCREEN2... - قائمة بالشاشات للتبديل من خلالها عند استخدام الأوامر screen-next (مرتبط بـ "Tab" افتراضيًا) و screen-prev ("Shift+Tab"). الخيارات الصالحة، إذا تم تمكينها وقت التجميع، هي:

help, playlist, browse, library, search, lyrics, outputs, chat, song, keydef

الافتراضي هو playlist browser، أي أن مفتاح "Tab" يبدل فقط بين هاتين الصفحتين.

library-page-tags = TAG1 TAG2 ... - قائمة بالعلامات لتجميع صفحة المكتبة. الافتراضي هو artist album.

search-mode = MODE - وضع البحث الافتراضي لشاشة البحث. MODE يجب أن يكون أحد العناوين، الفنان، الألبوم، اسم الملف، والفنان+العنوان، أو فهرس عددي (0 للعناوين، 1 للفنان، إلخ).

auto-center = yes|no - تمكين/تعطيل وضع التوسيط التلقائي. عند تمكين وضع التوسيط التلقائي، يقوم ncmpc بتوسيط المسار الحالي في نافذة قائمة التشغيل.

scroll-offset = NUM - الاحتفاظ بما لا يقل عن NUM سطور فوق وأسفل المؤشر في نوافذ القائمة، إن أمكن.

find-show-last = yes|no - عرض الاستعلام الأخير بدلاً من سطر فارغ للبحث.

find-wrap = yes|no - وضع البحث المغلف.

wrap-around = yes|no - حركة المؤشر المغلفة.

bell-on-wrap = yes|no - إصدار صوت جرس عند اكتمال البحث.

audible-bell = yes|no - تشغيل جرس مسموع عند التنبيهات.

visible-bell = yes|no - جرس مرئي عند التنبيهات.

crossfade-time = CROSSFADE TIME - وقت الانتقال التلقائي الافتراضي بالثواني.

seek-time = NUM - البحث للأمام/للخلف بمقدار NUM ثانية.

lyrics-timeout = NUM - يوقف تنزيل كلمات أغنية بعد انتهاء وقت الانتظار NUM ثانية، إذا كان NUM أكبر من صفر.

jump-prefix-only = yes|no - عند استخدام أمر القفز، ابحث عن البادئة لإدخال. هذا يعني أن كتابة "m" ستبدأ في أول إدخال يبدأ بـ "m".

lyrics-autosave = yes|no - حفظ كلمات الأغاني تلقائيًا بعد استلامها.

lyrics-show-plugin = yes|no - عرض اسم المكون الإضافي المستخدم لاستلام كلمات الأغاني في شاشة كلمات الأغاني.

text-editor = EDITOR - محرر النصوص المستخدم لتحرير كلمات الأغاني.

chat-prefix = PREFIX - إضافة PREFIX إلى الرسائل المرسلة مع شاشة الدردشة. افتراضيًا، يتم تمييزها باسم المستخدم الحالي محاطًا بـ < و > ومسافة (أي "<name> ").

second-column = yes|no - عرض طول الأغنية في عمود ثانٍ.

العرض

welcome-screen-list = yes|no - عرض قائمة بالشاشات في السطر العلوي.

wide-cursor = yes|no - جعل المؤشر بعرض الشاشة.

hardware-cursor = yes|no - استخدام مؤشر الأجهزة الخاص بالطرفية بدلاً من الألوان العكسية.

hide-cursor = NUM - إخفاء مؤشر قائمة التشغيل بعد NUM ثانية من عدم النشاط.

scroll = yes|no - تمرير العنوان إذا كان طويلاً جدًا للشاشة.

scroll-sep = STRING - الفاصل لإظهاره في نهاية العنوان المتمرر.

list-format = SONG FORMAT - التنسيق المستخدم لعرض الأغاني في النافذة الرئيسية.

search-format = SONG FORMAT - التنسيق المستخدم لعرض الأغاني في نافذة البحث. الافتراضي هو استخدام list-format.

status-format = SONG FORMAT - التنسيق المستخدم لعرض الأغاني في سطر الحالة.

status-message-time = TIME - الوقت، بالثواني، الذي سيتم فيه عرض رسائل الحالة.

display-time = yes|no - عرض الوقت في شريط الحالة عند عدم النشاط.

timedisplay-type = elapsed|remaining|none - يحدد ما إذا كان سيتم عرض الوقت المتبقي أو المنقضي في نافذة الحالة. الافتراضي هو المنقضي. none لا يعرض الوقت على الإطلاق.

visible-bitrate = yes|no - عرض معدل البت في شريط الحالة عند تشغيل بث.

set-xterm-title = yes|no - تغيير عنوان XTerm (لن يقوم ncmpc باستعادة العنوان)
