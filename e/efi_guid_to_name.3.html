efi_guid_to_name(3)

الحزمة: efivar

التراخيص: LGPL-2.1-or-later

EFI_GET_VARIABLE(3) دليل الدوال المكتبية EFI_GET_VARIABLE(3)

الاسم

efi_variables_supported, efi_del_variable, efi_get_variable, efi_get_variable_attributes, efi_get_variable_size, efi_set_variable - التعامل مع متغيرات UEFI

الملخص

#include <efivar.h>

int efi_variables_supported(void);
int efi_del_variable(efi_guid_t guid, const char *name);
int efi_get_variable(efi_guid_t guid, const char *name,
				 void **data, ssize_t *data_size,
				 uint32_t *attributes);
int efi_get_variable_attributes(efi_guid_t guid, const char *name,
						  uint32_t *attributes);
int efi_get_variable_exists(efi_guid_t guid, const char *name);
int efi_get_variable_size(efi_guid_t guid, const char *name,
					 size_t *size);
int efi_append_variable(efi_guid_t guid, const char *name,
				 void *data, size_t data_size,
				 uint32_t attributes);
int efi_set_variable(efi_guid_t guid, const char *name,
				 void *data, size_t data_size,
				 uint32_t attributes, mode_t mode);
int efi_get_next_variable_name(efi_guid_t **guid, char **name);
int efi_str_to_guid(const char *s, efi_guid_t *guid);
int efi_guid_to_str(const efi_guid_t *guid, char **sp);
int efi_name_to_guid(const char *name, efi_guid_t *guid);
int efi_id_guid_to_guid(const char *id_guid, efi_guid_t *guid);
int efi_guid_to_name(efi_guid_t *guid, char **name);
int efi_guid_to_id_guid(efi_guid_t *guid, char **id_guid);
int efi_guid_to_symbol(efi_guid_t *guid, char **symbol);
int efi_symbol_to_guid(const char *symbol, efi_guid_t *guid);

الوصف

efi_variables_supported() تختبر ما إذا كانت ميزة متغيرات UEFI مدعومة على الجهاز الحالي.

efi_del_variable() تحذف المتغير المحدد بواسطة guid و name.

efi_get_variable() تحصل على المتغير المحدد بواسطة guid و name. يتم تخزين القيمة في data، وحجمها في data_size، ويتم تخزين سماتها في attributes.

efi_get_variable_attributes() تحصل على سمات المتغير المحدد بواسطة guid و name.

efi_get_variable_exists() تحصل على ما إذا كان المتغير المحدد بواسطة guid و name موجودًا.

efi_get_variable_size() تحصل على حجم البيانات للمتغير المحدد بواسطة guid و name.

efi_append_variable() تلحق data بحجم size بالمتغير المحدد بواسطة guid و name.

efi_set_variable() تعيّن المتغير المحدد بواسطة guid و name، وتعّين وضع الملف إلى mode، مع مراعاة umask. لاحظ أن الوضع لن يستمر بعد إعادة التشغيل، وأن الأذونات تنطبق فقط إذا كانت على أنظمة تستخدم efivarfs.

efi_get_next_variable_name() تتنقل عبر المتغيرات الموجودة حاليًا، وتعيد guid واسمًا.

efi_str_to_guid() تحلل GUID UEFI من تنسيق سلسلة إلى efi_guid_t يقدمها المتصل.

efi_guid_to_str() تنشئ تمثيلًا سلسلة لـ UEFI GUID. إذا كان sp هو NULL، فإنها تعيد حجم السلسلة. إذا لم يكن sp هو NULL ولكن *sp هو NULL، فإنها تخصص سلسلة وتعيدها. تقع على عاتق المتصل مسؤولية تحرير هذه السلسلة. إذا لم يكن sp هو NULL و *sp ليس NULL، فإن efi_guid_to_str() تفترض وجود تخصيص بالحجم المناسب وتستخدمه.

efi_name_to_guid() تترجم من اسم معروف إلى efi_guid_t يقدمها المتصل.

efi_guid_to_name() تترجم من efi_guid_t إلى اسم معروف. إذا لم يكن لدى GUID المقدم اسم معروف، فإن هذه الدالة تعادل efi_guid_to_str().

efi_guid_to_id_guid() تترجم من efi_guid_t إلى {ID GUID}. إذا كان لدى GUID المقدم اسم معروف، فسيكون {ID GUID} بالشكل "{name_here}". إذا لم يكن كذلك، فسيكون بالشكل "{66b2af1c-6211-4082-95cb-9f73a4795a7e}".

efi_id_guid_to_guid() تترجم من {ID GUID} إلى efi_guid_t يقدمها المتصل.

efi_guid_to_symbol() تترجم من efi_guid_t إلى اسم رمز C فريد (ضمن libefivar). هذه الأسماء الرمزية مفيدة للطباعة كمعرّف فريد وسهل التحليل، وهي أيضًا متوفرة من المكتبة وملفاتها الرأسية.

efi_symbol_to_guid() تترجم من اسم رمز libefivar efi_guid_$FOO إلى efi_guid_t يقدمها المتصل.

قيمة الإرجاع

efi_variables_supported() تعيد true إذا كانت المتغيرات مدعومة على الأجهزة قيد التشغيل، و false إذا لم تكن كذلك.

efi_get_next_variable_name() تعيد 0 عند اكتمال التكرار، و 1 عند عدم اكتمال التكرار، و -1 في حالة حدوث خطأ. في حالة حدوث خطأ، يتم تعيين errno(3) بشكل مناسب.

efi_del_variable(), efi_get_variable(), efi_get_variable_attributes(), efi_get_variable_exists(), efi_get_variable_size(), efi_append_variable(), efi_set_variable(), efi_str_to_guid(), efi_guid_to_str(), efi_name_to_guid(), و efi_guid_to_name() تعيد قيمة سالبة في حالة الخطأ وصفرًا في حالة النجاح.
