الوصف

أحد قواعد البيانات المتاحة مع مفتاح تبديل خدمة الأسماء (NSS) هو قاعدة بيانات المستعارات، التي تحتوي على المستعارات البريدية. (لمعرفة قواعد البيانات المدعومة، جرب getent --help.) يتم توفير ست وظائف للوصول إلى قاعدة بيانات المستعارات.

تقوم الوظيفة getaliasent() بإرجاع مؤشر إلى بنية تحتوي على معلومات المجموعة من قاعدة بيانات المستعارات. في المرة الأولى التي يتم استدعاؤها فيها، تقوم بإرجاع الإدخال الأول؛ وبعد ذلك، تقوم بإرجاع الإدخالات المتتالية.

تقوم الوظيفة setaliasent() بإعادة تعيين مؤشر الملف إلى بداية قاعدة بيانات المستعارات.

تقوم الوظيفة endaliasent() بإغلاق قاعدة بيانات المستعارات.

getaliasent_r() هي النسخة القابلة لإعادة الدخول للوظيفة السابقة. يتم تخزين البنية المطلوبة عبر الوسيط الأول، ولكن يجب على المبرمج ملء الوسطاء الآخرين أيضًا. عدم توفير مساحة كافية يتسبب في فشل الوظيفة.

تأخذ الوظيفة getaliasbyname() الوسيط name وتقوم بالبحث في قاعدة بيانات المستعارات. يتم إرجاع الإدخال كمؤشر إلى struct aliasent.

getaliasbyname_r() هي النسخة القابلة لإعادة الدخول للوظيفة السابقة. يتم تخزين البنية المطلوبة عبر الوسيط الثاني، ولكن يجب على المبرمج ملء الوسطاء الآخرين أيضًا. عدم توفير مساحة كافية يتسبب في فشل الوظيفة.

تم تعريف struct aliasent في <aliases.h>:


struct aliasent {
    char    *alias_name;             /* اسم المستعار */
    size_t   alias_members_len;
    char   **alias_members;          /* قائمة أسماء المستعارات */
    int      alias_local;
};

قيمة الإرجاع

تقوم الوظيفتان getaliasent_r() و getaliasbyname_r() بإرجاع قيمة غير صفرية عند حدوث خطأ.

الملفات

قاعدة بيانات المستعارات الافتراضية هي الملف /etc/aliases. يمكن تغيير ذلك في ملف /etc/nsswitch.conf.

أمثلة

المثال التالي يتم تجميعه مع gcc example.c -o example. سيقوم بعرض جميع الأسماء في قاعدة بيانات المستعارات.

#include <aliases.h>
#include <errno.h>
#include <stdio.h>
#include <stdlib.h>
int
main(void)
{
    struct aliasent *al;
    setaliasent();
    for (;;) {
        al = getaliasent();
        if (al == NULL)
            break;
        printf("Name: %s\n", al->alias_name);
    }
    if (errno) {
        perror("reading alias");
        exit(EXIT_FAILURE);
    }
    endaliasent();
    exit(EXIT_SUCCESS);
}
