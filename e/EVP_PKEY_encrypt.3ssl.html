EVP_PKEY_encrypt(3ssl)

الحزمة: openssl

التراخيص: Apache-2.0

EVP_PKEY_ENCRYPT(3ssl) OpenSSL EVP_PKEY_ENCRYPT(3ssl)

الاسم

EVP_PKEY_encrypt_init_ex, EVP_PKEY_encrypt_init, EVP_PKEY_encrypt - التشفير باستخدام خوارزمية المفتاح العام

الملخص

#include <openssl/evp.h>
int EVP_PKEY_encrypt_init(EVP_PKEY_CTX *ctx);
int EVP_PKEY_encrypt_init_ex(EVP_PKEY_CTX *ctx, const OSSL_PARAM params[]);
int EVP_PKEY_encrypt(EVP_PKEY_CTX *ctx,
                     unsigned char *out, size_t *outlen,
                     const unsigned char *in, size_t inlen);

الوصف

تقوم الدالة EVP_PKEY_encrypt_init() بتهيئة سياق خوارزمية المفتاح العام باستخدام المفتاح pkey لعملية تشفير.

تقوم الدالة EVP_PKEY_encrypt_init_ex() بتهيئة سياق خوارزمية المفتاح العام باستخدام المفتاح pkey لعملية تشفير وتعيين params الخاصة بالخوارزمية.

تقوم الدالة EVP_PKEY_encrypt() بإجراء عملية تشفير بالمفتاح العام باستخدام ctx. يتم تحديد البيانات المراد تشفيرها باستخدام المعاملين in و inlen. إذا كان out يساوي NULL، فسيتم كتابة الحد الأقصى لحجم مخزن الإخراج في المعامل outlen. إذا لم يكن out يساوي NULL، فيجب أن يحتوي المعامل outlen قبل الاستدعاء على طول مخزن out. إذا نجح الاستدعاء، فسيتم كتابة البيانات المشفرة في out وسيتم كتابة كمية البيانات المكتوبة في outlen.

ملاحظات

بعد استدعاء EVP_PKEY_encrypt_init()، يمكن إجراء عمليات تحكم خاصة بالخوارزمية لتعيين أي معلمات مناسبة للعملية. يمكن تضمين هذه العمليات في استدعاء EVP_PKEY_encrypt_init_ex().

يمكن استدعاء الدالة EVP_PKEY_encrypt() أكثر من مرة على نفس السياق إذا تم إجراء عدة عمليات باستخدام نفس المعلمات.

قيم الإرجاع

EVP_PKEY_encrypt_init() و EVP_PKEY_encrypt_init_ex() و EVP_PKEY_encrypt() ترجع 1 للنجاح و 0 أو قيمة سالبة للفشل. على وجه الخصوص، يشير قيمة الإرجاع -2 إلى أن العملية غير مدعومة بواسطة خوارزمية المفتاح العام.

أمثلة

تشفير البيانات باستخدام OAEP (لمفاتيح RSA). انظر أيضًا PEM_read_PUBKEY(3) أو d2i_X509(3) لمعرفة طرق تحميل مفتاح عام. يمكنك أيضًا ببساطة تعيين 'eng = NULL;' للبدء بتنفيذ RSA الافتراضي لـ OpenSSL:

#include <openssl/evp.h>
#include <openssl/rsa.h>
#include <openssl/engine.h>
EVP_PKEY_CTX *ctx;
ENGINE *eng;
unsigned char *out, *in;
size_t outlen, inlen;
EVP_PKEY *key;
/*
 * ملاحظة: يفترض أن eng و key و in و inlen تم إعدادها بالفعل،
 * وأن key هو مفتاح RSA عام
 */
ctx = EVP_PKEY_CTX_new(key, eng);
if (!ctx)
    /* حدث خطأ */
if (EVP_PKEY_encrypt_init(ctx) <= 0)
    /* خطأ */
if (EVP_PKEY_CTX_set_rsa_padding(ctx, RSA_PKCS1_OAEP_PADDING) <= 0)
    /* خطأ */
/* تحديد طول المخزن المؤقت */
if (EVP_PKEY_encrypt(ctx, NULL, &outlen, in, inlen) <= 0)
    /* خطأ */
out = OPENSSL_malloc(outlen);
if (!out)
    /* فشل تخصيص الذاكرة */
if (EVP_PKEY_encrypt(ctx, out, &outlen, in, inlen) <= 0)
    /* خطأ */
/* البيانات المشفرة هي outlen بايت مكتوبة في المخزن المؤقت out */
