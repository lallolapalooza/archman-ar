EVP_PKEY_is_a(3ssl)

الحزمة: openssl

التراخيص: Apache-2.0

EVP_PKEY_IS_A(3ssl) OpenSSL EVP_PKEY_IS_A(3ssl)

الاسم

EVP_PKEY_is_a, EVP_PKEY_can_sign, EVP_PKEY_type_names_do_all, EVP_PKEY_get0_type_name, EVP_PKEY_get0_description, EVP_PKEY_get0_provider - وظائف نوع المفتاح وقدراته

الملخص

#include <openssl/evp.h>
int EVP_PKEY_is_a(const EVP_PKEY *pkey, const char *name);
int EVP_PKEY_can_sign(const EVP_PKEY *pkey);
int EVP_PKEY_type_names_do_all(const EVP_PKEY *pkey,
                               void (*fn)(const char *name, void *data),
                               void *data);
const char *EVP_PKEY_get0_type_name(const EVP_PKEY *key);
const char *EVP_PKEY_get0_description(const EVP_PKEY *key);
const OSSL_PROVIDER *EVP_PKEY_get0_provider(const EVP_PKEY *key);

الوصف

EVP_PKEY_is_a() تتحقق مما إذا كان نوع المفتاح لـ pkey هو name.

EVP_PKEY_can_sign() تتحقق مما إذا كانت الوظيفة لنوع المفتاح لـ pkey تدعم التوقيع. لا يتم إجراء أي فحص آخر، مثل ما إذا كان pkey يحتوي على مفتاح خاص.

EVP_PKEY_type_names_do_all() تجتاز جميع الأسماء لنوع المفتاح لـ pkey، وتستدعي fn مع كل اسم و data. على سبيل المثال، قد يتم تسمية مفتاح RSA EVP_PKEY بكل من "RSA" و "rsaEncryption". يعتمد ترتيب الأسماء على تنفيذ الموفر الذي يمتلك المفتاح.

EVP_PKEY_get0_type_name() تُرجع اسم نوع المفتاح الأول الذي تم العثور عليه لـ pkey المعطى. لاحظ أن pkey قد يكون لديه مرادفات متعددة مرتبطة به. في هذه الحالة، يعتمد الاسم الذي سيتم إرجاعه على تنفيذ الموفر الذي يمتلك المفتاح. يحتفظ كائن pkey بملكية السلسلة المرجعة ولا ينبغي تحريرها بواسطة المتصل.

EVP_PKEY_get0_description() تُرجع وصفًا لنوع EVP_PKEY، مخصص للعرض والاستهلاك البشري. الوصف هو تقدير لتنفيذ نوع المفتاح.

EVP_PKEY_get0_provider() تُرجع موفر EVP_KEYMGMT(3) لـ EVP_PKEY.

قيم الإرجاع

EVP_PKEY_is_a() تُرجع 1 إذا كان pkey له نوع المفتاح name، وإلا تُرجع 0.

EVP_PKEY_can_sign() تُرجع 1 إذا كانت وظيفة نوع المفتاح pkey تدعم التوقيع، وإلا تُرجع 0.

EVP_PKEY_get0_type_name() تُرجع الاسم الذي تم العثور عليه أو NULL عند حدوث خطأ.

EVP_PKEY_get0_description() تُرجع الوصف إذا تم العثور عليه أو NULL إذا لم يكن كذلك.

EVP_PKEY_get0_provider() تُرجع الموفر إذا تم العثور عليه أو NULL إذا لم يكن كذلك.

EVP_PKEY_type_names_do_all() تُرجع 1 إذا تم استدعاء الدالة الراجعة لجميع الأسماء. قيمة الإرجاع 0 تعني أنه لم يتم استدعاء الدالة الراجعة لأي أسماء.

أمثلة

EVP_PKEY_is_a()

ستحدد الموفرات المحملة وأنواع المفاتيح التي تدعمها في النهاية ما هو name الممكن استخدامه مع EVP_PKEY_is_a(). نحن نعلم أن الموفر الافتراضي يدعم مفاتيح RSA و DH و DSA و EC، لذا يمكننا استخدام هذا كمثال تقريبي:

#include <openssl/evp.h>
...
    /* |pkey| هو EVP_PKEY* */
    if (EVP_PKEY_is_a(pkey, "RSA")) {
        BIGNUM *modulus = NULL;
        if (EVP_PKEY_get_bn_param(pkey, "n", &modulus))
            /* افعل ما تريد بالمعامل */
        BN_free(modulus);
    }

EVP_PKEY_can_sign()

#include <openssl/evp.h>
...
    /* |pkey| هو EVP_PKEY* */
    if (!EVP_PKEY_can_sign(pkey)) {
        fprintf(stderr, "ليس مفتاح توقيع!");
        exit(1);
    }
    /* قم بالتوقيع على شيء ما... */
