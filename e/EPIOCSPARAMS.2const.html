ioctl_eventpoll(2) دليل استدعاءات النظام ioctl_eventpoll(2)

الاسم

ioctl_eventpoll, EPIOCSPARAMS, EPIOCGPARAMS - عمليات ioctl() لموصوفات epoll

المكتبة

مكتبة C القياسية (libc, -lc)

المُلَخَّص

#include <sys/epoll.h>  /* تعريف الثوابت EPIOC* */
#include <sys/ioctl.h>
int ioctl(int fd, EPIOCSPARAMS, const struct epoll_params *argp);
int ioctl(int fd, EPIOCGPARAMS, struct epoll_params *argp);
#include <sys/epoll.h>
struct epoll_params {
    uint32_t  busy_poll_usecs;   /* عدد الميكروثواني لاستقصاء الـ busy poll */
    uint16_t  busy_poll_budget;  /* الحد الأقصى للحزم لكل استقصاء */
    uint8_t   prefer_busy_poll;  /* تفضيل منطقي */
    /* توسيد البنية ليكون مضاعفاً لـ 64 بت */
    uint8_t   __pad;             /* يجب أن يكون صفراً */
};

الوصف

EPIOCSPARAMS
تعيين بنية epoll_params لتكوين عملية epoll. ارجع إلى وصف البنية أدناه لمعرفة التكوين المدعوم.
EPIOCGPARAMS
الحصول على إعدادات تكوين epoll_params الحالية.

يجب إجراء جميع العمليات الموضحة أعلاه على موصوف epoll، والذي يمكن الحصول عليه باستدعاء epoll_create(2) أو epoll_create1(2).

بنية epoll_params

يشير argp.busy_poll_usecs إلى عدد الميكروثواني التي سيقوم فيها مكدس الشبكة بإجراء استقصاء نشط. خلال هذه الفترة الزمنية، سيتم استقصاء جهاز الشبكة بشكل متكرر للحصول على حزم. لا يمكن أن يتجاوز هذا القيمة INT_MAX.

يشير argp.busy_poll_budget إلى الحد الأقصى لعدد الحزم التي سيقوم مكدس الشبكة باستردادها في كل محاولة استقصاء. لا يمكن أن تتجاوز هذه القيمة NAPI_POLL_WEIGHT (والتي تبلغ 64 اعتبارًا من Linux 6.9)، إلا إذا تم تشغيل العملية باستخدام CAP_NET_ADMIN.

argp.prefer_busy_poll هو حقل منطقي ويجب أن يكون إما 0 (معطل) أو 1 (ممكّن). إذا تم تمكينه، فهذا يشير إلى مكدس الشبكة أن الاستقصاء النشط هو الطريقة المفضلة لمعالجة بيانات الشبكة ويجب على مكدس الشبكة منح التطبيق فرصة لإجراء استقصاء نشط. بدون هذا الخيار، قد تستمر الأنظمة ذات النشاط العالي في معالجة الشبكة من خلال الطريقة العادية لمقاطعات IRQs التي تؤدي إلى تشغيل softIRQ و NAPI.

يجب أن يكون argp.__pad صفراً.

قيمة الإرجاع

عند النجاح، يتم إرجاع 0. عند الفشل، يتم إرجاع -1، ويتم تعيين errno للإشارة إلى الخطأ.

الأخطاء

EOPNOTSUPP
لم يتم تجميع النواة مع دعم الاستقصاء النشط.
EINVAL
fd ليس موصوف ملف صالح.
EINVAL
argp.__pad ليس صفراً.
EINVAL
argp.busy_poll_usecs يتجاوز INT_MAX.
EINVAL
argp.prefer_busy_poll ليس 0 أو 1.
EPERM
يتم تشغيل العملية بدون CAP_NET_ADMIN ويتجاوز argp.busy_poll_budget المحدد NAPI_POLL_WEIGHT.
EFAULT
argp هو عنوان غير صالح.

المعايير

Linux.

التاريخ

أمثلة

/* رمز لتعيين معلمات epoll لتمكين الاستقصاء النشط */
int epollfd = epoll_create1(0);
struct epoll_params params;
if (epollfd == -1) {
    perror("epoll_create1");
    exit(EXIT_FAILURE);
}
memset(&params, 0, sizeof(struct epoll_params));
params.busy_poll_usecs = 25;
params.busy_poll_budget = 8;
params.prefer_busy_poll = 1;
if (ioctl(epollfd, EPIOCSPARAMS, &params) == -1) {
    perror("ioctl");
    exit(EXIT_FAILURE);
}
/* رمز لإظهار كيفية استرداد الإعدادات الحالية */
memset(&params, 0, sizeof(struct epoll_params));
if (ioctl(epollfd, EPIOCGPARAMS, &params) == -1) {
    perror("ioctl");
    exit(EXIT_FAILURE);
}
/* يحتوي هيكل params الآن على المعلمات الحالية */
fprintf(stderr, "epoll usecs: %lu\n", params.busy_poll_usecs);
fprintf(stderr, "epoll packet budget: %u\n", params.busy_poll_budget);
fprintf(stderr, "epoll prefer busy poll: %u\n", params.prefer_busy_poll);
