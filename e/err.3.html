err(3) دليل وظائف المكتبة err(3)

الاسم

err, verr, errx, verrx, warn, vwarn, warnx, vwarnx - رسائل خطأ منسقة

المكتبة

مكتبة C القياسية (libc، -lc)

الملخص

#include <err.h>
[[noreturn]] void err(int eval, const char *fmt, ...);
[[noreturn]] void errx(int eval, const char *fmt, ...);
void warn(const char *fmt, ...);
void warnx(const char *fmt, ...);
#include <stdarg.h>
[[noreturn]] void verr(int eval, const char *fmt, va_list args);
[[noreturn]] void verrx(int eval, const char *fmt, va_list args);
void vwarn(const char *fmt, va_list args);
void vwarnx(const char *fmt, va_list args);

الوصف

تقوم عائلة وظائف err() و warn() بعرض رسالة خطأ منسقة على المخرج القياسي للأخطاء. في جميع الحالات، يتم إخراج الجزء الأخير من اسم البرنامج، ثم نقطتين رأسيتين، ومسافة. إذا كان الوسيط fmt غير NULL، يتم إخراج رسالة الخطأ المنسقة على غرار printf(3). يتم إنهاء المخرج بحرف سطر جديد.

تضيف وظائف err() و verr() و warn() و vwarn() رسالة خطأ تم الحصول عليها من strerror(3) بناءً على المتغير العام errno، مسبوقة بنقطتين رأسيتين أخريين ومسافة ما لم يكن الوسيط fmt NULL.

وظائف errx() و warnx() لا تضيف رسالة خطأ.

وظائف err() و verr() و errx() و verrx() لا تعود، بل تنتهي بقيمة الوسيط eval.

السمات

لشرح المصطلحات المستخدمة في هذا القسم، راجع attributes(7).

الواجهة السمة القيمة

أمثلة

عرض سلسلة معلومات errno الحالية والخروج:


p = malloc(size);
if (p == NULL)
    err(EXIT_FAILURE, NULL);
fd = open(file_name, O_RDONLY, 0);
if (fd == -1)
    err(EXIT_FAILURE, "%s", file_name);

عرض رسالة خطأ والخروج:


if (tm.tm_hour < START_TIME)
    errx(EXIT_FAILURE, "too early, wait until %s",
            start_time_string);

التحذير من خطأ:


fd = open(raw_device, O_RDONLY, 0);
if (fd == -1)
    warnx("%s: %s: trying the block device",
            raw_device, strerror(errno));
fd = open(block_device, O_RDONLY, 0);
if (fd == -1)
    err(EXIT_FAILURE, "%s", block_device);
