efi_symbol_to_guid(3)

الحزمة: efivar

التراخيص: LGPL-2.1-or-later

EFI_GET_VARIABLE(3) دليل وظائف المكتبة EFI_GET_VARIABLE(3)

الاسم

efi_variables_supported, efi_del_variable, efi_get_variable, efi_get_variable_attributes, efi_get_variable_size, efi_set_variable - التلاعب بمتغيرات UEFI

مُلَخَّص

#include <efivar.h>

int efi_variables_supported(void);
int efi_del_variable(efi_guid_t guid, const char *name);
int efi_get_variable(efi_guid_t guid, const char *name,
				 void **data, ssize_t *data_size,
				 uint32_t *attributes);
int efi_get_variable_attributes(efi_guid_t guid, const char *name,
						  uint32_t *attributes);
int efi_get_variable_exists(efi_guid_t guid, const char *name);
int efi_get_variable_size(efi_guid_t guid, const char *name,
					 size_t *size);
int efi_append_variable(efi_guid_t guid, const char *name,
				 void *data, size_t data_size,
				 uint32_t attributes);
int efi_set_variable(efi_guid_t guid, const char *name,
				 void *data, size_t data_size,
				 uint32_t attributes, mode_t mode);
int efi_get_next_variable_name(efi_guid_t **guid, char **name);
int efi_str_to_guid(const char *s, efi_guid_t *guid);
int efi_guid_to_str(const efi_guid_t *guid, char **sp);
int efi_name_to_guid(const char *name, efi_guid_t *guid);
int efi_id_guid_to_guid(const char *id_guid, efi_guid_t *guid);
int efi_guid_to_name(efi_guid_t *guid, char **name);
int efi_guid_to_id_guid(efi_guid_t *guid, char **id_guid);
int efi_guid_to_symbol(efi_guid_t *guid, char **symbol);
int efi_symbol_to_guid(const char *symbol, efi_guid_t *guid);

الوصف

efi_variables_supported() تختبر ما إذا كانت خدمة متغيرات UEFI مدعومة على الجهاز الحالي.

efi_del_variable() تحذف المتغير المحدد بواسطة guid و name.

efi_get_variable() تحصل على المتغير المحدد بواسطة guid و name. يتم تخزين القيمة في data، حجمها في data_size، وخصائصها مخزنة في attributes.

efi_get_variable_attributes() تحصل على خصائص للمتغير المحدد بواسطة guid و name.

efi_get_variable_exists() تحصل على ما إذا كان المتغير المحدد بواسطة guid و name موجودًا.

efi_get_variable_size() تحصل على حجم البيانات للمتغير المحدد بواسطة guid و name.

efi_append_variable() تضيف data بحجم size إلى المتغير المحدد بواسطة guid و name.

efi_set_variable() تضبط المتغير المحدد بواسطة guid و name، وتضبط وضع الملف إلى mode، مع مراعاة umask. لاحظ أن الوضع لن يستمر عبر إعادة التشغيل، وأن الأذونات تنطبق فقط إذا كانت على أنظمة تستخدم efivarfs.

efi_get_next_variable_name() تتكرر عبر المتغيرات الموجودة حاليًا، وتمرر guid واسمًا.

efi_str_to_guid() تحلل GUID UEFI من شكل سلسلة نصية إلى efi_guid_t يقدمها المتصل.

efi_guid_to_str() تنشئ تمثيلاً نصيًا لـ UEFI GUID. إذا كان sp NULL، فإنها تُرجع حجم السلسلة النصية. إذا كان sp ليس NULL ولكن *sp هو NULL، فإنها تقوم بتخصيص سلسلة نصية وتعيدها. تقع على عاتق المتصل مسؤولية تحرير هذه السلسلة النصية. إذا كان sp ليس NULL و *sp ليس NULL، فإن efi_guid_to_str() تفترض وجود تخصيص بالحجم المناسب وتستخدمه.

efi_name_to_guid() تترجم من اسم معروف إلى efi_guid_t يقدمها المتصل.

efi_guid_to_name() تترجم من efi_guid_t إلى اسم معروف. إذا لم يكن لـ GUID المقدم اسم معروف، فإن هذه الوظيفة تعادل efi_guid_to_str().

efi_guid_to_id_guid() تترجم من efi_guid_t إلى {ID GUID}. إذا كان لـ GUID المقدم اسم معروف، فإن {ID GUID} سيكون بالصيغة "name_here". إذا لم يكن كذلك، فسيكون بالصيغة "{66b2af1c-6211-4082-95cb-9f73a4795a7e}".

efi_id_guid_to_guid() تترجم من {ID GUID} إلى efi_guid_t يقدمها المتصل.

efi_guid_to_symbol() تترجم من efi_guid_t إلى اسم رمز C فريد (داخل libefivar). هذه الأسماء الرمزية مفيدة للطباعة كمعرف فريد وسهل التحليل، كما أنها مقدمة من المكتبة وملفاتها الرأسية.

efi_symbol_to_guid() تترجم من اسم رمز efi_guid_$FOO في libefivar إلى efi_guid_t يقدمها المتصل.

قيمة الإرجاع

efi_variables_supported() تُرجع true إذا كانت المتغيرات مدعومة على الجهاز قيد التشغيل، و false إذا لم تكن كذلك.

efi_get_next_variable_name() تُرجع 0 عند اكتمال التكرار، 1 عندما لا يكتمل التكرار، و -1 عند حدوث خطأ. في حالة حدوث خطأ، يتم تعيين errno(3) بشكل مناسب.

efi_del_variable(), efi_get_variable(), efi_get_variable_attributes(), efi_get_variable_exists(), efi_get_variable_size(), efi_append_variable(), efi_set_variable(), efi_str_to_guid(), efi_guid_to_str(), efi_name_to_guid(), و efi_guid_to_name() تُرجع قيمة سالبة عند حدوث خطأ وصفر عند النجاح.
