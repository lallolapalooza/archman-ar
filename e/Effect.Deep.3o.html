type ('a, 'b) continuation = ('a, 'b) continuation

('a,'b) continuation هو استمرارية محدودة تتوقع قيمة 'a وتعيد قيمة 'b.

val continue : ('a, 'b) continuation -> 'a -> 'b

continue k x تستأنف الاستمرارية k بتمرير x إلى k.

يثير Continuation_already_resumed إذا تم استئناف الاستمرارية بالفعل.

val discontinue : ('a, 'b) continuation -> exn -> 'b

discontinue k e تستأنف الاستمرارية k برفع الاستثناء e في k.

val discontinue_with_backtrace : ('a, 'b) continuation -> exn -> Printexc.raw_backtrace -> 'b

discontinue_with_backtrace k e bt تستأنف الاستمرارية k برفع الاستثناء e في k باستخدام bt كنقطة أصل للاستثناء.

type ('a, 'b) handler = {
retc : 'a -> 'b ;
exnc : exn -> 'b ;
effc : 'c. 'c Effect.t -> (('c, 'b) continuation -> 'b) option ;
}

('a,'b) handler هو سجل معالج بثلاثة حقول -- retc هو معالج القيمة، exnc يعالج الاستثناءات، و effc يعالج التأثيرات التي تقوم بها العملية المغلفة بالمعالج.

val match_with : ('c -> 'a) -> 'c -> ('a, 'b) handler -> 'b

match_with f v h تشغل العملية f v في المعالج h.

type 'a effect_handler = {
effc : 'b. 'b Effect.t -> (('b, 'a) continuation -> 'a) option ;
}

'a effect_handler هو معالج عميق مع معالج قيمة هويّة fun x -> x ومعالج استثناء يرفع أي استثناء fun e -> raise e.

val try_with : ('b -> 'a) -> 'b -> 'a effect_handler -> 'a

try_with f v h تشغل العملية f v تحت المعالج h.

val get_callstack : ('a, 'b) continuation -> int -> Printexc.raw_backtrace

get_callstack c n يعيد وصفاً للمكدس العلوي للاستدعاء على الاستمرارية c، بحد أقصى n مدخل.
