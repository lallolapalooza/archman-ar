EVP_PKEY_print_private(3ssl)

الحزمة: openssl

التراخيص: Apache-2.0

EVP_PKEY_PRINT_PRIVATE(3ssl) OpenSSL EVP_PKEY_PRINT_PRIVATE(3ssl)

الاسم

EVP_PKEY_print_public, EVP_PKEY_print_private, EVP_PKEY_print_params, EVP_PKEY_print_public_fp, EVP_PKEY_print_private_fp, EVP_PKEY_print_params_fp - روتينات طباعة خوارزمية المفتاح العام

المُلخص

#include <openssl/evp.h>
int EVP_PKEY_print_public(BIO *out, const EVP_PKEY *pkey,
                          int indent, ASN1_PCTX *pctx);
int EVP_PKEY_print_public_fp(FILE *fp, const EVP_PKEY *pkey,
                             int indent, ASN1_PCTX *pctx);
int EVP_PKEY_print_private(BIO *out, const EVP_PKEY *pkey,
                           int indent, ASN1_PCTX *pctx);
int EVP_PKEY_print_private_fp(FILE *fp, const EVP_PKEY *pkey,
                              int indent, ASN1_PCTX *pctx);
int EVP_PKEY_print_params(BIO *out, const EVP_PKEY *pkey,
                          int indent, ASN1_PCTX *pctx);
int EVP_PKEY_print_params_fp(FILE *fp, const EVP_PKEY *pkey,
                             int indent, ASN1_PCTX *pctx);

الوصف

تقوم الدوال EVP_PKEY_print_public() و EVP_PKEY_print_private() و EVP_PKEY_print_params() بطباعة المكونات العامة أو الخاصة أو مكونات المعلمات للمفتاح pkey على التوالي. يتم إرسال المفتاح إلى BIO out بصيغة قابلة للقراءة البشرية. يشير المعامل indent إلى مدى يجب أن يكون الإخراج مزاحًا.

يسمح المعامل pctx بتعديل إخراج الطباعة بدقة باستخدام خيارات طباعة ASN1. إذا تم تعيين pctx إلى NULL، فسيتم استخدام القيم الافتراضية.

تقوم الدوال EVP_PKEY_print_public_fp() و EVP_PKEY_print_private_fp() و EVP_PKEY_print_params_fp() بنفس الشيء مثل الدوال المعتمدة على BIO ولكنها تستخدم FILE fp بدلاً من ذلك.

ملاحظات

حاليًا، لا تتضمن أي من خوارزميات المفتاح العام أي خيارات في المعامل pctx.

إذا كان المفتاح لا يتضمن جميع المكونات التي تشير إليها الدالة، فسيتم طباعة المكونات الموجودة في المفتاح فقط. على سبيل المثال، تمرير مفتاح عام إلى EVP_PKEY_print_private() سيؤدي إلى طباعة المكونات العامة فقط.

قيم الإرجاع

جميع هذه الدوال ترجع 1 للنجاح و 0 أو قيمة سالبة للفشل. على وجه الخصوص، تشير قيمة إرجاع -2 إلى أن العملية غير مدعومة بواسطة خوارزمية المفتاح العام.

انظر أيضًا

EVP_PKEY_CTX_new(3), EVP_PKEY_keygen(3)

التاريخ

تمت إضافة الدوال EVP_PKEY_print_public() و EVP_PKEY_print_private() و EVP_PKEY_print_params() في OpenSSL 1.0.0.

تمت إضافة الدوال EVP_PKEY_print_public_fp() و EVP_PKEY_print_private_fp() و EVP_PKEY_print_params_fp() في OpenSSL 3.0.
