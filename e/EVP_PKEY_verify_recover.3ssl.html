EVP_PKEY_verify_recover(3ssl)

الحزمة: openssl

التراخيص: Apache-2.0

EVP_PKEY_VERIFY_RECOVER(3ssl) OpenSSL EVP_PKEY_VERIFY_RECOVER(3ssl)

الاسم

EVP_PKEY_verify_recover_init, EVP_PKEY_verify_recover_init_ex, EVP_PKEY_verify_recover_init_ex2, EVP_PKEY_verify_recover - استعادة التوقيع باستخدام خوارزمية مفتاح عام

الملخص

#include <openssl/evp.h>
int EVP_PKEY_verify_recover_init(EVP_PKEY_CTX *ctx);
int EVP_PKEY_verify_recover_init_ex(EVP_PKEY_CTX *ctx,
                                    const OSSL_PARAM params[]);
int EVP_PKEY_verify_recover_init_ex2(EVP_PKEY_CTX *ctx, EVP_SIGNATURE *algo,
                                     const OSSL_PARAM params[]);
int EVP_PKEY_verify_recover(EVP_PKEY_CTX *ctx,
                            unsigned char *rout, size_t *routlen,
                            const unsigned char *sig, size_t siglen);

الوصف

تقوم EVP_PKEY_verify_recover_init() بتهيئة سياق خوارزمية المفتاح العام ctx للتوقيع باستخدام الخوارزمية المعطاة عند إنشاء السياق باستخدام EVP_PKEY_CTX_new(3) أو متغيراتها. تُستخدم الخوارزمية لجلب طريقة EVP_SIGNATURE ضمنيًا، انظر "السحب الضمني" في provider(7) لمزيد من المعلومات حول السحب الضمني.

EVP_PKEY_verify_recover_init_ex() هي نفس EVP_PKEY_verify_recover_init() ولكنها تقوم أيضًا بتعيين المعلمات params المقدمة في السياق قبل العودة.

EVP_PKEY_verify_recover_init_ex2() هي نفس EVP_PKEY_verify_recover_init_ex()، ولكنها تعمل مع طريقة EVP_SIGNATURE algo تم جلبها صراحةً. لا يمكن استخدام السياق ctx بدون مفتاح محمل مسبقًا مع هذه الوظيفة. اعتمادًا على الخوارزمية التي تم جلبها، قد تكون بعض التفاصيل المتعلقة بمعالجة الإدخال إلى EVP_PKEY_verify() محددة مسبقًا، وفي هذه الحالة، قد لا يمكن عادةً تغيير تلك التفاصيل. انظر "ملاحظات" أدناه للحصول على شرح أعمق.

تقوم الدالة EVP_PKEY_verify_recover() باستعادة البيانات الموقعة باستخدام ctx. يتم تحديد التوقيع باستخدام المعلمات sig و siglen. إذا كانت rout فارغة (NULL)، فسيتم كتابة الحد الأقصى لحجم مخزن الإخراج في المعلمة routlen. إذا لم تكن rout فارغة (NULL)، فيجب أن تحتوي المعلمة routlen قبل الاستدعاء على طول المخزن المؤقت rout، وإذا كان الاستدعاء ناجحًا، يتم كتابة البيانات المستعادة إلى rout وكمية البيانات المكتوبة إلى routlen.

ملاحظات

عادةً ما يكون التطبيق مهتمًا فقط بما إذا كانت عملية التحقق من التوقيع ناجحة، وفي هذه الحالات يجب استخدام الدالة EVP_verify().

ومع ذلك، في بعض الأحيان يكون من المفيد الحصول على البيانات التي تم توقيعها أصلاً باستخدام عملية التوقيع. فقط بعض خوارزميات المفاتيح العامة يمكنها استعادة التوقيع بهذه الطريقة (على سبيل المثال، RSA في وضع الحشو PKCS).

بعد استدعاء EVP_PKEY_verify_recover_init()، يمكن إجراء عمليات تحكم خاصة بالخوارزمية لتعيين أي معلمات مناسبة للعملية.

بعد استدعاء EVP_PKEY_verify_recover_init_ex2()، قد لا تكون عمليات التحكم الخاصة بالخوارزمية مطلوبة إذا كانت الخوارزمية المختارة تعني أن تلك الضوابط محددة مسبقًا (وغير قابلة للتغيير).

يمكن استدعاء الدالة EVP_PKEY_verify_recover() أكثر من مرة على نفس السياق إذا تم إجراء عدة عمليات باستخدام نفس المعلمات.

قيم الإرجاع

EVP_PKEY_verify_recover_init() و EVP_PKEY_verify_recover() ترجع 1 للنجاح و 0 أو قيمة سالبة للفشل. على وجه الخصوص، يشير رمز الإرجاع -2 إلى أن العملية غير مدعومة من قبل خوارزمية المفتاح العام.

أمثلة

استعادة التجزئة التي تم توقيعها أصلاً باستخدام PKCS#1 و SHA256:

#include <openssl/evp.h>
#include <openssl/rsa.h>
EVP_PKEY_CTX *ctx;
unsigned char *rout, *sig;
size_t routlen, siglen;
EVP_PKEY *verify_key;
/*
 * ملاحظة: يفترض أن verify_key و sig و siglen تم إعدادهما بالفعل
 * وأن verify_key هو مفتاح RSA عام
 */
ctx = EVP_PKEY_CTX_new(verify_key, NULL /* لا يوجد محرك */);
if (!ctx)
    /* حدث خطأ */
if (EVP_PKEY_verify_recover_init(ctx) <= 0)
    /* خطأ */
if (EVP_PKEY_CTX_set_rsa_padding(ctx, RSA_PKCS1_PADDING) <= 0)
    /* خطأ */
if (EVP_PKEY_CTX_set_signature_md(ctx, EVP_sha256()) <= 0)
    /* خطأ */
/* تحديد طول المخزن المؤقت */
if (EVP_PKEY_verify_recover(ctx, NULL, &routlen, sig, siglen) <= 0)
    /* خطأ */
rout = OPENSSL_malloc(routlen);
if (!rout)
    /* فشل malloc */
if (EVP_PKEY_verify_recover(ctx, rout, &routlen, sig, siglen) <= 0)
    /* خطأ */
/* البيانات المستعادة هي routlen بايت مكتوبة في المخزن المؤقت rout */
