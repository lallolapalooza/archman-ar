Encode::Encoder(3perl) دليل مبرمجي Perl Encode::Encoder(3perl)

الاسم

Encode::Encoder -- مشفر موجه للكائنات

الملخص

use Encode::Encoder;
# Encode::encode("ISO-8859-1", $data); 
Encode::Encoder->new($data)->iso_8859_1; # طريقة الكائنات
# اختصار
use Encode::Encoder qw(encoder);
encoder($data)->iso_8859_1;
# يمكنك تكديسها!
encoder($data)->iso_8859_1->base64;  # بشرط أن يكون base64() معرفًا
# يمكنك استخدامه كفك تشفير أيضًا
encoder($base64)->bytes('base64')->latin1;
# تحويل إلى سلسلة نصية
print encoder($data)->utf8->latin1;  # يطبع السلسلة النصية بترميز latin1
# تحويل إلى رقم
encoder("\x{abcd}\x{ef}g")->utf8 == 6; # صحيح. bytes::length($data)

ملخص

Encode::Encoder يسمح لك باستخدام Encode بأسلوب موجه للكائنات. هذا ليس فقط أكثر بديهية من النهج الوظيفي، ولكنه أيضًا أكثر فائدة عندما تريد تكديس الترميزات. افترض أنك تريد تحويل سلسلة UTF-8 الخاصة بك إلى Latin1 ثم إلى Base64: يمكنك ببساطة أن تقول

my $base64 = encoder($utf8)->latin1->base64;

بدلاً من

my $latin1 = encode("latin1", $utf8);
my $base64 = encode_base64($utf8);

أو بطريقة أقل إرهاقًا وأكثر تعقيدًا

my $base64 = encode_base64(encode("latin1", $utf8));

الوصف

إليك كيفية استخدام هذه الوحدة.

  • هناك متغيران مثيلان على الأقل مخزنان في مرجع تجزئة، {data} و {encoding}.
  • عندما لا توجد طريقة، تأخذ اسم الطريقة كاسم للترميز وتقوم بتشفير بيانات المثيل بالترميز. إذا نجحت، يتم تعيين ترميز المثيل وفقًا لذلك.
  • يمكنك استرداد النتيجة عبر ->data ولكن عادةً لا تحتاج إلى ذلك لأن عامل التشغيل النصي ("") يتم تجاوزه للقيام بذلك بالضبط.

الطرق المعرفة مسبقًا

هذه الوحدة تعرف الطرق التالية مسبقًا:

$e = Encode::Encoder->new([$data, $encoding]);
يعيد كائن مشفر. يتم تهيئة بياناته بـ $data إذا كانت موجودة، ويتم تعيين ترميزه إلى $encoding إذا كانت موجودة.

عند حذف $encoding، فإنه يأخذ utf8 افتراضيًا إذا كانت $data بالفعل بتنسيق utf8 أو "" (سلسلة فارغة) بخلاف ذلك.

encoder()
هو اسم مستعار لـ Encode::Encoder->new(). هذا يتم تصديره عند الطلب.
$e->data([$data])
عند وجود $data، يتم تعيين بيانات المثيل إلى $data ويعيد الكائن نفسه. بخلاف ذلك، يتم إرجاع بيانات المثيل الحالية.
$e->encoding([$encoding])
عند وجود $encoding، يتم تعيين ترميز المثيل إلى $encoding ويعيد الكائن نفسه. بخلاف ذلك، يتم إرجاع ترميز المثيل الحالي.
$e->bytes([$encoding])
يفك تشفير بيانات المثيل من $encoding، أو ترميز المثيل إذا تم حذفه. إذا كان التحويل ناجحًا، فسيتم تعيين ترميز المثيل إلى "".

تم اختيار الاسم bytes عمدًا لتجنب تلوث مساحة الأسماء - يمكن استخدام هذه الوحدة كفئة أساسية لذلك يتم تجنب أسماء الطرق التي تظهر في Encode::Encoding.

مثال: محول base64

تم تصميم هذه الوحدة للعمل مع Encode::Encoding. لجعل مثال محول Base64 أعلاه يعمل بالفعل، يمكنك كتابة وحدة كهذه:

package Encode::Base64;
use parent 'Encode::Encoding';
__PACKAGE__->Define('base64');
use MIME::Base64;
sub encode{ 
    my ($obj, $data) = @_; 
    return encode_base64($data);
}
sub decode{
    my ($obj, $data) = @_; 
    return decode_base64($data);
}
1;
__END__

وستكون وحدة المستدعي لديك شيئًا كهذا:

use Encode::Encoder;
use Encode::Base64;
# الآن يمكنك القيام بما يلي حقًا
encoder($data)->iso_8859_1->base64;
encoder($base64)->bytes('base64')->latin1;

تحميل العمليات

تقوم هذه الوحدة بتحميل عمليتين، النص ("") والرقم (0+).

يقوم النص بطباعة البيانات داخل الكائن.

يقوم الرقم بإرجاع عدد البايتات في بيانات المثيل.

إنها مفيدة عندما تريد طباعة أو معرفة حجم البيانات.
