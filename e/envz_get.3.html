الملخص

#include <envz.h>
error_t envz_add(char **restrict envz, size_t *restrict envz_len,
               const char *restrict name, const char *restrict value);
char *envz_entry(const char *restrict envz, size_t envz_len,
               const char *restrict name);
char *envz_get(const char *restrict envz, size_t envz_len,
               const char *restrict name);
error_t envz_merge(char **restrict envz, size_t *restrict envz_len,
               const char *restrict envz2, size_t envz2_len,
               int override);
void envz_remove(char **restrict envz, size_t *restrict envz_len,
               const char *restrict name);
void envz_strip(char **restrict envz, size_t *restrict envz_len);

الوصف

هذه الوظائف خاصة بـ glibc.

متجه argz هو مؤشر إلى مخزن مؤقت للحروف مع طول، انظر argz_add(3). متجه envz هو متجه argz خاص، وهو الذي تكون فيه السلاسل بالشكل "name=value". كل شيء بعد أول '=' يعتبر القيمة. إذا لم يكن هناك '=', فسيتم اعتبار القيمة NULL. (بينما القيمة في حالة وجود '=' لاحقة هي السلسلة الفارغة "".)

تستخدم هذه الوظائف للتعامل مع متجهات envz.

envz_add() تضيف السلسلة "name=value" (في حالة value ليست NULL) أو "name" (في حالة value هي NULL) إلى متجه envz (*envz, *envz_len) وتحدّث *envz و *envz_len. إذا كان هناك مدخل بنفس الاسم name موجودًا، فسيتم إزالته.

envz_entry() تبحث عن name في متجه envz (envz, envz_len) وتعيد المدخل إذا تم العثور عليه، أو NULL إذا لم يتم العثور عليه.

envz_get() تبحث عن name في متجه envz (envz, envz_len) وتعيد القيمة إذا تم العثور عليها، أو NULL إذا لم يتم العثور عليها. (لاحظ أن القيمة يمكن أن تكون NULL أيضًا، وذلك عندما يكون هناك مدخل للاسم name بدون علامة '='.)

envz_merge() تضيف كل مدخل في envz2 إلى *envz، كما لو كان باستخدام envz_add(). إذا كان override صحيحًا، فإن القيم الموجودة في envz2 ستحل محل تلك التي لها نفس الاسم في *envz، وإلا فلا.

envz_remove() تزيل المدخل للاسم name من (*envz, *envz_len) إذا كان هناك واحد.

envz_strip() تزيل جميع الإدخالات ذات القيمة NULL.

قيمة الإرجاع

جميع وظائف envz التي تقوم بتخصيص الذاكرة لها نوع إرجاع error_t (نوع صحيح)، وتعيد 0 للنجاح، و ENOMEM في حالة حدوث خطأ في التخصيص.

أمثلة

#include <envz.h>
#include <stdio.h>
#include <stdlib.h>
int
main(int argc, char *argv[], char *envp[])
{
    char    *str;
    size_t  e_len = 0;
    for (size_t i = 0; envp[i] != NULL; i++)
        e_len += strlen(envp[i]) + 1;
    str = envz_entry(*envp, e_len, "HOME");
    printf("%s\n", str);
    str = envz_get(*envp, e_len, "HOME");
    printf("%s\n", str);
    exit(EXIT_SUCCESS);
}
