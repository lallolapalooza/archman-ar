EVP_ASYM_CIPHER_free(3ssl)

الحزمة: openssl

التراخيص: Apache-2.0

EVP_ASYM_CIPHER_FREE(3ssl) OpenSSL EVP_ASYM_CIPHER_FREE(3ssl)

الاسم

EVP_ASYM_CIPHER_fetch, EVP_ASYM_CIPHER_free, EVP_ASYM_CIPHER_up_ref, EVP_ASYM_CIPHER_is_a, EVP_ASYM_CIPHER_get0_provider, EVP_ASYM_CIPHER_do_all_provided, EVP_ASYM_CIPHER_names_do_all, EVP_ASYM_CIPHER_get0_name, EVP_ASYM_CIPHER_get0_description, EVP_ASYM_CIPHER_gettable_ctx_params, EVP_ASYM_CIPHER_settable_ctx_params - وظائف لإدارة كائنات خوارزميات EVP_ASYM_CIPHER

const char *EVP_ASYM_CIPHER_get0_name(const EVP_ASYM_CIPHER *cipher);
int EVP_ASYM_CIPHER_is_a(const EVP_ASYM_CIPHER *cipher, const char *name);
OSSL_PROVIDER *EVP_ASYM_CIPHER_get0_provider(const EVP_ASYM_CIPHER *cipher);
void EVP_ASYM_CIPHER_do_all_provided(OSSL_LIB_CTX *libctx,
                                     void (*fn)(EVP_ASYM_CIPHER *cipher,
                                                void *arg),
                                     void *arg);
int EVP_ASYM_CIPHER_names_do_all(const EVP_ASYM_CIPHER *cipher,
                                 void (*fn)(const char *name, void *data),

const OSSL_PARAM *EVP_ASYM_CIPHER_settable_ctx_params(const EVP_ASYM_CIPHER *cip);

الوصف

EVP_ASYM_CIPHER_fetch() تجلب التنفيذ للخوارزمية algorithm المحددة من أي مزود يقدمها، ضمن المعايير التي تحددها properties وفي نطاق سياق المكتبة المحدد ctx (انظر OSSL_LIB_CTX(3)). ستوفر الخوارزمية وظائف لتنفيذ مهام متعلقة بالتشفير غير المتماثل مثل التشفير وفك التشفير غير المتماثل. انظر "ALGORITHM FETCHING" في crypto(7) لمزيد من المعلومات.

يجب تحرير القيمة المعادة لاحقًا باستخدام EVP_ASYM_CIPHER_free().

EVP_ASYM_CIPHER_free() تقلل من عدد المراجع لهيكل EVP_ASYM_CIPHER. عادةً ما يتم الحصول على هذا الهيكل من استدعاء سابق لـ EVP_ASYM_CIPHER_fetch(). إذا انخفض عدد المراجع إلى 0، يتم تحرير الهيكل. إذا كان الوسيط NULL، فلا يتم فعل شيء.

EVP_ASYM_CIPHER_up_ref() تزيد من عدد المراجع لهيكل EVP_ASYM_CIPHER.

EVP_ASYM_CIPHER_is_a() تعيد 1 إذا كان cipher يمثل تنفيذًا لخوارزمية يمكن التعرف عليها بالاسم name، وإلا فإنها تعيد 0.

EVP_ASYM_CIPHER_get0_provider() تعيد المزود الذي تم جلب cipher منه.

EVP_ASYM_CIPHER_do_all_provided() تجتاز جميع خوارزميات EVP_ASYM_CIPHER التي تنفذها جميع المزودين النشطين في سياق المكتبة المحدد libctx، ولكل تنفيذ، تستدعي الوظيفة المعطاة fn مع طريقة التنفيذ و arg المعطى كوسيط.

EVP_ASYM_CIPHER_get0_name() تعيد اسم الخوارزمية من التنفيذ المقدم للخوارزمية cipher المعطاة. لاحظ أن cipher قد يكون لديها أسماء مرادفة متعددة مرتبطة بها. في هذه الحالة، يتم إرجاع الاسم الأول من تعريف الخوارزمية. يحتفظ كائن cipher بملكية السلسلة المعادة ولا يجب تحريرها بواسطة المتصل.

EVP_ASYM_CIPHER_names_do_all() تجتاز جميع الأسماء الخاصة بـ cipher، وتستدعي fn مع كل اسم و data.

EVP_ASYM_CIPHER_get0_description() تعيد وصفًا للخوارزمية cipher، مخصص للعرض والاستهلاك البشري. الوصف يعود لتقدير تنفيذ cipher.

EVP_ASYM_CIPHER_gettable_ctx_params() و EVP_ASYM_CIPHER_settable_ctx_params() تعيدان مصفوفة ثابتة من OSSL_PARAM(3) تصف أسماء وأنواع معلمات المفتاح التي يمكن استردادها أو تعيينها بواسطة خوارزمية تشفير المفتاح باستخدام EVP_PKEY_CTX_get_params(3) و EVP_PKEY_CTX_set_params(3).

قيم الإرجاع

EVP_ASYM_CIPHER_fetch() تعيد مؤشرًا إلى EVP_ASYM_CIPHER للنجاح أو NULL في حالة الفشل.

EVP_ASYM_CIPHER_up_ref() تعيد 1 للنجاح أو 0 بخلاف ذلك.

EVP_ASYM_CIPHER_names_do_all() تعيد 1 إذا تم استدعاء الدالة المساعدة لجميع الأسماء. قيمة إرجاع 0 تعني أن الدالة المساعدة لم يتم استدعاؤها لأي اسم.

EVP_ASYM_CIPHER_gettable_ctx_params() و EVP_ASYM_CIPHER_settable_ctx_params() تعيدان مصفوفة ثابتة من OSSL_PARAM(3) أو NULL عند حدوث خطأ.

انظر أيضًا

"ALGORITHM FETCHING" في crypto(7)، OSSL_PROVIDER(3)
