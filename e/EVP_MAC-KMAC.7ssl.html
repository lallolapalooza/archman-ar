EVP_MAC-KMAC(7ssl)

الحزمة: openssl

التراخيص: Apache-2.0

EVP_MAC-KMAC(7ssl) OpenSSL EVP_MAC-KMAC(7ssl)

الاسم

EVP_MAC-KMAC, EVP_MAC-KMAC128, EVP_MAC-KMAC256 - تطبيقات KMAC EVP_MAC

الوصف

دعم حساب KMAC MACs من خلال واجهة برمجة تطبيقات EVP_MAC.

الهوية

يتم التعرف على هذه التطبيقات بواحدة من هذه الأسماء والخصائص، لاستخدامها مع EVP_MAC_fetch():

"KMAC-128", "provider=default" أو "provider=fips"
"KMAC-256", "provider=default" أو "provider=fips"

المعلمات المدعومة

يمكن العثور على الوصف العام لهذه المعلمات في "PARAMETERS" في EVP_MAC(3).

يمكن تعيين جميع هذه المعلمات (باستثناء "block-size") باستخدام EVP_MAC_CTX_set_params(). علاوة على ذلك، يمكن استرداد المعلمة "size" باستخدام EVP_MAC_CTX_get_params()، أو باستخدام EVP_MAC_CTX_get_mac_size(). لا ينبغي أن يتجاوز طول المعلمة "size" (بايت) حجم size_t. وبالمثل، يمكن استرداد المعلمة "block-size" باستخدام EVP_MAC_CTX_get_params()، أو باستخدام EVP_MAC_CTX_get_block_size().

"key" (OSSL_MAC_PARAM_KEY) <سلسلة ثمانية>
تعيين مفتاح MAC. تعيين هذه المعلمة يعادل تمرير key إلى EVP_MAC_init(3). يجب أن يكون طول المفتاح (بالبايت) في النطاق 4...512.
"custom" (OSSL_MAC_PARAM_CUSTOM) <سلسلة ثمانية>
تعيين سلسلة التخصيص. وهي قيمة اختيارية بحد أقصى 512 بايت، وهي فارغة افتراضيًا.
"size" (OSSL_MAC_PARAM_SIZE) <عدد صحيح غير مؤشر>
تعيين حجم MAC. افتراضيًا، هو 32 لـ "KMAC-128" و 64 لـ "KMAC-256".
"block-size" (OSSL_MAC_PARAM_BLOCK_SIZE) <عدد صحيح غير مؤشر>
الحصول على حجم كتلة MAC. هو 168 لـ "KMAC-128" و 136 لـ "KMAC-256".
"xof" (OSSL_MAC_PARAM_XOF) <عدد صحيح>
من المتوقع أن تكون قيمة المعلمة "xof" هي 1 أو 0. استخدم 1 لتمكين وضع XOF. القيمة الافتراضية هي 0.
"fips-indicator" (OSSL_MAC_PARAM_FIPS_APPROVED_INDICATOR) <int>
تم وصف هذه المعلمة القابلة للتعيين في provider-mac(7).
"no-short-mac" (OSSL_MAC_PARAM_FIPS_NO_SHORT_MAC) <عدد صحيح>
تم وصف هذه المعلمة القابلة للتعيين في provider-mac(7). تستخدم من قبل موفر FIPS الخاص بـ OpenSSL ويتم تحديد الحد الأدنى لطول الإخراج لـ KMAC بواسطة NIST's SP 800-185 8.4.2.
"key-check" (OSSL_MAC_PARAM_FIPS_KEY_CHECK) <عدد صحيح>
تم وصف هذه المعلمة القابلة للتعيين في provider-mac(7).

يجب تعيين المعلمتين "custom" و "no-short-mac" كجزء من أو قبل استدعاء EVP_MAC_init(). يمكن تعيين المعلمتين "xof" و "size" في أي وقت قبل EVP_MAC_final(). يتم تعيين المعلمة "key" كجزء من استدعاء EVP_MAC_init()، ولكن يمكن تعيينها قبله بدلاً من ذلك.

أمثلة

#include <openssl/evp.h>
#include <openssl/params.h>
static int do_kmac(const unsigned char *in, size_t in_len,
                   const unsigned char *key, size_t key_len,
                   const unsigned char *custom, size_t custom_len,
                   int xof_enabled, unsigned char *out, int out_len)
{
    EVP_MAC_CTX *ctx = NULL;
    EVP_MAC *mac = NULL;
    OSSL_PARAM params[4], *p;
    int ret = 0;
    size_t l = 0;
    mac = EVP_MAC_fetch(NULL, "KMAC-128", NULL);
    if (mac == NULL)
        goto err;
    ctx = EVP_MAC_CTX_new(mac);
    /* يمكن تحرير mac بعد استخدامه بواسطة EVP_MAC_CTX_new */
    EVP_MAC_free(mac);
    if (ctx == NULL)
        goto err;
    /*
     * إعداد المعلمات المطلوبة قبل استدعاء EVP_MAC_init()
     * يمكن أيضًا استخدام المعلمات OSSL_MAC_PARAM_XOF و OSSL_MAC_PARAM_SIZE في هذه النقطة.
     */
    p = params;
    *p++ = OSSL_PARAM_construct_octet_string(OSSL_MAC_PARAM_KEY,
                                             (void *)key, key_len);
    if (custom != NULL && custom_len != 0)
      *p++ = OSSL_PARAM_construct_octet_string(OSSL_MAC_PARAM_CUSTOM,
                                               (void *)custom, custom_len);
    *p = OSSL_PARAM_construct_end();
    if (!EVP_MAC_CTX_set_params(ctx, params))
        goto err;
    if (!EVP_MAC_init(ctx))
        goto err;
    /*
     * ملاحظة: يمكن تعيين المعلمات الاختيارية التالية في أي وقت
     * قبل EVP_MAC_final().
     */
    p = params;
    *p++ = OSSL_PARAM_construct_int(OSSL_MAC_PARAM_XOF, &xof_enabled);
    *p++ = OSSL_PARAM_construct_int(OSSL_MAC_PARAM_SIZE, &out_len);
    *p = OSSL_PARAM_construct_end();
    if (!EVP_MAC_CTX_set_params(ctx, params))
        goto err;
    /* قد يتم استدعاء التحديث عدة مرات هنا لإدخال متدفق */
    if (!EVP_MAC_update(ctx, in, in_len))
        goto err;
    if (!EVP_MAC_final(ctx, out, &l, out_len))
        goto err;
    ret = 1;
err:
    EVP_MAC_CTX_free(ctx);
    return ret;
}
