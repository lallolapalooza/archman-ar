efi_get_variable_attributes(3)

الحزمة: efivar

التراخيص: LGPL-2.1-or-later

EFI_GET_VARIABLE(3) دليل وظائف المكتبة EFI_GET_VARIABLE(3)

الاسم

efi_variables_supported, efi_del_variable, efi_get_variable, efi_get_variable_attributes, efi_get_variable_size, efi_set_variable - التعامل مع متغيرات UEFI

الملخص

#include <efivar.h>

int efi_variables_supported(void);
int efi_del_variable(efi_guid_t guid, const char *name);
int efi_get_variable(efi_guid_t guid, const char *name,
				 void **data, ssize_t *data_size,
				 uint32_t *attributes);
int efi_get_variable_attributes(efi_guid_t guid, const char *name,
						  uint32_t *attributes);
int efi_get_variable_exists(efi_guid_t guid, const char *name);
int efi_get_variable_size(efi_guid_t guid, const char *name,
					 size_t *size);
int efi_append_variable(efi_guid_t guid, const char *name,
				 void *data, size_t data_size,
				 uint32_t attributes);
int efi_set_variable(efi_guid_t guid, const char *name,
				 void *data, size_t data_size,
				 uint32_t attributes, mode_t mode);
int efi_get_next_variable_name(efi_guid_t **guid, char **name);
int efi_str_to_guid(const char *s, efi_guid_t *guid);
int efi_guid_to_str(const efi_guid_t *guid, char **sp);
int efi_name_to_guid(const char *name, efi_guid_t *guid);
int efi_id_guid_to_guid(const char *id_guid, efi_guid_t *guid);
int efi_guid_to_name(efi_guid_t *guid, char **name);
int efi_guid_to_id_guid(efi_guid_t *guid, char **id_guid);
int efi_guid_to_symbol(efi_guid_t *guid, char **symbol);
int efi_symbol_to_guid(const char *symbol, efi_guid_t *guid);

الوصف

efi_variables_supported() تختبر ما إذا كانت خاصية متغيرات UEFI مدعومة على الجهاز الحالي.

efi_del_variable() تحذف المتغير المحدد بواسطة guid و name.

efi_get_variable() تحصل على المتغير المحدد بواسطة guid و name. يتم تخزين القيمة في data، حجمها في data_size، وتخزن سماتها في attributes.

efi_get_variable_attributes() تحصل على سمات المتغير المحدد بواسطة guid و name.

efi_get_variable_exists() تحصل على ما إذا كان المتغير المحدد بواسطة guid و name موجودًا.

efi_get_variable_size() تحصل على حجم البيانات للمتغير المحدد بواسطة guid و name.

efi_append_variable() تلحق data بالحجم size إلى المتغير المحدد بواسطة guid و name.

efi_set_variable() تعين المتغير المحدد بواسطة guid و name، وتعّين وضع الملف إلى mode، وتخضع لـ umask. لاحظ أن الوضع لن يستمر عبر إعادة التشغيل، وأن الأذونات تنطبق فقط إذا كانت على أنظمة تستخدم efivarfs.

efi_get_next_variable_name() تتنقل عبر المتغيرات الموجودة حاليًا، وتعيد guid واسم.

efi_str_to_guid() تقوم بتحليل GUID UEFI من شكل سلسلة نصية إلى efi_guid_t يقدمها المستدعي.

efi_guid_to_str() تنشئ تمثيلًا نصيًا لـ GUID UEFI. إذا كان sp هو NULL، فإنها تعيد حجم السلسلة النصية. إذا كان sp ليس NULL ولكن *sp هو NULL، فإنها تخصص سلسلة نصية وتعيدها. تقع على عاتق المستدعي مسؤولية تحرير هذه السلسلة. إذا كان sp ليس NULL و *sp ليس NULL، فإن efi_guid_to_str() تفترض وجود تخصيص بالحجم المناسب وتستخدمه.

efi_name_to_guid() تترجم من اسم معروف إلى efi_guid_t يقدمها المستدعي.

efi_guid_to_name() تترجم من efi_guid_t إلى اسم معروف. إذا لم يكن للـ GUID المقدم اسم معروف، فإن هذه الوظيفة تعادل efi_guid_to_str().

efi_guid_to_id_guid() تترجم من efi_guid_t إلى {ID GUID}. إذا كان للـ GUID المقدم اسم معروف، فإن {ID GUID} سيكون بالشكل "{name_here}". إذا لم يكن، فسيكون بالشكل "{66b2af1c-6211-4082-95cb-9f73a4795a7e}".

efi_id_guid_to_guid() تترجم من {ID GUID} إلى efi_guid_t يقدمها المستدعي.

efi_guid_to_symbol() تترجم من efi_guid_t إلى اسم رمز C فريد (داخل libefivar). هذه الأسماء الرمزية مفيدة للطباعة كمعرف فريد وسهل التحليل، كما أنها توفرها المكتبة وملفاتها الرأسية.

efi_symbol_to_guid() تترجم من اسم رمز libefivar efi_guid_$FOO إلى efi_guid_t يقدمها المستدعي.

قيمة الإرجاع

efi_variables_supported() تعيد true إذا كانت المتغيرات مدعومة على العتاد قيد التشغيل، و false إذا لم تكن كذلك.

efi_get_next_variable_name() تعيد 0 عند اكتمال التكرار، 1 عند عدم اكتمال التكرار، و -1 في حالة الخطأ. في حالة الخطأ، يتم تعيين errno(3) بشكل مناسب.

efi_del_variable(), efi_get_variable(), efi_get_variable_attributes(), efi_get_variable_exists(), efi_get_variable_size(), efi_append_variable(), efi_set_variable(), efi_str_to_guid(), efi_guid_to_str(), efi_name_to_guid(), و efi_guid_to_name() تعيد قيمة سالبة عند الخطأ وصفر عند النجاح.
