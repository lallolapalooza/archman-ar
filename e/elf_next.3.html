elf_next(3)

الحزمة: libelf

التراخيص: GPL-2.0-or-later OR LGPL-3.0-or-later

ELF_NEXT(3) دليل مبرمج Libelf ELF_NEXT(3)

الاسم

elf_next - تقدم واصف ELF إلى العضو التالي في الأرشيف

الملخص

#include <libelf.h>
Elf_Cmd elf_next(Elf *elf);

الوصف

تقدم واصف ELF المرتبط بملف أرشيف إلى عضو الأرشيف التالي المتاح.

يمكن استخدام واصفات ELF التي تم تهيئتها من ملف أرشيف لاسترداد واصفات ELF لأعضاء الأرشيف واحدًا تلو الآخر باستخدام elf_begin(3). تقوم elf_next(3) بتحديث واصف الأرشيف بحيث تُرجع elf_begin(3) واصف ELF للعضو التالي في الأرشيف. انظر قسم أمثلة أدناه.

قيمة الإرجاع

إذا كان elf يشير إلى عضو في الأرشيف، فقم بتحديث حالة واصف ELF للأرشيف الأب المرتبط بـ elf بحيث يمكن استرداد عضو الأرشيف التالي باستخدام elf_begin(3). قم بإرجاع Elf_Cmd الذي تم استخدامه مع elf_begin(3) لتهيئة elf.

إذا لم يتم تهيئة elf من ملف أرشيف أو لم تعد هناك أعضاء أرشيف أخرى، فإن elf_next(3) تُرجع ELF_C_NULL.

أمثلة

  /* افتح الأرشيف.  */
  fd = open (archive_name, O_RDONLY);
  if (fd == -1)
    {
      printf ("لا يمكن فتح ملف الأرشيف `%s'", fname);
      exit (1);
    }
  /* اضبط إصدار ELF.  */
  elf_version (EV_CURRENT);
  /* أنشئ واصف ELF للأرشيف.  */
  cmd = ELF_C_READ;
  elf = elf_begin (fd, cmd, NULL);
  if (elf == NULL)
    {
      printf ("لا يمكن إنشاء واصف ELF: %s\n", elf_errmsg (-1));
      exit (1);
    }
  /* تحقق من أن هذا واصف للأرشيف.  */
  if (elf_kind (elf) != ELF_K_AR)
    {
      printf ("`%s' ليس أرشيفًا\n", fname);
      exit (1);
    }
  /* احصل على أعضاء الأرشيف واحدًا تلو الآخر.  */
  while ((subelf = elf_begin (fd, cmd, elf)) != NULL)
    {
      /* عالج subelf هنا */
      [...]
      /* تقوم elf_next بتحديث elf، الأرشيف الأب، بحيث تُرجع الاستدعاء التالي
         لـ elf_begin عضو الأرشيف التالي.  */
      cmd = elf_next (subelf);
      if (elf_end (subelf) != 0)
        {
          printf ("خطأ أثناء تحرير واصف ELF الفرعي: %s\n",
                  elf_errmsg (-1));
          exit (1);
        }
    }
  elf_end (elf);
  close (fd);
