errc(3bsd)

الحزمة: libbsd

التراخيص: custom

errc(3bsd) 3bsd errc(3bsd)

الاسم

errc، verrc، warnc، vwarnc — رسائل خطأ مُنسقة

المكتبة

المكتبة "libbsd"

الملخص

#include <err.h> (راجع libbsd(7) لاستخدام التضمين.)
void
errc(int status, int code, const char *fmt, ...);

void
verrc(int status, int code, const char *fmt, va_list args);

void
warnc(int code, const char *fmt, ...);

void
vwarnc(int code, const char *fmt, va_list args);

الوصف

عائلة الدوال err() و warn() تعرض رسالة خطأ مُنسقة على مجرى الخطأ القياسي. في جميع الحالات، يتم إخراج المكون الأخير لاسم البرنامج، متبوعًا بنقطتين رأسيتين (':') ومسافة. النص الذي يلي يعتمد على الدالة التي يتم استدعاؤها. قد يكون مواصفات fmt (والوسائط المرتبطة بها) أي تنسيق مسموح به بواسطة printf(3) أو NULL. إذا لم تكن الوسيطة fmt NULL، فسيتم إخراج رسالة الخطأ المنسقة.

تقوم جميع الدوال بإخراج سلسلة رسالة خطأ مرتبطة بقيمة خطأ (راجع strerror(3))، مسبوقة بنقطتين رأسيتين ومسافة إذا لم تكن fmt NULL. أي، يكون الإخراج كما يلي:

اسم_البرنامج: fmt: سلسلة رسالة الخطأ

إذا لم تكن fmt NULL، أو:

اسم_البرنامج: سلسلة رسالة الخطأ

إذا كانت كذلك.

الوسيطة code تُستخدم كقيمة خطأ بدلاً من القيمة الحالية للمتغير العام errno.

في جميع الحالات، يتبع الإخراج سطر جديد.

الدوال errc() و verrc() لا تعود، بل تخرج بقيمة الوسيطة status.

أمثلة

عرض سلسلة معلومات errno الحالية والخروج:

if ((p = malloc(size)) == NULL)
	err(1, NULL);
if ((fd = open(file_name, O_RDONLY, 0)) == -1)
	err(1, "%s", file_name);

عرض رسالة خطأ والخروج:

if (tm.tm_hour < START_TIME)
	errx(1, "too early, wait until %s", start_time_string);

التحذير من خطأ:

if ((fd = open(raw_device, O_RDONLY, 0)) == -1)
	warnx("%s: %s: trying the block device",
	    raw_device, strerror(errno));
if ((fd = open(block_device, O_RDONLY, 0)) == -1)
	err(1, "%s", block_device);

راجع أيضًا

err(3) exit(3)، perror(3)، printf(3)، strerror(3)

التاريخ

ظهرت الدوال errc()، verrc()، warnc()، و vwarnc() لأول مرة في FreeBSD 3.0، NetBSD 7.0، و OpenBSD 5.6.

تحذيرات

من المهم أبدًا عدم تمرير سلسلة تحتوي على بيانات مقدمة من المستخدم كتنسيق بدون استخدام ‘%s’. يمكن للمهاجم وضع محددات تنسيق في السلسلة لتشويه المكدس، مما يؤدي إلى ثغرة أمنية محتملة. ينطبق هذا حتى لو تم بناء السلسلة "يدويًا" باستخدام دالة مثل snprintf()، حيث قد تحتوي السلسلة الناتجة على محددات تحويل مقدمة من المستخدم ليتم تفسيرها لاحقًا بواسطة عائلة دوال err() و warn().

تأكد دائمًا من استخدام الصيغة الآمنة الصحيحة:

errc(1, 0, "%s", string);
