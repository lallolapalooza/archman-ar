Encode::Guess(3perl) دليل مبرمجي بيرل Encode::Guess(3perl)

الاسم

Encode::Guess -- تخمين الترميز من البيانات

الملخص

# إذا كنت متأكدًا أن $data لن تحتوي على أي شيء زائف
use Encode;
use Encode::Guess qw/euc-jp shiftjis 7bit-jis/;
my $utf8 = decode("Guess", $data);
my $data = encode("Guess", $utf8);   # هذا لا يعمل!
# طريقة أكثر تفصيلاً
use Encode::Guess;
my $enc = guess_encoding($data, qw/euc-jp shiftjis 7bit-jis/);
ref($enc) or die "لا يمكن التخمين: $enc"; # التعامل مع الخطأ بهذه الطريقة
$utf8 = $enc->decode($data);
# أو
$utf8 = decode($enc->name, $data)

مُلخص

تتيح لك Encode::Guess تخمين الترميز الذي تم ترميز البيانات به، أو على الأقل محاولة ذلك.

الوصف

بشكل افتراضي، يتحقق فقط من ascii و utf8 و UTF-16/32 مع BOM.

use Encode::Guess; # ascii/utf8/BOMed UTF

للاستخدام العملي أكثر، يجب عليك تقديم أسماء الترميزات للتحقق منها (المشتبه بهم كما يلي). يمكن أن يكون اسم المشتبه بهم إما أسماء قياسية أو أسماء مستعارة.

تحذير: بخلاف UTF-(16|32)، لا يتم إزالة BOM في utf8 تلقائيًا.

# يحاول جميع ترميزات اليابانية الرئيسية أيضًا
 use Encode::Guess qw/euc-jp shiftjis 7bit-jis/;

إذا تم تعيين المتغير $Encode::Guess::NoUTFAutoGuess إلى قيمة صحيحة، فلن يتم تطبيق أي استدلالات على UTF8/16/32، وسيقتصر الناتج على المشتبه بهم و "ascii".

Encode::Guess->set_suspects
يمكنك أيضًا تغيير قائمة المشتبه بهم الداخلية عبر طريقة "set_suspects". 
use Encode::Guess;
Encode::Guess->set_suspects(qw/euc-jp shiftjis 7bit-jis/);
Encode::Guess->add_suspects
أو يمكنك استخدام طريقة "add_suspects". الفرق هو أن "set_suspects" تمسح قائمة المشتبه بهم الحالية بينما تضيف "add_suspects". 
use Encode::Guess;
Encode::Guess->add_suspects(qw/euc-jp shiftjis 7bit-jis/);
# الآن المشتبه بهم هم euc-jp، shiftjis، 7bit-jis، و
# euc-kr، euc-cn، و big5-eten
Encode::Guess->add_suspects(qw/euc-kr euc-cn big5-eten/);
Encode::decode("Guess" ...)
عندما تكون راضيًا عن قائمة المشتبه بهم، يمكنك الآن 
my $utf8 = Encode::decode("Guess", $data);
Encode::Guess->guess($data)
لكنها ستتعطل إذا:
  • بقي اثنان أو أكثر من المشتبه بهم
  • لم يبق مشتبه بهم

لذلك يجب أن تجرب هذا بدلاً من ذلك:

my $decoder = Encode::Guess->guess($data);

عند النجاح، $decoder هو كائن موثق في Encode::Encoding. لذلك يمكنك الآن القيام بهذا:

my $utf8 = $decoder->decode($data);

عند الفشل، يحتوي $decoder الآن على رسالة خطأ، لذا سيكون كل شيء على النحو التالي:

my $decoder = Encode::Guess->guess($data);
die $decoder unless ref($decoder);
my $utf8 = $decoder->decode($data);
guess_encoding($data, [, قائمة المشتبه بهم])
يمكنك أيضًا تجربة وظيفة "guess_encoding" المصدرة افتراضيًا. تأخذ $data للتحقق منها وتأخذ أيضًا قائمة المشتبه بهم اختياريًا. قائمة المشتبه بهم الاختيارية * لا تنعكس * في قائمة المشتبه بهم الداخلية. 
my $decoder = guess_encoding($data, qw/euc-jp euc-kr euc-cn/);
die $decoder unless ref($decoder);
my $utf8 = $decoder->decode($data);
# تحقق فقط من ascii، utf8 و UTF-(16|32) مع BOM
my $decoder = guess_encoding($data);

محاذير

  • بسبب الخوارزمية المستخدمة، فإن سلاسل ISO-8859 والترميزات الأخرى ذات البايت الواحد لا تعمل جيدًا إلا إذا كان أحد ترميزات ISO-8859 هو المشتبه به الوحيد (إلى جانب ascii و utf8). 
    use Encode::Guess;
# ربما يكون مقبولاً
my $decoder = guess_encoding($data, 'latin1');
# بالتأكيد غير مقبول
my $decoder = guess_encoding($data, qw/latin1 greek/);

    السبب هو أن Encode::Guess تخمن الترميز عن طريق التجربة والخطأ. يقوم أولاً بتقسيم $data إلى أسطر ويحاول فك ترميز السطر لكل مشتبه به. يستمر في العمل حتى يتم استبعاد جميع الترميزات باستثناء واحدة من قائمة المشتبه بهم. سلاسل ISO-8859 ناجحة جدًا في معظم الحالات (لأنها تملأ تقريبًا جميع نقاط الترميز في \x00-\xff).

  • لا تخلط بين الترميزات القياسية الوطنية وترميزات البائع المقابلة. 
    # فكرة سيئة للغاية
my $decoder
   = guess_encoding($data, qw/shiftjis MacJapanese cp932/);

    السبب هو أن ترميز البائع هو عادة مجموعة شاملة للغة القياسية، لذا يصبح غامضًا جدًا في معظم الحالات.

  • من ناحية أخرى، فإن خلط ترميزات اللغات الوطنية المختلفة يعمل تلقائيًا ما لم تكن $data قصيرة جدًا للسماح بالتخمين. 
    # هذا مقبول إذا كانت $data طويلة بما فيه الكفاية
my $decoder =  
 guess_encoding($data, qw/euc-cn
                          euc-jp shiftjis 7bit-jis
                          euc-kr
                          big5-eten/);
  • لا تضع الكثير من المشتبه بهم! لا تجرب شيئًا كهذا! 
    my $decoder = guess_encoding($data, 
                             Encode->encodings(":all"));

في النهاية، هو مجرد تخمين. يجب أن تكون دائمًا صريحًا عندما يتعلق الأمر بالترميزات. ولكن هناك بيئات، خاصة اليابانية، حيث يعد التخمين ضروريًا. استخدم هذه الوحدة بحذر.

قيد الإنجاز

Encode::Guess لا يعمل على منصات EBCDIC.
