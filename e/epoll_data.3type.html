epoll_event(3type) epoll_event(3type)

الاسم

epoll_event, epoll_data, epoll_data_t - حدث epoll

المكتبة

مكتبة C القياسية (libc)

الملخص

#include <sys/epoll.h>
struct epoll_event {
    uint32_t      events;  /* أحداث Epoll */
    epoll_data_t  data;    /* متغير بيانات المستخدم */
};
union epoll_data {
    void     *ptr;
    int       fd;
    uint32_t  u32;
    uint64_t  u64;
};
typedef union epoll_data  epoll_data_t;

الوصف

يقوم الهيكل epoll_event بتحديد البيانات التي يجب على النواة حفظها وإرجاعها عندما يصبح واصف الملف المقابل جاهزًا.

الإصدارات

اختلافات مكتبة C/النواة

توفر رؤوس نواة Linux أيضًا هذا النوع، بتعريف مختلف قليلاً:


#include <linux/eventpoll.h>
struct epoll_event {
    __poll_t  events;
    __u64     data;
};
