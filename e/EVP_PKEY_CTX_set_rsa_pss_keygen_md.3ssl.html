EVP_PKEY_CTX_set_rsa_pss_keygen_md(3ssl)

الحزمة: openssl

التراخيص: Apache-2.0

EVP_PKEY_CTX_SET_RSA_PSS_KEYGEN_MD(3ssl) OpenSSL EVP_PKEY_CTX_SET_RSA_PSS_KEYGEN_MD(3ssl)

الاسم

EVP_PKEY_CTX_set_rsa_pss_keygen_md, EVP_PKEY_CTX_set_rsa_pss_keygen_md_name, EVP_PKEY_CTX_set_rsa_pss_keygen_mgf1_md, EVP_PKEY_CTX_set_rsa_pss_keygen_mgf1_md_name, EVP_PKEY_CTX_set_rsa_pss_keygen_saltlen - دوال دعم خوارزمية EVP_PKEY RSA-PSS

ملخص

#include <openssl/rsa.h>
int EVP_PKEY_CTX_set_rsa_pss_keygen_md(EVP_PKEY_CTX *pctx,
                                       const EVP_MD *md);
int EVP_PKEY_CTX_set_rsa_pss_keygen_md_name(EVP_PKEY_CTX *ctx,
                                            const char *mdname,
                                            const char *mdprops);
int EVP_PKEY_CTX_set_rsa_pss_keygen_mgf1_md(EVP_PKEY_CTX *pctx,
                                            const EVP_MD *md);
int EVP_PKEY_CTX_set_rsa_pss_keygen_mgf1_md_name(EVP_PKEY_CTX *pctx,
                                                 const char *mdname);
int EVP_PKEY_CTX_set_rsa_pss_keygen_saltlen(EVP_PKEY_CTX *pctx,
                                            int saltlen);

الوصف

هذه هي الدوال التي تطبق RSA-PSS(7).

التوقيع والتحقق

الماكرو EVP_PKEY_CTX_set_rsa_padding() مدعوم ولكن يتم إرجاع خطأ إذا تم محاولة تعيين وضع الحشو إلى أي شيء آخر غير PSS. بخلاف ذلك، يكون مشابهًا لإصدار RSA.

يتم استخدام الماكرو EVP_PKEY_CTX_set_rsa_pss_saltlen() لتعيين طول الملح. إذا كان المفتاح لديه قيود استخدام، يتم إرجاع خطأ إذا تم محاولة تعيين طول الملح إلى أقل من الحد الأدنى للقيمة. بخلاف ذلك، يكون مشابهًا لعملية RSA باستثناء اكتشاف طول الملح (باستخدام RSA_PSS_SALTLEN_AUTO) غير مدعوم للتحقق إذا كان المفتاح لديه قيود استخدام.

تُستخدم الدالتان EVP_PKEY_CTX_set_signature_md(3) و EVP_PKEY_CTX_set_rsa_mgf1_md(3) لتعيين خوارزميات التجزئة و MGF1 على التوالي. إذا كان المفتاح لديه قيود استخدام، يتم إرجاع خطأ إذا تم محاولة تعيين التجزئة إلى أي شيء آخر غير القيمة المقيدة. بخلاف ذلك، تكون هذه مشابهة لإصدارات RSA.

توليد المفاتيح

كما هو الحال مع توليد مفاتيح RSA، فإن الماكروين EVP_PKEY_CTX_set_rsa_keygen_bits() و EVP_PKEY_CTX_set_rsa_keygen_pubexp() مدعومان لـ RSA-PSS: لهما نفس المعنى تمامًا كما هو الحال بالنسبة لخوارزمية RSA.

يمكن تحديد قيود اختيارية للمعلمات عند توليد مفتاح PSS. إذا تم تعيين أي قيود (باستخدام الماكروات الموضحة أدناه) فإن جميع المعلمات تكون مقيدة. على سبيل المثال، يؤدي تعيين حد أدنى لطول الملح أيضًا إلى تقييد خوارزميات التجزئة و MGF1. إذا كانت هناك أي قيود مطبقة، فإنها تنعكس في المعلمات المقابلة للمفتاح العام عند (على سبيل المثال) توقيع طلب شهادة.

EVP_PKEY_CTX_set_rsa_pss_keygen_md() تقيد خوارزمية التجزئة التي يمكن للمفتاح الذي تم إنشاؤه استخدامها إلى md. EVP_PKEY_CTX_set_rsa_pss_keygen_md_name() تقوم بنفس الشيء، ولكنها تمرر الخوارزمية بالاسم بدلاً من EVP_MD.

EVP_PKEY_CTX_set_rsa_pss_keygen_mgf1_md() تقيد خوارزمية MGF1 التي يمكن للمفتاح الذي تم إنشاؤه استخدامها إلى md. EVP_PKEY_CTX_set_rsa_pss_keygen_mgf1_md_name() تقوم بنفس الشيء، ولكنها تمرر الخوارزمية بالاسم بدلاً من EVP_MD.

EVP_PKEY_CTX_set_rsa_pss_keygen_saltlen() تقيد الحد الأدنى لطول الملح إلى saltlen.

ملاحظات

يمكن الحصول على سياق لخوارزمية RSA-PSS عن طريق استدعاء:

EVP_PKEY_CTX *pctx = EVP_PKEY_CTX_new_id(EVP_PKEY_RSA_PSS, NULL);

قيم الإرجاع

جميع هذه الدوال ترجع 1 للنجاح و 0 أو قيمة سالبة للفشل. على وجه الخصوص، يشير معدل الإرجاع -2 إلى أن العملية غير مدعومة من قبل خوارزمية المفتاح العام.
