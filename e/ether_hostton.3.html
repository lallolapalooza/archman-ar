ether_aton(3) دليل وظائف المكتبة ether_aton(3)

الاسم

ether_aton, ether_ntoa, ether_ntohost, ether_hostton, ether_line, ether_ntoa_r, ether_aton_r - إجراءات معالجة عناوين الإيثرنت

المكتبة

مكتبة C القياسية (libc, -lc)

ملخص

#include <netinet/ether.h>
char *ether_ntoa(const struct ether_addr *addr);
struct ether_addr *ether_aton(const char *asc);
int ether_ntohost(char *hostname, const struct ether_addr *addr);
int ether_hostton(const char *hostname, struct ether_addr *addr);
int ether_line(const char *line, struct ether_addr *addr,
               char *hostname);
/* ملحقات GNU */
char *ether_ntoa_r(const struct ether_addr *addr, char *buf);
struct ether_addr *ether_aton_r(const char *asc,
                                struct ether_addr *addr);

الوصف

تقوم ether_aton() بتحويل عنوان مضيف الإيثرنت ذي الـ 48 بت asc من صيغة الأرقام الست عشرية والنقاط القياسية إلى بيانات ثنائية بترتيب بايتات الشبكة وتعيد مؤشراً إليها في مخزن مؤقت مخصص ثابتًا، والذي ستقوم الاستدعاءات اللاحقة بالكتابة فوقه. تعيد ether_aton() قيمة NULL إذا كان العنوان غير صالح.

تقوم وظيفة ether_ntoa() بتحويل عنوان مضيف الإيثرنت addr المعطى بترتيب بايتات الشبكة إلى سلسلة نصية بصيغة الأرقام الست عشرية والنقاط القياسية، مع حذف الأصفار البادئة. يتم إعادة السلسلة النصية في مخزن مؤقت مخصص ثابتًا، والذي ستقوم الاستدعاءات اللاحقة بالكتابة فوقه.

تقوم وظيفة ether_ntohost() بربط عنوان إيثرنت باسم المضيف المقابل في /etc/ethers وتعيد قيمة غير صفرية إذا لم يتم العثور عليه.

تقوم وظيفة ether_hostton() بربط اسم مضيف باسم مضيف الإيثرنت المقابل في /etc/ethers وتعيد قيمة غير صفرية إذا لم يتم العثور عليه.

تقوم وظيفة ether_line() بتحليل سطر بصيغة /etc/ethers (عنوان إيثرنت متبوعاً بمسافات بيضاء متبوعة باسم مضيف؛ '#' يقدم تعليقًا) وتعيد زوجًا من العنوان واسم المضيف، أو قيمة غير صفرية إذا لم يتمكن من التحليل. يجب أن يكون المخزن المؤقت الذي يشير إليه hostname طويلًا بما فيه الكفاية - على سبيل المثال، يجب أن يكون له نفس طول line.

الوظائف ether_ntoa_r() و ether_aton_r() هي إصدارات آمنة قابلة لإعادة الاستخدام للوظائف ether_ntoa() و ether_aton() على التوالي، ولا تستخدم مخازن مؤقتة ثابتة.

يتم تعريف البنية ether_addr في <net/ethernet.h> كالتالي:


struct ether_addr {
    uint8_t ether_addr_octet[6];
}
