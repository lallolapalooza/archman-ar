ECDSA_SIG_new(3ssl)

الحزمة: openssl

التراخيص: Apache-2.0

ECDSA_SIG_NEW(3ssl) OpenSSL ECDSA_SIG_NEW(3ssl)

الاسم

ECDSA_SIG_new, ECDSA_SIG_free, ECDSA_SIG_get0, ECDSA_SIG_get0_r, ECDSA_SIG_get0_s, ECDSA_SIG_set0 - وظائف لإنشاء وتدمير ومعالجة كائنات ECDSA_SIG

الملخص

#include <openssl/ecdsa.h>
ECDSA_SIG *ECDSA_SIG_new(void);
void ECDSA_SIG_free(ECDSA_SIG *sig);
void ECDSA_SIG_get0(const ECDSA_SIG *sig, const BIGNUM **pr, const BIGNUM **ps);
const BIGNUM *ECDSA_SIG_get0_r(const ECDSA_SIG *sig);
const BIGNUM *ECDSA_SIG_get0_s(const ECDSA_SIG *sig);
int ECDSA_SIG_set0(ECDSA_SIG *sig, BIGNUM *r, BIGNUM *s);

الوصف

ECDSA_SIG هو هيكل غامض يتكون من عددين BIGNUMs لقيم r و s لتوقيع خوارزمية التوقيع الرقمي المنحنى الإهليلجي (ECDSA) (انظر FIPS186-4 أو X9.62). تم استخدام كائن ECDSA_SIG بشكل أساسي من قبل الدوال المنخفضة المستوى التي تم إهمالها والموصوفة في ECDSA_sign(3)، ولا يزال مطلوبًا لكي تتمكن من تعيين أو الحصول على قيم r و s من توقيع أو إليه. يستخدم هذا بشكل أساسي لأغراض الاختبار كما هو موضح في "أمثلة".

ECDSA_SIG_new() تقوم بتخصيص هيكل ECDSA_SIG فارغ. ملاحظة: قبل OpenSSL 1.1.0، تم تهيئة المكونات r و s.

ECDSA_SIG_free() تقوم بتحرير هيكل ECDSA_SIG sig. إذا كان الوسيط NULL، فلن يتم فعل أي شيء.

ECDSA_SIG_get0() تُرجع مؤشرات داخلية لقيم r و s الموجودة في sig وتخزنها في *pr و *ps على التوالي. يمكن أن يكون المؤشر pr أو ps NULL، وفي هذه الحالة لا يتم إرجاع القيمة المقابلة.

يمكن أيضًا استرداد القيم r و s بشكل منفصل بواسطة الدالتين المقابلتين ECDSA_SIG_get0_r() و ECDSA_SIG_get0_s() على التوالي.

يمكن تعيين قيم r و s غير NULL على sig عن طريق استدعاء ECDSA_SIG_set0(). ينقل استدعاء هذه الدالة إدارة الذاكرة للقيم إلى كائن ECDSA_SIG، وبالتالي يجب ألا يقوم المتصل بتحرير القيم التي تم تمريرها.

انظر i2d_ECDSA_SIG(3) و d2i_ECDSA_SIG(3) للحصول على معلومات حول ترميز وفك ترميز توقيعات ECDSA من/إلى DER.

قيم الإرجاع

ECDSA_SIG_new() تُرجع NULL إذا فشل التخصيص.

ECDSA_SIG_set0() تُرجع 1 عند النجاح أو 0 عند الفشل.

ECDSA_SIG_get0_r() و ECDSA_SIG_get0_s() تُرجعان القيمة المقابلة، أو NULL إذا لم يتم تعيينها.

أمثلة

استخراج قيم r و s للتوقيع من signature ECDSA بحجم signaturelen:

ECDSA_SIG *obj;
const BIGNUM *r, *s;
/* تحميل توقيع في كائن ECDSA_SIG */
obj = d2i_ECDSA_SIG(NULL, &signature, signaturelen);
if (obj == NULL)
    /* خطأ */
r = ECDSA_SIG_get0_r(obj);
s = ECDSA_SIG_get0_s(obj);
if (r == NULL || s == NULL)
    /* خطأ */
/* استخدم BN_bn2binpad() هنا للتحويل إلى مصفوفات بايت r و s */
/*
 * لا تحاول الوصول إلى I<r> أو I<s> بعد استدعاء ECDSA_SIG_free()،
 * حيث سيتم تحريرهما كلاهما بهذا الاستدعاء.
 */
ECDSA_SIG_free(obj);

تحويل مصفوفات بايت r و s إلى signature ECDSA بحجم signaturelen:

ECDSA_SIG *obj = NULL;
unsigned char *signature = NULL;
size_t signaturelen;
BIGNUM *rbn = NULL, *sbn = NULL;
obj = ECDSA_SIG_new();
if (obj == NULL)
    /* خطأ */
rbn = BN_bin2bn(r, rlen, NULL);
sbn = BN_bin2bn(s, slen, NULL);
if (rbn == NULL || sbn == NULL)
    /* خطأ */
if (!ECDSA_SIG_set0(obj, rbn, sbn))
    /* خطأ */
/* قم بتعيين هذه إلى NULL لأنها مملوكة الآن بواسطة obj */
rbn = sbn = NULL;
signaturelen = i2d_ECDSA_SIG(obj, &signature);
if (signaturelen <= 0)
    /* خطأ */
/*
 * يمكن الآن تمرير هذا التوقيع إلى L<EVP_DigestVerify(3)>
 * أو L<EVP_DigestVerifyFinal(3)>
 */
BN_free(rbn);
BN_free(sbn);
OPENSSL_free(signature);
ECDSA_SIG_free(obj);
