ether_aton(3) دليل وظائف المكتبة ether_aton(3)

الاسم

ether_aton, ether_ntoa, ether_ntohost, ether_hostton, ether_line, ether_ntoa_r, ether_aton_r - إجراءات معالجة عناوين Ethernet

المكتبة

مكتبة C القياسية (libc، -lc)

الملخص

#include <netinet/ether.h>
char *ether_ntoa(const struct ether_addr *addr);
struct ether_addr *ether_aton(const char *asc);
int ether_ntohost(char *hostname, const struct ether_addr *addr);
int ether_hostton(const char *hostname, struct ether_addr *addr);
int ether_line(const char *line, struct ether_addr *addr,
               char *hostname);
/* ملحقات GNU */
char *ether_ntoa_r(const struct ether_addr *addr, char *buf);
struct ether_addr *ether_aton_r(const char *asc,
                                struct ether_addr *addr);

الوصف

تقوم الدالة ether_aton() بتحويل عنوان Ethernet المكون من 48 بت asc من التدوين القياسي للأرقام الست عشرية والنقاط إلى بيانات ثنائية بترتيب بايت الشبكة وإرجاع مؤشر إليه في مخزن مؤقت مخصص ثابت، والذي ستقوم الاستدعاءات اللاحقة بالكتابة فوقه. ترجع الدالة ether_aton() قيمة NULL إذا كان العنوان غير صالح.

تقوم الدالة ether_ntoa() بتحويل عنوان Ethernet addr المعطى بترتيب بايت الشبكة إلى سلسلة بتدوين قياسي للأرقام الست عشرية والنقاط، مع حذف الأصفار البادئة. يتم إرجاع السلسلة في مخزن مؤقت مخصص ثابت، والذي ستقوم الاستدعاءات اللاحقة بالكتابة فوقه.

تقوم الدالة ether_ntohost() بربط عنوان Ethernet باسم المضيف المقابل في /etc/ethers وترجع قيمة غير صفرية إذا تعذر العثور عليه.

تقوم الدالة ether_hostton() بربط اسم المضيف بعنوان Ethernet المقابل في /etc/ethers وترجع قيمة غير صفرية إذا تعذر العثور عليه.

تقوم الدالة ether_line() بتحليل سطر بتنسيق /etc/ethers (عنوان Ethernet متبوعًا بمسافة بيضاء ثم اسم مضيف؛ '#' يبدأ تعليقًا) وترجع زوجًا من العنوان واسم المضيف، أو قيمة غير صفرية إذا تعذر تحليله. يجب أن يكون المخزن المؤقت الذي يشير إليه hostname طويلاً بما يكفي — على سبيل المثال، أن يكون له نفس طول line.

الدالتان ether_ntoa_r() و ether_aton_r() هما إصدارات قابلة لإعادة الدخول وآمنة للخيوط للدالتين ether_ntoa() و ether_aton() على التوالي، ولا تستخدمان مخازن مؤقتة ثابتة.

يتم تعريف البنية ether_addr في <net/ethernet.h> على النحو التالي:


struct ether_addr {
    uint8_t ether_addr_octet[6];
}
