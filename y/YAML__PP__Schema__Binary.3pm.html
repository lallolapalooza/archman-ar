YAML__PP__Schema__Binary(3pm)

الحزمة: perl-yaml-pp

التراخيص: PerlArtistic, GPL

YAML::PP::Schema::Binary(3) توثيق Perl المساهم به من قبل المستخدم YAML::PP::Schema::Binary(3)

الاسم

YAML::PP::Schema::Binary - مخطط لتحميل البيانات الثنائية

ملخص

use YAML::PP;
my $yp = YAML::PP->new( schema => [qw/ + Binary /] );
# أو
my ($binary, $same_binary) = $yp->load_string(<<'EOM');
--- !!binary "\
  R0lGODlhDAAMAIQAAP//9/X17unp5WZmZgAAAOfn515eXvPz7Y6OjuDg4J+fn5\
  OTk6enp56enmlpaWNjY6Ojo4SEhP/++f/++f/++f/++f/++f/++f/++f/++f/+\
  +f/++f/++f/++f/++f/++SH+Dk1hZGUgd2l0aCBHSU1QACwAAAAADAAMAAAFLC\
  AgjoEwnuNAFOhpEMTRiggcz4BNJHrv/zCFcLiwMWYNG84BwwEeECcgggoBADs="
--- !!binary |
  R0lGODlhDAAMAIQAAP//9/X17unp5WZmZgAAAOfn515eXvPz7Y6OjuDg4J+fn5
  OTk6enp56enmlpaWNjY6Ojo4SEhP/++f/++f/++f/++f/++f/++f/++f/++f/+
  +f/++f/++f/++f/++f/++SH+Dk1hZGUgd2l0aCBHSU1QACwAAAAADAAMAAAFLC
  AgjoEwnuNAFOhpEMTRiggcz4BNJHrv/zCFcLiwMWYNG84BwwEeECcgggoBADs=
# القيمة الثنائية أعلاه هي سهم صغير مشفر كصورة gif.
EOM

الوصف

انظر https://yaml.org/type/binary.html

عن طريق إضافة بادئة لسلسلة ثنائية مشفرة بـ base64 بـ علامة "!!binary"، يمكن فك تشفيرها تلقائيًا عند التحميل.

لاحظ أن منطق التفريغ ربما يكون معطلاً، انظر https://github.com/perlpunk/YAML-PP-p5/issues/28.

الاقتراحات مرحب بها.
