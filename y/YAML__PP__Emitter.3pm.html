YAML__PP__Emitter(3pm)

الحزمة: perl-yaml-pp

التراخيص: PerlArtistic, GPL

YAML::PP::Emitter(3) وثائق مساهمة من مستخدمي بيرل YAML::PP::Emitter(3)

الاسم

YAML::PP::Emitter - إصدار الأحداث

الملخص

my $emitter = YAML::PP::Emitter->new(
    indent => 4,
);
$emitter->init;
$emitter->stream_start_event;
$emitter->document_start_event({ implicit => 1 });
$emitter->sequence_start_event;
$emitter->scalar_event({ value => $input, style => $style });
$emitter->sequence_end_event;
$emitter->document_end_event({ implicit => 1 });
$emitter->stream_end_event;
my $yaml = $emitter->writer->output;
$emitter->finish;

الوصف

يقوم المصدر بإصدار الأحداث إلى YAML. يوفر طرقًا لكل نوع حدث. الوسائط هي نفس الوسائط تقريبًا مثل الأحداث من YAML::PP::Parser.

الطرق

new
my $emitter = YAML::PP::Emitter->new(
    indent => 4,
);
