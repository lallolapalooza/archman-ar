yield(n)

الحزمة: tcl

التراخيص: TCL

coroutine(n) أوامر Tcl المدمجة coroutine(n)

الاسم

coroutine, yield, yieldto - إنشاء وإنتاج قيم من الكوروتينات (coroutines)

الملخص

coroutine name command ?arg...?
yield ?value?
yieldto command ?arg...?
name ?value...?

الوصف

يقوم الأمر coroutine بإنشاء سياق كوروتين جديد (مع أمر مرتبط) باسم name وينفذ هذا السياق عن طريق استدعاء command، وتمرير الوسائط المتبقية الأخرى دون مزيد من التفسير. بمجرد أن يعود command بشكل طبيعي أو مع استثناء (مثل خطأ)، يتم حذف سياق الكوروتين name.

داخل السياق، يمكن توليد قيم كنتائج عن طريق استخدام الأمر yield؛ إذا لم يتم توفير value، يتم استخدام السلسلة الفارغة. عند استدعاء ذلك، سيتم تعليق السياق وسيعيد الأمر coroutine الوسيطة إلى yield. يمكن بعد ذلك استئناف تنفيذ السياق عن طريق استدعاء أمر السياق، وتمرير قيمة واحدة اختياريًا لاستخدامها كنتيجة لاستدعاء yield الذي تسبب في تعليق السياق. إذا لم ينتج سياق الكوروتين أبدًا واستعاد بدلاً من ذلك تقليديًا، فستكون نتيجة الأمر coroutine هي نتيجة تقييم السياق.

يمكن للكوروتين أيضًا تعليق تنفيذه عن طريق استخدام الأمر yieldto، الذي بدلاً من العودة، يتنازل عن التنفيذ لأمر يسمى command (يتم حله في سياق الكوروتين) ويمكن تمرير *أي* عدد من الوسائط إليه. نظرًا لأن كل كوروتين لديه أمر سياق، يمكن استخدام yieldto لنقل التحكم مباشرة من كوروتين إلى آخر (هذا مستحسن فقط إذا كان الكوروتينان يتوقعان حدوث ذلك) ولكن *أي* أمر يمكن أن يكون الهدف. إذا تم تعليق كوروتين بهذه الآلية، يمكن استئناف معالجة الكوروتين عن طريق استدعاء أمر السياق وتمرير عدد اختياري من الوسائط. ستكون القيمة المعادة من استدعاء yieldto هي قائمة الوسائط التي تم تمريرها إلى أمر السياق؛ الأمر متروك للمتصل لتحديد ما يجب فعله بهذه القيم.

الطريقة الموصى بها لكتابة نسخة من yield تسمح بالاستئناف مع وسائط متعددة هي باستخدام yieldto والأمر return، كالتالي:

proc yieldm {value} {
    yieldto return -level 0 $value
}

يمكن أيضًا حذف الكوروتين عن طريق تدمير الأمر name، ويمكن استرداد اسم الكوروتين الحالي عن طريق استخدام info coroutine. إذا كانت هناك تتبعات حذف على المتغيرات في تنفيذ الكوروتين، فستعمل في النقطة التي يتم فيها حذف الكوروتين صراحةً (أو، بشكل طبيعي، إذا عاد الأمر تقليديًا).

في النقطة التي يتم فيها استدعاء command، سيكون مساحة الاسم الحالية هي مساحة الاسم العامة ولن تكون هناك إطارات مكدس فوقها (بمعنى upvar و uplevel). ومع ذلك، سيتم تحديد الأمر الذي يجب استدعاؤه في مساحة الاسم التي تم فيها استدعاء الأمر coroutine.

أمثلة

يوضح هذا المثال كوروتين سينتج تسلسلًا لا نهائيًا من القيم الزوجية، وحلقة تستهلك أول عشرة منها.

proc allNumbers {} {
    yield
    set i 0
    while 1 {
        yield $i
        incr i 2
    }
}
coroutine nextNumber allNumbers
for {set i 0} {$i < 10} {incr i} {
    puts "received [nextNumber]"
}
rename nextNumber {}

في هذا المثال، يعمل الكوروتين على جمع الوسائط التي تم تمريرها إليه.

coroutine accumulator apply {{} {
    set x 0
    while 1 {
        incr x [yield $x]
    }
}}
for {set i 0} {$i < 10} {incr i} {
    puts "$i -> [accumulator $i]"
}

يوضح هذا المثال استخدام الكوروتينات لتنفيذ خوارزمية غربال إراتوستين الكلاسيكية للعثور على الأعداد الأولية. لاحظ إنشاء الكوروتينات داخل كوروتين.

proc filterByFactor {source n} {
    yield [info coroutine]
    while 1 {
        set x [$source]
        if {$x % $n} {
            yield $x
        }
    }
}
coroutine allNumbers apply {{} {while 1 {yield [incr x]}}}
coroutine eratosthenes apply {c {
    yield
    while 1 {
        set n [$c]
        yield $n
        set c [coroutine prime$n filterByFactor $c $n]
    }
}} allNumbers
for {set i 1} {$i <= 20} {incr i} {
    puts "prime#$i = [eratosthenes]"
}

يوضح هذا المثال كيف يمكن تمرير قيمة عبر مجموعة من ثلاثة كوروتينات تتنازل عن التنفيذ لبعضها البعض:

proc juggler {name target {value ""}} {
    if {$value eq ""} {
        set value [yield [info coroutine]]
    }
    while {$value ne ""} {
        puts "$name : $value"
        set value [string range $value 0 end-1]
        lassign [yieldto $target $value] value
    }
}
coroutine j1 juggler Larry [
    coroutine j2 juggler Curly [
        coroutine j3 juggler Moe j1]] "Nyuck!Nyuck!Nyuck!"

الدلالات التفصيلية

يوضح هذا المثال أن الكوروتينات تبدأ من مساحة الاسم العامة، وأن حل command يحدث قبل إنشاء مكدس الكوروتين.

proc report {where level} {
    # Where was the caller called from?
    set ns [uplevel 2 {namespace current}]
    yield "made $where $level context=$ns name=[info coroutine]"
}
proc example {} {
    report outer [info level]
}
namespace eval demo {
    proc example {} {
        report inner [info level]
    }
    proc makeExample {} {
        puts "making from [info level]"
        puts [coroutine coroEg example]
    }
    makeExample
}

والذي ينتج الإخراج أدناه. على وجه الخصوص، يمكننا أن نرى أن تلاعب المكدس قد حدث (بمقارنة المستويات من السطر الأول والثاني) وأن المستوى الأب في الكوروتين هو مساحة الاسم العامة. يمكننا أيضًا أن نرى أن أسماء الكوروتينات محلية لمساحة الاسم الحالية إذا لم يتم التأهل لها، وأن الكوروتينات قد تتنازل عن التنفيذ عند عمق (على سبيل المثال، في الإجراءات المسماة).

making from 2
made inner 1 context=:: name=::demo::coroEg
