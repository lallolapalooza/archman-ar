bspwm(1)

الحزمة: bspwm

التراخيص: BSD-2-Clause

DESKTOP NODE SETTINGS

focus_follows_pointer

عند تركيز نافذة، ضع المؤشر عليها.

pointer_follows_focus

عند تركيز نافذة، ضع المؤشر في وسطها.

pointer_follows_monitor

عند تركيز شاشة، ضع المؤشر في وسطها.

mapping_events_count

تعامل مع أحداث إخطار التعيين mapping_events_count التالية. تشير القيمة السالبة إلى أنه يجب التعامل مع كل حدث.

ignore_ewmh_focus

تجاهل طلبات التركيز EWMH القادمة من التطبيقات.

ignore_ewmh_fullscreen

منع انتقالات حالة ملء الشاشة التي تنشأ عن طلب EWMH. القيم الممكنة هي: none، all، أو قائمة مفصولة بفواصل من القيم التالية: enter، exit.

ignore_ewmh_struts

تجاهل تلميحات الهياكل من العملاء الذين يطلبون حجز مساحة (مثل أشرطة المهام).

center_pseudo_tiled

توسيط النوافذ المزيفة (pseudo tiled) داخل مستطيلات التجانب الخاصة بها. القيمة الافتراضية هي true.

remove_disabled_monitors

اعتبر الشاشات المعطلة منفصلة.

remove_unplugged_monitors

قم بإزالة الشاشات المفصولة.

merge_overlapping_monitors

ادمج الشاشات المتداخلة (تبقى الأكبر).

إعدادات الشاشات وأسطح المكتب

top_padding، right_padding، bottom_padding، left_padding

مساحة الحشو المضافة على جوانب الشاشة أو سطح المكتب.

إعدادات سطح المكتب

window_gap

حجم الفجوة التي تفصل بين النوافذ.

إعدادات العقدة

border_width

عرض حدود النافذة.

honor_size_hints

إذا كانت القيمة true، فقم بتطبيق تلميحات حجم النافذة ICCCM على جميع النوافذ. إذا كانت القيمة floating، فقم بتطبيقها فقط على النوافذ العائمة والنوافذ المزيفة. إذا كانت القيمة tiled، فقم بتطبيقها فقط على النوافذ المتجانبة. إذا كانت القيمة false، فلا تقم بتطبيقها. الافتراضي هو false.

روابط المؤشر

click_to_focus

ركز النافذة (أو الشاشة) الموجودة تحت المؤشر إذا لم تكن القيمة none.

pointer_modifier + button1

قم بتحريك النافذة الموجودة تحت المؤشر.

pointer_modifier + button2

قم بتغيير حجم النافذة الموجودة تحت المؤشر عن طريق سحب أقرب جانب.

pointer_modifier + button3

قم بتغيير حجم النافذة الموجودة تحت المؤشر عن طريق سحب أقرب زاوية.

يمكن تعديل سلوك pointer_modifier + button<n> من خلال إعداد pointer_action<n>.

الأحداث

report

راجع القسم التالي لوصف التنسيق.

monitor_add <monitor_id> <monitor_name> <monitor_geometry>

تمت إضافة شاشة.

monitor_rename <monitor_id> <old_name> <new_name>

تمت إعادة تسمية شاشة.

monitor_remove <monitor_id>

تمت إزالة شاشة.

monitor_swap <src_monitor_id> <dst_monitor_id>

تم تبديل شاشة.

monitor_focus <monitor_id>

تم تركيز شاشة.

monitor_geometry <monitor_id> <monitor_geometry>

تغير هندسة الشاشة.

desktop_add <monitor_id> <desktop_id> <desktop_name>

تمت إضافة سطح مكتب.

desktop_rename <monitor_id> <desktop_id> <old_name> <new_name>

تمت إعادة تسمية سطح مكتب.

desktop_remove <monitor_id> <desktop_id>

تمت إزالة سطح مكتب.

desktop_swap <src_monitor_id> <src_desktop_id> <dst_monitor_id> <dst_desktop_id>

تم تبديل سطح مكتب.

desktop_transfer <src_monitor_id> <src_desktop_id> <dst_monitor_id>

تم نقل سطح مكتب.

desktop_focus <monitor_id> <desktop_id>

تم تركيز سطح مكتب.

desktop_activate <monitor_id> <desktop_id>

تم تنشيط سطح مكتب.

desktop_layout <monitor_id> <desktop_id> tiled|monocle

تغير تخطيط سطح المكتب.

node_add <monitor_id> <desktop_id> <ip_id> <node_id>

تمت إضافة عقدة.

node_remove <monitor_id> <desktop_id> <node_id>

تمت إزالة عقدة.

node_swap <src_monitor_id> <src_desktop_id> <src_node_id> <dst_monitor_id> <dst_desktop_id> <dst_node_id>

تم تبديل عقدة.

node_transfer <src_monitor_id> <src_desktop_id> <src_node_id> <dst_monitor_id> <dst_desktop_id> <dst_node_id>

تم نقل عقدة.

node_focus <monitor_id> <desktop_id> <node_id>

تم تركيز عقدة.

node_activate <monitor_id> <desktop_id> <node_id>

تم تنشيط عقدة.

node_presel <monitor_id> <desktop_id> <node_id> (dir DIR|ratio RATIO|cancel)

تم اختيار عقدة مسبقًا.

node_stack <node_id_1> below|above <node_id_2>

تم تكديس عقدة أسفل أو فوق عقدة أخرى.

node_geometry <monitor_id> <desktop_id> <node_id> <node_geometry>

تغير هندسة النافذة.

node_state <monitor_id> <desktop_id> <node_id> tiled|pseudo_tiled|floating|fullscreen on|off

تغيرت حالة النافذة.

node_flag <monitor_id> <desktop_id> <node_id> hidden|sticky|private|locked|marked|urgent on|off

تغير أحد علامات العقدة.

node_layer <monitor_id> <desktop_id> <node_id> below|normal|above

تغيرت طبقة النافذة.

pointer_action <monitor_id> <desktop_id> <node_id> move|resize_corner|resize_side begin|end

حدث إجراء للمؤشر.

يرجى ملاحظة أن bspwm يقوم بتهيئة الشاشات قبل قراءة الرسائل على مقبسه، لذلك لا يمكن استقبال أحداث الشاشة الأولية.

تنسيق التقرير

تتكون كل رسالة حدث تقرير من عناصر مفصولة بنقطتين.

كل عنصر له الشكل <type><value> حيث <type> هو الحرف الأول للعنصر.

M<monitor_name>

الشاشة المركزة.

m<monitor_name>

الشاشة غير المركزة.

O<desktop_name>

سطح المكتب المركز المشغول.

o<desktop_name>

سطح المكتب غير المركز المشغول.

F<desktop_name>

سطح المكتب المركز الفارغ.

f<desktop_name>

سطح المكتب غير المركز الفارغ.

U<desktop_name>

سطح المكتب المركز العاجل.

u<desktop_name>

سطح المكتب غير المركز العاجل.

L(T|M)

تخطيط سطح المكتب المركز للشاشة.

T(T|P|F|=|@)

حالة العقدة المركزة لسطح مكتب مركز.

G(S?P?L?M?)

العلامات النشطة للعقدة المركزة لسطح مكتب مركز.

متغيرات البيئة

BSPWM_SOCKET

مسار المقبس المستخدم للتواصل بين bspc و bspwm. إذا لم يكن معرفًا، يتم استخدام المسار التالي: /tmp/bspwm<host_name>_<display_number>_<screen_number>-socket.
