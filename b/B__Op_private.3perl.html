B::Op_private(3perl) دليل مبرمجي Perl B::Op_private(3perl)

الاسم

B::Op_private - تعريفات علامة OP op_private

الملخص

use B::Op_private;
# تفاصيل العلامة للبت 7 من op_private الخاص بـ OP_AELEM:
my $name  = $B::Op_private::bits{aelem}{7}; # OPpLVAL_INTRO
my $value = $B::Op_private::defines{$name}; # 128
my $label = $B::Op_private::labels{$name};  # LVINTRO
# حقل البتات في المواقع 5..6 من op_private الخاص بـ OP_AELEM:
my $bf  = $B::Op_private::bits{aelem}{6};
my $mask = $bf->{bitmask}; # الخ

الوصف

توفر هذه الوحدة أربعة جداول عامة:

%B::Op_private::bits
%B::Op_private::defines
%B::Op_private::labels
%B::Op_private::ops_using

التي تحتوي على معلومات حول المعاني لكل عملية للبتات في حقل op_private.

%bits

يتم الفهرسة هذه حسب اسم العملية ثم رقم البت (0..7). بالنسبة لعلامات البتات الفردية، فإنها تُرجع اسم التعريف (إن وجد) لتلك البتة:

$B::Op_private::bits{aelem}{7} eq 'OPpLVAL_INTRO';

بالنسبة لحقول البتات، فإنها تُرجع مرجع جدول يحتوي على تفاصيل الحقل. وسيتم إرجاع نفس المرجع لجميع مواضع البتات التي تشكل حقل البتات؛ لذلك على سبيل المثال، كلاهما يُرجع نفس مرجع الجدول:

$bitfield = $B::Op_private::bits{aelem}{5};
$bitfield = $B::Op_private::bits{aelem}{6};

الشكل العام لمرجع الجدول هذا هو

{
    # نطاق البت والقناع؛ هذه موجودة دائمًا.
    bitmin        => 5,
    bitmax        => 6,
    bitmask       => 0x60,
    # (المفاتيح المتبقية اختيارية)
    # أسماء أي تعريفات تم طلبها:
    mask_def      => 'OPpFOO_MASK',
    baseshift_def => 'OPpFOO_SHIFT',
    bitcount_def  => 'OPpFOO_BITS',
    # إذا كانت موجودة، فستعرض Concise وما إلى ذلك القيمة مع بادئة 'FOO='.
    # إذا كانت تساوي '-'، فستعامل Concise حقل البتات كبتات خام ولن تحاول تفسيرها.
    label         => 'FOO',
    # إذا كانت موجودة، تحدد أسماء بعض التعريفات وتسميات العرض المستخدمة
    # لتعيين معنى لقيم عددية معينة داخل حقل البتات؛ على سبيل المثال، يتم عرض 3
    # على أنه 'C'.
    enum          => [ qw(
                         1   OPpFOO_A  A
                         2   OPpFOO_B  B
                         3   OPpFOO_C  C
                     )],
};

%defines

يعطي هذا قيمة كل تعريف "OPp"، على سبيل المثال.

$B::Op_private::defines{OPpLVAL_INTRO} == 128;

%labels

يعطي هذا تسمية العرض القصيرة لكل تعريف، كما يستخدمها "B::Concise" و "perl -Dx"، على سبيل المثال.

$B::Op_private::labels{OPpLVAL_INTRO} eq 'LVINTRO';

إذا كانت التسمية تساوي '-'، فستعامل Concise البتة كبتة خام ولن تحاول عرضها بشكل رمزي.

%ops_using

لكل تعريف، يعطي هذا مرجعًا لمصفوفة من أسماء العمليات التي تستخدم العلامة.

@ops_using_lvintro = @{ $B::Op_private::ops_using{OPp_LVAL_INTRO} };
