الاسم

bsearch - بحث ثنائي في مصفوفة مرتبة

الملخص

#include <stdlib.h>

void *bsearch(const void *key, const void *base,
                size_t nmemb, size_t size,
                int (*compar)(const void *, const void *));

الوصف

تقوم الدالة bsearch بالبحث عن قيمة key في مصفوفة base. يجب أن تكون المصفوفة مرتبة بترتيب تصاعدي وفقًا للدالة compar. يجب أن تكون nmemb هي عدد العناصر في المصفوفة، ويجب أن تكون size هي حجم كل عنصر في البايت.

تأخذ الدالة compar مؤشرين إلى عنصرين من المصفوفة وتقوم بإرجاع:

  • قيمة سالبة إذا كان العنصر الأول أصغر من العنصر الثاني.
  • صفر إذا كان العنصران متساويين.
  • قيمة موجبة إذا كان العنصر الأول أكبر من العنصر الثاني.

إذا تم العثور على العنصر، تعيد bsearch مؤشرًا إلى العنصر المطابق. إذا لم يتم العثور على العنصر، تعيد NULL.

قيمة الإرجاع

تعيد bsearch مؤشرًا إلى العنصر المطابق إذا تم العثور عليه، وإلا فإنها تعيد NULL.

الأخطاء

لا تحدد مواصفات POSIX أي أخطاء للدالة bsearch.

أمثلة

البحث عن عدد صحيح في مصفوفة مرتبة:

#include <stdio.h>
#include <stdlib.h>

int compare_ints(const void *a, const void *b) {
    int arg1 = *(const int *)a;
    int arg2 = *(const int *)b;
    if (arg1 < arg2) return -1;
    if (arg1 > arg2) return 1;
    return 0;
}

int main() {
    int arr[] = {10, 20, 30, 40, 50};
    int key = 30;
    int nmemb = sizeof(arr) / sizeof(arr[0]);
    int *ptr = bsearch(&key, arr, nmemb, sizeof(int), compare_ints);

    if (ptr) {
        printf("Element found: %d\n", *ptr);
    } else {
        printf("Element not found.\n");
    }

    return 0;
}

استخدام التطبيق

تعد bsearch مفيدة عند البحث عن عناصر في مجموعات بيانات كبيرة تم فرزها مسبقًا. نظرًا لطبيعة البحث الثنائي، فإنها تتطلب أن تكون البيانات مرتبة، ولكنها توفر أداءً أفضل بكثير من البحث الخطي للمجموعات الكبيرة.

المنطق

تم تضمين bsearch في POSIX لتوفير طريقة قياسية للبحث الثنائي الفعال في المصفوفات. تعتمد على مبدأ البحث الثنائي الذي يقلل بشكل كبير من عدد المقارنات المطلوبة مقارنة بالبحث الخطي.

التوجهات المستقبلية

لا توجد توجهات مستقبلية معروفة أو تغييرات متوقعة للدالة bsearch.
