الاسم

bt_traversal - وظائف استكشاف/استعلام الشجرة النحوية المجردة (AST) في مكتبة btparse

الملخص

AST * bt_next_entry (AST * entry_list, 
                     AST * prev_entry)
AST * bt_next_field  (AST * entry, AST * prev, char ** name)
AST * bt_next_value  (AST * head, 
                      AST * prev,
                      bt_nodetype_t * nodetype,
                      char ** text)
bt_metatype_t bt_entry_metatype (AST * entry)
char * bt_entry_type (AST * entry)
char * bt_entry_key   (AST * entry)
char * bt_get_text   (AST * node)

الوصف

تُستخدم الوظائف الموصوفة هنا لاستكشاف واستعلام الأشجار النحوية المجردة (ASTs) التي تُرجعها وظائف الإدخال الموضحة في bt_input. تُستخدم وظائف "bt_next" الثلاثة (bt_next_entry()، bt_next_field()، و bt_next_value()) لاستكشاف قائمة المدخلات، وقائمة الحقول داخل مدخل معين، وقائمة القيم البسيطة المرتبطة بحقل معين، على التوالي. تُستخدم الوظائف الأخرى فقط لاستكشاف العقد المختلفة في الشجرة للحصول على المعلومات المفيدة التي تحتوي عليها.

وظائف الاستكشاف

bt_next_entry()
AST * bt_next_entry (AST * entry_list, 
                     AST * prev_entry)

تُستخدم لاستكشاف القائمة المتصلة من المدخلات التي تُرجعها bt_parse_file() (انظر bt_input). في المكالمة الأولى، يجب توفير "NULL" لـ "prev_entry"، وسيتم إرجاع مؤشر إلى رأس القائمة. في المكالمات اللاحقة، قم بتمرير القيمة المرجعة السابقة كـ "prev_entry"؛ تقوم الوظيفة بإرجاع المدخل التالي في القائمة، أو "NULL" إذا لم تكن هناك مدخلات أخرى. كما تُرجع "NULL" إذا كان "entry_list" أو "prev_entry" غير صالحين.

على سبيل المثال (مع تجاهل معالجة الأخطاء وتعريفات المتغيرات):

entries = bt_parse_file (filename, options, &status);
entry = NULL;
while (entry = bt_next_entry (entries, entry))
{
   /* معالجة المدخل */
}
bt_next_field()
AST * bt_next_field  (AST * entry, AST * prev, char ** name)

تُستخدم لاستكشاف قائمة الحقول في مدخل تعريف عادي أو ماكرو. (يجب عليك استدعاء bt_entry_metatype() لتحديد ما إذا كان لديك النوع الصحيح من المدخلات قبل استدعاء bt_next_field().) يجب أن يكون "entry" مؤشرًا إلى AST لمدخل واحد، كما تم إرجاعه بواسطة bt_parse_entry()، bt_parse_entry_s()، أو bt_next_entry(). في المكالمة الأولى، قم بتوفير "NULL" لـ "prev"؛ ستُرجع bt_next_field() مؤشرًا إلى الحقل الأول في "entry"، أو "NULL" إذا كان "entry" لا يحتوي على حقول (على سبيل المثال، إذا كان مدخل تعليق أو مقدمة). في المكالمات اللاحقة، قم بتمرير القيمة المرجعة السابقة كـ "prev"؛ ستستمر bt_next_field() في إرجاع مؤشرات إلى ASTs فرعية للحقول طالما كان ذلك منطقيًا. يمكن بعد ذلك تمرير هذه المؤشرات إلى bt_next_value() أو bt_get_text() للحصول على قيمة الحقل.

على سبيل المثال، يمكن أن يكون جسم الحلقة في المثال السابق:

field = NULL;
while (field = bt_next_field (entry, field, &field_name))
{
   /* معالجة الحقل */
}
bt_next_value()
AST * bt_next_value (AST * head, 
                     AST * prev,
                     bt_nodetype_t * nodetype,
                     char ** text)

تستكشف قائمة القيم البسيطة التي تشكل قيمة حقل واحد. (تذكر أن القيمة البسيطة هي إما سلسلة نصية محاطة بعلامات اقتباس، أو استدعاء ماكرو، أو رقم. القيمة المركبة هي قائمة من هذه مفصولة بـ '#' في الإدخال الأصلي. اعتمادًا على خيارات معالجة السلاسل النصية المستخدمة عند تحليل البيانات، قد يتم الحفاظ على طبيعة "قائمة القيم البسيطة" للبيانات الأصلية في AST الذي تستكشفه، وفي هذه الحالة ستحتاج إلى حلقة bt_next_value().

تعمل bt_next_value() بشكل مشابه لـ bt_next_entry() و bt_next_field(): في المكالمة الأولى، توفر "NULL" لـ "prev"، وفي المكالمات اللاحقة توفر القيمة المرجعة السابقة. تُرجع "NULL" عندما لا تكون هناك قيم بسيطة أخرى لإرجاعها. كما تقوم بتعيين *nodetype و *text إلى المعلومات المقابلة من عقدة القيمة البسيطة. سيكون *nodetype أحد "BTAST_STRING"، "BTAST_MACRO"، أو "BTAST_NUMBER"؛ سيشير *text إلى نفس السلسلة النصية التي تشير إليها عقدة AST (لم يتم نسخها لك)، لذا لا تتلاعب بها.

على سبيل المثال، يمكن استبدال جسم الحلقة في مثال bt_next_field() بما يلي:

value = NULL;
while (value = bt_next_field (field, value, &nodetype, &text))
{
   switch (nodetype)
   {
      case BTAST_STRING:    /* معالجة السلسلة النصية */
      case BTAST_MACRO:     /* معالجة الماكرو */
      case BTAST_NUMBER:    /* معالجة الرقم */
   }
}

انظر أيضًا "bt_get_text".

وظائف الاستعلام

bt_entry_metatype()
bt_metatype_t bt_entry_metatype (AST * entry)

تُرجع النوع الفوقي (metatype) لمدخل. (تذكر أن النوع الفوقي هو نوع تعدادي تُشتق قيمه من النوع المحدد للمدخل؛ على سبيل المثال، مدخل @comment له نوع "comment" ونوع فوقي "BTE_COMMENT". العلاقة بين النوع والنوع الفوقي واضحة بالمثل لـ "BTE_PREAMBLE"؛ "BTE_MACRODEF" يقابل مدخلات @string؛ و "BTE_REGULAR" يقابل أي نوع آخر.)

تُرجع "BTE_UNKNOWN" إذا كان "entry" غير صالح (أي، "NULL" أو ليس مؤشرًا إلى AST مدخل).

bt_entry_type()
char * bt_entry_type (AST * entry)

تُرجع نوع المدخل. تذكر أن النوع هو الاسم الذي يظهر بعد الحرف '@' في الإدخال الأصلي. تُرجع "NULL" إذا كان "entry" غير صالح (أي، "NULL" أو ليس مؤشرًا إلى AST مدخل).

bt_entry_key()
char * bt_entry_key (AST * entry)

تُرجع مفتاح الاستشهاد لمدخل عادي. (مفتاح الاستشهاد هو الاسم الذي يظهر بعد المحدد المفتوح للمدخل في مدخل عادي.) تُرجع "NULL" إذا كان "entry" غير صالح (أي، "NULL" أو ليس مؤشرًا إلى AST لمدخل عادي).

bt_get_text()
char * bt_get_text (AST * node)
