الحزمة: perl-text-bibtex

التراخيص: PerlArtistic, GPL

الاسم

bt_format_names, bt_free_name_format, bt_set_format_text, bt_set_format_options, bt_format_name - تنسيق أسماء BibTeX

نبذة مختصرة

typedef struct bt_name_format bt_name_format;
void bt_free_name_format (bt_name_format * format);
void bt_set_format_text (bt_name_format * format,

الوصف

تستخدم هذه الوظائف لإنشاء وتخصيص وتطبيق تنسيقات الأسماء. يمكنك إنشاء تنسيق اسم مع عدد من الأجزاء المطلوبة، مثل "fvlj" (first-middle-last-junior) أو "vljf" (middle-first-last-junior)، ثم تخصيص كيفية عرض كل جزء. بعد ذلك، يمكنك استخدام التنسيق لتطبيق الأسماء.

  • أولاً، يتم إنشاء التنسيق باستخدام bt_create_name_format(). يتم تمرير سلسلة تصف ترتيب أجزاء الاسم المطلوبة، على سبيل المثال، "fvlj". يتم تحديد ما إذا كان يجب اختصار الأسماء الأولى والمتوسطة (عادةً ما يتم اختصارها).

  • ثم، يتم تطبيق بعض القواعد الافتراضية. على سبيل المثال، إذا تم تحديد أن الجزء الأخير هو "jr"، فسيتم عرض "." بعد الاختصار. هذا يغير "J R Smith" (بافتراض أن "J" هو الجزء الأول و"R" هو الجزء الأوسط) إلى "J. R. Smith"، وهو الشكل المرغوب عادةً. (إذا كنت *لا* تريد النقاط، سيتعين عليك تعيين النص اللاحق للرمز بنفسك باستخدام bt_set_format_text().)

  • بعد ذلك، إذا كان "jr" موجودًا ويأتي مباشرة بعد "last" (وهو ما يحدث دائمًا تقريبًا)، يتم تعيين النص السابق للجزء "jr" إلى ", "، ويتم تعيين طريقة ربط الأجزاء لـ "last" إلى "BTJ_NOTHING". هذا يغير "John Smith Jr" (حيث تأتي المسافة التي تلي "Smith" من تنسيق الاسم الأخير بطريقة ربط الأجزاء "BTJ_SPACE") إلى "John Smith, Jr" (حيث يرتبط ", " الآن بـ "Jr" - بهذه الطريقة، إذا لم يكن هناك جزء "jr"، فلن تتم طباعة ", ".)

  • أخيرًا، إذا كان "first" موجودًا ويأتي مباشرة بعد "jr" أو "last" (وهو ما يحدث عادةً في تنسيقات "last-name first")، يتم تطبيق نفس النوع من الحيل: يتم تعيين النص السابق للجزء "first" إلى ", "، ويتم تعيين طريقة ربط الأجزاء للجزء السابق (إما "jr" أو "last") إلى "BTJ_NOTHING".

على الرغم من أن كل هذه القواعد معقدة إلى حد ما، إلا أنها تعني أنك عادةً ما تكون معفيًا من القيام بأي تخصيص لتنسيق الاسم. بالتأكيد هذا هو الحال إذا كنت تحتاج فقط إلى ترتيبات أجزاء "fvlj" و "vljf"، وتريد فقط اختصار الاسم الأول، وتريد نقاطًا بعد الرموز المختصرة، ومسافات غير قابلة للكسر في الأماكن "الصحيحة"، وفواصل في الأماكن التقليدية.

إذا كنت تريد شيئًا غير عادي - على سبيل المثال، رموز مختصرة مجمعة معًا بدون علامات ترقيم، أو أسماء أخيرة مختصرة - ستحتاج إلى تخصيص تنسيق الاسم قليلاً باستخدام bt_set_format_text() و bt_set_format_options().

bt_free_name_format()
void bt_free_name_format (bt_name_format * format)

يحرر تنسيق اسم تم إنشاؤه بواسطة bt_create_name_format().

bt_set_format_text()
void bt_set_format_text (bt_name_format * format,
                         bt_namepart part,
                         char * pre_part,
                         char * post_part,
                         char * pre_token,
                         char * post_token)

يسمح لك بتخصيص بعض أو كل النص المحيط لجزء اسم واحد. قم بتوفير "NULL" لأي جزء من النص لا تريد تغييره.

على سبيل المثال، لنفترض أنك تريد تنسيق اسم يختصر الأسماء الأولى، ولكن بدون أي علامات ترقيم بعد الرموز المختصرة. يمكنك إنشاء وتخصيص التنسيق على النحو التالي:

format = bt_create_name_format ("fvlj", TRUE);
bt_set_format_text (format,
                    BTN_FIRST,       /* جزء الاسم المراد تخصيصه */
                    NULL, NULL,      /* نص سابق ولاحق للجزء */
                    NULL, "");       /* سلسلة فارغة للنص اللاحق للرمز */

بدون استدعاء bt_set_format_text()، كان "format" سينتج أسماء منسقة مثل "J. R. Smith". بعد تعيين النص اللاحق للرموز للأسماء الأولى إلى ""، سيصبح هذا الاسم "J R Smith".

bt_set_format_options()
void bt_set_format_options (bt_name_format * format,
                            bt_namepart part,
                            boolean abbrev,
                            bt_joinmethod join_tokens,
                            bt_joinmethod join_part)

يسمح بتخصيص إضافي لتنسيق الاسم: يمكنك تعيين علامة الاختصار وطريقتي ربط الرموز. للأسف، لا توجد آلية لترك قيمة دون تغيير؛ يجب عليك تعيين كل شيء باستخدام bt_set_format_options().

على سبيل المثال، لنفترض أن مجرد إسقاط النقاط من الرموز المختصرة في الاسم الأول ليس كافياً؛ فأنت *حقًا* تريد توفير المساحة عن طريق تجميع الرموز المختصرة معًا: "JR Smith" بدلاً من "J R Smith". بافتراض أن الاستدعاءين في المثال أعلاه قد تم إجراؤهما، فإن ما يلي سيكمل المهمة:

bt_set_format_options (format, BTN_FIRST,
                       TRUE,         /* الاحتفاظ بنفس القيمة لعلامة الاختصار */
                       BTJ_NOTHING,  /* تجميع الرموز معًا */
                       BTJ_SPACE);   /* مسافة بعد الرمز الأخير للجزء */

لاحظ أننا للأسف اضطررنا إلى معرفة (وتوفير) القيم الحالية لعلامة الاختصار وطريقة ربط ما بعد الجزء، على الرغم من أننا كنا نقوم فقط بتعيين طريقة ربط داخل الجزء.

bt_format_name()
char * bt_format_name (bt_name * name, bt_name_format * format)

بمجرد إنشاء تنسيق اسم وتخصيصه حسب رغبتك، يمكنك استخدامه لتنسيق أي عدد من الأسماء التي تم تقسيمها باستخدام "bt_split_name" (انظر bt_split_names). ببساطة قم بتمرير بنية الاسم وبنية تنسيق الاسم، وسيتم إرجاع سلسلة جديدة مخصصة تحتوي على الاسم المنسق إليك. تقع على عاتقك مسؤولية free() هذه السلسلة.
