val_free_dane(3)

الحزمة: dnssec-tools

التراخيص: BSD

val_getdaneinfo(3) دليل المبرمج val_getdaneinfo(3)

الاسم

val_getdaneinfo() - إجراء التحقق المتزامن من سجلات TLSA

val_dane_submit() - إجراء التحقق غير المتزامن من سجلات TLSA

val_dane_match() - التحقق من معلومات TLSA مقابل البيانات المقدمة.

val_dane_check() - التحقق من معلومات TLSA لاتصال SSL (OpenSSL فقط)

val_free_dane() - تحرير الذاكرة المرتبطة ببنية نتائج DANE.

p_dane_error() - إرجاع سلسلة الخطأ لرمز خطأ DANE المحدد.

الملخص

#include <validator/validator.h>
#include <validator/val_dane.h>
int val_getdaneinfo(val_context_t *ctx,
                  const char *name,
                  struct val_daneparams *params,
                  struct val_danestatus **dres);
int val_dane_submit(val_context_t *ctx,
                  const char *name,
                  struct val_daneparams *params,
                  val_dane_callback callback,
                  void *callback_data,
                  val_async_status **status);
int val_dane_match(val_context_t *ctx,
                 struct val_danestatus *dane_cur,
                 const unsigned char *databytes,
                 int databyteslen);
#include <openssl/ssl.h>
int val_dane_check(val_context_t *ctx,
                 SSL *con,
                 struct val_danestatus *danestatus,
                 int *do_pathval);
void val_free_dane(struct val_danestatus *dres);
const char *p_dane_error(int rc);

الوصف

val_getdaneinfo() يقوم بالبحث المتزامن عن سجل TLSA المرتبط بالاسم name المحدد ويرجع قائمة مرتبطة بجميع سجلات التحقق تلك. val_dane_submit() يقوم بنفس البحث بطريقة غير متزامنة ويستدعي الدالة callback مع وسائط callback_data عند اكتمال البحث. التعريف النوعي للدالة callback هو:

typedef int (*val_dane_callback)(void *callback_data,
                               int retval,
                               struct val_danestatus **dres);

توفر الوسيطة status مقبضًا لطلب عدم التزامن لتمكين العمليات المستقبلية (مثل إلغاء الطلب). لمزيد من المعلومات حول الكائن val_async_status راجع draft-hayatnagarkar-dnsext-validator-api.

اسم DNS الفعلي الذي يمتلك سجل TLSA في DNS له بادئة بالشكل _<port>._<proto>. val_getdaneinfo() سيقوم بإنشاء البادئة المذكورة أعلاه تلقائيًا؛ لذا لا ينبغي أن يحتوي اسم name الذي يقدمه المستخدم على هذه البادئة. الـ

يجب تقديم معلمات بحث TLSA في الوسيطة params، وهي مؤشر إلى البنية التالية:

struct val_daneparams {
  int port;
  int proto;
};

تستخدم حقلا port و proto في إنشاء بادئة اسم TLSA الموضحة أعلاه.

تُعاد نتائج بحث TLSA في الوسيطة dres، وهي مؤشر إلى قائمة مرتبطة بالبنى بالشكل التالي:

struct val_danestatus {
  long ttl;
  int usage;
  int selector;
  int type;
  size_t datalen;
  unsigned char *data;
  struct val_danestatus *next;
};

الحقل ttl هو وقت البقاء المرتبط بسجل TLSA. يجب على التطبيق عدم تخزين (واستخدام) سجل TLSA هذا بعد مدة صلاحيته. تتوافق حقلا usage و selector و type مع الحقول الثلاثة الأولى من RDATA في TLSA كما هو موضح في rfc6698. تُعاد بيانات ارتباط الشهادة في الحقل data ولها طول datalen بايت. قد يكون هناك أكثر من سجل TLSA واحد مرتبط باسم معين، ويشير الحقل next إلى السجل التالي في هذه القائمة.

بالنظر إلى قائمة مرتبطة ببنى TLSA في dres، يمكن استخدام val_dane_match() للتحقق مما إذا كانت بيانات ارتباط الشهادة لعنصر معين في هذه القائمة تتطابق مع البيانات المشفرة DER المقدمة في databytes بطول databyteslen.

تبسط الدالة val_dane_check() عملية المطابقة عندما يستخدم OpenSSL لتوفير دعم SSL/TLS ضمن التطبيق. تتكرر هذه الدالة تلقائيًا عبر جميع العناصر في dres وتقارن بيانات ارتباط الشهادة مع شهادات SSL/TLS المرتبطة باتصال SSL con. يمكّن بروتوكول DANE حالات استخدام معينة تسمح بإدخال مرساة ثقة جديدة عبر DNSSEC. تشير قيمة do_pathval إلى ما إذا كان يجب على التطبيق المتابعة مع التحقق من المسار X509 لهذا الاتصال وفقًا للاستخدام الذي تم ترميزه في سجل TLSA.

تقوم الدالة val_free_dane() بتحرير الذاكرة المرتبطة بالقائمة المرتبطة المشار إليها بواسطة dres.

تحدد المعلمة ctx في جميع الدوال المذكورة أعلاه سياق التحقق، والذي يمكن تعيينه إلى NULL للقيم الافتراضية (راجع libval(3) و dnsval.conf لمزيد من التفاصيل حول سياقات التحقق وسياسة التحقق).

قيم الإرجاع

val_getdaneinfo() و val_dane_submit() ترجعان VAL_DANE_NOERROR عند النجاح، و VAL_DANE_MALFORMED_TLSA أو VAL_DANE_INTERNAL_ERROR للحالات الخطأ. تُعاد القيمة VAL_DANE_NOTVALIDATED إذا تعذر التحقق من سجل TLSA عبر DNSSEC. تُعاد القيمة VAL_DANE_IGNORE_TLSA إذا كان سجل TLSA للاسم المحدد غائبًا بشكل قابل للإثبات.

يمكن أن تحتوي القيمة retval التي تُعاد كوسيطة لـ val_dane_callback() على أحدها: VAL_DANE_NOERROR (للنجاح)، VAL_DANE_INTERNAL_ERROR (للحالات الخطأ)، أو VAL_DANE_CANCELLED (عند إلغاء طلب غير متزامن).

val_dane_match() و val_dane_check() ترجعان VAL_DANE_NOERROR عند النجاح، VAL_DANE_INTERNAL_ERROR للحالات الخطأ العامة، و VAL_DANE_CHECK_FAILED إذا تعذر مطابقة سجل TLSA بنجاح مع بيانات ارتباط الشهادة المقدمة.

يمكن استخدام الدالة p_dane_error() لتحويل رموز خطأ DANE المتعلقة إلى قيمة سلسلة خطأ.
