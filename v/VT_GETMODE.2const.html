ioctl_vt(2) دليل استدعاءات النظام ioctl_vt(2)

الاسم

ioctl_vt - عمليات تحكم دخل/خرج لمحطات طرفية ووحدات افتراضية

الملخص

#include <linux/vt.h>  /* تعريف الثوابت VT_* */
#include <sys/ioctl.h>
int ioctl(int fd, unsigned long op, void *argp);

الوصف

عمليات ioctl(2) التالية الخاصة بنظام لينكس مدعومة لمحطات طرفية ووحدات افتراضية:

VT_OPENQRY
تعيد الوحدة الافتراضية المتاحة الأولى (غير المفتوحة). argp تشير إلى متغير من نوع int يتم تعيين قيمة رقم الوحدة الافتراضية له (1 <= *argp <= MAX_NR_CONSOLES).
VT_GETMODE
احصل على وضع الوحدة الافتراضية النشطة. argp تشير إلى

struct vt_mode {
    char  mode;    /* وضع الوحدة الافتراضية */
    char  waitv;   /* إذا تم التعيين، علق عند الكتابة إذا لم تكن نشطة */
    short relsig;  /* إشارة لإطلاقها عند عملية الإصدار */
    short acqsig;  /* إشارة لإطلاقها عند الاستحواذ */
    short frsig;   /* غير مستخدم (تم التعيين إلى 0) */
};

والتي يتم تعيينها إلى وضع الوحدة الافتراضية النشطة. يتم تعيين mode إلى إحدى القيم التالية:
VT_AUTO تبديل تلقائي للوحدة الافتراضية
VT_PROCESS العملية تتحكم في التبديل
VT_ACKACQ إقرار بالتبديل
VT_SETMODE
تعيين وضع الوحدة الافتراضية النشطة. argp تشير إلى struct vt_mode.
VT_GETSTATE
احصل على معلومات حالة الوحدة الافتراضية العامة. argp تشير إلى

struct vt_stat {
    unsigned short v_active;  /* الوحدة الافتراضية النشطة */
    unsigned short v_signal;  /* الإشارة المراد إرسالها */
    unsigned short v_state;   /* قناع بتات الوحدة الافتراضية */
};

لكل وحدة افتراضية قيد الاستخدام، يتم تعيين البت المقابل في عضو v_state. (لينكس 1.0 إلى لينكس 1.1.92).
VT_RELDISP
إصدار شاشة عرض.
VT_ACTIVATE
التبديل إلى الوحدة الافتراضية argp (1 <= argp <= MAX_NR_CONSOLES).
VT_WAITACTIVE
الانتظار حتى يتم تنشيط الوحدة الافتراضية argp.
VT_DISALLOCATE
إلغاء تخصيص الذاكرة المرتبطة بالوحدة الافتراضية argp. (منذ لينكس 1.1.54).
VT_RESIZE
تعيين حجم الشاشة في نواة النظام. argp تشير إلى

struct vt_sizes {
    unsigned short v_rows;       /* عدد الصفوف */
    unsigned short v_cols;       /* عدد الأعمدة */
    unsigned short v_scrollsize; /* لم يعد مستخدمًا */
};

لاحظ أن هذا لا يغير وضع الفيديو. انظر resizecons(8). (منذ لينكس 1.1.54).
VT_RESIZEX
تعيين معلمات الشاشة المختلفة في نواة النظام. argp تشير إلى 

struct vt_consize {
    unsigned short v_rows;  /* عدد الصفوف */
    unsigned short v_cols;  /* عدد الأعمدة */
    unsigned short v_vlin;  /* عدد صفوف البكسل
                               على الشاشة */
    unsigned short v_clin;  /* عدد صفوف البكسل
                               لكل حرف */
    unsigned short v_vcol;  /* عدد أعمدة البكسل
                               على الشاشة */
    unsigned short v_ccol;  /* عدد أعمدة البكسل
                               لكل حرف */
};

قد يتم تعيين أي معلمة إلى الصفر، مما يعني "لا تغيير"، ولكن إذا تم تعيين معلمات متعددة، فيجب أن تكون متسقة ذاتيًا. لاحظ أن هذا لا يغير وضع الفيديو. انظر resizecons(8). (منذ لينكس 1.3.3).
VT_GETCONSIZECSRPOS
الحصول على حجم الوحدة الطرفية وموضع المؤشر. argp تشير إلى

struct vt_consizecsrpos {
    __u16 con_rows;  /* عدد صفوف الوحدة الطرفية */
    __u16 con_cols;  /* عدد أعمدة الوحدة الطرفية */
    __u16 csr_row;   /* الصف الحالي للمؤشر */
    __u16 csr_col;   /* العمود الحالي للمؤشر */
};

(منذ لينكس 6.16.)

قيمة الإرجاع

عند النجاح، يتم إرجاع 0 (باستثناء ما هو مشار إليه). عند الفشل، يتم إرجاع -1، ويتم تعيين errno للإشارة إلى الخطأ.

الأخطاء

EINVAL
argp غير صالح.

المعايير

لينكس.

انظر أيضًا
