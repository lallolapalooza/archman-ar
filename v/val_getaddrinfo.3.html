val_getaddrinfo(3)

الحزمة: dnssec-tools

التراخيص: BSD

val_getaddrinfo(3) دليل المبرمج val_getaddrinfo(3)

الاسم

val_getaddrinfo() - الحصول على عنوان شبكة تم التحقق منه بواسطة DNSSEC وترجمة الخدمة

ملخص

#include <validator/validator.h>
int val_getaddrinfo(const struct val_context *ctx,
                    const char *nodename,
                    const char *servname,
                    const struct addrinfo *hints,
                    struct addrinfo **res,
                    val_status_t * val_status);
int val_getnameinfo(val_context_t * ctx,
                    const struct sockaddr *sa,
                    socklen_t salen,
                    char *host,
                    size_t hostlen,
                    char *serv,
                    size_t servlen,
                    int flags,
                    val_status_t * val_status);

الوصف

val_getaddrinfo() و val_getnameinfo يقومان بالتحقق من صحة استعلامات DNS بواسطة DNSSEC. إنهما يهدفان إلى أن يكونا بدائل مدركة لـ DNSSEC لـ getaddrinfo(3) و getnameinfo(3).

val_getaddrinfo() يعيد قيمة عنوان شبكة من النوع struct addrinfo في المعلمة res. val_getnameinfo تستخدم لتحويل بنية sockaddr إلى زوج من أسماء المضيف وسلاسل الخدمة.

val_status يعطي قيمة حالة التحقق المدمجة لجميع الإجابات التي أعادتها كل دالة من الدوال. val_istrusted() و val_isvalidated() يمكن استخدامها لتحديد مدى موثوقية البيانات و p_val_status() يمكن استخدامها لعرض قيمة الحالة للمستخدم بصيغة ASCII (انظر libval(3) لمزيد من المعلومات).

المعلمة ctx تحدد سياق التحقق، والذي يمكن تعيينه إلى NULL للقيم الافتراضية (انظر libval(3) و dnsval.conf لمزيد من التفاصيل حول سياقات التحقق و سياسة التحقق).

المعلمات nodename و servname و hints لها بناء جملة ودلالات مشابهة للمعلمات المقابلة للدالة الأصلية getaddrinfo(). المعلمة res مشابهة للمعلمة res للدالة getaddrinfo(). يرجى الرجوع إلى صفحة الدليل لـ getaddrinfo(3) لمزيد من التفاصيل حول هذه المعلمات.

قيم الإرجاع

الدالة val_getaddrinfo() تعيد 0 عند النجاح ورمز خطأ غير صفري عند الفشل. *res سيشير إلى قائمة مرتبطة مخصصة ديناميكيًا من هياكل addrinfo عند النجاح وستكون NULL إذا لم يكن هناك إجابة متاحة.

المعلمة val_status تعطي مؤشراً لمدى موثوقية البيانات. إذا كان *res هو NULL، فإن هذه القيمة تعطي مؤشراً لما إذا كان عدم وجود البيانات يمكن الوثوق به أم لا.

مثال

#include <stdio.h>
#include <stdlib.h>
#include <validator.h>
int main(int argc, char *argv[])
{
         struct addrinfo *ainfo = NULL;
         int retval;
         if (argc < 2) {
                 printf("Usage: %s <hostname>\n", argv[0]);
                 exit(1);
         }
         retval = val_getaddrinfo(NULL, argv[1], NULL, NULL, &ainfo);
         if ((retval == 0) && (ainfo != NULL)) {
                 printf("Validation Status = %d [%s]\n",
                        ainfo->ai_val_status,
                        p_val_status(ainfo->ai_val_status));
                 val_freeaddrinfo(ainfo);
         }
         return 0;
}
