vfs_fileid(8)

الحزمة: samba

التراخيص: GPL-3.0-or-later

VFS_FILEID(8) أدوات إدارة النظام VFS_FILEID(8)

الاسم

vfs_fileid - يقوم بإنشاء هياكل file_id مع معرفات أجهزة فريدة لإعدادات التجميع. كما يضيف طرقًا لكسر اتساق القفل عن قصد لأدلة معينة

ملخص

كائنات vfs = fileid

الوصف

وحدة VFS هذه جزء من مجموعة samba(7).

يستخدم Samba هياكل file_id لتحديد الملفات بشكل فريد لأغراض القفل. افتراضيًا، يحتوي file_id على معرف الجهاز ورقم الدليل (inode) الذي تم إرجاعه بواسطة استدعاء النظام stat(). نظرًا لأن file_id هو معرف فريد للملف، فيجب أن يكون هو نفسه على جميع العقد في إعداد تجميع. تتجاوز هذه الوحدة عملية SMB_VFS_FILE_ID_CREATE وتقوم بإنشاء معرف الجهاز بناءً على الخوارزمية المحددة (انظر الخيار "fileid:algorithm").

عند استخدام خوارزمية fsname أو fsid، يتطلب الأمر استدعاء stat() و statfs() لجميع أنظمة الملفات المثبتة لإنشاء file_id. إذا كان نظام ملفات NFS، على سبيل المثال، غير مستجيب، فقد يؤدي هذا الاستدعاء إلى تعطيل عملية smbd وتصبح غير مستجيبة. استخدم خيارات "fileid:fstype deny" أو "fileid:fstype allow" أو "fileid:mntdir deny" أو "fileid:mntdir allow" لتجاهل أنظمة الملفات التي قد لا تستجيب.

الخيارات

fileid:algorithm = ALGORITHM

الخوارزميات المتاحة هي fsname و fsid و next_module. القيمة الافتراضية هي fsname. بالإضافة إلى الخوارزميات القديمة التالية: fsname_nodirs و fsname_norootdir و fsname_norootdir_ext و hostname.

خوارزمية fsname تقوم بإنشاء معرف الجهاز عن طريق تجزئة اسم جهاز النواة.

خوارزمية fsid تقوم بإنشاء معرف الجهاز من f_fsid الذي تم إرجاعه من استدعاء النظام statfs().

خوارزمية next_module تسمح للوحدة VFS التالية في سلسلة الوحدات بإنشاء المعرف. يتم استخدام هذا بشكل أساسي بالاقتران مع ميزات "nolock" المختلفة التي توفرها وحدة fileid.

الخوارزمية القديمة hostname تقوم بإنشاء معرف جهاز فريد عن طريق تجزئة اسم المضيف ومعرف الجهاز منخفض المستوى. كما أنها تتضمن fileid:nolock_all_inodes=yes. يمكن استخدام هذا لكسر اتساق القفل عن قصد في بيئة التجميع ومع fileid:nolock_max_slots بين العمليات المحلية داخل المضيف. ملاحظة: لا تستخدم هذا دون معرفة ما تفعله! إنه يكسر دلالات SMB ويمكن أن يؤدي إلى تلف البيانات! هذا يتضمن fileid:nolock_all_inodes=yes.

الخوارزمية القديمة fsname_nodirs هي اسم مستعار لاستخدام خوارزمية fsname مع fileid:nolock_all_dirs=yes. ملاحظة: لا تستخدم هذا دون معرفة ما تفعله! إنه يكسر دلالات SMB! راجع fileid:nolock_paths للحصول على نهج أكثر دقة.

الخوارزمية القديمة fsname_norootdir هي اسم مستعار لاستخدام خوارزمية fsname مع fileid:nolock_paths = ".". هذا يعني أنه يمكن استخدامه لكسر اتساق القفل عن قصد للدليل الجذر للمشاركة.

الخوارزمية القديمة fsname_norootdir_ext هي اسم مستعار لاستخدام خوارزمية fsname مع fileid:nolock_paths = "." و fileid:nolock_max_slots = 18446744073709551615. هذا يعني أنه يمكن استخدامه لكسر اتساق القفل تمامًا للدليل الجذر للمشاركة. حتى العمليات المحلية لم تعد متسقة في قفلها.

fileid:mapping = ALGORITHM

هذا الخيار هو الإصدار القديم من خيار fileid:algorithm، والذي تم استخدامه في الإصدارات السابقة من ميزة تعيين fileid في إصدارات Samba 3.0 المخصصة.

fileid:fstype deny = LIST

قائمة أنواع أنظمة الملفات التي يجب تجاهلها لإنشاء file_id.

fileid:fstype allow = LIST

قائمة أنواع أنظمة الملفات التي يُسمح بها لإنشاء file_id. إذا تم تعيين هذا الخيار، فسيتم تجاهل أنواع أنظمة الملفات غير المدرجة هنا.

fileid:mntdir deny = LIST

قائمة نقاط تثبيت أنظمة الملفات التي يجب تجاهلها لإنشاء file_id.

fileid:mntdir allow = LIST

قائمة نقاط تثبيت أنظمة الملفات التي يُسمح بها لإنشاء file_id. إذا تم تعيين هذا الخيار، فسيتم تجاهل نقاط تثبيت أنظمة الملفات غير المدرجة هنا.

fileid:nolock_max_slots = NUMBER(1-18446744073709551615)

يغير هذا الخيار سلوك خوارزمية nolock بطريقة تكسر أيضًا اتساق القفل بين العمليات الفردية على نفس المضيف. الافتراضي هو وجود فتحة واحدة متزامنة فقط لكل مضيف. بزيادة عدد الفتحات، يمكنك تحديد عدد العمليات المتزامنة التي يمكنها العمل على دليل معين دون تنازع، يجب أن يكون العدد أكبر عادةً من عدد وحدات المعالجة المركزية المنطقية، ربما 2 ضعف عدد وحدات المعالجة المركزية.

fileid:nolock_all_dirs = BOOL

يشغل هذا الخيار استخدام سلوك fileid nolock لجميع أدلة الدليل، والتي يمكن استخدامها لكسر اتساق القفل لجميع الأدلة عن قصد. ملاحظة: لا تستخدم هذا دون معرفة ما تفعله! إنه يكسر دلالات SMB! راجع fileid:nolock_paths للحصول على نهج أكثر دقة.

fileid:nolock_all_inodes = BOOL

يشغل هذا الخيار استخدام خوارزمية fileid nolock لجميع أدلة الدليل، والتي يمكن استخدامها لكسر اتساق القفل لجميع الأدلة عن قصد. ملاحظة: لا تستخدم هذا دون معرفة ما تفعله! إنه يكسر دلالات SMB ويمكن أن يؤدي إلى تلف البيانات! راجع fileid:nolock_paths للحصول على نهج أكثر دقة.

fileid:nolock_paths = LIST

يحدد هذا الخيار قائمة مسارات تشير إلى الملفات و/أو الأدلة، والتي يجب أن تستخدم خوارزمية fileid nolock لكسر اتساق القفل لها عن قصد. يمكن أن تكون المسارات المحددة نسبية لدليل جذر المشاركة أو مطلقة. الأسماء هي مسارات يونكس حساسة لحالة الأحرف! لاحظ أن جميع المسارات يتم تقييمها فقط عند اتصال الشجرة، عند الاتصال بالمشاركة، ومن هناك يتم فقط مقارنة معرف الجهاز ورقم الدليل المتعلقين من استدعاء النظام stat(). ستؤدي المسارات غير الموجودة إلى رسالة بمستوى تسجيل 0. ملاحظة: يجب استخدام هذا الخيار بحذر لأنه يكسر دلالات SMB! ولكنه قد يساعد في المواقف التي يكون فيها دليل معين (عادةً للقراءة فقط) محل نزاع كبير.

fileid:nolockinode = NUMBER

يشغل هذا الخيار القديم استخدام سلوك fileid nolock للدليل المحدد، مع تجاهل أي معرف جهاز. يمكن استخدام هذا لكسر اتساق القفل للملف أو الدليل المقابل في بيئة التجميع. استخدام خيار fileid:nolock_paths أكثر مرونة وأسهل في الاستخدام.

أمثلة

استخدام وحدة fileid مع خوارزمية fsid:

        [global]
	vfs objects = fileid
	fileid:algorithm = fsid

استخدام وحدة fileid لتجنب التحميل على أدلة (عادةً للقراءة فقط) محل نزاع كبير.

        [global]
	vfs objects = fileid
	fileid:algorithm = next_module
	fileid:nolock_paths = . ContendedFolder1 /path/to/contended.exe
	fileid:nolock_max_slots = 256
