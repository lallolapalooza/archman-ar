الحزمة: dnssec-tools

التراخيص: BSD

val_getdaneinfo(3) دليل المبرمج val_getdaneinfo(3)

الاسم

val_getdaneinfo(3) - إجراء التحقق المتزامن لسجلات TLSA

val_dane_submit(3) - إجراء التحقق غير المتزامن لسجلات TLSA

val_dane_match(3) - التحقق من معلومات TLSA مقابل البيانات المقدمة.

val_dane_check(3) - التحقق من معلومات TLSA لاتصال SSL (OpenSSL فقط)

val_free_dane(3) - تحرير الذاكرة المرتبطة بهيكل نتيجة DANE.

p_dane_error(3) - إرجاع سلسلة الخطأ لرمز خطأ DANE معين.

الملخص

#include <validator/validator.h>
#include <validator/val_dane.h>
int val_getdaneinfo(val_context_t *ctx,
                  const char *name,
                  struct val_daneparams *params,
                  struct val_danestatus **dres);
int val_dane_submit(val_context_t *ctx,
                  const char *name,
                  struct val_daneparams *params,
                  val_dane_callback callback,
                  void *callback_data,
                  val_async_status **status);
int val_dane_match(val_context_t *ctx,
                 struct val_danestatus *dane_cur,
                 const unsigned char *databytes,
                 int databyteslen);
#include <openssl/ssl.h>
int val_dane_check(val_context_t *ctx,
                 SSL *con,
                 struct val_danestatus *danestatus,
                 int *do_pathval);
void val_free_dane(struct val_danestatus *dres);
const char *p_dane_error(int rc);

الوصف

val_getdaneinfo() يجري بحثًا متزامنًا عن سجل TLSA المرتبط بالاسم المحدد name ويعيد قائمة مرتبطة بجميع سجلات المتحقق منها. val_dane_submit() يجري نفس البحث بطريقة غير متزامنة ويستدعي الدالة callback مع الوسائط callback_data عند اكتمال البحث. تعريف الدالة callback هو كالتالي:

typedef int (*val_dane_callback)(void *callback_data,
                               int retval,
                               struct val_danestatus **dres);

يوفر الوسيط status مقبضًا لطلب غير متزامن لتمكين العمليات المستقبلية (مثل إلغاء الطلب). لمزيد من المعلومات حول كائن val_async_status، راجع draft-hayatnagarkar-dnsext-validator-api.

الاسم الفعلي لنظام أسماء النطاقات الذي يمتلك سجل TLSA في DNS له بادئة بالشكل _<port>._<proto>. val_getdaneinfo() سيقوم ببناء البادئة المذكورة أعلاه تلقائيًا؛ لذلك لا ينبغي أن يحتوي قيمة name المقدمة من المستخدم على هذه البادئة. الـ

يجب توفير معلمات البحث عن TLSA في الوسيط params، وهو مؤشر إلى الهيكل التالي:

struct val_daneparams {
  int port;
  int proto;
};

يتم استخدام حقلي port و proto في بناء بادئة اسم TLSA الموصوفة أعلاه.

يتم إرجاع نتائج البحث عن TLSA في الوسيط dres، وهو مؤشر إلى قائمة مرتبطة بالهياكل التالية:

struct val_danestatus {
  long ttl;
  int usage;
  int selector;
  int type;
  size_t datalen;
  unsigned char *data;
  struct val_danestatus *next;
};

الحقل ttl هو وقت البقاء المرتبط بسجل TLSA. يجب على التطبيق عدم تخزين (واستخدام) سجل TLSA هذا بعد انقضاء وقت البقاء الخاص به. المجالات usage و selector و type تتوافق مع المجالات الثلاثة الأولى من RDATA الخاص بـ TLSA كما هو موصوف في rfc6698. يتم إرجاع بيانات ارتباط الشهادة TLSA في الحقل data ولها طول datalen بايت. قد يكون هناك أكثر من سجل TLSA واحد مرتبط باسم معين، ويشير الحقل next إلى السجل التالي في هذه القائمة.

بالنظر إلى قائمة مرتبطة بهياكل TLSA في dres، يمكن استخدام val_dane_match() للتحقق مما إذا كانت بيانات ارتباط الشهادة لعنصر معين في هذه القائمة تتطابق مع البيانات المشفرة بصيغة DER المقدمة في databytes بطول databyteslen.

تقوم الدالة val_dane_check() بتبسيط عملية المطابقة عندما يتم استخدام OpenSSL لتوفير دعم SSL/TLS ضمن التطبيق. تتكرر هذه الدالة تلقائيًا على جميع العناصر في dres وتقارن بيانات ارتباط الشهادة مع شهادات SSL/TLS المرتبطة باتصال SSL con. يمكّن بروتوكول DANE حالات استخدام معينة تسمح بإدخال مراس جدد عبر DNSSEC. تشير قيمة do_pathval إلى ما إذا كان يجب على التطبيق المضي قدمًا في التحقق من مسار X509 لهذا الاتصال وفقًا للاستخدام الذي تم ترميزه في سجل TLSA.

تقوم الدالة val_free_dane() بتحرير الذاكرة المرتبطة بالقائمة المرتبطة التي يشير إليها dres.

الوسيط ctx في جميع الوظائف المذكورة أعلاه يحدد سياق التحقق، والذي يمكن تعيينه إلى NULL للقيم الافتراضية (راجع libval(3) و dnsval.conf لمزيد من التفاصيل حول سياقات التحقق وسياسة التحقق).

قيم الإرجاع

val_getdaneinfo() و val_dane_submit() ترجع VAL_DANE_NOERROR عند النجاح، و VAL_DANE_MALFORMED_TLSA أو VAL_DANE_INTERNAL_ERROR للحالات الخطأ. يتم إرجاع القيمة VAL_DANE_NOTVALIDATED إذا تعذر التحقق من سجل TLSA عبر DNSSEC. يتم إرجاع القيمة VAL_DANE_IGNORE_TLSA إذا كان سجل TLSA للاسم المحدد غائبًا بشكل قاطع.

يمكن أن تحتوي القيمة retval التي تم إرجاعها كوسيط للدالة val_dane_callback() على أحد الخيارات التالية: VAL_DANE_NOERROR (للنجاح)، VAL_DANE_INTERNAL_ERROR (للحالات الخطأ)، أو VAL_DANE_CANCELLED (عند إلغاء الطلب غير المتزامن).

val_dane_match() و val_dane_check() ترجع VAL_DANE_NOERROR عند النجاح، و VAL_DANE_INTERNAL_ERROR للحالات الخطأ العامة، و VAL_DANE_CHECK_FAILED إذا تعذر مطابقة سجل TLSA بنجاح مع بيانات ارتباط الشهادة المقدمة.

يمكن استخدام الدالة p_dane_error() لتحويل رموز الأخطاء المتعلقة بـ DANE إلى قيمة نصية للخطأ.
