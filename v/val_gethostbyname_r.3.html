val_gethostbyname_r(3)

الحزمة: dnssec-tools

التراخيص: BSD

val_gethostbyname(3) دليل المبرمج val_gethostbyname(3)

الاسم

val_gethostbyname(), val_gethostbyname2(), val_gethostbyname_r(), val_gethostbyname2_r() - الحصول على إدخال مضيف شبكة تم التحقق منه بواسطة DNSSEC

الملخص

#include <validator/validator.h>
extern int h_errno;
struct hostent *val_gethostbyname(const val_context_t *ctx,
                                  const char *name,
                                  val_status_t *val_status);
struct hostent *val_gethostbyname2(const val_context_t *ctx,
                                   const char *name,
                                   int af,
                                   val_status_t *val_status);
int val_gethostbyname_r(const val_context_t *ctx,
                        const char *name,
                        struct hostent *ret,
                        char *buf,
                        size_t buflen,
                        struct hostent **result,
                        int *h_errnop,
                        val_status_t *val_status);
int val_gethostbyname2_r(const val_context_t *ctx,
                         const char *name,
                         int af,
                         struct hostent *ret,
                         char *buf,
                         size_t buflen,
                         struct hostent **result,
                         int *h_errnop,
                         val_status_t *val_status);
  struct hostent *val_gethostbyaddr(val_context_t * ctx,
                                    const char *addr,
                                    int len,
                                    int type, 
                                    val_status_t * val_status);
  int val_gethostbyaddr_r(val_context_t * ctx,

                          int type,
                          struct hostent *ret,
                          char *buf,
                          int buflen,
                          struct hostent **result,
                          int *h_errnop,
                          val_status_t * val_status);

الوصف

val_gethostbyname()، val_gethostbyname2()، val_gethostbyname_r()، val_gethostbyname2_r()، val_gethostbyaddr() and val_gethostbyaddr_r() تنفذ التحقق من صحة استعلامات DNS باستخدام DNSSEC. ترجع هذه الدوال إدخال مضيف شبكة من النوع struct hostent وهي إصدارات مدركة لـ DNSSEC للدوال gethostbyname(3)، gethostbyname2(3)، gethostbyname_r()، gethostbyname2_r()، gethostbyaddr() and gethostbyaddr_r() على التوالي. (انظر gethostbyname(3) لمزيد من المعلومات حول النوع struct hostent).

val_gethostbyname()، val_gethostbyname_r()، val_gethostbyaddr()، and val_gethostbyaddr_r() تدعم عناوين IPv4 فقط. val_gethostbyname2() and val_gethostbyname2_r() تدعم كلاً من عناوين IPv4 و IPv6.

الدوال val_gethostbyname_r()، val_gethostbyname2_r() and val_gethostbyaddr_r() هي إصدارات قابلة لإعادة الدخول ويمكن استخدامها بأمان في التطبيقات متعددة الخيوط.

المعامل ctx يحدد سياق التحقق، والذي يمكن تعيينه إلى NULL للقيم الافتراضية (انظر libval(3) and dnsval.conf لمزيد من التفاصيل حول سياقات التحقق وسياسة التحقق).

val_gethostbyname()، val_gethostbyname2() and val_gethostbyaddr() تقوم بتعيين المتغير العام h_errno لإرجاع رمز خطأ المحلل. الإصدارات القابلة لإعادة الدخول val_gethostbyname_r()، val_gethostbyname2_r() and val_gethostbyaddr_r() تستخدم المعامل h_errnop لإرجاع هذه القيمة. هذا يضمن سلامة الخيوط، عن طريق تجنب المتغير العام h_errno. يجب ألا يكون h_errnop NULL. (انظر صفحة الدليل الخاصة بـ gethostbyname(3) للقيم الممكنة لـ h_errno.)

المعاملات name، af، ret، buf، buflen، and result لها نفس الصيغة والدلالات مثل المعاملات المقابلة للدوال الأصلية gethostbyname*() and gethostbyaddr*(). انظر صفحة الدليل الخاصة بـ gethostbyname(3) لمزيد من التفاصيل حول هذه المعاملات.

المعامل val_status يستخدم لإرجاع رمز خطأ المدقق ويجب ألا يكون NULL. يمكن استخدام val_istrusted() and val_isvalidated() لتحديد مدى موثوقية البيانات ويمكن استخدام p_val_status() لعرض قيمة الحالة للمستخدم بتنسيق ASCII (انظر libval(3) لمزيد من المعلومات).

قيم الإرجاع

الدوال val_gethostbyname()، val_gethostbyname2()، and val_gethostbyaddr() ترجع مؤشراً إلى بنية hostent عندما تتمكن من حل اسم المضيف المحدد (مع أو بدون التحقق من DNSSEC)، و NULL إذا لم تكن البيانات متاحة. الذاكرة للقيمة المرجعة مخصصة بشكل ثابت بواسطة هاتين الدالتين. وبالتالي، يجب على المتصل عدم تحرير الذاكرة للقيمة المرجعة.

الدوال val_gethostbyname_r()، val_gethostbyname2_r() and val_gethostbyaddr_r() ترجع 0 عندما تتمكن من حل اسم المضيف المحدد (مع أو بدون التحقق من DNSSEC)، ورمز خطأ غير صفري عند الفشل.

الدوال val_gethostbyaddr() and val_gethostbyaddr_r() ترجع 0 عندما تتمكن من حل اسم المضيف المحدد (مع أو بدون التحقق من DNSSEC)، ورمز خطأ غير صفري عند الفشل.

المعامل val_status يعطي مؤشرًا على موثوقية البيانات. إذا كانت بنية hostent المرجعة NULL، فإن هذه القيمة تعطي مؤشرًا على ما إذا كان عدم وجود البيانات يمكن الوثوق به أم لا.

مثال

#include <stdio.h>
#include <stdlib.h>
#include <validator.h>

   int main(int argc, char *argv[])
   {
        int val_status;
        struct hostent *h = NULL;
        if (argc < 2) {
            printf("Usage: %s <hostname>\n", argv[0]);
            exit(1);
        }
        h = val_gethostbyname(NULL, argv[1], &val_status);
        printf("h_errno = %d [%s]\n", h_errno,
            hstrerror(h_errno));
        if (h) {
            printf("Validation Status = %d [%s]\n", val_status,
                   p_val_status(val_status));
        }
        return 0;
   }

ملاحظات

هذه الدوال لا تقرأ حاليًا ترتيب البحث من /etc/hosts.conf. في الوقت الحالي، يتم تعيين الترتيب الافتراضي لاستشارة ملف /etc/hosts أولاً ثم الاستعلام عن DNS.

الإصدارات الحالية لهذه الدوال لا تدعم عمليات بحث NIS.
