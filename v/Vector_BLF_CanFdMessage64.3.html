uint16_t Vector::BLF::CanFdMessage64::bitCount {}

الطول الكامل للرسالة بالبت. عدد بتات الرسالة.

uint32_t Vector::BLF::CanFdMessage64::btrCfgArb {}

معدل البت المستخدم في مرحلة التحكيم. تكوين توقيت البت CAN أو CAN-FD لمرحلة التحكيم، قد يكون 0 إذا لم يكن مدعومًا بواسطة العتاد/برنامج التشغيل.

  • بت 0-7: تردد الكوارتز
  • بت 8-15: مقسم التردد (Prescaler)
  • بت 16-23: دورات BTL
  • بت 24-31: نقطة العينة

uint32_t Vector::BLF::CanFdMessage64::btrCfgData {}

معدل البت المستخدم في مرحلة البيانات. تكوين توقيت البت CAN-FD لمرحلة البيانات، قد يكون 0 إذا لم يكن مدعومًا بواسطة العتاد/برنامج التشغيل. انظر btrCfgArb.

uint8_t Vector::BLF::CanFdMessage64::channel {}

قناة التطبيق. القناة التي تم إرسال الإطار عليها أو استقباله.

std::vector<uint8_t> Vector::BLF::CanFdMessage64::data {}

بيانات CAN FD. بايتات البيانات (قد يكون الطول الفعلي أقصر من 64 بايت، وفقًا لقيمة dlc).

uint8_t Vector::BLF::CanFdMessage64::dir {}

اتجاه الرسالة.

uint8_t Vector::BLF::CanFdMessage64::dlc {}

CAN dlc. رمز طول البيانات للإطار.

  • DLC=0-8: CAN=0-8, CAN FD=0-8
  • DLC=9: CAN=8, CAN FD=12
  • DLC=10: CAN=8, CAN FD=16
  • DLC=11: CAN=8, CAN FD=20
  • DLC=12: CAN=8, CAN FD=24
  • DLC=13: CAN=8, CAN FD=32
  • DLC=14: CAN=8, CAN FD=48
  • DLC=15: CAN=8, CAN FD=64

uint8_t Vector::BLF::CanFdMessage64::extDataOffset {}
