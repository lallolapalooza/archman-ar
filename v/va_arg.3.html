stdarg(3) دليل وظائف المكتبة stdarg(3)

الاسم

stdarg, va_start, va_arg, va_end, va_copy - قوائم الوسائط المتغيرة

المكتبة

مكتبة C القياسية (libc, -lc)

النبذة

#include <stdarg.h>
void va_start(va_list ap, last);
type va_arg(va_list ap, type);
void va_end(va_list ap);
void va_copy(va_list dest, va_list src);

الوصف

يمكن استدعاء دالة بعدد متغير من الوسائط بأنواع مختلفة. يتضمن ملف الرأس <stdarg.h> نوعًا va_list ويعرّف ثلاث وحدات ماكرو للتقدم خلال قائمة الوسائط التي لا تعرفها الدالة المستدعاة من حيث العدد والأنواع.

يجب أن تعرف الدالة المستدعاة كائنًا من النوع va_list يتم استخدامه بواسطة وحدات الماكرو va_start() و va_arg() و va_end().

va_start()

تقوم وحدة الماكرو va_start() بتهيئة ap للاستخدام اللاحق بواسطة va_arg() و va_end()، ويجب استدعاؤها أولاً.

الوسيط last هو اسم الوسيط الأخير قبل قائمة الوسائط المتغيرة، أي الوسيط الأخير الذي تعرف الدالة المستدعية نوعه.

نظرًا لأنه قد يتم استخدام عنوان هذا الوسيط في وحدة الماكرو va_start()، فلا ينبغي تعريفه كمتغير سجل، أو كنوع دالة أو مصفوفة.

va_arg()

تتوسع وحدة الماكرو va_arg() إلى تعبير له النوع والقيمة للوسيط التالي في الاستدعاء. الوسيط ap هو va_list ap الذي تم تهيئته بواسطة va_start(). كل استدعاء لـ va_arg() يعدل ap بحيث يعيد الاستدعاء التالي الوسيط التالي. الوسيط type هو اسم نوع محدد بحيث يمكن الحصول على نوع مؤشر إلى كائن له النوع المحدد ببساطة عن طريق إضافة * إلى type.

أول استخدام لوحدة الماكرو va_arg() بعد استخدام وحدة الماكرو va_start() يعيد الوسيط الذي يلي last. الاستدعاءات المتتالية تعيد قيم الوسائط المتبقية.

إذا لم يكن هناك وسيط تالٍ، أو إذا كان type غير متوافق مع نوع الوسيط الفعلي التالي (كما تمت ترقيته وفقًا لترقيات الوسائط الافتراضية)، فس تحدث أخطاء عشوائية.

إذا تم تمرير ap إلى دالة تستخدم va_arg(ap,type)، فإن قيمة ap غير معرفة بعد عودة تلك الدالة.

va_end()

يجب أن يقابل كل استدعاء لـ va_start() استدعاء مطابق لـ va_end() في نفس الدالة. بعد الاستدعاء va_end(ap)، يكون المتغير ap غير معرف. يمكن إجراء اجتيازات متعددة للقائمة، يتم تغليف كل منها بواسطة va_start() و va_end(). قد تكون va_end() وحدة ماكرو أو دالة.

va_copy()

تقوم وحدة الماكرو va_copy() بنسخ قائمة الوسائط المتغيرة (المهيأة مسبقًا) src إلى dest. يكون السلوك كما لو تم تطبيق va_start() على dest مع نفس الوسيط last، متبوعًا بنفس عدد استدعاءات va_arg() التي استخدمت للوصول إلى الحالة الحالية لـ src.

تنفيذ واضح قد يجعل va_list مؤشرًا إلى إطار المكدس للدالة المتغيرة. في مثل هذا الإعداد (الأكثر شيوعًا إلى حد بعيد) لا يبدو أن هناك أي مشكلة في التعيين


va_list aq = ap;

لسوء الحظ، هناك أيضًا أنظمة تجعلها مصفوفة من المؤشرات (بطول 1)، وهناك تحتاج إلى


va_list aq;
*aq = *ap;

أخيرًا، في الأنظمة التي يتم فيها تمرير الوسائط في السجلات، قد يكون من الضروري أن تقوم va_start() بتخصيص الذاكرة، وتخزين الوسائط هناك، بالإضافة إلى مؤشر يوضح الوسيط التالي، بحيث يمكن لـ va_arg() الانتقال عبر القائمة. الآن يمكن لـ va_end() تحرير الذاكرة المخصصة مرة أخرى. لاستيعاب هذا الموقف، تضيف C99 وحدة ماكرو va_copy()، بحيث يمكن استبدال التعيين أعلاه بـ


va_list aq;
va_copy(aq, ap);
...
va_end(aq);

يجب أن يقابل كل استدعاء لـ va_copy() استدعاء مطابق لـ va_end() في نفس الدالة. بعض الأنظمة التي لا توفر va_copy() لديها __va_copy بدلاً من ذلك، لأن هذا كان الاسم المستخدم في الاقتراح المسود.

تحذيرات

بخلاف وحدات الماكرو التاريخية varargs، لا تسمح وحدات الماكرو stdarg للمبرمجين بترميز دالة بدون وسائط ثابتة. يولد هذا المشكل عملًا بشكل أساسي عند تحويل كود varargs إلى كود stdarg، ولكنه يخلق أيضًا صعوبات للدوال المتغيرة التي ترغب في تمرير جميع وسائطها إلى دالة تأخذ وسيط va_list، مثل vfprintf(3).

أمثلة

تأخذ الدالة foo سلسلة من أحرف التنسيق وتطبع الوسيط المرتبط بكل حرف تنسيق بناءً على النوع.

#include <stdio.h>
#include <stdarg.h>
void
foo(char *fmt, ...)   /* '...' هو صيغة C لدالة متغيرة */
{
    va_list ap;
    int d;
    char c;
    char *s;
    va_start(ap, fmt);
    while (*fmt)
        switch (*fmt++) {
        case 's':              /* سلسلة نصية */
            s = va_arg(ap, char *);
            printf("string %s\n", s);
            break;
        case 'd':              /* int */
            d = va_arg(ap, int);
            printf("int %d\n", d);
            break;
        case 'c':              /* char */
            /* تحتاج إلى تحويل هنا لأن va_arg
               تأخذ فقط الأنواع التي تمت ترقيتها بالكامل */
            c = (char) va_arg(ap, int);
            printf("char %c\n", c);
            break;
        }
    va_end(ap);
}
