verrc(3bsd)

الحزمة: libbsd

التراخيص: custom

errc(3bsd) 3bsd errc(3bsd)

الاسم

errc، verrc، warnc، vwarnc — رسائل خطأ مهيأة

المكتبة

مكتبة «libbsd»

الملخص

#include <err.h> (انظر libbsd(7) لاستخدام الاستيراد.)
void
errc(int status, int code, const char *fmt, ...);

void
verrc(int status, int code, const char *fmt, va_list args);

void
warnc(int code, const char *fmt, ...);

void
vwarnc(int code, const char *fmt, va_list args);

الوصف

تقوم عائلة الدوال err() و warn() بعرض رسالة خطأ مهيأة على مخرج الخطأ القياسي. في جميع الحالات، يتم إخراج الجزء الأخير من اسم البرنامج، متبوعًا بنقطتين («:») ومسافة. يعتمد النص الذي يتبع ذلك على الدالة التي يتم استدعاؤها. قد تكون مواصفات fmt (والوسائط المرتبطة بها) أي تنسيق مسموح به بواسطة printf(3) أو NULL. إذا لم يكن الوسيط fmt هو NULL، يتم إخراج رسالة الخطأ المهيأة.

تخرج الدوال جميعها رسالة خطأ مرتبطة بقيمة خطأ (انظر strerror(3))، تسبقها نقطتان ومسافة إذا لم يكن fmt هو NULL. أي أن الإخراج يكون كالتالي:

progname: fmt: error message string

إذا لم يكن fmt هو NULL، أو:

progname: error message string

إذا كان كذلك.

يتم استخدام الوسيط code كقيمة خطأ بدلاً من القيمة الحالية للمتغير العام errno.

في جميع الحالات، يتم إخراج سطر جديد بعد الإخراج.

الدوال errc() و verrc() لا تعود، بل تخرج بالقيمة الخاصة بالوسيط status.

أمثلة

عرض معلومات errno الحالية والخروج:

if ((p = malloc(size)) == NULL)
	err(1, NULL);
if ((fd = open(file_name, O_RDONLY, 0)) == -1)
	err(1, "%s", file_name);

عرض رسالة خطأ والخروج:

if (tm.tm_hour < START_TIME)
	errx(1, "too early, wait until %s", start_time_string);

التحذير من خطأ:

if ((fd = open(raw_device, O_RDONLY, 0)) == -1)
	warnx("%s: %s: trying the block device",
	    raw_device, strerror(errno));
if ((fd = open(block_device, O_RDONLY, 0)) == -1)
	err(1, "%s", block_device);

انظر أيضًا

err(3) exit(3)، perror(3)، printf(3)، strerror(3)

تاريخ

ظهرت الدوال errc()، verrc()، warnc()، و vwarnc() لأول مرة في FreeBSD 3.0، NetBSD 7.0، و OpenBSD 5.6.

تحذيرات

من المهم عدم تمرير سلسلة تحتوي على بيانات مقدمة من المستخدم كتنسيق دون استخدام «%s». يمكن للمهاجم وضع محددات تنسيق في السلسلة لتخريب المكدس، مما يؤدي إلى ثغرة أمنية محتملة. ينطبق هذا حتى لو تم بناء السلسلة «يدويًا» باستخدام دالة مثل snprintf()، حيث قد تحتوي السلسلة الناتجة على محددات تحويل خاصة بالمستخدم ليتم تفسيرها لاحقًا بواسطة عائلة دوال err() و warn().

تأكد دائمًا من استخدام الصيغة الآمنة الصحيحة:

errc(1, 0, "%s", string);
