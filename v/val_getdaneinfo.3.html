val_getdaneinfo(3)

الحزمة: dnssec-tools

التراخيص: BSD

val_getdaneinfo(3) دليل المبرمج val_getdaneinfo(3)

الاسم

val_getdaneinfo() - إجراء تحقق متزامن لسجلات TLSA

val_dane_submit() - إجراء تحقق غير متزامن لسجلات TLSA

val_dane_match() - التحقق من معلومات TLSA مقابل البيانات المقدمة.

val_dane_check() - التحقق من معلومات TLSA لاتصال SSL (OpenSSL فقط)

val_free_dane() - تحرير الذاكرة المرتبطة بهيكل نتيجة DANE.

p_dane_error() - إرجاع سلسلة الخطأ لرمز خطأ DANE معين.

الملخص

#include <validator/validator.h>
#include <validator/val_dane.h>
int val_getdaneinfo(val_context_t *ctx,
                  const char *name,
                  struct val_daneparams *params,
                  struct val_danestatus **dres);
int val_dane_submit(val_context_t *ctx,
                  const char *name,
                  struct val_daneparams *params,
                  val_dane_callback callback,
                  void *callback_data,
                  val_async_status **status);
int val_dane_match(val_context_t *ctx,
                 struct val_danestatus *dane_cur,
                 const unsigned char *databytes,
                 int databyteslen);
#include <openssl/ssl.h>
int val_dane_check(val_context_t *ctx,
                 SSL *con,
                 struct val_danestatus *danestatus,
                 int *do_pathval);
void val_free_dane(struct val_danestatus *dres);
const char *p_dane_error(int rc);

الوصف

val_getdaneinfo() تجري بحثًا متزامنًا عن سجل TLSA المرتبط بـ name المعطى وتعيد قائمة مرتبطة بجميع سجلات المدققة هذه. val_dane_submit() تجري البحث نفسه بطريقة غير متزامنة وتستدعي الدالة callback مع وسائط callback_data عند اكتمال البحث. للدالة الراجعة التعريف التالي:

typedef int (*val_dane_callback)(void *callback_data,
                               int retval,
                               struct val_danestatus **dres);

الوسيط status يوفر مقبضًا للطلب غير المتزامن لتمكين المشغلين المستقبليين (مثل إلغاء الطلب). لمزيد من المعلومات حول الكائن val_async_status راجع draft-hayatnagarkar-dnsext-validator-api.

اسم DNS الفعلي الذي يمتلك سجل TLSA في DNS له بادئة بالشكل _\<port>._<proto>. val_getdaneinfo() ستنشئ البادئة المذكورة أعلاه تلقائيًا؛ لذا يجب ألا يحتوي name الذي يقدمه المستخدم على هذه البادئة. الـ

يجب توفير معلمات البحث عن TLSA في الوسيط params، وهو مؤشر إلى الهيكل التالي:

struct val_daneparams {
  int port;
  int proto;
};

تستخدم حقلي port و proto في بناء بادئة اسم TLSA الموضحة أعلاه.

يتم إرجاع نتائج البحث عن TLSA في الوسيط dres، وهو مؤشر إلى قائمة مرتبطة بالهياكل بالشكل أدناه:

struct val_danestatus {
  long ttl;
  int usage;
  int selector;
  int type;
  size_t datalen;
  unsigned char *data;
  struct val_danestatus *next;
};

حقل ttl هو وقت البقاء المرتبط بسجل TLSA. يجب على التطبيق ألا يخزن (ويستخدم) سجل TLSA هذا بعد انتهاء مدة صلاحيته. تتوافق حقلا usage و selector و type مع الحقول الثلاثة الأولى من RDATA الخاص بـ TLSA كما هو موضح في rfc6698. يتم إرجاع بيانات ارتباط الشهادة الخاصة بـ TLSA في حقل data ويبلغ طولها datalen بايت. يمكن أن يكون هناك أكثر من سجل TLSA واحد مرتبط باسم معين، ويشير حقل next إلى السجل التالي في هذه القائمة.

بالنظر إلى قائمة مرتبطة بهياكل TLSA في dres، يمكن استخدام val_dane_match() للتحقق مما إذا كانت بيانات ارتباط الشهادة لعنصر معين في هذه القائمة تتطابق مع البيانات المشفرة بتنسيق DER المقدمة في databytes بطول databyteslen.

تبسط الدالة val_dane_check() عملية المطابقة عند استخدام OpenSSL لتوفير دعم SSL/TLS داخل التطبيق. تتكرر هذه الدالة تلقائيًا عبر جميع العناصر في dres وتقارن بيانات ارتباط الشهادة مع شهادات SSL/TLS المرتبطة باتصال SSL con. يسمح بروتوكول DANE بحالات استخدام معينة تسمح بإدخال مرساة ثقة جديدة عبر DNSSEC. تشير قيمة do_pathval إلى ما إذا كان يجب على التطبيق المتابعة في التحقق من مسار X509 لهذا الاتصال وفقًا للاستخدام الذي تم ترميزه في سجل TLSA.

تقوم الدالة val_free_dane() بتحرير الذاكرة المرتبطة بالقائمة المرتبطة التي يشير إليها dres.

الوسيط ctx في جميع الدوال المذكورة أعلاه يحدد سياق التحقق، والذي يمكن تعيينه إلى NULL للقيم الافتراضية (انظر libval(3) و dnsval.conf لمزيد من التفاصيل حول سياقات التحقق وسياسة التحقق).

قيم الإرجاع

val_getdaneinfo() و val_dane_submit() ترجعان VAL_DANE_NOERROR عند النجاح، و VAL_DANE_MALFORMED_TLSA أو VAL_DANE_INTERNAL_ERROR لحالات الخطأ. يتم إرجاع القيمة VAL_DANE_NOTVALIDATED إذا لم يكن من الممكن التحقق من سجل TLSA عبر DNSSEC. يتم إرجاع القيمة VAL_DANE_IGNORE_TLSA إذا كان سجل TLSA للاسم المعطى غائبًا بشكل قابل للإثبات.

يمكن أن تحتوي القيمة retval التي يتم إرجاعها كوسيط للدالة val_dane_callback() على أحد الخيارات التالية: VAL_DANE_NOERROR (للنجاح)، VAL_DANE_INTERNAL_ERROR (لحالات الخطأ)، أو VAL_DANE_CANCELLED (عند إلغاء الطلب غير المتزامن).

val_dane_match() و val_dane_check() ترجعان VAL_DANE_NOERROR عند النجاح، و VAL_DANE_INTERNAL_ERROR لحالات الخطأ العامة، و VAL_DANE_CHECK_FAILED إذا لم يكن من الممكن مطابقة سجل TLSA بنجاح مع بيانات ارتباط الشهادة المقدمة.

يمكن استخدام الدالة p_dane_error() لتحويل رموز خطأ DANE ذات الصلة إلى قيمة سلسلة خطأ.
