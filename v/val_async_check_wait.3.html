val_async_check_wait(3)

الحزمة: dnssec-tools

التراخيص: BSD

libval_async(3) دليل المبرمج libval_async(3)

الاسم

val_async_submit() - يرسل طلباً للمعالجة غير المتزامنة لاستعلامات DNS.

val_async_select_info() - يحدد واصفات الملفات المناسبة للطلبات غير المتزامنة المعلقة.

val_async_check_wait() - يعالج انتهاء مهلة الاستجابة أو يعالج استجابات DNS لاستعلامات معلقة.

val_async_cancel() - يلغي طلب استعلام غير متزامن.

val_async_cancel_all() - يلغي جميع استعلامات DNS غير المتزامنة لسياق معين.

ملخص

#include <validator/validator.h>

typedef int (*val_async_event_cb)(val_async_status *async_status,
int event, val_context_t *context,
void *cb_data, val_cb_params_t *cbp);

int val_async_submit(val_context_t *context,
const char * name, int class,
int type, unsigned int flags,
val_async_event_cb callback, void *cb_data,
val_async_status **async_status);

int val_async_select_info(val_context_t *context,
fd_set *fds,
int *num_fds,
struct timeval *timeout);

int val_async_check_wait(val_context_t *context,
fd_set *pending_desc, int *nfds,
struct timeval *tv, unsigned int flags);

int val_async_cancel(val_context_t *context,
val_async_status *as,
unsigned int flags);

int val_async_cancel_all(val_context_t *context,
unsigned int flags);

الوصف

تسمح واجهة برمجة تطبيقات التحقق من DNSSEC غير المتزامنة للتطبيق بإرسال طلبات متعددة، والتي يمكن معالجتها بالتوازي. في معظم الحالات، سيؤدي ذلك إلى اكتمال التحقق بشكل أسرع بكثير من سلسلة من الطلبات المتزامنة.

يشير المعلم ctx في الوظائف المختلفة أدناه إلى سياق التحقق، والذي يمكن تعيينه إلى NULL للقيم الافتراضية (راجع libval(3) و dnsval.conf لمزيد من التفاصيل حول سياقات التحقق وسياسة التحقق).

تقوم الوظيفة val_async_submit() بإرسال طلب للمعالجة غير المتزامنة لاستعلامات DNS للبيانات المرتبطة بالاسم name والفئة class والنوع type المعطاة. يقوم الكائن async_status بتحديد طلب معين بشكل فريد ويوفر معالجة للعمليات المستقبلية على هذا الطلب غير المتزامن، بما في ذلك إلغاؤه قبل اكتمال البحث.

يؤثر المعلم flags على متى وكم مرة يتم استدعاء دالة الاستدعاء. الأعلام التالية معرفة.

VAL_AS_IGNORE_CACHE
لا تستخدم أي ذاكرة تخزين مؤقت داخلية للإجابات لهذا الاستعلام.
VAL_AS_NO_NEW_QUERIES
لا ترسل أي استعلامات جديدة. سيتم إرجاع الإجابات من الذاكرة المؤقتة الداخلية.
VAL_AS_NO_ANSWERS
لا يهتم المتصل بنتائج الإجابات. يمكن استخدام هذا لتهيئة الذاكرة المؤقتة.
VAL_AS_NO_CALLBACKS
لا تستدعي أي دوال استدعاء.
VAL_AS_NO_CANCEL_CALLBACKS
استدعاء دوال الاستدعاء مع النتائج، ولكن لا تستدع أي دوال استدعاء عند إلغاء الطلب.
VAL_AS_INTERIM_CALLBACKS
استدعاء دالة الاستدعاء مع النتائج المؤقتة. إذا لم يتم تحديد هذا العلم، فسيتم استدعاء دالة الاستدعاء فقط عندما تكون جميع نتائج التحقق جاهزة.

عندما تصبح نتائج الاستدعاء غير المتزامن متاحة، سيتم استدعاء دالة callback (إذا لم تكن NULL) مع القيمة cb_data، والتي تم توفيرها في الأصل لاستدعاء val_async_submit()، كواحدة من وسائطها. يتم إرجاع نتائج البحث في cb_data، وهو مؤشر إلى بنية val_cb_params_t المعروضة أدناه.

typedef struct val_cb_params_s {
    val_status_t             val_status;
    char                    *name;
    int                      class_h;
    int                      type_h;
    int                      retval;
    struct val_result_chain *results;
    struct val_answer_chain *answers;
} val_cb_params_t;

تحتوي بنية val_cb_params_t على معلمات الاستعلام الأصلية في name و class_h و type_h على التوالي، وقيمة الإرجاع لعملية البحث في retval، ومؤشرات إلى سلاسل results و answers (انظر libval(3) لمزيد من التفاصيل)، وحالة التحقق النهائية لعملية البحث في val_status. يجب على التطبيق تحرير الذاكرة المرتبطة بـ results و answers باستخدام val_free_result_chain() و val_free_answer_chain() على التوالي (انظر libval(3) لمزيد من التفاصيل).

عند اكتمال عملية البحث غير المتزامنة، يتم إرجاع رمز حدث في event. أنواع الأحداث التالية معرفة:

VAL_AS_EVENT_COMPLETED
اكتمل الطلب.
VAL_AS_EVENT_INTERIM
لا يزال الطلب قيد المعالجة، ولكن بعض النتائج المؤقتة متاحة.
VAL_AS_EVENT_CANCELED
تم إلغاء الطلب. أعضاء val_status و results و answers في بنية معامل الاستدعاء غير محددين.

تقوم الوظيفة val_async_select_info() بفحص جميع الطلبات غير المتزامنة المعلقة للسياق المحدد وتعين واصفات الملفات المناسبة، وقيمة المهلة، وقيمة واصف الملف القصوى استعدادًا لاستدعاء select(3).

يتم تعيين واصف الملف لكل مقبس ينتظر استجابة في المعلم fds ويتم تعيين max_fd إلى أعلى رقم واصف ملف لأي طلب غير متزامن معلق ما لم تكن هذه القيمة أقل من القيمة الحالية لـ max_fd، وفي هذه الحالة تظل دون تغيير. يتم تعيين حقل timeout إلى أقل قيمة مهلة لأي مهلة استعلام غير متزامن معلق أقل من القيمة الحالية في هذا الحقل بواسطة التطبيق.

بعد أن يستدعي التطبيق select(3)، يجب عليه أيضًا استدعاء val_async_check_wait() مع fd_set وعدد واصفات الملفات الجاهزة، ndfs، التي تم إرجاعها بواسطة select(). تقوم الوظيفة val_async_check_wait() بمعالجة انتهاء المهلة أو معالجة استجابات DNS للاستعلامات المعلقة. كما تستدعي دوال الاستدعاء للطلبات المكتملة.

توفر val_async_check_wait() وضعين للتشغيل. الأول للاستخدام مع تطبيق له حلقة select() الخاصة به. يقوم التطبيق بتعيين واصفات الملفات الخاصة به، ويستدعي val_async_select_info() لتعيين واصفات الملفات للاستعلامات المعلقة ويستدعي select(). يتم تمرير معلمات fds و nfds من select إلى val_async_check_wait ويتم تجاهل قيمة المهلة. إذا تم معالجة استجابات لواصف ملف، فسيتم مسح واصف الملف المناسب في fds وسيتم إنقاص nfds.

في وضع التشغيل الثاني، يمكن للتطبيق تعيين fds و nfds إلى NULL لتحديد قيمة لـ timeout. هنا، يتم استدعاء val_async_select_info() و select() داخليًا ويتم معالجة أي استجابات تم تلقيها قبل انتهاء صلاحية قيمة المهلة.

يمكن استخدام الوظيفة val_async_cancel() لإلغاء الطلب غير المتزامن المحدد بواسطة مقبضه as، بينما يمكن استخدام val_async_cancel_all() لإلغاء جميع الطلبات غير المتزامنة المرتبطة بسياق معين. يمكن تعيين العلم التالي لطلب الإلغاء.

VAL_AS_CANCEL_NO_CALLBACKS
لا تستدعي دوال الاستدعاء المكتملة أو الملغاة.

قيم الإرجاع

ترجع الوظيفة val_async_submit() القيمة VAL_NO_ERROR عند النجاح وإحدى القيم VAL_RESOURCE_UNAVAILABLE أو VAL_BAD_ARGUMENT أو VAL_INTERNAL_ERROR عند الفشل.

val_async_select_info() تُرجع VAL_NO_ERROR عند النجاح و VAL_BAD_ARGUMENT إذا تم تمرير وسيطة غير قانونية إلى الدالة.

val_async_check_wait() تُرجع 0 عندما لا يتم العثور على طلبات معلقة ورقم صحيح موجب عندما لا تزال الطلبات معلقة. قيمة أقل من الصفر عند حدوث خطأ.

val_async_cancel() و val_async_cancel_all() تُرجع VAL_NO_ERROR عند النجاح.
