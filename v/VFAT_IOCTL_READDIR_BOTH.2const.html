VFAT_IOCTL_READDIR_BOTH(2const) VFAT_IOCTL_READDIR_BOTH(2const)

الاسم

VFAT_IOCTL_READDIR_BOTH, VFAT_IOCTL_READDIR_SHORT - قراءة أسماء الملفات لدليل في نظام ملفات FAT

المكتبة

مكتبة C القياسية (libc، -lc)

ملخص

#include <linux/msdos_fs.h>  /* تعريف الثوابت VFAT_* */
#include <sys/ioctl.h>
int ioctl(int fd, VFAT_IOCTL_READDIR_BOTH,
          struct __fat_dirent entry[2]);
int ioctl(int fd, VFAT_IOCTL_READDIR_SHORT,
          struct __fat_dirent entry[2]);

الوصف

دائمًا ما يكون للملف أو الدليل في نظام ملفات FAT اسم قصير يتكون من 8 أحرف كبيرة كحد أقصى، يتبعه اختياريًا نقطة وما يصل إلى 3 أحرف كبيرة للاحقة الملف. إذا لم يكن اسم الملف الفعلي مناسبًا لهذا المخطط، فسيتم تخزينه كاسم طويل يصل إلى 255 حرف UTF-16.

يمكن قراءة الأسماء القصيرة في دليل باستخدام VFAT_IOCTL_READDIR_SHORT. تقرأ VFAT_IOCTL_READDIR_BOTH كل من الأسماء القصيرة والطويلة.

يجب أن يكون الوسيط fd واصف ملف لدليل. يكفي إنشاء واصف الملف عن طريق استدعاء open(2) باستخدام العلامة O_RDONLY. يمكن استخدام واصف الملف مرة واحدة فقط للتنقل عبر إدخالات الدليل عن طريق استدعاء ioctl(2) بشكل متكرر.

الوسيط entry هو مصفوفة من عنصرين من الهياكل التالية:


struct __fat_dirent {
    long            d_ino;
    __kernel_off_t  d_off;
    uint32_t short  d_reclen;
    char            d_name[256];
};

الإدخال الأول في المصفوفة مخصص للاسم القصير. الإدخال الثاني مخصص للاسم الطويل.

يتم ملء الحقول d_ino و d_off فقط للأسماء الطويلة. يحمل الحقل d_ino رقم العقدة (inode) للدليل. يحمل الحقل d_off إزاحة إدخال الملف في الدليل. نظرًا لأن هذه القيم غير متوفرة للأسماء القصيرة، يجب على كود المستخدم تجاهلها ببساطة.

يحتوي الحقل d_reclen على طول اسم الملف في الحقل d_name. للحفاظ على التوافق مع الإصدارات السابقة، فإن طول 0 للاسم القصير يشير إلى الوصول إلى نهاية الدليل. ومع ذلك، فإن الطريقة المفضلة لاكتشاف نهاية الدليل هي اختبار قيمة الإرجاع لـ ioctl(2). إذا لم يكن هناك اسم طويل، يتم تعيين الحقل d_reclen إلى 0 و d_name هي سلسلة حرفية بطول 0 للاسم الطويل.

قيمة الإرجاع

تشير قيمة الإرجاع 1 إلى أنه تم قراءة إدخال دليل جديد، وتشير قيمة الإرجاع 0 إلى الوصول إلى نهاية الدليل.

عند حدوث خطأ، يتم إرجاع -1، ويتم تعيين errno للإشارة إلى الخطأ.

الأخطاء

ENOENT
fd يشير إلى دليل تم إزالته، ولكنه لا يزال مفتوحًا.
ENOTDIR
fd لا يشير إلى دليل.

المعايير

لينكس.

التاريخ

لينكس 2.0.

أمثلة

يوضح البرنامج التالي استخدام ioctl(2) لإدراج دليل.

ما يلي تم تسجيله عند تطبيق البرنامج على الدليل /mnt/user:


$ ./fat_dir /mnt/user;
. -> ''
.. -> ''
ALONGF~1.TXT -> 'a long filename.txt'
UPPER.TXT -> ''
LOWER.TXT -> 'lower.txt'

مصدر البرنامج

#include <fcntl.h>
#include <linux/msdos_fs.h>
#include <stdio.h>
#include <stdlib.h>
#include <sys/ioctl.h>
#include <unistd.h>
int
main(int argc, char *argv[])
{
    int                  fd;
    int                  ret;
    struct __fat_dirent  entry[2];
    if (argc != 2) {
        printf("Usage: %s DIRECTORY\n", argv[0]);
        exit(EXIT_FAILURE);
    }
    /*
     * فتح واصف الملف للدليل.
     */
    fd = open(argv[1], O_RDONLY | O_DIRECTORY);
    if (fd == -1) {
        perror("open");
        exit(EXIT_FAILURE);
    }
    for (;;) {
        /*
         * قراءة إدخال الدليل التالي.
         */
        ret = ioctl(fd, VFAT_IOCTL_READDIR_BOTH, entry);
        /*
         * إذا حدث خطأ، تكون قيمة الإرجاع -1.
         * إذا تم الوصول إلى نهاية قائمة الدليل،
         * تكون قيمة الإرجاع 0.
         * للحفاظ على التوافق مع الإصدارات السابقة،
         * يتم أيضًا الإشارة إلى نهاية قائمة الدليل بواسطة d_reclen == 0.
         */
        if (ret < 1)
            break;
        /*
         * كتابة الاسم القصير والاسم الطويل.
         */
        printf("%s -> '%s'\n", entry[0].d_name, entry[1].d_name);
    }
    if (ret == -1) {
        perror("VFAT_IOCTL_READDIR_BOTH");
        exit(EXIT_FAILURE);
    }
    /*
     * إغلاق واصف الملف.
     */
    close(fd);
    exit(EXIT_SUCCESS);
}
