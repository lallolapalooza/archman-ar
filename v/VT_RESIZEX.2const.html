ioctl_vt(2) دليل النظام ioctl_vt(2)

الاسم

ioctl_vt - أوامر ioctl للمحطات الطرفية ووحدات التحكم الافتراضية

الملخص

#include <linux/vt.h>  /* تعريف الثوابت VT_* */
#include <sys/ioctl.h>
int ioctl(int fd, unsigned long op, void *argp);

الوصف

عمليات ioctl(2) التالية الخاصة بنظام لينكس مدعومة لمحطات التحكم الطرفية ووحدات التحكم الافتراضية.

VT_OPENQRY
يعيد أول وحدة تحكم متاحة (غير مفتوحة). argp يشير إلى int يتم تعيين قيمته إلى رقم وحدة التحكم الافتراضية (1 <= *argp <= MAX_NR_CONSOLES).
VT_GETMODE
الحصول على وضع وحدة التحكم الافتراضية النشطة. argp يشير إلى

struct vt_mode {
    char  mode;    /* وضع وحدة التحكم الافتراضية */
    char  waitv;   /* إذا تم تعيينه، توقف عند عمليات الكتابة إذا لم تكن نشطة */
    short relsig;  /* الإشارة لرفعها عند عملية التحرير */
    short acqsig;  /* الإشارة لرفعها عند الاستحواذ */
    short frsig;   /* غير مستخدم (تم تعيينه إلى 0) */
};

والتي يتم تعيينها لوضع وحدة التحكم الافتراضية النشطة. يتم تعيين mode إلى أحد هذه القيم:
VT_AUTO التبديل التلقائي لوحدات التحكم الافتراضية
VT_PROCESS التحكم بالتبديل بواسطة العمليات
VT_ACKACQ تأكيد التبديل
VT_SETMODE
تعيين وضع وحدة التحكم الافتراضية النشطة. argp يشير إلى struct vt_mode.
VT_GETSTATE
الحصول على معلومات حالة وحدات التحكم الافتراضية العامة. argp يشير إلى

struct vt_stat {
    unsigned short v_active;  /* وحدة التحكم الافتراضية النشطة */
    unsigned short v_signal;  /* الإشارة المراد إرسالها */
    unsigned short v_state;   /* قناع بت لوحدات التحكم الافتراضية */
};

لكل وحدة تحكم افتراضية قيد الاستخدام، يتم تعيين البت المقابل في عضو v_state. (لينكس 1.0 حتى لينكس 1.1.92).
VT_RELDISP
تحرير شاشة عرض.
VT_ACTIVATE
التبديل إلى وحدة التحكم الافتراضية argp (1 <= argp <= MAX_NR_CONSOLES).
VT_WAITACTIVE
الانتظار حتى يتم تنشيط وحدة التحكم الافتراضية argp.
VT_DISALLOCATE
إلغاء تخصيص الذاكرة المرتبطة بوحدة التحكم الافتراضية argp. (منذ لينكس 1.1.54).
VT_RESIZE
تعيين حجم الشاشة كما يعرفه النواة. argp يشير إلى

struct vt_sizes {
    unsigned short v_rows;       /* عدد الصفوف */
    unsigned short v_cols;       /* عدد الأعمدة */
    unsigned short v_scrollsize; /* لم يعد مستخدماً */
};

لاحظ أن هذا لا يغير وضع الفيديو. انظر resizecons(8). (منذ لينكس 1.1.54).
VT_RESIZEX
تعيين معلمات الشاشة المختلفة كما تعرفها النواة. argp يشير إلى

struct vt_consize {
    unsigned short v_rows;  /* عدد الصفوف */
    unsigned short v_cols;  /* عدد الأعمدة */
    unsigned short v_vlin;  /* عدد صفوف البكسل
                               على الشاشة */
    unsigned short v_clin;  /* عدد صفوف البكسل
                               لكل حرف */
    unsigned short v_vcol;  /* عدد أعمدة البكسل
                               على الشاشة */
    unsigned short v_ccol;  /* عدد أعمدة البكسل
                               لكل حرف */
};

يمكن تعيين أي معلمة إلى الصفر، مما يعني "لا تغيير"، ولكن إذا تم تعيين معلمات متعددة، فيجب أن تكون متسقة ذاتيًا. لاحظ أن هذا لا يغير وضع الفيديو. انظر resizecons(8). (منذ لينكس 1.3.3).
VT_GETCONSIZECSRPOS
الحصول على حجم وحدة التحكم وموضع المؤشر. argp يشير إلى

struct vt_consizecsrpos {
    __u16 con_rows;  /* عدد صفوف وحدة التحكم */
    __u16 con_cols;  /* عدد أعمدة وحدة التحكم */
    __u16 csr_row;   /* صف المؤشر الحالي */
    __u16 csr_col;   /* عمود المؤشر الحالي */
};

(منذ لينكس 6.16).

قيمة الإرجاع

عند النجاح، يتم إرجاع 0 (باستثناء الحالات المشار إليها). عند الفشل، يتم إرجاع -1، ويتم تعيين errno للإشارة إلى الخطأ.

الأخطاء

EINVAL
argp غير صالح.

المعايير

لينكس.

انظر أيضا
