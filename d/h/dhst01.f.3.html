subroutine dhst01 (integer n, integer ilo, integer ihi, double precision, dimension( lda, * ) a, integer lda, double precision, dimension( ldh, * ) h, integer ldh, double precision, dimension( ldq, * ) q, integer ldq, double precision, dimension( lwork ) work, integer lwork, double precision, dimension( 2 ) result)

DHST01

الغرض:

!>
!> DHST01 تختبر اختزال مصفوفة عامة A إلى الشكل الهيسنبرغي العلوي: A = Q*H*Q'.
!> يتم حساب نسبتي اختبار؛
!>
!> RESULT(1) = norm( A - Q*H*Q' ) / ( norm(A) * N * EPS )
!> RESULT(2) = norm( I - Q'*Q ) / ( N * EPS )
!>
!> يُفترض أن المصفوفة Q معطاة صراحة كما ستكون بعد DGEHRD + DORGHR.
!>
!> في هذا الإصدار، لا يتم استخدام ILO و IHI ويفترض أنهما 1 و
!> N على التوالي.
!>

المعاملات

N 
!>          N هو عدد صحيح
!>          رتبة المصفوفة A. N >= 0.
!>

ILO

!>          ILO هو عدد صحيح
!>

IHI

!>          IHI هو عدد صحيح
!>
!>          يُفترض أن A مثلثية عليا في الصفوف والأعمدة
!>          1:ILO-1 و IHI+1:N، لذا تختلف Q عن المصفوفة المطابقة فقط في
!>          الصفوف والأعمدة ILO+1:IHI.
!>

A

!>          A هي مصفوفة ذات دقة مزدوجة، بُعدها (LDA,N)
!>          المصفوفة الأصلية A بحجم n × n.
!>

LDA

!>          LDA هو عدد صحيح
!>          البعد الرئيسي للمصفوفة A. LDA >= max(1,N).
!>

H

!>          H هي مصفوفة ذات دقة مزدوجة، بُعدها (LDH,N)
!>          المصفوفة الهيسنبرغية العلوية H من الاختزال A = Q*H*Q'
!>          كما تم حسابها بواسطة DGEHRD. يُفترض أن H صفر تحت القطر الفرعي الأول.
!>

LDH

!>          LDH هو عدد صحيح
!>          البعد الرئيسي للمصفوفة H. LDH >= max(1,N).
!>

Q

!>          Q هي مصفوفة ذات دقة مزدوجة، بُعدها (LDQ,N)
!>          المصفوفة المتعامدة Q من الاختزال A = Q*H*Q' كما
!>          تم حسابها بواسطة DGEHRD + DORGHR.
!>

LDQ

!>          LDQ هو عدد صحيح
!>          البعد الرئيسي للمصفوفة Q. LDQ >= max(1,N).
!>

WORK

!>          WORK هي مصفوفة ذات دقة مزدوجة، بُعدها (LWORK)
!>

LWORK

!>          LWORK هو عدد صحيح
!>          طول المصفوفة WORK. LWORK >= 2*N*N.
!>

RESULT

!>          RESULT هي مصفوفة ذات دقة مزدوجة، بُعدها (2)
!>          RESULT(1) = norm( A - Q*H*Q' ) / ( norm(A) * N * EPS )
!>          RESULT(2) = norm( I - Q'*Q ) / ( N * EPS )
!>
