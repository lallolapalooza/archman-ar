detect_obj_free(3)

الحزمة: libchardet

التراخيص: MPL

detect_obj_free(3) دليل وظائف المكتبة detect_obj_free(3)

الاسم

chardet_obj_free - تحرير بنية DetectObject

الملخص

#include <chardet.h>
void chardet_obj_free (DetectObj **);

الوصف

تحرير موارد بنية DetectObject التي تم تخصيصها بواسطة واجهة برمجة تطبيقات chardet_obj_init

قيمة الإرجاع

void

مثال

#include <chardet.h>
int main (void) {
	DetectObj *obj;
	if ( (obj = detect_obj_init ()) == NULL ) {
		fprintf (stderr, "فشل تخصيص الذاكرة\n");
		return CHARDET_MEM_ALLOCATED_FAIL;
	}
	switch (detect ("¾È³çÇÏ¼¼¿ä", &obj)) {
		case CHARDET_OUT_OF_MEMORY :
			fprintf (stderr, "أثناء معالجة المقبض، حدث نفاد الذاكرة\n");
			detect_obj_free (&obj);
			return CHARDET_OUT_OF_MEMORY;
		case CHARDET_NULL_OBJECT :
			fprintf (stderr,
					"يجب تخصيص الذاكرة للوسيط الثاني لـ chardet() "
					"باستخدام واجهة برمجة تطبيقات detect_obj_init\n");
			return CHARDET_NULL_OBJECT;
	}
	# تحقق من دعم bom الكائن باستخدام الثابت CHARDET_BOM_CHECK
	printf ("الترميز: %s، الثقة: %f، وجود bom: %d\n", obj->encoding, obj->confidence, obj->bom);
	detect_obj_free (&obj);
    return 0;
}
