detect_destroy(3)

الحزمة: libchardet

التراخيص: MPL

detect_destroy(3) دليل وظائف المكتبة detect_destroy(3)

الاسم

detect_destroy - تحرير بنية Detector

الملخص

#include <chardet.h>

void detect_destroy (Detect ** handle);

الوصف

تحرير الموارد التي تم تخصيصها بواسطة واجهة برمجة تطبيقات detect_init.

قيمة الإرجاع

void

مثال

#include <chardet.h>
int main (void) {
	Detect    * d;
	DetectObj * obj;
	if ( (d = detect_init ()) == NULL ) {
		fprintf (stderr, "فشل تهيئة مؤشر chardet\n");
		return CHARDET_MEM_ALLOCATED_FAIL;
	}
	detect_reset (&d);
	if ( (obj = detect_obj_init ()) == NULL ) {
		fprintf (stderr, "فشل تخصيص الذاكرة\n");
		return CHARDET_MEM_ALLOCATED_FAIL;
	}
	switch (detect_handledata (&d, "안녕하세요", &obj)) {
		case CHARDET_OUT_OF_MEMORY :
			fprintf (stderr, "أثناء معالجة المقبض، حدث نفاد الذاكرة\n");
			detect_obj_free (&obj);
			return CHARDET_OUT_OF_MEMORY;
		case CHARDET_NULL_OBJECT :
			fprintf (stderr,
					"يجب أن يكون الوسيط الثالث لـ chardet_handledata() مخصصًا للذاكرة "
					"باستخدام واجهة برمجة تطبيقات detect_obj_init\n");
			return CHARDET_NULL_OBJECT;
	}
	# check support obj->bom with CHARDET_BOM_CHECK constant
	printf ("التشفير: %s، الثقة: %f، وجود bom: %d\n", obj->encoding, obj->confidence, obj->bom);
	detect_obj_free (&obj);
	detect_destroy (&d);
    return 0;
}
