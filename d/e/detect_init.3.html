الحزمة: libchardet

التراخيص: MPL

detect_init(3) دليل وظائف المكتبة detect_init(3)

الاسم

detect_init - تهيئة بنية Detect

ملخص

#include <chardet.h>
Detect * chardet_init (void);

الوصف

تهيئة بنية Detect.

قيمة الإرجاع

تعيد NULL عند الفشل، وتعيد بنية Detect عند النجاح.

typedef struct Detect_t {
	Detector *detect;
} Detect;

مثال

#include <chardet.h>
int main (void) {
	Detect    * d;
	DetectObj * obj;
	if ( (d = detect_init ()) == NULL ) {
		fprintf (stderr, "فشل تهيئة معالج chardet\n");
		return CHARDET_MEM_ALLOCATED_FAIL;
	}
	detect_reset (&d);
	if ( (obj = detect_obj_init ()) == NULL ) {
		fprintf (stderr, "فشل تخصيص الذاكرة\n");
		return CHARDET_MEM_ALLOCATED_FAIL;
	}
	switch (detect_handledata (&d, "¾È³çÇÏ¼¼¿ä", &obj)) {
		case CHARDET_OUT_OF_MEMORY :
			fprintf (stderr, "حدث نقص في الذاكرة عند معالجة المعالج\n");
			detect_obj_free (&obj);
			return CHARDET_OUT_OF_MEMORY;
		case CHARDET_NULL_OBJECT :
			fprintf (stderr,
					"يجب أن يكون الوسيط الثاني لدالة chardet() مخصصًا للذاكرة "
					"باستخدام واجهة برمجة التطبيقات detect_obj_init\n");
			return CHARDET_NULL_OBJECT;
	}
	# التحقق من دعم obj->bom باستخدام الثابت CHARDET_BOM_CHECK
	printf ("الترميز: %s، الثقة: %f، وجود bom: %d\n", obj->encoding, obj->confidence, obj->bom);
	detect_obj_free (&obj);
	detect_destroy (&d);
    return 0;
}
