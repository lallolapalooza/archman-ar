Devel__Cover__DB(3pm)

الحزمة: perl-devel-cover

التراخيص: PerlArtistic, GPL

Devel::Cover::DB(3) وثائق Perl التي ساهم بها المستخدم Devel::Cover::DB(3)

الاسم

Devel::Cover::DB - مقاييس تغطية الكود لـ Perl

الإصدار

الإصدار 1.51

الملخص

use Devel::Cover::DB;
my $db = Devel::Cover::DB->new(db => "my_coverage_db");
$db->print_summary([$file1, $file2], ["statement", "pod"]);

الوصف

توفر هذه الوحدة الوصول إلى قاعدة بيانات معلومات تغطية الكود.

الأساليب

جديد

my $db = Devel::Cover::DB->new(db => "my_coverage_db");

ينشئ قاعدة البيانات من قاعدة البيانات المحددة.

تغطية

my $cover = $db->cover;

تُرجع كائن Devel::Cover::DB::Cover. من هنا يمكن الوصول إلى جميع بيانات التغطية.

my $cover = $db->cover;
for my $file ($cover->items) {
    print "$file\n";
    my $f = $cover->file($file);
    for my $criterion ($f->items) {
        print "  $criterion\n";
        my $c = $f->criterion($criterion);
        for my $location ($c->items) {
            my $l = $c->location($location);
            print "    $location @$l\n";
        }
    }
}

ستكون البيانات للمعايير المختلفة بتنسيقات مختلفة، لذا سيتطلب ذلك معالجة خاصة. لم يتم توثيق هذا بعد، لذا فإن أفضل رهان لك في الوقت الحالي هو النظر في بعض التقارير الأبسط و/أو المصدر.

الأساليب في المثال أعلاه هي في الواقع أسماء مستعارة للأساليب في Devel::Cover::DB::Base (الفئة الأساسية لجميع فئات Devel::Cover::DB::*):

  • Devel::Cover::DB::Base->values

    اسم مستعار لـ Devel::Cover::DB::Cover->files، Devel::Cover::DB::File->criteria، Devel::Cover::DB::Criterion->locations، و Devel::Cover::DB::Location->data

  • Devel::Cover::DB::Base->get

    اسم مستعار لـ Devel::Cover::DB::Cover->file، Devel::Cover::DB::File->criterion، Devel::Cover::DB::Criterion->location، و Devel::Cover::DB::Location->datum

بدلاً من استدعاء $file->criterion("x") يمكنك أيضًا استدعاء $file->x.

هل_صالح

my $valid = $db->is_valid;

تُرجع صحيحًا إذا كان $db صالحًا (أو يبدو صالحًا، الوظيفة متساهلة جدًا).
