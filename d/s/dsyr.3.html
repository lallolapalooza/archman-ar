subroutine dsyr (character uplo, integer n, double precision alpha, double precision, dimension(*) x, integer incx, double precision, dimension(lda,*) a, integer lda)

DSYR

الغرض:

!>
!> DSYR   يقوم بعملية الرتبة المتماثلة 1
!>
!>    A := alpha*x*x**T + A,
!>
!> حيث alpha هو عدد قياسي حقيقي، x هو متجه من n عنصر، و A هي مصفوفة
!> متماثلة بحجم n × n.
!>

المعلمات

UPLO 
!>          UPLO هو CHARACTER*1
!>           عند الإدخال، يحدد UPLO ما إذا كان الجزء العلوي أو السفلي
!>           المثلثي من المصفوفة A سيتم الرجوع إليه على النحو التالي:
!>
!>              UPLO = 'U' أو 'u'   سيتم الرجوع فقط إلى الجزء العلوي
!>                                  المثلثي من A.
!>
!>              UPLO = 'L' أو 'l'   سيتم الرجوع فقط إلى الجزء السفلي
!>                                  المثلثي من A.
!>

N

!>          N هو INTEGER
!>           عند الإدخال، يحدد N رتبة المصفوفة A.
!>           يجب أن يكون N على الأقل صفرًا.
!>

ALPHA

!>          ALPHA هو DOUBLE PRECISION.
!>           عند الإدخال، يحدد ALPHA العدد القياسي alpha.
!>

X

!>          X هي مصفوفة DOUBLE PRECISION، بحجم لا يقل عن
!>           ( 1 + ( n - 1 )*abs( INCX ) ).
!>           قبل الدخول، يجب أن تحتوي المصفوفة X المزودة بزيادة على المتجه
!>           x ذي العناصر n.
!>

INCX

!>          INCX هو INTEGER
!>           عند الإدخال، يحدد INCX الزيادة لعناصر X.
!>           يجب ألا يكون INCX صفرًا.
!>

A

!>          A هي مصفوفة DOUBLE PRECISION، بحجم ( LDA, N )
!>           قبل الدخول مع UPLO = 'U' أو 'u'، يجب أن يحتوي الجزء العلوي
!>           المثلثي من المصفوفة A الذي بحجم n × n على الجزء العلوي
!>           المثلثي من المصفوفة المتماثلة، ولا يتم الرجوع إلى الجزء السفلي
!>           المثلثي بدقة من A. عند الخروج، يتم استبدال الجزء العلوي
!>           المثلثي من المصفوفة A بالجزء العلوي المثلثي من المصفوفة المحدثة.
!>           قبل الدخول مع UPLO = 'L' أو 'l'، يجب أن يحتوي الجزء السفلي
!>           المثلثي من المصفوفة A الذي بحجم n × n على الجزء السفلي
!>           المثلثي من المصفوفة المتماثلة، ولا يتم الرجوع إلى الجزء العلوي
!>           المثلثي بدقة من A. عند الخروج، يتم استبدال الجزء السفلي
!>           المثلثي من المصفوفة A بالجزء السفلي المثلثي من المصفوفة المحدثة.
!>

LDA

!>          LDA هو INTEGER
!>           عند الإدخال، يحدد LDA البعد الأول للمصفوفة A كما هو معلن
!>           في البرنامج (الفرعي) المستدعي. يجب أن يكون LDA على الأقل
!>           max( 1, n ).
!>
