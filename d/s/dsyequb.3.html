subroutine dsyequb (character uplo, integer n, double precision, dimension( lda, * ) a, integer lda, double precision, dimension( * ) s, double precision scond, double precision amax, double precision, dimension( * ) work, integer info)

DSYEQUB

الغرض:

!>
!> DSYEQUB تحسب معاملات القياس للصف والأعمدة بهدف موازنة مصفوفة متناظرة A
!> (بالنسبة للمعيار الإقليدي) وتقليل رقم شرطها. يتم حساب معاملات القياس S
!> بواسطة خوارزمية BIN (انظر المراجع) بحيث تكون المصفوفة الموسعة B ذات العناصر
!> B(i,j) = S(i)*A(i,j)*S(j) لها رقم شرط ضمن عامل N من أصغر رقم شرط ممكن
!> فوق جميع عمليات القياس القطرية الممكنة.
!>

المعلمات

UPLO 
!>          UPLO هو CHARACTER*1
!>          = 'U':  يتم تخزين الجزء العلوي من A;
!>          = 'L':  يتم تخزين الجزء السفلي من A.
!>

N

!>          N هو INTEGER
!>          رتبة المصفوفة A. N >= 0.
!>

A

!>          A هي مصفوفة DOUBLE PRECISION، أبعادها (LDA,N)
!>          المصفوفة المتناظرة N×N التي يتم حساب معاملات القياس لها.
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرئيسي للمصفوفة A. LDA >= max(1,N).
!>

S

!>          S هي مصفوفة DOUBLE PRECISION، أبعادها (N)
!>          إذا كانت INFO = 0، فإن S تحتوي على معاملات القياس لـ A.
!>

SCOND

!>          SCOND هو DOUBLE PRECISION
!>          إذا كانت INFO = 0، فإن S تحتوي على نسبة أصغر S(i) إلى
!>          أكبر S(i). إذا كان SCOND >= 0.1 ولم يكن AMAX كبيرًا جدًا أو صغيرًا جدًا،
!>          فلا داعي للقياس باستخدام S.
!>

AMAX

!>          AMAX هو DOUBLE PRECISION
!>          أكبر قيمة مطلقة لأي عنصر في المصفوفة. إذا كان AMAX قريبًا جدًا
!>          من تجاوز الحد الأقصى أو قريبًا جدًا من تجاوز الحد الأدنى،
!>          فيجب قياس المصفوفة.
!>

WORK

!>          WORK هي مصفوفة DOUBLE PRECISION، أبعادها (2*N)
!>

INFO

!>          INFO هو INTEGER
!>          = 0:  انتهاء ناجح
!>          < 0:  إذا كانت INFO = -i، فإن المعلمة i كانت ذات قيمة غير قانونية
!>          > 0:  إذا كانت INFO = i، فإن العنصر القطري i غير موجب.
!>
