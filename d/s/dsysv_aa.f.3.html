subroutine dsysv_aa (character uplo, integer n, integer nrhs, double precision, dimension( lda, * ) a, integer lda, integer, dimension( * ) ipiv, double precision, dimension( ldb, * ) b, integer ldb, double precision, dimension( * ) work, integer lwork, integer info)

DSYSV_AA يحسب حل نظام المعادلات الخطية A * X = B لمصفوفات SY

الغرض:

!>
!> DSYSV يحسب حل نظام المعادلات الخطية الحقيقي A * X = B،
!> حيث A مصفوفة متناظرة بحجم N*N و X و B مصفوفتان بحجم N*NRHS.
!>
!> يتم استخدام خوارزمية Aasen لتحليل A كـ
!> A = U**T * T * U، إذا كان UPLO = 'U'، أو
!> A = L * T * L**T، إذا كان UPLO = 'L'،
!> حيث U (أو L) هي حاصل ضرب مصفوفات التبديل والمصفوفات المثلثية العليا (السفلى) الأحادية،
!> و T هي مصفوفة متناظرة ثلاثية القطر. ثم يتم استخدام الشكل المحلل لـ A لحل نظام المعادلات A * X = B.
!>

المعاملات

UPLO 
!>          UPLO هو CHARACTER*1
!>          = 'U':  يتم تخزين المثلث العلوي لـ A؛
!>          = 'L':  يتم تخزين المثلث السفلي لـ A؛
!>

N

!>          N هو INTEGER
!>          عدد المعادلات الخطية، أي رتبة المصفوفة A. N >= 0.
!>

NRHS

!>          NRHS هو INTEGER
!>          عدد الأطراف اليمنى، أي عدد أعمدة المصفوفة B. NRHS >= 0.
!>

A

!>          A هي مصفوفة DOUBLE PRECISION، بحجم (LDA,N)
!>          عند الدخول، المصفوفة المتناظرة A. إذا كان UPLO = 'U'، فإن الجزء العلوي
!>          المثلثي من A بحجم N*N يحتوي على الجزء العلوي المثلثي للمصفوفة A،
!>          ولا يتم الرجوع إلى الجزء السفلي المثلثي الصارم لـ A. إذا كان UPLO = 'L'،
!>          فإن الجزء السفلي المثلثي من A بحجم N*N يحتوي على الجزء السفلي المثلثي للمصفوفة A،
!>          ولا يتم الرجوع إلى الجزء العلوي المثلثي الصارم لـ A.
!>
!>          عند الخروج، إذا كان INFO = 0، فإن المصفوفة ثلاثية القطر T والمضاعفات
!>          المستخدمة للحصول على العامل U أو L من التحليل A = U**T*T*U أو A = L*T*L**T
!>          كما تم حسابها بواسطة DSYTRF.
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرائد للمصفوفة A. LDA >= max(1,N).
!>

IPIV

!>          IPIV هي مصفوفة INTEGER، بحجم (N)
!>          عند الخروج، تحتوي على تفاصيل التبادلات، أي أنه تم تبديل الصف والعمود k
!>          لـ A مع الصف والعمود IPIV(k).
!>

B

!>          B هي مصفوفة DOUBLE PRECISION، بحجم (LDB,NRHS)
!>          عند الدخول، المصفوفة B ذات الطرف الأيمن بحجم N*NRHS.
!>          عند الخروج، إذا كان INFO = 0، فإن مصفوفة الحل X بحجم N*NRHS.
!>

LDB

!>          LDB هو INTEGER
!>          البعد الرائد للمصفوفة B. LDB >= max(1,N).
!>

WORK

!>          WORK هي مصفوفة DOUBLE PRECISION، بحجم (MAX(1,LWORK))
!>          عند الخروج، إذا كان INFO = 0، فإن WORK(1) تعيد قيمة LWORK المثلى.
!>

LWORK

!>          LWORK هو INTEGER
!>          طول WORK. LWORK >= MAX(1,2*N,3*N-2)، ولأفضل أداء، LWORK >= MAX(1,N*NB)،
!>          حيث NB هو حجم الكتلة الأمثل لـ DSYTRF_AA.
!>
!>          إذا كان LWORK = -1، فإنه يُفترض استعلام عن مساحة العمل؛
!>          تقوم الإجراءات الفرعية فقط بحساب الحجم الأمثل لمصفوفة WORK،
!>          وتعيد هذه القيمة كمدخل أول لمصفوفة WORK، ولا يتم إصدار أي رسالة خطأ
!>          متعلقة بـ LWORK بواسطة XERBLA.
!>

INFO

!>          INFO هو INTEGER
!>          = 0: خروج ناجح
!>          < 0: إذا كان INFO = -i، فإن المعامل i كان له قيمة غير قانونية
!>          > 0: إذا كان INFO = i، فإن D(i,i) هو صفر تمامًا.
!>               لقد اكتمل التحليل، لكن المصفوفة القطرية الفرعية D مفردة تمامًا،
!!>               لذلك لم يكن بالإمكان حساب الحل.
!>
