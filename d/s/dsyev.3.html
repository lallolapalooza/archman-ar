subroutine dsyev (jobz, uplo, n, a, lda, w, work, lwork, info)
DSYEV يحسب القيم الذاتية، اختياريًا، المتجهات الذاتية اليسرى و/أو اليمنى لمصفوفات SY

توثيق الوظيفة/الروتين الفرعي

subroutine dsyev (character jobz, character uplo, integer n, double precision, dimension( lda, * ) a, integer lda, double precision, dimension( * ) w, double precision, dimension( * ) work, integer lwork, integer info)

DSYEV يحسب القيم الذاتية، اختياريًا، المتجهات الذاتية اليسرى و/أو اليمنى لمصفوفات SY

الغرض:

!>
!> DSYEV يحسب جميع القيم الذاتية، واختياريًا، المتجهات الذاتية لمصفوفة
!> متناظرة حقيقية A.
!>

المعلمات

JOBZ 
!>          JOBZ is CHARACTER*1
!>          = 'N':  حساب القيم الذاتية فقط؛
!>          = 'V':  حساب القيم الذاتية والمتجهات الذاتية.
!>

UPLO

!>          UPLO is CHARACTER*1
!>          = 'U':  يتم تخزين الجزء العلوي من A؛
!>          = 'L':  يتم تخزين الجزء السفلي من A.
!>

N

!>          N is INTEGER
!>          رتبة المصفوفة A.  N >= 0.
!>

A

!>          A is DOUBLE PRECISION array, dimension (LDA, N)
!>          عند الدخول، المصفوفة المتناظرة A.  إذا كان UPLO = 'U'،
!>          فإن الجزء العلوي الذي يمتد من N-by-N في A يحتوي على
!>          الجزء العلوي من المصفوفة A.  إذا كان UPLO = 'L'،
!>          فإن الجزء السفلي الذي يمتد من N-by-N في A يحتوي على
!>          الجزء السفلي من المصفوفة A.
!>          عند الخروج، إذا كان JOBZ = 'V'، ثم إذا كان INFO = 0، فإن A تحتوي على
!>          المتجهات الذاتية المتعامدة للمصفوفة A.
!>          إذا كان JOBZ = 'N'، فإنه عند الخروج، يتم تدمير المثلث السفلي (إذا كان UPLO='L')
!>          أو المثلث العلوي (إذا كان UPLO='U') من A، بما في ذلك
!>          القطر.
!>

LDA

!>          LDA is INTEGER
!>          البعد الرائد للمصفوفة A.  LDA >= max(1,N).
!>

W

!>          W is DOUBLE PRECISION array, dimension (N)
!>          إذا كان INFO = 0، فإن القيم الذاتية مرتبة تصاعديًا.
!>

WORK

!>          WORK is DOUBLE PRECISION array, dimension (MAX(1,LWORK))
!>          عند الخروج، إذا كان INFO = 0، فإن WORK(1) تعيد LWORK المثلى.
!>

LWORK

!>          LWORK is INTEGER
!>          طول المصفوفة WORK.  LWORK >= max(1,3*N-1).
!>          للكفاءة المثلى، LWORK >= (NB+2)*N،
!>          حيث NB هو حجم الكتلة لـ DSYTRD الذي يتم إرجاعه بواسطة ILAENV.
!>
!>          إذا كان LWORK = -1، فإنه يُفترض استعلام مساحة العمل؛ يقوم الروتين
!>          بحساب حجم مصفوفة WORK الأمثل فقط، ويعيد
!>          هذه القيمة كإدخال أول لمصفوفة WORK، ولا يصدر رسالة خطأ
!>          متعلقة بـ LWORK بواسطة XERBLA.
!>

INFO

!>          INFO is INTEGER
!>          = 0:  خروج ناجح
!>          < 0:  إذا كان INFO = -i، فإن المعلمة i كانت لها قيمة غير قانونية
!>          > 0:  إذا كان INFO = i، فشل الخوارزمية في التقارب؛ i
!>                لم تتقارب عناصر خارج القطر في شكل ثلاثي الأقطار وسيط
!>                إلى الصفر.
!>
