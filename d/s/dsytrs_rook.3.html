توثيق الوظيفة/الإجراء الفرعي

subroutine dsytrs_rook (character uplo, integer n, integer nrhs, double precision, dimension( lda, * ) a, integer lda, integer, dimension( * ) ipiv, double precision, dimension( ldb, * ) b, integer ldb, integer info)

DSYTRS_ROOK

الغرض:

!>
!> DSYTRS_ROOK يحل نظام المعادلات الخطية A*X = B
!> بمصفوفة متناظرة حقيقية A باستخدام التحليل A = U*D*U**T أو
!> A = L*D*L**T المحسوبة بواسطة DSYTRF_ROOK.
!>

المعلمات

UPLO 
!>          UPLO is CHARACTER*1
!>          يحدد ما إذا كانت تفاصيل التحليل مخزنة
!>          كمصفوفة مثلثة عليا أو سفلى.
!>          = 'U':  مثلثة عليا، الشكل هو A = U*D*U**T;
!>          = 'L':  مثلثة سفلى، الشكل هو A = L*D*L**T.
!>

N

!>          N is INTEGER
!>          رتبة المصفوفة A.  N >= 0.
!>

NRHS

!>          NRHS is INTEGER
!>          عدد الأطراف اليمنى، أي عدد أعمدة
!>          المصفوفة B.  NRHS >= 0.
!>

A

!>          A is DOUBLE PRECISION array, dimension (LDA,N)
!>          المصفوفة القطرية الفرعية D والمضاعفات المستخدمة
!>          للحصول على العامل U أو L كما تم حسابه بواسطة DSYTRF_ROOK.
!>

LDA

!>          LDA is INTEGER
!>          البعد الرئيسي للمصفوفة A.  LDA >= max(1,N).
!>

IPIV

!>          IPIV is INTEGER array, dimension (N)
!>          تفاصيل التبادلات والبنية الفرعية لـ D
!>          كما تم تحديدها بواسطة DSYTRF_ROOK.
!>

B

!>          B is DOUBLE PRECISION array, dimension (LDB,NRHS)
!>          عند الدخول، مصفوفة الطرف الأيمن B.
!>          عند الخروج، مصفوفة الحل X.
!>

LDB

!>          LDB is INTEGER
!>          البعد الرئيسي للمصفوفة B.  LDB >= max(1,N).
!>

INFO

!>          INFO is INTEGER
!>          = 0:  خروج ناجح
!>          < 0:  إذا كانت INFO = -i، فقد كان المعامل i ذا قيمة غير قانونية
!>
