subroutine dsyt01_3 (character uplo, integer n, double precision, dimension( lda, * ) a, integer lda, double precision, dimension( ldafac, * ) afac, integer ldafac, double precision, dimension( * ) e, integer, dimension( * ) ipiv, double precision, dimension( ldc, * ) c, integer ldc, double precision, dimension( * ) rwork, double precision resid)

DSYT01_3

الغرض:

!>
!> DSYT01_3 تعيد بناء مصفوفة متماثلة غير محددة A من تحليلها ذو
!> عامل L*D*L' أو U*D*U' المحسوب بواسطة DSYTRF_RK
!> (أو DSYTRF_BK) وتحسب الباقي
!>    norm( C - A ) / ( N * norm(A) * EPS ),
!> حيث C هي المصفوفة المعاد بناؤها و EPS هو خطأ الآلة.
!>

المعلمات

UPLO 
!>          UPLO هو CHARACTER*1
!>          يحدد ما إذا كان الجزء المثلثي العلوي أو السفلي من
!>          المصفوفة المتماثلة A مخزنًا:
!>          = 'U':  مثلثي علوي
!>          = 'L':  مثلثي سفلي
!>

N

!>          N هو INTEGER
!>          عدد الصفوف والأعمدة للمصفوفة A. N >= 0.
!>

A

!>          A هو مصفوفة DOUBLE PRECISION، ذات أبعاد (LDA,N)
!>          المصفوفة المتماثلة الأصلية A.
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرئيسي للمصفوفة A. LDA >= max(1,N)
!>

AFAC

!>          AFAC هو مصفوفة DOUBLE PRECISION، ذات أبعاد (LDAFAC,N)
!>          قطر المصفوفة القطرية المتوازية D والعوامل U أو L
!>          المحسوبة بواسطة DSYTRF_RK و DSYTRF_BK:
!>            أ) فقط العناصر القطرية للمصفوفة القطرية المتوازية المتماثلة
!>               D على قطر A، أي D(k,k) = A(k,k)؛
!>               (يجب توفير عناصر فوق القطر (أو تحت القطر) للمصفوفة D
!>                عند الإدخال في المصفوفة E)، و
!>            ب) إذا كان UPLO = 'U': العامل U في الجزء العلوي من القطر A.
!>               إذا كان UPLO = 'L': العامل L في الجزء السفلي من القطر A.
!>

LDAFAC

!>          LDAFAC هو INTEGER
!>          البعد الرئيسي للمصفوفة AFAC.
!>          LDAFAC >= max(1,N).
!>

E

!>          E هي مصفوفة DOUBLE PRECISION، ذات أبعاد (N)
!>          عند الإدخال، تحتوي على العناصر فوق القطر (أو تحت القطر)
!>          للمصفوفة القطرية المتوازية المتماثلة D
!>          مع كتل قطرية بحجم 1x1 أو 2x2، حيث
!>          إذا كان UPLO = 'U': E(i) = D(i-1,i),i=2:N, E(1) لا تتم الإشارة إليه؛
!>          إذا كان UPLO = 'L': E(i) = D(i+1,i),i=1:N-1, E(N) لا تتم الإشارة إليه.
!>

IPIV

!>          IPIV هي مصفوفة INTEGER، ذات أبعاد (N)
!>          مؤشرات المحور من DSYTRF_RK (أو DSYTRF_BK).
!>

C

!>          C هي مصفوفة DOUBLE PRECISION، ذات أبعاد (LDC,N)
!>

LDC

!>          LDC هو INTEGER
!>          البعد الرئيسي للمصفوفة C. LDC >= max(1,N).
!>

RWORK

!>          RWORK هي مصفوفة DOUBLE PRECISION، ذات أبعاد (N)
!>

RESID

!>          RESID هو DOUBLE PRECISION
!>          إذا كان UPLO = 'L'، norm(L*D*L' - A) / ( N * norm(A) * EPS )
!>          إذا كان UPLO = 'U'، norm(U*D*U' - A) / ( N * norm(A) * EPS )
!>
