subroutine dsyev_2stage (character jobz, character uplo, integer n, double precision, dimension( lda, * ) a, integer lda, double precision, dimension( * ) w, double precision, dimension( * ) work, integer lwork, integer info)

DSYEV_2STAGE تحسب القيم الذاتية، وبشكل اختياري، المتجهات الذاتية اليسرى و/أو اليمنى لمصفوفات SY

الغرض:

!>
!> DSYEV_2STAGE تحسب جميع القيم الذاتية، وبشكل اختياري، المتجهات الذاتية
!> لمصفوفة متماثلة حقيقية A باستخدام تقنية المرحلتين (2stage)
!> للاختزال إلى مصفوفة ثلاثية الأقطار.
!>

المعلمات

JOBZ 
!>          JOBZ is CHARACTER*1
!>          = 'N':  حساب القيم الذاتية فقط؛
!>          = 'V':  حساب القيم الذاتية والمتجهات الذاتية.
!>                  غير متاح في هذا الإصدار.
!>

UPLO

!>          UPLO is CHARACTER*1
!>          = 'U':  يتم تخزين الجزء العلوي من A؛
!>          = 'L':  يتم تخزين الجزء السفلي من A.
!>

N

!>          N is INTEGER
!>          رتبة المصفوفة A. N >= 0.
!>

A

!>          A is DOUBLE PRECISION array, dimension (LDA, N)
!>          عند الدخول، المصفوفة المتماثلة A. إذا كان UPLO = 'U'،
!>          يحتوي الجزء العلوي من A بحجم N*N على الجزء العلوي
!>          من المصفوفة A. إذا كان UPLO = 'L'، يحتوي الجزء السفلي
!>          من A بحجم N*N على الجزء السفلي من المصفوفة A.
!>          عند الخروج، إذا كان JOBZ = 'V'، وإذا كان INFO = 0،
!>          تحتوي A على المتجهات الذاتية المتعامدة للمصفوفة A.
!>          إذا كان JOBZ = 'N'، فعند الخروج، يتم تدمير الجزء السفلي
!>          (إذا كان UPLO='L') أو الجزء العلوي (إذا كان UPLO='U')
!>          من A، بما في ذلك القطر الرئيسي.
!>

LDA

!>          LDA is INTEGER
!>          البعد الرائد للمصفوفة A. LDA >= max(1,N).
!>

W

!>          W is DOUBLE PRECISION array, dimension (N)
!>          إذا كان INFO = 0، فإن القيم الذاتية مرتبة تصاعدياً.
!>

WORK

!>          WORK is DOUBLE PRECISION array, dimension (MAX(1,LWORK))
!>          عند الخروج، إذا كان INFO = 0، فإن WORK(1) يعيد LWORK المثلى.
!>

LWORK

!>          LWORK is INTEGER
!>          طول المصفوفة WORK. LWORK >= 1، عندما N <= 1؛
!>          وإلا
!>          إذا كان JOBZ = 'N' و N > 1، يجب الاستعلام عن LWORK.
!>                                   LWORK = MAX(1, dimension) حيث
!>                                   dimension = max(stage1,stage2) + (KD+1)*N + 2*N
!>                                             = N*KD + N*max(KD+1,FACTOPTNB)
!>                                               + max(2*KD*KD, KD*NTHREADS)
!>                                               + (KD+1)*N + 2*N
!>                                   حيث KD هو حجم التجميع للاختزال،
!>                                   FACTOPTNB هو التجميع المستخدم بواسطة خوارزمية QR أو LQ
!>                                   عادةً ما يكون FACTOPTNB=128 اختيارًا جيدًا
!>                                   NTHREADS هو عدد الخيوط المستخدمة عند
!>                                   تمكين تجميع openMP، وإلا =1.
!>          إذا كان JOBZ = 'V' و N > 1، يجب الاستعلام عن LWORK. غير متاح بعد
!>
!>          إذا كان LWORK = -1، يفترض استعلام مساحة العمل؛ تقوم الروتين
!>          بحساب حجم مساحة العمل المثلى فقط، وتعيده
!>          كإدخال أول في مصفوفة WORK، ولا يتم إصدار أي رسالة خطأ
!>          متعلقة بـ LWORK بواسطة XERBLA.
!>

INFO

!>          INFO is INTEGER
!>          = 0:  انتهاء ناجح
!>          < 0:  إذا كان INFO = -i، فإن المعامل i كان له قيمة غير قانونية
!>          > 0:  إذا كان INFO = i، فشلت الخوارزمية في التقارب؛ i
!>                العناصر خارج القطر الرئيسي للشكل الثلاثي الأقطار الوسيط
!>                لم تتقارب إلى الصفر.
!>
