subroutine dsbmv (character uplo, integer n, integer k, double precision alpha, double precision, dimension(lda,*) a, integer lda, double precision, dimension(*) x, integer incx, double precision beta, double precision, dimension(*) y, integer incy)

DSBMV

الغرض:

!>
!> DSBMV  يقوم بعملية المصفوفة-المتجه
!>
!>    y := alpha*A*x + beta*y,
!>
!> حيث alpha و beta عبارة عن قيم قياسية، x و y هما متجهان بعناصر n،
!> و A عبارة عن مصفوفة نطاق متناظرة بحجم n × n، مع k نطاقات علوية.
!>

المعاملات

UPLO 
!>          UPLO هو CHARACTER*1
!>           عند الإدخال، يحدد UPLO ما إذا كان الجزء العلوي أو السفلي
!>           من المصفوفة الشريطية A يتم توفيره كما يلي:
!>
!>              UPLO = 'U' أو 'u'   يتم توفير الجزء العلوي من A.
!>
!>              UPLO = 'L' أو 'l'   يتم توفير الجزء السفلي من A.
!>

N

!>          N هو INTEGER
!>           عند الإدخال، يحدد N رتبة المصفوفة A.
!>           يجب أن يكون N على الأقل صفر.
!>

K

!>          K هو INTEGER
!>           عند الإدخال، يحدد K عدد النطاقات العلوية للمصفوفة A.
!>           يجب أن يحقق K  0 .le. K.
!>

ALPHA

!>          ALPHA هو DOUBLE PRECISION.
!>           عند الإدخال، يحدد ALPHA القيمة القياسية alpha.
!>

A

!>          A هو مصفوفة DOUBLE PRECISION، أبعادها ( LDA, N )
!>           قبل الإدخال مع UPLO = 'U' أو 'u'، يجب أن يحتوي الجزء الرائد ( k + 1 )
!>           × N من المصفوفة A على الجزء العلوي من المصفوفة الشريطية المتناظرة،
!>           يتم توفيره عمودًا بعمود، مع القطر الرئيسي للمصفوفة في الصف
!>           ( k + 1 ) من المصفوفة، والنطاق العلوي الأول يبدأ من الموضع 2 في الصف k، وهكذا.
!>           يتم تجاهل الجزء المثلثي العلوي الأيسر بحجم k × k من المصفوفة A.
!>           سيقوم مقطع البرنامج التالي بنقل الجزء العلوي من مصفوفة نطاق متناظرة
!>           من التخزين الكامل التقليدي إلى تخزين النطاق:
!>
!>                 DO 20, J = 1, N
!>                    M = K + 1 - J
!>                    DO 10, I = MAX( 1, J - K ), J
!>                       A( M + I, J ) = matrix( I, J )
!>              10    CONTINUE
!>              20 CONTINUE
!>
!>           قبل الإدخال مع UPLO = 'L' أو 'l'، يجب أن يحتوي الجزء الرائد ( k + 1 )
!>           × N من المصفوفة A على الجزء السفلي من المصفوفة الشريطية المتناظرة،
!>           يتم توفيره عمودًا بعمود، مع القطر الرئيسي للمصفوفة في الصف 1
!>           من المصفوفة، والنطاق الفرعي الأول يبدأ من الموضع 1 في الصف 2، وهكذا.
!>           يتم تجاهل الجزء المثلثي السفلي الأيمن بحجم k × k من المصفوفة A.
!>           سيقوم مقطع البرنامج التالي بنقل الجزء السفلي من مصفوفة نطاق متناظرة
!>           من التخزين الكامل التقليدي إلى تخزين النطاق:
!>
!>                 DO 20, J = 1, N
!>                    M = 1 - J
!>                    DO 10, I = J, MIN( N, J + K )
!>                       A( M + I, J ) = matrix( I, J )
!>              10    CONTINUE
!>              20 CONTINUE
!>

LDA

!>          LDA هو INTEGER
!>           عند الإدخال، يحدد LDA البعد الأول لـ A كما تم تعريفه
!>           في البرنامج المستدعي (الفرعي). يجب أن يكون LDA على الأقل
!>           ( k + 1 ).
!>

X

!>          X هو مصفوفة DOUBLE PRECISION، أبعادها على الأقل
!>           ( 1 + ( n - 1 )*abs( INCX ) ).
!>           قبل الإدخال، يجب أن تحتوي المصفوفة X المتزايدة على
!>           المتجه x.
!>

INCX

!>          INCX هو INTEGER
!>           عند الإدخال، يحدد INCX الزيادة لعناصر X.
!>           يجب ألا يكون INCX صفرًا.
!>

BETA

!>          BETA هو DOUBLE PRECISION.
!>           عند الإدخال، يحدد BETA القيمة القياسية beta.
!>

Y

!>          Y هو مصفوفة DOUBLE PRECISION، أبعادها على الأقل
!>           ( 1 + ( n - 1 )*abs( INCY ) ).
!>           قبل الإدخال، يجب أن تحتوي المصفوفة Y المتزايدة على
!>           المتجه y. عند الخروج، يتم استبدال y بالمتجه y المحدث.
!>

INCY

!>          INCY هو INTEGER
!>           عند الإدخال، يحدد INCY الزيادة لعناصر Y.
!>           يجب ألا يكون INCY صفرًا.
!>
