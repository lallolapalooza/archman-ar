subroutine dsyequb (character uplo, integer n, double precision, dimension( lda, * ) a, integer lda, double precision, dimension( * ) s, double precision scond, double precision amax, double precision, dimension( * ) work, integer info)

DSYEQUB

الغرض:

!>
!> DSYEQUB تحسب عوامل القياس للصفوف والأعمدة المخصصة لمصفوفة متماثلة A
!> (فيما يتعلق بمعيار إقليدس) وتقلل من عدد حالاتها المشروطة. يتم حساب عوامل القياس S
!> بواسطة خوارزمية BIN (انظر المراجع) بحيث تكون المصفوفة المقاسة B ذات العناصر
!> B(i,j) = S(i)*A(i,j)*S(j) لها عدد حالات مشروطة في حدود عامل N
!> من أصغر عدد حالات مشروطة ممكنة على جميع القياسات القطرية الممكنة.
!>

المعلمات

UPLO 
!>          UPLO هو CHARACTER*1
!>          = 'U':  يتم تخزين الجزء العلوي من A؛
!>          = 'L':  يتم تخزين الجزء السفلي من A؛
!>

N

!>          N هو INTEGER
!>          رتبة المصفوفة A. N >= 0.
!>

A

!>          A هو مصفوفة DOUBLE PRECISION، أبعادها (LDA,N)
!>          المصفوفة المتماثلة N×N التي يتم حساب عوامل القياس الخاصة بها.
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرئيسي للمصفوفة A. LDA >= max(1,N).
!>

S

!>          S هو مصفوفة DOUBLE PRECISION، أبعادها (N)
!>          إذا كان INFO = 0، تحتوي S على عوامل القياس لـ A.
!>

SCOND

!>          SCOND هو DOUBLE PRECISION
!>          إذا كان INFO = 0، تحتوي S على نسبة أصغر S(i) إلى
!>          أكبر S(i). إذا كان SCOND >= 0.1 وكان AMAX ليس كبيرًا جدًا
!>          ولا صغيرًا جدًا، فلا داعي للقياس بـ S.
!>

AMAX

!>          AMAX هو DOUBLE PRECISION
!>          أكبر قيمة مطلقة لأي عنصر في المصفوفة. إذا كان AMAX قريبًا جدًا
!>          من فيضان أو قريبًا جدًا من تدفق سفلي، فيجب قياس المصفوفة.
!>

WORK

!>          WORK هو مصفوفة DOUBLE PRECISION، أبعادها (2*N)
!>

INFO

!>          INFO هو INTEGER
!>          = 0:  إنهاء ناجح
!>          < 0:  إذا كان INFO = -i، فإن المعلمة i كانت لها قيمة غير قانونية
!>          > 0:  إذا كان INFO = i، فإن العنصر القطري i غير موجب.
!>
