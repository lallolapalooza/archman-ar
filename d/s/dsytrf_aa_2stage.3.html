وثائق الوظيفة/الوحدة الفرعية

subroutine dsytrf_aa_2stage (character uplo, integer n, double precision, dimension( lda, * ) a, integer lda, double precision, dimension( * ) tb, integer ltb, integer, dimension( * ) ipiv, integer, dimension( * ) ipiv2, double precision, dimension( * ) work, integer lwork, integer info)

DSYTRF_AA_2STAGE

الغرض:

!>
!> DSYTRF_AA_2STAGE تحسب تحليل مصفوفة متناظرة حقيقية A
!> باستخدام خوارزمية Aasen. شكل التحليل هو
!>
!>    A = U**T*T*U  أو  A = L*T*L**T
!>
!> حيث U (أو L) هو حاصل ضرب مصفوفات التبديل والمصفوفات المثلثية العليا (السفلى)
!> والوحدات، و T هي مصفوفة نطاق متناظرة ذات نطاق NB (يتم تحديد NB داخليًا وتخزينه في TB( 1 )، و T
!> يتم تحليلها LU مع تحديد المحور الجزئي).
!>
!> هذا هو الإصدار المجمّع من الخوارزمية، ويستدعي BLAS المستوى 3.
!>

المعلمات

UPLO 
!>          UPLO هو CHARACTER*1
!>          = 'U':  يتم تخزين المثلث العلوي من A؛
!>          = 'L':  يتم تخزين المثلث السفلي من A.
!>

N

!>          N هو INTEGER
!>          ترتيب المصفوفة A.  N >= 0.
!>

A

!>          A هي مصفوفة DOUBLE PRECISION، أبعادها (LDA,N)
!>          عند الإدخال، المصفوفة المتناظرة A.  إذا كانت UPLO = 'U'، فإن الجزء المثلثي العلوي
!>          الرئيسي من A بقياس N*N يحتوي على الجزء المثلثي العلوي
!>          من المصفوفة A، ولا يتم الرجوع إلى الجزء المثلثي السفلي البحت
!>          من A. إذا كانت UPLO = 'L'، فإن الجزء المثلثي السفلي
!>          الرئيسي من A بقياس N*N يحتوي على الجزء المثلثي السفلي
!>          من المصفوفة A، ولا يتم الرجوع إلى الجزء المثلثي العلوي البحت
!>          من A.
!>
!>          عند الخروج، يتم تخزين L أسفل (أو أعلى) الكتل الفرعية،
!>          عندما تكون UPLO = 'L' (أو 'U').
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرئيسي للمصفوفة A.  LDA >= max(1,N).
!>

TB

!>          TB هي مصفوفة DOUBLE PRECISION، أبعادها (MAX(1,LTB))
!>          عند الخروج، تفاصيل تحليل LU للمصفوفة النطاقية.
!>

LTB

!>          LTB هو INTEGER
!>          حجم المصفوفة TB. LTB >= MAX(1,4*N)، مستخدمة داخليًا
!>          لتحديد NB بحيث LTB >= (3*NB+1)*N.
!>
!>          إذا كانت LTB = -1، فإنه يفترض استعلام مساحة العمل؛
!>          تقوم الوحدة الفرعية فقط بحساب الحجم الأمثل لـ LTB،
!>          وتعيد هذا الحجم كالإدخال الأول لـ TB،
!>          ولا يتم إصدار أي رسالة خطأ متعلقة بـ LTB بواسطة XERBLA.
!>

IPIV

!>          IPIV هي مصفوفة INTEGER، أبعادها (N)
!>          عند الخروج، تحتوي على تفاصيل التبادلات، أي
!>          تم تبديل الصف والعمود k من A مع
!>          الصف والعمود IPIV(k).
!>

IPIV2

!>          IPIV2 هي مصفوفة INTEGER، أبعادها (N)
!>          عند الخروج، تحتوي على تفاصيل التبادلات، أي
!>          تم تبديل الصف والعمود k من T مع
!>          الصف والعمود IPIV2(k).
!>

WORK

!>          WORK هي مساحة عمل DOUBLE PRECISION بحجم (MAX(1,LWORK))
!>

LWORK

!>          LWORK هو INTEGER
!>          حجم WORK. LWORK >= MAX(1,N)، مستخدم داخليًا
!>          لتحديد NB بحيث LWORK >= N*NB.
!>
!>          إذا كانت LWORK = -1، فإنه يفترض استعلام مساحة العمل؛
!>          تقوم الوحدة الفرعية فقط بحساب الحجم الأمثل لمصفوفة WORK،
!>          وتعيد هذا الحجم كالإدخال الأول لمصفوفة WORK،
!>          ولا يتم إصدار أي رسالة خطأ متعلقة بـ LWORK بواسطة XERBLA.
!>

INFO

!>          INFO هو INTEGER
!>          = 0:  انتهاء ناجح
!>          < 0:  إذا كانت INFO = -i، فإن الوسيط i كان له قيمة غير قانونية.
!>          > 0:  إذا كانت INFO = i، فشل تحليل LU النطاقي في العمود i
!>
