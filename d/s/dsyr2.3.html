subroutine dsyr2 (character uplo, integer n, double precision alpha, double precision, dimension(*) x, integer incx, double precision, dimension(*) y, integer incy, double precision, dimension(lda,*) a, integer lda)

DSYR2

الغرض:

!>
!> DSYR2  يقوم بعملية الرتبة 2 المتماثلة
!>
!>    A := alpha*x*y**T + alpha*y*x**T + A,
!>
!> حيث alpha هو قيمة قياسية، x و y هما متجهان يحتويان على n عنصر، و A هي مصفوفة متماثلة بحجم n × n.
!>

المعلمات

UPLO 
!>          UPLO هو CHARACTER*1
!>           عند الإدخال، تحدد UPLO ما إذا كان الجزء المثلثي العلوي أو السفلي من المصفوفة A سيتم الإشارة إليه على النحو التالي:
!>
!>              UPLO = 'U' أو 'u'   يتم الإشارة فقط إلى الجزء المثلثي العلوي من A.
!>
!>              UPLO = 'L' أو 'l'   يتم الإشارة فقط إلى الجزء المثلثي السفلي من A.
!>

N

!>          N هو INTEGER
!>           عند الإدخال، تحدد N رتبة المصفوفة A.
!>           يجب أن تكون N على الأقل صفر.
!>

ALPHA

!>          ALPHA هو DOUBLE PRECISION.
!>           عند الإدخال، تحدد ALPHA القيمة القياسية alpha.
!>

X

!>          X هو مصفوفة DOUBLE PRECISION، بحجم لا يقل عن
!>           ( 1 + ( n - 1 )*abs( INCX ) ).
!>           قبل الإدخال، يجب أن تحتوي المصفوفة X التي تم زيادة عناصرها على المتجه x المكون من n عنصر.
!>

INCX

!>          INCX هو INTEGER
!>           عند الإدخال، تحدد INCX الزيادة لعناصر X. يجب ألا تكون INCX صفرًا.
!>

Y

!>          Y هو مصفوفة DOUBLE PRECISION، بحجم لا يقل عن
!>           ( 1 + ( n - 1 )*abs( INCY ) ).
!>           قبل الإدخال، يجب أن تحتوي المصفوفة Y التي تم زيادة عناصرها على المتجه y المكون من n عنصر.
!>

INCY

!>          INCY هو INTEGER
!>           عند الإدخال، تحدد INCY الزيادة لعناصر Y. يجب ألا تكون INCY صفرًا.
!>

A

!>          A هي مصفوفة DOUBLE PRECISION، بحجم ( LDA, N )
!>           قبل الإدخال مع UPLO = 'U' أو 'u'، يجب أن يحتوي الجزء العلوي المثلثي الرائد بحجم n × n من المصفوفة A على الجزء العلوي المثلثي من المصفوفة المتماثلة، ولا يتم الإشارة إلى الجزء السفلي المثلثي الصارم من A. عند الخروج، يتم استبدال الجزء العلوي المثلثي من المصفوفة A بالجزء العلوي المثلثي من المصفوفة المحدثة.
!>           قبل الإدخال مع UPLO = 'L' أو 'l'، يجب أن يحتوي الجزء السفلي المثلثي الرائد بحجم n × n من المصفوفة A على الجزء السفلي المثلثي من المصفوفة المتماثلة، ولا يتم الإشارة إلى الجزء العلوي المثلثي الصارم من A. عند الخروج، يتم استبدال الجزء السفلي المثلثي من المصفوفة A بالجزء السفلي المثلثي من المصفوفة المحدثة.
!>

LDA

!>          LDA هو INTEGER
!>           عند الإدخال، تحدد LDA البعد الأول للمصفوفة A كما هو معلن في البرنامج (الفرعي) المستدعي. يجب أن تكون LDA على الأقل
!>           max( 1, n ).
!>
