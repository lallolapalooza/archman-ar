توثيق الوظيفة/الروتين الفرعي

subroutine dsytri (character uplo, integer n, double precision, dimension( lda, * ) a, integer lda, integer, dimension( * ) ipiv, double precision, dimension( * ) work, integer info)

DSYTRI

الغرض:

!>
!> DSYTRI تحسب معكوس مصفوفة حقيقية متماثلة غير محددة
!> A باستخدام التحليل A = U*D*U**T أو A = L*D*L**T الذي تم حسابه بواسطة
!> DSYTRF.
!>

المعلمات

UPLO 
!>          UPLO هو CHARACTER*1
!>          يحدد ما إذا كانت تفاصيل التحليل مخزنة
!>          كمصفوفة مثلثة علوية أو سفلية.
!>          = 'U':  مثلثة علوية، الشكل هو A = U*D*U**T؛
!>          = 'L':  مثلثة سفلية، الشكل هو A = L*D*L**T.
!>

N

!>          N هو INTEGER
!>          رتبة المصفوفة A.  N >= 0.
!>

A

!>          A هي مصفوفة DOUBLE PRECISION، البعد (LDA,N)
!>          عند الإدخال، المصفوفة القطرية المتوازية D والمعاملات
!>          المستخدمة للحصول على العامل U أو L الذي تم حسابه بواسطة DSYTRF.
!>
!>          عند الخروج، إذا كان INFO = 0، فإن معكوس المصفوفة الأصلية (المتماثل)
!>          هو. إذا كان UPLO = 'U'، يتم تشكيل الجزء العلوي المثلثي من المعكوس
!>          ولا يتم الرجوع إلى الجزء من A أسفل القطر؛ إذا كان UPLO = 'L' يتم تشكيل
!>          الجزء السفلي المثلثي من المعكوس ولا يتم الرجوع إلى الجزء من A فوق
!>          القطر.
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرائد للمصفوفة A.  LDA >= max(1,N).
!>

IPIV

!>          IPIV هي مصفوفة INTEGER، البعد (N)
!>          تفاصيل التبادلات وهيكل D المتوازي
!>          كما تم تحديده بواسطة DSYTRF.
!>

WORK

!>          WORK هي مصفوفة DOUBLE PRECISION، البعد (N)
!>

INFO

!>          INFO هو INTEGER
!>          = 0: الخروج بنجاح
!>          < 0: إذا كان INFO = -i، فإن المعلمة i-th كانت لها قيمة غير قانونية
!>          > 0: إذا كان INFO = i، D(i,i) = 0؛ المصفوفة مفردة ولا يمكن حساب معكوسها.
!>
