subroutine dsyt22 (integer itype, character uplo, integer n, integer m, integer kband, double precision, dimension( lda, * ) a, integer lda, double precision, dimension( * ) d, double precision, dimension( * ) e, double precision, dimension( ldu, * ) u, integer ldu, double precision, dimension( ldv, * ) v, integer ldv, double precision, dimension( * ) tau, double precision, dimension( * ) work, double precision, dimension( 2 ) result)

DSYT22

الغرض:

!>
!>      DSYT22  عموماً يتحقق من تحليل للشكل
!>
!>              A U = U S
!>
!>      حيث A متماثل، أعمدة U متعامدة، و S قطري (إذا كان KBAND=0)
!>      أو متماثل ثلاثي الأقطار (إذا كان KBAND=1). إذا كان ITYPE=1،
!>      فإن U ممثلة كمصفوفة كثيفة، وإلا فإن U يتم التعبير عنها كناتج
!>      لتحويلات هاوسهولدر، التي يتم تخزين متجهاتاتها في المصفوفة
!>       وثوابتها القياسية في
!>      سنستخدم الحرف
!>      للإشارة إلى ناتج تحويلات هاوسهولدر
!>      (والتي يجب أن تكون مساوية لـ U).
!>
!>      على وجه التحديد، إذا كان ITYPE=1، فإن:
!>
!>              RESULT(1) = | U**T A U - S | / ( |A| m ulp ) و
!>              RESULT(2) = | I - U**T U | / ( m ulp )
!> 
!>  ITYPE   INTEGER
!>          يحدد نوع الاختبارات التي سيتم إجراؤها.
!>          1: U ممثلة كمصفوفة متعامدة كثيفة:
!>             RESULT(1) = | A - U S U**T | / ( |A| n ulp ) و
!>             RESULT(2) = | I - U U**T | / ( n ulp )
!>
!>  UPLO    CHARACTER
!>          إذا كان UPLO='U'، فسيتم استخدام المثلث العلوي من A ولن
!>          يتم الرجوع إلى المثلث السفلي (الصارم). إذا كان UPLO='L'،
!>          فسيتم استخدام المثلث السفلي من A ولن يتم الرجوع إلى المثلث
!>          العلوي (الصارم).
!>          لا يتم تعديله.
!>
!>  N       INTEGER
!>          حجم المصفوفة. إذا كان صفرًا، فلن يفعل DSYT22 شيئًا.
!>          يجب أن يكون على الأقل صفرًا.
!>          لا يتم تعديله.
!>
!>  M       INTEGER
!>          عدد أعمدة U. إذا كان صفرًا، فلن يفعل DSYT22 شيئًا.
!>          يجب أن يكون على الأقل صفرًا.
!>          لا يتم تعديله.
!>
!>  KBAND   INTEGER
!>          عرض النطاق للمصفوفة. قد يكون صفرًا أو واحدًا فقط.
!>          إذا كان صفرًا، فإن S قطري، ولا يتم الرجوع إلى E. إذا كان
!>          واحدًا، فإن S متماثل ثلاثي الأقطار.
!>          لا يتم تعديله.
!>
!>  A       DOUBLE PRECISION array, dimension (LDA , N)
!>          المصفوفة الأصلية (غير المحللة). يُفترض أنها متماثلة، وسيتم
!>          الرجوع فقط إلى الجزء العلوي (UPLO='U') أو فقط الجزء السفلي
!>          (UPLO='L').
!>          لا يتم تعديله.
!>
!>  LDA     INTEGER
!>          البعد الرائد لـ A. يجب أن يكون على الأقل 1 وعلى الأقل N.
!>          لا يتم تعديله.
!>
!>  D       DOUBLE PRECISION array, dimension (N)
!>          القطر للمصفوفة (المتماثلة ثلاثية الأقطار).
!>          لا يتم تعديله.
!>
!>  E       DOUBLE PRECISION array, dimension (N)
!>          القطر غير القطري للمصفوفة (المتماثلة ثلاثية الأقطار).
!>          يتم تجاهل E(1)، E(2) هو العنصر (1,2) و (2,1)، إلخ.
!>          لا يتم الرجوع إليه إذا كان KBAND=0.
!>          لا يتم تعديله.
!>
!>  U       DOUBLE PRECISION array, dimension (LDU, N)
!>          إذا كان ITYPE=1 أو 3، فإن هذه تحتوي على المصفوفة المتعامدة
!>          في التحليل، ممثلة كمصفوفة كثيفة. إذا كان ITYPE=2،
!>          فلا يتم الرجوع إليها.
!>          لا يتم تعديله.
!>
!>  LDU     INTEGER
!>          البعد الرائد لـ U. يجب أن يكون LDU على الأقل N وعلى الأقل 1.
!>          لا يتم تعديله.
!>
!>  V       DOUBLE PRECISION array, dimension (LDV, N)
!>          إذا كان ITYPE=2 أو 3، فإن المثلث السفلي لهذه المصفوفة يحتوي
!>          على متجهات هاوسهولدر التي تصف المصفوفة المتعامدة في التحليل.
!>          إذا كان ITYPE=1، فلا يتم الرجوع إليها.
!>          لا يتم تعديله.
!>
!>  LDV     INTEGER
!>          البعد الرائد لـ V. يجب أن يكون LDV على الأقل N وعلى الأقل 1.
!>          لا يتم تعديله.
!>
!>  TAU     DOUBLE PRECISION array, dimension (N)
!>          إذا كان ITYPE >= 2، فإن TAU(j) هو العامل القياسي لـ
!>          v(j) v(j)**T في تحويل هاوسهولدر H(j) للناتج
!!>          U = H(1)...H(n-2)
!>          إذا كان ITYPE < 2، فلا يتم الرجوع إلى TAU.
!>          لا يتم تعديله.
!>
!>  WORK    DOUBLE PRECISION array, dimension (2*N**2)
!>          مساحة عمل.
!>          يتم تعديلها.
!>
!>  RESULT  DOUBLE PRECISION array, dimension (2)
!>          القيم التي تم حسابها بواسطة الاختبارين الموصوفين أعلاه.
!>          القيم محدودة حاليًا بـ 1/ulp، لتجنب تجاوز السعة.
!>          يتم تعديل RESULT(1) دائمًا. يتم تعديل RESULT(2) فقط
!>          إذا كان LDU على الأقل N.
!>          يتم تعديلها.
!>
