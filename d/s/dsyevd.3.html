DSYEVD يحسب القيم الذاتية، اختياريًا، المتجهات الذاتية اليسرى و/أو اليمنى لمصفوفات SY

توثيق الوظيفة/الروتين الفرعي

subroutine dsyevd (character jobz, character uplo, integer n, double precision, dimension( lda, * ) a, integer lda, double precision, dimension( * ) w, double precision, dimension( * ) work, integer lwork, integer, dimension( * ) iwork, integer liwork, integer info)

DSYEVD يحسب القيم الذاتية، اختياريًا، المتجهات الذاتية اليسرى و/أو اليمنى لمصفوفات SY

الغرض:

!>
!> DSYEVD يحسب جميع القيم الذاتية، واختياريًا، المتجهات الذاتية لمصفوفة
!> متماثلة حقيقية A. إذا كانت المتجهات الذاتية مطلوبة، فإنه يستخدم خوارزمية
!> الانقسام والغزو.
!>
!> بسبب الاستخدام الكبير لـ BLAS من المستوى 3، تحتاج DSYEVD إلى مساحة عمل
!> N**2 أكثر من DSYEVX.
!>

المعلمات

JOBZ 
!>          JOBZ هو CHARACTER*1
!>          = 'N':  حساب القيم الذاتية فقط؛
!>          = 'V':  حساب القيم الذاتية والمتجهات الذاتية.
!>

UPLO

!>          UPLO هو CHARACTER*1
!>          = 'U':  يتم تخزين المثلث العلوي لـ A؛
!>          = 'L':  يتم تخزين المثلث السفلي لـ A.
!>

N

!>          N هو INTEGER
!>          رتبة المصفوفة A.  N >= 0.
!>

A

!>          A هي مصفوفة DOUBLE PRECISION، أبعادها (LDA, N)
!>          عند الدخول، المصفوفة المتماثلة A.  إذا كانت UPLO = 'U'، فإن
!>          الجزء العلوي المثلثي العلوي لـ A من الحجم N-by-N يحتوي على
!>          الجزء العلوي المثلثي للمصفوفة A.  إذا كانت UPLO = 'L'، فإن
!>          الجزء السفلي المثلثي السفلي لـ A من الحجم N-by-N يحتوي على
!>          الجزء السفلي المثلثي للمصفوفة A.
!>          عند الخروج، إذا كانت JOBZ = 'V'، وإذا كانت INFO = 0، فإن A تحتوي على
!>          المتجهات الذاتية المتعامدة للمصفوفة A.
!>          إذا كانت JOBZ = 'N'، فإن عند الخروج، يتم تدمير المثلث السفلي (إذا كانت UPLO='L')
!>          أو المثلث العلوي (إذا كانت UPLO='U') لـ A، بما في ذلك القطر.
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرائد للمصفوفة A.  LDA >= max(1,N).
!>

W

!>          W هي مصفوفة DOUBLE PRECISION، أبعادها (N)
!>          إذا كانت INFO = 0، فإن القيم الذاتية مرتبة تصاعديًا.
!>

WORK

!>          WORK هي مصفوفة DOUBLE PRECISION، أبعادها (MAX(1,LWORK))
!>          عند الخروج، إذا كانت INFO = 0، فإن WORK(1) تعيد LWORK المثلى.
!>

LWORK

!>          LWORK هو INTEGER
!>          بعد المصفوفة WORK.
!>          إذا كانت N <= 1،               يجب أن يكون LWORK على الأقل 1.
!>          إذا كانت JOBZ = 'N' و N > 1، يجب أن يكون LWORK على الأقل 2*N+1.
!>          إذا كانت JOBZ = 'V' و N > 1، يجب أن يكون LWORK على الأقل
!>                                                1 + 6*N + 2*N**2.
!>
!>          إذا كانت LWORK = -1، فيفترض استعلام عن مساحة العمل؛ الروتين
!>          يقوم فقط بحساب الأحجام المثلى لمصفوفتي WORK و IWORK،
!>          ويعيد هذه القيم كمدخلات أولى لمصفوفتي WORK و IWORK، ولا يتم إصدار
!>          أي رسالة خطأ تتعلق بـ LWORK أو LIWORK بواسطة XERBLA.
!>

IWORK

!>          IWORK هي مصفوفة INTEGER، أبعادها (MAX(1,LIWORK))
!>          عند الخروج، إذا كانت INFO = 0، فإن IWORK(1) تعيد LIWORK المثلى.
!>

LIWORK

!>          LIWORK هو INTEGER
!>          بعد المصفوفة IWORK.
!>          إذا كانت N <= 1،                يجب أن يكون LIWORK على الأقل 1.
!>          إذا كانت JOBZ  = 'N' و N > 1، يجب أن يكون LIWORK على الأقل 1.
!>          إذا كانت JOBZ  = 'V' و N > 1، يجب أن يكون LIWORK على الأقل 3 + 5*N.
!>
!>          إذا كانت LIWORK = -1، فيفترض استعلام عن مساحة العمل؛ الروتين
!>          يقوم فقط بحساب الأحجام المثلى لمصفوفتي WORK و IWORK،
!>          ويعيد هذه القيم كمدخلات أولى لمصفوفتي WORK و IWORK، ولا يتم إصدار
!>          أي رسالة خطأ تتعلق بـ LWORK أو LIWORK بواسطة XERBLA.
!>

INFO

!>          INFO هو INTEGER
!>          = 0:  خروج ناجح
!>          < 0:  إذا كانت INFO = -i، فإن الوسيط i كان له قيمة غير قانونية
!>          > 0:  إذا كانت INFO = i و JOBZ = 'N'، فإن الخوارزمية فشلت
!>                في التقارب؛ لم تتقارب i عنصرًا خارج القطر في شكل ثلاثي
!>                وسيط إلى الصفر؛
!>                إذا كانت INFO = i و JOBZ = 'V'، فإن الخوارزمية فشلت
!>                في حساب قيمة ذاتية أثناء العمل على المصفوفة الفرعية
!>                الواقعة في الصفوف والأعمدة INFO/(N+1) حتى
!>                mod(INFO,N+1).
!>
