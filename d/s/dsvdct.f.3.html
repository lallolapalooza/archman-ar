subroutine dsvdct (integer n, double precision, dimension( * ) s, double precision, dimension( * ) e, double precision shift, integer num)

DSVDCT

الغرض:

!>
!> DSVDCT يحسب العدد NUM من القيم الذاتية لمصفوفة ثلاثية الأقطار 2*N × 2*N
!> T والتي هي أقل من أو تساوي SHIFT. يتم تشكيل T بوضع أصفار على القطر الرئيسي
!> وجعل الأقطار الفرعية تساوي S(1), E(1), S(2), E(2), ... , E(N-1), S(N).
!> إذا كانت SHIFT موجبة، فإن NUM تساوي N بالإضافة إلى عدد القيم المنفردة لمصفوفة
!> ثنائية الأقطار B والتي هي أقل من أو تساوي SHIFT. هنا، B تحتوي على مدخلات قطرية
!> S(1), ..., S(N) ومدخلات فوق القطر الرئيسي E(1), ... E(N-1).
!> إذا كانت SHIFT سالبة، فإن NUM تساوي عدد القيم المنفردة لـ B والتي هي أكبر من أو تساوي -SHIFT.
!>
!> انظر W. Kahan, Report CS41, Computer Science Dept., Stanford University,
!> July 21, 1966
!>

المعاملات

N 
!>          N هو عدد صحيح (INTEGER)
!>          بعد المصفوفة ثنائية الأقطار B.
!>

S

!>          S هو مصفوفة دقة مزدوجة (DOUBLE PRECISION) ، البعد (N)
!>          مدخلات القطر الرئيسي للمصفوفة ثنائية الأقطار B.
!>

E

!>          E هو مصفوفة دقة مزدوجة (DOUBLE PRECISION) ببعد (N-1)
!>          مدخلات فوق القطر الرئيسي للمصفوفة ثنائية الأقطار B.
!>

SHIFT

!>          SHIFT هو دقة مزدوجة (DOUBLE PRECISION)
!>          الإزاحة، تستخدم كما هو موضح في الغرض.
!>

NUM

!>          NUM هو عدد صحيح (INTEGER)
!>          عدد القيم الذاتية لـ T الأقل من أو تساوي SHIFT.
!>
