subroutine dsytrf_aa (character uplo, integer n, double precision, dimension( lda, * ) a, integer lda, integer, dimension( * ) ipiv, double precision, dimension( * ) work, integer lwork, integer info)

DSYTRF_AA

الغرض:

!>
!> DSYTRF_AA تحسب تحليل مصفوفة متماثلة حقيقية A
!> باستخدام خوارزمية Aasen. شكل التحليل هو
!>
!>    A = U**T*T*U  أو  A = L*T*L**T
!>
!> حيث U (أو L) هو حاصل ضرب مصفوفات التبديل والمصفوفات المثلثية العليا (السفلى)
!> الوحدوية، و T هي مصفوفة متماثلة ثلاثية الأقطار.
!>
!> هذا هو الإصدار المجمع للخوارزمية، والذي يستدعي BLAS من المستوى 3.
!>

المعلمات

UPLO 
!>          UPLO هو CHARACTER*1
!>          = 'U':  يتم تخزين المثلث العلوي من A؛
!>          = 'L':  يتم تخزين المثلث السفلي من A.
!>

N

!>          N هو INTEGER
!>          ترتيب المصفوفة A.  N >= 0.
!>

A

!>          A هي مصفوفة DOUBLE PRECISION، ذات أبعاد (LDA,N)
!>          عند الإدخال، المصفوفة المتماثلة A. إذا كان UPLO = 'U'، فإن
!>          المثلث العلوي N-by-N للمصفوفة A يحتوي على الجزء العلوي
!>          المثلثي للمصفوفة A، والجزء السفلي المثلثي تمامًا
!>          من A لا يتم الرجوع إليه. إذا كان UPLO = 'L'، فإن
!>          الجزء السفلي المثلثي N-by-N للمصفوفة A يحتوي على الجزء السفلي
!>          المثلثي للمصفوفة A، والجزء العلوي المثلثي تمامًا
!>          من A لا يتم الرجوع إليه.
!>
!>          عند الخروج، يتم تخزين المصفوفة ثلاثية الأقطار في الأقطار
!>          والأقطار الفرعية لـ A أسفل (أو أعلى) الأقطار مباشرةً،
!>          ويتم تخزين L أسفل (أو أعلى) الأقطار الفرعية، عندما يكون UPLO
!>          هو 'L' (أو 'U').
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرائد للمصفوفة A.  LDA >= max(1,N).
!>

IPIV

!>          IPIV هي مصفوفة INTEGER، ذات أبعاد (N)
!>          عند الخروج، تحتوي على تفاصيل التبادلات، أي
!>          تم تبديل الصف والعمود k من A مع
!>          الصف والعمود IPIV(k).
!>

WORK

!>          WORK هي مصفوفة DOUBLE PRECISION، ذات أبعاد (MAX(1,LWORK))
!>          عند الخروج، إذا كان INFO = 0، فإن WORK(1) تعيد LWORK الأمثل.
!>

LWORK

!>          LWORK هو INTEGER
!>          طول WORK.
!>          LWORK >= 1، إذا كان N <= 1، و LWORK >= 2*N، بخلاف ذلك.
!>          للحصول على الأداء الأمثل LWORK >= N*(1+NB)، حيث NB هو
!>          حجم الكتلة الأمثل، الذي تعيده ILAENV.
!>
!>          إذا كان LWORK = -1، فإنه يُفترض استعلام عن مساحة العمل؛ الروتين
!>          يقوم فقط بحساب الحجم الأمثل للمصفوفة WORK، ويعيد
!>          هذه القيمة كالإدخال الأول للمصفوفة WORK، ولا يتم إصدار أي رسالة خطأ
!>          متعلقة بـ LWORK بواسطة XERBLA.
!>

INFO

!>          INFO هو INTEGER
!>          = 0:  انتهاء ناجح
!>          < 0:  إذا كان INFO = -i، فإن وسيطة i كانت قيمة غير قانونية.
!>
