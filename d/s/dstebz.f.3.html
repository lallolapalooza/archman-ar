subroutine dstebz (character range, character order, integer n, double precision vl, double precision vu, integer il, integer iu, double precision abstol, double precision, dimension( * ) d, double precision, dimension( * ) e, integer m, integer nsplit, double precision, dimension( * ) w, integer, dimension( * ) iblock, integer, dimension( * ) isplit, double precision, dimension( * ) work, integer, dimension( * ) iwork, integer info)

DSTEBZ

الغرض:

!>
!> DSTEBZ يحسب القيم الذاتية لمصفوفة ثلاثية الأقطار متناظرة T. قد يطلب المستخدم
!> الحصول على جميع القيم الذاتية، أو جميع القيم الذاتية في الفترة نصف المفتوحة (VL, VU]،
!> أو القيم الذاتية من IL إلى IU.
!>
!> لتجنب تجاوز السعة، يجب قياس المصفوفة بحيث لا يتجاوز أكبر عنصر فيها
!> قيمة overflow**(1/2) * underflow**(1/4) بالقيمة المطلقة، ولأقصى دقة،
!> يجب ألا تكون أصغر بكثير من ذلك.
!>
!> انظر W. Kahan، تقرير CS41، قسم علوم الكمبيوتر، جامعة ستانفورد، 21 يوليو 1966.
!>

المعاملات

RANGE 
!>          RANGE هو CHARACTER*1
!>          = 'A': ()   سيتم العثور على جميع القيم الذاتية.
!>          = 'V': () سيتم العثور على جميع القيم الذاتية في الفترة نصف المفتوحة
!>                           (VL, VU].
!>          = 'I': () سيتم العثور على القيم الذاتية من IL إلى IU (للمصفوفة بأكملها).
!>

ORDER

!>          ORDER هو CHARACTER*1
!>          = 'B': () سيتم تجميع القيم الذاتية حسب
!>                              الكتلة المنفصلة (انظر IBLOCK، ISPLIT) وسيتم
!>                              ترتيبها من الأصغر إلى الأكبر داخل الكتلة.
!>          = 'E': ()
!>                              سيتم ترتيب القيم الذاتية للمصفوفة بأكملها
!>                              من الأصغر إلى الأكبر.
!>

N

!>          N هو INTEGER
!>          رتبة المصفوفة ثلاثية الأقطار T. N >= 0.
!>

VL

!>          VL هو DOUBLE PRECISION
!>
!>          إذا كان RANGE='V'، الحد الأدنى للفترة التي سيتم البحث فيها عن القيم الذاتية.
!>          لن يتم إرجاع القيم الذاتية الأقل من أو تساوي VL، أو الأكبر من VU. VL < VU.
!>          لا يتم الاستشهاد بها إذا كان RANGE = 'A' أو 'I'.
!>

VU

!>          VU هو DOUBLE PRECISION
!>
!>          إذا كان RANGE='V'، الحد الأعلى للفترة التي سيتم البحث فيها عن القيم الذاتية.
!>          لن يتم إرجاع القيم الذاتية الأقل من أو تساوي VL، أو الأكبر من VU. VL < VU.
!>          لا يتم الاستشهاد بها إذا كان RANGE = 'A' أو 'I'.
!>

IL

!>          IL هو INTEGER
!>
!>          إذا كان RANGE='I'، مؤشر أصغر قيمة ذاتية سيتم إرجاعها.
!>          1 <= IL <= IU <= N، إذا كان N > 0؛ IL = 1 و IU = 0 إذا كان N = 0.
!>          لا يتم الاستشهاد بها إذا كان RANGE = 'A' أو 'V'.
!>

IU

!>          IU هو INTEGER
!>
!>          إذا كان RANGE='I'، مؤشر أكبر قيمة ذاتية سيتم إرجاعها.
!>          1 <= IL <= IU <= N، إذا كان N > 0؛ IL = 1 و IU = 0 إذا كان N = 0.
!>          لا يتم الاستشهاد بها إذا كان RANGE = 'A' أو 'V'.
!>

ABSTOL

!>          ABSTOL هو DOUBLE PRECISION
!>          التسامح المطلق للقيم الذاتية. تعتبر القيمة الذاتية (أو الكتلة) في مكانها
!>          إذا تم تحديد أنها تقع في فترة عرضها ABSTOL أو أقل.
!>          إذا كان ABSTOL أقل من أو يساوي الصفر، فسيتم استخدام ULP*|T|، حيث |T|
!>          تعني معيار T ذي النمط 1.
!>
!>          سيتم حساب القيم الذاتية بأكبر قدر من الدقة عندما يتم تعيين ABSTOL
!>          إلى ضعف حد الحد الأدنى للتدفق 2*DLAMCH('S')، وليس صفرًا.
!>

D

!>          D هو مصفوفة DOUBLE PRECISION، البعد (N)
!>          عناصر القطر N للمصفوفة ثلاثية الأقطار T.
!>

E

!>          E هو مصفوفة DOUBLE PRECISION، البعد (N-1)
!>          عناصر خارج القطر (N-1) للمصفوفة ثلاثية الأقطار T.
!>

M

!>          M هو INTEGER
!>          العدد الفعلي للقيم الذاتية التي تم العثور عليها. 0 <= M <= N.
!>          (انظر أيضًا وصف INFO=2,3.)
!>

NSPLIT

!>          NSPLIT هو INTEGER
!>          عدد الكتل القطرية في المصفوفة T.
!>          1 <= NSPLIT <= N.
!>

W

!>          W هو مصفوفة DOUBLE PRECISION، البعد (N)
!>          عند الخروج، ستحتوي العناصر M الأولى من W على القيم الذاتية.
!>          (قد تستخدم DSTEBZ العناصر N-M المتبقية كمساحة عمل.)
!>

IBLOCK

!>          IBLOCK هو مصفوفة INTEGER، البعد (N)
!>          عند كل صف/عمود j حيث يكون E(j) صفرًا أو صغيرًا، تعتبر المصفوفة T
!>          منقسمة إلى مصفوفة قطرية الكتلة. عند الخروج، إذا كان INFO = 0،
!>          تحدد IBLOCK(i) الكتلة (من 1 إلى عدد الكتل) التي تنتمي إليها القيمة الذاتية W(i).
!>          (قد تستخدم DSTEBZ العناصر N-M المتبقية كمساحة عمل.)
!>

ISPLIT

!>          ISPLIT هو مصفوفة INTEGER، البعد (N)
!>          نقاط الانقسام، التي تنقسم عندها T إلى مصفوفات فرعية.
!>          تتكون المصفوفة الفرعية الأولى من الصفوف/الأعمدة 1 إلى ISPLIT(1)،
!>          الثانية من الصفوف/الأعمدة ISPLIT(1)+1 حتى ISPLIT(2)، إلخ،
!>          وتتكون NSPLIT-th من الصفوف/الأعمدة ISPLIT(NSPLIT-1)+1 حتى ISPLIT(NSPLIT)=N.
!>          (سيتم استخدام أول NSPLIT عنصر فقط بالفعل، ولكن نظرًا لأنه لا يمكن للمستخدم
!>          معرفة قيمة NSPLIT التي ستكون مسبقًا، يجب حجز N كلمة لـ ISPLIT.)
!>

WORK

!>          WORK هو مصفوفة DOUBLE PRECISION، البعد (4*N)
!>

IWORK

!>          IWORK هو مصفوفة INTEGER، البعد (3*N)
!>

INFO

!>          INFO هو INTEGER
!>          = 0:  خروج ناجح
!>          < 0:  إذا كان INFO = -i، فإن المعامل i كان له قيمة غير قانونية
!>          > 0:  فشلت بعض أو كل القيم الذاتية في التقارب أو لم يتم حسابها:
!>                =1 أو 3: فشل التنصيف في التقارب لبعض القيم الذاتية؛ يتم تمييز هذه
!>                        القيم الذاتية برقم كتلة سالب. التأثير هو أن القيم الذاتية
!>                        قد لا تكون دقيقة مثل التسامح المطلق والنسبي.
!>                        عادة ما يكون هذا ناتجًا عن حساب غير دقيق بشكل غير متوقع.
!>                =2 أو 3: RANGE='I' فقط: لم يتم العثور على جميع القيم الذاتية
!>                        IL:IU.
!>                        التأثير: M < IU+1-IL
!>                        السبب: حساب غير رتيب، مما يؤدي إلى تسلسل ستورم غير رتيب.
!>                        الحل: إعادة الحساب باستخدام RANGE='A'، واختيار القيم الذاتية IL:IU.
!>                        في بعض الحالات، قد يؤدي زيادة PARAMETER إلى نجاح الأمور.
!>                = 4:    RANGE='I'، وفترة غيرشجورين المستخدمة مبدئيًا كانت صغيرة جدًا.
!>                        لم يتم حساب أي قيم ذاتية.
!>
