توثيق الوظيفة/الروتين الفرعي

subroutine dsfrk (character transr, character uplo, character trans, integer n, integer k, double precision alpha, double precision, dimension( lda, * ) a, integer lda, double precision beta, double precision, dimension( * ) c)

DSFRK تنفذ عملية متناظرة من الرتبة k لمصفوفة بتنسيق RFP.

الغرض:

!>
!> روتين على غرار BLAS من المستوى 3 لـ C بتنسيق RFP.
!>
!> DSFRK تنفذ إحدى العمليات المتناظرة من الرتبة k
!>
!>    C := alpha*A*A**T + beta*C,
!>
!> أو
!>
!>    C := alpha*A**T*A + beta*C,
!>
!> حيث alpha و beta هي أعداد قياسية حقيقية، C هي مصفوفة متناظرة بحجم n × n
!> و A هي مصفوفة بحجم n × k في الحالة الأولى ومصفوفة k × n
!> في الحالة الثانية.
!>

المعاملات

TRANSR 
!>          TRANSR هو CHARACTER*1
!>          = 'N':  يتم تخزين الشكل الطبيعي لـ RFP A;
!>          = 'T':  يتم تخزين الشكل المنقول لـ RFP A.
!>

UPLO

!>          UPLO هو CHARACTER*1
!>           عند الإدخال، يحدد UPLO ما إذا كان الجزء العلوي أو السفلي
!>           من المصفوفة C سيتم الإشارة إليه كالتالي:
!>
!>              UPLO = 'U' or 'u'   سيتم الإشارة فقط إلى الجزء العلوي من C
!>                                  .
!>
!>              UPLO = 'L' or 'l'   سيتم الإشارة فقط إلى الجزء السفلي من C
!>                                  .
!>
!>           غير متغير عند الخروج.
!>

TRANS

!>          TRANS هو CHARACTER*1
!>           عند الإدخال، يحدد TRANS العملية التي سيتم تنفيذها كالتالي:
!>
!>              TRANS = 'N' or 'n'   C := alpha*A*A**T + beta*C.
!>
!>              TRANS = 'T' or 't'   C := alpha*A**T*A + beta*C.
!>
!>           غير متغير عند الخروج.
!>

N

!>          N هو INTEGER
!>           عند الإدخال، يحدد N رتبة المصفوفة C. يجب أن يكون N
!>           على الأقل صفر.
!>           غير متغير عند الخروج.
!>

K

!>          K هو INTEGER
!>           عند الإدخال مع TRANS = 'N' or 'n'، يحدد K عدد
!>           الأعمدة للمصفوفة A، وعند الإدخال مع TRANS = 'T'
!>           أو 't'، يحدد K عدد الصفوف للمصفوفة A. يجب أن يكون K
!>           على الأقل صفر.
!>           غير متغير عند الخروج.
!>

ALPHA

!>          ALPHA هو DOUBLE PRECISION
!>           عند الإدخال، يحدد ALPHA العدد القياسي alpha.
!>           غير متغير عند الخروج.
!>

A

!>          A هو مصفوفة DOUBLE PRECISION، أبعاد (LDA,ka)
!>           حيث KA
!>           هو K عندما TRANS = 'N' or 'n'، وهو N بخلاف ذلك. قبل
!>           الدخول مع TRANS = 'N' or 'n'، يجب أن يحتوي الجزء الأمامي بحجم N×K
!>           من المصفوفة A على المصفوفة A، بخلاف ذلك يجب أن يحتوي الجزء الأمامي بحجم K×N
!>           من المصفوفة A على المصفوفة A.
!>           غير متغير عند الخروج.
!>

LDA

!>          LDA هو INTEGER
!>           عند الإدخال، يحدد LDA البعد الأول لـ A كما تم الإعلان عنه
!>           في البرنامج (البرنامج الفرعي) المستدعي. عندما TRANS = 'N' or 'n'
!>           فيجب أن يكون LDA على الأقل max( 1, n )، بخلاف ذلك
!>           يجب أن يكون LDA على الأقل max( 1, k ).
!>           غير متغير عند الخروج.
!>

BETA

!>          BETA هو DOUBLE PRECISION
!>           عند الإدخال، يحدد BETA العدد القياسي beta.
!>           غير متغير عند الخروج.
!>

C

!>          C هو مصفوفة DOUBLE PRECISION، أبعاد (NT)
!>           NT = N*(N+1)/2. عند الإدخال، المصفوفة المتناظرة C بتنسيق
!>           RFP. يوصف تنسيق RFP بواسطة TRANSR و UPLO و N.
!>
