subroutine dsygst (integer itype, character uplo, integer n, double precision, dimension( lda, * ) a, integer lda, double precision, dimension( ldb, * ) b, integer ldb, integer info)

DSYGST

الغرض:

!>
!> DSYGST تقلل مشكلة القيم الذاتية المتماثلة المحددة المعممة
!> إلى الشكل القياسي.
!>
!> إذا كان ITYPE = 1، فإن المشكلة هي A*x = lambda*B*x،
!> ويتم استبدال A بـ inv(U**T)*A*inv(U) أو inv(L)*A*inv(L**T)
!>
!> إذا كان ITYPE = 2 أو 3، فإن المشكلة هي A*B*x = lambda*x أو
!> B*A*x = lambda*x، ويتم استبدال A بـ U*A*U**T أو L**T*A*L.
!>
!> يجب أن يكون B قد تم تحليله سابقًا كـ U**T*U أو L*L**T بواسطة DPOTRF.
!>

المعاملات

ITYPE 
!>          ITYPE هو عدد صحيح
!>          = 1: حساب inv(U**T)*A*inv(U) أو inv(L)*A*inv(L**T)؛
!>          = 2 أو 3: حساب U*A*U**T أو L**T*A*L.
!>

UPLO

!>          UPLO هو حرف واحد
!>          = 'U':  يتم تخزين المثلث العلوي من A ويتم تحليل B كـ
!>                  U**T*U؛
!>          = 'L':  يتم تخزين المثلث السفلي من A ويتم تحليل B كـ
!>                  L*L**T.
!>

N

!>          N هو عدد صحيح
!>          رتبة المصفوفتين A و B.  N >= 0.
!>

A

!>          A هو مصفوفة ذات دقة مزدوجة، أبعاد (LDA,N)
!>          عند الدخول، المصفوفة المتماثلة A.  إذا كان UPLO = 'U'، فإن
!>          الجزء العلوي المثلثي من A بالحجم N*N يحتوي على الجزء
!>          المثلثي العلوي للمصفوفة A، ولا يتم الرجوع إلى الجزء السفلي
!>          المثلثي الصارم من A.  إذا كان UPLO = 'L'، فإن الجزء السفلي
!>          المثلثي من A بالحجم N*N يحتوي على الجزء المثلثي السفلي
!>          للمصفوفة A، ولا يتم الرجوع إلى الجزء العلوي المثلثي الصارم من A.
!>
!>          عند الخروج، إذا كان INFO = 0، فإن المصفوفة المحولة، مخزنة بنفس
!>          تنسيق A.
!>

LDA

!>          LDA هو عدد صحيح
!>          البعد الرئيسي للمصفوفة A.  LDA >= max(1,N).
!>

B

!>          B هو مصفوفة ذات دقة مزدوجة، أبعاد (LDB,N)
!>          العامل المثلثي من التحليل Cholesky لـ B،
!>          كما تم إرجاعه بواسطة DPOTRF.
!>

LDB

!>          LDB هو عدد صحيح
!>          البعد الرئيسي للمصفوفة B.  LDB >= max(1,N).
!>

INFO

!>          INFO هو عدد صحيح
!>          = 0:  الخروج الناجح
!>          < 0:  إذا كان INFO = -i، فإن المعامل i كان له قيمة غير قانونية
!>
