subroutine dsgt01 (integer itype, character uplo, integer n, integer m, double precision, dimension( lda, * ) a, integer lda, double precision, dimension( ldb, * ) b, integer ldb, double precision, dimension( ldz, * ) z, integer ldz, double precision, dimension( * ) d, double precision, dimension( * ) work, double precision, dimension( * ) result)

DSGT01

الغرض:

!>
!> DDGT01 يتحقق من تحليل للصيغة
!>
!>    A Z   =  B Z D أو
!>    A B Z =  Z D أو
!>    B A Z =  Z D
!>
!> حيث A هي مصفوفة متناظرة، B هي
!> موجبة تمامًا متناظرة، Z متعامدة، و D قطرية.
!>
!> يتم حساب إحدى نسب الاختبار التالية:
!>
!> ITYPE = 1:  RESULT(1) = | A Z - B Z D | / ( |A| |Z| n ulp )
!>
!> ITYPE = 2:  RESULT(1) = | A B Z - Z D | / ( |A| |Z| n ulp )
!>
!> ITYPE = 3:  RESULT(1) = | B A Z - Z D | / ( |A| |Z| n ulp )
!>

المعلمات

ITYPE 
!>          ITYPE هو عدد صحيح
!>          شكل مشكلة القيمة الذاتية المعممة المتناظرة.
!>          = 1:  A*z = (lambda)*B*z
!>          = 2:  A*B*z = (lambda)*z
!>          = 3:  B*A*z = (lambda)*z
!>

UPLO

!>          UPLO هو حرف*1
!>          يحدد ما إذا كان الجزء العلوي أو السفلي الثلاثي للمصفوفتين
!>          المتناظرتين A و B مخزنًا.
!>          = 'U':  مثلثي علوي
!>          = 'L':  مثلثي سفلي
!>

N

!>          N هو عدد صحيح
!>          ترتيب المصفوفة A. N >= 0.
!>

M

!>          M هو عدد صحيح
!>          عدد القيم الذاتية التي تم العثور عليها. 0 <= M <= N.
!>

A

!>          A هو مصفوفة double precision، حجم (LDA, N)
!>          المصفوفة المتناظرة الأصلية A.
!>

LDA

!>          LDA هو عدد صحيح
!>          البعد الرئيسي للمصفوفة A. LDA >= max(1,N).
!>

B

!>          B هو مصفوفة double precision، حجم (LDB, N)
!>          المصفوفة الأصلية المتناظرة الموجبة تمامًا B.
!>

LDB

!>          LDB هو عدد صحيح
!>          البعد الرئيسي للمصفوفة B. LDB >= max(1,N).
!>

Z

!>          Z هي مصفوفة double precision، حجم (LDZ, M)
!>          المتجهات الذاتية المحسوبة لمشكلة القيمة الذاتية المعممة.
!>

LDZ

!>          LDZ هو عدد صحيح
!>          البعد الرئيسي للمصفوفة Z. LDZ >= max(1,N).
!>

D

!>          D هي مصفوفة double precision، حجم (M)
!>          القيم الذاتية المحسوبة لمشكلة القيمة الذاتية المعممة.
!>

WORK

!>          WORK هي مصفوفة double precision، حجم (N*N)
!>

RESULT

!>          RESULT هي مصفوفة double precision، حجم (1)
!>          نسبة الاختبار كما هو موضح أعلاه.
!>
