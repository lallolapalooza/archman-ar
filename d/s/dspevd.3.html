توثيق الوظيفة/الروتين الفرعي

subroutine dspevd (character jobz, character uplo, integer n, double precision, dimension( * ) ap, double precision, dimension( * ) w, double precision, dimension( ldz, * ) z, integer ldz, double precision, dimension( * ) work, integer lwork, integer, dimension( * ) iwork, integer liwork, integer info)

DSPEVD يحسب القيم الذاتية، واختياريًا، المتجهات الذاتية اليسرى و/أو اليمنى للمصفوفات الأخرى

الغرض:

!>
!> DSPEVD يحسب جميع القيم الذاتية، واختياريًا، المتجهات الذاتية
!> لمصفوفة متناظرة حقيقية A بتخزين مضغوط. إذا كانت المتجهات الذاتية
!> مطلوبة، فإنه يستخدم خوارزمية قسمة وفتح.
!>
!>

المعلمات

JOBZ 
!>          JOBZ هو CHARACTER*1
!>          = 'N': حساب القيم الذاتية فقط؛
!>          = 'V': حساب القيم الذاتية والمتجهات الذاتية.
!>

UPLO

!>          UPLO هو CHARACTER*1
!>          = 'U': المثلث العلوي لـ A مخزن؛
!>          = 'L': المثلث السفلي لـ A مخزن.
!>

N

!>          N هو INTEGER
!>          ترتيب المصفوفة A. N >= 0.
!>

AP

!>          AP هو مصفوفة DOUBLE PRECISION، البعد (N*(N+1)/2)
!>          عند الإدخال، المثلث العلوي أو السفلي للمصفوفة المتناظرة
!>          A، مضغوطة عموديًا في مصفوفة خطية. يتم تخزين العمود j من A
!>          في المصفوفة AP كالتالي:
!>          إذا كان UPLO = 'U'، AP(i + (j-1)*j/2) = A(i,j) لـ 1<=i<=j؛
!>          إذا كان UPLO = 'L'، AP(i + (j-1)*(2*n-j)/2) = A(i,j) لـ j<=i<=n.
!>
!>          عند الخروج، يتم الكتابة فوق AP بالقيم التي تم إنشاؤها أثناء
!>          الاختزال إلى شكل ثلاثي الأقطار. إذا كان UPLO = 'U'، يتم الكتابة
!>          فوق القطر الرئيسي والقطر الفائق الأول للمصفوفة ثلاثية الأقطار T
!>          فوق العناصر المقابلة لـ A، وإذا كان UPLO = 'L'، يتم الكتابة فوق
!>          القطر الرئيسي والقطر الفرعي الأول لـ T فوق
!>          العناصر المقابلة لـ A.
!>

W

!>          W هي مصفوفة DOUBLE PRECISION، البعد (N)
!>          إذا كان INFO = 0، القيم الذاتية بترتيب تصاعدي.
!>

Z

!>          Z هي مصفوفة DOUBLE PRECISION، البعد (LDZ, N)
!>          إذا كان JOBZ = 'V'، ثم إذا كان INFO = 0، تحتوي Z على المتجهات
!>          الذاتية المتعامدة للمصفوفة A، مع احتواء العمود i من Z
!>          على المتجه الذاتي المرتبط بـ W(i).
!>          إذا كان JOBZ = 'N'، فلن يتم استخدام Z.
!>

LDZ

!>          LDZ هو INTEGER
!>          البعد الرئيسي للمصفوفة Z. LDZ >= 1، وإذا
!>          كان JOBZ = 'V'، فإن LDZ >= max(1,N).
!>

WORK

!>          WORK هي مصفوفة DOUBLE PRECISION، البعد (MAX(1,LWORK))
!>          عند الخروج، إذا كان INFO = 0، فإن WORK(1) تعيد LWORK المطلوبة.
!>

LWORK

!>          LWORK هو INTEGER
!>          بعد المصفوفة WORK.
!>          إذا كان N <= 1، يجب أن يكون LWORK على الأقل 1.
!>          إذا كان JOBZ = 'N' و N > 1، يجب أن يكون LWORK على الأقل 2*N.
!>          إذا كان JOBZ = 'V' و N > 1، يجب أن يكون LWORK على الأقل
!>                                                 1 + 6*N + N**2.
!>
!>          إذا كان LWORK = -1، فسيتم افتراض استعلام مساحة العمل؛ تقوم الروتين
!>          فقط بحساب الأحجام المطلوبة للمصفوفتين WORK و IWORK،
!>          وتعيد هذه القيم كأول إدخال في المصفوفتين WORK و IWORK، ولا يتم
!>          إصدار أي رسالة خطأ تتعلق بـ LWORK أو LIWORK بواسطة XERBLA.
!>

IWORK

!>          IWORK هي مصفوفة INTEGER، البعد (MAX(1,LIWORK))
!>          عند الخروج، إذا كان INFO = 0، فإن IWORK(1) تعيد LIWORK المطلوبة.
!>

LIWORK

!>          LIWORK هو INTEGER
!>          بعد المصفوفة IWORK.
!>          إذا كان JOBZ = 'N' أو N <= 1، يجب أن يكون LIWORK على الأقل 1.
!>          إذا كان JOBZ = 'V' و N > 1، يجب أن يكون LIWORK على الأقل 3 + 5*N.
!>
!>          إذا كان LIWORK = -1، فسيتم افتراض استعلام مساحة العمل؛ تقوم الروتين
!>          فقط بحساب الأحجام المطلوبة للمصفوفتين WORK و IWORK،
!>          وتعيد هذه القيم كأول إدخال في المصفوفتين WORK و IWORK، ولا يتم
!>          إصدار أي رسالة خطأ تتعلق بـ LWORK أو LIWORK بواسطة XERBLA.
!>

INFO

!>          INFO هو INTEGER
!>          = 0: خروج ناجح
!>          < 0: إذا كان INFO = -i، فإن الوسيط i كان له قيمة غير قانونية.
!>          > 0: إذا كان INFO = i، فشلت الخوارزمية في التقارب؛ i
!>                العناصر خارج القطر الرئيسي لشكل ثلاثي الأقطار وسيط
!!                لم تتقارب إلى الصفر.
!>
