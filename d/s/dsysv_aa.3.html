subroutine dsysv_aa (character uplo, integer n, integer nrhs, double precision, dimension( lda, * ) a, integer lda, integer, dimension( * ) ipiv, double precision, dimension( ldb, * ) b, integer ldb, double precision, dimension( * ) work, integer lwork, integer info)

DSYSV_AA تحسب حل نظام المعادلات الخطية A * X = B لمصفوفات SY

الغرض:

!>
!> DSYSV تحسب حل نظام معادلات خطية حقيقي
!>    A * X = B,
!> حيث A مصفوفة متناظرة بحجم N×N و X و B مصفوفتان بحجم N×NRHS.
!>
!> يتم استخدام خوارزمية Aasen لتحليل A إلى:
!>    A = U**T * T * U,  إذا كان UPLO = 'U', أو
!>    A = L * T * L**T,  إذا كان UPLO = 'L',
!> حيث U (أو L) هو حاصل ضرب مصفوفات تبديل ومصفوفات مثلثة عليا (سفلى) وحدية،
!> و T هي مصفوفة ثلاثية متناظرة. ثم يتم استخدام الشكل المحلل لـ A لحل
!> نظام المعادلات A * X = B.
!>

المعاملات

UPLO 
!>          UPLO هو CHARACTER*1
!>          = 'U':  يتم تخزين الجزء العلوي من A؛
!>          = 'L':  يتم تخزين الجزء السفلي من A؛
!>

N

!>          N هو INTEGER
!>          عدد المعادلات الخطية، أي رتبة المصفوفة A.  N >= 0.
!>

NRHS

!>          NRHS هو INTEGER
!>          عدد الأطراف اليمنى، أي عدد أعمدة المصفوفة B.  NRHS >= 0.
!>

A

!>          A هي مصفوفة DOUBLE PRECISION، بأبعاد (LDA,N)
!>          عند الإدخال، المصفوفة المتناظرة A.  إذا كان UPLO = 'U'، فإن الجزء
!>          العلوي الرئيسي بحجم N×N من A يحتوي على الجزء العلوي
!>          من المصفوفة A، ولا يتم الإشارة إلى الجزء السفلي الصارم
!>          من A. إذا كان UPLO = 'L'، فإن الجزء السفلي الرئيسي بحجم N×N
!>          من A يحتوي على الجزء السفلي من المصفوفة A، ولا يتم الإشارة إلى الجزء
!>          العلوي الصارم من A.
!>
!>          عند الخروج، إذا كان INFO = 0، فإن المصفوفة الثلاثية T والمضاعفات
!>          المستخدمة للحصول على العامل U أو L من التحليل
!>          A = U**T*T*U أو A = L*T*L**T كما تم حسابها بواسطة
!>          DSYTRF.
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرئيسي للمصفوفة A.  LDA >= max(1,N).
!>

IPIV

!>          IPIV هي مصفوفة INTEGER، بأبعاد (N)
!>          عند الخروج، تحتوي على تفاصيل التبادلات، أي أن الصف والعمود k
!>          من A تم تبادلهما مع الصف والعمود IPIV(k).
!>

B

!>          B هي مصفوفة DOUBLE PRECISION، بأبعاد (LDB,NRHS)
!>          عند الإدخال، مصفوفة الطرف الأيمن B بحجم N×NRHS.
!>          عند الخروج، إذا كان INFO = 0، فإن مصفوفة الحل X بحجم N×NRHS.
!>

LDB

!>          LDB هو INTEGER
!>          البعد الرئيسي للمصفوفة B.  LDB >= max(1,N).
!>

WORK

!>          WORK هي مصفوفة DOUBLE PRECISION، بأبعاد (MAX(1,LWORK))
!>          عند الخروج، إذا كان INFO = 0، فإن WORK(1) ترجع LWORK المثلى.
!>

LWORK

!>          LWORK هو INTEGER
!>          طول WORK.  LWORK >= MAX(1,2*N,3*N-2)، وللحصول على أفضل
!>          أداء، LWORK >= MAX(1,N*NB)، حيث NB هو حجم الكتلة الأمثل لـ DSYTRF_AA.
!>
!>          إذا كان LWORK = -1، فإنه يُفترض استعلام عن مساحة العمل؛
!>          يقوم الروتين فقط بحساب الحجم الأمثل لمصفوفة WORK، ويرجع هذا
!>          الحجم كالإدخال الأول لمصفوفة WORK، ولا يتم إصدار أي رسالة خطأ
!>          متعلقة بـ LWORK بواسطة XERBLA.
!>

INFO

!>          INFO هو INTEGER
!>          = 0: اكتمال ناجح
!>          < 0: إذا كان INFO = -i، فإن المعامل i كان له قيمة غير قانونية
!>          > 0: إذا كان INFO = i، فإن D(i,i) يساوي صفرًا تمامًا. لقد اكتمل
!>               التحليل، لكن المصفوفة القطرية العادية D منفردة تمامًا،
!>               لذلك لم يكن بالإمكان حساب الحل.
!>
