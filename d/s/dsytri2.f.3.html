subroutine dsytri2 (character uplo, integer n, double precision, dimension( lda, * ) a, integer lda, integer, dimension( * ) ipiv, double precision, dimension( * ) work, integer lwork, integer info)

DSYTRI2

الغرض:

!>
!> DSYTRI2 يحسب معكوس المصفوفة المتماثلة غير المحددة ذات الدقة المزدوجة
!> A باستخدام التحليل A = U*D*U**T أو A = L*D*L**T الذي تم حسابه بواسطة
!> DSYTRF. DSYTRI2 يحدد البعد الرائد لمساحة العمل
!> قبل استدعاء DSYTRI2X الذي يحسب بالفعل المعكوس.
!>

المعلمات

UPLO 
!>          UPLO هو CHARACTER*1
!>          يحدد ما إذا كانت تفاصيل التحليل مخزنة
!>          كمصفوفة مثلثة علوية أو سفلية.
!>          = 'U':  مثلثة علوية، الشكل هو A = U*D*U**T؛
!>          = 'L':  مثلثة سفلية، الشكل هو A = L*D*L**T.
!>

N

!>          N هو INTEGER
!>          رتبة المصفوفة A.  N >= 0.
!>

A

!>          A هي مصفوفة DOUBLE PRECISION، البعد (LDA,N)
!>          عند الإدخال، المصفوفة القطرية المربعة D والعوامل
!>          المستخدمة للحصول على العامل U أو L كما تم حسابه بواسطة DSYTRF.
!>
!>          عند الخروج، إذا كان INFO = 0، فإن معكوس (المتماثل) للمصفوفة الأصلية.
!>          إذا كان UPLO = 'U'، يتم تشكيل الجزء المثلثي العلوي من المعكوس
!>          والجزء A تحت القطر لا يتم الرجوع إليه؛ إذا كان UPLO = 'L' يتم تشكيل
!>          الجزء المثلثي السفلي من المعكوس والجزء A فوق القطر
!>          لا يتم الرجوع إليه.
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرائد للمصفوفة A.  LDA >= max(1,N).
!>

IPIV

!>          IPIV هي مصفوفة INTEGER، البعد (N)
!>          تفاصيل التبادلات وبنية المصفوفة D
!>          كما تم تحديده بواسطة DSYTRF.
!>

WORK

!>          WORK هي مصفوفة DOUBLE PRECISION، البعد (MAX(1,LWORK))
!>

LWORK

!>          LWORK هو INTEGER
!>          بعد المصفوفة WORK.
!>          إذا كان N = 0، LWORK >= 1، وإلا LWORK >= (N+NB+1)*(NB+3).
!>          إذا كان LWORK = -1، فإنه يُفترض استعلام عن مساحة العمل؛ الروتين
!>          يحسب:
!>              - البعد الأمثل لمصفوفة WORK، يعيد
!>          هذه القيمة كإدخال أول لمصفوفة WORK،
!>              - ولا يتم إصدار أي رسالة خطأ متعلقة بـ LWORK بواسطة XERBLA.
!>

INFO

!>          INFO هو INTEGER
!>          = 0: خروج ناجح
!>          < 0: إذا كان INFO = -i، فإن الوسيط i كان له قيمة غير قانونية
!>          > 0: إذا كان INFO = i، D(i,i) = 0؛ المصفوفة مفردة ولا يمكن حساب معكوسها.
!>
