subroutine dsytrf (character uplo, integer n, double precision, dimension( lda, * ) a, integer lda, integer, dimension( * ) ipiv, double precision, dimension( * ) work, integer lwork, integer info)

DSYTRF

الغرض:

!>
!> DSYTRF تحسب تحليل مصفوفة متماثلة حقيقية A باستخدام
!> طريقة المحورية القطرية لـ Bunch-Kaufman. شكل
!> التحليل هو
!>
!>    A = U**T*D*U  أو  A = L*D*L**T
!>
!> حيث U (أو L) هو حاصل ضرب مصفوفات التبديل والمصفوفات المثلثية العلوية (السفلية) الوحدوية،
!> و D هي مصفوفة قطرية متماثلة وكتلية ذات كتل قطرية 1*1 و 2*2.
!>
!> هذا هو الإصدار المكتل للخوارزمية، والذي يستدعي BLAS المستوى 3.
!>

المعلمات

UPLO 
!>          UPLO هي CHARACTER*1
!>          = 'U':  يتم تخزين المثلث العلوي للمصفوفة A؛
!>          = 'L':  يتم تخزين المثلث السفلي للمصفوفة A.
!>

N

!>          N هو INTEGER
!>          رتبة المصفوفة A.  N >= 0.
!>

A

!>          A هي مصفوفة DOUBLE PRECISION، ذات البعد (LDA,N)
!>          عند الإدخال، المصفوفة المتماثلة A.  إذا كان UPLO = 'U'، فإن الجزء العلوي
!>          المثلثي الذي بحجم N*N في A يحتوي على الجزء العلوي
!>          المثلثي للمصفوفة A، ولا يتم الرجوع إلى الجزء السفلي
!>          المثلثي الصارم للمصفوفة A.  إذا كان UPLO = 'L'، فإن الجزء السفلي
!>          المثلثي الذي بحجم N*N في A يحتوي على الجزء السفلي
!>          المثلثي للمصفوفة A، ولا يتم الرجوع إلى الجزء العلوي
!>          المثلثي الصارم للمصفوفة A.
!>
!>          عند الخروج، المصفوفة القطرية الكتلية D والمعاملات المستخدمة
!>          للحصول على العامل U أو L (انظر أدناه لمزيد من التفاصيل).
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرئيسي للمصفوفة A.  LDA >= max(1,N).
!>

IPIV

!>          IPIV هي مصفوفة INTEGER، ذات البعد (N)
!>          تفاصيل التبادلات وهيكل الكتل لـ D.
!>          إذا كان IPIV(k) > 0، فإنه تم تبديل الصفوف والأعمدة k و IPIV(k)
!>          و D(k,k) هي كتلة قطرية 1*1.
!>          إذا كان UPLO = 'U' و IPIV(k) = IPIV(k-1) < 0، فإنه تم تبديل الصفوف
!>          والأعمدة k-1 و -IPIV(k) و D(k-1:k,k-1:k)
!>          هي كتلة قطرية 2*2.  إذا كان UPLO = 'L' و IPIV(k) =
!>          IPIV(k+1) < 0، فإنه تم تبديل الصفوف والأعمدة k+1 و -IPIV(k)
!>          و D(k:k+1,k:k+1) هي كتلة قطرية 2*2.
!>

WORK

!>          WORK هي مصفوفة DOUBLE PRECISION، ذات البعد (MAX(1,LWORK))
!>          عند الخروج، إذا كان INFO = 0، فإن WORK(1) يعيد LWORK الأمثل.
!>

LWORK

!>          LWORK هو INTEGER
!>          طول WORK.  LWORK >= 1.  لأفضل أداء
!>          LWORK >= N*NB، حيث NB هو حجم الكتلة الذي تعيده ILAENV.
!>
!>          إذا كان LWORK = -1، فسيتم افتراض استعلام مساحة العمل؛ الروتين
!>          يحسب فقط الحجم الأمثل لمصفوفة WORK، ويعيد
!>          هذه القيمة كإدخال أول لمصفوفة WORK، ولا يتم إصدار أي رسالة خطأ
!>          متعلقة بـ LWORK بواسطة XERBLA.
!>

INFO

!>          INFO هو INTEGER
!>          = 0:  خروج ناجح
!>          < 0:  إذا كان INFO = -i، فإن الوسيط i لديه قيمة غير قانونية
!>          > 0:  إذا كان INFO = i، فإن D(i,i) يساوي صفرًا بالضبط.  تم
!>                اكتمال التحليل، ولكن المصفوفة القطرية الكتلية D
!>                هي مفردة بالضبط، وسيحدث قسمة على صفر إذا
!>                تم استخدامها لحل نظام من المعادلات.
!>

تفاصيل إضافية:

!>
!>  إذا كان UPLO = 'U'، فإن A = U**T*D*U، حيث
!>     U = P(n)*U(n)* ... *P(k)U(k)* ...,
!>  أي أن U هو حاصل ضرب حدود P(k)*U(k)، حيث k تتناقص من n إلى
!>  1 بخطوات 1 أو 2، و D هي مصفوفة قطرية كتلية ذات كتل
!>  قطرية 1*1 و 2*2 هي D(k). P(k) هي مصفوفة تبديل كما
!>  تحددها IPIV(k)، و U(k) هي مصفوفة مثلثية علوية وحدوية، بحيث
!>  إذا كانت الكتلة القطرية D(k) من الرتبة s (s = 1 أو 2)، فإن
!>
!>             (   I    v    0   )   k-s
!>     U(k) =  (   0    I    0   )   s
!>             (   0    0    I   )   n-k
!>                k-s   s   n-k
!>
!>  إذا كان s = 1، فإن D(k) تستبدل A(k,k)، و v تستبدل A(1:k-1,k).
!>  إذا كان s = 2، فإن المثلث العلوي لـ D(k) يستبدل A(k-1,k-1)، A(k-1,k)،
!>  و A(k,k)، و v تستبدل A(1:k-2,k-1:k).
!>
!>  إذا كان UPLO = 'L'، فإن A = L*D*L**T، حيث
!>     L = P(1)*L(1)* ... *P(k)*L(k)* ...,
!>  أي أن L هو حاصل ضرب حدود P(k)*L(k)، حيث k تزداد من 1 إلى
!>  n بخطوات 1 أو 2، و D هي مصفوفة قطرية كتلية ذات كتل
!>  قطرية 1*1 و 2*2 هي D(k). P(k) هي مصفوفة تبديل كما
!>  تحددها IPIV(k)، و L(k) هي مصفوفة مثلثية سفلية وحدوية، بحيث
!>  إذا كانت الكتلة القطرية D(k) من الرتبة s (s = 1 أو 2)، فإن
!>
!>             (   I    0     0   )  k-1
!>     L(k) =  (   0    I     0   )  s
!>             (   0    v     I   )  n-k-s+1
!>                k-1   s  n-k-s+1
!>
!>  إذا كان s = 1، فإن D(k) تستبدل A(k,k)، و v تستبدل A(k+1:n,k).
!>  إذا كان s = 2، فإن المثلث السفلي لـ D(k) يستبدل A(k,k)، A(k+1,k)،
!>  و A(k+1,k+1)، و v تستبدل A(k+2:n,k:k+1).
!>
