توثيق الوظيفة/الإجراء الفرعي

subroutine dsygv (integer itype, character jobz, character uplo, integer n, double precision, dimension( lda, * ) a, integer lda, double precision, dimension( ldb, * ) b, integer ldb, double precision, dimension( * ) w, double precision, dimension( * ) work, integer lwork, integer info)

DSYGV

الغرض:

!>
!> DSYGV يحسب جميع القيم الذاتية، اختيارياً، المتجهات الذاتية
!> لمشكلة التماثل القياسية المعممة المحددة، من الشكل
!> A*x=(lambda)*B*x,  A*Bx=(lambda)*x,  أو B*A*x=(lambda)*x.
!> هنا يُفترض أن A و B متماثلان وأن B موجب محدد أيضًا.
!>

المعلمات

ITYPE 
!>          ITYPE هو عدد صحيح
!>          يحدد نوع المشكلة المراد حلها:
!>          = 1:  A*x = (lambda)*B*x
!>          = 2:  A*B*x = (lambda)*x
!>          = 3:  B*A*x = (lambda)*x
!>

JOBZ

!>          JOBZ هو حرف واحد
!>          = 'N':  حساب القيم الذاتية فقط؛
!>          = 'V':  حساب القيم الذاتية والمتجهات الذاتية.
!>

UPLO

!>          UPLO هو حرف واحد
!>          = 'U':  المثلثات العلوية من A و B مخزنة؛
!>          = 'L':  المثلثات السفلية من A و B مخزنة.
!>

N

!>          N هو عدد صحيح
!>          رتبة المصفوفتين A و B.  N >= 0.
!>

A

!>          A هي مصفوفة دقة مزدوجة، البعد (LDA, N)
!>          عند الإدخال، المصفوفة المتماثلة A.  إذا كان UPLO = 'U'، فإن
!>          الجزء العلوي المثلثي N-by-N الرائد من A يحتوي على
!>          الجزء العلوي المثلثي من المصفوفة A.  إذا كان UPLO = 'L'، فإن
!>          الجزء السفلي المثلثي الرائد N-by-N من A يحتوي على
!>          الجزء السفلي المثلثي من المصفوفة A.
!>
!>          عند الخروج، إذا كان JOBZ = 'V'، فإذا كان INFO = 0، تحتوي A على
!>          المصفوفة Z للمتجهات الذاتية.  يتم تطبيع المتجهات الذاتية
!>          بشكل التالي:
!>          إذا كان ITYPE = 1 أو 2، Z**T*B*Z = I؛
!>          إذا كان ITYPE = 3، Z**T*inv(B)*Z = I.
!>          إذا كان JOBZ = 'N'، ف عند الخروج، يتم تدمير المثلث العلوي (إذا كان UPLO='U')
!>          أو المثلث السفلي (إذا كان UPLO='L') من A، بما في ذلك
!>          القطر.
!>

LDA

!>          LDA هو عدد صحيح
!>          البعد الرائد للمصفوفة A.  LDA >= max(1,N).
!>

B

!>          B هي مصفوفة دقة مزدوجة، البعد (LDB, N)
!>          عند الإدخال، المصفوفة المتماثلة المحددة موجبة B.
!>          إذا كان UPLO = 'U'، فإن الجزء العلوي المثلثي الرائد N-by-N من B
!>          يحتوي على الجزء العلوي المثلثي من المصفوفة B.
!>          إذا كان UPLO = 'L'، فإن الجزء السفلي المثلثي الرائد N-by-N من B
!>          يحتوي على الجزء السفلي المثلثي من المصفوفة B.
!>
!>          عند الخروج، إذا كان INFO <= N، يتم استبدال الجزء من B الذي يحتوي على المصفوفة
!>          بواسطة العامل المثلثي U أو L من تحليل Cholesky
!>          B = U**T*U أو B = L*L**T.
!>

LDB

!>          LDB هو عدد صحيح
!>          البعد الرائد للمصفوفة B.  LDB >= max(1,N).
!>

W

!>          W هي مصفوفة دقة مزدوجة، البعد (N)
!>          إذا كان INFO = 0، فإن القيم الذاتية بترتيب تصاعدي.
!>

WORK

!>          WORK هي مصفوفة دقة مزدوجة، البعد (MAX(1,LWORK))
!>          عند الخروج، إذا كان INFO = 0، فإن WORK(1) يعيد LWORK الأمثل.
!>

LWORK

!>          LWORK هو عدد صحيح
!>          طول المصفوفة WORK.  LWORK >= max(1,3*N-1).
!>          للكفاءة المثلى، LWORK >= (NB+2)*N،
!>          حيث NB هو حجم الكتلة لـ DSYTRD الذي تم إرجاعه بواسطة ILAENV.
!>
!>          إذا كان LWORK = -1، ف يُفترض استعلام مساحة العمل؛ الروتين
!>          يحسب فقط الحجم الأمثل للمصفوفة WORK، ويعيد
!>          هذه القيمة كإدخال أول للمصفوفة WORK، ولا يتم إصدار أي رسالة خطأ
!>          متعلقة بـ LWORK بواسطة XERBLA.
!>

INFO

!>          INFO هو عدد صحيح
!>          = 0:  انتهاء ناجح
!>          < 0:  إذا كان INFO = -i، فإن الوسيط i كان له قيمة غير قانونية
!>          > 0:  DPOTRF أو DSYEV أعاد رمز خطأ:
!>             <= N:  إذا كان INFO = i، فشل DSYEV في التقارب؛
!>                    i عناصر خارج القطر لمصفوفة ثلاثية الأقطار وسيطة
!>                    لم تتقارب إلى الصفر؛
!>             > N:   إذا كان INFO = N + i، لـ 1 <= i <= N، فإن
!>                    المُحدِّد الرئيسي الرائد من الرتبة i لـ B ليس موجبًا.
!>                    لم يكن من الممكن إكمال تحليل B ولم
!>                    يتم حساب أي قيم ذاتية أو متجهات ذاتية.
!>
