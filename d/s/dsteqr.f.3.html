توثيق الوظيفة/الروتين الفرعي

subroutine dsteqr (character compz, integer n, double precision, dimension( * ) d, double precision, dimension( * ) e, double precision, dimension( ldz, * ) z, integer ldz, double precision, dimension( * ) work, integer info)

DSTEQR

الغرض:

!>
!> DSTEQR يحسب جميع القيم الذاتية، وبشكل اختياري، المتجهات الذاتية لمصفوفة ثلاثية قطرية متناظرة باستخدام طريقة QL أو QR الضمنية.
!> يمكن أيضًا العثور على المتجهات الذاتية لمصفوفة متناظرة كاملة أو نطاقية
!> إذا تم استخدام DSYTRD أو DSPTRD أو DSBTRD لتقليل هذه المصفوفة
!> إلى شكل ثلاثي قطري.
!>

المعلمات

COMPZ 
!>          COMPZ is CHARACTER*1
!>          = 'N':  احسب القيم الذاتية فقط.
!>          = 'V':  احسب القيم الذاتية والمتجهات الذاتية للمصفوفة الأصلية
!>                  المتناظرة. عند الإدخال، يجب أن تحتوي Z على
!>                  المصفوفة المتعامدة المستخدمة لتقليل المصفوفة الأصلية
!>                  إلى شكل ثلاثي قطري.
!>          = 'I':  احسب القيم الذاتية والمتجهات الذاتية للمصفوفة
!>                  ثلاثية القطرية. Z تم تهيئتها إلى مصفوفة الوحدة.
!>

N

!>          N is INTEGER
!>          ترتيب المصفوفة.  N >= 0.
!>

D

!>          D is DOUBLE PRECISION array, dimension (N)
!>          عند الإدخال، العناصر القطرية للمصفوفة ثلاثية القطرية.
!>          عند الخروج، إذا كان INFO = 0، فإن القيم الذاتية بترتيب تصاعدي.
!>

E

!>          E is DOUBLE PRECISION array, dimension (N-1)
!>          عند الإدخال، عناصر القطر السفلي (n-1) للمصفوفة ثلاثية القطرية.
!>          عند الخروج، تم تدمير E.
!>

Z

!>          Z is DOUBLE PRECISION array, dimension (LDZ, N)
!>          عند الإدخال، إذا كان  COMPZ = 'V'، فإن Z تحتوي على
!>          المصفوفة المتعامدة المستخدمة في الاختزال إلى شكل ثلاثي قطري.
!>          عند الخروج، إذا كان INFO = 0، فإن إذا كان COMPZ = 'V'، تحتوي Z على
!>          المتجهات الذاتية المتعامدة المنتظمة للمصفوفة المتناظرة الأصلية،
!>          وإذا كان COMPZ = 'I'، تحتوي Z على المتجهات الذاتية المتعامدة المنتظمة
!>          للمصفوفة ثلاثية القطرية المتناظرة.
!>          إذا كان COMPZ = 'N'، فلا يتم الرجوع إلى Z.
!>

LDZ

!>          LDZ is INTEGER
!>          البعد الرئيسي للمصفوفة Z.  LDZ >= 1، وإذا
!>          تم طلب المتجهات الذاتية، فإن LDZ >= max(1,N).
!>

WORK

!>          WORK is DOUBLE PRECISION array, dimension (max(1,2*N-2))
!>          إذا كان COMPZ = 'N'، فلا يتم الرجوع إلى WORK.
!>

INFO

!>          INFO is INTEGER
!>          = 0:  خروج ناجح
!>          < 0:  إذا كان INFO = -i، فإن المعلمة i كان لها قيمة غير قانونية
!>          > 0:  فشلت الخوارزمية في العثور على جميع القيم الذاتية في
!>                إجمالي 30*N تكرار؛ إذا كان INFO = i، فإن i
!>                عناصر E لم تتقارب إلى الصفر؛ عند الخروج، تحتوي D
!>                و E على عناصر مصفوفة ثلاثية قطرية متناظرة
!>                والتي تشبه متعامدة المصفوفة الأصلية.
!>
