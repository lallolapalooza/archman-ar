subroutine dsterf (integer n, double precision, dimension( * ) d, double precision, dimension( * ) e, integer info)

DSTERF

الغرض:

!>
!> DSTERF يحسب جميع القيم الذاتية لمصفوفة ثلاثية قطرية متناظرة
!> باستخدام صيغة Pal-Walker-Kahan من خوارزمية QL أو QR.
!>

المعلمات

N 
!>          N هو عدد صحيح (INTEGER)
!>          رتبة المصفوفة.  N >= 0.
!>

D

!>          D هو مصفوفة دقة مزدوجة (DOUBLE PRECISION)، البعد (N)
!>          عند الدخول، العناصر القطرية n للمصفوفة الثلاثية القطرية.
!>          عند الخروج، إذا كان INFO = 0، القيم الذاتية بترتيب تصاعدي.
!>

E

!>          E هو مصفوفة دقة مزدوجة (DOUBLE PRECISION)، البعد (N-1)
!>          عند الدخول، العناصر تحت القطرية n-1 للمصفوفة الثلاثية القطرية.
!>          عند الخروج، تم تدمير E.
!>

INFO

!>          INFO هو عدد صحيح (INTEGER)
!>          = 0:  انتهاء ناجح
!>          < 0:  إذا كان INFO = -i، فإن المعامل i كان له قيمة غير قانونية
!>          > 0:  فشلت الخوارزمية في إيجاد جميع القيم الذاتية في
!>                إجمالي 30*N تكرارات؛ إذا كان INFO = i، فإن i
!>                عناصر E لم تتقارب إلى الصفر.
!>
