subroutine dsprfs (character uplo, integer n, integer nrhs, double precision, dimension( * ) ap, double precision, dimension( * ) afp, integer, dimension( * ) ipiv, double precision, dimension( ldb, * ) b, integer ldb, double precision, dimension( ldx, * ) x, integer ldx, double precision, dimension( * ) ferr, double precision, dimension( * ) berr, double precision, dimension( * ) work, integer, dimension( * ) iwork, integer info)

DSPRFS

الغرض:

!>
!> DSPRFS يحسن الحل المحسوب لنظام المعادلات الخطية
!> عندما تكون مصفوفة المعاملات متناظرة غير محددة
!> ومعبأة، ويوفر حدود الخطأ وتقديرات الخطأ الرجعي
!> للحل.
!>

المعلمات

UPLO 
!>          UPLO is CHARACTER*1
!>          = 'U':  يتم تخزين المثلث العلوي للمصفوفة A؛
!>          = 'L':  يتم تخزين المثلث السفلي للمصفوفة A.
!>

N

!>          N is INTEGER
!>          رتبة المصفوفة A.  N >= 0.
!>

NRHS

!>          NRHS is INTEGER
!>          عدد الأطراف اليمنى، أي عدد الأعمدة
!>          للمصفوفتين B و X.  NRHS >= 0.
!>

AP

!>          AP is DOUBLE PRECISION array, dimension (N*(N+1)/2)
!>          المثلث العلوي أو السفلي للمصفوفة المتناظرة A، مجمعة
!>          عمودياً في مصفوفة خطية. يتم تخزين العمود j للمصفوفة A
!>          في المصفوفة AP كالتالي:
!>          إذا كان UPLO = 'U'، فإن AP(i + (j-1)*j/2) = A(i,j) لـ 1<=i<=j؛
!>          إذا كان UPLO = 'L'، فإن AP(i + (j-1)*(2*n-j)/2) = A(i,j) لـ j<=i<=n.
!>

AFP

!>          AFP is DOUBLE PRECISION array, dimension (N*(N+1)/2)
!>          الشكل المفكك للمصفوفة A. تحتوي AFP على المصفوفة
!>          القطرية الكتلوية D والمعاملات المستخدمة للحصول على
!>          العامل U أو L من التحليل A = U*D*U**T أو
!>          A = L*D*L**T كما تم حسابه بواسطة DSPTRF، مجمعة
!>          كمصفوفة مثلثة.
!>

IPIV

!>          IPIV is INTEGER array, dimension (N)
!>          تفاصيل التبادلات والهيكل الكتلي لـ D
!>          كما تم تحديده بواسطة DSPTRF.
!>

B

!>          B is DOUBLE PRECISION array, dimension (LDB,NRHS)
!>          مصفوفة الطرف الأيمن B.
!>

LDB

!>          LDB is INTEGER
!>          البعد الرئيسي للمصفوفة B.  LDB >= max(1,N).
!>

X

!>          X is DOUBLE PRECISION array, dimension (LDX,NRHS)
!>          عند الإدخال، مصفوفة الحل X، كما تم حسابها بواسطة DSPTRS.
!>          عند الخروج، مصفوفة الحل المحسنة X.
!>

LDX

!>          LDX is INTEGER
!>          البعد الرئيسي للمصفوفة X.  LDX >= max(1,N).
!>

FERR

!>          FERR is DOUBLE PRECISION array, dimension (NRHS)
!>          حد الخطأ الأمامي المقدر لكل متجه حل
!>          X(j) (العمود j لمصفوفة الحل X).
!>          إذا كان XTRUE هو الحل الحقيقي المقابل لـ X(j)، فإن FERR(j)
!>          هو تقدير للحد الأعلى لقيمة القيمة المطلقة لـ (X(j) - XTRUE) مقسومة على القيمة المطلقة لـ
!>          أكبر عنصر في X(j). التقدير موثوق به مثل
!>          التقدير لـ RCOND، وهو دائمًا تقريبًا أكثر بقليل من اللازم للخطأ الحقيقي.
!>

BERR

!>          BERR is DOUBLE PRECISION array, dimension (NRHS)
!>          الخطأ النسبي المكوني لكل متجه حل
!>          X(j) (أي أصغر تغيير نسبي في
!>          أي عنصر في A أو B يجعل X(j) حلاً دقيقًا).
!>

WORK

!>          WORK is DOUBLE PRECISION array, dimension (3*N)
!>

IWORK

!>          IWORK is INTEGER array, dimension (N)
!>

INFO

!>          INFO is INTEGER
!>          = 0:  إنهاء ناجح
!>          < 0:  إذا كان INFO = -i، فقد كان المعامل i قيمة غير قانونية
!>
