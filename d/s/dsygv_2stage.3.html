subroutine dsygv_2stage (integer itype, character jobz, character uplo, integer n, double precision, dimension( lda, * ) a, integer lda, double precision, dimension( ldb, * ) b, integer ldb, double precision, dimension( * ) w, double precision, dimension( * ) work, integer lwork, integer info)

DSYGV_2STAGE

الغرض:

!>
!> DSYGV_2STAGE يحسب جميع القيم الذاتية، واختياريًا، المتجهات الذاتية
!> لمشكلة قيمية متناظرة-محددة معممة حقيقية، من الشكل
!> A*x=(lambda)*B*x,  A*Bx=(lambda)*x,  or B*A*x=(lambda)*x.
!> هنا يُفترض أن A و B متناظرتان و B أيضًا
!> موجبة محددة.
!> يستخدم هذا الروتين تقنية المرحلتين (2stage) للاختزال إلى الشكل ثلاثي القطر
!> والتي أظهرت أداءً أعلى على الأجهزة الحديثة ولأحجام كبيرة N>2000.
!>

المعلمات

ITYPE 
!>          ITYPE هو عدد صحيح (INTEGER)
!>          يحدد نوع المشكلة المراد حلها:
!>          = 1:  A*x = (lambda)*B*x
!>          = 2:  A*B*x = (lambda)*x
!>          = 3:  B*A*x = (lambda)*x
!>

JOBZ

!>          JOBZ هو حرف (CHARACTER*1)
!>          = 'N':  حساب القيم الذاتية فقط؛
!>          = 'V':  حساب القيم الذاتية والمتجهات الذاتية.
!>                  غير متاح في هذا الإصدار.
!>

UPLO

!>          UPLO هو حرف (CHARACTER*1)
!>          = 'U':  المثلثات العلوية لـ A و B مخزنة؛
!>          = 'L':  المثلثات السفلية لـ A و B مخزنة.
!>

N

!>          N هو عدد صحيح (INTEGER)
!>          رتبة المصفوفتين A و B.  N >= 0.
!>

A

!>          A هي مصفوفة من الأعداد الحقيقية المزدوجة الدقة (DOUBLE PRECISION)، البعد (LDA, N)
!>          عند الإدخال، المصفوفة المتناظرة A.  إذا كان UPLO = 'U'، فإن
!>          الجزء العلوي المثلثي الذي أبعاده N*N في A يحتوي على
!>          الجزء العلوي المثلثي من المصفوفة A.  إذا كان UPLO = 'L'، فإن
!>          الجزء السفلي المثلثي الذي أبعاده N*N في A يحتوي على
!>          الجزء السفلي المثلثي من المصفوفة A.
!>
!>          عند الخروج، إذا كان JOBZ = 'V'، ثم إذا كان INFO = 0، فإن A تحتوي على
!>          المصفوفة Z للمتجهات الذاتية.  المتجهات الذاتية مُطبّعة
!>          كما يلي:
!>          إذا كان ITYPE = 1 أو 2، Z**T*B*Z = I؛
!>          إذا كان ITYPE = 3، Z**T*inv(B)*Z = I.
!>          إذا كان JOBZ = 'N'، فإنه عند الخروج، الجزء العلوي المثلثي (إذا كان UPLO='U')
!>          أو الجزء السفلي المثلثي (إذا كان UPLO='L') من A، بما في ذلك
!>          القطر الرئيسي، يتم تدميره.
!>

LDA

!>          LDA هو عدد صحيح (INTEGER)
!>          البعد الرئيسي للمصفوفة A.  LDA >= max(1,N).
!>

B

!>          B هي مصفوفة من الأعداد الحقيقية المزدوجة الدقة (DOUBLE PRECISION)، البعد (LDB, N)
!>          عند الإدخال، المصفوفة المتناظرة الموجبة المحددة B.
!>          إذا كان UPLO = 'U'، فإن الجزء العلوي المثلثي الذي أبعاده N*N في B
!>          يحتوي على الجزء العلوي المثلثي من المصفوفة B.
!>          إذا كان UPLO = 'L'، فإن الجزء السفلي المثلثي الذي أبعاده N*N في B
!>          يحتوي على الجزء السفلي المثلثي من المصفوفة B.
!>
!>          عند الخروج، إذا كان INFO <= N، فإن الجزء من B الذي يحتوي على المصفوفة
!>          يتم استبداله بالعامل المثلثي U أو L من تحليل Cholesky
!>          B = U**T*U أو B = L*L**T.
!>

LDB

!>          LDB هو عدد صحيح (INTEGER)
!>          البعد الرئيسي للمصفوفة B.  LDB >= max(1,N).
!>

W

!>          W هي مصفوفة من الأعداد الحقيقية المزدوجة الدقة (DOUBLE PRECISION)، البعد (N)
!>          إذا كان INFO = 0، فإن القيم الذاتية مرتبة تصاعديًا.
!>

WORK

!>          WORK هي مصفوفة من الأعداد الحقيقية المزدوجة الدقة (DOUBLE PRECISION)، البعد (MAX(1,LWORK))
!>          عند الخروج، إذا كان INFO = 0، فإن WORK(1) تُرجع قيمة LWORK المثلى.
!>

LWORK

!>          LWORK هو عدد صحيح (INTEGER)
!>          طول المصفوفة WORK. LWORK >= 1، عندما N <= 1؛
!>          بخلاف ذلك
!>          إذا كان JOBZ = 'N' و N > 1، يجب الاستعلام عن LWORK.
!>                                   LWORK = MAX(1, dimension) حيث
!>                                   dimension = max(stage1,stage2) + (KD+1)*N + 2*N
!>                                             = N*KD + N*max(KD+1,FACTOPTNB)
!>                                               + max(2*KD*KD, KD*NTHREADS)
!>                                               + (KD+1)*N + 2*N
!>                                   حيث KD هو حجم الكتلة للاختزال،
!>                                   FACTOPTNB هو الحجم المستخدم بواسطة خوارزمية QR أو LQ
!>                                   عادةً ما يكون FACTOPTNB=128 خيارًا جيدًا
!>                                   NTHREADS هو عدد الخيوط المستخدمة عند
!>                                   تم تجميع openMP، وإلا =1.
!>          إذا كان JOBZ = 'V' و N > 1، يجب الاستعلام عن LWORK. غير متاح بعد.
!>
!>          إذا كان LWORK = -1، فسيتم افتراض استعلام مساحة العمل؛ الروتين
!>          يحسب فقط الحجم الأمثل لمصفوفة WORK، ويُرجع
!>          هذه القيمة كالإدخال الأول للمصفوفة WORK، ولا تصدر رسالة خطأ
!>          متعلقة بـ LWORK بواسطة XERBLA.
!>

INFO

!>          INFO هو عدد صحيح (INTEGER)
!>          = 0:  انتهاء ناجح
!>          < 0:  إذا كان INFO = -i، فإن المعلمة i كانت ذات قيمة غير قانونية
!>          > 0:  DPOTRF أو DSYEV أعادت رمز خطأ:
!>             <= N:  إذا كان INFO = i، فشلت DSYEV في التقارب؛
!>                    i عناصر خارج القطر لمصفوفة ثلاثية الأبعاد وسيطة
!>                    لم تتقارب إلى الصفر؛
!>             > N:   إذا كان INFO = N + i، حيث 1 <= i <= N، فإن
!>                    المُعدِّل الرئيسي الجزئي من الرتبة i لـ B ليس موجبًا.
!>                    لم يتمكن من إكمال تحليل B
!>                    ولم يتم حساب أي قيم ذاتية أو متجهات ذاتية.
!>
