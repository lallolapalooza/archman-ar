توثيق الوظيفة/الروتين الفرعي

subroutine dsyt01 (character uplo, integer n, double precision, dimension( lda, * ) a, integer lda, double precision, dimension( ldafac, * ) afac, integer ldafac, integer, dimension( * ) ipiv, double precision, dimension( ldc, * ) c, integer ldc, double precision, dimension( * ) rwork, double precision resid)

DSYT01

الغرض:

!>
!> DSYT01 يعيد بناء مصفوفة متماثلة غير محددة A من
!> عامل L*D*L' أو U*D*U' الخاص بها ويحسب المتبقي
!>    norm( C - A ) / ( N * norm(A) * EPS ),
!> حيث C هي المصفوفة المعاد بناؤها و EPS هو الخطأ المطبعي للآلة.
!>

المعلمات

UPLO 
!>          UPLO is CHARACTER*1
!>          يحدد ما إذا كان الجزء العلوي أو السفلي المثلثي للمصفوفة
!>          المتماثلة A مخزنًا:
!>          = 'U':  مثلث علوي
!>          = 'L':  مثلث سفلي
!>

N

!>          N is INTEGER
!>          عدد الصفوف والأعمدة للمصفوفة A.  N >= 0.
!>

A

!>          A is DOUBLE PRECISION array, dimension (LDA,N)
!>          المصفوفة المتماثلة الأصلية A.
!>

LDA

!>          LDA is INTEGER
!>          البعد الرئيسي للمصفوفة A.  LDA >= max(1,N)
!>

AFAC

!>          AFAC is DOUBLE PRECISION array, dimension (LDAFAC,N)
!>          الشكل المفكك للمصفوفة A.  AFAC يحتوي على المصفوفة
!>          القطرية الكتلية D والعوامل المستخدمة للحصول على
!>          العامل L أو U من عامل L*D*L' أو U*D*U' الكتلي
!>          المحسوب بواسطة DSYTRF.
!>

LDAFAC

!>          LDAFAC is INTEGER
!>          البعد الرئيسي للمصفوفة AFAC.  LDAFAC >= max(1,N).
!>

IPIV

!>          IPIV is INTEGER array, dimension (N)
!>          مؤشرات المحور من DSYTRF.
!>

C

!>          C is DOUBLE PRECISION array, dimension (LDC,N)
!>

LDC

!>          LDC is INTEGER
!>          البعد الرئيسي للمصفوفة C.  LDC >= max(1,N).
!>

RWORK

!>          RWORK is DOUBLE PRECISION array, dimension (N)
!>

RESID

!>          RESID is DOUBLE PRECISION
!>          إذا كان UPLO = 'L'، norm(L*D*L' - A) / ( N * norm(A) * EPS )
!>          إذا كان UPLO = 'U'، norm(U*D*U' - A) / ( N * norm(A) * EPS )
!>
