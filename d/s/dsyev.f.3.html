subroutine dsyev (jobz, uplo, n, a, lda, w, work, lwork, info)
DSYEV يحسب القيم الذاتية، واختيارياً، المتجهات الذاتية اليسرى و/أو اليمنى لمصفوفات SY

توثيق الوظيفة/الإجراء الفرعي

subroutine dsyev (character jobz, character uplo, integer n, double precision, dimension( lda, * ) a, integer lda, double precision, dimension( * ) w, double precision, dimension( * ) work, integer lwork, integer info)

DSYEV يحسب القيم الذاتية، واختيارياً، المتجهات الذاتية لمصفوفات SY

الغرض:

!>
!> DSYEV يحسب جميع القيم الذاتية، واختيارياً، المتجهات الذاتية لمصفوفة
!> متناظرة حقيقية A.
!>

المعلمات

JOBZ 
!>          JOBZ هو CHARACTER*1
!>          = 'N':  حساب القيم الذاتية فقط؛
!>          = 'V':  حساب القيم الذاتية والمتجهات الذاتية.
!>

UPLO

!>          UPLO هو CHARACTER*1
!>          = 'U':  يتم تخزين المثلث العلوي من A؛
!>          = 'L':  يتم تخزين المثلث السفلي من A.
!>

N

!>          N هو INTEGER
!>          رتبة المصفوفة A.  N >= 0.
!>

A

!>          A هي مصفوفة DOUBLE PRECISION، البعد (LDA, N)
!>          عند الدخول، المصفوفة المتناظرة A.  إذا كان UPLO = 'U'،
!>          الجزء العلوي المثلثي N-by-N من A يحتوي على
!>          الجزء العلوي المثلثي للمصفوفة A.  إذا كان UPLO = 'L'،
!>          الجزء السفلي المثلثي N-by-N من A يحتوي على
!>          الجزء السفلي المثلثي للمصفوفة A.
!>          عند الخروج، إذا كان JOBZ = 'V'، ثم إذا كان INFO = 0، تحتوي A على
!>          المتجهات الذاتية المتعامدة للمصفوفة A.
!>          إذا كان JOBZ = 'N'، ثم عند الخروج يتم تدمير المثلث السفلي (إذا كان UPLO='L')
!>          أو المثلث العلوي (إذا كان UPLO='U') من A، بما في ذلك
!>          القطر.
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرئيسي للمصفوفة A.  LDA >= max(1,N).
!>

W

!>          W هي مصفوفة DOUBLE PRECISION، البعد (N)
!>          إذا كان INFO = 0، القيم الذاتية بترتيب تصاعدي.
!>

WORK

!>          WORK هي مصفوفة DOUBLE PRECISION، البعد (MAX(1,LWORK))
!>          عند الخروج، إذا كان INFO = 0، تعيد WORK(1) قيمة LWORK المثلى.
!>

LWORK

!>          LWORK هو INTEGER
!>          طول المصفوفة WORK.  LWORK >= max(1,3*N-1).
!>          لأقصى كفاءة، LWORK >= (NB+2)*N،
!>          حيث NB هو حجم الكتلة لـ DSYTRD الذي يتم إرجاعه بواسطة ILAENV.
!>
!>          إذا كان LWORK = -1، فسيتم افتراض استعلام عن مساحة العمل؛ تقوم الروتين
!>          بحساب الحجم الأمثل لمصفوفة WORK فقط، وتعيد
!>          هذه القيمة كإدخال أول لمصفوفة WORK، ولا يتم إصدار أي رسالة خطأ
!>          متعلقة بـ LWORK بواسطة XERBLA.
!>

INFO

!>          INFO هو INTEGER
!>          = 0:  انتهاء ناجح
!>          < 0:  إذا كان INFO = -i، كان الوسيط i له قيمة غير قانونية
!>          > 0:  إذا كان INFO = i، فشلت الخوارزمية في التقارب؛ i
!>                العناصر غير القطرية لشكل ثلاثي الأقطار وسيط
!>                لم تتقارب إلى الصفر.
!>
