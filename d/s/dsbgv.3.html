subroutine dsbgv (character jobz, character uplo, integer n, integer ka, integer kb, double precision, dimension( ldab, * ) ab, integer ldab, double precision, dimension( ldbb, * ) bb, integer ldbb, double precision, dimension( * ) w, double precision, dimension( ldz, * ) z, integer ldz, double precision, dimension( * ) work, integer info)

DSBGV

الغرض:

!>
!> DSBGV يحسب جميع القيم الذاتية، واختيارياً، المتجهات الذاتية
!> لمشكلة قيمة خاصة متماثلة محددة مع نطاق عام حقيقي،
!> للصيغة A*x=(lambda)*B*x. هنا يُفترض أن A و B متماثلتان
!> وضيقتا النطاق، وأن B موجبة محددة.
!>

المعلمات

JOBZ 
!>          JOBZ is CHARACTER*1
!>          = 'N':  حساب القيم الذاتية فقط؛
!>          = 'V':  حساب القيم الذاتية والمتجهات الذاتية.
!>

UPLO

!>          UPLO is CHARACTER*1
!>          = 'U':  المثلثات العلوية لـ A و B مخزنة؛
!>          = 'L':  المثلثات السفلية لـ A و B مخزنة.
!>

N

!>          N is INTEGER
!>          ترتيب المصفوفات A و B.  N >= 0.
!>

KA

!>          KA is INTEGER
!>          عدد الأقطار الفائقة للمصفوفة A إذا كان UPLO = 'U'،
!>          أو عدد الأقطار الفرعية إذا كان UPLO = 'L'. KA >= 0.
!>

KB

!>          KB is INTEGER
!>          عدد الأقطار الفائقة للمصفوفة B إذا كان UPLO = 'U'،
!>          أو عدد الأقطار الفرعية إذا كان UPLO = 'L'. KB >= 0.
!>

AB

!>          AB is DOUBLE PRECISION array, dimension (LDAB, N)
!>          عند الإدخال، المثلث العلوي أو السفلي للمصفوفة المتماثلة
!>          ذات النطاق A، مخزنة في أول ka+1 صفوف من المصفوفة.  العمود
!>          j من A مخزن في العمود j من المصفوفة AB
!>          كما يلي:
!>          إذا كان UPLO = 'U'، AB(ka+1+i-j,j) = A(i,j) لـ max(1,j-ka)<=i<=j؛
!>          إذا كان UPLO = 'L'، AB(1+i-j,j)    = A(i,j) لـ j<=i<=min(n,j+ka).
!>
!>          عند الخروج، يتم تدمير محتويات AB.
!>

LDAB

!>          LDAB is INTEGER
!>          البعد الرائد للمصفوفة AB.  LDAB >= KA+1.
!>

BB

!>          BB is DOUBLE PRECISION array, dimension (LDBB, N)
!>          عند الإدخال، المثلث العلوي أو السفلي للمصفوفة المتماثلة
!>          ذات النطاق B، مخزنة في أول kb+1 صفوف من المصفوفة.  العمود
!>          j من B مخزن في العمود j من المصفوفة BB
!>          كما يلي:
!>          إذا كان UPLO = 'U'، BB(kb+1+i-j,j) = B(i,j) لـ max(1,j-kb)<=i<=j؛
!>          إذا كان UPLO = 'L'، BB(1+i-j,j)    = B(i,j) لـ j<=i<=min(n,j+kb).
!>
!>          عند الخروج، العامل S من تحليل Cholesky المنفصل
!>          B = S**T*S، كما يتم إرجاعه بواسطة DPBSTF.
!>

LDBB

!>          LDBB is INTEGER
!>          البعد الرائد للمصفوفة BB.  LDBB >= KB+1.
!>

W

!>          W is DOUBLE PRECISION array, dimension (N)
!>          إذا كان INFO = 0، فالقيم الذاتية مرتبة تصاعدياً.
!>

Z

!>          Z is DOUBLE PRECISION array, dimension (LDZ, N)
!>          إذا كان JOBZ = 'V'، فعندما يكون INFO = 0، تحتوي Z على المصفوفة Z
!>          للمتجهات الذاتية، حيث يحمل العمود i من Z المتجه الذاتي
!>          المرتبط بـ W(i). يتم تطبيع المتجهات الذاتية بحيث Z**T*B*Z = I.
!>          إذا كان JOBZ = 'N'، فلن يتم استخدام Z.
!>

LDZ

!>          LDZ is INTEGER
!>          البعد الرائد للمصفوفة Z.  LDZ >= 1، وإذا كان
!>          JOBZ = 'V'، فإن LDZ >= N.
!>

WORK

!>          WORK is DOUBLE PRECISION array, dimension (3*N)
!>

INFO

!>          INFO is INTEGER
!>          = 0:  انتهاء ناجح
!>          < 0:  إذا كان INFO = -i، فقد كان الوسيط i غير صالح
!>          > 0:  إذا كان INFO = i، و i هو:
!>             <= N:  فشل الخوارزمية في التقارب:
!>                    i عنصر خارج القطر لمصفوفة ثلاثية الأقطار وسيطة
!>                    لم تتقارب إلى الصفر؛
!>             > N:   إذا كان INFO = N + i، لـ 1 <= i <= N، فإن DPBSTF
!>                    أرجع INFO = i: B ليست موجبة محددة.
!>                    تعذر إكمال تحليل B ولم يتم حساب أي قيم ذاتية أو
!>                    متجهات ذاتية.
!>
