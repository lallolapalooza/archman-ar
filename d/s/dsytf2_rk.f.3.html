subroutine dsytf2_rk (uplo, n, a, lda, e, ipiv, info)
DSYTF2_RK يحسب تحليل مصفوفة حقيقية متناظرة غير محددة باستخدام طريقة التنقيب القطري المقيدة لـ Bunch-Kaufman (rokk) (خوارزمية BLAS2 غير المحظورة).

توثيق الوظيفة/الإجراء الفرعي

subroutine dsytf2_rk (character uplo, integer n, double precision, dimension( lda, * ) a, integer lda, double precision, dimension( * ) e, integer, dimension( * ) ipiv, integer info)

DSYTF2_RK يحسب تحليل مصفوفة حقيقية متناظرة غير محددة باستخدام طريقة التنقيب القطري المقيدة لـ Bunch-Kaufman (rook) (خوارزمية BLAS2 غير المحظورة).

الغرض:

!> DSYTF2_RK يحسب تحليل مصفوفة حقيقية متناظرة A
!> باستخدام طريقة التنقيب القطري المقيدة لـ Bunch-Kaufman (rook):
!>
!>    A = P*U*D*(U**T)*(P**T) أو A = P*L*D*(L**T)*(P**T)،
!>
!> حيث U (أو L) هي مصفوفة مثلثية علوية (أو سفلية) وحيدة،
!> U**T (أو L**T) هي معكوس U (أو L)، P هي مصفوفة تبديل، P**T هي معكوس P، و D هي مصفوفة متناظرة وقطرية كتلية مع كتل قطرية 1x1 و 2x2.
!>
!> هذه هي النسخة غير المحظورة من الخوارزمية، تستدعي BLAS المستوى 2.
!> لمزيد من المعلومات انظر قسم التفاصيل الإضافية.
!>

المعاملات

UPLO 
!>          UPLO هو CHARACTER*1
!>          يحدد ما إذا كان الجزء المثلثي العلوي أو السفلي من المصفوفة المتناظرة A مخزنًا:
!>          = 'U':  مثلثي علوي
!>          = 'L':  مثلثي سفلي
!>

N

!>          N هو INTEGER
!>          رتبة المصفوفة A.  N >= 0.
!>

A

!>          A هو مصفوفة DOUBLE PRECISION، أبعادها (LDA,N)
!>          عند الدخول، المصفوفة المتناظرة A.
!>            إذا كان UPLO = 'U': الجزء العلوي المثلثي N×N العلوي من A يحتوي على الجزء العلوي المثلثي للمصفوفة A،
!>            والجزء السفلي المثلثي الصارم من A لا تتم الإشارة إليه.
!>
!>            إذا كان UPLO = 'L': الجزء السفلي المثلثي N×N السفلي من A يحتوي على الجزء السفلي المثلثي للمصفوفة A،
!>            والجزء العلوي المثلثي الصارم من A لا تتم الإشارة إليه.
!>
!>          عند الخروج، تحتوي على:
!>            أ) عناصر قطرية فقط من المصفوفة القطرية الكتلية المتناظرة D على قطر A، أي D(k,k) = A(k,k)؛
!>               (العناصر فوق القطرية (أو تحت القطرية) من D
!>                مخزنة عند الخروج في المصفوفة E)، و
!>            ب) إذا كان UPLO = 'U': العامل U في الجزء فوق القطري من A.
!>               إذا كان UPLO = 'L': العامل L في الجزء تحت القطري من A.
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرائد للمصفوفة A.  LDA >= max(1,N).
!>

E

!>          E هو مصفوفة DOUBLE PRECISION، أبعادها (N)
!>          عند الخروج، تحتوي على العناصر فوق القطرية (أو تحت القطرية)
!>          للمصفوفة القطرية الكتلية المتناظرة D
!>          مع كتل قطرية 1x1 أو 2x2، حيث
!>          إذا كان UPLO = 'U': E(i) = D(i-1,i)، i=2:N، E(1) يتم تعيينه إلى 0؛
!>          إذا كان UPLO = 'L': E(i) = D(i+1,i)، i=1:N-1، E(N) يتم تعيينه إلى 0.
!>
!>          ملاحظة: لكتلة قطرية 1x1 D(k)، حيث
!>          1 <= k <= N، يتم تعيين العنصر E(k) إلى 0 في كلتا
!>          حالتَي UPLO = 'U' أو UPLO = 'L'.
!>

IPIV

!>          IPIV هو مصفوفة INTEGER، أبعادها (N)
!>          IPIV تصف مصفوفة التبديل P في تحليل
!>          المصفوفة A كالتالي. القيمة المطلقة لـ IPIV(k)
!>          تمثل فهرس الصف والعمود اللذين تم
!>          تبديلهما مع الصف والعمود k-th. قيمة UPLO
!>          تصف ترتيب تطبيق التبادلات.
!>          أيضًا، علامة IPIV تمثل البنية الكتلية
!>          للمصفوفة القطرية الكتلية المتناظرة D مع كتل قطرية 1x1 أو 2x2
!>          والتي تتوافق مع تبادل 1 أو 2
!>          في كل خطوة تحليل. لمزيد من المعلومات انظر قسم التفاصيل الإضافية.
!>
!>          إذا كان UPLO = 'U'،
!>          (في ترتيب التحليل، k تتناقص من N إلى 1):
!>            أ) إدخال إيجابي واحد IPIV(k) > 0 يعني:
!>               D(k,k) هي كتلة قطرية 1x1.
!>               إذا كان IPIV(k) != k، تم تبديل الصفوف والأعمدة k و IPIV(k)
!>               في المصفوفة A(1:N,1:N)؛
!>               إذا كان IPIV(k) = k، لم يحدث تبديل.
!>
!>            ب) زوج من الإدخالات المتتالية السالبة
!>               IPIV(k) < 0 و IPIV(k-1) < 0 يعني:
!>               D(k-1:k,k-1:k) هي كتلة قطرية 2x2.
!>               (ملاحظة: الإدخالات السالبة في IPIV تظهر في أزواج فقط).
!>               1) إذا كان -IPIV(k) != k، تم تبديل الصفوف والأعمدة
!>                  k و -IPIV(k)
!>                  في المصفوفة A(1:N,1:N).
!>                  إذا كان -IPIV(k) = k، لم يحدث تبديل.
!>               2) إذا كان -IPIV(k-1) != k-1، تم تبديل الصفوف والأعمدة
!>                  k-1 و -IPIV(k-1)
!>                  في المصفوفة A(1:N,1:N).
!>                  إذا كان -IPIV(k-1) = k-1، لم يحدث تبديل.
!>
!>            ج) في كلتا الحالتين أ) وب)، دائمًا ABS( IPIV(k) ) <= k.
!>
!>            د) ملاحظة: أي إدخال IPIV(k) هو دائمًا غير صفري عند الخروج.
!>
!>          إذا كان UPLO = 'L'،
!>          (في ترتيب التحليل، k تزداد من 1 إلى N):
!>            أ) إدخال إيجابي واحد IPIV(k) > 0 يعني:
!>               D(k,k) هي كتلة قطرية 1x1.
!>               إذا كان IPIV(k) != k، تم تبديل الصفوف والأعمدة k و IPIV(k)
!>               في المصفوفة A(1:N,1:N).
!>               إذا كان IPIV(k) = k، لم يحدث تبديل.
!>
!>            ب) زوج من الإدخالات المتتالية السالبة
!>               IPIV(k) < 0 و IPIV(k+1) < 0 يعني:
!>               D(k:k+1,k:k+1) هي كتلة قطرية 2x2.
!>               (ملاحظة: الإدخالات السالبة في IPIV تظهر في أزواج فقط).
!>               1) إذا كان -IPIV(k) != k، تم تبديل الصفوف والأعمدة
!>                  k و -IPIV(k)
!>                  في المصفوفة A(1:N,1:N).
!>                  إذا كان -IPIV(k) = k، لم يحدث تبديل.
!>               2) إذا كان -IPIV(k+1) != k+1، تم تبديل الصفوف والأعمدة
!>                  k-1 و -IPIV(k-1)
!>                  في المصفوفة A(1:N,1:N).
!>                  إذا كان -IPIV(k+1) = k+1، لم يحدث تبديل.
!>
!>            ج) في كلتا الحالتين أ) وب)، دائمًا ABS( IPIV(k) ) >= k.
!>
!>            د) ملاحظة: أي إدخال IPIV(k) هو دائمًا غير صفري عند الخروج.
!>

INFO

!>          INFO هو INTEGER
!>          = 0: خروج ناجح
!>
!>          < 0: إذا كان INFO = -k، كان المعامل k غير قانوني.
!>
!>          > 0: إذا كان INFO = k، فإن المصفوفة A منفردة، وذلك بسبب:
!>                 إذا كان UPLO = 'U': العمود k في الجزء
!>                 المثلثي العلوي من A يحتوي على جميع الأصفار.
!>                 إذا كان UPLO = 'L': العمود k في الجزء
!>                 المثلثي السفلي من A يحتوي على جميع الأصفار.
!>
!>               لذلك D(k,k) يساوي صفر تمامًا، والعناصر فوق القطرية
!>               للعمود k من U (أو العناصر تحت القطرية لـ
!>               العمود k من L) كلها أصفار. تم إكمال التحليل،
!>               ولكن المصفوفة القطرية الكتلية D منفردة تمامًا، وسيحدث
!>               القسمة على صفر إذا تم استخدامها لحل نظام معادلات.
!>
!>               ملاحظة: INFO يخزن فقط أول حدوث
!>               للتفرد، أي حدوث لاحق للتفرد
!>               لا يتم تخزينه في INFO على الرغم من أن التحليل
!>               يكتمل دائمًا.
!>
