subroutine dsteqr (character compz, integer n, double precision, dimension( * ) d, double precision, dimension( * ) e, double precision, dimension( ldz, * ) z, integer ldz, double precision, dimension( * ) work, integer info)

DSTEQR

الغرض:

!>
!> DSTEQR تحسب جميع القيم الذاتية، واختيارياً، المتجهات الذاتية لمصفوفة ثلاثية متناظرة
!> باستخدام طريقة QL أو QR الضمنية.
!> يمكن أيضاً إيجاد المتجهات الذاتية لمصفوفة متناظرة كاملة أو شريطية
!> إذا تم استخدام DSYTRD أو DSPTRD أو DSBTRD لتقليل هذه المصفوفة
!> إلى شكل ثلاثي.
!>

المعلمات

COMPZ 
!>          COMPZ is CHARACTER*1
!>          = 'N':  Compute eigenvalues only. (حساب القيم الذاتية فقط.)
!>          = 'V':  Compute eigenvalues and eigenvectors of the original
!>                  symmetric matrix.  On entry, Z must contain the
!>                  orthogonal matrix used to reduce the original matrix
!>                  to tridiagonal form. (حساب القيم الذاتية والمتجهات الذاتية
!>                  للمصفوفة المتناظرة الأصلية. عند الدخول، يجب أن تحتوي Z على
!>                  المصفوفة المتعامدة المستخدمة لتقليل المصفوفة الأصلية
!>                  إلى شكل ثلاثي.)
!>          = 'I':  Compute eigenvalues and eigenvectors of the
!>                  tridiagonal matrix.  Z is initialized to the identity
!>                  matrix. (حساب القيم الذاتية والمتجهات الذاتية
!>                  للمصفوفة الثلاثية. يتم تهيئة Z إلى مصفوفة الوحدة.)
!>

N

!>          N is INTEGER
!>          The order of the matrix.  N >= 0. (رتبة المصفوفة. N >= 0.)
!>

D

!>          D is DOUBLE PRECISION array, dimension (N)
!>          On entry, the diagonal elements of the tridiagonal matrix.
!>          On exit, if INFO = 0, the eigenvalues in ascending order.
!>          (عند الدخول، العناصر القطرية للمصفوفة الثلاثية.
!>          عند الخروج، إذا كانت INFO = 0، القيم الذاتية بترتيب تصاعدي.)
!>

E

!>          E is DOUBLE PRECISION array, dimension (N-1)
!>          On entry, the (n-1) subdiagonal elements of the tridiagonal
!>          matrix.
!>          On exit, E has been destroyed.
!>          (عند الدخول، عناصر تحت القطر (n-1) للمصفوفة الثلاثية.
!>          عند الخروج، تم تدمير E.)
!>

Z

!>          Z is DOUBLE PRECISION array, dimension (LDZ, N)
!>          On entry, if  COMPZ = 'V', then Z contains the orthogonal
!>          matrix used in the reduction to tridiagonal form.
!>          On exit, if INFO = 0, then if  COMPZ = 'V', Z contains the
!>          orthonormal eigenvectors of the original symmetric matrix,
!>          and if COMPZ = 'I', Z contains the orthonormal eigenvectors
!>          of the symmetric tridiagonal matrix.
!>          If COMPZ = 'N', then Z is not referenced.
!>          (عند الدخول، إذا كانت COMPZ = 'V'، فإن Z تحتوي على المصفوفة
!>          المتعامدة المستخدمة في الاختزال إلى شكل ثلاثي.
!>          عند الخروج، إذا كانت INFO = 0، فإن Z تحتوي على المتجهات الذاتية
!>          المتعامدة للمصفوفة المتناظرة الأصلية إذا كانت COMPZ = 'V'،
!>          وإذا كانت COMPZ = 'I'، فإن Z تحتوي على المتجهات الذاتية
!>          المتعامدة للمصفوفة الثلاثية المتناظرة.
!>          إذا كانت COMPZ = 'N'، فلن يتم الرجوع إلى Z.)
!>

LDZ

!>          LDZ is INTEGER
!>          The leading dimension of the array Z.  LDZ >= 1, and if
!>          eigenvectors are desired, then  LDZ >= max(1,N).
!>          (البعد الرائد للمصفوفة Z. LDZ >= 1، وإذا كانت المتجهات الذاتية مطلوبة،
!>          فإن LDZ >= max(1,N).)
!>

WORK

!>          WORK is DOUBLE PRECISION array, dimension (max(1,2*N-2))
!>          If COMPZ = 'N', then WORK is not referenced.
!>          (WORK هي مصفوفة DOUBLE PRECISION، البعد (max(1,2*N-2)).
!>          إذا كانت COMPZ = 'N'، فلن يتم الرجوع إلى WORK.)
!>

INFO

!>          INFO is INTEGER
!>          = 0:  successful exit (خروج ناجح)
!>          < 0:  if INFO = -i, the i-th argument had an illegal value
!>                (إذا كانت INFO = -i، كانت الوسيطة i ذات قيمة غير قانونية)
!>          > 0:  the algorithm has failed to find all the eigenvalues in
!>                a total of 30*N iterations; if INFO = i, then i
!>                elements of E have not converged to zero; on exit, D
!>                and E contain the elements of a symmetric tridiagonal
!>                matrix which is orthogonally similar to the original
!>                matrix.
!>                (فشل الخوارزمية في العثور على جميع القيم الذاتية في
!>                إجمالي 30*N تكرارًا؛ إذا كانت INFO = i، فإن i
!>                عنصرًا من E لم تتقارب إلى الصفر؛ عند الخروج، تحتوي D
!>                و E على عناصر مصفوفة ثلاثية متناظرة
!>                متشابهة بشكل متعامد مع المصفوفة الأصلية.)
!>
