subroutine dsyt22 (integer itype, character uplo, integer n, integer m, integer kband, double precision, dimension( lda, * ) a, integer lda, double precision, dimension( * ) d, double precision, dimension( * ) e, double precision, dimension( ldu, * ) u, integer ldu, double precision, dimension( ldv, * ) v, integer ldv, double precision, dimension( * ) tau, double precision, dimension( * ) work, double precision, dimension( 2 ) result)

DSYT22

الغرض:

!>
!>      DSYT22  بشكل عام يتحقق من تفكيك من الشكل
!>
!>              A U = U S
!>
!>      حيث A متماثلة، وأعمدة U متعامدة، و S قطرية (إذا كان KBAND=0) أو
!>      متماثلة شبه قطرية (إذا كان KBAND=1). إذا كان ITYPE=1، فإن U ممثلة
!>      كمصفوفة كثيفة، وإلا فإن U معبر عنها كحاصل ضرب تحويلات هاوسهولدر،
!>      التي يتم تخزين المتجهات الخاصة بها في المصفوفة  وتوابعها الثابتة في
!>      المصفوفة . سنستخدم الرمز
!>       للإشارة إلى حاصل ضرب تحويلات هاوسهولدر (والذي يجب أن يساوي U).
!>      بشكل خاص، إذا كان ITYPE=1، فإن:
!>
!>              RESULT(1) = | U**T A U - S | / ( |A| m ulp ) و
!>              RESULT(2) = | I - U**T U | / ( m ulp )
!> 
!>  ITYPE   INTEGER
!>          يحدد نوع الاختبارات التي سيتم إجراؤها.
!>          1: U ممثلة كمصفوفة متعامدة كثيفة:
!>             RESULT(1) = | A - U S U**T | / ( |A| n ulp ) و
!>             RESULT(2) = | I - U U**T | / ( n ulp )
!>
!>  UPLO    CHARACTER
!>          إذا كان UPLO='U'، فسيتم استخدام المثلث العلوي لـ A ولن يتم
!>          الوصول إلى المثلث السفلي (الصارم). إذا كان UPLO='L'، فسيتم
!>          استخدام المثلث السفلي لـ A ولن يتم الوصول إلى المثلث العلوي (الصارم).
!>          لم يتم تعديله.
!>
!>  N       INTEGER
!>          حجم المصفوفة. إذا كان صفراً، فإن DSYT22 لا تفعل شيئًا.
!>          يجب أن يكون على الأقل صفرًا.
!>          لم يتم تعديله.
!>
!>  M       INTEGER
!>          عدد أعمدة U. إذا كان صفراً، فإن DSYT22 لا تفعل شيئًا.
!>          يجب أن يكون على الأقل صفرًا.
!>          لم يتم تعديله.
!>
!>  KBAND   INTEGER
!>          عرض النطاق الترددي للمصفوفة. يمكن أن يكون صفرًا أو واحدًا فقط.
!>          إذا كان صفرًا، فإن S قطرية، ولا يتم استخدام E. إذا كان واحدًا،
!>          فإن S متماثلة شبه قطرية.
!>          لم يتم تعديله.
!>
!>  A       DOUBLE PRECISION array, dimension (LDA , N)
!>          المصفوفة الأصلية (غير المفككة). يُفترض أنها متماثلة، وسيتم
!>          الوصول فقط إلى الجزء العلوي (UPLO='U') أو الجزء السفلي (UPLO='L')
!>          فقط.
!>          لم يتم تعديله.
!>
!>  LDA     INTEGER
!>          البعد الرائد لـ A. يجب أن يكون على الأقل 1 وعلى الأقل N.
!>          لم يتم تعديله.
!>
!>  D       DOUBLE PRECISION array, dimension (N)
!>          القطر الرئيسي للمصفوفة (المتماثلة شبه القطرية).
!>          لم يتم تعديله.
!>
!>  E       DOUBLE PRECISION array, dimension (N)
!>          العناصر خارج القطر الرئيسي للمصفوفة (المتماثلة شبه القطرية).
!>          يتم تجاهل E(1)، و E(2) هو العنصر (1,2) و (2,1)، وما إلى ذلك.
!>          لا يتم استخدامه إذا كان KBAND=0.
!>          لم يتم تعديله.
!>
!>  U       DOUBLE PRECISION array, dimension (LDU, N)
!>          إذا كان ITYPE=1 أو 3، فهذه تحتوي على المصفوفة المتعامدة في
!>          التفكيك، معبرًا عنها كمصفوفة كثيفة. إذا كان ITYPE=2، فلن يتم
!>          استخدامها.
!>          لم يتم تعديله.
!>
!>  LDU     INTEGER
!>          البعد الرائد لـ U. يجب أن يكون LDU على الأقل N وعلى الأقل 1.
!>          لم يتم تعديله.
!>
!>  V       DOUBLE PRECISION array, dimension (LDV, N)
!>          إذا كان ITYPE=2 أو 3، فإن المثلث السفلي لهذه المصفوفة يحتوي
!>          على متجهات هاوسهولدر التي تصف المصفوفة المتعامدة في التفكيك.
!>          إذا كان ITYPE=1، فلن يتم استخدامها.
!>          لم يتم تعديله.
!>
!>  LDV     INTEGER
!>          البعد الرائد لـ V. يجب أن يكون LDV على الأقل N وعلى الأقل 1.
!>          لم يتم تعديله.
!>
!>  TAU     DOUBLE PRECISION array, dimension (N)
!>          إذا كان ITYPE >= 2، فإن TAU(j) هو العامل القياسي لـ
!>          v(j) v(j)**T في تحويل هاوسهولدر H(j) لـ
!>          حاصل الضرب  U = H(1)...H(n-2)
!>          إذا كان ITYPE < 2، فلن يتم استخدام TAU.
!>          لم يتم تعديله.
!>
!>  WORK    DOUBLE PRECISION array, dimension (2*N**2)
!>          مساحة العمل.
!>          تم تعديله.
!>
!>  RESULT  DOUBLE PRECISION array, dimension (2)
!>          القيم المحسوبة بواسطة الاختبارين الموصوفين أعلاه.
!>          القيم محدودة حاليًا بـ 1/ulp، لتجنب تجاوز السعة.
!>          يتم تعديل RESULT(1) دائمًا. يتم تعديل RESULT(2) فقط
!>          إذا كان LDU على الأقل N.
!>          تم تعديله.
!>
