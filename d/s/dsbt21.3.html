توثيق الوظيفة/الروتين الفرعي

subroutine dsbt21 (character uplo, integer n, integer ka, integer ks, double precision, dimension( lda, * ) a, integer lda, double precision, dimension( * ) d, double precision, dimension( * ) e, double precision, dimension( ldu, * ) u, integer ldu, double precision, dimension( * ) work, double precision, dimension( 2 ) result)

DSBT21

الغرض:

!>
!> DSBT21  بشكل عام يتحقق من تفكيك بالشكل
!>
!>         A = U S U**T
!>
!> حيث **T يعني تبديل (transpose)، A متناظر شريطي، U متعامد،
!> و S قطري (إذا كان KS=0) أو متناظر ثلاثي الأقطار (إذا كان KS=1).
!>
!> على وجه التحديد:
!>
!>         RESULT(1) = | A - U S U**T | / ( |A| n ulp ) و
!>         RESULT(2) = | I - U U**T | / ( n ulp )
!>

المعلمات:

UPLO 
!>          UPLO is CHARACTER
!>          إذا كان UPLO='U'، سيتم استخدام المثلث العلوي للمصفوفتين A و V،
!>          ولن تتم الإشارة إلى المثلث السفلي (الصارم).
!>          إذا كان UPLO='L'، سيتم استخدام المثلث السفلي للمصفوفتين A و V،
!>          ولن تتم الإشارة إلى المثلث العلوي (الصارم).
!>

N

!>          N is INTEGER
!>          حجم المصفوفة. إذا كان صفرًا، فإن DSBT21 لا تفعل شيئًا.
!>          يجب أن يكون أكبر من أو يساوي صفرًا.
!>

KA

!>          KA is INTEGER
!>          عرض النطاق الترددي للمصفوفة A. يجب أن يكون أكبر من أو يساوي صفرًا.
!>          إذا كان أكبر من N-1، فسيتم استخدام max( 0, N-1 ).
!>

KS

!>          KS is INTEGER
!>          عرض النطاق الترددي للمصفوفة S. يمكن أن يكون صفرًا أو واحدًا فقط.
!>          إذا كان صفرًا، فإن S قطرية، ولا تتم الإشارة إلى E.
!>          إذا كان واحدًا، فإن S متناظرة ثلاثية الأقطار.
!>

A

!>          A is DOUBLE PRECISION array, dimension (LDA, N)
!>          المصفوفة الأصلية (غير المفككة). يُفترض أنها متناظرة،
!>          وسيتم الإشارة فقط إلى الجزء العلوي (UPLO='U') أو الجزء السفلي (UPLO='L') فقط.
!>

LDA

!>          LDA is INTEGER
!>          البعد الرائد للمصفوفة A. يجب أن يكون أكبر من أو يساوي 1
!>          وأكبر من أو يساوي min( KA, N-1 ).
!>

D

!>          D is DOUBLE PRECISION array, dimension (N)
!>          القطر للمصفوفة S (ثلاثية الأقطار المتناظرة).
!>

E

!>          E is DOUBLE PRECISION array, dimension (N-1)
!>          العناصر خارج القطر للمصفوفة S (ثلاثية الأقطار المتناظرة).
!>          E(1) هو العنصر (1,2) و (2,1)، E(2) هو العنصر (2,3) و (3,2)، إلخ.
!>          لا تتم الإشارة إليه إذا كان KS=0.
!>

U

!>          U is DOUBLE PRECISION array, dimension (LDU, N)
!>          المصفوفة المتعامدة في التفكيك، معبر عنها كمصفوفة كثيفة
!>          (أي، ليست كناتج تحويلات هاوسهولدر، تحويلات جيفنز، إلخ.)
!>

LDU

!>          LDU is INTEGER
!>          البعد الرائد للمصفوفة U. يجب أن يكون LDU أكبر من أو يساوي N
!>          وأكبر من أو يساوي 1.
!>

WORK

!>          WORK is DOUBLE PRECISION array, dimension (N**2+N)
!>

RESULT

!>          RESULT is DOUBLE PRECISION array, dimension (2)
!>          القيم المحسوبة بواسطة الاختبارين الموصوفين أعلاه.
!>          القيم محدودة حاليًا بـ 1/ulp، لتجنب تجاوز السعة.
!>
