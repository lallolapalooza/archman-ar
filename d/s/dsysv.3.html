توثيق الوظيفة/الوحدة الفرعية

subroutine dsysv (character uplo, integer n, integer nrhs, double precision, dimension( lda, * ) a, integer lda, integer, dimension( * ) ipiv, double precision, dimension( ldb, * ) b, integer ldb, double precision, dimension( * ) work, integer lwork, integer info)

DSYSV تحسب حل نظام المعادلات الخطية A * X = B لمصفوفات SY

الغرض:

!>
!> DSYSV تحسب حل نظام المعادلات الخطية الحقيقية
!>    A * X = B,
!> حيث A مصفوفة متناظرة N*N و X و B مصفوفتان N*NRHS.
!>
!> تستخدم طريقة القسمة القطرية لتحليل A إلى
!>    A = U * D * U**T,  إذا كان UPLO = 'U', أو
!>    A = L * D * L**T,  إذا كان UPLO = 'L',
!> حيث U (أو L) هي حاصل ضرب تبديل ومصفوفات مثلثة عليا (سفلى) أحادية،
!> و D مصفوفة متناظرة وقطرية مدمجة تحتوي على كتل قطرية 1*1 و 2*2.
!> ثم يتم استخدام الشكل المحلل لـ A لحل نظام المعادلات A * X = B.
!>

المعاملات

UPLO 
!>          UPLO هو CHARACTER*1
!>          = 'U':  المثلث العلوي لـ A مخزن؛
!>          = 'L':  المثلث السفلي لـ A مخزن.
!>

N

!>          N هو INTEGER
!>          عدد المعادلات الخطية، أي، رتبة المصفوفة A.  N >= 0.
!>

NRHS

!>          NRHS هو INTEGER
!>          عدد الأطراف اليمنى، أي، عدد أعمدة المصفوفة B.  NRHS >= 0.
!>

A

!>          A هو مصفوفة DOUBLE PRECISION، حجمها (LDA,N)
!>          عند الإدخال، المصفوفة المتناظرة A.  إذا كان UPLO = 'U'، فإن الجزء
!>          العلوي المثلثي N*N الرائد في A يحتوي على الجزء العلوي
!>          المثلثي للمصفوفة A، ولا يتم الإشارة إلى الجزء المثلثي السفلي
!>          الصارم لـ A.  إذا كان UPLO = 'L'، فإن الجزء
!>          السفلي المثلثي N*N الرائد في A يحتوي على الجزء السفلي
!>          المثلثي للمصفوفة A، ولا يتم الإشارة إلى الجزء العلوي
!>          المثلثي الصارم لـ A.
!>
!>          عند الخروج، إذا كان INFO = 0، فإن المصفوفة القطرية المدمجة D
!>          والمضاعفات المستخدمة للحصول على العامل U أو L من
!>          التحليل A = U*D*U**T أو A = L*D*L**T كما تم حسابها بواسطة
!>          DSYTRF.
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرائد للمصفوفة A.  LDA >= max(1,N).
!>

IPIV

!>          IPIV هو مصفوفة INTEGER، حجمها (N)
!>          تفاصيل التبادلات وهيكل الكتلة لـ D، كما
!>          تم تحديدها بواسطة DSYTRF.  إذا كان IPIV(k) > 0، فقد تم تبديل
!>          الصفوف والأعمدة k و IPIV(k)، و D(k,k) هي كتلة قطرية 1*1.
!>          إذا كان UPLO = 'U' و IPIV(k) = IPIV(k-1) < 0، فقد تم تبديل
!>          الصفوف والأعمدة k-1 و -IPIV(k) و D(k-1:k,k-1:k) هي كتلة قطرية 2*2.
!>          إذا كان UPLO = 'L' و IPIV(k) = IPIV(k+1) < 0، فقد تم تبديل
!>          الصفوف والأعمدة k+1 و -IPIV(k) و D(k:k+1,k:k+1) هي كتلة قطرية 2*2.
!>

B

!>          B هو مصفوفة DOUBLE PRECISION، حجمها (LDB,NRHS)
!>          عند الإدخال، مصفوفة الطرف الأيمن N*NRHS B.
!>          عند الخروج، إذا كان INFO = 0، فإن مصفوفة الحل X ذات الحجم N*NRHS.
!>

LDB

!>          LDB هو INTEGER
!>          البعد الرائد للمصفوفة B.  LDB >= max(1,N).
!>

WORK

!>          WORK هو مصفوفة DOUBLE PRECISION، حجمها (MAX(1,LWORK))
!>          عند الخروج، إذا كان INFO = 0، فإن WORK(1) يعيد قيمة LWORK المثلى.
!>

LWORK

!>          LWORK هو INTEGER
!>          طول WORK.  LWORK >= 1، ولأفضل أداء
!>          LWORK >= max(1,N*NB)، حيث NB هو حجم الكتلة الأمثل لـ
!>          DSYTRF.
!>          لـ LWORK < N، سيتم إجراء TRS باستخدام المستوى BLAS 2
!>          لـ LWORK >= N، سيتم إجراء TRS باستخدام المستوى BLAS 3
!>
!>          إذا كان LWORK = -1، فسيتم افتراض استعلام مساحة العمل؛ تقوم الوحدة الفرعية
!>          بحساب حجم مساحة العمل الأمثل فقط، وتعيد
!>          هذه القيمة كإدخال أول لمساحة العمل، ولا يتم إصدار أي رسالة خطأ
!>          متعلقة بـ LWORK بواسطة XERBLA.
!>

INFO

!>          INFO هو INTEGER
!>          = 0: خروج ناجح
!>          < 0: إذا كان INFO = -i، فإن المعامل i كان له قيمة غير قانونية
!>          > 0: إذا كان INFO = i، فإن D(i,i) تساوي صفرًا بالضبط.  تم إكمال التحليل،
!>               ولكن المصفوفة القطرية المدمجة D مفردة تمامًا، لذلك لم يكن من الممكن
!>               حساب الحل.
!>
