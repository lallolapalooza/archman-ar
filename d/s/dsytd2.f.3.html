subroutine dsytd2 (uplo, n, a, lda, d, e, tau, info)

DSYTD2 تقلل مصفوفة متماثلة إلى شكل ثلاثي قطري حقيقي متماثل عبر تحويل تشابه متعامد (خوارزمية غير مجمعة).

توثيق الوظيفة/الإجراء الفرعي

subroutine dsytd2 (character uplo, integer n, double precision, dimension( lda, * ) a, integer lda, double precision, dimension( * ) d, double precision, dimension( * ) e, double precision, dimension( * ) tau, integer info)

DSYTD2 تقلل مصفوفة متماثلة حقيقية إلى شكل ثلاثي قطري T عبر تحويل تشابه متعامد: Q**T * A * Q = T.

الغرض:

!>
!> DSYTD2 تقلل مصفوفة متماثلة حقيقية A إلى شكل ثلاثي قطري T عبر تحويل تشابه متعامد: Q**T * A * Q = T.
!>

المعلمات

UPLO 
!>          UPLO is CHARACTER*1
!>          يحدد ما إذا كان الجزء المثلثي العلوي أو السفلي من المصفوفة المتماثلة A مخزنًا:
!>          = 'U':  مثلثي علوي
!>          = 'L':  مثلثي سفلي
!>

N

!>          N is INTEGER
!>          ترتيب المصفوفة A.  N >= 0.
!>

A

!>          A is DOUBLE PRECISION array, dimension (LDA,N)
!>          عند الإدخال، المصفوفة المتماثلة A.  إذا كان UPLO = 'U'، فإن الجزء المثلثي العلوي n x n الرائد من A يحتوي على الجزء المثلثي العلوي من المصفوفة A، والجزء السفلي المثلثي الصارم من A غير مشار إليه.  إذا كان UPLO = 'L'، فإن الجزء المثلثي السفلي n x n الرائد من A يحتوي على الجزء المثلثي السفلي من المصفوفة A، والجزء العلوي المثلثي الصارم من A غير مشار إليه.
!>          عند الخروج، إذا كان UPLO = 'U'، يتم استبدال القطر والعنصر فوق القطري الأول من A بالعناصر المقابلة للمصفوفة الثلاثية القطرية T، والعناصر فوق العنصر فوق القطري الأول، مع المصفوفة TAU، تمثل المصفوفة المتعامدة Q كناتج للمنحنيات الأولية؛ إذا كان UPLO = 'L'، يتم استبدال القطر والعنصر تحت القطري الأول من A بالعناصر المقابلة للمصفوفة الثلاثية القطرية T، والعناصر تحت العنصر تحت القطري الأول، مع المصفوفة TAU، تمثل المصفوفة المتعامدة Q كناتج للمنحنيات الأولية. انظر التفاصيل الإضافية.
!>

LDA

!>          LDA is INTEGER
!>          البعد الرائد للمصفوفة A.  LDA >= max(1,N).
!>

D

!>          D is DOUBLE PRECISION array, dimension (N)
!>          العناصر القطرية للمصفوفة الثلاثية القطرية T:
!>          D(i) = A(i,i).
!>

E

!>          E is DOUBLE PRECISION array, dimension (N-1)
!>          العناصر خارج القطر للمصفوفة الثلاثية القطرية T:
!>          E(i) = A(i,i+1) إذا كان UPLO = 'U'، E(i) = A(i+1,i) إذا كان UPLO = 'L'.
!>

TAU

!>          TAU is DOUBLE PRECISION array, dimension (N-1)
!>          عوامل القياس للمنحنيات الأولية (انظر التفاصيل الإضافية).
!>

INFO

!>          INFO is INTEGER
!>          = 0:  خروج ناجح
!>          < 0:  إذا كان INFO = -i، فإن المعلمة i كان لها قيمة غير قانونية.
!>

تفاصيل إضافية:

!>
!>  إذا كان UPLO = 'U'، يتم تمثيل المصفوفة Q كناتج للمنحنيات الأولية
!>  Q = H(n-1) . . . H(2) H(1).
!>
!>  كل H(i) لها الشكل
!>
!>     H(i) = I - tau * v * v**T
!>
!>  حيث tau هو رقم قياسي حقيقي، و v هو متجه حقيقي مع
!>  v(i+1:n) = 0 و v(i) = 1؛ v(1:i-1) مخزن عند الخروج في
!>  A(1:i-1,i+1)، و tau في TAU(i).
!>
!>  إذا كان UPLO = 'L'، يتم تمثيل المصفوفة Q كناتج للمنحنيات الأولية
!>  Q = H(1) H(2) . . . H(n-1).
!>
!>  كل H(i) لها الشكل
!>
!>     H(i) = I - tau * v * v**T
!>
!>  حيث tau هو رقم قياسي حقيقي، و v هو متجه حقيقي مع
!>  v(1:i) = 0 و v(i+1) = 1؛ v(i+2:n) مخزن عند الخروج في A(i+2:n,i)،
!>  و tau في TAU(i).
!>
!>  محتويات A عند الخروج موضحة بالأمثلة التالية
!>  مع n = 5:
!>
!>  إذا كان UPLO = 'U':                       إذا كان UPLO = 'L':
!>
!>    (  d   e   v2  v3  v4 )              (  d                  )
!>    (      d   e   v3  v4 )              (  e   d              )
!>    (          d   e   v4 )              (  v1  e   d          )
!>    (              d   e  )              (  v1  v2  e   d      )
!>    (                  d  )              (  v1  v2  v3  e   d  )
!>
!>  حيث d و e يمثلان العناصر القطرية وغير القطرية لـ T، و vi
!>  يمثل عنصرًا من المتجه الذي يحدد H(i).
!>

تعريف في السطر 172 من الملف dsytd2.f.
