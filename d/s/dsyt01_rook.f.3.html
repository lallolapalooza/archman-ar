توثيق الوظيفة/الإجراء الفرعي

subroutine dsyt01_rook (character uplo, integer n, double precision, dimension( lda, * ) a, integer lda, double precision, dimension( ldafac, * ) afac, integer ldafac, integer, dimension( * ) ipiv, double precision, dimension( ldc, * ) c, integer ldc, double precision, dimension( * ) rwork, double precision resid)

DSYT01_ROOK

الغرض:

!>
!> DSYT01_ROOK تعيد بناء مصفوفة متماثلة غير محددة A من تفكيكها L*D*L' أو U*D*U'
!> وتحسب معيار الباقي
!>    norm( C - A ) / ( N * norm(A) * EPS ),
!> حيث C هي المصفوفة المعاد بناؤها و EPS هو الخطأ المطبعي الآلي.
!>

المعلمات

UPLO 
!>          UPLO هو CHARACTER*1
!>          يحدد ما إذا كان الجزء العلوي أو السفلي المثلثي للمصفوفة المتماثلة A مخزنًا:
!>          = 'U':  مثلثي علوي
!>          = 'L':  مثلثي سفلي
!>

N

!>          N هو INTEGER
!>          عدد الصفوف والأعمدة للمصفوفة A.  N >= 0.
!>

A

!>          A هي مصفوفة DOUBLE PRECISION، حجمها (LDA,N)
!>          المصفوفة المتماثلة الأصلية A.
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرائد للمصفوفة A.  LDA >= max(1,N)
!>

AFAC

!>          AFAC هي مصفوفة DOUBLE PRECISION، حجمها (LDAFAC,N)
!>          الشكل المفكك للمصفوفة A.  AFAC تحتوي على المصفوفة القطرية الكتلية D
!>          والمضاعفات المستخدمة للحصول على العامل L أو U من تفكيك L*D*L' أو U*D*U'
!>          كما تم حسابه بواسطة DSYTRF_ROOK.
!>

LDAFAC

!>          LDAFAC هو INTEGER
!>          البعد الرائد للمصفوفة AFAC.  LDAFAC >= max(1,N).
!>

IPIV

!>          IPIV هي مصفوفة INTEGER، حجمها (N)
!>          فهارس المحور من DSYTRF_ROOK.
!>

C

!>          C هي مصفوفة DOUBLE PRECISION، حجمها (LDC,N)
!>

LDC

!>          LDC هو INTEGER
!>          البعد الرائد للمصفوفة C.  LDC >= max(1,N).
!>

RWORK

!>          RWORK هي مصفوفة DOUBLE PRECISION، حجمها (N)
!>

RESID

!>          RESID هو DOUBLE PRECISION
!>          إذا كان UPLO = 'L'، فإن norm(L*D*L' - A) / ( N * norm(A) * EPS )
!>          إذا كان UPLO = 'U'، فإن norm(U*D*U' - A) / ( N * norm(A) * EPS )
!>
