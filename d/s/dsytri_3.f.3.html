subroutine dsytri_3 (character uplo, integer n, double precision, dimension( lda, * ) a, integer lda, double precision, dimension( * ) e, integer, dimension( * ) ipiv, double precision, dimension( * ) work, integer lwork, integer info)

DSYTRI_3

الغرض:

!> DSYTRI_3 تحسب معكوس مصفوفة متماثلة غير محددة حقيقية
!> A باستخدام التحليل الذي حسبته DSYTRF_RK أو DSYTRF_BK:
!>
!>     A = P*U*D*(U**T)*(P**T) أو A = P*L*D*(L**T)*(P**T),
!>
!> حيث U (أو L) هي مصفوفة وحدوية علوية (أو سفلية)،
!> U**T (أو L**T) هي منقولة U (أو L)، P هي مصفوفة تبديل،
!> P**T هي منقولة P، و D هي مصفوفة متماثلة وقطرية كتلية
!> ذات كتل قطرية 1x1 و 2x2.
!>
!> DSYTRI_3 تحدد البعد الرائد لمساحة العمل
!> قبل استدعاء DSYTRI_3X التي تحسب المعكوس فعليًا. هذا هو
!> الإصدار المكتلي للخوارزمية، الذي يستدعي BLAS من المستوى 3.
!>

المعاملات

UPLO 
!>          UPLO هو CHARACTER*1
!>          يحدد ما إذا كانت تفاصيل التحليل مخزنة كمصفوفة
!>          مثلثة علوية أم سفلية.
!>          = 'U':  يتم تخزين المثلث العلوي لـ A؛
!>          = 'L':  يتم تخزين المثلث السفلي لـ A.
!>

N

!>          N هو INTEGER
!>          ترتيب المصفوفة A. N >= 0.
!>

A

!>          A هي مصفوفة DOUBLE PRECISION، البعد (LDA,N)
!>          عند الإدخال، عناصر القطر للمصفوفة القطرية الكتلية
!>          المتماثلة D وعوامل U أو L كما حسبتها DSYTRF_RK و DSYTRF_BK:
!>            أ) عناصر القطر فقط للمصفوفة القطرية الكتلية
!>               المتماثلة D على قطر A، أي D(k,k) = A(k,k)؛
!>               (يجب توفير عناصر فوق القطر (أو تحت القطر) لـ D
!>                في المصفوفة E عند الإدخال)، و
!>            ب) إذا كان UPLO = 'U': العامل U في الجزء فوق القطري لـ A.
!>               إذا كان UPLO = 'L': العامل L في الجزء تحت القطري لـ A.
!>
!>          عند الخروج، إذا كان INFO = 0، فإن معكوس المصفوفة الأصلية
!>          المتماثلة.
!>             إذا كان UPLO = 'U': يتم تشكيل الجزء العلوي من المعكوس
!>             ولا تتم الإشارة إلى الجزء السفلي من A؛
!>             إذا كان UPLO = 'L': يتم تشكيل الجزء السفلي من المعكوس
!>             ولا تتم الإشارة إلى الجزء العلوي من A.
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرائد للمصفوفة A. LDA >= max(1,N).
!>

E

!>          E هي مصفوفة DOUBLE PRECISION، البعد (N)
!>          عند الإدخال، تحتوي على عناصر فوق القطر (أو تحت القطر)
!>          للمصفوفة القطرية الكتلية المتماثلة D
!>          ذات الكتل القطرية 1x1 أو 2x2، حيث
!>          إذا كان UPLO = 'U': E(i) = D(i-1,i),i=2:N, E(1) لا تتم الإشارة إليه؛
!>          إذا كان UPLO = 'L': E(i) = D(i+1,i),i=1:N-1, E(N) لا تتم الإشارة إليه.
!>
!>          ملاحظة: لكتلة قطرية 1x1 D(k)، حيث
!>          1 <= k <= N، لا تتم الإشارة إلى العنصر E(k) في كلتا
!>          حالات UPLO = 'U' أو UPLO = 'L'.
!>

IPIV

!>          IPIV هي مصفوفة INTEGER، البعد (N)
!>          تفاصيل التبادلات وهيكل الكتل لـ D
!>          كما تم تحديده بواسطة DSYTRF_RK أو DSYTRF_BK.
!>

WORK

!>          WORK هي مصفوفة DOUBLE PRECISION، البعد (MAX(1,LWORK)).
!>          عند الخروج، إذا كان INFO = 0، فإن WORK(1) تعيد حجم LWORK الأمثل.
!>

LWORK

!>          LWORK هو INTEGER
!>          طول WORK.
!>          إذا كان N = 0، LWORK >= 1، وإلا LWORK >= (N+NB+1)*(NB+3).
!>
!>          إذا كان LWORK = -1، فسيتم افتراض استعلام مساحة العمل؛
!>          يقوم الروتين فقط بحساب الحجم الأمثل لمساحة العمل
!>          الأمثل، ويعيد هذا الحجم كالإدخال الأول لمصفوفة
!>          WORK، ولا يتم إصدار أي رسالة خطأ متعلقة بـ
!>          LWORK بواسطة XERBLA.
!>

INFO

!>          INFO هو INTEGER
!>          = 0: خروج ناجح
!>          < 0: إذا كان INFO = -i، فإن المعامل i كان له قيمة غير قانونية
!>          > 0: إذا كان INFO = i، D(i,i) = 0؛ المصفوفة مفردة ولا يمكن
!>               حساب معكوسها.
!>
