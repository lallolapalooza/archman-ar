وثائق الوظيفة/الروتين الفرعي

subroutine dstev (character jobz, integer n, double precision, dimension( * ) d, double precision, dimension( * ) e, double precision, dimension( ldz, * ) z, integer ldz, double precision, dimension( * ) work, integer info)

DSTEV تحسب جميع القيم الذاتية، واختياريًا، المتجهات الذاتية لمصفوفة ثلاثية القطر المتناظرة الحقيقية A.

الغرض:

!>
!> DSTEV تحسب جميع القيم الذاتية، واختياريًا، المتجهات الذاتية لمصفوفة
!> ثلاثية القطر المتناظرة الحقيقية A.
!>

المعلمات

JOBZ 
!>          JOBZ is CHARACTER*1
!>          = 'N':  حساب القيم الذاتية فقط؛
!>          = 'V':  حساب القيم الذاتية والمتجهات الذاتية.
!>

N

!>          N is INTEGER
!>          رتبة المصفوفة.  N >= 0.
!>

D

!>          D is DOUBLE PRECISION array, dimension (N)
!>          عند الإدخال، العناصر القطرية n للمصفوفة ثلاثية القطر
!>          A.
!>          عند الخروج، إذا كان INFO = 0، القيم الذاتية بترتيب تصاعدي.
!>

E

!>          E is DOUBLE PRECISION array, dimension (N-1)
!>          عند الإدخال، العناصر تحت القطرية (n-1) للمصفوفة ثلاثية القطر
!>          A، مخزنة في العناصر 1 إلى N-1 من E.
!>          عند الخروج، يتم تدمير محتويات E.
!>

Z

!>          Z is DOUBLE PRECISION array, dimension (LDZ, N)
!>          إذا كان JOBZ = 'V'، فعندئذٍ، إذا كان INFO = 0، فإن Z يحتوي على
!>          المتجهات الذاتية المتعامدة للمصفوفة A، مع احتوء العمود i من Z
!>          على المتجه الذاتي المرتبط بـ D(i).
!>          إذا كان JOBZ = 'N'، فلا يتم استخدام Z.
!>

LDZ

!>          LDZ is INTEGER
!>          البعد الرئيسي للمصفوفة Z.  LDZ >= 1، وإذا
!>          كان JOBZ = 'V'، فإن LDZ >= max(1,N).
!>

WORK

!>          WORK is DOUBLE PRECISION array, dimension (max(1,2*N-2))
!>          إذا كان JOBZ = 'N'، فلا يتم استخدام WORK.
!>

INFO

!>          INFO is INTEGER
!>          = 0:  خروج ناجح
!>          < 0:  إذا كان INFO = -i، كان المعامل i له قيمة غير قانونية
!>          > 0:  إذا كان INFO = i، فشل الخوارزمية في التقارب؛ i
!>                لم تتقارب العناصر غير القطرية من E إلى الصفر.
!>
