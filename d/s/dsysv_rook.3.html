subroutine dsysv_rook (character uplo, integer n, integer nrhs, double precision, dimension( lda, * ) a, integer lda, integer, dimension( * ) ipiv, double precision, dimension( ldb, * ) b, integer ldb, double precision, dimension( * ) work, integer lwork, integer info)

DSYSV_ROOK تحسب حل نظام المعادلات الخطية A * X = B لمصفوفات SY

الغرض:

!>
!> DSYSV_ROOK تحسب حل نظام المعادلات الخطية الحقيقية
!>    A * X = B,
!> حيث A هي مصفوفة متناظرة N×N و X و B هما مصفوفتان N×NRHS.
!>
!> يتم استخدام طريقة التبادل القطري لتحليل A إلى
!>    A = U * D * U**T,  إذا كان UPLO = 'U', أو
!>    A = L * D * L**T,  إذا كان UPLO = 'L',
!> حيث U (أو L) هو حاصل ضرب مصفوفات التباديل والمصفوفات المثلثية العلوية (السفلية) الوحدوية، و D متناظرة وقطرية مصفوفات متفرعة تحتوي على كتل قطرية 1×1 و 2×2.
!>
!> يتم استدعاء DSYTRF_ROOK لحساب تحليل مصفوفة متناظرة حقيقية A باستخدام طريقة التبادل القطري المحدود Bunch-Kaufman ().
!>
!> يتم استخدام الشكل المحلل من A لحل نظام المعادلات A * X = B عن طريق استدعاء DSYTRS_ROOK.
!>

المعاملات

UPLO 
!>          UPLO هو CHARACTER*1
!>          = 'U':  يتم تخزين الجزء العلوي من A؛
!>          = 'L':  يتم تخزين الجزء السفلي من A؛
!>

N

!>          N هو INTEGER
!>          عدد المعادلات الخطية، أي رتبة المصفوفة A.  N >= 0.
!>

NRHS

!>          NRHS هو INTEGER
!>          عدد الأطراف اليمنى، أي عدد أعمدة المصفوفة B.  NRHS >= 0.
!>

A

!>          A هي مصفوفة DOUBLE PRECISION، أبعادها (LDA,N)
!>          عند الإدخال، المصفوفة المتناظرة A.  إذا كان UPLO = 'U'، فإن الجزء العلوي من A الذي أبعاده N×N يحتوي على الجزء العلوي للمصفوفة A، ولا يتم الإشارة إلى الجزء المثلثي السفلي الصارم لـ A.  إذا كان UPLO = 'L'، فإن الجزء السفلي من A الذي أبعاده N×N يحتوي على الجزء السفلي للمصفوفة A، ولا يتم الإشارة إلى الجزء المثلثي العلوي الصارم لـ A.
!>
!>          عند الخروج، إذا كان INFO = 0، فإن المصفوفة المتفرعة القطرية D والمضاعفات المستخدمة للحصول على العامل U أو L من التحليل A = U*D*U**T أو A = L*D*L**T كما تم حسابها بواسطة DSYTRF_ROOK.
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرئيسي للمصفوفة A.  LDA >= max(1,N).
!>

IPIV

!>          IPIV هي مصفوفة INTEGER، أبعادها (N)
!>          تفاصيل التبادلات وهيكل الكتلة D، كما تم تحديدها بواسطة DSYTRF_ROOK.
!>
!>          إذا كان UPLO = 'U':
!>               إذا كان IPIV(k) > 0، فقد تم تبديل الصفوف والأعمدة k و IPIV(k) و D(k,k) هي كتلة قطرية 1×1.
!>
!>               إذا كان IPIV(k) < 0 و IPIV(k-1) < 0، فقد تم تبديل الصفوف والأعمدة k و -IPIV(k) وتم تبديل الصفوف والأعمدة k-1 و -IPIV(k-1)،
!>               D(k-1:k,k-1:k) هي كتلة قطرية 2×2.
!>
!>          إذا كان UPLO = 'L':
!>               إذا كان IPIV(k) > 0، فقد تم تبديل الصفوف والأعمدة k و IPIV(k) و D(k,k) هي كتلة قطرية 1×1.
!>
!>               إذا كان IPIV(k) < 0 و IPIV(k+1) < 0، فقد تم تبديل الصفوف والأعمدة k و -IPIV(k) وتم تبديل الصفوف والأعمدة k+1 و -IPIV(k+1)،
!>               D(k:k+1,k:k+1) هي كتلة قطرية 2×2.
!>

B

!>          B هي مصفوفة DOUBLE PRECISION، أبعادها (LDB,NRHS)
!>          عند الإدخال، المصفوفة B ذات الأبعاد N×NRHS.
!>          عند الخروج، إذا كان INFO = 0، فإن المصفوفة X ذات الأبعاد N×NRHS.
!>

LDB

!>          LDB هو INTEGER
!>          البعد الرئيسي للمصفوفة B.  LDB >= max(1,N).
!>

WORK

!>          WORK هي مصفوفة DOUBLE PRECISION، أبعادها (MAX(1,LWORK))
!>          عند الخروج، إذا كان INFO = 0، فإن WORK(1) تعيد LWORK المثلى.
!>

LWORK

!>          LWORK هو INTEGER
!>          طول WORK.  LWORK >= 1، ولأفضل أداء LWORK >= max(1,N*NB)، حيث NB هو حجم الكتلة الأمثل لـ DSYTRF_ROOK.
!>
!>          سيتم إجراء TRS باستخدام BLAS المستوى 2
!>
!>          إذا كان LWORK = -1، فسيتم افتراض استعلام مساحة العمل؛ تقوم الروتين فقط بحساب الحجم الأمثل للمصفوفة WORK، وتعيد هذا الحجم كإدخال أول للمصفوفة WORK، ولا يصدر XERBLA أي رسالة خطأ تتعلق بـ LWORK.
!>

INFO

!>          INFO هو INTEGER
!>          = 0: خروج ناجح
!>          < 0: إذا كان INFO = -i، فإن المعامل i كان له قيمة غير قانونية
!>          > 0: إذا كان INFO = i، فإن D(i,i) يساوي صفر تمامًا.  تم إكمال التحليل، ولكن المصفوفة المتفرعة القطرية D هي مصفوفة مفردة تمامًا، وبالتالي لم يكن من الممكن حساب الحل.
!>
