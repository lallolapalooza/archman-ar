DJabberd__Component(3pm)

الحزمة: perl-djabberd

التراخيص: GPL, PerlArtistic

DJabberd::Component(3) توثيق Perl مساهم به من قبل المستخدم DJabberd::Component(3)

الاسم

DJabberd::Component - فئة مجردة تمثل مكونًا في DJabberd

الملخص

package MyPackage::DJabberd::MyComponent;
use base DJabberd::Component;

# مثال لمكون يقبل أوامر XEP-0050 Ad-Hoc

sub finalize {
    my ($self, $opts) = @_;

    # دع الفئة الأصلية تنهي
    $self->SUPER::finalize;

    $self->register_iq_handler(
        "set-{http://jabber.org/protocol/commands}command",
        sub { $self->handle_adhoc_command(@_) },
    );
}

sub handle_adhoc_command {
    my ($self, $vhost, $stanza) = @_;

    # تعامل مع حالة DJabberd::IQ في $stanza،
    # أو أرسل نوعًا من الاستجابة للخطأ.
}

توفر هذه الفئة فئة أساسية لجميع مكونات DJabberd. يمكن للمكونات التي ترث من هذه الفئة استخدامها مباشرة بواسطة الخادم كملحقات تسليم، أو استخدامها مباشرة بواسطة فئات أخرى.

انظر DJabberd::Component::Example للحصول على مثال لتنفيذ مكون.

الاستخدام

هذه الفئة هي تخصص لـ DJabberd::Agent موجهة لأولئك الذين يرغبون في إنشاء "مكونات"، وهي عوامل برمجية تتعامل مع مجال XMPP كامل، بما في ذلك جميع العقد داخل هذا المجال.

في معظم الحالات، سيستخدم مستخدمو هذه الفئة واجهة برمجة التطبيقات التي تعرضها DJabberd::Agent. ومع ذلك، هناك بعض السلوكيات الخاصة التي يجب ملاحظتها وبعض الطرق الإضافية التي ليست جزءًا من فئة الوكيل الأساسية.

handles_destination($self, $to_jid, $vhost)

توفر هذه الفئة إصدارًا معدلاً من هذه الطريقة يستجيب بقيمة صحيحة إذا وفقط إذا كان مجال JID المعطى مطابقًا لمجال المكون نفسه. من غير المرجح أن تحتاج إلى تجاوز هذه الطريقة.

handle_stanza($self, $vhost, $stanza)

الإصدار المعدل لهذه الطريقة له سلوك خاص عند التعامل مع الحالات الموجهة إلى عقد داخل المجال الذي يتعامل معه المكون الخاص بك. (أي، JIDs بالشكل "user@example.com"، بدلاً من "example.com" فقط.)

عند استلام مثل هذه الحالة، سيتم استدعاء الطريقة "get_node" لاسترداد كائن يمثل هذه العقدة.

إذا كنت ترغب في التعامل على مستوى منخفض مع جميع الحالات الواردة، يمكنك تجاوز هذه الطريقة. كما هو الحال مع DJabberd::Agent، فإن تجاوز هذه الطريقة سيعطل جميع طرق المعالجة عالية المستوى ما لم تفوض إلى الطريقة المتجاوزة.

إذا كنت ترغب في الاحتفاظ بطرق المعالجة عالية المستوى ولكنك لا ترغب في التعامل مع العقد ككائنات منفصلة، فقد ترغب في تجاوز هذه الطريقة لاستدعاء الإصدار الأصلي من DJabberd::Agent على النحو التالي:

sub handle_stanza {
    $self->DJabberd::Agent::handle_stanza(@_);
}

مع وجود هذه الطريقة، لن يتم استخدام الطريقة "get_node" بعد الآن وسيتم تمرير جميع الحالات المناسبة إلى طرق المعالجة الخاصة بهذه النسخة بغض النظر عن اسم عقدة JID المستهدفة.

get_node($self, $nodename)

يتم استدعاؤها عندما ترغب هذه الفئة في العثور على وكيل للتعامل مع عقدة معينة داخل مجال هذا المكون.

ستعيد معظم التنفيذات العادية لهذه الطريقة نسخة من فئة فرعية لـ DJabberd::Agent::Node، ولكن في الواقع أي فئة فرعية لـ DJabberd::Agent ستعمل طالما أنها ستقبل الحالات الموجهة إلى اسم العقدة المعطى.

vcard($self)

الإصدار المعدل لهذه الطريقة سيعيد بطاقة vCard حيث يتم تعيين الاسم الكامل لاسم مجال المكون. قد تقوم الفئات الفرعية بتجاوز هذه الطريقة للقيام بشيء أكثر إثارة.

identities($self)

الإصدار المعدل لهذه الطريقة يعيد هوية واحدة وعامة تشير إلى أن المكون هو "فرع" في شجرة اكتشاف الخدمة. تم تضمين هذا لأن بعض العملاء يبدون انزعاجًا إذا كانت JID تدعم الاكتشاف ولكن ليس لديها هويات.

انظر أيضًا

انظر أيضًا DJabberd::Agent، DJabberd::Agent::Node.
