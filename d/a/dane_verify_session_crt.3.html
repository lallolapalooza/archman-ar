dane_verify_session_crt(3)

dane_verify_session_crt(3) gnutls dane_verify_session_crt(3)

الاسم

dane_verify_session_crt - وظيفة API

ملخص

#include <gnutls/dane.h>

int dane_verify_session_crt(dane_state_t s, gnutls_session_t session, const char * hostname, const char * proto, unsigned int port, unsigned int sflags, unsigned int vflags, unsigned int * verify);

الوسائط

dane_state_t s
هيكل حالة DANE (قد يكون NULL)
gnutls_session_t session
جلسة gnutls
const char * hostname
اسم المضيف المرتبط بالسلسلة
const char * proto
بروتوكول الخدمة المتصلة (مثل tcp)
unsigned int port
منفذ الخدمة المتصلة (مثل 443)
unsigned int sflags
علامات لتهيئة s (إذا كان NULL)
unsigned int vflags
علامات التحقق؛ قائمة OR'ed لـ dane_verify_flags_t.
unsigned int * verify
قائمة OR'ed لـ dane_verify_status_t.

الوصف

ستقوم هذه الوظيفة بالتحقق من سلسلة شهادات الجلسة مقابل قيود CA و/أو الشهادة المتاحة عبر DANE. انظر dane_verify_crt() لمزيد من المعلومات.

لن يتحقق هذا من صحة السلسلة؛ ما لم يكن التحقق من DANE مقصورًا على الشهادات النهائية، فيجب القيام بذلك بشكل منفصل باستخدام gnutls_certificate_verify_peers3().

يعيد

رمز خطأ سالب عند الخطأ و DANE_E_SUCCESS (0) عندما تم تحليل إدخالات DANE بنجاح، بغض النظر عما إذا كانت قد تم التحقق منها (راجع verify للحصول على هذه المعلومات). إذا لم يتم العثور على إدخالات قابلة للاستخدام، فسيتم إرجاع DANE_E_REQUESTED_DATA_NOT_AVAILABLE.
