Date__Format(3pm)

الحزمة: perl-timedate

التراخيص: PerlArtistic

Date::Format(3) وثائق Perl المساهمة من المستخدم Date::Format(3)

الاسم

Date::Format - روتينات تنسيق التاريخ

الملخص

use Date::Format;

@lt = localtime(time);

print time2str($template, time);
print strftime($template, @lt);

print time2str($template, time, $zone);
print strftime($template, @lt, $zone);

print ctime(time);
print asctime(@lt);

print ctime(time, $zone);
print asctime(@lt, $zone);

الوصف

توفر هذه الوحدة روتينات لتنسيق التواريخ إلى سلاسل نصية ASCII. وهي تتوافق مع روتينات مكتبة C "strftime" و "ctime".

time2str(TEMPLATE, TIME [, ZONE])
"time2str" تحول "TIME" إلى سلسلة نصية ASCII باستخدام مواصفات التحويل المعطاة في "TEMPLATE". "ZONE" إذا تم تقديمه يحدد المنطقة المطلوبة للخرج، "ZONE" افتراضياً هو منطقتك الحالية.
strftime(TEMPLATE, TIME [, ZONE])
"strftime" مشابه لـ "time2str" باستثناء أن الوقت يتم تمريره كمصفوفة، مثل المصفوفة التي تعيدها "localtime".
ctime(TIME [, ZONE])
"ctime" يستدعي "time2str" مع الوسائط المعطاة باستخدام مواصفات التحويل "%a %b %e %T %Y\n"
asctime(TIME [, ZONE])
"asctime" يستدعي "time2str" مع الوسائط المعطاة باستخدام مواصفات التحويل "%a %b %e %T %Y\n"

دعم متعدد اللغات

تستطيع Date::Format التنسيق بعدة لغات عن طريق إنشاء كائن خاص باللغة واستدعاء الأساليب، انظر Date::Language

my $lang = Date::Language->new('German');
$lang->time2str("%a %b %e %T %Y\n", time);

مواصفات التحويل

يتم استبدال كل مواصفات تحويل بأحرف مناسبة كما هو موضح في القائمة التالية. يتم تحديد الأحرف المناسبة بواسطة فئة LC_TIME في لوكال البرنامج.

%%      نسبة مئوية
%a      اختصار يوم الأسبوع
%A      يوم الأسبوع
%b      اختصار الشهر
%B      الشهر
%c      تنسيق ctime: Sat Nov 19 21:05:57 1994
%C      تنسيق ctime: Sat Nov 19 21:05:57 1994
%d      الرقم العشري ليوم الشهر، مع الأصفار البادئة (مثل 01..31)
%e      مثل %d، ولكن يتم استبدال الصفر البادئ بمسافة (مثل  1..32)
%D      MM/DD/YY
%G      رقم أسبوع GPS (أسابيع منذ 6 يناير 1980)
%h      اختصار الشهر
%H      الساعة، 24 ساعة، أصفار بادئة)
%I      الساعة، 12 ساعة، أصفار بادئة)
%j      اليوم في السنة
%k      الساعة
%l      الساعة، 12 ساعة
%L      رقم الشهر، بدءًا من 1
%m      رقم الشهر، بدءًا من 01
%M      الدقائق، أصفار بادئة
%n      سطر جديد
%o      اليوم الزخرفي للشهر - "1st", "2nd", "25th", إلخ.
%p      صباحًا أو مساءً
%P      صباحًا أو مساءً (نعم، %p و %P معكوسان :)
%q      رقم الربع، بدءًا من 1
%r      تنسيق الوقت: 09:05:57 PM
%R      تنسيق الوقت: 21:05
%s      الثواني منذ الحقبة، بالتوقيت العالمي المنسق
%S      الثواني، أصفار بادئة
%t      علامة جدولة
%T      تنسيق الوقت: 21:05:57
%U      رقم الأسبوع، الأحد هو أول يوم في الأسبوع
%w      اليوم من الأسبوع، رقميًا، الأحد = 0
%W      رقم الأسبوع، الاثنين هو أول يوم في الأسبوع
%x      تنسيق التاريخ: 11/19/94
%X      تنسيق الوقت: 21:05:57
%y      السنة (رقمين)
%Y      السنة (4 أرقام)
%Z      المنطقة الزمنية بالنص. مثال: PST
%z      المنطقة الزمنية بالتنسيق -/+0000

%d، %e، %H، %I، %j، %k، %l، %m، %M، %q، %y و %Y يمكن إخراجها بالأرقام الرومانية عن طريق إضافة البادئة "O" للحرف، على سبيل المثال %OY ستخرج السنة بالأرقام الرومانية.

قيد

الوظائف في هذه الوحدة مقيدة بنطاق الوقت الذي يمكن تمثيله بنوع بيانات time_t، أي 1901-12-13 20:45:53 GMT إلى 2038-01-19 03:14:07 GMT.
