dane_query_data(3) gnutls dane_query_data(3)

الاسم

dane_query_data - دالة API

ملخص

#include <gnutls/dane.h>

int dane_query_data(dane_query_t q, unsigned int idx, unsigned int * usage, unsigned int * type, unsigned int * match, gnutls_datum_t * data);

المعاملات

dane_query_t q
بنية نتيجة الاستعلام
unsigned int idx
فهرس استجابة الاستعلام.
unsigned int * usage
استخدام الشهادة (راجع dane_cert_usage_t)
unsigned int * type
نوع الشهادة (راجع dane_cert_type_t)
unsigned int * match
نوع مطابقة DANE (راجع dane_match_type_t)
gnutls_datum_t * data
بيانات DANE.

الوصف

ستقوم هذه الدالة بتوفير بيانات DANE من استجابة الاستعلام.

المُرجعات

عند النجاح، يتم إرجاع DANE_E_SUCCESS (0)، وإلا يتم إرجاع قيمة خطأ سالبة.
