Data__UUID(3pm)

الحزمة: perl-data-uuid

التراخيص: LicenseRef-custom

UUID(3) وثائق Perl مساهمة من المستخدم UUID(3)

الاسم

Data::UUID - معرفات فريدة عالميًا (GUIDs/UUIDs)

انظر بدلاً من ذلك؟

توفر الوحدة Data::GUID واجهة أخرى لتوليد GUIDs. في الوقت الحالي، تعتمد على Data::UUID، ولكن قد لا تفعل ذلك في المستقبل. قد تكون واجهتها أكثر مباشرة قليلاً لبرنامج Perl العادي.

الملخص

use Data::UUID;

$ug    = Data::UUID->new;
$uuid1 = $ug->create();
$uuid2 = $ug->create_from_name(<namespace>, <name>);
$res   = $ug->compare($uuid1, $uuid2);
$str   = $ug->to_string( $uuid );
$uuid  = $ug->from_string( $str );

الوصف

توفر هذه الوحدة إطار عمل لتوليد UUIDs الإصدار 3 (معرفات فريدة عالميًا، تُعرف أيضًا باسم GUIDs (معرفات فريدة عالميًا). UUID هو 128 بت، ويُضمن أن يكون مختلفًا عن جميع UUIDs/GUIDs الأخرى التي تم إنشاؤها حتى عام 3400 م.

تم استخدام UUIDs في الأصل في نظام حوسبة الشبكة (NCS) ولاحقًا في بيئة الحوسبة الموزعة لمؤسسة البرمجيات المفتوحة (OSF). حاليًا، تعتمد العديد من التقنيات المختلفة على UUIDs لتوفير هوية فريدة لمكونات البرامج المختلفة. على سبيل المثال، تستخدم Microsoft COM/DCOM GUIDs بشكل مكثف لتحديد الفئات والتطبيقات والمكونات بشكل فريد عبر الأنظمة المتصلة بالشبكة.

تم وصف الخوارزمية لتوليد UUID، المستخدمة بواسطة هذا الامتداد، في المسودة الدولية "UUIDs and GUIDs" بواسطة Paul J. Leach و Rich Salz. (انظر RFC 4122). وهي توفر إطار عمل فعال وموثوق بشكل معقول لتوليد UUIDs وتدعم معدلات تخصيص عالية نسبيًا - 10 ملايين في الثانية لكل آلة - وبالتالي فهي مناسبة لتحديد كل من الكائنات قصيرة العمر جدًا والدائمة جدًا على نظام معين وكذلك عبر الشبكة.

توفر هذه الوحدة عدة طرق لإنشاء UUID. في جميع الطرق، "<namespace>" هو UUID و "<name>" هو سلسلة نصية مجانية.

# ينشئ UUID ثنائي (قيمة ثنائية بطول 16 بايت).
$ug->create();
$ug->create_bin();
# ينشئ UUID ثنائي (قيمة ثنائية بطول 16 بايت) بناءً على مساحة اسم
# معينة وسلسلة اسم.
$ug->create_from_name(<namespace>, <name>);
$ug->create_from_name_bin(<namespace>, <name>);
# ينشئ سلسلة UUID، باستخدام تنسيق سلسلة UUID التقليدي،
# مثل: 4162F712-1DD2-11B2-B17E-C09EFE1DC403
# لاحظ أن الأرقام A-F مكتوبة بأحرف كبيرة، وهو ما يتعارض مع rfc4122
$ug->create_str();
$ug->create_from_name_str(<namespace>, <name>);
# ينشئ سلسلة UUID كسلسلة ست عشرية،
# مثل: 0x4162F7121DD211B2B17EC09EFE1DC403
# لاحظ أن الأرقام A-F مكتوبة بأحرف كبيرة، وهو ما يتعارض مع rfc4122
$ug->create_hex();
$ug->create_from_name_hex(<namespace>, <name>);
# ينشئ سلسلة UUID كسلسلة مشفرة بـ Base64
$ug->create_b64();
$ug->create_from_name_b64(<namespace>, <name>);
يمكن تحويل UUIDs الثنائية إلى سلاسل قابلة للطباعة باستخدام الطرق التالية:
# التحويل إلى تمثيل سلسلة تقليدي
$ug->to_string(<uuid>);
# التحويل إلى سلسلة ست عشرية (باستخدام أحرف كبيرة، بدلاً من صغيرة)
$ug->to_hexstring(<uuid>);
# التحويل إلى سلسلة مشفرة بـ Base64
$ug->to_b64string(<uuid>);
على العكس، يمكن تحويل سلاسل UUIDs مرة أخرى إلى شكل ثنائي:
# إعادة إنشاء UUID ثنائي من سلسلة
$ug->from_string(<uuid>);
$ug->from_hexstring(<uuid>);
# إعادة إنشاء UUID ثنائي من سلسلة مشفرة بـ Base64
$ug->from_b64string(<uuid>);
أخيرًا، يمكن مقارنة اثنين من UUIDs الثنائية باستخدام الطريقة التالية:
# يعيد -1، 0، أو 1 اعتمادًا على ما إذا كان uuid1 أقل
# من، يساوي، أو أكبر من uuid2
$ug->compare(<uuid1>, <uuid2>);

أمثلة:

use Data::UUID;
# هذا ينشئ UUID جديدًا في شكل سلسلة، بناءً على مساحة الاسم القياسية
# UUID NameSpace_URL والاسم "www.mycompany.com"
$ug = Data::UUID->new;
print $ug->create_from_name_str(NameSpace_URL, "www.mycompany.com");

تصدير

تسمح الوحدة بتصدير العديد من UUIDs مساحات الأسماء القياسية:

NameSpace_DNS
NameSpace_URL
NameSpace_OID
NameSpace_X500
