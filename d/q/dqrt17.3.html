توثيق الدالة/الروتين الفرعي

double precision function dqrt17 (character trans, integer iresid, integer m, integer n, integer nrhs, double precision, dimension( lda, * ) a, integer lda, double precision, dimension( ldx, * ) x, integer ldx, double precision, dimension( ldb, * ) b, integer ldb, double precision, dimension( ldb, * ) c, double precision, dimension( lwork ) work, integer lwork)

DQRT17

الغرض:

!>
!> DQRT17 يحسب النسبة
!>
!>    norm(R**T * op(A)) / ( norm(A) * alpha * max(M,N,NRHS) * EPS ),
!>
!> حيث R = B - op(A)*X، op(A) هي A أو A**T، اعتمادًا على TRANS، EPS
!> هي الـ epsilon للجهاز، و
!>
!>    alpha = norm(B) إذا كان IRESID = 1 (مشكلة الصفر المتبقي)
!>    alpha = norm(R) إذا كان IRESID = 2 (في حالات أخرى).
!>
!> المعيار المستخدم هو المعيار 1.
!>

المعلمات

TRANS 
!>          TRANS هو CHARACTER*1
!>          يحدد ما إذا كان منقول A يستخدم أم لا.
!>          = 'N':  لا يوجد منقول، op(A) = A.
!>          = 'T':  منقول، op(A) = A**T.
!>

IRESID

!>          IRESID هو INTEGER
!>          IRESID = 1 يشير إلى مشكلة الصفر المتبقي.
!>          IRESID = 2 يشير إلى متبقي غير صفري.
!>

M

!>          M هو INTEGER
!>          عدد الصفوف في المصفوفة A.
!>          إذا كان TRANS = 'N'، عدد الصفوف في المصفوفة B.
!>          إذا كان TRANS = 'T'، عدد الصفوف في المصفوفة X.
!>

N

!>          N هو INTEGER
!>          عدد الأعمدة في المصفوفة A.
!>          إذا كان TRANS = 'N'، عدد الصفوف في المصفوفة X.
!>          إذا كان TRANS = 'T'، عدد الصفوف في المصفوفة B.
!>

NRHS

!>          NRHS هو INTEGER
!>          عدد أعمدة المصفوفتين X و B.
!>

A

!>          A هي مصفوفة DOUBLE PRECISION، حجمها (LDA,N)
!>          المصفوفة A ذات الأبعاد m × n.
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرائد للمصفوفة A. LDA >= M.
!>

X

!>          X هي مصفوفة DOUBLE PRECISION، حجمها (LDX,NRHS)
!>          إذا كان TRANS = 'N'، المصفوفة X ذات الأبعاد n × nrhs.
!>          إذا كان TRANS = 'T'، المصفوفة X ذات الأبعاد m × nrhs.
!>

LDX

!>          LDX هو INTEGER
!>          البعد الرائد للمصفوفة X.
!>          إذا كان TRANS = 'N'، LDX >= N.
!>          إذا كان TRANS = 'T'، LDX >= M.
!>

B

!>          B هي مصفوفة DOUBLE PRECISION، حجمها (LDB,NRHS)
!>          إذا كان TRANS = 'N'، المصفوفة B ذات الأبعاد m × nrhs.
!>          إذا كان TRANS = 'T'، المصفوفة B ذات الأبعاد n × nrhs.
!>

LDB

!>          LDB هو INTEGER
!>          البعد الرائد للمصفوفة B.
!>          إذا كان TRANS = 'N'، LDB >= M.
!>          إذا كان TRANS = 'T'، LDB >= N.
!>

C

!>          C هي مصفوفة DOUBLE PRECISION، حجمها (LDB,NRHS)
!>

WORK

!>          WORK هي مصفوفة DOUBLE PRECISION، حجمها (LWORK)
!>

LWORK

!>          LWORK هو INTEGER
!>          طول المصفوفة WORK. LWORK >= NRHS*(M+N).
!>
