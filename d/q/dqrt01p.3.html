توثيق الوظيفة/الروتين الفرعي

subroutine dqrt01p (integer m, integer n, double precision, dimension( lda, * ) a, double precision, dimension( lda, * ) af, double precision, dimension( lda, * ) q, double precision, dimension( lda, * ) r, integer lda, double precision, dimension( * ) tau, double precision, dimension( lwork ) work, integer lwork, double precision, dimension( * ) rwork, double precision, dimension( * ) result)

DQRT01P

الغرض:

!>
!> DQRT01P تختبر DGEQRFP، والتي تحسب تحليل QR لمصفوفة m-by-n
!> A، وتختبر جزئياً DORGQR التي تشكل المصفوفة المتعامدة m-by-m
!> Q.
!>
!> DQRT01P تقارن R مع Q'*A، وتتحقق من أن Q متعامدة.
!>

المعلمات

M 
!>          M هو عدد صحيح (INTEGER)
!>          عدد صفوف المصفوفة A.  M >= 0.
!>

N

!>          N هو عدد صحيح (INTEGER)
!>          عدد أعمدة المصفوفة A.  N >= 0.
!>

A

!>          A هي مصفوفة من نوع double precision، ذات بعد (LDA,N)
!>          المصفوفة m-by-n A.
!>

AF

!>          AF هي مصفوفة من نوع double precision، ذات بعد (LDA,N)
!>          تفاصيل تحليل QR للمصفوفة A، كما تم إرجاعها بواسطة DGEQRFP.
!>          راجع DGEQRFP لمزيد من التفاصيل.
!>

Q

!>          Q هي مصفوفة من نوع double precision، ذات بعد (LDA,M)
!>          المصفوفة المتعامدة m-by-m Q.
!>

R

!>          R هي مصفوفة من نوع double precision، ذات بعد (LDA,max(M,N))
!>

LDA

!>          LDA هو عدد صحيح (INTEGER)
!>          البعد الرئيسي للمصفوفات A، AF، Q و R.
!>          LDA >= max(M,N).
!>

TAU

!>          TAU هي مصفوفة من نوع double precision، ذات بعد (min(M,N))
!>          العوامل القياسية للمُفاعلات الأولية، كما تم إرجاعها
!>          بواسطة DGEQRFP.
!>

WORK

!>          WORK هي مصفوفة من نوع double precision، ذات بعد (LWORK)
!>

LWORK

!>          LWORK هو عدد صحيح (INTEGER)
!>          بعد المصفوفة WORK.
!>

RWORK

!>          RWORK هي مصفوفة من نوع double precision، ذات بعد (M)
!>

RESULT

!>          RESULT هي مصفوفة من نوع double precision، ذات بعد (2)
!>          نسب الاختبار:
!>          RESULT(1) = norm( R - Q'*A ) / ( M * norm(A) * EPS )
!>          RESULT(2) = norm( I - Q'*Q ) / ( M * EPS )
!>
