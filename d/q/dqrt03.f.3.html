توثيق الوظيفة/الروتين الفرعي

subroutine dqrt03 (integer m, integer n, integer k, double precision, dimension( lda, * ) af, double precision, dimension( lda, * ) c, double precision, dimension( lda, * ) cc, double precision, dimension( lda, * ) q, integer lda, double precision, dimension( * ) tau, double precision, dimension( lwork ) work, integer lwork, double precision, dimension( * ) rwork, double precision, dimension( * ) result)

DQRT03

الغرض:

!>
!> DQRT03 تختبر DORMQR، التي تحسب Q*C، Q'*C، C*Q أو C*Q'.
!>
!> تقارن DQRT03 نتائج استدعاء DORMQR مع نتائج
!> تكوين Q صراحةً عن طريق استدعاء DORGQR ثم أداء ضرب المصفوفات
!> عن طريق استدعاء DGEMM.
!>

المعلمات

M 
!>          M هو عدد صحيح (INTEGER)
!>          رتبة المصفوفة المتعامدة Q.  M >= 0.
!>

N

!>          N هو عدد صحيح (INTEGER)
!>          عدد الصفوف أو الأعمدة للمصفوفة C؛ C هي m × n إذا
!>          تم تطبيق Q من اليسار، أو n × m إذا تم تطبيق Q من
!>          اليمين.  N >= 0.
!>

K

!>          K هو عدد صحيح (INTEGER)
!>          عدد المتجهات المنعكسة الأولية التي يُعرف حاصل ضربها
!>          المصفوفة المتعامدة Q.  M >= K >= 0.
!>

AF

!>          AF هي مصفوفة ذات دقة مزدوجة (DOUBLE PRECISION)، بأبعاد (LDA,N)
!>          تفاصيل تحليل QR لمصفوفة m × n، كما
!>          تم إرجاعها بواسطة DGEQRF. انظر DGEQRF لمزيد من التفاصيل.
!>

C

!>          C هي مصفوفة ذات دقة مزدوجة (DOUBLE PRECISION)، بأبعاد (LDA,N)
!>

CC

!>          CC هي مصفوفة ذات دقة مزدوجة (DOUBLE PRECISION)، بأبعاد (LDA,N)
!>

Q

!>          Q هي مصفوفة ذات دقة مزدوجة (DOUBLE PRECISION)، بأبعاد (LDA,M)
!>

LDA

!>          LDA هو عدد صحيح (INTEGER)
!>          البعد الرائد للمصفوفات AF، C، CC، و Q.
!>

TAU

!>          TAU هي مصفوفة ذات دقة مزدوجة (DOUBLE PRECISION)، بأبعاد (min(M,N))
!>          العوامل القياسية للمتجهات المنعكسة الأولية المقابلة
!>          لتحليل QR في AF.
!>

WORK

!>          WORK هي مصفوفة ذات دقة مزدوجة (DOUBLE PRECISION)، بأبعاد (LWORK)
!>

LWORK

!>          LWORK هو عدد صحيح (INTEGER)
!>          طول WORK. يجب أن يكون LWORK على الأقل M، ويجب أن يكون
!>          M*NB، حيث NB هو حجم الكتلة لهذا البيئة.
!>

RWORK

!>          RWORK هي مصفوفة ذات دقة مزدوجة (DOUBLE PRECISION)، بأبعاد (M)
!>

RESULT

!>          RESULT هي مصفوفة ذات دقة مزدوجة (DOUBLE PRECISION)، بأبعاد (4)
!>          نسب الاختبار تقارن بين تقنيتين لضرب مصفوفة عشوائية C
!>          بمصفوفة متعامدة m × m تسمى Q.
!>          RESULT(1) = norm( Q*C - Q*C )  / ( M * norm(C) * EPS )
!>          RESULT(2) = norm( C*Q - C*Q )  / ( M * norm(C) * EPS )
!>          RESULT(3) = norm( Q'*C - Q'*C )/ ( M * norm(C) * EPS )
!>          RESULT(4) = norm( C*Q' - C*Q' )/ ( M * norm(C) * EPS )
!>
