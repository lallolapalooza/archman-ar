subroutine dgbequb (integer m, integer n, integer kl, integer ku, double precision, dimension( ldab, * ) ab, integer ldab, double precision, dimension( * ) r, double precision, dimension( * ) c, double precision rowcnd, double precision colcnd, double precision amax, integer info)

DGBEQUB

الغرض:

!>
!> DGBEQUB تحسب معاملات القياس للصف والأعمدة المصممة لموازنة مصفوفة M-by-N
!> A وتقلل عدد حالاتها. تعيد R عوامل قياس الصف و C عوامل قياس الأعمدة،
!> التي يتم اختيارها لمحاولة جعل أكبر عنصر في كل صف وعمود من المصفوفة B
!> بعناصر B(i,j)=R(i)*A(i,j)*C(j) لها قيمة مطلقة لا تتجاوز الأساس.
!>
!> R(i) و C(j) مقيدة بأن تكون قوة للأساس بين
!> SMLNUM = أصغر رقم آمن و BIGNUM = أكبر رقم آمن. استخدام
!> عوامل القياس هذه لا يضمن تقليل عدد حالات A ولكنه يعمل بشكل جيد في الممارسة.
!>
!> يختلف هذا الروتين عن DGEEQU بتقييد عوامل القياس
!> بقوة للأساس. باستثناء الحمل الزائد والنقصان، فإن القياس ب
!> هذه العوامل لا يسبب أي أخطاء إضافية في التقريب. ومع ذلك، فإن
!> مقادير العناصر المقاسة لم تعد تقريبًا 1 ولكنها تقع
!> بين sqrt(radix) و 1/sqrt(radix).
!>

المعلمات

M 
!>          M هو عدد صحيح (INTEGER)
!>          عدد صفوف المصفوفة A.  M >= 0.
!>

N

!>          N هو عدد صحيح (INTEGER)
!>          عدد أعمدة المصفوفة A.  N >= 0.
!>

KL

!>          KL هو عدد صحيح (INTEGER)
!>          عدد الأقطار الفرعية داخل نطاق A.  KL >= 0.
!>

KU

!>          KU هو عدد صحيح (INTEGER)
!>          عدد الأقطار فوق الرئيسية داخل نطاق A.  KU >= 0.
!>

AB

!>          AB هو مصفوفة دقة مزدوجة (DOUBLE PRECISION array)، الأبعاد (LDAB,N)
!>          عند الدخول، المصفوفة A بتخزين نطاق، في الصفوف 1 إلى KL+KU+1.
!>          يتم تخزين العمود j من A في العمود j من
!>          المصفوفة AB كما يلي:
!>          AB(KU+1+i-j,j) = A(i,j)  لـ max(1,j-KU) <= i <= min(N,j+kl)
!>

LDAB

!>          LDAB هو عدد صحيح (INTEGER)
!>          البعد الرئيسي للمصفوفة A.  LDAB >= max(1,M).
!>

R

!>          R هو مصفوفة دقة مزدوجة (DOUBLE PRECISION array)، الأبعاد (M)
!>          إذا كانت INFO = 0 أو INFO > M، تحتوي R على عوامل قياس الصفوف
!>          لـ A.
!>

C

!>          C هو مصفوفة دقة مزدوجة (DOUBLE PRECISION array)، الأبعاد (N)
!>          إذا كانت INFO = 0، تحتوي C على عوامل قياس الأعمدة لـ A.
!>

ROWCND

!>          ROWCND هو دقة مزدوجة (DOUBLE PRECISION)
!>          إذا كانت INFO = 0 أو INFO > M، تحتوي ROWCND على نسبة
!>          أصغر R(i) إلى أكبر R(i). إذا كانت ROWCND >= 0.1 و
!>          AMAX ليست كبيرة جدًا أو صغيرة جدًا، فلا داعي
!>          للقياس بـ R.
!>

COLCND

!>          COLCND هو دقة مزدوجة (DOUBLE PRECISION)
!>          إذا كانت INFO = 0، تحتوي COLCND على نسبة
!>          أصغر C(i) إلى أكبر C(i). إذا كانت COLCND >= 0.1، فلا داعي
!>          للقياس بـ C.
!>

AMAX

!>          AMAX هو دقة مزدوجة (DOUBLE PRECISION)
!>          القيمة المطلقة لأكبر عنصر في المصفوفة. إذا كانت AMAX قريبة جدًا
!>          من الحمل الزائد أو قريبة جدًا من النقصان، فيجب قياس المصفوفة.
!>

INFO

!>          INFO هو عدد صحيح (INTEGER)
!>          = 0:  انتهاء ناجح
!>          < 0:  إذا كانت INFO = -i، فإن المعلمة i كان لها قيمة غير قانونية
!>          > 0:  إذا كانت INFO = i، و i هو
!>                <= M:  الصف i من A يساوي صفر تمامًا
!>                >  M:  العمود (i-M) من A يساوي صفر تمامًا
!>
