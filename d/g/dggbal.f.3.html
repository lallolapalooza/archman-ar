subroutine dggbal (character job, integer n, double precision, dimension( lda, * ) a, integer lda, double precision, dimension( ldb, * ) b, integer ldb, integer ilo, integer ihi, double precision, dimension( * ) lscale, double precision, dimension( * ) rscale, double precision, dimension( * ) work, integer info)

DGGBAL

الغرض:

!>
!> DGGBAL يوازن زوجًا من المصفوفات الحقيقية العامة (A,B).  هذا
!> يتضمن، أولاً، تبديل A و B بتحويلات تشابه لعزل القيم الذاتية في
!> العناصر من 1 إلى ILO-1 ومن IHI+1 إلى N على القطر؛ وثانياً، تطبيق
!> تحويل تشابه قطري على الصفوف والأعمدة ILO إلى IHI لجعل الصفوف
!> والأعمدة قريبة قدر الإمكان من المعيار. كلا الخطوتين اختياريتان.
!>
!> قد يقلل الموازنة من المعيار 1 للمصفوفات، ويحسن
!> دقة القيم الذاتية و/أو المتجهات الذاتية المحسوبة في
!> مشكلة القيمة الذاتية المعممة A*x = lambda*B*x.
!>

المعلمات

JOB 
!>          JOB هو CHARACTER*1
!>          يحدد العمليات التي سيتم إجراؤها على A و B:
!>          = 'N':  لا شيء:  فقط اضبط ILO = 1، IHI = N، LSCALE(I) = 1.0
!>                  و RSCALE(I) = 1.0 من أجل i = 1,...,N.
!>          = 'P':  تبديل فقط؛
!>          = 'S':  قياس فقط؛
!>          = 'B':  تبديل وقياس كلاهما.
!>

N

!>          N هو INTEGER
!>          ترتيب المصفوفات A و B.  N >= 0.
!>

A

!>          A هو مصفوفة DOUBLE PRECISION، ذات البعد (LDA,N)
!>          عند الدخول، المصفوفة A المدخلة.
!>          عند الخروج، يتم استبدال A بالمصفوفة المتوازنة.
!>          إذا كان JOB = 'N'، لم يتم الرجوع إلى A.
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرائد للمصفوفة A. LDA >= max(1,N).
!>

B

!>          B هو مصفوفة DOUBLE PRECISION، ذات البعد (LDB,N)
!>          عند الدخول، المصفوفة B المدخلة.
!>          عند الخروج، يتم استبدال B بالمصفوفة المتوازنة.
!>          إذا كان JOB = 'N'، لم يتم الرجوع إلى B.
!>

LDB

!>          LDB هو INTEGER
!>          البعد الرائد للمصفوفة B. LDB >= max(1,N).
!>

ILO

!>          ILO هو INTEGER
!>

IHI

!>          IHI هو INTEGER
!>          يتم ضبط ILO و IHI على أعداد صحيحة بحيث عند الخروج
!>          A(i,j) = 0 و B(i,j) = 0 إذا كان i > j و
!>          j = 1,...,ILO-1 أو i = IHI+1,...,N.
!>          إذا كان JOB = 'N' أو 'S'، فإن ILO = 1 و IHI = N.
!>

LSCALE

!>          LSCALE هي مصفوفة DOUBLE PRECISION، ذات البعد (N)
!>          تفاصيل التبادلات وعوامل القياس المطبقة
!>          على الجانب الأيسر من A و B. إذا كان P(j) هو فهرس
!>          الصف المتبادل مع الصف j، و D(j)
!>          هو عامل القياس المطبق على الصف j، فإن
!>            LSCALE(j) = P(j)    من أجل J = 1,...,ILO-1
!>                      = D(j)    من أجل J = ILO,...,IHI
!>                      = P(j)    من أجل J = IHI+1,...,N.
!>          ترتيب التبادلات التي تتم هو N إلى IHI+1،
!>          ثم 1 إلى ILO-1.
!>

RSCALE

!>          RSCALE هي مصفوفة DOUBLE PRECISION، ذات البعد (N)
!>          تفاصيل التبادلات وعوامل القياس المطبقة
!>          على الجانب الأيمن من A و B. إذا كان P(j) هو فهرس
!>          العمود المتبادل مع العمود j، و D(j)
!>          هو عامل القياس المطبق على العمود j، فإن
!>            LSCALE(j) = P(j)    من أجل J = 1,...,ILO-1
!>                      = D(j)    من أجل J = ILO,...,IHI
!>                      = P(j)    من أجل J = IHI+1,...,N.
!>          ترتيب التبادلات التي تتم هو N إلى IHI+1،
!>          ثم 1 إلى ILO-1.
!>

WORK

!>          WORK هي مصفوفة DOUBLE PRECISION، ذات البعد (lwork)
!>          يجب أن يكون lwork على الأقل max(1,6*N) عندما يكون JOB = 'S' أو 'B'،
!>          وعلى الأقل 1 عندما يكون JOB = 'N' أو 'P'.
!>

INFO

!>          INFO هو INTEGER
!>          = 0:  الانتهاء بنجاح
!>          < 0:  إذا كان INFO = -i، فإن المعلمة i كانت ذات قيمة غير قانونية.
!>
