subroutine dgetsls (character trans, integer m, integer n, integer nrhs, double precision, dimension( lda, * ) a, integer lda, double precision, dimension( ldb, * ) b, integer ldb, double precision, dimension( * ) work, integer lwork, integer info)

DGETSLS

الغرض:

!>
!> DGETSLS يحل أنظمة خطية حقيقية مفرطة التحديد أو ناقصة التحديد
!> تتضمن مصفوفة A بحجم M-by-N، باستخدام عامل QR نحيف طويل أو عامل LQ قصير عريض
!> للمصفوفة A.
!>
!> يُفترض أن A لها رتبة كاملة، ولا يتم توفير سوى حماية أساسية
!> ضد المصفوفات ناقصة الرتبة. هذا الإجراء الفرعي يكتشف فقط
!> نقص الرتبة التام، حيث يكون عنصر قطري في العامل المثلثي
!> لـ A يساوي صفرًا بالضبط.
!>
!> من الممكن أن تكون قيمة واحدة (أو أكثر) من العناصر القطرية للعامل المثلثي
!> لـ A أرقامًا صغيرة جدًا دون أن يقوم هذا الإجراء الفرعي بالإشارة
!> إلى خطأ. قد تكون الحلول المحسوبة لمثل هذه المصفوفات شبه ناقصة الرتبة
!> أقل دقة بسبب فقدان الدقة الرقمية.
!>
!>
!> الخيارات التالية متاحة:
!>
!> 1. إذا كان TRANS = 'N' و m >= n:  إيجاد الحل الأصغر مربعات لنظام
!>    مفرط التحديد، أي حل مشكلة المربعات الصغرى
!>                 minimize || B - A*X ||.
!>
!> 2. إذا كان TRANS = 'N' و m < n:  إيجاد الحل الأدنى معيار لنظام
!>    ناقص التحديد A * X = B.
!>
!> 3. إذا كان TRANS = 'T' و m >= n:  إيجاد الحل الأدنى معيار لنظام
!>    ناقص التحديد A**T * X = B.
!>
!> 4. إذا كان TRANS = 'T' و m < n:  إيجاد الحل الأصغر مربعات لنظام
!>    مفرط التحديد، أي حل مشكلة المربعات الصغرى
!>                 minimize || B - A**T * X ||.
!>
!> يمكن التعامل مع عدة متجهات للجانب الأيمن b ومتجهات الحل x
!> في استدعاء واحد؛ يتم تخزينها كأعمدة للمصفوفة
!> B ذات الجانب الأيمن و X ذات الحل.
!>

المعلمات

TRANS 
!>          TRANS هو CHARACTER*1
!>          = 'N': النظام الخطي يتضمن A؛
!>          = 'T': النظام الخطي يتضمن A**T.
!>

M

!>          M هو INTEGER
!>          عدد الصفوف في المصفوفة A.  M >= 0.
!>

N

!>          N هو INTEGER
!>          عدد الأعمدة في المصفوفة A.  N >= 0.
!>

NRHS

!>          NRHS هو INTEGER
!>          عدد الجوانب اليمنى، أي عدد
!>          أعمدة المصفوفتين B و X. NRHS >=0.
!>

A

!>          A هو مصفوفة DOUBLE PRECISION، ذات بُعد (LDA,N)
!>          عند الإدخال، المصفوفة A بحجم M-by-N.
!>          عند الخروج،
!>          تُستبدل A بتفاصيل عاملها QR أو LQ
!>          كما تم إرجاعه بواسطة DGEQR أو DGELQ.
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرائد للمصفوفة A.  LDA >= max(1,M).
!>

B

!>          B هي مصفوفة DOUBLE PRECISION، ذات بُعد (LDB,NRHS)
!>          عند الإدخال، مصفوفة B لمتجهات الجانب الأيمن، مخزنة
!>          كأعمدة؛ B بحجم M-by-NRHS إذا كان TRANS = 'N'، أو N-by-NRHS
!>          إذا كان TRANS = 'T'.
!>          عند الخروج، إذا كان INFO = 0، تُستبدل B بمتجهات الحل
!>          مخزنة كأعمدة:
!>          إذا كان TRANS = 'N' و m >= n، فإن الصفوف من 1 إلى n في B تحتوي على متجهات
!>          الحل الأصغر مربعات.
!>          إذا كان TRANS = 'N' و m < n، فإن الصفوف من 1 إلى N في B تحتوي على
!>          متجهات الحل الأدنى معيار؛
!>          إذا كان TRANS = 'T' و m >= n، فإن الصفوف من 1 إلى M في B تحتوي على
!>          متجهات الحل الأدنى معيار؛
!>          إذا كان TRANS = 'T' و m < n، فإن الصفوف من 1 إلى M في B تحتوي على
!>          متجهات الحل الأصغر مربعات.
!>

LDB

!>          LDB هو INTEGER
!>          البعد الرائد للمصفوفة B. LDB >= MAX(1,M,N).
!>

WORK

!>          (مساحة عمل) مصفوفة DOUBLE PRECISION، ذات بُعد (MAX(1,LWORK))
!>          عند الخروج، إذا كان INFO = 0، تحتوي WORK(1) على الحجم الأمثل (أو الحد الأدنى
!>          أو الأمثل، إذا افترض استعلام) LWORK.
!>          انظر LWORK للتفاصيل.
!>

LWORK

!>          LWORK هو INTEGER
!>          بُعد المصفوفة WORK. LWORK >= 1.
!>          إذا كان LWORK = -1 أو -2، يُفترض استعلام مساحة عمل.
!>          إذا كان LWORK = -1، يقوم الإجراء الفرعي بحساب الحجم الأمثل لـ WORK لـ
!>          أداء أمثل وإرجاع هذا الحجم في WORK(1).
!>          إذا كان LWORK = -2، يقوم الإجراء الفرعي بحساب الحجم الأدنى لـ WORK و
!>          يرجع هذا الحجم في WORK(1).
!>

INFO

!>          INFO هو INTEGER
!>          = 0:  انتهاء ناجح
!>          < 0:  إذا كان INFO = -i، كان المعامل i غير قانوني
!>          > 0:  إذا كان INFO =  i، العنصر القطري i في
!>                العامل المثلثي لـ A هو صفر تمامًا، لذا فإن A ليس لديها
!>                رتبة كاملة؛ لم يكن من الممكن حساب الحل الأصغر مربعات.
!>
