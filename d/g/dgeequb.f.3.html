subroutine dgeequb (integer m, integer n, double precision, dimension( lda, * ) a, integer lda, double precision, dimension( * ) r, double precision, dimension( * ) c, double precision rowcnd, double precision colcnd, double precision amax, integer info)

DGEEQUB

الغرض:

!>
!> DGEEQUB يحسب معاملات القياس للصفوف والأعمدة المقصودة لموازنة مصفوفة
!> M × N، A، وتقليل رقم شرطها. R ترجع عوامل قياس الصفوف و C عوامل قياس الأعمدة،
!> المختارة لمحاولة جعل أكبر عنصر في كل صف وعمود للمصفوفة B ذات
!> العناصر B(i,j)=R(i)*A(i,j)*C(j) لها قيمة مطلقة لا تزيد عن الأساس.
!>
!> R(i) و C(j) مقيدان بكونهما قوة للأساس بين
!> SMLNUM = أصغر رقم آمن و BIGNUM = أكبر رقم آمن. استخدام
!> عوامل القياس هذه لا يضمن تقليل رقم شرط A ولكنه يعمل بشكل جيد في الممارسة العملية.
!>
!> هذا الروتين يختلف عن DGEEQU بتقييد عوامل القياس
!> إلى قوة للأساس. بخلاف الفائض والتدفق الناقص، فإن القياس بـ
!> هذه العوامل لا يقدم أخطاء تقريب إضافية. ومع ذلك، فإن
!> قيم الأعمدة المقاسة لم تعد تقريبًا 1 ولكنها تقع
!> بين sqrt(radix) و 1/sqrt(radix).
!>

المعاملات

M 
!>          M هو عدد صحيح (INTEGER)
!>          عدد صفوف المصفوفة A. M >= 0.
!>

N

!>          N هو عدد صحيح (INTEGER)
!>          عدد أعمدة المصفوفة A. N >= 0.
!>

A

!>          A هو مصفوفة دقة مزدوجة (DOUBLE PRECISION array)، ذات البعد (LDA,N)
!>          المصفوفة M × N التي سيتم حساب معاملات موازنتها.
!>

LDA

!>          LDA هو عدد صحيح (INTEGER)
!>          البعد الرائد للمصفوفة A. LDA >= max(1,M).
!>

R

!>          R هو مصفوفة دقة مزدوجة (DOUBLE PRECISION array)، ذات البعد (M)
!>          إذا كانت INFO = 0 أو INFO > M، تحتوي R على عوامل قياس الصفوف
!>          لـ A.
!>

C

!>          C هو مصفوفة دقة مزدوجة (DOUBLE PRECISION array)، ذات البعد (N)
!>          إذا كانت INFO = 0، تحتوي C على عوامل قياس الأعمدة لـ A.
!>

ROWCND

!>          ROWCND هو دقة مزدوجة (DOUBLE PRECISION)
!>          إذا كانت INFO = 0 أو INFO > M، تحتوي ROWCND على نسبة
!>          أصغر R(i) إلى أكبر R(i). إذا كانت ROWCND >= 0.1 و
!>          AMAX ليست كبيرة جدًا أو صغيرة جدًا، فلا فائدة من
!>          القياس بـ R.
!>

COLCND

!>          COLCND هو دقة مزدوجة (DOUBLE PRECISION)
!>          إذا كانت INFO = 0، تحتوي COLCND على نسبة
!>          أصغر C(i) إلى أكبر C(i). إذا كانت COLCND >= 0.1، فلا فائدة من
!>          القياس بـ C.
!>

AMAX

!>          AMAX هو دقة مزدوجة (DOUBLE PRECISION)
!>          القيمة المطلقة لأكبر عنصر في المصفوفة. إذا كانت AMAX قريبة جدًا
!>          من الفائض أو قريبة جدًا من التدفق الناقص، فيجب قياس المصفوفة.
!>

INFO

!>          INFO هو عدد صحيح (INTEGER)
!>          = 0: نجاح الخروج
!>          < 0: إذا كانت INFO = -i، كان المعامل i ذا قيمة غير قانونية
!>          > 0: إذا كانت INFO = i، و i هو
!>                <= M: الصف i من A هو صفر تمامًا
!>                >  M: العمود (i-M) من A هو صفر تمامًا
!>
