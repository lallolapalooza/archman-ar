subroutine dgerqs (integer m, integer n, integer nrhs, double precision, dimension( lda, * ) a, integer lda, double precision, dimension( * ) tau, double precision, dimension( ldb, * ) b, integer ldb, double precision, dimension( lwork ) work, integer lwork, integer info)

DGERQS

الغرض:

!>
!> حساب حل ذي معيار أدنى
!>     min || A*X - B ||
!> باستخدام تحليل RQ
!>     A = R*Q
!> الذي تم حسابه بواسطة DGERQF.
!>

المعلمات

M 
!>          M هو عدد صحيح
!>          عدد صفوف المصفوفة A.  M >= 0.
!>

N

!>          N هو عدد صحيح
!>          عدد أعمدة المصفوفة A.  N >= M >= 0.
!>

NRHS

!>          NRHS هو عدد صحيح
!>          عدد أعمدة B.  NRHS >= 0.
!>

A

!>          A هو مصفوفة DOUBLE PRECISION، البعد (LDA,N)
!>          تفاصيل تحليل RQ للمصفوفة الأصلية A كما
!>          تم إرجاعه بواسطة DGERQF.
!>

LDA

!>          LDA هو عدد صحيح
!>          البعد الرائد للمصفوفة A.  LDA >= M.
!>

TAU

!>          TAU هو مصفوفة DOUBLE PRECISION، البعد (M)
!>          تفاصيل المصفوفة المتعامدة Q.
!>

B

!>          B هو مصفوفة DOUBLE PRECISION، البعد (LDB,NRHS)
!>          عند الإدخال، متجهات الطرف الأيمن للنظام الخطي.
!>          عند الخروج، متجهات الحل X.  كل متجه حل
!>          موجود في الصفوف 1:N من عمود في B.
!>

LDB

!>          LDB هو عدد صحيح
!>          البعد الرائد للمصفوفة B. LDB >= max(1,N).
!>

WORK

!>          WORK هو مصفوفة DOUBLE PRECISION، البعد (LWORK)
!>

LWORK

!>          LWORK هو عدد صحيح
!>          طول المصفوفة WORK.  يجب أن يكون LWORK على الأقل NRHS،
!>          ويجب أن يكون على الأقل NRHS*NB، حيث NB هو حجم الكتلة
!>          لهذه البيئة.
!>

INFO

!>          INFO هو عدد صحيح
!>          = 0: إنهاء ناجح
!>          < 0: إذا كان INFO = -i، فإن وسيطة i كان لديها قيمة غير قانونية
!>
