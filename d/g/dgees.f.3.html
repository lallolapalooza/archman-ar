توثيق الوظيفة/الروتين الفرعي

subroutine dgees (character jobvs, character sort, external select, integer n, double precision, dimension( lda, * ) a, integer lda, integer sdim, double precision, dimension( * ) wr, double precision, dimension( * ) wi, double precision, dimension( ldvs, * ) vs, integer ldvs, double precision, dimension( * ) work, integer lwork, logical, dimension( * ) bwork, integer info)

DGEES تحسب القيم الذاتية، الشكل لشور، واختياريًا، مصفوفة متجهات شور لمصفوفات GE

الغرض:

!>
!> DGEES تحسب، لمصفوفة حقيقية غير متناظرة N × N وهي A،
!> القيم الذاتية، الشكل لشور الحقيقي T، واختياريًا، مصفوفة متجهات شور Z.  هذا يعطي تحليل شور A = Z*T*(Z**T).
!>
!> اختياريًا، تقوم أيضًا بترتيب القيم الذاتية على القطر للشكل لشور بحيث تكون القيم الذاتية المحددة في الجزء العلوي الأيسر.
!> تشكل الأعمدة الرائدة لـ Z أساسًا متعامدًا للمتجه الفراغي المقابل للقيم الذاتية المحددة.
!>
!> تكون المصفوفة في الشكل لشور الحقيقي إذا كانت شبه مثلثية عليا مع كتل 1 × 1 و 2 × 2. سيتم توحيد الكتل 2 × 2 في
!> الشكل
!>         [  a  b  ]
!>         [  c  a  ]
!>
!> حيث b*c < 0. القيم الذاتية لمثل هذه الكتلة هي a ± sqrt(bc).
!>

المعلمات

JOBVS 
!>          JOBVS هو CHARACTER*1
!>          = 'N': لم يتم حساب متجهات شور؛
!>          = 'V': تم حساب متجهات شور.
!>

SORT

!>          SORT هو CHARACTER*1
!>          يحدد ما إذا كان سيتم ترتيب القيم الذاتية على
!>          قطر شكل شور أم لا.
!>          = 'N': لم يتم ترتيب القيم الذاتية؛
!>          = 'S': تم ترتيب القيم الذاتية (انظر SELECT).
!>

SELECT

!>          SELECT هي وظيفة LOGICAL لمعاملين من نوع DOUBLE PRECISION
!>          يجب تعريف SELECT خارجيًا في الروتين الفرعي المستدعي.
!>          إذا كان SORT = 'S'، يتم استخدام SELECT لتحديد القيم الذاتية لترتيبها
!>          إلى الجزء العلوي الأيسر من شكل شور.
!>          إذا كان SORT = 'N'، لا يتم الرجوع إلى SELECT.
!>          يتم تحديد القيمة الذاتية WR(j)+sqrt(-1)*WI(j) إذا كان
!>          SELECT(WR(j),WI(j)) صحيحًا؛ أي، إذا تم تحديد إحدى زوج من القيم الذاتية المترافقة المعقدة،
!>          فإنه يتم تحديد كلتا القيمتين الذاتيتين المعقدتين.
!>          لاحظ أن القيمة الذاتية المعقدة المحددة قد لا
!>          تحقق SELECT(WR(j),WI(j)) = .TRUE. بعد الترتيب، نظرًا لأن
!>          الترتيب قد يغير قيمة القيم الذاتية المعقدة
!>          (خاصة إذا كانت القيمة الذاتية غير مستقرة)؛ في هذه
!>          الحالة يتم تعيين INFO إلى N+2 (انظر INFO أدناه).
!>

N

!>          N هو INTEGER
!>          رتبة المصفوفة A. N >= 0.
!>

A

!>          A هي مصفوفة DOUBLE PRECISION، بُعدها (LDA,N)
!>          عند الإدخال، المصفوفة A من N × N.
!>          عند الخروج، تم استبدال A بشكل شور الحقيقي T.
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرئيسي للمصفوفة A. LDA >= max(1,N).
!>

SDIM

!>          SDIM هو INTEGER
!>          إذا كان SORT = 'N'، فإن SDIM = 0.
!>          إذا كان SORT = 'S'، فإن SDIM = عدد القيم الذاتية (بعد الترتيب)
!>                         التي يكون SELECT صحيحًا لها. (الأزواج المترافقة المعقدة التي يكون SELECT صحيحًا لأحدها
!>                         تُحتسب كـ 2.)
!>

WR

!>          WR هي مصفوفة DOUBLE PRECISION، بُعدها (N)
!>

WI

!>          WI هي مصفوفة DOUBLE PRECISION، بُعدها (N)
!>          تحتوي WR و WI على الأجزاء الحقيقية والتخيلية،
!>          على التوالي، للقيم الذاتية المحسوبة بنفس الترتيب
!>          الذي تظهر به على قطر شكل شور الناتج T.
!>          ستظهر الأزواج المترافقة المعقدة من القيم الذاتية
!>          بشكل متتالي مع القيمة الذاتية ذات الجزء التخيلي الموجب أولاً.
!>

VS

!>          VS هي مصفوفة DOUBLE PRECISION، بُعدها (LDVS,N)
!>          إذا كان JOBVS = 'V'، فإن VS تحتوي على المصفوفة المتعامدة Z لمتجهات شور.
!>          إذا كان JOBVS = 'N'، لا يتم الرجوع إلى VS.
!>

LDVS

!>          LDVS هو INTEGER
!>          البعد الرئيسي للمصفوفة VS. LDVS >= 1؛ إذا كان
!>          JOBVS = 'V'، فإن LDVS >= N.
!>

WORK

!>          WORK هي مصفوفة DOUBLE PRECISION، بُعدها (MAX(1,LWORK))
!>          عند الخروج، إذا كان INFO = 0، فإن WORK(1) يحتوي على LWORK الأمثل.
!>

LWORK

!>          LWORK هو INTEGER
!>          بُعد المصفوفة WORK. LWORK >= max(1,3*N).
!>          لأداء جيد، يجب أن يكون LWORK أكبر بشكل عام.
!>
!>          إذا كان LWORK = -1، فيُفترض استعلام عن مساحة عمل؛ الروتين
!>          يحسب فقط الحجم الأمثل لمصفوفة WORK، ويعيد
!>          هذه القيمة كالإدخال الأول في مصفوفة WORK، ولا يصدر أي رسالة خطأ
!>          بخصوص LWORK بواسطة XERBLA.
!>

BWORK

!>          BWORK هي مصفوفة LOGICAL، بُعدها (N)
!>          لا يتم الرجوع إليها إذا كان SORT = 'N'.
!>

INFO

!>          INFO هو INTEGER
!>          = 0: خروج ناجح
!>          < 0: إذا كان INFO = -i، فإن المعامل i كان له قيمة غير قانونية.
!>          > 0: إذا كان INFO = i، و i هو
!>             <= N: فشل خوارزمية QR في حساب جميع
!>                   القيم الذاتية؛ العناصر 1:ILO-1 و i+1:N من WR و WI
!>                   تحتوي على تلك القيم الذاتية التي تقاربت؛ إذا كان
!>                   JOBVS = 'V'، فإن VS تحتوي على المصفوفة التي تحول A
!>                   إلى شكل شور المتقارب جزئيًا.
!>             = N+1: لم يكن من الممكن إعادة ترتيب القيم الذاتية لأن بعض
!>                   القيم الذاتية كانت قريبة جدًا من الانفصال (المشكلة
!>                   غير مستقرة للغاية)؛
!>             = N+2: بعد إعادة الترتيب، غيرت تقريبات الدقة قيم بعض
!>                   القيم الذاتية المعقدة بحيث لم تعد القيم الذاتية الرائدة في
!>                   شكل شور تحقق SELECT=.TRUE. قد يكون هذا ناتجًا أيضًا عن
!>                   نقص التدفق بسبب التحجيم.
!>
