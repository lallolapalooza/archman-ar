subroutine dgtts2 (integer itrans, integer n, integer nrhs, double precision, dimension( * ) dl, double precision, dimension( * ) d, double precision, dimension( * ) du, double precision, dimension( * ) du2, integer, dimension( * ) ipiv, double precision, dimension( ldb, * ) b, integer ldb)

DGTTS2 يحل نظام معادلات خطية مع مصفوفة ثلاثية الأقطار باستخدام تحليل LU الذي تم حسابه بواسطة sgttrf.

الغرض:

!>
!> DGTTS2 يحل أحد أنظمة المعادلات
!>    A*X = B  أو  A**T*X = B,
!> مع مصفوفة ثلاثية الأقطار A باستخدام تحليل LU الذي تم حسابه
!> بواسطة DGTTRF.
!>

المعلمات

ITRANS 
!>          ITRANS هو عدد صحيح (INTEGER)
!>          يحدد شكل نظام المعادلات.
!>          = 0:  A * X = B  (بدون تبديل)
!>          = 1:  A**T* X = B  (تبديل)
!>          = 2:  A**T* X = B  (تبديل مترافق = تبديل)
!>

N

!>          N هو عدد صحيح (INTEGER)
!>          رتبة المصفوفة A.
!>

NRHS

!>          NRHS هو عدد صحيح (INTEGER)
!>          عدد الأطراف اليمنى، أي عدد أعمدة
!>          المصفوفة B.  NRHS >= 0.
!>

DL

!>          DL هي مصفوفة ذات دقة مزدوجة (DOUBLE PRECISION)، بعد (N-1)
!>          المضاعفات التي تحدد المصفوفة L من
!>          تحليل LU لـ A.
!>

D

!>          D هي مصفوفة ذات دقة مزدوجة (DOUBLE PRECISION)، بعد (N)
!>          العناصر القطرية n للقطر العلوي للمصفوفة U من
!>          تحليل LU لـ A.
!>

DU

!>          DU هي مصفوفة ذات دقة مزدوجة (DOUBLE PRECISION)، بعد (N-1)
!>          عناصر القطر العلوي الأول لـ U.
!>

DU2

!>          DU2 هي مصفوفة ذات دقة مزدوجة (DOUBLE PRECISION)، بعد (N-2)
!>          عناصر القطر العلوي الثاني لـ U.
!>

IPIV

!>          IPIV هي مصفوفة أعداد صحيحة (INTEGER)، بعد (N)
!>          مؤشرات المحور؛ لـ 1 <= i <= n، تم تبديل الصف i للمصفوفة
!>          مع الصف IPIV(i).  سيكون IPIV(i) دائمًا إما
!>          i أو i+1؛ يشير IPIV(i) = i إلى عدم الحاجة إلى تبديل الصف.
!>

B

!>          B هي مصفوفة ذات دقة مزدوجة (DOUBLE PRECISION)، بعد (LDB,NRHS)
!>          عند الإدخال، المصفوفة الخاصة بمتجهات الطرف الأيمن B.
!>          عند الخروج، يتم استبدال B بمتجهات الحل X.
!>

LDB

!>          LDB هو عدد صحيح (INTEGER)
!>          البعد الرئيسي للمصفوفة B.  LDB >= max(1,N).
!>
