توثيق الوظيفة/الإجراء الفرعي

subroutine dgemlqt (character side, character trans, integer m, integer n, integer k, integer mb, double precision, dimension( ldv, * ) v, integer ldv, double precision, dimension( ldt, * ) t, integer ldt, double precision, dimension( ldc, * ) c, integer ldc, double precision, dimension( * ) work, integer info)

DGEMLQT

الغرض:

!>
!> DGEMLQT بالكتابة فوق المصفوفة العادية M-by-N C بـ
!>
!>                 SIDE = 'L'     SIDE = 'R'
!> TRANS = 'N':      Q C            C Q
!> TRANS = 'T':   Q**T C            C Q**T
!>
!> حيث Q هي مصفوفة متعامدة حقيقية معرفة كناتج K
!> عاكسات أولية:
!>
!>       Q = H(1) H(2) . . . H(K) = I - V T V**T
!>
!> تم إنشاؤها باستخدام التمثيل المدمج WY كما تم إرجاعه بواسطة DGELQT.
!>
!> Q من الرتبة M إذا كان SIDE = 'L' ومن الرتبة N إذا كان SIDE = 'R'.
!>

المعلمات

SIDE 
!>          SIDE is CHARACTER*1
!>          = 'L': تطبيق Q أو Q**T من اليسار؛
!>          = 'R': تطبيق Q أو Q**T من اليمين.
!>

TRANS

!>          TRANS is CHARACTER*1
!>          = 'N':  عدم النقل، تطبيق Q؛
!>          = 'C':  النقل، تطبيق Q**T.
!>

M

!>          M is INTEGER
!>          عدد الصفوف في المصفوفة C. M >= 0.
!>

N

!>          N is INTEGER
!>          عدد الأعمدة في المصفوفة C. N >= 0.
!>

K

!>          K is INTEGER
!>          عدد العاكسات الأولية التي يحدد منتجها
!>          المصفوفة Q.
!>          إذا كان SIDE = 'L'، فإن M >= K >= 0؛
!>          إذا كان SIDE = 'R'، فإن N >= K >= 0.
!>

MB

!>          MB is INTEGER
!>          حجم الكتلة المستخدمة لتخزين T. K >= MB >= 1.
!>          يجب أن تكون هذه هي نفس قيمة MB المستخدمة لإنشاء T
!>          في DGELQT.
!>

V

!>          V is DOUBLE PRECISION array, dimension
!>                               (LDV,M) if SIDE = 'L',
!>                               (LDV,N) if SIDE = 'R'
!>          يجب أن يحتوي الصف i على المتجه الذي يحدد
!>          العاكس الأولي H(i)، لـ i = 1,2,...,k، كما تم إرجاعه بواسطة
!>          DGELQT في الصفوف K الأولى من مصفوفة الوسيط A.
!>

LDV

!>          LDV is INTEGER
!>          البعد الرئيسي للمصفوفة V. LDV >= max(1,K).
!>

T

!>          T is DOUBLE PRECISION array, dimension (LDT,K)
!>          العوامل المثلثية العلوية للعواكس المجمعة
!>          كما تم إرجاعه بواسطة DGELQT، مخزنة كمصفوفة MB-by-K.
!>

LDT

!>          LDT is INTEGER
!>          البعد الرئيسي للمصفوفة T. LDT >= MB.
!>

C

!>          C is DOUBLE PRECISION array, dimension (LDC,N)
!>          عند الدخول، المصفوفة M-by-N C.
!>          عند الخروج، يتم الكتابة فوق C بواسطة Q C، Q**T C، C Q**T أو C Q.
!>

LDC

!>          LDC is INTEGER
!>          البعد الرئيسي للمصفوفة C. LDC >= max(1,M).
!>

WORK

!>          WORK is DOUBLE PRECISION array. The dimension of
!>          WORK is N*MB if SIDE = 'L', or  M*MB if SIDE = 'R'.
!>

INFO

!>          INFO is INTEGER
!>          = 0:  الخروج الناجح
!>          < 0:  إذا كان INFO = -i، فإن الوسيط i كان له قيمة غير قانونية
!>
