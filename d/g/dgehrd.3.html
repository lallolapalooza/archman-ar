subroutine dgehrd (integer n, integer ilo, integer ihi, double precision, dimension( lda, * ) a, integer lda, double precision, dimension( * ) tau, double precision, dimension( * ) work, integer lwork, integer info)

DGEHRD

الغرض:

!>
!> DGEHRD يقلل مصفوفة عامة حقيقية A إلى شكل هسنبرغ علوي H بواسطة
!> تحويل تشابه متعامد:  Q**T * A * Q = H .
!>

المعلمات

N 
!>          N هو عدد صحيح (INTEGER)
!>          رتبة المصفوفة A.  N >= 0.
!>

ILO

!>          ILO هو عدد صحيح (INTEGER)
!>

IHI

!>          IHI هو عدد صحيح (INTEGER)
!>
!>          يفترض أن A بالفعل مثلثية علوية في الصفوف
!>          والأعمدة 1:ILO-1 و IHI+1:N. عادة ما يتم تعيين ILO و IHI
!>          بواسطة استدعاء سابق لـ DGEBAL؛ وإلا يجب تعيينهما
!>          إلى 1 و N على التوالي. انظر التفاصيل الإضافية.
!>          1 <= ILO <= IHI <= N، إذا كان N > 0؛ ILO=1 و IHI=0، إذا كان N=0.
!>

A

!>          A هي مصفوفة دقة مزدوجة (DOUBLE PRECISION)، ذات أبعاد (LDA,N)
!>          عند الإدخال، المصفوفة العامة N×N المراد تقليلها.
!>          عند الخروج، يتم الكتابة فوق المثلث العلوي والقطر الفرعي الأول من A
!>          بواسطة المصفوفة هسنبرغ العلوية H، والعناصر الموجودة أسفل القطر الفرعي الأول،
!>          بواسطة المصفوفة TAU، تمثل المصفوفة المتعامدة Q كناتج للمنعكسات الأولية.
!>          انظر التفاصيل الإضافية.
!>

LDA

!>          LDA هو عدد صحيح (INTEGER)
!>          البعد الرئيسي للمصفوفة A.  LDA >= max(1,N).
!>

TAU

!>          TAU هي مصفوفة دقة مزدوجة (DOUBLE PRECISION)، ذات أبعاد (N-1)
!>          العوامل العددية للمنعكسات الأولية (انظر التفاصيل الإضافية).
!>          يتم تعيين العناصر 1:ILO-1 و IHI:N-1 من TAU إلى
!>          صفر.
!>

WORK

!>          WORK هي مصفوفة دقة مزدوجة (DOUBLE PRECISION)، ذات أبعاد (MAX(1,LWORK))
!>          عند الخروج، إذا كان INFO = 0، فإن WORK(1) يعيد LWORK المثلى.
!>

LWORK

!>          LWORK هو عدد صحيح (INTEGER)
!>          طول المصفوفة WORK.  LWORK >= max(1,N).
!>          لأداء جيد، يجب أن يكون LWORK أكبر بشكل عام.
!>
!>          إذا كان LWORK = -1، يفترض استعلام عن مساحة العمل؛ يقوم الروتين
!>          بحساب حجم مساحة العمل المثلى فقط، ويعيد
!>          هذه القيمة كإدخال أول في مساحة العمل، ولا يصدر رسالة خطأ
!>          متعلقة بـ LWORK بواسطة XERBLA.
!>

INFO

!>          INFO هو عدد صحيح (INTEGER)
!>          = 0:  انتهاء ناجح
!>          < 0:  إذا كان INFO = -i، كان الوسيط i غير قانوني.
!>

التفاصيل الإضافية:

!>
!>  المصفوفة Q ممثلة كناتج لـ (ihi-ilo) من المنعكسات الأولية
!>
!>     Q = H(ilo) H(ilo+1) . . . H(ihi-1).
!>
!>  كل H(i) له الشكل
!>
!>     H(i) = I - tau * v * v**T
!>
!>  حيث tau هو عدد حقيقي، و v هو متجه حقيقي مع
!>  v(1:i) = 0, v(i+1) = 1 و v(ihi+1:n) = 0؛ v(i+2:ihi) يتم تخزينه عند
!>  الخروج في A(i+2:ihi,i)، و tau في TAU(i).
!>
!>  يتم توضيح محتويات A بالمثال التالي، مع
!>  n = 7, ilo = 2 و ihi = 6:
!>
!>  عند الإدخال،                        عند الخروج،
!>
!>  ( a   a   a   a   a   a   a )    (  a   a   h   h   h   h   a )
!>  (     a   a   a   a   a   a )    (      a   h   h   h   h   a )
!>  (     a   a   a   a   a   a )    (      h   h   h   h   h   h )
!>  (     a   a   a   a   a   a )    (      v2  h   h   h   h   h )
!>  (     a   a   a   a   a   a )    (      v2  v3  h   h   h   h )
!>  (     a   a   a   a   a   a )    (      v2  v3  v4  h   h   h )
!>  (                         a )    (                          a )
!>
!>  حيث a يمثل عنصرًا من المصفوفة الأصلية A، h يمثل عنصرًا معدلًا
!>  من مصفوفة هسنبرغ العلوية H، و vi يمثل عنصرًا من المتجه الذي
!>  يحدد H(i).
!>
!>  هذا الملف هو تعديل طفيف لروتين DGEHRD في LAPACK-3.0
!>  يتضمن التحسينات المقترحة بواسطة Quintana-Orti و
!>  Van de Geijn (2006). (انظر DLAHR2.)
!>

التعريف في السطر 166 من الملف dgehrd.f.
