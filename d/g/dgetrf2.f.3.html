توثيق الوظيفة/الروتين الفرعي

recursive subroutine dgetrf2 (integer m, integer n, double precision, dimension( lda, * ) a, integer lda, integer, dimension( * ) ipiv, integer info)

DGETRF2

الغرض:

!>
!> تحسب DGETRF2 عامل LU لمصفوفة عامة M-by-N، A
!> باستخدام التنصيف الجزئي مع تبديلات الصفوف.
!>
!> العامل له الشكل
!>    A = P * L * U
!> حيث P مصفوفة تبديل، L مصفوفة مثلثية سفلية ذات عناصر قطرية موحدة
!> (شبه منحرف سفلي إذا كان m > n)، و U مصفوفة مثلثية علوية
!> (شبه منحرف علوي إذا كان m < n).
!>
!> هذه هي النسخة العودية للخوارزمية. تقسم
!> المصفوفة إلى أربع مصفوفات فرعية:
!>
!>        [  A11 | A12  ]  حيث A11 هي n1 × n1 و A22 هي n2 × n2
!>    A = [ -----|----- ]  مع n1 = min(m,n)/2
!>        [  A21 | A22  ]       n2 = n-n1
!>
!>                                       [ A11 ]
!> يستدعي الروتين الفرعي نفسه لعامل [ --- ]،
!>                                       [ A12 ]
!>                 [ A12 ]
!> يقوم بالتبديلات على [ --- ]، يحل A12، يحدث A22،
!>                 [ A22 ]
!>
!> ثم يستدعي نفسه لعامل A22 ويقوم بالتبديلات على A21.
!>
!>

المعاملات

M 
!>          M هو عدد صحيح (INTEGER)
!>          عدد صفوف المصفوفة A. M >= 0.
!>

N

!>          N هو عدد صحيح (INTEGER)
!>          عدد أعمدة المصفوفة A. N >= 0.
!>

A

!>          A هي مصفوفة ذات دقة مزدوجة (DOUBLE PRECISION)، بأبعاد (LDA,N)
!>          عند الإدخال، المصفوفة M-by-N المراد تحليلها.
!>          عند الخروج، عوامل L و U من التحليل
!>          A = P*L*U؛ لا يتم تخزين عناصر الوحدة القطرية لـ L.
!>

LDA

!>          LDA هو عدد صحيح (INTEGER)
!>          البعد الرائد للمصفوفة A. LDA >= max(1,M).
!>

IPIV

!>          IPIV هي مصفوفة أعداد صحيحة (INTEGER)، بأبعاد (min(M,N))
!>          فهارس التنصيف؛ من أجل 1 <= i <= min(M,N)، تم تبديل الصف i
!>          من المصفوفة مع الصف IPIV(i).
!>

INFO

!>          INFO هو عدد صحيح (INTEGER)
!>          = 0: خروج ناجح
!>          < 0: إذا كان INFO = -i، فإن المعامل i كان له قيمة غير قانونية
!>          > 0: إذا كان INFO = i، فإن U(i,i) يساوي صفرًا تمامًا. تم إكمال التحليل،
!>                ولكن العامل U هو مفرد تمامًا، وسينتج قسمة على صفر إذا تم استخدامه
!>                لحل نظام من المعادلات.
!>
