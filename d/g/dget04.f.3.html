subroutine dget04 (integer n, integer nrhs, double precision, dimension( ldx, * ) x, integer ldx, double precision, dimension( ldxact, * ) xact, integer ldxact, double precision rcond, double precision resid)

DGET04

الغرض:

!>
!> DGET04 يحسب الفرق بين حل محسوب والحل الحقيقي لنظام المعادلات الخطية.
!>
!> RESID =  ( norm(X-XACT) * RCOND ) / ( norm(XACT) * EPS ),
!> حيث RCOND هو مقلوب العدد المشروط و EPS هو التمثيل العشري الآلي.
!>

المعلمات

N 
!>          N هو عدد صحيح (INTEGER)
!>          عدد الصفوف في المصفوفات X و XACT.  N >= 0.
!>

NRHS

!>          NRHS هو عدد صحيح (INTEGER)
!>          عدد الأعمدة في المصفوفات X و XACT.  NRHS >= 0.
!>

X

!>          X هو مصفوفة الدقة المزدوجة (DOUBLE PRECISION)، البعد (LDX,NRHS)
!>          متجهات الحل المحسوبة.  كل متجه مخزن كعمود في المصفوفة X.
!>

LDX

!>          LDX هو عدد صحيح (INTEGER)
!>          البعد الرائد للمصفوفة X.  LDX >= max(1,N).
!>

XACT

!>          XACT هو مصفوفة الدقة المزدوجة (DOUBLE PRECISION)، البعد( LDX, NRHS )
!>          متجهات الحل الدقيقة.  كل متجه مخزن كعمود في المصفوفة XACT.
!>

LDXACT

!>          LDXACT هو عدد صحيح (INTEGER)
!>          البعد الرائد للمصفوفة XACT.  LDXACT >= max(1,N).
!>

RCOND

!>          RCOND هو الدقة المزدوجة (DOUBLE PRECISION)
!>          مقلوب العدد المشروط لمصفوفة المعاملات في نظام المعادلات.
!>

RESID

!>          RESID هو الدقة المزدوجة (DOUBLE PRECISION)
!>          الحد الأقصى على متجهات الحل NRHS لـ
!>          ( norm(X-XACT) * RCOND ) / ( norm(XACT) * EPS )
!>
