الغرض:

!>
!> DGGSVP3 يحسب المصفوفات المتعامدة U، V و Q بحيث
!>
!>                    N-K-L  K    L
!>  U**T*A*Q =     K ( 0    A12  A13 )  إذا كان M-K-L >= 0؛
!>                 L ( 0     0   A23 )
!>             M-K-L ( 0     0    0  )
!>
!>                  N-K-L  K    L
!>         =     K ( 0    A12  A13 )  إذا كان M-K-L < 0؛
!>             M-K ( 0     0   A23 )
!>
!>                  N-K-L  K    L
!>  V**T*B*Q =   L ( 0     0   B13 )
!>             P-L ( 0     0    0  )
!>
!> حيث المصفوفة A12 ذات الأبعاد K×K والمصفوفة B13 ذات الأبعاد L×L غير منفردة
!> ومثلثية عليا؛ A23 مثلثية عليا ذات الأبعاد L×L إذا كان M-K-L >= 0،
!> وإلا فإن A23 شبه منحرف عليا ذات الأبعاد (M-K)×L. K+L = الرتبة العددية الفعالة
!> للمصفوفة (A**T,B**T)**T ذات الأبعاد (M+P)×N.
!>
!> هذا التحلل هو خطوة المعالجة المسبقة لحساب تحليل القيم المفردة المعممة (GSVD)، انظر الوحدة الفرعية
!> DGGSVD3.
!>

المعلمات:

JOBU 
!>          JOBU هي CHARACTER*1
!>          = 'U':  يتم حساب المصفوفة المتعامدة U؛
!>          = 'N':  لا يتم حساب U.
!>

JOBV

!>          JOBV هي CHARACTER*1
!>          = 'V':  يتم حساب المصفوفة المتعامدة V؛
!>          = 'N':  لا يتم حساب V.
!>

JOBQ

!>          JOBQ هي CHARACTER*1
!>          = 'Q':  يتم حساب المصفوفة المتعامدة Q؛
!>          = 'N':  لا يتم حساب Q.
!>

M

!>          M هو INTEGER
!>          عدد صفوف المصفوفة A.  M >= 0.
!>

P

!>          P هو INTEGER
!>          عدد صفوف المصفوفة B.  P >= 0.
!>

N

!>          N هو INTEGER
!>          عدد أعمدة المصفوفات A و B.  N >= 0.
!>

A

!>          A هي مصفوفة DOUBLE PRECISION، البعد (LDA,N)
!>          عند الإدخال، المصفوفة A ذات الأبعاد M×N.
!>          عند الخروج، تحتوي A على المصفوفة المثلثية (أو شبه المنحرف)
!>          الموصوفة في قسم الغرض.
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرائد للمصفوفة A. LDA >= max(1,M).
!>

B

!>          B هي مصفوفة DOUBLE PRECISION، البعد (LDB,N)
!>          عند الإدخال، المصفوفة B ذات الأبعاد P×N.
!>          عند الخروج، تحتوي B على المصفوفة المثلثية الموصوفة في
!>          قسم الغرض.
!>

LDB

!>          LDB هو INTEGER
!>          البعد الرائد للمصفوفة B. LDB >= max(1,P).
!>

TOLA

!>          TOLA هي DOUBLE PRECISION
!>

TOLB

!>          TOLB هي DOUBLE PRECISION
!>
!>          TOLA و TOLB هما العتبتان لتحديد الرتبة العددية الفعالة
!>          للمصفوفة B وكتلة فرعية من A. بشكل عام،
!>             TOLA = MAX(M,N)*norm(A)*MACHEPS،
!>             TOLB = MAX(P,N)*norm(B)*MACHEPS.
!>          قد يؤثر حجم TOLA و TOLB على حجم أخطاء الخلفية
!>          للتحليل.
!>

K

!>          K هو INTEGER
!>

L

!>          L هو INTEGER
!>
!>          عند الخروج، تحدد K و L أبعاد الكتل الفرعية
!>          الموصوفة في قسم الغرض.
!>          K + L = الرتبة العددية الفعالة لـ (A**T,B**T)**T.
!>

U

!>          U هي مصفوفة DOUBLE PRECISION، البعد (LDU,M)
!>          إذا كان JOBU = 'U'، فإن U تحتوي على المصفوفة المتعامدة U.
!>          إذا كان JOBU = 'N'، لا يتم الرجوع إلى U.
!>

LDU

!>          LDU هو INTEGER
!>          البعد الرائد للمصفوفة U. LDU >= max(1,M) إذا كان
!>          JOBU = 'U'؛ LDU >= 1 بخلاف ذلك.
!>

V

!>          V هي مصفوفة DOUBLE PRECISION، البعد (LDV,P)
!>          إذا كان JOBV = 'V'، فإن V تحتوي على المصفوفة المتعامدة V.
!>          إذا كان JOBV = 'N'، لا يتم الرجوع إلى V.
!>

LDV

!>          LDV هو INTEGER
!>          البعد الرائد للمصفوفة V. LDV >= max(1,P) إذا كان
!>          JOBV = 'V'؛ LDV >= 1 بخلاف ذلك.
!>

Q

!>          Q هي مصفوفة DOUBLE PRECISION، البعد (LDQ,N)
!>          إذا كان JOBQ = 'Q'، فإن Q تحتوي على المصفوفة المتعامدة Q.
!>          إذا كان JOBQ = 'N'، لا يتم الرجوع إلى Q.
!>

LDQ

!>          LDQ هو INTEGER
!>          البعد الرائد للمصفوفة Q. LDQ >= max(1,N) إذا كان
!>          JOBQ = 'Q'؛ LDQ >= 1 بخلاف ذلك.
!>

IWORK

!>          IWORK هي مصفوفة INTEGER، البعد (N)
!>

TAU

!>          TAU هي مصفوفة DOUBLE PRECISION، البعد (N)
!>

WORK

!>          WORK هي مصفوفة DOUBLE PRECISION، البعد (MAX(1,LWORK))
!>          عند الخروج، إذا كان INFO = 0، فإن WORK(1) يعيد LWORK الأمثل.
!>

LWORK

!>          LWORK هو INTEGER
!>          بعد المصفوفة WORK. LWORK >= 1.
!>
!>          إذا كان LWORK = -1، فيفترض استعلام مساحة العمل؛ تقوم الوحدة فقط
!>          بحساب الحجم الأمثل للمصفوفة WORK، وتعيد
!>          هذه القيمة كالدخول الأول للمصفوفة WORK، ولا تصدر رسالة خطأ
!>          متعلقة بـ LWORK بواسطة XERBLA.
!>

INFO

!>          INFO هو INTEGER
!>          = 0:  تم الخروج بنجاح
!>          < 0:  إذا كان INFO = -i، فإن المعلمة i كانت ذات قيمة غير قانونية.
!>
