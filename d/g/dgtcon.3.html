وثائق الوظيفة/الإجراء الفرعي

subroutine dgtcon (character norm, integer n, double precision, dimension( * ) dl, double precision, dimension( * ) d, double precision, dimension( * ) du, double precision, dimension( * ) du2, integer, dimension( * ) ipiv, double precision anorm, double precision rcond, double precision, dimension( * ) work, integer, dimension( * ) iwork, integer info)

DGTCON

الغرض:

!>
!> DGTCON يقدر مقلوب رقم الشرط لمصفوفة ثلاثية قطرية حقيقية A
!> باستخدام تحليل LU كما تم حسابه بواسطة DGTTRF.
!>
!> يتم الحصول على تقدير لـ norm(inv(A))، ويتم حساب مقلوب رقم الشرط
!> كـ RCOND = 1 / (ANORM * norm(inv(A))).
!>

المعلمات

NORM 
!>          NORM is CHARACTER*1
!>          يحدد ما إذا كان رقم شرط الـ 1-norm أو الـ infinity-norm مطلوبًا:
!>          = '1' or 'O':  1-norm;
!>          = 'I':         Infinity-norm.
!>

N

!>          N is INTEGER
!>          رتبة المصفوفة A.  N >= 0.
!>

DL

!>          DL is DOUBLE PRECISION array, dimension (N-1)
!>          المضاعفات (n-1) التي تحدد المصفوفة L من تحليل LU للمصفوفة A
!>          كما تم حسابها بواسطة DGTTRF.
!>

D

!>          D is DOUBLE PRECISION array, dimension (N)
!>          العناصر القطرية n للمصفوفة المثلثية العليا U من تحليل LU للمصفوفة A.
!>

DU

!>          DU is DOUBLE PRECISION array, dimension (N-1)
!>          العناصر (n-1) فوق القطرية الأولى لـ U.
!>

DU2

!>          DU2 is DOUBLE PRECISION array, dimension (N-2)
!>          العناصر (n-2) فوق القطرية الثانية لـ U.
!>

IPIV

!>          IPIV is INTEGER array, dimension (N)
!>          مؤشرات المحور؛ لـ 1 <= i <= n، تم تبديل الصف i من المصفوفة
!>          مع الصف IPIV(i).  سيكون IPIV(i) دائمًا إما i أو i+1؛
!>          IPIV(i) = i يشير إلى أنه لم يكن هناك حاجة لتبديل الصف.
!>

ANORM

!>          ANORM is DOUBLE PRECISION
!>          إذا كان NORM = '1' أو 'O'، فإن 1-norm للمصفوفة الأصلية A.
!>          إذا كان NORM = 'I'، فإن infinity-norm للمصفوفة الأصلية A.
!>

RCOND

!>          RCOND is DOUBLE PRECISION
!>          مقلوب رقم شرط المصفوفة A، محسوبًا كـ RCOND = 1/(ANORM * AINVNM)،
!>          حيث AINVNM هو تقدير لـ 1-norm لـ inv(A) محسوب في هذا الروتين.
!>

WORK

!>          WORK is DOUBLE PRECISION array, dimension (2*N)
!>

IWORK

!>          IWORK is INTEGER array, dimension (N)
!>

INFO

!>          INFO is INTEGER
!>          = 0:  انتهاء ناجح
!>          < 0:  إذا كان INFO = -i، فإن الوسيط i كان له قيمة غير قانونية
!>
