subroutine dget40 (double precision rmax, integer lmax, integer, dimension( 2 ) ninfo, integer knt, integer nin)

DGET40

الغرض:

!>
!> DGET40 تختبر DTGEXC، وهو إجراء لتبديل الكتل المجاورة (إما
!> 1 × 1 أو 2 × 2) على القطر لمصفوفة تبادلية في شكل Schur العام الحقيقي.
!> وبالتالي، يحسب DTGEXC مصفوفات متعامدة Q و Z بحيث
!>
!> Q' * ( [ A B ], [ D E ] ) * Z = ( [ C1 B1 ], [ F1 E1 ] )
!> ( [ 0 C ] [   F ] )        ( [ 0  A1 ] [    D1] )
!>
!> حيث (C1,F1) مشابه لـ (C,F) و (A1,D1) مشابه لـ (A,D).
!> يُفترض أن كل من (A,D) و (C,F) في الشكل القياسي
!> ويتم إرجاع (A1,D1) و (C1,F1) بنفس
!> الخصائص.
!>
