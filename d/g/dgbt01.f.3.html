توثيق الوظيفة/الإجراء الفرعي

subroutine dgbt01 (integer m, integer n, integer kl, integer ku, double precision, dimension( lda, * ) a, integer lda, double precision, dimension( ldafac, * ) afac, integer ldafac, integer, dimension( * ) ipiv, double precision, dimension( * ) work, double precision resid)

DGBT01

الغرض:

!>
!> DGBT01 تعيد بناء مصفوفة A من تحليلها L*U وتحسب الباقي:
!>    norm(L*U - A) / ( N * norm(A) * EPS ),
!> حيث EPS هو ثابت دقة الآلة.
!>
!> يتم حساب التعبير L*U - A عمودًا تلو الآخر، لذا لا يتم تعديل A و
!> AFAC.
!>

المعلمات

M 
!>          M هو عدد صحيح (INTEGER)
!>          عدد صفوف المصفوفة A.  M >= 0.
!>

N

!>          N هو عدد صحيح (INTEGER)
!>          عدد أعمدة المصفوفة A.  N >= 0.
!>

KL

!>          KL هو عدد صحيح (INTEGER)
!>          عدد الأقطار الفرعية داخل شريط A.  KL >= 0.
!>

KU

!>          KU هو عدد صحيح (INTEGER)
!>          عدد الأقطار العلوية داخل شريط A.  KU >= 0.
!>

A

!>          A هي مصفوفة من النوع DOUBLE PRECISION، البعد (LDA,N)
!>          المصفوفة الأصلية A بتخزين شريطي، مخزنة في الصفوف من 1 إلى
!>          KL+KU+1.
!>

LDA

!>          LDA هو عدد صحيح (INTEGER).
!>          البعد الرائد للمصفوفة A.  LDA >= max(1,KL+KU+1).
!>

AFAC

!>          AFAC هي مصفوفة من النوع DOUBLE PRECISION، البعد (LDAFAC,N)
!>          الشكل المحلل للمصفوفة A.  AFAC تحتوي على العوامل الشريطية
!>          L و U من تحليل L*U، كما تم حسابه بواسطة
!>          DGBTRF.  يتم تخزين U كمصفوفة مثلثية علوية شريطية مع
!>          KL+KU قطر علوي في الصفوف من 1 إلى KL+KU+1، والعوامل
!>          المستخدمة أثناء التحليل مخزنة في الصفوف
!>          KL+KU+2 إلى 2*KL+KU+1.  انظر DGBTRF لمزيد من التفاصيل.
!>

LDAFAC

!>          LDAFAC هو عدد صحيح (INTEGER)
!>          البعد الرائد للمصفوفة AFAC.
!>          LDAFAC >= max(1,2*KL*KU+1).
!>

IPIV

!>          IPIV هي مصفوفة من النوع INTEGER، البعد (min(M,N))
!>          فهارس المحور من DGBTRF.
!>

WORK

!>          WORK هي مصفوفة من النوع DOUBLE PRECISION، البعد (2*KL+KU+1)
!>

RESID

!>          RESID هو DOUBLE PRECISION
!>          norm(L*U - A) / ( N * norm(A) * EPS )
!>
