وثائق الوظائف/الإجراءات الفرعية

subroutine dgttrs (character trans, integer n, integer nrhs, double precision, dimension( * ) dl, double precision, dimension( * ) d, double precision, dimension( * ) du, double precision, dimension( * ) du2, integer, dimension( * ) ipiv, double precision, dimension( ldb, * ) b, integer ldb, integer info)

DGTTRS

الغرض:

!>
!> DGTTRS تحل أحد أنظمة المعادلات
!>    A*X = B  أو  A**T*X = B,
!> مع مصفوفة ثلاثية قطرية A باستخدام التحليل LU الذي تم حسابه
!> بواسطة DGTTRF.
!>

المعاملات

TRANS 
!>          TRANS is CHARACTER*1
!>          يحدد شكل نظام المعادلات.
!>          = 'N':  A * X = B  (بدون تبديل)
!>          = 'T':  A**T* X = B  (تبديل)
!>          = 'C':  A**T* X = B  (تبديل مترافق = تبديل)
!>

N

!>          N is INTEGER
!>          رتبة المصفوفة A.
!>

NRHS

!>          NRHS is INTEGER
!>          عدد الأطراف اليمنى، أي عدد أعمدة
!>          المصفوفة B.  NRHS >= 0.
!>

DL

!>          DL is DOUBLE PRECISION array, dimension (N-1)
!>          المضاعفات (n-1) التي تحدد المصفوفة L من
!>          تحليل LU للمصفوفة A.
!>

D

!>          D is DOUBLE PRECISION array, dimension (N)
!>          العناصر القطرية n للمصفوفة المثلثية العليا U من
!>          تحليل LU للمصفوفة A.
!>

DU

!>          DU is DOUBLE PRECISION array, dimension (N-1)
!>          العناصر (n-1) للقطر العلوي الأول لـ U.
!>

DU2

!>          DU2 is DOUBLE PRECISION array, dimension (N-2)
!>          العناصر (n-2) للقطر العلوي الثاني لـ U.
!>

IPIV

!>          IPIV is INTEGER array, dimension (N)
!>          مؤشرات المحور؛ لـ 1 <= i <= n، تم تبديل الصف i من المصفوفة
!>          مع الصف IPIV(i).  سيكون IPIV(i) دائمًا إما
!>          i أو i+1؛ IPIV(i) = i يشير إلى عدم الحاجة إلى تبديل صف.
!>

B

!>          B is DOUBLE PRECISION array, dimension (LDB,NRHS)
!>          عند الدخول، المصفوفة للمتجهات الطرفية اليمنى B.
!>          عند الخروج، يتم استبدال B بمتجهات الحل X.
!>

LDB

!>          LDB is INTEGER
!>          البعد الرائد للمصفوفة B.  LDB >= max(1,N).
!>

INFO

!>          INFO is INTEGER
!>          = 0:  خروج ناجح
!>          < 0:  إذا كان INFO = -i، فإن المعامل i كان له قيمة غير قانونية
!>
