subroutine dgelsx (integer m, integer n, integer nrhs, double precision, dimension( lda, * ) a, integer lda, double precision, dimension( ldb, * ) b, integer ldb, integer, dimension( * ) jpvt, double precision rcond, integer rank, double precision, dimension( * ) work, integer info)

DGELSX يحل أنظمة مفرطة التحديد أو ناقصة التحديد لمصفوفات GE

الغرض:

!>
!> هذا الإجراء الفرعي مهمل وتم استبداله بالإجراء الفرعي DGELSY.
!>
!> DGELSX يحسب الحد الأدنى من حلول المعيار لمشكلة أقل مربعات خطية حقيقية:
!>     تصغير || A * X - B ||
!> باستخدام تحلل متعامد كامل لـ A. A هي مصفوفة M-by-N
!!> والتي قد تكون ذات رتبة ناقصة.
!>
!> يمكن التعامل مع عدة متجهات جانبية يمنى b ومتجهات حل x
!> في استدعاء واحد؛ يتم تخزينها كأعمدة في
!> المصفوفة الجانبية اليمنى B ذات الأبعاد M-by-NRHS
!> ومصفوفة الحل X ذات الأبعاد N-by-NRHS.
!>
!> يقوم الإجراء الفرعي أولاً بحساب تحليل QR مع تنقيط الأعمدة:
!>     A * P = Q * [ R11 R12 ]
!>                 [  0  R22 ]
!> حيث يتم تعريف R11 كأكبر مصفوفة فرعية رائدة تقدير شرطها
!> أقل من 1/RCOND. ترتيب R11، RANK،
!> هو الرتبة الفعالة لـ A.
!>
!> بعد ذلك، يعتبر R22 ضئيلاً، ويتم القضاء على R12
!> باستخدام تحويلات متعامدة من اليمين، مما يؤدي إلى
!> التحلل المتعامد الكامل:
!>    A * P = Q * [ T11 0 ] * Z
!>                [  0  0 ]
!> ثم الحل الأدنى للمعايير هو
!>    X = P * Z**T [ inv(T11)*Q1**T*B ]
!>                 [        0         ]
!> حيث تتكون Q1 من أول RANK أعمدة من Q.
!>

المعلمات

M 
!>          M هو عدد صحيح (INTEGER)
!>          عدد صفوف المصفوفة A. M >= 0.
!>

N

!>          N هو عدد صحيح (INTEGER)
!>          عدد أعمدة المصفوفة A. N >= 0.
!>

NRHS

!>          NRHS هو عدد صحيح (INTEGER)
!>          عدد الأسطر الجانبية اليمنى، أي عدد
!>          أعمدة المصفوفات B و X. NRHS >= 0.
!>

A

!>          A هي مصفوفة DOUBLE PRECISION، ذات أبعاد (LDA,N)
!>          عند الدخول، المصفوفة A ذات الأبعاد M-by-N.
!>          عند الخروج، تم استبدال A بتفاصيل
!>          التحلل المتعامد الكامل الخاص بها.
!>

LDA

!>          LDA هو عدد صحيح (INTEGER)
!>          البعد الرئيسي للمصفوفة A. LDA >= max(1,M).
!>

B

!>          B هي مصفوفة DOUBLE PRECISION، ذات أبعاد (LDB,NRHS)
!>          عند الدخول، المصفوفة الجانبية اليمنى B ذات الأبعاد M-by-NRHS.
!>          عند الخروج، مصفوفة الحل X ذات الأبعاد N-by-NRHS.
!>          إذا كانت m >= n والرتبة = n، فإن مجموع مربعات البواقي
!>          للحل في العمود i يُعطى بمجموع
!>          المربعات للعناصر N+1:M في هذا العمود.
!>

LDB

!>          LDB هو عدد صحيح (INTEGER)
!>          البعد الرئيسي للمصفوفة B. LDB >= max(1,M,N).
!>

JPVT

!>          JPVT هي مصفوفة أعداد صحيحة (INTEGER)، ذات أبعاد (N)
!>          عند الدخول، إذا كان JPVT(i) .ne. 0، فإن العمود i من A هو
!>          عمود أولي، وإلا فهو عمود حر. قبل
!>          تحليل QR لـ A، يتم تبديل جميع الأعمدة الأولية
!>          إلى المواضع الرائدة؛ فقط الأعمدة الحرة المتبقية
!>          يتم نقلها نتيجة لتنقيط الأعمدة
!>          خلال التحلل.
!>          عند الخروج، إذا كان JPVT(i) = k، فإن العمود i من A*P
!>          كان العمود k من A.
!>

RCOND

!>          RCOND هو DOUBLE PRECISION
!>          يستخدم RCOND لتحديد الرتبة الفعالة لـ A، والتي
!>          يتم تعريفها على أنها ترتيب أكبر مصفوفة فرعية مثلثة رائدة
!>          R11 في تحليل QR مع تنقيط A،
!>          التي تقدير رقم حالتها < 1/RCOND.
!>

RANK

!>          RANK هو عدد صحيح (INTEGER)
!>          الرتبة الفعالة لـ A، أي ترتيب المصفوفة الفرعية
!>          R11. هذا هو نفس ترتيب المصفوفة الفرعية T11
!>          في التحلل المتعامد الكامل لـ A.
!>

WORK

!>          WORK هي مصفوفة DOUBLE PRECISION، ذات أبعاد
!>                      (max( min(M,N)+3*N, 2*min(M,N)+NRHS )),
!>

INFO

!>          INFO هو عدد صحيح (INTEGER)
!>          = 0:  انتهاء ناجح
!>          < 0:  إذا كان INFO = -i، فإن المعلمة i كانت لها قيمة غير قانونية
!>
