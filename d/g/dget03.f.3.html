توثيق الوظيفة/الروتين الفرعي

subroutine dget03 (integer n, double precision, dimension( lda, * ) a, integer lda, double precision, dimension( ldainv, * ) ainv, integer ldainv, double precision, dimension( ldwork, * ) work, integer ldwork, double precision, dimension( * ) rwork, double precision rcond, double precision resid)

DGET03

الغرض:

!>
!> DGET03 يحسب الباقي لضرب مصفوفة عامة في معكوسها:
!>    norm( I - AINV*A ) / ( N * norm(A) * norm(AINV) * EPS ),
!> حيث EPS هو الخطأ المطبعي.
!>

المعلمات

N 
!>          N هو عدد صحيح
!>          عدد الصفوف والأعمدة للمصفوفة A.  N >= 0.
!>

A

!>          A هي مصفوفة Double Precision، أبعادها (LDA,N)
!>          المصفوفة الأصلية A بحجم N x N.
!>

LDA

!>          LDA هو عدد صحيح
!>          البعد الرئيسي للمصفوفة A.  LDA >= max(1,N).
!>

AINV

!>          AINV هي مصفوفة Double Precision، أبعادها (LDAINV,N)
!>          معكوس المصفوفة A.
!>

LDAINV

!>          LDAINV هو عدد صحيح
!>          البعد الرئيسي للمصفوفة AINV.  LDAINV >= max(1,N).
!>

WORK

!>          WORK هي مصفوفة Double Precision، أبعادها (LDWORK,N)
!>

LDWORK

!>          LDWORK هو عدد صحيح
!>          البعد الرئيسي للمصفوفة WORK.  LDWORK >= max(1,N).
!>

RWORK

!>          RWORK هي مصفوفة Double Precision، أبعادها (N)
!>

RCOND

!>          RCOND هو Double Precision
!>          مقلوب رقم شرط A، المحسوب كـ
!>          ( 1/norm(A) ) / norm(AINV).
!>

RESID

!>          RESID هو Double Precision
!>          norm(I - AINV*A) / ( N * norm(A) * norm(AINV) * EPS )
!>
