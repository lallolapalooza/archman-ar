subroutine dgeqrt (integer m, integer n, integer nb, double precision, dimension( lda, * ) a, integer lda, double precision, dimension( ldt, * ) t, integer ldt, double precision, dimension( * ) work, integer info)

DGEQRT

الغرض:

!>
!> DGEQRT يحسب عامل QR مسدود لمصفوفة حقيقية M-by-N تسمى A
!> باستخدام تمثيل WY المضغوط لـ Q.
!>

المعلمات

M 
!>          M هو عدد صحيح (INTEGER)
!>          عدد صفوف المصفوفة A.  M >= 0.
!>

N

!>          N هو عدد صحيح (INTEGER)
!>          عدد أعمدة المصفوفة A.  N >= 0.
!>

NB

!>          NB هو عدد صحيح (INTEGER)
!>          حجم الكتلة الذي سيتم استخدامه في QR المسدود.  MIN(M,N) >= NB >= 1.
!>

A

!>          A هو مصفوفة من نوع DOUBLE PRECISION، بأبعاد (LDA,N)
!>          عند الدخول، المصفوفة A ذات الأبعاد M-by-N.
!>          عند الخروج، العناصر على القطر وما فوقه في المصفوفة
!>          تحتوي على المصفوفة شبه المنحرفة العلوية R ذات الأبعاد min(M,N)-by-N (R
!>          مثلثة علوية إذا كان M >= N)؛ العناصر تحت القطر
!>          هي أعمدة V.
!>

LDA

!>          LDA هو عدد صحيح (INTEGER)
!>          البعد الرائد للمصفوفة A.  LDA >= max(1,M).
!>

T

!>          T هي مصفوفة من نوع DOUBLE PRECISION، بأبعاد (LDT,MIN(M,N))
!>          المُنعكسات القطاعية المثلثية العلوية مخزنة بتنسيق مضغوط
!>          كسلسلة من القطاعات المثلثية العلوية.  انظر أدناه
!>          للمزيد من التفاصيل.
!>

LDT

!>          LDT هو عدد صحيح (INTEGER)
!>          البعد الرائد للمصفوفة T.  LDT >= NB.
!>

WORK

!>          WORK هي مصفوفة من نوع DOUBLE PRECISION، بأبعاد (NB*N)
!>

INFO

!>          INFO هو عدد صحيح (INTEGER)
!>          = 0:  اكتمال ناجح
!>          < 0:  إذا كان INFO = -i، فإن الوسيط i كان له قيمة غير قانونية
!>

تفاصيل إضافية:

!>
!>  المصفوفة V تخزن المُنعكسات الأولية H(i) في العمود i
!>  أسفل القطر. على سبيل المثال، إذا كانت M=5 و N=3، فإن المصفوفة V هي
!>
!>               V = (  1       )
!>                   ( v1  1    )
!>                   ( v1 v2  1 )
!>                   ( v1 v2 v3 )
!>                   ( v1 v2 v3 )
!>
!>  حيث تمثل الـ vi's المتجهات التي تحدد H(i)، والتي يتم إرجاعها
!>  في المصفوفة A.  الـ 1s على طول قطر V لا يتم تخزينها في A.
!>
!>  لتكن K=MIN(M,N).  عدد القطاعات هو B = ceiling(K/NB)، حيث يكون كل
!>  قطاع بحجم NB باستثناء القطاع الأخير، والذي يكون بحجم
!>  IB = K - (B-1)*NB.  لكل من الـ B قطاعات، يتم حساب عامل مُنعكس قطاعي
!>  علوي: T1، T2، ... TB.  تُخزن الـ T ذات الأبعاد NB-by-NB (و IB-by-IB
!>  للقطاع الأخير) في المصفوفة T ذات الأبعاد NB-by-K كما يلي:
!>
!>               T = (T1 T2 ... TB).
!>
