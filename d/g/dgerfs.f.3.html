subroutine dgerfs (character trans, integer n, integer nrhs, double precision, dimension( lda, * ) a, integer lda, double precision, dimension( ldaf, * ) af, integer ldaf, integer, dimension( * ) ipiv, double precision, dimension( ldb, * ) b, integer ldb, double precision, dimension( ldx, * ) x, integer ldx, double precision, dimension( * ) ferr, double precision, dimension( * ) berr, double precision, dimension( * ) work, integer, dimension( * ) iwork, integer info)

DGERFS

الغرض:

!>
!> DGERFS يحسن الحل المحسوب لنظام المعادلات الخطية
!> ويوفر حدود الخطأ وتقديرات الخطأ الخلفي للحل.
!>

المعلمات

TRANS 
!>          TRANS is CHARACTER*1
!>          يحدد شكل نظام المعادلات:
!>          = 'N':  A * X = B     (لا يوجد تبديل)
!>          = 'T':  A**T * X = B  (تبديل)
!>          = 'C':  A**H * X = B  (التبديل المترافق = تبديل)
!>

N

!>          N is INTEGER
!>          رتبة المصفوفة A.  N >= 0.
!>

NRHS

!>          NRHS is INTEGER
!>          عدد الأطراف اليمنى، أي عدد أعمدة
!>          المصفوفتين B و X.  NRHS >= 0.
!>

A

!>          A is DOUBLE PRECISION array, dimension (LDA,N)
!>          المصفوفة الأصلية A بحجم N*N.
!>

LDA

!>          LDA is INTEGER
!>          البعد الرئيسي للمصفوفة A.  LDA >= max(1,N).
!>

AF

!>          AF is DOUBLE PRECISION array, dimension (LDAF,N)
!>          العوامل L و U من تحليل A = P*L*U
!>          كما تم حسابه بواسطة DGETRF.
!>

LDAF

!>          LDAF is INTEGER
!>          البعد الرئيسي للمصفوفة AF.  LDAF >= max(1,N).
!>

IPIV

!>          IPIV is INTEGER array, dimension (N)
!>          مؤشرات المحور من DGETRF؛ لـ 1<=i<=N، الصف i من
!>          المصفوفة تم تبديله مع الصف IPIV(i).
!>

B

!>          B is DOUBLE PRECISION array, dimension (LDB,NRHS)
!>          مصفوفة الطرف الأيمن B.
!>

LDB

!>          LDB is INTEGER
!>          البعد الرئيسي للمصفوفة B.  LDB >= max(1,N).
!>

X

!>          X is DOUBLE PRECISION array, dimension (LDX,NRHS)
!>          عند الدخول، مصفوفة الحل X، كما تم حسابها بواسطة DGETRS.
!>          عند الخروج، مصفوفة الحل المحسنة X.
!>

LDX

!>          LDX is INTEGER
!>          البعد الرئيسي للمصفوفة X.  LDX >= max(1,N).
!>

FERR

!>          FERR is DOUBLE PRECISION array, dimension (NRHS)
!>          التقدير لحد الخطأ الأمامي لكل متجه حل
!>          X(j) (العمود j من مصفوفة الحل X).
!>          إذا كان XTRUE هو الحل الحقيقي المقابل لـ X(j)، فإن FERR(j)
!>          هو تقدير لحد أعلى للقيمة المطلقة لـ (X(j) - XTRUE) مقسومة على القيمة المطلقة لـ
!>          أكبر عنصر في X(j).  التقدير موثوق به بقدر
!>          التقدير لـ RCOND، وهو تقريبًا دائمًا زيادة طفيفة
!>          للخطأ الحقيقي.
!>

BERR

!>          BERR is DOUBLE PRECISION array, dimension (NRHS)
!>          الخطأ الخلفي النسبي المكوني لكل متجه حل
!>          X(j) (أي، أصغر تغيير نسبي في
!>          أي عنصر من A أو B يجعل X(j) حلاً دقيقًا).
!>

WORK

!>          WORK is DOUBLE PRECISION array, dimension (3*N)
!>

IWORK

!>          IWORK is INTEGER array, dimension (N)
!>

INFO

!>          INFO is INTEGER
!>          = 0:  خروج ناجح
!>          < 0:  إذا كان INFO = -i، فإن الوسيط i كان لديه قيمة غير قانونية
!>
