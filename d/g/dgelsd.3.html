subroutine dgelsd (integer m, integer n, integer nrhs, double precision, dimension( lda, * ) a, integer lda, double precision, dimension( ldb, * ) b, integer ldb, double precision, dimension( * ) s, double precision rcond, integer rank, double precision, dimension( * ) work, integer lwork, integer, dimension( * ) iwork, integer info)

DGELSD يحسب الحل الأدنى للمعايير لمشكلة المربعات الصغرى لمصفوفات GE

الغرض:

!>
!> DGELSD يحسب الحل الأدنى للمعايير لمشكلة المربعات الصغرى الخطية الحقيقية:
!>     minimize 2-norm(| b - A*x |)
!> باستخدام تحليل القيم المفردة (SVD) للمصفوفة A. المصفوفة A هي مصفوفة M-by-N
!> قد تكون ذات رتبة ناقصة.
!>
!> يمكن التعامل مع العديد من متجهات الطرف الأيمن b ومتجهات الحل x في
!> استدعاء واحد؛ يتم تخزينها كأعمدة في المصفوفة اليمنى B
!> ومتجه الحل X (كلاهما M-by-NRHS).
!>
!> يتم حل المشكلة في ثلاث خطوات:
!> (1) اختزال مصفوفة المعاملات A إلى شكل ثنائي القطر باستخدام
!>     تحويلات هاوسهولدر، مما يؤدي إلى اختزال المشكلة الأصلية
!>     إلى مشكلة BLS.
!> (2) حل مشكلة BLS باستخدام نهج التقسيم والغزو.
!> (3) تطبيق جميع تحويلات هاوسهولدر للخلف لحل
!>     مشكلة المربعات الصغرى الأصلية.
!>
!> يتم تحديد الرتبة الفعالة للمصفوفة A عن طريق اعتبار القيم المفردة
!> التي تقل عن RCOND مضروبة في أكبر قيمة مفردة كأصفار.
!>
!>

المعاملات

M 
!>          M هو عدد صحيح (INTEGER)
!>          عدد الصفوف في A. M >= 0.
!>

N

!>          N هو عدد صحيح (INTEGER)
!>          عدد الأعمدة في A. N >= 0.
!>

NRHS

!>          NRHS هو عدد صحيح (INTEGER)
!>          عدد الأطراف اليمنى، أي عدد الأعمدة
!>          في المصفوفتين B و X. NRHS >= 0.
!>

A

!>          A هي مصفوفة ذات دقة مزدوجة (DOUBLE PRECISION)، ذات بُعد (LDA,N)
!>          عند الدخول، مصفوفة A ذات الأبعاد M-by-N.
!>          عند الخروج، تم تدمير A.
!>

LDA

!>          LDA هو عدد صحيح (INTEGER)
!>          البعد الرائد للمصفوفة A. LDA >= max(1,M).
!>

B

!>          B هي مصفوفة ذات دقة مزدوجة (DOUBLE PRECISION)، ذات بُعد (LDB,NRHS)
!>          عند الدخول، مصفوفة الطرف الأيمن B ذات الأبعاد M-by-NRHS.
!>          عند الخروج، يتم استبدال B بمصفوفة الحل X ذات الأبعاد N-by-NRHS.
!>          إذا كانت m >= n و RANK = n، فإن مجموع مربعات الخطأ
!>          للحل في العمود i يُعطى بمجموع مربعات العناصر n+1:m في ذلك العمود.
!>

LDB

!>          LDB هو عدد صحيح (INTEGER)
!>          البعد الرائد للمصفوفة B. LDB >= max(1,max(M,N)).
!>

S

!>          S هي مصفوفة ذات دقة مزدوجة (DOUBLE PRECISION)، ذات بُعد (min(M,N))
!>          القيم المفردة للمصفوفة A بترتيب تنازلي.
!>          رقم الشرط للمصفوفة A بالنسبة للقيمة 2 = S(1)/S(min(m,n)).
!>

RCOND

!>          RCOND هي دقة مزدوجة (DOUBLE PRECISION)
!>          يُستخدم RCOND لتحديد الرتبة الفعالة للمصفوفة A.
!>          تُعامل القيم المفردة S(i) <= RCOND*S(1) كأصفار.
!>          إذا كان RCOND < 0، يتم استخدام دقة الآلة بدلاً من ذلك.
!>

RANK

!>          RANK هو عدد صحيح (INTEGER)
!>          الرتبة الفعالة للمصفوفة A، أي عدد القيم المفردة
!>          التي تكون أكبر من RCOND*S(1).
!>

WORK

!>          WORK هي مصفوفة ذات دقة مزدوجة (DOUBLE PRECISION)، ذات بُعد (MAX(1,LWORK))
!>          عند الخروج، إذا كان INFO = 0، فإن WORK(1) يعيد قيمة LWORK المثلى.
!>

LWORK

!>          LWORK هو عدد صحيح (INTEGER)
!>          بُعد المصفوفة WORK. يجب أن يكون LWORK على الأقل 1.
!>          يعتمد الحد الأدنى الدقيق لمساحة العمل المطلوبة على M،
!>          N و NRHS. طالما أن LWORK على الأقل
!>              12*N + 2*N*SMLSIZ + 8*N*NLVL + N*NRHS + (SMLSIZ+1)**2,
!>          إذا كانت M أكبر من أو تساوي N أو
!>              12*M + 2*M*SMLSIZ + 8*M*NLVL + M*NRHS + (SMLSIZ+1)**2,
!>          إذا كانت M أقل من N، فإن الكود سينفذ بشكل صحيح.
!>          SMLSIZ يتم إرجاعه بواسطة ILAENV وهو يساوي الحد الأقصى
!>          لحجم المشاكل الفرعية في أسفل شجرة الحساب (عادة حوالي 25)، و
!>             NLVL = MAX( 0, INT( LOG_2( MIN( M,N )/(SMLSIZ+1) ) ) + 1 )
!>          لأداء جيد، يجب أن يكون LWORK أكبر بشكل عام.
!>
!>          إذا كان LWORK = -1، فسيتم افتراض استعلام مساحة العمل؛ الروتين
!>          يحسب فقط الحجم الأمثل لمصفوفة WORK، يعيد
!>          هذه القيمة كالإدخال الأول لمصفوفة WORK، ولا يتم إصدار رسالة خطأ
!>          متعلقة بـ LWORK بواسطة XERBLA.
!>

IWORK

!>          IWORK هي مصفوفة أعداد صحيحة (INTEGER)، ذات بُعد (MAX(1,LIWORK))
!>          LIWORK >= max(1, 3 * MINMN * NLVL + 11 * MINMN),
!>          حيث MINMN = MIN( M,N ).
!>          عند الخروج، إذا كان INFO = 0، فإن IWORK(1) يعيد قيمة LIWORK الدنيا.
!>

INFO

!>          INFO هو عدد صحيح (INTEGER)
!>          = 0:  انتهاء ناجح
!>          < 0:  إذا كان INFO = -i، فإن المعامل i كان له قيمة غير قانونية.
!>          > 0:  فشل الخوارزمية لحساب SVD في التقارب؛
!>                إذا كان INFO = i، فإن i عنصر غير قطري في شكل ثنائي قطري وسيط
!>                لم يتقارب إلى الصفر.
!>
