subroutine dgrqts (integer m, integer p, integer n, double precision, dimension( lda, * ) a, double precision, dimension( lda, * ) af, double precision, dimension( lda, * ) q, double precision, dimension( lda, * ) r, integer lda, double precision, dimension( * ) taua, double precision, dimension( ldb, * ) b, double precision, dimension( ldb, * ) bf, double precision, dimension( ldb, * ) z, double precision, dimension( ldb, * ) t, double precision, dimension( ldb, * ) bwk, integer ldb, double precision, dimension( * ) taub, double precision, dimension( lwork ) work, integer lwork, double precision, dimension( * ) rwork, double precision, dimension( 4 ) result)

DGRQTS

الغرض:

!>
!> DGRQTS تختبر DGGRQF، التي تحسب تحليل GR لمصفوفة A بحجم M × N ومصفوفة B بحجم P × N: A = R*Q و B = Z*T*Q.
!>

المعلمات

M 
!>          M هو عدد صحيح (INTEGER)
!>          عدد صفوف المصفوفة A.  M >= 0.
!>

P

!>          P هو عدد صحيح (INTEGER)
!>          عدد صفوف المصفوفة B.  P >= 0.
!>

N

!>          N هو عدد صحيح (INTEGER)
!>          عدد أعمدة المصفوفتين A و B.  N >= 0.
!>

A

!>          A هي مصفوفة من النوع DOUBLE PRECISION، بأبعاد (LDA,N)
!>          المصفوفة A بحجم M × N.
!>

AF

!>          AF هي مصفوفة من النوع DOUBLE PRECISION، بأبعاد (LDA,N)
!>          تفاصيل تحليل GR للمصفوفتين A و B، كما تم إرجاعها
!>          بواسطة DGGRQF، انظر SGGRQF لمزيد من التفاصيل.
!>

Q

!>          Q هي مصفوفة من النوع DOUBLE PRECISION، بأبعاد (LDA,N)
!>          المصفوفة المتعامدة Q بحجم N × N.
!>

R

!>          R هي مصفوفة من النوع DOUBLE PRECISION، بأبعاد (LDA,MAX(M,N))
!>

LDA

!>          LDA هو عدد صحيح (INTEGER)
!>          البعد الرئيسي للمصفوفات A, AF, R و Q.
!>          LDA >= max(M,N).
!>

TAUA

!>          TAUA هي مصفوفة من النوع DOUBLE PRECISION، بأبعاد (min(M,N))
!>          العوامل القياسية للانعكاسات الأولية، كما تم إرجاعها
!>          بواسطة DGGQRC.
!>

B

!>          B هي مصفوفة من النوع DOUBLE PRECISION، بأبعاد (LDB,N)
!>          عند الإدخال، المصفوفة B بحجم P × N.
!>

BF

!>          BF هي مصفوفة من النوع DOUBLE PRECISION، بأبعاد (LDB,N)
!>          تفاصيل تحليل GQR للمصفوفتين A و B، كما تم إرجاعها
!>          بواسطة DGGRQF، انظر SGGRQF لمزيد من التفاصيل.
!>

Z

!>          Z هي مصفوفة من النوع DOUBLE PRECISION، بأبعاد (LDB,P)
!>          المصفوفة المتعامدة Z بحجم P × P.
!>

T

!>          T هي مصفوفة من النوع DOUBLE PRECISION، بأبعاد (LDB,max(P,N))
!>

BWK

!>          BWK هي مصفوفة من النوع DOUBLE PRECISION، بأبعاد (LDB,N)
!>

LDB

!>          LDB هو عدد صحيح (INTEGER)
!>          البعد الرئيسي للمصفوفات B, BF, Z و T.
!>          LDB >= max(P,N).
!>

TAUB

!>          TAUB هي مصفوفة من النوع DOUBLE PRECISION، بأبعاد (min(P,N))
!>          العوامل القياسية للانعكاسات الأولية، كما تم إرجاعها
!>          بواسطة DGGRQF.
!>

WORK

!>          WORK هي مصفوفة من النوع DOUBLE PRECISION، بأبعاد (LWORK)
!>

LWORK

!>          LWORK هو عدد صحيح (INTEGER)
!>          بعد المصفوفة WORK، LWORK >= max(M,P,N)**2.
!>

RWORK

!>          RWORK هي مصفوفة من النوع DOUBLE PRECISION، بأبعاد (M)
!>

RESULT

!>          RESULT هي مصفوفة من النوع DOUBLE PRECISION، بأبعاد (4)
!>          نسب الاختبار:
!>            RESULT(1) = norm( R - A*Q' ) / ( MAX(M,N)*norm(A)*ULP)
!>            RESULT(2) = norm( T*Q - Z'*B ) / (MAX(P,N)*norm(B)*ULP)
!>            RESULT(3) = norm( I - Q'*Q ) / ( N*ULP )
!>            RESULT(4) = norm( I - Z'*Z ) / ( P*ULP )
!>
