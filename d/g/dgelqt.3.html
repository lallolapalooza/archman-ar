توثيق الوظيفة/الإجراء الفرعي

subroutine dgelqt (integer m, integer n, integer mb, double precision, dimension( lda, * ) a, integer lda, double precision, dimension( ldt, * ) t, integer ldt, double precision, dimension( * ) work, integer info)

DGELQT

الغرض:

!>
!> DGELQT تحسب تحليل LQ مسدود لمصفوفة حقيقية M-by-N تسمى A
!> باستخدام تمثيل WY المدمج لـ Q.
!>

المعلمات

M 
!>          M هو عدد صحيح (INTEGER).
!>          عدد صفوف المصفوفة A.  M >= 0.
!>

N

!>          N هو عدد صحيح (INTEGER).
!>          عدد أعمدة المصفوفة A.  N >= 0.
!>

MB

!>          MB هو عدد صحيح (INTEGER).
!>          حجم الكتلة (block size) الذي سيتم استخدامه في التحليل QR المسدود.  MIN(M,N) >= MB >= 1.
!>

A

!>          A هي مصفوفة من نوع double precision، بأبعاد (LDA,N).
!>          عند الإدخال، المصفوفة A التي أبعادها M-by-N.
!>          عند الخروج، تحتوي العناصر على القطر الرئيسي وما تحته في المصفوفة على
!>          المصفوفة شبه المنحرفة السفلية L (بحجم M-by-MIN(M,N)) (L تكون مثلثية سفلية إذا كان M <= N)؛
!>          تحتوي العناصر فوق القطر الرئيسي على صفوف V.
!>

LDA

!>          LDA هو عدد صحيح (INTEGER).
!>          البعد الرائد للمصفوفة A.  LDA >= max(1,M).
!>

T

!>          T هي مصفوفة من نوع double precision، بأبعاد (LDT,MIN(M,N)).
!>          المعاملات الأساسية المثلثية العلوية المخزنة في شكل مدمج
!>          كسلسلة من الكتل المثلثية العلوية.  انظر أدناه
!>          لمزيد من التفاصيل.
!>

LDT

!>          LDT هو عدد صحيح (INTEGER).
!>          البعد الرائد للمصفوفة T.  LDT >= MB.
!>

WORK

!>          WORK هي مصفوفة من نوع double precision، بأبعاد (MB*N).
!>

INFO

!>          INFO هو عدد صحيح (INTEGER).
!>          = 0:  انتهاء ناجح
!>          < 0:  إذا كان INFO = -i، فإن الوسيط i كان له قيمة غير قانونية
!>

تفاصيل إضافية:

!>
!>  تخزن المصفوفة V المعاملات الأساسية H(i) في الصف i
!>  فوق القطر الرئيسي. على سبيل المثال، إذا كان M=5 و N=3، فإن المصفوفة V هي
!>
!>               V = (  1  v1 v1 v1 v1 )
!>                   (     1  v2 v2 v2 )
!>                   (         1 v3 v3 )
!>
!>
!>  حيث تمثل الـ vi's المتجهات التي تحدد H(i)، والتي يتم إرجاعها
!>  في المصفوفة A.  الواحدات على طول قطر V غير مخزنة في A.
!>  لتكن K=MIN(M,N).  عدد الكتل هو B = ceiling(K/MB)، حيث كل
!>  كتلة من الرتبة MB باستثناء الكتلة الأخيرة، والتي هي من الرتبة
!>  IB = K - (B-1)*MB.  لكل كتلة من الكتل B، يتم حساب معامل مثلثي علوي مدمج
!>  : T1، T2، ...، TB.  يتم تخزين T's بحجم MB-by-MB (و IB-by-IB
!>  للكتلة الأخيرة) في المصفوفة T بحجم MB-by-K كما يلي
!>
!>               T = (T1 T2 ... TB).
!>
