subroutine dgemqr (character side, character trans, integer m, integer n, integer k, double precision, dimension( lda, * ) a, integer lda, double precision, dimension( * ) t, integer tsize, double precision, dimension( ldc, * ) c, integer ldc, double precision, dimension( * ) work, integer lwork, integer info)

DGEMQR

الغرض:

!>
!> DGEMQR يستبدل المصفوفة العامة الحقيقية M-by-N المسماة C بـ
!>
!>                      SIDE = 'L'     SIDE = 'R'
!>      TRANS = 'N':      Q * C          C * Q
!>      TRANS = 'T':      Q**T * C       C * Q**T
!>
!> حيث Q مصفوفة متعامدة حقيقية معرفة كناتج
!> للمُنعكِسات الأولية المحظورة المحسوبة بواسطة تحليل QR للأشكال الطويلة والضيقة (DGEQR)
!>
!>

المعاملات

SIDE 
!>          SIDE هو CHARACTER*1
!>          = 'L': تطبيق Q أو Q**T من اليسار؛
!>          = 'R': تطبيق Q أو Q**T من اليمين.
!>

TRANS

!>          TRANS هو CHARACTER*1
!>          = 'N':  عدم التطبيق، تطبيق Q؛
!>          = 'T':  تطبيق منقول Q، تطبيق Q**T.
!>

M

!>          M هو INTEGER
!>          عدد صفوف المصفوفة A.  M >=0.
!>

N

!>          N هو INTEGER
!>          عدد أعمدة المصفوفة C. N >= 0.
!>

K

!>          K هو INTEGER
!>          عدد المُنعكِسات الأولية التي يُعرّف ناتجها
!>          المصفوفة Q.
!>          إذا كان SIDE = 'L'، فإن M >= K >= 0؛
!>          إذا كان SIDE = 'R'، فإن N >= K >= 0.
!>

A

!>          A هي مصفوفة DOUBLE PRECISION، بُعدها (LDA,K)
!>          جزء من هيكل البيانات لتمثيل Q كما تم إرجاعه بواسطة DGEQR.
!>

LDA

!>          LDA هو INTEGER
!>          البُعد الرائد للمصفوفة A.
!>          إذا كان SIDE = 'L'، فإن LDA >= max(1,M)؛
!>          إذا كان SIDE = 'R'، فإن LDA >= max(1,N).
!>

T

!>          T هي مصفوفة DOUBLE PRECISION، بُعدها (MAX(5,TSIZE)).
!>          جزء من هيكل البيانات لتمثيل Q كما تم إرجاعه بواسطة DGEQR.
!>

TSIZE

!>          TSIZE هو INTEGER
!>          بُعد المصفوفة T. TSIZE >= 5.
!>

C

!>          C هي مصفوفة DOUBLE PRECISION، بُعدها (LDC,N)
!>          عند الإدخال، المصفوفة C ذات الأبعاد M-by-N.
!>          عند الخروج، يتم استبدال C بـ Q*C أو Q**T*C أو C*Q**T أو C*Q.
!>

LDC

!>          LDC هو INTEGER
!>          البُعد الرائد للمصفوفة C. LDC >= max(1,M).
!>

WORK

!>          (مساحة عمل) مصفوفة DOUBLE PRECISION، بُعدها (MAX(1,LWORK))
!>          عند الخروج، إذا كان INFO = 0، فإن WORK(1) يُرجع الحد الأدنى LWORK.
!>

LWORK

!>          LWORK هو INTEGER
!>          بُعد المصفوفة WORK. LWORK >= 1.
!>          إذا كان LWORK = -1، فسيتم افتراض استعلام مساحة العمل. تقوم الروتين
!>          فقط بحساب حجم المصفوفة WORK، وترجع هذه القيمة كـ WORK(1)، ولن يصدر
!>          XERBLA أي رسالة خطأ متعلقة بـ WORK.
!>

INFO

!>          INFO هو INTEGER
!>          = 0:  انتهاء ناجح
!>          < 0:  إذا كان INFO = -i، فإن الوسيط i كان له قيمة غير قانونية
!>

!>
!> هذه التفاصيل خاصة بتطبيق LAPACK هذا. يجب على المستخدمين عدم
!> اعتبارها مسلمات. قد تتغير هذه التفاصيل في المستقبل، ومن غير المرجح
!> أن تكون صحيحة لتطبيق LAPACK آخر. هذه التفاصيل ذات صلة إذا أراد
!> شخص ما محاولة فهم الكود. وهي ليست جزءًا من الواجهة.
!>
!> في هذا الإصدار،
!>
!>          T(2): حجم كتلة الصف (MB)
!>          T(3): حجم كتلة العمود (NB)
!>          T(6:TSIZE): هيكل البيانات اللازم لـ Q، محسوب بواسطة
!>                           DLATSQR أو DGEQRT
!>
!>  اعتمادًا على أبعاد المصفوفة M و N، وأحجام كتل الصفوف والأعمدة MB و NB
!>  التي يتم إرجاعها بواسطة ILAENV، سيستخدم DGEQR إما
!>  DLATSQR (إذا كانت المصفوفة طويلة وضيقة) أو DGEQRT لحساب
!>  تحليل QR.
!>  سيستخدم هذا الإصدار من DGEMQR إما DLAMTSQR أو DGEMQRT لـ
!>  ضرب المصفوفة Q بمصفوفة أخرى.
!>  مزيد من التفاصيل في DLATMSQR أو DGEMQRT.
!>
!>
