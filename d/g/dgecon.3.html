توثيق الوظيفة/الروتين الفرعي

subroutine dgecon (character norm, integer n, double precision, dimension( lda, * ) a, integer lda, double precision anorm, double precision rcond, double precision, dimension( * ) work, integer, dimension( * ) iwork, integer info)

DGECON

الغرض:

!>
!> DGECON تقدّر مقلوب رقم الشرط لمصفوفة حقيقية عامة A، إما في المعيار 1-norm
!> أو المعيار اللانهائي (infinity-norm)، باستخدام التحليل LU الذي تم حسابه بواسطة
!> DGETRF.
!>
!> يتم الحصول على تقدير لـ norm(inv(A))، ويتم حساب مقلوب رقم الشرط على أنه
!>    RCOND = 1 / ( norm(A) * norm(inv(A)) ).
!>

المعلمات

NORM 
!>          NORM هو CHARACTER*1
!>          يحدد ما إذا كان رقم الشرط 1-norm أو رقم الشرط اللانهائي
!>          مطلوبًا:
!>          = '1' أو 'O':  1-norm؛
!>          = 'I':         Infinity-norm.
!>

N

!>          N هو INTEGER
!>          ترتيب المصفوفة A.  N >= 0.
!>

A

!>          A هو مصفوفة DOUBLE PRECISION، حجمها (LDA,N)
!>          العوامل L و U من تحليل A = P*L*U
!>          كما تم حسابها بواسطة DGETRF.
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرئيسي للمصفوفة A.  LDA >= max(1,N).
!>

ANORM

!>          ANORM هو DOUBLE PRECISION
!>          إذا كان NORM = '1' أو 'O'، فإن المعيار 1-norm للمصفوفة الأصلية A.
!>          إذا كان NORM = 'I'، فإن المعيار اللانهائي للمصفوفة الأصلية A.
!>

RCOND

!>          RCOND هو DOUBLE PRECISION
!>          مقلوب رقم الشرط للمصفوفة A،
!>          محسوبًا كـ RCOND = 1/(norm(A) * norm(inv(A))).
!>

WORK

!>          WORK هي مصفوفة DOUBLE PRECISION، حجمها (4*N)
!>

IWORK

!>          IWORK هي مصفوفة INTEGER، حجمها (N)
!>

INFO

!>          INFO هو INTEGER
!>          = 0:  انتهاء ناجح
!>          < 0:  إذا كان INFO = -i، فإن المعلمة i كانت لها قيمة غير قانونية.
!>                قيم NaN غير قانونية لـ ANORM، وتنتشر إلى
!>                المعلمة الناتجة RCOND.
!>                اللانهاية غير قانونية لـ ANORM، وتنتشر إلى المخرج
!>                المعلمة RCOND كـ 0.
!>          = 1:  إذا كان RCOND = NaN، أو
!>                   RCOND = Inf، أو
!>                   المعيار المحسوب لمقلوب A هو 0.
!>                في الحالة الأخيرة، يتم إرجاع RCOND = 0.
!>
