subroutine dgeqrf (integer m, integer n, double precision, dimension( lda, * ) a, integer lda, double precision, dimension( * ) tau, double precision, dimension( * ) work, integer lwork, integer info)

DGEQRF

الغرض:

!>
!> DGEQRF تحسب تحليل QR لمصفوفة حقيقية M-by-N A:
!>
!>    A = Q * ( R ),
!>            ( 0 )
!>
!> حيث:
!>
!>    Q هي مصفوفة متعامدة M-by-M؛
!>    R هي مصفوفة شبه علوية N-by-N؛
!>    0 هي مصفوفة صفرية (M-N)-by-N، إذا كان M > N.
!>
!>

المعلمات

M 
!>          M هو عدد صحيح (INTEGER)
!>          عدد صفوف المصفوفة A.  M >= 0.
!>

N

!>          N هو عدد صحيح (INTEGER)
!>          عدد أعمدة المصفوفة A.  N >= 0.
!>

A

!>          A هي مصفوفة ذات دقة مزدوجة (DOUBLE PRECISION)، الأبعاد (LDA,N)
!>          عند الإدخال، المصفوفة M-by-N A.
!>          عند الخروج، تحتوي العناصر على ومن فوق القطر الرئيسي للمصفوفة
!>          على المصفوفة شبه العلوية R ذات الأبعاد min(M,N)-by-N (R
!>          هي شبه علوية إذا كان m >= n)؛ العناصر أسفل القطر الرئيسي،
!>          مع المصفوفة TAU، تمثل المصفوفة المتعامدة Q كـ
!>          ناتج ضرب min(m,n) من المستويات الأولية (انظر التفاصيل الإضافية).
!>

LDA

!>          LDA هو عدد صحيح (INTEGER)
!>          البعد القيادي للمصفوفة A.  LDA >= max(1,M).
!>

TAU

!>          TAU هي مصفوفة ذات دقة مزدوجة (DOUBLE PRECISION)، الأبعاد (min(M,N))
!>          العوامل القياسية للمستويات الأولية (انظر التفاصيل الإضافية).
!>

WORK

!>          WORK هي مصفوفة ذات دقة مزدوجة (DOUBLE PRECISION)، الأبعاد (MAX(1,LWORK))
!>          عند الخروج، إذا كان INFO = 0، فإن WORK(1) يعيد قيمة LWORK المثلى.
!>

LWORK

!>          LWORK هو عدد صحيح (INTEGER)
!>          بعد المصفوفة WORK.
!>          LWORK >= 1، إذا كان MIN(M,N) = 0، و LWORK >= N، بخلاف ذلك.
!>          للأداء الأمثل LWORK >= N*NB، حيث NB
!>          هو حجم القطعة الأمثل.
!>
!>          إذا كان LWORK = -1، فسيتم افتراض استعلام مساحة العمل؛ تقوم الروتين
!>          بحساب حجم مصفوفة WORK الأمثل فقط، وإعادة
!>          هذه القيمة كأول إدخال في مصفوفة WORK، ولا يتم إصدار أي رسالة خطأ
!>          متعلقة بـ LWORK بواسطة XERBLA.
!>

INFO

!>          INFO هو عدد صحيح (INTEGER)
!>          = 0:  انتهاء ناجح
!>          < 0:  إذا كان INFO = -i، فإن الوسيط i كان له قيمة غير قانونية
!>

تفاصيل إضافية:

!>
!>  يتم تمثيل المصفوفة Q كناتج ضرب مستويات أولية
!>
!>     Q = H(1) H(2) . . . H(k)، حيث k = min(m,n).
!>
!>  كل H(i) له الشكل
!>
!>     H(i) = I - tau * v * v**T
!>
!>  حيث tau هو عامل قياسي حقيقي، و v هو متجه حقيقي مع
!>  v(1:i-1) = 0 و v(i) = 1؛ يتم تخزين v(i+1:m) عند الخروج في A(i+1:m,i)،
!>  و tau في TAU(i).
!>
