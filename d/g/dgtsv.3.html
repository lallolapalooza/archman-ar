subroutine dgtsv (integer n, integer nrhs, double precision, dimension( * ) dl, double precision, dimension( * ) d, double precision, dimension( * ) du, double precision, dimension( ldb, * ) b, integer ldb, integer info)

DGTSV تحسب حل نظام المعادلات الخطية A * X = B لمصفوفات GT

الغرض:

!>
!> DGTSV  يحل المعادلة
!>
!>    A*X = B,
!>
!> حيث A مصفوفة ثلاثية قطرية (tridiagonal) بحجم n × n، عن طريق الحذف الغاوسي مع
!> المحورية الجزئية (partial pivoting).
!>
!> لاحظ أنه يمكن حل المعادلة  A**T*X = B  عن طريق تبديل ترتيب
!> الوسائط DU و DL.
!>

المعاملات

N 
!>          N هو عدد صحيح (INTEGER)
!>          رتبة المصفوفة A.  N >= 0.
!>

NRHS

!>          NRHS هو عدد صحيح (INTEGER)
!>          عدد الأطراف اليمنى (right hand sides)، أي عدد أعمدة
!>          المصفوفة B.  NRHS >= 0.
!>

DL

!>          DL هي مصفوفة ذات دقة مزدوجة (DOUBLE PRECISION array)، البعد (N-1)
!>          عند الإدخال، يجب أن تحتوي DL على العناصر الفرعية السفلية (sub-diagonal)
!>          البالغ عددها (n-1) للمصفوفة A.
!>
!>          عند الخروج، يتم الكتابة فوق DL بالعناصر (n-2) من
!>          المصفوفة المثلثية العلوية U الناتجة عن التحليل LU للمصفوفة A،
!>          في DL(1), ..., DL(n-2).
!>

D

!>          D هي مصفوفة ذات دقة مزدوجة (DOUBLE PRECISION array)، البعد (N)
!>          عند الإدخال، يجب أن تحتوي D على العناصر القطرية (diagonal) للمصفوفة A.
!>
!>          عند الخروج، يتم الكتابة فوق D بالعناصر القطرية البالغ عددها n للمصفوفة U.
!>

DU

!>          DU هي مصفوفة ذات دقة مزدوجة (DOUBLE PRECISION array)، البعد (N-1)
!>          عند الإدخال، يجب أن تحتوي DU على العناصر الفوق قطرية (super-diagonal)
!>          البالغ عددها (n-1) للمصفوفة A.
!>
!>          عند الخروج، يتم الكتابة فوق DU بالعناصر البالغ عددها (n-1) من
!>          المصفوفة الفوق قطرية الأولى للمصفوفة U.
!>

B

!>          B هي مصفوفة ذات دقة مزدوجة (DOUBLE PRECISION array)، البعد (LDB,NRHS)
!>          عند الإدخال، مصفوفة الطرف الأيمن B ذات الحجم N × NRHS.
!>          عند الخروج، إذا كان INFO = 0، فإنها تحتوي على مصفوفة الحل X ذات الحجم N × NRHS.
!>

LDB

!>          LDB هو عدد صحيح (INTEGER)
!>          البعد الرئيسي (leading dimension) للمصفوفة B.  LDB >= max(1,N).
!>

INFO

!>          INFO هو عدد صحيح (INTEGER)
!>          = 0: خروج ناجح
!>          < 0: إذا كان INFO = -i، فإن المعامل i-th كان له قيمة غير قانونية
!>          > 0: إذا كان INFO = i، فإن U(i,i) تساوي صفر تمامًا، ولم يتم حساب الحل.
!>               لم تكتمل عملية التحليل إلا إذا كان i = N.
!>
