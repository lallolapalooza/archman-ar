وثائق الوظيفة/الروتين الفرعي

subroutine dgbt02 (character trans, integer m, integer n, integer kl, integer ku, integer nrhs, double precision, dimension( lda, * ) a, integer lda, double precision, dimension( ldx, * ) x, integer ldx, double precision, dimension( ldb, * ) b, integer ldb, double precision, dimension( * ) rwork, double precision resid)

DGBT02

الغرض:

!>
!> DGBT02 يحسب الباقي لحل نظام المعادلات النطاقي op(A)*X = B:
!>    RESID = norm(B - op(A)*X) / ( norm(op(A)) * norm(X) * EPS ),
!> حيث op(A) = A أو A**T، اعتمادًا على TRANS، و EPS هو
!> الأبسلون الآلي.
!> المعيار المستخدم هو المعيار 1.
!>

المعلمات

TRANS 
!>          TRANS هو CHARACTER*1
!>          يحدد شكل نظام المعادلات:
!>          = 'N':  A    * X = B  (بدون تبديل)
!>          = 'T':  A**T * X = B  (تبديل)
!>          = 'C':  A**H * X = B  (تبديل مترافق = تبديل)
!>

M

!>          M هو INTEGER
!>          عدد صفوف المصفوفة A.  M >= 0.
!>

N

!>          N هو INTEGER
!>          عدد أعمدة المصفوفة A.  N >= 0.
!>

KL

!>          KL هو INTEGER
!>          عدد الأقطار الفرعية داخل نطاق A.  KL >= 0.
!>

KU

!>          KU هو INTEGER
!>          عدد الأقطار الفائقة داخل نطاق A.  KU >= 0.
!>

NRHS

!>          NRHS هو INTEGER
!>          عدد أعمدة B.  NRHS >= 0.
!>

A

!>          A هو مصفوفة DOUBLE PRECISION، أبعاد (LDA,N)
!>          المصفوفة الأصلية A بتخزين نطاقي، مخزنة في الصفوف من 1 إلى
!>          KL+KU+1.
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرئيسي للمصفوفة A.  LDA >= max(1,KL+KU+1).
!>

X

!>          X هي مصفوفة DOUBLE PRECISION، أبعاد (LDX,NRHS)
!>          حلول المتجهات المحسوبة لنظام المعادلات الخطية.
!>

LDX

!>          LDX هو INTEGER
!>          البعد الرئيسي للمصفوفة X.  إذا كان TRANS = 'N'،
!>          LDX >= max(1,N)؛ إذا كان TRANS = 'T' أو 'C'، LDX >= max(1,M).
!>

B

!>          B هي مصفوفة DOUBLE PRECISION، أبعاد (LDB,NRHS)
!>          عند الدخول، المتجهات الطرفية اليمنى لنظام المعادلات الخطية.
!>          عند الخروج، يتم استبدال B بالفرق B - A*X.
!>

LDB

!>          LDB هو INTEGER
!>          البعد الرئيسي للمصفوفة B.  إذا كان TRANS = 'N'،
!>          LDB >= max(1,M)؛ إذا كان TRANS = 'T' أو 'C'، LDB >= max(1,N).
!>

RWORK

!>          RWORK هي مصفوفة DOUBLE PRECISION، أبعاد (MAX(1,LRWORK))،
!>          حيث LRWORK >= M إذا كان TRANS = 'T' أو 'C'؛ بخلاف ذلك، لا يتم
!>          الرجوع إلى RWORK.
!>

RESID

!>          RESID هو DOUBLE PRECISION
!>          الحد الأقصى لعدد الأطراف اليمنى لـ
!>          norm(B - op(A)*X) / ( norm(op(A)) * norm(X) * EPS ).
!>
