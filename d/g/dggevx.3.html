subroutine dggevx (character balanc, character jobvl, character jobvr, character sense, integer n, double precision, dimension( lda, * ) a, integer lda, double precision, dimension( ldb, * ) b, integer ldb, double precision, dimension( * ) alphar, double precision, dimension( * ) alphai, double precision, dimension( * ) beta, double precision, dimension( ldvl, * ) vl, integer ldvl, double precision, dimension( ldvr, * ) vr, integer ldvr, integer ilo, integer ihi, double precision, dimension( * ) lscale, double precision, dimension( * ) rscale, double precision abnrm, double precision bbnrm, double precision, dimension( * ) rconde, double precision, dimension( * ) rcondv, double precision, dimension( * ) work, integer lwork, integer, dimension( * ) iwork, logical, dimension( * ) bwork, integer info)

DGGEVX تحسب القيم الذاتية، واختياريًا، المتجهات الذاتية اليسرى و/أو اليمنى لمصفوفات GE

الغرض:

!>
!> DGGEVX تحسب للزوج من المصفوفات الحقيقية غير المتماثلة (A,B) بحجم N*N
!> القيم الذاتية المعممة، واختياريًا، المتجهات الذاتية المعممة اليسرى و/أو اليمنى.
!>
!> اختياريًا، تقوم أيضًا بحساب تحويل موازنة لتحسين
!> شروط القيم الذاتية والمتجهات الذاتية (ILO, IHI,
!> LSCALE, RSCALE, ABNRM, و BBNRM)، أرقام الشروط المقلوبة لـ
!> القيم الذاتية (RCONDE)، وأرقام الشروط المقلوبة لـ
!> المتجهات الذاتية اليمنى (RCONDV).
!>
!> القيمة الذاتية المعممة لزوج المصفوفات (A,B) هي عدد ثابت
!> lambda أو نسبة alpha/beta = lambda، بحيث تكون A - lambda*B
!> منفردة (singular). يتم تمثيلها عادةً بالزوج (alpha,beta)،
!> حيث يوجد تفسير معقول لـ beta=0، وحتى عندما يكون كلاهما صفراً.
!>
!> المتجه الذاتي الأيمن v(j) المقابل للقيمة الذاتية lambda(j)
!> لـ (A,B) يحقق
!>
!>                  A * v(j) = lambda(j) * B * v(j) .
!>
!> المتجه الذاتي الأيسر u(j) المقابل للقيمة الذاتية lambda(j)
!> لـ (A,B) يحقق
!>
!>                  u(j)**H * A  = lambda(j) * u(j)**H * B.
!>
!> حيث u(j)**H هو المرافق المتجهي (conjugate-transpose) لـ u(j).
!>
!>

المعلمات:

BALANC 
!>          BALANC هو CHARACTER*1
!>          يحدد خيار الموازنة المطلوب تنفيذه.
!>          = 'N':  لا تقم بالقياس القطري أو التبديل؛
!>          = 'P':  قم بالتبديل فقط؛
!>          = 'S':  قم بالقياس فقط؛
!>          = 'B':  قم بالتبديل والقياس معًا.
!>          سيتم حساب أرقام الشروط المقلوبة للمصفوفات
!>          بعد التبديل و/أو الموازنة. التبديل
!>          لا يغير أرقام الشروط (في الحساب الدقيق)،
!>          لكن الموازنة تغيرها.
!>

JOBVL

!>          JOBVL هو CHARACTER*1
!>          = 'N':  لا تحسب المتجهات الذاتية المعممة اليسرى؛
!>          = 'V':  احسب المتجهات الذاتية المعممة اليسرى.
!>

JOBVR

!>          JOBVR هو CHARACTER*1
!>          = 'N':  لا تحسب المتجهات الذاتية المعممة اليمنى؛
!>          = 'V':  احسب المتجهات الذاتية المعممة اليمنى.
!>

SENSE

!>          SENSE هو CHARACTER*1
!>          يحدد أي أرقام شروط مقلوبة يتم حسابها.
!>          = 'N':  لا يتم حساب أي منها؛
!>          = 'E':  يتم حسابها للقيم الذاتية فقط؛
!>          = 'V':  يتم حسابها للمتجهات الذاتية فقط؛
!>          = 'B':  يتم حسابها للقيم الذاتية والمتجهات الذاتية.
!>

N

!>          N هو INTEGER
!>          رتبة المصفوفات A, B, VL, و VR.  N >= 0.
!>

A

!>          A هي مصفوفة DOUBLE PRECISION، الأبعاد (LDA, N)
!>          عند الدخول، المصفوفة A في الزوج (A,B).
!>          عند الخروج، تم استبدال A. إذا كان JOBVL='V' أو JOBVR='V'
!>          أو كلاهما، فإن A تحتوي على الجزء الأول من شكل Schur الحقيقي
!>          لإصدارات المصفوفات A و B المدخلة.
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرئيسي لـ A.  LDA >= max(1,N).
!>

B

!>          B هي مصفوفة DOUBLE PRECISION، الأبعاد (LDB, N)
!>          عند الدخول، المصفوفة B في الزوج (A,B).
!>          عند الخروج، تم استبدال B. إذا كان JOBVL='V' أو JOBVR='V'
!>          أو كلاهما، فإن B تحتوي على الجزء الثاني من شكل Schur الحقيقي
!>          لإصدارات المصفوفات A و B المدخلة.
!>

LDB

!>          LDB هو INTEGER
!>          البعد الرئيسي لـ B.  LDB >= max(1,N).
!>

ALPHAR

!>          ALPHAR هي مصفوفة DOUBLE PRECISION، الأبعاد (N)
!>

ALPHAI

!>          ALPHAI هي مصفوفة DOUBLE PRECISION، الأبعاد (N)
!>

BETA

!>          BETA هي مصفوفة DOUBLE PRECISION، الأبعاد (N)
!>          عند الخروج، (ALPHAR(j) + ALPHAI(j)*i)/BETA(j)، j=1,...,N، ستكون
!>          القيم الذاتية المعممة. إذا كانت ALPHAI(j) صفرًا، فإن
!>          القيمة الذاتية j حقيقية؛ إذا كانت موجبة، فإن القيمتين الذاتيتين j
!>          و (j+1) هما زوجان مترافقان مركبان، مع
!>          ALPHAI(j+1) سالبة.
!>
!>          ملاحظة: قد تتجاوز النسب ALPHAR(j)/BETA(j) و ALPHAI(j)/BETA(j)
!>          أو تقل بشكل كبير، وقد تكون BETA(j) صفراً.
!>          لذلك، يجب على المستخدم تجنب حساب النسبة
!>          ALPHA/BETA بشكل ساذج. ومع ذلك، ستكون ALPHAR و ALPHAI دائمًا أقل
!>          من، وعادة ما تكون قابلة للمقارنة مع norm(A) في المقدار،
!>          و BETA دائمًا أقل من، وعادة ما تكون قابلة للمقارنة مع norm(B).
!>

VL

!>          VL هي مصفوفة DOUBLE PRECISION، الأبعاد (LDVL,N)
!>          إذا كان JOBVL = 'V'، يتم تخزين المتجهات الذاتية اليسرى u(j)
!>          واحدًا تلو الآخر في أعمدة VL، بنفس ترتيب
!>          قيمها الذاتية. إذا كانت القيمة الذاتية j حقيقية، فإن
!>          u(j) = VL(:,j)، العمود j من VL. إذا كانت القيمتان الذاتيتان j و
!>          (j+1) تشكلان زوجًا مترافقًا مركبًا، فإن
!>          u(j) = VL(:,j)+i*VL(:,j+1) و u(j+1) = VL(:,j)-i*VL(:,j+1).
!>          سيتم قياس كل متجه ذاتي بحيث يكون المكون الأكبر
!>          لديه abs(الجزء الحقيقي) + abs(الجزء التخيلي) = 1.
!>          غير مستخدم إذا كان JOBVL = 'N'.
!>

LDVL

!>          LDVL هو INTEGER
!>          البعد الرئيسي للمصفوفة VL. LDVL >= 1، وإذا
!>          كان JOBVL = 'V'، فإن LDVL >= N.
!>

VR

!>          VR هي مصفوفة DOUBLE PRECISION، الأبعاد (LDVR,N)
!>          إذا كان JOBVR = 'V'، يتم تخزين المتجهات الذاتية اليمنى v(j)
!>          واحدًا تلو الآخر في أعمدة VR، بنفس ترتيب
!>          قيمها الذاتية. إذا كانت القيمة الذاتية j حقيقية، فإن
!>
