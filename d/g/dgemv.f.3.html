توثيق الوظيفة/الروتين الفرعي

subroutine dgemv (character trans, integer m, integer n, double precision alpha, double precision, dimension(lda,*) a, integer lda, double precision, dimension(*) x, integer incx, double precision beta, double precision, dimension(*) y, integer incy)

DGEMV

الغرض:

!>
!> DGEMV  تقوم بإحدى عمليات المصفوفة والمتجه التالية:
!>
!>    y := alpha*A*x + beta*y,   أو   y := alpha*A**T*x + beta*y,
!>
!> حيث alpha و beta هما عددان قياسيان، x و y هما متجهان، و A مصفوفة
!> بأبعاد m × n.
!>

المعلمات

TRANS 
!>          TRANS هو CHARACTER*1
!>           عند الإدخال، تحدد TRANS العملية التي سيتم تنفيذها كالتالي:
!>
!>              TRANS = 'N' أو 'n'   y := alpha*A*x + beta*y.
!>
!>              TRANS = 'T' أو 't'   y := alpha*A**T*x + beta*y.
!>
!>              TRANS = 'C' أو 'c'   y := alpha*A**T*x + beta*y.
!>

M

!>          M هو INTEGER
!>           عند الإدخال، تحدد M عدد صفوف المصفوفة A.
!>           يجب أن تكون M أكبر من أو تساوي الصفر.
!>

N

!>          N هو INTEGER
!>           عند الإدخال، تحدد N عدد أعمدة المصفوفة A.
!>           يجب أن تكون N أكبر من أو تساوي الصفر.
!>

ALPHA

!>          ALPHA هو DOUBLE PRECISION.
!>           عند الإدخال، تحدد ALPHA العدد القياسي alpha.
!>

A

!>          A هو مصفوفة DOUBLE PRECISION، بأبعاد ( LDA, N )
!>           قبل الإدخال، يجب أن يحتوي الجزء الرائد بأبعاد m × n من المصفوفة A
!>           على معاملات المصفوفة.
!>

LDA

!>          LDA هو INTEGER
!>           عند الإدخال، تحدد LDA البعد الأول للمصفوفة A كما تم تعريفه
!>           في البرنامج (الفرعي) المستدعي. يجب أن تكون LDA أكبر من أو
!>           تساوي 1 أو m.
!>

X

!>          X هو مصفوفة DOUBLE PRECISION، بأبعاد على الأقل
!>           ( 1 + ( n - 1 )*abs( INCX ) ) عندما TRANS = 'N' أو 'n'
!>           وعلى الأقل
!>           ( 1 + ( m - 1 )*abs( INCX ) ) خلاف ذلك.
!>           قبل الإدخال، يجب أن تحتوي المصفوفة X ذات الزيادات على
!>           المتجه x.
!>

INCX

!>          INCX هو INTEGER
!>           عند الإدخال، يحدد INCX الزيادة لعناصر X. يجب ألا يكون INCX صفرًا.
!>

BETA

!>          BETA هو DOUBLE PRECISION.
!>           عند الإدخال، يحدد BETA العدد القياسي beta. عندما يتم
!>           توفير BETA كصفر، لا يلزم تعيين Y عند الإدخال.
!>

Y

!>          Y هو مصفوفة DOUBLE PRECISION، بأبعاد على الأقل
!>           ( 1 + ( m - 1 )*abs( INCY ) ) عندما TRANS = 'N' أو 'n'
!>           وعلى الأقل
!>           ( 1 + ( n - 1 )*abs( INCY ) ) خلاف ذلك.
!>           قبل الإدخال عندما يكون BETA غير صفري، يجب أن تحتوي المصفوفة Y
!>           ذات الزيادات على المتجه y. عند الخروج، يتم استبدال Y
!>           بالمتجه y المحدث.
!>           إذا كانت m أو n صفرًا، فلا يتم الرجوع إلى Y وتنفذ الوظيفة
!>           عودة سريعة.
!>

INCY

!>          INCY هو INTEGER
!>           عند الإدخال، يحدد INCY الزيادة لعناصر Y. يجب ألا يكون INCY صفرًا.
!>
