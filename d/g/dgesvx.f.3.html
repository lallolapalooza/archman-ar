subroutine dgesvx (character fact, character trans, integer n, integer nrhs, double precision, dimension( lda, * ) a, integer lda, double precision, dimension( ldaf, * ) af, integer ldaf, integer, dimension( * ) ipiv, character equed, double precision, dimension( * ) r, double precision, dimension( * ) c, double precision, dimension( ldb, * ) b, integer ldb, double precision, dimension( ldx, * ) x, integer ldx, double precision rcond, double precision, dimension( * ) ferr, double precision, dimension( * ) berr, double precision, dimension( * ) work, integer, dimension( * ) iwork, integer info)

DGESVX تحسب حل نظام المعادلات الخطية A * X = B لمصفوفات GE

الغرض:

!>
!> تستخدم DGESVX تحليل LU لحساب حل نظام المعادلات الخطية الحقيقي
!>    A * X = B,
!> حيث A مصفوفة N*N و X و B مصفوفتان N*NRHS.
!>
!> كما يتم توفير حدود الخطأ على الحل وتقدير للحالة.
!>

الوصف:

!>
!> يتم تنفيذ الخطوات التالية:
!>
!> 1. إذا كان FACT = 'E'، يتم حساب عوامل القياس الحقيقية لتحقيق توازن
!>    النظام:
!>       TRANS = 'N':  diag(R)*A*diag(C)     *inv(diag(C))*X = diag(R)*B
!>       TRANS = 'T': (diag(R)*A*diag(C))**T *inv(diag(R))*X = diag(C)*B
!>       TRANS = 'C': (diag(R)*A*diag(C))**H *inv(diag(R))*X = diag(C)*B
!>    يعتمد ما إذا كان سيتم تحقيق توازن النظام أم لا على
!>    قياس المصفوفة A، ولكن إذا تم استخدام تحقيق التوازن، يتم استبدال A بـ
!>    diag(R)*A*diag(C) و B بـ diag(R)*B (إذا كان TRANS='N')
!>    أو diag(C)*B (إذا كان TRANS = 'T' أو 'C').
!>
!> 2. إذا كان FACT = 'N' أو 'E'، يتم استخدام تحليل LU لتحليل
!>    المصفوفة A (بعد تحقيق التوازن إذا كان FACT = 'E') على النحو التالي
!>       A = P * L * U,
!>    حيث P مصفوفة تبديل، L مصفوفة مثلثية سفلية وحدة، و U مصفوفة مثلثية عليا.
!>
!> 3. إذا كان بعض U(i,i)=0، بحيث تكون U مفردة تمامًا، فإن الوحدة
!>    تُرجع مع INFO = i. بخلاف ذلك، يتم استخدام الشكل المحلل لـ A لتقدير
!>    رقم حالة المصفوفة A. إذا كان
!>    معكوس رقم الحالة أقل من دقة الآلة،
!>    يتم إرجاع INFO = N+1 كتحذير، ولكن الوحدة لا تزال تستمر في
!>    حل X وحساب حدود الخطأ كما هو موضح أدناه.
!>
!> 4. يتم حل نظام المعادلات لـ X باستخدام الشكل المحلل لـ A.
!>
!> 5. يتم تطبيق تحسين تكراري لتحسين الحل المحسوب
!>    وتحسين وحساب حدود الخطأ والأخطاء الخلفية له.
!>
!> 6. إذا تم استخدام تحقيق التوازن، يتم ضرب المصفوفة X مسبقًا بـ
!>    diag(C) (إذا كان TRANS = 'N') أو diag(R) (إذا كان TRANS = 'T' أو 'C') بحيث
!>    تحل النظام الأصلي قبل تحقيق التوازن.
!>

المعلمات

FACT 
!>          FACT is CHARACTER*1
!>          يحدد ما إذا كان الشكل المحلل للمصفوفة A مزودًا عند الإدخال، وإذا لم يكن كذلك،
!>          هل يجب تحقيق توازن المصفوفة A قبل تحليلها.
!>          = 'F':  عند الإدخال، تحتوي AF و IPIV على الشكل المحلل لـ A.
!>                  إذا كان EQUED ليس 'N'، فقد تم تحقيق توازن المصفوفة A
!>                  بعوامل القياس المعطاة بواسطة R و C.
!>                  لا يتم تعديل A و AF و IPIV.
!>          = 'N':  سيتم نسخ المصفوفة A إلى AF وتحليلها.
!>          = 'E':  سيتم تحقيق توازن المصفوفة A إذا لزم الأمر، ثم
!>                  نسخها إلى AF وتحليلها.
!>

TRANS

!>          TRANS is CHARACTER*1
!>          يحدد شكل نظام المعادلات:
!>          = 'N':  A * X = B     (بدون تبديل)
!>          = 'T':  A**T * X = B  (تبديل)
!>          = 'C':  A**H * X = B  (تبديل)
!>

N

!>          N is INTEGER
!>          عدد المعادلات الخطية، أي ترتيب
!>          المصفوفة A.  N >= 0.
!>

NRHS

!>          NRHS is INTEGER
!>          عدد الأطراف اليمنى، أي عدد الأعمدة
!>          للمصفوفتين B و X.  NRHS >= 0.
!>

A

!>          A is DOUBLE PRECISION array, dimension (LDA,N)
!>          عند الإدخال، المصفوفة N*N A.  إذا كان FACT = 'F' و EQUED
!>          ليس 'N'، فيجب أن تكون A قد حققت توازنًا بواسطة عوامل القياس
!>          في R و/أو C.  لا يتم تعديل A إذا كان FACT = 'F' أو
!>          'N'، أو إذا كان FACT = 'E' و EQUED = 'N' عند الخروج.
!>
!>          عند الخروج، إذا كان EQUED .ne. 'N'، يتم قياس A على النحو التالي:
!>          EQUED = 'R':  A := diag(R) * A
!>          EQUED = 'C':  A := A * diag(C)
!>          EQUED = 'B':  A := diag(R) * A * diag(C).
!>

LDA

!>          LDA is INTEGER
!>          البعد الرئيسي للمصفوفة A.  LDA >= max(1,N).
!>

AF

!>          AF is DOUBLE PRECISION array, dimension (LDAF,N)
!>          إذا كان FACT = 'F'، فإن AF هو معامل إدخال ويحتوي عند الإدخال
!>          على العوامل L و U من التحليل
!>          A = P*L*U كما تم حسابه بواسطة DGETRF. إذا كان EQUED .ne. 'N'، فإن
!>          AF هو الشكل المحلل للمصفوفة المتوازنة A.
!>
!>          إذا كان FACT = 'N'، فإن AF هو معامل إخراج ويحتوي عند الخروج
!>          على العوامل L و U من تحليل A = P*L*U
!>          للمصفوفة الأصلية A.
!>
!>          إذا كان FACT = 'E'، فإن AF هو معامل إخراج ويحتوي عند الخروج
!>          على العوامل L و U من تحليل A = P*L*U
!>          للمصفوفة المتوازنة A (انظر وصف A لشكل المصفوفة المتوازنة).
!>

LDAF

!>          LDAF is INTEGER
!>          البعد الرئيسي للمصفوفة AF.  LDAF >= max(1,N).
!>

IPIV

!>          IPIV is INTEGER array, dimension (N)
!>          إذا كان FACT = 'F'، فإن IPIV هو معامل إدخال ويحتوي عند الإدخال
!>          على مؤشرات المحور من تحليل A = P*L*U
!>          كما تم حسابه بواسطة DGETRF؛ تم تبديل الصف i من المصفوفة
!>          مع الصف IPIV(i).
!>
!>          إذا كان FACT = 'N'، فإن IPIV هو معامل إخراج ويحتوي عند الخروج
!>          على مؤشرات المحور من تحليل A = P*L*U
!>          للمصفوفة الأصلية A.
!>
!>          إذا كان FACT = 'E'، فإن IPIV هو معامل إخراج ويحتوي عند الخروج
!>          على مؤشرات المحور من تحليل A = P*L*U
!>          للمصفوفة المتوازنة A.
!>

EQUED

!>          EQUED is CHARACTER*1
!>          يحدد شكل تحقيق التوازن الذي تم إجراؤه.
!>          = 'N':  لا يوجد تحقيق توازن (دائما صحيح إذا كان FACT = 'N').
!>          = 'R':  تحقيق توازن الصف، أي تم ضرب A مسبقًا بـ
!>                  diag(R).
!>          = 'C':  تحقيق توازن العمود، أي تم ضرب A لاحقًا بـ
!>                  diag(C).
!>          = 'B':  تحقيق توازن الصف والعمود، أي تم استبدال A بـ
!>                  diag(R) * A * diag(C).
!>          EQUED هو معامل إدخال إذا كان FACT = 'F'؛ بخلاف ذلك، فهو معامل
!>          إخراج.
!>

R

!>          R is DOUBLE PRECISION array, dimension (N)
!>          إذا كان EQUED = 'R' أو 'B'، فإن R يحتوي على عوامل قياس الصف
!>          التي تستخدمها الوحدة لتحقيق توازن المصفوفة A.
!>          إذا كان EQUED = 'N' أو 'C'، فإن R غير مستخدم.
!>

C

!>          C is DOUBLE PRECISION array, dimension (N)
!>          إذا كان EQUED = 'C' أو 'B'، فإن C يحتوي على عوامل قياس العمود
!>          التي تستخدمها الوحدة لتحقيق توازن المصفوفة A.
!>          إذا كان EQUED = 'N' أو 'R'، فإن C غير مستخدم.
!>

B

!>          B is DOUBLE PRECISION array, dimension (LDB,NRHS)
!>          عند الإدخال، يحتوي على الطرف الأيمن M*NRHS لمصفوفة B.
!>
!>          عند الخروج، إذا كان EQUED = 'R' أو 'B'، فإن B يتم قياسه
!>          على النحو التالي:
!>             B := diag(R) * B.
!>

LDB

!>          LDB is INTEGER
!>          البعد الرئيسي للمصفوفة B.  LDB >= max(1,N).
!>

X

!>          X is DOUBLE PRECISION array, dimension (LDX,NRHS)
!>          عند الخروج، يحتوي على حل النظام A * X = B.
!>          إذا كان EQUED = 'N' أو 'R'، يتم استعادة X بحيث يحل النظام الأصلي.
!>          إذا كان EQUED = 'C' أو 'B'، يتم قياس X على النحو التالي:
!>             X := diag(C) * X.
!>

LDX

!>          LDX is INTEGER
!>          البعد الرئيسي للمصفوفة X.  LDX >= max(1,N).
!>

RCOND

!>          RCOND is DOUBLE PRECISION
!>          عند الخروج، يقدر معكوس رقم حالة المصفوفة A.
!>

FERR

!>          FERR is DOUBLE PRECISION array, dimension (NRHS)
!>          عند الخروج، يحتوي على تقديرات للخطأ الأمامي لكل عمود من أعمدة الحل X.
!>          FERR(j) هو تقدير للخطأ الأمامي النسبي المعطى بـ
!>          max( |A X(j) - B(j)|_1 / ( |A|*|X(j)|_1 + |B(j)|_1 ) ).
!>

BERR

!>          BERR is DOUBLE PRECISION array, dimension (NRHS)
!>          عند الخروج، يحتوي على تقديرات للخطأ الخلفي لكل عمود من أعمدة الحل X.
!>          BERR(j) هو تقدير للخطأ الخلفي النسبي المعطى بـ
!>          max( |A X(j) - B(j)|_1 / ( |A|*|X(j)|_1 ) ).
!>

WORK

!>          WORK is DOUBLE PRECISION array, dimension (max(2*N,max(1,2*N-1)))
!>          متغير مؤقت.
!>

IWORK

!>          IWORK is INTEGER array, dimension (N)
!>          متغير مؤقت.
!>

INFO

!>          INFO is INTEGER
!>          عند الخروج، INFO = 0 إذا كانت الوحدة تعمل بشكل صحيح.
!>          INFO < 0: إذا كان المعامل -k يعني أن معامل الإدخال k-th كان به
!>                    خطأ.
!>          INFO = k > 0: إذا كان k هو فهرس القطر الرئيسي للمدخل
!>                      الذي كان صفره في U.  لذلك، فإن المصفوفة A مفردة
!>                      ولا يمكن إكمال تحليلها.  قد لا يكون
!>                      الباقي من التحسين أو الحلول موثوقًا به.
!>          INFO = N+1: المصفوفة A مفردة.  تقدير رقم الحالة
!>                      (RCOND) أقل من دقة الآلة.  ومع ذلك، تم
!>                      إكمال التحسين والحلول.
!>
