وثائق الوظيفة/الإجراء الفرعي

subroutine dgeqlf (integer m, integer n, double precision, dimension( lda, * ) a, integer lda, double precision, dimension( * ) tau, double precision, dimension( * ) work, integer lwork, integer info)

DGEQLF

الغرض:

!>
!> DGEQLF يحسب تحليل QL لمصفوفة حقيقية M-by-N وهي A:
!> A = Q * L.
!>

المعاملات

M 
!>          M هو عدد صحيح (INTEGER)
!>          عدد الصفوف في المصفوفة A.  M >= 0.
!>

N

!>          N هو عدد صحيح (INTEGER)
!>          عدد الأعمدة في المصفوفة A.  N >= 0.
!>

A

!>          A هي مصفوفة (array) ذات دقة مزدوجة (DOUBLE PRECISION)، ذات بُعد (LDA,N)
!>          عند الإدخال، المصفوفة A ذات الأبعاد M-by-N.
!>          عند الخروج،
!>          إذا كان m >= n، فإن المثلث السفلي للمصفوفة الفرعية
!>          A(m-n+1:m,1:n) يحتوي على المصفوفة المثلثية السفلية L ذات الأبعاد N-by-N؛
!>          إذا كان m <= n، فإن العناصر على وعلى أسفل القطر الفوقي رقم (n-m)
!>          تحتوي على المصفوفة شبه المنحرفة السفلية L ذات الأبعاد M-by-N؛
!>          العناصر المتبقية، مع مصفوفة TAU، تمثل
!>          المصفوفة المتعامدة Q كناتج لعوامل انعكاس أولية
!>          (انظر التفاصيل الإضافية).
!>

LDA

!>          LDA هو عدد صحيح (INTEGER)
!>          البعد الأمامي للمصفوفة A.  LDA >= max(1,M).
!>

TAU

!>          TAU هي مصفوفة (array) ذات دقة مزدوجة (DOUBLE PRECISION)، ذات بُعد (min(M,N))
!>          العوامل القياسية لعوامل الانعكاس الأولية (انظر التفاصيل الإضافية).
!>

WORK

!>          WORK هي مصفوفة (array) ذات دقة مزدوجة (DOUBLE PRECISION)، ذات بُعد (MAX(1,LWORK))
!>          عند الخروج، إذا كان INFO = 0، فإن WORK(1) يعيد حجم LWORK الأمثل.
!>

LWORK

!>          LWORK هو عدد صحيح (INTEGER)
!>          بُعد مصفوفة WORK.
!>          LWORK >= 1، إذا كان MIN(M,N) = 0، و LWORK >= N، بخلاف ذلك.
!>          للحصول على الأداء الأمثل LWORK >= N*NB، حيث NB هو
!>          حجم الكتلة الأمثل.
!>
!>          إذا كان LWORK = -1، فإنه يُفترض استعلام عن مساحة العمل؛ تقوم الروتين
!>          بحساب حجم مصفوفة WORK الأمثل فقط، وإعادة
!>          هذه القيمة كالإدخال الأول في مصفوفة WORK، ولا يتم إصدار أي رسالة خطأ
!>          متعلقة بـ LWORK بواسطة XERBLA.
!>

INFO

!>          INFO هو عدد صحيح (INTEGER)
!>          = 0:  انتهاء ناجح
!>          < 0:  إذا كان INFO = -i، فإن المعامل i كان له قيمة غير قانونية
!>

تفاصيل إضافية:

!>
!>  يتم تمثيل المصفوفة Q كناتج لعوامل انعكاس أولية
!>
!>     Q = H(k) . . . H(2) H(1)، حيث k = min(m,n).
!>
!>  كل H(i) لها الشكل
!>
!>     H(i) = I - tau * v * v**T
!>
!>  حيث tau هو مقياس حقيقي، و v هو متجه حقيقي مع
!>  v(m-k+i+1:m) = 0 و v(m-k+i) = 1؛ يتم تخزين v(1:m-k+i-1) عند الخروج في
!>  A(1:m-k+i-1,n-k+i)، و tau في TAU(i).
!>
