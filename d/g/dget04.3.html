subroutine dget04 (integer n, integer nrhs, double precision, dimension( ldx, * ) x, integer ldx, double precision, dimension( ldxact, * ) xact, integer ldxact, double precision rcond, double precision resid)

DGET04

الغرض:

!>
!> DGET04 يحسب الفرق بين الحل المحسوب والحل الحقيقي لنظام المعادلات الخطية.
!>
!> RESID =  ( norm(X-XACT) * RCOND ) / ( norm(XACT) * EPS ),
!> حيث RCOND هو مقلوب الرقم الشرطي و EPS هو رقم الآلة العائم.
!>

المعلمات

N 
!>          N هو عدد صحيح (INTEGER)
!>          عدد الصفوف في المصفوفتين X و XACT.  N >= 0.
!>

NRHS

!>          NRHS هو عدد صحيح (INTEGER)
!>          عدد الأعمدة في المصفوفتين X و XACT.  NRHS >= 0.
!>

X

!>          X هو مصفوفة دقة مزدوجة (DOUBLE PRECISION array)، البعد (LDX,NRHS)
!>          متجهات الحل المحسوبة. يتم تخزين كل متجه كعمود في المصفوفة X.
!>

LDX

!>          LDX هو عدد صحيح (INTEGER)
!>          البعد الرائد للمصفوفة X.  LDX >= max(1,N).
!>

XACT

!>          XACT هو مصفوفة دقة مزدوجة (DOUBLE PRECISION array)، البعد( LDX, NRHS )
!>          متجهات الحل الدقيقة. يتم تخزين كل متجه كعمود في المصفوفة XACT.
!>

LDXACT

!>          LDXACT هو عدد صحيح (INTEGER)
!>          البعد الرائد للمصفوفة XACT.  LDXACT >= max(1,N).
!>

RCOND

!>          RCOND هو دقة مزدوجة (DOUBLE PRECISION)
!>          مقلوب الرقم الشرطي للمصفوفة المعاملة في نظام المعادلات.
!>

RESID

!>          RESID هو دقة مزدوجة (DOUBLE PRECISION)
!>          الحد الأقصى لمتجهات الحل NRHS لـ
!>          ( norm(X-XACT) * RCOND ) / ( norm(XACT) * EPS )
!>
