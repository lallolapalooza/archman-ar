توثيق الوظيفة/الروتين الفرعي

subroutine dgemlq (character side, character trans, integer m, integer n, integer k, double precision, dimension( lda, * ) a, integer lda, double precision, dimension( * ) t, integer tsize, double precision, dimension( ldc, * ) c, integer ldc, double precision, dimension( * ) work, integer lwork, integer info)

DGEMLQ

الغرض:

!>
!>     DGEMLQ تستبدل المصفوفة C الحقيقية العامة M-by-N بـ
!>
!>                    SIDE = 'L'     SIDE = 'R'
!>    TRANS = 'N':      Q * C          C * Q
!>    TRANS = 'T':      Q**T * C       C * Q**T
!>    حيث Q مصفوفة متعامدة حقيقية معرفة كحاصل ضرب
!>    مُعكِسات مسدّسة مُغَلّفة محسوبة بواسطة تحليل LQ مسدود
!>    (DGELQ)
!>
!>

المعلمات

SIDE 
!>          SIDE is CHARACTER*1
!>          = 'L': تطبيق Q أو Q**T من اليسار؛
!>          = 'R': تطبيق Q أو Q**T من اليمين.
!>

TRANS

!>          TRANS is CHARACTER*1
!>          = 'N':  بدون تبديل، تطبيق Q؛
!>          = 'T':  تبديل، تطبيق Q**T.
!>

M

!>          M is INTEGER
!>          عدد صفوف المصفوفة A.  M >=0.
!>

N

!>          N is INTEGER
!>          عدد أعمدة المصفوفة C. N >= 0.
!>

K

!>          K is INTEGER
!>          عدد المُعكِسات المسدّسة التي يُعرّف حاصل ضربها
!>          المصفوفة Q.
!>          إذا كان SIDE = 'L'، فإن M >= K >= 0؛
!>          إذا كان SIDE = 'R'، فإن N >= K >= 0.
!>
!>

A

!>          A is DOUBLE PRECISION array, dimension
!>                               (LDA,M) if SIDE = 'L',
!>                               (LDA,N) if SIDE = 'R'
!>          جزء من هيكل البيانات لتمثيل Q كما تم إرجاعه بواسطة DGELQ.
!>

LDA

!>          LDA is INTEGER
!>          البعد الرائد للمصفوفة A. LDA >= max(1,K).
!>

T

!>          T is DOUBLE PRECISION array, dimension (MAX(5,TSIZE)).
!>          جزء من هيكل البيانات لتمثيل Q كما تم إرجاعه بواسطة DGELQ.
!>

TSIZE

!>          TSIZE is INTEGER
!>          بعد المصفوفة T. TSIZE >= 5.
!>

C

!>          C is DOUBLE PRECISION array, dimension (LDC,N)
!>          عند الإدخال، المصفوفة C بحجم M-by-N.
!>          عند الخروج، يتم استبدال C بـ Q*C أو Q**T*C أو C*Q**T أو C*Q.
!>

LDC

!>          LDC is INTEGER
!>          البعد الرائد للمصفوفة C. LDC >= max(1,M).
!>

WORK

!>          (مساحة عمل) DOUBLE PRECISION array, dimension (MAX(1,LWORK))
!>          عند الخروج، إذا كان INFO = 0، يتم إرجاع WORK(1) الحد الأدنى LWORK.
!>

LWORK

!>          LWORK is INTEGER
!>          بعد المصفوفة WORK. LWORK >= 1.
!>          إذا كان LWORK = -1، فإنه يُفترض وجود استعلام مساحة عمل. تقوم الروتين
!>          فقط بحساب حجم مصفوفة WORK، وإرجاع هذه القيمة كـ WORK(1)، ولا يتم
!>          إصدار أي رسالة خطأ متعلقة بـ WORK بواسطة XERBLA.
!>

INFO

!>          INFO is INTEGER
!>          = 0:  نجاح الخروج
!>          < 0:  إذا كان INFO = -i، فإن المعلمة i كانت ذات قيمة غير قانونية
!>

!>
!>          T(2): حجم كتلة الصف (MB)
!>          T(3): حجم كتلة العمود (NB)

!>                           DLASWLQ أو DGELQT

!>  اعتمادًا على أبعاد المصفوفة M و N، وأحجام كتل الصفوف والأعمدة

!>  DLASWLQ (إذا كانت المصفوفة عريضة وقصيرة) أو DGELQT لحساب
!>  تحليل LQ.
!>  سيستخدم هذا الإصدار من DGEMLQ إما DLAMSWLQ أو DGEMLQT لـ
!>  ضرب المصفوفة Q بمصفوفة أخرى.
