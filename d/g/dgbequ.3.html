subroutine dgbequ (integer m, integer n, integer kl, integer ku, double precision, dimension( ldab, * ) ab, integer ldab, double precision, dimension( * ) r, double precision, dimension( * ) c, double precision rowcnd, double precision colcnd, double precision amax, integer info)

DGBEQU

الغرض:

!>
!> DGBEQU يحسب عوامل قياس الصف والعمود المعدة لموازنة مصفوفة A ذات النطاق M-by-N
!> وتقليل رقم الشرط الخاص بها. تعيد R عوامل قياس الصف و C عوامل القياس العمودي،
!> التي تم اختيارها لمحاولة جعل أكبر عنصر في كل صف وعمود من المصفوفة B
!> بعناصر B(i,j)=R(i)*A(i,j)*C(j) له قيمة مطلقة 1.
!>
!> R(i) و C(j) مقيدان بأن يكونا بين SMLNUM = أصغر رقم آمن و BIGNUM = أكبر رقم آمن.
!> استخدام عوامل القياس هذه لا يضمن تقليل رقم شرط A ولكنه يعمل بشكل جيد في الممارسة.
!>

المعلمات

M 
!>          M هو عدد صحيح (INTEGER)
!>          عدد صفوف المصفوفة A.  M >= 0.
!>

N

!>          N هو عدد صحيح (INTEGER)
!>          عدد أعمدة المصفوفة A.  N >= 0.
!>

KL

!>          KL هو عدد صحيح (INTEGER)
!>          عدد الأقطار الفرعية داخل نطاق A.  KL >= 0.
!>

KU

!>          KU هو عدد صحيح (INTEGER)
!>          عدد الأقطار الفرعية العلوية داخل نطاق A.  KU >= 0.
!>

AB

!>          AB هو مصفوفة ذات دقة مزدوجة (DOUBLE PRECISION)، ذات أبعاد (LDAB,N)
!>          المصفوفة ذات النطاق A، مخزنة في الصفوف من 1 إلى KL+KU+1.  يتم تخزين
!>          العمود j من A في العمود j من المصفوفة AB كما يلي:
!>          AB(ku+1+i-j,j) = A(i,j) لـ max(1,j-ku) <= i <= min(m,j+kl).
!>

LDAB

!>          LDAB هو عدد صحيح (INTEGER)
!>          البعد الرائد للمصفوفة AB.  LDAB >= KL+KU+1.
!>

R

!>          R هي مصفوفة ذات دقة مزدوجة (DOUBLE PRECISION)، ذات أبعاد (M)
!>          إذا كان INFO = 0، أو INFO > M، فإن R تحتوي على عوامل قياس الصف
!>          لـ A.
!>

C

!>          C هي مصفوفة ذات دقة مزدوجة (DOUBLE PRECISION)، ذات أبعاد (N)
!>          إذا كان INFO = 0، فإن C تحتوي على عوامل القياس العمودي لـ A.
!>

ROWCND

!>          ROWCND هي دقة مزدوجة (DOUBLE PRECISION)
!>          إذا كان INFO = 0 أو INFO > M، فإن ROWCND تحتوي على نسبة أصغر R(i)
!>          إلى أكبر R(i).  إذا كان ROWCND >= 0.1 وكان AMAX ليس كبيراً جداً أو
!>          صغيراً جداً، فلا فائدة من القياس بـ R.
!>

COLCND

!>          COLCND هي دقة مزدوجة (DOUBLE PRECISION)
!>          إذا كان INFO = 0، فإن COLCND تحتوي على نسبة أصغر C(i) إلى أكبر C(i).
!>          إذا كان COLCND >= 0.1، فلا فائدة من القياس بـ C.
!>

AMAX

!>          AMAX هي دقة مزدوجة (DOUBLE PRECISION)
!>          القيمة المطلقة لأكبر عنصر مصفوفة.  إذا كان AMAX قريبًا جدًا من
!>          الفيضان أو قريبًا جدًا من الاندفاع، فيجب قياس المصفوفة.
!>

INFO

!>          INFO هو عدد صحيح (INTEGER)
!>          = 0:  خرج ناجح
!>          < 0:  إذا كان INFO = -i، فإن المعلمة i كانت لها قيمة غير قانونية
!>          > 0:  إذا كان INFO = i، و i هو
!>                <= M:  الصف i من A هو صفر تمامًا
!>                >  M:  العمود (i-M) من A هو صفر تمامًا
!>
