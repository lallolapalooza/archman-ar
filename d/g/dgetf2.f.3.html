subroutine dgetf2 (m, n, a, lda, ipiv, info)
DGETF2 تحسب تحليل LU لمصفوفة عامة m-by-n باستخدام التقسيم الجزئي مع تبادلات الصفوف (خوارزمية غير محظورة).

توثيق الوظيفة/الروتين الفرعي

subroutine dgetf2 (integer m, integer n, double precision, dimension( lda, * ) a, integer lda, integer, dimension( * ) ipiv, integer info)

DGETF2 تحسب تحليل LU لمصفوفة عامة m-by-n باستخدام التقسيم الجزئي مع تبادلات الصفوف (خوارزمية غير محظورة).

الغرض:

!>
!> DGETF2 تحسب تحليل LU لمصفوفة عامة A بحجم m-by-n
!> باستخدام التقسيم الجزئي مع تبادلات الصفوف.
!>
!> التحليل له الصيغة
!>    A = P * L * U
!> حيث P هي مصفوفة تبديل، L مصفوفة مثلثة سفلية بعناصر قطرية واحدية
!> (شبه منحرف سفلي إذا كان m > n)، و U مصفوفة مثلثة علوية
!> (شبه منحرف علوي إذا كان m < n).
!>
!> هذه هي النسخة ذات المستوى 2 BLAS ذات النمط اليميني للخوارزمية.
!>

المعلمات

M 
!>          M هو عدد صحيح
!>          عدد صفوف المصفوفة A.  M >= 0.
!>

N

!>          N هو عدد صحيح
!>          عدد أعمدة المصفوفة A.  N >= 0.
!>

A

!>          A هي مصفوفة ذات دقة مزدوجة، بحجم (LDA,N)
!>          عند الإدخال، المصفوفة m × n المطلوب تحليلها.
!>          عند الخروج، عوامل L و U من التحليل
!>          A = P*L*U؛ عناصر القطر الأحادية لـ L غير مخزنة.
!>

LDA

!>          LDA هو عدد صحيح
!>          البعد الرئيسي للمصفوفة A.  LDA >= max(1,M).
!>

IPIV

!>          IPIV هي مصفوفة أعداد صحيحة، بحجم (min(M,N))
!>          فهارس التقسيم؛ من أجل 1 <= i <= min(M,N)، تم تبديل الصف i من
!>          المصفوفة مع الصف IPIV(i).
!>

INFO

!>          INFO هو عدد صحيح
!>          = 0: خروج ناجح
!>          < 0: إذا كان INFO = -k، فإن الوسيط k كان له قيمة غير قانونية
!>          > 0: إذا كان INFO = k، فإن U(k,k) يساوي صفرًا تمامًا. تم
!>               إكمال التحليل، ولكن العامل U منفرد تمامًا، وسيحدث
!>               قسمة على صفر إذا تم استخدامه لحل نظام معادلات.
!>
