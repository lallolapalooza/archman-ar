dgedmdq(3)

الحزمة: lapack-doc

التراخيص: custom

!>    JOBQ (مدخل) CHARACTER*1
!>    يحدد ما إذا كان سيتم حساب مصفوفة التقويم (orthonormal matrix)
!>    من تحلل QR وإرجاعها بشكل صريح.
!>    'Q' :: سيتم حساب المصفوفة Q لتحليل QR لمصفوفة لقطات البيانات
!>           وتخزينها في المصفوفة F. انظر وصف F.
!>    'N' :: لن يتم حساب المصفوفة Q بشكل صريح.
!>

JOBT

!>    JOBT (مدخل) CHARACTER*1
!>    يحدد ما إذا كان سيتم إرجاع العامل المثلثي العلوي (upper triangular factor)
!>    من تحلل QR.
!>    'R' :: سيتم إرجاع المصفوفة R لتحليل QR لمصفوفة لقطات البيانات F
!>           في المصفوفة Y. انظر وصف Y وتفاصيل إضافية.
!>    'N' :: لن يتم إرجاع المصفوفة R.
!>

JOBF

!>    JOBF (مدخل) CHARACTER*1
!>    يحدد ما إذا كان سيتم تخزين المعلومات اللازمة للمعالجة اللاحقة
!>    (مثل حساب متجهات Ritz المحسنة).
!>    'R' :: سيتم حساب المصفوفة اللازمة لتحسين متجهات Ritz وتخزينها
!>           في المصفوفة B. انظر وصف B.
!>    'E' :: سيتم حساب المتجهات الذاتية غير المتدرجة (unscaled eigenvectors)
!>           لـ DMD الدقيق وإرجاعها في المصفوفة B. انظر وصف B.
!>    'N' :: لن يتم حساب أي بيانات لتحسين المتجهات الذاتية.
!>    لتكون مفيدة عند الخروج، يتطلب هذا الخيار JOBQ='Q'.
!>

WHTSVD

!>    WHTSVD (مدخل) INTEGER, WHTSVD في { 1, 2, 3, 4 }
!>    يسمح باختيار خوارزمية SVD (تحليل القيم المفردة) من مكتبة LAPACK.
!>    1 :: DGESVD (خوارزمية QR SVD)
!>    2 :: DGESDD (خوارزمية Divide and Conquer؛ إذا توفرت مساحة عمل كافية،
!>         فهذا هو الخيار الأسرع)
!>    3 :: DGESVDQ (خوارزمية QR SVD المسبقة المعالجة؛ هذا والخيار 4 هما
!>         الخياران الأكثر دقة)
!>    4 :: DGEJSV (خوارزمية Jacobi SVD المسبقة المعالجة؛ هذا والخيار 3 هما
!>         الخياران الأكثر دقة)
!>    بالنسبة للطرق الأربعة المذكورة أعلاه، يمكن أن يحدث اختلاف كبير
!>    في دقة القيم المفردة الصغيرة إذا اختلفت المعايير (norms) للقطات
!>    بحيث تكون X غير مشروطة بشدة (severely ill-conditioned). إذا كانت
!>    القيم المفردة الصغيرة (أصغر من EPS*||X||) ذات أهمية وكان JOBS=='N'،
!>    فإن الخيارين (3, 4) يعطيان النتائج الأكثر دقة، حيث يكون الخيار 4 أفضل
!>    قليلاً وله خلفية نظرية أقوى.
!>    إذا كان JOBS=='S'، أي سيتم تطبيع أعمدة X، فستعطي جميع الطرق نتائج
!>    دقيقة متساوية تقريبًا.
!>

M

!>    M (مدخل) INTEGER, M >= 0
!>    بعد الفضاء الحركي (state space dimension) (عدد صفوف F).
!>

N

!>    N (مدخل) INTEGER, 0 <= N <= M
!>    عدد لقطات البيانات من مسار واحد، مأخوذة في أوقات منفصلة متساوية المسافات.
!>    هذا هو عدد أعمدة F.
!>

F

!>    F (مدخل/مخرج) REAL(KIND=WP) مصفوفة M × N
!>    > عند الدخول،
!>    أعمدة F هي سلسلة لقطات البيانات من مسار واحد، مأخوذة في أوقات منفصلة
!>    متساوية المسافات. يُفترض أن معايير الأعمدة لـ F تقع في نطاق الأرقام
!>    العائمة المطبعة (normalized floating point numbers).
!>    < عند الخروج،
!>    إذا كان JOBQ == 'Q'، فإن المصفوفة F تحتوي على المصفوفة التقويمية/العامل Q
!>    لتحليل QR لمصفوفة لقطات البيانات الأولية F. انظر وصف JOBQ.
!>    إذا كان JOBQ == 'N'، فإن الإدخالات في F تحت القطر الرئيسي تمامًا
!>    تحتوي، على أساس عمودي، على معلومات حول متجهات هاوسهولدر (Householder vectors)،
!>    كما هو مُرجع بواسطة DGEQRF. المعلومات المتبقية لاستعادة المصفوفة التقويمية
!>    لتحليل QR الأولي مخزنة في WORK(1:N). انظر وصف WORK.
!>

LDF

!>    LDF (مدخل) INTEGER, LDF >= M
!>    البعد الرائد (leading dimension) للمصفوفة F.
!>

X

!>    X (مساحة عمل/مخرج) REAL(KIND=WP) مصفوفة MIN(M,N) × (N-1)
!>    تُستخدم X كمساحة عمل للاحتفاظ بتمثيلات لقطات البيانات الـ N-1 الرائدة
!>    في الأساس التقويمي المحسوب في تحليل QR لـ F.
!>    عند الخروج، تحتوي الأعمدة K الرائدة لـ X على المتجهات المفردة اليسرى (left singular vectors)
!>    الـ K الرائدة للمحتوى الموصوف أعلاه لـ X. لرفعها إلى فضاء المتجهات المفردة اليسرى U(:,1:K)
!>    للبيانات المدخلة، اضربها مسبقًا بعامل Q من تحليل QR الأولي.
!>    انظر أوصاف F، K، V، و Z.
!>

LDX

!>    LDX (مدخل) INTEGER, LDX >= N
!>    البعد الرائد للمصفوفة X.
!>

Y

!>    Y (مساحة عمل/مخرج) REAL(KIND=WP) مصفوفة MIN(M,N) × (N-1)
!>    تُستخدم Y كمساحة عمل للاحتفاظ بتمثيلات لقطات البيانات الـ N-1 الأخيرة
!>    في الأساس التقويمي المحسوب في تحليل QR لـ F.
!>    عند الخروج،
!>    إذا كان JOBT == 'R'، فإن Y تحتوي على العامل المثلثي العلوي MIN(M,N) × N
!>    من تحليل QR لمصفوفة لقطات البيانات F.
!>

LDY

!>    LDY (مدخل) INTEGER, LDY >= N
!>    البعد الرائد للمصفوفة Y.
!>

NRNK

!>    NRNK (مدخل) INTEGER
!>    يحدد وضع حساب الرتبة العددية (numerical rank)، أي كيفية اقتطاع (truncate)
!>    القيم المفردة الصغيرة لمصفوفة الإدخال X. عند الدخول، إذا كان:
!>    NRNK = -1 :: القيمة المفردة i-th sigma(i) تُقتطع إذا كانت sigma(i) <= TOL*sigma(1)
!>                 هذا الخيار موصى به.
!>    NRNK = -2 :: القيمة المفردة i-th sigma(i) تُقتطع إذا كانت sigma(i) <= TOL*sigma(i-1)
!>                 هذا الخيار مُضمّن لأغراض البحث والتطوير (R&D).
!>                 يتطلب SVD عالي الدقة، والذي قد لا يكون ممكنًا.
!>    يمكن فرض الرتبة العددية باستخدام قيمة موجبة لـ NRNK على النحو التالي:
!>    0 < NRNK <= N-1 :: سيتم استخدام NRNK كحد أقصى من القيم المفردة الأكبر.
!>                     إذا كان عدد القيم المفردة غير الصفرية المحسوبة أقل من NRNK،
!>                     فسيتم استخدام هذه القيم غير الصفرية فقط وسيكون البعد الفعلي المستخدم
!>                     أقل من NRNK. يتم إرجاع العدد الفعلي للقيم المفردة غير الصفرية
!>                     في المتغير K. انظر وصف K.
!>

TOL

!>     TOL (مدخل) REAL(KIND=WP), 0 <= TOL < 1
!>     هامش الخطأ (tolerance) لاقتطاع القيم المفردة الصغيرة.
!>     انظر وصف NRNK.
!>

K

!>     K (مخرج) INTEGER, 0 <= K <= N
!>     بعد أساس SVD/POD لأول N-1 لقطة بيانات (أعمدة F) وعدد أزواج Ritz المحسوبة.
!>     يتم تحديد قيمة K وفقًا للقاعدة التي تم تعيينها بواسطة المعلمات NRNK و TOL.
!>     انظر أوصاف NRNK و TOL.
!>

REIG

!>    REIG (مخرج) REAL(KIND=WP) مصفوفة (N-1) × 1
!>    الإدخالات K (K<=N) الرائدة في REIG تحتوي على الأجزاء الحقيقية
!>    للقيم الذاتية المحسوبة REIG(1:K) + sqrt(-1)*IMEIG(1:K).
!>    انظر أوصاف K، IMEIG، و Z.
!>

IMEIG

!>    IMEIG (مخرج) REAL(KIND=WP) مصفوفة (N-1) × 1
!>    الإدخالات K (K 0، فإن زوج القيم الذاتية المترافقة المعقدة
!>    يُقرأ كالتالي:
!>    LAMBDA(i)   = REIG(i) + sqrt(-1)*IMAG(i)
!>    LAMBDA(i+1) = REIG(i) - sqrt(-1)*IMAG(i)
!>    أي أن أزواج المترافقات المعقدة لها فهارس متتالية (i, i+1)، مع سرد
!>    الجزء التخيلي الموجب أولاً.
!>    انظر أوصاف K، REIG، و Z.
!>

Z

!>    Z (مساحة عمل/مخرج) REAL(KIND=WP) مصفوفة M × (N-1)
!>    إذا كان JOBZ == 'V' فإن:
!>       Z تحتوي على متجهات Ritz الحقيقية على النحو التالي:
!>       إذا كان IMEIG(i) = 0، فإن Z(:,i) هو متجه ذاتي للقيمة الذاتية i-th Ritz.
!>       إذا كان IMEIG(i) > 0 (و IMEIG(i+1) < 0) فإن:
!>       [Z(:,i) Z(:,i+1)] يمتد فضاءً ثابتًا (invariant subspace) والقيم الذاتية Ritz
!>       المستخرجة من هذا الفضاء هي
!>       REIG(i) + sqrt(-1)*IMEIG(i) و
!>       REIG(i) - sqrt(-1)*IMEIG(i).
!>       المتجهات الذاتية المقابلة هي
!>       Z(:,i) + sqrt(-1)*Z(:,i+1) و
!>       Z(:,i) - sqrt(-1)*Z(:,i+1)، على التوالي.
!>    إذا كان JOBZ == 'F'، فإن الأوصاف المذكورة أعلاه تنطبق على
!>    أعمدة Z*V، حيث أعمدة V هي المتجهات الذاتية لنسبة رايلي (Rayleigh quotient) K×K،
!>    و Z متعامدة (orthonormal). أعمدة V لها بنية مماثلة:
!>    إذا كان IMEIG(i) == 0 فإن Z*V(:,i) هو متجه ذاتي، وإذا كان
!>    IMEIG(i) > 0 فإن Z*V(:,i) + sqrt(-1)*Z*V(:,i+1) و
!>                      Z*V(:,i) - sqrt(-1)*Z*V(:,i+1)
!>    هما المتجهات الذاتية لـ LAMBDA(i) و LAMBDA(i+1).
!>    انظر أوصاف REIG، IMEIG، X، و V.
!>

LDZ

!>    LDZ (مدخل) INTEGER, LDZ >= M
!>    البعد الرائد للمصفوفة Z.
!>

RES

!>    RES (مخرج) REAL(KIND=WP) مصفوفة (N-1) × 1
!>    تحتوي RES(1:K) على البواقي (residuals) لأزواج Ritz المحسوبة K.
!>    إذا كانت LAMBDA(i) حقيقية، فإن:
!>       RES(i) = || A * Z(:,i) - LAMBDA(i)*Z(:,i))||_2.
!>    إذا كان [LAMBDA(i), LAMBDA(i+1)] زوجًا مترافقًا معقدًا، فإن:
!>    RES(i)=RES(i+1) = || A * Z(:,i:i+1) - Z(:,i:i+1) *B||_F
!>    حيث B = [ real(LAMBDA(i)) imag(LAMBDA(i)) ]
!>              [-imag(LAMBDA(i)) real(LAMBDA(i)) ].
!>    يتحقق ما يلي:
!>    RES(i)   = || A*ZC(:,i)   - LAMBDA(i)  *ZC(:,i)   ||_2
!>    RES(i+1) = || A*ZC(:,i+1) - LAMBDA(i+1)*ZC(:,i+1) ||_2
!>    حيث ZC(:,i)   =  Z(:,i) + sqrt(-1)*Z(:,i+1)
!>          ZC(:,i+1) =  Z(:,i) - sqrt(-1)*Z(:,i+1)
!>    انظر وصف Z.
!>

B

!>    B (مخرج) REAL(KIND=WP) مصفوفة MIN(M,N) × (N-1).
!>    إذا كان JOBF == 'R'، فإن B(1:N,1:K) تحتوي على A*U(:,1:K)، ويمكن
!>    استخدامها لحساب المتجهات المحسنة؛ انظر المزيد من التفاصيل في المراجع المقدمة.
!>    إذا كان JOBF == 'E'، فإن B(1:N,1;K) تحتوي على
!>    A*U(:,1:K)*W(1:K,1:K)، وهي المتجهات من DMD الدقيق، حتى التحجيم (scaling)
!>    بمقلوب القيم الذاتية.
!>    في كلتا الحالتين، يمكن رفع محتوى B إلى البعد الأصلي للبيانات المدخلة
!>    عن طريق الضرب المسبق بعامل Q من تحليل QR الأولي.
!>    هنا تشير A إلى ضغط للمشغل الأساسي.
!>    انظر أوصاف F و X.
!>    إذا كان JOBF == 'N'، فلن يتم الرجوع إلى B.
!>

LDB

!>    LDB (مدخل) INTEGER, LDB >= MIN(M,N)
!>    البعد الرائد للمصفوفة B.
!>

V

!>    V (مساحة عمل/مخرج) REAL(KIND=WP) مصفوفة (N-1) × (N-1)
!>    عند الخروج، تحتوي V(1:K,1:K) على المتجهات الذاتية K لنسبة رايلي.
!>    يتم إرجاع المتجهات الذاتية لزوج القيم الذاتية المترافقة المعقدة
!>    في شكل حقيقي كما هو موضح في وصف Z. متجهات Ritz (التي تم إرجاعها في Z)
!>    هي حاصل ضرب X و V؛ انظر أوصاف
