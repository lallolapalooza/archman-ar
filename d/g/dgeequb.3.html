subroutine dgeequb (integer m, integer n, double precision, dimension( lda, * ) a, integer lda, double precision, dimension( * ) r, double precision, dimension( * ) c, double precision rowcnd, double precision colcnd, double precision amax, integer info)

DGEEQUB

الغرض:

!>
!> DGEEQUB يحسب مقاييس الصفوف والأعمدة المخصصة لموازنة مصفوفة A ذات الأبعاد M×N
!> وتقليل رقم شرطها. R ترجع عوامل قياس الصفوف و C عوامل قياس الأعمدة، المختارة
!> لمحاولة جعل أكبر عنصر في كل صف وعمود من المصفوفة B التي عناصرها
!> B(i,j)=R(i)*A(i,j)*C(j) لها قيمة مطلقة لا تزيد عن الأساس.
!>
!> R(i) و C(j) مقيدة لتكون قوة الأساس بين
!> SMLNUM = أصغر رقم آمن و BIGNUM = أكبر رقم آمن. استخدام
!> عوامل القياس هذه لا يضمن تقليل رقم شرط A ولكنه يعمل بشكل جيد في الممارسة.
!>
!> هذا الإجراء يختلف عن DGEEQU بتقييد عوامل القياس
!> إلى قوة الأساس. باستثناء الفائض والنقصان، فإن القياس بواسطة
!> هذه العوامل لا يضيف أخطاء تقريب إضافية. ومع ذلك، فإن
!> مقادير العناصر المقاسة لم تعد تقريبًا 1 ولكنها تقع
!> بين sqrt(radix) و 1/sqrt(radix).
!>

المعلمات

M 
!>          M هو INTEGER
!>          عدد صفوف المصفوفة A.  M >= 0.
!>

N

!>          N هو INTEGER
!>          عدد أعمدة المصفوفة A.  N >= 0.
!>

A

!>          A هو مصفوفة DOUBLE PRECISION، أبعادها (LDA,N)
!>          المصفوفة ذات الأبعاد M×N التي سيتم حساب عوامل موازنتها.
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرائد للمصفوفة A.  LDA >= max(1,M).
!>

R

!>          R هو مصفوفة DOUBLE PRECISION، أبعادها (M)
!>          إذا كانت INFO = 0 أو INFO > M، فإن R تحتوي على عوامل قياس الصفوف
!>          لـ A.
!>

C

!>          C هو مصفوفة DOUBLE PRECISION، أبعادها (N)
!>          إذا كانت INFO = 0، فإن C تحتوي على عوامل قياس الأعمدة لـ A.
!>

ROWCND

!>          ROWCND هو DOUBLE PRECISION
!>          إذا كانت INFO = 0 أو INFO > M، فإن ROWCND تحتوي على نسبة
!>          أصغر R(i) إلى أكبر R(i).  إذا كان ROWCND >= 0.1
!>          و AMAX ليست كبيرة جدًا ولا صغيرة جدًا، فلا فائدة من
!>          القياس بواسطة R.
!>

COLCND

!>          COLCND هو DOUBLE PRECISION
!>          إذا كانت INFO = 0، فإن COLCND تحتوي على نسبة
!>          أصغر C(i) إلى أكبر C(i).  إذا كان COLCND >= 0.1، فلا فائدة من
!>          القياس بواسطة C.
!>

AMAX

!>          AMAX هو DOUBLE PRECISION
!>          القيمة المطلقة لأكبر عنصر في المصفوفة. إذا كان AMAX قريبًا جدًا
!>          من الفائض أو قريبًا جدًا من النقصان، فيجب قياس
!>          المصفوفة.
!>

INFO

!>          INFO هو INTEGER
!>          = 0:  انتهاء ناجح
!>          < 0:  إذا كانت INFO = -i، فإن المعلمة i كانت ذات قيمة غير قانونية
!>          > 0:  إذا كانت INFO = i، و i هو
!>                <= M:  الصف i من A هو صفر تمامًا
!>                >  M:  العمود (i-M) من A هو صفر تمامًا
!>
