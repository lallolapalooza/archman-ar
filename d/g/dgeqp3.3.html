توثيق الوظيفة/الروتين الفرعي

subroutine dgeqp3 (integer m, integer n, double precision, dimension( lda, * ) a, integer lda, integer, dimension( * ) jpvt, double precision, dimension( * ) tau, double precision, dimension( * ) work, integer lwork, integer info)

DGEQP3

الغرض:

!>
!> DGEQP3 تحسب تحليل QR مع تبديل الأعمدة لمصفوفة A: A*P = Q*R باستخدام BLAS المستوى 3.
!>

المعلمات

M 
!>          M هو عدد صحيح
!>          عدد صفوف المصفوفة A. M >= 0.
!>

N

!>          N هو عدد صحيح
!>          عدد أعمدة المصفوفة A.  N >= 0.
!>

A

!>          A هي مصفوفة من نوع DOUBLE PRECISION، الأبعاد (LDA,N)
!>          عند الإدخال، المصفوفة A بحجم M*N.
!>          عند الخروج، تحتوي المثلث العلوي للمصفوفة على المصفوفة شبه المنحرفة العليا R بحجم min(M,N)-by-N؛ العناصر الموجودة أسفل
!>          القطر، جنبًا إلى جنب مع المصفوفة TAU، تمثل المصفوفة المتعامدة Q كناتج لـ min(M,N) عاكسات ابتدائية.
!>

LDA

!>          LDA هو عدد صحيح
!>          البعد الرئيسي للمصفوفة A. LDA >= max(1,M).
!>

JPVT

!>          JPVT هي مصفوفة من نوع صحيح، الأبعاد (N)
!>          عند الإدخال، إذا كانت JPVT(J).ne.0، يتم تبديل العمود J للمصفوفة A إلى مقدمة A*P (عمود رائد)؛ إذا كانت JPVT(J)=0،
!>          فإن العمود J للمصفوفة A هو عمود حر.
!>          عند الخروج، إذا كانت JPVT(J)=K، فإن العمود J للمصفوفة A*P هو
!>          العمود K للمصفوفة A.
!>

TAU

!>          TAU هي مصفوفة من نوع DOUBLE PRECISION، الأبعاد (min(M,N))
!>          عوامل قياسية للعواكس الابتدائية.
!>

WORK

!>          WORK هي مصفوفة من نوع DOUBLE PRECISION، الأبعاد (MAX(1,LWORK))
!>          عند الخروج، إذا كانت INFO=0، فإن WORK(1) ترجع قيمة LWORK المثلى.
!>

LWORK

!>          LWORK هو عدد صحيح
!>          بعد المصفوفة WORK. LWORK >= 3*N+1.
!>          لأداء مثالي، LWORK >= 2*N+( N+1 )*NB، حيث NB
!>          هو حجم الكتلة الأمثل.
!>
!>          إذا كانت LWORK = -1، فسيتم افتراض استعلام مساحة عمل؛ الروتين
!>          يحسب فقط الحجم الأمثل للمصفوفة WORK، ويرجع
!>          هذه القيمة كالإدخال الأول للمصفوفة WORK، ولا يتم إصدار رسالة خطأ
!>          متعلقة بـ LWORK بواسطة XERBLA.
!>

INFO

!>          INFO هو عدد صحيح
!>          = 0: خروج ناجح.
!>          < 0: إذا كانت INFO = -i، فقد كان الوسيط i غير قانوني.
!>

تفاصيل إضافية:

!>
!>  يتم تمثيل المصفوفة Q كناتج للعواكس الابتدائية
!>
!>     Q = H(1) H(2) . . . H(k)، حيث k = min(m,n).
!>
!>  كل H(i) له الشكل
!>
!>     H(i) = I - tau * v * v**T
!>
!>  حيث tau هو عدد حقيقي، و v هو متجه حقيقي/مركب
!>  مع v(1:i-1) = 0 و v(i) = 1؛ يتم تخزين v(i+1:m) عند الخروج في
!>  A(i+1:m,i)، و tau في TAU(i).
!>
