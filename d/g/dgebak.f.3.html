توثيق الوظيفة/الروتين الفرعي

subroutine dgebak (character job, character side, integer n, integer ilo, integer ihi, double precision, dimension( * ) scale, integer m, double precision, dimension( ldv, * ) v, integer ldv, integer info)

DGEBAK

الغرض:

!>
!> DGEBAK تشكل المتجهات الذاتية اليمنى أو اليسرى لمصفوفة عامة حقيقية
!> عن طريق التحويل العكسي على المتجهات الذاتية المحسوبة للمصفوفة
!> المتوازنة الناتجة عن DGEBAL.
!>

المعلمات

JOB 
!>          JOB is CHARACTER*1
!>          يحدد نوع التحويل العكسي المطلوب:
!>          = 'N': لا تفعل شيئا، ارجع فوراً؛
!>          = 'P': قم بالتحويل العكسي للتبديل فقط؛
!>          = 'S': قم بالتحويل العكسي للقياس فقط؛
!>          = 'B': قم بالتحويل العكسي للتبديل والقياس كليهما.
!>          يجب أن يكون JOB هو نفس الوسيط JOB الذي تم توفيره لـ DGEBAL.
!>

SIDE

!>          SIDE is CHARACTER*1
!>          = 'R':  V تحتوي على متجهات ذاتية يمنى؛
!>          = 'L':  V تحتوي على متجهات ذاتية يسري.
!>

N

!>          N is INTEGER
!>          عدد صفوف المصفوفة V.  N >= 0.
!>

ILO

!>          ILO is INTEGER
!>

IHI

!>          IHI is INTEGER
!>          الأعداد الصحيحة ILO و IHI التي حددتها DGEBAL.
!>          1 <= ILO <= IHI <= N, if N > 0; ILO=1 and IHI=0, if N=0.
!>

SCALE

!>          SCALE is DOUBLE PRECISION array, dimension (N)
!>          تفاصيل التبديل وعوامل القياس، كما تم إرجاعها
!>          من DGEBAL.
!>

M

!>          M is INTEGER
!>          عدد أعمدة المصفوفة V.  M >= 0.
!>

V

!>          V is DOUBLE PRECISION array, dimension (LDV,M)
!>          عند الدخول، مصفوفة المتجهات الذاتية اليمنى أو اليسرى المطلوب
!>          تحويلها، كما تم إرجاعها من DHSEIN أو DTREVC.
!>          عند الخروج، يتم استبدال V بالمتجهات الذاتية المحولة.
!>

LDV

!>          LDV is INTEGER
!>          البعد الرئيسي للمصفوفة V. LDV >= max(1,N).
!>

INFO

!>          INFO is INTEGER
!>          = 0:  خروج ناجح
!>          < 0:  إذا كان INFO = -i, فإن الوسيط i كان له قيمة غير قانونية.
!>
