subroutine dgqrts (integer n, integer m, integer p, double precision, dimension( lda, * ) a, double precision, dimension( lda, * ) af, double precision, dimension( lda, * ) q, double precision, dimension( lda, * ) r, integer lda, double precision, dimension( * ) taua, double precision, dimension( ldb, * ) b, double precision, dimension( ldb, * ) bf, double precision, dimension( ldb, * ) z, double precision, dimension( ldb, * ) t, double precision, dimension( ldb, * ) bwk, integer ldb, double precision, dimension( * ) taub, double precision, dimension( lwork ) work, integer lwork, double precision, dimension( * ) rwork, double precision, dimension( 4 ) result)

DGQRTS

الغرض:

!>
!> DGQRTS تختبر DGGQRF، التي تحسب عامل GQR لمصفوفة A بحجم N-بـ M ومصفوفة B بحجم N-بـ P: A = Q*R و B = Q*T*Z.
!>

المعلمات

N 
!>          N هو عدد صحيح (INTEGER)
!>          عدد صفوف المصفوفات A و B. N >= 0.
!>

M

!>          M هو عدد صحيح (INTEGER)
!>          عدد أعمدة المصفوفة A. M >= 0.
!>

P

!>          P هو عدد صحيح (INTEGER)
!>          عدد أعمدة المصفوفة B. P >= 0.
!>

A

!>          A هي مصفوفة ذات دقة مزدوجة (DOUBLE PRECISION)، ذات أبعاد (LDA,M).
!>          المصفوفة A بحجم N-بـ M.
!>

AF

!>          AF هي مصفوفة ذات دقة مزدوجة (DOUBLE PRECISION)، ذات أبعاد (LDA,N).
!>          تفاصيل عامل GQR للمصفوفة A و B، كما تم إرجاعها بواسطة DGGQRF، انظر SGGQRF لمزيد من التفاصيل.
!>

Q

!>          Q هي مصفوفة متعامدة ذات دقة مزدوجة (DOUBLE PRECISION)، ذات أبعاد (LDA,N).
!>          المصفوفة المتعامدة Q بحجم M-بـ M.
!>

R

!>          R هي مصفوفة ذات دقة مزدوجة (DOUBLE PRECISION)، ذات أبعاد (LDA,MAX(M,N)).
!>

LDA

!>          LDA هو عدد صحيح (INTEGER).
!>          البعد الرئيسي للمصفوفات A، AF، R، و Q.
!>          LDA >= max(M,N).
!>

TAUA

!>          TAUA هي مصفوفة ذات دقة مزدوجة (DOUBLE PRECISION)، ذات أبعاد (min(M,N)).
!>          العوامل العددية للمحددات الأولية، كما تم إرجاعها بواسطة DGGQRF.
!>

B

!>          B هي مصفوفة ذات دقة مزدوجة (DOUBLE PRECISION)، ذات أبعاد (LDB,P).
!>          عند الدخول، المصفوفة B بحجم N-بـ P.
!>

BF

!>          BF هي مصفوفة ذات دقة مزدوجة (DOUBLE PRECISION)، ذات أبعاد (LDB,N).
!>          تفاصيل عامل GQR للمصفوفة A و B، كما تم إرجاعها بواسطة DGGQRF، انظر SGGQRF لمزيد من التفاصيل.
!>

Z

!>          Z هي مصفوفة متعامدة ذات دقة مزدوجة (DOUBLE PRECISION)، ذات أبعاد (LDB,P).
!>          المصفوفة المتعامدة Z بحجم P-بـ P.
!>

T

!>          T هي مصفوفة ذات دقة مزدوجة (DOUBLE PRECISION)، ذات أبعاد (LDB,max(P,N)).
!>

BWK

!>          BWK هي مصفوفة ذات دقة مزدوجة (DOUBLE PRECISION)، ذات أبعاد (LDB,N).
!>

LDB

!>          LDB هو عدد صحيح (INTEGER).
!>          البعد الرئيسي للمصفوفات B، BF، Z، و T.
!>          LDB >= max(P,N).
!>

TAUB

!>          TAUB هي مصفوفة ذات دقة مزدوجة (DOUBLE PRECISION)، ذات أبعاد (min(P,N)).
!>          العوامل العددية للمحددات الأولية، كما تم إرجاعها بواسطة DGGRQF.
!>

WORK

!>          WORK هي مصفوفة ذات دقة مزدوجة (DOUBLE PRECISION)، ذات أبعاد (LWORK).
!>

LWORK

!>          LWORK هو عدد صحيح (INTEGER).
!>          بعد المصفوفة WORK، LWORK >= max(N,M,P)**2.
!>

RWORK

!>          RWORK هي مصفوفة ذات دقة مزدوجة (DOUBLE PRECISION)، ذات أبعاد (max(N,M,P)).
!>

RESULT

!>          RESULT هي مصفوفة ذات دقة مزدوجة (DOUBLE PRECISION)، ذات أبعاد (4).
!>          نسب الاختبار:
!>            RESULT(1) = norm( R - Q'*A ) / ( MAX(M,N)*norm(A)*ULP)
!>            RESULT(2) = norm( T*Z - Q'*B ) / (MAX(P,N)*norm(B)*ULP)
!>            RESULT(3) = norm( I - Q'*Q ) / ( M*ULP )
!>            RESULT(4) = norm( I - Z'*Z ) / ( P*ULP )
!>
