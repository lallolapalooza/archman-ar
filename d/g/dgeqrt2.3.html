توثيق الوظيفة/الروتين الفرعي

subroutine dgeqrt2 (integer m, integer n, double precision, dimension( lda, * ) a, integer lda, double precision, dimension( ldt, * ) t, integer ldt, integer info)

DGEQRT2 تحسب عامل QR لمصفوفة حقيقية عامة M-by-N A، باستخدام تمثيل WY المضغوط لـ Q.

الغرض:

!>
!> DGEQRT2 تحسب عامل QR لمصفوفة حقيقية M-by-N A،
!> باستخدام تمثيل WY المضغوط لـ Q.
!>

المعلمات

M 
!>          M هو عدد صحيح (INTEGER)
!>          عدد صفوف المصفوفة A.  M >= N.
!>

N

!>          N هو عدد صحيح (INTEGER)
!>          عدد أعمدة المصفوفة A.  N >= 0.
!>

A

!>          A هو مصفوفة مزدوجة الدقة (DOUBLE PRECISION)، الأبعاد (LDA,N)
!>          عند الإدخال، العناصر الموجودة على القطر الرئيسي وفوقه تحتوي على المصفوفة المثلثية العلوية R بحجم N-by-N؛
!>          العناصر الموجودة أسفل القطر تحتوي على أعمدة V. راجع أدناه لمزيد من التفاصيل.
!>

LDA

!>          LDA هو عدد صحيح (INTEGER)
!>          البعد الرائد للمصفوفة A.  LDA >= max(1,M).
!>

T

!>          T هو مصفوفة مزدوجة الدقة (DOUBLE PRECISION)، الأبعاد (LDT,N)
!>          المصفوفة المثلثية العلوية بحجم N-by-N للعامل المنعكس الكتلي.
!>          العناصر الموجودة على القطر الرئيسي وفوقه تحتوي على العامل المنعكس الكتلي T؛
!>          العناصر الموجودة أسفل القطر غير مستخدمة.
!>          راجع أدناه لمزيد من التفاصيل.
!>

LDT

!>          LDT هو عدد صحيح (INTEGER)
!>          البعد الرائد للمصفوفة T.  LDT >= max(1,N).
!>

INFO

!>          INFO هو عدد صحيح (INTEGER)
!>          = 0: اكتمال بنجاح
!>          < 0: إذا كان INFO = -i، فإن المعلمة i كانت ذات قيمة غير قانونية
!>

تفاصيل إضافية:

!>
!>  تخزن المصفوفة V العوامل المنعكسة الأولية H(i) في العمود i
!>  أسفل القطر. على سبيل المثال، إذا كان M=5 و N=3، فإن المصفوفة V تكون
!>
!>               V = (  1       )
!>                   ( v1  1    )
!>                   ( v1 v2  1 )
!>                   ( v1 v2 v3 )
!>                   ( v1 v2 v3 )
!>
!>  حيث تمثل v1's المتجهات التي تحدد H(i)، والتي يتم إرجاعها
!>  في المصفوفة A.  الواحدات على طول قطر V لا يتم تخزينها في A.  العامل المنعكس الكتلي H يُعطى بعد ذلك بـ
!>
!>               H = I - V * T * V**T
!>
!>  حيث V**T هو منقول V.
!>
