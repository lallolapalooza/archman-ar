subroutine dgelss (integer m, integer n, integer nrhs, double precision, dimension( lda, * ) a, integer lda, double precision, dimension( ldb, * ) b, integer ldb, double precision, dimension( * ) s, double precision rcond, integer rank, double precision, dimension( * ) work, integer lwork, integer info)

DGELSS يحل أنظمة مصفوفات GE المحددة أو الناقصة

الغرض:

!>
!> DGELSS يحسب الحل الأدنى للمعيار لمسألة أقل المربعات الخطية الحقيقية:
!>
!> تقليل معيار 2 لـ | b - A*x |.
!>
!> باستخدام تفكيك القيم المفردة (SVD) للمصفوفة A. A هي مصفوفة M × N
!> قد تكون رتبتها ناقصة.
!>
!> يمكن التعامل مع عدة متجهات للجانب الأيمن b ومتجهات الحل x في استدعاء واحد؛
!> يتم تخزينها كأعمدة للمصفوفة B ذات الجانب الأيمن M × NRHS والمصفوفة X ذات الحل N × NRHS.
!>
!> يتم تحديد الرتبة الفعالة للمصفوفة A عن طريق التعامل مع القيم المفردة التي تقل عن
!> RCOND مضروبة في أكبر قيمة مفردة على أنها صفر.
!>

المعلمات

M 
!>          M هو عدد صحيح (INTEGER)
!>          عدد الصفوف في المصفوفة A. M >= 0.
!>

N

!>          N هو عدد صحيح (INTEGER)
!>          عدد الأعمدة في المصفوفة A. N >= 0.
!>

NRHS

!>          NRHS هو عدد صحيح (INTEGER)
!>          عدد الجوانب اليمنى، أي عدد أعمدة المصفوفتين B و X. NRHS >= 0.
!>

A

!>          A هي مصفوفة ذات دقة مزدوجة (DOUBLE PRECISION)، البعد (LDA,N)
!>          عند الإدخال، مصفوفة A بحجم M × N.
!>          عند الخروج، يتم استبدال أول min(m,n) صفوف من A
!>          بمتجهاتها المفردة اليمنى، مخزنة صفًا بصف.
!>

LDA

!>          LDA هو عدد صحيح (INTEGER)
!>          البعد الرائد للمصفوفة A. LDA >= max(1,M).
!>

B

!>          B هي مصفوفة ذات دقة مزدوجة (DOUBLE PRECISION)، البعد (LDB,NRHS)
!>          عند الإدخال، مصفوفة الجانب الأيمن B بحجم M × NRHS.
!>          عند الخروج، يتم استبدال B بمصفوفة الحل X بحجم N × NRHS.
!>          إذا كانت m >= n و RANK = n، فإن مجموع مربعات الباقي للحل في العمود i
!>          يعطى بمجموع مربعات العناصر n+1:m في ذلك العمود.
!>

LDB

!>          LDB هو عدد صحيح (INTEGER)
!>          البعد الرائد للمصفوفة B. LDB >= max(1,max(M,N)).
!>

S

!>          S هي مصفوفة ذات دقة مزدوجة (DOUBLE PRECISION)، البعد (min(M,N))
!>          القيم المفردة للمصفوفة A بترتيب تنازلي.
!>          رقم شرط A في معيار 2 = S(1)/S(min(m,n)).
!>

RCOND

!>          RCOND هو دقة مزدوجة (DOUBLE PRECISION)
!>          يستخدم RCOND لتحديد الرتبة الفعالة للمصفوفة A.
!>          القيم المفردة S(i) <= RCOND*S(1) تعتبر صفرًا.
!>          إذا كانت RCOND < 0، يتم استخدام دقة الآلة بدلاً من ذلك.
!>

RANK

!>          RANK هو عدد صحيح (INTEGER)
!>          الرتبة الفعالة للمصفوفة A، أي عدد القيم المفردة
!>          التي تكون أكبر من RCOND*S(1).
!>

WORK

!>          WORK هي مصفوفة ذات دقة مزدوجة (DOUBLE PRECISION)، البعد (MAX(1,LWORK))
!>          عند الخروج، إذا كانت INFO = 0، فإن WORK(1) يعيد LWORK الأمثل.
!>

LWORK

!>          LWORK هو عدد صحيح (INTEGER)
!>          بعد مصفوفة WORK. LWORK >= 1، وأيضًا:
!>          LWORK >= 3*min(M,N) + max( 2*min(M,N), max(M,N), NRHS )
!>          لأداء جيد، يجب أن يكون LWORK أكبر بشكل عام.
!>
!>          إذا كانت LWORK = -1، فسيتم افتراض استعلام مساحة العمل؛
!>          يقوم الروتين فقط بحساب الحجم الأمثل لمصفوفة WORK، ويعيد
!>          هذه القيمة كإدخال أول لمصفوفة WORK، ولا يتم إصدار أي رسالة خطأ
!>          متعلقة بـ LWORK بواسطة XERBLA.
!>

INFO

!>          INFO هو عدد صحيح (INTEGER)
!>          = 0: خروج ناجح
!>          < 0: إذا كانت INFO = -i، فإن المعلمة i كانت لها قيمة غير قانونية.
!>          > 0: فشل خوارزمية حساب SVD في التقارب؛
!>                إذا كانت INFO = i، فإن i عنصرًا خارج القطر لمصفوفة ثنائية القطر وسيطة
!>                لم تتقارب إلى الصفر.
!>
