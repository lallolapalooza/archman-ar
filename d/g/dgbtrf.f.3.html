توثيق الوظيفة/الروتين الفرعي

subroutine dgbtrf (integer m, integer n, integer kl, integer ku, double precision, dimension( ldab, * ) ab, integer ldab, integer, dimension( * ) ipiv, integer info)

DGBTRF

الغرض:

!>
!> تقوم DGBTRF بحساب تحليل LU لمصفوفة نطاقية حقيقية m-by-n A
!> باستخدام تبديل الصفوف الجزئي.
!>
!> هذه هي النسخة المجمعة للخوارزمية، تستدعي BLAS المستوى 3.
!>

المعلمات

M 
!>          M هو عدد صحيح (INTEGER)
!>          عدد صفوف المصفوفة A.  M >= 0.
!>

N

!>          N هو عدد صحيح (INTEGER)
!>          عدد أعمدة المصفوفة A.  N >= 0.
!>

KL

!>          KL هو عدد صحيح (INTEGER)
!>          عدد الأقطار الفرعية داخل نطاق A.  KL >= 0.
!>

KU

!>          KU هو عدد صحيح (INTEGER)
!>          عدد الأقطار الفوقية داخل نطاق A.  KU >= 0.
!>

AB

!>          AB هو مصفوفة ذات دقة مزدوجة (DOUBLE PRECISION array)، البعد (LDAB,N)
!>          عند الإدخال، المصفوفة A بتخزين النطاق، في الصفوف KL+1 إلى
!>          2*KL+KU+1؛ لا يلزم تعيين الصفوف 1 إلى KL من المصفوفة.
!>          يتم تخزين العمود j من A في العمود j من
!>          المصفوفة AB على النحو التالي:
!>          AB(kl+ku+1+i-j,j) = A(i,j) لـ max(1,j-ku)<=i<=min(m,j+kl)
!>
!>          عند الخروج، تفاصيل التحليل: يتم تخزين U كمصفوفة نطاقية علوية مثلثية
!>          مع KL+KU أقطار فوقية في
!>          الصفوف 1 إلى KL+KU+1، ويتم تخزين المضاعفات المستخدمة أثناء
!>          التحليل في الصفوف KL+KU+2 إلى 2*KL+KU+1.
!>          راجع أدناه لمزيد من التفاصيل.
!>

LDAB

!>          LDAB هو عدد صحيح (INTEGER)
!>          البعد الرائد للمصفوفة AB.  LDAB >= 2*KL+KU+1.
!>

IPIV

!>          IPIV هو مصفوفة أعداد صحيحة (INTEGER array)، البعد (min(M,N))
!>          فهارس المحور؛ لـ 1 <= i <= min(M,N)، تم تبديل الصف i من
!>          المصفوفة مع الصف IPIV(i).
!>

INFO

!>          INFO هو عدد صحيح (INTEGER)
!>          = 0: خروج ناجح
!>          < 0: إذا كان INFO = -i، كان المعامل i له قيمة غير قانونية
!>          > 0: إذا كان INFO = +i، فإن U(i,i) يساوي صفرًا تمامًا. تم
!>               إكمال التحليل، لكن العامل U هو
!>               فردي تمامًا، وسيحدث قسمة على صفر إذا تم استخدامه
!>               لحل نظام معادلات.
!>

!>

!>  M = N = 6، KL = 2، KU = 1:
!>
!>  عند الإدخال:                       عند الخروج:
!>
!>      *    *    *    +    +    +       *    *    *   u14  u25  u36
!>      *    *    +    +    +    +       *    *   u13  u24  u35  u46
!>      *   a12  a23  a34  a45  a56      *   u12  u23  u34  u45  u56
!>     a11  a22  a33  a44  a55  a66     u11  u22  u33  u44  u55  u66
!>     a21  a32  a43  a54  a65   *      m21  m32  m43  m54  m65   *
!>     a31  a42  a53  a64   *    *      m31  m42  m53  m64   *    *
!>
!>  العناصر المشار إليها بـ * غير مستخدمة بواسطة الروتين؛ العناصر المشار
!>  إليها بـ + لا يلزم تعيينها عند الإدخال، ولكنها مطلوبة بواسطة الروتين لتخزين
!>  عناصر U بسبب الملء الناتج عن تبديل الصفوف.
!>
