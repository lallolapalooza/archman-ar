توثيق الوظيفة/الروتين الفرعي

subroutine dgetrs (character trans, integer n, integer nrhs, double precision, dimension( lda, * ) a, integer lda, integer, dimension( * ) ipiv, double precision, dimension( ldb, * ) b, integer ldb, integer info)

DGETRS

الغرض:

!>
!> DGETRS تحل نظام المعادلات الخطية
!>    A * X = B  أو  A**T * X = B
!> مع مصفوفة عامة N-by-N A باستخدام تحليل LU الذي تم حسابه
!> بواسطة DGETRF.
!>

المعلمات

TRANS 
!>          TRANS is CHARACTER*1
!>          يحدد شكل نظام المعادلات:
!>          = 'N':  A * X = B  (بدون تبديل)
!>          = 'T':  A**T* X = B  (تبديل)
!>          = 'C':  A**T* X = B  (تبديل مترافق = تبديل)
!>

N

!>          N is INTEGER
!>          رتبة المصفوفة A.  N >= 0.
!>

NRHS

!>          NRHS is INTEGER
!>          عدد الأطراف اليمنى، أي عدد الأعمدة
!>          للمصفوفة B.  NRHS >= 0.
!>

A

!>          A is DOUBLE PRECISION array, dimension (LDA,N)
!>          العوامل L و U من تحليل A = P*L*U
!>          كما تم حسابه بواسطة DGETRF.
!>

LDA

!>          LDA is INTEGER
!>          البعد القيادي للمصفوفة A.  LDA >= max(1,N).
!>

IPIV

!>          IPIV is INTEGER array, dimension (N)
!>          فهارس المحاور من DGETRF؛ لكل 1<=i<=N، تم تبديل الصف i من
!>          المصفوفة مع الصف IPIV(i).
!>

B

!>          B is DOUBLE PRECISION array, dimension (LDB,NRHS)
!>          عند الإدخال، مصفوفة الطرف الأيمن B.
!>          عند الخروج، مصفوفة الحل X.
!>

LDB

!>          LDB is INTEGER
!>          البعد القيادي للمصفوفة B.  LDB >= max(1,N).
!>

INFO

!>          INFO is INTEGER
!>          = 0:  خروج ناجح
!>          < 0:  إذا كان INFO = -i، فإن المعلمة i كانت لها قيمة غير قانونية
!>
