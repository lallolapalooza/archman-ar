توثيق الوظيفة/الروتين الفرعي

subroutine dgttrs (character trans, integer n, integer nrhs, double precision, dimension( * ) dl, double precision, dimension( * ) d, double precision, dimension( * ) du, double precision, dimension( * ) du2, integer, dimension( * ) ipiv, double precision, dimension( ldb, * ) b, integer ldb, integer info)

DGTTRS

الغرض:

!>
!> DGTTRS يحل أحد أنظمة المعادلات
!>    A*X = B  أو  A**T*X = B,
!> مع مصفوفة ثلاثية القطر A باستخدام التحليل LU الذي تم حسابه
!> بواسطة DGTTRF.
!>

المعلمات

TRANS 
!>          TRANS هو CHARACTER*1
!>          يحدد شكل نظام المعادلات.
!>          = 'N':  A * X = B  (بدون تبديل)
!>          = 'T':  A**T* X = B  (تبديل)
!>          = 'C':  A**T* X = B  (تبديل مرافق = تبديل)
!>

N

!>          N هو INTEGER
!>          رتبة المصفوفة A.
!>

NRHS

!>          NRHS هو INTEGER
!>          عدد الأطراف اليمنى، أي عدد الأعمدة
!>          للمصفوفة B.  NRHS >= 0.
!>

DL

!>          DL هو مصفوفة DOUBLE PRECISION، ذات البعد (N-1)
!>          المضروبات (n-1) التي تحدد المصفوفة L من
!>          تحليل LU للمصفوفة A.
!>

D

!>          D هو مصفوفة DOUBLE PRECISION، ذات البعد (N)
!>          العناصر القطرية n للمصفوفة المثلثية العلوية U من
!>          تحليل LU للمصفوفة A.
!>

DU

!>          DU هو مصفوفة DOUBLE PRECISION، ذات البعد (N-1)
!>          العناصر (n-1) في القطر العلوي الأول لـ U.
!>

DU2

!>          DU2 هو مصفوفة DOUBLE PRECISION، ذات البعد (N-2)
!>          العناصر (n-2) في القطر العلوي الثاني لـ U.
!>

IPIV

!>          IPIV هو مصفوفة INTEGER، ذات البعد (N)
!>          مؤشرات المحور؛ لـ 1 <= i <= n، تم تبديل الصف i للمصفوفة
!>          مع الصف IPIV(i).  سيكون IPIV(i) دائمًا إما
!>          i أو i+1؛ يشير IPIV(i) = i إلى أنه لم يتم إجراء تبديل صف
!>          مطلوب.
!>

B

!>          B هي مصفوفة DOUBLE PRECISION، ذات البعد (LDB,NRHS)
!>          عند الدخول، المصفوفة التي تحتوي على متجهات الطرف الأيمن B.
!>          عند الخروج، يتم استبدال B بمتجهات الحل X.
!>

LDB

!>          LDB هو INTEGER
!>          البعد الرائد للمصفوفة B.  LDB >= max(1,N).
!>

INFO

!>          INFO هو INTEGER
!>          = 0:  انتهاء ناجح
!>          < 0:  إذا كان INFO = -i، فإن المعامل i كان له قيمة غير قانونية
!>
