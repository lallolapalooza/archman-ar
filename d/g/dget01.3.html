توثيق الوظيفة/الإجراء الفرعي

subroutine dget01 (integer m, integer n, double precision, dimension( lda, * ) a, integer lda, double precision, dimension( ldafac, * ) afac, integer ldafac, integer, dimension( * ) ipiv, double precision, dimension( * ) rwork, double precision resid)

DGET01

الغرض:

!>
!> DGET01 يعيد بناء مصفوفة A من تحليلها L*U ويحسب الباقي
!>    norm(L*U - A) / ( N * norm(A) * EPS ),
!> حيث EPS هو خطأ الآلة.
!>

المعلمات

M 
!>          M هو عدد صحيح
!>          عدد صفوف المصفوفة A.  M >= 0.
!>

N

!>          N هو عدد صحيح
!>          عدد أعمدة المصفوفة A.  N >= 0.
!>

A

!>          A هي مصفوفة ذات دقة مزدوجة، البعد (LDA,N)
!>          المصفوفة الأصلية M x N التي هي A.
!>

LDA

!>          LDA هو عدد صحيح
!>          البعد الرائد للمصفوفة A.  LDA >= max(1,M).
!>

AFAC

!>          AFAC هي مصفوفة ذات دقة مزدوجة، البعد (LDAFAC,N)
!>          الشكل المفكك للمصفوفة A.  AFAC يحتوي على العوامل
!>          L و U من التحليل L*U كما تم حسابه بواسطة DGETRF.
!>          يتم الكتابة فوقه بالمصفوفة المعاد بناؤها، ثم بفارق
!>          L*U - A.
!>

LDAFAC

!>          LDAFAC هو عدد صحيح
!>          البعد الرائد للمصفوفة AFAC.  LDAFAC >= max(1,M).
!>

IPIV

!>          IPIV هي مصفوفة أعداد صحيحة، البعد (N)
!>          مؤشرات المحور من DGETRF.
!>

RWORK

!>          RWORK هي مصفوفة ذات دقة مزدوجة، البعد (M)
!>

RESID

!>          RESID هو دقة مزدوجة
!>          norm(L*U - A) / ( N * norm(A) * EPS )
!>
