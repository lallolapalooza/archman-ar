توثيق الوظيفة/الإجراء الفرعي

subroutine dgtt01 (integer n, double precision, dimension( * ) dl, double precision, dimension( * ) d, double precision, dimension( * ) du, double precision, dimension( * ) dlf, double precision, dimension( * ) df, double precision, dimension( * ) duf, double precision, dimension( * ) du2, integer, dimension( * ) ipiv, double precision, dimension( ldwork, * ) work, integer ldwork, double precision, dimension( * ) rwork, double precision resid)

DGTT01

الغرض:

!>
!> DGTT01 يعيد بناء مصفوفة ثلاثية الأقطار A من تحليلها LU
!> ويحسب الباقي
!>    norm(L*U - A) / ( norm(A) * EPS )،
!> حيث EPS هو الجزء العائم من الآلة.
!>

المعلمات

N 
!>          N هو عدد صحيح
!>          رتبة المصفوفة A. N >= 0.
!>

DL

!>          DL هي مصفوفة DOUBLE PRECISION، البعد (N-1)
!>          عناصر تحت القطر الفرعي (n-1) للمصفوفة A.
!>

D

!>          D هي مصفوفة DOUBLE PRECISION، البعد (N)
!>          عناصر القطر الرئيسي للمصفوفة A.
!>

DU

!>          DU هي مصفوفة DOUBLE PRECISION، البعد (N-1)
!>          عناصر فوق القطر الفرعي (n-1) للمصفوفة A.
!>

DLF

!>          DLF هي مصفوفة DOUBLE PRECISION، البعد (N-1)
!>          المضاعفات (n-1) التي تحدد المصفوفة L من
!>          تحليل LU للمصفوفة A.
!>

DF

!>          DF هي مصفوفة DOUBLE PRECISION، البعد (N)
!>          عناصر القطر الرئيسي N للمصفوفة المثلثية العلوية U من
!>          تحليل LU للمصفوفة A.
!>

DUF

!>          DUF هي مصفوفة DOUBLE PRECISION، البعد (N-1)
!>          عناصر القطر فوق الرئيسي الأول للمصفوفة U.
!>

DU2

!>          DU2 هي مصفوفة DOUBLE PRECISION، البعد (N-2)
!>          عناصر القطر فوق الرئيسي الثاني للمصفوفة U.
!>

IPIV

!>          IPIV هي مصفوفة INTEGER، البعد (N)
!>          مؤشرات المحور؛ لـ 1 <= i <= n، تم تبديل الصف i للمصفوفة
!>          مع الصف IPIV(i). سيكون IPIV(i) دائمًا إما
!>          i أو i+1؛ IPIV(i) = i يشير إلى عدم الحاجة لتبديل الصف.
!>

WORK

!>          WORK هي مصفوفة DOUBLE PRECISION، البعد (LDWORK,N)
!>

LDWORK

!>          LDWORK هو عدد صحيح
!>          البعد الرئيسي للمصفوفة WORK. LDWORK >= max(1,N).
!>

RWORK

!>          RWORK هي مصفوفة DOUBLE PRECISION، البعد (N)
!>

RESID

!>          RESID هي DOUBLE PRECISION
!>          الباقي المُقاس:  norm(L*U - A) / (norm(A) * EPS)
!>
