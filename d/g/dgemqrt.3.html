توثيق الوظيفة/الروتين الفرعي

subroutine dgemqrt (character side, character trans, integer m, integer n, integer k, integer nb, double precision, dimension( ldv, * ) v, integer ldv, double precision, dimension( ldt, * ) t, integer ldt, double precision, dimension( ldc, * ) c, integer ldc, double precision, dimension( * ) work, integer info)

DGEMQRT

Purpose:

!>
!> DGEMQRT يحل محل مصفوفة C العامة الحقيقية M-by-N بـ
!>
!>                 SIDE = 'L'     SIDE = 'R'
!> TRANS = 'N':      Q C            C Q
!> TRANS = 'T':   Q**T C            C Q**T
!>
!> حيث Q مصفوفة متعامدة حقيقية معرفة كحاصل ضرب K
!> عواكس ابتدائية:
!>
!>       Q = H(1) H(2) . . . H(K) = I - V T V**T
!>
!> تم إنشاؤها باستخدام تمثيل WY المضغوط كما تم إرجاعه بواسطة DGEQRT.
!>
!> Q من الرتبة M إذا كان SIDE = 'L' ومن الرتبة N إذا كان SIDE = 'R'.
!>

Parameters

SIDE 
!>          SIDE is CHARACTER*1
!>          = 'L': تطبيق Q أو Q**T من اليسار؛
!>          = 'R': تطبيق Q أو Q**T من اليمين.
!>

TRANS

!>          TRANS is CHARACTER*1
!>          = 'N':  بدون تبديل، تطبيق Q؛
!>          = 'C':  تبديل، تطبيق Q**T.
!>

M

!>          M is INTEGER
!>          عدد صفوف المصفوفة C. M >= 0.
!>

N

!>          N is INTEGER
!>          عدد أعمدة المصفوفة C. N >= 0.
!>

K

!>          K is INTEGER
!>          عدد العواكس الابتدائية التي يحدد حاصل ضربها
!>          المصفوفة Q.
!>          إذا كان SIDE = 'L'، فإن M >= K >= 0؛
!>          إذا كان SIDE = 'R'، فإن N >= K >= 0.
!>

NB

!>          NB is INTEGER
!>          حجم الكتلة المستخدم لتخزين T. K >= NB >= 1.
!>          يجب أن تكون هذه نفس قيمة NB المستخدمة لإنشاء T
!>          في DGEQRT.
!>

V

!>          V is DOUBLE PRECISION array, dimension (LDV,K)
!>          يجب أن يحتوي العمود i على المتجه الذي يحدد
!>          العاكس الابتدائي H(i)، لـ i = 1,2,...,k، كما تم إرجاعه بواسطة
!>          DGEQRT في أول K أعمدة من مصفوفة الوسائط A.
!>

LDV

!>          LDV is INTEGER
!>          البعد الرئيسي للمصفوفة V.
!>          إذا كان SIDE = 'L'، فإن LDA >= max(1,M)؛
!>          إذا كان SIDE = 'R'، فإن LDA >= max(1,N).
!>

T

!>          T is DOUBLE PRECISION array, dimension (LDT,K)
!>          العوامل المثلثية العليا للعواكس المجمعة
!>          كما تم إرجاعه بواسطة DGEQRT، مخزنة كمصفوفة NB-by-N.
!>

LDT

!>          LDT is INTEGER
!>          البعد الرئيسي للمصفوفة T. LDT >= NB.
!>

C

!>          C is DOUBLE PRECISION array, dimension (LDC,N)
!>          عند الإدخال، مصفوفة C بحجم M-by-N.
!>          عند الخروج، يتم استبدال C بـ Q C، Q**T C، C Q**T أو C Q.
!>

LDC

!>          LDC is INTEGER
!>          البعد الرئيسي للمصفوفة C. LDC >= max(1,M).
!>

WORK

!>          WORK is DOUBLE PRECISION array. The dimension of
!>          WORK is N*NB if SIDE = 'L', or  M*NB if SIDE = 'R'.
!>          (حجم WORK هو N*NB إذا كان SIDE = 'L'، أو M*NB إذا كان SIDE = 'R'.)
!>

INFO

!>          INFO is INTEGER
!>          = 0:  نجاح الخروج
!>          < 0:  إذا كان INFO = -i، فإن الوسيط i كان له قيمة غير قانونية
!>
