DGEMMTR

توثيق الوظيفة/الروتين الفرعي

subroutine dgemmtr (character uplo, character transa, character transb, integer n, integer k, double precision alpha, double precision, dimension(lda,*) a, integer lda, double precision, dimension(ldb,*) b, integer ldb, double precision beta, double precision, dimension(ldc,*) c, integer ldc)

DGEMMTR

الغرض:

!>
!> DGEMMTR  تقوم بإحدى عمليات المصفوفة-المصفوفة
!>
!>    C := alpha*op( A )*op( B ) + beta*C,
!>
!> حيث  op( X )  هي واحدة من:
!>
!>    op( X ) = X   أو   op( X ) = X**T,
!>
!> alpha و beta هما عددان قياسيان، و A، B، و C هي مصفوفات، مع كون  op( A )  مصفوفة
!> من حجم n في k، و op( B )  مصفوفة من حجم k في n، و C مصفوفة من حجم n في n.
!> وبالتالي، فإن الروتين الفرعي يصل فقط إلى الجزء المثلثي العلوي أو السفلي
!> من المصفوفة الناتجة C ويقوم بتحديثه. يمكن استخدام هذا السلوك إذا كان
!> من المعروف أن المصفوفة الناتجة C متناظرة.
!>

المعاملات

UPLO 
!>          UPLO هو CHARACTER*1
!>           عند الدخول، يحدد UPLO ما إذا كان الجزء المثلثي السفلي أو العلوي
!>           من C سيتم الوصول إليه وتحديثه.
!>
!>              UPLO = 'L' أو 'l'، يتم استخدام الجزء المثلثي السفلي من C.
!>
!>              UPLO = 'U' أو 'u'، يتم استخدام الجزء المثلثي العلوي من C.
!>

TRANSA

!>          TRANSA هو CHARACTER*1
!>           عند الدخول، يحدد TRANSA شكل op( A ) المراد استخدامه في
!>           ضرب المصفوفات على النحو التالي:
!>
!>              TRANSA = 'N' أو 'n'،  op( A ) = A.
!>
!>              TRANSA = 'T' أو 't'،  op( A ) = A**T.
!>
!>              TRANSA = 'C' أو 'c'،  op( A ) = A**T.
!>

TRANSB

!>          TRANSB هو CHARACTER*1
!>           عند الدخول، يحدد TRANSB شكل op( B ) المراد استخدامه في
!>           ضرب المصفوفات على النحو التالي:
!>
!>              TRANSB = 'N' أو 'n'،  op( B ) = B.
!>
!>              TRANSB = 'T' أو 't'،  op( B ) = B**T.
!>
!>              TRANSB = 'C' أو 'c'،  op( B ) = B**T.
!>

N

!>          N هو INTEGER
!>           عند الدخول، يحدد N عدد الصفوف والأعمدة
!>           للمصفوفة C، وعدد أعمدة op(B)، وعدد
!>           صفوف op(A). يجب أن يكون N على الأقل صفرًا.
!>

K

!>          K هو INTEGER
!>           عند الدخول، يحدد K عدد أعمدة المصفوفة
!>           op( A ) وعدد صفوف المصفوفة op( B ). يجب أن يكون K
!>           على الأقل صفرًا.
!>

ALPHA

!>          ALPHA هو DOUBLE PRECISION.
!>           عند الدخول، يحدد ALPHA العدد القياسي alpha.
!>

A

!>          A هي مصفوفة DOUBLE PRECISION، بأبعاد ( LDA, ka )، حيث ka هي
!>           k  عندما  TRANSA = 'N' أو 'n'،  و n  بخلاف ذلك.
!>           قبل الدخول مع  TRANSA = 'N' أو 'n'،  يجب أن يحتوي الجزء الرائد
!>           بحجم n في k  من المصفوفة  A  على المصفوفة  A،  وإلا
!>           يجب أن يحتوي الجزء الرائد  بحجم  k في m  من المصفوفة  A  على
!>           المصفوفة A.
!>

LDA

!>          LDA هو INTEGER
!>           عند الدخول، يحدد LDA البعد الأول للمصفوفة A كما تم تعريفه
!>           في البرنامج (الفرعي) المستدعي. عندما يكون  TRANSA = 'N' أو 'n'
!>           فيجب أن يكون LDA على الأقل  max( 1, n )،  وإلا
!>           يجب أن يكون LDA على الأقل  max( 1, k ).
!>

B

!>          B هي مصفوفة DOUBLE PRECISION، بأبعاد ( LDB, kb )، حيث kb هي
!>           n  عندما  TRANSB = 'N' أو 'n'،  و k  بخلاف ذلك.
!>           قبل الدخول مع  TRANSB = 'N' أو 'n'،  يجب أن يحتوي الجزء الرائد
!>           بحجم k في n  من المصفوفة  B  على المصفوفة  B،  وإلا
!>           يجب أن يحتوي الجزء الرائد  بحجم  n في k  من المصفوفة  B  على
!>           المصفوفة B.
!>

LDB

!>          LDB هو INTEGER
!>           عند الدخول، يحدد LDB البعد الأول للمصفوفة B كما تم تعريفه
!>           في البرنامج (الفرعي) المستدعي. عندما يكون  TRANSB = 'N' أو 'n'
!>           فيجب أن يكون LDB على الأقل  max( 1, k )،  وإلا
!>           يجب أن يكون LDB على الأقل  max( 1, n ).
!>

BETA

!>          BETA هو DOUBLE PRECISION.
!>           عند الدخول، يحدد BETA العدد القياسي beta.  عندما يتم
!>           توفير BETA  بالصفر، فلا حاجة لتعيين C عند الدخول.
!>

C

!>          C هي مصفوفة DOUBLE PRECISION، بأبعاد ( LDC, N )
!>           قبل الدخول، يجب أن يحتوي الجزء الرائد
!>           بحجم n في n  من المصفوفة  C  على المصفوفة  C،  باستثناء عندما
!>           يكون beta صفرًا، وفي هذه الحالة لا يلزم تعيين C عند الدخول.
!>           عند الخروج، يتم استبدال الجزء العلوي أو السفلي المثلثي للمصفوفة
!>           C  بالمصفوفة
!>           ( alpha*op( A )*op( B ) + beta*C )  بحجم n في n.
!>

LDC

!>          LDC هو INTEGER
!>           عند الدخول، يحدد LDC البعد الأول للمصفوفة C كما تم تعريفه
!>           في البرنامج (الفرعي) المستدعي. يجب أن يكون LDC على الأقل
!>           max( 1, n ).
!>
