توثيق الوظيفة/الروتين الفرعي

subroutine dget51 (integer itype, integer n, double precision, dimension( lda, * ) a, integer lda, double precision, dimension( ldb, * ) b, integer ldb, double precision, dimension( ldu, * ) u, integer ldu, double precision, dimension( ldv, * ) v, integer ldv, double precision, dimension( * ) work, double precision result)

DGET51

الغرض:

!>
!>      DGET51  بشكل عام يتحقق من تفكيك بالشكل:
!>
!>              A = U B V'
!>
!>      حيث ' تعني منقول (transpose) و U و V متعامدتان.
!>
!>      بشكل خاص، إذا كان ITYPE=1
!>
!>              RESULT = | A - U B V' | / ( |A| n ulp )
!>
!>      إذا كان ITYPE=2، فإن:
!>
!>              RESULT = | A - B | / ( |A| n ulp )
!>
!>      إذا كان ITYPE=3، فإن:
!>
!>              RESULT = | I - UU' | / ( n ulp )
!>

المعلمات

ITYPE 
!>          ITYPE هو عدد صحيح (INTEGER)
!>          يحدد نوع الاختبارات التي سيتم إجراؤها.
!>          =1: RESULT = | A - U B V' | / ( |A| n ulp )
!>          =2: RESULT = | A - B | / ( |A| n ulp )
!>          =3: RESULT = | I - UU' | / ( n ulp )
!>

N

!>          N هو عدد صحيح (INTEGER)
!>          حجم المصفوفة. إذا كان صفراً، فإن DGET51 لا تفعل شيئاً.
!>          يجب أن يكون على الأقل صفر.
!>

A

!>          A هي مصفوفة دقة مزدوجة (DOUBLE PRECISION array)، بعدها (LDA, N)
!>          المصفوفة الأصلية (غير المفككة).
!>

LDA

!>          LDA هو عدد صحيح (INTEGER)
!>          البعد الرئيسي لـ A. يجب أن يكون على الأقل 1
!>          وعلى الأقل N.
!>

B

!>          B هي مصفوفة دقة مزدوجة (DOUBLE PRECISION array)، بعدها (LDB, N)
!>          المصفوفة المفككة.
!>

LDB

!>          LDB هو عدد صحيح (INTEGER)
!>          البعد الرئيسي لـ B. يجب أن يكون على الأقل 1
!>          وعلى الأقل N.
!>

U

!>          U هي مصفوفة دقة مزدوجة (DOUBLE PRECISION array)، بعدها (LDU, N)
!>          المصفوفة المتعامدة على الجانب الأيسر في
!>          التفكيك.
!>          لا يتم الرجوع إليها إذا كان ITYPE=2
!>

LDU

!>          LDU هو عدد صحيح (INTEGER)
!>          البعد الرئيسي لـ U. يجب أن يكون LDU على الأقل N و
!>          على الأقل 1.
!>

V

!>          V هي مصفوفة دقة مزدوجة (DOUBLE PRECISION array)، بعدها (LDV, N)
!>          المصفوفة المتعامدة على الجانب الأيسر في
!>          التفكيك.
!>          لا يتم الرجوع إليها إذا كان ITYPE=2
!>

LDV

!>          LDV هو عدد صحيح (INTEGER)
!>          البعد الرئيسي لـ V. يجب أن يكون LDV على الأقل N و
!>          على الأقل 1.
!>

WORK

!>          WORK هي مصفوفة دقة مزدوجة (DOUBLE PRECISION array)، بعدها (2*N**2)
!>

RESULT

!>          RESULT هو دقة مزدوجة (DOUBLE PRECISION)
!>          القيم المحسوبة بواسطة الاختبار المحدد بواسطة ITYPE.
!>          القيمة محدودة حالياً بـ 1/ulp، لتجنب تجاوز السعة.
!>          يتم تمييز الأخطاء بواسطة RESULT=10/ulp.
!>
