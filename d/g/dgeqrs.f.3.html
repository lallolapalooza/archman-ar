توثيق الوظيفة/الإجراء الفرعي

subroutine dgeqrs (integer m, integer n, integer nrhs, double precision, dimension( lda, * ) a, integer lda, double precision, dimension( * ) tau, double precision, dimension( ldb, * ) b, integer ldb, double precision, dimension( lwork ) work, integer lwork, integer info)

DGEQRS

الغرض:

!>
!> حل مشكلة المربعات الصغرى
!>     min || A*X - B ||
!> باستخدام تحليل QR
!>     A = Q*R
!> الذي تم حسابه بواسطة DGEQRF.
!>

المعلمات

M 
!>          M هو عدد صحيح (INTEGER)
!>          عدد صفوف المصفوفة A.  M >= 0.
!>

N

!>          N هو عدد صحيح (INTEGER)
!>          عدد أعمدة المصفوفة A.  M >= N >= 0.
!>

NRHS

!>          NRHS هو عدد صحيح (INTEGER)
!>          عدد أعمدة B.  NRHS >= 0.
!>

A

!>          A هي مصفوفة ذات دقة مضاعفة (DOUBLE PRECISION array)، البعد (LDA,N)
!>          تفاصيل تحليل QR للمصفوفة الأصلية A كما
!>          تم إرجاعها بواسطة DGEQRF.
!>

LDA

!>          LDA هو عدد صحيح (INTEGER)
!>          البعد الرئيسي للمصفوفة A.  LDA >= M.
!>

TAU

!>          TAU هي مصفوفة ذات دقة مضاعفة (DOUBLE PRECISION array)، البعد (N)
!>          تفاصيل المصفوفة المتعامدة Q.
!>

B

!>          B هي مصفوفة ذات دقة مضاعفة (DOUBLE PRECISION array)، البعد (LDB,NRHS)
!>          عند الإدخال، مصفوفة الجانب الأيمن B ذات الأبعاد m-by-nrhs.
!>          عند الخروج، مصفوفة الحل X ذات الأبعاد n-by-nrhs.
!>

LDB

!>          LDB هو عدد صحيح (INTEGER)
!>          البعد الرئيسي للمصفوفة B. LDB >= M.
!>

WORK

!>          WORK هي مصفوفة ذات دقة مضاعفة (DOUBLE PRECISION array)، البعد (LWORK)
!>

LWORK

!>          LWORK هو عدد صحيح (INTEGER)
!>          طول المصفوفة WORK.  يجب أن تكون LWORK على الأقل NRHS،
!>          ويجب أن تكون على الأقل NRHS*NB، حيث NB هو حجم الكتلة
!>          لهذه البيئة.
!>

INFO

!>          INFO هو عدد صحيح (INTEGER)
!>          = 0: الخروج بنجاح
!>          < 0: إذا كان INFO = -i، فإن المعامل i كان له قيمة غير قانونية
!>
