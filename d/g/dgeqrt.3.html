subroutine dgeqrt (integer m, integer n, integer nb, double precision, dimension( lda, * ) a, integer lda, double precision, dimension( ldt, * ) t, integer ldt, double precision, dimension( * ) work, integer info)

DGEQRT

الغرض:

!>
!> DGEQRT يحسب عامل QR المقطع لمصفوفة حقيقية M-by-N A
!> باستخدام تمثيل WY المدمج لـ Q.
!>

المعلمات

M 
!>          M هو عدد صحيح (INTEGER)
!>          عدد صفوف المصفوفة A.  M >= 0.
!>

N

!>          N هو عدد صحيح (INTEGER)
!>          عدد أعمدة المصفوفة A.  N >= 0.
!>

NB

!>          NB هو عدد صحيح (INTEGER)
!>          حجم المقطع المراد استخدامه في QR المقطع.  MIN(M,N) >= NB >= 1.
!>

A

!>          A هي مصفوفة (double precision)، أبعادها (LDA,N)
!>          عند الإدخال، المصفوفة A ذات الأبعاد M-by-N.
!>          عند الخروج، العناصر على القطر وما فوقه في المصفوفة
!>          تحتوي على المصفوفة الشبه منحرفة العلوية R ذات الأبعاد min(M,N)-by-N (R
!>          هي مصفوفة مثلثية علوية إذا كان M >= N)؛ العناصر أسفل القطر
!>          هي أعمدة V.
!>

LDA

!>          LDA هو عدد صحيح (INTEGER)
!>          البعد الرئيسي للمصفوفة A.  LDA >= max(1,M).
!>

T

!>          T هي مصفوفة (double precision)، أبعادها (LDT,MIN(M,N))
!>          المُنعكسات القطعية المثلثية العلوية المخزنة بشكل مدمج
!>          كسلسلة من القطع المثلثية العلوية.  انظر أدناه
!>          لمزيد من التفاصيل.
!>

LDT

!>          LDT هو عدد صحيح (INTEGER)
!>          البعد الرئيسي للمصفوفة T.  LDT >= NB.
!>

WORK

!>          WORK هي مصفوفة (double precision)، أبعادها (NB*N)
!>

INFO

!>          INFO هو عدد صحيح (INTEGER)
!>          = 0:  الخروج بنجاح
!>          < 0:  إذا كان INFO = -i، فإن المعامل i كان له قيمة غير قانونية
!>

تفاصيل إضافية:

!>
!>  المصفوفة V تخزن المُنعكسات الأولية H(i) في العمود i
!>  أسفل القطر. على سبيل المثال، إذا كانت M=5 و N=3، فإن المصفوفة V هي
!>
!>               V = (  1       )
!>                   ( v1  1    )
!>                   ( v1 v2  1 )
!>                   ( v1 v2 v3 )
!>                   ( v1 v2 v3 )
!>
!>  حيث تمثل الـ vi's المتجهات التي تحدد H(i)، والتي يتم إرجاعها
!>  في المصفوفة A.  الواحدات الموجودة على طول قطر V لا يتم تخزينها في A.
!>
!>  ليكن K = MIN(M,N).  عدد المقاطع هو B = سقف(K/NB)، حيث كل
!>  مقطع له رتبة NB باستثناء المقطع الأخير، والذي له رتبة
!>  IB = K - (B-1)*NB.  لكل من المقاطع B، يتم حساب عامل مُنعكس قِطعي علوي
!>  مثلثي: T1, T2, ..., TB.  يتم تخزين الـ NB-by-NB (والـ IB-by-IB
!>  للمقطع الأخير) T's في المصفوفة T ذات الأبعاد NB-by-K كـ
!>
!>               T = (T1 T2 ... TB).
!>
