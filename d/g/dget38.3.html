subroutine dget38 (double precision, dimension( 3 ) rmax, integer, dimension( 3 ) lmax, integer, dimension( 3 ) ninfo, integer knt, integer nin)

DGET38

الغرض:

!>
!> DGET38 تختبر DTRSEN، وهو إجراء لتقدير أرقام شروط مجموعة قيم ذاتية
!> و/أو الفضاء الجزئي الموجه الأيمن المرتبط بها.
!>
!> يتم قراءة مصفوفات الاختبار من ملف برقم وحدة منطقي NIN.
!>

المعاملات

RMAX 
!>          RMAX هو مصفوفة DOUBLE PRECISION، حجمها (3)
!>          قيم أكبر نسب الاختبار.
!>          RMAX(1) = أكبر البواقي من DHST01 أو مقارنة
!>                    بين استدعاءات مختلفة لـ DTRSEN
!>          RMAX(2) = أكبر خطأ في مقلوب أرقام الشروط
!>                    مع مراعاة شروطها
!>          RMAX(3) = أكبر خطأ في مقلوب أرقام الشروط
!>                    بدون مراعاة شروطها (قد تكون أكبر من RMAX(2))
!>

LMAX

!>          LMAX هو مصفوفة INTEGER، حجمها (3)
!>          LMAX(i) هو رقم المثال حيث يتم تحقيق أكبر نسبة اختبار
!>          RMAX(i). أيضًا:
!>          إذا أعادت DGEHRD القيمة INFO غير الصفرية في المثال i، فإن LMAX(1)=i
!>          إذا أعادت DHSEQR القيمة INFO غير الصفرية في المثال i، فإن LMAX(2)=i
!>          إذا أعادت DTRSEN القيمة INFO غير الصفرية في المثال i، فإن LMAX(3)=i
!>

NINFO

!>          NINFO هو مصفوفة INTEGER، حجمها (3)
!>          NINFO(1) = عدد مرات إعادة DGEHRD للقيمة INFO غير الصفرية
!>          NINFO(2) = عدد مرات إعادة DHSEQR للقيمة INFO غير الصفرية
!>          NINFO(3) = عدد مرات إعادة DTRSEN للقيمة INFO غير الصفرية
!>

KNT

!>          KNT هو INTEGER
!>          العدد الإجمالي للأمثلة التي تم اختبارها.
!>

NIN

!>          NIN هو INTEGER
!>          رقم وحدة الإدخال المنطقية.
!>
