subroutine dgeesx (character jobvs, character sort, external select, character sense, integer n, double precision, dimension( lda, * ) a, integer lda, integer sdim, double precision, dimension( * ) wr, double precision, dimension( * ) wi, double precision, dimension( ldvs, * ) vs, integer ldvs, double precision rconde, double precision rcondv, double precision, dimension( * ) work, integer lwork, integer, dimension( * ) iwork, integer liwork, logical, dimension( * ) bwork, integer info)

DGEESX تحسب القيم الذاتية، وشكل شور، واختياريًا، مصفوفة متجهات شور لمصفوفات GE

الغرض:

!>
!> DGEESX يحسب لمصفوفة حقيقية غير متناظرة A بحجم N×N،
!> القيم الذاتية، شكل شور الحقيقي T، واختياريًا، مصفوفة متجهات شور Z.  هذا يعطي تفكيك شور A = Z*T*(Z**T).
!>
!> اختياريًا، فإنه يرتب القيم الذاتية على القطر لشكل شور
!> بحيث تكون القيم الذاتية المحددة في الجزء العلوي الأيسر؛
!> يحسب رقم شرط عكسي لمتوسط القيم الذاتية المحددة (RCONDE)؛ ويحسب رقم شرط عكسي
!> للفضاء المتجهي الأيمن المقابل للقيم الذاتية المحددة (RCONDV).  الأعمدة الرائدة لـ Z تشكل أساسًا متعامدًا
!> لهذا الفضاء المتجهي.
!>
!> لمزيد من الشرح حول أرقام الشروط العكسية RCONDE و RCONDV، انظر القسم 4.10 من دليل مستخدمي LAPACK (حيث
!> تسمى هذه الكميات s و sep على التوالي).
!>
!> المصفوفة الحقيقية تكون في شكل شور الحقيقي إذا كانت شبه مثلثة عليا
!> مع كتل 1×1 و 2×2. سيتم توحيد الكتل 2×2 بالشكل
!>           [  a  b  ]
!>           [  c  a  ]
!>
!> حيث b*c < 0. القيم الذاتية لمثل هذه الكتلة هي a +- sqrt(bc).
!>

المعلمات

JOBVS 
!>          JOBVS هو CHARACTER*1
!>          = 'N': لا يتم حساب متجهات شور؛
!>          = 'V': يتم حساب متجهات شور.
!>

SORT

!>          SORT هو CHARACTER*1
!>          يحدد ما إذا كان سيتم ترتيب القيم الذاتية على القطر
!>          لشكل شور أم لا.
!>          = 'N': لا يتم ترتيب القيم الذاتية؛
!>          = 'S': يتم ترتيب القيم الذاتية (انظر SELECT).
!>

SELECT

!>          SELECT هي وظيفة LOGICAL لوسيطين DOUBLE PRECISION
!>          يجب تعريف SELECT بشكل خارجي في الإجراء الفرعي المستدعي.
!>          إذا كان SORT = 'S'، يتم استخدام SELECT لاختيار القيم الذاتية لترتيبها
!>          في الجزء العلوي الأيسر من شكل شور.
!>          إذا كان SORT = 'N'، لا يتم الرجوع إلى SELECT.
!>          يتم اختيار قيمة ذاتية مركبة WR(j)+sqrt(-1)*WI(j) إذا كان
!>          SELECT(WR(j),WI(j)) صحيحًا؛ أي، إذا تم اختيار إحدى
!>          القيم الذاتية المزدوجة المترافقة، فسيتم اختيار كليهما.
!>          لاحظ أن قيمة ذاتية مركبة مختارة قد لا تفي بعد الآن
!>          بـ SELECT(WR(j),WI(j)) = .TRUE. بعد الترتيب، نظرًا لأن
!>          الترتيب قد يغير قيمة القيم الذاتية المركبة
!>          (خاصة إذا كانت القيمة الذاتية غير مستقرة)؛ في هذه
!>          الحالة قد يتم تعيين INFO إلى N+3 (انظر INFO أدناه).
!>

SENSE

!>          SENSE هو CHARACTER*1
!>          يحدد أرقام الشروط العكسية التي يتم حسابها.
!>          = 'N': لا يتم حساب أي منها؛
!>          = 'E': يتم حسابه لمتوسط القيم الذاتية المحددة فقط؛
!>          = 'V': يتم حسابه للفضاء المتجهي الأيمن المحدد فقط؛
!>          = 'B': يتم حسابه لكليهما.
!>          إذا كان SENSE = 'E'، 'V' أو 'B'، يجب أن يكون SORT = 'S'.
!>

N

!>          N هو INTEGER
!>          رتبة المصفوفة A. N >= 0.
!>

A

!>          A هو مصفوفة DOUBLE PRECISION، البعد (LDA, N)
!>          عند الدخول، المصفوفة A بحجم N×N.
!>          عند الخروج، يتم استبدال A بشكل شور الحقيقي T.
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرائد للمصفوفة A. LDA >= max(1,N).
!>

SDIM

!>          SDIM هو INTEGER
!>          إذا كان SORT = 'N'، فإن SDIM = 0.
!>          إذا كان SORT = 'S'، فإن SDIM = عدد القيم الذاتية (بعد الترتيب)
!>                         التي يكون SELECT لها صحيحًا. (الأزواج المركبة
!>                         التي يكون SELECT لها صحيحًا لأي من
!>                         القيمتين الذاتيتين تُحسب كـ 2.)
!>

WR

!>          WR هي مصفوفة DOUBLE PRECISION، البعد (N)
!>

WI

!>          WI هي مصفوفة DOUBLE PRECISION، البعد (N)
!>          تحتوي WR و WI على الأجزاء الحقيقية والتخيلية، على التوالي،
!>          للقيم الذاتية المحسوبة، بنفس الترتيب الذي تظهر به
!>          على القطر لشكل شور الناتج T. تظهر الأزواج المترافقة
!>          للقيم الذاتية بشكل متتالٍ مع القيمة الذاتية التي لها الجزء التخيلي الموجب أولاً.
!>

VS

!>          VS هي مصفوفة DOUBLE PRECISION، البعد (LDVS,N)
!>          إذا كان JOBVS = 'V'، فإن VS يحتوي على المصفوفة المتعامدة Z لمتجهات شور.
!>          إذا كان JOBVS = 'N'، لا يتم الرجوع إلى VS.
!>

LDVS

!>          LDVS هو INTEGER
!>          البعد الرائد للمصفوفة VS. LDVS >= 1، وإذا كان
!>          JOBVS = 'V'، فإن LDVS >= N.
!>

RCONDE

!>          RCONDE هو DOUBLE PRECISION
!>          إذا كان SENSE = 'E' أو 'B'، فإن RCONDE يحتوي على رقم الشرط
!>          العكسي لمتوسط القيم الذاتية المحددة.
!>          لا يتم الرجوع إليه إذا كان SENSE = 'N' أو 'V'.
!>

RCONDV

!>          RCONDV هو DOUBLE PRECISION
!>          إذا كان SENSE = 'V' أو 'B'، فإن RCONDV يحتوي على رقم الشرط
!>          العكسي للفضاء المتجهي الأيمن المحدد.
!>          لا يتم الرجوع إليه إذا كان SENSE = 'N' أو 'E'.
!>

WORK

!>          WORK هي مصفوفة DOUBLE PRECISION، البعد (MAX(1,LWORK))
!>          عند الخروج، إذا كان INFO = 0، فإن WORK(1) يعيد LWORK الأمثل.
!>

LWORK

!>          LWORK هو INTEGER
!>          بعد المصفوفة WORK. LWORK >= max(1,3*N).
!>          أيضًا، إذا كان SENSE = 'E' أو 'V' أو 'B'،
!>          LWORK >= N+2*SDIM*(N-SDIM)، حيث SDIM هو عدد
!>          القيم الذاتية المحددة التي تم حسابها بواسطة هذا الروتين.
!>          لاحظ أن N+2*SDIM*(N-SDIM) <= N+N*N/2. لاحظ أيضًا أن الخطأ يتم فقط
!>          إرجاعه إذا كان LWORK < max(1,3*N)، ولكن إذا كان SENSE = 'E' أو 'V' أو
!>          'B' فقد لا يكون هذا كافيًا.
!>          لأداء جيد، يجب أن يكون LWORK أكبر بشكل عام.
!>
!>          إذا كان LWORK = -1، فيفترض استعلام مساحة العمل؛ الروتين
!>          يحسب فقط الحدود العليا للأحجام المثلى للمصفوفات
!>          WORK و IWORK، ويعيد هذه القيم كأول
!>          إدخالات في مصفوفتي WORK و IWORK، ولا يتم إصدار أي رسائل خطأ
!>          متعلقة بـ LWORK أو LIWORK بواسطة XERBLA.
!>

IWORK

!>          IWORK هي مصفوفة INTEGER، البعد (MAX(1,LIWORK))
!>          عند الخروج، إذا كان INFO = 0، فإن IWORK(1) يعيد LIWORK الأمثل.
!>

LIWORK

!>          LIWORK هو INTEGER
!>          بعد المصفوفة IWORK.
!>          LIWORK >= max(1,N).
!>          إذا كان SENSE = 'E' أو 'V' أو 'B'،
!>          LIWORK >= N + 2*SDIM.
!>
!>          إذا كان LIWORK = -1، فيفترض استعلام مساحة العمل؛ الروتين
!>          يحسب فقط الحدود العليا للأحجام المثلى للمصفوفات
!>          WORK و IWORK، ويعيد هذه القيم كأول
!>          إدخالات في مصفوفتي WORK و IWORK، ولا يتم إصدار أي رسائل خطأ
!>          متعلقة بـ LWORK أو LIWORK بواسطة XERBLA.
!>

BWORK

!>          BWORK هي مصفوفة LOGICAL، البعد (MAX(1,N))
!>          (هذه المصفوفة هي مؤقتة ولا يتم استخدامها إلا إذا تم استدعاء
!>          SELECT بواسطة DGEESX).
!>

INFO

!>          INFO هو INTEGER
!>          = 0: التنفيذ الناجح.
!>          < 0: إذا كان INFO = -i، فإن المعلمة i-th هي التي بها مشكلة.
!>          > 0: فشل خوارزمية التقارب.
!>             = 1: فشل في التقارب في غضون 30*N تكرار.
!>                    القيمة الذاتية المعقدة قد تكون غير صحيحة.
!>             = 2: فشل في إيجاد متجهات شور.
!>                    يمكن أن تكون القيمة الذاتية المعقدة غير صحيحة.
!>             = 3: فشل في التقارب لـ RCONDE.
!>                    يمكن أن تكون القيمة الذاتية المعقدة غير صحيحة.
!>             = 4: فشل في التقارب لـ RCONDV.
!>                    يمكن أن تكون القيمة الذاتية المعقدة غير صحيحة.
!>             = N+1: فشل في إيجاد متجهات شور،
!>                    لا يمكن التعامل مع بعض الأخطاء.
!>             = N+2: فشل في إيجاد متجهات شور.
!>                    لا يمكن التعامل مع بعض الأخطاء.
!>             = N+3: خطأ في ترتيب القيم الذاتية.
!>                    قد تكون القيمة الذاتية المعقدة غير صحيحة.
!>
