توثيق الوظيفة/الروتين الفرعي

subroutine dgetri (integer n, double precision, dimension( lda, * ) a, integer lda, integer, dimension( * ) ipiv, double precision, dimension( * ) work, integer lwork, integer info)

DGETRI

الغرض:

!>
!> DGETRI يحسب معكوس المصفوفة باستخدام تحلل LU
!> الذي تم حسابه بواسطة DGETRF.
!>
!> هذه الطريقة تعكس U ثم تحسب inv(A) عن طريق حل النظام
!> inv(A)*L = inv(U) لـ inv(A).
!>

المعلمات

N 
!>          N هو عدد صحيح (INTEGER)
!>          رتبة المصفوفة A.  N >= 0.
!>

A

!>          A هي مصفوفة ذات دقة مزدوجة (DOUBLE PRECISION array)، حجمها (LDA,N)
!>          عند الإدخال، عوامل L و U من التحلل
!>          A = P*L*U كما تم حسابها بواسطة DGETRF.
!>          عند الخروج، إذا كان INFO = 0، فإن معكوس المصفوفة الأصلية A.
!>

LDA

!>          LDA هو عدد صحيح (INTEGER)
!>          البعد الرئيسي للمصفوفة A.  LDA >= max(1,N).
!>

IPIV

!>          IPIV هي مصفوفة أعداد صحيحة (INTEGER array)، حجمها (N)
!>          مؤشرات المحور من DGETRF؛ لـ 1<=i<=N، تم تبديل الصف i للمصفوفة
!>          مع الصف IPIV(i).
!>

WORK

!>          WORK هي مصفوفة ذات دقة مزدوجة (DOUBLE PRECISION array)، حجمها (MAX(1,LWORK))
!>          عند الخروج، إذا كان INFO=0، فإن WORK(1) يعيد LWORK الأمثل.
!>

LWORK

!>          LWORK هو عدد صحيح (INTEGER)
!>          بعد مصفوفة WORK.  LWORK >= max(1,N).
!>          لأداء أمثل LWORK >= N*NB، حيث NB هو
!>          حجم الكتلة الأمثل الذي تم إرجاعه بواسطة ILAENV.
!>
!>          إذا كان LWORK = -1، فسيتم افتراض استعلام مساحة العمل؛ يقوم الروتين
!>          بحساب حجم مساحة العمل الأمثل فقط، ويعيد
!>          هذه القيمة كإدخال أول لمصفوفة WORK، ولا يتم إصدار رسالة خطأ
!>          متعلقة بـ LWORK بواسطة XERBLA.
!>

INFO

!>          INFO هو عدد صحيح (INTEGER)
!>          = 0:  خروج ناجح
!>          < 0:  إذا كان INFO = -i، فإن الوسيط i كان لديه قيمة غير قانونية
!>          > 0:  إذا كان INFO = i، فإن U(i,i) هو صفر تمامًا؛ المصفوفة
!>                منفردة ولا يمكن حساب معكوسها.
!>
