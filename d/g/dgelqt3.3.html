recursive subroutine dgelqt3 (m, n, a, lda, t, ldt, info)
DGELQT3 تحسب بشكل تكراري تحليل LQ لمصفوفة عامة حقيقية أو معقدة باستخدام تمثيل WY المدمج للمصفوفة Q.

توثيق الوظيفة/الروتين الفرعي

recursive subroutine dgelqt3 (integer m, integer n, double precision, dimension( lda, * ) a, integer lda, double precision, dimension( ldt, * ) t, integer ldt, integer info)

DGELQT3 تحسب بشكل تكراري تحليل LQ لمصفوفة عامة حقيقية أو معقدة باستخدام تمثيل WY المدمج للمصفوفة Q.

الغرض:

!>
!> DGELQT3 تحسب بشكل تكراري تحليل LQ لمصفوفة حقيقية M-by-N
!> A، باستخدام تمثيل WY المدمج للمصفوفة Q.
!>
!> بناءً على خوارزمية Elmroth و Gustavson،
!> IBM J. Res. Develop. Vol 44 No. 4 July 2000.
!>

المعاملات

M 
!>          M هو عدد صحيح (INTEGER)
!>          عدد صفوف المصفوفة A.  M =< N.
!>

N

!>          N هو عدد صحيح (INTEGER)
!>          عدد أعمدة المصفوفة A.  N >= 0.
!>

A

!>          A هي مصفوفة من نوع DOUBLE PRECISION، ذات بعد (LDA,N)
!>          عند الدخول، تحتوي العناصر على القطر الرئيسي وما تحته على المصفوفة
!>          المثلثة السفلى L بحجم N-by-N؛ العناصر فوق القطر الرئيسي هي صفوف V.
!>          انظر أدناه لمزيد من التفاصيل.
!>

LDA

!>          LDA هو عدد صحيح (INTEGER)
!>          البعد الرائد للمصفوفة A.  LDA >= max(1,M).
!>

T

!>          T هي مصفوفة من نوع DOUBLE PRECISION، ذات بعد (LDT,N)
!>          المصفوفة المثلثية العليا بحجم N-by-N للعامل الانعكاسي الكتلي.
!>          العناصر على القطر الرئيسي وما فوقه تحتوي على الانعكاس الكتلي T؛
!>          العناصر تحت القطر الرئيسي غير مستخدمة.
!>          انظر أدناه لمزيد من التفاصيل.
!>

LDT

!>          LDT هو عدد صحيح (INTEGER)
!>          البعد الرائد للمصفوفة T.  LDT >= max(1,N).
!>

INFO

!>          INFO هو عدد صحيح (INTEGER)
!>          = 0: انتهاء ناجح
!>          < 0: إذا كان INFO = -i، فإن الوسيط i كان له قيمة غير قانونية
!>

تفاصيل إضافية:

!>
!>  المصفوفة V تخزن الانعكاسات الأولية H(i) في الصف i
!>  فوق القطر الرئيسي. على سبيل المثال، إذا كانت M=5 و N=3، فإن المصفوفة V هي
!>
!>               V = (  1  v1 v1 v1 v1 )
!>                   (     1  v2 v2 v2 )
!>                   (     1  v3 v3 v3 )
!>
!>
!>  حيث تمثل الـ vi's المتجهات التي تحدد H(i)، والتي يتم إرجاعها
!>  في المصفوفة A.  الأرقام 1 على طول قطر V غير مخزنة في A.  ثم يتم إعطاء
!>  الانعكاس الكتلي H بواسطة
!>
!>               H = I - V * T * V**T
!>
!>  حيث V**T هو منقول (transpose) V.
!>
!>  للحصول على تفاصيل الخوارزمية، انظر Elmroth و Gustavson (المشار إليهما أعلاه).
!>
