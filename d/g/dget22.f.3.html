توثيق الوظيفة/الروتين الفرعي

subroutine dget22 (character transa, character transe, character transw, integer n, double precision, dimension( lda, * ) a, integer lda, double precision, dimension( lde, * ) e, integer lde, double precision, dimension( * ) wr, double precision, dimension( * ) wi, double precision, dimension( * ) work, double precision, dimension( 2 ) result)

DGET22

الغرض:

!>
!> DGET22 يقوم بفحص المتجهات الذاتية.
!>
!> الاختبار الأساسي هو:
!>
!>    RESULT(1) = | A E  -  E W | / ( |A| |E| ulp )
!>
!> باستخدام المعيار 1. كما أنه يختبر تطبيع E:
!>
!>    RESULT(2) = max | m-norm(E(j)) - 1 | / ( n ulp )
!>                 j
!>
!> حيث E(j) هو المتجه الذاتي j-th، والمعيار m هو المعيار الأقصى لـ
!> متجه. إذا كان المتجه الذاتي مركبًا، كما هو محدد من WI(j)
!> غير صفري، فإن المعيار الأقصى للمتجه ( er + i*ei ) هو الحد الأقصى
!> لـ
!>    |er(1)| + |ei(1)|, ... , |er(n)| + |ei(n)|
!>
!> W هو مصفوفة قطرية مقسمة، مع كتلة 1 × 1 لكل قيمة ذاتية حقيقية
!> وكتلة 2 × 2 لكل زوج مترافق مركب.
!> إذا كانت القيم الذاتية j و j+1 زوجًا مترافقًا مركبًا، بحيث
!> WR(j) = WR(j+1) = wr و WI(j) = - WI(j+1) = wi، فإن الكتلة 2 × 2
!> المقابلة للزوج ستكون:
!>
!>    (  wr  wi  )
!>    ( -wi  wr  )
!>
!> ضرب هذه الكتلة لمصفوفة n × 2 ( ur ui ) على اليمين
!> سيكون هو نفسه ضرب ur + i*ui بـ wr + i*wi.
!>
!> لمعالجة مخططات مختلفة لتخزين المتجهات الذاتية اليسرى، هناك
!> خيارات لاستخدام A-transpose بدلاً من A، و E-transpose بدلاً من E،
!> و/أو W-transpose بدلاً من W.
!>

المعلمات

TRANSA 
!>          TRANSA is CHARACTER*1
!>          يحدد ما إذا كان A منقولًا أم لا.
!>          = 'N':  لا يوجد نقل
!>          = 'T':  نقل
!>          = 'C':  نقل مترافق (= نقل)
!>

TRANSE

!>          TRANSE is CHARACTER*1
!>          يحدد ما إذا كان E منقولًا أم لا.
!>          = 'N':  لا يوجد نقل، المتجهات الذاتية في أعمدة E
!>          = 'T':  نقل، المتجهات الذاتية في صفوف E
!>          = 'C':  نقل مترافق (= نقل)
!>

TRANSW

!>          TRANSW is CHARACTER*1
!>          يحدد ما إذا كان W منقولًا أم لا.
!>          = 'N':  لا يوجد نقل
!>          = 'T':  نقل، استخدم -WI(j) بدلاً من WI(j)
!>          = 'C':  نقل مترافق، استخدم -WI(j) بدلاً من WI(j)
!>

N

!>          N is INTEGER
!>          ترتيب المصفوفة A. N >= 0.
!>

A

!>          A is DOUBLE PRECISION array, dimension (LDA,N)
!>          المصفوفة التي توجد متجهاتها الذاتية في E.
!>

LDA

!>          LDA is INTEGER
!>          البعد الرئيسي للمصفوفة A. LDA >= max(1,N).
!>

E

!>          E is DOUBLE PRECISION array, dimension (LDE,N)
!>          مصفوفة المتجهات الذاتية. إذا كان TRANSE = 'N'، يتم تخزين
!>          المتجهات الذاتية في أعمدة E، إذا كان TRANSE = 'T' أو 'C'، يتم
!>          تخزين المتجهات الذاتية في صفوف E.
!>

LDE

!>          LDE is INTEGER
!>          البعد الرئيسي للمصفوفة E. LDE >= max(1,N).
!>

WR

!>          WR is DOUBLE PRECISION array, dimension (N)
!>

WI

!>          WI is DOUBLE PRECISION array, dimension (N)
!>
!>          الأجزاء الحقيقية والتخيلية للقيم الذاتية لـ A.
!>          يتم الإشارة إلى القيم الذاتية الحقيقية البحتة بـ WI(j) = 0.
!>          يتم الإشارة إلى الأزواج المترافقة المركبة بـ WR(j)=WR(j+1) و
!>          WI(j) = - WI(j+1) غير صفري؛ يُفترض أن الجزء الحقيقي مخزن
!>          في الصف/العمود j والجزء التخيلي في
!>          الصف/العمود (j+1).
!>

WORK

!>          WORK is DOUBLE PRECISION array, dimension (N*(N+1))
!>

RESULT

!>          RESULT is DOUBLE PRECISION array, dimension (2)
!>          RESULT(1) = | A E  -  E W | / ( |A| |E| ulp )
!>          RESULT(2) = max | m-norm(E(j)) - 1 | / ( n ulp )
!>                       j
!>
