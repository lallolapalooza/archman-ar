توثيق الوظيفة/الروتين الفرعي

subroutine dgelss (integer m, integer n, integer nrhs, double precision, dimension( lda, * ) a, integer lda, double precision, dimension( ldb, * ) b, integer ldb, double precision, dimension( * ) s, double precision rcond, integer rank, double precision, dimension( * ) work, integer lwork, integer info)

DGELSS تحل أنظمة مصفوفات GE المفرطة التحديد أو الناقصة التحديد

الغرض:

!>
!> DGELSS تحسب الحل الأدنى للمعيار لمشكلة أقل المربعات الخطية الحقيقية:
!>
!> تقليل المعيار 2 لـ | b - A*x |.
!>
!> باستخدام تفكيك القيم المفردة (SVD) للمصفوفة A. A هي مصفوفة M × N
!> قد تكون ناقصة الرتبة.
!>
!> يمكن التعامل مع عدة متجهات جانب أيمن b ومتجهات حل x في استدعاء واحد؛
!> يتم تخزينها كأعمدة في المصفوفة اليمنى B ذات الأبعاد M × NRHS
!> ومصفوفة الحل X ذات الأبعاد N × NRHS.
!>
!> يتم تحديد الرتبة الفعالة للمصفوفة A عن طريق معاملة القيم المفردة
!> التي تقل عن RCOND مضروبة في أكبر قيمة مفردة على أنها صفر.
!>

المعلمات

M 
!>          M هو عدد صحيح (INTEGER)
!>          عدد صفوف المصفوفة A. M >= 0.
!>

N

!>          N هو عدد صحيح (INTEGER)
!>          عدد أعمدة المصفوفة A. N >= 0.
!>

NRHS

!>          NRHS هو عدد صحيح (INTEGER)
!>          عدد الجوانب اليمنى، أي عدد أعمدة المصفوفتين B و X. NRHS >= 0.
!>

A

!>          A هي مصفوفة من نوع double precision، بأبعاد (LDA,N)
!>          عند الإدخال، مصفوفة A ذات الأبعاد M × N.
!>          عند الخروج، يتم استبدال الصفوف الأولى min(m,n) من A
!>          متجهاتها المفردة اليمنى، مخزنة صفًا بصف.
!>

LDA

!>          LDA هو عدد صحيح (INTEGER)
!>          البعد الرائد للمصفوفة A. LDA >= max(1,M).
!>

B

!>          B هي مصفوفة من نوع double precision، بأبعاد (LDB,NRHS)
!>          عند الإدخال، مصفوفة الجانب الأيمن B ذات الأبعاد M × NRHS.
!>          عند الخروج، يتم استبدال B بمصفوفة الحل X ذات الأبعاد N × NRHS.
!>          إذا كان m >= n و RANK = n، فإن مجموع مربعات الباقي للحل في
!>          العمود i يُعطى بمجموع مربعات العناصر n+1:m في ذلك العمود.
!>

LDB

!>          LDB هو عدد صحيح (INTEGER)
!>          البعد الرائد للمصفوفة B. LDB >= max(1,max(M,N)).
!>

S

!>          S هي مصفوفة من نوع double precision، بأبعاد (min(M,N))
!>          القيم المفردة للمصفوفة A بترتيب تنازلي.
!>          العدد الشرطي للمصفوفة A في المعيار 2 = S(1)/S(min(m,n)).
!>

RCOND

!>          RCOND هو double precision
!>          يُستخدم RCOND لتحديد الرتبة الفعالة للمصفوفة A.
!>          تُعامل القيم المفردة S(i) <= RCOND*S(1) على أنها صفر.
!>          إذا كان RCOND < 0، يتم استخدام دقة الآلة بدلاً من ذلك.
!>

RANK

!>          RANK هو عدد صحيح (INTEGER)
!>          الرتبة الفعالة للمصفوفة A، أي عدد القيم المفردة
!>          الأكبر من RCOND*S(1).
!>

WORK

!>          WORK هي مصفوفة من نوع double precision، بأبعاد (MAX(1,LWORK))
!>          عند الخروج، إذا كان INFO = 0، تعيد WORK(1) قيمة LWORK المثلى.
!>

LWORK

!>          LWORK هو عدد صحيح (INTEGER)
!>          بعد المصفوفة WORK. LWORK >= 1، وأيضًا:
!>          LWORK >= 3*min(M,N) + max( 2*min(M,N), max(M,N), NRHS )
!>          لأداء جيد، يجب أن يكون LWORK بشكل عام أكبر.
!>
!>          إذا كان LWORK = -1، فسيتم افتراض استعلام عن مساحة العمل؛
!>          تحسب الروتين فقط الحجم الأمثل للمصفوفة WORK، وتعيد
!>          هذه القيمة كإدخال أول للمصفوفة WORK، ولا يتم إصدار أي رسالة خطأ
!>          متعلقة بـ LWORK بواسطة XERBLA.
!>

INFO

!>          INFO هو عدد صحيح (INTEGER)
!>          = 0: انتهاء ناجح
!>          < 0: إذا كان INFO = -i، فإن المعلمة i لها قيمة غير قانونية.
!>          > 0: فشل خوارزمية حساب SVD في التقارب؛
!>                إذا كان INFO = i، فإن i عنصرًا خارج القطر لمصفوفة ثنائية
!>                وسيطة لم تتقارب إلى الصفر.
!>
