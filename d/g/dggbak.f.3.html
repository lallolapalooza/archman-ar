subroutine dggbak (character job, character side, integer n, integer ilo, integer ihi, double precision, dimension( * ) lscale, double precision, dimension( * ) rscale, integer m, double precision, dimension( ldv, * ) v, integer ldv, integer info)

DGGBAK

الغرض:

!>
!> DGGBAK تشكل المتجهات الذاتية اليمنى أو اليسرى لمشكلة القيمة الذاتية المعممة الحقيقية A*x = lambda*B*x، عن طريق التحويل العكسي على
!> المتجهات الذاتية المحسوبة للزوج المتوازن من المصفوفات الناتجة عن DGGBAL.
!>

المعلمات

JOB 
!>          JOB is CHARACTER*1
!>          يحدد نوع التحويل العكسي المطلوب:
!>          = 'N':  لا تفعل شيئاً، ارجع فوراً؛
!>          = 'P':  قم بالتحويل العكسي للتبديل فقط؛
!>          = 'S':  قم بالتحويل العكسي للتوسيع فقط؛
!>          = 'B':  قم بالتحويلات العكسية للتبديل والتوسيع معاً.
!>          يجب أن يكون JOB هو نفسه الوسيط JOB المقدم إلى DGGBAL.
!>

SIDE

!>          SIDE is CHARACTER*1
!>          = 'R':  تحتوي V على المتجهات الذاتية اليمنى؛
!>          = 'L':  تحتوي V على المتجهات الذاتية اليسرى.
!>

N

!>          N is INTEGER
!>          عدد صفوف المصفوفة V.  N >= 0.
!>

ILO

!>          ILO is INTEGER
!>

IHI

!>          IHI is INTEGER
!>          الأعداد الصحيحة ILO و IHI التي تحددها DGGBAL.
!>          1 <= ILO <= IHI <= N، إذا كان N > 0؛ ILO=1 و IHI=0، إذا كان N=0.
!>

LSCALE

!>          LSCALE is DOUBLE PRECISION array, dimension (N)
!>          تفاصيل التباديل و/أو عوامل التوسيع المطبقة
!>          على الجانب الأيسر من A و B، كما تم إرجاعه بواسطة DGGBAL.
!>

RSCALE

!>          RSCALE is DOUBLE PRECISION array, dimension (N)
!>          تفاصيل التباديل و/أو عوامل التوسيع المطبقة
!>          على الجانب الأيمن من A و B، كما تم إرجاعه بواسطة DGGBAL.
!>

M

!>          M is INTEGER
!>          عدد أعمدة المصفوفة V.  M >= 0.
!>

V

!>          V is DOUBLE PRECISION array, dimension (LDV,M)
!>          عند الدخول، مصفوفة المتجهات الذاتية اليمنى أو اليسرى التي سيتم
!>          تحويلها، كما تم إرجاعه بواسطة DTGEVC.
!>          عند الخروج، يتم الكتابة فوق V بالمتجهات الذاتية المحولة.
!>

LDV

!>          LDV is INTEGER
!>          البعد الرئيسي للمصفوفة V. LDV >= max(1,N).
!>

INFO

!>          INFO is INTEGER
!>          = 0:  خروج ناجح.
!>          < 0:  إذا كان INFO = -i، فإن الوسيط i كان له قيمة غير قانونية.
!>
