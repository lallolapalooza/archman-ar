توثيق الوظيفة/الإجراء الفرعي

subroutine dtpcon (character norm, character uplo, character diag, integer n, double precision, dimension( * ) ap, double precision rcond, double precision, dimension( * ) work, integer, dimension( * ) iwork, integer info)

DTPCON

الغرض:

!>
!> DTPCON تقدر معكوس رقم الشرط لمصفوفة مثلثية مضغوطة A، إما بمعيار 1 أو
!> بمعيار ما لا نهاية.
!>
!> يتم حساب معيار A ويتم الحصول على تقدير لـ
!> norm(inv(A))، ثم يتم حساب معكوس رقم الشرط على أنه
!>    RCOND = 1 / ( norm(A) * norm(inv(A)) ).
!>

المعلمات

NORM 
!>          NORM is CHARACTER*1
!>          يحدد ما إذا كان رقم الشرط القياسي 1 أو رقم الشرط القياسي
!>          ما لا نهاية مطلوبًا:
!>          = '1' or 'O':  المعيار 1؛
!>          = 'I':         معيار ما لا نهاية.
!>

UPLO

!>          UPLO is CHARACTER*1
!>          = 'U':  A مثلثية علوية؛
!>          = 'L':  A مثلثية سفلية.
!>

DIAG

!>          DIAG is CHARACTER*1
!>          = 'N':  A مثلثية غير وحدة؛
!>          = 'U':  A مثلثية وحدة.
!>

N

!>          N is INTEGER
!>          ترتيب المصفوفة A.  N >= 0.
!>

AP

!>          AP is DOUBLE PRECISION array, dimension (N*(N+1)/2)
!>          المصفوفة المثلثية العلوية أو السفلية A، مضغوطة عمودياً في
!>          مصفوفة خطية. يتم تخزين العمود j من A في المصفوفة
!>          AP كما يلي:
!>          إذا كان UPLO = 'U'، فإن AP(i + (j-1)*j/2) = A(i,j) لـ 1<=i<=j؛
!>          إذا كان UPLO = 'L'، فإن AP(i + (j-1)*(2n-j)/2) = A(i,j) لـ j<=i<=n.
!>          إذا كان DIAG = 'U'، فلا يتم الرجوع إلى عناصر القطر لـ A
!>          ويفترض أنها 1.
!>

RCOND

!>          RCOND is DOUBLE PRECISION
!>          معكوس رقم شرط المصفوفة A،
!>          محسوب على أنه RCOND = 1/(norm(A) * norm(inv(A))).
!>

WORK

!>          WORK is DOUBLE PRECISION array, dimension (3*N)
!>

IWORK

!>          IWORK is INTEGER array, dimension (N)
!>

INFO

!>          INFO is INTEGER
!>          = 0:  انتهاء ناجح
!>          < 0:  إذا كان INFO = -i، فإن الوسيط i كان له قيمة غير قانونية
!>
