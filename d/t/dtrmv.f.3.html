subroutine dtrmv (character uplo, character trans, character diag, integer n, double precision, dimension(lda,*) a, integer lda, double precision, dimension(*) x, integer incx)

DTRMV

الغرض:

!>
!> DTRMV  تقوم بإحدى عمليات المصفوفة والمتجه التالية:
!>
!>    x := A*x,   أو   x := A**T*x,
!>
!> حيث x متجه من n عنصر و A مصفوفة مثلثية علوية أو سفلية، واحدة أو غير واحدة، بحجم n × n.
!>

المعاملات

UPLO 
!>          UPLO هو CHARACTER*1
!>           عند الإدخال، تحدد UPLO ما إذا كانت المصفوفة مثلثية علوية أم سفلية
!>           كما يلي:
!>
!>              UPLO = 'U' أو 'u'   A هي مصفوفة مثلثية علوية.
!>
!>              UPLO = 'L' أو 'l'   A هي مصفوفة مثلثية سفلية.
!>

TRANS

!>          TRANS هو CHARACTER*1
!>           عند الإدخال، تحدد TRANS العملية المراد تنفيذها كما يلي:
!>
!>              TRANS = 'N' أو 'n'   x := A*x.
!>
!>              TRANS = 'T' أو 't'   x := A**T*x.
!>
!>              TRANS = 'C' أو 'c'   x := A**T*x.
!>

DIAG

!>          DIAG هو CHARACTER*1
!>           عند الإدخال، تحدد DIAG ما إذا كانت A واحدة أم لا
!>           مثلثية كما يلي:
!>
!>              DIAG = 'U' أو 'u'   يفترض أن A مثلثية واحدة.
!>
!>              DIAG = 'N' أو 'n'   لا يفترض أن A مثلثية
!>                                  وحدات.
!>

N

!>          N هو INTEGER
!>           عند الإدخال، تحدد N رتبة المصفوفة A.
!>           يجب أن تكون N أكبر من أو تساوي صفر.
!>

A

!>          A هي مصفوفة DOUBLE PRECISION، بأبعاد ( LDA, N )
!>           قبل الإدخال مع UPLO = 'U' أو 'u'، يجب أن تحتوي الجزء العلوي المثلثي
!>           المحدد بحجم n × n من المصفوفة A على المصفوفة المثلثية العلوية
!>           ولا يتم الرجوع إلى الجزء المثلثي السفلي الصارم من A.
!>           قبل الإدخال مع UPLO = 'L' أو 'l'، يجب أن تحتوي الجزء السفلي المثلثي
!>           المحدد بحجم n × n من المصفوفة A على المصفوفة المثلثية السفلية
!>           ولا يتم الرجوع إلى الجزء المثلثي العلوي الصارم من A.
!>           لاحظ أنه عندما يكون DIAG = 'U' أو 'u'، لا يتم الرجوع إلى العناصر القطرية
!>           لـ A أيضًا، ولكن يفترض أنها وحدات.
!>

LDA

!>          LDA هو INTEGER
!>           عند الإدخال، تحدد LDA البعد الأول لـ A كما تم تعريفه
!>           في البرنامج (الفرعي) المستدعي. يجب أن يكون LDA على الأقل
!>           max( 1, n ).
!>

X

!>          X هي مصفوفة DOUBLE PRECISION، بأبعاد لا تقل عن
!>           ( 1 + ( n - 1 )*abs( INCX ) ).
!>           قبل الإدخال، يجب أن تحتوي المصفوفة X المعززة على المتجه x
!>           المكون من n عنصر. عند الخروج، يتم استبدال X
!>           بالمتجه المحول x.
!>

INCX

!>          INCX هو INTEGER
!>           عند الإدخال، تحدد INCX الزيادة للعناصر في
!>           X. يجب ألا تكون INCX صفرًا.
!>
