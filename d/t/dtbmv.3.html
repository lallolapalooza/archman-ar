subroutine dtbmv (character uplo, character trans, character diag, integer n, integer k, double precision, dimension(lda,*) a, integer lda, double precision, dimension(*) x, integer incx)

DTBMV

الغرض:

!>
!> DTBMV  يجري إحدى عمليات المصفوفة والمتجه:
!>
!>    x := A*x,   أو   x := A**T*x,
!>
!> حيث x متجه مكون من n و A مصفوفة مثلثة نطاقية، وحدية أو غير وحدية، علوية أو سفلية،
!> مع ( k + 1 ) أقطار.
!>

المعلمات

UPLO 
!>          UPLO is CHARACTER*1
!>           عند الإدخال، يحدد UPLO ما إذا كانت المصفوفة علوية أم سفلية
!>           على النحو التالي:
!>
!>              UPLO = 'U' or 'u'   A هي مصفوفة مثلثة علوية.
!>
!>              UPLO = 'L' or 'l'   A هي مصفوفة مثلثة سفلية.
!>

TRANS

!>          TRANS is CHARACTER*1
!>           عند الإدخال، يحدد TRANS العملية المراد تنفيذها على النحو
!>           التالي:
!>
!>              TRANS = 'N' or 'n'   x := A*x.
!>
!>              TRANS = 'T' or 't'   x := A**T*x.
!>
!>              TRANS = 'C' or 'c'   x := A**T*x.
!>

DIAG

!>          DIAG is CHARACTER*1
!>           عند الإدخال، يحدد DIAG ما إذا كانت A وحدية أم لا
!>           مثلثة:
!>
!>              DIAG = 'U' or 'u'   يفترض أن A وحدية مثلثة.
!>
!>              DIAG = 'N' or 'n'   لا يفترض أن A وحدية
!>                                  مثلثة.
!>

N

!>          N is INTEGER
!>           عند الإدخال، يحدد N ترتيب المصفوفة A.
!>           يجب أن يكون N على الأقل صفرًا.
!>

K

!>          K is INTEGER
!>           عند الإدخال مع UPLO = 'U' أو 'u'، يحدد K عدد
!>           الأقطار فوق الرئيسية للمصفوفة A.
!>           عند الإدخال مع UPLO = 'L' أو 'l'، يحدد K عدد
!>           الأقطار تحت الرئيسية للمصفوفة A.
!>           يجب أن يحقق K  0 .le. K.
!>

A

!>          A is DOUBLE PRECISION array, dimension ( LDA, N )
!>           قبل الإدخال مع UPLO = 'U' أو 'u'، يجب أن يحتوي الجزء
!>           ( k + 1 ) بواسطة n الرائد من المصفوفة A على الجزء
!>           المثلث العلوي النطاقي من مصفوفة المعاملات، المقدمة عمودًا
!>           بعمود، مع القطر الرئيسي للمصفوفة في الصف
!>           ( k + 1 ) من المصفوفة، ويبدأ القطر الفوقي الأول من الموضع
!>           2 في الصف k، وهكذا. لا تتم الإشارة إلى الجزء العلوي الأيسر
!>           k × k من المصفوفة A.
!>           سيقوم مقطع البرنامج التالي بنقل مصفوفة نطاقية مثلثة علوية
!>           من تخزين المصفوفة الكاملة التقليدية إلى تخزين النطاق:
!>
!>                 DO 20, J = 1, N
!>                    M = K + 1 - J
!>                    DO 10, I = MAX( 1, J - K ), J
!>                       A( M + I, J ) = matrix( I, J )
!>              10    CONTINUE
!>              20 CONTINUE
!>
!>           قبل الإدخال مع UPLO = 'L' أو 'l'، يجب أن يحتوي الجزء
!>           ( k + 1 ) بواسطة n الرائد من المصفوفة A على الجزء
!>           المثلث السفلي النطاقي من مصفوفة المعاملات، المقدمة عمودًا
!>           بعمود، مع القطر الرئيسي للمصفوفة في الصف 1
!>           من المصفوفة، ويبدأ القطر السفلي الأول من الموضع 1 في
!>           الصف 2، وهكذا. لا تتم الإشارة إلى الجزء السفلي الأيمن
!>           k × k من المصفوفة A.
!>           سيقوم مقطع البرنامج التالي بنقل مصفوفة نطاقية مثلثة سفلية
!>           من تخزين المصفوفة الكاملة التقليدية إلى تخزين النطاق:
!>
!>                 DO 20, J = 1, N
!>                    M = 1 - J
!>                    DO 10, I = J, MIN( N, J + K )
!>                       A( M + I, J ) = matrix( I, J )
!>              10    CONTINUE
!>              20 CONTINUE
!>
!>           لاحظ أنه عندما يكون DIAG = 'U' أو 'u'، فإن عناصر المصفوفة A
!>           المقابلة للعناصر القطرية للمصفوفة لا تتم الإشارة إليها،
!>           ولكن يفترض أنها الوحدة.
!>

LDA

!>          LDA is INTEGER
!>           عند الإدخال، يحدد LDA البعد الأول لـ A كما تم تعريفه
!>           في البرنامج (الفرعي) المستدعي. يجب أن يكون LDA على الأقل
!>           ( k + 1 ).
!>

X

!>          X is DOUBLE PRECISION array, dimension at least
!>           ( 1 + ( n - 1 )*abs( INCX ) ).
!>           قبل الإدخال، يجب أن يحتوي المتجه X المتزايد على المتجه x
!>           المكون من n عنصر. عند الخروج، يتم استبدال X
!>           بالمتجه x المحول.
!>

INCX

!>          INCX is INTEGER
!>           عند الإدخال، يحدد INCX الزيادة لعناصر X. يجب أن لا يكون INCX صفرًا.
!>
