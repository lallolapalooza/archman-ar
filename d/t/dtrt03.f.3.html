توثيق الوظيفة/الدالة الفرعية

subroutine dtrt03 (character uplo, character trans, character diag, integer n, integer nrhs, double precision, dimension( lda, * ) a, integer lda, double precision scale, double precision, dimension( * ) cnorm, double precision tscal, double precision, dimension( ldx, * ) x, integer ldx, double precision, dimension( ldb, * ) b, integer ldb, double precision, dimension( * ) work, double precision resid)

DTRT03

الغرض:

!>
!> DTRT03 يحسب الباقي لحل نظام معادلات مثلثي محسوب المقياس A*x = s*b  أو  A'*x = s*b.
!> هنا A مصفوفة مثلثية، A' هي منقولة A، s هو عامل قياسي، و x و b مصفوفتان N × NRHS.
!> نسبة الاختبار هي القيمة القصوى على عدد الأطراف اليمنى لـ
!>    norm(s*b - op(A)*x) / ( norm(op(A)) * norm(x) * EPS ),
!> حيث op(A) تعني A أو A' و EPS هو جزء من الألفاظ الآلية.
!>

المعلمات

UPLO 
!>          UPLO هو CHARACTER*1
!>          يحدد ما إذا كانت المصفوفة A مثلثية علوية أم سفلية.
!>          = 'U':  مثلثية علوية
!>          = 'L':  مثلثية سفلية
!>

TRANS

!>          TRANS هو CHARACTER*1
!>          يحدد العملية المطبقة على A.
!>          = 'N':  A *x = s*b  (بدون منقول)
!>          = 'T':  A'*x = s*b  (منقول)
!>          = 'C':  A'*x = s*b  (منقول مترافق = منقول)
!>

DIAG

!>          DIAG هو CHARACTER*1
!>          يحدد ما إذا كانت المصفوفة A مثلثية موحدة أم لا.
!>          = 'N':  غير موحدة مثلثية
!>          = 'U':  موحدة مثلثية
!>

N

!>          N هو INTEGER
!>          رتبة المصفوفة A.  N >= 0.
!>

NRHS

!>          NRHS هو INTEGER
!>          عدد الأطراف اليمنى، أي عدد أعمدة المصفوفتين X و B.  NRHS >= 0.
!>

A

!>          A هو DOUBLE PRECISION array، البعد (LDA,N)
!>          المصفوفة المثلثية A.  إذا كان UPLO = 'U'، فإن الجزء العلوي المثلثي
!>          الرئيسي N × N في المصفوفة A يحتوي على المصفوفة المثلثية العلوية،
!>          ولا يتم الرجوع إلى الجزء المثلثي السفلي الصارم لـ A.
!>          إذا كان UPLO = 'L'، فإن الجزء السفلي المثلثي الرئيسي N × N في
!>          المصفوفة A يحتوي على المصفوفة المثلثية السفلية، ولا يتم الرجوع إلى
!>          الجزء المثلثي العلوي الصارم لـ A.
!>          إذا كان DIAG = 'U'، فلا يتم الرجوع إلى عناصر القطر في A ويُفترض أنها 1.
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرئيسي للمصفوفة A.  LDA >= max(1,N).
!>

SCALE

!>          SCALE هو DOUBLE PRECISION
!>          عامل القياس s المستخدم في حل النظام المثلثي.
!>

CNORM

!>          CNORM هو DOUBLE PRECISION array، البعد (N)
!>          القيمة المطلقة لـ 1-norms لأعمدة A، باستثناء القطر.
!>

TSCAL

!>          TSCAL هو DOUBLE PRECISION
!>          عامل القياس المستخدم في حساب 1-norms في CNORM.
!>          CNORM يحتوي فعليًا على قيم norms أعمدة TSCAL*A.
!>

X

!>          X هو DOUBLE PRECISION array، البعد (LDX,NRHS)
!>          متجهات الحل المحسوبة لنظام المعادلات الخطية.
!>

LDX

!>          LDX هو INTEGER
!>          البعد الرئيسي للمصفوفة X.  LDX >= max(1,N).
!>

B

!>          B هو DOUBLE PRECISION array، البعد (LDB,NRHS)
!>          متجهات الطرف الأيمن لنظام المعادلات الخطية.
!>

LDB

!>          LDB هو INTEGER
!>          البعد الرئيسي للمصفوفة B.  LDB >= max(1,N).
!>

WORK

!>          WORK هو DOUBLE PRECISION array، البعد (N)
!>

RESID

!>          RESID هو DOUBLE PRECISION
!>          القيمة القصوى على عدد الأطراف اليمنى لـ
!>          norm(op(A)*x - s*b) / ( norm(op(A)) * norm(x) * EPS ).
!>
