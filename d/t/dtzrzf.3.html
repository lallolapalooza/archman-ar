توثيق الوظيفة/الروتين الفرعي

subroutine dtzrzf (integer m, integer n, double precision, dimension( lda, * ) a, integer lda, double precision, dimension( * ) tau, double precision, dimension( * ) work, integer lwork, integer info)

DTZRZF

الغرض:

!>
!> يقوم DTZRZF بتحويل مصفوفة A شبه مثلثة علوية حقيقية من حجم M-by-N ( M<=N )
!> إلى شكل مثلثي علوي بواسطة تحويلات متعامدة.
!>
!> يتم تحليل المصفوفة شبه المثلثية العلوية A إلى
!>
!>    A = ( R  0 ) * Z,
!>
!> حيث Z هي مصفوفة متعامدة N-by-N و R هي مصفوفة مثلثية علوية M-by-M.
!>

المعلمات

M 
!>          M هو عدد صحيح (INTEGER)
!>          عدد صفوف المصفوفة A. M >= 0.
!>

N

!>          N هو عدد صحيح (INTEGER)
!>          عدد أعمدة المصفوفة A. N >= M.
!>

A

!>          A هي مصفوفة ذات دقة مزدوجة (DOUBLE PRECISION)، ذات أبعاد (LDA,N)
!>          عند الدخول، يجب أن يحتوي الجزء العلوي شبه المثلثي من المصفوفة A بأبعاد M-by-N
!>          على المصفوفة المراد تحليلها.
!>          عند الخروج، يحتوي الجزء العلوي المثلثي من A بأبعاد M-by-M على
!>          المصفوفة المثلثية العلوية R، والعناصر من M+1 إلى N في الصفوف M الأولى من A،
!>          مع المصفوفة TAU، تمثل المصفوفة المتعامدة Z كناتج لمُحولات أساسية M.
!>

LDA

!>          LDA هو عدد صحيح (INTEGER)
!>          البعد الرئيسي للمصفوفة A. LDA >= max(1,M).
!>

TAU

!>          TAU هي مصفوفة ذات دقة مزدوجة (DOUBLE PRECISION)، ذات أبعاد (M)
!>          العوامل القياسية للمُحولات الأساسية.
!>

WORK

!>          WORK هي مصفوفة ذات دقة مزدوجة (DOUBLE PRECISION)، ذات أبعاد (MAX(1,LWORK))
!>          عند الخروج، إذا كان INFO = 0، يعيد WORK(1) قيمة LWORK المثلى.
!>

LWORK

!>          LWORK هو عدد صحيح (INTEGER)
!>          بعد المصفوفة WORK. LWORK >= max(1,M).
!>          للأداء الأمثل LWORK >= M*NB، حيث NB
!>          هو حجم الكتلة الأمثل.
!>
!>          إذا كان LWORK = -1، يفترض أن هذا استعلام مساحة عمل؛ يقوم الروتين
!>          فقط بحساب الحجم الأمثل للمصفوفة WORK، ويعيد
!>          هذه القيمة كإدخال أول في المصفوفة WORK، ولا يتم إصدار أي رسالة خطأ
!>          متعلقة بـ LWORK بواسطة XERBLA.
!>

INFO

!>          INFO هو عدد صحيح (INTEGER)
!>          = 0:  انتهاء ناجح
!>          < 0:  إذا كان INFO = -i، فإن الوسيط i كان له قيمة غير قانونية
!>

تفاصيل إضافية:

!>
!>  يمكن حساب المصفوفة Z بحجم N-by-N بواسطة
!>
!>     Z =  Z(1)*Z(2)* ... *Z(M)
!>
!>  حيث يتم إعطاء كل Z(k) بحجم N-by-N بواسطة
!>
!>     Z(k) = I - tau(k)*v(k)*v(k)**T
!>
!>  مع v(k) هو متجه الصف k للمصفوفة M-by-N
!>
!>     V = ( I   A(:,M+1:N) )
!>
!>  I هي مصفوفة الوحدة M-by-M، A(:,M+1:N)
!>  هي المخرجات المخزنة في A عند الخروج من DTZRZF،
!>  و tau(k) هو العنصر k في المصفوفة TAU.
!>
!>
