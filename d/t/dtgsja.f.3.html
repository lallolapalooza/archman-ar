subroutine dtgsja (jobu، jobv، jobq، m، p، n، k، l، a، lda، b، ldb، tola، tolb، alpha، beta، u، ldu، v، ldv، q، ldq، work، ncycle، info)

DTGSJA

الغرض:

!>
!> DTGSJA تحسب التحلل المتفرد المعمم (GSVD)
!> لمصفوفتين مثلثيتين علويتين (أو شبه منحرف) حقيقيتين A و B.
!>
!> عند الدخول، يُفترض أن للمصفوفتين A و B الأشكال التالية،
!> والتي قد يتم الحصول عليها بواسطة الروتين الفرعي للمعالجة المسبقة DGGSVP
!> من مصفوفة عامة M-by-N A ومصفوفة P-by-N B:
!>
!>              N-K-L  K    L
!>    A =    K ( 0    A12  A13 ) إذا كانت M-K-L >= 0؛
!>           L ( 0     0   A23 )
!>       M-K-L ( 0     0    0  )
!>
!>            N-K-L  K    L
!>    A =  K ( 0    A12  A13 ) إذا كانت M-K-L < 0؛
!>       M-K ( 0     0   A23 )
!>
!>            N-K-L  K    L
!>    B =  L ( 0     0   B13 )
!>       P-L ( 0     0    0  )
!>
!> حيث مصفوفة A12 بحجم K-by-K ومصفوفة B13 بحجم L-by-L غير مفردة
!> مثلثية علوية؛ A23 مثلثية علوية بحجم L-by-L إذا كانت M-K-L >= 0،
!> وإلا فإن A23 هي شبه منحرف علوية بحجم (M-K)-by-L.
!>
!> عند الخروج،
!>
!>        U**T *A*Q = D1*( 0 R ),    V**T *B*Q = D2*( 0 R ),
!>
!> حيث U، V، و Q هي مصفوفات متعامدة.
!> R هي مصفوفة مثلثية علوية غير مفردة، و D1 و D2 هي
!> مصفوفات "قطرية"، وهي ذات هياكل كالتالي:
!>
!> إذا كانت M-K-L >= 0،
!>
!>                     K  L
!>        D1 =     K ( I  0 )
!>                 L ( 0  C )
!>             M-K-L ( 0  0 )
!>
!>                   K  L
!>        D2 = L   ( 0  S )
!>             P-L ( 0  0 )
!>
!>                N-K-L  K    L
!>   ( 0 R ) = K (  0   R11  R12 ) K
!>             L (  0    0   R22 ) L
!>
!> حيث
!>
!>   C = diag( ALPHA(K+1), ... , ALPHA(K+L) )،
!>   S = diag( BETA(K+1),  ... , BETA(K+L) )،
!>   C**2 + S**2 = I.
!>
!>   يتم تخزين R في A(1:K+L,N-K-L+1:N) عند الخروج.
!>
!> إذا كانت M-K-L < 0،
!>
!>                K M-K K+L-M
!>     D1 =   K ( I  0    0   )
!>          M-K ( 0  C    0   )
!>
!>                  K M-K K+L-M
!>     D2 =   M-K ( 0  S    0   )
!>          K+L-M ( 0  0    I   )
!>            P-L ( 0  0    0   )
!>
!>                N-K-L  K   M-K  K+L-M
!> ( 0 R ) =    K ( 0    R11  R12  R13  )
!>           M-K ( 0     0   R22  R23  )
!>         K+L-M ( 0     0    0   R33  )
!>
!> حيث
!> C = diag( ALPHA(K+1), ... , ALPHA(M) )،
!> S = diag( BETA(K+1),  ... , BETA(M) )،
!> C**2 + S**2 = I.
!>
!> يتم تخزين R = ( R11 R12 R13 ) في A(1:M, N-K-L+1:N) والجزء R33
!>     (  0  R22 R23 )
!> في B(M-K+1:L,N+M-K-L+1:N) عند الخروج.
!>
!> حساب المصفوفات التحويلية المتعامدة U، V أو Q اختياري.
!> قد يتم تشكيل هذه المصفوفات صراحةً، أو قد يتم ضربها في المصفوفات المدخلة
!> U1، V1، أو Q1.
!>

المعلمات

JOBU 
!>          JOBU is CHARACTER*1
!>          = 'U':  يجب أن تحتوي U على مصفوفة متعامدة U1 عند الدخول،
!>                  ويتم إرجاع الناتج U1*U؛
!>          = 'I':  يتم تهيئة U إلى مصفوفة الوحدة،
!>                  ويتم إرجاع المصفوفة المتعامدة U؛
!>          = 'N':  لا يتم حساب U.
!>

JOBV

!>          JOBV is CHARACTER*1
!>          = 'V':  يجب أن تحتوي V على مصفوفة متعامدة V1 عند الدخول،
!>                  ويتم إرجاع الناتج V1*V؛
!>          = 'I':  يتم تهيئة V إلى مصفوفة الوحدة،
!>                  ويتم إرجاع المصفوفة المتعامدة V؛
!>          = 'N':  لا يتم حساب V.
!>

JOBQ

!>          JOBQ is CHARACTER*1
!>          = 'Q':  يجب أن تحتوي Q على مصفوفة متعامدة Q1 عند الدخول،
!>                  ويتم إرجاع الناتج Q1*Q؛
!>          = 'I':  يتم تهيئة Q إلى مصفوفة الوحدة،
!>                  ويتم إرجاع المصفوفة المتعامدة Q؛
!>          = 'N':  لا يتم حساب Q.
!>

M

!>          M is INTEGER
!>          عدد صفوف المصفوفة A.  M >= 0.
!>

P

!>          P is INTEGER
!>          عدد صفوف المصفوفة B.  P >= 0.
!>

N

!>          N is INTEGER
!>          عدد أعمدة المصفوفتين A و B.  N >= 0.
!>

K

!>          K is INTEGER
!>

L

!>          L is INTEGER
!>
!>          تحدد K و L الكتل الفرعية في المصفوفتين المدخلتين A و B:
!>          A23 = A(K+1:MIN(K+L,M),N-L+1:N) و B13 = B(1:L,N-L+1:N)
!>          من A و B، والتي سيتم حساب GSVD الخاصة بها بواسطة DTGSJA.
!>          انظر التفاصيل الإضافية.
!>

A

!>          A is DOUBLE PRECISION array, dimension (LDA,N)
!>          عند الدخول، المصفوفة A بحجم M-by-N.
!>          عند الخروج، A(N-K+1:N,1:MIN(K+L,M) ) تحتوي على المصفوفة
!>          المثلثية R أو جزء من R. انظر الغرض للتفاصيل.
!>

LDA

!>          LDA is INTEGER
!>          البعد الرئيسي للمصفوفة A. LDA >= max(1,M).
!>

B

!>          B is DOUBLE PRECISION array, dimension (LDB,N)
!>          عند الدخول، المصفوفة B بحجم P-by-N.
!>          عند الخروج، إذا لزم الأمر، B(M-K+1:L,N+M-K-L+1:N) تحتوي على
!>          جزء من R. انظر الغرض للتفاصيل.
!>

LDB

!>          LDB is INTEGER
!>          البعد الرئيسي للمصفوفة B. LDB >= max(1,P).
!>

TOLA

!>          TOLA is DOUBLE PRECISION
!>

TOLB

!>          TOLB is DOUBLE PRECISION
!>
!>          TOLA و TOLB هما معايير التقارب لإجراء التكرار Jacobi-
!>          Kogbetliantz. بشكل عام، هما نفس المستخدمين في خطوة المعالجة المسبقة،
!>          على سبيل المثال
!>              TOLA = max(M,N)*norm(A)*MAZHEPS,
!>              TOLB = max(P,N)*norm(B)*MAZHEPS.
!>

ALPHA

!>          ALPHA is DOUBLE PRECISION array, dimension (N)
!>

BETA

!>          BETA is DOUBLE PRECISION array, dimension (N)
!>
!>          عند الخروج، تحتوي ALPHA و BETA على أزواج القيم المتفردة
!>          المعممة للمصفوفتين A و B؛
!>            ALPHA(1:K) = 1،
!>            BETA(1:K)  = 0،
!>          وإذا كانت M-K-L >= 0،
!>            ALPHA(K+1:K+L) = diag(C)،
!>            BETA(K+1:K+L)  = diag(S)،
!>          أو إذا كانت M-K-L < 0،
!>            ALPHA(K+1:M)= C, ALPHA(M+1:K+L)= 0
!>            BETA(K+1:M) = S, BETA(M+1:K+L) = 1.
!>          علاوة على ذلك، إذا كانت K+L < N،
!>            ALPHA(K+L+1:N) = 0 و
!>            BETA(K+L+1:N)  = 0.
!>

U

!>          U is DOUBLE PRECISION array, dimension (LDU,M)
!>          عند الدخول، إذا كانت JOBU = 'U'، فإن U
