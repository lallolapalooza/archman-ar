subroutine dtrt03 (character uplo, character trans, character diag, integer n, integer nrhs, double precision, dimension( lda, * ) a, integer lda, double precision scale, double precision, dimension( * ) cnorm, double precision tscal, double precision, dimension( ldx, * ) x, integer ldx, double precision, dimension( ldb, * ) b, integer ldb, double precision, dimension( * ) work, double precision resid)

DTRT03

الغرض:

!>
!> DTRT03 يحسب الباقي لحل نظام المعادلات المثلثية المُقاس A*x = s*b  أو  A'*x = s*b.
!> هنا A هي مصفوفة مثلثية، A' هي منقولة A، s هو ثابت عددي، و x و b هما مصفوفتان N × NRHS.  نسبة الاختبار هي
!> الحد الأقصى على عدد الأطراف اليمنى لـ
!>    norm(s*b - op(A)*x) / ( norm(op(A)) * norm(x) * EPS ),
!> حيث op(A) تعني A أو A' و EPS هو معامل الآلة.
!>

المعاملات

UPLO 
!>          UPLO is CHARACTER*1
!>          يحدد ما إذا كانت المصفوفة A مثلثية علوية أو سفلية.
!>          = 'U':  مثلثية علوية
!>          = 'L':  مثلثية سفلية
!>

TRANS

!>          TRANS is CHARACTER*1
!>          يحدد العملية المطبقة على A.
!>          = 'N':  A *x = s*b  (بدون منقول)
!>          = 'T':  A'*x = s*b  (منقول)
!>          = 'C':  A'*x = s*b  (منقول مترافق = منقول)
!>

DIAG

!>          DIAG is CHARACTER*1
!>          يحدد ما إذا كانت المصفوفة A مثلثية أحادية أم لا.
!>          = 'N':  غير أحادية
!>          = 'U':  أحادية
!>

N

!>          N is INTEGER
!>          رتبة المصفوفة A.  N >= 0.
!>

NRHS

!>          NRHS is INTEGER
!>          عدد الأطراف اليمنى، أي عدد أعمدة المصفوفتين X و B.  NRHS >= 0.
!>

A

!>          A is DOUBLE PRECISION array, dimension (LDA,N)
!>          المصفوفة المثلثية A.  إذا كان UPLO = 'U'، فإن الجزء العلوي المثلثي الرئيسي للمصفوفة A ببعد n × n يحتوي على المصفوفة المثلثية العلوية، والجزء السفلي المثلثي الصارم للمصفوفة A لا تتم الإشارة إليه.  إذا كان UPLO = 'L'، فإن الجزء السفلي المثلثي الرئيسي للمصفوفة A ببعد n × n يحتوي على المصفوفة المثلثية السفلية، والجزء العلوي المثلثي الصارم للمصفوفة A لا تتم الإشارة إليه.  إذا كان DIAG = 'U'، فإن العناصر القطرية لـ A لا تتم الإشارة إليها أيضًا ويُفترض أنها 1.
!>

LDA

!>          LDA is INTEGER
!>          البعد الرئيسي للمصفوفة A.  LDA >= max(1,N).
!>

SCALE

!>          SCALE is DOUBLE PRECISION
!>          عامل القياس s المستخدم في حل النظام المثلثي.
!>

CNORM

!>          CNORM is DOUBLE PRECISION array, dimension (N)
!>          معايير الأعمدة للمصفوفة A، لا تشمل القطر.
!>

TSCAL

!>          TSCAL is DOUBLE PRECISION
!>          عامل القياس المستخدم في حساب معايير الأعمدة في CNORM.
!>          CNORM يحتوي بالفعل على معايير الأعمدة لـ TSCAL*A.
!>

X

!>          X is DOUBLE PRECISION array, dimension (LDX,NRHS)
!>          متجهات الحل المحسوبة لأنظمة المعادلات الخطية.
!>

LDX

!>          LDX is INTEGER
!>          البعد الرئيسي للمصفوفة X.  LDX >= max(1,N).
!>

B

!>          B is DOUBLE PRECISION array, dimension (LDB,NRHS)
!>          متجهات الطرف الأيمن لأنظمة المعادلات الخطية.
!>

LDB

!>          LDB is INTEGER
!>          البعد الرئيسي للمصفوفة B.  LDB >= max(1,N).
!>

WORK

!>          WORK is DOUBLE PRECISION array, dimension (N)
!>

RESID

!>          RESID is DOUBLE PRECISION
!>          الحد الأقصى على عدد الأطراف اليمنى لـ
!>          norm(op(A)*x - s*b) / ( norm(op(A)) * norm(x) * EPS ).
!>
