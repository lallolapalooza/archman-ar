توثيق الوظيفة/الروتين الفرعي

subroutine dtpt05 (character uplo, character trans, character diag, integer n, integer nrhs, double precision, dimension( * ) ap, double precision, dimension( ldb, * ) b, integer ldb, double precision, dimension( ldx, * ) x, integer ldx, double precision, dimension( ldxact, * ) xact, integer ldxact, double precision, dimension( * ) ferr, double precision, dimension( * ) berr, double precision, dimension( * ) reslts)

DTPT05

الغرض:

!>
!> DTPT05 تختبر حدود الخطأ من التنقيح التكراري للحل المحسوب لنظام المعادلات A*X = B،
!> حيث A مصفوفة مثلثية بتنسيق تخزين مدمج.
!>
!> RESLTS(1) = اختبار حد الخطأ
!>           = norm(X - XACT) / ( norm(X) * FERR )
!>
!> تُرجع قيمة كبيرة إذا لم تكن هذه النسبة أقل من واحد.
!>
!> RESLTS(2) = الباقي من روتين التنقيح التكراري
!>           = الحد الأقصى لـ BERR / ( (n+1)*EPS + (*) )، حيث
!>             (*) = (n+1)*UNFL / (min_i (abs(A)*abs(X) +abs(b))_i )
!>

المعلمات

UPLO 
!>          UPLO هو CHARACTER*1
!>          يحدد ما إذا كانت المصفوفة A علوية أم سفلية.
!>          = 'U':  مثلثية علوية
!>          = 'L':  مثلثية سفلية
!>

TRANS

!>          TRANS هو CHARACTER*1
!>          يحدد شكل نظام المعادلات.
!>          = 'N':  A * X = B  (بدون تبديل)
!>          = 'T':  A'* X = B  (تبديل)
!>          = 'C':  A'* X = B  (تبديل مرافق = تبديل)
!>

DIAG

!>          DIAG هو CHARACTER*1
!>          يحدد ما إذا كانت المصفوفة A مثلثية وحدة أم لا.
!>          = 'N':  غير مثلثية وحدة
!>          = 'U':  مثلثية وحدة
!>

N

!>          N هو INTEGER
!>          عدد صفوف المصفوفات X، B، و XACT، وترتيب المصفوفة A. N >= 0.
!>

NRHS

!>          NRHS هو INTEGER
!>          عدد أعمدة المصفوفات X، B، و XACT. NRHS >= 0.
!>

AP

!>          AP هو مصفوفة DOUBLE PRECISION، البعد (N*(N+1)/2)
!>          المصفوفة المثلثية العلوية أو السفلية A، مدمجة عمودياً في
!>          مصفوفة خطية. يتم تخزين العمود j من A في المصفوفة
!>          AP بالشكل التالي:
!>          إذا كان UPLO = 'U'، فإن AP(i + (j-1)*j/2) = A(i,j) لـ 1<=i<=j؛
!>          إذا كان UPLO = 'L'، فإن AP(i + (j-1)*(2n-j)/2) = A(i,j) لـ j<=i<=n.
!>          إذا كان DIAG = 'U'، فإن العناصر القطرية لـ A لا يتم الرجوع إليها
!>          ويفترض أنها 1.
!>

B

!>          B هي مصفوفة DOUBLE PRECISION، البعد (LDB,NRHS)
!>          متجهات الطرف الأيمن لنظام المعادلات الخطية.
!>

LDB

!>          LDB هو INTEGER
!>          البعد الرئيسي للمصفوفة B. LDB >= max(1,N).
!>

X

!>          X هي مصفوفة DOUBLE PRECISION، البعد (LDX,NRHS)
!>          متجهات الحل المحسوبة. يتم تخزين كل متجه كعمود في المصفوفة X.
!>

LDX

!>          LDX هو INTEGER
!>          البعد الرئيسي للمصفوفة X. LDX >= max(1,N).
!>

XACT

!>          XACT هي مصفوفة DOUBLE PRECISION، البعد (LDX,NRHS)
!>          متجهات الحل الدقيقة. يتم تخزين كل متجه كعمود في المصفوفة XACT.
!>

LDXACT

!>          LDXACT هو INTEGER
!>          البعد الرئيسي للمصفوفة XACT. LDXACT >= max(1,N).
!>

FERR

!>          FERR هي مصفوفة DOUBLE PRECISION، البعد (NRHS)
!>          حدود الخطأ الأمامي المقدرة لكل متجه حل X. إذا كان XTRUE هو الحل الحقيقي،
!>          فإن FERR يحد من قيمة أكبر إدخال في (X - XTRUE) مقسومة على
!>          قيمة أكبر إدخال في X.
!>

BERR

!>          BERR هي مصفوفة DOUBLE PRECISION، البعد (NRHS)
!>          الخطأ الرجعي النسبي المكون لكل متجه حل (أي، أصغر تغيير نسبي في أي إدخال
!>          لـ A أو B يجعل X حلاً دقيقًا).
!>

RESLTS

!>          RESLTS هي مصفوفة DOUBLE PRECISION، البعد (2)
!>          القيمة القصوى لمتجهات الحل NRHS للنسب التالية:
!>          RESLTS(1) = norm(X - XACT) / ( norm(X) * FERR )
!>          RESLTS(2) = BERR / ( (n+1)*EPS + (*) )
!>
