subroutine dtrt02 (character uplo, character trans, character diag, integer n, integer nrhs, double precision, dimension( lda, * ) a, integer lda, double precision, dimension( ldx, * ) x, integer ldx, double precision, dimension( ldb, * ) b, integer ldb, double precision, dimension( * ) work, double precision resid)

DTRT02

الغرض:

!>
!> DTRT02 يحسب الباقي للحل المحسوب لنظام المعادلات الخطية المثلثية op(A)*X = B، حيث A هي
!> مصفوفة مثلثية. نسبة الاختبار هي الحد الأقصى على
!>    norm(b - op(A)*x) / ( ||op(A)||_1 * norm(x) * EPS )،
!> حيث op(A) = A أو A**T، b هو عمود B، x هو متجه الحل، و EPS هو دقة الآلة.
!> المعيار المستخدم هو المعيار 1.
!>

المعاملات

UPLO 
!>          UPLO هو CHARACTER*1
!>          يحدد ما إذا كانت المصفوفة A علوية أم سفلية مثلثية.
!>          = 'U':  علوية مثلثية
!>          = 'L':  سفلية مثلثية
!>

TRANS

!>          TRANS هو CHARACTER*1
!>          يحدد العملية المطبقة على A.
!>          = 'N':  A    * X = B  (بدون تبديل)
!>          = 'T':  A**T * X = B  (تبديل)
!>          = 'C':  A**H * X = B  (تبديل مرافق = تبديل)
!>

DIAG

!>          DIAG هو CHARACTER*1
!>          يحدد ما إذا كانت المصفوفة A وحدوية مثلثية أم لا.
!>          = 'N':  غير وحدوية مثلثية
!>          = 'U':  وحدوية مثلثية
!>

N

!>          N هو INTEGER
!>          رتبة المصفوفة A. N >= 0.
!>

NRHS

!>          NRHS هو INTEGER
!>          عدد الجوانب اليمنى، أي عدد أعمدة
!>          المصفوفتين X و B. NRHS >= 0.
!>

A

!>          A هو مصفوفة DOUBLE PRECISION، البعد (LDA,N)
!>          المصفوفة المثلثية A. إذا كان UPLO = 'U'، فإن الجزء العلوي المثلثي
!>          المتجه n × n من المصفوفة A يحتوي على المصفوفة المثلثية العلوية،
!>          ولا يتم الرجوع إلى الجزء السفلي المثلثي الصارم لـ A. إذا كان UPLO = 'L'،
!>          فإن الجزء السفلي المثلثي المتجه n × n من المصفوفة A يحتوي على
!>          المصفوفة المثلثية السفلية، ولا يتم الرجوع إلى الجزء العلوي المثلثي
!>          الصارم لـ A. إذا كان DIAG = 'U'، فإن العناصر القطرية لـ A
!>          لا يتم الرجوع إليها أيضًا ويُفترض أنها 1.
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرائد للمصفوفة A. LDA >= max(1,N).
!>

X

!>          X هي مصفوفة DOUBLE PRECISION، البعد (LDX,NRHS)
!>          متجهات الحل المحسوبة لنظام المعادلات الخطية.
!>

LDX

!>          LDX هو INTEGER
!>          البعد الرائد للمصفوفة X. LDX >= max(1,N).
!>

B

!>          B هي مصفوفة DOUBLE PRECISION، البعد (LDB,NRHS)
!>          متجهات الجانب الأيمن لنظام المعادلات الخطية.
!>

LDB

!>          LDB هو INTEGER
!>          البعد الرائد للمصفوفة B. LDB >= max(1,N).
!>

WORK

!>          WORK هي مصفوفة DOUBLE PRECISION، البعد (N)
!>

RESID

!>          RESID هو DOUBLE PRECISION
!>          الحد الأقصى على عدد الجوانب اليمنى لـ
!>          norm(op(A)*X - B) / ( norm(op(A)) * norm(X) * EPS ).
!>
