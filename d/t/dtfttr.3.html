subroutine dtfttr (transr, uplo, n, arf, a, lda, info)

DTFTTR ينسخ مصفوفة مثلثية من التنسيق المعبأ بالكامل المستطيل (TF) إلى التنسيق الكامل القياسي (TR).

توثيق الوظيفة/الإجراء الفرعي

subroutine dtfttr (character transr, character uplo, integer n, double precision, dimension( 0: * ) arf, double precision, dimension( 0: lda-1, 0: * ) a, integer lda, integer info)

DTFTTR ينسخ مصفوفة مثلثية A من التنسيق المعبأ بالكامل المستطيل (TF) إلى التنسيق الكامل القياسي (TR).

الغرض:

!>
!> DTFTTR ينسخ مصفوفة مثلثية A من التنسيق المعبأ بالكامل المستطيل (TF) إلى التنسيق الكامل القياسي (TR).
!>

المعلمات

TRANSR 
!>          TRANSR هو CHARACTER*1
!>          = 'N':  ARF في تنسيق عادي؛
!>          = 'T':  ARF في تنسيق منقول.
!>

UPLO

!>          UPLO هو CHARACTER*1
!>          = 'U':  A مثلثية عليا؛
!>          = 'L':  A مثلثية سفلى.
!>

N

!>          N هو INTEGER
!>          رتبة المصفوفات ARF و A. N >= 0.
!>

ARF

!>          ARF هو مصفوفة DOUBLE PRECISION، البعد (N*(N+1)/2).
!>          عند الدخول، الجزء العلوي (إذا كان UPLO = 'U') أو السفلي (إذا كان UPLO = 'L')
!>          من المصفوفة A بتنسيق RFP. انظر أدناه لمزيد من التفاصيل.
!>

A

!>          A هو مصفوفة DOUBLE PRECISION، البعد (LDA,N)
!>          عند الخروج، المصفوفة المثلثية A. إذا كان UPLO = 'U'، فإن
!>          الجزء العلوي المثلثي N-by-N الرائد من المصفوفة A يحتوي على
!>          المصفوفة المثلثية العليا، ولا يتم الرجوع إلى الجزء المثلثي السفلي البحت
!>          من A. إذا كان UPLO = 'L'، فإن
!>          الجزء السفلي المثلثي N-by-N الرائد من المصفوفة A يحتوي على
!>          المصفوفة المثلثية السفلى، ولا يتم الرجوع إلى الجزء المثلثي العلوي البحت
!>          من A.
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرائد للمصفوفة A. LDA >= max(1,N).
!>

INFO

!>          INFO هو INTEGER
!>          = 0:  انتهاء ناجح
!>          < 0:  إذا كان INFO = -i، فقد كانت المعلمة i ذات قيمة غير قانونية
!>

تفاصيل إضافية:

!>
!>  ننظر أولاً إلى تنسيق التعبئة الكاملة المستطيلة (RFP) عندما تكون N
!>  زوجية. نقدم مثالًا حيث N = 6.
!>
!>      AP علوية             AP سفلية
!>
!>   00 01 02 03 04 05       00
!>      11 12 13 14 15       10 11
!>         22 23 24 25       20 21 22
!>            33 34 35       30 31 32 33
!>               44 45       40 41 42 43 44
!>                  55       50 51 52 53 54 55
!>
!>
!>  إذا كان TRANSR = 'N'. تحتفظ RFP بـ AP كما يلي:
!>  بالنسبة لـ UPLO = 'U'، يتكون شبه المنحرف العلوي A(0:5,0:2) من الأعمدة الثلاثة الأخيرة
!>  من AP العلوي. يتكون المثلث السفلي A(4:6,0:2) من
!>  منقول الأعمدة الثلاثة الأولى من AP العلوي.
!>  بالنسبة لـ UPLO = 'L'، يتكون شبه المنحرف السفلي A(1:6,0:2) من الأعمدة الثلاثة الأولى
!>  من AP السفلي. يتكون المثلث العلوي A(0:2,0:2) من
!>  منقول الأعمدة الثلاثة الأخيرة من AP السفلي.
!>  هذا يغطي الحالة N زوجية و TRANSR = 'N'.
!>
!>         RFP A                   RFP A
!>
!>        03 04 05                33 43 53
!>        13 14 15                00 44 54
!>        23 24 25                10 11 55
!>        33 34 35                20 21 22
!>        00 44 45                30 31 32
!>        01 11 55                40 41 42
!>        02 12 22                50 51 52
!>
!>
!>  الآن إذا كان TRANSR = 'T'. RFP A في كلتا حالتي UPLO هو فقط
!>  منقول RFP A أعلاه. لذلك يحصل المرء على:
!>
!>
!>           RFP A                   RFP A
!>
!>     03 13 23 33 00 01 02    33 00 10 20 30 40 50
!>     04 14 24 34 44 11 12    43 44 11 21 31 41 51
!>     05 15 25 35 45 55 22    53 54 55 22 32 42 52
!>
!>
!>  ثم ننظر إلى تنسيق التعبئة الكاملة المستطيلة (RFP) عندما تكون N
!>  فردية. نقدم مثالًا حيث N = 5.
!>
!>     AP علوية                 AP سفلية
!>
!>   00 01 02 03 04              00
!>      11 12 13 14              10 11
!>         22 23 24              20 21 22
!>            33 34              30 31 32 33
!>               44              40 41 42 43 44
!>
!>
!>  إذا كان TRANSR = 'N'. تحتفظ RFP بـ AP كما يلي:
!>  بالنسبة لـ UPLO = 'U'، يتكون شبه المنحرف العلوي A(0:4,0:2) من الأعمدة الثلاثة الأخيرة
!>  من AP العلوي. يتكون المثلث السفلي A(3:4,0:1) من
!>  منقول العمودين الأولين من AP العلوي.
!>  بالنسبة لـ UPLO = 'L'، يتكون شبه المنحرف السفلي A(0:4,0:2) من الأعمدة الثلاثة الأولى
!>  من AP السفلي. يتكون المثلث العلوي A(0:1,1:2) من
!>  منقول العمودين الأخيرين من AP السفلي.
!>  هذا يغطي الحالة N فردية و TRANSR = 'N'.
!>
!>         RFP A                   RFP A
!>
!>        02 03 04                00 33 43
!>        12 13 14                10 11 44
!>        22 23 24                20 21 22
!>        00 33 34                30 31 32
!>        01 11 44                40 41 42
!>
!>  الآن إذا كان TRANSR = 'T'. RFP A في كلتا حالتي UPLO هو فقط
!>  منقول RFP A أعلاه. لذلك يحصل المرء على:
!>
!>           RFP A                   RFP A
!>
!>     02 12 22 00 01             00 10 20 30 40 50
!>     03 13 23 33 11             33 11 21 31 41 51
!>     04 14 24 34 44             43 44 22 32 42 52
!>
