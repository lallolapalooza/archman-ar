subroutine dtgsen (integer ijob, logical wantq, logical wantz, logical, dimension( * ) select, integer n, double precision, dimension( lda, * ) a, integer lda, double precision, dimension( ldb, * ) b, integer ldb, double precision, dimension( * ) alphar, double precision, dimension( * ) alphai, double precision, dimension( * ) beta, double precision, dimension( ldq, * ) q, integer ldq, double precision, dimension( ldz, * ) z, integer ldz, integer m, double precision pl, double precision pr, double precision, dimension( * ) dif, double precision, dimension( * ) work, integer lwork, integer, dimension( * ) iwork, integer liwork, integer info)

DTGSEN

الغرض:

!>
!> DTGSEN يعيد ترتيب التحلل القياسي الحقيقي المعمم لزوج مصفوفات حقيقي (A, B)
!> (من حيث تحويل تكافؤ متعامد Q**T * (A, B) * Z)، بحيث تظهر مجموعة مختارة من
!> القيم الذاتية في الكتل القطرية الأمامية للمصفوفة شبه المثلثية العلوية A
!> والمصفوفة المثلثية العلوية B. الأعمدة الأمامية لـ Q و Z تشكل قواعد متعامدة
!> لمساحات القيم الذاتية اليسرى واليمنى المقابلة (مساحات التفاضل).
!> يجب أن تكون (A, B) في شكل قياسي حقيقي معمّم (كما تم إرجاعه بواسطة DGGES)،
!> أي أن A شبه مثلثية علوية بكتل قطرية 1x1 و 2x2. B مثلثية علوية.
!>
!> DTGSEN يحسب أيضًا القيم الذاتية المعممة
!>
!>             w(j) = (ALPHAR(j) + i*ALPHAI(j))/BETA(j)
!>
!> لزوج المصفوفات المعاد ترتيبها (A, B).
!>
!> اختياريًا، يحسب DTGSEN تقديرات لأرقام الشروط العكسية للقيم الذاتية
!> ومساحات القيم الذاتية. هذه هي Difu[(A11,B11), (A22,B22)] و Difl[(A11,B11),
!> (A22,B22)]، أي الفصل (الفصول) بين أزواج المصفوفات (A11, B11) و (A22,B22)
!> التي تتوافق مع المجموعة المختارة والقيم الذاتية خارج المجموعة، على التوالي،
!> ومعايير المساحات الذاتية اليسرى واليمنى المتعلقة بالمجموعة المختارة في كتلة (1,1).
!>

المعلمات

IJOB 
!>          IJOB هو عدد صحيح
!>          يحدد ما إذا كانت أرقام الشروط مطلوبة للمجموعة المختارة من القيم الذاتية
!>          (PL و PR) أو مساحات التفاضل (Difu و Difl):
!>           =0: فقط إعادة الترتيب فيما يتعلق بـ SELECT. لا إضافات.
!>           =1: مقلوب معايير المساحات الذاتية اليسرى واليمنى المتعلقة بالمجموعة
!>               المختارة (PL و PR).
!>           =2: حدود عليا لـ Difu و Difl. تقدير قائم على معيار F (DIF(1:2)).
!>           =3: تقدير لـ Difu و Difl. تقدير قائم على معيار 1 (DIF(1:2)).
!>               حوالي 5 مرات أغلى من IJOB = 2.
!>           =4: حساب PL و PR و DIF (أي 0، 1 و 2 أعلاه): إصدار اقتصادي
!>               للحصول على كل شيء.
!>           =5: حساب PL و PR و DIF (أي 0، 1 و 3 أعلاه)
!>

WANTQ

!>          WANTQ هو منطقي
!>          .TRUE. : تحديث مصفوفة التحويل اليسرى Q؛
!>          .FALSE.: لا تحديث Q.
!>

WANTZ

!>          WANTZ هو منطقي
!>          .TRUE. : تحديث مصفوفة التحويل اليمنى Z؛
!>          .FALSE.: لا تحديث Z.
!>

SELECT

!>          SELECT هو مصفوفة منطقية، أبعادها (N)
!>          تحدد SELECT القيم الذاتية في المجموعة المختارة.
!>          لتحديد قيمة ذاتية حقيقية w(j)، يجب تعيين SELECT(j) إلى
!>          .TRUE.. لتحديد زوج قيم ذاتية مترافقة مركبة
!>          w(j) و w(j+1)، المقابلة لكتلة قطرية 2x2،
!>          يجب تعيين إما SELECT(j) أو SELECT(j+1) أو كليهما إلى
!>          .TRUE.؛ يجب أن يكون زوج القيم الذاتية المترافقة المركبة
!>          إما مدرجًا بالكامل في المجموعة أو مستبعدًا بالكامل.
!>

N

!>          N هو عدد صحيح
!>          ترتيب المصفوفات A و B. N >= 0.
!>

A

!>          A هي مصفوفة DOUBLE PRECISION، أبعادها (LDA,N)
!>          عند الإدخال، المصفوفة شبه المثلثية العلوية A، مع (A, B) في
!>          الشكل القياسي الحقيقي المعمم.
!>          عند الخروج، يتم الكتابة فوق A بالمصفوفة المعاد ترتيبها A.
!>

LDA

!>          LDA هو عدد صحيح
!>          البعد الرائد للمصفوفة A. LDA >= max(1,N).
!>

B

!>          B هي مصفوفة DOUBLE PRECISION، أبعادها (LDB,N)
!>          عند الإدخال، المصفوفة المثلثية العلوية B، مع (A, B) في
!>          الشكل القياسي الحقيقي المعمم.
!>          عند الخروج، يتم الكتابة فوق B بالمصفوفة المعاد ترتيبها B.
!>

LDB

!>          LDB هو عدد صحيح
!>          البعد الرائد للمصفوفة B. LDB >= max(1,N).
!>

ALPHAR

!>          ALPHAR هي مصفوفة DOUBLE PRECISION، أبعادها (N)
!>

ALPHAI

!>          ALPHAI هي مصفوفة DOUBLE PRECISION، أبعادها (N)
!>

BETA

!>          BETA هي مصفوفة DOUBLE PRECISION، أبعادها (N)
!>
!>          عند الخروج، (ALPHAR(j) + ALPHAI(j)*i)/BETA(j)، j=1,...,N، ستكون
!>          القيم الذاتية المعممة. ALPHAR(j) + ALPHAI(j)*i
!>          و BETA(j)،j=1,...,N هي أقطار الشكل القياسي المعقد (S,T)
!>          الذي سينتج إذا تم تقليل الكتل القطرية 2x2 للشكل القياسي الحقيقي المعمم
!>          لـ (A,B) بشكل أكبر إلى شكل مثلثي باستخدام تحويلات وحدوية معقدة.
!>          إذا كانت ALPHAI(j) صفرًا، فإن القيمة الذاتية j حقيقية؛ إذا كانت
!>          موجبة، فإن القيمتين الذاتيتين j و (j+1) هما زوج مترافق مركب،
!>          مع ALPHAI(j+1) سالبًا.
!>

Q

!>          Q هي مصفوفة DOUBLE PRECISION، أبعادها (LDQ,N)
!>          عند الإدخال، إذا كانت WANTQ = .TRUE.، فإن Q هي مصفوفة N × N.
!>          عند الخروج، تم ضرب Q مسبقًا بمصفوفة التحويل المتعامدة اليسرى
!>          التي تعيد ترتيب (A, B)؛ الأعمدة M الأمامية لـ Q تشكل قواعد متعامدة
!>          للمساحات الذاتية اليسرى (مساحات التفاضل) المحددة.
!>          إذا كانت WANTQ = .FALSE.، لا يتم الرجوع إلى Q.
!>

LDQ

!>          LDQ هو عدد صحيح
!>          البعد الرائد للمصفوفة Q. LDQ >= 1؛
!>          وإذا كانت WANTQ = .TRUE.، فإن LDQ >= N.
!>

Z

!>          Z هي مصفوفة DOUBLE PRECISION، أبعادها (LDZ,N)
!>          عند الإدخال، إذا كانت WANTZ = .TRUE.، فإن Z هي مصفوفة N × N.
!>          عند الخروج، تم ضرب Z مسبقًا بمصفوفة التحويل المتعامدة اليسرى
!>          التي تعيد ترتيب (A, B)؛ الأعمدة M الأمامية لـ Z تشكل قواعد متعامدة
!>          للمساحات الذاتية اليسرى (مساحات التفاضل) المحددة.
!>          إذا كانت WANTZ = .FALSE.، لا يتم الرجوع إلى Z.
!>

LDZ

!>          LDZ هو عدد صحيح
!>          البعد الرائد للمصفوفة Z. LDZ >= 1؛
!>          إذا كانت WANTZ = .TRUE.، فإن LDZ >= N.
!>

M

!>          M هو عدد صحيح
!>          بعد المساحات الذاتية اليسرى واليمنى المحددة (مساحات التفاضل). 0 <= M <= N.
!>
