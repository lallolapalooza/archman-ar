subroutine dtrmm (character side, character uplo, character transa, character diag, integer m, integer n, double precision alpha, double precision, dimension(lda,*) a, integer lda, double precision, dimension(ldb,*) b, integer ldb)

DTRMM

الغرض:

!>
!> DTRMM  تقوم بإحدى عمليات المصفوفة-المصفوفة:
!>
!>    B := alpha*op( A )*B,   أو   B := alpha*B*op( A ),
!>
!> حيث  alpha  هو قيمة قياسية،  B  هي مصفوفة m × n،  A  هي مصفوفة
!> قطرية علوية أو سفلية، موحدة أو غير موحدة، و op( A ) هي إحدى العمليات التالية:
!>
!>    op( A ) = A   أو   op( A ) = A**T.
!>

المعلمات

SIDE 
!>          SIDE هو CHARACTER*1
!>           عند الدخول، يحدد SIDE ما إذا كانت op( A ) تضرب B من
!>           اليسار أو اليمين كما يلي:
!>
!>              SIDE = 'L' أو 'l'   B := alpha*op( A )*B.
!>
!>              SIDE = 'R' أو 'r'   B := alpha*B*op( A ).
!>

UPLO

!>          UPLO هو CHARACTER*1
!>           عند الدخول، يحدد UPLO ما إذا كانت المصفوفة A علوية أو
!>           سفلية مثلثة كما يلي:
!>
!>              UPLO = 'U' أو 'u'   A هي مصفوفة مثلثة علوية.
!>
!>              UPLO = 'L' أو 'l'   A هي مصفوفة مثلثة سفلية.
!>

TRANSA

!>          TRANSA هو CHARACTER*1
!>           عند الدخول، يحدد TRANSA شكل op( A ) للاستخدام في
!>           ضرب المصفوفات كما يلي:
!>
!>              TRANSA = 'N' أو 'n'   op( A ) = A.
!>
!>              TRANSA = 'T' أو 't'   op( A ) = A**T.
!>
!>              TRANSA = 'C' أو 'c'   op( A ) = A**T.
!>

DIAG

!>          DIAG هو CHARACTER*1
!>           عند الدخول، يحدد DIAG ما إذا كانت A موحدة مثلثة أم لا
!>           كما يلي:
!>
!>              DIAG = 'U' أو 'u'   A مفترضة أنها موحدة مثلثة.
!>
!>              DIAG = 'N' أو 'n'   A ليست مفترضة أنها موحدة
!>                                  مثلثة.
!>

M

!>          M هو INTEGER
!>           عند الدخول، يحدد M عدد صفوف B. يجب أن يكون M
!>           على الأقل صفر.
!>

N

!>          N هو INTEGER
!>           عند الدخول، يحدد N عدد أعمدة B.  يجب أن يكون N
!>           على الأقل صفر.
!>

ALPHA

!>          ALPHA هو DOUBLE PRECISION.
!>           عند الدخول، يحدد ALPHA القيمة القياسية  alpha. عندما تكون  alpha
!>           صفرًا، لا يتم الرجوع إلى A ولا يلزم تعيين B قبل
!>           الدخول.
!>

A

!>           A هو مصفوفة DOUBLE PRECISION، أبعادها ( LDA, k )، حيث k هو m
!>           عندما SIDE = 'L' أو 'l'  و n  عندما SIDE = 'R' أو 'r'.
!>           قبل الدخول مع UPLO = 'U' أو 'u'، يجب أن يحتوي الجزء العلوي
!>           المثلثي العلوي من المصفوفة A على حجم k × k
!>           المصفوفة المثلثية، ولا يتم الرجوع إلى الجزء السفلي
!>           المثلثي السفلي من A.
!>           قبل الدخول مع UPLO = 'L' أو 'l'، يجب أن يحتوي الجزء السفلي
!>           المثلثي السفلي من المصفوفة A على حجم k × k
!>           المصفوفة المثلثية، ولا يتم الرجوع إلى الجزء العلوي
!>           المثلثي العلوي من A.
!>           لاحظ أنه عندما يكون DIAG = 'U' أو 'u'، لا يتم الرجوع إلى العناصر
!>           القطرية لـ A أيضًا، ولكن يُفترض أنها واحد.
!>

LDA

!>          LDA هو INTEGER
!>           عند الدخول، يحدد LDA البعد الأول لـ A كما هو معلن
!>           في البرنامج المستدعي (الفرعي).  عندما يكون SIDE = 'L' أو 'l'،
!>           يجب أن يكون LDA  على الأقل max( 1, m )، وعندما يكون SIDE = 'R' أو 'r'
!>           يجب أن يكون LDA على الأقل max( 1, n ).
!>

B

!>          B هو مصفوفة DOUBLE PRECISION، أبعادها ( LDB, N )
!>           قبل الدخول، يجب أن يحتوي الجزء الأمامي بحجم m × n من المصفوفة B على
!>           المصفوفة B، وعند الخروج يتم استبدالها بالمصفوفة
!>           المحولة.
!>

LDB

!>          LDB هو INTEGER
!>           عند الدخول، يحدد LDB البعد الأول لـ B كما هو معلن
!>           في البرنامج المستدعي (الفرعي). يجب أن يكون LDB
!>           على الأقل max( 1, m ).
!>
