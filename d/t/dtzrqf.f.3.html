subroutine dtzrqf (integer m, integer n, double precision, dimension( lda, * ) a, integer lda, double precision, dimension( * ) tau, integer info)

DTZRQF

الغرض:

!>
!> هذا الروتين مهمل وتم استبداله بالروتين DTZRZF.
!>
!> يقوم DTZRQF بتقليل المصفوفة A ذات الشكل شبه المنحرف العلوي الحقيقي M-by-N (M<=N)
!> إلى شكل مثلثي علوي بواسطة تحويلات متعامدة.
!>
!> يتم تحليل المصفوفة شبه المنحرفة العلوية A على النحو التالي:
!>
!>    A = ( R  0 ) * Z,
!>
!> حيث Z مصفوفة متعامدة N-by-N و R مصفوفة مثلثية علوية M-by-M.
!>

المعلمات

M 
!>          M هو عدد صحيح (INTEGER)
!>          عدد صفوف المصفوفة A.  M >= 0.
!>

N

!>          N هو عدد صحيح (INTEGER)
!>          عدد أعمدة المصفوفة A.  N >= M.
!>

A

!>          A هي مصفوفة (DOUBLE PRECISION array)، ذات بعد (LDA,N)
!>          عند الإدخال، يجب أن يحتوي الجزء العلوي شبه المنحرف من المصفوفة A،
!>          بحجم M-by-N، على المصفوفة المراد تحليلها.
!>          عند الخروج، يحتوي الجزء المثلثي العلوي من A بحجم M-by-M على
!>          المصفوفة المثلثية العلوية R، والعناصر من M+1 إلى N في الصفوف M الأولى من A،
!>          مع المصفوفة TAU، تمثل المصفوفة المتعامدة Z كناتج لـ M عاكسات أولية.
!>

LDA

!>          LDA هو عدد صحيح (INTEGER)
!>          البعد الرئيسي للمصفوفة A.  LDA >= max(1,M).
!>

TAU

!>          TAU هي مصفوفة (DOUBLE PRECISION array)، ذات بعد (M)
!>          العوامل القياسية للعوامل الأولية.
!>

INFO

!>          INFO هو عدد صحيح (INTEGER)
!>          = 0:  انتهاء ناجح
!>          < 0:  إذا كان INFO = -i، فإن المعامل i كان له قيمة غير قانونية
!>

تفاصيل إضافية:

!>
!>  يتم الحصول على التحليل باستخدام طريقة هاوسهولدر. مصفوفة التحويل
!>  kth، Z( k )، والتي تستخدم لإدخال أصفار في الصف ( m - k + 1 ) من A،
!>  يتم إعطاؤها بالشكل
!>
!>     Z( k ) = ( I     0   )،
!>              ( 0  T( k ) )
!>
!>  حيث
!>
!>     T( k ) = I - tau*u( k )*u( k )**T،   u( k ) = (   1    )،
!>                                                   (   0    )
!>                                                   ( z( k ) )
!>
!>  يتم اختيار tau و z( k ) لإفناء العناصر في الصف kth من X.
!>
!>  يتم إرجاع القيمة العددية tau في العنصر kth من TAU والمتجه
!>  u( k ) في الصف kth من A، بحيث تكون عناصر z( k ) موجودة في
!>  a( k, m + 1 )، ...، a( k, n ). يتم إرجاع عناصر R في
!>  الجزء المثلثي العلوي من A.
!>
!>  يتم إعطاء Z بواسطة
!>
!>     Z =  Z( 1 ) * Z( 2 ) * ... * Z( m ).
!>
