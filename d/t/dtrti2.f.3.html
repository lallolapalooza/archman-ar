subroutine dtrti2 (uplo, diag, n, a, lda, info)
DTRTI2 تحسب معكوس مصفوفة مثلثة (خوارزمية غير محظورة).

توثيق الوظيفة/الروتين الفرعي

subroutine dtrti2 (character uplo, character diag, integer n, double precision, dimension( lda, * ) a, integer lda, integer info)

DTRTI2 تحسب معكوس مصفوفة مثلثة (خوارزمية غير محظورة).

الغرض:

!>
!> DTRTI2 تحسب معكوس مصفوفة مثلثة عليا أو سفلية حقيقية
!>
!> هذا هو الإصدار المستوى 2 من BLAS للخوارزمية.
!>

المعلمات

UPLO 
!>          UPLO is CHARACTER*1
!>          يحدد ما إذا كانت المصفوفة A مثلثة عليا أم سفلية.
!>          = 'U':  مثلثة عليا
!>          = 'L':  مثلثة سفلية
!>

DIAG

!>          DIAG is CHARACTER*1
!>          يحدد ما إذا كانت المصفوفة A مثلثة وحدوية أم لا.
!>          = 'N':  مثلثة غير وحدوية
!>          = 'U':  مثلثة وحدوية
!>

N

!>          N is INTEGER
!>          رتبة المصفوفة A.  N >= 0.
!>

A

!>          A is DOUBLE PRECISION array, dimension (LDA,N)
!>          عند الدخول، الجزء العلوي أو السفلي المثلثي من المصفوفة A
!>          بأبعاد n x n يحتوي على المصفوفة المثلثية. إذا كان UPLO = 'U'،
!>          فإن الجزء العلوي المثلثي من المصفوفة A بحدود n x n يحتوي
!>          على المصفوفة المثلثية العليا، والجزء السفلي المثلثي الصارم
!>          من A لا يتم الاستشهاد به. إذا كان UPLO = 'L'، فإن الجزء
!>          السفلي المثلثي من المصفوفة A بحدود n x n يحتوي على المصفوفة
!>          المثلثية السفلى، والجزء العلوي المثلثي الصارم من A لا يتم
!>          الاستشهاد به. إذا كان DIAG = 'U'، فإن العناصر القطرية
!>          لـ A لا يتم الاستشهاد بها أيضًا ويُفترض أنها 1.
!>
!>          عند الخروج، معكوس المصفوفة الأصلية (المثلثة)، بنفس تنسيق
!>          التخزين.
!>

LDA

!>          LDA is INTEGER
!>          البعد الرائد للمصفوفة A.  LDA >= max(1,N).
!>

INFO

!>          INFO is INTEGER
!>          = 0: خروج ناجح
!>          < 0: إذا كان INFO = -k، فقد كان المعامل k لديه قيمة غير قانونية
!>
