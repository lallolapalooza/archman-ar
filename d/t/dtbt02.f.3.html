subroutine dtbt02 (character uplo, character trans, character diag, integer n, integer kd, integer nrhs, double precision, dimension( ldab, * ) ab, integer ldab, double precision, dimension( ldx, * ) x, integer ldx, double precision, dimension( ldb, * ) b, integer ldb, double precision, dimension( * ) work, double precision resid)

DTBT02

الغرض:

!>
!> DTBT02 يحسب الباقي للحل المحسوب لنظام المعادلات الخطية المثلثي op(A)*X = B، عندما تكون A مصفوفة ذات نطاق مثلثي. نسبة الاختبار هي الحد الأقصى لـ
!>    norm(b - op(A)*x) / ( ||op(A)||_1 * norm(x) * EPS )،
!> حيث op(A) = A أو A**T، b هو عمود B، x هو متجه الحل، و EPS هي الدقة الآلية.
!> المعيار المستخدم هو المعيار 1.
!>

المعلمات

UPLO 
!>          UPLO is CHARACTER*1
!>          يحدد ما إذا كانت المصفوفة A علوية أو سفلية مثلثية.
!>          = 'U':  مثلثية علوية
!>          = 'L':  مثلثية سفلية
!>

TRANS

!>          TRANS is CHARACTER*1
!>          يحدد العملية المطبقة على A.
!>          = 'N':  A    * X = B  (بدون تبديل)
!>          = 'T':  A**T * X = B  (تبديل)
!>          = 'C':  A**H * X = B  (تبديل مرافق = تبديل)
!>

DIAG

!>          DIAG is CHARACTER*1
!>          يحدد ما إذا كانت المصفوفة A وحدة مثلثية أم لا.
!>          = 'N':  غير وحدة مثلثية
!>          = 'U':  وحدة مثلثية
!>

N

!>          N is INTEGER
!>          ترتيب المصفوفة A.  N >= 0.
!>

KD

!>          KD is INTEGER
!>          عدد الأقطار العلوية أو الأقطار السفلية للمصفوفة المثلثية ذات النطاق A.  KD >= 0.
!>

NRHS

!>          NRHS is INTEGER
!>          عدد الأطراف اليمنى، أي عدد أعمدة المصفوفات X و B.  NRHS >= 0.
!>

AB

!>          AB is DOUBLE PRECISION array, dimension (LDAB,N)
!>          المصفوفة المثلثية ذات النطاق العلوية أو السفلية A، مخزنة في
!>          الصفوف الـ kd+1 الأولى من المصفوفة. يتم تخزين العمود j من A
!>          في العمود j من المصفوفة AB كالتالي:
!>          if UPLO = 'U', AB(kd+1+i-j,j) = A(i,j) for max(1,j-kd)<=i<=j;
!>          if UPLO = 'L', AB(1+i-j,j)    = A(i,j) for j<=i<=min(n,j+kd).
!>

LDAB

!>          LDAB is INTEGER
!>          البعد الرائد للمصفوفة AB.  LDAB >= KD+1.
!>

X

!>          X is DOUBLE PRECISION array, dimension (LDX,NRHS)
!>          متجهات الحل المحسوبة لنظام المعادلات الخطية.
!>

LDX

!>          LDX is INTEGER
!>          البعد الرائد للمصفوفة X.  LDX >= max(1,N).
!>

B

!>          B is DOUBLE PRECISION array, dimension (LDB,NRHS)
!>          متجهات الطرف الأيمن لنظام المعادلات الخطية.
!>

LDB

!>          LDB is INTEGER
!>          البعد الرائد للمصفوفة B.  LDB >= max(1,N).
!>

WORK

!>          WORK is DOUBLE PRECISION array, dimension (N)
!>

RESID

!>          RESID is DOUBLE PRECISION
!>          الحد الأقصى على عدد الأطراف اليمنى لـ
!>          norm(op(A)*x - b) / ( norm(op(A)) * norm(x) * EPS ).
!>
