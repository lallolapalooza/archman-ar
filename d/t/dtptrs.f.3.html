subroutine dtptrs (character uplo, character trans, character diag, integer n, integer nrhs, double precision, dimension( * ) ap, double precision, dimension( ldb, * ) b, integer ldb, integer info)

DTPTRS

الغرض:

!>
!> DTPTRS يحل نظامًا مثلثيًا بالشكل:
!>
!>    A * X = B  أو  A**T * X = B،
!>
!> حيث A هي مصفوفة مثلثية من الدرجة N مخزنة بتنسيق مضغوط، و B هي مصفوفة N-by-NRHS.
!>
!> يتحقق هذا الروتين الفرعي من أن A غير منفردة، ولكن يجب أن يلاحظ المتصلون أنه يتم اكتشاف
!> التفرد المطلق فقط. من الممكن أن تكون عناصر قطرية واحدة أو أكثر من A أرقامًا صغيرة جدًا
!!> دون أن يشير هذا الروتين الفرعي إلى خطأ.
!>
!> إذا كان فقدان الدقة الرقمية المحتمل بسبب المصفوفات القريبة من التفرد مصدر قلق،
!> فيجب على المتصل التحقق من أن A غير منفردة ضمن بعض التفاوت قبل استدعاء هذا الروتين الفرعي.
!>

المعلمات

UPLO 
!>          UPLO هو CHARACTER*1
!>          = 'U':  A هي مثلثية عليا؛
!>          = 'L':  A هي مثلثية سفلى.
!>

TRANS

!>          TRANS هو CHARACTER*1
!>          يحدد شكل نظام المعادلات:
!>          = 'N':  A * X = B  (بدون تبديل)
!>          = 'T':  A**T * X = B  (تبديل)
!>          = 'C':  A**H * X = B  (مرافق تبديل = تبديل)
!>

DIAG

!>          DIAG هو CHARACTER*1
!>          = 'N':  A ليست مثلثية موحدة؛
!>          = 'U':  A هي مثلثية موحدة.
!>

N

!>          N هو INTEGER
!>          رتبة المصفوفة A.  N >= 0.
!>

NRHS

!>          NRHS هو INTEGER
!>          عدد الجوانب اليمنى، أي عدد أعمدة
!>          المصفوفة B.  NRHS >= 0.
!>

AP

!>          AP هو مصفوفة DOUBLE PRECISION، البعد (N*(N+1)/2)
!>          المصفوفة المثلثية A، العليا أو السفلى، مضغوطة عمودياً في
!!>          مصفوفة خطية.  يتم تخزين العمود j من A في المصفوفة
!>          AP كما يلي:
!>          إذا كان UPLO = 'U'، فإن AP(i + (j-1)*j/2) = A(i,j) لـ 1<=i<=j؛
!>          إذا كان UPLO = 'L'، فإن AP(i + (j-1)*(2*n-j)/2) = A(i,j) لـ j<=i<=n.
!>

B

!>          B هي مصفوفة DOUBLE PRECISION، البعد (LDB,NRHS)
!>          عند الإدخال، مصفوفة الجانب الأيمن B.
!>          عند الخروج، إذا كان INFO = 0، فإن مصفوفة الحل X.
!>

LDB

!>          LDB هو INTEGER
!>          البعد الرائد للمصفوفة B.  LDB >= max(1,N).
!>

INFO

!>          INFO هو INTEGER
!>          = 0:  خروج ناجح
!>          < 0:  إذا كان INFO = -i، فإن المعامل i كان له قيمة غير قانونية
!>          > 0:  إذا كان INFO = i، فإن العنصر القطري i من A يساوي صفرًا بالضبط،
!>                مما يشير إلى أن المصفوفة منفردة وأن
!>                الحلول X لم يتم حسابها.
!>
