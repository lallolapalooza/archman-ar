subroutine dtrtrs (character uplo, character trans, character diag, integer n, integer nrhs, double precision, dimension( lda, * ) a, integer lda, double precision, dimension( ldb, * ) b, integer ldb, integer info)

DTRTRS

الغرض:

!>
!> DTRTRS يحل نظامًا مثلثيًا من الشكل
!>
!>    A * X = B  أو  A**T * X = B,
!>
!> حيث A مصفوفة مثلثية من الرتبة N، و B مصفوفة N-by-NRHS.
!>
!> هذا الإجراء الفرعي يتحقق من أن A غير مفردة، ولكن يجب على المتصلين ملاحظة أن التفرد التام فقط هو الذي يتم اكتشافه. من الممكن أن تكون هناك عناصر قطرية واحدة أو أكثر في A هي أرقام صغيرة جدًا (subnormally tiny) دون أن يشير هذا الإجراء الفرعي إلى خطأ.
!>
!> إذا كان فقدان الدقة العددية المحتمل بسبب المصفوفات القريبة من المفردة يمثل مصدر قلق، فيجب على المتصل التحقق من أن A غير مفردة ضمن نطاق معين قبل استدعاء هذا الإجراء الفرعي.
!>

المعاملات

UPLO 
!>          UPLO هو CHARACTER*1
!>          = 'U':  A مصفوفة مثلثية عليا؛
!>          = 'L':  A مصفوفة مثلثية سفلى.
!>

TRANS

!>          TRANS هو CHARACTER*1
!>          يحدد شكل نظام المعادلات:
!>          = 'N':  A * X = B  (بدون تبديل)
!>          = 'T':  A**T * X = B  (مبدلة)
!>          = 'C':  A**H * X = B  (منسقة مبدلة = مبدلة)
!>

DIAG

!>          DIAG هو CHARACTER*1
!>          = 'N':  A مصفوفة مثلثية غير موحدة؛
!>          = 'U':  A مصفوفة مثلثية موحدة.
!>

N

!>          N هو INTEGER
!>          رتبة المصفوفة A.  N >= 0.
!>

NRHS

!>          NRHS هو INTEGER
!>          عدد الأطراف اليمنى، أي عدد أعمدة
!>          المصفوفة B.  NRHS >= 0.
!>

A

!>          A هي مصفوفة DOUBLE PRECISION، الأبعاد (LDA,N)
!>          المصفوفة المثلثية A.  إذا كان UPLO = 'U'، فإن الجزء العلوي المثلثي الرائد N-by-N
!>          من المصفوفة A يحتوي على المصفوفة المثلثية العليا، والجزء السفلي المثلثي الصارم
!>          من A لا يتم الرجوع إليه.  إذا كان UPLO = 'L'، فإن الجزء السفلي المثلثي الرائد N-by-N
!>          من المصفوفة A يحتوي على المصفوفة المثلثية السفلى، والجزء العلوي المثلثي الصارم من A لا يتم
!>          الرجوع إليه.  إذا كان DIAG = 'U'، فإن العناصر القطرية لـ A
!>          لا يتم الرجوع إليها أيضًا ويُفترض أنها 1.
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرائد للمصفوفة A.  LDA >= max(1,N).
!>

B

!>          B هي مصفوفة DOUBLE PRECISION، الأبعاد (LDB,NRHS)
!>          عند الإدخال، مصفوفة الطرف الأيمن B.
!>          عند الخروج، إذا كان INFO = 0، فإن مصفوفة الحل X.
!>

LDB

!>          LDB هو INTEGER
!>          البعد الرائد للمصفوفة B.  LDB >= max(1,N).
!>

INFO

!>          INFO هو INTEGER
!>          = 0:  انتهاء ناجح
!>          < 0: إذا كان INFO = -i، فإن المعامل i كان له قيمة غير قانونية
!>          > 0:  إذا كان INFO = i، فإن العنصر القطري i لـ A يساوي صفرًا بالضبط،
!>               مما يشير إلى أن المصفوفة مفردة ولم يتم حساب الحلول
!>               X.
!>
