subroutine dtpqrt (integer m, integer n, integer l, integer nb, double precision, dimension( lda, * ) a, integer lda, double precision, dimension( ldb, * ) b, integer ldb, double precision, dimension( ldt, * ) t, integer ldt, double precision, dimension( * ) work, integer info)

DTPQRT

الغرض:

!>
!> DTPQRT يحسب تحليل QR محظور لمصفوفة حقيقية
!>  C، والتي تتكون من
!>  كتلة مثلثة A وكتلة خماسية B، باستخدام تمثيل WY المدمج لـ Q.
!>

المعلمات

M 
!>          M هو صحيح (INTEGER)
!>          عدد صفوف المصفوفة B.
!>          M >= 0.
!>

N

!>          N هو صحيح (INTEGER)
!>          عدد أعمدة المصفوفة B، وترتيب
!>          المصفوفة المثلثية A.
!>          N >= 0.
!>

L

!>          L هو صحيح (INTEGER)
!>          عدد صفوف الجزء شبه المنحرف العلوي من B.
!>          MIN(M,N) >= L >= 0. انظر التفاصيل الإضافية.
!>

NB

!>          NB هو صحيح (INTEGER)
!>          حجم الكتلة المستخدم في QR المحظور. N >= NB >= 1.
!>

A

!>          A هي مصفوفة DOUBLE PRECISION، أبعادها (LDA,N)
!>          عند الإدخال، المصفوفة المثلثية العلوية A ذات الأبعاد N × N.
!>          عند الخروج، تحتوي العناصر على القطر وفوقه في المصفوفة
!>          على المصفوفة المثلثية العلوية R.
!>

LDA

!>          LDA هو صحيح (INTEGER)
!>          البعد الرائد للمصفوفة A. LDA >= max(1,N).
!>

B

!>          B هي مصفوفة DOUBLE PRECISION، أبعادها (LDB,N)
!>          عند الإدخال، المصفوفة الخماسية B ذات الأبعاد M × N. أول M-L صفوف
!>          مستطيلة، وآخر L صفوف شبه منحرفة علوية.
!>          عند الخروج، تحتوي B على المصفوفة الخماسية V. انظر التفاصيل الإضافية.
!>

LDB

!>          LDB هو صحيح (INTEGER)
!>          البعد الرائد للمصفوفة B. LDB >= max(1,M).
!>

T

!>          T هي مصفوفة DOUBLE PRECISION، أبعادها (LDT,N)
!>          المُرتدات المثلثية العلوية المحظورة مخزنة بتنسيق مدمج
!>          كسلسلة من الكتل المثلثية العلوية. انظر التفاصيل الإضافية.
!>

LDT

!>          LDT هو صحيح (INTEGER)
!>          البعد الرائد للمصفوفة T. LDT >= NB.
!>

WORK

!>          WORK هي مصفوفة DOUBLE PRECISION، أبعادها (NB*N)
!>

INFO

!>          INFO هو صحيح (INTEGER)
!>          = 0:  انتهاء ناجح

التفاصيل الإضافية:

!>
!>  المصفوفة المدخلة C هي مصفوفة (N+M)-by-N
!>
!>               C = [ A ]
!>                   [ B ]
!>
!>  حيث A هي مصفوفة مثلثة علوية N-by-N، و B هي مصفوفة خماسية M-by-N
!>  تتكون من مصفوفة مستطيلة B1 بأبعاد (M-L)-by-N في الأعلى ومصفوفة
!>  شبه منحرفة علوية L-by-N تسمى B2 في الأسفل:
!>
!>               B = [ B1 ]  <- مستطيلة بأبعاد (M-L)-by-N
!>                   [ B2 ]  <- شبه منحرفة علوية بأبعاد L-by-N.
!>
!>  تتكون المصفوفة شبه المنحرفة العلوية B2 من أول L صفوف من مصفوفة
!>  مثلثة علوية N-by-N، حيث 0 <= L <= MIN(M,N). إذا كانت L=0،
!>  فإن B مستطيلة M-by-N؛ إذا كانت M=L=N، فإن B مثلثة علوية.
!>
!>  تخزن المصفوفة W المُرتدات الأولية H(i) في العمود i
!>  تحت القطر (لـ A) في المصفوفة المدخلة C ذات الأبعاد (N+M)-by-N
!>
!>               C = [ A ]  <- مثلثة علوية N-by-N
!>                   [ B ]  <- خماسية M-by-N
!>
!>  بحيث يمكن تمثيل W كـ
!>
!>               W = [ I ]  <- هوية، N-by-N
!>                   [ V ]  <- M-by-N، نفس شكل B.
!>
!>  وبالتالي، فإن جميع المعلومات اللازمة لـ W موجودة عند الخروج في B،
!>  والتي نسميها V أعلاه. لاحظ أن V لها نفس شكل B؛ أي،
!>
!>               V = [ V1 ] <- مستطيلة بأبعاد (M-L)-by-N
!>                   [ V2 ] <- شبه منحرفة علوية بأبعاد L-by-N.
!>
!>  تمثل أعمدة V المتجهات التي تحدد H(i).
!>
!>  عدد الكتل هو B = سقف(N/NB)، حيث كل
!>  كتلة من الرتبة NB باستثناء الكتلة الأخيرة، والتي تكون من الرتبة
!>  IB = N - (B-1)*NB. لكل من الكتل B، يتم حساب مُرتد عامل مثلث علوي:
!>  T1, T2, ..., TB. يتم تخزين T ذات الأبعاد NB-by-NB (و IB-by-IB
!>  للكتلة الأخيرة) في المصفوفة T ذات الأبعاد NB-by-N كـ
!>
!>               T = [T1 T2 ... TB].
!>
