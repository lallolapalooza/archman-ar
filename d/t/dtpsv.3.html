subroutine dtpsv (character uplo, character trans, character diag, integer n, double precision, dimension(*) ap, double precision, dimension(*) x, integer incx)

DTPSV

الغرض:

!>
!> DTPSV  يحل أحد أنظمة المعادلات
!>
!>    A*x = b,   أو   A**T*x = b,
!>
!> حيث b و x هما متجهان مكونان من n عنصر و A هي مصفوفة مثلثة علوية أو سفلية
!> مربعة بحجم n × n، وحدة أو غير وحدة، مقدمة في شكل مضغوط.
!>
!> لا يتضمن هذا الإجراء الفرعي أي اختبار للانفراد أو قرب الانفراد.
!> يجب إجراء مثل هذه الاختبارات قبل استدعاء هذا الإجراء الفرعي.
!>

المعلمات

UPLO 
!>          UPLO هو CHARACTER*1
!>           عند الإدخال، يحدد UPLO ما إذا كانت المصفوفة مثلثة علوية أو سفلية
!>           على النحو التالي:
!>
!>              UPLO = 'U' أو 'u'   A هي مصفوفة مثلثة علوية.
!>
!>              UPLO = 'L' أو 'l'   A هي مصفوفة مثلثة سفلية.
!>

TRANS

!>          TRANS هو CHARACTER*1
!>           عند الإدخال، يحدد TRANS المعادلات التي سيتم حلها على النحو
!>           التالي:
!>
!>              TRANS = 'N' أو 'n'   A*x = b.
!>
!>              TRANS = 'T' أو 't'   A**T*x = b.
!>
!>              TRANS = 'C' أو 'c'   A**T*x = b.
!>

DIAG

!>          DIAG هو CHARACTER*1
!>           عند الإدخال، يحدد DIAG ما إذا كانت A مثلثة وحدة أم لا
!>           على النحو التالي:
!>
!>              DIAG = 'U' أو 'u'   يفترض أن A مثلثة وحدة.
!>
!>              DIAG = 'N' أو 'n'   لا يفترض أن A مثلثة
!>                                  وحدة.
!>

N

!>          N هو INTEGER
!>           عند الإدخال، يحدد N رتبة المصفوفة A.
!>           يجب أن يكون N صفرًا أو أكبر.
!>

AP

!>          AP هو مصفوفة DOUBLE PRECISION، بعدها على الأقل
!>           ( ( n*( n + 1 ) )/2 ).
!>           قبل الدخول مع UPLO = 'U' أو 'u'، يجب أن تحتوي المصفوفة AP
!>           على المصفوفة المثلثة العلوية مضغوطة بشكل متسلسل،
!>           عمودًا بعد عمود، بحيث يحتوي AP( 1 ) على a( 1, 1 )،
!>           يحتوي AP( 2 ) و AP( 3 ) على a( 1, 2 ) و a( 2, 2 )
!>           على التوالي، وهكذا.
!>           قبل الدخول مع UPLO = 'L' أو 'l'، يجب أن تحتوي المصفوفة AP
!>           على المصفوفة المثلثة السفلية مضغوطة بشكل متسلسل،
!>           عمودًا بعد عمود، بحيث يحتوي AP( 1 ) على a( 1, 1 )،
!>           يحتوي AP( 2 ) و AP( 3 ) على a( 2, 1 ) و a( 3, 1 )
!>           على التوالي، وهكذا.
!>           لاحظ أنه عندما يكون DIAG = 'U' أو 'u'، لا يتم الرجوع إلى العناصر
!>           القطرية لـ A، ولكن يفترض أنها الوحدة.
!>

X

!>          X هو مصفوفة DOUBLE PRECISION، بعدها على الأقل
!>           ( 1 + ( n - 1 )*abs( INCX ) ).
!>           قبل الدخول، يجب أن تحتوي المصفوفة X المزودة بفواصل متزايدة على
!>           المتجه الأيمن b المكون من n عنصر. عند الخروج، يتم الكتابة فوق X
!>           بالمتجه الحل x.
!>

INCX

!>          INCX هو INTEGER
!>           عند الإدخال، يحدد INCX الزيادة لعناصر X.
!>           يجب ألا يكون INCX صفرًا.
!>
