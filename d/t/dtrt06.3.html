توثيق الوظيفة/الروتين الفرعي

subroutine dtrt06 (double precision rcond, double precision rcondc, character uplo, character diag, integer n, double precision, dimension( lda, * ) a, integer lda, double precision, dimension( * ) work, double precision rat)

DTRT06

الغرض:

!>
!> DTRT06 يحسب نسبة اختبار تقارن RCOND (المقلوب
!> رقم الحالة لمصفوفة مثلثية A) و RCONDC، التقدير
!> المحسوب بواسطة DTRCON.  تستخدم معلومات حول المصفوفة المثلثية A
!> إذا كان أحد التقديرين صفرًا والآخر غير صفري لاتخاذ قرار ما إذا كان
!> التسرب السفلي في التقدير مبررًا.
!>

المعلمات

RCOND 
!>          RCOND هو DOUBLE PRECISION
!>          تقدير المقلوب لرقم الحالة الذي تم الحصول عليه بواسطة
!>          تشكيل معكوس المصفوفة A وحساب
!>          RCOND = 1/( norm(A) * norm(inv(A)) ).
!>

RCONDC

!>          RCONDC هو DOUBLE PRECISION
!>          تقدير المقلوب لرقم الحالة المحسوب بواسطة
!>          DTRCON.
!>

UPLO

!>          UPLO هو CHARACTER
!>          يحدد ما إذا كانت المصفوفة A علوية أم سفلية مثلثية.
!>          = 'U':  مثلثية علوية
!>          = 'L':  مثلثية سفلية
!>

DIAG

!>          DIAG هو CHARACTER
!>          يحدد ما إذا كانت المصفوفة A وحدة مثلثية أم لا.
!>          = 'N':  مثلثية غير وحدة
!>          = 'U':  مثلثية وحدة
!>

N

!>          N هو INTEGER
!>          رتبة المصفوفة A.  N >= 0.
!>

A

!>          A هو DOUBLE PRECISION array, dimension (LDA,N)
!>          المصفوفة المثلثية A.  إذا كان UPLO = 'U'، فإن الجزء العلوي
!>          المثلثي العلوي بحجم n × n في المصفوفة A يحتوي على المصفوفة
!>          المثلثية العلوية، ولا تتم الإشارة إلى الجزء المثلثي السفلي الصارم
!>          من A.  إذا كان UPLO = 'L'، فإن الجزء السفلي
!>          المثلثي السفلي بحجم n × n في المصفوفة A يحتوي على المصفوفة
!>          المثلثية السفلية، ولا تتم الإشارة إلى الجزء المثلثي العلوي الصارم
!>          من A.  إذا كان DIAG = 'U'، فإن العناصر القطرية لـ A
!>          لا تتم الإشارة إليها أيضًا ويفترض أنها 1.
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرئيسي للمصفوفة A.  LDA >= max(1,N).
!>

WORK

!>          WORK هو DOUBLE PRECISION array, dimension (N)
!>

RAT

!>          RAT هو DOUBLE PRECISION
!>          نسبة الاختبار.  إذا كان كل من RCOND و RCONDC غير صفريين،
!>             RAT = MAX( RCOND, RCONDC )/MIN( RCOND, RCONDC ) - 1.
!>          إذا كان RAT = 0، فإن التقديرين متطابقان تمامًا.
!>
