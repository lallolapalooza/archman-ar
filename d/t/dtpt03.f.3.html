subroutine dtpt03 (character uplo, character trans, character diag, integer n, integer nrhs, double precision, dimension( * ) ap, double precision scale, double precision, dimension( * ) cnorm, double precision tscal, double precision, dimension( ldx, * ) x, integer ldx, double precision, dimension( ldb, * ) b, integer ldb, double precision, dimension( * ) work, double precision resid)

DTPT03

الغرض:

!>
!> DTPT03 يحسب الباقي لحل نظام المعادلات المثلثي المُقيَّس A*x = s*b  أو  A'*x = s*b  عندما يتم تخزين المصفوفة المثلثية A في صيغة مدمجة. هنا A' هو منقول A، s هو عدد قياسي، و x و b هما مصفوفتان N × NRHS. نسبة الاختبار هي الحد الأقصى لعدد الأطراف اليمنى من
!>    norm(s*b - op(A)*x) / ( norm(op(A)) * norm(x) * EPS )
!> حيث op(A) يعني A أو A' و EPS هي الدقة الآلية.
!>

المعلمات:

UPLO 
!>          UPLO هو CHARACTER*1
!>          يحدد ما إذا كانت المصفوفة A مثلثية علوية أو سفلية.
!>          = 'U':  مثلثية علوية
!>          = 'L':  مثلثية سفلية
!>

TRANS

!>          TRANS هو CHARACTER*1
!>          يحدد العملية المطبقة على A.
!>          = 'N':  A *x = s*b  (بدون منقول)
!>          = 'T':  A'*x = s*b  (منقول)
!>          = 'C':  A'*x = s*b  (منقول مترافق = منقول)
!>

DIAG

!>          DIAG هو CHARACTER*1
!>          يحدد ما إذا كانت المصفوفة A مثلثية وحدية أم لا.
!>          = 'N':  مثلثية غير وحدية
!>          = 'U':  مثلثية وحدية
!>

N

!>          N هو INTEGER
!>          ترتيب المصفوفة A.  N >= 0.
!>

NRHS

!>          NRHS هو INTEGER
!>          عدد الأطراف اليمنى، أي عدد أعمدة المصفوفتين X و B.  NRHS >= 0.
!>

AP

!>          AP هو مصفوفة DOUBLE PRECISION، بُعدها (N*(N+1)/2)
!>          المصفوفة المثلثية العلوية أو السفلية A، مدمجة عمودياً في
!>          مصفوفة خطية.  يتم تخزين العمود j من A في المصفوفة
!>          AP كما يلي:
!>          إذا كان UPLO = 'U'، فإن AP((j-1)*j/2 + i) = A(i,j) لـ 1<=i<=j؛
!>          إذا كان UPLO = 'L'،
!>             AP((j-1)*(n-j) + j*(j+1)/2 + i-j) = A(i,j) لـ j<=i<=n.
!>

SCALE

!>          SCALE هو DOUBLE PRECISION
!>          عامل القياس s المستخدم في حل النظام المثلثي.
!>

CNORM

!>          CNORM هو مصفوفة DOUBLE PRECISION، بُعدها (N)
!>          المعايير 1 لأعمدة A، بدون حساب القطر.
!>

TSCAL

!>          TSCAL هو DOUBLE PRECISION
!>          عامل القياس المستخدم في حساب المعايير 1 في CNORM.
!>          CNORM يحتوي فعلياً على معايير الأعمدة لـ TSCAL*A.
!>

X

!>          X هو مصفوفة DOUBLE PRECISION، بُعدها (LDX,NRHS)
!>          متجهات الحل المحسوبة لأنظمة المعادلات الخطية.
!>

LDX

!>          LDX هو INTEGER
!>          البعد الرئيسي للمصفوفة X.  LDX >= max(1,N).
!>

B

!>          B هو مصفوفة DOUBLE PRECISION، بُعدها (LDB,NRHS)
!>          متجهات الطرف الأيمن لأنظمة المعادلات الخطية.
!>

LDB

!>          LDB هو INTEGER
!>          البعد الرئيسي للمصفوفة B.  LDB >= max(1,N).
!>

WORK

!>          WORK هو مصفوفة DOUBLE PRECISION، بُعدها (N)
!>

RESID

!>          RESID هو DOUBLE PRECISION
!>          الحد الأقصى لعدد الأطراف اليمنى من
!>          norm(op(A)*x - s*b) / ( norm(op(A)) * norm(x) * EPS ).
!>
