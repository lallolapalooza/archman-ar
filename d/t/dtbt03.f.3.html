subroutine dtbt03 (character uplo, character trans, character diag, integer n, integer kd, integer nrhs, double precision, dimension( ldab, * ) ab, integer ldab, double precision scale, double precision, dimension( * ) cnorm, double precision tscal, double precision, dimension( ldx, * ) x, integer ldx, double precision, dimension( ldb, * ) b, integer ldb, double precision, dimension( * ) work, double precision resid)

DTBT03

الغرض:

!>
!> DTBT03 يحسب الباقي لحل نظام مثلثي لمعادلات مصقولة A*x = s*b أو A'*x = s*b
!> عندما تكون A مصفوفة شريطية مثلثية. هنا A' هي منقولة A، s هو عدد سلمي،
!> و x و b هما مصفوفتان N × NRHS. نسبة الاختبار هي الحد الأقصى
!> على عدد الأطراف اليمنى لـ
!>    norm(s*b - op(A)*x) / ( norm(op(A)) * norm(x) * EPS )،
!> حيث يمثل op(A) A أو A' و EPS هو اصغر رقم في الآلة.
!>

المعلمات

UPLO 
!>          UPLO هو CHARACTER*1
!>          يحدد ما إذا كانت المصفوفة A مثلثية علوية أو سفلية.
!>          = 'U':  مثلثية علوية
!>          = 'L':  مثلثية سفلية
!>

TRANS

!>          TRANS هو CHARACTER*1
!>          يحدد العملية المطبقة على A.
!>          = 'N':  A *x = b  (بدون منقول)
!>          = 'T':  A'*x = b  (منقول)
!>          = 'C':  A'*x = b  (منقول مترافق = منقول)
!>

DIAG

!>          DIAG هو CHARACTER*1
!>          يحدد ما إذا كانت المصفوفة A مثلثية وحدية أم لا.
!>          = 'N':  مثلثية غير وحدية
!>          = 'U':  مثلثية وحدية
!>

N

!>          N هو INTEGER
!>          رتبة المصفوفة A.  N >= 0.
!>

KD

!>          KD هو INTEGER
!>          عدد الأقطار الفائقة أو الأقطار الفرعية للمصفوفة
!>          الشريطية المثلثية A.  KD >= 0.
!>

NRHS

!>          NRHS هو INTEGER
!>          عدد الأطراف اليمنى، أي عدد أعمدة
!>          المصفوفتين X و B.  NRHS >= 0.
!>

AB

!>          AB هو مصفوفة DOUBLE PRECISION، بأبعاد (LDAB,N)
!>          المصفوفة الشريطية المثلثية العلوية أو السفلية A، مخزنة في
!>          الصفوف k+1 الأولى من المصفوفة. يتم تخزين العمود j من A
!>          في العمود j من المصفوفة AB كما يلي:
!>          إذا كان UPLO = 'U'، فإن AB(kd+1+i-j,j) = A(i,j) لـ max(1,j-kd)<=i<=j؛
!>          إذا كان UPLO = 'L'، فإن AB(1+i-j,j)    = A(i,j) لـ j<=i<=min(n,j+kd).
!>

LDAB

!>          LDAB هو INTEGER
!>          البعد الرئيسي للمصفوفة AB.  LDAB >= KD+1.
!>

SCALE

!>          SCALE هو DOUBLE PRECISION
!>          العامل السلمي s المستخدم في حل النظام المثلثي.
!>

CNORM

!>          CNORM هو مصفوفة DOUBLE PRECISION، بأبعاد (N)
!>          المعايير 1 لأعمدة A، بدون حساب القطر.
!>

TSCAL

!>          TSCAL هو DOUBLE PRECISION
!>          العامل السلمي المستخدم في حساب المعايير 1 في CNORM.
!>          CNORM تحتوي فعليًا على معايير الأعمدة لـ TSCAL*A.
!>

X

!>          X هي مصفوفة DOUBLE PRECISION، بأبعاد (LDX,NRHS)
!>          متجهات الحل المحسوبة لنظام المعادلات
!>          الخطية.
!>

LDX

!>          LDX هو INTEGER
!>          البعد الرئيسي للمصفوفة X.  LDX >= max(1,N).
!>

B

!>          B هي مصفوفة DOUBLE PRECISION، بأبعاد (LDB,NRHS)
!>          متجهات الطرف الأيمن لنظام المعادلات
!>          الخطية.
!>

LDB

!>          LDB هو INTEGER
!>          البعد الرئيسي للمصفوفة B.  LDB >= max(1,N).
!>

WORK

!>          WORK هي مصفوفة DOUBLE PRECISION، بأبعاد (N)
!>

RESID

!>          RESID هو DOUBLE PRECISION
!>          الحد الأقصى على عدد الأطراف اليمنى لـ
!>          norm(op(A)*x - s*b) / ( norm(op(A)) * norm(x) * EPS ).
!>
