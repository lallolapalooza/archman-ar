subroutine dtbrfs (character uplo, character trans, character diag, integer n, integer kd, integer nrhs, double precision, dimension( ldab, * ) ab, integer ldab, double precision, dimension( ldb, * ) b, integer ldb, double precision, dimension( ldx, * ) x, integer ldx, double precision, dimension( * ) ferr, double precision, dimension( * ) berr, double precision, dimension( * ) work, integer, dimension( * ) iwork, integer info)

DTBRFS

الغرض:

!>
!> يوفر DTBRFS حدود الخطأ وتقديرات الخطأ الخلفي لحل نظام المعادلات الخطية
!> مع مصفوفة معاملية ذات نطاق مثلثي.
!>
!> يجب حساب مصفوفة الحل X بواسطة DTBTRS أو بطريقة أخرى
!> قبل الدخول إلى هذا الروتين. DTBRFS لا يقوم بالتحسين التكراري
!> لأنه لا يمكنه تحسين الخطأ الخلفي.
!>

المعلمات

UPLO 
!>          UPLO is CHARACTER*1
!>          = 'U':  A is upper triangular; (A مثلثة عليا)
!>          = 'L':  A is lower triangular. (A مثلثة سفلى)
!>

TRANS

!>          TRANS is CHARACTER*1
!>          Specifies the form of the system of equations: (يحدد شكل نظام المعادلات:)
!>          = 'N':  A * X = B  (No transpose) (بدون تبديل)
!>          = 'T':  A**T * X = B  (Transpose) (تبديل)
!>          = 'C':  A**H * X = B  (Conjugate transpose = Transpose) (تبديل مرافق = تبديل)
!>

DIAG

!>          DIAG is CHARACTER*1
!>          = 'N':  A is non-unit triangular; (A غير أحادية مثلثة)
!>          = 'U':  A is unit triangular. (A أحادية مثلثة)
!>

N

!>          N is INTEGER
!>          The order of the matrix A.  N >= 0. (رتبة المصفوفة A. N >= 0.)
!>

KD

!>          KD is INTEGER
!>          The number of superdiagonals or subdiagonals of the
!>          triangular band matrix A.  KD >= 0. (عدد الأقطار العليا أو السفلى للمصفوفة A ذات النطاق المثلثي. KD >= 0.)
!>

NRHS

!>          NRHS is INTEGER
!>          The number of right hand sides, i.e., the number of columns
!>          of the matrices B and X.  NRHS >= 0. (عدد الأطراف اليمنى، أي عدد أعمدة المصفوفتين B و X. NRHS >= 0.)
!>

AB

!>          AB is DOUBLE PRECISION array, dimension (LDAB,N)
!>          The upper or lower triangular band matrix A, stored in the
!>          first kd+1 rows of the array. The j-th column of A is stored
!>          in the j-th column of the array AB as follows:
!>          if UPLO = 'U', AB(kd+1+i-j,j) = A(i,j) for max(1,j-kd)<=i<=j;
!>          if UPLO = 'L', AB(1+i-j,j)    = A(i,j) for j<=i<=min(n,j+kd).
!>          If DIAG = 'U', the diagonal elements of A are not referenced
!>          and are assumed to be 1.
!>          (مصفوفة النطاق المثلثي العلوي أو السفلي A، مخزنة في
!>          الصفوف kd+1 الأولى من المصفوفة. يتم تخزين العمود j من A
!>          في العمود j من المصفوفة AB كالتالي:
!>          إذا كان UPLO = 'U'، فإن AB(kd+1+i-j,j) = A(i,j) عندما يكون max(1,j-kd)<=i<=j؛
!>          إذا كان UPLO = 'L'، فإن AB(1+i-j,j) = A(i,j) عندما يكون j<=i<=min(n,j+kd).
!>          إذا كان DIAG = 'U'، فإن عناصر القطر الرئيسي لـ A غير مرجعية
!>          ويفترض أن تكون 1.)
!>

LDAB

!>          LDAB is INTEGER
!>          The leading dimension of the array AB.  LDAB >= KD+1. (البعد الرئيسي للمصفوفة AB. LDAB >= KD+1.)
!>

B

!>          B is DOUBLE PRECISION array, dimension (LDB,NRHS)
!>          The right hand side matrix B. (مصفوفة الطرف الأيمن B.)
!>

LDB

!>          LDB is INTEGER
!>          The leading dimension of the array B.  LDB >= max(1,N). (البعد الرئيسي للمصفوفة B. LDB >= max(1,N).)
!>

X

!>          X is DOUBLE PRECISION array, dimension (LDX,NRHS)
!>          The solution matrix X. (مصفوفة الحل X.)
!>

LDX

!>          LDX is INTEGER
!>          The leading dimension of the array X.  LDX >= max(1,N). (البعد الرئيسي للمصفوفة X. LDX >= max(1,N).)
!>

FERR

!>          FERR is DOUBLE PRECISION array, dimension (NRHS)
!>          The estimated forward error bound for each solution vector
!>          X(j) (the j-th column of the solution matrix X).
!>          If XTRUE is the true solution corresponding to X(j), FERR(j)
!>          is an estimated upper bound for the magnitude of the largest
!>          element in (X(j) - XTRUE) divided by the magnitude of the
!>          largest element in X(j).  The estimate is as reliable as
!>          the estimate for RCOND, and is almost always a slight
!>          overestimate of the true error.
!>          (حد الخطأ الأمامي المقدر لكل متجه حل
!>          X(j) (العمود j من مصفوفة الحل X).
!>          إذا كان XTRUE هو الحل الحقيقي المقابل لـ X(j)، فإن FERR(j)
!>          هو حد علوي مقدر لحجم أكبر عنصر في (X(j) - XTRUE) مقسومًا على حجم
!>          أكبر عنصر في X(j). التقدير موثوق به مثل
!>          التقدير لـ RCOND، وهو دائمًا تقريبًا تقدير زائد طفيف للخطأ الحقيقي.)
!>

BERR

!>          BERR is DOUBLE PRECISION array, dimension (NRHS)
!>          The componentwise relative backward error of each solution
!>          vector X(j) (i.e., the smallest relative change in
!>          any element of A or B that makes X(j) an exact solution).
!>          (خطأ الخلفية النسبي لكل عنصر لمتجه الحل
!>          X(j) (أي، أصغر تغيير نسبي في
!>          أي عنصر في A أو B يجعل X(j) حلاً دقيقًا).)
!>

WORK

!>          WORK is DOUBLE PRECISION array, dimension (3*N) (مصفوفة WORK من النوع DOUBLE PRECISION، البعد (3*N))
!>

IWORK

!>          IWORK is INTEGER array, dimension (N) (مصفوفة IWORK من النوع INTEGER، البعد (N))
!>

INFO

!>          INFO is INTEGER
!>          = 0:  successful exit (نجاح الخروج)
!>          < 0:  if INFO = -i, the i-th argument had an illegal value (إذا كان INFO = -i، فإن المعلمة i كانت لها قيمة غير قانونية)
!>
