توثيق الوظيفة/الروتين الفرعي

subroutine dtrsyl (character trana, character tranb, integer isgn, integer m, integer n, double precision, dimension( lda, * ) a, integer lda, double precision, dimension( ldb, * ) b, integer ldb, double precision, dimension( ldc, * ) c, integer ldc, double precision scale, integer info)

DTRSYL

الغرض:

!>
!> DTRSYL يحل معادلة المصفوفة السيلفسترية الحقيقية:
!>
!>    op(A)*X + X*op(B) = scale*C أو
!>    op(A)*X - X*op(B) = scale*C,
!>
!> حيث op(A) = A أو A**T، و A و B كلاهما شبه مثلثي علوي.
!> A هي M-by-M و B هي N-by-N؛ C الجانب الأيمن و X الحل هما M-by-N؛
!> و scale هو عامل قياس إخراج، يتم تعيينه <= 1 لتجنب تجاوز السعة في X.
!>
!> يجب أن تكون A و B في شكل شور القياسي (كما تم إرجاعه بواسطة DHSEQR)،
!> أي، شبه مثلثي كتلي مع كتل قطرية 1x1 و 2x2؛ كل كتلة قطرية 2x2
!> لها عناصر قطرية متساوية وعناصر خارج القطر بإشارة معاكسة.
!>

المعلمات

TRANA 
!>          TRANA هو CHARACTER*1
!>          يحدد خيار op(A):
!>          = 'N': op(A) = A    (لا يوجد تبديل)
!>          = 'T': op(A) = A**T (تبديل)
!>          = 'C': op(A) = A**H (تبديل مرافق = تبديل)
!>

TRANB

!>          TRANB هو CHARACTER*1
!>          يحدد خيار op(B):
!>          = 'N': op(B) = B    (لا يوجد تبديل)
!>          = 'T': op(B) = B**T (تبديل)
!>          = 'C': op(B) = B**H (تبديل مرافق = تبديل)
!>

ISGN

!>          ISGN هو INTEGER
!>          يحدد الإشارة في المعادلة:
!>          = +1: حل op(A)*X + X*op(B) = scale*C
!>          = -1: حل op(A)*X - X*op(B) = scale*C
!>

M

!>          M هو INTEGER
!>          رتبة المصفوفة A، وعدد الصفوف في المصفوفتين X و C. M >= 0.
!>

N

!>          N هو INTEGER
!>          رتبة المصفوفة B، وعدد الأعمدة في المصفوفتين X و C. N >= 0.
!>

A

!>          A هي مصفوفة DOUBLE PRECISION، البعد (LDA,M)
!>          المصفوفة شبه المثلثية العلوية A، في شكل شور القياسي.
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرئيسي للمصفوفة A. LDA >= max(1,M).
!>

B

!>          B هي مصفوفة DOUBLE PRECISION، البعد (LDB,N)
!>          المصفوفة شبه المثلثية العلوية B، في شكل شور القياسي.
!>

LDB

!>          LDB هو INTEGER
!>          البعد الرئيسي للمصفوفة B. LDB >= max(1,N).
!>

C

!>          C هي مصفوفة DOUBLE PRECISION، البعد (LDC,N)
!>          عند الدخول، مصفوفة C ذات الـ M-by-N للجانب الأيمن.
!>          عند الخروج، يتم استبدال C بالمصفوفة X للحل.
!>

LDC

!>          LDC هو INTEGER
!>          البعد الرئيسي للمصفوفة C. LDC >= max(1,M)
!>

SCALE

!>          SCALE هو DOUBLE PRECISION
!>          عامل القياس، scale، يتم تعيينه <= 1 لتجنب تجاوز السعة في X.
!>

INFO

!>          INFO هو INTEGER
!>          = 0: خروج ناجح
!>          < 0: إذا كان INFO = -i، فإن المعامل i لديه قيمة غير قانونية
!>          = 1: A و B لهما قيم ذاتية مشتركة أو قريبة جدًا؛ تم استخدام قيم
!>               مضطربة لحل المعادلة (لكن المصفوفتين A و B لم تتغير).
!>
