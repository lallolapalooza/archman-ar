توثيق الوظيفة/الروتين الفرعي

subroutine dtrt01 (character uplo, character diag, integer n, double precision, dimension( lda, * ) a, integer lda, double precision, dimension( ldainv, * ) ainv, integer ldainv, double precision rcond, double precision, dimension( * ) work, double precision resid)

DTRT01

الغرض:

!>
!> DTRT01 يحسب الباقي لمصفوفة مثلثية A مضروبة في معكوسها:
!>    RESID = norm( A*AINV - I ) / ( N * norm(A) * norm(AINV) * EPS ),
!> حيث EPS هو ثابت آلة الحساب.
!>

المعلمات

UPLO 
!>          UPLO is CHARACTER*1
!>          يحدد ما إذا كانت المصفوفة A علوية أو سفلية مثلثية.
!>          = 'U':  علوية
!>          = 'L':  سفلية
!>

DIAG

!>          DIAG is CHARACTER*1
!>          يحدد ما إذا كانت المصفوفة A وحدوية أم لا.
!>          = 'N':  غير وحدوية
!>          = 'U':  وحدوية
!>

N

!>          N is INTEGER
!>          ترتيب المصفوفة A.  N >= 0.
!>

A

!>          A is DOUBLE PRECISION array, dimension (LDA,N)
!>          المصفوفة المثلثية A.  إذا كانت UPLO = 'U'، فإن الجزء العلوي
!>          المثلثي العلوي الذي مساحته n × n من الصفيف A يحتوي على
!>          المصفوفة المثلثية العلوية، ولا تتم الإشارة إلى الجزء السفلي
!>          المثلثي الصارم لـ A.  إذا كانت UPLO = 'L'، فإن الجزء السفلي
!>          المثلثي السفلي الذي مساحته n × n من الصفيف A يحتوي على
!>          المصفوفة المثلثية السفلية، ولا تتم الإشارة إلى الجزء العلوي
!>          المثلثي الصارم لـ A.  إذا كانت DIAG = 'U'، فإن العناصر القطرية
!>          لـ A لا تتم الإشارة إليها أيضًا ويُفترض أنها 1.
!>

LDA

!>          LDA is INTEGER
!>          البعد الرئيسي للصفيف A.  LDA >= max(1,N).
!>

AINV

!>          AINV is DOUBLE PRECISION array, dimension (LDAINV,N)
!>          عند الإدخال، معكوس المصفوفة A (المثلثية)، بنفس تنسيق التخزين
!>          مثل A.
!>          عند الخروج، يتم تدمير محتويات AINV.
!>

LDAINV

!>          LDAINV is INTEGER
!>          البعد الرئيسي للصفيف AINV.  LDAINV >= max(1,N).
!>

RCOND

!>          RCOND is DOUBLE PRECISION
!>          المقلوب المعامل الشرطي لـ A، المحسوب كـ
!>          1/(norm(A) * norm(AINV)).
!>

WORK

!>          WORK is DOUBLE PRECISION array, dimension (N)
!>

RESID

!>          RESID is DOUBLE PRECISION
!>          norm(A*AINV - I) / ( N * norm(A) * norm(AINV) * EPS )
!>
