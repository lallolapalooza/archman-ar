subroutine dtzrzf (integer m, integer n, double precision, dimension( lda, * ) a, integer lda, double precision, dimension( * ) tau, double precision, dimension( * ) work, integer lwork, integer info)

DTZRZF

الغرض:

!>
!> DTZRZF يختزل مصفوفة A ذات الشكل شبه المنحرف العلوي M-by-N ( M<=N )
!> إلى شكل مثلثي علوي بواسطة تحويلات متعامدة.
!>
!> يتم تحليل المصفوفة شبه المنحرفة العلوية A إلى
!>
!>    A = ( R  0 ) * Z,
!>
!> حيث Z هي مصفوفة متعامدة N-by-N و R هي مصفوفة مثلثية علوية M-by-M.
!>

المعلمات

M 
!>          M هو عدد صحيح (INTEGER)
!>          عدد صفوف المصفوفة A.  M >= 0.
!>

N

!>          N هو عدد صحيح (INTEGER)
!>          عدد أعمدة المصفوفة A.  N >= M.
!>

A

!>          A هي مصفوفة ذات دقة مزدوجة (DOUBLE PRECISION)، أبعادها (LDA,N)
!>          عند الدخول، يجب أن يحتوي الجزء العلوي الشبه منحرف من A بأبعاد M-by-N
!>          على المصفوفة المراد تحليلها.
!>          عند الخروج، يحتوي الجزء المثلثي العلوي M-by-M من A
!>          على المصفوفة المثلثية العلوية R، والعناصر من M+1 إلى
!>          N في الصفوف الأولى من A، مع المصفوفة TAU، تمثل
!>          المصفوفة المتعامدة Z كناتج لـ M عاكسات أساسية.
!>

LDA

!>          LDA هو عدد صحيح (INTEGER)
!>          البعد الرائد للمصفوفة A.  LDA >= max(1,M).
!>

TAU

!>          TAU هي مصفوفة ذات دقة مزدوجة (DOUBLE PRECISION)، أبعادها (M)
!>          عوامل قياسية للعواكس الأساسية.
!>

WORK

!>          WORK هي مصفوفة ذات دقة مزدوجة (DOUBLE PRECISION)، أبعادها (MAX(1,LWORK))
!>          عند الخروج، إذا كان INFO = 0، فإن WORK(1) تعيد قيمة LWORK المثلى.
!>

LWORK

!>          LWORK هو عدد صحيح (INTEGER)
!>          بعد المصفوفة WORK.  LWORK >= max(1,M).
!>          للأداء الأمثل LWORK >= M*NB، حيث NB هو
!>          حجم الكتلة الأمثل.
!>
!>          إذا كان LWORK = -1، فيفترض استعلام عن مساحة العمل؛ الروتين
!>          يحسب فقط الحجم الأمثل للمصفوفة WORK، ويعيد
!>          هذه القيمة كإدخال أول في المصفوفة WORK، ولا يصدر رسالة خطأ
!>          متعلقة بـ LWORK بواسطة XERBLA.
!>

INFO

!>          INFO هو عدد صحيح (INTEGER)
!>          = 0:  انتهاء ناجح
!>          < 0:  إذا كان INFO = -i، فقد كانت الوسيطة i ذات قيمة غير قانونية
!>

تفاصيل إضافية:

!>
!>  يمكن حساب المصفوفة N-by-N Z عن طريق
!>
!>     Z =  Z(1)*Z(2)* ... *Z(M)
!>
!>  حيث كل Z(k) بحجم N-by-N تُعطى بواسطة
!>
!>     Z(k) = I - tau(k)*v(k)*v(k)**T
!>
!>  مع v(k) هو متجه الصف k للمصفوفة M-by-N
!>
!>     V = ( I   A(:,M+1:N) )
!>
!>  I هي مصفوفة الوحدة M-by-M، A(:,M+1:N)
!>  هو الناتج المخزن في A عند الخروج من DTZRZF،
!>  و tau(k) هو العنصر k من المصفوفة TAU.
!>
!>
