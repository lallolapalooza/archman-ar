subroutine dorm2l (character side, character trans, integer m, integer n, integer k, double precision, dimension( lda, * ) a, integer lda, double precision, dimension( * ) tau, double precision, dimension( ldc, * ) c, integer ldc, double precision, dimension( * ) work, integer info)

DORM2L تضرب مصفوفة عامة بمصفوفة متعامدة ناتجة عن تحليل QL تم تحديده بواسطة sgeqlf (خوارزمية غير محظورة).

الغرض:

!>
!> DORM2L تستبدل المصفوفة العامة الحقيقية m x n، C، بـ:
!>
!>       Q * C  إذا كانت SIDE = 'L' و TRANS = 'N'، أو
!>
!>       Q**T * C  إذا كانت SIDE = 'L' و TRANS = 'T'، أو
!>
!>       C * Q  إذا كانت SIDE = 'R' و TRANS = 'N'، أو
!>
!>       C * Q**T إذا كانت SIDE = 'R' و TRANS = 'T'،
!>
!> حيث Q مصفوفة متعامدة حقيقية معرفة كناتج لـ k
!> عاكسات أولية
!>
!>       Q = H(k) . . . H(2) H(1)
!>
!> كما تم إرجاعها بواسطة DGEQLF. Q لها ترتيب m إذا كانت SIDE = 'L' وترتيب n
!> إذا كانت SIDE = 'R'.
!>

المعلمات

SIDE 
!>          SIDE هي CHARACTER*1
!>          = 'L': تطبيق Q أو Q**T من اليسار
!>          = 'R': تطبيق Q أو Q**T من اليمين
!>

TRANS

!>          TRANS هي CHARACTER*1
!>          = 'N': تطبيق Q (بدون تبديل)
!>          = 'T': تطبيق Q**T (تبديل)
!>

M

!>          M هي INTEGER
!>          عدد صفوف المصفوفة C. M >= 0.
!>

N

!>          N هي INTEGER
!>          عدد أعمدة المصفوفة C. N >= 0.
!>

K

!>          K هي INTEGER
!>          عدد العاكسات الأولية التي يحدد حاصل ضربها
!>          المصفوفة Q.
!>          إذا كانت SIDE = 'L'، فإن M >= K >= 0؛
!>          إذا كانت SIDE = 'R'، فإن N >= K >= 0.
!>

A

!>          A هي DOUBLE PRECISION array، dimension (LDA,K)
!>          يجب أن يحتوي العمود i على المتجه الذي يحدد
!>          العاكس الأولي H(i)، لكل i = 1، 2، ...، k، كما تم إرجاعه بواسطة
!>          DGEQLF في الأعمدة k الأخيرة من مصفوفة الوسائط A الخاصة به.
!>

LDA

!>          LDA هي INTEGER
!>          البعد الرائد للمصفوفة A.
!>          إذا كانت SIDE = 'L'، فإن LDA >= max(1,M)؛
!>          إذا كانت SIDE = 'R'، فإن LDA >= max(1,N).
!>

TAU

!>          TAU هي DOUBLE PRECISION array، dimension (K)
!>          يجب أن تحتوي TAU(i) على عامل القياس للعاكس
!>          الأولي H(i)، كما تم إرجاعه بواسطة DGEQLF.
!>

C

!>          C هي DOUBLE PRECISION array، dimension (LDC,N)
!>          عند الإدخال، المصفوفة C بحجم m x n.
!>          عند الخروج، يتم استبدال C بـ Q*C أو Q**T*C أو C*Q**T أو C*Q.
!>

LDC

!>          LDC هي INTEGER
!>          البعد الرائد للمصفوفة C. LDC >= max(1,M).
!>

WORK

!>          WORK هي DOUBLE PRECISION array، dimension
!>                                   (N) إذا كانت SIDE = 'L'،
!>                                   (M) إذا كانت SIDE = 'R'
!>

INFO

!>          INFO هي INTEGER
!>          = 0: خروج ناجح
!>          < 0: إذا كان INFO = -i، فإن الوسيط i كان لديه قيمة غير قانونية
!>
