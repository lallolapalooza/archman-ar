subroutine dorgtsqr_row (integer m, integer n, integer mb, integer nb, double precision, dimension( lda, * ) a, integer lda, double precision, dimension( ldt, * ) t, integer ldt, double precision, dimension( * ) work, integer lwork, integer info)

DORGTSQR_ROW

الغرض:

!>
!> DORGTSQR_ROW تنشئ مصفوفة حقيقية M-by-N Q_out بأعمدة متعامدة
!> من خرج DLATSQR. هذه الأعمدة المتعاطفة N هي الأعمدة الأولى
!> لناتج المصفوفات المتعامدة المعقدة Q(k)_in من الرتبة M،
!> والتي يتم إرجاعها بواسطة DLATSQR بتنسيق خاص.
!>
!>      Q_out = أول_أعمدة_N( Q(1)_in * Q(2)_in * ... * Q(k)_in ).
!>
!> يتم تخزين المصفوفات المدخلة Q(k)_in في كتل صفية وعمودية في A.
!> انظر توثيق DLATSQR لمزيد من التفاصيل حول تنسيق
!> Q(k)_in، حيث يتم تمثيل كل Q(k)_in بواسطة تحويلات
!> هاوسهولدر كتلية. تستدعي هذه الروتين روتيناً مساعداً DLARFB_GETT،
!> حيث يتم أداء الحساب على كل كتلة فردية. تقوم الخوارزمية
!> أولاً بمسح كتل الأعمدة ذات الحجم NB من اليمين إلى اليسار
!> بدءاً من كتلة الصف السفلي وتستمر إلى كتلة الصف العلوي
!> (لذلك _ROW في اسم الروتين). هذا المسح بترتيب عكسي
!> لترتيب إنشاء كتل الخرج بواسطة DLATSQR.
!>

المعلمات

M 
!>          M هو عدد صحيح (INTEGER)
!>          عدد صفوف المصفوفة A.  M >= 0.
!>

N

!>          N هو عدد صحيح (INTEGER)
!>          عدد أعمدة المصفوفة A. M >= N >= 0.
!>

MB

!>          MB هو عدد صحيح (INTEGER)
!>          حجم كتلة الصف المستخدم بواسطة DLATSQR لإرجاع
!>          المصفوفات A و T. MB > N.
!>          (لاحظ أنه إذا كان MB > M، يتم استخدام M بدلاً من MB
!>          كحجم كتلة الصف).
!>

NB

!>          NB هو عدد صحيح (INTEGER)
!>          حجم كتلة العمود المستخدم بواسطة DLATSQR لإرجاع
!>          المصفوفات A و T. NB >= 1.
!>          (لاحظ أنه إذا كان NB > N، يتم استخدام N بدلاً من NB
!>          كحجم كتلة العمود).
!>

A

!>          A هي مصفوفة مزدوجة الدقة (DOUBLE PRECISION)، البعد (LDA,N)
!>
!>          عند الإدخال:
!>
!>             العناصر على القطر والأعلى منه لا تستخدم كمدخلات.
!>             العناصر تحت القطر تمثل المصفوفة السفلية شبه القطرية
!>             الموحدة V المحسوبة بواسطة DLATSQR والتي تحدد
!>             المصفوفات المدخلة Q_in(k) (الواحدات على القطر لا يتم تخزينها).
!>             انظر DLATSQR لمزيد من التفاصيل.
!>
!>          عند الخروج:
!>
!>             تحتوي المصفوفة A على مصفوفة متعامدة M-by-N Q_out،
!>             أي أن أعمدة A هي متجهات وحدة متعامدة.
!>

LDA

!>          LDA هو عدد صحيح (INTEGER)
!>          البعد الرئيسي للمصفوفة A.  LDA >= max(1,M).
!>

T

!>          T هي مصفوفة مزدوجة الدقة (DOUBLE PRECISION)،
!>          البعد (LDT, N * NIRB)
!>          حيث NIRB = عدد كتل الصف المدخلة
!>                     = MAX( 1, CEIL((M-N)/(MB-N)) )
!>          ليكن NICB = عدد كتل العمود المدخلة
!>                   = CEIL(N/NB)
!>
!>          المحددات الكتلية المثلثية العلوية المستخدمة لتحديد
!>          المصفوفات المدخلة Q_in(k)، k=(1:NIRB*NICB). يتم تخزين
!>          المحددات الكتلية بتنسيق مضغوط في NIRB من تسلسلات
!>          المحددات الكتلية. يتم تخزين كل تسلسل من تسلسلات المحددات
!>          الكتلية NIRB في كتلة عمودية NB-by-N أكبر من T
!>          وتتكون من NICB من كتل عمودية أصغر NB-by-NB
!>          مثلثة علوية. انظر DLATSQR لمزيد من التفاصيل حول تنسيق
!>          T.
!>

LDT

!>          LDT هو عدد صحيح (INTEGER)
!>          البعد الرئيسي للمصفوفة T.
!>          LDT >= max(1,min(NB,N)).
!>

WORK

!>          (مساحة عمل) مصفوفة مزدوجة الدقة (DOUBLE PRECISION)، البعد (MAX(1,LWORK))
!>          عند الخروج، إذا كان INFO = 0، تعيد WORK(1) قيمة LWORK المثلى.
!>

LWORK

!>          LWORK هو عدد صحيح (INTEGER)
!>          بعد المصفوفة WORK.
!>          LWORK >= NBLOCAL * MAX(NBLOCAL,(N-NBLOCAL))،
!>          حيث NBLOCAL=MIN(NB,N).
!>          إذا كان LWORK = -1، فإنه يُفترض استعلام عن مساحة عمل.
!>          تقوم الروتين بحساب حجم مساحة العمل المثلى فقط،
!>          وتعيد هذه القيمة كالإدخال الأول لمصفوفة WORK،
!>          ولا تصدر رسالة خطأ متعلقة بـ LWORK بواسطة XERBLA.
!>

INFO

!>          INFO هو عدد صحيح (INTEGER)
!>          = 0:  خروج ناجح
!>          < 0:  إذا كان INFO = -i، فإن الوسيط i كان له قيمة غير قانونية
!>
