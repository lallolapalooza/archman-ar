توثيق الوظيفة/الإجراء الفرعي

subroutine dorglq (integer m, integer n, integer k, double precision, dimension( lda, * ) a, integer lda, double precision, dimension( * ) tau, double precision, dimension( * ) work, integer lwork, integer info)

DORGLQ

الغرض:

!>
!> DORGLQ يقوم بإنشاء مصفوفة Q حقيقية من النوع M x N ذات صفوف متعامدة
!> ، والتي تُعرّف كأول M صف من حاصل ضرب K من المرايا الأولية من الرتبة N
!>
!>       Q  =  H(k) . . . H(2) H(1)
!>
!> كما تم إرجاعها بواسطة DGELQF.
!>

المعلمات

M 
!>          M هو عدد صحيح (INTEGER)
!>          عدد صفوف المصفوفة Q. M >= 0.
!>

N

!>          N هو عدد صحيح (INTEGER)
!>          عدد أعمدة المصفوفة Q. N >= M.
!>

K

!>          K هو عدد صحيح (INTEGER)
!>          عدد المرايا الأولية التي يحدد حاصل ضربها المصفوفة Q.
!>          M >= K >= 0.
!>

A

!>          A هو مصفوفة ذات دقة مزدوجة (DOUBLE PRECISION array)، البعد (LDA,N)
!>          عند الإدخال، يجب أن يحتوي الصف i على المتجه الذي يحدد
!>          المرآة الأولية H(i)، للصف i = 1,2,...,k، كما تم إرجاعها
!>          بواسطة DGELQF في أول K صف من مصفوفة الوسائط A الخاصة بها.
!>          عند الخروج، المصفوفة Q من النوع M x N.
!>

LDA

!>          LDA هو عدد صحيح (INTEGER)
!>          البعد الأول للمصفوفة A. LDA >= max(1,M).
!>

TAU

!>          TAU هي مصفوفة ذات دقة مزدوجة (DOUBLE PRECISION array)، البعد (K)
!>          يجب أن تحتوي TAU(i) على العامل القياسي للمرآة الأولية
!>          H(i)، كما تم إرجاعها بواسطة DGELQF.
!>

WORK

!>          WORK هي مصفوفة ذات دقة مزدوجة (DOUBLE PRECISION array)، البعد (MAX(1,LWORK))
!>          عند الخروج، إذا كان INFO = 0، فإن WORK(1) ترجع قيمة LWORK المثلى.
!>

LWORK

!>          LWORK هو عدد صحيح (INTEGER)
!>          بعد المصفوفة WORK. LWORK >= max(1,M).
!>          للحصول على الأداء الأمثل، LWORK >= M*NB ، حيث NB
!>          هو حجم الكتلة الأمثل.
!>
!>          إذا كان LWORK = -1 ، فيُفترض استعلام مساحة العمل؛ تقوم الروتين
!>          فقط بحساب الحجم الأمثل للمصفوفة WORK، وترجع
!>          هذه القيمة كالإدخال الأول للمصفوفة WORK، ولا يصدر أي رسالة خطأ
!>          متعلقة بـ LWORK بواسطة XERBLA.
!>

INFO

!>          INFO هو عدد صحيح (INTEGER)
!>          = 0:  الخروج بنجاح
!>          < 0:  إذا كان INFO = -i، فإن الوسيط i له قيمة غير قانونية
!>
