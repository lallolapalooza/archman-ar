subroutine dormlq (character side, character trans, integer m, integer n, integer k, double precision, dimension( lda, * ) a, integer lda, double precision, dimension( * ) tau, double precision, dimension( ldc, * ) c, integer ldc, double precision, dimension( * ) work, integer lwork, integer info)

DORMLQ

الغرض:

!>
!> DORMLQ يستبدل المصفوفة الحقيقية C ذات الأبعاد M*N بـ
!>
!>                 SIDE = 'L'     SIDE = 'R'
!> TRANS = 'N':      Q * C          C * Q
!> TRANS = 'T':      Q**T * C       C * Q**T
!>
!> حيث Q مصفوفة متعامدة حقيقية معرفة كناتج لـ k
!> عاكسات أولية
!>
!>       Q = H(k) . . . H(2) H(1)
!>
!> كما تم إرجاعه بواسطة DGELQF. Q من الرتبة M إذا كانت SIDE = 'L' ومن الرتبة N
!> إذا كانت SIDE = 'R'.
!>

المعلمات

SIDE 
!>          SIDE هو CHARACTER*1
!>          = 'L': تطبيق Q أو Q**T من اليسار؛
!>          = 'R': تطبيق Q أو Q**T من اليمين.
!>

TRANS

!>          TRANS هو CHARACTER*1
!>          = 'N':  بدون تبديل، تطبيق Q؛
!>          = 'T':  تبديل، تطبيق Q**T.
!>

M

!>          M هو INTEGER
!>          عدد صفوف المصفوفة C. M >= 0.
!>

N

!>          N هو INTEGER
!>          عدد أعمدة المصفوفة C. N >= 0.
!>

K

!>          K هو INTEGER
!>          عدد العاكسات الأولية التي ينتج عنها حاصل الضرب الذي يحدد
!>          المصفوفة Q.
!>          إذا كانت SIDE = 'L'، فإن M >= K >= 0؛
!>          إذا كانت SIDE = 'R'، فإن N >= K >= 0.
!>

A

!>          A هو مصفوفة DOUBLE PRECISION، بأبعاد
!>                               (LDA,M) إذا كانت SIDE = 'L'،
!>                               (LDA,N) إذا كانت SIDE = 'R'
!>          الصف i يجب أن يحتوي على المتجه الذي يحدد
!>          العاكس الأولي H(i)، لـ i = 1,2,...,k، كما تم إرجاعه بواسطة
!>          DGELQF في الصفوف k الأولى من مصفوفة الوسيط A.
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرائد للمصفوفة A. LDA >= max(1,K).
!>

TAU

!>          TAU هو مصفوفة DOUBLE PRECISION، بأبعاد (K)
!>          يجب أن يحتوي TAU(i) على العامل القياسي للعكس
!>          الأولي H(i)، كما تم إرجاعه بواسطة DGELQF.
!>

C

!>          C هو مصفوفة DOUBLE PRECISION، بأبعاد (LDC,N)
!>          عند الدخول، مصفوفة C ذات الأبعاد M*N.
!>          عند الخروج، يتم استبدال C بـ Q*C أو Q**T*C أو C*Q**T أو C*Q.
!>

LDC

!>          LDC هو INTEGER
!>          البعد الرائد للمصفوفة C. LDC >= max(1,M).
!>

WORK

!>          WORK هو مصفوفة DOUBLE PRECISION، بأبعاد (MAX(1,LWORK))
!>          عند الخروج، إذا كان INFO = 0، فإن WORK(1) يرجع LWORK الأمثل.
!>

LWORK

!>          LWORK هو INTEGER
!>          بعد مصفوفة WORK.
!>          إذا كانت SIDE = 'L'، فإن LWORK >= max(1,N)؛
!>          إذا كانت SIDE = 'R'، فإن LWORK >= max(1,M).
!>          لأداء جيد، يجب أن يكون LWORK أكبر بشكل عام.
!>
!>          إذا كانت LWORK = -1، فيفترض استعلام مساحة العمل؛ الروتين
!>          يحسب فقط الحجم الأمثل لمصفوفة WORK، ويرجع
!>          هذه القيمة كمدخل أول لمصفوفة WORK، ولا يتم إصدار أي رسالة خطأ
!>          متعلقة بـ LWORK بواسطة XERBLA.
!>

INFO

!>          INFO هو INTEGER
!>          = 0:  انتهاء ناجح
!>          < 0:  إذا كان INFO = -i، فقد كانت وسيطة i غير قانونية
!>
