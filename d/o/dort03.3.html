subroutine dort03 (character*( * ) rc, integer mu, integer mv, integer n, integer k, double precision, dimension( ldu, * ) u, integer ldu, double precision, dimension( ldv, * ) v, integer ldv, double precision, dimension( * ) work, integer lwork, double precision result, integer info)

DORT03

الغرض:

!>
!> DORT03 يقارن مصفوفتين متعامدتين U و V لمعرفة ما إذا كانت صفوفهما أو أعمدتهما
!> تتمدد في نفس المساحات. يتم فحص الصفوف إذا كان RC = 'R'، ويتم فحص الأعمدة إذا كان RC = 'C'.
!>
!> RESULT هو الحد الأقصى لـ
!>
!>    | V*V' - I | / ( MV ulp )، إذا كان RC = 'R'، أو
!>
!>    | V'*V - I | / ( MV ulp )، إذا كان RC = 'C'،
!>
!> والحد الأقصى للصفوف (أو الأعمدة) من 1 إلى K لـ
!>
!>    | U(i) - S*V(i) |/ ( N ulp )
!>
!> حيث S هي +-1 (مختارة لتقليل التعبير)، U(i) هو الصف (العمود) i من U، و V(i) هو الصف (العمود) i من V.
!>

المعلمات

RC 
!>          RC هو CHARACTER*1
!>          إذا كان RC = 'R' يتم مقارنة صفوف U و V.
!>          إذا كان RC = 'C' يتم مقارنة أعمدة U و V.
!>

MU

!>          MU هو INTEGER
!>          عدد صفوف U إذا كان RC = 'R'، وعدد
!>          الأعمدة إذا كان RC = 'C'. إذا كان MU = 0، فإن DORT03 لا يفعل شيئًا.
!>          يجب أن يكون MU على الأقل صفرًا.
!>

MV

!>          MV هو INTEGER
!>          عدد صفوف V إذا كان RC = 'R'، وعدد
!>          الأعمدة إذا كان RC = 'C'. إذا كان MV = 0، فإن DORT03 لا يفعل شيئًا.
!>          يجب أن يكون MV على الأقل صفرًا.
!>

N

!>          N هو INTEGER
!>          إذا كان RC = 'R'، عدد أعمدة المصفوفتين U و V،
!>          وإذا كان RC = 'C'، عدد صفوف U و V. إذا كان N = 0
!>          فإن DORT03 لا يفعل شيئًا. يجب أن يكون N على الأقل صفرًا.
!>

K

!>          K هو INTEGER
!>          عدد الصفوف أو الأعمدة في U و V للمقارنة.
!>          0 <= K <= max(MU,MV).
!>

U

!>          U هو مصفوفة DOUBLE PRECISION، البعد (LDU,N)
!>          المصفوفة الأولى للمقارنة. إذا كان RC = 'R'، فإن U هي MU × N، و
!>          إذا كان RC = 'C'، فإن U هي N × MU.
!>

LDU

!>          LDU هو INTEGER
!>          البعد الرائد لـ U. إذا كان RC = 'R'، فإن LDU >= max(1,MU)،
!>          وإذا كان RC = 'C'، فإن LDU >= max(1,N).
!>

V

!>          V هو مصفوفة DOUBLE PRECISION، البعد (LDV,N)
!>          المصفوفة الثانية للمقارنة. إذا كان RC = 'R'، فإن V هي MV × N، و
!>          إذا كان RC = 'C'، فإن V هي N × MV.
!>

LDV

!>          LDV هو INTEGER
!>          البعد الرائد لـ V. إذا كان RC = 'R'، فإن LDV >= max(1,MV)،
!>          وإذا كان RC = 'C'، فإن LDV >= max(1,N).
!>

WORK

!>          WORK هي مصفوفة DOUBLE PRECISION، البعد (LWORK)
!>

LWORK

!>          LWORK هو INTEGER
!>          طول المصفوفة WORK. للحصول على أفضل أداء، يجب أن يكون LWORK
!>          على الأقل N*N إذا كان RC = 'C' أو M*M إذا كان RC = 'R'، ولكن
!>          سيتم إجراء الاختبارات حتى لو كان LWORK = 0.
!>

RESULT

!>          RESULT هو DOUBLE PRECISION
!>          القيمة المحسوبة بواسطة الاختبار الموضح أعلاه. يتم
!>          تحديد RESULT بـ 1/ulp لتجنب تجاوز السعة.
!>

INFO

!>          INFO هو INTEGER
!>          0  يشير إلى خروج ناجح
!>          -k يشير إلى أن المعلمة k كانت لها قيمة غير قانونية
!>
