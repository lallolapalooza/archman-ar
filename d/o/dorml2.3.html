subroutine dorml2 (character side, character trans, integer m, integer n, integer k, double precision, dimension( lda, * ) a, integer lda, double precision, dimension( * ) tau, double precision, dimension( ldc, * ) c, integer ldc, double precision, dimension( * ) work, integer info)

DORML2 تضرب مصفوفة عامة بمصفوفة متعامدة من عامل تحلل LQ تم تحديده بواسطة sgelqf (خوارزمية غير محظورة).

الغرض:

!>
!> DORML2 تستبدل مصفوفة C العامة ذات الأبعاد m × n بـ
!>
!>       Q * C  إذا كانت SIDE = 'L' و TRANS = 'N'، أو
!>
!>       Q**T* C  إذا كانت SIDE = 'L' و TRANS = 'T'، أو
!>
!>       C * Q  إذا كانت SIDE = 'R' و TRANS = 'N'، أو
!>
!>       C * Q**T إذا كانت SIDE = 'R' و TRANS = 'T'،
!>
!> حيث Q هي مصفوفة متعامدة حقيقية معرفة كناتج لـ k
!> عواكس أولية
!>
!>       Q = H(k) . . . H(2) H(1)
!>
!> كما تم إرجاعه بواسطة DGELQF. Q من الرتبة m إذا كانت SIDE = 'L' ومن الرتبة n
!> إذا كانت SIDE = 'R'.
!>

المعلمات

SIDE 
!>          SIDE is CHARACTER*1
!>          = 'L': تطبيق Q أو Q**T من اليسار
!>          = 'R': تطبيق Q أو Q**T من اليمين
!>

TRANS

!>          TRANS is CHARACTER*1
!>          = 'N': تطبيق Q  (بدون تبديل)
!>          = 'T': تطبيق Q**T (تبديل)
!>

M

!>          M is INTEGER
!>          عدد صفوف المصفوفة C. M >= 0.
!>

N

!>          N is INTEGER
!>          عدد أعمدة المصفوفة C. N >= 0.
!>

K

!>          K is INTEGER
!>          عدد العواكس الأولية التي يحدد ناتجها
!>          المصفوفة Q.
!>          إذا كانت SIDE = 'L'، فإن M >= K >= 0؛
!>          إذا كانت SIDE = 'R'، فإن N >= K >= 0.
!>

A

!>          A is DOUBLE PRECISION array, dimension
!>                               (LDA,M) if SIDE = 'L',
!>                               (LDA,N) if SIDE = 'R'
!>          يجب أن يحتوي الصف i على المتجه الذي يحدد
!>          العاكس الأولي H(i)، لـ i = 1,2,...,k، كما تم إرجاعه بواسطة
!>          DGELQF في الصفوف k الأولى من مصفوفة الوسيط A الخاصة به.
!>

LDA

!>          LDA is INTEGER
!>          البعد الرئيسي للمصفوفة A. LDA >= max(1,K).
!>

TAU

!>          TAU is DOUBLE PRECISION array, dimension (K)
!>          يجب أن يحتوي TAU(i) على العامل العددي للعكس الأولي
!>          H(i)، كما تم إرجاعه بواسطة DGELQF.
!>

C

!>          C is DOUBLE PRECISION array, dimension (LDC,N)
!>          عند الإدخال، مصفوفة C ذات الأبعاد m × n.
!>          عند الخروج، يتم استبدال C بـ Q*C أو Q**T*C أو C*Q**T أو C*Q.
!>

LDC

!>          LDC is INTEGER
!>          البعد الرئيسي للمصفوفة C. LDC >= max(1,M).
!>

WORK

!>          WORK is DOUBLE PRECISION array, dimension
!>                                   (N) if SIDE = 'L',
!>                                   (M) if SIDE = 'R'
!>

INFO

!>          INFO is INTEGER
!>          = 0: خروج ناجح
!>          < 0: إذا كانت INFO = -i، فإن الوسيط i كان له قيمة غير قانونية
!>
