توثيق الوظيفة/الروتين الفرعي

subroutine dorg2l (integer m, integer n, integer k, double precision, dimension( lda, * ) a, integer lda, double precision, dimension( * ) tau, double precision, dimension( * ) work, integer info)

DORG2L تنشئ جزءًا أو كل المصفوفة المتعامدة Q من تحليل QL الذي حدده sgeqlf (خوارزمية غير محظورة).

الغرض:

!>
!> DORG2L تنشئ مصفوفة Q حقيقية بأبعاد m × n بأعمدة متعامدة،
!> والتي تُعرّف بأنها آخر n أعمدة من حاصل ضرب k عاكسات ابتدائية
!> من الرتبة m
!>
!>       Q  =  H(k) . . . H(2) H(1)
!>
!> كما تم إرجاعها بواسطة DGEQLF.
!>

المعلمات

M 
!>          M هو عدد صحيح (INTEGER)
!>          عدد صفوف المصفوفة Q. M >= 0.
!>

N

!>          N هو عدد صحيح (INTEGER)
!>          عدد أعمدة المصفوفة Q. M >= N >= 0.
!>

K

!>          K هو عدد صحيح (INTEGER)
!>          عدد العاكسات الابتدائية التي يُعرّف حاصل ضربها المصفوفة Q.
!>          N >= K >= 0.
!>

A

!>          A هي مصفوفة ذات دقة مزدوجة (DOUBLE PRECISION)، بأبعاد (LDA,N)
!>          عند الإدخال، يجب أن يحتوي العمود (n-k+i) على المتجه الذي
!>          يُعرّف العاكس الابتدائي H(i)، لكل i = 1,2,...,k، كما
!>          تم إرجاعه بواسطة DGEQLF في آخر k أعمدة من حجته المصفوفية
!>          A.
!>          عند الخروج، المصفوفة Q بأبعاد m × n.
!>

LDA

!>          LDA هو عدد صحيح (INTEGER)
!>          البعد الأول للمصفوفة A. LDA >= max(1,M).
!>

TAU

!>          TAU هي مصفوفة ذات دقة مزدوجة (DOUBLE PRECISION)، بأبعاد (K)
!>          يجب أن يحتوي TAU(i) على العامل العددي للعاكس الابتدائي H(i)،
!>          كما تم إرجاعه بواسطة DGEQLF.
!>

WORK

!>          WORK هي مصفوفة ذات دقة مزدوجة (DOUBLE PRECISION)، بأبعاد (N)
!>

INFO

!>          INFO هو عدد صحيح (INTEGER)
!>          = 0: انتهاء ناجح
!>          < 0: إذا كان INFO = -i، فإن الوسيط i لديه قيمة غير قانونية
!>
