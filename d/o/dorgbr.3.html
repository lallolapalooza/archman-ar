توثيق الوظيفة/الروتين الفرعي

subroutine dorgbr (character vect, integer m, integer n, integer k, double precision, dimension( lda, * ) a, integer lda, double precision, dimension( * ) tau, double precision, dimension( * ) work, integer lwork, integer info)

DORGBR

الغرض:

!>
!> DORGBR يقوم بتوليد إحدى المصفوفات المتعامدة الحقيقية Q أو P**T
!> التي تم تحديدها بواسطة DGEBRD عند اختزال مصفوفة حقيقية A إلى شكل مصفوفة ثنائية القطر: A = Q * B * P**T.  يتم تعريف Q و P**T كناتج ضرب
!> للمنعكسات الأولية H(i) أو G(i) على التوالي.
!>
!> إذا كان VECT = 'Q'، يُفترض أن A كانت مصفوفة M-by-K، وأن Q
!> هي من الدرجة M:
!> إذا كان m >= k، فإن Q = H(1) H(2) . . . H(k) ويعيد DORGBR الأعمدة n الأولى
!> من Q، حيث m >= n >= k؛
!> إذا كان m < k، فإن Q = H(1) H(2) . . . H(m-1) ويعيد DORGBR المصفوفة Q كمصفوفة
!> M-by-M.
!>
!> إذا كان VECT = 'P'، يُفترض أن A كانت مصفوفة K-by-N، وأن P**T
!> هي من الدرجة N:
!> إذا كان k < n، فإن P**T = G(k) . . . G(2) G(1) ويعيد DORGBR الصفوف m الأولى
!> من P**T، حيث n >= m >= k؛
!> إذا كان k >= n، فإن P**T = G(n-1) . . . G(2) G(1) ويعيد DORGBR المصفوفة P**T كمصفوفة
!> N-by-N.
!>

المعاملات

VECT 
!>          VECT هو CHARACTER*1
!>          يحدد ما إذا كانت المصفوفة Q أو المصفوفة P**T مطلوبة، كما هو محدد في التحويل المطبق بواسطة DGEBRD:
!>          = 'Q':  توليد Q؛
!>          = 'P':  توليد P**T.
!>

M

!>          M هو INTEGER
!>          عدد صفوف المصفوفة Q أو P**T المطلوب إعادتها.
!>          M >= 0.
!>

N

!>          N هو INTEGER
!>          عدد أعمدة المصفوفة Q أو P**T المطلوب إعادتها.
!>          N >= 0.
!>          إذا كان VECT = 'Q'، فإن M >= N >= min(M,K)؛
!>          إذا كان VECT = 'P'، فإن N >= M >= min(N,K).
!>

K

!>          K هو INTEGER
!>          إذا كان VECT = 'Q'، فإن عدد أعمدة المصفوفة الأصلية M-by-K
!>          التي تم اختزالها بواسطة DGEBRD.
!>          إذا كان VECT = 'P'، فإن عدد صفوف المصفوفة الأصلية K-by-N
!>          التي تم اختزالها بواسطة DGEBRD.
!>          K >= 0.
!>

A

!>          A هي مصفوفة DOUBLE PRECISION، أبعادها (LDA,N)
!>          عند الدخول، المتجهات التي تحدد المنعكسات الأولية،
!>          كما أعادتها DGEBRD.
!>          عند الخروج، المصفوفة M-by-N Q أو P**T.
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرائد للمصفوفة A. LDA >= max(1,M).
!>

TAU

!>          TAU هي مصفوفة DOUBLE PRECISION، أبعادها
!>                                (min(M,K)) إذا كان VECT = 'Q'
!>                                (min(N,K)) إذا كان VECT = 'P'
!>          يجب أن تحتوي TAU(i) على العامل العددي للمنعكس الأولي
!>          H(i) أو G(i)، والذي يحدد Q أو P**T، كما
!>          أعادته DGEBRD في مصفوفة المعاملات الخاصة بها TAUQ أو TAUP.
!>

WORK

!>          WORK هي مصفوفة DOUBLE PRECISION، أبعادها (MAX(1,LWORK))
!>          عند الخروج، إذا كان INFO = 0، فإن WORK(1) تعيد LWORK الأمثل.
!>

LWORK

!>          LWORK هو INTEGER
!>          بعد المصفوفة WORK. LWORK >= max(1,min(M,N)).
!>          لأداء أمثل، LWORK >= min(M,N)*NB، حيث NB
!>          هو حجم الكتلة الأمثل.
!>
!>          إذا كان LWORK = -1، فيُفترض استعلام عن مساحة العمل؛ يقوم الروتين
!>          فقط بحساب الحجم الأمثل للمصفوفة WORK، ويعيد
!>          هذه القيمة كإدخال أول للمصفوفة WORK، ولا يتم إصدار رسالة خطأ
!>          متعلقة بـ LWORK بواسطة XERBLA.
!>

INFO

!>          INFO هو INTEGER
!>          = 0:  الخروج الناجح
!>          < 0:  إذا كان INFO = -i، فإن المعامل i كان له قيمة غير قانونية
!>
