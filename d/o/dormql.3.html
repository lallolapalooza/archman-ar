subroutine dormql (character side, character trans, integer m, integer n, integer k, double precision, dimension( lda, * ) a, integer lda, double precision, dimension( * ) tau, double precision, dimension( ldc, * ) c, integer ldc, double precision, dimension( * ) work, integer lwork, integer info)

DORMQL

الغرض:

!>
!> DORMQL يستبدل مصفوفة C الحقيقية M-by-N بـ
!>
!>                 SIDE = 'L'     SIDE = 'R'
!> TRANS = 'N':      Q * C          C * Q
!> TRANS = 'T':      Q**T * C       C * Q**T
!>
!> حيث Q هي مصفوفة متعامدة حقيقية معرفة كناتج لـ k
!> عواكس أولية
!>
!>       Q = H(k) . . . H(2) H(1)
!>
!> كما تم إرجاعها بواسطة DGEQLF. Q من الرتبة M إذا كانت SIDE = 'L' ومن الرتبة N
!> إذا كانت SIDE = 'R'.
!>

المعلمات

SIDE 
!>          SIDE هو CHARACTER*1
!>          = 'L': تطبيق Q أو Q**T من اليسار؛
!>          = 'R': تطبيق Q أو Q**T من اليمين.
!>

TRANS

!>          TRANS هو CHARACTER*1
!>          = 'N':  بدون تبديل، تطبيق Q؛
!>          = 'T':  تبديل، تطبيق Q**T.
!>

M

!>          M هو INTEGER
!>          عدد صفوف المصفوفة C. M >= 0.
!>

N

!>          N هو INTEGER
!>          عدد أعمدة المصفوفة C. N >= 0.
!>

K

!>          K هو INTEGER
!>          عدد العواكس الأولية التي يعرّف ناتجها
!>          المصفوفة Q.
!>          إذا كانت SIDE = 'L'، فإن M >= K >= 0؛
!>          إذا كانت SIDE = 'R'، فإن N >= K >= 0.
!>

A

!>          A هو مصفوفة DOUBLE PRECISION، البعد (LDA,K)
!>          يجب أن تحتوي العمود i على المتجه الذي يحدد
!>          العاكس الأولي H(i)، لـ i = 1، 2، ... ، k، كما تم إرجاعه بواسطة
!>          DGEQLF في الأعمدة k الأخيرة من وسيطة المصفوفة A الخاصة به.
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرائد للمصفوفة A.
!>          إذا كانت SIDE = 'L'، فإن LDA >= max(1,M)؛
!>          إذا كانت SIDE = 'R'، فإن LDA >= max(1,N).
!>

TAU

!>          TAU هو مصفوفة DOUBLE PRECISION، البعد (K)
!>          يجب أن يحتوي TAU(i) على عامل القياس للعامل الأولي
!>          H(i)، كما تم إرجاعه بواسطة DGEQLF.
!>

C

!>          C هو مصفوفة DOUBLE PRECISION، البعد (LDC,N)
!>          عند الدخول، المصفوفة M-by-N C.
!>          عند الخروج، يتم استبدال C بـ Q*C أو Q**T*C أو C*Q**T أو C*Q.
!>

LDC

!>          LDC هو INTEGER
!>          البعد الرائد للمصفوفة C. LDC >= max(1,M).
!>

WORK

!>          WORK هي مصفوفة DOUBLE PRECISION، البعد (MAX(1,LWORK))
!>          عند الخروج، إذا كان INFO = 0، فإن WORK(1) يعيد حجم LWORK الأمثل.
!>

LWORK

!>          LWORK هو INTEGER
!>          بعد مصفوفة WORK.
!>          إذا كانت SIDE = 'L'، فإن LWORK >= max(1,N)؛
!>          إذا كانت SIDE = 'R'، فإن LWORK >= max(1,M).
!>          لأداء جيد، يجب أن يكون LWORK بشكل عام أكبر.
!>
!>          إذا كان LWORK = -1، فسيتم افتراض استعلام عن مساحة العمل؛ تقوم الروتين
!>          بحساب حجم مصفوفة WORK الأمثل فقط، وتعيد
!>          هذه القيمة كالإدخال الأول لمصفوفة WORK، ولا يتم إصدار رسالة خطأ
!>          متعلقة بـ LWORK بواسطة XERBLA.
!>

INFO

!>          INFO هو INTEGER
!>          = 0:  إنهاء ناجح
!>          < 0:  إذا كان INFO = -i، فإن الوسيط i كان لديه قيمة غير قانونية
!>
