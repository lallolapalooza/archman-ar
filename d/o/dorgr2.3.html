subroutine dorgr2 (m, n, k, a, lda, tau, work, info)
DORGR2 تولد جزءًا أو كل المصفوفة المتعامدة Q من تحليل RQ تم تحديده بواسطة sgerqf (خوارزمية غير محظورة).

توثيق الوظيفة/الإجراء الفرعي

subroutine dorgr2 (integer m, integer n, integer k, double precision, dimension( lda, * ) a, integer lda, double precision, dimension( * ) tau, double precision, dimension( * ) work, integer info)

DORGR2 تولد جزءًا أو كل المصفوفة المتعامدة Q من تحليل RQ تم تحديده بواسطة sgerqf (خوارزمية غير محظورة).

الغرض:

!>
!> DORGR2 تولد مصفوفة Q حقيقية من حجم m × n ذات صفوف متعامدة،
!> والتي تُعرّف بأنها آخر m صفوف من حاصل ضرب k انعكاسات أولية
!> من الرتبة n
!>
!>       Q  =  H(1) H(2) . . . H(k)
!>
!> كما تم إرجاعها بواسطة DGERQF.
!>

المعلمات

M 
!>          M هو عدد صحيح (INTEGER)
!>          عدد صفوف المصفوفة Q. M >= 0.
!>

N

!>          N هو عدد صحيح (INTEGER)
!>          عدد أعمدة المصفوفة Q. N >= M.
!>

K

!>          K هو عدد صحيح (INTEGER)
!>          عدد الانعكاسات الأولية التي يُعرّف حاصل ضربها المصفوفة Q. M >= K >= 0.
!>

A

!>          A هي مصفوفة من نوع double precision، أبعادها (LDA,N)
!>          عند الإدخال، يجب أن يحتوي الصف (m-k+i) على المتجه الذي
!>          يعرّف الانعكاس الأولي H(i)، لكل i = 1،2،...،k، كما
!>          تم إرجاعه بواسطة DGERQF في آخر k صفوف من مصفوفة الوسيط A.
!>          عند الخروج، المصفوفة Q بحجم m × n.
!>

LDA

!>          LDA هو عدد صحيح (INTEGER)
!>          البعد الأول للمصفوفة A. LDA >= max(1,M).
!>

TAU

!>          TAU هي مصفوفة من نوع double precision، أبعادها (K)
!>          يجب أن تحتوي TAU(i) على المعامل القياسي للانعكاس الأولي
!>          H(i)، كما تم إرجاعه بواسطة DGERQF.
!>

WORK

!>          WORK هي مصفوفة من نوع double precision، أبعادها (M)
!>

INFO

!>          INFO هو عدد صحيح (INTEGER)
!>          = 0: خروج ناجح
!>          < 0: إذا كان INFO = -i، فإن الوسيط i له قيمة غير قانونية
!>
