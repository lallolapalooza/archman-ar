subroutine dort01 (character rowcol, integer m, integer n, double precision, dimension( ldu, * ) u, integer ldu, double precision, dimension( * ) work, integer lwork, double precision resid)

DORT01

الغرض:

!>
!> يتحقق DORT01 من أن المصفوفة U متعامدة عن طريق حساب النسبة
!>
!>    RESID = norm( I - U*U' ) / ( n * EPS )، إذا كان ROWCOL = 'R'،
!> أو
!>    RESID = norm( I - U'*U ) / ( m * EPS )، إذا كان ROWCOL = 'C'.
!>
!> بدلاً من ذلك، إذا لم يكن هناك مساحة عمل كافية لتشكيل
!> I - U*U' أو I - U'*U، يتم حساب النسبة على أنها
!>
!>    RESID = abs( I - U*U' ) / ( n * EPS )، إذا كان ROWCOL = 'R'،
!> أو
!>    RESID = abs( I - U'*U ) / ( m * EPS )، إذا كان ROWCOL = 'C'.
!>
!> حيث EPS هي دقة الآلة. يتم استخدام ROWCOL فقط إذا كان m = n؛
!> إذا كان m > n، يُفترض أن ROWCOL هو 'C'، وإذا كان m < n، يُفترض أن ROWCOL هو
!> 'R'.
!>

المعلمات

ROWCOL 
!>          ROWCOL هو CHARACTER
!>          يحدد ما إذا كان يجب التحقق من صفوف أو أعمدة U للتعامد
!>          . يستخدم فقط إذا كان M = N.
!>          = 'R':  تحقق من صفوف U المتعامدة
!>          = 'C':  تحقق من أعمدة U المتعامدة
!>

M

!>          M هو INTEGER
!>          عدد صفوف المصفوفة U.
!>

N

!>          N هو INTEGER
!>          عدد أعمدة المصفوفة U.
!>

U

!>          U هي مصفوفة DOUBLE PRECISION، البعد (LDU,N)
!>          المصفوفة المتعامدة U. يتم التحقق من U للأعمدة المتعامدة
!>          إذا كان m > n أو إذا كان m = n و ROWCOL = 'C'. يتم التحقق من U
!>          للصفوف المتعامدة إذا كان m < n أو إذا كان m = n و ROWCOL = 'R'.
!>

LDU

!>          LDU هو INTEGER
!>          البعد الرئيسي للمصفوفة U. LDU >= max(1,M).
!>

WORK

!>          WORK هي مصفوفة DOUBLE PRECISION، البعد (LWORK)
!>

LWORK

!>          LWORK هو INTEGER
!>          طول المصفوفة WORK. للحصول على أفضل أداء، يجب أن يكون LWORK
!>          على الأقل N*(N+1) إذا كان ROWCOL = 'C' أو M*(M+1) إذا كان
!>          ROWCOL = 'R'، ولكن سيتم إجراء الاختبار حتى لو كان LWORK = 0.
!>

RESID

!>          RESID هو DOUBLE PRECISION
!>          RESID = norm( I - U * U' ) / ( n * EPS )، إذا كان ROWCOL = 'R'، أو
!>          RESID = norm( I - U' * U ) / ( m * EPS )، إذا كان ROWCOL = 'C'.
!>
