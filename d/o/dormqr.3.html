subroutine dormqr (character side, character trans, integer m, integer n, integer k, double precision, dimension( lda, * ) a, integer lda, double precision, dimension( * ) tau, double precision, dimension( ldc, * ) c, integer ldc, double precision, dimension( * ) work, integer lwork, integer info)

DORMQR

الغرض:

!>
!> DORMQR يستبدل مصفوفة C الحقيقية ذات الأبعاد M-by-N بـ
!>
!>                 SIDE = 'L'     SIDE = 'R'
!> TRANS = 'N':      Q * C          C * Q
!> TRANS = 'T':      Q**T * C       C * Q**T
!>
!> حيث Q هي مصفوفة متعامدة حقيقية معرفة كناتج لـ k
!> عاكسات أولية
!>
!>       Q = H(1) H(2) . . . H(k)
!>
!> كما تم إرجاعه بواسطة DGEQRF. Q هي من الرتبة M إذا كان SIDE = 'L' ومن الرتبة N
!> إذا كان SIDE = 'R'.
!>

المعلمات

SIDE 
!>          SIDE is CHARACTER*1
!>          = 'L': تطبيق Q أو Q**T من اليسار؛
!>          = 'R': تطبيق Q أو Q**T من اليمين.
!>

TRANS

!>          TRANS is CHARACTER*1
!>          = 'N':  بدون تبديل، تطبيق Q؛
!>          = 'T':  تبديل، تطبيق Q**T.
!>

M

!>          M is INTEGER
!>          عدد صفوف المصفوفة C. M >= 0.
!>

N

!>          N is INTEGER
!>          عدد أعمدة المصفوفة C. N >= 0.
!>

K

!>          K is INTEGER
!>          عدد العاكسات الأولية التي يحدد حاصل ضربها
!>          المصفوفة Q.
!>          إذا كان SIDE = 'L'، فإن M >= K >= 0؛
!>          إذا كان SIDE = 'R'، فإن N >= K >= 0.
!>

A

!>          A is DOUBLE PRECISION array, dimension (LDA,K)
!>          يجب أن يحتوي العمود i على المتجه الذي يحدد
!>          العاكس الأولي H(i)، من أجل i = 1،2،...،k، كما تم إرجاعه بواسطة
!>          DGEQRF في الأعمدة k الأولى من وسيطة المصفوفة A.
!>

LDA

!>          LDA is INTEGER
!>          البعد الرائد للمصفوفة A.
!>          إذا كان SIDE = 'L'، فإن LDA >= max(1,M)؛
!>          إذا كان SIDE = 'R'، فإن LDA >= max(1,N).
!>

TAU

!>          TAU is DOUBLE PRECISION array, dimension (K)
!>          يجب أن يحتوي TAU(i) على العامل القياسي للعاكس
!>          الأولي H(i)، كما تم إرجاعه بواسطة DGEQRF.
!>

C

!>          C is DOUBLE PRECISION array, dimension (LDC,N)
!>          عند الدخول، مصفوفة C ذات الأبعاد M-by-N.
!>          عند الخروج، يتم استبدال C بـ Q*C أو Q**T*C أو C*Q**T أو C*Q.
!>

LDC

!>          LDC is INTEGER
!>          البعد الرائد للمصفوفة C. LDC >= max(1,M).
!>

WORK

!>          WORK is DOUBLE PRECISION array, dimension (MAX(1,LWORK))
!>          عند الخروج، إذا كان INFO = 0، فإن WORK(1) يرجع LWORK الأمثل.
!>

LWORK

!>          LWORK is INTEGER
!>          بعد المصفوفة WORK.
!>          إذا كان SIDE = 'L'، فإن LWORK >= max(1,N)؛
!>          إذا كان SIDE = 'R'، فإن LWORK >= max(1,M).
!>          للحصول على أداء جيد، يجب أن يكون LWORK بشكل عام أكبر.
!>
!>          إذا كان LWORK = -1، فسيتم افتراض استعلام مساحة العمل؛ تقوم الوحدة الفرعية
!>          فقط بحساب الحجم الأمثل للمصفوفة WORK، وترجع
!>          هذه القيمة كالإدخال الأول للمصفوفة WORK، ولا يتم إصدار أي رسالة خطأ
!>          متعلقة بـ LWORK بواسطة XERBLA.
!>

INFO

!>          INFO is INTEGER
!>          = 0:  خروج ناجح
!>          < 0:  إذا كان INFO = -i، فإن الوسيط i كان لديه قيمة غير قانونية
!>
