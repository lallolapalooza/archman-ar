وثائق الوظيفة/الإجراء الفرعي

subroutine dorglq (integer m, integer n, integer k, double precision, dimension( lda, * ) a, integer lda, double precision, dimension( * ) tau, double precision, dimension( * ) work, integer lwork, integer info)

DORGLQ

الغرض:

!>
!> DORGLQ تولد مصفوفة Q حقيقية من حجم M × N ذات صفوف متعامدة،
!> والتي تُعرّف على أنها الصفوف M الأولى من حاصل ضرب K من المفاعلات الأولية
!> ذات الرتبة N
!>
!>       Q  =  H(k) . . . H(2) H(1)
!>
!> كما تم إرجاعها بواسطة DGELQF.
!>

المعلمات

M 
!>          M هو عدد صحيح (INTEGER)
!>          عدد صفوف المصفوفة Q. M >= 0.
!>

N

!>          N هو عدد صحيح (INTEGER)
!>          عدد أعمدة المصفوفة Q. N >= M.
!>

K

!>          K هو عدد صحيح (INTEGER)
!>          عدد المفاعلات الأولية التي تحدد حاصل ضربها المصفوفة
!>          Q. M >= K >= 0.
!>

A

!>          A هي مصفوفة من نوع DOUBLE PRECISION، ذات بُعد (LDA,N)
!>          عند الإدخال، يجب أن يحتوي الصف i على المتجه الذي يحدد
!>          المفاعل الأولي H(i)، للصف i = 1,2,...,k، كما تم إرجاعه
!>          بواسطة DGELQF في الصفوف K الأولى من مصفوفة المعامل A الخاصة به.
!>          عند الخروج، المصفوفة Q بحجم M × N.
!>

LDA

!>          LDA هو عدد صحيح (INTEGER)
!>          البُعد الأول للمصفوفة A. LDA >= max(1,M).
!>

TAU

!>          TAU هي مصفوفة من نوع DOUBLE PRECISION، ذات بُعد (K)
!>          يجب أن يحتوي TAU(i) على المعامل القياسي للمفاعل
!>          الأولي H(i)، كما تم إرجاعه بواسطة DGELQF.
!>

WORK

!>          WORK هي مصفوفة من نوع DOUBLE PRECISION، ذات بُعد (MAX(1,LWORK))
!>          عند الخروج، إذا كان INFO = 0، فإن WORK(1) يُرجع LWORK الأمثل.
!>

LWORK

!>          LWORK هو عدد صحيح (INTEGER)
!>          بُعد المصفوفة WORK. LWORK >= max(1,M).
!>          للحصول على أفضل أداء، LWORK >= M*NB، حيث NB هو
!>          حجم الكتلة الأمثل.
!>
!>          إذا كان LWORK = -1، فسيتم افتراض استعلام عن مساحة العمل؛ يقوم الإجراء الفرعي
!>          فقط بحساب الحجم الأمثل للمصفوفة WORK، ويرجع
!>          هذه القيمة كالإدخال الأول للمصفوفة WORK، ولن تصدر رسالة خطأ
!>          متعلقة بـ LWORK بواسطة XERBLA.
!>

INFO

!>          INFO هو عدد صحيح (INTEGER)
!>          = 0:  انتهاء ناجح
!>          < 0:  إذا كان INFO = -i، فإن المعامل i له قيمة غير قانونية
!>
