توثيق الوظيفة/الروتين الفرعي

subroutine dormtr (character side, character uplo, character trans, integer m, integer n, double precision, dimension( lda, * ) a, integer lda, double precision, dimension( * ) tau, double precision, dimension( ldc, * ) c, integer ldc, double precision, dimension( * ) work, integer lwork, integer info)

DORMTR

الغرض:

!>
!> DORMTR تستبدل مصفوفة C الحقيقية العامة M-by-N بـ
!>
!>                 SIDE = 'L'     SIDE = 'R'
!> TRANS = 'N':      Q * C          C * Q
!> TRANS = 'T':      Q**T * C       C * Q**T
!>
!> حيث Q هي مصفوفة متعامدة حقيقية من الرتبة nq، مع nq = m إذا
!> SIDE = 'L' و nq = n إذا SIDE = 'R'. تُعرف Q بأنها حاصل ضرب
!> nq-1 عاكسات أولية، كما تم إرجاعها بواسطة DSYTRD:
!>
!> إذا كان UPLO = 'U'، فإن Q = H(nq-1) . . . H(2) H(1)؛
!>
!> إذا كان UPLO = 'L'، فإن Q = H(1) H(2) . . . H(nq-1).
!>

المعلمات

SIDE 
!>          SIDE هو CHARACTER*1
!>          = 'L': تطبيق Q أو Q**T من اليسار؛
!>          = 'R': تطبيق Q أو Q**T من اليمين.
!>

UPLO

!>          UPLO هو CHARACTER*1
!>          = 'U': المثلث العلوي من A يحتوي على العاكسات الأولية
!>                 من DSYTRD؛
!>          = 'L': المثلث السفلي من A يحتوي على العاكسات الأولية
!>                 من DSYTRD.
!>

TRANS

!>          TRANS هو CHARACTER*1
!>          = 'N':  بدون تحويل، تطبيق Q؛
!>          = 'T':  تحويل، تطبيق Q**T.
!>

M

!>          M هو INTEGER
!>          عدد صفوف المصفوفة C. M >= 0.
!>

N

!>          N هو INTEGER
!>          عدد أعمدة المصفوفة C. N >= 0.
!>

A

!>          A هو مصفوفة DOUBLE PRECISION، الأبعاد
!>                               (LDA,M) إذا كان SIDE = 'L'
!>                               (LDA,N) إذا كان SIDE = 'R'
!>          المتجهات التي تعرف العاكسات الأولية، كما
!>          تم إرجاعها بواسطة DSYTRD.
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرائد للمصفوفة A.
!>          LDA >= max(1,M) إذا كان SIDE = 'L'؛ LDA >= max(1,N) إذا كان SIDE = 'R'.
!>

TAU

!>          TAU هو مصفوفة DOUBLE PRECISION، الأبعاد
!>                               (M-1) إذا كان SIDE = 'L'
!>                               (N-1) إذا كان SIDE = 'R'
!>          يجب أن تحتوي TAU(i) على العامل القياسي للعاكس الأولي
!>          H(i)، كما تم إرجاعها بواسطة DSYTRD.
!>

C

!>          C هو مصفوفة DOUBLE PRECISION، الأبعاد (LDC,N)
!>          عند الدخول، المصفوفة C ذات الأبعاد M-by-N.
!>          عند الخروج، يتم استبدال C بـ Q*C أو Q**T*C أو C*Q**T أو C*Q.
!>

LDC

!>          LDC هو INTEGER
!>          البعد الرائد للمصفوفة C. LDC >= max(1,M).
!>

WORK

!>          WORK هو مصفوفة DOUBLE PRECISION، الأبعاد (MAX(1,LWORK))
!>          عند الخروج، إذا كان INFO = 0، ترجع WORK(1) قيمة LWORK المثلى.
!>

LWORK

!>          LWORK هو INTEGER
!>          بعد المصفوفة WORK.
!>          إذا كان SIDE = 'L'، فإن LWORK >= max(1,N)؛
!>          إذا كان SIDE = 'R'، فإن LWORK >= max(1,M).
!>          للحصول على الأداء الأمثل، LWORK >= N*NB إذا كان SIDE = 'L'،
!>          و LWORK >= M*NB إذا كان SIDE = 'R'، حيث NB هو حجم الكتلة الأمثل.
!>
!>          إذا كان LWORK = -1، فيفترض استعلام عن مساحة العمل؛ يقوم الروتين
!>          فقط بحساب الحجم الأمثل للمصفوفة WORK، ويرجع
!>          هذه القيمة كإدخال أول في المصفوفة WORK، ولا تصدر رسالة خطأ
!>          متعلقة بـ LWORK بواسطة XERBLA.
!>

INFO

!>          INFO هو INTEGER
!>          = 0:  خروج ناجح
!>          < 0:  إذا كان INFO = -i، كان الوسيط i له قيمة غير قانونية
!>
