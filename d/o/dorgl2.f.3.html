subroutine dorgl2 (integer m, integer n, integer k, double precision, dimension( lda, * ) a, integer lda, double precision, dimension( * ) tau, double precision, dimension( * ) work, integer info)

DORGL2

الغرض:

!>
!> DORGL2 تنشئ مصفوفة حقيقية m x n ذات صفوف متعامدة،
!> والتي تُعرّف على أنها أول m صفوف من حاصل ضرب k من المعكوسات الأولية
!> من الرتبة n
!>
!>       Q  =  H(k) . . . H(2) H(1)
!>
!> كما تم إرجاعها بواسطة DGELQF.
!>

المعاملات

M 
!>          M هو عدد صحيح (INTEGER)
!>          عدد صفوف المصفوفة Q. M >= 0.
!>

N

!>          N هو عدد صحيح (INTEGER)
!>          عدد أعمدة المصفوفة Q. N >= M.
!>

K

!>          K هو عدد صحيح (INTEGER)
!>          عدد المعكوسات الأولية التي يُعرّف حاصل ضربها المصفوفة Q. M >= K >= 0.
!>

A

!>          A هي مصفوفة ذات دقة مزدوجة (DOUBLE PRECISION)، حجمها (LDA,N)
!>          عند الدخول، يجب أن يحتوي الصف i على المتجه الذي يُعرّف
!>          المعكوس الأولي H(i)، لكل i = 1,2,...,k، كما تم إرجاعه
!>          بواسطة DGELQF في أول k صفوف من مصفوفة الوسيط A.
!>          عند الخروج، المصفوفة m x n Q.
!>

LDA

!>          LDA هو عدد صحيح (INTEGER)
!>          البعد الأول للمصفوفة A. LDA >= max(1,M).
!>

TAU

!>          TAU هي مصفوفة ذات دقة مزدوجة (DOUBLE PRECISION)، حجمها (K)
!>          يجب أن تحتوي TAU(i) على عامل الضرب للمعكوس الأولي
!>          H(i)، كما تم إرجاعه بواسطة DGELQF.
!>

WORK

!>          WORK هي مصفوفة ذات دقة مزدوجة (DOUBLE PRECISION)، حجمها (M)
!>

INFO

!>          INFO هو عدد صحيح (INTEGER)
!>          = 0: خروج ناجح
!>          < 0: إذا كان INFO = -i، فإن الوسيط i يحتوي على قيمة غير قانونية
!>
