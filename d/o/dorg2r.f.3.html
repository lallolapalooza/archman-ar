subroutine dorg2r (m, n, k, a, lda, tau, work, info)
DORG2R يقوم بتوليد جزء من المصفوفة المتعامدة Q من تحليل QR الذي تم تحديده بواسطة sgeqrf (خوارزمية غير محظورة).

وثائق الوظيفة/الروتين الفرعي

subroutine dorg2r (integer m, integer n, integer k, double precision, dimension( lda, * ) a, integer lda, double precision, dimension( * ) tau, double precision, dimension( * ) work, integer info)

DORG2R يقوم بتوليد جزء من المصفوفة المتعامدة Q من تحليل QR الذي تم تحديده بواسطة sgeqrf (خوارزمية غير محظورة).

الغرض:

!>
!> DORG2R تقوم بتوليد مصفوفة Q حقيقية بحجم m × n بأعمدة متعامدة،
!> والتي تُعرّف على أنها أول n أعمدة من حاصل ضرب k عاكسات أولية
!> ذات رتبة m
!>
!>       Q  =  H(1) H(2) . . . H(k)
!>
!> كما تم إرجاعه بواسطة DGEQRF.
!>

المعاملات

M 
!>          M هو عدد صحيح (INTEGER)
!>          عدد صفوف المصفوفة Q. M >= 0.
!>

N

!>          N هو عدد صحيح (INTEGER)
!>          عدد أعمدة المصفوفة Q. M >= N >= 0.
!>

K

!>          K هو عدد صحيح (INTEGER)
!>          عدد العاكسات الأولية التي يحدد حاصل ضربها المصفوفة Q.
!>          N >= K >= 0.
!>

A

!>          A هي مصفوفة ذات دقة مزدوجة (DOUBLE PRECISION array)،
!>          ذات بعد (LDA,N).
!>          عند الدخول، يجب أن يحتوي العمود i على المتجه الذي
!>          يحدد العاكس الأولي H(i)، من أجل i = 1,2,...,k، كما
!>          تم إرجاعه بواسطة DGEQRF في أول k أعمدة من مصفوفة الوسائط A.
!>          عند الخروج، المصفوفة Q بحجم m × n.
!>

LDA

!>          LDA هو عدد صحيح (INTEGER)
!>          البعد الأول للمصفوفة A. LDA >= max(1,M).
!>

TAU

!>          TAU هي مصفوفة ذات دقة مزدوجة (DOUBLE PRECISION array)،
!>          ذات بعد (K).
!>          يجب أن يحتوي TAU(i) على عامل القياس للعاكس الأولي
!>          H(i)، كما تم إرجاعه بواسطة DGEQRF.
!>

WORK

!>          WORK هي مصفوفة ذات دقة مزدوجة (DOUBLE PRECISION array)،
!>          ذات بعد (N).
!>

INFO

!>          INFO هو عدد صحيح (INTEGER)
!>          = 0: الخروج ناجح
!>          < 0: إذا كان INFO = -i، فإن المعامل i يحتوي على قيمة غير قانونية
!>
