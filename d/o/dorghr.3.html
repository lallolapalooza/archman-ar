subroutine dorghr (integer n, integer ilo, integer ihi, double precision, dimension( lda, * ) a, integer lda, double precision, dimension( * ) tau, double precision, dimension( * ) work, integer lwork, integer info)

DORGHR

الغرض:

!>
!> DORGHR تنشئ مصفوفة متعامدة حقيقية Q معرفة كناتج لـ IHI-ILO انعكاسات أولية من الرتبة N، كما تم إرجاعها بواسطة DGEHRD:
!>
!> Q = H(ilo) H(ilo+1) . . . H(ihi-1).
!>

المعلمات

N 
!>          N هو عدد صحيح (INTEGER)
!>          رتبة المصفوفة Q. N >= 0.
!>

ILO

!>          ILO هو عدد صحيح (INTEGER)
!>

IHI

!>          IHI هو عدد صحيح (INTEGER)
!>
!>          يجب أن تحمل ILO و IHI نفس القيم كما في استدعاء DGEHRD السابق. Q تساوي مصفوفة الوحدة باستثناء الجزء الفرعي
!>          Q(ilo+1:ihi,ilo+1:ihi).
!>          1 <= ILO <= IHI <= N، إذا كان N > 0؛ ILO=1 و IHI=0، إذا كان N=0.
!>

A

!>          A هو مصفوفة Double Precision، ذات الأبعاد (LDA,N)
!>          عند الدخول، المتجهات التي تحدد الانعكاسات الأولية،
!>          كما تم إرجاعها بواسطة DGEHRD.
!>          عند الخروج، المصفوفة المتعامدة Q ذات الأبعاد N×N.
!>

LDA

!>          LDA هو عدد صحيح (INTEGER)
!>          البعد الرائد للمصفوفة A. LDA >= max(1,N).
!>

TAU

!>          TAU هو مصفوفة Double Precision، ذات الأبعاد (N-1)
!>          يجب أن تحتوي TAU(i) على العامل القياسي للانعكاس الأولي
!>          H(i)، كما تم إرجاعها بواسطة DGEHRD.
!>

WORK

!>          WORK هو مصفوفة Double Precision، ذات الأبعاد (MAX(1,LWORK))
!>          عند الخروج، إذا كان INFO = 0، فإن WORK(1) ترجع LWORK الأمثل.
!>

LWORK

!>          LWORK هو عدد صحيح (INTEGER)
!>          بعد المصفوفة WORK. LWORK >= IHI-ILO.
!>          لأداء أمثل LWORK >= (IHI-ILO)*NB، حيث NB
!>          هو حجم الكتلة الأمثل.
!>
!>          إذا كان LWORK = -1، فيفترض استعلام عن مساحة العمل؛ الروتين
!>          يحسب فقط الحجم الأمثل للمصفوفة WORK، ويعيد
!>          هذه القيمة كالإدخال الأول للمصفوفة WORK، ولا يتم إصدار أي رسالة خطأ
!>          تتعلق بـ LWORK بواسطة XERBLA.
!>

INFO

!>          INFO هو عدد صحيح (INTEGER)
!>          = 0:  انتهاء ناجح
!>          < 0:  إذا كان INFO = -i، كان الوسيط i غير قانوني
!>
