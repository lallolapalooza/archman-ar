وثائق الوظيفة/الروتين الفرعي

subroutine dormbr (character vect, character side, character trans, integer m, integer n, integer k, double precision, dimension( lda, * ) a, integer lda, double precision, dimension( * ) tau, double precision, dimension( ldc, * ) c, integer ldc, double precision, dimension( * ) work, integer lwork, integer info)

DORMBR

الغرض:

!>
!> إذا كان VECT = 'Q'، فإن DORMBR تستبدل المصفوفة العامة الحقيقية M-by-N C بـ
!>                 SIDE = 'L'     SIDE = 'R'
!> TRANS = 'N':      Q * C          C * Q
!> TRANS = 'T':      Q**T * C       C * Q**T
!>
!> إذا كان VECT = 'P'، فإن DORMBR تستبدل المصفوفة العامة الحقيقية M-by-N C بـ
!>                 SIDE = 'L'     SIDE = 'R'
!> TRANS = 'N':      P * C          C * P
!> TRANS = 'T':      P**T * C       C * P**T
!>
!> هنا Q و P**T هما المصفوفتان المتعامدتان اللتان تم تحديدهما بواسطة DGEBRD عند
!> تقليل المصفوفة A الحقيقية إلى شكل ثنائي القطر: A = Q * B * P**T. يتم تعريف Q و
!> P**T كحاصل ضرب من الانعكاسات الأولية H(i) و G(i) على التوالي.
!>
!> لنفترض أن nq = m إذا كان SIDE = 'L' و nq = n إذا كان SIDE = 'R'. وبالتالي فإن nq هو
!> ترتيب المصفوفة المتعامدة Q أو P**T التي يتم تطبيقها.
!>
!> إذا كان VECT = 'Q'، يُفترض أن A كانت مصفوفة NQ-by-K:
!> إذا كان nq >= k، فإن Q = H(1) H(2) . . . H(k);
!> إذا كان nq < k، فإن Q = H(1) H(2) . . . H(nq-1).
!>
!> إذا كان VECT = 'P'، يُفترض أن A كانت مصفوفة K-by-NQ:
!> إذا كان k < nq، فإن P = G(1) G(2) . . . G(k);
!> إذا كان k >= nq، فإن P = G(1) G(2) . . . G(nq-1).
!>

المعلمات:

VECT 
!>          VECT هو CHARACTER*1
!>          = 'Q': تطبيق Q أو Q**T;
!>          = 'P': تطبيق P أو P**T.
!>

SIDE

!>          SIDE هو CHARACTER*1
!>          = 'L': تطبيق Q أو Q**T أو P أو P**T من اليسار;
!>          = 'R': تطبيق Q أو Q**T أو P أو P**T من اليمين.
!>

TRANS

!>          TRANS هو CHARACTER*1
!>          = 'N':  لا يوجد تبديل، تطبيق Q  أو P;
!>          = 'T':  تبديل، تطبيق Q**T أو P**T.
!>

M

!>          M هو INTEGER
!>          عدد صفوف المصفوفة C. M >= 0.
!>

N

!>          N هو INTEGER
!>          عدد أعمدة المصفوفة C. N >= 0.
!>

K

!>          K هو INTEGER
!>          إذا كان VECT = 'Q'، فإن عدد الأعمدة في المصفوفة الأصلية
!>          التي تم تقليلها بواسطة DGEBRD.
!>          إذا كان VECT = 'P'، فإن عدد الصفوف في المصفوفة الأصلية
!>          التي تم تقليلها بواسطة DGEBRD.
!>          K >= 0.
!>

A

!>          A هي مصفوفة DOUBLE PRECISION، البعد
!>                                (LDA,min(nq,K)) إذا كان VECT = 'Q'
!>                                (LDA,nq)        إذا كان VECT = 'P'
!>          المتجهات التي تحدد الانعكاسات الأولية H(i) و
!>          G(i)، التي تحدد حاصل ضربها المصفوفتين Q و P، كما
!>          تم إرجاعها بواسطة DGEBRD.
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرائد للمصفوفة A.
!>          إذا كان VECT = 'Q'، فإن LDA >= max(1,nq);
!>          إذا كان VECT = 'P'، فإن LDA >= max(1,min(nq,K)).
!>

TAU

!>          TAU هي مصفوفة DOUBLE PRECISION، البعد (min(nq,K))
!>          يجب أن تحتوي TAU(i) على العامل القياسي للانعكاس الأولي
!>          H(i) أو G(i) الذي يحدد Q أو P، كما تم إرجاعه
!>          بواسطة DGEBRD في وسيط المصفوفة TAUQ أو TAUP.
!>

C

!>          C هي مصفوفة DOUBLE PRECISION، البعد (LDC,N)
!>          عند الإدخال، المصفوفة C بحجم M-by-N.
!>          عند الخروج، يتم استبدال C بـ Q*C أو Q**T*C أو C*Q**T أو C*Q
!>          أو P*C أو P**T*C أو C*P أو C*P**T.
!>

LDC

!>          LDC هو INTEGER
!>          البعد الرائد للمصفوفة C. LDC >= max(1,M).
!>

WORK

!>          WORK هي مصفوفة DOUBLE PRECISION، البعد (MAX(1,LWORK))
!>          عند الخروج، إذا كان INFO = 0، فإن WORK(1) يعيد قيمة LWORK المثلى.
!>

LWORK

!>          LWORK هو INTEGER
!>          بعد المصفوفة WORK.
!>          إذا كان SIDE = 'L'، فإن LWORK >= max(1,N);
!>          إذا كان SIDE = 'R'، فإن LWORK >= max(1,M).
!>          للحصول على الأداء الأمثل، LWORK >= N*NB إذا كان SIDE = 'L'، و
!>          LWORK >= M*NB إذا كان SIDE = 'R'، حيث NB هو حجم الكتلة الأمثل.
!>
!>          إذا كان LWORK = -1، فسيتم افتراض استعلام عن مساحة العمل؛ يقوم الروتين
!>          بحساب حجم مساحة العمل المثلى فقط، وإعادة هذه القيمة كأول إدخال لمساحة العمل،
!>          ولا يتم إصدار أي رسالة خطأ متعلقة بـ LWORK بواسطة XERBLA.
!>

INFO

!>          INFO هو INTEGER
!>          = 0:  اكتمال بنجاح
!>          < 0:  إذا كان INFO = -i، فإن الوسيط i كان له قيمة غير قانونية
!>
