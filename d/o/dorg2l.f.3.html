subroutine dorg2l (integer m, integer n, integer k, double precision, dimension( lda, * ) a, integer lda, double precision, dimension( * ) tau, double precision, dimension( * ) work, integer info)

DORG2L تولد جزءًا أو كل مصفوفة Q المتعامدة من تحليل QL الذي حدده sgeqlf (خوارزمية غير محظورة).

الغرض:

!>
!> DORG2L تولد مصفوفة Q حقيقية بحجم m × n بأعمدة متعامدة،
!> والتي يتم تعريفها كآخر n أعمدة لحاصل ضرب k من عاكسات أولية
!> من الرتبة m
!>
!>       Q  =  H(k) . . . H(2) H(1)
!>
!> كما تم إرجاعه بواسطة DGEQLF.
!>

المعلمات

M 
!>          M هو عدد صحيح (INTEGER)
!>          عدد الصفوف في المصفوفة Q. M >= 0.
!>

N

!>          N هو عدد صحيح (INTEGER)
!>          عدد الأعمدة في المصفوفة Q. M >= N >= 0.
!>

K

!>          K هو عدد صحيح (INTEGER)
!>          عدد العاكسات الأولية التي يحدد حاصل ضربها المصفوفة Q. N >= K >= 0.
!>

A

!>          A هي مصفوفة من نوع DOUBLE PRECISION، أبعادها (LDA,N)
!>          عند الدخول، يجب أن يحتوي العمود (n-k+i) على المتجه الذي
!>          يحدد العاكس الأولي H(i)، من أجل i = 1,2,...,k، كما
!>          تم إرجاعه بواسطة DGEQLF في آخر k أعمدة لمعامل المصفوفة A الخاص به.
!>          عند الخروج، المصفوفة Q بحجم m × n.
!>

LDA

!>          LDA هو عدد صحيح (INTEGER)
!>          البعد الأول للمصفوفة A. LDA >= max(1,M).
!>

TAU

!>          TAU هي مصفوفة من نوع DOUBLE PRECISION، أبعادها (K)
!>          يجب أن تحتوي TAU(i) على العامل القياسي للعاكس
!>          الأولي H(i)، كما تم إرجاعه بواسطة DGEQLF.
!>

WORK

!>          WORK هي مصفوفة من نوع DOUBLE PRECISION، أبعادها (N)
!>

INFO

!>          INFO هو عدد صحيح (INTEGER)
!>          = 0: خروج ناجح
!>          < 0: إذا كان INFO = -i، فإن المعامل i له قيمة غير قانونية
!>
