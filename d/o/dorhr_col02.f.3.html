subroutine dorhr_col02 (integer m, integer n, integer mb1, integer nb1, integer nb2, double precision, dimension(6) result)

DORHR_COL02

الغرض:

!>
!> DORHR_COL02 تختبر DORGTSQR_ROW و DORHR_COL داخل DGETSQRHRT
!> (التي تستدعي DLATSQR، DORGTSQR_ROW و DORHR_COL) باستخدام DGEMQRT.
!> لذلك، يجب اختبار DLATSQR (جزء من DGEQR)، DGEMQRT (جزء من DGEMQR)
!> قبل هذا الاختبار.
!>
!>

المعلمات

M 
!>          M هو عدد صحيح
!>          عدد الصفوف في مصفوفة الاختبار.
!>

N

!>          N هو عدد صحيح
!>          عدد الأعمدة في مصفوفة الاختبار.
!>

MB1

!>          MB1 هو عدد صحيح
!>          عدد الصفوف في الكتلة الصفية في مصفوفة الاختبار المدخلة.
!>

NB1

!>          NB1 هو عدد صحيح
!>          عدد الأعمدة في الكتلة العمودية في مصفوفة الاختبار المدخلة.
!>

NB2

!>          NB2 هو عدد صحيح
!>          عدد الأعمدة في الكتلة العمودية في مصفوفة الاختبار المخرجة.
!>

RESULT

!>          RESULT هو مصفوفة ذات دقة مزدوجة، بأبعاد (6)
!>          نتائج كل من الاختبارات الستة أدناه.
!>
!>            A هي مصفوفة اختبار مدخلة m-by-n ليتم تحليلها.
!>            بحيث A = Q_gr * ( R )
!>                               ( 0 ),
!>
!>            Q_qr هي مصفوفة متعامدة ضمنية m-by-m Q، وهي نتيجة
!>            التحليل بتمثيل WY المكدس،
!>            مخزنة بتنسيق الإخراج ZGEQRT.
!>
!>            R هي مصفوفة مثلثة علوية n-by-n،
!>
!>            0 هي مصفوفة صفرية (m-n)-by-n،
!>
!>            Q هي مصفوفة متعامدة صريحة m-by-m Q = Q_gr * I
!>
!>            C هي مصفوفة عشوائية m-by-n،
!>
!>            D هي مصفوفة عشوائية n-by-m.
!>
!>          الاختبارات الستة هي:
!>
!>          RESULT(1) = |R - (Q**H) * A| / ( eps * m * |A| )
!>            يعادل الاختبار لـ | A - Q * R | / (eps * m * |A|)،
!>
!>          RESULT(2) = |I - (Q**H) * Q| / ( eps * m )،
!>
!>          RESULT(3) = | Q_qr * C - Q * C | / (eps * m * |C|)،
!>
!>          RESULT(4) = | (Q_gr**H) * C - (Q**H) * C | / (eps * m * |C|)
!>
!>          RESULT(5) = | D * Q_qr - D * Q | / (eps * m * |D|)
!>
!>          RESULT(6) = | D * (Q_qr**H) - D * (Q**H) | / (eps * m * |D|)،
!>
!>          حيث:
!>            Q_qr * C، (Q_gr**H) * C، D * Q_qr، D * (Q_qr**H) يتم
!>            حسابها باستخدام DGEMQRT،
!>
!>            Q * C، (Q**H) * C، D * Q، D * (Q**H) يتم
!>            حسابها باستخدام DGEMM.
!>
