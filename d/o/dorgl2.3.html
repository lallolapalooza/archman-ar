توثيق الوظيفة/الروتين الفرعي

subroutine dorgl2 (integer m, integer n, integer k, double precision, dimension( lda, * ) a, integer lda, double precision, dimension( * ) tau, double precision, dimension( * ) work, integer info)

DORGL2

الغرض:

!>
!> DORGL2 يولد مصفوفة Q حقيقية بأبعاد m x n مع صفوف متعامدة،
!> والتي تُعرّف على أنها أول m صفوف من حاصل ضرب k من المرتفعات الابتدائية
!> ذات الرتبة n
!>
!>       Q  =  H(k) . . . H(2) H(1)
!>
!> كما يتم إرجاعها بواسطة DGELQF.
!>

المعلمات

M 
!>          M هو عدد صحيح (INTEGER)
!>          عدد صفوف المصفوفة Q. M >= 0.
!>

N

!>          N هو عدد صحيح (INTEGER)
!>          عدد أعمدة المصفوفة Q. N >= M.
!>

K

!>          K هو عدد صحيح (INTEGER)
!>          عدد المرتفعات الابتدائية التي يحدد حاصل ضربها المصفوفة Q.
!>          M >= K >= 0.
!>

A

!>          A هي مصفوفة ذات دقة مزدوجة (DOUBLE PRECISION)، أبعادها (LDA,N)
!>          عند الإدخال، يجب أن يحتوي الصف i على المتجه الذي يحدد
!>          المرتفع الابتدائي H(i)، للمتغيرات i = 1,2,...,k، كما تم
!>          إرجاعه بواسطة DGELQF في أول k صفوف من مصفوفة الوسيط A.
!>          عند الخروج، المصفوفة Q بأبعاد m x n.
!>

LDA

!>          LDA هو عدد صحيح (INTEGER)
!>          البعد الأول للمصفوفة A. LDA >= max(1,M).
!>

TAU

!>          TAU هي مصفوفة ذات دقة مزدوجة (DOUBLE PRECISION)، أبعادها (K)
!>          يجب أن يحتوي TAU(i) على العامل القياسي للمرتفع
!>          الابتدائي H(i)، كما تم إرجاعه بواسطة DGELQF.
!>

WORK

!>          WORK هي مصفوفة ذات دقة مزدوجة (DOUBLE PRECISION)، أبعادها (M)
!>

INFO

!>          INFO هو عدد صحيح (INTEGER)
!>          = 0: خروج ناجح
!>          < 0: إذا كان INFO = -i، فإن الوسيط i له قيمة غير قانونية
!>
