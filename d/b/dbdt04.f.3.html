وثائق الوظيفة/الروتين الفرعي

subroutine dbdt04 (character uplo, integer n, double precision, dimension( * ) d, double precision, dimension( * ) e, double precision, dimension( * ) s, integer ns, double precision, dimension( ldu, * ) u, integer ldu, double precision, dimension( ldvt, * ) vt, integer ldvt, double precision, dimension( * ) work, double precision resid)

DBDT04

الغرض:

!>
!> DBDT04 يعيد بناء مصفوفة ثنائية القطر B من تحليلها المنفرد (الجزئي) SVD:
!>    S = U' * B * V
!> حيث U و V مصفوفتان متعامدتان و S مصفوفة قطرية.
!>
!> نسبة الاختبار لاختبار تحليل القيمة المنفردة هي
!>    RESID = norm( S - U' * B * V ) / ( n * norm(B) * EPS )
!> حيث VT = V' و EPS هي دقة الآلة.
!>

المعلمات

UPLO 
!>          UPLO هو CHARACTER*1
!>          يحدد ما إذا كانت المصفوفة B ثنائية القطر علوية أم سفلية.
!>          = 'U':  ثنائية القطر علوية
!>          = 'L':  ثنائية القطر سفلية
!>

N

!>          N هو INTEGER
!>          رتبة المصفوفة B.
!>

D

!>          D هو مصفوفة DOUBLE PRECISION، ذات البعد (N)
!>          العناصر القطرية n للمصفوفة ثنائية القطر B.
!>

E

!>          E هو مصفوفة DOUBLE PRECISION، ذات البعد (N-1)
!>          العناصر القطرية الفائقة n-1 للمصفوفة B
!>          إذا كان UPLO = 'U'، أو العناصر القطرية الفرعية n-1 للمصفوفة B إذا
!>          كان UPLO = 'L'.
!>

S

!>          S هو مصفوفة DOUBLE PRECISION، ذات البعد (NS)
!>          القيم المنفردة من تحليل القيمة المنفردة (الجزئي) لـ B، مرتبة بترتيب
!>          تنازلي.
!>

NS

!>          NS هو INTEGER
!>          عدد القيم/المتجهات المنفردة من تحليل القيمة المنفردة (الجزئي) لـ B.
!>

U

!>          U هو مصفوفة DOUBLE PRECISION، ذات البعد (LDU,NS)
!>          المصفوفة المتعامدة n × ns U في S = U' * B * V.
!>

LDU

!>          LDU هو INTEGER
!>          البعد الرائد للمصفوفة U.  LDU >= max(1,N)
!>

VT

!>          VT هو مصفوفة DOUBLE PRECISION، ذات البعد (LDVT,N)
!>          المصفوفة المتعامدة n × ns V في S = U' * B * V.
!>

LDVT

!>          LDVT هو INTEGER
!>          البعد الرائد للمصفوفة VT.
!>

WORK

!>          WORK هو مصفوفة DOUBLE PRECISION، ذات البعد (2*N)
!>

RESID

!>          RESID هو DOUBLE PRECISION
!>          نسبة الاختبار:  norm(S - U' * B * V) / ( n * norm(B) * EPS )
!>
