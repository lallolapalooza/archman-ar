توثيق الوظيفة/الإجراء الفرعي

subroutine dbdt01 (integer m, integer n, integer kd, double precision, dimension( lda, * ) a, integer lda, double precision, dimension( ldq, * ) q, integer ldq, double precision, dimension( * ) d, double precision, dimension( * ) e, double precision, dimension( ldpt, * ) pt, integer ldpt, double precision, dimension( * ) work, double precision resid)

DBDT01

الغرض:

!>
!> DBDT01 تعيد بناء مصفوفة عامة A من شكلها ثنائي القطر
!>    A = Q * B * P**T
!> حيث Q (m x min(m,n)) و P**T (min(m,n) x n) هي مصفوفات متعامدة
!> و B ثنائية القطر.
!>
!> نسبة الاختبار لاختبار الاختزال هي
!>    RESID = norm(A - Q * B * P**T) / ( n * norm(A) * EPS )
!> حيث EPS هي دقة الآلة.
!>

المعلمات

M 
!>          M هو عدد صحيح (INTEGER)
!>          عدد الصفوف للمصفوفتين A و Q.
!>

N

!>          N هو عدد صحيح (INTEGER)
!>          عدد الأعمدة للمصفوفتين A و P**T.
!>

KD

!>          KD هو عدد صحيح (INTEGER)
!>          إذا كان KD = 0، فإن B قطرية ولا يتم الاستشهاد بالمصفوفة E.
!>          إذا كان KD = 1، تم إجراء الاختزال بواسطة xGEBRD؛ B قطرية علوية
!>          إذا كان M >= N، وقطرية سفلية إذا كان M < N.
!>          إذا كان KD = -1، تم إجراء الاختزال بواسطة xGBBRD؛ B قطرية علوية دائمًا.
!>

A

!>          A هي مصفوفة double precision، ذات أبعاد (LDA,N)
!>          المصفوفة A ذات الأبعاد m x n.
!>

LDA

!>          LDA هو عدد صحيح (INTEGER)
!>          البعد الرائد للمصفوفة A. LDA >= max(1,M).
!>

Q

!>          Q هي مصفوفة double precision، ذات أبعاد (LDQ,N)
!>          المصفوفة المتعامدة Q ذات الأبعاد m x min(m,n) في الاختزال
!>          A = Q * B * P**T.
!>

LDQ

!>          LDQ هو عدد صحيح (INTEGER)
!>          البعد الرائد للمصفوفة Q. LDQ >= max(1,M).
!>

D

!>          D هي مصفوفة double precision، ذات أبعاد (min(M,N))
!>          العناصر القطرية للمصفوفة ثنائية القطر B.
!>

E

!>          E هي مصفوفة double precision، ذات أبعاد (min(M,N)-1)
!>          العناصر القطرية العلوية للمصفوفة ثنائية القطر B إذا كان
!>          m >= n، أو العناصر القطرية السفلية لـ B إذا كان m < n.
!>

PT

!>          PT هي مصفوفة double precision، ذات أبعاد (LDPT,N)
!>          المصفوفة المتعامدة P**T ذات الأبعاد min(m,n) x n في الاختزال
!>          A = Q * B * P**T.
!>

LDPT

!>          LDPT هو عدد صحيح (INTEGER)
!>          البعد الرائد للمصفوفة PT.
!>          LDPT >= max(1,min(M,N)).
!>

WORK

!>          WORK هي مصفوفة double precision، ذات أبعاد (M+N)
!>

RESID

!>          RESID هو double precision
!>          نسبة الاختبار:
!>          norm(A - Q * B * P**T) / ( n * norm(A) * EPS )
!>
