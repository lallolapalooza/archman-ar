توثيق الوظيفة/الروتين الفرعي

subroutine dbdt02 (integer m, integer n, double precision, dimension( ldb, * ) b, integer ldb, double precision, dimension( ldc, * ) c, integer ldc, double precision, dimension( ldu, * ) u, integer ldu, double precision, dimension( * ) work, double precision resid)

DBDT02

الغرض:

!>
!> DBDT02 تختبر تغيير الأساس C = U**H * B عن طريق حساب الباقي
!>
!>    RESID = norm(B - U * C) / ( max(m,n) * norm(B) * EPS ),
!>
!> حيث B و C هي مصفوفات M × N ، U هي مصفوفة متعامدة M × M ،
!> و EPS هي دقة الآلة.
!>

المعاملات

M 
!>          M هو INTEGER
!>          عدد صفوف المصفوفات B و C ورتبة المصفوفة Q.
!>

N

!>          N هو INTEGER
!>          عدد أعمدة المصفوفات B و C.
!>

B

!>          B هو مصفوفة DOUBLE PRECISION، البعد (LDB,N)
!>          المصفوفة B بحجم m × n.
!>

LDB

!>          LDB هو INTEGER
!>          البعد الرئيسي للمصفوفة B.  LDB >= max(1,M).
!>

C

!>          C هو مصفوفة DOUBLE PRECISION، البعد (LDC,N)
!>          المصفوفة C بحجم m × n، يُفترض أنها تحتوي على U**H * B.
!>

LDC

!>          LDC هو INTEGER
!>          البعد الرئيسي للمصفوفة C.  LDC >= max(1,M).
!>

U

!>          U هو مصفوفة DOUBLE PRECISION، البعد (LDU,M)
!>          المصفوفة المتعامدة U بحجم m × m.
!>

LDU

!>          LDU هو INTEGER
!>          البعد الرئيسي للمصفوفة U.  LDU >= max(1,M).
!>

WORK

!>          WORK هي مصفوفة DOUBLE PRECISION، البعد (M)
!>

RESID

!>          RESID هو DOUBLE PRECISION
!>          RESID = norm(B - U * C) / ( max(m,n) * norm(B) * EPS ),
!>
