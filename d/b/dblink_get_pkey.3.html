dblink_get_pkey(3)

الحزمة: postgresql

التراخيص: PostgreSQL

DBLINK_GET_PKEY(3) توثيق PostgreSQL 18.1 DBLINK_GET_PKEY(3)

الاسم

dblink_get_pkey - يعيد مواضع وأسماء حقول المفتاح الأساسي لعلاقة

الملخص

dblink_get_pkey(text relname) returns setof dblink_pkey_results

الوصف

dblink_get_pkey يوفر معلومات حول المفتاح الأساسي لعلاقة في قاعدة البيانات المحلية. هذا مفيد أحيانًا في إنشاء استعلامات ليتم إرسالها إلى قواعد بيانات بعيدة.

الوسائط

relname

اسم علاقة محلية، على سبيل المثال foo أو myschema.mytab. قم بتضمين علامات الاقتباس المزدوجة إذا كان الاسم يحتوي على أحرف مختلطة أو يحتوي على أحرف خاصة، على سبيل المثال "FooBar"; بدون علامات اقتباس، سيتم تحويل السلسلة إلى أحرف صغيرة.

قيمة الإرجاع

يعيد صفًا واحدًا لكل حقل مفتاح أساسي، أو لا يعيد أي صفوف إذا لم يكن للعلاقة مفتاح أساسي. يتم تعريف نوع صف النتيجة على أنه

CREATE TYPE dblink_pkey_results AS (position int, colname text);

عمود الموضع يبدأ ببساطة من 1 إلى N؛ إنه رقم الحقل ضمن المفتاح الأساسي، وليس الرقم ضمن أعمدة الجدول.

أمثلة

CREATE TABLE foobar (
    f1 int,
    f2 int,
    f3 int,
    PRIMARY KEY (f1, f2, f3)
);
CREATE TABLE
SELECT * FROM dblink_get_pkey('foobar');
 position | colname
----------+---------
        1 | f1
        2 | f2
        3 | f3
(3 rows)
