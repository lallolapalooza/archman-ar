المعلمات

NOUT 
!>          NOUT هو عدد صحيح
!>                رقم الوحدة للخرج.
!>

NN

!>          NN هو عدد صحيح
!>                عدد القيم N الموجودة في المتجه NVAL.
!>

NVAL

!>          NVAL هو مصفوفة أعداد صحيحة، أبعادها (NN)
!>                قيم البعد N للمصفوفة.
!>

THRESH

!>          THRESH هو ضعف الدقة (double precision)
!>                القيمة الحدية لنسب الاختبار. يتم تضمين نتيجة
!>                في ملف الخرج إذا كانت RESULT >= THRESH. لطباعة
!>                كل نسبة اختبار، استخدم THRESH = 0.
!>

A

!>          A هي مصفوفة ضعف الدقة (double precision)، أبعادها (LDA,NMAX)
!>

LDA

!>          LDA هو عدد صحيح
!>                البعد الرئيسي للمصفوفة A. LDA >= max(1,NMAX).
!>

ARF

!>          ARF هي مصفوفة ضعف الدقة (double precision)، أبعادها ((NMAX*(NMAX+1))/2).
!>

WORK

!>          WORK هي مصفوفة ضعف الدقة (double precision)، أبعادها ( NMAX )
!>
