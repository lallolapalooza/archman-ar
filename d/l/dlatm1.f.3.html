subroutine dlatm1 (integer mode, double precision cond, integer irsign, integer idist, integer, dimension( 4 ) iseed, double precision, dimension( * ) d, integer n, integer info)

DLATM1

الغرض:

!>
!>    DLATM1 يحسب مدخلات D(1..N) كما هو محدد بواسطة
!>    MODE، COND و IRSIGN. IDIST و ISEED يحددان توليد
!>    الأرقام العشوائية. يتم استدعاء DLATM1 بواسطة DLATMR لتوليد
!>    مصفوفات اختبار عشوائية لبرامج LAPACK.
!>

المعلمات

MODE 
!>          MODE هو عدد صحيح
!>           عند الإدخال يصف كيفية حساب D:
!>           MODE = 0 يعني عدم تغيير D.
!>           MODE = 1 يضبط D(1)=1 و D(2:N)=1.0/COND
!>           MODE = 2 يضبط D(1:N-1)=1 و D(N)=1.0/COND
!>           MODE = 3 يضبط D(I)=COND**(-(I-1)/(N-1))
!>           MODE = 4 يضبط D(i)=1 - (i-1)/(N-1)*(1 - 1/COND)
!>           MODE = 5 يضبط D إلى أرقام عشوائية في النطاق
!>                    ( 1/COND ، 1 ) بحيث تكون لوغاريتماتها
!>                    موزعة بشكل منتظم.
!>           MODE = 6 يضبط D إلى أرقام عشوائية من نفس التوزيع
!>                    مثل بقية المصفوفة.
!>           MODE < 0 له نفس المعنى مثل ABS(MODE)، باستثناء
!>              أن ترتيب عناصر D معكوس.
!>           لذلك إذا كان MODE موجبًا، فإن D لها مدخلات تتراوح من
!>              1 إلى 1/COND، إذا كان سالبًا، من 1/COND إلى 1،
!>           لم يتم تعديله.
!>

COND

!>          COND هو دقة مزدوجة
!>           عند الإدخال، يستخدم كما هو موضح تحت MODE أعلاه.
!>           إذا تم استخدامه، يجب أن يكون >= 1. لم يتم تعديله.
!>

IRSIGN

!>          IRSIGN هو عدد صحيح
!>           عند الإدخال، إذا لم يكن MODE -6 أو 0 أو 6، يحدد إشارة
!>           مدخلات D
!>           0 => ترك مدخلات D دون تغيير
!>           1 => ضرب كل مدخل من D بـ 1 أو -1 باحتمالية 0.5
!>

IDIST

!>          IDIST هو عدد صحيح
!>           عند الإدخال، يحدد IDIST نوع التوزيع المراد استخدامه
!>           لتوليد مصفوفة عشوائية.
!>           1 => منتظم ( 0, 1 )
!>           2 => منتظم ( -1, 1 )
!>           3 => طبيعي ( 0, 1 )
!>           لم يتم تعديله.
!>

ISEED

!>          ISEED هو مصفوفة أعداد صحيحة، أبعادها ( 4 )
!>           عند الإدخال، يحدد ISEED البذرة لمولد الأرقام
!>           العشوائية. يستخدم مولد الأرقام
!>           العشوائية سلسلة خطية متوافقة محدودة بـ
!>           أعداد صحيحة صغيرة، ولذلك يجب أن تنتج أرقامًا عشوائية
!>           مستقلة عن الجهاز. يتم تغيير قيم ISEED عند
!>           الخروج، ويمكن استخدامها في الاستدعاء التالي لـ DLATM1
!>           لمواصلة نفس تسلسل الأرقام العشوائية.
!>           تم التغيير عند الخروج.
!>

D

!>          D هي مصفوفة دقة مزدوجة، أبعادها ( N )
!>           المصفوفة المراد حسابها وفقًا لـ MODE، COND و IRSIGN.
!>           قد يتم تغييرها عند الخروج إذا كان MODE غير صفري.
!>

N

!>          N هو عدد صحيح
!>           عدد مدخلات D. لم يتم تعديله.
!>

INFO

!>          INFO هو عدد صحيح
!>            0  => إنهاء طبيعي
!>           -1  => إذا لم يكن MODE ضمن النطاق -6 إلى 6
!>           -2  => إذا لم يكن MODE -6 أو 0 أو 6، و
!>                  IRSIGN ليس 0 أو 1
!>           -3  => إذا لم يكن MODE -6 أو 0 أو 6 و COND أقل من 1
!>           -4  => إذا كان MODE يساوي 6 أو -6 و IDIST ليس ضمن النطاق 1 إلى 3
!>           -7  => إذا كان N سالبًا
!>
