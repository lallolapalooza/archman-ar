توثيق الوظيفة/الروتين الفرعي

subroutine dlaptm (integer n, integer nrhs, double precision alpha, double precision, dimension( * ) d, double precision, dimension( * ) e, double precision, dimension( ldx, * ) x, integer ldx, double precision beta, double precision, dimension( ldb, * ) b, integer ldb)

DLAPTM

الغرض:

!>
!> تقوم DLAPTM بضرب مصفوفة N × NRHS من X بمصفوفة ثلاثية قطرية متناظرة A
!> وتخزن النتيجة في مصفوفة B.  تأخذ العملية الشكل
!>
!>    B := alpha * A * X + beta * B
!>
!> حيث يمكن أن يكون alpha إما 1. أو -1. ويمكن أن يكون beta 0. أو 1. أو -1.
!>

المعلمات

N 
!>          N هو عدد صحيح (INTEGER)
!>          رتبة المصفوفة A.  N >= 0.
!>

NRHS

!>          NRHS هو عدد صحيح (INTEGER)
!>          عدد الأطراف اليمنى، أي عدد أعمدة المصفوفتين X و B.
!>

ALPHA

!>          ALPHA هو عدد حقيقي مزدوج الدقة (DOUBLE PRECISION)
!>          المقدار الثابت alpha.  يجب أن يكون ALPHA 1. أو -1.؛ وإلا،
!>          يفترض أنه 0.
!>

D

!>          D هي مصفوفة ذات عدد حقيقي مزدوج الدقة (DOUBLE PRECISION)، بُعدها (N)
!>          العناصر القطرية n للمصفوفة الثلاثية القطرية A.
!>

E

!>          E هي مصفوفة ذات عدد حقيقي مزدوج الدقة (DOUBLE PRECISION)، بُعدها (N-1)
!>          العناصر تحت القطرية أو فوق القطرية (n-1) للمصفوفة A.
!>

X

!>          X هي مصفوفة ذات عدد حقيقي مزدوج الدقة (DOUBLE PRECISION)، بُعدها (LDX,NRHS)
!>          المصفوفة X ذات الأبعاد N × NRHS.
!>

LDX

!>          LDX هو عدد صحيح (INTEGER)
!>          البعد الرائد للمصفوفة X.  LDX >= max(N,1).
!>

BETA

!>          BETA هو عدد حقيقي مزدوج الدقة (DOUBLE PRECISION)
!>          المقدار الثابت beta.  يجب أن يكون BETA 0. أو 1. أو -1.؛ وإلا،
!>          يفترض أنه 1.
!>

B

!>          B هي مصفوفة ذات عدد حقيقي مزدوج الدقة (DOUBLE PRECISION)، بُعدها (LDB,NRHS)
!>          عند الدخول، المصفوفة B ذات الأبعاد N × NRHS.
!>          عند الخروج، يتم استبدال B بالتعبير المصفوفي
!>          B := alpha * A * X + beta * B.
!>

LDB

!>          LDB هو عدد صحيح (INTEGER)
!>          البعد الرائد للمصفوفة B.  LDB >= max(N,1).
!>
