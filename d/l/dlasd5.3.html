subroutine dlasd5 (i, d, z, delta, rho, dsigma, work)
DLASD5 يحسب الجذر التربيعي للقيمة الذاتية i-th لتعديل متناظر إيجابي من الرتبة الأولى لمصفوفة قطرية 2 × 2. يستخدم بواسطة sbdsdc.

توثيق الوظيفة/الإجراء الفرعي

subroutine dlasd5 (integer i, double precision, dimension( 2 ) d, double precision, dimension( 2 ) z, double precision, dimension( 2 ) delta, double precision rho, double precision dsigma, double precision, dimension( 2 ) work)

DLASD5 يحسب الجذر التربيعي للقيمة الذاتية i-th لتعديل متناظر إيجابي من الرتبة الأولى لمصفوفة قطرية 2 × 2. يستخدم بواسطة sbdsdc.

الغرض:

!>
!> تقوم هذه الإجراءات الفرعية بحساب الجذر التربيعي للقيمة الذاتية I-th
!> لتعديل متناظر إيجابي من الرتبة الأولى لمصفوفة قطرية 2 × 2
!>
!>            diag( D ) * diag( D ) +  RHO * Z * transpose(Z) .
!>
!> يُفترض أن تكون المدخلات القطرية في المصفوفة D تلبي:
!>
!>            0 <= D(i) < D(j)  for  i < j .
!>
!> نفترض أيضًا أن RHO > 0 وأن المعيار الإقليدي للمتجه Z هو واحد.
!>

المعلمات

I 
!>          I is INTEGER
!>         فهرس القيمة الذاتية المطلوب حسابها. I = 1 أو I = 2.
!>

D

!>          D is DOUBLE PRECISION array, dimension ( 2 )
!>         القيم الذاتية الأصلية. نفترض 0 <= D(1) < D(2).
!>

Z

!>          Z is DOUBLE PRECISION array, dimension ( 2 )
!>         مكونات المتجه التحديثي.
!>

DELTA

!>          DELTA is DOUBLE PRECISION array, dimension ( 2 )
!>         يحتوي على (D(j) - sigma_I) في المكون j.
!>         يحتوي المتجه DELTA على المعلومات الضرورية
!>         لبناء المتجهات الذاتية.
!>

RHO

!>          RHO is DOUBLE PRECISION
!>         العدد الثابت في صيغة التحديث المتناظر.
!>

DSIGMA

!>          DSIGMA is DOUBLE PRECISION
!>         القيمة المحسوبة sigma_I، القيمة الذاتية المحدثة I-th.
!>

WORK

!>          WORK is DOUBLE PRECISION array, dimension ( 2 )
!>         يحتوي WORK على (D(j) + sigma_I) في المكون j.
!>
