subroutine dlaqsy (character uplo, integer n, double precision, dimension( lda, * ) a, integer lda, double precision, dimension( * ) s, double precision scond, double precision amax, character equed)

DLAQSY تقوم بقياس مصفوفة متماثلة/هرميتية، باستخدام عوامل القياس التي حسبتها spoequ.

الغرض:

!>
!> DLAQSY تقوم بموازنة مصفوفة متماثلة A باستخدام عوامل القياس
!> في المتجه S.
!>

المعلمات

UPLO 
!>          UPLO هو CHARACTER*1
!>          يحدد ما إذا كان الجزء المثلثي العلوي أو السفلي من المصفوفة
!>          المتماثلة A مخزنًا.
!>          = 'U':  مثلثي علوي
!>          = 'L':  مثلثي سفلي
!>

N

!>          N هو INTEGER
!>          ترتيب المصفوفة A.  N >= 0.
!>

A

!>          A هي مصفوفة DOUBLE PRECISION، ذات بعد (LDA,N)
!>          عند الإدخال، المصفوفة المتماثلة A.  إذا كان UPLO = 'U'، فإن الجزء
!>          المثلثي العلوي n × n الرئيسي من A يحتوي على الجزء المثلثي
!>          العلوي من المصفوفة A، ولا تتم الإشارة إلى الجزء المثلثي السفلي
!>          الصارم من A.  إذا كان UPLO = 'L'، فإن الجزء المثلثي السفلي n × n
!>          الرئيسي من A يحتوي على الجزء المثلثي السفلي من المصفوفة A، ولا
!>          تتم الإشارة إلى الجزء المثلثي العلوي الصارم من A.
!>
!>          عند الخروج، إذا كان EQUED = 'Y'، فإن المصفوفة المتوازنة:
!>          diag(S) * A * diag(S).
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرئيسي للمصفوفة A.  LDA >= max(N,1).
!>

S

!>          S هي مصفوفة DOUBLE PRECISION، ذات بعد (N)
!>          عوامل القياس لـ A.
!>

SCOND

!>          SCOND هو DOUBLE PRECISION
!>          نسبة أصغر S(i) إلى أكبر S(i).
!>

AMAX

!>          AMAX هو DOUBLE PRECISION
!>          القيمة المطلقة لأكبر عنصر في المصفوفة.
!>

EQUED

!>          EQUED هو CHARACTER*1
!>          يحدد ما إذا تمت الموازنة أم لا.
!>          = 'N':  لم تتم الموازنة.
!>          = 'Y':  تمت الموازنة، أي تم استبدال A بـ
!>                  diag(S) * A * diag(S).
!>

المعلمات الداخلية:

!>  THRESH هو قيمة عتبة تستخدم لتحديد ما إذا كان يجب إجراء القياس
!>  بناءً على نسبة عوامل القياس.  إذا كان SCOND < THRESH،
!>  يتم إجراء القياس.
!>
!>  LARGE و SMALL هما قيمتا عتبة تستخدمان لتحديد ما إذا كان يجب إجراء القياس
!>  بناءً على الحجم المطلق لأكبر عنصر في المصفوفة.
!>  إذا كان AMAX > LARGE أو AMAX < SMALL، يتم إجراء القياس.
!>
