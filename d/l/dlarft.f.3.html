recursive subroutine dlarft (direct, storev, n, k, v, ldv, tau, t, ldt)
DLARFT تشكل العامل المثلثي T للانعكاس الكتلي H = I - vtvH

توثيق الوظيفة/الإجراء الفرعي

recursive subroutine dlarft (character direct, character storev, integer n, integer k, double precision, dimension( ldv, * ) v, integer ldv, double precision, dimension( * ) tau, double precision, dimension( ldt, * ) t, integer ldt)

DLARFT تشكل العامل المثلثي T للانعكاس الكتلي H = I - vtvH

الغرض:

!>
!> DLARFT تشكل العامل المثلثي T للانعكاس الكتلي الحقيقي H من الرتبة n،
!> والذي يُعرّف كناتج لـ k انعكاسات أولية.
!>
!> إذا كان DIRECT = 'F'، فإن H = H(1) H(2) . . . H(k) و T يكون مثلثيًا علويًا؛
!>
!> إذا كان DIRECT = 'B'، فإن H = H(k) . . . H(2) H(1) و T يكون مثلثيًا سفليًا.
!>
!> إذا كان STOREV = 'C'، فإن المتجه الذي يُعرّف الانعكاس الأولي
!> H(i) يتم تخزينه في العمود i من المصفوفة V، و
!>
!>    H  =  I - V * T * V**T
!>
!> إذا كان STOREV = 'R'، فإن المتجه الذي يُعرّف الانعكاس الأولي
!> H(i) يتم تخزينه في الصف i من المصفوفة V، و
!>
!>    H  =  I - V**T * T * V
!>

المعلمات

DIRECT 
!>          DIRECT هو CHARACTER*1
!>          يحدد ترتيب ضرب الانعكاسات الأولية لتشكيل الانعكاس الكتلي:
!>          = 'F': H = H(1) H(2) . . . H(k) (أمامي)
!>          = 'B': H = H(k) . . . H(2) H(1) (خلفي)
!>

STOREV

!>          STOREV هو CHARACTER*1
!>          يحدد كيفية تخزين المتجهات التي تُعرّف الانعكاسات الأولية
!>          (انظر أيضًا المزيد من التفاصيل):
!>          = 'C': عموديًا
!>          = 'R': أفقيًا
!>

N

!>          N هو INTEGER
!>          رتبة الانعكاس الكتلي H. N >= 0.
!>

K

!>          K هو INTEGER
!>          رتبة العامل المثلثي T (= عدد الانعكاسات الأولية). K >= 1.
!>

V

!>          V هي مصفوفة DOUBLE PRECISION، حجمها
!>                               (LDV,K) إذا كان STOREV = 'C'
!>                               (LDV,N) إذا كان STOREV = 'R'
!>          المصفوفة V. انظر المزيد من التفاصيل.
!>

LDV

!>          LDV هو INTEGER
!>          البعد الرئيسي للمصفوفة V.
!>          إذا كان STOREV = 'C'، فإن LDV >= max(1,N)؛ إذا كان STOREV = 'R'، فإن LDV >= K.
!>

TAU

!>          TAU هي مصفوفة DOUBLE PRECISION، حجمها (K)
!>          يجب أن تحتوي TAU(i) على العامل القياسي للانعكاس الأولي
!>          H(i).
!>

T

!>          T هي مصفوفة DOUBLE PRECISION، حجمها (LDT,K)
!>          العامل المثلثي T من الرتبة k × k للانعكاس الكتلي.
!>          إذا كان DIRECT = 'F'، فإن T يكون مثلثيًا علويًا؛ إذا كان DIRECT = 'B'، فإن T يكون
!>          مثلثيًا سفليًا. الجزء المتبقي من المصفوفة لا يُستخدم.
!>

LDT

!>          LDT هو INTEGER
!>          البعد الرئيسي للمصفوفة T. LDT >= K.
!>

جامعة تينيسي

جامعة كاليفورنيا بيركلي

المزيد من التفاصيل:

!>
!>  شكل المصفوفة V وتخزين المتجهات التي تُعرّف H(i) يتضح بشكل أفضل من خلال المثال التالي
!>  مع n = 5 و k = 3. العناصر التي تساوي 1 لا يتم تخزينها.
!>
!>  DIRECT = 'F' و STOREV = 'C':         DIRECT = 'F' و STOREV = 'R':
!>
!>               V = (  1       )                 V = (  1 v1 v1 v1 v1 )
!>                   ( v1  1    )                     (     1 v2 v2 v2 )
!>                   ( v1 v2  1 )                     (        1 v3 v3 )
!>                   ( v1 v2 v3 )
!>                   ( v1 v2 v3 )
!>
!>  DIRECT = 'B' و STOREV = 'C':         DIRECT = 'B' و STOREV = 'R':
!>
!>               V = ( v1 v2 v3 )                 V = ( v1 v1  1       )
!>                   ( v1 v2 v3 )                     ( v2 v2 v2  1    )
!>                   (  1 v2 v3 )                     ( v3 v3 v3 v3  1 )
!>                   (     1 v3 )
!>                   (        1 )
!>
