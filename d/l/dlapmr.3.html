توثيق الوظيفة/الإجراء الفرعي

subroutine dlapmr (logical forwrd, integer m, integer n, double precision, dimension( ldx, * ) x, integer ldx, integer, dimension( * ) k)

DLAPMR يعيد ترتيب صفوف المصفوفة X كما هو محدد بواسطة متجه التبديل K(1),K(2),...,K(M) للأعداد الصحيحة 1,...,M. إذا كان FORWRD = .TRUE.، تبديل أمامي: X(K(I),*) يتم نقله إلى X(I,*) لـ I = 1,2,...,M. إذا كان FORWRD = .FALSE.، تبديل خلفي: X(I,*) يتم نقله إلى X(K(I),*) لـ I = 1,2,...,M.

الغرض:

!>
!> DLAPMR يعيد ترتيب صفوف المصفوفة M x N المصفوفة X كما هو محدد
!> بواسطة التبديل K(1),K(2),...,K(M) للأعداد الصحيحة 1,...,M.
!> إذا كان FORWRD = .TRUE.، تبديل أمامي:
!>
!>      X(K(I),*) يتم نقله إلى X(I,*) لـ I = 1,2,...,M.
!>
!> إذا كان FORWRD = .FALSE.، تبديل خلفي:
!>
!>      X(I,*) يتم نقله إلى X(K(I),*) لـ I = 1,2,...,M.
!>

المعاملات

FORWRD 
!>          FORWRD هو LOGICAL (منطقي)
!>          = .TRUE.، تبديل أمامي
!>          = .FALSE.، تبديل خلفي
!>

M

!>          M هو INTEGER (عدد صحيح)
!>          عدد صفوف المصفوفة X. M >= 0.
!>

N

!>          N هو INTEGER (عدد صحيح)
!>          عدد أعمدة المصفوفة X. N >= 0.
!>

X

!>          X هو مصفوفة DOUBLE PRECISION (دقة مزدوجة) ، البعد (LDX,N)
!>          عند الإدخال، المصفوفة M x N X.
!>          عند الخروج، تحتوي X على المصفوفة المبدلة X.
!>

LDX

!>          LDX هو INTEGER (عدد صحيح)
!>          البعد الرئيسي للمصفوفة X، LDX >= MAX(1,M).
!>

K

!>          K هو مصفوفة INTEGER (عدد صحيح) ، البعد (M)
!>          عند الإدخال، تحتوي K على متجه التبديل. يتم استخدام K
!>          كمساحة عمل داخلية، ولكن يتم إعادة تعيينها إلى قيمتها الأصلية عند
!>          الخروج.
!>
