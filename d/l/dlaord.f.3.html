subroutine dlaord (character job, integer n, double precision, dimension( * ) x, integer incx)

DLAORD

الغرض:

!>
!> DLAORD يرتب عناصر المتجه x بترتيب تصاعدي أو تنازلي.
!>

المعلمات

JOB 
!>          JOB هو CHARACTER
!>          = 'I':  فرز بترتيب تصاعدي
!>          = 'D':  فرز بترتيب تنازلي
!>

N

!>          N هو INTEGER
!>          طول المتجه X.
!>

X

!>          X هو مصفوفة DOUBLE PRECISION، بعد
!>                         (1+(N-1)*INCX)
!>          عند الدخول، المتجه بطول n ليتم فرزه.
!>          عند الخروج، يتم فرز المتجه x بالترتيب المحدد.
!>

INCX

!>          INCX هو INTEGER
!>          الفجوة بين العناصر المتتالية لـ X. INCX >= 0.
!>
