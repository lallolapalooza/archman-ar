توثيق الوظيفة/الروتين الفرعي

subroutine dlauum (character uplo, integer n, double precision, dimension( lda, * ) a, integer lda, integer info)

DLAUUM تحسب حاصل ضرب UUH أو LHL، حيث U و L مصفوفات مثلثة عليا أو سفلى (خوارزمية محظورة).

الغرض:

!>
!> DLAUUM تحسب حاصل ضرب U * U**T أو L**T * L، حيث يتم تخزين العامل المثلثي U أو L
!> في الجزء العلوي أو السفلي من المصفوفة A.
!>
!> إذا كان UPLO = 'U' أو 'u' فسيتم تخزين المثلث العلوي من النتيجة،
!> والكتابة فوق العامل U في A.
!> إذا كان UPLO = 'L' أو 'l' فسيتم تخزين المثلث السفلي من النتيجة،
!> والكتابة فوق العامل L في A.
!>
!> هذا هو الشكل المحظور للخوارزمية، يستدعي BLAS المستوى 3.
!>

المعلمات

UPLO 
!>          UPLO هو CHARACTER*1
!>          يحدد ما إذا كان العامل المثلثي المخزن في المصفوفة A
!>          هو مثلث علوي أو سفلي:
!>          = 'U':  مثلث علوي
!>          = 'L':  مثلث سفلي
!>

N

!>          N هو INTEGER
!>          رتبة العامل المثلثي U أو L. N >= 0.
!>

A

!>          A هي مصفوفة DOUBLE PRECISION، حجمها (LDA,N)
!>          عند الدخول، العامل المثلثي U أو L.
!>          عند الخروج، إذا كان UPLO = 'U'، فسيتم الكتابة فوق المثلث العلوي لـ A
!>          بالمثلث العلوي لحاصل الضرب U * U**T؛
!>          إذا كان UPLO = 'L'، فسيتم الكتابة فوق المثلث السفلي لـ A
!>          بالمثلث السفلي لحاصل الضرب L**T * L.
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرائد للمصفوفة A. LDA >= max(1,N).
!>

INFO

!>          INFO هو INTEGER
!>          = 0: انتهاء ناجح
!>          < 0: إذا كان INFO = -k، فإن المعلمة k لديها قيمة غير قانونية
!>
