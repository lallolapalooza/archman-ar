توثيق الوظيفة/الإجراء الفرعي

subroutine dlakf2 (integer m, integer n, double precision, dimension( lda, * ) a, integer lda, double precision, dimension( lda, * ) b, double precision, dimension( lda, * ) d, double precision, dimension( lda, * ) e, double precision, dimension( ldz, * ) z, integer ldz)

DLAKF2

الغرض:

!>
!> تشكيل المصفوفة 2*M*N × 2*M*N
!>
!>        Z = [ kron(In, A)  -kron(B', Im) ]
!>            [ kron(In, D)  -kron(E', Im) ],
!>
!> حيث In هي المصفوفة الهوية بحجم n و X' هي معكوس X. kron(X, Y) هو ضرب كرونيكر بين المصفوفات X
!> و Y.
!>

المعلمات

M 
!>          M هو عدد صحيح
!>          حجم المصفوفة، يجب أن يكون >= 1.
!>

N

!>          N هو عدد صحيح
!>          حجم المصفوفة، يجب أن يكون >= 1.
!>

A

!>          A هو double precision، بعد ( LDA, M )
!>          المصفوفة A في المصفوفة الناتجة Z.
!>

LDA

!>          LDA هو عدد صحيح
!>          البعد الرئيسي للمصفوفات A, B, D, و E. ( LDA >= M+N )
!>

B

!>          B هو double precision، بعد ( LDA, N )
!>

D

!>          D هو double precision، بعد ( LDA, M )
!>

E

!>          E هو double precision، بعد ( LDA, N )
!>
!>          المصفوفات المستخدمة في تشكيل المصفوفة الناتجة Z.
!>

Z

!>          Z هو double precision، بعد ( LDZ, 2*M*N )
!>          المصفوفة الناتجة كرونيكر M*N*2 × M*N*2 (انظر أعلاه.)
!>

LDZ

!>          LDZ هو عدد صحيح
!>          البعد الرئيسي للمصفوفة Z. ( LDZ >= 2*M*N )
!>
