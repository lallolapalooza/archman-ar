subroutine dlasq3 (integer i0, integer n0, double precision, dimension( * ) z, integer pp, double precision dmin, double precision sigma, double precision desig, double precision qmax, integer nfail, integer iter, integer ndiv, logical ieee, integer ttype, double precision dmin1, double precision dmin2, double precision dn, double precision dn1, double precision dn2, double precision g, double precision tau)

DLASQ3 يتحقق من الانكماش، يحسب إزاحة ويستدعي dqds. يستخدم بواسطة sbdsqr.

الغرض:

!>
!> DLASQ3 يتحقق من الانكماش، يحسب إزاحة (TAU) ويستدعي dqds.
!> في حالة الفشل يقوم بتغيير الإزاحات، ويحاول مرة أخرى حتى تكون المخرجات
!> موجبة.
!>

المعاملات

I0 
!>          I0 هو عدد صحيح
!>         المؤشر الأول.
!>

N0

!>          N0 هو عدد صحيح
!>         المؤشر الأخير.
!>

Z

!>          Z هو مصفوفة بدقة مزدوجة، البعد ( 4*N0 )
!>         Z يحتوي على مصفوفة qd.
!>

PP

!>          PP هو عدد صحيح
!>         PP=0 للطنين، PP=1 للصدى.
!>         PP=2 يشير إلى أنه تم تطبيق القلب على مصفوفة Z
!>         وأنه لا ينبغي إجراء الاختبارات الأولية للانكماش.
!>

DMIN

!>          DMIN هو دقة مزدوجة
!>         الحد الأدنى لقيمة d.
!>

SIGMA

!>          SIGMA هو دقة مزدوجة
!>         مجموع الإزاحات المستخدمة في المقطع الحالي.
!>

DESIG

!>          DESIG هو دقة مزدوجة
!>         الجزء السفلي من SIGMA
!>

QMAX

!>          QMAX هو دقة مزدوجة
!>         الحد الأقصى لقيمة q.
!>

NFAIL

!>          NFAIL هو عدد صحيح

ITER

NDIV

!>         قم بزيادة NDIV بمقدار 1 لكل قسمة.
!>

IEEE

TTYPE
