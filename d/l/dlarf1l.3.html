وثائق الوظيفة/الروتين الفرعي

subroutine dlarf1l (character side, integer m, integer n, double precision, dimension( * ) v, integer incv, double precision tau, double precision, dimension( ldc, * ) c, integer ldc, double precision, dimension( * ) work)

DLARF1L يطبق عاكسة عنصرية حقيقية H على مصفوفة حقيقية m × n C، إما من اليسار أو من اليمين.

الغرض:

!>
!> DLARF1L يطبق عاكسة عنصرية حقيقية H على مصفوفة حقيقية m × n
!> C، من اليسار أو من اليمين. يتم تمثيل H بالشكل
!>
!>       H = I - tau * v * v**T
!>
!> حيث tau هو عدد قياسي حقيقي و v هو متجه حقيقي.
!>
!> إذا كان tau = 0، يتم اعتبار H مصفوفة الوحدة.
!>

المعلمات

SIDE 
!>          SIDE هو CHARACTER*1
!>          = 'L': تكوين  H * C
!>          = 'R': تكوين  C * H
!>

M

!>          M هو INTEGER
!>          عدد صفوف المصفوفة C.
!>

N

!>          N هو INTEGER
!>          عدد أعمدة المصفوفة C.
!>

V

!>          V هو مصفوفة DOUBLE PRECISION، الأبعاد
!>                     (1 + (M-1)*abs(INCV)) إذا كانت SIDE = 'L'
!>                  أو (1 + (N-1)*abs(INCV)) إذا كانت SIDE = 'R'
!>          المتجه v في تمثيل H. لا يتم استخدام V إذا
!>          كان TAU = 0.
!>

INCV

!>          INCV هو INTEGER
!>          الزيادة بين عناصر v. INCV <> 0.
!>

TAU

!>          TAU هو DOUBLE PRECISION
!>          القيمة tau في تمثيل H.
!>

C

!>          C هي مصفوفة DOUBLE PRECISION، الأبعاد (LDC,N)
!>          عند الإدخال، المصفوفة m × n C.
!>          عند الخروج، يتم استبدال C بالمصفوفة H * C إذا كانت SIDE = 'L'،
!>          أو C * H إذا كانت SIDE = 'R'.
!>

LDC

!>          LDC هو INTEGER
!>          البعد الرائد للمصفوفة C. LDC >= max(1,M).
!>

WORK

!>          WORK هي مصفوفة DOUBLE PRECISION، الأبعاد
!>                         (N) إذا كانت SIDE = 'L'
!>                      أو (M) إذا كانت SIDE = 'R'
!>
