توثيق الوظيفة/الروتين الفرعي

subroutine dlattr (integer imat, character uplo, character trans, character diag, integer, dimension( 4 ) iseed, integer n, double precision, dimension( lda, * ) a, integer lda, double precision, dimension( * ) b, double precision, dimension( * ) work, integer info)

DLATTR

الغرض:

!>
!> DLATTR يولد مصفوفة مثلثية للاختبار.
!> IMAT و UPLO يحددان بشكل فريد خصائص المصفوفة الاختبارية،
!> والتي يتم إرجاعها في المصفوفة A.
!>

المعاملات

IMAT 
!>          IMAT هو عدد صحيح
!>          مفتاح عدد صحيح يصف أي مصفوفة سيتم إنشاؤها لهذا
!>          المسار.
!>

UPLO

!>          UPLO هو حرف بطول 1
!>          يحدد ما إذا كانت المصفوفة A ستكون مثلثية عليا أم سفلية.
!>          = 'U':  مثلثية عليا
!>          = 'L':  مثلثية سفلية
!>

TRANS

!>          TRANS هو حرف بطول 1
!>          يحدد ما إذا كانت المصفوفة أو منقولتها ستستخدم.
!>          = 'N':  بدون منقول
!>          = 'T':  منقول
!>          = 'C':  منقول مترافق (= منقول)
!>

DIAG

!>          DIAG هو حرف بطول 1
!>          يحدد ما إذا كانت المصفوفة A مثلثية وحدة أم لا.
!>          = 'N':  مثلثية غير وحدة
!>          = 'U':  مثلثية وحدة
!>

ISEED

!>          ISEED هي مصفوفة أعداد صحيحة، بأبعاد (4)
!>          متجه البذرة لمولد الأرقام العشوائية (يستخدم في
!>          DLATMS).  يتم تعديله عند الخروج.
!>

N

!>          N هو عدد صحيح
!>          ترتيب المصفوفة المراد إنشاؤها.
!>

A

!>          A هي مصفوفة دقة مزدوجة، بأبعاد (LDA,N)
!>          المصفوفة المثلثية A.  إذا كان UPLO = 'U'، فإن الجزء العلوي
!>          المثلثي العلوي من المصفوفة A بحجم N*N يحتوي على المصفوفة
!>          المثلثية العليا، والجزء السفلي المثلثي الأدنى من
!>          A لا يتم الرجوع إليه.  إذا كان UPLO = 'L'، فإن الجزء السفلي
!>          المثلثي الأدنى من المصفوفة A بحجم N*N يحتوي على المصفوفة
!>          المثلثية السفلى، والجزء العلوي المثلثي الأدنى من A لا
!>          يتم الرجوع إليه.  إذا كان DIAG = 'U'، يتم ضبط العناصر القطرية
!>          لـ A بحيث يكون A(k,k) = k لـ 1 <= k <= n.
!>

LDA

!>          LDA هو عدد صحيح
!>          البعد الرائد للمصفوفة A.  LDA >= max(1,N).
!>

B

!>          B هي مصفوفة دقة مزدوجة، بأبعاد (N)
!>          المتجه الطرف الأيمن، إذا كان IMAT > 10.
!>

WORK

!>          WORK هي مصفوفة دقة مزدوجة، بأبعاد (3*N)
!>

INFO

!>          INFO هو عدد صحيح
!>          = 0:  إنهاء ناجح
!>          < 0: إذا كان INFO = -k، كان المعامل k ذو قيمة غير قانونية
!>
