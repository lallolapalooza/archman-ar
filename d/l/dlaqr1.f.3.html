توثيق الوظيفة/الإجراء الفرعي

subroutine dlaqr1 (integer n, double precision, dimension( ldh, * ) h, integer ldh, double precision sr1, double precision si1, double precision sr2, double precision si2, double precision, dimension( * ) v)

DLAQR1 تحدد مضاعفاً قياسياً للعمود الأول من حاصل ضرب مصفوفة 2x2 أو 3x3 H مع الإزاحات المحددة.

الغرض:

!>
!>      بالنظر إلى مصفوفة H بحجم 2x2 أو 3x3، تحدد DLAQR1 المتجه v كمضاعف قياسي
!>      للعمود الأول من حاصل الضرب
!>
!>      (*)  K = (H - (sr1 + i*si1)*I)*(H - (sr2 + i*si2)*I)
!>
!>      مع القياس لتجنب تجاوزات الأرقام ومعظم حالات النقصان. يُفترض أن أحد الأمرين التاليين صحيح:
!>
!>              1) sr1 = sr2 و si1 = -si2
!>          أو
!>              2) si1 = si2 = 0.
!>
!>      هذا مفيد لبدء تشكيل الفقاعات ذات الإزاحات الضمنية المزدوجة
!>      في خوارزمية QR.
!>

المعلمات

N 
!>          N هو عدد صحيح (INTEGER).
!>              رتبة المصفوفة H. يجب أن تكون N إما 2 أو 3.
!>

H

!>          H هي مصفوفة من النوع DOUBLE PRECISION، أبعادها (LDH,N).
!>              المصفوفة H بحجم 2x2 أو 3x3 في (*).
!>

LDH

!>          LDH هو عدد صحيح (INTEGER).
!>              البعد الرائد للمصفوفة H كما تم تعريفه في
!>              الإجراء المستدعي.  LDH >= N
!>

SR1

!>          SR1 هو من النوع DOUBLE PRECISION.
!>

SI1

!>          SI1 هو من النوع DOUBLE PRECISION.
!>

SR2

!>          SR2 هو من النوع DOUBLE PRECISION.
!>

SI2

!>          SI2 هو من النوع DOUBLE PRECISION.
!>              الإزاحات في (*).
!>

V

!>          V هي مصفوفة من النوع DOUBLE PRECISION، أبعادها (N).
!>              مضاعف قياسي للعمود الأول من المصفوفة K في (*).
!>
