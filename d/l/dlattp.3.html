subroutine dlattp (integer imat, character uplo, character trans, character diag, integer, dimension( 4 ) iseed, integer n, double precision, dimension( * ) a, double precision, dimension( * ) b, double precision, dimension( * ) work, integer info)

DLATTP

الغرض:

!>
!> DLATTP يولد مصفوفة اختبار مثلثة في تخزين مكدس.
!> IMAT و UPLO يحددان خصائص مصفوفة الاختبار بشكل فريد،
!> والتي يتم إرجاعها في المصفوفة AP.
!>

المعاملات

IMAT 
!>          IMAT هو عدد صحيح (INTEGER)
!>          مفتاح عدد صحيح يصف المصفوفة التي سيتم إنشاؤها لهذا
!>          المسار.
!>

UPLO

!>          UPLO هو حرف (CHARACTER*1)
!>          يحدد ما إذا كانت المصفوفة A ستكون مثلثة علوية أو سفلية.
!>          = 'U':  مثلثة علوية
!>          = 'L':  مثلثة سفلية
!>

TRANS

!>          TRANS هو حرف (CHARACTER*1)
!>          يحدد ما إذا كانت المصفوفة أو منقولتها سيتم استخدامها.
!>          = 'N':  بدون منقولة
!>          = 'T':  منقولة
!>          = 'C':  منقولة مترافقة (= منقولة)
!>

DIAG

!>          DIAG هو حرف (CHARACTER*1)
!>          يحدد ما إذا كانت المصفوفة A مثلثة وحدوية أم لا.
!>          = 'N':  مثلثة غير وحدوية
!>          = 'U':  مثلثة وحدوية
!>

ISEED

!>          ISEED هي مصفوفة أعداد صحيحة (INTEGER array)، ذات بعد (4)
!>          متجه البذرة لمولد الأرقام العشوائية (المستخدم في
!>          DLATMS). يتم تعديله عند الخروج.
!>

N

!>          N هو عدد صحيح (INTEGER)
!>          رتبة المصفوفة التي سيتم إنشاؤها.
!>

A

!>          A هي مصفوفة دقة مزدوجة (DOUBLE PRECISION array)، ذات بعد (N*(N+1)/2)
!>          المصفوفة المثلثة A، مكدسة عمودياً في
!>          مصفوفة خطية. يتم تخزين العمود j من A في المصفوفة
!>          AP كما يلي:
!>          إذا كان UPLO = 'U'، فإن AP((j-1)*j/2 + i) = A(i,j) لـ 1<=i<=j؛
!>          إذا كان UPLO = 'L'،
!>             AP((j-1)*(n-j) + j*(j+1)/2 + i-j) = A(i,j) لـ j<=i<=n.
!>

B

!>          B هي مصفوفة دقة مزدوجة (DOUBLE PRECISION array)، ذات بعد (N)
!>          المتجه الأيمن، إذا كان IMAT > 10.
!>

WORK

!>          WORK هي مصفوفة دقة مزدوجة (DOUBLE PRECISION array)، ذات بعد (3*N)
!>

INFO

!>          INFO هو عدد صحيح (INTEGER)
!>          = 0:  انتهاء ناجح
!>          < 0: إذا كان INFO = -k، فإن المعامل k كان له قيمة غير قانونية
!>
