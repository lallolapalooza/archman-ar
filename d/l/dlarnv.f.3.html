subroutine dlarnv (idist, iseed, n, x)
DLARNV تُرجع متجهًا من الأرقام العشوائية من توزيع منتظم أو طبيعي.

توثيق الوظيفة/الإجراء الفرعي

subroutine dlarnv (integer idist, integer, dimension( 4 ) iseed, integer n, double precision, dimension( * ) x)

DLARNV تُرجع متجهًا من n رقمًا حقيقيًا عشوائيًا من توزيع منتظم أو طبيعي.

الغرض:

!>
!> DLARNV تُرجع متجهًا من n رقمًا حقيقيًا عشوائيًا من توزيع منتظم أو
!> طبيعي.
!>

المعلمات

IDIST 
!>          IDIST هو عدد صحيح (INTEGER).
!>          يحدد توزيع الأرقام العشوائية:
!>          = 1:  منتظم (0,1)
!>          = 2:  منتظم (-1,1)
!>          = 3:  طبيعي (0,1)
!>

ISEED

!>          ISEED هو مصفوفة أعداد صحيحة (INTEGER array)، حجمها (4).
!>          عند الإدخال، البذرة لمولد الأرقام العشوائية؛ يجب أن تكون
!>          عناصر المصفوفة بين 0 و 4095، ويجب أن يكون ISEED(4)
!>          فرديًا.
!>          عند الخروج، يتم تحديث البذرة.
!>

N

!>          N هو عدد صحيح (INTEGER).
!>          عدد الأرقام العشوائية التي سيتم إنشاؤها.
!>

X

!>          X هي مصفوفة دقة مزدوجة (DOUBLE PRECISION array)، حجمها (N).
!>          الأرقام العشوائية التي تم إنشاؤها.
!>
