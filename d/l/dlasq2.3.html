subroutine dlasq2 (n, z, info)
DLASQ2 تحسب جميع القيم الذاتية للمصفوفة ثلاثية الأقطار المتناظرة المحددة الإيجابية المرتبطة بمصفوفة qd Z بدقة نسبية عالية. تستخدم بواسطة sbdsqr و sstegr.

توثيق الوظيفة/الإجراء الفرعي

subroutine dlasq2 (integer n, double precision, dimension( * ) z, integer info)

DLASQ2 تحسب جميع القيم الذاتية للمصفوفة ثلاثية الأقطار المتناظرة المحددة الإيجابية المرتبطة بمصفوفة qd Z بدقة نسبية عالية. تستخدم بواسطة sbdsqr و sstegr.

الغرض:

!>
!> DLASQ2 تحسب جميع القيم الذاتية للمصفوفة ثلاثية الأقطار المتناظرة
!> المحددة الإيجابية المرتبطة بمصفوفة qd Z بدقة نسبية عالية،
!> في غياب التشويش والفيضان والتجاوز.
!>
!> لرؤية العلاقة بين Z والمصفوفة ثلاثية الأقطار، دع L تكون
!> مصفوفة قطرية سفلية وحدوية مع الأقطار الفرعية Z(2,4,6,,..)
!> ودع U تكون مصفوفة قطرية علوية مع 1 فوقها وقطر
!> Z(1,3,5,,..). المصفوفة ثلاثية الأقطار هي L*U أو، إذا كنت تفضل،
!> المصفوفة ثلاثية الأقطار المتناظرة التي هي مشابهة لها.
!>
!> ملاحظة: DLASQ2 تعرف متغيرًا منطقيًا، IEEE، وهو صحيح
!> في الأجهزة التي تتبع معيار الفاصلة العائمة ieee-754 في
!> تعاملها مع اللانهائيات و NaNs، وخطأ بخلاف ذلك. يتم تمرير هذا المتغير
!> إلى DLASQ3.
!>

المعلمات

N 
!>          N هو عدد صحيح (INTEGER)
!>        عدد الصفوف والأعمدة في المصفوفة. N >= 0.
!>

Z

!>          Z هي مصفوفة double precision، البعد ( 4*N )
!>        عند الإدخال Z تحمل مصفوفة qd. عند الخروج، تحتفظ الإدخالات من 1 إلى N
!>        بالقيم الذاتية بترتيب تنازلي، Z( 2*N+1 ) تحتفظ بالمجموع (trace)،
!>        و Z( 2*N+2 ) تحتفظ بمجموع القيم الذاتية. إذا
!>        N > 2، فإن Z( 2*N+3 ) تحتفظ بعداد التكرارات، Z( 2*N+4 )
!>        تحتفظ بـ NDIVS/NIN^2، و Z( 2*N+5 ) تحتفظ بنسبة
!>        التحولات التي فشلت.
!>

INFO

!>          INFO هو عدد صحيح (INTEGER)
!>        = 0: خروج ناجح
!>        < 0: إذا كانت المعلمة i عددًا قياسيًا وكانت لها قيمة غير قانونية، فإن INFO = -i،
!>             إذا كانت المعلمة i مصفوفة وكان للإدخال j قيمة غير قانونية، فإن
!>             INFO = -(i*100+j)
!>        > 0: فشلت الخوارزمية
!>              = 1، تم تحديد تقسيم بقيمة موجبة في E
!>              = 2، لم يتم تشخيص كتلة Z الحالية بعد 100*N
!>                   تكرار (في الحلقة الداخلية). عند الخروج Z تحتفظ
!>                   بمصفوفة qd بنفس القيم الذاتية لـ Z المعطاة.
!>              = 3، لم يتم استيفاء معيار الإنهاء للحلقة الخارجية
!>                   (أنشأت البرنامج أكثر من N كتلة غير مختزلة)
!>
