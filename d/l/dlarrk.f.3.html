subroutine dlarrk (n, iw, gl, gu, d, e2, pivmin, reltol, w, werr, info)
DLARRK تحسب قيمة ذاتية لمصفوفة ثلاثية الأقطار متناظرة بدقة مناسبة.

توثيق الوظيفة/الروتين الفرعي

subroutine dlarrk (integer n, integer iw, double precision gl, double precision gu, double precision, dimension( * ) d, double precision, dimension( * ) e2, double precision pivmin, double precision reltol, double precision w, double precision werr, integer info)

DLARRK تحسب قيمة ذاتية لمصفوفة ثلاثية الأقطار متناظرة بدقة مناسبة.

الغرض:

!>
!> DLARRK تحسب قيمة ذاتية لمصفوفة ثلاثية الأقطار متناظرة
!> بدقة مناسبة. هذا هو رمز مساعد سيتم استدعاؤه من DSTEMR.
!>
!> لتجنب الفائض، يجب قياس المصفوفة بحيث لا يتجاوز عنصرها الأكبر
!> قيمة overflow**(1/2) * underflow**(1/4) بالقيمة المطلقة، ولأقصى
!> دقة، يجب ألا تكون أصغر بكثير من ذلك.
!>
!> انظر W. Kahan ، التقرير CS41، قسم علوم الحاسوب، جامعة ستانفورد
!> ، 21 يوليو 1966.
!>

المعلمات

N 
!>          N هو عدد صحيح (INTEGER)
!>          رتبة المصفوفة الثلاثية الأقطار T. N >= 0.
!>

IW

!>          IW هو عدد صحيح (INTEGER)
!>          فهرس القيم الذاتية المطلوب إرجاعها.
!>

GL

!>          GL هو دقة مزدوجة (DOUBLE PRECISION)
!>

GU

!>          GU هو دقة مزدوجة (DOUBLE PRECISION)
!>          حد علوي وسفلي للقيمة الذاتية.
!>

D

!>          D هو مصفوفة دقة مزدوجة (DOUBLE PRECISION)، البعد (N)
!>          عناصر القطر N للمصفوفة الثلاثية الأقطار T.
!>

E2

!>          E2 هو مصفوفة دقة مزدوجة (DOUBLE PRECISION)، البعد (N-1)
!>          العناصر المربعة خارج القطر (N-1) للمصفوفة الثلاثية الأقطار T.
!>

PIVMIN

!>          PIVMIN هو دقة مزدوجة (DOUBLE PRECISION)
!>          الحد الأدنى للمحور المسموح به في تسلسل ستورم لـ T.
!>

RELTOL

!>          RELTOL هو دقة مزدوجة (DOUBLE PRECISION)
!>          الحد الأدنى للعرض النسبي للفترة. عندما تكون الفترة
!>          أضيق من RELTOL مضروبة في الطرف الأكبر (من حيث القيمة المطلقة)،
!>          تعتبر أنها صغيرة بما فيه الكفاية، أي، تم التقارب.
!>          ملاحظة: يجب أن تكون هذه القيمة على الأقل epsilon الآلة * radix.
!>

W

!>          W هو دقة مزدوجة (DOUBLE PRECISION)
!>

WERR

!>          WERR هو دقة مزدوجة (DOUBLE PRECISION)
!>          حد الخطأ على تقريب القيمة الذاتية المقابلة
!>          في W.
!>

INFO

!>          INFO هو عدد صحيح (INTEGER)
!>          = 0:       تقاربت القيمة الذاتية
!>          = -1:      لم تتقارب القيمة الذاتية
!>
