توثيق الوظيفة/الإجراء الفرعي

subroutine dlarrf (integer n, double precision, dimension( * ) d, double precision, dimension( * ) l, double precision, dimension( * ) ld, integer clstrt, integer clend, double precision, dimension( * ) w, double precision, dimension( * ) wgap, double precision, dimension( * ) werr, double precision spdiam, double precision clgapl, double precision clgapr, double precision pivmin, double precision sigma, double precision, dimension( * ) dplus, double precision, dimension( * ) lplus, double precision, dimension( * ) work, integer info)

DLARRF تجد تمثيلاً قوياً نسبياً جديداً بحيث تكون واحدة على الأقل من القيم الذاتية معزولة نسبياً.

الغرض:

!>
!> بالنظر إلى التمثيل الأولي L D L^T ومجموعة القيم الذاتية القريبة نسبياً
!> (بمقياس نسبي)، W( CLSTRT )، W( CLSTRT+1 )، ... W( CLEND )، تجد DLARRF
!> تمثيلاً قوياً نسبياً جديداً L D L^T - SIGMA I = L(+) D(+) L(+)^T بحيث
!> تكون واحدة على الأقل من القيم الذاتية لـ L(+) D(+) L(+)^T معزولة نسبياً.
!>

المعلمات

N 
!>          N هو عدد صحيح
!>          ترتيب المصفوفة (مصفوفة فرعية، إذا تم تقسيم المصفوفة).
!>

D

!>          D هو مصفوفة دقة مزدوجة، البعد (N)
!>          عناصر القطر N للمصفوفة القطرية D.
!>

L

!>          L هو مصفوفة دقة مزدوجة، البعد (N-1)
!>          عناصر تحت القطر (N-1) للمصفوفة ثنائية القطرية الوحدية
!>          L.
!>

LD

!>          LD هو مصفوفة دقة مزدوجة، البعد (N-1)
!>          عناصر L(i)*D(i) تحت القطر (N-1).
!>

CLSTRT

!>          CLSTRT هو عدد صحيح
!>          فهرس أول قيمة ذاتية في المجموعة.
!>

CLEND

!>          CLEND هو عدد صحيح
!>          فهرس آخر قيمة ذاتية في المجموعة.
!>

W

!>          W هو مصفوفة دقة مزدوجة، البعد
!>          البعد >= (CLEND-CLSTRT+1)
!>          القيم الذاتية التقريبية لـ L D L^T بترتيب تصاعدي.
!>          W( CLSTRT ) إلى W( CLEND ) تشكل مجموعة القيم الذاتية القريبة
!>          نسبياً.
!>

WGAP

!>          WGAP هو مصفوفة دقة مزدوجة، البعد
!>          البعد >= (CLEND-CLSTRT+1)
!>          الفصل عن القيمة الذاتية المجاورة اليمنى في W.
!>

WERR

!>          WERR هو مصفوفة دقة مزدوجة، البعد
!>          البعد >= (CLEND-CLSTRT+1)
!>          WERR تحتوي على نصف عرض فترة عدم اليقين
!>          للقيمة الذاتية التقريبية المقابلة في W.
!>

SPDIAM

!>          SPDIAM هو دقة مزدوجة
!>          تقدير للقطر الطيفي تم الحصول عليه من
!>          فترات Gerschgorin.
!>

CLGAPL

!>          CLGAPL هو دقة مزدوجة
!>

CLGAPR

!>          CLGAPR هو دقة مزدوجة
!>          الفجوة المطلقة على كل طرف من المجموعة.
!>          تم تعيينها بواسطة الروتين المتصل للحماية من التحولات القريبة جداً
!>          من القيم الذاتية خارج المجموعة.
!>

PIVMIN

!>          PIVMIN هو دقة مزدوجة
!>          الارتكاز الأدنى المسموح به في تسلسل ستيورم.
!>

SIGMA

!>          SIGMA هو دقة مزدوجة
!>          التحويل المستخدم لتشكيل L(+) D(+) L(+)^T.
!>

DPLUS

!>          DPLUS هو مصفوفة دقة مزدوجة، البعد (N)
!>          عناصر القطر N للمصفوفة القطرية D(+).
!>

LPLUS

!>          LPLUS هو مصفوفة دقة مزدوجة، البعد (N-1)
!>          العناصر (N-1) الأولى لـ LPLUS تحتوي على عناصر تحت القطر
!>          للمصفوفة ثنائية القطرية الوحدية L(+).
!>

WORK

!>          WORK هو مصفوفة دقة مزدوجة، البعد (2*N)
!>          مساحة عمل.
!>

INFO

!>          INFO هو عدد صحيح
!>          يشير إلى معالجة ناجحة (=0) أو فشل (=1)
!>
