subroutine dlasd0 (n, sqre, d, e, u, ldu, vt, ldvt, smlsiz, iwork, work, info)
DLASD0 تحسب القيم المنفردة لمصفوفة علوية ثنائية القطر حقيقية n-by-m B مع القطر d والمحور e. تستخدم بواسطة sbdsdc.

توثيق الدالة/الإجراء الفرعي

subroutine dlasd0 (integer n, integer sqre, double precision, dimension( * ) d, double precision, dimension( * ) e, double precision, dimension( ldu, * ) u, integer ldu, double precision, dimension( ldvt, * ) vt, integer ldvt, integer smlsiz, integer, dimension( * ) iwork, double precision, dimension( * ) work, integer info)

DLASD0 تحسب القيم المنفردة لمصفوفة علوية ثنائية القطر حقيقية n-by-m B مع القطر D والمحور E. تستخدم بواسطة sbdsdc.

الغرض:

!>
!> باستخدام نهج التقسيم والغزو، تقوم DLASD0 بحساب
!> تفكيك القيم المنفردة (SVD) لمصفوفة B ثنائية القطر العلوية الحقيقية N-by-M
!> مع القطر D والمحور E، حيث M = N + SQRE.
!> تحسب الخوارزمية المصفوفات المتعامدة U و VT بحيث
!> B = U * S * VT. يتم استبدال القيم المنفردة S في D.
!>
!> تقوم الإجراء الفرعي ذي الصلة، DLASDA، بحساب القيم المنفردة فقط،
!> واختيارياً، المتجهات المنفردة في شكل مضغوط.
!>

المعلمات

N 
!>          N هو عدد صحيح
!>         عند الإدخال، بعد الصفوف للمصفوفة ثنائية القطر العلوية.
!>         هذا هو أيضًا بعد الصف الرئيسي Q.
!>

SQRE

!>          SQRE هو عدد صحيح
!>         يحدد بعد الأعمدة للمصفوفة ثنائية القطر.
!>         = 0: المصفوفة ثنائية القطر لها بعد العمود M = N؛
!>         = 1: المصفوفة ثنائية القطر لها بعد العمود M = N+1؛
!>

D

!>          D هو مصفوفة ذات دقة مزدوجة، بعد (N)
!>         عند الإدخال، يحتوي D على القطر الرئيسي للمصفوفة
!>         ثنائية القطر.
!>         عند الخروج من D، إذا كان INFO = 0، فإنه يحتوي على قيمه المنفردة.
!>

E

!>          E هو مصفوفة ذات دقة مزدوجة، بعد (M-1)
!>         يحتوي على إدخالات تحت القطر للمصفوفة ثنائية القطر.
!>         عند الخروج، تم تدمير E.
!>

U

!>          U هي مصفوفة ذات دقة مزدوجة، بعد (LDU, N)
!>         عند الخروج، يحتوي U على المتجهات المنفردة اليسرى،
!>          إذا تم تمرير U كـ (N, N) Identity.
!>

LDU

!>          LDU هو عدد صحيح
!>         عند الإدخال، البعد الرائد لـ U.
!>

VT

!>          VT هي مصفوفة ذات دقة مزدوجة، بعد (LDVT, M)
!>         عند الخروج، يحتوي VT**T على المتجهات المنفردة اليمنى،
!>          إذا تم تمرير VT كـ (M, M) Identity.
!>

LDVT

!>          LDVT هو عدد صحيح
!>         عند الإدخال، البعد الرائد لـ VT.
!>

SMLSIZ

!>          SMLSIZ هو عدد صحيح
!>         عند الإدخال، الحد الأقصى لحجم المشاكل الفرعية في
!>         أسفل شجرة الحساب.
!>

IWORK

!>          IWORK هي مصفوفة عدد صحيح، بعد (8*N)
!>

WORK

!>          WORK هي مصفوفة ذات دقة مزدوجة، بعد (3*M**2+2*M)
!>

INFO

!>          INFO هو عدد صحيح
!>          = 0: خروج ناجح.
!>          < 0: إذا كان INFO = -i، فإن المعلمة i كانت قيمة غير قانونية.
!>          > 0: إذا كان INFO = 1، لم تتقارب قيمة منفردة.
!>
