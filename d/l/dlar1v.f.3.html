subroutine dlar1v (integer n, integer b1, integer bn, double precision lambda, double precision, dimension( * ) d, double precision, dimension( * ) l, double precision, dimension( * ) ld, double precision, dimension( * ) lld, double precision pivmin, double precision gaptol, double precision, dimension( * ) z, logical wantnc, integer negcnt, double precision ztz, double precision mingma, integer r, integer, dimension( * ) isuppz, double precision nrminv, double precision resid, double precision rqcorr, double precision, dimension( * ) work)

DLAR1V يحسب العمود r (المُقاس) للمصفوفة العكسية للمصفوفة الفرعية من الصفوف b1 إلى bn للمصفوفة الثلاثية القطرية LDLT - λI.

الغرض:

!>
!> DLAR1V يحسب العمود r (المُقاس) للمصفوفة العكسية للمصفوفة الفرعية
!> من الصفوف B1 إلى BN للمصفوفة الثلاثية القطرية
!> L D L**T - sigma I. عندما تكون sigma قريبة من قيمة ذاتية،
!> فإن المتجه المحسوب يكون متجهًا ذاتيًا دقيقًا. عادةً، يتوافق r
!> مع الفهرس حيث يكون المتجه الذاتي هو الأكبر قيمة مطلقة.
!> الخطوات التالية تحقق هذا الحساب:
!> (أ) تحويل qd الثابت، L D L**T - sigma I = L(+) D(+) L(+)**T،
!!> (ب) تحويل qd التقدمي، L D L**T - sigma I = U(-) D(-) U(-)**T،
!> (ج) حساب العناصر القطرية للمصفوفة العكسية لـ
!>     L D L**T - sigma I عن طريق دمج التحويلات أعلاه، واختيار
!>     r كفهرس حيث تكون القطرية العكسية (من بين)
!>     الأكبر قيمة مطلقة.
!> (د) حساب العمود r (المُقاس) للمصفوفة العكسية باستخدام
!>     التحليل الملتوي الذي تم الحصول عليه بدمج الجزء العلوي من
!>     التحويل الثابت والجزء السفلي من التحويل التقدمي.
!>

المعلمات

N 
!>          N هو INTEGER
!>           رتبة المصفوفة L D L**T.
!>

B1

!>          B1 هو INTEGER
!>           الفهرس الأول للمصفوفة الفرعية لـ L D L**T.
!>

BN

!>          BN هو INTEGER
!>           الفهرس الأخير للمصفوفة الفرعية لـ L D L**T.
!>

LAMBDA

!>          LAMBDA هو DOUBLE PRECISION
!>           الإزاحة. لحساب متجه ذاتي دقيق،
!>           يجب أن تكون LAMBDA تقريبًا جيدًا لقيمة ذاتية
!>           لـ L D L**T.
!>

L

!>          L هو مصفوفة DOUBLE PRECISION، أبعادها (N-1)
!>           عناصر تحت القطر الفرعية (n-1) للمصفوفة الوحدوية ثنائية القطرية
!>           L، في العناصر من 1 إلى N-1.
!>

D

!>          D هو مصفوفة DOUBLE PRECISION، أبعادها (N)
!>           عناصر القطرية n للمصفوفة القطرية D.
!>

LD

!>          LD هو مصفوفة DOUBLE PRECISION، أبعادها (N-1)
!>           عناصر L(i)*D(i) البالغ عددها n-1.
!>

LLD

!>          LLD هو مصفوفة DOUBLE PRECISION، أبعادها (N-1)
!>           عناصر L(i)*L(i)*D(i) البالغ عددها n-1.
!>

PIVMIN

!>          PIVMIN هو DOUBLE PRECISION
!>           أصغر محور في تسلسل ستورم.
!>

GAPTOL

!>          GAPTOL هو DOUBLE PRECISION
!>           تفاوت يشير إلى متى تكون مدخلات المتجه الذاتي ضئيلة
!>           بالنسبة لمساهمتها في الباقي.
!>

Z

!>          Z هو مصفوفة DOUBLE PRECISION، أبعادها (N)
!>           عند الإدخال، يجب تعيين جميع مدخلات Z إلى 0.
!>           عند الإخراج، تحتوي Z على العمود r (المُقاس) للمصفوفة العكسية.
!>           القياس هو بحيث Z(R) يساوي 1.
!>

WANTNC

!>          WANTNC هو LOGICAL
!>           يحدد ما إذا كان يجب حساب NEGCNT.
!>

NEGCNT

!>          NEGCNT هو INTEGER
!>           إذا كان WANTNC هو .TRUE.، فإن NEGCNT = عدد المحاور < pivmin
!>           في تحليل المصفوفة L D L**T، و NEGCNT = -1 بخلاف ذلك.
!>

ZTZ

!>          ZTZ هو DOUBLE PRECISION
!>           مربع المعيار 2 للمتجه Z.
!>

MINGMA

!>          MINGMA هو DOUBLE PRECISION
!>           معكوس أكبر (من حيث القيمة المطلقة) عنصر قطري
!>           في المصفوفة العكسية لـ L D L**T - sigma I.
!>

R

!>          R هو INTEGER
!>           فهرس الالتواء للتحليل الملتوي المستخدم لحساب
!>           Z.
!>           عند الإدخال، 0 <= R <= N. إذا تم إدخال R كـ 0، يتم تعيين R إلى
!>           الفهرس حيث يكون (L D L**T - sigma I)^{-1} أكبر
!>           قيمة مطلقة. إذا كان 1 <= R <= N، فإن R يبقى دون تغيير.
!>           عند الإخراج، تحتوي R على فهرس الالتواء المستخدم لحساب Z.
!>           من الناحية المثالية، يحدد R موضع أكبر عنصر في
!>           المتجه الذاتي.
!>

ISUPPZ

!>          ISUPPZ هو مصفوفة INTEGER، أبعادها (2)
!>           دعم المتجه في Z، أي أن المتجه Z
!>           غير صفري فقط في العناصر من ISUPPZ(1) إلى ISUPPZ( 2 ).
!>

NRMINV

!>          NRMINV هو DOUBLE PRECISION
!>           NRMINV = 1/SQRT( ZTZ )
!>

RESID

!>          RESID هو DOUBLE PRECISION
!>           باقي المتجه FP.
!>           RESID = ABS( MINGMA )/SQRT( ZTZ )
!>

RQCORR

!>          RQCORR هو DOUBLE PRECISION
!>           تصحيح حاصل قسمة رايلي لـ LAMBDA.
!>           RQCORR = MINGMA*TMP
!>

WORK

!>          WORK هو مصفوفة DOUBLE PRECISION، أبعادها (4*N)
!>
