وثائق الوظيفة/الروتين الفرعي

subroutine dlaqp2rk (integer m, integer n, integer nrhs, integer ioffset, integer kmax, double precision abstol, double precision reltol, integer kp1, double precision maxc2nrm, double precision, dimension( lda, * ) a, integer lda, integer k, double precision maxc2nrmk, double precision relmaxc2nrmk, integer, dimension( * ) jpiv, double precision, dimension( * ) tau, double precision, dimension( * ) vn1, double precision, dimension( * ) vn2, double precision, dimension( * ) work, integer info)

DLAQP2RK تحسب عامل QR مقطوع (مرتبة K) مع تدوير الأعمدة لمصفوفة حقيقية باستخدام BLAS من المستوى 2، وتستبدل مصفوفة B بحجم m-by-nrhs بـ Q**T * B.

الغرض:

!>
!> DLAQP2RK تحسب عامل QR مقطوع (مرتبة K) أو عامل QR كامل مع تدوير
!> الأعمدة لمصفوفة حقيقية
!> كتلة A(IOFFSET+1:M,1:N) على النحو التالي:
!>
!>   A * P(K) = Q(K) * R(K).
!>
!> يستخدم الروتين الفرعي BLAS من المستوى 2. الكتلة A(1:IOFFSET,1:N)
!> يتم تدويرها وفقًا لذلك، ولكن لا يتم تحليلها.
!>
!> يقوم الروتين الفرعي أيضًا باستبدال كتلة المصفوفة ذات الجوانب اليمنى B
!> المخزنة في A(IOFFSET+1:M,N+1:N+NRHS) بـ Q(K)**T * B.
!>

المعاملات

M 
!>          M هو عدد صحيح (INTEGER)
!>          عدد الصفوف في المصفوفة A. M >= 0.
!>

N

!>          N هو عدد صحيح (INTEGER)
!>          عدد الأعمدة في المصفوفة A. N >= 0.
!>

NRHS

!>          NRHS هو عدد صحيح (INTEGER)
!>          عدد الجوانب اليمنى، أي عدد
!>          الأعمدة في المصفوفة B. NRHS >= 0.
!>

IOFFSET

!>          IOFFSET هو عدد صحيح (INTEGER)
!>          عدد صفوف المصفوفة A التي يجب تدويرها
!>          ولكن لا يتم تحليلها. IOFFSET >= 0.
!>
!>          IOFFSET يمثل أيضًا عدد أعمدة المصفوفة الأصلية A_orig
!>          التي تم تحليلها
!>          في الخطوات السابقة.
!>

KMAX

!>          KMAX هو عدد صحيح (INTEGER)
!>
!>          معيار التوقف الأول للتحليل. KMAX >= 0.
!>
!>          الحد الأقصى لعدد أعمدة المصفوفة A التي سيتم تحليلها،
!>          أي الحد الأقصى لمرتبة التحليل.
!>
!>          أ) إذا كان KMAX >= min(M-IOFFSET,N)، فسيتم تجاهل معيار
!>                التوقف هذا، ويتم تحليل الأعمدة
!>                اعتمادًا على ABSTOL و RELTOL.
!>
!>          ب) إذا كان KMAX = 0، فسيتم استيفاء معيار التوقف هذا
!>             عند الإدخال ويخرج الروتين الفرعي فورًا.
!>             هذا يعني أنه لم يتم إجراء التحليل،
!>             ولم يتم تعديل المصفوفات A و B والمصفوفات TAU و IPIV.
!>

ABSTOL

!>          ABSTOL هو دقة مزدوجة (DOUBLE PRECISION)، لا يمكن أن تكون NaN.
!>
!>          معيار التوقف الثاني للتحليل.
!>
!>          التفاوت المطلق (عتبة التوقف) لـ
!>          معيار العمود الأقصى 2-norm للمصفوفة المتبقية.
!>          يتقارب الخوارزمية (توقف التحليل) عندما
!>          يكون معيار العمود الأقصى 2-norm للمصفوفة المتبقية
!>          أقل من أو يساوي ABSTOL.
!>
!>          أ) إذا كان ABSTOL < 0.0، فلن يتم استخدام معيار التوقف هذا،
!>                ويقوم الروتين الفرعي بتحليل الأعمدة اعتمادًا
!>                على KMAX و RELTOL.
!>                يشمل ذلك حالة ABSTOL = -Inf.
!>
!>          ب) إذا كان 0.0 <= ABSTOL، فسيتم استخدام القيمة المدخلة
!>                لـ ABSTOL.
!>

RELTOL

!>          RELTOL هو دقة مزدوجة (DOUBLE PRECISION)، لا يمكن أن تكون NaN.
!>
!>          معيار التوقف الثالث للتحليل.
!>
!>          التفاوت (عتبة التوقف) لنسبة
!>          معيار العمود الأقصى 2-norm للمصفوفة المتبقية إلى الحد الأقصى
!>          لمعيار العمود 2-norm للمصفوفة الأصلية A_orig. تتقارب الخوارزمية
!>          (توقف التحليل)، عندما تكون هذه النسبة
!>          أقل من أو تساوي RELTOL.
!>
!>          أ) إذا كان RELTOL < 0.0، فلن يتم استخدام معيار التوقف هذا،
!>                ويقوم الروتين الفرعي بتحليل الأعمدة اعتمادًا
!>                على KMAX و ABSTOL.
!>                يشمل ذلك حالة RELTOL = -Inf.
!>
!>          د) إذا كان 0.0 <= RELTOL، فسيتم استخدام القيمة المدخلة
!>                لـ RELTOL.
!>

KP1

!>          KP1 هو عدد صحيح (INTEGER)
!>          مؤشر العمود ذي المعيار 2-norm الأقصى في
!>          المصفوفة الأصلية الكاملة A_orig التي تم تحديدها في
!>          الروتين الرئيسي DGEQP3RK. 1 <= KP1 <= N_orig_mat.
!>

MAXC2NRM

!>          MAXC2NRM هو دقة مزدوجة (DOUBLE PRECISION)
!>          معيار العمود الأقصى 2-norm للمصفوفة الأصلية الكاملة
!>          A_orig المحسوب في الروتين الرئيسي DGEQP3RK.
!>          MAXC2NRM >= 0.
!>

A

!>          A هي مصفوفة دقة مزدوجة (DOUBLE PRECISION)، أبعادها (LDA,N+NRHS)
!>          عند الإدخال:
!>              المصفوفة A بحجم M-by-N والمصفوفة B بحجم M-by-NRHS، كما في
!>
!>                                  N     NRHS
!>              array_A   =   M  [ mat_A, mat_B ]
!>
!>          عند الخروج:
!>          1. العناصر في الكتلة A(IOFFSET+1:M,1:K) أسفل
!>             القطر مع المصفوفة TAU تمثل
!>             المصفوفة المتعامدة Q(K) كناتج عن انعكاسات أولية.
!>          2. الكتلة المثلثية العليا للمصفوفة A المخزنة
!>             في A(IOFFSET+1:M,1:K) هي العامل المثلثي الناتج.
!>          3. الكتلة A المخزنة في A(1:IOFFSET,1:N)
!>             تم تدويرها وفقًا لذلك، ولكن لم يتم تحليلها.
!>          4. باقي المصفوفة A، الكتلة A(IOFFSET+1:M,K+1:N+NRHS).
!>             يحتوي الجزء الأيسر A(IOFFSET+1:M,K+1:N) من هذه الكتلة
!>             على باقي المصفوفة A، وإذا كان NRHS > 0، فإن الجزء الأيمن من الكتلة
!>             A(IOFFSET+1:M,N+1:N+NRHS) يحتوي على الكتلة
!>             للمصفوفة ذات الجوانب اليمنى B. تم تحديث كلتا هاتين الكتلتين
!>             بالضرب من اليسار بواسطة Q(K)**T.
!>

LDA

!>          LDA هو عدد صحيح (INTEGER)
!>          البعد الرائد للمصفوفة A. LDA >= max(1,M).
!>

K

!>          K هو عدد صحيح (INTEGER)
!>          مرتبة تحليل المصفوفة A، أي مرتبة
!>          العامل R، والتي هي نفس عدد الصفوف غير الصفرية
!>          للعامل R. 0 <= K <= min(M-IOFFSET,KMAX,N).
!>
!>          K يمثل أيضًا عدد المتجهات هاوسهولدر غير الصفرية.
!>

MAXC2NRMK

!>          MAXC2NRMK هو دقة مزدوجة (DOUBLE PRECISION)
!>          معيار العمود الأقصى 2-norm للمصفوفة المتبقية،
!>          عندما توقف التحليل عند المرتبة K. MAXC2NRMK >= 0.
!>

RELMAXC2NRMK

!>          RELMAXC2NRMK هو دقة مزدوجة (DOUBLE PRECISION)
!>          النسبة MAXC2NRMK / MAXC2NRM لمعيار العمود الأقصى
!>          2-norm للمصفوفة المتبقية (عندما توقف التحليل
!>          عند المرتبة K) إلى معيار العمود الأقصى
!>          2-norm للمصفوفة الأصلية الكاملة A. RELMAXC2NRMK >= 0.
!>

JPIV

!>          JPIV هي مصفوفة أعداد صحيحة (INTEGER array)، أبعادها (N)
!>          مؤشرات تدوير الأعمدة
