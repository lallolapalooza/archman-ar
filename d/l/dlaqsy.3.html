subroutine dlaqsy (character uplo, integer n, double precision, dimension( lda, * ) a, integer lda, double precision, dimension( * ) s, double precision scond, double precision amax, character equed)

DLAQSY تقوم بتوسيع مصفوفة متماثلة/هيرميتية، باستخدام عوامل القياس المحسوبة بواسطة spoequ.

الغرض:

!>
!> DLAQSY تقوم بمعادلة مصفوفة متماثلة A باستخدام عوامل القياس
!> في المتجه S.
!>

المعاملات

UPLO 
!>          UPLO هو CHARACTER*1
!>          يحدد ما إذا كان الجزء العلوي أو السفلي المثلثي من المصفوفة
!>          المتماثلة A مخزنًا.
!>          = 'U':  مثلث علوي
!>          = 'L':  مثلث سفلي
!>

N

!>          N هو INTEGER
!>          رتبة المصفوفة A.  N >= 0.
!>

A

!>          A هو مصفوفة DOUBLE PRECISION، البعد (LDA,N)
!>          عند الإدخال، المصفوفة المتماثلة A.  إذا كان UPLO = 'U'، فإن الجزء
!>          العلوي المثلثي n × n الرائد في A يحتوي على الجزء العلوي
!>          المثلثي من المصفوفة A، والجزء السفلي المثلثي الصارم من A لا يتم
!>          الإشارة إليه.  إذا كان UPLO = 'L'، فإن الجزء السفلي
!>          المثلثي n × n الرائد في A يحتوي على الجزء السفلي
!>          المثلثي من المصفوفة A، والجزء العلوي المثلثي الصارم من A لا يتم
!>          الإشارة إليه.
!>
!>          عند الخروج، إذا كان EQUED = 'Y'، المصفوفة المعادلة:
!>          diag(S) * A * diag(S).
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرائد للمصفوفة A.  LDA >= max(N,1).
!>

S

!>          S هو مصفوفة DOUBLE PRECISION، البعد (N)
!>          عوامل القياس لـ A.
!>

SCOND

!>          SCOND هو DOUBLE PRECISION
!>          نسبة أصغر S(i) إلى أكبر S(i).
!>

AMAX

!>          AMAX هو DOUBLE PRECISION
!>          القيمة المطلقة لأكبر مدخل في المصفوفة.
!>

EQUED

!>          EQUED هو CHARACTER*1
!>          يحدد ما إذا تمت المعادلة أم لا.
!>          = 'N':  لم تتم المعادلة.
!>          = 'Y':  تمت المعادلة، أي تم استبدال A بـ
!>                  diag(S) * A * diag(S).
!>

المعاملات الداخلية:

!>  THRESH هو قيمة حدية تستخدم لتحديد ما إذا كان يجب إجراء القياس
!>  بناءً على نسبة عوامل القياس.  إذا كان SCOND < THRESH،
!>  يتم إجراء القياس.
!>
!>  LARGE و SMALL هما قيمتان حدّيتان تستخدمان لتحديد ما إذا كان يجب
!>  إجراء القياس بناءً على الحجم المطلق لأكبر عنصر في المصفوفة.
!>  إذا كان AMAX > LARGE أو AMAX < SMALL، يتم إجراء القياس.
!>
