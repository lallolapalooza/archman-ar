subroutine dlarra (integer n, double precision, dimension( * ) d, double precision, dimension( * ) e, double precision, dimension( * ) e2, double precision spltol, double precision tnrm, integer nsplit, integer, dimension( * ) isplit, integer info)

DLARRA يحسب نقاط التقسيم بالحد المحدد.

الغرض:

!>
!> حساب نقاط التقسيم بالحد SPLTOL.
!> DLARRA يضبط أي عناصر خارج القطر إلى الصفر.
!>

المعلمات

N 
!>          N هو عدد صحيح (INTEGER)
!>          ترتيب المصفوفة. N > 0.
!>

D

!>          D هو مصفوفة (array) ذات دقة مزدوجة (DOUBLE PRECISION)
!>          بحجم (N).
!>          عند الإدخال، تحتوي العناصر القطرية N للمصفوفة الثلاثية
!>          القطر T.
!>

E

!>          E هو مصفوفة (array) ذات دقة مزدوجة (DOUBLE PRECISION)
!>          بحجم (N).
!>          عند الإدخال، تحتوي العناصر (N-1) الأولى على عناصر
!>          القطر الفرعي للمصفوفة الثلاثية القطر T؛ لا تحتاج E(N)
!>          إلى التعيين.
!>          عند الخروج، يتم ضبط العناصر E( ISPLIT( I ) )،
!>          حيث 1 <= I <= NSPLIT، إلى الصفر، بينما تبقى العناصر
!>          الأخرى في E دون تغيير.
!>

E2

!>          E2 هو مصفوفة (array) ذات دقة مزدوجة (DOUBLE PRECISION)
!>          بحجم (N).
!>          عند الإدخال، تحتوي العناصر (N-1) الأولى على مربعات
!>          عناصر القطر الفرعي للمصفوفة الثلاثية القطر T؛
!>          لا تحتاج E2(N) إلى التعيين.
!>          عند الخروج، تم ضبط العناصر E2( ISPLIT( I ) )،
!>          حيث 1 <= I <= NSPLIT، إلى الصفر.
!>

SPLTOL

!>          SPLTOL هو دقة مزدوجة (DOUBLE PRECISION)
!>          الحد لعملية التقسيم. يمكن استخدام معيارين:
!>          SPLTOL<0 : معيار يعتمد على قيمة خارج القطر المطلقة
!>          SPLTOL>0 : معيار يحافظ على الدقة النسبية
!>

TNRM

!>          TNRM هو دقة مزدوجة (DOUBLE PRECISION)
!>          معيار المصفوفة T.
!>

NSPLIT

!>          NSPLIT هو عدد صحيح (INTEGER)
!>          عدد الكتل التي تنقسم إليها T. 1 <= NSPLIT <= N.
!>

ISPLIT

!>          ISPLIT هي مصفوفة (array) ذات عدد صحيح (INTEGER)
!>          بحجم (N).
!>          نقاط التقسيم، التي تنقسم عندها T إلى كتل.
!>          الكتلة الأولى تتكون من الصفوف/الأعمدة من 1 إلى ISPLIT(1)،
!>          الثانية من الصفوف/الأعمدة ISPLIT(1)+1 حتى ISPLIT(2)،
!>          وهكذا، والكتلة رقم NSPLIT تتكون من الصفوف/الأعمدة
!>          ISPLIT(NSPLIT-1)+1 حتى ISPLIT(NSPLIT)=N.
!>

INFO

!>          INFO هو عدد صحيح (INTEGER)
!>          = 0:  خروج ناجح
!>
