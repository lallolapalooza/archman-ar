subroutine dlals0 (integer icompq, integer nl, integer nr, integer sqre, integer nrhs, double precision, dimension( ldb, * ) b, integer ldb, double precision, dimension( ldbx, * ) bx, integer ldbx, integer, dimension( * ) perm, integer givptr, integer, dimension( ldgcol, * ) givcol, integer ldgcol, double precision, dimension( ldgnum, * ) givnum, integer ldgnum, double precision, dimension( ldgnum, * ) poles, double precision, dimension( * ) difl, double precision, dimension( ldgnum, * ) difr, double precision, dimension( * ) z, integer k, double precision c, double precision s, double precision, dimension( * ) work, integer info)

DLALS0 يطبق عوامل الضرب الخلفية في حل مشكلة المربعات الصغرى باستخدام نهج تفرقة ودمج SVD. يستخدم بواسطة sgelsd.

الغرض:

!>
!> DLALS0 يطبق عوامل الضرب الخلفية إما لمصفوفة المتجهات المفردة اليسرى أو اليمنى لمصفوفة قطرية مضافة إلى صف
!> لمصفوفة الطرف الأيمن B في حل مشكلة المربعات الصغرى باستخدام نهج SVD للتفرقة والدمج.
!>
!> لمصفوفة المتجهات المفردة اليسرى، تتضمن ثلاثة أنواع من المصفوفات المتعامدة:
!>
!> (1L) دورانات جيفنز: عدد هذه الدورانات هو GIVPTR؛ تم تخزين أزواج الأعمدة/الصفوف التي تم تطبيقها عليها في GIVCOL؛
!> وتم تخزين قيم C و S لهذه الدورانات في GIVNUM.
!>
!> (2L) التبديل. الصف (NL+1) من B سيتم نقله إلى الصف الأول، ولـ J=2:N، سيتم نقل الصف PERM(J) من B إلى
!> الصف J.
!>
!> (3L) مصفوفة المتجهات المفردة اليسرى للمصفوفة المتبقية.
!>
!> لمصفوفة المتجهات المفردة اليمنى، تتضمن أربعة أنواع من المصفوفات المتعامدة:
!>
!> (1R) مصفوفة المتجهات المفردة اليمنى للمصفوفة المتبقية.
!>
!> (2R) إذا كان SQRE = 1، دورة جيفنز إضافية لتوليد الفضاء الصفري الأيمن.
!>
!> (3R) التحويل العكسي لـ (2L).
!>
!> (4R) التحويل العكسي لـ (1L).
!>

المعاملات

ICOMPQ 
!>          ICOMPQ هو INTEGER
!>         يحدد ما إذا كان سيتم حساب المتجهات المفردة في شكل محسوب:
!>         = 0: مصفوفة المتجهات المفردة اليسرى.
!>         = 1: مصفوفة المتجهات المفردة اليمنى.
!>

NL

!>          NL هو INTEGER
!>         البعد الصفي للكتلة العلوية. NL >= 1.
!>

NR

!>          NR هو INTEGER
!>         البعد الصفي للكتلة السفلية. NR >= 1.
!>

SQRE

!>          SQRE هو INTEGER
!>         = 0: الكتلة السفلية هي مصفوفة مربعة بحجم NR-by-NR.
!>         = 1: الكتلة السفلية هي مصفوفة مستطيلة بحجم NR-by-(NR+1).
!>
!>         المصفوفة ثنائية القطر لها بعد صفي N = NL + NR + 1،
!>         وبعد عمودي M = N + SQRE.
!>

NRHS

!>          NRHS هو INTEGER
!>         عدد أعمدة B و BX. يجب أن يكون NRHS على الأقل 1.
!>

B

!>          B هو مصفوفة DOUBLE PRECISION، البعد ( LDB, NRHS )
!>         عند الإدخال، تحتوي B على الطرف الأيمن لمشكلة المربعات الصغرى في الصفوف من 1 إلى M. عند الإخراج، تحتوي B على
!> الحل X في الصفوف من 1 إلى N.
!>

LDB

!>          LDB هو INTEGER
!>         البعد الرائد لـ B. يجب أن يكون LDB على الأقل
!>         max(1,MAX( M, N ) ).
!>

BX

!>          BX هو مصفوفة DOUBLE PRECISION، البعد ( LDBX, NRHS )
!>

LDBX

!>          LDBX هو INTEGER
!>         البعد الرائد لـ BX.
!>

PERM

!>          PERM هي مصفوفة INTEGER، البعد ( N )
!>         التبادلات (من التفريغ والفرز) المطبقة
!>         على الكتلتين.
!>

GIVPTR

!>          GIVPTR هو INTEGER
!>         عدد دورانات جيفنز التي حدثت في هذه
!>         المشكلة الفرعية.
!>

GIVCOL

!>          GIVCOL هي مصفوفة INTEGER، البعد ( LDGCOL, 2 )
!>         يشير كل زوج من الأرقام إلى زوج من الصفوف/الأعمدة
!>         المشتركة في دوران جيفنز.
!>

LDGCOL

!>          LDGCOL هو INTEGER
!>         البعد الرائد لـ GIVCOL، يجب أن يكون على الأقل N.
!>

GIVNUM

!>          GIVNUM هي مصفوفة DOUBLE PRECISION، البعد ( LDGNUM, 2 )
!>         يشير كل رقم إلى قيمة C أو S المستخدمة في
!>         دوران جيفنز المقابل.
!>

LDGNUM

!>          LDGNUM هو INTEGER
!>         البعد الرائد للمصفوفات DIFR، POLES، و
!>         GIVNUM، يجب أن يكون على الأقل K.
!>

POLES

!>          POLES هي مصفوفة DOUBLE PRECISION، البعد ( LDGNUM, 2 )
!>         عند الإدخال، تحتوي POLES(1:K, 1) على القيم المفردة المحدثة
!>         الناتجة عن حل المعادلة العلمانية، و
!>         POLES(1:K, 2) هي مصفوفة تحتوي على الأقطاب في
!>         المعادلة العلمانية.
!>

DIFL

!>          DIFL هي مصفوفة DOUBLE PRECISION، البعد ( K ).
!>         عند الإدخال، DIFL(I) هي المسافة بين القيمة المفردة المحدثة
!>         (غير المفروغة) رقم I والقيمة المفردة القديمة (غير المفروغة)
!>         رقم I.
!>

DIFR

!>          DIFR هي مصفوفة DOUBLE PRECISION، البعد ( LDGNUM, 2 ).
!>         عند الإدخال، تحتوي DIFR(I, 1) على المسافات بين القيمة المفردة المحدثة
!>         (غير المفروغة) رقم I والقيمة المفردة القديمة (غير المفروغة)
!>         رقم I+1. و DIFR(I, 2) هو
!>         عامل التطبيع للمتجه المفرد الأيمن رقم I.
!>

Z

!>          Z هي مصفوفة DOUBLE PRECISION، البعد ( K )
!>         تحتوي على مكونات متجه الصف المحدث بعد التفريغ
!>         .
!>

K

!>          K هو INTEGER
!>         تحتوي على بعد المصفوفة غير المفروغة،
!>         هذا هو ترتيب المعادلة العلمانية المرتبطة. 1 <= K <=N.
!>

C

!>          C هو DOUBLE PRECISION
!>         تحتوي C على قمامة إذا كان SQRE = 0 وقيمة C لدوران
!>         جيفنز المتعلق بالفضاء الصفري الأيمن إذا كان SQRE = 1.
!>

S

!>          S هو DOUBLE PRECISION
!>         تحتوي S على قمامة إذا كان SQRE = 0 وقيمة S لدوران
!>         جيفنز المتعلق بالفضاء الصفري الأيمن إذا كان SQRE = 1.
!>

WORK

!>          WORK هي مصفوفة DOUBLE PRECISION، البعد ( K )
!>

INFO

!>          INFO هو INTEGER
!>          = 0:  انتهاء ناجح.
!>          < 0:  إذا كان INFO = -i، كان الوسيط i غير قانوني.
!>
