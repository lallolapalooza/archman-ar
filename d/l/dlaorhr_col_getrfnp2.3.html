recursive subroutine dlaorhr_col_getrfnp2 (integer m, integer n, double precision, dimension( lda, * ) a, integer lda, double precision, dimension( * ) d, integer info)

DLAORHR_COL_GETRFNP2

الغرض:

!>
!> DLAORHR_COL_GETRFNP2 تحسب تحليل LU المعدل بدون
!> تبديل لمصفوفة حقيقية عامة M-by-N تسمى A. التحليل له الشكل:
!>
!>     A - S = L * U,
!>
!> حيث:
!>    S هي مصفوفة إشارة قطرية m-by-n مع القطر D، بحيث
!>    D(i) = S(i,i), 1 <= i <= min(M,N). يتم بناء القطر D
!>    على أنه D(i)=-SIGN(A(i,i))، حيث A(i,i) هي القيمة بعد إجراء
!>    خطوات الاستبعاد الغاوسي i-1. هذا يعني أن العنصر القطري
!>    في كل خطوة من خطوات الاستبعاد الغاوسي يكون
!>    على الأقل واحد بالقيمة المطلقة (بحيث لا يكون القسمة على صفر
!>    ممكنة أثناء القسمة على العنصر القطري)؛
!>
!>    L هي مصفوفة مثلثة سفلية M-by-N بعناصر قطرية وحدات (شبه منحرف سفلي إذا كان M > N)؛
!>
!>    و U هي مصفوفة مثلثة علوية M-by-N
!>    (شبه منحرف علوي إذا كان M < N).
!>
!> هذا الإجراء الفرعي هو إجراء فرعي مساعد يستخدم في إجراء إعادة بناء هاوسهولدر
!> DORHR_COL. في DORHR_COL، يتم تطبيق هذا الإجراء الفرعي على مصفوفة
!> A بالمصفوفة M-by-N ذات الأعمدة المتعامدة، حيث يتم تحديد كل
!> عنصر بواحد بالقيمة المطلقة. مع اختيار
!> المصفوفة S أعلاه، يمكن إظهار أن العنصر القطري في كل
!> خطوة من خطوات الاستبعاد الغاوسي هو الأكبر (بالقيمة المطلقة) في
!> العمود على القطر أو أسفله، بحيث لا يلزم التبديل
!> للاستقرار العددي [1].
!>
!> لمزيد من التفاصيل حول خوارزمية إعادة بناء هاوسهولدر،
!> بما في ذلك تحليل LU المعدل، انظر [1].
!>
!> هذا هو الإصدار التكراري لخوارزمية تحليل LU.
!> لنطلق على A - S اسم B. تقسم الخوارزمية المصفوفة B إلى أربع
!> مصفوفات فرعية:
!>
!>        [  B11 | B12  ]  حيث B11 هي n1 × n1،
!>    B = [ -----|----- ]        B21 هي (m-n1) × n1،
!>        [  B21 | B22  ]        B12 هي n1 × n2،
!>                               B22 هي (m-n1) × n2،
!>                               مع n1 = min(m,n)/2, n2 = n-n1.
!>
!>
!> يستدعي الإجراء الفرعي نفسه لتحليل B11، ويحل لـ B21،
!> ويحل لـ B12، ويحدث B22، ثم يستدعي نفسه لتحليل B22.
!>
!> لمزيد من التفاصيل حول خوارزمية LU التكرارية، انظر [2].
!>
!> يتم استدعاء DLAORHR_COL_GETRFNP2 لتحليل كتلة بواسطة
!> الإجراء المكتلي DLAORHR_COL_GETRFNP، الذي يستخدم رمزًا مكتليًا يستدعي
!> BLAS المستوى 3 لتحديث المصفوفة الفرعية. ومع ذلك، فإن DLAORHR_COL_GETRFNP2
!> مكتفٍ ذاتيًا ويمكن استخدامه بدون DLAORHR_COL_GETRFNP.
!>
!> [1] ,
!>     G. Ballard, J. Demmel, L. Grigori, M. Jacquelin, H.D. Nguyen,
!>     E. Solomonik, J. Parallel Distrib. Comput.,
!>     vol. 85, pp. 3-31, 2015.
!>
!> [2] , F. Gustavson, IBM J. of Res. and Dev.,
!>     vol. 41, no. 6, pp. 737-755, 1997.
!>

المعلمات

M 
!>          M is INTEGER
!>          عدد صفوف المصفوفة A. M >= 0.
!>

N

!>          N is INTEGER
!>          عدد أعمدة المصفوفة A. N >= 0.
!>

A

!>          A is DOUBLE PRECISION array, dimension (LDA,N)
!>          عند الدخول، المصفوفة M-by-N المراد تحليلها.
!>          عند الخروج، العوامل L و U من التحليل
!>          A-S=L*U؛ لا يتم تخزين عناصر القطر الوحدات لـ L.
!>

LDA

!>          LDA is INTEGER
!>          البعد الرائد للمصفوفة A. LDA >= max(1,M).
!>

D

!>          D is DOUBLE PRECISION array, dimension min(M,N)
!>          العناصر القطرية لمصفوفة الإشارة القطرية S ذات الحجم M-by-N،
!>          D(i) = S(i,i)، حيث 1 <= i <= min(M,N). يمكن أن تكون العناصر
!>          فقط زائد أو ناقص واحد.
!>

INFO

!>          INFO is INTEGER
!>          = 0:  انتهاء ناجح
!>          < 0:  إذا كان INFO = -i، فإن الوسيط i كان له قيمة غير قانونية
!>
