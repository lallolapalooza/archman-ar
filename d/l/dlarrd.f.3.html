subroutine dlarrd (character range, character order, integer n, double precision vl, double precision vu, integer il, integer iu, double precision, dimension( * ) gers, double precision reltol, double precision, dimension( * ) d, double precision, dimension( * ) e, double precision, dimension( * ) e2, double precision pivmin, integer nsplit, integer, dimension( * ) isplit, integer m, double precision, dimension( * ) w, double precision, dimension( * ) werr, double precision wl, double precision wu, integer, dimension( * ) iblock, integer, dimension( * ) indexw, double precision, dimension( * ) work, integer, dimension( * ) iwork, integer info)

DLARRD يحسب القيم الذاتية لمصفوفة ثلاثية متناظرة بدقة مناسبة.

الغرض:

!>
!> DLARRD يحسب القيم الذاتية لمصفوفة ثلاثية متناظرة T بدقة مناسبة. هذا هو رمز مساعد يتم استدعاؤه من DSTEMR.
!> قد يطلب المستخدم جميع القيم الذاتية، أو جميع القيم الذاتية
!> في الفترة نصف المفتوحة (VL, VU]، أو القيم الذاتية من IL-th إلى IU-th.
!>
!> لتجنب الفيضان، يجب قياس المصفوفة بحيث لا يتجاوز أكبر عنصر فيها
!> قيمة overflow**(1/2) * underflow**(1/4) بالقيمة المطلقة، وللحصول على أقصى دقة،
!> لا ينبغي أن تكون أصغر بكثير من ذلك.
!>
!> انظر W. Kahan، Report CS41، قسم علوم الحاسوب، جامعة ستانفورد،
!> 21 يوليو 1966.
!>

المعلمات

RANGE 
!>          RANGE is CHARACTER*1
!>          = 'A': ()   سيتم العثور على جميع القيم الذاتية.
!>          = 'V': () سيتم العثور على جميع القيم الذاتية في الفترة نصف المفتوحة
!>                           (VL, VU].
!>          = 'I': () سيتم العثور على القيم الذاتية من IL-th إلى IU-th (للمصفوفة
!>                           بأكملها).
!>

ORDER

!>          ORDER is CHARACTER*1
!>          = 'B': () سيتم تجميع القيم الذاتية حسب
!>                              كتلة الفصل (انظر IBLOCK، ISPLIT) و
!>                              ترتيبها من الأصغر إلى الأكبر داخل
!>                              الكتلة.
!>          = 'E': ()
!>                              سيتم ترتيب القيم الذاتية للمصفوفة بأكملها
!>                              من الأصغر إلى
!>                              الأكبر.
!>

N

!>          N is INTEGER
!>          رتبة المصفوفة الثلاثية T. N >= 0.
!>

VL

!>          VL is DOUBLE PRECISION
!>          إذا كانت RANGE='V'، الحد الأدنى للفترة التي
!>          سيتم البحث فيها عن القيم الذاتية. القيم الذاتية الأقل من أو تساوي
!>          VL، أو الأكبر من VU، لن يتم إرجاعها. VL < VU.
!>          غير مشار إليها إذا كانت RANGE = 'A' أو 'I'.
!>

VU

!>          VU is DOUBLE PRECISION
!>          إذا كانت RANGE='V'، الحد الأعلى للفترة التي
!>          سيتم البحث فيها عن القيم الذاتية. القيم الذاتية الأقل من أو تساوي
!>          VL، أو الأكبر من VU، لن يتم إرجاعها. VL < VU.
!>          غير مشار إليها إذا كانت RANGE = 'A' أو 'I'.
!>

IL

!>          IL is INTEGER
!>          إذا كانت RANGE='I'، فهرس
!>          أصغر قيمة ذاتية سيتم إرجاعها.
!>          1 <= IL <= IU <= N، إذا كانت N > 0؛ IL = 1 و IU = 0 إذا كانت N = 0.
!>          غير مشار إليها إذا كانت RANGE = 'A' أو 'V'.
!>

IU

!>          IU is INTEGER
!>          إذا كانت RANGE='I'، فهرس
!>          أكبر قيمة ذاتية سيتم إرجاعها.
!>          1 <= IL <= IU <= N، إذا كانت N > 0؛ IL = 1 و IU = 0 إذا كانت N = 0.
!>          غير مشار إليها إذا كانت RANGE = 'A' أو 'V'.
!>

GERS

!>          GERS is DOUBLE PRECISION array, dimension (2*N)
!>          فترات Gerschgorin الـ N (فترة Gerschgorin الـ i هي
!>          (GERS(2*i-1), GERS(2*i)).
!>

RELTOL

!>          RELTOL is DOUBLE PRECISION
!>          الحد الأدنى للعرض النسبي للفترة. عندما تكون الفترة
!>          أضيق من RELTOL مضروبة في النهاية الأكبر (بالقيمة المطلقة)
!>          للفترة، تعتبر أنها
!>          صغيرة بما فيه الكفاية، أي، متقاربة. ملاحظة: يجب أن يكون هذا
!>          دائمًا على الأقل radix*machine epsilon.
!>

D

!>          D is DOUBLE PRECISION array, dimension (N)
!>          العناصر القطرية الـ N للمصفوفة الثلاثية T.
!>

E

!>          E is DOUBLE PRECISION array, dimension (N-1)
!>          العناصر خارج القطر الـ (N-1) للمصفوفة الثلاثية T.
!>

E2

!>          E2 is DOUBLE PRECISION array, dimension (N-1)
!>          مربعات العناصر خارج القطر الـ (N-1) للمصفوفة الثلاثية T.
!>

PIVMIN

!>          PIVMIN is DOUBLE PRECISION
!>          الحد الأدنى للمحور المسموح به في متتالية ستورم لـ T.
!>

NSPLIT

!>          NSPLIT is INTEGER
!>          عدد الكتل القطرية في المصفوفة T.
!>          1 <= NSPLIT <= N.
!>

ISPLIT

!>          ISPLIT is INTEGER array, dimension (N)
!>          نقاط التقسيم، التي تنقسم عندها T إلى مصفوفات فرعية.
!>          تتكون المصفوفة الفرعية الأولى من الصفوف/الأعمدة من 1 إلى ISPLIT(1)،
!>          الثانية من الصفوف/الأعمدة ISPLIT(1)+1 حتى ISPLIT(2)،
!>          وهكذا، والكتلة NSPLIT-th تتكون من الصفوف/الأعمدة
!>          ISPLIT(NSPLIT-1)+1 حتى ISPLIT(NSPLIT)=N.
!>          (سيتم استخدام أول NSPLIT عنصر فقط فعليًا، ولكن
!>          بما أن المستخدم لا يمكنه معرفة قيمة NSPLIT مسبقًا،
!>          يجب حجز N كلمة لـ ISPLIT.)
!>

M

!>          M is INTEGER
!>          العدد الفعلي للقيم الذاتية التي تم العثور عليها. 0 <= M <= N.
!>          (انظر أيضًا وصف INFO=2، 3.)
!>

W

!>          W is DOUBLE PRECISION array, dimension (N)
!>          عند الخروج، ستحتوي العناصر M الأولى في W على
!>          تقديرات القيم الذاتية. DLARRD يحسب فترة
!>          I_j = (a_j, b_j] التي تشمل القيمة الذاتية j. تقدير القيمة الذاتية
!>          يُعطى كمنتصف الفترة
!>          W(j)= ( a_j + b_j)/2. الخطأ المقابل مقيد بـ
!>          WERR(j) = abs( a_j - b_j)/2
!>

WERR

!>          WERR is DOUBLE PRECISION array, dimension (N)
!>          حد الخطأ على تقدير القيمة الذاتية المقابل
!>          في W.
!>

WL

!>          WL is DOUBLE PRECISION
!>

WU

!>          WU is DOUBLE PRECISION
!>          الفترة (WL, WU] تحتوي على جميع القيم الذاتية المطلوبة.
!>          إذا كانت RANGE='V'، فإن WL=VL و WU=VU.
!>          إذا كانت RANGE='A'، فإن WL و WU هما حدود Gerschgorin العالمية
!>                        على الطيف.
!>          إذا كانت RANGE='I'، يتم حساب WL و WU بواسطة DLAEBZ من
!>                        الفهرس
