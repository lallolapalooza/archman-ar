subroutine dlaqge (m, n, a, lda, r, c, rowcnd, colcnd, amax, equed)
DLAQGE تقوم بقياس مصفوفة مستطيلة عامة، باستخدام عوامل قياس الصفوف والأعمدة التي تم حسابها بواسطة sgeequ.

توثيق الوظيفة/الروتين الفرعي

subroutine dlaqge (integer m, integer n, double precision, dimension( lda, * ) a, integer lda, double precision, dimension( * ) r, double precision, dimension( * ) c, double precision rowcnd, double precision colcnd, double precision amax, character equed)

DLAQGE تقوم بقياس مصفوفة مستطيلة عامة، باستخدام عوامل قياس الصفوف والأعمدة التي تم حسابها بواسطة sgeequ.

الغرض:

!>
!> DLAQGE تقوم بموازنة مصفوفة عامة M في N باستخدام عوامل القياس للصفوف والأعمدة في المتجهات R و C.
!>

المعلمات

M 
!>          M هو عدد صحيح (INTEGER)
!>          عدد صفوف المصفوفة A.  M >= 0.
!>

N

!>          N هو عدد صحيح (INTEGER)
!>          عدد أعمدة المصفوفة A.  N >= 0.
!>

A

!>          A هي مصفوفة من النوع DOUBLE PRECISION، بأبعاد (LDA,N)
!>          عند الإدخال، المصفوفة A بحجم M في N.
!>          عند الخروج، المصفوفة المتوازنة. انظر EQUED لشكل المصفوفة المتوازنة.
!>

LDA

!>          LDA هو عدد صحيح (INTEGER)
!>          البعد الرئيسي للمصفوفة A.  LDA >= max(M,1).
!>

R

!>          R هي مصفوفة من النوع DOUBLE PRECISION، بأبعاد (M)
!>          عوامل قياس الصفوف لـ A.
!>

C

!>          C هي مصفوفة من النوع DOUBLE PRECISION، بأبعاد (N)
!>          عوامل قياس الأعمدة لـ A.
!>

ROWCND

!>          ROWCND هو DOUBLE PRECISION
!>          نسبة أصغر R(i) إلى أكبر R(i).
!>

COLCND

!>          COLCND هو DOUBLE PRECISION
!>          نسبة أصغر C(i) إلى أكبر C(i).
!>

AMAX

!>          AMAX هو DOUBLE PRECISION
!>          القيمة المطلقة لأكبر عنصر في المصفوفة.
!>

EQUED

!>          EQUED هو CHARACTER*1
!>          يحدد شكل الموازنة التي تم إجراؤها.
!>          = 'N':  لا توجد موازنة
!>          = 'R':  موازنة الصفوف، أي تم ضرب A مسبقًا بـ
!>                  diag(R).
!>          = 'C':  موازنة الأعمدة، أي تم استبدال A بضربها لاحقًا بـ
!>                  diag(C).
!>          = 'B':  موازنة الصفوف والأعمدة معًا، أي تم استبدال A بـ
!>                  diag(R) * A * diag(C).
!>

المعلمات الداخلية:

!>  THRESH هو قيمة حدية تستخدم لتحديد ما إذا كان ينبغي إجراء قياس الصفوف أو الأعمدة
!>  بناءً على نسبة عوامل قياس الصفوف أو الأعمدة. إذا كان ROWCND < THRESH، يتم إجراء قياس الصفوف، وإذا
!>  كان COLCND < THRESH، يتم إجراء قياس الأعمدة.
!>
!>  LARGE و SMALL هي قيم حدية تستخدم لتحديد ما إذا كان ينبغي إجراء قياس الصفوف
!>  بناءً على الحجم المطلق لأكبر عنصر في المصفوفة. إذا كان AMAX > LARGE أو AMAX < SMALL، يتم إجراء قياس الصفوف.
!>
