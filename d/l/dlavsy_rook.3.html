subroutine dlavsy_rook (character uplo, character trans, character diag, integer n, integer nrhs, double precision, dimension( lda, * ) a, integer lda, integer, dimension( * ) ipiv, double precision, dimension( ldb, * ) b, integer ldb, integer info)

DLAVSY_ROOK

الغرض:

!>
!> DLAVSY_ROOK  يجري إحدى عمليات المصفوفة والمتجه
!>    x := A*x  أو  x := A'*x,
!> حيث x متجه يتكون من N عنصر، و A هو أحد العوامل
!> من التحليل الجزئي U*D*U' أو L*D*L' المحسوب بواسطة DSYTRF_ROOK.
!>
!> إذا كان TRANS = 'N'، يضرب بـ U  أو U * D  (أو L  أو L * D)
!> إذا كان TRANS = 'T'، يضرب بـ U' أو D * U' (أو L' أو D * L')
!> إذا كان TRANS = 'C'، يضرب بـ U' أو D * U' (أو L' أو D * L')
!>

المعلمات

UPLO 
!>          UPLO هو CHARACTER*1
!>          يحدد ما إذا كان العامل المخزن في A علويًا أم سفليًا
!>          مثلثيًا.
!>          = 'U':  مثلثي علوي
!>          = 'L':  مثلثي سفلي
!>

TRANS

!>          TRANS هو CHARACTER*1
!>          يحدد العملية المراد تنفيذها:
!>          = 'N':  x := A*x
!>          = 'T':  x := A'*x
!>          = 'C':  x := A'*x
!>

DIAG

!>          DIAG هو CHARACTER*1
!>          يحدد ما إذا كانت الكتل القطرية وحدات
!>          أم لا.  إذا كان من المفترض أن تكون الكتل القطرية وحدات،
!>          فإن A = U أو A = L، وإلا فإن A = U*D أو A = L*D.
!>          = 'U':  من المفترض أن تكون الكتل القطرية وحدات.
!>          = 'N':  من المفترض أن تكون الكتل القطرية غير وحدات.
!>

N

!>          N هو INTEGER
!>          عدد الصفوف والأعمدة للمصفوفة A.  N >= 0.
!>

NRHS

!>          NRHS هو INTEGER
!>          عدد الأطراف اليمنى، أي عدد المتجهات
!>          x المراد ضربها بـ A.  NRHS >= 0.
!>

A

!>          A هي مصفوفة DOUBLE PRECISION، أبعادها (LDA,N)
!>          المصفوفة القطرية الكتلوية D والمعاملات المستخدمة
!>          للحصول على العامل U أو L كما تم حسابه بواسطة DSYTRF_ROOK.
!>          مخزنة كمصفوفة ثلاثية ثنائية الأبعاد.
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرائد للمصفوفة A.  LDA >= max(1,N).
!>

IPIV

!>          IPIV هي مصفوفة INTEGER، أبعادها (N)
!>          تفاصيل التبادلات والهيكل الكتلي لـ D،
!>          كما تم تحديده بواسطة DSYTRF_ROOK.
!>
!>          إذا كان UPLO = 'U':
!>               إذا كان IPIV(k) > 0، فقد تم تبديل الصفوف والأعمدة k و IPIV(k)
!>               و D(k,k) هي كتلة قطرية بحجم 1x1.
!>               (إذا كان IPIV( k ) = k، لم يتم إجراء أي تبديل).
!>
!>               إذا كان IPIV(k) < 0 و IPIV(k-1) < 0، فقد تم تبديل الصفوف
!>               والأعمدة k و -IPIV(k) وتم تبديل الصفوف
!>               والأعمدة k-1 و -IPIV(k-1)،
!>               D(k-1:k,k-1:k) هي كتلة قطرية بحجم 2x2.
!>
!>          إذا كان UPLO = 'L':
!>               إذا كان IPIV(k) > 0، فقد تم تبديل الصفوف والأعمدة k و IPIV(k)
!>               و D(k,k) هي كتلة قطرية بحجم 1x1.
!>               (إذا كان IPIV( k ) = k، لم يتم إجراء أي تبديل).
!>
!>               إذا كان IPIV(k) < 0 و IPIV(k+1) < 0، فقد تم تبديل الصفوف
!>               والأعمدة k و -IPIV(k) وتم تبديل الصفوف
!>               والأعمدة k+1 و -IPIV(k+1)،
!>               D(k:k+1,k:k+1) هي كتلة قطرية بحجم 2x2.
!>

B

!>          B هي مصفوفة DOUBLE PRECISION، أبعادها (LDB,NRHS)
!>          عند الدخول، تحتوي B على NRHS متجهات بطول N.
!>          عند الخروج، يتم استبدال B بمنتج A * B.
!>

LDB

!>          LDB هو INTEGER
!>          البعد الرائد للمصفوفة B.  LDB >= max(1,N).
!>

INFO

!>          INFO هو INTEGER
!>          = 0: خروج ناجح
!>          < 0: إذا كان INFO = -k، فإن المعلمة k كانت لها قيمة غير قانونية
!>
