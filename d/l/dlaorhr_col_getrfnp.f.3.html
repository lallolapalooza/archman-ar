subroutine dlaorhr_col_getrfnp (integer m, integer n, double precision, dimension( lda, * ) a, integer lda, double precision, dimension( * ) d, integer info)

DLAORHR_COL_GETRFNP

الغرض:

!>
!> DLAORHR_COL_GETRFNP يحسب تفكيك LU المعدل بدون
!> تبديل لمصفوفة حقيقية عامة M-by-N وهي A. التفكيك له الشكل:
!>
!>     A - S = L * U,
!>
!> حيث:
!>    S هي مصفوفة إشارة قطرية m-by-n مع القطر D، بحيث
!>    D(i) = S(i,i)، 1 <= i <= min(M,N). يتم بناء القطر D
!>    كـ D(i)=-SIGN(A(i,i))، حيث A(i,i) هي القيمة بعد إجراء
!>    i-1 خطوة من الحذف الغاوسي. هذا يعني أن العنصر القطري
!>    في كل خطوة من خطوات الحذف الغاوسي
!>    هو على الأقل واحد في القيمة المطلقة (بحيث لا يكون القسمة على صفر
!>    ممكنة أثناء القسمة على العنصر القطري)؛
!>
!>    L هي مصفوفة مثلثة سفلية M-by-N بعناصر قطرية وحدات
!>    (شبه منحرف سفلي إذا كان M > N)؛
!>
!>    و U هي مصفوفة مثلثة علوية M-by-N
!>    (شبه منحرف علوي إذا كان M < N).
!>
!> هذا الروتين هو روتين مساعد يستخدم في روتين إعادة بناء هاوسهولدر
!> DORHR_COL. في DORHR_COL، يتم تطبيق هذا الروتين
!> على مصفوفة A من حجم M-by-N بأعمدة متعامدة، حيث يتم
!> تقييد كل عنصر بواحد في القيمة المطلقة. مع اختيار
!> المصفوفة S أعلاه، يمكن إظهار أن العنصر القطري في كل
!> خطوة من خطوات الحذف الغاوسي
!> هو الأكبر (في القيمة المطلقة) في العمود على أو تحت القطر،
!> بحيث لا يلزم تبديل من أجل الاستقرار العددي [1].
!>
!> لمزيد من التفاصيل حول خوارزمية إعادة بناء هاوسهولدر،
!> بما في ذلك تفكيك LU المعدل، انظر [1].
!>
!> هذه هي النسخة المكدسة التي تنظر إلى اليمين من الخوارزمية،
!> تستدعي BLAS المستوى 3 لتحديث المصفوفة الفرعية. لتفكيك كتلة،
!> يستدعي هذا الروتين الروتين التكراري DLAORHR_COL_GETRFNP2.
!>
!> [1] ,
!>     G. Ballard, J. Demmel, L. Grigori, M. Jacquelin, H.D. Nguyen,
!>     E. Solomonik, J. Parallel Distrib. Comput.,
!>     vol. 85, pp. 3-31, 2015.
!>

المعلمات

M 
!>          M هو INTEGER
!>          عدد صفوف المصفوفة A. M >= 0.
!>

N

!>          N هو INTEGER
!>          عدد أعمدة المصفوفة A. N >= 0.
!>

A

!>          A هي مصفوفة DOUBLE PRECISION، ذات بعد (LDA,N)
!>          عند الإدخال، المصفوفة M-by-N المراد تفكيكها.
!>          عند الخروج، العوامل L و U من التفكيك
!>          A-S=L*U؛ لا يتم تخزين عناصر الوحدة القطرية لـ L.
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرائد للمصفوفة A. LDA >= max(1,M).
!>

D

!>          D هي مصفوفة DOUBLE PRECISION، ذات بعد min(M,N)
!>          العناصر القطرية لمصفوفة الإشارة القطرية S بحجم M-by-N،
!>          D(i) = S(i,i)، حيث 1 <= i <= min(M,N). يمكن أن تكون العناصر
!>          فقط زائد أو ناقص واحد.
!>

INFO

!>          INFO هو INTEGER
!>          = 0:  انتهاء ناجح
!>          < 0:  إذا كان INFO = -i، فإن الوسيط i كان له قيمة غير قانونية
!>
