subroutine dlagtf (n, a, lambda, b, c, tol, d, in, info)
DLAGTF تحسب تحليل LU للمصفوفة T-λI، حيث T مصفوفة ثلاثية عامة، و λ قيمة قياسية، باستخدام التقسيم الجزئي مع تبديلات الصفوف.

وثائق الوظيفة/الوحدة الفرعية

subroutine dlagtf (integer n, double precision, dimension( * ) a, double precision lambda, double precision, dimension( * ) b, double precision, dimension( * ) c, double precision tol, double precision, dimension( * ) d, integer, dimension( * ) in, integer info)

DLAGTF تحسب تحليل LU للمصفوفة T-λI، حيث T مصفوفة ثلاثية عامة، و λ قيمة قياسية، باستخدام التقسيم الجزئي مع تبديلات الصفوف.

الغرض:

!>
!> DLAGTF تحلل المصفوفة (T - lambda*I)، حيث T مصفوفة ثلاثية n×n
!> و lambda قيمة قياسية، إلى:
!>
!>    T - lambda*I = PLU,
!>
!> حيث P مصفوفة تبديل، L مصفوفة ثلاثية سفلية وحدوية مع على الأكثر عنصر واحد غير صفري تحت القطر الرئيسي لكل عمود، و U مصفوفة مثلثة علوية مع على الأكثر عنصرين غير صفريين فوق القطر الرئيسي لكل عمود.
!>
!> يتم الحصول على التحليل عن طريق الحذف الغاوسي مع التقسيم الجزئي
!> والقياس الضمني للصفوف.
!>
!> يتم تضمين المعلمة LAMBDA في الروتين حتى يمكن استخدام DLAGTF، بالاشتراك مع DLAGTS، للحصول على المتجهات الذاتية لـ T عن طريق التكرار العكسي.
!>

المعلمات

N 
!>          N هو عدد صحيح (INTEGER).
!>          رتبة المصفوفة T.
!>

A

!>          A هو مصفوفة دقة مزدوجة (DOUBLE PRECISION)، البعد (N).
!>          عند الإدخال، يجب أن تحتوي A على العناصر القطرية لـ T.
!>
!>          عند الخروج، يتم الكتابة فوق A بالعناصر القطرية n للمصفوفة المثلثية العلوية U لتحليل T.
!>

LAMBDA

!>          LAMBDA هو دقة مزدوجة (DOUBLE PRECISION).
!>          عند الإدخال، القيمة القياسية lambda.
!>

B

!>          B هو مصفوفة دقة مزدوجة (DOUBLE PRECISION)، البعد (N-1).
!>          عند الإدخال، يجب أن تحتوي B على عناصر فوق القطر الرئيسي (n-1) لـ T.
!>
!>          عند الخروج، يتم الكتابة فوق B بعناصر فوق القطر الرئيسي (n-1) للمصفوفة L لتحليل T.
!>

C

!>          C هو مصفوفة دقة مزدوجة (DOUBLE PRECISION)، البعد (N-1).
!>          عند الإدخال، يجب أن تحتوي C على عناصر تحت القطر الرئيسي (n-1) لـ T.
!>
!>          عند الخروج، يتم الكتابة فوق C بعناصر تحت القطر الرئيسي (n-1) للمصفوفة L لتحليل T.
!>

TOL

!>          TOL هو دقة مزدوجة (DOUBLE PRECISION).
!>          عند الإدخال، قيمة تحمل نسبية تستخدم للإشارة إلى ما إذا كانت المصفوفة (T - lambda*I) قريبة من أن تكون منفردة أم لا. يجب اختيار TOL عادةً كخطأ نسبي تقريبي في عناصر T. على سبيل المثال، إذا كانت عناصر T صحيحة تقريبًا إلى 4 أرقام معنوية، فيجب تعيين TOL إلى حوالي 5*10**(-4). إذا تم توفير TOL أقل من eps، حيث eps هي الدقة النسبية للآلة، فسيتم استخدام القيمة eps بدلاً من TOL.
!>

D

!>          D هو مصفوفة دقة مزدوجة (DOUBLE PRECISION)، البعد (N-2).
!>          عند الخروج، يتم الكتابة فوق D بالعناصر الثانية فوق القطر الرئيسي (n-2) للمصفوفة U لتحليل T.
!>

IN

!>          IN هو مصفوفة أعداد صحيحة (INTEGER)، البعد (N).
!>          عند الخروج، تحتوي IN على تفاصيل مصفوفة التبديل P. إذا حدث تبديل في الخطوة k من الحذف، فإن IN(k) = 1، وإلا فإن IN(k) = 0. العنصر IN(n) يعيد أصغر عدد صحيح موجب j بحيث
!>
!>             abs( u(j,j) ) <= norm( (T - lambda*I)(j) )*TOL،
!>
!>          حيث تشير norm( A(j) ) إلى مجموع القيم المطلقة للصف j من المصفوفة A. إذا لم يكن هناك مثل هذا j، فسيتم إرجاع IN(n) على أنه صفر. إذا تم إرجاع IN(n) بقيمة موجبة، فإن عنصرًا قطريًا في U صغير، مما يشير إلى أن (T - lambda*I) منفردة أو قريبة من أن تكون منفردة.
!>

INFO

!>          INFO هو عدد صحيح (INTEGER).
!>          = 0:  اكتمال ناجح
!>          < 0:  إذا كان INFO = -k، فإن المعلمة k كانت لها قيمة غير قانونية.
!>
