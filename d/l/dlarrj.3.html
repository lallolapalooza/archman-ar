وثائق الوظائف/الإجراءات الفرعية

subroutine dlarrj (integer n, double precision, dimension( * ) d, double precision, dimension( * ) e2, integer ifirst, integer ilast, double precision rtol, integer offset, double precision, dimension( * ) w, double precision, dimension( * ) werr, double precision, dimension( * ) work, integer, dimension( * ) iwork, double precision pivmin, double precision spdiam, integer info)

DLARRJ تنفذ تحسين التقديرات الأولية للقيم الذاتية للمصفوفة T.

الغرض:

!>
!> بالنظر إلى التقديرات الأولية للقيم الذاتية لـ T، تقوم DLARRJ
!> بإجراء التنصيف لتحسين القيم الذاتية لـ T،
!> W( IFIRST-OFFSET ) خلال W( ILAST-OFFSET )، إلى دقة أعلى.
!> يتم إدخال التخمينات الأولية لهذه القيم الذاتية في W،
!> وتقدير الخطأ المقابل لهذه التخمينات في WERR.
!> أثناء التنصيف، يتم الاحتفاظ بالفترات [left, right]
!> عن طريق تخزين نقاطها الوسطى وأنصاف أقطارها
!> في المصفوفات W و WERR على التوالي.
!>

المعلمات

N 
!>          N هو عدد صحيح (INTEGER)
!>          رتبة المصفوفة.
!>

D

!>          D هي مصفوفة دقة مزدوجة (DOUBLE PRECISION)، البعد (N)
!>          العناصر القطرية N للمصفوفة T.
!>

E2

!>          E2 هي مصفوفة دقة مزدوجة (DOUBLE PRECISION)، البعد (N-1)
!>          مربعات العناصر شبه القطرية (N-1) للمصفوفة T.
!>

IFIRST

!>          IFIRST هو عدد صحيح (INTEGER)
!>          فهرس أول قيمة ذاتية سيتم حسابها.
!>

ILAST

!>          ILAST هو عدد صحيح (INTEGER)
!>          فهرس آخر قيمة ذاتية سيتم حسابها.
!>

RTOL

!>          RTOL هو دقة مزدوجة (DOUBLE PRECISION)
!>          التفاوت للتقارب لفترات التنصيف.
!>          فترة [LEFT,RIGHT] تعتبر متقاربة إذا كان
!>          RIGHT-LEFT < RTOL*MAX(|LEFT|,|RIGHT|).
!>

OFFSET

!>          OFFSET هو عدد صحيح (INTEGER)
!>          الإزاحة للمصفوفات W و WERR، أي العناصر من IFIRST-OFFSET
!>          إلى ILAST-OFFSET في هاتين المصفوفتين سيتم استخدامها.
!>

W

!>          W هي مصفوفة دقة مزدوجة (DOUBLE PRECISION)، البعد (N)
!>          عند الإدخال، W( IFIRST-OFFSET ) خلال W( ILAST-OFFSET ) هي
!>          تقديرات للقيم الذاتية لـ L D L^T المفهرسة من IFIRST إلى
!>          ILAST.
!>          عند الإخراج، يتم تحسين هذه التقديرات.
!>

WERR

!>          WERR هي مصفوفة دقة مزدوجة (DOUBLE PRECISION)، البعد (N)
!>          عند الإدخال، WERR( IFIRST-OFFSET ) خلال WERR( ILAST-OFFSET ) هي
!>          الأخطاء في تقديرات العناصر المقابلة في W.
!>          عند الإخراج، يتم تحسين هذه الأخطاء.
!>

WORK

!>          WORK هي مصفوفة دقة مزدوجة (DOUBLE PRECISION)، البعد (2*N)
!>          مساحة عمل.
!>

IWORK

!>          IWORK هي مصفوفة عدد صحيح (INTEGER)، البعد (2*N)
!>          مساحة عمل.
!>

PIVMIN

!>          PIVMIN هو دقة مزدوجة (DOUBLE PRECISION)
!>          الارتكاز الأدنى في تسلسل ستورم لـ T.
!>

SPDIAM

!>          SPDIAM هو دقة مزدوجة (DOUBLE PRECISION)
!>          القطر الطيفي لـ T.
!>

INFO

!>          INFO هو عدد صحيح (INTEGER)
!>          علامة الخطأ.
!>
