وثائق الوظيفة/الإجراء الفرعي

subroutine dla_wwaddw (integer n, double precision, dimension( * ) x, double precision, dimension( * ) y, double precision, dimension( * ) w)

DLA_WWADDW تضيف متجهًا إلى متجه مزدوج المفرد.

الغرض:

!>
!>    DLA_WWADDW تضيف المتجه W إلى المتجه المزدوج المفرد (X, Y).
!>
!>    هذا يعمل لجميع الحسابات العشرية ذات النقطة العائمة الثنائية والسداسية عشرية
!>    الحالية من IBM، ولكن ليس للعشري.
!>

المعلمات

N 
!>          N هو عدد صحيح
!>            طول المتجهات X، Y، و W.
!>

X

!>          X هو مصفوفة ذات دقة مزدوجة، البعد (N)
!>            الجزء الأول من المتجه المتراكم المزدوج المفرد.
!>

Y

!>          Y هو مصفوفة ذات دقة مزدوجة، البعد (N)
!>            الجزء الثاني من المتجه المتراكم المزدوج المفرد.
!>

W

!>          W هو مصفوفة ذات دقة مزدوجة، البعد (N)
!>            المتجه المراد إضافته.
!>
