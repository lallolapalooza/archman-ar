توثيق الوظيفة/الإجراء الفرعي

subroutine dlaqp2 (integer m, integer n, integer offset, double precision, dimension( lda, * ) a, integer lda, integer, dimension( * ) jpvt, double precision, dimension( * ) tau, double precision, dimension( * ) vn1, double precision, dimension( * ) vn2, double precision, dimension( * ) work)

DLAQP2 يحسب تحليل QR مع تبديل الأعمدة لكتلة المصفوفة.

الغرض:

!>
!> DLAQP2 يحسب تحليل QR مع تبديل الأعمدة لكتلة A(OFFSET+1:M,1:N).
!> الكتلة A(1:OFFSET,1:N) يتم تبديلها وفقًا لذلك، ولكن لا يتم تحليلها.
!>

المعلمات

M 
!>          M هو عدد صحيح
!>          عدد صفوف المصفوفة A. M >= 0.
!>

N

!>          N هو عدد صحيح
!>          عدد أعمدة المصفوفة A. N >= 0.
!>

OFFSET

!>          OFFSET هو عدد صحيح
!>          عدد صفوف المصفوفة A التي يجب تبديلها
!>          ولكن لا يتم تحليلها. OFFSET >= 0.
!>

A

!>          A هي مصفوفة ذات دقة مزدوجة، البعد (LDA,N)
!>          عند الإدخال، المصفوفة A ذات الأبعاد M×N.
!>          عند الخروج، الجزء العلوي المثلثي من الكتلة A(OFFSET+1:M,1:N) هو
!>          العامل المثلثي الناتج؛ العناصر في الكتلة
!>          A(OFFSET+1:M,1:N) تحت القطر، بالإضافة إلى المصفوفة TAU، تمثل
!>          المصفوفة المتعامدة Q كحاصل ضرب لمُبدِلات أولية. تم تبديل الكتلة
!>          A(1:OFFSET,1:N) وفقًا لذلك، ولكن لم يتم تحليلها.
!>

LDA

!>          LDA هو عدد صحيح
!>          البعد الرائد للمصفوفة A. LDA >= max(1,M).
!>

JPVT

!>          JPVT هي مصفوفة أعداد صحيحة، البعد (N)
!>          عند الإدخال، إذا كان JPVT(i) .ne. 0، يتم تبديل العمود i من A
!>          إلى مقدمة A*P (عمود رائد)؛ إذا كان JPVT(i) = 0،
!>          فإن العمود i من A هو عمود حر.
!>          عند الخروج، إذا كان JPVT(i) = k، فإن العمود i من A*P
!>          كان العمود k من A.
!>

TAU

!>          TAU هي مصفوفة ذات دقة مزدوجة، البعد (min(M,N))
!>          العوامل القياسية للمُبدِلات الأولية.
!>

VN1

!>          VN1 هي مصفوفة ذات دقة مزدوجة، البعد (N)
!>          المتجه الذي يحتوي على معايير الأعمدة الجزئية.
!>

VN2

!>          VN2 هي مصفوفة ذات دقة مزدوجة، البعد (N)
!>          المتجه الذي يحتوي على معايير الأعمدة الدقيقة.
!>

WORK

!>          WORK هي مصفوفة ذات دقة مزدوجة، البعد (N)
!>
