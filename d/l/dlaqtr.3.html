subroutine dlaqtr (ltran, lreal, n, t, ldt, b, w, scale, x, work, info)
DLAQTR يحل نظام معادلات شبه مثلثي حقيقي، أو نظام شبه مثلثي معقد ذو شكل خاص، بالحسابات الحقيقية.

توثيق الوظيفة/الروتين الفرعي

subroutine dlaqtr (logical ltran, logical lreal, integer n, double precision, dimension( ldt, * ) t, integer ldt, double precision, dimension( * ) b, double precision w, double precision scale, double precision, dimension( * ) x, double precision, dimension( * ) work, integer info)

DLAQTR يحل نظام شبه مثلثي حقيقي، أو نظام شبه مثلثي معقد ذو شكل خاص، بالحسابات الحقيقية.

الغرض:

!>
!> DLAQTR يحل النظام شبه المثلثي الحقيقي
!>
!>              op(T)*p = scale*c,               if LREAL = .TRUE.
!>
!> أو الأنظمة شبه المثلثية المعقدة
!>
!>            op(T + iB)*(p+iq) = scale*(c+id),  if LREAL = .FALSE.
!>
!> بالحسابات الحقيقية، حيث T شبه مثلثي علوي.
!> إذا كانت LREAL = .FALSE.، فإن أول كتلة قطرية لـ T يجب أن تكون
!> 1 × 1، و B هي المصفوفة ذات البنية الخاصة
!>
!>                B = [ b(1) b(2) ... b(n) ]
!>                    [       w            ]
!>                    [           w        ]
!>                    [              .     ]
!>                    [                 w  ]
!>
!> op(A) = A أو A**T، A**T يمثل منقول المصفوفة A.
!>
!> عند الإدخال، X = [ c ]. عند الإخراج، X = [ p ].
!>               [ d ]                  [ q ]
!>
!> هذا الروتين الفرعي مصمم لتقدير رقم الشرط في الروتين DTRSNA.
!>

المعلمات

LTRAN 
!>          LTRAN is LOGICAL
!>          عند الإدخال، تحدد LTRAN خيار المرافق التبادلي:
!>             = .FALSE.,    op(T+i*B) = T+i*B,
!>             = .TRUE.,     op(T+i*B) = (T+i*B)**T.
!>

LREAL

!>          LREAL is LOGICAL
!>          عند الإدخال، تحدد LREAL بنية مصفوفة الإدخال:
!>             = .FALSE.,    الإدخال معقد
!>             = .TRUE.,     الإدخال حقيقي
!>

N

!>          N is INTEGER
!>          عند الإدخال، تحدد N رتبة T+i*B. N >= 0.
!>

T

!>          T is DOUBLE PRECISION array, dimension (LDT,N)
!>          عند الإدخال، تحتوي T على مصفوفة في شكل كانونيكالي شوري.
!>          إذا كانت LREAL = .FALSE.، فيجب أن تكون الكتلة القطرية الأولى لـ T
!>          1 × 1.
!>

LDT

!>          LDT is INTEGER
!>          البعد الرائد للمصفوفة T. LDT >= max(1,N).
!>

B

!>          B is DOUBLE PRECISION array, dimension (N)
!>          عند الإدخال، تحتوي B على العناصر لتشكيل المصفوفة
!>          B كما هو موضح أعلاه.
!>          إذا كانت LREAL = .TRUE.، لا يتم الرجوع إلى B.
!>

W

!>          W is DOUBLE PRECISION
!>          عند الإدخال، W هو العنصر القطري للمصفوفة B.
!>          إذا كانت LREAL = .TRUE.، لا يتم الرجوع إلى W.
!>

SCALE

!>          SCALE is DOUBLE PRECISION
!>          عند الإخراج، SCALE هو عامل القياس.
!>

X

!>          X is DOUBLE PRECISION array, dimension (2*N)
!>          عند الإدخال، تحتوي X على الجانب الأيمن من النظام.
!>          عند الإخراج، يتم الكتابة فوق X بالحل.
!>

WORK

!>          WORK is DOUBLE PRECISION array, dimension (N)
!>

INFO

!>          INFO is INTEGER
!>          عند الإخراج، يتم تعيين INFO إلى
!>             0: انتهاء ناجح.
!>               1: تم تشويش بعض الكتل القطرية 1 × 1 برقم صغير SMIN
!>                  للحفاظ على عدم الانفراد.
!>               2: تم تشويش بعض الكتل القطرية 2 × 2 برقم صغير في DLALN2
!>                  للحفاظ على عدم الانفراد.
!>          ملاحظة: تحقيقًا للسرعة، لا يتحقق هذا الروتين من أخطاء المدخلات.
!>
