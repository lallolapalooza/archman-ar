subroutine dlaqr5 (logical wantt, logical wantz, integer kacc22, integer n, integer ktop, integer kbot, integer nshfts, double precision, dimension( * ) sr, double precision, dimension( * ) si, double precision, dimension( ldh, * ) h, integer ldh, integer iloz, integer ihiz, double precision, dimension( ldz, * ) z, integer ldz, double precision, dimension( ldv, * ) v, integer ldv, double precision, dimension( ldu, * ) u, integer ldu, integer nv, double precision, dimension( ldwv, * ) wv, integer ldwv, integer nh, double precision, dimension( ldwh, * ) wh, integer ldwh)

DLAQR5 تجري عملية مسح QR واحدة متعددة التحولات صغيرة الانتفاخ.

الغرض:

!>
!>    DLAQR5، الذي تم استدعاؤه بواسطة DLAQR0، يجري
!>    مسح QR واحد متعدد التحولات صغير الانتفاخ.
!>

المعلمات

WANTT 
!>          WANTT هو LOGICAL
!>             WANTT = .true. إذا كان عامل Schur شبه المستوي
!>             يجري حسابه. يتم تعيين WANTT إلى .false. خلاف ذلك.
!>

WANTZ

!>          WANTZ هو LOGICAL
!>             WANTZ = .true. إذا كان عامل Schur المتعامد
!>             يجري حسابه. يتم تعيين WANTZ إلى .false. خلاف ذلك.
!>

KACC22

!>          KACC22 هو INTEGER بقيمة 0، 1، أو 2.
!>             يحدد وضع حساب تحديثات المتعامد البعيدة عن القطر.
!>        = 0: DLAQR5 لا تجمع الانعكاسات ولا تستخدم ضرب المصفوفات
!>             لتحديث إدخالات المصفوفة البعيدة عن القطر.
!>        = 1: DLAQR5 تجمع الانعكاسات وتستخدم ضرب المصفوفات
!>             لتحديث إدخالات المصفوفة البعيدة عن القطر.
!>        = 2: نفس KACC22 = 1. كان هذا الخيار يستخدم للاستفادة
!>             من هيكل 2x2 أثناء عمليات ضرب المصفوفات، ولكن
!>             هذا لم يعد مدعومًا.
!>

N

!>          N هو INTEGER
!>             N هو رتبة مصفوفة Hessenberg H التي تعمل عليها هذه
!>             الروتين الفرعي.
!>

KTOP

!>          KTOP هو INTEGER
!>

KBOT

!>          KBOT هو INTEGER
!>             هذه هي الصفوف والأعمدة الأولى والأخيرة
!>             لمصفوفة قطرية معزولة يتم تطبيق مسح QR عليها.
!>             يُفترض دون تحقق أن
!>                       إما KTOP = 1  أو   H(KTOP,KTOP-1) = 0
!>             و
!>                       إما KBOT = N  أو   H(KBOT+1,KBOT) = 0.
!>

NSHFTS

!>          NSHFTS هو INTEGER
!>             NSHFTS يعطي عدد التحولات المتزامنة. يجب أن يكون NSHFTS
!>             موجبًا وزوجيًا.
!>

SR

!>          SR هو مصفوفة DOUBLE PRECISION، ذات البعد (NSHFTS)
!>

SI

!>          SI هو مصفوفة DOUBLE PRECISION، ذات البعد (NSHFTS)
!>             تحتوي SR على الأجزاء الحقيقية و SI على الأجزاء التخيلية
!>             لتحولات الأصل NSHFTS المتزامنة التي تحدد
!>             مسح QR المتعدد التحولات. قد يتم إعادة ترتيب SR و SI
!>             عند الخروج.
!>

H

!>          H هو مصفوفة DOUBLE PRECISION، ذات البعد (LDH,N)
!>             عند الإدخال، تحتوي H على مصفوفة Hessenberg. عند الخروج،
!>             يتم تطبيق مسح QR متعدد التحولات مع التحولات SR(J)+i*SI(J)
!>             على المصفوفة القطرية المعزولة في الصفوف والأعمدة من KTOP
!>             إلى KBOT.
!>

LDH

!>          LDH هو INTEGER
!>             LDH هو البعد الرائد لـ H كما تم تعريفه في
!>             الإجراء الاستدعاء. LDH >= MAX(1,N).
!>

ILOZ

!>          ILOZ هو INTEGER
!>

IHIZ

!>          IHIZ هو INTEGER
!>             حدد صفوف Z التي يجب تطبيق التحويلات عليها
!>             إذا كانت WANTZ = .TRUE.. 1 <= ILOZ <= IHIZ <= N
!>

Z

!>          Z هو مصفوفة DOUBLE PRECISION، ذات البعد (LDZ,IHIZ)
!>             إذا كانت WANTZ = .TRUE.، فسيتم تجميع تحويل
!>             التشابه المتعامد لمسح QR في
!>             Z(ILOZ:IHIZ,ILOZ:IHIZ) من اليمين.
!>             إذا كانت WANTZ = .FALSE.، فإن Z غير مستخدمة.
!>

LDZ

!>          LDZ هو INTEGER
!>             LDA هو البعد الرائد لـ Z كما تم تعريفه في
!>             الإجراء الاستدعاء. LDZ >= N.
!>

V

!>          V هو مصفوفة DOUBLE PRECISION، ذات البعد (LDV,NSHFTS/2)
!>

LDV

!>          LDV هو INTEGER
!>             LDV هو البعد الرائد لـ V كما تم تعريفه في
!>             الإجراء الاستدعاء. LDV >= 3.
!>

U

!>          U هو مصفوفة DOUBLE PRECISION، ذات البعد (LDU,2*NSHFTS)
!>

LDU

!>          LDU هو INTEGER
!>             LDU هو البعد الرائد لـ U كما تم تعريفه في
!>             في الإجراء الاستدعاء. LDU >= 2*NSHFTS.
!>

NV

!>          NV هو INTEGER
!>             NV هو عدد الصفوف في WV المتاحة للمساحة.
!>             NV >= 1.
!>

WV

!>          WV هو مصفوفة DOUBLE PRECISION، ذات البعد (LDWV,2*NSHFTS)
!>

LDWV

!>          LDWV هو INTEGER
!>             LDWV هو البعد الرائد لـ WV كما تم تعريفه في
!>             في الإجراء الاستدعاء. LDWV >= NV.
!>

NH

!>          NH هو INTEGER
!>             NH هو عدد الأعمدة في المصفوفة WH المتاحة
!>             للمساحة. NH >= 1.
!>

WH

!>          WH هو مصفوفة DOUBLE PRECISION، ذات البعد (LDWH,NH)
!>

LDWH

!>          LDWH هو INTEGER
!>             البعد الرائد لـ WH كما تم تعريفه في
!>             الإجراء الاستدعاء. LDWH >= 2*NSHFTS.
!>
