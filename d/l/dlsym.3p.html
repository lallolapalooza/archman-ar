الاسم

dlsym - الحصول على عنوان رمز من مكتبة مشتركة تم تحميلها

الملخص


#include <dlfcn.h>

void *dlsym(void *handle, const char *symbol);

الوصف

الدالة dlsym تقوم باسترداد عنوان رمز (متغير أو دالة) من مكتبة مشتركة تم تحميلها مسبقًا بواسطة dlopen.

  • handle: هو مقبض المكتبة المشتركة الذي تم إرجاعه بواسطة استدعاء dlopen. إذا كان RTLD_DEFAULT، فسيتم البحث عن الرمز في جميع المكتبات المحملة. إذا كان RTLD_NEXT، فسيتم البحث عن الرمز في المكتبة التالية بعد المكتبة التي قامت باستدعاء dlsym.
  • symbol: هو اسم الرمز الذي سيتم البحث عنه.

قيمة الإرجاع

عند النجاح، تُرجع dlsym عنوان الرمز المطلوب. عند الفشل، تُرجع NULL.

يمكنك التمييز بين فشل البحث عن رمز وعدم وجود الرمز باستخدام dlerror.

الأخطاء

إذا فشلت dlsym، يمكن استدعاء dlerror للحصول على وصف نصي للخطأ.

أمثلة


#include <stdio.h>
#include <stdlib.h>
#include <dlfcn.h>

int main() {
    void *handle;
    char *error;
    int (*func)();

    handle = dlopen("libm.so.6", RTLD_LAZY);
    if (!handle) {
        fprintf(stderr, "%s\n", dlerror());
        exit(EXIT_FAILURE);
    }

    // مسح أي أخطاء سابقة
    dlerror();

    func = dlsym(handle, "sin");

    error = dlerror();
    if (error != NULL) {
        fprintf(stderr, "%s\n", error);
        dlclose(handle);
        exit(EXIT_FAILURE);
    }

    printf("Address of sin: %p\n", (void*)func);

    dlclose(handle);
    return 0;
}

استخدام التطبيق

يتم استخدام dlsym عادةً عند تطوير مكونات إضافية (plugins) أو عند الحاجة إلى تحميل مكتبات ديناميكيًا في وقت التشغيل.

الأساس المنطقي

تسمح dlsym للتطبيقات بالوصول إلى الوظائف والمتغيرات داخل المكتبات المشتركة دون الحاجة إلى ربطها بشكل ثابت أثناء الترجمة. هذا يوفر مرونة كبيرة، خاصة في البيئات التي تتطلب تحميل مكونات جديدة أو تحديث المكتبات دون إعادة ترجمة التطبيق بأكمله.

الاتجاهات المستقبلية

لا توجد اتجاهات مستقبلية محددة للدالة dlsym نفسها، حيث أنها جزء أساسي من واجهة تحميل المكتبات الديناميكية. قد تتطور آليات إدارة المكتبات المشتركة، ولكن الدالة الأساسية للوصول إلى الرموز من المتوقع أن تظل كما هي.
