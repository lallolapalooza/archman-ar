توثيق الوظيفة/الإجراء الفرعي

subroutine dlarf1f (character side, integer m, integer n, double precision, dimension( * ) v, integer incv, double precision tau, double precision, dimension( ldc, * ) c, integer ldc, double precision, dimension( * ) work)

DLARF1F يطبق انعكاساً عنصرياً حقيقياً H على مصفوفة مستطيلة حقيقية m × n C، من اليسار أو اليمين.

الغرض:

!>
!> DLARF1F يطبق انعكاساً عنصرياً حقيقياً H على مصفوفة مستطيلة حقيقية
!> C، من اليسار أو اليمين. يُمثَّل H بالصيغة
!>
!>       H = I - tau * v * v**T
!>
!> حيث tau هو قيمة قياسية حقيقية و v هو متجه حقيقي.
!>
!> إذا كانت tau = 0، فسيتم اعتبار H مصفوفة الوحدة.
!>

المعاملات

SIDE 
!>          SIDE هو CHARACTER*1
!>          = 'L': تشكيل  H * C
!>          = 'R': تشكيل  C * H
!>

M

!>          M هو INTEGER
!>          عدد صفوف المصفوفة C.
!>

N

!>          N هو INTEGER
!>          عدد أعمدة المصفوفة C.
!>

V

!>          V هو مصفوفة DOUBLE PRECISION، أبعادها
!>                     (1 + (M-1)*abs(INCV)) إذا كان SIDE = 'L'
!>                  أو (1 + (N-1)*abs(INCV)) إذا كان SIDE = 'R'
!>          المتجه v في تمثيل H. لا يتم استخدام V إذا كان
!>          TAU = 0. لا يتم الإشارة إلى V(1) أو تعديله.
!>

INCV

!>          INCV هو INTEGER
!>          الزيادة بين عناصر v. INCV <> 0.
!>

TAU

!>          TAU هو DOUBLE PRECISION
!>          القيمة tau في تمثيل H.
!>

C

!>          C هي مصفوفة DOUBLE PRECISION، أبعادها (LDC,N)
!>          عند الدخول، المصفوفة C ذات الأبعاد m × n.
!>          عند الخروج، يتم استبدال C بالمصفوفة H * C إذا كان SIDE = 'L'،
!>          أو C * H إذا كان SIDE = 'R'.
!>

LDC

!>          LDC هو INTEGER
!>          البعد الرائد للمصفوفة C. LDC >= max(1,M).
!>

WORK

!>          WORK هي مصفوفة DOUBLE PRECISION، أبعادها
!>                         (N) إذا كان SIDE = 'L'
!>                      أو (M) إذا كان SIDE = 'R'
!>
