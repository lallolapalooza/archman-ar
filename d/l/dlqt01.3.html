توثيق الوظائف/الإجراءات الفرعية

subroutine dlqt01 (integer m, integer n, double precision, dimension( lda, * ) a, double precision, dimension( lda, * ) af, double precision, dimension( lda, * ) q, double precision, dimension( lda, * ) l, integer lda, double precision, dimension( * ) tau, double precision, dimension( lwork ) work, integer lwork, double precision, dimension( * ) rwork, double precision, dimension( * ) result)

DLQT01

الغرض:

!>
!> DLQT01 تختبر DGELQF، التي تحسب عاملة LQ للمصفوفة A ذات الأبعاد m × n،
!> وتختبر جزئيًا DORGLQ التي تشكل المصفوفة المتعامدة Q ذات الأبعاد n × n.
!>
!> DLQT01 تقارن L مع A*Q'، وتتحقق من أن Q متعامدة.
!>

المعلمات

M 
!>          M هو عدد صحيح (INTEGER)
!>          عدد الصفوف في المصفوفة A.  M >= 0.
!>

N

!>          N هو عدد صحيح (INTEGER)
!>          عدد الأعمدة في المصفوفة A.  N >= 0.
!>

A

!>          A هي مصفوفة من النوع DOUBLE PRECISION، أبعادها (LDA,N)
!>          المصفوفة A ذات الأبعاد m × n.
!>

AF

!>          AF هي مصفوفة من النوع DOUBLE PRECISION، أبعادها (LDA,N)
!>          تفاصيل عاملة LQ للمصفوفة A، كما تم إرجاعها بواسطة DGELQF.
!>          انظر DGELQF لمزيد من التفاصيل.
!>

Q

!>          Q هي مصفوفة من النوع DOUBLE PRECISION، أبعادها (LDA,N)
!>          المصفوفة المتعامدة Q ذات الأبعاد n × n.
!>

L

!>          L هي مصفوفة من النوع DOUBLE PRECISION، أبعادها (LDA,max(M,N))
!>

LDA

!>          LDA هو عدد صحيح (INTEGER)
!>          البعد الرئيسي للمصفوفات A، AF، Q، و L.
!>          LDA >= max(M,N).
!>

TAU

!>          TAU هي مصفوفة من النوع DOUBLE PRECISION، أبعادها (min(M,N))
!>          العوامل القياسية للمنعكسات الأولية، كما تم إرجاعها
!>          بواسطة DGELQF.
!>

WORK

!>          WORK هي مصفوفة من النوع DOUBLE PRECISION، أبعادها (LWORK)
!>

LWORK

!>          LWORK هو عدد صحيح (INTEGER)
!>          بعد المصفوفة WORK.
!>

RWORK

!>          RWORK هي مصفوفة من النوع DOUBLE PRECISION، أبعادها (max(M,N))
!>

RESULT

!>          RESULT هي مصفوفة من النوع DOUBLE PRECISION، أبعادها (2)
!>          نسب الاختبار:
!>          RESULT(1) = norm( L - A*Q' ) / ( N * norm(A) * EPS )
!>          RESULT(2) = norm( I - Q*Q' ) / ( N * EPS )
!>
