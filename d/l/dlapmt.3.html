subroutine dlapmt (logical forwrd, integer m, integer n, double precision, dimension( ldx, * ) x, integer ldx, integer, dimension( * ) k)

DLAPMT يقوم بترتيب عمودية مسبق أو عكسي لمصفوفة.

الغرض:

!>
!> DLAPMT يعيد ترتيب أعمدة المصفوفة M x N المسماة X حسب
!> الترتيب K(1),K(2),...,K(N) للأعداد الصحيحة 1,...,N.
!> إذا كان FORWRD = .TRUE.، ترتيب مسبق:
!>
!>      يتم نقل X(*,K(J)) إلى X(*,J) لـ J = 1,2,...,N.
!>
!> إذا كان FORWRD = .FALSE.، ترتيب عكسي:
!>
!>      يتم نقل X(*,J) إلى X(*,K(J)) لـ J = 1,2,...,N.
!>

المعلمات

FORWRD 
!>          FORWRD هو LOGICAL (منطقي)
!>          = .TRUE.، ترتيب مسبق
!>          = .FALSE.، ترتيب عكسي
!>

M

!>          M هو INTEGER (عدد صحيح)
!>          عدد صفوف المصفوفة X. M >= 0.
!>

N

!>          N هو INTEGER (عدد صحيح)
!>          عدد أعمدة المصفوفة X. N >= 0.
!>

X

!>          X هو مصفوفة DOUBLE PRECISION (دقة مزدوجة)، ذات أبعاد (LDX,N)
!>          عند الدخول، المصفوفة X ذات الأبعاد M x N.
!>          عند الخروج، تحتوي X على المصفوفة المعاد ترتيبها X.
!>

LDX

!>          LDX هو INTEGER (عدد صحيح)
!>          البعد الرئيسي للمصفوفة X، LDX >= MAX(1,M).
!>

K

!>          K هي مصفوفة INTEGER (عدد صحيح)، ذات أبعاد (N)
!>          عند الدخول، تحتوي K على متجه الترتيب. يتم استخدام K
!>          كمساحة عمل داخلية، ولكن يتم إعادة تعيينها إلى قيمتها الأصلية
!>          عند الخروج.
!>
