توثيق الوظيفة/الإجراء الفرعي

إجراء فرعي متكرر dlarft (character direct, character storev, integer n, integer k, double precision, dimension( ldv, * ) v, integer ldv, double precision, dimension( * ) tau, double precision, dimension( ldt, * ) t, integer ldt)

DLARFT يشكل عامل المثلث T للمنعكس الكتل H = I - vtvH

الغرض:

!>
!> DLARFT يشكل عامل المثلث T للمنعكس الكتل الحقيقي H
!> من الرتبة n، والذي يتم تعريفه كحاصل ضرب k منعكسات ابتدائية.
!>
!> إذا كان DIRECT = 'F'، فإن H = H(1) H(2) . . . H(k) و T مثلثية عليا؛
!>
!> إذا كان DIRECT = 'B'، فإن H = H(k) . . . H(2) H(1) و T مثلثية سفلى.
!>
!> إذا كان STOREV = 'C'، فإن المتجه الذي يعرف المنعكس الابتدائي
!> H(i) مخزن في العمود i من المصفوفة V، و
!>
!>    H  =  I - V * T * V**T
!>
!> إذا كان STOREV = 'R'، فإن المتجه الذي يعرف المنعكس الابتدائي
!> H(i) مخزن في الصف i من المصفوفة V، و
!>
!>    H  =  I - V**T * T * V
!>

المعلمات

DIRECT 
!>          DIRECT هو CHARACTER*1
!>          يحدد ترتيب ضرب المنعكسات الابتدائية
!>          لتشكيل المنعكس الكتل:
!>          = 'F': H = H(1) H(2) . . . H(k) (أمامي)
!>          = 'B': H = H(k) . . . H(2) H(1) (خلفي)
!>

STOREV

!>          STOREV هو CHARACTER*1
!>          يحدد كيفية تخزين المتجهات التي تعرف المنعكسات الابتدائية
!>          (انظر أيضاً التفاصيل الإضافية):
!>          = 'C': عمودياً
!>          = 'R': صفياً
!>

N

!>          N هو INTEGER
!>          رتبة المنعكس الكتل H. N >= 0.
!>

K

!>          K هو INTEGER
!>          رتبة عامل المثلث T (= عدد المنعكسات الابتدائية). K >= 1.
!>

V

!>          V هي مصفوفة DOUBLE PRECISION، البعد
!>                               (LDV,K) إذا كان STOREV = 'C'
!>                               (LDV,N) إذا كان STOREV = 'R'
!>          المصفوفة V. انظر التفاصيل الإضافية.
!>

LDV

!>          LDV هو INTEGER
!>          البعد الرئيسي للمصفوفة V.
!>          إذا كان STOREV = 'C'، فإن LDV >= max(1,N)؛ إذا كان STOREV = 'R'، فإن LDV >= K.
!>

TAU

!>          TAU هي مصفوفة DOUBLE PRECISION، البعد (K)
!>          يجب أن تحتوي TAU(i) على عامل القياس للمنعكس الابتدائي
!>          H(i).
!>

T

!>          T هي مصفوفة DOUBLE PRECISION، البعد (LDT,K)
!>          عامل المثلث k × k T للمنعكس الكتل.
!>          إذا كان DIRECT = 'F'، فإن T مثلثية عليا؛ إذا كان DIRECT = 'B'، فإن T مثلثية
!>          سفلى. لم يتم استخدام باقي المصفوفة.
!>

LDT

!>          LDT هو INTEGER
!>          البعد الرئيسي للمصفوفة T. LDT >= K.
!>

تفاصيل إضافية:

!>
!>  شكل المصفوفة V وتخزين المتجهات التي تعرف H(i) يوضحه المثال التالي مع n = 5 و
!>  k = 3. العناصر التي تساوي 1 غير مخزنة.
!>
!>  DIRECT = 'F' و STOREV = 'C':         DIRECT = 'F' و STOREV = 'R':
!>
!>               V = (  1       )                 V = (  1 v1 v1 v1 v1 )
!>                   ( v1  1    )                     (     1 v2 v2 v2 )
!>                   ( v1 v2  1 )                     (        1 v3 v3 )
!>                   ( v1 v2 v3 )
!>                   ( v1 v2 v3 )
!>
!>  DIRECT = 'B' و STOREV = 'C':         DIRECT = 'B' و STOREV = 'R':
!>
!>               V = ( v1 v2 v3 )                 V = ( v1 v1  1       )
!>                   ( v1 v2 v3 )                     ( v2 v2 v2  1    )
!>                   (  1 v2 v3 )                     ( v3 v3 v3 v3  1 )
!>                   (     1 v3 )
!>                   (        1 )
!>
