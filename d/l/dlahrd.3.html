subroutine dlahrd (n, k, nb, a, lda, tau, t, ldt, y, ldy)
DLAHRD تقلل أول nb عمودًا من مصفوفة A مستطيلة عامة بحيث تصبح العناصر الموجودة أسفل القطر الفرعي k صفرًا، وتعيد مصفوفات مساعدة مطلوبة لتطبيق التحويل على الجزء غير المختزل من A.

توثيق الوظيفة/الروتين الفرعي

subroutine dlahrd (integer n, integer k, integer nb, double precision, dimension( lda, * ) a, integer lda, double precision, dimension( nb ) tau, double precision, dimension( ldt, nb ) t, integer ldt, double precision, dimension( ldy, nb ) y, integer ldy)

DLAHRD تقلل أول nb عمودًا من مصفوفة A مستطيلة عامة بحيث تصبح العناصر الموجودة أسفل القطر الفرعي k صفرًا، وتعيد مصفوفات مساعدة مطلوبة لتطبيق التحويل على الجزء غير المختزل من A.

الغرض:

!>
!> هذا الروتين مهمل وتم استبداله بالروتين DLAHR2.
!>
!> DLAHRD تقلل أول NB عمودًا من مصفوفة A عامة n-by-(n-k+1)
!> بحيث تكون العناصر الموجودة أسفل القطر الفرعي k صفرًا. يتم
!> إجراء الاختزال بواسطة تحويل تشابه متعامد
!> Q**T * A * Q. يعيد الروتين المصفوفتين V و T اللتين تحددان
!> Q كمُعكِس كتلة I - V*T*V**T، وكذلك المصفوفة Y = A * V * T.
!>

!>          K هو عدد صحيح (INTEGER)
!>          إزاحة الاختزال. يتم اختزال العناصر الموجودة أسفل القطر الفرعي
!>          k في أول NB عمودًا إلى صفر.
!>

!>          عدد الأعمدة التي سيتم اختزالها.
!>

A

!>          A هي مصفوفة ذات دقة مزدوجة (DOUBLE PRECISION)، ذات بُعد (LDA,N-K+1)
!>          عند الإدخال، المصفوفة العامة n-by-(n-k+1) A.
!>          عند الخروج، يتم استبدال العناصر الموجودة على القطر الفرعي k وفوقه في
!>          أول NB عمودًا بالعناصر المقابلة للمصفوفة المختزلة؛ العناصر الموجودة أسفل
!>          القطر الفرعي k، مع مصفوفة TAU، تمثل المصفوفة Q كـ
!>          حاصل ضرب مُعكِسات أولية. الأعمدة الأخرى لـ A لم تتغير. انظر التفاصيل الإضافية.
!>

!>

T

!>          T هي مصفوفة ذات دقة مزدوجة (DOUBLE PRECISION)، ذات بُعد (LDT,NB)

!>

LDT

!>          LDT هو عدد صحيح (INTEGER)
!>          البعد الرائد للمصفوفة T. LDT >= NB.
!>

Y

!>          Y هي مصفوفة ذات دقة مزدوجة (DOUBLE PRECISION)، ذات بُعد (LDY,NB)

تفاصيل إضافية:

!>
!>  يتم تمثيل المصفوفة Q كحاصل ضرب nb من المُعكِسات الأولية
!>
!>     Q = H(1) H(2) . . . H(nb).
!>
!>  كل H(i) لها الشكل
!>
!>     H(i) = I - tau * v * v**T
!>
!>  حيث tau هو عامل قياسي حقيقي، و v هو متجه حقيقي مع
!>  v(1:i+k-1) = 0، v(i+k) = 1؛ v(i+k+1:n) مخزن عند الخروج في
!>  A(i+k+1:n,i)، و tau في TAU(i).
!>
!>  تُشكل عناصر المتجهات v المصفوفة (n-k+1)-by-nb
!>  V المطلوبة، مع T و Y، لتطبيق التحويل على
!>  الجزء غير المختزل من المصفوفة، باستخدام تحديث بالشكل:
!>  A := (I - V*T*V**T) * (A - Y*V**T).
!>
!>  محتويات A عند الخروج موضحة بالمثال التالي
!>  مع n = 7، k = 3 و nb = 2:
!>
!>     ( a   h   a   a   a )
!>     ( a   h   a   a   a )
!>     ( a   h   a   a   a )
!>     ( h   h   a   a   a )
!>     ( v1  h   a   a   a )
!>     ( v1  v2  a   a   a )
!>     ( v1  v2  a   a   a )
!>
!>  حيث يمثل a عنصرًا من المصفوفة الأصلية A، ويمثل h عنصرًا معدلاً
!>  من المصفوفة هسنبرغ العلوية H، ويمثل vi عنصرًا من المتجه الذي يحدد H(i).
!>
