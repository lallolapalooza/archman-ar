subroutine dlavsp (character uplo, character trans, character diag, integer n, integer nrhs, double precision, dimension( * ) a, integer, dimension( * ) ipiv, double precision, dimension( ldb, * ) b, integer ldb, integer info)

DLAVSP

الغرض:

!>
!> DLAVSP  تنفذ إحدى عمليات المصفوفة-المتجه
!>    x := A*x  أو  x := A'*x,
!> حيث x متجه من N عنصر و A هي إحدى العوامل
!> من تحليل الكتلة U*D*U' أو L*D*L' الذي تم حسابه بواسطة DSPTRF.
!>
!> إذا كان TRANS = 'N'، تضرب في U  أو U * D  (أو L  أو L * D)
!> إذا كان TRANS = 'T'، تضرب في U' أو D * U' (أو L' أو D * L' )
!> إذا كان TRANS = 'C'، تضرب في U' أو D * U' (أو L' أو D * L' )
!>

المعاملات

UPLO 
!>          UPLO هو CHARACTER*1
!>          يحدد ما إذا كان العامل المخزن في A علويًا أم سفليًا
!>          مثلثيًا.
!>          = 'U':  مثلثي علوي
!>          = 'L':  مثلثي سفلي
!>

TRANS

!>          TRANS هو CHARACTER*1
!>          يحدد العملية المراد تنفيذها:
!>          = 'N':  x := A*x
!>          = 'T':  x := A'*x
!>          = 'C':  x := A'*x
!>

DIAG

!>          DIAG هو CHARACTER*1
!>          يحدد ما إذا كانت الكتل القطرية وحدات
!>          أم لا. إذا كان من المفترض أن تكون الكتل القطرية وحدات،
!>          فإن A = U أو A = L، وإلا فإن A = U*D أو A = L*D.
!>          = 'U':  من المفترض أن تكون الكتل القطرية وحدات.
!>          = 'N':  من المفترض أن تكون الكتل القطرية غير وحدات.
!>

N

!>          N هو INTEGER
!>          عدد الصفوف والأعمدة في المصفوفة A.  N >= 0.
!>

NRHS

!>          NRHS هو INTEGER
!>          عدد الأطراف اليمنى، أي عدد المتجهات
!>          x المراد ضربها في A.  NRHS >= 0.
!>

A

!>          A هي مصفوفة DOUBLE PRECISION، البعد (N*(N+1)/2)
!>          المصفوفة القطرية الكتلية D والمضاعفات المستخدمة
!>          للحصول على العامل U أو L، مخزنة كمصفوفة مثلثية معبأة
!>          كما تم حسابها بواسطة DSPTRF.
!>

IPIV

!>          IPIV هي مصفوفة INTEGER، البعد (N)
!>          فهارس المحاور من DSPTRF.
!>

B

!>          B هي مصفوفة DOUBLE PRECISION، البعد (LDB,NRHS)
!>          عند الدخول، تحتوي B على NRHS متجه بطول N.
!>          عند الخروج، يتم الكتابة فوق B بالناتج A * B.
!>

LDB

!>          LDB هو INTEGER
!>          البعد الرئيسي للمصفوفة B.  LDB >= max(1,N).
!>

INFO

!>          INFO هو INTEGER
!>          = 0: خروج ناجح
!>          < 0: إذا كان INFO = -k، كان المعامل k غير قانوني
!>
