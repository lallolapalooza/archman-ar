توثيق الوظيفة/الإجراء الفرعي

subroutine dlarra (integer n, double precision, dimension( * ) d, double precision, dimension( * ) e, double precision, dimension( * ) e2, double precision spltol, double precision tnrm, integer nsplit, integer, dimension( * ) isplit, integer info)

DLARRA تحسب نقاط التقسيم مع العتبة المحددة.

الغرض:

!>
!> احسب نقاط التقسيم بالعتبة SPLTOL.
!> DLARRA تعيّن أي عناصر خارج القطر إلى الصفر.
!>

المعلمات

N 
!>          N هو عدد صحيح (INTEGER)
!>          رتبة المصفوفة. N > 0.
!>

D

!>          D هو مصفوفة من النوع double precision، البعد (N)
!>          عند الإدخال، العناصر القطرية N للمصفوفة ثلاثية الأقطار T.
!>

E

!>          E هو مصفوفة من النوع double precision، البعد (N)
!>          عند الإدخال، تحتوي أول (N-1) مداخل على العناصر تحت القطرية
!>          للمصفوفة ثلاثية الأقطار T؛ لا حاجة لتعيين E(N).
!>          عند الخروج، يتم تعيين الإدخالات E( ISPLIT( I ) )، 1 <= I <= NSPLIT،
!>          إلى الصفر، ولا يتم المساس بالإدخالات الأخرى لـ E.
!>

E2

!>          E2 هو مصفوفة من النوع double precision، البعد (N)
!>          عند الإدخال، تحتوي أول (N-1) مداخل على مربعات
!>          العناصر تحت القطرية للمصفوفة ثلاثية الأقطار T؛
!>          لا حاجة لتعيين E2(N).
!>          عند الخروج، تم تعيين الإدخالات E2( ISPLIT( I ) )،
!>          1 <= I <= NSPLIT، إلى الصفر.
!>

SPLTOL

!>          SPLTOL هو double precision
!>          العتبة للتقسيم. يمكن استخدام معيارين:
!>          SPLTOL<0 : معيار يعتمد على قيمة خارج القطر المطلقة
!>          SPLTOL>0 : معيار يحافظ على الدقة النسبية
!>

TNRM

!>          TNRM هو double precision
!>          معيار المصفوفة.
!>

NSPLIT

!>          NSPLIT هو عدد صحيح (INTEGER)
!>          عدد الكتل التي تنقسم إليها T. 1 <= NSPLIT <= N.
!>

ISPLIT

!>          ISPLIT هو مصفوفة من النوع integer، البعد (N)
!>          نقاط التقسيم، التي تنقسم عندها T إلى كتل.
!>          الكتلة الأولى تتكون من الصفوف/الأعمدة من 1 إلى ISPLIT(1)،
!>          الثانية من الصفوف/الأعمدة ISPLIT(1)+1 إلى ISPLIT(2)،
!>          وهكذا، والكتلة NSPLIT تتكون من الصفوف/الأعمدة
!>          ISPLIT(NSPLIT-1)+1 إلى ISPLIT(NSPLIT)=N.
!>

INFO

!>          INFO هو عدد صحيح (INTEGER)
!>          = 0: تم الخروج بنجاح
!>
