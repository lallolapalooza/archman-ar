subroutine dlagsy (integer n, integer k, double precision, dimension( * ) d, double precision, dimension( lda, * ) a, integer lda, integer, dimension( 4 ) iseed, double precision, dimension( * ) work, integer info)

DLAGSY

الغرض:

!>
!> DLAGSY تولد مصفوفة متناظرة حقيقية A، عن طريق الضرب المسبق واللاحق
!> لمصفوفة قطرية D عشوائية بمصفوفة متعامدة عشوائية:
!> A = U*D*U'. ثم يمكن تقليل عرض النطاق شبه للـ K بواسطة تحويلات
!> متعامدة إضافية.
!>

المعاملات

N 
!>          N هو عدد صحيح (INTEGER)
!>          رتبة المصفوفة A.  N >= 0.
!>

K

!>          K هو عدد صحيح (INTEGER)
!>          عدد الأقطار الفرعية غير الصفرية داخل نطاق A.
!>          0 <= K <= N-1.
!>

D

!>          D هي مصفوفة ذات دقة مزدوجة (DOUBLE PRECISION)، البعد (N)
!>          العناصر القطرية للمصفوفة القطرية D.
!>

A

!>          A هي مصفوفة ذات دقة مزدوجة (DOUBLE PRECISION)، البعد (LDA,N)
!>          المصفوفة المتناظرة N x N المولدة (يتم تخزين المصفوفة الكاملة).
!>

LDA

!>          LDA هو عدد صحيح (INTEGER)
!>          البعد الرائد للمصفوفة A.  LDA >= N.
!>

ISEED

!>          ISEED هي مصفوفة أعداد صحيحة (INTEGER)، البعد (4)
!>          عند الإدخال، البذرة لمولد الأرقام العشوائية؛ يجب أن تكون عناصر
!>          المصفوفة بين 0 و 4095، ويجب أن يكون ISEED(4) فردياً.
!>          عند الخروج، يتم تحديث البذرة.
!>

WORK

!>          WORK هي مصفوفة ذات دقة مزدوجة (DOUBLE PRECISION)، البعد (2*N)
!>

INFO

!>          INFO هو عدد صحيح (INTEGER)
!>          = 0: خروج ناجح
!>          < 0: إذا كان INFO = -i، فإن المعامل i كان له قيمة غير قانونية
!>
