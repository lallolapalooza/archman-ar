subroutine dlapll (integer n, double precision, dimension( * ) x, integer incx, double precision, dimension( * ) y, integer incy, double precision ssmin)

DLAPLL تقيس الاعتماد الخطي لمتجهين.

الغرض:

!>
!> بالنظر إلى المتجهين العموديين X و Y، لنكن
!>
!>                      A = ( X Y ).
!>
!> يقوم الإجراء الفرعي أولاً بحساب تحلل QR للمصفوفة A = Q*R،
!> ثم يحسب SVD للمصفوفة المثلثية العلوية 2x2 R.
!> القيمة المفردة الأصغر لـ R يتم إرجاعها في SSMIN، والتي تستخدم
!> لقياس الاعتماد الخطي للمتجهين X و Y.
!>

المعلمات

N 
!>          N هو عدد صحيح (INTEGER)
!>          طول المتجهين X و Y.
!>

X

!>          X هو مصفوفة من النوع DOUBLE PRECISION،
!>                         ذات بُعد (1+(N-1)*INCX)
!>          عند الدخول، يحتوي X على المتجه N-vector X.
!>          عند الخروج، يتم استبدال X.
!>

INCX

!>          INCX هو عدد صحيح (INTEGER)
!>          الزيادة بين العناصر المتتالية لـ X. INCX > 0.
!>

Y

!>          Y هو مصفوفة من النوع DOUBLE PRECISION،
!>                         ذات بُعد (1+(N-1)*INCY)
!>          عند الدخول، يحتوي Y على المتجه N-vector Y.
!>          عند الخروج، يتم استبدال Y.
!>

INCY

!>          INCY هو عدد صحيح (INTEGER)
!>          الزيادة بين العناصر المتتالية لـ Y. INCY > 0.
!>

SSMIN

!>          SSMIN هو DOUBLE PRECISION
!>          أصغر قيمة مفردة للمصفوفة N-by-2 A = ( X Y ).
!>
