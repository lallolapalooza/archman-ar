توثيق الوظيفة/الروتين الفرعي

subroutine dlaptm (integer n, integer nrhs, double precision alpha, double precision, dimension( * ) d, double precision, dimension( * ) e, double precision, dimension( ldx, * ) x, integer ldx, double precision beta, double precision, dimension( ldb, * ) b, integer ldb)

DLAPTM

الغرض:

!>
!> تقوم DLAPTM بضرب مصفوفة X ذات الأبعاد N × NRHS بمصفوفة A المتناظرة ثلاثية القطر
!> وتخزين النتيجة في مصفوفة B. العملية لها الشكل
!>
!>    B := alpha * A * X + beta * B
!>
!> حيث قد يكون alpha إما 1. أو -1. وقد يكون beta 0. أو 1. أو -1.
!>

المعلمات

N 
!>          N هو عدد صحيح (INTEGER)
!>          رتبة المصفوفة A.  N >= 0.
!>

NRHS

!>          NRHS هو عدد صحيح (INTEGER)
!>          عدد الأيادي اليمنى، أي عدد أعمدة المصفوفتين X و B.
!>

ALPHA

!>          ALPHA هو دقة مضاعفة (DOUBLE PRECISION)
!>          المقدار القياسي alpha.  يجب أن يكون ALPHA إما 1. أو -1.؛ بخلاف ذلك،
!>          يفترض أنه 0.
!>

D

!>          D هي مصفوفة دقة مضاعفة (DOUBLE PRECISION)، ذات بعد (N)
!>          العناصر القطرية N للمصفوفة ثلاثية القطر A.
!>

E

!>          E هي مصفوفة دقة مضاعفة (DOUBLE PRECISION)، ذات بعد (N-1)
!>          العناصر تحت القطرية أو فوق القطرية (N-1) لـ A.
!>

X

!>          X هي مصفوفة دقة مضاعفة (DOUBLE PRECISION)، ذات بعد (LDX,NRHS)
!>          المصفوفة X ذات الأبعاد N × NRHS.
!>

LDX

!>          LDX هو عدد صحيح (INTEGER)
!>          البعد الرئيسي للمصفوفة X.  LDX >= max(N,1).
!>

BETA

!>          BETA هو دقة مضاعفة (DOUBLE PRECISION)
!>          المقدار القياسي beta.  يجب أن يكون BETA إما 0. أو 1. أو -1.؛ بخلاف ذلك،
!>          يفترض أنه 1.
!>

B

!>          B هي مصفوفة دقة مضاعفة (DOUBLE PRECISION)، ذات بعد (LDB,NRHS)
!>          عند الإدخال، المصفوفة B ذات الأبعاد N × NRHS.
!>          عند الخروج، يتم استبدال B بالمصفوفة التعبيرية
!>          B := alpha * A * X + beta * B.
!>

LDB

!>          LDB هو عدد صحيح (INTEGER)
!>          البعد الرئيسي للمصفوفة B.  LDB >= max(N,1).
!>
