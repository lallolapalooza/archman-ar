subroutine dlaeda (n, tlvls, curlvl, curpbm, prmptr, perm, givptr, givcol, givnum, q, qptr, z, ztemp, info)
DLAEDA تستخدم بواسطة DSTEDC. تقوم بحساب المتجه Z الذي يحدد التعديل الرتبي-واحد للمصفوفة القطرية. تستخدم عندما تكون المصفوفة الأصلية كثيفة.

توثيق الوظيفة/الوحدة الفرعية

subroutine dlaeda (integer n, integer tlvls, integer curlvl, integer curpbm, integer, dimension( * ) prmptr, integer, dimension( * ) perm, integer, dimension( * ) givptr, integer, dimension( 2, * ) givcol, double precision, dimension( 2, * ) givnum, double precision, dimension( * ) q, integer, dimension( * ) qptr, double precision, dimension( * ) z, double precision, dimension( * ) ztemp, integer info)

DLAEDA تستخدم بواسطة DSTEDC. تقوم بحساب المتجه Z الذي يحدد التعديل الرتبي-واحد للمصفوفة القطرية. تستخدم عندما تكون المصفوفة الأصلية كثيفة.

الغرض:

!>
!> DLAEDA تحسب المتجه Z المقابل لخطوة الدمج في الخطوة CURLVLth من عملية الدمج
!> مع TLVLS خطوة للمشكلة CURPBMth.
!>

المعلمات

N 
!>          N هو عدد صحيح (INTEGER)
!>         بعد المصفوفة المتماثلة الثلاثية. N >= 0.
!>

TLVLS

!>          TLVLS هو عدد صحيح (INTEGER)
!>         العدد الإجمالي لمستويات الدمج في شجرة التقسيم والتغلب الشاملة.
!>

CURLVL

!>          CURLVL هو عدد صحيح (INTEGER)
!>         المستوى الحالي في روتين الدمج الشامل،
!>         0 <= curlvl <= tlvls.
!>

CURPBM

!>          CURPBM هو عدد صحيح (INTEGER)
!>         المشكلة الحالية في المستوى الحالي في الدمج الشامل
!>         (العد من أعلى اليسار إلى أسفل اليمين).
!>

PRMPTR

!>          PRMPTR هو مصفوفة أعداد صحيحة (INTEGER array)، الأبعاد (N lg N)
!>         تحتوي على قائمة من المؤشرات التي تشير إلى مكان تخزين تبديل
!>         المستوى في PERM.  PRMPTR(i+1) - PRMPTR(i)
!>         يشير إلى حجم التبديل، وبشكل عرضي، حجم المشكلة الكاملة وغير المنفصلة.
!>

PERM

!>          PERM هو مصفوفة أعداد صحيحة (INTEGER array)، الأبعاد (N lg N)
!>         تحتوي على التبديلات (من الفصل والفرز) التي سيتم
!>         تطبيقها على كل كتلة فكرية.
!>

GIVPTR

!>          GIVPTR هو مصفوفة أعداد صحيحة (INTEGER array)، الأبعاد (N lg N)
!>         تحتوي على قائمة من المؤشرات التي تشير إلى مكان تخزين دورانات
!>         Givens للمستوى في GIVCOL.  GIVPTR(i+1) - GIVPTR(i)
!>         يشير إلى عدد دورانات Givens.
!>

GIVCOL

!>          GIVCOL هو مصفوفة أعداد صحيحة (INTEGER array)، الأبعاد (2, N lg N)
!>         كل زوج من الأرقام يشير إلى زوج من الأعمدة التي سيحدث فيها
!>         دوران Givens.
!>

GIVNUM

!>          GIVNUM هو مصفوفة أرقام مزدوجة الدقة (DOUBLE PRECISION array)، الأبعاد (2, N lg N)
!>         كل رقم يشير إلى قيمة S التي سيتم استخدامها في دوران
!>         Givens المقابل.
!>

Q

!>          Q هو مصفوفة أرقام مزدوجة الدقة (DOUBLE PRECISION array)، الأبعاد (N**2)
!>         تحتوي على كتل الأفكار المربعة من المستويات السابقة،
!>         المواقع الأولية للكتل معطاة بواسطة QPTR.
!>

QPTR

!>          QPTR هو مصفوفة أعداد صحيحة (INTEGER array)، الأبعاد (N+2)
!>         تحتوي على قائمة من المؤشرات التي تشير إلى مكان تخزين كتلة
!>         الفكرة في Q.  SQRT( QPTR(i+1) - QPTR(i) ) يشير
!>         إلى حجم الكتلة.
!>

Z

!>          Z هو مصفوفة أرقام مزدوجة الدقة (DOUBLE PRECISION array)، الأبعاد (N)
!>         عند الخرج، تحتوي هذه المصفوفة على المتجه المحدث (الصف الأخير
!>         لمصفوفة المتجهات الفكرية الفرعية الأولى والصف الأول
!>         لمصفوفة المتجهات الفكرية الفرعية الثانية).
!>

ZTEMP

!>          ZTEMP هو مصفوفة أرقام مزدوجة الدقة (DOUBLE PRECISION array)، الأبعاد (N)
!>

INFO

!>          INFO هو عدد صحيح (INTEGER)
!>          = 0:  خروج ناجح.
!>          < 0:  إذا كانت INFO = -i، فإن المعلمة i-th كانت لها قيمة غير قانونية.
!>
