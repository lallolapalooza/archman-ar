subroutine dlaqsb (character uplo, integer n, integer kd, double precision, dimension( ldab, * ) ab, integer ldab, double precision, dimension( * ) s, double precision scond, double precision amax, character equed)

DLAQSB يقوم بقياس مصفوفة متناظرة/هرمية ذات نطاق، باستخدام عوامل القياس التي تم حسابها بواسطة spbequ.

الغرض:

!>
!> DLAQSB يوازن مصفوفة متناظرة ذات نطاق A باستخدام عوامل القياس
!> في المتجه S.
!>

المعلمات

UPLO 
!>          UPLO هو CHARACTER*1
!>          يحدد ما إذا كان الجزء العلوي أو السفلي المثلثي للمصفوفة
!>          المتناظرة A مخزنًا.
!>          = 'U':  مثلثي علوي
!>          = 'L':  مثلثي سفلي
!>

N

!>          N هو INTEGER
!>          رتبة المصفوفة A.  N >= 0.
!>

KD

!>          KD هو INTEGER
!>          عدد الأقطار الفائقة للمصفوفة A إذا كانت UPLO = 'U'،
!>          أو عدد الأقطار الفرعية إذا كانت UPLO = 'L'.  KD >= 0.
!>

AB

!>          AB هو مصفوفة DOUBLE PRECISION، ذات البعد (LDAB,N)
!>          عند الإدخال، الجزء العلوي أو السفلي المثلثي للمصفوفة المتناظرة
!>          ذات النطاق A، المخزنة في أول KD+1 صفوف من المصفوفة.  العمود
!>          j من A مخزن في العمود j من المصفوفة AB
!>          كما يلي:
!>          إذا كانت UPLO = 'U'، فإن AB(kd+1+i-j,j) = A(i,j) لـ max(1,j-kd)<=i<=j؛
!>          إذا كانت UPLO = 'L'، فإن AB(1+i-j,j)    = A(i,j) لـ j<=i<=min(n,j+kd).
!>
!>          عند الخروج، إذا كانت INFO = 0، فإن العامل المثلثي U أو L من
!>          تحليل Cholesky A = U**T*U أو A = L*L**T للمصفوفة ذات النطاق
!>          A، بنفس تنسيق التخزين مثل A.
!>

LDAB

!>          LDAB هو INTEGER
!>          البعد الرائد للمصفوفة AB.  LDAB >= KD+1.
!>

S

!>          S هي مصفوفة DOUBLE PRECISION، ذات البعد (N)
!>          عوامل القياس للمصفوفة A.
!>

SCOND

!>          SCOND هو DOUBLE PRECISION
!>          نسبة أصغر S(i) إلى أكبر S(i).
!>

AMAX

!>          AMAX هو DOUBLE PRECISION
!>          القيمة المطلقة لأكبر عنصر في المصفوفة.
!>

EQUED

!>          EQUED هو CHARACTER*1
!>          يحدد ما إذا تم إجراء القياس أم لا.
!>          = 'N':  لم يتم إجراء القياس.
!>          = 'Y':  تم إجراء القياس، أي تم استبدال A بـ
!>                  diag(S) * A * diag(S).
!>

المعلمات الداخلية:

!>  THRESH هو حد يستخدم لتحديد ما إذا كان ينبغي إجراء القياس
!>  بناءً على نسبة عوامل القياس.  إذا كان SCOND < THRESH،
!>  يتم إجراء القياس.
!>
!>  LARGE و SMALL هي حدود تستخدم لتحديد ما إذا كان ينبغي إجراء القياس
!>  بناءً على الحجم المطلق لأكبر عنصر في المصفوفة.
!>  إذا كان AMAX > LARGE أو AMAX < SMALL، يتم إجراء القياس.
!>
