توثيق الوظيفة/الروتين الفرعي

subroutine dlarz (character side, integer m, integer n, integer l, double precision, dimension( * ) v, integer incv, double precision tau, double precision, dimension( ldc, * ) c, integer ldc, double precision, dimension( * ) work)

DLARZ يطبق عاكس أولي (كما تم إرجاعه بواسطة stzrzf) على مصفوفة عامة.

الغرض:

!>
!> DLARZ يطبق عاكس أولي حقيقي H على مصفوفة C حقيقية M-by-N،
!> من اليسار أو اليمين. يتم تمثيل H بالشكل
!>
!>       H = I - tau * v * v**T
!>
!> حيث tau هو قيمة قياسية حقيقية و v هو متجه حقيقي.
!>
!> إذا كان tau = 0، فإن H تعتبر مصفوفة الوحدة.
!>
!>
!> H هو حاصل ضرب k عواكس أولية كما تم إرجاعه بواسطة DTZRZF.
!>

المعلمات

SIDE 
!>          SIDE هو CHARACTER*1
!>          = 'L': شكل  H * C
!>          = 'R': شكل  C * H
!>

M

!>          M هو INTEGER
!>          عدد صفوف المصفوفة C.
!>

N

!>          N هو INTEGER
!>          عدد أعمدة المصفوفة C.
!>

L

!>          L هو INTEGER
!>          عدد عناصر المتجه V التي تحتوي على
!>          الجزء الهادف من متجهات هاوسهولدر.
!>          إذا كان SIDE = 'L'، فإن M >= L >= 0، إذا كان SIDE = 'R'، فإن N >= L >= 0.
!>

V

!>          V هو مصفوفة DOUBLE PRECISION، البعد (1+(L-1)*abs(INCV))
!>          المتجه v في تمثيل H كما تم إرجاعه بواسطة
!>          DTZRZF. لا يتم استخدام V إذا كان TAU = 0.
!>

INCV

!>          INCV هو INTEGER
!>          الزيادة بين عناصر v. INCV <> 0.
!>

TAU

!>          TAU هو DOUBLE PRECISION
!>          القيمة tau في تمثيل H.
!>

C

!>          C هي مصفوفة DOUBLE PRECISION، البعد (LDC,N)
!>          عند الدخول، المصفوفة C بالمقاس M-by-N.
!>          عند الخروج، يتم استبدال C بالمصفوفة H * C إذا كان SIDE = 'L'،
!>          أو C * H إذا كان SIDE = 'R'.
!>

LDC

!>          LDC هو INTEGER
!>          البعد الرئيسي للمصفوفة C. LDC >= max(1,M).
!>

WORK

!>          WORK هي مصفوفة DOUBLE PRECISION، البعد
!>                         (N) إذا كان SIDE = 'L'
!>                      أو (M) إذا كان SIDE = 'R'
!>
