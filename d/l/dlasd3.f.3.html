subroutine dlasd3 (nl, nr, sqre, k, d, q, ldq, dsigma, u, ldu, u2, ldu2, vt, ldvt, vt2, ldvt2, idxc, ctot, z, info)
DLASD3 يجد جميع الجذور التربيعية لجذور المعادلة العلمانية، كما تحددها القيم في D و Z، ثم يحدث المتجهات الفردية عن طريق ضرب المصفوفة. يستخدم بواسطة sbdsdc.

توثيق الدالة/الروتين الفرعي

subroutine dlasd3 (integer nl, integer nr, integer sqre, integer k, double precision, dimension( * ) d, double precision, dimension( ldq, * ) q, integer ldq, double precision, dimension( * ) dsigma, double precision, dimension( ldu, * ) u, integer ldu, double precision, dimension( ldu2, * ) u2, integer ldu2, double precision, dimension( ldvt, * ) vt, integer ldvt, double precision, dimension( ldvt2, * ) vt2, integer ldvt2, integer, dimension( * ) idxc, integer, dimension( * ) ctot, double precision, dimension( * ) z, integer info)

DLASD3 يجد جميع الجذور التربيعية لجذور المعادلة العلمانية، كما تحددها القيم في D و Z، ثم يحدث المتجهات الفردية عن طريق ضرب المصفوفة. يستخدم بواسطة sbdsdc.

الغرض:

!>
!> DLASD3 يجد جميع الجذور التربيعية لجذور المعادلة العلمانية،
!> كما تحددها القيم في D و Z. يقوم بإجراء المكالمات المناسبة لـ DLASD4
!> ثم يقوم بتحديث المتجهات الفردية عن طريق ضرب المصفوفة.
!>
!> يتم استدعاء DLASD3 من DLASD1.
!>

المعلمات

NL 
!>          NL هو عدد صحيح
!>         بعد الصف للمجموعة العلوية. NL >= 1.
!>

NR

!>          NR هو عدد صحيح
!>         بعد الصف للمجموعة السفلية. NR >= 1.
!>

SQRE

!>          SQRE هو عدد صحيح
!>         = 0: المجموعة السفلية هي مصفوفة مربعة NR-by-NR.
!>         = 1: المجموعة السفلية هي مصفوفة مستطيلة NR-by-(NR+1).
!>
!>         تحتوي المصفوفة ثنائية القطر على N = NL + NR + 1 صفوف و
!>         M = N + SQRE >= N أعمدة.
!>

K

!>          K هو عدد صحيح
!>         حجم المعادلة العلمانية، 1 =< K = < N.
!>

D

!>          D هي مصفوفة DOUBLE PRECISION، البعد (K)
!>         عند الخروج الجذور التربيعية لجذور المعادلة العلمانية،
!>         مرتبة تصاعدياً.
!>

Q

!>          Q هي مصفوفة DOUBLE PRECISION، البعد (LDQ,K)
!>

LDQ

!>          LDQ هو عدد صحيح
!>         البعد الرائد للمصفوفة Q. LDQ >= K.
!>

DSIGMA

!>          DSIGMA هي مصفوفة DOUBLE PRECISION، البعد (K)
!>         تحتوي العناصر K الأولى لهذه المصفوفة على الجذور القديمة
!>         لمشكلة التحديث المخففة. هذه هي الأقطاب
!>         للمعادلة العلمانية.
!>

U

!>          U هي مصفوفة DOUBLE PRECISION، البعد (LDU, N)
!>         تحتوي الأعمدة N - K الأخيرة من هذه المصفوفة على المتجهات
!>         الفردية اليسرى المخففة.
!>

LDU

!>          LDU هو عدد صحيح
!>         البعد الرائد للمصفوفة U. LDU >= N.
!>

U2

!>          U2 هي مصفوفة DOUBLE PRECISION، البعد (LDU2, N)
!>         تحتوي الأعمدة K الأولى لهذه المصفوفة على المتجهات
!>         الفردية اليسرى غير المخففة للمشكلة المنقسمة.
!>

LDU2

!>          LDU2 هو عدد صحيح
!>         البعد الرائد للمصفوفة U2. LDU2 >= N.
!>

VT

!>          VT هي مصفوفة DOUBLE PRECISION، البعد (LDVT, M)
!>         تحتوي الأعمدة M - K الأخيرة من VT**T على المتجهات
!>         الفردية اليمنى المخففة.
!>

LDVT

!>          LDVT هو عدد صحيح
!>         البعد الرائد للمصفوفة VT. LDVT >= N.
!>

VT2

!>          VT2 هي مصفوفة DOUBLE PRECISION، البعد (LDVT2, N)
!>         تحتوي الأعمدة K الأولى من VT2**T على المتجهات
!>         الفردية اليمنى غير المخففة للمشكلة المنقسمة.
!>

LDVT2

!>          LDVT2 هو عدد صحيح
!>         البعد الرائد للمصفوفة VT2. LDVT2 >= N.
!>

IDXC

!>          IDXC هي مصفوفة عدد صحيح، البعد ( N )
!>         التبديل المستخدم لترتيب أعمدة U (وصفوف
!>         VT) إلى ثلاث مجموعات: المجموعة الأولى تحتوي على إدخالات غير صفرية
!>         فقط عند NL +1 أو قبله؛ الثانية
!>         تحتوي على إدخالات غير صفرية فقط عند NL+2 أو بعده؛
!>         والثالثة كثيفة. العمود الأول من U والصف
!>         من VT يعاملان بشكل منفصل.
!>
!>         يجب تبديل صفوف المتجهات الفردية التي تم العثور عليها بواسطة DLASD4
!>         بنفس الطريقة قبل إجراء ضرب المصفوفات.
!>

CTOT

!>          CTOT هي مصفوفة عدد صحيح، البعد ( 4 )
!>         عدد إجمالي أنواع الأعمدة المختلفة
!>         في U (أو صفوف في VT)، كما هو موضح في IDXC. العمود الرابع
!>         من النوع هو أي عمود تم تخفيفه.
!>

Z

!>          Z هي مصفوفة DOUBLE PRECISION، البعد (K)
!>         تحتوي العناصر K الأولى لهذه المصفوفة على مكونات
!>         متجه الصف المحدث المعدل للتخفيف.
!>

INFO

!>          INFO هو عدد صحيح
!>         = 0: خروج ناجح.
!>         < 0: إذا كان INFO = -i، فقد كان الوسيط i ذو قيمة غير قانونية.
!>         > 0: إذا كان INFO = 1، لم يتقارب المتجه الفردي
!>
