توثيق الوظيفة/الإجراء الفرعي

subroutine dlaein (logical rightv, logical noinit, integer n, double precision, dimension( ldh, * ) h, integer ldh, double precision wr, double precision wi, double precision, dimension( * ) vr, double precision, dimension( * ) vi, double precision, dimension( ldb, * ) b, integer ldb, double precision, dimension( * ) work, double precision eps3, double precision smlnum, double precision bignum, integer info)

DLAEIN تحسب متجهًا ذاتيًا يمينًا أو يسارًا محددًا لمصفوفة هسنبرغ علوية باستخدام التكرار العكسي.

الغرض:

!>
!> DLAEIN تستخدم التكرار العكسي لإيجاد متجه ذاتي يميني أو يساري
!> يتوافق مع القيمة الذاتية (WR,WI) لمصفوفة هسنبرغ علوية حقيقية H.
!>

المعاملات

RIGHTV 
!>          RIGHTV هو LOGICAL (منطقي)
!>          = .TRUE. : حساب المتجه الذاتي اليميني؛
!>          = .FALSE.: حساب المتجه الذاتي اليساري.
!>

NOINIT

!>          NOINIT هو LOGICAL (منطقي)
!>          = .TRUE. : لم يتم توفير متجه ابتدائي في (VR,VI).
!>          = .FALSE.: تم توفير متجه ابتدائي في (VR,VI).
!>

N

!>          N هو INTEGER (عدد صحيح)
!>          ترتيب المصفوفة H. N >= 0.
!>

H

!>          H هي مصفوفة DOUBLE PRECISION (دقة مضاعفة)
!>          مصفوفة هسنبرغ العلوية H.
!>

LDH

!>          LDH هو INTEGER (عدد صحيح)
!>          البعد الرئيسي للمصفوفة H. LDH >= max(1,N).
!>

WR

!>          WR هو DOUBLE PRECISION (دقة مضاعفة)
!>

WI

!>          WI هو DOUBLE PRECISION (دقة مضاعفة)
!>          الأجزاء الحقيقية والتخيلية للقيمة الذاتية لـ H التي يتم
!>          حساب المتجه الذاتي اليميني أو اليساري المقابل لها.
!>

VR

!>          VR هي مصفوفة DOUBLE PRECISION (دقة مضاعفة)
!>

VI

!>          VI هي مصفوفة DOUBLE PRECISION (دقة مضاعفة)
!>          عند الإدخال، إذا كان NOINIT = .FALSE. و WI = 0.0، يجب أن تحتوي
!>          VR على متجه بداية حقيقي للتكرار العكسي باستخدام القيمة
!>          الذاتية الحقيقية WR؛ إذا كان NOINIT = .FALSE. و WI.ne.0.0،
!>          يجب أن تحتوي VR و VI على الأجزاء الحقيقية والتخيلية لمتجه
!>          بداية مركب للتكرار العكسي باستخدام القيمة الذاتية المركبة
!>          (WR,WI)؛ بخلاف ذلك، لا يلزم تعيين VR و VI.
!>          عند الخروج، إذا كان WI = 0.0 (قيمة ذاتية حقيقية)، تحتوي VR
!>          على المتجه الذاتي الحقيقي المحسوب؛ إذا كان WI.ne.0.0 (قيمة
!>          ذاتية مركبة)، تحتوي VR و VI على الأجزاء الحقيقية والتخيلية
!>          للمتجه الذاتي المركب المحسوب. يتم تسوية المتجه الذاتي بحيث
!>          يكون للمكون ذي الحجم الأكبر حجم 1؛ هنا، يتم أخذ حجم عدد
!>          مركب (x,y) على أنه |x| + |y|.
!>          لا يتم استخدام VI إذا كان WI = 0.0.
!>

B

!>          B هي مصفوفة DOUBLE PRECISION (دقة مضاعفة)
!>

LDB

!>          LDB هو INTEGER (عدد صحيح)
!>          البعد الرئيسي للمصفوفة B. LDB >= N+1.
!>

WORK

!>          WORK هي مصفوفة DOUBLE PRECISION (دقة مضاعفة)
!>

EPS3

!>          EPS3 هو DOUBLE PRECISION (دقة مضاعفة)
!>          قيمة صغيرة تعتمد على الآلة تستخدم لإزعاج القيم الذاتية
!>          القريبة، واستبدال المحاور الصفرية.
!>

SMLNUM

!>          SMLNUM هو DOUBLE PRECISION (دقة مضاعفة)
!>          قيمة قريبة من عتبة الاستنفاغ تعتمد على الآلة.
!>

BIGNUM

!>          BIGNUM هو DOUBLE PRECISION (دقة مضاعفة)
!>          قيمة قريبة من عتبة الفيضان تعتمد على الآلة.
!>

INFO

!>          INFO هو INTEGER (عدد صحيح)
!>          = 0:  خروج ناجح
!>          = 1:  لم يتقارب التكرار العكسي؛ تم تعيين VR إلى التكرار
!>                الأخير، وكذلك VI إذا كان WI.ne.0.0.
!>
