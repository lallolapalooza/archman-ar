DLARZB يطبق عاكس كتلة أو معكوسه على مصفوفة عامة.

توثيق الوظيفة/الروتين الفرعي

subroutine dlarzb (character side, character trans, character direct, character storev, integer m, integer n, integer k, integer l, double precision, dimension( ldv, * ) v, integer ldv, double precision, dimension( ldt, * ) t, integer ldt, double precision, dimension( ldc, * ) c, integer ldc, double precision, dimension( ldwork, * ) work, integer ldwork)

DLARZB يطبق عاكس كتلة H أو معكوسه H**T على مصفوفة عامة موزعة M-by-N C من اليسار أو اليمين.

الغرض:

!>
!> DLARZB يطبق عاكس كتلة حقيقي H أو معكوسه H**T على
!> مصفوفة موزعة حقيقية M-by-N C من اليسار أو اليمين.
!>
!> حاليًا، يتم دعم STOREV = 'R' و DIRECT = 'B' فقط.
!>

المعلمات

SIDE 
!>          SIDE هي CHARACTER*1
!>          = 'L': تطبيق H أو H**T من اليسار
!>          = 'R': تطبيق H أو H**T من اليمين
!>

TRANS

!>          TRANS هي CHARACTER*1
!>          = 'N': تطبيق H (بدون تبديل)
!>          = 'C': تطبيق H**T (تبديل)
!>

DIRECT

!>          DIRECT هي CHARACTER*1
!>          يشير إلى كيفية تشكيل H من حاصل ضرب عواكس أولية
!>          = 'F': H = H(1) H(2) . . . H(k) (أمامي، غير مدعوم بعد)
!>          = 'B': H = H(k) . . . H(2) H(1) (خلفي)
!>

STOREV

!>          STOREV هي CHARACTER*1
!>          يشير إلى كيفية تخزين المتجهات التي تحدد العواكس الأولية:
!>          = 'C': عمودي (غير مدعوم بعد)
!>          = 'R': صفي
!>

M

!>          M هي INTEGER
!>          عدد صفوف المصفوفة C.
!>

N

!>          N هي INTEGER
!>          عدد أعمدة المصفوفة C.
!>

K

!>          K هي INTEGER
!>          رتبة المصفوفة T (= عدد العواكس الأولية
!>          التي يحدد حاصل ضربها العاكس الكتلة).
!>

L

!>          L هي INTEGER
!>          عدد الأعمدة في المصفوفة V التي تحتوي على
!>          الجزء الهادف من عواكس هاوسولدر.
!>          إذا كانت SIDE = 'L'، فإن M >= L >= 0، إذا كانت SIDE = 'R'، فإن N >= L >= 0.
!>

V

!>          V هي مصفوفة DOUBLE PRECISION، البعد (LDV,NV).
!>          إذا كانت STOREV = 'C'، فإن NV = K؛ إذا كانت STOREV = 'R'، فإن NV = L.
!>

LDV

!>          LDV هي INTEGER
!>          البعد الرئيسي للمصفوفة V.
!>          إذا كانت STOREV = 'C'، فإن LDV >= L؛ إذا كانت STOREV = 'R'، فإن LDV >= K.
!>

T

!>          T هي مصفوفة DOUBLE PRECISION، البعد (LDT,K)
!>          المصفوفة المثلثية K-by-K T في تمثيل
!>          العاكس الكتلي.
!>

LDT

!>          LDT هي INTEGER
!>          البعد الرئيسي للمصفوفة T. LDT >= K.
!>

C

!>          C هي مصفوفة DOUBLE PRECISION، البعد (LDC,N)
!>          عند الدخول، المصفوفة C ذات البعد M-by-N.
!>          عند الخروج، يتم استبدال C بـ H*C أو H**T*C أو C*H أو C*H**T.
!>

LDC

!>          LDC هي INTEGER
!>          البعد الرئيسي للمصفوفة C. LDC >= max(1,M).
!>

WORK

!>          WORK هي مصفوفة DOUBLE PRECISION، البعد (LDWORK,K)
!>

LDWORK

!>          LDWORK هي INTEGER
!>          البعد الرئيسي للمصفوفة WORK.
!>          إذا كانت SIDE = 'L'، فإن LDWORK >= max(1,N)؛
!>          إذا كانت SIDE = 'R'، فإن LDWORK >= max(1,M).
!>
