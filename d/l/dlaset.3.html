subroutine dlaset (character uplo, integer m, integer n, double precision alpha, double precision beta, double precision, dimension( lda, * ) a, integer lda)

DLASET تهيئة العناصر غير القطرية والعناصر القطرية لمصفوفة بقيم معينة.

الغرض:

!>
!> DLASET تهيئة مصفوفة A بحجم m-by-n إلى BETA على القطر و
!> ALPHA على العناصر غير القطرية.
!>

المعاملات

UPLO 
!>          UPLO هو CHARACTER*1
!>          يحدد جزء المصفوفة A الذي سيتم تعيينه.
!>          = 'U':      الجزء العلوي المثلثي يتم تعيينه؛ الجزء السفلي تمامًا
!>                      من المصفوفة A لا يتغير.
!>          = 'L':      الجزء السفلي المثلثي يتم تعيينه؛ الجزء العلوي تمامًا
!>                      من المصفوفة A لا يتغير.
!>          بخلاف ذلك:  تُعيّن المصفوفة A بأكملها.
!>

M

!>          M هو INTEGER
!>          عدد صفوف المصفوفة A.  M >= 0.
!>

N

!>          N هو INTEGER
!>          عدد أعمدة المصفوفة A.  N >= 0.
!>

ALPHA

!>          ALPHA هو DOUBLE PRECISION
!>          الثابت الذي سيتم تعيين العناصر غير القطرية إليه.
!>

BETA

!>          BETA هو DOUBLE PRECISION
!>          الثابت الذي سيتم تعيين العناصر القطرية إليه.
!>

A

!>          A هو مصفوفة DOUBLE PRECISION، الأبعاد (LDA,N)
!>          عند الخروج، يتم تعيين الجزء العلوي m-by-n من A كما يلي:
!>
!>          إذا كان UPLO = 'U'، A(i,j) = ALPHA، 1<=i<=j-1، 1<=j<=n،
!>          إذا كان UPLO = 'L'، A(i,j) = ALPHA، j+1<=i<=m، 1<=j<=n،
!>          بخلاف ذلك،     A(i,j) = ALPHA، 1<=i<=m، 1<=j<=n، i.ne.j،
!>
!>          و، لجميع UPLO، A(i,i) = BETA، 1<=i<=min(m,n).
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرئيسي للمصفوفة A.  LDA >= max(1,M).
!>
