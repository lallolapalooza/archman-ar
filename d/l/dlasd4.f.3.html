subroutine dlasd4 (n, i, d, z, delta, rho, sigma, work, info)
DLASD4 يحسب الجذر التربيعي للقيمة الذاتية المحدثة i لمصفوفة قطرية موجبة موجبة موجبة موجبة من الرتبة الأولى. تستخدم بواسطة dbdsdc.

توثيق الوظيفة/الإجراء الفرعي

subroutine dlasd4 (integer n, integer i, double precision, dimension( * ) d, double precision, dimension( * ) z, double precision, dimension( * ) delta, double precision rho, double precision sigma, double precision, dimension( * ) work, integer info)

DLASD4 يحسب الجذر التربيعي للقيمة الذاتية المحدثة i لمصفوفة قطرية موجبة موجبة موجبة موجبة من الرتبة الأولى. تستخدم بواسطة dbdsdc.

الغرض:

!>
!> هذه الإجراء الفرعي يحسب الجذر التربيعي للقيمة الذاتية المحدثة I
!> لمصفوفة قطرية موجبة موجبة موجبة موجبة من الرتبة الأولى
!> لمصفوفة قطرية موجبة مدخلاتها معطاة كمربعات
!> للمدخلات المقابلة في المصفوفة d، و
!>
!>        0 <= D(i) < D(j)  for  i < j
!>
!> و RHO > 0. يتم ترتيب ذلك بواسطة الروتين المستدعي، وهو
!> ليس خسارة في العمومية. نظام الرتبة الأولى المحدث هو بالتالي
!>
!>        diag( D ) * diag( D ) +  RHO * Z * Z_transpose.
!>
!> حيث نفترض أن المعيار الإقليدي لـ Z هو 1.
!>
!> تتكون الطريقة من تقريب الدوال الكسرية في
!> المعادلة العلمانية بواسطة دوال كسرية استيفائية أبسط.
!>

المعلمات

N 
!>          N هو INTEGER
!>         طول جميع المصفوفات.
!>

I

!>          I هو INTEGER
!>         فهرس القيمة الذاتية المراد حسابها.  1 <= I <= N.
!>

D

!>          D هي مصفوفة DOUBLE PRECISION، البعد ( N )
!>         القيم الذاتية الأصلية.  يفترض أنها في
!>         الترتيب، 0 <= D(I) < D(J)  for I < J.
!>

Z

!>          Z هي مصفوفة DOUBLE PRECISION، البعد ( N )
!>         مكونات متجه التحديث.
!>

DELTA

!>          DELTA هي مصفوفة DOUBLE PRECISION، البعد ( N )
!>         إذا كان N .ne. 1، فإن DELTA تحتوي على (D(j) - sigma_I) في المكون j.
!>         إذا كان N = 1، فإن DELTA(1) = 1.  المتجه DELTA
!>         يحتوي على المعلومات اللازمة لبناء
!>         المتجهات الذاتية (المفردة).
!>

RHO

!>          RHO هو DOUBLE PRECISION
!>         الثابت في صيغة التحديث المتماثل.
!>

SIGMA

!>          SIGMA هو DOUBLE PRECISION
!>         القيمة sigma_I المحسوبة، القيمة الذاتية المحدثة I.
!>

WORK

!>          WORK هي مصفوفة DOUBLE PRECISION، البعد ( N )
!>         إذا كان N .ne. 1، فإن WORK تحتوي على (D(j) + sigma_I) في المكون j.
!>         إذا كان N = 1، فإن WORK( 1 ) = 1.
!>

INFO

!>          INFO هو INTEGER
!>         = 0:  انتهاء ناجح
!>         > 0:  إذا كان INFO = 1، فشلت عملية التحديث.
!>

المعلمات الداخلية:

!>  المتغير المنطقي ORGATI (origin-at-i?) يستخدم للتمييز
!>  ما إذا كان D(i) أو D(i+1) يعاملان كنقطة أصل.
!>
!>            ORGATI = .true.    نقطة الأصل عند i
!>            ORGATI = .false.   نقطة الأصل عند i+1
!>
!>  المتغير المنطقي SWTCH3 (switch-for-3-poles?) لملاحظة
!>  ما إذا كنا نعمل مع ثلاثة أقطاب!
!>
!>  MAXIT هو الحد الأقصى لعدد التكرارات المسموح بها لكل
!>  قيمة ذاتية.
!>
