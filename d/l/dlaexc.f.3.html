subroutine dlaexc (logical wantq, integer n, double precision, dimension( ldt, * ) t, integer ldt, double precision, dimension( ldq, * ) q, integer ldq, integer j1, integer n1, integer n2, double precision, dimension( * ) work, integer info)

DLAEXC تبادل كتل قطرية متجاورة لمصفوفة شبه ثلاثية علوية حقيقية في شكل كانونيكي لشتور، عن طريق تحويل تشابه متعامد.

الغرض:

!>
!> DLAEXC تبادل الكتل القطرية المتجاورة T11 و T22 من الرتبة 1 أو 2 في
!> مصفوفة شبه ثلاثية علوية T عن طريق تحويل تشابه متعامد.
!>
!> يجب أن تكون T في شكل كانونيكي لشتور، أي، كتلة ثلاثية علوية
!> بكتل قطرية 1x1 و 2x2؛ كل كتلة قطرية 2x2
!> لها عناصر قطرية متساوية وعناصرها خارج القطر ذات
!> إشارة متعاكسة.
!>

المعلمات

WANTQ 
!>          WANTQ هو LOGICAL
!>          = .TRUE.: تجميع التحويل في المصفوفة Q؛
!>          = .FALSE.: لا تجمع التحويل.
!>

N

!>          N هو INTEGER
!>          رتبة المصفوفة T. N >= 0.
!>

T

!>          T هي مصفوفة DOUBLE PRECISION، الأبعاد (LDT,N)
!>          عند الإدخال، المصفوفة شبه الثلاثية العلوية T، في شكل
!>          كانونيكي لشتور.
!>          عند الخروج، المصفوفة المحدثة T، مرة أخرى في شكل كانونيكي لشتور.
!>

LDT

!>          LDT هو INTEGER
!>          البعد الرائد للمصفوفة T. LDT >= max(1,N).
!>

Q

!>          Q هي مصفوفة DOUBLE PRECISION، الأبعاد (LDQ,N)
!>          عند الإدخال، إذا كان WANTQ هو .TRUE.، فإن المصفوفة المتعامدة Q.
!>          عند الخروج، إذا كان WANTQ هو .TRUE.، فإن المصفوفة المحدثة Q.
!>          إذا كان WANTQ هو .FALSE.، فإن Q غير مستخدمة.
!>

LDQ

!>          LDQ هو INTEGER
!>          البعد الرائد للمصفوفة Q.
!>          LDQ >= 1؛ وإذا كان WANTQ هو .TRUE.، فإن LDQ >= N.
!>

J1

!>          J1 هو INTEGER
!>          فهرس الصف الأول للكتلة الأولى T11.
!>

N1

!>          N1 هو INTEGER
!>          رتبة الكتلة الأولى T11. N1 = 0، 1 أو 2.
!>

N2

!>          N2 هو INTEGER
!>          رتبة الكتلة الثانية T22. N2 = 0، 1 أو 2.
!>

WORK

!>          WORK هي مصفوفة DOUBLE PRECISION، الأبعاد (N)
!>

INFO

!>          INFO هو INTEGER
!>          = 0: الخروج الناجح
!>          = 1: ستكون المصفوفة المحولة T بعيدة جدًا عن شكل شتور
!>               ؛ لم يتم تبديل الكتل و T و Q
!>               لم تتغير.
!>
