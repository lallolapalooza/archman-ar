subroutine dlahrd (n, k, nb, a, lda, tau, t, ldt, y, ldy)
DLAHRD تقلل أول nb عمود من مصفوفة A المستطيلة العامة بحيث تصبح العناصر الموجودة تحت القطر الفرعي k صفرًا، وتعيد مصفوفات مساعدة مطلوبة لتطبيق التحويل على الجزء غير المخفض من A.

توثيق الوظيفة/الإجراء الفرعي

subroutine dlahrd (integer n, integer k, integer nb, double precision, dimension( lda, * ) a, integer lda, double precision, dimension( nb ) tau, double precision, dimension( ldt, nb ) t, integer ldt, double precision, dimension( ldy, nb ) y, integer ldy)

DLAHRD تقلل أول nb عمود من مصفوفة A المستطيلة العامة بحيث تصبح العناصر الموجودة تحت القطر الفرعي k صفرًا، وتعيد مصفوفات مساعدة مطلوبة لتطبيق التحويل على الجزء غير المخفض من A.

الغرض:

!>
!> تم إهمال هذا الروتين وتم استبداله بالروتين DLAHR2.
!>
!> DLAHRD تقلل أول NB عمود من مصفوفة A العامة n-by-(n-k+1)
!> بحيث تصبح العناصر الموجودة تحت القطر الفرعي k صفرًا.
!> يتم إجراء التخفيض بواسطة تحويل تشابه متعامد
!> Q**T * A * Q. يعيد الروتين المصفوفات V و T التي تحدد
!> Q كمُعكِس كتلة I - V*T*V**T، وكذلك المصفوفة Y = A * V * T.
!>

التفاصيل الإضافية:

!>
!>  يتم تمثيل المصفوفة Q كحاصل ضرب لـ nb معكوس ابتدائي
!>
!>     Q = H(1) H(2) . . . H(nb).
!>
!>  كل H(i) لها الشكل
!>
!>     H(i) = I - tau * v * v**T
!>
!>  حيث tau هو عامل قياسي حقيقي، و v هو متجه حقيقي مع
!>  v(1:i+k-1) = 0، v(i+k) = 1؛ v(i+k+1:n) مخزن عند الخروج في
!>  A(i+k+1:n,i)، و tau في TAU(i).
!>
!>  عناصر المتجهات v تشكل معًا المصفوفة V ذات الأبعاد (n-k+1)-by-nb
!>  المطلوبة، مع T و Y، لتطبيق التحويل على
!>  الجزء غير المخفض من المصفوفة، باستخدام تحديث بالشكل:
!>  A := (I - V*T*V**T) * (A - Y*V**T).
!>
!>  محتويات A عند الخروج موضحة بالمثال التالي
!>  مع n = 7، k = 3 و nb = 2:
!>
!>     ( a   h   a   a   a )
!>     ( a   h   a   a   a )
!>     ( a   h   a   a   a )
!>     ( h   h   a   a   a )
!>     ( v1  h   a   a   a )
!>     ( v1  v2  a   a   a )
!>     ( v1  v2  a   a   a )
!>
!>  حيث يشير a إلى عنصر من المصفوفة الأصلية A، ويشير h إلى
!>  عنصر معدل من المصفوفة هسنبرغ العليا H، ويشير vi إلى عنصر
!>  من المتجه الذي يحدد H(i).
!>
