double precision function dla_porcond (uplo, n, a, lda, af, ldaf, cmode, c, info, work, iwork)
DLA_PORCOND تقدر رقم الشرط لـ Skeel لمصفوفة متماثلة موجبة محددة.

توثيق الوظيفة/الإجراء الفرعي

double precision function dla_porcond (character uplo, integer n, double precision, dimension( lda, * ) a, integer lda, double precision, dimension( ldaf, * ) af, integer ldaf, integer cmode, double precision, dimension( * ) c, integer info, double precision, dimension( * ) work, integer, dimension( * ) iwork)

DLA_PORCOND تقدر رقم الشرط لـ Skeel للمصفوفة المتماثلة الموجبة المحددة.

الغرض:

!>
!>    DLA_PORCOND تقدر رقم الشرط لـ Skeel لـ op(A) * op2(C)
!>    حيث يتم تحديد op2 بواسطة CMODE على النحو التالي
!>    CMODE =  1    op2(C) = C
!>    CMODE =  0    op2(C) = I
!>    CMODE = -1    op2(C) = inv(C)
!>    يتم حساب رقم الشرط لـ Skeel  cond(A) = norminf( |inv(A)||A| )
!>    عن طريق حساب عوامل القياس R بحيث
!>    diag(R)*A*op2(C) تكون متوازنة الصفوف ويتم حساب رقم الشرط القياسي
!>    لمعيار لانهائي.
!>

المعلمات

UPLO 
!>          UPLO هو CHARACTER*1
!>       = 'U':  المثلث العلوي لـ A مخزن؛
!>       = 'L':  المثلث السفلي لـ A مخزن.
!>

N

!>          N هو INTEGER
!>     عدد المعادلات الخطية، أي ترتيب
!>     المصفوفة A.  N >= 0.
!>

A

!>          A هو مصفوفة DOUBLE PRECISION، الأبعاد (LDA,N)
!>     عند الإدخال، المصفوفة N-by-N A.
!>

LDA

!>          LDA هو INTEGER
!>     البعد الرائد للمصفوفة A.  LDA >= max(1,N).
!>

AF

!>          AF هو مصفوفة DOUBLE PRECISION، الأبعاد (LDAF,N)
!>     العامل المثلثي U أو L من تحليل Cholesky
!>     A = U**T*U أو A = L*L**T، كما تم حسابه بواسطة DPOTRF.
!>

LDAF

!>          LDAF هو INTEGER
!>     البعد الرائد للمصفوفة AF.  LDAF >= max(1,N).
!>

CMODE

!>          CMODE هو INTEGER
!>     يحدد op2(C) في الصيغة op(A) * op2(C) على النحو التالي:
!>     CMODE =  1    op2(C) = C
!>     CMODE =  0    op2(C) = I
!>     CMODE = -1    op2(C) = inv(C)
!>

C

!>          C هو مصفوفة DOUBLE PRECISION، الأبعاد (N)
!>     المتجه C في الصيغة op(A) * op2(C).
!>

INFO

!>          INFO هو INTEGER
!>       = 0:  انتهاء ناجح.
!>     i > 0:  المعلمة i غير صالحة.
!>

WORK

!>          WORK هو مصفوفة DOUBLE PRECISION، الأبعاد (3*N).
!>     مساحة العمل.
!>

IWORK

!>          IWORK هو مصفوفة INTEGER، الأبعاد (N).
!>     مساحة العمل.
!>
