subroutine dla_geamv (trans, m, n, alpha, a, lda, x, incx, beta, y, incy)

  DLA_GEAMV تحسب حاصل ضرب مصفوفة ومتجه باستخدام مصفوفة عامة لحساب حدود الخطأ.

وثائق الوظيفة/الروتين الفرعي

subroutine dla_geamv (integer trans, integer m, integer n, double precision alpha, double precision, dimension( lda, * ) a, integer lda, double precision, dimension( * ) x, integer incx, double precision beta, double precision, dimension( * ) y, integer incy)

DLA_GEAMV تحسب حاصل ضرب مصفوفة ومتجه باستخدام مصفوفة عامة لحساب حدود الخطأ.

الغرض:

!>
!> DLA_GEAMV  تنفذ إحدى عمليات ضرب المصفوفة-المتجه:
!>
!>         y := alpha*abs(A)*abs(x) + beta*abs(y),
!>    أو   y := alpha*abs(A)**T*abs(x) + beta*abs(y),
!>
!> حيث alpha و beta هما ثابتان، x و y هما متجهان، و A هي مصفوفة
!> بأبعاد m × n.
!>
!> هذه الوظيفة تستخدم بشكل أساسي في حساب حدود الخطأ.
!> للحماية من النقصان أثناء التقييم، يتم إزاحة المكونات في
!> المتجه الناتج عن (N+1) أضعاف عتبة النقصان. لمنع الأخطاء الكبيرة بشكل غير ضروري
!> للهياكل الكتلية المدمجة في المصفوفات العامة،
!> لا يتم إزاحة المكونات الصفرية. تعتبر المدخلات الصفرية
!> إذا كانت جميع عمليات الضرب المتضمنة
!> في حساب هذا المدخل لديها معامل واحد على الأقل صفري.
!>

المعلمات

TRANS 
!>          TRANS هو عدد صحيح (INTEGER).
!>           عند الإدخال، تحدد TRANS العملية التي سيتم تنفيذها كما يلي:
!>
!>             BLAS_NO_TRANS      y := alpha*abs(A)*abs(x) + beta*abs(y)
!>             BLAS_TRANS         y := alpha*abs(A**T)*abs(x) + beta*abs(y)
!>             BLAS_CONJ_TRANS    y := alpha*abs(A**T)*abs(x) + beta*abs(y)
!>
!>           لا تتغير عند الخروج.
!>

M

!>          M هو عدد صحيح (INTEGER).
!>           عند الإدخال، يحدد M عدد صفوف المصفوفة A.
!>           يجب أن يكون M أكبر من أو يساوي صفر.
!>           لا يتغير عند الخروج.
!>

N

!>          N هو عدد صحيح (INTEGER).
!>           عند الإدخال، يحدد N عدد أعمدة المصفوفة A.
!>           يجب أن يكون N أكبر من أو يساوي صفر.
!>           لا يتغير عند الخروج.
!>

ALPHA

!>          ALPHA هو رقم عشري دقيق (DOUBLE PRECISION).
!>           عند الإدخال، يحدد ALPHA الثابت alpha.
!>           لا يتغير عند الخروج.
!>

A

!>          A هو مصفوفة بأرقام عشرية دقيقة (DOUBLE PRECISION array)، بأبعاد ( LDA, n ).
!>           قبل الإدخال، يجب أن يحتوي الجزء الرئيسي بأبعاد m × n من المصفوفة A على
!>           معاملات المصفوفة.
!>           لا يتغير عند الخروج.
!>

LDA

!>          LDA هو عدد صحيح (INTEGER).
!>           عند الإدخال، يحدد LDA البعد الأول للمصفوفة A كما هو معلن
!>           في البرنامج (الروتين الفرعي) المستدعي. يجب أن يكون LDA على الأقل
!>           max( 1, m ).
!>           لا يتغير عند الخروج.
!>

X

!>          X هي مصفوفة بأرقام عشرية دقيقة (DOUBLE PRECISION array)، بأبعاد
!>           ( 1 + ( n - 1 )*abs( INCX ) ) عندما TRANS = 'N' أو 'n'
!>           وعلى الأقل
!>           ( 1 + ( m - 1 )*abs( INCX ) ) في حالات أخرى.
!>           قبل الإدخال، يجب أن يحتوي المتجه X ذو الفواصل الزمنية المحددة على
!>           المتجه x.
!>           لا يتغير عند الخروج.
!>

INCX

!>          INCX هو عدد صحيح (INTEGER).
!>           عند الإدخال، يحدد INCX الفاصل الزمني لعناصر X. يجب أن لا يكون INCX صفرًا.
!>           لا يتغير عند الخروج.
!>

BETA

!>          BETA هو رقم عشري دقيق (DOUBLE PRECISION).
!>           عند الإدخال، يحدد BETA الثابت beta. عندما يكون BETA
!>           مقدمًا كصفر، فلا يلزم تعيين Y عند الإدخال.
!>           لا يتغير عند الخروج.
!>

Y

!>          Y هي مصفوفة بأرقام عشرية دقيقة (DOUBLE PRECISION array)،
!>           بأبعاد على الأقل
!>           ( 1 + ( m - 1 )*abs( INCY ) ) عندما TRANS = 'N' أو 'n'
!>           وعلى الأقل
!>           ( 1 + ( n - 1 )*abs( INCY ) ) في حالات أخرى.
!>           قبل الإدخال عندما يكون BETA غير صفري، يجب أن يحتوي المتجه Y ذو الفواصل الزمنية المحددة على
!>           المتجه y. عند الخروج، يتم استبدال Y بـ
!>           المتجه المحدث y.
!>           إذا كان m أو n صفراً، فإن Y لا يتم الرجوع إليه وتقوم الوظيفة
!>           بالعودة السريعة.
!>

INCY

!>          INCY هو عدد صحيح (INTEGER).
!>           عند الإدخال، يحدد INCY الفاصل الزمني لعناصر Y. يجب أن لا يكون INCY صفرًا.
!>           لا يتغير عند الخروج.
!>
!>  روتين فرعي من المستوى 2 من BLAS.
!>
