توثيق الوظيفة/الروتين الفرعي

subroutine dlalsd (character uplo, integer smlsiz, integer n, integer nrhs, double precision, dimension( * ) d, double precision, dimension( * ) e, double precision, dimension( ldb, * ) b, integer ldb, double precision rcond, integer rank, double precision, dimension( * ) work, integer, dimension( * ) iwork, integer info)

DLALSD يستخدم التحلل القيمي المنفرد للمصفوفة A لحل مشكلة المربعات الصغرى.

الغرض:

!>
!> DLALSD يستخدم التحلل القيمي المنفرد للمصفوفة A لحل مشكلة المربعات الصغرى
!> لإيجاد X التي تقلل من المعيار الإقليدي لكل عمود في A*X-B، حيث A مصفوفة
!> علوية ثنائية القطر بحجم N×N، و X و B بحجم N×NRHS. الحل X يستبدل B.
!>
!> القيم المنفردة لـ A الأصغر من RCOND مضروبة في أكبر قيمة منفردة
!> تُعامل على أنها صفر في حل مشكلة المربعات الصغرى؛ في هذه الحالة يتم إرجاع
!> حل بأقل معيار. القيم المنفردة الفعلية تُرجع في D بترتيب تصاعدي.
!>
!>

المعلمات

UPLO 
!>          UPLO is CHARACTER*1
!>         = 'U': D و E يحددان مصفوفة علوية ثنائية القطر.
!>         = 'L': D و E يحددان مصفوفة سفلية ثنائية القطر.
!>

SMLSIZ

!>          SMLSIZ is INTEGER
!>         الحد الأقصى لحجم المشكلات الفرعية في أسفل شجرة الحساب.
!>

N

!>          N is INTEGER
!>         بعد المصفوفة ثنائية القطر. N >= 0.
!>

NRHS

!>          NRHS is INTEGER
!>         عدد أعمدة B. يجب أن يكون NRHS على الأقل 1.
!>

D

!>          D is DOUBLE PRECISION array, dimension (N)
!>         عند الدخول، يحتوي D على القطر الرئيسي للمصفوفة ثنائية القطر.
!>         عند الخروج، إذا كان INFO = 0، يحتوي D على قيمه المنفردة.
!>

E

!>          E is DOUBLE PRECISION array, dimension (N-1)
!>         يحتوي على عناصر القطر العلوي للمصفوفة ثنائية القطر.
!>         عند الخروج، تم تدمير E.
!>

B

!>          B is DOUBLE PRECISION array, dimension (LDB,NRHS)
!>         عند الدخول، يحتوي B على الجوانب اليمنى لمشكلة المربعات الصغرى.
!>         عند الخروج، يحتوي B على الحل X.
!>

LDB

!>          LDB is INTEGER
!>         البعد الرئيسي لـ B في البرنامج الفرعي المستدعي.
!>         يجب أن يكون LDB على الأقل max(1,N).
!>

RCOND

!>          RCOND is DOUBLE PRECISION
!>         القيم المنفردة لـ A التي تقل عن أو تساوي RCOND مضروبة في
!>         أكبر قيمة منفردة تُعامل على أنها صفر في حل مشكلة المربعات الصغرى.
!>         إذا كان RCOND سالبًا، يتم استخدام الدقة الآلية بدلاً من ذلك.
!>         على سبيل المثال، إذا كانت diag(S)*X=B هي مشكلة المربعات الصغرى،
!>         حيث diag(S) هي مصفوفة قطرية للقيم المنفردة، سيكون الحل
!>         X(i) = B(i) / S(i) إذا كانت S(i) أكبر من RCOND*max(S)،
!>         و X(i) = 0 إذا كانت S(i) أقل من أو تساوي RCOND*max(S).
!>

RANK

!>          RANK is INTEGER
!>         عدد القيم المنفردة لـ A الأكبر من RCOND مضروبة في أكبر قيمة منفردة.
!>

WORK

!>          WORK is DOUBLE PRECISION array, dimension at least
!>         (9*N + 2*N*SMLSIZ + 8*N*NLVL + N*NRHS + (SMLSIZ+1)**2),
!>         حيث NLVL = max(0, INT(log_2 (N/(SMLSIZ+1))) + 1).
!>

IWORK

!>          IWORK is INTEGER array, dimension at least
!>         (3*N*NLVL + 11*N)
!>

INFO

!>          INFO is INTEGER
!>         = 0:  انتهاء ناجح.
!>         < 0:  إذا كان INFO = -i، فإن المعلمة i كانت لها قيمة غير قانونية.
!>         > 0:  فشل الخوارزمية في حساب قيمة منفردة أثناء العمل على
!>               المصفوفة الفرعية الموجودة في الصفوف والأعمدة
!>               INFO/(N+1) حتى MOD(INFO,N+1).
!>
