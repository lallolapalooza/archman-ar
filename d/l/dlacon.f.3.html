توثيق الوظيفة/الروتين الفرعي

subroutine dlacon (integer n, double precision, dimension( * ) v, double precision, dimension( * ) x, integer, dimension( * ) isgn, double precision est, integer kase)

DLACON يقدّر معيار 1 لمصفوفة مربعة حقيقية، باستخدام الاتصال العكسي لتقييم ضربات المتجه المصفوفي.

الغرض:

!>
!> DLACON يقدر معيار 1 لمصفوفة حقيقية مربعة A.
!> يتم استخدام الاتصال العكسي لتقييم ضربات المتجه المصفوفي.
!>

المعاملات

N 
!>          N هو عدد صحيح (INTEGER).
!>         رتبة المصفوفة. N >= 1.
!>

V

!>          V هي مصفوفة من نوع DOUBLE PRECISION، أبعادها (N).
!>         عند العودة النهائية، V = A*W، حيث EST = norm(V)/norm(W)
!>         (W لا يتم إرجاعها).
!>

X

!>          X هي مصفوفة من نوع DOUBLE PRECISION، أبعادها (N).
!>         عند العودة المتوسطة، يجب استبدال X بـ
!>               A * X،   إذا كان KASE=1،
!>               A**T * X،  إذا كان KASE=2،
!>         ويجب إعادة استدعاء DLACON مع جميع المعاملات الأخرى
!>         دون تغيير.
!>

ISGN

!>          ISGN هي مصفوفة من نوع INTEGER، أبعادها (N).
!>

EST

!>          EST هو من نوع DOUBLE PRECISION.
!>         عند الدخول مع KASE = 1 أو 2 و JUMP = 3، يجب أن يكون EST
!>         دون تغيير عن الاستدعاء السابق لـ DLACON.
!>         عند الخروج، EST هو تقدير (حد أدنى) لـ norm(A).
!>

KASE

!>          KASE هو عدد صحيح (INTEGER).
!>         عند الاستدعاء الأولي لـ DLACON، يجب أن يكون KASE = 0.
!>         عند العودة المتوسطة، سيكون KASE = 1 أو 2، مشيرًا
!>         إلى ما إذا كان يجب استبدال X بـ A * X  أو A**T * X.
!>         عند العودة النهائية من DLACON، سيكون KASE = 0 مرة أخرى.
!>
