subroutine dlaeda (n, tlvls, curlvl, curpbm, prmptr, perm, givptr, givcol, givnum, q, qptr, z, ztemp, info)
DLAEDA تستخدم بواسطة DSTEDC. تحسب متجه Z الذي يحدد التعديل رتبة-واحد للمصفوفة القطرية. تستخدم عندما تكون المصفوفة الأصلية كثيفة.

توثيق الوظيفة/الإجراء الفرعي

subroutine dlaeda (integer n, integer tlvls, integer curlvl, integer curpbm, integer, dimension( * ) prmptr, integer, dimension( * ) perm, integer, dimension( * ) givptr, integer, dimension( 2, * ) givcol, double precision, dimension( 2, * ) givnum, double precision, dimension( * ) q, integer, dimension( * ) qptr, double precision, dimension( * ) z, double precision, dimension( * ) ztemp, integer info)

DLAEDA تستخدم بواسطة DSTEDC. تحسب متجه Z الذي يحدد التعديل رتبة-واحد للمصفوفة القطرية. تستخدم عندما تكون المصفوفة الأصلية كثيفة.

الغرض:

!>
!> DLAEDA تحسب متجه Z المقابل لخطوة الدمج في خطوة CURLVLth من عملية الدمج مع TLVLS خطوة في مشكلة CURPBMth.
!>

المعاملات

N 
!>          N هو عدد صحيح
!>         بعد المصفوفة الثلاثية المتماثلة. N >= 0.
!>

TLVLS

!>          TLVLS هو عدد صحيح
!>         إجمالي عدد مستويات الدمج في شجرة التقسيم والحل الشاملة.
!>

CURLVL

!>          CURLVL هو عدد صحيح
!>         المستوى الحالي في روتين الدمج الشامل،
!>         0 <= curlvl <= tlvls.
!>

CURPBM

!>          CURPBM هو عدد صحيح
!>         المشكلة الحالية في المستوى الحالي في الروتين الشامل للدمج
!>         (العد من أعلى اليسار إلى أسفل اليمين).
!>

PRMPTR

!>          PRMPTR هو مصفوفة أعداد صحيحة، ذات بعد (N lg N)
!>         تحتوي على قائمة بالمؤشرات التي تشير إلى مكان تخزين تبديل المستوى في PERM.
!>         PRMPTR(i+1) - PRMPTR(i) يشير إلى حجم التبديل وبشكل عرضي حجم المشكلة الكاملة غير المحددة.
!>

PERM

!>          PERM هي مصفوفة أعداد صحيحة، ذات بعد (N lg N)
!>         تحتوي على التباديل (من التحديد والفرز) التي سيتم تطبيقها على كل كتلة قيمة ذاتية.
!>

GIVPTR

!>          GIVPTR هي مصفوفة أعداد صحيحة، ذات بعد (N lg N)
!>         تحتوي على قائمة بالمؤشرات التي تشير إلى مكان تخزين دورانات جيفنز للمستوى في GIVCOL.
!>         GIVPTR(i+1) - GIVPTR(i) يشير إلى عدد دورانات جيفنز.
!>

GIVCOL

!>          GIVCOL هي مصفوفة أعداد صحيحة، ذات بعد (2, N lg N)
!>         كل زوج من الأرقام يشير إلى زوج من الأعمدة التي ستحدث فيها دورانات جيفنز.
!>

GIVNUM

!>          GIVNUM هي مصفوفة دقة مزدوجة، ذات بعد (2, N lg N)
!>         كل رقم يشير إلى قيمة S التي سيتم استخدامها في دورانات جيفنز المقابلة.
!>

Q

!>          Q هي مصفوفة دقة مزدوجة، ذات بعد (N**2)
!>         تحتوي على كتل القيم الذاتية المربعة من المستويات السابقة،
!>         يتم إعطاء مواضع البداية للكتل بواسطة QPTR.
!>

QPTR

!>          QPTR هي مصفوفة أعداد صحيحة، ذات بعد (N+2)
!>         تحتوي على قائمة بالمؤشرات التي تشير إلى مكان تخزين كتلة القيم الذاتية في Q.
!>         SQRT( QPTR(i+1) - QPTR(i) ) تشير إلى حجم الكتلة.
!>

Z

!>          Z هي مصفوفة دقة مزدوجة، ذات بعد (N)
!>         عند الإخراج، يحتوي هذا المتجه على المتجه المحدث (الصف الأخير من مصفوفة المتجهات الذاتية الفرعية الأولى والصف الأول
!>         من مصفوفة المتجهات الذاتية الفرعية الثانية).
!>

ZTEMP

!>          ZTEMP هي مصفوفة دقة مزدوجة، ذات بعد (N)
!>

INFO

!>          INFO هو عدد صحيح
!>          = 0: خروج ناجح.
!>          < 0: إذا كان INFO = -i، فإن المعامل i كان له قيمة غير قانونية.
!>
