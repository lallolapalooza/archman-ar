توثيق الوظيفة/الروتين الفرعي

subroutine dlaein (logical rightv, logical noinit, integer n, double precision, dimension( ldh, * ) h, integer ldh, double precision wr, double precision wi, double precision, dimension( * ) vr, double precision, dimension( * ) vi, double precision, dimension( ldb, * ) b, integer ldb, double precision, dimension( * ) work, double precision eps3, double precision smlnum, double precision bignum, integer info)

DLAEIN تحسب متجهًا خاصًا أيمن أو أيسر محددًا لمصفوفة هسنبرغ علوية باستخدام التكرار العكسي.

الغرض:

!>
!> DLAEIN يستخدم التكرار العكسي لإيجاد متجه خاص أيمن أو أيسر
!> المقابل للقيمة الذاتية (WR,WI) لمصفوفة هسنبرغ علوية حقيقية H.
!>

المعلمات

RIGHTV 
!>          RIGHTV هو LOGICAL
!>          = .TRUE. : حساب المتجه الخاص الأيمن؛
!>          = .FALSE.: حساب المتجه الخاص الأيسر.
!>

NOINIT

!>          NOINIT هو LOGICAL
!>          = .TRUE. : لم يتم توفير متجه أولي في (VR,VI).
!>          = .FALSE.: تم توفير متجه أولي في (VR,VI).
!>

N

!>          N هو INTEGER
!>          رتبة المصفوفة H.  N >= 0.
!>

H

!>          H هو مصفوفة DOUBLE PRECISION، البعد (LDH,N)
!>          مصفوفة هسنبرغ العلوية H.
!>

LDH

!>          LDH هو INTEGER
!>          البعد الرئيسي للمصفوفة H.  LDH >= max(1,N).
!>

WR

!>          WR هو DOUBLE PRECISION
!>

WI

!>          WI هو DOUBLE PRECISION
!>          الأجزاء الحقيقية والخيالية للقيمة الذاتية لـ H التي
!>          يجب حساب المتجه الخاص الأيمن أو الأيسر المقابل لها.
!>

VR

!>          VR هو مصفوفة DOUBLE PRECISION، البعد (N)
!>

VI

!>          VI هو مصفوفة DOUBLE PRECISION، البعد (N)
!>          عند الإدخال، إذا كان NOINIT = .FALSE. و WI = 0.0، يجب أن تحتوي VR
!>          على متجه بداية حقيقي للتكرار العكسي باستخدام القيمة الذاتية الحقيقية WR؛
!>          إذا كان NOINIT = .FALSE. و WI.ne.0.0، يجب أن تحتوي VR و VI
!>          على الأجزاء الحقيقية والخيالية لمتجه بداية مركب للتكرار العكسي
!>          باستخدام القيمة الذاتية المركبة (WR,WI)؛ بخلاف ذلك، لا يلزم تعيين VR و VI.
!>          عند الخروج، إذا كان WI = 0.0 (قيمة ذاتية حقيقية)، تحتوي VR على
!>          المتجه الخاص الحقيقي المحسوب؛ إذا كان WI.ne.0.0 (قيمة ذاتية مركبة)،
!>          تحتوي VR و VI على الأجزاء الحقيقية والخيالية للمتجه الخاص المركب
!>          المحسوب. يتم تطبيع المتجه الخاص بحيث يكون للمكون ذي القيمة المطلقة الأكبر قيمة مطلقة 1؛
!>          هنا، القيمة المطلقة لعدد مركب (x,y) تُعتبر |x| + |y|.
!>          لا يتم الرجوع إلى VI إذا كان WI = 0.0.
!>

B

!>          B هو مصفوفة DOUBLE PRECISION، البعد (LDB,N)
!>

LDB

!>          LDB هو INTEGER
!>          البعد الرئيسي للمصفوفة B.  LDB >= N+1.
!>

WORK

!>          WORK هو مصفوفة DOUBLE PRECISION، البعد (N)
!>

EPS3

!>          EPS3 هو DOUBLE PRECISION
!>          قيمة صغيرة تعتمد على الآلة تُستخدم لإزاحة القيم الذاتية القريبة،
!>          واستبدال الأصفار المحورية.
!>

SMLNUM

!>          SMLNUM هو DOUBLE PRECISION
!>          قيمة قريبة من عتبة التقارب الأدنى، تعتمد على الآلة.
!>

BIGNUM

!>          BIGNUM هو DOUBLE PRECISION
!>          قيمة قريبة من عتبة تجاوز الحد، تعتمد على الآلة.
!>

INFO

!>          INFO هو INTEGER
!>          = 0:  إنهاء ناجح
!>          = 1:  لم يتقارب التكرار العكسي؛ تم تعيين VR إلى التكرار الأخير،
!>                وكذلك VI إذا كان WI.ne.0.0.
!>
