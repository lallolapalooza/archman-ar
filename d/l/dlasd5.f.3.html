subroutine dlasd5 (i, d, z, delta, rho, dsigma, work)
DLASD5 تحسب الجذر التربيعي للقيمة الذاتية i-th لتعديل موجب متناظر من الرتبة واحد لمصفوفة قطرية 2x2. تستخدم بواسطة sbdsdc.

توثيق الوظيفة/الإجراء الفرعي

subroutine dlasd5 (integer i, double precision, dimension( 2 ) d, double precision, dimension( 2 ) z, double precision, dimension( 2 ) delta, double precision rho, double precision dsigma, double precision, dimension( 2 ) work)

DLASD5 تحسب الجذر التربيعي للقيمة الذاتية i-th لتعديل موجب متناظر من الرتبة واحد لمصفوفة قطرية 2x2. تستخدم بواسطة sbdsdc.

الغرض:

!>
!> هذه الإجراء الفرعي يحسب الجذر التربيعي للقيمة الذاتية I-th
!> لتعديل موجب متناظر من الرتبة واحد لمصفوفة قطرية 2x2
!>
!>            diag( D ) * diag( D ) +  RHO * Z * transpose(Z) .
!>
!> يُفترض أن المدخلات القطرية في المصفوفة D تحقق
!>
!>            0 <= D(i) < D(j)  من أجل  i < j .
!>
!> نفترض أيضًا أن RHO > 0 وأن المعيار الإقليدي للمتجه
!> Z هو واحد.
!>

المعلمات

I 
!>          I هو عدد صحيح (INTEGER)
!>         فهرس القيمة الذاتية المراد حسابها.  I = 1 أو I = 2.
!>

D

!>          D هي مصفوفة من النوع DOUBLE PRECISION، ذات بعد ( 2 )
!>         القيم الذاتية الأصلية. نفترض أن 0 <= D(1) < D(2).
!>

Z

!>          Z هي مصفوفة من النوع DOUBLE PRECISION، ذات بعد ( 2 )
!>         مكونات المتجه التحديثي.
!>

DELTA

!>          DELTA هي مصفوفة من النوع DOUBLE PRECISION، ذات بعد ( 2 )
!>         تحتوي على (D(j) - sigma_I) في المكون j.
!>         المتجه DELTA يحتوي على المعلومات اللازمة
!>         لبناء المتجهات الذاتية.
!>

RHO

!>          RHO هو النوع DOUBLE PRECISION
!>         المقياس في صيغة التحديث المتناظر.
!>

DSIGMA

!>          DSIGMA هو النوع DOUBLE PRECISION
!>         القيمة sigma_I المحسوبة، القيمة الذاتية المحدثة I-th.
!>

WORK

!>          WORK هي مصفوفة من النوع DOUBLE PRECISION، ذات بعد ( 2 )
!>         تحتوي WORK على (D(j) + sigma_I) في المكون j.
!>
