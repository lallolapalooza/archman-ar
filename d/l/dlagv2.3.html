subroutine dlagv2 (double precision, dimension( lda, * ) a, integer lda, double precision, dimension( ldb, * ) b, integer ldb, double precision, dimension( 2 ) alphar, double precision, dimension( 2 ) alphai, double precision, dimension( 2 ) beta, double precision csl, double precision snl, double precision csr, double precision snr)

DLAGV2 تحسب تحليل Schur المعمم لمصفوفة قلمية حقيقية 2x2 (A,B) حيث B مثلثية عليا.

الغرض:

!>
!> DLAGV2 تحسب تحليل Schur المعمم لمصفوفة قلمية حقيقية 2x2
!> (A,B) حيث B مثلثية عليا. هذا الروتين
!> يحسب مصفوفتي دوران متعامدتين (CSL, SNL و CSR, SNR) بحيث
!>
!> 1) إذا كان للقلم (A,B) قيمان ذاتيان حقيقيان (بما في ذلك الأنواع 0/0 أو 1/0)
!>    حينها،
!>
!>    [ a11 a12 ] := [  CSL  SNL ] [ a11 a12 ] [  CSR -SNR ]
!>    [  0  a22 ]    [ -SNL  CSL ] [ a21 a22 ] [  SNR  CSR ]
!>
!>    [ b11 b12 ] := [  CSL  SNL ] [ b11 b12 ] [  CSR -SNR ]
!>    [  0  b22 ]    [ -SNL  CSL ] [  0  b22 ] [  SNR  CSR ],
!>
!> 2) إذا كان للقلم (A,B) زوج من القيم الذاتية المترافقة المركبة،
!>    حينها
!>
!>    [ a11 a12 ] := [  CSL  SNL ] [ a11 a12 ] [  CSR -SNR ]
!>    [ a21 a22 ]    [ -SNL  CSL ] [ a21 a22 ] [  SNR  CSR ]
!>
!>    [ b11  0  ] := [  CSL  SNL ] [ b11 b12 ] [  CSR -SNR ]
!>    [  0  b22 ]    [ -SNL  CSL ] [  0  b22 ] [  SNR  CSR ]
!>
!>    حيث b11 >= b22 > 0.
!>
!>

المعلمات

A 
!>          A هي مصفوفة DOUBLE PRECISION، بأبعاد (LDA, 2)
!>          عند الدخول، المصفوفة A بالحجم 2x2.
!>          عند الخروج، يتم استبدال A بـ "جزء A" من نموذج Schur المعمم.
!>

LDA

!>          LDA هو عدد صحيح (INTEGER).
!>          البعد الرئيسي للمصفوفة A. LDA >= 2.
!>

B

!>          B هي مصفوفة DOUBLE PRECISION، بأبعاد (LDB, 2).
!>          عند الدخول، المصفوفة B المثلثية العليا بالحجم 2x2.
!>          عند الخروج، يتم استبدال B بـ "جزء B" من نموذج Schur المعمم.
!>

LDB

!>          LDB هو عدد صحيح (INTEGER).
!>          البعد الرئيسي للمصفوفة B. LDB >= 2.
!>

ALPHAR

!>          ALPHAR هي مصفوفة DOUBLE PRECISION، بأبعاد (2).
!>

ALPHAI

!>          ALPHAI هي مصفوفة DOUBLE PRECISION، بأبعاد (2).
!>

BETA

!>          BETA هي مصفوفة DOUBLE PRECISION، بأبعاد (2).
!>          (ALPHAR(k)+i*ALPHAI(k))/BETA(k) هي القيم الذاتية للقلم (A,B)، k=1,2، i = sqrt(-1).
!>          لاحظ أن BETA(k) قد تكون صفرًا.
!>

CSL

!>          CSL هو DOUBLE PRECISION.
!>          جيب تمام (cosine) مصفوفة الدوران اليسرى.
!>

SNL

!>          SNL هو DOUBLE PRECISION.
!>          جيب (sine) مصفوفة الدوران اليسرى.
!>

CSR

!>          CSR هو DOUBLE PRECISION.
!>          جيب تمام (cosine) مصفوفة الدوران اليمنى.
!>

SNR

!>          SNR هو DOUBLE PRECISION.
!>          جيب (sine) مصفوفة الدوران اليمنى.
!>
