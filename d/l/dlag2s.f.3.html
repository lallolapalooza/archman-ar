subroutine dlag2s (m, n, a, lda, sa, ldsa, info)
DLAG2S تحول مصفوفة دقة مزدوجة إلى مصفوفة دقة واحدة.

توثيق الوظيفة/الإجراء الفرعي

subroutine dlag2s (integer m, integer n, double precision, dimension( lda, * ) a, integer lda, real, dimension( ldsa, * ) sa, integer ldsa, integer info)

DLAG2S تحول مصفوفة دقة مزدوجة إلى مصفوفة دقة واحدة.

الغرض:

!>
!> DLAG2S تحول مصفوفة DOUBLE PRECISION، A، إلى مصفوفة
!> SINGLE PRECISION، SA.
!>
!> RMAX هو تجاوز الحساب للعمليات الحسابية بـ SINGLE PRECISION
!> تتحقق DLAG2S من أن جميع مدخلات A تقع بين -RMAX و
!> RMAX. إذا لم يكن الأمر كذلك، يتم إيقاف التحويل وإثارة علامة.
!>
!> هذا إجراء فرعي مساعد لذلك لا يوجد فحص للمعاملات.
!>

المعاملات

M 
!>          M هو عدد صحيح
!>          عدد صفوف المصفوفة A.  M >= 0.
!>

N

!>          N هو عدد صحيح
!>          عدد أعمدة المصفوفة A.  N >= 0.
!>

A

!>          A هي مصفوفة DOUBLE PRECISION، البعد (LDA,N)
!>          عند الإدخال، مصفوفة المعاملات A ذات الأبعاد M×N.
!>

LDA

!>          LDA هو عدد صحيح
!>          البعد الرائد للمصفوفة A.  LDA >= max(1,M).
!>

SA

!>          SA هي مصفوفة REAL، البعد (LDSA,N)
!>          عند الخروج، إذا كان INFO=0، فإن مصفوفة المعاملات SA ذات الأبعاد M×N؛
!>          إذا كان INFO>0، فإن محتوى SA غير محدد.
!>

LDSA

!>          LDSA هو عدد صحيح
!>          البعد الرائد للمصفوفة SA.  LDSA >= max(1,M).
!>

INFO

!>          INFO هو عدد صحيح
!>          = 0:  انتهاء ناجح.
!>          = 1:  إدخال في المصفوفة A أكبر من حد تجاوز SINGLE PRECISION،
!>                في هذه الحالة، محتوى SA عند الخروج غير محدد.
!>
