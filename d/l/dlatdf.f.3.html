subroutine dlatdf (ijob, n, z, ldz, rhs, rdsum, rdscal, ipiv, jpiv)


DLATDF يستخدم تحليل LU للمصفوفة n × n التي تم حسابها بواسطة sgetc2 ويحسب مساهمة في تقدير Dif العكسي.

توثيق الوظيفة/الإجراء الفرعي

subroutine dlatdf (integer ijob, integer n, double precision, dimension( ldz, * ) z, integer ldz, double precision, dimension( * ) rhs, double precision rdsum, double precision rdscal, integer, dimension( * ) ipiv, integer, dimension( * ) jpiv)

DLATDF يستخدم تحليل LU للمصفوفة n × n التي تم حسابها بواسطة DGETC2 ويحسب مساهمة في تقدير Dif العكسي عن طريق حل Z * x = b لـ x ، واختيار b في الطرف الأيمن بحيث يكون المعيار العكسي لـ x أكبر ما يمكن. عند الدخول، RHS = b يحتوي على المساهمة من الأنظمة الفرعية التي تم حلها سابقًا، وعند الخروج RHS = x.

تحليل Z الذي أعاده DGETC2 له الشكل Z = P * L * U * Q، حيث P و Q هما مصفوفتان للتبديل. L هي مصفوفة مثلثة سفلى مع عناصر قطرية وحدوية، و U هي مصفوفة مثلثة عليا.

الغرض:

!>
!> DLATDF يستخدم تحليل LU للمصفوفة n × n Z التي تم حسابها بواسطة
!> DGETC2 ويحسب مساهمة في تقدير Dif العكسي
!> عن طريق حل Z * x = b لـ x ، واختيار الطرف الأيمن b بحيث
!> يكون المعيار العكسي لـ x أكبر ما يمكن. عند الدخول RHS = b يحتوي على
!> المساهمة من الأنظمة الفرعية التي تم حلها سابقًا، وعند الخروج RHS = x.
!>
!> تحليل Z الذي أعاده DGETC2 له الشكل Z = P*L*U*Q،
!> حيث P و Q هما مصفوفتان للتبديل. L هي مصفوفة مثلثة سفلى مع
!> عناصر قطرية وحدوية و U هي مصفوفة مثلثة عليا.
!>

المعلمات

IJOB 
!>          IJOB هو عدد صحيح (INTEGER)
!>          IJOB = 2: أولاً، يتم حساب متجه صفري تقريبي e
!>              لـ Z باستخدام DGECON، ويتم تطبيع e وحل
!>              Zx = +-e - f مع الإشارة التي تعطي قيمة أكبر
!>              للمعيار الثنائي (2-norm) لـ x. التكلفة حوالي 5 مرات
!>              أعلى من الافتراضي.
!>          IJOB .ne. 2: استراتيجية انتظار محلية حيث يتم اختيار كل مدخل
!>              للطرف الأيمن b ليكون إما +1 أو -1 (الافتراضي).
!>

N

!>          N هو عدد صحيح (INTEGER)
!>          عدد أعمدة المصفوفة Z.
!>

Z

!>          Z هي مصفوفة من نوع DOUBLE PRECISION، بأبعاد (LDZ, N)
!>          عند الدخول، الجزء LU من تحليل المصفوفة n × n
!>          Z التي تم حسابها بواسطة DGETC2:  Z = P * L * U * Q
!>

LDZ

!>          LDZ هو عدد صحيح (INTEGER)
!>          البعد الرائد للمصفوفة Z.  LDA >= max(1, N).
!>

RHS

!>          RHS هي مصفوفة من نوع DOUBLE PRECISION، بأبعاد (N)
!>          عند الدخول، يحتوي RHS على مساهمات من أنظمة فرعية أخرى.
!>          عند الخروج، يحتوي RHS على حل النظام الفرعي مع
!>          المدخلات وفقًا لقيمة IJOB (انظر أعلاه).
!>

RDSUM

!>          RDSUM هو DOUBLE PRECISION
!>          عند الدخول، مجموع مربعات المساهمات المحسوبة لـ
!>          تقدير Dif الذي يتم حسابه بواسطة DTGSYL، حيث تم
!>          استبعاد عامل القياس RDSCAL (انظر أدناه).
!>          عند الخروج، مجموع المربعات المقابل المحدث بـ
!>          المساهمات من النظام الفرعي الحالي.
!>          إذا كان TRANS = 'T'، فلا يتم لمس RDSUM.
!>          ملاحظة: RDSUM يكون منطقيًا فقط عندما يتم استدعاء DTGSY2 بواسطة
!>          STGSYL.
!>

RDSCAL

!>          RDSCAL هو DOUBLE PRECISION
!>          عند الدخول، عامل القياس المستخدم لمنع تجاوز التدفق في RDSUM.
!>          عند الخروج، يتم تحديث RDSCAL فيما يتعلق بالمساهمات
!>          الحالية في RDSUM.
!>          إذا كان TRANS = 'T'، فلا يتم لمس RDSCAL.
!>          ملاحظة: RDSCAL يكون منطقيًا فقط عندما يتم استدعاء DTGSY2 بواسطة
!>                DTGSYL.
!>

IPIV

!>          IPIV هي مصفوفة من نوع INTEGER، بأبعاد (N).
!>          فهارس المحور؛ لـ 1 <= i <= N، تم تبديل الصف i في
!>          المصفوفة مع الصف IPIV(i).
!>

JPIV

!>          JPIV هي مصفوفة من نوع INTEGER، بأبعاد (N).
!>          فهارس المحور؛ لـ 1 <= j <= N، تم تبديل العمود j في
!>          المصفوفة مع العمود JPIV(j).
!>
