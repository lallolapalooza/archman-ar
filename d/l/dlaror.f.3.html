توثيق الوظيفة/الروتين الفرعي

subroutine dlaror (character side, character init, integer m, integer n, double precision, dimension( lda, * ) a, integer lda, integer, dimension( 4 ) iseed, double precision, dimension( * ) x, integer info)

DLAROR

الغرض:

!>
!> تقوم DLAROR بضرب المصفوفة A ذات الأبعاد M × N مسبقًا أو لاحقًا بمصفوفة متعامدة عشوائية U، مع الكتابة فوق A. قد يتم تهيئة A اختياريًا إلى مصفوفة الوحدة قبل الضرب بـ U. يتم إنشاء U باستخدام
!> طريقة G.W. Stewart (SIAM J. Numer. Anal. 17، 1980، 403-409).
!>

المعاملات

SIDE 
!>          SIDE is CHARACTER*1
!>          يحدد ما إذا كانت A مضروبة من اليسار أو اليمين بواسطة U.
!>          = 'L':         اضرب A من اليسار (ضرب مسبق) بواسطة U
!>          = 'R':         اضرب A من اليمين (ضرب لاحق) بواسطة U'
!>          = 'C' or 'T':  اضرب A من اليسار بواسطة U ومن اليمين
!>                          بواسطة U' (هنا، U' تعني U-transpose.)
!>

INIT

!>          INIT is CHARACTER*1
!>          يحدد ما إذا كان يجب تهيئة A إلى
!>          مصفوفة الوحدة أم لا.
!>          = 'I':  قم بتهيئة A إلى (قسم من) مصفوفة الوحدة
!>                   قبل تطبيق U.
!>          = 'N':  لا تهيئة.  طبق U على المصفوفة المدخلة A.
!>
!>          يمكن استخدام INIT = 'I' لإنشاء مصفوفات متعامدة مربعة أو مستطيلة
!>          مصفوفات:
!>
!>          بالنسبة لـ M = N و SIDE = 'L' أو 'R'، ستكون الصفوف متعامدة
!>          على بعضها البعض، وكذلك الأعمدة.
!>
!>          إذا كان M < N، ينتج SIDE = 'R' مصفوفة كثيفة صفوفها متعامدة
!>          وأعمدتها ليست كذلك، بينما SIDE = 'L'
!>          ينتج مصفوفة صفوفها متعامدة، وأول
!>          M أعمدة متعامدة، وأعمدتها المتبقية
!>          صفر.
!>
!>          إذا كان M > N، ينتج SIDE = 'L' مصفوفة كثيفة أعمدتها
!>          متعامدة وصفوفها ليست كذلك، بينما SIDE = 'R'
!>          ينتج مصفوفة أعمدتها متعامدة، وأول
!>          M صفوف متعامدة، وصفوفها المتبقية
!>          صفر.
!>

M

!>          M is INTEGER
!>          عدد صفوف A.
!>

N

!>          N is INTEGER
!>          عدد أعمدة A.
!>

A

!>          A is DOUBLE PRECISION array, dimension (LDA, N)
!>          عند الإدخال، المصفوفة A.
!>          عند الخروج، يتم الكتابة فوقها بواسطة U A ( إذا كان SIDE = 'L' )،
!>           أو بواسطة A U ( إذا كان SIDE = 'R' )،
!>           أو بواسطة U A U' ( إذا كان SIDE = 'C' أو 'T').
!>

LDA

!>          LDA is INTEGER
!>          البعد الرائد للمصفوفة A.  LDA >= max(1,M).
!>

ISEED

!>          ISEED is INTEGER array, dimension (4)
!>          عند الإدخال، تحدد ISEED بذرة مولد الأرقام العشوائية.
!>          يجب أن تكون عناصر المصفوفة بين 0 و 4095؛
!>          إذا لم يكن كذلك، فسيتم تقليلها حسب 4096. أيضًا، يجب أن يكون ISEED(4)
!>          فرديًا.  يستخدم مولد الأرقام العشوائية تسلسلًا خطيًا متطابقًا
!>          محدودًا بأعداد صحيحة صغيرة، وبالتالي
!>          يجب أن ينتج أرقامًا عشوائية مستقلة عن الآلة.  قيم ISEED
!>          تتغير عند الخروج، ويمكن استخدامها في
!>          الاستدعاء التالي لـ DLAROR لمتابعة نفس تسلسل الأرقام العشوائية.
!>

X

!>          X is DOUBLE PRECISION array, dimension (3*MAX( M, N ))
!>          مساحة عمل بطول
!>              2*M + N إذا كان SIDE = 'L'،
!>              2*N + M إذا كان SIDE = 'R'،
!>              3*N     إذا كان SIDE = 'C' أو 'T'.
!>

INFO

!>          INFO is INTEGER
!>          علامة خطأ.  يتم تعيينها إلى:
!>          = 0:  عودة طبيعية
!>          < 0:  إذا كان INFO = -k، فإن المعامل k كان له قيمة غير قانونية
!>          = 1:  إذا كانت الأرقام العشوائية التي تم إنشاؤها بواسطة DLARND سيئة.
!>
