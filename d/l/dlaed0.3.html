subroutine dlaed0 (icompq, qsiz, n, d, e, q, ldq, qstore, ldqs, work, iwork, info)
DLAED0 تستخدم بواسطة DSTEDC. تحسب جميع القيم الذاتية والمتجهات الذاتية المقابلة لمصفوفة ثلاثية القطر متماثلة غير مختزلة باستخدام طريقة فرق تسد.

توثيق الوظيفة/الإجراء الفرعي

subroutine dlaed0 (integer icompq, integer qsiz, integer n, double precision, dimension( * ) d, double precision, dimension( * ) e, double precision, dimension( ldq, * ) q, integer ldq, double precision, dimension( ldqs, * ) qstore, integer ldqs, double precision, dimension( * ) work, integer, dimension( * ) iwork, integer info)

DLAED0 تستخدم بواسطة DSTEDC. تحسب جميع القيم الذاتية والمتجهات الذاتية المقابلة لمصفوفة ثلاثية القطر متماثلة غير مختزلة باستخدام طريقة فرق تسد.

الغرض:

!>
!> DLAED0 تحسب جميع القيم الذاتية والمتجهات الذاتية المقابلة لمصفوفة ثلاثية القطر متماثلة
!> باستخدام طريقة فرق تسد.
!>

المعاملات

ICOMPQ 
!>          ICOMPQ هو عدد صحيح
!>          = 0:  حساب القيم الذاتية فقط.
!>          = 1:  حساب المتجهات الذاتية للمصفوفة المتماثلة الكثيفة الأصلية
!>                أيضًا. عند الإدخال، تحتوي Q على المصفوفة المتعامدة المستخدمة
!>                لتقليل المصفوفة الأصلية إلى شكل ثلاثي القطر.
!>          = 2:  حساب القيم الذاتية والمتجهات الذاتية للمصفوفة ثلاثية القطر.
!>

QSIZ

!>          QSIZ هو عدد صحيح
!>         بعد المصفوفة المتعامدة المستخدمة لتقليل
!>         المصفوفة الكاملة إلى شكل ثلاثي القطر.  QSIZ >= N إذا كان ICOMPQ = 1.
!>

N

!>          N هو عدد صحيح
!>         بعد المصفوفة المتماثلة ثلاثية القطر.  N >= 0.
!>

D

!>          D هو مصفوفة Double Precision
!>         عند الإدخال، القطر الرئيسي للمصفوفة ثلاثية القطر.
!>         عند الخروج، قيمها الذاتية.
!>

E

!>          E هو مصفوفة Double Precision، البعد (N-1)
!>         العناصر خارج القطر للمصفوفة ثلاثية القطر.
!>         عند الخروج، تم تدمير E.
!>

Q

!>          Q هو مصفوفة Double Precision، البعد (LDQ, N)
!>         عند الإدخال، يجب أن تحتوي Q على مصفوفة متعامدة N×N.
!>         إذا كان ICOMPQ = 0، فلن يتم الرجوع إلى Q.
!>         إذا كان ICOMPQ = 1، عند الإدخال، Q هو مجموعة فرعية من أعمدة
!>                          المصفوفة المتعامدة المستخدمة لتقليل المصفوفة الكاملة
!>                          إلى شكل ثلاثي القطر المقابل
!>                          للمجموعة الفرعية من المصفوفة الكاملة التي يتم
!>                          تحليلها في هذا الوقت.
!>         إذا كان ICOMPQ = 2، عند الإدخال، ستكون Q هي مصفوفة الوحدة.
!>                          عند الخروج، تحتوي Q على المتجهات الذاتية للمصفوفة
!>                          ثلاثية القطر.
!>

LDQ

!>          LDQ هو عدد صحيح
!>         البعد الرائد للمصفوفة Q.  إذا كانت المتجهات الذاتية
!>         مرغوبة، فإن  LDQ >= max(1,N).  في أي حال،  LDQ >= 1.
!>

QSTORE

!>          QSTORE هو مصفوفة Double Precision، البعد (LDQS, N)
!>         يتم الرجوع إليه فقط عندما يكون ICOMPQ = 1.  يستخدم لتخزين أجزاء من
!>         مصفوفة المتجهات الذاتية عندما تحدث عملية ضرب المصفوفة المحدثة.
!>

LDQS

!>          LDQS هو عدد صحيح
!>         البعد الرائد للمصفوفة QSTORE.  إذا كان ICOMPQ = 1،
!>         فإن  LDQS >= max(1,N).  في أي حال،  LDQS >= 1.
!>

WORK

!>          WORK هي مصفوفة Double Precision،
!>         إذا كان ICOMPQ = 0 أو 1، يجب أن يكون بعد WORK على الأقل
!>                     1 + 3*N + 2*N*lg N + 3*N**2
!>                     ( lg( N ) = أصغر عدد صحيح k
!>                                 بحيث 2^k >= N )
!>         إذا كان ICOMPQ = 2، يجب أن يكون بعد WORK على الأقل
!>                     4*N + N**2.
!>

IWORK

!>          IWORK هي مصفوفة Integer،
!>         إذا كان ICOMPQ = 0 أو 1، يجب أن يكون بعد IWORK على الأقل
!>                        6 + 6*N + 5*N*lg N.
!>                        ( lg( N ) = أصغر عدد صحيح k
!>                                    بحيث 2^k >= N )
!>         إذا كان ICOMPQ = 2، يجب أن يكون بعد IWORK على الأقل
!>                        3 + 5*N.
!>

INFO

!>          INFO هو عدد صحيح
!>          = 0:  خروج ناجح.
!>          < 0:  إذا كان INFO = -i، فإن المعامل i كان له قيمة غير قانونية.
!>          > 0:  فشلت الخوارزمية في حساب قيمة ذاتية أثناء العمل على المصفوفة الفرعية
!>                الواقعة في الصفوف والأعمدة
!>                INFO/(N+1) حتى mod(INFO,N+1).
!>
