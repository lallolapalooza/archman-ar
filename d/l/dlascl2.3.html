subroutine dlascl2 (integer m, integer n, double precision, dimension( * ) d, double precision, dimension( ldx, * ) x, integer ldx)

DLASCL2 تقوم بالتحجيم القطري لمصفوفة.

الغرض:

!>
!> DLASCL2 تقوم بالتحجيم القطري لمصفوفة:
!>   x <-- D * x
!> حيث يتم تخزين المصفوفة القطرية D كمتجه.
!>
!> ليتم استبدالها في النهاية بـ BLAS_dge_diag_scale في معيار BLAS الجديد.
!>

المعاملات

M 
!>          M هو عدد صحيح (INTEGER)
!>     عدد صفوف D و X. M >= 0.
!>

N

!>          N هو عدد صحيح (INTEGER)
!>     عدد أعمدة X. N >= 0.
!>

D

!>          D هو مصفوفة من نوع DOUBLE PRECISION، بطول M
!>     المصفوفة القطرية D، مخزنة كمتجه بطول M.
!>

X

!>          X هو مصفوفة من نوع DOUBLE PRECISION، أبعادها (LDX,N)
!>     عند الإدخال، المصفوفة X المراد تحجيمها بواسطة D.
!>     عند الخروج، المصفوفة المحجمة.
!>

LDX

!>          LDX هو عدد صحيح (INTEGER)
!>     البعد الرائد للمصفوفة X. LDX >= M.
!>
