subroutine dla_lin_berr (integer n, integer nz, integer nrhs, double precision, dimension( n, nrhs ) res, double precision, dimension( n, nrhs ) ayb, double precision, dimension( nrhs ) berr)

DLA_LIN_BERR تحسب خطأ رجعي نسبي مكون بمكون.

الغرض:

!>
!>    DLA_LIN_BERR تحسب خطأ رجعي نسبي مكون بمكون من
!>    الصيغة
!>        max(i) ( abs(R(i)) / ( abs(op(A_s))*abs(Y) + abs(B_s) )(i) )
!>    حيث abs(Z) هي القيمة المطلقة المكونة بمكون للمصفوفة
!>    أو المتجه Z.
!>

المعلمات

N 
!>          N هو عدد صحيح (INTEGER)
!>     عدد المعادلات الخطية، أي رتبة
!>     المصفوفة A. N >= 0.
!>

NZ

!>          NZ هو عدد صحيح (INTEGER)
!>     نضيف (NZ+1)*SLAMCH( 'Safe minimum' ) إلى R(i) في البسط للحماية
!>     من البواقي الصفرية بشكل مضلل. القيمة الافتراضية هي N.
!>

NRHS

!>          NRHS هو عدد صحيح (INTEGER)
!>     عدد الأطراف اليمنى، أي عدد الأعمدة
!>     للمصفوفات AYB, RES, و BERR. NRHS >= 0.
!>

RES

!>          RES هي مصفوفة دقة مزدوجة (DOUBLE PRECISION array)، البعد (N,NRHS)
!>     مصفوفة البواقي، أي المصفوفة R في صيغة الخطأ الرجعي النسبي أعلاه.
!>

AYB

!>          AYB هي مصفوفة دقة مزدوجة (DOUBLE PRECISION array)، البعد (N, NRHS)
!>     المقام في صيغة الخطأ الرجعي النسبي أعلاه، أي
!>     المصفوفة abs(op(A_s))*abs(Y) + abs(B_s). المصفوفات A، Y، و B
!>     من التنقيح التكراري (راجع dla_gerfsx_extended.f).
!>

BERR

!>          BERR هي مصفوفة دقة مزدوجة (DOUBLE PRECISION array)، البعد (NRHS)
!>     الخطأ الرجعي النسبي المكون بمكون من الصيغة أعلاه.
!>
