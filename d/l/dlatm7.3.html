subroutine dlatm7 (integer mode, double precision cond, integer irsign, integer idist, integer, dimension( 4 ) iseed, double precision, dimension( * ) d, integer n, integer rank, integer info)

DLATM7

الغرض:

!>
!>    DLATM7 يحسب عناصر D كما هو محدد بواسطة MODE
!>    COND و IRSIGN. IDIST و ISEED تحدد توليد الأرقام العشوائية. يتم استدعاء DLATM7 بواسطة DLATMT لتوليد
!>    مصفوفات اختبار عشوائية.
!> 
!>  MODE   - INTEGER
!>           عند الإدخال يصف كيف سيتم حساب D:
!>           MODE = 0 يعني عدم تغيير D.
!>
!>           MODE = 1 يضبط D(1)=1 و D(2:RANK)=1.0/COND
!>           MODE = 2 يضبط D(1:RANK-1)=1 و D(RANK)=1.0/COND
!>           MODE = 3 يضبط D(I)=COND**(-(I-1)/(RANK-1)) I=1:RANK
!>
!>           MODE = 4 يضبط D(i)=1 - (i-1)/(N-1)*(1 - 1/COND)
!>           MODE = 5 يضبط D لأرقام عشوائية في النطاق
!>                    ( 1/COND , 1 ) بحيث تكون لوغاريتماتها
!>                    موزعة بانتظام.
!>           MODE = 6 يضبط D لأرقام عشوائية من نفس التوزيع
!>                    مثل بقية المصفوفة.
!>           MODE < 0 له نفس المعنى مثل ABS(MODE)، باستثناء أن
!>              ترتيب عناصر D يتم عكسه.
!>           وبالتالي إذا كان MODE موجبًا، فإن D يحتوي على عناصر تتراوح من
!>              1 إلى 1/COND، وإذا كان سالبًا، من 1/COND إلى 1،
!>           غير معدل.
!>
!>  COND   - DOUBLE PRECISION
!>           عند الإدخال، يستخدم كما هو موضح تحت MODE أعلاه.
!>           إذا تم استخدامه، يجب أن يكون >= 1. غير معدل.
!>
!>  IRSIGN - INTEGER
!>           عند الإدخال، إذا لم يكن MODE -6 أو 0 أو 6، يحدد إشارة
!>           عناصر D
!>           0 => ترك عناصر D دون تغيير
!>           1 => ضرب كل عنصر من D بـ 1 أو -1 باحتمالية .5
!>
!>  IDIST  - CHARACTER*1
!>           عند الإدخال، يحدد IDIST نوع التوزيع الذي سيتم
!>           استخدامه لتوليد مصفوفة عشوائية.
!>           1 => UNIFORM( 0, 1 )
!>           2 => UNIFORM( -1, 1 )
!>           3 => NORMAL( 0, 1 )
!>           غير معدل.
!>
!>  ISEED  - INTEGER array, dimension ( 4 )
!>           عند الإدخال، يحدد ISEED بذرة مولد الأرقام
!>           العشوائية. يستخدم مولد الأرقام
!>           العشوائية تسلسل خطي متطابق محدود بأعداد صحيحة صغيرة
!>           صغيرة، وبالتالي يجب أن ينتج أرقامًا عشوائية مستقلة عن الآلة.
!>           قيم ISEED تتغير عند الخروج، ويمكن استخدامها في الاستدعاء التالي لـ DLATM7
!>           لمواصلة نفس تسلسل الأرقام العشوائية.
!>           تم التغيير عند الخروج.
!>
!>  D      - DOUBLE PRECISION array, dimension ( MIN( M , N ) )
!>           المصفوفة التي سيتم حسابها وفقًا لـ MODE و COND و IRSIGN.
!>           قد يتم تغييرها عند الخروج إذا كان MODE غير صفري.
!>
!>  N      - INTEGER
!>           عدد عناصر D. غير معدل.
!>
!>  RANK   - INTEGER
!>           رتبة المصفوفة التي سيتم إنشاؤها للوضع 1،2،3 فقط.
!>           D( RANK+1:N ) = 0.
!>           غير معدل.
!>
!>  INFO   - INTEGER
!>            0  => إنهاء طبيعي
!>           -1  => إذا كان MODE ليس في النطاق -6 إلى 6
!>           -2  => إذا لم يكن MODE -6 أو 0 أو 6، و
!>                  IRSIGN ليس 0 أو 1
!>           -3  => إذا لم يكن MODE -6 أو 0 أو 6 و COND أقل من 1
!>           -4  => إذا كان MODE يساوي 6 أو -6 و IDIST ليس في النطاق 1 إلى 3
!>           -7  => إذا كان N سالبًا
!>
