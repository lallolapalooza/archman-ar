subroutine dlsets (integer m, integer p, integer n, double precision, dimension( lda, * ) a, double precision, dimension( lda, * ) af, integer lda, double precision, dimension( ldb, * ) b, double precision, dimension( ldb, * ) bf, integer ldb, double precision, dimension( * ) c, double precision, dimension( * ) cf, double precision, dimension( * ) d, double precision, dimension( * ) df, double precision, dimension( * ) x, double precision, dimension( lwork ) work, integer lwork, double precision, dimension( * ) rwork, double precision, dimension( 2 ) result)

DLSETS

الغرض:

!>
!> DLSETS تختبر DGGLSE - روتين فرعي لحل مشكلة المربعات الصغرى المقيدة خطياً (LSE).
!>

المعلمات

M 
!>          M هو عدد صحيح
!>          عدد الصفوف في المصفوفة A.  M >= 0.
!>

P

!>          P هو عدد صحيح
!>          عدد الصفوف في المصفوفة B.  P >= 0.
!>

N

!>          N هو عدد صحيح
!>          عدد الأعمدة في المصفوفتين A و B.  N >= 0.
!>

A

!>          A هي مصفوفة من نوع DOUBLE PRECISION، البعد (LDA,N)
!>          المصفوفة A بحجم M×N.
!>

AF

!>          AF هي مصفوفة من نوع DOUBLE PRECISION، البعد (LDA,N)
!>

LDA

!>          LDA هو عدد صحيح
!>          البعد الرائد للمصفوفات A, AF, Q و R.
!>          LDA >= max(M,N).
!>

B

!>          B هي مصفوفة من نوع DOUBLE PRECISION، البعد (LDB,N)
!>          المصفوفة A بحجم P×N.
!>

BF

!>          BF هي مصفوفة من نوع DOUBLE PRECISION، البعد (LDB,N)
!>

LDB

!>          LDB هو عدد صحيح
!>          البعد الرائد للمصفوفات B, BF, V و S.
!>          LDB >= max(P,N).
!>

C

!>          C هي مصفوفة من نوع DOUBLE PRECISION، البعد( M )
!>          المتجه C في مشكلة LSE.
!>

CF

!>          CF هي مصفوفة من نوع DOUBLE PRECISION، البعد( M )
!>

D

!>          D هي مصفوفة من نوع DOUBLE PRECISION، البعد( P )
!>          المتجه D في مشكلة LSE.
!>

DF

!>          DF هي مصفوفة من نوع DOUBLE PRECISION، البعد( P )
!>

X

!>          X هي مصفوفة من نوع DOUBLE PRECISION، البعد( N )
!>          المتجه الحل X في مشكلة LSE.
!>

WORK

!>          WORK هي مصفوفة من نوع DOUBLE PRECISION، البعد (LWORK)
!>

LWORK

!>          LWORK هو عدد صحيح
!>          بعد المصفوفة WORK.
!>

RWORK

!>          RWORK هي مصفوفة من نوع DOUBLE PRECISION، البعد (M)
!>

RESULT

!>          RESULT هي مصفوفة من نوع DOUBLE PRECISION، البعد (2)
!>          نسب الاختبار:
!>            RESULT(1) = norm( A*x - c )/ norm(A)*norm(X)*EPS
!>            RESULT(2) = norm( B*x - d )/ norm(B)*norm(X)*EPS
!>
