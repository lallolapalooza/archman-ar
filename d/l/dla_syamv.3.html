subroutine dla_syamv (uplo, n, alpha, a, lda, x, incx, beta, y, incy)


DLA_SYAMV تحسب ناتج ضرب مصفوفة ومتجه باستخدام مصفوفة غير محددة متماثلة لحساب حدود الخطأ.

توثيق الوظيفة/الإجراء الفرعي

subroutine dla_syamv (integer uplo, integer n, double precision alpha, double precision, dimension( lda, * ) a, integer lda, double precision, dimension( * ) x, integer incx, double precision beta, double precision, dimension( * ) y, integer incy)

DLA_SYAMV تحسب ناتج ضرب مصفوفة ومتجه باستخدام مصفوفة غير محددة متماثلة لحساب حدود الخطأ.

الغرض:

!>
!> DLA_SYAMV  تقوم بعملية المصفوفة والمتجه
!>
!>         y := alpha*abs(A)*abs(x) + beta*abs(y),
!>
!> حيث alpha و beta هما قيمتان قياسيتان، x و y هما متجهان، و A مصفوفة
!> متماثلة بحجم n × n.
!>
!> تُستخدم هذه الوظيفة بشكل أساسي في حساب حدود الخطأ.
!> للحماية من التدفق السفلي أثناء التقييم، يتم تعديل مكونات
!> المتجه الناتج بعيداً عن الصفر بمقدار (N+1)
!> أضعاف عتبة التدفق السفلي. لمنع الأخطاء الكبيرة بشكل غير ضروري
!> للهياكل الكتلية المضمنة في المصفوفات العامة،
!> لا يتم تعديل المكونات الصفرية.
!>  يعتبر الإدخال الصفري
!>  إذا كانت جميع عمليات الضرب المتضمنة
!> في حساب هذا الإدخال تحتوي على مضاعف صفري واحد على الأقل.
!>

المعلمات

UPLO 
!>          UPLO هو عدد صحيح (INTEGER).
!>           عند الدخول، يحدد UPLO ما إذا كان الجزء المثلثي العلوي أو السفلي
!>           من المصفوفة A سيتم الرجوع إليه كما يلي:
!>
!>              UPLO = BLAS_UPPER   فقط الجزء المثلثي العلوي من A
!>                                  سيتم الرجوع إليه.
!>
!>              UPLO = BLAS_LOWER   فقط الجزء المثلثي السفلي من A
!>                                  سيتم الرجوع إليه.
!>
!>           لا يتغير عند الخروج.
!>

N

!>          N هو عدد صحيح (INTEGER).
!>           عند الدخول، يحدد N عدد أعمدة المصفوفة A.
!>           يجب أن يكون N على الأقل صفرًا.
!>           لا يتغير عند الخروج.
!>

ALPHA

!>          ALPHA هو دقة مزدوجة (DOUBLE PRECISION).
!>           عند الدخول، يحدد ALPHA القيمة القياسية alpha.
!>           لا يتغير عند الخروج.
!>

A

!>          A هو مصفوفة دقة مزدوجة (DOUBLE PRECISION array)، الأبعاد ( LDA, n ).
!>           قبل الدخول، يجب أن يحتوي الجزء الرئيسي n × n من المصفوفة A
!>           على مصفوفة المعاملات.
!>           لا يتغير عند الخروج.
!>

LDA

!>          LDA هو عدد صحيح (INTEGER).
!>           عند الدخول، يحدد LDA البعد الأول لـ A كما تم تعريفه
!>           في البرنامج (الفرعي) المستدعي. يجب أن يكون LDA على الأقل
!>           max( 1, n ).
!>           لا يتغير عند الخروج.
!>

X

!>          X هي مصفوفة دقة مزدوجة (DOUBLE PRECISION array)، الأبعاد
!>           ( 1 + ( n - 1 )*abs( INCX ) )
!>           قبل الدخول، يجب أن تحتوي المصفوفة المتزايدة X على
!>           المتجه x.
!>           لا يتغير عند الخروج.
!>

INCX

!>          INCX هو عدد صحيح (INTEGER).
!>           عند الدخول، يحدد INCX الزيادة لعناصر X.
!>           يجب ألا يكون INCX صفرًا.
!>           لا يتغير عند الخروج.
!>

BETA

!>          BETA هو دقة مزدوجة (DOUBLE PRECISION).
!>           عند الدخول، يحدد BETA القيمة القياسية beta. عندما يتم
!>           توفير BETA بصفر، لا يلزم تعيين Y عند الدخول.
!>           لا يتغير عند الخروج.
!>

Y

!>          Y هي مصفوفة دقة مزدوجة (DOUBLE PRECISION array)، الأبعاد
!>           ( 1 + ( n - 1 )*abs( INCY ) )
!>           قبل الدخول مع BETA غير صفرية، يجب أن تحتوي المصفوفة المتزايدة Y
!>           على المتجه y. عند الخروج، سيتم استبدال Y بالمتجه y
!>           المحدث.
!>

INCY

!>          INCY هو عدد صحيح (INTEGER).
!>           عند الدخول، يحدد INCY الزيادة لعناصر Y.
!>           يجب ألا يكون INCY صفرًا.
!>           لا يتغير عند الخروج.
!>
