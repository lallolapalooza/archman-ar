subroutine dlascl (character type, integer kl, integer ku, double precision cfrom, double precision cto, integer m, integer n, double precision, dimension( lda, * ) a, integer lda, integer info)

DLASCL تضرب المصفوفة الحقيقية M × N المسماة A في القياسي الحقيقي CTO/CFROM.

الغرض:

!>
!> DLASCL تضرب المصفوفة الحقيقية M × N المسماة A في القياسي الحقيقي
!> CTO/CFROM. يتم ذلك دون فيض علوي/سفلي طالما أن النتيجة النهائية
!> CTO*A(I,J)/CFROM لا تفيض علويًا/سفليًا. TYPE تحدد أن
!> A قد تكون كاملة، أو مثلثية علوية، أو مثلثية سفلية، أو علوية هسنبرغ،
!> أو نطاقية.
!>

المعلمات

TYPE 
!>          TYPE هو CHARACTER*1
!>          TYPE تشير إلى نوع تخزين المصفوفة المدخلة.
!>          = 'G':  A هي مصفوفة كاملة.
!>          = 'L':  A هي مصفوفة مثلثة سفلية.
!>          = 'U':  A هي مصفوفة مثلثة علوية.
!>          = 'H':  A هي مصفوفة هسنبرغ علوية.
!>          = 'B':  A هي مصفوفة نطاقية متماثلة مع عرض نطاق سفلي KL
!>                  وعرض نطاق علوي KU ويتم تخزين النصف السفلي فقط.
!>          = 'Q':  A هي مصفوفة نطاقية متماثلة مع عرض نطاق سفلي KL
!>                  وعرض نطاق علوي KU ويتم تخزين النصف العلوي فقط.
!>          = 'Z':  A هي مصفوفة نطاقية مع عرض نطاق سفلي KL وعرض نطاق علوي
!>                  KU. انظر DGBTRF لتفاصيل التخزين.
!>

KL

!>          KL هو INTEGER
!>          عرض النطاق السفلي للمصفوفة A. يتم الإشارة إليه فقط إذا كان TYPE = 'B',
!>          'Q', أو 'Z'.
!>

KU

!>          KU هو INTEGER
!>          عرض النطاق العلوي للمصفوفة A. يتم الإشارة إليه فقط إذا كان TYPE = 'B',
!>          'Q', أو 'Z'.
!>

CFROM

!>          CFROM هو DOUBLE PRECISION
!>

CTO

!>          CTO هو DOUBLE PRECISION
!>
!>          تضرب المصفوفة A في CTO/CFROM. يتم حساب A(I,J)
!>          دون فيض علوي/سفلي إذا كانت النتيجة النهائية CTO*A(I,J)/CFROM
!>          يمكن تمثيلها دون فيض علوي/سفلي. يجب أن تكون CFROM
!>          غير صفرية.
!>

M

!>          M هو INTEGER
!>          عدد صفوف المصفوفة A.  M >= 0.
!>

N

!>          N هو INTEGER
!>          عدد أعمدة المصفوفة A.  N >= 0.
!>

A

!>          A هي مصفوفة DOUBLE PRECISION، البعد (LDA,N)
!>          المصفوفة التي سيتم ضربها في CTO/CFROM. انظر TYPE لنوع
!>          التخزين.
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرائد للمصفوفة A.
!>          إذا كان TYPE = 'G', 'L', 'U', 'H', LDA >= max(1,M);
!>             TYPE = 'B', LDA >= KL+1;
!>             TYPE = 'Q', LDA >= KU+1;
!>             TYPE = 'Z', LDA >= 2*KL+KU+1.
!>

INFO

!>          INFO هو INTEGER
!>          0  - خروج ناجح
!>          <0 - إذا كان INFO = -i، فإن المعامل i كان له قيمة غير قانونية.
!>
