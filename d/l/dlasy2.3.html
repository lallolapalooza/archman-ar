subroutine dlasy2 (logical ltranl, logical ltranr, integer isgn, integer n1, integer n2, double precision, dimension( ldtl, * ) tl, integer ldtl, double precision, dimension( ldtr, * ) tr, integer ldtr, double precision, dimension( ldb, * ) b, integer ldb, double precision scale, double precision, dimension( ldx, * ) x, integer ldx, double precision xnorm, integer info)

DLASY2 تحل معادلة سيلفستر المصفوفية حيث تكون المصفوفات من الرتبة 1 أو 2.

الغرض:

!>
!> DLASY2 تحل للمصفوفة X من حجم N1 في N2، حيث 1 <= N1,N2 <= 2، في المعادلة
!>
!>        op(TL)*X + ISGN*X*op(TR) = SCALE*B,
!>
!> حيث TL هي N1 في N1، TR هي N2 في N2، B هي N1 في N2، و ISGN = 1 أو
!> -1. op(T) = T أو T**T، حيث T**T تمثل منقول T.
!>

المعلمات

LTRANL 
!>          LTRANL هو LOGICAL
!>          عند الدخول، تحدد LTRANL op(TL):
!>             = .FALSE., op(TL) = TL,
!>             = .TRUE., op(TL) = TL**T.
!>

LTRANR

!>          LTRANR هو LOGICAL
!>          عند الدخول، تحدد LTRANR op(TR):
!>            = .FALSE., op(TR) = TR,
!>            = .TRUE., op(TR) = TR**T.
!>

ISGN

!>          ISGN هو INTEGER
!>          عند الدخول، تحدد ISGN إشارة المعادلة
!>          كما هو موضح سابقاً. ISGN يمكن أن تكون فقط 1 أو -1.
!>

N1

!>          N1 هو INTEGER
!>          عند الدخول، تحدد N1 رتبة المصفوفة TL.
!>          N1 يمكن أن تكون فقط 0، 1 أو 2.
!>

N2

!>          N2 هو INTEGER
!>          عند الدخول، تحدد N2 رتبة المصفوفة TR.
!>          N2 يمكن أن تكون فقط 0، 1 أو 2.
!>

TL

!>          TL هو مصفوفة DOUBLE PRECISION، الأبعاد (LDTL,2)
!>          عند الدخول، تحتوي TL على مصفوفة N1 في N1.
!>

LDTL

!>          LDTL هو INTEGER
!>          البعد الرئيسي للمصفوفة TL. LDTL >= max(1,N1).
!>

TR

!>          TR هو مصفوفة DOUBLE PRECISION، الأبعاد (LDTR,2)
!>          عند الدخول، تحتوي TR على مصفوفة N2 في N2.
!>

LDTR

!>          LDTR هو INTEGER
!>          البعد الرئيسي للمصفوفة TR. LDTR >= max(1,N2).
!>

B

!>          B هو مصفوفة DOUBLE PRECISION، الأبعاد (LDB,2)
!>          عند الدخول، تحتوي المصفوفة B من حجم N1 في N2 على الجانب الأيمن
!>          من المعادلة.
!>

LDB

!>          LDB هو INTEGER
!>          البعد الرئيسي للمصفوفة B. LDB >= max(1,N1).
!>

SCALE

!>          SCALE هو DOUBLE PRECISION
!>          عند الخروج، تحتوي SCALE على عامل القياس. يتم اختيار SCALE
!>          لتكون أقل من أو تساوي 1 لمنع تجاوز سعة الحل.
!>

X

!>          X هو مصفوفة DOUBLE PRECISION، الأبعاد (LDX,2)
!>          عند الخروج، تحتوي X على الحل من حجم N1 في N2.
!>

LDX

!>          LDX هو INTEGER
!>          البعد الرئيسي للمصفوفة X. LDX >= max(1,N1).
!>

XNORM

!>          XNORM هو DOUBLE PRECISION
!>          عند الخروج، XNORM هي المعيار اللانهائي للحل.
!>

INFO

!>          INFO هو INTEGER
!>          عند الخروج، يتم تعيين INFO إلى:
!>             0: خروج ناجح.
!>             1: TL و TR لديهما قيم ذاتية متقاربة جداً، لذا يتم إزعاج TL أو
!>                TR للحصول على معادلة غير مفردة.
!>          ملاحظة: حرصاً على السرعة، لا تتحقق هذه الروتين الفرعي من المدخلات للأخطاء.
!>
