subroutine dlagtf (n, a, lambda, b, c, tol, d, in, info)

DLAGTF تحسب تحليل LU للمصفوفة T-λI، حيث T مصفوفة ثلاثية قطرية عامة، وλ قيمة قياسية، باستخدام التقسيم الجزئي مع تبديلات الصفوف.

توثيق الوظيفة/الإجراء الفرعي

subroutine dlagtf (integer n, double precision, dimension( * ) a, double precision lambda, double precision, dimension( * ) b, double precision, dimension( * ) c, double precision tol, double precision, dimension( * ) d, integer, dimension( * ) in, integer info)

DLAGTF تحسب تحليل LU للمصفوفة T-λI، حيث T مصفوفة ثلاثية قطرية عامة، وλ قيمة قياسية، باستخدام التقسيم الجزئي مع تبديلات الصفوف.

الغرض:

!>
!> DLAGTF تحلل المصفوفة (T - lambda*I)، حيث T مصفوفة ثلاثية قطرية من الرتبة n
!> و lambda قيمة قياسية، كالتالي:
!>
!>    T - lambda*I = PLU,
!>
!> حيث P مصفوفة تبديل، L مصفوفة ثلاثية قطرية سفلية وحدة بعناصر فرعية قطرية غير صفرية على الأكثر لكل عمود، و U
!> مصفوفة مثلثة عليا بعناصر فائقة قطرية غير صفرية على الأكثر لكل عمود.
!>
!> يتم الحصول على التحليل عن طريق الحذف الغاوسي مع التقسيم الجزئي
!> والقياس الضمني للصفوف.
!>
!> يتم تضمين المعامل LAMBDA في الروتين بحيث يمكن استخدام DLAGTF، بالاشتراك مع DLAGTS،
!> للحصول على المتجهات الذاتية لـ T عن طريق التكرار العكسي.
!>

المعاملات

N 
!>          N هو عدد صحيح (INTEGER)
!>          رتبة المصفوفة T.
!>

A

!>          A هو مصفوفة من نوع DOUBLE PRECISION، بأبعاد (N)
!>          عند الإدخال، يجب أن تحتوي A على العناصر القطرية للمصفوفة T.
!>
!>          عند الخروج، يتم الكتابة فوق A بالعناصر القطرية n للمصفوفة المثلثية العليا
!>          U الخاصة بتحليل T.
!>

LAMBDA

!>          LAMBDA هو نوع DOUBLE PRECISION
!>          عند الإدخال، القيمة القياسية lambda.
!>

B

!>          B هو مصفوفة من نوع DOUBLE PRECISION، بأبعاد (N-1)
!>          عند الإدخال، يجب أن تحتوي B على العناصر الفائقة القطرية (n-1) للمصفوفة
!>          T.
!>
!>          عند الخروج، يتم الكتابة فوق B بالعناصر الفائقة القطرية (n-1)
!>          للمصفوفة U الخاصة بتحليل T.
!>

C

!>          C هو مصفوفة من نوع DOUBLE PRECISION، بأبعاد (N-1)
!>          عند الإدخال، يجب أن تحتوي C على العناصر الفرعية القطرية (n-1) للمصفوفة
!>          T.
!>
!>          عند الخروج، يتم الكتابة فوق C بالعناصر الفرعية القطرية (n-1)
!>          للمصفوفة L الخاصة بتحليل T.
!>

TOL

!>          TOL هو نوع DOUBLE PRECISION
!>          عند الإدخال، هامش تسامح نسبي يستخدم للإشارة إلى ما إذا كانت المصفوفة (T - lambda*I)
!>          شبه مفردة أم لا. يجب عادةً اختيار TOL بحيث يكون تقريبًا أكبر خطأ نسبي
!>          في عناصر T. على سبيل المثال، إذا كانت عناصر T صحيحة لحوالي 4 أرقام معنوية،
!>          فيجب تعيين TOL إلى حوالي 5*10**(-4). إذا تم توفير TOL بقيمة أقل من eps،
!>          حيث eps هو الدقة النسبية للآلة، فيتم استخدام قيمة eps بدلاً من TOL.
!>

D

!>          D هو مصفوفة من نوع DOUBLE PRECISION، بأبعاد (N-2)
!>          عند الخروج، يتم الكتابة فوق D بالعناصر الفائقة القطرية الثانية (n-2)
!>          للمصفوفة U الخاصة بتحليل T.
!>

IN

!>          IN هو مصفوفة من نوع INTEGER، بأبعاد (N)
!>          عند الخروج، تحتوي IN على تفاصيل مصفوفة التبديل P. إذا
!>          حدث تبديل في الخطوة k من عملية الحذف، فإن IN(k) = 1، وإلا فإن IN(k) = 0.
!>          القيمة IN(n) تعود بأصغر عدد صحيح موجب j بحيث
!>
!>             abs( u(j,j) ) <= norm( (T - lambda*I)(j) )*TOL,
!>
!>          حيث يشير norm( A(j) ) إلى مجموع القيم المطلقة
!>          للصف j من المصفوفة A. إذا لم يكن مثل هذا j موجودًا، يتم إرجاع IN(n)
!>          على أنه صفر. إذا تم إرجاع IN(n) كقيمة موجبة، فإن عنصرًا قطريًا في U صغير،
!>          مما يشير إلى أن (T - lambda*I) مفردة أو شبه مفردة،
!>

INFO

!>          INFO هو عدد صحيح (INTEGER)
!>          = 0:  خروج ناجح
!>          < 0:  إذا كان INFO = -k، فإن المعامل k كان له قيمة غير قانونية
!>
