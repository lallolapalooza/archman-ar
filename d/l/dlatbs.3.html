subroutine dlatbs (character uplo, character trans, character diag, character normin, integer n, integer kd, double precision, dimension( ldab, * ) ab, integer ldab, double precision, dimension( * ) x, double precision scale, double precision, dimension( * ) cnorm, integer info)

DLATBS يحل نظامًا مثلثيًا نطاقيًا.

الغرض:

!>
!> DLATBS يحل أحد الأنظمة المثلثية
!>
!>    A *x = s*b  أو  A**T*x = s*b
!>
!> مع القياس لمنع الفيضان، حيث A مصفوفة نطاقية مثلثية علوية أو سفلية. هنا A**T
!> يرمز إلى منقول A، و x و b متجهات من n عنصر، و s هو عامل قياس، عادة ما يكون
!!> أصغر من أو يساوي 1، يتم اختياره بحيث تكون مكونات x أقل من
!> عتبة الفيضان. إذا لم يتسبب النظام غير المقاس في حدوث فيضان، فسيتم استدعاء
!> روتين BLAS من المستوى 2 DTBSV. إذا كانت المصفوفة A مفردة (A(j,j) = 0 لبعض j)،
!> فسيتم تعيين s إلى 0 ويتم إرجاع حل غير تافه لـ A*x = 0.
!>

المعلمات

UPLO 
!>          UPLO هو CHARACTER*1
!>          يحدد ما إذا كانت المصفوفة A علوية أم سفلية.
!>          = 'U':  مثلثية علوية
!>          = 'L':  مثلثية سفلية
!>

TRANS

!>          TRANS هو CHARACTER*1
!>          يحدد العملية المطبقة على A.
!>          = 'N':  حل A * x = s*b  (بدون تبديل)
!>          = 'T':  حل A**T* x = s*b  (تبديل)
!>          = 'C':  حل A**T* x = s*b  (تبديل مترافق = تبديل)
!>

DIAG

!>          DIAG هو CHARACTER*1
!>          يحدد ما إذا كانت المصفوفة A مثلثية وحدوية أم لا.
!>          = 'N':  مثلثية غير وحدوية
!>          = 'U':  مثلثية وحدوية
!>

NORMIN

!>          NORMIN هو CHARACTER*1
!>          يحدد ما إذا كان CNORM قد تم تعيينه أم لا.
!>          = 'Y':  يحتوي CNORM على معايير الأعمدة عند الدخول
!>          = 'N':  لم يتم تعيين CNORM عند الدخول. عند الخروج، سيتم
!>                  حساب المعايير وتخزينها في CNORM.
!>

N

!>          N هو INTEGER
!>          رتبة المصفوفة A.  N >= 0.
!>

KD

!>          KD هو INTEGER
!>          عدد الأقطار الفرعية أو الأقطار الفوقية في
!>          المصفوفة المثلثية A.  KD >= 0.
!>

AB

!>          AB هي مصفوفة DOUBLE PRECISION، ذات أبعاد (LDAB,N)
!>          المصفوفة النطاقية المثلثية العلوية أو السفلية A، مخزنة في
!>          أول KD+1 صفوف من المصفوفة. يتم تخزين العمود j من A
!>          في العمود j من المصفوفة AB كما يلي:
!>          إذا كان UPLO = 'U'، فإن AB(kd+1+i-j,j) = A(i,j) لـ max(1,j-kd)<=i<=j؛
!>          إذا كان UPLO = 'L'، فإن AB(1+i-j,j) = A(i,j) لـ j<=i<=min(n,j+kd).
!>

LDAB

!>          LDAB هو INTEGER
!>          البعد الرائد للمصفوفة AB.  LDAB >= KD+1.
!>

X

!>          X هي مصفوفة DOUBLE PRECISION، ذات أبعاد (N)
!>          عند الدخول، الجانب الأيمن b للنظام المثلثي.
!>          عند الخروج، يتم استبدال X بمتجه الحل x.
!>

SCALE

!>          SCALE هو DOUBLE PRECISION
!>          عامل القياس s للنظام المثلثي
!>             A * x = s*b  أو  A**T* x = s*b.
!>          إذا كان SCALE = 0، فإن المصفوفة A مفردة أو مقاسة بشكل سيء،
!>          والمتجه x هو حل دقيق أو تقريبي لـ A*x = 0.
!>

CNORM

!>          CNORM هي مصفوفة DOUBLE PRECISION، ذات أبعاد (N)
!>
!>          إذا كان NORMIN = 'Y'، فإن CNORM هو وسيط إدخال و CNORM(j)
!>          يحتوي على معيار الجزء غير القطري للعمود j
!>          من A.  إذا كان TRANS = 'N'، يجب أن يكون CNORM(j) أكبر من أو يساوي
!>          المعيار اللانهائي، وإذا كان TRANS = 'T' أو 'C'، يجب أن يكون CNORM(j)
!>          أكبر من أو يساوي المعيار 1.
!>
!>          إذا كان NORMIN = 'N'، فإن CNORM هو وسيط إخراج و CNORM(j)
!>          يعيد المعيار 1 للجزء غير القطري للعمود j
!>          من A.
!>

INFO

!>          INFO هو INTEGER
!>          = 0:  خروج ناجح
!>          < 0:  إذا كان INFO = -k، فإن الوسيط k كان له قيمة غير قانونية
!>

تفاصيل إضافية:

!>
!> يتم حساب حد تقريبي لـ x؛ إذا كان هذا الحد أقل من الفيضان،
!> يتم استدعاء DTBSV، وإلا، يتم استخدام رمز محدد يتحقق من احتمال
!> حدوث فيضان أو قسمة على صفر عند كل عملية.
!>
!> يتم استخدام مخطط عمودي لحل A*x = b. الخوارزمية الأساسية
!> إذا كانت A مثلثية سفلية هي
!>
!>      x[1:n] := b[1:n]
!>      for j = 1, ..., n
!>           x(j) := x(j) / A(j,j)
!>           x[j+1:n] := x[j+1:n] - x(j) * A[j+1:n,j]
!>      end
!>
!> نحدد الحدود على مكونات x بعد j تكرار للحلقة:
!>    M(j) = حد على x[1:j]
!>    G(j) = حد على x[j+1:n]
!> في البداية، لنفترض M(0) = 0 و G(0) = max{x(i), i=1,...,n}.
!>
!> ثم للتكرار j+1 لدينا
!>    M(j+1) <= G(j) / | A(j+1,j+1) |
!>    G(j+1) <= G(j) + M(j+1) * | A[j+2:n,j+1] |
!>           <= G(j) ( 1 + CNORM(j+1) / | A(j+1,j+1) | )
!>
!> حيث CNORM(j+1) أكبر من أو يساوي المعيار اللانهائي
!> للعمود j+1 من A، باستثناء القطر. وبالتالي
!>
!>    G(j) <= G(0) product ( 1 + CNORM(i) / | A(i,i) | )
!>                  1<=i<=j
!> و
!>
!>    |x(j)| <= ( G(0) / |A(j,j)| ) product ( 1 + CNORM(i) / |A(i,i)| )
!>                                   1<=i< j
!>
!> بما أن |x(j)| <= M(j)، نستخدم روتين BLAS من المستوى 2 DTBSV إذا كان
!> مقلوب أكبر M(j)، j=1,..,n، أكبر من
!> max(underflow, 1/overflow).
!>
!> يستخدم الحد على x(j) أيضًا لتحديد متى يمكن إجراء خطوة في
!> المخطط العمودي دون خوف من الفيضان. إذا
!> كان الحد المحسوب أكبر من ثابت كبير، يتم قياس x لمنع
!> الفيضان، ولكن إذا حدث فيضان للحد، يتم تعيين x إلى 0، و x(j) إلى
!> 1، و scale إلى 0، ويتم العثور على حل غير تافه لـ A*x = 0.
!>
!> وبالمثل، يتم استخدام مخطط صفي لحل A**T*x = b. الخوارزمية الأساسية
!> لـ A مثلثية علوية هي
!>
!>       for j = 1, ..., n
!>            x(j) := ( b(j) - A[1:j-1,j]**T * x[1:j-1] ) / A(j,j)
!>       end
!>
!> نقوم بحساب حدين في وقت واحد
!>       G(j) = حد على ( b(i) - A[1:i-1,i]**T * x[1:i-1] )، 1<=i<=j
!>       M(j) = حد على x(i)، 1<=i<=j
!>
!> القيم الأولية هي G(0) = 0، M(0) = max{b(i)، i=1,..,n}، ونحن
!> نضيف الشرط G(j) >= G(j-1) و M(j) >= M(j-1) لـ j >= 1.
!> ثم حد x(j) هو
!>
!>       M(j) <= M(j-1) * ( 1 + CNORM(j) ) / | A(j,j) |
!>
!>            <= M(0) * product ( ( 1 + CNORM(i) ) / |A(i,i)| )
!>                      1<=i<=j
!>
!> ويمكننا استدعاء DTBSV بأمان إذا كان 1/M(n) و 1/G(n) كلاهما أكبر
!> من max(underflow, 1/overflow).
!>
