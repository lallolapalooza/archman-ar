subroutine dlamrg (n1, n2, a, dtrd1, dtrd2, index)
DLAMRG ينشئ قائمة تبديل لدمج مدخلات مجموعتين مرتبتين بشكل مستقل في مجموعة واحدة مرتبة بترتيب تصاعدي.

توثيق الوظيفة/الإجراء الفرعي

subroutine dlamrg (integer n1, integer n2, double precision, dimension( * ) a, integer dtrd1, integer dtrd2, integer, dimension( * ) index)

DLAMRG ينشئ قائمة تبديل لدمج مدخلات مجموعتين مرتبتين بشكل مستقل في مجموعة واحدة مرتبة بترتيب تصاعدي.

الغرض:

!>
!> DLAMRG سينشئ قائمة تبديل ستقوم بدمج عناصر A (التي تتكون من مجموعتين
!> مرتبتين بشكل مستقل) في مجموعة واحدة مرتبة بترتيب تصاعدي.
!>

المعلمات

N1 
!>          N1 هو عدد صحيح (INTEGER)
!>

N2

!>          N2 هو عدد صحيح (INTEGER)
!>         تحتوي هذه الوسائط على الأطوال المقابلة للقائمتين المرتبتين المراد
!>         دمجهما.
!>

A

!>          A هو مصفوفة من النوع double precision، بأبعاد (N1+N2)
!>         تحتوي أول N1 عناصر من A على قائمة من الأرقام المرتبة إما
!>         بترتيب تصاعدي أو تنازلي. وكذلك الأمر بالنسبة لآخر N2 عناصر.
!>

DTRD1

!>          DTRD1 هو عدد صحيح (INTEGER)
!>

DTRD2

!>          DTRD2 هو عدد صحيح (INTEGER)
!>         هذه هي الخطوات التي سيتم اتخاذها عبر المصفوفة A.
!>         الخطوات المسموح بها هي 1 و -1. وهي تشير إلى ما إذا كانت مجموعة
!>         فرعية من A مرتبة بترتيب تصاعدي (DTRDx = 1) أو تنازلي
!>         (DTRDx = -1).
!>

INDEX

!>          INDEX هي مصفوفة من النوع INTEGER، بأبعاد (N1+N2)
!>         عند الخروج، ستحتوي هذه المصفوفة على تبديل بحيث إذا كان B( I ) =
!>         A( INDEX( I ) ) لـ I=1,N1+N2، فإن B سيكون مرتبًا بترتيب
!>         تصاعدي.
!>
