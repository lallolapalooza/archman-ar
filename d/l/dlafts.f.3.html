subroutine dlafts (character*3 type, integer m, integer n, integer imat, integer ntests, double precision, dimension( * ) result, integer, dimension( 4 ) iseed, double precision thresh, integer iounit, integer ie)

DLAFTS

الغرض:

!>
!>    DLAFTS تختبر متجه النتائج مقابل قيمة الحد الأقصى لمعرفة الاختبارات التي فشلت
!>    لهذا النوع من المصفوفات في تجاوز الحد الأقصى.
!>    المخرجات إلى الملف المحدد بواسطة الوحدة IOUNIT.
!> 
!>  TYPE   - CHARACTER*3
!>           عند الإدخال، يحدد TYPE نوع المصفوفة المراد استخدامها في
!>           رسائل الطباعة.
!>           غير معدلة.
!>
!>  N      - INTEGER
!>           عند الإدخال، يحدد N ترتيب مصفوفة الاختبار.
!>           غير معدلة.
!>
!>  IMAT   - INTEGER
!>           عند الإدخال، يحدد IMAT نوع مصفوفة الاختبار.
!>           يتم طباعة قائمة بالأنواع المختلفة بواسطة DLAHD2
!>           إلى ملف المخرجات إذا فشل اختبار في تجاوز الحد الأقصى.
!>           غير معدلة.
!>
!>  NTESTS - INTEGER
!>           عند الإدخال، NTESTS هو عدد الاختبارات التي تم إجراؤها على
!>           الروتينات الفرعية في المسار المحدد بواسطة TYPE.
!>           غير معدلة.
!>
!>  RESULT - DOUBLE PRECISION               مصفوفة بحجم ( NTESTS )
!>           عند الإدخال، تحتوي RESULT على نسب الاختبارات
!>           التي تم إجراؤها في البرنامج المستدعي.
!>           غير معدلة.
!>
!>  ISEED  - INTEGER            مصفوفة بحجم ( 4 )
!>           تحتوي على البذرة العشوائية التي ولّدت المصفوفة المستخدمة
!>           للاختبارات التي نسبها في RESULT.
!>           غير معدلة.
!>
!>  THRESH - DOUBLE PRECISION
!>           عند الإدخال، يحدد THRESH الحد الأقصى المقبول لـ
!>           نسب الاختبارات. إذا كان RESULT( K ) > THRESH، فإن الاختبار K
!>           لم يتجاوز الحد الأقصى وسيتم طباعة رسالة.
!>           غير معدلة.
!>
!>  IOUNIT - INTEGER
!>           عند الإدخال، يحدد IOUNIT رقم وحدة الملف
!>           التي ستتم طباعة الرسائل إليها.
!>           غير معدلة.
!>
!>  IE     - INTEGER
!>           عند الإدخال، يحتوي IE على عدد الاختبارات التي فشلت
!>           في تجاوز الحد الأقصى حتى الآن.
!>           يتم تحديثه عند الخروج إذا فشلت أي من النسب في RESULT أيضًا.
!>
