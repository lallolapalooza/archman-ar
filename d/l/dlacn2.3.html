توثيق الوظيفة/الروتين الفرعي

subroutine dlacn2 (integer n, double precision, dimension( * ) v, double precision, dimension( * ) x, integer, dimension( * ) isgn, double precision est, integer kase, integer, dimension( 3 ) isave)

DLACN2 تقدر قيمة المعيار 1 لمصفوفة مربعة حقيقية، باستخدام الاتصال العكسي لتقييم نواتج ضرب المصفوفة والمتجه.

الغرض:

!>
!> DLACN2 يقدر قيمة المعيار 1 لمصفوفة مربعة حقيقية A.
!> يتم استخدام الاتصال العكسي لتقييم نواتج ضرب المصفوفة والمتجه.
!>

المعلمات

N 
!>          N هو عدد صحيح (INTEGER)
!>         مرتبة المصفوفة.  N >= 1.
!>

V

!>          V هي مصفوفة من نوع double precision، حجمها (N)
!>         عند العودة النهائية، V = A*W، حيث EST = norm(V)/norm(W)
!>         (W لا يتم إرجاعه).
!>

X

!>          X هي مصفوفة من نوع double precision، حجمها (N)
!>         عند العودة الوسيطة، يجب استبدال X بـ
!>               A * X،   إذا كان KASE=1،
!>               A**T * X،  إذا كان KASE=2،
!>         ويجب إعادة استدعاء DLACN2 بجميع المعلمات الأخرى
!>         دون تغيير.
!>

ISGN

!>          ISGN هي مصفوفة أعداد صحيحة (INTEGER)، حجمها (N)
!>

EST

!>          EST هو double precision
!>         عند الدخول مع KASE = 1 أو 2 و ISAVE(1) = 3، يجب أن يكون EST
!>         دون تغيير عن الاستدعاء السابق لـ DLACN2.
!>         عند الخروج، EST هو تقدير (حد أدنى) لـ norm(A).
!>

KASE

!>          KASE هو عدد صحيح (INTEGER)
!>         عند الاستدعاء الأولي لـ DLACN2، يجب أن يكون KASE = 0.
!>         عند العودة الوسيطة، سيكون KASE = 1 أو 2، مما يشير
!>         إلى ما إذا كان يجب استبدال X بـ A * X  أو A**T * X.
!>         عند العودة النهائية من DLACN2، سيكون KASE = 0 مرة أخرى.
!>

ISAVE

!>          ISAVE هي مصفوفة أعداد صحيحة (INTEGER)، حجمها (3)
!>         تستخدم ISAVE لحفظ المتغيرات بين استدعاءات DLACN2
!>
