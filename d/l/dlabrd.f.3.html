الاسم

dlabrd - وظيفة subroutin لتصغير مصفوفة متناظرة شبه متعامدة.

الملخص

```fortran SUBROUTINE DLABRD( RCM, L, N, NB, A, LDA, D, E, TAUQ, TAUP, X, LDX, Y, LDY ) DOUBLE PRECISION A(LDA, *), D( * ), E( * ), TAUQ( * ), TAUP( * ), X(LDX, * ), Y(LDY, *) INTEGER L, N, NB, LDA, LDX, LDY LOGICAL RCM ```

الوصف

DLABRD تقوم بتصغير مصفوفة متناظرة شبه متعامدة A، مع الاهتمام بتعريفها (U^T A V) = D + T، حيث U و V مصفوفتان متعامدتان، و D مصفوفة قطرية، و T مصفوفة مثلثة علوية.

تقوم DLABRD بتقليص الجزء العلوي من A إلى شكل شبه متعامد، مع الاحتفاظ بالجزء السفلي من A.

الوسائط

RCM (IN)

LOGICAL. إذا كانت RCM .TRUE.، فيتم حساب U و V بحيث U^T A V = D + T. إذا كانت RCM .FALSE.، فيتم حساب U و V بحيث V^T A U = D + T.

L (IN)

INTEGER. عدد الصفوف التي يتم الاحتفاظ بها في الجزء السفلي من A. L >= 0.

N (IN)

INTEGER. ترتيب المصفوفة A. N >= 0.

NB (IN)

INTEGER. عرض المصفوفة الشبه متعامدة (block width). NB >= 0.

A (INOUT)

DOUBLE PRECISION A(LDA, *). على الدخل، المصفوفة الأصلية. على الخرج، تم استبدال الجزء العلوي من A بـ D و T.

LDA (IN)

INTEGER. درجة الصف للمصفوفة A. LDA >= max(1, N).

D (OUT)

DOUBLE PRECISION D( * ). على الخرج، المصفوفة القطرية D.

E (OUT)

DOUBLE PRECISION E( * ). على الخرج، المصفوفة الشبه قطرية E. E(i) تحتوي على العنصر (i, i+1) من T.

TAUQ (OUT)

DOUBLE PRECISION TAUQ( * ). على الخرج، المتجهات الموجهة (vectors) التي تولد U.

TAUP (OUT)

DOUBLE PRECISION TAUP( * ). على الخرج، المتجهات الموجهة (vectors) التي تولد V.

X (OUT)

DOUBLE PRECISION X(LDX, *). على الخرج، مصفوفة U.

LDX (IN)

INTEGER. درجة الصف للمصفوفة X. LDX >= max(1, N).

Y (OUT)

DOUBLE PRECISION Y(LDY, *). على الخرج، مصفوفة V.

LDY (IN)

INTEGER. درجة الصف للمصفوفة Y. LDY >= max(1, N).

التفاصيل

تم تطوير هذه الخوارزمية على أساس الخوارزميات المستخدمة في LAPACK.

ملاحظة

هذه الدالة هي جزء من مكتبة LAPACK.

أمثلة

مثال على استخدام DLABRD: 

      PROGRAM EXAMPLE
      IMPLICIT NONE
      INTEGER, PARAMETER :: N = 5
      INTEGER, PARAMETER :: LDA = N, LDX = N, LDY = N
      INTEGER, PARAMETER :: L = 2, NB = 2
      DOUBLE PRECISION A(LDA,N), D(N), E(N-1), TAUQ(N), TAUP(N)
      DOUBLE PRECISION X(LDX,N), Y(LDY,N)
      LOGICAL RCM
      INTEGER I, J

      ! Initialize A
      DO J = 1, N
         DO I = 1, N
            A(I,J) = DBLE( I + J )
         END DO
      END DO

      RCM = .TRUE.

      CALL DLABRD(RCM, L, N, NB, A, LDA, D, E, TAUQ, TAUP, X, LDX, Y, LDY)

      PRINT *, "Diagonal matrix D:"
      DO I = 1, N
         PRINT *, D(I)
      END DO

      PRINT *, "Superdiagonal matrix E:"
      DO I = 1, N-1
         PRINT *, E(I)
      END DO

      PRINT *, "Matrix U:"
      DO I = 1, N
         DO J = 1, N
            PRINT *, X(I,J)
         END DO
      END DO

      PRINT *, "Matrix V:"
      DO I = 1, N
         DO J = 1, N
            PRINT *, Y(I,J)
         END DO
      END DO

      END PROGRAM EXAMPLE
