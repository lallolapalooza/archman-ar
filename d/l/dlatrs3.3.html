subroutine dlatrs3 (character uplo, character trans, character diag, character normin, integer n, integer nrhs, double precision, dimension( lda, * ) a, integer lda, double precision, dimension( ldx, * ) x, integer ldx, double precision, dimension( * ) scale, double precision, dimension( * ) cnorm, double precision, dimension( * ) work, integer lwork, integer info)

DLATRS3 يحل نظام معادلات مثلثي مع عوامل القياس المعينة لمنع الفائض.

الغرض:

!>
!> DLATRS3 يحل أحد الأنظمة المثلثية
!>
!>    A * X = B * diag(scale)  أو  A**T * X = B * diag(scale)
!>
!> مع القياس لمنع الفائض. هنا A هي مصفوفة علوية أو سفلية مثلثية، A**T ترمز إلى منقول A. X و B هي
!> مصفوفتان n × nrhs ومتجه scale مكون من nrhs عامل قياس. عادة ما يكون عامل القياس scale(j) أصغر من أو
!> يساوي 1، مختار بحيث يكون X(:,j) أصغر من حد الفائض.
!> إذا كانت المصفوفة A مفردة (A(j,j) = 0 لبعض j)، فسيتم إرجاع حل غير تافه لـ A*X = 0. إذا كان النظام
!> مقاسًا بشكل سيء لدرجة أن الحل لا يمكن تمثيله كـ
!> (1/scale(k))*X(:,k)، فسيتم إرجاع x(:,k) = 0 و scale(k).
!>
!> هذه نسخة BLAS-3 من LATRS لحل عدة أطراف يمنى في وقت واحد.
!>
!>

المعلمات

UPLO 
!>          UPLO هو CHARACTER*1
!>          يحدد ما إذا كانت المصفوفة A علوية أم سفلية مثلثية.
!>          = 'U':  علوية مثلثية
!>          = 'L':  سفلية مثلثية
!>

TRANS

!>          TRANS هو CHARACTER*1
!>          يحدد العملية المطبقة على A.
!>          = 'N':  حل A * x = s*b  (بدون تبديل)
!>          = 'T':  حل A**T* x = s*b  (تبديل)
!>          = 'C':  حل A**T* x = s*b  (تبديل مرافق = تبديل)
!>

DIAG

!>          DIAG هو CHARACTER*1
!>          يحدد ما إذا كانت المصفوفة A وحدة مثلثية أم لا.
!>          = 'N':  غير وحدة مثلثية
!>          = 'U':  وحدة مثلثية
!>

NORMIN

!>          NORMIN هو CHARACTER*1
!>          يحدد ما إذا كان CNORM قد تم تعيينه أم لا.
!>          = 'Y':  يحتوي CNORM على معايير الأعمدة عند الدخول
!>          = 'N':  لم يتم تعيين CNORM عند الدخول. عند الخروج، سيتم
!>                  حساب المعايير وتخزينها في CNORM.
!>

N

!>          N هو INTEGER
!>          ترتيب المصفوفة A.  N >= 0.
!>

NRHS

!>          NRHS هو INTEGER
!>          عدد أعمدة X.  NRHS >= 0.
!>

A

!>          A هو مصفوفة DOUBLE PRECISION، البعد (LDA,N)
!>          المصفوفة المثلثية A.  إذا كان UPLO = 'U'، فإن الجزء العلوي المثلثي من n × n
!>          الرائد في المصفوفة A يحتوي على المصفوفة المثلثية العلوية، والجزء السفلي المثلثي الصارم
!>          من A لا تتم الإشارة إليه. إذا كان UPLO = 'L'، فإن الجزء السفلي المثلثي من n × n
!>          الرائد في المصفوفة A يحتوي على المصفوفة المثلثية السفلية، والجزء العلوي المثلثي الصارم من A لا تتم
!>          الإشارة إليه. إذا كان DIAG = 'U'، فإن العناصر القطرية لـ A
!>          لا تتم الإشارة إليها أيضًا وتُفترض أنها 1.
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرائد للمصفوفة A.  LDA >= max (1,N).
!>

X

!>          X هي مصفوفة DOUBLE PRECISION، البعد (LDX,NRHS)
!>          عند الدخول، الطرف الأيمن B للنظام المثلثي.
!>          عند الخروج، يتم استبدال X بمصفوفة الحل X.
!>

LDX

!>          LDX هو INTEGER
!>          البعد الرائد للمصفوفة X.  LDX >= max (1,N).
!>

SCALE

!>          SCALE هي مصفوفة DOUBLE PRECISION، البعد (NRHS)
!>          عامل القياس s(k) للنظام المثلثي
!>          A * x(:,k) = s(k)*b(:,k)  أو  A**T* x(:,k) = s(k)*b(:,k).
!>          إذا كان SCALE = 0، فإن المصفوفة A مفردة أو مقاسة بشكل سيء.
!>          إذا تم مواجهة A(j,j) = 0، فسيتم إرجاع متجه غير تافه x(:,k)
!>          وهو حل دقيق أو تقريبي لـ A*x(:,k) = 0. إذا كان النظام
!>          مقاسًا بشكل سيء لدرجة أن الحل لا يمكن تقديمه كـ x(:,k) * 1/s(k)، فسيتم إرجاع x(:,k) = 0.
!>

CNORM

!>          CNORM هي مصفوفة DOUBLE PRECISION، البعد (N)
!>
!>          إذا كان NORMIN = 'Y'، فإن CNORM هو معلمة إدخال و CNORM(j)
!>          يحتوي على معيار الجزء غير القطري للعمود j
!>          من A. إذا كان TRANS = 'N'، يجب أن يكون CNORM(j) أكبر من أو يساوي
!>          معيار اللانهاية، وإذا كان TRANS = 'T' أو 'C'، يجب أن يكون CNORM(j)
!>          أكبر من أو يساوي معيار 1.
!>
!>          إذا كان NORMIN = 'N'، فإن CNORM هو معلمة إخراج و CNORM(j)
!>          يعيد معيار 1 للجزء غير القطري من العمود j
!>          من A.
!>

WORK

!>          WORK هي مصفوفة DOUBLE PRECISION، البعد (MAX(1,LWORK)).
!>          عند الخروج، إذا كان INFO = 0، فإن WORK(1) يعيد الحجم الأمثل
!>          لـ WORK.
!>

LWORK

!>          LWORK هو INTEGER
!>          بعد المصفوفة WORK.
!>
!>          إذا كان MIN(N,NRHS) = 0، فإن LWORK >= 1، وإلا
!>          LWORK >= MAX(1, 2*NBA * MAX(NBA, MIN(NRHS, 32))، حيث
!>          NBA = (N + NB - 1)/NB و NB هو حجم الكتلة الأمثل.
!>
!>          إذا كان LWORK = -1، فسيتم افتراض استعلام مساحة العمل؛ يقوم الروتين
!>          بحساب الأبعاد المثلى للمصفوفة WORK فقط، ويعيد
!>          هذه القيمة كإدخال أول للمصفوفة WORK، ولا يصدر XERBLA أي رسالة خطأ
!>          متعلقة بـ LWORK.
!>

INFO

!>          INFO هو INTEGER
!>          = 0:  انتهاء ناجح
!>          < 0:  إذا كان INFO = -k، فإن المعلمة k كان لها قيمة غير قانونية
!>
