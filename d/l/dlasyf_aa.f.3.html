subroutine dlasyf_aa (character uplo, integer j1, integer m, integer nb, double precision, dimension( lda, * ) a, integer lda, integer, dimension( * ) ipiv, double precision, dimension( ldh, * ) h, integer ldh, double precision, dimension( * ) work)

DLASYF_AA

الغرض:

!>
!> DLATRF_AA يحلل لوحة من مصفوفة متماثلة حقيقية A باستخدام
!> خوارزمية Aasen. تتكون اللوحة من مجموعة من صفوف NB من A
!> عندما يكون UPLO هو U، أو مجموعة من أعمدة NB عندما يكون UPLO هو L.
!>
!> لتحليل اللوحة، تتطلب خوارزمية Aasen الصف الأخير، أو العمود،
!> من اللوحة السابقة. يتم تعيين الصف الأول، أو العمود، من A ليكون الصف الأول، أو العمود،
!> من مصفوفة الوحدة،
!> والتي تستخدم لتحليل اللوحة الأولى.
!>
!> يتم تخزين الصف J من U، أو العمود J من L، الناتج في
!> الصف (J-1)، أو العمود، من A (بدون الأقطار الأحادية)، بينما
!> يتم الكتابة فوق القطر الفرعي والقطري لـ A بتلك الخاصة بـ T.
!>
!>

المعلمات

UPLO 
!>          UPLO هو CHARACTER*1
!>          = 'U':  يتم تخزين الجزء العلوي من A؛
!>          = 'L':  يتم تخزين الجزء السفلي من A.
!>

J1

!>          J1 هو INTEGER
!>          موقع الصف الأول، أو العمود، للوحة
!>          ضمن المصفوفة الفرعية لـ A، الممررة إلى هذا الروتين، على سبيل المثال،
!>          عند الاستدعاء بواسطة DSYTRF_AA، بالنسبة للوحة الأولى، J1 هو 1،
!>          بينما بالنسبة للوحات المتبقية، J1 هو 2.
!>

M

!>          M هو INTEGER
!>          بعد المصفوفة الفرعية. M >= 0.
!>

NB

!>          NB هو INTEGER
!>          بعد اللوحة المراد تحليلها.
!>

A

!>          A هو مصفوفة DOUBLE PRECISION، البعد (LDA,M) لـ
!>          اللوحة الأولى، بينما البعد (LDA,M+1) لـ
!>          اللوحات المتبقية.
!>
!>          عند الدخول، تحتوي A على الصف الأخير، أو العمود، من
!>          اللوحة السابقة، والمصفوفة الفرعية اللاحقة لـ A
!>          المراد تحليلها، باستثناء اللوحة الأولى، فقط
!>          يتم تمرير اللوحة.
!>
!>          عند الخروج، يتم تحليل اللوحة الرائدة.
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرائد للمصفوفة A.  LDA >= max(1,M).
!>

IPIV

!>          IPIV هو مصفوفة INTEGER، البعد (M)
!>          تفاصيل تبادلات الصفوف والأعمدة،
!>          تم تبديل الصف والعمود k مع الصف والعمود IPIV(k).
!>

H

!>          H هي مساحة عمل DOUBLE PRECISION، البعد (LDH,NB).
!>
!>

LDH

!>          LDH هو INTEGER
!>          البعد الرائد لمساحة العمل H. LDH >= max(1,M).
!>

WORK

!>          WORK هي مساحة عمل DOUBLE PRECISION، البعد (M).
!>
