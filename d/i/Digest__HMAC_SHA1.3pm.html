Digest__HMAC_SHA1(3pm)

الحزمة: perl-digest-hmac

التراخيص: PerlArtistic

Digest::HMAC_SHA1(3) وثائق Perl المساهمة من المستخدم Digest::HMAC_SHA1(3)

الاسم

Digest::HMAC_SHA1 - التجزئة المفتاحية لمصادقة الرسائل

الملخص

# أسلوب وظيفي
use Digest::HMAC_SHA1 qw(hmac_sha1 hmac_sha1_hex);
$digest = hmac_sha1($data, $key);
print hmac_sha1_hex($data, $key);
# أسلوب الكائنات
use Digest::HMAC_SHA1;
$hmac = Digest::HMAC_SHA1->new($key);
$hmac->add($data);
$hmac->addfile(*FILE);
$digest = $hmac->digest;
$digest = $hmac->hexdigest;
$digest = $hmac->b64digest;

الوصف

توفر هذه الوحدة التجزئة HMAC-SHA-1.

انظر أيضا

Digest::HMAC, Digest::SHA, Digest::HMAC_MD5
