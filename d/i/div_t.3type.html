div_t(3type) div_t(3type)

الاسم

div_t, ldiv_t, lldiv_t, imaxdiv_t - ناتج وحاصل قسمة عدد صحيح

المكتبة

مكتبة C القياسية (libc)

الملخص

#include <stdlib.h>
typedef struct {
    int quot; /* ناتج القسمة */
    int rem;  /* الباقي */
} div_t;
typedef struct {
    long quot; /* ناتج القسمة */
    long rem;  /* الباقي */
} ldiv_t;
typedef struct {
    long long quot; /* ناتج القسمة */
    long long rem;  /* الباقي */
} lldiv_t;
#include <inttypes.h>
typedef struct {
    intmax_t quot; /* ناتج القسمة */
    intmax_t rem;  /* الباقي */
} imaxdiv_t;

الوصف

[[l]l]div_t هو نوع القيمة التي يتم إرجاعها بواسطة الدالة [[l]l]div(3).

imaxdiv_t هو نوع القيمة التي يتم إرجاعها بواسطة الدالة imaxdiv(3).
