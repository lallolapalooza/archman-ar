subroutine drscl (integer n, double precision sa, double precision, dimension( * ) sx, integer incx)

DRSCL يضرب متجهًا في مقلوب ثابت حقيقي.

الغرض:

!>
!> DRSCL يضرب متجهًا حقيقيًا مكونًا من N عنصرًا x في الثابت الحقيقي 1/a.
!> يتم ذلك بدون تجاوز أو نقص في الأرقام طالما أن
!> النتيجة النهائية x/a لا تتجاوز أو تنقص في الأرقام.
!>

المعلمات

N 
!>          N هو عدد صحيح
!>          عدد مكونات المتجه x.
!>

SA

!>          SA هو دقة مزدوجة
!>          الثابت a الذي يستخدم لقسمة كل مكون من مكونات x.
!>          يجب أن يكون SA >= 0، وإلا سيقوم الروتين الفرعي بالقسمة على صفر.
!>

SX

!>          SX هو مصفوفة دقة مزدوجة، أبعادها
!>                         (1+(N-1)*abs(INCX))
!>          المتجه x المكون من N عنصرًا.
!>

INCX

!>          INCX هو عدد صحيح
!>          الزيادة بين القيم المتتالية للمتجه SX.
!>          > 0:  SX(1) = X(1) and SX(1+(i-1)*INCX) = x(i),     1< i<= n
!>
