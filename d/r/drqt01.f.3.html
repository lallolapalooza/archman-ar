توثيق الوظيفة/الإجراء الفرعي

subroutine drqt01 (integer m, integer n, double precision, dimension( lda, * ) a, double precision, dimension( lda, * ) af, double precision, dimension( lda, * ) q, double precision, dimension( lda, * ) r, integer lda, double precision, dimension( * ) tau, double precision, dimension( lwork ) work, integer lwork, double precision, dimension( * ) rwork, double precision, dimension( * ) result)

DRQT01

الغرض:

!>
!> DRQT01 تختبر DGERQF، التي تحسب تحليل RQ لمصفوفة m-by-n
!> A، وتختبر جزئياً DORGRQ التي تشكل المصفوفة المتعامدة n-by-n
!> Q.
!>
!> DRQT01 تقارن R مع A*Q'، وتتحقق من أن Q متعامدة.
!>

المعلمات

M 
!>          M هو عدد صحيح (INTEGER)
!>          عدد صفوف المصفوفة A.  M >= 0.
!>

N

!>          N هو عدد صحيح (INTEGER)
!>          عدد أعمدة المصفوفة A.  N >= 0.
!>

A

!>          A هي مصفوفة دقة مزدوجة (DOUBLE PRECISION)، ذات بعد (LDA,N)
!>          المصفوفة A ذات الأبعاد m-by-n.
!>

AF

!>          AF هي مصفوفة دقة مزدوجة (DOUBLE PRECISION)، ذات بعد (LDA,N)
!>          تفاصيل تحليل RQ للمصفوفة A، كما تم إرجاعها بواسطة DGERQF.
!>          راجع DGERQF لمزيد من التفاصيل.
!>

Q

!>          Q هي مصفوفة دقة مزدوجة (DOUBLE PRECISION)، ذات بعد (LDA,N)
!>          المصفوفة المتعامدة Q ذات الأبعاد n-by-n.
!>

R

!>          R هي مصفوفة دقة مزدوجة (DOUBLE PRECISION)، ذات بعد (LDA,max(M,N))
!>

LDA

!>          LDA هو عدد صحيح (INTEGER)
!>          البعد الرئيسي للمصفوفات A, AF, Q, و L.
!>          LDA >= max(M,N).
!>

TAU

!>          TAU هي مصفوفة دقة مزدوجة (DOUBLE PRECISION)، ذات بعد (min(M,N))
!>          العوامل العددية للمُسَلِّمات الأولية، كما تم إرجاعها
!>          بواسطة DGERQF.
!>

WORK

!>          WORK هي مصفوفة دقة مزدوجة (DOUBLE PRECISION)، ذات بعد (LWORK)
!>

LWORK

!>          LWORK هو عدد صحيح (INTEGER)
!>          بعد المصفوفة WORK.
!>

RWORK

!>          RWORK هي مصفوفة دقة مزدوجة (DOUBLE PRECISION)، ذات بعد (max(M,N))
!>

RESULT

!>          RESULT هي مصفوفة دقة مزدوجة (DOUBLE PRECISION)، ذات بعد (2)
!>          نسب الاختبار:
!>          RESULT(1) = norm( R - A*Q' ) / ( N * norm(A) * EPS )
!>          RESULT(2) = norm( I - Q*Q' ) / ( N * EPS )
!>
