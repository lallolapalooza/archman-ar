توثيق الوظيفة/الروتين الفرعي

subroutine drqt03 (integer m, integer n, integer k, double precision, dimension( lda, * ) af, double precision, dimension( lda, * ) c, double precision, dimension( lda, * ) cc, double precision, dimension( lda, * ) q, integer lda, double precision, dimension( * ) tau, double precision, dimension( lwork ) work, integer lwork, double precision, dimension( * ) rwork, double precision, dimension( * ) result)

DRQT03

الغرض:

!>
!> DRQT03 تختبر DORMRQ، التي تحسب Q*C، Q'*C، C*Q أو C*Q'.
!>
!> تقارن DRQT03 نتائج استدعاء DORMRQ مع نتائج
!> تكوين Q بشكل صريح عن طريق استدعاء DORGRQ ثم إجراء ضرب المصفوفات
!> عن طريق استدعاء DGEMM.
!>

المعلمات

M 
!>          M هو عدد صحيح (INTEGER)
!>          عدد الصفوف أو الأعمدة للمصفوفة C؛ C هي n*m إذا
!>          تم تطبيق Q من اليسار، أو m*n إذا
!>          تم تطبيق Q من اليمين.  M >= 0.
!>

N

!>          N هو عدد صحيح (INTEGER)
!>          رتبة المصفوفة المتعامدة Q.  N >= 0.
!>

K

!>          K هو عدد صحيح (INTEGER)
!>          عدد الانعكاسات الأولية التي يشكل حاصل ضربها المصفوفة المتعامدة Q.
!>          N >= K >= 0.
!>

AF

!>          AF هي مصفوفة (DOUBLE PRECISION array)، البعد (LDA,N)
!>          تفاصيل تحليل RQ للمصفوفة m*n، كما
!>          تعود من DGERQF. انظر SGERQF لمزيد من التفاصيل.
!>

C

!>          C هي مصفوفة (DOUBLE PRECISION array)، البعد (LDA,N)
!>

CC

!>          CC هي مصفوفة (DOUBLE PRECISION array)، البعد (LDA,N)
!>

Q

!>          Q هي مصفوفة (DOUBLE PRECISION array)، البعد (LDA,N)
!>

LDA

!>          LDA هو عدد صحيح (INTEGER)
!>          البعد الرئيسي للمصفوفات AF، C، CC، و Q.
!>

TAU

!>          TAU هي مصفوفة (DOUBLE PRECISION array)، البعد (min(M,N))
!>          العوامل القياسية للانعكاسات الأولية المقابلة
!>          لتحليل RQ في AF.
!>

WORK

!>          WORK هي مصفوفة (DOUBLE PRECISION array)، البعد (LWORK)
!>

LWORK

!>          LWORK هو عدد صحيح (INTEGER)
!>          طول WORK.  يجب أن يكون LWORK على الأقل M، ويجب أن يكون
!>          M*NB، حيث NB هو حجم الكتلة لهذه البيئة.
!>

RWORK

!>          RWORK هي مصفوفة (DOUBLE PRECISION array)، البعد (M)
!>

RESULT

!>          RESULT هي مصفوفة (DOUBLE PRECISION array)، البعد (4)
!>          تقارن نسب الاختبار تقنيتين لضرب مصفوفة عشوائية C
!>          بمصفوفة متعامدة n*n Q.
!>          RESULT(1) = norm( Q*C - Q*C )  / ( N * norm(C) * EPS )
!>          RESULT(2) = norm( C*Q - C*Q )  / ( N * norm(C) * EPS )
!>          RESULT(3) = norm( Q'*C - Q'*C )/ ( N * norm(C) * EPS )
!>          RESULT(4) = norm( C*Q' - C*Q' )/ ( N * norm(C) * EPS )
!>
