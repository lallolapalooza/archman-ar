subroutine drotm (integer n, double precision, dimension(*) dx, integer incx, double precision, dimension(*) dy, integer incy, double precision, dimension(5) dparam)

DROTM

الغرض:

!>
!>    تطبيق تحويل جيفنز المعدل، H، على المصفوفة 2 × N
!>
!>    (DX**T) ، حيث **T يعني النقل (transpose). عناصر DX موجودة في
!>    (DY**T)
!>
!>    DX(LX+I*INCX)، I = 0 إلى N-1، حيث LX = 1 إذا كان INCX .GE. 0، وإلا
!>    LX = (-INCX)*N، وبالمثل لـ SY باستخدام LY و INCY.
!>    مع DPARAM(1)=DFLAG، تمتلك H أحد الأشكال التالية..
!>
!>    DFLAG=-1.D0     DFLAG=0.D0        DFLAG=1.D0     DFLAG=-2.D0
!>
!>      (DH11  DH12)    (1.D0  DH12)    (DH11  1.D0)    (1.D0  0.D0)
!>    H=(          )    (          )    (          )    (          )
!>      (DH21  DH22)،   (DH21  1.D0)،   (-1.D0 DH22)،   (0.D0  1.D0).
!>    انظر DROTMG لوصف تخزين البيانات في DPARAM.
!>

المعلمات

N 
!>          N هو عدد صحيح (INTEGER)
!>         عدد العناصر في المتجه (المتجهات) المدخلة
!>

DX

!>          DX هو مصفوفة مزدوجة الدقة (DOUBLE PRECISION)، بأبعاد ( 1 + ( N - 1 )*abs( INCX ) )
!>

INCX

!>          INCX هو عدد صحيح (INTEGER)
!>         التباعد في التخزين بين عناصر DX
!>

DY

!>          DY هو مصفوفة مزدوجة الدقة (DOUBLE PRECISION)، بأبعاد ( 1 + ( N - 1 )*abs( INCY ) )
!>

INCY

!>          INCY هو عدد صحيح (INTEGER)
!>         التباعد في التخزين بين عناصر DY
!>

DPARAM

!>          DPARAM هي مصفوفة مزدوجة الدقة (DOUBLE PRECISION)، بأبعاد (5)
!>     DPARAM(1)=DFLAG
!>     DPARAM(2)=DH11
!>     DPARAM(3)=DH21
!>     DPARAM(4)=DH12
!>     DPARAM(5)=DH22
!>
