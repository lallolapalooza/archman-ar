subroutine drscl (integer n, double precision sa, double precision, dimension( * ) sx, integer incx)

DRSCL يضرب متجهًا في مقلوب عدد قياسي حقيقي.

الغرض:

!>
!> DRSCL يضرب متجهًا حقيقيًا مكونًا من N عنصرًا x بالعدد القياسي الحقيقي 1/a.
!> يتم ذلك دون تجاوز السعة أو نقص السعة طالما أن
!> النتيجة النهائية x/a لا تتجاوز السعة أو تنقص منها.
!>

المعاملات

N 
!>          N هو عدد صحيح (INTEGER)
!>          عدد مكونات المتجه x.
!>

SA

!>          SA هو دقة مزدوجة (DOUBLE PRECISION)
!>          العدد القياسي a الذي يستخدم لتقسيم كل مكون من مكونات x.
!>          يجب أن يكون SA >= 0، وإلا فإن الإجراء الفرعي سيقسم على صفر.
!>

SX

!>          SX هو مصفوفة بدقة مزدوجة (DOUBLE PRECISION)، أبعادها
!>                         (1+(N-1)*abs(INCX))
!>          المتجه x المكون من N عنصرًا.
!>

INCX

!>          INCX هو عدد صحيح (INTEGER)
!>          الزيادة بين القيم المتتالية للمتجه SX.
!>          > 0:  SX(1) = X(1) و SX(1+(i-1)*INCX) = x(i)،     1< i<= n
!>
