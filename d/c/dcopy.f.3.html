subroutine dcopy (integer n, double precision, dimension(*) dx, integer incx, double precision, dimension(*) dy, integer incy)

DCOPY

الغرض:

!>
!>    DCOPY يقوم بنسخ متجه، x، إلى متجه، y.
!>    يستخدم حلقات غير ملولبة للأبعاد المتزايدة التي تساوي 1.
!>

المعلمات

N 
!>          N هو عدد صحيح (INTEGER)
!>         عدد العناصر في المتجه (المتجهات) المدخلة
!>

DX

!>          DX هو مصفوفة بدقة مزدوجة (DOUBLE PRECISION array)، أبعادها ( 1 + ( N - 1 )*abs( INCX ) )
!>

INCX

!>          INCX هو عدد صحيح (INTEGER)
!>         تباعد التخزين بين عناصر DX
!>

DY

!>          DY هو مصفوفة بدقة مزدوجة (DOUBLE PRECISION array)، أبعادها ( 1 + ( N - 1 )*abs( INCY ) )
!>

INCY

!>          INCY هو عدد صحيح (INTEGER)
!>         تباعد التخزين بين عناصر DY
!>
