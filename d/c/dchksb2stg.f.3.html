subroutine dchksb2stg (integer nsizes, integer, dimension( * ) nn, integer nwdths, integer, dimension( * ) kk, integer ntypes, logical, dimension( * ) dotype, integer, dimension( 4 ) iseed, double precision thresh, integer nounit, double precision, dimension( lda, * ) a, integer lda, double precision, dimension( * ) sd, double precision, dimension( * ) se, double precision, dimension( * ) d1, double precision, dimension( * ) d2, double precision, dimension( * ) d3, double precision, dimension( ldu, * ) u, integer ldu, double precision, dimension( * ) work, integer lwork, double precision, dimension( * ) result, integer info)

DCHKSB2STG

الغرض:

!>
!> DCHKSB2STG تختبر تقليل مصفوفة شريطية متناظرة إلى صيغة ثلاثية القطر،
!> المستخدمة مع مشكلة القيم الذاتية المتناظرة.
!>
!> DSBTRD تقوم بتحليل مصفوفة شريطية متناظرة A إلى A = U S U' ، حيث ' تعني
!> منقول، S متناظرة ثلاثية الأقطار، و U متعامدة.
!> يمكن لـ DSBTRD استخدام إما الجزء السفلي أو الجزء العلوي فقط
!> من A؛ DCHKSB2STG تتحقق من كلتا الحالتين.
!>
!> DSYTRD_SB2ST تقوم بتحليل مصفوفة شريطية متناظرة A إلى A = U S U' ،
!> حيث ' تعني منقول، S متناظرة ثلاثية الأقطار، و U متعامدة.
!> يمكن لـ DSYTRD_SB2ST استخدام إما الجزء السفلي أو الجزء العلوي فقط
!> من A؛ DCHKSB2STG تتحقق من كلتا الحالتين.
!>
!> DSTEQR تقوم بتحليل S إلى S = Z D1 Z'.
!> D1 هي مصفوفة القيم الذاتية المحسوبة عندما لا يتم حساب Z
!> ومن S الناتجة من DSBTRD (تستخدم كمرجع لـ DSYTRD_SB2ST)
!> D2 هي مصفوفة القيم الذاتية المحسوبة عندما لا يتم حساب Z
!> ومن S الناتجة من DSYTRD_SB2ST .
!> D3 هي مصفوفة القيم الذاتية المحسوبة عندما لا يتم حساب Z
!> ومن S الناتجة من DSYTRD_SB2ST .
!>
!> عندما يتم استدعاء DCHKSB2STG، يتم تحديد عدد من أحجام المصفوفات (NN)، وعدد
!> من عرض النطاقات (KK)، وعدد من أنواع المصفوفات (DOTYPE).
!> لكل حجم (NN(j))، ولكل عرض نطاق (KK(k)) أقل من أو
!> يساوي الحجم، ولكل نوع مصفوفة، سيتم إنشاء مصفوفة واحدة
!> واستخدامها لاختبار الروتين الخاص بتقليل النطاقات المتناظرة. لكل
!> مصفوفة، سيتم إجراء عدد من الاختبارات:
!>
!> (1)     | A - V S V' | / ( |A| n ulp )  محسوبة بواسطة DSBTRD مع
!>                                         UPLO='U'
!>
!> (2)     | I - UU' | / ( n ulp )
!>
!> (3)     | A - V S V' | / ( |A| n ulp )  محسوبة بواسطة DSBTRD مع
!>                                         UPLO='L'
!>
!> (4)     | I - UU' | / ( n ulp )
!>
!> (5)     | D1 - D2 | / ( |D1| ulp )      حيث D1 محسوبة بواسطة
!>                                         DSBTRD مع UPLO='U' و
!>                                         D2 محسوبة بواسطة
!>                                         DSYTRD_SB2ST مع UPLO='U'
!>
!> (6)     | D1 - D3 | / ( |D1| ulp )      حيث D1 محسوبة بواسطة
!>                                         DSBTRD مع UPLO='U' و
!>                                         D3 محسوبة بواسطة
!>                                         DSYTRD_SB2ST مع UPLO='L'
!>
!> يتم تحديد أحجام المصفوفات بواسطة مصفوفة NN(1:NSIZES)؛ قيمة
!> كل عنصر NN(j) تحدد حجمًا واحدًا.
!> يتم تحديد أنواع المصفوفات بواسطة مصفوفة منطقية DOTYPE( 1:NTYPES )؛
!> إذا كان DOTYPE(j) هو .TRUE.، فسيتم إنشاء مصفوفة من النوع j.
!> حاليًا، قائمة الأنواع الممكنة هي:
!>
!> (1)  المصفوفة الصفرية.
!> (2)  المصفوفة المحايدة.
!>
!> (3)  مصفوفة قطرية ذات مدخلات متباعدة بانتظام
!>      1, ..., ULP  وعلامات عشوائية.
!>      (ULP = (أول رقم أكبر من 1) - 1)
!> (4)  مصفوفة قطرية ذات مدخلات متباعدة هندسيًا
!>      1, ..., ULP  وعلامات عشوائية.
!> (5)  مصفوفة قطرية ذات مدخلات  1, ULP, ..., ULP
!>      وعلامات عشوائية.
!>
!> (6)  نفس (4)، ولكن مضروبة في مربع جذر (حد الفيضان).
!> (7)  نفس (4)، ولكن مضروبة في مربع جذر (حد التدفق السفلي).
!>
!> (8)  مصفوفة من الشكل U' D U، حيث U متعامدة و
!>      D لها مدخلات متباعدة بانتظام 1, ..., ULP مع علامات عشوائية
!>      على القطر.
!>
!> (9)  مصفوفة من الشكل U' D U، حيث U متعامدة و
!>      D لها مدخلات متباعدة هندسيًا 1, ..., ULP مع علامات عشوائية
!>      على القطر.
!>
!> (10) مصفوفة من الشكل U' D U، حيث U متعامدة و
!>      D لها مدخلات  1, ULP,..., ULP مع علامات
!>      عشوائية على القطر.
!>
!> (11) نفس (8)، ولكن مضروبة في مربع جذر (حد الفيضان).
!> (12) نفس (8)، ولكن مضروبة في مربع جذر (حد التدفق السفلي).
!>
!> (13) مصفوفة متناظرة ذات مدخلات عشوائية مختارة من (-1,1).
!> (14) نفس (13)، ولكن مضروبة في مربع جذر (حد الفيضان).
!> (15) نفس (13)، ولكن مضروبة في مربع جذر (حد التدفق السفلي).
!>

المعلمات

NSIZES 
!>          NSIZES هو عدد صحيح (INTEGER)
!>          عدد أحجام المصفوفات المراد استخدامها. إذا كان صفرًا،
!>          فإن DCHKSB2STG لا تفعل شيئًا. يجب أن يكون على الأقل صفرًا.
!>

NN

!>          NN هي مصفوفة أعداد صحيحة (INTEGER array)، ذات أبعاد (NSIZES)
!>          مصفوفة تحتوي على الأحجام المراد استخدامها للمصفوفات.
!>          سيتم تخطي القيم الصفرية. يجب أن تكون القيم على الأقل
!>          صفرًا.
!>

NWDTHS

!>          NWDTHS هو عدد صحيح (INTEGER)
!>          عدد عروض النطاقات المراد استخدامها. إذا كان صفرًا،
!>          فإن DCHKSB2STG لا تفعل شيئًا. يجب أن يكون على الأقل صفرًا.
!>

KK

!>          KK هي مصفوفة أعداد صحيحة (INTEGER array)، ذات أبعاد (NWDTHS)
!>          مصفوفة تحتوي على عروض النطاقات المراد استخدامها للمصفوفات
!>          الشريطية. يجب أن تكون القيم على الأقل صفرًا.
!>

NTYPES

!>          NTYPES هو عدد صحيح (INTEGER)
!>          عدد العناصر في DOTYPE. إذا كان صفرًا، فإن DCHKSB2STG
!>          لا تفعل شيئًا. يجب أن يكون على الأقل صفرًا. إذا كان MAXTYP+1
!>          و NSIZES يساوي 1، فسيتم تعريف نوع إضافي، MAXTYP+1،
!>          وهو استخدام أي مصفوفة موجودة في A. هذا
!>          مفيد فقط إذا كان DOTYPE(1:MAXTYP) هو .FALSE. و
!>          DOTYPE(MAXTYP+1) هو .TRUE. .
!>

DOTYPE

!>          DOTYPE هي مصفوفة منطقية (LOGICAL array)، ذات أبعاد (NTYPES)
!>          إذا كان DOTYPE(j) هو .TRUE.، فلكل حجم في NN سيتم إنشاء
!>          مصفوفة بهذا الحجم ومن النوع j.
!>          إذا كان NTYPES أصغر من الحد الأقصى لعدد الأنواع
!>          المعرفة (PARAMETER MAXTYP)، فلن يتم إنشاء الأنواع NTYPES+1 حتى
!>          MAXTYP. إذا كان NTYPES أكبر
!>          من MAXTYP، فسيتم تجاهل DOTYPE(MAXTYP+1) حتى DOTYPE(NTYPES).
!>

ISEED

!>          ISEED هي مصفوفة أعداد صحيحة (INTEGER array)، ذات أبعاد (4)
!>          عند الدخول، تحدد ISEED بذرة مولد الأرقام العشوائية.
!>          يجب أن تكون عناصر المصفوفة بين 0 و 4095؛
!>          إذا لم تكن كذلك، فسيتم تقليلها باستخدام باقي القسمة على 4096. أيضًا، يجب أن يكون ISEED(4)
!>          فرديًا. يستخدم مولد الأرقام العشوائية تسلسل خطي توافقي محدود بأعداد صحيحة صغيرة، وبالتالي
!>          يجب أن ينتج أرقامًا عشوائية مستقلة عن الآلة. يتم تغيير
!>          قيم ISEED عند الخروج، ويمكن استخدامها في
!>          الاستدعاء التالي لـ DCHKSB2STG لمواصلة نفس تسلسل الأرقام
!>          العشوائية.
!>

THRESH
