subroutine dckglm (integer nn, integer, dimension( * ) mval, integer, dimension( * ) pval, integer, dimension( * ) nval, integer nmats, integer, dimension( 4 ) iseed, double precision thresh, integer nmax, double precision, dimension( * ) a, double precision, dimension( * ) af, double precision, dimension( * ) b, double precision, dimension( * ) bf, double precision, dimension( * ) x, double precision, dimension( * ) work, double precision, dimension( * ) rwork, integer nin, integer nout, integer info)

DCKGLM

الغرض:

!>
!> DCKGLM تختبر DGGGLM - روتين فرعي لحل مشكلة النموذج الخطي المعمم.
!>

المعلمات

NN 
!>          NN هو عدد صحيح
!>          عدد قيم N، M و P الموجودة في المتجهات
!>          NVAL، MVAL و PVAL.
!>

MVAL

!>          MVAL هو مصفوفة أعداد صحيحة، ذات بعد (NN)
!>          قيم البعد M لأعمدة المصفوفة.
!>

PVAL

!>          PVAL هو مصفوفة أعداد صحيحة، ذات بعد (NN)
!>          قيم البعد P لأعمدة المصفوفة.
!>

NVAL

!>          NVAL هو مصفوفة أعداد صحيحة، ذات بعد (NN)
!>          قيم البعد N لصفوف المصفوفة.
!>

NMATS

!>          NMATS هو عدد صحيح
!>          عدد أنواع المصفوفات التي سيتم اختبارها لكل تركيبة
!>          من أبعاد المصفوفة. إذا كان NMATS >= NTYPES (العدد الأقصى
!>          لأنواع المصفوفات)، فسيتم إنشاء جميع الأنواع المختلفة للاختبار.
!>          إذا كان NMATS < NTYPES، تتم قراءة سطر إدخال آخر للحصول على
!>          أرقام أنواع المصفوفات التي سيتم استخدامها.
!>

ISEED

!>          ISEED هي مصفوفة أعداد صحيحة، ذات بعد (4)
!>          عند الإدخال، بذرة مولد الأرقام العشوائية. يجب أن تكون
!>          عناصر المصفوفة بين 0 و 4095، وإلا سيتم تقليلها بمعامل 4096،
!>          ويجب أن يكون ISEED(4) فرديًا.
!>          عند الخروج، البذرة التالية في تسلسل الأرقام العشوائية بعد
!>          إنشاء جميع مصفوفات الاختبار.
!>

THRESH

!>          THRESH هو دقة مزدوجة
!>          القيمة الحدية لنسب الاختبار. يتم تضمين النتيجة
!>          في ملف الإخراج إذا كان RESID >= THRESH. لطباعة
!>          كل نسبة اختبار، استخدم THRESH = 0.
!>

NMAX

!>          NMAX هو عدد صحيح
!>          القيمة القصوى المسموح بها لـ M أو N، المستخدمة في تحديد
!>          أبعاد مصفوفات العمل.
!>

A

!>          A هي مصفوفة دقة مزدوجة، ذات بعد (NMAX*NMAX)
!>

AF

!>          AF هي مصفوفة دقة مزدوجة، ذات بعد (NMAX*NMAX)
!>

B

!>          B هي مصفوفة دقة مزدوجة، ذات بعد (NMAX*NMAX)
!>

BF

!>          BF هي مصفوفة دقة مزدوجة، ذات بعد (NMAX*NMAX)
!>

X

!>          X هي مصفوفة دقة مزدوجة، ذات بعد (4*NMAX)
!>

RWORK

!>          RWORK هي مصفوفة دقة مزدوجة، ذات بعد (NMAX)
!>

WORK

!>          WORK هي مصفوفة دقة مزدوجة، ذات بعد (NMAX*NMAX)
!>

NIN

!>          NIN هو رقم وحدة للإدخال.
!>

NOUT

!>          NOUT هو رقم وحدة للإخراج.
!>

INFO

!>          INFO هو عدد صحيح
!>          = 0 : انتهاء ناجح
!>          > 0 : إذا أعاد DLATMS رمز خطأ، يتم إرجاع قيمته المطلقة.
!>
