subroutine dpbrfs (character uplo, integer n, integer kd, integer nrhs, double precision, dimension( ldab, * ) ab, integer ldab, double precision, dimension( ldafb, * ) afb, integer ldafb, double precision, dimension( ldb, * ) b, integer ldb, double precision, dimension( ldx, * ) x, integer ldx, double precision, dimension( * ) ferr, double precision, dimension( * ) berr, double precision, dimension( * ) work, integer, dimension( * ) iwork, integer info)

DPBRFS

الغرض:

!>
!> DPBRFS يحسن الحل المحسوب لنظام المعادلات الخطية
!> عندما تكون مصفوفة المعاملات متماثلة موجبة محددة
!> وذات نطاق، ويوفر حدود خطأ وتقديرات خطأ رجعي
!> للحل.
!>

المعلمات

UPLO 
!>          UPLO is CHARACTER*1
!>          = 'U':  المثلث العلوي لـ A مخزن؛
!>          = 'L':  المثلث السفلي لـ A مخزن.
!>

N

!>          N is INTEGER
!>          رتبة المصفوفة A.  N >= 0.
!>

KD

!>          KD is INTEGER
!>          عدد الأقطار الفائقة للمصفوفة A إذا كان UPLO = 'U'،
!>          أو عدد الأقطار الفرعية إذا كان UPLO = 'L'.  KD >= 0.
!>

NRHS

!>          NRHS is INTEGER
!>          عدد الجوانب اليمنى، أي عدد الأعمدة
!>          للمصفوفتين B و X.  NRHS >= 0.
!>

AB

!>          AB is DOUBLE PRECISION array, dimension (LDAB,N)
!>          المثلث العلوي أو السفلي للمصفوفة الشريطية المتماثلة A،
!>          مخزنة في أول KD+1 صفوف من المصفوفة.  العمود j
!>          للمصفوفة A مخزن في العمود j من المصفوفة AB على النحو التالي:
!>          إذا كان UPLO = 'U'، AB(kd+1+i-j,j) = A(i,j) لـ max(1,j-kd)<=i<=j؛
!>          إذا كان UPLO = 'L'، AB(1+i-j,j)    = A(i,j) لـ j<=i<=min(n,j+kd).
!>

LDAB

!>          LDAB is INTEGER
!>          البعد الرائد للمصفوفة AB.  LDAB >= KD+1.
!>

AFB

!>          AFB is DOUBLE PRECISION array, dimension (LDAFB,N)
!>          العامل المثلثي U أو L من تحليل تشوليسكي
!>          A = U**T*U أو A = L*L**T للمصفوفة الشريطية A كما تم حسابه بواسطة
!>          DPBTRF، بنفس تنسيق التخزين مثل A (انظر AB).
!>

LDAFB

!>          LDAFB is INTEGER
!>          البعد الرائد للمصفوفة AFB.  LDAFB >= KD+1.
!>

B

!>          B is DOUBLE PRECISION array, dimension (LDB,NRHS)
!>          مصفوفة الجانب الأيمن B.
!>

LDB

!>          LDB is INTEGER
!>          البعد الرائد للمصفوفة B.  LDB >= max(1,N).
!>

X

!>          X is DOUBLE PRECISION array, dimension (LDX,NRHS)
!>          عند الإدخال، مصفوفة الحل X، كما تم حسابها بواسطة DPBTRS.
!>          عند الخروج، مصفوفة الحل المحسنة X.
!>

LDX

!>          LDX is INTEGER
!>          البعد الرائد للمصفوفة X.  LDX >= max(1,N).
!>

FERR

!>          FERR is DOUBLE PRECISION array, dimension (NRHS)
!>          حد الخطأ الأمامي المقدر لكل متجه حل
!>          X(j) (العمود j للمصفوفة الحل X).
!>          إذا كان XTRUE هو الحل الحقيقي المقابل لـ X(j)، فإن FERR(j)
!>          هو حد أعلى مقدر لقيمة الجزء الأكبر
!>          من (X(j) - XTRUE) مقسومًا على قيمة الجزء الأكبر
!>          من X(j). التقدير موثوق به بقدر
!>          موثوقية تقدير RCOND، وهو دائمًا ما يكون أكثر من اللازم قليلاً
!>          للخطأ الحقيقي.
!>

BERR

!>          BERR is DOUBLE PRECISION array, dimension (NRHS)
!>          الخطأ الرجعي النسبي المكوني لكل متجه حل
!>          X(j) (أي، أصغر تغيير نسبي في
!>          أي عنصر من A أو B يجعل X(j) حلاً دقيقًا).
!>

WORK

!>          WORK is DOUBLE PRECISION array, dimension (3*N)
!>

IWORK

!>          IWORK is INTEGER array, dimension (N)
!>

INFO

!>          INFO is INTEGER
!>          = 0:  خروج ناجح
!>          < 0:  إذا كان INFO = -i، فإن الوسيط i كان له قيمة غير قانونية
!>
