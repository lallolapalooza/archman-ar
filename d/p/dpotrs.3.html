subroutine dpotrs (character uplo, integer n, integer nrhs, double precision, dimension( lda, * ) a, integer lda, double precision, dimension( ldb, * ) b, integer ldb, integer info)

DPOTRS

الغرض:

!>
!> DPOTRS يحل نظام المعادلات الخطية A*X = B مع مصفوفة متماثلة
!> موجبة محددة A باستخدام تحليل Cholesky
!> A = U**T*U أو A = L*L**T الذي تم حسابه بواسطة DPOTRF.
!>

المعلمات

UPLO 
!>          UPLO هو CHARACTER*1
!>          = 'U':  يتم تخزين الجزء العلوي من A؛
!>          = 'L':  يتم تخزين الجزء السفلي من A.
!>

N

!>          N هو INTEGER
!>          رتبة المصفوفة A.  N >= 0.
!>

NRHS

!>          NRHS هو INTEGER
!>          عدد الأطراف اليمنى، أي عدد الأعمدة
!>          للمصفوفة B.  NRHS >= 0.
!>

A

!>          A هي مصفوفة DOUBLE PRECISION، البعد (LDA,N)
!>          العامل المثلثي U أو L من تحليل Cholesky
!>          A = U**T*U أو A = L*L**T، كما تم حسابه بواسطة DPOTRF.
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرائد للمصفوفة A.  LDA >= max(1,N).
!>

B

!>          B هي مصفوفة DOUBLE PRECISION، البعد (LDB,NRHS)
!>          عند الإدخال، مصفوفة الطرف الأيمن B.
!>          عند الخروج، مصفوفة الحل X.
!>

LDB

!>          LDB هو INTEGER
!>          البعد الرائد للمصفوفة B.  LDB >= max(1,N).
!>

INFO

!>          INFO هو INTEGER
!>          = 0:  خروج ناجح
!>          < 0:  إذا كان INFO = -i، كان المعامل i له قيمة غير قانونية
!>
