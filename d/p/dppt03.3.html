subroutine dppt03 (character uplo, integer n, double precision, dimension( * ) a, double precision, dimension( * ) ainv, double precision, dimension( ldwork, * ) work, integer ldwork, double precision, dimension( * ) rwork, double precision rcond, double precision resid)

DPPT03

الغرض:

!>
!> DPPT03 يحسب الباقي لمصفوفة متماثلة مكدسة مضروبة في معكوسها:
!>    norm( I - A*AINV ) / ( N * norm(A) * norm(AINV) * EPS ),
!> حيث EPS هو الرقم الأقل دقة في الآلة.
!>

المعلمات

UPLO 
!>          UPLO هو CHARACTER*1
!>          يحدد ما إذا كان الجزء العلوي أو السفلي المثلثي للمصفوفة
!>          المتماثلة A مخزنًا:
!>          = 'U':  مثلثي علوي
!>          = 'L':  مثلثي سفلي
!>

N

!>          N هو INTEGER
!>          عدد الصفوف والأعمدة في المصفوفة A.  N >= 0.
!>

A

!>          A هو مصفوفة DOUBLE PRECISION، أبعادها (N*(N+1)/2)
!>          المصفوفة المتماثلة الأصلية A، مخزنة كمصفوفة
!>          مثلثية مكدسة.
!>

AINV

!>          AINV هو مصفوفة DOUBLE PRECISION، أبعادها (N*(N+1)/2)
!>          المعكوس (المتماثل) للمصفوفة A، مخزن كمصفوفة
!>          مثلثية مكدسة.
!>

WORK

!>          WORK هو مصفوفة DOUBLE PRECISION، أبعادها (LDWORK,N)
!>

LDWORK

!>          LDWORK هو INTEGER
!>          البعد الرائد للمصفوفة WORK.  LDWORK >= max(1,N).
!>

RWORK

!>          RWORK هو مصفوفة DOUBLE PRECISION، أبعادها (N)
!>

RCOND

!>          RCOND هو DOUBLE PRECISION
!>          مقلوب رقم الشرط للمصفوفة A، محسوب كـ
!>          ( 1/norm(A) ) / norm(AINV).
!>

RESID

!>          RESID هو DOUBLE PRECISION
!>          norm(I - A*AINV) / ( N * norm(A) * norm(AINV) * EPS )
!>
