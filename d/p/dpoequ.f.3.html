subroutine dpoequ (integer n, double precision, dimension( lda, * ) a, integer lda, double precision, dimension( * ) s, double precision scond, double precision amax, integer info)

DPOEQU

الغرض:

!>
!> DPOEQU يحسب عوامل قياس للصف والأعمدة المقصود بها موازنة مصفوفة متماثلة موجبة محددة A وتقليل عدد شرطتها
!> (بالنسبة للمعيار الثنائي). تحتوي S على عوامل القياس،
!> S(i) = 1/sqrt(A(i,i))، المختارة بحيث تكون المصفوفة المقاسة B ذات
!> العناصر B(i,j) = S(i)*A(i,j)*S(j) لها واحدات على القطر. هذا
!> الاختيار لـ S يضع عدد شرطة B في حدود عامل N لعدد الشرطة الأصغر الممكن
!> على جميع القياسات القطرية الممكنة.
!>

المعلمات

N 
!>          N هو عدد صحيح (INTEGER)
!>          ترتيب المصفوفة A.  N >= 0.
!>

A

!>          A هي مصفوفة من نوع DOUBLE PRECISION، أبعادها (LDA,N)
!>          المصفوفة المتماثلة الموجبة المحددة من حجم N*N التي سيتم
!>          حساب عوامل قياسها. يتم الإشارة فقط إلى العناصر القطرية لـ A.
!>

LDA

!>          LDA هو عدد صحيح (INTEGER)
!>          البعد الرئيسي للمصفوفة A.  LDA >= max(1,N).
!>

S

!>          S هي مصفوفة من نوع DOUBLE PRECISION، أبعادها (N)
!>          إذا كانت INFO = 0، تحتوي S على عوامل القياس لـ A.
!>

SCOND

!>          SCOND هو DOUBLE PRECISION
!>          إذا كانت INFO = 0، تحتوي S على نسبة أصغر S(i) إلى
!>          أكبر S(i). إذا كانت SCOND >= 0.1 و AMAX ليست كبيرة جدًا أو صغيرة جدًا،
!>          فلا داعي للقياس بواسطة S.
!>

AMAX

!>          AMAX هو DOUBLE PRECISION
!>          القيمة المطلقة لأكبر عنصر في المصفوفة. إذا كان AMAX قريبًا جدًا
!>          من تجاوز الحد الأقصى أو قريبًا جدًا من الحد الأدنى، فيجب قياس
!>          المصفوفة.
!>

INFO

!>          INFO هو عدد صحيح (INTEGER)
!>          = 0:  الخروج بنجاح
!>          < 0:  إذا كانت INFO = -i، فإن المعامل i كان له قيمة غير قانونية
!>          > 0:  إذا كانت INFO = i، فإن العنصر القطري i غير موجب.
!>
