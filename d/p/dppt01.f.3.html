subroutine dppt01 (character uplo, integer n, double precision, dimension( * ) a, double precision, dimension( * ) afac, double precision, dimension( * ) rwork, double precision resid)

DPPT01

الغرض:

!>
!> DPPT01 تعيد بناء مصفوفة متماثلة محددة موجبة A
!> من تحليلها L*L' أو U'*U وتحسب الباقي
!>    norm( L*L' - A ) / ( N * norm(A) * EPS ) أو
!>    norm( U'*U - A ) / ( N * norm(A) * EPS )،
!> حيث EPS هي ثابت الآلة (machine epsilon).
!>

المعلمات

UPLO 
!>          UPLO هو CHARACTER*1
!>          يحدد ما إذا كان الجزء المثلثي العلوي أو السفلي للمصفوفة
!>          المتماثلة A مخزنًا:
!>          = 'U':  مثلثي علوي
!>          = 'L':  مثلثي سفلي
!>

N

!>          N هو INTEGER
!>          عدد الصفوف والأعمدة للمصفوفة A.  N >= 0.
!>

A

!>          A هو مصفوفة DOUBLE PRECISION، بأبعاد (N*(N+1)/2)
!>          المصفوفة المتماثلة الأصلية A، مخزنة كمصفوفة
!>          مثلثية مجمعة.
!>

AFAC

!>          AFAC هو مصفوفة DOUBLE PRECISION، بأبعاد (N*(N+1)/2)
!>          عند الدخول، العامل L أو U من تحليل
!>          L*L' أو U'*U للمصفوفة A، مخزن كمصفوفة
!>          مثلثية مجمعة.
!>          يتم استبدالها بالمصفوفة المعاد بناؤها، ثم بالفرق
!>          L*L' - A (أو U'*U - A).
!>

RWORK

!>          RWORK هو مصفوفة DOUBLE PRECISION، بأبعاد (N)
!>

RESID

!>          RESID هو DOUBLE PRECISION
!>          إذا كان UPLO = 'L'، فإن norm(L*L' - A) / ( N * norm(A) * EPS )
!>          إذا كان UPLO = 'U'، فإن norm(U'*U - A) / ( N * norm(A) * EPS )
!>
