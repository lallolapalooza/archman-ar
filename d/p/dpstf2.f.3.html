subroutine dpstf2 (character uplo, integer n, double precision, dimension( lda, * ) a, integer lda, integer, dimension( n ) piv, integer rank, double precision tol, double precision, dimension( 2*n ) work, integer info)

DPSTF2 تحسب تفكيك Cholesky مع التحويل الكامل لمصفوفة حقيقية متناظرة موجبة شبه محددة.

الغرض:

!>
!> DPSTF2 تحسب تفكيك Cholesky مع التحويل الكامل
!> لمصفوفة A متناظرة حقيقية موجبة شبه محددة.
!>
!> التفكيك له الشكل
!>    P**T * A * P = U**T * U ,  إذا كان UPLO = 'U',
!>    P**T * A * P = L  * L**T,  إذا كان UPLO = 'L',
!> حيث U مصفوفة مثلثة عليا و L مصفوفة مثلثة سفلى، و
!> P مخزنة كمتجه PIV.
!>
!> هذه الخوارزمية لا تحاول التحقق من أن A موجبة
!> شبه محددة. هذا الإصدار من الخوارزمية يستدعي BLAS مستوى 2.
!>

المعاملات

UPLO 
!>          UPLO هو CHARACTER*1
!>          يحدد ما إذا كان الجزء المثلثي العلوي أو السفلي من
!>          المصفوفة المتناظرة A مخزناً.
!>          = 'U':  مثلثي علوي
!>          = 'L':  مثلثي سفلي
!>

N

!>          N هو INTEGER
!>          رتبة المصفوفة A.  N >= 0.
!>

A

!>          A هو مصفوفة DOUBLE PRECISION، بأبعاد (LDA,N)
!>          عند الدخول، المصفوفة المتناظرة A.  إذا كان UPLO = 'U'، فإن الجزء
!>          المثلثي العلوي العلوي N × N من A يحتوي على الجزء
!>          المثلثي العلوي للمصفوفة A، والجزء المثلثي السفلي الصارم
!>          من A لا يتم الرجوع إليه. إذا كان UPLO = 'L'، فإن الجزء
!>          المثلثي السفلي العلوي N × N من A يحتوي على الجزء
!>          المثلثي السفلي للمصفوفة A، والجزء المثلثي العلوي الصارم
!>          من A لا يتم الرجوع إليه.
!>
!>          عند الخروج، إذا كان INFO = 0، فإن العامل U أو L من تفكيك
!>          Cholesky كما هو مذكور أعلاه.
!>

PIV

!>          PIV هو مصفوفة INTEGER، بأبعاد (N)
!>          PIV بحيث تكون المدخلات غير الصفرية P( PIV(K), K ) = 1.
!>

RANK

!>          RANK هو INTEGER
!>          رتبة A المعطاة بعدد الخطوات التي أكملتها الخوارزمية.
!>

TOL

!>          TOL هو DOUBLE PRECISION
!>          تحمل مستخدم محدد. إذا كان TOL < 0، فسيتم استخدام N*U*MAX( A( K,K ) ).
!>          تتوقف الخوارزمية عند الخطوة (K-1) إذا كان المحور <= TOL.
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرائد للمصفوفة A.  LDA >= max(1,N).
!>

WORK

!>          WORK هو مصفوفة DOUBLE PRECISION، بأبعاد (2*N)
!>          مساحة عمل.
!>

INFO

!>          INFO هو INTEGER
!>          < 0: إذا كان INFO = -K، فإن المعامل K كان له قيمة غير قانونية،
!>          = 0: اكتملت الخوارزمية بنجاح، و
!>          > 0: المصفوفة A إما ناقصة الرتبة مع رتبة محسوبة
!>               كما هو معطى في RANK، أو ليست موجبة شبه محددة. انظر
!>               القسم 7 من مذكرة عمل LAPACK رقم 161 لمزيد من
!>               المعلومات.
!>
