subroutine dpstrf (character uplo, integer n, double precision, dimension( lda, * ) a, integer lda, integer, dimension( n ) piv, integer rank, double precision tol, double precision, dimension( 2*n ) work, integer info)

DPSTRF تحسب تفكيك Cholesky مع تثبيت كامل لمصفوفة حقيقية متناظرة موجبة شبه محددة.

الغرض:

!>
!> DPSTRF تحسب تفكيك Cholesky مع تثبيت كامل
!> لمصفوفة حقيقية متناظرة موجبة شبه محددة A.
!>
!> التفكيك له الشكل
!>    P**T * A * P = U**T * U ,  إذا كان UPLO = 'U',
!>    P**T * A * P = L  * L**T,  إذا كان UPLO = 'L',
!> حيث U مصفوفة مثلثة عليا و L مصفوفة مثلثة سفلى،
!> و P مخزنة كمتجه PIV.
!>
!> هذه الخوارزمية لا تحاول التحقق من أن A موجبة شبه محددة.
!> هذا الإصدار من الخوارزمية يستدعي BLAS المستوى 3.
!>

المعلمات

UPLO 
!>          UPLO هو CHARACTER*1
!>          يحدد ما إذا كان الجزء العلوي أو السفلي المثلثي
!>          للمصفوفة المتناظرة A مخزنًا.
!>          = 'U':  مثلث علوي
!>          = 'L':  مثلث سفلي
!>

N

!>          N هو INTEGER
!>          رتبة المصفوفة A.  N >= 0.
!>

A

!>          A هو مصفوفة DOUBLE PRECISION، حجمها (LDA,N)
!>          عند الدخول، المصفوفة المتناظرة A.  إذا كان UPLO = 'U'، فإن
!>          الجزء العلوي المثلثي من A الذي أبعاده n x n يحتوي على
!>          الجزء العلوي المثلثي للمصفوفة A، ولا يتم الرجوع إلى الجزء
!>          السفلي المثلثي الصارم من A.  إذا كان UPLO = 'L'، فإن
!>          الجزء السفلي المثلثي من A الذي أبعاده n x n يحتوي على
!>          الجزء السفلي المثلثي للمصفوفة A، ولا يتم الرجوع إلى الجزء
!>          العلوي المثلثي الصارم من A.
!>
!>          عند الخروج، إذا كان INFO = 0، فإن العامل U أو L من تفكيك
!>          Cholesky المذكور أعلاه.
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرئيسي للمصفوفة A.  LDA >= max(1,N).
!>

PIV

!>          PIV هو مصفوفة INTEGER، حجمها (N)
!>          PIV بحيث تكون المدخلات غير الصفرية هي P( PIV(K), K ) = 1.
!>

RANK

!>          RANK هو INTEGER
!>          رتبة A المعطاة بعدد الخطوات التي أكملتها الخوارزمية.
!>

TOL

!>          TOL هو DOUBLE PRECISION
!>          تسامح معرف من قبل المستخدم. إذا كان TOL < 0، فسيتم استخدام
!>          N*U*MAX( A(K,K) ). تنتهي الخوارزمية عند الخطوة (K-1)
!>          إذا كان المحور <= TOL.
!>

WORK

!>          WORK هو مصفوفة DOUBLE PRECISION، حجمها (2*N)
!>          مساحة عمل.
!>

INFO

!>          INFO هو INTEGER
!>          < 0: إذا كان INFO = -K، فإن الوسيط K كان له قيمة غير قانونية،
!>          = 0: اكتملت الخوارزمية بنجاح، و
!>          > 0: المصفوفة A إما ناقصة الرتبة مع رتبة محسوبة كما هو
!>               معاد في RANK، أو ليست موجبة شبه محددة. راجع
!>               القسم 7 من مذكرة عمل LAPACK #161 لمزيد من
!>               المعلومات.
!>
