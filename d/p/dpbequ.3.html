subroutine dpbequ (character uplo, integer n, integer kd, double precision, dimension( ldab, * ) ab, integer ldab, double precision, dimension( * ) s, double precision scond, double precision amax, integer info)

DPBEQU

الغرض:

!>
!> DPBEQU يحسب عوامل القياس للصفوف والأعمدة المقصود بها موازنة مصفوفة
!> موجبة محددة ومتناظرة ذات نطاق A وتقليل عدد شرطها (بالنسبة إلى المعيار الثنائي).
!> يحتوي S على عوامل القياس، S(i) = 1/sqrt(A(i,i))، المختار بحيث تكون المصفوفة
!> المقاسة B ذات العناصر B(i,j) = S(i)*A(i,j)*S(j) لها واحدات على القطر.
!> هذا الاختيار لـ S يضع عدد شرط B في حدود عامل N من أصغر عدد شرط ممكن
!> عبر جميع عمليات القياس القطرية الممكنة.
!>

المعلمات

UPLO 
!>          UPLO هو CHARACTER*1
!>          = 'U':  يتم تخزين الجزء العلوي المثلثي من A؛
!>          = 'L':  يتم تخزين الجزء السفلي المثلثي من A.
!>

N

!>          N هو INTEGER
!>          ترتيب المصفوفة A. N >= 0.
!>

KD

!>          KD هو INTEGER
!>          عدد الأقطار الفائقة للمصفوفة A إذا كانت UPLO = 'U'،
!>          أو عدد الأقطار الفرعية إذا كانت UPLO = 'L'. KD >= 0.
!>

AB

!>          AB هو مصفوفة DOUBLE PRECISION، ذات أبعاد (LDAB,N)
!>          الجزء العلوي أو السفلي المثلثي من المصفوفة المتناظرة ذات النطاق A،
!>          مخزن في الصفوف KD+1 الأولى من المصفوفة. يتم تخزين العمود j
!>          من A في العمود j من المصفوفة AB على النحو التالي:
!>          إذا كانت UPLO = 'U'، AB(kd+1+i-j,j) = A(i,j) لـ max(1,j-kd)<=i<=j؛
!>          إذا كانت UPLO = 'L'، AB(1+i-j,j)    = A(i,j) لـ j<=i<=min(n,j+kd).
!>

LDAB

!>          LDAB هو INTEGER
!>          البعد الرئيسي للمصفوفة A. LDAB >= KD+1.
!>

S

!>          S هي مصفوفة DOUBLE PRECISION، ذات أبعاد (N)
!>          إذا كانت INFO = 0، تحتوي S على عوامل القياس لـ A.
!>

SCOND

!>          SCOND هو DOUBLE PRECISION
!>          إذا كانت INFO = 0، تحتوي S على نسبة أصغر S(i) إلى
!>          أكبر S(i). إذا كانت SCOND >= 0.1 وكان AMAX ليس كبيرًا جدًا
!>          ولا صغيرًا جدًا، فلا فائدة من القياس بـ S.
!>

AMAX

!>          AMAX هو DOUBLE PRECISION
!>          القيمة المطلقة لأكبر عنصر في المصفوفة. إذا كان AMAX قريبًا جدًا
!>          من تجاوز الحد الأقصى أو قريبًا جدًا من الحد الأدنى، يجب قياس
!>          المصفوفة.
!>

INFO

!>          INFO هو INTEGER
!>          = 0:  انتهاء ناجح
!>          < 0:  إذا كانت INFO = -i، فإن المعامل i له قيمة غير قانونية.
!>          > 0:  إذا كانت INFO = i، فإن العنصر القطري i غير موجب.
!>
