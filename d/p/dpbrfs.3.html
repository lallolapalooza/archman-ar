subroutine dpbrfs (character uplo, integer n, integer kd, integer nrhs, double precision, dimension( ldab, * ) ab, integer ldab, double precision, dimension( ldafb, * ) afb, integer ldafb, double precision, dimension( ldb, * ) b, integer ldb, double precision, dimension( ldx, * ) x, integer ldx, double precision, dimension( * ) ferr, double precision, dimension( * ) berr, double precision, dimension( * ) work, integer, dimension( * ) iwork, integer info)

DPBRFS

الغرض:

!>
!> DPBRFS يحسن الحل المحسوب لنظام المعادلات الخطية
!> عندما تكون مصفوفة المعاملات متماثلة وموجبة مؤكدة
!> ونطاقية، ويوفر حدود الخطأ وتقديرات الخطأ الخلفي
!> للحل.
!>

المعلمات

UPLO 
!>          UPLO is CHARACTER*1
!>          = 'U':  المثلث العلوي للمصفوفة A مخزن؛
!>          = 'L':  المثلث السفلي للمصفوفة A مخزن.
!>

N

!>          N is INTEGER
!>          رتبة المصفوفة A. N >= 0.
!>

KD

!>          KD is INTEGER
!>          عدد الأقطار الفائقة للمصفوفة A إذا كان UPLO = 'U'،
!>          أو عدد الأقطار الفرعية إذا كان UPLO = 'L'. KD >= 0.
!>

NRHS

!>          NRHS is INTEGER
!>          عدد الأطراف اليمنى، أي عدد الأعمدة
!>          للمصفوفتين B و X. NRHS >= 0.
!>

AB

!>          AB is DOUBLE PRECISION array, dimension (LDAB,N)
!>          المثلث العلوي أو السفلي للمصفوفة النطاقية المتماثلة A،
!>          مخزنة في أول KD+1 صف من المصفوفة. العمود j
!>          للمصفوفة A مخزن في العمود j للمصفوفة AB كالتالي:
!>          إذا كان UPLO = 'U', AB(kd+1+i-j,j) = A(i,j) لـ max(1,j-kd)<=i<=j؛
!>          إذا كان UPLO = 'L', AB(1+i-j,j)    = A(i,j) لـ j<=i<=min(n,j+kd).
!>

LDAB

!>          LDAB is INTEGER
!>          البعد الرئيسي للمصفوفة AB. LDAB >= KD+1.
!>

AFB

!>          AFB is DOUBLE PRECISION array, dimension (LDAFB,N)
!>          العامل المثلثي U أو L من تحليل Cholesky
!>          A = U**T*U أو A = L*L**T للمصفوفة النطاقية A كما تم حسابه بواسطة
!>          DPBTRF، بنفس تنسيق التخزين مثل A (انظر AB).
!>

LDAFB

!>          LDAFB is INTEGER
!>          البعد الرئيسي للمصفوفة AFB. LDAFB >= KD+1.
!>

B

!>          B is DOUBLE PRECISION array, dimension (LDB,NRHS)
!>          مصفوفة الطرف الأيمن B.
!>

LDB

!>          LDB is INTEGER
!>          البعد الرئيسي للمصفوفة B. LDB >= max(1,N).
!>

X

!>          X is DOUBLE PRECISION array, dimension (LDX,NRHS)
!>          عند الإدخال، مصفوفة الحل X، كما تم حسابه بواسطة DPBTRS.
!>          عند الخروج، مصفوفة الحل المحسنة X.
!>

LDX

!>          LDX is INTEGER
!>          البعد الرئيسي للمصفوفة X. LDX >= max(1,N).
!>

FERR

!>          FERR is DOUBLE PRECISION array, dimension (NRHS)
!>          الحد التقديري للخطأ الأمامي لكل حل متجه
!>          X(j) (العمود j من مصفوفة الحل X).
!>          إذا كان XTRUE هو الحل الحقيقي المقابل لـ X(j)، فإن FERR(j)
!>          هو حد علوي مقدر لقيمة المطلقة لأكبر
!>          عنصر في (X(j) - XTRUE) مقسومًا على القيمة المطلقة لـ
!>          أكبر عنصر في X(j). التقدير موثوق به مثل
!>          التقدير لـ RCOND، وهو دائمًا تقريبًا تقدير أعلى قليلاً للخطأ الحقيقي.
!>

BERR

!>          BERR is DOUBLE PRECISION array, dimension (NRHS)
!>          خطأ الخلفية النسبي المكون لكل حل متجه
!>          X(j) (أي، أصغر تغيير نسبي في
!>          أي عنصر في A أو B يجعل X(j) حلاً دقيقًا).
!>

WORK

!>          WORK is DOUBLE PRECISION array, dimension (3*N)
!>

IWORK

!>          IWORK is INTEGER array, dimension (N)
!>

INFO

!>          INFO is INTEGER
!>          = 0:  انتهاء ناجح
!>          < 0:  إذا كان INFO = -i، فإن المعلمة i كان لها قيمة غير قانونية
!>
