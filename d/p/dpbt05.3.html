توثيق الوظيفة/الروتين الفرعي

subroutine dpbt05 (character uplo, integer n, integer kd, integer nrhs, double precision, dimension( ldab, * ) ab, integer ldab, double precision, dimension( ldb, * ) b, integer ldb, double precision, dimension( ldx, * ) x, integer ldx, double precision, dimension( ldxact, * ) xact, integer ldxact, double precision, dimension( * ) ferr, double precision, dimension( * ) berr, double precision, dimension( * ) reslts)

DPBT05

الغرض:

!>
!> DPBT05 تختبر حدود الخطأ الناتجة عن التحسين التكراري للحل المحسوب
!> لنظام المعادلات A*X = B، حيث A هي مصفوفة متناظرة شريطية.
!>
!> RESLTS(1) = اختبار حد الخطأ
!>           = norm(X - XACT) / ( norm(X) * FERR )
!>
!> يتم إرجاع قيمة كبيرة إذا لم تكن هذه النسبة أقل من واحد.
!>
!> RESLTS(2) = الخطأ المتبقي من روتين التحسين التكراري
!>           = القيمة القصوى لـ BERR / ( NZ*EPS + (*) )، حيث
!>             (*) = NZ*UNFL / (min_i (abs(A)*abs(X) +abs(b))_i )
!>             و NZ = أقصى عدد من العناصر غير الصفرية في أي صف من A، بالإضافة إلى 1
!>

المعلمات

UPLO 
!>          UPLO is CHARACTER*1
!>          يحدد ما إذا كان الجزء العلوي أو السفلي المثلثي للمصفوفة
!>          المتناظرة A مخزنًا.
!>          = 'U':  مثلثي علوي
!>          = 'L':  مثلثي سفلي
!>

N

!>          N is INTEGER
!>          عدد الصفوف للمصفوفات X، B، و XACT، وترتيب المصفوفة A. N >= 0.
!>

KD

!>          KD is INTEGER
!>          عدد الأقطار الفائقة للمصفوفة A إذا كان UPLO = 'U'،
!>          أو عدد الأقطار الفرعية إذا كان UPLO = 'L'. KD >= 0.
!>

NRHS

!>          NRHS is INTEGER
!>          عدد الأعمدة للمصفوفات X، B، و XACT. NRHS >= 0.
!>

AB

!>          AB is DOUBLE PRECISION array, dimension (LDAB,N)
!>          الجزء العلوي أو السفلي المثلثي للمصفوفة الشريطية المتناظرة A،
!>          مخزن في أول KD+1 صف من المصفوفة. يتم تخزين العمود j
!>          من A في العمود j من المصفوفة AB كما يلي:
!>          إذا كان UPLO = 'U'، AB(kd+1+i-j,j) = A(i,j) لـ max(1,j-kd) <= i <= j؛
!>          إذا كان UPLO = 'L'، AB(1+i-j,j)    = A(i,j) لـ j <= i <= min(n,j+kd).
!>

LDAB

!>          LDAB is INTEGER
!>          البعد الرائد للمصفوفة AB. LDAB >= KD+1.
!>

B

!>          B is DOUBLE PRECISION array, dimension (LDB,NRHS)
!>          متجهات الجانب الأيمن لأنظمة المعادلات الخطية.
!>

LDB

!>          LDB is INTEGER
!>          البعد الرائد للمصفوفة B. LDB >= max(1,N).
!>

X

!>          X is DOUBLE PRECISION array, dimension (LDX,NRHS)
!>          متجهات الحل المحسوبة. يتم تخزين كل متجه كعمود
!>          من المصفوفة X.
!>

LDX

!>          LDX is INTEGER
!>          البعد الرائد للمصفوفة X. LDX >= max(1,N).
!>

XACT

!>          XACT is DOUBLE PRECISION array, dimension (LDX,NRHS)
!>          متجهات الحل الدقيقة. يتم تخزين كل متجه كعمود
!>          من المصفوفة XACT.
!>

LDXACT

!>          LDXACT is INTEGER
!>          البعد الرائد للمصفوفة XACT. LDXACT >= max(1,N).
!>

FERR

!>          FERR is DOUBLE PRECISION array, dimension (NRHS)
!>          حدود الخطأ الأمامي المقدرة لكل متجه حل
!>          X. إذا كان XTRUE هو الحل الحقيقي، فإن FERR يقدر حجم
!>          أكبر مدخل في (X - XTRUE) مقسومًا على حجم
!>          أكبر مدخل في X.
!>

BERR

!>          BERR is DOUBLE PRECISION array, dimension (NRHS)
!>          الخطأ الخلفي النسبي المكون من كل متجه حل
!>          (أي، أصغر تغيير نسبي في أي مدخل في A
!>          أو B يجعل X حلاً دقيقًا).
!>

RESLTS

!>          RESLTS is DOUBLE PRECISION array, dimension (2)
!>          الحد الأقصى لمتجهات الحل NRHS من النسب التالية:
!>          RESLTS(1) = norm(X - XACT) / ( norm(X) * FERR )
!>          RESLTS(2) = BERR / ( NZ*EPS + (*) )
!>
