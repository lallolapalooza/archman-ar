subroutine dpprfs (character uplo, integer n, integer nrhs, double precision, dimension( * ) ap, double precision, dimension( * ) afp, double precision, dimension( ldb, * ) b, integer ldb, double precision, dimension( ldx, * ) x, integer ldx, double precision, dimension( * ) ferr, double precision, dimension( * ) berr, double precision, dimension( * ) work, integer, dimension( * ) iwork, integer info)

DPPRFS

الغرض:

!>
!> DPPRFS يحسن الحل المحسوب لنظام المعادلات الخطية
!> عندما تكون مصفوفة المعاملات متماثلة وموجبة محددة
!> ومعبأة، ويوفر حدود الخطأ وتقديرات الخطأ الخلفي
!> للحل.
!>

المعلمات

UPLO 
!>          UPLO is CHARACTER*1
!>          = 'U':  يتم تخزين المثلث العلوي للمصفوفة A؛
!>          = 'L':  يتم تخزين المثلث السفلي للمصفوفة A.
!>

N

!>          N is INTEGER
!>          رتبة المصفوفة A. N >= 0.
!>

NRHS

!>          NRHS is INTEGER
!>          عدد الأطراف اليمنى، أي عدد الأعمدة
!>          للمصفوفتين B و X. NRHS >= 0.
!>

AP

!>          AP is DOUBLE PRECISION array, dimension (N*(N+1)/2)
!>          المثلث العلوي أو السفلي للمصفوفة المتماثلة A، معبأ
!>          عموديًا في مصفوفة خطية. يتم تخزين العمود j من A
!>          في المصفوفة AP كما يلي:
!>          إذا كان UPLO = 'U'، فإن AP(i + (j-1)*j/2) = A(i,j) لـ 1<=i<=j؛
!>          إذا كان UPLO = 'L'، فإن AP(i + (j-1)*(2n-j)/2) = A(i,j) لـ j<=i<=n.
!>

AFP

!>          AFP is DOUBLE PRECISION array, dimension (N*(N+1)/2)
!>          العامل المثلثي U أو L من تحليل Cholesky
!>          A = U**T*U أو A = L*L**T، كما تم حسابه بواسطة DPPTRF/ZPPTRF،
!>          معبأ عموديًا في مصفوفة خطية بنفس تنسيق A
!>          (انظر AP).
!>

B

!>          B is DOUBLE PRECISION array, dimension (LDB,NRHS)
!>          المصفوفة الطرفية اليمنى B.
!>

LDB

!>          LDB is INTEGER
!>          البعد الرائد للمصفوفة B. LDB >= max(1,N).
!>

X

!>          X is DOUBLE PRECISION array, dimension (LDX,NRHS)
!>          عند الإدخال، مصفوفة الحل X، كما تم حسابها بواسطة DPPTRS.
!>          عند الخروج، مصفوفة الحل المحسنة X.
!>

LDX

!>          LDX is INTEGER
!>          البعد الرائد للمصفوفة X. LDX >= max(1,N).
!>

FERR

!>          FERR is DOUBLE PRECISION array, dimension (NRHS)
!>          تقدير حد الخطأ الأمامي لكل متجه حل
!>          X(j) (العمود j من مصفوفة الحل X).
!>          إذا كان XTRUE هو الحل الحقيقي المقابل لـ X(j)، فإن FERR(j)
!>          هو تقدير للحد الأعلى لمقدار أكبر
!>          عنصر في (X(j) - XTRUE) مقسومًا على مقدار
!>          أكبر عنصر في X(j). التقدير موثوق به بقدر
!>          موثوقية تقدير RCOND، وهو تقريبًا دائمًا
!>          تقدير زائد قليلاً للخطأ الحقيقي.
!>

BERR

!>          BERR is DOUBLE PRECISION array, dimension (NRHS)
!>          الخطأ النسبي الخلفي المكون من عنصر لعنصر لكل متجه حل
!>          X(j) (أي، أصغر تغيير نسبي في
!>          أي عنصر في A أو B يجعل X(j) حلاً دقيقًا).
!>

WORK

!>          WORK is DOUBLE PRECISION array, dimension (3*N)
!>

IWORK

!>          IWORK is INTEGER array, dimension (N)
!>

INFO

!>          INFO is INTEGER
!>          = 0:  خروج ناجح
!>          < 0:  إذا كان INFO = -i، فإن الوسيط i كان لديه قيمة غير قانونية
!>
