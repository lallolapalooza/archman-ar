توثيق الوظيفة/الروتين الفرعي

subroutine dpot03 (character uplo, integer n, double precision, dimension( lda, * ) a, integer lda, double precision, dimension( ldainv, * ) ainv, integer ldainv, double precision, dimension( ldwork, * ) work, integer ldwork, double precision, dimension( * ) rwork, double precision rcond, double precision resid)

DPOT03

الغرض:

!>
!> DPOT03 يحسب الباقي لمصفوفة متماثلة مضروبة في معكوسها:
!>    norm( I - A*AINV ) / ( N * norm(A) * norm(AINV) * EPS ),
!> حيث EPS هو اصطناع الآلة.
!>

المعاملات

UPLO 
!>          UPLO هو CHARACTER*1
!>          يحدد ما إذا كان الجزء العلوي أو السفلي المثلثي للمصفوفة
!>          المتماثلة A مخزنًا:
!>          = 'U':  مثلثي علوي
!>          = 'L':  مثلثي سفلي
!>

N

!>          N هو INTEGER
!>          عدد الصفوف والأعمدة للمصفوفة A.  N >= 0.
!>

A

!>          A هو مصفوفة DOUBLE PRECISION، أبعاد (LDA,N)
!>          المصفوفة الأصلية المتماثلة A.
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرائد للمصفوفة A.  LDA >= max(1,N)
!>

AINV

!>          AINV هو مصفوفة DOUBLE PRECISION، أبعاد (LDAINV,N)
!>          عند الدخول، معكوس المصفوفة A، مخزن كمصفوفة متماثلة
!>          بنفس تنسيق A.
!>          في هذا الإصدار، يتم توسيع AINV إلى مصفوفة كاملة و
!>          ضربها في A، لذلك سيتم تغيير المثلث المعاكس لـ AINV؛
!>          أي، إذا تم تخزين الجزء المثلثي العلوي من AINV،
!>          فإن الجزء المثلثي السفلي سيستخدم كمساحة عمل.
!>

LDAINV

!>          LDAINV هو INTEGER
!>          البعد الرائد للمصفوفة AINV.  LDAINV >= max(1,N).
!>

WORK

!>          WORK هي مصفوفة DOUBLE PRECISION، أبعاد (LDWORK,N)
!>

LDWORK

!>          LDWORK هو INTEGER
!>          البعد الرائد للمصفوفة WORK.  LDWORK >= max(1,N).
!>

RWORK

!>          RWORK هي مصفوفة DOUBLE PRECISION، أبعاد (N)
!>

RCOND

!>          RCOND هو DOUBLE PRECISION
!>          مقلوب رقم الشرط لـ A، محسوب كـ
!>          ( 1/norm(A) ) / norm(AINV).
!>

RESID

!>          RESID هو DOUBLE PRECISION
!>          norm(I - A*AINV) / ( N * norm(A) * norm(AINV) * EPS )
!>
