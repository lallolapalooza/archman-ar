subroutine dppsv (character uplo, integer n, integer nrhs, double precision, dimension( * ) ap, double precision, dimension( ldb, * ) b, integer ldb, integer info)

DPPSV يحسب الحل لنظام المعادلات الخطية A * X = B للمصفوفات الأخرى

الغرض:

!>
!> DPPSV يحسب الحل لنظام معادلات خطية حقيقي
!>    A * X = B,
!> حيث A مصفوفة N × N متناظرة وموجبة محددة مخزنة بتنسيق مضغوط و X و B مصفوفتان N × NRHS.
!>
!> يتم استخدام تحليل Cholesky لتحليل A إلى
!>    A = U**T* U,  إذا كان UPLO = 'U', أو
!>    A = L * L**T,  إذا كان UPLO = 'L',
!> حيث U مصفوفة مثلثة عليا و L مصفوفة مثلثة سفلى.  ثم يتم استخدام الشكل المحلل لـ A لحل نظام المعادلات A * X = B.
!>

المعلمات

UPLO 
!>          UPLO هو CHARACTER*1
!>          = 'U':  يتم تخزين المثلث العلوي لـ A؛
!>          = 'L':  يتم تخزين المثلث السفلي لـ A.
!>

N

!>          N هو INTEGER
!>          عدد المعادلات الخطية، أي رتبة المصفوفة A.  N >= 0.
!>

NRHS

!>          NRHS هو INTEGER
!>          عدد الأطراف اليمنى، أي عدد أعمدة المصفوفة B.  NRHS >= 0.
!>

AP

!>          AP هو مصفوفة DOUBLE PRECISION، البعد (N*(N+1)/2)
!>          عند الإدخال، المثلث العلوي أو السفلي للمصفوفة المتناظرة A، مضغوطة حسب العمود في مصفوفة خطية.  يتم تخزين العمود j من A
!>          في المصفوفة AP على النحو التالي:
!>          إذا كان UPLO = 'U'، فإن AP(i + (j-1)*j/2) = A(i,j) لـ 1<=i<=j؛
!>          إذا كان UPLO = 'L'، فإن AP(i + (j-1)*(2n-j)/2) = A(i,j) لـ j<=i<=n.
!>          راجع التفاصيل أدناه.
!>
!>          عند الخروج، إذا كان INFO = 0، فإن العامل U أو L من تحليل Cholesky
!!>          A = U**T*U أو A = L*L**T، بنفس تنسيق التخزين لـ A.
!>

B

!>          B هي مصفوفة DOUBLE PRECISION، البعد (LDB,NRHS)
!>          عند الإدخال، مصفوفة الطرف الأيمن B ذات البعد N × NRHS.
!>          عند الخروج، إذا كان INFO = 0، فإن مصفوفة الحل X ذات البعد N × NRHS.
!>

LDB

!>          LDB هو INTEGER
!>          البعد الأمامي للمصفوفة B.  LDB >= max(1,N).
!>

INFO

!>          INFO هو INTEGER
!>          = 0:  خروج ناجح
!>          < 0:  إذا كان INFO = -i، فإن المعلمة i كان لها قيمة غير قانونية
!>          > 0:  إذا كان INFO = i، فإن العامل الرئيسي الأساسي من الرتبة i
!>                لـ A ليس موجبًا، لذلك لم يكن من الممكن إكمال التحليل
!>                ولم يتم حساب الحل.
!>
