توثيق الوظيفة/الروتين الفرعي

subroutine dppt02 (character uplo, integer n, integer nrhs, double precision, dimension( * ) a, double precision, dimension( ldx, * ) x, integer ldx, double precision, dimension( ldb, * ) b, integer ldb, double precision, dimension( * ) rwork, double precision resid)

DPPT02

الغرض:

!>
!> DPPT02 يحسب الباقي في حل نظام
!> المعادلات الخطية A*x = b  عند استخدام تخزين مضغوط للمصفوفة المعاملة.  النسبة المحسوبة هي
!>
!>    RESID = norm(B - A*X) / ( norm(A) * norm(X) * EPS),
!>
!> حيث EPS هي الدقة الآلية.
!>

المعلمات

UPLO 
!>          UPLO is CHARACTER*1
!>          يحدد ما إذا كان الجزء العلوي أو السفلي المثلثي من المصفوفة المتماثلة A مخزنًا:
!>          = 'U':  مثلثي علوي
!>          = 'L':  مثلثي سفلي
!>

N

!>          N is INTEGER
!>          عدد الصفوف والأعمدة للمصفوفة A.  N >= 0.
!>

NRHS

!>          NRHS is INTEGER
!>          عدد أعمدة B، مصفوفة الأطراف اليمنى.
!>          NRHS >= 0.
!>

A

!>          A is DOUBLE PRECISION array, dimension (N*(N+1)/2)
!>          المصفوفة الأصلية المتماثلة A، مخزنة كمصفوفة
!>          مثلثية مضغوطة.
!>

X

!>          X is DOUBLE PRECISION array, dimension (LDX,NRHS)
!>          متجهات الحل المحسوبة لنظام المعادلات الخطية.
!>

LDX

!>          LDX is INTEGER
!>          البعد الرائد للمصفوفة X.   LDX >= max(1,N).
!>

B

!>          B is DOUBLE PRECISION array, dimension (LDB,NRHS)
!>          عند الإدخال، متجهات الطرف الأيمن لنظام المعادلات الخطية.
!>          عند الخروج، يتم استبدال B بالفرق B - A*X.
!>

LDB

!>          LDB is INTEGER
!>          البعد الرائد للمصفوفة B.  LDB >= max(1,N).
!>

RWORK

!>          RWORK is DOUBLE PRECISION array, dimension (N)
!>

RESID

!>          RESID is DOUBLE PRECISION
!>          القيمة القصوى لعدد الأطراف اليمنى
!>          norm(B - A*X) / ( norm(A) * norm(X) * EPS ).
!>
