subroutine dptt05 (integer n, integer nrhs, double precision, dimension( * ) d, double precision, dimension( * ) e, double precision, dimension( ldb, * ) b, integer ldb, double precision, dimension( ldx, * ) x, integer ldx, double precision, dimension( ldxact, * ) xact, integer ldxact, double precision, dimension( * ) ferr, double precision, dimension( * ) berr, double precision, dimension( * ) reslts)

DPTT05

الغرض:

!>
!> DPTT05 تختبر حدود الخطأ من التنقيح التكراري للحل المحسوب لنظام المعادلات A*X = B، حيث A هي مصفوفة ثلاثية القطر متناظرة من الرتبة n.
!>
!> RESLTS(1) = اختبار حد الخطأ
!>           = norm(X - XACT) / ( norm(X) * FERR )
!>
!> تُرجع قيمة كبيرة إذا لم تكن هذه النسبة أقل من واحد.
!>
!> RESLTS(2) = الباقي من روتين التنقيح التكراري
!>           = الحد الأقصى لـ BERR / ( NZ*EPS + (*) )، حيث
!>             (*) = NZ*UNFL / (min_i (abs(A)*abs(X) +abs(b))_i )
!>             و NZ = أقصى عدد من العناصر غير الصفرية في أي صف من A، بالإضافة إلى 1
!>

المعاملات

N 
!>          N هو عدد صحيح (INTEGER)
!>          عدد صفوف المصفوفات X، B، و XACT، ورتبة المصفوفة A.  N >= 0.
!>

NRHS

!>          NRHS هو عدد صحيح (INTEGER)
!>          عدد أعمدة المصفوفات X، B، و XACT.
!>          NRHS >= 0.
!>

D

!>          D هي مصفوفة ذات دقة مزدوجة (DOUBLE PRECISION)، البعد (N)
!>          عناصر القطر الرئيسي للمصفوفة ثلاثية القطر A.
!>

E

!>          E هي مصفوفة ذات دقة مزدوجة (DOUBLE PRECISION)، البعد (N-1)
!>          عناصر تحت القطر الرئيسي للمصفوفة ثلاثية القطر A.
!>

B

!>          B هي مصفوفة ذات دقة مزدوجة (DOUBLE PRECISION)، البعد (LDB,NRHS)
!>          متجهات الطرف الأيمن لنظام المعادلات الخطية.
!>

LDB

!>          LDB هو عدد صحيح (INTEGER)
!>          البعد الرئيسي للمصفوفة B.  LDB >= max(1,N).
!>

X

!>          X هي مصفوفة ذات دقة مزدوجة (DOUBLE PRECISION)، البعد (LDX,NRHS)
!>          متجهات الحل المحسوبة.  يتم تخزين كل متجه كعمود في المصفوفة X.
!>

LDX

!>          LDX هو عدد صحيح (INTEGER)
!>          البعد الرئيسي للمصفوفة X.  LDX >= max(1,N).
!>

XACT

!>          XACT هي مصفوفة ذات دقة مزدوجة (DOUBLE PRECISION)، البعد (LDX,NRHS)
!>          متجهات الحل الدقيقة.  يتم تخزين كل متجه كعمود في المصفوفة XACT.
!>

LDXACT

!>          LDXACT هو عدد صحيح (INTEGER)
!>          البعد الرئيسي للمصفوفة XACT.  LDXACT >= max(1,N).
!>

FERR

!>          FERR هي مصفوفة ذات دقة مزدوجة (DOUBLE PRECISION)، البعد (NRHS)
!>          تقديرات حدود الخطأ الأمامي لكل متجه حل X.  إذا كانت XTRUE هي الحل الحقيقي، فإن FERR يحد من مقدار أكبر مدخل في (X - XTRUE) مقسومًا على مقدار أكبر مدخل في X.
!>

BERR

!>          BERR هي مصفوفة ذات دقة مزدوجة (DOUBLE PRECISION)، البعد (NRHS)
!>          الخطأ الخلفي النسبي المكوني لكل متجه حل (أي، أصغر تغيير نسبي في أي مدخل في A أو B يجعل X حلاً دقيقًا).
!>

RESLTS

!>          RESLTS هي مصفوفة ذات دقة مزدوجة (DOUBLE PRECISION)، البعد (2)
!>          أقصى قيم على متجهات الحل NRHS للنسب:
!>          RESLTS(1) = norm(X - XACT) / ( norm(X) * FERR )
!>          RESLTS(2) = BERR / ( NZ*EPS + (*) )
!>
