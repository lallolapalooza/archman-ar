subroutine dpbt02 (character uplo, integer n, integer kd, integer nrhs, double precision, dimension( lda, * ) a, integer lda, double precision, dimension( ldx, * ) x, integer ldx, double precision, dimension( ldb, * ) b, integer ldb, double precision, dimension( * ) rwork, double precision resid)

DPBT02

الغرض:

!>
!> DPBT02 يحسب الباقي لحل نظام معادلات متماثل نطاقي  A*x = b:
!>    RESID = norm( B - A*X ) / ( norm(A) * norm(X) * EPS)
!> حيث EPS هو دقة الآلة.
!>

المعاملات

UPLO 
!>          UPLO is CHARACTER*1
!>          يحدد ما إذا كان الجزء العلوي أو السفلي المثلثي للمصفوفة المتماثلة A مخزنًا:
!>          = 'U':  مثلثي علوي
!>          = 'L':  مثلثي سفلي
!>

N

!>          N is INTEGER
!>          عدد الصفوف والأعمدة للمصفوفة A.  N >= 0.
!>

KD

!>          KD is INTEGER
!>          عدد الأقطار الفائقة للمصفوفة A إذا كان UPLO = 'U'،
!>          أو عدد الأقطار الفرعية إذا كان UPLO = 'L'.  KD >= 0.
!>

NRHS

!>          NRHS is INTEGER
!>          عدد الأطراف اليمنى. NRHS >= 0.
!>

A

!>          A is DOUBLE PRECISION array, dimension (LDA,N)
!>          المصفوفة النطاقية المتماثلة الأصلية A.  إذا كان UPLO = 'U'،
!>          يتم تخزين الجزء العلوي المثلثي من A كمصفوفة نطاقية؛ إذا
!>          كان UPLO = 'L'، يتم تخزين الجزء السفلي المثلثي من A.  يتم تخزين
!>          الأعمدة للمثلث المناسب في أعمدة A ويتم تخزين الأقطار للمثلث
!>          في صفوف A.  انظر DPBTRF لمزيد من التفاصيل.
!>

LDA

!>          LDA is INTEGER.
!>          البعد الرئيسي للمصفوفة A.  LDA >= max(1,KD+1).
!>

X

!>          X is DOUBLE PRECISION array, dimension (LDX,NRHS)
!>          متجهات الحل المحسوبة لنظام المعادلات الخطية.
!>

LDX

!>          LDX is INTEGER
!>          البعد الرئيسي للمصفوفة X.   LDX >= max(1,N).
!>

B

!>          B is DOUBLE PRECISION array, dimension (LDB,NRHS)
!>          عند الإدخال، متجهات الطرف الأيمن لنظام المعادلات الخطية.
!>          عند الخروج، يتم استبدال B بـ B - A*X.
!>

LDB

!>          LDB is INTEGER
!>          البعد الرئيسي للمصفوفة B.  LDB >= max(1,N).
!>

RWORK

!>          RWORK is DOUBLE PRECISION array, dimension (N)
!>

RESID

!>          RESID is DOUBLE PRECISION
!>          الحد الأقصى لعدد الأطراف اليمنى لـ
!>          norm(B - A*X) / ( norm(A) * norm(X) * EPS ).
!>
