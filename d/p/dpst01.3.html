توثيق الوظيفة/الروتين الفرعي

subroutine dpst01 (character uplo, integer n, double precision, dimension( lda, * ) a, integer lda, double precision, dimension( ldafac, * ) afac, integer ldafac, double precision, dimension( ldperm, * ) perm, integer ldperm, integer, dimension( * ) piv, double precision, dimension( * ) rwork, double precision resid, integer rank)

DPST01

الغرض:

!>
!> DPST01 يعيد بناء مصفوفة متماثلة موجبة نصف محددة A
!> من عواملها L أو U ومصفوفة التبديل P ويحسب
!> الباقي
!>    norm( P*L*L'*P' - A ) / ( N * norm(A) * EPS ) أو
!>    norm( P*U'*U*P' - A ) / ( N * norm(A) * EPS )،
!> حيث EPS هي وحدة الآلة.
!>

المعلمات

UPLO 
!>          UPLO is CHARACTER*1
!>          تحدد ما إذا كان الجزء العلوي أو السفلي المثلثي من المصفوفة المتماثلة A مخزناً:
!>          = 'U':  مثلثي علوي
!>          = 'L':  مثلثي سفلي
!>

N

!>          N is INTEGER
!>          عدد الصفوف والأعمدة للمصفوفة A.  N >= 0.
!>

A

!>          A is DOUBLE PRECISION array, dimension (LDA,N)
!>          المصفوفة المتماثلة الأصلية A.
!>

LDA

!>          LDA is INTEGER
!>          البعد الرائد للمصفوفة A.  LDA >= max(1,N)
!>

AFAC

!>          AFAC is DOUBLE PRECISION array, dimension (LDAFAC,N)
!>          العامل L أو U من تحليل L*L' أو U'*U
!>          لـ A.
!>

LDAFAC

!>          LDAFAC is INTEGER
!>          البعد الرائد للمصفوفة AFAC.  LDAFAC >= max(1,N).
!>

PERM

!>          PERM is DOUBLE PRECISION array, dimension (LDPERM,N)
!>          تتم الكتابة فوقها بالمصفوفة المعاد بناؤها، ثم بالفرق P*L*L'*P' - A (أو P*U'*U*P' - A)
!>

LDPERM

!>          LDPERM is INTEGER
!>          البعد الرائد للمصفوفة PERM.
!>          LDAPERM >= max(1,N).
!>

PIV

!>          PIV is INTEGER array, dimension (N)
!>          PIV بحيث تكون الإدخالات غير الصفرية
!>          P( PIV( K ), K ) = 1.
!>

RWORK

!>          RWORK is DOUBLE PRECISION array, dimension (N)
!>

RESID

!>          RESID is DOUBLE PRECISION
!>          If UPLO = 'L', norm(L*L' - A) / ( N * norm(A) * EPS )
!>          If UPLO = 'U', norm(U'*U - A) / ( N * norm(A) * EPS )
!>

RANK

!>          RANK is INTEGER
!>          عدد القيم المفردة غير الصفرية للمصفوفة A.
!>
