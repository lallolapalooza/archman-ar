subroutine dptsv (integer n, integer nrhs, double precision, dimension( * ) d, double precision, dimension( * ) e, double precision, dimension( ldb, * ) b, integer ldb, integer info)

DPTSV تحسب حل نظام المعادلات الخطية A * X = B لمصفوفات PT

الغرض:

!>
!> DPTSV يحسب حل نظام المعادلات الخطية الحقيقي
!> A*X = B، حيث A مصفوفة قطرية ثلاثية متناظرة وموجبة محددة بحجم N×N،
!> و X و B مصفوفتان بحجم N×NRHS.
!>
!> يتم تحليل A إلى A = L*D*L**T، ثم يتم استخدام الشكل المحلل لـ A
!> لحل نظام المعادلات.
!>

المعلمات

N 
!>          N هو عدد صحيح (INTEGER)
!>          رتبة المصفوفة A. N >= 0.
!>

NRHS

!>          NRHS هو عدد صحيح (INTEGER)
!>          عدد الأطراف اليمنى، أي عدد أعمدة المصفوفة B. NRHS >= 0.
!>

D

!>          D هو مصفوفة دقيقة مزدوجة (DOUBLE PRECISION)، البعد (N)
!>          عند الإدخال، العناصر القطرية الـ N للمصفوفة القطرية الثلاثية
!>          A. عند الخروج، العناصر القطرية الـ N للمصفوفة القطرية
!>          D من تحليل A = L*D*L**T.
!>

E

!>          E هو مصفوفة دقيقة مزدوجة (DOUBLE PRECISION)، البعد (N-1)
!>          عند الإدخال، عناصر تحت القطر الـ (N-1) للمصفوفة القطرية الثلاثية
!>          A. عند الخروج، عناصر تحت القطر الـ (N-1) للعامل القطري الأحادي
!>          L من تحليل L*D*L**T للمصفوفة A. (يمكن اعتبار E أيضًا
!>          القطر العلوي للعامل القطري الأحادي U من تحليل U**T*D*U
!>          للمصفوفة A.)
!>

B

!>          B هو مصفوفة دقيقة مزدوجة (DOUBLE PRECISION)، البعد (LDB,NRHS)
!>          عند الإدخال، مصفوفة الطرف الأيمن B بحجم N×NRHS.
!>          عند الخروج، إذا كان INFO = 0، مصفوفة الحل X بحجم N×NRHS.
!>

LDB

!>          LDB هو عدد صحيح (INTEGER)
!>          البعد الرائد للمصفوفة B. LDB >= max(1,N).
!>

INFO

!>          INFO هو عدد صحيح (INTEGER)
!>          = 0:  انتهاء ناجح
!>          < 0:  إذا كان INFO = -i، فإن الوسيط i كان له قيمة غير قانونية
!>          > 0:  إذا كان INFO = i، فإن المحدد الرئيسي الرائد من الرتبة i
!>                ليس موجبًا، ولم يتم حساب الحل. لم يكتمل التحليل
!>                إلا إذا كان i = N.
!>
