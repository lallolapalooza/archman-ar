وثائق الوظيفة/الروتين الفرعي

subroutine dptts2 (integer n, integer nrhs, double precision, dimension( * ) d, double precision, dimension( * ) e, double precision, dimension( ldb, * ) b, integer ldb)

DPTTS2 يحل نظامًا ثلاثي الأقطار بالشكل AX=B باستخدام تحليل L D LH الذي تم حسابه بواسطة spttrf.

الغرض:

!>
!> DPTTS2 يحل نظامًا ثلاثي الأقطار بالشكل
!>    A * X = B
!> باستخدام تحليل L*D*L**T لـ A الذي تم حسابه بواسطة DPTTRF. D هي مصفوفة
!> قطرية محددة في المتجه D، L هي مصفوفة ثنائية القطر الوحدة التي يتم تحديد عناصرها
!> تحت القطر الرئيسي في المتجه E، و X و B هما مصفوفتان N × NRHS.
!>

المعاملات

N 
!>          N هو عدد صحيح (INTEGER)
!>          رتبة المصفوفة ثلاثية الأقطار A. N >= 0.
!>

NRHS

!>          NRHS هو عدد صحيح (INTEGER)
!>          عدد الأطراف اليمنى، أي عدد أعمدة المصفوفة B. NRHS >= 0.
!>

D

!>          D هي مصفوفة ذات دقة مزدوجة (DOUBLE PRECISION)، البعد (N)
!>          العناصر القطرية N للمصفوفة القطرية D من تحليل
!>          L*D*L**T لـ A.
!>

E

!>          E هي مصفوفة ذات دقة مزدوجة (DOUBLE PRECISION)، البعد (N-1)
!>          العناصر تحت القطرية (N-1) للعامل ثنائي القطر الوحدة L من تحليل
!>          L*D*L**T لـ A. يمكن اعتبار E أيضًا على أنها القطر الفوقي للعامل ثنائي القطر الوحدة U من
!>          التحليل A = U**T*D*U.
!>

B

!>          B هي مصفوفة ذات دقة مزدوجة (DOUBLE PRECISION)، البعد (LDB,NRHS)
!>          عند الدخول: متجهات الطرف الأيمن B لنظام المعادلات الخطية.
!>          عند الخروج: متجهات الحل X.
!>

LDB

!>          LDB هو عدد صحيح (INTEGER)
!>          البعد الرئيسي للمصفوفة B. LDB >= max(1,N).
!>
