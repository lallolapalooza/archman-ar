subroutine dpotf2 (character uplo, integer n, double precision, dimension( lda, * ) a, integer lda, integer info)

DPOTF2 تحسب تحليل Cholesky لمصفوفة متماثلة/هرمية موجبة محددة (خوارزمية غير محظورة).

الغرض:

!>
!> DPOTF2 تحسب تحليل Cholesky لمصفوفة A الحقيقية المتماثلة
!> موجبة المحددة.
!>
!> يكون التحليل على الشكل
!>    A = U**T * U ,  إذا كان UPLO = 'U', أو
!>    A = L  * L**T,  إذا كان UPLO = 'L',
!> حيث U مصفوفة مثلثة علوية و L مصفوفة مثلثة سفلية.
!>
!> هذه هي النسخة غير المحظورة من الخوارزمية، تستدعي BLAS المستوى 2.
!>

المعلمات

UPLO 
!>          UPLO هو CHARACTER*1
!>          يحدد ما إذا كان الجزء العلوي أو السفلي المثلثي من المصفوفة
!>          المتماثلة A مخزناً.
!>          = 'U':  مثلثي علوي
!>          = 'L':  مثلثي سفلي
!>

N

!>          N هو INTEGER
!>          ترتيب المصفوفة A.  N >= 0.
!>

A

!>          A هو مصفوفة DOUBLE PRECISION، بأبعاد (LDA,N)
!>          عند الإدخال، المصفوفة المتماثلة A.  إذا كان UPLO = 'U'، فإن الجزء
!>          المثلثي العلوي n x n الرائد في A يحتوي على الجزء المثلثي
!>          العلوي للمصفوفة A، ولا يتم الرجوع إلى الجزء المثلثي السفلي
!>          الصارم في A.  إذا كان UPLO = 'L'، فإن الجزء المثلثي السفلي
!>          n x n الرائد في A يحتوي على الجزء المثلثي السفلي للمصفوفة A،
!>          ولا يتم الرجوع إلى الجزء المثلثي العلوي الصارم في A.
!>
!>          عند الخروج، إذا كان INFO = 0، فإن العامل U أو L من تحليل
!>          Cholesky A = U**T *U  أو A = L*L**T.
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرائد للمصفوفة A.  LDA >= max(1,N).
!>

INFO

!>          INFO هو INTEGER
!>          = 0: نجاح الخروج
!>          < 0: إذا كان INFO = -k، فإن المعلمة k كانت ذات قيمة غير قانونية
!>          > 0: إذا كان INFO = k، فإن الحد الرئيسي الرائد من الدرجة k
!>               ليس موجباً، ولم يكن من الممكن إكمال التحليل.
!>
