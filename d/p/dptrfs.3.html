subroutine dptrfs (integer n, integer nrhs, double precision, dimension( * ) d, double precision, dimension( * ) e, double precision, dimension( * ) df, double precision, dimension( * ) ef, double precision, dimension( ldb, * ) b, integer ldb, double precision, dimension( ldx, * ) x, integer ldx, double precision, dimension( * ) ferr, double precision, dimension( * ) berr, double precision, dimension( * ) work, integer info)

DPTRFS

الغرض:

!>
!> DPTRFS يحسن الحل المحسوب لنظام المعادلات الخطية
!> عندما تكون مصفوفة المعاملات موجبة محددة متناظرة
!> وثلاثية الأقطار، ويوفر حدودًا للأخطاء وتقديرات للخطأ الخلفي
!> للحل.
!>

المعاملات

N 
!>          N هو عدد صحيح (INTEGER)
!>          ترتيب المصفوفة A.  N >= 0.
!>

NRHS

!>          NRHS هو عدد صحيح (INTEGER)
!>          عدد الأطراف اليمنى، أي عدد أعمدة
!>          المصفوفة B.  NRHS >= 0.
!>

D

!>          D هو مصفوفة من النوع DOUBLE PRECISION
!>          الأبعاد (N)
!>          العناصر القطرية n للمصفوفة ثلاثية الأقطار A.
!>

E

!>          E هو مصفوفة من النوع DOUBLE PRECISION
!>          الأبعاد (N-1)
!>          العناصر تحت القطرية (n-1) للمصفوفة ثلاثية الأقطار A.
!>

DF

!>          DF هو مصفوفة من النوع DOUBLE PRECISION
!>          الأبعاد (N)
!>          العناصر القطرية n للمصفوفة القطرية D من
!>          التحليل المحسوب بواسطة DPTTRF.
!>

EF

!>          EF هو مصفوفة من النوع DOUBLE PRECISION
!>          الأبعاد (N-1)
!>          العناصر تحت القطرية (n-1) للعامل ثنائي القطر الوحدة
!>          L من التحليل المحسوب بواسطة DPTTRF.
!>

B

!>          B هو مصفوفة من النوع DOUBLE PRECISION
!>          الأبعاد (LDB,NRHS)
!>          المصفوفة الطرفية اليمنى B.
!>

LDB

!>          LDB هو عدد صحيح (INTEGER)
!>          البعد الرئيسي للمصفوفة B.  LDB >= max(1,N).
!>

X

!>          X هو مصفوفة من النوع DOUBLE PRECISION
!>          الأبعاد (LDX,NRHS)
!>          عند الإدخال، المصفوفة الحل X، كما حسبت بواسطة DPTTRS.
!>          عند الخروج، المصفوفة الحل المحسنة X.
!>

LDX

!>          LDX هو عدد صحيح (INTEGER)
!>          البعد الرئيسي للمصفوفة X.  LDX >= max(1,N).
!>

FERR

!>          FERR هو مصفوفة من النوع DOUBLE PRECISION
!>          الأبعاد (NRHS)
!>          حد الخطأ الأمامي لكل متجه الحل
!>          X(j) (العمود j من مصفوفة الحل X).
!>          إذا كان XTRUE هو الحل الحقيقي المقابل لـ X(j)، فإن FERR(j)
!>          هو تقدير للحد العلوي لقيمة أكبر
!>          عنصر في (X(j) - XTRUE) مقسومًا على قيمة أكبر
!>          عنصر في X(j).
!>

BERR

!>          BERR هو مصفوفة من النوع DOUBLE PRECISION
!>          الأبعاد (NRHS)
!>          الخطأ النسبي المكون لكل متجه حل
!>          X(j) (أي، أصغر تغيير نسبي في
!>          أي عنصر من A أو B يجعل X(j) حلاً دقيقًا).
!>

WORK

!>          WORK هو مصفوفة من النوع DOUBLE PRECISION
!>          الأبعاد (2*N)
!>

INFO

!>          INFO هو عدد صحيح (INTEGER)
!>          = 0:  خروج ناجح
!>          < 0:  إذا كان INFO = -i، فإن المعامل i كان له قيمة غير قانونية
!>
