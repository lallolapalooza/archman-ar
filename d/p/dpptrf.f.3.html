subroutine dpptrf (character uplo, integer n, double precision, dimension( * ) ap, integer info)

DPPTRF

الغرض:

!>
!> DPPTRF يحسب تحليل Cholesky لمصفوفة متماثلة موجبة محددة A مخزنة بتنسيق مضغوط.
!>
!> التحليل له الشكل
!>    A = U**T * U,  إذا كان UPLO = 'U', أو
!>    A = L  * L**T,  إذا كان UPLO = 'L',
!> حيث U مصفوفة مثلثية عليا و L مصفوفة مثلثية سفلى.
!>

المعلمات

UPLO 
!>          UPLO هو CHARACTER*1
!>          = 'U':  يتم تخزين المثلث العلوي لـ A؛
!>          = 'L':  يتم تخزين المثلث السفلي لـ A.
!>

N

!>          N هو INTEGER
!>          رتبة المصفوفة A.  N >= 0.
!>

AP

!>          AP هو مصفوفة DOUBLE PRECISION، حجمها (N*(N+1)/2)
!>          عند الإدخال، المثلث العلوي أو السفلي للمصفوفة المتماثلة
!>          A، مضغوطة حسب العمود في مصفوفة خطية. يتم تخزين العمود j للمصفوفة A
!>          في المصفوفة AP كما يلي:
!>          إذا كان UPLO = 'U', AP(i + (j-1)*j/2) = A(i,j) لـ 1<=i<=j؛
!>          إذا كان UPLO = 'L', AP(i + (j-1)*(2n-j)/2) = A(i,j) لـ j<=i<=n.
!>          انظر أدناه لمزيد من التفاصيل.
!>
!>          عند الخروج، إذا كان INFO = 0، فإن العامل المثلثي U أو L من
!>          تحليل Cholesky A = U**T*U أو A = L*L**T، بنفس
!>          تنسيق التخزين مثل A.
!>

INFO

!>          INFO هو INTEGER
!>          = 0:  انتهاء ناجح
!>          < 0:  إذا كان INFO = -i، فإن المعامل i كان له قيمة غير قانونية
!>          > 0:  إذا كان INFO = i، فإن القطر الرئيسي الرائد من الرتبة i
!>                ليس موجبًا، ولم يتمكن التحليل من الاكتمال.
!>
