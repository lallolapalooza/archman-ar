subroutine dpstrf (character uplo, integer n, double precision, dimension( lda, * ) a, integer lda, integer, dimension( n ) piv, integer rank, double precision tol, double precision, dimension( 2*n ) work, integer info)

DPSTRF تحسب تحليل Cholesky مع تبديل كامل لمصفوفة حقيقية متناظرة موجبة شبه محددة.

الغرض:

!>
!> DPSTRF تحسب تحليل Cholesky مع تبديل كامل لمصفوفة حقيقية متناظرة موجبة شبه محددة A.
!>
!> التحلل له الشكل
!>    P**T * A * P = U**T * U ,  if UPLO = 'U',
!>    P**T * A * P = L  * L**T,  if UPLO = 'L',
!> حيث U مصفوفة مثلثية علوية و L مصفوفة مثلثية سفلية، و
!> P مخزنة كمتجه PIV.
!>
!> هذه الخوارزمية لا تحاول التحقق من أن A موجبة شبه محددة. هذا الإصدار من الخوارزمية يستدعي BLAS مستوى 3.
!>

المعاملات

UPLO 
!>          UPLO is CHARACTER*1
!>          يحدد ما إذا كان الجزء المثلثي العلوي أو السفلي للمصفوفة المتناظرة A مخزنًا.
!>          = 'U':  مثلثي علوي
!>          = 'L':  مثلثي سفلي
!>

N

!>          N is INTEGER
!>          رتبة المصفوفة A.  N >= 0.
!>

A

!>          A is DOUBLE PRECISION array, dimension (LDA,N)
!>          عند الدخول، المصفوفة المتناظرة A.  إذا كان UPLO = 'U'، فإن الجزء العلوي
!>          المثلثي العلوي من A الذي حجمه n × n يحتوي على الجزء المثلثي العلوي
!>          للمصفوفة A، ولا تتم الإشارة إلى الجزء المثلثي السفلي الصارم من A.  إذا كان UPLO = 'L'، فإن الجزء
!>          المثلثي السفلي من A الذي حجمه n × n يحتوي على الجزء المثلثي السفلي
!>          للمصفوفة A، ولا تتم الإشارة إلى الجزء المثلثي العلوي الصارم من A.
!>
!>          عند الخروج، إذا كان INFO = 0، فإن العامل U أو L من تحليل Cholesky
!>          كما هو مذكور أعلاه.
!>

LDA

!>          LDA is INTEGER
!>          البعد الرئيسي للمصفوفة A.  LDA >= max(1,N).
!>

PIV

!>          PIV is INTEGER array, dimension (N)
!>          PIV بحيث تكون المدخلات غير الصفرية هي P( PIV(K), K ) = 1.
!>

RANK

!>          RANK is INTEGER
!>          رتبة A المعطاة بعدد الخطوات التي أكملتها الخوارزمية.
!>

TOL

!>          TOL is DOUBLE PRECISION
!>          التسامح المحدد من قبل المستخدم. إذا كان TOL < 0، فسيتم استخدام N*U*MAX( A(K,K) ).
!>          تنتهي الخوارزمية عند الخطوة (K-1) إذا كان المحور <= TOL.
!>

WORK

!>          WORK is DOUBLE PRECISION array, dimension (2*N)
!>          مساحة عمل.
!>

INFO

!>          INFO is INTEGER
!>          < 0: إذا كان INFO = -K، فإن المعامل K كان له قيمة غير قانونية،
!>          = 0: اكتملت الخوارزمية بنجاح، و
!>          > 0: المصفوفة A إما أن تكون ذات رتبة ناقصة مع رتبة محسوبة
!>               كما هو معطى في RANK، أو ليست موجبة شبه محددة. انظر
!>               القسم 7 من ملاحظة عمل LAPACK #161 لمزيد من
!>               المعلومات.
!>
