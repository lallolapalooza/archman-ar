توثيق الوظيفة/الروتين الفرعي

subroutine dpocon (character uplo, integer n, double precision, dimension( lda, * ) a, integer lda, double precision anorm, double precision rcond, double precision, dimension( * ) work, integer, dimension( * ) iwork, integer info)

DPOCON

الغرض:

!>
!> DPOCON تقدير معكوس عدد الشرط (بمعيار 1) لمصفوفة متماثلة حقيقية موجبة محددة
!> باستخدام تحليل Cholesky A = U**T*U أو A = L*L**T الذي تم حسابه بواسطة DPOTRF.
!>
!> يتم الحصول على تقدير لـ norm(inv(A))، ويتم حساب معكوس عدد الشرط على أنه RCOND = 1 / (ANORM * norm(inv(A))).
!>

المعاملات

UPLO 
!>          UPLO هو CHARACTER*1
!>          = 'U':  يتم تخزين المثلث العلوي لـ A;
!>          = 'L':  يتم تخزين المثلث السفلي لـ A.
!>

N

!>          N هو INTEGER
!>          رتبة المصفوفة A.  N >= 0.
!>

A

!>          A هو مصفوفة DOUBLE PRECISION، ذات أبعاد (LDA,N)
!>          العامل المثلثي U أو L من تحليل Cholesky
!>          A = U**T*U أو A = L*L**T، كما تم حسابه بواسطة DPOTRF.
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرئيسي للمصفوفة A.  LDA >= max(1,N).
!>

ANORM

!>          ANORM هو DOUBLE PRECISION
!>          معيار 1 (أو معيار ما لا نهاية) للمصفوفة المتماثلة A.
!>

RCOND

!>          RCOND هو DOUBLE PRECISION
!>          معكوس عدد الشرط للمصفوفة A،
!>          محسوب على أنه RCOND = 1/(ANORM * AINVNM)، حيث AINVNM هو
!>          تقدير لمعيار 1 لـ inv(A) محسوب في هذا الروتين.
!>

WORK

!>          WORK هي مصفوفة DOUBLE PRECISION، ذات أبعاد (3*N)
!>

IWORK

!>          IWORK هي مصفوفة INTEGER، ذات أبعاد (N)
!>

INFO

!>          INFO هو INTEGER
!>          = 0:  انتهاء ناجح
!>          < 0:  إذا كان INFO = -i، فإن المعامل i كان له قيمة غير قانونية
!>
