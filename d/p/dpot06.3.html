subroutine dpot06 (character uplo, integer n, integer nrhs, double precision, dimension( lda, * ) a, integer lda, double precision, dimension( ldx, * ) x, integer ldx, double precision, dimension( ldb, * ) b, integer ldb, double precision, dimension( * ) rwork, double precision resid)

DPOT06

الغرض:

!>
!> DPOT06 يحسب الباقي لحل نظام المعادلات الخطية  A*x = b :
!>    RESID = norm(B - A*X,inf) / ( norm(A,inf) * norm(X,inf) * EPS ),
!> حيث EPS هو الخطأ المعياري للآلة.
!>

المعلمات

UPLO 
!>          UPLO is CHARACTER*1
!>          يحدد ما إذا كان الجزء العلوي أو السفلي المثلثي للمصفوفة المتماثلة A مخزنًا:
!>          = 'U':  مثلثي علوي
!>          = 'L':  مثلثي سفلي
!>

N

!>          N is INTEGER
!>          عدد الصفوف والأعمدة للمصفوفة A.  N >= 0.
!>

NRHS

!>          NRHS is INTEGER
!>          عدد الأعمدة في B، مصفوفة الأطراف اليمنى.
!>          NRHS >= 0.
!>

A

!>          A is DOUBLE PRECISION array, dimension (LDA,N)
!>          المصفوفة الأصلية A بحجم M x N.
!>

LDA

!>          LDA is INTEGER
!>          البعد الرئيسي للمصفوفة A.  LDA >= max(1,N).
!>

X

!>          X is DOUBLE PRECISION array, dimension (LDX,NRHS)
!>          متجهات الحل المحسوبة لنظام المعادلات الخطية.
!>

LDX

!>          LDX is INTEGER
!>          البعد الرئيسي للمصفوفة X.  إذا كان TRANS = 'N'،
!>          LDX >= max(1,N)؛ إذا كان TRANS = 'T' أو 'C'، LDX >= max(1,N).
!>

B

!>          B is DOUBLE PRECISION array, dimension (LDB,NRHS)
!>          عند الإدخال، متجهات الطرف الأيمن لنظام المعادلات الخطية.
!>          عند الخروج، يتم استبدال B بالفرق B - A*X.
!>

LDB

!>          LDB is INTEGER
!>          البعد الرئيسي للمصفوفة B.  إذا كان TRANS = 'N'،
!>          LDB >= max(1,M)؛ إذا كان TRANS = 'T' أو 'C'، LDB >= max(1,N).
!>

RWORK

!>          RWORK is DOUBLE PRECISION array, dimension (N)
!>

RESID

!>          RESID is DOUBLE PRECISION
!>          الحد الأقصى على عدد الأطراف اليمنى لـ
!>          norm(B - A*X) / ( norm(A) * norm(X) * EPS ).
!>
