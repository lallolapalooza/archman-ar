توثيق الوظيفة/الإجراء الفرعي

subroutine dpbt05 (character uplo, integer n, integer kd, integer nrhs, double precision, dimension( ldab, * ) ab, integer ldab, double precision, dimension( ldb, * ) b, integer ldb, double precision, dimension( ldx, * ) x, integer ldx, double precision, dimension( ldxact, * ) xact, integer ldxact, double precision, dimension( * ) ferr, double precision, dimension( * ) berr, double precision, dimension( * ) reslts)

DPBT05

الغرض:

!>
!> DPBT05 يختبر حدود الخطأ من التحسين التكراري للحل المحسوب لنظام المعادلات A*X = B، حيث A هي مصفوفة شريطية متناظرة.
!>
!> RESLTS(1) = اختبار حد الخطأ
!>           = norm(X - XACT) / ( norm(X) * FERR )
!>
!> يتم إرجاع قيمة كبيرة إذا كانت هذه النسبة ليست أقل من واحد.
!>
!> RESLTS(2) = المتبقي من روتين التحسين التكراري
!>           = الحد الأقصى لـ BERR / ( NZ*EPS + (*) )، حيث
!>             (*) = NZ*UNFL / (min_i (abs(A)*abs(X) +abs(b))_i )
!>             و NZ = الحد الأقصى لعدد العناصر غير الصفرية في أي صف من A، بالإضافة إلى 1
!>

المعلمات

UPLO 
!>          UPLO هو CHARACTER*1
!>          يحدد ما إذا كان الجزء المثلثي العلوي أو السفلي من المصفوفة المتناظرة A مخزنًا.
!>          = 'U':  مثلثي علوي
!>          = 'L':  مثلثي سفلي
!>

N

!>          N هو INTEGER
!>          عدد صفوف المصفوفات X، B، و XACT، وترتيب المصفوفة A. N >= 0.
!>

KD

!>          KD هو INTEGER
!>          عدد الأقطار الفائقة للمصفوفة A إذا كان UPLO = 'U'،
!>          أو عدد الأقطار الفرعية إذا كان UPLO = 'L'. KD >= 0.
!>

NRHS

!>          NRHS هو INTEGER
!>          عدد أعمدة المصفوفات X، B، و XACT.
!>          NRHS >= 0.
!>

AB

!>          AB هي مصفوفة DOUBLE PRECISION، بأبعاد (LDAB,N)
!>          الجزء العلوي أو السفلي من مصفوفة الشريط المتناظرة A،
!>          مخزن في أول KD+1 صف من المصفوفة. يتم تخزين العمود j من A
!>          في العمود j من المصفوفة AB كالتالي:
!>          إذا كان UPLO = 'U'، AB(kd+1+i-j,j) = A(i,j) لـ max(1,j-kd)<=i<=j;
!>          إذا كان UPLO = 'L'، AB(1+i-j,j)    = A(i,j) لـ j<=i<=min(n,j+kd).
!>

LDAB

!>          LDAB هو INTEGER
!>          البعد الرئيسي للمصفوفة AB. LDAB >= KD+1.
!>

B

!>          B هي مصفوفة DOUBLE PRECISION، بأبعاد (LDB,NRHS)
!>          متجهات الجانب الأيمن لنظام المعادلات الخطية.
!>

LDB

!>          LDB هو INTEGER
!>          البعد الرئيسي للمصفوفة B. LDB >= max(1,N).
!>

X

!>          X هي مصفوفة DOUBLE PRECISION، بأبعاد (LDX,NRHS)
!>          متجهات الحل المحسوبة. يتم تخزين كل متجه كعمود في المصفوفة X.
!>

LDX

!>          LDX هو INTEGER
!>          البعد الرئيسي للمصفوفة X. LDX >= max(1,N).
!>

XACT

!>          XACT هي مصفوفة DOUBLE PRECISION، بأبعاد (LDX,NRHS)
!>          متجهات الحل الدقيقة. يتم تخزين كل متجه كعمود في المصفوفة XACT.
!>

LDXACT

!>          LDXACT هو INTEGER
!>          البعد الرئيسي للمصفوفة XACT. LDXACT >= max(1,N).
!>

FERR

!>          FERR هي مصفوفة DOUBLE PRECISION، بأبعاد (NRHS)
!>          تقديرات حدود الخطأ الأمامي لكل متجه حل X. إذا كان XTRUE هو الحل الحقيقي،
!>          فإن FERR يحد من حجم أكبر مدخل في (X - XTRUE) مقسومًا على حجم أكبر مدخل في X.
!>

BERR

!>          BERR هي مصفوفة DOUBLE PRECISION، بأبعاد (NRHS)
!>          الخطأ الخلفي النسبي المكون من كل متجه حل (أي، أصغر تغيير نسبي في أي مدخل من A أو B
!>          يجعل X حلاً دقيقًا).
!>

RESLTS

!>          RESLTS هي مصفوفة DOUBLE PRECISION، بأبعاد (2)
!>          الحد الأقصى على متجهات الحل NRHS للنسب التالية:
!>          RESLTS(1) = norm(X - XACT) / ( norm(X) * FERR )
!>          RESLTS(2) = BERR / ( NZ*EPS + (*) )
!>
