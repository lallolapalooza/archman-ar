توثيق الوظيفة/الروتين الفرعي

subroutine dpteqr (character compz, integer n, double precision, dimension( * ) d, double precision, dimension( * ) e, double precision, dimension( ldz, * ) z, integer ldz, double precision, dimension( * ) work, integer info)

DPTEQR

الغرض:

!>
!> DPTEQR تحسب جميع القيم الذاتية، وبشكل اختياري، المتجهات الذاتية لمصفوفة
!> ثلاثية قطرية متناظرة موجبة محددة عن طريق أولاً تحليل المصفوفة
!> باستخدام DPTTRF، ثم استدعاء DBDSQR لحساب قيم التفرد للعامل ثنائي القطر.
!>
!> هذا الروتين يحسب القيم الذاتية للمصفوفة ثلاثية القطرية الموجبة
!> المحددة بدقة نسبية عالية. هذا يعني أنه إذا تراوحت القيم الذاتية
!> عبر العديد من الأوامر من حيث الحجم، فسيتم حساب القيم الذاتية الصغيرة
!> والمتجهات الذاتية المقابلة لها بدقة أكبر مما هو عليه الحال، على سبيل المثال،
!> مع طريقة QR القياسية.
!>
!> يمكن أيضًا العثور على المتجهات الذاتية لمصفوفة كاملة أو نطاقية متناظرة
!> موجبة محددة إذا تم استخدام DSYTRD أو DSPTRD أو DSBTRD
!> لتقليل هذه المصفوفة إلى شكل ثلاثي قطري. (ومع ذلك، قد يستبعد التقليل إلى
!> شكل ثلاثي قطري إمكانية الحصول على دقة نسبية عالية في القيم الذاتية الصغيرة
!> للمصفوفة الأصلية، إذا تراوحت هذه القيم الذاتية عبر العديد من الأوامر من حيث الحجم.)
!>

المعلمات

COMPZ 
!>          COMPZ is CHARACTER*1
!>          = 'N':  احسب القيم الذاتية فقط.
!>          = 'V':  احسب المتجهات الذاتية للمصفوفة المتناظرة الأصلية
!>                  أيضاً. يحتوي الصف Z على المصفوفة المتعامدة
!>                  المستخدمة لتقليل المصفوفة الأصلية إلى
!>                  شكل ثلاثي قطري.
!>          = 'I':  احسب المتجهات الذاتية للمصفوفة ثلاثية القطرية
!>                  أيضاً.
!>

N

!>          N is INTEGER
!>          رتبة المصفوفة. N >= 0.
!>

D

!>          D is DOUBLE PRECISION array, dimension (N)
!>          عند الإدخال، العناصر القطرية n للمصفوفة ثلاثية القطرية.
!>          عند الخروج العادي، يحتوي D على القيم الذاتية، بترتيب تنازلي.
!>

E

!>          E is DOUBLE PRECISION array, dimension (N-1)
!>          عند الإدخال، عناصر النطاق الفرعي (n-1) للمصفوفة ثلاثية القطرية.
!>          عند الخروج، تم تدمير E.
!>

Z

!>          Z is DOUBLE PRECISION array, dimension (LDZ, N)
!>          عند الإدخال، إذا كان COMPZ = 'V'، فإن المصفوفة المتعامدة المستخدمة في
!>          التقليل إلى شكل ثلاثي قطري.
!>          عند الخروج، إذا كان COMPZ = 'V'، فإن المتجهات الذاتية المتعامدة العادية
!>          للمصفوفة المتناظرة الأصلية؛
!>          إذا كان COMPZ = 'I'، فإن المتجهات الذاتية المتعامدة العادية
!>          للمصفوفة ثلاثية القطرية.
!>          إذا كان INFO > 0 عند الخروج، يحتوي Z على المتجهات الذاتية المرتبطة
!>          بالقيم الذاتية المخزنة فقط.
!>          إذا كان COMPZ = 'N'، فلن يتم الرجوع إلى Z.
!>

LDZ

!>          LDZ is INTEGER
!>          البعد الرائد للمصفوفة Z. LDZ >= 1، وإذا كان
!>          COMPZ = 'V' أو 'I'، فإن LDZ >= max(1,N).
!>

WORK

!>          WORK is DOUBLE PRECISION array, dimension (4*N)
!>

INFO

!>          INFO is INTEGER
!>          = 0:  خروج ناجح.
!>          < 0:  إذا كان INFO = -i، فإن الوسيط i كان لديه قيمة غير قانونية.
!>          > 0:  إذا كان INFO = i، وكان i هو:
!>                <= N  فشل تحليل Cholesky للمصفوفة
!>                      لأن المتجه الرئيسي الأمامي من الرتبة i لم يكن موجبًا.
!>                > N   فشل خوارزمية SVD في التقارب؛
!>                      إذا كان INFO = N+i، فإن i من العناصر خارج القطر
!>                      للعامل ثنائي القطر لم تتقارب إلى الصفر.
!>
