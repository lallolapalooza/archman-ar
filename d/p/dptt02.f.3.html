توثيق الوظيفة/الروتين الفرعي

subroutine dptt02 (integer n, integer nrhs, double precision, dimension( * ) d, double precision, dimension( * ) e, double precision, dimension( ldx, * ) x, integer ldx, double precision, dimension( ldb, * ) b, integer ldb, double precision resid)

DPTT02

الغرض:

!>
!> DPTT02 يحسب الباقي لحل نظام المعادلات المتناظر ثلاثي القطر:
!>    RESID = norm(B - A*X) / (norm(A) * norm(X) * EPS),
!> حيث EPS هو الخطأ المطبعي.
!>

المعلمات

N 
!>          N هو عدد صحيح
!>          رتبة المصفوفة A.
!>

NRHS

!>          NRHS هو عدد صحيح
!>          عدد الأطراف اليمنى، أي عدد أعمدة
!>          المصفوفتين B و X.  NRHS >= 0.
!>

D

!>          D هي مصفوفة ذات دقة مزدوجة، البعد (N)
!>          عناصر القطر الرئيسية لثلاثي الأقطار A.
!>

E

!>          E هي مصفوفة ذات دقة مزدوجة، البعد (N-1)
!>          عناصر تحت القطر الرئيسية لثلاثي الأقطار A.
!>

X

!>          X هي مصفوفة ذات دقة مزدوجة، البعد (LDX,NRHS)
!>          مصفوفة الحلول X بعمود NRHS.
!>

LDX

!>          LDX هو عدد صحيح
!>          البعد الرئيسي للمصفوفة X.  LDX >= max(1,N).
!>

B

!>          B هي مصفوفة ذات دقة مزدوجة، البعد (LDB,NRHS)
!>          عند الإدخال، مصفوفة الأطراف اليمنى B.
!>          عند الخروج، يتم استبدال B بالفرق B - A*X.
!>

LDB

!>          LDB هو عدد صحيح
!>          البعد الرئيسي للمصفوفة B.  LDB >= max(1,N).
!>

RESID

!>          RESID هو دقة مزدوجة
!>          norm(B - A*X) / (norm(A) * norm(X) * EPS)
!>
