JLP(JLP)

الحزمة: judy

التراخيص: LGPL

اسم

JLP, JLL, JLFE, JLNE, JLLE, JLPE - واجهات برمجة تطبيقاتJudyL.

موجز

#include <Judy.h>
Word_t   Index;
Word_t   *PValue;
Pvoid_t  PJLArray;
int      Rc_int;

/* Search for the last index present that is equal to or less than Index */
JLL(PValue, PJLArray, Index);

/* Search for the previous index present that is less than Index */
JLP(PValue, PJLArray, Index);

/* Search for the first index absent that is equal to or greater than Index */
JLFE(Rc_int, PJLArray, Index);

/* Search for the next index absent that is greater than Index */
JLNE(Rc_int, PJLArray, Index);

/* Search for the last index absent that is equal to or less than Index */
JLLE(Rc_int, PJLArray, Index);

/* Search for the previous index absent that is less than Index */
JLPE(Rc_int, PJLArray, Index);

الوصف

JudyL هي عبارة عن هياكل بيانات ديناميكية تعمل بكفاءة عالية وتشغل مساحة ذاكرة قليلة، وهي مصممة لتخزين أزواج من الفهرس/القيمة (index/value) حيث تكون الفهارس أعدادًا صحيحة غير سالبة. واجهات برمجة التطبيقات (APIs) الموضحة هنا هي واجهات سياق عكسية (reverse context) تعمل مع JudyL.

JLL(PValue, PJLArray, Index) // JudyLFirst()
البحث (بما في ذلك) عن أول فهرس موجود في المصفوفة. (ابدأ بـ Index = 0 للعثور على أول فهرس في المصفوفة).
JLN(PValue, PJLArray, Index) // JudyLNext()
البحث (حصري) عن الفهرس التالي الموجود الذي يزيد عن Index الممرر. يستخدم JLN() عادةً لمتابعة مسح الفهارس الموجودة بترتيب تصاعدي في مصفوفة JudyL.
JLF(PValue, PJLArray, Index) // JudyLLast()
البحث (بما في ذلك) عن آخر فهرس موجود مساوٍ أو أقل من Index الممرر. (ابدأ بـ Index = -1، أي جميع الآحاد، للعثور على آخر فهرس في المصفوفة). يستخدم JLL() عادةً *لبدء* مسح بترتيب تنازلي للفهارس الموجودة في مصفوفة JudyL.
JLP(PValue, PJLArray, Index) // JudyLPrev()
البحث (حصري) عن الفهرس السابق الموجود الذي يقل عن Index الممرر. يستخدم JLP() عادةً *لمتابعة* مسح بترتيب تنازلي للفهارس الموجودة في مصفوفة JudyL، أو لتحديد موقع "جيران" لفهرس معين.
JLFE(Rc_int, PJLArray, Index) // JudyLFirstEmpty()
البحث (بما في ذلك) عن أول فهرس غائب مساوٍ أو أكبر من Index الممرر. (ابدأ بـ Index = 0 للعثور على أول فهرس غائب في المصفوفة).
JLNE(Rc_int, PJLArray, Index) // JudyLNextEmpty()
البحث (حصري) عن الفهرس الغائب التالي الذي يزيد عن Index الممرر.
JLLE(Rc_int, PJLArray, Index) // JudyLLastEmpty()
البحث (بما في ذلك) عن آخر فهرس غائب مساوٍ أو أقل من Index الممرر. (ابدأ بـ Index = -1، أي جميع الآحاد، للعثور على آخر فهرس غائب في المصفوفة).
JLPE(Rc_int, PJLArray, Index) // JudyLPrevEmpty()
البحث (حصري) عن الفهرس الغائب السابق الذي يقل عن Index الممرر.

مصفوفات JudyL متعددة الأبعاد

يعد تخزين مؤشر إلى مصفوفة JudyL أخرى في Value لمصفوفة JudyL طريقة بسيطة لدعم المصفوفات متعددة الأبعاد الديناميكية. هذه المصفوفات (أو الأشجار) المبنية باستخدام مصفوفات JudyL سريعة جدًا وفعالة في استخدام الذاكرة. (في الواقع، هذا هو كيف يتم تنفيذ JudySL و JudyHS). يمكن تحقيق عدد تعسفي من الأبعاد بهذه الطريقة. لإنهاء عدد الأبعاد (أو الشجرة)، يتم تمييز مؤشر Value للإشارة إلى أنه *لا* يشير إلى مصفوفة Judy أخرى. يتم استخدام علامة JLAP_INVALID في أقل بت(بتات) من المؤشر. بعد إزالة العلامة JLAP_INVALID، يتم استخدامها كمؤشر لبيانات المستخدم. يحدد ملف الترويسة Judy.h JLAP_INVALID. انظر جزء الكود أدناه.

ملاحظة: الإصدار الحالي من Judy.h غيّر هذه العلامة من 0x4 إلى 0x1 للسماح *لمخصص الذاكرة* (malloc) الذي لا يعيد الذاكرة على حدود محاذاة 8 بايت (مثل الإصدارات القديمة من valgrind).

يمكن استخدام مقطع الكود المثال التالي لتحديد ما إذا كان المؤشر يشير إلى JudyL آخر أم لا:

PValue = (PWord_t)PMultiDimArray;
for (Dim = 0; ;Dim++)
{
   if (PValue == (PWord_t)NULL) goto IndexNotFound;
   /* الانتقال إلى البعد التالي في المصفوفة */
   JLG(PValue, (Pvoid_t)*PValue, Index[Dim]);
   /* التحقق مما إذا كان مؤشرًا إلى مخزن مؤقت للمستخدم: */
   if (*PValue & JLAP_INVALID)) break;
}
UPointer = (UPointer_t) (*PValue & ~JLAP_INVALID);  // إخفاء (mask) وتحويل.
printf("User object pointer is 0x%lx\n", (Word_t) UPointer);
       ...

ملاحظة: يعمل هذا لأن *malloc()* يضمن إرجاع مؤشر بأقل بت(بتات) == 0x0. يجب عليك إزالة JLAP_INVALID قبل استخدام المؤشر.

مثال

قراءة سلسلة من أزواج الفهرس/القيمة من الإدخال القياسي، وتخزينها في مصفوفة JudyL، ثم طباعتها بترتيب فرز. ثم زيارة كل الفهارس المخزنة بترتيب تصاعدي، ثم تنازلي، وحذف كل فهرس أثناء المرور التنازلي.

#include <stdio.h>
#include <Judy.h>
Word_t   Index;                     /* فهرس المصفوفة */
Word_t   Value;                     /* قيمة عنصر المصفوفة */
Word_t * PValue;                    /* مؤشر إلى قيمة العنصر */
int      Rc_int;                    /* رمز الاستجابة */
Pvoid_t  PJLArray = (Pvoid_t) NULL; /* تهيئة مصفوفة JudyL */
while (scanf("%lu %lu", &Index, &Value))
{
    JLI(PValue, PJLArray, Index);
    If (PValue == PJERR) goto process_malloc_failure;
    *PValue = Value;                 /* تخزين القيمة الجديدة */
}
/* بعد ذلك، قم بزيارة جميع الفهارس المخزنة بترتيب فرز، أولاً تصاعديًا،
   ثم تنازليًا، واحذف كل فهرس أثناء المرور التنازلي. */
Index = 0;
JLF(PValue, PJLArray, Index);
while (PValue != NULL)
{
    printf("%lu %lu\n", Index, *PValue));
    JLN(PValue, PJLArray, Index);
}
Index = -1;
JLL(PValue, PJLArray, Index);
while (PValue != NULL)
{
    printf("%lu %lu\n", Index, *PValue));
    JLD(Rc_int, PJLArray, Index);
    if (Rc_int == JERR) goto process_malloc_failure;
    JLP(PValue, PJLArray, Index);
}
