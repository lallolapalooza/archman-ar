الوصف

JudyLLast() - يبحث عن أكبر فهرس يكون أقل من أو يساوي الفهرس المقدم.

JudyLLastEmpty() - يبحث عن أول فهرس غير موجود، والذي يكون أكبر من أو يساوي الفهرس المقدم.

JudyLPrev() - يبحث عن الفهرس السابق الذي يكون أقل من الفهرس المقدم.

JudyLPrevEmpty() - يبحث عن الفهرس السابق غير الموجود، والذي يكون أقل من الفهرس المقدم.

JudyLFirstEmpty() - يبحث عن أول فهرس غير موجود، والذي يكون أكبر من أو يساوي الفهرس المقدم.

JudyLNextEmpty() - يبحث عن الفهرس التالي غير الموجود، والذي يكون أكبر من الفهرس المقدم.

JudyLLastEmpty() - يبحث عن آخر فهرس غير موجود، والذي يكون أقل من أو يساوي الفهرس المقدم.

JudyLFirstEmpty() - يبحث عن أول فهرس غير موجود، والذي يكون أقل من أو يساوي الفهرس المقدم.

JudyLP() - يستخدم لبدء تكرار للفهارس بترتيب عكسي في مصفوفة JudyL.

JudyJ() - يستخدم للبحث عن أصغر فهرس يكون أقل من أو يساوي الفهرس المقدم.

JudyJL() - يستخدم للبحث عن أكبر فهرس يكون أقل من أو يساوي الفهرس المقدم.

JudyJLL() - يستخدم للبحث عن أكبر فهرس يكون أقل من أو يساوي الفهرس المقدم.

JudyJLE() - يستخدم للبحث عن أكبر فهرس يكون أقل من أو يساوي الفهرس المقدم.

JudyJLP() - يستخدم لمواصلة تكرار الفهارس بترتيب عكسي في مصفوفة JudyL، أو لتحديد موقع "جار" لفهرس معين.

JudyJLPE() - يستخدم للبحث عن الفهرس السابق غير الموجود، والذي يكون أقل من الفهرس المقدم.

JudyJLFE() - يستخدم للبحث عن أول فهرس غير موجود، والذي يكون أكبر من أو يساوي الفهرس المقدم.

JudyJLNE() - يستخدم للبحث عن الفهرس التالي غير الموجود، والذي يكون أكبر من الفهرس المقدم.

مصفوفات JudyL متعددة الأبعاد

يعد تخزين مؤشر لمصفوفة JudyL أخرى في قيمة مصفوفة JudyL طريقة بسيطة لدعم المصفوفات الديناميكية متعددة الأبعاد. هذه المصفوفات (أو الأشجار) المبنية بمصفوفات JudyL سريعة جدًا وفعالة من حيث الذاكرة. (في الواقع، هذه هي الطريقة التي يتم بها تنفيذ JudySL و JudyHS). يمكن إجراء أي عدد من الأبعاد بهذه الطريقة. لإنهاء عدد الأبعاد (أو الشجرة)، يتم تمييز المؤشر إلى القيمة لكي لا يشير إلى مصفوفة JudyL أخرى. يتم استخدام العلامة JLAP_INVALID في البت الأكثر أهمية للمؤشر. بعد إزالة العلامة JLAP_INVALID، يتم استخدامها كمؤشر لبيانات المستخدم. يعرف الملف Judy.h JLAP_INVALID. انظر المقطع البرمجي أدناه.

ملاحظة: النسخة الجديدة من Judy.h غيرت هذه العلامة من 0x4 إلى 0x1 للسماح لـ malloc() بعدم إرجاع ذاكرة في حدود مصفوفة 8 بايت (كما في الإصدارات القديمة من valgrind).

يمكن استخدام مثال الكود التالي لتحديد ما إذا كان المؤشر يشير إلى مصفوفة JudyL أخرى:

PValue = (PWord_t) PMultiDimArray;
for (Dim = 0; ;Dim++)
    if (PValue == (PWord_t)NULL) goto IndexNotFound;
    /* التقدم إلى البعد التالي في المصفوفة */
JLG(PValue, (Pvoid_t)*PValue, Index[Dim]);
    /* التحقق مما إذا كان المؤشر إلى مخزن المستخدم المؤقت: */
if (*PValue & JLAP_INVALID) break;
UPointer = (UPointer_t) (*PValue & ~JLAP_INVALID);  // قناع وتحويل.
printf("User object pointer is 0x%lx\n", (Word_t) UPointer);
...

ملاحظة: هذا يعمل لأن malloc() يضمن إرجاع مؤشر تكون فيه البتات الأكثر أهمية == 0x0. يجب عليك إزالة JLAP_INVALID قبل استخدام المؤشر.

الأخطاء

مثال

يقرأ سلسلة من أزواج الفهرس/القيمة من الإدخال القياسي، ويخزنها في مصفوفة JudyL، ثم يطبعها بترتيب مرتب.

#include <stdio.h>
#include <Judy.h>
Word_t   Index;                     // فهرس المصفوفة
Word_t   Value;                     // قيمة عنصر المصفوفة
Word_t * PValue;                    // مؤشر إلى قيمة عنصر المصفوفة
int      Rc_int;                    // رمز الاستجابة
Pvoid_t  PJLArray = (Pvoid_t) NULL; // تهيئة مصفوفة JudyL
while (scanf("%lu %lu", &Index, &Value))
    JLI(PValue, PJLArray, Index);
    If (PValue == PJERR) goto process_malloc_failure;
    *PValue = Value;                 // تخزين القيمة الجديدة
    // تاليًا، قم بزيارة جميع الفهارس المخزنة بالترتيب التصاعدي، ثم التنازلي، وحذف كل فهرس أثناء المرور التنازلي.
Index = 0;
JLF(PValue, PJLArray, Index);
while (PValue != NULL)
    printf("%lu %lu\n", Index, *PValue));
JLN(PValue, PJLArray, Index);
Index = -1;
JLL(PValue, PJLArray, Index);
while (PValue != NULL)
    printf("%lu %lu\n", Index, *PValue));
JLD(Rc_int, PJLArray, Index);
if (Rc_int == JERR) goto process_malloc_failure;
JLP(PValue, PJLArray, Index);
    // ...
