JLFA(JLFA)

الحزمة: judy

التراخيص: LGPL

الاسم

Judy

الملخص

Judy(3)

الوصف

Judy هي مكتبة وظائف لمعالجة مصفوفات JudyL. يمكن استخدامها لتخزين مؤشر إلى مصفوفة JudyL في قيمة مصفوفة JudyL، مما يسمح بمعالجة المصفوفات متعددة الأبعاد الديناميكية بسهولة. هذه المصفوفات (أو الأشجار) المبنية باستخدام مصفوفات JudyL سريعة جدًا وفعالة من حيث الذاكرة. (في الواقع، هذه هي الطريقة التي يتم بها تنفيذ JudySL و JudyHS). يمكن إجراء أي عدد من الأبعاد بهذه الطريقة. لإنهاء عدد الأبعاد (أو الشجرة)، يتم وضع علامة على مؤشر القيمة كعدم الإشارة إلى مصفوفة JudyL أخرى. يتم استخدام العلامة JLAP_INVALID في أقل بت أهمية من العنوان. بعد إزالة العلامة JLAP_INVALID، يتم استخدامها كمؤشر إلى بيانات المستخدم. يحدد ملف الرأس Judy.h JLAP_INVALID. انظر المقتطف البرمجي أدناه.

ملاحظة: تغيرت العلامة الحالية في ملف Judy.h من 0x4 إلى 0x1 للسماح باستدعاء malloc() لا يعيد ذاكرة على حدود محاذاة بـ 8 بايت (كما في الإصدارات القديمة من valgrind).

يمكن استخدام مثال الكود التالي لتحديد ما إذا كان المؤشر يشير إلى مصفوفة JudyL أخرى:

PValue = (PWord_t)PMultiDimArray;
for (Dim = 0; ;Dim++)
{
   if (PValue == (PWord_t)NULL) goto IndexNotFound;
   /* الانتقال إلى البعد التالي للمصفوفة */
JLG(PValue, (Pvoid_t)*PValue, Index[Dim]);
   /* التحقق مما إذا كان المؤشر يشير إلى مخزن مؤقت للمستخدم: */
if (*PValue & JLAP_INVALID)) break;
UPointer = (UPointer_t) (*PValue & ~JLAP_INVALID);  // قناع وتحويل.
printf("مؤشر كائن المستخدم هو 0x%lx\n", (Word_t) UPointer);
...

ملاحظة: يعمل هذا لأن malloc() يضمن إعادة مؤشر يحتوي على أقل بتات أهمية == 0x0. يجب عليك إزالة JLAP_INVALID قبل استخدام المؤشر.

مثال

قراءة سلسلة من أزواج الفهرس/القيمة من الإدخال القياسي، وتخزينها في مصفوفة JudyL، ثم طباعتها بترتيب فرز.

#include <stdio.h>
#include <Judy.h>
Word_t   Index;                     // فهرس المصفوفة
Word_t   Value;                     // قيمة عنصر المصفوفة
Word_t * PValue;                    // مؤشر إلى قيمة عنصر المصفوفة
int      Rc_int;                    // رمز العودة
Pvoid_t  PJLArray = (Pvoid_t) NULL; // تهيئة مصفوفة JudyL
while (scanf("%lu %lu", &Index, &Value))
{
JLI(PValue, PJLArray, Index);
If (PValue == PJERR) goto process_malloc_failure;
*PValue = Value;                 // تخزين قيمة جديدة
}
Index = 0;
JLF(PValue, PJLArray, Index);
while (PValue != NULL)
{
printf("%lu %lu\n", Index, *PValue));
JLN(PValue, PJLArray, Index);
}
Index = -1;
JLL(PValue, PJLArray, Index);
while (PValue != NULL)
{
printf("%lu %lu\n", Index, *PValue));
JLD(Rc_int, PJLArray, Index);
If (Rc_int == JERR) goto process_malloc_failure;
JLP(PValue, PJLArray, Index);
}
