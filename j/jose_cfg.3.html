توثيق التعريفات النوعية

typedef jose_cfg_t jose_cfg_auto_t

يحدد jose_cfg_t الذي يستدعي jose_cfg_decref() عند نهاية النطاق. على سبيل المثال:

void foo() {
    jose_cfg_auto_t *cfg = jose_cfg();
    // jose_cfg_decref() يتم استدعاؤه ضمنيًا
}

توثيق الوظائف

jose_cfg_t* jose_cfg (void)

ينشئ مثيل تكوين جديد.

الاسترجاع:

مثيل تكوين مخصص حديثًا.

jose_cfg_t* jose_cfg_incref (jose_cfg_t * cfg)

يزيد عدد مرات الإشارة لمثيل التكوين. تنجح هذه الوظيفة دائمًا.

المعلمات:

cfg سياق التكوين.

الاسترجاع:

قيمة cfg (للراحة).

void jose_cfg_decref (jose_cfg_t * cfg)

يقلل عدد مرات الإشارة لمثيل التكوين. عندما يصل عدد مرات الإشارة إلى الصفر، يتم تحرير مثيل التكوين.

المعلمات:

cfg سياق التكوين.

void jose_cfg_set_err_func (jose_cfg_t * cfg, jose_cfg_err_t * err, void * misc)

يضبط دالة معالج الأخطاء لمثيل التكوين هذا. سيتم تمرير قيمة misc إلى دالة معالج الأخطاء.

يمكنك تمرير NULL إلى err للعودة إلى معالج الأخطاء الافتراضي.

المعلمات:

cfg سياق التكوين.
err دالة معالج الأخطاء التي ترغب في تمكينها.
misc البيانات المتنوعة التي ترغب في تمريرها إلى معالج الأخطاء.

void* jose_cfg_get_err_misc (jose_cfg_t * cfg)

يحصل على البيانات المتنوعة المرتبطة بمعالج الأخطاء الحالي.

المعلمات:

cfg سياق التكوين.

الاسترجاع:

البيانات المتنوعة المرتبطة بمعالج الأخطاء.

void jose_cfg_err (jose_cfg_t * cfg, uint64_t err, const char * fmt, ...)

تقديم خطأ. سيتم استدعاء معالج الأخطاء مع الخطأ المقدم.

المعلمات:

cfg سياق التكوين (اختياري).
err الرقم المقابل لنوع الخطأ هذا.
fmt سلسلة تنسيق على غرار printf().
... وسائط التنسيق على غرار printf().
