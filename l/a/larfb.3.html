روتين فرعي clarfb (side, trans, direct, storev, m, n, k, complex, dimension( ldv, * ) v, ldv, t, ldt, c, ldc, work, ldwork)

CLARFB يطبق عاكس كتلة أو المترافق المترافق له على مصفوفة مستطيلة عامة.

الغرض:

!>
!> CLARFB يطبق عاكس كتلة معقد H أو مترافقه H**H على مصفوفة C مستطيلة معقدة M-by-N،
!> من اليسار أو اليمين.
!>

المعاملات

SIDE 
!>          SIDE هو CHARACTER*1
!>          = 'L': تطبيق H أو H**H من اليسار
!>          = 'R': تطبيق H أو H**H من اليمين
!>

TRANS

!>          TRANS هو CHARACTER*1
!>          = 'N': تطبيق H (بدون تدوير)
!>          = 'C': تطبيق H**H (التدوير المترافق)
!>

DIRECT

!>          DIRECT هو CHARACTER*1
!>          يشير إلى كيفية تكوين H من ناتج عواكس أولية
!>          = 'F': H = H(1) H(2) . . . H(k) (أمامي)
!>          = 'B': H = H(k) . . . H(2) H(1) (خلفي)
!>

STOREV

!>          STOREV هو CHARACTER*1
!>          يشير إلى كيفية تخزين المتجهات التي تعرف العواكس الأولية:
!>          = 'C': عموديًا
!>          = 'R': أفقيًا
!>

M

!>          M هو INTEGER
!>          عدد صفوف المصفوفة C.
!>

N

!>          N هو INTEGER
!>          عدد أعمدة المصفوفة C.
!>

K

!>          K هو INTEGER
!>          مرتبة المصفوفة T (= عدد العواكس الأولية التي يشكل ناتجها العاكس الكتلة).
!>          إذا كان SIDE = 'L'، فإن M >= K >= 0؛
!>          إذا كان SIDE = 'R'، فإن N >= K >= 0.
!>

V

!>          V هو مصفوفة COMPLEX، الأبعاد
!>                                (LDV,K) إذا كان STOREV = 'C'
!>                                (LDV,M) إذا كان STOREV = 'R' و SIDE = 'L'
!>                                (LDV,N) إذا كان STOREV = 'R' و SIDE = 'R'
!>          المصفوفة V. انظر تفاصيل إضافية.
!>

LDV

!>          LDV هو INTEGER
!>          البعد الرائد للمصفوفة V.
!>          إذا كان STOREV = 'C' و SIDE = 'L'، فإن LDV >= max(1,M)؛
!>          إذا كان STOREV = 'C' و SIDE = 'R'، فإن LDV >= max(1,N)؛
!>          إذا كان STOREV = 'R'، فإن LDV >= K.
!>

T

!>          T هي مصفوفة COMPLEX، الأبعاد (LDT,K)
!>          المصفوفة المثلثية K-by-K T في تمثيل العاكس الكتلة.
!>

LDT

!>          LDT هو INTEGER
!>          البعد الرائد للمصفوفة T. LDT >= K.
!>

C

!>          C هي مصفوفة COMPLEX، الأبعاد (LDC,N)
!>          عند الإدخال، المصفوفة C بحجم M-by-N.
!>          عند الخروج، يتم استبدال C بـ H*C أو H**H*C أو C*H أو C*H**H.
!>

LDC

!>          LDC هو INTEGER
!>          البعد الرائد للمصفوفة C. LDC >= max(1,M).
!>

WORK

!>          WORK هي مصفوفة COMPLEX، الأبعاد (LDWORK,K)
!>

LDWORK

!>          LDWORK هو INTEGER
!>          البعد الرائد للمصفوفة WORK.
!>          إذا كان SIDE = 'L'، فإن LDWORK >= max(1,N)؛
!>          إذا كان SIDE = 'R'، فإن LDWORK >= max(1,M).
!>

!>  k = 3. الجزء المثلثي من V (بما في ذلك قطره) غير مشار إليه.
!>
!>  DIRECT = 'F' و STOREV = 'C':         DIRECT = 'F' و STOREV = 'R':
!>
!>               V = (  1       )                 V = (  1 v1 v1 v1 v1 )
!>                   ( v1  1    )                     (     1 v2 v2 v2 )
!>                   ( v1 v2  1 )                     (        1 v3 v3 )

!>                   ( v1 v2 v3 )
!>
!>  DIRECT = 'B' و STOREV = 'C':         DIRECT = 'B' و STOREV = 'R':
!>
!>               V = ( v1 v2 v3 )                 V = ( v1 v1  1       )
!>                   ( v1 v2 v3 )                     ( v2 v2 v2  1    )
!>                   (  1 v2 v3 )                     ( v3 v3 v3 v3  1 )
!>                   (     1 v3 )
!>                   (        1 )
!>
