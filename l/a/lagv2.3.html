الوصف التفصيلي

توثيق الوظيفة

procedure dlagv2 (double precision, dimension( lda, * ) a, integer lda, double precision, dimension( ldb, * ) b, integer ldb, double precision, dimension( 2 ) alphar, double precision, dimension( 2 ) alphai, double precision, dimension( 2 ) beta, double precision csl, double precision snl, double precision csr, double precision snr)

DLAGV2 تحسب عامل شور المعمم لمصفوفة قلم حقيقية 2x2 (A,B) حيث B هي مصفوفة مثلثة عليا.

الغرض:

!>
!> DLAGV2 تحسب عامل شور المعمم لمصفوفة قلم حقيقية 2x2
!> (A,B) حيث B هي مصفوفة مثلثة عليا. هذه الروتين
!> تحسب مصفوفات متعامدة (دوران) معطاة بواسطة CSL, SNL و CSR,
!> SNR بحيث
!>
!> 1) إذا كانت المصفوفة (A,B) تمتلك قيمتين ذاتيين حقيقيتين (تشمل أنواع 0/0 أو 1/0
!>    )، فإن
!>
!>    [ a11 a12 ] := [  CSL  SNL ] [ a11 a12 ] [  CSR -SNR ]
!>    [  0  a22 ]    [ -SNL  CSL ] [ a21 a22 ] [  SNR  CSR ]
!>
!>    [ b11 b12 ] := [  CSL  SNL ] [ b11 b12 ] [  CSR -SNR ]
!>    [  0  b22 ]    [ -SNL  CSL ] [  0  b22 ] [  SNR  CSR ],
!>
!> 2) إذا كانت المصفوفة (A,B) تمتلك زوجًا من القيم الذاتية المترافقة المعقدة،
!>    فإن
!>
!>    [ a11 a12 ] := [  CSL  SNL ] [ a11 a12 ] [  CSR -SNR ]
!>    [ a21 a22 ]    [ -SNL  CSL ] [ a21 a22 ] [  SNR  CSR ]
!>
!>    [ b11  0  ] := [  CSL  SNL ] [ b11 b12 ] [  CSR -SNR ]
!>    [  0  b22 ]    [ -SNL  CSL ] [  0  b22 ] [  SNR  CSR ]
!>
!>    حيث b11 >= b22 > 0.
!>
!>

المعلمات

A 
!>          A هي مصفوفة DOUBLE PRECISION، الأبعاد (LDA, 2)
!>          عند الدخول، المصفوفة 2 × 2 A.
!>          عند الخروج، يتم الكتابة فوق A بواسطة "جزء A" من شكل شور المعمم.
!>

LDA

!>          LDA هو عدد صحيح (INTEGER)
!>          البعد الرئيسي للمصفوفة A. LDA >= 2.
!>

B

!>          B هي مصفوفة DOUBLE PRECISION، الأبعاد (LDB, 2)
!>          عند الدخول، المصفوفة المثلثة العليا 2 × 2 B.
!>          عند الخروج، يتم الكتابة فوق B بواسطة "جزء B" من شكل شور المعمم.
!>

LDB

!>          LDB هو عدد صحيح (INTEGER)
!>          البعد الرئيسي للمصفوفة B. LDB >= 2.
!>

ALPHAR

!>          ALPHAR هي مصفوفة DOUBLE PRECISION، الأبعاد (2)
!>

ALPHAI

!>          ALPHAI هي مصفوفة DOUBLE PRECISION، الأبعاد (2)
!>

BETA

!>          BETA هي مصفوفة DOUBLE PRECISION، الأبعاد (2)
!>          (ALPHAR(k)+i*ALPHAI(k))/BETA(k) هي القيم الذاتية لـ
!>          القلم (A,B)، k=1,2، i = الجذر التربيعي لـ (-1). لاحظ أن BETA(k) قد
!>          يكون صفرًا.
!>

CSL

!>          CSL هو DOUBLE PRECISION
!>          جيب تمام مصفوفة الدوران اليسرى.
!>

SNL

!>          SNL هو DOUBLE PRECISION
!>          جيب مصفوفة الدوران اليسرى.
!>

CSR

!>          CSR هو DOUBLE PRECISION
!>          جيب تمام مصفوفة الدوران اليمنى.
!>

SNR

!>          SNR هو DOUBLE PRECISION
!>          جيب مصفوفة الدوران اليمنى.
!>
