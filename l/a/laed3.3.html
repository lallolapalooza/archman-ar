توثيق الدوال

subroutine dlaed3 (integer k, integer n, integer n1, double precision, dimension( * ) d, double precision, dimension( ldq, * ) q, integer ldq, double precision rho, double precision, dimension( * ) dlambda, double precision, dimension( * ) q2, integer, dimension( * ) indx, integer, dimension( * ) ctot, double precision, dimension( * ) w, double precision, dimension( * ) s, integer info)

DLAED3 تستخدم بواسطة DSTEDC. تجد جذور المعادلة العلمانية وتحدث المتجهات الذاتية. تستخدم عندما تكون المصفوفة الأصلية ثلاثية الأقطار.

الغرض:

!>
!> DLAED3 تجد جذور المعادلة العلمانية، كما يتم تعريفها بواسطة القيم في D، W، و RHO، بين 1 و K.
!> تقوم بالاستدعاءات المناسبة لـ DLAED4 ثم تحدث المتجهات الذاتية بضرب مصفوفة المتجهات الذاتية للأنظمة الذاتية المزدوجة
!> التي يتم دمجها بواسطة مصفوفة المتجهات الذاتية للنظام K-by-K الذي يتم حله هنا.
!>
!>

المعاملات

K 
!>          K هو عدد صحيح (INTEGER)
!>          عدد الحدود في الدالة الكسرية التي سيتم حلها بواسطة DLAED4. K >= 0.
!>

N

!>          N هو عدد صحيح (INTEGER)
!>          عدد الصفوف والأعمدة في مصفوفة Q.
!>          N >= K (قد ينتج عن التقليص N>K).
!>

N1

!>          N1 هو عدد صحيح (INTEGER)
!>          موقع آخر قيمة ذاتية في المصفوفة الفرعية الرائدة.
!>          min(1,N) <= N1 <= N/2.
!>

D

!>          D هي مصفوفة DOUBLE PRECISION، البعد (N)
!>          تحتوي D(I) على القيم الذاتية المحدثة لـ
!>          1 <= I <= K.
!>

Q

!>          Q هي مصفوفة DOUBLE PRECISION، البعد (LDQ,N)
!>          في البداية، تستخدم الأعمدة الأولى كمساحة عمل.
!>          عند الخرج، تحتوي الأعمدة من 1 إلى K على
!>          المتجهات الذاتية المحدثة.
!>

LDQ

!>          LDQ هو عدد صحيح (INTEGER)
!>          البعد الرائد للمصفوفة Q. LDQ >= max(1,N).
!>

RHO

!>          RHO هو DOUBLE PRECISION
!>          قيمة المعامل في معادلة التحديث ذات الرتبة الأولى.
!>          RHO >= 0 مطلوب.
!>

DLAMBDA

!>          DLAMBDA هي مصفوفة DOUBLE PRECISION، البعد (K)
!>          تحتوي العناصر K الأولى من هذه المصفوفة على الجذور القديمة
!>          لمشكلة التحديث المخفضة. هذه هي أقطاب
!>          المعادلة العلمانية.
!>

Q2

!>          Q2 هي مصفوفة DOUBLE PRECISION، البعد (LDQ2*N)
!>          تحتوي الأعمدة K الأولى من هذه المصفوفة على المتجهات الذاتية غير المخفضة
!>          للمشكلة المنقسمة.
!>

INDX

!>          INDX هي مصفوفة عدد صحيح (INTEGER)، البعد (N)
!>          التبديل المستخدم لترتيب أعمدة مصفوفة Q المخفضة
!>          إلى ثلاث مجموعات (انظر DLAED2).
!>          يجب تبديل صفوف المتجهات الذاتية التي تم العثور عليها بواسطة DLAED4
!>          بالمثل قبل أن تتم عملية ضرب المصفوفات.
!>

CTOT

!>          CTOT هي مصفوفة عدد صحيح (INTEGER)، البعد (4)
!>          عدد إجمالي لأنواع الأعمدة المختلفة
!>          في Q، كما هو موصوف في INDX. النوع الرابع من الأعمدة هو أي
!>          عمود تم تخفيضه.
!>

W

!>          W هي مصفوفة DOUBLE PRECISION، البعد (K)
!>          تحتوي العناصر K الأولى من هذه المصفوفة على مكونات
!>          متجه التحديث المعدل بالخفض. يتم تدميره عند
!>          الخرج.
!>

S

!>          S هي مصفوفة DOUBLE PRECISION، البعد (N1 + 1)*K
!>          ستحتوي على المتجهات الذاتية للمصفوفة المصلحة والتي
!>          سيتم ضربها في المتجهات الذاتية المتراكمة سابقًا
!>          لتحديث النظام.
!>

INFO

!>          INFO هو عدد صحيح (INTEGER)
!>          = 0: الخروج بنجاح.
!>          < 0: إذا كان INFO = -i، فإن المعامل i كان له قيمة غير قانونية.
!>          > 0: إذا كان INFO = 1، لم تتقارب قيمة ذاتية
!>
