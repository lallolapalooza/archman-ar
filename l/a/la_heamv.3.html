subroutine cla_heamv (integer uplo, integer n, real alpha, complex, dimension( lda, * ) a, integer lda, complex, dimension( * ) x, integer incx, real beta, real, dimension( * ) y, integer incy)

CLA_HEAMV تحسب حاصل ضرب مصفوفة-متجه باستخدام مصفوفة مترافقة غير محددة لحساب حدود الخطأ.

الغرض:

!>
!> CLA_SYAMV  تقوم بعملية المصفوفة-المتجه
!>
!>         y := alpha*abs(A)*abs(x) + beta*abs(y),
!>
!> حيث alpha و beta عددان، x و y متجهان، و A مصفوفة متماثلة بحجم n × n.
!>
!> تستخدم هذه الوظيفة بشكل أساسي في حساب حدود الخطأ.
!> للحماية من نقص القيم (underflow) أثناء التقييم، يتم تعديل مكونات
!> المتجه الناتج بعيدًا عن الصفر بمقدار (N+1)
!> مضروبة في الحد الأدنى للقيم (underflow threshold). لمنع الأخطاء الكبيرة
!> بشكل غير ضروري للهياكل الكتلية المضمنة في المصفوفات العامة،
!> لا يتم تعديل المكونات الصفرية. تعتبر القيمة الصفرية
!> إذا كانت جميع عمليات الضرب المتضمنة
!> في حساب هذه القيمة تحتوي على عامل ضرب صفري واحد على الأقل.
!>

المعاملات

UPLO 
!>          UPLO هو عدد صحيح (INTEGER).
!>           عند الإدخال، يحدد UPLO ما إذا كان الجزء العلوي أو السفلي
!>           من المصفوفة A سيتم الرجوع إليه كما يلي:
!>
!>              UPLO = BLAS_UPPER   يتم الرجوع فقط إلى الجزء العلوي من A.
!>
!>              UPLO = BLAS_LOWER   يتم الرجوع فقط إلى الجزء السفلي من A.
!>
!>           لا يتغير عند الخروج.
!>

N

!>          N هو عدد صحيح (INTEGER).
!>           عند الإدخال، يحدد N عدد أعمدة المصفوفة A.
!>           يجب أن يكون N على الأقل صفرًا.
!>           لا يتغير عند الخروج.
!>

ALPHA

!>          ALPHA هو قيمة حقيقية (REAL).
!>           عند الإدخال، يحدد ALPHA العدد ألفا.
!>           لا يتغير عند الخروج.
!>

A

!>          A هي مصفوفة مركبة (COMPLEX) بأبعاد ( LDA, n ).
!>           قبل الإدخال، يجب أن يحتوي الجزء الرئيسي بحجم m × n من المصفوفة A على
!>           معاملات المصفوفة.
!>           لا تتغير عند الخروج.
!>

LDA

!>          LDA هو عدد صحيح (INTEGER).
!>           عند الإدخال، يحدد LDA البعد الأول للمصفوفة A كما تم تعريفه
!>           في البرنامج (الفرعي) المستدعي. يجب أن يكون LDA على الأقل
!>           max( 1, n ).
!>           لا يتغير عند الخروج.
!>

X

!>          X هي مصفوفة مركبة (COMPLEX) بأبعاد
!>           ( 1 + ( n - 1 )*abs( INCX ) )
!>           قبل الإدخال، يجب أن تحتوي المصفوفة X المعدلة على
!>           المتجه x.
!>           لا يتغير عند الخروج.
!>

INCX

!>          INCX هو عدد صحيح (INTEGER).
!>           عند الإدخال، يحدد INCX الزيادة لعناصر X.
!>           يجب ألا يكون INCX صفرًا.
!>           لا يتغير عند الخروج.
!>

BETA

!>          BETA هي قيمة حقيقية (REAL).
!>           عند الإدخال، يحدد BETA العدد بيتا. عندما يتم توفير BETA
!>           كصفر، فلا حاجة لتعيين Y عند الإدخال.
!>           لا يتغير عند الخروج.
!>

Y

!>          Y هي مصفوفة حقيقية (REAL) بأبعاد
!>           ( 1 + ( n - 1 )*abs( INCY ) )
!>           قبل الإدخال مع BETA غير صفر، يجب أن تحتوي المصفوفة Y المعدلة
!>           على المتجه y. عند الخروج، يتم الكتابة فوق Y بالمتجه
!>           المحدث y.
!>

INCY

!>          INCY هو عدد صحيح (INTEGER).
!>           عند الإدخال، يحدد INCY الزيادة لعناصر Y.
!>           يجب ألا يكون INCY صفرًا.
!>           لا يتغير عند الخروج.
!>

ZLA_HEAMV تحسب حاصل ضرب مصفوفة-متجه باستخدام مصفوفة مترافقة غير محددة لحساب حدود الخطأ.

الغرض:

!>
!> ZLA_SYAMV  تقوم بعملية المصفوفة-المتجه
!>
!>         y := alpha*abs(A)*abs(x) + beta*abs(y),
!>
!> حيث alpha و beta عددان، x و y متجهان، و A مصفوفة متماثلة بحجم n × n.
!>
!> تستخدم هذه الوظيفة بشكل أساسي في حساب حدود الخطأ.
!> للحماية من نقص القيم (underflow) أثناء التقييم، يتم تعديل مكونات
!> المتجه الناتج بعيدًا عن الصفر بمقدار (N+1)
!> مضروبة في الحد الأدنى للقيم (underflow threshold). لمنع الأخطاء الكبيرة
!> بشكل غير ضروري للهياكل الكتلية المضمنة في المصفوفات العامة،
!> لا يتم تعديل المكونات الصفرية. تعتبر القيمة الصفرية
!> إذا كانت جميع عمليات الضرب المتضمنة
!> في حساب هذه القيمة تحتوي على عامل ضرب صفري واحد على الأقل.
!>
