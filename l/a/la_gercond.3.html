توثيق الدالة

دالة حقيقية cla_gercond_c (trans, n, a, lda, af, ldaf, ipiv, c, capply, info, work, rwork)

CLA_GERCOND_C تحسب رقم شرط المعيار اللانهائي لـ op(A)*inv(diag(c)) للمصفوفات العامة.

الغرض:

!>
!>
!>    CLA_GERCOND_C تحسب رقم شرط المعيار اللانهائي لـ
!>    op(A) * inv(diag(C)) حيث C متجه حقيقي.
!>

المعاملات

TRANS 
!>          TRANS هو CHARACTER*1
!>     يحدد شكل نظام المعادلات:
!>       = 'N':  A * X = B     (بدون تبديل)
!>       = 'T':  A**T * X = B  (تبديل)
!>       = 'C':  A**H * X = B  (تبديل مرافق = تبديل)
!>

N

!>          N هو INTEGER
!>     عدد المعادلات الخطية، أي ترتيب
!>     المصفوفة A.  N >= 0.
!>

A

!>          A هي مصفوفة COMPLEX، حجمها (LDA,N)
!>     عند الإدخال، المصفوفة N*N A
!>

LDA

!>          LDA هو INTEGER
!>     البعد الرائد للمصفوفة A.  LDA >= max(1,N).
!>

AF

!>          AF هي مصفوفة COMPLEX، حجمها (LDAF,N)
!>     العوامل L و U من تفكيك
!>     A = P*L*U كما تم حسابه بواسطة CGETRF.
!>

LDAF

!>          LDAF هو INTEGER
!>     البعد الرائد للمصفوفة AF.  LDAF >= max(1,N).
!>

IPIV

!>          IPIV هي مصفوفة INTEGER، حجمها (N)
!>     فهارس المحور من تفكيك A = P*L*U
!>     كما تم حسابه بواسطة CGETRF؛ تم تبديل الصف i من المصفوفة
!>     مع الصف IPIV(i).
!>

C

!>          C هي مصفوفة REAL، حجمها (N)
!>     المتجه C في الصيغة op(A) * inv(diag(C)).
!>

CAPPLY

!>          CAPPLY هو LOGICAL
!>     إذا كان .TRUE. فقم بالوصول إلى المتجه C في الصيغة أعلاه.
!>

INFO

!>          INFO هو INTEGER
!>       = 0:  خروج ناجح.
!>     i > 0:  المعامل i غير صالح.
!>

WORK

!>          WORK هي مصفوفة COMPLEX، حجمها (2*N).
!>     مساحة عمل.
!>

RWORK

!>          RWORK هي مصفوفة REAL، حجمها (N).
!>     مساحة عمل.
!>

دالة مزدوجة الدقة dla_gercond (trans, n, a, lda, af, ldaf, ipiv, cmode, c, info, work, iwork)

DLA_GERCOND تقدر رقم شرط Skeel لمصفوفة عامة.

الغرض:

!>
!>
!>    DLA_GERCOND تقدر رقم شرط Skeel لمصفوفة عامة A.
!>
!>    رقم شرط Skeel هو مقياس لمدى حساسية حل نظام خطي
!>    AX = B للتغيرات الصغيرة في A أو B.
!>
!>    يتم حساب رقم شرط Skeel على النحو التالي:
!>
!>    ||A||_inf * ||inv(A)||_inf
!>
!>    حيث ||.||_inf هو المعيار اللانهائي.
!>
!>    يتم تقدير رقم شرط Skeel باستخدام تقنيات تكرارية.
!>

المعاملات

TRANS 
!>          TRANS هو CHARACTER*1
!>     يحدد شكل المصفوفة A:
!>       = 'N':  A (بدون تبديل)
!>       = 'T':  A**T (تبديل)
!>       = 'C':  A**H (تبديل مرافق)
!>

N

!>          N هو INTEGER
!>     ترتيب المصفوفة A. N >= 0.
!>

A

!>          A هي مصفوفة DOUBLE PRECISION، حجمها (LDA,N)
!>     عند الإدخال، المصفوفة N*N A.
!>

LDA

!>          LDA هو INTEGER
!>     البعد الرائد للمصفوفة A. LDA >= max(1,N).
!>

AF

!>          AF هي مصفوفة DOUBLE PRECISION، حجمها (LDAF,N)
!>     العوامل L و U من تفكيك PA = LU كما تم حسابه بواسطة DGETRF.
!>

LDAF

!>          LDAF هو INTEGER
!>     البعد الرائد للمصفوفة AF. LDAF >= max(1,N).
!>

IPIV

!>          IPIV هي مصفوفة INTEGER، حجمها (N)
!>     فهارس المحور من تفكيك PA = LU كما تم حسابه بواسطة DGETRF؛
!>     تم تبديل الصف i من المصفوفة مع الصف IPIV(i).
!>

CMODE

!>          CMODE هو INTEGER
!>     نمط حساب المعيار:
!>       = 0:  المعيار اللانهائي.
!>       = 1:  المعيار 1.
!>       = 2:  المعيار 2.
!>       = -1: المعيار -1.
!>       = -2: المعيار -2.
!>

C

!>          C هي مصفوفة DOUBLE PRECISION، حجمها (N)
!>     عند الإخراج، يمثل تقدير رقم شرط Skeel.
!>

INFO

!>          INFO هو INTEGER
!>       = 0:  خروج ناجح.
!>     i > 0:  المعامل i غير صالح.
!>

WORK

!>          WORK هي مصفوفة DOUBLE PRECISION، حجمها (N).
!>     مساحة عمل.
!>

IWORK

!>          IWORK هي مصفوفة INTEGER، حجمها (N).
!>     مساحة عمل.
!>
