subroutine clals0 (integer icompq, integer nl, integer nr, integer sqre, integer nrhs, complex, dimension( ldb, * ) b, integer ldb, complex, dimension( ldbx, * ) bx, integer ldbx, integer, dimension( * ) perm, integer givptr, integer, dimension( ldgcol, * ) givcol, integer ldgcol, real, dimension( ldgnum, * ) givnum, integer ldgnum, real, dimension( ldgnum, * ) poles, real, dimension( * ) difl, real, dimension( ldgnum, * ) difr, real, dimension( * ) z, integer k, real c, real s, real, dimension( * ) rwork, integer info)

CLALS0 تطبق عوامل الضرب الخلفية في حل مشكلة المربعات الصغرى باستخدام نهج SVD للتقسيم والغزو. تستخدم بواسطة sgelsd.

الغرض:

!>
!> CLALS0 تطبق عوامل الضرب الخلفية إما لمصفوفة المتجهات المفردة اليسرى أو اليمنى
!> لمصفوفة قطرية مضافة إلى مصفوفة الجانب الأيمن B في حل مشكلة المربعات الصغرى
!> باستخدام نهج SVD للتقسيم والغزو.
!>
!> لمصفوفة المتجهات المفردة اليسرى، تشمل ثلاثة أنواع من المصفوفات المتعامدة:
!>
!> (1L) دورانات جيفنز: عدد هذه الدورانات هو GIVPTR؛ الأزواج من الأعمدة/الصفوف التي
!>      تم تطبيقها عليها مخزنة في GIVCOL؛ وقيم C و S لهذه الدورانات مخزنة في GIVNUM.
!>
!> (2L) التبديل. يتم نقل الصف (NL+1) من B إلى الصف الأول، ولـ J=2:N، يتم نقل الصف
!>      PERM(J) من B إلى الصف J.
!>
!> (3L) مصفوفة المتجهات المفردة اليسرى للمصفوفة المتبقية.
!>
!> لمصفوفة المتجهات المفردة اليمنى، تشمل أربعة أنواع من المصفوفات المتعامدة:
!>
!> (1R) مصفوفة المتجهات المفردة اليمنى للمصفوفة المتبقية.
!>
!> (2R) إذا كان SQRE = 1، دورة جيفنز إضافية لتوليد الفضاء الصفري الأيمن.
!>
!> (3R) التحويل العكسي لـ (2L).
!>
!> (4R) التحويل العكسي لـ (1L).
!>

المعلمات

ICOMPQ 
!>          ICOMPQ هو عدد صحيح (INTEGER)
!>         يحدد ما إذا كانت المتجهات المفردة سيتم حسابها بالشكل المفكك:
!>         = 0: مصفوفة المتجهات المفردة اليسرى.
!>         = 1: مصفوفة المتجهات المفردة اليمنى.
!>

NL

!>          NL هو عدد صحيح (INTEGER)
!>         بعد الصفوف للكتلة العلوية. NL >= 1.
!>

NR

!>          NR هو عدد صحيح (INTEGER)
!>         بعد الصفوف للكتلة السفلية. NR >= 1.
!>

SQRE

!>          SQRE هو عدد صحيح (INTEGER)
!>         = 0: الكتلة السفلية هي مصفوفة مربعة بحجم NR-by-NR.
!>         = 1: الكتلة السفلية هي مصفوفة مستطيلة بحجم NR-by-(NR+1).
!>
!>         المصفوفة ثنائية القطر لها بعد صفوف N = NL + NR + 1،
!>         وبعد أعمدة M = N + SQRE.
!>

NRHS

!>          NRHS هو عدد صحيح (INTEGER)
!>         عدد أعمدة B و BX. يجب أن يكون NRHS على الأقل 1.
!>

B

!>          B هي مصفوفة مركبة (COMPLEX)، البعد ( LDB, NRHS )
!>         عند الإدخال، تحتوي B على الجوانب اليمنى لمشكلة المربعات الصغرى في الصفوف
!>         من 1 إلى M. عند الإخراج، تحتوي B على الحل X في الصفوف من 1 إلى N.
!>

LDB

!>          LDB هو عدد صحيح (INTEGER)
!>         البعد الرائد لـ B. يجب أن يكون LDB على الأقل
!>         max(1,MAX( M, N ) ).
!>

BX

!>          BX هي مصفوفة مركبة (COMPLEX)، البعد ( LDBX, NRHS )
!>

LDBX

!>          LDBX هو عدد صحيح (INTEGER)
!>         البعد الرائد لـ BX.
!>

PERM

!>          PERM هي مصفوفة أعداد صحيحة (INTEGER array)، البعد ( N )
!>         التبادلات (من إزالة التحيز والفرز) المطبقة
!>         على الكتلين.
!>

GIVPTR

!>          GIVPTR هو عدد صحيح (INTEGER)
!>         عدد دورانات جيفنز التي حدثت في هذه
!>         المشكلة الفرعية.
!>

GIVCOL

!>          GIVCOL هي مصفوفة أعداد صحيحة (INTEGER array)، البعد ( LDGCOL, 2 )
!>         كل زوج من الأرقام يشير إلى زوج من الصفوف/الأعمدة
!>         المشاركة في دوران جيفنز.
!>

LDGCOL

!>          LDGCOL هو عدد صحيح (INTEGER)
!>         البعد الرائد لـ GIVCOL، يجب أن يكون على الأقل N.
!>

GIVNUM

!>          GIVNUM هي مصفوفة حقيقية (REAL array)، البعد ( LDGNUM, 2 )
!>         كل رقم يشير إلى قيمة C أو S المستخدمة في
!>         دوران جيفنز المقابل.
!>

LDGNUM

!>          LDGNUM هو عدد صحيح (INTEGER)
!>         البعد الرائد لمصفوفات DIFR و POLES و
!>         GIVNUM، يجب أن يكون على الأقل K.
!>

POLES

!>          POLES هي مصفوفة حقيقية (REAL array)، البعد ( LDGNUM, 2 )
!>         عند الإدخال، تحتوي POLES(1:K, 1) على القيم المفردة
!>         المحدثة الناتجة عن حل المعادلة العلمانية، و
!>         POLES(1:K, 2) هي مصفوفة تحتوي على الأقطاب في
!>         المعادلة العلمانية.
!>

DIFL

!>          DIFL هي مصفوفة حقيقية (REAL array)، البعد ( K ).
!>         عند الإدخال، DIFL(I) هي المسافة بين القيمة المفردة
!>         المحدثة (غير المزاحة) i-th والقيمة المفردة القديمة
!>         (غير المزاحة) i-th.
!>

DIFR

!>          DIFR هي مصفوفة حقيقية (REAL array)، البعد ( LDGNUM, 2 ).
!>         عند الإدخال، تحتوي DIFR(I, 1) على المسافات بين القيمة المفردة
!>         المحدثة (غير المزاحة) i-th والقيمة المفردة القديمة
!>         (غير المزاحة) i+1-th. و DIFR(I, 2) هي
!>         عامل التطبيع للمتجه المفرد الأيمن i-th.
!>

Z

!>          Z هي مصفوفة حقيقية (REAL array)، البعد ( K )
!>         تحتوي على مكونات متجه الصف المحدث المعدل بالإزالة.
!>

K

!>          K هو عدد صحيح (INTEGER)
!>         يحتوي على بعد المصفوفة غير المزاحة،
!>         هذا هو رتبة المعادلة العلمانية ذات الصلة. 1 <= K <=N.
!>

C

!>          C هو عدد حقيقي (REAL)
!>         تحتوي C على قيم غير هامة إذا كان SQRE = 0 وقيمة C لدورة جيفنز
!>         المتعلقة بالفضاء الصفري الأيمن إذا كان SQRE = 1.
!>

S

!>          S هو عدد حقيقي (REAL)
!>         تحتوي S على قيم غير هامة إذا كان S

subroutine clals0 (integer icompq, integer nl, integer nr, integer sqre, integer nrhs, complex, dimension( ldb, * ) b, integer ldb, complex, dimension( ldbx, * ) bx, integer ldbx, integer, dimension( * ) perm, integer givptr, integer, dimension( ldgcol, * ) givcol, integer ldgcol, complex, dimension( ldgnum, * ) givnum, integer ldgnum, complex, dimension( ldgnum, * ) poles, complex, dimension( * ) difl, complex, dimension( ldgnum, * ) difr, complex, dimension( * ) z, integer k, complex c, complex s, complex, dimension( * ) work, integer info)

CLALS0 applies back multiplying factors in solving the least squares problem using divide and conquer SVD approach. Used by sgelsd.

Purpose:

!>
!> CLALS0 applies back the multiplying factors of either the left or the
!> right singular vector matrix of a diagonal matrix appended by a row
!> to the right hand side matrix B in solving the least squares problem
!> using the divide-and-conquer SVD approach.
!>
!> For the left singular vector matrix, three types of orthogonal
!> matrices are involved:
!>
!> (1L) Givens rotations: the number of such rotations is GIVPTR; the
!>      pairs of columns/rows they were applied to are stored in GIVCOL;
!>      and the C- and S-values of these rotations are stored in GIVNUM.
!>
!> (2L) Permutation. The (NL+1)-st row of B is to be moved to the first
!>      row, and for J=2:N, PERM(J)-th row of B is to be moved to the
!>      J-th row.
!>
!> (3L) The left singular vector matrix of the remaining matrix.
!>
!> For the right singular vector matrix, four types of orthogonal
!> matrices are involved:
!>
!> (1R) The right singular vector matrix of the remaining matrix.
!>
!> (2R) If SQRE = 1, one extra Givens rotation to generate the right
!>      null space.
!>
!> (3R) The inverse transformation of (2L).
!>
!> (4R) The inverse transformation of (1L).
!>

Parameters

ICOMPQ 
!>          ICOMPQ is INTEGER
!>         Specifies whether singular vectors are to be computed in
!>         factored form:
!>         = 0: Left singular vector matrix.
!>         = 1: Right singular vector matrix.
!>

NL

!>          NL is INTEGER
!>         The row dimension of the upper block. NL >= 1.
!>

NR

!>          NR is INTEGER
!>         The row dimension of the lower block. NR >= 1.
!>

SQRE

!>          SQRE is INTEGER
!>         = 0: the lower block is an NR-by-NR square matrix.
!>         = 1: the lower block is an NR-by-(NR+1) rectangular matrix.
!>
!>         The bidiagonal matrix has row dimension N = NL + NR + 1,
!>         and column dimension M = N + SQRE.
!>

NRHS

!>          NRHS is INTEGER
!>         The number of columns of B and BX. NRHS must be at least 1.
!>

B

!>          B is COMPLEX array, dimension ( LDB, NRHS )
!>         On input, B contains the right hand sides of the least
!>         squares problem in rows 1 through M. On output, B contains
!>         the solution X in rows 1 through N.
!>

LDB

!>          LDB is INTEGER
!>         The leading dimension of B. LDB must be at least
!>         max(1,MAX( M, N ) ).
!>

BX

!>          BX is COMPLEX array, dimension ( LDBX, NRHS )
!>

LDBX

!>          LDBX is INTEGER
!>         The leading dimension of BX.
!>

PERM

!>          PERM is INTEGER array, dimension ( N )
!>         The permutations (from deflation and sorting) applied
!>         to the two blocks.
!>

GIVPTR

!>          GIVPTR is INTEGER
!>         The number of Givens rotations which took place in this
!>         subproblem.
!>

GIVCOL

!>          GIVCOL is INTEGER array, dimension ( LDGCOL, 2 )
!>         Each pair of numbers indicates a pair of rows/columns
!>         involved in a Givens rotation.
!>

LDGCOL

!>          LDGCOL is INTEGER
!>         The leading dimension of GIVCOL, must be at least N.
!>

GIVNUM

!>          GIVNUM is COMPLEX array, dimension ( LDGNUM, 2 )
!>         Each number indicates the C or S value used in the
!>         corresponding Givens rotation.
!>

LDGNUM

!>          LDGNUM is INTEGER
!>         The leading dimension of arrays DIFR, POLES and
!>         GIVNUM, must be at least K.
!>

POLES

!>          POLES is COMPLEX array, dimension ( LDGNUM, 2 )
!>         On entry, POLES(1:K, 1) contains the new singular
!>         values obtained from solving the secular equation, and
!>         POLES(1:K, 2) is an array containing the poles in the secular
!>         equation.
!>

DIFL

!>          DIFL is COMPLEX array, dimension ( K ).
!>         On entry, DIFL(I) is the distance between I-th updated
!>         (undeflated) singular value and the I-th (undeflated) old
!>         singular value.
!>

DIFR

!>          DIFR is COMPLEX array, dimension ( LDGNUM, 2 ).
!>         On entry, DIFR(I, 1) contains the distances between I-th
!>         updated (undeflated) singular value and the I+1-th
!>         (undeflated) old singular value. And DIFR(I, 2) is the
!>         normalizing factor for the I-th right singular vector.
!>

Z

!>          Z is COMPLEX array, dimension ( K )
!>         Contain the components of the deflation-adjusted updating row
!>         vector.
!>

K

!>          K is INTEGER
!>         Contains the dimension of the non-deflated matrix,
!>         This is the order of the related secular equation. 1 <= K <=N.
!>

C

!>          C is COMPLEX
!>         C contains garbage if SQRE =0 and the C-value of a Givens
!>         rotation related to the right null space if SQRE = 1.
!>

S

!>          S is COMPLEX
!>         S contains garbage if SQRE =0 and the S-value of a Givens
!>         rotation related to the right null space if SQRE = 1.
!>

WORK

!>          WORK is COMPLEX array, dimension ( K )
!>

INFO

!>          INFO is INTEGER
!>          = 0:  successful exit.
!>          < 0:  if INFO = -i, the i-th argument had an illegal value.
!>

ZLALS0 applies back multiplying factors in solving the least squares problem using divide and conquer SVD approach. Used by sgelsd.

Purpose:

!>
!> ZLALS0 applies back the multiplying factors of either the left or the
!> right singular vector matrix of a diagonal matrix appended by a row
!> to the right hand side matrix B in solving the least squares problem
!> using the divide-and-conquer SVD approach.
!>
!> For the left singular vector matrix, three types of orthogonal
!> matrices are involved:
!>
!> (1L) Givens rotations: the number of such rotations is GIVPTR; the
!>      pairs of columns/rows they were applied to are stored in GIVCOL;
!>      and the C- and S-values of these rotations are stored in GIVNUM.
!>

!>      row, and for J=2:N, PERM(J)-th row of B is to be moved to the

!> (1R) The right singular vector matrix of the remaining matrix.
!>
!> (2R) If SQRE = 1, one extra Givens rotation to generate the right

NAME

zlals0

a subroutine for the recursive computation of the singular value decomposition of a bidiagonal matrix

SYNOPSIS

 SUBROUTINE ZLALS0(
  LCASE,      MB,             PNORM,          X,              LX,
  SIGMA,      SIGMAX,         SSEND,          WA,             DWA,

  CSL,            CSR,            SOL,            SOR,            LAMBDA,

 INTEGER     MB, LX, LDUT, LDUTT, LDVT, LDVT1, INFO
 REAL*8      PNORM, SIGMAX, SSEND
 COMPLEX*16  X( LX, * ), WA( * ), U( LDUT, * ), UT( LDUTT, * ), VT( LDVT, * ),
             RT( LDVT1, * ), SUMQ( * ), Q( * ), S( * ), E( * ), U1( * ),
             U2( * ), SIGMA( * ), CSL( * ), CSR( * ), SOL( * ), SOR( * ),
             LAMBDA( * ), W( * ), WORK( * )

Detailed Description

ZLALS0 computes the singular values (and the related right singular vectors) of a bidiagonal matrix. This subroutine is called by the top level driver ZLASDT.

Note that the routine is for the case when the matrix is the result of the QR decomposition of a matrix C, where C has been reduced to upper triangular form.

The singular values are computed in an increasing order.
