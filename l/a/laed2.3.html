روتين فرعي dlaed2 (k, n, n1, d, q, ldq, indxq, rho, z, dlambda, w, q2, indx, indxc, indxp, coltyp, info)
DLAED2 يُستخدم بواسطة DSTEDC. يدمج القيم الذاتية وينتفخ المعادلة العلمانية. يُستخدم عندما تكون المصفوفة الأصلية ثلاثية الأقطار. روتين فرعي slaed2 (k, n, n1, d, q, ldq, indxq, rho, z, dlambda, w, q2, indx, indxc, indxp, coltyp, info)
SLAED2 يُستخدم بواسطة SSTEDC. يدمج القيم الذاتية وينتفخ المعادلة العلمانية. يُستخدم عندما تكون المصفوفة الأصلية ثلاثية الأقطار.

روتين فرعي dlaed2 (integer k, integer n, integer n1, double precision, dimension( * ) d, double precision, dimension( ldq, * ) q, integer ldq, integer, dimension( * ) indxq, double precision rho, double precision, dimension( * ) z, double precision, dimension( * ) dlambda, double precision, dimension( * ) w, double precision, dimension( * ) q2, integer, dimension( * ) indx, integer, dimension( * ) indxc, integer, dimension( * ) indxp, integer, dimension( * ) coltyp, integer info)

DLAED2 يُستخدم بواسطة DSTEDC. يدمج القيم الذاتية وينتفخ المعادلة العلمانية. يُستخدم عندما تكون المصفوفة الأصلية ثلاثية الأقطار.

الغرض:

!>
!> يدمج DLAED2 مجموعتي القيم الذاتية معًا في مجموعة واحدة
!> مرتبة. ثم يحاول تقليص حجم المشكلة.
!> هناك طريقتان يمكن أن يحدث بهما الانتفاخ: عندما تكون قيمتان ذاتيتان
!> أو أكثر متقاربتين أو إذا كان هناك عنصر صغير جدًا في
!> متجه Z. لكل حالة من هذه الحالات يتم تقليل رتبة مشكلة المعادلة
!> العلمانية المرتبطة بمقدار واحد.
!>

المعاملات

K 
!>          K هو INTEGER
!>         عدد القيم الذاتية غير المنتفخة، ورتبة المعادلة
!>         العلمانية المرتبطة. 0 <= K <=N.
!>

N

!>          N هو INTEGER
!>         بُعد المصفوفة المتماثلة ثلاثية الأقطار.  N >= 0.
!>

N1

!>          N1 هو INTEGER
!>         موقع آخر قيمة ذاتية في المصفوفة الفرعية البادئة.
!>         min(1,N) <= N1 <= N/2.
!>

D

!>          D هو مصفوفة DOUBLE PRECISION، البُعد (N)
!>         عند الدخول، يحتوي D على القيم الذاتية للمصفوفتين الفرعيتين
!>         المراد دمجهما.
!>         عند الخروج، يحتوي D على القيم الذاتية المحدثة اللاحقة (N-K)
!>         (تلك التي تم انتفاخها) مرتبة بترتيب تصاعدي.
!>

Q

!>          Q هو مصفوفة DOUBLE PRECISION، البُعد (LDQ, N)
!>         عند الدخول، يحتوي Q على المتجهات الذاتية لمصفوفتين فرعيتين في
!>         الكتلتين المربعتين بزوايا عند (1,1)، (N1,N1)
!>         و (N1+1, N1+1)، (N,N).
!>         عند الخروج، يحتوي Q على المتجهات الذاتية المحدثة اللاحقة (N-K)
!>         (تلك التي تم انتفاخها) في أعمدته الأخيرة N-K.
!>

LDQ

!>          LDQ هو INTEGER
!>         البُعد الرئيسي للمصفوفة Q.  LDQ >= max(1,N).
!>

INDXQ

!>          INDXQ هو مصفوفة INTEGER، البُعد (N)
!>         التبديل الذي يرتب المشكلتين الفرعيتين
!>         في D بشكل منفصل بترتيب تصاعدي. لاحظ أن العناصر في النصف
!>         الثاني من هذا التبديل يجب أن تُضاف إليها N1 أولاً
!>         إلى قيمها. يتم تدميرها عند الخروج.
!>

RHO

!>          RHO هو DOUBLE PRECISION
!>         عند الدخول، العنصر خارج القطري المرتبط بالقطع
!>         من الرتبة-1 الذي فصل في الأصل المصفوفتين الفرعيتين اللتين
!>         يتم إعادة دمجهما الآن.
!>         عند الخروج، تم تعديل RHO إلى القيمة المطلوبة بواسطة
!>         DLAED3.
!>

Z

!>          Z هو مصفوفة DOUBLE PRECISION، البُعد (N)
!>         عند الدخول، يحتوي Z على متجه التحديث (الصف الأخير
!>         من مصفوفة المتجهات الذاتية الفرعية الأولى والصف الأول من
!>         مصفوفة المتجهات الذاتية الفرعية الثانية).
!>         عند الخروج، تم تدمير محتويات Z بواسطة عملية
!>         التحديث.
!>

DLAMBDA

!>          DLAMBDA هو مصفوفة DOUBLE PRECISION، البُعد (N)
!>         نسخة من أول K قيم ذاتية والتي ستُستخدم بواسطة
!>         DLAED3 لتكوين المعادلة العلمانية.
!>

W

!>          W هو مصفوفة DOUBLE PRECISION، البُعد (N)
!>         أول k قيم من متجه z النهائي المعدل بالانتفاخ
!>         والتي ستُمرر إلى DLAED3.
!>

Q2

!>          Q2 هو مصفوفة DOUBLE PRECISION، البُعد (N1**2+(N-N1)**2)
!>         نسخة من أول K متجهات ذاتية والتي ستُستخدم بواسطة
!>         DLAED3 في ضرب مصفوفات (DGEMM) لحل المتجهات الذاتية
!>         الجديدة.
!>

INDX

!>          INDX هو مصفوفة INTEGER، البُعد (N)
!>         التبديل المستخدم لترتيب محتويات DLAMBDA بترتيب
!>         تصاعدي.
!>

INDXC

!>          INDXC هو مصفوفة INTEGER، البُعد (N)
!>         التبديل المستخدم لترتيب أعمدة مصفوفة Q المنتفخة
!>         في ثلاث مجموعات:  المجموعة الأولى تحتوي على عناصر
!>         غير صفرية فقط عند N1 وما فوقها، والثانية تحتوي على
!>         عناصر غير صفرية فقط تحت N1، والثالثة كثيفة.
!>

INDXP

!>          INDXP هو مصفوفة INTEGER، البُعد (N)
!>         التبديل المستخدم لوضع قيم D المنتفخة في نهاية
!>         المصفوفة.  INDXP(1:K) يشير إلى قيم D غير المنتفخة
!>         و INDXP(K+1:N) يشير إلى القيم الذاتية المنتفخة.
!>

COLTYP

!>          COLTYP هو مصفوفة INTEGER، البُعد (N)
!>         أثناء التنفيذ، تسمية تشير إلى أي من الأنواع
!>         التالية يكون عمود في مصفوفة Q2:
!>         1 : غير صفري في النصف العلوي فقط؛
!>         2 : كثيف؛
!>         3 : غير صفري في النصف السفلي فقط؛
!>         4 : منتفخ.
!>         عند الخروج، COLTYP(i) هو عدد الأعمدة من النوع i،
!>         لـ i=1 إلى 4 فقط.
!>

INFO

!>          INFO هو INTEGER
!>          = 0:  خروج ناجح.
!>          < 0:  إذا كان INFO = -i، فإن المعامل i له قيمة غير قانونية.
!>
