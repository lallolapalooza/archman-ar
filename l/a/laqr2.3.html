توثيق الوظيفة

subroutine claqr2 (logical wantt, logical wantz, integer n, integer ktop, integer kbot, integer nw, complex, dimension( ldh, * ) h, integer ldh, integer iloz, integer ihiz, complex, dimension( ldz, * ) z, integer ldz, integer ns, integer nd, complex, dimension( * ) sh, complex, dimension( ldv, * ) v, integer ldv, integer nh, complex, dimension( ldt, * ) t, integer ldt, integer nv, complex, dimension( ldwv, * ) wv, integer ldwv, complex, dimension( * ) work, integer lwork)

CLAQR2 تجري التحويلات المتشابهة الوحدوية لمصفوفة هسنبرغ للكشف عن القيم الذاتية المتوافقة بالكامل من مصفوفة فرعية رئيسية لاحقة (تحديد مبكر عدواني).

الغرض:

!>
!>    CLAQR2 مطابق لـ CLAQR3 باستثناء أنه يتجنب
!>    الاستدعاء العودي عن طريق استدعاء CLAHQR بدلاً من CLAQR4.
!>
!>    تحديد مبكر عدواني:
!>
!>    تقبل هذه الوحدة الفرعية مصفوفة هسنبرغ العلوية
!>    H وتجري تحويلاً متشابهاً وحدوياً
!>    مصمم للكشف عن القيم الذاتية المتوافقة بالكامل وتحديدها من
!>    مصفوفة فرعية رئيسية لاحقة. على الإخراج تم استبدال H بـ
!>    مصفوفة هسنبرغ جديدة هي اضطراب لـ
!>    تحويل متشابه وحدوي لـ H. يؤمل أن يكون الإصدار النهائي لـ H يحتوي على العديد من الإدخالات الفرعية الصفرية.
!>

المعلمات

WANTT 
!>          WANTT هو LOGICAL
!>          إذا كان .TRUE.، يتم تحديث مصفوفة هسنبرغ H بالكامل
!>          بحيث يمكن حساب عامل Schur المثلثي (بالتعاون مع الوحدة الفرعية المستدعية).
!>          إذا كان .FALSE.، يتم تحديث H بشكل كافٍ فقط للحفاظ على
!>          القيم الذاتية.
!>

WANTZ

!>          WANTZ هو LOGICAL
!>          إذا كان .TRUE.، يتم تحديث المصفوفة الوحدوية Z
!>          بحيث يمكن حساب عامل Schur الوحدوي
!>          (بالتعاون مع الوحدة الفرعية المستدعية).
!>          إذا كان .FALSE.، لا يتم الرجوع إلى Z.
!>

N

!>          N هو INTEGER
!>          رتبة المصفوفة H و (إذا كانت WANTZ .TRUE.)
!>          رتبة المصفوفة الوحدوية Z.
!>

KTOP

!>          KTOP هو INTEGER
!>          يفترض أن KTOP = 1 أو H(KTOP,KTOP-1)=0.
!>          يحدد KBOT و KTOP معًا كتلة معزولة
!>          على طول القطر الرئيسي لمصفوفة هسنبرغ.
!>

KBOT

!>          KBOT هو INTEGER
!>          يفترض بدون فحص أن KBOT = N أو H(KBOT+1,KBOT)=0.
!>          يحدد KBOT و KTOP معًا كتلة معزولة
!>          على طول القطر الرئيسي لمصفوفة هسنبرغ.
!>

NW

!>          NW هو INTEGER
!>          حجم نافذة التحديد. 1 <= NW <= (KBOT-KTOP+1).
!>

H

!>          H هي مصفوفة COMPLEX، البعد (LDH,N)
!>          عند الإدخال، يخزن القسم N × N الأولي من H مصفوفة
!>          هسنبرغ التي تخضع للتحديد المبكر العدواني.
!>          عند الإخراج، تم تحويل H بواسطة تحويل متشابه وحدوي،
!>          تم اضطرابه، وأعيد إلى شكل هسنبرغ الذي (يؤمل) لديه بعض
!>          الإدخالات الفرعية الصفرية.
!>

LDH

!>          LDH هو INTEGER
!>          البعد الرائد لـ H كما هو معلن في الوحدة الفرعية المستدعية.
!>          N <= LDH
!>

ILOZ

!>          ILOZ هو INTEGER
!>

IHIZ

!>          IHIZ هو INTEGER
!>          حدد صفوف Z التي يجب تطبيق التحويلات عليها
!>          إذا كانت WANTZ .TRUE.. 1 <= ILOZ <= IHIZ <= N.
!>

Z

!>          Z هي مصفوفة COMPLEX، البعد (LDZ,N)
!>          إذا كانت WANTZ .TRUE.، فعند الإخراج، تم تجميع التحويل
!>          المتشابه الوحدوي المذكور أعلاه في Z(ILOZ:IHIZ,ILOZ:IHIZ) من اليمين.
!>          إذا كانت WANTZ .FALSE.، لا يتم الرجوع إلى Z.
!>

LDZ

!>          LDZ هو INTEGER
!>          البعد الرائد لـ Z كما هو معلن في الوحدة الفرعية المستدعية.
!>          1 <= LDZ.
!>

NS

!>          NS هو INTEGER
!>          عدد القيم الذاتية التقريبية (غير المتوافقة)
!>          المُرجعة في SR و SI التي يمكن استخدامها كتحويلات بواسطة
!>          الوحدة الفرعية المستدعية.
!>

ND

!>          ND هو INTEGER
!>          عدد القيم الذاتية المتوافقة التي كشفت عنها هذه
!>          الوحدة الفرعية.
!>

SH

!>          SH هي مصفوفة COMPLEX، البعد (KBOT)
!>          عند الإخراج، القيم الذاتية التقريبية التي يمكن
!>          استخدامها للتحويلات مخزنة في SH(KBOT-ND-NS+1)
!>          حتى SR(KBOT-ND). القيم الذاتية المتوافقة
!>          مخزنة في SH(KBOT-ND+1) حتى SH(KBOT).
!>

V

!>          V هي مصفوفة COMPLEX، البعد (LDV,NW)
!>          مصفوفة عمل NW × NW.
!>

LDV

!>          LDV هو INTEGER
!>          البعد الرائد لـ V كما هو معلن في الوحدة الفرعية المستدعية.
!>          NW <= LDV
!>

NH

!>          NH هو INTEGER
!>          عدد أعمدة T. NH >= NW.
!>

T

!>          T هي مصفوفة COMPLEX، البعد (LDT,NW)
!>

LDT

!>          LDT هو INTEGER
!>          البعد الرائد لـ T كما هو معلن في الوحدة الفرعية المستدعية.
!>          NW <= LDT
!>

NV

!>          NV هو INTEGER
!>          عدد صفوف مصفوفة العمل WV المتاحة لـ
!>          مساحة العمل. NV >= NW.
!>

WV

!>          WV هي مصفوفة COMPLEX، البعد (LDWV,NW)
!>

LDWV

!>          LDWV هو INTEGER
!>          البعد الرائد لـ W كما هو معلن في الوحدة الفرعية المستدعية.
!>          NW
