توثيق الوظائف

subroutine dlarrb (integer n, double precision, dimension( * ) d, double precision, dimension( * ) lld, integer ifirst, integer ilast, double precision rtol1, double precision rtol2, integer offset, double precision, dimension( * ) w, double precision, dimension( * ) wgap, double precision, dimension( * ) werr, double precision, dimension( * ) work, integer, dimension( * ) iwork, double precision pivmin, double precision spdiam, integer twist, integer info)

DLARRB يوفر تقسيمًا محدودًا لتحديد مواقع القيم الذاتية لمزيد من الدقة.

الغرض:

!>
!> نظرًا للتمثيل القوي نسبيًا (RRR) L D L^T، تقوم DLARRB
!> بالتقسيم لتحسين القيم الذاتية لـ L D L^T،
!> W( IFIRST-OFFSET ) إلى W( ILAST-OFFSET )، لمزيد من الدقة. المدخلات
!> التقديرية لهذه القيم الذاتية موجودة في W، والتقدير المقابل
!> للخطأ في هذه التقديرات وفجواتها موجود في WERR
!> و WGAP، على التوالي. أثناء التقسيم، يتم
!> الحفاظ على الفترات [left, right] عن طريق تخزين نقاط المنتصف
!> ونصف العرض في المصفوفات W و WERR على التوالي.
!>

المعلمات

N 
!>          N هو عدد صحيح
!>          رتبة المصفوفة.
!>

D

!>          D هي مصفوفة double precision
!>          العناصر القطرية N للمصفوفة القطرية D.
!>

LLD

!>          LLD هي مصفوفة double precision، البعد (N-1)
!>          العناصر (N-1) L(i)*L(i)*D(i).
!>

IFIRST

!>          IFIRST هو عدد صحيح
!>          فهرس أول قيمة ذاتية سيتم حسابها.
!>

ILAST

!>          ILAST هو عدد صحيح
!>          فهرس آخر قيمة ذاتية سيتم حسابها.
!>

RTOL1

!>          RTOL1 هو double precision
!>

RTOL2

!>          RTOL2 هو double precision
!>          التسامح لتقارب فترات التقسيم.
!>          فترة [LEFT,RIGHT] تقاربت إذا
!>          RIGHT-LEFT < MAX( RTOL1*GAP, RTOL2*MAX(|LEFT|,|RIGHT|) )
!>          حيث GAP هي المسافة (المقدرة) إلى أقرب
!>          قيمة ذاتية.
!>

OFFSET

!>          OFFSET هو عدد صحيح
!>          إزاحة لمصفوفات W، WGAP و WERR، أي، سيتم استخدام
!>          العناصر IFIRST-OFFSET إلى ILAST-OFFSET من هذه المصفوفات.
!>

W

!>          W هي مصفوفة double precision، البعد (N)
!>          عند الإدخال، W( IFIRST-OFFSET ) إلى W( ILAST-OFFSET ) هي
!>          تقديرات للقيم الذاتية لـ L D L^T مفهرسة IFIRST إلى
!>          ILAST.
!>          عند الخرج، تم تحسين هذه التقديرات.
!>

WGAP

!>          WGAP هي مصفوفة double precision، البعد (N-1)
!>          عند الإدخال، الفجوات (المقدرة) بين القيم الذاتية المتتالية
!>          لـ L D L^T، أي WGAP(I-OFFSET) هي الفجوة بين
!>          القيم الذاتية I و I+1. لاحظ أنه إذا كان IFIRST = ILAST
!>          فيجب تعيين WGAP(IFIRST-OFFSET) إلى الصفر.
!>          عند الخرج، تم تحسين هذه الفجوات.
!>

WERR

!>          WERR هي مصفوفة double precision، البعد (N)
!>          عند الإدخال، WERR( IFIRST-OFFSET ) إلى WERR( ILAST-OFFSET ) هي
!>          الأخطاء في تقديرات العناصر المقابلة في W.
!>          عند الخرج، تم تحسين هذه الأخطاء.
!>

WORK

!>          WORK هي مصفوفة double precision، البعد (2*N)
!>          مساحة عمل.
!>

IWORK

!>          IWORK هي مصفوفة عدد صحيح، البعد (2*N)
!>          مساحة عمل.
!>

PIVMIN

!>          PIVMIN هو double precision
!>          أصغر محور في تسلسل ستورم.
!>

SPDIAM

!>          SPDIAM هو double precision
!>          القطر الطيفي للمصفوفة.
!>

TWIST

!>          TWIST هو عدد صحيح
!>          فهرس الالتواء للتحليل الملتوي المستخدم
!>          للعد السلبي.
!>          TWIST = N: حساب العد السلبي من L D L^T - LAMBDA I = L+ D+ L+^T
!>          TWIST = 1: حساب العد السلبي من L D L^T - LAMBDA I = U- D- U-^T
!>          TWIST = R: حساب العد السلبي من L D L^T - LAMBDA I = N(r) D(r) N(r)
!>

INFO

!>          INFO هو عدد صحيح
!>          علم الخطأ.
!>
