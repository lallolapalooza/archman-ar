subroutine dlar1v (integer n, integer b1, integer bn, double precision lambda, double precision, dimension( * ) d, double precision, dimension( * ) l, double precision, dimension( * ) ld, double precision, dimension( * ) lld, double precision pivmin, double precision gaptol, double precision, dimension( * ) z, logical wantnc, integer negcnt, double precision ztz, double precision mingma, integer r, integer, dimension( * ) isuppz, double precision nrminv, double precision resid, double precision rqcorr, double precision, dimension( * ) work)

DLAR1V يحسب العمود r (المُقاس) للمصفوفة العكسية للمصفوفة الجزئية في الصفوف b1 إلى bn للمصفوفة ثلاثية القطر LDLT - λI.

الغرض:

!>
!> DLAR1V يحسب العمود r (المُقاس) للمصفوفة العكسية للمصفوفة الجزئية
!> في الصفوف B1 إلى BN للمصفوفة ثلاثية القطر L D L**T - sigma I. عندما يكون sigma قريبًا من قيمة ذاتية،
!> فإن المتجه المحسوب هو متجه ذاتي دقيق. عادةً، يتوافق r
!> مع الدليل حيث يكون المتجه الذاتي أكبر من حيث القيمة المطلقة.
!> الخطوات التالية تحقق هذه الحسابات:
!> (أ) تحويل qd الثابت، L D L**T - sigma I = L(+) D(+) L(+)**T,
!> (ب) تحويل qd التقدمي، L D L**T - sigma I = U(-) D(-) U(-)**T,
!> (ج) حساب العناصر القطرية للمصفوفة العكسية لـ
!> L D L**T - sigma I عن طريق دمج التحويلات المذكورة أعلاه، واختيار
!> r كدليل حيث تكون القطرية للمصفوفة العكسية (واحدة من)
!> الأكبر من حيث القيمة المطلقة.
!> (د) حساب العمود r (المُقاس) للمصفوفة العكسية باستخدام
!> العامل الملتوي الذي تم الحصول عليه عن طريق دمج الجزء العلوي من
!> التحويل الثابت والجزء السفلي من التحويل التقدمي.
!>

ZTZ

R

!>          R هو عدد صحيح (INTEGER)

!>           الدليل حيث تكون (L D L**T - sigma I)^-1 أكبر من حيث القيمة المطلقة.
!>           إذا كان 1 <= R <= N، فإن R لا يتغير.
!>           عند الإخراج، يحتوي R على مؤشر الالتواء المستخدم لحساب Z.
!>           من الناحية المثالية، يحدد R موضع أكبر مدخل في المتجه الذاتي.
!>

ISUPPZ

!>          ISUPPZ هو مصفوفة أعداد صحيحة (INTEGER)، البعد (2)
!>           دعم المتجه في Z، أي أن المتجه Z
!>           غير صفري فقط في العناصر من ISUPPZ(1) إلى ISUPPZ(2).
!>

NRMINV

!>          NRMINV هو عدد دقيق مزدوج (DOUBLE PRECISION)

RESID

!>           RESID = ABS( MINGMA )/SQRT( ZTZ )
!>

RQCORR

!>           RQCORR = MINGMA*TMP

WORK
