labnag(1)

الحزمة: labwc

التراخيص: GPL-2.0-only

labnag(1) دليل الأوامر العامة labnag(1)

الاسم

labnag - عرض مربع حوار مع رسالة وأزرار

ملخص

labnag [خيارات...]

الخيارات

-B, --button <نص> [<إجراء>]

إنشاء زر بالنص نص والذي ينفذ اختياريًا إجراء عند الضغط عليه. يمكن تعريف أزرار متعددة بتوفير العلامة عدة مرات. ستظهر الأزرار بالترتيب الذي يتم توفيره به من اليسار إلى اليمين.

-Z, --button-dismiss <نص> [<إجراء>]

إنشاء زر بالنص نص والذي ينفذ اختياريًا إجراء عند الضغط عليه، ويغلق labnag. يمكن تعريف أزرار متعددة بتوفير العلامة عدة مرات. ستظهر الأزرار بالترتيب الذي يتم توفيره به من اليسار إلى اليمين.

-d, --debug

تمكين التصحيح.

-e, --edge top|bottom

تحديد الحافة المراد استخدامها.

-y, --layer overlay|top|bottom|background

تحديد الطبقة المراد استخدامها.

-f, --font <خط>

تحديد الخط المراد استخدامه.

-h, --help

عرض رسالة المساعدة والخروج.

-l, --detailed-message

قراءة رسالة مفصلة من الإدخال القياسي. سيتم إضافة زر للتبديل بين التفاصيل. تظهر التفاصيل في منطقة نص متعددة الأسطر قابلة للتمرير.

-L, --detailed-button <نص>

تحديد نص الزر الذي يبدل التفاصيل. ليس لهذا الخيار أي تأثير إذا لم تكن هناك رسالة مفصلة. الافتراضي هو تبديل التفاصيل.

-m, --message <رسالة>

تحديد نص الرسالة.

-o, --output <مخرج>

تحديد المخرج المراد استخدامه. يجب أن يكون هذا اسم xdg_output.

-t, --timeout

تحديد مدة إغلاق مربع الحوار. الافتراضي هو 5 ثوانٍ.

-x, --exclusive-zone

استخدام المنطقة الحصرية. الافتراضي هو false.

-v, --version

عرض رقم الإصدار والخروج.

خيارات المظهر

--background-color <RRGGBB[AA]>

تحديد لون الخلفية.

--button-border-color <RRGGBB[AA]>

تحديد لون حدود الزر.

--border-bottom-color <RRGGBB[AA]>

تحديد لون الحد السفلي.

--button-background-color <RRGGBB[AA]>

تحديد لون خلفية الأزرار.

--text-color <RRGGBB[AA]>

تحديد لون النص.

--button-text-color <RRGGBB[AA]>

تحديد لون نص الزر.

--border-bottom-size <الحجم>

تحديد سمك الحد السفلي.

--message-padding <المسافة>

تحديد المسافة البادئة للرسالة.

--details-background-color <RRGGBB[AA]>

تحديد لون خلفية التفاصيل.

--details-border-size <الحجم>

تحديد سمك حدود التفاصيل.

--button-border-size <الحجم>

تحديد سمك حدود الزر.

--button-gap <الفجوة>

تحديد حجم الفجوة بين الأزرار.

--button-dismiss-gap <الفجوة>

تحديد حجم الفجوة بين زر الإغلاق وزر آخر.

--button-margin-right <الهامش>

تحديد الهامش من يمين زر الإغلاق إلى الحافة.

--button-padding <المسافة>

تحديد المسافة البادئة لنص الزر.

مثال

هذا مثال بسيط لواجهة رسومية لتسجيل الخروج باستخدام labnag.

#!/bin/sh
# تسجيل الخروج باستخدام labnag
labnag \
 -f "Hack Regular 10"\
 -m "اختر خيار تسجيل الخروج الخاص بك"\
 -Z "  القفل   " "gtklock -d"\
 -Z " تسجيل الخروج  " "labwc -e"\
 -Z "إيقاف التشغيل " "systemctl poweroff"\
 -Z " إعادة التشغيل " "systemctl reboot"\
 -Z "السبات " "systemctl hibernate"\
 -Z " الإسبات " "systemctl suspend"\
 -Z " إلغاء  "\
 --background-color 00ffff\
 --button-background-color 00ffff\
 --button-border-color 00ccccaa\
 --text-color 000000\
 --button-text-color 000000\
 --button-gap 8\
 --button-margin-right 0\
 --button-padding 5\
 --button-border-size 2\
 -t 60
2025-10-15
