توثيق الوظيفة

subroutine dlaed9 (integer k, integer kstart, integer kstop, integer n, double precision, dimension( * ) d, double precision, dimension( ldq, * ) q, integer ldq, double precision rho, double precision, dimension( * ) dlambda, double precision, dimension( * ) w, double precision, dimension( lds, * ) s, integer lds, integer info)

DLAED9 يستخدم بواسطة DSTEDC. يبحث عن جذور المعادلة العلمانية ويحدث المتجهات الذاتية. يستخدم عندما تكون المصفوفة الأصلية كثيفة.

الغرض:

!>
!> DLAED9 تبحث عن جذور المعادلة العلمانية، كما هو محدد بالقيم في D، Z، و RHO، بين KSTART و KSTOP. تقوم بإجراء المكالمات المناسبة لـ DLAED4 ثم تخزن مصفوفة المتجهات الذاتية الجديدة لاستخدامها في حساب المستوى التالي من متجهات Z.
!>

المعلمات

K 
!>          K هو عدد صحيح
!>          عدد الحدود في الدالة الكسرية التي سيتم حلها بواسطة
!>          DLAED4.  K >= 0.
!>

KSTART

!>          KSTART هو عدد صحيح
!>

KSTOP

!>          KSTOP هو عدد صحيح
!>          القيم الذاتية المحدثة Lambda(I)، KSTART <= I <= KSTOP
!>          سيتم حسابها.  1 <= KSTART <= KSTOP <= K.
!>

N

!>          N هو عدد صحيح
!>          عدد الصفوف والأعمدة في المصفوفة Q.
!>          N >= K (قد ينتج عن الحذف N > K).
!>

D

!>          D هي مصفوفة DOUBLE PRECISION، حجمها (N)
!>          D(I) يحتوي على القيم الذاتية المحدثة
!>          لـ KSTART <= I <= KSTOP.
!>

Q

!>          Q هي مصفوفة DOUBLE PRECISION، حجمها (LDQ,N)
!>

LDQ

!>          LDQ هو عدد صحيح
!>          البعد الرائد للمصفوفة Q.  LDQ >= max( 1, N ).
!>

RHO

!>          RHO هي DOUBLE PRECISION
!>          قيمة المعامل في معادلة التحديث ذات الرتبة واحد.
!>          مطلوب RHO >= 0.
!>

DLAMBDA

!>          DLAMBDA هي مصفوفة DOUBLE PRECISION، حجمها (K)
!>          تحتوي العناصر K الأولى من هذه المصفوفة على الجذور القديمة
!>          لمشكلة التحديث المخففة.  هذه هي الأقطاب
!>          للمعادلة العلمانية.
!>

W

!>          W هي مصفوفة DOUBLE PRECISION، حجمها (K)
!>          تحتوي العناصر K الأولى من هذه المصفوفة على مكونات
!>          متجه التحديث المعدل بالحذف.
!>

S

!>          S هي مصفوفة DOUBLE PRECISION، حجمها (LDS, K)
!>          ستحتوي على المتجهات الذاتية للمصفوفة التي تم إصلاحها والتي
!>          سيتم تخزينها لحساب متجه Z لاحقًا و
!>          ضربها في المتجهات الذاتية المتراكمة سابقًا
!>          لتحديث النظام.
!>

LDS

!>          LDS هو عدد صحيح
!>          البعد الرائد لـ S.  LDS >= max( 1, K ).
!>

INFO

!>          INFO هو عدد صحيح
!>          = 0:  خروج ناجح.
!>          < 0:  إذا كان INFO = -i، فإن الوسيط i كان له قيمة غير قانونية.
!>          > 0:  إذا كان INFO = 1، لم تتقارب قيمة ذاتية
!>
