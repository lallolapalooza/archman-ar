subroutine clahqr (logical wantt, logical wantz, integer n, integer ilo, integer ihi, complex, dimension( ldh, * ) h, integer ldh, complex, dimension( * ) w, integer iloz, integer ihiz, complex, dimension( ldz, * ) z, integer ldz, integer info)

CLAHQR تحسب القيم الذاتية وتحلل شور لمصفوفة هسنبرج علوية، باستخدام خوارزمية QR ذات الإزاحة المزدوجة/الإزاحة المفردة.

الغرض:

!>
!>    CLAHQR هي روتين مساعد يتم استدعاؤه بواسطة CHSEQR لتحديث
!>    القيم الذاتية وتحليل شور الذي تم حسابه بالفعل بواسطة CHSEQR، عن طريق
!>    التعامل مع المصفوفة الفرعية هسنبرج في الصفوف والأعمدة ILO إلى
!>    IHI.
!>

المعاملات

WANTT 
!>          WANTT هو LOGICAL
!>          = .TRUE. : يتم طلب شكل شور الكامل T؛
!>          = .FALSE. : يتم طلب القيم الذاتية فقط.
!>

WANTZ

!>          WANTZ هو LOGICAL
!>          = .TRUE. : يتم طلب مصفوفة متجهات شور Z؛
!>          = .FALSE. : لا يتم طلب متجهات شور.
!>

N

!>          N هو INTEGER
!>          ترتيب المصفوفة H. N >= 0.
!>

ILO

!>          ILO هو INTEGER
!>

IHI

!>          IHI هو INTEGER
!>          من المفترض أن تكون H بالفعل علوية مثلثة في الصفوف و
!>          الأعمدة IHI+1:N، وأن H(ILO,ILO-1) = 0 (ما لم يكن ILO = 1).
!>          تعمل CLAHQR بشكل أساسي مع المصفوفة الفرعية هسنبرج في الصفوف
!>          والأعمدة ILO إلى IHI، ولكنها تطبق تحويلات على H بأكملها
!>          إذا كانت WANTT هي .TRUE..
!>          1 <= ILO <= max(1,IHI); IHI <= N.
!>

H

!>          H هو مصفوفة COMPLEX، البعد (LDH,N)
!>          عند الإدخال، مصفوفة هسنبرج العلوية H.
!>          عند الخروج، إذا كان INFO يساوي صفر وإذا كانت WANTT هي .TRUE.، فإن H
!>          تكون علوية مثلثة في الصفوف والأعمدة ILO:IHI. إذا كان INFO
!>          يساوي صفر وإذا كانت WANTT هي .FALSE.، فإن محتويات H
!>          غير محددة عند الخروج. حالة الخروج لـ H في حالة
!>          INF إيجابي موجودة أدناه تحت وصف INFO.
!>

LDH

!>          LDH هو INTEGER
!>          البعد الرئيسي للمصفوفة H. LDH >= max(1,N).
!>

W

!>          W هي مصفوفة COMPLEX، البعد (N)
!>          يتم تخزين القيم الذاتية المحسوبة ILO إلى IHI في
!>          العناصر المقابلة لـ W. إذا كانت WANTT هي .TRUE.، فإن
!>          القيم الذاتية مخزنة بنفس ترتيب الأقطار
!>          للشكل شور المعادة في H، مع W(i) = H(i,i).
!>

ILOZ

!>          ILOZ هو INTEGER
!>

IHIZ

!>          IHIZ هو INTEGER
!>          حدد الصفوف من Z التي يجب تطبيق التحويلات عليها
!>          إذا كانت WANTZ هي .TRUE..
!>          1 <= ILOZ <= ILO; IHI <= IHIZ <= N.
!>

Z

!>          Z هي مصفوفة COMPLEX، البعد (LDZ,N)
!>          إذا كانت WANTZ هي .TRUE.، عند الإدخال يجب أن تحتوي Z على
!>          مصفوفة التحويلات الحالية Z التي تراكمت بواسطة CHSEQR، وعند
!>          الخروج تم تحديث Z؛ يتم تطبيق التحويلات فقط على
!>          المصفوفة الفرعية Z(ILOZ:IHIZ,ILO:IHI).
!>          إذا كانت WANTZ هي .FALSE.، فلا يتم الرجوع إلى Z.
!>

LDZ

!>          LDZ هو INTEGER
!>          البعد الرئيسي للمصفوفة Z. LDZ >= max(1,N).
!>

INFO

!>          INFO هو INTEGER
!>           = 0:  خروج ناجح
!>           > 0:  إذا كان INFO = i، فشلت CLAHQR في حساب جميع
!>                  القيم الذاتية ILO إلى IHI في إجمالي 30 تكرارًا
!>                  لكل قيمة ذاتية؛ العناصر i+1:ihi من W تحتوي على
!>                  تلك القيم الذاتية التي تم حسابها بنجاح.
!>
!>                  إذا كان INFO > 0 وكانت WANTT هي .FALSE.، فعند الخروج،
!>                  القيم الذاتية غير المتقاربة المتبقية هي
!>                  القيم الذاتية لمصفوفة هسنبرج العلوية
!>                  في الصفوف والأعمدة ILO خلال INFO من القيمة النهائية،
!>                  الناتجة لـ H.
!>
!>                  إذا كان INFO > 0 وكانت WANTT هي .TRUE.، فعند الخروج
!>          (*)       (القيمة الأولية لـ H)*U  = U*(القيمة النهائية لـ H)
!>                  حيث U هي مصفوفة متعامدة. القيمة النهائية
!>                  لـ H هي هسنبرج علوية ومثلثة في
!>                  الصفوف والأعمدة INFO+1 إلى IHI.
!>
!>                  إذا كان INFO > 0 وكانت WANTZ هي .TRUE.، فعند الخروج
!>                      (القيمة النهائية لـ Z)  = (القيمة الأولية لـ Z)*U
!>                  حيث U هي المصفوفة المتعامدة في (*)
!>                  (بغض النظر عن قيمة WANTT.)
!>
