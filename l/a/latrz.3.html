subroutine clatrz (integer m, integer n, integer l, complex, dimension( lda, * ) a, integer lda, complex, dimension( * ) tau, complex, dimension( * ) work)

CLATRZ تعامل مصفوفة شبه منحرفة علوية باستخدام تحويلات وحدوية.

الغرض:

!>
!> CLATRZ تعامل المصفوفة شبه المنحرفة العلوية المعقدة M-by-(M+L)
!> [ A1 A2 ] = [ A(1:M,1:M) A(1:M,N-L+1:N) ] على شكل ( R 0 ) * Z بواسطة
!> تحويلات وحدوية، حيث Z مصفوفة وحدوية (M+L)-by-(M+L) و R و A1 مصفوفتان مثلثيتان علويتان M-by-M.
!>

المعاملات

M 
!>          M هو عدد صحيح (INTEGER)
!>          عدد صفوف المصفوفة A.  M >= 0.
!>

N

!>          N هو عدد صحيح (INTEGER)
!>          عدد أعمدة المصفوفة A.  N >= 0.
!>

L

!>          L هو عدد صحيح (INTEGER)
!>          عدد أعمدة المصفوفة A التي تحتوي على الجزء ذي المعنى من متجهات هاوسهولدر. N-M >= L >= 0.
!>

A

!>          A هي مصفوفة معقدة (COMPLEX)، البعد (LDA,N)
!>          عند الدخول، يجب أن يحتوي الجزء العلوي شبه المنحرف M-by-N من المصفوفة A على المصفوفة المطلوب تحليلها.
!>          عند الخروج، يحتوي الجزء العلوي المثلثي M-by-M من A على المصفوفة المثلثية العلوية R، وعناصر N-L+1 إلى N للصفوف M الأولى من A، مع المصفوفة TAU، على تمثيل المصفوفة الوحدوية Z كناتج لـ M عاكس أولي.
!>

LDA

!>          LDA هو عدد صحيح (INTEGER)
!>          البعد الرئيسي للمصفوفة A.  LDA >= max(1,M).
!>

TAU

!>          TAU هي مصفوفة معقدة (COMPLEX)، البعد (*)
!>          العوامل العددية للعواكس الأولية.
!>

WORK

!>          WORK هي مصفوفة معقدة (COMPLEX)، البعد (*)
!>

تفاصيل إضافية:

!>
!>  يتم الحصول على التحليل بطريقة هاوسهولدر.  المصفوفة التحويلية k، Z( k )، التي تستخدم لإدخال أصفار في
!>  الصف ( m - k + 1 ) من A، معطاة في الصيغة
!>
!>     Z( k ) = ( I     0   )،
!>              ( 0  T( k ) )
!>
!>  حيث
!>
!>     T( k ) = I - tau*u( k )*u( k )**H,   u( k ) = (   1    )،
!>                                                 (   0    )
!>                                                 ( z( k ) )
!>
!>  يتم اختيار tau وهو عدد و z( k ) وهو متجه من l عنصر لإبطال عناصر الصف k من A2.
!>
!>  يتم إرجاع العدد tau في العنصر k من TAU والمتجه u( k ) في الصف k من A2، بحيث تكون عناصر z( k )
!>  في a( k, l + 1 )، ...، a( k, n ). يتم إرجاع عناصر R في
!>  الجزء المثلثي العلوي من A1.
!>
!>  يتم إعطاء Z بواسطة
!>
!>     Z =  Z( 1 ) * Z( 2 ) * ... * Z( m ).
!>

الوصف التفصيلي

تستخدم هذه الإجراءات طريقة هاوسهولدر لتفكيك مصفوفة شبه منحرفة علوية.

يتم استخدام الوظيفة slatrz مع المصفوفات الحقيقية والوظيفة zlatrz مع المصفوفات المركبة.

تستقبل كلتا الوظيفتين مصفوفة علوية شبه منحرفة A بحيث M صفوف و N عمود. تمثل L الأبعاد الإضافية للمصفوفة. تتمثل المهمة في إيجاد مصفوفة علوية ثلاثية R ومصفوفة وحدوية Z بحيث A = R * Z. يتم تخزين R في الجزء العلوي الأيسر من A، ويتم تمثيل Z كناتج للمنحدرات الأولية التي يتم تخزينها في الجزء العلوي الأيمن من A.

!>
!>     Z =  Z(1) * Z(2) * ... * Z(m).
!>

subroutine zlatrz (integer m, integer n, integer l, complex*16, dimension( lda, * ) a, integer lda, complex*16, dimension( * ) tau, complex*16, dimension( * ) work)

ZLATRZ تقوم بتفكيك مصفوفة شبه منحرفة علوية باستخدام تحويلات وحدوية.

الغرض:

!>
!> ZLATRZ تفكك المصفوفة العلوية شبه المنحرفة المركبة M-by-(M+L)
!> [ A1 A2 ] = [ A(1:M,1:M) A(1:M,N-L+1:N) ] إلى ( R  0 ) * Z بواسطة
!> تحويلات وحدوية، حيث Z مصفوفة وحدوية (M+L)-by-(M+L) و R و A1
!> مصفوفتان علويتان ثلاثيتان M-by-M.
!>

المعلمات

M 
!>          M هو عدد صحيح (INTEGER)
!>          عدد صفوف المصفوفة A. M >= 0.
!>

N

!>          N هو عدد صحيح (INTEGER)
!>          عدد أعمدة المصفوفة A. N >= 0.
!>

L

!>          L هو عدد صحيح (INTEGER)
!>          عدد أعمدة المصفوفة A التي تحتوي على الجزء الهادف
!>          لمتجهات هاوسهولدر. N-M >= L >= 0.
!>

A

!>          A هي مصفوفة مركبة*16 (COMPLEX*16 array)، ذات البعد (LDA,N)
!>          عند الدخول، يجب أن يحتوي الجزء العلوي شبه المنحرف ذو الأبعاد
!>          M-by-N الرائد في المصفوفة A على المصفوفة المراد تفكيكها.
!>          عند الخروج، يحتوي الجزء العلوي الثلاثي ذو الأبعاد M-by-M
!>          الرائد في A على المصفوفة العلوية الثلاثية R، وتمثل العناصر
!>          من N-L+1 إلى N من الصفوف الأولى لـ A، مع المصفوفة TAU،
!>          المصفوفة الوحدوية Z كناتج لـ M من الانعكاسات الأولية.
!>

LDA

!>          LDA هو عدد صحيح (INTEGER)
!>          البعد الرائد للمصفوفة A. LDA >= max(1,M).
!>

TAU

!>          TAU هي مصفوفة مركبة*16 (COMPLEX*16 array)، ذات البعد (M)
!>          العوامل العددية للانعكاسات الأولية.
!>

WORK

!>          WORK هي مصفوفة مركبة*16 (COMPLEX*16 array)، ذات البعد (M)
!>
