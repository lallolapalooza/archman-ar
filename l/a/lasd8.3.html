subroutine dlasd8 (integer icompq, integer k, double precision, dimension( * ) d, double precision, dimension( * ) z, double precision, dimension( * ) vf, double precision, dimension( * ) vl, double precision, dimension( * ) difl, double precision, dimension( lddifr, * ) difr, integer lddifr, double precision, dimension( * ) dsigma, double precision, dimension( * ) work, integer info)

DLASD8 تجد الجذور التربيعية لجذور المعادلة العلمانية، وتخزن، لكل عنصر في D، المسافة إلى أقرب قطبين له. تستخدم بواسطة sbdsdc.

الهدف:

!>
!> DLASD8 تجد الجذور التربيعية لجذور المعادلة العلمانية،
!> كما تم تعريفها بواسطة القيم في DSIGMA و Z. تقوم بإجراء الاستدعاءات المناسبة
!> لـ DLASD4، وتخزن، لكل عنصر في D، المسافة
!> إلى أقرب قطبين لها (العناصر في DSIGMA). كما أنها تحدث
!> المصفوفات VF و VL، المكونات الأولى والأخيرة لجميع
!> المتجهات المفردة اليمنى للمصفوفة ثنائية القطر الأصلية.
!>
!> يتم استدعاء DLASD8 من DLASD6.
!>

المعلمات

ICOMPQ 
!>          ICOMPQ هو عدد صحيح
!>          يحدد ما إذا كان سيتم حساب المتجهات المفردة في
!>          شكل مسطور في الروتين المستدعي:
!>          = 0: حساب القيم المفردة فقط.
!>          = 1: حساب المتجهات المفردة في شكل مسطور أيضًا.
!>

K

!>          K هو عدد صحيح
!>          عدد الحدود في الدالة الكسرية التي سيتم حلها
!>          بواسطة DLASD4.  K >= 1.
!>

D

!>          D هي مصفوفة double precision، حجمها ( K )
!>          عند الخروج، تحتوي D على القيم المفردة المحدثة.
!>

Z

!>          Z هي مصفوفة double precision، حجمها ( K )
!>          عند الدخول، تحتوي العناصر K الأولى لهذه المصفوفة على
!>          مكونات متجه الصف المحدث المضبوط بالانكماش.
!>          عند الخروج، يتم تحديث Z.
!>

VF

!>          VF هي مصفوفة double precision، حجمها ( K )
!>          عند الدخول، تحتوي VF على معلومات تم تمريرها عبر DBEDE8.
!>          عند الخروج، تحتوي VF على المكونات K الأولى لـ
!>          المكونات الأولى لجميع المتجهات المفردة اليمنى للمصفوفة ثنائية القطر.
!>

VL

!>          VL هي مصفوفة double precision، حجمها ( K )
!>          عند الدخول، تحتوي VL على معلومات تم تمريرها عبر DBEDE8.
!>          عند الخروج، تحتوي VL على المكونات K الأولى لـ
!>          المكونات الأخيرة لجميع المتجهات المفردة اليمنى للمصفوفة ثنائية القطر.
!>

DIFL

!>          DIFL هي مصفوفة double precision، حجمها ( K )
!>          عند الخروج، DIFL(I) = D(I) - DSIGMA(I).
!>

DIFR

!>          DIFR هي مصفوفة double precision،
!>                   حجمها ( LDDIFR, 2 ) إذا كان ICOMPQ = 1 و
!>                   حجمها ( K ) إذا كان ICOMPQ = 0.
!>          عند الخروج، DIFR(I,1) = D(I) - DSIGMA(I+1)، DIFR(K,1) غير معرف
!>          ولن يتم الإشارة إليه.
!>
!>          إذا كان ICOMPQ = 1، فإن DIFR(1:K,2) هي مصفوفة تحتوي على
!>          عوامل التطبيع لمصفوفة المتجهات المفردة اليمنى.
!>

LDDIFR

!>          LDDIFR هو عدد صحيح
!>          البعد الرائد لـ DIFR، يجب أن يكون على الأقل K.
!>

DSIGMA

!>          DSIGMA هي مصفوفة double precision، حجمها ( K )
!>          عند الدخول، تحتوي العناصر K الأولى لهذه المصفوفة على
!>          الجذور القديمة لمشكلة التحديث المنكمشة. هذه هي الأقطاب
!>          للمعادلة العلمانية.
!>

WORK

!>          WORK هي مصفوفة double precision، حجمها (3*K)
!>

INFO

!>          INFO هو عدد صحيح
!>          = 0:  الخروج بنجاح.
!>          < 0:  إذا كان INFO = -i، فإن المعامل i كان له قيمة غير قانونية.
!>          > 0:  إذا كان INFO = 1، لم تتقارب قيمة مفردة
!>

NAME

lasd8

computes the L D Q^T factorization of a diagonal matrix plus low rank matrix

SYNOPSIS

    SUBROUTINE LASD8(
       ICOMPQ, NL, NR, SQRE, K, DIFL, DIFR, D, Z,
       QD, QL, LDQ, QS, V, LDV, PK, PKV, WORK, IWORK, INFO
    )

        LOGICAL :: ICOMPQ
        INTEGER :: NL
        INTEGER :: NR
        INTEGER :: SQRE
        INTEGER :: K
        REAL :: DIFL( * )
        REAL :: DIFR( * )
        REAL :: D( * )
        REAL :: Z( * )
        REAL :: QD( * )
        REAL :: QL( * )
        INTEGER :: LDQ
        REAL :: QS( * )
        REAL :: V( * )
        INTEGER :: LDV
        REAL :: PK( * )
        REAL :: PKV( * )
        REAL :: WORK( * )
        INTEGER :: IWORK( * )
        INTEGER :: INFO

Detailed Description

This routine is called by the master routine LASD7 to compute the L D Q^T factorization of the matrix R in the equation:

|| R - sigma I || = 0

where R is the diagonal matrix R = diag( D ) and sigma is a shift. The equation is in the form:

Diag( D ) + Z * V^T = sigma I

Here Z is a diagonal matrix and V is a matrix of rank 1. LASD8 uses the same logic as LASD6. See the documentation of LASD6 for detailed explanation.

Arguments:

ICOMPQ
LOGICAL. Options to preserve eigenvectors.
NL
INTEGER. The number of non-zero elements in the diagonal matrix D.
NR
INTEGER. The number of non-zero elements in the diagonal matrix D.
SQRE
INTEGER. The square size of the current block of the matrix.
K
INTEGER. The number of columns of V, W, etc. K = NL + NR + 1.
DIFL
REAL array, dimension ( K ). On exit, DIFL(I) = D(I) - DSIGMA(I).
DIFR
REAL array, dimension ( K ). On exit, DIFR(I) = D(I) - DSIGMA(I+1).
D
REAL array, dimension ( K ). The initial values of D are the diagonal elements of the matrix. On exit, D contains the extremal singular values (in sorted order).
Z
REAL array, dimension ( K ). The diagonal elements of the matrix Z.
QD
REAL array, dimension ( K ). On exit, QD contains the diagonal elements of the matrix Q D.
QL
REAL array, dimension ( LDQ, 2 ). On exit, QL contains the left singular vectors of the bidiagonal matrix.
LDQ
INTEGER. The leading dimension of the array QL and QS.
QS
REAL array, dimension ( LDQ, 2 ). On exit, QS contains the right singular vectors of the bidiagonal matrix.
V
REAL array, dimension ( LDV, K ). The matrix V of rank K.
LDV
INTEGER. The leading dimension of V.
PK
REAL array, dimension ( K ).
PKV
REAL array, dimension ( K ).
WORK
REAL array, dimension (3*K).
IWORK
INTEGER array, dimension (3*K).
INFO
INTEGER.
  • = 0: successful exit.
  • < 0: if INFO = -i, the i-th argument had an illegal value.
  • > 0: if INFO = 1, the updating process failed.

Function Documentation

The function computes the L D Q^T factorization of a diagonal matrix plus low rank matrix.

Parameters:

ICOMPQ

!>          ICOMPQ is LOGICAL
!>          = .TRUE.: Compute the vectors Q and Z for the rank-one
!>                  update of the singular vector matrix.
!>          = .FALSE.: Compute the vectors only for the diagonal matrix D.
!>

NL

!>          NL is INTEGER
!>          Number of non-zero elements in D, number of negative
!>          singular values.
!>

NR

!>          NR is INTEGER
!>          Number of positive singular values.
!>

SQRE

!>          SQRE is INTEGER
!>          Size of the current block. SQRE = NR + NL + 1.
!>

K

!>          K is INTEGER
!>          The dimension of the matrix, K = NL + NR + 1.
!>

DIFL

!>          DIFL is REAL array, dimension ( K )
!>          On exit, DIFL(I) = D(I) - DSIGMA(I)
!>

DIFR

!>          DIFR is REAL array, dimension ( K )
!>          On exit, DIFR(I) = D(I) - DSIGMA(I+1).
!>

D

!>          D is REAL array, dimension ( K )
!>          On entry, D contains the diagonal elements of the matrix
!>          D. On exit, D contains the extremal singular values in
!>          sorted order.
!>

Z

!>          Z is REAL array, dimension ( K )
!>          On entry, Z contains the diagonal elements of the matrix Z.
!>

QD

!>          QD is REAL array, dimension ( K )
!>          On exit, QD contains the diagonal elements of the matrix Q D.
!>

QL

!>          QL is REAL array, dimension ( LDQ, 2 )
!>          On exit, QL contains the left singular vectors of the
!>          bidiagonal matrix.
!>

LDQ

!>          LDQ is INTEGER
!>          The leading dimension of QL and QS.
!>

QS

!>          QS is REAL array, dimension ( LDQ, 2 )
!>          On exit, QS contains the right singular vectors of the
!>          bidiagonal matrix.
!>

V

!>          V is REAL array, dimension ( LDV, K )
!>          On entry, V contains the first K components of the columns

LDV

PK

PKV

!>

WORK

!>          WORK is REAL array, dimension (3*K)

IWORK

!>          IWORK is INTEGER array, dimension (3*K)
