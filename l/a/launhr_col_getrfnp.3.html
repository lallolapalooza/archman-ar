subroutine claunhr_col_getrfnp (integer m, integer n, complex, dimension( lda, * ) a, integer lda, complex, dimension( * ) d, integer info)

CLAUNHR_COL_GETRFNP

الغرض:

!>
!> تقوم CLAUNHR_COL_GETRFNP بحساب عامل LU المعدل بدون
!> تبديل لمصفوفة A العامة المعقدة M-by-N. العامل له الشكل:
!>
!>     A - S = L * U,
!>
!> حيث:
!>    S هي مصفوفة علامة قطرية m-by-n مع القطر D، بحيث أن
!>    D(i) = S(i,i)، 1 <= i <= min(M,N). يتم بناء القطر D
!>    على أنه D(i)=-SIGN(A(i,i))، حيث A(i,i) هي القيمة بعد تنفيذ
!>    خطوات الاستبعاد الغاوسي i-1. هذا يعني أن العنصر القطري
!>    في كل خطوة من خطوات الاستبعاد الغاوسي
!>    على الأقل واحد في القيمة المطلقة (بحيث لا يكون هناك قسمة على
!>    صفر ممكنة أثناء القسمة على العنصر القطري)؛
!>
!>    L هي مصفوفة قطرية سفلية M-by-N مع عناصر قطرية وحدية
!>    (شبه منحرفة سفلية إذا كانت M > N)؛
!>
!>    و U هي مصفوفة قطرية علوية M-by-N
!>    (شبه منحرفة علوية إذا كانت M < N).
!>
!> هذه الروتين هي روتين مساعد تستخدم في روتين إعادة بناء هاوسهولدر
!> CUNHR_COL. في CUNHR_COL، يتم تطبيق هذه الروتين على مصفوفة
!> A ذات الأعمدة المتعامدة M-by-N، حيث كل عنصر محدود بواحد في القيمة
!> المطلقة. مع اختيار
!> المصفوفة S أعلاه، يمكن إظهار أن العنصر القطري في كل
!> خطوة من خطوات الاستبعاد الغاوسي هو الأكبر (في القيمة المطلقة) في
!> العمود على أو تحت القطر، بحيث لا يتطلب تبديل
!> للاستقرار العددي [1].
!>
!> لمزيد من التفاصيل حول خوارزمية إعادة بناء هاوسهولدر،
!> بما في ذلك عامل LU المعدل، انظر [1].
!>
!> هذا هو الإصدار المحظور الذي ينظر إلى اليمين من الخوارزمية،
!> يستدعي BLAS المستوى 3 لتحديث المصفوفة الفرعية. لعامل كتلة،
!> تستدعي هذه الروتين الروتين العودي CLAUNHR_COL_GETRFNP2.
!>
!> [1] ,
!>     G. Ballard, J. Demmel, L. Grigori, M. Jacquelin, H.D. Nguyen,
!>     E. Solomonik, J. Parallel Distrib. Comput.,
!>     vol. 85, pp. 3-31, 2015.
!>

المعلمات

M 
!>          M هو عدد صحيح
!>          عدد صفوف المصفوفة A. M >= 0.
!>

N

!>          N هو عدد صحيح
!>          عدد أعمدة المصفوفة A. N >= 0.
!>

A

!>          A هي مصفوفة معقدة، أبعادها (LDA,N)
!>          عند الإدخال، المصفوفة M-by-N المراد تحليلها.
!>          عند الخروج، عوامل L و U من التحليل
!>          A-S=L*U؛ عناصر القطر الوحدية لـ L غير مخزنة.
!>

LDA

!>          LDA هو عدد صحيح
!>          البعد الرائد للمصفوفة A. LDA >= max(1,M).
!>

D

!>          D هي مصفوفة معقدة، أبعادها min(M,N)
!>          العناصر القطرية لمصفوفة الإشارة القطرية S ذات الأبعاد M-by-N،
!>          D(i) = S(i,i)، حيث 1 <= i <= min(M,N). يمكن أن تكون العناصر
!>          فقط ( +1.0, 0.0 ) أو (-1.0, 0.0 ).
!>

INFO

!>          INFO هو عدد صحيح
!>          = 0:  انتهاء ناجح
!>          < 0:  إذا كان INFO = -i، فإن المعلمة i كانت لها قيمة غير قانونية
!>
