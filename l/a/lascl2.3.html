توثيق الوظيفة

subroutine clascl2 (integer m, integer n, real, dimension( * ) d, complex, dimension( ldx, * ) x, integer ldx)

CLASCL2 تقوم بقياس قطري للمصفوفة.

الغرض:

!>
!> CLASCL2 تقوم بقياس قطري للمصفوفة:
!>   x <-- D * x
!> حيث يتم تخزين المصفوفة القطرية REAL D كمصفوفة.
!>
!> سيتم استبدالها في النهاية بـ BLAS_cge_diag_scale في معيار BLAS الجديد
!>

المعلمات

M 
!>          M هو عدد صحيح
!>     عدد صفوف D و X. M >= 0.
!>

N

!>          N هو عدد صحيح
!>     عدد أعمدة X. N >= 0.
!>

D

!>          D هو مصفوفة REAL، طولها M
!>     المصفوفة القطرية D، مخزنة كمتجه بطول M.
!>

X

!>          X هي مصفوفة COMPLEX، أبعادها (LDX,N)
!>     عند الإدخال، المصفوفة X التي سيتم قياسها بواسطة D.
!>     عند الخروج، المصفوفة المقاسة.
!>

LDX

!>          LDX هو عدد صحيح
!>     البعد الرائد للمصفوفة X. LDX >= M.
!>
