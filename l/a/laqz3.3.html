!>

N

ILO

!>

IHI

!>

NW

!>          NW is INTEGER
!>          الحجم المطلوب لنافذة التفريغ.
!>

A

!>          A is DOUBLE PRECISION array, dimension (LDA, N)
!>

LDA

!>          LDA is INTEGER
!>          البعد الرائد للمصفوفة A.  LDA >= max( 1, N ).
!>

B

!>          B is DOUBLE PRECISION array, dimension (LDB, N)
!>

LDB

!>          LDB is INTEGER
!>          البعد الرائد للمصفوفة B.  LDB >= max( 1, N ).
!>

Q

!>          Q is DOUBLE PRECISION array, dimension (LDQ, N)
!>

LDQ

!>          LDQ is INTEGER
!>

Z

!>          Z is DOUBLE PRECISION array, dimension (LDZ, N)
!>

LDZ

!>          LDZ is INTEGER
!>

NS

!>          NS is INTEGER
!>          عدد القيم الذاتية غير المتقاربة المتاحة للاستخدام كتحويلات.
!>

ND

!>          ND is INTEGER
!>          عدد القيم الذاتية المتقاربة التي تم العثور عليها.
!>

ALPHAR

!>          ALPHAR is DOUBLE PRECISION array, dimension (N)
!>          الأجزاء الحقيقية لكل قيمة قياسية ألفا تحدد قيمة ذاتية
!>          لـ GNEP.
!>

ALPHAI

!>          ALPHAI is DOUBLE PRECISION array, dimension (N)
!>          الأجزاء التخيلية لكل قيمة قياسية ألفا تحدد قيمة ذاتية
!>          لـ GNEP.
!>          إذا كانت ALPHAI(j) صفرًا، فإن القيمة الذاتية j حقيقية؛ إذا
!>          كانت موجبة، فإن القيمتين الذاتيتين j و (j+1) هما زوج
!>          مترافق تخيلي، حيث ALPHAI(j+1) = -ALPHAI(j).
!>

BETA

!>          BETA is DOUBLE PRECISION array, dimension (N)
!>          القيم القياسية بيتا التي تحدد القيم الذاتية لـ GNEP.
!>          معًا، الكميات ألفا = (ALPHAR(j),ALPHAI(j)) و
!>          بيتا = BETA(j) تمثل القيمة الذاتية j لزوج المصفوفات
!>          (A,B)، في أحد الأشكال lambda = alpha/beta أو
!>          mu = beta/alpha. نظرًا لأن lambda أو mu يمكن أن يفيضا،
!>          لا ينبغي حسابهما بشكل عام.
!>

QC

!>          QC is DOUBLE PRECISION array, dimension (LDQC, NW)
!>

LDQC

!>          LDQC is INTEGER
!>

ZC

!>          ZC is DOUBLE PRECISION array, dimension (LDZC, NW)
!>

LDZC

!>          LDZ is INTEGER
!>

WORK

!>          WORK is DOUBLE PRECISION array, dimension (MAX(1,LWORK))
!>          عند الخروج، إذا كان INFO >= 0، فإن WORK(1) تعيد LWORK الأمثل.
!>

LWORK

!>          LWORK is INTEGER
!>          بعد المصفوفة WORK.  LWORK >= max(1,N).
!>
!>          إذا كان LWORK = -1، فسيتم افتراض استعلام مساحة العمل؛
!>          تقوم الإجراء فقط بحساب الحجم الأمثل لمصفوفة WORK،
!>          وتعيد هذا الحجم كإدخال أول لمصفوفة WORK، ولا يصدر
!>          أي رسالة خطأ تتعلق بـ LWORK بواسطة XERBLA.
!>

REC

!>          REC is INTEGER
!>             يشير REC إلى مستوى الاستدعاء التكراري الحالي. يجب تعيينه
!>             إلى 0 عند الاستدعاء الأول.
!>

INFO

!>          INFO is INTEGER
!>          = 0: خروج ناجح
!>          < 0: إذا كان INFO = -i، فإن المعلمة i كان لها قيمة غير قانونية
!>
