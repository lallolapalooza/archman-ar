SLASD7 تدمج مجموعتي القيم المفردة معًا في مجموعة واحدة مرتبة. ثم تحاول تفريغ حجم المشكلة. تستخدم بواسطة sbdsdc.

توثيق الوظيفة

روتين فرعي dlasd7 (integer icompq, integer nl, integer nr, integer sqre, integer k, double precision, dimension( * ) d, double precision, dimension( * ) z, double precision, dimension( * ) zw, double precision, dimension( * ) vf, double precision, dimension( * ) vfw, double precision, dimension( * ) vl, double precision, dimension( * ) vlw, double precision alpha, double precision beta, double precision, dimension( * ) dsigma, integer, dimension( * ) idx, integer, dimension( * ) idxp, integer, dimension( * ) idxq, integer, dimension( * ) perm, integer givptr, integer, dimension( ldgcol, * ) givcol, integer ldgcol, double precision, dimension( ldgnum, * ) givnum, integer ldgnum, double precision c, double precision s, integer info)

DLASD7 تدمج مجموعتي القيم المفردة معًا في مجموعة واحدة مرتبة. ثم تحاول تفريغ حجم المشكلة. تستخدم بواسطة sbdsdc.

الغرض:

!>
!> DLASD7 تدمج مجموعتي القيم المفردة معًا في مجموعة واحدة مرتبة. ثم تحاول
!> تفريغ حجم المشكلة. هناك طريقتان يمكن أن يحدث بهما التفريغ: عندما تكون قيمتان مفردتان أو أكثر قريبتين من بعضهما البعض أو إذا كان هناك إدخال صغير في متجه Z. لكل حدوث من هذا القبيل، يتم تقليل رتبة مشكلة المعادلة العلمانية ذات الصلة بواحد.
!>
!> يتم استدعاء DLASD7 من DLASD6.
!>

المعلمات

ICOMPQ 
!>          ICOMPQ هو INTEGER
!>          يحدد ما إذا كان سيتم حساب المتجهات المفردة
!>          بشكل مضغوط، على النحو التالي:
!>          = 0: حساب القيم المفردة فقط.
!>          = 1: حساب المتجهات المفردة للمصفوفة
!>               ثنائية القطر العليا بشكل مضغوط.
!>

NL

!>          NL هو INTEGER
!>         البعد الصف للمصفوفة الفرعية العلوية. NL >= 1.
!>

NR

!>          NR هو INTEGER
!>         البعد الصف للمصفوفة الفرعية السفلية. NR >= 1.
!>

SQRE

!>          SQRE هو INTEGER
!>         = 0: المصفوفة الفرعية السفلية هي مصفوفة مربعة NR-by-NR.
!>         = 1: المصفوفة الفرعية السفلية هي مصفوفة مستطيلة NR-by-(NR+1).
!>
!>         تحتوي المصفوفة ثنائية القطر على
!>         N = NL + NR + 1 صفوف و
!>         M = N + SQRE >= N أعمدة.
!>

K

!>          K هو INTEGER
!>         يحتوي على بعد المصفوفة غير المفرغة، هذه هي
!>         رتبة المعادلة العلمانية ذات الصلة. 1 <= K <=N.
!>

D

!>          D هو مصفوفة DOUBLE PRECISION، البعد ( N )
!>         عند الدخول تحتوي D على القيم المفردة للمصفوفتين الفرعيتين
!>         المزمع دمجهما. عند الخروج تحتوي D على القيم المفردة المحدثة
!>         (التي تم تفريغها) مرتبة بترتيب تصاعدي.
!>

Z

!>          Z هو مصفوفة DOUBLE PRECISION، البعد ( M )
!>         عند الخروج تحتوي Z على متجه الصف المحدث في المعادلة العلمانية.
!>

ZW

!>          ZW هو مصفوفة DOUBLE PRECISION، البعد ( M )
!>         مساحة عمل لـ Z.
!>

VF

!>          VF هو مصفوفة DOUBLE PRECISION، البعد ( M )
!>         عند الدخول، تحتوي VF(1:NL+1) على المكونات الأولى لجميع
!>         المتجهات المفردة اليمنى للمصفوفة الفرعية العلوية؛ وتحتوي VF(NL+2:M)
!>         على المكونات الأولى لجميع المتجهات المفردة اليمنى
!>         للمصفوفة الفرعية السفلية. عند الخروج، تحتوي VF على المكونات الأولى
!>         لجميع المتجهات المفردة اليمنى للمصفوفة ثنائية القطر.
!>

VFW

!>          VFW هو مصفوفة DOUBLE PRECISION، البعد ( M )
!>         مساحة عمل لـ VF.
!>

VL

!>          VL هو مصفوفة DOUBLE PRECISION، البعد ( M )
!>         عند الدخول، تحتوي VL(1:NL+1) على المكونات الأخيرة لجميع
!>         المتجهات المفردة اليمنى للمصفوفة الفرعية العلوية؛ وتحتوي VL(NL+2:M)
!>         على المكونات الأخيرة لجميع المتجهات المفردة اليمنى
!>         للمصفوفة الفرعية السفلية. عند الخروج، تحتوي VL على المكونات الأخيرة
!>         لجميع المتجهات المفردة اليمنى للمصفوفة ثنائية القطر.
!>

VLW

!>          VLW هو مصفوفة DOUBLE PRECISION، البعد ( M )
!>         مساحة عمل لـ VL.
!>

ALPHA

!>          ALPHA هو DOUBLE PRECISION
!>         يحتوي على العنصر القطري المرتبط بالصف المضاف.
!>

BETA

!>          BETA هو DOUBLE PRECISION
!>         يحتوي على العنصر غير القطري المرتبط بالصف المضاف.
!>

DSIGMA

!>          DSIGMA هو مصفوفة DOUBLE PRECISION، البعد ( N )
!>         تحتوي على نسخة من العناصر القطرية (K-1 قيم مفردة
!!>         وصفر واحد) في المعادلة العلمانية.
!>

IDX

!>          IDX هو مصفوفة INTEGER، البعد ( N )
!>         ستحتوي هذه على التبديل المستخدم لفرز محتويات
!>         D بترتيب تصاعدي.
!>

IDXP

!>          IDXP هو مصفوفة INTEGER، البعد ( N )
!>         ستحتوي هذه على التبديل المستخدم لوضع القيم
!>         المفرغة من D في نهاية المصفوفة. عند الخروج، يشير IDXP(2:K)
!>         إلى قيم D غير المفرغة ويشير IDXP(K+1:N)
!>         إلى القيم المفردة المفرغة.
!>

IDXQ

!>          IDXQ هو مصفوفة INTEGER، البعد ( N )
!>         تحتوي هذه على التبديل الذي يفرز بشكل منفصل الفرعين
!>         المشكلة في D بترتيب تصاعدي. ملاحظة أنه يجب أولاً نقل الإدخالات
!>         في النصف الأول من هذا التبديل بموضع واحد إلى الخلف؛ ويجب أولاً إضافة
!>         NL+1 إلى قيم الإدخالات في النصف الثاني.
!>

PERM

!>          PERM هو مصفوفة INTEGER، البعد ( N )
!>         التبديلات (من التفريغ والفرز) التي سيتم تطبيقها
!>         على كل كتلة مفردة. غير مشار إليها إذا كان ICOMPQ = 0.
!>

GIVPTR

!>          GIVPTR هو INTEGER
!>         عدد تحويلات جيفينز التي حدثت في هذه
!>         المشكلة الفرعية. غير مشار إليها إذا كان ICOMPQ = 0.
!>

GIVCOL

!>          GIVCOL هو مصفوفة INTEGER، البعد ( LDGCOL, 2 )
!>         يشير كل زوج من الأرقام إلى زوج من الأعمدة ليحدث فيها
