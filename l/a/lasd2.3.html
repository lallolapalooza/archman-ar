الاسم

lasd2 - lasd2: خطوة الانقسام والدمج (D&C): التفريغ

الموجز

الوظائف


الإجراء الفرعي dlasd2 (nl, nr, sqre, k, d, z, alpha, beta, u, ldu, vt, ldvt, dsigma, u2, ldu2, vt2, ldvt2, idxp, idx, idxc, idxq, coltyp, info)
DLASD2 تدمج مجموعتي القيم المفردة معًا في مجموعة واحدة مرتبة. تستخدم بواسطة sbdsdc. الإجراء الفرعي slasd2 (nl, nr, sqre, k, d, z, alpha, beta, u, ldu, vt, ldvt, dsigma, u2, ldu2, vt2, ldvt2, idxp, idx, idxc, idxq, coltyp, info)
SLASD2 تدمج مجموعتي القيم المفردة معًا في مجموعة واحدة مرتبة. تستخدم بواسطة sbdsdc.

الوصف التفصيلي

توثيق الوظيفة

الإجراء الفرعي dlasd2 (integer nl, integer nr, integer sqre, integer k, double precision, dimension( * ) d, double precision, dimension( * ) z, double precision alpha, double precision beta, double precision, dimension( ldu, * ) u, integer ldu, double precision, dimension( ldvt, * ) vt, integer ldvt, double precision, dimension( * ) dsigma, double precision, dimension( ldu2, * ) u2, integer ldu2, double precision, dimension( ldvt2, * ) vt2, integer ldvt2, integer, dimension( * ) idxp, integer, dimension( * ) idx, integer, dimension( * ) idxc, integer, dimension( * ) idxq, integer, dimension( * ) coltyp, integer info)

DLASD2 تدمج مجموعتي القيم المفردة معًا في مجموعة واحدة مرتبة. تستخدم بواسطة sbdsdc.

الغرض:

!>
!> DLASD2 تدمج مجموعتي القيم المفردة معًا في مجموعة واحدة مرتبة. ثم تحاول تفريغ حجم المشكلة.
!> هناك طريقتان يمكن أن يحدث بهما التفريغ: عندما تكون هناك قيم مفردة اثنتان أو أكثر قريبة من بعضها البعض
!> أو إذا كان هناك إدخال صغير جدًا في المتجه Z. لكل حدوث من هذا القبيل، يتم تقليل رتبة مشكلة
!> المعادلة العلمانية المرتبطة بمقدار واحد.
!>
!> يتم استدعاء DLASD2 من DLASD1.
!>

المعلمات

NL 
!>          NL هو عدد صحيح (INTEGER)
!>         بعد الصف للكتلة العلوية. NL >= 1.
!>

NR

!>          NR هو عدد صحيح (INTEGER)
!>         بعد الصف للكتلة السفلية. NR >= 1.
!>

SQRE

!>          SQRE هو عدد صحيح (INTEGER)
!>         = 0: الكتلة السفلية مصفوفة مربعة بحجم NR-by-NR.
!>         = 1: الكتلة السفلية مصفوفة مستطيلة بحجم NR-by-(NR+1).
!>
!>         المصفوفة ثنائية القطر لديها N = NL + NR + 1 صفوف
!>         و M = N + SQRE >= N أعمدة.
!>

K

!>          K هو عدد صحيح (INTEGER)
!>         يحتوي على بعد المصفوفة غير المفروغة،
!>         هذه هي رتبة المعادلة العلمانية المرتبطة. 1 <= K <= N.
!>

D

!>          D هو مصفوفة double precision، بعدها (N)
!>         عند الإدخال، يحتوي D على القيم المفردة للمصفوفتين الفرعيتين
!>         المراد دمجهما. عند الخروج، يحتوي D على القيم المفردة المحدثة
!>         اللاحقة (N-K) (تلك التي تم تفريغها) مرتبة في
!>         ترتيب متزايد.
!>

Z

!>          Z هو مصفوفة double precision، بعدها (N)
!>         عند الخروج، يحتوي Z على المتجه الصف المحدث في
!>         المعادلة العلمانية.
!>

ALPHA

!>          ALPHA هو double precision
!>         يحتوي على العنصر القطري المرتبط بالصف المضاف.
!>

BETA

!>          BETA هو double precision
!>         يحتوي على العنصر غير القطري المرتبط بالصف المضاف.
!>

U

!>          U هو مصفوفة double precision، بعدها (LDU,N)
!>         عند الإدخال، يحتوي U على المتجهات المفردة اليسرى لمصفوفتين فرعيتين
!>         في الكتلتين المربعتين اللتين لهما زوايا عند (1,1)،
!>         (NL, NL)، و (NL+2, NL+2)، (N,N).
!>         عند الخروج، يحتوي U على المتجهات المفردة اليسرى المحدثة
!>         اللاحقة (N-K) (تلك التي تم تفريغها) في أعمدتها الأخيرة N-K.
!>

LDU

!>          LDU هو عدد صحيح (INTEGER)
!>         البعد الرائد للمصفوفة U. LDU >= N.
!>

VT

!>          VT هو مصفوفة double precision، بعدها (LDVT,M)
!>         عند الإدخال، يحتوي VT**T على المتجهات المفردة اليمنى لمصفوفتين فرعيتين
!>         في الكتلتين المربعتين اللتين لهما زوايا عند (1,1)،
!>         (NL+1, NL+1)، و (NL+2, NL+2)، (M,M).
!>         عند الخروج، يحتوي VT**T على المتجهات المفردة اليمنى المحدثة
!>         اللاحقة (N-K) (تلك التي تم تفريغها) في أعمدتها الأخيرة N-K.
!>         في حالة SQRE =1، الصف الأخير من VT يمتد فراغ العدم الأيمن.
!>

LDVT

!>          LDVT هو عدد صحيح (INTEGER)
!>         البعد الرائد للمصفوفة VT. LDVT >= M.
!>

DSIGMA

!>          DSIGMA هو مصفوفة double precision، بعدها (N)
!>         تحتوي على نسخة من العناصر القطرية (K-1 قيمة مفردة
!>         وصفر واحد) في المعادلة العلمانية.
!>

U2

!>          U2 هو مصفوفة double precision، بعدها (LDU2,N)
!>         تحتوي على نسخة من أول K-1 متجه مفرد أيسر والتي
!>         سيتم استخدامها بواسطة DLASD3 في ضرب مصفوفة (DGEMM) لحل
!>         المتجهات المفردة اليسرى الجديدة. يتم ترتيب U2 في أربع
!>         كتل. تحتوي الكتلة الأولى على عمود به 1 عند NL+1 وصفر
!>         في كل مكان آخر؛ تحتوي الكتلة الثانية على إدخالات غير صفرية
!>         فقط عند NL وما فوق؛ الثالثة تحتوي على إدخالات غير صفرية
!>         فقط تحت NL+1؛ والرابعة كثيفة.
!>

LDU2

!>          LDU2 هو عدد صحيح (INTEGER)
!>         البعد الرائد للمصفوفة U2. LDU2 >= N.
!>

VT2

!>          VT2 هو مصفوفة double precision، بعدها (LDVT2,N)
!>         يحتوي VT2**T على نسخة من أول K متجه مفرد أيمن
!>         التي سيتم استخدامها بواسطة DLASD3 في ضرب مصفوفة (DGEMM) لحل
!>         المتجهات المفردة اليمنى الجديدة. يتم ترتيب VT2 في
!>         ثلاث كتل. تحتوي الكتلة الأولى على صف يتوافق
!>         مع العنصر القطري الخاص 0 في SIGMA؛ تحتوي الكتلة الثانية
!>         على إدخالات غير صفرية فقط عند NL +1 وما قبله؛ الكتلة الثالثة
!>         تحتوي على إدخالات غير صفرية فقط عند NL +2 وما بعده.
!>

LDVT2

!>          LDVT2 هو عدد صحيح (INTEGER)
!>         البعد الرائد للمصفوفة VT2. LDVT2 >= M.
!>

IDXP

!>          IDXP هو مصفوفة عدد صحيح (INTEGER) بعدها (N)
!>         سيحتوي هذا على التبديل المستخدم لوضع القيم المفروغة
!>         من D في نهاية المصفوفة. عند الخروج، IDXP(2:K)
!>         يشير إلى قيم D غير المفروغة و IDXP(K+1:N)
!>         يشير إلى القيم المفردة المفروغة.
!>

IDX

!>          IDX هو مصفوفة عدد صحيح (INTEGER) بعدها (N)
!>         سيحتوي هذا على التبديل المستخدم لفرز محتويات
!>         D في ترتيب تصاعدي.
!>

IDXC

!>          IDXC هو مصفوفة عدد صحيح (INTEGER) بعدها (N)
!>         سيحتوي هذا على التبديل المستخدم لترتيب الأعمدة
!>         من مصفوفة U المفروغة إلى ثلاث مجموعات: المجموعة الأولى
!>         تحتوي على إدخالات غير صفرية فقط عند NL وما فوق، الثانية
!>         تحتوي على إدخالات غير صفرية فقط تحت NL+2، والثالثة
!>         كثيفة.
!>

IDXQ

!>          IDXQ هو مصفوفة عدد صحيح (INTEGER) بعدها (N)
!>         هذه تحتوي على التبديل الذي يفرز بشكل منفصل الاثنين
