subroutine dlag2 (double precision, dimension( lda, * ) a, integer lda, double precision, dimension( ldb, * ) b, integer ldb, double precision safmin, double precision scale1, double precision scale2, double precision wr1, double precision wr2, double precision wi)

DLAG2 تحسب القيم الذاتية لمشكلة القيم الذاتية المعممة 2x2، مع التحجيم حسب الضرورة لتجنب الزيادة/النقصان.

الغرض:

!>
!> DLAG2 تحسب القيم الذاتية لمشكلة القيم الذاتية المعممة 2x2  A - w B، مع التحجيم حسب الضرورة لتجنب الزيادة/النقصان.
!>
!> عامل التحجيم  ينتج عنه معادلة القيم الذاتية المعدلة
!>
!>     s A - w B
!>
!> حيث  s  هو عامل تحجيم غير سالب يتم اختياره بحيث لا تتجاوز  w،  w B،
!> و  s A  الحد الأقصى (overflow) ولا تقل عن الحد الأدنى (underflow)، إن أمكن.
!>

المعاملات

A 
!>          A هو مصفوفة DOUBLE PRECISION، ذات بعد (LDA, 2)
!>          عند الإدخال، المصفوفة 2x2 A. يُفترض أن حجم النمط 1 الخاص بها
!>          أقل من 1/SAFMIN. المدخلات الأقل من
!>          sqrt(SAFMIN)*norm(A) معرضة للمعاملة على أنها صفر.
!>

LDA

!>          LDA هو عدد صحيح INTEGER
!>          البعد الرائد للمصفوفة A. LDA >= 2.
!>

B

!>          B هي مصفوفة DOUBLE PRECISION، ذات بعد (LDB, 2)
!>          عند الإدخال، المصفوفة المثلثية العليا 2x2 B. يُفترض أن
!>          حجم النمط 1 لـ B أقل من 1/SAFMIN. يجب أن تكون
!>          الأقطار على الأقل sqrt(SAFMIN) مضروبة في أكبر
!>          عنصر في B (بالقيمة المطلقة)؛ إذا كان القطر أصغر
!>          من ذلك، فسيتم استخدام  +/- sqrt(SAFMIN) بدلاً من
!>          ذلك القطر.
!>

LDB

!>          LDB هو عدد صحيح INTEGER
!>          البعد الرائد للمصفوفة B. LDB >= 2.
!>

SAFMIN

!>          SAFMIN هو DOUBLE PRECISION
!>          أصغر عدد موجب s.t. 1/SAFMIN لا يسبب
!>          الزيادة (overflow). (يجب أن يكون هذا دائمًا DLAMCH('S') - وهو
!>          معامل لتجنب الاضطرار إلى استدعاء DLAMCH بشكل متكرر.)
!>

SCALE1

!>          SCALE1 هو DOUBLE PRECISION
!>          عامل تحجيم يستخدم لتجنب الزيادة/النقصان في
!>          معادلة القيم الذاتية التي تحدد القيمة الذاتية الأولى. إذا
!>          كانت القيم الذاتية مركبة، فإن القيم الذاتية هي
!>          ( WR1  +/-  WI i ) / SCALE1  (والتي قد تقع خارج
!>          نطاق الأس للجهاز)، SCALE1=SCALE2، و SCALE1
!>          سيكون دائمًا موجبًا. إذا كانت القيم الذاتية حقيقية، فإن
!>          القيمة الذاتية (الحقيقية) الأولى هي  WR1 / SCALE1 ، ولكن هذا قد
!>          يسبب زيادة أو نقصان (overflow or underflow)، وفي الواقع، قد يكون SCALE1 صفرًا أو
!>          أقل من عتبة النقصان إذا كانت القيمة الذاتية الدقيقة كبيرة بما فيه الكفاية.
!>

SCALE2

!>          SCALE2 هو DOUBLE PRECISION
!>          عامل تحجيم يستخدم لتجنب الزيادة/النقصان في
!>          معادلة القيم الذاتية التي تحدد القيمة الذاتية الثانية. إذا
!>          كانت القيم الذاتية مركبة، فإن SCALE2=SCALE1. إذا كانت
!>          القيم الذاتية حقيقية، فإن القيمة الذاتية (الحقيقية) الثانية هي
!>          WR2 / SCALE2 ، ولكن هذا قد يسبب زيادة أو نقصان (overflow or underflow)، وفي الواقع، قد يكون SCALE2 صفرًا أو أقل من عتبة النقصان
!>          إذا كانت القيمة الذاتية الدقيقة كبيرة بما فيه الكفاية.
!>

WR1

!>          WR1 هو DOUBLE PRECISION
!>          إذا كانت القيمة الذاتية حقيقية، فإن WR1 هو SCALE1 مضروبًا في
!>          القيمة الذاتية الأقرب إلى العنصر (2,2) من A B**(-1). إذا كانت
!>          القيمة الذاتية مركبة، فإن WR1=WR2 هو SCALE1 مضروبًا في الجزء الحقيقي
!>          من القيم الذاتية.
!>

WR2

!>          WR2 هو DOUBLE PRECISION
!>          إذا كانت القيمة الذاتية حقيقية، فإن WR2 هو SCALE2 مضروبًا في
!>          القيمة الذاتية الأخرى. إذا كانت القيمة الذاتية مركبة، فإن
!>          WR1=WR2 هو SCALE1 مضروبًا في الجزء الحقيقي من القيم الذاتية.
!>

WI

!>          WI هو DOUBLE PRECISION
!>          إذا كانت القيمة الذاتية حقيقية، فإن WI هو صفر. إذا كانت
!>          القيمة الذاتية مركبة، فإن WI هو SCALE1 مضروبًا في الجزء التخيلي
!>          من القيم الذاتية. سيكون WI دائمًا غير سالب.
!>
