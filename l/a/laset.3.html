توثيق الوظيفة

subroutine claset (character uplo, integer m, integer n, complex alpha, complex beta, complex, dimension( lda, * ) a, integer lda)

CLASET تهيئة العناصر خارج القطر والعناصر القطرية للمصفوفة إلى قيم معطاة.

الغرض:

!>
!> CLASET تهيئة مصفوفة ثنائية الأبعاد A إلى BETA على القطر و
!> ALPHA على خارج الأقطار.
!>

المعلمات

UPLO 
!>          UPLO هو CHARACTER*1
!>          يحدد الجزء من المصفوفة A الذي سيتم تعيينه.
!>          = 'U':      يتم تعيين الجزء المثلثي العلوي. يبقى الجزء السفلي
!>                      دون تغيير.
!>          = 'L':      يتم تعيين الجزء المثلثي السفلي. يبقى الجزء العلوي
!>                      دون تغيير.
!>          بخلاف ذلك: يتم تعيين كل المصفوفة A.
!>

M

!>          M هو INTEGER
!>          عند الدخول، M تحدد عدد صفوف A.
!>

N

!>          N هو INTEGER
!>          عند الدخول، N تحدد عدد أعمدة A.
!>

ALPHA

!>          ALPHA هو COMPLEX
!>          جميع عناصر المصفوفة خارج القطر يتم تعيينها إلى ALPHA.
!>

BETA

!>          BETA هو COMPLEX
!>          جميع عناصر المصفوفة القطرية يتم تعيينها إلى BETA.
!>

A

!>          A هو مصفوفة COMPLEX، بأبعاد (LDA,N)
!>          عند الدخول، المصفوفة A بحجم m × n.
!>          عند الخروج، A(i,j) = ALPHA، 1 <= i <= m، 1 <= j <= n، i.ne.j؛
!>                   A(i,i) = BETA، 1 <= i <= min(m,n)
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرائد للمصفوفة A. LDA >= max(1,M).
!>

subroutine dlaset (character uplo, integer m, integer n, double precision alpha, double precision beta, double precision, dimension( lda, * ) a, integer lda)

DLASET تهيئة العناصر خارج القطر والعناصر القطرية للمصفوفة إلى قيم معطاة.

الغرض:

!>
!> DLASET تهيئة مصفوفة بحجم m × n A إلى BETA على القطر و
!> ALPHA على خارج الأقطار.
!>

المعلمات

UPLO 
!>          UPLO هو CHARACTER*1
!>          يحدد الجزء من المصفوفة A الذي سيتم تعيينه.
!>          = 'U':      يتم تعيين الجزء المثلثي العلوي؛ الجزء السفلي تمامًا
!>                      من A لا يتغير.
!>          = 'L':      يتم تعيين الجزء المثلثي السفلي؛ الجزء العلوي تمامًا
!>                      من A لا يتغير.
!>          بخلاف ذلك: يتم تعيين كل المصفوفة A.
!>

M

!>          M هو INTEGER
!>          عدد صفوف المصفوفة A. M >= 0.
!>

N

!>          N هو INTEGER
!>          عدد أعمدة المصفوفة A. N >= 0.
!>

ALPHA

!>          ALPHA هو DOUBLE PRECISION
!>          الثابت الذي سيتم تعيين العناصر خارج القطر إليه.
!>

BETA

!>          BETA هو DOUBLE PRECISION
!>          الثابت الذي سيتم تعيين العناصر القطرية إليه.
!>

A

!>          A هو مصفوفة DOUBLE PRECISION، بأبعاد (LDA,N)
!>          عند الخروج، يتم تعيين الجزء الفرعي الرئيسي بحجم m × n من A على النحو التالي:
!>
!>          إذا كان UPLO = 'U'، فإن A(i,j) = ALPHA، 1<=i<=j-1، 1<=j<=n،
!>          إذا كان UPLO = 'L'، فإن A(i,j) = ALPHA، j+1<=i<=m، 1<=j<=n،
!>          بخلاف ذلك،     A(i,j) = ALPHA، 1<=i<=m، 1<=j<=n، i.ne.j،
!>
!>          و، لجميع UPLO، A(i,i) = BETA، 1<=i<=min(m,n).
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرائد للمصفوفة A. LDA >= max(1,M).
!>
