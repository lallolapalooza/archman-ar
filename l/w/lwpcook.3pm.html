"Sh" id='LARGE_DOCUMENTS'>مستندات كبيرة

إذا كان المستند الذي تريد جلبه كبيرًا جدًا بحيث لا يمكن الاحتفاظ به في الذاكرة، فلديك خياران. يمكنك إرشاد المكتبة لكتابة محتوى المستند إلى ملف (الحجة الثانية $ua->request() هي اسم ملف):

use LWP::UserAgent;
$ua = LWP::UserAgent->new;
my $req = HTTP::Request->new(GET =>
    'http://www.cpan.org/CPAN/authors/id/O/OA/OALDERS/libwww-perl-6.26.tar.gz');
$res = $ua->request($req, "libwww-perl.tar.gz");
if ($res->is_success) {
   print "ok\n";
}
else {
   print $res->status_line, "\n";
}

أو يمكنك معالجة المستند أثناء وصوله (الحجة الثانية $ua->request() هي مرجع كود):

use LWP::UserAgent;
$ua = LWP::UserAgent->new;
$URL = 'ftp://ftp.isc.org/pub/rfc/rfc-index.txt';
my $expected_length;
my $bytes_received = 0;
my $res =
   $ua->request(HTTP::Request->new(GET => $URL),
             sub {
                 my($chunk, $res) = @_;
                 $bytes_received += length($chunk);
                 unless (defined $expected_length) {
                    $expected_length = $res->content_length || 0;
                 }
                 if ($expected_length) {
                      printf STDERR "%d%% - ",
                                100 * $bytes_received / $expected_length;
                 }
                 print STDERR "$bytes_received bytes received\n";
                 # XXX Should really do something with the chunk itself
                 # print $chunk;
             });
 print $res->status_line, "\n";
