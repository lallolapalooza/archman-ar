lp(4) دليل برامج التشغيل lp(4)

الاسم

lp - أجهزة الطابعة الخطية

ملخص

#include <linux/lp.h>

الإعداد

lp[0–2] هي أجهزة حرفية للطابعات الخطية المتوازية؛ لها رقم رئيسي 6 ورقم فرعي 0–2. الأرقام الفرعية تتوافق مع عناوين منافذ الطابعة الأساسية 0x03bc، 0x0378، و 0x0278. عادة ما تكون ذات وضع 220 ومملوكة للمستخدم root والمجموعة lp. يمكنك استخدام منافذ الطابعة إما مع الاستقصاء أو مع المقاطعات. يُنصح بالمقاطعات عند توقع حركة مرور عالية، على سبيل المثال، للطابعات الليزرية. بالنسبة للطابعات النقطية النموذجية، عادة ما يكون الاستقصاء كافياً. الافتراضي هو الاستقصاء.

الوصف

مكالمات ioctl(2) التالية مدعومة:

int ioctl(int fd, LPTIME, int arg)
يضبط مقدار الوقت الذي ينتظره برنامج التشغيل قبل إعادة فحص الطابعة عندما يبدو أن مخزن الطابعة ممتلئ إلى arg. إذا كان لديك طابعة سريعة، قلل هذا الرقم؛ إذا كان لديك طابعة بطيئة، فزد هذا الرقم. هذا بالـ مئة من الثانية، الافتراضي 2 هو 0.02 ثانية. يؤثر فقط على برنامج تشغيل الاستقصاء.
int ioctl(int fd, LPCHAR, int arg)
يضبط الحد الأقصى لعدد تكرارات الانتظار النشط الذي يقوم به برنامج تشغيل الاستقصاء أثناء الانتظار حتى تصبح الطابعة جاهزة لاستقبال حرف إلى arg. إذا كان الطباعة بطيئة جدًا، فزد هذا الرقم؛ إذا أصبح النظام بطيئًا جدًا، فقلل هذا الرقم. الافتراضي هو 1000. يؤثر فقط على برنامج تشغيل الاستقصاء.
int ioctl(int fd, LPABORT, int arg)
إذا كان arg هو 0، فسيعيد برنامج تشغيل الطابعة المحاولة عند حدوث أخطاء، وإلا فسيتوقف. الافتراضي هو 0.
int ioctl(int fd, LPABORTOPEN, int arg)
إذا كان arg هو 0، فسيتم إلغاء open(2) عند حدوث خطأ، وإلا فسيتم تجاهل الخطأ. الافتراضي هو تجاهله.
int ioctl(int fd, LPCAREFUL, int arg)
إذا كان arg هو 0، فيجب أن تكون إشارات نفاد الورق، وإيقاف التشغيل، والأخطاء خاطئة عند جميع عمليات الكتابة؛ وإلا فسيتم تجاهلها. الافتراضي هو تجاهلها.
int ioctl(int fd, LPWAIT, int arg)
يضبط عدد تكرارات الانتظار النشط للانتظار قبل وميض الطابعة لقبول حرف تم كتابته للتو، وعدد التكرارات للانتظار قبل إطفاء الوميض مرة أخرى، إلى arg. تقول المواصفات أن هذا الوقت يجب أن يكون 0.5 ميكروثانية، ولكن الخبرة أظهرت أن التأخير الناتج عن الكود كافٍ بالفعل. لهذا السبب، القيمة الافتراضية هي 0. هذا يستخدم لكل من برنامج تشغيل الاستقصاء وبرنامج تشغيل المقاطعات.
int ioctl(int fd, LPSETIRQ, int arg)
تتطلب هذه المكالمة ioctl(2) امتيازات المستخدم الخارق. تأخذ int يحتوي على IRQ الجديد كوسيط. كأثر جانبي، سيتم إعادة تعيين الطابعة. عندما يكون arg هو 0، سيتم استخدام برنامج تشغيل الاستقصاء، وهو الافتراضي أيضًا.
int ioctl(int fd, LPGETIRQ, int *arg)
يخزن IRQ المستخدم حاليًا في arg.
int ioctl(int fd, LPGETSTATUS, int *arg)
يخزن قيمة منفذ الحالة في arg. الأرقام الثنائية لها المعاني التالية:
LP_PBUSY إدخال مشغول معكوس، مرتفع نشط
LP_PACK إدخال إقرار غير متغير، منخفض نشط
LP_POUTPA إدخال نفاد الورق غير متغير، مرتفع نشط
LP_PSELECD إدخال محدد غير متغير، مرتفع نشط
LP_PERRORP إدخال خطأ غير متغير، منخفض نشط
ارجع إلى دليل الطابعة الخاص بك لمعرفة معنى الإشارات. لاحظ أن الأرقام الثنائية غير الموثقة قد يتم تعيينها أيضًا، اعتمادًا على الطابعة الخاصة بك.
int ioctl(int fd, LPRESET)
يعيد تعيين الطابعة. لا يتم استخدام وسيط.

الملفات
