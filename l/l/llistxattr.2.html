BUGS

The llistxattr() function can loop infinitely if the filesystem attribute list contains a symbolic link that points to itself. This is a rare and unlikely scenario.

EXAMPLES

To list all extended attributes of a file named myfile.txt:

    #include <sys/types.h>
    #include <linux/types.h>
    #include <stdio.h>
    #include <stdlib.h>
    #include <string.h>
    #include <unistd.h>
    #include <sys/xattr.h>

    int
    main(void)
    {
        char buf[1024];
        ssize_t buflen;
        char *key, *val;
        size_t keylen, vallen;

        buflen = llistxattr("myfile.txt", buf, sizeof(buf));
        if (buflen == -1) {
            perror("llistxattr");
            exit(EXIT_FAILURE);
        }

        if (buflen == 0) {
            printf("No extended attributes.");
        }

        key = buf;
        while (buflen > 0) {
            printf("%s", key); /* Print attribute name */

            /* Get attribute value */
            vallen = getxattr("myfile.txt", key, NULL, 0);
            if (vallen == -1) {
                if (errno != ENODATA) {
                    perror("getxattr (size)");
                    exit(EXIT_FAILURE);
                }
                /* Attribute might have been removed between calls */
                vallen = 0;
            }

            if (vallen > 0) {
                val = malloc(vallen);
                if (val == NULL) {
                    perror("malloc");
                    exit(EXIT_FAILURE);
                }

                if (getxattr("myfile.txt", key, val, vallen) == -1) {
                    perror("getxattr");
                    free(val);
                    exit(EXIT_FAILURE);
                }

                printf(" = %.*s", (int)vallen, val);
                free(val);
            } else if (vallen == 0) {
                printf("<no value>");
            }
            printf("\n");

            /* Forward to next attribute key. */
            keylen = strlen(key) + 1;
            buflen -= keylen;
            key += keylen;
        }

        exit(EXIT_SUCCESS);
    }

الأخطاء

يمكن للدالة llistxattr() أن تدخل في حلقة لا نهائية إذا كانت قائمة سمات نظام الملفات تحتوي على رابط رمزي يشير إلى نفسه. هذا سيناريو نادر وغير مرجح.

أمثلة

لعرض جميع السمات الموسعة لملف يسمى myfile.txt:

    #include <sys/types.h>
    #include <linux/types.h>
    #include <stdio.h>
    #include <stdlib.h>
    #include <string.h>
    #include <unistd.h>
    #include <sys/xattr.h>

    int
    main(void)
    {
        char buf[1024];
        ssize_t buflen;
        char *key, *val;
        size_t keylen, vallen;

        buflen = llistxattr("myfile.txt", buf, sizeof(buf));
        if (buflen == -1) {
            perror("llistxattr");
            exit(EXIT_FAILURE);
        }

        if (buflen == 0) {
            printf("لا توجد سمات موسعة.");
        }

        key = buf;
        while (buflen > 0) {
            printf("%s", key); /* طباعة اسم السمة */

            /* الحصول على قيمة السمة */
            vallen = getxattr("myfile.txt", key, NULL, 0);
            if (vallen == -1) {
                if (errno != ENODATA) {
                    perror("getxattr (size)");
                    exit(EXIT_FAILURE);
                }
                /* قد تكون السمة قد تمت إزالتها بين الاستدعاءات */
                vallen = 0;
            }

            if (vallen > 0) {
                val = malloc(vallen);
                if (val == NULL) {
                    perror("malloc");
                    exit(EXIT_FAILURE);
                }

                if (getxattr("myfile.txt", key, val, vallen) == -1) {
                    perror("getxattr");
                    free(val);
                    exit(EXIT_FAILURE);
                }

                printf(" = %.*s", (int)vallen, val);
                free(val);
            } else if (vallen == 0) {
                printf("<بدون قيمة>");
            }
            printf("\n");

            /* التقدم إلى مفتاح السمة التالي. */
            keylen = strlen(key) + 1;
            buflen -= keylen;
            key += keylen;
        }

        exit(EXIT_SUCCESS);
    }
