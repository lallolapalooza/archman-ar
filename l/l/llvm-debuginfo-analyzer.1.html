حالة الاختبار 5 - نطاق معجمي غير صحيح للمتغير

يتم استخدام المثال أدناه لإظهار مخرجات مختلفة تم إنشاؤها بواسطة llvm-debuginfo-analyzer. قمنا بتجميع المثال لأهداف X86 Codeview و ELF باستخدام إصدارات حديثة من Clang و GCC و MSVC (-O0 -g).

// definitions.h
#ifdef _MSC_VER
  #define forceinline __forceinline
#elif defined(__clang__)
  #if __has_attribute(__always_inline__)
    #define forceinline inline __attribute__((__always_inline__))
  #else
    #define forceinline inline
  #endif
#elif defined(__GNUC__)
  #define forceinline inline __attribute__((__always_inline__))
#else
  #define forceinline inline
  #error
#endif

نظرًا لأن الاختبار يعتمد على خيارات المترجم المضمنة، فإن الملف الرأسي أعلاه يحدد forceinline.

#include "definitions.h"
 1  #include "definitions.h"
 2  forceinline int InlineFunction(int Param) {
 3    int Var_1 = Param;
 4    {
 5      int Var_2 = Param + Var_1;
 6      Var_1 = Var_2;
 7    }
 8    return Var_1;
 9  }
10
11  int test(int Param_1, int Param_2) {
12    int A = Param_1;
13    A += InlineFunction(Param_2);
14    return A;
15  }
