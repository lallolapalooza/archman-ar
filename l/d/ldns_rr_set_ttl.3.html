ldns_rr_set_ttl(3)

الحزمة: ldns

التراخيص: مخصص: BSD

ldns(3) دليل وظائف المكتبة ldns(3)

الاسم

ldns_rr_set_owner, ldns_rr_set_ttl, ldns_rr_set_type, ldns_rr_set_rd_count, ldns_rr_set_class, ldns_rr_set_rdf - تعيين سمات ldns_rr

الملخص

#include <stdint.h>
#include <stdbool.h>

#include <ldns/ldns.h>

void ldns_rr_set_owner(ldns_rr *rr, ldns_rdf *owner);

void ldns_rr_set_ttl(ldns_rr *rr, uint32_t ttl);

void ldns_rr_set_type(ldns_rr *rr, ldns_rr_type rr_type);

void ldns_rr_set_rd_count(ldns_rr *rr, size_t count);

void ldns_rr_set_class(ldns_rr *rr, ldns_rr_class rr_class);

ldns_rdf* ldns_rr_set_rdf(ldns_rr *rr, const ldns_rdf *f, size_t position);

الوصف

ldns_rr_set_owner() تقوم بتعيين المالك في بنية rr.
*rr: rr للعمل عليه
*owner: يتم تعيينه لهذا المالك
تعيد void

ldns_rr_set_ttl() تقوم بتعيين ttl في بنية rr.
*rr: rr للعمل عليه
ttl: يتم تعيينه لهذا ttl
تعيد void

ldns_rr_set_type() تقوم بتعيين النوع في rr.
*rr: rr للعمل عليه
rr_type: يتم تعيينه لهذا النوع
تعيد void

ldns_rr_set_rd_count() تقوم بتعيين rd_count في rr.
*rr: rr للعمل عليه
count: يتم تعيينه لهذا العدد
تعيد void

ldns_rr_set_class() تقوم بتعيين الفئة في rr.
*rr: rr للعمل عليه
rr_class: يتم تعيينه لهذه الفئة
تعيد void

ldns_rr_set_rdf() تقوم بتعيين عضو rdf، وسيتم تعيينه في الموضع المحدد. يتم إرجاع القيمة القديمة، مثل pop.
*rr: rr للعمل عليه
*f: rdf المراد تعيينه
position: الموضع الذي سيتم تعيين rdf فيه
تعيد القيمة القديمة في rr، NULL عند الفشل
