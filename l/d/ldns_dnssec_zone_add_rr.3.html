ldns_dnssec_zone_add_rr(3)

الحزمة: ldns

التراخيص: custom:BSD

ldns(3) دليل وظائف المكتبة ldns(3)

الاسم

ldns_dnssec_zone_find_rrset, ldns_dnssec_zone_new, ldns_dnssec_zone_free, ldns_dnssec_zone_add_rr, ldns_dnssec_zone_names_print, ldns_dnssec_zone_print, ldns_dnssec_zone_add_empty_nonterminals - وظائف لـ ldns_dnssec_zone

الملخص

#include <stdint.h>
#include <stdbool.h>

#include <ldns/ldns.h>

ldns_dnssec_rrsets* ldns_dnssec_zone_find_rrset(const ldns_dnssec_zone *zone, const ldns_rdf *dname, ldns_rr_type type);

ldns_dnssec_zone* ldns_dnssec_zone_new(void);

void ldns_dnssec_zone_free(ldns_dnssec_zone *zone);

ldns_status ldns_dnssec_zone_add_rr(ldns_dnssec_zone *zone, ldns_rr *rr);

void ldns_dnssec_zone_names_print(FILE *out, const ldns_rbtree_t *tree, bool print_soa);

void ldns_dnssec_zone_print(FILE *out, const ldns_dnssec_zone *zone);

ldns_status ldns_dnssec_zone_add_empty_nonterminals(ldns_dnssec_zone *zone);

الوصف

ldns_dnssec_zone_find_rrset() البحث عن مجموعة سجلات الموارد (RRset) بالاسم والنوع المعطيين في المنطقة.

zone: بنية المنطقة للبحث فيها عن مجموعة سجلات الموارد (RRset).
dname: اسم النطاق لمجموعة سجلات الموارد (RRset) المراد البحث عنها.
type: نوع مجموعة سجلات الموارد (RRset) المراد البحث عنها.
تُرجع مجموعة سجلات الموارد (RRset)، أو NULL إذا لم تكن موجودة.

ldns_dnssec_zone_new() إنشاء بنية dnssec_zone جديدة.
تُرجع البنية المخصصة.

ldns_dnssec_zone_free() تحرير بنية المنطقة المعطاة، وشجرة سجلات أسماء (dnssec_names) الخاصة بها. يتم *عدم* تحرير سجلات الموارد (ldns_rr) الفردية داخل تلك الأسماء.
*zone: المنطقة المراد تحريرها.

ldns_dnssec_zone_add_rr() إضافة سجل الموارد (RR) المعطى إلى المنطقة. تبحث عما إذا كانت هناك بنية dnssec_name بالاسم الموجود. إذا كان الأمر كذلك، فأضفه إليها، وإلا فأنشئ بنية جديدة. معالجة خاصة لسجلات NSEC و RRSIG.

zone: المنطقة التي سيتم إضافة سجل الموارد (RR) إليها.
rr: سجل الموارد (RR) المراد إضافته.
تُرجع LDNS_STATUS_OK عند النجاح، ورمز خطأ بخلاف ذلك.

ldns_dnssec_zone_names_print() طباعة شجرة بنى ldns_dnssec_name إلى واصف الملف.

out: واصف الملف لطباعة الأسماء عليه.
tree: شجرة بنى ldns_dnssec_name المراد طباعتها.
print_soa: إذا كانت صحيحة، تطبع سجلات SOA، وإذا كانت خاطئة، تتجاهلها.

ldns_dnssec_zone_print() طباعة المنطقة الكاملة إلى واصف الملف المعطى.

out: واصف الملف الذي سيتم الطباعة عليه.
zone: بنية dnssec_zone المراد طباعتها.

ldns_dnssec_zone_add_empty_nonterminals() إضافة بنى dnssec_name صريحة للنهايات الفارغة (empty nonterminals) في هذه المنطقة. (هذا ضروري لتوليد NSEC3).

zone: المنطقة المراد التحقق منها من أجل النهايات الفارغة. تُرجع LDNS_STATUS_OK عند النجاح.
