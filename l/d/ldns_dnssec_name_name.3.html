ldns_dnssec_name_name(3)

الحزمة: ldns

التراخيص: custom:BSD

ldns(3) دليل وظائف المكتبة ldns(3)

الاسم

ldns_dnssec_name_new, ldns_dnssec_name_new_frm_rr, ldns_dnssec_name_free, ldns_dnssec_name_name, ldns_dnssec_name_set_name, ldns_dnssec_name_set_nsec, ldns_dnssec_name_cmp, ldns_dnssec_name_add_rr, ldns_dnssec_name_find_rrset, ldns_dnssec_name_print - وظائف لـ ldns_dnssec_name

الملخص

#include <stdint.h>
#include <stdbool.h>

#include <ldns/ldns.h>

ldns_dnssec_name* ldns_dnssec_name_new(void);

ldns_dnssec_name* ldns_dnssec_name_new_frm_rr(ldns_rr *rr);

void ldns_dnssec_name_free(ldns_dnssec_name *name);

ldns_rdf* ldns_dnssec_name_name(const ldns_dnssec_name *name);

void ldns_dnssec_name_set_name(ldns_dnssec_name *name, ldns_rdf *dname);

void ldns_dnssec_name_set_nsec(ldns_dnssec_name *name, ldns_rr *nsec);

int ldns_dnssec_name_cmp(const void *a, const void *b);

ldns_status ldns_dnssec_name_add_rr(ldns_dnssec_name *name, ldns_rr *rr);

ldns_dnssec_rrsets* ldns_dnssec_name_find_rrset(const ldns_dnssec_name *name, ldns_rr_type type);

void ldns_dnssec_name_print(FILE *out, const ldns_dnssec_name *name);

الوصف

ldns_dnssec_name_new() إنشاء بنية بيانات جديدة لاسم dnssec
تعيد البنية المخصصة

ldns_dnssec_name_new_frm_rr() إنشاء بنية بيانات جديدة لاسم dnssec لـ RR المعطى

rr: الـ RR الذي سيتم اشتقاق الخصائص منه، وإضافته إلى الاسم

ldns_dnssec_name_free() تحرير بنية الاسم والـ rrs والـ rrsets الخاصة بها. لا يتم تحرير سجلات ldns_rr الفردية الموجودة فيها

name: البنية التي سيتم تحريرها

ldns_dnssec_name_name() تعيد اسم النطاق لبنية dnssec_name المعطاة

name: بنية dnssec التي سيتم الحصول على اسم النطاق منها
تعيد اسم النطاق

ldns_dnssec_name_set_name() تحدد اسم النطاق لبنية dnssec_name المعطاة

name: بنية dnssec التي سيتم تحديد اسم النطاق الخاص بها
dname: اسم النطاق الذي سيتم تحديده به. هذه البيانات ليست منسوخة.

ldns_dnssec_name_set_nsec() تحدد NSEC(3) RR لبنية dnssec_name المعطاة

name: بنية dnssec التي سيتم تحديد اسم النطاق الخاص بها
nsec: الـ nsec rr الذي سيتم تحديده به. هذه البيانات ليست منسوخة.

ldns_dnssec_name_cmp() تقارن أسماء النطاقات للمدخلات المعطاة في ترتيبها القياسي.

a: أول dnssec_name للمقارنة
b: ثاني dnssec_name للمقارنة
تعيد -1 إذا كان اسم النطاق لـ a يأتي قبل اسم النطاق لـ b في الترتيب القياسي، 1 إذا كان العكس، و 0 إذا كانا متساويين

ldns_dnssec_name_add_rr() تُدرج الـ rr المعطى في المكان الصحيح في dnssec_name الحالية. لا يتم إجراء أي تحقق مما إذا كان الاسم مطابقًا

name: ldns_dnssec_name التي سيتم إضافة الـ RR إليها
rr: الـ RR الذي سيتم إضافته
تعيد LDNS_STATUS_OK عند النجاح، رمز خطأ عند الفشل

ldns_dnssec_name_find_rrset() البحث عن RRset بالنوع المعطى ضمن بنية الاسم هذه

name: الاسم الذي سيتم البحث عن RRset فيه
type: نوع RRset الذي سيتم البحث عنه
تعيد RRset، أو NULL إذا لم يكن موجودًا

ldns_dnssec_name_print() تطبع الـ RRs في بنية dnssec name إلى واصف الملف المعطى

out: واصف الملف الذي سيتم الطباعة إليه
name: بنية الاسم التي سيتم طباعة محتوياتها
