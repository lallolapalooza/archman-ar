ldns_update_prcount(3)

الحزمة: ldns

التراخيص: custom:BSD

ldns(3) دليل وظائف المكتبة ldns(3)

الاسم

ldns_update_zocount, ldns_update_prcount, ldns_update_upcount, ldns_update_adcount - إرجاع عدادات حزم التحديث

الاختصار

#include <stdint.h>
#include <stdbool.h>

#include <ldns/ldns.h>

uint16_t ldns_update_zocount(const ldns_pkt *p);

uint16_t ldns_update_prcount(const ldns_pkt *p);

uint16_t ldns_update_upcount(const ldns_pkt *p);


ldns_update_adcount();

p: الحزمة
يعيد عدد pr
