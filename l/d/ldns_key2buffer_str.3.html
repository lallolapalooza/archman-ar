ldns_key2buffer_str(3)

الحزمة: ldns

التراخيص: custom:BSD

ldns(3) دليل وظائف المكتبة ldns(3)

الاسم

ldns_pkt2buffer_str, ldns_pktheader2buffer_str, ldns_rr2buffer_str, ldns_rr_list2buffer_str, ldns_rdf2buffer_str, ldns_key2buffer_str, ldns_pkt2buffer_wire, ldns_rr2buffer_wire, ldns_rdf2buffer_wire, ldns_rrsig2buffer_wire, ldns_rr_rdata2buffer_wire - تحويلات المستوى الأدنى

النبذة

#include <stdint.h>
#include <stdbool.h>

#include <ldns/ldns.h>

ldns_status ldns_pkt2buffer_str(ldns_buffer *output, const ldns_pkt *pkt);

ldns_status ldns_pktheader2buffer_str(ldns_buffer *output, const ldns_pkt *pkt);

ldns_status ldns_rr2buffer_str(ldns_buffer *output, const ldns_rr *rr);

ldns_status ldns_rr_list2buffer_str(ldns_buffer *output, const ldns_rr_list *list);

ldns_status ldns_rdf2buffer_str(ldns_buffer *output, const ldns_rdf *rdf);

ldns_status ldns_key2buffer_str(ldns_buffer *output, const ldns_key *k);

ldns_status ldns_pkt2buffer_wire(ldns_buffer *output, const ldns_pkt *pkt);

ldns_status ldns_rr2buffer_wire(ldns_buffer *output, const ldns_rr *rr, int section);

ldns_status ldns_rdf2buffer_wire(ldns_buffer *output, const ldns_rdf *rdf);

ldns_status ldns_rrsig2buffer_wire(ldns_buffer *output, const ldns_rr *sigrr);

ldns_status ldns_rr_rdata2buffer_wire(ldns_buffer *output, const ldns_rr *rr);

الوصف

ldns_pkt2buffer_str() يحول البيانات في حزمة DNS إلى تنسيق العرض (كنص char *) ويضيفها إلى المخزن المؤقت المحدد

output: مؤشر إلى المخزن المؤقت لإضافة البيانات إليه
pkt: مؤشر الحزمة للتحويل
يعيد الحالة

ldns_pktheader2buffer_str() يحول رأس الحزمة إلى تنسيق العرض ويضيفه إلى المخزن المؤقت للإخراج
output: المخزن المؤقت لإلحاق الإخراج به
pkt: الحزمة التي يتم تحويل رأسها
يعيد ldns_status

ldns_rr2buffer_str() يحول البيانات في سجل الموارد إلى تنسيق العرض (كنص char *) ويضيفها إلى المخزن المؤقت المحدد. يتم تعليق تنسيق العرض لسجل DNSKEY بتعليقات توضح المعرف والنوع والحجم للمفتاح.

output: مؤشر إلى المخزن المؤقت لإضافة البيانات إليه
rr: مؤشر حقل rr للتحويل
يعيد الحالة

ldns_rr_list2buffer_str() يحول قائمة rr إلى تنسيق العرض ويضيفها إلى المخزن المؤقت للإخراج
output: المخزن المؤقت لإلحاق الإخراج به
list: قائمة ldns_rr المراد طباعتها
يعيد ldns_status

ldns_rdf2buffer_str() يحول البيانات في حقل rdata إلى تنسيق العرض (كنص char *) ويضيفها إلى المخزن المؤقت المحدد

output: مؤشر إلى المخزن المؤقت لإضافة البيانات إليه
rdf: مؤشر حقل rdafa الذي يحتوي على البيانات
يعيد الحالة

ldns_key2buffer_str() يحول البيانات في حزمة DNS إلى تنسيق العرض (كنص char *) ويضيفها إلى المخزن المؤقت المحدد

output: مؤشر إلى المخزن المؤقت لإضافة البيانات إليه
k: مؤشر المفتاح الخاص للتحويل
يعيد الحالة

ldns_pkt2buffer_wire() ينسخ بيانات الحزمة إلى المخزن المؤقت بتنسيق السلك
*output: المخزن المؤقت لإلحاق النتيجة به
*pkt: الحزمة للتحويل
يعيد ldns_status

ldns_rr2buffer_wire() ينسخ بيانات rr إلى المخزن المؤقت بتنسيق السلك
*output: المخزن المؤقت لإلحاق النتيجة به
*rr: سجل الموارد للتحويل
section: القسم في الحزمة الذي من المفترض أن يكون فيه هذا السجل (لتحديد ما إذا كان سيتم إضافة rdata أم لا)
يعيد ldns_status

ldns_rdf2buffer_wire() ينسخ بيانات rdata إلى المخزن المؤقت بتنسيق السلك
*output: المخزن المؤقت لإلحاق النتيجة به
*rdf: rdata للتحويل
يعيد ldns_status

ldns_rrsig2buffer_wire() يحول rrsig إلى تنسيق السلك ولكنه يستبعد rdata لـ rrsig. هذا مطلوب في التحقق من DNSSEC
output: المخزن المؤقت لإلحاق النتيجة به
sigrr: سجل التوقيع المراد التعامل معه
يعيد ldns_status

ldns_rr_rdata2buffer_wire() يحول rdata الخاص بـ rr إلى تنسيق السلك، مع استبعاد اسم المالك وكل الأشياء قبل rdata. هذا مطلوب في حساب علامة المفتاح DNSSEC، وحساب ds من المفتاح وربما في أماكن أخرى.

*output: المخزن المؤقت الذي سيتم وضع النتيجة فيه
*rr: rr المراد التعامل معه
يعيد ldns_status
