ldns_dnssec_rrsets_set_type(3)

الحزمة: ldns

التراخيص: مخصص: BSD

ldns(3) دليل وظائف المكتبة ldns(3)

الاسم

ldns_dnssec_rrsets_new, ldns_dnssec_rrsets_free, ldns_dnssec_rrsets_type, ldns_dnssec_rrsets_set_type, ldns_dnssec_rrsets_add_rr, ldns_dnssec_rrsets_print - وظائف لـ ldns_dnssec_rrsets

ملخص

#include <stdint.h>
#include <stdbool.h>

#include <ldns/ldns.h>

ldns_dnssec_rrsets* ldns_dnssec_rrsets_new(void);

void ldns_dnssec_rrsets_free(ldns_dnssec_rrsets *rrsets);

ldns_rr_type ldns_dnssec_rrsets_type(const ldns_dnssec_rrsets *rrsets);

ldns_status ldns_dnssec_rrsets_set_type(ldns_dnssec_rrsets *rrsets, ldns_rr_type type);

ldns_status ldns_dnssec_rrsets_add_rr(ldns_dnssec_rrsets *rrsets, ldns_rr *rr);

void ldns_dnssec_rrsets_print(FILE *out, const ldns_dnssec_rrsets *rrsets, bool follow);

الوصف

ldns_dnssec_rrsets_new() تنشئ قائمة جديدة (إدخال) من RRsets
تعيد الهيكل الذي تم تخصيصه حديثًا

ldns_dnssec_rrsets_free() تحرر قائمة rrsets و rrs الخاصة بها، ولكن *ليس* سجلات ldns_rr في المجموعات

rrsets: هيكل البيانات الذي سيتم تحريره

ldns_dnssec_rrsets_type() تعيد نوع rr الخاص بـ rrset (وهو رأس القائمة المعطاة)

rrsets: rrset للحصول على نوعه
تعيد نوع rr

ldns_dnssec_rrsets_set_type() تعيين نوع RR لـ rrset (وهو رأس القائمة المعطاة)

rrsets: rrset لتعيين نوعه
type: النوع المراد تعيينه
تعيد LDNS_STATUS_OK عند النجاح

ldns_dnssec_rrsets_add_rr() إضافة ldns_rr إلى RRset المقابل في القائمة المعطاة من RRsets. إذا لم يكن موجودًا، قم بإضافته كـ RRset جديد بسجل واحد.

rrsets: قائمة rrsets لإضافة RR إليها
rr: rr المراد إضافته إلى قائمة rrsets
تعيد LDNS_STATUS_OK عند النجاح

ldns_dnssec_rrsets_print() اطبع قائمة rrsets المعطاة إلى واصف الملف المحدد

out: واصف الملف للطباعة إليه
rrsets: قائمة RRsets للطباعة
follow: إذا تم تعيينها على false، اطبع فقط RRset الأول
