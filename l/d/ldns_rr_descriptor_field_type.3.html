ldns_rr_descriptor_field_type(3)

الحزمة: ldns

التراخيص: custom:BSD

ldns(3) دليل الوظائف المكتبية ldns(3)

الاسم

ldns_rr_descriptor, ldns_rr_descript, ldns_rr_descriptor_minimum, ldns_rr_descriptor_maximum, ldns_rr_descriptor_field_type - واصفات حقول السجل (rdata)

ملخص

#include <stdint.h>
#include <stdbool.h>

#include <ldns/ldns.h>

const ldns_rr_descriptor* ldns_rr_descript(uint16_t type);

size_t ldns_rr_descriptor_minimum(const ldns_rr_descriptor *descriptor);

size_t ldns_rr_descriptor_maximum(const ldns_rr_descriptor *descriptor);

ldns_rdf_type ldns_rr_descriptor_field_type(const ldns_rr_descriptor *descriptor, size_t field);

الوصف

ldns_rr_descriptor
يحتوي على جميع المعلومات المتعلقة بأنواع سجلات الموارد (resource record).

يحتوي هذا الهيكل، لجميع أنواع سجلات الموارد، على حقول السجلات (rdata) المعرفة.


struct ldns_struct_rr_descriptor
{
نوع سجل الموارد الذي يتم وصفه هنا:
ldns_rr_type _type;
الاسم النصي لنوع سجل الموارد.:
const char *_name;
الحد الأدنى لعدد حقول السجلات (rdata) في سجلات الموارد من هذا النوع.:
uint8_t _minimum;
الحد الأقصى لعدد حقول السجلات (rdata) في سجلات الموارد من هذا النوع.:
uint8_t _maximum;
مواصفات التنسيق السلكي للسجل، أي أنواع حقول السجلات (rdata) بترتيبها المقابل.:
const ldns_rdf_type *_wireformat;
أنواع rdf الخاصة:
ldns_rdf_type _variable;
يحدد ما إذا كان يمكن استخدام الضغط لأسماء النطاقات (dnames) في نوع سجل الموارد هذا.:
ldns_rr_compress _compress;
عدد أسماء النطاقات (DNAMEs) في سلسلة _wireformat، للتحليل.:
uint8_t _dname_count;
};
typedef struct ldns_struct_rr_descriptor ldns_rr_descriptor;

ldns_rr_descript() تُرجع واصف سجل الموارد لنوع سجل الموارد المحدد.

type: قيمة النوع لنوع سجل الموارد
تُرجع ldns_rr_descriptor لهذا النوع

ldns_rr_descriptor_minimum() تُرجع الحد الأدنى لعدد حقول السجلات (rdata) لنوع سجل الموارد الذي يصفه هذا الواصف.

descriptor: لنوع سجل موارد
تُرجع الحد الأدنى لعدد حقول السجلات (rdata)

ldns_rr_descriptor_maximum() تُرجع الحد الأقصى لعدد حقول السجلات (rdata) لنوع سجل الموارد الذي يصفه هذا الواصف.

descriptor: لنوع سجل موارد
تُرجع الحد الأقصى لعدد حقول السجلات (rdata)

ldns_rr_descriptor_field_type() تُرجع نوع rdf لرقم حقل السجل (rdata) المحدد لنوع سجل الموارد المحدد للواصف المحدد.

descriptor: لنوع سجل موارد
field: رقم الحقل
تُرجع نوع rdf للحقل
