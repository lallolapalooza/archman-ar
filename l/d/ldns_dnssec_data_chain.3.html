الوصف

ldns_dnssec_data_chain
هيكل سلسلة يحتوي على جميع بيانات DNSSEC اللازمة
للتحقق من مجموعة سجلات (rrset)
struct ldns_dnssec_data_chain_struct
{
ldns_rr_list *rrset;
ldns_rr_list *signatures;
ldns_rr_type parent_type;
ldns_dnssec_data_chain *parent;
ldns_pkt_rcode packet_rcode;
ldns_rr_type packet_qtype;
bool packet_nodata;
};

typedef struct ldns_dnssec_data_chain_struct ldns_dnssec_data_chain;

ldns_dnssec_data_chain_struct()

ldns_dnssec_trust_tree
هيكل شجرة يحتوي على علاقة بيانات DNSSEC،
وحالتها التشفيرية.

تم اشتقاق هذه الشجرة من سلسلة بيانات (data_chain)، ويمكن استخدامها
للتحقق مما إذا كانت هناك علاقة بين مجموعة سجلات (RRSET)
ومفتاح موثوق به. تحتوي الشجرة فقط على مؤشرات إلى
سلسلة البيانات (data_chain)، ولذلك يجب *عدم* تحرير() (free())
سلسلة البيانات طالما أن هناك شجرة ثقة مشتقة منها.

مثال للشجرة:

لكل توقيع هناك أصل؛ إذا كان مؤشر الأصل (parent) فارغًا (null)،
فلم يتم العثور عليه ولم يكن هناك إنكار (denial)؛ بخلاف ذلك، فهي شجرة تحتوي
إما على DNSKEY، أو DS، أو NSEC rr
struct ldns_dnssec_trust_tree_struct
{
ldns_rr *rr;
/* مجموعة السجلات الكاملة التي كان هذا السجل جزءًا منها */
ldns_rr_list *rrset;
ldns_dnssec_trust_tree *parents[LDNS_DNSSEC_TRUST_TREE_MAX_PARENTS];
ldns_status parent_status[LDNS_DNSSEC_TRUST_TREE_MAX_PARENTS];
/** لأغراض التصحيح، أضف التوقيعات أيضًا (قد تحتاج
إلى تلك إذا كانت تحتوي على أخطاء) */
ldns_rr *parent_signature[LDNS_DNSSEC_TRUST_TREE_MAX_PARENTS];
size_t parent_count;
};

typedef struct ldns_dnssec_trust_tree_struct ldns_dnssec_trust_tree;
