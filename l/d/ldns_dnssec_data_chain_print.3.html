ldns_dnssec_data_chain_print(3)

الحزمة: ldns

التراخيص: custom:BSD

ldns(3) دليل وظائف المكتبة ldns(3)

الاسم

ldns_dnssec_data_chain_new, ldns_dnssec_data_chain_free, ldns_dnssec_data_chain_deep_free, ldns_dnssec_build_data_chain, ldns_dnssec_data_chain_print - إنشاء، تدمير وطباعة ldns_chain

الملخص

#include <stdint.h>
#include <stdbool.h>

#include <ldns/ldns.h>

ldns_dnssec_data_chain* ldns_dnssec_data_chain_new(void);

void ldns_dnssec_data_chain_free(ldns_dnssec_data_chain *chain);

void ldns_dnssec_data_chain_deep_free(ldns_dnssec_data_chain *chain);

ldns_dnssec_data_chain* ldns_dnssec_build_data_chain(ldns_resolver *res, const uint16_t qflags, const ldns_rr_list *data_set, const ldns_pkt *pkt, ldns_rr *orig_rr);

void ldns_dnssec_data_chain_print(FILE *out, const ldns_dnssec_data_chain *chain);

الوصف

ldns_dnssec_data_chain_new() تنشئ بنية dnssec_chain جديدة
تُرجع ldns_dnssec_data_chain *

ldns_dnssec_data_chain_free() تحرر بنية dnssec_data_chain

*chain: السلسلة المراد تحريرها

ldns_dnssec_data_chain_deep_free() تحرر بنية dnssec_data_chain، وجميع البيانات التي تحتويها

*chain: بنية dnssec_data_chain المراد تحريرها

ldns_dnssec_build_data_chain() تبني ldns_dnssec_data_chain، والتي تحتوي على جميع بيانات DNSSEC اللازمة لاشتقاق شجرة الثقة لاحقًا

سيتم استنساخ data_set

*res: بنية المحلل الاستعلامات الإضافية المطلوبة
qflags: أعلام التحلل
*data_set: مجموعة السجلات الأصلية التي تنتهي فيها السلسلة
*pkt: اختياري، يمكن أن يحتوي على الحزمة الأصلية (وبالتالي التوقيعات وربما المفتاح)
*orig_rr: سجل الموارد الأصلي

تُرجع سلسلة بيانات DNSSEC

ldns_dnssec_data_chain_print() تطبع ldns_dnssec_data_chain إلى مجرى الملف المعطى

*out: مجرى الملف للطباعة إليه
*chain: بنية ldns_dnssec_data_chain المراد طباعتها
