ldns_dnssec_name_find_rrset(3)

الحزمة: ldns

التراخيص: custom:BSD

ldns(3) دليل وظائف المكتبة ldns(3)

الاسم

ldns_dnssec_name_new, ldns_dnssec_name_new_frm_rr, ldns_dnssec_name_free, ldns_dnssec_name_name, ldns_dnssec_name_set_name, ldns_dnssec_name_set_nsec, ldns_dnssec_name_cmp, ldns_dnssec_name_add_rr, ldns_dnssec_name_find_rrset, ldns_dnssec_name_print - وظائف لـ ldns_dnssec_name

الملخص

#include <stdint.h>
#include <stdbool.h>

#include <ldns/ldns.h>

ldns_dnssec_name* ldns_dnssec_name_new(void);

ldns_dnssec_name* ldns_dnssec_name_new_frm_rr(ldns_rr *rr);

void ldns_dnssec_name_free(ldns_dnssec_name *name);

ldns_rdf* ldns_dnssec_name_name(const ldns_dnssec_name *name);

void ldns_dnssec_name_set_name(ldns_dnssec_name *name, ldns_rdf *dname);

void ldns_dnssec_name_set_nsec(ldns_dnssec_name *name, ldns_rr *nsec);

int ldns_dnssec_name_cmp(const void *a, const void *b);

ldns_status ldns_dnssec_name_add_rr(ldns_dnssec_name *name, ldns_rr *rr);

ldns_dnssec_rrsets* ldns_dnssec_name_find_rrset(const ldns_dnssec_name *name, ldns_rr_type type);

void ldns_dnssec_name_print(FILE *out, const ldns_dnssec_name *name);

الوصف

ldns_dnssec_name_new() إنشاء بنية بيانات جديدة لاسم dnssec
تُرجع البنية المخصصة

ldns_dnssec_name_new_frm_rr() إنشاء بنية بيانات جديدة لاسم dnssec لـ RR المحدد

rr: RR لاشتقاق الخصائص منه، ولإضافته إلى الاسم

ldns_dnssec_name_free() تحرير بنية الاسم وسجلات RR ومجموعات RR الخاصة بها. لا يتم تحرير سجلات ldns_rr الفردية الموجودة بداخلها

name: البنية المراد تحريرها

ldns_dnssec_name_name() تُرجع اسم النطاق لبنية dnssec_name المعطاة

name: اسم dnssec للحصول على اسم النطاق منه
تُرجع اسم النطاق

ldns_dnssec_name_set_name() تعيين اسم النطاق لبنية dnssec_name المعطاة

name: اسم dnssec لتعيين اسم النطاق الخاص به
dname: اسم النطاق لتعيينه إليه. لا يتم نسخ هذه البيانات.

ldns_dnssec_name_set_nsec() تعيين RR NSEC(3) لبنية dnssec_name المعطاة

name: اسم dnssec لتعيين اسم النطاق الخاص به
nsec: RR nsec لتعيينه إليه. لا يتم نسخ هذه البيانات.

ldns_dnssec_name_cmp() مقارنة أسماء النطاقات للحججتين بترتيبهما القياسي.

a: أول dnssec_name للمقارنة
b: ثاني dnssec_name للمقارنة
تُرجع -1 إذا كان اسم النطاق لـ a يأتي قبل اسم النطاق لـ b في الترتيب القياسي، 1 إذا كان العكس، و 0 إذا كانا متساويين

ldns_dnssec_name_add_rr() إدراج RR المعطى في المكان الصحيح في dnssec_name الحالي. لا يتم إجراء أي فحص للتأكد من تطابق الاسم

name: ldns_dnssec_name لإضافة RR إليه
rr: RR المراد إضافته
تُرجع LDNS_STATUS_OK عند النجاح، رمز خطأ بخلاف ذلك

ldns_dnssec_name_find_rrset() البحث عن مجموعة RR بالنوع المحدد داخل بنية الاسم هذه

name: الاسم الذي سيتم البحث عن مجموعة RR فيه
type: نوع مجموعة RR المراد البحث عنها
تُرجع مجموعة RR، أو NULL إذا لم تكن موجودة

ldns_dnssec_name_print() طباعة سجلات RR في بنية اسم dnssec إلى واجهة الملف المعطاة

out: واجهة الملف المراد الطباعة إليها
name: بنية الاسم لطباعة محتوياتها
