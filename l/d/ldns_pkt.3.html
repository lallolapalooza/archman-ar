الوصف

ldns_pkt
حزمة DNS

يحتوي هذا الهيكل على حزمة DNS كاملة (سواء استعلام أو رد).

إنه التمثيل الكامل لما ترسله فعليًا إلى
خادم الأسماء، وما يعيده (بافتراض أنك العميل هنا).
struct ldns_struct_pkt
{
القسم الرأسي:
ldns_hdr *_header;
/* عناصر إضافية مطلوبة في الحزمة */
an rdf (A أو AAAA) مع عنوان IP الخاص بالخادم الذي جاءت منه:
ldns_rdf *_answerfrom;
ختم زمني لوقت إرسال الحزمة أو إنشائها:
struct timeval timestamp;
مدة الاستعلام الذي تعد هذه الحزمة ردًا عليه:
uint32_t _querytime;
حجم التنسيق السلكي للحزمة بالبايت:
size_t _size;
اختياري tsig rr:
ldns_rr *_tsig_rr;
حجم ذاكرة التخزين المؤقت المتاح لـ EDNS0، راجع RFC2671:
uint16_t _edns_udp_size;
EDNS0 Extended rcode:
uint8_t _edns_extended_rcode;
EDNS Version:
uint8_t _edns_version;
/* علم وجود OPT pseudo-RR */
uint8_t _edns_present;
بتات بيانات EDNS المحجوزة:
uint16_t _edns_z;
بيانات rdata عشوائية لـ EDNS:
ldns_rdf *_edns_data;
بيانات EDNS المهيكلة:
ldns_edns_option_list *_edns_list;
قسم الاستعلام:
ldns_rr_list *_question;
قسم الرد:
ldns_rr_list *_answer;
قسم السلطة:
ldns_rr_list *_authority;
القسم الإضافي:
ldns_rr_list *_additional;
};
typedef struct ldns_struct_pkt ldns_pkt;
