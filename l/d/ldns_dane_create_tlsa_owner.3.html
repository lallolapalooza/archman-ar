ldns_dane_create_tlsa_owner(3)

الحزمة: ldns

التراخيص: custom:BSD

ldns(3) دليل دوال المكتبة ldns(3)

الاسم

ldns_dane_create_tlsa_rr, ldns_dane_create_tlsa_owner, ldns_dane_cert2rdf, ldns_dane_select_certificate - دوال إنشاء سجلات TLSA

الملخص

#include <stdint.h>
#include <stdbool.h>

#include <ldns/ldns.h>

ldns_status ldns_dane_create_tlsa_rr(ldns_rr** tlsa, ldns_tlsa_certificate_usage certificate_usage, ldns_tlsa_selector selector, ldns_tlsa_matching_type matching_type, X509* cert);

ldns_status ldns_dane_create_tlsa_owner(ldns_rdf** tlsa_owner, const ldns_rdf* name, uint16_t port, ldns_dane_transport transport);

ldns_status ldns_dane_cert2rdf(ldns_rdf** rdf, X509* cert, ldns_tlsa_selector selector, ldns_tlsa_matching_type matching_type);

ldns_status ldns_dane_select_certificate(X509** selected_cert, X509* cert, STACK_OF(X509)* extra_certs, X509_STORE* pkix_validation_store, ldns_tlsa_certificate_usage cert_usage, int index);

الوصف

ldns_dane_create_tlsa_rr() تنشئ سجل مورد TLSA من الشهادة. لا يتم إجراء أي تحقق PKIX! تُستخدم الشهادة المقدمة كبيانات بغض النظر عن قيمة usage certificate.

tlsa: سجل مورد TLSA الذي تم إنشاؤه.
certificate_usage: القيمة لحقل استخدام الشهادة
selector: القيمة لحقل المحدد
matching_type: القيمة لنوع المطابقة
cert: الشهادة التي سيتم تمثيل بياناتها

تُرجع LDNS_STATUS_OK عند النجاح أو رمز خطأ بخلاف ذلك.

ldns_dane_create_tlsa_owner() تنشئ اسم نطاق يتكون من الاسم المعطى، مسبوقًا بمنفذ الخدمة ونوع النقل: _<EM>port</EM>._<EM>transport</EM>.<EM>name</EM>.

tlsa_owner: اسم النطاق الذي تم إنشاؤه.
name: اسم النطاق الذي يجب أن يسبقه.
port: رقم منفذ الخدمة الذي يجب إنشاء الاسم له.
transport: النقل الذي يجب إنشاء الاسم له.
تُرجع LDNS_STATUS_OK عند النجاح أو رمز خطأ بخلاف ذلك.

ldns_dane_cert2rdf() تنشئ RDF من النوع LDNS_RDF_TYPE_HEX استنادًا إلى البيانات الثنائية التي يختارها المحدد ويتم تشفيرها باستخدام نوع المطابقة.

rdf: RDF الذي تم إنشاؤه من النوع LDNS_RDF_TYPE_HEX.
cert: الشهادة التي يتم اختيار البيانات منها
selector: الشهادة الكاملة أو المفتاح العام
matching_type: البيانات الكاملة أو تجزئة SHA256 أو SHA512 للبيانات المختارة
تُرجع LDNS_STATUS_OK عند النجاح أو رمز خطأ بخلاف ذلك.

ldns_dane_select_certificate() تحدد الشهادة من cert، extra_certs أو pkix_validation_store استنادًا إلى قيمة cert_usage والفهرس.

selected_cert: الشهادة المحددة.
cert: الشهادة للتحقق منها (أو عدم التحقق منها)
extra_certs: الشهادات الوسيطة التي قد تكون ضرورية أثناء التحقق. قد تكون NULL، إلا عند استخدام الشهادة "تأكيد مرساة الثقة" لأن مرساة الثقة يجب تقديمها. (بخلاف ذلك، اختر "شهادة صادرة عن النطاق!")
pkix_validation_store: تُستخدم عندما يكون استخدام الشهادة هو "قيد شهادة CA" أو "قيد شهادة الخدمة" للتحقق من الشهادة، وفي حالة "قيد شهادة CA"، تحديد CA. عندما يكون pkix_validation_store NULL، يتم إيقاف التحقق صراحةً، ويكون السلوك عندئذٍ هو نفسه بالنسبة لـ "تأكيد مرساة الثقة" و"شهادة صادرة عن النطاق" على التوالي.
cert_usage: أي شهادة للاستخدام وكيفية التحقق.
index: يُستخدم لتحديد مرساة الثقة عندما يكون استخدام الشهادة "تأكيد مرساة الثقة". 0 هو آخر شهادة في سلسلة التحقق. 1 هي التي تسبق الأخيرة، وهكذا. عندما يكون الفهرس -1، تُستخدم آخر شهادة والتي يجب أن تكون موقعة ذاتيًا. يمكن أن يساعد ذلك في التأكد من أن مرساة الثقة المقصودة (الموقعة ذاتيًا) موجودة بالفعل في extra_certs (وهو شرط DANE).
تُرجع LDNS_STATUS_OK عند النجاح أو رمز خطأ بخلاف ذلك.
