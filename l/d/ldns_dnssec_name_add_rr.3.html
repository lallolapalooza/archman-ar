ldns_dnssec_name_add_rr(3)

الحزمة: ldns

التراخيص: custom:BSD

ldns(3) دليل وظائف المكتبة ldns(3)

الاسم

ldns_dnssec_name_new, ldns_dnssec_name_new_frm_rr, ldns_dnssec_name_free, ldns_dnssec_name_name, ldns_dnssec_name_set_name, ldns_dnssec_name_set_nsec, ldns_dnssec_name_cmp, ldns_dnssec_name_add_rr, ldns_dnssec_name_find_rrset, ldns_dnssec_name_print - وظائف لـ ldns_dnssec_name

الملخص

#include <stdint.h>
#include <stdbool.h>

#include <ldns/ldns.h>

ldns_dnssec_name* ldns_dnssec_name_new(void);

ldns_dnssec_name* ldns_dnssec_name_new_frm_rr(ldns_rr *rr);

void ldns_dnssec_name_free(ldns_dnssec_name *name);

ldns_rdf* ldns_dnssec_name_name(const ldns_dnssec_name *name);

void ldns_dnssec_name_set_name(ldns_dnssec_name *name, ldns_rdf *dname);

void ldns_dnssec_name_set_nsec(ldns_dnssec_name *name, ldns_rr *nsec);

int ldns_dnssec_name_cmp(const void *a, const void *b);

ldns_status ldns_dnssec_name_add_rr(ldns_dnssec_name *name, ldns_rr *rr);

ldns_dnssec_rrsets* ldns_dnssec_name_find_rrset(const ldns_dnssec_name *name, ldns_rr_type type);

void ldns_dnssec_name_print(FILE *out, const ldns_dnssec_name *name);

الوصف

ldns_dnssec_name_new() إنشاء هيكل بيانات جديد لاسم dnssec
يعيد الهيكل المخصص

ldns_dnssec_name_new_frm_rr() إنشاء هيكل بيانات جديد لاسم dnssec لـ RR المعطى

rr: هو RR للمشتقة منه الخصائص، ولإضافته إلى الاسم

ldns_dnssec_name_free() يحرر هيكل الاسم و rrs و rrsets الخاصة به. لا يتم تحرير سجلات ldns_rr الفردية الموجودة بداخله

name: الهيكل المراد تحريره

ldns_dnssec_name_name() يعيد اسم النطاق لهيكل dnssec_name المعطى

name: اسم dnssec للحصول على اسم النطاق منه
يعيد اسم النطاق

ldns_dnssec_name_set_name() يحدد اسم النطاق لهيكل dnssec_name المعطى

name: اسم dnssec لتحديد اسم النطاق الخاص به
dname: اسم النطاق لتحديده به. هذه البيانات *غير* منسوخة.

ldns_dnssec_name_set_nsec() يحدد RR من نوع NSEC(3) لهيكل dnssec_name المعطى

name: اسم dnssec لتحديد اسم النطاق الخاص به
nsec: RR من نوع nsec لتحديده به. هذه البيانات *غير* منسوخة.

ldns_dnssec_name_cmp() يقارن أسماء النطاقات للحجتين بترتيبها القانوني.

a: أول dnssec_name للمقارنة
b: ثاني dnssec_name للمقارنة
يعيد -1 إذا جاء اسم النطاق لـ a قبل اسم النطاق لـ b بالترتيب القانوني، 1 إذا كان العكس، و 0 إذا كانا متساويين

ldns_dnssec_name_add_rr() يدرج RR المعطى في المكان الصحيح في dnssec_name الحالي. لا يتم إجراء أي فحص للتأكد من تطابق الاسم

name: ldns_dnssec_name لإضافة RR إليه
rr: RR المراد إضافته
يعيد LDNS_STATUS_OK عند النجاح، رمز خطأ بخلاف ذلك

ldns_dnssec_name_find_rrset() يبحث عن RRset بالنوع المعطى داخل هيكل الاسم هذا

name: الاسم للبحث فيه عن RRset
type: نوع RRset المراد البحث عنه
يعيد RRset، أو NULL إذا لم يكن موجودًا

ldns_dnssec_name_print() يطبع RRs في هيكل dnssec name إلى واصف الملف المعطى

out: واصف الملف للطباعة إليه
name: هيكل الاسم لطباعة محتوياته
