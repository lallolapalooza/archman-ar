ldns_rr_owner(3)

الحزمة: ldns

التراخيص: custom:BSD

ldns(3) دليل وظائف المكتبة ldns(3)

الاسم

ldns_rr_rdf, ldns_rr_owner, ldns_rr_rd_count, ldns_rr_ttl, ldns_rr_get_class - الوصول إلى حقول بيانات الاستجابة (rdata) على ldns_rr

الملخص

#include <stdint.h>
#include <stdbool.h>

#include <ldns/ldns.h>

ldns_rdf* ldns_rr_rdf(const ldns_rr *rr, size_t nr);

ldns_rdf* ldns_rr_owner(const ldns_rr *rr);

size_t ldns_rr_rd_count(const ldns_rr *rr);

uint32_t ldns_rr_ttl(const ldns_rr *rr);

ldns_rr_class ldns_rr_get_class(const ldns_rr *rr);

الوصف

ldns_rr_rdf() تُرجع عداد عضو حقل بيانات الاستجابة (rdata).
*rr: السجل (rr) المطلوب التشغيل عليه
nr: رقم حقل بيانات الاستجابة (rdf) المطلوب إرجاعه
تُرجع ldns_rdf *

ldns_rr_owner() تُرجع اسم المالك لهيكل السجل (rr).
*rr: السجل (rr) المطلوب التشغيل عليه
تُرجع ldns_rdf *

ldns_rr_rd_count() تُرجع عدد حقول بيانات الاستجابة (rd_count) لهيكل السجل (rr).
*rr: السجل (rr) المطلوب القراءة منه
تُرجع عدد حقول بيانات الاستجابة للسجل

ldns_rr_ttl() تُرجع قيمة TTL للسجل (rr).
*rr: السجل (rr) المطلوب القراءة منه
تُرجع قيمة TTL للسجل

ldns_rr_get_class() تُرجع فئة السجل (rr).
*rr: السجل (rr) المطلوب القراءة منه
تُرجع فئة السجل
