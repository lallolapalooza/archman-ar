الوصف

ldns_buffer
تنفيذ المخازن المؤقتة لتسهيل العمليات

يمكن لمخازن ldns المؤقتة أن تحتوي على معلومات عشوائية، لكل بايت. يمكنك الكتابة
إلى نهاية المخزن المؤقت الحالية، والقراءة من الموضع الحالي،
والوصول إلى أي بيانات بداخله.

كمثال على استخدام المخازن المؤقتة، انظر إلى الشيفرة المصدرية للملف \ref host2str.c
struct ldns_struct_buffer
{
الموضع الحالي المستخدم للقراءة/الكتابة:
size_t _position;

حد القراءة/الكتابة:
size_t _limit;

كمية البيانات التي يمكن للمخزن المؤقت احتوائها:
size_t _capacity;

البيانات الموجودة في المخزن المؤقت:
uint8_t *_data;

إذا كان المخزن المؤقت ثابتًا، فلا يمكن تغيير حجمه:
unsigned _fixed : 1;

/** الحالة الحالية للمخزن المؤقت. إذا فشلت الكتابة إلى المخزن المؤقت
* لأي سبب، يتم تغيير هذه القيمة. بهذه الطريقة، يمكنك إجراء
* عمليات كتابة متعددة بالتسلسل والتحقق من النجاح لاحقًا. */
ldns_status _status;
};
typedef struct ldns_struct_buffer ldns_buffer;

ldns_buffer_new() تنشئ مخزنًا مؤقتًا جديدًا بالسعة المحددة.

capacity: الحجم (بالبايت) المخصص للمخزن المؤقت
تُرجع المخزن المؤقت الذي تم إنشاؤه

ldns_buffer_new_frm_data() تنشئ مخزنًا مؤقتًا بالبيانات المحددة. يتم نسخ البيانات وتخصيص الذاكرة. المخزن المؤقت ليس ثابتًا ويمكن تغيير حجمه باستخدام buffer_reserve().

buffer: مؤشر إلى المخزن المؤقت لوضع البيانات فيه
data: البيانات التي سيتم تضمينها في المخزن المؤقت
size: حجم البيانات

ldns_buffer_clear() تمسح المخزن المؤقت وتجعله جاهزًا للكتابة. يتم تعيين حد المخزن المؤقت إلى السعة ويتم تعيين الموضع إلى 0.
buffer: المخزن المؤقت المراد مسحه

ldns_buffer_printf() تطبع في المخزن المؤقت، مع زيادة السعة إذا لزم الأمر باستخدام buffer_reserve(). يتم تعيين موضع المخزن المؤقت إلى '\0' المنهية. تُرجع عدد الأحرف المكتوبة (باستثناء '\0' المنهية) أو -1 عند الفشل.

ldns_buffer_free() تحرر المخزن المؤقت.
*buffer: المخزن المؤقت المراد تحريره
تُرجع void

ldns_buffer_copy() تنسخ محتويات المخزن المؤقت من `from` إلى المخزن المؤقت `result` ثم تقوم بقلب المخزن المؤقت `result`. سيتم اقتطاع البيانات بصمت إذا كان المخزن المؤقت `result` صغيرًا جدًا.
*result: المخزن المؤقت الناتج الذي يتم النسخ إليه.
*from: ما يتم نسخه إلى `result`.

ldns_buffer_export() تجعل المخزن المؤقت ثابتًا وتُرجع مؤشرًا إلى البيانات. يكون المتصل مسؤولاً عن تحرير النتيجة.
*buffer: المخزن المؤقت المراد تصديره
تُرجع void

ldns_buffer_export2str() تصدر وتُرجع البيانات في المخزن المؤقت كسلسلة حرفية `char *` منتهية بحرف '\0'. يجب على المستدعي تحرير السلسلة المرجعة. يجب أن يكون المخزن المؤقت في وضع الكتابة وبالتالي لا يجب أن يكون قد تم قلبه. يصبح المخزن المؤقت ثابتًا بعد عودة هذه الدالة.

buffer: مخزن مؤقت يحتوي على بيانات من نوع `char *`
تُرجع بيانات من نوع `char *` منتهية بحرف '\0'، أو NULL عند حدوث خطأ

ldns_buffer2str() تُرجع نسخة من البيانات في المخزن المؤقت كسلسلة حرفية `char *` منتهية بحرف '\0'. يجب على المستدعي تحرير السلسلة المرجعة. يجب أن يكون المخزن المؤقت في وضع الكتابة وبالتالي لا يجب أن يكون قد تم قلبه.

buffer: مخزن مؤقت يحتوي على بيانات من نوع `char *`
تُرجع بيانات من نوع `char *` منتهية بحرف '\0'، أو NULL عند حدوث خطأ
