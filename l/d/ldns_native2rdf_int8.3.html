ldns_native2rdf_int8(3)

الحزمة: ldns

التراخيص: مخصص: BSD

ldns(3) دليل وظائف المكتبة ldns(3)

الاسم

ldns_native2rdf_int8, ldns_native2rdf_int16, ldns_native2rdf_int32, ldns_native2rdf_int16_data, ldns_rdf2native_int8, ldns_rdf2native_int16, ldns_rdf2native_int32, ldns_rdf2native_sockaddr_storage, ldns_rdf2native_time_t - وظائف تحويل الأرقام في rdf

الملخص

#include <stdint.h>
#include <stdbool.h>

#include <ldns/ldns.h>

ldns_rdf* ldns_native2rdf_int8(ldns_rdf_type type, uint8_t value);

ldns_rdf* ldns_native2rdf_int16(ldns_rdf_type type, uint16_t value);

ldns_rdf* ldns_native2rdf_int32(ldns_rdf_type type, uint32_t value);

ldns_rdf* ldns_native2rdf_int16_data(size_t size, uint8_t *data);

uint8_t ldns_rdf2native_int8(const ldns_rdf *rd);

uint16_t ldns_rdf2native_int16(const ldns_rdf *rd);

uint32_t ldns_rdf2native_int32(const ldns_rdf *rd);


ldns_rdf2native_sockaddr_storage();

time_t ldns_rdf2native_time_t(const ldns_rdf *rd);

الوصف

ldns_native2rdf_int8() تُرجع rdf التي تحتوي على تمثيل uint8_t الأصلي.
type: نوع ldns_rdf المراد استخدامه
value: قيمة uint8_t المراد استخدامها
تُرجع ldns_rdf* مع القيمة المحولة

ldns_native2rdf_int16() تُرجع rdf التي تحتوي على تمثيل uint16_t الأصلي.
type: نوع ldns_rdf المراد استخدامه
value: قيمة uint16_t المراد استخدامها
تُرجع ldns_rdf* مع القيمة المحولة

ldns_native2rdf_int32() تُرجع rdf التي تحتوي على قيمة int32 المعطاة.

نظرًا لأن أنواع rdf المتعددة يمكن أن تحتوي على int32، يجب تحديد النوع
type: نوع ldns_rdf المراد استخدامه
value: قيمة uint32_t المراد استخدامها
تُرجع ldns_rdf* مع القيمة المحولة

ldns_native2rdf_int16_data() تُرجع rdf من نوع int16_data تحتوي على البيانات الموجودة في المصفوفة المعطاة، مسبوقة بـ int16 يحدد الطول.

يتم نسخ الذاكرة، وتُرجع LDNS_RDF_TYPE_INT16DATA
size: حجم البيانات
*data: مؤشر إلى البيانات الفعلية

تُرجع ldns_rd* الـ rdf مع البيانات

ldns_rdf2native_int8() تُرجع تمثيل uint8_t الأصلي من الـ rdf.
rd: الـ ldns_rdf المراد العمل عليه
تُرجع uint8_t القيمة المستخرجة

ldns_rdf2native_int16() تُرجع تمثيل uint16_t الأصلي من الـ rdf.
rd: الـ ldns_rdf المراد العمل عليه
تُرجع uint16_t القيمة المستخرجة

ldns_rdf2native_int32() تُرجع تمثيل uint32_t الأصلي من الـ rdf.
rd: الـ ldns_rdf المراد العمل عليه
تُرجع uint32_t القيمة المستخرجة

ldns_rdf2native_sockaddr_storage()

ldns_rdf2native_time_t() تُرجع تمثيل time_t الأصلي من الـ rdf.
rd: الـ ldns_rdf المراد العمل عليه
تُرجع time_t القيمة المستخرجة (32 بت حاليًا)
