ldns_rr_list2buffer_str(3)

التراخيص: custom:BSD

#include <stdint.h>

ldns_status ldns_rr2buffer_str(ldns_buffer *output, const ldns_rr *rr);

الوصف

ldns_pkt2buffer_str() تقوم بتحويل البيانات في حزمة DNS إلى صيغة عرض (كنص char *) وتلحقها بالمخزن المؤقت المحدد.

output: مؤشر إلى المخزن المؤقت لإلحاق البيانات به.
pkt: مؤشر إلى الحزمة المطلوب تحويلها.
تعيد الحالة (status).

ldns_pktheader2buffer_str() تقوم بتحويل رأس الحزمة إلى صيغة عرض وتلحقها بالمخزن المؤقت للإخراج.
output: المخزن المؤقت لإلحاق الإخراج به.
pkt: الحزمة المطلوب تحويل رأسها.
تعيد ldns_status.

ldns_rr2buffer_str() تقوم بتحويل البيانات في سجل الموارد إلى صيغة عرض (كنص char *) وتلحقها بالمخزن المؤقت المحدد. صيغة العرض لسجل DNSKEY تكون مصحوبة بتعليقات توضح معرف المفتاح، ونوعه، وحجمه.

output: مؤشر إلى المخزن المؤقت لإلحاق البيانات به.
rr: مؤشر إلى حقل rr المطلوب تحويله.
تعيد الحالة (status).

ldns_rr_list2buffer_str() تقوم بتحويل قائمة سجلات الموارد (rr_list) إلى صيغة عرض وتلحقها بالمخزن المؤقت للإخراج.
output: المخزن المؤقت لإلحاق الإخراج به.
list: قائمة سجلات الموارد (ldns_rr_list) المطلوب طباعتها.
تعيد ldns_status.

ldns_rdf2buffer_str() تقوم بتحويل البيانات في حقل rdata إلى صيغة عرض (كنص char *) وتلحقها بالمخزن المؤقت المحدد.

output: مؤشر إلى المخزن المؤقت لإلحاق البيانات به.
rdf: مؤشر إلى حقل rdata الذي يحتوي على البيانات.
تعيد الحالة (status).

ldns_key2buffer_str() تقوم بتحويل البيانات في حزمة DNS إلى صيغة عرض (كنص char *) وتلحقها بالمخزن المؤقت المحدد.

output: مؤشر إلى المخزن المؤقت لإلحاق البيانات به.
k: مؤشر إلى المفتاح الخاص المطلوب تحويله.
تعيد الحالة (status).

ldns_pkt2buffer_wire() تنسخ بيانات الحزمة إلى المخزن المؤقت بصيغة الأسلاك (wire format).
*output: المخزن المؤقت الذي سيتم إلحاق النتيجة به.
*pkt: الحزمة المطلوب تحويلها.
تعيد ldns_status.

ldns_rr2buffer_wire() تنسخ بيانات سجل الموارد (rr) إلى المخزن المؤقت بصيغة الأسلاك.
*output: المخزن المؤقت الذي سيتم إلحاق النتيجة به.
*rr: سجل الموارد المطلوب تحويله.
section: القسم في الحزمة الذي من المفترض أن يكون فيه سجل الموارد هذا (لتحديد ما إذا كان سيتم إضافة rdata أم لا).
تعيد ldns_status.

ldns_rdf2buffer_wire() تنسخ بيانات rdata إلى المخزن المؤقت بصيغة الأسلاك.
*output: المخزن المؤقت الذي سيتم إلحاق النتيجة به.
*rdf: rdata المطلوب تحويله.
تعيد ldns_status.

ldns_rrsig2buffer_wire() تقوم بتحويل RRSIG إلى صيغة الأسلاك ولكن مع استثناء rdata الخاص بـ RRSIG. هذا مطلوب في التحقق من DNSSEC.
output: المخزن المؤقت الذي سيتم إلحاق النتيجة به.
sigrr: سجل التوقيع (signature rr) المطلوب التعامل معه.
تعيد ldns_status.

ldns_rr_rdata2buffer_wire() تقوم بتحويل rdata الخاص بسجل الموارد إلى صيغة الأسلاك، مع استثناء اسم المالك وكل الأشياء التي تسبق rdata. هذا مطلوب في حساب علامة مفتاح DNSSEC، وحساب ds من المفتاح، وربما في أماكن أخرى.

*output: المخزن المؤقت الذي سيتم وضع النتيجة فيه.
*rr: سجل الموارد المطلوب التعامل معه.
تعيد ldns_status.
