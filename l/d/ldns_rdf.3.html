ldns_rdf(3)

الحزمة: ldns

التراخيص: custom:BSD

ldns(3) دليل وظائف المكتبة ldns(3)

الاسم

ldns_rdf, ldns_rdf_type - نوع حقل rdata

الملخص

#include <stdint.h>
#include <stdbool.h>

#include <ldns/ldns.h>

الوصف

ldns_rdf
حقل بيانات السجل الموارد.

البيانات هي مصفوفة بايتات بترتيب شبكي، يتم تحديد حجمها بواسطة
حقل الحجم (16 بت). لتحليلها بشكل صحيح، استخدم النوع
المحدد في حقل النوع (16 بت) بقيمة من \ref ldns_rdf_type.
struct ldns_struct_rdf
{
حجم البيانات (بالبايتات):
size_t _size;
نوع البيانات:
ldns_rdf_type _type;
مؤشر إلى البيانات (البايتات الخام):
void *_data;
};
typedef struct ldns_struct_rdf ldns_rdf;

ldns_rdf_type
الأنواع المختلفة لحقول RDATA.
enum ldns_enum_rdf_type
{
لا شيء:
LDNS_RDF_TYPE_NONE,
اسم المجال:
LDNS_RDF_TYPE_DNAME,
8 بت:
LDNS_RDF_TYPE_INT8,
16 بت:
LDNS_RDF_TYPE_INT16,
32 بت:
LDNS_RDF_TYPE_INT32,
سجل A:
LDNS_RDF_TYPE_A,
سجل AAAA:
LDNS_RDF_TYPE_AAAA,
سلسلة نصية:
LDNS_RDF_TYPE_STR,
بيانات APL:
LDNS_RDF_TYPE_APL,
سلسلة B32:
LDNS_RDF_TYPE_B32_EXT,
سلسلة B64:
LDNS_RDF_TYPE_B64,
سلسلة سداسية عشرية:
LDNS_RDF_TYPE_HEX,
رموز أنواع NSEC:
LDNS_RDF_TYPE_NSEC,
نوع سجل A:
LDNS_RDF_TYPE_TYPE,
فئة A:
LDNS_RDF_TYPE_CLASS,
خوارزمية الشهادة:
LDNS_RDF_TYPE_CERT_ALG,
خوارزمية A:
LDNS_RDF_TYPE_ALG,
أنواع غير معروفة:
LDNS_RDF_TYPE_UNKNOWN,
وقت (32 بت):
LDNS_RDF_TYPE_TIME,
فترة:
LDNS_RDF_TYPE_PERIOD,
وقت TSIG 48 بت:
LDNS_RDF_TYPE_TSIGTIME,
/** يمثل خوارزمية المفتاح العام، وحقول HIT والمفتاح العام
* لأنواع سجلات HIP. يتم استخدام نوع rdf خاص بـ HIP بسبب
* التنسيق غير المعتاد في تنسيق السلك (انظر RFC 5205 القسم 5) */
LDNS_RDF_TYPE_HIP,
/** بيانات rdata من نوع أي متغيرة الطول حيث يتم
* تحديد الطول بواسطة البايتين الأولين */
LDNS_RDF_TYPE_INT16_DATA,
البروتوكول وخرائط المنافذ:
LDNS_RDF_TYPE_SERVICE,
بيانات الموقع:
LDNS_RDF_TYPE_LOC,
الخدمات المعروفة:
LDNS_RDF_TYPE_WKS,
NSAP:
LDNS_RDF_TYPE_NSAP,
ATMA:
LDNS_RDF_TYPE_ATMA,
IPSECKEY:
LDNS_RDF_TYPE_IPSECKEY,
ملح NSEC3:
LDNS_RDF_TYPE_NSEC3_SALT,
سلسلة NSEC3 base32 (مع بايت الطول على السلك:
LDNS_RDF_TYPE_NSEC3_NEXT_OWNER,

/** 4 أعداد صحيحة قصيرة ممثلة بـ 4 * 16 بت من الأرقام السداسية العشرية
* مفصولة بنقطتين. لـ NID و L64.
*/
LDNS_RDF_TYPE_ILNP64,

6 * 8 بت أرقام سداسية عشرية مفصولة بشرطات. لـ EUI48.:
LDNS_RDF_TYPE_EUI48,
8 * 8 بت أرقام سداسية عشرية مفصولة بشرطات. لـ EUI64.:
LDNS_RDF_TYPE_EUI64,

/** تسلسل غير صفري من الأحرف والأرقام الأبجدية الأبجدية بحالة أحرف صغيرة.
* لـ CAA.
*/
LDNS_RDF_TYPE_TAG,

/** ترميز <character-string> لقيمة الحقل كما هو محدد
* [RFC1035]، القسم 5.1.، مرمّز كـ rdata متبقي.
* لـ CAA.
*/
LDNS_RDF_TYPE_LONG_STR,

/** منذ سجلات RFC7218 TLSA يمكن تقديمها بأسماء رمزية،
* ومن هنا تأتي أنواع حقول rdata هذه. ولكن كما هو الحال مع DNSKEYs،
* يكون الإخراج دائمًا رقميًا.
*/
LDNS_RDF_TYPE_CERTIFICATE_USAGE,
LDNS_RDF_TYPE_SELECTOR,
LDNS_RDF_TYPE_MATCHING_TYPE,

draft-ietf-mboned-driad-amt-discovery *:
LDNS_RDF_TYPE_AMTRELAY,

draft-ietf-dnsop-svcb-https *:
LDNS_RDF_TYPE_SVCPARAMS,

/* أسماء مستعارة */
LDNS_RDF_TYPE_BITMAP = LDNS_RDF_TYPE_NSEC
};
typedef enum ldns_enum_rdf_type ldns_rdf_type;
