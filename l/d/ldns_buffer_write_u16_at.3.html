الوصف

ldns_buffer_write_at() تكتب البيانات المعطاة إلى المخزن المؤقت في الموضع المحدد
buffer: المخزن المؤقت
at: الموضع (بعدد البايتات) لكتابة البيانات فيه
data: مؤشر إلى البيانات المراد كتابتها في المخزن المؤقت
count: عدد البايتات من البيانات المراد كتابتها

ldns_buffer_write() تكتب عدد count من البايتات من البيانات إلى الموضع الحالي للمخزن المؤقت
buffer: المخزن المؤقت
data: البيانات المراد كتابتها
count: طول البيانات المراد كتابتها

ldns_buffer_write_string_at() تنسخ السلسلة النصية المعطاة (المنتهية بـ null) إلى الموضع المحدد في المخزن المؤقت
buffer: المخزن المؤقت
at: الموضع في المخزن المؤقت
str: السلسلة النصية المراد كتابتها

ldns_buffer_write_string() تنسخ السلسلة النصية المعطاة (المنتهية بـ null) إلى الموضع الحالي في المخزن المؤقت
buffer: المخزن المؤقت
str: السلسلة النصية المراد كتابتها

ldns_buffer_write_u8_at() تكتب البايت المعطى من البيانات في الموضع المعطى في المخزن المؤقت
buffer: المخزن المؤقت
at: الموضع في المخزن المؤقت
data: البايت الثماني المراد كتابته

ldns_buffer_write_u8() تكتب البايت المعطى من البيانات في الموضع الحالي في المخزن المؤقت
buffer: المخزن المؤقت
data: البايت الثماني المراد كتابته

ldns_buffer_write_u16_at() تكتب العدد الصحيح المكون من 2 بايت في الموضع المعطى في المخزن المؤقت
buffer: المخزن المؤقت
at: الموضع في المخزن المؤقت
data: العدد ذو الـ 16 بت المراد كتابته

ldns_buffer_write_u16() تكتب العدد الصحيح المكون من 2 بايت في الموضع الحالي في المخزن المؤقت
buffer: المخزن المؤقت

ldns_buffer_read_at() تنسخ عدد count من البايتات من البيانات في الموضع المعطى إلى مصفوفة البيانات المعطاة
buffer: المخزن المؤقت
at: الموضع في المخزن المؤقت للبدء
data: المخزن المؤقت للنسخ إليه
count: طول البيانات المراد نسخها

ldns_buffer_read() تنسخ عدد count من البايتات من البيانات في الموضع الحالي إلى مصفوفة البيانات المعطاة
buffer: المخزن المؤقت

count: طول البيانات المراد نسخها

ldns_buffer_read_u8_at() تُرجع قيمة البايت في الموضع المعطى في المخزن المؤقت

ldns_buffer_write_u16_at(3)

NAME

ldns_buffer_write_u16_at - Write a 16-bit unsigned integer at a given position

LIBRARY

ldns

SYNOPSIS

#include <stdint.h>
#include <ldns/ldns.h>

bool
ldns_buffer_write_u16_at(ldns_buffer* buffer, size_t position, uint16_t value);

DESCRIPTION

Writes a 16-bit unsigned integer of value into the buffer at the given position.

RETURN VALUE

Returns true on success, false on failure.

AUTHOR

Peter Koch <kro@methyl.daemon.nl>
