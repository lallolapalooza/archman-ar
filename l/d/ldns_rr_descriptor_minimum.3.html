ldns_rr_descriptor_minimum(3)

الحزمة: ldns

التراخيص: custom:BSD

ldns(3) دليل وظائف المكتبة ldns(3)

الاسم

ldns_rr_descriptor, ldns_rr_descript, ldns_rr_descriptor_minimum, ldns_rr_descriptor_maximum, ldns_rr_descriptor_field_type - واصفات حقول rdata

الملخص

#include <stdint.h>
#include <stdbool.h>

#include <ldns/ldns.h>

const ldns_rr_descriptor* ldns_rr_descript(uint16_t type);

size_t ldns_rr_descriptor_minimum(const ldns_rr_descriptor *descriptor);

size_t ldns_rr_descriptor_maximum(const ldns_rr_descriptor *descriptor);

ldns_rdf_type ldns_rr_descriptor_field_type(const ldns_rr_descriptor *descriptor, size_t field);

الوصف

ldns_rr_descriptor
يحتوي على جميع المعلومات حول أنواع سجلات الموارد.

يحتوي هذا الهيكل، لجميع أنواع rr، على حقول rdata التي تم تعريفها.
struct ldns_struct_rr_descriptor
{
نوع rr الذي يتم وصفه هنا:
ldns_rr_type _type;
الاسم النصي لنوع rr.
const char *_name;
الحد الأدنى لعدد حقول rdata في RRs من هذا النوع.
uint8_t _minimum;
الحد الأقصى لعدد حقول rdata في RRs من هذا النوع.
uint8_t _maximum;
مواصفات تنسيق الأسلاك لـ rr، أي أنواع حقول rdata بترتيبها.
const ldns_rdf_type *_wireformat;
أنواع rdf الخاصة:
ldns_rdf_type _variable;
يحدد ما إذا كان يمكن استخدام الضغط لـ dnames في نوع rr هذا.
ldns_rr_compress _compress;
عدد DNAMEs في سلسلة _wireformat، للتحليل.
uint8_t _dname_count;
};
typedef struct ldns_struct_rr_descriptor ldns_rr_descriptor;

ldns_rr_descript() تُرجع واصف سجل الموارد لنوع rr المحدد.

type: قيمة النوع لنوع rr
تُرجع ldns_rr_descriptor لهذا النوع

ldns_rr_descriptor_minimum() تُرجع الحد الأدنى لعدد حقول rdata لنوع rr الذي يصفه هذا الواصف.

descriptor: لنوع rr
تُرجع الحد الأدنى لعدد حقول rdata

ldns_rr_descriptor_maximum() تُرجع الحد الأقصى لعدد حقول rdata لنوع rr الذي يصفه هذا الواصف.

descriptor: لنوع rr
تُرجع الحد الأقصى لعدد حقول rdata

ldns_rr_descriptor_field_type() تُرجع نوع rdf لرقم حقل rdata المحدد لنوع rr المحدد للواصف المحدد.

descriptor: لنوع rr
field: رقم الحقل
تُرجع نوع rdf للحقل
