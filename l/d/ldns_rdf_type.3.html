ldns_rdf_type(3)

الحزمة: ldns

التراخيص: custom:BSD

ldns(3) دليل وظائف المكتبة ldns(3)

الاسم

ldns_rdf, ldns_rdf_type - نوع حقل البيانات (rdata)

الملخص

#include <stdint.h>
#include <stdbool.h>

#include <ldns/ldns.h>

الوصف

ldns_rdf
حقل بيانات السجل المورد (Resource record data field).

البيانات هي مصفوفة من البايتات بترتيب شبكي، ويتم تحديد حجمها بواسطة
الحقل (بحجم 16 بت). لتحليلها بشكل صحيح، استخدم النوع
المحدد في الحقل (بحجم 16 بت) بقيمة من \ref ldns_rdf_type.
struct ldns_struct_rdf
{
حجم البيانات (بالبايت):
size_t _size;
نوع البيانات:
ldns_rdf_type _type;
مؤشر إلى البيانات (بايتات خام):
void *_data;
};
typedef struct ldns_struct_rdf ldns_rdf;

ldns_rdf_type
الأنواع المختلفة لحقول RDATA.
enum ldns_enum_rdf_type
{
لا شيء:
LDNS_RDF_TYPE_NONE,
اسم النطاق:
LDNS_RDF_TYPE_DNAME,
8 بت:
LDNS_RDF_TYPE_INT8,
16 بت:
LDNS_RDF_TYPE_INT16,
32 بت:
LDNS_RDF_TYPE_INT32,
سجل A:
LDNS_RDF_TYPE_A,
سجل AAAA:
LDNS_RDF_TYPE_AAAA,
سلسلة نصية:
LDNS_RDF_TYPE_STR,
بيانات APL:
LDNS_RDF_TYPE_APL,
سلسلة B32:
LDNS_RDF_TYPE_B32_EXT,
سلسلة B64:
LDNS_RDF_TYPE_B64,
سلسلة سداسية عشرية:
LDNS_RDF_TYPE_HEX,
رموز أنواع NSEC:
LDNS_RDF_TYPE_NSEC,
نوع RR:
LDNS_RDF_TYPE_TYPE,
فئة:
LDNS_RDF_TYPE_CLASS,
خوارزمية الشهادة:
LDNS_RDF_TYPE_CERT_ALG,
خوارزمية:
LDNS_RDF_TYPE_ALG,
أنواع غير معروفة:
LDNS_RDF_TYPE_UNKNOWN,
وقت (32 بت):
LDNS_RDF_TYPE_TIME,
فترة:
LDNS_RDF_TYPE_PERIOD,
وقت TSIG (48 بت):
LDNS_RDF_TYPE_TSIGTIME,
/** يمثل خوارزمية المفتاح العام، حقول HIT والمفتاح العام
* لأنواع RR الخاصة بـ HIP. يتم استخدام نوع rr خاص بـ HIP بسبب
* التخطيط غير العادي في تنسيق السلك (انظر RFC 5205 القسم 5) */
LDNS_RDF_TYPE_HIP,
/** بيانات rdata من نوع أي متغير الطول حيث يتم
* تحديد الطول بواسطة البايتين الأولين */
LDNS_RDF_TYPE_INT16_DATA,
قوائم بت البروتوكول والمنفذ:
LDNS_RDF_TYPE_SERVICE,
بيانات الموقع:
LDNS_RDF_TYPE_LOC,
خدمات معروفة:
LDNS_RDF_TYPE_WKS,
NSAP:
LDNS_RDF_TYPE_NSAP,
ATMA:
LDNS_RDF_TYPE_ATMA,
IPSECKEY:
LDNS_RDF_TYPE_IPSECKEY,
ملح NSEC3:
LDNS_RDF_TYPE_NSEC3_SALT,
سلسلة B32 NSEC3 (مع بايت طول في السلك:
LDNS_RDF_TYPE_NSEC3_NEXT_OWNER,

/** 4 أرقام قصيرة ممثلة بـ 4 * 16 رقم سداسي عشري
* مفصولة بنقطتين. لـ NID و L64.
*/
LDNS_RDF_TYPE_ILNP64,

6 * 8 أرقام سداسية عشرية مفصولة بشرطات. لـ EUI48.:
LDNS_RDF_TYPE_EUI48,
8 * 8 أرقام سداسية عشرية مفصولة بشرطات. لـ EUI64.:
LDNS_RDF_TYPE_EUI64,

/** تسلسل غير صفري من الأحرف والأرقام الأبجدية الإنجليزية (US-ASCII) بأحرف صغيرة.
* لـ CAA.
*/
LDNS_RDF_TYPE_TAG,

/** ترميز <character-string> لحقل القيمة كما هو محدد
* في [RFC1035]، القسم 5.1.، مرمّز كبيانات rdata متبقية.
* لـ CAA.
*/
LDNS_RDF_TYPE_LONG_STR,

/** نظرًا لأن سجلات TLSA في RFC7218 يمكن أن تُعطى بأسماء رمزية،
* ومن ثم هذه الأنواع من حقول rdata. ولكن كما هو الحال مع DNSKEYs، فإن الإخراج
* رقمي دائمًا.
*/
LDNS_RDF_TYPE_CERTIFICATE_USAGE,
LDNS_RDF_TYPE_SELECTOR,
LDNS_RDF_TYPE_MATCHING_TYPE,

مسودة-ietf-mboned-driad-amt-discovery *:
LDNS_RDF_TYPE_AMTRELAY,

مسودة-ietf-dnsop-svcb-https *:
LDNS_RDF_TYPE_SVCPARAMS,

/* أسماء مستعارة */
LDNS_RDF_TYPE_BITMAP = LDNS_RDF_TYPE_NSEC
};
typedef enum ldns_enum_rdf_type ldns_rdf_type;
