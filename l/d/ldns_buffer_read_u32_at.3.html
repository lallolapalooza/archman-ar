الوصف

تقوم ldns_buffer_write_at() بكتابة البيانات المعطاة في المخزن المؤقت في الموضع المحدد
buffer: المخزن المؤقت
at: الموضع (بعدد البايتات) لكتابة البيانات فيه
data: مؤشر إلى البيانات المراد كتابتها في المخزن المؤقت
count: عدد البايتات من البيانات المراد كتابتها

تقوم ldns_buffer_write() بكتابة عدد count من البايتات من البيانات في الموضع الحالي للمخزن المؤقت
buffer: المخزن المؤقت
data: البيانات المراد كتابتها
count: طول البيانات المراد كتابتها

تقوم ldns_buffer_write_string_at() بنسخ السلسلة المعطاة (المحددة بنهاية فارغة) إلى الموضع المحدد في المخزن المؤقت
buffer: المخزن المؤقت
at: الموضع في المخزن المؤقت
str: السلسلة المراد كتابتها

    الاسم

    ldns_buffer_read_u32_at - قراءة 32 بت غير موقّعة من مخزن مؤقت

    الصيغة

    #include <ldns/buffer.h>

uint32_t
ldns_buffer_read_u32_at(ldns_buffer *buffer, size_t offset);

    الوصف

    تقوم هذه الدالة بقراءة 32 بت غير موقّعة من buffer بدءًا من offset .

    المعاملات

    buffer
    المخزن المؤقت للقراءة منه.
    offset
    المسافة البادئة التي يجب القراءة منها.

    القيمة المعادة

    القيمة 32 بت غير الموقّعة التي تمت قراءتها.

    مثال

    ldns_buffer *buffer = ldns_buffer_new();
ldns_buffer_write_u32(buffer, 12345);
uint32_t value = ldns_buffer_read_u32_at(buffer, 0);
/* value is 12345 */
ldns_buffer_free(buffer);

    انظر أيضًا
