ldns_rdf_new(3)

الحزمة: ldns

التراخيص: custom:BSD

ldns(3) دليل وظائف المكتبة ldns(3)

الاسم

ldns_rdf_new, ldns_rdf_clone, ldns_rdf_new_frm_data, ldns_rdf_new_frm_str, ldns_rdf_new_frm_fp, ldns_rdf_free, ldns_rdf_deep_free, ldns_rdf_print - إنشاء وتدمير وطباعة ldns_rdf

ملخص

#include <stdint.h>
#include <stdbool.h>

#include <ldns/ldns.h>

ldns_rdf* ldns_rdf_new(ldns_rdf_type type, size_t size, void *data);

ldns_rdf* ldns_rdf_clone(const ldns_rdf *rd);

ldns_rdf* ldns_rdf_new_frm_data(ldns_rdf_type type, size_t size, const void *data);

ldns_rdf* ldns_rdf_new_frm_str(ldns_rdf_type type, const char *str);

ldns_status ldns_rdf_new_frm_fp(ldns_rdf **r, ldns_rdf_type type, FILE *fp);

void ldns_rdf_free(ldns_rdf *rd);

void ldns_rdf_deep_free(ldns_rdf *rd);

void ldns_rdf_print(FILE *output, const ldns_rdf *rdf);

الوصف

ldns_rdf_new() يقوم بتخصيص بنية rdf جديدة وملئها. هذه الوظيفة لا تنسخ المحتويات من المخزن المؤقت، على عكس ldns_rdf_new_frm_data()
type: نوع الـ rdf
size: حجم المخزن المؤقت
data: مؤشر إلى المخزن المؤقت المراد نسخه
يعيد بنية الـ rdf الجديدة أو NULL في حالة الفشل

ldns_rdf_clone() ينسخ بنية rdf. يتم نسخ البيانات.
rd: الـ rdf المراد نسخه
يعيد بنية rdf جديدة

ldns_rdf_new_frm_data() يقوم بتخصيص بنية rdf جديدة وملئها. هذه الوظيفة تنسخ المحتويات من المخزن المؤقت، على عكس ldns_rdf_new()
type: نوع الـ rdf
size: حجم المخزن المؤقت
data: مؤشر إلى المخزن المؤقت المراد نسخه
يعيد بنية الـ rdf الجديدة أو NULL في حالة الفشل

ldns_rdf_new_frm_str() ينشئ rdf جديد من سلسلة نصية.
type: النوع المراد استخدامه
str: السلسلة النصية المراد استخدامها
يعيد ldns_rdf* أو NULL في حالة حدوث خطأ

ldns_rdf_new_frm_fp() ينشئ rdf جديد من ملف يحتوي على سلسلة نصية.
r: الـ rdf الجديد
type: النوع المراد استخدامه
fp: مؤشر الملف المراد استخدامه
يعيد LDNS_STATUS_OK أو الخطأ

ldns_rdf_free() يحرر بنية rdf، تاركًا مؤشر البيانات سليمًا.
rd: المؤشر المراد تحريره
يعيد void

ldns_rdf_deep_free() يحرر بنية rdf _ويحرر_ البيانات. يجب أن يتم إنشاء الـ rdf باستخدام _new_frm_data
rd: بنية الـ rdf المراد تحريرها
يعيد void

ldns_rdf_print() يطبع البيانات في حقل rdata إلى تيار الملف المحدد (بتنسيق العرض)

output: تيار الملف المراد الطباعة إليه
rdf: حقل rdata المراد طباعته
يعيد void
