ldns_verify_notime(3)

الحزمة: ldns

التراخيص: custom:BSD

ldns(3) دليل وظائف المكتبة ldns(3)

الاسم

ldns_verify, ldns_verify_rrsig, ldns_verify_rrsig_keylist, ldns_verify_rrsig_keylist_notime, ldns_verify_notime - التحقق من صحة rrsigs

الملخص

#include <stdint.h>
#include <stdbool.h>

#include <ldns/ldns.h>

ldns_status ldns_verify(ldns_rr_list *rrset, ldns_rr_list *rrsig, const ldns_rr_list *keys, ldns_rr_list *good_keys);

ldns_status ldns_verify_rrsig(ldns_rr_list *rrset, ldns_rr *rrsig, ldns_rr *key);

ldns_status ldns_verify_rrsig_keylist(ldns_rr_list *rrset, ldns_rr *rrsig, const ldns_rr_list *keys, ldns_rr_list *good_keys);

ldns_status ldns_verify_rrsig_keylist_notime(const ldns_rr_list *rrset, const ldns_rr *rrsig, const ldns_rr_list *keys, ldns_rr_list *good_keys);

ldns_status ldns_verify_notime(ldns_rr_list *rrset, ldns_rr_list *rrsig, const ldns_rr_list *keys, ldns_rr_list *good_keys);

الوصف

ldns_verify() تتحقق من صحة قائمة التوقيعات لمجموعة سجلات rrset واحدة.

rrset: مجموعة السجلات المراد التحقق منها
rrsig: قائمة بالتوقيعات للتحقق منها
keys: قائمة بالمفاتيح للتحقق باستخدامها
good_keys: إذا كانت هذه قائمة (تم تهيئتها)، فسيتم إضافة المؤشر إلى المفاتيح من المفاتيح التي تتحقق من صحة أحد التوقيعات إليها
تُرجع الحالة LDNS_STATUS_OK إذا كان هناك مفتاح صحيح واحد على الأقل

ldns_verify_rrsig() تتحقق من صحة rrsig بمفتاح واحد
rrset: مجموعة السجلات
rrsig: توقيع rrsig المراد التحقق منه
key: المفتاح المراد استخدامه
تُرجع رسالة حالة حول ما إذا كان التحقق قد نجح.

ldns_verify_rrsig_keylist() تتحقق من صحة rrsig. يتم تجربة جميع المفاتيح في مجموعة المفاتيح.
rrset: مجموعة السجلات المراد التحقق منها
rrsig: توقيع مجموعة السجلات
keys: المفاتيح المراد تجربتها
good_keys: إذا كانت هذه قائمة (تم تهيئتها)، فسيتم إضافة المؤشر إلى المفاتيح من المفاتيح التي تتحقق من صحة أحد التوقيعات إليها
تُرجع قائمة بالمفاتيح التي تتحقق من صحة rrsig + rrset. تُرجع الحالة LDNS_STATUS_OK إذا كان مفتاح واحد على الأقل متطابقًا. وإلا، خطأ.

ldns_verify_rrsig_keylist_notime() تتحقق من صحة rrsig. يتم تجربة جميع المفاتيح في مجموعة المفاتيح. لا يتم التحقق من الوقت.
rrset: مجموعة السجلات المراد التحقق منها
rrsig: توقيع مجموعة السجلات
keys: المفاتيح المراد تجربتها
good_keys: إذا كانت هذه قائمة (تم تهيئتها)، فسيتم إضافة المؤشر إلى المفاتيح من المفاتيح التي تتحقق من صحة أحد التوقيعات إليها
تُرجع قائمة بالمفاتيح التي تتحقق من صحة rrsig + rrset. تُرجع الحالة LDNS_STATUS_OK إذا كان مفتاح واحد على الأقل متطابقًا. وإلا، خطأ.

ldns_verify_notime() تتحقق من صحة قائمة التوقيعات لمجموعة سجلات rrset واحدة، ولكن مع تجاهل الوقت. لا يتم التحقق من البداية والنهاية.

rrset: مجموعة السجلات المراد التحقق منها
rrsig: قائمة بالتوقيعات للتحقق منها
keys: قائمة بالمفاتيح للتحقق باستخدامها
good_keys: إذا كانت هذه قائمة (تم تهيئتها)، فسيتم إضافة المؤشر إلى المفاتيح من المفاتيح التي تتحقق من صحة أحد التوقيعات إليها
تُرجع الحالة LDNS_STATUS_OK إذا كان هناك مفتاح صحيح واحد على الأقل
