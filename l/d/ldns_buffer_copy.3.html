الوصف

ldns_buffer
تنفيذ للمخازن المؤقتة لتسهيل العمليات

يمكن أن تحتوي مخازن ldns المؤقتة على معلومات عشوائية، لكل بايت. يمكنك الكتابة
إلى النهاية الحالية للمخزن المؤقت، والقراءة من الموضع الحالي،
والوصول إلى أي بيانات بداخله.

كمثال لاستخدام المخازن المؤقتة، انظر الشيفرة المصدرية لـ \ref host2str.c
struct ldns_struct_buffer
{
الموضع الحالي المستخدم للقراءة/الكتابة:
size_t _position;

حد القراءة/الكتابة:
size_t _limit;

مقدار البيانات التي يمكن أن يحتوي عليها المخزن المؤقت:
size_t _capacity;

البيانات الموجودة في المخزن المؤقت:
uint8_t *_data;

إذا كان المخزن المؤقت ثابتًا فلا يمكن تغيير حجمه:
unsigned _fixed : 1;

/** الحالة الحالية للمخزن المؤقت. إذا فشلت الكتابة إلى المخزن المؤقت
* لأي سبب، يتم تغيير هذه القيمة. بهذه الطريقة، يمكنك إجراء
* عمليات كتابة متعددة بالتتابع والتحقق من النجاح لاحقًا. */
ldns_status _status;
};
typedef struct ldns_struct_buffer ldns_buffer;

ldns_buffer_new() تنشئ مخزنًا مؤقتًا جديدًا بالسعة المحددة.

capacity: الحجم (بالبايت) المخصص للمخزن المؤقت
تُرجع المخزن المؤقت الذي تم إنشاؤه

ldns_buffer_new_frm_data() تنشئ مخزنًا مؤقتًا بالبيانات المحددة. يتم نسخ البيانات وتخصيص الذاكرة. المخزن المؤقت ليس ثابتًا ويمكن تغيير حجمه باستخدام buffer_reserve().

buffer: مؤشر إلى المخزن المؤقت لوضع البيانات فيه
data: البيانات المراد تغليفها في المخزن المؤقت
size: حجم البيانات

ldns_buffer_clear() تمسح المخزن المؤقت وتجعله جاهزًا للكتابة. يتم تعيين حد المخزن المؤقت إلى السعة والموضع إلى 0.
buffer: المخزن المؤقت المراد مسحه

ldns_buffer_printf() تطبع إلى المخزن المؤقت، وتزيد السعة إذا لزم الأمر باستخدام buffer_reserve(). يتم تعيين موضع المخزن المؤقت إلى '\0' النهائي. تُرجع عدد الأحرف المكتوبة (باستثناء '\0' النهائي) أو -1 في حالة الفشل.

ldns_buffer_free() تحرر المخزن المؤقت.
*buffer: المخزن المؤقت المراد تحريره
تُرجع void

ldns_buffer_copy() تنسخ محتويات المخزن المؤقت from إلى المخزن المؤقت result ثم تقلب المخزن المؤقت result. سيتم اقتطاع البيانات بصمت إذا كان المخزن المؤقت result صغيرًا جدًا.
*result: المخزن المؤقت الناتج الذي يتم النسخ إليه.
*from: ما يجب نسخه إلى result.

ldns_buffer_export() تجعل المخزن المؤقت ثابتًا وتُرجع مؤشرًا إلى البيانات. يتحمل المستدعي مسؤولية تحرير النتيجة.
*buffer: المخزن المؤقت المراد تصديره
تُرجع void

ldns_buffer_export2str() تصدر وتُرجع البيانات الموجودة في المخزن المؤقت كسلسلة نصية char * منتهية بحرف null. يجب على المستدعي تحرير السلسلة النصية المرجعة. يجب أن يكون المخزن المؤقت في وضع الكتابة وبالتالي قد لا يكون قد تم قلبه. يصبح المخزن المؤقت ثابتًا بعد عودة هذه الدالة.

buffer: المخزن المؤقت الذي يحتوي على بيانات char *
تُرجع بيانات char * منتهية بحرف null، أو NULL في حالة حدوث خطأ

ldns_buffer2str() تُرجع نسخة من البيانات الموجودة في المخزن المؤقت كسلسلة نصية char * منتهية بحرف null. يجب على المستدعي تحرير السلسلة النصية المرجعة. يجب أن يكون المخزن المؤقت في وضع الكتابة وبالتالي قد لا يكون قد تم قلبه.

buffer: المخزن المؤقت الذي يحتوي على بيانات char *
تُرجع بيانات char * منتهية بحرف null، أو NULL في حالة حدوث خطأ
