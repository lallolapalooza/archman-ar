ldns_buffer_flip(3)

الحزمة: ldns

التراخيص: custom:BSD

ldns(3) دليل وظائف المكتبة ldns(3)

الاسم

ldns_buffer_flip, ldns_buffer_rewind, ldns_buffer_position, ldns_buffer_set_position, ldns_buffer_skip - تحديد موقع المخزن المؤقت

الملخص

#include <stdint.h>
#include <stdbool.h>

#include <ldns/ldns.h>

void ldns_buffer_flip(ldns_buffer *buffer);

void ldns_buffer_rewind(ldns_buffer *buffer);

size_t ldns_buffer_position(const ldns_buffer *buffer);

void ldns_buffer_set_position(ldns_buffer *buffer, size_t mark);

void ldns_buffer_skip(ldns_buffer *buffer, ssize_t count);

الوصف

ldns_buffer_flip() تجعل المخزن المؤقت جاهزًا لقراءة البيانات التي تم كتابتها إلى المخزن المؤقت. يتم تعيين حد المخزن المؤقت إلى الموضع الحالي ويتم تعيين الموضع إلى 0.

buffer: المخزن المؤقت لقلبه
تُرجع void

ldns_buffer_rewind() تجعل المخزن المؤقت جاهزًا لإعادة قراءة البيانات. يتم إعادة تعيين موضع المخزن المؤقت إلى 0.
buffer: المخزن المؤقت لإعادة اللف

ldns_buffer_position() تُرجع الموضع الحالي في المخزن المؤقت (بعدد البايتات)
buffer: المخزن المؤقت
تُرجع الموضع الحالي

ldns_buffer_set_position() تعيين موضع المخزن المؤقت إلى MARK. يجب أن يكون الموضع أقل من أو يساوي حد المخزن المؤقت.
buffer: المخزن المؤقت
mark: العلامة المراد استخدامها

ldns_buffer_skip() يغير موضع المخزن المؤقت بمقدار COUNT بايت. يجب ألا يتم نقل الموضع خلف حد المخزن المؤقت أو قبل بداية المخزن المؤقت.
buffer: المخزن المؤقت
count: العدد المراد استخدامه
