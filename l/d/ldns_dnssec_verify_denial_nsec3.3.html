ldns_dnssec_verify_denial_nsec3(3)

الحزمة: ldns

التراخيص: custom:BSD

ldns(3) دليل وظائف المكتبة ldns(3)

الاسم

ldns_dnssec_verify_denial, ldns_dnssec_verify_denial_nsec3 - التحقق من إنكار الوجود

الملخص

#include <stdint.h>
#include <stdbool.h>

#include <ldns/ldns.h>

ldns_status ldns_dnssec_verify_denial(ldns_rr *rr, ldns_rr_list *nsecs, ldns_rr_list *rrsigs);

ldns_status ldns_dnssec_verify_denial_nsec3(ldns_rr *rr, ldns_rr_list *nsecs, ldns_rr_list *rrsigs, ldns_pkt_rcode packet_rcode, ldns_rr_type packet_qtype, bool packet_nodata);

الوصف

ldns_dnssec_verify_denial() إنكار الوجود ليس مجرد نهر في مصر

rr: سجل الموارد (الاستعلام) للتحقق من إنكار وجوده
nsecs: قائمة سجلات موارد NSEC التي يُفترض أن تنكر وجود سجل الموارد
rrsigs: سجل الموارد RRSIG الذي يغطي سجلات موارد NSEC
يعيد LDNS_STATUS_OK إذا كانت سجلات موارد NSEC تنكر الوجود، رمز خطأ يحتوي على سبب عدم قيامها بذلك بخلاف ذلك

ldns_dnssec_verify_denial_nsec3() إنكار الوجود باستخدام سجلات NSEC3. نظرًا لأن NSEC3 أكثر تعقيدًا قليلاً من الإنكار العادي، فبعض وسائط السياق مطلوبة

rr: سجل الموارد (الاستعلام) للتحقق من إنكار وجوده
nsecs: قائمة سجلات موارد NSEC3 التي يُفترض أن تنكر وجود سجل الموارد
rrsigs: سجل الموارد RRSIG الذي يغطي سجلات موارد NSEC3
packet_rcode: قيمة RCODE للحزمة التي قدمت سجلات موارد NSEC3
packet_qtype: نوع سجل الموارد الأصلي للاستعلام
packet_nodata: صحيح إذا كانت الحزمة المقدمة تحتوي على قسم إجابة فارغ
يعيد LDNS_STATUS_OK إذا كانت سجلات موارد NSEC3 تنكر الوجود، رمز خطأ يحتوي على سبب عدم قيامها بذلك بخلاف ذلك
