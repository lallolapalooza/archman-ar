ldns_dnssec_name_new(3)

الحزمة: ldns

التراخيص: custom:BSD

ldns(3) دليل وظائف المكتبة ldns(3)

الاسم

ldns_dnssec_name_new, ldns_dnssec_name_new_frm_rr, ldns_dnssec_name_free, ldns_dnssec_name_name, ldns_dnssec_name_set_name, ldns_dnssec_name_set_nsec, ldns_dnssec_name_cmp, ldns_dnssec_name_add_rr, ldns_dnssec_name_find_rrset, ldns_dnssec_name_print - وظائف لـ ldns_dnssec_name

#include <stdint.h>
#include <stdbool.h>

#include <ldns/ldns.h>

void ldns_dnssec_name_set_nsec(ldns_dnssec_name *name, ldns_rr *nsec);

int ldns_dnssec_name_cmp(const void *a, const void *b);

الوصف

ldns_dnssec_name_new() ينشئ بنية بيانات جديدة لاسم dnssec
يعيد البنية المخصصة

ldns_dnssec_name_new_frm_rr() ينشئ بنية بيانات جديدة لاسم dnssec للسجل RR المحدد

rr: السجل RR لاستخلاص الخصائص منه، وإضافته إلى الاسم

ldns_dnssec_name_free() يحرر بنية الاسم وسجلاتها ومجموعات سجلاتها. لا يتم تحرير سجلات ldns_rr الفردية الموجودة فيها

name: البنية المراد تحريرها

ldns_dnssec_name_name() يعيد اسم النطاق لبنية dnssec_name المحددة

name: اسم dnssec للحصول على اسم النطاق منه
يعيد اسم النطاق

ldns_dnssec_name_set_name() يضبط اسم النطاق لبنية dnssec_name المحددة

name: بنية dnssec لضبط اسم النطاق الخاص بها
dname: اسم النطاق المراد ضبطه. هذه البيانات *لا* يتم نسخها.

ldns_dnssec_name_set_nsec() يضبط سجل NSEC(3) لبنية dnssec_name المحددة

name: بنية dnssec لضبط اسم النطاق الخاص بها
nsec: سجل nsec المراد ضبطه. هذه البيانات *لا* يتم نسخها.

ldns_dnssec_name_cmp() يقارن أسماء النطاقات للحجتين في ترتيبها القياسي.

a: أول dnssec_name للمقارنة
b: ثاني dnssec_name للمقارنة
يعيد -1 إذا جاء اسم النطاق الخاص بـ a قبل اسم النطاق الخاص بـ b في الترتيب القياسي، 1 إذا كان العكس، و 0 إذا كانا متساويين

ldns_dnssec_name_add_rr() يدرج سجل RR المحدد في المكان الصحيح في بنية dnssec_name الحالية. لا يتم إجراء أي فحص لمعرفة ما إذا كان الاسم متطابقًا

name: ldns_dnssec_name لإضافة سجل RR إليه
rr: سجل RR المراد إضافته
يعيد LDNS_STATUS_OK عند النجاح، رمز خطأ بخلاف ذلك

ldns_dnssec_name_find_rrset() يبحث عن مجموعة سجلات RR بالنوع المحدد داخل بنية الاسم هذه

name: الاسم الذي سيتم البحث فيه عن مجموعة سجلات RR
type: نوع مجموعة سجلات RR المراد البحث عنها
يعيد مجموعة سجلات RR، أو NULL إذا لم تكن موجودة

ldns_dnssec_name_print() يطبع سجلات RR في بنية dnssec_name إلى واصف الملف المحدد

out: واصف الملف الذي سيتم الطباعة إليه
name: بنية الاسم التي سيتم طباعة محتوياتها
