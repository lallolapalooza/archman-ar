ldns_duration2string(3)

الحزمة: ldns

التراخيص: custom:BSD

ldns(3) دليل وظائف المكتبة ldns(3)

الاسم

ldns_duration_type, ldns_duration_create, ldns_duration_create_from_string, ldns_duration_cleanup, ldns_duration_compare, ldns_duration2string, ldns_duration2time - نوع المدة والوظائف ذات الصلة

الملخص

#include <stdint.h>
#include <stdbool.h>

#include <ldns/ldns.h>

ldns_duration_type* ldns_duration_create(void);

ldns_duration_type* ldns_duration_create_from_string(const char* str);

void ldns_duration_cleanup(ldns_duration_type* duration);

int ldns_duration_compare(const ldns_duration_type* d1, const ldns_duration_type* d2);

char* ldns_duration2string(const ldns_duration_type* duration);

time_t ldns_duration2time(const ldns_duration_type* duration);

الوصف

ldns_duration_type
مدة.

struct ldns_duration_struct
{
time_t years;
time_t months;
time_t weeks;
time_t days;
time_t hours;
time_t minutes;
time_t seconds;
};

typedef struct ldns_duration_struct ldns_duration_type;

ldns_duration_create() إنشاء مدة 'فورية' جديدة.
يعيد ldns_duration_type* المدة المنشأة

ldns_duration_create_from_string() إنشاء مدة من سلسلة نصية.
str: مدة بصيغة سلسلة نصية
يعيد ldns_duration_type* المدة المنشأة

ldns_duration_cleanup() تنظيف المدة.
duration: المدة المراد تنظيفها

ldns_duration_compare() مقارنة المدد.
d1: مدة واحدة
d2: مدة أخرى
يعيد int 0 إذا كانت متساوية، -1 إذا كانت d1 < d2، 1 إذا كانت d2 < d1

ldns_duration2string() تحويل مدة إلى سلسلة نصية.
duration: المدة المراد تحويلها
يعيد char* مدة بصيغة سلسلة نصية

ldns_duration2time() تحويل مدة إلى وقت.
duration: المدة المراد تحويلها
يعيد time_t مدة بصيغة وقت
