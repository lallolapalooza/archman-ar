ldap_url(3)

الحزمة: libldap

التراخيص: custom

LDAP_URL(3) دليل وظائف المكتبة LDAP_URL(3)

الاسم

ldap_is_ldap_url, ldap_url_parse, ldap_free_urldesc - إجراءات محدد موقع الموارد الموحد LDAP

المكتبة

OpenLDAP LDAP (libldap, -lldap)

الخلاصة

#include <ldap.h>
int ldap_is_ldap_url( const char *url )
int ldap_url_parse( const char *url, LDAPURLDesc **ludpp )
typedef struct ldap_url_desc {
    char *      lud_scheme;     /* مخطط URI */
    char *      lud_host;       /* مضيف LDAP للاتصال به */
    int         lud_port;       /* منفذ على المضيف */
    char *      lud_dn;         /* قاعدة للبحث */
    char **     lud_attrs;      /* قائمة السمات */
    int         lud_scope;      /* قيمة LDAP_SCOPE_... */
    char *      lud_filter;     /* مرشح بحث LDAP */
    char **     lud_exts;       /* ملحقات LDAP */
    int         lud_crit_exts;  /* صحيح إذا كانت أي ملحقات حرجة */
    /* قد تحتوي على حقول إضافية للاستخدام الداخلي */
} LDAPURLDesc;
void ldap_free_urldesc( LDAPURLDesc *ludp );

الوصف

تدعم هذه الإجراءات استخدام عناوين URL لـ LDAP (محدِّدات مواقع الموارد الموحدة) كما هو مفصل في RFC 4516. تبدو عناوين URL لـ LDAP كالتالي:

  ldap://hostport/dn[?attrs[?scope[?filter[?exts]]]]
حيث:
  hostport هو اسم مضيف مع ":portnumber" اختياري
  dn هو قاعدة البحث
  attrs هي قائمة سمات مفصولة بفواصل لطلبها
  scope هو أحد هذه الكلمات الثلاث:
    base one sub (افتراضي=base)
  filter هو المرشح
  exts هي مجموعة معترف بها من ملحقات LDAP و/أو API.
مثال:
  ldap://ldap.example.net/dc=example,dc=net?cn,sn?sub?(cn=*)

يتم أيضًا التسامح مع عناوين URL المغلفة بأقواس زاوية و/أو مسبوقة بـ "URL:". قد يتم تحليل مخططات LDAP البديلة مثل ldaps:// و ldapi:// باستخدام الإجراءات أدناه أيضًا.

ldap_is_ldap_url() تعيد قيمة غير صفرية إذا كان url يبدو كعنوان URL لـ LDAP (على عكس نوع آخر من عناوين URL). يمكن استخدامه كفحص سريع لعنوان URL لـ LDAP؛ يجب استخدام إجراء ldap_url_parse() إذا كانت هناك حاجة إلى فحص أكثر شمولاً.

ldap_url_parse() يفكك عنوان URL لـ LDAP المقدم في url إلى أجزائه المكونة. إذا نجح، يتم إرجاع صفر، ويتم تخصيص وصف عنوان URL لـ LDAP، وملؤه، ويتم تعيين ludpp ليشير إليه. إذا حدث خطأ، يتم إرجاع رمز خطأ عنوان URL غير صفري.

يجب استدعاء ldap_free_urldesc() لتحرير وصف عنوان URL لـ LDAP الذي تم الحصول عليه من استدعاء ldap_url_parse().
