ldns_rr_print(3)

الحزمة: ldns

التراخيص: custom:BSD

ldns(3) دليل وظائف المكتبة ldns(3)

الاسم

ldns_rr_new, ldns_rr_new_frm_type, ldns_rr_new_frm_str, ldns_rr_new_frm_fp, ldns_rr_free, ldns_rr_print - إنشاء، حذف وطباعة ldns_rr

الملخص

#include <stdint.h>
#include <stdbool.h>

#include <ldns/ldns.h>

ldns_rr* ldns_rr_new(void);

ldns_rr* ldns_rr_new_frm_type(ldns_rr_type t);

ldns_status ldns_rr_new_frm_str(ldns_rr **n, const char *str, uint32_t default_ttl, const ldns_rdf *origin, ldns_rdf **prev);

ldns_status ldns_rr_new_frm_fp(ldns_rr **rr, FILE *fp, uint32_t *default_ttl, ldns_rdf **origin, ldns_rdf **prev);

void ldns_rr_free(ldns_rr *rr);

void ldns_rr_print(FILE *output, const ldns_rr *rr);

الوصف

ldns_rr_new() تنشئ بنية rr جديدة.
تُرجع ldns_rr *

ldns_rr_new_frm_type() تنشئ بنية rr جديدة، بناءً على النوع المحدد. تخصص مساحة كافية لاحتواء جميع الأجزاء.

ldns_rr_new_frm_str() تنشئ rr من سلسلة نصية. يجب أن تكون السلسلة النصية rr مكتملة، مثل اسم المالك &lt;مسافة&gt; TTL &lt;مسافة&gt; CLASS &lt;مسافة&gt; TYPE &lt;مسافة&gt; RDATA.
n: الـ rr المراد إرجاعه
str: السلسلة النصية للتحويل
default_ttl: قيمة TTL الافتراضية للـ rr. إذا كانت 0، فسيتم استخدام DEF_TTL
origin: عند ما يكون المالك نسبيًا، قم بإضافة هذا. يجب على المستدعي استدعاء ldns_rdf_deep_free له.
prev: اسم المالك السابق. إذا كانت هذه القيمة NULL، فإن الدالة تستبدل هذه القيمة باسم المالك الموجود في هذه السلسلة النصية. يجب على المستدعي استدعاء ldns_rdf_deep_free له.
تُرجع رسالة حالة تصف خطأ أو LDNS_STATUS_OK

ldns_rr_new_frm_fp() تنشئ rr جديدة من ملف يحتوي على سلسلة نصية.
rr: الـ rr الجديد
fp: مؤشر الملف المراد استخدامه
default_ttl: مؤشر إلى TTL افتراضي للـ rr. إذا كانت NULL، فسيتم استخدام DEF_TTL وسيتم تحديث المؤشر إذا كان الملف يحتوي على توجيه $TTL
origin: عند ما يكون المالك نسبيًا، قم بإضافة هذا وسيتم تحديث المؤشر إذا كان الملف يحتوي على توجيه $ORIGIN. يجب على المستدعي استدعاء ldns_rdf_deep_free له.
prev: عند ما يكون المالك عبارة عن مسافات بيضاء، استخدم هذا كـ * اسم المالك وسيتم تحديث المؤشر بعد الاستدعاء. يجب على المستدعي استدعاء ldns_rdf_deep_free له.
تُرجع ldns_status مع خطأ أو LDNS_STATUS_OK

ldns_rr_free() تحرر بنية RR
*rr: الـ RR المراد تحريره
تُرجع void

ldns_rr_print() تطبع البيانات في سجل الموارد إلى تدفق الملف المحدد (بالتنسيق التقديمي)

output: تدفق الملف للطباعة عليه
rr: سجل الموارد المراد طباعته
تُرجع void
