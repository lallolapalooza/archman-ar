ldns_wire2pkt(3)

الحزمة: ldns

التراخيص: custom:BSD

ldns(3) دليل وظائف المكتبة ldns(3)

الاسم

ldns_wire2rr, ldns_wire2pkt, ldns_wire2rdf, ldns_wire2dname - تحويل من تنسيق سلكي إلى نوع مضيف

الملخص

#include <stdint.h>
#include <stdbool.h>

#include <ldns/ldns.h>

ldns_status ldns_wire2rr(ldns_rr **rr, const uint8_t *wire, size_t max, size_t *pos, ldns_pkt_section section);

ldns_status ldns_wire2pkt(ldns_pkt **packet, const uint8_t *data, size_t len);

ldns_status ldns_wire2rdf(ldns_rr *rr, const uint8_t *wire, size_t max, size_t *pos);

ldns_status ldns_wire2dname(ldns_rdf **dname, const uint8_t *wire, size_t max, size_t *pos);

الوصف

تقوم ldns_wire2rr() بتحويل البيانات الموجودة على مصفوفة البايت uint8_t (بتنسيق سلكي) إلى سجل مورد DNS. ستقوم هذه الدالة بتهيئة وتخصيص مساحة للذاكرة لهيكل rr. تتم إضافة طول بيانات السلك لهذا rr إلى القيمة *pos.

rr: مؤشر إلى الهيكل الذي سيحتوي على قيمة rdata
wire: مؤشر إلى المخزن المؤقت بالبيانات
max: طول المخزن المؤقت للبيانات (بالبايت)
pos: موضع الـ rr في المخزن المؤقت (أي عدد البايتات من بداية المخزن المؤقت)
section: القسم في الحزمة الذي يُقصد به الـ rr
تُرجع LDNS_STATUS_OK إذا نجح كل شيء، خطأ بخلاف ذلك

تقوم ldns_wire2pkt() بتحويل البيانات الموجودة على مصفوفة البايت uint8_t (بتنسيق سلكي) إلى حزمة DNS. ستقوم هذه الدالة بتهيئة وتخصيص مساحة للذاكرة لهيكل الحزمة.

packet: مؤشر إلى الهيكل الذي سيحتوي على الحزمة
data: مؤشر إلى المخزن المؤقت بالبيانات
len: طول المخزن المؤقت للبيانات (بالبايت)
تُرجع LDNS_STATUS_OK إذا نجح كل شيء، خطأ بخلاف ذلك

تقوم ldns_wire2rdf() بتحويل البيانات الموجودة على مصفوفة البايت uint8_t (بتنسيق سلكي) إلى حقول rdata في DNS، وإضافتها إلى قائمة rdfs الخاصة بالـ rr المعطى. ستقوم هذه الدالة بتهيئة وتخصيص مساحة للذاكرة لهياكل dname. تتم إضافة طول بيانات السلك لهذه الـ rdfs إلى القيمة *pos.

يتم قراءة جميع الـ rdfs التي تنتمي إلى الـ RR؛ يجب ألا يحتوي الـ rr على أي rdfs بعد. يتم إرجاع خطأ إذا لم يكن بالإمكان تحليل التنسيق.

rr: مؤشر إلى هيكل ldns_rr الذي سيحتوي على قيمة rdata
wire: مؤشر إلى المخزن المؤقت بالبيانات
max: طول المخزن المؤقت للبيانات (بالبايت)
pos: موضع الـ rdf في المخزن المؤقت (أي عدد البايتات من بداية المخزن المؤقت)
تُرجع LDNS_STATUS_OK إذا نجح كل شيء، خطأ بخلاف ذلك

تقوم ldns_wire2dname() بتحويل البيانات الموجودة على مصفوفة البايت uint8_t (بتنسيق سلكي) إلى حقل rdata dname في DNS. ستقوم هذه الدالة بتهيئة وتخصيص مساحة للذاكرة لهيكل dname. تتم إضافة طول بيانات السلك لهذا الـ rdf إلى القيمة *pos.

dname: مؤشر إلى الهيكل الذي سيحتوي على قيمة rdata
wire: مؤشر إلى المخزن المؤقت بالبيانات
max: طول المخزن المؤقت للبيانات (بالبايت)
pos: موضع الـ rdf في المخزن المؤقت (أي عدد البايتات من بداية المخزن المؤقت)
تُرجع LDNS_STATUS_OK إذا نجح كل شيء، خطأ بخلاف ذلك
