ldns_pkt_new(3)

الحزمة: ldns

التراخيص: custom:BSD

ldns(3) دليل وظائف المكتبة ldns(3)

الاسم

ldns_pkt_new, ldns_pkt_free, ldns_pkt_print, ldns_pkt_query_new, ldns_pkt_query_new_frm_str, ldns_pkt_reply_type - إنشاء، تدمير وطباعة حزم ldns_pkt

الملخص

#include <stdint.h>
#include <stdbool.h>

#include <ldns/ldns.h>

ldns_pkt* ldns_pkt_new(void);

void ldns_pkt_free(ldns_pkt *packet);

void ldns_pkt_print(FILE *output, const ldns_pkt *pkt);

ldns_pkt* ldns_pkt_query_new(ldns_rdf *rr_name, ldns_rr_type rr_type, ldns_rr_class rr_class, uint16_t flags);

ldns_status ldns_pkt_query_new_frm_str(ldns_pkt **p, const char *rr_name, ldns_rr_type rr_type, ldns_rr_class rr_class , uint16_t flags);

ldns_pkt_type ldns_pkt_reply_type(const ldns_pkt *p);

الوصف

ldns_pkt_new() تقوم بتخصيص وتهيئة بنية ldns_pkt.
ترجع مؤشرًا إلى الحزمة الجديدة.

ldns_pkt_free() تقوم بتحرير بنية الحزمة وجميع البيانات التي تحتوي عليها.
packet: بنية الحزمة المراد تحريرها.
ترجع void (لا شيء).

ldns_pkt_print() تطبع البيانات في حزمة DNS إلى مجرى الملف المعطى (بالتنسيق التقديمي).

output: مجرى الملف الذي سيتم الطباعة إليه.
pkt: الحزمة المراد طباعتها.
ترجع void (لا شيء).

ldns_pkt_query_new() تنشئ حزمة تحتوي على استعلام بالاسم والنوع والفئة المعطاة.
rr_name: الاسم المراد الاستعلام عنه (لا يتم نسخه). الحزمة المُرجعة ستتولى ملكية rr_name، لذا يجب على المستدعي عدم تحريرها.
rr_type: النوع المراد الاستعلام عنه.
rr_class: الفئة المراد الاستعلام عنها.
flags: أعلام الحزمة.
ترجع ldns_pkt* مؤشرًا إلى الحزمة الجديدة.

ldns_pkt_query_new_frm_str() تنشئ حزمة استعلام بالاسم والنوع والفئة المعطاة.
p: الحزمة المراد إرجاعها.
rr_name: الاسم المراد الاستعلام عنه (كسلسلة نصية).
rr_type: النوع المراد الاستعلام عنه.
rr_class: الفئة المراد الاستعلام عنها.
flags: أعلام الحزمة.
ترجع LDNS_STATUS_OK أو رسالة ldns_status مع الخطأ.

ldns_pkt_reply_type() تبحث داخل الحزمة لتحديد نوعها، مثل AUTH، NXDOMAIN، REFERRAL، إلخ.
p: الحزمة المراد فحصها.
ترجع نوع الحزمة.
