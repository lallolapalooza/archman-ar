ldns_bskipcs(3)

الحزمة: ldns

التراخيص: custom:BSD

ldns(3) دليل الوظائف المكتبية ldns(3)

الاسم

ldns_bget_token, ldns_bgetc, ldns_bskipcs - الحصول على رموز من المخازن المؤقتة

الموجز

#include <stdint.h>
#include <stdbool.h>

#include <ldns/ldns.h>

ssize_t ldns_bget_token(ldns_buffer *b, char *token, const char *delim, size_t limit);

int ldns_bgetc(ldns_buffer *buffer);

void ldns_bskipcs(ldns_buffer *buffer, const char *s);

الوصف

تقوم ldns_bget_token() بإرجاع رمز/حرف من المخزن المؤقت b. تتعامل هذه الدالة مع ( و ) في المخزن المؤقت، وتتجاهلها عند العثور عليها.
*b: المخزن المؤقت للقراءة منه
*token: يتم وضع الرمز هنا
*delim: الأحرف التي يجب أن يتوقف التحليل عندها
*limit: مقدار القراءة. إذا كان 0، يتم استخدام الحد الأقصى المدمج
ترجع 0 عند حدوث خطأ أو الوصول إلى نهاية الملف (EOF) للمخزن المؤقت b. بخلاف ذلك، ترجع طول ما تمت قراءته

تقوم ldns_bgetc() بإرجاع الحرف التالي من المخزن المؤقت. تقدم مؤشر الموضع بمقدار 1. عند الوصول إلى نهاية المخزن المؤقت، ترجع EOF. هذا هو المكافئ للدالة getc() للمخزن المؤقت.
*buffer: المخزن المؤقت للقراءة منه
ترجع EOF عند الفشل، بخلاف ذلك ترجع الحرف

تقوم ldns_bskipcs() بتخطي جميع الأحرف الموجودة في السلسلة المعطاة في المخزن المؤقت، وتنقل الموضع إلى أول حرف غير موجود في *s.
*buffer: المخزن المؤقت للاستخدام
*s: الأحرف المطلوب تخطيها
ترجع void
