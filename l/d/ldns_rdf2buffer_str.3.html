ldns_rdf2buffer_str(3)

#include <stdint.h>

ldns_status ldns_rr2buffer_str(ldns_buffer *output, const ldns_rr *rr);

ldns_status ldns_rr_rdata2buffer_wire(ldns_buffer *output, const ldns_rr *rr);

الوصف

ldns_pkt2buffer_str() تحول البيانات في حزمة DNS إلى صيغة العرض (كسلسلة نصية) وتلحقها بالمخزن المؤقت المحدد.

output: مؤشر إلى المخزن المؤقت لإلحاق البيانات به
pkt: المؤشر إلى الحزمة للتحويل
تُرجع الحالة

ldns_pktheader2buffer_str() تحول رأس الحزمة إلى صيغة العرض وتلحقها بالمخزن المؤقت للمخرجات.
output: المخزن المؤقت لإلحاق المخرجات به
pkt: الحزمة التي يتم تحويل رأسها
تُرجع ldns_status

ldns_rr2buffer_str() تحول البيانات في سجل الموارد إلى صيغة العرض (كسلسلة نصية) وتلحقها بالمخزن المؤقت المحدد. صيغة العرض لسجل DNSKEY موصوفة بتعليقات توضح المعرف والنوع والحجم للمفتاح.

output: مؤشر إلى المخزن المؤقت لإلحاق البيانات به
rr: المؤشر إلى حقل rr المراد تحويله
تُرجع الحالة

ldns_rr_list2buffer_str() تحول قائمة rr إلى صيغة العرض وتلحقها بالمخزن المؤقت للمخرجات.
output: المخزن المؤقت لإلحاق المخرجات به
list: قائمة ldns_rr المراد طباعتها
تُرجع ldns_status

ldns_rdf2buffer_str() تحول البيانات في حقل rdata إلى صيغة العرض (كسلسلة نصية) وتلحقها بالمخزن المؤقت المحدد.

output: مؤشر إلى المخزن المؤقت لإلحاق البيانات به
rdf: المؤشر إلى حقل rdafa الذي يحتوي على البيانات
تُرجع الحالة

ldns_key2buffer_str() تحول البيانات في حزمة DNS إلى صيغة العرض (كسلسلة نصية) وتلحقها بالمخزن المؤقت المحدد.

output: مؤشر إلى المخزن المؤقت لإلحاق البيانات به
k: المؤشر إلى المفتاح الخاص المراد تحويله
تُرجع الحالة

ldns_pkt2buffer_wire() تنسخ بيانات الحزمة إلى المخزن المؤقت بصيغة الشبكة.
*output: المخزن المؤقت لإلحاق النتيجة به
*pkt: الحزمة المراد تحويلها
تُرجع ldns_status

ldns_rr2buffer_wire() تنسخ بيانات rr إلى المخزن المؤقت بصيغة الشبكة.
*output: المخزن المؤقت لإلحاق النتيجة به
*rr: سجل المورد المراد تحويله
section: القسم في الحزمة الذي يُفترض أن يكون فيه هذا السجل (لتحديد ما إذا كان سيتم إضافة rdata أم لا).
تُرجع ldns_status

ldns_rdf2buffer_wire() تنسخ بيانات rdata إلى المخزن المؤقت بصيغة الشبكة.
*output: المخزن المؤقت لإلحاق النتيجة به
*rdf: rdata المراد تحويله
تُرجع ldns_status

ldns_rrsig2buffer_wire() تحول rrsig إلى صيغة الشبكة ولكن تستبعد rdata الخاصة بـ rrsig. هذا مطلوب في التحقق من DNSSEC.
output: المخزن المؤقت لإلحاق النتيجة به
sigrr: سجل التوقيع المراد العمل عليه
تُرجع ldns_status

ldns_rr_rdata2buffer_wire() تحول rdata لسجل rr إلى صيغة الشبكة، مع استبعاد اسم المالك وكل الأشياء التي تسبق rdata. هذا مطلوب في حساب مفتاح العلامة DNSSEC، وحساب ds من المفتاح، وربما في أماكن أخرى.

*output: المخزن المؤقت لوضع النتيجة فيه
*rr: rr المراد العمل عليه
تُرجع ldns_status
