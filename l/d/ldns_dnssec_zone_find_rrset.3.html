ldns_dnssec_zone_find_rrset(3)

الحزمة: ldns

التراخيص: custom:BSD

ldns(3) دليل وظائف المكتبة ldns(3)

الاسم

ldns_dnssec_zone_find_rrset, ldns_dnssec_zone_new, ldns_dnssec_zone_free, ldns_dnssec_zone_add_rr, ldns_dnssec_zone_names_print, ldns_dnssec_zone_print, ldns_dnssec_zone_add_empty_nonterminals - وظائف لـ ldns_dnssec_zone

الملخص

#include <stdint.h>
#include <stdbool.h>

#include <ldns/ldns.h>

ldns_dnssec_rrsets* ldns_dnssec_zone_find_rrset(const ldns_dnssec_zone *zone, const ldns_rdf *dname, ldns_rr_type type);

ldns_dnssec_zone* ldns_dnssec_zone_new(void);

void ldns_dnssec_zone_free(ldns_dnssec_zone *zone);

ldns_status ldns_dnssec_zone_add_rr(ldns_dnssec_zone *zone, ldns_rr *rr);

void ldns_dnssec_zone_names_print(FILE *out, const ldns_rbtree_t *tree, bool print_soa);

void ldns_dnssec_zone_print(FILE *out, const ldns_dnssec_zone *zone);

ldns_status ldns_dnssec_zone_add_empty_nonterminals(ldns_dnssec_zone *zone);

الوصف

ldns_dnssec_zone_find_rrset() البحث عن مجموعة سجلات الموارد (RRset) بالاسم والنوع المعطيين في المنطقة

zone: هيكل المنطقة للبحث فيها عن مجموعة سجلات الموارد (RRset)
dname: اسم النطاق لمجموعة سجلات الموارد (RRset) المراد العثور عليها
type: نوع مجموعة سجلات الموارد (RRset) المراد العثور عليها
إرجاع مجموعة سجلات الموارد (RRset)، أو NULL إذا لم تكن موجودة

ldns_dnssec_zone_new() إنشاء هيكل dnssec_zone جديد
إرجاع الهيكل المخصص

ldns_dnssec_zone_free() تحرير هيكل المنطقة المعطى، وشجرة أشجارها (rbtree) لأسماء dnssec_names. سجلات الموارد (RR) الفردية ضمن هذه الأسماء *لا* يتم تحريرها.
*zone: المنطقة المراد تحريرها

ldns_dnssec_zone_add_rr() إضافة سجل الموارد (RR) المعطى إلى المنطقة. تبحث فيما إذا كان هناك اسم dnssec_name بنفس الاسم موجودًا. إذا كان كذلك، قم بإضافته إليه، وإلا فقم بإنشاء اسم جديد. معالجة خاصة لـ NSEC و RRSIG متوفرة

zone: المنطقة التي سيتم إضافة سجل الموارد (RR) إليها
rr: سجل الموارد (RR) المراد إضافته
إرجاع LDNS_STATUS_OK عند النجاح، ورمز خطأ بخلاف ذلك

ldns_dnssec_zone_names_print() طباعة شجرة أشجار (rbtree) لهياكل ldns_dnssec_name إلى واصف الملف

out: واصف الملف لطباعة الأسماء عليه
tree: شجرة أشجار (rbtree) لهياكل ldns_dnssec_name للطباعة
print_soa: إذا كانت صحيحة، اطبع سجلات SOA، وإذا كانت خاطئة، فتجاوزها

ldns_dnssec_zone_print() طباعة المنطقة الكاملة إلى واصف الملف المعطى

out: واصف الملف للطباعة عليه
zone: منطقة dnssec_zone للطباعة

ldns_dnssec_zone_add_empty_nonterminals() إضافة هياكل dnssec_name صريحة لـ nonterminals الفارغة في هذه المنطقة. (هذا ضروري لإنشاء NSEC3)

zone: المنطقة للتحقق من nonterminals الفارغة. إرجاع LDNS_STATUS_OK عند النجاح.
