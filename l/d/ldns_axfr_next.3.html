ldns_axfr_next(3)

الحزمة: ldns

التراخيص: custom:BSD

ldns(3) دليل وظائف المكتبة ldns(3)

الاسم

ldns_axfr_start, ldns_axfr_next, ldns_axfr_abort, ldns_axfr_complete, ldns_axfr_last_pkt - وظائف لنقل المنطقة بالكامل

الملخص

#include <stdint.h>
#include <stdbool.h>

#include <ldns/ldns.h>


ldns_axfr_start();

ldns_rr* ldns_axfr_next(ldns_resolver *resolver);

void ldns_axfr_abort(ldns_resolver *resolver);

bool ldns_axfr_complete(const ldns_resolver *resolver);

ldns_pkt* ldns_axfr_last_pkt(const ldns_resolver *res);

الوصف

ldns_axfr_start()

ldns_axfr_next() الحصول على الدفق التالي من سجلات الموارد (RRs) في AXFR
resolver: المحلل لاستخدامه. يجب استدعاء ldns_axfr_start() أولاً.
يعيد ldns_rr سجل الموارد التالي من دفق AXFR. بعد الحصول على سجل الموارد الذي تم إرجاعه (ليس NULL: عند حدوث خطأ)، تحقق مما إذا كانت ldns_axfr_complete() صحيحة لمعرفة ما إذا كان نقل المنطقة قد اكتمل.

ldns_axfr_abort() إلغاء نقل قيد التقدم.
resolver: المحلل المستخدم.

ldns_axfr_complete() تعيد صحيحًا إذا اكتمل نقل axfr (أي تم العثور على سجلات SOA 2 ولم يتم مواجهة أخطاء.
resolver: المحلل المستخدم.
يعيد bool صحيحًا إذا اكتمل نقل axfr بدون خطأ.

ldns_axfr_last_pkt() يعيد مؤشرًا إلى آخر ldns_pkt تم إرساله بواسطة الخادم في نقل AXFR، وهو مفيد على سبيل المثال للحصول على رمز الخطأ عند الفشل.
res: المحلل الذي تم استخدامه في نقل axfr.
يعيد ldns_pkt آخر حزمة تم إرسالها.
