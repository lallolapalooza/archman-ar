ldns_verify_rrsig_dsa(3)

الحزمة: ldns

التراخيص: مخصص:BSD

ldns(3) دليل وظائف المكتبة ldns(3)

الاسم

ldns_verify_rrsig_dsa, ldns_verify_rrsig_rsasha1, ldns_verify_rrsig_rsamd5 - التحقق من صحة مخازن بيانات التوقيع

الملخص

#include <stdint.h>
#include <stdbool.h>

#include <ldns/ldns.h>

ldns_status ldns_verify_rrsig_dsa(ldns_buffer *sig, ldns_buffer *rrset, ldns_buffer *key);

ldns_status ldns_verify_rrsig_rsasha1(ldns_buffer *sig, ldns_buffer *rrset, ldns_buffer *key);

ldns_status ldns_verify_rrsig_rsamd5(ldns_buffer *sig, ldns_buffer *rrset, ldns_buffer *key);

الوصف

تقوم الدالة ldns_verify_rrsig_dsa() بالتحقق من صحة مخزن بيانات يحتوي على بيانات التوقيع (DSA) لمخزن بيانات يحتوي على بيانات مجموعة السجلات (rrset) مع مخزن بيانات يحتوي على بيانات المفتاح.

sig: بيانات التوقيع
rrset: بيانات مجموعة السجلات، مرتبة ومعالجة للتحقق
key: بيانات المفتاح

تقوم الدالة ldns_verify_rrsig_rsasha1() بالتحقق من صحة مخزن بيانات يحتوي على بيانات التوقيع (RSASHA1) لمخزن بيانات يحتوي على بيانات مجموعة السجلات (rrset) مع مخزن بيانات يحتوي على بيانات المفتاح.

sig: بيانات التوقيع
rrset: بيانات مجموعة السجلات، مرتبة ومعالجة للتحقق
key: بيانات المفتاح

تقوم الدالة ldns_verify_rrsig_rsamd5() بالتحقق من صحة مخزن بيانات يحتوي على بيانات التوقيع (RSAMD5) لمخزن بيانات يحتوي على بيانات مجموعة السجلات (rrset) مع مخزن بيانات يحتوي على بيانات المفتاح.

sig: بيانات التوقيع
rrset: بيانات مجموعة السجلات، مرتبة ومعالجة للتحقق
key: بيانات المفتاح
