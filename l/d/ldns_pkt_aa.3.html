الوصف

ldns_pkt_id()

ldns_pkt_qr() قراءة بت الاستعلام/الاستجابة (qr) للحزمة
p: الحزمة
القيمة المرجعة: قيمة البت

ldns_pkt_aa() قراءة بت المصادقة (aa) للحزمة
p: الحزمة
القيمة المرجعة: قيمة البت

ldns_pkt_tc() قراءة بت القطع (tc) للحزمة
p: الحزمة
القيمة المرجعة: قيمة البت

ldns_pkt_rd() قراءة بت التكرار المطلوب (rd) للحزمة
p: الحزمة
القيمة المرجعة: قيمة البت

ldns_pkt_cd() قراءة بت التحقق من تعطيل (cd) للحزمة
p: الحزمة
القيمة المرجعة: قيمة البت

ldns_pkt_ra() قراءة بت التكرار متاح (ra) للحزمة
p: الحزمة
القيمة المرجعة: قيمة البت

ldns_pkt_ad() قراءة بت المصادقة متاحة (ad) للحزمة
p: الحزمة
القيمة المرجعة: قيمة البت

ldns_pkt_get_opcode() قراءة رمز العملية (opcode) للحزمة
p: الحزمة
القيمة المرجعة: رمز العملية

ldns_pkt_get_rcode() إرجاع رمز الاستجابة (rcode) للحزمة
p: الحزمة
القيمة المرجعة: رمز الاستجابة

ldns_pkt_qdcount() إرجاع عدد استعلامات qd في الحزمة
p: الحزمة
القيمة المرجعة: عدد qd

ldns_pkt_ancount() إرجاع عدد الإجابات an في الحزمة
p: الحزمة
القيمة المرجعة: عدد an

ldns_pkt_nscount() إرجاع عدد خوادم الأسماء ns في الحزمة
p: الحزمة
القيمة المرجعة: عدد ns

ldns_pkt_arcount() إرجاع عدد السجلات الإضافية ar في الحزمة
p: الحزمة
القيمة المرجعة: عدد ar

ldns_pkt_answerfrom() إرجاع معلومة المصدر للحزمة
p: الحزمة
القيمة المرجعة: اسم الخادم

ldns_pkt_querytime() إرجاع وقت الاستعلام للحزمة
p: الحزمة
القيمة المرجعة: وقت الاستعلام

ldns_pkt_size() إرجاع حجم الحزمة بالبايت
p: الحزمة
القيمة المرجعة: الحجم

ldns_pkt_tsig() إرجاع سجل TSIG الزائف للحزمة
p: الحزمة
القيمة المرجعة: سجل TSIG

ldns_pkt_question() إرجاع قسم الأسئلة للحزمة
p: الحزمة
القيمة المرجعة: القسم

ldns_pkt_answer() إرجاع قسم الإجابات للحزمة
p: الحزمة
القيمة المرجعة: القسم

ldns_pkt_authority() إرجاع قسم السلطة للحزمة
p: الحزمة
القيمة المرجعة: القسم

ldns_pkt_additional() إرجاع القسم الإضافي للحزمة
p: الحزمة
القيمة المرجعة: القسم

ldns_pkt_get_section_clone() إرجاع جميع قوائم السجلات (rr_list) في الحزمة. يتم نسخ القوائم بدلاً من إرجاع مؤشرات.
p: الحزمة للبحث فيها
s: القسم (الأقسام) المطلوب إرجاعها
القيمة المرجعة: ldns_rr_list بالسجلات أو NULL إذا لم يتم العثور على أي شيء

ldns_pkt_rr_list_by_name() إرجاع جميع السجلات (rr) باسم معين من حزمة. يمكن اختياريًا تحديد القسم في الحزمة
p: الحزمة
r: الاسم
s: قسم الحزمة
القيمة المرجعة: قائمة بالسجلات أو NULL إذا لم يتم العثور على أي شيء

ldns_pkt_rr_list_by_type() إرجاع جميع السجلات (rr) بنوع معين من حزمة. يمكن اختياريًا تحديد القسم في الحزمة
p: الحزمة
t: النوع
s: قسم الحزمة
القيمة المرجعة: قائمة بالسجلات أو NULL إذا لم يتم العثور على أي شيء

ldns_pkt_rr_list_by_name_and_type() إرجاع جميع السجلات (rr) بنوع واسم معين من حزمة. يمكن اختياريًا تحديد القسم في الحزمة
packet: الحزمة
ownername: الاسم
type: النوع
sec: قسم الحزمة
القيمة المرجعة: قائمة بالسجلات أو NULL إذا لم يتم العثور على أي شيء
