ldns_key_new(3)

الحزمة: ldns

التراخيص: custom:BSD

ldns(3) دليل وظائف المكتبة ldns(3)

الاسم

ldns_key_new, ldns_key_new_frm_algorithm, ldns_key_new_frm_fp, ldns_key_new_frm_fp_l, ldns_key_new_frm_fp_rsa, ldns_key_new_frm_fp_rsa_l, ldns_key_new_frm_fp_dsa, ldns_key_new_frm_fp_dsa_l - إنشاء مفتاح ldns_key

الملخص

#include <stdint.h>
#include <stdbool.h>

#include <ldns/ldns.h>

ldns_key* ldns_key_new(void);

ldns_key* ldns_key_new_frm_algorithm(ldns_signing_algorithm a, uint16_t size);

ldns_status ldns_key_new_frm_fp(ldns_key **k, FILE *fp);

ldns_status ldns_key_new_frm_fp_l(ldns_key **k, FILE *fp, int *line_nr);

RSA* ldns_key_new_frm_fp_rsa(FILE *fp);

RSA* ldns_key_new_frm_fp_rsa_l(FILE *fp, int *line_nr);

DSA* ldns_key_new_frm_fp_dsa(FILE *fp);

DSA* ldns_key_new_frm_fp_dsa_l(FILE *fp, int *line_nr);

الوصف

ldns_key_new() ينشئ هيكل مفتاح فارغ جديد
يعيد هيكل ldns_key * جديد

ldns_key_new_frm_algorithm() ينشئ مفتاحًا جديدًا بناءً على الخوارزمية

a: الخوارزمية المراد استخدامها
size: عدد البايتات لحجم المفتاح
يعيد هيكل ldns_key جديدًا مع المفتاح

ldns_key_new_frm_fp() ينشئ مفتاحًا خاصًا جديدًا بناءً على محتويات الملف المشار إليه بواسطة fp.

يجب أن يكون الملف بتنسيق Private-key-format v1.x.

k: هيكل ldns_key الجديد
fp: مؤشر الملف المراد استخدامه
يعيد خطأ أو LDNS_STATUS_OK

ldns_key_new_frm_fp_l() ينشئ مفتاحًا خاصًا جديدًا بناءً على محتويات الملف المشار إليه بواسطة fp

يجب أن يكون الملف بتنسيق Private-key-format v1.x.

k: هيكل ldns_key الجديد
fp: مؤشر الملف المراد استخدامه
line_nr: مؤشر إلى عدد صحيح يحتوي على رقم السطر الحالي (لأغراض التصحيح)
يعيد خطأ أو LDNS_STATUS_OK

ldns_key_new_frm_fp_rsa() وظيفة مساعدة frm_fp. تقوم هذه الوظيفة بتحليل باقي ملف المفتاح الخاص (RSA) الذي تم إنشاؤه من bind9
fp: الملف المراد تحليله
يعيد NULL عند الفشل، وإلا هيكل RSA

ldns_key_new_frm_fp_rsa_l() وظيفة مساعدة frm_fp. تقوم هذه الوظيفة بتحليل باقي ملف المفتاح الخاص (RSA) الذي تم إنشاؤه من bind9
fp: الملف المراد تحليله
line_nr: مؤشر إلى عدد صحيح يحتوي على رقم السطر الحالي (لأغراض التصحيح)
يعيد NULL عند الفشل، وإلا هيكل RSA

ldns_key_new_frm_fp_dsa() وظيفة مساعدة frm_fp. تقوم هذه الوظيفة بتحليل باقي ملف المفتاح الخاص (DSA)
fp: الملف المراد تحليله
يعيد NULL عند الفشل، وإلا هيكل RSA

ldns_key_new_frm_fp_dsa_l() وظيفة مساعدة frm_fp. تقوم هذه الوظيفة بتحليل باقي ملف المفتاح الخاص (DSA)
fp: الملف المراد تحليله
line_nr: مؤشر إلى عدد صحيح يحتوي على رقم السطر الحالي (لأغراض التصحيح)
يعيد NULL عند الفشل، وإلا هيكل RSA
