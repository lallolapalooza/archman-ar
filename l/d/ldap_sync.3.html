ldap_sync(3)

الحزمة: libldap

التراخيص: custom

LDAP_SYNC(3) دليل دوال المكتبة LDAP_SYNC(3)

الاسم

ldap_sync_init, ldap_sync_init_refresh_only, ldap_sync_init_refresh_and_persist, ldap_sync_poll - روتينيات مزامنة LDAP

المكتبة

OpenLDAP LDAP (libldap, -lldap)

الملخص

#include <ldap.h>
int ldap_sync_init(ldap_sync_t *ls, int mode);
int ldap_sync_init_refresh_only(ldap_sync_t *ls);
int ldap_sync_init_refresh_and_persist(ldap_sync_t *ls);
int ldap_sync_poll(ldap_sync_t *ls);
ldap_sync_t * ldap_sync_initialize(ldap_sync_t *ls);
void ldap_sync_destroy(ldap_sync_t *ls, int freeit);

typedef int (*ldap_sync_search_reference_f)(ldap_sync_t *ls,

الوصف

توفر هذه الروتينيات واجهة لعملية مزامنة محتوى LDAP (RFC 4533). تتطلب بنية ldap_sync_t ليتم إعدادها بمعلمات مطلوبة لمراحل مختلفة من العملية؛ يشمل ذلك تعيين بعض المعالجات للأحداث الخاصة. جميع المعالجات تأخذ مؤشرًا إلى بنية ldap_sync_t كوسيط أول، ومؤشرًا إلى بنية LDAPMessage كما تم استلامها من الخادم بواسطة مكتبة العميل، بالإضافة إلى وسائط أخرى محددة في بعض الأحيان.

أعضاء بنية ldap_sync_t هم:

char *ls_base
قاعدة البحث؛ افتراضيًا، الخيار BASE في ldap.conf(5).
int ls_scope
نطاق البحث (أحد خيارات LDAP_SCOPE_BASE، LDAP_SCOPE_ONELEVEL، LDAP_SCOPE_SUBORDINATE أو LDAP_SCOPE_SUBTREE؛ انظر ldap.h للتفاصيل).
char *ls_filter
المرشح (RFC 4515)؛ افتراضيًا، (objectClass=*).
char **ls_attrs
السمات المطلوبة؛ افتراضيًا NULL، مما يشير إلى جميع سمات المستخدم.
int ls_timelimit
حد الوقت المطلوب (بالثواني)؛ افتراضيًا 0، للإشارة إلى عدم وجود حد.
int ls_sizelimit
حد الحجم المطلوب (بالإدخالات)؛ افتراضيًا 0، للإشارة إلى عدم وجود حد.
int ls_timeout
المهلة الزمنية المرغوبة أثناء الاستقصاء باستخدام ldap_sync_poll(3). قيمة -1 تعني أن الاستقصاء يحظر، لذا لن تعود ldap_sync_poll(3) حتى يتم تلقي رسالة؛ قيمة 0 تعني أن الاستقصاء يعود فورًا، بغض النظر عما إذا كان هناك أي استجابة متاحة أم لا؛ قيمة موجبة تمثل المهلة الزمنية التي ستنتظرها الدالة ldap_sync_poll(3) للاستجابة قبل العودة، ما لم يتم تلقي رسالة؛ في هذه الحالة، تعود ldap_sync_poll(3) بمجرد توفر الرسالة.
ldap_sync_search_entry_f ls_search_entry
دالة يتم استدعاؤها عند إرجاع إدخال. الوسيط msg هو LDAPMessage الذي يحتوي على searchResultEntry؛ يمكن تحليله باستخدام روتينيات واجهة برمجة تطبيقات العميل العادية، مثل ldap_get_dn(3)، ldap_first_attribute(3)، وهكذا. يحتوي الوسيط entryUUID على entryUUID للإدخال. يحدد الوسيط phase نوع العملية: أحد خيارات LDAP_SYNC_CAPI_PRESENT، LDAP_SYNC_CAPI_ADD، LDAP_SYNC_CAPI_MODIFY، LDAP_SYNC_CAPI_DELETE؛ في حالة LDAP_SYNC_CAPI_PRESENT أو LDAP_SYNC_CAPI_DELETE، يتم احتواء DN فقط في LDAPMessage؛ في حالة LDAP_SYNC_CAPI_MODIFY، يتم احتواء الإدخال بالكامل في LDAPMessage، ويكون التطبيق مسؤولاً عن تحديد الاختلافات بين العرض الجديد للإدخال المقدم من المتصل والبيانات المعروفة بالفعل.
ldap_sync_search_reference_f ls_search_reference
دالة يتم استدعاؤها عند إرجاع مرجع بحث. الوسيط msg هو LDAPMessage الذي يحتوي على searchResultReference؛ يمكن تحليله باستخدام روتينيات واجهة برمجة تطبيقات العميل العادية، مثل ldap_parse_reference(3).
ldap_sync_intermediate_f ls_intermediate
دالة يتم استدعاؤها عند حدوث شيء ذي صلة أثناء مرحلة تحديث البحث، والتي يتم تمييزها برسالة من نوع intermediateResponse. الوسيط msg هو LDAPMessage الذي يحتوي على الاستجابة الوسيطة؛ يمكن تحليله باستخدام روتينيات واجهة برمجة تطبيقات العميل العادية، مثل ldap_parse_intermediate(3). يحتوي الوسيط syncUUIDs على مصفوفة من UUIDs للإدخالات التي تعتمد على قيمة الوسيط phase. في حالة LDAP_SYNC_CAPI_PRESENTS، يتم الدخول إلى مرحلة "presents"؛ هذا يعني أن التسلسل التالي من النتائج سيتكون من إدخالات في حالة مزامنة "present". في حالة LDAP_SYNC_CAPI_DELETES، يتم الدخول إلى مرحلة "deletes"؛ هذا يعني أن التسلسل التالي من النتائج سيتكون من إدخالات في حالة مزامنة "delete". في حالة LDAP_SYNC_CAPI_PRESENTS_IDSET، تحتوي الرسالة على مجموعة من UUIDs للإدخالات الموجودة؛ فهي تحل محل مرحلة "presents". في حالة LDAP_SYNC_CAPI_DELETES_IDSET، تحتوي الرسالة على مجموعة من UUIDs للإدخالات التي تم حذفها؛ فهي تحل محل مرحلة "deletes". في حالة LDAP_SYNC_CAPI_DONE، تم إرجاع مرحلة "presents" مع تعيين "refreshDone" إلى "TRUE" للإشارة إلى أن مرحلة التحديث لـ refreshAndPersist قد انتهت، ويجب على العميل البدء في الاستقصاء. باستثناء حالات LDAP_SYNC_CAPI_PRESENTS_IDSET و LDAP_SYNC_CAPI_DELETES_IDSET، يكون syncUUIDs هو NULL.
ldap_sync_search_result_f ls_search_result
دالة يتم استدعاؤها عند إرجاع searchResultDone. في refreshAndPersist، يمكن أن يحدث هذا فقط عندما يقرر الخادم أن البحث يجب أن يتوقف. الوسيط msg هو LDAPMessage الذي يحتوي على الاستجابة؛ يمكن تحليله باستخدام روتينيات واجهة برمجة تطبيقات العميل العادية، مثل ldap_parse_result(3). الوسيط refreshDeletes ليس ذا صلة في هذه الحالة؛ يجب أن يكون دائمًا -1.
void *ls_private
مؤشر إلى بيانات خاصة. يمكن للعميل تسجيل مؤشر هنا إلى البيانات التي قد تحتاجها المعالجات المذكورة أعلاه.
LDAP *ls_ld
مؤشر إلى بنية LDAP يتم استخدامها للاتصال بالخادم. تقع على عاتق العميل مسؤولية تهيئة البنية وتوفير المصادقة والأمان المناسبين.
