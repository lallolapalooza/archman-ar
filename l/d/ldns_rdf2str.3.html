ldns_rdf2str(3)

الحزمة: ldns

التراخيص: custom:BSD

ldns(3) دليل وظائف المكتبة ldns(3)

الاسم

ldns_rr2str, ldns_pkt2str, ldns_rdf2str, ldns_rr_list2str, ldns_key2str - وظائف للتحويل إلى سلسلة نصية

الملخص

#include <stdint.h>
#include <stdbool.h>

#include <ldns/ldns.h>

char* ldns_rr2str(const ldns_rr *rr);

char* ldns_pkt2str(const ldns_pkt *pkt);

char* ldns_rdf2str(const ldns_rdf *rdf);

char* ldns_rr_list2str(const ldns_rr_list *rr_list);

char* ldns_key2str(const ldns_key *k);

الوصف

ldns_rr2str() تقوم بتحويل البيانات في سجل الموارد إلى تنسيق العرض وتعيدها كسلسلة نصية (char *). تذكر تحريرها.

rr: حقل بيانات السجل المراد تحويله.
يعيد بيانات سلسلة نصية (char *) منتهية بصفر، أو NULL في حالة حدوث خطأ.

ldns_pkt2str() تقوم بتحويل البيانات في حزمة DNS إلى تنسيق العرض وتعيدها كسلسلة نصية (char *). تذكر تحريرها.

pkt: حزمة DNS المراد تحويلها.
يعيد بيانات سلسلة نصية (char *) منتهية بصفر، أو NULL في حالة حدوث خطأ.

ldns_rdf2str() تقوم بتحويل البيانات في حقل بيانات السجل إلى تنسيق العرض وتعيدها كسلسلة نصية (char *). تذكر تحريرها.

rdf: حقل بيانات السجل المراد تحويله.
يعيد بيانات سلسلة نصية (char *) منتهية بصفر، أو NULL في حالة حدوث خطأ.

ldns_rr_list2str() تقوم بتحويل قائمة سجلات الموارد إلى تنسيق العرض وتعيدها كسلسلة نصية (char *). تذكر تحريرها.

rr_list: قائمة سجلات الموارد المراد تحويلها إلى نص.
يعيد بيانات سلسلة نصية (char *) منتهية بصفر، أو NULL في حالة حدوث خطأ.

ldns_key2str() تقوم بتحويل مفتاح خاص إلى تنسيق العرض النصي وتعيده كسلسلة نصية (char *). تذكر تحريرها.

k: المفتاح المراد تحويله إلى نص.
يعيد بيانات سلسلة نصية (char *) منتهية بصفر، أو NULL في حالة حدوث خطأ.
