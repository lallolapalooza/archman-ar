ldns_get_rr_list_addr_by_name(3)

الحزمة: ldns

التراخيص: custom:BSD

ldns(3) دليل دوال المكتبة ldns(3)

الاسم

ldns_get_rr_list_addr_by_name, ldns_get_rr_list_name_by_addr - الحصول على عناوين بالاسم أو أسماء بالعنوان

المُلخَّص

#include <stdint.h>
#include <stdbool.h>

#include <ldns/ldns.h>

ldns_rr_list* ldns_get_rr_list_addr_by_name(ldns_resolver *r, const ldns_rdf *name, ldns_rr_class c, uint16_t flags);

ldns_rr_list* ldns_get_rr_list_name_by_addr(ldns_resolver *r, const ldns_rdf *addr, ldns_rr_class c, uint16_t flags);

الوصف

ldns_get_rr_list_addr_by_name() اطلب من المحل الاستعلام عن الاسم وارجع جميع سجلات العناوين
r: المحل المراد استخدامه
name: الاسم المراد البحث عنه
c: الفئة المراد استخدامها
flags: أعط بعض العلامات الاختيارية للاستعلام

ldns_get_rr_list_name_by_addr() اطلب من المحل الاستعلام عن العنوان وارجع الاسم
r: المحل المراد استخدامه
addr: العنوان المراد البحث عنه
c: الفئة المراد استخدامها
flags: أعط بعض العلامات الاختيارية للاستعلام
