ldns_zone_print_fmt(3)

الحزمة: ldns

التراخيص: custom:BSD

ldns(3) دليل وظائف المكتبة ldns(3)

الاسم

ldns_zone, ldns_zone_new, ldns_zone_free, ldns_zone_deep_free, ldns_zone_new_frm_fp, ldns_zone_new_frm_fp_l, ldns_zone_print, ldns_zone_print_fmt - إنشاء وتدمير وطباعة ldns_zone

#include <stdint.h>
#include <stdbool.h>

#include <ldns/ldns.h>

ldns_zone * ldns_zone_new(void);

void ldns_zone_free(ldns_zone *zone);

void ldns_zone_deep_free(ldns_zone *zone);

ldns_status ldns_zone_new_frm_fp(ldns_zone **z, FILE *fp, const ldns_rdf *origin, uint32_t ttl, ldns_rr_class c);

ldns_status ldns_zone_new_frm_fp_l(ldns_zone **z, FILE *fp, const ldns_rdf *origin, uint32_t ttl, ldns_rr_class c, int *line_nr);

void ldns_zone_print(FILE *output, const ldns_zone *z);

void ldns_zone_print_fmt(FILE *output, const ldns_output_format *fmt, const ldns_zone *z);

الوصف

ldns_zone
منطقة DNS

قائمة من سجلات الموارد (RR's) مع
معلومات إضافية تأتي من سجل SOA
ملاحظة: لم يتم فعل شيء لجعل هذا فعالاً (حتى الآن).
struct ldns_struct_zone
{
سجل SOA يحدد المنطقة:
ldns_rr *_soa;
/* بشكل أساسي المنطقة هي قائمة بسجلات الموارد */
ldns_rr_list *_rrs;
/* يمكننا تغيير هذا ليكون شجرة b إلخ. إلخ. todo */
};
typedef struct ldns_struct_zone ldns_zone;

ldns_zone_new() تنشئ بنية ldns_zone جديدة
تُرجع مؤشرًا إلى بنية ldns_zone

ldns_zone_free() تحرر الذاكرة المخصصة للمنطقة، وبنية rr_list فيها
zone: المنطقة المراد تحريرها

ldns_zone_deep_free() تحرر الذاكرة المخصصة للمنطقة، وسجل SOA فيها، وبنية rr_list فيها، بما في ذلك سجلات الموارد فيها. إلخ.
zone: المنطقة المراد تحريرها

ldns_zone_new_frm_fp() تنشئ منطقة جديدة من ملف
z: المنطقة الجديدة
*fp: مؤشر الملف للاستخدام
*origin: أصل المنطقة
ttl: TTL الافتراضي للاستخدام
c: فئة DNS الافتراضية للاستخدام (IN)

تُرجع رسالة ldns_status بخطأ أو LDNS_STATUS_OK

ldns_zone_new_frm_fp_l() تنشئ منطقة جديدة من ملف، وتتتبع ترقيم الأسطر
z: المنطقة الجديدة
*fp: مؤشر الملف للاستخدام
*origin: أصل المنطقة
ttl: TTL الافتراضي للاستخدام
c: فئة DNS الافتراضية للاستخدام (IN)
line_nr: يُستخدم لرسالة الخطأ، للوصول إلى رقم السطر

تُرجع رسالة ldns_status بخطأ أو LDNS_STATUS_OK

ldns_zone_print() تطبع بنية منطقة * إلى المخرج. ملاحظة: يتم تضمين سجل SOA في هذا الإخراج
output: واصف الملف للطباعة إليه
z: المنطقة المراد طباعتها

ldns_zone_print_fmt() تطبع بنية منطقة * إلى المخرج. ملاحظة: يتم تضمين سجل SOA في هذا الإخراج
output: واصف الملف للطباعة إليه
fmt: تنسيق التمثيل النصي
z: المنطقة المراد طباعتها
