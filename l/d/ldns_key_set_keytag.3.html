ldns_key_set_keytag(3)

الحزمة: ldns

التراخيص: مخصص:BSD

ldns(3) دليل دوال المكتبة ldns(3)

الاسم

ldns_key_set_algorithm, ldns_key_set_rsa_key, ldns_key_set_dsa_key, ldns_key_set_hmac_key, ldns_key_set_origttl, ldns_key_set_inception, ldns_key_set_expiration, ldns_key_set_pubkey_owner, ldns_key_set_keytag, ldns_key_set_flags, ldns_key_list_set_key_count, ldns_key_algo_supported - تعيين سمات ldns_key

ملخص

#include <stdint.h>
#include <stdbool.h>

#include <ldns/ldns.h>

void ldns_key_set_algorithm(ldns_key *k, ldns_signing_algorithm l);

void ldns_key_set_rsa_key(ldns_key *k, RSA *r);

void ldns_key_set_dsa_key(ldns_key *k, DSA *d);

void ldns_key_set_hmac_key(ldns_key *k, unsigned char *hmac);

void ldns_key_set_origttl(ldns_key *k, uint32_t t);

void ldns_key_set_inception(ldns_key *k, uint32_t i);

void ldns_key_set_expiration(ldns_key *k, uint32_t e);

void ldns_key_set_pubkey_owner(ldns_key *k, ldns_rdf *r);

void ldns_key_set_keytag(ldns_key *k, uint16_t tag);

void ldns_key_set_flags(ldns_key *k, uint16_t flags);

void ldns_key_list_set_key_count(ldns_key_list *key, size_t count);

int ldns_key_algo_supported(int algo);

الوصف

ldns_key_set_algorithm() تعيين خوارزمية المفتاح
k: المفتاح
l: الخوارزمية

ldns_key_set_rsa_key() تعيين بيانات مفتاح RSA. يجب على المستخدم تحرير بيانات RSA.
k: المفتاح
r: بيانات RSA

ldns_key_set_dsa_key() تعيين بيانات مفتاح DSA. يجب على المستخدم تحرير بيانات DSA.
k: المفتاح
d: بيانات DSA

ldns_key_set_hmac_key() تعيين بيانات مفتاح HMAC
k: المفتاح
hmac: بيانات المفتاح الأولية

ldns_key_set_origttl() تعيين TTL الأصلي للمفتاح
k: المفتاح
t: TTL

ldns_key_set_inception() تعيين تاريخ بدء صلاحية المفتاح (بالثواني بعد الحقبة)
k: المفتاح
i: تاريخ البدء

ldns_key_set_expiration() تعيين تاريخ انتهاء صلاحية المفتاح (بالثواني بعد الحقبة)
k: المفتاح
e: تاريخ الانتهاء

ldns_key_set_pubkey_owner() تعيين مالك المفتاح العام
k: المفتاح
r: المالك

ldns_key_set_keytag() تعيين علامة المفتاح
k: المفتاح
tag: علامة المفتاح

ldns_key_set_flags() تعيين أعلام المفتاح
k: المفتاح
flags: الأعلام

ldns_key_list_set_key_count() تعيين عدد المفاتيح في قائمة المفاتيح إلى count
key: المفتاح
count: العدد

ldns_key_algo_supported() التحقق مما إذا كانت خوارزمية المفتاح مدعومة
algo: رقم خوارزمية التوقيع.
يعيد true إذا كانت مدعومة.
