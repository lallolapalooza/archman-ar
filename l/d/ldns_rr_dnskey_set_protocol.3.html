ldns_rr_dnskey_set_protocol(3)

الحزمة: ldns

التراخيص: custom:BSD

ldns(3) دليل دوال المكتبة ldns(3)

الاسم

ldns_rr_dnskey_flags, ldns_rr_dnskey_set_flags, ldns_rr_dnskey_protocol, ldns_rr_dnskey_set_protocol, ldns_rr_dnskey_algorithm, ldns_rr_dnskey_set_algorithm, ldns_rr_dnskey_key, ldns_rr_dnskey_set_key - الحصول على وتعيين حقول بيانات DNSKEY RR

الملخص

#include <stdint.h>
#include <stdbool.h>

#include <ldns/ldns.h>

ldns_rdf* ldns_rr_dnskey_flags(const ldns_rr *r);

bool ldns_rr_dnskey_set_flags(ldns_rr *r, ldns_rdf *f);

ldns_rdf* ldns_rr_dnskey_protocol(const ldns_rr *r);

bool ldns_rr_dnskey_set_protocol(ldns_rr *r, ldns_rdf *f);

ldns_rdf* ldns_rr_dnskey_algorithm(const ldns_rr *r);

bool ldns_rr_dnskey_set_algorithm(ldns_rr *r, ldns_rdf *f);

ldns_rdf* ldns_rr_dnskey_key(const ldns_rr *r);

bool ldns_rr_dnskey_set_key(ldns_rr *r, ldns_rdf *f);

الوصف

ldns_rr_dnskey_flags() تُرجع العلامات لسجل LDNS_RR_TYPE_DNSKEY
r: السجل المرجعي
تُرجع ldns_rdf* بالعلامات أو NULL عند الفشل

ldns_rr_dnskey_set_flags() تُعيّن العلامات لسجل LDNS_RR_TYPE_DNSKEY
r: السجل المرجعي للاستخدام
f: العلامات المراد تعيينها
تُرجع true عند النجاح، و false بخلاف ذلك

ldns_rr_dnskey_protocol() تُرجع البروتوكول لسجل LDNS_RR_TYPE_DNSKEY
r: السجل المرجعي
تُرجع ldns_rdf* بالبروتوكول أو NULL عند الفشل

ldns_rr_dnskey_set_protocol() تُعيّن البروتوكول لسجل LDNS_RR_TYPE_DNSKEY
r: السجل المرجعي للاستخدام
f: البروتوكول المراد تعيينه
تُرجع true عند النجاح، و false بخلاف ذلك

ldns_rr_dnskey_algorithm() تُرجع الخوارزمية لسجل LDNS_RR_TYPE_DNSKEY
r: السجل المرجعي
تُرجع ldns_rdf* بالخوارزمية أو NULL عند الفشل

ldns_rr_dnskey_set_algorithm() تُعيّن الخوارزمية لسجل LDNS_RR_TYPE_DNSKEY
r: السجل المرجعي للاستخدام
f: الخوارزمية المراد تعيينها
تُرجع true عند النجاح، و false بخلاف ذلك

ldns_rr_dnskey_key() تُرجع بيانات المفتاح لسجل LDNS_RR_TYPE_DNSKEY
r: السجل المرجعي
تُرجع ldns_rdf* ببيانات المفتاح أو NULL عند الفشل

ldns_rr_dnskey_set_key() تُعيّن بيانات المفتاح لسجل LDNS_RR_TYPE_DNSKEY
r: السجل المرجعي للاستخدام
f: بيانات المفتاح المراد تعيينها
تُرجع true عند النجاح، و false بخلاف ذلك
