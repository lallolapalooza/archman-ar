ldns_buffer_begin(3)

الحزمة: ldns

التراخيص: custom:BSD

ldns(3) دليل وظائف المكتبة ldns(3)

الاسم

ldns_buffer_limit, ldns_buffer_set_limit, ldns_buffer_capacity, ldns_buffer_set_capacity, ldns_buffer_reserve, ldns_buffer_at, ldns_buffer_begin, ldns_buffer_end, ldns_buffer_current - حدود المخزن المؤقت ومؤشراته

الاختصار

#include <stdint.h>
#include <stdbool.h>

#include <ldns/ldns.h>

size_t ldns_buffer_limit(const ldns_buffer *buffer);

void ldns_buffer_set_limit(ldns_buffer *buffer, size_t limit);

size_t ldns_buffer_capacity(const ldns_buffer *buffer);

bool ldns_buffer_set_capacity(ldns_buffer *buffer, size_t capacity);

bool ldns_buffer_reserve(ldns_buffer *buffer, size_t amount);

uint8_t * ldns_buffer_at(const ldns_buffer *buffer, size_t at);

uint8_t * ldns_buffer_begin(const ldns_buffer *buffer);

uint8_t * ldns_buffer_end(const ldns_buffer *buffer);

uint8_t * ldns_buffer_current(const ldns_buffer *buffer);

الوصف

ldns_buffer_limit() تُرجع الحد الأقصى لحجم المخزن المؤقت \param[in] buffer
تُرجع الحجم

ldns_buffer_set_limit() تُغيّر حد المخزن المؤقت. إذا كان موضع المخزن المؤقت أكبر من الحد الجديد، يتم تعيين الموضع إلى الحد.
buffer: المخزن المؤقت
limit: الحد الجديد

ldns_buffer_capacity() تُرجع عدد البايتات التي يمكن للمخزن المؤقت استيعابها.
buffer: المخزن المؤقت
تُرجع عدد البايتات

ldns_buffer_set_capacity() تُغيّر سعة المخزن المؤقت. يتم إعادة تخصيص البيانات، لذلك قد تصبح أي مؤشرات إلى البيانات غير صالحة. يتم تعيين حد المخزن المؤقت إلى سعة المخزن المؤقت الجديدة.
buffer: المخزن المؤقت
capacity: السعة المراد استخدامها
تُرجع ما إذا كان هذا قد فشل أو نجح

ldns_buffer_reserve() تضمن أن المخزن المؤقت يمكن أن يستوعب على الأقل AMOUNT بايتات إضافية. يتم زيادة سعة المخزن المؤقت إذا لزم الأمر باستخدام buffer_set_capacity().

يتم دائمًا تعيين حد المخزن المؤقت إلى السعة (التي قد تكون قد زادت).
buffer: المخزن المؤقت
amount: الكمية المراد استخدامها
تُرجع ما إذا كان هذا قد فشل أو نجح

ldns_buffer_at() تُرجع مؤشرًا إلى البيانات في الموضع المحدد.
buffer: المخزن المؤقت
at: الموضع
تُرجع المؤشر إلى البيانات

ldns_buffer_begin() تُرجع مؤشرًا إلى بداية المخزن المؤقت (البيانات في الموضع 0).
buffer: المخزن المؤقت
تُرجع المؤشر

ldns_buffer_end() تُرجع مؤشرًا إلى نهاية المخزن المؤقت (البيانات عند حد المخزن المؤقت).
buffer: المخزن المؤقت
تُرجع المؤشر

ldns_buffer_current() تُرجع مؤشرًا إلى البيانات في الموضع الحالي للمخزن المؤقت.
buffer: المخزن المؤقت
تُرجع المؤشر
