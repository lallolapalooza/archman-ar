ldns_wire2dname(3)

الحزمة: ldns

التراخيص: custom:BSD

ldns(3) دليل وظائف المكتبة ldns(3)

الاسم

ldns_wire2rr, ldns_wire2pkt, ldns_wire2rdf, ldns_wire2dname - التحويل من تنسيق السلك إلى النوع المضيف

الملخص

#include <stdint.h>
#include <stdbool.h>

#include <ldns/ldns.h>

ldns_status ldns_wire2rr(ldns_rr **rr, const uint8_t *wire, size_t max, size_t *pos, ldns_pkt_section section);

ldns_status ldns_wire2pkt(ldns_pkt **packet, const uint8_t *data, size_t len);

ldns_status ldns_wire2rdf(ldns_rr *rr, const uint8_t *wire, size_t max, size_t *pos);

ldns_status ldns_wire2dname(ldns_rdf **dname, const uint8_t *wire, size_t max, size_t *pos);

الوصف

تقوم ldns_wire2rr() بتحويل البيانات الموجودة في مصفوفة البايتات uint8_t (بتنسيق السلك) إلى سجل مورد DNS. ستقوم هذه الوظيفة بتهيئة وتخصيص مساحة ذاكرة لهيكل rr. تتم إضافة طول بيانات السلك لهذا السجل إلى قيمة *pos.

rr: مؤشر إلى الهيكل الذي سيحتوي على قيمة rdata
wire: مؤشر إلى المخزن المؤقت بالبيانات
max: طول المخزن المؤقت للبيانات (بالبايت)
pos: موضع السجل في المخزن المؤقت (أي عدد البايتات من بداية المخزن المؤقت)
section: القسم في الحزمة الذي يهدف إليه السجل
ترجع LDNS_STATUS_OK إذا نجح كل شيء، وإلا ترجع خطأ

تقوم ldns_wire2pkt() بتحويل البيانات الموجودة في مصفوفة البايتات uint8_t (بتنسيق السلك) إلى حزمة DNS. ستقوم هذه الوظيفة بتهيئة وتخصيص مساحة ذاكرة لهيكل الحزمة.

packet: مؤشر إلى الهيكل الذي سيحتوي على الحزمة
data: مؤشر إلى المخزن المؤقت بالبيانات
len: طول المخزن المؤقت للبيانات (بالبايت)
ترجع LDNS_STATUS_OK إذا نجح كل شيء، وإلا ترجع خطأ

تقوم ldns_wire2rdf() بتحويل البيانات الموجودة في مصفوفة البايتات uint8_t (بتنسيق السلك) إلى حقول rdata الخاصة بـ DNS، وإضافتها إلى قائمة rdfs الخاصة بالسجل المعطى. ستقوم هذه الوظيفة بتهيئة وتخصيص مساحة ذاكرة لهياكل dname. تتم إضافة طول بيانات السلك لهذه الـ rdfs إلى قيمة *pos.

يتم قراءة جميع الـ rdfs التي تنتمي إلى السجل؛ يجب ألا يحتوي السجل على أي rdfs بعد. يتم إرجاع خطأ إذا لم يكن من الممكن تحليل التنسيق.

rr: مؤشر إلى هيكل ldns_rr الذي سيحتوي على قيمة rdata
wire: مؤشر إلى المخزن المؤقت بالبيانات
max: طول المخزن المؤقت للبيانات (بالبايت)
pos: موضع الـ rdf في المخزن المؤقت (أي عدد البايتات من بداية المخزن المؤقت)
ترجع LDNS_STATUS_OK إذا نجح كل شيء، وإلا ترجع خطأ

تقوم ldns_wire2dname() بتحويل البيانات الموجودة في مصفوفة البايتات uint8_t (بتنسيق السلك) إلى حقل rdata لـ dname الخاص بـ DNS. ستقوم هذه الوظيفة بتهيئة وتخصيص مساحة ذاكرة لهيكل dname. تتم إضافة طول بيانات السلك لهذا الـ rdf إلى قيمة *pos.

dname: مؤشر إلى الهيكل الذي سيحتوي على قيمة rdata
wire: مؤشر إلى المخزن المؤقت بالبيانات
max: طول المخزن المؤقت للبيانات (بالبايت)
pos: موضع الـ rdf في المخزن المؤقت (أي عدد البايتات من بداية المخزن المؤقت)
ترجع LDNS_STATUS_OK إذا نجح كل شيء، وإلا ترجع خطأ
