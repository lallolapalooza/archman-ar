ldns_dane_select_certificate(3)

الحزمة: ldns

التراخيص: custom:BSD

ldns(3) دليل وظائف المكتبة ldns(3)

الاسم

ldns_dane_create_tlsa_rr, ldns_dane_create_tlsa_owner, ldns_dane_cert2rdf, ldns_dane_select_certificate - وظائف إنشاء سجل TLSA

الملخص

#include <stdint.h>
#include <stdbool.h>

#include <ldns/ldns.h>

ldns_status ldns_dane_create_tlsa_rr(ldns_rr** tlsa, ldns_tlsa_certificate_usage certificate_usage, ldns_tlsa_selector selector, ldns_tlsa_matching_type matching_type, X509* cert);

ldns_status ldns_dane_create_tlsa_owner(ldns_rdf** tlsa_owner, const ldns_rdf* name, uint16_t port, ldns_dane_transport transport);

ldns_status ldns_dane_cert2rdf(ldns_rdf** rdf, X509* cert, ldns_tlsa_selector selector, ldns_tlsa_matching_type matching_type);

ldns_status ldns_dane_select_certificate(X509** selected_cert, X509* cert, STACK_OF(X509)* extra_certs, X509_STORE* pkix_validation_store, ldns_tlsa_certificate_usage cert_usage, int index);

الوصف

ldns_dane_create_tlsa_rr() تنشئ سجلاً لمورد TLSA من الشهادة. لا يتم إجراء أي تحقق من PKIX! يتم استخدام الشهادة المقدمة كبيانات بغض النظر عن قيمة استخدام الشهادة.

tlsa: سجل مورد TLSA الذي تم إنشاؤه.
certificate_usage: القيمة الخاصة بحقل استخدام الشهادة.
selector: القيمة الخاصة بحقل المحدد.
matching_type: القيمة الخاصة بنوع المطابقة.
cert: الشهادة التي سيتم تمثيل بياناتها.

تُرجع LDNS_STATUS_OK عند النجاح أو رمز خطأ بخلاف ذلك.

ldns_dane_create_tlsa_owner() تنشئ اسم مجال يتكون من الاسم المقدم، مسبوقًا بمنفذ الخدمة ونوع النقل: _<EM>port</EM>._<EM>transport</EM>.<EM>name</EM>.

tlsa_owner: اسم المجال الذي تم إنشاؤه.
name: اسم المجال الذي يجب أن يسبقه.
port: رقم منفذ الخدمة الذي يجب إنشاء الاسم من أجله.
transport: النقل الذي يجب إنشاء الاسم من أجله.
تُرجع LDNS_STATUS_OK عند النجاح أو رمز خطأ بخلاف ذلك.

ldns_dane_cert2rdf() تنشئ RDF من نوع LDNS_RDF_TYPE_HEX بناءً على البيانات الثنائية التي اختارها المحدد وتم ترميزها باستخدام نوع المطابقة.

rdf: RDF الذي تم إنشاؤه من نوع LDNS_RDF_TYPE_HEX.
cert: الشهادة التي يتم اختيار البيانات منها.
selector: الشهادة الكاملة أو المفتاح العام.
matching_type: البيانات الكاملة أو تجزئة SHA256 أو SHA512 للبيانات المختارة.
تُرجع LDNS_STATUS_OK عند النجاح أو رمز خطأ بخلاف ذلك.

ldns_dane_select_certificate() تحدد الشهادة من cert، أو extra_certs، أو pkix_validation_store بناءً على قيمة cert_usage و index.

selected_cert: الشهادة المحددة.
cert: الشهادة المراد التحقق منها (أو عدم التحقق منها).
extra_certs: الشهادات الوسيطة التي قد تكون ضرورية أثناء التحقق. قد تكون NULL، إلا عند استخدام الشهادة "تأكيد مرساة الثقة" لأنه يجب تقديم مرساة الثقة (وإلا اختر "شهادة صادرة عن النطاق!").
pkix_validation_store: يُستخدم عندما يكون استخدام الشهادة هو "قيد شهادة CA" أو "قيد شهادة الخدمة" للتحقق من صحة الشهادة، وفي حالة "قيد شهادة CA"، لتحديد CA. عندما يكون pkix_validation_store NULL، يتم إيقاف التحقق صراحةً، ويكون السلوك بعد ذلك هو نفسه كما هو الحال مع "تأكيد مرساة الثقة" و "شهادة صادرة عن النطاق" على التوالي.
cert_usage: أي شهادة يجب استخدامها وكيفية التحقق منها.
index: يُستخدم لتحديد مرساة الثقة عندما يكون استخدام الشهادة هو "تأكيد مرساة الثقة". 0 هي الشهادة الأخيرة في سلسلة التحقق. 1 هي التي قبل الأخيرة، إلخ. عندما يكون index -1، يتم استخدام الشهادة الأخيرة التي يجب أن تكون موقعة ذاتيًا. يمكن أن يساعد هذا في التأكد من أن مرساة الثقة المقصودة (الموقعة ذاتيًا) موجودة بالفعل في extra_certs (وهو شرط DANE).

تُرجع LDNS_STATUS_OK عند النجاح أو رمز خطأ بخلاف ذلك.
