ldns_verify_rrsig_keylist_notime(3)

الحزمة: ldns

التراخيص: custom:BSD

ldns(3) دليل وظائف المكتبة ldns(3)

الاسم

ldns_verify, ldns_verify_rrsig, ldns_verify_rrsig_keylist, ldns_verify_rrsig_keylist_notime, ldns_verify_notime - التحقق من RRSIGs

ملخص

#include <stdint.h>
#include <stdbool.h>

#include <ldns/ldns.h>

ldns_status ldns_verify(ldns_rr_list *rrset, ldns_rr_list *rrsig, const ldns_rr_list *keys, ldns_rr_list *good_keys);

ldns_status ldns_verify_rrsig(ldns_rr_list *rrset, ldns_rr *rrsig, ldns_rr *key);

ldns_status ldns_verify_rrsig_keylist(ldns_rr_list *rrset, ldns_rr *rrsig, const ldns_rr_list *keys, ldns_rr_list *good_keys);

ldns_status ldns_verify_rrsig_keylist_notime(const ldns_rr_list *rrset, const ldns_rr *rrsig, const ldns_rr_list *keys, ldns_rr_list *good_keys);

ldns_status ldns_verify_notime(ldns_rr_list *rrset, ldns_rr_list *rrsig, const ldns_rr_list *keys, ldns_rr_list *good_keys);

الوصف

ldns_verify() تتحقق من قائمة التواقيع لمجموعة RR واحدة.

rrset: مجموعة RR للتحقق منها
rrsig: قائمة بالتوقيعات للتحقق منها
keys: قائمة بالمفاتيح للتحقق باستخدامها
good_keys: إذا كانت هذه قائمة (تم تهيئتها)، فإن المؤشر إلى المفاتيح من المفاتيح التي تتحقق من صحة أحد التوقيعات تتم إضافتها إليها
تُرجع الحالة LDNS_STATUS_OK إذا كان هناك مفتاح صحيح واحد على الأقل

ldns_verify_rrsig() تتحقق من RRSIG باستخدام مفتاح واحد
rrset: مجموعة RR
rrsig: RRSIG للتحقق منه
key: المفتاح المستخدم
تُرجع رسالة الحالة سواء نجح التحقق.

ldns_verify_rrsig_keylist() تتحقق من RRSIG. يتم تجربة جميع المفاتيح في مجموعة المفاتيح.
rrset: مجموعة RR للتحقق منها
rrsig: توقيع مجموعة RR
keys: المفاتيح التي سيتم تجربتها
good_keys: إذا كانت هذه قائمة (تم تهيئتها)، فإن المؤشر إلى المفاتيح من المفاتيح التي تتحقق من صحة أحد التوقيعات تتم إضافتها إليها
تُرجع قائمة بالمفاتيح التي تتحقق من صحة RRSIG + rrset. تُرجع الحالة LDNS_STATUS_OK إذا تطابق مفتاح واحد على الأقل. وإلا، يحدث خطأ.

ldns_verify_rrsig_keylist_notime() تتحقق من RRSIG. يتم تجربة جميع المفاتيح في مجموعة المفاتيح. لا يتم التحقق من الوقت.
rrset: مجموعة RR للتحقق منها
rrsig: توقيع مجموعة RR
keys: المفاتيح التي سيتم تجربتها
good_keys: إذا كانت هذه قائمة (تم تهيئتها)، فإن المؤشر إلى المفاتيح من المفاتيح التي تتحقق من صحة أحد التوقيعات تتم إضافتها إليها
تُرجع قائمة بالمفاتيح التي تتحقق من صحة RRSIG + rrset. تُرجع الحالة LDNS_STATUS_OK إذا تطابق مفتاح واحد على الأقل. وإلا، يحدث خطأ.

ldns_verify_notime() تتحقق من قائمة التواقيع لمجموعة RR واحدة، ولكن تتجاهل الوقت. لا يتم التحقق من تاريخ البدء وانتهاء الصلاحية.

rrset: مجموعة RR للتحقق منها
rrsig: قائمة بالتوقيعات للتحقق منها
keys: قائمة بالمفاتيح للتحقق باستخدامها
good_keys: إذا كانت هذه قائمة (تم تهيئتها)، فإن المؤشر إلى المفاتيح من المفاتيح التي تتحقق من صحة أحد التوقيعات تتم إضافتها إليها
تُرجع الحالة LDNS_STATUS_OK إذا كان هناك مفتاح صحيح واحد على الأقل
