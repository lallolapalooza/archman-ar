ldns_buffer_remaining_at(3)

الحزمة: ldns

التراخيص: custom:BSD

ldns(3) دليل وظائف المكتبة ldns(3)

الاسم

ldns_buffer_remaining_at، ldns_buffer_remaining، ldns_buffer_available_at، ldns_buffer_available، ldns_buffer_status، ldns_buffer_status_ok - التحقق من حالة المخزن المؤقت

الملخص

#include <stdint.h>
#include <stdbool.h>

#include <ldns/ldns.h>

size_t ldns_buffer_remaining_at(const ldns_buffer *buffer, size_t at);

size_t ldns_buffer_remaining(const ldns_buffer *buffer);

int ldns_buffer_available_at(const ldns_buffer *buffer, size_t at, size_t count);

int ldns_buffer_available(const ldns_buffer *buffer, size_t count);

ldns_status ldns_buffer_status(const ldns_buffer *buffer);

bool ldns_buffer_status_ok(const ldns_buffer *buffer);

الوصف

ldns_buffer_remaining_at() تُرجع عدد البايتات المتبقية بين الموضع المحدد والحد.
buffer: المخزن المؤقت
at: الموضع المحدد
تُرجع عدد البايتات

ldns_buffer_remaining() تُرجع عدد البايتات المتبقية بين موضع المخزن المؤقت وحدوده.
buffer: المخزن المؤقت
تُرجع عدد البايتات

ldns_buffer_available_at() تتحقق مما إذا كان المخزن المؤقت يحتوي على COUNT بايتات متاحة على الأقل. قبل القراءة أو الكتابة، يجب على المتصل التأكد من توفر مساحة كافية!
buffer: المخزن المؤقت
at: الموضع المحدد
count: مقدار ما هو متاح
تُرجع صحيح أو خطأ (كرقم صحيح؟)

ldns_buffer_available() تتحقق مما إذا كان المخزن المؤقت يحتوي على count بايتات متاحة في الموضع الحالي
buffer: المخزن المؤقت
count: مقدار ما هو متاح
تُرجع صحيح أو خطأ (كرقم صحيح؟)

ldns_buffer_status() تُرجع حالة المخزن المؤقت \param[in] buffer
تُرجع الحالة

ldns_buffer_status_ok() تُرجع صحيح إذا كانت حالة المخزن المؤقت هي LDNS_STATUS_OK، خطأ بخلاف ذلك
buffer: المخزن المؤقت
تُرجع صحيح أو خطأ
