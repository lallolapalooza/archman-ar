ldns_dnssec_name
هيكل يحتوي على جميع سجلات الموارد لاسم نطاق
بما في ذلك NSEC3 المشتقة، إذا وجدت
struct ldns_struct_dnssec_name
{
/**
* مؤشر إلى dname يحتوي على الاسم.
* يشير عادةً إلى اسم المالك لأول RR من أول RRset
*/
ldns_rdf *name;
/**
* عادةً، يكون الاسم مؤشرًا إلى اسم المالك لأول rr لهذا
* الاسم، ولكن في بعض الأحيان لا توجد بيانات فعلية للإشارة إليها،
* على سبيل المثال في
* الأسماء التي تمثل محطات طرفية فارغة. إذا كان الأمر كذلك، اضبط name_alloced على true للإشارة
* إلى أنه يجب أيضًا تحرير هذه البيانات عند تحرير الاسم
*/
bool name_alloced;
/**
* RRsets لهذا الاسم
*/
ldns_dnssec_rrsets *rrsets;
/**
* NSEC يشير إلى الاسم التالي (أو NSEC3 يشير إلى NSEC3 التالي)
*/
ldns_rr *nsec;
/**
* توقيعات لسجل NSEC
*/
ldns_dnssec_rrs *nsec_signatures;
/**
* على عكس ما يشير إليه الاسم is_glue، يتم تعيين هذا الحقل إلى true بواسطة
* ldns_dnssec_zone_mark_glue() أو ldns_dnssec_zone_mark_and_get_glue()
* عندما يتم حجب الاسم الذي يمثله هيكل dnssec_name هذا.
* الأسماء التي تحتوي على rrsets محجوبة أخرى وسجلات مع غراء على
* نقطة التفويض لن يكون لديها هذا البول تعيين إلى true.
* لا ينبغي قراءة هذا الحقل مباشرة، ولكن فقط عبر
* الدالة ldns_dnssec_name_is_glue()!
*/
bool is_glue;
/**
* مؤشر لتخزين الاسم المجزأ (يستخدم فقط عند وجوده في منطقة NSEC3
*/
ldns_rdf *hashed_name;
};

typedef struct ldns_struct_dnssec_name ldns_dnssec_name;

ldns_dnssec_rrs
قائمة مرتبطة أحاديًا لـ rrs
struct ldns_struct_dnssec_rrs
{
ldns_rr *rr;
ldns_dnssec_rrs *next;
};

typedef struct ldns_struct_dnssec_rrs ldns_dnssec_rrs;

ldns_dnssec_rrsets
قائمة مرتبطة أحاديًا لـ RRsets
struct ldns_struct_dnssec_rrsets
{
ldns_dnssec_rrs *rrs;
ldns_rr_type type;
ldns_dnssec_rrs *signatures;
ldns_dnssec_rrsets *next;
};

typedef struct ldns_struct_dnssec_rrsets ldns_dnssec_rrsets;
