level.3bobcat

الحزمة: libbobcat

التراخيص: GPL

FBB::level(3bobcat) معالجة السجلات FBB::level(3bobcat)

الاسم

FBB::level - معالج يحدد مستوى تسجيل كائنات FBB::Log

الملخص

#include <bobcat/level>
خيار الربط: -lbobcat

الوصف

يُستخدم كائن FBB::level لتحديد `قوى` إدراج المعلومات التي يتم إدراجها في كائنات FBB::Log. عند إدراجها في كائنات Log، تحدد `قوة` عمليات الإدراج اللاحقة (أو استدعاءات write) في تلك الكائنات Log.

على العكس من ذلك، قد تحدد كائنات Log `مقاومات` إدراج من خلال أعضاء setLevel الخاصة بها. إذا كانت `قوة` level تساوي أو تتجاوز `مقاومة` كائن Log، يتم تنفيذ الإدراج، وإلا يتم تجاهل الإدراج. قد يحتوي بيان إدراج واحد على استدعاءات level متعددة. إذا كان الأمر كذلك، فإن كل استدعاء level يقوم بتحديث `القوة` لعمليات الإدراج التي تلي استدعاء level.

بشكل افتراضي، يتم إكمال عمليات الإدراج في كائنات Log التي لا تسبقها عمليات إدراج level.

عند إدراج كائن level في نوع آخر من كائنات std::ostream، فإن كائن level لا يقوم بأي إجراءات.

المنشئات

o
level(size_t force):
يُستخدم هذا المنشئ في بيانات الإدراج، وعادةً ما يتم إدراجه ككائن مجهول في كائن FBB::Log.

منشئات النسخ والنقل (وعوامل التعيين) متاحة.

العامل المعالج الزائد

o
std::ostream &::operator<<(std::ostream &ostr, FBB::level const &force):
إذا كان يمكن تحويل ostr إلى كائن FBB::Log، فسيتم تنفيذ عمليات الإدراج اللاحقة إذا تجاوزت force `مقاومة` كائن Log كما هو محدد بواسطة عضوه setLevel. إذا كان ostr لا يشير فعليًا إلى كائن Log، فسيتم إرجاع ostr دون أي إجراء إضافي، ويتعامل ostr مع عمليات الإدراج اللاحقة بطريقته الافتراضية.

مثال

#include <iostream>
#include <iomanip>
//#include <bobcat/log>
#include "../log"
#include <bobcat/level>
using namespace std;
using namespace FBB;
int main()
{
//    Log &log = Log::initialize("&1"); // يستخدم كائن Log الثابت
    Log log;                        // تعريف صريح لكائن Log
//    log.open("/tmp/out");           // أو مرة واحدة: Log log{ "/tmp/out" }
    log << "هذه الرسالة مكتوبة إلى cout" << nl <<
           setw(16) << ’ ’ << "احتلال عدة أسطر\n";
    log.off();
    log << "هذه الرسالة غير معروضة\n";
    log << "هذه الرسالة غير معروضة\n";
    log << fnl;
    log << "هذه الرسالة غير معروضة\n";
    log.on(2);
    log << "هذه الرسالة معروضة\n";
    log << level(0) << "غير معروض" << level(2) << "معروض بالمستوى 2\n";
    log << level(3) << "في المستوى (3)" << level(1) << "غير معروض" << fnl;
    log << "سطر جديد منفصل\n";
    log << level(2) << "مرة أخرى في العمل\n";
    log << "السطر النهائي\n";
    log.str("ab");
    log(’a’) << "مرحباً a!" << endl;
    log(’b’) << "مرحباً b!" << nl <<
                setw(16) << ’ ’ << "حتى الآن، كل شيء على ما يرام" << endl;
    log << "غير معروض" << endl;
    log(’c’) << "غير معروض\n";
    log << "غير معروض\n";
    log.setLevel(2);
    log << level(2) << "مرة أخرى في العمل\n";
}
