loc_database_new(3)

الحزمة: libloc

التراخيص: LGPL-2.1-or-later

LOC_DATABASE_NEW(3)   LOC_DATABASE_NEW(3)

اسم

loc_database_new - إنشاء سياق libloc جديد

الملخص

#include <libloc/libloc.h>
#include <libloc/database.h>

struct loc_database;

int loc_database_new(struct loc_ctx* ctx, struct loc_database** database, FILE* f);

عد المراجع:

struct loc_database* loc_database_ref(struct loc_database* db);

struct loc_database* loc_database_unref(struct loc_database* db);

الوصول إلى بعض البيانات:

time_t loc_database_created_at(struct loc_database* db);

const char* loc_database_get_vendor(struct loc_database* db);

const char* loc_database_get_description(struct loc_database* db);

const char* loc_database_get_license(struct loc_database* db);

الوصف

loc_database_new() يفتح قاعدة بيانات جديدة من واصف الملف المعطى. يمكن إغلاق واصف الملف بعد هذه العملية لأن الدالة تنشئ نسختها الخاصة.

إذا تم فتح قاعدة البيانات بنجاح، يتم إرجاع صفر. بخلاف ذلك، سيشير رمز إرجاع غير صفري إلى حدوث خطأ وسيتم تعيين errno بشكل مناسب.

يمكن استرداد بيانات وصفية مختلفة حول قاعدة البيانات باستخدام loc_database_created_at() و loc_database_get_vendor() و loc_database_get_description() و loc_database_get_license().
