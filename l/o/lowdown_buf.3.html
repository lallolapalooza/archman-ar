lowdown_buf(3)

الحزمة: lowdown

التراخيص: ISC

LOWDOWN_BUF(3) دليل وظائف المكتبة LOWDOWN_BUF(3)

الاسم

lowdown_buf — تحليل مخزن Markdown إلى إخراج منسق

المكتبة

المكتبة "liblowdown"

الملخص

#include <sys/queue.h>
#include <stdio.h>
#include <lowdown.h>

int
lowdown_buf(const struct lowdown_opts *opts, const char *buf, size_t bufsz, char **ret, size_t *retsz, struct lowdown_metaq *metaq);

الوصف

يقوم بتحليل مخزن lowdown(5) buf بحجم bufsz إلى مخزن إخراج ret بحجم retsz وفقًا لتكوين opts. يتم تحديد تنسيق الإخراج بواسطة opts->type. إذا تم تعيين LOWDOWN_METADATA في opts->feat ولم يكن metaq NULL، فسيتم ملء metaq بالبيانات الوصفية المقدمة بتنسيق الإخراج المحدد.

المستدعي مسؤول عن تحرير ret و metaq.

قيم الإرجاع

تُرجع صفرًا عند الفشل، وغير صفر عند النجاح. عند الفشل، تكون القيم المشار إليها بواسطة res و rsz غير محددة.

أمثلة

يقوم المثال التالي بتحليل الإدخال القياسي إلى مستند HTML5 قائم بذاته. يمكّن هذا المقتطف الحواشي، والروابط التلقائية، والجداول، والنص الفائق، والنص المشطوب، وكتل التعليمات البرمجية المحددة، و Commonmark، وقوائم التعريف، وسمات الصور الموسعة، ومعالجة البيانات الوصفية. يمر الإخراج عبر HTML خام وله طباعة ذكية.

struct lowdown_opts opts;
char *buf = NULL, *obuf;
char rbuf[1024];
size_t sz, bufsz = 0, obufsz;

while (!(feof(stdin) || ferror(stdin))) {
	sz = fread(rbuf, 1, sizeof(rbuf), stdin);
	if (sz == 0)
		err(1, "fread");
	buf = realloc(buf, bufsz + sz);
	if (buf == NULL)
		err(1, NULL);
	memcpy(buf + bufsz, rbuf, sz);
	bufsz += sz;
}

if (ferror(stdin))
	err(1, "fread");

memset(&opts, 0, sizeof(struct lowdown_opts));
opts.type = LOWDOWN_HTML;
opts.feat = LOWDOWN_FOOTNOTES |
	LOWDOWN_AUTOLINK |
	LOWDOWN_TABLES |
	LOWDOWN_SUPER |
	LOWDOWN_STRIKE |
	LOWDOWN_FENCED |
	LOWDOWN_COMMONMARK |
	LOWDOWN_DEFLIST |
	LOWDOWN_IMG_EXT |
	LOWDOWN_METADATA;
opts.oflags = LOWDOWN_HTML_HEAD_IDS |
	LOWDOWN_HTML_NUM_ENT |
	LOWDOWN_HTML_OWASP |
	LOWDOWN_SMARTY |
	LOWDOWN_STANDALONE;
if (!lowdown_buf(&opts, buf, bufsz, &obuf, &obufsz, NULL))
	errx(1, "lowdown_buf");
fwrite(buf, 1, bufsz, stdout);
free(buf);
free(obuf);
