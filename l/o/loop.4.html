loop(4) دليل برامج تشغيل الأجهزة loop(4)

الاسم

loop, loop-control - أجهزة الحلقة

الملخص

#include <linux/loop.h>

الوصف

جهاز الحلقة هو جهاز كتلة يقوم بربط كتل البيانات الخاصة به ليس بجهاز مادي مثل القرص الصلب أو مشغل الأقراص الضوئية، ولكن بكتل ملف عادي في نظام ملفات أو بجهاز كتلة آخر. يمكن أن يكون هذا مفيدًا على سبيل المثال لتوفير جهاز كتلة لصورة نظام ملفات مخزنة في ملف، بحيث يمكن تثبيتها باستخدام الأمر mount(8). يمكنك القيام بذلك:


$ dd if=/dev/zero of=file.img bs=1MiB count=10;
$ sudo losetup /dev/loop4 file.img;
$ sudo mkfs -t ext4 /dev/loop4;
$ sudo mkdir /myloopdev;
$ sudo mount /dev/loop4 /myloopdev;

راجع losetup(8) لمثال آخر.

يمكن تحديد وظيفة نقل لكل جهاز حلقة لأغراض التشفير وفك التشفير.

توفر عمليات ioctl(2) التالية جهاز كتلة الحلقة:

LOOP_SET_FD
ربط جهاز الحلقة بالملف المفتوح الذي يتم تمرير واصف الملف الخاص به كوصف (ثالث) وسيطة ioctl(2).
LOOP_CLR_FD
فصل جهاز الحلقة عن أي واصف ملف.
LOOP_SET_STATUS
تعيين حالة جهاز الحلقة باستخدام (ثالث) وسيطة ioctl(2). هذه الوسيطة هي مؤشر إلى هيكل loop_info، محدد في <linux/loop.h> كما يلي:

struct loop_info {
    int           lo_number;      /* ioctl للقراءة/الكتابة */
    dev_t         lo_device;      /* ioctl للقراءة/الكتابة */
    unsigned long lo_inode;       /* ioctl للقراءة/الكتابة */
    dev_t         lo_rdevice;     /* ioctl للقراءة/الكتابة */
    int           lo_offset;
    int           lo_encrypt_type;
    int           lo_encrypt_key_size;  /* ioctl للكتابة فقط */
    int           lo_flags;       /* ioctl للقراءة/الكتابة (للقراءة فقط قبل
                                     Linux 2.6.25) */
    char          lo_name[LO_NAME_SIZE];
    unsigned char lo_encrypt_key[LO_KEY_SIZE];
                                  /* ioctl للكتابة فقط */
    unsigned long lo_init[2];
    char          reserved[4];
};

يجب أن يكون نوع التشفير (lo_encrypt_type) أحد الأنواع التالية: LO_CRYPT_NONE, LO_CRYPT_XOR, LO_CRYPT_DES, LO_CRYPT_FISH2, LO_CRYPT_BLOW, LO_CRYPT_CAST128, LO_CRYPT_IDEA, LO_CRYPT_DUMMY, LO_CRYPT_SKIPJACK، أو (منذ Linux 2.6.0) LO_CRYPT_CRYPTOAPI.
حقل lo_flags هو قناع بت يمكن أن يتضمن صفرًا أو أكثر مما يلي:
LO_FLAGS_READ_ONLY
جهاز الحلقة للقراءة فقط.
LO_FLAGS_AUTOCLEAR (منذ Linux 2.6.25)
سيتم حذف جهاز الحلقة تلقائيًا عند آخر إغلاق.
LO_FLAGS_PARTSCAN (منذ Linux 3.2)
السماح بمسح الأقسام تلقائيًا.
LO_FLAGS_DIRECT_IO (منذ Linux 4.10)
استخدام وضع الإدخال/الإخراج المباشر للوصول إلى الملف الأساسي.
القيم الوحيدة لـ lo_flags التي يمكن تعديلها بواسطة LOOP_SET_STATUS هي LO_FLAGS_AUTOCLEAR و LO_FLAGS_PARTSCAN.
LOOP_GET_STATUS
الحصول على حالة جهاز الحلقة. يجب أن تكون (ثالث) وسيطة ioctl(2) مؤشرًا إلى struct loop_info.
LOOP_CHANGE_FD (منذ Linux 2.6.5)
تبديل المخزن الأساسي لجهاز الحلقة إلى الملف الجديد المحدد بواسطة واصف الملف المقدم في (ثالث) وسيطة ioctl(2)، وهو عدد صحيح. هذه العملية ممكنة فقط إذا كان جهاز الحلقة للقراءة فقط وكان المخزن الأساسي الجديد بنفس حجم ونوع المخزن الأساسي القديم.
LOOP_SET_CAPACITY (منذ Linux 2.6.30)
تغيير حجم جهاز حلقة حي. يمكن للمرء تغيير حجم المخزن الأساسي الأساسي ثم استخدام هذه العملية حتى يتعرف مشغل الحلقة على الحجم الجديد. هذه العملية لا تأخذ أي وسيطة.
LOOP_SET_DIRECT_IO (منذ Linux 4.10)
تعيين وضع الإدخال/الإخراج المباشر على جهاز الحلقة، بحيث يمكن استخدامه لفتح الملف الأساسي. (ثالث) وسيطة ioctl(2) هي قيمة unsigned long. تمثل القيمة غير الصفرية وضع الإدخال/الإخراج المباشر.
LOOP_SET_BLOCK_SIZE (منذ Linux 4.14)
تعيين حجم الكتلة لجهاز الحلقة. (ثالث) وسيطة ioctl(2) هي قيمة unsigned long. يجب أن تكون هذه القيمة قوة للعدد 2 في النطاق [512، pagesize]؛ وإلا، تحدث خطأ EINVAL.
LOOP_CONFIGURE (منذ Linux 5.8)
إعداد وتكوين جميع معلمات جهاز الحلقة في خطوة واحدة باستخدام (ثالث) وسيطة ioctl(2). هذه الوسيطة هي مؤشر إلى هيكل loop_config، محدد في <linux/loop.h> كما يلي:

struct loop_config {
    __u32               fd;
    __u32               block_size;
    struct loop_info64  info;
    __u64               __reserved[8];
};

بالإضافة إلى القيام بما يمكن أن يقوم به LOOP_SET_STATUS، يمكن أيضًا استخدام LOOP_CONFIGURE للقيام بما يلي:
•
تعيين حجم الكتلة الصحيح فورًا عن طريق تعيين loop_config.block_size؛
•
طلب وضع الإدخال/الإخراج المباشر بشكل صريح عن طريق تعيين LO_FLAGS_DIRECT_IO في loop_config.info.lo_flags؛ و
•
طلب وضع القراءة فقط بشكل صريح عن طريق تعيين LO_FLAGS_READ_ONLY في loop_config.info.lo_flags.

منذ Linux 2.6، هناك عمليتان جديدتان لـ ioctl(2):

LOOP_SET_STATUS64
LOOP_GET_STATUS64
هذه العمليات مشابهة لـ LOOP_SET_STATUS و LOOP_GET_STATUS الموصوفة أعلاه ولكنها تستخدم هيكل loop_info64، الذي يحتوي على بعض الحقول الإضافية ونطاق أكبر لبعض الحقول الأخرى:

struct loop_info64 {
    uint64_t lo_device;           /* ioctl للقراءة/الكتابة */
    uint64_t lo_inode;            /* ioctl للقراءة/الكتابة */
    uint64_t lo_rdevice;          /* ioctl للقراءة/الكتابة */
    uint64_t lo_offset;
    uint64_t lo_sizelimit;  /* بايت، 0 == أقصى ما هو متاح */
    uint32_t lo_number;           /* ioctl للقراءة/الكتابة */
    uint32_t lo_encrypt_type;
    uint32_t lo_encrypt_key_size; /* ioctl للكتابة فقط */
    uint32_t lo_flags; i          /* ioctl للقراءة/الكتابة (للقراءة فقط قبل
                                     Linux 2.6.25) */
    uint8_t  lo_file_name[LO_NAME_SIZE];
    uint8_t  lo_crypt_name[LO_NAME_SIZE];
    uint8_t  lo_encrypt_key[LO_KEY_SIZE]; /* ioctl للكتابة فقط */
    uint64_t lo_init[2];
};

/dev/loop-control

منذ Linux 3.1، يوفر النواة جهاز /dev/loop-control، والذي يسمح لتطبيق بالعثور ديناميكيًا على جهاز مجاني، وإضافة وإزالة أجهزة الحلقة من النظام. لأداء هذه العمليات، يقوم المرء أولاً بفتح /dev/loop-control ثم يستخدم إحدى عمليات ioctl(2) التالية:

LOOP_CTL_GET_FREE
تخصيص أو العثور على جهاز حلقة مجاني للاستخدام. عند النجاح، يتم إرجاع رقم الجهاز كوصف
