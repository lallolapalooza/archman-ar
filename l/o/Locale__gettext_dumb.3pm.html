Locale__gettext_dumb(3pm)

الحزمة: perl-libintl-perl

التراخيص: LGPL

Locale::gettext_dumb(3) وثائق Perl المقدمة من المستخدم Locale::gettext_dumb(3)

الاسم

Locale::gettext_dumb - تنفيذ غير مدرك للمحلي للترجمة الرسائل Uniforum

الملخص

use Locale::gettext_dumb qw(:locale_h :libintl_h);
# في العادة، لن ترغب في تضمين هذه الوحدة مباشرة ولكن بهذه الطريقة:
use Locale::Messages;

my $selected = Locale::Messages->select_package ('gettext_dumb');
gettext $msgid;

dcpgettext $domainname, $msgctxt, $msgid, LC_MESSAGES;
npgettext $msgctxt, $msgid, $msgid_plural, $count;
dnpgettext $domainname, $msgctxt, $msgid, $msgid_plural, $count;
dcnpgettext $domainname, $msgctxt, $msgid, $msgid_plural, $count, LC_MESSAGES;

الوصف

هام! هذه الوحدة تجريبية. قد لا تعمل كما هو موصوف!

الوحدة Locale::gettext_dumb تفعل بالضبط نفس الشيء مثل Locale::gettext_xs(3pm) أو Locale::gettext_pp(3pm).

بينما تستخدم كلتا الوحدتين الأخريين POSIX::setlocale() لتحديد المحل المحدد حاليًا، فإن هذه الواجهة الخلفية تفحص فقط متغيرات البيئة LANGUAGE، LANG، LC_ALL، LC_MESSAGES (بهذا الترتيب)، عندما تحاول تحديد موقع كتالوج رسائل (ملف .mo).

تم تقديم هذه الفئة في libintl-perl 1.22.

الاستخدام

يجب عدم استخدام هذه الوحدة لبرامج سطح المكتب أو النصوص التي تعمل محليًا. لماذا؟ إذا استخدمت كتالوج رسائل على سبيل المثال باللغة الدنماركية بصيغة UTF-8 (da_DA.UTF8) ولكن إعداد المحل للنظام هو اللغة الروسية بصيغة KOI8-R (ru_RU.KOI8-R) فقد تنتج مخرجات غير صالحة، إما تسلسلات متعددة البايتات غير صالحة أو نص غير صالح، اعتمادًا على كيفية النظر إليها.

سيحدث هذا، عند خلط المخرجات من Locale::gettext_pp مع المخرجات التي تعتمد على المحل من نظام التشغيل مثل محتويات المتغير "$!"، وظائف تنسيق التاريخ والوقت (localtime()، gmtime()، POSIX::strftime() إلخ)، تنسيق الأرقام باستخدام printf() وما شابه.

سيناريو الاستخدام النموذجي يبدو كالتالي:

لديك تطبيق خادم (على سبيل المثال، تطبيق ويب) من المفترض أن يعرض مجموعة ثابتة من الرسائل بالعديد من اللغات. إذا كنت ترغب في القيام بذلك باستخدام Locale::gettext_xs(3pm) أو Locale::gettext_pp(3pm)، يجب عليك تثبيت بيانات المحل لجميع هذه اللغات. وإلا، فإن ترجمة الرسائل لن تعمل.

مع Locale::gettext_dumb(3pm) يمكنك تخفيف هذه المتطلبات، وعرض الرسائل لجميع اللغات التي لديك لها ملفات mo.

من ناحية أخرى، ستصل قريبًا إلى حدود هذا النهج. يتطلب تقريبًا أي تطبيق أكثر من مجرد ترجمة رسائل بسيطة للعولمة. تريد تنسيق التواريخ والأوقات، وتريد عرض الأرقام بالتنسيق الصحيح للغات المختارة، وقد ترغب في عرض رسائل خطأ النظام ("$!").

عمليًا، لا تزال Locale::gettext_dumb(3pm) مفيدة في هذه السيناريوهات. سيتعين على المستخدمين لديك التعايش مع حقيقة أن المخرجات المقدمة هي بلغات مختلفة أو لمحلات مختلفة، عندما لا يكون محل "هم" مثبتًا على نظامك.

الخلط بين المخرجات في مجموعات أحرف مختلفة أكثر خطورة، ولكن يمكن تجنب ذلك بسهولة. فقط تأكد من أن Locale::gettext_dump يستخدم UTF-8 (على سبيل المثال، عن طريق تعيين متغير البيئة OUTPUT_CHARSET أو عن طريق استدعاء bind_textdomain_codeset()) وتأكد من أن محل النظام يستخدم أيضًا UTF-8، على سبيل المثال "en_US.UTF8". إذا فشل ذلك، فقم بالتبديل إلى محل يستخدم مجموعة فرعية من UTF-8. عمليًا، سيكون هذا US-ASCII، مجموعة الأحرف المستخدمة في المحل الافتراضي "C" أو "POSIX".

سيقوم تطبيقك بعد ذلك، إلى حد ما، بخلط المخرجات لتوطينات أو لغات مختلفة. ولكن هذا تحت سيطرتك بالكامل.

مثال

انظر أعلاه! في العادة، لا يجب عليك استخدام هذه الوحدة! ومع ذلك، لنفترض أنك قرأت التحذيرات. في تطبيق ويب، ستفعل شيئًا كهذا:

use Locale::TextDomain qw (com.example.yourapp);
use Locale::Messages qw (nl_putenv LC_ALL bindtextdomain
                         bind_textdomain_codeset);
use Locale::Util qw (web_set_locale);
use POSIX qw (setlocale);

# حاول أولاً التبديل إلى المحل الذي طلبه المستخدم. إذا
# كنت تعرفه، يمكنك محاولة تمريره إلى setlocale هكذا:
#
#   my $hardcoded_locale = 'fr_FR.UTF-8';
#   my $success = POSIX::setlocale (LC_ALL, $hardcoded_locale);
#
# ومع ذلك، نحاول أن ندع libintl-perl يقوم بعمل أفضل لنا:
my $success = web_set_locale $ENV{HTTP_ACCEPT_LANGUAGE},
                             $ENV{HTTP_ACCEPT_CHARSET};
# ملاحظة: إذا كان تطبيقك يجبر استخدام UTF-8 لمخرجاته
# يجب عليك تمرير 'UTF-8' كوسيط ثانٍ لـ web_set_locale
# بدلاً من $ENV{HTTP_ACCEPT_CHARSET}.

if (!$success) {
    # لم يعمل. قم بالتبديل إلى الواجهة الصماء
    # لـ Locale::Messages.
    Locale::Messages->select_package ('gettext_dumb');

    # وحاول التبديل إلى محل افتراضي:
    if (!setlocale (LC_ALL, 'en_US.UTF-8')) {
        # لا يزال لا يوجد حظ. فرض US-ASCII على الأقل:
        setlocale (LC_ALL, 'C');
    }
    bind_textdomain_codeset 'com.example.yourapp', 'utf-8';
}

إذا كان تطبيقك يجبر استخدام UTF-8، يجب عليك تجاهل متغير البيئة
