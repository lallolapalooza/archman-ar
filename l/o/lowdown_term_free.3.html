lowdown_term_free(3)

الحزمة: lowdown

التراخيص: ISC

LOWDOWN_TERM_FREE(3) دليل وظائف المكتبة LOWDOWN_TERM_FREE(3)

الاسم

lowdown_term_free — تحرير عارض طرفية Markdown

المكتبة

المكتبة "liblowdown"

الملخص

#include <sys/queue.h>
#include <stdio.h>
#include <lowdown.h>

void
lowdown_term_free(void *arg);

الوصف

يحرر عارض الطرفية الذي تم إنشاؤه بواسطة lowdown_term_new(3). إذا كان arg هو NULL، فإن الدالة لا تفعل شيئًا.

أمثلة

يقوم المثال التالي بتحليل b بطول bsz ويخرج بتنسيق طرفية ANSI.

struct lowdown_buf *out;
struct lowdown_doc *doc;
struct lowdown_node *n;
void *rndr;

if (setlocale(LC_CTYPE, "en_US.UTF-8") == NULL)
	err(1, NULL);

if ((doc = lowdown_doc_new(NULL)) == NULL)
	err(1, NULL);
if ((n = lowdown_doc_parse(doc, NULL, b, bsz, NULL)) == NULL)
	err(1, NULL);
if ((out = lowdown_buf_new(256)) == NULL)
	err(1, NULL);
if ((rndr = lowdown_term_new(NULL)) == NULL)
	err(1, NULL);
if (!lowdown_term_rndr(out, rndr, n))
	err(1, NULL);

fwrite(out->data, 1, out->size, stdout);

lowdown_term_free(rndr);
lowdown_buf_free(out);
lowdown_node_free(n);
lowdown_doc_free(doc);

راجع أيضًا

lowdown(3)، lowdown_term_new(3)
