loop(4) دليل مشغلات الأجهزة loop(4)

اسم

loop, loop-control - أجهزة الحلقة

ملخص

#include <linux/loop.h>

وصف

يقوم جهاز الحلقة بربط كتل بياناته ليس بجهاز فعلي مثل القرص الصلب أو محرك الأقراص الضوئية، ولكن بكتل ملف عادي في نظام ملفات أو بجهاز كتلة آخر. يمكن أن يكون هذا مفيدًا على سبيل المثال لتوفير جهاز كتلة لصورة نظام ملفات مخزنة في ملف، بحيث يمكن تحميلها باستخدام الأمر mount(8). يمكنك القيام بذلك


$ dd if=/dev/zero of=file.img bs=1MiB count=10;
$ sudo losetup /dev/loop4 file.img;
$ sudo mkfs -t ext4 /dev/loop4;
$ sudo mkdir /myloopdev;
$ sudo mount /dev/loop4 /myloopdev;

انظر losetup(8) لمثال آخر.

يمكن تحديد وظيفة نقل لكل جهاز حلقة لأغراض التشفير وفك التشفير.

توفر عمليات ioctl(2) التالية جهاز كتلة الحلقة:

LOOP_SET_FD
ربط جهاز الحلقة بالملف المفتوح الذي تم تمرير واصف الملف الخاص به كوسيطة ioctl(2) (الثالثة).
LOOP_CLR_FD
فصل جهاز الحلقة عن أي واصف ملف.
LOOP_SET_STATUS
تعيين حالة جهاز الحلقة باستخدام الوسيطة (الثالثة) لـ ioctl(2). هذه الوسيطة هي مؤشر إلى بنية loop_info، المعرفة في <linux/loop.h> على النحو التالي:

struct loop_info {
    int           lo_number;      /* ioctl r/o */
    dev_t         lo_device;      /* ioctl r/o */
    unsigned long lo_inode;       /* ioctl r/o */
    dev_t         lo_rdevice;     /* ioctl r/o */
    int           lo_offset;
    int           lo_encrypt_type;
    int           lo_encrypt_key_size;  /* ioctl w/o */
    int           lo_flags;       /* ioctl r/w (r/o before
                                     Linux 2.6.25) */
    char          lo_name[LO_NAME_SIZE];
    unsigned char lo_encrypt_key[LO_KEY_SIZE];
                                  /* ioctl w/o */
    unsigned long lo_init[2];
    char          reserved[4];
};

يجب أن يكون نوع التشفير (lo_encrypt_type) أحد القيم التالية: LO_CRYPT_NONE، LO_CRYPT_XOR، LO_CRYPT_DES، LO_CRYPT_FISH2، LO_CRYPT_BLOW، LO_CRYPT_CAST128، LO_CRYPT_IDEA، LO_CRYPT_DUMMY، LO_CRYPT_SKIPJACK، أو (منذ Linux 2.6.0) LO_CRYPT_CRYPTOAPI.
حقل lo_flags هو قناع بت يمكن أن يشمل صفرًا أو أكثر من القيم التالية:
LO_FLAGS_READ_ONLY
جهاز الحلقة للقراءة فقط.
LO_FLAGS_AUTOCLEAR (منذ Linux 2.6.25)
سيتم حذف جهاز الحلقة تلقائيًا عند إغلاقه للمرة الأخيرة.
LO_FLAGS_PARTSCAN (منذ Linux 3.2)
السماح بالمسح التلقائي للأقسام.
LO_FLAGS_DIRECT_IO (منذ Linux 4.10)
استخدام وضع الإدخال/الإخراج المباشر للوصول إلى الملف الخلفي.
القيم الوحيدة لـ lo_flags التي يمكن تعديلها بواسطة LOOP_SET_STATUS هي LO_FLAGS_AUTOCLEAR و LO_FLAGS_PARTSCAN.
LOOP_GET_STATUS
الحصول على حالة جهاز الحلقة. يجب أن تكون الوسيطة (الثالثة) لـ ioctl(2) مؤشرًا إلى struct loop_info.
LOOP_CHANGE_FD (منذ Linux 2.6.5)
تبديل المخزن الخلفي لجهاز الحلقة إلى الملف الجديد الذي تم تحديده بواسطة واصف الملف المقدم في الوسيطة (الثالثة) لـ ioctl(2)، وهو عدد صحيح. هذه العملية ممكنة فقط إذا كان جهاز الحلقة للقراءة فقط وكان المخزن الخلفي الجديد بنفس حجم ونوع المخزن الخلفي القديم.
LOOP_SET_CAPACITY (منذ Linux 2.6.30)
تغيير حجم جهاز حلقة مباشر. يمكن للمرء تغيير حجم المخزن الخلفي الأساسي ثم استخدام هذه العملية حتى يتعرف مشغل الحلقة على الحجم الجديد. هذه العملية لا تأخذ وسيطة.
LOOP_SET_DIRECT_IO (منذ Linux 4.10)
تعيين وضع الإدخال/الإخراج المباشر (DIRECT I/O) على جهاز الحلقة، بحيث يمكن استخدامه لفتح الملف الخلفي. الوسيطة (الثالثة) لـ ioctl(2) هي قيمة غير موقعة (unsigned long). قيمة غير صفرية تمثل وضع الإدخال/الإخراج المباشر.
LOOP_SET_BLOCK_SIZE (منذ Linux 4.14)
تعيين حجم الكتلة لجهاز الحلقة. الوسيطة (الثالثة) لـ ioctl(2) هي قيمة غير موقعة (unsigned long). يجب أن تكون هذه القيمة قوة للرقم 2 في النطاق [512،pagesize]؛ بخلاف ذلك، ستكون هناك نتيجة خطأ EINVAL.
LOOP_CONFIGURE (منذ Linux 5.8)
إعداد وتهيئة جميع معلمات جهاز الحلقة في خطوة واحدة باستخدام الوسيطة (الثالثة) لـ ioctl(2). هذه الوسيطة هي مؤشر إلى بنية loop_config، المعرفة في <linux/loop.h> على النحو التالي:

struct loop_config {
    __u32               fd;
    __u32               block_size;
    struct loop_info64  info;
    __u64               __reserved[8];
};

بالإضافة إلى القيام بما يمكن لـ LOOP_SET_STATUS القيام به، يمكن لـ LOOP_CONFIGURE أيضًا القيام بما يلي:
•
تعيين حجم الكتلة الصحيح فورًا عن طريق تعيين loop_config.block_size؛
•
طلب وضع الإدخال/الإخراج المباشر بشكل صريح عن طريق تعيين LO_FLAGS_DIRECT_IO في loop_config.info.lo_flags؛ و
•
طلب وضع القراءة فقط بشكل صريح عن طريق تعيين LO_FLAGS_READ_ONLY في loop_config.info.lo_flags.

منذ Linux 2.6، هناك عمليتان جديدتان لـ ioctl(2):

LOOP_SET_STATUS64
LOOP_GET_STATUS64
هذه مشابهة لـ LOOP_SET_STATUS و LOOP_GET_STATUS الموصوفة أعلاه ولكنها تستخدم البنية loop_info64، والتي تحتوي على بعض الحقول الإضافية ونطاق أكبر لبعض الحقول الأخرى:

struct loop_info64 {
    uint64_t lo_device;           /* ioctl r/o */
    uint64_t lo_inode;            /* ioctl r/o */
    uint64_t lo_rdevice;          /* ioctl r/o */
    uint64_t lo_offset;
    uint64_t lo_sizelimit;  /* bytes, 0 == max available */
    uint32_t lo_number;           /* ioctl r/o */
    uint32_t lo_encrypt_type;
    uint32_t lo_encrypt_key_size; /* ioctl w/o */
    uint32_t lo_flags; i          /* ioctl r/w (r/o before
                                     Linux 2.6.25) */
    uint8_t  lo_file_name[LO_NAME_SIZE];
    uint8_t  lo_crypt_name[LO_NAME_SIZE];
    uint8_t  lo_encrypt_key[LO_KEY_SIZE]; /* ioctl w/o */
    uint64_t lo_init[2];
};

/dev/loop-control

منذ Linux 3.1، يوفر النواة جهاز /dev/loop-control، الذي يسمح للتطبيق بالعثور ديناميكيًا على جهاز فارغ، وإضافة أجهزة حلقة وإزالتها من النظام. لتنفيذ هذه العمليات، يجب أولاً فتح /dev/loop-control ثم استخدام إحدى عمليات ioctl(2) التالية:

LOOP_CTL_GET_FREE
تخصيص أو العثور على جهاز حلقة فارغ للاستخدام. عند النجاح، يتم إرجاع رقم الجهاز كنتيجة للمكالمة. هذه العملية لا تأخذ وسيطة.
LOOP_CTL_ADD
إضافة جهاز حلقة جديد يتم تحديد رقم جهازه

هذا الموقع غير مرتبط بـ Arch Linux ولا تابع له بأي شكل من الأشكال. المحتوى هو ترجمة فقط لصفحات دليل Arch المنشورة على man.archlinux.org.

loop-control(4) — archlinux man pages

NAME

loop-control - loop device control

SYNOPSIS

#include <linux/loop>

ioctl(loopctl_fd, LOOP_CTL_GET_FREE, &devnr);

ioctl(loopctl_fd, LOOP_CTL_ADD, &devnr);

ioctl(loopctl_fd, LOOP_CTL_REMOVE, &devnr);

DESCRIPTION

The loop device control interface is exposed via the special device file /dev/loop-control.

The following ioctl(2) operations are supported:

LOOP_CTL_GET_FREE
Get the number of the next available (unused) loop device. The third argument is a pointer to a long integer into which the device number will be stored. The device number is then used to construct the name of the loop device file (e.g., /dev/loop0). On success, the device number is returned as the result of the call. If there are no free loop devices, the call fails with the error ENFILE.
LOOP_CTL_ADD
Add a new loop device. The third ioctl(2) argument specifies the device number of the loop device to be added as a long integer. This is typically a device number obtained via LOOP_CTL_GET_FREE. On success, the device index is returned as the result of the call. If the device is already allocated, the call fails with the error EEXIST.
LOOP_CTL_REMOVE
Remove the loop device whose device number is specified as a long integer in the third ioctl(2) argument. On success, the device number is returned as the result of the call. If the device is in use, the call fails with the error EBUSY.

FILES

/dev/loop*
The loop block special device files.

EXAMPLES

The program below uses the /dev/loop-control device to find a free loop device, opens the loop device, opens a file to be used as the underlying storage for the device, and then associates the loop device with the backing store. The following shell session demonstrates the use of the program:


$ dd if=/dev/zero of=file.img bs=1MiB count=10;
10+0 records in
10+0 records out
10485760 bytes (10 MB) copied, 0.00609385 s, 1.7 GB/s
$ sudo ./mnt_loop file.img;
loopname = /dev/loop5

Program source

#include <err.h>
#include <fcntl.h>
#include <linux/loop.h>
#include <sys/ioctl.h>
#include <stdio.h>
#include <stdlib.h>
#include <unistd.h>
int
main(int argc, char *argv[])
{
    int loopctlfd, loopfd, backingfile;
    long devnr;
    char loopname[4096];
    if (argc != 2) {
        fprintf(stderr, "Usage: %s backing-file\n", argv[0]);
        exit(EXIT_FAILURE);
    }
    loopctlfd = open("/dev/loop-control", O_RDWR);
    if (loopctlfd == -1)
        err(EXIT_FAILURE, "open: /dev/loop-control");
    devnr = ioctl(loopctlfd, LOOP_CTL_GET_FREE);
    if (devnr == -1)
        err(EXIT_FAILURE, "ioctl-LOOP_CTL_GET_FREE");
    sprintf(loopname, "/dev/loop%ld", devnr);
    printf("loopname = %s\n", loopname);
    loopfd = open(loopname, O_RDWR);
    if (loopfd == -1)
        err(EXIT_FAILURE, "open: loopname");
    backingfile = open(argv[1], O_RDWR);
    if (backingfile == -1)
        err(EXIT_FAILURE, "open: backing-file");
    if (ioctl(loopfd, LOOP_SET_FD, backingfile) == -1)
        err(EXIT_FAILURE, "ioctl-LOOP_SET_FD");
    exit(EXIT_SUCCESS);
}
