Locale__Codes__LangVar(3pm)

الحزمة: perl-locale-codes

التراخيص: GPL, PerlArtistic

Locale::Codes::LangVar(3) وثائق Perl المساهمة من المستخدم Locale::Codes::LangVar(3)

الاسم

Locale::Codes::LangVar - وحدة للتعامل مع مجموعات رموز langvar

الملخص

use Locale::Codes::LangVar;
$name = code2langvar(CODE);
$code = langvar2code(NAME);
@codes   = all_langvar_codes();
@names   = all_langvar_names();

الوصف

توفر هذه الوحدة الوصول إلى مجموعات رموز langvar.

يرجى الرجوع إلى وثيقة Locale::Codes::Types للحصول على وصف لمجموعات الرموز المتاحة.

تأخذ معظم الروتينات وسيطًا إضافيًا اختياريًا يحدد مجموعة الرموز المراد استخدامها. يمكن تحديد مجموعة الرموز باستخدام اسم مجموعة رموز، أو الثابت perl المحدد في الوثيقة أعلاه. إذا لم يتم تحديدها، فسيتم استخدام مجموعة الرموز الافتراضية.

الروتينات

تقوم جميع الروتينات في هذه الوحدة باستدعاء الطريقة المناسبة في وحدة Locale::Codes، باستخدام كائن من النوع: langvar. يرجى الرجوع إلى وثائق وحدة Locale::Codes للحصول على تفاصيل حول كل وظيفة.

يتم تصدير الوظائف التالية تلقائيًا:

code2langvar(CODE [,CODESET] [,'retired'])
انظر code2name في Locale::Codes
code2langvars(CODE [,CODESET])
انظر code2names في Locale::Codes
langvar2code(NAME [,CODESET] [,'retired'])
انظر name2code في Locale::Codes
langvar_code2code(CODE ,CODESET ,CODESET2)
انظر code2code في Locale::Codes
all_langvar_codes([CODESET] [,'retired'])
انظر all_codes في Locale::Codes
all_langvar_names([CODESET] [,'retired'])
انظر all_names في Locale::Codes

الوظائف التالية غير مصدّرة ويجب استدعاؤها بالكامل مع اسم الحزمة:

Locale::Codes::Langvar::show_errors(FLAG)
بشكل افتراضي، سينتج الإدخال غير الصالح نتائج فارغة، ولكن لا توجد أخطاء. عن طريق تمرير قيمة غير صفرية لـ FLAG، سيتم إنتاج الأخطاء.

انظر show_errors في Locale::Codes ولكن لاحظ أن الافتراضي للوحدات غير الكائنية هو عدم إنتاج أخطاء.

Locale::Codes::Langvar::rename_langvar(CODE ,NEW_NAME [,CODESET])
انظر rename_code في Locale::Codes
Locale::Codes::Langvar::add_langvar(CODE ,NAME [,CODESET])
انظر add_code في Locale::Codes
Locale::Codes::Langvar::delete_langvar(CODE [,CODESET])
انظر delete_code في Locale::Codes
Locale::Codes::Langvar::add_langvar_alias(NAME ,NEW_NAME)
انظر add_alias في Locale::Codes
Locale::Codes::Langvar::delete_langvar_alias(NAME)
انظر delete_alias في Locale::Codes
Locale::Codes::Langvar::rename_langvar_code(CODE ,NEW_CODE [,CODESET])
انظر replace_code في Locale::Codes
Locale::Codes::Langvar::add_langvar_code_alias(CODE ,NEW_CODE [,CODESET])
انظر add_code_alias في Locale::Codes
Locale::Codes::Langvar::delete_langvar_code_alias(CODE [,CODESET])
انظر delete_code_alias في Locale::Codes
