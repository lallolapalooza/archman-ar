FBB::LocalServerSocket(3bobcat) مقبس خادم نطاق Unix FBB::LocalServerSocket(3bobcat)

الاسم

FBB::LocalServerSocket - مقبس خادم نطاق Unix يقبل طلبات الاتصال

الملخص

#include <bobcat/localserversocket>
خيار الربط: -lbobcat

الوصف

يُعرّف FBB::LocalServerSocket مقبس خادم نطاق Unix، يستمع لطلبات الاتصال من المضيف المحلي باستخدام مقبس نطاق Unix. يمكن قبول طلبات الاتصال إما في الأوضاع الحاجزة أو غير الحاجزة. عند قبول اتصال يُعاد مقبس يمكن استخدامه لقراءة المعلومات من العميل الذي طلب الاتصال أو كتابة المعلومات إليه. المقبس الذي يتم إتاحته هو واصف ملف يمكن استخدامه لتهيئة std::istream و/أو std::ostream. يُستخدم std::istream لقراءة المعلومات من عملية العميل؛ ويُستخدم std::ostream لإرسال المعلومات إلى عملية العميل. وبما أن المقبس يمكن اعتباره واصف ملف فإن الفئات المتاحة FBB::IFdStream وFBB::IFdStreamBuf وFBB::OFdStream وFBB::OFdStreamBuf يمكن استخدامها بشكل مفيد هنا. لاحظ أن توفر مقبس لا يعني أن هذا يحدد بروتوكول الاتصال. لا يزال الأمر مسؤولية المبرمج للامتثال لبروتوكول موجود أو تنفيذ بروتوكول مصمم خصيصًا. الحالة الأخيرة تعني أن تسلسل عمليات الإدخال والإخراج يُحدده المبرمج.

التعداد

يُعرّف التعداد التالي في الفئة FBB::LocalServerSocket:

enum Socket
يحمل هذا التعداد قيمتين:

o
KEEP:
عند تحديد هذه القيمة في وقت البناء، لا يُزال الملف الذي يمثل مقبس نطاق Unix عند تدمير LocalServerSocket.
o
UNLINK:
عند تحديد هذه القيمة في وقت البناء، يُزال الملف الذي يمثل مقبس نطاق Unix عند تدمير LocalServerSocket. وبما أنه لا يمكن إعادة استخدام المقبس، فهذه هي القيمة الافتراضية المستخدمة مع باني LocalServerSocket.

الباني

o
LocalServerSocket():
ينشئ هذا الباني كائن FBB::LocalServerSocket فارغًا (غير فعّال). قبل استخدامه، يجب استدعاء العضو open() (انظر أدناه).
o
LocalServerSocket(string const &name, Socket action = UNLINK) throw (Exception):
يهيّئ هذا الباني كائن FBB::LocalServerSocket، الذي سيستمع لطلبات الاتصال باستخدام مقبس نطاق Unix المسمّى. يُطرح FBB::Exception إذا تعذّر إنشاء المقبس. إذا أُعطي الباني وسيطًا ثانيًا FBB::LocalServerSocket::KEEP، فلا يُفصل المقبس المُنشأ عند تدمير كائن FBB::LocalServerSocket. إنشاء المقبس لا يعني أن كائن FBB::LocalServerSocket يستمع فعليًا للاتصالات. لبدء الاستماع، يجب استدعاء العضو listen().

بانيات النسخ والنقل (ومعاملات الإسناد) غير متاحة.

دوال الأعضاء

o
size_t accept():
يُعيد العضو accept() قيمة size_t وهي واصف ملف (مقبس) يمكن استخدامه للتواصل مع العميل الذي طلب الاتصال. في البرامج الأكثر تعقيدًا يمكن تمرير واصف الملف (المقبس) المُعاد إلى فئة مشتقة من FBB::Fork، تتعامل مع التواصل مع العملية الفرعية كعملية (فرعية) منفصلة.
o
void listen(size_t backlog = 5, bool blocking = true):
يحدد العضو listen() الطريقة التي سيستمع بها FBB::LocalServerSocket للعملاء الذين يطلبون اتصالًا. يمكن استخدامه مرة واحدة فقط مع FBB::LocalServerSocket. يُطرح كائن FBB::Exception إذا فشل الاستماع.
يحدد وسيط backlog للعضو listen() حجم قائمة الانتظار الداخلية لـ FBB::LocalServerSocket التي قد تُخزَّن فيها طلبات الاتصال بانتظار دورها للخدمة. عندما تنتظر backlog طلبات ويصل طلب آخر، فإن ذلك الطلب يُفقد.
يُستخدم الوسيط الثاني للعضو، blocking، للتحكم في وضع الحجب. افتراضيًا، يُستخدم الحجب، وسينتظر listen() حتى يتم إنشاء اتصال. هذا مقبول في الحالات التي يتصل فيها العملاء بشكل غير متكرر ولفترات زمنية قصيرة نسبيًا. بخلاف ذلك، في البرامج الأكثر تعقيدًا، يمكن استخدام كائن FBB::Selector لاستشعار الإدخال على مقبس الخادم و/أو على مقابس العملاء المختلفة.
o
void open(string const &name, Socket action = UNLINK):
يهيّئ هذا العضو كائن FBB::LocalServerSocket للاستخدام. يجب استخدامه فقط بالاقتران مع الباني الافتراضي. بعد open() سيتمكن كائن FBB:::LocalServerSocket من الاستماع لطلبات الاتصال باستخدام مقبس نطاق Unix المسمّى. يُطرح FBB::Exception إذا تعذّر إنشاء المقبس. إذا قُدّم الوسيط الثاني FBB::LocalServerSocket::KEEP، فلا يُفصل المقبس المُنشأ عند تدمير كائن FBB::LocalServerSocket. إنشاء المقبس لا يعني أن كائن FBB::LocalServerSocket يستمع فعليًا للاتصالات. لبدء الاستماع، يجب استدعاء العضو listen() بعد ذلك.

مثال

انظر أيضًا مثال localclientsocket(3bobcat).

    #include <iostream>
    #include <bobcat/localserversocket>
    #include <bobcat/ifdstream>
    #include <bobcat/ofdstream>
    using namespace std;
    using namespace FBB;
    int main(int argc, char **argv)
    try
    {
        if (argc == 1)
        {
            cerr << "قدّم اسم ملف محلي، مثل /tmp/uds\n";
            return 1;
        }
        LocalServerSocket server(argv[1]);
        cerr << "الخادم يستخدم `" << argv[1] << "’" << endl;
        cout <<
            "ينتهي الخادم عندما يتلقى `q’ واحدة على سطر\n"
            "ينتهي الاتصال عندما لا يُستقبل أي إدخال بعد الآن.\n"
            "ثم يصبح اتصال آخر ممكنًا" << endl;
        server.listen();                // الاستماع في وضع الحجب
        while (true)
        {
            int fd = server.accept();
            cerr << "واصف ملف العميل = " << fd << ", " << endl;
            IFdStream in(fd);           // تدفق للقراءة من العميل
            OFdStream out(fd);          // تدفق للكتابة إلى العميل
            string cmd;
            while (getline(in, cmd))
            {
                cout << "تم الاستلام: " << cmd << endl;
                out << "تم الاستلام: " << cmd << "\r" << endl;
                if (cmd[0] == ’q’)
                    return 0;
            }
            cout << "جاهز لاتصال آخر\n";
        }
    }
    catch (Exception const &err)
    {
        cerr <<
            err.what() << endl <<
            "مقبس الخادم على المنفذ " << argv[1] <<
            " لا يمكن فتحه" << endl;
        return -1;
    }
