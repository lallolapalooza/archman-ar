location(1)

الحزمة: python-location

التراخيص: LGPL-2.1-or-later

LOCATION(1)   LOCATION(1)

اسم

location - استعلام قاعدة بيانات الموقع

ملخص

location export --directory=DIR [--format=FORMAT] [--family=ipv6|ipv4] [ASN|CC ...]
location get-as ASN [ASN...]
location list-countries [--show-name] [--show-continent]
location list-networks-by-as ASN
location list-networks-by-cc COUNTRY_CODE
location list-networks-by-flags [--anonymous-proxy|--satellite-provider|--anycast|--drop]
location lookup ADDRESS [ADDRESS...]
location search-as STRING
location update [--cron=daily|weekly|monthly]
location verify
location version

الوصف

يسترد location معلومات من قاعدة بيانات الموقع. يمكن استخدام هذه البيانات لتحديد موقع شخص ما على الإنترنت ولبناء مجموعات قواعد جدار الحماية لحظر الوصول من ASes أو بلدان معينة.

خيارات

--database FILE, -d FILE

مسار قاعدة البيانات التي يتم فتحها.

إذا تم تجاهل هذا الخيار، فسيتم فتح قاعدة بيانات النظام.

--quiet

تمكين الوضع الهادئ

--debug

تمكين وضع التصحيح

أوامر

export [--directory=DIR] [--format=FORMAT] [--family=ipv6|ipv4] [ASN|CC ...]

يقوم هذا الأمر بتصدير قاعدة البيانات بأكملها إلى الدليل المحدد.

يمكن تصفية الإخراج عن طريق تصدير عائلة عناوين معينة فقط، أو عن طريق تمرير قائمة من رموز البلدان و/أو ASNs. الافتراضي هو تصدير جميع البلدان المعروفة.

يمكن اختيار تنسيق الإخراج باستخدام المعلمة --format. للتنسيقات الممكنة، يرجى الرجوع أدناه.

إذا تم تجاهل الخيار --directory، فسيتم كتابة الإخراج إلى stdout، وهو أمر مفيد عندما تريد تحميل أي تصديرات مخصصة مباشرة في nftables أو ipset.

get-as ASN [ASN...]

يعيد هذا الأمر اسم المؤسسة المالكة للنظام المستقل (Autonomous System).

list-countries [--show-name] [--show-continent]

يسرد جميع البلدان المعروفة لقاعدة البيانات.

مع المعلمات الاختيارية --show-name و --show-continent، سيتم طباعة الاسم ورمز القارة أيضًا.

list-networks-by-as [--family=[ipv6|ipv4]] [--format=FORMAT] ASN

يسرد جميع الشبكات التي تنتمي إلى هذا النظام المستقل.

يمكن استخدام المعلمة --family لتصفية الإخراج لعناوين IPv6 أو IPv4 فقط.

يمكن للمعلمة --format تغيير الإخراج بحيث يمكن تحميله مباشرة في برامج أخرى. للتفاصيل انظر أدناه.

list-networks-by-cc [--family=[ipv6|ipv4]] [--format=FORMAT] COUNTRY_CODE

يسرد جميع الشبكات التي تنتمي إلى بلد معين.

يجب ترميز البلد بتدوين ISO3166 Alpha-2.

انظر أعلاه لاستخدام المعلمتين --family و --format.

list-networks-by-flags [--family=[ipv6|ipv4]] [--format=FORMAT] [--anonymous-proxy|--satellite-provider|--anycast|--drop]

يسرد جميع الشبكات التي تحمل علامة معينة.

انظر أعلاه لاستخدام المعلمتين --family و --format.

list-bogons [--family=[ipv6|ipv4]] [--format=FORMAT]

يسرد جميع الـ bogons (أي الشبكات غير المعروفة لقاعدة البيانات).

انظر أعلاه لاستخدام المعلمتين --family و --format.

lookup ADDRESS [ADDRESS...]

يعيد هذا الأمر الشبكة التي تم العثور فيها على عنوان IP المحدد وكذلك نظامها المستقل إذا كانت هذه المعلومات متاحة.

search-as STRING

يسرد جميع الأنظمة المستقلة التي تطابق السلسلة المحددة.

سيتم إجراء البحث دون مراعاة حالة الأحرف.

update

سيحاول هذا الأمر تحديث قاعدة البيانات المحلية.

سينتهي برمز عودة صفر إذا تم تحديث قاعدة البيانات بنجاح. 1 عند حدوث خطأ، 2 عند استدعاء غير صالح، و 3 إذا كانت قاعدة البيانات هي بالفعل أحدث إصدار.

يتيح الخيار --cron الحد من التحديثات مرة واحدة يوميًا (daily)، مرة واحدة أسبوعيًا (weekly)، أو مرة واحدة شهريًا (monthly). إذا تم استدعاء المهمة، ولكن تم تحديث قاعدة البيانات مؤخرًا، فستتخطى التحديث.

verify

يتحقق من قاعدة البيانات التي تم تنزيلها.

version

يعرض معلومات الإصدار لقاعدة البيانات التي تم تنزيلها.

--help

يعرض نص مساعدة قصير حول استخدام هذا البرنامج.

--version

يعرض إصدار البرنامج ويخرج.

رموز الخروج

سيخرج الأمر location عادةً بالرمز صفر. إذا كانت هناك مشكلة ولم يمكن تنفيذ الإجراء المطلوب، فسيكون رمز الخروج غير مساوٍ للصفر.

التنسيقات

تسمح بعض الأوامر بتحديد تنسيق الإخراج. هذا مفيد إذا كان يجب استيراد البيانات المصدرة إلى مرشح الحزم على سبيل المثال. يتم فهم التنسيقات التالية:

•list (الافتراضي): يسرد فقط جميع الشبكات، واحدة في كل سطر
•ipset: لـ ipset
•nftables: لـ nftables
•xt_geoip: يعيد قائمة بالشبكات ليتم تحميلها في وحدة نواة xt_geoip

كيف يعمل

يقوم برنامج التنزيل بفحص سجل DNS لأحدث إصدار من قاعدة البيانات. ثم سيحاول تنزيل ملف بهذا الإصدار من خادم مرآة. إذا كان الملف الذي تم تنزيله قديمًا، فسيتم تجربة المرآة التالية حتى نجد ملفًا حديثًا بما فيه الكفاية.
