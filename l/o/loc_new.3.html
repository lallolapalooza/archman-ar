loc_new(3)

الحزمة: libloc

التراخيص: LGPL-2.1-or-later

LOC_NEW(3)   LOC_NEW(3)

الاسم

loc_new - إنشاء سياق libloc جديد

الملخص

#include <libloc/libloc.h>

struct loc_ctx;

int loc_new(struct loc_ctx** ctx);

struct loc_ctx* loc_ref(struct loc_ctx* ctx);

struct loc_ctx* loc_unref(struct loc_ctx* ctx);

الوصف

كل عملية في libloc تتطلب إعداد سياق أولاً. يتم ذلك عن طريق استدعاء loc_new(3).

في كل مرة يحتفظ فيها جزء آخر من شفرتك بمرجع للسياق، ستحتاج إلى استدعاء loc_ref() لزيادة عداد المراجع. إذا لم تعد بحاجة إلى السياق، فستحتاج إلى استدعاء loc_unref().
