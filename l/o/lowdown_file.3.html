lowdown_file(3)

الحزمة: lowdown

التراخيص: ISC

LOWDOWN_FILE(3) دليل وظائف المكتبة LOWDOWN_FILE(3)

الاسم

lowdown_file — تحليل ملف Markdown إلى إخراج منسق

المكتبة

المكتبة “liblowdown”

ملخص

#include <sys/queue.h>
#include <stdio.h>
#include <lowdown.h>

int
lowdown_file(const struct lowdown_opts *opts, FILE *in, char **ret, size_t *retsz, struct lowdown_metaq *metaq);

الوصف

يقوم بتحليل تدفق ملف lowdown(5) in إلى مخزن إخراج ret بحجم retsz وفقًا للإعدادات opts. يتم تحديد تنسيق الإخراج بواسطة opts->type. إذا تم تعيين LOWDOWN_METADATA في opts->feat وكان metaq ليس NULL، يتم ملء metaq بالبيانات التعريفية المقدمة بالتنسيق المحدد.

عند النجاح، يكون المتصل مسؤولاً عن تحرير ret و metaq.

قيم الإرجاع

يعيد صفرًا عند الفشل، وغير صفر عند النجاح. عند الفشل، تكون القيم التي يشير إليها res و rsz غير محددة.

أمثلة

يقوم ما يلي بتحليل الإدخال القياسي إلى مستند HTML5 مستقل. يُمكّن التعليقات التوضيحية، والروابط التلقائية، والجداول، والنص الفائق، والخطوط المتقاطعة، وكتل التعليمات البرمجية المسوّرة، والـ commonmark، وقوائم التعريف، وسمات الصور الموسعة، ومعالجة البيانات التعريفية. يمر الإخراج عبر HTML خام وله طباعة ذكية.

struct lowdown_opts opts;
char *buf;
size_t bufsz;

memset(&opts, 0, sizeof(struct lowdown_opts));
opts.type = LOWDOWN_HTML;
opts.feat = LOWDOWN_FOOTNOTES |
	LOWDOWN_AUTOLINK |
	LOWDOWN_TABLES |
	LOWDOWN_SUPER |
	LOWDOWN_STRIKE |
	LOWDOWN_FENCED |
	LOWDOWN_COMMONMARK |
	LOWDOWN_DEFLIST |
	LOWDOWN_IMG_EXT |
	LOWDOWN_METADATA;
opts.oflags = LOWDOWN_HTML_HEAD_IDS |
	LOWDOWN_HTML_NUM_ENT |
	LOWDOWN_HTML_OWASP |
	LOWDOWN_SMARTY |
	LOWDOWN_STANDALONE;
if (!lowdown_file(&opts, stdin, &buf, &bufsz, NULL))
	errx(1, "lowdown_file");
fwrite(buf, 1, bufsz, stdout);
free(buf);
