enum uv_process_flags
Flags to be set on the flags field of uv_process_options_t.
enum uv_process_flags {
    /*
    * Set the child process' user id.
    */
    UV_PROCESS_SETUID = (1 << 0),
    /*
    * Set the child process' group id.
    */
    UV_PROCESS_SETGID = (1 << 1),
    /*
    * Do not wrap any arguments in quotes, or perform any other escaping, when
    * converting the argument list into a command line string. This option is
    * only meaningful on Windows systems. On Unix it is silently ignored.
    */
    UV_PROCESS_WINDOWS_VERBATIM_ARGUMENTS = (1 << 2),
    /*
    * Spawn the child process in a detached state - this will make it a process
    * group leader, and will effectively enable the child to keep running after
    * the parent exits. Note that the child process will still keep the
    * parent's event loop alive unless the parent process calls uv_unref() on
    * the child's process handle.
    */
    UV_PROCESS_DETACHED = (1 << 3),
    /*
    * Hide the subprocess window that would normally be created. This option is
    * only meaningful on Windows systems. On Unix it is silently ignored.
    */
    UV_PROCESS_WINDOWS_HIDE = (1 << 4),
    /*
    * Hide the subprocess console window that would normally be created. This
    * option is only meaningful on Windows systems. On Unix it is silently
    * ignored.
    */
    UV_PROCESS_WINDOWS_HIDE_CONSOLE = (1 << 5),
    /*
    * Hide the subprocess GUI window that would normally be created. This
    * option is only meaningful on Windows systems. On Unix it is silently
    * ignored.
    */
    UV_PROCESS_WINDOWS_HIDE_GUI = (1 << 6),
    /*
     * On Windows, if the path to the program to execute, specified in
     * uv_process_options_t's file field, has a directory component,
     * search for the exact file name before trying variants with
     * extensions like '.exe' or '.cmd'.
     */
    UV_PROCESS_WINDOWS_FILE_PATH_EXACT_NAME = (1 << 7)
};
type uv_stdio_container_t
Container for each stdio handle or fd passed to a child process.
typedef struct uv_stdio_container_s {
    uv_stdio_flags flags;
    union {
        uv_stream_t* stream;
        int fd;
    } data;
} uv_stdio_container_t;
enum uv_stdio_flags
Flags specifying how a stdio should be transmitted to the child process.
typedef enum {
    /*
    * The following four options are mutually-exclusive, and define
    * the operation to perform for the corresponding file descriptor
    * in the child process:
    */
    /*
    * No file descriptor will be provided (or redirected to
    * `/dev/null` if it is fd 0, 1 or 2).
    */
    UV_IGNORE = 0x00,
    /*
    * Open a new pipe into `data.stream`, per the flags below. The
    * `data.stream` field must point to a uv_pipe_t object that has
    * been initialized with `uv_pipe_init(loop, data.stream, ipc);`,
    * but not yet opened or connected.
    /*
    UV_CREATE_PIPE = 0x01,
    /*
    * The child process will be given a duplicate of the parent's
    * file descriptor given by `data.fd`.
    */
    UV_INHERIT_FD = 0x02,
    /*
    * The child process will be given a duplicate of the parent's
    * file descriptor being used by the stream handle given by
    * `data.stream`.
    */
    UV_INHERIT_STREAM = 0x04,
    /*
    * When UV_CREATE_PIPE is specified, UV_READABLE_PIPE and UV_WRITABLE_PIPE
    * determine the direction of flow, from the child process's perspective. Both
    * flags may be specified to create a duplex data stream.
    */
    UV_READABLE_PIPE = 0x10,
    UV_WRITABLE_PIPE = 0x20,
    /*
    * When UV_CREATE_PIPE is specified, specifying UV_NONBLOCK_PIPE opens the
    * handle in non-blocking mode in the child. This may cause loss of data,
    * if the child is not designed to handle to encounter this mode,
    * but can also be significantly more efficient.
    */
    UV_NONBLOCK_PIPE = 0x40
} uv_stdio_flags;

TCP handles provide an abstraction over TCP sockets. They are stream handles that support both uv_listen() and uv_connect().

uv_tcp_t is a 'subclass' of uv_stream_t.

Bind the TCP handle to an address. addr should point to either a struct sockaddr_in or struct sockaddr_in6.

The flags argument can be used to set socket options at bind time. Currently supported flags are:

  • UV_TCP_REUSEADDR: Enable the SO_REUSEADDR socket option. This allows a socket to bind to an address that is already in use, which is useful for server applications that need to be restarted quickly.

Changed in version 1.49.0: added the UV_TCP_REUSEPORT flag.

NOTE:

The UV_TCP_REUSEPORT flag is only available on Linux 3.9+, DragonFlyBSD 3.6+, FreeBSD 12.0+, Solaris 11.4, and AIX 7.2.5+. On other platforms, this function will return an UV_ENOTSUP error.

SO_REUSEPORT is an extension to the socket API that allows multiple socket instances to be bound to the same network address and port. This can be used for load balancing that distributes incoming connections across all listening sockets in multiple processes or threads.

Returns 0 on success, or an error code < 0 on failure.

جدولة عمل مجمع الخيوط

يوفر libuv مجمع خيوط يمكن استخدامه لتشغيل كود المستخدم والحصول على إشعار في خيط الحلقة. يستخدم مجمع الخيوط هذا داخليًا لتشغيل جميع عمليات نظام الملفات، بالإضافة إلى طلبات getaddrinfo و getnameinfo.

حجمه الافتراضي هو 4، ولكن يمكن تغييره وقت البدء عن طريق تعيين متغير البيئة UV_THREADPOOL_SIZE إلى أي قيمة (الحد الأقصى المطلق هو 1024).

تم التغيير في الإصدار 1.30.0: زاد الحد الأقصى المسموح به لـ UV_THREADPOOL_SIZE من 128 إلى 1024.

تم التغيير في الإصدار 1.45.0: أصبحت للخيوط الآن مكدس بحجم 8 ميجابايت بدلاً من الافتراضي (المنخفض أحيانًا) للمنصة.

تم التغيير في الإصدار 1.50.0: أصبحت للخيوط الآن اسم افتراضي هو libuv-worker.

مجمع الخيوط عالمي ومشترك بين جميع حلقات الأحداث. عندما يستخدم دالة معينة مجمع الخيوط (مثل عند استخدام uv_queue_work()) يقوم libuv بتخصيص تهيئة الحد الأقصى لعدد الخيوط المسموح به بواسطة UV_THREADPOOL_SIZE. المزيد من الخيوط عادة ما يعني المزيد من الإنتاجية ولكن بصمة ذاكرة أعلى. ومع ذلك، تنمو مكدسات الخيوط بشكل كسول على معظم المنصات.

ملاحظة:

لاحظ أنه على الرغم من استخدام مجمع خيوط عالمي مشترك عبر جميع حلقات الأحداث، فإن الدوال ليست آمنة للخيوط.

أساسيات libuv

تفرض libuv أسلوب برمجة **غير متزامن**، **موجه بالأحداث**. مهمتها الأساسية هي توفير حلقة أحداث وإشعارات قائمة على الاستدعاء لـ I/O وأنشطة أخرى. توفر libuv أدوات أساسية مثل المؤقتات، ودعم الشبكات غير المحظورة، والوصول غير المتزامن لنظام الملفات، والعمليات الفرعية، والمزيد.

حلقات الأحداث

في البرمجة الموجهة بالأحداث، يعبر التطبيق عن اهتمامه بأحداث معينة ويستجيب لها عند حدوثها. مسؤولية جمع الأحداث من نظام التشغيل أو مراقبة مصادر الأحداث الأخرى تتولاها libuv، ويمكن للمستخدم تسجيل استدعاءات ليتم استدعاؤها عند وقوع حدث. عادة ما تستمر حلقة الأحداث في العمل إلى الأبد. في الكود الزائف:

``` while (true) { wait_for_event(); process_all_events(); } ```
. تستخدم عمليات نظام الملفات دوال حظر داخليًا، ولكنها تستدعي هذه الدوال في مجمع مؤشرات الترابط وتقوم بإخطار المراقبين المسجلين لدى حلقة الأحداث عندما يكون تفاعل التطبيق مطلوبًا.

جميع دوال نظام الملفات لها شكلان - متزامن و غير متزامن.

يتم استدعاء الأشكال المتزامنة تلقائيًا (و تحظر) إذا كانت رد الاتصال فارغة (null). القيمة المرجعة للدوال هي رمز خطأ libuv. هذا عادة ما يكون مفيدًا فقط للمكالمات المتزامنة. يتم استدعاء الشكل غير المتزامن عند تمرير رد اتصال وتكون القيمة المرجعة 0.

يتم الحصول على واصف الملف باستخدام

يتم إغلاق واصفات الملفات باستخدام

ردود اتصال عمليات نظام الملفات لها التوقيع:

لنلقِ نظرة على تطبيق بسيط لأمر cat. نبدأ بتسجيل رد اتصال عند فتح الملف:

المجال result لـ uv_fs_t هو واصف الملف في حالة رد اتصال uv_fs_open. إذا تم فتح الملف بنجاح، نبدأ في قراءته.

    if (req->result < 0) {
        fprintf(stderr, "Read error: %s\n", uv_strerror(req->result));
    }
    else if (req->result == 0) {
        uv_fs_t close_req;
        // متزامن
        uv_fs_close(uv_default_loop(), &close_req, open_req.result, NULL);
    }
    else if (req->result > 0) {
        iov.len = req->result;
        uv_fs_write(uv_default_loop(), &write_req, 1, &iov, 1, -1, on_write);
    }
}

في حالة استدعاء القراءة، يجب تمرير مخزن مؤقت تم تهيئته والذي سيتم ملؤه بالبيانات قبل تشغيل رد اتصال القراءة. عمليات uv_fs_* تتوافق تقريبًا بشكل مباشر مع دوال POSIX معينة، لذا يتم الإشارة إلى نهاية الملف (EOF) في هذه الحالة بواسطة result يساوي 0. في حالة التدفقات أو الأنابيب، سيتم تمرير الثابت UV_EOF كحالة بدلاً من ذلك.

هنا ترى نمطًا شائعًا عند كتابة برامج غير متزامنة. يتم إجراء استدعاء uv_fs_close() بشكل متزامن. عادةً ما يتم تنفيذ المهام التي تحدث لمرة واحدة، أو التي يتم إجراؤها كجزء من مرحلة البدء أو الإنهاء بشكل متزامن، حيث أننا نهتم بالإدخال/الإخراج السريع عندما يقوم البرنامج بمهمته الأساسية ويتعامل مع مصادر إدخال/إخراج متعددة. بالنسبة للمهام المنفردة، غالبًا ما يكون فرق الأداء ضئيلًا وقد يؤدي إلى كود أبسط.

الكتابة إلى نظام الملفات بسيطة بالمثل باستخدام uv_fs_write(). سيتم تشغيل رد الاتصال الخاص بك بعد اكتمال الكتابة. في حالتنا، يقوم رد الاتصال ببساطة بتشغيل القراءة التالية. وبالتالي، تتقدم القراءة والكتابة بالتوازي عبر ردود الاتصال.

المخازن المؤقتة والتدفقات

مقبض الإدخال/الإخراج الأساسي في libuv هو التدفق (uv_stream_t). مقابس TCP، ومقابس UDP، والأنابيب لملفات الإدخال/الإخراج، والاتصال بين العمليات (IPC) كلها تُعامل على أنها فئات فرعية من التدفقات.

يتم تهيئة التدفقات باستخدام دوال مخصصة لكل فئة فرعية، ثم يتم تشغيلها باستخدام

int uv_read_start(uv_stream_t*, uv_alloc_cb alloc_cb, uv_read_cb read_cb);
int uv_read_stop(uv_stream_t*);
int uv_write(uv_write_t* req, uv_stream_t* handle,
             const uv_buf_t bufs[], unsigned int nbufs, uv_write_cb cb);

دوال التدفقات أبسط في الاستخدام من دوال نظام الملفات وسيقوم libuv تلقائيًا بمواصلة القراءة من التدفق عند استدعاء uv_read_start() مرة واحدة، حتى يتم استدعاء uv_read_stop().

الوحدة المنفصلة للبيانات هي المخزن المؤقت -- uv_buf_t. هذا مجرد مجموعة من مؤشر إلى البايتات (uv_buf_t.base) والطول (uv_buf_t.len). uv_buf_t خفيف الوزن ويتم تمريره بالقيمة. ما يتطلب الإدارة هو البايتات الفعلية، والتي يجب تخصيصها وتحريرها بواسطة التطبيق.

خطأ:

هذا البرنامج لا يعمل دائمًا، يحتاج إلى شيء أفضل

uvtee/main.c - قراءة على الأنابيب

Introduction

libuv is a library that provides asynchronous I/O primitives based on event loops. It's commonly used for network programming and cross-platform I/O operations. This document provides a high-level overview of libuv's features and usage.

Processes

libuv provides facilities for creating and managing child processes, including communication via pipes.

The following example demonstrates how to copy data from standard input to a file and standard output. It uses the uv_pipe_t handle type.

uvtee/main.c - setup

#include <uv.h>
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <fcntl.h> // For O_CREAT, O_RDWR
#include <unistd.h> // For write, close

// Global variables for handles
uv_loop_t *loop;
uv_pipe_t stdin_pipe;
uv_pipe_t stdout_pipe;
uv_pipe_t file_pipe;

// Forward declarations
void alloc_buffer(uv_handle_t* handle, size_t suggested_size, uv_buf_t* buf);
void read_stdin(uv_stream_t* stream, ssize_t nread, const uv_buf_t* buf);
void on_stdout_write(uv_write_t* req, int status);
void on_file_write(uv_write_t* req, int status);
void free_write_req(uv_write_t* req);
void write_data(uv_stream_t* dest, size_t size, uv_buf_t buf, uv_write_cb cb);

int main(int argc, char **argv) {
    if (argc <= 1) {
        fprintf(stderr, "Usage: %s <filename>\n", argv[0]);
        return 1;
    }

    loop = uv_default_loop();

    // Initialize pipes for stdin, stdout, and file
    uv_pipe_init(loop, &stdin_pipe, 0);
    uv_pipe_open(&stdin_pipe, 0); // 0 is the file descriptor for stdin

    uv_pipe_init(loop, &stdout_pipe, 0);
    uv_pipe_open(&stdout_pipe, 1); // 1 is the file descriptor for stdout

    uv_fs_t file_req;
    // Open the file for creation, reading, and writing with read/write permissions
    int fd = uv_fs_open(loop, &file_req, argv[1], O_CREAT | O_RDWR, 0644, NULL);
    if (fd < 0) {
        fprintf(stderr, "Error opening file: %s\n", uv_strerror(fd));
        return 1;
    }

    uv_pipe_init(loop, &file_pipe, 0);
    uv_pipe_open(&file_pipe, fd); // Associate pipe with the opened file descriptor

    // Start reading from stdin
    uv_read_start((uv_stream_t*)&stdin_pipe, alloc_buffer, read_stdin);

    // Run the event loop
    return uv_run(loop, UV_RUN_DEFAULT);
}

// Buffer allocation callback
void alloc_buffer(uv_handle_t* handle, size_t suggested_size, uv_buf_t* buf) {
    *buf = uv_buf_init((char*) malloc(suggested_size), suggested_size);
}

// Structure to hold write request and buffer
typedef struct {
    uv_write_t req;
    uv_buf_t buf;
} write_req_t;

// Function to free write request resources
void free_write_req(uv_write_t* req) {
    write_req_t* wr = (write_req_t*) req;
    free(wr->buf.base);
    free(wr);
}

// Callback for writing to stdout
void on_stdout_write(uv_write_t* req, int status) {
    if (status < 0) {
        fprintf(stderr, "Stdout write error: %s\n", uv_strerror(status));
    }
    free_write_req(req);
}

// Callback for writing to the file
void on_file_write(uv_write_t* req, int status) {
    if (status < 0) {
        fprintf(stderr, "File write error: %s\n", uv_strerror(status));
    }
    free_write_req(req);
}

// Utility function to write data to a stream
void write_data(uv_stream_t* dest, size_t size, uv_buf_t buf, uv_write_cb cb) {
    write_req_t* req = (write_req_t*) malloc(sizeof(write_req_t));
    if (!req) {
        fprintf(stderr, "Failed to allocate write request\n");
        free(buf.base); // Free the original buffer if allocation fails
        return;
    }
    // Allocate a new buffer for the write request to copy data into
    req->buf = uv_buf_init((char*) malloc(size), size);
    if (!req->buf.base) {
        fprintf(stderr, "Failed to allocate buffer for write request\n");
        free(buf.base); // Free the original buffer
        free(req);      // Free the request itself
        return;
    }
    memcpy(req->buf.base, buf.base, size);
    uv_write((uv_write_t*) req, (uv_stream_t*)dest, &req->buf, 1, cb);
}


// Read callback for stdin
void read_stdin(uv_stream_t* stream, ssize_t nread, const uv_buf_t* buf) {
    if (nread < 0) {
        // Error or EOF
        if (nread == UV_EOF) {
            // End of file reached, close all handles
            uv_close((uv_handle_t*)&stdin_pipe, NULL);
            uv_close((uv_handle_t*)&stdout_pipe, NULL);
            uv_close((uv_handle_t*)&file_pipe, NULL);
        } else {
            fprintf(stderr, "Read error: %s\n", uv_strerror(nread));
            // Close handles on other read errors as well
            uv_close((uv_handle_t*)&stdin_pipe, NULL);
            uv_close((uv_handle_t*)&stdout_pipe, NULL);
            uv_close((uv_handle_t*)&file_pipe, NULL);
        }
    } else if (nread > 0) {
        // Data read, write it to stdout and the file
        write_data((uv_stream_t *)&stdout_pipe, nread, *buf, on_stdout_write);
        write_data((uv_stream_t *)&file_pipe, nread, *buf, on_file_write);
    }

    // Free the buffer allocated by alloc_buffer, as write_data makes copies
    if (buf->base) {
        free(buf->base);
    }
}

The standard malloc is sufficient here, but you can use any memory allocation scheme. For example, node.js uses its own slab allocator which associates buffers with V8 objects.

The read callback nread parameter is less than 0 on any error. This error might be EOF, in which case we close all the streams, using the generic close function uv_close() which deals with the handle based on its internal type. Otherwise nread is a non-negative number and we can attempt to write that many bytes to the output streams. Finally remember that buffer allocation and deallocation is application responsibility, so we free the data.

The allocation callback may return a buffer with length zero if it fails to allocate memory. In this case, the read callback is invoked with error UV_ENOBUFS. libuv will continue to attempt to read the stream though, so you must explicitly call uv_close() if you want to stop when allocation fails.

The read callback may be called with nread = 0, indicating that at this point there is nothing to be read. Most applications will just ignore this.

File change events

All modern operating systems provide APIs to put watches on individual files or directories and be informed when the files are modified. libuv wraps common file change notification libraries [1]. This is one of the more inconsistent parts of libuv. File change notification systems are themselves extremely varied across platforms so getting everything working everywhere is difficult. To demonstrate, I'm going to build a simple utility which runs a command whenever any of the watched files change:

./onchange <command> <file1> [file2] ...

NOTE:

Currently this example only works on OSX and Windows. Refer to the notes of uv_fs_event_start function.

The file change notification is started using uv_fs_event_init():

onchange/main.c - The setup

int main(int argc, char **argv) {
    if (argc <= 2) {
        fprintf(stderr, "Usage: %s <command> <file1> [file2 ...]\n", argv[0]);
        return 1;
    }
    loop = uv_default_loop();
    command = argv[1];
    while (argc-- > 2) {
        fprintf(stderr, "Adding watch on %s\n", argv[argc]);
        uv_fs_event_t *fs_event_req = malloc(sizeof(uv_fs_event_t));
        uv_fs_event_init(loop, fs_event_req);
        // The recursive flag watches subdirectories too.
        uv_fs_event_start(fs_event_req, run_command, argv[argc], UV_FS_EVENT_RECURSIVE);
    }
    return uv_run(loop, UV_RUN_DEFAULT);
}

The third argument is the actual file or directory to monitor. The last argument, flags, can be:

/*
 * Flags to be passed to uv_fs_event_start().
 */
enum uv_fs_event_flags {
    UV_FS_EVENT_WATCH_ENTRY = 1,
    UV_FS_EVENT_STAT = 2,
    UV_FS_EVENT_RECURSIVE = 4
};

UV_FS_EVENT_WATCH_ENTRY and UV_FS_EVENT_STAT don't do anything (yet). UV_FS_EVENT_RECURSIVE will start watching subdirectories as well on supported platforms.

The callback will receive the following arguments:

1.
uv_fs_event_t *handle - The handle. The path field of the handle is the file on which the watch was set.
2.
const char *filename - If a directory is being monitored, this is the file which was changed. Only non-null on Linux and Windows. May be null even on those platforms.
3.
int events - one of UV_RENAME or UV_CHANGE, or a bitwise OR of both.
4.
int status - If status < 0, there is an libuv error.

In our example we simply print the arguments and run the command using system().

onchange/main.c - file change notification callback

void run_command(uv_fs_event_t *handle, const char *filename, int events, int status) {
    char path[1024];
    size_t size = 1023;
    // Does not handle error if path is longer than 1023.
    uv_fs_event_getpath(handle, path, &size);
    path[size] = '\0';
    fprintf(stderr, "Change detected in %s: ", path);
    if (events & UV_RENAME)
        fprintf(stderr, "renamed");
    if (events & UV_CHANGE)
        fprintf(stderr, "changed");
    fprintf(stderr, " %s\n", filename ? filename : "");
    system(command);
}

مؤشرات الترابط (Threads)

انتظر لحظة؟ لماذا نحن نتحدث عن مؤشرات الترابط؟ أليست حلقات الأحداث (event loops) هي الطريقة للقيام ببرمجة واسعة النطاق (web-scale programming)؟ حسناً... لا. مؤشرات الترابط لا تزال هي الوسيط الذي تقوم به المعالجات بعملها. وبالتالي فإن مؤشرات الترابط مفيدة للغاية في بعض الأحيان، حتى لو كان عليك التنقل عبر العديد من أدوات المزامنة (synchronization primitives).

تُستخدم مؤشرات الترابط داخليًا لمحاكاة الطبيعة غير المتزامنة لجميع استدعاءات النظام. يستخدم libuv أيضًا مؤشرات الترابط للسماح لك، التطبيق، بأداء مهمة بشكل غير متزامن تكون في الواقع محظورة، عن طريق إنشاء مؤشر ترابط وجمع النتيجة عند الانتهاء.

جانب ملحوظ من مرافق مؤشرات الترابط في libuv هو أنها قسم مستقل بذاته داخل libuv. بينما تعتمد الميزات الأخرى بشكل وثيق على حلقة الأحداث ومبادئ الاستدعاء العكسي (callbacks)، فإن مؤشرات الترابط مستقلة تمامًا، فهي تحظر حسب الحاجة، وتشير إلى الأخطاء مباشرة عبر قيم الإرجاع، وكما هو موضح في المثال الأول، فإنها لا تتطلب حتى حلقة أحداث قيد التشغيل.

واجهة برمجة تطبيقات مؤشرات الترابط في libuv محدودة للغاية لأن دلالات وصيغة مؤشرات الترابط تختلف على جميع المنصات، مع مستويات مختلفة من الاكتمال.

يفترض هذا الفصل الافتراض التالي: هناك حلقة أحداث واحدة فقط، تعمل في مؤشر ترابط واحد (مؤشر الترابط الرئيسي). لا يتفاعل أي مؤشر ترابط آخر مع حلقة الأحداث (باستثناء استخدام uv_async_send).

الأقفال (Read-Write Locks)

أقفال القراءة والكتابة (Read-write locks) هي آلية وصول أكثر دقة. يمكن لقارئين الوصول إلى الذاكرة المشتركة في نفس الوقت. لا يمكن للكاتب الحصول على القفل عندما يكون تحت يد القارئ. لا يمكن للقارئ أو الكاتب الحصول على القفل عندما يكون الكاتب يحمله. غالبًا ما تستخدم أقفال القراءة والكتابة في قواعد البيانات. إليك مثال بسيط.

#include <stdio.h>
#include <uv.h>
uv_barrier_t blocker;
uv_rwlock_t numlock;
int shared_num;
void reader(void *n)
{
    int num = *(int *)n;
    int i;
    for (i = 0; i < 20; i++) {
        uv_rwlock_rdlock(&numlock);
        printf("Reader %d: acquired lock\n", num);
        printf("Reader %d: shared num = %d\n", num, shared_num);
        uv_rwlock_rdunlock(&numlock);
        printf("Reader %d: released lock\n", num);
    }
    uv_barrier_wait(&blocker);
}
void writer(void *n)
{
    int num = *(int *)n;
    int i;
    for (i = 0; i < 20; i++) {
        uv_rwlock_wrlock(&numlock);
        printf("Writer %d: acquired lock\n", num);
        shared_num++;
        printf("Writer %d: incremented shared num = %d\n", num, shared_num);
        uv_rwlock_wrunlock(&numlock);
        printf("Writer %d: released lock\n", num);
    }
    uv_barrier_wait(&blocker);
}
int main()
{
    uv_barrier_init(&blocker, 4);
    shared_num = 0;
    uv_rwlock_init(&numlock);
    uv_thread_t threads[3];
    int thread_nums[] = {1, 2, 1};
    uv_thread_create(&threads[0], reader, &thread_nums[0]);
    uv_thread_create(&threads[1], reader, &thread_nums[1]);
    uv_thread_create(&threads[2], writer, &thread_nums[2]);
    uv_barrier_wait(&blocker);
    uv_barrier_destroy(&blocker);
    uv_rwlock_destroy(&numlock);
    return 0;
}

العمليات

تقدم libuv إدارة كبيرة للعمليات الفرعية، وتجريد الاختلافات بين المنصات وتسمح بالتواصل مع العملية الفرعية باستخدام تدفقات أو أنابيب مسماة.

النمط الشائع في Unix هو أن تقوم كل عملية بعمل شيء واحد وتقوم به بشكل جيد. في مثل هذه الحالة، غالبًا ما تستخدم العملية عمليات فرعية متعددة لتحقيق المهام (على غرار استخدام الأنابيب في الأصداف). قد يكون نموذج العمليات المتعددة مع الرسائل أسهل في التفكير فيه مقارنة بنموذج مع الخيوط والذاكرة المشتركة.

الاستنتاج الشائع ضد البرامج المستندة إلى الأحداث هو أنها لا تستطيع الاستفادة من النوى المتعددة في أجهزة الكمبيوتر الحديثة. في برنامج متعدد الخيوط، يمكن للنواة إجراء جدولة وتعيين خيوط مختلفة لنوى مختلفة، مما يحسن الأداء. ولكن حلقة الأحداث لديها خيط واحد فقط. يمكن أن يكون الحل البديل هو تشغيل عمليات متعددة بدلاً من ذلك، مع تشغيل كل عملية لحلقة أحداث، وتعيين كل عملية لنواة CPU منفصلة.

إنشاء عمليات فرعية

أبسط حالة هي عندما تريد ببساطة تشغيل عملية ومعرفة متى تنتهي. يتم تحقيق ذلك باستخدام uv_spawn.

spawn/main.c

uv_loop_t *loop;
uv_process_t child_req;
uv_process_options_t options;
int main() {
    loop = uv_default_loop();
    char* args[3];
    args[0] = "mkdir";
    args[1] = "test-dir";
    args[2] = NULL;
    options.exit_cb = on_exit;
    options.file = "mkdir";
    options.args = args;
    int r;
    if ((r = uv_spawn(loop, &child_req, &options))) {
        fprintf(stderr, "%s\n", uv_strerror(r));
        return 1;
    } else {
        fprintf(stderr, "Launched process with ID %d\n", child_req.pid);
    }
    return uv_run(loop, UV_RUN_DEFAULT);
}

ملاحظة:

يتم تهيئة options ضمنيًا بالأصفار نظرًا لأنها متغير عام. إذا قمت بتغيير options إلى متغير محلي، فتذكر تهيئته لإصفار جميع الحقول غير المستخدمة:

بعد استدعاء uv_spawn، سيحتوي uv_process_t.pid على معرف العملية للعملية الفرعية.

سيتم استدعاء رد النداء للخروج مع حالة الخروج و نوع الإشارة التي سببت الخروج.

لاحظ أنه من المهم **عدم** استدعاء uv_close قبل رد النداء للخروج.

spawn/main.c

void on_exit(uv_process_t *req, int64_t exit_status, int term_signal) {
    fprintf(stderr, "Process exited with status %" PRId64 ", signal %d\n", exit_status, term_signal);
    uv_close((uv_handle_t*) req, NULL);

**مطلوب** إغلاق مراقب العملية بعد خروج العملية.

تغيير معلمات العملية

قبل تشغيل العملية الفرعية، يمكنك التحكم في بيئة التنفيذ باستخدام حقول في uv_process_options_t.

تغيير دليل التنفيذ

قم بتعيين uv_process_options_t.cwd إلى الدليل المقابل.

تعيين متغيرات البيئة

uv_process_options_t.env هي مصفوفة من السلاسل المنتهية بصفر، كل منها بالشكل VAR=VALUE المستخدمة لإعداد متغيرات البيئة للعملية. قم بتعيين هذا إلى NULL لاستلهام البيئة من العملية الأم (هذه).

علامات الخيارات

يؤدي تعيين uv_process_options_t.flags إلى OR بين بتات العلامات التالية، إلى تعديل سلوك العملية الفرعية:

  • UV_PROCESS_SETUID - تعيين معرف مستخدم التنفيذ للعملية الفرعية إلى uv_process_options_t.uid.
  • UV_PROCESS_SETGID - تعيين معرف مجموعة التنفيذ للعملية الفرعية إلى uv_process_options_t.gid.

تغيير UID/GID مدعوم فقط على Unix، وسيفشل uv_spawn على Windows مع UV_ENOTSUP.

  • UV_PROCESS_WINDOWS_VERBATIM_ARGUMENTS - لا يتم إجراء اقتباس أو تهريب لـ uv_process_options_t.args على Windows. يتم تجاهلها على Unix.
  • UV_PROCESS_DETACHED - يبدأ العملية الفرعية في جلسة جديدة، والتي ستستمر في العمل بعد خروج العملية الأم. انظر المثال أدناه.

فصل العمليات

يمكن استخدام تمرير العلامة UV_PROCESS_DETACHED لتشغيل الشياطين (daemons)، أو العمليات الفرعية المستقلة عن العملية الأم بحيث لا يؤثر خروج العملية الأم عليها.

detach/main.c

int main() {
    loop = uv_default_loop();
    char* args[3];
    args[0] = "sleep";
    args[1] = "100";
    args[2] = NULL;
    options.exit_cb = NULL;
    options.file = "sleep";
    options.args = args;
    options.flags = UV_PROCESS_DETACHED;
    int r;
    if ((r = uv_spawn(loop, &child_req, &options))) {
        fprintf(stderr, "%s\n", uv_strerror(r));
        return 1;
    }
    fprintf(stderr, "Launched sleep with PID %d\n", child_req.pid);
    uv_unref((uv_handle_t*) &child_req);
    return uv_run(loop, UV_RUN_DEFAULT);

فقط تذكر أن المراقب لا يزال يراقب العملية الفرعية، لذلك لن يخرج برنامجك. استخدم uv_unref() إذا كنت تريد أن تكون أكثر إطلاق ونسيان.

إرسال إشارات إلى العمليات

تغلف libuv استدعاء النظام kill(2) القياسي على Unix وتنفذ استدعاءً بخصائص مماثلة على Windows، مع تحذير واحد: جميع SIGTERM و SIGINT و SIGKILL تؤدي إلى إنهاء العملية. توقيع uv_kill هو:

uv_err_t uv_kill(int pid, int signum);

للعمليات التي تم تشغيلها باستخدام libuv، يمكنك استخدام uv_process_kill بدلاً من ذلك، والذي يقبل المراقب uv_process_t كوسيط أول، بدلاً من معرف العملية (pid). في هذه الحالة، **تذكر استدعاء** uv_close على المراقب _بعد_ استدعاء رد النداء للخروج.

الإشارات

استخدم uv_signal_init() لتهيئة مراقب وربطه بحلقة. للاستماع إلى إشارات معينة على هذا المراقب، استخدم uv_signal_start() مع وظيفة المراقب. يمكن ربط كل مراقب برقم إشارة واحد فقط، مع قيام الاستدعاءات اللاحقة لـ uv_signal_start() بالكتابة فوق الارتباطات السابقة. استخدم uv_signal_stop() لإيقاف المراقبة. إليك مثال صغير يوضح الإمكانيات المختلفة:

signal/main.c

#include <stdio.h>
#include <stdlib.h>
#include <unistd.h>
#include <uv.h>
uv_loop_t* create_loop()
{
    uv_loop_t *loop = malloc(sizeof(uv_loop_t));
    if (loop) {
      uv_loop_init(loop);
    }
    return loop;
}
void signal_handler(uv_signal_t *handle, int signum)
{
    printf("Signal received: %d\n", signum);
    uv_signal_stop(handle);
}
// two signal handlers in one loop
void thread1_worker(void *userp)
{
    uv_loop_t *loop1 = create_loop();
    uv_signal_t sig1a, sig1b;
    uv_signal_init(loop1, &sig1a);
    uv_signal_start(&sig1a, signal_handler, SIGUSR1);
    uv_signal_init(loop1, &sig1b);
    uv_signal_start(&sig1b, signal_handler, SIGUSR1);
    uv_run(loop1, UV_RUN_DEFAULT);
}
// two signal handlers, each in its own loop
void thread2_worker(void *userp)
{
    uv_loop_t *loop2 = create_loop();
    uv_loop_t *loop3 = create_loop();
    uv_signal_t sig2;
    uv_signal_init(loop2, &sig2);
    uv_signal_start(&sig2, signal_handler, SIGUSR1);
    uv_signal_t sig3;
    uv_signal_init(loop3, &sig3);
    uv_signal_start(&sig3, signal_handler, SIGUSR1);
    while (uv_run(loop2, UV_RUN_NOWAIT) || uv_run(loop3, UV_RUN_NOWAIT)) {
    }
}
int main()
{
    printf("PID %d\n", getpid());
    uv_thread_t thread1, thread2;
    uv_thread_create(&thread1, thread1_worker, 0);
    uv_thread_create(&thread2, thread2_worker, 0);
    uv_thread_join(&thread1);
    uv_thread_join(&thread2);
    return 0;
}

ملاحظة:

uv_run(loop, UV_RUN_NOWAIT) يشبه uv_run(loop, UV_RUN_ONCE) من حيث أنه سيعالج حدثًا واحدًا فقط. UV_RUN_ONCE يحجب إذا لم تكن هناك أحداث معلقة، بينما UV_RUN_NOWAIT يعود فورًا. نستخدم NOWAIT حتى لا يتم تجويع إحدى الحلقات لأن الأخرى لا تحتوي على نشاط معلق.

أرسل SIGUSR1 إلى العملية، وستجد المراقب

يجري استدعاؤه 4 مرات، مرة واحدة لكل uv_signal_t. يتوقف المعالج ببساطة عن كل مقبض، بحيث يخرج البرنامج. هذا النوع من الإرسال إلى جميع المعالجات مفيد للغاية. يمكن لخادم يستخدم حلقات أحداث متعددة ضمان حفظ جميع البيانات بأمان قبل الإنهاء، ببساطة عن طريق إضافة كل حلقة لمراقب لـ SIGINT.

إدخال/إخراج العملية الفرعية

لدى العملية الجديدة العادية، مجموعة خاصة بها من واصفات الملفات، حيث تمثل 0 و 1 و 2 stdin و stdout و stderr على التوالي. في بعض الأحيان قد ترغب في مشاركة واصفات الملفات مع العملية الفرعية. على سبيل المثال، ربما يقوم تطبيقك بتشغيل أمر فرعي وتريد أن تذهب جميع الأخطاء إلى ملف السجل، ولكن تجاهل stdout. لهذا، ترغب في أن يكون stderr للعملية الفرعية هو نفسه stderr للعملية الأصلية. في هذه الحالة، يدعم libuv وراثة واصفات الملفات. في هذه العينة، نقوم باستدعاء برنامج الاختبار، وهو:

proc-streams/test.c

#include <stdio.h>
int main()
{
    fprintf(stderr, "This is stderr\n");
    printf("This is stdout\n");
    return 0;
}

يقوم برنامج proc-streams الفعلي بتشغيل هذا أثناء مشاركة stderr فقط. يتم تعيين واصفات الملفات للعملية الفرعية باستخدام حقل stdio في uv_process_options_t. أولاً، قم بتعيين حقل stdio_count إلى عدد واصفات الملفات التي يتم تعيينها. uv_process_options_t.stdio هو مصفوفة من uv_stdio_container_t، وهو:

typedef struct uv_stdio_container_s {
    uv_stdio_flags flags;
    union {
        uv_stream_t* stream;
        int fd;
    } data;
} uv_stdio_container_t;

حيث يمكن أن تحمل الأعلام (flags) قيمًا متعددة. استخدم UV_IGNORE إذا لم يكن سيتم استخدامه. إذا تم تمييز الحقول الثلاثة الأولى لـ stdio على أنها UV_IGNORE، فسيتم إعادة توجيهها إلى /dev/null.

نظرًا لأننا نريد تمرير واصف موجود، سنستخدم UV_INHERIT_FD. ثم نقوم بتعيين fd إلى stderr.

proc-streams/main.c

int main() {
    loop = uv_default_loop();
    /* ... */
    options.stdio_count = 3;
    uv_stdio_container_t child_stdio[3];
    child_stdio[0].flags = UV_IGNORE;
    child_stdio[1].flags = UV_IGNORE;
    child_stdio[2].flags = UV_INHERIT_FD;
    child_stdio[2].data.fd = 2;
    options.stdio = child_stdio;
    options.exit_cb = on_exit;
    options.file = args[0];
    options.args = args;
    int r;
    if ((r = uv_spawn(loop, &child_req, &options))) {
        fprintf(stderr, "%s\n", uv_strerror(r));
        return 1;
    }
    return uv_run(loop, UV_RUN_DEFAULT);
}

إذا قمت بتشغيل proc-stream، سترى أن السطر "This is stderr" فقط هو الذي سيتم عرضه. حاول تمييز stdout على أنه موروث وشاهد الإخراج.

التواصل بين العملية الأم والعملية الفرعية (IPC)

يمكن أن يكون لدى العملية الأم والعملية الفرعية اتصال أحادي أو ثنائي الاتجاه عبر أنبوب (pipe) تم إنشاؤه عن طريق تعيين uv_stdio_container_t.flags إلى تركيبة بتية (bitwise combination) من UV_CREATE_PIPE و UV_READABLE_PIPE أو UV_WRITABLE_PIPE. علامة القراءة/الكتابة من منظور العملية الفرعية. في هذه الحالة، يجب تعيين حقل uv_stream_t* stream ليشير إلى مثيل uv_pipe_t تم تهيئته ولم يتم فتحه.

التواصل بين العمليات العشوائية (IPC)

عند استخدام مقابس المجال، يتم اتباع نمط العميل-الخادم عادةً مع قيام منشئ/مالك المقبس بدور الخادم. بعد الإعداد الأولي، لا يختلف إرسال الرسائل عن TCP، لذلك سنعيد استخدام مثال خادم الصدى.

pipe-echo-server/main.c

$ socat - /path/to/socket

العميل الذي يرغب في الاتصال بمقبس مجال سيستخدم:

void uv_pipe_connect(uv_connect_t *req, uv_pipe_t *handle, const char *name, uv_connect_cb cb);

حيث سيشير name إلى echo.sock أو ما شابه. على أنظمة Unix، يجب أن يشير name إلى ملف صالح (على سبيل المثال، /tmp/echo.sock). على Windows، يتبع name تنسيق \\?\pipe\echo.sock.

إرسال واصفات الملفات عبر الأنابيب

الشيء الرائع في مقابس المجال هو أنه يمكن تبادل واصفات الملفات بين العمليات عن طريق إرسالها عبر مقبس مجال. هذا يسمح للعمليات بتسليم إدخالها/إخراجها إلى عمليات أخرى. تشمل التطبيقات خوادم موازنة الحمل، وعمليات العمال، وطرق أخرى للاستفادة المثلى من وحدة المعالجة المركزية. يدعم libuv حاليًا فقط إرسال مقابس TCP أو أنابيب أخرى عبر الأنابيب.

لتوضيح ذلك، سننظر في تنفيذ خادم صدى يقوم بتسليم العملاء إلى عمليات العمال بترتيب دوري (round-robin). هذا البرنامج معقد بعض الشيء، وبينما يتم تضمين مقتطفات فقط في الكتاب، يوصى بقراءة الكود الكامل لفهمه حقًا.

عملية العامل بسيطة جدًا، حيث يتم تسليم واصف الملف إليها بواسطة المشرف.

multi-echo-server/worker.c

uv_loop_t *loop;

int main() {

multi-echo-server/worker.c

void on_new_connection(uv_stream_t *q, ssize_t nread, const uv_buf_t *buf) {
    if (nread < 0) {
        if (nread != UV_EOF)
            fprintf(stderr, "Read error %s\n", uv_err_name(nread));
        uv_close((uv_handle_t*) q, NULL);
        return;
    }
    uv_pipe_t *pipe = (uv_pipe_t*) q;
    if (!uv_pipe_pending_count(pipe)) {
        fprintf(stderr, "No pending count\n");
        return;
    }
    uv_handle_type pending = uv_pipe_pending_type(pipe);
    assert(pending == UV_TCP);
    uv_tcp_t *client = (uv_tcp_t*) malloc(sizeof(uv_tcp_t));
    uv_tcp_init(loop, client);
    if (uv_accept(q, (uv_stream_t*) client) == 0) {
        uv_os_fd_t fd;
        uv_fileno((const uv_handle_t*) client, &fd);
        fprintf(stderr, "Worker %d: Accepted fd %d\n", getpid(), fd);
        uv_read_start((uv_stream_t*) client, alloc_buffer, echo_read);
    }
    else {
        uv_close((uv_handle_t*) client, NULL);
    }
}

أولاً، نستدعي uv_pipe_pending_count() للتأكد من وجود مقبض متاح للقراءة. إذا كان برنامجك يمكنه التعامل مع أنواع مختلفة من المقابض، يمكن استخدام uv_pipe_pending_type() لتحديد النوع. على الرغم من أن accept قد تبدو غريبة في هذا الكود، إلا أنها منطقية. ما تفعله accept تقليديًا هو الحصول على واجهة ملف (العميل) من واجهة ملف أخرى (مقبس الاستماع). وهذا هو بالضبط ما نفعله هنا. جلب واجهة الملف (client) من queue. من هذه النقطة، يقوم العامل بعمل خادم صدى قياسي.

multi-echo-server/main.c

إيقاف حلقة الأحداث

يمكن استخدام uv_stop() لإيقاف حلقة الأحداث. في أقرب وقت يمكن أن تتوقف الحلقة عن العمل هو * في التكرار التالي*، وربما لاحقًا. هذا يعني أن الأحداث الجاهزة للمعالجة في هذا التكرار من الحلقة ستظل تتم معالجتها، لذا لا يمكن استخدام uv_stop() كمفتاح إيقاف. عندما يتم استدعاء uv_stop()، لن تحظر الحلقة لعمليات الإدخال / الإخراج في هذا التكرار. قد يكون فهم دلالات هذه الأشياء صعبًا بعض الشيء، لذلك دعنا ننظر إلى uv_run() حيث يحدث كل تدفق التحكم.
