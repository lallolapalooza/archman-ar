libssh2_session_supported_algs(3)

الحزمة: libssh2

التراخيص: BSD

libssh2_session_supported_algs(3) libssh2 libssh2_session_supported_algs(3)

الاسم

libssh2_session_supported_algs - الحصول على قائمة الخوارزميات المدعومة

الملخص

#include <libssh2.h>
int
libssh2_session_supported_algs(LIBSSH2_SESSION* session,
                               int method_type,
                               const char*** algs);

الوصف

session - مثيل لـ LIBSSH2_SESSION تم تهيئته (ستستخدم الدالة مؤشرها إلى دالة تخصيص الذاكرة). method_type - نوع الطريقة. انظر libssh2_session_method_pref(3). algs - عنوان مؤشر سيشير إلى مصفوفة من الخوارزميات التي تم إرجاعها.

الحصول على قائمة الخوارزميات المدعومة لـ method_type المحدد. معلمة method_type مكافئة لـ method_type في libssh2_session_method_pref(3). في حالة النجاح، ستقوم الدالة بتخصيص الكمية المناسبة من الذاكرة. عند عدم الحاجة إليها، يجب إلغاء تخصيصها عن طريق استدعاء libssh2_free(3). عند فشل هذه الدالة، لا يجب القيام بذلك.

للحصول على قائمة بجميع خوارزميات الضغط المدعومة، يجب استدعاء libssh2_session_flag(session, LIBSSH2_FLAG_COMPRESS, 1) قبل استدعاء هذه الدالة، وإلا سيتم إرجاع "none" فقط.

في حالة النجاح، ستقوم الدالة بتخصيص المصفوفة وملئها بالخوارزميات المدعومة (نفس الأسماء كما هي محددة في RFC 4253). المصفوفة ليست منتهية بـ NULL.

مثال

#include "libssh2.h"
const char **algorithms;
int rc, i;
LIBSSH2_SESSION *session;
/* تهيئة الجلسة */
session = libssh2_session_init();
rc = libssh2_session_supported_algs(session,
                                    LIBSSH2_METHOD_CRYPT_CS,
                                    &algorithms);
if(rc > 0) {
    /* نجح الاستدعاء، قم بشيء ما مع قائمة الخوارزميات
       (مثل عرضها)... */
    printf("الخوارزميات المتناظرة المدعومة:\n");
    for(i = 0; i < rc; i++)
        printf("\t%s\n", algorithms[i]);
    /* ... وتحرير الذاكرة المخصصة عند عدم الحاجة إليها */
    libssh2_free(session, algorithms);
}
else {
    /* فشل الاستدعاء، معالجة الأخطاء */
}

قيمة الإرجاع

في حالة النجاح، سيتم إرجاع عدد الخوارزميات التي تم إرجاعها (أي رقم موجب). في حالة الفشل، يتم إرجاع رمز خطأ (رقم سالب، انظر أدناه). لا يجب إرجاع 0 أبدًا.

الأخطاء

LIBSSH2_ERROR_BAD_USE - عنوان algs غير صالح.

LIBSSH2_ERROR_METHOD_NOT_SUPPORTED - نوع الطريقة غير معروف.

LIBSSH2_ERROR_INVAL - خطأ داخلي (عادة لا ينبغي أن يحدث).

LIBSSH2_ERROR_ALLOC - فشل تخصيص الذاكرة.
