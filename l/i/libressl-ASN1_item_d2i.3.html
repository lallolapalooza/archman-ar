libressl-ASN1_item_d2i(3)

الحزمة: libressl

التراخيص: ISC, custom:OpenSSL

ASN1_item_d2i(3) دليل وظائف المكتبة ASN1_item_d2i(3)

الاسم

ASN1_item_d2i، ASN1_item_d2i_bio، ASN1_item_d2i_fp، d2i_ASN1_TYPE، ASN1_item_i2d، ASN1_item_i2d_bio، ASN1_item_i2d_fp، i2d_ASN1_TYPE، ASN1_item_dup، ASN1_item_print — فك تشفير وتشفير كائنات ASN.1

الملخص

مكتبة «libcrypto»
#include <openssl/asn1.h>

ASN1_TYPE *

int
ASN1_item_i2d(ASN1_VALUE *val_in, unsigned char **der_out, const ASN1_ITEM *it);

int
ASN1_item_i2d_bio(const ASN1_ITEM *it, BIO *out_bio, void *val_in);

int
ASN1_item_i2d_fp(const ASN1_ITEM *it, FILE *out_fp, void *val_in);

i2d_ASN1_TYPE(ASN1_TYPE *val_in, unsigned char **der_out);

الوصف

تقوم هذه الوظائف بتحويل قيم ASN.1 من ترميز BER إلى هياكل C داخلية ("d2i") والعكس صحيح ("i2d"). على عكس هياكل C التي تحتوي على مؤشرات إلى كائنات فرعية، فإن BER هو ترميز تسلسلي، مناسب للنقل عبر الشبكة وللتخزين في ملف.

ASN1_item_d2i() تفسر *der_in كمصفوفة بايت مشفرة بصيغة DER أو BER وتفك تشفير قيمة واحدة من النوع it ممثلة بما يصل إلى length بايت. إذا نجحت العملية، يتم تقدم *der_in إلى البايت التالي للبيانات التي تم تحليلها.

إذا نجح فك التشفير وكان val_out أو *val_out هو NULL، يتم تخصيص كائن جديد.

إذا نجح فك التشفير وكان *val_out غير NULL، فمن المفترض أنه يشير إلى كائن صالح مملوء ويتم محاولة إعادة استخدامه. يجب ألا يكون هيكلًا فارغًا مثل ذلك الذي تم إرجاعه بواسطة ASN1_item_new(3) أو بواسطة إحدى الوظائف المختلفة الخاصة بالنوع *_new(). هذه القدرة على "إعادة الاستخدام" موجودة للتوافق مع الإصدارات السابقة، ولكن استخدامها غير مستحسن بشدة؛ انظر قسم الأخطاء أدناه.

ASN1_item_d2i_bio() و ASN1_item_d2i_fp() مشابهتان لـ ASN1_item_d2i() باستثناء أنهما تقرأان من BIO أو FILE، على التوالي.

d2i_ASN1_TYPE() مشابهة لـ ASN1_item_d2i() باستثناء أنها لا تتطلب تحديد نوع مرغوب فيه من قبل المستخدم، ولكنها بدلاً من ذلك ترجع كائن غلاف ASN1_TYPE يحتوي على كل من النوع والقيمة الموجودة في الإدخال.

ASN1_item_i2d() تقوم بتشفير الكائن المشار إليه بواسطة val_in إلى تنسيق DER.

إذا لم يكن *der_out يساوي NULL، فإنها تكتب البيانات المشفرة بتنسيق DER إلى المخزن المؤقت في *der_out وتزيد المؤشر ليشير إلى ما بعد البيانات المكتوبة للتو. في هذه الحالة، تقع على عاتق المستخدم مسؤولية التأكد من أن المخزن المؤقت المشار إليه بواسطة *der_out طويل بما يكفي، بحيث لا يحدث تجاوز للمخزن المؤقت.

إذا كان *der_out يساوي NULL، يتم تخصيص ذاكرة لمخزن مؤقت، ولا يتم زيادة *der_out، ولكنه يشير إلى بداية البيانات المكتوبة لل تو.

إذا كان der_out يساوي NULL، فلن يتم كتابة البايتات المشفرة في أي مكان ولكن سيتم تجاهلها. بالنسبة لكائنات val_in ذات حجم التشفير المتغير، يتم استخدام هذا أحيانًا للعثور أولاً على عدد البايتات التي سيتم كتابتها. بعد ذلك، يتم تخصيص كمية كافية من الذاكرة قبل استدعاء ASN1_item_i2d() مرة أخرى. غالبًا ما لا تكون تقنية الاستدعاء المزدوجة الصريحة هذه ضرورية لأن تقنية التخصيص التلقائي الموضحة في الفقرة السابقة يمكن استخدامها.

ASN1_item_i2d_bio() و ASN1_item_i2d_fp() مشابهتان لـ ASN1_item_i2d() باستثناء أنهما تكتبان إلى BIO أو FILE، على التوالي.

i2d_ASN1_TYPE() مشابهة لـ ASN1_item_i2d() باستثناء أنه لا يتم توفير النوع والقيمة بشكل منفصل، ولكن في شكل كائن ASN1_TYPE واحد.

ASN1_item_dup() تقوم بإنشاء نسخة عميقة من val_in عن طريق استدعاء ASN1_item_i2d() و ASN1_item_d2i().

قيم الإرجاع

إذا نجحت، ترجع ASN1_item_d2i()، ASN1_item_d2i_bio()، ASN1_item_d2i_fp()، و d2i_ASN1_TYPE() مؤشرًا إلى قيمة ASN.1 التي تم فك تشفيرها. بالإضافة إلى ذلك، إذا لم يكن val_out يساوي NULL، يتم كتابة المؤشر أيضًا إلى *val_out. إذا حدث خطأ، يتم إرجاع NULL.

ASN1_item_i2d() و i2d_ASN1_TYPE() ترجع عدد البايتات المكتوبة أو قيمة سالبة إذا حدث خطأ.

ASN1_item_i2d_bio() و ASN1_item_i2d_fp() ترجع 1 للنجاح أو 0 للفشل.

ASN1_item_dup() ترجع كائن ASN1_VALUE الجديد أو NULL إذا حدث خطأ.
