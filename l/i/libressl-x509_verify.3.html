libressl-x509_verify(3)

الحزمة: libressl

التراخيص: ISC, custom:OpenSSL

X509_VERIFY(3) دليل وظائف المكتبة X509_VERIFY(3)

الاسم

x509_verify، x509_verify_ctx_new، x509_verify_ctx_free، x509_verify_ctx_set_max_depth، x509_verify_ctx_set_max_signatures، x509_verify_ctx_set_max_chains، x509_verify_ctx_set_purpose، x509_verify_ctx_set_intermediates، x509_verify_ctx_error_string، x509_verify_ctx_error_depth، x509_verify_ctx_chain — اكتشاف والتحقق من سلاسل شهادات X.509

الملخص

#include <openssl/x509_verify.h>

size_t
x509_verify(X509_VERIFY_CTX *ctx, X509 *leaf, char *name);

X509_VERIFY_CTX *
x509_verify_ctx_new(STACK_OF(X509) *roots);

void
x509_verify_ctx_free(X509_VERIFY_CTX *ctx);

int
x509_verify_ctx_set_max_depth(X509_VERIFY_CTX *ctx, size_t max);

int
x509_verify_ctx_set_max_signatures(X509_VERIFY_CTX *ctx, size_t max);

int
x509_verify_ctx_set_max_chains(X509_VERIFY_CTX *ctx, size_t max);

int
x509_verify_ctx_set_purpose(X509_VERIFY_CTX *ctx, int purpose_id);

int
x509_verify_ctx_set_intermediates(X509_VERIFY_CTX *ctx, STACK_OF(X509) *intermediates);

const char *
x509_verify_ctx_error_string(X509_VERIFY_CTX *ctx);

size_t
x509_verify_ctx_error_depth(X509_VERIFY_CTX *ctx);

STACK_OF(X509) *
x509_verify_ctx_chain(X509_VERIFY_CTX *ctx, size_t index);

الوصف

تحاول الدالة x509_verify() اكتشاف والتحقق من صحة جميع سلاسل الشهادات للاسم name من الشهادة الورقية leaf بناءً على المعلمات في ctx. يمكن بناء والتحقق من صحة عدة سلاسل. لا يتم إجراء فحص الإلغاء بواسطة هذه الدالة، ويجب على المتصل القيام به على أي سلاسل معادة إذا رغب في ذلك.

x509_verify_ctx_new() تخصص سياقًا جديدًا باستخدام الجذور الموثوقة roots. في حالة النجاح، فإنها تزيد عدد مرات الرجوع لـ roots.

x509_verify_ctx_free() تفرغ ctx وتقلل عدد مرات الرجوع للجذور roots والشهادات الوسيطة intermediates المرتبطة بها. إذا كان ctx هو NULL، فلا يحدث أي إجراء.

x509_verify_ctx_set_max_depth() تحدد أقصى عمق لسلاسل الشهادات التي سيتم إنشاؤها بـ max، والتي يمكن أن تكون في النطاق من 1 إلى الحد الافتراضي 32.

x509_verify_ctx_set_max_signatures() تحدد الحد الأقصى لعدد عمليات توقيع المفتاح العام التي سيتم استخدامها عند التحقق من صحة سلاسل الشهادات إلى max، والتي يمكن أن تكون في النطاق من 1 إلى 100000. الحد الافتراضي هو 256.

x509_verify_ctx_set_max_chains() تحدد الحد الأقصى لعدد السلاسل التي يمكن إرجاعها بـ max، والتي يمكن أن تكون في النطاق من 1 إلى الحد الافتراضي 8.

x509_verify_ctx_set_purpose() تحدد الغرض من الشهادة للتحقق من صحتها بـ purpose_id. يمكن استخدام الثوابت X509_PURPOSE_* المدرجة في X509_check_purpose(3).

x509_verify_ctx_set_intermediates() توفر بعض الشهادات الوسيطة، والتي يتم استلامها عادةً من الطرف المقابل، ليتم استخدامها لبناء السلاسل. في حالة النجاح، تزيد هذه الدالة من عدد مرات الرجوع لـ intermediates.

x509_verify_ctx_error_string() تستخرج وصفًا لآخر خطأ تم مواجهته بواسطة استدعاء سابق لـ x509_verify() من ctx.

x509_verify_ctx_error_depth() تستخرج عمق آخر خطأ تم مواجهته بواسطة استدعاء سابق لـ x509_verify() من ctx.

x509_verify_ctx_chain() تستخرج السلسلة التي تم التحقق من صحتها مع الفهرس index من ctx بعد استدعاء سابق لـ x509_verify(). يبدأ الفهرس index من 0، ويعتبر خطأ تمرير رقم أكبر من أو يساوي القيمة المعادة من x509_verify(). المؤشر المعطى ليس نسخة، ولا يتم زيادة عدد مرات الرجوع له.

قيم الإرجاع

x509_verify() تعيد عدد السلاسل التي تم إنشاؤها والتحقق من صحتها بنجاح أو 0 عند الفشل.

x509_verify_ctx_new() تعيد سياقًا مخصصًا جديدًا أو NULL عند الفشل.

x509_verify_ctx_set_max_depth()، x509_verify_ctx_set_max_signatures()، x509_verify_ctx_set_max_chains()، x509_verify_ctx_set_purpose()، و x509_verify_ctx_set_intermediates() تعيد 1 عند النجاح أو 0 عند الفشل.

x509_verify_ctx_error_string() تعيد مؤشرًا إلى سلسلة نصية قابلة للقراءة لوصف الخطأ. إذا لم يحدث خطأ، يتم إرجاع "ok".

x509_verify_ctx_chain() تعيد مؤشرًا داخليًا إلى سلسلة تم التحقق من صحتها أو NULL إذا كان index أكبر من أو يساوي عدد السلاسل التي تم إنشاؤها والتحقق من صحتها بنجاح. يصبح المؤشر المعطى غير صالح عند تدمير ctx.
