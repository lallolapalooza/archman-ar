libressl-EVP_PKEY_new(3)

الحزمة: libressl

التراخيص: ISC, custom:OpenSSL

EVP_PKEY_NEW(3) دليل المكتبة EVP_PKEY_NEW(3)

الاسم

EVP_PKEY_new، EVP_PKEY_up_ref، EVP_PKEY_free، EVP_PKEY_new_raw_private_key، EVP_PKEY_new_raw_public_key، EVP_PKEY_new_mac_key، EVP_PKEY_get_raw_private_key، EVP_PKEY_get_raw_public_key — وظائف تخصيص المفاتيح العامة والخاصة والتعامل مع المفاتيح الأولية

الوصف

يُستخدم الهيكل EVP_PKEY بواسطة وظائف OpenSSL المختلفة التي تتطلب مفتاحًا عامًا أو خاصًا عامًا دون الرجوع إلى خوارزمية معينة.

تقوم الدالة EVP_PKEY_new() بتخصيص هيكل EVP_PKEY فارغ. يتم تعيين عدد المراجع إلى 1. لإضافة مفتاح خاص أو عام إليه، استخدم الدوال الموصوفة في EVP_PKEY_set1_RSA(3).

تقوم الدالة EVP_PKEY_up_ref() بزيادة عدد المراجع لـ pkey بمقدار 1.

تقوم الدالة EVP_PKEY_free() بتقليل عدد المراجع لـ pkey بمقدار 1، وإذا وصل عدد المراجع إلى الصفر، يتم تحريره. إذا كان pkey مؤشر NULL، فلا يحدث أي إجراء.

تقوم الدالة EVP_PKEY_new_raw_private_key() بتخصيص EVP_PKEY جديد. يتم توفير NID لخوارزمية مفتاح عام تدعم المفاتيح الخاصة الأولية، أي EVP_PKEY_HMAC، EVP_PKEY_X25519، أو EVP_PKEY_ED25519، في الوسيط type و rawlen بايتات من بيانات المفتاح الخاص الأولي من هذا النوع في rawpriv. يتم اشتقاق بيانات المفتاح العام تلقائيًا من بيانات المفتاح الخاص المعطاة، إذا كان ذلك مناسبًا لنوع الخوارزمية. يتم تجاهل الوسيط ENGINE *engine دائمًا ويوصى بتمرير NULL.

تعمل الدالة EVP_PKEY_new_raw_public_key() بنفس الطريقة التي تعمل بها EVP_PKEY_new_raw_private_key() باستثناء أن rawpub يشير إلى بيانات المفتاح العام الأولي. يتم تهيئة هيكل EVP_PKEY بدون أي معلومات مفتاح خاص. أنواع الخوارزميات التي تدعم المفاتيح العامة الأولية هي EVP_PKEY_X25519 و EVP_PKEY_ED25519.

الدالة EVP_PKEY_new_mac_key() هي دالة مهملة تحقق نفس التأثير الذي تحققه EVP_PKEY_new_raw_private_key() بطريقة أكثر تعقيدًا وتعمل فقط مع type من EVP_PKEY_HMAC.

تكتب الدالة EVP_PKEY_get_raw_private_key() ما يصل إلى *rawlen بايت من بيانات المفتاح الخاص الأولي إلى المخزن المؤقت الذي يبدأ عند rawpriv وتخزن عدد البايتات المكتوبة في *rawlen. التطبيق المتصل مسؤول عن ضمان أن المخزن المؤقت كبير بما يكفي لاستقبال بيانات المفتاح الخاص. إذا كان الوسيط rawpriv هو NULL، يتم تخزين عدد البايتات المطلوبة لحمل المفتاح في *rawlen. هذه الدالة تعمل فقط مع الخوارزميات التي تدعم المفاتيح الخاصة الأولية. حاليًا، هذه هي EVP_PKEY_HMAC، EVP_PKEY_X25519، و EVP_PKEY_ED25519.

تشبه الدالة EVP_PKEY_get_raw_public_key() الدالة EVP_PKEY_get_raw_private_key() باستثناء أنها تكتب بيانات المفتاح العام الأولي. هذه الدالة تعمل فقط مع الخوارزميات التي تدعم المفاتيح العامة الأولية. حاليًا، هذه هي EVP_PKEY_X25519 و EVP_PKEY_ED25519.

قيم الإرجاع

EVP_PKEY_new()، EVP_PKEY_new_raw_private_key()، EVP_PKEY_new_raw_public_key()، و EVP_PKEY_new_mac_key() ترجع إما هيكل EVP_PKEY الجديد المخصص أو NULL إذا حدث خطأ.

EVP_PKEY_up_ref()، EVP_PKEY_get_raw_private_key()، و EVP_PKEY_get_raw_public_key() ترجع 1 للنجاح أو 0 للفشل.

أمثلة

يستوعب الكود التالي رسالة باستخدام HMAC-SHA256:

/* مفتاح وهمي: عادة ما يتم تعيينه من مصدر آخر */
const unsigned char *key = "key";
const size_t key_len = strlen(key);

const char *msg = "The quick brown fox jumps over the lazy dog";
const size_t msg_len = strlen(msg);

unsigned char *out_mac;
size_t out_len, i;

EVP_PKEY *pkey;
EVP_MD_CTX *md_ctx;

pkey = EVP_PKEY_new_raw_private_key(EVP_PKEY_HMAC, NULL,
    key, key_len);
if (pkey == NULL)
	err(1, "EVP_PKEY_new_raw_private_key");

md_ctx = EVP_MD_CTX_new();
if (md_ctx == NULL)
	err(1, "EVP_MD_CTX_new");

if (EVP_DigestSignInit(md_ctx, NULL, EVP_sha256(), NULL, pkey) == 0)
	err(1, "EVP_DigestSignInit");
if (EVP_DigestSign(md_ctx, NULL, &out_len, msg, msg_len) == 0)
	err(1, "EVP_DigestSign(NULL)");
if ((out_mac = calloc(1, out_len)) == NULL)
	err(1, "calloc");
if (EVP_DigestSign(md_ctx, out_mac, &out_len, msg, msg_len) == 0)
	err(1, "EVP_DigestSign(MAC)");

EVP_MD_CTX_free(md_ctx);
EVP_PKEY_free(pkey);

printf(" MAC = ");
for (i = 0; i < out_len; i++)
	printf("%02x", out_mac[i]);
printf("\n");
free(out_mac);

على الرغم من أن اسم النوع EVP_PKEY كان مخصصًا في الأصل لـ «المفتاح الخاص» وتم تصميم واجهة برمجة التطبيقات EVP_DigestSignInit(3) للتوقيعات الرقمية في سياق التشفير بالمفاتيح العامة، إلا أنهما يستخدمان هنا أيضًا لأن MAC يتطلب أيضًا مفتاحًا، حتى لو كان مفتاحًا متماثلًا.

يمكن استخدام نفس الكود للتوقيع باستخدام Ed25519 عن طريق جعل المفتاح ED25519_PRIVATE_KEY_LENGTH = 32 بايت
