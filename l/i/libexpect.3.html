libexpect(3)

الحزمة: expect

التراخيص: custom

wnl و exp_spawnv يحاكيان إجراءات الـ shell في البحث عن ملف تنفيذي في قائمة من الأدلة. يتم الحصول على قائمة الأدلة من البيئة.

معالجة التوقع

بينما يمكن استخدام read() لقراءة المعلومات من عملية تم إنشاؤها بواسطة exp_spawnl أو exp_spawnv، يتم توفير وظائف أكثر ملاءمة. وهي كالتالي:

int
exp_expectl(fd,type1,pattern1,[re1,],value1,type2,...,exp_end);
int fd;
enum exp_type type;
char *pattern1, *pattern2, ...;
regexp *re1, *re2, ...;
int value1, value2, ...;


int
exp_fexpectl(fp,type1,pattern1,[re1,]value1,type2,...,exp_end);
FILE *fp;
enum exp_type type;
char *pattern1, *pattern2, ...;
regexp *re1, *re2, ...;
int value1, value2, ...;
enum exp_type {
exp_end,
exp_glob,
exp_exact,
exp_regexp,
exp_compiled,
exp_null,
};
struct exp_case {
char *pattern;
regexp *re;
enum exp_type type;
int value;
};
int
exp_expectv(fd,cases);
int fd;
struct exp_case *cases;
int
exp_fexpectv(fp,cases);
FILE *fp;
struct exp_case *cases;
extern int exp_timeout;
extern char *exp_match;
extern char *exp_match_end;
extern char *exp_buffer;
extern char *exp_buffer_end;
extern int exp_match_max;
extern int exp_full_buffer;
extern int exp_remove_nulls;

تنتظر الوظائف حتى تتطابق المخرجات من عملية مع أحد الأنماط، أو تمر فترة زمنية محددة، أو يتم رؤية نهاية الملف (EOF).

الوسيط الأول لكل وظيفة هو إما واصف ملف أو دفق. تصف مجموعات الوسائط المتتالية الأنماط والقيم الصحيحة المرتبطة بها التي سيتم إرجاعها عند تطابق النمط.

الوسيط type هو أحد أربعة قيم. exp_end تشير إلى عدم وجود المزيد من الأنماط. exp_glob تشير إلى أن النمط هو نمط سلسلة على غرار glob. exp_exact تشير إلى أن النمط هو سلسلة بالضبط. exp_regexp تشير إلى أن النمط هو نمط سلسلة على غرار regexp. exp_compiled تشير إلى أن النمط هو نمط سلسلة على غرار regexp، وأن شكله المترجم متاح أيضًا. exp_null تشير إلى أن النمط هو null (لأغراض التصحيح، يجب أن يتبع نمط السلسلة أيضًا).

إذا لم يتم توفير الشكل المترجم مع الوظائف exp_expectl و exp_fexpectl، فسيتم تجاهل أي ترجمة نمط يتم إجراؤها داخليًا بعد عودة الوظيفة. ستقوم الوظائف exp_expectv و exp_fexpectv بترجمة الأنماط تلقائيًا ولن تتجاهلها. بدلاً من ذلك، يجب على المستخدم التخلص منها، عن طريق استدعاء free لكل نمط. من الضروري فقط التخلص منها، في المرة الأخيرة التي يتم فيها استخدام الحالات.

يتم تخزين الأنماط الفرعية لـ Regexp المطابقة في regexp المترجم. بافتراض أن "re" يحتوي على regexp مترجم، يمكن العثور على السلسلة المتطابقة في re->startp[0]. يمكن العثور على سلاسل المطابقة الفرعية (وفقًا للأقواس) في النمط الأصلي في re->startp[1]، re->startp[2]، وهكذا، حتى re->startp[9]. نهايات السلاسل المقابلة هي re->endp[x] حيث x هو نفس الفهرس الخاص ببداية السلسلة.

يطابق exp_null إذا ظهر null في الإدخال. يجب تعيين المتغير exp_remove_nulls إلى 0 لمنع إزالة nulls تلقائيًا. بشكل افتراضي، يتم تعيين exp_remove_nulls إلى 1 ويتم إزالة nulls تلقائيًا.

تكون الوظائف exp_expectv و exp_fexpectv مفيدة عندما لا يكون عدد الأنماط معروفًا مسبقًا. في هذه الحالة، يتم توفير المجموعات في مصفوفة. يتم تحديد نهاية المصفوفة بواسطة struct exp_case مع النوع exp_end. لبقية هذا المناقشة، سيشار إلى هذه الوظائف بشكل عام باسم expect.

إذا تطابق نمط، يتم إرجاع القيمة الصحيحة المقابلة. لا يلزم أن تكون القيم فريدة، ولكن يجب أن تكون موجبة لتجنب الخلط بينها وبين EXP_EOF، EXP_TIMEOUT، أو EXP_FULLBUFFER. عند EOF أو انتهاء المهلة، يتم إرجاع القيمة EXP_EOF أو EXP_TIMEOUT. فترة انتهاء المهلة الافتراضية هي 10 ثوانٍ ولكن يمكن تغييرها عن طريق تعيين المتغير exp_timeout. القيمة -1 تعطّل انتهاء المهلة. تتسبب القيمة 0 في عودة وظيفة expect فورًا (أي، استطلاع) بعد قراءة واحدة(). ومع ذلك، يجب أن يسبقها وظيفة مثل select، poll، أو معاودة اتصال مدير الأحداث لضمان وجود بيانات للقراءة.

إذا كانت المتغير exp_full_buffer يساوي 1، فسيتم إرجاع EXP_FULLBUFFER إذا امتلأ exp_buffer بدون تطابق أي نمط.

عندما تعود وظيفة expect، يشير exp_buffer إلى مخزن الأحرف الذي كان يتم النظر فيه للمطابقة. يشير exp_buffer_end إلى ما بعد آخر حرف في exp_buffer. إذا حدث تطابق، يشير exp_match إلى exp_buffer حيث بدأ التطابق. يشير exp_match_end إلى حرف واحد بعد نهاية التطابق.

في كل مرة تصل فيها بيانات إدخال جديدة، تتم مقارنتها بكل نمط بالترتيب الذي تم إدراجه به. وبالتالي، يمكنك اختبار عدم وجود تطابق بجعل النمط الأخير شيئًا مضمونًا للظهور، مثل مطالبة. في المواقف التي لا توجد فيها مطالبة، يجب عليك التحقق من EXP_TIMEOUT (تمامًا كما لو كنت تتفاعل يدويًا). يمكن العثور على المزيد من الفلسفة والاستراتيجيات حول تحديد أنماط expect في الوثائق الخاصة ببرنامج expect نفسه. انظر SEE ALSO أدناه.

الأنماط هي تعبيرات نمطية قياسية من طراز C-shell. على سبيل المثال، يبحث الجزء التالي عن تسجيل دخول ناجح، كما هو الحال في حوار telnet.

switch (exp_expectl(
	exp_glob,"connected",CONN,
	exp_glob,"busy",BUSY,
	exp_glob,"failed",ABORT,
	exp_glob,"invalid password",ABORT,
	exp_end)) {
case CONN:	/* logged in successfully */
	break;
case BUSY:	/* couldn't log in at the moment */
	break;
case EXP_TIMEOUT:
case ABORT:	/* can't log in at any moment! */
	break;
default: /* problem with expect */
}

تعتبر النجوم (كما في المثال أعلاه) اختصارًا مفيدًا لتجاهل أحرف إنهاء الأسطر والتفاصيل الأخرى. يجب أن تتطابق الأنماط مع كامل مخرجات العملية الحالية (منذ آخر قراءة على الواصف أو الدفق. يمكن أن يؤدي أكثر من 2000 بايت من المخرجات إلى "نسيان" البايتات السابقة. يمكن تغيير هذا عن طريق تعيين المتغير exp_match_max. لاحظ أن القيم الكبيرة بشكل مفرط يمكن أن تبطئ مطابقة الأنماط.

التشغيل في الخلفية

extern int exp_disconnected;
int exp_disconnect();

من الممكن نقل عملية إلى الخلفية بعد أن تبدأ في العمل. الاستخدام النموذجي لهذا هو قراءة كلمات المرور ثم الانتقال إلى الخلفية للنوم قبل استخدام كلمات المرور للقيام بالعمل الفعلي.

لتحريك عملية إلى الخلفية، قم بعمل fork، واستدعاء exp_disconnect() في العملية الفرعية و exit() في العملية الأم. هذا يفصل عمليتك عن الطرفية المتحكمة. إذا كنت ترغب في نقل عملية إلى الخلفية بطريقة مختلفة، يجب عليك تعيين المتغير exp_disconnected إلى 1. هذا يسمح ببدء العمليات التي تم إنشاؤها بعد هذه النقطة بشكل صحيح.

تعدد الإرسال

بشكل افتراضي، تقوم وظائف expect بحظر قراءة واصف ملف واحد. إذا كنت ترغب في الانتظار على الأنماط من واصفات ملفات متعددة، استخدم select، poll، أو مدير الأحداث. ستخبرك هذه الوظائف بواصف الملف الجاهز للقراءة.

عندما يكون واصف ملف جاهزًا للقراءة، يمكنك استخدام وظائف expect لإجراء قراءة واحدة فقط عن طريق تعيين انتهاء المهلة إلى 0.

التحكم بالتابع

void
exp_slave_control(fd,enable)
int fd;
int enable;

يتم عادةً التقاطpty تلقائيًا بواسطة وظائف expect. ومع ذلك، إذا كنت ترغب في إصدار ioctl، على سبيل المثال، مباشرة على جهاز التابع، يجب عليك تعطيل الالتقاط مؤقتًا.

يمكن التحكم في التقاطpty باستخدام exp_slave_control. الوسيط الأول هو واصف الملف المقابل للعملية المنشأة. الوسيط الثاني هو 0 إذا كان سيتم تعطيل الالتقاط و 1 إذا كان سيتم تمكينه.

الأخطاء

تشير جميع الوظائف إلى الأخطاء عن طريق إرجاع -1 وتعيين errno.

تتم كتابة الأخطاء التي تحدث بعد fork وظائف spawn (على سبيل المثال، محاولة إنشاء برنامج غير موجود) إلى stderr العملية، وسيتم قراءتها بواسطة أول expect.

الإشارات

extern int exp_reading;
extern jmp_buf exp_readenv;

يستخدم expect alarm() لانتهاء المهلة، وبالتالي إذا قمت بإنشاء إنذارات أثناء expect، فسوف تنتهي المهلة مبكرًا.

داخليًا، يقوم expect باستدعاء read() والتي يمكن مقاطعتها بواسطة الإشارات. إذا قمت بتعريف معالجات الإشارات، يمكنك اختيار إعادة تشغيل أو إنهاء قراءة expect الداخلية. المتغير exp_reading صحيح إذا (وفقط إذا) تمت مقاطعة قراءة expect. longjmp(exp_readenv,EXP_ABORT) سيؤدي إلى إنهاء القراءة. longjmp(exp_readenv,EXP_RESTART) سيعيد تشغيل القراءة.

التسجيل

extern int exp_loguser;
extern int exp_logfile_all
extern FILE *exp_logfile;

إذا كان exp_loguser غير صفري، يرسل expect أي مخرجات من العملية المنشأة إلى stdout. نظرًا لأن البرامج التفاعلية تعكس عادةً مدخلاتها، فإن هذا يكفي عادةً لإظهار جانبي المحادثة. إذا كان exp_logfile غير صفري أيضًا، يتم كتابة نفس المخرجات إلى الدفق المحدد بواسطة exp_logfile. إذا كان exp_logfile_all غير صفري، فسيتم كتابة exp_logfile بغض النظر عن قيمة exp_loguser.

التصحيح

بينما أعتبر المكتبة سهلة الاستخدام، أعتقد أن برنامج expect المستقل أسهل بكثير، بكثير، من العمل مع C ومراحل التحرير والترجمة والتصحيح المعتادة. على عكس برامج C النموذجية، فإن معظم التصحيح لا يتعلق بجعل مترجم C يقبل برامجك - بل يتعلق بجعل الحوار صحيحًا. أيضًا، ترجمة النصوص من expect إلى C ليست ضرورية عادةً. على سبيل المثال، سرعة الحوارات التفاعلية لا تمثل مشكلة في الغالب. لذا يرجى تجربة برنامج 'expect' المستقل أولاً. أشك في أنه حل أكثر ملاءمة لمعظم الناس من المكتبة.

ومع ذلك، إذا شعرت بالضرورة لتصحيح الأخطاء في C، فإليك بعض الأدوات لمساعدتك.



DEBUGGING


extern int exp_is_debugging;
extern FILE *exp_debugfile;



While expect dialogues seem very intuitive, trying to codify them
    in a program can reveal many surprises in a program's interface. Therefore a
    variety of debugging aids are available. They are controlled by the above
    variables, all 0 by default.


Debugging information internal to expect is sent to stderr
    when exp_is_debugging is non-zero. The debugging information includes
    every character received, and every attempt made to match the current input
    against the patterns. In addition, non-printable characters are translated
    to a printable form. For example, a control-C appears as a caret followed by
    a C. If exp_logfile is non-zero, this information is also written to
    that stream.


If exp_debugfile is non-zero, all normal and debugging
    information is written to that stream, regardless of the value of
    exp_is_debugging.

CAVEATS

The stream versions of the expect functions are much slower than the file descriptor versions because there is no way to portably read an unknown number of bytes without the potential of timing out. Thus, characters are read one at a time. You are therefore strongly encouraged to use the file descriptor versions of expect (although, automated versions of interactive programs don't usually demand high speed anyway).

You can actually get the best of both worlds, writing with the usual stream functions and reading with the file descriptor versions of expect as long as you don't attempt to intermix other stream input functions (e.g., fgetc). To do this, pass fileno(stream) as the file descriptor each time. Fortunately, there is little reason to use anything but the expect functions when reading from interactive programs.

There is no matching exp_pclose to exp_popen (unlike popen and pclose). It only takes two functions to close down a connection (fclose() followed by waiting on the pid), but it is not uncommon to separate these two actions by large time intervals, so the function seems of little value.

If you are running on a Cray running Unicos (all I know for sure from experience), you must run your compiled program as root or setuid. The problem is that the Cray only allows root processes to open ptys. You should observe as much precautions as possible: If you don't need permissions, setuid(0) only immediately before calling one of the spawn functions and immediately set it back afterwards.

Normally, spawn takes little time to execute. If you notice spawn taking a significant amount of time, it is probably encountering ptys that are wedged. A number of tests are run on ptys to avoid entanglements with errant processes. (These take 10 seconds per wedged pty.) Running expect with the -d option will show if expect is encountering many ptys in odd states. If you cannot kill the processes to which these ptys are attached, your only recourse may be to reboot.

BUGS

The exp_fexpect functions don't work at all under HP-UX - it appears to be a bug in getc. Follow the advice (above) about using the exp_expect functions (which doesn't need to call getc). If you fix the problem (before I do - please check the latest release) let me know.

SEE ALSO

An alternative to this library is the expect program. expect interprets scripts written in a high-level language which direct the dialogue. In addition, the user can take control and interact directly when desired. If it is not absolutely necessary to write your own C program, it is much easier to use expect to perform the entire interaction. It is described further in the following references:

"expect: Curing Those Uncontrollable Fits of Interactivity" by Don Libes, Proceedings of the Summer 1990 USENIX Conference, Anaheim, California, June 11-15, 1990.

"Using expect to Automate System Administration Tasks" by Don Libes, Proceedings of the 1990 USENIX Large Installation Systems Administration Conference, Colorado Springs, Colorado, October 17-19, 1990.

expect(1), alarm(3), read(2), write(2), fdopen(3), execve(2), execvp(3), longjmp(3), pty(4).

There are several examples C programs in the test directory of expect's source distribution which use the expect library.
