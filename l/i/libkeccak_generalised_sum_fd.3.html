libkeccak_generalised_sum_fd(3)

الحزمة: libkeccak

التراخيص: custom:ISC

LIBKECCAK_GENERALISED_SUM_FD(3) دليل وظائف المكتبة LIBKECCAK_GENERALISED_SUM_FD(3)

الاسم

libkeccak_generalised_sum_fd - حساب تجزئة الملف

مُلَخَّص

#include <libkeccak.h>
int libkeccak_generalised_sum_fd(int fd, struct libkeccak_state *state, const struct libkeccak_spec *spec,
                                 const char *suffix, void *hashsum);

الربط باستخدام -lkeccak.

الوصف

تقوم الدالة libkeccak_generalised_sum_fd() بحساب تجزئة الملف، الذي تم تحديده بواسطة واصف الملف fd (ويجب أن يكون في بداية الملف). يتم تحديد خوارزمية التجزئة بواسطة *spec و suffix، حيث *spec هو ضبط الخوارزمية و suffix هو البتات الملحقة بالرسالة (أو NULL إذا لم يكن هناك أي شيء.)

يتم تخزين التجزئة في شكل ثنائي في hashsum. يجب أن يكون حجم تخصيص hashsum على الأقل (((spec->output + 7) / 8) * sizeof(char)).

يجب عدم تهيئة *state إلا إذا كان spec هو NULL. تقوم libkeccak_generalised_sum_fd() بتهيئة *state بنفسها إلا إذا كان spec هو NULL. لذلك سيكون هناك تسرب للذاكرة إذا كان *state مهيأ بالفعل وكان spec غير NULL.

قيم الإرجاع

تعيد الدالة libkeccak_generalised_sum_fd() القيمة 0 عند اكتمال العملية بنجاح. عند حدوث خطأ، يتم إرجاع -1 ويتم تعيين errno لوصف الخطأ.

الأخطاء

قد تفشل الدالة libkeccak_generalised_sum_fd() لأي سبب، باستثناء تلك التي تؤدي إلى تعيين errno إلى EINTR، والمحددة للدوال read(2)، malloc(3)، و realloc(3).

ملاحظات

كن على علم، تقوم libkeccak_generalised_sum_fd() بتجزئة الملف حتى الوصول إلى النهاية. بالنسبة للأنابيب والمقابس، يعني هذا حتى يتم إغلاق الملف. ولكن بالنسبة لأجهزة الأحرف، هذا يعني عادةً عدم الانتهاء أبدًا. لذلك، فإن محاولة تجزئة الملفات في /dev هي عادة فكرة سيئة. لا تتحقق libkeccak_generalised_sum_fd() من طول الملف أو نوعه قبل التجزئة لأن هذا يمكن أن يحد مما يمكنك فعله، ويجعل المكتبة أكثر تعقيدًا.

libkeccak_generalised_sum_fd() لا تتوقف إذا تم مقاطعتها (read(2) ترجع EINTR.)

تفترض libkeccak_generalised_sum_fd() أن جميع المعلومات غير حساسة، ولن تقوم بالتالي بأي مسح آمن للمعلومات.

libkeccak_generalised_sum_fd() لا تتحقق من صحة ضبط الخوارزمية.

مثال

يحسب هذا المثال تجزئة Keccak[b = 1024, c = 576, n = 256] من الإدخال من الإدخال القياسي، ويطبع التجزئة، في شكل سداسي عشري، إلى الإخراج القياسي.

struct libkeccak_state state;
struct libkeccak_spec spec;
char binhash[256 / 8];
char hexhash[256 / 8 * 2 + 1];
spec.bitrate = 1024;
spec.capacity = 576;
spec.output = 256;
if (libkeccak_generalised_sum_fd(STDIN_FILENO, &state, &spec, NULL, binhash) < 0)
	goto fail;
libkeccak_behex_lower(hexhash, binhash, sizeof(binhash));
printf("%s\n", hexhash);
libkeccak_state_destroy(&state);
