libressl-d2i_SSL_SESSION(3)

الحزمة: libressl

التراخيص: ISC, custom:OpenSSL

D2I_SSL_SESSION(3) دليل وظائف المكتبة D2I_SSL_SESSION(3)

الاسم

d2i_SSL_SESSION، i2d_SSL_SESSION — تحويل كائن SSL_SESSION من/إلى تمثيل ASN1

الملخص

المكتبة “libssl” libcrypto
#include <openssl/ssl.h>

SSL_SESSION *
d2i_SSL_SESSION(SSL_SESSION **a, const unsigned char **pp, long length);

int
i2d_SSL_SESSION(SSL_SESSION *in, unsigned char **pp);

الوصف

d2i_SSL_SESSION() يحول التمثيل الخارجي ASN1 لجلسة SSL/TLS، المخزن كبيانات ثنائية في الموقع pp بطول length، إلى كائن SSL_SESSION.

i2d_SSL_SESSION() يحول كائن SSL_SESSION in إلى التمثيل ASN1 ويخزنه في موقع الذاكرة الذي يشير إليه pp. يتم إرجاع طول التمثيل ASN1 الناتج. إذا كان pp مؤشر NULL، يتم حساب الطول فقط وإرجاعه.

SSL_SESSION يتم بناؤه من عدة أجزاء تم تخصيصها باستخدام malloc(3)؛ لذلك لا يمكن نقله أو نسخه أو تخزينه مباشرة. لتخزين بيانات الجلسة على القرص أو في قاعدة بيانات، يجب تحويلها إلى تمثيل ASN1 ثنائي.

عند استخدام d2i_SSL_SESSION()، يتم تخصيص كائن SSL_SESSION تلقائيًا. عدد المراجع هو 1، لذا يجب إزالة الجلسة صراحة باستخدام SSL_SESSION_free(3)، ما لم يتم تولي كائن SSL_SESSION بالكامل، عند استدعائه داخل get_session_cb()، انظر SSL_CTX_sess_set_get_cb(3).

SSL_SESSION تحتفظ بمعلومات الربط الداخلية لقائمة ذاكرة التخزين المؤقت للجلسة عند إدراجها في ذاكرة التخزين المؤقت للجلسة لأحد كائنات SSL_CTX. لذلك، يجب استخدام كائن SSL_SESSION واحد، بغض النظر عن عدد المراجع، فقط مع كائن SSL_CTX واحد (وكائنات SSL المنشأة من هذا الكائن SSL_CTX).

عند استخدام i2d_SSL_SESSION()، يجب أن يكون موقع الذاكرة الذي يشير إليه pp كبيرًا بما يكفي لاستيعاب التمثيل الثنائي للجلسة. لا يوجد حد معروف لحجم تمثيل ASN1 الناتج، لذا قم باستدعاء i2d_SSL_SESSION() أولاً مع pp=NULL للحصول على الحجم المشفر، قبل تخصيص الكمية المطلوبة من الذاكرة واستدعاء i2d_SSL_SESSION() مرة أخرى. لاحظ أن هذا سيؤدي إلى تقديم القيمة الموجودة في *pp، لذا من الضروري حفظ نسخة من التخصيص الأصلي. على سبيل المثال:

char	*p, *pp;
int	 elen, len;

elen = i2d_SSL_SESSION(sess, NULL);
p = pp = malloc(elen);
if (p != NULL) {
	len = i2d_SSL_SESSION(sess, &pp);
	assert(elen == len);
	assert(p + len == pp);
}

قيم الإرجاع

d2i_SSL_SESSION() ترجع مؤشرًا إلى كائن SSL_SESSION الذي تم تخصيصه حديثًا. في حالة الفشل، يتم إرجاع مؤشر NULL ويمكن استرداد رسالة الخطأ من مكدس الأخطاء.

i2d_SSL_SESSION() ترجع حجم تمثيل ASN1 بالبايت. عندما تكون الجلسة غير صالحة، يتم إرجاع 0 ولا يتم إجراء أي عملية.

انظر أيضًا

d2i_X509(3)، ssl(3)، SSL_CTX_sess_set_get_cb(3)، SSL_SESSION_free(3)

التاريخ

d2i_SSL_SESSION() و i2d_SSL_SESSION() ظهرت لأول مرة في SSLeay 0.5.2 وكانت متاحة منذ OpenBSD 2.4.
