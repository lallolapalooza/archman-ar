libsres(3)

الحزمة: dnssec-tools

التراخيص: BSD

libsres(3) دليل المبرمج libsres(3)

الاسم

query_send(), response_rcv(), get() - إرسال استعلامات واستقبال ردود من خادم أسماء DNS.

clone_ns(), clone_ns_list(), free_name_server(), free_name_servers() - إدارة قوائم خوادم الأسماء

print_response() - عرض الإجابات التي تم إرجاعها من خادم الأسماء

الملخص

#include <resolver.h>
int query_send(const char    *name,
          const unsigned short type,
          const unsigned short class,
          struct name_server  *nslist,
          int                 edns0_size,
          int                 *trans_id);
int response_recv(int         *trans_id,
          fd_set              *pending_desc,
          struct timeval      *closest_event,
          struct name_server  **respondent,
          unsigned char       **response,
          size_t              *response_length);
int get(const char          *name_n,
        const unsigned short type_h,
        const unsigned short class_h,
        struct name_server  *nslist,
        struct name_server  **respondent,
        unsigned char       **response,
        size_t              *response_length);
int clone_ns(struct name_server **cloned_ns, 
        struct name_server *ns);
int clone_ns_list(struct name_server **ns_list,
                  struct name_server *orig_ns_list);
void free_name_server(struct name_server **ns);
void free_name_servers(struct name_server **ns);
void print_response(unsigned char *response, 
          size_t response_length);

الوصف

تقوم الدالة query_send() بإرسال استعلام إلى خوادم الأسماء المحددة في nslist. يتكون الاستعلام من الثلاثي <name, class, type> ويوفر trans_id مقبضًا لهذه المعاملة داخل مكتبة libsres. يمكن تعيين حجم المخزن المؤقت المعلن عنه في خيار EDNS0 باستخدام الوسيط ends0_size.

تقوم الدالة response_recv() بإرجاع الإجابات، إن وجدت، من خادم الأسماء الذي يستجيب للاستعلام المحدد بواسطة trans_id. تكون الاستجابة متاحة في response ويتم إرجاع خادم الأسماء المستجيب في respondent. يتم إرجاع طول الاستجابة بالبايت في response_length.

توفر الدالة get() غلافًا حول الدالتين query_send() و response_recv(). بعد إرسال طلب، فإنها تنتظر حتى يتم استلام استجابة من بعض خوادم الأسماء أو حتى انتهاء صلاحية الطلب. لا تتبع مكتبة libsres الإحالات تلقائيًا؛ يتم التعامل مع الاستجابات التي تحتوي على إحالات كاستجابات صالحة.

يتم تخصيص الذاكرة التي تشير إليها *respondent داخليًا بواسطة مكتبة libsres ويجب على المستدعي تحريرها باستخدام free_name_server(). يمكن تحرير قائمة كاملة من خوادم الأسماء باستخدام free_name_servers(). يمكن إنشاء نسخة من خادم أسماء باستخدام clone_ns() ونسخة من قائمة خوادم الأسماء باستخدام clone_ns_list().

توفر print_response() طريقة سهلة لعرض الإجابات التي تم إرجاعها في response بواسطة خادم الأسماء.

يتم تعريف البنية name_server في resolver.h على النحو التالي:

#define NS_MAXCDNAME    255
struct name_server
{
    unsigned char ns_name_n[NS_MAXCDNAME];
    void *ns_tsig;
    unsigned int ns_security_options;
    unsigned int ns_status;
    unsigned long ns_options;
    int ns_retry;
    int ns_retrans;
    struct name_server *ns_next;
    int ns_number_of_addresses;
    struct sockaddr_storage **ns_address;
};
ns_name_n
اسم المنطقة التي يعد خادم الأسماء هذا مسؤولاً عنها.
ns_tsig
مفتاح tsig الذي يجب استخدامه لحماية الرسائل المرسلة إلى خادم الأسماء هذا. هذا الحقل غير مستخدم حاليًا ويجب تعيينه إلى NULL.
ns_security_options
خيارات الأمان للمنطقة. هذا الحقل غير مستخدم حاليًا ويجب تعيينه إلى ZONE_USE_NOTHING.
ns_status
حالة المنطقة. يشير هذا الحقل إلى كيفية الحصول على معلومات المنطقة. يجب على المستدعي تعيين هذه القيمة إلى SR_ZI_STATUS_UNSET. المناطق التي تم الحصول على معلوماتها من خلال الإحالات لها قيمة SR_ZI_STATUS_LEARNED لهذا الحقل.
ns_options
يحدد علامات إضافية للمحلل. العلامات المعرفة حاليًا هي SR_QUERY_RECURSE، والتي تعين علامة "Recursion Desired"؛ SR_QUERY_SET_DO، والتي تعين بت "DNSSEC OK"؛ في رأس EDNS0؛ SR_QUERY_SET_CD، والتي تعين بت "DNSSEC CD"؛ في رأس EDNS0؛ و SR_QUERY_DEBUG، والتي تمكن التصحيح. تعين SR_QUERY_VALIDATING_STUB_FLAGS كل من SR_QUERY_SET_DO و SR_QUERY_SET_CD.
ns_retry
يحدد الحد الأقصى لعدد المحاولات التي يجب إجراؤها للحصول على اسم من خادم أسماء غير مستجيب قبل الاستسلام.
ns_retrans
يحدد فترة إعادة الإرسال بالثواني للاستعلامات المرسلة إلى خوادم الأسماء غير المستجيبة.
ns_next
عنوان خادم الأسماء التالي في القائمة.
ns_number_of_addresses
عدد العناصر في المصفوفة ns_addresses. هذا الحقل غير مستخدم حاليًا.
ns_addresses
عنوان IP لخادم الأسماء.

رموز أخرى مصدَّرة

تقوم مكتبة libsres أيضًا بتصدير دوال BIND التالية، والتي يمكن العثور على وثائقها في مصادر ووثائق BIND:

res_nametoclass
res_nametotype
ns_name_ntop
ns_name_pton
ns_name_unpack
ns_parse_ttl
p_class
p_section
p_type

تمت زيادة الدالة p_type() المصدرة من libsres بحيث تتعرف على رموز أنواع DNSSEC المختلفة مثل DNSKEY و RRSIG و NSEC و NSEC3 و DLV.

قيم الإرجاع

SR_UNSET
لا يوجد خطأ.
SR_CALL_ERROR
تم تمرير معلمة غير صالحة إلى get() أو query_send() أو response_recv().
SR_INTERNAL_ERROR
واجه المحلل خطأ داخليًا.
SR_TSIG_ERROR
واجه المحلل خطأ متعلقًا بـ TSIG. هذا غير مطبق حاليًا.
SR_NO_ANSWER
لم يتم استلام أي إجابات من أي خادم أسماء.
SR_NO_ANSWER_YET
لا توجد إجابة متاحة حاليًا؛ الاستعلام لا يزال نشطًا.
SR_HEADER_ERROR
كان طول وعدد السجلات في الرأس غير صحيحين.
SR_NXDOMAIN
الاسم الذي تم الاستعلام عنه لم يكن موجودًا.
SR_FORMERR
لم يتمكن خادم الأسماء من تحليل رسالة الاستعلام.
SR_SERVFAIL
لم يكن خادم الأسماء قابلاً للوصول.
SR_NOTIMPL
وظيفة معينة لم يتم تنفيذها بعد.
SR_REFUSED
رفض خادم الأسماء الإجابة على هذا الاستعلام.
SR_DNS_GENERIC_FAILURE
فشل آخر تم إرجاعه بواسطة خادم الأسماء وينعكس في رسالة RCODE التي تم إرجاعها.
SR_EDNS_VERSION_ERROR

قيم الإرجاع

SR_EDNS_VERSION_NOT_SUPPORTED
لم يتم التعرف على إصدار EDNS.
SR_NAME_EXPANSION_FAILURE
حدث فشل أثناء محاولة توسيع اسم نطاق مضغوط.

الحالة الحالية

لا يوجد حاليًا دعم لـ IPv6.

هناك دعم محدود لتحديد سياسة المحلل؛ لا يزال يتعين تغيير أعضاء struct name_server.
