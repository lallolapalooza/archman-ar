libressl-tls_config_set_session_id(3)

الحزمة: libressl

التراخيص: ISC, custom:OpenSSL

TLS_CONFIG_SET_SESSION_ID(3) دليل وظائف المكتبة TLS_CONFIG_SET_SESSION_ID(3)

الاسم

tls_config_set_session_fd، tls_config_set_session_id، tls_config_set_session_lifetime، tls_config_add_ticket_key — تكوين استئناف مصافحات TLS

الشكل العام

مكتبة "libtls" libssl libcrypto
#include <tls.h>

int
tls_config_set_session_fd(struct tls_config *config, int session_fd);

int
tls_config_set_session_id(struct tls_config *config, const unsigned char *session_id, size_t len);

int
tls_config_set_session_lifetime(struct tls_config *config, int lifetime);

int
tls_config_add_ticket_key(struct tls_config *config, uint32_t keyrev, unsigned char *key, size_t keylen);

الوصف

tls_config_set_session_fd() يقوم بتعيين واصف ملف لاستخدامه لإدارة بيانات جلسات TLS (العميل فقط). يجب أن يكون واصف الملف المقدم ملفًا عاديًا وأن يكون مملوكًا للمستخدم الحالي، مع تقييد الأذونات للسماح فقط للمالك بقراءة الملف وكتابته (0600). إذا كان للملف طول غير صفري، فسيحاول العميل قراءة بيانات الجلسة من هذا الملف واستئناف جلسة TLS السابقة مع الخادم. عند نجاح المصافحة، سيتم تحديث الملف ببيانات الجلسة الحالية، إن وجدت. يتحمل المتصل مسؤولية إغلاق واصف الملف هذا، بعد تحرير جميع سياقات TLS التي تم تكوينها لاستخدامه عبر tls_free().

tls_config_set_session_id() يقوم بتعيين معرف الجلسة الذي سيستخدمه خادم TLS عند تمكين الجلسات (الخادم فقط). افتراضيًا، يتم استخدام قيمة عشوائية.

tls_config_set_session_lifetime() يقوم بتعيين فترة الحياة لاستخدامها لجلسات TLS (الخادم فقط). يتم تعطيل دعم الجلسة إذا تم تحديد فترة حياة صفر، وهو الافتراضي.

tls_config_add_ticket_key() يضيف مفتاحًا يستخدم لتشفير وتصديق تذاكر TLS (الخادم فقط). افتراضيًا، يتم إنشاء المفاتيح وتدويرها تلقائيًا بناءً على فترة حياتها. يجب استخدام هذه الوظيفة فقط لمزامنة مفتاح تشفير التذكرة عبر عمليات متعددة. سيؤدي إعادة إضافة مفتاح معروف إلى حدوث خطأ، ما لم يكن هو أحدث مفتاح تم إضافته.

قيم الإرجاع

تعيد هذه الوظائف 0 عند النجاح أو -1 عند الخطأ.
