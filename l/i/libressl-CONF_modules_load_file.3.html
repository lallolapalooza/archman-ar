libressl-CONF_modules_load_file(3)

الحزمة: libressl

التراخيص: ISC, custom:OpenSSL

CONF_MODULES_LOAD_FILE(3) دليل وظائف المكتبة CONF_MODULES_LOAD_FILE(3)

الاسم

CONF_modules_load_file، CONF_modules_load، X509_get_default_cert_area — وظائف تهيئة OpenSSL

الملخص

المكتبة «libcrypto»
#include <openssl/conf.h>

int
CONF_modules_load_file(const char *filename، const char *appname، unsigned long flags);

int
CONF_modules_load(const CONF *cnf، const char *appname، unsigned long flags);

#include <openssl/x509.h>

const char *
X509_get_default_cert_area(void);

الوصف

تقوم الوظيفة CONF_modules_load_file() بتهيئة OpenSSL باستخدام الملف filename بتنسيق openssl.cnf(5) واسم التطبيق appname. إذا كان filename هو NULL، فسيتم استخدام ملف تهيئة OpenSSL القياسي /etc/ssl/openssl.cnf. إذا كان appname هو NULL، فسيتم استخدام اسم تطبيق OpenSSL القياسي «openssl_conf». يمكن تخصيص السلوك باستخدام flags.

راجع قسم أمثلة لمعرفة الوظائف الإضافية التي قد تحتاج إلى استدعائها. قد يكون استدعاء وظائف التهيئة بالترتيب الصحيح للتأثير المقصود أمرًا صعبًا لأن العديد من وظائف التهيئة تستدعي بعضها البعض داخليًا.

CONF_modules_load() مشابهة لـ CONF_modules_load_file() باستثناء أنها تقرأ معلومات التهيئة من cnf.

العلامات التالية flags معترف بها حاليًا:

CONF_MFLAGS_IGNORE_ERRORS
تجاهل الأخطاء التي تعيدها وحدات التهيئة الفردية. افتراضيًا، يعتبر خطأ الوحدة الأول قاتلًا ولن يتم تحميل أي وحدات أخرى.
CONF_MFLAGS_SILENT
لا تقم بإضافة أي معلومات خطأ. افتراضيًا، تضيف جميع أخطاء الوحدات معلومات خطأ إلى قائمة الأخطاء.
CONF_MFLAGS_NO_DSO
تعطيل تحميل وحدات التهيئة من DSOs. هذه العلامة متوفرة للتوافق وليس لها تأثير.
CONF_MFLAGS_IGNORE_MISSING_FILE
دع CONF_modules_load_file() تتجاهل ملفات التهيئة المفقودة. افتراضيًا، يعيد الملف المفقود خطأ.
CONF_MFLAGS_DEFAULT_SECTION
إذا لم يكن appname هو NULL ولكنه غير موجود، فالرجوع إلى القسم الافتراضي «openssl_conf».

باستخدام CONF_modules_load_file() مع العلامات المناسبة، يمكن للتطبيق تخصيص تهيئة التطبيق لتناسب احتياجاته على أفضل وجه. في بعض الحالات، يكون استخدام ملف التهيئة اختياريًا وغيابه ليس خطأ: في هذه الحالة، سيتم تعيين CONF_MFLAGS_IGNORE_MISSING_FILE.

قد يتم التعامل مع الأخطاء أثناء التهيئة بشكل مختلف من قبل تطبيقات مختلفة. على سبيل المثال، في بعض الحالات، قد يصدر الخطأ ببساطة رسالة تحذير وقد يستمر التطبيق. في حالات أخرى، قد يعتبر التطبيق أن خطأ ملف التهيئة قاتل ويخرج على الفور.

يمكن للتطبيقات استخدام وظيفة CONF_modules_load() إذا كانت ترغب في تحميل ملف تهيئة بنفسها ولديها تحكم أدق في كيفية معالجة الأخطاء.

قيم الإرجاع

CONF_modules_load_file() و CONF_modules_load() ترجع 1 للنجاح وصفر أو قيمة سالبة للفشل. إذا لم يتم تجاهل أخطاء الوحدات، فإن رمز الإرجاع سيعكس قيمة الإرجاع للوحدة الفاشلة (والتي ستكون دائمًا صفرًا أو سالبة).

X509_get_default_cert_area() ترجع مؤشرًا إلى السلسلة الثابتة «/etc/ssl».

الملفات

/etc/ssl
دليل التهيئة القياسي
/etc/ssl/openssl.cnf
ملف التهيئة القياسي

أمثلة

تحميل ملف تهيئة وطباعة أي أخطاء والخروج (يعتبر الملف المفقود قاتلًا):

if (CONF_modules_load_file(NULL, NULL, 0) <= 0) {
	fprintf(stderr, "FATAL: error loading configuration file\n");
	ERR_print_errors_fp(stderr);
	exit(1);
}

تحميل ملف التهيئة الافتراضي باستخدام القسم المشار إليه بواسطة «myapp»، مع تحمل الملفات المفقودة، ولكن الخروج عند الأخطاء الأخرى:

if (CONF_modules_load_file(NULL, "myapp",
    CONF_MFLAGS_IGNORE_MISSING_FILE) <= 0) {
	fprintf(stderr, "FATAL: error loading configuration file\n");
	ERR_print_errors_fp(stderr);
	exit(1);
}

تحميل ملف قسم تهيئة مخصص بدلاً من الملف القياسي، مع طباعة تحذيرات فقط عند الخطأ، مع تجاهل ملف التهيئة المفقود:

OPENSSL_no_config();
if (CONF_modules_load_file("/something/app.cnf", "myapp",
    CONF_MFLAGS_IGNORE_MISSING_FILE) <= 0) {
	fprintf(stderr, "WARNING: error loading configuration file\n");
	ERR_print_errors_fp(stderr);
}

في المثال السابق، يلزم استدعاء OPENSSL_no_config(3) أولاً لقمع التحميل التلقائي لملف التهيئة القياسي.

تحميل وتحليل ملف التهيئة يدويًا، مع معالجة مخصصة للأخطاء:

FILE	*fp;
CONF	*cnf = NULL;
long	 eline;

fp = fopen("/somepath/app.cnf", "r");
if (fp == NULL) {
	fprintf(stderr, "Error opening configuration file\n");
	/* Other missing configuration file behaviour */
} else {
	cnf = NCONF_new(NULL);
	if (NCONF_load_fp(cnf, fp, &eline) == 0) {
		fprintf(stderr, "Error on line %ld of configuration file\n",
		    eline);
		ERR_print_errors_fp(stderr);
		/* Other malformed configuration file behaviour */
	} else if (CONF_modules_load(cnf, "appname", 0) <= 0) {
		fprintf(stderr, "Error configuring application\n");
		ERR_print_errors_fp(stderr);
		/* Other configuration error behaviour */
	}
	fclose(fp);
	NCONF_free(cnf);
}
