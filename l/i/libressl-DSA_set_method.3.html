libressl-DSA_set_method(3)

الحزمة: libressl

التراخيص: ISC، مخصص:OpenSSL

DSA_SET_METHOD(3) دليل الدوال المكتبية DSA_SET_METHOD(3)

الاسم

DSA_set_default_method، DSA_get_default_method، DSA_set_method، DSA_new_method، DSA_OpenSSL — اختيار طريقة DSA

الملخص

المكتبة “libcrypto”
#include <openssl/dsa.h>

void
DSA_set_default_method(const DSA_METHOD *meth);

const DSA_METHOD *
DSA_get_default_method(void);

int
DSA_set_method(DSA *dsa, const DSA_METHOD *meth);

DSA *
DSA_new_method(ENGINE *engine);

DSA_METHOD *
DSA_OpenSSL(void);

الوصف

يحتوي كائن DSA_METHOD على مؤشرات إلى الدوال المستخدمة لعمليات DSA. افتراضيًا، يتم استخدام التنفيذ الداخلي الذي تم إرجاعه بواسطة DSA_OpenSSL(). عن طريق اختيار طريقة أخرى، يمكن استخدام تطبيقات بديلة مثل مسرعات الأجهزة.

DSA_set_default_method() يختار meth كطريقة افتراضية لجميع هياكل DSA التي تم إنشاؤها لاحقًا.

DSA_get_default_method() يعيد مؤشرًا إلى الطريقة الافتراضية الحالية.

DSA_set_method() يختار meth لتنفيذ جميع العمليات باستخدام المفتاح dsa. هذا يحل محل DSA_METHOD المستخدم بواسطة مفتاح DSA. من الممكن أن يكون لمفاتيح DSA التي تعمل فقط مع تطبيقات DSA_METHOD معينة، وفي مثل هذه الحالات قد يكون لمحاولة تغيير DSA_METHOD للمفتاح نتائج غير متوقعة.

DSA_new_method() تخصص وتهيئ هيكل DSA. يتم تجاهل الوسيط engine ويتم استخدام الطريقة الافتراضية التي يتم التحكم فيها بواسطة DSA_set_default_method().

يتم تعريف هيكل DSA_METHOD على النحو التالي:

struct {
	/* اسم التنفيذ */
	const char *name;
	/* توقيع */
	DSA_SIG *(*dsa_do_sign)(const unsigned char *dgst, int dlen,
	    DSA *dsa);
	/* حساب مسبق لـ k^-1 و r */
	int (*dsa_sign_setup)(DSA *dsa, BN_CTX *ctx_in, BIGNUM **kinvp,
	    BIGNUM **rp);
	/* تحقق */
	int (*dsa_do_verify)(const unsigned char *dgst, int dgst_len,
	    DSA_SIG *sig, DSA *dsa);
	/* يتم استدعاؤه عند DSA_new */
	int (*init)(DSA *DSA);
	/* يتم استدعاؤه عند DSA_free */
	int (*finish)(DSA *DSA);
	int flags;
} DSA_METHOD;

قيم الإرجاع

DSA_OpenSSL() و DSA_get_default_method() تعيد مؤشرات إلى DSA_METHOD المعني.

DSA_set_method() تعيد 1 عند النجاح أو 0 عند الفشل. حاليًا، لا يمكن أن تفشل.

DSA_new_method() تعيد NULL وتعيين رمز خطأ يمكن الحصول عليه بواسطة ERR_get_error(3) إذا فشل التخصيص. بخلاف ذلك، تعيد مؤشرًا إلى الهيكل المخصص حديثًا.
