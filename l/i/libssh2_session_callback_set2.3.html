libssh2_session_callback_set2(3)

الحزمة: libssh2

التراخيص: BSD

libssh2_session_callback_set2(3) libssh2 libssh2_session_callback_set2(3)

الاسم

libssh2_session_callback_set2 - تعيين دالة رد اتصال

الهيكل

#include <libssh2.h>
libssh2_cb_generic *
libssh2_session_callback_set2(LIBSSH2_SESSION *session, int cbtype,
                              libssh2_cb_generic *callback);

الوصف

يعين معالج رد اتصال مخصص لكائن جلسة تم تهيئته مسبقًا. يتم تشغيل ردود الاتصال عن طريق استلام حزم خاصة في طبقة النقل. لتعطيل رد اتصال، قم بتعيينه إلى NULL.

session - مثيل الجلسة كما تم إرجاعه بواسطة libssh2_session_init_ex(3)

cbtype - نوع رد الاتصال. أحد الأنواع المدرجة في أنواع ردود الاتصال.

callback - مؤشر إلى دالة رد اتصال مخصصة. يجب أن يتطابق بروتوكول هذه الدالة مع ماكرو تعريف رد الاتصال المرتبط.

أنواع ردود الاتصال

LIBSSH2_CALLBACK_IGNORE
يتم استدعاؤه عند استلام رسالة SSH_MSG_IGNORE
LIBSSH2_CALLBACK_DEBUG
يتم استدعاؤه عند استلام رسالة SSH_MSG_DEBUG
LIBSSH2_CALLBACK_DISCONNECT
يتم استدعاؤه عند استلام رسالة SSH_MSG_DISCONNECT
LIBSSH2_CALLBACK_MACERROR
يتم استدعاؤه عند اكتشاف MAC غير متطابق في طبقة النقل. إذا أعادت الدالة 0، فسيتم قبول الحزمة على الرغم من ذلك.
LIBSSH2_CALLBACK_X11
يتم استدعاؤه عند قبول اتصال X11
LIBSSH2_CALLBACK_SEND
يتم استدعاؤه عندما ترغب libssh2 في إرسال بيانات عبر الاتصال. يمكن تعيينه إلى دالة مخصصة للتعامل مع الإدخال/الإخراج بطريقتك الخاصة.

بروتوكول رد الاتصال:

ssize_t sendcb(libssh2_socket_t sockfd, const void *buffer,
               size_t length, int flags, void **abstract);

sockfd هو المقبس الذي سيتم الكتابة إليه، buffer يشير إلى البيانات المراد إرسالها، length هو حجم البيانات، flags هي العلامات التي كان سيتم استخدامها لاستدعاء send()، و abstract هو مؤشر إلى المؤشر المجرد المعين في استدعاء libssh2_session_init_ex(3).

تعيد دالة رد الاتصال عدد البايتات المرسلة، أو -1 في حالة حدوث خطأ. يمكن إعادة رمز الإرجاع الخاص -EAGAIN للإشارة إلى أنه تم إلغاء الإرسال لمنع الحظر وأنه بحاجة إلى استدعائه مرة أخرى.

LIBSSH2_CALLBACK_RECV
يتم استدعاؤه عندما ترغب libssh2 في قراءة بيانات من الاتصال. يمكن تعيينه إلى دالة مخصصة للتعامل مع الإدخال/الإخراج بطريقتك الخاصة.

بروتوكول رد الاتصال:

ssize_t recvcb(libssh2_socket_t sockfd, void *buffer,
               size_t length, int flags, void **abstract);

sockfd هو المقبس الذي سيتم القراءة منه، buffer المكان الذي سيتم تخزين البيانات المستلمة فيه، length هو حجم المخزن المؤقت، flags هي العلامات التي كان سيتم استخدامها لاستدعاء recv()، و abstract هو مؤشر إلى المؤشر المجرد المعين في استدعاء libssh2_session_init_ex(3).

تعيد دالة رد الاتصال عدد البايتات المقروءة، أو -1 في حالة حدوث خطأ. يمكن إعادة رمز الإرجاع الخاص -EAGAIN للإشارة إلى أنه تم إلغاء القراءة لمنع الحظر وأنه بحاجة إلى استدعائه مرة أخرى.

LIBSSH2_CALLBACK_AUTHAGENT
يتم استدعاؤه أثناء عملية المصادقة للسماح للعميل بالاتصال بـ ssh-agent وإجراء أي إعدادات، مثل تكوين الوكيل أو إضافة مفاتيح.

بروتوكول رد الاتصال:

void authagent(LIBSSH2_SESSION* session, LIBSSH2_CHANNEL *channel,
               void **abstract);
LIBSSH2_CALLBACK_AUTHAGENT_IDENTITIES
لا يتم استدعاؤه بواسطة libssh2. العميل مسؤول عن استدعاء هذه الطريقة عند استلام رسالة SSH2_AGENTC_REQUEST_IDENTITIES.

بروتوكول رد الاتصال:

void identities(LIBSSH2_SESSION* session, void *buffer,
                const char *agent_path,
                void **abstract)

يجب ملء buffer بواسطة رد الاتصال. قد تقوم عملاء مختلفون بتنفيذ ذلك بشكل مختلف. على سبيل المثال، قد يقوم عميل بتمرير unsigned char ** لهذا المعامل، بينما قد يقوم عميل آخر بتمرير مؤشر إلى هيكل.

بغض النظر عن نوع المخزن المؤقت المستخدم، سيحتاج العميل إلى إعادة قائمة بالهويات بالتنسيق التالي.

uint32 طول المخزن المؤقت uint32 عدد الإدخالات إدخالات

حيث يكون كل إدخال في قائمة الإدخالات بالتنسيق التالي:

سلسلة بيانات cstring تعليق

agent_path المسار إلى ssh-agent قيد التشغيل على جهاز العميل، والذي يمكن من خلاله سرد الهويات.

LIBSSH2_CALLBACK_AUTHAGENT_SIGN
لا يتم استدعاؤه بواسطة libssh2. العميل مسؤول عن استدعاء هذه الطريقة عند استلام رسالة SSH2_AGENTC_SIGN_REQUEST.

بروتوكول رد الاتصال:

void sign(LIBSSH2_SESSION* session,
          unsigned char *blob, unsigned int blen,
          const unsigned char *data, unsigned int dlen,
          unsigned char **sig, unsigned int *sig_len,
          const char *agent_path,
          void **abstract);

عند التفاعل مع ssh-agent مثبت على النظام العميل، يمكن لهذه الطريقة استدعاء libssh2_agent_sign(3) لإجراء عملية التوقيع.

قيمة الإرجاع

مؤشر إلى معالج رد الاتصال السابق. يعيد NULL إذا لم يتم تعيين معالج رد اتصال سابق أو كان نوع رد الاتصال غير معروف.
