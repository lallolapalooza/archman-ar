libressl-EVP_PKEY_CTX_set_hkdf_md(3)

الحزمة: libressl

التراخيص: ISC, custom:OpenSSL

EVP_PKEY_CTX_SET_HKDF_MD(3) دليل الدوال المكتبية EVP_PKEY_CTX_SET_HKDF_MD(3)

الاسم

EVP_PKEY_CTX_set_hkdf_md، EVP_PKEY_CTX_set1_hkdf_salt، EVP_PKEY_CTX_set1_hkdf_key، EVP_PKEY_CTX_add1_hkdf_info، EVP_PKEY_CTX_hkdf_mode — خوارزمية اشتقاق المفاتيح المستندة إلى HMAC (الاستخلاص والتوسيع)

الموجز

مكتبة «libcrypto»
#include <openssl/evp.h>
#include <openssl/kdf.h>

int
EVP_PKEY_CTX_hkdf_mode(EVP_PKEY_CTX *pctx, int mode);

int
EVP_PKEY_CTX_set_hkdf_md(EVP_PKEY_CTX *pctx, const EVP_MD *md);

int
EVP_PKEY_CTX_set1_hkdf_salt(EVP_PKEY_CTX *pctx, unsigned char *salt, int saltlen);

int
EVP_PKEY_CTX_set1_hkdf_key(EVP_PKEY_CTX *pctx, unsigned char *key, int keylen);

int
EVP_PKEY_CTX_add1_hkdf_info(EVP_PKEY_CTX *pctx, unsigned char *info, int infolen);

الوصف

تنفيذ خوارزمية EVP_PKEY_HKDF وظيفة اشتقاق المفاتيح HKDF. يتبع HKDF نموذج "الاستخلاص ثم التوسيع"، حيث تتكون وظيفة اشتقاق المفاتيح منطقياً من وحدتين. تأخذ المرحلة الأولى مادة المفتاح المدخلة و"تستخلص" منها مفتاحًا شبه عشوائيًا ثابت الطول K. تقوم المرحلة الثانية "بتوسيع" المفتاح K إلى عدة مفاتيح شبه عشوائية إضافية (ناتج وظيفة اشتقاق المفاتيح).

EVP_PKEY_CTX_hkdf_mode() يضبط الوضع لعملية HKDF. هناك ثلاثة أوضاع محددة حاليًا:

EVP_PKEY_HKDEF_MODE_EXTRACT_AND_EXPAND
هذا هو الوضع الافتراضي. سيقوم استدعاء EVP_PKEY_derive(3) على EVP_PKEY_CTX تم إعداده لـ HKDF بإجراء استخلاص متبوعًا بعملية توسيع دفعة واحدة. سيكون المفتاح المشتق المُعاد هو النتيجة بعد عملية التوسيع. المفتاح شبه العشوائي ثابت الطول الوسيط K غير مُعاد.

في هذا الوضع، يجب تعيين قيم الهضم (digest) والمفتاح والملح (salt) والمعلومات (info) قبل اشتقاق مفتاح أو حدوث خطأ.

EVP_PKEY_HKDEF_MODE_EXTRACT_ONLY
في هذا الوضع، سيقوم استدعاء EVP_PKEY_derive(3) بإجراء عملية الاستخلاص فقط. ستكون القيمة المُعادة هي المفتاح شبه العشوائي ثابت الطول الوسيط K.

يجب تعيين قيم الهضم والمفتاح والملح قبل اشتقاق مفتاح أو حدوث خطأ.

EVP_PKEY_HKDEF_MODE_EXPAND_ONLY
في هذا الوضع، سيقوم استدعاء EVP_PKEY_derive(3) بإجراء عملية التوسيع فقط. يجب تعيين المفتاح المدخل إلى المفتاح شبه العشوائي ثابت الطول الوسيط K المُعاد من عملية استخلاص سابقة.

يجب تعيين قيم الهضم والمفتاح والمعلومات قبل اشتقاق مفتاح أو حدوث خطأ.

EVP_PKEY_CTX_set_hkdf_md() يضبط خوارزمية الهضم (digest) المرتبطة بـ HKDF.

EVP_PKEY_CTX_set1_hkdf_salt() يضبط الملح إلى saltlen بايت من المخزن المؤقت salt. يتم استبدال أي قيمة موجودة.

EVP_PKEY_CTX_set1_hkdf_key() يضبط المفتاح إلى keylen بايت من المخزن المؤقت key. يتم استبدال أي قيمة موجودة.

EVP_PKEY_CTX_add1_hkdf_info() يضبط قيمة المعلومات إلى infolen بايت من المخزن المؤقت info. إذا كانت قيمة ما مضبوطة بالفعل، فسيتم إلحاقها بالقيمة الموجودة.

عناصر تحكم السلسلة

يدعم HKDF أيضًا عمليات تحكم قائمة على السلاسل عبر EVP_PKEY_CTX_ctrl_str(3). يستخدم المعامل type "md" القيمة value المقدمة كاسم لخوارزمية الهضم المراد استخدامها. المعامل type "mode" يقبل "EXTRACT_AND_EXPAND"، "EXTRACT_ONLY" و "EXPAND_ONLY" كـ value لتحديد الوضع المراد استخدامه. المعاملات type "salt"، "key" و "info" تستخدم المعامل value المقدم كمفتاح، أو مفتاح، أو معلومات. الأسماء "hexsalt"، "hexkey" و "hexinfo" متشابهة باستثناء أنها تأخذ سلسلة سداسية عشرية يتم تحويلها إلى ثنائي.

ملاحظات

جميع هذه الدوال منفذة كـ وحدات ماكرو.

يمكن الحصول على سياق لـ HKDF عن طريق استدعاء:

EVP_PKEY_CTX *pctx = EVP_PKEY_CTX_new_id(EVP_PKEY_HKDF, NULL);

لا يمكن أن يتجاوز الطول الإجمالي لمخزن المعلومات المؤقت 1024 بايت: يجب أن يكون هذا كافياً لأي استخدام عادي لـ HKDF.

يتم تحديد طول المخرج لعملية توسيع HKDF عبر معامل الطول لوظيفة EVP_PKEY_derive(3). نظرًا لأن طول مخرج HKDF متغير، فإن تمرير مخزن مؤقت NULL كوسيلة للحصول على الطول المطلوب ليس ذا مغزى مع HKDF في أي وضع يقوم بعملية توسيع. بدلاً من ذلك، يجب على المتصل تخصيص مخزن مؤقت بالطول المطلوب، وتمرير هذا المخزن المؤقت إلى EVP_PKEY_derive(3) بالإضافة إلى (مؤشر تم تهيئته إلى) الطول المطلوب. يُسمح بتمرير مخزن مؤقت NULL للحصول على الطول عند استخدام EVP_PKEY_HKDEF_MODE_EXTRACT_ONLY.

قيم الإرجاع

ترجع جميع هذه الدوال 1 للنجاح و 0 أو قيمة سالبة للفشل. على وجه الخصوص، يشير رمز الإرجاع -2 إلى أن العملية غير مدعومة من قبل خوارزمية المفتاح العام.

أمثلة

يشتق هذا المثال 10 بايت باستخدام SHA-256 مع المفتاح السري "secret"، وقيمة الملح "salt"، وقيمة المعلومات "label":

EVP_PKEY_CTX *pctx;
unsigned char out[10];
size_t outlen = sizeof(out);

if ((pctx = EVP_PKEY_CTX_new_id(EVP_PKEY_HKDF, NULL)) == NULL)
	/* خطأ */

if (EVP_PKEY_derive_init(pctx) <= 0)
	/* خطأ */
if (EVP_PKEY_CTX_set_hkdf_md(pctx, EVP_sha256()) <= 0)
	/* خطأ */
if (EVP_PKEY_CTX_set1_hkdf_salt(pctx, "salt", 4) <= 0)
	/* خطأ */
if (EVP_PKEY_CTX_set1_hkdf_key(pctx, "secret", 6) <= 0)
	/* خطأ */
if (EVP_PKEY_CTX_add1_hkdf_info(pctx, "label", 5) <= 0)
	/* خطأ */
if (EVP_PKEY_derive(pctx, out, &outlen) <= 0)
	/* خطأ */
