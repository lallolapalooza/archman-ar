libssh2_sftp_lstat(3)

الحزمة: libssh2

التراخيص: BSD

libssh2_sftp_lstat(3) libssh2 libssh2_sftp_lstat(3)

الاسم

libssh2_sftp_lstat - ماكرو اختصار لاستدعاءات libssh2_sftp_stat_ex(3)

الملخص

#include <libssh2.h>
#include <libssh2_sftp.h>
int
libssh2_sftp_lstat(LIBSSH2_SFTP *sftp, const char *path,
                   LIBSSH2_SFTP_ATTRIBUTES *attrs);

الوصف

هذا ماكرو معرف في ملف رأس عام لـ libssh2 يستخدم الوظيفة الأساسية libssh2_sftp_stat_ex(3).
