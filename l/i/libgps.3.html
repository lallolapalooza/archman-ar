libgps(3)

LIBGPS(3) توثيق GPSD LIBGPS(3)

الاسم

libgps - مكتبة خدمة C للتواصل مع برنامج GPS daemon

الملخص

C:

#include <gps.h>
int gps_open(char * server, char * port, struct gps_data_t * gpsdata)
int gps_send(struct gps_data_t * gpsdata, char * fmt, ...)
int gps_read(struct gps_data_t * gpsdata, char * message,
             int message_size)
bool gps_waiting(const struct gps_data_t * gpsdata, int timeout)
char * gps_data(const struct gps_data_t * gpsdata)
int gps_unpack(char * buf, struct gps_data_t * gpsdata)
int gps_close(struct gps_data_t * gpsdata)
int gps_stream(struct gps_data_t * gpsdata, unsigned int flags,
               void * data)
int gps_mainloop(struct gps_data_t * gpsdata, int timeout,
                 void (* hook)(struct gps_data_t *gpsdata))
const char * gps_errstr(int err)

Python:

import gps
session = gps.gps(host="localhost", port="2947")
session.stream(flags=gps.WATCH_JSON)
while 0 == session.read():
    process(session)
del session

الوصف

C

libgps هي مكتبة خدمة تدعم التواصل مع نسخة من برنامج gpsd(8)، قم بربطها بخيار الرابط -lgps. قد تحتاج بعض الأنظمة أيضًا إلى -lm.


تحذير

احرص على الشرطية في كودك بناءً على رموز إصدارات API الرئيسية والثانوية في gps.h؛ من الأفضل فرض فشل الترجمة إذا كان GPSD_API_MAJOR_VERSION ليس إصدارًا تتعرف عليه. راجع موقع مشروع GPSD لمزيد من المعلومات حول البروتوكول وتغييرات API.

جميع الدوال الموصوفة هنا تستخدم بنية gps_data_t. استشر gps.h لمعرفة المزيد حول gps_data_t، أعضاء بياناتها، البنى المرتبطة بها، الطوابع الزمنية المرتبطة بها. لاحظ أن المعلومات ستتراكم في بنية الجلسة بمرور الوقت، وأن الحقل 'valid' لا يتم تعيينه تلقائيًا إلى الصفر بواسطة كل gps_read(). يعود الأمر إلى العميل لتصفير هذا الحقل عند الاقتضاء ومراقبة طوابع زمنية للإصلاحات والجمل.


تحذير

تقوم gps_data_t بتعيين متغيرات النقطة العائمة إلى NaN عندما تكون القيمة الفعلية للمتغير غير معروفة. تحقق من جميع الأعداد العشرية والدبل باستخدام isfinite() قبل استخدامها. isnan() غير كافية!

gps_open()

تستدعي gps_open() لتهيئة بنية gps_data_t لاحتواء البيانات التي تم جمعها بواسطة GPS، وتقوم بإعداد الوصول إلى gpsd(8) عبر المقبس أو التصدير المشترك للذاكرة. التصدير المشترك للذاكرة أسرع، ولكنه لا يحمل معلومات حول أحداث تفعيل وإلغاء تفعيل الجهاز ولن يسمح لك بمراقبة حركة مرور حزم الجهاز.

ترجع gps_open() 0 عند النجاح، و -1 عند حدوث أخطاء وهي قابلة لإعادة الدخول. يتم تعيين errno اعتمادًا على الخطأ الذي تم إرجاعه من واجهة المقبس أو الذاكرة المشتركة؛ راجع gps.h للقيم والشروحات؛ راجع أيضًا gps_errstr(). قد يكون عنوان المضيف اسم DNS، أو رباعي نقاط IPv4، أو عنوان IPv6، أو القيمة الخاصة GPSD_SHARED_MEMORY التي تشير إلى تصدير الذاكرة المشتركة؛ ستفعل المكتبة الشيء الصحيح لأي من هذه.

gps_close()

تنهي gps_close() الجلسة ويجب استدعاؤها فقط بعد استدعاء ناجح لـ gps_open(). ترجع 0 عند النجاح، و -1 عند حدوث أخطاء. يرجع إغلاق واجهة الذاكرة المشتركة دائمًا 0، بينما يمكن أن يؤدي إغلاق المقبس إلى حدوث خطأ. في حالة حدوث خطأ في إغلاق المقبس، سيتم تعيين errno من استدعاء نظام close().

gps_send()

تكتب gps_send() أمرًا إلى برنامج gpsd. لا تفعل شيئًا عند استخدام تصدير الذاكرة المشتركة. يجب أن يكون الوسيط الثاني سلسلة تنسيق تحتوي على عناصر من مجموعة الأوامر الموثقة في gpsd(8). قد تحتوي على عناصر % كما هو الحال في sprintf(3)، والتي سيتم ملؤها من أي وسائط تالية. ترجع هذه الدالة -1 إذا كان هناك خطأ في كتابة على مستوى Unix، وإلا 0. يرجى قراءة قسم LIMITATIONS للحصول على معلومات إضافية وتحذيرات. راجع gps_stream() كبديل محتمل.

gps_read()

تقبل gps_read() استجابة، أو سلسلة من الاستجابات، من برنامج gpsd وتقوم بفك تشفير الاستجابة إلى gps_data_t. افتراضيًا، تقوم هذه الدالة بقراءة متوقفة لبيانات من برنامج gpsd أو جلب من الذاكرة المشتركة؛ ترجع عدد البايتات المقروءة للنجاح، و -1 مع تعيين errno عند خطأ قراءة على مستوى Unix، و -1 مع عدم تعيين errno إذا تم إغلاق المقبس إلى برنامج gpsd أو إذا كان مقطع الذاكرة المشتركة غير متاح، و 0 إذا لم تكن هناك بيانات متاحة.

عادة ما يكون الوسيط الثاني لـ gps_read() هو NULL، والوسيط الثالث هو صفر. إذا أراد تطبيقك رؤية البيانات الخام من برنامج gpsd، فقم بتعيين الوسيط الثاني إلى عنوان مخزن الرسائل الخاص بك، والوسيط الثالث هو حجم مخزنك. استخدم بحذر؛ قد لا تكون هذه سلسلة منتهية بـ NUL إذا تم تمكين WATCH_RAW.

gps_waiting()

يمكن استخدام gps_waiting() للتحقق مما إذا كانت هناك بيانات جديدة من برنامج gpsd. الوسيط الثاني هو الحد الأقصى للوقت للانتظار (بالميكروثانية) قبل العودة. ترجع صحيحة إذا كانت هناك مدخلات في الانتظار، خاطئة عند انتهاء الوقت (لا توجد بيانات في الانتظار) أو حالة خطأ. عند استخدام تصدير المقبس، تعد هذه الدالة غلافًا مناسبًا لاستدعاء select(2)، وتقوم بتصفير errno عند الدخول؛ يمكنك اختبار errno بعد الخروج للحصول على مزيد من المعلومات حول حالات الخطأ.

تحذير: تحت واجهة الذاكرة المشتركة، توجد نافذة سباق صغيرة بين gps_waiting() واستدعاء gps_read() التالي؛ في هذا السياق، نظرًا لأن الأخير لا ينتظر، فمن الأفضل كتابة حلقة قراءة بسيطة.

gps_mainloop()

تسمح gps_mainloop() باستدعاء دالة الخطاف المقدمة باستمرار كلما كانت هناك بيانات gpsd. الوسيط الثاني هو الحد الأقصى للوقت للانتظار (بالميكروثانية) قبل الخروج من الحلقة (وإرجاع القيمة -1). سترجع أيضًا قيمة سالبة عند حدوث أخطاء مختلفة.

gps_unpack()

تقوم gps_unpack() بتحليل JSON من مخزن الوسيط إلى الهدف من وسيط مؤشر بنية الجلسة. مضمنة في حالة رغبة تطبيقك في إدارة I/O المقبس بنفسك.

gps_data()

ترجع gps_data() محتويات مخزن بيانات العميل (ترجع NULL عند استخدام تصدير الذاكرة المشتركة). استخدم بحذر؛ قد لا تكون هذه سلسلة منتهية بـ NUL إذا تم تمكين WATCH_RAW.

gps_stream()

تطلب gps_stream() من gpsd بث التقارير التي لديه إليك، لتكون متاحة عند الاستطلاع (غير متاحة عند استخدام تصدير الذاكرة المشتركة). الوسيط الثاني هو قناع علامة يحدد بتات سياسة مختلفة؛ راجع القائمة أدناه. يُسمح باستدعاء gps_stream() أكثر من مرة بأقنعة علامات مختلفة.

WATCH_DEVICE

تقييد المشاهدة إلى جهاز محدد. يتم تقديم مسار الجهاز كسلسلة للوسيط الثالث (data).

WATCH_DISABLE

تعطيل أوضاع الإبلاغ المحددة بواسطة علامات WATCH_ الأخرى.

WATCH_ENABLE

تمكين أوضاع الإبلاغ المحددة بواسطة علامات WATCH_ الأخرى. هذا هو الافتراضي.

WATCH_JSON

تمكين الإبلاغ بتنسيق JSON عن البيانات. إذا تم تعيين WATCH_ENABLE، ولم يتم تعيين أي علامات WATCH أخرى، فهذا هو الافتراضي.

WATCH_NEWSTYLE

فرض إصدار تهيئة JSON والحصول على استجابات جديدة النمط. هذا هو الافتراضي.

WATCH_NMEA

تمكين الإبلاغ الزائف NMEA الذي تم إنشاؤه على الأجهزة الثنائية.

WATCH_OLDSTYLE

فرض إصدار أمر W أو R والحصول على استجابات قديمة النمط. تحذير: ستتم إزالة هذه العلامة (والقدرة) في إصدار مستقبلي.

WATCH_RARE

تمكين الإبلاغ عن الحزم الثنائية بتنسيق سداسي عشري مشفر.

WATCH_RAW

تمكين المرور الحرفي للحزم الثنائية.

WATCH_SCALED

عند الإبلاغ عن بيانات AIS أو Subframe، قم بقياس الكميات الصحيحة إلى أعداد عشرية إذا كانت لها مقسوم أو صيغة عرض مرتبطة بها.

gps_errstr()

ترجع gps_errstr() سلسلة ASCII (باللغة الإنجليزية) تصف الخطأ الذي يشير إليه قيمة غير صفرية من gps_op
