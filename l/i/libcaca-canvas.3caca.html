الاسم

libcaca-canvas - صيغة لوحة الرسم libcaca (الإصدار 1)

جميع الأنواع هي big endian.

struct
{
magic:
   uint8_t caca_header[2];    // "CA"
   uint8_t caca_file_type[2]; // "CV"
canvas_header:
   uint32_t control_size;     // حجم التحكم (canvas_data - canvas_header)
   uint32_t data_size;        // حجم البيانات (EOF - canvas_data)
   uint16_t version;          // إصدار صيغة لوحة الرسم
                              //  بت 0: تم التعيين إلى 1 إذا كانت لوحة الرسم متوافقة
                              //         مع الإصدار 1 من الصيغة
                              //  بت 1-15: غير مستخدمة بعد، يجب أن تكون 0
   uint32_t frames;           // عدد الإطارات
   uint16_t flags;            // أعلام الميزات
                              //  بت 0-15: غير مستخدمة بعد، يجب أن تكون 0
frame_info:
   struct
   {
      uint32_t width;         // عرض الإطار
      uint32_t height;        // ارتفاع الإطار
      uint32_t duration;      // مدة الإطار بالمللي ثانية، 0 لـ
                              // عدم تحديد مدة
      uint32_t attr;          // سمة سياق الرسومات
      int32_t cursor_x;       // إحداثي X للمؤشر
      int32_t cursor_y;       // إحداثي Y للمؤشر
      int32_t handle_x;       // إحداثي X للمقبض
      int32_t handle_y;       // إحداثي Y للمقبض
   }
   frame_list[frames];
control_extension_1:
control_extension_2:
   ...
control_extension_N:
   ...                        // محجوز للاستخدام المستقبلي
canvas_data:
   uint8_t data[data_size];   // بيانات لوحة الرسم
data_extension_1:
data_extension_2:
   ...
data_extension_N:
   ...                        // محجوز للاستخدام المستقبلي
};
