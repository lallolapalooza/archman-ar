libkeccak_shakesum_fd(3)

الحزمة: libkeccak

التراخيص: custom:ISC

LIBKECCAK_SHAKESUM_FD(3) دليل وظائف المكتبة LIBKECCAK_SHAKESUM_FD(3)

الاسم

libkeccak_shakesum_fd - حساب مجموع SHAKE لملف

ملخص

#include <libkeccak.h>
int libkeccak_shakesum_fd(int fd, struct libkeccak_state *state, long int semicapacity, long int output, void *hashsum);

ربط مع -lkeccak.

الوصف

تقوم الدالة libkeccak_shakesum_fd() بحساب مجموع SHAKE لملف، يتم تحديد واصف الملف بواسطة fd (ويجب أن يكون في بداية الملف). يتم ضبط خوارزمية التجزئة بواسطة المعاملات semicapacity و output؛ فهما يحددان نصف السعة ومخرج الإخراج، على التوالي، بالبت.

يتم تخزين المجموع في شكل ثنائي في hashsum. يجب أن يكون حجم تخصيص hashsum على الأقل (((output + 7) / 8) * sizeof(char)).

*state يجب ألا تكون مهيأة. تقوم libkeccak_shakesum_fd() بتهيأة *state بنفسها. لذلك سيكون هناك تسرب في الذاكرة إذا كان *state مهيأ بالفعل.

قيم الإرجاع

ترجع الدالة libkeccak_shakesum_fd() القيمة 0 عند اكتمال العملية بنجاح. عند حدوث خطأ، يتم إرجاع -1 وتعيين errno لوصف الخطأ.

الأخطاء

قد تفشل الدالة libkeccak_shakesum_fd() لأي سبب، باستثناء تلك التي تؤدي إلى تعيين errno إلى EINTR، والمحددة للدوال read(2)، malloc(3)، و realloc(3).

ملاحظات

كن على علم، تقوم libkeccak_shakesum_fd() بتجزئة الملف حتى نهاية الملف. بالنسبة للأنابيب والمقابس، هذا يعني حتى يتم إغلاق الملف. ولكن بالنسبة لأجهزة الأحرف، هذا يعني عادةً عدم الانتهاء أبدًا. لذلك، فإن محاولة تجزئة الملفات في /dev هي فكرة سيئة عادةً. لا تتحقق libkeccak_shakesum_fd() من طول الملف أو نوعه قبل التجزئة لأن هذا يمكن أن يحد مما يمكنك القيام به، ويجعل المكتبة أكثر تعقيدًا.

libkeccak_shakesum_fd() لا تتوقف عند المقاطعة (read(2) يرجع EINTR.)

تفترض libkeccak_shakesum_fd() أن جميع المعلومات غير حساسة، ولن تقوم بالتالي بأي مسح آمن للمعلومات.

libkeccak_shakesum_fd() لا تتحقق من صحة ضبط الخوارزمية.

مثال

يقوم هذا المثال بحساب مجموع SHAKE256(, 512) من الإدخال من stdin، ويطبع المجموع، في شكل سداسي عشري، إلى stdout.

struct libkeccak_state state;
if (libkeccak_shakesum_fd(STDIN_FILENO, &state, 256, 512, binhash) < 0)
	goto fail;
libkeccak_behex_lower(hexhash, binhash, sizeof(binhash));
printf("%s\n", hexhash);
libkeccak_state_destroy(&state);
