libkeccak_update(3)

الحزمة: libkeccak

التراخيص: مخصص:ISC

LIBKECCAK_UPDATE(3) دليل دوال المكتبة LIBKECCAK_UPDATE(3)

الاسم

libkeccak_update - تجزئة جزئية لرسالة مع الاستعادة

الاختصار

#include <libkeccak.h>
int libkeccak_update(struct libkeccak_state *state, const void *msg, size_t msglen);

الربط مع -lkeccak.

الوصف

الدالة libkeccak_update() تستمر (أو تبدأ) في تجزئة رسالة. يتم تخزين الحالة الحالية للتجزئة في *state، وسيتم تحديثها. سيتم تجزئة الرسالة المحددة بواسطة المعامل msg بحجم البايت المحدد بواسطة المعامل msglen.

قد تعيد الدالة libkeccak_update() تخصيص مخزن مؤقت لقطع الرسالة الخاص بالحالة. عند القيام بذلك، تحاول أن تفعل ذلك بأكبر قدر ممكن من الأمان، بدلاً من أكبر قدر ممكن من السرعة.

قيم الإرجاع

تُرجع الدالة libkeccak_update() القيمة 0 عند اكتمالها بنجاح. عند حدوث خطأ، يتم إرجاع -1 ويتم تعيين errno لوصف الخطأ.

الأخطاء

قد تفشل الدالة libkeccak_update() لأي سبب تحدده الدالة malloc(3).

ملاحظات

لا يجوز أن يكون أي من المعاملين NULL أو 0.

مثال

يحسب هذا المثال تجزئة Keccak[b = 1024, c = 576, n = 256] للمدخلات من stdin، ويطبع التجزئة، بصيغة سداسية عشرية، إلى stdout.

struct libkeccak_state state;
struct libkeccak_spec spec;
char binhash[256 / 8];
char hexhash[256 / 8 * 2 + 1];
char chunk[4 << 10];
ssize_t len;
spec.bitrate = 1024;
spec.capacity = 576;
spec.output = 256;
if (libkeccak_state_initialise(&state, &spec) < 0)
	goto fail;
for (;;) {
	len = read(STDIN_FILENO, chunk, sizeof(chunk));
	if ((len < 0) && (errno == EINTR))
		continue;
	if (len < 0)
		goto fail;
	if (len == 0)
		break;
	if (libkeccak_update(&state, chunk, (size_t)len) < 0)
		goto fail;
}
if (libkeccak_digest(&state, NULL, 0, 0, "", binhash) < 0)
	goto fail;
libkeccak_behex_lower(hexhash, binhash, sizeof(binhash));
printf("%s\n", hexhash);
libkeccak_state_destroy(&state);
