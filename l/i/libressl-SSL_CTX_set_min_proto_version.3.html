libressl-SSL_CTX_set_min_proto_version(3)

الحزمة: libressl

التراخيص: ISC, custom:OpenSSL

SSL_CTX_SET_MIN_PROTO_VERSION(3) دليل المكتبة SSL_CTX_SET_MIN_PROTO_VERSION(3)

الاسم

SSL_CTX_set_min_proto_version، SSL_CTX_set_max_proto_version، SSL_CTX_get_min_proto_version، SSL_CTX_get_max_proto_version، SSL_set_min_proto_version، SSL_set_max_proto_version، SSL_get_min_proto_version، SSL_get_max_proto_version — الحصول على وتعيين الحد الأدنى والأقصى لإصدار البروتوكول المدعوم

الملخص

#include <openssl/ssl.h>

الوصف

تسترجع هذه الدوال أو تعين الحد الأدنى والأقصى لإصدارات البروتوكول المدعومة لـ ctx أو ssl. يعمل هذا بالاقتران مع الخيارات التي تم تعيينها عبر SSL_CTX_set_options(3) والتي تجعل من الممكن أيضًا تعطيل إصدارات بروتوكول معينة. استخدم هذه الدوال بدلاً من تعطيل إصدارات البروتوكول المحددة.

سيؤدي تعيين الحد الأدنى أو الأقصى للإصدار إلى 0 إلى تمكين إصدارات البروتوكول وصولاً إلى أدنى أو وصولاً إلى أعلى إصدار مدعوم من قبل المكتبة، على التوالي.

الإصدارات المدعومة حاليًا هي TLS1_VERSION، TLS1_1_VERSION، و TLS1_2_VERSION لـ TLS و DTLS1_VERSION و DTLS1_2_VERSION لـ DTLS.

في تطبيقات أخرى، قد يتم تنفيذ هذه الدوال كـ وحدات ماكرو.

قيم الإرجاع

تُرجع دوال الضبط القيمة 1 عند النجاح أو 0 عند الفشل.

تُرجع دوال الاسترجاع الإصدار المُكوّن أو 0 إذا تم تكوين ctx أو ssl لاستخدام أدنى أو أعلى إصدار مدعوم من قبل المكتبة تلقائيًا.

انظر أيضًا

ssl(3)، SSL_CTX_new(3)، SSL_CTX_set_options(3)

التاريخ

ظهرت دوال الضبط لأول مرة في BoringSSL في ديسمبر 2014، بأسماء أقصر بدون الجزء proto_. بعد عامين، أدرجتها OpenSSL في إصدارها 1.1.0، وغيرت الأسماء بشكل غير ضروري؛ تنهدت جوجل وتبنت الأسماء الأطول بعد شهر واحد. كانت متاحة منذ OpenBSD 6.2.

ظهرت دوال الاسترجاع لأول مرة في OpenSSL 1.1.0g وكانت متاحة منذ OpenBSD 6.3.
