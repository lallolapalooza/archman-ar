Lingua__EN__Inflect(3pm)

الحزمة: perl-lingua-en-inflect

التراخيص: PerlArtistic

Lingua::EN::Inflect(3) وثائق Perl من مساهمة المستخدم Lingua::EN::Inflect(3)

الاسم

Lingua::EN::Inflect - تحويل المفرد إلى جمع. اختيار "a" أو "an".

الإصدار

يوثق هذا المستند الإصدار 1.905 من Lingua::EN::Inflect

الملخص

use Lingua::EN::Inflect qw ( PL PL_N PL_V PL_ADJ NO NUM
                 PL_eq PL_N_eq PL_V_eq PL_ADJ_eq
                 A AN
                 PART_PRES
                 ORD NUMWORDS
                 WORDLIST
                 inflect classical
                 def_noun def_verb def_adj def_a def_an );
# تكوين الجمع بشكل غير مشروط
     print "The plural of ", $word, " is ", PL($word), "\n";
# تكوين الجمع بشكل شرطي
     print "I saw $cat_count ", PL("cat",$cat_count), "\n";
# تكوين صيغ الجمع لأجزاء معينة من الكلام
     print PL_N("I",$N1), PL_V("saw",$N1),
       PL_ADJ("my",$N2), PL_N("saw",$N2), "\n";
# التعامل مع "0/1/N" -> "no/1/N" الترجمة:
     print "There ", PL_V("was",$errors), NO(" error",$errors), "\n";
# استخدام العد الافتراضي:
     print NUM($N1,""), PL("I"), PL_V(" saw"), NUM($N2), PL_N(" saw");
     print "There ", NUM($errors,''), PL_V("was"), NO(" error"), "\n";
# مقارنة كلمتين "غير حساسين للعدد":
     print "same\n"      if PL_eq($word1, $word2);
     print "same noun\n" if PL_N_eq($word1, $word2);
     print "same verb\n" if PL_V_eq($word1, $word2);
     print "same adj.\n" if PL_ADJ_eq($word1, $word2);
# إضافة "a" أو "an" الصحيحة لكلمة معينة:
     print "Did you want ", A($thing), " or ", AN($idea), "\n";
# تحويل الأرقام إلى ترتيبية (مثل 1->1st, 2->2nd, 3->3rd, إلخ.)
     print "It was", ORD($position), " from the left\n";
# تحويل الأرقام إلى كلمات (مثل 1->"one", 101->"one hundred and one", إلخ.)
# في سياق قياسي: احصل على سلسلة واحدة...
   $words = NUMWORDS(1234);      # "one thousand, two hundred and thirty-four"
   $words = NUMWORDS(ORD(1234)); # "one thousand, two hundred and thirty-fourth"
# في سياق قائمة: احصل على قائمة سلاسل، واحدة لكل "جزء"...
   @words = NUMWORDS(1234);    # ("one thousand","two hundred and thirty-four")
# المعلمات الاختيارية تغير الترجمة:
   $words = NUMWORDS(12345, group=>1);
               # "one, two, three, four, five"
   $words = NUMWORDS(12345, group=>2);
               # "twelve, thirty-four, five"
   $words = NUMWORDS(12345, group=>3);
               # "one twenty-three, forty-five"
   $words = NUMWORDS(1234, 'and'=>'');
               # "one thousand, two hundred thirty-four"
   $words = NUMWORDS(1234, 'and'=>', plus');
               # "one thousand, two hundred, plus thirty-four"
   $words = NUMWORDS(555_1202, group=>1, zero=>'oh');
               # "five, five, five, one, two, oh, two"
   $words = NUMWORDS(555_1202, group=>1, one=>'unity');
               # "five, five, five, unity, two, zero, two"
   $words = NUMWORDS(123.456, group=>1, decimal=>'mark');
               # "one two three mark four five six"
# الأرقام الأسلوب الحرفي فقط أقل من عتبة معينة...
   $words = NUMWORDS(   9, threshold=>10);    # "nine"
   $words = NUMWORDS(  10, threshold=>10);    # "ten"
   $words = NUMWORDS(  11, threshold=>10);    # "11"
   $words = NUMWORDS(1000, threshold=>10);    # "1,000"
# ضم الكلمات في قائمة:
   $list = WORDLIST("apple", "banana", "carrot");
               # "apple, banana, and carrot"
   $list = WORDLIST("apple", "banana");
               # "apple and banana"
   $list = WORDLIST("apple", "banana", "carrot", {final_sep=>""});
               # "apple, banana and carrot"
# طلب صيغ الجمع "الكلاسيكية" (مثل: "focus" -> "foci", "cherub" -> "cherubim")
     classical;          # استخدام كل صيغ الجمع الكلاسيكية
     classical 1;           #  استخدام كل صيغ الجمع الكلاسيكية
     classical 0;           #  استخدام كل صيغ الجمع الحديثة (افتراضي)
     classical 'zero';      #  "no error" بدلاً من "no errors"
     classical zero=>1;     #  "no error" بدلاً من "no errors"
     classical zero=>0;     #  "no errors" بدلاً من "no error"
     classical 'herd';      #  "2 buffalo" بدلاً من "2 buffalos"
     classical herd=>1;     #  "2 buffalo" بدلاً من "2 buffalos"
     classical herd=>0;     #  "2 buffalos" بدلاً من "2 buffalo"
     classical 'persons';   # "2 chairpersons" بدلاً من "2 chairpeople"
     classical persons=>1;  # "2 chairpersons" بدلاً من "2 chairpeople"
     classical persons=>0;  # "2 chairpeople" بدلاً من "2 chairpersons"
     classical 'ancient';   # "2 formulae" بدلاً من "2 formulas"
     classical ancient=>1;  # "2 formulae" بدلاً من "2 formulas"
     classical ancient=>0;  # "2 formulas" بدلاً من "2 formulae"
# استيفاء "PL()", "PL_N()", "PL_V()", "PL_ADJ()", A()", "AN()"
# "NUM()" و "ORD()" داخل السلاسل:
     print inflect("The plural of $word is PL($word)\n");
     print inflect("I saw $cat_count PL(cat,$cat_count)\n");
     print inflect("PL(I,$N1) PL_V(saw,$N1) PL(a,$N2) PL_N(saw,$N2)\n");
     print inflect("NUM($N1,)PL(I) PL_V(saw) NUM($N2,)PL(a) PL_N(saw)\n");
     print inflect("I saw NUM($cat_count) PL(cat)\n");
     print inflect("There PL_V(was,$errors) NO(error,$errors)\n");
     print inflect("There NUM($errors,)PL_V(was) NO(error)\n");
     print inflect("Did you want A($thing) or AN($idea)\n");
     print inflect("It was ORD($position) from the left\n");
# إضافة تصريفات معرفة من قبل المستخدم (تجاوز القواعد المدمجة):
     def_noun  "VAX"  => "VAXen";  # مفرد => جمع
     def_verb  "will" => "shall",  # ضمير المتكلم المفرد => الجمع
               "will" => "will",   # ضمير المخاطب المفرد => الجمع
               "will" => "will";   # ضمير الغائب المفرد => الجمع
     def_adj   "hir"  => "their";  # مفرد => جمع
     def_a     "h";                # "AY HALWAYS SEZ 'HAITCH'!"
     def_an    "horrendous.*";     # "AN HORRENDOUS AFFECTATION"

الوصف

[ملاحظة: هذه الوحدة في وضع الصيانة الصارم الآن. ألقِ نظرة على وحدة Lingua::EN::Inflexion الأحدث، والتي توفر واجهة أنظف وأكثر ملاءمة، ولديها العديد من الميزات الأخرى (بما في ذلك التحويل من الجمع إلى المفرد)، وهي أيضاً أفضل اختبارًا بكثير. إذا كان لديك كود موجود يعتمد على Lingua::EN::Inflect، فراجع القسم من الوثائق بعنوان "تحويل من LINGUA::EN::INFLECT". ]

توفر الإجراءات القابلة للتصدير من Lingua::EN::Inflect صيغ الجمع، اختيار "a"/"an" للكلمات الإنجليزية، و معالجة الأرقام ككلمات.

تتوفر صيغ الجمع لجميع الأسماء، ومعظم الأفعال، وبعض الصفات. حيثما كان ذلك مناسباً، تتوفر أيضاً متغيرات "كلاسيكية" (على سبيل المثال: "brother" -> "brethren"، "dogma" -> "dogmata"، إلخ.).

يتم توفير اختيار "a"/"an" المستند إلى النطق لجميع الكلمات الإنجليزية، ومعظم الاختصارات.

من الممكن أيضاً تحويل الأرقام (1،2،3) إلى أعداد ترتيبية (1st، 2nd، 3rd) وإلى كلمات إنجليزية ("one"، "two"، "three).

عند إنشاء هذه التصريفات، تتبع Lingua::EN::Inflect قاموس أكسفورد الإنجليزي وإرشادات Fowler's Modern English Usage، مفضلةً الأول حيث يختلف الاثنان.

الوحدة مبنية على الإملاء البريطاني القياسي، ولكنها مصممة للتعامل مع المتغيرات الأمريكية الشائعة أيضاً. لا يتم التعامل صراحةً مع العامية أو المصطلحات المتخصصة أو اللهجات الإنجليزية الأخرى.

حيثما توجد صيغ تصريف اثنتان أو أكثر لكلمة واحدة (عادةً صيغة "كلاسيكية" وصيغة "حديثة")، تفضل Lingua::EN::Inflect الصيغة الأكثر شيوعاً (عادةً الصيغة "الحديثة")، ما لم يتم تحديد معالجة "كلاسيكية" (انظر "الصيغ الحديثة مقابل الكلاسيكية").

تصريف الجمع

جميع روتينات التصريف الفرعية لـ "PL_..." تأخذ الكلمة المراد تصريفها كوسيط أول وتعيد التصريف المقابل. لاحظ أن جميع هذه الروتينات الفرعية تتوقع شكل المفرد للكلمة. نتائج تمرير شكل الجمع غير محددة (ومن غير المرجح أن تكون صحيحة).

تأخذ الروتينات الفرعية "PL_..." أيضًا وسيطًا ثانيًا اختياريًا، والذي يشير إلى "عدد" الكلمة النحوي (أو كلمة أخرى يجب أن تتفق معها الكلمة المصرفة). إذا تم توفير وسيط "العدد" ولم يكن 1 (أو "one" أو "a"، أو أي صفة أخرى تدل على المفرد)، يتم إرجاع شكل الجمع للكلمة. إذا كان وسيط "العدد" يدل على المفرد، يتم إرجاع الكلمة نفسها (غير المصرفة). إذا تم حذف وسيط العدد، يتم إرجاع شكل الجمع بشكل غير مشروط.

الروتينات الفرعية المختلفة هي:

PL_N($;$)
الروتين الفرعي القابل للتصدير PL_N() يأخذ اسمًا أو ضميرًا إنجليزيًا مفردًا ويعيد جمعه. يتم التعامل مع الضمائر في حالة الرفع ("I" -> "we") وحالة النصب ("me" -> "us")، وكذلك الضمائر الملكية ("mine" -> "ours").
PL_V($;$)
الروتين الفرعي القابل للتصدير PL_V() يأخذ شكل المفرد لفعل مصرف (أي، فعل في "شخص" و "صيغة" صحيحة بالفعل) ويعيد تصريف الجمع المقابل.
PL_ADJ($;$)
الروتين الفرعي القابل للتصدير PL_ADJ() يأخذ شكل المفرد لأنواع معينة من الصفات ويعيد شكل الجمع المقابل. الصفات التي يتم التعامل معها بشكل صحيح تشمل: الصفات "الرقمية" ("a" -> "some")، الصفات الإشارية ("this" -> "these"، "that" -> "those")، والصفات الملكية ("my" -> "our"، "cat's" -> "cats'"، "child's" -> "childrens'"، إلخ).
PL($;$)
الروتين الفرعي القابل للتصدير PL() يأخذ اسمًا أو ضميرًا أو فعلًا أو صفة إنجليزية مفردة ويعيد شكل جمعها. حيث يكون للكلمة أكثر من تصريف واحد حسب نوع الكلمة (على سبيل المثال، الاسم "thought" يصرف إلى "thoughts"، والفعل "thought" إلى "thought")، يتم تفضيل معنى الاسم (المفرد) على معنى الفعل (المفرد).

لذلك، PL("knife") ستعيد "knives" (معاملة "knife" كاسم مفرد)، بينما PL("knifes") ستعيد "knife" (معاملة "knifes" كفعل للشخص الثالث المفرد).

الغموض المتأصل في مثل هذه الحالات يشير إلى أنه، عند معرفة نوع الكلمة، يجب استخدام "PL_N" أو "PL_V" أو "PL_ADJ" مفضلة على "PL".

لاحظ أن جميع هذه الروتينات الفرعية تتجاهل أي مسافات بيضاء محيطة بالكلمة التي يتم تصريفها، ولكنها تحتفظ بهذه المسافات البيضاء عند إرجاع النتيجة. على سبيل المثال، "PL(" cat  ")" تعيد " cats  ".

تصاريف الجمع الرقمية

الروتينات الفرعية "PL_..." تعيد فقط الكلمة المصرفة، وليس العدد الذي تم استخدامه لتصريفها. وبالتالي، لإنتاج "I saw 3 ducks"، من الضروري استخدام:

print "I saw $N ", PL_N($animal,$N), "\n";

نظرًا لأن الغرض المعتاد من إنتاج الجمع هو جعله يتفق مع عدد سابق، فإن Lingua::EN::Inflect يوفر روتينًا فرعيًا قابلًا للتصدير (NO($;$)) والذي، عند إعطائه كلمة وعدد (اختياري)، يعيد العدد متبوعًا بالكلمة المصرفة بشكل صحيح. وبالتالي يمكن إعادة كتابة المثال السابق:

print "I saw ", NO($animal,$N), "\n";

بالإضافة إلى ذلك، إذا كان العدد صفرًا (أو أي مصطلح آخر يدل على الصفر، مثل "zero" أو "nil"، إلخ)، يتم استبدال العدد بالكلمة "no". وبالتالي، إذا كانت قيمة $N صفرًا، فإن المثال السابق سيطبع ما هو أكثر أناقة:

I saw no animals

بدلاً من:

I saw 0 animals

لاحظ أن اسم الروتين الفرعي عبارة عن تلاعب بالألفاظ: الروتين الفرعي يعيد إما رقمًا (No.) أو "no"، قبل الكلمة المصرفة.

تصاريف الجمع بكلمات و بفواصل

الروتين الفرعي NO() يأخذ وسيطًا ثالثًا اختياريًا: تجزئة (hash) من الخيارات المسماة التي تهيئ سلوكه.

الخيار 'words_below' يخبر NO() ما هي الأعداد الأخرى (أي بخلاف الصفر) التي يجب تحويلها إلى كلمات. على سبيل المثال:

for my $count (0..12) {
    print NO('cat', $count, {words_below => 10}), "\n";
}

سيؤدي ذلك إلى طباعة:

no cats
one cat
two cats
three cats
four cats
five cats
six cats
seven cats
eight cats
nine cats
10 cats
11 cats
12 cats

الخياران 'comma' و 'comma_every' يحددان ما إذا كانت الأعداد التي ينتجها NO() تحتوي على فواصل أم لا. أي:

2001 space odysseys

مقابل:

2,001 space odysseys

عادةً، يتم إنتاج الأعداد بدون فواصل، ولكن إذا تم تحديد 'comma' أو 'comma_every'، فسيتم إضافة الفواصل كما هو مطلوب.

الخيار 'comma' يحدد الحرف الذي سيتم استخدامه كفاصلة. افتراضيًا هو ','، ولكن يمكن تعيينه إلى أي شيء مناسب:

print NO('Euro', $amount, {comma=>'.'});
# يطبع:  1.000.000 Euros

الخيار 'comma_every' يحدد عدد الأحرف بين الفواصل. افتراضيًا هو 3، ولكن يمكن تعيينه إلى أي عدد صحيح موجب:

print NO('Euro', $amount, {comma_every=>4});
# يطبع:  100,0000 Euros

لاحظ أنه يمكنك تعيين كلا الخيارين في نفس الوقت، إذا أردت:

print NO('Euro', $amount, {comma_every=>2, comma=>'_'});
# يطبع:  1_00_00_00 Euros

تقليل عدد الأعداد المطلوبة

في بعض السياقات، تجعل الحاجة إلى توفير عدد صريح للروتينات الفرعية المختلفة "PL_..." تكرارًا مملًا. على سبيل المثال:

print PL_ADJ("This",$errors), PL_N(" error",$errors),
      PL_V(" was",$errors), " fatal.\n";

لذلك، يوفر Lingua::EN::Inflect روتينًا فرعيًا قابلًا للتصدير (NUM($;$)) يمكن استخدامه لتعيين "عدد افتراضي" دائم. إذا تم تعيين قيمة كهذه، فسيتم استخدامها لاحقًا كلما تم حذف وسيط "العدد" الاختياري الثاني. يمكن بعد ذلك إزالة القيمة الافتراضية المعينة عن طريق استدعاء NUM() بدون وسائط. وبالتالي يمكننا إعادة كتابة المثال السابق:

NUM($errors);
print PL_ADJ("This"), PL_N(" error"), PL_V(" was"), "fatal.\n";
NUM();

عادةً، تعيد NUM() وسيطها الأول، بحيث يمكن أيضًا "تضمينها" في سياقات مثل:

print NUM($errors), PL_N(" error"), PL_V(" was"), " detected.\n"
print PL_ADJ("This"), PL_N(" error"), PL_V(" was"), "fatal.\n"
    if $severity > 1;

ومع ذلك، في سياقات معينة (انظر "INTERPOLATING INFLECTIONS IN STRINGS") يُفضل أن تعيد NUM() سلسلة فارغة. لذلك، توفر NUM() وسيطًا ثانيًا اختياريًا. إذا تم توفير هذا الوسيط (أي، إذا تم تعريفه) وتقييمه إلى false، فإن "NUM" يعيد سلسلة فارغة بدلاً من وسيطه الأول. على سبيل المثال:

print NUM($errors,0), NO("error"), PL_V(" was"), " detected.\n";
print PL_ADJ("This"), PL_N(" error"), PL_V(" was"), "fatal.\n"
    if $severity > 1;

المساواة غير الحساسة للعدد

يوفر Lingua::EN::Inflect أيضًا حلاً لمشكلة مقارنة الكلمات ذات الجمع المختلف من خلال الروتينات الفرعية القابلة للتصدير PL_eq($$)، و PL_N_eq($$)، و PL_V_eq($$)، و PL_ADJ_eq($$). كل من هذه الروتينات الفرعية تأخذ سلسلتين، وتقارنهما باستخدام روتين التصريف الفرعي المقابل للجمع (PL()، PL_N()، PL_V()، و PL_ADJ() على التوالي).

تعيد المقارنة true إذا:

  • كانت السلاسل متساوية "eq"، أو
  • كانت إحدى السلاسل متساوية "eq" لشكل جمع للسلسلة الأخرى، أو
  • كانت السلاسل شكلين جمع مختلفين لنفس الكلمة.

لذلك، كلها تعيد true:

PL_eq("index","index")      # تعيد "eq"
PL_eq("index","indexes")    # تعيد "s:p"
PL_eq("index","indices")    # تعيد "s:p"
PL_eq("indexes","index")    # تعيد "p:s"
PL_eq("indices","index")    # تعيد "p:s"
PL_eq("indices","indexes")  # تعيد "p:p"
PL_eq("indexes","indices")  # تعيد "p:p"
PL_eq("indices","indices")  # تعيد "eq"

كما هو موضح بالتعليقات في المثال السابق، فإن القيمة الفعلية التي تعيدها روتينات "PL_eq" المختلفة تشفر أي من قواعد المساواة الثلاث نجحت: يتم إرجاع "eq" إذا كان

الأرقام ككلمات

NUMWORDS(32) returns "thirty-two". It requires a scalar context; otherwise, it will return a list of the constituent words:

print NUMWORDS(1234567);

will (misprint) print:

one milliontwo hundred and thirty-four thousandfive hundred and sixty-seven

To get readable output, make sure the call in in scalar context:

print scalar NUMWORDS(1234567);

Non-digits (apart from an optional leading plus or minus sign, any decimal points, and ordinal suffixes -- see below) are silently ignored, so the following all produce identical results:

NUMWORDS(5551202);
NUMWORDS(5_551_202);
NUMWORDS("5,551,202");
NUMWORDS("555-1202");

That last case is a little awkward since it's almost certainly a phone number, and "five million, five hundred and fifty-one thousand, two hundred and two" probably isn't what's wanted.

To overcome this, NUMWORDS() takes an optional named argument, 'group', which changes how numbers are translated. The argument must be a positive integer less than four, which indicated how the digits of the number are to be grouped. If the argument is 1, then each digit is translated separately. If the argument is 2, pairs of digits (starting from the left) are grouped together. If the argument is 3, triples of numbers (again, from the left) are grouped. Hence:

NUMWORDS("555-1202", group=>1)

returns "five, five, five, one, two, zero, two", whilst:

NUMWORDS("555-1202", group=>2)

returns "fifty-five, fifty-one, twenty, two", and:

NUMWORDS("555-1202", group=>3)

returns "five fifty-five, one twenty, two".

Phone numbers are often written in words as "five..five..five..one..two..zero..two", which is also easy to achieve:

join '..', NUMWORDS("555-1202", group=>1)

"NUMWORDS" also handles decimal fractions. Hence:

NUMWORDS("1.2345")

returns "one point two three four five" in a scalar context and "("one","point","two","three","four","five")") in an array context. Exponent form ("1.234e56") is not yet handled.

Multiple decimal points are only translated in one of the "grouping" modes. Hence:

NUMWORDS(101.202.303)

returns "one hundred and one point two zero two three zero three", whereas:

NUMWORDS(101.202.303, group=>1)

returns "one zero one point two zero two point three zero three".

The digit '0' is unusual in that in may be translated to English as "zero", "oh", or "nought". To cater for this diversity, "NUMWORDS" may be passed a named argument, 'zero', which may be set to the desired translation of '0'. For example:

print join "..", NUMWORDS("555-1202", group=>3, zero=>'oh')

prints "five..five..five..one..two..oh..two". By default, zero is rendered as "zero".

Likewise, the digit '1' may be rendered as "one" or "a/an" (or very occasionally other variants), depending on the context. So there is a 'one' argument as well:

print NUMWORDS($_, one=>'a solitary', zero=>'no more'),
      PL(" bottle of beer on the wall\n", $_)
           for (3,2,1,0);
# prints:
#     three bottles of beer on the wall
#     two bottles of beer on the wall
#     a solitary bottle of beer on the wall
#     no more bottles of beer on the wall

Care is needed if the word "a/an" is to be used as a 'one' value. Unless the next word is known in advance, it's almost always necessary to use the "A" function as well:

    print A( NUMWORDS(1, one=>'a') . " $_\n")
     for qw(cat aardvark ewe hour);
# prints:
#     a cat
#     an aardvark
#     a ewe
#     an hour

Another major regional variation in number translation is the use of "and" in certain contexts. The named argument 'and' allows the programmer to specify how "and" should be handled. Hence:

print scalar NUMWORDS("765", 'and'=>'')

prints "seven hundred sixty-five", instead of "seven hundred and sixty-five". By default, the "and" is included.

The translation of the decimal point is also subject to variation (with "point", "dot", and "decimal" being the favorites). The named argument 'decimal' allows the programmer to how the decimal point should be rendered. Hence:

print scalar NUMWORDS("666.124.64.101", group=>3, decimal=>'dot')

prints "six sixty-six, dot, one twenty-four, dot, sixty-four, dot, one zero one" By default, the decimal point is rendered as "point".

"NUMWORDS" also handles the ordinal forms of numbers. So:

print scalar NUMWORDS('1st');
print scalar NUMWORDS('3rd');
print scalar NUMWORDS('202nd');
print scalar NUMWORDS('1000000th');

print:

first
third
two hundred and twenty-second
one millionth

Two common idioms in this regard are:

print scalar NUMWORDS(ORD($number));

and:

print scalar ORD(NUMWORDS($number));

These are identical in effect, except when $number contains a decimal:

$number = 99.09;
print scalar NUMWORDS(ORD($number));    # ninety-ninth point zero nine
print scalar ORD(NUMWORDS($number));    # ninety-nine point zero ninth

Use whichever you feel is most appropriate.

تحويل قوائم الكلمات إلى عبارات

When creating a list of words, commas are used between adjacent items, except if the items contain commas, in which case semicolons are used. But if there are less than two items, the commas/semicolons are omitted entirely. The final item also has a conjunction (usually "and" or "or") before it. And although it's technically incorrect (and sometimes misleading), some people prefer to omit the comma before that final conjunction, even when there are more than two items.

That's complicated enough to warrant its own subroutine: WORDLIST(). This subroutine expects a list of words, possibly with one or more hash references containing options. It returns a string that joins the list together in the normal English usage. For example:

print "You chose ", WORDLIST(@selected_items), "\n";
# You chose barley soup, roast beef, and Yorkshire pudding
print "You chose ", WORDLIST(@selected_items, {final_sep=>""}), "\n";
# You chose barley soup, roast beef and Yorkshire pudding
print "Please chose ", WORDLIST(@side_orders, {conj=>"or"}), "\n";
# Please chose salad, vegetables, or ice-cream

The available options are:

Option named    Specifies                Default value
conj            Final conjunction        "and"
sep             Inter-item separator     ","
last_sep        Final separator          value of 'sep' option

تضمين التصريفات في النصوص

By far the commonest use of the inflection subroutines is to produce message strings for various purposes. For example:

print NUM($errors), PL_N(" error"), PL_V(" was"), " detected.\n";
print PL_ADJ("This"), PL_N(" error"), PL_V(" was"), "fatal.\n"
        if $severity > 1;

Unfortunately the need to separate each subroutine call detracts significantly from the readability of the resulting code. To ameliorate this problem, Lingua::EN::Inflect provides an exportable string-interpolating subroutine (inflect($)), which recognizes calls to the various inflection subroutines within a string and interpolates them appropriately.

Using "inflect" the previous example could be rewritten:

print inflect "NUM($errors) PL_N(error) PL_V(was) detected.\n";
print inflect "PL_ADJ(This) PL_N(error) PL_V(was) fatal.\n"
        if $severity > 1;

Note that "inflect" also correctly handles calls to the NUM() subroutine (whether interpolated or antecedent). The inflect() subroutine has a related extra feature, in that it automatically cancels any "default number" value before it returns its interpolated string. This means that calls to NUM() which are embedded in an inflect()-interpolated string do not "escape" and interfere with subsequent inflections.

التصريفات الحديثة مقابل الكلاسيكية

Certain words, mainly of Latin or Ancient Greek origin, can form plurals either using the standard English "-s" suffix, or with their original Latin or Greek inflections. For example:

PL("stigma")            # -> "stigmas" or "stigmata"
PL("torus")             # -> "toruses" or "tori"
PL("index")             # -> "indexes" or "indices"
PL("millennium")        # -> "millenniums" or "millennia"
PL("ganglion")          # -> "ganglions" or "ganglia"
PL("octopus")           # -> "octopuses" or "octopodes"

Lingua::EN::Inflect caters to such words by providing an "alternate state" of inflection known as "classical mode". By default, words are inflected using their contemporary English plurals, but if classical mode is invoked, the more traditional plural forms are returned instead.
