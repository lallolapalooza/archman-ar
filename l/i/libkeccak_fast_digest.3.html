libkeccak_fast_digest(3)

الحزمة: libkeccak

التراخيص: مخصص:ISC

LIBKECCAK_FAST_DIGEST(3) دليل وظائف المكتبة LIBKECCAK_FAST_DIGEST(3)

الاسم

libkeccak_fast_digest - إكمال تجزئة الرسالة بدون مسح

الملخص

#include <libkeccak.h>
int libkeccak_fast_digest(struct libkeccak_state *state, const void *msg, size_t msglen,
                          size_t bits, const char *suffix, void *hashsum);

ربط مع -lkeccak.

الوصف

تقوم وظيفة libkeccak_fast_digest() بامتصاص الجزء الأخير من (أو كل) الرسالة، وتعيد تجزئة الرسالة بأكملها. يتم تحديد الجزء الأخير من الرسالة بواسطة المعامل msg، ويتم تحديد حجمه بالبايت بواسطة المعامل msglen. إذا تمت معالجة الرسالة بأكملها بالفعل عن طريق استدعاء وظيفة libkeccak_update(3)، libkeccak_fast_update(3)، أو libkeccak_zerocopy_update(3) (بنفس المؤشر على state)، يجب تعيين msg و msglen إلى NULL و 0، على التوالي.

إذا لم تتكون الرسالة من عدد صحيح من البايتات، فيجب تحديد عدد البتات، مقسومًا على 8، في الرسالة في المعامل bits. يجب أن يحسب msglen فقط عدد البايتات الكاملة، أي، الجزء الصحيح من عدد البتات في الرسالة مقسومًا على 8.

يجب أن يكون suffix سلسلة نصية منتهية بـ NUL تحتوي على أحرف '1' و '0' من ASCII، تمثل البتات التي يجب إلحاقها بالرسالة. إذا كانت هذه السلسلة فارغة، يمكن استخدام NULL بدلاً من ذلك. يستخدم هذا لاختيار خوارزمية التجزئة. بالنسبة لـ Keccak الخالصة، يتم استخدام NULL أو "". بالنسبة للخوارزميات الأخرى، يتم استخدام الثوابت LIBKECCAK_SHA3_SUFFIX (لـ SHA-3)، LIBKECCAK_RAWSHAKE_SUFFIX (لـ RawSHAKE)، و LIBKECCAK_SHAKE_SUFFIX (لـ SHAKE)، أو نتيجة وظيفة libkeccak_cshake_suffix(3) (لـ cSHAKE).

سيتم تخزين تجزئة الرسالة في hashsum، إلا إذا كان hashsum هو NULL (مما يزيد من أداء الاستدعاء.) سيتم كتابة ما مجموعه ((state->n + 7) / 8) بايت إلى بداية hashsum. لذلك، يحتاج hashsum على الأقل إلى تخصيص بحجم هذا العدد من البايتات.

قد تعيد وظيفة libkeccak_fast_digest() تخصيص مخزن مؤقت لرسائل الحالة. عند القيام بذلك، تحاول القيام بذلك بأسرع ما يمكن، بدلاً من التأكد من إزالة المعلومات في التخصيص القديم بشكل آمن إذا كان هناك حاجة إلى تخصيص جديد.

قيم الإرجاع

تُرجع الوظيفة libkeccak_fast_digest() القيمة 0 عند اكتمالها بنجاح. عند حدوث خطأ، يتم إرجاع -1 ويتم تعيين errno لوصف الخطأ.

الأخطاء

قد تفشل وظيفة libkeccak_fast_digest() لأي سبب تحدده وظيفة realloc(3).

مثال

يقوم هذا المثال بحساب تجزئة Keccak[b = 1024, c = 576, n = 256] للمدخلات من stdin، ويطبع التجزئة، بالشكل السداسي عشري، إلى stdout.

struct libkeccak_state state;
struct libkeccak_spec spec;
char binhash[256 / 8];
char hexhash[256 / 8 * 2 + 1];
char chunk[4 << 10];
ssize_t len;
spec.bitrate = 1024;
spec.capacity = 576;
spec.output = 256;
if (libkeccak_state_initialise(&state, &spec) < 0)
	goto fail;
for (;;) {
	len = read(STDIN_FILENO, chunk, sizeof(chunk));
	if ((len < 0) && (errno == EINTR))
		continue;
	if (len < 0)
		goto fail;
	if (len == 0)
		break;
	if (libkeccak_fast_update(&state, chunk, (size_t)len) < 0)
		goto fail;
}
if (libkeccak_fast_digest(&state, NULL, 0, 0, "", binhash) < 0)
	goto fail;
libkeccak_behex_lower(hexhash, binhash, sizeof(binhash));
printf("%s\n", hexhash);
libkeccak_state_fast_destroy(&state);
