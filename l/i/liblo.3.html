liblo(3)

الحزمة: liblo

التراخيص: LGPL-2.1-or-later

void lo_address_free (lo_address t)

تحرير الذاكرة المستخدمة بواسطة كائن lo_address.

int lo_address_get_ttl (lo_address t)

الحصول على قيمة Time-to-Live لعنوان وجهة معين.

المعلمات

t عنوان OSC.

القيمة المعادة

عدد صحيح يحدد نطاق رسالة UDP المتعددة البث (multicast).

lo_address lo_address_new (const char * host, const char * port)

تعريف وجهة OSC، معطاة عنوان IP ورقم المنفذ. نفس وظيفة lo_address_new_with_proto()، ولكن باستخدام UDP.

المعلمات

host عنوان IP أو رقم، أو NULL للجهاز المحلي.
port رقم منفذ عشري أو اسم خدمة.

يمكن استخدام كائن lo_address كوجهة لرسائل OSC.

ملاحظة: إذا كنت ترغب في تلقي ردود من وجهة هذا العنوان، يجب عليك إما توفير lo_server باستخدام lo_send_from() أو lo_send_message_from() لإرسال رسالتك، وإلا سيتم استخدام آخر lo_server أو lo_server_thread تم إنشاؤه كمقبس الرد. قد يحصل المستقبل البعيد على عنوان الرد عن طريق استدعاء lo_message_get_source() في معالج الرسائل. انظر example_tcp_echo_server.c للحصول على مثال لكيفية إنشاء اتصال ثنائي الاتجاه.

lo_address lo_address_new_from_url (const char * url)

إنشاء كائن lo_address من عنوان URL الخاص بـ OSC. مثال: "osc.udp://localhost:4444/my/path/"

lo_address lo_address_new_with_proto (int proto, const char * host, const char * port)

تعريف وجهة OSC، معطاة عنوان IP ورقم المنفذ، وتحديد البروتوكول.

المعلمات

proto البروتوكول المراد استخدامه، يجب أن يكون أحد LO_UDP، LO_TCP أو LO_UNIX.
host عنوان IP أو رقم، أو NULL للجهاز المحلي.
port رقم منفذ عشري أو اسم خدمة.

يمكن استخدام كائن lo_address كوجهة لرسائل OSC.

ملاحظة: إذا كنت ترغب في تلقي ردود من وجهة هذا العنوان، يجب عليك أولاً إنشاء كائن lo_server_thread أو lo_server لتلقي الردود. آخر كائن lo_server(_thread) تم إنشاؤه سيكون المستقبل.

void lo_address_set_ttl (lo_address t, int ttl)

تعيين قيمة Time-to-Live لعنوان وجهة معين. هذا مطلوب لإرسال رسائل UDP المتعددة البث (multicast). قيمة 1 (الحالة العادية) تبقي الرسالة ضمن الشبكة الفرعية، بينما 255 تعني نطاقًا عالميًا غير مقيد.

المعلمات

t عنوان OSC.
ttl عدد صحيح يحدد نطاق رسالة UDP المتعددة البث (multicast).

void * lo_blob_dataptr (lo_blob b)

إرجاع مؤشر إلى بداية بيانات الـ blob للسماح بتغيير المحتويات. إذا كان الحجم 0، فسيعيد مؤشر NULL.

uint32_t lo_blob_datasize (lo_blob b)

إرجاع مقدار البيانات الصالحة في كائن lo_blob. إذا كنت تريد معرفة حجم التخزين، استخدم lo_arg_size().

void lo_blob_free (lo_blob b)

تحرير الذاكرة التي يشغلها الـ blob.

lo_blob lo_blob_new (int32_t size, const void * data)

إنشاء نوع OSC blob جديد.

المعلمات

size مقدار المساحة المطلوب تخصيصها في هيكل الـ blob.
data البيانات التي سيتم استخدامها لتهيئة الـ blob، يجب أن يكون طولها bytes.

int lo_send (lo_address targ, const char * path, const char * type, ...)

إرسال رسالة بتنسيق OSC إلى العنوان المحدد.

المعلمات

targ عنوان OSC الوجهة
path مسار OSC الذي سيتم تسليم الرسالة إليه
type أنواع عناصر البيانات في الرسالة، الأنواع محددة في lo_osc_types.h
... قيم البيانات المراد إرسالها. يجب أن تتوافق أنواع الوسائط المقدمة هنا مع الأنواع المحددة في معلمة النوع.

مثال:

lo_send(t, "/foo/bar", "ff", 0.1f, 23.0f);

القيمة المعادة

-1 عند الفشل.

int lo_send_from (lo_address targ, lo_server from, lo_timetag ts, const char * path, const char * type, ...)

إرسال رسالة بتنسيق OSC إلى العنوان المحدد، من نفس مقبس الخادم المحدد. إذا كان خادم liblo يستقبل هذه الرسالة، فيمكنه الرد عن طريق الحصول على عنوان الخادم عن طريق استدعاء lo_message_get_source() في معالج الرسائل، مع تمرير lo_message المقدم كوسيط إلى lo_method_handler. بهذه الآلية يمكن إنشاء اتصال ثنائي الاتجاه عن طريق إعداد lo_server أو lo_server_thread على كلا الجانبين.

المعلمات

targ عنوان OSC الوجهة
from الخادم الذي سيتم إرسال الرسالة منه (يمكن أن يكون NULL لاستخدام مقبس جديد)
ts طابع الوقت OSC الذي سيتم معالجة الرسالة فيه (يمكن أن يكون LO_TT_IMMEDIATE إذا كنت لا تريد إرفاق طابع وقت)
path مسار OSC الذي سيتم تسليم الرسالة إليه
type أنواع عناصر البيانات في الرسالة، الأنواع محددة في lo_osc_types.h
... قيم البيانات المراد إرسالها. يجب أن تتوافق أنواع الوسائط المقدمة هنا مع الأنواع المحددة في معلمة النوع.

مثال:

serv = lo_server_new(NULL, err);
lo_server_add_method(serv, "/reply", "ss", reply_handler, NULL);
lo_send_from(t, serv, LO_TT_IMMEDIATE, "/foo/bar", "ff", 0.1f, 23.0f);

انظر example_tcp_echo_server.c للحصول على مثال لكيفية استخدام lo_message_get_source() كما هو موضح.

القيمة المعادة

عند النجاح، عدد البايتات المرسلة، أو -1 عند الفشل.

int lo_send_timestamped (lo_address targ, lo_timetag ts, const char * path, const char * type, ...)

إرسال رسالة بتنسيق OSC إلى العنوان المحدد، مجدولة للمعالجة في وقت لاحق في المستقبل.

المعلمات

targ عنوان OSC الوجهة
ts طابع الوقت OSC الذي سيتم معالجة الرسالة فيه
path مسار OSC الذي سيتم تسليم الرسالة إليه
type أنواع عناصر البيانات في الرسالة، الأنواع محددة في lo_osc_types.h
... قيم البيانات المراد إرسالها. يجب أن تتوافق أنواع الوسائط المقدمة هنا مع الأنواع المحددة في معلمة النوع.

مثال:

lo_timetag now;<br>
lo_timetag_now(&now);<br>
lo_send_timestamped(t, now, "/foo/bar", "ff", 0.1f, 23.0f);

القيمة المعادة

عند النجاح، عدد البايتات المرسلة، أو -1 عند الفشل.

void lo_version (char * verstr, int verstr_size, int * major, int * minor, char * extra, int extra_size, int * lt_major, int * lt_minor, int * lt_bug)

الحصول على معلومات حول إصدار liblo المستخدم حاليًا. جميع المعلمات اختيارية ويمكن إعطاؤها القيمة 0 إذا لم تكن هذه المعلومات مطلوبة. على سبيل المثال، للحصول على الإصدار كسلسلة نصية فقط، استدعِ lo_version(str, size, 0, 0, 0, 0, 0, 0, 0);

تشير حقول "lt"، التي تسمى إصدار ABI، إلى نظام ترقيم libtool للتوافق الثنائي للواجهة، ولا ترتبط برقم إصدار المكتبة. عادة ما يتم ترميز هذه المعلومات في اسم ملف المكتبة المشتركة.

عادةً ما تتوافق السلسلة النصية المعادة في 'verstr' مع $major.$minor$extra، على سبيل المثال، "0.28rc". إذا لم تكن هناك معلومات 'extra'، على سبيل المثال، "0.28"، فستحصل 'extra' على سلسلة فارغة.

المعلمات

verstr مخزن مؤقت لاستقبال سلسلة نصية تصف إصدار المكتبة.
verstr_size حجم المخزن المؤقت المشار إليه بواسطة السلسلة النصية.
major موقع لاستقبال الإصدار الرئيسي للمكتبة.
minor موقع لاستقبال الإصدار الفرعي للمكتبة.
extra موقع لاستقبال السلسلة الإضافية لإصدار المكتبة.
extra_size حجم المخزن المؤقت المشار إليه بواسطة extra.
lt_major موقع لاستقبال الإصدار الرئيسي لـ ABI.
lt_minor موقع لاستقبال الإصدار الفرعي لـ ABI.
lt_bug موقع لاستقبال إصدار 'إصلاح الأخطاء' لـ ABI.

مشار إليه بواسطة lo::version().
