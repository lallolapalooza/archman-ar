الوصف

تنفّذ هذه المكتبة جداول تجزئة ديناميكية مُتحقَّقة النوع. يمكن أن تكون مدخلات جدول التجزئة هياكل اعتباطية. عادةً تتكون من حقول المفتاح والقيمة.

lh_<type>_new() تُنشئ بنية LHASH_OF(<type>) جديدة لتخزين مدخلات بيانات اعتباطية، وتوفّر دالتي الاستدعاء للتجزئة والمقارنة لاستخدامهما في تنظيم مدخلات الجدول. تأخذ دالة الاستدعاء للتجزئة مؤشرًا إلى مدخل جدول كوسيط لها وتُعيد قيمة تجزئة من نوع unsigned long لحقل مفتاحها. عادةً تُقتطع قيمة التجزئة إلى قوة للعدد 2، لذا تأكد من أن دالة التجزئة الخاصة بك تُعيد بتات منخفضة الرتبة ممزوجة جيدًا. تأخذ دالة الاستدعاء للمقارنة وسيطين (مؤشرين إلى مدخلي جدول تجزئة)، وتُعيد 0 إذا كانت مفاتيحهما متساوية، وقيمة غير صفرية خلاف ذلك. إذا كان جدول التجزئة الخاص بك سيحتوي على عناصر من نوع معين وكانت دالتا الاستدعاء للتجزئة والمقارنة تُجزّئان وتقارنان هذه الأنواع، فيمكن استخدام وحدات الماكرو DECLARE_LHASH_HASH_FN() و IMPLEMENT_LHASH_COMP_FN() لإنشاء أغلفة دالة استدعاء للنماذج الأولية المطلوبة بواسطة lh_<type>_new(). توفّر هذه عمليات تحويل لكل متغير قبل استدعاء دوال الاستدعاء الخاصة بالنوع التي كتبها مؤلف التطبيق. تُعرَّف وحدات الماكرو هذه، بالإضافة إلى تلك المستخدمة لدوال الاستدعاء doall، كما يلي؛

#define	DECLARE_LHASH_HASH_FN(name, o_type) \
	unsigned long name##_LHASH_HASH(const void *);
#define	IMPLEMENT_LHASH_HASH_FN(name, o_type) \
	unsigned long name##_LHASH_HASH(const void *arg) { \
		const o_type *a = arg; \
		return name##_hash(a); }
#define	LHASH_HASH_FN(name) name##_LHASH_HASH

#define	DECLARE_LHASH_COMP_FN(name, o_type) \
	int name##_LHASH_COMP(const void *, const void *);
#define	IMPLEMENT_LHASH_COMP_FN(name, o_type) \
	int name##_LHASH_COMP(const void *arg1, const void *arg2) { \
		const o_type *a = arg1;		    \
		const o_type *b = arg2; \
		return name##_cmp(a,b); }
#define	LHASH_COMP_FN(name) name##_LHASH_COMP

#define	DECLARE_LHASH_DOALL_FN(name, o_type) \
	void name##_LHASH_DOALL(void *);
#define	IMPLEMENT_LHASH_DOALL_FN(name, o_type) \
	void name##_LHASH_DOALL(void *arg) { \
		o_type *a = arg; \
		name##_doall(a); }
#define	LHASH_DOALL_FN(name) name##_LHASH_DOALL

#define	DECLARE_LHASH_DOALL_ARG_FN(name, o_type, a_type) \
	void name##_LHASH_DOALL_ARG(void *, void *);
#define	IMPLEMENT_LHASH_DOALL_ARG_FN(name, o_type, a_type) \
	void name##_LHASH_DOALL_ARG(void *arg1, void *arg2) { \
		o_type *a = arg1; \
		a_type *b = arg2; \
		name##_doall_arg(a, b); }
#define	LHASH_DOALL_ARG_FN(name) name##_LHASH_DOALL_ARG

مثال على جدول تجزئة يخزّن (مؤشرات إلى) هياكل من النوع 'STUFF' يمكن تعريفه كما يلي؛

/* احسب قيمة التجزئة لـ 'tohash' (منفّذة في مكان آخر) */
unsigned long STUFF_hash(const STUFF *tohash);
/* رتّب 'arg1' و 'arg2' (منفّذة في مكان آخر) */
int stuff_cmp(const STUFF *arg1, const STUFF *arg2);
/* أنشئ دوال غلاف آمنة النوع للاستخدام في داخل LHASH */
static IMPLEMENT_LHASH_HASH_FN(stuff, STUFF);
static IMPLEMENT_LHASH_COMP_FN(stuff, STUFF);
/* ... */
int main(int argc, char *argv[]) {
        /* أنشئ جدول التجزئة الجديد باستخدام أغلفة التجزئة/المقارنة */
        LHASH_OF(STUFF) *hashtable =
	    lh_STUFF_new(LHASH_HASH_FN(STUFF_hash),
        	LHASH_COMP_FN(STUFF_cmp));
	/* ... */
}

lh_<type>_free() تُحرّر بنية LHASH_OF(<type>) table. لن تُحرَّر مدخلات جدول التجزئة المخصصة؛ فكّر في استخدام lh_<type>_doall() لإلغاء تخصيص أي مدخلات متبقية في جدول التجزئة (انظر أدناه).

lh_<type>_insert() تُدرج البنية التي يشير إليها data في table. إذا كان هناك بالفعل مدخل بنفس المفتاح، فسيُستبدل القيمة القديمة. لاحظ أن lh_<type>_insert() تخزّن المؤشرات، ولا تُنسخ البيانات.

lh_<type>_delete() تحذف مدخلًا من table.

lh_<type>_retrieve() تبحث عن مدخل في table. عادةً، data هي بنية مع تعيين حقل (حقول) المفتاح؛ ستُعيد الدالة مؤشرًا إلى بنية مكتملة التعبئة.

lh_<type>_doall() ستستدعي، لكل مدخل في جدول التجزئة، func مع عنصر البيانات كوسيط لها. بالنسبة لـ lh_<type>_doall() و lh_<type>_doall_arg()، ينبغي تجنّب تحويل مؤشر الدالة في دوال الاستدعاء (انظر ملاحظات) — بدلًا من ذلك استخدم وحدات الماكرو للتصريح/التنفيذ لإنشاء أغلفة مُتحقَّقة النوع تحوّل المتغيرات قبل استدعاء دوال الاستدعاء الخاصة بالنوع. يوضَّح مثال على ذلك هنا حيث تُستخدم دالة الاستدعاء لتنظيف الموارد للعناصر في جدول التجزئة قبل إلغاء تخصيص جدول التجزئة نفسه:

/* نظّف الموارد التابعة لـ 'a' (هذا منفّذ في مكان آخر) */
void STUFF_cleanup_doall(STUFF *a);
/* نفّذ غلافًا متوافقًا مع النموذج الأولي لـ &qu

يمكن أن تحذف دالة الاستدعاء (callback) عناصر من جدول التجزئة، لكن ليس من الآمن إدراج عناصر جديدة.

lh_<type>_doall_arg() هي نفسها lh_<type>_doall() باستثناء أن func ستُستدعى مع arg كوسيط ثانٍ ويجب أن تكون func من النوع LHASH_DOALL_ARG_FN_TYPE (نموذج أولي لدالة استدعاء يُمرَّر له كل من مدخل الجدول ووسيط إضافي). كما هو الحال مع lh_<type>_doall()، يمكنك بدلاً من ذلك الإعلان عن دالة الاستدعاء الخاصة بك بنموذج أولي يطابق الأنواع التي تتعامل معها واستخدام وحدات الماكرو declare/implement لإنشاء أغلفة متوافقة تحوّل المتغيرات قبل استدعاء دوال الاستدعاء الخاصة بنوعك. يوضح المثال التالي ذلك (طباعة جميع مدخلات جدول التجزئة إلى BIO يوفره المستدعي):

/* نفّذ غلافًا متوافقًا مع النموذج الأولي لـ "STUFF_print" */

lh_<type>_error() يمكن استخدامها لتحديد ما إذا حدث خطأ في العملية الأخيرة.

قيم الإرجاع

lh_<type>_new() تُرجع NULL عند حدوث خطأ، وإلا مؤشرًا إلى بنية LHASH الجديدة.

عند استبدال مدخل في جدول التجزئة، تُرجع lh_<type>_insert() القيمة التي يجري استبدالها. تُرجع NULL في التشغيل العادي وعند حدوث خطأ.

تُرجع lh_<type>_delete() المدخل الذي يجري حذفه. تُرجع NULL إذا لم تكن هناك قيمة كهذه في جدول التجزئة.

تُرجع lh_<type>_retrieve() مدخل جدول التجزئة إذا عُثر عليه، أو NULL بخلاف ذلك.

تُرجع lh_<type>_error() القيمة 1 إذا حدث خطأ في العملية الأخيرة، أو 0 بخلاف ذلك.

ملاحظات

توجد وحدات الماكرو وأنواع دوال الاستدعاء المختلفة الخاصة بـ LHASH لجعل من الممكن كتابة كود مُتحقَّق من الأنواع دون اللجوء إلى تحويل النماذج الأولية للدوال — وهو شر يجعل كود التطبيق أصعب بكثير في التدقيق/التحقق ويفتح أيضًا نافذة فرصة لتلف المكدس وأخطاء أخرى يصعب العثور عليها. كما أنه، على ما يبدو، يخالف ANSI-C.

يعتبر كود LHASH مدخلات الجدول بيانات ثابتة. وعلى هذا النحو، يمثل داخليًا العناصر المُدرَجة عبر lh_<type>_insert() بنوع مؤشر const void *. هذا هو السبب في أن دوال الاستدعاء مثل تلك المستخدمة بواسطة lh_<type>_doall() و lh_<type>_doall_arg() تعلن عن نماذجها الأولية باستخدام "const"، حتى بالنسبة للمعاملات التي تعيد تمرير مؤشرات بيانات عناصر الجدول — للاتساق، تكون البيانات التي يوفرها المستخدم "const" في جميع الأوقات بقدر ما يتعلق الأمر بكود LHASH. ومع ذلك، بما أن المستدعين هم أنفسهم من يوفرون هذه المؤشرات، فيمكنهم اختيار ما إذا كان ينبغي لهم أيضًا معاملة كل هذه المعاملات كثوابت.

على سبيل المثال، قد يُدار جدول تجزئة بواسطة كود ليس لديه، لأسباب تتعلق بالتغليف، سوى وصول "const" إلى البيانات المفهرسة في جدول التجزئة (أي تُرجع كـ "const" من مكان آخر في كودهم) — في هذه الحالة تكون نماذج LHASH الأولية مناسبة كما هي. على العكس من ذلك، إذا كان المستدعي مسؤولاً عن عمر البيانات المعنية، فقد يرغب في إجراء تعديلات على عنصر الجدول الذي يُعاد تمريره في دوال الاستدعاء lh_<type>_doall() أو lh_<type>_doall_arg() (انظر مثال "STUFF_cleanup" أعلاه). إذا كان الأمر كذلك، يمكن للمستدعي إما تحويل "const" بعيدًا (إذا كانوا يوفرون دوال الاستدعاء الخام بأنفسهم) أو استخدام وحدات الماكرو للإعلان عن/تنفيذ دوال الغلاف بدون أنواع "const".

المستدعون الذين لديهم فقط وصول "const" إلى البيانات التي يفهرسونها في جدول، ومع ذلك يعلنون عن دوال استدعاء بدون أنواع ثابتة (أو يحوّلون "const" بعيدًا بأنفسهم)، فإنهم بذلك يخلقون مخاطرهم/أخطاءهم الخاصة دون أن تشجعهم واجهة برمجة التطبيقات على ذلك. في ملاحظة ذات صلة، يجب على أولئك الذين يدققون الكود إيلاء اهتمام خاص لأي حالات من وحدات الماكرو DECLARE/IMPLEMENT_LHASH_DOALL_[ARG_]_FN التي توفر أنواعًا بدون أي محددات "const".

التفاصيل الداخلية

يعتمد الوصف التالي على وثائق SSLeay:

تنفذ مكتبة lhash جدول تجزئة موصوفًا في Communications of the ACM في عام 1991. ما يجعل جدول التجزئة هذا مختلفًا هو أنه مع امتلاء الجدول، يزداد حجم جدول التجزئة (أو ينقص) عبر reallocarray(3). عند إجراء 'تغيير الحجم'، بدلاً من إعادة توزيع جميع التجزئات على ضعف عدد 'الدلاء'، يُقسَّم دلو واحد. لذا عند إجراء 'توسيع'، لا توجد سوى تكلفة دنيا لإعادة توزيع بعض القيم. ستتسبب عمليات الإدراج اللاحقة في المزيد من عمليات إعادة توزيع 'الدلو' الواحد ولكن لن تكون هناك أبدًا تكلفة كبيرة مفاجئة بسبب إعادة توزيع جميع 'الدلاء'.

تُحفظ حالة جدول تجزئة معين في بنية LHASH. يُتخذ قرار زيادة أو تقليل حجم جدول التجزئة اعتمادًا على 'الحمل' (load) لجدول التجزئة. الحمل هو عدد العناصر في جدول التجزئة مقسومًا على حجم جدول التجزئة. القيم الافتراضية هي كما يلي. إذا (hash->up_load < load) => توسيع. إذا (hash->down_load > load) => تقليص. القيمة الافتراضية لـ up_load هي 1 والقيمة الافتراضية لـ down_load هي 2. يمكن تعديل هذه الأرقام بواسطة التطبيق بمجرد العبث بمتغيري up_load و down_load. يُحفظ 'الحمل' في صورة مضروبة في 256. لذا فإن hash->up_load=8*256 ستؤدي إلى تعيين حمل قدره 8.

إذا كنت مهتمًا بالأداء، فإن الحقل الذي يجب مراقبته هو num_comp_calls. تتتبع مكتبة التجزئة قيمة 'التجزئة' لكل عنصر، لذا عند إجراء بحث، تُقارن 'التجزئات'، وإذا كان هناك تطابق، يُجرى مقارنة كاملة، وتُزاد hash->num_comp_calls. إذا لم تكن num_comp_calls مساوية لـ num_delete زائد num_retrieve، فهذا يعني أن دالة التجزئة الخاصة بك تولّد تجزئات متطابقة لقيم مختلفة. ربما يستحق الأمر تغيير دالة التجزئة الخاصة بك إذا كان الأمر كذلك لأنه حتى لو كان جدول التجزئة الخاص بك يحتوي على 10 عناصر في 'دلو'، فيمكن البحث فيه باستخدام 10 مقارنات من نوع unsigned long و10 عمليات اجتياز لقائمة مترابطة. سيكون هذا أقل تكلفة بكثير من 10 استدعاءات لدالة المقارنة الخاصة بك.

lh_strhash() هي دالة تجزئة نصية تجريبية. نظرًا لأن روتينات LHASH عادةً تُمرَّر لها بنى، فإن هذه الروتين لن يُمرَّر عادةً إلى lh_<type>_new()، بل سيُستخدم في الدالة المُمرَّرة إلى lh_<type>_new().

التاريخ

ظهرت lh_new()، وlh_free()، وlh_insert()، وlh_delete()، وlh_retrieve()، وlh_doall()، وlh_strhash() في SSLeay 0.4 أو أقدم. ظهرت lh_doall_arg() لأول مرة في SSLeay 0.5.1. هذه الدوال متاحة منذ OpenBSD 2.4.

أُضيفت lh_<type>_error() في SSLeay 0.9.1b.

في OpenSSL 0.9.7، غُيِّرت جميع دوال lhash التي كانت تُمرَّر لها مؤشرات دوال لتحسين أمان الأنواع، وأصبحت أنواع الدوال LHASH_COMP_FN_TYPE، وLHASH_HASH_FN_TYPE، وLHASH_DOALL_FN_TYPE، و LHASH_DOALL_ARG_FN_TYPE متاحة.

في OpenSSL 1.0.0، أُعيد تجديد واجهة lhash لتحسين التحقق من الأنواع بشكل أكبر.

الأخطاء

تُرجع lh_<type>_insert() NULL في كل من النجاح والخطأ.
