libressl-EVP_PKEY_sign(3)

الحزمة: libressl

التراخيص: ISC, custom:OpenSSL

EVP_PKEY_SIGN(3) دليل وظائف المكتبة EVP_PKEY_SIGN(3)

الاسم

EVP_PKEY_sign_init، EVP_PKEY_sign — توقيع باستخدام خوارزمية مفتاح عام

الملخص

المكتبة «libcrypto»
#include <openssl/evp.h>

int
EVP_PKEY_sign_init(EVP_PKEY_CTX *ctx);

int
EVP_PKEY_sign(EVP_PKEY_CTX *ctx, unsigned char *sig, size_t *siglen, const unsigned char *tbs, size_t tbslen);

الوصف

تقوم الدالة EVP_PKEY_sign_init() بتهيئة سياق خوارزمية مفتاح عام باستخدام المفتاح ctx->pkey لعملية توقيع.

تقوم الدالة EVP_PKEY_sign() بتنفيذ عملية توقيع مفتاح عام باستخدام ctx. يتم تحديد البيانات المراد توقيعها باستخدام المعاملات tbs و tbslen. إذا كان sig هو NULL، فسيتم كتابة الحد الأقصى لحجم المخزن المؤقت للإخراج في المعامل siglen. إذا كان sig ليس NULL، فيجب أن يحتوي المعامل siglen قبل الاستدعاء على طول المخزن المؤقت لـ sig. إذا نجح الاستدعاء، فسيتم كتابة التوقيع في sig وكمية البيانات المكتوبة في siglen.

EVP_PKEY_sign() لا تقوم بتجزئة البيانات المراد توقيعها، ولذلك تستخدم عادةً لتوقيع الملخصات (digests). لتوقيع رسائل عشوائية، راجع واجهات التوقيع EVP_DigestSignInit(3) و EVP_SignInit(3) بدلاً من ذلك.

بعد استدعاء EVP_PKEY_sign_init()، يمكن إجراء عمليات تحكم خاصة بالخوارزمية لتعيين أي معلمات مناسبة للعملية؛ راجع EVP_PKEY_CTX_ctrl(3).

يمكن استدعاء الدالة EVP_PKEY_sign() أكثر من مرة على نفس السياق إذا تم تنفيذ عدة عمليات باستخدام نفس المعلمات.

قيم الإرجاع

EVP_PKEY_sign_init() و EVP_PKEY_sign() ترجع 1 للنجاح و 0 أو قيمة سالبة للفشل. على وجه الخصوص، يشير رمز الإرجاع -2 إلى أن العملية غير مدعومة بواسطة خوارزمية المفتاح العام.

أمثلة

توقيع البيانات باستخدام RSA مع حشو PKCS#1 وملخص SHA-256:

#include <openssl/evp.h>
#include <openssl/rsa.h>

EVP_PKEY_CTX *ctx;
/* md هو ملخص SHA-256 في هذا المثال. */
unsigned char *md, *sig;
size_t mdlen = 32, siglen;
EVP_PKEY *signing_key;

/*
 * ملاحظة: يفترض أن يكون signing_key و md معدين قبل الخطوة التالية.
 * يجب أن يكون signing_key مفتاح RSA خاصًا ويجب أن يشير md
 * إلى ملخص SHA-256 المراد توقيعه.
 */
ctx = EVP_PKEY_CTX_new(signing_key, NULL /* لا يوجد محرك */);
if (!ctx)
	/* حدث خطأ */
if (EVP_PKEY_sign_init(ctx) <= 0)
	/* خطأ */
if (EVP_PKEY_CTX_set_rsa_padding(ctx, RSA_PKCS1_PADDING) <= 0)
	/* خطأ */
if (EVP_PKEY_CTX_set_signature_md(ctx, EVP_sha256()) <= 0)
	/* خطأ */

/* تحديد طول المخزن المؤقت */
if (EVP_PKEY_sign(ctx, NULL, &siglen, md, mdlen) <= 0)
	/* خطأ */

sig = malloc(siglen);

if (!sig)
	/* فشل malloc */

if (EVP_PKEY_sign(ctx, sig, &siglen, md, mdlen) <= 0)
	/* خطأ */

/* التوقيع هو siglen بايت مكتوبة في المخزن المؤقت sig */
