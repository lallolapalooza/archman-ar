libressl-EVP_PKEY_verify(3)

الحزمة: libressl

التراخيص: ISC, custom:OpenSSL

EVP_PKEY_VERIFY(3) دليل وظائف المكتبة EVP_PKEY_VERIFY(3)

الاسم

EVP_PKEY_verify_init، EVP_PKEY_verify — التحقق من التوقيع باستخدام خوارزمية مفتاح عام

الملخص

المكتبة “libcrypto”
#include <openssl/evp.h>

int
EVP_PKEY_verify_init(EVP_PKEY_CTX *ctx);

int
EVP_PKEY_verify(EVP_PKEY_CTX *ctx, const unsigned char *sig, size_t siglen, const unsigned char *tbs, size_t tbslen);

الوصف

تقوم الدالة EVP_PKEY_verify_init() بتهيئة سياق خوارزمية مفتاح عام باستخدام المفتاح ctx->pkey لعملية التحقق من التوقيع.

تقوم الدالة EVP_PKEY_verify() بتنفيذ عملية التحقق من المفتاح العام باستخدام ctx. يتم تحديد التوقيع باستخدام المعلمات sig و siglen. يتم تحديد البيانات التي تم التحقق منها (أي البيانات التي يُعتقد أنها تم توقيعها في الأصل) باستخدام المعلمات tbs و tbslen.

بعد استدعاء EVP_PKEY_verify_init()، يمكن إجراء عمليات تحكم خاصة بالخوارزمية لتعيين أي معلمات مناسبة للعملية.

يمكن استدعاء الدالة EVP_PKEY_verify() أكثر من مرة على نفس السياق إذا تم إجراء عدة عمليات باستخدام نفس المعلمات.

قيم الإرجاع

EVP_PKEY_verify_init() و EVP_PKEY_verify() ترجع 1 إذا كان التحقق ناجحًا و 0 إذا فشل. على عكس الدوال الأخرى، فإن قيمة الإرجاع 0 من EVP_PKEY_verify() تشير فقط إلى أن التوقيع لم يتم التحقق منه بنجاح. أي أن tbs لم يتطابق مع البيانات الأصلية أو أن التوقيع كان ذو شكل غير صالح. هذا لا يشير إلى خطأ أكثر خطورة.

تشير القيمة السالبة إلى خطأ بخلاف فشل التحقق من التوقيع. على وجه الخصوص، تشير قيمة الإرجاع -2 إلى أن العملية غير مدعومة من قبل خوارزمية المفتاح العام.

أمثلة

التحقق من التوقيع باستخدام PKCS#1 و SHA-256 digest:

#include <openssl/evp.h>
#include <openssl/rsa.h>

EVP_PKEY_CTX *ctx;
unsigned char *md, *sig;
size_t mdlen, siglen;
EVP_PKEY *verify_key;

/*
 * يفترض أن verify_key، sig، siglen، md، و mdlen تم إعدادها بالفعل
 * وأن verify_key هو مفتاح RSA عام.
 */
ctx = EVP_PKEY_CTX_new(verify_key, NULL);
if (!ctx)
	/* حدث خطأ */
if (EVP_PKEY_verify_init(ctx) <= 0)
	/* خطأ */
if (EVP_PKEY_CTX_set_rsa_padding(ctx, RSA_PKCS1_PADDING) <= 0)
	/* خطأ */
if (EVP_PKEY_CTX_set_signature_md(ctx, EVP_sha256()) <= 0)
	/* خطأ */

/* تنفيذ العملية */
ret = EVP_PKEY_verify(ctx, sig, siglen, md, mdlen);

/*
 * ret == 1 يشير إلى النجاح، 0 فشل التحقق،
 * و < 0 خطأ آخر.
 */
