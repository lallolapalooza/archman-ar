libressl-EVP_PKEY_derive(3)

الحزمة: libressl

التراخيص: ISC, custom:OpenSSL

EVP_PKEY_DERIVE(3) دليل الدوال المكتبية EVP_PKEY_DERIVE(3)

الاسم

EVP_PKEY_derive_init، EVP_PKEY_derive_set_peer، EVP_PKEY_CTX_get0_peerkey، EVP_PKEY_derive — اشتقاق سر مشترك لخوارزمية المفتاح العام

الملخص

مكتبة "libcrypto"
#include <openssl/evp.h>

int
EVP_PKEY_derive_init(EVP_PKEY_CTX *ctx);

int
EVP_PKEY_derive_set_peer(EVP_PKEY_CTX *ctx, EVP_PKEY *peerkey);

EVP_PKEY *
EVP_PKEY_CTX_get0_peerkey(EVP_PKEY_CTX *ctx);

int
EVP_PKEY_derive(EVP_PKEY_CTX *ctx, unsigned char *key, size_t *keylen);

الوصف

EVP_PKEY_derive_init() تهيئة سياق خوارزمية المفتاح العام ctx لاشتقاق السر المشترك باستخدام كائن EVP_PKEY المخزن بالفعل في ctx. توفر المكتبة دعماً مدمجاً للمفاتيح ذات EVP_PKEY_base_id(3) EVP_PKEY_DH، EVP_PKEY_EC، EVP_PKEY_HKDF، و EVP_PKEY_X25519.

بعد استدعاء EVP_PKEY_derive_init() ، يمكن إجراء عمليات تحكم خاصة بالخوارزمية بشكل اختياري لتعيين أي معلمات مناسبة للعملية.

EVP_PKEY_derive_set_peer() يقوم بتهيئة ctx، والتي يجب أن تكون قد تم تهيئتها بالفعل مع EVP_PKEY_derive_init()، أو EVP_PKEY_encrypt_init(3)، أو EVP_PKEY_decrypt_init(3)، لاستخدام peerkey، والذي هو عادة مفتاح عام. في حالة النجاح، يتم زيادة عدد مراجع peerkey بمقدار واحد. بالتالي، يحتاج المتصل إلى استدعاء EVP_PKEY_free(3) على peerkey عندما لا يحتاج المتصل إليه بعد الآن، حتى لو كان لا يزال قيد الاستخدام بواسطة ctx.

EVP_PKEY_derive() يشتق سراً مشتركاً باستخدام ctx. إذا كان key هو NULL، فسيتم كتابة الحد الأقصى لحجم مخزن الإخراج في المعلمة keylen. إذا لم يكن key هو NULL، فيجب أن يحتوي المعلمة keylen قبل الاستدعاء على طول مخزن key. إذا كان الاستدعاء ناجحاً، فسيتم كتابة السر المشترك في key والكمية المكتوبة من البيانات في keylen.

يمكن استدعاء الدالة EVP_PKEY_derive() أكثر من مرة على نفس السياق إذا تم إجراء عدة عمليات باستخدام نفس المعلمات.

قيم الإرجاع

EVP_PKEY_derive_init()، EVP_PKEY_derive_set_peer()، و EVP_PKEY_derive() ترجع 1 للنجاح و 0 أو قيمة سالبة للفشل. على وجه الخصوص، يشير الإرجاع -2 إلى أن العملية غير مدعومة من قبل خوارزمية المفتاح العام.

بالنسبة لـ EVP_PKEY_derive_set_peer()، يمكن أن يحدث الإرجاع -1 على سبيل المثال إذا لم يتم تهيئة ctx بشكل صحيح، أو لا يحتوي على EVP_PKEY يمكن استرداده باستخدام EVP_PKEY_CTX_get0_pkey(3)، أو لا تتطابق EVP_PKEY_id(3) لكلا المفتاحين، أو EVP_PKEY_cmp_parameters(3) تُبلغ عن عدم تطابق معلمات المفتاح.

EVP_PKEY_derive() يفشل بإرجاع -1 على سبيل المثال إذا لم يتم تهيئة ctx بنجاح باستخدام EVP_PKEY_derive_init().

EVP_PKEY_CTX_get0_peerkey() ترجع مؤشراً داخلياً إلى peerkey المستخدم بواسطة ctx دون زيادة عدد مراجعته.

أمثلة

اشتقاق سر مشترك (على سبيل المثال مفاتيح DH أو EC):

#include <openssl/evp.h>
#include <openssl/rsa.h>

EVP_PKEY_CTX *ctx;
unsigned char *skey;
size_t skeylen;
EVP_PKEY *pkey, *peerkey;

/* يفترض أن pkey و peerkey قد تم إعدادهما بالفعل. */
ctx = EVP_PKEY_CTX_new(pkey, NULL);
if (!ctx)
	/* حدث خطأ */
if (EVP_PKEY_derive_init(ctx) <= 0)
	/* خطأ */
if (EVP_PKEY_derive_set_peer(ctx, peerkey) <= 0)
	/* خطأ */

/* تحديد طول المخزن */
if (EVP_PKEY_derive(ctx, NULL, &skeylen) <= 0)
	/* خطأ */

skey = malloc(skeylen);

if (!skey)
	/* فشل malloc */

if (EVP_PKEY_derive(ctx, skey, &skeylen) <= 0)
	/* خطأ */

/* السر المشترك هو skey بايت مكتوبة في المخزن skey */
