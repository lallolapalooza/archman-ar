الاسم

pm_system - تشغيل برنامج Netpbm مع إدخال وإخراج البرنامج

الملخص


#include <netpbm/pm_system.h>
pm_system(void                  stdinFeeder(int, void *),
          void *          const feederParm,
          void                  stdoutAccepter(int, void *),
          void *          const accepterParm,
          const char *    const shellCommand);
pm_system_lp(const char *  const progName,
             void                stdinFeeder(int, void *),
             void *        const feederParm,
             void                stdoutAccepter(int, void *),
             void *        const accepterParm,
             ...);
pm_system_vp(const char *  const progName,
             const char ** const argArray,
             void                stdinFeeder(int, void *),
             void *        const feederParm,
             void                stdoutAccepter(int, void *),
             void *        const accepterParm);

مثال

هذا المثال البسيط يحول صورة PNM من الإدخال القياسي إلى صورة JFIF (JPEG) من الإخراج القياسي. في هذه الحالة، لا يقوم pm_system() بأكثر مما يقوم به system().


    pm_system(NULL, NULL, NULL, NULL, "pnmtojpeg");

هذا المثال يفعل نفس الشيء، ولكنه ينقل البيانات عبر مخازن الذاكرة لتوضيح الاستخدام مع مخازن الذاكرة، ونضيف مرحلة لتقليص حجم الصورة أيضًا:

#include <netpbm/pm_system.h>
char              pnmData[100*1024];   /* يجب أن تكون ملف الإدخال < 100K */
char              jfifData[100*1024];
struct bufferDesc pnmBuffer;
struct bufferDesc jfifBuffer;
unsigned int      jfifSize;
pnmBuffer.size = fread(pnmData, 1, sizeof(pnmData), stdin);
pnmBuffer.buffer = pnmData;
pnmBuffer.bytesTransferredP = NULL;
jfifBuffer.size = sizeof(jfifData);
jfifBuffer.buffer = jfifData;
jfifBuffer.bytesTransferredP = &jfifSize; 
pm_system(&pm_feed_from_memory, &pnmBuffer,
          &pm_accept_to_memory, &jfifBuffer,
          "pamscale .5 | pnmtojpeg");
fwrite(jfifData, 1, jfifSize, stdout);

هذا المثال يقرأ صورة إلى هياكل PAM الخاصة بمكتبة libnetpbm، ثم يضيئها، ثم يكتبها، لتوضيح استخدام pm_system مع هياكل PAM.

#include <netpbm/pam.h>
#include <netpbm/pm_system.h>
struct pam       inpam;
struct pam       outpam;
tuple **         inTuples;
tuple **         outTuples;
struct pamtuples inPamtuples;
struct pamtuples outPamtuples;
inTuples = pnm_readpam(stdin, &inpam, sizeof(inpam));
outpam = inpam;
inPamtuples.pamP = &inpam;
inPamtuples.tuplesP = &inTuples;
outPamtuples.pamP = &outpam;
outPamtuples.tuplesP = &outTuples;
pm_system(&pm_feed_from_pamtuples, &inPamtuples,
          &pm_accept_to_pamtuples, &outPamtuples,
          "ppmbrighten -v 100");
outpam.file = stdout;
pnm_writepam(&outpam, outTuples);

الوصف

هذه الدوال المكتبية هي جزء من Netpbm(1)

pm_system() تشبه إلى حد كبير الروتين الفرعي system() الخاص بمكتبة C القياسية. يقوم بتشغيل shell ولديه هذا الـ shell ينفذ أمر shell تحدده. ولكن pm_system() يمنحك مزيدًا من التحكم في الإدخال القياسي والإخراج القياسي لأمر الـ shell هذا مقارنة بـ system(). يمرر system() إلى أمر الـ shell كمدخل قياسي ومخرج قياسي ما هو المدخل القياسي والمخرج القياسي للعملية التي تستدعي system(). ولكن مع pm_system()، يمكنك تحديد كمعاملات روتينات فرعية لتنفيذها لتوليد تدفق الإدخال القياسي لأمر الـ shell ومعالجة تدفق الإخراج القياسي لأمر الـ shell.

يمكن لروتين التغذية الخاص بك للإدخال القياسي توليد التدفق بطرق لا حصر لها. يمنحه pm_system() واصف ملف لأنبوب للكتابة إليه التدفق الذي يولده. يربط pm_system() الطرف الآخر من هذا الأنبوب بالإدخال القياسي لأمر الـ shell.

وبالمثل، يمكن لروتين القبول الخاص بك للإخراج القياسي فعل أي شيء يريده بالتدفق الذي يحصل عليه. يمنحه pm_system() واصف ملف لأنبوب للقراءة منه التدفق. يربط pm_system() الطرف الآخر من هذا الأنبوب بالإخراج القياسي لأمر الـ shell.

المعامل stdinFeeder هو مؤشر دالة يحدد الروتين الفرعي الخاص بك لتغذية الإدخال القياسي. يقوم pm_system() بتشغيله في عملية فرعية وينتظر تلك العملية حتى تنتهي (ويقبل حالة إكمالها) قبل العودة. feederParm هو المعامل الذي يمرره pm_system() إلى الروتين الفرعي؛ إنه مبهم لـ pm_system().

إذا مررت stdinFeeder = NULL، فإن pm_system() ببساطة يمرر تدفق الإدخال القياسي الحالي الخاص بك إلى أمر الـ shell (كما يفعل system())، ولا ينشئ عملية فرعية.

المعامل stdoutAccepter هو مؤشر دالة يحدد الروتين الفرعي الخاص بك لقبول الإخراج القياسي. تستدعيه pm_system() في العملية الحالية. accepterParm هو معامل مشابه لـ feederParm.

إذا مررت stdoutAccepter = NULL، فإن pm_system() ببساطة يمرر تدفق الإخراج القياسي الحالي الخاص بك إلى أمر الـ shell (كما يفعل system().

المعامل shellCommand هو سلسلة تنتهي بحرف NULL تحتوي على أمر الـ shell الذي يجب على الـ shell تنفيذه. يمكن أن يكون أي أمر له معنى للـ shell ويمكنه أخذ أنبوب للإدخال القياسي والإخراج القياسي. مثال:


    ppmbrighten -v 100 | pamdepth 255 | pamscale .5

ينشئ pm_system() عملية فرعية لتشغيل الـ shell وينتظر تلك العملية حتى تنتهي (ويقبل حالة إكمالها) قبل العودة.

pm_system_lp()

pm_system_lp() يشبه pm_system() باستثناء أنه بدلاً من تشغيل shell، الذي بدوره يشغل عادةً برنامجًا آخر، يمكنك تشغيل برنامج من اختيارك مباشرة.

المعامل progName يحدد البرنامج المراد تشغيله، بنفس الطريقة كما هو الحال مع execlp() أو أمر shell: إذا كان يحتوي على شرطة مائلة (/)، فهو الاسم الكامل للملف الذي يحتوي على البرنامج. إذا لم يكن كذلك، فهو اسم يجب البحث عنه في مسار البحث عن البرامج الخاص بالنظام (يتم تحديده بواسطة متغير البيئة PATH).

تحدد وسائط البرنامج بنفس طريقة execlp(): باستخدام الوسائط المتغيرة في نهاية قائمة وسائط pm_system_lp(). كل منها عبارة عن سلسلة تنتهي بحرف NULL. يجب أن يكون الوسيط الأخير NULL لإخبار pm_system_lp() بمكان انتهاء الوسائط.

لاحظ أن الوسيط الأول ('arg0') لبرنامج ما هو تقليديًا الكلمة الأولى للأمر المستخدم لتشغيل البرنامج، كما لو كان يتم تشغيله لأمر shell. بمعنى آخر، عادةً ما يكون اسم البرنامج.

مثال:


    pm_system_lp('pnmtojpeg', NULL, NULL, NULL, NULL,
                 'pnmtojpeg', 'mypicture.jpg', '-quality=50', NULL);

pm_system_lp() أكثر أمانًا بكثير من pm_system() عندما يحسب برنامجك الوسائط أو يحصل عليها من مستخدم. إذا قمت بإنشاء أمر shell باستخدام مثل هذه الوسائط، ما لم تكن حذرًا حقًا، فقد ينتهي بك الأمر بإنشاء أمر shell يقوم بشيء مختلف تمامًا عما كنت تقصده، لأن الوسيط يمكن أن يحتوي على أحرف ذات معنى للـ shell مثل '|'.

يمكن أن يكون pm_system_lp() أيضًا أسرع بكثير من pm_system()، لأنه يتجاوز عملية تشغيل الـ shell بأكملها.

pm_system_vp()

pm_system_vp() يشبه pm_system_lp() باستثناء أنه بدلاً من توفير وسائط البرنامج كوسائط متغيرة، فإنك توفرها كمصفوفة، كما هو الحال مع execvp(). يحدد عنصر NULL في المصفوفة نهاية الوسائط.

مثال:


    const char * argArray[3];
    argArray[0] = 'pnmtojpeg';
    argArray[1] = '-quality=50';
