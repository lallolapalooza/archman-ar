libkeccak_degeneralise_spec(3)

الحزمة: libkeccak

التراخيص: custom:ISC

LIBKECCAK_DEGENERALISE_SPEC(3) دليل وظائف المكتبة LIBKECCAK_DEGENERALISE_SPEC(3)

الاسم

libkeccak_degeneralise_spec - تعيين جميع معلمات المواصفات إلى تلقائي

ملخص

#include <libkeccak.h>
int libkeccak_degeneralise_spec(struct libkeccak_generalised_spec *spec, struct libkeccak_spec *output_spec);

الربط مع -lkeccak.

الوصف

ستقوم الدالة libkeccak_degeneralise_spec() بحل المعلمات التلقائية في *spec وترجمة المعلمات إلى *output_spec، بحيث يمكن استخدامها للتجزئة.

ستقوم الدالة بتعديل كل من *spec و *output_spec.

يجب عليك استدعاء الدالة libkeccak_spec_check(3) بعد استدعاء libkeccak_degeneralise_spec().

struct libkeccak_generalised_spec {
	long int bitrate;    /* معدل البت (بالبت) */
	long int capacity;   /* السعة (بالبت) */
	long int output;     /* حجم الإخراج (بالبت) */
	long int state_size; /* حجم الحالة (بالبت) */
	long int word_size;  /* حجم الكلمة (بالبت) */
};

قيم الإرجاع

تعيد الدالة libkeccak_degeneralise_spec() القيمة 0 إذا كانت الإعدادات قابلة للاستخدام. وإلا، فستعيد أحد الثوابت التالية.

LIBKECCAK_GENERALISED_SPEC_ERROR_STATE_NONPOSITIVE
حجم الحالة المحدد غير موجب.
LIBKECCAK_GENERALISED_SPEC_ERROR_STATE_TOO_LARGE
تجاوز حجم الحالة المحدد الحد المدعوم (حاليًا 1600 بت).
LIBKECCAK_GENERALISED_SPEC_ERROR_STATE_MOD_25
حجم الحالة المحدد، بالبت، لم يكن مكافئًا لـ 0 معدل 25. مما يعني أن حجم الحالة لا يمكن أن يغطي جميع المسارات بشكل متكافئ.
LIBKECCAK_GENERALISED_SPEC_ERROR_WORD_NONPOSITIVE
حجم الكلمة المحدد غير موجب.
LIBKECCAK_GENERALISED_SPEC_ERROR_WORD_TOO_LARGE
تجاوز حجم الكلمة المحدد الحد المدعوم (حاليًا 64 بت).
LIBKECCAK_GENERALISED_SPEC_ERROR_STATE_WORD_INCOHERENCY
حجم الحالة المحدد ليس أكبر بـ 25 مرة بالضبط من حجم الكلمة.
LIBKECCAK_GENERALISED_SPEC_ERROR_CAPACITY_NONPOSITIVE
السعة المحددة غير موجبة.
LIBKECCAK_GENERALISED_SPEC_ERROR_CAPACITY_MOD_8
لم تكن السعة المحددة مكافئة لـ 0 معدل 8، أي أنها لم تكن بالبايت الكامل.
LIBKECCAK_GENERALISED_SPEC_ERROR_BITRATE_NONPOSITIVE
معدل البت المحدد غير موجب.
LIBKECCAK_GENERALISED_SPEC_ERROR_BITRATE_MOD_8
لم يكن معدل البت المحدد مكافئًا لـ 0 معدل 8، أي أنه لم يكن بالبايت الكامل.
LIBKECCAK_GENERALISED_SPEC_ERROR_OUTPUT_NONPOSITIVE
حجم الإخراج المحدد غير موجب.
LIBKECCAK_GENERALISED_SPEC_ERROR_STATE_BITRATE_CAPACITY_INCONSISTENCY
مجموع معدل البت والسعة لا يساوي حجم الحالة (25 ضعف حجم الكلمة).

لاحظ أنه قد يكون هناك أكثر من خطأ واحد. يتم إرجاع الخطأ الأول الذي تم اكتشافه فقط.

الأخطاء

لا يمكن أن تفشل الدالة libkeccak_degeneralise_spec().

مثال

يقوم هذا المثال بتكوين struct libkeccak_spec لتحديد الإعدادات لـ Keccak[c = 512]:

int r;
struct libkeccak_spec spec;
struct libkeccak_generalised_spec gspec;
libkeccak_generalised_spec_initialise(&gspec);
gspec.capacity = 512;
if ((r = libkeccak_degeneralise_spec(&gspec, &spec)))
    goto fail_degeneralise_spec;
if ((r = libkeccak_spec_check(&spec)));
    goto fail_spec_check;
