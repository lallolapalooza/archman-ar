List__UtilsBy(3pm)

الحزمة: perl-list-utilsby

التراخيص: GPL, PerlArtistic

List::UtilsBy(3) وثائق بيرل من مساهمة المستخدم List::UtilsBy(3)

الاسم

"List::UtilsBy" - دوال مساعدة للقوائم من الدرجة العليا

الملخص

use List::UtilsBy qw( nsort_by min_by );
use File::stat qw( stat );
my @files_by_age = nsort_by { stat($_)->mtime } @files;
my $shortest_name = min_by { length } @names;

الوصف

توفر هذه الوحدة عددًا من دوال مساعدة القوائم، تأخذ جميعها كتلة رمزية أولية للتحكم في سلوكها. وهي تنويعات على دوال بيرل أساسية مماثلة أو دوال "List::Util" بأسماء مشابهة، ولكنها تستخدم الكتلة للتحكم في سلوكها. على سبيل المثال، تأخذ دالة بيرل الأساسية "sort" قائمة قيم وتعيدها، مرتبة حسب قيمتها النصية. تقوم الدالة "sort_by" بفرزها وفقًا للقيمة النصية التي تعيدها الدالة الإضافية، عند إعطائها كل قيمة.

my @names_sorted = sort @names;
my @people_sorted = sort_by { $_->name } @people;

الدوال

تمت إضافة جميع الدوال منذ الإصدار 0.04 ما لم يُذكر خلاف ذلك، حيث تم إعادة تسمية الأسماء الأصلية للإصدارات السابقة.

sort_by

@vals = sort_by { KEYFUNC } @vals

تعيد قائمة القيم مرتبة وفقًا للقيم النصية التي تعيدها كتلة أو دالة "KEYFUNC". قد يكون الاستخدام النموذجي لهذا هو فرز الكائنات وفقًا للقيمة النصية لواجهة وصول معينة، مثل

sort_by { $_->name } @people

يتم استدعاء دالة المفتاح في سياق قياسي، حيث يتم تمرير كل قيمة بدورها كـ $_ وكوسيط وحيد في المعلمات، @_. ثم يتم فرز القيم وفقًا للمقارنات النصية على القيم التي تم إرجاعها.

هذا يعادل

sort { $a->name cmp $b->name } @people

باستثناء أنها تضمن أن يتم تنفيذ واجهة الوصول "name" مرة واحدة فقط لكل قيمة.

أحد حالات الاستخدام المثيرة للاهتمام هو فرز السلاسل النصية التي قد تحتوي على أرقام مضمنة فيها "بشكل طبيعي"، بدلاً من الفرز الهجائي.

sort_by { s/(\d+)/sprintf "%09d", $1/eg; $_ } @strings

يقوم هذا بفرز السلاسل النصية عن طريق إنشاء مفاتيح فرز تقوم بإلحاق الأصفار بالأرقام المضمنة إلى مستوى معين (9 أرقام في هذه الحالة)، مما يساعد على ضمان أن الفرز الهجائي يضعها في الترتيب الصحيح.

nsort_by

@vals = nsort_by { KEYFUNC } @vals

مشابه لـ "sort_by" ولكنه يقارن قيم المفاتيح الخاصة به عدديًا.

rev_sort_by

rev_nsort_by

@vals = rev_sort_by { KEYFUNC } @vals
@vals = rev_nsort_by { KEYFUNC } @vals

منذ الإصدار 0.06.

مشابه لـ "sort_by" و "nsort_by" ولكنه يعيد القائمة بترتيب عكسي. يعادل

@vals = reverse sort_by { KEYFUNC } @vals

باستثناء أن هذه الدوال أكثر كفاءة قليلاً لأنها تتجنب عملية "reverse" النهائية.

max_by

$optimal = max_by { KEYFUNC } @vals
@optimal = max_by { KEYFUNC } @vals

يعيد القيمة (الأولى) من @vals التي تعطي أكبر نتيجة عددية من دالة المفتاح.

my $tallest = max_by { $_->height } @people
use File::stat qw( stat );
my $newest = max_by { stat($_)->mtime } @files;

في سياق قياسي، يتم إرجاع أول قيمة قصوى. في سياق قائمة، يتم إرجاع قائمة بجميع القيم القصوى. يمكن استخدام هذا للحصول على مواضع أخرى غير الأولى، إذا كان الترتيب مهمًا.

إذا تم استدعاؤها على قائمة فارغة، يتم إرجاع قائمة فارغة.

للتناسق مع دالة "nsort_by"، يتم توفير هذا أيضًا تحت الاسم "nmax_by" نظرًا لأنه يتصرف عدديًا.

min_by

$optimal = min_by { KEYFUNC } @vals
@optimal = min_by { KEYFUNC } @vals

مشابه لـ "max_by" ولكنه يعيد القيم التي تعطي أصغر نتيجة عددية من دالة المفتاح. متوفر أيضًا باسم "nmin_by"

minmax_by

( $minimal, $maximal ) = minmax_by { KEYFUNC } @vals

منذ الإصدار 0.11.

مشابه لاستدعاء كل من "min_by" و "max_by" بنفس دالة المفتاح على نفس القائمة. هذا الإصدار أكثر كفاءة من استدعاء الدالتين الأخريين بشكل فردي، لأنه يتطلب عملًا أقل بشكل عام. في حالة التعادل، يتم إرجاع العنصر الأمثل الأول الذي تم العثور عليه في كل حالة فقط. متوفر أيضًا باسم "nminmax_by".

uniq_by

@vals = uniq_by { KEYFUNC } @vals

يعيد قائمة بالمجموعة الفرعية من القيم التي تعيد كتلة دالة المفتاح قيمًا فريدة لها. يتم اختيار القيمة الأولى التي تنتج مفتاحًا معينًا، ويتم رفض القيم اللاحقة.

my @some_fruit = uniq_by { $_->colour } @fruit;

لاختيار القيمة الأخيرة لكل مفتاح بدلاً من ذلك، قم بعكس قائمة الإدخال. إذا كان ترتيب النتائج مهمًا، فلا تنسَ عكس النتيجة أيضًا:

my @some_fruit = reverse uniq_by { $_->colour } reverse @fruit;

نظرًا لأن القيم التي تعيدها دالة المفتاح تُستخدم كمفاتيح تجزئة، فيجب أن تكون سلاسل نصية، أو على الأقل أن تتصرف بشكل جيد كسلاسل نصية (مثل الأرقام، أو مراجع الكائنات التي تتجاوز السلسلة النصية بطريقة مناسبة).

partition_by

%parts = partition_by { KEYFUNC } @vals

يعيد قائمة مفتاح/قيمة من مراجع ARRAY تحتوي على جميع القيم الأصلية موزعة وفقًا لنتيجة كتلة دالة المفتاح. ستكون كل قيمة مرجع ARRAY يحتوي على جميع القيم التي أعادت السلسلة النصية من دالة المفتاح، بترتيبها الأصلي.

my %balls_by_colour = partition_by { $_->colour } @balls;

نظرًا لأن القيم التي تعيدها دالة المفتاح تُستخدم كمفاتيح تجزئة، فيجب أن تكون سلاسل نصية، أو على الأقل أن تتصرف بشكل جيد كسلاسل نصية (مثل الأرقام، أو مراجع الكائنات التي تتجاوز السلسلة النصية بطريقة مناسبة).

count_by

%counts = count_by { KEYFUNC } @vals

منذ الإصدار 0.07.

يعيد قائمة مفتاح/قيمة من الأعداد الصحيحة، مما يعطي عدد المرات التي أعادت فيها كتلة دالة المفتاح المفتاح، لكل قيمة في القائمة.

my %count_of_balls = count_by { $_->colour } @balls;

نظرًا لأن القيم التي تعيدها دالة المفتاح تُستخدم كمفاتيح تجزئة، فيجب أن تكون سلاسل نصية، أو على الأقل أن تتصرف بشكل جيد كسلاسل نصية (مثل الأرقام، أو مراجع الكائنات التي تتجاوز السلسلة النصية بطريقة مناسبة).

zip_by

@vals = zip_by { ITEMFUNC } \@arr0, \@arr1, \@arr2,...

يعيد قائمة بكل قيمة أعادتها كتلة الدالة، عند استدعائها بقيم من عبر كل مرجع ARRAY معطى. ستكون كل قيمة في القائمة التي تم إرجاعها نتيجة للدالة التي تم استدعاؤها بالوسائط في هذا الموضع، من عبر كل من المصفوفات المعطاة.

my @transposition = zip_by { [ @_ ] } @matrix;
my @names = zip_by { "$_[1], $_[0]" } \@firstnames, \@surnames;
print zip_by { "$_[0] => $_[1]\n" } [ keys %hash ], [ values %hash ];

إذا كانت بعض المصفوفات أقصر من غيرها، فستتصرف الدالة كما لو كانت تحتوي على "undef" في المواضع اللاحقة. السطران التاليان يعادلان:

zip_by { f(@_) } [ 1, 2, 3 ], [ "a", "b" ]
f( 1, "a" ), f( 2, "b" ), f( 3, undef )

يتم استدعاء دالة العنصر بواسطة "map"، لذلك إذا أعادت قائمة، يتم تضمين القائمة بأكملها في النتيجة. يمكن أن يكون هذا مفيدًا، على سبيل المثال، لتوليد تجزئة من قائمتين منفصلتين من المفاتيح والقيم.
