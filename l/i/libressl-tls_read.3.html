libressl-tls_read(3)

الحزمة: libressl

التراخيص: ISC, custom:OpenSSL

TLS_READ(3) دليل وظائف المكتبة TLS_READ(3)

الاسم

tls_read، tls_write، tls_handshake، tls_error، tls_close — استخدام اتصال TLS

ملخص

مكتبة “libtls” libssl libcrypto
#include <tls.h>

ssize_t
tls_read(struct tls *ctx، void *buf، size_t buflen);

ssize_t
tls_write(struct tls *ctx، const void *buf، size_t buflen);

int
tls_handshake(struct tls *ctx);

const char *
tls_error(struct tls *ctx);

int
tls_close(struct tls *ctx);

الوصف

tls_read() تقرأ buflen بايت من البيانات من المقبس إلى buf. تُرجع كمية البيانات المقروءة.

tls_write() تكتب buflen بايت من البيانات من buf إلى المقبس. تُرجع كمية البيانات المكتوبة.

tls_handshake() تنفذ بشكل صريح مصافحة TLS. لا يلزم استدعاء هذه الوظيفة إلا إذا كنت بحاجة إلى ضمان اكتمال المصافحة، حيث أن كل من tls_read() و tls_write() تقومان تلقائيًا بتنفيذ مصافحة TLS عند الضرورة.

يمكن استخدام الوظيفة tls_error() لاسترداد سلسلة نصية تحتوي على مزيد من المعلومات حول أحدث خطأ يتعلق بسياق (context).

tls_close() تغلق الاتصال بعد الاستخدام. سيتم إغلاق طبقة TLS فقط، ويكون المستدعي مسؤولاً عن إغلاق واصفات الملفات، إلا إذا تم إنشاء الاتصال باستخدام tls_connect(3) أو tls_connect_servername(3). بعد إغلاق الاتصال، يمكن تمرير ctx إلى tls_free(3).

قيم الإرجاع

tls_read() و tls_write() تُرجعان حجمًا عند النجاح أو -1 عند حدوث خطأ.

tls_handshake() و tls_close() تُرجعان 0 عند النجاح أو -1 عند حدوث خطأ.

لكل من وظائف tls_read()، tls_write()، tls_handshake()، و tls_close() قيمتان خاصتان للإرجاع:

TLS_WANT_POLLIN
يحتاج واصف الملف الخاص بالقراءة الأساسي إلى أن يكون قابلاً للقراءة للمتابعة.
TLS_WANT_POLLOUT
يحتاج واصف الملف الخاص بالكتابة الأساسي إلى أن يكون قابلاً للكتابة للمتابعة.

في حالة واصفات الملفات المحظورة، يجب تكرار نفس استدعاء الوظيفة على الفور. في حالة واصفات الملفات غير المحظورة، يجب تكرار نفس استدعاء الوظيفة عند استيفاء الشرط المطلوب.

لا يمكن للمستدعين لهذه الوظائف الاعتماد على قيمة المتغير العام errno. لمنع سوء التعامل مع حالات الخطأ، تقوم كل من tls_read()، و tls_write()، و tls_handshake()، و tls_close() بتصفير errno بشكل صريح.

tls_error() تُرجع NULL إذا لم يحدث خطأ مع ctx خلال أو منذ آخر استدعاء لـ tls_handshake()، أو tls_read()، أو tls_write()، أو tls_close()، أو tls_reset(3) التي تتضمن ctx، أو إذا فشل تخصيص الذاكرة عند محاولة تجميع السلسلة النصية التي تصف أحدث خطأ متعلق بـ ctx.

أمثلة

يوضح المثال التالي كيفية التعامل مع كتابات TLS على واصف ملفات محظور:

...
while (len > 0) {
	ssize_t ret;

	ret = tls_write(ctx, buf, len);
	if (ret == TLS_WANT_POLLIN || ret == TLS_WANT_POLLOUT)
		continue;
	if (ret == -1)
		errx(1, "tls_write: %s", tls_error(ctx));
	buf += ret;
	len -= ret;
}
...

يوضح المثال التالي كيفية التعامل مع كتابات TLS على واصف ملفات غير محظور باستخدام poll(2):

...
pfd[0].fd = fd;
pfd[0].events = POLLIN|POLLOUT;
while (len > 0) {
	nready = poll(pfd, 1, 0);
	if (nready == -1)
		err(1, "poll");
	if ((pfd[0].revents & (POLLERR|POLLNVAL)))
		errx(1, "bad fd %d", pfd[0].fd);
	if ((pfd[0].revents & (pfd[0].events|POLLHUP))) {
		ssize_t ret;

		ret = tls_write(ctx, buf, len);
		if (ret == TLS_WANT_POLLIN)
			pfd[0].events = POLLIN;
		else if (ret == TLS_WANT_POLLOUT)
			pfd[0].events = POLLOUT;
		else if (ret == -1)
			errx(1, "tls_write: %s", tls_error(ctx));
		else {
			buf += ret;
			len -= ret;
		}
	}
}
...

التاريخ

tls_read()، tls_write()، tls_error()، و tls_close() ظهرت في OpenBSD 5.6 وتم تسميتها نهائيًا في OpenBSD 5.7.

tls_handshake() ظهرت في OpenBSD 5.9.

تحذيرات

الوظيفة tls_error() تُرجع مؤشرًا داخليًا. يجب ألا يقوم التطبيق بتحريره، وإلا فسيحدث خطأ تحرير مزدوج. سيصبح المؤشر غير صالح عندما يحدث الخطأ التالي مع ctx. وبناءً على ذلك، إذا احتاج التطبيق إلى الرسالة في وقت لاحق، فيجب عليه نسخ السلسلة النصية قبل استدعاء وظيفة libtls التالية التي تتضمن ctx، أو قد تحدث نتيجة لذلك أخطاء تجزئة (segmentation fault) أو وصول قراءة لبيانات غير مقصودة.
