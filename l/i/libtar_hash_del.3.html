libtar_hash_del(3)

الحزمة: libtar

التراخيص: BSD

libtar_hash_new(3) استدعاءات مكتبة C libtar_hash_new(3)

الاسم

libtar_hash_new, libtar_hash_free, libtar_hash_next, libtar_hash_prev, libtar_hash_getkey, libtar_hash_search, libtar_hash_add, libtar_hash_del - روتينيات جدول التجزئة

الخلاصة

#include <libtar.h>

libtar_hash_t *libtar_hash_new(int num, int (*hashfunc)());
void libtar_hash_free(libtar_hash_t *h, void (*freefunc)());
int libtar_hash_next(libtar_hash_t *h, libtar_hashptr_t *hp);
int libtar_hash_prev(libtar_hash_t *h, libtar_hashptr_t *hp);
int libtar_hash_search(libtar_hash_t *h, libtar_hashptr_t *hp, void *data, int (*matchfunc)());
int libtar_hash_getkey(libtar_hash_t *h, libtar_hashptr_t *hp, void *data, int (*matchfunc)());
int libtar_hash_add(libtar_hash_t *h, void *data);
int libtar_hash_del(libtar_hash_t *h, libtar_hashptr_t *hp);

الوصف

تقوم الدالة libtar_hash_new() بإنشاء جدول تجزئة جديد بـ num من الدلاء واستخدام دالة التجزئة المشار إليها بواسطة hashfunc. إذا كانت hashfunc هي NULL، يتم استخدام دالة تجزئة افتراضية مصممة للسلاسل النصية ASCII ذات 7 بت.

تقوم الدالة libtar_hash_free() بتحرير جميع الذاكرة المرتبطة بهيكل التجزئة h. إذا لم تكن freefunc هي NULL، يتم استدعاؤها لتحرير الذاكرة المرتبطة بكل عقدة في التجزئة.

تُستخدم الدالتان libtar_hash_next() و libtar_hash_prev() للتنقل عبر جدول التجزئة. يحتوي الهيكل libtar_hashptr_t على حقلين: bucket، الذي يشير إلى الدلو الحالي في جدول التجزئة، و node، وهو مؤشر إلى العقدة الحالية في الدلو الحالي. لبدء التشغيل من بداية أو نهاية جدول التجزئة، يجب على المتصل تهيئة hp.bucket إلى -1 و hp.node إلى NULL.

تقوم الدالة libtar_hash_search() بالبحث بشكل تكراري عبر جدول التجزئة h حتى تجد عقدة يتطابق محتواها مع data باستخدام دالة المطابقة matchfunc. يبدأ البحث من الموقع الذي يشير إليه hp.

تستخدم الدالة libtar_hash_getkey() دالة التجزئة المرتبطة بـ h لتحديد الدلو الذي يجب أن يكون data فيه، وتبحث فقط في ذلك الدلو عن عقدة مطابقة باستخدام matchfunc. يبدأ البحث من الموقع الذي يشير إليه hp.

تضيف الدالة libtar_hash_add() data إلى جدول التجزئة h.

تقوم الدالة libtar_hash_del() بإزالة العقدة المشار إليها بواسطة hp.

قيمة الإرجاع

تُرجع الدالة libtar_hash_new() مؤشرًا إلى هيكل جدول التجزئة الجديد، أو NULL في حالة حدوث خطأ.

تُرجع الدالتان libtar_hash_next() و libtar_hash_prev() القيمة 1 عند إرجاع بيانات صالحة، و 0 عند نهاية جدول التجزئة.

تُرجع الدالتان libtar_hash_getkey() و libtar_hash_search() القيمة 1 عند العثور على تطابق، أو 0 بخلاف ذلك.

تُرجع الدالة libtar_hash_add() القيمة 0 عند النجاح، أو -1 في حالة حدوث خطأ (وتضبط errno).

تُرجع الدالة libtar_hash_del() القيمة 0 عند النجاح، أو -1 في حالة حدوث خطأ (وتضبط errno).
