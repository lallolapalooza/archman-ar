libssh2_sftp_fstat_ex(3)

الحزمة: libssh2

التراخيص: BSD

libssh2_sftp_fstat_ex(3) libssh2 libssh2_sftp_fstat_ex(3)

الاسم

libssh2_sftp_fstat_ex - الحصول على أو تعيين سمات على مقبض ملف SFTP

الملخص

#include <libssh2.h>
#include <libssh2_sftp.h>
int
libssh2_sftp_fstat_ex(LIBSSH2_SFTP_HANDLE *handle,
                      LIBSSH2_SFTP_ATTRIBUTES *attrs, int setstat)
#define libssh2_sftp_fstat(handle, attrs) \
    libssh2_sftp_fstat_ex((handle), (attrs), 0)
#define libssh2_sftp_fsetstat(handle, attrs) \
    libssh2_sftp_fstat_ex((handle), (attrs), 1)

الوصف

handle - مقبض ملف SFTP كما تم إرجاعه بواسطة libssh2_sftp_open_ex(3)

attrs - مؤشر إلى بنية LIBSSH2_SFTP_ATTRIBUTES لتعيين بيانات التعريف للملف منها أو إليها اعتمادًا على قيمة setstat.

setstat - عندما تكون غير صفرية، سيتم تحديث بيانات التعريف للملف بالبيانات الموجودة في attrs وفقًا لقيم attrs->flags والسمات الأعضاء الأخرى ذات الصلة.

الحصول على أو تعيين بيانات من نوع statbuf لمثيل LIBSSH2_SFTP_HANDLE معين.

أنواع البيانات

LIBSSH2_SFTP_ATTRIBUTES هو بنية معرفة بـ typedef والتي يتم تعريفها على النحو التالي

struct _LIBSSH2_SFTP_ATTRIBUTES {
    /* إذا تم تعيين بت flags & ATTR_* ، فإن القيمة في هذه
     * البنية ستكون ذات معنى. بخلاف ذلك يجب تجاهلها
     */
    unsigned long flags;
    /* حجم الملف، بالبايت */
    libssh2_uint64_t filesize;
    /* تمثيل رقمي للمستخدم والمجموعة المالكة
     * للملف
     */
    unsigned long uid, gid;
    /* قناع بت للأذونات */
    unsigned long permissions;
    /* وقت الوصول ووقت تعديل الملف */
    unsigned long atime, mtime;
};

ستجد مجموعة كاملة من التعريفات والوحدات الماكرو لتحديد العلامات والأذونات في ملف الرأس libssh2_sftp.h، ولكن بعضها الأكثر شيوعًا هو:

للتحقق من أذونات مستخدم معينة، فإن مجموعة التعريفات تكون بالنمط LIBSSH2_SFTP_S_I<action><who> حيث <action> هو R أو W أو X للقراءة أو الكتابة أو التنفيذ و <who> هو USR أو GRP أو OTH للمستخدم أو المجموعة أو الآخر. لذلك، للتحقق مما إذا كان الملف قابلاً للقراءة من قبل المستخدم، استخدم البت LIBSSH2_SFTP_S_IRUSR بينما تريد معرفة ما إذا كان قابلاً للتنفيذ للآخرين، استخدم LIBSSH2_SFTP_S_IXOTH وهكذا.

للتحقق من أنواع الملفات المحددة، كنت ستستخدم سابقًا (قبل libssh2 1.2.5) وحدات الماكرو القياسية posix S_IS***()، ولكن منذ 1.2.5 يوفر libssh2 مجموعته الخاصة من وحدات الماكرو لهذه الوظيفة:

LIBSSH2_SFTP_S_ISLNK
اختبار للرابط الرمزي
LIBSSH2_SFTP_S_ISREG
اختبار لملف عادي
LIBSSH2_SFTP_S_ISDIR
اختبار لدليل
LIBSSH2_SFTP_S_ISCHR
اختبار لملف خاص بالحرف
LIBSSH2_SFTP_S_ISBLK
اختبار لملف خاص بالكتلة
LIBSSH2_SFTP_S_ISFIFO
اختبار لأنابيب أو ملفات FIFO خاصة
LIBSSH2_SFTP_S_ISSOCK
اختبار لمقبس

قيمة الإرجاع

إرجاع 0 عند النجاح أو قيمة سالبة عند الفشل. ترجع LIBSSH2_ERROR_EAGAIN عندما يتم حظرها بخلاف ذلك. بينما LIBSSH2_ERROR_EAGAIN هو رقم سالب، إلا أنه ليس فشلاً حقيقيًا.

الأخطاء

LIBSSH2_ERROR_ALLOC - فشل استدعاء تخصيص ذاكرة داخلي.

LIBSSH2_ERROR_SOCKET_SEND - غير قادر على إرسال البيانات عبر المقبس.

LIBSSH2_ERROR_SOCKET_TIMEOUT -

LIBSSH2_ERROR_SFTP_PROTOCOL - تم استلام استجابة بروتوكول SFTP غير صالحة على المقبس، أو تسببت عملية SFTP في إرجاع رمز خطأ بواسطة الخادم.

التوفر

هذه الوظيفة موجودة منذ زمن طويل، ولكن تم تقديم معظم التعريفات LIBSSH2_SFTP_S_* في libssh2 0.14 وتم تقديم وحدات الماكرو LIBSSH2_SFTP_S_IS***() في libssh2 1.2.5.
