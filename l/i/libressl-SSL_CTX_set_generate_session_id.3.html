libressl-SSL_CTX_set_generate_session_id(3)

الحزمة: libressl

التراخيص: ISC, custom:OpenSSL

SSL_CTX_SET_GENERATE_SESSION_ID(3) دليل وظائف المكتبة SSL_CTX_SET_GENERATE_SESSION_ID(3)

الاسم

SSL_CTX_set_generate_session_id، SSL_set_generate_session_id، SSL_has_matching_session_id، GEN_SESSION_CB — التلاعب بإنشاء معرفات جلسات SSL (الخادم فقط)

الملخص

مكتبة "libssl"
#include <openssl/ssl.h>

typedef int
(*GEN_SESSION_CB)(const SSL *ssl, unsigned char *id, unsigned int *id_len);

int
SSL_CTX_set_generate_session_id(SSL_CTX *ctx, GEN_SESSION_CB cb);

int
SSL_set_generate_session_id(SSL *ssl, GEN_SESSION_CB cb);

int
SSL_has_matching_session_id(const SSL *ssl, const unsigned char *id, unsigned int id_len);

الوصف

SSL_CTX_set_generate_session_id() تعين وظيفة رد الاتصال لإنشاء معرفات جلسات جديدة لـ SSL/TLS لـ ctx لتكون cb.

SSL_set_generate_session_id() تعين وظيفة رد الاتصال لإنشاء معرفات جلسات جديدة لـ SSL/TLS لـ ssl لتكون cb.

SSL_has_matching_session_id() تتحقق مما إذا كانت جلسة ذات معرف id (بطول id_len) موجودة بالفعل في ذاكرة التخزين المؤقت للجلسات الداخلية للسياق الأصل لـ ssl.

عند إنشاء جلسة جديدة بين العميل والخادم، يقوم الخادم بإنشاء معرف جلسة. معرف الجلسة هو تسلسل عشوائي من البايتات. طول معرف الجلسة يتراوح بين 1 و 32 بايت. معرف الجلسة ليس حرجًا من الناحية الأمنية ولكنه يجب أن يكون فريدًا للخادم. بالإضافة إلى ذلك، يتم إرسال معرف الجلسة بوضوح عند إعادة استخدام الجلسة، لذلك يجب ألا يحتوي على معلومات حساسة.

بدون تعيين وظيفة رد الاتصال، سيقوم خادم OpenSSL بإنشاء معرف جلسة فريد من أرقام عشوائية زائفة بأقصى طول ممكن. باستخدام وظيفة رد الاتصال، يمكن تغيير معرف الجلسة ليشمل معلومات إضافية مثل، على سبيل المثال، معرف المضيف لتحسين موازنة التحميل أو تقنيات التخزين المؤقت الخارجية.

تتلقى وظيفة رد الاتصال مؤشرًا إلى موقع الذاكرة لوضع id فيه ومؤشرًا إلى الحد الأقصى للطول المسموح به id_len. لا يضمن أن يكون حجم المخزن المؤقت في الموقع id هو id_len فقط. يُسمح لوظيفة رد الاتصال فقط بإنشاء معرف أقصر وتقليل id_len. يجب ألا تزيد وظيفة رد الاتصال id_len أبدًا أو تكتب في الموقع id بما يتجاوز الحد المعطى.

يتم ملء الموقع id بـ 0x00 قبل استدعاء وظيفة رد الاتصال، لذلك قد تقوم وظيفة رد الاتصال بملء جزء فقط من الطول الممكن وترك id_len دون تغيير مع الحفاظ على القابلية للتكرار.

نظرًا لأنه يجب تمييز الجلسات، يجب أن تكون معرفات الجلسات فريدة. بدون وظيفة رد الاتصال، يتم استخدام رقم عشوائي، بحيث يكون احتمال إنشاء نفس معرف الجلسة صغيرًا للغاية (2^256 لـ TLSv1). لضمان تفرد معرف الجلسة الذي تم إنشاؤه، يجب على وظيفة رد الاتصال استدعاء SSL_has_matching_session_id() وإنشاء معرف آخر في حالة حدوث تعارض. إذا لم يتم حل تعارض المعرف، فسيفشل المصافحة. إذا قامت التطبيق بترميز، على سبيل المثال، معرف مضيف فريد، ورقم عملية فريد، ورقم تسلسلي فريد في معرف الجلسة، يمكن تحقيق التفرد بسهولة بدون إضافة عشوائية (ومع ذلك، يجب الحرص على عدم تسريب أي معلومات سرية بهذه الطريقة). إذا لم يتمكن التطبيق من ضمان التفرد، فمن المستحسن استخدام id_len الأقصى وملء البايتات غير المستخدمة لترميز معلومات خاصة ببيانات عشوائية لتجنب التصادمات.

SSL_has_matching_session_id() سيستعلم فقط من ذاكرة التخزين المؤقت للجلسات الداخلية، وليس الخارجية. نظرًا لأنه يتم إنشاء معرف الجلسة قبل اكتمال المصافحة، فإنه لا يتم إضافته فورًا إلى ذاكرة التخزين المؤقت. إذا كان خيط آخر يستخدم نفس ذاكرة التخزين المؤقت للجلسات الداخلية، فقد تحدث حالة سباق حيث يقوم خيط آخر بإنشاء نفس معرف الجلسة. يمكن أن تحدث التصادمات أيضًا عند استخدام ذاكرة تخزين مؤقت خارجية للجلسات، نظرًا لأنه لا يتم اختبار ذاكرة التخزين المؤقت الخارجية باستخدام SSL_has_matching_session_id() وتنطبق نفس حالة السباق.

يجب أن تعيد وظيفة رد الاتصال 0 إذا لم تتمكن من إنشاء معرف جلسة لأي سبب، وتعيد 1 عند النجاح.

قيم الإرجاع

SSL_CTX_set_generate_session_id() و SSL_set_generate_session_id() تعيدان دائمًا 1.

SSL_has_matching_session_id() تعيد 1 إذا كانت هناك جلسة أخرى بنفس المعرف موجودة بالفعل في ذاكرة التخزين المؤقت.

أمثلة

وظيفة رد الاتصال المدرجة ستنشئ معرف جلسة مع معرف الخادم المحدد، وستملأ الباقي ببايتات عشوائية زائفة:

const char session_id_prefix = "www-18";

#define MAX_SESSION_ID_ATTEMPTS 10
static int
generate_session_id(const SSL *ssl, unsigned char *id,
    unsigned int *id_len)
{
	unsigned int count = 0;

	do {
		RAND_pseudo_bytes(id, *id_len);
		/*
		 * Prefix the session_id with the required prefix. NB: If
		 * our prefix is too long, clip it – but there will be
		 * worse effects anyway, e.g., the server could only
		 * possibly create one session ID (the prefix!) so all
		 * future session negotiations will fail due to conflicts.
		 */
		memcpy(id, session_id_prefix,
		    (strlen(session_id_prefix) < *id_len) ?
		    strlen(session_id_prefix) : *id_len);
	} while (SSL_has_matching_session_id(ssl, id, *id_len) &&
	    (++count < MAX_SESSION_ID_ATTEMPTS));

	if (count >= MAX_SESSION_ID_ATTEMPTS)
		return 0;
	return 1;
}
