الاسم

libcaca-font - تنسيق خطوط libcaca (الإصدار 1)

جميع الأنواع هي big endian.

struct
{
magic:
   uint8_t caca_header[2];    // "CA\xCA"
   uint8_t caca_file_type[2]; // "FT"
font_header:
   uint32_t control_size;     // حجم التحكم (font_data - font_header)
   uint32_t data_size;        // حجم البيانات (EOF - font_data)
   uint16_t version;          // إصدار تنسيق الخط
                              //  بت 0: تم تعيينه إلى 1 إذا كان الخط متوافقًا
                              //         مع الإصدار 1 من التنسيق
                              //  بتات 1-15: غير مستخدمة بعد، يجب أن تكون 0
   uint16_t blocks;           // عدد الكتل في الخط
   uint32_t glyphs;           // العدد الإجمالي للأشكال في الخط
   uint16_t bpp;              // بت لكل بكسل لبيانات الشكل (القيم الصحيحة
                              // هي 1، 2، 4 و 8)
   uint16_t width;            // عرض الشكل القياسي
   uint16_t height;           // ارتفاع الشكل القياسي
   uint16_t maxwidth;         // أقصى عرض للشكل
   uint16_t maxheight;        // أقصى ارتفاع للشكل
   uint16_t flags;            // أعلام الميزات
                              //  بت 0: تم تعيينه إلى 1 إذا كان الخط ثابت العرض
                              //  بتات 1-15: غير مستخدمة بعد، يجب أن تكون 0
block_info:
   struct
   {
      uint32_t start;         // فهرس Unicode لأول شكل
      uint32_t stop;          // فهرس Unicode لآخر شكل + 1
      uint32_t index;         // فهرس معلومات الشكل لأول شكل
   }
   block_list[blocks];
glyph_info:
   struct
   {
      uint16_t width;         // عرض الشكل بالبكسل
      uint16_t height;        // ارتفاع الشكل بالبكسل
      uint32_t data_offset;   // الإزاحة (بدءًا من البيانات) للبيانات
                              // للحرف الأول
   }
   glyph_list[glyphs];
control_extension_1:
control_extension_2:
   ...
control_extension_N:
   ...                        // محجوز للاستخدام المستقبلي
font_data:
   uint8_t data[data_size];   // بيانات الشكل
data_extension_1:
data_extension_2:
   ...
data_extension_N:
   ...                        // محجوز للاستخدام المستقبلي
};
