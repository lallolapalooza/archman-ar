الاسم

X25519, ed25519 - منحنى X25519 ومنحنى Ed25519

ملخص

#include <openssl/x25519.h>

#include <openssl/ed25519.h>

int X25519_keypair(X25519_KEY *key);

int X25519_public_from_private(X25519_PUBLIC_KEY *pub, const X25519_PRIVATE_KEY *priv);

int X25519_dh(X25519_KEY *out, const X25519_PRIVATE_KEY *priv, const X25519_PUBLIC_KEY *peer);

int Ed25519_keypair(Ed25519_KEY *key);

int Ed25519_public_from_private(Ed25519_PUBLIC_KEY *pub, const Ed25519_PRIVATE_KEY *priv);

int Ed25519_sign(Ed25519_SIGNATURE *sig, const Ed25519_PRIVATE_KEY *priv, const uint8_t *msg, size_t msglen);

int Ed25519_verify(const Ed25519_SIGNATURE *sig, const Ed25519_PUBLIC_KEY *pub, const uint8_t *msg, size_t msglen);

الوصف

تُقدم هذه الوظائف واجهات لـ X25519 و Ed25519.

X25519 هي دالة اشتقاق مفتاح (KDF) بناءً على منحنى Elliptic Curve Diffie-Hellman (ECDH) باستخدام منحنى Montogomery الذي يطلق عليه X25519. تُستخدم لتوليد أزواج المفاتيح (خاص/عام) وإنشاء اتفاقيات مفاتيح مشتركة.

Ed25519 هي خوارزمية توقيع رقمي منحنى بيضاوي. تُستخدم لتوليد أزواج مفاتيح التوقيع (خاص/عام) وإنشاء التوقيعات والتحقق منها.

قيم الإرجاع

ترجع جميع الوظائف 1 عند النجاح و 0 عند الفشل.

أمثلة

مثال على استخدام X25519 لاتفاقية مفتاح: 


X25519_KEY my_key;
X25519_keypair(&my_key);

X25519_KEY peer_key;
// ... الحصول على مفتاح العميل العام ...

X25519_KEY shared_secret;
X25519_dh(&shared_secret, &my_key.priv, &peer_key.pub);

مثال على استخدام Ed25519 للتوقيع والتحقق: 


Ed25519_KEY my_signing_key;
Ed25519_keypair(&my_signing_key);

uint8_t message[] = "Hello, world!";
Ed25519_SIGNATURE signature;
Ed25519_sign(&signature, &my_signing_key.priv, message, sizeof(message));

Ed25519_verify(&signature, &my_signing_key.pub, message, sizeof(message));
