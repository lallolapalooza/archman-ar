دوال libalpm متنوعة

توثيق النوع المعرف

typedef int64_t alpm_time_t

نوع الوقت المستخدم بواسطة libalpm. يمثل طابع زمني يونكس

توثيق نوع التعداد

enum alpm_caps

تعداد للميزات الممكنة وقت الترجمة.

المُعدِّد

ALPM_CAPABILITY_NLS
الترجمة المحلية
ALPM_CAPABILITY_DOWNLOADER
القدرة على التنزيل.
ALPM_CAPABILITY_SIGNATURES
فحص التوقيعات.

توثيق الدوال

int alpm_capabilities (void )

الحصول على إمكانيات المكتبة.

يعيد

قناع بت للإمكانيات

char * alpm_compute_md5sum (const char * filename)

الحصول على ملخص md5 للملف.

المعلمات

filename اسم الملف

يعيد

الملخص عند النجاح، NULL عند الخطأ

char * alpm_compute_sha256sum (const char * filename)

الحصول على ملخص sha256 للملف.

المعلمات

filename اسم الملف

يعيد

الملخص عند النجاح، NULL عند الخطأ

int alpm_sandbox_setup_child (alpm_handle_t * handle, const char * sandboxuser, const char * sandbox_path)

خفض الامتيازات عن طريق التبديل إلى مستخدم مختلف.

المعلمات

handle مقبض السياق
sandboxuser المستخدم للتبديل إليه
sandbox_path إذا لم يكن NULL، قصر الكتابات على مسار نظام الملفات هذا

يعيد

0 عند النجاح، -1 عند الفشل

alpm_pkg_t * alpm_sync_get_new_version (alpm_pkg_t * pkg, alpm_list_t * dbs_sync)

التحقق من وجود إصدار جديد للحزمة pkg في قواعد بيانات المزامنة. إذا ظهرت نفس الحزمة في قواعد بيانات متعددة، سيتم التحقق من الأولى فقط

هذا يتحقق فقط من قاعدة بيانات المزامنة بحثًا عن إصدار أحدث. لا يصل إلى الشبكة على الإطلاق. انظر alpm_db_update لتحديث قاعدة بيانات.

int alpm_unlock (alpm_handle_t * handle)

إزالة ملف قفل قاعدة البيانات.

المعلمات

handle مقبض السياق

يعيد

0 عند النجاح، -1 عند الخطأ

ملاحظة

آمن للاستدعاء من داخل معالجات الإشارات.

const char * alpm_version (void )

الحصول على إصدار المكتبة.

يعيد

إصدار المكتبة، مثال "6.0.4"
