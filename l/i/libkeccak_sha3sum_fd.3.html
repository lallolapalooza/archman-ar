libkeccak_sha3sum_fd(3)

الحزمة: libkeccak

التراخيص: custom:ISC

LIBKECCAK_SHA3SUM_FD(3) دليل وظائف المكتبة LIBKECCAK_SHA3SUM_FD(3)

الاسم

libkeccak_sha3sum_fd - حساب مجموع تجزئة SHA-3 لملف

الموجز

#include <libkeccak.h>
int libkeccak_sha3sum_fd(int fd, struct libkeccak_state *state, long int output, void *hashsum);

الربط مع -lkeccak.

الوصف

تقوم الدالة libkeccak_sha3sum_fd() بحساب مجموع تجزئة SHA-3 لملف، يتم تحديد واصف الملف الخاص به بواسطة fd (ويجب أن يكون في بداية الملف.) يتم ضبط خوارزمية التجزئة بواسطة المعامل output؛ فهو يحدد حجم الخرج، بالبت.

يتم تخزين التجزئة في شكل ثنائي في hashsum. يجب أن يكون تخصيص hashsum بحجم لا يقل عن (((output + 7) / 8) * sizeof(char)).

*state يجب ألا يكون مهيئًا. تقوم libkeccak_sha3sum_fd() بتهيئته بنفسها. لذلك سيكون هناك تسرب للذاكرة إذا كان *state مهيئًا بالفعل.

قيم الإرجاع

تُرجع الدالة libkeccak_sha3sum_fd() القيمة 0 عند الانتهاء بنجاح. عند حدوث خطأ، يتم إرجاع -1 ويتم تعيين errno لوصف الخطأ.

الأخطاء

قد تفشل الدالة libkeccak_sha3sum_fd() لأي سبب، باستثناء الأسباب التي تؤدي إلى تعيين errno إلى EINTR، والمحددة للدوال read(2)، malloc(3)، و realloc(3).

ملاحظات

كن على علم، تقوم libkeccak_sha3sum_fd() بتجزئة الملف حتى نهايته. بالنسبة للأنابيب والمقابس، هذا يعني حتى يتم إغلاق الملف. ولكن بالنسبة للأجهزة الحرفية، هذا يعني عادةً عدم الانتهاء أبدًا. لذلك، فإن محاولة تجزئة الملفات في /dev عادة ما تكون فكرة سيئة. لا تتحقق libkeccak_sha3sum_fd() من طول الملف أو نوعه قبل التجزئة لأن هذا قد يحد مما يمكنك فعله، ويجعل المكتبة أكثر تعقيدًا.

libkeccak_sha3sum_fd() لا تتوقف إذا تم مقاطعتها (read(2) تُرجع EINTR.)

تفترض libkeccak_sha3sum_fd() أن جميع المعلومات غير حساسة، ولن تقوم بالتالي بأي مسح آمن للمعلومات.

libkeccak_sha3sum_fd() لا تتحقق من ضبط الخوارزمية.

مثال

يحسب هذا المثال تجزئة SHA3-256 للمدخلات من الإدخال القياسي (stdin)، ويطبع التجزئة، بتنسيق ست عشري، إلى الإخراج القياسي (stdout).

struct libkeccak_state state;
if (libkeccak_sha3sum_fd(STDIN_FILENO, &state, 256, binhash) < 0)
	goto fail;
libkeccak_behex_lower(hexhash, binhash, sizeof(binhash));
printf("%s\n", hexhash);
libkeccak_state_destroy(&state);
