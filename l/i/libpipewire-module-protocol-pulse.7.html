الاسم

libpipewire-module-protocol-pulse - بروتوكول Pulse

الوصف

تقوم هذه الوحدة بتطبيق خادم PulseAudio كامل فوق PipeWire.

هذا هو تطبيق الخادم فقط، ومن المتوقع أن تستخدم العملاء مكتبة عملاء PulseAudio الأصلية. يوفر هذا مستوى عالٍ من التوافق مع التطبيقات الحالية؛ في الواقع، ستستمر جميع أدوات PulseAudio المعتادة مثل pavucontrol و pactl و pamon و paplay في العمل كما كانت من قبل.

عادةً ما يتم تحميل هذه الوحدة كجزء من عملية pipewire مستقلة، تسمى pipewire-pulse، باستخدام ملف التكوين pipewire-pulse.conf.

يقوم خادم pulse بتطبيق ذاكرة تخزين مؤقت للعينة غير متوفرة حاليًا في PipeWire.

اسم الوحدة

libpipewire-module-protocol-pulse

خيارات الوحدة

يمكن أن تكون وسائط الوحدة هي محتويات pulse.properties، ولكن يوصى بإنشاء قسم pulse.properties منفصل في ملف التكوين للسماح بالتجاوزات.

خصائص PULSE.PROPERTIES

يمكن تقديم قسم تكوين يحتوي على خصائص الخادم.

# ~/.config/pipewire/pipewire-pulse.conf.d/custom.conf
pulse.properties = {
    # العناوين التي يستمع عليها هذا الخادم
    server.address = [
        "unix:native"
        #"unix:/tmp/something"              # يمكن استخدام المسارات المطلقة
        #"tcp:4713"                         # IPv4 و IPv6 على جميع العناوين
        #"tcp:[::]:9999"                    # IPv6 على جميع العناوين
        #"tcp:127.0.0.1:8888"               # IPv4 على عنوان واحد
        #
        #{ address = "tcp:4713"             # العنوان
        #  max-clients = 64                 # الحد الأقصى لعدد العملاء
        #  listen-backlog = 32              # قائمة انتظار الخادم
        #  client.access = "restricted"     # أذونات العملاء
        #}
    ]
    #server.dbus-name       = "org.pulseaudio.Server"
    #pulse.allow-module-loading = true
    #pulse.min.req          = 128/48000     # 2.7 مللي ثانية
    #pulse.default.req      = 960/48000     # 20 مللي ثانية
    #pulse.min.frag         = 128/48000     # 2.7 مللي ثانية
    #pulse.default.frag     = 96000/48000   # 2 ثانية
    #pulse.default.tlength  = 96000/48000   # 2 ثانية
    #pulse.min.quantum      = 128/48000     # 2.7 مللي ثانية
    #pulse.default.format   = F32
    #pulse.default.position = [ FL FR ]
    #pulse.idle.timeout     = 0
}
pulse.properties.rules = [
    {   matches = [ { cpu.vm.name = !null } ]
        actions = {
            update-props = {
                # سيتم تطبيق هذه التجاوزات فقط عند التشغيل في جهاز افتراضي.
                pulse.min.quantum = 1024/48000      # 22 مللي ثانية
            }
        }
    }
]

خيارات الاتصال

...
server.address = [
    "unix:native"
    # "tcp:4713"
]
...

العناوين التي يستمع عليها الخادم عند البدء. قم بإلغاء التعليق على الإدخال tcp:4713 لجعل الخادم يستمع أيضًا على مقبس tcp. هذا يعادل تحميل libpipewire-module-native-protocol-tcp.

هناك أيضًا صيغة أكثر تفصيلاً بقليل مع المزيد من الخيارات:

....
server.address = [
  {  address = "tcp:4713"             # العنوان
     max-clients = 64                 # الحد الأقصى لعدد العملاء
     listen-backlog = 32              # قائمة انتظار الخادم
     client.access = "restricted"     # أذونات العملاء
  }
....

استخدم client.access لاستخدام إحدى طرق الوصول لتقييد الأذونات الممنوحة للعملاء المتصلين عبر هذا العنوان.

افتراضيًا، يتم منح الوصول إلى الشبكة أذونات "مقيّدة". مدير الجلسة مسؤول عن تعيين الأذونات للعملاء ذوي الأذونات المقيدة (عادةً أذونات القراءة فقط).

server.dbus-name       = "org.pulseaudio.Server"

اسم DBus الذي سيتم حجزه للخادم. إذا كان لديك خوادم متعددة، قد ترغب في تغيير الاسم.

pulse.allow-module-loading = true

افتراضيًا، يُسمح للعملاء بتحميل الوحدات وإلغاء تحميلها. يمكنك تعطيل هذه الميزة بهذا الخيار.

خيارات التخزين المؤقت للتشغيل

pulse.min.req = 128/48000              # 2.7 مللي ثانية

الحد الأدنى لكمية البيانات المطلوب للعملاء. سيتم تقييد القيم المطلوبة من العميل إلى هذه القيمة. تقليل هذه القيمة مع tlength يمكن أن يقلل من زمن الاستجابة إذا أراد العميل ذلك، ولكنه يزيد من استخدام وحدة المعالجة المركزية.

pulse.default.req = 960/48000          # 20 مللي ثانية

الكمية الافتراضية للبيانات المطلوبة للعملاء. إذا لم يحدد العميل أي قيمة، فسيتم استخدام هذه القيمة الافتراضية. تقليل هذه القيمة مع tlength يمكن أن يقلل من زمن الاستجابة ولكنه يزيد من استخدام وحدة المعالجة المركزية.

pulse.default.tlength = 96000/48000    # 2 ثانية

الكمية المستهدفة للبيانات للتخزين المؤقت على جانب الخادم. إذا لم يحدد العميل قيمة، فسيتم استخدام هذه القيمة الافتراضية. القيم الأقل يمكن أن تقلل من زمن الاستجابة.

خيارات التخزين المؤقت للتسجيل

pulse.min.frag = 128/48000             # 2.7 مللي ثانية

الحد الأدنى لحجم المخزن المؤقت لالتقاط البيانات قبل إرسالها إلى العميل. سيتم تقييد القيمة المطلوبة من العميل إلى هذه القيمة. تقليل هذه القيمة يمكن أن يقلل من زمن الاستجابة على حساب زيادة استخدام وحدة المعالجة المركزية.

pulse.default.frag = 96000/48000       # 2 ثانية

الحجم الافتراضي للمخزن المؤقت لالتقاط البيانات قبل إرسالها إلى العميل. إذا لم يحدد العميل أي قيمة، فسيتم استخدام هذه القيمة الافتراضية. تقليل هذه القيمة يمكن أن يقلل من زمن الاستجابة على حساب زيادة استخدام وحدة المعالجة المركزية.

خيارات الجدولة

pulse.min.quantum = 128/48000          # 2.7 مللي ثانية

الكمية الدنيا (حجم المخزن المؤقت بالعينات) للاستخدام مع عملاء pulseaudio. يتم حساب هذه القيمة بناءً على frag و req/tlength لتدفقات التسجيل والتشغيل على التوالي ويتم تقييدها بعد ذلك بهذه القيمة لضمان عدم طلب عملاء pulseaudio كميات صغيرة جدًا. قد يقلل تقليل هذه القيمة من زمن الاستجابة على حساب زيادة استخدام وحدة المعالجة المركزية.

خيارات التنسيق

pulse.default.format = F32

ستستخدم بعض الوحدات هذا التنسيق افتراضيًا عندما لا يتم تقديم أي تنسيق آخر. هذا يعادل خيار PulseAudio الافتراضي-sample-format في /etc/pulse/daemon.conf.

pulse.default.position = [ FL FR ]

ستستخدم بعض الوحدات خريطة القنوات هذه (مع عدد قنواتها) افتراضيًا. هذا يعادل خيارات PulseAudio الافتراضي-sample-channels و default-channel-map في /etc/pulse/daemon.conf.

خيارات التصحيحات (Quirk)

pulse.fix.format = "S16LE"

عندما يستخدم تدفق علامة FIX_FORMAT، قم بتثبيت التنسيق على هذه القيمة. عادةً ما يتم تثبيت التنسيق على sink/source الذي يتصل به التدفق. عند تعيين تنسيق غير صالح (null أو "")، يتم تجاهل علامة FIX_FORMAT.

pulse.fix.rate = 48000

عندما يستخدم تدفق علامة FIX_RATE، قم بتثبيت معدل العينة على هذه القيمة. عادةً ما يتم تثبيت المعدل على sink/source الذي يتصل به التدفق. عند تعيين معدل 0، يتم تجاهل علامة FIX_RATE.

pulse.fix.position = "[ FL FR ]"

عندما يستخدم تدفق علامة FIX_CHANNELS، قم بتثبيت القنوات على هذه القيمة. عادةً ما يتم تثبيت القنوات على sink/source الذي يتصل به التدفق. عند تعيين موضع غير صالح (null أو "")، يتم تجاهل علامة FIX_CHANNELS.

pulse.idle.timeout = 0

لا ترسل بعض العملاء بيانات عندما يجب أن يفعلوا ويسببون نقصًا (underruns). عند تعيين هذا الخيار، سيتم تعيين هؤلاء العملاء إلى وضع الإيقاف المؤقت إذا تعرضوا لنقص لمدة معينة بالثواني. هذا يضمن إمكانية تعليق sinks وتوفير طاقة البطارية. عند استئناف العميل، سيتم إلغاء الإيقاف المؤقت مرة أخرى. القيمة 0 تقوم بتعطيل هذه الميزة.
