libressl-EVP_PKEY_asn1_get_count(3)

الحزمة: libressl

التراخيص: ISC, custom:OpenSSL

EVP_PKEY_asn1_get_count(3) دليل المكتبة EVP_PKEY_asn1_get_count(3)

الاسم

EVP_PKEY_asn1_get_count، EVP_PKEY_asn1_get0، EVP_PKEY_get0_asn1، EVP_PKEY_asn1_find، EVP_PKEY_asn1_find_str، EVP_PKEY_asn1_get0_info — تعداد طرق ASN.1 للمفاتيح العامة

الملخص

المكتبة «libcrypto»
#include <openssl/evp.h>

int
EVP_PKEY_asn1_get_count(void);

const EVP_PKEY_ASN1_METHOD *
EVP_PKEY_asn1_get0(int idx);

const EVP_PKEY_ASN1_METHOD *
EVP_PKEY_get0_asn1(const EVP_PKEY *pkey);

const EVP_PKEY_ASN1_METHOD *
EVP_PKEY_asn1_find(ENGINE **engine, int type);

const EVP_PKEY_ASN1_METHOD *
EVP_PKEY_asn1_find_str(ENGINE **engine, const char *str, int len);

int
EVP_PKEY_asn1_get0_info(int *ppkey_id, int *pkey_base_id, int *ppkey_flags, const char **pinfo, const char **ppem_str, const EVP_PKEY_ASN1_METHOD *ameth);

الوصف

EVP_PKEY_asn1_get_count() تُرجع عدد طرق ASN.1 للمفاتيح العامة المتاحة.

EVP_PKEY_asn1_get0() تُرجع طريقة ASN.1 للمفتاح العام idx. يجب أن تكون قيمة idx في النطاق من صفر إلى EVP_PKEY_asn1_get_count() - 1.

EVP_PKEY_asn1_find() تبحث عن الطريقة بالمعرف type (NID)، والذي يمكن أن يكون أيًا من القيم التي قد تُرجعها EVP_PKEY_base_id(3) و EVP_PKEY_id(3). إذا كان engine ليس NULL، فسيتم تعيين *engine إلى NULL.

EVP_PKEY_asn1_find_str() تبحث عن الطريقة بسلسلة نوع PEM المعطاة بالبايتات الـ len الأولى من str. إذا كان len يساوي -1، فيتم استخدام strlen(3) لـ str بدلاً من ذلك. سلاسل نوع PEM المدعومة افتراضيًا مدرجة في صفحة دليل EVP_PKEY_base_id(3). تمامًا مثل EVP_PKEY_asn1_find()، إذا كان engine ليس NULL، فسيتم تعيين *engine إلى NULL.

EVP_PKEY_asn1_get0_info() تسترجع معرف المفتاح العام كما تُرجعه EVP_PKEY_id(3)، ومعرف المفتاح الأساسي كما تُرجعه EVP_PKEY_base_id(3) (كلاهما معرفات NIDs)، وأي علامات، ومؤشرات داخلية مملوكة لـ ameth تشير إلى سلسلة وصف الطريقة وسلسلة نوع PEM الخاصة بها.

يمكن أن تحدث بتات العلامات التالية، مجمعة معًا في *ppkey_flags:

ASN1_PKEY_ALIAS
كائن ameth هذا يعمل كاسم مستعار لكائن EVP_PKEY_ASN1_METHOD آخر ولن يتم إرجاعه أبدًا من EVP_PKEY_asn1_find() أو EVP_PKEY_asn1_find_str().
ASN1_PKEY_DYNAMIC
هذه العلامة غير مستخدمة. كان يمكن سابقًا استخدامها لتمييز كائن ameth كمخصص ديناميكيًا.
ASN1_PKEY_SIGPARAM_NULL
إذا استخدم سياق التوقيع ctx مفتاحًا خاصًا EVP_PKEY مرتبطًا بـ ameth هذا، فأخبر ASN1_item_sign_ctx(3) باستخدام نوع معامل V_ASN1_NULL بدلاً من الافتراضي V_ASN1_UNDEF عند ترميز كائنات AlgorithmIdentifier ASN.1 باستخدام X509_ALGOR_set0(3). على وجه الخصوص، يتم استخدام هذا لـ EVP_PKEY_RSA.

قيم الإرجاع

EVP_PKEY_asn1_get_count() تُرجع عدد طرق المفاتيح العامة المتاحة.

EVP_PKEY_asn1_get0() تُرجع طريقة مفتاح عام أو NULL إذا كان idx خارج النطاق.

EVP_PKEY_get0_asn1() تُرجع طريقة المفتاح العام المستخدمة بواسطة pkey.

EVP_PKEY_asn1_find() و EVP_PKEY_asn1_find_str() تُرجعان طريقة مفتاح عام مطابق أو NULL إذا لم يتم العثور على تطابق.

EVP_PKEY_asn1_get0_info() تُرجع 1 عند النجاح أو 0 عند الفشل.

انظر أيضًا

EVP_PKEY_base_id(3)، EVP_PKEY_new(3)
