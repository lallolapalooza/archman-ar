libressl-X509_STORE_CTX_set_flags(3)

الحزمة: libressl

التراخيص: ISC, custom:OpenSSL

X509_STORE_CTX_SET_FLAGS(3) دليل وظائف المكتبة X509_STORE_CTX_SET_FLAGS(3)

الاسم

X509_STORE_CTX_set_flags، X509_STORE_CTX_set_time، X509_STORE_CTX_set_depth، X509_STORE_CTX_set_trust، X509_STORE_CTX_set_purpose، X509_STORE_CTX_get0_param، X509_STORE_CTX_set0_param، X509_STORE_CTX_set_default — تهيئة معلمات X509_STORE_CTX

الملخص

مكتبة “libcrypto”
#include <openssl/x509_vfy.h>

void
X509_STORE_CTX_set_flags(X509_STORE_CTX *ctx, unsigned long flags);

void
X509_STORE_CTX_set_time(X509_STORE_CTX *ctx, unsigned long dummy, time_t time);

void
X509_STORE_CTX_set_depth(X509_STORE_CTX *ctx, int depth);

int
X509_STORE_CTX_set_trust(X509_STORE_CTX *ctx, int trust);

int
X509_STORE_CTX_set_purpose(X509_STORE_CTX *ctx, int purpose);

X509_VERIFY_PARAM *
X509_STORE_CTX_get0_param(X509_STORE_CTX *ctx);

void
X509_STORE_CTX_set0_param(X509_STORE_CTX *ctx, X509_VERIFY_PARAM *param);

int
X509_STORE_CTX_set_default(X509_STORE_CTX *ctx, const char *name);

الوصف

تعمل هذه الوظائف على كائن X509_VERIFY_PARAM المستخدم بواسطة ctx. عادة، يتم استدعاء X509_STORE_CTX_init(3) على ctx قبل هذه الوظائف، و X509_verify_cert(3) بعدها.

X509_STORE_CTX_set_flags() تقوم بتعيين علامات معلمة التحقق الداخلية إلى flags. انظر X509_VERIFY_PARAM_set_flags(3) لوصف علامات التحقق.

X509_STORE_CTX_set_time() تقوم بتعيين time التحقق باستخدام X509_VERIFY_PARAM_set_time(3). يتم تجاهل الوسيط dummy.

X509_STORE_CTX_set_depth() تقوم بتعيين depth التحقق القصوى باستخدام X509_VERIFY_PARAM_set_depth(3). هذا هو الحد الأقصى لعدد شهادات CA غير الموثوقة التي يمكن أن تظهر في سلسلة.

X509_STORE_CTX_set_trust() تقوم بتعيين معرف trust الذي يمكن تعيينه أيضًا باستخدام X509_VERIFY_PARAM_set_trust(3). إذا كان الوسيط trust هو 0 أو غير صالح أو تم تعيين معرف الثقة بالفعل إلى قيمة غير صفرية في الكائن X509_VERIFY_PARAM، فلا يحدث أي إجراء.

X509_STORE_CTX_set_purpose() تقوم بتعيين معرف purpose الذي يمكن تعيينه أيضًا باستخدام X509_VERIFY_PARAM_set_purpose(3). إذا كان الوسيط purpose هو 0 أو حدث أي خطأ، فلا يتم تغيير أي شيء.

فيما يلي، يسمى معرف الثقة الموجود في الكائن X509_PURPOSE المرتبط بـ purpose "الثقة المرتبطة".

تفشل الوظيفة إذا كان الوسيط purpose أو الثقة المرتبطة به غير صالحين ولكن ليس 0؛ بخلاف ذلك، تقوم X509_STORE_CTX_set_purpose() أيضًا بما يعادل استدعاء X509_STORE_CTX_set_trust() مع الثقة المرتبطة.

إذا تم تعيين معرف الغرض بالفعل إلى قيمة غير صفرية في الكائن X509_VERIFY_PARAM، فلن يتم تغييره، حتى لو كان الوسيط purpose صالحًا أيضًا.

X509_STORE_CTX_get0_param() تقوم باسترداد مؤشر داخلي لمعلمات التحقق المرتبطة بـ ctx.

X509_STORE_CTX_set0_param() تقوم بتعيين مؤشر معلمة التحقق الداخلي إلى param. بعد هذا الاستدعاء، يجب عدم استخدام param.

X509_STORE_CTX_set_default() تبحث عن طريقة التحقق الافتراضية وتعينها إلى name. يستخدم هذا الوظيفة X509_VERIFY_PARAM_lookup(3) للعثور على مجموعة معلمات مناسبة من name وينسخها باستخدام X509_VERIFY_PARAM_inherit(3).

قيم الإرجاع

X509_STORE_CTX_set_trust() ترجع 1 إذا كان الوسيط trust هو 0 أو صالحًا أو 0 إذا كان غير صالح ولكن ليس 0. قيمة الإرجاع 1 لا تعني أن معرف الثقة المخزن في الكائن X509_VERIFY_PARAM قد تم تغييره.

X509_STORE_CTX_set_purpose() ترجع 1 إذا كان كل من الوسيط purpose والثقة المرتبطة به هما 0 أو صالحين. ترجع 0 إذا كان الوسيط purpose أو الثقة المرتبطة به غير صالحين ولكن ليس 0. قيمة الإرجاع 1 لا تعني أن أي بيانات قد تم تغييرها.

X509_STORE_CTX_get0_param() ترجع مؤشرًا إلى هيكل X509_VERIFY_PARAM أو NULL إذا حدث خطأ.

X509_STORE_CTX_set_default() ترجع 1 للنجاح أو 0 إذا حدث خطأ.

الأخطاء

يمكن استرداد التشخيصات التالية باستخدام ERR_get_error(3)، ERR_GET_REASON(3)، و ERR_reason_error_string(3):

X509_R_UNKNOWN_TRUST_ID "معرف ثقة غير معروف"
تم استدعاء X509_STORE_CTX_set_trust() بوسيط trust غير صالح ولكن ليس 0. قد ترجع تطبيقات أخرى أيضًا هذا عند استدعاء X509_STORE_CTX_set_purpose() بوسيط purpose مع ثقة مرتبطة غير صالحة.
X509_R_UNKNOWN_PURPOSE_ID "معرف غرض غير معروف"
الوسيط purpose غير صالح ولكن ليس 0.

الوظائف الأخرى لا تقدم أي تشخيصات.
