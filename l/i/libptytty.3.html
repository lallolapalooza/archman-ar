الحزمة: libptytty

التراخيص: GPL

libptytty(3) LIBPTYTTY libptytty(3)

الاسم

libptytty - معالجة مستقلة وآمنة لأنظمة pty/tty وقواعد بيانات utmp/wtmp/lastlog

الخلاصة

cc ... -lptytty
#include <libptytty.h>
// C++
ptytty *pty = ptytty::create ();
if (!pty->get ())
  // خطأ في تخصيص pty
if (نريد utmp)
  pty->login (process_pid, 0, "remote.host");
else if (نريد utmp و wtmp/lastlog)
  pty->login (process_pid, 1, "remote.host");
// لقد انتهينا منه
delete pty;
// C
PTYTTY pty = ptytty_create ();
if (!ptytty_get (pty))
  // خطأ في تخصيص pty
if (نريد utmp)
  ptytty_login (pty, process_pid, 0, "remote.host");
else if (نريد utmp و wtmp/lastlog)
  ptytty_login (pty, process_pid, 1, "remote.host");
// لقد انتهينا منه
ptytty_delete (pty);

انظر أيضًا إلى دليل eg/، والذي يحتوي حاليًا على الملف c-sample.c الذي يقوم بإنشاء shell تسجيل دخول من C باستخدام libptytty.

الوصف

Libptytty هي مكتبة صغيرة توفر إدارة pseudo-tty بطريقة مستقلة عن نظام التشغيل. تم إنشاؤها من الإحباط بسبب الاختلافات العديدة في معالجة pty/tty في أنظمة التشغيل المختلفة للاستخدام داخل "rxvt-unicode".

بالإضافة إلى توفير مجرد إدارة pty/tty، فإنها توفر أيضًا دعمًا لقاعدة بيانات الجلسات (تحديثات utmp و wtmp/lastlog الاختيارية لجلسات تسجيل الدخول).

كما أنها تدعم الانقسام (fork'ing) بعد بدء التشغيل وإنزال الامتيازات في العملية المتصلة، لذلك في حالة تعرض العملية المتصلة للاختراق من قبل المستخدم الذي يبدأ البرنامج، هناك ما يمكن كسبه أقل، حيث أن عملية المساعد فقط هي التي تعمل بامتيازات (مثل setuid/setgid)، مما يقلل مساحة الهجوم بشكل كبير.

Libptytty مكتوبة بلغة C++، لكنها توفر أيضًا واجهة برمجة تطبيقات C فقط.

التثبيت

Libptytty تستخدم "CMake" كنظام بناء. لبناء libptytty، قم بتثبيت "CMake" وقم بتشغيل الأوامر التالية من دليل مصدر libptytty أو من دليل بناء منفصل:

cmake -DCMAKE_INSTALL_PREFIX=<prefix> -DBUILD_SHARED_LIBS=ON <path/to/libptytty>
cmake --build .
cmake --install .

اعتبارات الأمان

من الأهمية بمكان أن تقرأ على الأقل الفقرة التالية

إذا كنت تكتب برنامجًا نموذجيًا يشبه الطرفية ويريد فقط pty واحدًا أو أكثر، فيجب عليك استدعاء الدالة "ptytty::init ()" (باللغة C: دالة "ptytty_init ()") كأول شيء في برنامجك:

int main (int argc, char *argv[])
{
   // لا تفعل شيئًا هنا
   ptytty::init ();
   // في C: ptytty_init ();
   // تهيئة، تحليل الوسائط، إلخ.
}

يقوم هذا بفحص ما إذا كان البرنامج يعمل بصلاحيات setuid أو setgid. إذا كان الأمر كذلك، فسيقوم بإنشاء عملية مساعد وإسقاط الامتيازات.

تحتاج بعض البرامج إلى تحكم أدق في ما إذا ومتى يتم بدء هذه العملية المساعدة، وإذا وكيفية إسقاط الامتيازات. بالنسبة لتلك البرامج، تكون الدوال "ptytty::use_helper" و "ptytty::drop_privileges" (وربما "ptytty::sanitise_stdfd") أكثر فائدة.

واجهة C++: الصنف ptytty

دوال ثابتة

ptytty::init ()
الطريقة الافتراضية لتهيئة libptytty. يجب استدعاؤها فورًا كأول شيء في دالة "main"، أو قبل ذلك، على سبيل المثال، أثناء وقت الإنشاء الثابت. كلما كان ذلك أبكر، كان أفضل.

تستدعي هذه الدالة "sanitise_stdfd" ثم تتحقق مما إذا كان البرنامج يعمل بصلاحيات setuid/setgid وإذا كان الأمر كذلك، تقوم بإنشاء عملية مساعد لإدارة pty/tty. ثم تقوم بإسقاط الامتيازات بالكامل، بحيث يعمل البرنامج الفعلي بدون صلاحيات setuid/setgid.

عند الفشل، تطرح هذه الدالة استثناء "ptytty_error".

ptytty::use_helper ()
يحاول بدء عملية مساعد تحتفظ بالامتيازات حتى عندما لا تفعل العملية المتصلة ذلك. يتم استدعاء هذا عادةً من "ptytty::init" عندما تكتشف أن البرنامج يعمل بصلاحيات setuid أو setgid، ولكن يمكن استدعاؤه يدويًا إذا كان إسقاط الامتيازات عند بدء التشغيل غير مناسب، أو عندما لا تعمل بصلاحيات setuid/setgid ولكنك تريد إسقاط الامتيازات (على سبيل المثال، عند العمل كخدمة تبدأها العملية الجذر).

لن تحاول هذه الدالة بدء أكثر من عملية مساعد واحدة. يمكن عادةً استخدام نفس العملية المساعدة من العملية التي تبدأها وجميع العمليات المنشقة (fork'ed) (وليس exec'ed) منها.

عند الفشل، تطرح هذه الدالة استثناء "ptytty_error".

ptytty::drop_privileges ()
يسقط الامتيازات بالكامل، أي يضبط معرف المستخدم الحقيقي والفعال والمحفوظ لمعرف المستخدم الحقيقي. مفيد للتأكد من أن العملية لا تعمل بامتيازات خاصة.

عند الفشل، تطرح هذه الدالة استثناء "ptytty_error".

ptytty::sanitise_stdfd ()
يتحقق مما إذا كانت واصفات الملفات 0 و 1 و 2 (stdin، stdout، و stderr) صالحة (مفتوحة) وإذا لم تكن كذلك، يقوم بتوصيلها بـ /dev/tty أو /dev/null إذا أمكن. هذا ضروري لأن libptytty قد ترغب في إخراج رسائل الخطأ إلى تلك الواصفات، والتي قد تكون متصلة في وقت إخراج رسالة الخطأ بشيء غير مناسب فتحه البرنامج غير المتوقع نفسه (يمكن أن يكون هذا مشكلة أمنية).

عند الفشل، تطرح هذه الدالة استثناء "ptytty_error".

bool success = ptytty::send_fd (int socket, int fd)
دالة مساعدة لإرسال واصف ملف عبر مقبس مجال يونكس. تعيد true إذا نجحت، false بخلاف ذلك. يتم عرض هذه الدالة فقط لراحتك وليست ضرورية للتشغيل العادي.
int fd = ptytty::recv_fd (int socket)
دالة مساعدة لاستقبال واصف ملف عبر مقبس مجال يونكس. تعيد واصف الملف إذا نجحت و "-1" بخلاف ذلك. يتم عرض هذه الدالة فقط لراحتك وليست ضرورية للتشغيل العادي.
ptytty *pty = ptytty::create ()
ينشئ كائن ptytty جديدًا. لا يقوم الإنشاء حتى الآن بأي شيء بخلاف تخصيص الهيكل.

يتم استخدام دالة ثابتة لأن تطبيق ptytty الفعلي يمكن أن يختلف في وقت التشغيل، لذا تحتاج إلى تسهيل إنشاء كائن ديناميكي.

أعضاء بيانات ديناميكية / متعلقة بالجلسات وطرق

int pty_fd = pty->pty
int tty_fd = pty->tty
تحتوي هذه الأعضاء على واصفات ملفات pty و tty، على التوالي. تحتوي مبدئيًا على "-1" حتى استدعاء ناجح لـ "ptytty::get".
bool success = pty->get ()
يحاول العثور على زوج pty/tty وتخصيصه وتهيئته. يعيد "true" عند النجاح.

إذا كانت العملية المساعدة قيد التشغيل وكان هناك خطأ في البروتوكول، تطرح هذه الدالة استثناء "ptytty_error".

pty->login (int cmd_pid, bool login_shell, const char *hostname)
ينشئ إدخالاً في قاعدة بيانات (قواعد بيانات) الجلسة للنظام (utmp، wtmp، lastlog). يجب أن يكون "cmd_pid" هو معرف العملية الذي يمثل الجلسة (مثل shell تسجيل الدخول)، "login_shell" يحدد ما إذا كانت الجلسة مرتبطة بتسجيل دخول، مما يؤثر على ما إذا كان سيتم إنشاء إدخالات wtmp و lastlog، و "hostname" يجب أن يحدد
