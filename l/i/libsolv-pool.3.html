السمات العامة

void *appdata

مؤشر للاستخدام المجاني من قبل مستخدم المكتبة. تحرير المجمع ببساطة يتجاهل المؤشر.

Stringpool ss

مجمع السلاسل الموحدة.

Reldep *rels

مجمع الاعتماديات العلائقية الموحدة.

int nrels

عدد الاعتماديات العلائقية المخصصة.

Repo **repos

مصفوفة مؤشرات المستودعات، مفهرسة بمعرف المستودع.

int nrepos

عدد عناصر مصفوفة المستودعات المخصصة، أي حجم مصفوفة المستودعات.

int urepos

عدد عناصر مصفوفة المستودعات المستخدمة (أي غير الصفرية).

Repo *installed

مؤشر إلى المستودع الذي يحتوي على الحزم المثبتة. لك حرية قراءة هذه السمة، ولكن يجب استخدام pool_set_installed() إذا كنت ترغب في تغييرها.

Solvable *solvables

مصفوفة كائنات Solvable.

int nsolvables

عدد كائنات Solvable، أي حجم مصفوفة solvables. لاحظ أن المصفوفة قد تحتوي على solvables محررة، في هذه الحالة سيكون مؤشر المستودع للمنفذ صفراً.

int disttype

نوع التوزيع لنظامك، على سبيل المثال DISTTYPE_DEB. لك حرية قراءة هذه السمة، ولكن يجب استخدام pool_setdisttype() إذا كنت ترغب في تغييرها.

Id *whatprovidesdata

مخزن معرّفات متعدد الأغراض يحتوي على مصفوفات معرّفات منتهية بصفر. pool_whatprovides() تُرجع إزاحة في هذه البيانات.

Map *considered

خريطة بت اختيارية يمكنها جعل المكتبة تتجاهل solvables. إذا تم تعيين خريطة بت، فسيتم اعتبار solvables التي تحتوي على بت معين في خريطة البت عند معرفها فقط قابلة للاستخدام.

int debugmask

قناع يحدد أي أحداث التصحيح يجب الإبلاغ عنها. تقوم pool_setdebuglevel() بتعيين هذا القناع.

Datapos pos

كائن يخزن بعض المواقع في بيانات المستودع. وظائف مثل dataiterator_set_pos() تقوم بتعيين هذا الكائن، والوصول إلى البيانات باستخدام معرف solvable زائف SOLVID_POS يستخدمه.

Queue pooljobs

قائمة انتظار يمكن تخزين وظائف محلل ثابتة فيها. تتم إضافة هذه الوظائف تلقائيًا عند استدعاء solver_solve()، وهي مفيدة لتخزين بيانات التكوين مثل الحزم التي يجب تثبيتها بتعدد الإصدارات.

الإنشاء والتدمير

Pool *pool_create();

إنشاء مثيل جديد لمجمع.

void pool_free(Pool *pool);

تحرير مجمع وجميع البيانات التي يحتوي عليها، على سبيل المثال، solvables والمستودعات والسلاسل.

التصحيح وإبلاغ الأخطاء

الثوابت

SOLV_FATAL

الإبلاغ عن الخطأ واستدعاء "exit(1)" بعد ذلك. لا يمكنك إخفاء هذا المستوى. يتم الإبلاغ إلى stderr بدلاً من stdout.

SOLV_ERROR

يستخدم للإبلاغ عن الأخطاء. يتم الإبلاغ إلى stderr بدلاً من stdout.

SOLV_WARN

يستخدم للإبلاغ عن التحذيرات.

SOLV_DEBUG_STATS

يستخدم للإبلاغ عن البيانات الإحصائية.

SOLV_DEBUG_RULE_CREATION

يستخدم للإبلاغ عن معلومات حول إنشاء المحلل للقواعد.

SOLV_DEBUG_PROPAGATE

يستخدم للإبلاغ عن معلومات حول عملية نشر قواعد الوحدة للمحلل.

SOLV_DEBUG_ANALYZE

يستخدم للإبلاغ عن معلومات حول آلية توليد القواعد المتعلمة للمحلل.

SOLV_DEBUG_UNSOLVABLE

يستخدم للإبلاغ عن معلومات حول تعامل المحلل مع القواعد المتعارضة.

SOLV_DEBUG_SOLUTIONS

يستخدم للإبلاغ عن معلومات حول إنشاء المحلل لحلول لحل المشاكل.

SOLV_DEBUG_POLICY

يستخدم للإبلاغ عن معلومات حول بحث المحلل عن حل أمثل.

SOLV_DEBUG_RESULT

تستخدمها وظائف التصحيح لإخراج النتائج.

SOLV_DEBUG_JOB

يستخدم للإبلاغ عن معلومات حول عملية توليد قواعد الوظائف.

SOLV_DEBUG_SOLVER

يستخدم للإبلاغ عن معلومات حول ما يفعله المحلل حاليًا.

SOLV_DEBUG_TRANSACTION

يستخدم للإبلاغ عن معلومات حول عملية توليد وترتيب المعاملات.

SOLV_DEBUG_TO_STDERR

اكتب رسائل التصحيح إلى stderr بدلاً من stdout.

الوظائف

void pool_debug(Pool *pool, int type, const char *format, ...);

الإبلاغ عن رسالة من النوع type. يمكنك تصفية رسائل التصحيح عن طريق تعيين قناع تصحيح.

void pool_setdebuglevel(Pool *pool, int level);

تعيين قناع تصحيح محدد مسبقًا. المستوى الأعلى يعني عمومًا المزيد من البتات المعينة في القناع، وبالتالي طباعة المزيد من الرسائل.

void pool_setdebugmask(Pool *pool, int mask);

تعيين قناع التصحيح لتصفية رسائل التصحيح.

int pool_error(Pool *pool, int ret, const char *format, ...);

تعيين سلسلة الخطأ للمجمع. يتم استخدام القيمة ret ببساطة كقيمة إرجاع للوظيفة حتى تتمكن من كتابة كود مثل return pool_error(...);. إذا كان قناع التصحيح يحتوي على بت SOLV_ERROR، فسيتم أيضًا استدعاء pool_debug() بالرسالة والنوع SOLV_ERROR.

extern char *pool_errstr(Pool *pool);

إرجاع سلسلة الخطأ الحالية المخزنة في المجمع. كما هو الحال مع قيمة errno في libc، تكون السلسلة ذات مغزى فقط بعد أن ترجع وظيفة خطأ.

void pool_setdebugcallback(Pool *pool, void (*debugcallback)(Pool *, void *data, int type, const char *str), void *debugcallbackdata);

تعيين وظيفة رد اتصال تصحيح مخصصة. بدلاً من الكتابة إلى stdout أو stderr، سيتم استدعاء وظيفة رد الاتصال.

تكوين المجمع

الثوابت

DISTTYPE_RPM

يستخدم للأنظمة التي تستخدم rpm كمدير حزم منخفض المستوى.

DISTTYPE_DEB

يستخدم للأنظمة التي تستخدم dpkg كمدير حزم منخفض المستوى.

DISTTYPE_ARCH

يستخدم للأنظمة التي تستخدم مدير حزم Arch Linux.

DISTTYPE_HAIKU

يستخدم للأنظمة التي تستخدم حزم Haiku.

POOL_FLAG_PROMOTEEPOCH

ترقية حقبة الاعتمادية المقدمة إلى الاعتمادية المطلوبة إذا لم تكن تحتوي على حقبة. استخدم في وقت ما في إصدارات rpm القديمة، لا تحتاج الأنظمة الحديثة إلى هذا أبدًا.

POOL_FLAG_FORBIDSELFCONFLICTS

حظر تثبيت الحزم التي تتعارض مع نفسها. تسمح Debian دائمًا بالحزم المتعارضة مع نفسها، وكان rpm يحظرها ولكنه تحول إلى السماح بها مؤخرًا.

POOL_FLAG_OBSOLETEUSESPROVIDES

جعل الاعتمادية من نوع obsolete تتطابق مع provides بدلاً من مجرد اسم وإصدار الحزم. استخدمت إصدارات rpm القديمة جدًا الاسم/الإصدار، ثم تم تبديلها إلى provides ولاحقًا تم تبديلها مرة أخرى إلى الاسم/الإصدار فقط.

POOL_FLAG_IMPLICITOBSOLETEUSESPROVIDES

يعتبر obsolescence الضمني آلية داخلية لإزالة الحزمة القديمة عند التحديث. ث

إدارة المجمع

الثوابت

POOL_FLAG_OBSOLETEUSESCOLORS

تنفيذ multilib الخاص بـ Rpm (المستخدم في RedHat و Fedora) يميز بين حزم 32 بت و 64 بت (المصطلح هو أن لديها لونًا مختلفًا). إذا تم تعيين obsoleteusescolors ، فلن تلغي الحزم ذات الألوان المختلفة بعضها البعض.

POOL_FLAG_IMPLICITOBSOLETEUSESCOLORS

نفس POOL_FLAG_OBSOLETEUSESCOLORS ، ولكن يتم استخدامه لمعرفة ما إذا كان يمكن تثبيت الحزم من نفس الاسم بالتوازي. بالنسبة لأنظمة Fedora الحالية ، يجب أن يكون POOL_FLAG_OBSOLETEUSESCOLORS خاطئًا ويجب أن يكون POOL_FLAG_IMPLICITOBSOLETEUSESCOLORS صحيحًا (هذا هو الافتراضي إذا تم تعريف FEDORA عند تجميع libsolv).

POOL_FLAG_NOINSTALLEDOBSOLETES

تأخذ الإصدارات الجديدة من rpm في الاعتبار ما تلغيه الحزم المثبتة عند التحقق من الاعتماديات ، وبالتالي قد لا تتمكن من تثبيت حزمة تلغيها حزمة أخرى مثبتة ، ما لم تقم أيضًا بمسح الحزمة الأخرى.

POOL_FLAG_HAVEDISTEPOCH

أضافت Mandriva حقلاً جديدًا يسمى distepoch يتم التحقق منه في مقارنة الإصدارات إذا كانت epoch / version / release لحزمتين متطابقة.

POOL_FLAG_NOOBSOLETESMULTIVERSION

إذا تم تثبيت حزمة في وضع multiversionmode ، اعتاد rpm على تجاهل كل من إلغاءات الضمنية واعتماد إلغاء الحزمة. تم تغيير هذا إلى تجاهل إلغاءات الضمنية فقط ، وبالتالي يمكنك تثبيت إصدارات متعددة من نفس الاسم ، ولكن لا تزال الحزم الملغاة تتم إزالتها.

POOL_FLAG_ADDFILEPROVIDESFILTERED

اجعل طريقة addfileprovides تضيف الملفات فقط من المواقع القياسية (أي الدلائل "bin" و "etc"). هذا مفيد إذا كان لديك عدد قليل فقط من الحزم التي تستخدم تبعيات ملفات غير قياسية ، ولكنك لا تزال ترغب في السرعة التي تولدها addfileprovides ().

الوظائف

int pool_setdisttype(Pool *pool, int disttype);

قم بتعيين نوع الحزمة لنظامك. يتم استخدام disttype على سبيل المثال لتحديد دلالات مقارنة الحزم. يجب أن يتطابق disttype الافتراضي لـ Libsolv مع مدير الحزم الخاص بنظامك ، لذلك تحتاج فقط إلى استخدام هذه الدالة إذا كنت ترغب في استخدام المكتبة لحل مشاكل التعبئة لأنظمة مختلفة. ترجع الدالة disttype القديم عند النجاح ، و -1 إذا كان disttype الجديد غير مدعوم. لاحظ أن أي استدعاءات لـ pool_setarch و pool_setarchpolicy يجب أن تأتي بعد استدعاء pool_setdisttype ، حيث أنها تستفيد من معرف بنية noarch/any/all.

int pool_set_flag(Pool *pool, int flag, int value);

قم بتعيين علامة إلى قيمة جديدة. ترجع القيمة القديمة للعلامة.

int pool_get_flag(Pool *pool, int flag);

احصل على قيمة علامة المجمع. انظر قسم الثوابت لمعنى العلامات.

void pool_set_rootdir(Pool *pool, const char *rootdir);

قم بتعيين دليل جذر محدد. تدعم بعض وظائف المكتبة علامة تخبر الدالة بوضع بادئة لـ rootdir لأسماء الملفات والأدلة.

const char *pool_get_rootdir(Pool *pool);

أرجع القيمة الحالية لدليل الجذر.

char *pool_prepend_rootdir(Pool *pool, const char *dir);

أضف بادئة لدليل الجذر إلى سلسلة الوسيط dir. تم تخصيص السلسلة المرجعة حديثًا وتحتاج إلى تحريرها بعد الاستخدام.

char *pool_prepend_rootdir_tmp(Pool *pool, const char *dir);

نفس pool_prepend_rootdir ، ولكن يستخدم مساحة مؤقتة للمجمع للتخصيص.

void pool_set_installed(Pool *pool, Repo *repo);

قم بتعيين المستودع الذي يجب التعامل معه على أنه مستودع "مثبت"، أي المستودع الذي يحتوي على معلومات حول الحزم المثبتة.

void pool_set_languages(Pool *pool, const char **languages, int nlanguages);

قم بتعيين لغة نظامك. توفر المكتبة وظائف بحث ترجع سلاسل نصية محلية ، على سبيل المثال لوصف الحزم. يمكنك تعيين مصفوفة من اللغات لتوفير آلية احتياطية إذا لم تكن لغة ما متاحة.

void pool_setarch(Pool *pool, const char *arch);

قم بتعيين بنية نظامك. يتم استخدام البنية لتحديد الحزم القابلة للتثبيت والحزم التي لا يمكن تثبيتها. الوسيط arch هو عادةً قيمة "machine" لاستدعاء نظام "uname".

void pool_setarchpolicy(Pool *, const char *);

قم بتعيين سياسة البنية لنظامك. هذا هو الإصدار العام لـ pool_setarch (في الواقع ، يستدعي pool_setarch pool_setarchpolicy داخليًا). انظر القسم حول سياسات البنية لمزيد من المعلومات.

void pool_addvendorclass(Pool *pool, const char **vendorclass);

أضف فئة معادلة بائع جديدة إلى النظام. تحدد فئة معادلة البائع ما إذا كان يمكن استبدال حزمة مثبتة من بائع واحد بحزمة قادمة من بائع مختلف. يجب أن يكون الوسيط vendorclass مصفوفة من السلاسل النصية المنتهية بـ NULL. انظر القسم حول سياسات البائع لمزيد من المعلومات.

void pool_setvendorclasses(Pool *pool, const char **vendorclasses);

قم بتعيين جميع معادلات البائع المسموح بها. يجب أن تكون الوسيط vendorclasses مصفوفة منتهية بـ NULL تتكون من جميع الفئات المسموح بها مدمجة. يجب أن تنتهي كل فئة نفسها بـ NULL ، وبالتالي تنتهي الفئة الأخيرة بعنصرين NULL ، أحدهما لإنهاء الفئة والآخر لإنهاء قائمة الفئات.

void pool_set_custom_vendorcheck(Pool *pool, int (*vendorcheck)(Pool *, Solvable *, Solvable *));

حدد آلية تحقق مخصصة للبائع. يمكنك استخدام هذا إذا كانت آلية فئة معادلة البائع الداخلية لـ libsolv لا تتناسب مع احتياجاتك.

void pool_setloadcallback(Pool *pool, int (*cb)(Pool *, Repodata *, void *), void *loadcbdata);

حدد دالة رد اتصال يتم استدعاؤها عند الحاجة إلى تحميل بيانات وصفية للمستودع عند الطلب. انظر القسم حول التحميل عند الطلب في دليل libsolv-repodata.

void pool_setnamespacecallback(Pool *pool, Id (*cb)(Pool *, void *, Id, Id), void *nscbdata);

حدد دالة رد اتصال لتنفيذ دعم مساحة أسماء مخصصة. انظر القسم حول تبعيات مساحة الأسماء.

إدارة مجمع المعرف

الثوابت

ID_EMPTY

معرف السلسلة الفارغة ، وهو دائمًا المعرف 1.

REL_LT

يمثل علاقة "<".

REL_EQ

يمثل علاقة "=" .

REL_GT

يمثل علاقة ">" . يمكنك استخدام مجموعات من REL_GT و REL_EQ و REL_LT التي تم وضع علامة OR عليها لإنشاء أي علاقة تريدها.

REL_AND

عملية AND منطقية ، يمكن أن تكون أجزاء "name" و "evr" من العلاقة تبعيتين فرعيتين. يجب أن تتطابق الحزم مع كلا جزأي التبعية.

REL_OR

عملية OR منطقية ، يمكن أن تكون أجزاء "name" و "evr" من العلاقة تبعيتين فرعيتين. يمكن للحزم مطابقة أي جزء من التبعية.

REL_WITH

مثل REL_AND ، ولكن يجب أن تتطابق الحزم مع كلا التبعيتين في وقت واحد. انظر القسم حول التبعيات المنطقية لمزيد من المعلومات.

REL_NAMESPACE

علاقة مساحة أسماء خاصة. انظر القسم حول تبعيات مساحة الأسماء لمزيد من المعلومات.

REL_ARCH

تبعية تصفية بنية. جزء "name" من العلاقة هو تبعية فرعية ، وجزء "evr" هو معرف بنية يجب أن تمتلكها الحزم المطابقة (لاحظ أن هذا تطابق تام مع تجاهل سياسات البنية).

REL_FILECONFLICT

تبعية تعارض ملفات داخلية تستخدم لتمثيل تعارضات الملفات. انظر دالة pool_add_fileconflicts_deps().

REL_COND

تبعية شرطية ، يتم النظر في التبعية الفرعية "name" فقط إذا تم الوفاء بالتبعية الفرعية "evr". انظر القسم حول التبعيات المنطقية لمزيد من المعلومات.

REL_UNLESS

تبعية شرطية ، يتم النظر في التبعية الفرعية "name" فقط إذا لم يتم الوفاء بالتبعية الفرعية "evr". انظر القسم حول التبعيات المنطقية لمزيد من المعلومات.

REL_COMPAT

تبعية متوافقة تستخدم في Haiku لتمثيل نطاقات الإصدارات. جزء "name" هو الإصدار الفعلي ، وجزء "evr" هو إصدار التوافق العكسي.

REL_KIND

تبعية زائفة تحد من الحلول إلى نوع معين. من المتوقع أن يكون النوع بادئة لاسم الحل ، على سبيل المثال "patch:foo" سيكون من النوع "patch". يتم دعم "REL_KIND" فقط في وظائف التحديد.

REL_MULTIARCH

تعليق debian multiarch. القيمة الأكثر شيوعًا لجزء "evr" هي "any".

REL_ELSE

جزء else من تبعية "REL_COND" أو "REL_UNLESS". انظر القسم حول التبعيات المنطقية.

REL_ERROR

تبعية غير صالحة. هذا مفيد لتشفير أخطاء تحليل التبعيات.

الوظائف

وظائف مساعدة

char *pool_alloctmpspace(Pool *pool, int len);

تخصيص مساحة في منطقة الذاكرة المؤقتة للمجمع. لهذه المساحة عمر محدود، وسيتم تحريرها تلقائيًا بعد عدد ثابت (حالياً 16) من مكالمات `pool_alloctmpspace()` الأخرى.

void pool_freetmpspace(Pool *pool, const char *space);

إعادة المساحة التي تم تخصيصها باستخدام `pool_alloctmpspace()` إلى النظام. ليس عليك استخدام هذه الوظيفة، حيث يتم استعادة المساحة تلقائيًا، ولكن يمكن أن تكون مفيدة لتمديد عمر مؤشرات أخرى إلى منطقة الذاكرة المؤقتة للمجمع.

const char *pool_bin2hex(Pool *pool, const unsigned char *buf, int len);

تحويل بعض البيانات الثنائية إلى سداسي عشري، وإرجاع سلسلة محرفية مخصصة في منطقة الذاكرة المؤقتة للمجمع.

char *pool_tmpjoin(Pool *pool, const char *str1, const char *str2, const char *str3);

دمج ثلاث سلاسل محرفية وإرجاع النتيجة في منطقة الذاكرة المؤقتة للمجمع. يمكنك استخدام وسائط `NULL` إذا كنت ترغب فقط في دمج عدد أقل من السلاسل.

char *pool_tmpappend(Pool *pool, const char *str1, const char *str2, const char *str3);

مثل `pool_tmpjoin()`، ولكن إذا كان الوسيط الأول هو آخر مساحة مخصصة في منطقة الذاكرة المؤقتة للمجمع، فسيتم استبدالها بنتيجة الدمج ولن يتم استخدام خانة ذاكرة مؤقتة جديدة. وبالتالي يمكنك دمج أكثر من ثلاث سلاسل عن طريق مزيج من استدعاء `pool_tmpjoin()` واحد واستدعاءات `pool_tmpappend()` متعددة. لاحظ أن المؤشر `str1` لم يعد قابلاً للاستخدام بعد الاستدعاء.

بحث البيانات

الثوابت

SOLVID_POS

استخدام موضع البيانات المخزن في المجمع للبحث بدلاً من البحث عن بيانات قابلة للحل.

SOLVID_META

استخدام البيانات المخزنة في القسم الوصفي للمستودع (أو منطقة بيانات المستودع) بدلاً من البحث عن بيانات قابلة للحل. هذه الثابتة لا تعمل مع وظائف البحث الخاصة بالمجمع، استخدمها لوظائف البحث الخاصة بالمستودع أو بيانات المستودع بدلاً من ذلك. تم إدراجها فقط من أجل الاكتمال.

الوظائف

const char *pool_lookup_str(Pool *pool, Id solvid, Id keyname);

إرجاع قيمة السلسلة النصية المخزنة تحت السمة `keyname` في الحل `solvid`.

unsigned long long pool_lookup_num(Pool *pool, Id solvid, Id keyname, unsigned long long notfound);

إرجاع الرقم غير المؤشر 64 بت المخزن تحت السمة `keyname` في الحل `solvid`. إذا لم يتم العثور على مثل هذا الرقم، فسيتم إرجاع قيمة الوسيط `notfound` بدلاً من ذلك.

Id pool_lookup_id(Pool *pool, Id solvid, Id keyname);

إرجاع المعرف `Id` المخزن تحت السمة `keyname` في الحل `solvid`.

int pool_lookup_idarray(Pool *pool, Id solvid, Id keyname, Queue *q);

ملء الطابور `q` بمحتويات مصفوفة المعرفات المخزنة تحت السمة `keyname` في الحل `solvid`. إرجاع "1" إذا تم العثور على مصفوفة، وإلا سيكون الطابور فارغًا وسيتم إرجاع "0".

int pool_lookup_void(Pool *pool, Id solvid, Id keyname);

إرجاع "1" إذا تم تخزين قيمة فارغة تحت السمة `keyname` في الحل `solvid`، وإلا "0".

const char *pool_lookup_checksum(Pool *pool, Id solvid, Id keyname, Id *typep);

إرجاع المجموع الاختباري المخزن تحت السمة `keyname` في الحل `solvid`. سيتم إرجاع نوع المجموع الاختباري عبر المؤشر `typep`. إذا لم يتم العثور على مثل هذا المجموع الاختباري، فسيتم إرجاع `NULL` وسيتم تعيين النوع إلى صفر. لاحظ أن النتيجة مخزنة في منطقة الذاكرة المؤقتة للمجمع.

const unsigned char *pool_lookup_bin_checksum(Pool *pool, Id solvid, Id keyname, Id *typep);

إرجاع المجموع الاختباري المخزن تحت السمة `keyname` في الحل `solvid`. إرجاع المجموع الاختباري كبيانات ثنائية، يمكنك استخدام النوع المرجع لحساب طول المجموع الاختباري. لا حاجة لمنطقة ذاكرة مؤقتة.

const char *pool_lookup_deltalocation(Pool *pool, Id solvid, unsigned int *medianrp);

هذه وظيفة بحث مساعدة لإرجاع موقع دلتا لـ دلتا rpm. نظرًا لعدم إمكانية تخزين الحلول للدلتا، يجب عليك استخدام `SOLVID_POS` كوسيط وتعيين مؤشر `datapos` للمجمع للإشارة إلى بيانات دلتا rpm صالحة.

void pool_search(Pool *pool, Id solvid, Id keyname, const char *match, int flags, int (*callback)(void *cbdata, Solvable *s, Repodata *data, Repokey *key, KeyValue *kv), void *cbdata);

قم بإجراء بحث على جميع البيانات المخزنة في المجمع. يمكنك تحديد منطقة البحث باستخدام وسيطات `solvid` و `keyname`. يمكن مطابقة القيم اختياريًا مع الوسيط `match`، استخدم `NULL` إذا كنت لا تريد هذا المطابقة. راجع صفحة دليل Dataiterator لمعرفة أوضاع المطابقة الممكنة ووسيط `flags`. لجميع القيم (المتطابقة)، يتم استدعاء دالة الاستدعاء مع وسيط الاستدعاء `cbdata` والبيانات التي تصف القيمة.

وظائف المهام والاختيار

تتكون المهمة من معرفين، `how` و `what`. يصف الجزء `how` الإجراء، وعلامات المهمة، وطريقة الاختيار بينما الجزء `what` هو مدخل لعملية الاختيار. يتكون الاختيار من طابور يضم مهام متعددة (وبالتالي يجب أن يكون عدد العناصر في الطابور مضاعفًا للرقم اثنين). راجع صفحة دليل Solver لمزيد من المعلومات حول المهام.

const char *pool_job2str(Pool *pool, Id how, Id what, Id flagmask);

تحويل مهمة إلى سلسلة نصية. مفيدة لأغراض التصحيح. يمكن استخدام `flagmask` لإخفاء علامات المهمة، استخدم "0" إذا كنت لا تريد رؤية مثل هذه العلامات، "-1" لرؤية جميع العلامات، أو مزيج من العلامات التي تريد رؤيتها.

void pool_job2solvables(Pool *pool, Queue *pkgs, Id how, Id what);

إرجاع قائمة بالحلول التي تحددها المهمة المحددة.

int pool_isemptyupdatejob(Pool *pool, Id how, Id what);

إرجاع "1" إذا كانت المهمة مهمة تحديث لا تعمل مع أي حزمة مثبتة، أي أن المهمة هي في الأساس لا شيء. يمكنك استخدام هذا لتحويل مهام التحديث التي لا تفعل شيئًا إلى مهام تثبيت (كما تفعل مديري الحزم مثل "zypper").

const char *pool_selection2str(Pool *pool, Queue *selection, Id flagmask);

تحويل اختيار إلى سلسلة نصية. مفيدة لأغراض التصحيح. راجع وظيفة `pool_job2str()` للوسيط `flagmask`.

متفرقات

void pool_freeallrepos(Pool *pool, int reuseids);

تحرير جميع المستودعات من المجمع (بما في ذلك جميع الحلول). إذا كان `reuseids` صحيحًا، فإن جميع معرفات الحلول قابلة لإعادة الاستخدام في المرة القادمة التي يتم فيها إنشاء الحلول.

void pool_clear_pos(Pool *pool);

مسح موضع البيانات المخزن في المجمع.

سياسات المعمارية

تحدد سياسة المعمارية قائمة بالمعماريات التي يمكن تثبيتها على النظام، وكذلك العلاقة بينها (أي الترتيب). يمكن تحديد المعماريات باستخدام ثلاثة أحرف مختلفة:

':'

لا توجد علاقة بين المعماريات. لا يمكن استبدال حزمة من معمارية واحدة بأخرى.

'>'

المعمارية الأولى أفضل من الثانية. يمكن استبدال حزمة مثبتة من المعمارية الثانية بواحدة من المعمارية الأولى والعكس صحيح. سيختار الحل المعمارية الأفضل إذا كانت الإصدارات متساوية.

'='

المعماريتان قابلتان للتبادل بحرية. تستخدم لتعريف أسماء مستعارة للمعمارية.

مثال على ذلك سيكون 'x86_64:i686=athlon>i586'. هذا يعني أن حزم x86_64 لا يمكن استبدالها إلا بحزم x86_64 أخرى، ويمكن استبدال حزم i686 بحزم i686 و i586 (ولكن سيتم تفضيل حزم i686)، و athlon هو اسم آخر لمعمارية i686.

يمكنك إيقاف فحوصات استبدال المعمارية باستخدام علامة `SOLVER_FLAG_ALLOW_ARCHCHANGE` الخاصة بالحل.
