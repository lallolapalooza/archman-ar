LIST(3) دليل وظائف المكتبة LIST(3)

الاسم

LIST_EMPTY, LIST_ENTRY, LIST_FIRST, LIST_FOREACH, LIST_HEAD, LIST_HEAD_INITIALIZER, LIST_INIT, LIST_INSERT_AFTER, LIST_INSERT_BEFORE, LIST_INSERT_HEAD, LIST_NEXT, LIST_REMOVE - تنفيذ قائمة مرتبطة مزدوجة

المكتبة

مكتبة C القياسية (libc, -lc)

الملخص

#include <sys/queue.h>
LIST_ENTRY(TYPE);
LIST_HEAD(HEADNAME, TYPE);
LIST_HEAD LIST_HEAD_INITIALIZER(LIST_HEAD head);
void LIST_INIT(LIST_HEAD *head);
int LIST_EMPTY(LIST_HEAD *head);
void LIST_INSERT_HEAD(LIST_HEAD *head,
                        struct TYPE *elm, LIST_ENTRY NAME);
void LIST_INSERT_BEFORE(struct TYPE *listelm,
                        struct TYPE *elm, LIST_ENTRY NAME);
void LIST_INSERT_AFTER(struct TYPE *listelm,
                        struct TYPE *elm, LIST_ENTRY NAME);
struct TYPE *LIST_FIRST(LIST_HEAD *head);
struct TYPE *LIST_NEXT(struct TYPE *elm, LIST_ENTRY NAME);
LIST_FOREACH(struct TYPE *var, LIST_HEAD *head, LIST_ENTRY NAME);
void LIST_REMOVE(struct TYPE *elm, LIST_ENTRY NAME);

الوصف

تعرّف هذه الماكرو وتعمل على القوائم المرتبطة المزدوجة.

في تعريفات الماكرو، TYPE هو اسم بنية يحددها المستخدم، والتي يجب أن تحتوي على حقل من النوع LIST_ENTRY، واسمه NAME. الوسيط HEADNAME هو اسم بنية يحددها المستخدم يجب تعريفها باستخدام الماكرو LIST_HEAD().

الإنشاء

يتم ترؤيس القائمة ببنية يتم تعريفها بواسطة الماكرو LIST_HEAD(). تحتوي هذه البنية على مؤشر واحد للعنصر الأول في القائمة. ترتبط العناصر مزدوجًا بحيث يمكن إزالة عنصر تعسفي دون اجتياز القائمة. يمكن إضافة عناصر جديدة إلى القائمة بعد عنصر موجود، أو قبل عنصر موجود، أو في بداية القائمة. يتم تعريف بنية LIST_HEAD على النحو التالي:


LIST_HEAD(HEADNAME, TYPE) head;

حيث struct HEADNAME هي البنية المراد تعريفها، و struct TYPE هو نوع العناصر التي سيتم ربطها بالقائمة. يمكن لاحقًا تعريف مؤشر إلى رأس القائمة على النحو التالي:


struct HEADNAME *headp;

(الاسمين head و headp قابلان للاختيار من قبل المستخدم.)

LIST_ENTRY() يعرف بنية تربط العناصر في القائمة.

LIST_HEAD_INITIALIZER() يقيم إلى مُهيئ للقائمة head.

LIST_INIT() تهيئة القائمة التي يشير إليها head.

LIST_EMPTY() يقيم إلى صحيح إذا لم تكن هناك عناصر في القائمة.

الإدراج

LIST_INSERT_HEAD() يُدرج العنصر الجديد elm في بداية القائمة.

LIST_INSERT_BEFORE() يُدرج العنصر الجديد elm قبل العنصر listelm.

LIST_INSERT_AFTER() يُدرج العنصر الجديد elm بعد العنصر listelm.

التجوال

LIST_FIRST() يعيد العنصر الأول في القائمة، أو NULL إذا كانت القائمة فارغة.

LIST_NEXT() يعيد العنصر التالي في القائمة، أو NULL إذا كان هذا هو الأخير.

LIST_FOREACH() يجتاز القائمة المشار إليها بواسطة head في الاتجاه الأمامي، ويعين كل عنصر بدوره إلى var.

الإزالة

LIST_REMOVE() يزيل العنصر elm من القائمة.

قيمة الإرجاع

LIST_EMPTY() تعيد قيمة غير صفرية إذا كانت القائمة فارغة، وصفر إذا كانت القائمة تحتوي على إدخال واحد على الأقل.

LIST_FIRST() و LIST_NEXT() تعيد مؤشرًا إلى البنية TYPE الأولى أو التالية، على التوالي.

LIST_HEAD_INITIALIZER() تعيد مُهيئًا يمكن تعيينه لرأس القائمة head.

المعايير

BSD.

التاريخ

4.4BSD.

أخطاء

LIST_FOREACH() لا تسمح بإزالة أو تحرير var داخل الحلقة، لأن ذلك سيتداخل مع التجوال. LIST_FOREACH_SAFE()، الموجودة في BSD ولكنها غير موجودة في glibc، تصلح هذا القيد عن طريق السماح بإزالة var بأمان من القائمة وتحريرها من داخل الحلقة دون التأثير على التجوال.

أمثلة

#include <stddef.h>
#include <stdio.h>
#include <stdlib.h>
#include <sys/queue.h>
struct entry {
    int data;
    LIST_ENTRY(entry) entries;              /* List */
};
LIST_HEAD(listhead, entry);
int
main(void)
{
    struct entry *n1, *n2, *n3, *np;
    struct listhead head;                   /* List head */
    int i;
    LIST_INIT(&head);                       /* Initialize the list */
    n1 = malloc(sizeof(struct entry));      /* Insert at the head */
    LIST_INSERT_HEAD(&head, n1, entries);
    n2 = malloc(sizeof(struct entry));      /* Insert after */
    LIST_INSERT_AFTER(n1, n2, entries);
    n3 = malloc(sizeof(struct entry));      /* Insert before */
    LIST_INSERT_BEFORE(n2, n3, entries);
    i = 0;                                  /* Forward traversal */
    LIST_FOREACH(np, &head, entries)
        np->data = i++;
    LIST_REMOVE(n2, entries);               /* Deletion */
    free(n2);
                                            /* Forward traversal */
    LIST_FOREACH(np, &head, entries)
        printf("%i\n", np->data);
                                            /* List deletion */
    n1 = LIST_FIRST(&head);
    while (n1 != NULL) {
        n2 = LIST_NEXT(n1, entries);
        free(n1);
        n1 = n2;
    }
    LIST_INIT(&head);
    exit(EXIT_SUCCESS);
}
