SLIST_HEAD(slisthead, entry) head =
    SLIST_HEAD_INITIALIZER(head);
struct slisthead *headp;		/* رأس القائمة المرتبطة برأس واحد. */
struct entry {
	...
	SLIST_ENTRY(entry) entries;	/* القائمة المرتبطة برأس واحد. */
	...
} *n1, *n2, *n3, *np;

SLIST_INIT(&head);			/* تهيئة القائمة. */

n1 = malloc(sizeof(struct entry));	/* الإدراج في الرأس. */
SLIST_INSERT_HEAD(&head, n1, entries);

n2 = malloc(sizeof(struct entry));	/* الإدراج بعد. */
SLIST_INSERT_AFTER(n1, n2, entries);

SLIST_REMOVE(&head, n2, entry, entries);/* الحذف. */
free(n2);

n3 = SLIST_FIRST(&head);
SLIST_REMOVE_HEAD(&head, entries);	/* الحذف من الرأس. */
free(n3);
					/* اجتياز أمامي. */
SLIST_FOREACH(np, &head, entries)
	np-> ...
					/* اجتياز أمامي آمن. */
SLIST_FOREACH_SAFE(np, &head, entries, np_temp) {
	np->do_stuff();
	...
	SLIST_REMOVE(&head, np, entry, entries);
	free(np);
}

while (!SLIST_EMPTY(&head)) {		/* حذف القائمة. */
	n1 = SLIST_FIRST(&head);
	SLIST_REMOVE_HEAD(&head, entries);
	free(n1);
}

طوابير الذيل المرتبطة برأس واحد

طابور ذيل مرتبط برأس واحد يتم ترأسه بواسطة هيكل معرف بواسطة الماكرو STAILQ_HEAD. يحتوي هذا الهيكل على زوج من المؤشرات، أحدهما إلى العنصر الأول في طابور الذيل والآخر إلى العنصر الأخير في طابور الذيل. ترتبط العناصر برأس واحد لتقليل استهلاك المساحة وأعباء معالجة المؤشرات إلى الحد الأدنى، مقابل تكلفة إزالة عناصر عشوائية تستغرق O(n). يمكن إضافة عناصر جديدة إلى طابور الذيل بعد عنصر موجود، أو في رأس طابور الذيل، أو في نهاية طابور الذيل. يتم تعريف هيكل STAILQ_HEAD على النحو التالي:

STAILQ_HEAD(HEADNAME, TYPE) head;

حيث HEADNAME هو اسم الهيكل الذي سيتم تعريفه، و TYPE هو نوع العناصر التي سيتم ربطها في طابور الذيل. يمكن لاحقًا تعريف مؤشر إلى رأس طابور الذيل على النحو التالي:

struct HEADNAME *headp;

(أسماء head و headp قابلة للاختيار من قبل المستخدم.)

الماكرو STAILQ_HEAD_INITIALIZER يقيم إلى مُهيئ لطابور الذيل head.

الماكرو STAILQ_CONCAT يربط طابور الذيل الذي ترأسه head2 بنهاية الطابور الذي ترأسه head1، ويزيل جميع الإدخالات من الطابور الأول.

الماكرو STAILQ_EMPTY يقيم إلى صحيح إذا لم تكن هناك عناصر في طابور الذيل.

الماكرو STAILQ_ENTRY يعرف هيكلًا يربط العناصر في طابور الذيل.

الماكرو STAILQ_FIRST يُرجع العنصر الأول في طابور الذيل أو NULL إذا كان طابور الذيل فارغًا.

الماكرو STAILQ_FOREACH يجتاز طابور الذيل المشار إليه بواسطة head في الاتجاه الأمامي، ويخصص كل عنصر بدوره إلى var.

الماكرو STAILQ_FOREACH_FROM يعمل بنفس طريقة STAILQ_FOREACH عندما يكون var NULL، وإلا فإنه يعامل var كعنصر STAILQ تم العثور عليه سابقًا ويبدأ الحلقة عند var بدلاً من العنصر الأول في STAILQ المشار إليه بواسطة head.

الماكرو STAILQ_FOREACH_SAFE يجتاز طابور الذيل المشار إليه بواسطة head في الاتجاه الأمامي، ويخصص كل عنصر بدوره إلى var. ومع ذلك، على عكس STAILQ_FOREACH()، يُسمح هنا بحذف var وكذلك تحريره من داخل الحلقة بأمان دون التأثير على الاجتياز.

الماكرو STAILQ_FOREACH_FROM_SAFE يعمل بنفس طريقة STAILQ_FOREACH_SAFE عندما يكون var NULL، وإلا فإنه يعامل var كعنصر STAILQ تم العثور عليه سابقًا ويبدأ الحلقة عند var بدلاً من العنصر الأول في STAILQ المشار إليه بواسطة head.

الماكرو STAILQ_INIT يقوم بتهيئة طابور الذيل المشار إليه بواسطة head.

الماكرو STAILQ_INSERT_HEAD يُدرج العنصر الجديد elm في رأس طابور الذيل.

الماكرو STAILQ_INSERT_TAIL يُدرج العنصر الجديد elm في نهاية طابور الذيل.

الماكرو STAILQ_INSERT_AFTER يُدرج العنصر الجديد elm بعد العنصر listelm.

الماكرو STAILQ_LAST يُرجع العنصر الأخير في طابور الذيل. إذا كان طابور الذيل فارغًا، فإن القيمة المرجعة هي NULL.

الماكرو STAILQ_NEXT يُرجع العنصر التالي في طابور الذيل، أو NULL إذا كان هذا العنصر هو الأخير.

الماكرو STAILQ_REMOVE_AFTER يحذف العنصر الذي يلي elm من طابور الذيل. على عكس STAILQ_REMOVE، هذا الماكرو لا يجتاز طابور الذيل بأكمله.

الماكرو STAILQ_REMOVE_HEAD يحذف العنصر الموجود في رأس طابور الذيل. للحصول على أقصى قدر من الكفاءة، يجب على العناصر التي يتم حذفها من رأس طابور الذيل استخدام هذا الماكرو صراحةً بدلاً من الماكرو العام STAILQ_REMOVE.

الماكرو STAILQ_REMOVE يحذف العنصر elm من طابور الذيل. يجب تجنب استخدام هذا الماكرو لأنه يجتاز القائمة بأكملها. يجب استخدام طابور ذيل مزدوج الارتباط إذا كان هذا الماكرو مطلوبًا في مسارات التعليمات البرمجية ذات الاستخدام العالي أو للعمل على طوابير ذيل طويلة.

الماكرو STAILQ_SWAP يتبادل محتويات head1 و head2.

مثال طابور ذيل مرتبط برأس واحد

STAILQ_HEAD(stailhead, entry) head =
    STAILQ_HEAD_INITIALIZER(head);
struct stailhead *headp;		/* رأس طابور ذيل مرتبط برأس واحد. */
struct entry {
	...
	STAILQ_ENTRY(entry) entries;	/* طابور الذيل. */
	...
} *n1, *n2, *n3, *np;

STAILQ_INIT(&head);			/* تهيئة الطابور. */

n1 = malloc(sizeof(struct entry));	/* الإدراج في الرأس. */
STAILQ_INSERT_HEAD(&head, n1, entries);

n1 = malloc(sizeof(struct entry));	/* الإدراج في الذيل. */
STAILQ_INSERT_TAIL(&head, n1, entries);

n2 = malloc(sizeof(struct entry));	/* الإدراج بعد. */
STAILQ_INSERT_AFTER(&head, n1, n2, entries);
					/* الحذف. */
STAILQ_REMOVE(&head, n2, entry, entries);
free(n2);
					/* الحذف من الرأس. */
n3 = STAILQ_FIRST(&head);
STAILQ_REMOVE_HEAD(&head, entries);
free(n3);
					/* اجتياز أمامي. */
STAILQ_FOREACH(np, &head, entries)
	np-> ...
					/* اجتياز أمامي آمن. */
STAILQ_FOREACH_SAFE(np, &head, entries, np_temp) {
	np->do_stuff();
	...
	STAILQ_REMOVE(&head, np, entry, entries);
	free(np);
}
					/* حذف طابور الذيل. */
while (!STAILQ_EMPTY(&head)) {
	n1 = STAILQ_FIRST(&head);
	STAILQ_REMOVE_HEAD(&head, entries);
	free(n1);
}
					/* حذف أسرع لطابور الذيل. */
n1 = STAILQ_FIRST(&head);
while (n1 != NULL) {
	n2 = STAILQ_NEXT(n1, entries);
	free(n1);
	n1 = n2;
}
STAILQ_INIT(&head);

القوائم

يتم ترأس القائمة بواسطة هيكل معرف بواسطة الماكرو LIST_HEAD. يحتوي هذا الهيكل على مؤشر واحد إلى العنصر الأول في القائمة. ترتبط العناصر ارتباطًا مزدوجًا بحيث يمكن حذف عنصر عشوائي دون اجتياز القائمة. يمكن إضافة عناصر جديدة إلى القائمة بعد عنصر موجود، أو قبل عنصر موجود، أو في رأس القائمة. يتم تعريف هيكل LIST_HEAD على النحو التالي:

LIST_HEAD(HEADNAME, TYPE) head;

حيث HEADNAME هو اسم الهيكل الذي سيتم تعريفه، و TYPE هو نوع العناصر التي سيتم ربطها في القائمة. يمكن لاحقًا تعريف مؤشر إلى رأس القائمة على النحو التالي:

struct HEADNAME *headp;

(أسماء head و headp قابلة للاختيار من قبل المستخدم.)

الماكرو LIST_HEAD_INITIALIZER يقيم إلى مُهيئ للقائمة head.

الماكرو LIST_CONCAT يربط القائمة التي ترأسها head2 بنهاية القائمة التي ترأسها head1، ويزيل جميع الإدخالات من القائمة الأولى. يجب تجنب استخدام هذا الماكرو لأنه يجتاز القائمة head1 بأكملها. يجب استخدام طابور ذيل إذا كان هذا الماكرو مطلوبًا في مسارات التعليمات البرمجية ذات الاستخدام العالي أو للعمل على قوائم طويلة.

الماكرو LIST_EMPTY يقيم إلى صحيح إذا لم تكن هناك عناصر في القائمة.

الماكرو LIST_ENTRY يعرف هيكلًا يربط العناصر في القائمة.

الماكرو LIST_FIRST يُرجع العنصر الأول في القائمة أو NULL إذا كانت القائمة فارغة.

الماكرو LIST_FOREACH يجتاز القائمة المشار إليها بواسطة head

/var> في الاتجاه الأمامي، وتعيين كل عنصر بالتناوب إلى var.

الماكرو LIST_FOREACH_FROM يتصرف بشكل مطابق لـ LIST_FOREACH عندما var يكون NULL، وإلا فإنه يعامل var كعنصر تم العثور عليه مسبقًا في القائمة ويبدأ الدورة من var بدلاً من العنصر الأول في القائمة المشار إليها بواسطة head.

الماكرو LIST_FOREACH_SAFE يتجول في القائمة المشار إليها بواسطة head في الاتجاه الأمامي، ويقوم بتعيين كل عنصر بالتناوب إلى var. ومع ذلك، على عكس LIST_FOREACH()، يُسمح هنا بكل من إزالة var وتحريره بأمان دون التدخل في المسار.

الماكرو LIST_FOREACH_FROM_SAFE يتصرف بشكل مطابق لـ LIST_FOREACH_SAFE عندما var يكون NULL، وإلا فإنه يعامل var كعنصر تم العثور عليه مسبقًا في القائمة ويبدأ الدورة من var بدلاً من العنصر الأول في القائمة المشار إليها بواسطة head.

الماكرو LIST_INIT يقوم بتهيئة القائمة المشار إليها بواسطة head.

الماكرو LIST_INSERT_HEAD يقوم بإدراج العنصر الجديد elm في مقدمة القائمة.

الماكرو LIST_INSERT_AFTER يقوم بإدراج العنصر الجديد elm بعد العنصر listelm.

الماكرو LIST_INSERT_BEFORE يقوم بإدراج العنصر الجديد elm قبل العنصر listelm.

الماكرو LIST_NEXT يعيد العنصر التالي في القائمة، أو NULL إذا كان هذا هو الأخير.

الماكرو LIST_PREV يعيد العنصر السابق في القائمة، أو NULL إذا كان هذا هو الأول. يجب أن تحتوي القائمة head على العنصر elm.

الماكرو LIST_REMOVE يزيل العنصر elm من القائمة.

الماكرو LIST_SWAP يتبادل محتويات head1 و head2.

مثال القائمة

LIST_HEAD(listhead, entry) head =
    LIST_HEAD_INITIALIZER(head);
struct listhead *headp;			/* رأس القائمة. */
struct entry {
	...
	LIST_ENTRY(entry) entries;	/* القائمة. */
	...
} *n1, *n2, *n3, *np, *np_temp;

LIST_INIT(&head);			/* تهيئة القائمة. */

n1 = malloc(sizeof(struct entry));	/* الإدراج في المقدمة. */
LIST_INSERT_HEAD(&head, n1, entries);

n2 = malloc(sizeof(struct entry));	/* الإدراج بعد. */
LIST_INSERT_AFTER(n1, n2, entries);

n3 = malloc(sizeof(struct entry));	/* الإدراج قبل. */
LIST_INSERT_BEFORE(n2, n3, entries);

LIST_REMOVE(n2, entries);		/* الحذف. */
free(n2);
					/* المسار الأمامي. */
LIST_FOREACH(np, &head, entries)
	np-> ...

					/* المسار الأمامي الآمن. */
LIST_FOREACH_SAFE(np, &head, entries, np_temp) {
	np->do_stuff();
	...
	LIST_REMOVE(np, entries);
	free(np);
}

while (!LIST_EMPTY(&head)) {		/* حذف القائمة. */
	n1 = LIST_FIRST(&head);
	LIST_REMOVE(n1, entries);
	free(n1);
}

n1 = LIST_FIRST(&head);			/* حذف أسرع للقائمة. */
while (n1 != NULL) {
	n2 = LIST_NEXT(n1, entries);
	free(n1);
	n1 = n2;
}
LIST_INIT(&head);

طوابير الذيل

يتم تعريف طابور ذيل بواسطة الماكرو TAILQ_HEAD. تحتوي هذه البنية على زوج من المؤشرات، أحدهما يشير إلى العنصر الأول في طابور الذيل والآخر يشير إلى العنصر الأخير في طابور الذيل. العناصر مرتبطة بشكل مزدوج، مما يسمح بحذف أي عنصر دون اجتياز طابور الذيل. يمكن إضافة عناصر جديدة إلى طابور الذيل بعد عنصر موجود، أو قبل عنصر موجود، أو في مقدمة طابور الذيل، أو في نهاية طابور الذيل. يتم تعريف بنية TAILQ_HEAD كالتالي:

TAILQ_HEAD(HEADNAME, TYPE) head;

حيث HEADNAME هو اسم البنية المراد تعريفها و TYPE هو نوع العناصر المرتبطة في طابور الذيل. يمكن بعد ذلك تعريف مؤشر إلى رأس طابور الذيل كالتالي:

struct HEADNAME *headp;

(الأسماء head و headp قابلة للاختيار من قبل المستخدم.)

الماكرو TAILQ_HEAD_INITIALIZER يقيم إلى مُهيئ لطابور الذيل head.

الماكرو TAILQ_CONCAT يربط طابور الذيل المشار إليه بواسطة head2 بنهاية طابور الذيل المشار إليه بواسطة head1، مع حذف جميع العناصر من الأول.

الماكرو TAILQ_EMPTY يقيم إلى صحيح إذا لم تكن هناك عناصر في طابور الذيل.

الماكرو TAILQ_ENTRY يعرّف بنية تربط عناصر طابور الذيل.

الماكرو TAILQ_FIRST يعيد العنصر الأول في طابور الذيل أو NULL إذا كان طابور الذيل فارغًا.

الماكرو TAILQ_FOREACH يتجول في طابور الذيل المشار إليه بواسطة head في الاتجاه الأمامي، ويقوم بتعيين كل عنصر بالتناوب إلى var. يتم تعيين var إلى NULL إذا انتهت الدورة بشكل طبيعي، أو إذا لم تكن هناك عناصر.

الماكرو TAILQ_FOREACH_FROM يتصرف بشكل مطابق لـ TAILQ_FOREACH عندما يكون var NULL، وإلا فإنه يعامل var كعنصر تم العثور عليه مسبقًا في طابور الذيل ويبدأ الدورة في var بدلاً من العنصر الأول في طابور الذيل المشار إليه بواسطة head.

الماكرو TAILQ_FOREACH_REVERSE يتجول في طابور الذيل المشار إليه بواسطة head في الاتجاه العكسي، ويقوم بتعيين كل عنصر بالتناوب إلى var.

الماكرو TAILQ_FOREACH_REVERSE_FROM يتصرف بشكل مطابق لـ TAILQ_FOREACH_REVERSE عندما يكون var NULL، وإلا فإنه يعامل var كعنصر تم العثور عليه مسبقًا في طابور الذيل ويبدأ الدورة العكسية في var بدلاً من العنصر الأخير في طابور الذيل المشار إليه بواسطة head.

الماكرو TAILQ_FOREACH_SAFE و TAILQ_FOREACH_REVERSE_SAFE يتجولان في القائمة المشار إليها بواسطة head في الاتجاه الأمامي أو العكسي على التوالي، ويقومان بتعيين كل عنصر بالتناوب إلى var. ومع ذلك، على عكس إصداراتهم غير الآمنة، تسمح TAILQ_FOREACH و TAILQ_FOREACH_REVERSE بإزالة var وتحريره بأمان دون التدخل في المسار.

الماكرو TAILQ_FOREACH_FROM_SAFE يتصرف بشكل مطابق لـ TAILQ_FOREACH_SAFE عندما يكون var NULL، وإلا فإنه يعامل var كعنصر تم العثور عليه مسبقًا في طابور الذيل ويبدأ الدورة في var بدلاً من العنصر الأول في طابور الذيل المشار إليه بواسطة head.

الماكرو TAILQ_FOREACH_REVERSE_FROM_SAFE يتصرف بشكل مطابق لـ TAILQ_FOREACH_REVERSE_SAFE عندما يكون var NULL، وإلا فإنه يعامل var كعنصر تم العثور عليه مسبقًا في طابور الذيل ويبدأ الدورة العكسية في var بدلاً من العنصر الأخير في طابور الذيل المشار إليه بواسطة head.

الماكرو TAILQ_INIT يقوم بتهيئة طابور الذيل المشار إليه بواسطة head.

الماكرو TAILQ_INSERT_HEAD يقوم بإدراج العنصر الجديد elm في مقدمة طابور الذيل.

الماكرو TAILQ_INSERT_TAIL يقوم بإدراج العنصر الجديد elm في نهاية طابور الذيل.

الماكرو TAILQ_INSERT_AFTER يقوم بإدراج العنصر الجديد elm بعد العنصر listelm.

الماكرو TAILQ_INSERT_BEFORE يقوم بإدراج العنصر الجديد elm قبل العنصر listelm.

الماكرو TAILQ_LAST يعيد العنصر الأخير في طابور الذيل. إذا كان طابور الذيل فارغًا، تكون القيمة المعادلة NULL.

الماكرو TAILQ_NEXT يعيد العنصر التالي في طابور الذيل، أو NULL إذا كان هذا هو الأخير.

الماكرو TAILQ_PREV يعيد العنصر السابق في طابور الذيل، أو NULL إذا كان هذا هو الأول.

الماكرو TAILQ_REMOVE يزيل العنصر elm من طابور الذيل.

الماكرو TAILQ_SWAP يتبادل محتويات head1 و head2.

التشخيص


عند تصحيح أخطاء queue(3)، قد يكون من المفيد تتبع تغييرات قائمة الانتظار. لتمكين التتبع، قم بتعريف الماكرو QUEUE_MACRO_DEBUG_TRACE في وقت الترجمة.


قد يكون من المفيد أيضًا تشويش المؤشرات التي تم فك ربطها من قائمة الانتظار، للكشف عن الاستخدام بعد الإزالة. لتمكين تشويش المؤشرات، قم بتعريف الماكرو QUEUE_MACRO_DEBUG_TRASH في وقت الترجمة. الماكرو QMD_IS_TRASHED(void *ptr) يعيد قيمة صحيحة إذا تم تشويش ptr بواسطة الخيار QUEUE_MACRO_DEBUG_TRASH.
