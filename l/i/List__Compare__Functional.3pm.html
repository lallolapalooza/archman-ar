الحزمة: perl-list-compare

التراخيص: GPL, PerlArtistic

أو

$memb_arr_ref = is_member_which_ref( {
    lists => [ \@Llist, \@Rlist ]،  # القيمة هي مرجع مصفوفة
    item  => 'baker'،               # القيمة هي سلسلة نصية

في المثال أعلاه، ستكون $memb_arr_ref:

لأن 'baker' موجود في @Llist و @Rlist، اللتين تحتفظان بالمواقع 0 و 1، على التوالي، في قائمة الوسائط التي تم تمريرها إلى new().

ملاحظة: تختبر الدالتان is_member_which() و "is_member_which_ref" سلسلة نصية واحدة فقط في كل مرة وبالتالي تأخذان وسيطًا واحدًا فقط. لاختبار أكثر من سلسلة نصية واحدة في المرة الواحدة، راجع الدالة التالية، are_members_which().

  • تحديد في "أي" (إن وجدت) من القوائم التي تم تمريرها كوسائط يمكن العثور على سلسلة نصية واحدة أو أكثر. يتم وضع القوائم التي يتم البحث فيها في مصفوفة، ومرجع إليها هو الوسيط الأول الذي تم تمريره إلى are_members_which(). يتم أيضًا وضع السلاسل النصية المراد اختبارها في مصفوفة، ومرجع إليها هو الوسيط الثاني الذي تم تمريره إلى تلك الدالة. 
    $memb_hash_ref =
    are_members_which( [ \@Llist, \@Rlist ] ،
                       [ qw| abel baker fargo hilton zebra | ]
                     );

    أو

    $memb_hash_ref = are_members_which( {
    lists => [ \@Llist, \@Rlist ]،                    # القيمة هي مرجع مصفوفة
    items => [ qw| abel baker fargo hilton zebra | ]، # القيمة هي مرجع مصفوفة
} );

    القيمة المرجعة هي مرجع إلى جدول تجزئة (hash) للمصفوفات. المفتاح لكل عنصر في هذا الجدول هو السلسلة النصية التي يتم اختبارها. قيمة كل عنصر هي مرجع إلى مصفوفة مجهولة عناصرها هي تلك الفهارس في قائمة وسائط المنشئ التي تتوافق مع القوائم التي تحتوي على السلاسل النصية التي يتم اختبارها. في الأمثلة أعلاه، سيكون $memb_hash_ref:

    {
     abel     => [ 0    ]،
     baker    => [ 0, 1 ]،
     fargo    => [ 0, 1 ]،
     hilton   => [    1 ]،
     zebra    => [      ]،
};

    ملاحظة: يمكن أن تأخذ are_members_which() أكثر من وسيط؛ تأخذ is_member_which() و is_member_which_ref() وسيطًا واحدًا فقط. على عكس تلك الدوال، ترجع are_members_which() مرجع جدول تجزئة.

  • تحديد ما إذا كان يمكن العثور على سلسلة نصية معينة في أي من القوائم التي تم تمريرها كوسائط. ترجع 1 إذا تم العثور على سلسلة نصية محددة في أي من القوائم و 0 إذا لم يتم العثور عليها. 
    $found = is_member_any( [ \@Llist, \@Rlist ] ، [ 'abel' ] );

    أو

    $found = is_member_any( {
    lists => [ \@Llist, \@Rlist ]، # القيمة هي مرجع مصفوفة
    item  => 'abel'،               # القيمة هي سلسلة نصية
} );

    في المثال أعلاه، سيكون $found هو 1 لأن 'abel' موجود في قائمة واحدة أو أكثر من القوائم التي تم تمريرها كوسائط إلى new().

  • تحديد ما إذا كان يمكن العثور على سلسلة نصية أو سلاسل نصية محددة في أي من القوائم التي تم تمريرها كوسائط. يتم وضع القوائم التي يتم البحث فيها في مصفوفة، ومرجع إليها هو الوسيط الأول الذي تم تمريره إلى are_members_any(). يتم أيضًا وضع السلاسل النصية المراد اختبارها في مصفوفة، ومرجع إليها هو الوسيط الثاني الذي تم تمريره إلى تلك الدالة. 
    $memb_hash_ref =
    are_members_any( [ \@Llist, \@Rlist ] ،
                     [ qw| abel baker fargo hilton zebra | ]
                   );

    أو

    lists => [ \@Llist, \@Rlist ]،                    # القيمة هي مرجع مصفوفة
    items => [ qw| abel baker fargo hilton zebra | ]، # القيمة هي مرجع مصفوفة
} );

    القيمة المرجعة هي مرجع إلى جدول تجزئة حيث المفتاح لكل عنصر هو السلسلة النصية التي يتم اختبارها والقيمة هي 1 إذا تم العثور على السلسلة النصية في أي من القوائم و 0 إذا لم يتم العثور عليها. في الأمثلة أعلاه، سيكون $memb_hash_ref:

    {
     abel     => 1,
     baker    => 1,
     fargo    => 1,
     hilton   => 1,
     zebra    => 0,
};

    قيمة "zebra" هي 0 لأن "zebra" غير موجودة في أي من القوائم التي تم تمريرها كوسائط إلى are_members_any().
