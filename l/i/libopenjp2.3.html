الاسم

libopenjp2 - مكتبة لقراءة وكتابة ملفات الصور JPEG2000.

الملخص

#include <openjpeg.h>

تنسيقات التحويل

PGX: imagetopgx() / pgxtoimage()

PXM: imagetopnm() / pnmtoimage()

BMP: imagetobmp() / bmptoimage()

TIF: imagetotif() / tiftoimage()

RAW: imagetoraw() / rawtoimage()

TGA: imagetotga() / tgatoimage()

PNG: imagetopng() / pngtoimage()

YUV: imagetoyuv() / yuvtoimage() (MJ2)

معلمات فك الضغط

typedef struct opj_dparameters
{
/*
قم بتعيين عدد مستويات الدقة الأعلى التي سيتم تجاهلها.
يتم قسمة دقة الصورة فعليًا على 2 مرفوعة للقوة
عدد المستويات المهملة.
يقتصر عامل التقليل على أصغر عدد إجمالي
مستويات التحلل بين البلاطات.
إذا كان != 0 ، فإن البعد الأصلي مقسومًا على 2 ^ (reduce)؛
إذا كان == 0 أو غير مستخدم، يتم فك ضغط الصورة بالدقة الكاملة
*/
int cp_reduce;
/*
قم بتعيين الحد الأقصى لعدد طبقات الجودة التي سيتم فك ضغطها.
إذا كانت هناك طبقات جودة أقل من العدد المحدد ،
يتم فك ضغط جميع طبقات الجودة.
إذا كان != 0 ، يتم فك ضغط الطبقات "layer" الأولى فقط؛
إذا كان == 0 أو غير مستخدم، يتم فك ضغط جميع طبقات الجودة
*/
int cp_layer;


/*معلمات تشفير سطر الأوامر (غير مستخدمة داخل المكتبة) */
/* اسم ملف الإدخال */
char infile[OPJ_PATH_LEN];
/* اسم ملف الإخراج */
char outfile[OPJ_PATH_LEN];
/* تنسيق ملف الإدخال: انظر OPJ_CODEC_FORMAT */
int decod_format;
/* تنسيق ملف الإخراج */
int cod_format;


/*معلمات فك ضغط JPWL */
/* ينشط إمكانيات تصحيح JPWL */
bool jpwl_correct;
/* العدد المتوقع للمكونات */
int jpwl_exp_comps;
/* الحد الأقصى لعدد البلاطات */
int jpwl_max_tiles;


/*
حدد ما إذا كان سيتم إجراء فك الضغط على
تدفق الشفرة بأكمله، أو سيقتصر على الرأس الرئيسي
يحد فك الضغط من الرأس الرئيسي
لتكون قادرة على استخراج خصائص تدفق الشفرة
إذا كان == NO_LIMITATION، يتم فك ضغط تدفق الشفرة بالكامل؛
إذا كان == LIMIT_TO_MAIN_HEADER، يتم فك ضغط الرأس الرئيسي فقط؛
*/
OPJ_LIMIT_DECODING cp_limit_decoding;
} opj_dparameters_t;

typedef struct opj_cparameters
{
/* حجم البلاطة: tile_size_on = false (ليس في الوسيط)
أو tile_size_on = true (في الوسيط) */
bool tile_size_on;
/* XTOsiz */
int cp_tx0;
/* YTOsiz */
int cp_ty0;
/* XTsiz */
int cp_tdx;
/* YTsiz */
int cp_tdy;
/* تخصيص حسب المعدل / التشوه */
int cp_disto_alloc;
/* تخصيص حسب الطبقة الثابتة */
int cp_fixed_alloc;
/* إضافة جودة ثابتة */
int cp_fixed_quality;
/* طبقة ثابتة */
int *cp_matrice;
/* تعليق للترميز */
char *cp_comment;
/* نمط الترميز */
int csty;
/* ترتيب التقدم:
PROG_UNKNOWN, LRCP (الافتراضي)، RLCP، RPCL، PCRL، CPRL */
OPJ_PROG_ORDER prog_order;
/* تغييرات ترتيب التقدم */
opj_poc_t POC[32];
/* عدد تغييرات ترتيب التقدم (POC)، الافتراضي: 0 */
int numpocs;
/* عدد الطبقات */
int tcp_numlayers;
/* معدلات الطبقات */
float tcp_rates[100];
/* قيم psnr مختلفة للطبقات المتتالية */
float tcp_distoratio[100];
/* عدد المستويات */
int numresolution;
/* عرض كتلة الشفرة الأولي، الافتراضي: 64 */
int cblockw_init;
/* ارتفاع كتلة الشفرة الأولي، الافتراضي: 64 */
int cblockh_init;
/* وضع التبديل (cblk_style) */
/* 1: استخدام DWT 9-7 غير القابل للعكس،
0: استخدام ضغط بدون فقدان (الافتراضي) */
int irreversible;
/* منطقة الاهتمام: المكون المتأثر في [0..3]،
-1 يعني عدم وجود منطقة اهتمام */
int roi_compno;
/* منطقة الاهتمام: قيمة الإزاحة لأعلى */
int roi_shift;
/* عدد مواصفات حجم المنطقة */
int res_spec;
/* الأولي /* عرض المقطع */
int prcw_init[J2K_MAXRLVLS];
/* ارتفاع المقطع الأولي */
int prch_init[J2K_MAXRLVLS];


/* معلمات مشفر سطر الأوامر (لا تستخدم داخل المكتبة) */
/* اسم ملف الإدخال */
char infile[OPJ_PATH_LEN];
/* اسم ملف الإخراج */
char outfile[OPJ_PATH_LEN];
/* تم إيقاف استخدامه. يتم التعامل مع إنشاء الفهرس الآن باستخدام
الدالة opj_encode_with_info(). اضبطه على NULL */
int index_on;
/* تم إيقاف استخدامه. يتم التعامل مع إنشاء الفهرس الآن باستخدام
الدالة opj_encode_with_info(). اضبطه على NULL */
char index[OPJ_PATH_LEN];
/* تشفير الصورة الفرعية: إزاحة الصورة الأصلية في اتجاه x */
int image_offset_x0;
/* تشفير الصورة الفرعية: إزاحة الصورة الأصلية في اتجاه y */
int image_offset_y0;
/* قيمة الاستعيان لـ dx */
int subsampling_dx;
/* قيمة الاستعيان لـ dy */
int subsampling_dy;
/* تنسيق ملف الإدخال */
int decod_format;
/* تنسيق ملف الإخراج: انظر OPJ_CODEC_FORMAT */
int cod_format;


/* معلمات تشفير JPWL */
/* تمكين كتابة EPC في MH، وبالتالي تفعيل JPWL */
bool jpwl_epc_on;
/* طريقة حماية الأخطاء لـ MH (0,1,16,32,37-128) */
int jpwl_hprot_MH;
/* رقم المقطع لمواصفات حماية الرأس (>=0) */
int jpwl_hprot_TPH_tileno[JPWL_MAX_NO_TILESPECS];
/* طرق حماية الأخطاء لـ TPHs (0,1,16,32,37-128) */
int jpwl_hprot_TPH[JPWL_MAX_NO_TILESPECS];
/* رقم المقطع لمواصفات حماية الحزم (>=0) */
int jpwl_pprot_tileno[JPWL_MAX_NO_PACKSPECS];
/* رقم الحزمة لمواصفات حماية الحزم (>=0) */
int jpwl_pprot_packno[JPWL_MAX_NO_PACKSPECS];
/* طرق حماية الأخطاء للحزم (0,1,16,32,37-128) */
int jpwl_pprot[JPWL_MAX_NO_PACKSPECS];
/* تمكين كتابة ESD، (0=لا/1/2 بايت) */
int jpwl_sens_size;
/* حجم عنوان الحساسية (0=تلقائي/2/4 بايت) */
int jpwl_sens_addr;
/* نطاق الحساسية (0-3) */
int jpwl_sens_range;
/* طريقة الحساسية لـ MH (-1=لا، 0-7) */
int jpwl_sens_MH;
/* رقم المقطع لمواصفات الحساسية (>=0) */
int jpwl_sens_TPH_tileno[JPWL_MAX_NO_TILESPECS];
/* طرق الحساسية لـ TPHs (-1=لا، 0-7) */
int jpwl_sens_TPH[JPWL_MAX_NO_TILESPECS];


/* الامتثال للسينما الرقمية: OFF-غير متوافق،
CINEMA2K_24, CINEMA2K_48, CINEMA4K_24 */
OPJ_CINEMA_MODE cp_cinema;
/* الحد الأقصى لمعدل كل مكون.
إذا كان == 0، لا يتم اعتبار حد حجم المكون */
int max_comp_size;
/* اسم التعريف */
OPJ_RSIZ_CAPABILITIES cp_rsiz;
/* إنشاء جزء المقطع */
char tp_on;
/* علامة لإنشاء جزء المقطع */
char tp_flag;
/* MCT (تحويل متعدد المكونات) */
char tcp_mct;
} opj_cparameters_t;
