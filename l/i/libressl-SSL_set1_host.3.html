libressl-SSL_set1_host(3)

الحزمة: libressl

التراخيص: ISC, custom:OpenSSL

SSL_SET1_HOST(3) دليل وظائف المكتبة SSL_SET1_HOST(3)

الاسم

SSL_set1_host، SSL_set_hostflags، SSL_get0_peername — معلمات التحقق من خادم SSL

ملخص

المكتبة "libssl" libcrypto
#include <openssl/ssl.h>

int
SSL_set1_host(SSL *ssl، const char *hostname);

void
SSL_set_hostflags(SSL *ssl، unsigned int flags);

const char *
SSL_get0_peername(SSL *ssl);

الوصف

SSL_set1_host() يقوم بتكوين فحص اسم المضيف للخادم في عميل ssl، مع تعيين اسم DNS المتوقع لـ hostname ومسح أي اسم مضيف تم تحديده مسبقًا. إذا كان hostname هو NULL أو سلسلة فارغة، فلن يتم إجراء فحوصات الاسم على شهادة النظير. إذا تم تحديد hostname غير فارغ، فإن التحقق من الشهادة يقوم تلقائيًا بفحص اسم المضيف للنظير عبر X509_check_host(3) مع تعيين flags إلى 0.

SSL_set_hostflags() يحدد العلامات التي سيتم تمريرها إلى X509_check_host(3) عند تطبيق فحوصات الاسم، افتراضيًا تكون قيمة العلامات 0. راجع X509_check_host(3) للحصول على قائمة العلامات المتاحة ومعناها.

SSL_get0_peername() يعيد اسم DNS أو الاسم الشائع للموضوع من شهادة النظير التي تطابقت مع أحد المعرفات المرجعية. ما لم يتم تعطيل مطابقة الأحرف البديلة، قد يكون الاسم المتطابق في شهادة النظير اسمًا بديلاً. يشير معرف مرجعي يبدأ بـ " ." إلى بادئة نطاق أب بدلاً من اسم ثابت. في هذه الحالة، قد يكون اسم النظير المتطابق نطاقًا فرعيًا للمعرف المرجعي. السلسلة المعاد ملكيتها للمكتبة وليست صالحة بمجرد مسح كائن ssl المرتبط أو تحريره، أو إذا حدث إعادة التفاوض. يجب على التطبيقات عدم تحرير القيمة المعادة.

يُنصح عملاء SSL باستخدام هذه الدوال بدلاً من استدعاء X509_check_host(3) بشكل صريح.

قيم الإرجاع

SSL_set1_host() يعيد 1 للنجاح أو 0 للفشل.

SSL_get0_peername() يعيد اسم النظير المتطابق أو NULL إذا لم يكن التحقق من اسم النظير قابلاً للتطبيق أو لم يتم مطابقة اسم نظير موثوق به. استخدم SSL_get_verify_result(3) لتحديد ما إذا كان التحقق قد نجح.

أمثلة

تتحقق الاستدعاءات أدناه من اسم المضيف. الأحرف البديلة مدعومة، ولكن يجب أن تتطابق مع التسمية بأكملها. يتم استرداد الاسم الفعلي المتطابق في الشهادة (والذي قد يكون بديلاً)، ويجب نسخه بواسطة التطبيق إذا كان سيتم الاحتفاظ به بعد عمر اتصال SSL.

if (!SSL_set1_host(ssl, "smtp.example.com"))
	/* خطأ */

/* XXX: قم بأداء مصافحة SSL_connect() وتعامل مع الأخطاء هنا */

if (SSL_get_verify_result(ssl) == X509_V_OK) {
	const char *peername = SSL_get0_peername(ssl);

	if (peername != NULL)
		/* كانت فحوصات الاسم في النطاق وطابقت peername */
}
