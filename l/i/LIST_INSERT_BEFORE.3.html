LIST(3) صفحات دليل C Library LIST(3)

الاسم

LIST_EMPTY, LIST_ENTRY, LIST_FIRST, LIST_FOREACH, LIST_HEAD, LIST_HEAD_INITIALIZER, LIST_INIT, LIST_INSERT_AFTER, LIST_INSERT_BEFORE, LIST_INSERT_HEAD, LIST_NEXT, LIST_REMOVE - تطبيق قائمة مزدوجة الربط

المكتبة

مكتبة C القياسية (libc, -lc)

ملخص

#include <sys/queue.h>
LIST_ENTRY(TYPE);
LIST_HEAD(HEADNAME, TYPE);
LIST_HEAD LIST_HEAD_INITIALIZER(LIST_HEAD head);
void LIST_INIT(LIST_HEAD *head);
int LIST_EMPTY(LIST_HEAD *head);
void LIST_INSERT_HEAD(LIST_HEAD *head,
                        struct TYPE *elm, LIST_ENTRY NAME);
void LIST_INSERT_BEFORE(struct TYPE *listelm,
                        struct TYPE *elm, LIST_ENTRY NAME);
void LIST_INSERT_AFTER(struct TYPE *listelm,
                        struct TYPE *elm, LIST_ENTRY NAME);
struct TYPE *LIST_FIRST(LIST_HEAD *head);
struct TYPE *LIST_NEXT(struct TYPE *elm, LIST_ENTRY NAME);
LIST_FOREACH(struct TYPE *var, LIST_HEAD *head, LIST_ENTRY NAME);
void LIST_REMOVE(struct TYPE *elm, LIST_ENTRY NAME);

الوصف

تعرّف هذه الماكرو وتتعامل مع قوائم مزدوجة الربط.

في تعريفات الماكرو، TYPE هو اسم بنية معرفة من قبل المستخدم، ويجب أن تحتوي على حقل من نوع LIST_ENTRY، يسمى NAME. الوسيط HEADNAME هو اسم بنية معرفة من قبل المستخدم، ويجب تعريفه باستخدام الماكرو LIST_HEAD().

الإنشاء

يتم ترأس القائمة ببنية معرفة بواسطة الماكرو LIST_HEAD(). تحتوي هذه البنية على مؤشر فريد للعنصر الأول في القائمة. ترتبط العناصر مزدوجة، بحيث يمكن إزالة عنصر عشوائي دون المرور عبر القائمة. يمكن إضافة عناصر جديدة إلى القائمة بعد عنصر موجود، أو قبله، أو في مقدمة القائمة. بنية LIST_HEAD تعرف على النحو التالي:


LIST_HEAD(HEADNAME, TYPE) head;

حيث struct HEADNAME هي البنية المراد تعريفها، و struct TYPE هو نوع العناصر المرتبطة بالقائمة. يمكن بعد ذلك تعريف عنوان مؤشر إلى مقدمة القائمة على النحو التالي:


struct HEADNAME *headp;

(الأسماء head و headp قابلة للاختيار من قبل المستخدم.)

LIST_ENTRY() تعرّف بنية تربط عناصر القائمة.

LIST_HEAD_INITIALIZER() تُقيّم إلى مُهيئ للقائمة head.

LIST_INIT() تهيئ القائمة المشار إليها بواسطة head.

LIST_EMPTY() تُرجع صحيحًا إذا كانت القائمة فارغة، وخطأ إذا كانت تحتوي على عنصر واحد على الأقل.

الإدراج

LIST_INSERT_HEAD() يُدرج العنصر elm في مقدمة القائمة.

LIST_INSERT_BEFORE() يُدرج العنصر elm قبل العنصر listelm.

LIST_INSERT_AFTER() يُدرج العنصر elm بعد العنصر listelm.

التنقل

LIST_FIRST() تُرجع العنصر الأول في القائمة، أو NULL إذا كانت القائمة فارغة.

LIST_NEXT() تُرجع العنصر التالي في القائمة، أو NULL إذا كان هو الأخير.

LIST_FOREACH() يتنقل عبر القائمة المشار إليها بواسطة head في الاتجاه الأمامي، ويعيّن كل عنصر إلى var.

الإزالة

LIST_REMOVE() تزيل العنصر elm من القائمة.

قيمة الإرجاع

LIST_EMPTY() تُرجع قيمة غير صفرية إذا كانت القائمة فارغة، وصفر إذا كانت تحتوي على عنصر واحد على الأقل.

LIST_FIRST() و LIST_NEXT() تُرجعان مؤشرًا إلى بنية TYPE الأولى أو التالية، على التوالي.

LIST_HEAD_INITIALIZER() تُرجع مُهيئًا يمكن تعيينه إلى القائمة head.

المعايير

BSD.

التاريخ

4.4BSD.

أخطاء

LIST_FOREACH() لا تسمح بإزالة أو تحرير var داخل الحلقة، لأن ذلك قد يتداخل مع التنقل. LIST_FOREACH_SAFE()، الموجودة في BSDs ولكن ليس في glibc، تعالج هذا القيد عن طريق السماح بإزالة وتحرير var داخل الحلقة دون التأثير على التنقل.

أمثلة

#include <stddef.h>
#include <stdio.h>
#include <stdlib.h>
#include <sys/queue.h>
struct entry {
    int data;
    LIST_ENTRY(entry) entries;              /* قائمة */
};
LIST_HEAD(listhead, entry);
int
main(void)
{
    struct entry *n1, *n2, *n3, *np;
    struct listhead head;                   /* قائمة مرؤوسة */
    int i;
    LIST_INIT(&head);                       /* تهيئة القائمة */
    n1 = malloc(sizeof(struct entry));      /* إدراج في المقدمة */
    LIST_INSERT_HEAD(&head, n1, entries);
    n2 = malloc(sizeof(struct entry));      /* إدراج بعد */
    LIST_INSERT_AFTER(n1, n2, entries);
    n3 = malloc(sizeof(struct entry));      /* إدراج قبل */
    LIST_INSERT_BEFORE(n2, n3, entries);
    i = 0;                                  /* تنقل أمامي */
    LIST_FOREACH(np, &head, entries)
        np->data = i++;
    LIST_REMOVE(n2, entries);               /* إزالة */
    free(n2);
                                            /* تنقل أمامي */
    LIST_FOREACH(np, &head, entries)
        printf("%i\n", np->data);
                                            /* إزالة القائمة */
    n1 = LIST_FIRST(&head);
    while (n1 != NULL) {
        n2 = LIST_NEXT(n1, entries);
        free(n1);
        n1 = n2;
    }
    LIST_INIT(&head);
    exit(EXIT_SUCCESS);
}
