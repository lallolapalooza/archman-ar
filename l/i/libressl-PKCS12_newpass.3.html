libressl-PKCS12_newpass(3)

الحزمة: libressl

التراخيص: ISC, custom:OpenSSL

PKCS12_NEWPASS(3) دليل الدوال المكتبية PKCS12_NEWPASS(3)

الاسم

PKCS12_newpass — تغيير كلمة مرور هيكل PKCS#12

الملخص

المكتبة “libcrypto”
#include <openssl/pkcs12.h>

int
PKCS12_newpass(PKCS12 *p12, const char *oldpass, const char *newpass);

الوصف

PKCS12_newpass() تغير كلمة مرور هيكل PKCS#12.

p12 هو مؤشر إلى هيكل PKCS#12. oldpass هي كلمة المرور الحالية و newpass هي كلمة المرور الجديدة.

إذا كان هيكل PKCS#12 لا يحتوي على كلمة مرور، استخدم السلسلة الفارغة "" لـ oldpass. تمرير NULL لـ oldpass يؤدي إلى فشل PKCS12_newpass().

إذا تم استخدام كلمة المرور الخاطئة لـ oldpass، ستفشل الدالة بسبب خطأ في التحقق من MAC. في حالات نادرة، لا يحتوي هيكل PKCS#12 على MAC: في هذه الحالة، ستفشل عادةً بسبب خطأ في حشو فك التشفير.

قيم الإرجاع

عند النجاح، يتم إرجاع 1؛ وإلا، يتم إرجاع 0 ويمكن استرداد رمز خطأ باستخدام ERR_get_error(3).

أمثلة

يقوم هذا المثال بتحميل ملف PKCS#12، وتغيير كلمة المرور الخاصة به، وكتابة النتيجة إلى ملف جديد.

#include <stdio.h>
#include <stdlib.h>
#include <openssl/pem.h>
#include <openssl/err.h>
#include <openssl/pkcs12.h>

int main(int argc, char **argv)
{
	FILE *fp;
	PKCS12 *p12;
	if (argc != 5) {
		fprintf(stderr,
		    "Usage: pkread p12file password newpass opfile\n");
		return 1;
	}
	if ((fp = fopen(argv[1], "rb")) == NULL) {
		fprintf(stderr, "Error opening file %s\n", argv[1]);
		return 1;
	}
	p12 = d2i_PKCS12_fp(fp, NULL);
	fclose(fp);
	if (p12 == NULL) {
		fprintf(stderr, "Error reading PKCS#12 file\n");
		ERR_print_errors_fp(stderr);
		return 1;
	}
	if (PKCS12_newpass(p12, argv[2], argv[3]) == 0) {
		fprintf(stderr, "Error changing password\n");
		ERR_print_errors_fp(stderr);
		PKCS12_free(p12);
		return 1;
	}
	if ((fp = fopen(argv[4], "wb")) == NULL) {
		fprintf(stderr, "Error opening file %s\n", argv[4]);
		PKCS12_free(p12);
		return 1;
	}
	i2d_PKCS12_fp(fp, p12);
	PKCS12_free(p12);
	fclose(fp);
	return 0;
}

التاريخ

PKCS12_newpass() ظهرت لأول مرة في OpenSSL 0.9.5 وهي متاحة منذ OpenBSD 2.7.

الأخطاء

تنسيق كلمة المرور هو سلسلة ASCII منتهية بصفر يتم تحويلها إلى شكل Unicode داخليًا. نتيجة لذلك، لا يمكن توفير بعض كلمات المرور لهذه الدالة.
