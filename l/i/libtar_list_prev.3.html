libtar_list_prev(3)

الحزمة: libtar

التراخيص: BSD

libtar_list_new(3) استدعاءات مكتبة C libtar_list_new(3)

#include <libtar.h>

void libtar_list_del(libtar_list_t *l, libtar_node_t **n);
int libtar_list_search(libtar_list_t *l, libtar_node_t **n, void *data, int (*matchfunc)());

libtar_list_t *libtar_list_dup(libtar_list_t *l);
libtar_list_t *libtar_list_merge(int (*cmpfunc)(), int flags, libtar_list_t *list1, libtar_list_t *list2);

الوصف

تقوم الدالة libtar_list_new() بإنشاء قائمة جديدة. يجب أن تكون قيمة الوسيط flags إحدى القيم التالية:

LIST_USERFUNC
يشير الوسيط cmpfunc إلى دالة يوفرها المستخدم تحدد ترتيب القائمة.
LIST_STACK
استخدام القائمة كمكدس. يتم إضافة العناصر الجديدة إلى مقدمة القائمة.
LIST_QUEUE
استخدام القائمة كطابور. يتم إضافة العناصر الجديدة إلى نهاية القائمة.

تقوم الدالة libtar_list_free() بإلغاء تخصيص كل الذاكرة المرتبطة بالقائمة l. إذا لم يكن freefunc يساوي NULL، فسيتم استدعاؤه لتحرير الذاكرة المرتبطة بكل عقدة في القائمة.

تقوم الدالة libtar_list_add() بإضافة العنصر الذي يشير إليه data إلى القائمة l. سيتم تحديد موضع العنصر الجديد بواسطة الأعلام التي تم تمريرها إلى libtar_list_new() عند إنشاء القائمة.

تقوم الدالة libtar_list_add_str() بتقسيم السلسلة النصية str إلى رموز باستخدام محددات الفصل الموجودة في السلسلة النصية delim. يتم إضافة الرموز الناتجة إلى القائمة l باستخدام libtar_list_add().

تقوم الدالة libtar_list_search() بالبحث عن عنصر يطابق data باستخدام دالة المطابقة matchfunc. إذا كان matchfunc يساوي NULL، فسيتم استخدام دالة مطابقة افتراضية مصممة للسلاسل النصية ASCII. يبدأ البحث من العقدة التي يشير إليها n.

تقوم الدالة libtar_list_del() بإزالة الإدخال الذي يشير إليه n من القائمة التي يشير إليها l.

تقوم الدالة libtar_list_dup() بإنشاء نسخة من القائمة l باستخدام ذاكرة مخصصة ديناميكيًا.

تقوم الدالة libtar_list_merge() بإنشاء قائمة جديدة باستخدام flags و cmpfunc، بنفس طريقة libtar_list_new(). ثم تقوم بإضافة جميع العناصر من list1 و list2 باستخدام libtar_list_add().

قيمة الإرجاع

تقوم الدوال libtar_list_new() و libtar_list_dup() و libtar_list_merge() بإرجاع مؤشر إلى بنية القائمة الجديدة، أو NULL في حالة حدوث خطأ.

تقوم الدوال libtar_list_next() و libtar_list_prev() و libtar_list_search() بإرجاع 1 عندما يتم إرجاع بيانات صالحة، أو 0 بخلاف ذلك.

تقوم الدوال libtar_list_add() و libtar_list_add_str() بإرجاع 0 عند النجاح، أو -1 عند حدوث خطأ.
