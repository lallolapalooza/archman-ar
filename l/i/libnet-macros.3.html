توثيق تعريفات الوحدات

#define for_each_context_in_cq(l) for (l = libnet_cq_head(); libnet_cq_last(); l = libnet_cq_next())

يوفر واجهة للتكرار عبر طابور السياقات في libnet. قبل استدعاء هذه الوحدة، تأكد من تعيين الطابور باستخدام libnet_cq_head().

#define IN6ADDR_ERROR_INIT

رمز خطأ IPv6

#define LIBNET_DONT_RESOLVE 0

تستخدم لوظائف تحليل الأسماء في libnet، تحدد أنه لا ينبغي إجراء عمليات بحث DNS وأن عنوان IP يجب أن يبقى في شكل رقمي.

#define LIBNET_ERRBUF_SIZE 0x100

مخزن أخطاء libnet طوله 256 بايت.

#define LIBNET_MAX_PACKET 0xffff

أكبر حجم يمكن أن يكون لحزمة IP - 65,535 بايت.

#define LIBNET_MAXOPTION_SIZE 0x28

يمكن أن تكون خيارات IP و TCP بطول يصل إلى 40 بايت.

#define LIBNET_OFF 1

تستخدم في عدة أماكن، لتحديد 'تشغيل' أو 'واحد'

#define LIBNET_ON 0

تستخدم في عدة أماكن، لتحديد 'تشغيل' أو 'واحد'

#define LIBNET_PR2 0

تستخدم لـ libnet_get_prand() لتحديد تصرف الوظيفة

#define LIBNET_RESOLVE 1

تستخدم لوظائف تحليل الأسماء في libnet، تحدد أنه يمكن إجراء بحث DNS إذا لزم الأمر لحل عنوان IP إلى شكل معياري.
