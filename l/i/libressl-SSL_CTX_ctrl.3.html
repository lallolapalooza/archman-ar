libressl-SSL_CTX_ctrl(3)

الحزمة: libressl

التراخيص: ISC, custom:OpenSSL

SSL_CTX_CTRL(3) دليل المكتبة SSL_CTX_CTRL(3)

الاسم

SSL_CTX_ctrl, SSL_CTX_callback_ctrl, SSL_ctrl, SSL_callback_ctrl — وظائف معالجة داخلية لكائنات SSL_CTX و SSL

الملخص

المكتبة “libssl” libcrypto
#include <openssl/ssl.h>

long
SSL_CTX_ctrl(SSL_CTX *ctx, int cmd, long larg, void *parg);

long
SSL_CTX_callback_ctrl(SSL_CTX *, int cmd, void (*fp)());

long
SSL_ctrl(SSL *ssl, int cmd, long larg, void *parg);

long
SSL_callback_ctrl(SSL *, int cmd, void (*fp)());

الوصف

تُستخدم عائلة الوظائف SSL_*_ctrl() لمعالجة إعدادات كائنات SSL_CTX و SSL. اعتمادًا على الأمر cmd، يتم تقييم الوسائط larg، parg، أو fp. لا ينبغي استدعاء هذه الوظائف مباشرة أبدًا. تتوفر جميع الوظائف المطلوبة من خلال وظائف أو ماكرو أخرى.

قيم الإرجاع

تعتمد قيم الإرجاع لوظائف SSL*_ctrl() على الأمر المقدم عبر المعلمة cmd().

ssl(3), SSL_CTX_add_extra_chain_cert(3), SSL_CTX_sess_number(3), SSL_CTX_sess_set_cache_size(3), SSL_CTX_set_max_cert_list(3), SSL_CTX_set_mode(3), SSL_CTX_set_options(3), SSL_CTX_set_session_cache_mode(3), SSL_CTX_set_tlsext_servername_callback(3), SSL_CTX_set_tlsext_status_cb(3), SSL_CTX_set_tlsext_ticket_key_cb(3), SSL_get_server_tmp_key(3), SSL_num_renegotiations(3), SSL_session_reused(3), SSL_set_max_send_fragment(3)

التاريخ

SSL_CTX_ctrl() و SSL_ctrl() ظهرت لأول مرة في SSLeay 0.8.0 وكانت متاحة منذ OpenBSD 2.4.

SSL_CTX_callback_ctrl() و SSL_callback_ctrl() ظهرت لأول مرة في OpenSSL 0.9.5 وكانت متاحة منذ OpenBSD 2.7.
