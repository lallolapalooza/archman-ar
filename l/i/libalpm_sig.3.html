توثيق هياكل البيانات

struct alpm_pgpkey_t

مفتاح PGP.

حقول البيانات:

alpm_time_t created متى تم إنشاء المفتاح.

void * data بيانات المفتاح الفعلية.

char * email بريد المالك للمفتاح.

alpm_time_t expires متى تنتهي صلاحية المفتاح.

char * fingerprint بصمة المفتاح.

unsigned int length طول المفتاح.

char * name اسم المالك للمفتاح.

char pubkey_algo حرف يمثل خوارزمية التشفير المستخدمة بواسطة المفتاح العام.
؟ = غير معروف R = RSA D = DSA E = EDDSA

unsigned int revoked هل تم إلغاء المفتاح؟

char * uid معرف المستخدم للمفتاح.

struct alpm_sigresult_t

نتيجة التوقيع.

يحتوي على المفتاح، والحالة، والصلاحية لتوقيع معين.

حقول البيانات:

alpm_pgpkey_t key مفتاح التوقيع.

alpm_sigstatus_t status حالة التوقيع.

alpm_sigvalidity_t validity صلاحية التوقيع.

struct alpm_siglist_t

قائمة التوقيعات.

يحتوي على عدد التوقيعات الموجودة ومؤشر إلى مصفوفة من النتائج. حجم المصفوفة هو count.

حقول البيانات:

size_t count عدد النتائج في المصفوفة.

alpm_sigresult_t * results مصفوفة من نتائج التوقيع.

توثيق أنواع التعداد

enum alpm_siglevel_t

خيارات التحقق من توقيع PGP.

المُعداد

ALPM_SIG_PACKAGE
الحزم تتطلب توقيعًا.
ALPM_SIG_PACKAGE_OPTIONAL
الحزم لا تتطلب توقيعًا، ولكن تحقق من الحزم التي لديها توقيعات.
ALPM_SIG_PACKAGE_MARGINAL_OK
السماح للحزم بتوقيعات ذات ثقة هامشية.
ALPM_SIG_PACKAGE_UNKNOWN_OK
السماح للحزم بتوقيعات غير معروفة الثقة.
ALPM_SIG_DATABASE
قواعد البيانات تتطلب توقيعًا.
ALPM_SIG_DATABASE_OPTIONAL
قواعد البيانات لا تتطلب توقيعًا، ولكن تحقق من قواعد البيانات التي لديها توقيعات.
ALPM_SIG_DATABASE_MARGINAL_OK
السماح لقواعد البيانات بتوقيعات ذات ثقة هامشية.
ALPM_SIG_DATABASE_UNKNOWN_OK
السماح لقواعد البيانات بتوقيعات غير معروفة الثقة.
ALPM_SIG_USE_DEFAULT
مستوى التوقيع الافتراضي.

enum alpm_sigstatus_t

رموز حالة التحقق من توقيع PGP.

المُعداد

ALPM_SIGSTATUS_VALID
التوقيع صحيح.
ALPM_SIGSTATUS_KEY_EXPIRED
انتهت صلاحية المفتاح.
ALPM_SIGSTATUS_SIG_EXPIRED
انتهت صلاحية التوقيع.
ALPM_SIGSTATUS_KEY_UNKNOWN
المفتاح غير موجود في سلسلة المفاتيح.
ALPM_SIGSTATUS_KEY_DISABLED
تم تعطيل المفتاح.
ALPM_SIGSTATUS_INVALID
التوقيع غير صحيح.

enum alpm_sigvalidity_t

مستوى الثقة لمفتاح PGP.

المُعداد

ALPM_SIGVALIDITY_FULL
التوقيع موثوق به بالكامل.
ALPM_SIGVALIDITY_MARGINAL
التوقيع موثوق به بشكل هامشي.
ALPM_SIGVALIDITY_NEVER
التوقيع غير موثوق به أبدًا.
ALPM_SIGVALIDITY_UNKNOWN
التوقيع له ثقة غير معروفة.

int alpm_db_check_pgp_signature (alpm_db_t * db, alpm_siglist_t * siglist)

تحقق من توقيع PGP لقاعدة البيانات المعطاة.

المعاملات

db قاعدة البيانات للتحقق منها
siglist مؤشر لتخزين نتائج التوقيع

القيم المعادة

فك ترميز توقيع محمل بصيغة base64.

المعاملات

base64_data التوقيع لمحاولة فك ترميزه
data البيانات بعد فك الترميز؛ يجب تحريرها بواسطة المتصل
data_len طول البيانات المعادة

القيم المعادة

0 عند النجاح، -1 عند الفشل في فك الترميز بشكل صحيح

int alpm_extract_keyid (alpm_handle_t * handle, const char * identifier, const unsigned char * sig, const size_t len, alpm_list_t ** keys)

المعاملات

handle مقبض السياق

sig توقيع PGP
len طول التوقيع
keys مؤشر لتخزين معرفات المفاتيح

القيم المعادة

0 عند النجاح، -1 عند حدوث خطأ

المعاملات

siglist مؤشر لتخزين نتائج التوقيع
