libsolv-constantids(3)

الحزمة: libsolv

التراخيص: BSD-3-Clause

مجموع التحقق "solvable:checksum"

مجموع التحقق للحزمة الثنائية. راجع قسم أنواع البيانات للحصول على قائمة بالخوارزميات المدعومة.

SOLVABLE_PKGID "solvable:pkgid"

سلسلة تعريف حزمة. بالنسبة لحزم rpm، هذا هو ملخص md5sum لرأس الحزمة والحمولة.

SOLVABLE_HDRID "solvable:hdrid"

سلسلة تعريف حزمة. بالنسبة لحزم rpm، هذا هو ملخص sha1sum لرأس الحزمة فقط.

SOLVABLE_LEADSIGID "solvable:leadsigid"

سلسلة تعريف جزء التوقيع للحزمة. بالنسبة لحزم rpm، هذا هو ملخص md5sum من بداية الملف حتى رأس الحزمة (أي أنه يتضمن المقدمة، ورأس التوقيع، والحشو).

SOLVABLE_HEADEREND "solvable:headerend"

إزاحة الحمولة في حزم rpm الثنائية. يمكنك استخدام هذه المعلومات لتنزيل الرأس فقط إذا كنت ترغب في عرض معلومات غير مدرجة في بيانات المستودع الوصفية.

SOLVABLE_CHANGELOG "solvable:changelog"

المصفوفة التي تحتوي على جميع هياكل سجل التغييرات.

SOLVABLE_CHANGELOG_AUTHOR "solvable:changelog:author"

مؤلف إدخال سجل التغييرات.

SOLVABLE_CHANGELOG_TIME "solvable:changelog:time"

عدد الثواني منذ بداية حقبة يونكس عندما تمت كتابة إدخال سجل التغييرات.

SOLVABLE_CHANGELOG_TEXT "solvable:changelog:text"

نص إدخال سجل التغييرات.

سمات خاصة للحزم

RPM_RPMDBID "rpm:dbid"

معرف قاعدة بيانات rpm لهذه الحزمة (المثبتة). عادة ما يكون رقمًا صحيحًا صغيرًا.

SOLVABLE_PATCHCATEGORY "solvable:patchcategory"

حقل الفئة لحزم التحديثات. يجب تسميتها "update:category" بدلاً من ذلك.

UPDATE_REBOOT "update:reboot"

إذا كان هذا السمة موجودًا، فيجب إعادة تشغيل النظام بعد تثبيت التحديث.

UPDATE_RESTART "update:restart"

إذا كان هذا السمة موجودًا، فيجب تشغيل مدير البرامج مرة أخرى بعد تثبيت التحديث.

UPDATE_RELOGIN "update:relogin"

إذا كان هذا السمة موجودًا، فيجب على المستخدم تسجيل الخروج ثم تسجيل الدخول مرة أخرى بعد تثبيت التحديث.

UPDATE_MESSAGE "update:message"

رسالة يجب عرضها للمستخدم لتحذيره بشأن أي شيء غير قياسي.

UPDATE_SEVERITY "update:severity"

مدى خطورة التحديث.

UPDATE_RIGHTS "update:rights"

أي حقوق قانونية أو غيرها للتحديث.

UPDATE_COLLECTION "update:collection"

المصفوفة التي تحتوي على قائمة الحزم للتحديث.

UPDATE_COLLECTION_NAME "update:collection:name"

اسم الحزمة التي تم تحديثها.

UPDATE_COLLECTION_EVR "update:collection:evr"

إصدار الحزمة التي تم تحديثها.

UPDATE_COLLECTION_ARCH "update:collection:arch"

هيكلية الحزمة التي تم تحديثها.

UPDATE_COLLECTION_FILENAME "update:collection:filename"

اسم ملف الحزمة التي تم تحديثها.

UPDATE_REFERENCE "update:reference"

المصفوفة التي تحتوي على قائمة المراجع للتحديث.

UPDATE_REFERENCE_TYPE "update:reference:type"

نوع المرجع، على سبيل المثال، bugzilla.

UPDATE_REFERENCE_HREF "update:reference:href"

عنوان URL للمرجع.

UPDATE_REFERENCE_ID "update:reference:id"

سلسلة التعريف للمرجع، على سبيل المثال، رقم الخطأ.

UPDATE_REFERENCE_TITLE "update:reference:title"

عنوان المرجع، على سبيل المثال، ملخص الخطأ.

PRODUCT_REFERENCEFILE "product:referencefile"

اسم الملف الأساسي للمنتج في الحزمة.

PRODUCT_SHORTLABEL "product:shortlabel"

سلسلة تعريف للمنتج.

PRODUCT_DISTPRODUCT "product:distproduct"

مهمل، لا تستخدم. كان اسم منتج SUSE Code-10.

PRODUCT_DISTVERSION "product:distversion"

مهمل، لا تستخدم. كان إصدار منتج SUSE Code-10.

PRODUCT_TYPE "product:type"

نوع المنتج، على سبيل المثال، "base".

PRODUCT_URL "product:url"

مصفوفة من عناوين URL للمنتجات.

PRODUCT_URL_TYPE "product:url:type"

مصفوفة من أنواع عناوين URL للمنتجات.

PRODUCT_FLAGS "product:flags"

مصفوفة من علامات المنتج.

PRODUCT_PRODUCTLINE "product:productline"

سلسلة خط منتج تستخدم لتسجيل المنتج.

PRODUCT_REGISTER_TARGET "product:regtarget"

هدف لتسجيل المنتج.

PRODUCT_REGISTER_RELEASE "product:regrelease"

سلسلة إصدار لتسجيل المنتج.

PUBKEY_KEYID "pubkey:keyid"

معرف المفتاح العام (keyid)، يتكون من 8 بايت بالترميز الست عشري.

PUBKEY_FINGERPRINT "pubkey:fingerprint"

البصمة الرقمية للمفتاح العام، عادة ما تكون ملخص sha1sum بالترميز الست عشري. المفاتيح RSA القديمة من النوع V3 تستخدم ملخص md5sum بدلاً من ذلك.

PUBKEY_EXPIRES "pubkey:expires"

عدد الثواني منذ بداية حقبة يونكس عند انتهاء صلاحية المفتاح العام.

PUBKEY_SUBKEYOF "pubkey:subkeyof"

معرف المفتاح العام الرئيسي للمفاتيح الفرعية.

PUBKEY_DATA "pubkey:data"

بيانات MPI للمفتاح العام.

SOLVABLE_ISVISIBLE "solvable:isvisible"

سمة تصف ما إذا كان يجب عرض الحزمة للمستخدم أم لا. تستخدم لأنماط SUSE.

SOLVABLE_CATEGORY "solvable:category"

فئة النمط.

SOLVABLE_INCLUDES "solvable:includes"

قائمة بالأنماط الأخرى التي يتضمنها هذا النمط.

SOLVABLE_EXTENDS "solvable:extends"

قائمة بالأنماط الأخرى التي يوسعها هذا النمط.

SOLVABLE_ICON "solvable:icon"

اسم أيقونة النمط.

SOLVABLE_ORDER "solvable:order"

مؤشر ترتيب للنمط.

SUSETAGS_SHARE_NAME "susetags:share:name"

سمة داخلية لتنفيذ سجلات susetags المشتركة. تحتفظ باسم الحزمة المستخدم لمشاركة السمات.

SUSETAGS_SHARE_EVR "susetags:share:evr"

سمة داخلية لتنفيذ سجلات susetags المشتركة. تحتفظ بإصدار الحزمة المستخدم لمشاركة السمات.

SUSETAGS_SHARE_ARCH "susetags:share:arch"

سمة داخلية لتنفيذ سجلات susetags المشتركة. تحتفظ بهيكلية الحزمة المستخدم لمشاركة السمات.

هياكل الحزم

قيم الهيكلية المعرفة مسبقًا للهياكل الشائعة الاستخدام.

ARCH_SRC "src"

تستخدم للحزم الثنائية التي تحتوي على مصادر الحزمة.

ARCH_NOSRC "nosrc"

تستخدم للحزم الثنائية التي تحتوي على بعض مصادر الحزمة، ولكن ليس كلها (بسبب القيود).

ARCH_NOARCH "noarch"

يمكن تثبيت هذه الحزمة على أي هيكلية. تستخدم لـ rpm.

ARCH_ALL "all"

يمكن تثبيت هذه الحزمة على أي هيكلية. تستخدم لـ Debian.

ARCH_ANY "any"

يمكن تثبيت هذه الحزمة على أي هيكلية. تستخدم لـ Archlinux و Haiku.

معرفات التبعيات

المساحات الاسمية هي معدلات خاصة تغير معنى التبعية. يتم إنشاء تبعيات المساحة الاسمية باستخدام علامة REL_NAMESPACE. لجعل مساحات الأسماء المخصصة تعمل، يجب عليك تنفيذ وظيفة رد اتصال للمساحة الاسمية.

تقوم علامات التبعية بتقسيم مصفوفة التبعية إلى جزأين بمعاني مختلفة.

NAMESPACE_MODALIAS "namespace:modalias"

التبعية هي تبعية modalias خاصة تطابق الأجهزة المثبتة.

NAMESPACE_SPLITPROVIDES "namespace:splitprovides"

التبعية هي تبعية splitprovides خاصة تستخدم لتنفيذ التحديثات التي تتضمن تقسيم حزمة. تحتوي تبعية splitprovides على اسم ملف واسم حزمة، ويتم مطابقتها إذا تم تثبيت حزمة بالاسم المقدم الذي يحتوي على اسم الملف. تم تنفيذ مساحة الاسم هذه في libsolv، لذلك لا تحتاج إلى رد اتصال.

NAMESPACE_LANGUAGE "namespace:language"

التبعية تصف لغة. يجب أن تعيد وظيفة رد الاتصال القيمة true إذا تم اختيار اللغة من قبل المستخدم.

NAMESPACE_FILESYSTEM "namespace:filesystem"

التبعية تصف نظام ملفات. يجب أن تعيد وظيفة رد الاتصال القيمة true إذا كان نظام الملفات مطلوبًا.

NAMESPACE_OTHERPROVIDERS "namespace:otherproviders"

هذه حيلة للسماح بالحزم المتعارضة ذاتيًا. ليست هناك حاجة إليها مع إصدار rpm الحالي، لذا لا تستخدم مساحة الاسم هذه.

SOLVABLE_PREREQMARKER "solvable:prereqmarker"

تقوم هذه العلامة بفصل التبعيات المطلوبة العادية عن المتطلبات المسبقة. ليست هناك حاجة إليها لحل التبعيات، ولكن يتم استخدامها بواسطة خوارزمية ترتيب المعاملات عند وجود دورة تبعية n

هذا الموقع غير مرتبط بـ Arch Linux ولا تابع له بأي شكل من الأشكال. المحتوى هو ترجمة فقط لصفحات دليل Arch المنشورة على man.archlinux.org.
