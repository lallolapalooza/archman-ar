libressl-DH_set_method(3)

الحزمة: libressl

التراخيص: ISC, custom:OpenSSL

DH_SET_METHOD(3) دليل وظائف المكتبة DH_SET_METHOD(3)

الاسم

DH_set_default_method، DH_get_default_method، DH_set_method، DH_new_method، DH_OpenSSL — تحديد طريقة DH

الملخص

مكتبة "libcrypto"
#include <openssl/dh.h>

void
DH_set_default_method(const DH_METHOD *meth);

const DH_METHOD *
DH_get_default_method(void);

int
DH_set_method(DH *dh, const DH_METHOD *meth);

DH *
DH_new_method(ENGINE *engine);

const DH_METHOD *
DH_OpenSSL(void);

الوصف

يحتوي كائن DH_METHOD على مؤشرات إلى الوظائف المستخدمة في عمليات Diffie-Hellman. افتراضيًا، يتم استخدام التنفيذ الداخلي الذي تم إرجاعه بواسطة DH_OpenSSL(). من خلال اختيار طريقة أخرى، يمكن استخدام تطبيقات بديلة مثل مسرعات الأجهزة.

DH_set_default_method() تختار meth كطريقة افتراضية لجميع هياكل DH التي يتم إنشاؤها لاحقًا.

DH_get_default_method() ترجع مؤشرًا إلى الطريقة الافتراضية الحالية.

DH_set_method() تختار meth لتنفيذ جميع العمليات باستخدام المفتاح dh. هذا يحل محل DH_METHOD التي يستخدمها مفتاح dh. من الممكن أن يكون لدى مفاتيح DH التي تعمل فقط مع تطبيقات DH_METHOD معينة، وفي مثل هذه الحالات، قد يكون لمحاولة تغيير DH_METHOD للمفتاح نتائج غير متوقعة.

DH_new_method() تخصص وتهيئ هيكل DH. يتم تجاهل الوسيط engine ويتم استخدام الطريقة الافتراضية التي يتم التحكم فيها بواسطة DH_set_default_method().

يتم تعريف هيكل DH_METHOD على النحو التالي:

typedef struct dh_meth_st
{
     /* اسم التطبيق */
	const char *name;

     /* إنشاء قيم DH الخاصة والعامة لاتفاق المفتاح */
        int (*generate_key)(DH *dh);

     /* حساب السر المشترك */
        int (*compute_key)(unsigned char *key, BIGNUM *pub_key, DH *dh);

     /* حساب r = a ^ p mod m (قد يكون NULL لبعض التطبيقات) */
        int (*bn_mod_exp)(DH *dh, BIGNUM *r, BIGNUM *a, const BIGNUM *p,
                                const BIGNUM *m, BN_CTX *ctx,
                                BN_MONT_CTX *m_ctx);

     /* يتم استدعاؤه عند DH_new */
        int (*init)(DH *dh);

     /* يتم استدعاؤه عند DH_free */
        int (*finish)(DH *dh);

        int flags;

        char *app_data; /* ?? */

} DH_METHOD;

قيم الإرجاع

DH_OpenSSL() و DH_get_default_method() ترجع مؤشرات إلى DH_METHOD المعنية.

DH_set_method() ترجع 1 عند النجاح أو 0 عند الفشل. حاليًا، لا يمكن أن تفشل.

DH_new_method() ترجع NULL وتعيين رمز خطأ يمكن الحصول عليه بواسطة ERR_get_error(3) إذا فشل التخصيص. بخلاف ذلك، فإنها ترجع مؤشرًا إلى الهيكل المخصص حديثًا.
