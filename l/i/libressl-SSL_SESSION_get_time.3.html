libressl-SSL_SESSION_get_time(3)

الحزمة: libressl

التراخيص: ISC, custom:OpenSSL

SSL_SESSION_GET_TIME(3) دليل وظائف المكتبة SSL_SESSION_GET_TIME(3)

الاسم

SSL_SESSION_get_time، SSL_SESSION_set_time، SSL_SESSION_get_timeout، SSL_SESSION_set_timeout، SSL_get_time، SSL_set_time، SSL_get_timeout، SSL_set_timeout — استرجاع وتعديل أوقات الجلسة وإعدادات مهلة الانتهاء

الملخص

المكتبة “libssl” libcrypto
#include <openssl/ssl.h>

long
SSL_SESSION_get_time(const SSL_SESSION *s);

long
SSL_SESSION_set_time(SSL_SESSION *s, long tm);

long
SSL_SESSION_get_timeout(const SSL_SESSION *s);

long
SSL_SESSION_set_timeout(SSL_SESSION *s, long tm);

long
SSL_get_time(const SSL_SESSION *s);

long
SSL_set_time(SSL_SESSION *s, long tm);

long
SSL_get_timeout(const SSL_SESSION *s);

long
SSL_set_timeout(SSL_SESSION *s, long tm);

الوصف

SSL_SESSION_get_time() تُرجع الوقت الذي تم فيه إنشاء الجلسة s. يُعطى الوقت بالثواني منذ بداية الحقبة، وبالتالي فهو متوافق مع الوقت الذي توفره الدالة time(3).

SSL_SESSION_set_time() تستبدل وقت إنشاء الجلسة s بالقيمة المختارة tm.

SSL_SESSION_get_timeout() تُرجع قيمة مهلة الانتهاء المحددة للجلسة s بالثواني.

SSL_SESSION_set_timeout() تحدد قيمة مهلة الانتهاء للجلسة s بالثواني إلى tm.

الدوال SSL_get_time()، SSL_set_time()، SSL_get_timeout()، و SSL_set_timeout() هي أسماء مستعارة لنظائرها من الدوال SSL_SESSION_*().

يتم انتهاء صلاحية الجلسات بفحص وقت الإنشاء وقيمة مهلة الانتهاء. يتم تعيين كليهما وقت إنشاء الجلسة إلى الوقت الفعلي وقيمة مهلة الانتهاء الافتراضية في وقت الإنشاء، على التوالي، كما تم تعيينه بواسطة SSL_CTX_set_timeout(3). باستخدام هذه الدوال، من الممكن تمديد أو تقصير فترة حياة الجلسة.

قيم الإرجاع

SSL_SESSION_get_time() و SSL_SESSION_get_timeout() تُرجع القيم الصالحة حاليًا.

SSL_SESSION_set_time() و SSL_SESSION_set_timeout() تُرجع 1 عند النجاح.

إذا تم تمرير المؤشر NULL للجلسة s إلى أي من الدوال، يتم إرجاع 0.

انظر أيضًا

التاريخ

SSL_get_time()، SSL_get_timeout()، و SSL_set_timeout() ظهرت في SSLeay 0.4 أو أقدم. SSL_set_time() ظهرت لأول مرة في SSLeay 0.5.2. SSL_SESSION_get_time()، SSL_SESSION_set_time()، SSL_SESSION_get_timeout()، و SSL_SESSION_set_timeout() ظهرت لأول مرة في SSLeay 0.9.0. كل هذه الدوال كانت متاحة منذ OpenBSD 2.4.
