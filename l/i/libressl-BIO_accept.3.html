libressl-BIO_accept(3)

الحزمة: libressl

التراخيص: ISC, custom:OpenSSL

الاسم

BIO_sock_non_fatal_error, BIO_sock_should_retry, BIO_set_tcp_ndelay, BIO_get_host_ip, BIO_get_port, BIO_get_accept_socket, BIO_accept, BIO_socket_nbio - وظائف إدارة سوكتات BIO

الموجز

#include <openssl/bio.h>
int BIO_sock_non_fatal_error(int errnum);
int BIO_sock_should_retry(int retval);
int BIO_set_tcp_ndelay(BIO *bio, int on);
int BIO_get_host_ip(const char *hostname, unsigned char *ip);
int BIO_get_port(const char *host_port, unsigned char *ip, unsigned short *port);
BIO *BIO_get_accept_socket(const char *host_port, int prepend_newline);
BIO *BIO_accept(BIO *bio);
int BIO_socket_nbio(BIO *bio, int nonblock);

الوصف

تتعامل هذه الوظائف مع تكوين سوكتات BIO والتحقق من الأخطاء.

BIO_sock_non_fatal_error()
تتحقق مما إذا كان errnum يمثل خطأ غير قاتل في سوكت.
BIO_sock_should_retry()
تتحقق مما إذا كان يجب إعادة محاولة عملية سوكت بناءً على قيمة الإرجاع والخطأ.
BIO_set_tcp_ndelay()
تضبط أو تلغي خيار TCP_NODELAY على سوكت BIO.
BIO_get_host_ip()
تحصل على عنوان IP الخاص بمضيف معين.
BIO_get_port()
تحصل على عنوان IP ورقم المنفذ من سلسلة نصية.
BIO_get_accept_socket()
تنشئ سوكت استماع جاهزًا لقبول الاتصالات.
BIO_accept()
تقبل اتصالًا واردًا على سوكت استماع BIO.
BIO_socket_nbio()
تضبط أو تلغي وضع الإدخال/الإخراج غير الحاجب (non-blocking) لسوكت BIO.

قيم الإرجاع

BIO_sock_non_fatal_error() تُرجع 1 إذا كان errnum هو EAGAIN، EALREADY، EINPROGRESS، EINTR، أو ENOTCONN، و 0 بخلاف ذلك، حتى لو كان errnum يساوي 0.

BIO_sock_should_retry() تُرجع 1 إذا كان BIO_sock_non_fatal_error(errno) يساوي 1 وكان retval يساوي 0 أو -1، أو 0 بخلاف ذلك.

BIO_set_tcp_ndelay() تُرجع 0 عند النجاح أو -1 عند الفشل.

الأخطاء

إذا فشلت BIO_get_host_ip()، BIO_get_port()، أو BIO_get_accept_socket() أو فشلت BIO_accept() بشكل قاتل، يمكن استرداد التشخيصات التالية باستخدام ERR_get_error(3)، ERR_GET_REASON(3)، و ERR_reason_error_string(3):

BIO_R_ACCEPT_ERROR "خطأ قبول"
accept(2) فشلت بشكل قاتل في BIO_accept().
BIO_R_BAD_HOSTNAME_LOOKUP "بحث اسم مضيف سيء"
getaddrinfo(3) فشل أو كان hostname NULL في BIO_get_host_ip()، أو getaddrinfo(3) فشل في BIO_get_accept_socket().
BIO_R_INVALID_ARGUMENT "وسيط غير صالح"
getaddrinfo(3) فشل في BIO_get_port().
ERR_R_MALLOC_FAILURE "فشل malloc"
فشل تخصيص الذاكرة في BIO_get_accept_socket()، أو BIO_accept() نجحت لكنها لم تتمكن من تخصيص ذاكرة لـ *addr. بالنسبة لـ BIO_accept()، فإن واصف الملف المُرجع صالح، ويمكن محاولة الاتصال مع الطرف الآخر باستخدامه.
BIO_R_NO_PORT_SPECIFIED "لم يتم تحديد منفذ"
كان الوسيط servname NULL في BIO_get_port()، أو كان host_port NULL أو انتهى بعد النقطتين الأوليين في BIO_get_accept_socket().
BIO_R_NULL_PARAMETER "وسيط فارغ"
كان الوسيط addr NULL في BIO_accept().
BIO_R_UNABLE_TO_BIND_SOCKET "غير قادر على ربط السوكت"
bind(2) فشل في BIO_get_accept_socket().
BIO_R_UNABLE_TO_CREATE_SOCKET "غير قادر على إنشاء سوكت"
socket(2) فشل في BIO_get_accept_socket().
BIO_R_UNABLE_TO_LISTEN_SOCKET "غير قادر على الاستماع على السوكت"
listen(2) فشل في BIO_get_accept_socket().

التاريخ

BIO_sock_should_retry() ظهرت لأول مرة في SSLeay 0.6.5 والوظائف الأخرى باستثناء BIO_socket_nbio() في SSLeay 0.8.0. كانت جميعها متاحة منذ OpenBSD 2.4.

BIO_socket_nbio() ظهرت لأول مرة في SSLeay 0.9.1 وكانت متاحة منذ OpenBSD 2.6.
