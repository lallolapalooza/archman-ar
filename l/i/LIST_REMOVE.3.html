LIST(3) دليل الوظائف القياسية (libc، -lc) LIST(3)

NAME

LIST_EMPTY, LIST_ENTRY, LIST_FIRST, LIST_FOREACH, LIST_HEAD, LIST_HEAD_INITIALIZER, LIST_INIT, LIST_INSERT_AFTER, LIST_INSERT_BEFORE, LIST_INSERT_HEAD, LIST_NEXT, LIST_REMOVE - تطبيق قائمة مزدوجة

LIBRARY

مكتبة C القياسية (libc، -lc)

SYNOPSIS

#include <sys/queue.h>
LIST_ENTRY(TYPE);
LIST_HEAD(HEADNAME, TYPE);
LIST_HEAD LIST_HEAD_INITIALIZER(LIST_HEAD head);
void LIST_INIT(LIST_HEAD *head);
int LIST_EMPTY(LIST_HEAD *head);
void LIST_INSERT_HEAD(LIST_HEAD *head,
                        struct TYPE *elm, LIST_ENTRY NAME);
void LIST_INSERT_BEFORE(struct TYPE *listelm,
                        struct TYPE *elm, LIST_ENTRY NAME);
void LIST_INSERT_AFTER(struct TYPE *listelm,
                        struct TYPE *elm, LIST_ENTRY NAME);
struct TYPE *LIST_FIRST(LIST_HEAD *head);
struct TYPE *LIST_NEXT(struct TYPE *elm, LIST_ENTRY NAME);
LIST_FOREACH(struct TYPE *var, LIST_HEAD *head, LIST_ENTRY NAME);
void LIST_REMOVE(struct TYPE *elm, LIST_ENTRY NAME);

الوصف

تعرف هذه الماكرو وتتعامل مع القوائم المزدوجة.

في تعريفات الماكرو، TYPE هو اسم هيكل معرف من قبل المستخدم، والذي يجب أن يحتوي على حقل من نوع LIST_ENTRY، ويسمى NAME. الوسيط HEADNAME هو اسم هيكل معرف من قبل المستخدم يجب تعريفه باستخدام الماكرو LIST_HEAD().

الإنشاء

تُدار القائمة بواسطة هيكل معرف باستخدام الماكرو LIST_HEAD(). يحتوي هذا الهيكل على مؤشر للعنصر الأول في القائمة. ترتبط العناصر بشكل مزدوج، مما يسمح بإزالة أي عنصر دون اجتياز القائمة. يمكن إضافة عناصر جديدة إلى القائمة بعد عنصر موجود، قبل عنصر موجود، أو في بداية القائمة. يتم تعريف هيكل LIST_HEAD هكذا:


LIST_HEAD(HEADNAME, TYPE) head;

حيث struct HEADNAME هو الهيكل المراد تعريفه، و struct TYPE هو نوع العناصر المراد ربطها في القائمة. يمكن لاحقًا تعريف عنوان مؤشر إلى رأس القائمة هكذا:


struct HEADNAME *headp;

(أسماء head و headp قابلة للاختيار من قبل المستخدم.)

LIST_ENTRY() يعرف هيكلًا يربط عناصر القائمة.

LIST_HEAD_INITIALIZER() يُقيّم إلى مُهيئ للقائمة head.

LIST_INIT() يقوم بتهيئة القائمة المشار إليها بواسطة head.

LIST_EMPTY() يُقيّم إلى صحيح إذا كانت القائمة فارغة.

الإدراج

LIST_INSERT_HEAD() يدرج العنصر الجديد elm في بداية القائمة.

LIST_INSERT_BEFORE() يدرج العنصر الجديد elm قبل العنصر listelm.

LIST_INSERT_AFTER() يدرج العنصر الجديد elm بعد العنصر listelm.

التجوال

LIST_FIRST() يعيد العنصر الأول في القائمة، أو NULL إذا كانت القائمة فارغة.

LIST_NEXT() يعيد العنصر التالي في القائمة، أو NULL إذا كان الأخير.

LIST_FOREACH() يتجول في القائمة المشار إليها بواسطة head في الاتجاه الأمامي، ويخصص كل عنصر للمتغير var بالتناوب.

الإزالة

LIST_REMOVE() يزيل العنصر elm من القائمة.

قيمة الإرجاع

LIST_EMPTY() يعيد قيمة غير صفرية إذا كانت القائمة فارغة، وصفر إذا كانت تحتوي على عنصر واحد على الأقل.

LIST_FIRST() و LIST_NEXT() يعيدان مؤشرًا إلى هيكل TYPE الأول أو التالي، على التوالي.

LIST_HEAD_INITIALIZER() يعيد مُهيئًا يمكن تعيينه للقائمة head.

المعايير

BSD.

التاريخ

4.4BSD.

أخطاء

LIST_FOREACH() لا يسمح بإزالة أو تحرير var داخل الحلقة، لأن ذلك سيتداخل مع التجوال. LIST_FOREACH_SAFE()، الموجود في BSDs ولكن ليس في glibc، يحل هذه المشكلة بالسماح بإزالة وتحرير var داخل الحلقة دون التأثير على التجوال.

أمثلة

#include <stddef.h>
#include <stdio.h>
#include <stdlib.h>
#include <sys/queue.h>
struct entry {
    int data;
    LIST_ENTRY(entry) entries;              /* قائمة */
};
LIST_HEAD(listhead, entry);
int
main(void)
{
    struct entry *n1, *n2, *n3, *np;
    struct listhead head;                   /* رأس القائمة */
    int i;
    LIST_INIT(&head);                       /* تهيئة القائمة */
    n1 = malloc(sizeof(struct entry));      /* الإدراج في البداية */
    LIST_INSERT_HEAD(&head, n1, entries);
    n2 = malloc(sizeof(struct entry));      /* الإدراج بعد */
    LIST_INSERT_AFTER(n1, n2, entries);
    n3 = malloc(sizeof(struct entry));      /* الإدراج قبل */
    LIST_INSERT_BEFORE(n2, n3, entries);
    i = 0;                                  /* التجوال الأمامي */
    LIST_FOREACH(np, &head, entries)
        np->data = i++;
    LIST_REMOVE(n2, entries);               /* الإزالة */
    free(n2);
                                            /* التجوال الأمامي */
    LIST_FOREACH(np, &head, entries)
        printf("%i\n", np->data);
                                            /* إزالة القائمة */
    n1 = LIST_FIRST(&head);
    while (n1 != NULL) {
        n2 = LIST_NEXT(n1, entries);
        free(n1);
        n1 = n2;
    }
    LIST_INIT(&head);
    exit(EXIT_SUCCESS);
}
