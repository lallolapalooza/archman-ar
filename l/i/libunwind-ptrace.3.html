libunwind-ptrace(3)

الحزمة: libunwind

التراخيص: MIT

LIBUNWIND-PTRACE(3libunwind) مكتبة البرمجة LIBUNWIND-PTRACE(3libunwind)

الاسم

libunwind-ptrace -- دعم ptrace() في libunwind

الملخص

#include <libunwind-ptrace.h>

unw_accessors_t _UPT_accessors;

void *_UPT_create(pid_t);
void _UPT_destroy(void *);

int _UPT_find_proc_info(unw_addr_space_t, unw_word_t, unw_proc_info_t *, int, void *);
void _UPT_put_unwind_info(unw_addr_space_t, unw_proc_info_t *, void *);
int _UPT_get_dyn_info_list_addr(unw_addr_space_t, unw_word_t *, void *);
int _UPT_access_mem(unw_addr_space_t, unw_word_t, unw_word_t *, int, void *);
int _UPT_access_reg(unw_addr_space_t, unw_regnum_t, unw_word_t *, int, void *);
int _UPT_access_fpreg(unw_addr_space_t, unw_regnum_t, unw_fpreg_t *, int, void *);
int _UPT_get_proc_name(unw_addr_space_t, unw_word_t, char *, size_t, unw_word_t *, void *);
int _UPT_resume(unw_addr_space_t, unw_cursor_t *, void *);

الوصف

تجعل استدعاء النظام ptrace(2) من الممكن لعملية الحصول على وصول إلى حالة الجهاز والذاكرة الافتراضية لعملية أخرى. مع مجموعة صحيحة من روتينات رد النداء، من الممكن لذلك ربط libunwind بعملية أخرى عبر ptrace(2). في حين أنه ليس من الصعب جدًا القيام بذلك مباشرة، فإن libunwind يسهل هذه المهمة أكثر من خلال توفير روتينات جاهزة للاستخدام لهذا الغرض. الروتينات والمتغيرات التي تنفذ هذه الميزة تستخدم بادئة اسم _UPT، والتي تقف لـ "unwind via ptrace".

التطبيق الذي يرغب في استخدام موجه libunwind ptrace البعيد يحتاج إلى اتخاذ الخطوات التالية.

1.
إنشاء مساحة عنوان libunwind جديدة تمثل العملية المستهدفة. يتم ذلك عن طريق استدعاء unw_create_addr_space(). في كثير من الحالات، سيكون التطبيق ببساطة يرغب في تمرير عنوان _UPT_accessors كوسيطة أولى لهذا الروتين. القيام بذلك سيضمن أن libunwind ستكون قادرة على فك ضغط العملية المستهدفة بشكل صحيح.
2.
تفعيل وضع ptrace على العملية المستهدفة، إما عن طريق إنشاء عملية جديدة، استدعاء PTRACE_TRACEME، ثم بدء البرنامج المستهدف (عبر execve(2))، أو عن طريق إرفاق مباشرة بعملية قيد التشغيل بالفعل (عبر PTRACE_ATTACH).
3.
بمجرد معرفة معرف العملية (pid) للعملية المستهدفة، يمكن إنشاء بنية معلومات UPT عن طريق استدعاء _UPT_create()، مع تمرير pid للعملية المستهدفة كوسيطة وحيدة.
4.
يجب بعد ذلك تمرير المؤشر المبهم الذي تم إرجاعه كوسيطة "arg" (الوسيطة الثالثة) إلى unw_init_remote().

في ظروف خاصة، قد يفضل التطبيق استخدام أجزاء فقط من موجه libunwind ptrace البعيد. لهذا السبب، تتوفر روتينات رد النداء الفردية (_UPT_find_proc_info()، _UPT_put_unwind_info()، إلخ) أيضًا للاستخدام المباشر. بالطبع، يمكن أيضًا الحصول على عناوين هذه الروتينات من _UPT_accessors، ولكن القيام بذلك سيمنع التهيئة الثابتة. أيضًا، عند استخدام _UPT_accessors، سيتم ربط جميع روتينات رد النداء في التطبيق، حتى لو لم يتم استدعاؤها فعليًا.

يمكن استخدام الروتين _UPT_resume() لاستئناف تنفيذ العملية المستهدفة. يقوم ببساطة باستدعاء ptrace(2) بقيمة أمر PTRACE_CONT.

عندما ينتهي التطبيق من استخدام libunwind على العملية المستهدفة، يجب استدعاء _UPT_destroy()، وتمرير المؤشر المبهم الذي تم إرجاعه من استدعاء _UPT_create() إليه. هذا يضمن تحرير جميع الذاكرة والموارد الأخرى.

التوفر

بما أن ptrace(2) يعمل داخل آلة واحدة فقط، فإن موجه libunwind ptrace بحكم تعريفه غير متوفر في إصدارات libunwind المعدة للفك المتقاطع.

سلامة الخيوط

يفترض موجه libunwind ptrace البعيد أن بنية معلومات UPT واحدة لا تتم مشاركتها أبدًا بين الخيوط. بسبب هذا، لا يتم استخدام قفل صريح. طالما أن خيطًا واحدًا فقط يستخدم بنية معلومات UPT في أي وقت معين، فإن هذه الميزة آمنة للخيوط.

قيمة الإرجاع

قد ترجع _UPT_create() مؤشرًا فارغًا إذا فشلت في إنشاء بنية معلومات UPT لأي سبب. بالنسبة للتنفيذ الحالي، السبب الوحيد الذي قد يفشل فيه هذا الاستدعاء هو عندما يكون النظام خارج الذاكرة.

الملفات

libunwind-ptrace.h

ملف الرأس الذي يجب تضمينه عند استخدام الواجهة المعرفة بواسطة هذه المكتبة.
-lunwind-ptrace -lunwind-generic

مفاتيح الربط التي يجب إضافتها عند بناء برنامج يستخدم الوظائف المعرفة بواسطة هذه المكتبة.

مثال

    #include <libunwind-ptrace.h>
    #include <stdio.h>
    #include <stdlib.h>
    int
    main (int argc, char **argv)
    {
      if (argc != 2) {
        fprintf (stderr, "usage: %s PID\n", argv[0]);
        exit (EXIT_FAILURE);
      }
      char *endptr;
      pid_t target_pid = strtoul (argv[1], &endptr, 10);
      if (target_pid == 0 && argv[1] == endptr) {
        fprintf (stderr, "usage: %s PID\n", argv[0]);
        exit (EXIT_FAILURE);
      }
      unw_addr_space_t as = unw_create_addr_space (&_UPT_accessors, 0);
      if (!as) {
        fprintf (stderr, "unw_create_addr_space() failed");
        exit (EXIT_FAILURE);
      }
      void *ui = _UPT_create (target_pid);
      if (!ui) {
        fprintf (stderr, "_UPT_create() failed");
        exit (EXIT_FAILURE);
      }
      unw_cursor_t cursor;
      int ret = unw_init_remote (&cursor, as, ui);
      if (ret < 0) {
        fprintf (stderr, "unw_init_remote() failed: ret=%d\n", ret);
        exit (EXIT_FAILURE);
      }
      do {
        unw_proc_info_t pi;
        ret = unw_get_proc_info (&cursor, &pi);
        if (ret == -UNW_ENOINFO) {
          fprintf (stdout, "no info\n");
        } else if (ret >= 0) {
          printf ("\tproc=%#016lx-%#016lx\thandler=%#016lx lsda=%#016lx",
                  (long) pi.start_ip, (long) pi.end_ip,
                  (long) pi.handler, (long) pi.lsda);
        }
        ret = unw_step (&cursor);
      } while (ret > 0);
      if (ret < 0) {
        fprintf (stderr, "unwind failed with ret=%d\n", ret);
        exit (EXIT_FAILURE);
      }
      _UPT_destroy (ui);
      unw_destroy_addr_space (as);
      exit (EXIT_SUCCESS);
    }
