libssh2_trace_sethandler(3)

الحزمة: libssh2

التراخيص: BSD

libssh2_trace_sethandler(3) libssh2 libssh2_trace_sethandler(3)

الاسم

libssh2_trace_sethandler - تعيين معالج إخراج التتبع

الملخص

#include <libssh2.h>
typedef void (*libssh2_trace_handler_func)(LIBSSH2_SESSION *session,
                                           void *context,
                                           const char *data,
                                           size_t length);
int
libssh2_trace_sethandler(LIBSSH2_SESSION *session,
                         void *context,
                         libssh2_trace_handler_func callback);

الوصف

يقوم libssh2_trace_sethandler بتثبيت معالج إخراج التتبع لتطبيقك. بشكل افتراضي، عند تشغيل التتبع عبر استدعاء libssh2_trace()، يتم كتابة جميع المخرجات إلى stderr. عن طريق استدعاء هذه الدالة وتمرير مؤشر دالة يطابق النموذج الأولي لـ libssh2_trace_handler_func، ستستدعي libssh2 مرة أخرى أثناء إنشاء مخرجات التتبع. يمكن استخدام هذا لالتقاط مخرجات التتبع ووضعها في ملف سجل أو نافذة تشخيص. لا يكون لهذه الدالة أي تأثير ما لم يتم بناء libssh2 لدعم هذا الخيار، وقد لا يكون "بناء الإصدار" النموذجي كذلك.

يمكن استخدام context لتمرير بيانات عشوائية محددة من قبل المستخدم مرة أخرى إلى رد الاتصال عند استدعائها.
