LIST_VACIO(3) أدلة الوظائف القياسية LIST_VACIO(3)

NAME

LIST_VACIO, LIST_ENTRY, LIST_PRIMERO, LIST_FOREACH, LIST_PRIMERO, LIST_HEAD, LIST_HEAD_INITIALIZER, LIST_INIT, LIST_VACIO, LIST_INSERT_AFTER, LIST_INSERT_BEFORE, LIST_INSERT_HEAD, LIST_PRIMERO, LIST_SUCESOR, LIST_REMOVE - تنفيذ قائمة مزدوجة

LIBRARY

مكتبة وظائف C القياسية (libc,  -lc)

SYNOPSIS

#include <sys/queue.h>
LIST_ENTRY(TYPE);
LIST_HEAD(HEADNAME, TYPE);
LIST_HEAD LIST_HEAD_INITIALIZER(LIST_HEAD head);
void LIST_INIT(LIST_HEAD *head);
int LIST_VACIO(LIST_HEAD *head);
void LIST_INSERT_HEAD(LIST_HEAD *head,
                        struct TYPE *elm, LIST_ENTRY NAME);
void LIST_INSERT_BEFORE(struct TYPE *listelm,
                        struct TYPE *elm, LIST_ENTRY NAME);
void LIST_INSERT_AFTER(struct TYPE *listelm,
                        struct TYPE *elm, LIST_ENTRY NAME);
struct TYPE *LIST_PRIMERO(LIST_HEAD *head);
struct TYPE *LIST_SUCESOR(struct TYPE *elm, LIST_ENTRY NAME);
LIST_FOREACH(struct TYPE *var, LIST_HEAD *head, LIST_ENTRY NAME);
void LIST_REMOVE(struct TYPE *elm, LIST_ENTRY NAME);

DESCRIPTION

تعرف هذه الماكرو وتقوم بالعمليات على قوائم مزدوجة.

في تعريفات الماكرو، TYPE هو اسم هيكل معرف من قبل المستخدم، والذي يجب أن يحتوي على حقل من النوع LIST_ENTRY، مسمى NAME. الوسيط HEADNAME هو اسم هيكل معرف من قبل المستخدم، والذي يجب تعريفه باستخدام الماكرو LIST_HEAD().

إنشاء

يتم ترؤيس القائمة بواسطة هيكل معرف بواسطة الماكرو LIST_HEAD(). يحتوي هذا الهيكل على مؤشر واحد للعنصر الأول في القائمة. ترتبط العناصر بطريقة مزدوجة، مما يسمح بإزالة أي عنصر دون اجتياز القائمة. يمكن إضافة عناصر جديدة إلى القائمة بعد عنصر موجود، قبل عنصر موجود، أو في بداية القائمة. يتم تعريف هيكل LIST_HEAD على النحو التالي:


LIST_HEAD(HEADNAME, TYPE) head;

حيث struct HEADNAME هو الهيكل المراد تعريفه و struct TYPE هو نوع العناصر المرتبطة بالقائمة. يمكن بعد ذلك تعريف عنوان مؤشر بداية القائمة على النحو التالي:


struct HEADNAME *headp;

(الاسمين head و headp يمكن للمستخدم اختيارهما.)

LIST_ENTRY() يعرّف هيكلًا يربط عناصر القائمة.

LIST_HEAD_INITIALIZER() يُقيّم إلى مُهيئ للقائمة head.

LIST_INIT() يُهيئ القائمة التي يشير إليها head.

LIST_VACIO() يُرجع صحيحًا إذا كانت القائمة فارغة.

إدراج

LIST_INSERT_HEAD() يُدرج العنصر الجديد elm في بداية القائمة.

LIST_INSERT_BEFORE() يُدرج العنصر الجديد elm قبل العنصر listelm.

LIST_INSERT_AFTER() يُدرج العنصر الجديد elm بعد العنصر listelm.

اجتياز

LIST_PRIMERO() يُرجع العنصر الأول في القائمة، أو NULL إذا كانت القائمة فارغة.

LIST_SUCESOR() يُرجع العنصر التالي في القائمة، أو NULL إذا كان هو الأخير.

LIST_FOREACH() يجتاز القائمة المشار إليها بواسطة head في الاتجاه الأمامي، مع تعيين كل عنصر إلى var.

إزالة

LIST_REMOVE() يزيل العنصر elm من القائمة.

قيمة الإرجاع

LIST_VACIO() يُرجع قيمة غير صفرية إذا كانت القائمة فارغة، وصفرًا إذا كانت تحتوي على عنصر واحد على الأقل.

LIST_PRIMERO() و LIST_SUCESOR() يُرجعان مؤشرًا إلى هيكل من النوع TYPE، على التوالي.

LIST_HEAD_INITIALIZER() يُرجع مُهيئ يمكن تعيينه للقائمة head.

المعايير

BSD.

التاريخ

4.4BSD.

مشاكل

LIST_FOREACH() لا يسمح بإزالة var أو تحريره داخل الحلقة، حيث قد يتداخل ذلك مع الاجتياز. LIST_FOREACH_SAFE()، الموجود في BSD ولكن ليس في glibc، يحل هذه المشكلة بالسماح بإزالة var من القائمة وتحريره داخل الحلقة دون التداخل مع الاجتياز.

أمثلة

#include <stddef.h>
#include <stdio.h>
#include <stdlib.h>
#include <sys/queue.h>
struct entry {
    int data;
    LIST_ENTRY(entry) entries;              /* قائمة */
};
LIST_HEAD(listhead, entry);
int
main(void)
{
    struct entry *n1, *n2, *n3, *np;
    struct listhead head;                   /* رأس القائمة */
    int i;
    LIST_INIT(&head);                       /* تهيئة القائمة */
    n1 = malloc(sizeof(struct entry));      /* إدراج في البداية */
    LIST_INSERT_HEAD(&head, n1, entries);
    n2 = malloc(sizeof(struct entry));      /* إدراج بعد */
    LIST_INSERT_AFTER(n1, n2, entries);
    n3 = malloc(sizeof(struct entry));      /* إدراج قبل */
    LIST_INSERT_BEFORE(n2, n3, entries);
    i = 0;                                  /* اجتياز أمامي */
    LIST_FOREACH(np, &head, entries)
        np->data = i++;
    LIST_REMOVE(n2, entries);               /* إزالة */
    free(n2);
                                            /* اجتياز أمامي */
    LIST_FOREACH(np, &head, entries)
        printf("%i\n", np->data);
                                            /* إزالة */
    n1 = LIST_PRIMERO(&head);
    while (n1 != NULL) {
        n2 = LIST_SUCESOR(n1, entries);
        free(n1);
        n1 = n2;
    }
    LIST_INIT(&head);
    exit(EXIT_SUCCESS);
}
