libressl-HMAC(3)

الحزمة: libressl

التراخيص: ISC, custom:OpenSSL

HMAC(3) دليل وظائف المكتبة HMAC(3)

الاسم

HMAC، HMAC_CTX_new، HMAC_CTX_reset، HMAC_CTX_free، HMAC_Init_ex، HMAC_Update، HMAC_Final، HMAC_CTX_copy، HMAC_CTX_set_flags، HMAC_CTX_get_md، HMAC_size — رمز مصادقة رسالة HMAC

الملخص

المكتبة “libcrypto”
#include <openssl/hmac.h>

unsigned char *

HMAC_CTX *
HMAC_CTX_new(void);

int
HMAC_CTX_reset(HMAC_CTX *ctx);

void
HMAC_CTX_free(HMAC_CTX *ctx);

int
HMAC_Init_ex(HMAC_CTX *ctx، const void *key، int key_len، const EVP_MD *md، ENGINE *engine);

int
HMAC_Update(HMAC_CTX *ctx، const unsigned char *data، size_t len);

int
HMAC_Final(HMAC_CTX *ctx، unsigned char *md، unsigned int *len);

int
HMAC_CTX_copy(HMAC_CTX *dctx، HMAC_CTX *sctx);

HMAC_CTX_set_flags(HMAC_CTX *ctx، unsigned long flags);

const EVP_MD *
HMAC_CTX_get_md(const HMAC_CTX *ctx);

الوصف

HMAC هو MAC (رمز مصادقة الرسالة)، أي وظيفة تجزئة مفتاحية تُستخدم لمصادقة الرسائل، وهي تستند إلى وظيفة تجزئة.

HMAC() يحسب رمز مصادقة الرسالة لـ n بايت في d باستخدام وظيفة التجزئة evp_md والمفتاح key بطول key_len بايت.

يضع النتيجة في md، والتي يجب أن تكون لديها مساحة لمخرجات وظيفة التجزئة، والتي لا تزيد عن EVP_MAX_MD_SIZE بايت. يتم وضع حجم المخرجات في md_len، ما لم يكن NULL.

evp_md يمكن أن يكون EVP_sha1(3)، EVP_ripemd160(3)، إلخ.

HMAC_CTX_new() يخصص ويهيئ كائن HMAC_CTX جديد.

HMAC_CTX_reset() يقوم بمسح ويهيئ ctxHMAC_CTX_free() وتم إنشاؤه حديثًا باستخدام HMAC_CTX_new().

HMAC_CTX_free() يمسح المفتاح والبيانات الأخرى من ctx، ويحرر أي موارد مرتبطة، ثم يحرر ctx نفسه.

يمكن استخدام الوظائف التالية إذا لم يتم تخزين الرسالة بالكامل في الذاكرة:

HMAC_Init_ex() يُعد أو يعيد استخدام ctxevp_md والمفتاح key. يمكن أن يكون أي منهما NULL، وفي هذه الحالة يتم إعادة استخدام الموجود. يجب أن يكون ctxHMAC_CTX_new() قبل الاستخدام الأول في هذه الوظيفة. إذا تم استدعاء HMAC_Init_ex() مع NULL key ولكن evp_md ليس NULL وليس نفس الهضم السابق الذي استخدمه ctx، فسيتم إرجاع خطأ لأن إعادة استخدام مفتاح موجود مع هضم مختلف غير مدعومة. يتم تجاهل الوسيط ENGINE *engine دائمًا، ويوصى بتمرير NULL.

HMAC_Update() يمكن استدعاؤه بشكل متكرر مع أجزاء من الرسالة المراد مصادقتها (len بايت في data).

HMAC_Final() يضع رمز مصادقة الرسالة في md، والذي يجب أن يكون لديه مساحة لمخرجات وظيفة التجزئة.

HMAC_CTX_copy() ينسخ كل الحالة الداخلية من sctx إلى dctx.

HMAC_CTX_set_flags() يطبق العلامات المحددة على كائنات EVP_MD_CTX الداخلية. قيم العلامات الممكنة EVP_MD_CTX_FLAG_* محددة في <openssl/evp.h>.

HMAC_size() يعيد طول مخرجات وظيفة التجزئة الأساسية بالبايت. يتم تنفيذه كـ ماكرو.

قيم الإرجاع

HMAC() يعيد مؤشرًا إلى رمز مصادقة الرسالة أو NULL إذا حدث خطأ.

HMAC_CTX_new() يعيد مؤشرًا إلى كائن HMAC_CTX الجديد أو NULL إذا حدث خطأ.

HMAC_CTX_reset()، HMAC_Init_ex()، HMAC_Update()، HMAC_Final()، و HMAC_CTX_copy() تعيد 1 للنجاح أو 0 إذا حدث خطأ.

HMAC_CTX_get_md() يعيد خوارزمية التجزئة التي تم تعيينها سابقًا لـ ctx باستخدام HMAC_Init_ex()، أو NULL إذا لم يتم تعيين أي منها.

HMAC_size() يعيد الطول بالبايت لمخرجات وظيفة التجزئة الأساسية أو 0 عند حدوث خطأ.

انظر أيضًا

CMAC_Init(3)، EVP_DigestInit(3)

المعايير

RFC 2104

التاريخ

HMAC()، HMAC_Update()، HMAC_Final()، و HMAC_size() ظهرت لأول مرة في SSLeay 0.9.0 وكانت متاحة منذ OpenBSD 2.4.

HMAC_Init_ex() ظهرت لأول مرة في OpenSSL 0.9.7 وكانت متاحة منذ OpenBSD 3.2.

HMAC_CTX_set_flags() ظهرت لأول مرة في OpenSSL 0.9.7f وكانت متاحة منذ OpenBSD 3.8.
