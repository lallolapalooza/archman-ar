libkeccak_rawshakesum_fd(3)

الحزمة: libkeccak

التراخيص: مخصص:ISC

LIBKECCAK_RAWSHAKESUM_FD(3) دليل وظائف المكتبة LIBKECCAK_RAWSHAKESUM_FD(3)

الاسم

libkeccak_rawshakesum_fd - حساب مجموع تجزئة RawSHAKE لملف

الملخص

#include <libkeccak.h>
int libkeccak_rawshakesum_fd(int fd, struct libkeccak_state *state, long int semicapacity, long int output, void *hashsum);

اربط باستخدام -lkeccak.

الوصف

تقوم وظيفة libkeccak_rawshakesum_fd() بحساب مجموع تجزئة RawSHAKE لملف، يتم تحديد واصف الملف الخاص به بواسطة fd (ويجب أن يكون في بداية الملف.) يتم ضبط خوارزمية التجزئة بواسطة المعلمات semicapacity و output؛ فهما تحددان نصف السعة وحجم الإخراج، على التوالي، بالبت.

يتم تخزين التجزئة في شكل ثنائي في hashsum. يجب أن يكون حجم تخصيص hashsum على الأقل (((output + 7) / 8) * sizeof(char)).

يجب عدم تهيئة *state. تقوم libkeccak_rawshakesum_fd() بتهيئة *state بنفسها. لذلك سيكون هناك تسرب للذاكرة إذا كانت *state مهيأة بالفعل.

قيم الإرجاع

ترجع وظيفة libkeccak_rawshakesum_fd() القيمة 0 عند اكتمال التنفيذ بنجاح. عند حدوث خطأ، يتم إرجاع -1 ويتم تعيين errno لوصف الخطأ.

الأخطاء

قد تفشل وظيفة libkeccak_rawshakesum_fd() لأي سبب، باستثناء تلك التي تؤدي إلى تعيين errno إلى EINTR، والمحددة للوظائف read(2)، malloc(3)، و realloc(3).

ملاحظات

كن على علم، تقوم libkeccak_rawshakesum_fd() بتجزئة الملف حتى الوصول إلى النهاية. بالنسبة للأنابيب والمقابس، يعني هذا حتى إغلاق الملف. ولكن بالنسبة لأجهزة الأحرف، يعني هذا عادةً عدم الانتهاء أبدًا. لذلك، فإن محاولة تجزئة الملفات في /dev هي عادة فكرة سيئة. لا تتحقق libkeccak_rawshakesum_fd() من طول الملف أو نوعه قبل التجزئة لأن هذا يمكن أن يحد مما يمكنك القيام به، ويجعل المكتبة أكثر تعقيدًا.

libkeccak_rawshakesum_fd() لا تتوقف إذا تمت مقاطعتها (read(2) ترجع EINTR.)

تفترض libkeccak_rawshakesum_fd() أن جميع المعلومات غير حساسة، ولن تقوم بالتالي بأي مسح آمن للمعلومات.

libkeccak_rawshakesum_fd() لا تتحقق من ضبط الخوارزمية.

مثال

يحسب هذا المثال تجزئة RawSHAKE256(, 512) للمدخلات من الإدخال القياسي، ويطبع التجزئة، في شكل سداسي عشري، إلى الإخراج القياسي.

struct libkeccak_state state;
if (libkeccak_rawshakesum_fd(STDIN_FILENO, &state, 256, 512, binhash) < 0)
	goto fail;
libkeccak_behex_lower(hexhash, binhash, sizeof(binhash));
printf("%s\n", hexhash);
libkeccak_state_destroy(&state);
