Path::IsDev::Role::Matcher::Child::BaseName::MatchRegexp(3) وثائق Perl مساهم بها من المستخدم Path::IsDev::Role::Matcher::Child::BaseName::MatchRegexp(3)

الاسم

Path::IsDev::Role::Matcher::Child::BaseName::MatchRegexp - المطابقة عندما يحتوي المسار على ملف فرعي يطابق تعبيرًا

الإصدار

version 1.001003

الطرق

"child_basename_matchregexp"

$class->child_basename_matchregexp( $result_object, $regexp );

نظرًا لتعبير عادي $regexp، قم بالمطابقة إذا كان أي من "$result_object->path->children" يطابق التعبير العادي المعطى.

if ( $self->child_basename_matchregexp( $result_object, qr/^Change(.*)$/i ) ) {
    # يحتوي result_object->path() على طفل واحد على الأقل يطابق التعبير العادي
}

الطرق الخاصة

if ( $class->_this_child_matchregexp( $result_object, $child_path, $regexp ) ) {
    ...
