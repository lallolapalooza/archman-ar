الاسم

pam_winbind.conf - ملف إعدادات وحدة PAM لوحدة Winbind

الوصف

يتضمن ملف الإعدادات هذا مجموعة samba(7).

pam_winbind.conf هو ملف الإعدادات لوحدة PAM pam_winbind. انظر pam_winbind(8) لمزيد من التفاصيل.

الملخص

ملف إعدادات pam_winbind.conf هو ملف إعدادات كلاسيكي من نوع ini. يوجد قسم واحد فقط (عام) حيث يتم تعريف خيارات مختلفة.

الخيارات

يدعم pam_winbind العديد من الخيارات التي يمكن تعيينها إما في ملفات إعدادات PAM أو في ملف إعدادات pam_winbind الموجود في /etc/security/pam_winbind.conf. الخيارات من ملف إعدادات PAM لها الأسبقية على تلك الموجودة في ملف إعدادات pam_winbind.conf.

debug = yes|no

يعطي مخرجات تصحيح الأخطاء إلى syslog. الافتراضي هو "no".

debug_state = yes|no

يعطي مخرجات تصحيح أخطاء مفصلة لحالة PAM إلى syslog. الافتراضي هو "no".

require_membership_of = [SID أو الاسم]

إذا تم تعيين هذا الخيار، فستنجح pam_winbind فقط إذا كان المستخدم عضوًا في SID أو الاسم المحدد. يمكن أن يكون SID هو SID مجموعة، أو SID اسم مستعار، أو حتى SID مستخدم. من الممكن أيضًا إعطاء اسم بدلاً من SID. يجب أن يكون لهذا الاسم الشكل: MYDOMAIN\mygroup أو MYDOMAIN\myuser (حيث يمثل الحرف '\' قيمة معلمة winbind separator). من الممكن أيضًا استخدام UPN بالشكل user@REALM أو group@REALM. ستقوم pam_winbind، في هذه الحالة، بالبحث عن SID داخليًا. لاحظ أن الاسم قد لا يحتوي على أي مسافات. لذلك يوصى باستخدام SIDs فقط. يمكنك التحقق من قائمة SIDs التي ينتمي إليها المستخدم باستخدام wbinfo --user-sids=SID. هذا الإعداد فارغ افتراضيًا.

يعمل هذا الخيار فقط أثناء مصادقة كلمة المرور، ولن يقيد الوصول إذا لم تكن كلمة المرور مطلوبة لأي سبب (مثل تسجيل الدخول المستند إلى مفتاح SSH).

try_first_pass = yes|no

افتراضيًا، تحاول pam_winbind الحصول على رمز المصادقة من وحدة سابقة. إذا لم يكن هناك رمز متاح، فإنها تطلب من المستخدم كلمة المرور القديمة. مع هذا الخيار، تتوقف pam_winbind بخطأ إذا لم يكن هناك رمز مصادقة من وحدة سابقة. إذا لم تكن كلمة المرور الأساسية صالحة، فستطالب PAM بكلمة مرور. الافتراضي هو "no".

krb5_auth = yes|no

يمكن لـ pam_winbind المصادقة باستخدام Kerberos عندما تتواصل winbindd مع متحكم نطاق Active Directory. يجب تمكين مصادقة Kerberos بهذا المعلمة. عندما لا تنجح مصادقة Kerberos (على سبيل المثال، بسبب عدم تزامن الساعة)، ستعود winbindd إلى مصادقة samlogon عبر MSRPC. عند استخدام هذه المعلمة بالاقتران مع winbind refresh tickets، ستحافظ winbind على تحديث تذكرة منح التذاكر (TGT) الخاصة بك عن طريق تحديثها عند الضرورة. الافتراضي هو "no".

krb5_ccache_type = [type]

عندما يتم تكوين pam_winbind لمحاولة مصادقة Kerberos عن طريق تمكين خيار krb5_auth، يمكنها تخزين تذكرة منح التذاكر (TGT) المستلمة في ذاكرة تخزين مؤقت للاعتماديات. يمكن التحكم في نوع ذاكرة التخزين المؤقت للاعتماديات بهذا الخيار. القيم المدعومة هي: KCM أو KEYRING (عندما تدعمها مكتبة Kerberos الخاصة بالنظام ونظام التشغيل)، FILE و DIR (عندما تدعم مكتبة Kerberos الخاصة بالنظام نوع DIR). في حالة FILE، سيتم إنشاء ذاكرة تخزين مؤقت للاعتماديات بالشكل /tmp/krb5cc_UID - في حالة DIR تحتاج إلى تحديد دليل. يتم استبدال UID بمعرف المستخدم الرقمي. يتم إنشاء دليل UID. يجب أن يكون المسار حتى الدليل موجودًا بالفعل. تحقق من تفاصيل تطبيق Kerberos.

عند استخدام نوع KEYRING، الآلية المدعومة هي "KEYRING:persistent:UID"، والتي تستخدم ذاكرة تخزين مفاتيح نواة Linux لتخزين الاعتماديات على أساس كل UID. لـ KEYRING قيودها. نظرًا لأنها ذاكرة نواة آمنة، على سبيل المثال، لا يمكن تخزين كميات كبيرة من الاعتماديات.

عند استخدام نوع KCM، الآلية المدعومة هي "KCM:UID"، والتي تستخدم مدير اعتماديات Kerberos لتخزين الاعتماديات على أساس كل UID بشكل مشابه لـ KEYRING. هذا هو الخيار الموصى به في أحدث توزيعات Linux، حيث يوفر مدير اعتماديات Kerberos. إذا لم يكن الأمر كذلك، فنحن نقترح استخدام KEYRING لأنه الطريقة الأكثر أمانًا وقابلية للتنبؤ.

من الممكن أيضًا تحديد مسارات ملفات مخصصة واستخدام النمط "%u" لاستبدال معرف المستخدم الرقمي. أمثلة:

krb5_ccache_type = DIR:/run/user/%u/krb5cc

سيؤدي هذا إلى إنشاء ملف ذاكرة تخزين مؤقت للاعتماديات في الدليل المحدد.

krb5_ccache_type = FILE:/tmp/krb5cc_%u

سيؤدي هذا إلى إنشاء ملف ذاكرة تخزين مؤقت للاعتماديات.

اتركه فارغًا لإجراء مصادقة Kerberos فقط دون وجود ذاكرة تخزين مؤقت للتذكرة بعد نجاح تسجيل الدخول. هذا الإعداد فارغ افتراضيًا.

cached_login = yes|no

تسمح Winbind بتسجيل الدخول باستخدام الاعتماديات المخزنة مؤقتًا عندما يكون winbind offline logon ممكّنًا. لاستخدام هذه الميزة من وحدة PAM، يجب تعيين هذا الخيار. الافتراضي هو "no".

silent = yes|no

لا تصدر أي رسائل. الافتراضي هو "no".

mkhomedir = yes|no

إنشاء دليل المنزل للمستخدم عند الطلب، الخيار صالح في كتلة جلسة PAM. الافتراضي هو "no".

warn_pwd_expire = days

يحدد عدد الأيام قبل أن تبدأ pam_winbind في التحذير بشأن كلمات المرور التي ستنتهي صلاحيتها. الافتراضي هو 14 يومًا.

pwd_change_prompt = yes|no

توليد مطالبة لتغيير كلمة مرور منتهية الصلاحية. الافتراضي هو "no".
