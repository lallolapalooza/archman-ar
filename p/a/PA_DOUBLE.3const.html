المكتبة

مكتبة C القياسية (libc, -lc)

ملخص

#include <printf.h>
int register_printf_specifier(int spec, printf_function func,
                              printf_arginfo_size_function arginfo);
int register_printf_modifier(const wchar_t *str);
int register_printf_type(printf_va_arg_function fct);

الدوال المرجعية

typedef int printf_function(FILE *stream, const struct printf_info *info,
                              const void *const args[]);
typedef int printf_arginfo_size_function(const struct printf_info *info,
                              size_t n, int argtypes[n], int size[n]);
typedef void printf_va_arg_function(void *mem, va_list *ap);

الوصف

تُستخدم هذه الدوال لتوسيع و/أو تعديل سلوك عائلة دوال printf(3).

register_printf_specifier()

تسجل هذه الدالة محول تنسيق مخصص لعائلة دوال printf(3).

spec
الحرف الذي سيتم استخدامه كمحول تنسيق في سلسلة التنسيق.
func
الدالة المرجعية التي سيتم تنفيذها بواسطة عائلة دوال printf(3) لتنسيق وسائط الإدخال إلى تدفق الإخراج stream.
stream
تدفق الإخراج الذي سيتم طباعة النتيجة إليه. يمثل هذا التدفق النتيجة بشفافية، حتى في حالات الدوال التي تكتب إلى سلسلة.
info
الهيكل الذي يحتوي على معلومات السياق، بما في ذلك المعدلات المحددة في سلسلة التنسيق. يحتوي هذا على نفس محتويات arginfo.
args
مصفوفة من المؤشرات إلى الوسائط التي سيتم استخدامها في دالة شبيهة بـ printf(3).
arginfo
الدالة المرجعية التي سيتم تنفيذها بواسطة عائلة دوال printf(3) لمعرفة عدد الوسائط التي يجب تحليلها لمحول التنسيق المخصص وكذلك أنواعها.
info
الهيكل الذي يحتوي على معلومات السياق، بما في ذلك المعدلات المحددة في سلسلة التنسيق. يحتوي هذا على نفس محتويات func.
n
عدد الوسائط المتبقية للتحليل.
argtypes
يجب تعيين هذا المصفوفة لتحديد نوع كل وسيط سيتم تحليله. يمثل كل عنصر أحد الوسائط التي سيتم تحليلها، بنفس الترتيب الذي يتم تمريرها به إلى دالة شبيهة بـ printf(3). يجب تعيين كل عنصر إلى نوع أساسي (PA_*) من التعداد أعلاه، أو نوع مخصص، و ORed اختياريًا مع معدل طول (PA_FLAG_*).

يتم تحديد النوع باستخدام أحد الثوابت التالية:

PA_INT
int.
PA_CHAR
int، محول إلى char.
PA_WCHAR
wchar_t.
PA_STRING
const char*، سلسلة منتهية بـ '\0'.
PA_WSTRING
const wchar_t*، سلسلة أحرف عريضة منتهية بـ L'\0'.
PA_POINTER
void*.
PA_FLOAT
float.
PA_DOUBLE
double.
PA_LAST
TODO.
size
للأنواع المخصصة، يجب أيضًا تحديد حجمها (بالبايت) من خلال هذه المصفوفة. بخلاف ذلك، لا تستخدمها.

يتم استدعاء arginfo قبل func ويهيئ المعلومات اللازمة لاستدعاء func.

قيمة الإرجاع

register_printf_specifier() و register_printf_modifier() و register_printf_type() تُرجع 0 عند النجاح، أو -1 عند حدوث خطأ.

الدوال المرجعية

يجب أن تُرجع الدالة المرجعية من نوع printf_function عدد الأحرف المكتوبة، أو -1 عند حدوث خطأ.

يجب أن تُرجع الدالة المرجعية من نوع printf_arginfo_size_function عدد الوسائط المراد تحليلها بواسطة محول التنسيق هذا. كما أنها تمرر معلومات حول نوع تلك الوسائط إلى الاستدعاء من خلال argtypes. عند حدوث خطأ، يجب أن تُرجع -1.

الأخطاء

EINVAL
المحدد لم يكن حرفًا صالحًا.
