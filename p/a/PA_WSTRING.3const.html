المعايير

GNU.

التاريخ

register_printf_function(3) هي دالة أقدم مماثلة لـ register_printf_specifier()، وهي الآن مهملة. لا يمكن لهذه الدالة التعامل مع الأنواع المعرفة من قبل المستخدم.

register_printf_specifier() تحل محل register_printf_function(3).

أمثلة

يسجل مثال البرنامج التالي مواصفات 'b' و 'B' لطباعة الأعداد الصحيحة بتنسيق ثنائي، مما يعكس قواعد مواصفات التحويل غير الموقعة الأخرى مثل 'x' و 'u'. يمكن استخدام هذا للطباعة بالصيغة الثنائية قبل C23.

#include <limits.h>
#include <stddef.h>
#include <stdint.h>
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <sys/param.h>
#include <printf.h>
#define GROUP_SEP  '\'
struct Printf_Pad {
    char    ch;
    size_t  len;
};
static int b_printf(FILE *stream, const struct printf_info *info,
                    const void *const args[]);
static int b_arginf_sz(const struct printf_info *info,

static size_t b_pad_len(const struct printf_info *info,
                    ptrdiff_t bin_len);

                    const struct printf_info *info);
static ssize_t b_pad_zeros(FILE *stream, const struct printf_info *info,
                    ptrdiff_t min_len);
static ssize_t b_print_number(FILE *stream,
                    const struct printf_info *info,
                    const char bin[UINTMAX_WIDTH],
                    size_t min_len, size_t bin_len);
static char pad_ch(const struct printf_info *info);
static ssize_t pad_spaces(FILE *stream, size_t pad_len);

static int
b_printf(FILE *stream, const struct printf_info *info,
         const void *const args[])
{
    char               bin[UINTMAX_WIDTH];
    size_t             min_len, bin_len;
    ssize_t            len, tmp;
    struct Printf_Pad  pad = {0};
    len = 0;
    min_len = b_bin_repr(bin, info, args[0]);
    bin_len = b_bin_len(info, min_len);
    pad.ch = pad_ch(info);
    if (pad.ch == ' ')
        pad.len = b_pad_len(info, bin_len);
    /* الحشو بـ ' ' (محاذاة لليمين) */
    if ((pad.ch == ' ') && !info->left) {
        tmp = pad_spaces(stream, pad.len);
        if (tmp == EOF)
            return EOF;
        len += tmp;
    }
    /* بادئة "0b"/"0B" */
    if (info->alt) {
        tmp = b_print_prefix(stream, info);
        if (tmp == EOF)
            return EOF;
        len += tmp;
    }
    /* الحشو بـ '0' */
    if (pad.ch == '0') {
        tmp = b_pad_zeros(stream, info, min_len);
        if (tmp == EOF)
            return EOF;
        len += tmp;
    }
    /* طباعة الرقم (بما في ذلك الأصفار البادئة لملء الدقة) */
    tmp = b_print_number(stream, info, bin, min_len, bin_len);
    if (tmp == EOF)
        return EOF;
    len += tmp;
    /* الحشو بـ ' ' (محاذاة لليسار) */
    if (info->left) {
        tmp = pad_spaces(stream, pad.len);
        if (tmp == EOF)
            return EOF;
        len += tmp;
    }
    return len;
}
static int
b_arginf_sz(const struct printf_info *info, size_t n, int argtypes[n],
            [[maybe_unused]] int size[n])
{
    if (n < 1)
        return -1;
    if (info->is_long_double)
        argtypes[0] = PA_INT | PA_FLAG_LONG_LONG;
    else if (info->is_long)
        argtypes[0] = PA_INT | PA_FLAG_LONG;
    else
        argtypes[0] = PA_INT;
    return 1;
}
static uintmax_t
b_value(const struct printf_info *info, const void *arg)
{
    if (info->is_long_double)
        return *(const unsigned long long *)arg;
    if (info->is_long)
        return *(const unsigned long *)arg;
    /* short and char are both promoted to int */
    return *(const unsigned int *)arg;
}
static size_t
b_bin_repr(char bin[UINTMAX_WIDTH],
           const struct printf_info *info, const void *arg)
{
    size_t     min_len;
    uintmax_t  val;
    val = b_value(info, arg);
    bin[0] = '0';
    for (min_len = 0; val; min_len++) {
        bin[min_len] = '0' + (val % 2);
        val >>= 1;
    }
    return MAX(min_len, 1);
}
static size_t
b_bin_len(const struct printf_info *info, ptrdiff_t min_len)
{
    return MAX(info->prec, min_len);
}
static size_t
b_pad_len(const struct printf_info *info, ptrdiff_t bin_len)
{
    ptrdiff_t  pad_len;
    pad_len = info->width - bin_len;
    if (info->alt)
        pad_len -= 2;
    if (info->group)
        pad_len -= (bin_len - 1) / 4;
    return MAX(pad_len, 0);
}
static ssize_t
b_print_prefix(FILE *stream, const struct printf_info *info)
{
    ssize_t len;
    len = 0;
    if (fputc('0', stream) == EOF)
        return EOF;
    len++;
    if (fputc(info->spec, stream) == EOF)
        return EOF;
    len++;
    return len;
}
static ssize_t
b_pad_zeros(FILE *stream, const struct printf_info *info,
            ptrdiff_t min_len)
{
    ssize_t    len;
    ptrdiff_t  tmp;
    len = 0;
    tmp = info->width - (info->alt * 2);
    if (info->group)
        tmp -= tmp / 5 - !(tmp % 5);
    for (ptrdiff_t i = tmp - 1; i > min_len - 1; i--) {
        if (fputc('0', stream) == EOF)
            return EOF;
        len++;
        if (!info->group || (i % 4))
            continue;
        if (fputc(GROUP_SEP, stream) == EOF)
            return EOF;
        len++;
    }
    return len;
}
static ssize_t
b_print_number(FILE *stream, const struct printf_info *info,
               const char bin[UINTMAX_WIDTH],
               size_t min_len, size_t bin_len)
{
    ssize_t  len;
    len = 0;
    /* طباعة الأصفار البادئة لملء الدقة */
    for (size_t i = bin_len - 1; i > min_len - 1; i--) {
        if (fputc('0', stream) == EOF)
            return EOF;
        len++;
        if (!info->group || (i % 4))
            continue;
        if (fputc(GROUP_SEP, stream) == EOF)
            return EOF;
        len++;
    }
    /* طباعة الرقم */
    for (size_t i = min_len - 1; i < min_len; i--) {
        if (fputc(bin[i], stream) == EOF)
            return EOF;
        len++;
        if (!info->group || (i % 4) || !i)
            continue;
        if (fputc(GROUP_SEP, stream) == EOF)
            return EOF;
        len++;
    }
    return len;
}
static char
pad_ch(const struct printf_info *info)
{
    if ((info->prec != -1) || (info->pad == ' ') || info->left)
        return ' ';
    return '0';
}
static ssize_t
pad_spaces(FILE *stream, size_t pad_len)
{
    ssize_t  len;
    len = 0;
    for (size_t i = pad_len - 1; i < pad_len; i--) {
        if (fputc(' ', stream) == EOF)
            return EOF;
        len++;
    }
    return len;
}

PA_WSTRING(3const)

PA_WSTRING - Description of PA_WSTRING

LIBRARY

libpulse (part of PulseAudio)

SYNOPSIS

#include <pulse/introspect.h>

extern const char * const PA_WSTRING[];

DESCRIPTION

PA_WSTRING is an array of strings, each representing a state or status of a PulseAudio object. These strings are human-readable and can be used for debugging or display purposes.

RETURN VALUE

Not applicable. PA_WSTRING is a global constant array.

ERRORS

Not applicable.

STANDARDS

Not applicable. This is a PulseAudio-specific constant.

HISTORY

Part of PulseAudio since version 0.9.10.

EXAMPLES

While you can't directly use PA_WSTRING in a typical program without knowing the exact index, it's primarily used internally by PulseAudio functions. For instance, a function might use an integer value to index into this array to get a descriptive string for a state.

/* This is a conceptual example, as direct usage is uncommon */
int state_index = 5; /* Assuming 5 corresponds to a valid state */
if (state_index >= 0 && state_index < sizeof(PA_WSTRING) / sizeof(PA_WSTRING[0])) {
    const char *state_string = PA_WSTRING[state_index];
    printf("Current state: %s\n", state_string);
}
