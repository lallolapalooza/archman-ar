تسجيل_المحددات_التحويل

تسجل هذه الدالة محدد تحويل مخصص لوظائف عائلة printf(3).

محدد
الحرف الذي سيُستخدم كمحدد تحويل في سلسلة التنسيق.
وظيفة
دالة الاستدعاء التي سيتم تنفيذها بواسطة وظائف عائلة printf(3) لتنسيق وسائط الإدخال في تدفق الإخراج تدفق.
تدفق
تدفق الإخراج الذي سيتم طباعة نتيجة التنسيق عليه. يتم تمثيل هذا التدفق بشفافية حتى في حالات الوظائف التي تكتب إلى سلسلة نصية.
معلومات
الهيكل الذي يحتوي على معلومات السياق، بما في ذلك المعدلات المحددة في سلسلة التنسيق. يحتوي هذا على نفس محتويات معلومات_المعامل.
المعاملات
مصفوفة من المؤشرات إلى الوسائط التي يتم تمريرها إلى وظيفة شبيهة بـ printf(3).
معلومات_المعامل
دالة الاستدعاء التي سيتم تنفيذها بواسطة وظائف عائلة printf(3) لمعرفة عدد الوسائط التي يجب تحليلها لمحدد التحويل المخصص وأنواعها أيضًا.
معلومات
الهيكل الذي يحتوي على معلومات السياق، بما في ذلك المعدلات المحددة في سلسلة التنسيق. يحتوي هذا على نفس محتويات وظيفة.
عدد
عدد الوسائط المتبقية للتحليل.
أنواع_المعاملات
يجب تعيين هذه المصفوفة لتحديد نوع كل وسيط سيتم تحليله. يمثل كل عنصر في المصفوفة أحد الوسائط المراد تحليلها، بنفس الترتيب الذي يتم تمريره به إلى وظيفة شبيهة بـ printf(3). يجب تعيين كل عنصر إلى نوع أساسي (PA_) من التعداد أعلاه، أو نوع مخصص، ويمكن دمجه اختياريًا مع معدل طول (PA_FLAG_*).

PA_POINTER
void *.
PA_FLOAT

قيمة الإرجاع

تسجيل_المحددات_التحويل() و تسجيل_معدلات_التحويل() و تسجيل_نوع_التحويل() ترجع 0 عند النجاح، أو -1 عند حدوث خطأ.

دوال الاستدعاء_2

يجب أن ترجع دالة الاستدعاء من نوع وظيفة_تحويل_printf عدد الأحرف المكتوبة، أو -1 عند حدوث خطأ.

يجب أن ترجع دالة الاستدعاء من نوع وظيفة_حجم_معلومات_المعامل عدد الوسائط المراد تحليلها بواسطة هذا المحدد. كما أنها تمرر معلومات حول نوع هذه الوسائط إلى الاستدعاء من خلال أنواع_المعاملات. عند حدوث خطأ، يجب أن ترجع -1.

التاريخ

register_printf_function(3) هي دالة أقدم مشابهة لـ register_printf_specifier()، وهي الآن مهملة. لا يمكن لتلك الدالة التعامل مع الأنواع المعرفة من قبل المستخدم.

register_printf_specifier() تحل محل register_printf_function(3).

static int
b_printf(FILE *stream, const struct printf_info *info,
         const void *const args[])
{
    char               bin[UINTMAX_WIDTH];
    size_t             min_len, bin_len;
    ssize_t            len, tmp;
    struct Printf_Pad  pad = {0};
    len = 0;
    min_len = b_bin_repr(bin, info, args[0]);
    bin_len = b_bin_len(info, min_len);
    pad.ch = pad_ch(info);
    if (pad.ch == ' ')
        pad.len = b_pad_len(info, bin_len);
    /* التبطين بـ ' ' (محاذاة لليمين) */
    if ((pad.ch == ' ') && !info->left) {
        tmp = pad_spaces(stream, pad.len);
        if (tmp == EOF)
            return EOF;
        len += tmp;
    }
    /* بادئة "0b"/"0B" */
    if (info->alt) {
        tmp = b_print_prefix(stream, info);
        if (tmp == EOF)
            return EOF;
        len += tmp;
    }
    /* التبطين بـ '0' */
    if (pad.ch == '0') {
        tmp = b_pad_zeros(stream, info, min_len);
        if (tmp == EOF)
            return EOF;
        len += tmp;
    }
    /* طباعة الرقم (بما في ذلك الأصفار البادئة لملء الدقة) */
    tmp = b_print_number(stream, info, bin, min_len, bin_len);
    if (tmp == EOF)
        return EOF;
    len += tmp;
    /* التبطين بـ ' ' (محاذاة لليسار) */
    if (info->left) {
        tmp = pad_spaces(stream, pad.len);
        if (tmp == EOF)
            return EOF;
        len += tmp;
    }
    return len;
}
static int
b_arginf_sz(const struct printf_info *info, size_t n, int argtypes[n],
            [[maybe_unused]] int size[n])
{
    if (n < 1)
        return -1;
    if (info->is_long_double)
        argtypes[0] = PA_INT | PA_FLAG_LONG_LONG;
    else if (info->is_long)
        argtypes[0] = PA_INT | PA_FLAG_LONG;
    else
        argtypes[0] = PA_INT;
    return 1;
}
static uintmax_t
b_value(const struct printf_info *info, const void *arg)
{
    if (info->is_long_double)
        return *(const unsigned long long *)arg;
    if (info->is_long)
        return *(const unsigned long *)arg;
    /* يتم ترقية short و char كلاهما إلى int */
    return *(const unsigned int *)arg;
}
static size_t
b_bin_repr(char bin[UINTMAX_WIDTH],
           const struct printf_info *info, const void *arg)
{
    size_t     min_len;
    uintmax_t  val;
    val = b_value(info, arg);
    bin[0] = '0';
    for (min_len = 0; val; min_len++) {
        bin[min_len] = '0' + (val % 2);
        val >>= 1;
    }
    return MAX(min_len, 1);
}
static size_t
b_bin_len(const struct printf_info *info, ptrdiff_t min_len)
{
    return MAX(info->prec, min_len);
}
static size_t
b_pad_len(const struct printf_info *info, ptrdiff_t bin_len)
{
    ptrdiff_t  pad_len;
    pad_len = info->width - bin_len;
    if (info->alt)
        pad_len -= 2;
    if (info->group)
        pad_len -= (bin_len - 1) / 4;
    return MAX(pad_len, 0);
}
static ssize_t
b_print_prefix(FILE *stream, const struct printf_info *info)
{
    ssize_t len;
    len = 0;
    if (fputc('0', stream) == EOF)
        return EOF;
    len++;
    if (fputc(info->spec, stream) == EOF)
        return EOF;
    len++;
    return len;
}
static ssize_t
b_pad_zeros(FILE *stream, const struct printf_info *info,
            ptrdiff_t min_len)
{
    ssize_t    len;
    ptrdiff_t  tmp;
    len = 0;
    tmp = info->width - (info->alt * 2);
    if (info->group)
        tmp -= tmp / 5 - !(tmp % 5);
    for (ptrdiff_t i = tmp - 1; i > min_len - 1; i--) {
        if (fputc('0', stream) == EOF)
            return EOF;
        len++;
        if (!info->group || (i % 4))
            continue;
        if (fputc(GROUP_SEP, stream) == EOF)
            return EOF;
        len++;
    }
    return len;
}
static ssize_t
b_print_number(FILE *stream, const struct printf_info *info,
               const char bin[UINTMAX_WIDTH],
               size_t min_len, size_t bin_len)
{
    ssize_t  len;
    len = 0;
    /* طباعة الأصفار البادئة لملء الدقة */
    for (size_t i = bin_len - 1; i > min_len - 1; i--) {
        if (fputc('0', stream) == EOF)
            return EOF;
        len++;
        if (!info->group || (i % 4))
            continue;
        if (fputc(GROUP_SEP, stream) == EOF)
            return EOF;
        len++;
    }
    /* طباعة الرقم */
    for (size_t i = min_len - 1; i < min_len; i--) {
        if (fputc(bin[i], stream) == EOF)
            return EOF;
        len++;
        if (!info->group || (i % 4) || !i)
            continue;
        if (fputc(GROUP_SEP, stream) == EOF)
            return EOF;
        len++;
    }
    return len;
}
static char
pad_ch(const struct printf_info *info)
{
    if ((info->prec != -1) || (info->pad == ' ') || info->left)
        return ' ';
    return '0';
}
static ssize_t
pad_spaces(FILE *stream, size_t pad_len)
{
    ssize_t  len;
    len = 0;
    for (size_t i = pad_len - 1; i < pad_len; i--) {
        if (fputc(' ', stream) == EOF)
            return EOF;
        len++;
    }
    return len;
}
