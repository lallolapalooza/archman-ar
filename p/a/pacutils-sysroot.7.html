pacutils-sysroot(7)

الحزمة: pacutils

التراخيص: MIT

PACUTILS-SYSROOT(7) pacutils-sysroot PACUTILS-SYSROOT(7)

الاسم

pacutils-sysroot - إدارة نظام ضيف مثبت

الوصف

إدارة الضيوف باستخدام --root

عندما يصبح تثبيت يعتمد على libalpm مكسورًا إلى درجة لا يمكن فيها استخدام مدير الحزم نفسه، يجب إصلاحه عن طريق تحميل بيئة عمل وتثبيت التثبيت الذي يحتاج إلى إصلاح كضيف. عادةً ما يتم ذلك عن طريق تعيين جذر التثبيت للضيف باستخدام الخيار --root. تستخدم Libalpm فقط جذر التثبيت لتحديد مكان تثبيت وإزالة الملفات أثناء معاملات الحزمة. وهو مستقل تمامًا عن جميع خيارات التكوين الأخرى. هذا يجعل استخدام --root غير موثوق به لإدارة ضيف مثبت.

المشكلة الأولى هي أنه عند تغيير جذر التثبيت، لا يزال تكوين النظام المضيف قيد الاستخدام. نظرًا لأن جذر التثبيت هو خيار تكوين ("RootDir") لا يمكن استخدامه لتحميل ملف تكوين بديل. حتى إذا تمت إضافة --config لتحميل تكوين الضيف، فسيتم حل أي مسارات ("Include") لا تزال نسبية لجذر المضيف. الطريقة الوحيدة لاستخدام تكوين الضيف فعليًا دون استخدام "chroot" هي استخدام --config وإضافة مسار التركيب يدويًا إلى جميع مسارات ("Include").

الآن يتم استخدام ملفات تكوين الضيف، ولكن جميع المسارات المهيأة لا تزال تشير إلى مسارات تحت جذر المضيف، وليس الضيف. إذا تم توفير جذر تثبيت بشكل صريح، فسيقوم pacman و pacutils بتعيين قيم افتراضية لبعض، ولكن ليس كل، مسارات التكوين لتكون تحتها. هذا يعمل عادةً، لأن هذه المسارات نادرًا ما يتم تعيينها بشكل صريح، لذلك تقوم القيم الافتراضية بشكل عام بالشيء الصحيح. إذا تم تعيينها، ومع ذلك، سيتم استخدام القيم المهيأة دون تعديل. من أجل العمل بشكل موثوق على الضيف، يجب تعيين جميع مسارات التكوين بشكل نسبي إلى جذر التثبيت (باستثناء مسارات قاعدة البيانات وملف السجل التي ستكون افتراضية لمسارات داخل جذر التثبيت إذا لم يتم تعيينها).

تقديم Sysroot

هذا قدر كبير من العمل فقط لتشغيل عمليات على نظام مثبت. للسماح بالتشغيل الموثوق به على ضيف مثبت، أضاف pacman و pacutils مفهوم "sysroot". Sysroot هو ما يقصد عادة عند استخدام الخيار --root؛ سيعمل البرنامج كما لو أن sysroot كان بالفعل جذر نظام الملفات. هذا مشابه لاستخدام "chroot" للدخول إلى النظام المثبت قبل تشغيل البرنامج، ولكنه لا يزال يشغل نسخة المضيف. هناك طريقتان للقيام بذلك: chroot إلى sysroot بعد فترة وجيزة من بدء التشغيل قبل قراءة أي تكوين أو إضافة sysroot إلى بداية جميع المسارات.

قد يسبب استخدام "chroot" مباشرة مشاكل مع المكتبات التي تستخدم التحميل المتأخر لملفات التكوين أو الكائنات المشتركة. والجدير بالذكر أن glibc تؤخر تحميل بعض الكائنات حتى يتم استخدام وظائفها فعليًا. يمكن أن يتسبب هذا في عدم تطابق بين المكونات التي تم تحميلها قبل "chroot" وتلك التي تم تحميلها بعده. لتجنب هذه المشكلة، يجب تكوين libalpm بشكل عام لاستخدام المسارات تحت sysroot دون استدعاء "chroot" فعليًا.

توفر Pacutils إصدارات مدركة لـ sysroot من وظائف تحليل التكوين الخاصة بها لتبسيط العملية. لاحظ أن sysroot يتم إلحاقه بجميع المسارات أثناء تحليل التكوين. يجب على البرامج التي تستخدم روتين تحليل تكوين sysroot عدم إلحاق sysroot بمسارات التكوين المقدمة في سطر الأوامر.

أمثلة

#include <alpm.h>
#include <getopt.h>
#include <pacutils.h>
enum {
       FLAG_CONFIG = 1,
       FLAG_SYSROOT,
};
int main(int argc, char *argv[]) {
 const char *config_file = "/etc/pacman.conf";
 const char *sysroot = NULL;
       alpm_handle_t *handle = NULL;
       pu_config_t *config = NULL;
       struct option long_opts[] = {
               { "config"        , required_argument , NULL    , FLAG_CONFIG        } ,
               { "sysroot"       , required_argument , NULL    , FLAG_SYSROOT       } ,
               { 0, 0, 0, 0 },
       };
       while((c = getopt_long(argc, argv, short_opts, long_opts, NULL)) != -1) {
               switch(c) {
                       case FLAG_CONFIG: config_file = optarg; break;
                       case FLAG_SYSROOT: sysroot = optarg; break;
                       case '?': return 1; /* getopt_long already printed an error message */
               }
       }
       if((config = pu_ui_config_load_sysroot(NULL, config_file, sysroot)) == NULL)
               { return 1; } /* load_sysroot already printed an error message */
       if(!(handle = pu_initialize_handle_from_config(config))) {
               fprintf(stderr, "error: failed to initialize alpm.\n");
               return 1;
       }
       puts(alpm_option_get_root(handle));
       pu_config_free(config);
       alpm_release(handle);
       return 0;
}
