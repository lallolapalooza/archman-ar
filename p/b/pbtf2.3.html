subroutine cpbtf2 (character uplo, integer n, integer kd, complex, dimension( ldab, * ) ab, integer ldab, integer info)

CPBTF2 تحسب تحليل Cholesky لمصفوفة ذات نطاق هرمي/متماثل موجبة محددة (خوارزمية غير محظورة).

الغرض:

!>
!> CPBTF2 تحسب تحليل Cholesky لمصفوفة هرمية مركبة موجبة محددة A.
!>
!> التحليل له الشكل
!>    A = U**H * U ،  إذا كان UPLO = 'U', أو
!>    A = L  * L**H،  إذا كان UPLO = 'L',
!> حيث U مصفوفة مثلثية عليا، U**H هو المرافق المعكوس لـ U، و L مصفوفة مثلثية سفلى.
!>
!> هذا هو الإصدار غير المحظور للخوارزمية، الذي يستدعي BLAS المستوى 2.
!>

المعلمات

UPLO 
!>          UPLO هو CHARACTER*1
!>          يحدد ما إذا كان الجزء المثلثي العلوي أو السفلي للمصفوفة الهرمية A مخزناً:
!>          = 'U':  مثلثي علوي
!>          = 'L':  مثلثي سفلي
!>

N

!>          N هو INTEGER
!>          رتبة المصفوفة A.  N >= 0.
!>

KD

!>          KD هو INTEGER
!>          عدد الأقطار الفائقة للمصفوفة A إذا كان UPLO = 'U',
!>          أو عدد الأقطار الفرعية إذا كان UPLO = 'L'.  KD >= 0.
!>

AB

!>          AB هو مصفوفة COMPLEX، البعد (LDAB,N)
!>          عند الإدخال، الجزء العلوي أو السفلي من مصفوفة النطاق الهرمي
!>          A، مخزنة في أول KD+1 صفوف من المصفوفة.  العمود j من A مخزن في
!>          العمود j من المصفوفة AB كما يلي:
!>          إذا كان UPLO = 'U', AB(kd+1+i-j,j) = A(i,j) حيث max(1,j-kd)<=i<=j؛
!>          إذا كان UPLO = 'L', AB(1+i-j,j)    = A(i,j) حيث j<=i<=min(n,j+kd).
!>
!>          عند الخروج، إذا كان INFO = 0، فإن العامل المثلثي U أو L من تحليل Cholesky
!>          A = U**H *U أو A = L*L**H للمصفوفة النطاقية A، بنفس تنسيق التخزين مثل A.
!>

LDAB

!>          LDAB هو INTEGER
!>          البعد الرائد للمصفوفة AB.  LDAB >= KD+1.
!>

INFO

!>          INFO هو INTEGER
!>          = 0: خروج ناجح
!>          < 0: إذا كان INFO = -k، فإن الوسيط k كان له قيمة غير قانونية
!>          > 0: إذا كان INFO = k، فإن الأصغر الرئيسي الرائد من الرتبة k
!>               ليس موجباً، ولم يكن من الممكن إكمال التحليل.
!>

subroutine spbtf2 (character uplo, integer n, integer kd, real, dimension( ldab, * ) ab, integer ldab, integer info)

SPBTF2 يحسب تفكيك Cholesky لمصفوفة ذات نطاق متناظر/متشابك موجبة محددة (خوارزمية غير محظورة).

الغرض:

!>
!> SPBTF2 يحسب تفكيك Cholesky لمصفوفة A المتناظرة ذات النطاق والموجبة المحددة.
!>
!> التفكيك له الشكل
!>    A = U**T * U ، إذا كان UPLO = 'U'، أو
!>    A = L  * L**T، إذا كان UPLO = 'L'،
!> حيث U مصفوفة مثلثة علوية، U**T هي منقولة U، و L مصفوفة مثلثة سفلية.
!>
!> هذه هي النسخة غير المحظورة من الخوارزمية، والتي تستدعي BLAS المستوى 2.
!>

المعاملات

UPLO 
!>          UPLO هو CHARACTER*1
!>          يحدد ما إذا كان الجزء المثلثي العلوي أو السفلي للمصفوفة المتناظرة A
!>          مخزنًا:
!>          = 'U':  مثلث علوي
!>          = 'L':  مثلث سفلي
!>

N

!>          N هو INTEGER
!>          رتبة المصفوفة A. N >= 0.
!>

KD

!>          KD هو INTEGER
!>          عدد الأقطار الفائقة للمصفوفة A إذا كان UPLO = 'U'،
!>          أو عدد الأقطار الفرعية إذا كان UPLO = 'L'. KD >= 0.
!>

AB

!>          AB هو مصفوفة REAL، حجمها (LDAB,N)
!>          عند الدخول، الجزء المثلثي العلوي أو السفلي للمصفوفة ذات النطاق المتناظر
!>          A، مخزن في أول KD+1 صفوف من المصفوفة. يتم تخزين العمود j من A
!>          في العمود j من المصفوفة AB كما يلي:
!>          إذا كان UPLO = 'U'، AB(kd+1+i-j,j) = A(i,j) لـ max(1,j-kd) <= i <= j؛
!>          إذا كان UPLO = 'L'، AB(1+i-j,j) = A(i,j) لـ j <= i <= min(n,j+kd).
!>
!>          عند الخروج، إذا كان INFO = 0، فإن العامل المثلثي U أو L من تفكيك
!>          Cholesky A = U**T*U أو A = L*L**T للمصفوفة ذات النطاق A، بنفس تنسيق التخزين
!>          مثل A.
!>

LDAB

!>          LDAB هو INTEGER
!>          البعد الرائد للمصفوفة AB. LDAB >= KD+1.
!>

INFO

!>          INFO هو INTEGER
!>          = 0: خروج ناجح
!>          < 0: إذا كان INFO = -k، فإن المعامل k لديه قيمة غير قانونية
!>          > 0: إذا كان INFO = k، فإن المحدد الرئيسي الرائد من الرتبة k
!>               ليس موجبًا، ولم يكن من الممكن إكمال التفكيك.
!>
