subroutine cpbsv (character uplo, integer n, integer kd, integer nrhs, complex, dimension( ldab, * ) ab, integer ldab, complex, dimension( ldb, * ) b, integer ldb, integer info)

CPBSV تحسب حل نظام المعادلات الخطية المعقدة A * X = B

الغرض:

!>
!> CPBSV تحسب حل نظام المعادلات الخطية المعقدة
!>    A * X = B,
!> حيث A مصفوفة مجال هرميتية موجبة محددة N*N و X و B مصفوفتان N*NRHS.
!>
!> يتم استخدام تحليل Cholesky لتحليل A بالشكل
!>    A = U**H * U,  إذا كان UPLO = 'U', أو
!>    A = L * L**H,  إذا كان UPLO = 'L',
!> حيث U مصفوفة مجال مثلثة علوية، و L مصفوفة مجال مثلثة سفلية، بنفس عدد الأقطار الفوقية أو السفلية مثل A. ثم يستخدم الشكل المحلل لـ A لحل نظام المعادلات A * X = B.
!>

المعلمات

UPLO 
!>          UPLO هو CHARACTER*1
!>          = 'U':  يتم تخزين المثلث العلوي لـ A؛
!>          = 'L':  يتم تخزين المثلث السفلي لـ A.
!>

N

!>          N هو INTEGER
!>          عدد المعادلات الخطية، أي ترتيب المصفوفة A. N >= 0.
!>

KD

!>          KD هو INTEGER
!>          عدد الأقطار الفوقية للمصفوفة A إذا كان UPLO = 'U'،
!>          أو عدد الأقطار السفلية إذا كان UPLO = 'L'. KD >= 0.
!>

NRHS

!>          NRHS هو INTEGER
!>          عدد الجوانب اليمنى، أي عدد أعمدة المصفوفة B. NRHS >= 0.
!>

AB

!>          AB هو مصفوفة COMPLEX، البعد (LDAB,N)
!>          عند الإدخال، المثلث العلوي أو السفلي للمصفوفة المجال الهرميتية
!>          A، مخزنة في أول KD+1 صف من المصفوفة.
!>          يتم تخزين العمود j من A في العمود j من المصفوفة AB
!>          على النحو التالي:
!>          إذا كان UPLO = 'U', AB(KD+1+i-j,j) = A(i,j) لـ max(1,j-KD)<=i<=j؛
!>          إذا كان UPLO = 'L', AB(1+i-j,j)    = A(i,j) لـ j<=i<=min(N,j+KD).
!>          انظر أدناه لمزيد من التفاصيل.
!>
!>          عند الخروج، إذا كان INFO = 0، فإن العامل المثلث U أو L من
!>          تحليل Cholesky A = U**H*U أو A = L*L**H للمصفوفة المجال
!>          A، بنفس تنسيق التخزين مثل A.
!>

LDAB

!>          LDAB هو INTEGER
!>          البعد الرئيسي للمصفوفة AB. LDAB >= KD+1.
!>

B

!>          B هي مصفوفة COMPLEX، البعد (LDB,NRHS)
!>          عند الإدخال، مصفوفة الجانب الأيمن N*NRHS B.
!>          عند الخروج، إذا كان INFO = 0، فإن مصفوفة الحل N*NRHS X.
!>

LDB

!>          LDB هو INTEGER
!>          البعد الرئيسي للمصفوفة B. LDB >= max(1,N).
!>

INFO

!>          INFO هو INTEGER
!>          = 0:  انتهاء ناجح
!>          < 0:  إذا كان INFO = -i، فإن المعلمة i كانت قيمة غير قانونية
!>          > 0:  إذا كان INFO = i، فإن المحدد الرئيسي الرائد من الرتبة i
!>                لـ A ليس موجبًا، لذلك لم يكن بالإمكان إكمال التحليل
!>                ولم يتم حساب الحل.
!>

subroutine dbpsv (character uplo, integer n, integer kd, integer nrhs, double precision, dimension( ldab, * ) ab, integer ldab, double precision, dimension( ldb, * ) b, integer ldb, integer info)

DBPSV computes the solution to system of linear equations A * X = B for OTHER matrices

Purpose:

!>
!> DBPSV computes the solution to a double precision system of linear equations
!>    A * X = B,
!> where A is an N-by-N symmetric positive definite band matrix and X
!> and B are N-by-NRHS matrices.
!>
!> The Cholesky decomposition is used to factor A as
!>    A = U**T * U,  if UPLO = 'U', or
!>    A = L * L**T,  if UPLO = 'L',
!> where U is an upper triangular band matrix, and L is a lower
!> triangular band matrix, with the same number of superdiagonals or
!> subdiagonals as A.  The factored form of A is then used to solve the
!> system of equations A * X = B.
!>

Parameters

UPLO 
!>          UPLO is CHARACTER*1
!>          = 'U':  Upper triangle of A is stored;
!>          = 'L':  Lower triangle of A is stored.
!>

N

!>          N is INTEGER
!>          The number of linear equations, i.e., the order of the
!>          matrix A.  N >= 0.
!>

KD

!>          KD is INTEGER
!>          The number of superdiagonals of the matrix A if UPLO = 'U',
!>          or the number of subdiagonals if UPLO = 'L'.  KD >= 0.
!>

NRHS

!>          NRHS is INTEGER
!>          The number of right hand sides, i.e., the number of columns
!>          of the matrix B.  NRHS >= 0.
!>

AB

!>          AB is DOUBLE PRECISION array, dimension (LDAB,N)
!>          On entry, the upper or lower triangle of the symmetric band
!>          matrix A, stored in the first KD+1 rows of the array.  The
!>          j-th column of A is stored in the j-th column of the array AB
!>          as follows:
!>          if UPLO = 'U', AB(KD+1+i-j,j) = A(i,j) for max(1,j-KD)<=i<=j;
!>          if UPLO = 'L', AB(1+i-j,j)    = A(i,j) for j<=i<=min(N,j+KD).
!>          See below for further details.
!>
!>          On exit, if INFO = 0, the triangular factor U or L from the
!>          Cholesky factorization A = U**T*U or A = L*L**T of the band
!>          matrix A, in the same storage format as A.
!>

LDAB

!>          LDAB is INTEGER
!>          The leading dimension of the array AB.  LDAB >= KD+1.
!>

B

!>          B is DOUBLE PRECISION array, dimension (LDB,NRHS)
!>          On entry, the N-by-NRHS right hand side matrix B.
!>          On exit, if INFO = 0, the N-by-NRHS solution matrix X.
!>

LDB

!>          LDB is INTEGER
!>          The leading dimension of the array B.  LDB >= max(1,N).
!>

INFO

!>          INFO is INTEGER
!>          = 0:  successful exit
!>          < 0:  if INFO = -i, the i-th argument had an illegal value
!>          > 0:  if INFO = i, the leading principal minor of order i
!>                of A is not positive, so the factorization could not
!>                be completed, and the solution has not been computed.
!>

Author

Univ. of Tennessee

Univ. of California Berkeley

Univ. of Colorado Denver

NAG Ltd.

Further Details:

!>
!>  The band storage scheme is illustrated by the following example, when
!>  N = 6, KD = 2, and UPLO = 'U':
!>
!>  On entry:                       On exit:
!>
!>      *    *   a13  a24  a35  a46      *    *   u13  u24  u35  u46
!>      *   a12  a23  a34  a45  a56      *   u12  u23  u34  u45  u56
!>     a11  a22  a33  a44  a55  a66     u11  u22  u33  u44  u55  u66
!>
!>  Similarly, if UPLO = 'L' the format of A is as follows:
!>
!>  On entry:                       On exit:
!>
!>     a11  a22  a33  a44  a55  a66     l11  l22  l33  l44  l55  l66
!>     a21  a32  a43  a54  a65   *      l21  l32  l43  l54  l65   *
!>     a31  a42  a53  a64   *    *      l31  l42  l53  l64   *    *
!>
!>  Array elements marked * are not used by the routine.
!>

pbsv(3) - LAPACK

NAME

pbsv – a subroutine for solving a symmetric positive definite banded system of linear equations.

SYNOPSIS

      SUBROUTINE PBSV( UPLO, N, KD, NRHS, AB, LDAB, B, LDB, INFO )
      CHARACTER*1 UPLO
      INTEGER N, KD, NRHS, LDAB, LDB, INFO
      REAL    AB( LDAB, * ), B( LDB, * )

Detailed Description

PBSV is a general-purpose subroutine to solve the real symmetric positive definite banded system of linear equations

    A * X = B

where A is an N x N symmetric positive definite band matrix with K subdiagonals, X and B are N x NRHS matrixes, and LDA is the leading dimension of the array AB.

Each diagonal of the matrix A is stored in a vector. The i-th diagonal is stored in the i-th column of AB. The first column of AB stores the main diagonal (k=0), the second column stores the first super- and subdiagonals (k=1), and so on.

The matrix A is represented by the array AB.

The matrix B is represented by the array B.

A is an N x N symmetric positive definite band matrix.

The array AB stores the upper or lower triangle of the matrix A. If UPLO = 'U', the array AB stores the upper triangle. If UPLO = 'L', the array AB stores the lower triangle.

The array AB has dimensions (LDAB, N).

The array B has dimensions (LDB, NRHS).

The array INFO is an INTEGER.

On exit, INFO is set to:

    INFO = 0 : successful exit
    INFO < 0 : if INFO = -i, the i-th argument had an illegal value

  Subroutine pbsv(UPLO, N, KD, NRHS, AB, LDAB, B, LDB, INFO)
  Character*1 UPLO
  Integer N, KD, NRHS, LDAB, LDB, INFO
  Real AB(LDAB, *), B(LDB, *)

  !-----------------------------------------------------------------------
  !
  !  Purpose:
  !  =======
  !
  !  PBSV solves a symmetric positive definite banded system of linear
  !  equations
  !
  !     A * X = B
  !
  !  with a band matrix A, by using symmetric indefinite factorization in
  !  the form of a split Cholesky decomposition.
  !
  !  Arguments:
  !  ==========
  !
  !  UPLO    (input) CHARACTER*1
  !          Specifies whether the upper or lower triangular part of the
  !          band matrix A is supplied:
  !          = 'U': Upper Triangular
  !          = 'L': Lower Triangular
  !
  !  N       (input) INTEGER
  !          The order of the matrix A. N >= 0.
  !
  !  KD      (input) INTEGER
  !          The number of super-diagonals and sub-diagonals of the matrix
  !          A. KD >= 0.
  !
  !  NRHS    (input) INTEGER
  !          The number of right hand sides, i.e., the number of columns
  !          of the matrix B. NRHS >= 0.
  !
  !  AB      (input/output) REAL array, dimension (LDAB,N)
  !          On entry, the upper or lower triangle of the band matrix A,
  !          stored column by column in the manner in which a band matrix
  !          is typically represented. The width of the band (for the first
  !          KD super-diagonals and sub-diagonals) must be KD + 1.
  !          Column i contains the elements of the i-th diagonal starting
  !          from the main diagonal. Thus, AB(k,i) = A(i-k,i) for k=0 to KD.
  !          On exit, AB is overwritten by details of the factorization.
  !
  !  LDAB    (input) INTEGER
  !          The leading dimension of the array AB. LDAB >= KD + 1.
  !
  !  B       (input/output) REAL array, dimension (LDB,NRHS)
  !          On entry, the N x NRHS matrix of right hand side vectors, B.
  !          On exit, if INFO = 0, the N x NRHS matrix of solution vectors,
  !          X.
  !
  !  LDB     (input) INTEGER
  !          The leading dimension of the array B. LDB >= max(1,N).
  !
  !  INFO    (output) INTEGER
  !          = 0: successful exit
  !          < 0: if INFO = -i, then the i-th argument had an illegal
  !               value.
  !
  !  =====================================================================
  !
  !     .. Parameters ..
  !
  !     .. Local Scalars ..
  !
  !     .. Local Arrays ..
  !
  !     .. External Subroutines ..
  !
  !     .. Intrinsic Functions ..
  !
  !     .. Statement Functions ..
  !
  !     .. Executable Statements ..
  !
  !     Check the input parameters.
  !
  !
  !
  !     Quick return if N = 0.
  !
  !
  !     Band matrix storage layout:
  !     ---------------------------
  !     The following arrays illustrate the storage layout for KD = 2.
  !     The first column of AB contains the main diagonal A(I,I).
  !     The second column of AB contains the first super- and sub-diagonals.
  !     The third column of AB contains the second super- and sub-diagonals.
  !
  !     UPLO = 'U' (Upper triangular part is stored)
  !     AB(0, J) = A(J, J)   (Main diagonal)
  !     AB(1, J) = A(J-1, J) (First super-diagonal)
  !     AB(2, J) = A(J-2, J) (Second super-diagonal)
  !
  !     Example for N = 5, KD = 2:
  !
  !     Matrix A:
  !
  !     A = [ a11 a12 a13  0   0  ]
  !         [ a21 a22 a23 a24  0  ]
  !         [ a31 a32 a33 a34 a35 ]
  !         [ 0   a42 a43 a44 a45 ]
  !         [ 0    0  a53 a54 a55 ]
  !
  !     AB (stored upper triangle):
  !
  !     J=1  J=2  J=3  J=4  J=5
  !
  !     [ a11 a22 a33 a44 a55 ]  (Main diagonal, k=0)
  !     [ a12 a23 a34 a45  .  ]  (First super-diagonal, k=1)
  !     [ a13 a24  .   .   .  ]  (Second super-diagonal, k=2)
  !
  !     UPLO = 'L' (Lower triangular part is stored)
  !     AB(0, J) = A(J, J)   (Main diagonal)
  !     AB(1, J) = A(J, J-1) (First sub-diagonal)
  !     AB(2, J) = A(J, J-2) (Second sub-diagonal)
  !
  !     Example for N = 5, KD = 2:
  !
  !     Matrix A:
  !
  !     A = [ a11 a12 a13  0   0  ]
  !         [ a21 a22 a23 a24  0  ]
  !         [ a31 a32 a33 a34 a35 ]
  !         [ 0   a42 a43 a44 a45 ]
  !         [ 0    0  a53 a54 a55 ]
  !
  !     AB (stored lower triangle):
  !
  !     J=1  J=2  J=3  J=4  J=5
  !
  !     [ a11 a22 a33 a44 a55 ]  (Main diagonal, k=0)
  !     [ a21 a32 a43 a54  .  ]  (First sub-diagonal, k=1)
  !     [ a31 a42 a53  .   .  ]  (Second sub-diagonal, k=2)
  !
  !
  !
  !     This subroutine calls the panel routine ZPBTRF to factorize the
  !     matrix A as A = U' * D * U, where U is a band upper triangular
  !     matrix with unit diagonal elements and D is a band diagonal matrix.
  !
  !     Then, the solution is obtained by applying
  !     X = U'\(D\(U\B))
  !
  !     The arrays are indexed as follows:
  !
  !     l11  l22  l33  l44  l55
  !     l21  l32  l43  l54  l65   *
  !     l31  l42  l53  l64   *    *
  !     l41  l52   *    *    *    *
  !     l51   *    *    *    *    *
  !
  !     l11  l22  l33  l44  l55
  !     *    l32  l43  l54  l65
  !     *    *    l53  l64  l75
  !     *    *    *    l74  l85
  !     *    *    *    *    l95
  !
  !     u11  u22  u33  u44  u55
  !     u12  u23  u34  u45  u65   *
  !     u13  u24   *    *    *    *
  !
  !     u11  u22  u33  u44  u55
  !     *    u23  u34  u45  u55
  !     *    *    u34  u45  u55
  !
  !     l11  l22  l33  l44  l55
  !     l21  l32  l43  l54  l65   *
  !     l31  l42  l53  l64   *    *
  !
  !     l11  l22  l33  l44  l55
  !     *    l32  l43  l54  l65
  !     *    *    l53  l64  l75
  !
  !
  !     a21  a32  a43  a54  a65   *      l21  l32  l43  l54  l65   *
  !     a31  a42  a53  a64   *    *      l31  l42  l53  l64   *    *
  !
  !  Array elements marked * are not used by the routine.
