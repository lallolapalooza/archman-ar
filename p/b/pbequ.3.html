subroutine cpbequ (character uplo, integer n, integer kd, complex, dimension( ldab, * ) ab, integer ldab, real, dimension( * ) s, real scond, real amax, integer info)

CPBEQU

الغرض:

!>
!> CPBEQU تحسب معاملات القياس للصف والأعمدة المخصصة لموازنة مصفوفة
!> هرميتية موجبة محددة A وتقليل رقم حالتها (بالنسبة للمعيار الثنائي).
!> S تحتوي على عوامل القياس، S(i) = 1/sqrt(A(i,i))، المختارة بحيث تكون
!> المصفوفة المقاسة B بعناصر B(i,j) = S(i)*A(i,j)*S(j) لها قيم واحد على
!> القطر. هذا الاختيار لـ S يضع رقم حالة B ضمن عامل N من أصغر رقم حالة
!> ممكن عبر جميع قياسات الأقطار الممكنة.
!>

المعلمات

UPLO 
!>          UPLO هو CHARACTER*1
!>          = 'U':  المثلث العلوي لـ A مخزن؛
!>          = 'L':  المثلث السفلي لـ A مخزن.
!>

N

!>          N هو INTEGER
!>          ترتيب المصفوفة A. N >= 0.
!>

KD

!>          KD هو INTEGER
!>          عدد الأقطار الفائقة للمصفوفة A إذا كان UPLO = 'U'،
!>          أو عدد الأقطار الفرعية إذا كان UPLO = 'L'. KD >= 0.
!>

AB

!>          AB هو مصفوفة COMPLEX، حجمها (LDAB,N)
!>          المثلث العلوي أو السفلي للمصفوفة الهرميتية ذات النطاق،
!>          مخزنة في أول KD+1 صفوف من المصفوفة. يتم تخزين العمود j
!>          للمصفوفة A في العمود j من المصفوفة AB كالتالي:
!>          إذا كان UPLO = 'U'، AB(kd+1+i-j,j) = A(i,j) لـ max(1,j-kd)<=i<=j؛
!>          إذا كان UPLO = 'L'، AB(1+i-j,j)    = A(i,j) لـ j<=i<=min(n,j+kd).
!>

LDAB

!>          LDAB هو INTEGER
!>          البعد الرائد للمصفوفة A. LDAB >= KD+1.
!>

S

!>          S هي مصفوفة REAL، حجمها (N)
!>          إذا كان INFO = 0، تحتوي S على عوامل القياس لـ A.
!>

SCOND

!>          SCOND هو REAL
!>          إذا كان INFO = 0، تحتوي S على نسبة أصغر S(i) إلى
!>          أكبر S(i). إذا كان SCOND >= 0.1 وكان AMAX ليس كبيرًا جدًا
!>          ولا صغيرًا جدًا، فلا يستحق القياس بـ S.
!>

AMAX

!>          AMAX هو REAL
!>          القيمة المطلقة لأكبر عنصر في المصفوفة. إذا كان AMAX قريبًا جدًا
!>          من تجاوز الحد الأعلى أو قريبًا جدًا من تجاوز الحد الأدنى،
!>          يجب قياس المصفوفة.
!>

INFO

!>          INFO هو INTEGER
!>          = 0:  انتهاء ناجح
!>          < 0:  إذا كان INFO = -i، كان المعامل i ذو قيمة غير قانونية.
!>          > 0:  إذا كان INFO = i، كان العنصر القطري i غير موجب.
!>
