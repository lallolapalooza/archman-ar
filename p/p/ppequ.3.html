subroutine cppequ (character uplo, integer n, complex, dimension( * ) ap, real, dimension( * ) s, real scond, real amax, integer info)

CPPEQU

الغرض:

!>
!> CPPEQU تحسب مقاييس الصفوف والأعمدة المخصصة لموازنة مصفوفة هرميتية موجبة محددة A المخزنة في شكل مضغوط وتقليل
!> عدد حالتها (بالنسبة للقيمة الثنائية). S تحتوي على
!> عوامل القياس، S(i)=1/sqrt(A(i,i))، المختارة بحيث تكون المصفوفة المقاسة
!> B بعناصر B(i,j)=S(i)*A(i,j)*S(j) لها واحدات على القطر.
!> هذا الاختيار لـ S يضع عدد حالة B في حدود عامل N من
!> أصغر عدد حالة ممكن عبر جميع القياسات القطرية الممكنة.
!>

المعلمات

UPLO 
!>          UPLO هو CHARACTER*1
!>          = 'U':  المثلث العلوي لـ A مخزن؛
!>          = 'L':  المثلث السفلي لـ A مخزن.
!>

N

!>          N هو INTEGER
!>          ترتيب المصفوفة A. N >= 0.
!>

AP

!>          AP هو مصفوفة COMPLEX، البعد (N*(N+1)/2)
!>          المثلث العلوي أو السفلي للمصفوفة الهرميتية A، مضغوط
!>          عمودياً في مصفوفة خطية. يتم تخزين العمود j من A
!>          في المصفوفة AP كما يلي:
!>          إذا كان UPLO = 'U'، AP(i + (j-1)*j/2) = A(i,j) لـ 1<=i<=j؛
!>          إذا كان UPLO = 'L'، AP(i + (j-1)*(2n-j)/2) = A(i,j) لـ j<=i<=n.
!>

S

!>          S هو مصفوفة REAL، البعد (N)
!>          إذا كان INFO = 0، تحتوي S على عوامل القياس لـ A.
!>

SCOND

!>          SCOND هو REAL
!>          إذا كان INFO = 0، تحتوي S على نسبة أصغر S(i) إلى
!>          أكبر S(i). إذا كان SCOND >= 0.1 وكان AMAX ليس كبيراً جداً
!>          أو صغيراً جداً، فلا يستحق القياس بـ S.
!>

AMAX

!>          AMAX هو REAL
!>          القيمة المطلقة لأكبر عنصر في المصفوفة. إذا كان AMAX قريباً جداً
!>          من تجاوز الحد الأعلى أو قريباً جداً من تحت الحد الأدنى، يجب قياس المصفوفة.
!>

INFO

!>          INFO هو INTEGER
!>          = 0:  خروج ناجح
!>          < 0:  إذا كان INFO = -i، فإن المعلمة i كان لها قيمة غير قانونية
!>          > 0:  إذا كان INFO = i، فإن العنصر القطري i غير موجب.
!>
