pipewire-props(7)

الحزمة: pipewire

التراخيص: MIT, LGPL-2.1-or-later

يقوم معظم تحويلات العينات وإعادة التشكيل في العميل (أو في حالة خادم PulseAudio، في خادم pipewire-pulse الذي ينشئ التدفقات). هذا يضمن تفريغ جميع التحويلات إلى العملاء ويمكن للخادم التعامل مع تنسيق واحد فقط لأسباب تتعلق بالأداء.

فيما يلي شرح للخيارات التي يمكن ضبطها في محول العينات.

resample.quality = 4

جودة إعادة التشكيل. من 0 إلى 14، الافتراضي هو 4.

زيادة الجودة ستؤدي إلى قطع أفضل وتشوه أقل على حساب استهلاك (أكثر بكثير) لوحدة المعالجة المركزية. تم اختيار جودة افتراضية تبلغ 4 كتسوية جيدة بين الجودة والأداء دون تشوهات تقل بكثير عن النطاق المسموع.

راجع Infinite Wave لمقارنة الأداء.

resample.disable = false

تعطيل إعادة التشكيل بالكامل. سيكون العقدة قادرة فقط على التفاوض مع الرسم البياني عندما تكون معدلات العينات متوافقة.

معلمات خلاط القنوات

يمكن للمصادر والأحواض وتدفقات الالتقاط والإخراج تطبيق خلط القنوات على الإشارة الواردة.

عادةً ما لا يتم استخدام خلاط القنوات للأجهزة، وعادةً ما يتم عرض قنوات الجهاز كما هي. يتم فرض هذه السياسة عادةً بواسطة مدير الجلسة، لذا نشير إلى وثائقه هناك.

عادةً ما يتم تكوين تدفقات الإخراج والالتقاط لتخطيط قنوات الحوض/المصدر الذي تتصل به، وبالتالي ستقوم بخلط القنوات.

ينفذ خلاط القنوات أيضًا حجمًا برمجيًا. يتم إجراء تعديل الحجم هذا على تخطيط القناة الأصلي. مثال: تدفق إخراج ستيريو يتم رفعه إلى 5.1 لديه تحكم في مستوى الصوت الأيسر والأيمن.

channelmix.disable = false

يعطل خلاط القنوات تمامًا. سيكون التدفق قادرًا فقط على الارتباط بالمصادر/الأحواض المتوافقة بنفس تخطيط القناة بالضبط.

channelmix.min-volume = 0.0

channelmix.max-volume = 10.0

يعطي قيم الحد الأدنى والأقصى للحجم المسموح به. سيتم قص أي حجم يتم تعيينه إلى هذه القيم.

channelmix.normalize = false

يضمن أنه أثناء هذا الخلط وإعادة التشكيل، يتم الحفاظ على مستوى 0 ديسيبل الأصلي، بحيث لا يبدو أي شيء أهدأ/أعلى بشكل كبير.

بينما يمنع هذا الخيار التشويش، فإنه يمكن أن ينتج عنه حجم منخفض جدًا في بعض الحالات. قم بزيادة مستوى الصوت في تلك الحالة أو قم بتعطيل التطبيع.

channelmix.lock-volumes = false

يعطل تمامًا تغييرات مستوى الصوت أو كتم الصوت. الافتراضي هو false.

channelmix.mix-lfe = true

يخلط قناة التأثيرات منخفضة التردد في القناة المركزية الأمامية أو زوج الاستيريو. قد يعزز هذا النطاق الديناميكي للإشارة إذا لم يكن هناك مضخم صوت ويمكن للسماعات إعادة إنتاج إشارة التردد المنخفض.

channelmix.upmix = true

يُمكّن رفع مستوى الصوت للقناة المركزية الأمامية (FC) عندما تحتوي الوجهة على قناة FC. يتم استخدام مجموع قنوات الستيريو ويمكن استخدام مرشح تمرير منخفض اختياري (انظر channelmix.fc-cutoff).

كما يتم تمكين رفع مستوى الصوت لـ LFE عندما يتم تعيين channelmix.lfe-cutoff إلى شيء آخر غير 0 وتحتوي الوجهة على قناة LFE. يتم إنتاج قناة LFE عن طريق إضافة قنوات الستيريو.

إذا كان channelmix.upmix صحيحًا، فسيتم أيضًا تمكين رفع مستوى الصوت للقنوات الخلفية والتحكم فيه باستخدام خاصية channelmix-upmix-method.

channelmix.upmix-method = psd

يتم توفير 3 طرق لإنتاج القنوات الخلفية في الصوت المحيطي:
1.
none. لا يتم إنتاج قنوات خلفية.
2.
simple. يتم نسخ القنوات الأمامية إلى الخلفية. هذا سريع ولكنه قد ينتج عنه تأثيرات طور.
3.
psd. يتم إنتاج القنوات الخلفية من الصوت المحيطي الأمامي الأيسر والأيمن (الفرق بين القنوات). تتم إضافة تأخير ومرشح طور اختياري للإشارة الخلفية لجعل الصوت أكبر.

channelmix.lfe-cutoff = 150

تطبيق مرشح تمرير منخفض على التأثيرات منخفضة التردد. يتم التعبير عن القيمة بالهرتز. تحتوي مضخمات الصوت النموذجية على قطع عند حوالي 150 و 200. القيمة الافتراضية 0 تعطّل الميزة.

channelmix.fc-cutoff = 12000

تطبيق مرشح تمرير منخفض على تردد المركز الأمامي. يتم التعبير عن القيمة بالهرتز.

نظرًا لأن المركز الأمامي يحتوي على الحوارات، فإن تردد القطع النموذجي هو 12000 هرتز.

هذا الخيار نشط فقط عند تمكين الرفع.

channelmix.rear-delay = 12.0

تطبيق تأخير بالمللي ثانية عند رفع القنوات الخلفية. هذا يحسن تخصيص الصوت. تأخير نموذجي يبلغ 12 مللي ثانية هو الافتراضي.

هذا نشط فقط عند استخدام طريقة الرفع psd.

channelmix.stereo-widen = 0.0

يطرح بعض إشارة المركز الأمامي من قنوات الاستيريو. هذا ينقل الحوارات أكثر إلى مكبر الصوت المركزي ويترك الصوت المحيطي في قنوات الاستيريو.

هذا نشط فقط عند تمكين الرفع وخلط قناة مركزية أمامية.

channelmix.hilbert-taps = 0

سيطبق هذا الخيار تحول طور بزاوية 90 درجة على القنوات الخلفية لتحسين التخصيص. تحتاج الصنبور إلى أن تكون بين 15 و 255 مع نتائج أكثر دقة (واستهلاك أكبر لوحدة المعالجة المركزية) للقيم الأعلى.

هذا نشط فقط عند استخدام طريقة الرفع psd.

dither.noise = 0

سيضيف هذا الخيار N بت من البيانات العشوائية إلى الإشارة. عندما لا يتم تحديد dither.method، ستتغير البيانات العشوائية بين [-(1<< (N-1))، 0] كل 1024 عينة. مع dither.method، يتم تضخيم ضوضاء التدرج بمقدار 1<<(N-1) بت.

يمكن استخدام هذا لإبقاء بعض المضخمات قيد التشغيل خلال فترات الصمت. عادة ما يكون بت أو بتان من الضوضاء كافيين، وإلا ستصبح الضوضاء مسموعة. يستخدم هذا عادةً مع session.suspend-timeout-seconds لتعطيل الإيقاف المؤقت في مدير الجلسة.

لاحظ أن PipeWire يستخدم عمليات النقطة العائمة بدقة 24 بت لجميع معالجة الصوت. التحويل إلى تنسيقات عينات عدد صحيح 24 بت يكون عديم الفقد، والتحويل إلى تنسيقات عينات عدد صحيح 32 بت يكون مجرد حشو بأصفار في النهاية. هذا يعني أن ضوضاء التدرج تكون دائمًا في الـ 24 بت الأكثر أهمية.

dither.method = none

يمكن إجراء تدرج اختياري على الإشارة المكمّاة.

هناك 6 أوضاع متاحة:

1.
none لا يتم إجراء أي تدرج.
2.
rectangular التدرج مع توزيع ضوضاء مستطيل. هذا يضيف بتات عشوائية في النطاق [-0.5، 0.5] إلى الإشارة بتوزيع متساوٍ.
3.
triangular التدرج مع توزيع ضوضاء مثلثي. هذا يضيف بتات عشوائية في النطاق [-1.0، 1.0] إلى الإشارة بتوزيع مثلثي حول 0.0.
4.
triangular-hf التدرج مع توزيع ضوضاء مثلثي مائل.
5.
wannamaker3 يتم إجراء تشكيل ضوضاء إضافي على التدرج المثلثي المائل لنقل الضوضاء إلى نطاق أقل سماعًا. هذا يستخدم مرشح ضوضاء "F-Weighted" الموصوف بواسطة Wannamaker.
6.
shaped5 يتم إجراء تشكيل ضوضاء إضافي على التدرج المثلثي لنقل الضوضاء إلى نطاق أقل سماعًا. هذا يستخدم مرشح Lipshitz.

التدرج مفيد فقط للتحويل إلى تنسيق بأقل من 24 بت وسيتم تعطيله بخلاف ذلك.

معلمات التصحيح

debug.wav-path = ""

اجعل التدفق يكتب أيضًا العينات الخام إلى ملف WAV لأغراض التصحيح.

معلمات أخرى

هذه تتحكم في ميزات تقنية منخفضة المستوى:

clock.quantum-limit

راجع pipewire.conf(5)

resample.peaks = false # boolean

بدلاً من إعادة التشكيل فعليًا، قم بإنتاج قيم سعة الذروة كمخرجات. يتم استخدام هذا لمراقبة مستوى الصوت، حيث يتم تعيينه كخاصية لتدفق "التسجيل".

resample.prefill = false # boolean

املأ مخازن إعادة التشكيل مسبقًا بالصمت. هذا يؤثر على العينات الأولية التي تنتجها أداة إعادة التشكيل.

adapter.auto-port-config = null # JSON

إذا تم تحديده، قم بتكوين منافذ العقدة عند إنشائها، بدلاً من ترك ذلك لمدير الجلسة للقيام به. هذا مفيد (فقط) للتكوينات الدنيا بدون مدير جلسة.

القيمة هي SPA JSON بالشكل:

{
    mode = "none",          # "none", "passthrough", "convert", "dsp"
    monitor = false,        # boolean
    control = false,        # boolean
    position = "preserve"   # "unknown", "aux", "preserve"
}


راجع spa_param_port_config للمعنى.

خصائص ALSA

خصائص المراقبة

alsa.use-acp # boolean

استخدم ملفات تعريف بطاقة ALSA (ACP) لتكوين الجهاز.

alsa.udev.expose-busy # boolean

اكشف عن بطاقة ALSA حتى لو كانت مشغولة / قيد الاستخدام. الافتراضي هو false. يمكن أن يكون هذا مفيدًا عندما تكون بعض وحدات PCM قيد الاستخدام بواسطة تطبيقات أخرى ولكن يجب أن تظل وحدات PCM الأخرى مكشوفة.

خصائص الجهاز

api.alsa.path # string

مسار جهاز ALSA كما يمكن استخدامه في snd_pcm_open() و snd_ctl_open().

api.alsa.use-ucm = true # boolean

عند تمكين ACP وتوفر تكوين UCM للجهاز، فإنه يتم استخدامه افتراضيًا بدلاً من ملفات تعريف ACP. يسمح لك هذا الخيار بتعطيل ذلك واستخدام ملفات تعريف ACP بدلاً من ذلك.

هذا الخيار لا يفعل شيئًا إذا تم تعيين api.alsa.use-acp على false.

api.alsa.soft-mixer = false # boolean

سيؤدي تعيين هذا الخيار إلى true إلى تعطيل خلاط الأجهزة للتحكم في مستوى الصوت وكتم الصوت. سيتم بعد ذلك استخدام جميع معالجة مستوى الصوت باستخدام حجم البرامج وكتم الصوت، مع ترك خلاط الأجهزة دون تغيير. يمكن أن يكون هذا مثيرًا للاهتمام لتجاوز الأخطاء في اكتشاف الخلاط أو الإبلاغ عن الديسيبل. سيظل خلاط الأجهزة مستخدمًا لكتم الصوت غير المستخدم

هذا الموقع غير مرتبط بـ Arch Linux ولا تابع له بأي شكل من الأشكال. المحتوى هو ترجمة فقط لصفحات دليل Arch المنشورة على man.archlinux.org.
