pidfd_send_signal(2)

NAME pidfd_send_signal — إرسال إشارة إلى معرّف عملية
LIBRARY مكتبة C القياسية (libc)، libpthread
SYNOPSIS 
#include <sys/types.h>
#include <signal.h>
#include <unistd.h>

int pidfd_send_signal(int pidfd, int sig, siginfo_t *info, unsigned int flags);
DESCRIPTION وظيفة pidfd_send_signal() ترسل الإشارة sig إلى العملية التي يمثلها معرّف العملية pidfd. pidfd هو معرّف ملف تم الحصول عليه بواسطة استدعاء pidfd_open(2).

يعتمد السلوك عند إرسال إشارة إلى عملية معينة على المعلمات التالية:

pidfd:
معرّف ملف يمثل العملية المستهدفة. يجب أن يكون هذا المعرّف تم الحصول عليه بواسطة pidfd_open(2).
sig:
رقم الإشارة التي سيتم إرسالها.
info:
معلومات إضافية عن الإشارة. إذا كانت NULL، فسيتم إرسال الإشارة بدون معلومات إضافية.
flags:
رايات لتعديل سلوك الإرسال. حاليًا، لا توجد رايات مدعومة. يجب أن تكون 0.

يمكن استخدام pidfd_send_signal() لإرسال أي إشارة صالحة، بما في ذلك SIGKILL و SIGSTOP.

على عكس kill(2)، لا يسمح pidfd_send_signal() بإرسال إشارات إلى مجموعات عمليات أو معرّفات عمليات سلبية.
RETURN_VALUE عند النجاح، يتم إرجاع 0. عند الفشل، يتم إرجاع -1، ويتم تعيين errno للإشارة إلى الخطأ.
ERRORS
EBADF:
pidfd ليس معرّف ملف صالحًا.
EINVAL:
sig غير صالح، أو info غير صالح، أو flags غير صفر.
ESRCH:
العملية التي يمثلها pidfd لم تعد موجودة.
EPERM:
لا يملك المتصل الإذن بإرسال الإشارة إلى العملية المستهدفة.
STANDARDS تمت إضافة pidfd_send_signal() في Linux kernel 5.1.
HISTORY

تم تقديم pidfd_send_signal() لأول مرة في Linux kernel 5.1.
NOTES

تعتبر pidfd_send_signal() طريقة أكثر أمانًا لإرسال الإشارات مقارنة بـ kill(2)، حيث أنها تستخدم معرّف ملف حصري للعملية بدلاً من معرّف العملية (PID) الذي قد يتم إعادة استخدامه.

عند إرسال إشارة باستخدام pidfd_send_signal()، يمكن أن يتضمن الحقل si_pid في بنية siginfo_t معرّف العملية للمتصل، ويشير si_signo إلى الإشارة المرسلة، و si_code إلى سبب الإشارة (عادةً ما يكون SI_PIDFD).

EXAMPLES

يقوم المثال التالي بفتح معرّف عملية لطفل، ثم يرسل إليه إشارة SIGTERM. 

#include <stdio.h>
#include <stdlib.h>
#include <unistd.h>
#include <signal.h>
#include <sys/types.h>
#include <sys/wait.h>

int main() {
    pid_t pid;
    int pidfd;

    pid = fork();
    if (pid == -1) {
        perror("fork");
        exit(EXIT_FAILURE);
    }

    if (pid == 0) { /* Child process */
        /* Keep the child process alive until it receives a signal */
        while (1) {
            pause();
        }
        exit(EXIT_SUCCESS);
    } else { /* Parent process */
        pidfd = pidfd_open(pid, 0);
        if (pidfd == -1) {
            perror("pidfd_open");
            exit(EXIT_FAILURE);
        }

        printf("Sending SIGTERM to PID %d...\n", pid);
        if (pidfd_send_signal(pidfd, SIGTERM, NULL, 0) == -1) {
            perror("pidfd_send_signal");
            close(pidfd);
            exit(EXIT_FAILURE);
        }

        wait(NULL); /* Wait for the child to terminate */
        close(pidfd);
        printf("Child process terminated.\n");
    }

    return 0;
}
