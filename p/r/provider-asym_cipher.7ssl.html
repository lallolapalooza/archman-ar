الاسم

provider-asym_cipher - دالات مكتبة asym_cipher <-> المزود

الملخص

#include <openssl/core_dispatch.h>
#include <openssl/core_names.h>
/*
 * لا توجد أي من هذه الدوال فعليًا، ولكن يتم عرضها بهذه الطريقة
 * لتوقيعات الدوال التي يتم توفيرها كإشارات دالة في مصفوفات OSSL_DISPATCH.
 */
/* إدارة السياق */
void *OSSL_FUNC_asym_cipher_newctx(void *provctx);
void OSSL_FUNC_asym_cipher_freectx(void *ctx);
void *OSSL_FUNC_asym_cipher_dupctx(void *ctx);
/* التشفير */
int OSSL_FUNC_asym_cipher_encrypt_init(void *ctx, void *provkey,
                                       const OSSL_PARAM params[]);
int OSSL_FUNC_asym_cipher_encrypt(void *ctx, unsigned char *out, size_t *outlen,
                                  size_t outsize, const unsigned char *in,
                                  size_t inlen);
/* فك التشفير */
int OSSL_FUNC_asym_cipher_decrypt_init(void *ctx, void *provkey,
                                       const OSSL_PARAM params[]);
int OSSL_FUNC_asym_cipher_decrypt(void *ctx, unsigned char *out, size_t *outlen,
                                  size_t outsize, const unsigned char *in,
                                  size_t inlen);
/* معلمات التشفير غير المتماثل */
int OSSL_FUNC_asym_cipher_get_ctx_params(void *ctx, OSSL_PARAM params[]);
const OSSL_PARAM *OSSL_FUNC_asym_cipher_gettable_ctx_params(void *provctx);
int OSSL_FUNC_asym_cipher_set_ctx_params(void *ctx, const OSSL_PARAM params[]);
const OSSL_PARAM *OSSL_FUNC_asym_cipher_settable_ctx_params(void *provctx);

الوصف

هذه الوثائق موجهة بشكل أساسي لمؤلفي المزودين. انظر provider(7) لمزيد من المعلومات.

تسمح عملية التشفير غير المتماثل (OSSL_OP_ASYM_CIPHER) للمزودين بتنفيذ خوارزميات التشفير غير المتماثل وجعلها متاحة للتطبيقات عبر وظائف API EVP_PKEY_encrypt(3)، EVP_PKEY_decrypt(3) والوظائف الأخرى ذات الصلة).

جميع "الدوال" المذكورة هنا يتم تمريرها كإشارات دالة بين libcrypto والمزود في مصفوفات OSSL_DISPATCH(3) عبر مصفوفات OSSL_ALGORITHM(3) التي يتم إرجاعها بواسطة دالة provider_query_operation() الخاصة بالمزود (انظر "دوال المزود" في provider-base(7)).

جميع هذه "الدوال" لها تعريف نوع دالة مطابق يسمى OSSL_FUNC_{name}_fn، ودالة مساعدة لاسترداد مؤشر الدالة من عنصر OSSL_DISPATCH(3) يسمى OSSL_FUNC_{name}. على سبيل المثال، "الدالة" OSSL_FUNC_asym_cipher_newctx() لها هذه:

typedef void *(OSSL_FUNC_asym_cipher_newctx_fn)(void *provctx);
static ossl_inline OSSL_FUNC_asym_cipher_newctx_fn
    OSSL_FUNC_asym_cipher_newctx(const OSSL_DISPATCH *opf);

يتم فهرسة مصفوفات OSSL_DISPATCH(3) بواسطة أرقام يتم توفيرها كـ macros في openssl-core_dispatch.h(7)، كما يلي:

OSSL_FUNC_asym_cipher_newctx               OSSL_FUNC_ASYM_CIPHER_NEWCTX
OSSL_FUNC_asym_cipher_freectx              OSSL_FUNC_ASYM_CIPHER_FREECTX
OSSL_FUNC_asym_cipher_dupctx               OSSL_FUNC_ASYM_CIPHER_DUPCTX
OSSL_FUNC_asym_cipher_encrypt_init         OSSL_FUNC_ASYM_CIPHER_ENCRYPT_INIT
OSSL_FUNC_asym_cipher_encrypt              OSSL_FUNC_ASYM_CIPHER_ENCRYPT
OSSL_FUNC_asym_cipher_decrypt_init         OSSL_FUNC_ASYM_CIPHER_DECRYPT_INIT
OSSL_FUNC_asym_cipher_decrypt              OSSL_FUNC_ASYM_CIPHER_DECRYPT
OSSL_FUNC_asym_cipher_get_ctx_params       OSSL_FUNC_ASYM_CIPHER_GET_CTX_PARAMS
OSSL_FUNC_asym_cipher_gettable_ctx_params  OSSL_FUNC_ASYM_CIPHER_GETTABLE_CTX_PARAMS
OSSL_FUNC_asym_cipher_set_ctx_params       OSSL_FUNC_ASYM_CIPHER_SET_CTX_PARAMS
OSSL_FUNC_asym_cipher_settable_ctx_params  OSSL_FUNC_ASYM_CIPHER_SETTABLE_CTX_PARAMS

قد لا ينفذ تنفيذ خوارزمية تشفير غير متماثل جميع هذه الدوال. لكي يكون مجموعة متسقة من الدوال، يجب على المزود تنفيذ OSSL_FUNC_asym_cipher_newctx و OSSL_FUNC_asym_cipher_freectx. ويجب عليه أيضًا تنفيذ كل من OSSL_FUNC_asym_cipher_encrypt_init و OSSL_FUNC_asym_cipher_encrypt، أو كل من OSSL_FUNC_asym_cipher_decrypt_init و OSSL_FUNC_asym_cipher_decrypt. OSSL_FUNC_asym_cipher_get_ctx_params اختياري ولكن إذا كان موجودًا فيجب أن يكون OSSL_FUNC_asym_cipher_gettable_ctx_params موجودًا أيضًا. وبالمثل، OSSL_FUNC_asym_cipher_set_ctx_params اختياري ولكن إذا كان موجودًا فيجب أن يكون OSSL_FUNC_asym_cipher_settable_ctx_params موجودًا أيضًا.

يجب على خوارزمية التشفير غير المتماثل أيضًا تنفيذ آلية لإنشاء أو تحميل أو استيراد المفاتيح عبر عملية إدارة المفاتيح (OSSL_OP_KEYMGMT). انظر provider-keymgmt(7) لمزيد من التفاصيل.

دوال إدارة السياق

يجب على OSSL_FUNC_asym_cipher_newctx() إنشاء وإرجاع مؤشر إلى بنية جانب المزود للاحتفاظ بمعلومات السياق أثناء عملية التشفير غير المتماثل. سيتم إعادة تمرير مؤشر إلى هذا السياق في عدد من استدعاءات دالة عملية التشفير غير المتماثل الأخرى. المعلمة provctx هي سياق المزود الذي تم إنشاؤه أثناء تهيئة المزود (انظر provider(7)).

يتم تمرير OSSL_FUNC_asym_cipher_freectx() مؤشرًا إلى سياق التشفير غير المتماثل الخاص بالمزود في المعلمة ctx. يجب على هذه الدالة تحرير أي موارد مرتبطة بهذا السياق.

يجب على OSSL_FUNC_asym_cipher_dupctx() تكرار سياق التشفير غير المتماثل الخاص بالمزود في المعلمة ctx وإرجاع نسخة مكررة.

دوال التشفير

تقوم OSSL_FUNC_asym_cipher_encrypt_init() بتهيئة سياق للتشفير غير المتماثل بالنظر إلى سياق تشفير غير متماثل خاص بالمزود في المعلمة ctx، ومؤشر إلى كائن مفتاح خاص بالمزود في المعلمة provkey. يجب تعيين المعلمات params، إذا لم تكن NULL، على السياق بطريقة مشابهة لاستخدام OSSL_FUNC_asym_cipher_set_ctx_params(). يجب أن يكون كائن المفتاح قد تم إنشاؤه أو تحميله أو استيراده مسبقًا إلى المزود باستخدام عملية إدارة المفاتيح (OSSL_OP_KEYMGMT) (انظر provider-keymgmt(7)). تقوم OSSL_FUNC_asym_cipher_encrypt() بإجراء التشفير الفعلي نفسه. يتم تمرير سياق تشفير غير متماثل تم تهيئته مسبقًا في المعلمة ctx. تشير البيانات المراد تشفيرها بواسطة المعلمة in التي يبلغ طولها inlen بايت. ما لم تكن out NULL، يجب كتابة البيانات المشفرة إلى الموقع الذي يشير إليه المعلمة out ويجب ألا تتجاوز outsize بايت. يجب كتابة طول البيانات المشفرة إلى *outlen. إذا كانت out NULL، فيجب كتابة الحد الأقصى لطول البيانات المشفرة إلى *outlen.

دوال فك التشفير

تقوم OSSL_FUNC_asym_cipher_decrypt_init() بتهيئة سياق لفك التشفير غير المتماثل بالنظر إلى سياق تشفير غير متماثل خاص بالمزود في المعلمة ctx، ومؤشر إلى كائن مفتاح خاص بالمزود في المعلمة provkey. يجب تعيين المعلمات params، إذا لم تكن NULL، على السياق بطريقة مشابهة لاستخدام OSSL_FUNC_asym_cipher_set_ctx_params(). يجب أن يكون كائن المفتاح قد تم إنشاؤه أو تحميله أو استيراده مسبقًا إلى المزود باستخدام عملية إدارة المفاتيح (OSSL_OP_KEYMGMT) (انظر provider-keymgmt(7)).

تقوم OSSL_FUNC_asym_cipher_decrypt() بإجراء فك التشفير الفعلي نفسه. يتم تمرير سياق تشفير غير متماثل تم تهيئته مسبقًا في المعلمة ctx. تشير البيانات المراد فك تشفيرها بواسطة المعلمة in التي يبلغ طولها inlen بايت. ما لم تكن out NULL، يجب كتابة البيانات المفكوكة إلى الموقع الذي يشير إليه المعلمة out ويجب ألا تتجاوز outsize بايت. يجب كتابة طول البيانات المفكوكة إلى *outlen. إذا كانت out NULL، فيجب كتابة الحد الأقصى لطول البيانات المفكوكة إلى *outlen.

معلمات التشفير غير المتماثل

انظر OSSL_PARAM(3) لمزيد من التفاصيل حول بنية المعلمات المستخدمة بواسطة OSSL_FUNC_asym_cipher_get_ctx_params()

NAME

provider-asym_cipher - Provider asymmetric cipher methods

SYNOPSIS

 #include <openssl/core_dispatch.h>

 /*
  * The following constants are used to identify specific provider
  * asymmetric cipher parameters.
  */
 #define OSSL_ASYM_CIPHER_PARAM_PAD_MODE "pad-mode"
 #define OSSL_ASYM_CIPHER_PARAM_OAEP_DIGEST "digest"
 #define OSSL_ASYM_CIPHER_PARAM_DIGEST "digest"
 #define OSSL_ASYM_CIPHER_PARAM_OAEP_DIGEST_PROPS "digest-props"
 #define OSSL_ASYM_CIPHER_PARAM_DIGEST_PROPS "digest-props"
 #define OSSL_ASYM_CIPHER_PARAM_MGF1_DIGEST "mgf1-digest"
 #define OSSL_ASYM_CIPHER_PARAM_MGF1_DIGEST_PROPS "mgf1-digest-props"
 #define OSSL_ASYM_CIPHER_PARAM_OAEP_LABEL "oaep-label"
 #define OSSL_ASYM_CIPHER_PARAM_TLS_CLIENT_VERSION "tls-client-version"
 #define OSSL_ASYM_CIPHER_PARAM_TLS_NEGOTIATED_VERSION "tls-negotiated-version"
 #define OSSL_ASYM_CIPHER_PARAM_IMPLICIT_REJECTION "implicit-rejection"
 #define OSSL_ASYM_CIPHER_PARAM_FIPS_APPROVED_INDICATOR "fips-indicator"
 #define OSSL_ASYM_CIPHER_PARAM_FIPS_KEY_CHECK "key-check"

 typedef OSSL_OP_ qīng_newctx(const OSSL_CORE_ENV *env, OSSL_DISPATCH *provider_dispatch);
 typedef void OSSL_OP_ qīng_freectx(void *ctx);

 typedef void OSSL_OP_ qīng_flushctx(void *ctx);
 typedef void *OSSL_OP_ qīng_dupctx(void *ctx);

 typedef int OSSL_OP_ qīng_set_ctx_params(void *ctx, const OSSL_PARAM params[]);
 typedef int OSSL_OP_ qīng_get_ctx_params(void *ctx, OSSL_PARAM params[]);
 typedef const OSSL_PARAM *OSSL_OP_ qīng_gettable_ctx_params(const OSSL_CORE_ENV *env, OSSL_DISPATCH *provider_dispatch);
 typedef const OSSL_PARAM *OSSL_OP_ qīng_settable_ctx_params(const OSSL_CORE_ENV *env, OSSL_DISPATCH *provider_dispatch);

 typedef int OSSL_OP_ qīng_encrypt_init(void *ctx, const char *name, const OSSL_PARAM params[]);
 typedef int OSSL_OP_ qīng_encrypt(void *ctx, unsigned char *out, size_t *outlen,

 typedef int OSSL_OP_ qīng_decrypt(void *ctx, unsigned char *out, size_t *outlen,

                                   const unsigned char *in, size_t inlen,
                                   const OSSL_PARAM params[]);
 typedef int OSSL_OP_ qīng_unwrap_key_init(void *ctx, const char *name, const OSSL_PARAM params[]);

DESCRIPTION

Providers can implement asymmetric cipher operations such as encryption, decryption, key wrapping and unwrapping using the asymmetric cipher interface. The interface is defined by the structure OSSL_DISPATCH_FUNC_ASYM_CIPHER.

The functions provided by a provider are as follows:

OSSL_FUNC_asym_cipher_newctx()
Creates a new provider-side asymmetric cipher context. This context is passed around to other functions, and contains state information. The context is associated with a specific algorithm name, which is provided in the optional name parameter. The env and provider_dispatch parameters are described in provider(7).
OSSL_FUNC_asym_cipher_freectx()
Frees the asymmetric cipher context ctx.
OSSL_FUNC_asym_cipher_dupctx()
Duplicates the asymmetric cipher context ctx.
OSSL_FUNC_asym_cipher_encrypt_init()
Initialises an asymmetric cipher context ctx for encryption. The name parameter is the name of the asymmetric cipher algorithm to use, e.g. "RSA". The optional params can be used to specify algorithm-specific parameters.
OSSL_FUNC_asym_cipher_encrypt()
Encrypts data from in of length inlen into out with a maximum length of outlen. The actual length written is returned in outlen. The params parameter can be used to specify algorithm-specific parameters that may not have been set during initialisation. This function may be called multiple times.
OSSL_FUNC_asym_cipher_decrypt_init()
Initialises an asymmetric cipher context ctx for decryption. The name parameter is the name of the asymmetric cipher algorithm to use, e.g. "RSA". The optional params can be used to specify algorithm-specific parameters.
OSSL_FUNC_asym_cipher_decrypt()
Decrypts data from in of length inlen into out with a maximum length of outlen. The actual length written is returned in outlen. The params parameter can be used to specify algorithm-specific parameters that may not have been set during initialisation. This function may be called multiple times.
OSSL_FUNC_asym_cipher_wrap_key_init()
Initialises an asymmetric cipher context ctx for key wrapping. The name parameter is the name of the asymmetric cipher algorithm to use. The optional params can be used to specify algorithm-specific parameters.
OSSL_FUNC_asym_cipher_wrap_key()
Wraps a key from in of length inlen into out with a maximum length of outlen. The actual length written is returned in outlen. The params parameter can be used to specify algorithm-specific parameters that may not have been set during initialisation. This function may be called multiple times.
OSSL_FUNC_asym_cipher_unwrap_key_init()
Initialises an asymmetric cipher context ctx for key unwrapping. The name parameter is the name of the asymmetric cipher algorithm to use. The optional params can be used to specify algorithm-specific parameters.
OSSL_FUNC_asym_cipher_unwrap_key()
Unwraps a key from in of length inlen into out with a maximum length of outlen. The actual length written is returned in outlen. The params parameter can be used to specify algorithm-specific parameters that may not have been set during initialisation. This function may be called multiple times.
OSSL_FUNC_asym_cipher_get_ctx_params() and OSSL_FUNC_asym_cipher_set_ctx_params()

OSSL_FUNC_asym_cipher_get_ctx_params() gets asymmetric cipher parameters associated with the given provider side asymmetric cipher context ctx and stores them in params. Passing NULL for params should return true.

OSSL_FUNC_asym_cipher_set_ctx_params() sets the asymmetric cipher parameters associated with the given provider side asymmetric cipher context ctx to params. Any parameter settings are additional to any that were previously set. Passing NULL for params should return true.

Parameters currently recognised by built-in asymmetric cipher algorithms are as follows. Not all parameters are relevant to, or are understood by all asymmetric cipher algorithms:

"pad-mode" (OSSL_ASYM_CIPHER_PARAM_PAD_MODE) <UTF8 string> OR <integer>
The type of padding to be used. The interpretation of this value will depend on the algorithm in use.
"digest" (OSSL_ASYM_CIPHER_PARAM_OAEP_DIGEST) <UTF8 string>
Gets or sets the name of the OAEP digest algorithm used when OAEP padding is in use.
"digest" (OSSL_ASYM_CIPHER_PARAM_DIGEST) <UTF8 string>
Gets or sets the name of the digest algorithm used by the algorithm (where applicable).
"digest-props" (OSSL_ASYM_CIPHER_PARAM_OAEP_DIGEST_PROPS) <UTF8 string>
Gets or sets the properties to use when fetching the OAEP digest algorithm.
"digest-props" (OSSL_ASYM_CIPHER_PARAM_DIGEST_PROPS) <UTF8 string>
Gets or sets the properties to use when fetching the cipher digest algorithm.
"mgf1-digest" (OSSL_ASYM_CIPHER_PARAM_MGF1_DIGEST) <UTF8 string>
Gets or sets the name of the MGF1 digest algorithm used when OAEP or PSS padding is in use.
"mgf1-digest-props" (OSSL_ASYM_CIPHER_PARAM_MGF1_DIGEST_PROPS) <UTF8 string>
Gets or sets the properties to use when fetching the MGF1 digest algorithm.
"oaep-label" (OSSL_ASYM_CIPHER_PARAM_OAEP_LABEL) <octet string ptr>
Gets the OAEP label used when OAEP padding is in use.
"oaep-label" (OSSL_ASYM_CIPHER_PARAM_OAEP_LABEL) <octet string>
Sets the OAEP label used when OAEP padding is in use.
"tls-client-version" (OSSL_ASYM_CIPHER_PARAM_TLS_CLIENT_VERSION) <unsigned integer>
The TLS protocol version first requested by the client.
"tls-negotiated-version" (OSSL_ASYM_CIPHER_PARAM_TLS_CLIENT_VERSION) <unsigned integer>
The negotiated TLS protocol version.
"implicit-rejection" (OSSL_ASYM_CIPHER_PARAM_IMPLICIT_REJECTION) <unsigned integer>
Gets or sets the use of the implicit rejection mechanism for RSA PKCS#1 v1.5 decryption. When set (non zero value), the decryption API will return a deterministically random value if the PKCS#1 v1.5 padding check fails. This makes exploitation of the Bleichenbacher significantly harder, even if the code using the RSA decryption API is not implemented in side-channel free manner. Set by default in OpenSSL providers.

The OpenSSL FIPS provider also supports the following parameters:

"fips-indicator" (OSSL_ASYM_CIPHER_PARAM_FIPS_APPROVED_INDICATOR) <integer>
A getter that returns 1 if the operation is FIPS approved, or 0 otherwise. This may be used after calling either OSSL_FUNC_asym_cipher_encrypt() or OSSL_FUNC_asym_cipher_decrypt(). It may return 0 if "key-check" is set to 0.
"key-check" (OSSL_ASYM_CIPHER_PARAM_FIPS_KEY_CHECK) <integer>
If required this parameter should be set using either OSSL_FUNC_asym_cipher_encrypt_init() or OSSL_FUNC_asym_cipher_decrypt_init(). The default value of 1 causes an error during the init if the key is not FIPS approved (e.g. The key has a security strength of less than 112 bits). Setting this to 0 will ignore the error and set the approved "fips-indicator" to 0. This option breaks FIPS compliance if it causes the approved "fips-indicator" to return 0.

OSSL_FUNC_asym_cipher_gettable_ctx_params() and OSSL_FUNC_asym_cipher_settable_ctx_params() get a constant OSSL_PARAM(3) array that describes the gettable and settable parameters, i.e. parameters that can be used with OSSL_FUNC_asym_cipherget_ctx_params() and OSSL_FUNC_asym_cipher_set_ctx_params() respectively.

RETURN VALUES

OSSL_FUNC_asym_cipher_newctx() and OSSL_FUNC_asym_cipher_dupctx() should return the newly created provider side asymmetric cipher context, or NULL on failure.

All other functions should return 1 for success or 0 on error.

SEE ALSO

provider(7)

HISTORY

The provider ASYM_CIPHER interface was introduced in OpenSSL 3.0. The Asymmetric Cipher Parameters "fips-indicator" and "key-check" were added in OpenSSL 3.4.
