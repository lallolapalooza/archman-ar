procps_misc(3)

الحزمة: procps-ng

التراخيص: GPL, LGPL

PROCPS_MISC(3) دليل وظائف المكتبة PROCPS_MISC(3)

الاسم

procps_misc - واجهة برمجة تطبيقات للمعلومات المتنوعة في نظام الملفات /proc

ملخص

#include <libproc2/misc.h>
تفاصيل المنصة
long         procps_cpu_count (void);
long         procps_hertz_get (void);
unsigned int procps_pid_length (void);
int          procps_linux_version (void);
تفاصيل وقت التشغيل
int   procps_container_uptime (double *uptime_secs);
int   procps_loadavg (double *av1, double *av5, double *av15);
int   procps_uptime (double *uptime_secs, double *idle_secs);
int   procps_uptime_snprint ( char *restrict str, size_t size, double uptime_secs, const int pretty);
char *procps_uptime_sprint (void);
char *procps_uptime_sprint_short (void);
int   procps_users (void);
تفاصيل مساحة الأسماء
int         procps_ns_get_id (const char *name);
const char *procps_ns_get_name (int id);
int         procps_ns_read_pid (int pid, struct procps_ns *nsp);
تحويل اسم القناع
int procps_sigmask_names (char *const str, size_t size , const char *sigmask);
int procps_capmask_names (char *const str, size_t size,  const char *capmask);
اربط مع -lproc2.

الوصف

procps_cpu_count() تُرجع عدد وحدات المعالجة المركزية النشطة حاليًا كما هو معرّف بواسطة sysconf(_SC_NPROCESSORS_ONLY) أو قيمة مفترضة 1.

procps_hertz_get() تُرجع عدد دورات الساعة في الثانية كما هو معرّف بواسطة sysconf(_SC_CLK_TCK) أو قيمة مفترضة 100. قسمة الدورات على هذه القيمة تعطي الثواني.

procps_pid_length() تُرجع أقصى طول سلسلة نصية لمعرف العملية (PID) على النظام. على سبيل المثال، إذا كانت أكبر قيمة ممكنة لمعرف العملية هي 123، فسيكون الطول 3. إذا كان الملف /proc/sys/kernel/pid_max غير قابل للقراءة، فسيتم افتراض القيمة 5.

procps_linux_version() تُرجع إصدار Linux الحالي كعدد صحيح مشفر. على الأنظمة غير Linux التي تحتوي على نظام ملفات proc محاكى، تُرجع هذه الوظيفة إصدار محاكاة Linux بدلاً من ذلك. يتكون الإصدار من ثلاثة أعداد صحيحة موجبة تمثل مستويات رئيسية وثانوية وتصحيحية. يتم توفير وحدات الماكرو التالية لتشفير إصدار Linux معين أو فصل مكونات الإصدار الحالي.

LINUX_VERSION( major , minor , patch )

LINUX_VERSION_MAJOR( ver )

LINUX_VERSION_MINOR( ver )

LINUX_VERSION_PATCH( ver )

procps_loadavg() تجلب متوسط تحميل النظام وتضع المتوسطات لمدة 1 و 5 و 15 دقيقة في المواقع المحددة بأي مؤشر ليس NULL.

procps_container_uptime() تُرجع وقت تشغيل الحاوية إلى الموقع المحدد بواسطة المؤشر إذا لم يكن NULL. يتم تحديد ذلك حاليًا بواسطة الوقت المنقضي لمعرف العملية 1 (PID 1)، والذي يمكن الحصول عليه بالأمر:


$ ps -o etimes 1
ELAPSED
12345

لا يوجد مفهوم لوقت الخمول في الحاوية، لذا فإن procps_container_uptime() لا تحتوي على هذا المعامل.

procps_uptime() تُرجع ثواني وقت التشغيل و/أو وقت الخمول إلى المواقع المحددة بأي مؤشر ليس NULL. تُرجع أنواع sprint سلسلة نصية سهلة القراءة في أحد شكلين.

HH:MM:SS up HH:MM, # users, load average: 1, 5, 15 MM averages

up HH, MM

procps_uptime_snprint() تستخدم المخزن المؤقت المحدد str بطول size لملء سلسلة وقت التشغيل بدلاً من المخزن المؤقت المخصص ثابتًا. تتطلب الوظيفة قيمة uptime_secs، والتي يمكن الحصول عليها من procps_uptime() أو procps_container_uptime() ويحدد العلامة pretty ما إذا كان الإخراج قياسيًا أم جميلًا/قصيرًا.

procps_users() تُرجع عدد المستخدمين على النظام. تأتي هذه القيمة من sd_get_sessions(3) ويتم عد جلسات الفئات user, user-early, و user-incomplete أو عن طريق التعداد عبر getutent(3).

procps_ns_get_id() تُرجع المعرف العددي (enum namespace_type) لمساحة الأسماء للاسم مساحة الأسماء المعطى name.

procps_ns_get_name() تُرجع اسم مساحة الأسماء للمعّرف المعطى id (enum namespace_type).

procps_ns_read_pid() تُرجع inodes مساحات الأسماء للعملية المعطاة في بنية procps_ns المشار إليها بواسطة nsp. ستظهر هذه الـ inodes بالترتيب المحدد بواسطة enum namespace_type.

enum namespace_type {
    PROCPS_NS_CGROUP,
    PROCPS_NS_IPC,
    PROCPS_NS_MNT,
    PROCPS_NS_NET,
    PROCPS_NS_PID,
    PROCPS_NS_TIME,
    PROCPS_NS_USER,
    PROCPS_NS_UTS
};

procps_sigmask_names() تملأ str بسلسلة نصية سهلة القراءة تصل إلى size بايت من الإشارات المعينة في قناع الإشارة السداسي عشري sigmask، انظر signal(7). إذا تجاوزت قائمة الإشارات size بايت، فسيتم اقتطاع السلسلة النصية وستنتهي بـ "+". يقع على عهدة البرنامج المتصل ضمان تخصيص str بشكل صحيح بـ size بايت على الأقل.

procps_capmask_names() تملأ str بسلسلة نصية سهلة القراءة تصل إلى size بايت من القدرات المعينة في قناع القدرات السداسي عشري capmask، انظر capabilities(7). إذا تجاوزت قائمة القدرات size بايت، فسيتم اقتطاع السلسلة النصية وستنتهي بـ "+". يقع على عهدة البرنامج المتصل ضمان تخصيص str بشكل صحيح بـ size بايت على الأقل.

بالنسبة لعملية ليس لديها قدرات أو لديها جميع القدرات، ستكون السلسلة النصية "-" و "full" على التوالي.

قيمة الإرجاع

الوظائف التي تُرجع ‘int’ أو ‘long’

سيتم الإشارة إلى خطأ برقم سالب هو دائمًا عكس قيمة errno.h معروفة بشكل جيد.

الوظائف التي تُرجع ‘address’

سيتم الإشارة إلى خطأ بمؤشر إرجاع NULL مع وجود السبب في قيمة errno الرسمية.

الملفات

/proc/loadavg
القيم الخام لمتوسط التحميل.
/proc/sys/kernel/osrelease
يحتوي على إصدار إصدار نواة Linux أو نظام الملفات proc.
/proc/sys/kernel/pid_max
يحتوي على القيمة التي تلتف عندها معرفات العمليات (PIDs)، وهي أكبر بواحد من أقصى قيمة لمعرف العملية.
/proc/uptime
القيم الخام لوقت تشغيل النظام ووقت الخمول.
/proc/PID/ns
يحتوي على مجموعة مساحات الأسماء لـ PID معين.
