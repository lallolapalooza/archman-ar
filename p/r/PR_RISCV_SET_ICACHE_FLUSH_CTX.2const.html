PR_RISCV_SET_ICACHE_FLUSH_CTX(2const) PR_RISCV_SET_ICACHE_FLUSH_CTX(2const)

الاسم

PR_RISCV_SET_ICACHE_FLUSH_CTX - تمكين/تعطيل تعليمات مسح ذاكرة التخزين المؤقت للتعليمات في مساحة المستخدم.

المكتبة

مكتبة C القياسية (libc, -lc)

الملخص

#include <linux/prctl.h>  /* تعريف ثوابت PR_* */
#include <sys/prctl.h>
int prctl(PR_RISCV_SET_ICACHE_FLUSH_CTX, unsigned long ctx,
          unsigned long scope);

الوصف

يمكن توفير السياق والنطاق باستخدام ctx و scope على التوالي.

عند تعيين النطاق إلى PR_RISCV_SCOPE_PER_PROCESS، يُسمح لجميع الخيوط في العملية بإصدار تعليمات مسح ذاكرة التخزين المؤقت للتعليمات. كلما تم ترحيل أي خيط في العملية، سيتم ضمان اتساق ذاكرة التخزين المؤقت للتعليمات الخاصة بـ hart المقابل مع تخزين التعليمات. هذا لا يفرض أي ضمانات خارج نطاق الترحيل. إذا قام خيط بتعديل تعليمات قد يحاول خيط آخر تنفيذها، فيجب على الخيط الآخر أن يصدر تعليمات مسح ذاكرة التخزين المؤقت للتعليمات قبل محاولة تنفيذ التعليمات المعدلة المحتملة. يجب أن يتم ذلك بواسطة برنامج مساحة المستخدم.

في سياق الخيط الواحد (على سبيل المثال، إذا تم تعيين النطاق إلى PR_RISCV_SCOPE_PER_THREAD)، يُسمح فقط للخيط الذي يستدعي هذه الدالة بإصدار تعليمات مسح ذاكرة التخزين المؤقت للتعليمات. عند ترحيل الخيط، سيتم ضمان اتساق ذاكرة التخزين المؤقت للتعليمات الخاصة بـ hart المقابل مع تخزين التعليمات.

على النوى التي تم تكوينها بدون SMP، فإن هذا prctl PR_RISCV_SET_ICACHE_FLUSH_CTX لا يفعل شيئًا حيث لن تحدث عمليات الترحيل عبر harts.

يمكن تحديد القيم التالية لـ ctx:

PR_RISCV_CTX_SW_FENCEI_ON (منذ Linux 6.10)
السماح بـ fence.i في مساحة المستخدم.
PR_RISCV_CTX_SW_FENCEI_OFF (منذ Linux 6.10)
منع fence.i في مساحة المستخدم. ستتأثر جميع الخيوط في العملية عند تعيين النطاق إلى PR_RISCV_SCOPE_PER_PROCESS. لذلك، يجب توخي الحذر؛ استخدم هذا العلم فقط عندما يمكنك ضمان أن أي خيط في العملية لن يصدر fence.i من هذه النقطة فصاعدًا.

يمكن تحديد القيم التالية لـ scope:

PR_RISCV_SCOPE_PER_PROCESS (منذ Linux 6.10)
ضمان أن ذاكرة التخزين المؤقت للتعليمات لأي خيط في هذه العملية متسقة مع تخزين التعليمات عند الترحيل.
PR_RISCV_SCOPE_PER_THREAD (منذ Linux 6.10)
ضمان أن ذاكرة التخزين المؤقت للتعليمات للخيط الحالي متسقة مع تخزين التعليمات عند الترحيل.

أمثلة

الملفات التالية مخصصة للترجمة والربط مع بعضها البعض. تقوم الدالة modify_instruction() باستبدال الجمع بصفر بجمع بواحد، مما يتسبب في تغيير تسلسل التعليمات في get_value() من إرجاع صفر إلى إرجاع واحد.

مصدر البرنامج: cmodx.c

#include <stdio.h>
#include <sys/prctl.h>
extern int get_value(void);
extern void modify_instruction(void);
int
main(void)
{
    int value = get_value();
    printf("القيمة قبل cmodx: %d\n", value);
    // استدعاء prctl قبل استدعاء fence.i الأول
    prctl(PR_RISCV_SET_ICACHE_FLUSH_CTX, PR_RISCV_CTX_SW_FENCEI_ON,
          PR_RISCV_SCOPE_PER_PROCESS);
    modify_instruction();
    // استدعاء prctl بعد استدعاء fence.i الأخير في العملية
    prctl(PR_RISCV_SET_ICACHE_FLUSH_CTX, PR_RISCV_CTX_SW_FENCEI_OFF,
          PR_RISCV_SCOPE_PER_PROCESS);
    value = get_value();
    printf("القيمة بعد cmodx: %d\n", value);
    return 0;
}

مصدر البرنامج: cmodx.S

.option norvc
.text
.global modify_instruction
modify_instruction:
lw a0, new_insn
lui a5,%hi(old_insn)

fence.i
ret

.global get_value

li a0, 0

addi a0, a0, 0
ret
.data

addi a0, a0, 1
