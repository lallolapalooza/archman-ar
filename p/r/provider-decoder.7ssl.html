الاسم

provider-decoder - مكتبة OSSL_DECODER <-> دوال المزود

الملخص

#include <openssl/core_dispatch.h>
/*
 * لا توجد أي من هذه دوال فعلية، ولكنها معروضة بهذه الطريقة لتوقيعات الدوال
 * التي تُعرض كمؤشرات دوال في مصفوفات OSSL_DISPATCH.
 */
/* وصول إلى معلمات الفك وتوصيفها */
const OSSL_PARAM *OSSL_FUNC_decoder_gettable_params(void *provctx);
int OSSL_FUNC_decoder_get_params(OSSL_PARAM params[]);
/* دوال لإنشاء / تدمير / معالجة سياق الفك */
void *OSSL_FUNC_decoder_newctx(void *provctx);
void OSSL_FUNC_decoder_freectx(void *ctx);
const OSSL_PARAM *OSSL_FUNC_decoder_settable_ctx_params(void *provctx);
int OSSL_FUNC_decoder_set_ctx_params(void *ctx, const OSSL_PARAM params[]);
/* دوال للتحقق من دعم التحديد */
int OSSL_FUNC_decoder_does_selection(void *provctx, int selection);
/* دوال لفك بيانات الكائن */
int OSSL_FUNC_decoder_decode(void *ctx, OSSL_CORE_BIO *in,
                             int selection,
                             OSSL_CALLBACK *data_cb, void *data_cbarg,
                             OSSL_PASSPHRASE_CALLBACK *cb, void *cbarg);
/* دوال لتصدير كائن مفكوك */
int OSSL_FUNC_decoder_export_object(void *ctx,
                                      const void *objref, size_t objref_sz,
                                      OSSL_CALLBACK *export_cb,
                                      void *export_cbarg);

الوصف

يُستخدم مصطلح "فك" في هذا الدليل. وهذا يشمل، ولكن لا يقتصر على، إلغاء التسلسل حيث يمكن لموفكات فردية أيضًا القيام بـ فك البيانات إلى تنسيقات بيانات وسيطة.

عملية DECODER هي طريقة عامة لإنشاء مرجع كائن خاص بالمزود أو بيانات مفككة وسيطة من شكل مشفر مقروء من OSSL_CORE_BIO المعطى. إذا أراد المتصل فك البيانات من الذاكرة، فيجب عليه توفير BIO_s_mem(3) BIO. يتم تمرير البيانات المفككة أو مرجع الكائن مع أي بيانات وصفية محتملة إلى metadata_cb كمعلمات OSSL_PARAM(3).

لا يحتاج المزود إلى معرفة المزيد عن مؤشر OSSL_CORE_BIO بخلاف كونه قادرًا على تمريره إلى استدعاءات BIO المناسبة (انظر "وظائف النواة" في provider-base(7)).

قد يكون تطبيق الفك جزءًا من سلسلة، حيث يتم تمرير البيانات من واحد إلى الآخر. على سبيل المثال، قد يكون هناك تطبيق لفك كائن من PEM إلى DER، وآخر يفك DER إلى كائن خاص بالمزود.

عادةً ما يُفترض أن تكون خطوة الفك الأخيرة في سلسلة الفك هي إنشاء كائن خاص بالمزود يُشار إليه بمرجع كائن. لاستيراد هذا الكائن إلى مزود مختلف، يمكن استدعاء OSSL_FUNC_decoder_export_object() كخطوة أخيرة في عملية الفك.

جميع "الدوال" المذكورة هنا يتم تمريرها كمؤشرات دوال بين libcrypto والمزود في مصفوفات OSSL_DISPATCH(3) عبر مصفوفات OSSL_ALGORITHM(3) التي يتم إرجاعها بواسطة دالة المزود provider_query_operation() (انظر "دوال المزود" في provider-base(7)).

جميع هذه "الدوال" لها تعريف نوع دالة مطابق اسمه OSSL_FUNC_{name}_fn، ودالة مساعدة لاسترداد مؤشر الدالة من عنصر OSSL_DISPATCH(3) يسمى OSSL_FUNC_{name}. على سبيل المثال، "الدالة" OSSL_FUNC_decoder_decode() لها ما يلي:

typedef int
    (OSSL_FUNC_decoder_decode_fn)(void *ctx, OSSL_CORE_BIO *in,
                                  int selection,
                                  OSSL_CALLBACK *data_cb, void *data_cbarg,
                                  OSSL_PASSPHRASE_CALLBACK *cb, void *cbarg);
static ossl_inline OSSL_FUNC_decoder_decode_fn*
    OSSL_FUNC_decoder_decode(const OSSL_DISPATCH *opf);

يتم فهرسة مصفوفات OSSL_DISPATCH(3) بواسطة أرقام يتم توفيرها كـ macros في openssl-core_dispatch.h(7)، كما يلي:

OSSL_FUNC_decoder_get_params          OSSL_FUNC_DECODER_GET_PARAMS
OSSL_FUNC_decoder_gettable_params     OSSL_FUNC_DECODER_GETTABLE_PARAMS
OSSL_FUNC_decoder_newctx              OSSL_FUNC_DECODER_NEWCTX
OSSL_FUNC_decoder_freectx             OSSL_FUNC_DECODER_FREECTX
OSSL_FUNC_decoder_set_ctx_params      OSSL_FUNC_DECODER_SET_CTX_PARAMS
OSSL_FUNC_decoder_settable_ctx_params OSSL_FUNC_DECODER_SETTABLE_CTX_PARAMS
OSSL_FUNC_decoder_does_selection      OSSL_FUNC_DECODER_DOES_SELECTION
OSSL_FUNC_decoder_decode              OSSL_FUNC_DECODER_DECODE
OSSL_FUNC_decoder_export_object       OSSL_FUNC_DECODER_EXPORT_OBJECT

الأسماء والخصائص

يجب أن يتطابق اسم التنفيذ مع نوع الكائن المستهدف الذي يفكه. على سبيل المثال، يجب تسمية تنفيذ يفك مفتاح RSA باسم "RSA". وبالمثل، يجب تسمية تنفيذ يفك بيانات DER من إدخال PEM باسم "DER".

يمكن استخدام الخصائص، كما هو محدد في عنصر مصفوفة OSSL_ALGORITHM(3) لكل تنفيذ المزود، لتحديد تفاصيل إضافية حول التنفيذ:

input
تُستخدم هذه الخاصية لتحديد تنسيق الإدخال الذي يمكن للتنفيذ فكه.

هذه الخاصية إلزامية.

مزودات OpenSSL تتعرف على أنواع الإدخال التالية:

pem
يقوم تنفيذ بهذا النوع من الإدخال بفك بيانات بتنسيق PEM.
der
يقوم تنفيذ بهذا النوع من الإدخال بفك بيانات بتنسيق DER.
msblob
يقوم تنفيذ بهذا النوع من الإدخال بفك بيانات بتنسيق MSBLOB.
pvk
يقوم تنفيذ بهذا النوع من الإدخال بفك بيانات بتنسيق PVK.
structure
تُستخدم هذه الخاصية لتحديد البنية المتوقع أن تحتوي عليها البيانات المفككة.

هذه الخاصية اختيارية.

الهياكل المعترف بها حاليًا بواسطة مفكات مدمجة:

"type-specific"
بنية خاصة بالنوع.
"pkcs8"
بنية وفقًا لمواصفات PKCS#8.
"SubjectPublicKeyInfo"
ترميز المفاتيح العامة وفقًا للمعلومات العامة للموضوع (Subject Public Key Info) في RFC 5280.

قيم هاتين الخاصيتين مفتوحة. يمكن لمزود أن يحدد أنواع إدخال وهياكل لا يعرف عنها libcrypto شيئًا.

تحديدات المجموعات الفرعية

في بعض الأحيان، يحتوي الكائن على أكثر من مجموعة فرعية واحدة من البيانات التي تهم المعالجة بشكل منفصل أو معًا. من الممكن تحديد المجموعات الفرعية التي يجب فكها، باستخدام مجموعة من البتات selection التي يتم تمريرها في int.

تعتمد مجموعة البتات هذه كليًا على نوع الكائن الخاص بالمزود الذي سيتم فكه. على سبيل المثال، يُفترض أن تكون هذه البتات هي نفسها تلك المستخدمة مع provider-keymgmt(7) (انظر "كائنات المفاتيح" في provider-keymgmt(7)) عندما يكون الكائن زوج مفاتيح غير متناظر - على سبيل المثال، OSSL_KEYMGMT_SELECT_PRIVATE_KEY إذا كان الكائن المراد فكه من المفترض أن يحتوي على مكونات مفتاح خاص.

يجب أن يوضح OSSL_FUNC_decoder_does_selection() ما إذا كان تنفيذ معين يدعم أيًا من التركيبات المعطاة بواسطة selection.

دوال السياق

OSSL_FUNC_decoder_newctx() تُرجع سياقًا لاستخدامه مع بقية الدوال.

OSSL_FUNC_decoder_freectx() تقوم بتحرير ctx المعطى كما تم إنشاؤه بواسطة OSSL_FUNC_decoder_newctx().

OSSL_FUNC_decoder_set_ctx_params() تقوم بتعيين بيانات السياق وفقًا للمعلمات من params التي تتعرف عليها. يجب تجاهل المعلمات غير المتعرف عليها. تمرير NULL لـ params يجب أن يُرجع صحيحًا.

OSSL_FUNC_decoder_settable_ctx_params() تُرجع مصفوفة OSSL_PARAM ثابتة تحدد المعلمات التي يمكن تعيينها إلى سياق المزود.

انظر أيضًا

OSSL_PARAM(3)، provider-base(7)، provider-keymgmt(7)

NAME

provider-decoder - Decoder interface for OpenSSL providers

SYNOPSIS

 OSSL_DECODER_CTX *OSSL_FUNC_decoder_newctx(void *vctx);
 int OSL_FUNC_decoder_set_ctx_params(OSSL_DECODER_CTX *ctx,
     const OSSL_PARAM params[]);
 const OSSL_PARAM *OSSL_FUNC_decoder_settable_ctx_params(OSSL_DECODER_CTX *ctx);
 int OSL_FUNC_decoder_does_selection(uint32_t selection);
 int OSL_FUNC_decoder_decode(OSSL_DECODER_CTX *ctx,
     void **out, const unsigned char **in, size_t inl,
     OSSL_PASSPHRASE_CALLBACK *pass, void *passarg);
 int OSL_FUNC_decoder_export_object(OSSL_DECODER_CTX *ctx,
     const void *objref, size_t objref_sz,
     OSSL_CORE_BIO *export_cb, void *export_cbarg);
 void OSL_FUNC_decoder_freectx(OSSL_DECODER_CTX *ctx);

DESCRIPTION

The DECODER interface is used by consumers of OpenSSL provider functionality to decode data into OpenSSL objects. This is typically used for loading cryptographic keys, parameters, or other data structures from external representations such as PEM or DER encoded data.

Providers that want to support decoding of certain data types implement functions within the DECODER dispatch table.

Context management

OSSL_FUNC_decoder_newctx() creates and initialises a decoder context. The supplied vctx is the general provider context.

OSSL_FUNC_decoder_freectx() cleans up and frees a decoder context.

Parameter management

OSSL_FUNC_decoder_set_ctx_params() sets parameters in the decoder context. The supported parameters are described by the OSSL_PARAM(3) array returned by OSSL_FUNC_decoder_settable_ctx_params().

OSSL_FUNC_decoder_settable_ctx_params() returns an OSSL_PARAM(3) array describing the parameters that OSSL_FUNC_decoder_set_ctx_params() can handle.

See OSSL_PARAM(3) for further details on the parameters structure used by OSSL_FUNC_decoder_set_ctx_params() and OSSL_FUNC_decoder_settable_ctx_params().

Export function

When a provider-native object is created by a decoder it would be unsuitable for direct use with a foreign provider. The export function allows for exporting the object into that foreign provider if the foreign provider supports the type of the object and provides an import function.

OSSL_FUNC_decoder_export_object() should export the object of size objref_sz referenced by objref as an OSSL_PARAM(3) array and pass that into the export_cb as well as the given export_cbarg.

Decoding functions

OSSL_FUNC_decoder_decode() should decode the data as read from the OSSL_CORE_BIO in to produce decoded data or an object to be passed as reference in an OSSL_PARAM(3) array along with possible other metadata that was decoded from the input. This OSSL_PARAM(3) array is then passed to the data_cb callback. The selection bits, if relevant, should determine what the input data should contain. The decoding functions also take an OSSL_PASSPHRASE_CALLBACK(3) function pointer along with a pointer to application data cbarg, which should be used when a pass phrase prompt is needed.

It's important to understand that the return value from this function is interpreted as follows:

True (1)
This means "carry on the decoding process", and is meaningful even though this function couldn't decode the input into anything, because there may be another decoder implementation that can decode it into something.

The data_cb callback should never be called when this function can't decode the input into anything.

False (0)
This means "stop the decoding process", and is meaningful when the input could be decoded into some sort of object that this function understands, but further treatment of that object results into errors that won't be possible for some other decoder implementation to get a different result.

The conditions to stop the decoding process are at the discretion of the implementation.

Decoder operation parameters

There are currently no operation parameters currently recognised by the built-in decoders.

Parameters currently recognised by the built-in pass phrase callback:

"info" (OSSL_PASSPHRASE_PARAM_INFO) <UTF8 string>
A string of information that will become part of the pass phrase prompt. This could be used to give the user information on what kind of object it's being prompted for.

RETURN VALUES

OSSL_FUNC_decoder_newctx() returns a pointer to a context, or NULL on failure.

OSSL_FUNC_decoder_set_ctx_params() returns 1, unless a recognised parameter was invalid or caused an error, for which 0 is returned.

OSSL_FUNC_decoder_settable_ctx_params() returns a pointer to an array of constant OSSL_PARAM(3) elements.

OSSL_FUNC_decoder_does_selection() returns 1 if the decoder implementation supports any of the selection bits, otherwise 0.

OSSL_FUNC_decoder_decode() returns 1 to signal that the decoding process should continue, or 0 to signal that it should stop.
