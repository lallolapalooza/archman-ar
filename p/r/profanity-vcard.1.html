profanity-vcard(1)

الحزمة: profanity-gtk

التراخيص: GPL3

man(1) عميل Profanity XMPP man(1)

الاسم

/vcard

الوصف

قراءة بطاقة vCard الخاصة بك أو بطاقة vCard لمستخدم، الحصول على صورة شخصية لمستخدم عبر بطاقة vCard الخاصة به، أو تعديل بطاقة vCard الخاصة بك. إذا لم يتم تقديم أي وسائط، فسيتم عرض بطاقة vCard الخاصة بك في نافذة جديدة، أو نافذة vCard موجودة.

الاستخدام

/vcard get [<اسم مستعار|جهة اتصال>]

/vcard photo open <اسم مستعار|جهة اتصال> [<فهرس>]

/vcard photo save <اسم مستعار|جهة اتصال> [output <مسار الملف>] [index <فهرس>]

/vcard set fullname <الاسم الكامل>

الوسائط

get [<اسم مستعار|جهة اتصال>]

احصل على بطاقة vCard الخاصة بك، إذا تم توفير اسم مستعار/جهة اتصال، فاحصل على بطاقة vCard الخاصة بهذا المستخدم.

photo open <اسم مستعار|جهة اتصال> [<فهرس>]

قم بتنزيل صورة مستخدم من بطاقة vCard الخاصة به إلى ملف، وافتحها. إذا لم يتم تحديد الفهرس، فقم بتنزيل الصورة الأولى (عادةً الصورة الرمزية) من بطاقة vCard الخاصة به.

photo save <اسم مستعار|جهة اتصال>

قم بتنزيل صورة مستخدم من بطاقة vCard الخاصة به إلى ملف. إذا لم يتم تحديد الفهرس، فقم بتنزيل الصورة الأولى (عادةً الصورة الرمزية) من بطاقة vCard الخاصة به. إذا لم يتم تحديد المخرج، فقم بتنزيل الصورة إلى دليل الصور الخاص بـ profanity.

photo open-self [<فهرس>]

قم بتنزيل صورة من بطاقة vCard الخاصة بك إلى ملف، وافتحها. إذا لم يتم تحديد الفهرس، فقم بتنزيل الصورة الأولى (عادةً الصورة الرمزية) من بطاقة vCard الخاصة بك.

photo save-self

قم بتنزيل صورة من بطاقة vCard الخاصة بك إلى ملف. إذا لم يتم تحديد الفهرس، فقم بتنزيل الصورة الأولى (عادةً الصورة الرمزية) من بطاقة vCard الخاصة بك. إذا لم يتم تحديد المخرج، فقم بتنزيل الصورة إلى دليل الصور الخاص بـ profanity. نفس وسائط `photo open`

set fullname <الاسم الكامل>

قم بتعيين الاسم الكامل لبطاقة vCard الخاصة بك إلى القيمة المحددة.

set name family <اسم العائلة>

قم بتعيين اسم عائلة بطاقة vCard الخاصة بك إلى القيمة المحددة.

set name given <الاسم الأول>

قم بتعيين الاسم الأول لبطاقة vCard الخاصة بك إلى القيمة المحددة.

set name middle <الاسم الأوسط>

قم بتعيين الاسم الأوسط لبطاقة vCard الخاصة بك إلى القيمة المحددة.

set name prefix <بادئة الاسم>

قم بتعيين بادئة اسم بطاقة vCard الخاصة بك إلى القيمة المحددة.

set name suffix <لاحقة الاسم>

قم بتعيين لاحقة اسم بطاقة vCard الخاصة بك إلى القيمة المحددة.

set <فهرس> [<قيمة>]

قم بتعيين الحقل الرئيسي في عنصر في بطاقة vCard الخاصة بك إلى القيمة المحددة، أو إذا لم يتم تحديد قيمة، فقم بتعديل الحقل في محرر. هذا يعمل فقط في العناصر التي تحتوي على حقل واحد.

set <فهرس> pobox <قيمة>

قم بتعيين صندوق البريد في عنصر عنوان في بطاقة vCard الخاصة بك إلى القيمة المحددة.

set <فهرس> extaddr <قيمة>

قم بتعيين العنوان الموسع في عنصر عنوان في بطاقة vCard الخاصة بك إلى القيمة المحددة.

set <فهرس> street <قيمة>

قم بتعيين الشارع في عنصر عنوان في بطاقة vCard الخاصة بك إلى القيمة المحددة.

set <فهرس> locality <قيمة>

قم بتعيين المدينة في عنصر عنوان في بطاقة vCard الخاصة بك إلى القيمة المحددة.

set <فهرس> region <قيمة>

قم بتعيين المنطقة في عنصر عنوان في بطاقة vCard الخاصة بك إلى القيمة المحددة.

set <فهرس> pocode <قيمة>

قم بتعيين رمز صندوق البريد في عنصر عنوان في بطاقة vCard الخاصة بك إلى القيمة المحددة.

set <فهرس> country <قيمة>

قم بتعيين البلد في عنصر عنوان في بطاقة vCard الخاصة بك إلى القيمة المحددة.

set <فهرس> type domestic|international

قم بتعيين نوع عنصر عنوان في بطاقة vCard الخاصة بك إلى محلي أو دولي.

set <فهرس> home on|off

قم بتعيين خيار المنزل في عنصر في بطاقة vCard الخاصة بك. (عنوان، هاتف، بريد إلكتروني فقط)

set <فهرس> work on|off

قم بتعيين خيار العمل في عنصر في بطاقة vCard الخاصة بك. (عنوان، هاتف، بريد إلكتروني فقط)

قم بتعيين خيار الصوت في عنصر هاتف في بطاقة vCard الخاصة بك.

set <فهرس> fax on|off

قم بتعيين خيار الفاكس في عنصر هاتف في بطاقة vCard الخاصة بك.

set <فهرس> pager on|off

قم بتعيين خيار جهاز النداء في عنصر هاتف في بطاقة vCard الخاصة بك.

set <فهرس> msg on|off

قم بتعيين خيار الرسائل في عنصر هاتف في بطاقة vCard الخاصة بك.

set <فهرس> cell on|off

قم بتعيين خيار الهاتف المحمول في عنصر هاتف في بطاقة vCard الخاصة بك.

set <فهرس> video on|off

قم بتعيين خيار الفيديو في عنصر هاتف في بطاقة vCard الخاصة بك.

set <فهرس> bbs on|off

قم بتعيين خيار BBS في عنصر هاتف في بطاقة vCard الخاصة بك.

قم بتعيين خيار المودم في عنصر هاتف في بطاقة vCard الخاصة بك.

قم بتعيين خيار ISDN في عنصر هاتف في بطاقة vCard الخاصة بك.

set <فهرس> pcs on|off

set <فهرس> preferred on|off

قم بتعيين الخيار المفضل في عنصر في بطاقة vCard الخاصة بك. (عنوان، هاتف، بريد إلكتروني فقط)

set <فهرس> parcel on|off

قم بتعيين خيار الطرود في عنصر عنوان في بطاقة vCard الخاصة بك.

set <فهرس> postal on|off

قم بتعيين الخيار البريدي في عنصر عنوان في بطاقة vCard الخاصة بك.

set <فهرس> internet on|off

set <فهرس> x400 on|off

قم بتعيين خيار X400 في عنوان بريد إلكتروني في بطاقة vCard الخاصة بك.
