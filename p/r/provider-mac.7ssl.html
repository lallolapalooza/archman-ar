provider-mac(7ssl)

الحزمة: openssl

التراخيص: Apache-2.0

PROVIDER-MAC(7ssl) OpenSSL PROVIDER-MAC(7ssl)

الاسم

provider-mac - وظائف مكتبة mac <-> المزود

موجز

#include <openssl/core_dispatch.h>
#include <openssl/core_names.h>
/*
 * لا توجد أي من هذه الوظائف بالفعل، ولكنها تعرض بهذه الطريقة لتوقيعات
 * الوظائف التي يتم تقديمها كمؤشرات وظائف في مصفوفات OSSL_DISPATCH.
 */
/* وظائف إدارة السياق */
void *OSSL_FUNC_mac_newctx(void *provctx);
void OSSL_FUNC_mac_freectx(void *mctx);
void *OSSL_FUNC_mac_dupctx(void *src);
/* وظائف التشفير/فك التشفير */
int OSSL_FUNC_mac_init(void *mctx, unsigned char *key, size_t keylen,
                       const OSSL_PARAM params[]);
int OSSL_FUNC_mac_init_skey(void *mctx, const void *key, const OSSL_PARAM params[]);
int OSSL_FUNC_mac_update(void *mctx, const unsigned char *in, size_t inl);
int OSSL_FUNC_mac_final(void *mctx, unsigned char *out, size_t *outl, size_t outsize);
/* واصفات معلمات MAC */
const OSSL_PARAM *OSSL_FUNC_mac_gettable_params(void *provctx);
const OSSL_PARAM *OSSL_FUNC_mac_gettable_ctx_params(void *mctx, void *provctx);
const OSSL_PARAM *OSSL_FUNC_mac_settable_ctx_params(void *mctx, void *provctx);
/* معلمات MAC */
int OSSL_FUNC_mac_get_params(OSSL_PARAM params[]);
int OSSL_FUNC_mac_get_ctx_params(void *mctx, OSSL_PARAM params[]);
int OSSL_FUNC_mac_set_ctx_params(void *mctx, const OSSL_PARAM params[]);

الوصف

هذه الوثائق موجهة بشكل أساسي لمؤلفي المزودين. انظر provider(7) لمزيد من المعلومات.

تتيح عملية MAC للمزودين تنفيذ خوارزميات mac وتقديمها للتطبيقات عبر وظائف API EVP_MAC_init(3)، EVP_MAC_update(3) و EVP_MAC_final(3).

جميع "الوظائف" المذكورة هنا يتم تمريرها كمؤشرات وظائف بين libcrypto والمزود في مصفوفات OSSL_DISPATCH(3) عبر مصفوفات OSSL_ALGORITHM(3) التي يتم إرجاعها من وظيفة provider_query_operation() الخاصة بالمزود (انظر "وظائف المزود" في provider-base(7)).

جميع هذه "الوظائف" لها نوع وظيفة مطابق يسمى OSSL_FUNC_{name}_fn، ووظيفة مساعدة لاسترداد مؤشر الوظيفة من عنصر OSSL_DISPATCH(3) يسمى OSSL_FUNC_{name}. على سبيل المثال، "الوظيفة" OSSL_FUNC_mac_newctx() لديها ما يلي:

typedef void *(OSSL_FUNC_mac_newctx_fn)(void *provctx);
static ossl_inline OSSL_FUNC_mac_newctx_fn
    OSSL_FUNC_mac_newctx(const OSSL_DISPATCH *opf);

يتم فهرسة مصفوفات OSSL_DISPATCH(3) بأرقام مقدمة كـ macros في openssl-core_dispatch.h(7)، على النحو التالي:

OSSL_FUNC_mac_newctx               OSSL_FUNC_MAC_NEWCTX
OSSL_FUNC_mac_freectx              OSSL_FUNC_MAC_FREECTX
OSSL_FUNC_mac_dupctx               OSSL_FUNC_MAC_DUPCTX
OSSL_FUNC_mac_init                 OSSL_FUNC_MAC_INIT
OSSL_FUNC_mac_init_skey            OSSL_FUNC_MAC_INIT_SKEY
OSSL_FUNC_mac_update               OSSL_FUNC_MAC_UPDATE
OSSL_FUNC_mac_final                OSSL_FUNC_MAC_FINAL
OSSL_FUNC_mac_get_params           OSSL_FUNC_MAC_GET_PARAMS
OSSL_FUNC_mac_get_ctx_params       OSSL_FUNC_MAC_GET_CTX_PARAMS
OSSL_FUNC_mac_set_ctx_params       OSSL_FUNC_MAC_SET_CTX_PARAMS
OSSL_FUNC_mac_gettable_params      OSSL_FUNC_MAC_GETTABLE_PARAMS
OSSL_FUNC_mac_gettable_ctx_params  OSSL_FUNC_MAC_GETTABLE_CTX_PARAMS
OSSL_FUNC_mac_settable_ctx_params  OSSL_FUNC_MAC_SETTABLE_CTX_PARAMS

قد لا تنفذ خوارزمية mac معينة كل هذه الوظائف. لكي تكون مجموعة متسقة من الوظائف، يجب تنفيذ الوظائف التالية على الأقل: OSSL_FUNC_mac_newctx()، OSSL_FUNC_mac_freectx()، واحدة على الأقل من OSSL_FUNC_mac_init() أو OSSL_FUNC_mac_init_skey()، OSSL_FUNC_mac_update()، OSSL_FUNC_mac_final(). جميع الوظائف الأخرى اختيارية.

وظائف إدارة السياق

يجب أن تقوم OSSL_FUNC_mac_newctx() بإنشاء وإرجاع مؤشر إلى هيكل جانب المزود لتخزين معلومات السياق أثناء عملية mac. سيتم تمرير مؤشر إلى هذا السياق مرة أخرى في عدد من استدعاءات وظائف عملية mac الأخرى. المعلمة provctx هي سياق المزود الذي تم إنشاؤه أثناء تهيئة المزود (انظر provider(7)).

يتم تمرير OSSL_FUNC_mac_freectx() مؤشرًا إلى سياق mac جانب المزود في المعلمة mctx. إذا استقبلت NULL كقيمة لـ mctx، فيجب ألا تفعل شيئًا سوى العودة. يجب أن تقوم هذه الوظيفة بتحرير أي موارد مرتبطة بهذا السياق.

يجب أن تقوم OSSL_FUNC_mac_dupctx() بتكرار سياق mac جانب المزود في المعلمة mctx وإرجاع نسخة مكررة.

وظائف التشفير/فك التشفير

تقوم OSSL_FUNC_mac_init() بتهيئة عملية mac بالنظر إلى سياق mac تم إنشاؤه حديثًا جانب المزود في المعلمة mctx. يتم تعيين params قبل تعيين مفتاح MAC key بطول keylen بايت.

OSSL_FUNC_mac_init_skey() مشابه ولكنه يستخدم كائنًا خاصًا بالمزود غير شفاف لتهيئة سياق MAC.

يتم استدعاء OSSL_FUNC_mac_update() لتوفير بيانات لحساب MAC لعملية mac مهيأة مسبقًا. تحتوي المعلمة mctx على مؤشر إلى سياق جانب المزود مهيأ مسبقًا. قد يتم استدعاء OSSL_FUNC_mac_update() عدة مرات لعملية mac واحدة.

تكمل OSSL_FUNC_mac_final() حساب MAC الذي بدأ من خلال استدعاءات OSSL_FUNC_mac_init() و OSSL_FUNC_mac_update() السابقة. تحتوي المعلمة mctx على مؤشر إلى السياق جانب المزود. يجب كتابة MAC الناتج إلى out وحجم البيانات المكتوبة إلى *outl، والتي يجب ألا تتجاوز outsize بايت. تنطبق نفس التوقعات على outsize كما هو موثق في EVP_MAC_final(3).

معلمات Mac

انظر OSSL_PARAM(3) لمزيد من التفاصيل حول هيكل المعلمات المستخدم بواسطة هذه الوظائف.

تقوم OSSL_FUNC_mac_get_params() بالحصول على تفاصيل قيم المعلمات المرتبطة بخوارزمية المزود وتخزينها في params.

تقوم OSSL_FUNC_mac_set_ctx_params() بتعيين معلمات mac المرتبطة بسياق mac المحدد جانب المزود mctx إلى params. أي إعدادات للمعلمات تكون إضافية لأي شيء تم تعيينه مسبقًا. تمرير NULL لـ params يجب أن يعود بنتيجة صحيحة.

تقوم OSSL_FUNC_mac_get_ctx_params() بالحصول على تفاصيل قيم المعلمات الحالية المعينة المرتبطة بسياق mac المحدد جانب المزود mctx وتخزينها في params. تمرير NULL لـ params يجب أن يعود بنتيجة صحيحة.

جميع وظائف OSSL_FUNC_mac_gettable_params()، OSSL_FUNC_mac_gettable_ctx_params()، و OSSL_FUNC_mac_settable_ctx_params() تقوم بإرجاع مصفوفات OSSL_PARAM(3) ثابتة كواصفات للمعلمات التي يمكن أن تتعامل معها OSSL_FUNC_mac_get_params()، OSSL_FUNC_mac_get_ctx_params()، و OSSL_FUNC_mac_set_ctx_params()، على التوالي. OSSL_FUNC_mac_gettable_ctx_params() و OSSL_FUNC_mac_settable_ctx_params() ستقوم بإرجاع المعلمات المرتبطة بسياق المزود mctx في حالته الحالية إذا لم يكن NULL. خلاف ذلك، ستقوم بإرجاع المعلمات المرتبطة بخوارزمية المزود provctx.

من المتوقع أن تتعامل جميع تطبيقات MAC مع المعلمات التالية:

مع OSSL_FUNC_set_ctx_params():
"key" (OSSL_MAC_PARAM_KEY) <سلسلة بايت>
تعيين المفتاح في سياق MAC المرتبط. هذا مطابق لتمرير وسيطة key إلى الوظيفة OSSL_FUNC_mac_init().
مع OSSL_FUNC_get_params():
"size" (OSSL_MAC_PARAM_SIZE) <عدد صحيح>
يمكن استخدامه للحصول على حجم MAC الافتراضي (والذي قد يكون الحجم الوحيد المسموح به للخوارزمية).
