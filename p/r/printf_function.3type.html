الوصف

تخدم هذه الوظائف لتوسيع و/أو تعديل سلوك عائلة وظائف printf(3).

register_printf_specifier()

تسجل هذه الوظيفة محدد تحويل مخصص لعائلة وظائف printf(3).

spec
الحرف الذي سيتم استخدامه كمحدد تحويل في سلسلة التنسيق.
func
وظيفة الاستدعاء التي سيتم تنفيذها بواسطة عائلة وظائف printf(3) لتنسيق الوسائط المدخلة إلى تدفق الإخراج stream.
stream
تدفق الإخراج حيث يجب طباعة الإخراج المنسق. يمثل هذا التدفق الإخراج بشفافية، حتى في حالة الوظائف التي تكتب إلى سلسلة.
info
بنية تحتوي على معلومات السياق، بما في ذلك المعدلات المحددة في سلسلة التنسيق. تحتوي هذه على نفس محتويات تلك الموجودة في arginfo.
args
مصفوفة من المؤشرات إلى الوسائط الخاصة بوظيفة شبيهة بـ printf(3).
arginfo
وظيفة الاستدعاء التي سيتم تنفيذها بواسطة عائلة وظائف printf(3) لمعرفة عدد الوسائط التي يجب تحليلها للمحدد المخصص وأنواعها أيضًا.
info
بنية تحتوي على معلومات السياق، بما في ذلك المعدلات المحددة في سلسلة التنسيق. تحتوي هذه على نفس محتويات تلك الموجودة في func.
n
عدد الوسائط المتبقية للتحليل.
argtypes
يجب تعيين هذه المصفوفة لتحديد نوع كل وسيط سيتم تحليله. يمثل كل عنصر في المصفوفة وسيطًا واحدًا سيتم تحليله، بنفس الترتيب الذي يتم به تمريره إلى وظيفة شبيهة بـ printf(3). يجب تعيين كل عنصر إلى نوع أساسي (PA_*) من التعداد أعلاه، أو نوع مخصص، وربما يتم دمجه OR مع معدل طول مناسب (PA_FLAG_*).

يتم تحديد النوع باستخدام أحد الثوابت التالية:

PA_INT
int.
PA_CHAR
int، تم تحويله إلى char.
PA_WCHAR
wchar_t.
PA_STRING
const char *، سلسلة منتهية بـ '\0'.
PA_WSTRING
const wchar_t *، سلسلة أحرف واسعة منتهية بـ L'\0'.
PA_POINTER
void *.
PA_FLOAT
float.
PA_DOUBLE
double.
PA_LAST
TODO.
size
بالنسبة للأنواع المعرفة من قبل المستخدم، يجب أيضًا تحديد حجم النوع (بالبايت) من خلال هذه المصفوفة. بخلاف ذلك، اتركها غير مستخدمة.

يتم استدعاء arginfo قبل func، ويقوم بإعداد بعض المعلومات اللازمة لاستدعاء func.

قيمة الإرجاع

register_printf_specifier() و register_printf_modifier() و register_printf_type() ترجع صفر عند النجاح، أو -1 عند حدوث خطأ.

استدعاءات

يجب أن ترجع استدعاءات النوع printf_function عدد الأحرف المكتوبة، أو -1 عند حدوث خطأ.

يجب أن ترجع استدعاءات النوع printf_arginfo_size_function عدد الوسائط التي سيتم تحليلها بواسطة هذا المحدد. كما أنها تمرر معلومات حول نوع تلك الوسائط إلى المستدعي من خلال argtypes. عند حدوث خطأ، يجب أن ترجع -1.

التاريخ

register_printf_function(3) هي دالة أقدم تشبه register_printf_specifier()، وهي الآن مهملة. لا يمكن لهذه الدالة التعامل مع أنواع معرفة من قبل المستخدم.

register_printf_specifier() تحل محل register_printf_function(3).

أمثلة

static int
b_printf(FILE *stream, const struct printf_info *info,
         const void *const args[])
{
    char               bin[UINTMAX_WIDTH];
    size_t             min_len, bin_len;
    ssize_t            len, tmp;
    struct Printf_Pad  pad = {0};
    len = 0;
    min_len = b_bin_repr(bin, info, args[0]);
    bin_len = b_bin_len(info, min_len);
    pad.ch = pad_ch(info);
    if (pad.ch == ' ')
        pad.len = b_pad_len(info, bin_len);
    /* Padding with ' ' (right aligned) */
    if ((pad.ch == ' ') && !info->left) {
        tmp = pad_spaces(stream, pad.len);
        if (tmp == EOF)
            return EOF;
        len += tmp;
    }
    /* "0b"/"0B" prefix */
    if (info->alt) {
        tmp = b_print_prefix(stream, info);
        if (tmp == EOF)
            return EOF;
        len += tmp;
    }
    /* Padding with '0' */
    if (pad.ch == '0') {
        tmp = b_pad_zeros(stream, info, min_len);
        if (tmp == EOF)
            return EOF;
        len += tmp;
    }
    /* Print number (including leading 0s to fill precision) */
    tmp = b_print_number(stream, info, bin, min_len, bin_len);
    if (tmp == EOF)
        return EOF;
    len += tmp;
    /* Padding with ' ' (left aligned) */
    if (info->left) {
        tmp = pad_spaces(stream, pad.len);
        if (tmp == EOF)
            return EOF;
        len += tmp;
    }
    return len;
}
static int
b_arginf_sz(const struct printf_info *info, size_t n, int argtypes[n],
            [[maybe_unused]] int size[n])
{
    if (n < 1)
        return -1;
    if (info->is_long_double)
        argtypes[0] = PA_INT | PA_FLAG_LONG_LONG;
    else if (info->is_long)
        argtypes[0] = PA_INT | PA_FLAG_LONG;
    else
        argtypes[0] = PA_INT;
    return 1;
}
static uintmax_t
b_value(const struct printf_info *info, const void *arg)
{
    if (info->is_long_double)
        return *(const unsigned long long *)arg;
    if (info->is_long)
        return *(const unsigned long *)arg;
    /* short and char are both promoted to int */
    return *(const unsigned int *)arg;
}
static size_t
b_bin_repr(char bin[UINTMAX_WIDTH],
           const struct printf_info *info, const void *arg)
{
    size_t     min_len;
    uintmax_t  val;
    val = b_value(info, arg);
    bin[0] = '0';
    for (min_len = 0; val; min_len++) {
        bin[min_len] = '0' + (val % 2);
        val >>= 1;
    }
    return MAX(min_len, 1);
}
static size_t
b_bin_len(const struct printf_info *info, ptrdiff_t min_len)
{
    return MAX(info->prec, min_len);
}
static size_t
b_pad_len(const struct printf_info *info, ptrdiff_t bin_len)
{
    ptrdiff_t  pad_len;
    pad_len = info->width - bin_len;
    if (info->alt)
        pad_len -= 2;
    if (info->group)
        pad_len -= (bin_len - 1) / 4;
    return MAX(pad_len, 0);
}
static ssize_t
b_print_prefix(FILE *stream, const struct printf_info *info)
{
    ssize_t len;
    len = 0;
    if (fputc('0', stream) == EOF)
        return EOF;
    len++;
    if (fputc(info->spec, stream) == EOF)
        return EOF;
    len++;
    return len;
}
static ssize_t
b_pad_zeros(FILE *stream, const struct printf_info *info,
            ptrdiff_t min_len)
{
    ssize_t    len;
    ptrdiff_t  tmp;
    len = 0;
    tmp = info->width - (info->alt * 2);
    if (info->group)
        tmp -= tmp / 5 - !(tmp % 5);
    for (ptrdiff_t i = tmp - 1; i > min_len - 1; i--) {
        if (fputc('0', stream) == EOF)
            return EOF;
        len++;
        if (!info->group || (i % 4))
            continue;
        if (fputc(GROUP_SEP, stream) == EOF)
            return EOF;
        len++;
    }
    return len;
}
static ssize_t
b_print_number(FILE *stream, const struct printf_info *info,
               const char bin[UINTMAX_WIDTH],
               size_t min_len, size_t bin_len)
{
    ssize_t  len;
    len = 0;
    /* Print leading zeros to fill precision */
    for (size_t i = bin_len - 1; i > min_len - 1; i--) {
        if (fputc('0', stream) == EOF)
            return EOF;
        len++;
        if (!info->group || (i % 4))
            continue;
        if (fputc(GROUP_SEP, stream) == EOF)
            return EOF;
        len++;
    }
    /* Print number */
    for (size_t i = min_len - 1; i < min_len; i--) {
        if (fputc(bin[i], stream) == EOF)
            return EOF;
        len++;
        if (!info->group || (i % 4) || !i)
            continue;
        if (fputc(GROUP_SEP, stream) == EOF)
            return EOF;
        len++;
    }
    return len;
}
static char
pad_ch(const struct printf_info *info)
{
    if ((info->prec != -1) || (info->pad == ' ') || info->left)
        return ' ';
    return '0';
}
static ssize_t
pad_spaces(FILE *stream, size_t pad_len)
{
    ssize_t  len;
    len = 0;
    for (size_t i = pad_len - 1; i < pad_len; i--) {
        if (fputc(' ', stream) == EOF)
            return EOF;
        len++;
    }
    return len;
}
