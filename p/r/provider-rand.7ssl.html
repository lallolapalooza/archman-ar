#include <openssl/core_dispatch.h>
#include <openssl/core_names.h>
/*
 * لا توجد أي من هذه الدوال في الواقع، ولكنها معروضة على هذا النحو
 * لتوقيعات الدوال التي يتم تقديمها كمؤشرات دالة في مصفوفات OSSL_DISPATCH.
 */
/* وظائف إدارة السياق */
void *OSSL_FUNC_rand_newctx(void *provctx, void *parent,
                            const OSSL_DISPATCH *parent_calls);
void OSSL_FUNC_rand_freectx(void *ctx);
/* وظائف مولد الأرقام العشوائية: NIST */
int OSSL_FUNC_rand_instantiate(void *ctx, unsigned int strength,
                               int prediction_resistance,
                               const unsigned char *pstr, size_t pstr_len,
                               const OSSL_PARAM params[]);
int OSSL_FUNC_rand_uninstantiate(void *ctx);
int OSSL_FUNC_rand_generate(void *ctx, unsigned char *out, size_t outlen,
                            unsigned int strength, int prediction_resistance,
                            const unsigned char *addin, size_t addin_len);
int OSSL_FUNC_rand_reseed(void *ctx, int prediction_resistance,
                          const unsigned char *ent, size_t ent_len,
                          const unsigned char *addin, size_t addin_len);
/* وظائف مولد الأرقام العشوائية: إضافية */
size_t OSSL_FUNC_rand_nonce(void *ctx, unsigned char *out, size_t outlen,
                            int strength, size_t min_noncelen,
                            size_t max_noncelen);
size_t OSSL_FUNC_rand_get_seed(void *ctx, unsigned char **buffer,
                               int entropy, size_t min_len, size_t max_len,
                               int prediction_resistance,
                               const unsigned char *adin, size_t adin_len);
void OSSL_FUNC_rand_clear_seed(void *ctx, unsigned char *buffer, size_t b_len);
int OSSL_FUNC_rand_verify_zeroization(void *ctx);
/* قفل السياق */
int OSSL_FUNC_rand_enable_locking(void *ctx);
int OSSL_FUNC_rand_lock(void *ctx);
void OSSL_FUNC_rand_unlock(void *ctx);
/* واصفات معلمات RAND */
const OSSL_PARAM *OSSL_FUNC_rand_gettable_params(void *provctx);
const OSSL_PARAM *OSSL_FUNC_rand_gettable_ctx_params(void *ctx, void *provctx);
const OSSL_PARAM *OSSL_FUNC_rand_settable_ctx_params(void *ctx, void *provctx);
/* معلمات RAND */
int OSSL_FUNC_rand_get_params(OSSL_PARAM params[]);
int OSSL_FUNC_rand_get_ctx_params(void *ctx, OSSL_PARAM params[]);
int OSSL_FUNC_rand_set_ctx_params(void *ctx, const OSSL_PARAM params[]);

الوصف

هذه الوثائق موجهة بشكل أساسي لمؤلفي المزودين. انظر provider(7) لمزيد من المعلومات.

تتيح عملية RAND للمزودين تنفيذ خوارزميات إنشاء الأرقام العشوائية ومصادر الأرقام العشوائية وجعلها متاحة للتطبيقات عبر وظيفة API EVP_RAND(3).

وظائف إدارة السياق

يجب على OSSL_FUNC_rand_newctx() إنشاء وإرجاع مؤشر إلى هيكل جانب المزود للاحتفاظ بمعلومات السياق أثناء عملية عشوائية. سيتم تمرير مؤشر إلى هذا السياق مرة أخرى في عدد من استدعاءات وظائف عملية العشوائية الأخرى. المعلمة provctx هي سياق المزود الذي تم إنشاؤه أثناء تهيئة المزود (انظر provider(7)). تحدد المعلمة parent مثيل عشوائي آخر ليتم استخدامه لأغراض البذر. إذا كان NULL وكانت المثيل المحدد يدعم ذلك، فسيتم استخدام نظام التشغيل للبذر. تشير المعلمة parent_calls إلى جدول الإرسال لـ parent. وبالتالي، لا يلزم أن يكون الوالد من نفس المزود مثل المثيل الجديد.

يتم تمرير OSSL_FUNC_rand_freectx() مؤشرًا إلى سياق العشوائي جانب المزود في المعلمة mctx. إذا استلمت NULL كقيمة ctx، فيجب ألا تفعل شيئًا سوى العودة. يجب أن تحرر هذه الوظيفة أي موارد مرتبطة بهذا السياق.

وظائف مولد الأرقام العشوائية: NIST

تقابل هذه الوظائف تلك المحددة في NIST SP 800-90A و SP 800-90C.

يتم استخدام OSSL_FUNC_rand_instantiate() لاستنساط DRBG ctx عند strength أمان مطلوب. بالإضافة إلى ذلك، يمكن طلب prediction_resistance. يمكن اختياريًا توفير مدخلات إضافية addin بطول addin_len بايت. يجب معالجة المعلمات المحددة في params قبل الاستنساخ.

يتم استخدام OSSL_FUNC_rand_uninstantiate() لإلغاء استنساخ DRBG ctx. بعد إلغاء استنساخه، يكون DRBG غير قادر على إنتاج مخرجات حتى يتم استنساخه من جديد.

يتم استخدام OSSL_FUNC_rand_generate() لإنشاء بايتات عشوائية من DRBG ctx. سيقوم بإنشاء outlen بايت ووضعها في المخزن المؤقت الذي يشير إليه out. ستحقق البايتات التي تم إنشاؤها strength الأمان المحدد، وإذا كان prediction_resistance صحيحًا، فسيتم إنتاج البايتات بعد إعادة البذر من مصدر إنتروبيا مباشر. يمكن اختياريًا توفير مدخلات إضافية addin بطول addin_len بايت.

وظائف مولد الأرقام العشوائية: إضافية

يتم استخدام OSSL_FUNC_rand_nonce() لإنشاء nonce بالقوة strength المعطاة بطول يتراوح من min_noncelen إلى max_noncelen. إذا كان المخزن المؤقت للإخراج out NULL، فيجب إرجاع طول الـ nonce.

يتم استخدام OSSL_FUNC_rand_get_seed() بواسطة المولدات الحتمية للحصول على مواد البذر الخاصة بها من والديها. ستحقق بايتات البذر مستوى الأمان المحدد البالغ entropy بت وسيكون هناك ما بين min_len و max_len بايت إجمالاً. إذا كان prediction_resistance صحيحًا، فسيتم إنتاج البايتات من مصدر إنتروبيا مباشر. يمكن اختياريًا توفير مدخلات إضافية addin بطول addin_len بايت. يتم إرجاع مؤشر إلى مادة البذر في *buffer ويجب تحريره بواسطة استدعاء لاحق لـ OSSL_FUNC_rand_clear_seed().

يقوم OSSL_FUNC_rand_clear_seed() بتحرير المخزن المؤقت للبذر buffer بطول b_len بايت تم تخصيصه سابقًا بواسطة OSSL_FUNC_rand_get_seed().

يتم استخدام OSSL_FUNC_rand_verify_zeroization() لتحديد ما إذا كانت الحالة الداخلية لـ DRBG هي صفر. هذه القدرة مطلوبة بموجب NIST كجزء من الاختبارات الذاتية، ومن غير المرجح أن تكون مفيدة في ظروف أخرى.

قفل السياق

عندما يتم استخدام DRBGs بواسطة خيوط متعددة، يجب توظيف قفل لضمان تشغيلها السليم. نظرًا لأن القفل يمثل عبئًا إضافيًا، فهو معطل افتراضيًا.

يسمح OSSL_FUNC_rand_enable_locking() بتشغيل القفل لـ DRBG وجميع DRBGs الأصلية الخاصة به. من هذا الاستدعاء فصاعدًا، يمكن استخدام DRBG بطريقة آمنة للخيوط.

يتم استخدام OSSL_FUNC_rand_lock() لقفل DRBG. بمجرد القفل، يتم ضمان الوصول الحصري.

يتم استخدام OSSL_FUNC_rand_unlock() لفتح DRBG.

معلمات العشوائية

انظر OSSL_PARAM(3) لمزيد من التفاصيل حول بنية المعلمات المستخدمة بواسطة هذه الوظائف.

تقوم OSSL_FUNC_rand_get_params() بالحصول على تفاصيل قيم المعلمات المرتبطة بخوارزمية المزود وتخزينها في params.

تقوم OSSL_FUNC_rand_set_ctx_params() بتعيين معلمات عشوائية مرتبطة بسياق العشوائي جانب المزود المحدد ctx إلى params. أي إعدادات معلمات إضافية لأي شيء تم تعيينه سابقًا. يجب أن ترجع القيمة true عند تمرير NULL لـ params.

تقوم OSSL_FUNC_rand_get_ctx_params() بالحصول على تفاصيل قيم المعلمات المعينة حاليًا المرتبطة بسياق العشوائي جانب المزود المحدد ctx وتخزينها في params. يجب أن ترجع القيمة true عند تمرير NULL لـ params.

تقوم OSSL_FUNC_rand_gettable_params() و OSSL_FUNC_rand_gettable_ctx_params() و OSSL_FUNC_rand_settable_ctx_params() جميعها بإرجاع مصفوفات OSSL_PARAM(3) ثابتة كواصفات للمعلمات التي يمكن لـ OSSL_FUNC_rand_get_params() و OSSL_FUNC_rand_get_ctx_params() و OSSL_FUNC_rand_set_ctx_params() التعامل معها، على التوالي. OSSL_FUNC_rand_gettable_ctx_params() و OSSL_FUNC_rand

NAME

provider-rand - OpenSSL provider random number generator interface

SYNOPSIS

This manual page describes the RAND interface for OpenSSL providers.

The functions, constants and types described here are intended to be used by provider implementors, not by applications.

A provider implements the RAND interface by providing implementations of the functions listed in the table under DESCRIPTION. Each function is identified by a unique integer constant, as shown in the table.

The implementation of the RAND interface is done via the OSSL_provider_get_capabilities() function, which should return information about the RAND capabilities that the provider supports. The information includes the names of the algorithms supported, and the corresponding function pointers for each operation.

The structure of the information returned by OSSL_provider_get_capabilities() for RAND is as follows:

const OSSL_PARAM *OSSL_provider_get_capabilities(void *provctx,
                                                 const char *cap,
                                                 const char *name);

If cap is "RAND", then name can be NULL or the name of a specific RAND algorithm (e.g. "default", "openssl"). The returned OSSL_PARAM array contains information about the algorithm(s).

The following table lists the RAND operation identifiers and their associated functions:

Operation
Function Signature
OSSL_RAND_GETTABLE_PARAMS
const OSSL_PARAM *(*gettable_params)(void *provctx);
OSSL_RAND_GETTABLE_CTX_PARAMS
const OSSL_PARAM *(*gettable_ctx_params)(void *provctx);
OSSL_RAND_GET_params
int (*get_params)(void *provctx, OSSL_PARAM params[]);
OSSL_RAND_SETTABLE_CTX_PARAMS
const OSSL_PARAM *(*settable_ctx_params)(void *provctx);
OSSL_RAND_SET_CTX_params
int (*set_ctx_params)(void *rand, OSSL_PARAM params[]);
OSSL_RAND_NEWCTX
void *(*newctx)(void *provctx, const char *name, const OSSL_DISPATCH *parent_rtbl);
OSSL_RAND_FREECTX
void (*freectx)(void *rand);
OSSL_RAND_GENERATE
int (*generate)(void *rand, unsigned char *buf, size_t len, int strength, int predict, void *caller_ctx);
OSSL_RAND_NONCE
int (*nonce)(void *rand, unsigned char *buf, size_t len);
OSSL_RAND_ENABLE_CALLBK
int (*enable_callbk)(void *rand);
OSSL_RAND_SECURITY_BITS
int (*security_bits)(void *rand);
OSSL_RAND_GET_SEED
int (*get_seed)(void *rand, unsigned char **pbuf, int entropy, size_t max_len);

DESCRIPTION

The OpenSSL RAND interface provides a way for applications to obtain random bytes. Providers can implement this interface to offer different random number generators (RNGs).

The RAND interface is based on the provider function OSSL_provider_get_capabilities(). A provider registers its RAND capabilities by returning an OSSL_PARAM(3) array from its implementation of OSSL_provider_get_capabilities() when the capability name is "RAND".

Each entry in the returned array describes a specific RAND algorithm supported by the provider. The entry must contain at least the following parameters:

"name" <UTF8 string>
The name of the RAND algorithm. This is the name that applications will use to refer to this RAND algorithm.
"implementation" <UTF8 string>
The name of the implementation. This is typically the name of the provider that provides this RAND algorithm.
"version" <unsigned integer>
The version number of the implementation.
"gettable_params" <UTF8 string>
A NULL-terminated list of parameter names that can be obtained from the provider-level RAND context using OSSL_FUNC_rand_gettable_params().
"settable_ctx_params" <UTF8 string>
A NULL-terminated list of parameter names that can be set for the provider-side RAND context using OSSL_FUNC_rand_settable_ctx_params().
"gettable_ctx_params" <UTF8 string>
A NULL-terminated list of parameter names that can be obtained from the provider-side RAND context using OSSL_FUNC_rand_gettable_ctx_params().
"callbacks" <UTF8 string>
A NULL-terminated list of function pointers for the RAND operations. Each function pointer is preceded by its operation identifier.

The available RAND operations and their corresponding function pointers are:

OSSL_RAND_GETTABLE_PARAMS
Returns a constant OSSL_PARAM(3) array describing the parameters that can be obtained from the provider context.
OSSL_RAND_GETTABLE_CTX_PARAMS
Returns a constant OSSL_PARAM(3) array describing the parameters that can be obtained from the RAND context.
OSSL_RAND_GET_params
Retrieves parameters from the provider context. The parameter names that can be retrieved are described by the array returned from OSSL_RAND_GETTABLE_PARAMS.
OSSL_RAND_SETTABLE_CTX_PARAMS
Returns a constant OSSL_PARAM(3) array describing the parameters that can be set for the RAND context.
OSSL_RAND_SET_CTX_PARAMS
Sets parameters for the RAND context. The parameter names that can be set are described by the array returned from OSSL_RAND_SETTABLE_CTX_PARAMS.
OSSL_RAND_NEWCTX
Creates a new RAND context for the specified algorithm. provctx is the provider context, name is the name of the algorithm (e.g. "default"), and parent_rtbl is a dispatch table of functions that the provider may need from the core OpenSSL library.
OSSL_RAND_FREECTX
Frees the RAND context.
OSSL_RAND_GENERATE
Generates cryptographically strong pseudo-random bytes. rand is the RAND context, buf is the buffer to fill with random bytes, len is the number of bytes to generate, strength is the security strength required (in bits), and predict is a flag indicating whether the output should be unpredictable (1) or just pseudo-random (0). caller_ctx is an optional context pointer from the caller.
OSSL_RAND_NONCE
Generates a nonce (number used once). rand is the RAND context, buf is the buffer to fill with the nonce, and len is the desired length of the nonce.
OSSL_RAND_ENABLE_CALLBK
Enables callbacks for the RAND context. This may be used internally by OpenSSL.
OSSL_RAND_SECURITY_BITS
Returns the security strength of the RAND algorithm in bits.
OSSL_RAND_GET_SEED
Retrieves seed material for the RAND algorithm. rand is the RAND context, pbuf is a pointer to a buffer that will be filled with seed material, entropy is the amount of entropy requested (in bits), and max_len is the maximum number of bytes to retrieve.

The function OSSL_FUNC_rand_settable_ctx_params() will return the parameters that can be set for the provider side context ctx. If ctx is NULL, it returns the parameters that can be set for the provider side algorithm provctx.

The function OSSL_FUNC_rand_gettable_ctx_params() will return the parameters associated with the provider side context ctx in its current state if it is not NULL. Otherwise, they return the parameters associated with the provider side algorithm provctx.

Parameters currently recognised by built-in rands are as follows. Not all parameters are relevant to, or are understood by all rands:

"state" (OSSL_RAND_PARAM_STATE) <integer>
Returns the state of the random number generator.
"strength" (OSSL_RAND_PARAM_STRENGTH) <unsigned integer>
Returns the bit strength of the random number generator.
"fips-indicator" (OSSL_RAND_PARAM_FIPS_APPROVED_INDICATOR) <integer>
A getter that returns 1 if the operation is FIPS approved, or 0 otherwise. This option is used by the OpenSSL FIPS provider and is not supported by all EVP_RAND sources.

For rands that are also deterministic random bit generators (DRBGs), these additional parameters are recognised. Not all parameters are relevant to, or are understood by all DRBG rands:

"reseed_requests" (OSSL_DRBG_PARAM_RESEED_REQUESTS) <unsigned integer>
Reads or set the number of generate requests before reseeding the associated RAND ctx.
"reseed_time_interval" (OSSL_DRBG_PARAM_RESEED_TIME_INTERVAL) <integer>
Reads or set the number of elapsed seconds before reseeding the associated RAND ctx.
"max_request" (OSSL_DRBG_PARAM_RESEED_REQUESTS) <unsigned integer>
Specifies the maximum number of bytes that can be generated in a single call to OSSL_FUNC_rand_generate.
"min_entropylen" (OSSL_DRBG_PARAM_MIN_ENTROPYLEN) <unsigned integer>
"max_entropylen" (OSSL_DRBG_PARAM_MAX_ENTROPYLEN) <unsigned integer>
Specify the minimum and maximum number of bytes of random material that can be used to seed the DRBG.
"min_noncelen" (OSSL_DRBG_PARAM_MIN_NONCELEN) <unsigned integer>
"max_noncelen" (OSSL_DRBG_PARAM_MAX_NONCELEN) <unsigned integer>
Specify the minimum and maximum number of bytes of nonce that can be used to instantiate the DRBG.
"max_perslen" (OSSL_DRBG_PARAM_MAX_PERSLEN) <unsigned integer>
"max_adinlen" (OSSL_DRBG_PARAM_MAX_ADINLEN) <unsigned integer>
Specify the minimum and maximum number of bytes of personalisation string that can be used with the DRBG.
"reseed_counter" (OSSL_DRBG_PARAM_RESEED_COUNTER) <unsigned integer>
Specifies the number of times the DRBG has been seeded or reseeded.
"digest" (OSSL_DRBG_PARAM_DIGEST) <UTF8 string>
"cipher" (OSSL_DRBG_PARAM_CIPHER) <UTF8 string>
"mac" (OSSL_DRBG_PARAM_MAC) <UTF8 string>
Sets the name of the underlying cipher, digest or MAC to be used. It must name a suitable algorithm for the DRBG that's being used.
"properties" (OSSL_DRBG_PARAM_PROPERTIES) <UTF8 string>
Sets the properties to be queried when trying to fetch an underlying algorithm. This must be given together with the algorithm naming parameter to be considered valid.

The OpenSSL FIPS provider also supports the following parameters:

"fips-indicator" (OSSL_DRBG_PARAM_FIPS_APPROVED_INDICATOR) <integer>
A getter that returns 1 if the operation is FIPS approved, or 0 otherwise. This may be used after calling OSSL_FUNC_rand_generate(). It may return 0 if the "digest-check" is set to 0.
"digest-check" (OSSL_DRBG_PARAM_FIPS_DIGEST_CHECK) <integer>
If required this parameter should be set before the digest is set. The default value of 1 causes an error when the digest is set if the digest is not FIPS approved (e.g. truncated digests). Setting this to 0 will ignore the error and set the approved "fips-indicator" to 0. This option breaks FIPS compliance if it causes the approved "fips-indicator" to return 0.

RETURN VALUES

OSSL_FUNC_rand_newctx() should return the newly created provider side rand context, or NULL on failure.

OSSL_FUNC_rand_gettable_params(), OSSL_FUNC_rand_gettable_ctx_params() and OSSL_FUNC_rand_settable_ctx_params() should return a constant OSSL_PARAM(3) array, or NULL if none is offered.

OSSL_FUNC_rand_nonce() returns the size of the generated nonce, or 0 on error.

OSSL_FUNC_rand_get_seed() returns the size of the generated seed, or 0 on error.

All of the remaining functions should return 1 for success or 0 on error.

NOTES

The RAND life-cycle is described in life_cycle-rand(7). Providers should ensure that the various transitions listed there are supported. At some point the EVP layer will begin enforcing the listed transitions.

HISTORY

The provider RAND interface was introduced in OpenSSL 3.0. The Rand Parameters "fips-indicator" and "digest-check" were added in OpenSSL 3.4.
