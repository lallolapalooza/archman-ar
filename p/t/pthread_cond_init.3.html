o be isيقظ (على سبيل المثال، إذا كان هناك خيطان فقط يتواصلان عبر x و y)، يمكن استخدام pthread_cond_signal() كبديل أكثر كفاءة بقليل من pthread_cond_broadcast(). في حالة الشك، استخدم pthread_cond_broadcast().

للانتظار حتى يصبح x أكبر من y مع مهلة قدرها 5 ثوانٍ، قم بما يلي:

struct timeval now;
struct timespec timeout;
int retcode;
pthread_mutex_lock(&mut);
gettimeofday(&now);
timeout.tv_sec = now.tv_sec + 5;
timeout.tv_nsec = now.tv_usec * 1000;
retcode = 0;
while (x <= y && retcode != ETIMEDOUT) {
        retcode = pthread_cond_timedwait(&cond, &mut, &timeout);
}
if (retcode == ETIMEDOUT) {
        /* حدث انتهاء المهلة */
} else {
        /* اعمل على x و y */
}
pthread_mutex_unlock(&mut);
