subroutine cptcon (integer n, real, dimension( * ) d, complex, dimension( * ) e, real anorm, real rcond, real, dimension( * ) rwork, integer info)

CPTCON

الغرض:

!>
!> CPTCON تحسب مقلوب رقم الشرط (بالقاعدة 1-norm) لمصفوفة مركبة متماثلة موجبة محددة ثلاثية الأقطار
!> باستخدام التحليل A = L*D*L**H أو A = U**H*D*U المحسوب بواسطة CPTTRF.
!>
!> يتم حساب Norm(inv(A)) بطريقة مباشرة، ويتم حساب مقلوب رقم الشرط كـ
!>                  RCOND = 1 / (ANORM * norm(inv(A))).
!>

المعلمات

N 
!>          N هو عدد صحيح (INTEGER)
!>          رتبة المصفوفة A.  N >= 0.
!>

D

!>          D هو مصفوفة حقيقية (REAL array)، بالبعد (N)
!>          العناصر القطرية n للعناصر القطرية للمصفوفة D من
!>          تحليل A، كما تم حسابه بواسطة CPTTRF.
!>

E

!>          E هو مصفوفة مركبة (COMPLEX array)، بالبعد (N-1)
!>          العناصر غير القطرية n-1 للعناصر غير القطرية للـ
!>          عامل ثنائي القطر الأحادي U أو L من تحليل A، كما تم حسابه بواسطة CPTTRF.
!>

ANORM

!>          ANORM هو عدد حقيقي (REAL)
!>          القاعدة 1-norm للمصفوفة الأصلية A.
!>

RCOND

!>          RCOND هو عدد حقيقي (REAL)
!>          مقلوب رقم الشرط للمصفوفة A،
!>          يتم حسابه كـ RCOND = 1/(ANORM * AINVNM)، حيث AINVNM هو
!>          القاعدة 1-norm للمصفوفة المعكوسة inv(A) التي تم حسابها في هذا الروتين.
!>

RWORK

!>          RWORK هو مصفوفة حقيقية (REAL array)، بالبعد (N)
!>

INFO

!>          INFO هو عدد صحيح (INTEGER)
!>          = 0:  خروج ناجح
!>          < 0:  إذا كانت INFO = -i، فإن المعامل i كان له قيمة غير قانونية
!>
