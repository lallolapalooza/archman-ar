pthread_key_create(3) دليل وظائف المكتبة pthread_key_create(3)

الاسم

pthread_key_create, pthread_key_delete, pthread_setspecific, pthread_getspecific - إدارة البيانات الخاصة بالخيط

الملخص

#include <pthread.h>
int pthread_key_create(pthread_key_t *key,
                       typeof(void (void *)) *destr_function;
int pthread_key_delete(pthread_key_t key);
int pthread_setspecific(pthread_key_t key, const void *pointer);
void * pthread_getspecific(pthread_key_t key);

الوصف

تحتاج البرامج غالبًا إلى متغيرات عامة أو ثابتة لها قيم مختلفة في خيوط مختلفة. نظرًا لأن الخيوط تتشارك مساحة ذاكرة واحدة، لا يمكن تحقيق ذلك باستخدام المتغيرات العادية. البيانات الخاصة بالخيط هي إجابة POSIX threads لهذه الحاجة.

يمتلك كل خيط كتلة ذاكرة خاصة به، تسمى منطقة البيانات الخاصة بالخيط، أو منطقة TSD للاختصار. يتم فهرسة هذه المنطقة بواسطة مفاتيح TSD. تربط منطقة TSD قيمًا من نوع void * بمفاتيح TSD. مفاتيح TSD مشتركة بين جميع الخيوط، ولكن القيمة المرتبطة بمفتاح TSD معين يمكن أن تكون مختلفة في كل خيط.

للتبسيط، يمكن اعتبار مناطق TSD كمصفوفات من مؤشرات void *، ومفاتيح TSD كفهارس عددية في هذه المصفوفات، وقيمة مفتاح TSD كقيمة العنصر المقابل في المصفوفة في الخيط المستدعي.

عند إنشاء خيط، تقوم منطقة TSD الخاصة به في البداية بربط NULL بجميع المفاتيح.

تقوم pthread_key_create() بتخصيص مفتاح TSD جديد. يتم تخزين المفتاح في الموقع الذي يشير إليه key. يوجد حد أقصى لعدد المفاتيح المخصصة في وقت معين وهو PTHREAD_KEYS_MAX. القيمة المرتبطة مبدئيًا بالمفتاح الذي تم إرجاعه هي NULL في جميع الخيوط التي يتم تنفيذها حاليًا.

الوسيط destr_function، إذا لم يكن NULL، يحدد دالة مدمرة مرتبطة بالمفتاح. عندما ينتهي خيط عبر pthread_exit() أو عبر الإلغاء، يتم استدعاء destr_function مع الوسيطات، وهي القيمة المرتبطة بالمفتاح في ذلك الخيط. لا يتم استدعاء destr_function إذا كانت تلك القيمة NULL. ترتيب استدعاء الدوال المدمرة عند إنهاء الخيط غير محدد.

قبل استدعاء الدالة المدمرة، يتم ربط قيمة NULL بالمفتاح في الخيط الحالي. ومع ذلك، قد تعيد دالة المدمرة ربط قيم غير NULL بهذا المفتاح أو بمفتاح آخر. للتعامل مع هذا، إذا ظلت هناك قيم غير NULL مرتبطة بدوال مدمرة بعد استدعاء جميع الدوال المدمرة لجميع القيم غير NULL، فسيتم تكرار العملية. يتوقف تنفيذ glibc عن العملية بعد PTHREAD_DESTRUCTOR_ITERATIONS تكرار، حتى لو بقيت بعض القيم غير NULL مع معرّفات مرتبطة. قد تستمر التطبيقات الأخرى في حلقة لا نهائية.

تقوم pthread_key_delete() بإلغاء تخصيص مفتاح TSD. لا تتحقق مما إذا كانت هناك قيم غير NULL مرتبطة بهذا المفتاح في الخيوط التي يتم تنفيذها حاليًا، ولا تستدعي الدالة المدمرة المرتبطة بالمفتاح.

تغير pthread_setspecific() القيمة المرتبطة بـ key في الخيط المستدعي، وتخزين pointer المقدم بدلاً منها.

تُرجع pthread_getspecific() القيمة المرتبطة حاليًا بـ key في الخيط المستدعي.

قيمة الإرجاع

pthread_key_create() و pthread_key_delete() و pthread_setspecific() ترجع 0 عند النجاح ورمز خطأ غير صفري عند الفشل. في حالة النجاح، تقوم pthread_key_create() بتخزين المفتاح المخصص حديثًا في الموقع الذي يشير إليه الوسيط key.

تُرجع pthread_getspecific() القيمة المرتبطة بـ key عند النجاح، و NULL عند الفشل.

الأخطاء

ترجع pthread_key_create() رمز الخطأ التالي عند الفشل:

EAGAIN
تم تخصيص مفاتيح PTHREAD_KEYS_MAX بالفعل.

ترجع pthread_key_delete() و pthread_setspecific() رمز الخطأ التالي عند الفشل:

EINVAL
key ليس مفتاح TSD صالحًا ومخصصًا.

ترجع pthread_getspecific() NULL إذا كان key ليس مفتاح TSD صالحًا ومخصصًا.

راجع أيضًا

pthread_create(3), pthread_exit(3), pthread_testcancel(3).

مثال

يقوم جزء الكود التالي بتخصيص مصفوفة أحرف خاصة بالخيط بحجم 100 حرف، مع إعادة تخصيص تلقائية عند انتهاء الخيط:

/* مفتاح المخزن المؤقت الخاص بالخيط */
static pthread_key_t buffer_key;
/* تهيئة المفتاح مرة واحدة فقط */
static pthread_once_t buffer_key_once = PTHREAD_ONCE_INIT;
/* تخصيص المخزن المؤقت الخاص بالخيط */
void buffer_alloc(void)
{
  pthread_once(&buffer_key_once, buffer_key_alloc);
  pthread_setspecific(buffer_key, malloc(100));
}
/* إرجاع المخزن المؤقت الخاص بالخيط */
char * get_buffer(void)
{
  return (char *) pthread_getspecific(buffer_key);
}
/* تخصيص المفتاح */
static void buffer_key_alloc()
{
  pthread_key_create(&buffer_key, buffer_destroy);
}
/* تحرير المخزن المؤقت الخاص بالخيط */
static void buffer_destroy(void * buf)
{
  free(buf);
}
