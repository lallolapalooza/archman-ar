pthread_cond_timedwait(3)

pthread_cond_init(3) دليل الدوال المكتبية pthread_cond_init(3)

الاسم

pthread_cond_init, pthread_cond_signal, pthread_cond_broadcast, pthread_cond_wait, pthread_cond_timedwait, pthread_cond_destroy - عمليات على الشروط

الملخص

#include <pthread.h>
pthread_cond_t cond = PTHREAD_COND_INITIALIZER;
int pthread_cond_init(pthread_cond_t *cond,
                      pthread_condattr_t *cond_attr);
int pthread_cond_signal(pthread_cond_t *cond);
int pthread_cond_broadcast(pthread_cond_t *cond);
int pthread_cond_wait(pthread_cond_t *cond, pthread_mutex_t *mutex);
int pthread_cond_timedwait(pthread_cond_t *cond, pthread_mutex_t *mutex,
                      const struct timespec *abstime);
int pthread_cond_destroy(pthread_cond_t *cond);

الوصف

الشرط (اختصار لـ "متغير الشرط") هو جهاز تزامن يسمح للخيوط بتعليق التنفيذ والتخلي عن المعالجات حتى يتم تلبية بعض المسندات على البيانات المشتركة. العمليات الأساسية على الشروط هي: إشارة الشرط (عندما يصبح المسند صحيحًا)، وانتظار الشرط، وتعليق تنفيذ الخيط حتى يقوم خيط آخر بإشارة الشرط.

يجب دائمًا ربط متغير الشرط بقفل (mutex) لتجنب حالة السباق حيث يقوم خيط بالتحضير للانتظار على متغير شرط، ويقوم خيط آخر بإشارة الشرط قبل أن ينتظر الخيط الأول فعليًا.

تقوم pthread_cond_init() بتهيئة متغير الشرط cond، باستخدام سمات الشرط المحددة في cond_attr، أو السمات الافتراضية إذا كان cond_attr فارغًا (NULL). لا يدعم تطبيق LinuxThreads أي سمات للشروط، وبالتالي يتم تجاهل المعامل cond_attr بالفعل.

يمكن أيضًا تهيئة المتغيرات من النوع pthread_cond_t بشكل ثابت، باستخدام الثابت PTHREAD_COND_INITIALIZER.

تقوم pthread_cond_signal() بإعادة تشغيل أحد الخيوط التي تنتظر على متغير الشرط cond. إذا لم تكن هناك خيوط تنتظر على cond، فلا يحدث شيء. إذا كانت عدة خيوط تنتظر على cond، فسيتم إعادة تشغيل واحد فقط، ولكن لا يتم تحديد أي منها.

تقوم pthread_cond_broadcast() بإعادة تشغيل جميع الخيوط التي تنتظر على متغير الشرط cond. لا يحدث شيء إذا لم تكن هناك خيوط تنتظر على cond.

تقوم pthread_cond_wait() بإلغاء قفل mutex بشكل ذري (وفقًا لـ pthread_unlock_mutex()) وتنتظر حتى يتم إشارة متغير الشرط cond. يتم تعليق تنفيذ الخيط ولا يستهلك أي وقت معالج حتى يتم إشارة متغير الشرط. يجب أن يكون mutex مقفلاً بواسطة الخيط المتصل عند الدخول إلى pthread_cond_wait(). قبل العودة إلى الخيط المتصل، تعيد pthread_cond_wait() قفل mutex (وفقًا لـ pthread_mutex_lock()).

يتم إلغاء قفل القفل والتعليق على متغير الشرط بشكل ذري. وبالتالي، إذا كان جميع الخيوط يحصلون دائمًا على القفل قبل إشارة الشرط، فهذا يضمن أنه لا يمكن إشارة الشرط (وبالتالي تجاهله) بين الوقت الذي يقفل فيه الخيط القفل والوقت الذي ينتظر فيه على متغير الشرط. انظر الملاحظات أدناه.

تقوم pthread_cond_timedwait() بإلغاء قفل mutex بشكل ذري وتنتظر على cond، كما تفعل pthread_cond_wait()، ولكنها تحدد أيضًا مدة الانتظار. إذا لم يتم إشارة cond خلال المدة المحددة بواسطة abstime، تتم إعادة قفل mutex وتعيد pthread_cond_timedwait() الخطأ ETIMEDOUT. يحدد المعامل abstime وقتًا مطلقًا، بنفس أصل time(2) و gettimeofday(2): abstime بقيمة 0 يتوافق مع 00:00:00 بتوقيت جرينتش، 1 يناير 1970.

تقوم pthread_cond_destroy() بتدمير متغير شرط، وتحرير الموارد التي قد يمتلكها. لا يجب أن تكون هناك خيوط تنتظر على متغير الشرط عند الدخول إلى pthread_cond_destroy(). في تنفيذ LinuxThreads، لا توجد موارد مرتبطة بمتغيرات الشروط، وبالتالي فإن pthread_cond_destroy() لا تفعل شيئًا فعليًا باستثناء التحقق من أن الشرط لا يحتوي على خيوط تنتظر.

الإلغاء

pthread_cond_wait() و pthread_cond_timedwait() هي نقاط إلغاء. إذا تم إلغاء خيط أثناء تعليقه في إحدى هذه الدوال، يتم استئناف تنفيذ الخيط على الفور، ثم يعيد قفل mutex كوسيط لـ pthread_cond_wait() و pthread_cond_timedwait()، وينفذ الإلغاء في النهاية. وبالتالي، يتم ضمان أن معالجات التنظيف تقوم بقفل mutex عند استدعائها.

السلامة من الإشارات غير المتزامنة

وظائف الشرط ليست آمنة للإشارات غير المتزامنة، ويجب عدم استدعائها من معالج الإشارة. على وجه الخصوص، قد يؤدي استدعاء pthread_cond_signal() أو pthread_cond_broadcast() من معالج الإشارة إلى إيقاف الخيط المتصل مؤقتًا (deadlock).

قيمة الإرجاع

جميع وظائف متغير الشرط تعيد 0 عند النجاح، ورمز خطأ غير صفري عند حدوث خطأ.

الأخطاء

pthread_cond_init()، pthread_cond_signal()، pthread_cond_broadcast()، و pthread_cond_wait() لا تعيد أبدًا رمز خطأ.

تعيد الدالة pthread_cond_timedwait() رموز الخطأ التالية عند حدوث خطأ:

ETIMEDOUT
لم يتم إشارة متغير الشرط حتى انقضاء المهلة المحددة بواسطة abstime.

تعيد الدالة pthread_cond_destroy() رمز الخطأ التالي عند حدوث خطأ:

EBUSY
هناك خيوط تنتظر حاليًا على cond.

انظر أيضًا

pthread_condattr_init(3)، pthread_mutex_lock(3)، pthread_mutex_unlock(3)، gettimeofday(2)، nanosleep(2).

ملاحظات

استخدم تنفيذ الدوال المقدمة حتى glibc 2.25 قفلاً داخليًا للبيانات. لم يدعم هذا القفل وراثة الأولوية وكان عرضة للانعكاس غير المحدود للأولوية، وهو أمر ظاهر في نظام في الوقت الفعلي.

بعد إعادة كتابة التنفيذ في glibc 2.25، تغير استخدام القفل الداخلي. يتم الحصول على القفل الداخلي دائمًا بواسطة الدوال التي ترسل إشارات pthread_cond_signal() و pthread_cond_broadcast(). يحصل الخيط المنتظر على القفل إذا تم مقاطعة عملية الانتظار. يمكن أن تحدث المقاطعة على سبيل المثال بسبب مهلة زمنية محددة، ويشار إليها بقيمة الخطأ ETIMEDOUTA، أو بسبب إشارة مستلمة، والتي يشار إليها بقيمة الخطأ EINTR.

مثال

اعتبر متغيرين مشتركين x و y، محميين بواسطة القفل mut، ومتغير شرط cond الذي سيتم إشارته عندما تصبح x أكبر من y.

int x,y;
pthread_mutex_t mut = PTHREAD_MUTEX_INITIALIZER;
pthread_cond_t cond = PTHREAD_COND_INITIALIZER;

يتم إجراء الانتظار حتى تصبح x أكبر من y على النحو التالي:

pthread_mutex_lock(&mut);
while (x <= y) {
        pthread_cond_wait(&cond, &mut);
}
/* تشغيل على x و y */
pthread_mutex_unlock(&mut);

يجب على التعديلات على x و y التي قد تتسبب في أن تصبح x أكبر من y إشارة الشرط إذا لزم الأمر:

pthread_mutex_lock(&mut);
/* تعديل x و y */
if (x > y) pthread_cond_broadcast(&cond);
pthread_mutex_unlock(&mut);
