pthread_key_create(3) دليل وظائف المكتبة pthread_key_create(3)

الاسم

pthread_key_create, pthread_key_delete, pthread_setspecific, pthread_getspecific - إدارة البيانات الخاصة بالخيط

الملخص

#include <pthread.h>
int pthread_key_create(pthread_key_t *key,
                       typeof(void (void *)) *destr_function;
int pthread_key_delete(pthread_key_t key);
int pthread_setspecific(pthread_key_t key, const void *pointer);
void * pthread_getspecific(pthread_key_t key);

الوصف

غالبًا ما تحتاج البرامج إلى متغيرات عامة أو ثابتة لها قيم مختلفة في خيوط مختلفة. نظرًا لأن الخيوط تشترك في مساحة ذاكرة واحدة، لا يمكن تحقيق ذلك باستخدام المتغيرات العادية. البيانات الخاصة بالخيط هي إجابة POSIX threads لهذه الحاجة.

يمتلك كل خيط كتلة ذاكرة خاصة به، وهي منطقة البيانات الخاصة بالخيط، أو TSD area اختصارًا. يتم فهرسة هذه المنطقة بواسطة مفاتيح TSD. تربط TSD keys قيمًا من نوع void * بمفاتيح TSD. مفاتيح TSD مشتركة بين جميع الخيوط، ولكن القيمة المرتبطة بمفتاح TSD معين يمكن أن تكون مختلفة في كل خيط.

للتوضيح، يمكن اعتبار مناطق TSD كمصفوفات من مؤشرات void *، ومفاتيح TSD كفهارس عددية لهذه المصفوفات، وقيمة مفتاح TSD كقيمة العنصر المقابل في المصفوفة في الخيط المستدعي.

عند إنشاء خيط، تقوم منطقة TSD الخاصة به في البداية بربط NULL بجميع المفاتيح.

تقوم pthread_key_create() بتخصيص مفتاح TSD جديد. يتم تخزين المفتاح في الموقع المشار إليه بواسطة key. يوجد حد أقصى وهو PTHREAD_KEYS_MAX للمفاتيح التي يمكن تخصيصها في وقت معين. القيمة المرتبطة مبدئيًا بالمفتاح المُرجع هي NULL في جميع الخيوط التي يتم تنفيذها حاليًا.

الوسيط destr_function، إذا لم يكن NULL، يحدد دالة مدمرة مرتبطة بالمفتاح. عند انتهاء خيط عبر pthread_exit() أو عن طريق الإلغاء، يتم استدعاء destr_function مع الوسيط وهو القيمة المرتبطة بالمفتاح في ذلك الخيط. لا يتم استدعاء destr_function إذا كانت تلك القيمة NULL. الترتيب الذي يتم به استدعاء دوال التدمير عند انتهاء الخيط غير محدد.

قبل استدعاء دالة التدمير، يتم ربط القيمة NULL بالمفتاح في الخيط الحالي. ومع ذلك، قد تعيد دالة التدمير ربط قيم غير NULL بهذا المفتاح أو مفتاح آخر. للتعامل مع هذا، إذا كانت لا تزال هناك بعض القيم غير NULL المرتبطة بدوال تدمير بعد استدعاء جميع دوال التدمير لجميع القيم غير NULL، فسيتم تكرار العملية. يتوقف تنفيذ glibc عن العملية بعد PTHREAD_DESTRUCTOR_ITERATIONS تكرار، حتى لو بقيت بعض القيم غير NULL مع موصفات مرتبطة. قد تدخل عمليات التنفيذ الأخرى في حلقة لا نهائية.

تقوم pthread_key_delete() بإلغاء تخصيص مفتاح TSD. لا تتحقق مما إذا كانت هناك قيم غير NULL مرتبطة بهذا المفتاح في الخيوط التي يتم تنفيذها حاليًا، ولا تستدعي دالة التدمير المرتبطة بالمفتاح.

تقوم pthread_setspecific() بتغيير القيمة المرتبطة بـ key في الخيط المستدعي، وتخزين pointer المُعطى بدلاً من ذلك.

تقوم pthread_getspecific() بإرجاع القيمة المرتبطة حاليًا بـ key في الخيط المستدعي.

قيمة الإرجاع

pthread_key_create() و pthread_key_delete() و pthread_setspecific() ترجع 0 عند النجاح ورمز خطأ غير صفري عند الفشل. عند النجاح، تقوم pthread_key_create() بتخزين المفتاح المخصص حديثًا في الموقع المشار إليه بواسطة وسيطها key.

pthread_getspecific() ترجع القيمة المرتبطة بـ key عند النجاح، و NULL عند الفشل.

الأخطاء

pthread_key_create() ترجع رمز الخطأ التالي عند حدوث خطأ:

EAGAIN
تم تخصيص مفاتيح PTHREAD_KEYS_MAX بالفعل.

pthread_key_delete() و pthread_setspecific() ترجعان رمز الخطأ التالي عند حدوث خطأ:

EINVAL
key ليس مفتاح TSD صالحًا ومخصصًا.

pthread_getspecific() ترجع NULL إذا كان key ليس مفتاح TSD صالحًا ومخصصًا.

انظر أيضًا

pthread_create(3), pthread_exit(3), pthread_testcancel(3).

مثال

يقوم جزء الكود التالي بتخصيص مخزن مؤقت خاص بالخيط بحجم 100 بايت، مع إعادة تخصيص تلقائي عند انتهاء الخيط:

/* مفتاح المخزن المؤقت الخاص بالخيط */
static pthread_key_t buffer_key;
/* تهيئة المفتاح مرة واحدة فقط */
static pthread_once_t buffer_key_once = PTHREAD_ONCE_INIT;
/* تخصيص المخزن المؤقت الخاص بالخيط */
void buffer_alloc(void)
{
  pthread_once(&buffer_key_once, buffer_key_alloc);
  pthread_setspecific(buffer_key, malloc(100));
}
/* إرجاع المخزن المؤقت الخاص بالخيط */
char * get_buffer(void)
{
  return (char *) pthread_getspecific(buffer_key);
}
/* تخصيص المفتاح */
static void buffer_key_alloc()
{
  pthread_key_create(&buffer_key, buffer_destroy);
}
/* تحرير المخزن المؤقت الخاص بالخيط */
static void buffer_destroy(void * buf)
{
  free(buf);
}
