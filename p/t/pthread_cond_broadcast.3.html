pthread_cond_init(3) دليل الدوال المكتبية pthread_cond_init(3)

الاسم

pthread_cond_init, pthread_cond_signal, pthread_cond_broadcast, pthread_cond_wait, pthread_cond_timedwait, pthread_cond_destroy - عمليات على الشروط

الملخص

#include <pthread.h>
pthread_cond_t cond = PTHREAD_COND_INITIALIZER;
int pthread_cond_init(pthread_cond_t *cond,
                      pthread_condattr_t *cond_attr);
int pthread_cond_signal(pthread_cond_t *cond);
int pthread_cond_broadcast(pthread_cond_t *cond);
int pthread_cond_wait(pthread_cond_t *cond, pthread_mutex_t *mutex);
int pthread_cond_timedwait(pthread_cond_t *cond, pthread_mutex_t *mutex,
                      const struct timespec *abstime);
int pthread_cond_destroy(pthread_cond_t *cond);

الوصف

الشرط (اختصار لـ "متغير الشرط") هو جهاز تزامن يسمح للخيوط بتعليق التنفيذ والتخلي عن المعالجات حتى يتم استيفاء افتراض على البيانات المشتركة. العمليات الأساسية على الشروط هي: إشارة الشرط (عندما يصبح الافتراض صحيحًا)، والانتظار للشرط، وتعليق تنفيذ الخيط حتى يقوم خيط آخر بإشارة الشرط.

يجب دائمًا ربط متغير الشرط بقفل (mutex)، لتجنب حالة السباق حيث يقوم خيط بإعداد الانتظار على متغير شرط وخيط آخر يقوم بإشارة الشرط قبل أن ينتظر الخيط الأول فعليًا عليه.

pthread_cond_init() تهيئة متغير الشرط cond، باستخدام سمات الشرط المحددة في cond_attr، أو السمات الافتراضية إذا كان cond_attr هو NULL. تطبيق LinuxThreads لا يدعم أي سمات للشروط، لذلك يتم تجاهل المعلمة cond_attr بالفعل.

يمكن أيضًا تهيئة المتغيرات من النوع pthread_cond_t بشكل ثابت، باستخدام الثابت PTHREAD_COND_INITIALIZER.

pthread_cond_signal() يعيد تشغيل أحد الخيوط التي تنتظر على متغير الشرط cond. إذا لم تكن هناك خيوط تنتظر على cond، فلا يحدث شيء. إذا كانت عدة خيوط تنتظر على cond، يتم إعادة تشغيل واحد بالضبط، ولكن لم يتم تحديد أي منها.

pthread_cond_broadcast() يعيد تشغيل جميع الخيوط التي تنتظر على متغير الشرط cond. لا يحدث شيء إذا لم تكن هناك خيوط تنتظر على cond.

pthread_cond_wait() يقوم بإلغاء قفل mutex (وفقًا لـ pthread_unlock_mutex()) بشكل ذري وينتظر حتى يتم إرسال إشارة إلى متغير الشرط cond. يتم تعليق تنفيذ الخيط ولا يستهلك أي وقت وحدة المعالجة المركزية حتى يتم إرسال إشارة إلى متغير الشرط. يجب أن يكون mutex مقفلاً بواسطة الخيط المتصل عند الدخول إلى pthread_cond_wait(). قبل العودة إلى الخيط المتصل، يقوم pthread_cond_wait() بإعادة الحصول على mutex (وفقًا لـ pthread_mutex_lock()).

إلغاء قفل القفل والانتظار على متغير الشرط يتم بشكل ذري. وبالتالي، إذا كان جميع الخيوط تحصل دائمًا على القفل قبل إرسال إشارة إلى الشرط، فهذا يضمن أنه لا يمكن إرسال إشارة إلى الشرط (وبالتالي تجاهله) بين الوقت الذي يقفل فيه الخيط القفل والوقت الذي ينتظر فيه على متغير الشرط. انظر تحذيرات أدناه.

pthread_cond_timedwait() يلغي قفل mutex بشكل ذري وينتظر على cond، كما يفعل pthread_cond_wait()، ولكنه يحدد أيضًا مدة الانتظار. إذا لم يتم إرسال إشارة إلى cond خلال الوقت المحدد بواسطة abstime، يتم إعادة الحصول على القفل mutex وتعيد pthread_cond_timedwait() الخطأ ETIMEDOUT. تحدد المعلمة abstime وقتًا مطلقًا، بنفس الأصل الذي تستخدمه time(2) و gettimeofday(2): abstime بقيمة 0 يقابل 00:00:00 بتوقيت جرينتش، 1 يناير 1970.

pthread_cond_destroy() يدمر متغير شرط، ويحرر الموارد التي قد يحتفظ بها. لا يجب أن تكون هناك خيوط تنتظر على متغير الشرط عند الدخول إلى pthread_cond_destroy(). في تطبيق LinuxThreads، لا توجد موارد مرتبطة بمتغيرات الشروط، وبالتالي فإن pthread_cond_destroy() لا يفعل شيئًا فعليًا باستثناء التحقق من أن الشرط لا يحتوي على خيوط منتظرة.

الإلغاء

pthread_cond_wait() و pthread_cond_timedwait() هي نقاط إلغاء. إذا تم إلغاء خيط أثناء تعليقه في إحدى هذه الدوال، فإن الخيط يستأنف التنفيذ فورًا، ثم يعيد قفل mutex الذي تم تمريره إلى pthread_cond_wait() و pthread_cond_timedwait()، وأخيرًا ينفذ عملية الإلغاء. نتيجة لذلك، يتم التأكيد على معالجات التنظيف أن mutex مقفل عند استدعائها.

أمان إشارة غير متزامن

دوال الشرط ليست آمنة للإشارات غير المتزامنة، ويجب عدم استدعائها من معالج إشارة. بشكل خاص، قد يؤدي استدعاء pthread_cond_signal() أو pthread_cond_broadcast() من معالج إشارة إلى تعليق الخيط المتصل.

قيمة الإرجاع

جميع دوال متغير الشرط ترجع 0 عند النجاح ورمز خطأ غير صفري عند الخطأ.

الأخطاء

pthread_cond_init() و pthread_cond_signal() و pthread_cond_broadcast() و pthread_cond_wait() لا ترجع أبداً رموز خطأ.

دالة pthread_cond_timedwait() ترجع رموز الخطأ التالية عند الخطأ:

ETIMEDOUT
لم يتم إرسال إشارة إلى متغير الشرط حتى انتهاء المهلة المحددة بواسطة abstime .

دالة pthread_cond_destroy() ترجع رمز الخطأ التالي عند الخطأ:

EBUSY
هناك خيوط تنتظر حاليًا على cond .

تحذيرات

تطبيق الدوال المقدمة حتى glibc 2.25 استخدم قفل بيانات داخلي. لم يدعم هذا القفل وراثة الأولوية وكان عرضة لـ "تضخيم الأولوية غير المحدود"، وهو أمر ظاهر على نظام في الوقت الفعلي.

بعد إعادة كتابة التطبيق في glibc 2.25، تغير استخدام القفل الداخلي. يتم الحصول على القفل الداخلي دائمًا بواسطة دوال الإشارة pthread_cond_signal() و pthread_cond_broadcast(). تحصل الدالة المنتظرة على القفل إذا تم مقاطعة العملية المنتظرة. يمكن أن تحدث المقاطعة على سبيل المثال بسبب مهلة محددة، ويشار إليها بقيمة الخطأ ETIMEDOUTA، أو بسبب إشارة تم تلقيها، والتي يشار إليها بقيمة الخطأ EINTR.

مثال

ضع في اعتبارك متغيرين مشتركين x و y، محميين بواسطة القفل mut، ومتغير شرط cond الذي سيتم إرسال إشارة إليه عندما تصبح x أكبر من y.

int x,y;
pthread_mutex_t mut = PTHREAD_MUTEX_INITIALIZER;
pthread_cond_t cond = PTHREAD_COND_INITIALIZER;

الانتظار حتى تصبح x أكبر من y يتم كالتالي:

pthread_mutex_lock(&mut);
while (x <= y) {
        pthread_cond_wait(&cond, &mut);
}
/* العمل على x و y */
pthread_mutex_unlock(&mut);

التعديلات على x و y التي قد تتسبب في أن تصبح x أكبر من y يجب أن ترسل إشارة إلى الشرط إذا لزم الأمر:

pthread_mutex_lock(&mut);
/* تعديل x و y */
if (x > y) pthread_cond_broadcast(&cond);
pthread_mutex_unlock(&mut);

إذا كان يمكن إثبات أن خيطًا واحدًا على الأكثر يحتاج إلى الاستيقاظ (على سبيل المثال، إذا كان هناك خيطان يتواصلان عبر

x و y)، يمكن استخدام pthread_cond_signal() كبديل أكثر كفاءة قليلاً لـ pthread_cond_broadcast(). في حالة الشك، استخدم pthread_cond_broadcast().

للانتظار حتى تصبح x أكبر من y مع مهلة 5 ثوانٍ، قم بما يلي:

struct timeval now;
struct timespec timeout;
int retcode;
pthread_mutex_lock(&mut);
gettimeofday(&now);
timeout.tv_sec = now.tv_sec + 5;
timeout.tv_nsec = now.tv_usec * 1000;
retcode = 0;
while (x <= y && retcode != ETIMEDOUT) {
        retcode = pthread_cond_timedwait(&cond, &mut, &timeout);
}
if (retcode == ETIMEDOUT) {
        /* حدث انتهاء المهلة */
} else {
        /* تعامل مع x و y */
}
pthread_mutex_unlock(&mut);
