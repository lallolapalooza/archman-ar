pthread_cond_init(3) دليل الدوال المكتبية pthread_cond_init(3)

الاسم

pthread_cond_init, pthread_cond_signal, pthread_cond_broadcast, pthread_cond_wait, pthread_cond_timedwait, pthread_cond_destroy - عمليات على الشروط

الملخص

#include <pthread.h>
pthread_cond_t cond = PTHREAD_COND_INITIALIZER;
int pthread_cond_init(pthread_cond_t *cond,
                      pthread_condattr_t *cond_attr);
int pthread_cond_signal(pthread_cond_t *cond);
int pthread_cond_broadcast(pthread_cond_t *cond);
int pthread_cond_wait(pthread_cond_t *cond, pthread_mutex_t *mutex);
int pthread_cond_timedwait(pthread_cond_t *cond, pthread_mutex_t *mutex,
                      const struct timespec *abstime);
int pthread_cond_destroy(pthread_cond_t *cond);

الوصف

الشرط (اختصار لـ "متغير الشرط") هو جهاز مزامنة يسمح للخيوط بتعليق التنفيذ والتخلي عن المعالجات حتى يتم تلبية شرط معين على البيانات المشتركة. العمليات الأساسية على الشروط هي: إشارة الشرط (عندما يصبح الشرط صحيحًا)، وانتظار الشرط، وتعليق تنفيذ الخيط حتى يشير خيط آخر إلى الشرط.

يجب دائمًا ربط متغير الشرط بـ mutex لتجنب حالة السباق حيث يستعد خيط للانتظار على متغير شرطي وخيط آخر يشير إلى الشرط قبل أن ينتظر الخيط الأول بالفعل.

يقوم pthread_cond_init() بتهيئة متغير الشرط cond، باستخدام سمات الشرط المحددة في cond_attr، أو السمات الافتراضية إذا كان cond_attr هو NULL. تنفيذ LinuxThreads لا يدعم أي سمات للشروط، لذا يتم تجاهل المعلمة cond_attr فعليًا.

يمكن أيضًا تهيئة المتغيرات من النوع pthread_cond_t بشكل ثابت، باستخدام الثابت PTHREAD_COND_INITIALIZER.

يقوم pthread_cond_signal() بإعادة تشغيل أحد الخيوط التي تنتظر على متغير الشرط cond. إذا لم يكن هناك خيوط تنتظر على cond، فلن يحدث شيء. إذا كانت هناك عدة خيوط تنتظر على cond، فسيتم إعادة تشغيل واحد بالضبط، ولكن لم يتم تحديد أي واحد.

يقوم pthread_cond_broadcast() بإعادة تشغيل جميع الخيوط التي تنتظر على متغير الشرط cond. لن يحدث شيء إذا لم يكن هناك خيوط تنتظر على cond.

يقوم pthread_cond_wait() بإلغاء قفل mutex (وفقًا لـ pthread_unlock_mutex()) وينتظر حتى يتم الإشارة إلى متغير الشرط cond. يتم تعليق تنفيذ الخيط ولا يستهلك أي وقت للمعالج حتى يتم الإشارة إلى متغير الشرط. يجب أن يكون mutex مقفلًا بواسطة الخيط المستدعي عند الدخول إلى pthread_cond_wait(). قبل العودة إلى الخيط المستدعي، يعيد pthread_cond_wait() الحصول على mutex (وفقًا لـ pthread_mutex_lock()).

يتم إلغاء قفل الـ mutex وتعليقه على متغير الشرط بشكل ذري. وبالتالي، إذا كان جميع الخيوط تحصل دائمًا على الـ mutex قبل الإشارة إلى الشرط، فهذا يضمن أنه لا يمكن الإشارة إلى الشرط (وبالتالي تجاهله) بين الوقت الذي يقفل فيه الخيط الـ mutex والوقت الذي ينتظر فيه متغير الشرط.

يقوم pthread_cond_timedwait() بإلغاء قفل mutex بشكل ذري وينتظر على cond، كما يفعل pthread_cond_wait()، ولكنه يحدد أيضًا مدة الانتظار. إذا لم يتم الإشارة إلى cond خلال المدة المحددة بواسطة abstime، فسيتم إعادة الحصول على الـ mutex mutex وستعيد pthread_cond_timedwait() الخطأ ETIMEDOUT. تحدد المعلمة abstime وقتًا مطلقًا، بنفس مصدر time(2) و gettimeofday(2): abstime بقيمة 0 يتوافق مع 00:00:00 GMT، 1 يناير 1970.

يقوم pthread_cond_destroy() بتدمير متغير شرط، مما يؤدي إلى تحرير الموارد التي قد يحتفظ بها. لا يجب أن تنتظر أي خيوط على متغير الشرط عند الدخول إلى pthread_cond_destroy(). في تنفيذ LinuxThreads، لا توجد موارد مرتبطة بمتغيرات الشروط، وبالتالي فإن pthread_cond_destroy() لا يفعل شيئًا فعليًا باستثناء التحقق من عدم وجود خيوط تنتظر على الشرط.

الإلغاء

pthread_cond_wait() و pthread_cond_timedwait() هي نقاط إلغاء. إذا تم إلغاء خيط أثناء تعليقه في إحدى هاتين الدالتين، فسيستأنف الخيط التنفيذ فورًا، ثم يعيد قفل mutex المقدم إلى pthread_cond_wait() و pthread_cond_timedwait()، ثم ينفذ الإلغاء. نتيجة لذلك، يتم التأكد من أن معالجات التنظيف تقوم بقفل mutex عندما يتم استدعاؤها.

سلامة الإشارة غير المتزامنة

دوال الشرط ليست آمنة للإشارة غير المتزامنة، ويجب عدم استدعائها من معالج إشارة. على وجه الخصوص، قد يؤدي استدعاء pthread_cond_signal() أو pthread_cond_broadcast() من معالج إشارة إلى إيقاف الخيط المستدعي.

قيمة الإرجاع

ترجع جميع دوال متغير الشرط 0 عند النجاح ورمز خطأ غير صفري عند حدوث خطأ.

الأخطاء

pthread_cond_init(), pthread_cond_signal(), pthread_cond_broadcast(), و pthread_cond_wait() لا تعيد أبدًا رمز خطأ.

تعيد الدالة pthread_cond_timedwait() رموز الخطأ التالية عند حدوث خطأ:

ETIMEDOUT
لم يتم الإشارة إلى متغير الشرط حتى انقضاء المهلة المحددة بواسطة abstime.

تعيد الدالة pthread_cond_destroy() رمز الخطأ التالي عند حدوث خطأ:

EBUSY
هناك خيوط تنتظر حاليًا على cond.

تنبيهات

استخدمت آلية تنفيذ الدوال المقدمة حتى الإصدار 2.25 من glibc قفل بيانات داخلي. لم يدعم هذا القفل توريث الأولوية وكان عرضة للانعكاس غير المحدود للأولوية، وهو ما كان ظاهرًا في نظام الوقت الفعلي.

بعد إعادة كتابة الآلية في glibc 2.25، تغير استخدام القفل الداخلي. يتم دائمًا الحصول على القفل الداخلي بواسطة الدوال التي تشير pthread_cond_signal() و pthread_cond_broadcast(). يحصل دالة الانتظار على القفل إذا تم مقاطعة عملية الانتظار. يمكن أن يكون سبب المقاطعة، على سبيل المثال، مهلة زمنية محددة، ويشار إليها بقيمة الخطأ ETIMEDOUTA، أو إشارة تم تلقيها، والتي يشار إليها بقيمة الخطأ EINTR.

مثال

ضع في اعتبارك متغيرين مشتركين x و y، محميين بواسطة الـ mutex mut، ومتغير شرط cond الذي يجب الإشارة إليه عندما يصبح x أكبر من y.

int x,y;
pthread_mutex_t mut = PTHREAD_MUTEX_INITIALIZER;
pthread_cond_t cond = PTHREAD_COND_INITIALIZER;

الانتظار حتى يصبح x أكبر من y يتم تنفيذه على النحو التالي:

pthread_mutex_lock(&mut);
while (x <= y) {
        pthread_cond_wait(&cond, &mut);
}
/* عملية على x و y */
pthread_mutex_unlock(&mut);

التعديلات على x و y التي قد تتسبب في أن يصبح x أكبر من y يجب أن تشير إلى الشرط إذا لزم الأمر:

pthread_mutex_lock(&mut);
/* تعديل x و y */
if (x > y) pthread_cond_broadcast(&cond);
pthread_mutex_unlock(&mut);
