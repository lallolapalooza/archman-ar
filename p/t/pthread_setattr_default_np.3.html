pthread_getattr_default_np(3) دليل وظائف المكتبة pthread_getattr_default_np(3)

الاسم

pthread_getattr_default_np, pthread_setattr_default_np, - الحصول على أو تعيين سمات الإنشاء الافتراضية للخيط

المكتبة

مكتبة خيوط POSIX (libpthread, -lpthread)

الملخص

#define _GNU_SOURCE             /* انظر feature_test_macros(7) */
#include <pthread.h>
int pthread_getattr_default_np(pthread_attr_t *attr);
int pthread_setattr_default_np(const pthread_attr_t *attr);

الوصف

تقوم الدالة pthread_setattr_default_np() بتعيين السمات الافتراضية المستخدمة لإنشاء خيط جديد — أي السمات التي تُستخدم عند استدعاء pthread_create(3) مع وسيط ثانٍ قيمته NULL. يتم تعيين السمات الافتراضية باستخدام السمات المقدمة في *attr، وهو كائن سمات خيط تم تهيئته مسبقًا. لاحظ التفاصيل التالية حول كائن السمات المقدم:

•
يجب أن تكون إعدادات السمة في الكائن صالحة.
•
يجب عدم تعيين سمة عنوان المكدس في الكائن.
•
تعيين سمة حجم المكدس إلى صفر يعني ترك حجم المكدس الافتراضي دون تغيير.

تقوم الدالة pthread_getattr_default_np() بتهيئة كائن سمات الخيط المشار إليه بواسطة attr بحيث يحتوي على السمات الافتراضية المستخدمة لإنشاء الخيط.

الأخطاء

EINVAL
(pthread_setattr_default_np()) أحد إعدادات السمة في attr غير صالح، أو تم تعيين سمة عنوان المكدس في attr.
ENOMEM
(pthread_setattr_default_np()) ذاكرة غير كافية.

السمات

للحصول على شرح للمصطلحات المستخدمة في هذا القسم، انظر attributes(7).

الواجهة السمة القيمة
pthread_getattr_default_np (), pthread_setattr_default_np () سلامة الخيوط MT-Safe

المعايير

GNU؛ وبالتالي اللاحقة "_np" (غير قابلة للنقل) في أسمائها.

التاريخ

glibc 2.18.

أمثلة

يستخدم البرنامج أدناه pthread_getattr_default_np() لجلب سمات إنشاء الخيط الافتراضية ثم يعرض إعدادات مختلفة من كائن سمات الخيط المُرجع. عند تشغيل البرنامج، نرى المخرجات التالية:


$ ./a.out;
حجم المكدس:          8388608
حجم الحارس:          4096
سياسة الجدولة:   SCHED_OTHER
أولوية الجدولة: 0
حالة الفصل:       JOINABLE
وراثة المجدول:   INHERIT

مصدر البرنامج

#define _GNU_SOURCE
#include <err.h>
#include <pthread.h>
#include <sched.h>
#include <stdio.h>
#include <stdlib.h>
static void
display_pthread_attr(pthread_attr_t *attr)
{
    int s;
    size_t stacksize;
    size_t guardsize;
    int policy;
    struct sched_param schedparam;
    int detachstate;
    int inheritsched;
    s = pthread_attr_getstacksize(attr, &stacksize);
    if (s != 0)
        errc(EXIT_FAILURE, s, "pthread_attr_getstacksize");
    printf("حجم المكدس:          %zu\n", stacksize);
    s = pthread_attr_getguardsize(attr, &guardsize);
    if (s != 0)
        errc(EXIT_FAILURE, s, "pthread_attr_getguardsize");
    printf("حجم الحارس:          %zu\n", guardsize);
    s = pthread_attr_getschedpolicy(attr, &policy);
    if (s != 0)
        errc(EXIT_FAILURE, s, "pthread_attr_getschedpolicy");
    printf("سياسة الجدولة:   %s\n",
           (policy == SCHED_FIFO) ? "SCHED_FIFO" :
           (policy == SCHED_RR) ? "SCHED_RR" :
           (policy == SCHED_OTHER) ? "SCHED_OTHER" : "[unknown]");
    s = pthread_attr_getschedparam(attr, &schedparam);
    if (s != 0)
        errc(EXIT_FAILURE, s, "pthread_attr_getschedparam");
    printf("أولوية الجدولة: %d\n", schedparam.sched_priority);
    s = pthread_attr_getdetachstate(attr, &detachstate);
    if (s != 0)
        errc(EXIT_FAILURE, s, "pthread_attr_getdetachstate");
    printf("حالة الفصل:       %s\n",
           (detachstate == PTHREAD_CREATE_DETACHED) ? "DETACHED" :
           (detachstate == PTHREAD_CREATE_JOINABLE) ? "JOINABLE" :
           "???");
    s = pthread_attr_getinheritsched(attr, &inheritsched);
    if (s != 0)
        errc(EXIT_FAILURE, s, "pthread_attr_getinheritsched");
    printf("وراثة المجدول:   %s\n",
           (inheritsched == PTHREAD_INHERIT_SCHED) ? "INHERIT" :
           (inheritsched == PTHREAD_EXPLICIT_SCHED) ? "EXPLICIT" :
           "???");
}
int
main(void)
{
    int s;
    pthread_attr_t attr;
    s = pthread_getattr_default_np(&attr);
    if (s != 0)
        errc(EXIT_FAILURE, s, "pthread_getattr_default_np");
    display_pthread_attr(&attr);
    exit(EXIT_SUCCESS);
}
