pthread_sigmask(3) دليل دوال المكتبة pthread_sigmask(3)

الاسم

pthread_sigmask - فحص وتغيير قناع الإشارات المحظورة

المكتبة

مكتبة خيوط POSIX (libpthread, -lpthread)

الملخص

#include <signal.h>
int pthread_sigmask(int how, const sigset_t *set, sigset_t *oldset);
متطلبات ماكرو اختبار الميزة لـ glibc (انظر feature_test_macros(7)):

pthread_sigmask():

    _POSIX_C_SOURCE >= 199506L || _XOPEN_SOURCE >= 500

الوصف

الدالة pthread_sigmask() تشبه الدالة sigprocmask(2)، مع الاختلاف أن استخدامها في البرامج متعددة الخيوط محدد صراحة بواسطة POSIX.1. ملاحظات على اختلافات أخرى مذكورة في هذه الصفحة.

لوصف الوسائط وعمل هذه الدالة، انظر sigprocmask(2).

قيمة الإرجاع

عند النجاح، ترجع pthread_sigmask() القيمة 0؛ عند حدوث خطأ، ترجع رقم خطأ.

الأخطاء

انظر sigprocmask(2).

السمات

لشرح المصطلحات المستخدمة في هذا القسم، انظر attributes(7).

الواجهة السمة القيمة
pthread_sigmask () أمان الخيط MT-Safe

المعايير

POSIX.1-2008.

التاريخ

POSIX.1-2001.

ملاحظات

يرث الخيط الجديد نسخة من قناع الإشارة الخاص بمنشئه.

تتجاهل دالة glibc pthread_sigmask() بصمت محاولات حظر إشارتين حقيقيتين تستخدمان داخليًا بواسطة تطبيق خيوط NPTL. انظر nptl(7) للتفاصيل.

أمثلة

يقوم البرنامج أدناه بحظر بعض الإشارات في الخيط الرئيسي، ثم ينشئ خيطًا مخصصًا لجلب تلك الإشارات عبر sigwait(3). تظهر جلسة shell التالية استخدامه:


$ ./a.out &
[1] 5423
$ kill -QUIT %1
Signal handling thread got signal 3
$ kill -USR1 %1
Signal handling thread got signal 10
$ kill -TERM %1
[1]+  Terminated              ./a.out

مصدر البرنامج

#include <err.h>
#include <errno.h>
#include <pthread.h>
#include <signal.h>
#include <stdio.h>
#include <stdlib.h>
#include <unistd.h>
/* دوال بسيطة لمعالجة الأخطاء */
static void *
sig_thread(void *arg)
{
    sigset_t *set = arg;
    int s, sig;
    for (;;) {
        s = sigwait(set, &sig);
        if (s != 0)
            errc(EXIT_FAILURE, s, "sigwait");
        printf("Signal handling thread got signal %d\n", sig);
    }
}
int
main(void)
{
    pthread_t thread;
    sigset_t set;
    int s;
    /* حظر SIGQUIT و SIGUSR1؛ الخيوط الأخرى التي تم إنشاؤها بواسطة main()
       سترث نسخة من قناع الإشارة.  */
    sigemptyset(&set);
    sigaddset(&set, SIGQUIT);
    sigaddset(&set, SIGUSR1);
    s = pthread_sigmask(SIG_BLOCK, &set, NULL);
    if (s != 0)
        errc(EXIT_FAILURE, s, "pthread_sigmask");
    s = pthread_create(&thread, NULL, &sig_thread, &set);
    if (s != 0)
        errc(EXIT_FAILURE, s, "pthread_create");
    /* يستمر الخيط الرئيسي في إنشاء خيوط أخرى و/أو القيام
       بأعمال أخرى.  */
    pause();            /* إيقاف مؤقت وهمي لاختبار البرنامج */
}
