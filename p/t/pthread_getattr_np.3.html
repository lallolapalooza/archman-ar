NAME

pthread_getattr_np, pthread_setattr_np — get or set thread attributes (non-standard)

LIBRARY

Pthreads library (libpthread, often linked as -lpthread)

This page describes the non-standard functions pthread_getattr_np and pthread_setattr_np.

SYNOPSIS

#include <pthread.h>

int pthread_getattr_np(pthread_t thread, pthread_attr_t *attr);
int pthread_setattr_np(pthread_t thread, const pthread_attr_t *attr);

DESCRIPTION

The pthread_getattr_np function retrieves the thread attributes object that was in effect when the thread specified by thread was created. The attributes are copied into the object pointed to by attr.

The pthread_setattr_np function sets the thread attributes object for the thread specified by thread to be the attributes contained in the object pointed to by attr. This function can only be used to modify attributes that affect future behavior of the thread, not attributes that only describe the state at the time of creation.

Currently, the only attribute that can be successfully modified using pthread_setattr_np is the stack size attribute. Modifications to other attributes result in an error.

The pthread_getattr_np and pthread_setattr_np functions are non-standard extensions, but are available on Linux and some other systems.

The attributes object attr must have been previously initialized using pthread_attr_init. After use, it should be destroyed using pthread_attr_destroy.

RETURN VALUE

On success, these functions return 0. On failure, they return a non-zero error number.

ERRORS

EINVAL
The value of attr is invalid, or the requested attribute modification is not supported.
ESRCH
No thread could be found corresponding to that thread ID.

ATTRIBUTES

Standard Attributes

For an explanation of the attributes associated with the objects on which [`pthread_getattr_np`](/man/pthread_getattr_np.3.en) and [`pthread_setattr_np`](/man/pthread_setattr_np.3.en) operate, see [`pthread_attr_init`](/man/pthread_attr_init.3.en).

STANDARDS

These functions are non-standard extensions. They are not specified by POSIX.1.

HISTORY

The pthread_getattr_np and pthread_setattr_np functions first appeared in Linux.

EXAMPLES

The following example demonstrates how to use pthread_getattr_np and pthread_setattr_np to retrieve and modify the stack size of a thread.

#include <stdio.h>
#include <stdlib.h>
#include <pthread.h>
#include <err.h>
#include <unistd.h>

#define THREAD_STACK_SIZE (1024 * 1024) /* 1MB */

static void *
thread_start(void *arg)
{
    printf("Thread started, stack size: %d bytes\n", THREAD_STACK_SIZE);
    pause(); /* Keep the thread alive until main thread exits */
    return NULL;
}

static pthread_attr_t *
get_thread_attributes_from_cl(int argc, char *argv[], pthread_attr_t *attrp)
{
    int s;
    size_t stack_size = -1;
    size_t guard_size = -1;
    int allocate_stack = 0;
    void *stack_addr = NULL;
    pthread_attr_t *ret_attrp = NULL;

    /* Default: use thread-default attributes */
    if (argc == 1)
        return NULL;

    s = pthread_attr_init(attrp);
    if (s != 0)
        errc(EXIT_FAILURE, s, "pthread_attr_init");

    ret_attrp = attrp; /* We are using a local attributes object */

    /* Parse command line arguments to set attributes */
    /* (Simplified parsing for demonstration) */
    if (argc > 1) {
        /* Example: parse stack size if provided */
        /* In a real application, you'd use getopt or similar */
        if (argc > 2 && strcmp(argv[1], "--stack-size") == 0) {
            stack_size = atoi(argv[2]);
            if (stack_size < PTHREAD_STACK_MIN) {
                fprintf(stderr, "Stack size too small\n");
                exit(EXIT_FAILURE);
            }
            /* Check if we need to allocate stack manually */
            if (argc > 3 && strcmp(argv[3], "--allocate-stack") == 0) {
                allocate_stack = 1;
            }
        }
        /* Example: parse guard size if provided */
        if (argc > 4 && strcmp(argv[4], "--guard-size") == 0) {
            guard_size = atoi(argv[5]);
        }
    }

    if (stack_size != -1) {
        if (!allocate_stack) {
            s = pthread_attr_setstacksize(attrp, stack_size);
            if (s != 0)
                errc(EXIT_FAILURE, s, "pthread_attr_setstacksize");
        } else {
            s = posix_memalign(&stack_addr, sysconf(_SC_PAGESIZE),
                               stack_size);
            if (s != 0)
                errc(EXIT_FAILURE, s, "posix_memalign");
            printf("Allocated thread stack at %p\n\n", stack_addr);
            s = pthread_attr_setstack(attrp, stack_addr, stack_size);
            if (s != 0)
                errc(EXIT_FAILURE, s, "pthread_attr_setstack");
        }
    }
    if (guard_size != -1) {
        s = pthread_attr_setguardsize(attrp, guard_size);
        if (s != 0)
            errc(EXIT_FAILURE, s, "pthread_attr_setguardsize");
    }
    return ret_attrp;
}

int
main(int argc, char *argv[])
{
    int s;
    pthread_t thr;
    pthread_attr_t attr;
    pthread_attr_t *attrp = NULL;    /* Set to &attr if we initialize
                                        a thread attributes object */

    attrp = get_thread_attributes_from_cl(argc, argv, &attr);

    if (attrp != NULL) {
        printf("Thread attributes object after initializations:\n");
        /* display_stack_related_attributes(attrp, "\t"); */ /* Assuming this function exists */
        printf("\n");
    }

    s = pthread_create(&thr, attrp, &thread_start, NULL);
    if (s != 0)
        errc(EXIT_FAILURE, s, "pthread_create");

    /* Retrieve and display stack size using pthread_getattr_np */
    pthread_attr_t retrieved_attr;
    s = pthread_getattr_np(thr, &retrieved_attr);
    if (s != 0) {
        errc(EXIT_FAILURE, s, "pthread_getattr_np");
    } else {
        size_t stack_size, guard_size;
        s = pthread_attr_getstacksize(&retrieved_attr, &stack_size);
        if (s == 0) {
            printf("Thread stack size (retrieved): %zu bytes\n", stack_size);
        }
        s = pthread_attr_getguardsize(&retrieved_attr, &guard_size);
        if (s == 0) {
            printf("Thread guard size (retrieved): %zu bytes\n", guard_size);
        }
        pthread_attr_destroy(&retrieved_attr); /* Clean up retrieved attributes */
    }

    /* Example of modifying stack size using pthread_setattr_np */
    /* Note: This modification might not be fully effective for already running threads */
    /* or might have limitations depending on the system implementation. */
    /* It's generally better to set attributes at thread creation time. */
    printf("\nAttempting to modify thread stack size to %d bytes using pthread_setattr_np...\n", THREAD_STACK_SIZE);
    pthread_attr_t new_attr;
    s = pthread_attr_init(&new_attr);
    if (s != 0) errc(EXIT_FAILURE, s, "pthread_attr_init for modification");
    s = pthread_attr_setstacksize(&new_attr, THREAD_STACK_SIZE);
    if (s != 0) errc(EXIT_FAILURE, s, "pthread_attr_setstacksize for modification");

    s = pthread_setattr_np(thr, &new_attr);
    if (s != 0) {
        fprintf(stderr, "Failed to set thread attributes using pthread_setattr_np: ");
        /* Check for specific errors if needed */
        if (s == EINVAL) fprintf(stderr, "EINVAL (Invalid attribute or unsupported modification)\n");
        else errc(EXIT_FAILURE, s, "pthread_setattr_np");
    } else {
        printf("pthread_setattr_np call successful (may not affect running thread's current stack).\n");
    }
    pthread_attr_destroy(&new_attr); /* Clean up modification attributes */


    if (attrp != NULL) {
        s = pthread_attr_destroy(attrp);
        if (s != 0)
            errc(EXIT_FAILURE, s, "pthread_attr_destroy");
    }

    pause();    /* Terminates when other thread calls exit() */
    return 0;
}

/* Dummy display_stack_related_attributes for compilation */
void display_stack_related_attributes(pthread_attr_t *attrp, const char *prefix) {
    size_t stack_size, guard_size;
    int s;

    s = pthread_attr_getstacksize(attrp, &stack_size);
    if (s == 0) printf("%sStack size: %zu\n", prefix, stack_size);

    s = pthread_attr_getguardsize(attrp, &guard_size);
    if (s == 0) printf("%sGuard size: %zu\n", prefix, guard_size);
}
