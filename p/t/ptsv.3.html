subroutine cptsv (integer n, integer nrhs, real, dimension( * ) d, complex, dimension( * ) e, complex, dimension( ldb, * ) b, integer ldb, integer info)

CPTSV يحسب حل نظام المعادلات الخطية المعقدة A * X = B لمصفوفات PT

الغرض:

!>
!> CPTSV يحسب حل نظام المعادلات الخطية المعقدة
!> A*X = B, حيث A هي مصفوفة ثلاثية قطرية متناظرة موجبة محددة من الرتبة N x N
!> و X و B هما مصفوفتان من الرتبة N x NRHS.
!>
!> يتم تحليل A إلى A = L*D*L**H، ثم يتم استخدام الشكل المحلل من A
!> لحل نظام المعادلات.
!>

المعلمات

N 
!>          N هو عدد صحيح (INTEGER)
!>          رتبة المصفوفة A. N >= 0.
!>

NRHS

!>          NRHS هو عدد صحيح (INTEGER)
!>          عدد الأطراف اليمنى، أي عدد أعمدة
!>          المصفوفة B. NRHS >= 0.
!>

D

!>          D هي مصفوفة حقيقية (REAL array)، البعد (N)
!>          عند الإدخال، العناصر القطرية n للمصفوفة ثلاثية القطر
!>          A. عند الخروج، العناصر القطرية n للمصفوفة القطرية
!>          D من التحليل A = L*D*L**H.
!>

E

!>          E هي مصفوفة معقدة (COMPLEX array)، البعد (N-1)
!>          عند الإدخال، العناصر تحت القطرية (n-1) للمصفوفة ثلاثية القطر
!>          A. عند الخروج، العناصر تحت القطرية (n-1) للعامل ثنائي القطر الموحد
!>          L من التحليل L*D*L**H للمصفوفة A. يمكن اعتبار E أيضًا كعناصر فوق القطرية للعامل ثنائي القطر الموحد
!>          U من التحليل U**H*D*U للمصفوفة A.
!>

B

!>          B هي مصفوفة معقدة (COMPLEX array)، البعد (LDB,NRHS)
!>          عند الإدخال، مصفوفة الطرف الأيمن B من الرتبة N x NRHS.
!>          عند الخروج، إذا كان INFO = 0، فإن مصفوفة الحل X من الرتبة N x NRHS.
!>

LDB

!>          LDB هو عدد صحيح (INTEGER)
!>          البعد الرائد للمصفوفة B. LDB >= max(1,N).
!>

INFO

!>          INFO هو عدد صحيح (INTEGER)
!>          = 0:  انتهاء ناجح
!>          < 0:  إذا كان INFO = -i، فإن المعلمة i كانت ذات قيمة غير قانونية
!>          > 0:  إذا كان INFO = i، فإن المحدد الرئيسي الرائد من الرتبة i
!>                ليس موجبًا، ولم يتم حساب الحل. لم يكتمل التحليل
!>                إلا إذا كان i = N.
!>
