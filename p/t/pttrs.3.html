subroutine cpttrs (character uplo, integer n, integer nrhs, real, dimension( * ) d, complex, dimension( * ) e, complex, dimension( ldb, * ) b, integer ldb, integer info)

CPTTRS

الغرض:

!>
!> CPTTRS يحل نظام ثلاثي الأقطار بالشكل
!>    A * X = B
!> باستخدام تحليل A = U**H*D*U أو A = L*D*L**H الذي تم حسابه بواسطة CPTTRF.
!> D هي مصفوفة قطرية محددة في المتجه D، U (أو L) هي مصفوفة ثنائية الأقطار وحدوية
!> والقطر الفائق (القطر الفرعي) محدد في
!> المتجه E، و X و B هي مصفوفتان N × NRHS.
!>

المعلمات

UPLO 
!>          UPLO هو CHARACTER*1
!>          يحدد شكل التحليل وما إذا كان
!>          المتجه E هو القطر الفائق للعامل ثنائي الأقطار العلوي
!>          U أو القطر الفرعي للعامل ثنائي الأقطار السفلي L.
!>          = 'U':  A = U**H*D*U, E هو القطر الفائق لـ U
!>          = 'L':  A = L*D*L**H, E هو القطر الفرعي لـ L
!>

N

!>          N هو INTEGER
!>          رتبة المصفوفة ثلاثية الأقطار A. N >= 0.
!>

NRHS

!>          NRHS هو INTEGER
!>          عدد الأطراف اليمنى، أي عدد أعمدة
!>          المصفوفة B. NRHS >= 0.
!>

D

!>          D هي مصفوفة REAL، بأبعاد (N)
!>          عناصر القطر N للمصفوفة القطرية D من
!>          التحليل A = U**H*D*U أو A = L*D*L**H.
!>

E

!>          E هي مصفوفة COMPLEX، بأبعاد (N-1)
!>          إذا كان UPLO = 'U'، فإن عناصر القطر الفائق (N-1) للعامل ثنائي الأقطار الوحدوي
!>          U من التحليل A = U**H*D*U.
!>          إذا كان UPLO = 'L'، فإن عناصر القطر الفرعي (N-1) للعامل ثنائي الأقطار الوحدوي
!>          L من التحليل A = L*D*L**H.
!>

B

!>          B هي مصفوفة COMPLEX، بأبعاد (LDB,NRHS)
!>          عند الدخول، متجهات الطرف الأيمن B لنظام
!>          المعادلات الخطية.
!>          عند الخروج، متجهات الحل X.
!>

LDB

!>          LDB هو INTEGER
!>          البعد الرئيسي للمصفوفة B. LDB >= max(1,N).
!>

INFO

!>          INFO هو INTEGER
!>          = 0: انتهاء ناجح
!>          < 0: إذا كان INFO = -k، فإن المعامل k كان له قيمة غير قانونية
!>
