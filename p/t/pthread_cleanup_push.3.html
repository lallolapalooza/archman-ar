pthread_cleanup_push(3) دليل وظائف المكتبة pthread_cleanup_push(3)

الاسم

pthread_cleanup_push, pthread_cleanup_pop - دفع وإخراج معالجات تنظيف إلغاء الخيط

المكتبة

مكتبة خيوط POSIX (libpthread, -lpthread)

الملخص

#include <pthread.h>
void pthread_cleanup_push(typeof(void (void *)) *routine, void *arg);
void pthread_cleanup_pop(int execute);

الوصف

تقوم هذه الوظائف بمعالجة مكدس معالجات تنظيف إلغاء الخيط الخاص بالخيط المستدعي. معالج التنظيف هو وظيفة يتم تنفيذها تلقائيًا عندما يتم إلغاء خيط (أو في ظروف مختلفة أخرى موصوفة أدناه)؛ قد تقوم، على سبيل المثال، بإلغاء قفل متبادل بحيث يصبح متاحًا للخيوط الأخرى في العملية.

تدفع وظيفة pthread_cleanup_push() routine إلى أعلى مكدس معالجات التنظيف. عندما يتم استدعاء routine لاحقًا، سيتم تمرير arg كوسيط لها.

تقوم وظيفة pthread_cleanup_pop() بإخراج المعالج في أعلى مكدس معالجات التنظيف، وتقوم بتنفيذه اختياريًا إذا كان execute غير صفري.

يتم إخراج معالج تنظيف الإلغاء وتنفيذه في الظروف التالية:

•
عندما يتم إلغاء خيط، يتم إخراج جميع معالجات التنظيف المكدسة وتنفيذها بترتيب عكسي للترتيب الذي تم دفعها به إلى المكدس.
•
عندما ينتهي خيط عن طريق استدعاء pthread_exit(3)، يتم تنفيذ جميع معالجات التنظيف كما هو موضح في النقطة السابقة. (لا يتم استدعاء معالجات التنظيف إذا انتهى الخيط عن طريق إجراء return من وظيفة بدء الخيط.)
•
عندما يستدعي خيط pthread_cleanup_pop() مع وسيط execute غير صفري، يتم إخراج المعالج الأعلى وتنفيذه.

يسمح POSIX.1 بتنفيذ pthread_cleanup_push() و pthread_cleanup_pop() كـ macros تتوسع إلى نص يحتوي على '{' و '}'، على التوالي. لهذا السبب، يجب على المتصل التأكد من أن استدعاءات هذه الوظائف مقترنة داخل نفس الوظيفة، وفي نفس مستوى التداخل المعجمي. (بمعنى آخر، يتم إنشاء معالج تنظيف فقط أثناء تنفيذ قسم محدد من التعليمات البرمجية.)

يؤدي استدعاء longjmp(3) (siglongjmp(3)) إلى نتائج غير محددة إذا تم إجراء أي استدعاء لـ pthread_cleanup_push() أو pthread_cleanup_pop() بدون الاستدعاء المطابق للزوج منذ ملء مخزن القفز بواسطة setjmp(3) (sigsetjmp(3)). وبالمثل، فإن استدعاء longjmp(3) (siglongjmp(3)) من داخل معالج تنظيف يؤدي إلى نتائج غير محددة ما لم يتم ملء مخزن القفز أيضًا بواسطة setjmp(3) (sigsetjmp(3)) داخل المعالج.

الإصدارات

على glibc، يتم تنفيذ وظيفتي pthread_cleanup_push() و pthread_cleanup_pop() كـ macros تتوسع إلى نص يحتوي على '{' و '}'، على التوالي. هذا يعني أن المتغيرات المعلنة ضمن نطاق الاستدعاءات المقترنة لهذه الوظائف ستكون مرئية فقط ضمن هذا النطاق.

يقول POSIX.1 أن تأثير استخدام return، break، continue، أو goto للخروج المبكر من كتلة محاطة بـ pthread_cleanup_push() و pthread_cleanup_pop() غير محدد. يجب على التطبيقات المتوافقة تجنب القيام بذلك.

المعايير

POSIX.1-2008.

التاريخ

POSIX.1-2001. glibc 2.0.

أمثلة

يقدم البرنامج أدناه مثالًا بسيطًا لاستخدام الوظائف الموصوفة في هذه الصفحة. يقوم البرنامج بإنشاء خيط ينفذ حلقة محاطة بـ pthread_cleanup_push() و pthread_cleanup_pop(). هذه الحلقة تزيد متغيرًا عامًا، cnt، مرة واحدة كل ثانية. اعتمادًا على وسائط سطر الأوامر المقدمة، يرسل الخيط الرئيسي طلب إلغاء إلى الخيط الآخر، أو يعين متغيرًا عامًا يتسبب في إنهاء الخيط الآخر لحلقته وإنهاءه بشكل طبيعي (عن طريق إجراء return).

في جلسة shell التالية، يرسل الخيط الرئيسي طلب إلغاء إلى الخيط الآخر:


$ ./a.out;
New thread started
cnt = 0
cnt = 1
Canceling thread
Called clean-up handler
Thread was canceled; cnt = 0

من الأعلى، نرى أن الخيط تم إلغاؤه، وأن معالج التنظيف تم استدعاؤه وقام بإعادة تعيين قيمة المتغير العام cnt إلى 0.

في التشغيل التالي، يعين البرنامج الرئيسي متغيرًا عامًا يتسبب في إنهاء الخيط الآخر بشكل طبيعي:


$ ./a.out x;
New thread started
cnt = 0
cnt = 1
Thread terminated normally; cnt = 2

من الأعلى، نرى أن معالج التنظيف لم يتم تنفيذه (لأن cleanup_pop_arg كان 0)، وبالتالي لم يتم إعادة تعيين قيمة cnt.

في التشغيل التالي، يعين البرنامج الرئيسي متغيرًا عامًا يتسبب في إنهاء الخيط الآخر بشكل طبيعي، ويوفر قيمة غير صفرية لـ cleanup_pop_arg:


$ ./a.out x 1;
New thread started
cnt = 0
cnt = 1
Called clean-up handler
Thread terminated normally; cnt = 0

في الأعلى، نرى أنه على الرغم من عدم إلغاء الخيط، تم تنفيذ معالج التنظيف، لأن الوسيط المعطى لـ pthread_cleanup_pop() كان غير صفري.

مصدر البرنامج

#include <err.h>
#include <errno.h>
#include <pthread.h>
#include <stdio.h>
#include <stdlib.h>
#include <sys/types.h>
#include <unistd.h>
static int done = 0;
static int cleanup_pop_arg = 0;
static int cnt = 0;
static void
cleanup_handler(void *arg)
{
    printf("Called clean-up handler\n");
    cnt = 0;
}
static void *
thread_start(void *arg)
{
    time_t curr;
    printf("New thread started\n");
    pthread_cleanup_push(cleanup_handler, NULL);
    curr = time(NULL);
    while (!done) {
        pthread_testcancel();           /* A cancelation point */
        if (curr < time(NULL)) {
            curr = time(NULL);
            printf("cnt = %d\n", cnt);  /* A cancelation point */
            cnt++;
        }
    }
    pthread_cleanup_pop(cleanup_pop_arg);
    return NULL;
}
int
main(int argc, char *argv[])
{
    pthread_t thr;
    int s;
    void *res;
    s = pthread_create(&thr, NULL, thread_start, NULL);
    if (s != 0)
        errc(EXIT_FAILURE, s, "pthread_create");
    sleep(2);           /* Allow new thread to run a while */
    if (argc > 1) {
        if (argc > 2)
            cleanup_pop_arg = atoi(argv[2]);
        done = 1;
    } else {
        printf("Canceling thread\n");
        s = pthread_cancel(thr);
        if (s != 0)
            errc(EXIT_FAILURE, s, "pthread_cancel");
    }
    s = pthread_join(thr, &res);
    if (s != 0)
        errc(EXIT_FAILURE, s, "pthread_join");
    if (res == PTHREAD_CANCELED)
        printf("Thread was canceled; cnt = %d\n", cnt);
    else
        printf("Thread terminated normally; cnt = %d\n", cnt);
    exit(EXIT_SUCCESS);
}

NAME

\n

pthread_cleanup_push, pthread_cleanup_pop - push and pop thread cleanup handlers

\n
\n\n
\n

LIBRARY

\n

Link with -pthread.

\n
\n\n
\n

SYNOPSIS

\n
#include <pthread.h>
\n
void pthread_cleanup_push(void (*routine)(void *), void *arg);
\n
void pthread_cleanup_pop(int execute);
\n
\n\n
\n

DESCRIPTION

\n

The pthread_cleanup_push() function installs a cancellation cleanup handler routine on the calling thread's stack. The function routine is a pointer to a function that takes a single argument of type void * and returns void. The argument passed to routine when it is invoked is arg.

\n

The pthread_cleanup_pop() function removes the handler that was most recently pushed onto the calling thread's stack by pthread_cleanup_push().

\n

When a thread is canceled (see pthread_cancel(3)), or when it calls pthread_exit(3), all cleanup handlers that have been pushed and not yet popped are executed. The handlers are executed in the reverse order in which they were pushed.

\n

If the argument execute to pthread_cleanup_pop() is non-zero, the handler removed by the call is also executed.

\n

The two functions are designed to be used as a pair. Typically, they surround a potentially long-running operation or a blocking call that might be canceled. The structure is:

\n
        pthread_cleanup_push(cleanup_handler, arg);\n        /* ... work that might be canceled ... */\n        pthread_cleanup_pop(execute_flag);\n
\n

The pthread_cleanup_push() and pthread_cleanup_pop() functions are implemented as macros that expand to code that calls underlying functions. This means that they must be used within the same lexical block (e.g., within the same function). You cannot use them across function calls, and they cannot be the last statement in a block if the block is within a compound statement (e.g., inside an if statement). This restriction is because the macros effectively add a goto statement.

\n

The pthread_cleanup_push() macro adds an entry to a linked list of cleanup handlers. The pthread_cleanup_pop() macro removes the entry.

\n

When a thread is canceled, the cancellation point is temporarily disabled, the handlers are executed, and then the cancellation point is re-enabled. The canceled thread then terminates.

\n

If pthread_cleanup_pop() is called with execute being zero, the handler is removed but not executed. This is useful for cases where the thread decides not to proceed with the operation that the handler was meant to clean up.

\n

The handler routine routine should be careful not to perform any operations that could lead to cancellation, such as calling pthread_testcancel(3) or other blocking functions, unless it has appropriately set up its own cleanup handlers.

\n
\n\n
\n

RETURN VALUE

\n

These functions do not return a value.

\n
\n\n
\n

ERRORS

\n

No errors are defined.

EXAMPLES

\n
#include <stdio.h>\n#include <stdlib.h>\n#include <pthread.h>\n#include <unistd.h>
\n
static void cleanup_handler(void *arg) {\n    printf(\"Cleanup handler called with arg: %s\\n\", (char *)arg);\n}
\n
static void *thread_func(void *arg) {\n    printf(\"Thread started.\\n\");\n    pthread_cleanup_push(cleanup_handler, \"First handler\");\n    printf(\"Cleanup handler pushed.\ Go to sleep.\n\");\n    sleep(5); /* Block here, might be canceled */\n    printf(\"Thread woke up. Pop handler.\n\");\n    pthread_cleanup_pop(1); /* Execute handler */\n    printf(\"Thread finished normally.\\n\");\n    return NULL;\n}
\n
int main() {\n    pthread_t tid;\n    int ret;\n\n    ret = pthread_create(&tid, NULL, thread_func, NULL);\n    if (ret != 0) {\n        perror(\"pthread_create\");\n        exit(EXIT_FAILURE);\n    }\n\n    sleep(1); /* Give thread time to start and push handler */\n\n    printf(\"Canceling thread...\n\");\n    ret = pthread_cancel(tid);\n    if (ret != 0) {\n        perror(\"pthread_cancel\");\n        exit(EXIT_FAILURE);\n    }\n\n    ret = pthread_join(tid, NULL);\n    if (ret != 0) {\n        perror(\"pthread_join\");\n        exit(EXIT_FAILURE);\n    }\n\n    printf(\"Thread joined. Main exiting.\\n\");\n    return 0;\n}
\n
A sample output would look like:\n\nThread started.\nCleanup handler pushed. Go to sleep.\nCanceling thread...\nCleanup handler called with arg: First handler\nThread joined. Main exiting.\n
\n

In the example above, the thread is created and immediately pushes a cleanup handler. It then calls sleep(), which is a cancellation point. After some time, the main thread cancels the created thread. When the cancellation request is processed (likely when sleep() returns or is interrupted), the cleanup handler is executed because pthread_cleanup_pop() was called with a non-zero argument (implicitly, as the thread terminates). The main thread then joins the canceled thread, which has already terminated.

\n
Another example:
\n
#include <stdio.h>\n#include <stdlib.h>\n#include <pthread.h>
\n
static void cleanup_resource(void *arg) {\n    printf(\"Releasing resource: %s\\n\", (char *)arg);\n    /* In a real application, this would free allocated memory, unlock mutexes, etc. */\n}
\n
static void *worker_thread(void *arg) {\n    printf(\"Worker thread started.\\n\");\n    pthread_cleanup_push(cleanup_resource, \"Resource A\");\n\n    /* Simulate acquiring a resource */\n    printf(\"Acquired Resource A.\\n\");\n\n    /* Maybe perform some operations */\n    printf(\"Performing work...\n\");\n    /* If the thread exits normally, we might not want to execute the cleanup handler */\n    /* For demonstration, let's pop without execution */\n    pthread_cleanup_pop(0); /* Pop the handler, but do not execute it */\n\n    printf(\"Worker thread finished normally (cleanup handler not executed automatically).\\n\");\n    return NULL;\n}\n\nint main() {\n    pthread_t tid;\n    int ret;\n\n    ret = pthread_create(&tid, NULL, worker_thread, NULL);\n    if (ret != 0) {\n        perror(\"pthread_create\");\n        exit(EXIT_FAILURE);\n    }\n\n    ret = pthread_join(tid, NULL);\n    if (ret != 0) {\n        perror(\"pthread_join\");\n        exit(EXIT_FAILURE);\n    }\n\n    printf(\"Main thread finished.\\n\");\n    return 0;\n}
\n
A sample output would look like:\n\nWorker thread started.\nAcquired Resource A.\nWorker thread finished normally (cleanup handler not executed automatically).\nMain thread finished.\n
\n

In this second example, the worker thread acquires a resource and pushes a cleanup handler. However, because the thread exits normally and pthread_cleanup_pop() is called with execute = 0, the cleanup handler is not executed automatically upon thread exit. If the thread were canceled while the handler was pushed, the handler would be executed.

\n

الاسم

\n

pthread_cleanup_push, pthread_cleanup_pop - دفع وإزالة معالجات تنظيف مؤشرات الترابط

\n
\n\n
\n

المكتبة

\n

اربط باستخدام -pthread.

\n
\n\n
\n

الملخص

\n
#include <pthread.h>
\n
void pthread_cleanup_push(void (*routine)(void *), void *arg);
\n
void pthread_cleanup_pop(int execute);
\n
\n\n
\n

الوصف

\n

pthread_cleanup_push() تقوم بتثبيت معالج تنظيف الإلغاء routine على مكدس مؤشر الترابط المستدعي. الدالة routine هي مؤشر إلى دالة تأخذ وسيطًا واحدًا من نوع void * وتعيد void. الوسيط الذي يتم تمريره إلى routine عند استدعائها هو arg.

\n

pthread_cleanup_pop() تقوم بإزالة المعالج الذي تم دفعه مؤخرًا إلى مكدس مؤشر الترابط المستدعي بواسطة pthread_cleanup_push().

\n

عندما يتم إلغاء مؤشر ترابط (انظر pthread_cancel(3))، أو عندما يستدعي pthread_exit(3)، يتم تنفيذ جميع معالجات التنظيف التي تم دفعها ولم يتم سحبها بعد. يتم تنفيذ المعالجات بترتيب عكسي للترتيب الذي تم دفعها به.

\n

إذا كان الوسيط execute إلى pthread_cleanup_pop() غير صفري، فسيتم أيضًا تنفيذ المعالج الذي تمت إزالته بواسطة الاستدعاء.

\n

تم تصميم الدالتين ليتم استخدامهما كزوج. عادةً، تحيط هاتان الدالتان بعملية قد تستغرق وقتًا طويلاً أو استدعاءً يحظر قد يتم إلغاؤه. الهيكل هو:

\n
        pthread_cleanup_push(cleanup_handler, arg);\n        /* ... العمل الذي قد يتم إلغاؤه ... */\n        pthread_cleanup_pop(execute_flag);\n
\n

تُنفذ pthread_cleanup_push() و pthread_cleanup_pop() كم وحدات ماكرو تتوسع إلى كود يستدعي دوال أساسية. هذا يعني أنه يجب استخدامها داخل نفس الكتلة المعجمية (على سبيل المثال، داخل نفس الدالة). لا يمكنك استخدامها عبر استدعاءات الدوال، ولا يمكن أن تكون البيان الأخير في كتلة إذا كانت الكتلة داخل بيان مركب (على سبيل المثال، داخل بيان if). هذا القيد يرجع إلى أن وحدات الماكرو تضيف فعليًا عبارة goto.

\n

pthread_cleanup_push() تضيف وحدة إدخال إلى قائمة مرتبطة بمعالجات التنظيف. pthread_cleanup_pop() تزيل الوحدة.

\n

عند إلغاء مؤشر ترابط، يتم تعطيل نقطة الإلغاء مؤقتًا، ويتم تنفيذ المعالجات، ثم يتم إعادة تمكين نقطة الإلغاء. ثم ينتهي مؤشر الترابط الملغى.

\n

إذا تم استدعاء pthread_cleanup_pop() مع execute يساوي صفر، يتم إزالة المعالج ولكن لا يتم تنفيذه. هذا مفيد للحالات التي يقرر فيها مؤشر الترابط عدم المتابعة في العملية التي كان المعالج مخصصًا لتنظيفها.

\n

يجب أن يكون المعالج routine حذرًا في عدم إجراء أي عمليات قد تؤدي إلى الإلغاء، مثل استدعاء pthread_testcancel(3) أو دوال أخرى تحظر، إلا إذا كان قد أعد معالجات التنظيف الخاصة به بشكل مناسب.

\n
\n\n
\n

التاريخ

\n

لم تتضمن معايير POSIX threads هذه الدوال في البداية. تمت إضافتها في POSIX.1-2001.

\n
\n\n
\n

أمثلة

\n
#include <stdio.h>\n#include <stdlib.h>\n#include <pthread.h>\n#include <unistd.h>
\n
static void cleanup_handler(void *arg) {\n    printf(\"تم استدعاء معالج التنظيف مع الوسيط: %s\\n\", (char *)arg);\n}
\n
static void *thread_func(void *arg) {\n    printf(\"بدأ مؤشر الترابط.\\n\");\n    pthread_cleanup_push(cleanup_handler, \"المعالج الأول\");\n    printf(\"تم دفع معالج التنظيف. الاستمرار في النوم.\n\");\n    sleep(5); /* احظر هنا، قد يتم إلغاء هذا */\n    printf(\"استيقظ مؤشر الترابط. سحب المعالج.\n\");\n    pthread_cleanup_pop(1); /* تنفيذ المعالج */\n    printf(\"انتهى مؤشر الترابط بشكل طبيعي.\\n\");\n    return NULL;\n}
\n
int main() {\n    pthread_t tid;\n    int ret;\n\n    ret = pthread_create(&tid, NULL, thread_func, NULL);\n    if (ret != 0) {\n        perror(\"pthread_create\");\n        exit(EXIT_FAILURE);\n    }\n\n    sleep(1); /* إعطاء وقت لمؤشر الترابط للبدء ودفع معالجه */\n\n    printf(\"الإلغاء قيد التقدم...\n\");\n    ret = pthread_cancel(tid);\n    if (ret != 0) {\n        perror(\"pthread_cancel\");\n        exit(EXIT_FAILURE);\n    }\n\n    ret = pthread_join(tid, NULL);\n    if (ret != 0) {\n        perror(\"pthread_join\");\n        exit(EXIT_FAILURE);\n    }\n\n    printf(\"تم الانضمام إلى مؤشر الترابط. إنهاء البرنامج الرئيسي.\n\");\n    return 0;\n}
\n
نتيجة نموذجية ستكون كالتالي:\n\nبدأ مؤشر الترابط.\nتم دفع معالج التنظيف. الاستمرار في النوم.\nالإلغاء قيد التقدم...\nتم استدعاء معالج التنظيف مع الوسيط: المعالج الأول\nتم الانضمام إلى مؤشر الترابط. إنهاء البرنامج الرئيسي.\n
\n

في المثال أعلاه، يتم إنشاء مؤشر ترابط ويدفع فورًا معالج تنظيف. ثم يستدعي sleep()، وهي نقطة إلغاء. بعد بعض الوقت، يلغي مؤشر الترابط الرئيسي مؤشر الترابط المنشأ. عندما تتم معالجة طلب الإلغاء (على الأرجح عند عودة sleep() أو مقاطعتها)، يتم تنفيذ معالج التنظيف لأنه تم استدعاء pthread_cleanup_pop() بوسيط غير صفري (ضمنياً، عند انتهاء مؤشر الترابط). ثم ينضم مؤشر الترابط الرئيسي إلى مؤشر الترابط الملغى، والذي يكون قد انتهى بالفعل.

\n
مثال آخر:
\n
#include <stdio.h>\n#include <stdlib.h>\n#include <pthread.h>
\n
static void cleanup_resource(void *arg) {\n    printf(\"تحرير المورد: %s\\n\", (char *)arg);\n    /* في تطبيق حقيقي، سيقوم هذا بتحرير الذاكرة المخصصة، وإلغاء قفل الأقفال، وما إلى ذلك. */\n}
\n
static void *worker_thread(void *arg) {\n    printf(\"بدأ مؤشر ترابط العامل.\\n\");\n    pthread_cleanup_push(cleanup_resource, \"المورد أ\");\n\n    /* محاكاة الحصول على مورد */\n    printf(\"تم الحصول على المورد أ.\\n\");\n\n    /* قد يتم إجراء بعض العمليات */\n    printf(\"جارٍ إجراء العمل...\n\");\n    /* إذا انتهى مؤشر الترابط بشكل طبيعي، فقد لا نرغب في تنفيذ معالج التنظيف */\n    /* للعرض التوضيحي، دعنا نسحب دون تنفيذ */\n    pthread_cleanup_pop(0); /* اسحب المعالج، ولكن لا تنفذه */\n\n    printf(\"انتهى مؤشر ترابط العامل بشكل طبيعي (لم يتم تنفيذ معالج التنظيف تلقائيًا).\\n\");\n    return NULL;\n}\n\nint main() {\n    pthread_t tid;\n    int ret;\n\n    ret = pthread_create(&tid, NULL, worker_thread, NULL);\n    if (ret != 0) {\n        perror(\"pthread_create\");\n        exit(EXIT_FAILURE);\n    }\n\n    ret = pthread_join(tid, NULL);\n    if (ret != 0) {\n        perror(\"pthread_join\");\n        exit(EXIT_FAILURE);\n    }\n\n    printf(\"انتهى مؤشر الترابط الرئيسي.\\n\");\n    return 0;\n}
\n
نتيجة نموذجية ستكون كالتالي:\n\nبدأ مؤشر ترابط العامل.\nتم الحصول على المورد أ.\nانتهى مؤشر ترابط العامل بشكل طبيعي (لم يتم تنفيذ معالج التنظيف تلقائيًا).\nانتهى مؤشر الترابط الرئيسي.\n
\n

في هذا المثال الثاني، يحصل مؤشر ترابط العامل على مورد ويدفع معالج تنظيف. ومع ذلك، نظرًا لأن مؤشر الترابط ينتهي بشكل طبيعي وتم استدعاء pthread_cleanup_pop() مع execute = 0، لم يتم تنفيذ معالج التنظيف تلقائيًا عند انتهاء مؤشر الترابط. إذا تم إلغاء مؤشر الترابط أثناء دفع المعالج، فسيتم تنفيذ المعالج.

\n
int main_cancel_example(int argc, char *argv[]) {\n    pthread_t tid;\n    int cnt = 0;\n    int ret;\n\n    ret = pthread_create(&tid, NULL, thread_func, NULL);\n    if (ret != 0) {\n        perror(\"pthread_create\");\n        exit(EXIT_FAILURE);\n    }\n\n    /* إعطاء مؤشر الترابط لحظة للبدء ودفع معالج التنظيف الخاص به */\n    sleep(1);\n\n    printf(\"الرئيسي: محاولة إلغاء مؤشر الترابط %lu\\n\", (unsigned long)tid);\n    ret = pthread_cancel(tid);\n    if (ret != 0) {\n        perror(\"pthread_cancel\");\n        exit(EXIT_FAILURE);\n    }\n\n    ret = pthread_join(tid, NULL);\n    if (ret != 0) {\n        perror(\"pthread_join\");\n        exit(EXIT_FAILURE);\n    }\n\n    printf(\"الرئيسي: تم الانضمام إلى مؤشر الترابط %lu.\\n\", (unsigned long)tid);\n    return cnt;\n    exit(EXIT_SUCCESS);\n}
