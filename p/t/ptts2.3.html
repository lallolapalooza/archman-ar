توثيق الوظيفة

روتين فرعي cptts2 (integer iuplo, integer n, integer nrhs, real, dimension( * ) d, complex, dimension( * ) e, complex, dimension( ldb, * ) b, integer ldb)

CPTTS2 يحل نظامًا ثلاثيًا على شكل AX=B باستخدام تحليل L D LH الذي تم حسابه بواسطة spttrf.

الغرض:

!>
!> CPTTS2 يحل نظامًا ثلاثيًا على شكل
!>    A * X = B
!> باستخدام التحليل A = U**H*D*U أو A = L*D*L**H الذي تم حسابه بواسطة CPTTRF.
!> D هي مصفوفة قطرية محددة في المتجه D، U (أو L) هي مصفوفة ثنائية قطرية موحدة
!> يكون قطرها العلوي (أو السفلي) محددًا في المتجه E، و X و B هما مصفوفتان N × NRHS.
!>

المعلمات

IUPLO 
!>          IUPLO هو عدد صحيح
!>          يحدد شكل التحليل وما إذا كان
!>          المتجه E هو القطر العلوي للعامل العلوي ثنائي القطر
!>          U أو القطر السفلي للعامل السفلي L.
!>          = 1:  A = U**H *D*U، E هو القطر العلوي لـ U
!>          = 0:  A = L*D*L**H، E هو القطر السفلي لـ L
!>

N

!>          N هو عدد صحيح
!>          رتبة المصفوفة الثلاثية A.  N >= 0.
!>

NRHS

!>          NRHS هو عدد صحيح
!>          عدد الأطراف اليمنى، أي عدد أعمدة
!>          المصفوفة B.  NRHS >= 0.
!>

D

!>          D هي مصفوفة REAL، البعد (N)
!>          عناصر القطر N للمصفوفة القطرية D من
!>          التحليل A = U**H *D*U أو A = L*D*L**H.
!>

E

!>          E هي مصفوفة COMPLEX، البعد (N-1)
!>          إذا كان IUPLO = 1، فإن عناصر القطر العلوي (N-1) للعامل ثنائي القطر الموحد
!>          U من التحليل A = U**H*D*U.
!>          إذا كان IUPLO = 0، فإن عناصر القطر السفلي (N-1) للعامل ثنائي القطر الموحد
!>          L من التحليل A = L*D*L**H.
!>

B

!>          B هي مصفوفة COMPLEX، البعد (LDB,NRHS)
!>          عند الدخول، متجهات الطرف الأيمن B لنظام
!>          المعادلات الخطية.
!>          عند الخروج، متجهات الحل، X.
!>

LDB

!>          LDB هو عدد صحيح
!>          البعد الرائد للمصفوفة B.  LDB >= max(1,N).
!>

DPTTS2 يحل نظامًا ثلاثيًا على شكل AX=B باستخدام تحليل L D LH الذي تم حسابه بواسطة spttrf.

الغرض:

!>
!> DPTTS2 يحل نظامًا ثلاثيًا على شكل
!>    A * X = B
!> باستخدام التحليل L*D*L**T للمصفوفة A الذي تم حسابه بواسطة DPTTRF.  D هي
!> مصفوفة قطرية محددة في المتجه D، L هي مصفوفة ثنائية قطرية موحدة
!> يكون قطرها السفلي محددًا في المتجه E، و X و B هما مصفوفتان N × NRHS.
!>

المعلمات

N 
!>          N هو عدد صحيح
!>          رتبة المصفوفة الثلاثية A.  N >= 0.
!>

NRHS

!>          NRHS هو عدد صحيح
!>          عدد الأطراف اليمنى، أي عدد أعمدة
!>          المصفوفة B.  NRHS >= 0.
!>

D

!>          D هي مصفوفة DOUBLE PRECISION، البعد (N)
!>          عناصر القطر N للمصفوفة القطرية D من
!>          التحليل L*D*L**T للمصفوفة A.
!>

E

!>          E هي مصفوفة DOUBLE PRECISION، البعد (N-1)
!>          عناصر القطر السفلي (N-1) للعامل ثنائي القطر الموحد
!>          L من التحليل L*D*L**T للمصفوفة A.  يمكن اعتبار E أيضًا
!>          كالقطر العلوي للعامل ثنائي القطر الموحد U من
!>          التحليل A = U**T*D*U.
!>

B

!>          B هي مصفوفة DOUBLE PRECISION، البعد (LDB,NRHS)
!>          عند الدخول، متجهات الطرف الأيمن B لنظام
!>          المعادلات الخطية.
!>          عند الخروج، متجهات الحل، X.
!>

LDB

!>          LDB هو عدد صحيح
!>          البعد الرائد للمصفوفة B.  LDB >= max(1,N).
!>
