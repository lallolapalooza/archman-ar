pthread_setname_np(3) دليل وظائف المكتبة pthread_setname_np(3)

الاسم

pthread_setname_np, pthread_getname_np - تعيين/الحصول على اسم خيط

المكتبة

مكتبة خيوط POSIX (libpthread, -lpthread)

الملخص

#define _GNU_SOURCE             /* انظر feature_test_macros(7) */
#include <pthread.h>
int pthread_setname_np(pthread_t thread, const char *name);
int pthread_getname_np(size_t size;
                       pthread_t thread, char name[size], size_t size);

الوصف

بشكل افتراضي، ترث جميع الخيوط التي تم إنشاؤها باستخدام pthread_create() اسم البرنامج. يمكن استخدام الدالة pthread_setname_np() لتعيين اسم فريد لخيط، والذي يمكن أن يكون مفيدًا لتصحيح الأخطاء في التطبيقات متعددة الخيوط. اسم الخيط هو سلسلة نصية ذات معنى بلغة C، وطولها مقيد بـ 16 حرفًا، بما في ذلك الحرف الصفري النهائي ('\0'). الوسيط thread يحدد الخيط الذي سيتم تغيير اسمه؛ name يحدد الاسم الجديد.

يمكن استخدام الدالة pthread_getname_np() لاسترداد اسم الخيط. الوسيط thread يحدد الخيط الذي سيتم استرداد اسمه. يتم استخدام المخزن المؤقت name لإرجاع اسم الخيط؛ size يحدد عدد البايتات المتاحة في name. يجب أن يكون المخزن المؤقت المحدد بواسطة name بطول 16 حرفًا على الأقل. سيكون اسم الخيط المُرجع في المخزن المؤقت الناتج منهيًا بحرف صفري.

قيمة الإرجاع

عند النجاح، تُرجع هذه الدوال 0؛ عند حدوث خطأ، تُرجع رقم خطأ غير صفري.

الأخطاء

يمكن أن تفشل الدالة pthread_setname_np() مع الخطأ التالي:

ERANGE
يتجاوز طول السلسلة المحددة التي يشير إليها name الحد المسموح به.

يمكن أن تفشل الدالة pthread_getname_np() مع الخطأ التالي:

ERANGE
المخزن المؤقت المحدد بواسطة name و size صغير جدًا للاحتفاظ باسم الخيط.

إذا فشلت أي من هاتين الدالتين في فتح /proc/self/task/tid/comm، فقد تفشل المكالمة بواحد من الأخطاء الموصوفة في open(2).

السمات

للحصول على شرح للمصطلحات المستخدمة في هذا القسم، انظر attributes(7).

الواجهة السمة القيمة
pthread_setname_np (), pthread_getname_np () سلامة الخيوط MT-Safe

المعايير

GNU؛ ومن هنا جاء اللاحقة "_np" (غير قابلة للنقل) في الأسماء.

التاريخ

glibc 2.12.

ملاحظات

تقوم pthread_setname_np() بالكتابة داخليًا إلى ملف comm الخاص بالخيط في نظام الملفات /proc: /proc/self/task/tid/comm. تقوم pthread_getname_np() باسترداده من نفس الموقع.

أمثلة

يوضح البرنامج أدناه استخدام pthread_setname_np() و pthread_getname_np().

تُظهر جلسة shell التالية تشغيلًا نموذجيًا للبرنامج:


$ ./a.out;
تم إنشاء خيط. الاسم الافتراضي هو: a.out
اسم الخيط بعد تعيينه هو THREADFOO.
^Z                           # تعليق البرنامج
[1]+  Stopped           ./a.out
$ ps H -C a.out -o 'pid tid cmd comm'
  PID   TID CMD                         COMMAND
 5990  5990 ./a.out                     a.out
 5990  5991 ./a.out                     THREADFOO
$ cat /proc/5990/task/5990/comm
a.out
$ cat /proc/5990/task/5991/comm
THREADFOO

مصدر البرنامج

#define _GNU_SOURCE
#include <err.h>
#include <errno.h>
#include <pthread.h>
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <unistd.h>
#define NAMELEN 16
static void *
threadfunc(void *parm)
{
    sleep(5);          // السماح للبرنامج الرئيسي بتعيين اسم الخيط
    return NULL;
}
int
main(int argc, char *argv[])
{
    pthread_t thread;
    int rc;
    char thread_name[NAMELEN];
    rc = pthread_create(&thread, NULL, threadfunc, NULL);
    if (rc != 0)
        errc(EXIT_FAILURE, rc, "pthread_create");
    rc = pthread_getname_np(thread, thread_name, NAMELEN);
    if (rc != 0)
        errc(EXIT_FAILURE, rc, "pthread_getname_np");
    printf("تم إنشاء خيط. الاسم الافتراضي هو: %s\n", thread_name);
    rc = pthread_setname_np(thread, (argc > 1) ? argv[1] : "THREADFOO");
    if (rc != 0)
        errc(EXIT_FAILURE, rc, "pthread_setname_np");
    sleep(2);
    rc = pthread_getname_np(thread, thread_name, NAMELEN);
    if (rc != 0)
        errc(EXIT_FAILURE, rc, "pthread_getname_np");
    printf("اسم الخيط بعد تعيينه هو %s.\n", thread_name);
    rc = pthread_join(thread, NULL);
    if (rc != 0)
        errc(EXIT_FAILURE, rc, "pthread_join");
    printf("تم الانتهاء\n");
    exit(EXIT_SUCCESS);
}
