pthread_key_create(3) دليل وظائف المكتبة pthread_key_create(3)

الاسم

pthread_key_create, pthread_key_delete, pthread_setspecific, pthread_getspecific - إدارة البيانات الخاصة بالخيط

الملخص

#include <pthread.h>
int pthread_key_create(pthread_key_t *key,
                       typeof(void (void *)) *destr_function;
int pthread_key_delete(pthread_key_t key);
int pthread_setspecific(pthread_key_t key, const void *pointer);
void * pthread_getspecific(pthread_key_t key);

الوصف

غالباً ما تحتاج البرامج إلى متغيرات عامة أو ثابتة لها قيم مختلفة في الخيوط المختلفة. نظراً لأن الخيوط تتشارك مساحة ذاكرة واحدة، لا يمكن تحقيق ذلك باستخدام المتغيرات العادية. البيانات الخاصة بالخيط هي إجابة POSIX threads لهذه الحاجة.

يمتلك كل خيط كتلة ذاكرة خاصة به، وهي منطقة البيانات الخاصة بالخيط، أو منطقة TSD اختصارًا. يتم فهرسة هذه المنطقة بواسطة مفاتيح TSD. تربط منطقة TSD قيمًا من النوع void * بمفاتيح TSD. مفاتيح TSD مشتركة بين جميع الخيوط، ولكن القيمة المرتبطة بمفتاح TSD معين يمكن أن تكون مختلفة في كل خيط.

للتوضيح، يمكن النظر إلى مناطق TSD على أنها مصفوفات من مؤشرات void *، ومفاتيح TSD كفهارس عددية لهذه المصفوفات، وقيمة مفتاح TSD كقيمة العنصر المقابل في المصفوفة في الخيط المستدعي.

عند إنشاء خيط، تقوم منطقة TSD الخاصة به بربط NULL مبدئياً بجميع المفاتيح.

تقوم pthread_key_create() بتخصيص مفتاح TSD جديد. يتم تخزين المفتاح في الموقع الذي يشير إليه key. يوجد حد أقصى يبلغ PTHREAD_KEYS_MAX لعدد المفاتيح المخصصة في وقت معين. القيمة المرتبطة مبدئياً بالمفتاح المُعاد هي NULL في جميع الخيوط التي يتم تنفيذها حالياً.

المعامل destr_function، إذا لم يكن NULL، يحدد دالة مدمرة مرتبطة بالمفتاح. عند إنهاء خيط عبر pthread_exit() أو عن طريق الإلغاء، يتم استدعاء destr_function مع الوسيطات هي القيمة المرتبطة بالمفتاح في ذلك الخيط. لا يتم استدعاء destr_function إذا كانت تلك القيمة NULL. الترتيب الذي يتم به استدعاء دوال المدمرات عند إنهاء الخيط غير محدد.

قبل استدعاء دالة المدمرة، يتم ربط القيمة NULL بالمفتاح في الخيط الحالي. ومع ذلك، قد تقوم دالة المدمرة بإعادة ربط قيم غير NULL بهذا المفتاح أو بمفتاح آخر. للتعامل مع هذا، إذا ظلت هناك بعض القيم غير NULL مع دوال مدمرة مرتبطة بها بعد استدعاء جميع الدوال المدمرة لجميع القيم غير NULL، يتم تكرار العملية. يتوقف تطبيق glibc عن العملية بعد PTHREAD_DESTRUCTOR_ITERATIONS تكرارًا، حتى لو بقيت بعض القيم غير NULL مع واصفات مرتبطة بها. قد تدخل التطبيقات الأخرى في حلقة لا نهائية.

تقوم pthread_key_delete() بإلغاء تخصيص مفتاح TSD. لا تتحقق مما إذا كانت القيم غير NULL مرتبطة بهذا المفتاح في الخيوط التي يتم تنفيذها حالياً، ولا تستدعي الدالة المدمرة المرتبطة بالمفتاح.

تقوم pthread_setspecific() بتغيير القيمة المرتبطة بـ key في الخيط المستدعي، وتخزين pointer المعطى بدلاً من ذلك.

تقوم pthread_getspecific() بإرجاع القيمة المرتبطة حاليًا بـ key في الخيط المستدعي.

قيمة الإرجاع

تقوم pthread_key_create() و pthread_key_delete() و pthread_setspecific() بإرجاع 0 عند النجاح ورمز خطأ غير صفري عند الفشل. في حالة النجاح، تقوم pthread_key_create() بتخزين المفتاح المخصص حديثًا في الموقع الذي يشير إليه وسيطتها key.

تقوم pthread_getspecific() بإرجاع القيمة المرتبطة بـ key عند النجاح، و NULL عند الفشل.

الأخطاء

ترجع pthread_key_create() رمز الخطأ التالي عند الفشل:

EAGAIN
تم تخصيص مفاتيح PTHREAD_KEYS_MAX بالفعل.

ترجع pthread_key_delete() و pthread_setspecific() رمز الخطأ التالي عند الفشل:

EINVAL
key ليس مفتاح TSD صالحًا ومخصصًا.

ترجع pthread_getspecific() NULL إذا كان key ليس مفتاح TSD صالحًا ومخصصًا.

انظر أيضًا

pthread_create(3), pthread_exit(3), pthread_testcancel(3).

مثال

يقوم جزء الكود التالي بتخصيص مصفوفة خاصة بالخيط بحجم 100 حرف، مع إعادة تخصيص تلقائية عند إنهاء الخيط:

/* مفتاح لمخزن الخيط المؤقت الخاص */
static pthread_key_t buffer_key;
/* تهيئة لمرة واحدة للمفتاح */
static pthread_once_t buffer_key_once = PTHREAD_ONCE_INIT;
/* تخصيص مخزن الخيط المؤقت الخاص */
void buffer_alloc(void)
{
  pthread_once(&buffer_key_once, buffer_key_alloc);
  pthread_setspecific(buffer_key, malloc(100));
}
/* إرجاع مخزن الخيط المؤقت الخاص */
char * get_buffer(void)
{
  return (char *) pthread_getspecific(buffer_key);
}
/* تخصيص المفتاح */
static void buffer_key_alloc()
{
  pthread_key_create(&buffer_key, buffer_destroy);
}
/* تحرير مخزن الخيط المؤقت الخاص */
static void buffer_destroy(void * buf)
{
  free(buf);
}
