PDF__API2(3pm)

PDF::API2(3) وثائق حزم Perl المساهمة PDF::API2(3)

الاسم

PDF::API2 - إنشاء، تعديل، وفحص ملفات PDF

الملخص

use PDF::API2;
# إنشاء ملف PDF فارغ
$pdf = PDF::API2->new();
# فتح ملف PDF موجود
$pdf = PDF::API2->open('some.pdf');
# إضافة صفحة فارغة
$page = $pdf->page();
# الحصول على صفحة موجودة
$page = $pdf->open_page($page_number);
# تعيين حجم الصفحة
$page->size('Letter');
# إضافة خط نوع قائم على النظام إلى PDF
$font = $pdf->font('Helvetica-Bold');
# إضافة خط نوع TrueType خارجي إلى PDF
$font = $pdf->font('/path/to/font.ttf');
# إضافة نص إلى الصفحة
$text = $page->text();
$text->font($font, 20);
$text->position(200, 700);
$text->text('Hello World!');
# حفظ ملف PDF
$pdf->save('/path/to/new.pdf');

طرق الإدخال/الإخراج

new

my $pdf = PDF::API2->new(%options);

إنشاء ملف PDF جديد.

الخيارات التالية متاحة:

  • file

    إذا كنت تعرف اسم الملف الذي سيتم تخزين ملف PDF فيه وترغب في فتحه للكتابة منذ البداية، فيمكنك تضمينه هنا. "file" يمكن أن يكون أيضًا مقبض ملف.

  • compress

    افتراضيًا، سيتم ضغط معظم محتوى PDF لتوفير المساحة. لإيقاف هذه الوظيفة (مفيد عادةً للاختبار أو التصحيح)، قم بتعيين "compress" إلى 0.

open

my $pdf = PDF::API2->open('/path/to/file.pdf', %options);

فتح ملف PDF موجود.

الخيار التالي متاح:

•
compress

افتراضيًا، سيتم ضغط معظم محتوى PDF لتوفير المساحة. لإيقاف هذه الوظيفة (مفيد عادةً للاختبار أو التصحيح)، قم بتعيين "compress" إلى 0.

save

$pdf->save('/path/to/file.pdf');

اكتب ملف PDF إلى القرص وأغلق الملف. اسم الملف اختياري إذا تم تحديده عند فتح أو إنشاء ملف PDF.

كأثر جانبي، يتم إزالة بنية المستند من الذاكرة عند حفظ الملف، لذا لن يكون مفيدًا بعد ذلك.

close

$pdf->close();

إغلاق ملف مفتوح (إذا كان ذا صلة) وإزالة بنية الكائن من الذاكرة.

يحتوي PDF::API2 على مراجع دائرية، لذا فإن هذه المكالمة ضرورية في العمليات الطويلة لتجنب استنفاد الذاكرة.

سيتم استدعاء هذه المكالمة تلقائيًا عند حفظ ملف PDF أو تحويله إلى سلسلة نصية. يجب عليك فقط استدعائها بشكل صريح إذا كنت تقرأ ملفات PDF ولا تكتبها.

from_string

my $pdf = PDF::API2->from_string($pdf_string, %options);

قراءة مستند PDF موجود في سلسلة نصية.

الخيار التالي متاح:

•
compress

افتراضيًا، سيتم ضغط معظم محتوى PDF لتوفير المساحة. لإيقاف هذه الوظيفة (مفيد عادةً للاختبار أو التصحيح)، قم بتعيين "compress" إلى 0.

to_string

my $string = $pdf->to_string();

إرجاع مستند PDF كسلسلة نصية.

كأثر جانبي، يتم إزالة بنية المستند من الذاكرة عند إنشاء السلسلة النصية، لذا لن يكون مفيدًا بعد ذلك.

طرق البيانات الوصفية

title

$title = $pdf->title();
$pdf = $pdf->title($title);

الحصول/تعيين/مسح عنوان المستند.

author

$author = $pdf->author();
$pdf = $pdf->author($author);

الحصول/تعيين/مسح اسم منشئ المستند.

subject

$subject = $pdf->subject();
$pdf = $pdf->subject($subject);

الحصول/تعيين/مسح موضوع المستند.

keywords

$keywords = $pdf->keywords();
$pdf = $pdf->keywords($keywords);

الحصول/تعيين/مسح سلسلة نصية مفصولة بمسافات للكلمات الرئيسية المرتبطة بالمستند.

creator

$creator = $pdf->creator();
$pdf = $pdf->creator($creator);

الحصول/تعيين/مسح اسم المنتج الذي أنشأ المستند قبل تحويله إلى PDF.

producer

$producer = $pdf->producer();
$pdf = $pdf->producer($producer);

الحصول/تعيين/مسح اسم المنتج الذي حول المستند الأصلي إلى PDF.

يقوم PDF::API2 بملء هذا الحقل عند إنشاء ملف PDF.

created

$date = $pdf->created();
$pdf = $pdf->created($date);

الحصول/تعيين/مسح تاريخ إنشاء المستند.

تنسيق التاريخ هو "D:YYYYMMDDHHmmSSOHH'mm"، حيث "D:" هو بادئة ثابتة تحدد السلسلة كنص تاريخ PDF. يمكن اقتطاع التاريخ في أي وقت بعد السنة. "O" هو واحد من "+", "-", أو "Z"، مع "HH'mm" يمثل فرقًا بالساعات عن التوقيت العالمي المنسق.

عند تعيين التاريخ، سيتم إضافة "D:" تلقائيًا إذا لم يتم تضمينه.

modified

$date = $pdf->modified();
$pdf = $pdf->modified($date);

الحصول/تعيين/مسح تاريخ تعديل المستند. تنسيق التاريخ هو نفسه كما في "created" سابقًا.

info_metadata

# الحصول على جميع المفاتيح والقيم
%info = $pdf->info_metadata();
# الحصول على قيمة مفتاح
$value = $pdf->info_metadata($key);
# تعيين قيمة مفتاح
$pdf = $pdf->info_metadata($key, $value);

الحصول/تعيين/مسح مفتاح في قاموس البيانات الوصفية للمستند. المفاتيح القياسية (title, author, إلخ) لها واجهات وصول خاصة بها، لذا فإن هذه الطريقة مخصصة بشكل أساسي للتفاعل مع البيانات الوصفية المخصصة.

مرر "undef" كقيمة لحذف المفتاح من القاموس.

xml_metadata

$xml = $pdf->xml_metadata();
$pdf = $pdf->xml_metadata($xml);

الحصول/تعيين تدفق البيانات الوصفية XML للمستند.

version

$version = $pdf->version($new_version);

الحصول/تعيين إصدار PDF (على سبيل المثال، 1.4).

is_encrypted

$boolean = $pdf->is_encrypted();

يعيد صحيحًا إذا كان ملف PDF المفتوح مشفرًا.

طرق الميزات التفاعلية

outline

$outline = $pdf->outline();

ينشئ (إذا لزم الأمر) ويعيد شجرة المخطط التفصيلي للمستند، والتي تُعرف أيضًا باسم إشاراتها المرجعية أو جدول المحتويات، اعتمادًا على قارئ PDF.

لفحص أو تعديل شجرة المخطط التفصيلي، راجع PDF::API2::Outline.

open_action

$pdf = $pdf->open_action($page, $location, @args);

تعيين الوجهة في ملف PDF التي يجب أن تكون

كائن خط اصطناعي جديد. انظر PDF::API2::Resource::Font::SynFont للتفاصيل.

standard_fonts

@names = $pdf->standard_fonts()

يعيد أسماء الخطوط القياسية الـ 14 (المضمنة). انظر PDF::API2::Resource::Font::CoreFont للتفاصيل.

is_standard_font

$boolean = PDF::API2->is_standard_font($name);

يعيد صحيحًا إذا كانت $name تطابقًا تامًا، حساسًا لحالة الأحرف، لأحد أسماء الخطوط القياسية.

font_path

@directories = PDF::API2->font_path()

يعيد قائمة الدلائل التي سيتم البحث فيها (بالترتيب) بالإضافة إلى الدليل الحالي عند إضافة خط إلى ملف PDF دون تضمين المسار الكامل لملف الخط.

add_to_font_path

@directories = PDF::API2->add_to_font_path('/my/fonts', '/path/to/fonts');

يضيف دليلًا واحدًا أو أكثر إلى قائمة المسارات التي سيتم البحث فيها عن ملفات الخطوط.

يعيد مسار البحث عن الخطوط.

set_font_path

@directories = PDF::API2->set_font_path('/my/fonts', '/path/to/fonts');

يستبدل مسار البحث عن الخطوط الحالي. يجب أن يكون هذا ضروريًا فقط إذا كنت بحاجة إلى إزالة دليل من المسار لسبب ما، أو إذا كنت بحاجة إلى إعادة ترتيب القائمة.

يعيد مسار البحث عن الخطوط.

طرق الرسومات

image

$object = $pdf->image($file, %options);

يستورد نوع صورة مدعوم ويعيد كائنًا يمكن وضعه كجزء من محتوى الصفحة:

my $pdf = PDF::API2->new();
my $page = $pdf->page();
my $image = $pdf->image('/path/to/image.jpg');
$page->object($image, 100, 100);
$pdf->save('sample.pdf');

$file يمكن أن يكون اسم ملف، أو مقبض ملف، أو كائن GD::Image.

راجع "object" في PDF::API2::Content للحصول على تفاصيل حول وضع الصور على صفحة بمجرد استيرادها.

يتم اكتشاف تنسيق الصورة عادةً تلقائيًا بناءً على امتداد الملف. عند تمرير مقبض ملف، سيتم اكتشاف تنسيقات الصور GIF و JPEG و PNG بناءً على رأس الملف.

إذا كان الملف يحتوي على امتداد غير نمطي أو كان مقبض الملف لنوع آخر من الصور، يمكنك تعيين خيار "format" إلى أحد الأنواع المدعومة: "gif"، "jpeg"، "png"، "pnm"، أو "tiff".

ملاحظة: تمر صور PNG التي تتضمن قناة ألفا (شفافية) بعملية بطيئة نسبيًا لتقسيم الصورة إلى مكونات RGB وألفا منفصلة كما هو مطلوب للصور في ملفات PDF. إذا كنت تواجه مشكلات في الأداء، فقم بتثبيت PDF::API2::XS أو Image::PNG::Libpng لتسريع هذه العملية بعشرة أضعاف؛ سيتم استخدام أي من الوحدتين تلقائيًا إذا كانت متاحة.

barcode

$object = $pdf->barcode($format, $code, %options);

يولد ويعيد رمز شريطي يمكن وضعه كجزء من محتوى الصفحة:

my $pdf = PDF::API2->new();
my $page = $pdf->page();
my $barcode = $pdf->barcode('ean13', '0123456789012');
$page->object($barcode, 100, 100);
my $qr_code = $pdf->barcode('qr', 'http://www.example.com');
$page->object($qr_code, 100, 300, 144 / $qr_code->width())
$pdf->save('sample.pdf');

$format يمكن أن يكون أحد "codabar"، "code128"، "code39" (المعروف أيضًا باسم 3 من 9)، "ean128"، "ean13"، "itf" (المعروف أيضًا باسم interleaved 2 of 5)، أو "qr".

$code هو القيمة التي سيتم ترميزها. أحرف البداية والنهاية مطلوبة فقط عندما تكون ثابتة (على سبيل المثال، لـ Codabar).

الخيارات التالية متاحة:

  • bar_width

    عرض أصغر شريط أو مساحة بالنقاط (72 نقطة = 1 بوصة).

    إذا كنت تتبع مواصفات تعطي عرض الشريط بالمل (أجزاء من الألف من البوصة)، استخدم هذا التحويل: "$points = $mils / 1000 * 72".

  • bar_height

    الارتفاع الأساسي للرمز الشريطي بالنقاط.

  • bar_extend

    إذا كان موجودًا وقابلًا للتطبيق، فسيتم تمديد الأشرطة للأحرف غير القابلة للطباعة (مثل أحرف البداية والنهاية) لأسفل بهذا العدد من النقاط، وسيتم عرض الأحرف القابلة للطباعة أسفل أشرطتها المقابلة.

    هذا ممكّن افتراضيًا لرموز EAN-13.

  • caption

    إذا كان موجودًا، فسيتم طباعة هذه القيمة، في المنتصف، أسفل الرمز الشريطي، ويجب أن تكون تمثيلاً قابلاً للقراءة البشرية للرمز الشريطي. يتم تجاهل هذا الخيار لرموز QR.

  • font

    كائن خط (تم إنشاؤه بواسطة "font") سيتم استخدامه لطباعة التسمية، أو الأحرف القابلة للطباعة عندما يتم تعيين "bar_extend".

    سيتم استخدام Helvetica افتراضيًا.

  • font_size

    حجم الخط المستخدم لطباعة التسمية أو الأحرف القابلة للطباعة.

    سيتم حساب الافتراضي بناءً على حجم الرمز الشريطي، إذا تم تعيين "bar_extend"، أو 10 بخلاف ذلك.

  • quiet_zone

    هامش، بالنقاط، سيتم وضعه قبل الحواف اليسرى والسفلية للرمز الشريطي (بما في ذلك التسمية، إذا وجدت). يُستخدم هذا لمساعدة الماسحات الضوئية على معرفة مكان بداية الرمز الشريطي ونهايته.

    الافتراضي هو عرض حرف واحد مشفر، أو أربعة مربعات لرموز QR.

  • bar_overflow

    يقلل العرض الأفقي للأشرطة بهذا المقدار بالنقاط للحساب عند انتشار الحبر عند الطباعة. يتم تجاهل هذا الخيار لرموز QR.

    الافتراضي هو 0.01 نقطة.

  • color

    ارسم الأشرطة باستخدام هذا اللون، والذي يمكن أن يكون أي قيمة مقبولة بواسطة "fillcolor" في PDF::API2::Content.

    الافتراضي هو الأسود.

تحتوي رموز QR على خيارات إضافية لتخصيص مستوى تصحيح الخطأ والإعدادات المتخصصة الأخرى.

colorspace

$colorspace = $pdf->colorspace($type, @arguments);

يمكن إضافة مساحات الألوان إلى ملف PDF إما للتحكم تحديدًا في إخراج اللون على جهاز معين (ألوان البقع، ألوان الجهاز) أو لتوفير المساحة عن طريق الحد من الألوان المتاحة إلى لوحة ألوان محددة (لوحة الألوان الآمنة للويب، ملف ACT).

بمجرد إضافتها إلى ملف PDF، يمكن استخدامها بدلاً من رموز HEX العادية أو الألوان المسماة:

my $pdf = PDF::API2->new();
my $page = $pdf->page();
my $content = $page->graphics();
# إضافة مساحات الألوان للون بقعة ولوحة ألوان الويب الآمنة
my $spot = $pdf->colorspace('spot', 'PANTONE Red 032 C', '#EF3340');
my $web = $pdf->colorspace('web');
# ملء باستخدام اللون البقعة مع تغطية 100%
$content->fill_color($spot, 1.0);
# رسم باستخدام اللون الأول من لوحة الويب الآمنة
$content->stroke_color($web, 0);
# إضافة مستطيل إلى الصفحة
$content->rectangle(100, 100, 200, 200);
$content->paint();
$pdf->save('sample.pdf');

أنواع مساحات الألوان التالية مدعومة

  • spot 
    my $spot = $pdf->colorspace('spot', $tint, $alt_color);

    تُستخدم ألوان البقع لإرشاد جهاز (عادةً طابعة) لاستخدام أو محاكاة لون حبر معين ($tint) لأجزاء من المستند. يتم توفير $alt_color للأجهزة (مثل عارضات PDF) التي لا تعرف كيفية إنتاج اللون المسمى. يمكن أن يكون إما تقريبًا للون بتنسيق RGB أو CMYK أو HSV، أو لونًا مختلفًا تمامًا (على سبيل المثال، 100% ماجنتا، %0F00) لتوضيح ما إذا كان لون البقعة لا يُستخدم كما هو متوقع.

  • web 
    my $web = $pdf->colorspace('web');

    لوحة الألوان الآمنة للويب هي مجموعة تاريخية من الألوان التي تم استخدامها عندما كانت العديد من أجهزة العرض تدعم 256 لونًا فقط.

  • act 
    my $act = $pdf->colorspace('act', $filename);

    يوفر ملف جدول ألوان Adobe (ACT) لوحة ألوان مخصصة يمكن الإشارة إليها بواسطة أوامر رسم الرسومات والنصوص في PDF.

  • device 
    my $devicen = $pdf->colorspace('device', @colorspaces);

    تسمح مساحة الألوان الخاصة بالجهاز بإخراج ألوان دقيقة على جهاز معين (عادةً ماكينة طباعة)، متجاوزة التفسير اللوني العادي الذي تجريه معالجات الصور النقطية (RIPs).

    مساحات الألوان الخاصة بالجهاز مطلوبة أيضًا إذا كنت ترغب في مزج ألوان البقع:

    my $pdf = PDF::API2->new();
my $page = $pdf->page();
my $content = $page->graphics();
# إنشاء مساحة ألوان جهاز بلونين
my $yellow = $pdf->colorspace('spot', 'Yellow', '%00F0');
my $spot = $pdf->colorspace('spot', 'PANTONE Red 032 C', '#EF3340');
my $device = $pdf->colorspace('device', $yellow, $spot);
# ملء باستخدام مزيج من 25% أصفر و 75% لون بقعة
$content->fill_color($device, 0.25, 0.75);
# رسم باستخدام 100% لون بقعة
$content->stroke_color($device, 0, 1);
# إضافة مستطيل إلى الصفحة
$content->rectangle(100, 100, 200, 200);
$content->paint();
$pdf->save('sample.pdf');

egstate

$resource = $pdf->egstate();
