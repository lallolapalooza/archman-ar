PDF__API2__Outline(3pm)

الحزمة: perl-pdf-api2

التراخيص: LGPL-2.1-or-later

PDF::API2::Outline(3) وثائق Perl المقدمة من المستخدم PDF::API2::Outline(3)

الاسم

PDF::API2::Outline - إدارة مخططات PDF (تُعرف أيضًا باسم الإشارات المرجعية)

الملخص

# الحصول/إنشاء شجرة المخطط ذات المستوى الأعلى
my $outline = $pdf->outline();
# إضافة إدخال
my $item = $outline->outline();
$item->title('الصفحة الأولى');
$item->destination($pdf->open_page(1));

الأساليب

فحص شجرة المخطط

has_children

my $boolean = $outline->has_children();

إرجاع صحيح إذا كان عنصر المخطط الحالي لديه عناصر فرعية.

count

my $integer = $outline->count();

إرجاع عدد الأحفاد المرئيين عند فتح (توسيع) عنصر المخطط الحالي.

first

my $child = $outline->first();

إرجاع أول ابن لمستوى المخطط الحالي، إذا كان موجودًا.

last

my $child = $outline->last();

إرجاع آخر ابن لمستوى المخطط الحالي، إذا كان موجودًا.

parent

my $parent = $outline->parent();

إرجاع الوالد للعنصر الحالي، إذا لم يكن في المستوى الأعلى لشجرة المخطط.

prev

my $sibling = $outline->prev();

إرجاع العنصر السابق لمستوى شجرة المخطط الحالي.

next

my $sibling = $outline->next();

إرجاع العنصر التالي لمستوى شجرة المخطط الحالي.

تعديل شجرة المخطط

outline

my $child = $outline->outline();

إضافة عنصر مخطط في نهاية قائمة الأبناء للمخطط الحالي.

insert_after

my $sibling = $outline->insert_after();

إضافة عنصر مخطط مباشرة بعد العنصر الحالي.

insert_before

$sibling = $outline->insert_before();

إضافة عنصر مخطط مباشرة قبل العنصر الحالي.

delete

$outline->delete();

إزالة عنصر المخطط الحالي من شجرة المخطط. إذا كان للعنصر أي أبناء، فسيتم حذفهم فعليًا لأنهم لن يعودوا مرتبطين.

is_open

# الحصول
my $boolean = $outline->is_open();
# تعيين
my $outline = $outline->is_open($boolean);

الحصول/تعيين ما إذا كان المخطط موسعًا أو مطويًا.

تعيين سمات المخطط

title

# الحصول
my $title = $outline->title();
# تعيين
$outline = $outline->title($text);

الحصول/تعيين عنوان عنصر المخطط.

destination

$outline = $outline->destination($destination, $location, @args);

تعيين صفحة الوجهة والموضع الاختياري للمخطط. $location و @args كما هو محدد في "destination" في PDF::API2::NamedDestination.

$destination يمكن أن يكون اختياريًا اسم وجهة مسماة معرفة في مكان آخر.

uri

$outline = $outline->uri($uri);

تشغيل عنوان URI - عادةً صفحة ويب - عند تفعيل عنصر المخطط.

launch

$outline->launch($file);

تشغيل تطبيق أو ملف عند تفعيل عنصر المخطط.

pdf

$outline = $outline->pdf($filename, $page_number, $location, @args);

فتح ملف PDF آخر إلى رقم صفحة معين (الصفحة الأولى هي صفر، وهي أيضًا الافتراضية). يمكن وضع الصفحة اختياريًا في موضع معين إذا تم تعيين $location و @args - انظر "destination" في PDF::API2::NamedDestination للإعدادات الممكنة.
