PDF__API2__Content(3pm)

الحزمة: perl-pdf-api2

التراخيص: LGPL-2.1-or-later

PDF::API2::Content(3) توثيق Perl من مساهمي المستخدم PDF::API2::Content(3)

الاسم

PDF::API2::Content - طرق لإضافة رسومات ونصوص إلى ملف PDF

الملخص

# ابدأ بصفحة PDF (جديدة أو مفتوحة)
my $pdf = PDF::API2->new();
my $page = $pdf->page();
# أضف كائن محتوى جديد
my $content = $page->graphics();
my $content = $page->text();
# ثم استدعِ الطرق أدناه لإضافة رسومات ونصوص إلى الصفحة.

تحويلات الإحداثيات

تغير الطرق في هذا القسم نظام الإحداثيات لكائن المحتوى الحالي نسبةً إلى بقية المستند.

التغييرات على نظام الإحداثيات تؤثر فقط على المسارات أو النصوص اللاحقة.

يؤدي استدعاء أي من الطرق في هذا القسم إلى إعادة تعيين نظام الإحداثيات قبل تطبيق تغييراته، ما لم يتم تعيين الخيار "relative".

translate

$content = $content->translate($x, $y);

ينقل نقطة الأصل على طول المحورين x و y.

rotate

$content = $content->rotate($degrees);

يدور نظام الإحداثيات عكس اتجاه عقارب الساعة.

استخدم قيمة سالبة للدوران في اتجاه عقارب الساعة.

scale

$content = $content->scale($x, $y);

يقوم بتغيير حجم (تمديد) أنظمة الإحداثيات على طول المحورين x و y. القيمة 1 لأي من $x أو $y تمثل 100% من الحجم (أي لا يوجد تغيير).

skew

$content = $content->skew($a, $b);

يشوه نظام الإحداثيات بمقدار $a درجة (عكس اتجاه عقارب الساعة) من المحور x و $b درجة (في اتجاه عقارب الساعة) من المحور y.

transform

$content = $content->transform(
    translate => [$x, $y],
    rotate    => $degrees,
    scale     => [$x, $y],
    skew      => [$a, $b],
    repeat    => $boolean,
);

يجري تحويلات إحداثيات متعددة، بالترتيب الموصى به من مواصفات PDF (نقل، دوران، تحجيم، ثم تشويه). الخيارات المحذوفة لن تتغير.

إذا كانت "repeat" صحيحة وإذا كانت هذه ليست أول مكالمة لطريقة تحويل، فسيتم إجراء التحويل السابق مرة أخرى، مع تعديله بأي وسائط أخرى مقدمة.

matrix

$graphics = $graphics->matrix($a, $b, $c, $d, $e, $f);
($a, $b, $c, $d, $e, $f) = $text->matrix($a, $b, $c, $d, $e, $f);

يضبط مصفوفة التحويل الحالية يدويًا. ما لم يكن لديك حاجة معينة لإدخال التحويلات يدويًا، يجب عليك استخدام طريقة "transform" بدلاً من ذلك.

تختلف القيمة المرجعة بناءً على ما إذا كان المتصل كائن محتوى رسومات أو كائن محتوى نصي.

حالة الرسومات

save

$content = $content->save();

يحفظ حالة الرسومات الحالية على مكدس.

restore

$content = $content->restore();

يستعيد أحدث حالة رسومات محفوظة، ويزيلها من المكدس.

line_width

$content = $content->line_width($points);

يضبط عرض الخط بالنقاط.

line_cap

$content = $content->line_cap($style);

يضبط الشكل الذي سيتم استخدامه في نهايات المسارات المفتوحة (والشرطات، إن وجدت) عند رسمها.

  • "butt" أو 0 = Butt Cap، الافتراضي

    ينتهي الخط عند نهاية المسار، دون أي بروز.

  • "round" أو 1 = Round Cap)

    يتم رسم قوس حول نهاية المسار بقطر يساوي عرض الخط، ويتم ملؤه.

  • "square" أو 2 = Projecting Square Cap

    يستمر الخط بعد نهاية المسار بنصف عرض الخط.

line_join

$content = $content->line_join($style);

يضبط نمط الاتصال الذي سيتم استخدامه عند الزوايا للمسار.

  • "miter" أو 0 = Miter Join، الافتراضي

    تمتد الحواف الخارجية للخط حتى تلتقي، حتى الحد المحدد أدناه. إذا تم تجاوز الحد، يتم استخدام اتصال مشطوف بدلاً من ذلك.

  • "round" أو 1 = Round Join

    يتم رسم دائرة بقطر يساوي عرض الخط حول نقطة الزاوية، مما ينتج عنه زاوية مستديرة.

  • "bevel" أو 2 = Bevel Join

    يتم رسم مثلث لملء الفجوة بين الخطين.

miter_limit

$content = $content->miter_limit($ratio);

يضبط حد الشطف عندما يكون نمط اتصال الخط هو Miter Join.

الـ $ratio هو الحد الأقصى لطول الشطف (من الزاوية الداخلية إلى الخارجية) مقسومًا على عرض الخط. أي شطف أعلى من هذه النسبة سيتم تحويله إلى اتصال مشطوف.

لا يوجد حد شطف افتراضي موثق.

line_dash_pattern

# خط صلب
$content = $content->line_dash_pattern();
# أطوال خطوط وفجوات متساوية
$content = $content->line_dash_pattern($length);
# أطوال خطوط وفجوات محددة
$content = $content->line_dash_pattern($line1, $gap1, $line2, $gap2, ...);
# إزاحة نقطة البداية
$content = $content->line_dash_pattern(
    pattern => [$line1, $gap1, $line2, $gap2, ...],
    offset => $points,
);

يضبط نمط الخط المتقطع.

إذا تم استدعاؤه بدون أي وسائط، فسيتم رسم خط صلب.

إذا تم استدعاؤه بوسيط واحد، فإن الشرطات والفجوات سيكون لها أطوال متساوية.

إذا تم استدعاؤه بوسيطين أو أكثر، فإن الوسائط تمثل أطوال متناوبة للشرطات والفجوات.

إذا تم استدعاؤه باستخدام جدول من الوسائط، يمكن تعيين مرحلة الشرطة، والتي تحدد المسافة داخل النمط التي يبدأ منها الشرطة.

flatness_tolerance

$content = $content->flatness_tolerance($tolerance);

يضبط الحد الأقصى للمسافة بالبكسل في الجهاز بين المسار الصحيح رياضيًا لمنحنى وتقريب مبني من شرائح خطية.

الـ $tolerance هو عدد صحيح بين 0 و 100، حيث يمثل 0 تسامح التسطيح الافتراضي للجهاز.

egstate

$content = $content->egstate($object);

يضيف كائن PDF::API2::Resource::ExtGState يحتوي على مجموعة من معلمات حالة الرسومات.

إنشاء المسار (الرسم)

لاحظ أن المسارات لن تظهر حتى يتم استدعاء طريقة رسم مسار ("stroke"، "fill"، أو "paint").

move

$content = $content->move($x, $y);

يبدأ مسارًا جديدًا عند الإحداثيات المحددة.

line

$content = $content->line($x, $y);

يمد المسار في خط من الإحداثيات الحالية إلى الإحداثيات المحددة.

hline

$content = $content->hline($x);
>

يمتد المسار في خط أفقي من الموضع الحالي إلى الإحداثي x المحدد.

vline

$content = $content->vline($x);

يمتد المسار في خط عمودي من الموضع الحالي إلى الإحداثي y المحدد.

polyline

$content = $content->polyline($x1, $y1, $x2, $y2, ...);

يمتد المسار من الموضع الحالي في خط واحد أو أكثر من الخطوط المستقيمة.

curve

$content = $content->curve($cx1, $cy1, $cx2, $cy2, $x, $y);

يمتد المسار في منحنى من النقطة الحالية إلى "($x, $y)"، باستخدام النقطتين المحددتين لإنشاء منحنى بيزييه تكعيبي.

spline

$content = $content->spline($cx1, $cy1, $x, $y);

يمتد المسار في منحنى من النقطة الحالية إلى "($x, $y)"، باستخدام النقطتين المحددتين لإنشاء منحنى متشعب.

arc

$content = $content->arc($x, $y, $major, $minor, $a, $b);

يمتد المسار على طول قوس من قطع ناقص مركزه عند "[$x, $y]". تمثل $major و $minor محاور القطع الناقص، ويتحرك القوس من درجة $a إلى درجة $b.

close

$content = $content->close();

يغلق المسار الحالي عن طريق تمديد خط من الموضع الحالي إلى موضع البداية.

إنشاء الأشكال (رسم)

التالي هي طرق ملائمة لرسم المسارات المغلقة.

لاحظ أن الأشكال لن تظهر حتى يتم استدعاء طريقة رسم المسار ("stroke" أو "fill" أو "paint").

rectangle

$content = $content->rectangle($x1, $y1, $x2, $y2);

ينشئ مسارًا جديدًا على شكل مستطيل، بين النقطتين "[$x1, $y1]" و "[$x2, $y2]".

circle

$content = $content->circle($x, $y, $radius);

ينشئ مسارًا دائريًا جديدًا مركزه عند "[$x, $y]" بنصف القطر المحدد.

ellipse

$content = $content->ellipse($x, $y, $major, $minor);

ينشئ مسارًا بيضاويًا جديدًا مركزه عند "[$x, $y]" بمحوريه الرئيسيين والثانويين المحددين.

pie

$content = $content->pie($x, $y, $major, $minor, $a, $b);

ينشئ مسارًا جديدًا على شكل إسفين من قطع ناقص مركزه عند "[$x, $y]" بمحوريه الرئيسيين والثانويين المحددين، ويمتد من الدرجة $a إلى الدرجة $b.

طلاء المسار (رسم)

stroke_color

$content->stroke_color($color, @arguments);

يحدد لون الخط، والذي يكون أسود بشكل افتراضي.

# استخدم لونًا مسمى
$content->stroke_color('blue');
# استخدم لون RGB (ابدأ بـ '#')
$content->stroke_color('#FF0000');
# استخدم لون CMYK (ابدأ بـ '%')
$content->stroke_color('%FF000000');
# استخدم لونًا مميزًا بتغطية 100%.
my $spot = $pdf->colorspace('spot', 'PANTONE Red 032 C', '#EF3340');
$content->stroke_color($spot, 1.0);

يمكن أن تحتوي ألوان RGB و CMYK على قيم أحادية البايت، أو ثنائية البايت، أو ثلاثية البايت، أو رباعية البايت لكل لون، اعتمادًا على مستوى الدقة المطلوب. على سبيل المثال، يمكن إعطاء اللون السماوي كـ %F000 أو %FFFF000000000000.

fill_color

$content->fill_color($color, @arguments);

يحدد لون التعبئة، والذي يكون أسود بشكل افتراضي. الوسائط هي نفسها كما في "stroke_color".

stroke

$content = $content->stroke();

يرسم المسار الحالي.

fill

$content = $content->fill(rule => $rule);

يملأ المسار الحالي.

$rule يصف المناطق التي يتم ملؤها عندما يتقاطع المسار مع نفسه.

  • nonzero (افتراضي)

    استخدم قاعدة عدد اللفات غير الصفرية. يميل هذا إلى تعني أن المساحة بأكملها المحاطة بالمسار يتم ملؤها، مع بعض الاستثناءات اعتمادًا على اتجاه المسار.

  • even-odd

    استخدم قاعدة الأعداد الفردية والزوجية. يميل هذا إلى تعني أن وجود التعبئة يتناوب في كل مرة يتم فيها تقاطع المسار.

انظر مواصفات PDF 1.7 القسم 8.5.3.3، الملء، لمزيد من التفاصيل.

paint

$content = $content->paint(rule => $rule);

يملأ ويرسم المسار الحالي. $rule كما هو موصوف في "fill".

clip

$content = $content->clip(rule => $rule);

يعدل مسار القطع الحالي (الذي يكون في البداية الصفحة بأكملها) عن طريق تقاطعه مع المسار الحالي بعد أمر رسم المسار التالي. $rule كما هو موصوف في "fill".

end

$content = $content->end();

ينهي المسار الحالي دون ملء أو رسم. يتم استخدامه في المقام الأول كأثر جانبي لتغيير مسار القطع الحالي.

كائنات خارجية

object

$content = $content->object($object, $x, $y, $scale_x, $scale_y);

يضع صورة أو كائنًا خارجيًا آخر (يُعرف أيضًا باسم XObject) على الصفحة في الموقع المحدد.

إذا تم حذف $x و $y، فسيتم وضع الكائن عند "[0, 0]".

بالنسبة للصور، تمثل $scale_x و $scale_y عرض وطول الصورة على الصفحة بالنقاط. إذا تم حذف $scale_x، فسيتم افتراض 72 بكسل لكل بوصة. إذا تم حذف $scale_y، فسيتم تغيير حجم الصورة بشكل متناسب بناءً على أبعاد الصورة.

بالنسبة للكائنات الخارجية الأخرى، يكون المقياس مضاعفًا، حيث يمثل 1 (الافتراضي) 100% (أي لا تغيير).

إذا تم إجراء تحويلات إحداثيات (انظر تحويلات الإحداثيات أعلاه)، فسيكون الموضع والمقياس نسبيين للإحداثيات المحدثة.

إذا لم تكن هناك حاجة لتحويلات إحداثيات، يمكن استدعاء هذه الطريقة مباشرة من كائن PDF::API2::Page بدلاً من ذلك.

حالة النص

يتم تطبيق جميع المعلمات التالية التي تأخذ حجمًا قبل إجراء أي تحجيم، لذلك لا تحتاج إلى تعديل القيم لمواجهة التحجيم.

font

$content = $content->font($font, $size);

يحدد الخط وحجم الخط. $font هو كائن تم إنشاؤه عن طريق استدعاء "font" في PDF::API2 لإضافة الخط إلى المستند.

my $pdf = PDF::API2->new();
my $page = $pdf->page();
my $text = $page->text();
my $font = $pdf->font('Helvetica');
$text->font($font, 24);
$text->position(72, 720);
$text->text('Hello, World!');
$pdf->save('sample.pdf');

character_spacing

$spacing = $content->character_spacing($spacing);

يحدد المسافة بين الأحرف. هذه المسافة صفر في البداية.

word_spacing

$spacing = $content->word_spacing($spacing);

يحدد المسافة بين الكلمات. هذه المسافة صفر في البداية (أي عرض المسافة فقط).

قد تؤثر المسافة بين الكلمات فقط على الخطوط البسيطة والخطوط المركبة حيث يكون حرف المسافة رمزًا أحادي البايت. هذا قيد لمواصفات PDF على الأقل اعتبارًا من الإصدار 1.7 (انظر القسم 9.3.3). من الممكن أن يدعم إصدار أحدث من المواصفات المسافة بين الكلمات في الخطوط التي تستخدم رموزًا متعددة البايت.

hscale

$scale = $content->hscale($scale);

يحدد/يحصل على نسبة قياس النص الأفقي. أدخل مقياسًا أكبر من 100 لتمديد النص، وأقل من 100 لضغط النص، أو 100 لتعطيل أي قياس موجود.

leading

$leading = $content->leading($leading);

يحدد/يحصل على مسافة الأسطر، وهي المسافة بين خطوط الأساس. هذه المسافة صفر في البداية (أي سيتم طباعة الأسطر فوق بعضها البعض).

render

$mode = $content->render($mode);

يحدد وضع عرض النص.

  • 0 = ملء النص
  • 1 = رسم خطوط النص (مخطط تفصيلي)
  • 2 = ملء النص، ثم رسمه
  • 3 = لا ملء ولا رسم للنص (غير مرئي)
  • 4 = ملء النص وإضافته إلى المسار للقطع
  • 5 = رسم خطوط النص وإضافته إلى المسار للقطع
  • 6 = ملء النص، ثم رسمه، وإضافته إلى المسار للقطع

rise

$distance = $content->rise($distance);

يضبط خط الأساس للأعلى أو للأسفل من موقعه الحالي. هذا صفر مبدئيًا.

استخدم هذا لإنشاء الأحرف العلوية أو السفلية (عادةً مع ضبط لحجم الخط أيضًا).

TEXT PLACEMENT

position

# Set
$content = $content->position($x, $y);
# Get
($x, $y) = $content->position();

إذا تم استدعاؤه مع وسيطات، فإنه يتحرك إلى بداية سطر النص، مزاحًا بمقدار $x و $y.

إذا لم يتم استدعاؤه مع وسيطات، فإنه يُرجع الموضع الحالي للمؤشر (قبل تأثير أي طريقة لتحويل الإحداثيات).

crlf

$content = $content->crlf();

ينتقل إلى بداية السطر التالي، بناءً على قيمة "leading".

إذا لم يتم تعريف "leading"، فسيتم استخدام قيمة افتراضية بنسبة 120٪ من حجم الخط.

text

my $width = $content->text($text, %options);

يضع النص على الصفحة. يُرجع عرض النص بالنقاط.

خيارات:

  • align

    أحد خيارات "left" (افتراضي)، "center"، أو "right". سيتم وضع النص بحيث يبدأ، أو يكون في المنتصف، أو ينتهي عند الموضع الحالي للنص، اعتمادًا على قيمة "align".

    في كل حالة، سينتقل الموضع إلى نهاية النص.

  • indent

    يحاذي النص لأسفل بالنقاط.

    إذا لم يكن "align" هو "left"، فسيتم تجاهل هذا الإعداد.

  • underline

    يضع خطًا تحت النص. يمكن أن تكون القيمة واحدة مما يلي:

  • auto

    يحدد مسافة الخط تحت النص بناءً على الخط وحجم الخط.

  • $distance

    يضبط مسافة الخط تحت النص يدويًا بالنقاط. القيمة الموجبة تحرك الخط للأسفل.

  • [$distance, $thickness, ...]

    يضبط كل من مسافة الخط تحت النص وسماكة الخط، وكلاهما بالنقاط.

    كرر هذه الوسيطات لتضمين خطوط سفلية متعددة.

text_justified

my $width = $content->text_justified($text, $width, %options);

مثل "text"، يملأ العرض المحدد عن طريق ضبط المسافات بين الكلمات.

paragraph

# Scalar context
$overflow_text = $content->paragraph($text, $width, $height, %options);
# Array context
($overflow, $height) = $content->paragraph($text, $width, $height, %options);

يملأ المستطيل بالنص المقدم، إذا أمكن.

في سياق المصفوفة، يُرجع النص المتبقي (إن وجد) من النص الموضوع والارتفاع المتبقي (غير المستخدم). في سياق العدد، يُرجع النص المتبقي (إن وجد).

تتبع المسافات بين الأسطر قيمة "leading"، إذا تم تعريفها، أو 120٪ من حجم الخط افتراضيًا.

خيارات

  • align

    يحدد محاذاة كل سطر من أسطر النص. يمكن أن تكون "left" (افتراضي)، "center"، "right"، أو "justified".

  • align-last

    يحدد محاذاة السطر الأخير من النص المحاذي. يمكن أن تكون "left" (افتراضي)، "center"، "right"، أو "justified".

  • underline

    كما هو موضح في "text".

text_width

my $width = $content->text_width($line, %overrides);

يُرجع عرض سطر نصي بناءً على الحالة الحالية لسمات المحتوى. يمكن تجاوز هذه بشكل اختياري:

my $width = $content->text_width($line,
    font => $font,
    size => $size,
    character_spacing => $spacing,
    word_spacing => $spacing,
    hscale => $scale,
);

انظر "MIGRATION" في PDF::API2 للحصول على مقدمة.

transform(%hyphen_prefixed_options);
يزيل الواصلات من أسماء الخيارات (يصبح "-translate" "translate"، وما إلى ذلك).
transform_rel
يتم استبداله بـ "transform"، مع تعيين القيمة "repeat" إلى true. يزيل الواصلات من أسماء الخيارات الأخرى.

يتم استبداله بـ "line_dash_pattern". يزيل الواصلات من أسماء الخيارات. يعيد تسمية "-shift" إلى "offset".
flatness
يتم استبداله بـ "flatness_tolerance".
poly
يتم استبداله بـ "move" (أول وسيطين) و "polyline" (الوسيطات المتبقية).
endpath
يتم استبداله بـ "end".
rect
يتم استبداله بـ "rectangle"، مع تحويل الوسيطات $w (الثالث) و $h (الرابع) إلى قيم الزاوية اليمنى العلوية: 
# Old
$content->rect($x, $y, $w, $h);
# New
$content->rectangle($x, $y, $x + $w, $y + $h);
rectxy
