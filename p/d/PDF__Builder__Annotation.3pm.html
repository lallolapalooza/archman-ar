PDF__Builder__Annotation(3pm)

الحزمة: perl-pdf-builder

التراخيص: LGPL-2.1-or-later

PDF::Builder::Annotation(3) وثائق Perl للمساهمين PDF::Builder::Annotation(3)

الاسم

PDF::Builder::Annotation - إضافة تعليقات توضيحية إلى ملف PDF

ترث من PDF::Builder::Basic::PDF::Dict

ملخص

my $pdf = PDF::Builder->new();
my $font = $pdf->font('Helvetica');
my $page1 = $pdf->page();
my $page2 = $pdf->page();
my $content = $page1->text();
my $message = 'اذهب إلى الصفحة 2';
my $size = 18;
$content->distance(1 * 72, 9 * 72);
$content->font($font, $size);
$content->text($message);
my $annotation = $page1->annotation();
my $width = $content->text_width($message);
$annotation->rect(1 * 72, 9 * 72, 1 * 72 + $width, 9 * 72 + $size);
$annotation->link($page2);
$pdf->save('sample.pdf');

الطرق

لاحظ أن معالجة التعليقات التوضيحية قد تختلف من قارئ إلى آخر. قد تكون مجموعة الأيقونات المتاحة أكبر أو أصغر من تلك المذكورة هنا، وبعض القراء ينشطون التعليق التوضيحي بنقرة واحدة بالماوس، بينما يتطلب البعض الآخر نقرة مزدوجة. قد لا تكون جميع الميزات المتوفرة هنا متاحة في جميع قراء PDF.

جديد

$annotation = PDF::Builder::Annotation->new()
يعيد كائن تعليق توضيحي (يتم استدعاؤه من $page->annotation()).

عادةً *لا* يكون من الضروري استدعاء هذه الطريقة بشكل صريح (انظر الأمثلة).

أنواع التعليقات التوضيحية

link

$annotation->link($page, %opts)
يعرّف التعليق التوضيحي كصفحة تشغيل مع الصفحة $page (داخل المستند *الحالي*) وخيارات %opts (rect، border، color، fit: انظر الأوصاف أدناه).

لاحظ أن $page *ليس* رقم صفحة بسيطًا، بل هو بنية صفحة مثل "$pdf->openpage(page_number)"، *أو* وجهة مسماة معرفة في مكان آخر.

pdf, pdfile, pdf_file

$annotation->pdf($pdffile, $page_number, %opts)
يعرّف التعليق التوضيحي كملف PDF بمسار ملف $pdffile، في الصفحة $page_number، وخيارات %opts (rect، border، color، fit: انظر الأوصاف أدناه). يختلف هذا عن استدعاء "link" في أن الهدف موجود في ملف PDF مختلف، وليس المستند الحالي.

$page_number هو رقم الصفحة الفعلي، بدءًا من 1: 1، 2، ...

أسماء بديلة: "pdfile" و "pdf_file"

في الأصل، تم تسمية هذه الطريقة "pdfile"، ثم "pdf_file" ولكن تغييرًا حديثًا في PDF::API2 جعلها "pdf". للتوافق، تم تغييرها إلى "pdf"، مع بقاء "pdfile" و "pdf_file" كألقاب.

launch, file

$annotation->launch($file, %opts)
يعرّف التعليق التوضيحي كملف تشغيل بمسار ملف $file (ملف محلي) وخيارات %opts (rect، border، color: انظر الأوصاف أدناه). *كيف* يتم عرض الملف يعتمد على نظام التشغيل، ونوع الملف، والتكوين أو التعيين المحلي.

اسم بديل: "file"

في الأصل، تم تسمية هذه الطريقة "file"، ولكن تغييرًا حديثًا في PDF::API2 جعلها "launch". للتوافق، تم تغييرها إلى "launch"، مع بقاء "file" كلقب.

uri, url

$annotation->uri($url, %opts)
يعرّف التعليق التوضيحي كعنوان URL تشغيل بعنوان URL $url وخيارات %opts (rect، border، color: انظر الأوصاف أدناه). عادةً ما يتم فتح هذه الصفحة في متصفح، وقد تكون بعيدة.

اسم بديل: "url"

في الأصل، تم تسمية هذه الطريقة "url"، ولكن تغييرًا حديثًا في PDF::API2 جعلها "uri". للتوافق، تم تغييرها إلى "uri"، مع بقاء "url" كلقب.

text

$annotation->text($text, %opts)
يعرّف التعليق التوضيحي كملاحظة نصية بالمحتوى $text وخيارات %opts (rect، color، text، open: انظر الأوصاف أدناه). قد يتضمن $text أسطرًا جديدة \n لعدة أسطر. يتم تجاهل خيار border، نظرًا لأنه يتم استخدام *أيقونة*.

الخيار "text" هو سلسلة تسمية النافذة المنبثقة، وليس ليتم الخلط بينه وبين $text الرئيسي.

تظهر الأيقونة في الزاوية العلوية اليسرى من مستطيل الاختيار "rect"، ويتم تحديد منطقة النقر النشطة بواسطة الأيقونة (إنها *ليست* مساوية للمستطيل). حجم الأيقونة ثابت، ولون الملء الخاص بها تم تحديده بواسطة "color".

خيارات إضافية:

icon => name_string
icon => reference
حدد **الأيقونة** المراد استخدامها. الافتراضي هو خاص بالقارئ (عادةً "Note")، وقد يتم تعريف أيقونات أخرى بواسطة القارئ. يُفترض أن "Comment"، "Key"، "Help"، "NewParagraph"، "Paragraph"، و "Insert" متاحة أيضًا على جميع قراء PDF. لاحظ أن *حالة* الاسم يجب أن تتطابق تمامًا. الأيقونة ذات حجم ثابت. أي إدخال قاموس AP سيتجاوز إعداد الأيقونة.

يمكن تمرير *مرجع* إلى أيقونة بدلاً من اسم.

opacity => value
حدد شفافية (عدم الشفافية، الكثافة) للأيقونة. تتراوح هذه القيمة من 0.0 (شفافة) إلى 1.0 (معتمة تمامًا)، وتنطبق على كل من المخطط الخارجي ولون الملء. الافتراضي هو 1.0.

markup

$annotation->markup($text, $PointList, $highlight, %opts)
يعرّف التعليق التوضيحي كملاحظة نصية بالمحتوى $text وخيارات %opts (color، text، open، opacity: انظر الأوصاف أدناه). قد يتضمن $text أسطرًا جديدة \n لعدة أسطر.

"text" هو سلسلة تسمية النافذة المنبثقة، وليس ليتم الخلط بينه وبين $text الرئيسي.

لا توجد أيقونة. بدلاً من ذلك، يتم تمييز النص المشروح بواسطة $PointList بإحدى الطرق الأربع المحددة بواسطة $highlight.

$PointList => [ 8n numbers ]
يتم إعطاء مجموعة واحدة أو أكثر من الإحداثيات الرقمية، والتي تحدد الشكل الرباعي (عادةً مستطيل) حول النص المراد تمييزه وجعله قابلاً للاختيار (النقر، لإظهار نص التعليق التوضيحي). هذه هي أربع مجموعات من إحداثيات "x,y"، معطاة (للنص من اليسار إلى اليمين) كحد علوي للأعلى اليسار إلى الأعلى اليمين ثم الحد السفلي للأسفل اليسار إلى الأسفل اليمين. **لاحظ أن هذا يختلف عما هو موثق (بشكل خاطئ) في مواصفات PDF!** من المهم إعطاء الإحداثيات بهذا الترتيب.

يمكن إعطاء مجموعات متعددة من أركان الشكل الرباعي، مثل النص المميز الذي يلتف إلى سطر (أو أسطر) جديدة. الحد الأدنى هو مجموعة واحدة (8 أرقام). أي إدخال قاموس AP سيتجاوز إعداد $PointList. أخيرًا، يتم إنشاء مستطيل الاختيار "Rect" *مباشرة خارج* الصندوق المحيط المحدب المحدد بواسطة $PointList.

$highlight => 'string'
التأثيرات التمييزية التالية مسموح بها. يجب أن تكون السلسلة *مكتوبة بشكل مطابق تمامًا* ومن حيث حالة الأحرف كما تم إعطاؤها:
الألوان المسماة (مثل 'black') غير مدعومة في الوقت الحالي.

color => [ ] أو ليس 1 أو 3 أو 4 أرقام 0.0-1.0
سيتم استخدام اللون الرمادي المتوسط (قيمة 0.5) إذا تم إعطاء لون غير صالح.
color => [ g ]
إذا كانت g بين 0.0 (أسود) و 1.0 (أبيض)، فسيكون لون التعبئة رماديًا.
color => [ r, g, b ]
إذا كانت r (الأحمر) و g (الأخضر) و b (الأزرق) كلها بين 0.0 و 1.0، فسيكون لون التعبئة هو تدرج RGB المحدد. [ 0، 0، 0 ] هو الأسود، [ 1، 1، 0 ] هو الأصفر، و [ 1، 1، 1 ] هو الأبيض.
color => [ c, m, y, k ]
إذا كانت c (السماوي) و m (الأرجواني) و y (الأصفر) و k (الأسود) كلها بين 0.0 و 1.0، فسيكون لون التعبئة هو تدرج CMYK المحدد. [ 0، 0، 0، 0 ] هو الأبيض، [ 1، 0، 1، 0 ] هو الأخضر، و [ 1، 1، 1، 1 ] هو الأسود.

نص سلسلة

text => string
حدد عنوان نصي اختياري للتعليق. يظهر هذا النص أو التعليق فقط كعنوان في النافذة المنبثقة التي تحتوي على الملف أو النص.
