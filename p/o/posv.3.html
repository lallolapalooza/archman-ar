روتين cposv (character uplo, integer n, integer nrhs, complex, dimension( lda, * ) a, integer lda, complex, dimension( ldb, * ) b, integer ldb, integer info)

CPOSV يحسب حل نظام المعادلات الخطية المعقدة A * X = B،

الغرض:

!>
!> CPOSV يحسب حل نظام المعادلات الخطية المعقدة
!>    A * X = B,
!> حيث A مصفوفة هيرميتية موجبة محددة من النوع N × N و X و B
!> مصفوفتان من النوع N × NRHS.
!>
!> يتم استخدام تحليل تشوليسكي لتحليل A بالشكل
!>    A = U**H* U,  إذا كان UPLO = 'U', أو
!>    A = L * L**H,  إذا كان UPLO = 'L',
!> حيث U مصفوفة مثلثة عليا و L مصفوفة مثلثة سفلى.
!> ثم يتم استخدام الصيغة المحللة لـ A لحل نظام المعادلات A * X = B.
!>

المعلمات

UPLO 
!>          UPLO هو CHARACTER*1
!>          = 'U':  يتم تخزين المثلث العلوي لـ A؛
!>          = 'L':  يتم تخزين المثلث السفلي لـ A.
!>

N

!>          N هو INTEGER
!>          عدد المعادلات الخطية، أي ترتيب
!>          المصفوفة A.  N >= 0.
!>

NRHS

!>          NRHS هو INTEGER
!>          عدد الأطراف اليمنى، أي عدد الأعمدة
!>          للمصفوفة B.  NRHS >= 0.
!>

A

!>          A هي مصفوفة COMPLEX، البعد (LDA,N)
!>          عند الدخول، المصفوفة الهيرميتية A.  إذا كان UPLO = 'U'، فإن
!>          الجزء العلوي المثلثي الرئيسي N × N من A يحتوي على الجزء
!>          المثلثي العلوي للمصفوفة A، والجزء المثلثي السفلي الصارم
!>          من A غير مشار إليه.  إذا كان UPLO = 'L'، فإن
!>          الجزء السفلي المثلثي الرئيسي N × N من A يحتوي على الجزء
!>          المثلثي السفلي للمصفوفة A، والجزء المثلثي العلوي الصارم
!>          من A غير مشار إليه.
!>
!>          عند الخروج، إذا كان INFO = 0، فإن العامل U أو L من تحليل تشوليسكي
!>          A = U**H*U أو A = L*L**H.
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرئيسي للمصفوفة A.  LDA >= max(1,N).
!>

B

!>          B هي مصفوفة COMPLEX، البعد (LDB,NRHS)
!>          عند الدخول، مصفوفة الطرف الأيمن B من النوع N × NRHS.
!>          عند الخروج، إذا كان INFO = 0، فإن مصفوفة الحل X من النوع N × NRHS.
!>

LDB

!>          LDB هو INTEGER
!>          البعد الرئيسي للمصفوفة B.  LDB >= max(1,N).
!>

INFO

!>          INFO هو INTEGER
!>          = 0:  خروج ناجح
!>          < 0:  إذا كان INFO = -i، فإن الوسيط i كان له قيمة غير قانونية
!>          > 0:  إذا كان INFO = i، فإن المحدد الرئيسي الرائد من الرتبة i
!>                لـ A ليس موجبًا، لذلك لم يكن من الممكن إكمال التحليل،
!>                ولم يتم حساب الحل.
!>
