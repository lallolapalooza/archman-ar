Pod__POM__Node(3pm)

الحزمة: perl-pod-pom

التراخيص: GPL-1.0-or-later, Artistic-2.0

Pod::POM::Node(3) وثائق Perl التي يساهم بها المستخدمون Pod::POM::Node(3)

الاسم

Pod::POM::Node - الفئة الأساسية لعقدة نموذج POM

الملخص

package Pod::POM::Node::Over;
use parent qw( Pod::POM::Node );
use vars qw( @ATTRIBS @ACCEPT $EXPECT $ERROR );
@ATTRIBS =   ( indent => 4 );
@ACCEPT  = qw( over item begin for text verbatim );
$EXPECT  =  q( back );
package main;
my $list = Pod::POM::Node::Over->new(8);
$list->add('item', 'First Item');
$list->add('item', 'Second Item');
...

الوصف

يصف هذا المستند العمل الداخلي لفئة Pod::POM::Node ويقدم شرحًا موجزًا للعلاقة بينها وبين فئاتها المشتقة. إنه دليل للهواة المهتمين بالتفاصيل الداخلية أكثر من كونه وثائق عامة للمستخدمين. راجع Pod::POM للحصول على معلومات حول استخدام الفئات.

تطبق هذه الفئة فئة عقدة أساسية يتم اشتقاقها لتمثيل عناصر مختلفة داخل نموذج كائن Pod.

package Pod::POM::Node::Over;
use parent qw( Pod::POM::Node );
use vars qw( @ATTRIBS $ERROR );
@ATTRIBS = ( indent => 4 );
package main;
my $list = Pod::POM::Node::Over->new();

يمكن تحديد ميزة العقدة عن طريق تحديد متغيرات معينة في الفئة المشتقة. يمكن استخدام القائمة @ATTRIBS للإشارة إلى السمات التي يجب أن تقبلها العقدة. في حالة عقدة "=over"، على سبيل المثال، يمكن تحديد سمة "indent" والتي يتم تعيينها افتراضيًا إلى 4.

package Pod::POM::Node::Over;
use parent qw( Pod::POM::Node );
use vars qw( @ATTRIBS $ERROR );
@ATTRIBS = ( indent => 4 );

يتوقع الطريقة new() الآن وسيطًا لتعيين قيمة المسافة البادئة، أو ستستخدم 4 كقيمة افتراضية إذا لم يتم توفير أي وسيط.

my $list = Pod::POM::Node::Over->new(8);    # indent: 8
my $list = Pod::POM::Node::Over->new();    # indent: 4

إذا كانت القيمة الافتراضية غير محددة، فإن الوسيط إلزامي.

package Pod::POM::Node::Head1;
use parent qw( Pod::POM::Node );
use vars qw( @ATTRIBS $ERROR );
@ATTRIBS = ( title => undef );
package main;
my $head = Pod::POM::Node::Head1->new('My Title');

إذا لم يتم توفير وسيط إلزامي، فإن المنشئ سيعيد undef للإشارة إلى الفشل. يتم تعيين المتغير $ERROR في الفئة المشتقة ليحتوي على سلسلة نصية بالشكل "$type expected an attribute $attribute".

# سيطبع خطأ: "head1 expected a title"
my $head = Pod::POM::Node::Head1->new()
    || die $Pod::POM::Node::Head1::ERROR;

للتسهيل، يمكن استدعاء الإجراء المساعد error() كطريقة فئة للحصول على هذه القيمة.

my $type = 'Pod::POM::Node::Head1';
my $head = $type->new()
    || die $type->error();

يمكن استخدام المتغير @ACCEPT للإشارة إلى أنواع العقد التي يتم قبولها كأبناء للعقدة.

package Pod::POM::Node::Head1;
use parent qw( Pod::POM::Node );
use vars qw( @ATTRIBS @ACCEPT $ERROR );
@ATTRIBS =   ( title => undef );
@ACCEPT  = qw( head2 over begin for text verbatim );

يمكن استدعاء الطريقة add() على عقدة لإضافة عقدة ابن جديدة كجزء من محتواها.

$head->add('over', 8);

يشير الوسيط الأول إلى نوع العقدة. يتم فحص القائمة @ACCEPT للتأكد من أن نوع عقدة الابن مقبول للعقدة الأم. إذا كان صالحًا، يتم استدعاء منشئ فئة عقدة الابن المقابلة مع تمرير أي وسيطات متبقية كسمات. ثم يتم إرجاع العقدة الجديدة.

my $list = $head->add('over', 8);

يمكن استدعاء الطريقة error() على العقدة الأم للحصول على أي أخطاء تم إنشاؤها بواسطة عقدة الابن.

my $list = $head->add('over', 8);
die $head->error() unless defined $list;

إذا كانت عقدة الابن غير مقبولة من قبل العقدة الأم، فإن الطريقة add() تعيد إحدى القيم IGNORE أو REDUCE أو REJECT، كما هو محدد في Pod::POM::Constants. تستخدم هذه القيم بواسطة وحدة المحلل Pod::POM لتنفيذ محلل بسيط من نوع shift/reduce.

في الحالة الأكثر شيوعًا، يتم إرجاع IGNORE للإشارة إلى أن العقدة الأم لا تعرف شيئًا عن عقدة الابن الجديدة. يستخدم المحلل هذا كمؤشر على أنه يجب عليه التراجع عبر مكدس التحليل حتى يجد عقدة يجب أن تقبل هذه العقدة الابن. من خلال هذه الآلية، يمكن للمحلل إنهاء كتل POD هذه ضمنيًا. على سبيل المثال، قائمة عنصر تبدأ بعلامة "=item" لن تقبل علامة "=item" أخرى ولكنها ستعيد IGNORE. سيتراجع المحلل حتى يجد العقدة "=over" التي يجب أن تقبلها. لذلك، فإن علامة "=item" جديدة تنهي ضمنيًا أي علامة "=item" سابقة.

يمكن استخدام المتغير $EXPECT للإشارة إلى نوع العقدة التي تتوقع عقدة أم إنهاء نفسها. عقدة "=over"، على سبيل المثال، يجب أن تنتهي دائمًا بعلامة "=back". عند حدوث هذه المطابقة، تعيد الطريقة add() القيمة REDUCE للإشارة إلى الإنهاء الناجح.

package Pod::POM::Node::Over;
use parent qw( Pod::POM::Node );
use vars qw( @ATTRIBS @ACCEPT $EXPECT $ERROR );
@ATTRIBS =   ( indent => 4 );
@ACCEPT  = qw( over item begin for text verbatim );
$EXPECT  =  q( back );
package main;
my $list = Pod::POM::Node::Over->new();
my $item = $list->add('item');
$list->add('back');                 # يعيد REDUCE

إذا لم يتم تحديد عقدة ابن في القائمة @ACCEPT أو لم تطابق أي $EXPECT محدد، يتم إرجاع REJECT. تقوم العقدة الأم بتعيين خطأ داخلي بالشكل "$type expected a terminator $expect". يستخدم المحلل هذا لاكتشاف علامات POD المفقودة. في معظم الحالات، يكون المحلل ذكيًا بما يكفي لإصلاح البنية غير الصحيحة وتحويل أي أخطاء إلى تحذيرات.

# سيطبع خطأ 'over expected a terminator back'
ref $list->add('head1', 'My Title')     # يعيد REJECT
    || die $list->error();

تحتوي كل عقدة على حقل 'type' يحتوي على سلسلة نصية بسيطة تشير إلى نوع العقدة، على سبيل المثال 'head1'، 'over'، إلخ. تشير المتغيرات $NODES و $NAMES (في الفئة الأساسية) إلى قوائم تجزئة تربط هذه الأسماء بأسماء الحزم والعكس (على سبيل المثال، head1 <=> Pod::POM::Node::Head1).

print $list->{ type };      # 'over'

يتم توفير طريقة AUTOLOAD للوصول إلى هذه العناصر الداخلية لأولئك الذين لا يرغبون في انتهاك تغليف الكائن.

print $list->type();

تحتوي العقد أيضًا على قائمة 'content' محملة في فئة Pod::POM::Node::Content، والتي تحتوي على محتوى (أبناء) للعقدة. تعيد الطريقة AUTOLOAD هذا كمرجع لقائمة أو كقائمة عناصر اعتمادًا على السياق الذي تم استدعاؤها فيه.

my $items = $list->content();
my @items = $list->content();

تحتوي كل عقدة أيضًا على قائمة 'content' لكل نوع ابن فردي يمكن قبوله.

my @items = $list->item();
my @text  = $list->text();
my @vtext = $list->verbatim();

تستخدم الطريقة present() لعرض العقدة من خلال عرض محدد. هذا ببساطة يربط نوع العقدة بطريقة يتم استدعاؤها على كائن العرض. وهذا ما يسمى 'double dispatch'.

my $view = 'Pod::POM::View::HTML';
print $list->present($view);

يتم بناء اسم الطريقة من نوع العقدة مسبوقًا بـ 'view_'. لذلك، فإن ما يلي يكون مكافئًا تقريبًا.

$list->present($view);
$view->view_list($list);

ميزة الأولى على الثانية هي، بالطبع، أن المتصل ليس بحاجة إلى معرفة أو تحديد نوع العقدة. العقدة نفسها في أفضل وضع لتحديد نوعها.
