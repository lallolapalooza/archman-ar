POE__Session(3pm)

الحزمة: perl-poe

التراخيص: GPL, PerlArtistic

POE::Session(3) وثائق Perl المساهمة من المستخدم POE::Session(3)

الاسم

POE::Session - مهمة عامة قائمة على الأحداث

الملخص

use POE; # يتضمن تلقائيًا POE::Kernel و POE::Session
POE::Session->create(
  inline_states => {
    _start => sub { $_[KERNEL]->yield("next") },
    next   => sub {
      print "tick...\n";
      $_[KERNEL]->delay(next => 1);
    },
  },
);
POE::Kernel->run();
exit;

يمكن لـ POE::Session أيضًا إرسال استدعاءات إلى أساليب الكائنات والفئات عبر دوال رد الاتصال "object_states" و "package_states".

الوصف

تقوم POE::Session وفئاتها الفرعية بترجمة الأحداث من الموزع العام لـ POE::Kernel إلى اتفاقيات الاستدعاء المناسبة لكود التطبيق. بعبارات نمط التصميم، تعد فئات POE::Session محولات بين POE::Kernel وكود التطبيق.

الجلسات التي يديرها POE::Kernel هي أقرب إلى هياكل المهام العامة. للأسف، هذان المفهومان المتباينان لهما اسمان متطابقان تقريبًا.

ملاحظة حول المصطلحات

ستشير هذه الوثائق إلى معالجات الأحداث باسم "الحالات" في بعض المواقف التي لا يمكن تجنبها. كانت الجلسات في الأصل تهدف إلى أن تكون آلات حالة قائمة على الأحداث، ولكن أغراضها تطورت مع مرور الوقت. لا يزال بعض المفردات القديمة موجودًا في واجهة برمجة التطبيقات (API) للتوافق مع الإصدارات السابقة، ومع ذلك.

بشكل مربك، POE::NFA هو فئة لتنفيذ آلات الحالة الحقيقية القائمة على الأحداث. تستخدم وثائقه "الحالة" بالمعنى الصحيح.

استخدام POE::Session

لـ POE::Session غرضان رئيسيان. أولاً، تقوم بربط أسماء الأحداث بالكود الذي سيعالجها. ثانيًا، تقوم بربط واجهة توزيع أحداث متسقة بهذه المعالجات.

ضع في اعتبارك "الملخص" على سبيل المثال. يتم إنشاء مثيل POE::Session باستخدام "inline_states" اثنتين، كل منهما تربط اسم حدث ("_start" و "next") بتوزيعة مضمنة. يضمن POE::Session أن "$_[KERNEL]" وما إلى ذلك ذات معنى داخل معالج الحدث.

يمكن أيضًا أن تكون معالجات الأحداث أساليب كائنات أو فئات، باستخدام "object_states" و "package_states" على التوالي. بناء الجملة create() يختلف عن "inline_states" ، ولكن اتفاقية الاستدعاء متطابقة تقريبًا.

لاحظ أنه لم يتم حفظ كائن POE::Session الذي تم إنشاؤه في متغير. يعطي كائن POE::Session الجديد نفسه إلى POE::Kernel، الذي يديره بعد ذلك وجميع الموارد التي يستخدمها.

من الممكن الاحتفاظ بمراجع لكائنات POE::Session الجديدة، ولكن هذا ليس ضروريًا عادةً. إذا لم يكن التطبيق حذرًا بشأن تنظيف هذه المراجع، فستقوم بإنشاء مراجع دائرية، والتي ستؤدي إلى تسرب الذاكرة عندما يقوم POE::Kernel بتدمير كائن POE::Session بشكل طبيعي. يوصى بالاحتفاظ بمعرف الجلسة بدلاً من ذلك.

اتفاقية استدعاء POE::Session

أكبر عقبة نحوية يواجهها معظم الأشخاص مع POE هي اتفاقية استدعاء POE::Session غير التقليدية. على سبيل المثال:

sub handle_event {
  my ($kernel, $heap, $parameter) = @_[KERNEL, HEAP, ARG0];
  ...;
}

أو استخدام $_[KERNEL] و $_[HEAP] و $_[ARG0] مضمنة، كما هو الحال في معظم الأمثلة.

ما يحدث هنا أساسي للغاية. يمرر Perl المعلمات إلى التوزيعات الفرعية أو الأساليب باستخدام المصفوفة @_. "KERNEL" و "HEAP" و "ARG0" وغيرها هي ثوابت يتم تصديرها بواسطة POE::Session (الذي يتم تضمينه مجانًا عندما يستخدم البرنامج POE).

لذلك $_[KERNEL] هو المعلمة KERNELth لمعالج الحدث. @_[HEAP, ARG0] هو شريحة من @_ تحتوي على معلمات HEAPth و ARG0th.

بينما يبدو هذا غريبًا، إلا أنه بناء جملة Perl بسيط وقانوني تمامًا. يستخدم POE ذلك لعدة أسباب:

1.
في الحالة الشائعة، يكون تمرير المعلمات في @_ أسرع من تمرير مراجع الهاش أو المصفوفة ثم فك مرجعيتها في المعالج.
2.
الأخطاء الإملائية في قوائم المعلمات المستندة إلى الهاش هي إما أخطاء وقت تشغيل دقيقة أو تتطلب فحصًا مستمرًا لوقت التشغيل. الثوابت معروفة إما في وقت التجميع، أو هي أخطاء واضحة في وقت التجميع.
3.
يمكن أن تنتقل المعلمات في المستقبل دون كسر كود التطبيق، حيث أن الإشارة إلى إزاحات @_ باستخدام الثوابت تسمح بذلك.
4.
لا يحتاج معظم معالجات الأحداث إلى كل @_. تسمح الشرائح للمعالجات باستخدام المعلمات التي يهتمون بها فقط.

معلمات POE::Session

يتلقى معالجات الأحداث معظم سياق وقت التشغيل الخاص بها في ما يصل إلى تسع معلمات رد اتصال. يوفر POE::Kernel العديد منها.

$_[OBJECT]

$_[OBJECT] هو $self لمعالجات الأحداث التي هي أسلوب كائن. إنها اسم الفئة (الحزمة) لمعالجات الأحداث المستندة إلى الفئة. إنها غير معرفة للتوزيعات الفرعية العادية، التي ليس لها قيمة خاصة تشبه $self.

"OBJECT" هي دائمًا صفر، نظرًا لأن $_[0] دائمًا $self أو $class في أساليب الكائنات والفئات. يتم استدعاء معالجات التوزيعات الفرعية مع عنصر نائب "غير معرف" في $_[0] حتى تظل الإزاحات الأخرى صالحة.

غالبًا ما يكون من المفيد لمعالجات الأحداث المستندة إلى الأساليب استدعاء أساليب أخرى في نفس الكائن. $_[OBJECT] يساعد في حدوث ذلك.

sub ui_update_everything {
  my $self = $_[OBJECT];
  $self->update_menu();
  $self->update_main_window();
  $self->update_status_line();
}

يمكنك أيضًا استخدام وراثة الأساليب. هنا نستدعي $self->a_method(@_). نظرًا لأن عامل التشغيل "-">" في Perl يقوم بإزاحة $self إلى بداية @_، يجب علينا أولاً إزاحة نسخة للحفاظ على إزاحات معلمات POE:

sub a_method {
  my $self = shift;
  $self->SUPER::a_method( @_ );
  # ... المزيد من العمل ...
}

$_[SESSION]

$_[SESSION] هو مرجع إلى كائن الجلسة الحالي. هذا يسمح لمعالجات الأحداث بالوصول إلى أساليب جلستهم. قد تقارن البرامج أيضًا $_[SESSION] بـ $_[SENDER] للتحقق من أن الأحداث داخل الجلسة لم تأت من جلسات أخرى.

يمكن أيضًا استخدام $_[SESSION] كوجهة لـ post() و call() داخل الجلسة. ومع ذلك، فإن yield() أكثر ملاءمة وكفاءة بشكل هامشي من "post($_[SESSION], ...)".

من سوء الأدب الوصول إلى جلسة أخرى مباشرة. النهج الموصى به هو معالجة الجلسة من خلال معالج الأحداث.

sub enable_trace {
  my $previous_trace = $_[SESSION]->option( trace => 1 );
  my $id = $_[SESSION]->ID;
  if ($previous_trace) {
    print "Session $id: dispatch trace is still on.\n";
  }
  else {
    print "Session $id: dispatch trace has been enabled.\n";
  }
}

$_[KERNEL]

المعلمة KERNELth هي دائمًا مرجع إلى مثيل POE::Kernel الفردي للتطبيق. يتم استخدامه في الغالب لاستدعاء أساليب POE::Kernel من معالجات الأحداث.

# تعيين مؤقت لمدة 10 ثوانٍ.
$_[KERNEL]->delay( time_is_up => 10 );

$_[HEAP]

يحتوي كل كائن POE::Session على مساحة اسم المتغيرات الخاصة به والمعروفة باسم "HEAP" للجلسة. يتم نمذجته وتسميته على غرار كومات ذاكرة العمليات (وليس كومات الأولوية). الكومات هي افتراضيًا مراجع هاش مجهولة، ولكن يمكن تهيئتها في create() لتكون أي شيء تقريبًا. لا يستخدم POE::Session نفسه $_[HEAP] أبدًا، على الرغم من أن بعض مكونات POE تفعل ذلك.

لا تتداخل الكومات بين الجلسات، على الرغم من أنه يمكن استخدام معلمة "heap" الخاصة بـ create() لجعل هذا يحدث.

instantiate() يقوم بإنشاء وإرجاع كائن POE::Session فارغ. يتم استدعاؤه مع CREATE_PARAMETERS في مرجع هاش قبل أن تقوم create() بمعالجتها. ستؤثر التعديلات على CREATE_PARAMETERS على كيفية تهيئة create() للجلسة الجديدة.

قد تقوم الفئات الفرعية بتجاوز instantiate() لتغيير بنية الجلسة الأساسية. قد تقوم بتوسيع instantiate() لإضافة معلمات جديدة إلى create().

يجب إزالة أي معلمات لا تتعرف عليها create() من CREATE_PARAMETERS قبل أن تعود instantiate(). ستفشل create() إذا اكتشفت معلمات غير معروفة.

تأكد من إرجاع $self من instantiate.

sub instantiate {
  my ($class, $create_params) = @_;
  # اجعل الفئة الأساسية تقوم بإنشاء الجلسة الجديدة.
  my $self = $class->SUPER::instantiate($create_parameters);
  # قم بتوسيع المعلمات التي تعرفها create().
  my $new_option = delete $create_parameters->{new_option};
  if (defined $new_option) {
    # ... قم بتخصيص $self هنا ...
  }
  return $self;
}

try_alloc START_ARGS

try_alloc() تستدعي session_alloc() الخاصة بـ POE::Kernel لتخصيص بنية جلسة والبدء في إدارة الجلسة داخل نواة POE. يتم استدعاؤها في نهاية create() الخاصة بـ POE::Session. إنها تعيد $self.

إنها خطاف للفئات الفرعية للتخصيص المتأخر للجلسة قبل أن تعود create(). قد تؤثر أيضًا على محتويات @_[ARG0..$#_] التي يتم تمريرها إلى معالج _start الخاص بالجلسة.

sub try_alloc {
  my ($self, @start_args) = @_;
  # قم بإجراء التهيئة المتأخرة.
  # ...
  # قم بتمرير $self إلى POE::Kernel.
  return $self->SUPER::try_alloc(@args);
}

أحداث POE::Session

يرجى عدم تعريف أحداث جديدة تبدأ بشرطة سفلية بادئة. تطالب POE بالأحداث /^_/ كأحداث خاصة بها.

تنشئ POE::Session حدثًا واحدًا فقط، وهو _default. جميع أحداث POE الداخلية الأخرى يتم إنشاؤها بواسطة (وموثقة في) POE::Kernel.

_default

_default هو معالج الأحداث الخاص بـ "AUTOLOAD". إذا لم تتمكن POE::Session من العثور على معالج في وقت الإرسال، فإنها تحاول إعادة توجيه الحدث إلى معالج _default بدلاً من ذلك.

إذا لم يكن هناك معالج _default، فإن POE::Session ستتجاهل الحدث بصمت ما لم يتم تعيين خيار "default".

للحفاظ على المعلومات الأصلية، يتم تغيير الحدث الأصلي قليلاً قبل إعادة توجيهه إلى معالج _default: يتم نقل معلمات الحدث الأصلية إلى مرجع مصفوفة في ARG1، ويتم تمرير اسم الحدث الأصلي إلى _default في ARG0.

sub handle_default {
  my ($event, $args) = @_[ARG0, ARG1];
  print(
    "Session ", $_[SESSION]->ID,
    " caught unhandled event $event with (@$args).\n"
  );
}

_default مرن للغاية. يمكن استخدامه لتصحيح الأخطاء، أو لمعالجة أسماء الأحداث التي تم إنشاؤها ديناميكيًا دون تعريف معالجاتها مسبقًا. بهذا المعنى، يعمل _default بشكل مشابه لـ Perl's "AUTOLOAD".

يمكن أيضًا استخدام _default كبند افتراضي أو "بخلاف ذلك" في عبارة switch. ضع في اعتبارك معالج إدخال يطلق أحداثًا بناءً على اسم أمر:

sub parse_command {
  my ($command, @parameters) = split /\s+/, $_[ARG0];
  $_[KERNEL]->post( "cmd_$command", @parameters );
}

يمكن استخدام معالج _default لإصدار أخطاء للأوامر غير المعروفة:

sub handle_default {
  my $event = $_[ARG0];
  return unless $event =~ /^cmd_(\S+)/;
  warn "Unknown command: $1\n";
}

يتم تنفيذ السلوك الافتراضي في POE::Session، لذلك قد يكون مختلفًا لأنواع الجلسات الأخرى.

ميزات تصحيح أخطاء POE::Session

تحتوي POE::Session على تأكيد تصحيح واحد، في الوقت الحالي.

ASSERT_STATES

يؤدي تعيين ASSERT_STATES إلى true إلى جعل كل جلسة تصدر تحذيرًا عندما يُطلب منها معالجة أحداث غير معروفة. يقوم Session.pm بتنفيذ جوهر ASSERT_STATES عن طريق تعيين خيار "default" إلى true بدلاً من false. انظر طريقة option() في وقت سابق في هذا المستند للحصول على تفاصيل حول خيار "default".

انظر أيضًا

POE::Kernel.

قسم SEE ALSO في POE يحتوي على جدول محتويات يغطي توزيع POE بأكمله.

الأخطاء

هناك احتمال أن تتصادم معرفات الجلسات بعد التفاف قيمة عدد صحيح لـ Perl. يمكن أن يحدث هذا بعد 4.29 مليار جلسة.

احذر من المراجع الدائرية

كما تعلم على الأرجح، فإن المرجع الدائري هو عندما يكون المتغير جزءًا من سلسلة مراجع تشير في النهاية إلى نفسها. لن تستعيد Perl الذاكرة المتضمنة في مثل هذه السلسلة من المراجع حتى يتم كسر السلسلة يدويًا.

هنا يتم إنشاء POE::Session يشير إلى نفسه عبر عددي خارجي. تقوم معالجات الأحداث باستيراد $session عبر إغلاقات يتم تخزينها بدورها داخل $session. حتى لو توقفت هذه الجلسة، فستظل المراجع الدائرية قائمة.

my $session;
$session = POE::Session->create(
  inline_states => {
    _start => sub {
      $_[HEAP]->{todo} = [ qw( step1 step2 step2a ) ],
      $_[KERNEL]->post( $session, 'next' );
    },
    next => sub {
      my $next = shift @{ $_[HEAP]->{todo} };
      return unless $next;
      $_[KERNEL]->post( $session, $next );
    }
    # ....
  }
);

مختزلة إلى جوهرها:

my %event_handlers;
$event_handler{_start} = sub { \%event_handlers };

لاحظ أيضًا أن إجراء فرعي مجهول ينشئ إغلاقًا لجميع المتغيرات المعجمية في النطاق الذي تم تعريفه فيه، حتى لو لم يشر إليها. لا يزال $session محتفظًا بمرجع دائري هنا:

my $self = $package->new;
my $session;
$session = POE::Session->create(
  inline_state => {
    _start => sub { $self->_start( @_[ARG0..$#_] ) }
  }
);

لتجنب ذلك، يمكن للجلسة تعيين اسم مستعار لنفسها. يمكن لأجزاء أخرى من البرنامج الإشارة إليها بعد ذلك بالاسم المستعار. في هذه الحالة، لا يحتاج المرء إلى تتبع الجلسة بنفسه (POE::Kernel سيفعل ذلك على أي حال).

POE::Session->create(
  inline_states => {
    _start => sub {
      $_[HEAP]->{todo} = [ qw( step1 step2 step2a ) ],
      $_[KERNEL]->alias_set('step_doer');
      $_[KERNEL]->post( 'step_doer', 'next' );
    },
    next => sub {
      my $next = shift @{ $_[HEAP]->{todo} };
      return unless $next;
      $_[KERNEL]->post( 'step_doer', $next );
    }
    # ....
  }
);

الأسماء المستعارة ليست مطلوبة حتى في المثال السابق لأن الجلسة تشير إلى نفسها. يمكن للمرء بدلاً من ذلك استخدام طريقة yield() الخاصة بـ POE::Kernel لنشر الحدث مرة أخرى إلى الجلسة الحالية:

next => sub {
  my $next = shift @{ $_[HEAP]->{todo} };
  return unless $next;
  $_[KERNEL]->yield( $next );
}

أو المعلمة "$\_[SESSION]" التي تم تمريرها إلى كل معالج أحداث، ولكن yield() أكثر كفاءة.

next => sub {
  my $next = shift @{ $_[HEAP]->{todo} };
  return unless $next;
  $_[KERNEL]->post( $_[SESSION], $next );
}

على غرار "$\_[SESSION]"، يمكن للجلسة الرد على مرسل الحدث عن طريق النشر إلى "$\_[SENDER]". هذا رائع للاستجابة للطلبات.

إذا كان البرنامج يجب أن يحتفظ بنوع من مرجع الجلسة الديناميكي، فمن الموصى به استخدام المعرف الرقمي للجلسة بدلاً من الكائن نفسه. يمكن تحويل معرف الجلسة مرة أخرى إلى الكائن الخاص به، ولكن post() يقبل معرفات الجلسات بالإضافة إلى الكائنات والأسماء المستعارة:

my $session_id;
$session_id = POE::Session->create(
  inline_states => {
    _start => sub {
      $_[HEAP]->{todo} = [ qw( step1 step2 step2a ) ],
      $_[KERNEL]->post( $session_id, 'next' );
    },
    # ....
  }
)->ID;
