subroutine cpoequb (integer n, complex, dimension( lda, * ) a, integer lda, real, dimension( * ) s, real scond, real amax, integer info)

CPOEQUB

الغرض:

!>
!> CPOEQUB تحسب عوامل القياس للصفوف والأعمدة المقصود بها موازنة مصفوفة هرمية موجبة محددة A وتقليل رقم شرطها
!> (بالنسبة للمعيار الثنائي). تحتوي S على عوامل القياس،
!> S(i) = 1/sqrt(A(i,i))، المختارة بحيث تحتوي المصفوفة الموسعة B ذات
!> العناصر B(i,j) = S(i)*A(i,j)*S(j) على الواحدات على القطر. هذا
!> الاختيار لـ S يضع رقم شرط B ضمن عامل N من
!> أصغر رقم شرط ممكن عبر جميع عوامل القياس القطرية الممكنة.
!>
!> تختلف هذه الروتين عن CPOEQU بتقييد عوامل القياس
!> إلى قوة أساس النظام. باستثناء التدفق الزائد والناقص، فإن القياس بواسطة
!> هذه العوامل لا يقدم أخطاء تقريب إضافية. ومع ذلك، فإن
!> العناصر القطرية الموسعة لم تعد تقارب 1 بل تقع
!> بين sqrt(radix) و 1/sqrt(radix).
!>

المعلمات

N 
!>          N هو عدد صحيح (INTEGER)
!>          رتبة المصفوفة A. N >= 0.
!>

A

!>          A هي مصفوفة مركبة (COMPLEX)، بأبعاد (LDA,N)
!>          المصفوفة الهرمية الموجبة المحددة بحجم N*N التي سيتم حساب عوامل القياس لها.
!>          يتم الإشارة فقط إلى العناصر القطرية لـ A.
!>

LDA

!>          LDA هو عدد صحيح (INTEGER)
!>          البعد الرائد للمصفوفة A. LDA >= max(1,N).
!>

S

!>          S هي مصفوفة حقيقية (REAL)، بأبعاد (N)
!>          إذا كانت INFO = 0، فإن S تحتوي على عوامل القياس لـ A.
!>

SCOND

!>          SCOND هو عدد حقيقي (REAL)
!>          إذا كانت INFO = 0، فإن S تحتوي على نسبة أصغر S(i) إلى
!>          أكبر S(i). إذا كانت SCOND >= 0.1 وكان AMAX ليس كبيراً جداً أو صغيراً جداً،
!>          فلا يستحق القياس بواسطة S.
!>

AMAX

!>          AMAX هو عدد حقيقي (REAL)
!>          القيمة المطلقة لأكبر عنصر في المصفوفة. إذا كان AMAX قريبًا جدًا
!>          من التدفق الزائد أو قريبًا جدًا من التدفق الناقص، يجب قياس المصفوفة.
!>

INFO

!>          INFO هو عدد صحيح (INTEGER)
!>          = 0: خروج ناجح
!>          < 0: إذا كانت INFO = -i، فإن الوسيط i كان له قيمة غير قانونية
!>          > 0: إذا كانت INFO = i، فإن العنصر القطري i غير موجب.
!>
