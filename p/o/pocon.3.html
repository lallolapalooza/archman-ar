subroutine cpocon (character uplo, integer n, complex, dimension( lda, * ) a, integer lda, real anorm, real rcond, complex, dimension( * ) work, real, dimension( * ) rwork, integer info)

CPOCON

الغرض:

!>
!> CPOCON يقدّر مقلوب رقم الشرط (بالنسبة للمعيار 1) لمصفوفة مركبة هرميتية موجبة محددة، A = U**H*U أو A = L*L**H، التي تم حسابها بواسطة CPOTRF.
!>
!> يتم الحصول على تقدير لـ norm(inv(A))، ويتم حساب مقلوب رقم الشرط على أنه RCOND = 1 / (ANORM * norm(inv(A))).
!>

المعلمات

UPLO 
!>          UPLO هو CHARACTER*1
!>          = 'U':  المثلث العلوي من A مخزن؛
!>          = 'L':  المثلث السفلي من A مخزن.
!>

N

!>          N هو INTEGER
!>          رتبة المصفوفة A.  N >= 0.
!>

A

!>          A هو مصفوفة COMPLEX، ذات بعد (LDA,N)
!>          العامل المثلثي U أو L من تفكيك Cholesky
!>          A = U**H*U أو A = L*L**H، كما تم حسابه بواسطة CPOTRF.
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرائد للمصفوفة A.  LDA >= max(1,N).
!>

ANORM

!>          ANORM هو REAL
!>          المعيار 1 (أو المعيار اللانهائي) للمصفوفة الهرميتية A.
!>

RCOND

!>          RCOND هو REAL
!>          مقلوب رقم شرط المصفوفة A،
!>          محسوب على أنه RCOND = 1/(ANORM * AINVNM)، حيث AINVNM هو تقدير
!>          للمعيار 1 لـ inv(A) المحسوب في هذا الروتين.
!>

WORK

!>          WORK هي مصفوفة COMPLEX، ذات بعد (2*N)
!>

RWORK

!>          RWORK هي مصفوفة REAL، ذات بعد (N)
!>

INFO

!>          INFO هو INTEGER
!>          = 0:  خروج ناجح
!>          < 0:  إذا كان INFO = -i، فإن الوسيط i كان له قيمة غير قانونية
!>
