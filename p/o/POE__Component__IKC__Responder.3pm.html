POE__Component__IKC__Responder(3pm)

الحزمة: perl-poe-component-ikc

التراخيص: GPL, PerlArtistic

IKC::Responder(3) توثيق Perl مقدم من المستخدمين IKC::Responder(3)

الاسم

POE::Component::IKC::Responder - معالج الحالة لـ POE IKC

ملخص

use POE;
use POE::Component::IKC::Responder;
create_ikc_responder();
...
$kernel->post('IKC', 'post', $to_state, $state);
$ikc->publish('my_name', [qw(state1 state2 state3)]);

الوصف

تطبق هذه الوحدة معالجة الحالة لـ POE IKC. يرسل المستجيب الحالات إلى النوى الخارجية ويستدعي الحالات في النواة المحلية بناءً على طلب النوى الخارجية.

هناك واجهتان للمستجيب. إما عن طريق إرسال الحالات إلى الجلسة 'IKC' أو عن طريق واجهة الكائن. على الرغم من أن الثانية أسرع، إلا أن السلوك الأفضل حيث أن POE نظام تعاوني.

الحالات/الطرق

"spawn"

POE::Component::IKC::Responder->spawn();

تقوم هذه الدالة بإنشاء جلسة المستجيب والكائن. عادةً، يقوم POE::Component::IKC::Client أو POE::Component::IKC::Server بالقيام بذلك نيابة عنك. ومع ذلك، في بعض التطبيقات، تريد التأكد من أن المستجيب نشط قبل أن تقوم بذلك.

"post"

يرسل طلب حالة إلى نواة خارجية. تعيد قيمة منطقية إذا تم إرسال الحالة وقيمة منطقية أخرى إذا لم يتمكن من إرسال الطلب إلى النواة الخارجية. هذا لا يعني بالضرورة أن النواة الخارجية تمكنت من نشر الحالة، على الرغم من أنها تفعل ذلك. المعلمات هي التالية:

"foreign_state"
مرآة لحالة النواة الخارجية. انظر POE::Component::IKC::Specifier.
"parameters"
مرجع لأي شيء تريد أن تحصل عليه الحالة الخارجية كـ ARG0. إذا كنت تريد تحديد معلمات متعددة، استخدم مرجع مصفوفة ودع الحالة الخارجية تفك مرجعها. 
$kernel->post('IKC', 'post', 
    {kernel=>'Syslog', session=>'logger', state=>'log'},
    [$faculty, $priority, $message];

أو

$ikc->post('poe://Syslog/logger/log', [$faculty, $priority, $message]);

هذا يسجل حالة مع مسجل افتراضي.

انظر "PROXY SENDER" أدناه.

"call"

هذا مطابق لـ "post"، باستثناء أنه يحتوي على معلمة ثالثة تصف الحالة التي ستتلقى القيمة المعادة من النواة الخارجية.

$kernel->post('IKC', 'call', 
            'poe://Pulse/timeserver/time', '',
            'poe:get_time');

أو

$ikc->call({kernel=>'Pulse', session=>'timeserver', state=>'time'},
            '', 'poe://me/get_time');

هذا يسأل النواة الخارجية 'Pulse' عن الوقت. يتم نشر الحالة 'get_time' للجلسة الحالية مع ما أعادته الحالة الخارجية.

ليس عليك نشر رسائل العودة، لأنها تنشر مؤقتًا. إلى أي مدى مؤقت؟ يمكن نشرها من نواة خارجية مرة واحدة فقط. هذه مشكلة بالطبع لأن شخصًا ما يمكنه النشر قبل العودة. هذه هي الحياة.

"foreign_state"
مطابق لمعلمة "post".
"parameters"
مطابق لمعلمة "post".
"rsvp"
معرف الحدث للعودة. أي، يتم استدعاء هذه الحالة بالقيمة المعادة من الحالة الخارجية. يمكن أن يكون مرآة لـ "foreign_state" أو مجرد اسم حالة في الجلسة الحالية.
$kernel->call('IKC', 'post', 
    {kernel=>'e-comm', session=>'CC', state=>'check'},
    {CC=>$cc, expiry=>$expiry}, folder=>$holder},
    'is_valid');
# أو
$ikc->call('poe://e-comm/CC/check',
    {CC=>$cc, expiry=>$expiry}, folder=>$holder},
    'poe://me/is_valid');

هذا يسأل خادم e-comm للتحقق مما إذا كان رقم بطاقة الائتمان "صالحًا". نعم، هذا سيكون على الأرجح استخدامًا مفرطًا.

الحالة "rsvp" لا تحتاج إلى النشر. يتعامل IKC مع تسجيل الحالة "rsvp" والسماح للنواة الخارجية بالنشر إليها.

انظر "PROXY SENDER" أدناه.

"default"

يضبط النواة الخارجية الافتراضية. يجب أن تكون متصلاً بالنواة الخارجية أولاً.

المعلمة الوحيدة هي اسم النواة الخارجية.

تعيد قيمة منطقية في حالة النجاح.

"register"

يسجل أسماء النوى الخارجية مع المستجيب. يتم ذلك أثناء مرحلة التفاوض لـ IKC وعادة ما تتم إدارته بواسطة POE::Component::IKC::Channel. سيحدد النواة الافتراضية إذا لم تكن هناك نواة افتراضية.

المعلمة الأولى هي اسم نواة فريد. المعلمة الثانية هي مرجع مصفوفة لأسماء مستعارة للنواة للتسجيل.

"unregister"

يلغي تسجيل أسماء النوى الخارجية مع المستجيب. يتم ذلك عندما ينفصل النواة الخارجية بواسطة POE::Component::IKC::Channel. إذا كانت هذه هي النواة الافتراضية، فلن تكون هناك نواة افتراضية.

المعلمة الأولى هي اسم نواة أو اسم مستعار. المعلمة الثانية هي مرجع مصفوفة لأسماء مستعارة للنواة لإلغاء التسجيل. المعلمة الثانية غبية بعض الشيء، لأنك إذا ألغيت تسجيل نواة بعيدة، فسيتم إلغاء تسجيل جميع الأسماء المستعارة الخاصة بها أيضًا.

"register_local"

يسجل أسماء مستعارة محلية جديدة مع المستجيب. يتم ذلك داخليًا بواسطة POE::Component::IKC::Server و POE::Component::IKC::Client. لن يحدد النواة الافتراضية.

المعلمة الأولى والوحيدة هي مرجع مصفوفة لأسماء مستعارة للنواة للتسجيل.

"publish"

يخبر IKC أن بعض الحالات في الجلسة الحالية متاحة للاستخدام من قبل جلسات خارجية.

"session"
اسم مستعار للجلسة التي ستتصل بها الجلسات الخارجية. يجب أن يكون الاسم المستعار مسجلاً بالفعل مع النواة المحلية.
"states"
مرجع مصفوفة للحالات التي يمكن نشرها بواسطة الجلسات الخارجية. 
$kernel->post('IKC', 'publish', 'me', [qw(foo bar baz)]);
# أو
$ikc->publish('me', [qw(foo bar baz)]);

"retract"

يخبر IKC أن بعض الحالات لم تعد متاحة للاستخدام من قبل جلسات خارجية. ليس عليك إلغاء جميع الحالات المنشورة.

"session"
مطابق لمعلمة "session" في "publish"
"states"
مطابق لمعلمة "states" في "publish". إذا لم يتم تحديدها، سيتم إلغاء نشر *جميع* الحالات المنشورة. 
$kernel->post('IKC', 'retract', 'me', [qw(foo mibble dot)]);
# أو
$ikc->retract('me', [qw(foo)]);

"published"

$list=$kernel->call(IKC=>'published', $session);

يعيد قائمة بجميع الحالات المنشورة.

$hash=$kernel->call(IKC=>'published');

يعيد مرجع hashref، مفتاح حسب معرفات الجلسة. القيم هي مراجع مصفوفة للحالات المنشورة بواسطة تلك الجلسة.

"session"
اسم مستعار للجلسة التي تريد قائمة الحالات لها.

"subscribe"

يشترك في جلسات أو حالات خارجية.

Callback states

يمكن مراقبة الحالات التالية:

"channel"
يتم استدعاؤها عندما يصبح القناة جاهزًا أو يتم إغلاقه. ARG3 هو إما "ready" أو "close". ARG4 هو المعرف الرقمي للجلسة الخاصة بالقناة. انظر "CHANNELS" أدناه.
"register"
يتم استدعاؤها عند تسجيل نواة أو اسم مستعار. هذا يعادل اكتمال مرحلة الاتصال.
"unregister"
يتم استدعاؤها عند إلغاء تسجيل نواة أو اسم مستعار. هذا يعادل انقطاع اتصال النواة البعيدة.
"subscribe"
يتم استدعاؤها عندما ينجح IKC في الاشتراك في جلسة بعيدة. ARG3 هو محدد IKC لما تم الاشتراك فيه. استخدم هذا للنشر إلى جلسة الوكيل.
"unsubscribe"
يتم استدعاؤها عندما ينجح IKC في إلغاء الاشتراك من جلسة بعيدة.
"shutdown"
يتم إعلامك عند محاولة القيام بفصل سليم لـ IKC وجميع الجلسات الطرفية. يتم استدعاء هذا مرة واحدة فقط، بعد أن ينشر شخص ما حدث IKC/shutdown.
"error"
يتم الإبلاغ عن الأخطاء في النوى المحلية والبعيدة. ARG3 هي العملية التي فشلت. ARG4 هي رسالة الخطأ. انظر "ERRORS" أدناه.
"data"
بعض البيانات (يمكن أن تكون قيمة مقياسية أو مرجعًا) يتم تمريرها إلى الاستدعاء. هذا يسمح بمزيد من السحر.

يتم استدعاء حالات الاستدعاء مع المعلمات التالية:

"ARG0"
اسم النواة الذي تم تمريره إلى poe://*/IKC/monitor
"ARG1"
معرف أو اسم مستعار للنواة البعيدة من وجهة نظر IKC.
"ARG2"
علامة. إذا كانت صحيحة، فإن ARG1 هو المعرف الفريد للنواة البعيدة، إذا كانت خاطئة، فإن ARG1 هو اسم مستعار. هذا مفيد بشكل أساسي عند مراقبة *&*، وهو في الواقع نوع من الجمود.
"ARG3"
"$state->{data}" أي بيانات تريدها.
"ARG4" ... "ARGN"
معلمات خاصة بالاستدعاء. انظر أعلاه.

في معظم الأوقات، سيكون ARG0 و ARG1 متساويين. الاستثناءات هي إذا كنت تراقب *&* أو إذا قمت بتوفير محدد حدث IKC كامل لـ IKC/monitor بدلاً من مجرد اسم نواة.

Short note about monitoring all kernels with "*"

هناك سببان قد تجد نفسك تراقب جميع النوى البعيدة: معرفة الأسماء مسبقًا وأسماء غير معروفة مسبقًا.

إذا كنت تعرف أسماء النوى مسبقًا، فقد يكون من الأفضل لك مراقبة اسم معين. ومع ذلك، قد تفضل إجراء مقارنة مشابهة مع ARG1 (باستخدام التعبيرات العادية، على سبيل المثال). سيكون هذا مفيدًا للتجميع، حيث قد تتبع النوى المتكررة اصطلاح تسمية مثل [application]-[host]، لذلك يمكنك مقارنة "ARG1" بـ "/^credit-/" لمعرفة ما إذا كنت تريد إعداد أشياء خاصة بتلك النواة.

دون معرفة اسم النواة مسبقًا، قد تقوم بنوع من الاكتشاف التلقائي أو مجرد المراقبة لتصحيح الأخطاء أو التسجيل أو المحاسبة. من الواضح أنك لا تريد القيام بالاكتشاف التلقائي لكل اسم مستعار لكل نواة، فقط للاسم القياسي، لذلك مطلوب ARG2.

الاسم

POE::Component::IKC::Responder

الصيغة

مهمل. يرجى استخدام

POE::Compontent::IKC::Responder->spawn();

الوصف

POE::Component::IKC::Responder هو مكوّن يسمح بإجراء اتصال مع نواة بعيدة عبر بروتوكول IKC. يُستخدم هذا المكوّن لإدارة الاتصالات ومعالجة الأخطاء وإجراء العمليات بين النوى.

الحالات/الطرق

يسمح المكوّن بإدارة حالات مثل connected وdisconnected وclosed وerror وwaiting. تشمل الطرق connect وdisconnect وclose وmonitor.

القنوات

يسمح المكوّن بإدارة قنوات مثل channel-read وchannel-write وchannel-error وchannel-close.

الأخطاء

الإصدارات السابقة من IKC لم تسمح بمراقبة الأخطاء في اتصال بشكل مناسب. بدءًا من 0.2400 بدأنا بمراقبة الأخطاء.

هناك مرحلتان يمكن أن تحدث فيهما أخطاء: عند فتح الاتصال وأثناء تبادل الرسائل. تُعالج هاتان المرحلتان بشكل مختلف.

تستخدم on_error في POE::Component::IKC::Client وon_error في POE::Component::IKC::Server لتلقي الأخطاء أثناء الاتصال. لاحظ أن هذا يشمل المصافحة الأولية لـ IKC.

sub on_error 
{
    my( $op, $errnum, $errstr ) = @_;
    # Handle this like you would any POE socket error
    # But remember you can't rely on your session being active
}

تستخدم monitor في error لتلقي الأخطاء أثناء تبادل الرسائل. ARG3 هو اسم العملية. ARG4 هو رسالة الخطأ. العمليات الحالية هي:

remote-request
لم تتمكن النواة البعيدة من تحليل طلب أُرسل من النواة المحلية.
remote-check
لم تنشر النواة البعيدة حدثًا أُرسل من النواة المحلية.
remote-resolve
لم تجد النواة البعيدة جلسة يمكنها معالجة الطلب.
remote-invocation
حدث خطأ في النواة البعيدة عند محاولة استدعاء معالج الطلب. ملاحظة: هذا لا يلتقط الأخطاء في معالج الطلب، لكنه يلتقط الأخطاء في thunk.
local-request
local-check
local-resolve
local-invocation
هذه العمليات الأربع هي المكافئ المحلي للعمليات الأربع السابقة. وهي مخصصة للتسجيل. بشكل عام لا تتطلب أي إجراءات.

لاحظ أن 'local' و'remote' تشيران إلى مكان حدوث العملية، وليس إلى مكان نشوء الطلب. على سبيل المثال، ترسل النواة A طلبًا poe://B/foo/bar إلى النواة B. لم تنشر النواة B ذلك الحدث. سترى المراقبات في النواة A رسالة remote-check. وسترى المراقبات في النواة B رسالة local-check.

channel-error
تلقي الأخطاء أثناء تبادل الرسائل. أخطاء القناة مكافئة لأخطاء عجلة POE. ستكون الرسالة "[$errnum] $errstr".
subscribe
خطأ في الاشتراك في جلسة بعيدة.
fork
فشل POE::Component::IKC::Server في التفريع.
resolve
خطأ عند محاولة العثور على نواة أو جلسة بعيدة.

مثال على مراقب لأحداث الأخطاء:

sub monitor_error
{   
    my( $self, $rid, $kernel, $real, $data, $op, $message ) = 
            @_[ OBJECT, ARG0 ... $#_ ];
    if( $op =~ /^channel-/ and $message =~ /\$(\d+)\](.*)/ ) {
        return unless $real;
        my( $errnum, $errstr ) = ( $1, $2 );
        if( $op eq 'channel-read' and $errnum == 0 ) {
            warn "Connection closed";
            return;
        }
    }
    warn "Error during $op: $message";
}

على وجه الخصوص، ستلاحظ أنك لا تفعل شيئًا عند اكتشاف إغلاق القناة. بدلاً من ذلك، يُوصى بمحاولة إعادة الاتصال في حدث "unregister".

"create_ikc_responder"

مرسل الوكيل

معالجات الأحداث التي تُستدعى عبر IKC لديها جلسة مرسل وكيل. يمكنك استخدامها لإرسال ردود إلى الجلسة البعيدة.

$poe_kernel->post( $_[SENDER], 'response', @args );

عادةً ما تكون جلسة الوكيل هذه متاحة أثناء استدعاء معالج الأحداث. يمكنك الاحتفاظ بها لمدة أطول عن طريق إنشاء مرجع خارجي:

$heap->{remote} = $_[SENDER]->ID;
$poe_kernel->refcount_increment( $heap->{remote}, 'MINE' );

سيكتشف POE::Component::IKC ذلك وسينشئ جلسة وكيل جديدة للاستدعاءات المستقبلية. بعد ذلك، ستكون مسؤولاً عن تحرير الجلسة:

$poe_kernel->refcount_decrement( $heap->{remote}, 'MINE' );

ملاحظة: ستحتاج إلى نشر أي حدث سيُنشر كرد.

الأخطاء

إرسال مراجع الجلسات أو coderef إلى نواة خارجية فكرة سيئة. في مرحلة ما سيكون من المرغوب اللجوء إلى المعاملات وتحويل أي مرجع جلسة إلى محدد حالة.

حالة "rsvp" في الاستدعاء مشكلة بعض الشيء. يسمح IKC بنشرها مرة واحدة، لكنه لا يتحقق مما إذا كانت النواة الخارجية هي الصحيحة.

لا يقوم "retract" بتحديث النوى الخارجية حول سحب جلسة أو حالة.

الاستدعاء "call() لحالة في جلسة نواة خارجية لا يعمل لأسباب واضحة.
