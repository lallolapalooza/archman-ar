POE__Component__IKC(3pm)

الحزمة: perl-poe-component-ikc

التراخيص: GPL, PerlArtistic

IKC(3) وثائق Perl المساهمة من المستخدم IKC(3)

الاسم

POE::Component::IKC -- الاتصال بين نواة POE

الملخص

خادم IKC

استخدم POE::Component::IKC::Server;
# قم بإنشاء جميع الجلسات الخاصة بك
POE::Component::IKC::Server->spawn(
            port=>30, name=>'Server'
    ); # هناك خيارات أخرى متاحة
$poe_kernel->run();

عميل IKC

استخدم POE::Component::IKC::Client;
POE::Component::IKC::Client->spawn( 
            host=>name, 
            port=>30, 
            name=>'Client',
            on_connect=>\&build);
$poe_kernel->run();
sub build
{
    # قم بإنشاء جلسات تعتمد على النواة الخارجية.
}

إرسال حالة إلى نواة خارجية

$kernel->post('IKC', 'post', "poe://Server/session/state", $ONE_arg);

IKC نظير إلى نظير.

يمكن للخادم الإرسال إلى العميل.

$kernel->post('IKC', 'post', 'poe://Client/session/state', $ONE_arg);

استدعاء حالة على نواة بعيدة

تكون دلالات الاستدعاء مستحيلة، لأنها ستتسبب في حظر POE. استدعاء IKC مختلف قليلاً. إنه 'post'، ولكنه يحتوي على معلمة RSVP إضافية.

$kernel->post('IKC', 'call', 'poe://Server/hello/world', $ONE_arg, 
              'poe:callback');

سيؤدي هذا إلى إرسال القيمة المرجعة للحالة الخارجية إلى الحالة 'callback' في الجلسة الحالية. قد ترغب في أن تكون الاستدعاءات في جلسة أخرى، لكنني لا أعتقد أن هذه فكرة جيدة.

$kernel->post('IKC', 'call', 'poe://Server/hello/world', $ONE_arg, 
              'poe:/elsewhere/hi');

ملاحظة: إذا استخدمت ->call('IKC')، فستعيد عدد الأنوية الخارجية التي تم إرسال الحالة إليها. هذه طريقة مفيدة لمعرفة ما إذا كنت لا تزال متصلاً بنواة خارجية.

بعض السحر

إذا تم نشر حالة بواسطة نواة خارجية، فإن $_[SENDER] يكون صالحًا فقط خلال تلك الحالة. ومع ذلك، ستتمكن من النشر مرة أخرى إليها.

$kernel->post($_[SENDER], 'something', 'the answer is foo');

يجب على المستدعي البعيد نشر الحالات حتى يمكن استدعاؤها، أليس كذلك؟

نشر / اشتراك

يجب عليك نشر واجهات الجلسة لتكون متاحة للأنوية البعيدة.

إذا اشتركت في جلسة بعيدة، يمكنك الوصول إليها كما لو كانت جلسة محلية.

أولاً، تنشر الجلسة واجهاتها:

$kernel->post('IKC', 'publish', 'session_alias',
                    [qw(state1 state2 state3 state4)], );

ثم تشترك نواة خارجية فيها:

# ابحث عن جلسة على جميع الأنوية الخارجية المعروفة
$kernel->post('IKC', 'subscribe', [qw(poe://*/session_alias/)]);
# ابحث عن جلسة على نواة خارجية محددة
$kernel->post('IKC', 'subscribe', [qw(poe://Pulse/timeserver)]);
# تأكد من أن الجلسة لديها حالة معينة
$kernel->post('IKC', 'subscribe', [qw(poe://*/timeserver/connect)]);

بعد الاشتراك، يتم إنشاء جلسة وكيل يمكن الوصول إليها مثل أي جلسة قديمة، على الرغم من أن ->call() يعمل بنفس طريقة ->post() لأسباب واضحة:

$kernel->post('poe:/Pulse/timeserver', 'state', $arg1, $arg2...);

حاليًا، يجب أن يكون اسم مستعار للجلسة المستخدم للنشر إلى الجلسة الوكيلة هو نفس الاسم المستخدم عند الاشتراك. نظرًا لأن الأنوية لها أسماء متعددة، إذا كنت تستخدم '*' كاسم للنواة عند الاشتراك، فقد لا يكون الاسم المستعار للجلسة كما تعتقد. انظر "المراقب" لمزيد من التفاصيل.

بالطبع، محاولة النشر إلى جلسة وكيل قبل إنشائها ستكون مشكلة. ليتم تنبيهك عند إنشاء الجلسة الوكيلة، يمكن تحديد حالة استدعاء،

$kernel->post('IKC', 'subscribe', [qw(poe://*/timeserver)], 
        'timeserver_subscribed');

سيتم استدعاء الاستدعاء مع قائمة بجميع الجلسات التي تمكن من الاشتراك فيها. يجب عليك التحقق من هذه القائمة قبل المتابعة. والأفضل من ذلك، يمكنك استخدام مراقب IKC (انظر أدناه).

يمكن للمرء أيضًا ترك POE::Component::IKC::Client->spawn يتعامل مع جميع التفاصيل.

POE::Component::IKC::Client->spawn(
        port=>31337, name=>$name,
        subscribe=>[qw(poe://*/timeserver)],
        on_connect=>\&create_me,
    );

يتم استدعاء 'on_connect' فقط عندما يتم قبول جميع الاشتراكات. إذا تم رفض اشتراك، فسيتخلى create_ikc_client. إذا تم الاستعلام عن نوى خارجية متعددة لجلسة (كما هو الحال في ما سبق)، يعتبر الاشتراك ناجحًا إذا قبلت نواة خارجية واحدة على الأقل الاشتراك.

للتراجع عن الأشياء:

$kernel->post(IKC=>'retract', 'session_alias'=>[qw(states)]);
$kernel->post(IKC=>'unsubscribe', [qw(poe://Pulse/timeserver)]);

المراقب

قل أنك تريد مراقبة جميع الأنوية البعيدة التي تتصل بك:

$kernel->post(IKC=>'monitor', '*'=>{register=>'some_event'});
sub some_event
{
    my($name, $real)=@_[ARG1, ARG2];
    print "- Remote kernel ", ($real ? '' : "alias "), "$name connected\n";
}

لاحقًا، تريد معرفة متى تنفصل جلسة بعيدة معينة:

$kernel->post(IKC=>'monitor', some_kernel=>{unregister=>'bye_bye'});

أو ربما تعتقد أن الجلسة يجب أن تقوم بالتنظيف والمغادرة كلما انتهى IKC.

$kernel->post(IKC=>'monitor', '*'=>{shutdown=>'other_event'});
sub other_event
{
    # قتل العجلات، والإنذارات، والاختيارات، والأسماء المستعارة هنا
}

انظر "monitor" في POE::Component::IKC::Responder لمزيد من التفاصيل.

إيقاف التشغيل

عندما تشعر أن الوقت مناسب وتريد التخلص من جميع الجلسات المتعلقة بـ IKC، قم بما يلي:

$kernel->post(IKC=>'shutdown');

ويجب أن تختفي جميعها. في أسوأ الأحوال، لا يزال لدى البعض أسماء مستعارة مسجلة، لكن هذا لن يمنع النواة من الخروج.

النواة المحلية

يمكنك النشر إلى النواة المحلية كما لو كانت بعيدة:

$kernel->post(IKC=>'post', 
                "poe://$kernel->ID/session/state'=>$ONE_arg);

ومع ذلك، لا يمكنك حاليًا الاشتراك في الجلسات المحلية. لا أعرف كيف سأحل هذا.

الوصف

هذا هو الاتصال بين نواة POE. يستخدم لنقل الأحداث من نواة POE إلى أخرى.

انظر أيضًا

POE::Component::IKC::Responder -- قلب النظام

POE::Component::IKC::Server -- إنشاء عملية تستمع إلى الأنوية الأخرى.

POE::Component::IKC::Client -- إنشاء عملية تتصل بالأنوية الأخرى.

POE::Component::IKC::ClientLite -- تنفيذ IKC خفيف الوزن للأماكن التي لا يمكنك فيها استخدام POE، مثل mod_perl.

POE::Component::IKC::Channel -- التعامل مع الاتصالات مع الأنوية الأخرى.

POE::Component::IKC::Proxy -- جلسة وكيل يتم إنشاؤها عند الاشتراك في جلسة بعيدة.

POE::Component::IKC::Freezer -- طريقة تسلسل Perl خالصة.

POE::Component::IKC::Specifier -- روتينات مساعدة لتحليل محددات IKC.
