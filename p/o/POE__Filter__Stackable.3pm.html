POE__Filter__Stackable(3pm)

الحزمة: perl-poe

التراخيص: GPL, PerlArtistic

POE::Filter::Stackable(3) وثائق Perl مساهمة من المستخدم POE::Filter::Stackable(3)

الاسم

POE::Filter::Stackable - تجميع كائنات POE::Filter متعددة

الملخص

#!perl
use POE qw(
  Wheel::FollowTail
  Filter::Line Filter::Grep Filter::Stackable
);
POE::Session->create(
  inline_states => {
    _start => sub {
      my $parse_input_as_lines = POE::Filter::Line->new();
      my $select_sudo_log_lines = POE::Filter::Grep->new(
        Put => sub { 1 },
        Get => sub {
          my $input = shift;
          return $input =~ /sudo\[\d+\]/i;
        },
      );
      my $filter_stack = POE::Filter::Stackable->new(
        Filters => [
          $parse_input_as_lines, # أولاً عند الإدخال، أخيراً عند الإخراج
          $select_sudo_log_lines, # أولاً عند الإخراج، أخيراً عند الإدخال
        ]
      );
      $_[HEAP]{tailor} = POE::Wheel::FollowTail->new(
        Filename => "/var/log/system.log",
        InputEvent => "got_log_line",
        Filter => $filter_stack,
      );
    },
    got_log_line => sub {
      print "Log: $_[ARG0]\n";
    }
  }
);
POE::Kernel->run();
exit;

الوصف

يجمع POE::Filter::Stackable بين عوامل تصفية متعددة بطريقة تجعلها تبدو كعامل تصفية واحد. تعمل جميع طرق POE::Filter المعتادة، ولكن يتم تمرير البيانات سرًا عبر عوامل التصفية المكدسة قبل إرجاعها. لا تحتاج كائنات POE::Wheel والبرامج المستقلة إلى تعديلات للعمل مع مكدس عوامل التصفية.

في "الملخص"، يتم دمج POE::Filter::Line و POE::Filter::Grep في عامل تصفية واحد يعيد نوعًا معينًا فقط من الأسطر. يمكن أن يكون هذا أكثر كفاءة من تصفية الأسطر في مساحة التطبيق، حيث قد تحتاج أحداث أقل إلى الإرسال والمعالجة.

داخليًا، يتم تخزين عوامل التصفية في مصفوفة.

ستتدفق البيانات المضافة بواسطة get_one_start() عبر مصفوفة عوامل التصفية بترتيب زيادة الفهرس. سيحصل عامل التصفية #0 على الفرصة الأولى لمعالجتها، يليه عامل التصفية #1 وهكذا. ستعيد استدعاء get_one() عنصرًا بعد مروره عبر عامل التصفية الأخير.

تمرر put() البيانات عبر عوامل التصفية بترتيب تنازلي للفهرس. ستمر البيانات عبر عامل التصفية ذي الفهرس الأعلى أولاً، وستعيد put() النتائج بعد مرور البيانات عبر عامل التصفية #0.

طرق عوامل التصفية العامة

بالإضافة إلى طرق POE::Filter المعتادة، يدعم POE::Filter::Stackable ما يلي أيضًا.

جديد

بشكل افتراضي، تنشئ new() مكدس عوامل تصفية فارغًا يتصرف مثل POE::Filter::Stream. يمكن تمرير معلمات اختيارية لتهيئة المكدس بمصفوفة من عوامل التصفية.

my $sudo_lines = POE::Filter::Stackable->new(
  Filters => [
    POE::Filter::Line->new(),
    POE::Filter::Grep->new(
      Put => sub { 1 }, # إخراج جميع العناصر
      Get => sub { shift() =~ /sudo\[\d+\]/i },
    ),
  ]
);

pop

تتصرف مثل pop() المضمنة في Perl لمكدس عوامل التصفية. تتم إزالة عامل التصفية ذي الفهرس الأعلى من المكدس وإعادته. يتم فقدان أي بيانات متبقية في مخزن الإدخال المؤقت لعامل التصفية، ولكن يمكن للتطبيق دائمًا استدعاء "get_pending" في POE::Filter على عامل التصفية المُعاد.

my $last_filter = $stackable->pop();
my $last_buffer = $last_filter->get_pending();

shift

تتصرف مثل shift() المضمنة في Perl لمكدس عوامل التصفية. تتم إزالة عامل التصفية ذي الفهرس 0 من المكدس وإعادته. يتم تمرير أي بيانات متبقية في مخزن الإدخال المؤقت لعامل التصفية إلى رأس المكدس الجديد، أو يتم فقدانها إذا أصبح المكدس فارغًا. يمكن للتطبيق أيضًا استدعاء "get_pending" في POE::Filter على عامل التصفية المُعاد لفحص مخزن الإدخال المؤقت لعامل التصفية.

my $first_filter = $stackable->shift();
my $first_buffer = $first_filter->get_pending();

push FILTER[, FILTER]

تضيف push() عامل تصفية (عوامل تصفية) جديدًا (جدد) إلى نهاية المكدس. ستقوم عوامل التصفية المدفوعة حديثًا بمعالجة الإدخال أخيرًا، وستتعامل مع الإخراج أولاً.

# عكس البيانات المقروءة عبر المكدس.
# تشفير rot13 للبيانات المرسلة عبر المكدس.
$stackable->push(
  POE::Filter::Map->(
    Get => sub { return scalar reverse shift() },
    Put => sub { local $_ = shift(); tr[a-zA-Z][n-za-mN-ZA-M]; $_ },
  )
);

unshift FILTER[, FILTER]

تضيف unshift() عامل تصفية (عوامل تصفية) جديدًا (جدد) إلى بداية المكدس. ستقوم عوامل التصفية المُضافة حديثًا بمعالجة الإدخال أولاً، وستتعامل مع الإخراج أخيرًا.

filters

تعيد filters() قائمة بعوامل التصفية داخل عامل التصفية Stackable، بترتيب المكدس الأصلي.

استدعاء "$filter_stack->filters()" في "الملخص" سيعيد قائمة بعاملي تصفية:

POE::Filter::Line=ARRAY(0x8b5ee0)
POE::Filter::Grep=ARRAY(0x8b5f7c)

filter_types

تعيد filter_types() قائمة بأسماء الفئات لكل عامل تصفية في المكدس، بترتيب المكدس الأصلي.

استدعاء "$filter_stack->filter_types()" في "الملخص" سيعيد قائمة باسمي فئة:

POE::FIlter::Line
POE::Filter::Grep

يمكن استبدالها بسهولة بـ:

my @filter_types = map { ref } $filter_stack->filters;

انظر أيضاً

POE لمزيد من المعلومات حول عوامل التصفية بشكل عام.

عوامل تصفية محددة، من بينها: POE::Filter::Block, POE::Filter::Grep, POE::Filter::HTTPD, POE::Filter::Line, POE::Filter::Map, POE::Filter::RecordBlock, POE::Filter::Reference, POE::Filter::Stream
