POD2__JA__MIME__Charset(3pm)

الحزمة: perl-mime-charset

التراخيص: PerlArtistic, GPL

POD2::JA::MIME::Charset(3) وثائق Perl مقدمة من المستخدم POD2::JA::MIME::Charset(3)

الاسم

MIME::Charset~[ja] - معلومات مجموعة الأحرف لـ MIME

الملخص

use MIME::Charset:
$charset = MIME::Charset->new("euc-jp");

الحصول على معلومات مجموعة الأحرف:

$benc = $charset->body_encoding; # مثال: "Q"
$cset = $charset->as_string; # مثال: "US-ASCII"
$henc = $charset->header_encoding; # مثال: "S"
$cset = $charset->output_charset; # مثال: "ISO-2022-JP"

تحويل بيانات النص:

($text, $charset, $encoding) =
    $charset->header_encode(
       "\xc9\xc2\xc5\xaa\xc0\xde\xc3\xef\xc5\xaa".
       "\xc7\xd1\xca\xaa\xbd\xd0\xce\xcf\xb4\xef",
       Charset => 'euc-jp');
# ... مثال: يعيد ("سلسلة نصية محولة", "ISO-2022-JP", "B")
($text, $charset, $encoding) =
    $charset->body_encode(
        "Collectioneur path\xe9tiquement ",
        Charset => 'latin1');
# ... مثال: يعيد ("السلسلة النصية الأصلية", "ISO-8859-1", "QUOTED-PRINTABLE")
$len = $charset->encoded_header_len(
    "Perl\xe8\xa8\x80\xe8\xaa\x9e",
    Charset => "utf-8",
    Encoding => "b");
# ... مثال: يعيد 28

عمل القيم الافتراضية للوحدة:

MIME::Charset::alias("csEUCKR", "euc-kr");
MIME::Charset::default("iso-8859-1");
MIME::Charset::fallback("us-ascii");

وظائف غير كائنية (سيتم إهمالها قريبًا):

use MIME::Charset qw(:info);
$benc = body_encoding("iso-8859-2"); # "Q"
$cset = canonical_charset("ANSI X3.4-1968"); # "US-ASCII"
$henc = header_encoding("utf-8"); # "S"
$cset = output_charset("shift_jis"); # "ISO-2022-JP"
use MIME::Charset qw(:trans);
($text, $charset, $encoding) =
    header_encode(
       "\xc9\xc2\xc5\xaa\xc0\xde\xc3\xef\xc5\xaa".
       "\xc7\xd1\xca\xaa\xbd\xd0\xce\xcf\xb4\xef",
       "euc-jp");
# ... (يعيد <النص المحول>, "ISO-2022-JP", "B")
($text, $charset, $encoding) =
    body_encode(
        "Collectioneur path\xe9tiquement ".
        "\xe9clectique de d\xe9chets",
        "latin1");
# ... (يعيد <النص الأصلي>, "ISO-8859-1", "QUOTED-PRINTABLE")
$len = encoded_header_len(
    "Perl\xe8\xa8\x80\xe8\xaa\x9e", "b", "utf-8"); # 28

الوصف

توفر MIME::Charset معلومات حول مجموعات الأحرف المستخدمة في رسائل MIME على الإنترنت.

التعريف

مجموعة الأحرف تشير إلى "character set" في MIME، وهي طريقة لترجمة تسلسل من البايتات إلى تسلسل من الأحرف. وهي تتضمن مفهوم كل من "مجموعة أحرف الرموز" (CCS) و "آلية ترميز الأحرف" (CES) وفقًا لـ ISO/IEC.

الترميز هو المصطلح الذي يشير إلى ذلك في MIME، وهو ...
