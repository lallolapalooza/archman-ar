POE__Filter__Map(3pm)

الحزمة: perl-poe

التراخيص: GPL, PerlArtistic

POE::Filter::Map(3) توثيق Perl المساهم به POE::Filter::Map(3)

الاسم

POE::Filter::Map - تحويل المدخلات و/أو المخرجات ضمن مكدس مرشحات

الملخص

#!perl
use POE qw(
  Wheel::FollowTail
  Filter::Line Filter::Map Filter::Stackable
);
POE::Session->create(
  inline_states => {
    _start => sub {
      my $parse_input_as_lines = POE::Filter::Line->new();
      my $redact_some_lines = POE::Filter::Map->new(
        Code => sub {
          my $input = shift;
          $input = "[REDACTED]" unless $input =~ /sudo\[\d+\]/i;
          return $input;
        },
      );
      my $filter_stack = POE::Filter::Stackable->new(
        Filters => [
          $parse_input_as_lines, # أول في الاستقبال، آخر في الإرسال
          $redact_some_lines, # أول في الإرسال، آخر في الاستقبال
        ]
      );
      $_[HEAP]{tailor} = POE::Wheel::FollowTail->new(
        Filename => "/var/log/system.log",
        InputEvent => "got_log_line",
        Filter => $filter_stack,
      );
    },
    got_log_line => sub {
      print "Log: $_[ARG0]\n";
    }
  }
);
POE::Kernel->run();
exit;

الوصف

يقوم POE::Filter::Map بتحويل البيانات داخل مكدس المرشحات. يمكن استخدامه لتحويل المدخلات، المخرجات، أو كليهما حسب كيفية إنشائه. تم تسمية هذا المرشح ونمذجته بعد الدالة المضمنة في Perl map().

تم تصميم POE::Filter::Map للدمج مع مرشحات أخرى من خلال POE::Filter::Stackable. في مثال "الملخص"، تم إنشاء مكدس مرشحات لتحليل السجلات كسطور وحذف جميع الإدخالات التي لا تتعلق بعملية sudo.

طرق المرشحات العامة

بالإضافة إلى طرق POE::Filter المعتادة، يدعم POE::Filter::Map أيضًا ما يلي:

new

يقوم new() بإنشاء كائن POE::Filter::Map جديد. يجب استدعاؤه إما بمعامل Code واحد، أو بمعاملات Put و Get. القيم الخاصة بـ Code، Put، و Get هي مراجع كود، عند استدعائها، تُرجع إصدارات محسنة من معاملاتها الوحيدة. سيتم استخدام دالة Code لكل من المدخلات والمخرجات، بينما تسمح دوال Get و Put بتصفية المدخلات والمخرجات بطرق مختلفة.

# فك تشفير rot13.
sub decrypt_rot13 {
  my $encrypted = shift;
  $encrypted =~ tr[a-zA-Z][n-za-mN-ZA-M];
  return $encrypted;
}
# تشفير rot13.
sub encrypt_rot13 {
  my $plaintext = shift;
  $plaintext =~ tr[a-zA-Z][n-za-mN-ZA-M];
  return $plaintext;
}
# فك تشفير rot13 عند المدخل، وتشفيره عند المخرج.
my $rot13_transcrypter = POE::Filter::Map->new(
  Get => \&decrypt_rot13,
  Put => \&encrypt_rot13,
);

Rot13 متناظر، لذلك يمكن تبسيط المثال أعلاه لاستخدام دالة Code واحدة.

my $rot13_transcrypter = POE::Filter::Map->new(
  Code => sub {
    local $_ = shift;
    tr[a-zA-Z][n-za-mN-ZA-M];
    return $_;
  }
);

modify

يقوم modify() بتغيير سلوك كائن POE::Filter::Map في وقت التشغيل. يقبل نفس المعاملات مثل new()، ويستبدل التحويلات الحالية بتحويلات جديدة.

# التبديل إلى تشفير "عكسي" للاختبار.
$rot13_transcrypter->modify(
  Code => sub { return scalar reverse shift }
);

انظر أيضًا

POE::Filter لمزيد من المعلومات حول المرشحات بشكل عام.

POE::Filter::Stackable لمزيد من التفاصيل حول تكديس المرشحات.
