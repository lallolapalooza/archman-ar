Pod__Select(3pm)

الحزمة: perl-pod-parser

التراخيص: PerlArtistic

Pod::Select(3) وثائق Perl مساهمة من المستخدم Pod::Select(3)

الاسم

Pod::Select, podselect() - استخراج أقسام POD المحددة من المدخلات

الملخص

use Pod::Select;
## تحديد كافة أقسام POD لكل ملف في @filelist
## وطباعة النتيجة على المخرج القياسي.
podselect(@filelist);
## نفس ما سبق، ولكن الكتابة إلى tmp.out
podselect({-output => "tmp.out"}, @filelist):
## التحديد من قائمة الملفات المعطاة، فقط أقسام POD التي تقع
## ضمن قسم من المستوى الأول يسمى أي مما يلي: NAME, SYNOPSIS, OPTIONS.
podselect({-sections => ["NAME|SYNOPSIS", "OPTIONS"]}, @filelist):
## تحديد قسم "DESCRIPTION" من PODs من STDIN وكتابة
## النتيجة إلى STDERR.
podselect({-output => ">&STDERR", -sections => ["DESCRIPTION"]}, \*STDIN);

أو

use Pod::Select;
## إنشاء كائن محلل لتحديد أقسام POD من المدخلات
$parser = new Pod::Select();
## تحديد كافة أقسام POD لكل ملف في @filelist
## وطباعة النتيجة إلى tmp.out.
$parser->parse_from_file("<&STDIN", "tmp.out");
## التحديد من قائمة الملفات المعطاة، فقط أقسام POD التي تقع
## ضمن قسم من المستوى الأول يسمى أي مما يلي: NAME, SYNOPSIS, OPTIONS.
$parser->select("NAME|SYNOPSIS", "OPTIONS");
for (@filelist) { $parser->parse_from_file($_); }
## تحديد قسمي "DESCRIPTION" و "SEE ALSO" من PODs من
## STDIN وكتابة النتيجة إلى STDERR.
$parser->select("DESCRIPTION");
$parser->add_selection("SEE ALSO");
$parser->parse_from_filehandle(\*STDIN, \*STDERR);

يتطلب

perl5.005, Pod::Parser, Exporter, Carp

التصدير

podselect()

الوصف

ملاحظة: تعتبر هذه الوحدة قديمة؛ الإصدارات الحديثة من Perl (5.31.1 وما فوق) ستقوم بإزالة Pod-Parser من النواة واستخدام Pod::Simple لجميع الأمور المتعلقة بـ POD.

podselect() هي وظيفة ستقوم باستخراج أقسام محددة من وثائق pod من تيار إدخال. يتم توفير هذه القدرة بواسطة الوحدة Pod::Select وهي فئة فرعية من Pod::Parser. توفر Pod::Select طريقة تسمى select() لتحديد مجموعة أقسام POD التي سيتم تحديدها للمعالجة/الطباعة. podselect() تقوم ببساطة بإنشاء كائن Pod::Select ثم تستدعي podselect() متبوعة بـ parse_from_file().

مواصفات الأقسام

يمكن إعطاء podselect() و Pod::Select::select() مواصفة قسم واحدة أو أكثر "لتحديد" النص الذي تتم معالجته فقط للمجموعة المطلوبة من الأقسام والأقسام الفرعية المقابلة لها. مواصفات القسم هي سلسلة نصية تحتوي على تعبيرات نمطية واحدة أو أكثر بأسلوب Perl مفصولة بشرطة مائلة ("/") . إذا كنت بحاجة إلى استخدام شرطة مائلة حرفيًا داخل عنوان قسم، يمكنك الهروب منها بشرطة مائلة عكسية ("\/").

الصيغة الرسمية لمواصفات القسم هي:

•
head1-title-regex/head2-title-regex/...

سيتم افتراض أي تعبيرات نمطية مفقودة أو فارغة كـ ".*". يرجى ملاحظة أن كل تعبير نمطي معطى يتم تثبيته ضمنيًا عن طريق إضافة "^" و "$" إلى البداية والنهاية. أيضًا، إذا بدأ تعبير نمطي معطى بحرف "!" ، فسيتم عكس التعبير (لذا فإن "!foo" سيطابق أي شيء ما عدا "foo").

فيما يلي بعض الأمثلة لمواصفات الأقسام.

  • مطابقة أقسام "NAME" و "SYNOPSIS" وجميع أقسامها الفرعية:

    "NAME|SYNOPSIS"

  • مطابقة فقط الأقسام الفرعية "Question" و "Answer" من قسم "DESCRIPTION":

    "DESCRIPTION/Question|Answer"

  • مطابقة قسم "Comments" الفرعي من جميع الأقسام:

    "/Comments"

  • مطابقة جميع الأقسام الفرعية من "DESCRIPTION" باستثناء "Comments":

    "DESCRIPTION/!Comments"

  • مطابقة قسم "DESCRIPTION" ولكن عدم مطابقة أي من أقسامه الفرعية:

    "DESCRIPTION/!.+"

  • مطابقة جميع الأقسام من المستوى الأعلى ولكن لا شيء من أقسامها الفرعية:

    "/!.+"

طرق الكائن

الطرق التالية مقدمة في هذه الوحدة. كل منها يأخذ مرجعًا للكائن نفسه كمعلمة أولى ضمنية.

curr_headings()

($head1, $head2, $head3, ...) = $parser->curr_headings();
$head1 = $parser->curr_headings(1);

ترجع هذه الطريقة قائمة بالعناوين والأقسام الفرعية النشطة حاليًا في المستند الذي تتم معالجته. تتوافق قائمة العناوين التي تم إرجاعها مع أحدث فقرة تمت معالجتها من المدخلات.

إذا تم تقديم وسيط، فيجب أن يتوافق مع رقم عنوان القسم المطلوب، وفي هذه الحالة يتم إرجاع عنوان القسم المحدد فقط. إذا لم يكن هناك عنوان قسم حالي في المستوى المحدد، فسيتم إرجاع "undef".

select()

$parser->select($section_spec1,$section_spec2,...);

تستخدم هذه الطريقة لتحديد أقسام وأقسام فرعية معينة من وثائق POD التي سيتم طباعتها و/أو معالجتها. يتم استبدال مجموعة الأقسام المحددة حاليًا بالمجموعة المعطاة من الأقسام. انظر add_selection() للإضافة إلى مجموعة الأقسام المحددة حاليًا.

يجب أن تكون كل من وسائط $section_spec مواصفات قسم كما هو موضح في "مواصفات الأقسام". يتم تحليل مواصفات الأقسام بواسطة هذه الطريقة ويتم تخزين التعبيرات النمطية الناتجة في الكائن المستدعي.

إذا لم يتم تقديم وسائط $section_spec، فسيتم مسح مجموعة الأقسام المحددة حاليًا (مما يعني أنه سيتم معالجة جميع الأقسام).

لا ينبغي تجاوز هذه الطريقة بشكل طبيعي بواسطة الفئات الفرعية.

add_selection()

$parser->add_selection($section_spec1,$section_spec2,...);

تستخدم هذه الطريقة للإضافة إلى الأقسام والأقسام الفرعية المحددة حالياً من وثائق POD التي سيتم طباعتها و/أو معالجتها. انظر <select()> لاستبدال الأقسام المحددة حاليًا.

يجب أن تكون كل من وسائط $section_spec مواصفات قسم كما هو موضح في "مواصفات الأقسام". يتم تحليل مواصفات الأقسام بواسطة هذه الطريقة ويتم تخزين التعبيرات النمطية الناتجة في الكائن المستدعي.

لا ينبغي تجاوز هذه الطريقة بشكل طبيعي بواسطة الفئات الفرعية.

clear_selections()

$parser->clear_selections();

لا تأخذ هذه الطريقة أي وسائط، ولها نفس تأثير استدعاء <select()> بدون وسائط.

match_section()

$boolean = $parser->match_section($heading1,$heading2,...);

ترجع قيمة صحيحة إذا كانت عناوين القسم والأقسام الفرعية المعطاة تطابق أي من مواصفات الأقسام المحددة حاليًا والنافذة من استدعاءات سابقة لـ select() و add_selection() (أو إذا لم تكن هناك أقسام محددة/ملغاة بشكل صريح).

الوسائط $heading1، $heading2، إلخ هي عناوين الأقسام والأقسام الفرعية، إلخ، التي يتم محاولة مطابقتها. إذا تم حذف $headingN، فإنها تأخذ القيمة الافتراضية لعنوان القسم المقابل الحالي في المدخلات.

لا ينبغي تجاوز هذه الطريقة بشكل طبيعي بواسطة الفئات الفرعية.
