POE__Filter__Block(3pm)

الحزمة: perl-poe

التراخيص: GPL, PerlArtistic

POE::Filter::Block(3) وثائق مساهمة من مستخدمي Perl POE::Filter::Block(3)

الاسم

POE::Filter::Block - ترجمة البيانات بين التدفقات والكتل

الملخص

#!perl
use warnings;
use strict;
use POE::Filter::Block;
my $filter = POE::Filter::Block->new( BlockSize => 8 );
# تطبع ثلاثة أسطر: abcdefgh، ijklmnop، qrstuvwx.
# تبقى البايتات "y" و "z" في المخزن المؤقت وتنتظر اكتمال الكتلة
# المكونة من 8 بايتات التالية.
$filter->get_one_start([ "abcdefghijklmnopqrstuvwxyz" ]);
while (1) {
  my $block = $filter->get_one();
  last unless @$block;
  print $block->[0], "\n";
}
# تطبع سطراً واحداً: yz123456
$filter->get_one_start([ "123456" ]);
while (1) {
  my $block = $filter->get_one();
  last unless @$block;
  print $block->[0], "\n";
}

الوصف

POE::Filter::Block يترجم البيانات بين التدفقات التسلسلية والكتل. يمكنه التعامل مع الكتل ذات الطول الثابت والكتل ذات البادئة الطول، ويمكن توسيعه للتعامل مع أنواع الكتل الأخرى.

تُستخدم الكتل ذات الطول الثابت عندما يتم استدعاء منشئ الكتلة (Block constructor) بقيمة BlockSize. وإلا فإن فلتر الكتلة يستخدم الكتل ذات البادئة الطول.

المستخدمون الذين يحددون أحجام كتل أقل من واحد يستحقون ما يحصلون عليه.

في وضع الطول المتغير، يمكن تحديد معلمة LengthCodec. يجب أن تكون قيمة LengthCodec مرجعًا إلى قائمة من دالتين: مشفر الطول، وفك تشفير الطول:

LengthCodec => [ \&encoder, \&decoder ]

يقوم المشفر بأخذ مرجع إلى مخزن مؤقت ويضيف طول المخزن المؤقت إليه في البداية. يقوم المشفر الافتراضي بإضافة التمثيل النصي لطول المخزن المؤقت وبايت chr(0) للفصل بين الطول والبيانات الفعلية:

sub _default_encoder {
  my $stuff = shift;
  substr($$stuff, 0, 0) = length($$stuff) . "\0";
  return;
}

يقوم مفك التشفير المقابل بإرجاع طول الكتلة بعد إزالته هو وفاصل البيانات من المخزن المؤقت. يعيد لا شيء إذا لم يتم تحديد طول.

sub _default_decoder {
  my $stuff = shift;
  unless ($$stuff =~ s/^(\d+)\0//s) {
    warn length($1), " strange bytes removed from stream"
      if $$stuff =~ s/^(\D+)//s;
    return;
  }
  return $1;
}

هذا المرشح يحتفظ بالكتل غير المكتملة حتى يتم إكمالها في مخزن مؤقت تأطيري. للتحكم في استخدام الذاكرة، يتم فرض حد أقصى لحجم المخزن المؤقت التأطيري. هذا الحد الأقصى للحجم افتراضيًا هو 512 ميجابايت (512 * 1024 * 1024 بايت). يمكنك تغيير حد الحجم هذا باستخدام المعلمة "MaxBuffer".

MaxBuffer => 1099511627776  # واحد تيرابايت!

يتم أيضًا تحديد حجم كل كتلة فردية. افتراضيًا، قد لا تكون كل كتلة أكثر من 64 ميجابايت. يمكنك تغيير حد الحجم هذا باستخدام المعلمة "MaxLength".

MaxLength => 10             # كتل صغيرة

تذكر أن MaxBuffer يجب أن يكون أكبر من MaxLength. علاوة على ذلك، يجب أن يكون لديه مساحة لبادئة الطول.

إذا تم تجاوز قيد "MaxLength" أو "MaxBuffer"، فسيقوم "POE::Filter::Bock" بإطلاق استثناء.

أساليب المرشح العامة

لا يحتوي POE::Filter::Block على أي أساليب عامة إضافية.

انظر أيضًا

يرجى الرجوع إلى POE::Filter للحصول على وثائق حول الواجهة الأساسية.

قسم "انظر أيضًا" في POE يحتوي على جدول محتويات يغطي توزيع POE بالكامل.

الأخطاء

أسلوب put() لا يتحقق من أحجام الكتل.
