الحزمة: pnetcdf-openmpi

PnetCDF(3) وظائف المكتبة PnetCDF(3)

الاسم

PnetCDF - مكتبة متوازية للوصول إلى الملفات بتنسيق Network Common Data Form (CDF، CDF-2 و CDF-5).

الملخص

#include <pnetcdf.h>

mpicc ... -lpnetcdf

يمكن العثور على التوثيق الكامل لمكتبات PnetCDF على موقع PnetCDF: http://cucis.ece.northwestern.edu/projects/PnetCDF/.

إصدار المكتبة

يصف هذا المستند واجهات برمجة التطبيقات لـ PnetCDF للغة البرمجة C.

const char* ncmpi_inq_libvers()

(يقابل nc_inq_libvers() في netCDF)

يعيد سلسلة نصية تحدد إصدار مكتبة PnetCDF، ووقت بنائها، مثل: "1.14.1 of 2025-07-31".

يجد الأمر RCS ident(1) سلسلة مثل "$Id: @(#) PnetCDF library version 1.14.1 of 2025-07-31 $" في المكتبة. يجد الأمر SCCS what(1) سلسلة مثل "PnetCDF library version 1.14.1 of 2025-07-31".

قيم الإرجاع

جميع وظائف PnetCDF (باستثناء ncmpi_inq_libvers() و ncmpi_strerror()) تعيد حالة عدد صحيح.

إذا كانت قيمة الحالة المرجعة هذه لا تساوي NC_NOERR (صفر)، فهذا يشير إلى حدوث خطأ. يمكن تحديد قيم الحالة الممكنة في ملف التضمين النظامي <errno.h> وفي "pnetcdf.h".

const char* ncmpi_strerror(int status)

(يقابل nc_strerror() في netCDF)

يعيد ترجمة نصية سلسلة لقيمة status، مثل "Attribute or variable name contains illegal characters" أو "No such file or directory".

عمليات الملفات

int ncmpi_create(MPI_Comm comm, const char path[], int cmode, MPI_Info info, int* ncid)

(يقابل nc_create() في netCDF)

ينشئ مجموعة بيانات netCDF جديدة في path بشكل جماعي بواسطة مجموعة من عمليات MPI المحددة بواسطة comm، ويعيد معرّف netCDF في ncid. يمكن أن تتضمن الوسيطة cmode عملية OR بت (bitwise-or) للعلامات التالية: NC_NOCLOBBER لحماية مجموعات البيانات الموجودة (الافتراضي هو NC_CLOBBER، الذي يمسحها بصمت)، NC_64BIT_OFFSET لإنشاء ملف بتنسيق الإزاحة 64 بت (CDF-2)، مقابل التنسيق الكلاسيكي، الافتراضي، أو NC_64BIT_DATA لإنشاء ملف بتنسيق البيانات 64 بت (CDF-5). استخدم إما NC_64BIT_OFFSET أو NC_64BIT_DATA. يسمح تنسيق الإزاحة 64 بت بإنشاء ملفات كبيرة جدًا مع قيود أقل بكثير من تنسيق netCDF الكلاسيكي، ولكن لا يمكن قراءتها إلا بواسطة مكتبة netCDF الإصدار 3.6 أو أحدث. يُحذر المستخدمون من أن الملفات التي تستخدم تنسيق الإزاحة 64 بت لن يتم التعرف عليها بواسطة تطبيقات netCDF المرتبطة بإصدار أقدم من مكتبة netCDF 3.6. ستتمكن التطبيقات المرتبطة بالإصدار 3.6 أو الأحدث من الوصول بشكل شفاف إما إلى التنسيق الكلاسيكي أو تنسيق الإزاحة 64 بت. يسمح تنسيق البيانات 64 بت بإنشاء متغيرات مصفوفة كبيرة جدًا. ملفات CDF-5 حاليًا لن يتم التعرف عليها بواسطة مكتبة netCDF 3 أو 4.

يجب أن تكون الوسيطة cmode متسقة بين جميع عمليات MPI التي تنشئ الملف بشكل جماعي. الوسيطة info هي كائن معلومات MPI. يمكن للمستخدمين استخدامه لتوفير تلميحات الوصول إلى الملفات لتحسين الأداء بشكل إضافي. تشمل التلميحات تلميحات MPI-IO الموجودة بالإضافة إلى التلميحات المعرفة والمستخدمة في PnetCDF.

عند إنشاء مجموعة بيانات netCDF، يتم فتحها في وضع NC_WRITE. عندما تعود هذه الوظيفة، تكون مجموعة بيانات netCDF الجديدة في وضع التعريف.

int ncmpi_open(MPI_Comm comm, const char path[], int mode, MPI_Info info, int* ncid)

(يقابل nc_open() في netCDF)

يفتح مجموعة بيانات netCDF موجودة في path بشكل جماعي بواسطة مجموعة من عمليات MPI المحددة بواسطة comm، ويعيد معرّف netCDF في ncid. يتم وصف نوع الوصول بواسطة المعلمة mode، والتي قد تتضمن عملية OR بت (bitwise-or) للعلامات التالية: NC_WRITE للوصول للقراءة والكتابة (الافتراضي هو القراءة فقط).

يجب أن تكون الوسيطة mode متسقة بين جميع عمليات MPI التي تفتح الملف بشكل جماعي. الوسيطة info هي كائن معلومات MPI. يمكن للمستخدمين استخدامه لتوفير تلميحات الوصول إلى الملفات لتحسين الأداء بشكل إضافي. تشمل التلميحات تلميحات MPI-IO الموجودة بالإضافة إلى التلميحات المعرفة والمستخدمة في PnetCDF.

int ncmpi_redef(int ncid)

(يقابل nc_redef() في netCDF)

يضع مجموعة بيانات netCDF مفتوحة في وضع التعريف، بحيث يمكن إضافة الأبعاد والمتغيرات والسمات أو إعادة تسميتها وحذف السمات.

int ncmpi_enddef(int ncid)

(يقابل nc_enddef() في netCDF)

يخرج مجموعة بيانات netCDF مفتوحة من وضع التعريف. يتم التحقق من التغييرات التي تم إجراؤها على مجموعة بيانات netCDF أثناء وجودها في وضع التعريف ودمجها في القرص إذا لم تحدث مشاكل. بعد استدعاء ناجح، يمكن قراءة بيانات المتغير أو كتابتها إلى مجموعة البيانات.

int ncmpi_sync(int ncid)

(يقابل nc_sync() في netCDF)

قد يتم تخزين البيانات التي كتبتها واجهات برمجة تطبيقات PnetCDF مؤقتًا بواسطة نظام ملفات محلي على كل عقدة حساب. هذه الواجهة تقوم بمسح البيانات المخزنة مؤقتًا عن طريق استدعاء MPI_File_sync.

int ncmpi_abort(int ncid)

(يقابل nc_abort() في netCDF)

لا تحتاج إلى استدعاء هذه الوظيفة. يتم استدعاء هذه الوظيفة تلقائيًا بواسطة ncmpi_close() إذا كانت netCDF في وضع التعريف وحدث خطأ ما مع الدمج. إذا لم تكن مجموعة بيانات netCDF في وضع التعريف، فإن هذه الوظيفة تعادل ncmpi_close(). إذا تم استدعاؤها بعد ncmpi_redef()، ولكن قبل ncmpi_enddef()، فلن يتم دمج التعريفات الجديدة وسيتم إغلاق مجموعة البيانات. إذا تم استدعاؤها بعد ncmpi_create() ولكن قبل ncmpi_enddef()، فإن مجموعة البيانات تختفي.

int ncmpi_close(int ncid)

(يقابل nc_close() في netCDF)

يغلق مجموعة بيانات netCDF مفتوحة. إذا كانت مجموعة البيانات في وضع التعريف، فسيتم استدعاء ncmpi_enddef() قبل الإغلاق. بعد إغلاق مجموعة البيانات، يمكن إعادة تخصيص معرّفها لمجموعة بيانات أخرى.

int ncmpi_def_dim(int ncid, const char name[], MPI_Offset len, int* dimid)

(يقابل nc_def_dim() في netCDF)

يضيف بعدًا جديدًا إلى مجموعة بيانات netCDF مفتوحة، والتي يجب أن تكون في وضع التعريف. name هو اسم البعد. إذا كان dimid ليس مؤشر NULL، فعند اكتمال الوظيفة بنجاح

هذا الموقع غير مرتبط بـ Arch Linux ولا تابع له بأي شكل من الأشكال. المحتوى هو ترجمة فقط لصفحات دليل Arch المنشورة على man.archlinux.org.

n dimid سيحتوي على معرف البعد الخاص بالبعد الذي تم إنشاؤه حديثًا.

int ncmpi_def_var(int ncid, const char name[], nc_type xtype, int ndims, const int dimids[], int* varid)

(يتوافق مع nc_def_var() في netCDF)

يضيف متغيرًا جديدًا إلى مجموعة بيانات netCDF. يجب أن تكون netCDF في وضع التعريف. إذا لم يكن varid فارغًا (NULL)، فسيتم تعيينه إلى معرف متغير netCDF. سيكون ndims عدد الأبعاد للمتغير. سيكون name اسم متغير netCDF. xtype هو النوع الخارجي لـ netCDF للمتغير ويجب أن يكون أحد NC_BYTE NC_CHAR، NC_SHORT، NC_INT، NC_FLOAT، أو NC_DOUBLE، لتنسيقات ملفات CDF-1 و CDF-2. تعرف CDF-5 أنواعًا خارجية إضافية: NC_UBYTE، NC_USHORT، NC_UINT، NC_INT64، و NC_UINT64. الوسيط dimids هو متجه بمعرفات الأبعاد ndims المقابلة لأبعاد المتغير.

يحدد إحداثيات قيمة بيانات netCDF التي سيتم الوصول إليها. تبدأ المؤشرات من 0؛ وبالتالي، على سبيل المثال، القيمة الأولى لمتغير ثنائي الأبعاد هي (0,0). يجب أن يكون حجم المتجه على الأقل مساوياً لرتبة متغير netCDF المرتبط ويجب أن تتوافق عناصره، بترتيبها، مع أبعاد المتغير.
const MPI_Offset start[]
يحدد نقطة البداية للوصول إلى قيم بيانات متغير netCDF من حيث الإحداثيات المؤشرية لزاوية قسم المصفوفة. تبدأ المؤشرات من 0؛ وبالتالي، فإن القيمة الأولى للبيانات لمتغير هي (0، 0، ...، 0). يجب أن يكون حجم المتجه على الأقل مساوياً لرتبة متغير netCDF المرتبط ويجب أن تتوافق عناصره، بترتيبها، مع أبعاد المتغير.
const MPI_Offset count[]
يحدد عدد المؤشرات المختارة على طول كل بعد من قسم المصفوفة. وبالتالي، للوصول إلى قيمة واحدة، على سبيل المثال، قم بتحديد count كـ (1، 1، ...، 1). لاحظ أنه بالنسبة لعمليات الإدخال/الإخراج المتفرقة، يجب تعديل هذه الوسيطة لتكون متوافقة مع الوسيطات stride و start بحيث لا يحاول التفاعل بين الثلاثة الوصول إلى إحداثي بيانات غير صالح. تتوافق عناصر متجه count، بترتيبها، مع أبعاد المتغير.
const MPI_Offset stride[]
يحدد فاصل أخذ العينات على طول كل بعد من متغير netCDF. تتوافق عناصر متجه الخطوة، بترتيبها، مع أبعاد متغير netCDF (stride[0] يعطي فاصل أخذ العينات على طول البعد الأبطأ تغيراً لمتغير netCDF). يتم تحديد فواصل أخذ العينات بوحدات مستقلة عن النوع من العناصر (قيمة 1 تحدد العناصر المتتالية لمتغير netCDF على طول البعد المقابل، قيمة 2 تحدد كل عنصر آخر، إلخ). يتم معاملة وسيطة الخطوة NULL كـ (1، 1، ...، 1).
const MPI_Offset imap[]
يحدد التعيين بين أبعاد متغير netCDF وبنية الذاكرة الداخلية لمصفوفة البيانات. تتوافق عناصر متجه تعيين المؤشر، بترتيبها، مع أبعاد متغير netCDF (imap[0] يعطي المسافة بين عناصر المصفوفة الداخلية التي تتوافق مع البعد الأبطأ تغيراً لمتغير netCDF). يتم تحديد المسافات بين العناصر بوحدات مستقلة عن النوع من العناصر (المسافة بين العناصر الداخلية التي تحتل مواقع ذاكرة متجاورة هي 1 وليست طول البايت للعنصر كما في netCDF 2). يشير المؤشر NULL إلى أن القيم الموجودة في الذاكرة لها نفس هيكل متغير netCDF المرتبط.

تعبئة المتغيرات مسبقًا

قبل الإصدار 1.6.1، لا يدعم PnetCDF تعبئة البيانات. الوضع الافتراضي للتعبئة في PnetCDF هو NC_NOFILL، وهذا يتعارض مع مكتبة netCDF التي وضعها الافتراضي هو NC_FILL. عندما يتم تمكين وضع التعبئة، يقوم PnetCDF بتعيين قيم جميع المتغيرات المعرّفة حديثًا ذات الطول المحدود (أي تلك التي لا تحتوي على بُعد غير محدود) إلى قيمة التعبئة الخاصة بكل متغير. يتم ذلك عند استدعاء ncmpi_enddef(). يمكن تغيير قيمة التعبئة للمتغير عن القيمة الافتراضية عن طريق تعريف السمة `_FillValue' للمتغير. يجب أن تحمل هذه السمة نفس نوع المتغير وأن تكون بطول واحد.

لا يتم تعبئة المتغيرات ذات البعد غير المحدود مسبقًا في PnetCDF. هذا يتعارض أيضًا مع netCDF، الذي يقوم بتعبئة متغيرات السجلات مسبقًا. في PnetCDF، يجب إجراء تعبئة متغير السجل عن طريق استدعاء ncmpi_fill_var_rec(). لاحظ أن هذا يملأ سجلاً واحدًا فقط للمتغير.

يمكن ضبط وضع التعبئة للملف بأكمله عن طريق ncmpi_set_fill(). يتوفر أيضًا ضبط وضع التعبئة لكل متغير من خلال ncmpi_def_var_fill(). في PnetCDF، يجب تغيير وضع التعبئة في وضع التعريف. في netCDF، هذا صحيح فقط للمتغيرات ذات الحجم الثابت. بالنسبة لمتغيرات السجلات، يمكن تغيير وضع التعبئة في أي وقت في NetCDF.
