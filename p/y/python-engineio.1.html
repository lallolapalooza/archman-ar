@eio.on('disconnect')
def on_disconnect(reason):
    if reason == eio.reason.CLIENT_DISCONNECT:
        print('client disconnection')
    elif reason == eio.reason.SERVER_DISCONNECT:
        print('the server kicked me out')
    else:
        print(f'disconnect reason: {reason}')

الاتصال بالخادم

يتم إنشاء الاتصال بالخادم عن طريق استدعاء الدالة connect():

eio.connect('http://localhost:5000')

في حالة عميل asyncio، تكون الدالة عبارة عن coroutine:

await eio.connect('http://localhost:5000')

عند الاتصال، يقوم الخادم بتعيين معرف جلسة فريد للعميل. يمكن للتطبيق العثور على هذا المعرف في السمة sid:

print('my sid is', eio.sid)

إرسال الرسائل

يمكن للعميل إرسال رسالة إلى الخادم باستخدام الدالة send():

eio.send({'foo': 'bar'})

أو في حالة asyncio، كـ coroutine:

await eio.send({'foo': 'bar'})

الوسيط الوحيد المقدم للدالة هو البيانات التي يتم تمريرها إلى الخادم. يمكن أن تكون البيانات من نوع str، bytes، dict أو list. البيانات المدرجة داخل القواميس والقوائم مقيدة أيضًا بهذه الأنواع.

يمكن استدعاء الدالة send() داخل معالج حدث كاستجابة لحدث من الخادم، أو في أي جزء آخر من التطبيق، بما في ذلك في المهام الخلفية.

قطع الاتصال بالخادم

في أي وقت، يمكن للعميل طلب قطع الاتصال بالخادم عن طريق استدعاء الدالة disconnect():

eio.disconnect()

بالنسبة لعميل asyncio، هذه coroutine:

await eio.disconnect()

إدارة المهام الخلفية

عند إنشاء اتصال عميل بالخادم، يتم إنشاء عدد قليل من المهام الخلفية للحفاظ على الاتصال نشطًا ومعالجة الأحداث الواردة. التطبيق الذي يعمل في الثريد الرئيسي حر في القيام بأي عمل، حيث أن هذا لن يمنع عمل عميل Engine.IO.

إذا لم يكن لدى التطبيق أي شيء للقيام به في الثريد الرئيسي ويريد فقط الانتظار حتى ينتهي الاتصال، يمكنه استدعاء الدالة wait():

eio.wait()

أو في إصدار asyncio:

await eio.wait()

لتسهيل الأمر على التطبيق، يتم توفير دالة مساعدة لبدء مهمة خلفية مخصصة:

def my_background_task(my_argument)
    # do some background work here!
    pass
eio.start_background_task(my_background_task, 123)

الوسائط الممررة إلى هذه الدالة هي دالة الخلفية وأي وسائط موضعية أو مسماة لاستدعاء الدالة بها.

هنا إصدار asyncio:

async def my_background_task(my_argument)
    # do some background work here!
    pass
eio.start_background_task(my_background_task, 123)

لاحظ أن هذه الدالة ليست coroutine، نظرًا لأنها لا تنتظر انتهاء دالة الخلفية، ولكن دالة الخلفية تفعل ذلك.

الدالة sleep() هي دالة مساعدة ثانية يتم توفيرها لصالح التطبيقات التي تعمل مع مهامها الخلفية الخاصة:

eio.sleep(2)

أو لـ asyncio:

await eio.sleep(2)

الوسيط الوحيد المقدم للدالة هو عدد الثواني للنوم.

التصحيح واستكشاف الأخطاء وإصلاحها

لمساعدتك في تصحيح الأخطاء، يمكن تكوين العميل لإخراج السجلات إلى الطرفية:

import engineio
# standard Python
eio = engineio.Client(logger=True)
# asyncio
eio = engineio.AsyncClient(logger=True)

يمكن تعيين الوسيط logger إلى True لإخراج السجلات إلى stderr، أو إلى كائن متوافق مع حزمة logging في Python حيث سيتم إصدار السجلات. قيمة False تعطّل التسجيل.

يمكن أن يساعد التسجيل في تحديد سبب مشاكل الاتصال، والانقطاعات غير المتوقعة، والمشاكل الأخرى.

خادم Engine.IO

تحتوي هذه الحزمة على خادمين لـ Engine.IO:

  • الفئة engineio.Server() تنشئ خادمًا متوافقًا مع مكتبة Python القياسية.
  • الفئة engineio.AsyncServer() تنشئ خادمًا متوافقًا مع حزمة asyncio.

طرق الخادمين متطابقة، مع الاختلاف الوحيد هو أنه في خادم asyncio، معظم الطرق يتم تنفيذها كـ coroutines.

التثبيت

لتثبيت خادم Python Engine.IO استخدم الأمر التالي:

pip install "python-engineio"

بالإضافة إلى الخادم، ستحتاج إلى اختيار إطار عمل غير متزامن أو خادم لاستخدامه معه. قائمة الحزم المدعومة مغطاة في قسم استراتيجيات النشر.

إنشاء مثيل للخادم

خادم Engine.IO هو مثيل للفئة engineio.Server. يمكن تحويل هذا المثيل إلى تطبيق WSGI قياسي عن طريق لفه بالفئة engineio.WSGIApp:

import engineio
# create a Engine.IO server
eio = engineio.Server()
# wrap with a WSGI application
app = engineio.WSGIApp(eio)

بالنسبة للخوادم القائمة على asyncio، توفر الفئة engineio.AsyncServer نفس الوظيفة، ولكن بتنسيق مناسب للـ coroutines. إذا رغبت في ذلك، يمكن للفئة engineio.ASGIApp تحويل الخادم إلى تطبيق ASGI قياسي:

# create a Engine.IO server
eio = engineio.AsyncServer()
# wrap with ASGI application
app = engineio.ASGIApp(eio)

يمكن لهذين الغلافين أيضًا العمل كوسيطين، لتمرير أي حركة مرور غير مخصصة لخادم Engine.IO إلى تطبيق آخر. هذا يسمح لخوادم Engine.IO بالتكامل بسهولة في تطبيقات WSGI أو ASGI الحالية:

from wsgi import app  # a Flask, Django, etc. application
app = engineio.WSGIApp(eio, app)

تقديم الملفات الثابتة

يمكن تكوين خادم Engine.IO لتقديم ملفات ثابتة للعملاء. هذا مفيد بشكل خاص لتسليم ملفات HTML و CSS و JavaScript إلى العملاء عندما يتم استخدام هذه الحزمة بدون إطار عمل مصاحب.

يتم تكوين الملفات الثابتة باستخدام قاموس Python حيث يكون كل زوج مفتاح/قيمة عبارة عن قاعدة تعيين ملف ثابت. في أبسط أشكاله، يحتوي هذا القاموس على عنوان URL واحد أو أكثر للملفات الثابتة كمفاتيح، والملفات المقابلة في الخادم كقيم:

static_files = {
    '/': 'latency.html',
    '/static/engine.io.js': 'static/engine.io.js',
    '/static/style.css': 'static/style.css',
}

مع هذا التكوين النموذجي، عندما يتلقى الخادم طلبًا لـ / (عنوان URL الجذر) فسوف يعيد محتويات الملف latency.html في الدليل الحالي، وسيعين نوع محتوى بناءً على امتداد الملف، في هذه الحالة text/html.

الملفات ذات امتدادات الملفات .html، .css، .js، .json، .jpg، .png، .gif و .txt يتم التعرف عليها تلقائيًا وتعيين نوع المحتوى الصحيح لها. بالنسبة للملفات ذات امتدادات الملفات الأخرى أو بدون امتداد ملف، يتم استخدام نوع المحتوى application/octet-stream كافتراضي.

إذا رغبت في ذلك، يمكن إعطاء نوع محتوى صريح لملف ثابت على النحو التالي:

static_files = {
    '/': {'filename': 'latency.html', 'content_type': 'text/plain'},
}

من الممكن أيضًا تكوين دليل كامل في قاعدة واحدة، بحيث يتم تقديم جميع الملفات فيه كملفات ثابتة:

static_files = {
    '/static': './public',
}

في هذا المثال، سيتم تقديم أي ملفات ذات عناوين URL تبدأ بـ /static مباشرة من مجلد public في الدليل الحالي، لذلك على سبيل المثال، عنوان URL /static/index.html سيعيد الملف المحلي ./public/index.html وعنوان URL /static/css/styles.css سيعيد الملف المحلي ./public/css/styles.css.

إذا تم طلب عنوان URL ينتهي بـ /، فسيتم إلحاق اسم ملف افتراضي هو index.html به. في المثال السابق، طلب عنوان URL /static/ سيعيد الملف المحلي ./public/index.html. يمكن تعيين اسم الملف الافتراضي لتقديمه للعناوين URL المنتهية بشرطة مائلة في قاموس الملفات الثابتة باستخدام مفتاح فارغ:

static_files = {
    '/static': './public',
    '': 'image.gif',
}

مع هذا التكوين، طلب عنوان URL /static/ سيعيد الملف المحلي ./public/image.gif. يمكن أيضًا تحديد نوع محتوى غير قياسي إذا لزم الأمر:

static_files = {
    '/static': './public',
    '': {'filename': 'image.gif', 'content_type': 'text/plain'},
}

ال 'static_files' يتم تمرير قاموس تكوين الملفات الثابتة كوسيط للفئات engineio.WSGIApp أو engineio.ASGIApp: 

# للتطبيقات القياسية WSGI
eio = engineio.Server()
app = engineio.WSGIApp(eio, static_files=static_files)
# لتطبيقات ASGI المستندة إلى asyncio
eio = engineio.AsyncServer()
app = engineio.ASGIApp(eio, static_files=static_files)

أسبقية التوجيه في هاتين الفئتين هي كما يلي:

  • أولاً، يتم فحص المسار مقابل مسار Engine.IO.
  • بعد ذلك، يتم فحص المسار مقابل تكوين الملفات الثابتة، إذا كان موجودًا.
  • إذا لم يتطابق المسار مع مسار Engine.IO أو أي ملف ثابت، يتم تمرير التحكم إلى التطبيق الثانوي إذا تم تكوينه، وإلا يتم إرجاع خطأ 404.

ملاحظة: خدمة الملفات الثابتة مخصصة للاستخدام أثناء التطوير فقط، ولذلك تفتقر إلى ميزات مهمة مثل التخزين المؤقت. لا تستخدم في بيئة الإنتاج.

تحديد معالجات الأحداث

للرد على الأحداث التي يتم تشغيلها بواسطة الاتصال أو العميل، يجب تعريف دوال معالج الأحداث باستخدام المُزخرف on:

@eio.on('connect')
    def on_connect(sid):
        print('تم توصيل عميل!')
    @eio.on('message')
    def on_message(sid, data):
        print('لقد تلقيت رسالة!')
    @eio.on('disconnect')
    def on_disconnect(sid, reason):
        print('تم فصل العميل! السبب:', reason)

بالنسبة لخادم asyncio، يمكن أن تكون معالجات الأحداث دوال عادية كما هو موضح أعلاه، أو يمكن أن تكون أيضًا coroutines:

@eio.on('message')
    async def on_message(sid, data):
        print('لقد تلقيت رسالة!')

الوسيط الذي تم تمريره إلى المُزخرف on هو اسم الحدث. الأحداث المدعومة هي connect و message و disconnect.

الوسيط sid الذي تم تمريره إلى جميع معالجات الأحداث هو معرف اتصال للعميل. ستستخدم جميع الأحداث من عميل واحد نفس قيمة sid.

يعتبر معالج connect هو المكان الذي يمكن للخادم فيه إجراء المصادقة. تستخدم القيمة التي يتم إرجاعها بواسطة هذا المعالج لتحديد ما إذا كان الاتصال مقبولًا أم مرفوضًا. عندما لا يُرجع المعالج أي قيمة (وهو نفس الشيء مثل إرجاع None) أو عندما يُرجع True، يتم قبول الاتصال. إذا قام المعالج بإرجاع False أو أي نوع بيانات متوافق مع JSON (سلسلة نصية، عدد صحيح، قائمة، أو قاموس) يتم رفض الاتصال. يؤدي الاتصال المرفوض إلى استجابة برمز الحالة 401.

يحتوي الوسيط data الذي تم تمريره إلى معالج الحدث 'message' على بيانات خاصة بالتطبيق قدمها العميل مع الحدث.

يتم استدعاء معالج disconnect لفصل العميل الذي بدأه العميل، أو فصل العميل الذي بدأه الخادم، أو انقطاع الاتصال العرضي، على سبيل المثال بسبب فشل الشبكة. الوسيط الثاني الذي تم تمريره إلى هذا المعالج يوفر سبب الانقطاع. مثال:

@eio.on('disconnect')
    def on_disconnect(sid, reason):
        if reason == eio.reason.CLIENT_DISCONNECT:
            print('العميل اختفى')
        elif reason == eio.reason.SERVER_DISCONNECT:
            print('تم طرد العميل')
        else:
            print(f'سبب الانقطاع: {reason}')

إرسال الرسائل

يمكن للخادم إرسال رسالة إلى أي عميل باستخدام الطريقة send():

eio.send(sid, {'foo': 'bar'})

أو في حالة asyncio، كـ coroutine:

await eio.send(sid, {'foo': 'bar'})

الوسيط الأول الذي تم تمريره إلى الطريقة هو معرف الاتصال للعميل المستلم. الوسيط الثاني هو البيانات التي يتم تمريرها إلى الخادم. يمكن أن تكون البيانات من نوع str أو bytes أو dict أو list. البيانات المدرجة داخل القواميس والقوائم مقيدة أيضًا بهذه الأنواع.

يمكن استدعاء الطريقة send() داخل معالج حدث كرد على حدث عميل، أو في أي جزء آخر من التطبيق، بما في ذلك في المهام الخلفية.

جلسات المستخدم

يمكن للخادم الاحتفاظ بمعلومات خاصة بالتطبيق في جلسة مستخدم مخصصة لكل عميل متصل. يمكن للتطبيقات استخدام جلسة المستخدم لكتابة أي تفاصيل حول المستخدم تحتاج إلى الاحتفاظ بها طوال مدة الاتصال، مثل أسماء المستخدمين أو معرفات المستخدمين.

يتم استخدام الطريقتين save_session() و get_session() لتخزين واسترداد المعلومات في جلسة المستخدم:

@eio.on('connect')
    def on_connect(sid, environ):
        username = authenticate_user(environ)
        eio.save_session(sid, {'username': username})
    @eio.on('message')
    def on_message(sid, data):
        session = eio.get_session(sid)
        print('رسالة من ', session['username'])

بالنسبة لخادم asyncio، هذه الطرق هي coroutines:

@eio.on('connect')
    async def on_connect(sid, environ):
        username = authenticate_user(environ)
        await eio.save_session(sid, {'username': username})
    @eio.on('message')
    async def on_message(sid, data):
        session = await eio.get_session(sid)
        print('رسالة من ', session['username'])

يمكن أيضًا التعامل مع الجلسة باستخدام مدير السياق session():

@eio.on('connect')
    def on_connect(sid, environ):
        username = authenticate_user(environ)
        with eio.session(sid) as session:
            session['username'] = username
    @eio.on('message')
    def on_message(sid, data):
        with eio.session(sid) as session:
            print('رسالة من ', session['username'])

بالنسبة لخادم asyncio، يتم استخدام مدير سياق غير متزامن:

@eio.on('connect')
    def on_connect(sid, environ):
        username = authenticate_user(environ)
        async with eio.session(sid) as session:
            session['username'] = username
    @eio.on('message')
    def on_message(sid, data):
        async with eio.session(sid) as session:
            print('رسالة من ', session['username'])

ملاحظة: يتم تدمير محتويات جلسة المستخدم عند فصل العميل.

فصل العميل

في أي وقت، يمكن للخادم فصل عميل عن الخادم عن طريق استدعاء الطريقة disconnect() وتمرير قيمة sid المعينة للعميل:

eio.disconnect(sid)

بالنسبة لعميل asyncio، هذه coroutine:

await eio.disconnect(sid)

إدارة المهام الخلفية

لتسهيل التطبيق، يتم توفير دالة مساعدة لبدء مهمة خلفية مخصصة:

def my_background_task(my_argument)
        # قم ببعض العمل في الخلفية هنا!
        pass
    eio.start_background_task(my_background_task, 123)

الوسائط التي تم تمريرها إلى هذه الطريقة هي الدالة الخلفية وأي وسائط موضعية أو مسماة لاستدعاء الدالة بها.

هذا هو إصدار asyncio:

async def my_background_task(my_argument)
        # قم ببعض العمل في الخلفية هنا!
        pass
    eio.start_background_task(my_background_task, 123)

لاحظ أن هذه الدالة ليست coroutine، لأنها لا تنتظر انتهاء الدالة الخلفية، لكن الدالة الخلفية كذلك.

الطريقة sleep() هي دالة مساعدة ثانية يتم توفيرها لصالح التطبيقات التي تعمل مع مهام خلفية خاصة بها:

eio.sleep(2)

أو لـ asyncio:

await eio.sleep(2)

الوسيط الوحيد الذي تم تمريره إلى الطريقة هو عدد الثواني للسبات.

import engineio

استراتيجيات النشر

تصف الأقسام التالية مجموعة متنوعة من استراتيجيات النشر لخوادم Engine.IO.

Uvicorn و Daphne وخوادم ASGI الأخرى

الفئة engineio.ASGIApp هي تطبيق متوافق مع ASGI يمكنه توجيه حركة مرور Engine.IO إلى مثيل engineio.AsyncServer:

eio = engineio.AsyncServer(async_mode='asgi')
    app = engineio.ASGIApp(eio)

إذا رغبت في ذلك، يمكن للفئة engineio.ASGIApp توجيه أي حركة مرور ليست Engine.IO إلى تطبيق ASGI آخر، مما يجعل من الممكن نشر تطبيق ويب ASGI قياسي وخادم Engine.IO كحزمة واحدة:

eio = engineio.AsyncServer(async_mode='asgi')
    app = engineio.ASGIApp(eio, other_app)

المثيل ASGIApp هو مثيل متوافق تمامًا مع ASGI يمكن نشره مع خادم ويب متوافق مع ASGI.

Aiohttp

ستستخدم مثيلات الفئة engineio.AsyncServer تلقائيًا aiohttp للعمليات غير المتزامنة إذا كانت المكتبة مثبتة. لطلب استخدامها بشكل صريح، يمكن إعطاء خيار async_mode في المنشئ:

يوضح المثال التالي كيفية بدء تطبيق Engine.IO باستخدام فئة العامل المعالج Gunicorn:

$ gunicorn -w 1 --threads 100 module:app

مع التكوين أعلاه، سيتمكن الخادم من التعامل مع ما يصل إلى 100 عميل متزامن.

ملاحظات قابلية التوسع

بروتوكول Engine.IO هو بروتوكول متصل بالحالة، مما يجعل التوسع الأفقي أكثر صعوبة. لنشر مجموعة من عمليات Engine.IO المستضافة على خادم واحد أو أكثر، يجب تلبية الشروط التالية:

  • يجب أن تكون كل عملية من عمليات خادم Engine.IO قادرة على التعامل مع طلبات متعددة بشكل متزامن. هذا مطلوب لأن عملاء الاستقصاء الطويل يرسلون طلبين بالتوازي. عمليات العمال التي لا يمكنها التعامل إلا مع طلب واحد في المرة الواحدة غير مدعومة.
  • يجب تكوين موازن التحميل لإعادة توجيه الطلبات من عميل إلى نفس العملية دائمًا. يطلق موازنو التحميل على هذا "الجلسات الثابتة" أو "تناسق الجلسة".

ضوابط الأصل المشترك

لأسباب أمنية، يفرض هذا الخادم سياسة الأصل نفسه افتراضيًا. عمليًا، هذا يعني ما يلي:

  • إذا كان طلب HTTP أو WebSocket الوارد يتضمن رأس Origin ، فيجب أن يتطابق هذا الرأس مع مخطط واستضافة عنوان URL للاتصال. في حالة عدم التطابق، يتم إرجاع استجابة رمز الحالة 400 ويتم رفض الاتصال.
  • لا توجد قيود مفروضة على الطلبات الواردة التي لا تتضمن رأس Origin.

إذا لزم الأمر، يمكن استخدام خيار cors_allowed_origins للسماح بأصول أخرى. يمكن تعيين هذه الوسيطة إلى سلسلة لتعيين أصل واحد مسموح به، أو إلى قائمة للسماح بأصول متعددة. يمكن استخدام قيمة خاصة '*' لإبلاغ الخادم بالسماح بجميع الأصول، ولكن يجب القيام بذلك بحذر، حيث يمكن أن يجعل هذا الخادم عرضة لهجمات تزوير الطلبات عبر المواقع (CSRF).

  • cookie -- إذا تم تعيينها كسلسلة نصية، فهي اسم ملف تعريف الارتباط HTTP الذي يرسله الخادم مرة أخرى إلى العميل والذي يحتوي على معرف جلسة العميل. إذا تم تعيينها كقاموس، يحتوي المفتاح 'name' على اسم ملف تعريف الارتباط والمفاتيح الأخرى تحدد سمات ملف تعريف الارتباط، حيث يمكن أن تكون قيمة كل سمة سلسلة نصية، أو دالة قابلة للاستدعاء بدون وسائط، أو قيمة منطقية. إذا تم تعيينها إلى None (الافتراضي)، فلا يتم إرسال ملف تعريف الارتباط إلى العميل.
  • cors_allowed_origins -- أصل أو قائمة الأصول المسموح لها بالاتصال بهذا الخادم. يُسمح فقط بنفس الأصل افتراضيًا. قم بتعيين هذا الوسيط إلى '*' أو ['*'] للسماح بجميع الأصول، أو إلى [] لتعطيل معالجة CORS.
  • cors_credentials -- ما إذا كان مسموحًا ببيانات الاعتماد (ملفات التعريف الارتباط، المصادقة) في الطلبات إلى هذا الخادم.
  • logger -- لتمكين التسجيل اضبط على True أو مرر كائن logger للاستخدام. لتعطيل التسجيل اضبط على False. لاحظ أن الأخطاء الفادحة يتم تسجيلها حتى عندما يكون logger هو False.
  • json -- وحدة json بديلة للاستخدام في ترميز وفك ترميز الحزم. يجب أن تحتوي وحدات json المخصصة على وظائف dumps و loads متوافقة مع إصدارات المكتبة القياسية.
  • async_handlers -- إذا تم تعيينها إلى True، فقم بتشغيل معالجات أحداث الرسائل في خيوط غير حاجزة. لتشغيل المعالجات بشكل متزامن، اضبطها على False. الافتراضي هو True.
  • monitor_clients -- إذا تم تعيينها إلى True، فستضمن مهمة في الخلفية إغلاق العملاء غير النشطين. اضبط على False لتعطيل مهمة المراقبة (غير موصى به). الافتراضي هو True.
  • transports -- قائمة وسائل النقل المسموح بها. وسائل النقل الصالحة هي 'polling' و 'websocket'. الافتراضي هو ['polling', 'websocket'].
  • kwargs -- محجوز للتوسعات المستقبلية، سيتم تجاهل أي وسائط إضافية مقدمة كوسائط كلمة رئيسية بصمت.

هذا مدير سياق يعيد قاموس جلسة المستخدم للعميل. أي تغييرات يتم إجراؤها على هذا القاموس داخل كتلة مدير السياق يتم حفظها مرة أخرى إلى الجلسة. مثال للاستخدام:

@eio.on('connect')
def on_connect(sid, environ):
    username = authenticate_user(environ)
    if not username:
        return False
    with eio.session(sid) as session:
        session['username'] = username
@eio.on('message')
def on_message(sid, msg):
    async with eio.session(sid) as session:
        print('received message from ', session['username'])
