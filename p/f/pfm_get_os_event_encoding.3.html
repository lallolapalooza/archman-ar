pfm_get_os_event_encoding(3)

LIBPFM(3) دليل مبرمج Linux LIBPFM(3)

الاسم

pfm_get_os_event_encoding - الحصول على ترميز الحدث لنظام تشغيل معين

الملخص

#include <perfmon/pfmlib.h>

int pfm_get_os_event_encoding(const char *str, int dfl_plm, pfm_os_t os,  void *arg);

الوصف

هذه هي الوظيفة الرئيسية لاسترداد ترميز حدث لواجهة نظام تشغيل معينة. يتم تحليل سلسلة الأحداث المقدمة في str وترميزها لنظام التشغيل المحدد بواسطة os. لا يمكن ترميز سوى حدث واحد لكل مكالمة. على هذا النحو، يمكن أن يحتوي str على اسم حدث رمزي واحد فقط. يتم ترميز الحدث للمراقبة في مستويات الامتياز المحددة بواسطة قناع dfl_plm، إذا كان مدعومًا، وإلا يتم تجاهل هذه المعلمة. يتم تمرير وسائط الإدخال والإخراج الخاصة بنظام التشغيل في arg.

يمكن أن تحتوي سلسلة الأحداث، str، على أقنعة فرعية للأحداث (umask) وأي معدلات أخرى مدعومة. يتم تحليل حدث واحد فقط من السلسلة. للراحة، من الممكن تمرير قائمة مفصولة بفواصل من الأحداث في str ولكن يتم ترميز الحدث الأول فقط.

القيم التالية مدعومة لـ os:

PFM_OS_NONE
تتسبب هذه القيمة في ترميز الحدث بشكل بحت كما هو محدد بواسطة عتاد PMU. يجب أن يكون الوسيط arg مؤشرًا إلى بنية pfm_pmu_encode_arg_t والتي يتم تعريفها على النحو التالي: 
typedef struct {
    uint64_t    *codes;
    char        **fstr;
    size_t      size;
    int         count;
    int         idx;
} pfm_pmu_encode_arg_t;

الحقول معرفة على النحو التالي:

codes
مؤشر إلى مصفوفة من القيم 64 بت. عند الإدخال، إذا كان codes فارغًا (NULL)، فإن المكتبة تقوم بتخصيص ما يلزم لتخزين ترميز الحدث. إذا لم يكن codes فارغًا عند الإدخال، فيجب أن يعكس count العدد الفعلي للعناصر. إذا كان count كبيرًا بما فيه الكفاية، تقوم المكتبة بتخزين الترميز في العنوان المقدم.
count
عند الإدخال، يحتوي الحقل على الحد الأقصى لعدد العناصر في المصفوفة codes. عند العودة، يحتوي على عدد الإدخالات الفعلية في codes. إذا كان codes فارغًا، فيجب أن يكون count صفرًا.
fstr
إذا كان المتصل مهتمًا باسترداد سلسلة الأحداث المؤهلة بالكامل حيث يتم تحديد جميع أقنعة الوحدات المستخدمة وجميع المعدلات، فيجب تعيين هذا الحقل إلى عنوان غير فارغ لمؤشر إلى سلسلة (char **). عند العودة، إذا لم يكن fstr فارغًا، فإن مؤشر السلسلة الذي تم تمريره عند الدخول يشير إلى سلسلة الأحداث. يتم تخصيص السلسلة ديناميكيًا ويجب تحريرها في النهاية بواسطة المتصل. إذا كان fstr فارغًا عند الدخول، فلا يتم إرجاع شيء في هذا الحقل. يظهر تسلسل الاستدعاء النموذجي على النحو التالي: 
   char *fstr = NULL
   pfm_pmu_encode_arg_t arg;
   arg.fstr = &fstr;
   ret = pfm_get_os_event_encoding("event",
                                   PFM_PLM0|PFM_PLM3,
                                   PFM_OS_NONE,
                                   &e);
   if (ret == PFM_SUCCESS) {
      printf("fstr=%s\n", fstr);
      free(fstr);
   }
size
يحتوي هذا الحقل على حجم البنية التي تم تمريرها. يستخدم هذا الحقل لتوفير قابلية التوسع للبنية دون المساس بالتوافق مع الإصدارات السابقة. يجب تعيين القيمة إلى sizeof(pfm_pmu_encode_arg_t). إذا تم تحديد قيمة 0 بدلاً من ذلك، تفترض المكتبة أن البنية التي تم تمريرها مطابقة للإصدار الأول من ABI الذي حجمه PFM_RAW_ENCODE_ABI0. وبالتالي، إذا تمت إضافة حقول بعد الإصدار الأول من ABI، فلن تقوم المكتبة بتعيينها. تتحقق المكتبة من أن البايتات التي تتجاوز ما تم تنفيذه هي أصفار.
idx
عند العودة، يحتوي هذا الحقل على المعرف الفريد غير الشفاف للحدث الموصوف في str. يمكن استخدام هذا الفهرس لاسترداد معلومات حول الحدث باستخدام pfm_get_event_info()، على سبيل المثال.
PFM_OS_PERF_EVENT, PFM_OS_PERF_EVENT_EXT
تتسبب هذه القيمة في ترميز الحدث لواجهة perf_event لنواة Linux (متوفرة منذ 2.6.31). يجب أن يكون arg مؤشرًا إلى بنية pfm_perf_encode_arg_t. توفر طبقة PFM_OS_PERF_EVENT المعدلات التي تصدرها عتاد PMU الأساسي، والتي قد يتجاوز بعضها فعليًا واجهة perf_event، مثل مستويات امتياز المراقبة. يوسع PFM_OS_PERF_EVENT_EXT PFM_OS_PERF_EVENT لإضافة معدلات يتم التحكم فيها فقط بواسطة واجهة perf_event، مثل فترة العينات (period)، والتردد (freq)، والوصول الحصري للموارد (excl). 
typedef struct {
    struct perf_event_attr *attr;
    char **fstr;
    size_t size;
    int idx;
    int cpu;
    int flags;
} pfm_perf_encode_arg_t;
الحقول معرفة على النحو التالي:
attr
مؤشر إلى بنية perf_event_attr كما هو محدد في perf_event.h. لا يمكن أن يكون هذا الحقل فارغًا (NULL) عند الدخول. لا يتم الكتابة فوق البنية بالكامل بواسطة المكالمة. تقوم المكتبة بتعديل الحقول التي تعرفها فقط، مما يسمح بعدم توافق ABI لـ perf_event بين المتصل والمكتبة.
fstr
نفس السلوك الموضح لـ PFM_OS_NONE أعلاه.
size
يحتوي هذا الحقل على حجم البنية التي تم تمريرها. يستخدم هذا الحقل لتوفير قابلية التوسع للبنية دون المساس بالتوافق مع الإصدارات السابقة. يجب تعيين القيمة إلى sizeof(pfm_perf_encode_arg_t). إذا تم تحديد قيمة 0 بدلاً من ذلك، تفترض المكتبة أن البنية التي تم تمريرها مطابقة للإصدار الأول من ABI الذي حجمه PFM_PERF_ENCODE_ABI0. وبالتالي، إذا تمت إضافة حقول بعد الإصدار الأول من ABI، فلن تقوم المكتبة بتعيينها. تتحقق المكتبة من أن البايتات التي تتجاوز ما تم تنفيذه هي أصفار.
idx
عند العودة، يحتوي هذا الحقل على المعرف الفريد غير الشفاف للحدث الموصوف في str. يمكن استخدام هذا الفهرس لاسترداد معلومات حول الحدث باستخدام pfm_get_event_info()، على سبيل المثال.
cpu
غير مستخدم بعد.
flags
غير مستخدم بعد.

إليك مثال على كيفية استخدام هذه الوظيفة مع PFM_OS_NONE:

#include <inttypes.h>
#include <err.h>
#include <perfmon/pfmlib.h>
int main(int argc, char **argv)
{
   pfm_pmu_encode_arg_t arg;
   int i, ret;
   ret = pfm_initialize();
   if (ret != PFM_SUCCESS)
      errx(1, "cannot initialize library %s", pfm_strerror(ret));
   memset(&arg, 0, sizeof(arg));
   ret = pfm_get_os_event_encoding("RETIRED_INSTRUCTIONS", PFM_PLM3, PFM_OS_NONE, &arg);
   if (ret != PFM_SUCCESS)
      err(1, "cannot get encoding %s", pfm_strerror(ret));
   for(i = 0; i < arg.count; i++)
      printf("count[%d]=0x%"PRIx64"\n", i, arg.codes[i]);
   free(arg.codes);
   return 0;
}

الإرجاع

تقوم الوظيفة بإرجاع ترميز الحدث لنظام التشغيل الذي تم تمريره في os في arg. يعتمد محتوى arg على الوسيط os. عند النجاح، يتم إرجاع PFM_SUCCESS وإلا يتم إرجاع رمز خطأ محدد.

الأخطاء

PFM_ERR_TOOSMALL
الوسيط code صغير جدًا للترميز.
PFM_ERR_INVAL
الوسيط code أو count هو NULL أو يحتوي str على أكثر من حدث رمزي واحد.
PFM_ERR_NOMEM
لا توجد ذاكرة كافية.
PFM_ERR_NOTFOUND
الحدث غير موجود.
