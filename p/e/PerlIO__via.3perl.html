الحزمة: perl

التراخيص: Artistic-1.0-Perl OR GPL-1.0-or-later

PerlIO::via(3perl) دليل مبرمجي بيرل PerlIO::via(3perl)

الاسم

PerlIO::via - فئة مساعدة لطبقات PerlIO المنفذة في بيرل

الملخص

use PerlIO::via::Layer;
open($fh,"<:via(Layer)",...);
use Some::Other::Package;
open($fh,">:via(Some::Other::Package)",...);

الوصف

تسمح الوحدة PerlIO::via لك بتطوير طبقات PerlIO في بيرل، دون الحاجة إلى الدخول في تفاصيل البرمجة بلغة C باستخدام XS كواجهة لـ Perl.

تم تضمين وحدة مثال واحدة، PerlIO::via::QuotedPrint، مع Perl 5.8.0، وتتوفر وحدات أمثلة أخرى من CPAN، مثل PerlIO::via::StripHTML و PerlIO::via::Base64. على سبيل المثال، تسمح وحدة PerlIO::via::StripHTML لك بما يلي:

use PerlIO::via::StripHTML;
open( my $fh, "<:via(StripHTML)", "index.html" );
my @line = <$fh>;

للحصول على نص ملف HTML في مصفوفة مع إزالة جميع علامات HTML تلقائيًا.

يرجى ملاحظة أنه إذا تم إنشاء الطبقة في مساحة اسم PerlIO::via::، فلا يلزم أن تكون مؤهلة بالكامل. ستقوم الوحدة PerlIO::via بوضع بادئة مساحة الاسم PerlIO::via:: إذا لم يكن اسم الوحدة المحدد موجودًا كاسم وحدة مؤهل بالكامل.

الطرق المتوقعة

لإنشاء وحدة Perl تنفذ طبقة PerlIO في Perl (على عكس C باستخدام XS كواجهة لـ Perl)، تحتاج إلى توفير بعض الإجراءات الفرعية التالية. يوصى بإنشاء وحدات Perl هذه في مساحة الاسم PerlIO::via::، بحيث يمكن العثور عليها بسهولة على CPAN واستخدام ميزة مساحة الاسم الافتراضية للوحدة PerlIO::via نفسها.

يرجى ملاحظة أن هذا مجال تطوير حديث في Perl وأن الواجهة الموضحة هنا تخضع للتغيير (ويؤمل أن يكون لديها وثائق أفضل والمزيد من الأمثلة).

في وصف الطرق أدناه، $fh سيكون مرجعًا إلى علامة (glob) يمكن التعامل معها كمقبض ملف perl. تشير إلى الطبقة أدناه. $fh لا يتم تمريره إذا كانت الطبقة في أسفل المكدس، ولهذا السبب وللحفاظ على مستوى معين من "التوافق" مع فئات TIEHANDLE، يتم تمريره أخيرًا.

$class->PUSHED([$mode,[$fh]])
يجب أن تعيد كائنًا أو الفئة، أو -1 في حالة الفشل. (قارن TIEHANDLE.) الوسيطات هي سلسلة وضع اختيارية ("r"، "w"، "w+"، ...) ومقبض ملف لطبقة PerlIO أدناه. إلزامية.

عندما يتم دفع الطبقة كجزء من استدعاء "open"، سيتم استدعاء "PUSHED" *قبل* حدوث الفتح الفعلي، سواء كان ذلك عبر "OPEN"، "SYSOPEN"، "FDOPEN" أو عن طريق السماح لطبقة أدنى بإجراء الفتح.

$obj->POPPED([$fh])
اختياري - يتم استدعاؤه عندما تكون الطبقة على وشك الإزالة.
$obj->UTF8($belowFlag,[$fh])
اختياري - إذا كان موجودًا، فسيتم استدعاؤه فورًا بعد عودة PUSHED. يجب أن يعيد قيمة صحيحة إذا كانت الطبقة تتوقع ترميز البيانات بـ UTF-8. إذا أعاد قيمة صحيحة، فإن النتيجة كما لو كان المستدعي قد قام بما يلي: 
":via(YourClass):utf8"

إذا لم يكن موجودًا أو إذا أعاد قيمة خاطئة، فسيتم ترك الدفق مع تعيين علامة UTF-8. ستكون الوسيطة $belowFlag صحيحة إذا كانت هناك طبقة أدنى وكانت تلك الطبقة تتوقع UTF-8.

$obj->OPEN($path,$mode,[$fh])
اختياري - إذا لم يكن موجودًا، تقوم طبقة أدنى بإجراء الفتح. إذا كان موجودًا، يتم استدعاؤه لعمليات الفتح العادية بعد دفع الطبقة. تخضع هذه الوظيفة للتغيير لأنه لا توجد طريقة سهلة لجعل طبقة أدنى تقوم بالفتح ثم استعادة التحكم.
$obj->BINMODE([$fh])
اختياري - إذا لم يكن موجودًا، يتم إزالة الطبقة عند binmode($fh) أو عند دفع ":raw". إذا كان موجودًا، يجب أن يعيد 0 عند النجاح، -1 عند الخطأ، أو undef لإزالة الطبقة.
$obj->FDOPEN($fd,[$fh])
اختياري - إذا لم يكن موجودًا، تقوم طبقة أدنى بإجراء الفتح. إذا كان موجودًا، يتم استدعاؤه بعد دفع الطبقة لعمليات الفتح التي تمرر واصف ملف رقمي. تخضع هذه الوظيفة للتغيير لأنه لا توجد طريقة سهلة لجعل طبقة أدنى تقوم بالفتح ثم استعادة التحكم.
$obj->SYSOPEN($path,$imode,$perm,[$fh])
اختياري - إذا لم يكن موجودًا، تقوم طبقة أدنى بإجراء الفتح. إذا كان موجودًا، يتم استدعاؤه بعد دفع الطبقة لعمليات الفتح بأسلوب sysopen التي تمرر وضعًا رقميًا وأذونات. تخضع هذه الوظيفة للتغيير لأنه لا توجد طريقة سهلة لجعل طبقة أدنى تقوم بالفتح ثم استعادة التحكم.
$obj->FILENO($fh)
يعيد قيمة رقمية لواصف ملف شبيه بـ Unix. يعيد -1 إذا لم يكن هناك واحد. اختياري. الافتراضي هو fileno($fh).
$obj->READ($buffer,$len,$fh)
يعيد عدد البايتات التي تم وضعها في $buffer (يجب أن تكون undef للإشارة إلى خطأ أو عدد صحيح غير سالب أقل من أو يساوي الحد الأدنى لـ $len وطول $buffer المحدث). اختياري. الافتراضي هو استخدام FILL بدلاً من ذلك.
$obj->WRITE($buffer,$fh)
يعيد عدد البايتات من $buffer التي تم كتابتها بنجاح.
$obj->FILL($fh)
يجب أن يعيد سلسلة لوضعها في المخزن المؤقت. اختياري. إذا لم يتم توفيره، يجب توفير READ أو رفض المقابض المفتوحة للقراءة في PUSHED.
$obj->CLOSE($fh)
يجب أن يعيد 0 عند النجاح، -1 عند الخطأ. اختياري.
$obj->SEEK($posn,$whence,$fh)
يجب أن يعيد 0 عند النجاح، -1 عند الخطأ. اختياري. الافتراضي هو الفشل، ولكن من المحتمل أن يتغير ذلك في المستقبل.
$obj->TELL($fh)
يعيد موضع الملف. اختياري. الافتراضي سيتم تحديده.
$obj->UNREAD($buffer,$fh)
يعيد عدد البايتات من $buffer التي تم حفظها بنجاح لإعادتها في مكالمات FILL/READ المستقبلية. اختياري. الافتراضي هو دفع البيانات إلى طبقة مؤقتة فوق هذه الطبقة.
$obj->FLUSH($fh)
مسح أي بيانات كتابة مخزنة مؤقتًا. قد يتم استدعاؤها على المقابض القابلة للقراءة أيضًا. يجب أن تعيد 0 عند النجاح، -1 عند الخطأ.
$obj->SETLINEBUF($fh)
اختياري. لا يعيد شيئًا.
$obj->CLEARERR($fh)
اختياري. لا يعيد شيئًا.
$obj->ERROR($fh)
اختياري. يعيد حالة الخطأ. الافتراضي هو عدم وجود خطأ حتى يتم تحديد آلية للإشارة إلى الخطأ (die؟).
$obj->EOF($fh)
اختياري. يعيد حالة نهاية الملف. الافتراضي هو دالة لقيمة الإرجاع من FILL أو READ.

أمثلة

تحقق من مساحة الاسم PerlIO::via:: على CPAN للحصول على أمثلة لطبقات PerlIO المنفذة في Perl. لإعطائك فكرة عن مدى بساطة تنفيذ طبقة PerlIO، يتم تضمين مثال بسيط هنا.

مثال - مقبض سداسي عشري

بالنظر إلى الوحدة التالية، PerlIO::via::Hex:

package PerlIO::via::Hex;
sub PUSHED
{
 my ($class,$mode,$fh) = @_;
 # عند الكتابة، نقوم بتخزين البيانات مؤقتًا
 my $buf = '';
 return bless \$buf,$class;
}
sub FILL
{
 my ($obj,$fh) = @_;
 my $line = <$fh>;
 return (defined $line) ? pack("H*", $line) : undef;
}
sub WRITE
{
 my ($obj,$buf,$fh) = @_;
 $$obj .= unpack("H*", $buf);
 return length($buf);
}
sub FLUSH
{
 my ($obj,$fh) = @_;
 print $fh $$obj or return -1;
 $$obj = '';
 return 0;
}
1;

يفتح الكود التالي مقبض إخراج سيحول أي إخراج إلى تفريغ سداسي عشري لبايتات الإخراج: على سبيل المثال، سيتم تحويل "A" إلى "41" (على الأجهزة المستندة إلى ASCII، على منصات EBCDIC سيصبح "A" "c1")

use PerlIO::via::Hex;
open(my $fh, ">:via(Hex)", "foo.hex");

والرمز التالي سيقرأ التفريغ السداسي عشري ويحوله تلقائيًا مرة أخرى إلى بايتات:

open(my $fh, "<:via(Hex)", "foo.hex");
