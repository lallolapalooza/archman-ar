perf-dlfilter(1)

الحزمة: perf

التراخيص: GPL-2.0-only

PERF-DLFILTER(1) دليل perf PERF-DLFILTER(1)

الاسم

perf-dlfilter - تصفية أحداث العينات باستخدام ملف كائن مشترك يتم تحميله ديناميكيًا

الملخص

perf script [--dlfilter file.so ] [ --dlarg arg ]...

الوصف

يُستخدم هذا الخيار لمعالجة البيانات من خلال مرشح مخصص مقدم بواسطة ملف كائن مشترك يتم تحميله ديناميكيًا. يمكن تمرير الوسائط باستخدام --dlarg واسترجاعها باستخدام perf_dlfilter_fns.args().

إذا كان file.so لا يحتوي على "/"، فسيتم العثور عليه إما في الدليل الحالي، أو في مسار أدوات perf التنفيذي وهو ~/libexec/perf-core/dlfilters لبناء وتثبيت محلي (راجع perf --exec-path)، أو في مسارات الرابط الديناميكي.

واجهة برمجة التطبيقات

تتكون واجهة برمجة التطبيقات للتصفية من ما يلي:

#include <perf/perf_dlfilter.h>
struct perf_dlfilter_fns perf_dlfilter_fns;
int start(void **data, void *ctx);
int stop(void *data, void *ctx);
int filter_event(void *data, const struct perf_dlfilter_sample *sample, void *ctx);
int filter_event_early(void *data, const struct perf_dlfilter_sample *sample, void *ctx);
const char *filter_description(const char **long_description);

إذا تم تنفيذها، سيتم استدعاء start في البداية، قبل أي استدعاءات لـ filter_event أو filter_event_early. أرجع 0 للإشارة إلى النجاح، أو أرجع رمز خطأ سالبًا. يمكن تعيين *data للاستخدام من قبل وظائف أخرى. ctx مطلوب للاستدعاءات إلى perf_dlfilter_fns، ولكن معظم perf_dlfilter_fns غير صالحة عند استدعائها من start.

إذا تم تنفيذها، سيتم استدعاء stop في النهاية، بعد أي استدعاءات لـ filter_event أو filter_event_early. أرجع 0 للإشارة إلى النجاح، أو أرجع رمز خطأ سالبًا. يتم تعيين data بواسطة start. ctx مطلوب للاستدعاءات إلى perf_dlfilter_fns، ولكن معظم perf_dlfilter_fns غير صالحة عند استدعائها من stop.

إذا تم تنفيذها، سيتم استدعاء filter_event لكل حدث عينة. أرجع 0 للاحتفاظ بحدث العينة، 1 لتصفيته، أو أرجع رمز خطأ سالبًا. يتم تعيين data بواسطة start. ctx مطلوب للاستدعاءات إلى perf_dlfilter_fns.

filter_event_early هو نفس filter_event باستثناء أنه يتم استدعاؤه قبل التصفية الداخلية.

إذا تم تنفيذها، يجب أن ترجع filter_description وصفًا من سطر واحد للمرشح، ووصفًا أطول اختياريًا.

لا تفترض أن وسيط sample صالح (قابل للإشارة إليه) بعد عودة filter_event و filter_event_early.

لا تفترض أن البيانات المشار إليها بواسطة المؤشرات في بنية perf_dlfilter_sample صالحة (قابلة للإشارة إليها) بعد عودة filter_event و filter_event_early.

بنية perf_dlfilter_sample

يتم تمرير filter_event و filter_event_early بنية perf_dlfilter_sample، والتي تحتوي على الحقول التالية:

/*
 * معلومات حدث عينة perf (وفقًا لـ perf script و <linux/perf_event.h>)
 */
struct perf_dlfilter_sample {
        __u32 size; /* حجم هذا الهيكل (للتوافق) */
        __u16 ins_lat;          /* راجع PERF_SAMPLE_WEIGHT_TYPE في <linux/perf_event.h> */
        __u16 p_stage_cyc;      /* راجع PERF_SAMPLE_WEIGHT_TYPE في <linux/perf_event.h> */
        __u64 ip;
        __s32 pid;
        __s32 tid;
        __u64 time;
        __u64 addr;
        __u64 id;
        __u64 stream_id;
        __u64 period;
        __u64 weight;           /* راجع PERF_SAMPLE_WEIGHT_TYPE في <linux/perf_event.h> */
        __u64 transaction;      /* راجع PERF_SAMPLE_TRANSACTION في <linux/perf_event.h> */
        __u64 insn_cnt; /* لتعليمات لكل دورة (IPC) */
        __u64 cyc_cnt;          /* لتعليمات لكل دورة (IPC) */
        __s32 cpu;
        __u32 flags;            /* راجع PERF_DLFILTER_FLAG_* أعلاه */
        __u64 data_src;         /* راجع PERF_SAMPLE_DATA_SRC في <linux/perf_event.h> */
        __u64 phys_addr;        /* راجع PERF_SAMPLE_PHYS_ADDR في <linux/perf_event.h> */
        __u64 data_page_size;   /* راجع PERF_SAMPLE_DATA_PAGE_SIZE في <linux/perf_event.h> */
        __u64 code_page_size;   /* راجع PERF_SAMPLE_CODE_PAGE_SIZE في <linux/perf_event.h> */
        __u64 cgroup;           /* راجع PERF_SAMPLE_CGROUP في <linux/perf_event.h> */
        __u8  cpumode;          /* راجع CPUMODE_MASK وما إلى ذلك في <linux/perf_event.h> */
        __u8  addr_correlates_sym; /* صحيح => يمكن استدعاء resolve_addr() */
        __u16 misc;             /* راجع perf_event_header في <linux/perf_event.h> */
        __u32 raw_size;         /* راجع PERF_SAMPLE_RAW في <linux/perf_event.h> */
        const void *raw_data;   /* راجع PERF_SAMPLE_RAW في <linux/perf_event.h> */
        __u64 brstack_nr;       /* عدد إدخالات brstack */
        const struct perf_branch_entry *brstack; /* راجع <linux/perf_event.h> */
        __u64 raw_callchain_nr; /* عدد إدخالات raw_callchain */
        const __u64 *raw_callchain; /* راجع <linux/perf_event.h> */
        const char *event;
        __s32 machine_pid;
        __s32 vcpu;
};

ملاحظة: machine_pid و vcpu ليست أعضاء أصلية، ولكن تمت إضافتها معًا لاحقًا. يمكن استخدام size لتحديد وجودها في وقت التشغيل. سيتم تعريف PERF_DLFILTER_HAS_MACHINE_PID إذا كانت موجودة في وقت الترجمة. على سبيل المثال:

#include <perf/perf_dlfilter.h>
#include <stddef.h>
#include <stdbool.h>
static inline bool have_machine_pid(const struct perf_dlfilter_sample *sample)
{
#ifdef PERF_DLFILTER_HAS_MACHINE_PID
        return sample->size >= offsetof(struct perf_dlfilter_sample, vcpu) + sizeof(sample->vcpu);
#else
        return false;
#endif
}

بنية perf_dlfilter_fns

يتم ملء بنية perf_dlfilter_fns بمؤشرات وظائف عند تحميل الملف. يمكن استدعاء الوظائف بواسطة filter_event أو filter_event_early.

struct perf_dlfilter_fns {
        const struct perf_dlfilter_al *(*resolve_ip)(void *ctx);
        const struct perf_dlfilter_al *(*resolve_addr)(void *ctx);
        char **(*args)(void *ctx, int *dlargc);
        __s32 (*resolve_address)(void *ctx, __u64 address, struct perf_dlfilter_al *al);
        const __u8 *(*insn)(void *ctx, __u32 *length);
        const char *(*srcline)(void *ctx, __u32 *line_number);
        struct perf_event_attr *(*attr)(void *ctx);
        __s32 (*object_code)(void *ctx, __u64 ip, void *buf, __u32 len);
        void (*al_cleanup)(void *ctx, struct perf_dlfilter_al *al);
        void *(*reserved[119])(void *);
};

resolve_ip ترجع معلومات حول ip.

resolve_addr ترجع معلومات حول addr (إذا كان addr_correlates_sym).

args ترجع الوسائط من خيارات --dlarg.

resolve_address توفر معلومات حول address. يجب تعيين al→size قبل الاستدعاء. ترجع 0 عند النجاح، -1 خلاف ذلك. استدع al_cleanup() (إذا كانت موجودة، انظر أدناه) عندما لا تكون بيانات al مطلوبة بعد الآن.

insn ترجع بايتات التعليمات والطول.

srcline ترجع اسم ملف المصدر ورقم السطر.

attr ترجع perf_event_attr، راجع <linux/perf_event.h>.

object_code يقرأ رمز الكائن ويرجع عدد البايتات المقروءة.

يجب استدعاء al_cleanup (إذا كانت موجودة، لذا تحقق من perf_dlfilter_fns.al_cleanup != NULL) بعد resolve_address() لتحرير أي موارد مرتبطة.

لا تفترض أن المؤشرات التي تم الحصول عليها عبر perf_dlfilter_fns صالحة (قابلة للإشارة إليها) بعد عودة filter_event و filter_event_early.

بنية perf_dlfilter_al

تحتوي بنية perf_dlfilter_al على معلومات حول عنوان.

/*
 * موقع العنوان (وفقًا لـ perf script)
 */
struct perf_dlfilter_al {
        __u32 size; /* حجم هذا الهيكل (للتوافق) */
        __u32 symoff;
        const char *sym;
        __u64 addr; /* العنوان المعين (من dso) */
        __u64 sym_start;
        __u64 sym_end;
        const char *dso;
        __u8  sym_binding; /* STB_LOCAL, STB_GLOBAL أو STB_WEAK، راجع <elf.h> */
        __u8  is_64_bit; /* صالح فقط إذا كان dso ليس NULL */
        __u8  is_kernel_ip; /* صحيح إذا كان في مساحة النواة */
        __u32 buildid_size;
        __u8 *buildid;
        /* الأعضاء أدناه تم تعبئتها فقط بواسطة */
};
