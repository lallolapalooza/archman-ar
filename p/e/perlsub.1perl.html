PERLSUB(1perl) دليل مرجع مبرمجي Perl PERLSUB(1perl)

الاسم

perlsub - الإجراءات الفرعية في Perl (الدوال المعرفة من قبل المستخدم)

الملخص

لتعريف الإجراءات الفرعية:

sub NAME;                       # إعلان "مسبق".
sub NAME(PROTO);                #  مثله، لكن مع النماذج الأولية
sub NAME : ATTRS;               #  مع السمات
sub NAME(PROTO) : ATTRS;        #  مع السمات والنماذج الأولية
sub NAME BLOCK                  # إعلان وتعريف.
sub NAME(PROTO) BLOCK           #  مثله، لكن مع النماذج الأولية
sub NAME : ATTRS BLOCK          #  مع السمات
sub NAME(PROTO) : ATTRS BLOCK   #  مع النماذج الأولية والسمات
use feature 'signatures';
sub NAME(SIG) BLOCK                     # مع التوقيع
sub NAME :ATTRS (SIG) BLOCK             # مع التوقيع والسمات
sub NAME :prototype(PROTO) (SIG) BLOCK  # مع التوقيع والنموذج الأولي

لتعريف إجراء فرعي مجهول في وقت التشغيل:

$subref = sub BLOCK;                    # بدون نموذج أولي
$subref = sub (PROTO) BLOCK;            # مع نموذج أولي
$subref = sub : ATTRS BLOCK;            # مع السمات
$subref = sub (PROTO) : ATTRS BLOCK;    # مع نموذج أولي وسمات
use feature 'signatures';
$subref = sub (SIG) BLOCK;          # مع التوقيع
$subref = sub : ATTRS(SIG) BLOCK;   # مع التوقيع والسمات

لاستيراد الإجراءات الفرعية:

use MODULE qw(NAME1 NAME2 NAME3);

لاستدعاء الإجراءات الفرعية:

NAME(LIST);     # استدعاء إجراء فرعي عادي.
NAME LIST;      # الأقواس اختيارية إذا كان معلنًا مسبقًا/مستوردًا.
&NAME(LIST);    # تجاوز النماذج الأولية.
&NAME;          # يجعل @_ الحالي مرئيًا للإجراء الفرعي المستدعى.

الوصف

مثل العديد من اللغات، توفر Perl إجراءات فرعية معرفة من قبل المستخدم. يمكن أن تكون هذه موجودة في أي مكان في البرنامج الرئيسي، أو محملة من ملفات أخرى عبر الكلمات المفتاحية "do"، "require"، أو "use"، أو مولدة أثناء التشغيل باستخدام "eval" أو الإجراءات الفرعية المجهولة. يمكنك حتى استدعاء دالة بشكل غير مباشر باستخدام متغير يحتوي على اسمها أو مرجع CODE.

نموذج Perl لاستدعاء الدوال وقيم الإرجاع بسيط: يتم تمرير جميع الدوال كمعاملات في قائمة مسطحة واحدة من القيم العددية، وجميع الدوال بالمثل تعيد إلى المستدعي قائمة مسطحة واحدة من القيم العددية. أي مصفوفات أو تجزئات في قوائم الاستدعاء والإرجاع هذه ستنهار، فاقدة هويتها--لكن يمكنك دائمًا استخدام التمرير بالمرجع بدلاً من ذلك لتجنب هذا. يمكن أن تحتوي قوائم الاستدعاء والإرجاع على أكبر أو أقل عدد من العناصر العددية كما تريد. (غالبًا ما تسمى الدالة بدون عبارة إرجاع صريحة إجراءً فرعيًا، لكن لا يوجد فرق حقيقي من منظور Perl.)

في إجراء فرعي يستخدم التوقيعات (انظر "التوقيعات" أدناه)، يتم تعيين الوسائط في متغيرات معجمية يقدمها التوقيع. في التنفيذ الحالي لـ Perl، يمكن الوصول إليها أيضًا في مصفوفة @_ بنفس الطريقة كما في الإجراءات الفرعية غير الموقعة، لكن الوصول إليها بهذه الطريقة غير مشجع الآن داخل إجراء فرعي يستخدم التوقيع.

في إجراء فرعي لا يستخدم التوقيعات، تظهر أي وسائط تمرر في المصفوفة @_. لذلك، إذا استدعيت دالة بوسيطين، فسيتم تخزينهما في $_[0] و $_[1]. المصفوفة @_ هي مصفوفة محلية، لكن عناصرها أسماء مستعارة للمعاملات العددية الفعلية. على وجه الخصوص، إذا تم تحديث عنصر $_[0]، يتم تحديث الوسيط المقابل (أو يحدث خطأ إذا لم يكن قابلًا للتحديث). إذا كان الوسيط عنصر مصفوفة أو تجزئة لم يكن موجودًا عند استدعاء الدالة، فإن ذلك العنصر يُنشأ فقط عندما (وإذا) تم تعديله أو أُخذ مرجع إليه. (بعض الإصدارات السابقة من Perl أنشأت العنصر سواء تم تعيين العنصر أم لا.) التعيين إلى المصفوفة بأكملها @_ يزيل هذا الاسم المستعار، ولا يحدّث أي وسائط.

عند عدم استخدام التوقيعات، لا توفر Perl بطريقة أخرى وسيلة لإنشاء معاملات رسمية مسماة. عمليًا كل ما تفعله هو التعيين إلى قائمة my() من هذه. المتغيرات التي لم يتم الإعلان عنها كخاصة هي متغيرات عامة. للتفاصيل الدقيقة حول إنشاء متغيرات خاصة، انظر "المتغيرات الخاصة عبر my()" و "القيم المؤقتة عبر local()". لإنشاء بيئات محمية لمجموعة من الدوال في حزمة منفصلة (وربما ملف منفصل)، انظر "الحزم" في perlmod.

يمكن استخدام عبارة "return" للخروج من إجراء فرعي، مع تحديد القيمة المعادة اختياريًا، والتي سيتم تقييمها في السياق المناسب (قائمة، عددي، أو فراغ) اعتمادًا على سياق استدعاء الإجراء الفرعي. إذا لم تحدد قيمة إرجاع، فإن الإجراء الفرعي يعيد قائمة فارغة في سياق القائمة، القيمة غير المعرفة في السياق العددي، أو لا شيء في سياق الفراغ. إذا أعدت واحدًا أو أكثر من التجميعات (المصفوفات والتجزئات)، فسيتم تسطيحها معًا في قائمة كبيرة واحدة لا يمكن تمييزها.

إذا لم يتم العثور على "return" وإذا كانت العبارة الأخيرة تعبيرًا، فسيتم إرجاع قيمتها. إذا كانت العبارة الأخيرة بنية تحكم حلقة مثل "foreach" أو "while"، فإن القيمة المعادة غير محددة. الإجراء الفرعي الفارغ يعيد القائمة الفارغة.

مثال:

sub max {
    my $max = shift(@_);
    foreach $foo (@_) {
        $max = $foo if $max < $foo;
    }
    return $max;
}
$bestday = max($mon,$tue,$wed,$thu,$fri);

مثال:

# احصل على سطر، بدمج أسطر المتابعة
#  التي تبدأ بمسافة بيضاء
sub get_line {
    $thisline = $lookahead;  # متغيرات عامة!
    LINE: while (defined($lookahead = <STDIN>)) {
        if ($lookahead =~ /^[ \t]/) {
            $thisline .= $lookahead;
        }
        else {
            last LINE;
        }
    }
    return $thisline;
}
$lookahead = <STDIN>;       # احصل على السطر الأول
while (defined($line = get_line())) {
    ...
}

التعيين إلى قائمة من المتغيرات الخاصة لتسمية وسائطك:

sub maybeset {
    my($key, $value) = @_;
    $Foo{$key} = $value unless $Foo{$key};
}

لأن التعيين ينسخ القيم، فإن هذا له أيضًا تأثير تحويل التمرير بالمرجع إلى تمرير بالقيمة. وإلا فإن الدالة حرة في إجراء تعديلات في المكان على @_ وتغيير قيم المستدعي.

upcase_in($v1, $v2);  # هذا يغير $v1 و $v2
sub upcase_in {
    for (@_) { tr/a-z/A-Z/ }
}

لا يُسمح لك بتعديل الثوابت بهذه الطريقة، بالطبع. إذا كان الوسيط حرفيًا فعليًا وحاولت تغييره، فستحصل على استثناء (من المفترض أنه قاتل). على سبيل المثال، هذا لن يعمل:

upcase_in("frederick");

سيكون أكثر أمانًا بكثير إذا كُتبت دالة upcase_in() لإرجاع نسخة من معاملاتها بدلاً من تغييرها في المكان:

($v3, $v4) = upcase($v1, $v2);  # هذا لا يغير $v1 و $v2
sub upcase {
    return unless defined wantarray;  # سياق فراغ، لا تفعل شيئًا
    my @parms = @_;
    for (@parms) { tr/a-z/A-Z/ }
    return wantarray ? @parms : $parms[0];
}

لاحظ كيف أن هذه الدالة (بدون نموذج أولي) لا تهتم سواء تم تمرير قيم عددية حقيقية أو مصفوفات. ترى Perl جميع الوسائط كقائمة معاملات واحدة كبيرة وطويلة ومسطحة في @_. هذه إحدى المجالات التي يتألق فيها أسلوب تمرير الوسائط البسيط في Perl. دالة upcase() ستعمل بشكل مثالي دون تغيير تعريف upcase() حتى لو غذيناها أشياء مثل هذه:

@newlist   = upcase(@list1, @list2);
@newlist   = upcase( split /:/, $var );

لا تكن مغرًى، مع ذلك، بفعل هذا:

(@x, @y)   = upcase(@list1, @list2);

مثل قائمة المعاملات الواردة المسطحة، فإن قائمة الإرجاع مسطحة أيضًا عند الإرجاع. لذا كل ما تمكنت من فعله هنا هو تخزين كل شيء في @x وجعل @y فارغة. انظر "التمرير بالمرجع" للبدائل.

يمكن استدعاء إجراء فرعي باستخدام بادئة "&" صريحة. علامة "&" اختيارية في Perl الحديثة، كما هي الأقواس إذا كان الإجراء الفرعي معلنًا مسبقًا. علامة "&" ليست اختيارية عند مجرد تسمية الإجراء الفرعي، كما هو الحال عندما يُستخدم كوسيط لـ defined() أو undef(). كما أنها ليست اختيارية عندما تريد إجراء استدعاء إجراء فرعي غير مباشر باسم إجراء فرعي أو مرجع باستخدام تراكيب &$subref() أو "&{$subref}()"، على الرغم من أن تدوين $subref->() يحل تلك المشكلة. انظر perlref```arabic

للمزيد حول كل ذلك.

يمكن استدعاء الروتينات الفرعية بشكل تكراري. إذا تم استدعاء روتين فرعي باستخدام صيغة "&"، فإن قائمة الوسائط اختيارية، وإذا تم حذفها، فلن يتم إعداد مصفوفة @_ للروتين الفرعي: تصبح مصفوفة @_ في وقت الاستدعاء مرئية للروتين الفرعي بدلاً من ذلك. هذه آلية كفاءة قد يرغب المستخدمون الجدد في تجنبها.

&foo(1,2,3);        # تمرير ثلاثة وسائط
foo(1,2,3);         # نفس الشيء
foo();              # تمرير قائمة وسائط فارغة
&foo();             # نفس الشيء
&foo;               # foo() تحصل على الوسائط الحالية، مثل foo(@_)!
use strict 'subs';
foo;                # مثل foo() إذا تم الإعلان المسبق عن sub foo، وإلا
                    # خطأ في وقت التصريف
no strict 'subs';
foo;                # مثل foo() إذا تم الإعلان المسبق عن sub foo، وإلا
                    # سلسلة حرفية "foo"

لا تجعل صيغة "&" قائمة الوسائط اختيارية فحسب، بل تعطل أيضًا أي فحص نموذجي للوسائط التي تقدمها. هذا جزئيًا لأسباب تاريخية، وجزئيًا لأن يكون لديك طريقة مريحة للتحايل إذا كنت تعرف ما تفعله. انظر "النماذج" أدناه.

منذ Perl 5.16.0، الرمز "__SUB__" متاح تحت use feature 'current_sub' و "use v5.16". سيتم تقييمه كمرجع للروتين الفرعي الذي يتم تشغيله حاليًا، والذي يسمح بالاستدعاءات التكرارية دون معرفة اسم الروتين الفرعي الخاص بك.

use v5.16;
my $factorial = sub {
    my ($x) = @_;
    return 1 if $x == 1;
    return($x * __SUB__->( $x - 1 ) );
};

سلوك "__SUB__" داخل كتلة كود تعبير عادي (مثل "/(?{...})/") خاضع للتغيير.

التوقيعات

تمتلك Perl آلية للسماح بتعريف المعلمات الرسمية للروتين الفرعي بواسطة بناء جملة خاص، منفصل عن الكود الإجرائي لجسم الروتين الفرعي. تُعرف قائمة المعلمات الرسمية بأنها توقيع.

يجب تمكين هذه الآلية قبل استخدامها. يتم تمكينها تلقائيًا بواسطة إعلان "use v5.36" (أو أعلى)، أو بشكل مباشر أكثر بواسطة "use feature 'signatures'"، في النطاق الحالي.

التوقيع هو جزء من جسم الروتين الفرعي. عادةً ما يكون جسم الروتين الفرعي مجرد كتلة من الكود محاطة بأقواس معقوفة، ولكن عند استخدام توقيع، يكون التوقيع قائمة بين قوسين تأتي مباشرة قبل الكتلة، بعد أي اسم أو سمات.

على سبيل المثال:

sub foo :lvalue ($x, $y = 1, @z) { .... }

يعلن التوقيع عن متغيرات معجمية موجودة في النطاق للكتلة. عند استدعاء الروتين الفرعي، يتولى التوقيع التحكم أولاً. يقوم بملء متغيرات التوقيع من قائمة الوسائط التي تم تمريرها. إذا لم تفي قائمة الوسائط بمتطلبات التوقيع، فسيتم طرح استثناء. عند اكتمال معالجة التوقيع، يتم تمرير التحكم إلى الكتلة.

يتم التعامل مع المعلمات الموضعية عن طريق تسمية متغيرات قياسية ببساطة في التوقيع. على سبيل المثال:

sub foo ($left, $right) {
    return $left + $right;
}

يأخذ معلمتين موضعيتين، يجب ملؤهما وقت التشغيل بوسيطين. بشكل افتراضي، المعلمات إلزامية، ولا يُسمح بتمرير وسائط أكثر من المتوقع. لذلك ما سبق يعادل

sub foo {
    die "Too many arguments for subroutine" unless @_ <= 2;
    die "Too few arguments for subroutine" unless @_ >= 2;
    my $left = $_[0];
    my $right = $_[1];
    return $left + $right;
}

يمكن تجاهل الوسيط عن طريق حذف الجزء الرئيسي من الاسم من تعريف المعلمة، وترك فقط رمز "$" عاريًا. على سبيل المثال:

sub foo ($first, $, $third) {
    return "first=$first, third=$third";
}

على الرغم من أن الوسيط المتجاهل لا ينتقل إلى متغير، إلا أنه لا يزال إلزاميًا للمستدعي تمريره.

تصبح المعلمة الموضعية اختيارية عن طريق إعطائها قيمة افتراضية، مفصولة باسم المعلمة بـ "=":

sub foo ($left, $right = 0) {
    return $left + $right;
}

يمكن استدعاء الروتين الفرعي أعلاه بوسيط واحد أو اثنين. يتم تقييم تعبير القيمة الافتراضية عند استدعاء الروتين الفرعي، لذلك يمكنه توفير قيم افتراضية مختلفة لاستدعاءات مختلفة. يتم تقييمه فقط إذا تم حذف الوسيط فعليًا من الاستدعاء. على سبيل المثال:

my $auto_id = 0;
sub foo ($thing, $id = $auto_id++) {
    print "$thing has ID $id";
}

يقوم تلقائيًا بتعيين معرفات تسلسلية مميزة للأشياء التي لم يتم توفير معرف لها من قبل المستدعي. يمكن لتعبير القيمة الافتراضية أيضًا الإشارة إلى معلمات سابقة في التوقيع، مما يجعل الافتراضي لمعلمة واحدة يختلف وفقًا للمعلمات السابقة. على سبيل المثال:

sub foo ($first_name, $surname, $nickname = $first_name) {
    print "$first_name $surname is known as \"$nickname\"";
}

منذ Perl 5.38، يمكن أيضًا كتابة تعبير القيمة الافتراضية باستخدام عامل التشغيل "//="، حيث سيتم تقييمه واستخدامه إذا قام المستدعي بحذف قيمة أو كانت القيمة المقدمة هي "undef".

sub foo ($name //= "world") {
    print "Hello, $name";
}
foo(undef);  # سيطبع "Hello, world"

Variables

Perl variables are of several kinds: scalars, arrays, and hashes. These are indicated by the leading character:

$
Scalars (numbers, strings, references)
@
Arrays (lists of scalars)
%
Hashes (key/value mappings)
&
Subroutines (unless qualified by a package)
*
Symbol table entries (globs)

Names following these sigils must start with a letter or underscore, and may contain letters, numbers, and underscores. Names are case-sensitive. (Note: a bareword that starts with a letter or underscore, and contains only letters, numbers, and underscores, can be interpreted as a subroutine name. Names starting with punctuation are reserved for Perl's special variables.)

By default, all variables are global. You may use the "my" keyword to create lexically scoped variables, or the "our" keyword to create lexical variables that are also visible outside the current package (but not outside the current file). You may also use the "state" keyword to create lexically scoped variables that persist between calls to a subroutine.

You can declare variables to be local to the current scope using "local". This is different from "my" and "our". See "Temporary Values via local()" below.

You may use the "use strict;" pragma to enforce stricter rules about variable usage, which often helps prevent bugs.

Lexical Variables

A lexical variable is one that is declared with "my", "our", or "state". It is visible only within the block in which it is declared, and within any nested blocks or subroutines called from that block. The scope of a lexical variable extends from its declaration through the end of the block in which it is declared.

The scope of a lexical variable declared in an "if", "unless", "while", or "for" statement "span" extends from its declaration through the rest of that conditional, including any "elsif" and "else" clauses, but not beyond it. See "Simple Statements" in perlsyn for information on the scope of variables in statements with modifiers.

The "foreach" loop defaults to scoping its index variable dynamically in the manner of "local". However, if the index variable is prefixed with the keyword "my", or if there is already a lexical by that name in scope, then a new lexical is created instead. Thus in the loop 

for my $i (1, 2, 3) {
    some_function();
}

the scope of $i extends to the end of the loop, but not beyond it, rendering the value of $i inaccessible within some_function().

Some users may wish to encourage the use of lexically scoped variables. As an aid to catching implicit uses to package variables, which are always global, if you say 

use strict 'vars';

then any variable mentioned from there to the end of the enclosing block must either refer to a lexical variable, be predeclared via "our" or "use vars", or else must be fully qualified with the package name. A compilation error results otherwise. An inner block may countermand this with "no strict 'vars'".

A "my" has both a compile-time and a run-time effect. At compile time, the compiler takes notice of it. The principal usefulness of this is to quiet "use strict 'vars'", but it is also essential for generation of closures as detailed in perlref. Actual initialization is delayed until run time, though, so it gets executed at the appropriate time, such as each time through a loop, for example.

Variables declared with "my" are not part of any package and are therefore never fully qualified with the package name. In particular, you're not allowed to try to make a package variable (or other global) lexical: 

my $pack::var;      # ERROR!  Illegal syntax

In fact, a package or global variable is still accessible using the fully qualified "::" notation even while a lexical of the same name is also visible: 

package main;
our $x = 10;
my  $x = 20;
print "$x and $::x\n";

That will print out 20 and 10.

You may declare "my" variables at the outermost scope of a file to hide any such identifiers from the world outside that file. This is similar in spirit to C's static variables when they are used at the file level. To do this with a subroutine requires the use of a closure (an anonymous function that accesses enclosing lexicals). If you want to create a private subroutine that cannot be called from outside that block, it can declare a lexical variable containing an anonymous sub reference: 

my $secret_version = '1.001-beta';
my $secret_sub = sub { print $secret_version };
$secret_sub->();

As long as the reference is never returned by any function within the module, no outside module can see the subroutine, because its name is not in any package's symbol table. Remember that it's not REALLY called $some_pack::secret_version or anything; it's just $secret_version, unqualified and unqualifiable.

This does not work with object methods, however; all object methods have to be in the symbol table of some package to be found. See "Function Templates" in perlref for something of a work-around to this.

Persistent Private Variables

There are two ways to build persistent private variables in Perl 5.10. First, you can simply use the "state" feature. Or, you can use closures, if you want to stay compatible with releases older than 5.10.

Persistent variables via state()

Beginning with Perl 5.10.0, you can declare variables with the "state" keyword in place of "my". For that to work, though, you must have enabled that feature beforehand, either by using the "feature" pragma, or by using "-E" on one-liners (see feature). Beginning with Perl 5.16, the "CORE::state" form does not require the "feature" pragma.

The "state" keyword creates a lexical variable (following the same scoping rules as "my") that persists from one subroutine call to the next. If a state variable resides inside an anonymous subroutine, then each copy of the subroutine has its own copy of the state variable. However, the value of the state variable will still persist between calls to the same copy of the anonymous subroutine. (Don't forget that "sub { ... }" creates a new subroutine each time it is executed.)

For example, the following code maintains a private counter, incremented each time the gimme_another() function is called: 

use feature 'state';
sub gimme_another { state $x; return ++$x }

And this example uses anonymous subroutines to create separate counters: 

use feature 'state';
sub create_counter {
    return sub { state $x; return ++$x }
}

Also, since $x is lexical, it can't be reached or modified by any Perl code outside.

When combined with variable declaration, simple assignment to "state" variables (as in "state $x = 42") is executed only the first time. When such statements are evaluated subsequent times, the assignment is ignored. The behavior of assignment to "state" declarations where the left hand side of the assignment involves any parentheses is currently undefined.

Persistent variables with closures

Just because a lexical variable is lexically (also called statically) scoped to its enclosing block, "eval", or "do" FILE, this doesn't mean that within a function it works like a C static. It normally works more like a C auto, but with implicit garbage collection.

Unlike local variables in C or C++, Perl's lexical variables don't necessarily get recycled just because their scope has exited. If something more permanent is still aware of the lexical, it will stick around. So long as something else references a lexical, that lexical won't be freed--which is as it should be. You wouldn't want memory being free until you were done using it, or kept around once you were done. Automatic garbage collection takes care of this for you.

This means that you can pass back or save away references to lexical variables, whereas to return a pointer to a C auto is a grave error. It also gives us a way to simulate C's function statics. Here's a mechanism for giving a function private variables with both lexical scoping and a static lifetime. If you do want to create something like C's static variables, just enclose the whole function in an extra block, and put the static variable outside the function but in the block. 

{
    my $secret_val = 0;
    sub gimme_another {
        return ++$secret_val;
    }
}

# $secret_val now becomes unreachable by the outside # world, but retains its value between calls to gimme_another

If this function is being sourced in from a separate file via "require" or "use", then this is probably just fine. If it's all in the main program, you'll need to arrange for the "my" to be executed early, either by putting the whole block above your main program, or more likely, placing merely a "BEGIN" code block around it to make sure it gets executed before your program starts to run: 

BEGIN {
    my $secret_val = 0;
    sub gimme_another {
        return ++$secret_val;
    }
}

See "BEGIN, UNITCHECK, CHECK, INIT and END" in perlmod about the special triggered code blocks, "BEGIN", "UNITCHECK", "CHECK", "INIT" and "END".

If declared at the outermost scope (the file scope), then lexicals work somewhat like C's file statics. They are available to all functions in that same file declared below them, but are inaccessible from outside that file. This strategy is sometimes used in modules to create private variables that the whole module can see.

```من داخل تلك الكتلة. إن "local" يعطي فقط قيمًا مؤقتة للمتغيرات العامة (بمعنى متغيرات الحزمة). وهو لا ينشئ متغيرًا محليًا. يُعرف هذا بالنطاق الديناميكي. أما النطاق المعجمي فيتم باستخدام "my"، والذي يعمل أقرب إلى إعلانات auto في لغة C.

بعض أنواع القيم القابلة للإسناد (lvalues) يمكن توطينها أيضًا: عناصر وشرائح المصفوفات والتجزئات، والشرطيات (بشرط أن تكون نتيجتها قابلة للتوطين دائمًا)، والمراجع الرمزية. وكما هو الحال مع المتغيرات البسيطة، فإن هذا ينشئ قيمًا جديدة ذات نطاق ديناميكي.

إذا أُعطي أكثر من متغير أو تعبير واحد إلى "local"، فيجب وضعها بين أقواس. يعمل هذا العامل بحفظ القيم الحالية لتلك المتغيرات في قائمة وسائطه على مكدس مخفي واستعادتها عند الخروج من الكتلة أو الروتين الفرعي أو eval. وهذا يعني أن الروتينات الفرعية المستدعاة يمكنها أيضًا الإشارة إلى المتغير المحلي، ولكن ليس إلى المتغير العام. ويمكن إسناد قائمة الوسائط إذا رغبت، مما يتيح لك تهيئة متغيراتك المحلية. (إذا لم يُعطَ مُهيّئ لمتغير معين، فإنه يُنشأ بقيمة غير معرّفة.)

لأن "local" عامل وقت تشغيل، فإنه يُنفَّذ في كل مرة عبر حلقة. وبالتالي، فمن الأكثر كفاءة توطين متغيراتك خارج الحلقة.

ملاحظة نحوية حول local()

إن "local" هو ببساطة معدِّل على تعبير lvalue. عندما تُسند إلى متغير مُوطَّن بـ "local"، فإن "local" لا يغيّر ما إذا كانت قائمته تُعامل كمتغير عددي أو كمصفوفة. لذا فإن

local($foo) = <STDIN>;
local @FOO = <STDIN>;

كلاهما يوفّر سياق قائمة للطرف الأيمن، بينما

local $foo = <STDIN>;

يوفّر سياقًا عدديًا.

توطين المتغيرات الخاصة

إذا وطّنت متغيرًا خاصًا، فستعطيه قيمة جديدة، لكن سحره لن يزول. وهذا يعني أن جميع الآثار الجانبية المرتبطة بهذا السحر لا تزال تعمل مع القيمة المُوطَّنة.

تتيح هذه الميزة عمل شيفرة كهذه:

# Read the whole contents of FILE in $slurp
{ local $/ = undef; $slurp = <FILE>; }

لاحظ، مع ذلك، أن هذا يقيّد توطين بعض القيم؛ فعلى سبيل المثال، تفشل العبارة التالية، اعتبارًا من Perl 5.10.0، بخطأ Modification of a read-only value attempted، لأن المتغير $1 سحري وقراءة فقط:

local $1 = 2;

أحد الاستثناءات هو المتغير العددي الافتراضي: بدءًا من Perl 5.14 فإن local($_) سيزيل دائمًا كل السحر من $_، لجعل من الممكن إعادة استخدام $_ بأمان في روتين فرعي.

تحذير: توطين المصفوفات والتجزئات المربوطة (tied) لا يعمل حاليًا كما هو موصوف. سيُصلح هذا في إصدار مستقبلي من Perl؛ في غضون ذلك، تجنّب الشيفرة التي تعتمد على أي سلوك معين لتوطين المصفوفات أو التجزئات المربوطة (توطين العناصر الفردية لا يزال مقبولًا). راجع "Localising Tied Arrays and Hashes Is Broken" في perl58delta لمزيد من التفاصيل.

توطين الـ globs

التركيب

local *name;

ينشئ مدخل جدول رموز جديدًا بالكامل للـ glob "name" في الحزمة الحالية. وهذا يعني أن جميع المتغيرات في خانة الـ glob الخاصة به ($name، @name، %name، &name، ومقبض الملف "name") تُعاد ضبطها ديناميكيًا.

وهذا يعني، من بين أمور أخرى، أن أي سحر تحمله تلك المتغيرات في النهاية يُفقد محليًا. بعبارة أخرى، فإن قول "local */" لن يكون له أي تأثير على القيمة الداخلية لفاصل سجلات الإدخال.

توطين عناصر الأنواع المركبة

يجدر أيضًا أخذ لحظة لشرح ما يحدث عندما تُوطّن عضوًا من نوع مركب (أي عنصر مصفوفة أو تجزئة). في هذه الحالة، يُوطَّن العنصر بالاسم. وهذا يعني أنه عندما ينتهي نطاق local()، ستُستعاد القيمة المحفوظة إلى عنصر التجزئة الذي سُمّي مفتاحه في local()، أو إلى عنصر المصفوفة الذي سُمّي فهرسه في local(). وإذا حُذف ذلك العنصر أثناء سريان مفعول local() (مثلًا بواسطة delete() من تجزئة أو shift() من مصفوفة)، فسيعود إلى الوجود، وربما يمدّ المصفوفة ويملأ العناصر المتخطاة بـ "undef". على سبيل المثال، إذا قلت

%hash = ( 'This' => 'is', 'a' => 'test' );
@ary  = ( 0..5 );
{
    local($ary[5]) = 6;
    local($hash{'a'}) = 'drill';
    while (my $e = pop(@ary)) {
        print "$e . . .\n";
        last unless $e > 3;
    }
    if (@ary) {
        $hash{'only a'} = 'test';
        delete $hash{'a'};
    }
}
print join(' ', map { "$_ $hash{$_}" } sort keys %hash),".\n";
print "The array has ",scalar(@ary)," elements: ",
    join(', ', map { defined $_ ? $_ : 'undef' } @ary),"\n";

سيطبع Perl

6 . . .
4 . . .
3 . . .
This is a test only a test.
The array has 6 elements: 0, 1, 2, undef, undef, 5

سلوك local() على الأعضاء غير الموجودة في الأنواع المركبة عرضة للتغيير في المستقبل. وسلوك local() على عناصر المصفوفة المحددة باستخدام فهارس سالبة مفاجئ بشكل خاص، ومن المرجح جدًا أن يتغير.

الحذف المُوطَّن لعناصر الأنواع المركبة

يمكنك استخدام التركيبين "delete local $array[$idx]" و "delete local $hash{key}" لحذف مدخل نوع مركب للكتلة الحالية واستعادته عند انتهائها. وهما يعيدان قيمة المصفوفة/التجزئة قبل التوطين، مما يعني أنهما مكافئان على التوالي لـ

do {
    my $val = $array[$idx];
    local  $array[$idx];
    delete $array[$idx];
    $val
}

و

do {
    my $val = $hash{key};
    local  $hash{key};
    delete $hash{key};
    $val
}

باستثناء أن "local" في هاتين الحالتين يكون نطاقه كتلة "do". والشرائح مقبولة أيضًا.

my %hash = (
    a => [ 7, 8, 9 ],
    b => 1,
)
{
    my $x = delete local $hash{a};
    # $x is [ 7, 8, 9 ]
    # %hash is (b => 1)
    {
        my @nums = delete local @$x[0, 2]
        # @nums is (7, 9)
        # $x is [ undef, 8 ]
        $x[0] = 999; # will be erased when the scope ends
    }
    # $x is back to [ 7, 8, 9 ]
}
# %hash is back to its original state

هذا التركيب مدعوم منذ Perl v5.12.

الروتينات الفرعية القابلة للإسناد (Lvalue)

من الممكن إرجاع قيمة قابلة للتعديل من روتين فرعي. للقيام بذلك، يجب أن تعلن أن الروتين الفرعي يعيد lvalue.

my $val;
sub canmod : lvalue {
    $val;  # or:  return $val;
}
sub nomod {
    $val;
}
canmod() = 5;   # assigns to $val
nomod()  = 5;   # ERROR

يُحدَّد سياق العددية/القائمة للروتين الفرعي وللطرف الأيمن من الإسناد كما لو أن استدعاء الروتين الفرعي قد استُبدل بقيمة عددية. على سبيل المثال، انظر إلى:

data(2,3) = get_data(3,4);

كلا الروتينين الفرعيين هنا يُستدعيان في سياق عددي، بينما في:

(data(2,3)) = get_data(3,4);

وفي:

(data(2),data(3)) = get_data(3,4);

تُستدعى جميع الروتينات الفرعية في سياق قائمة.

الروتينات الفرعية القابلة للإسناد مريحة، لكن يجب أن تضع في اعتبارك أنها، عند استخدامها مع الكائنات، قد تنتهك التغليف. يمكن لمعدِّل عادي التحقق من الوسيطة المقدمة قبل تعيين السمة التي يحميها، أما الروتين الفرعي القابل للإسناد فلا يمكنه ذلك. إذا كنت تتطلب أي معالجة خاصة عند تخزين القيم واسترجاعها، ففكر في استخدام وحدة CPAN المسماة Sentinel أو شيء مشابه.

الروتينات الفرعية المعجمية

بدءًا من Perl 5.18، يمكنك الإعلان عن روتين فرعي خاص باستخدام "my" أو "state". وكما هو الحال مع متغيرات الحالة، فإن الكلمة المفتاحية "state" متاحة فقط تحت "use feature 'state'" أو "use v5.10" أو أعلى.

قبل Perl 5.26، كانت الروتينات الفرعية المعجمية تُعتبر تجريبية وكانت متاحة فقط تحت pragma "use feature 'lexical_subs'". وكانت أيضًا تنتج تحذيرًا ما لم تُعطَّل فئة التحذيرات "experimental::lexical_subs".

هذه الروتينات الفرعية مرئية فقط داخل الكتلة التي أُعلنت فيها، وفقط بعد ذلك الإعلان:

# Include these two lines if your code is intended to run under Perl
# versions earlier than 5.26.
no warnings "experimental::lexical_subs";
use feature 'lexical_subs';
foo();              # calls the package/global subroutine
state sub foo {
    foo();          # also calls the package subroutine
}
foo();              # calls "state" sub
my $ref = \&foo;    # take a reference to "state" sub
my sub bar { ... }
bar();              # calls "my" sub

لا يمكنك (مباشرة) كتابة روتين فرعي معجمي تكراري:

# WRONG
my sub baz {
    baz();
}

يفشل هذا المثال لأن baz() يشير إلى الروتين الفرعي العام/الحزمة "baz"، وليس إلى الروتين الفرعي المعجمي الذي يُعرَّف حاليًا.

الحل هو استخدام "__SUB__":

my sub baz {
    __SUB__->();    # calls itself
}

من الممكن الإعلان المسبق عن روتين فرعي معجمي. يحترم تركيب تعريف الروتين الفرعي "sub foo {...}" أي إعلان سابق "my sub;" أو "state sub;". ومع ذلك، فإن استخدام هذا لتعريف روتينات فرعية تكرارية فكرة سيئة:

my sub baz;         # predeclaration
sub baz {           # define the "my" sub
    baz();          # WRONG: calls itself, but leaks memory
}

تمامًا مثل "my $f; $f = sub { $f->() }"، فإن هذا المثال يسرّب الذاكرة. فالاسم "baz" مرجع إلى الروتين الفرعي، والروتين الفرعي يستخدم الاسم "baz"؛ فيُبقيان بعضهما حيًا (راجع "Circular References" في perlref).

"state sub" مقابل "my sub"

ما الفرق بين "state" subs و "my" subs؟ في كل مرة يدخل التنفيذ كتلةً عند الإعلان عن "my" subs، تُنشأ نسخة جديدة من كل sub. تستمر "State" subroutines من تنفيذ واحد للكتلة الحاوية إلى التنفيذ التالي.

لذا، بشكل عام، "state" subroutines أسرع. لكن "my" subs ضرورية إذا أردت إنشاء closures:

sub whatever {
    my $x = shift;
    my sub inner {
        ... do something with $x ...
    }
    inner();
}

في هذا المثال، تُنشأ $x جديدة عند استدعاء "whatever"، وكذلك تُنشأ "inner" جديدة، والتي يمكنها رؤية $x الجديدة. أما "state" sub فسترى فقط $x من أول استدعاء لـ "whatever".

"our" subroutines

مثل "our $variable"، فإن "our sub" ينشئ alias معجميًا لـ package subroutine التي تحمل الاسم نفسه.

الاستخدامان الرئيسيان لهذا هما العودة إلى استخدام package sub داخل نطاق داخلي:

sub foo { ... }
sub bar {
    my sub foo { ... }
    {
        # need to use the outer foo here
        our sub foo;
        foo();
    }
}

وجعل subroutine مرئية لـ packages أخرى في النطاق نفسه:

package MySneakyModule;
our sub do_something { ... }
sub do_something_with_caller {
    package DB;
    () = caller 1;          # sets @DB::args
    do_something(@args);    # uses MySneakyModule::do_something
}

تمرير مدخلات جدول الرموز (typeglobs)

تحذير: كانت الآلية الموصوفة في هذا القسم في الأصل الطريقة الوحيدة لمحاكاة pass-by-reference في الإصدارات الأقدم من Perl. ورغم أنها لا تزال تعمل بشكل جيد في الإصدارات الحديثة، فإن آلية المراجع الجديدة أسهل عمومًا في التعامل. انظر أدناه.

أحيانًا لا تريد تمرير قيمة مصفوفة إلى subroutine بل اسمَها، حتى يتمكن subroutine من تعديل النسخة العامة منها بدلًا من العمل على نسخة محلية. في Perl يمكنك الإشارة إلى جميع الكائنات ذات اسم معين بوضع نجمة قبل الاسم: *foo. يُعرف هذا غالبًا باسم "typeglob"، لأن النجمة في المقدمة يمكن اعتبارها مطابقة wildcard لجميع أحرف البادئة الغريبة على المتغيرات وsubroutines وما شابه.

عند تقييمه، يُنتج typeglob قيمة scalar تمثل جميع الكائنات ذات ذلك الاسم، بما في ذلك أي filehandle أو format أو subroutine. وعند الإسناد إليه، فإنه يجعل الاسم المذكور يشير إلى أي قيمة "*" أُسندت إليه. مثال:

sub doubleary {
    local(*someary) = @_;
    foreach $elem (@someary) {
        $elem *= 2;
    }
}
doubleary(*foo);
doubleary(*bar);

تُمرَّر Scalars بالفعل بالمرجع، لذا يمكنك تعديل وسائط scalar دون استخدام هذه الآلية بالإشارة صراحةً إلى $_[0] وما شابه. يمكنك تعديل جميع عناصر مصفوفة بتمرير جميع العناصر كـ scalars، لكن عليك استخدام آلية "*" (أو آلية المرجع المكافئة) لـ "push" أو "pop" أو تغيير حجم مصفوفة. سيكون تمرير typeglob (أو المرجع) أسرع بالتأكيد.

حتى إذا لم ترغب في تعديل مصفوفة، فإن هذه الآلية مفيدة لتمرير مصفوفات متعددة في LIST واحد، لأن آلية LIST عادةً تدمج جميع قيم المصفوفات بحيث لا يمكنك استخراج المصفوفات الفردية. لمزيد عن typeglobs، انظر "Typeglobs and Filehandles" في perldata.

متى لا يزال يتعين استخدام local()

رغم وجود "my"، لا تزال هناك ثلاثة مواضع يتألق فيها العامل "local". في الواقع، في هذه المواضع الثلاثة، يجب عليك استخدام "local" بدلًا من "my".

1.
تحتاج إلى إعطاء متغير عام قيمة مؤقتة، خصوصًا $_.

المتغيرات العامة، مثل @ARGV أو متغيرات الترقيم، يجب أن تُحوَّل إلى "local" باستخدام local(). تقرأ هذه الكتلة /etc/motd، وتقسمه إلى أجزاء تفصلها أسطر من علامات يساوي متساوية، تُوضع في @Fields.

{
    local @ARGV = ("/etc/motd");
    local $/ = undef;
    local $_ = <>;
    @Fields = split /^\s*=+\s*$/;
}

على وجه الخصوص، من المهم تحويل $_ إلى "local" في أي روتين يُسند إليه. احترس من الإسنادات الضمنية في شرطيات "while".

2.
تحتاج إلى إنشاء file أو directory handle محلي أو دالة محلية.

الدالة التي تحتاج إلى filehandle خاص بها يجب أن تستخدم local() على typeglob كامل. يمكن استخدام هذا لإنشاء مدخلات جديدة في جدول الرموز:

sub ioqueue {
    local  (*READER, *WRITER);    # not my!
    pipe    (READER,  WRITER)     or die "pipe: $!";
    return (*READER, *WRITER);
}
($head, $tail) = ioqueue();

انظر وحدة Symbol لطريقة إنشاء مدخلات مجهولة في جدول الرموز.

لأن إسناد مرجع إلى typeglob ينشئ alias، يمكن استخدام هذا لإنشاء ما هو فعليًا دالة محلية، أو على الأقل alias محلي.

{
    local *grow = \&shrink; # only until this block exits
    grow();                # really calls shrink()
    move();                # if move() grow()s, it shrink()s too
}
grow();                    # get the real grow() again

انظر "Function Templates" في perlref لمزيد حول معالجة الدوال بالاسم بهذه الطريقة.

3.
تريد تغيير عنصر واحد فقط من مصفوفة أو hash مؤقتًا.

يمكنك تحويل عنصر واحد فقط من aggregate إلى "local". عادةً يُفعل هذا على dynamics:

{
    local $SIG{INT} = 'IGNORE';
    funct();                            # uninterruptible
}
# interruptibility automatically restored here

لكنه يعمل أيضًا على aggregates المعلنة معجميًا.

التمرير بالمرجع

إذا أردت تمرير أكثر من مصفوفة أو hash إلى دالة -- أو إرجاعها منها -- والحفاظ على سلامتها، فستحتاج إلى pass-by-reference صريح. قبل أن تفعل ذلك، تحتاج إلى فهم المراجع كما هو مفصل في perlref. قد لا يبدو هذا القسم منطقيًا كثيرًا لك بخلاف ذلك.

إليك بعض الأمثلة البسيطة. أولًا، لنمرر عدة مصفوفات إلى دالة ونجعلها تنفذ "pop" على جميعها، مع إرجاع قائمة جديدة بجميع عناصرها الأخيرة السابقة:

@tailings = popmany ( \@w, \@x, \@y, \@z );
sub popmany {
    my $aref;
    my @retlist;
    foreach $aref ( @_ ) {
        push @retlist, pop @$aref;
    }
    return @retlist;
}

إليك كيف يمكنك كتابة دالة تُرجع قائمة بالمفاتيح الموجودة في جميع الـ hashes الممررة إليها:

@common = inter( \%foo, \%bar, \%joe );
sub inter {
    my ($k, $href, %seen); # locals
    foreach $href (@_) {
        while ( $k = each %$href ) {
            $seen{$k}++;
        }
    }
    return grep { $seen{$_} == @_ } keys %seen;
}

حتى الآن، نستخدم فقط آلية إرجاع القائمة العادية. ماذا يحدث إذا أردت تمرير أو إرجاع hash؟ حسنًا، إذا كنت تستخدم واحدًا فقط منها، أو لا تمانع دمجها، فإن اصطلاح الاستدعاء العادي مقبول، رغم أنه مكلف قليلًا.

حيث يقع الناس في المشكلة هو هنا:

    (@w, @x) = func(@y, @z);
or
    (%w, %x) = func(%y, %z);

هذه الصيغة لن تعمل ببساطة. إنها تضبط @w أو %w فقط وتفرغ @x أو %x. إضافةً إلى أن الدالة لم تُمرَّر إليها مصفوفتان أو hashان منفصلان: بل استلمت قائمة طويلة واحدة في @_، كما هو الحال دائمًا.

إذا كان بإمكانك ترتيب أن يتعامل الجميع مع هذا عبر المراجع، فسيكون الكود أنظف، رغم أنه ليس جميل المظهر. إليك دالة تأخذ مرجعي مصفوفة كوسيطين، وتُرجع عنصري المصفوفة بترتيب عدد العناصر فيهما:

($wref, $xref) = func(\@y, \@z);
print "@$wref has more than @$xref\n";
sub func {
    my ($yref, $zref) = @_;
    if (@$yref > @$zref) {
        return ($yref, $zref);
    } else {
        return ($zref, $yref);
    }
}

اتضح أنه يمكنك في الواقع فعل هذا أيضًا:

(*w, *x) = func(\@y, \@z);
print "@w has more than @x\n";
sub func {
    local (*y, *z) = @_;
    if (@y > @z) {
        return (\@y, \@z);
    } else {
        return (\@z, \@y);
    }
}

هنا نستخدم typeglobs لعمل aliasing في جدول الرموز. الأمر دقيق بعض الشيء، كما أنه لن يعمل إذا كنت تستخدم متغيرات "my"، لأن globals فقط (حتى لو متنكرة كـ "local"s) هي الموجودة في جدول الرموز.

إذا كنت تمرر filehandles، فيمكنك عادةً استخدام typeglob المجرد فقط، مثل *STDOUT، لكن مراجع typeglobs تعمل أيضًا. على سبيل المثال:

splutter(\*STDOUT);
sub splutter {
    my $fh = shift;
    print $fh "her um well a hmmm\n";
}
$rec = get_rec(\*STDIN);
sub get_rec {
    my $fh = shift;
    return scalar <$fh>;
}

إذا كنت تخطط لإنشاء filehandles جديدة، يمكنك فعل هذا. لاحظ أنك تُرجع *FH المجرد فقط، وليس مرجعه.

sub openit {
    my $path = shift;
    local *FH;
    return open (FH, $path) ? *FH : undef;
}

النماذج الأولية

تدعم Perl نوعًا محدودًا جدًا من التحقق من الوسائط وقت الترجمة باستخدام function prototyping. يمكن الإعلان عن هذا إما في قسم PROTO أو باستخدام سمة prototype. إذا أعلنت أيًا من

sub mypush (\@@)
sub mypush :prototype(\@@)

فإن mypush() تأخذ وسائط تمامًا مثل push().

إذا كانت subroutine signature

مفعّلة (انظر "التوقيعات")، فإن صيغة PROTO الأقصر غير متاحة، لأنها ستتعارض مع التوقيعات. في هذه الحالة، لا يمكن الإعلان عن نموذج أولي إلا في شكل سمة.

يجب أن يكون إعلان الدالة مرئيًا في وقت الترجمة. يؤثر النموذج الأولي فقط على تفسير الاستدعاءات العادية للدالة، حيث تُعرّف العادية بأنها لا تستخدم علامة "&". بعبارة أخرى، إذا استدعيتها كدالة مدمجة، فإنها تتصرف كدالة مدمجة. وإذا استدعيتها كإجراء فرعي قديم الطراز (perl4)، فإنها تتصرف كإجراء فرعي قديم الطراز. وينتج عن هذه القاعدة بطبيعة الحال أن النماذج الأولية ليس لها أي تأثير على مراجع الإجراءات الفرعية مثل "\&foo" أو على استدعاءات الإجراءات الفرعية غير المباشرة مثل "&{$subref}()" أو $subref->().

لا تتأثر استدعاءات الطرق بالنماذج الأولية أيضًا، لأن الدالة المطلوب استدعاؤها غير محددة في وقت الترجمة، حيث يعتمد الكود الدقيق المستدعى على الوراثة.

نظرًا لأن الغرض من هذه الميزة هو في المقام الأول السماح لك بتعريف إجراءات فرعية تعمل مثل الدوال المدمجة، فإليك نماذج أولية لبعض الدوال الأخرى التي تُحلَّل تقريبًا تمامًا مثل الدوال المدمجة المقابلة.

مُعلنة كـ             تُستدعى كـ
sub mylink ($$)         mylink $old, $new
sub myvec ($$$)         myvec $var, $offset, 1
sub myindex ($$;$)      myindex getstring(), "substr"
sub mysyswrite ($$$;$)  mysyswrite $buf, 0, length($buf) - $off, $off
sub myreverse (@)       myreverse $x, $y, $z
sub myjoin ($@)         myjoin ":", $x, $y, $z
sub mypop (\@)          mypop @array
sub mysplice (\@$$@)    mysplice @array, 0, 2, @pushme
sub mykeys (\[%@])      mykeys $hashref->%*
sub myopen (*;$)        myopen HANDLE, $name
sub mypipe (**)         mypipe READHANDLE, WRITEHANDLE
sub mygrep (&@)         mygrep { /foo/ } $x, $y, $z
sub myrand (;$)         myrand 42
sub mytime ()           mytime

أي حرف نموذج أولي مسبوق بشرطة مائلة عكسية يمثل وسيطة فعلية يجب أن تبدأ بذلك الحرف (اختياريًا مسبوقًا بـ "my" أو "our" أو "local")، باستثناء "$"، الذي سيقبل أي تعبير قيمة قابلة للتعديل قياسي، مثل "$foo = 7" أو "my_function()->[0]". ستكون القيمة الممررة كجزء من @_ مرجعًا إلى الوسيطة الفعلية المعطاة في استدعاء الإجراء الفرعي، والتي يتم الحصول عليها بتطبيق "\" على تلك الوسيطة.

يمكنك استخدام تدوين مجموعة الشرطة المائلة العكسية "\[]" لتحديد أكثر من نوع وسيطة مسموح به. على سبيل المثال:

sub myref (\[$@%&*])

سيسمح باستدعاء myref() كـ

myref $var
myref @array
myref %hash
myref &sub
myref *glob

وستكون الوسيطة الأولى لـ myref() مرجعًا إلى قيمة قياسية، أو مصفوفة، أو تجزئة، أو إجراء فرعي، أو glob.

لأحرف النموذج الأولي غير المسبوقة بشرطة مائلة عكسية معانٍ خاصة. أي "@" أو "%" غير مسبوق بشرطة مائلة عكسية يلتهم جميع الوسيطات المتبقية، ويفرض سياق القائمة. الوسيطة الممثلة بـ "$" تفرض سياق القيمة القياسية. أما "&" فيتطلب إجراءً فرعيًا مجهولًا، والذي، إذا مُرر كوسيطة أولى، قد يبدو ككتلة مجردة: لا يتطلب الكلمة المفتاحية "sub" أو فاصلة لاحقة.

تسمح "*" للإجراء الفرعي بقبول كلمة مجردة، أو ثابت، أو تعبير قياسي، أو typeglob، أو مرجع إلى typeglob في ذلك الموضع. ستكون القيمة متاحة للإجراء الفرعي إما كقيمة قياسية بسيطة، أو (في الحالتين الأخيرتين) كمرجع إلى typeglob. إذا كنت ترغب في تحويل مثل هذه الوسيطات دائمًا إلى مرجع typeglob، فاستخدم Symbol::qualify_to_ref() كما يلي:

use Symbol 'qualify_to_ref';
sub foo (*) {
    my $fh = qualify_to_ref(shift, caller);
    ...
}

النموذج الأولي "+" هو بديل خاص لـ "$" يتصرف مثل "\[@%]" عند إعطائه متغير مصفوفة أو تجزئة حرفيًا، ولكنه بخلاف ذلك سيفرض سياق القيمة القياسية على الوسيطة. هذا مفيد للدوال التي يجب أن تقبل إما مصفوفة حرفية أو مرجع مصفوفة كوسيطة:

sub mypush (+@) {
    my $aref = shift;
    die "Not an array or arrayref" unless ref $aref eq 'ARRAY';
    push @$aref, @_;
}

عند استخدام النموذج الأولي "+"، يجب أن تتحقق دالتك من أن الوسيطة من نوع مقبول.

تفصل الفاصلة المنقوطة (";") الوسيطات الإلزامية عن الوسيطات الاختيارية. وهي زائدة عن الحاجة قبل "@" أو "%"، اللذين يلتهمان كل شيء آخر.

كحرف أخير في النموذج الأولي، أو قبل فاصلة منقوطة مباشرة، أو "@" أو "%"، يمكنك استخدام "_" بدلًا من "$": إذا لم تُقدَّم هذه الوسيطة، فسيُستخدم $_ بدلًا منها.

لاحظ كيف تُعامَل الأمثلة الثلاثة الأخيرة في الجدول أعلاه معاملة خاصة من قبل المحلل. يُحلَّل mygrep() كعامل قائمة حقيقي، ويُحلَّل myrand() كعامل أحادي حقيقي بأسبقية أحادية مماثلة لـ rand()، وmytime() حقًا بدون وسيطات، تمامًا مثل time(). أي، إذا قلت

mytime +2;

فستحصل على "mytime() + 2"، وليس mytime(2)، وهو ما كان سيُحلَّل به بدون نموذج أولي. إذا كنت تريد إجبار دالة أحادية على أن تكون لها نفس أسبقية عامل القائمة، فأضف ";" إلى نهاية النموذج الأولي:

sub mygetprotobynumber($;);
mygetprotobynumber $x > $y; # parsed as mygetprotobynumber($x > $y)

الشيء المثير للاهتمام بشأن "&" هو أنه يمكنك توليد صيغة جديدة به، بشرط أن يكون في الموضع الأولي:

sub try (&@) {
    my($try,$catch) = @_;
    eval { &$try };
    if ($@) {
        local $_ = $@;
        &$catch;
    }
}
sub catch (&) { $_[0] }
try {
    die "phooey";
} catch {
    /phooey/ and print "unphooey\n";
};

هذا يطبع "unphooey". (نعم، لا تزال هناك مسائل غير محلولة تتعلق برؤية @_. أنا أتجاهل ذلك السؤال في الوقت الحالي. (لكن لاحظ أنه إذا جعلنا @_ محددة معجميًا، فيمكن لتلك الإجراءات الفرعية المجهولة أن تعمل كإغلاقات... (يا إلهي، هل يبدو هذا ليبشيًا قليلًا؟ (لا يهم.))))

وإليك إعادة تنفيذ لعامل "grep" في Perl:

sub mygrep (&@) {
    my $code = shift;
    my @result;
    foreach $_ (@_) {
        push(@result, $_) if &$code;
    }
    @result;
}

يفضل بعض الناس نماذج أولية أبجدية رقمية كاملة. لقد تُركت الأحرف الأبجدية الرقمية عمدًا خارج النماذج الأولية لغرض صريح هو إضافة معاملات مسماة ورسمية في المستقبل يومًا ما. الهدف الرئيسي للآلية الحالية هو السماح لكتّاب الوحدات بتقديم تشخيصات أفضل لمستخدمي الوحدات. يرى Larry أن التدوين مفهوم تمامًا لمبرمجي Perl، وأنه لن يتطفل كثيرًا على جوهر الوحدة، ولن يجعل قراءتها أصعب. يتم تغليف ضوضاء السطر بصريًا في حبة صغيرة يسهل ابتلاعها.

إذا حاولت استخدام تسلسل أبجدي رقمي في نموذج أولي، فستولّد تحذيرًا اختياريًا - "Illegal character in prototype...". لسوء الحظ، سمحت الإصدارات الأقدم من Perl باستخدام النموذج الأولي طالما كانت بادئته نموذجًا أوليًا صالحًا. قد يُرقّى التحذير إلى خطأ قاتل في إصدار مستقبلي من Perl بمجرد إصلاح غالبية الكود المخالف.

من المحتمل أن يكون من الأفضل وضع نماذج أولية للدوال الجديدة، وليس إضافة النماذج الأولية بأثر رجعي إلى الدوال الأقدم. وذلك لأنك يجب أن تكون حذرًا بشكل خاص بشأن الفروض الصامتة لسياقات القائمة مقابل القيمة القياسية المختلفة. على سبيل المثال، إذا قررت أن دالة يجب أن تأخذ معاملًا واحدًا فقط، مثل هذا:

sub func ($) {
    my $n = shift;
    print "you gave me $n\n";
}

وكان شخص ما يستدعيها بمصفوفة أو تعبير يعيد قائمة:

func(@foo);
func( $text =~ /\w+/g );

فإنك بذلك قد أضفت تلقائيًا "scalar" أمام وسيطتهم، وهو ما قد يكون مفاجئًا أكثر من قليل. لن يُمرَّر @foo القديم الذي كان يحتوي على شيء واحد. بدلًا من ذلك، ستُمرَّر الآن إلى func() قيمة 1؛ أي عدد العناصر في @foo. وستُستدعى "m//g" في سياق القيمة القياسية، فبدلًا من قائمة كلمات ستعيد نتيجة منطقية وتقدم pos($text). آخ!

إذا كان للإجراء الفرعي كل من PROTO وBLOCK، فلن يُطبَّق النموذج الأولي حتى تُعرَّف BLOCK بالكامل. هذا يعني أن الدالة العودية ذات النموذج الأولي يجب أن تُعلَن مسبقًا لكي يصبح النموذج الأولي ساري المفعول، هكذا:

sub foo($$);
sub foo($$) {
    foo 1, 2;
}

كل هذا قوي جدًا بالطبع، ويجب استخدامه باعتدال فقط لجعل العالم مكانًا أفضل.

الدوال الثابتة

الدوال ذات النموذج الأولي "()" هي مرشحات محتملة للتضمين. إذا كانت النتيجة بعد التحسين وطي الثوابت إما ثابتًا أو قيمة قياسية محددة معجميًا ليس لها مراجع أخرى، فسيُستخدم بدلًا من استدعاءات الدالة المصنوعة بدون "&". الاستدعاءات المصنوعة باستخدام "&" لا تُضمَّن أبدًا. (انظر constant لطريقة سهلة لإعلان معظم الثوابت.)

ستُضمَّن جميع الدوال التالية:

sub pi ()           { 3.14159 }             # Not exact, but close.
sub PI ()           { 4 * atan2 1, 1 }      # As good as it gets,
                                            # and it's inlined, too!
sub ST_DEV ()       { 0 }
sub ST_INO ()       { 1 }
sub FLAG_FOO ()     { 1 << 8 }
sub FLAG_BAR ()     { 1 << 9 }
sub FLAG_MASK ()    { FLAG_FOO | FLAG_BAR }
sub OPT_BAZ ()      { not (0x1B58 & FLAG_MASK) }
sub N () { int(OPT_BAZ) / 3 }
sub FOO_SET () { 1 if FLAG_MASK & FLAG_FOO }
sub FOO_SET2 () { if (FLAG_MASK & FLAG_FOO) { 1 } }

(كن على علم بأن المثال الأخير لم يكن يُضمَّن دائمًا في Perl 5.20 وما قبله، والذي لم يفعل

تتصرف بشكل غير متسق مع الروتينات الفرعية التي تحتوي على نطاقات داخلية.) يمكنك إلغاء التضمين باستخدام "return" صريح:

sub baz_val () {
    if (OPT_BAZ) {
        return 23;
    }
    else {
        return 42;
    }
}
sub bonk_val () { return 12345 }

كما أُشير سابقًا، يمكنك أيضًا تعريف روتينات فرعية مضمّنة ديناميكيًا في وقت BEGIN إذا كان جسمها يتكون من متغير عددي ذي نطاق معجمي لا توجد له مراجع أخرى. سيتم تضمين المثال الأول فقط هنا:

BEGIN {
    my $var = 1;
    no strict 'refs';
    *INLINED = sub () { $var };
}
BEGIN {
    my $var = 1;
    my $ref = \$var;
    no strict 'refs';
    *NOT_INLINED = sub () { $var };
}

من التحذيرات غير الواضحة تمامًا بهذا الشأن (انظر [RT #79908]) ما يحدث إذا كان المتغير قابلًا للتعديل. على سبيل المثال:

BEGIN {
    my $x = 10;
    *FOO = sub () { $x };
    $x++;
}
print FOO(); # طُبع 10 قبل 5.32.0

منذ Perl 5.22 فصاعدًا أعطى هذا تحذيرًا بإهمال، ومنذ Perl 5.32 فصاعدًا أصبح خطأً في وقت التشغيل. سابقًا كان المتغير يُضمَّن فورًا، ويتوقف عن التصرف كمتغير معجمي عادي؛ لذا طُبع 10، وليس 11.

إذا كنت لا تزال تريد تضمين مثل هذا الروتين الفرعي (دون تحذير)، فتأكد من أن المتغير لا يُستخدم في سياق يمكن فيه تعديله خارج مكان تعريفه.

# جيد، لا تحذير
BEGIN {
    my $x = 54321;
    *INLINED = sub () { $x };
}
# خطأ
BEGIN {
    my $x;
    $x = 54321;
    *ALSO_INLINED = sub () { $x };
}

قدّم Perl 5.22 أيضًا السمة "const" كبديل. كانت تجريبية في البداية، لكنها أصبحت مستقرة في Perl 5.40. عند تطبيقها على روتين فرعي مجهول، فإنها تفرض استدعاء الروتين الفرعي عند تقييم تعبير "sub". تُلتقط القيمة المُعادة وتُحوَّل إلى روتين فرعي ثابت:

my $x = 54321;
*INLINED = sub : const { $x };
$x++;

ستكون القيمة المُعادة لـ "INLINED" في هذا المثال دائمًا 54321، بغض النظر عن التعديلات اللاحقة على $x. يمكنك أيضًا وضع أي كود عشوائي داخل الروتين الفرعي، وسيُنفَّذ فورًا وتُلتقط قيمته المُعادة بالطريقة نفسها.

إذا كنت تريد حقًا روتينًا فرعيًا بنموذج أولي "()" يُعيد متغيرًا معجميًا، فيمكنك بسهولة إجباره على عدم التضمين بإضافة "return" صريح:

BEGIN {
    my $x = 10;
    *FOO = sub () { return $x };
    $x++;
}
print FOO(); # يطبع 11

أسهل طريقة لمعرفة ما إذا كان الروتين الفرعي قد ضُمِّن هي استخدام B::Deparse. تأمل هذا المثال لروتينين فرعيين يُعيدان 1، أحدهما بنموذج أولي "()" مما يؤدي إلى تضمينه، والآخر بدونه (مع اقتطاع مخرجات deparse للتوضيح):

$ perl -MO=Deparse -e 'sub ONE { 1 } if (ONE) { print ONE if ONE }'
sub ONE {
    1;
}
if (ONE ) {
    print ONE() if ONE ;
}
$ perl -MO=Deparse -e 'sub ONE () { 1 } if (ONE) { print ONE if ONE }'
sub ONE () { 1 }
do {
    print 1
};

إذا أعدت تعريف روتين فرعي كان مؤهلًا للتضمين، فستحصل على تحذير افتراضيًا. يمكنك استخدام هذا التحذير لمعرفة ما إذا كان روتين فرعي معين يُعتبر قابلًا للتضمين، لأنه يختلف عن التحذير الخاص بتجاوز الروتينات الفرعية غير المضمّنة:

$ perl -e 'sub one () {1} sub one () {2}'
Constant subroutine one redefined at -e line 1.
$ perl -we 'sub one {1} sub one {2}'
Subroutine one redefined at -e line 1.

يُعتبر التحذير جادًا بما يكفي لعدم تأثره بمفتاح -w (أو غيابه) لأن الاستدعاءات المترجمة سابقًا للدالة ستظل تستخدم القيمة القديمة للدالة. إذا كنت بحاجة إلى القدرة على إعادة تعريف الروتين الفرعي، فتحتاج إلى ضمان عدم تضمينه، إما بإسقاط النموذج الأولي "()" (مما يغيّر دلالات الاستدعاء، فاحذر) أو بإحباط آلية التضمين بطريقة أخرى، مثل إضافة "return" صريح، كما ذُكر أعلاه:

sub not_inlined () { return 23 }

تجاوز الدوال المدمجة

يمكن تجاوز العديد من الدوال المدمجة، رغم أنه ينبغي محاولة ذلك فقط أحيانًا ولسبب وجيه. عادةً قد تفعل ذلك حزمة تحاول محاكاة وظائف مدمجة مفقودة على نظام غير يونكس.

لا يمكن إجراء التجاوز إلا باستيراد الاسم من وحدة في وقت الترجمة--الإعلان المسبق العادي لا يكفي. ومع ذلك، تتيح لك pragma "use subs"، فعليًا، الإعلان المسبق عن الروتينات الفرعية عبر صيغة الاستيراد، ويمكن لهذه الأسماء بعد ذلك تجاوز الأسماء المدمجة:

use subs 'chdir', 'chroot', 'chmod', 'chown';
chdir $somewhere;
sub chdir { ... }

للإشارة بشكل لا لبس فيه إلى الصيغة المدمجة، ضع قبل الاسم المدمج المؤهل الخاص بالحزمة "CORE::". على سبيل المثال، قول CORE::open() يشير دائمًا إلى open() المدمجة، حتى لو كانت الحزمة الحالية قد استوردت روتينًا فرعيًا آخر يُسمى &open() من مكان آخر. رغم أنه يبدو كاستدعاء دالة عادي، إلا أنه ليس كذلك: بادئة "CORE::" في تلك الحالة جزء من صيغة Perl، وتعمل مع أي كلمة مفتاحية، بغض النظر عما يوجد في حزمة "CORE". أخذ مرجع إليه، أي "\&CORE::open"، يعمل فقط مع بعض الكلمات المفتاحية. انظر CORE.

لا ينبغي لوحدات المكتبة عمومًا تصدير أسماء مدمجة مثل "open" أو "chdir" كجزء من قائمة @EXPORT الافتراضية، لأنها قد تتسلل إلى مساحة أسماء شخص آخر وتغيّر الدلالات بشكل غير متوقع. بدلًا من ذلك، إذا أضافت الوحدة ذلك الاسم إلى @EXPORT_OK، فسيصبح من الممكن للمستخدم استيراد الاسم صراحةً، وليس ضمنيًا. أي يمكنهم قول

use Module 'open';

وسيستورد تجاوز "open". لكن إذا قالوا

use Module;

فسيحصلون على الاستيرادات الافتراضية دون تجاوزات.

الآلية السابقة لتجاوز المدمجات مقيدة، عن قصد تام، بالحزمة التي تطلب الاستيراد. هناك طريقة ثانية تنطبق أحيانًا عندما ترغب في تجاوز مدمج في كل مكان، دون اعتبار لحدود مساحة الأسماء. يتحقق ذلك باستيراد روتين فرعي إلى مساحة الأسماء الخاصة "CORE::GLOBAL::". إليك مثال يستبدل بجرأة تامة عامل "glob" بشيء يفهم التعبيرات النمطية.

package REGlob;
require Exporter;
@ISA = 'Exporter';
@EXPORT_OK = 'glob';
sub import {
    my $pkg = shift;
    return unless @_;
    my $sym = shift;
    my $where = ($sym =~ s/^GLOBAL_// ? 'CORE::GLOBAL' : caller(0));
    $pkg->export($where, $sym, @_);
}
sub glob {
    my $pat = shift;
    my @got;
    if (opendir my $d, '.') {
        @got = grep /$pat/, readdir $d;
        closedir $d;
    }
    return @got;
}
1;

وإليك كيف يمكن إساءة استخدامه:

#use REGlob 'GLOBAL_glob';      # تجاوز glob() في جميع مساحات الأسماء
package Foo;
use REGlob 'glob';              # تجاوز glob() في Foo:: فقط
print for <^[a-z_]+\.pm\$>;     # عرض جميع الوحدات البراغماتية

يُظهر التعليق الأول مثالًا مصطنعًا، بل خطيرًا. بتجاوز "glob" عالميًا، ستفرض السلوك الجديد (والتخريبي) لعامل "glob" على كل مساحة أسماء، دون علم كامل أو تعاون الوحدات التي تملك تلك المساحات. بطبيعة الحال، ينبغي فعل ذلك بحذر شديد--إن كان لا بد من فعله أصلًا.

مثال "REGlob" أعلاه لا ينفذ كل الدعم اللازم لتجاوز عامل "glob" في Perl بشكل نظيف. للمدمج "glob" سلوكيات مختلفة حسب ظهوره في سياق عددي أو قائمة، لكن "REGlob" الخاص بنا لا يفعل ذلك. في الواقع، العديد من مدمجات Perl لها سلوكيات حساسة للسياق، ويجب دعمها بشكل كافٍ بواسطة تجاوز مكتوب بشكل صحيح. للحصول على مثال كامل الوظائف لتجاوز "glob"، ادرس تنفيذ "File::DosGlob" في المكتبة القياسية.

عندما تتجاوز مدمجًا، يجب أن يكون بديلك متسقًا (إن أمكن) مع الصيغة الأصلية المدمجة. يمكنك تحقيق ذلك باستخدام نموذج أولي مناسب. للحصول على النموذج الأولي لمدمج قابل للتجاوز، استخدم الدالة "prototype" بوسيط "CORE::builtin_name" (انظر "prototype" في perlfunc).

لاحظ مع ذلك أن بعض المدمجات لا يمكن التعبير عن صيغتها بنموذج أولي (مثل "system" أو "chomp"). إذا تجاوزتها فلن تتمكن من محاكاة صيغتها الأصلية بالكامل.

يمكن أيضًا تجاوز المدمجات "do" و"require" و"glob"، لكن بسبب سحر خاص، تُحفظ صيغتها الأصلية، ولا يلزمك تعريف نموذج أولي لبدائلها. (لا يمكنك تجاوز صيغة "do BLOCK"، مع ذلك).

لـ "require" سحر إضافي خاص مظلم: إذا استدعيت بديل "require" الخاص بك كـ "require Foo::Bar"، فسيستقبل فعليًا الوسيط "Foo/Bar.pm" في @_. انظر "require" في perlfunc.

وكما لاحظت من المثال السابق، إذا تجاوزت "glob"، فسيتم تجاوز عامل glob "<*>" أيضًا.

بطريقة مشابهة، تجاوز الدالة "readline" يتجاوز أيضًا عامل الإدخال/الإخراج المكافئ "<FILEHANDLE>". كذلك، تجاوز "readpipe" يتجاوز أيضًا العاملين `` و"qx//".

أخيرًا، بعض المدمجات (مثل "exists" أو "grep") لا يمكن تجاوزها.

التحميل التلقائي

إذا ك

إذا قمت باستدعاء روتين فرعي غير معرّف، فعادةً ستحصل على خطأ فوري وقاتل يشتكي من أن الروتين الفرعي غير موجود. (وبالمثل بالنسبة للروتينات الفرعية المستخدمة كطرق، عندما لا تكون الطريقة موجودة في أي صنف أساسي من حزمة الصنف.) ومع ذلك، إذا تم تعريف روتين فرعي "AUTOLOAD" في الحزمة أو الحزم المستخدمة لتحديد موقع الروتين الفرعي الأصلي، فإن روتين "AUTOLOAD" الفرعي هذا يُستدعى مع الوسائط التي كان سيتم تمريرها إلى الروتين الفرعي الأصلي. يظهر الاسم المؤهل بالكامل للروتين الفرعي الأصلي بشكل سحري في المتغير العام $AUTOLOAD لنفس الحزمة التي يوجد بها روتين "AUTOLOAD". لا يُمرَّر الاسم كوسيط عادي لأنه، حسنًا، فقط لأنه كذلك. (كاستثناء، يتم تخطي استدعاء طريقة لطريقة "import" أو "unimport" غير موجودة فقط. أيضًا، إذا كان روتين AUTOLOAD الفرعي هو XSUB، فهناك طرق أخرى لاسترداد اسم الروتين الفرعي. راجع "Autoloading with XSUBs" في perlguts للتفاصيل.)

تحمّل العديد من روتينات "AUTOLOAD" تعريفًا للروتين الفرعي المطلوب باستخدام eval()، ثم تنفّذ ذلك الروتين الفرعي باستخدام صيغة خاصة من goto() تمحو إطار المكدس لروتين "AUTOLOAD" دون أثر. (راجع المصدر الخاص بالوحدة القياسية الموثقة في AutoLoader، على سبيل المثال.) لكن روتين "AUTOLOAD" يمكنه أيضًا محاكاة الروتين فقط دون تعريفه أبدًا. على سبيل المثال، لنفترض أن دالة لم تكن معرّفة يجب أن تستدعي فقط "system" بتلك الوسائط. كل ما عليك فعله هو:

sub AUTOLOAD {
    our $AUTOLOAD;              # keep 'use strict' happy
    my $program = $AUTOLOAD;
    $program =~ s/.*:://;
    system($program, @_);
}
date();
who();
ls('-l');

في الواقع، إذا قمت بالإعلان المسبق عن الدوال التي تريد استدعاءها بهذه الطريقة، فلن تحتاج حتى إلى أقواس:

use subs qw(date who ls);
date;
who;
ls '-l';

مثال أكثر اكتمالاً على ذلك هو وحدة Shell على CPAN، والتي يمكنها التعامل مع استدعاءات الروتينات الفرعية غير المعرّفة كاستدعاءات لبرامج خارجية.

تتوفر آليات لمساعدة كتّاب الوحدات على تقسيم وحداتهم إلى ملفات قابلة للتحميل التلقائي. راجع الوحدة القياسية AutoLoader الموثقة في AutoLoader وفي AutoSplit، ووحدات SelfLoader القياسية في SelfLoader، والوثيقة الخاصة بإضافة دوال C إلى كود Perl في perlxs.

سمات الروتين الفرعي

قد يكون لإعلان أو تعريف روتين فرعي قائمة من السمات المرتبطة به. إذا كانت قائمة السمات هذه موجودة، فإنها تُقسَّم عند حدود المسافات أو النقطتين الرأسيتين وتُعامَل كما لو أنه قد تمت رؤية "use attributes". راجع attributes للتفاصيل حول السمات المدعومة حاليًا. على عكس القيد الخاص بـ "use attrs" المهجور، فإن صيغة "sub : ATTRLIST" تعمل على ربط السمات بإعلان مسبق، وليس فقط بتعريف روتين فرعي.

يجب أن تكون السمات صالحة كأسماء معرّفات بسيطة (بدون أي علامات ترقيم غير حرف '_'). قد تكون لديها قائمة معاملات ملحقة، والتي يتم التحقق منها فقط من حيث ما إذا كانت أقواسها ('(',')') متداخلة بشكل صحيح.

أمثلة على الصيغة الصالحة (رغم أن السمات غير معروفة):

sub fnord (&\%) : switch(10,foo(7,3))  :  expensive;
sub plugh () : Ugly('\(") :Bad;
sub xyzzy : _5x5 { ... }

أمثلة على الصيغة غير الصالحة:

sub fnord : switch(10,foo();    # ()-string not balanced
sub snoid : Ugly('(');          # ()-string not balanced
sub xyzzy : 5x5;                # "5x5" not a valid identifier
sub plugh : Y2::north;          # "Y2::north" not a simple identifier
sub snurt : foo + bar;          # "+" not a colon or space

تُمرَّر قائمة السمات كقائمة من السلاسل النصية الثابتة إلى الكود الذي يربطها بالروتين الفرعي. على وجه الخصوص، يبدو المثال الثاني للصيغة الصالحة أعلاه حاليًا هكذا من حيث كيفية تحليله واستدعائه:

use attributes __PACKAGE__, \&plugh, q[Ugly('\(")], 'Bad';

لمزيد من التفاصيل حول قوائم السمات ومعالجتها، راجع attributes و Attribute::Handlers.

انظر أيضًا

راجع "Function Templates" في perlref لمزيد من المعلومات حول المراجع والإغلاقات. راجع perlxs إذا كنت ترغب في معرفة المزيد حول استدعاء روتينات C الفرعية من Perl. راجع perlembed إذا كنت ترغب في معرفة المزيد حول استدعاء روتينات Perl الفرعية من C. راجع perlmod لمعرفة المزيد حول تجميع دوالك في ملفات منفصلة. راجع perlmodlib لمعرفة وحدات المكتبة التي تأتي قياسيًا على نظامك. راجع perlootut لمعرفة كيفية إجراء استدعاءات طرق الكائنات.
