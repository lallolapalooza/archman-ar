الوصف

بالنسبة لوكلاء نطاق مقاييس الأداء (PMDA) (PMDA(3)) التي تستخدم بروتوكولات PDU الثنائية للتواصل مع pmcd(1)، توفر الدالة pmdaMain تطبيقًا عامًا لحلقة المعالجة الرئيسية المدفوعة بواسطة PDU.

dispatch يصف كيفية معالجة كل PDU وارد. وهو متجه من مؤشرات الدوال، واحد لكل نوع طلب PDU، كما هو مستخدم في واجهة DSO لوكيل PMDA، وهو:

/*
 * تعريفات الواجهة لطرق PMDA
 */
typedef struct {
    int domain;         /* تعيين/إرجاع معرف نطاق مقاييس الأداء هنا */
    struct {
        unsigned int    pmda_interface: 8; /* إصدار واجهة PMDA DSO */
        unsigned int    pmapi_version : 8; /* إصدار PMAPI */
        unsigned int    flags : 16;        /* علامات ميزات اختيارية */
    } comm;             /* تعيين/إرجاع معلومات الاتصال والإصدار */
    int status;         /* تعيين/إرجاع حالة التهيئة هنا */
    union {
        struct {                              /* PMDA_INTERFACE_2 or _3 */
            pmdaExt *ext;
            int (*profile)(pmProfile *, pmdaExt *);
            int (*fetch)(int, pmID *, pmResult **, pmdaExt *);
            int (*desc)(pmID, pmDesc *, pmdaExt *);
            int (*instance)(pmInDom, int, char *, pmInResult **, pmdaExt *);
            int (*text)(int, int, char **, pmdaExt *);
            int (*store)(pmResult *, pmdaExt *);
        } two, three;
        struct {                              /* PMDA_INTERFACE_4 or _5 */
            pmdaExt *ext;
            int     (*profile)(pmProfile *, pmdaExt *);
            int     (*fetch)(int, pmID *, pmResult **, pmdaExt *);
            int     (*desc)(pmID, pmDesc *, pmdaExt *);
            int     (*instance)(pmInDom, int, char *, pmInResult **, pmdaExt *);
            int     (*text)(int, int, char **, pmdaExt *);
            int     (*store)(pmResult *, pmdaExt *);
            int     (*pmid)(char *, pmID *, pmdaExt *);
            int     (*name)(pmID, char ***, pmdaExt *);
            int     (*children)(char *, int, char ***, int **, pmdaExt *);
        } four, five;
        struct {                              /* PMDA_INTERFACE_6 */
            pmdaExt *ext;
            int     (*profile)(pmProfile *, pmdaExt *);
            int     (*fetch)(int, pmID *, pmResult **, pmdaExt *);
            int     (*desc)(pmID, pmDesc *, pmdaExt *);
            int     (*instance)(pmInDom, int, char *, pmInResult **, pmdaExt *);
            int     (*text)(int, int, char **, pmdaExt *);
            int     (*store)(pmResult *, pmdaExt *);
            int     (*pmid)(char *, pmID *, pmdaExt *);
            int     (*name)(pmID, char ***, pmdaExt *);
            int     (*children)(char *, int, char ***, int **, pmdaExt *);
            int     (*attribute)(int, int, const char *, int, pmdaExt *);
        } six;
    } version;
} pmdaInterface;

تم توسيع هذا الهيكل لتضمين إصدارات الواجهة المتعددة التي تطورت مع مرور الوقت. بالنسبة لـ pmdaMain، يتم تجاهل dispatch->domain و dispatch->status. يتم استخدام الحقل comm.pmda_interface لتحديد الواجهة التي يستخدمها PMDA. سيؤدي تعيين هذا الحقل إلى PMDA_INTERFACE_2 أو PMDA_INTERFACE_3 إلى إجبار pmdaMain على استخدام استدعاءات الرد في الهيكل version.two أو version.three. سيؤدي تعيين PMDA_INTERFACE_4 أو PMDA_INTERFACE_5 إلى إجبار pmdaMain على استخدام استدعاءات الرد في الهيكل version.four أو version.five، وبالمثل سيؤدي تعيين PMDA_INTERFACE_6 إلى إجبار pmdaMain على استخدام استدعاءات الرد في الهيكل version.six. أي قيمة أخرى ستؤدي إلى حدوث خطأ وإنهاء pmdaMain.

لاحظ أن استخدام dispatch كواجهة بين pmcd(1) وطرق PMDA يسمح بتنفيذ كل PMDA كما لو كان DSO، مع توفير pmdaMain لغلاف مريح يمكن استخدامه للتحويل من واجهة DSO إلى واجهة PDU الثنائية (daemon PMDA).

pmdaMain تعمل كحلقة مستمرة، ولا تعود إلا عند مواجهة نهاية الملف في واصف ملف إدخال PDU.

استدعاءات الرد

بالإضافة إلى استدعاءات معالجة PDU الفردية - pmdaProfile(3)، pmdaFetch(3)، pmdaDesc(3)، pmdaInstance(3)، pmdaText(3)، pmdaStore(3)، pmdaPMID(3)، pmdaName(3)، pmdaChildren(3)، و pmdaAttribute(3)، هناك استدعاءات رد أخرى يمكن أن تؤثر أو تعلم جميع معالجة PDU داخل PMDA، وهي check و done و end. يجب تعيين استدعاءات الرد هذه باستخدام pmdaSetCheckCallBack و pmdaSetDoneCallBack و pmdaSetEndContextCallBack.

إذا لم يكن check فارغًا، يتم استدعاؤه بعد استلام كل PDU (ولكن قبل معالجتها)، ويتم استدعاء done بعد إرسال كل PDU. إذا أعاد check قيمة أقل من الصفر (عادةً PM_ERR_AGAIN)، يتم تخطي معالجة PDU وفي معظم الحالات يتم إرجاع قيمة الدالة كـ PDU خطأ إلى pmcd(1) - يمكن استخدام هذا لوكلاء PMDA التي تتطلب نوعًا من بروتوكولات الاتصال المؤجل أو إعادة الاتصال للمصادر الأساسية لمقاييس الأداء، على سبيل المثال قاعدة بيانات علائقية (DBMS). لا يتم تمرير مؤشر الخطأ من check مرة أخرى إلى pmcd(1) في الحالات التي لا يكون فيها أي إقرار متوقعًا، على سبيل المثال لـ PDU_PROFILE.

يسمح استدعاء الرد end لوكيل PMDA بتتبع الحالة للعملاء الفرديين الذين يطلبون منه أداء إجراءات (معالجة PDU). باستخدام pmdaGetContext، يمكن لوكيل PMDA تحديد، في أي نقطة، معرف رقمي يحدد بشكل فريد أدوات العميل في الطرف البعيد لـ PMCD (في أوضاع السياق المحلي، يكون هذا المعرف دائمًا صفرًا). يصبح هذا مهمًا جدًا لمعالجة مقاييس الأحداث، حيث يجب نشر كل حدث مرة واحدة فقط لكل عميل مهتم. كما أنه يشكل أساس الآلية التي يتم بها تمرير معلومات الاتصال إلى وكيل PMDA، مثل بيانات الاعتماد (معرفات المستخدم والمجموعة) لأداة العميل.

يتم توفير آلية استدعاء رد أخيرة لمعالجة pmResult الذي تم بناؤه لـ PDU_RESULT استجابةً لطلب PDU_FETCH. افتراضيًا، سيقوم pmdaMain بتحرير pmResult بمجرد إرسال النتيجة إلى pmcd(1). بالنسبة لبعض وكلاء PMDA، هذا غير مناسب، على سبيل المثال، يتم تخصيص pmResult بشكل ثابت، أو يحتوي على مزيج من معلومات المخزن المؤقت PDU المثبتة ومعلومات مخصصة ديناميكيًا. يمكن استخدام pmdaSetResultCallBack لتحديد استدعاء رد بديل من pmdaMain.

التشخيص

قد تُلحق هذه الرسائل بملف سجل PMDA:

إصدار واجهة PMDA interface غير مدعوم
الإصدار interface غير مدعوم بواسطة pmdaMain.
نوع pdu غير معروف
استقبلت PMDA PDU من pmcd لا تتعرف عليه. قد يشير هذا إلى أن عملية pmcd تستخدم واجهة أكثر تقدمًا من pmdaMain.

إذا كانت خيارات التحكم في تصحيح الأخطاء الخاصة بـ PMAPI(3) مضبوطة على الخيار ``libpmda''، فسيتم الإبلاغ عن كل PDU يتم استقباله في ملف سجل PMDA.
