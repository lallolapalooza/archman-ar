الاسم

rpc - إجراءات مكتبية لاستدعاءات الإجراءات عن بعد

المكتبة

مكتبة C القياسية (libc، -lc)

الملخص والوصف

تسمح هذه الإجراءات لبرامج C بإجراء استدعاءات إجراءات على أجهزة أخرى عبر الشبكة. أولاً، يستدعي العميل إجراءً لإرسال حزمة بيانات إلى الخادم. عند استلام الحزمة، يستدعي الخادم إجراءً توزيعيًا لتنفيذ الخدمة المطلوبة، ثم يرسل ردًا. أخيرًا، تعود استدعاء الإجراء إلى العميل.

لاستخدام هذه الإجراءات، قم بتضمين ملف الرأس <rpc/rpc.h>.

تستخدم النماذج الأولية أدناه الأنواع التالية:

typedef int bool_t;
typedef typeof(bool_t (XDR *, void *, ...))  *xdrproc_t;
typedef typeof(bool_t (caddr_t resp, struct sockaddr_in *raddr)
    *resultproc_t;

ينشئ ويعيد مقبض مصادقة RPC يحتوي على معلومات المصادقة. المعلمة host هي اسم الجهاز الذي تم إنشاء المعلومات عليه؛ uid هو معرف المستخدم؛ gid هو معرف المجموعة الحالي للمستخدم؛ n و aup_gids تشير إلى مصفوفة معدودة للمجموعات التي ينتمي إليها المستخدم. من السهل انتحال هوية مستخدم.

يستدعي authunix_create() بالمعلمات المناسبة.
int callrpc(char *host, unsigned long prognum,
            unsigned long versnum, unsigned long procnum,
            xdrproc_t inproc, const char *in,
            xdrproc_t outproc, char *out);
يستدعي الإجراء البعيد المرتبط بـ prognum، versnum، و procnum على الجهاز host. المعلمة in هي عنوان وسيط (وسائط) الإجراء، و out هي العنوان الذي سيتم فيه وضع النتائج؛ inproc يستخدم لتشفير معاملات الإجراء، و outproc يستخدم لفك تشفير نتائج الإجراء. تعيد هذه الدالة الصفر إذا نجحت، أو قيمة enum clnt_stat محولة إلى عدد صحيح إذا فشلت. الدالة clnt_perrno() مفيدة لترجمة حالات الفشل إلى رسائل.

مثل callrpc()، إلا أن رسالة الاستدعاء يتم بثها إلى جميع شبكات البث المتصلة محليًا. في كل مرة تتلقى فيها ردًا، تستدعي هذه الدالة eachresult()، والتي يكون شكلها:

حيث out هو نفس out الذي تم تمريره إلى clnt_broadcast()، باستثناء أن ناتج الإجراء البعيد يتم فك تشفيره هناك؛ addr يشير إلى عنوان الجهاز الذي أرسل النتائج. إذا أعادت eachresult() صفرًا، تنتظر clnt_broadcast() المزيد من الردود؛ وإلا فإنها تعود بالحالة المناسبة.

clnt_destroy(CLIENT *clnt);
ماكرو يدمر مقبض RPC الخاص بالعميل. عادةً ما يتضمن التدمير إلغاء تخصيص هياكل البيانات الخاصة، بما في ذلك clnt نفسه. استخدام clnt غير معرف بعد استدعاء clnt_destroy(). إذا فتحت مكتبة RPC المقبس المرتبط، فستقوم بإغلاقه أيضًا. بخلاف ذلك، يظل المقبس مفتوحًا.
CLIENT *clnt_create(const char *host, unsigned long prog,

روتين عام لإنشاء عميل. host يحدد اسم المضيف البعيد حيث يوجد الخادم. proto يشير إلى نوع بروتوكول النقل الذي سيتم استخدامه. القيم المدعومة حاليًا لهذا الحقل هي "udp" و "tcp". يتم تعيين مهلات افتراضية، ولكن يمكن تعديلها باستخدام clnt_control().
تحذير: استخدام UDP له قصوره. نظرًا لأن رسائل RPC المستندة إلى UDP يمكن أن تحمل 8 كيلوبايت فقط من البيانات المشفرة، فلا يمكن استخدام هذا البروتوكول للإجراءات التي تتلقى معاملات كبيرة أو تعيد نتائج ضخمة.

ماكرو يستخدم لتغيير أو استرداد معلومات مختلفة حول كائن العميل. req يشير إلى نوع العملية، و info هو مؤشر إلى المعلومات. لكل من UDP و TCP، القيم المدعومة لـ req وأنواع وسائطها وماذا تفعل هي:

CLSET_TIMEOUT  struct timeval // تعيين المهلة الإجمالية
CLGET_TIMEOUT  struct timeval // استرداد المهلة الإجمالية

تأخذ نفس وسائط clnt_perrno()، ولكن بدلاً من إرسال رسالة إلى الخطأ القياسي تشير إلى سبب فشل استدعاء RPC، تعيد مؤشرًا إلى سلسلة تحتوي على الرسالة. تنتهي السلسلة بسطر جديد.

CLIENT *clntraw_create(unsigned long prognum, unsigned long versnum);
تقوم هذه الدالة بإنشاء عميل RPC وهمي للبرنامج البعيد prognum، الإصدار versnum. النقل المستخدم لتمرير الرسائل إلى الخدمة هو في الواقع مخزن مؤقت داخل مساحة عنوان العملية، لذلك يجب أن يعيش خادم RPC المقابل في نفس مساحة العنوان؛ انظر svcraw_create(). هذا يسمح بمحاكاة RPC واكتساب تكاليف RPC، مثل أوقات الرحلة ذهابًا وإيابًا، دون أي تدخل من النواة. تعيد هذه الدالة NULL إذا فشلت.
CLIENT *clnttcp_create(struct sockaddr_in *addr,
                unsigned long prognum, unsigned long versnum,
                int *sockp, unsigned int sendsz, unsigned int recvsz);
تقوم هذه الدالة بإنشاء عميل RPC للبرنامج البعيد prognum، الإصدار versnum؛ يستخدم العميل TCP/IP كنقل. يقع البرنامج البعيد في عنوان الإنترنت *addr. إذا كان addr->sin_port يساوي صفرًا، فسيتم تعيينه إلى المنفذ الفعلي الذي يستمع إليه البرنامج البعيد (يتم استشارة خدمة portmap البعيدة لهذه المعلومات). المعلمة sockp هي مقبس؛ إذا كانت RPC_ANYSOCK، فإن هذه الدالة تفتح مقبسًا جديدًا وتعيّن sockp. نظرًا لأن RPC المستند إلى TCP يستخدم إدخال/إخراج مخزن مؤقت، يمكن للمستخدم تحديد حجم مخازن الإرسال والاستقبال مع المعلمات sendsz وrecvsz؛ القيم الصفرية تختار القيم الافتراضية المناسبة. تعيد هذه الدالة NULL إذا فشلت.

متغير عام يتم تعيين قيمته بواسطة أي روتين إنشاء عميل RPC لا ينجح. استخدم الروتين clnt_pcreateerror() لطباعة سبب ذلك.
void svc_destroy(SVCXPRT *xprt);
ماكرو يدمر مقبض نقل خدمة RPC، xprt. عادة ما يتضمن التدمير تخصيص هياكل بيانات خاصة، بما في ذلك xprt نفسه. استخدام xprt غير محدد بعد استدعاء هذا الروتين.
fd_set svc_fdset;
متغير عام يعكس قناع بت واصف الملف للقراءة الخاص بجانب خدمة RPC؛ إنه مناسب كمعلمة لاستدعاء نظام select(2). هذا يهم فقط إذا كان منفذ الخدمة يقوم بمعالجة الأحداث غير المتزامنة الخاصة به، بدلاً من استدعاء svc_run(). هذا المتغير للقراءة فقط (لا تمرر عنوانه إلى select(2)!)، ومع ذلك قد يتغير بعد استدعاء svc_getreqset() أو أي روتينات إنشاء.
int svc_fds;
مشابه لـ svc_fdset، ولكنه محدود إلى 32 واصف ملف. تم إهمال هذه الواجهة بواسطة svc_fdset.
svc_freeargs(SVCXPRT *xprt, xdrproc_t inproc, char *in);
ماكرو يحرر أي بيانات تم تخصيصها بواسطة نظام RPC/XDR عند فك تشفير الوسائط لإجراء خدمة باستخدام svc_getargs(). يعيد هذا الروتين 1 إذا تم تحرير النتائج بنجاح، وصفر بخلاف ذلك.
svc_getargs(SVCXPRT *xprt, xdrproc_t inproc, char *in);
ماكرو يقوم بفك تشفير وسائط طلب RPC المرتبط بمقبض نقل خدمة RPC، xprt. المعلمة in هي العنوان الذي سيتم وضع الوسائط فيه؛ inproc هو روتين XDR المستخدم لفك تشفير الوسائط. يعيد هذا الروتين واحدًا إذا نجح فك التشفير، وصفر بخلاف ذلك.
struct sockaddr_in *svc_getcaller(SVCXPRT *xprt);
الطريقة المعتمدة للحصول على عنوان الشبكة للمتصل بإجراء مرتبط بمقبض نقل خدمة RPC، xprt.
void svc_getreqset(fd_set *rdfds);
هذا الروتين يهم فقط إذا لم يستدعِ منفذ الخدمة svc_run()، ولكنه بدلاً من ذلك ينفذ معالجة أحداث غير متزامنة مخصصة. يتم استدعاؤه عندما تحدد استدعاء نظام select(2) أن طلب RPC قد وصل على بعض مقابس RPC؛ rdfds هو قناع بت واصف الملف للقراءة الناتج. يعيد الروتين عندما تتم خدمة جميع المقابس المرتبطة بقيمة rdfds.
void svc_getreq(int rdfds);
مشابه لـ svc_getreqset()، ولكنه محدود إلى 32 واصف ملف. تم إهمال هذه الواجهة بواسطة svc_getreqset().
bool_t svc_register(SVCXPRT *xprt, unsigned long prognum,
                    unsigned long versnum,
                    typeof(void (struct svc_req *, SVCXPRT *)) *dispatch,
                    unsigned long protocol);
يربط prognum و versnum بإجراء توزيع الخدمة، dispatch. إذا كان protocol صفرًا، فلن يتم تسجيل الخدمة مع خدمة portmap. إذا كان protocol غير صفر، فسيتم إنشاء رسم خرائط للثلاثي [prognum,versnum,protocol] إلى xprt->xp_port مع خدمة portmap المحلية (بشكل عام protocol هو صفر، IPPROTO_UDP أو IPPROTO_TCP). يكون للإجراء dispatch الشكل التالي:

dispatch(struct svc_req *request, SVCXPRT *xprt);

يعيد الروتين svc_register() واحدًا إذا نجح، وصفر بخلاف ذلك.
void svc_run(void);
هذا الروتين لا يعود أبداً. ينتظر وصول طلبات RPC، ويستدعي إجراء الخدمة المناسب باستخدام svc_getreq() عند وصول أحدها. هذا الإجراء ينتظر عادةً حتى تعود استدعاء نظام select(2).
bool_t svc_sendreply(SVCXPRT *xprt, xdrproc_t outproc, char *out);
يتم استدعاؤه بواسطة روتين توزيع خدمة RPC لإرسال نتائج استدعاء إجراء بعيد. المعلمة xprt هي مقبض النقل المرتبط بالطلب؛ outproc هو روتين XDR المستخدم لترميز النتائج؛ و out هو عنوان النتائج. يعيد هذا الروتين واحدًا إذا نجح، وصفر بخلاف ذلك.
void svc_unregister(unsigned long prognum, unsigned long versnum);
إزالة جميع رسوم البيانات المزدوجة [prognum,versnum] إلى روتينات التوزيع، ورسم البيانات الثلاثية [prognum,versnum,*] إلى رقم المنفذ.
void svcerr_auth(SVCXPRT *xprt, enum auth_stat why);
يتم استدعاؤه بواسطة روتين توزيع خدمة يرفض تنفيذ استدعاء إجراء بعيد بسبب خطأ في المصادقة.
void svcerr_decode(SVCXPRT *xprt);
يتم استدعاؤه بواسطة روتين توزيع خدمة لا يمكنه فك تشفير وسائطه بنجاح. انظر أيضًا svc_getargs().
void svcerr_noproc(SVCXPRT *xprt);
يتم استدعاؤه بواسطة روتين توزيع خدمة لا ينفذ رقم الإجراء الذي يطلبه المتصل.
void svcerr_noprog(SVCXPRT *xprt);
يتم استدعاؤه عندما لا يكون البرنامج المطلوب مسجلاً في حزمة RPC. عادة لا يحتاج منفذو الخدمة إلى هذا الروتين.
void svcerr_progvers(SVCXPRT *xprt, unsigned long low_vers,
                     unsigned long high_vers);
يتم استدعاؤه عندما لا يكون الإصدار المطلوب من برنامج مسجلاً في حزمة RPC. عادة لا يحتاج منفذو الخدمة إلى هذا الروتين.
void svcerr_systemerr(SVCXPRT *xprt);
يتم استدعاؤه بواسطة روتين توزيع خدمة عندما يكتشف خطأ في النظام لا تغطيه بروتوكول معين. على سبيل المثال، إذا لم يعد بإمكان الخدمة تخصيص تخزين، فقد تستدعي هذا الروتين.
void svcerr_weakauth(SVCXPRT *xprt);
يتم استدعاؤه بواسطة روتين توزيع خدمة يرفض تنفيذ استدعاء إجراء بعيد بسبب عدم كفاية معلمات المصادقة. يستدعي الروتين svcerr_auth(xprt, AUTH_TOOWEAK).
SVCXPRT *svcfd_create(int fd, unsigned int sendsize,
                      unsigned int recvsize);
إنشاء خدمة فوق أي واصف ملف مفتوح. عادة، يكون واصف الملف هذا مقبسًا متصلاً لبروتوكول تدفق مثل TCP. تشير sendsize و recvsize إلى أحجام مخازن الإرسال والاستقبال. إذا كانت صفرًا، فسيتم اختيار افتراضي معقول.
SVCXPRT *svcraw_create(void);
ينشئ هذا الروتين نقل خدمة RPC وهمي، والذي يعيد مؤشرًا إليه. النقل هو في الواقع مخزن مؤقت داخل مساحة عنوان العملية، لذا يجب أن يعيش عميل RPC المقابل في نفس مساحة العنوان؛ انظر clntraw_create(). يسمح هذا الروتين بمحاكاة RPC واكتساب النفقات العامة لـ RPC (مثل أوقات الرحلة ذهابًا وإيابًا)، دون أي تدخل من النواة. يعيد هذا الروتين NULL إذا فشل.
SVCXPRT *svctcp_create(int sock, unsigned int send_buf_size,
                       unsigned int recv_buf_size);
ينشئ هذا الروتين نقل خدمة RPC قائم على TCP/IP، والذي يعيد مؤشرًا إليه. يرتبط النقل بالمقبس sock، والذي قد يكون RPC_ANYSOCK، وفي هذه الحالة يتم إنشاء مقبس جديد. إذا لم يكن المقبس مرتبطًا بمنفذ TCP محلي، فإن هذا الروتين يربطه بمنفذ عشوائي. عند اكتمال العملية، يكون xprt->xp_sock هو واصف مقبس النقل، و xprt->xp_port هو رقم منفذ النقل. يعيد هذا الروتين NULL إذا فشل. نظرًا لأن RPC المستند إلى TCP يستخدم الإدخال/الإخراج المخزن مؤقتًا، يمكن للمستخدمين تحديد حجم المخازن المؤقتة؛ تختار القيم الصفرية افتراضيات مناسبة.
SVCXPRT *svcudp_bufcreate(int sock, unsigned int sendsize,
                          unsigned int recosize);
ينشئ هذا الروتين نقل خدمة RPC قائم على UDP/IP، والذي يعيد مؤشرًا إليه. يرتبط النقل بالمقبس sock، والذي قد يكون RPC_ANYSOCK، وفي هذه الحالة يتم إنشاء مقبس جديد. إذا لم يكن المقبس مرتبطًا بمنفذ UDP محلي، فإن هذا الروتين يربطه بمنفذ عشوائي. عند اكتمال العملية، يكون xprt->xp_sock هو واصف مقبس النقل، و xprt->xp_port هو رقم منفذ النقل. يعيد هذا الروتين NULL إذا فشل.
هذا يسمح للمستخدم بتحديد الحد الأقصى لحجم الحزمة لرسائل RPC المستندة إلى UDP المرسلة والمستقبلة.
SVCXPRT *svcudp_create(int sock);
هذا الاستدعاء يعادل svcudp_bufcreate(sock,SZ,SZ) لبعض الحجم الافتراضي SZ.
bool_t xdr_accepted_reply(XDR *xdrs, struct accepted_reply *ar);
يستخدم لترميز رسائل رد RPC. هذا الروتين مفيد للمستخدمين الذين يرغبون في إنشاء رسائل على نمط RPC دون استخدام حزمة RPC.
bool_t xdr_authunix_parms(XDR *xdrs, struct authunix_parms *aupp);
يستخدم لوصف بيانات اعتماد UNIX. هذا الروتين مفيد للمستخدمين الذين يرغبون في إنشاء بيانات الاعتماد هذه دون استخدام حزمة مصادقة RPC.
void xdr_callhdr(XDR *xdrs, struct rpc_msg *chdr);
يستخدم لوصف رسائل رأس استدعاء RPC. هذا الروتين مفيد للمستخدمين الذين يرغبون في إنشاء رسائل على نمط RPC دون استخدام حزمة RPC.
bool_t xdr_callmsg(XDR *xdrs, struct rpc_msg *cmsg);
يُستخدم لوصف رسائل استدعاء RPC. هذه الروتين مفيدة للمستخدمين الذين يرغبون في إنشاء رسائل بأسلوب RPC دون استخدام حزمة RPC.
bool_t xdr_opaque_auth(XDR *xdrs, struct opaque_auth *ap);
يُستخدم لوصف رسائل معلومات مصادقة RPC. هذه الروتين مفيدة للمستخدمين الذين يرغبون في إنشاء رسائل بأسلوب RPC دون استخدام حزمة RPC.
bool_t xdr_pmap(XDR *xdrs, struct pmap *regs);
يُستخدم لوصف المعلمات للإجراءات المختلفة لـ portmap، خارجيًا. هذه الروتين مفيدة للمستخدمين الذين يرغبون في إنشاء هذه المعلمات دون استخدام واجهة pmap.
bool_t xdr_pmaplist(XDR *xdrs, struct pmaplist **rp);
يُستخدم لوصف قائمة تعيينات المنافذ، خارجيًا. هذه الروتين مفيدة للمستخدمين الذين يرغبون في إنشاء هذه المعلمات دون استخدام واجهة pmap.
bool_t xdr_rejected_reply(XDR *xdrs, struct rejected_reply *rr);
يُستخدم لوصف رسائل ردود RPC. هذه الروتين مفيدة للمستخدمين الذين يرغبون في إنشاء رسائل بأسلوب RPC دون استخدام حزمة RPC.
bool_t xdr_replymsg(XDR *xdrs, struct rpc_msg *rmsg);
يُستخدم لوصف رسائل ردود RPC. هذه الروتين مفيدة للمستخدمين الذين يرغبون في إنشاء رسائل بأسلوب RPC دون استخدام حزمة RPC.
void xprt_register(SVCXPRT *xprt);
بعد إنشاء معالجات نقل خدمة RPC، يجب عليها تسجيل نفسها في حزمة خدمة RPC. هذه الروتين تعدل المتغير العام svc_fds. عادةً لا يحتاج منفذو الخدمة إلى هذه الروتين.
void xprt_unregister(SVCXPRT *xprt);
قبل تدمير معالج نقل خدمة RPC، يجب عليه إلغاء تسجيل نفسه في حزمة خدمة RPC. هذه الروتين تعدل المتغير العام svc_fds. عادةً لا يحتاج منفذو الخدمة إلى هذه الروتين.
