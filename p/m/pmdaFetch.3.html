pmdaFetch(3)

الحزمة: pcp

التراخيص: GPL-2.0-or-later, LGPL-2.1-or-later, CC-BY-3.0

PMDAFETCH(3) دليل وظائف المكتبة PMDAFETCH(3)

الاسم

pmdaFetch, pmdaSetFetchCallBack - ملء بنية pmResult بقيم المقاييس المطلوبة

C الملخص

#include <pcp/pmapi.h>
#include <pcp/pmda.h>

int pmdaFetch(int numpmid,
pmID *pmidlist, pmResult **resp, pmdaExt *pmda);

void pmdaSetFetchCallBack(pmdaInterface *dispatch,
pmdaFetchCallBack callback);

cc ... -lpcp_pmda -lpcp

الوصف

pmdaFetch هو رد اتصال عام تستخدمه PMDA(3) لمعالجة طلب جلب من pmcd(1). يتم بدء الطلب من pmcd بواسطة عميل يستدعي pmFetch(3).

هذا واحد من ردود الاتصال العامة القليلة في libpcp_pmda (انظر PMDA(3)) والذي يعتبر غير مكتمل، ويتطلب طريقة pmdaFetchCallBack إضافية خاصة به. يجب تسجيل رد الاتصال الإضافي باستخدام pmdaSetFetchCallBack وله طريقة pmdaFetchCallBack النموذج الأولي التالي:

int func(pmdaMetric *mdesc, unsigned int inst, pmAtomValue *avp)

سيقوم pmdaFetch بتخصيص وتغيير حجم بنية نتيجة resp، لتخزين القيم للمقاييس numpmid المدرجة في pmidlist.

لكل مثيل مدرج في التكوين (انظر pmdaProfile(3)) لكل مقياس مدرج في pmidlist، يتم استدعاء طريقة pmdaFetchCallBack لملء بنية pmAtomValue المشار إليها بواسطة avp بقيمة لزوج مقياس-مثيل محدد معرف بواسطة واصف المقياس mdesc والمثيل inst. يتم بعد ذلك نسخ هذه القيمة إلى بنية pmResult.

يجب أن تعيد طريقة pmdaFetchCallBack قيمة أقل من الصفر للإشارة إلى خطأ، والحالات الأكثر احتمالاً هي PM_ERR_PMID إذا كان المقياس المحدد بواسطة mdesc غير معروف للطريقة، أو PM_ERR_INST إذا اعتقدت الطريقة أن المثيل inst غير معروف للمقياس المحدد بواسطة mdesc.

تعتمد رموز الأخطاء الناجحة على إصدار PMDA_INTERFACE الذي يستخدمه PMDA.

إذا كان PMDA يستخدم PMDA_INTERFACE_2، فعند النجاح يجب أن تعيد طريقة pmdaFetchCallBack القيمة 0.

إذا كان PMDA يستخدم PMDA_INTERFACE_3 أو PMDA_INTERFACE_4، فعند النجاح يجب أن تعيد طريقة pmdaFetchCallBack القيمة 1 إذا تم إرجاع قيمة عبر avp، وإلا 0 إذا لم تكن هناك قيم متاحة حاليًا لزوج المقياس-المثيل المطلوب على الرغم من أن mdesc و inst يبدوان معقولين.

إذا كان PMDA يستخدم PMDA_INTERFACE_5 أو أحدث، فعند النجاح يجب أن تعيد طريقة pmdaFetchCallBack القيمة PMDA_FETCH_STATIC (1) إذا كان يمكن تجاهل القيمة التي تم إرجاعها عبر avp بواسطة pmdaFetch بمجرد نسخها إلى بنية pmResult، وإلا PMDA_FETCH_DYNAMIC (2) إذا كانت القيمة التي تم إرجاعها عبر avp تستخدم إما حقلي vp أو cp للبنية pmAtomValue وتم تخصيص القيمة (المخزن المؤقت) المرتبطة باستخدام أحد malloc(3)، calloc(3)، realloc(3)، strdup (3) وما إلى ذلك ويجب على pmdaFetch تحرير الذاكرة عن طريق استدعاء free(3) بمجرد تخصيص مخزن مؤقت جديد ونسخ القيمة، وإلا PMDA_FETCH_NOVALUES (0) إذا لم تكن هناك قيم متاحة حاليًا لزوج المقياس-المثيل المطلوب على الرغم من أن mdesc و inst يبدوان معقولين.

إذا أعادت طريقة pmdaFetchCallBack قيمة لمثيل مقياس من نوع PM_TYPE_STRING أو PM_TYPE_AGGREGATE، فهناك حاجة إلى بعض العناية الخاصة - يجب على الطريقة إما استخدام مخزن مؤقت ثابت، أو تعيين avp->cp أو avp->vp إلى عنوان المخزن المؤقت وإرجاع PMDA_FETCH_STATIC، أو استخدام مخزن مؤقت مخصص ديناميكيًا، والاحتفاظ بمرجع ثابت لعنوان المخزن المؤقت، وإرجاع PMDA_FETCH_STATIC و free (3) أو realloc (3) أو إعادة استخدام المخزن المؤقت في المرة التالية التي يتم فيها استدعاء طريقة pmdaFetchCallBack، وإلا استخدم مخزن مؤقت مخصص ديناميكيًا وأرجع PMDA_FETCH_DYNAMIC.

مثال

تُستخدم المقاطع البرمجية التالية لـ PMDA افتراضي لديه مقاييس (A، B، C، و D) ونطاق مثيلات (X) مع مثيلين (X1 و X2). يمكن تعريف جداول وصف نطاق المثيلات والمقاييس (انظر pmdaInit(3)) كالتالي:

static pmdaInstid _X[] = {
    { 0, "X1" }, { 1, "X2" }
};

static pmdaIndom indomtab[] = {
#define X_INDOM 0
    { 0, 2, _X },
};

static pmdaMetric metrictab[] = {
/* A */
    { (void *)0,
      { PMDA_PMID(0,0), PM_TYPE_32, PM_INDOM_NULL,
        PM_SEM_INSTANT, {0,0,0,0,0,0} }, },
/* B */
    { (void *)0,
      { PMDA_PMID(0,1), PM_TYPE_DOUBLE, X_INDOM,
        PM_SEM_INSTANT, {0,1,0,0,PM_TIME_SEC,0} }, },
/* C */
    { (void *)0,
      { PMDA_PMID(0,2), PM_TYPE_STRING, PM_INDOM_NULL,
        PM_SEM_INSTANT, {0,0,0,0,0,0} }, },
/* D */
    { (void *)0,
      { PMDA_PMID(0,3), PM_TYPE_STRING, PM_INDOM_NULL,
        PM_SEM_INSTANT, {0,0,0,0,0,0} }, },
};

يمكن تعريف طريقة pmdaFetchCallBack لاستدعائها من pmdaFetch كالتالي:

int
myFetchCallBack(pmdaMetric *mdesc, unsigned int inst, pmAtomValue *avp)
{
  static char sbuf[20]; // إعادة استخدام هذا المخزن المؤقت
  char        *dbuf;    // تم تخصيصه باستخدام malloc
  switch (pmID_item(mdesc->m_desc.pmid)) {
    case 0:
      /* تعيين قيمة للمقياس A */;
      avp->l = ...
      break;
    case 1:
      switch (inst) {
          case 0:
	      /* تعيين قيمة للمقياس B، المثيل X1 */;
              avp->d = ...
              break;
          case 1:
	      /* تعيين قيمة للمقياس B، المثيل X2 */;
              avp->d = ...
              break;
          default:
              return PM_ERR_INST;
      }
    case 2:
      /* وضع القيمة للمقياس C في dbuf[] */
      memcpy(dbuf, ...);
      avp->cp = dbuf;
      break;
    case 3:
      avp->cp = (char *)malloc(somesize);
      /* وضع القيمة في avp->cp */
      pmsprintf(avp->cp, somesize, ...);
      return PMDA_FETCH_DYNAMIC;
    default:
      return PM_ERR_PMID;
  }
  return PMDA_FETCH_STATIC;
}

تنبيه

يجب أن يستخدم PMDA PMDA_INTERFACE_2 أو أحدث، كما هو محدد في الاستدعاء إلى pmdaDSO(3) أو pmdaDaemon(3).

التشخيص

تشير رسائل الخطأ التالية إلى وجود تضارب بين مساحة الأسماء، وجداول pmdaMetric و pmdaIndom التي تم تمريرها إلى pmdaInit(3)، ورد الاتصال المسجل للجلب:

pmdaFetch: المقياس المطلوب metric غير معرف
مقياس مطلوب metric غير مدرج في جدول pmdaMetric. قد تحتوي مساحة الأسماء لهذا PMDA(3) على مقاييس إضافية.
pmdaFetch: رد الاتصال للجلب لا يعالج PMID pmid
أعادت طريقة pmdaFetchCallBack القيمة PM_ERR_PMID. هذا يشير إلى أن مقياسًا قد يكون مدرجًا في جدول pmdaMetric، ولكنه غير مدعوم من قبل طريقة رد الاتصال.
pmdaFetch: رد الاتصال للجلب لا يعالج المثيل inst الخاص بـ PMID pmid
أعادت طريقة pmdaFetchCallBack القيمة PM_ERR_INST. هذا يشير إلى أن مثيلاً لمقياس مدرج في جدول pmdaIndom، ولكنه غير مدعوم من قبل طريقة رد الاتصال.
pmdaFetch: خطأ في رد الاتصال للجلب:
أعادت طريقة pmdaFetchCallBack نتيجة بخلاف PMDA_FETCH_NOVALUES، PMDA_FETCH_STATIC، PMDA_FETCH_DYNAMIC، PM_ERR_PMID أو PM_ERR_INST.
pmdaFetch: نوع الواصف ( type) للمقياس pmid خاطئ
نوع البيانات type المحدد للمقياس pmid في جدول pmdaMetric غير قانوني.

سيعيد pmdaFetch القيمة -errno إذا حدث خطأ
