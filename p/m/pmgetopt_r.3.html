PMGETOPTIONS(3) دليل وظائف المكتبة PMGETOPTIONS(3)

الاسم

pmgetopt_r، pmGetOptions، pmGetContextOptions، pmFreeOptions، pmUsageMessage - معالجة سطر الأوامر لأدوات PMAPI

C ملخص

#include <pcp/pmapi.h>

int pmgetopt_r(int argc، char *const *argv، pmOptions *opts);
int pmGetOptions(int argc،
char *const *argv، pmOptions *opts);

int pmGetContextOptions(int ctx، pmOptions *opts);
void pmUsageMessage(pmOptions *opts);
void pmFreeOptions(pmOptions *opts);

cc ... -lpcp

الوصف

توفر الوظيفة pmGetOptions خدمات معالجة خيارات سطر الأوامر لكل من أدوات المراقبة والمجمّع PMAPI(3). تم تصميمها على غرار النسخ المتوافقة مع الخيوط من واجهة برمجة تطبيقات GNU getopt_long(3)، وتختلف بشكل أساسي في تركيزها على توفير معالجة معممة لخيارات سطر أوامر PCP القياسية (بحكم الواقع) الموصوفة في PCPIntro(1). تتضمن هذه الخيارات الشائعة مواصفات المضيف والأرشيف، والنوافذ الزمنية، والمناطق الزمنية، وعدد العينات، والفواصل الزمنية، وما إلى ذلك.

الواجهة الأساسية هي pmGetOptions، والتي يجب تمرير وسيطات argc و argv إليها، كما تم تمريرها إلى وظيفة main() عند استدعاء البرنامج. يصف الوسيط opts النهائي مجموعة الخيارات الطويلة والقصيرة التي أداة مستعدة لمعالجتها، وبيانات وصفية أخرى حول كيفية معالجة هذه الخيارات.

تتعامل واجهة pmgetopt_r، المستخدمة داخليًا بواسطة pmGetOptions، بطريقة مشابهة، ولكنها لا تقوم بأي معالجة للخيارات الشائعة. إنها أكثر ملاءمة لعمليات مجمّع PCP، في حين تميل أدوات مراقب PCP إلى استخدام pmGetOptions.

تتكون الوسيط opts من مصفوفة من إدخالات pmLongOpts تصف الوسيطات، بالإضافة إلى الهيكل المغلف pmOptions، والتي يتم تعريفها على النحو التالي (الحقول الداخلية غير معروضة، للإيجاز):

typedef struct {
    const char *        long_opt;
    int                 has_arg;
    int                 short_opt;
    const char *        argname;
    const char *        message;
} pmLongOptions;
typedef struct {
    int                 version;
    int                 flags;
    const char *        short_options;
    pmLongOptions *     long_options;
    const char *        short_usage;
    pmOptionOverride    override;
    int                 index;
    int                 optind;
    int                 opterr;
    int                 optopt;
    char                *optarg;
    /* [internal fields, undocumented] */
    int                 errors;
    int                 context; /* PM_CONTEXT_{HOST,ARCHIVE,LOCAL} */
    int                 nhosts;
    int                 narchives;
    char **             hosts;
    char **             archives;
    struct timespec     start;
    struct timespec     finish;
    struct timespec     origin;
    struct timespec     interval;
    char *              align_optarg;
    char *              start_optarg;
    char *              finish_optarg;
    char *              origin_optarg;
    char *              guiport_optarg;
    char *              timezone;
    int                 samples;
    int                 guiport;
    int                 padding;
    unsigned int        guiflag : 1;
    unsigned int        tzflag  : 1;
    unsigned int        nsflag  : 1;
    unsigned int        Lflag   : 1;
    unsigned int        zeroes  : 28;
} pmOptions;

تصف الحقول الأولية flags و version كيف سيتم تفسير بقية هيكل pmOptions. يمكن تصفير هذه الحقول، مما يشير إلى تفسير افتراضي. بدلاً من ذلك، يمكن استخدام الماكرو PMAPI_VERSION لتحديد مستوى واجهة برمجة التطبيقات المراد استخدامه (حاليًا، يُسمح بقيم 3 أو أقل). الإصدار 2 هو الافتراضي، والإصدار 3 يقدم حقول نافذة زمنية وفواصل زمنية عالية الدقة (أي باستخدام struct timespec بدلاً من struct timeval). عند استخدام الشكل الأخير، قبل تضمين <pcp/pmapi.h>، يجب تعيين الماكرو PMAPI_VERSION إلى 3 لضمان تخطيط صحيح لهيكل pmOptions لاستخدامه بواسطة التطبيق. يمكن استخدام حقل flags لتعديل سلوك معالجة الخيارات كما هو موصوف في قسم ``قيم الأعلام'' أدناه.

يجب إنهاء مصفوفة هياكل long_options pmLongOpts بمانع ويمكن استخدام الماكرو PMAPI_OPTIONS_END لإحداث هذا الإنهاء. يمكن أن تكون السجلات الفردية داخل مصفوفة long_options من نوعين - رؤوس خيارات، أو خيارات فعلية. يتم إنشاء رأس خيارات باستخدام الماكرو PMAPI_OPTIONS_HEADER، ويستخدم لتجميع خيارات رسالة الاستخدام. يمكن إدراج نص حر في رسالة الاستخدام في أي نقطة باستخدام الماكرو PMAPI_OPTIONS_TEXT - هذا مخصص للنص التفسيري الإضافي الذي يغطي الاستخدام التفصيلي الذي يتجاوز نطاق الرؤوس أو الخيارات الفردية. بخلاف ذلك، يحدد إدخال المصفوفة خيارًا. يجب تسميتها (long_opt) إذا كان شكل الخيار الطويل مسموحًا به، وتحديد ما إذا كانت تأخذ وسيطًا أم لا (has_arg)، وتحديد الوسيط الأحادي الحرف المتغير (short_opt) إذا كان شكل الخيار القصير مسموحًا به، وأخيرًا تحديد كيفية تقديم الخيار في رسالة الاستخدام. يتكون هذا المكون الأخير من وصف قصير من كلمة واحدة للوسيط الاختياري (argname) ووصف من سطر واحد لما يفعله خيار سطر الأوامر (message).

يتم استخدام السلسلة short_usage أيضًا عند إنشاء رسالة الاستخدام. تشكل الجزء من رسالة الاستخدام الذي يلي اسم البرنامج (أي argv[0]).

السلسلة الاختيارية short_options هي سلسلة مواصفات خيار سطر الأوامر العادية لـ getopt، باستخدام أحرف فردية (تلك التي تحتوي على وسيطات محددة باستخدام الحرف ':') - كما تستخدمها جميع تطبيقات getopt.

يتم توفير مرفق لتوسيع المجموعة الحالية من الخيارات الشائعة بخيارات إضافية، وكذلك لإعادة تكليف الخيارات القياسية بأدوار غير قياسية لأدوات فردية. يتم تحقيق هذا الأخير باستخدام طريقة override، والتي تسمح بتوفير وظيفة استدعاء سيتم استدعاؤها عند تلقي كل وسيط، قبل المعالجة المشتركة. إذا أعادت هذه الوظيفة قيمة غير صفرية، فسيتم اختصار المعالجة المشتركة لهذا الخيار، وإلا ستستمر المعالجة. وبالتالي، فإن كل أداة عميل حرة في اختيار الخيارات القياسية التي ترغب في دعمها بالضبط - يمكن أن يكون كل ذلك، أو بعضه، أو لا شيء، وبغض النظر عما تختاره، لكل أداة دائمًا إمكانية الوصول إلى قدرة تحليل الخيارات الطويلة وتسهيل إنشاء رسالة الاستخدام.

تمتلئ حقول هيكل pmOptions المتبقية نتيجة لمعالجة الوسيطات، وهي واضحة بشكل كبير. يتم تأجيل المزيد من المناقشة لهذه في قسم ``قيم الأعلام'' أدناه. يحتوي حقل error على عدد الأخطاء المكتشفة أثناء معالجة الخيارات. يمكن أن تكون هذه أخطاء استخدام أو أخطاء وقت تشغيل، كما يشير إليها حقل flags (تم تعيينها، وتم تمريرها إلى المتصل). عادةً، ستفشل أداة سطر الأوامر في البدء بنجاح وستنتج رسالة خطأ (على سبيل المثال، عبر pmUsageMessage) إذا كان حقل error غير صفري في نهاية pmGetOptions أو pmGetContextOptions.

لا يمكن إجراء بعض المعالجات اللاحقة لخيارات سطر الأوامر إلا بعد أن تنشئ الأداة سياق PMAPI عبر pmNewContext(3). تشمل هذه المعالجة استخدام المناطق الزمنية المتعلقة بالسياق (-z)، ومعالجة النوافذ الزمنية (-A، -O، -S، -T) التي قد تتأثر بالمنطقة الزمنية، على سبيل المثال. الوظيفة pmGetContextOptions متاحة لهذه المواقف، وتقوم بإكمال أي معالجة متبقية لـ opts فيما يتعلق بمعرف السياق ctx المقدم.

تقوم الوظيفة pmUsageMessage بإنشاء رسالة استخدام للأداة، وتضمين كل من خيارات PCP القياسية والخيارات المخصصة لكل أداة، كما هو محدد بواسطة مصفوفة pmLongOptions. تدعم تجميع الخيارات (عبر PMAPI_OPTIONS_HEADER) بالإضافة إلى التنسيق الأنيق لجميع الخيارات - القصيرة والطويلة - ووسيطاتها، ورسائلها التفسيرية الفردية. ستقوم ببناء رسالة الاستخدام هذه باستخدام pmprintf(3) والتي ستصدر عليها استدعاء pmflush(3) واحد قبل العودة إلى المتصل، بشرط تعيين علامة PM_OPTFLAG_USAGE_ERR في flags، والتي سيحدث تلقائيًا أثناء تحليل الخيارات، عند اكتشاف أخطاء الاستخدام.

في مواقف معينة، مثل تسجيل قوائم مواصفات المضيف أو مسارات أرشيف PCP، قد تقوم روتين pmGetOptions بتخصيص ذاكرة، وتخزين مؤشرات إليها داخل opts. إذا أراد برنامج ما تحرير هذه الذاكرة
