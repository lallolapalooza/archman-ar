rpc(3) دليل وظائف المكتبة rpc(3)

الاسم

rpc - روتينات المكتبة لاستدعاءات الإجراءات عن بعد

المكتبة

مكتبة C القياسية (libc, -lc)

الملخص والوصف

تسمح هذه الروتينات لبرامج C بإجراء استدعاءات إجراءات على أجهزة أخرى عبر الشبكة. أولاً، يستدعي العميل إجراءً لإرسال حزمة بيانات إلى الخادم. عند استلام الحزمة، يستدعي الخادم روتين إرسال لأداء الخدمة المطلوبة، ثم يرسل رداً. أخيراً، يعود استدعاء الإجراء إلى العميل.

للاستفادة من هذه الروتينات، قم بتضمين ملف الرأس <rpc/rpc.h>.

النماذج الأولية أدناه تستفيد من الأنواع التالية:

typedef int bool_t;
typedef typeof(bool_t (XDR *, void *, ...))  *xdrproc_t;
typedef typeof(bool_t (caddr_t resp, struct sockaddr_in *raddr)
    *resultproc_t;

راجع ملفات الرأس للحصول على تعريفات الأنواع AUTH، CLIENT، SVCXPRT، و XDR.

void auth_destroy(AUTH *auth);
ماكرو يقوم بتدمير معلومات المصادقة المرتبطة بـ auth. عادة ما يتضمن التدمير إلغاء تخصيص هياكل البيانات الخاصة. استخدام auth غير محدد بعد استدعاء auth_destroy().
AUTH *authnone_create(void);
ينشئ ويعيد مقبض مصادقة RPC يمرر معلومات مصادقة غير قابلة للاستخدام مع كل استدعاء إجراء عن بعد. هذه هي المصادقة الافتراضية التي يستخدمها RPC.
AUTH *authunix_create(char *host, uid_t uid, gid_t gid,
                      int n, gid_t aup_gids[n]);
ينشئ ويعيد مقبض مصادقة RPC يحتوي على معلومات مصادقة. المعلمة host هي اسم الجهاز الذي تم فيه إنشاء المعلومات؛ uid هو معرف المستخدم؛ gid هو معرف المجموعة الحالي للمستخدم؛ n و aup_gids تشير إلى مصفوفة محسوبة للمجموعات التي ينتمي إليها المستخدم. من السهل انتحال شخصية مستخدم.
AUTH *authunix_create_default(void);
يستدعي authunix_create() بالمعلمات المناسبة.
int callrpc(char *host, unsigned long prognum,
            unsigned long versnum, unsigned long procnum,
            xdrproc_t inproc, const char *in,
            xdrproc_t outproc, char *out);
استدعاء الإجراء البعيد المرتبط بـ prognum و versnum و procnum على الجهاز host. المعلمة in هي عنوان وسيط (وسائط) الإجراء، و out هي العنوان الذي سيتم فيه وضع النتيجة (النتائج)؛ يستخدم inproc لتشفير معلمات الإجراء، ويستخدم outproc لفك تشفير نتائج الإجراء. يعيد هذا الروتين صفرًا إذا نجح، أو قيمة enum clnt_stat محولة إلى عدد صحيح إذا فشل. الروتين clnt_perrno() مفيد لترجمة حالات الفشل إلى رسائل.
تحذير: استدعاء الإجراءات البعيدة باستخدام هذا الروتين يستخدم UDP/IP كواسطة نقل؛ راجع clntudp_create() للقيود. ليس لديك تحكم في المهلات أو المصادقة باستخدام هذا الروتين.
enum clnt_stat clnt_broadcast(unsigned long prognum,
                     unsigned long versnum, unsigned long procnum,
                     xdrproc_t inproc, char *in,
                     xdrproc_t outproc, char *out,
                     resultproc_t eachresult);
مثل callrpc()، باستثناء أن رسالة الاستدعاء يتم بثها إلى جميع شبكات البث المتصلة محليًا. في كل مرة يستلم فيها ردًا، يستدعي هذا الروتين eachresult()، الذي يكون شكله:

eachresult(char *out, struct sockaddr_in *addr);

حيث out هو نفس out الذي تم تمريره إلى clnt_broadcast()، باستثناء أنه يتم فك تشفير خرج الإجراء البعيد هناك؛ addr يشير إلى عنوان الجهاز الذي أرسل النتائج. إذا أعاد eachresult() صفرًا، ينتظر clnt_broadcast() ردودًا إضافية؛ وإلا فإنه يعود بحالة مناسبة.
تحذير: مقابس البث محدودة الحجم بحد أقصى لوحدة نقل البيانات على رابط البيانات. بالنسبة للإيثرنت، تبلغ هذه القيمة 1500 بايت.
enum clnt_stat clnt_call(CLIENT *clnt, unsigned long procnum,
                    xdrproc_t inproc, char *in,
                    xdrproc_t outproc, char *out,
                    struct timeval tout);
ماكرو يستدعي الإجراء البعيد procnum المرتبط بمقبض العميل clnt، والذي تم الحصول عليه باستخدام روتين إنشاء عميل RPC مثل clnt_create(). المعلمة in هي عنوان وسيط (وسائط) الإجراء، و out هي العنوان الذي سيتم فيه وضع النتيجة (النتائج)؛ يستخدم inproc لتشفير معلمات الإجراء، ويستخدم outproc لفك تشفير نتائج الإجراء؛ tout هو الوقت المسموح به لعودة النتائج.
clnt_destroy(CLIENT *clnt);
ماكرو يدمر مقبض RPC الخاص بالعميل. عادة ما يتضمن التدمير إلغاء تخصيص هياكل البيانات الخاصة، بما في ذلك clnt نفسه. استخدام clnt غير محدد بعد استدعاء clnt_destroy(). إذا قامت مكتبة RPC بفتح المقبس المرتبط، فستقوم بإغلاقه أيضًا. بخلاف ذلك، يبقى المقبس مفتوحًا.
CLIENT *clnt_create(const char *host, unsigned long prog,
                    unsigned long vers, const char *proto);
روتين إنشاء عميل عام. يحدد host اسم المضيف البعيد حيث يوجد الخادم. يشير proto إلى نوع بروتوكول النقل المراد استخدامه. القيم المدعومة حاليًا لهذا الحقل هي "udp" و "tcp". يتم تعيين مهلات افتراضية، ولكن يمكن تعديلها باستخدام clnt_control().
تحذير: استخدام UDP له عيوبه. نظرًا لأن رسائل RPC المستندة إلى UDP يمكن أن تحتوي فقط على ما يصل إلى 8 كيلوبايت من البيانات المشفرة، لا يمكن استخدام هذه الواسطة للإجراءات التي تأخذ وسائط كبيرة أو تعيد نتائج ضخمة.
bool_t clnt_control(CLIENT *cl, int req, char *info);
ماكرو يستخدم لتغيير أو استرداد معلومات مختلفة حول كائن عميل. يشير req إلى نوع العملية، و info هو مؤشر إلى المعلومات. لكل من UDP و TCP، القيم المدعومة لـ req وأنواع وسائطها وما تفعله هي:

CLSET_TIMEOUT  struct timeval // تعيين المهلة الإجمالية
CLGET_TIMEOUT  struct timeval // الحصول على المهلة الإجمالية

ملاحظة: إذا قمت بتعيين المهلة باستخدام clnt_control()، فسيتم تجاهل معلمة المهلة التي تم تمريرها إلى clnt_call() في جميع الاستدعاءات المستقبلية.

CLGET_SERVER_ADDR  struct sockaddr_in
                // الحصول على عنوان الخادم

العمليات التالية صالحة لـ UDP فقط:

CLSET_RETRY_TIMEOUT  struct timeval // تعيين مهلة إعادة المحاولة
CLGET_RETRY_TIMEOUT  struct timeval // الحصول على مهلة إعادة المحاولة

مهلة إعادة المحاولة هي الوقت الذي ينتظره "RPC UDP" حتى يرد الخادم قبل إعادة إرسال الطلب.
clnt_freeres(CLIENT * clnt, xdrproc_t outproc, char *out);
ماكرو يقوم بتحرير أي بيانات تم تخصيصها بواسطة نظام RPC/XDR عند فك تشفير نتائج استدعاء RPC. المعلمة out هي عنوان النتائج، و outproc هو روتين XDR الذي يصف النتائج. يعيد هذا الروتين واحدًا إذا تم تحرير النتائج بنجاح، وصفرًا بخلاف ذلك.
void clnt_geterr(CLIENT *clnt, struct rpc_err

هذا الموقع غير مرتبط بـ Arch Linux ولا تابع له بأي شكل من الأشكال. المحتوى هو ترجمة فقط لصفحات دليل Arch المنشورة على man.archlinux.org.

يُستخدم لترميز رسائل استدعاء RPC. هذا الإجراء مفيد للمستخدمين الذين يرغبون في إنشاء رسائل بنمط RPC دون استخدام حزمة RPC.
struct rpc_createerr rpc_createerr;
متغير عام يتم تعيين قيمته بواسطة أي إجراء إنشاء عميل RPC لا ينجح. استخدم الإجراء clnt_pcreateerror() لطباعة السبب.
void svc_destroy(SVCXPRT *xprt);
ماكرو يدمر مقبض نقل خدمة RPC، xprt. عادة ما يتضمن التدمير إلغاء تخصيص هياكل البيانات الخاصة، بما في ذلك xprt نفسه. استخدام xprt غير محدد بعد استدعاء هذا الإجراء.
fd_set svc_fdset;
متغير عام يعكس قناع بت واصف الملف للقراءة الخاص بجانب خدمة RPC؛ وهو مناسب كمعامل لاستدعاء النظام select(2). هذا يهم فقط إذا كان مُنشئ الخدمة يقوم بمعالجة الأحداث غير المتزامنة الخاصة به، بدلاً من استدعاء svc_run(). هذا المتغير للقراءة فقط (لا تمرر عنوانه إلى select(2)!)، ومع ذلك، قد يتغير بعد استدعاءات svc_getreqset() أو أي إجراءات إنشاء.
int svc_fds;
مشابه لـ svc_fdset، ولكنه يقتصر على 32 واصف ملف. تم إهمال هذه الواجهة بواسطة svc_fdset.
svc_freeargs(SVCXPRT *xprt, xdrproc_t inproc, char *in);
ماكرو يحرر أي بيانات تم تخصيصها بواسطة نظام RPC/XDR عند فك ترميز المعاملات لإجراء خدمة باستخدام svc_getargs(). يعيد هذا الإجراء 1 إذا تم تحرير النتائج بنجاح، وصفر بخلاف ذلك.
svc_getargs(SVCXPRT *xprt, xdrproc_t inproc, char *in);
ماكرو يقوم بفك ترميز معاملات طلب RPC المرتبط بمقبض نقل خدمة RPC، xprt. المعامل in هو العنوان الذي سيتم وضع المعاملات فيه؛ inproc هو إجراء XDR المستخدم لفك ترميز المعاملات. يعيد هذا الإجراء واحدًا إذا نجح فك الترميز، وصفر بخلاف ذلك.
struct sockaddr_in *svc_getcaller(SVCXPRT *xprt);
الطريقة المعتمدة للحصول على عنوان الشبكة للمتصل بإجراء مرتبط بمقبض نقل خدمة RPC، xprt.
void svc_getreqset(fd_set *rdfds);
هذا الإجراء يهم فقط إذا كان مُنشئ الخدمة لا يستدعي svc_run()، ولكنه بدلاً من ذلك ينفذ معالجة أحداث غير متزامنة مخصصة. يتم استدعاؤه عندما يحدد استدعاء النظام select(2) أن طلب RPC قد وصل على بعض مآخذ RPC؛ rdfds هو قناع بت واصف الملف للقراءة الناتج. يعود الإجراء عندما تتم خدمة جميع المآخذ المرتبطة بالقيمة rdfds.
void svc_getreq(int rdfds);
مشابه لـ svc_getreqset()، ولكنه يقتصر على 32 واصف ملف. تم إهمال هذه الواجهة بواسطة svc_getreqset().
bool_t svc_register(SVCXPRT *xprt, unsigned long prognum,
                    unsigned long versnum,
                    typeof(void (struct svc_req *, SVCXPRT *)) *dispatch,
                    unsigned long protocol);
يربط prognum و versnum بإجراء إرسال الخدمة، dispatch. إذا كان protocol صفرًا، فلن يتم تسجيل الخدمة مع خدمة portmap. إذا كان protocol غير صفري، فيتم إنشاء تعيين للثلاثي [prognum,versnum,protocol] إلى xprt->xp_port مع خدمة portmap المحلية (بشكل عام protocol هو صفر، IPPROTO_UDP أو IPPROTO_TCP). للإجراء dispatch الشكل التالي:

dispatch(struct svc_req *request, SVCXPRT *xprt);

يعيد الإجراء svc_register() واحدًا إذا نجح، وصفر بخلاف ذلك.
void svc_run(void);
هذا الإجراء لا يعود أبدًا. ينتظر وصول طلبات RPC، ويستدعي إجراء الخدمة المناسب باستخدام svc_getreq() عند وصول واحد. عادة ما ينتظر هذا الإجراء استدعاء النظام select(2) للعودة.
bool_t svc_sendreply(SVCXPRT *xprt, xdrproc_t outproc, char *out);
يتم استدعاؤه بواسطة إجراء إرسال خدمة RPC لإرسال نتائج استدعاء إجراء عن بعد. المعامل xprt هو مقبض النقل المرتبط بالطلب؛ outproc هو إجراء XDR المستخدم لترميز النتائج؛ و out هو عنوان النتائج. يعيد هذا الإجراء واحدًا إذا نجح، وصفر بخلاف ذلك.
void svc_unregister(unsigned long prognum, unsigned long versnum);
إزالة جميع تعيينات الزوج [prognum,versnum] إلى إجراءات الإرسال، وتعريف الثلاثي [prognum,versnum,*] إلى رقم المنفذ.
void svcerr_auth(SVCXPRT *xprt, enum auth_stat why);
يتم استدعاؤه بواسطة إجراء إرسال خدمة يرفض أداء استدعاء إجراء عن بعد بسبب خطأ في المصادقة.
void svcerr_decode(SVCXPRT *xprt);
يتم استدعاؤه بواسطة إجراء إرسال خدمة لا يمكنه فك ترميز معاملاته بنجاح. انظر أيضًا svc_getargs().
void svcerr_noproc(SVCXPRT *xprt);
يتم استدعاؤه بواسطة إجراء إرسال خدمة لا ينفذ رقم الإجراء الذي يطلبه المتصل.
void svcerr_noprog(SVCXPRT *xprt);
يتم استدعاؤه عندما لا يتم تسجيل البرنامج المطلوب مع حزمة RPC. عادة لا يحتاج مُنشئو الخدمة إلى هذا الإجراء.
void svcerr_progvers(SVCXPRT *xprt, unsigned long low_vers,
                     unsigned long high_vers);
يتم استدعاؤه عندما لا يتم تسجيل الإصدار المطلوب من برنامج مع حزمة RPC. عادة لا يحتاج مُنشئو الخدمة إلى هذا الإجراء.
void svcerr_systemerr(SVCXPRT *xprt);
يتم استدعاؤه بواسطة إجراء إرسال خدمة عند اكتشاف خطأ في النظام لا يغطيه أي بروتوكول معين. على سبيل المثال، إذا لم يعد بإمكان الخدمة تخصيص مخزن، فقد تستدعي هذا الإجراء.
void svcerr_weakauth(SVCXPRT *xprt);
يتم استدعاؤه بواسطة إجراء إرسال خدمة يرفض أداء استدعاء إجراء عن بعد بسبب معاملات مصادقة غير كافية. يستدعي الإجراء svcerr_auth(xprt, AUTH_TOOWEAK).
SVCXPRT *svcfd_create(int fd, unsigned int sendsize,
                      unsigned int recvsize);
إنشاء خدمة فوق أي واصف ملف مفتوح. عادة، يكون واصف الملف هذا عبارة عن مقبس متصل لبروتوكول تدفق مثل TCP. sendsize و recvsize يشيران إلى أحجام المخازن المؤقتة للإرسال والاستقبال. إذا كانت صفرًا، يتم اختيار قيمة افتراضية معقولة.
SVCXPRT *svcraw_create(void);
ينشئ هذا الإجراء مقبض نقل خدمة RPC وهمي، والذي يعيد مؤشرًا إليه. النقل هو في الواقع مخزن مؤقت داخل مساحة عنوان العملية، لذا يجب أن يعيش عميل RPC المقابل في نفس مساحة العنوان؛ انظر clntraw_create(). يسمح هذا الإجراء بمحاكاة RPC والحصول على نفقات RPC (مثل أوقات الذهاب والعودة)، دون أي تدخل من النواة. يعيد هذا الإجراء NULL إذا فشل.
SVCXPRT *svctcp_create(int sock, unsigned int send_buf_size,
                       unsigned int recv_buf_size);
ينشئ هذا الإجراء مقبض نقل خدمة RPC يعتمد على TCP/IP، ويعيد مؤشرًا إليه. يرتبط النقل بالمقبس sock، والذي قد يكون RPC_ANYSOCK، وفي هذه الحالة يتم إنشاء مقبس جديد. إذا لم يكن المقبس مرتبطًا بمنفذ TCP محلي، يقوم هذا الإجراء بربطه بمنفذ عشوائي. عند الانتهاء، يكون xprt->xp_sock هو واصف مقبس النقل، و xprt->xp_port هو رقم منفذ النقل. يعيد هذا الإجراء NULL إذا فشل. نظرًا لأن RPC المستند إلى TCP يستخدم إدخال/إخراج مخزن مؤقت، يمكن للمستخدمين تحديد حجم المخازن المؤقتة؛ القيم صفر تختار قيمًا افتراضية مناسبة.
SVCXPRT *svcudp_bufcreate(int sock, unsigned int sendsize,
                          unsigned int recosize);
ينشئ هذا الإجراء مقبض نقل خدمة RPC يعتمد على UDP/IP، ويعيد مؤشرًا إليه. يرتبط النقل بالمقبس sock، والذي قد يكون RPC_ANYSOCK، وفي هذه الحالة يتم إنشاء مقبس جديد. إذا لم يكن المقبس مرتبطًا بمنفذ UDP محلي، يقوم هذا الإجراء بربطه بمنفذ عشوائي. عند الانتهاء، يكون xprt->xp_sock هو واصف مقبس النقل، و xprt->xp_port هو رقم منفذ النقل. يعيد هذا الإجراء NULL إذا فشل.
هذا يسمح للمستخدم بتحديد الحد الأقصى لحجم الحزمة لإرسال واستقبال رسائل RPC المستندة إلى UDP.
SVCXPRT *svcudp_create(int sock);
هذا الاستدعاء يعادل svcudp_bufcreate(sock,SZ,SZ) لبعض الأحجام الافتراضية SZ.
bool_t xdr_accepted_reply(XDR *xdrs, struct accepted_reply *ar);
يُستخدم لترميز رسائل رد RPC. هذا الإجراء مفيد للمستخدمين الذين يرغبون في إنشاء رسائل بنمط RPC دون استخدام حزمة RPC.
bool_t xdr_authunix_parms(XDR *xdrs, struct authunix_parms *aupp);
يُستخدم لوصف بيانات اعتماد UNIX. هذا الإجراء مفيد للمستخدمين الذين يرغبون في إنشاء بيانات الاعتماد هذه دون استخدام حزمة مصادقة RPC.
void xdr_callhdr(XDR *xdrs, struct rpc_msg *chdr);
يُستخدم لوصف رسائل رأس استدعاء RPC. هذا الإجراء مفيد للمستخدمين الذين يرغبون في إنشاء رسائل بنمط RPC دون استخدام حزمة RPC.
bool_t xdr_callmsg(XDR *xdrs
