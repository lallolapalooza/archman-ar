pmgenmap(1)

الحزمة: pcp

التراخيص: GPL-2.0-or-later, LGPL-2.1-or-later, CC-BY-3.0

PMGENMAP(1) دليل الأوامر العامة PMGENMAP(1)

الاسم

pmgenmap - إنشاء كود C لتبسيط التعامل مع مقاييس الأداء

الملخص

pmgenmap [ملف_الإدخال]

الوصف

بالنظر إلى قائمة واحدة أو أكثر من أسماء المقاييس في ملف_الإدخال أو من الإدخال القياسي، يقوم pmgenmap بإنشاء تعريفات C ووحدات ماكرو cpp(1) مناسبة للاستخدام عبر واجهة برمجة تطبيقات مقاييس الأداء (PMAPI) إلى الإخراج القياسي.

التعريفات التي ينتجها pmgenmap تبسط الترميز لتطبيقات العميل التي تستخدم PMAPI.

يجب أن يتكون الإدخال من قائمة واحدة أو أكثر من أسماء المقاييس بالشكل

اسم_القائمة {
    اسم_المقياس1 اسم_رمز1
    اسم_المقياس2 اسم_رمز2
    ...
}

والتي ستنتج تعريفات C و cpp(1) بالشكل

char *اسم_القائمة[] = {
#define اسم_رمز1 0
    "اسم_المقياس1",
#define اسم_رمز2 1
    "اسم_المقياس2",
    ...
};

تعريفات المصفوفة المنتجة مناسبة كمعلمات لـ pmLookupName(3) ويمكن استخدام الثوابت المعرفة بـ #define لفهرسة vsets في بنية pmResult التي تم إرجاعها بواسطة استدعاء pmFetch(3).

من الواضح أن اسم_القائمة يجب أن يتوافق مع قواعد تسمية معرفات C، ويجب أن يتوافق كل اسم_رمز مع قواعد تسمية وحدات ماكرو cpp(1)، ومن المتوقع أن يكون كل اسم_مقياس اسم مقياس أداء صالحًا (راجع PMNS(5) لمزيد من التفاصيل).

قد يتضمن الإدخال أسطر تعليقات بنمط sh، أي مع وجود '#' كأول حرف غير فارغ في السطر، ويتم ترجمة هذه الأسطر في الإخراج إلى تعليقات أحادية أو متعددة الأسطر بأسلوب K&R. على سبيل المثال، الإدخال:

# كتلة بادئة من التعليقات متعددة الأسطر
# مجموعة التهيئة
foo {
        a.b.c   ONE
        d.e.f.g TWO
        # كتلة مضمنة من التعليقات متعددة الأسطر
        # التعليقات ونص الحشو الممل
        xx.yy.zz        THREE
}
# تعليق سطر واحد لاحق

ينتج عنه الإخراج التالي:

/*
 * كتلة بادئة من التعليقات متعددة الأسطر
 * مجموعة التهيئة
 */
char *foo[] = {
#define ONE 0
        "a.b.c",
#define TWO 1
        "d.e.f.g",
/*
 * كتلة مضمنة من التعليقات متعددة الأسطر
 * التعليقات ونص الحشو الممل
 */
#define THREE 2
        "xx.yy.zz",
};
/* تعليق سطر واحد لاحق */

أمثلة

للاختصار، قمنا بإزالة رمز معالجة الأخطاء، وافترضنا أن المقاييس المختارة ليس لها قيم متعددة.

ملف الإدخال

mystats {
    kernel.percpu.cpu.idle     IDLE
    kernel.percpu.cpu.sys      SYS
    kernel.percpu.cpu.user     USER
    hinv.ncpu                       NCPU
}

ينتج عنه كود C التالي، مناسب لـ #include

/*
 * خريطة مساحة أسماء مقاييس الأداء
 * تم بناؤه بواسطة pmgenmap من الملف
 * mystats.metrics
 * في الأربعاء 28 ديسمبر 19:44:17 EST 1994
 *
 * لا تقم بتحرير هذا الملف!
 */
char *mystats[] = {
#define IDLE    0
        "kernel.percpu.cpu.idle",
#define SYS     1
        "kernel.percpu.cpu.sys",
#define USER    2
        "kernel.percpu.cpu.user",
#define NCPU    3
        "hinv.ncpu",
};

باستخدام الكود الذي تم إنشاؤه بواسطة pmgenmap، يمكننا الآن الحصول بسهولة على مقاييس من نظام فرعي لجمع مقاييس الأداء (PMCS) على النحو التالي:

#define MAX_PMID 4
    int         trip = 0;
    int         numpmid = sizeof(mystats)/sizeof(mystats[0]);
    double      duration;
    pmResult    *resp;
    pmResult    *prev;
    pmID        pmidlist[MAX_PMID];
    pmNewContext(PM_CONTEXT_HOST, "localhost");
    pmLookupName(numpmid, mystats, pmidlist);
    pmFetch(numpmid, pmidlist, &resp);
    printf("%d CPUs: %d usr   %d sys   %d   idle0,
           resp->vset[NCPU]->vlist[0].value.lval,
           resp->vset[USER]->vlist[0].value.lval,
           resp->vset[SYS]->vlist[0].value.lval,
           resp->vset[IDLE]->vlist[0].value.lval);

تمت إزالة بعض الاستدعاءات لضمان قابلية النقل من الكود أعلاه من أجل الوضوح - لا ينبغي استخدام المثال أعلاه كقالب للبرمجة. على وجه الخصوص، تم استخدام القيم الخام للمقاييس عندما كان يجب استدعاء pmLookupDesc(3) لتحديد دلالات كل مقياس.

تشمل الأمثلة الأكثر اكتمالاً التي توضح استخدام pmgenmap والتي يمكن استخدامها كأساس لتطوير البرامج عروض PCP التوضيحية، على سبيل المثال $PCP_DEMOS_DIR/pmclient.

الملفات

$PCP_VAR_DIR/pmns/*
ملفات مواصفات PMNS الافتراضية

بيئة PCP

تُستخدم متغيرات البيئة التي تبدأ بالبادئة PCP_ لضبط أسماء الملفات والأدلة المستخدمة بواسطة PCP. في كل تثبيت، يحتوي الملف /etc/pcp.conf على القيم المحلية لهذه المتغيرات. يمكن استخدام المتغير $PCP_CONF لتحديد ملف تكوين بديل، كما هو موضح في pcp.conf(5).
