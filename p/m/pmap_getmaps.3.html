void clnt_pcreateerror(const char *s);
طباعة رسالة إلى الخطأ القياسي توضح سبب عدم إنشاء مقبض عميل RPC. يتم تسبق الرسالة بالسلسلة النصية s ونقطتين. تستخدم عند فشل استدعاء clnt_create() أو clntraw_create() أو clnttcp_create() أو clntudp_create().
void clnt_perrno(enum clnt_stat stat);
طباعة رسالة إلى الخطأ القياسي تتوافق مع الحالة التي تشير إليها stat. تستخدم بعد callrpc().
clnt_perror(CLIENT *clnt, const char *s);
طباعة رسالة إلى الخطأ القياسي توضح سبب فشل استدعاء RPC؛ clnt هو المقبض المستخدم لإجراء الاستدعاء. يتم تسبق الرسالة بالسلسلة النصية s ونقطتين. تستخدم بعد clnt_call().
char *clnt_spcreateerror(const char *s);
مثل clnt_pcreateerror()، ولكنها ترجع سلسلة نصية بدلاً من طباعتها إلى الخطأ القياسي.
أخطاء: ترجع مؤشرًا إلى بيانات ثابتة يتم استبدالها عند كل استدعاء.
char *clnt_sperrno(enum clnt_stat stat);
تأخذ نفس الوسائط مثل clnt_perrno()، ولكن بدلاً من إرسال رسالة إلى الخطأ القياسي توضح سبب فشل استدعاء RPC، ترجع مؤشرًا إلى سلسلة نصية تحتوي على الرسالة. تنتهي السلسلة بسطر جديد.
يتم استخدام clnt_sperrno() بدلاً من clnt_perrno() إذا لم يكن لدى البرنامج خطأ قياسي (كما هو الحال غالبًا في برنامج يعمل كخادم)، أو إذا كان المبرمج لا يريد إخراج الرسالة باستخدام printf(3)، أو إذا كان المراد استخدام تنسيق رسالة مختلف عن الذي تدعمه clnt_perrno(). ملاحظة: على عكس clnt_sperror() و clnt_spcreateerror()، ترجع clnt_sperrno() مؤشرًا إلى بيانات ثابتة، ولكن النتيجة لن يتم استبدالها عند كل استدعاء.
char *clnt_sperror(CLIENT *rpch, const char *s);
مثل clnt_perror()، ولكن (مثل clnt_sperrno()) ترجع سلسلة نصية بدلاً من الطباعة إلى الخطأ القياسي.
أخطاء: ترجع مؤشرًا إلى بيانات ثابتة يتم استبدالها عند كل استدعاء.
CLIENT *clntraw_create(unsigned long prognum, unsigned long versnum);
يقوم هذا الروتين بإنشاء عميل RPC وهمي للبرنامج البعيد prognum، الإصدار versnum. آلية النقل المستخدمة لتمرير الرسائل إلى الخدمة هي في الواقع مخزن مؤقت داخل مساحة عنوان العملية، لذلك يجب أن يعيش خادم RPC المقابل في نفس مساحة العنوان؛ انظر svcraw_create(). هذا يسمح بمحاكاة RPC والحصول على تكاليف RPC، مثل أوقات الذهاب والعودة، دون أي تدخل من النواة. يعيد هذا الروتين NULL إذا فشل.
CLIENT *clnttcp_create(struct sockaddr_in *addr,
                unsigned long prognum, unsigned long versnum,
                int *sockp, unsigned int sendsz, unsigned int recvsz);
يقوم هذا الروتين بإنشاء عميل RPC للبرنامج البعيد prognum، الإصدار versnum؛ يستخدم العميل TCP/IP كآلية نقل. يقع البرنامج البعيد في عنوان الإنترنت *addr. إذا كان addr->sin_port صفرًا، فسيتم تعيينه إلى المنفذ الفعلي الذي يستمع عليه البرنامج البعيد (يتم استشارة خدمة portmap البعيدة لهذه المعلومات). الوسيط sockp هو مقبس؛ إذا كان RPC_ANYSOCK، يقوم هذا الروتين بفتحه وتعيين sockp. نظرًا لأن RPC المستند إلى TCP يستخدم إدخال/إخراج مخزن مؤقتًا، يمكن للمستخدم تحديد حجم المخازن المؤقتة للإرسال والاستقبال باستخدام الوسائط sendsz و recvsz؛ القيم صفر تختار الإعدادات الافتراضية المناسبة. يعيد هذا الروتين NULL إذا فشل.
CLIENT *clntudp_create(struct sockaddr_in *addr,
                unsigned long prognum, unsigned long versnum,
                struct timeval wait, int *sockp);
يقوم هذا الروتين بإنشاء عميل RPC للبرنامج البعيد prognum، الإصدار versnum؛ يستخدم العميل UDP/IP كآلية نقل. يقع البرنامج البعيد في عنوان الإنترنت addr. إذا كان addr->sin_port صفرًا، فسيتم تعيينه إلى المنفذ الفعلي الذي يستمع عليه البرنامج البعيد (يتم استشارة خدمة portmap البعيدة لهذه المعلومات). الوسيط sockp هو مقبس؛ إذا كان RPC_ANYSOCK، يقوم هذا الروتين بفتحه وتعيين sockp. آلية نقل UDP تعيد إرسال رسالة الاستدعاء بفواصل زمنية قدرها wait حتى يتم استلام استجابة أو حتى ينتهي وقت الاستدعاء. يتم تحديد الوقت الإجمالي لانتهاء وقت الاستدعاء بواسطة clnt_call().
تحذير: نظرًا لأن رسائل RPC المستندة إلى UDP يمكن أن تحتوي فقط على ما يصل إلى 8 كيلوبايت من البيانات المشفرة، لا يمكن استخدام آلية النقل هذه للإجراءات التي تأخذ وسائط كبيرة أو ترجع نتائج ضخمة.
CLIENT *clntudp_bufcreate(struct sockaddr_in *addr,
            unsigned long prognum, unsigned long versnum,
            struct timeval wait, int *sockp,
            unsigned int sendsize, unsigned int recosize);
يقوم هذا الروتين بإنشاء عميل RPC للبرنامج البعيد prognum، على versnum؛ يستخدم العميل UDP/IP كآلية نقل. يقع البرنامج البعيد في عنوان الإنترنت addr. إذا كان addr->sin_port صفرًا، فسيتم تعيينه إلى المنفذ الفعلي الذي يستمع عليه البرنامج البعيد (يتم استشارة خدمة portmap البعيدة لهذه المعلومات). الوسيط sockp هو مقبس؛ إذا كان RPC_ANYSOCK، يقوم هذا الروتين بفتحه وتعيين sockp. آلية نقل UDP تعيد إرسال رسالة الاستدعاء بفواصل زمنية قدرها wait حتى يتم استلام استجابة أو حتى ينتهي وقت الاستدعاء. يتم تحديد الوقت الإجمالي لانتهاء وقت الاستدعاء بواسطة clnt_call().
هذا يسمح للمستخدم بتحديد الحد الأقصى لحجم الحزمة لإرسال واستقبال رسائل RPC المستندة إلى UDP.
void get_myaddress(struct sockaddr_in *addr);
يضع عنوان IP للجهاز في *addr، دون استشارة روتينيات المكتبة التي تتعامل مع /etc/hosts. يتم دائمًا تعيين رقم المنفذ إلى htons(PMAPPORT).
struct pmaplist *pmap_getmaps(struct sockaddr_in *addr);
واجهة مستخدم لخدمة portmap، والتي تعيد قائمة بخرائط البرامج RPC الحالية إلى المنافذ على المضيف الموجود في عنوان IP *addr. يمكن لهذا الروتين أن يعيد NULL. الأمر rpcinfo -p يستخدم هذا الروتين.
unsigned short pmap_getport(struct sockaddr_in *addr,
                    unsigned long prognum, unsigned long versnum,
                    unsigned int protocol);
واجهة مستخدم لخدمة portmap، والتي تعيد رقم المنفذ الذي تنتظر عليه خدمة تدعم رقم البرنامج prognum، الإصدار versnum، وتتحدث بروتوكول النقل المرتبط بـ protocol. قيمة protocol هي على الأرجح IPPROTO_UDP أو IPPROTO_TCP. قيمة الإرجاع صفر تعني أن الخريطة غير موجودة أو أن نظام RPC فشل في الاتصال بخدمة portmap البعيدة. في الحالة الأخيرة، تحتوي المتغير العام rpc_createerr على حالة RPC.
enum clnt_stat pmap_rmtcall(struct sockaddr_in *addr,
                    unsigned long prognum, unsigned long versnum,
                    unsigned long procnum,
                    xdrproc_t inproc, char *in,
                    xdrproc_t outproc, char *out,
                    struct timeval tout, unsigned long *portp);
واجهة مستخدم لخدمة portmap، والتي توجه خدمة portmap على المضيف في عنوان IP *addr لإجراء استدعاء RPC نيابة عنك لإجراء على هذا المضيف. سيتم تعديل الوسيط *portp إلى رقم منفذ البرنامج إذا نجح الإجراء. يتم مناقشة تعريفات الوسائط الأخرى في callrpc() و clnt_call(). يجب استخدام هذا الإجراء لـ "ping" وليس لأي شيء آخر. انظر أيضًا clnt_broadcast().
bool_t pmap_set(unsigned long prognum, unsigned long versnum,
                int protocol, unsigned short port);
واجهة مستخدم لخدمة portmap، والتي تنشئ خريطة بين الثلاثي [prognum,versnum,protocol] و port على خدمة portmap للجهاز. قيمة protocol هي على الأرجح IPPROTO_UDP أو IPPROTO_TCP. يعيد هذا الروتين واحدًا إذا نجح، وصفرًا إذا فشل. يتم ذلك تلقائيًا بواسطة svc_register().
bool_t pmap_unset(unsigned long prognum, unsigned long versnum);
واجهة مستخدم لخدمة portmap، والتي تدمر جميع الخرائط بين الثلاثي [prognum,versnum,*] و المنافذ على خدمة portmap للجهاز. يعيد هذا الروتين واحدًا إذا نجح، وصفرًا إذا فشل.
int registerrpc(unsigned long prognum, unsigned long versnum,
                unsigned long procnum, typeof(char *(char *)) *procname,
                xdrproc_t inproc, xdrproc_t outproc);
تسجيل الإجراء procname مع حزمة خدمة RPC. إذا وصل طلب للبرنامج prognum، الإصدار versnum، والإجراء procnum، يتم استدعاء procname مع مؤشر إلى وسيطه (وسائطها)؛ يجب أن يعيد procname مؤشرًا إلى نتائجه الثابتة؛ يستخدم inproc لفك تشفير الوسائط بينما يستخدم outproc لتشفير النتائج. يعيد هذا الروتين صفرًا إذا نجح التسجيل، -1 بخلاف ذلك.
تحذير: يتم الوصول إلى الإجراءات البعيدة المسجلة بهذا الشكل باستخدام آلية نقل UDP/IP
