pmLookupDesc(3)

الحزمة: pcp

التراخيص: GPL-2.0-or-later، LGPL-2.1-or-later، CC-BY-3.0

PMLOOKUPDESC(3) دليل دوال المكتبة PMLOOKUPDESC(3)

الاسم

pmLookupDesc، pmLookupDescs - الحصول على أوصاف مقاييس الأداء

ملخص C

#include <pcp/pmapi.h>

int pmLookupDesc(pmID pmid, pmDesc *desc);
int pmLookupDescs(int numpmid, pmID *pmids, pmDesc *descs);

cc ... -lpcp

الوصف

بإعطاء معرّف مقياس الأداء (PMID) كـ pmid، تقوم روتين pmLookupDesc بملء بنية pmDesc المعطاة، المشار إليها بالمعامل desc، من سياق واجهة برمجة تطبيقات مقاييس الأداء (PMAPI) الحالي.

يوفر البديل pmLookupDescs وظائف مكافئة لـ numpmid من المقاييس دفعة واحدة، مع توفير مصفوفة pmids لمعرّفات المقاييس المراد البحث عنها. وهو أكثر كفاءة مع زيادة عدد المقاييس، لأنه يتجنب زمن الاستجابة ذهابًا وإيابًا من الطلبات الفردية المتعددة. لاحظ أن بروتوكول الخطأ يضمن وجود علاقة 1:1 بين عناصر descs وpmids، وبالتالي تحتوي كلتا القائمتين على numpmid من العناصر بالضبط. لهذا السبب، يُتوقع من المستدعي أن يكون قد خصص مسبقًا مصفوفة بالحجم المناسب لـ descs.

توفر بنية pmDesc جميع المعلومات المطلوبة لوصف مقياس أداء ومعالجته عبر PMAPI، ولها التعريف التالي.

/* واصف مقياس الأداء */
typedef struct {
    pmID    pmid;   /* معرّف فريد */
    int     type;   /* نوع البيانات الأساسي (انظر أدناه) */
    pmInDom indom;  /* نطاق المثيل */
    int     sem;    /* دلالات القيمة (انظر أدناه) *
    pmUnits units;  /* البُعد والوحدات (انظر أدناه) */
} pmDesc;
/* pmDesc.type -- نوع بيانات قيم المقياس */
#define PM_TYPE_NOSUPPORT        -1    /* غير مُنفّذ في هذا الإصدار */
#define PM_TYPE_32               0    /* عدد صحيح موقّع 32 بت */
#define PM_TYPE_U32              1    /* عدد صحيح غير موقّع 32 بت */
#define PM_TYPE_64               2    /* عدد صحيح موقّع 64 بت */
#define PM_TYPE_U64              3    /* عدد صحيح غير موقّع 64 بت */
#define PM_TYPE_FLOAT            4    /* نقطة عائمة 32 بت */
#define PM_TYPE_DOUBLE           5    /* نقطة عائمة 64 بت */
#define PM_TYPE_STRING           6    /* مصفوفة من char */
#define PM_TYPE_AGGREGATE        7    /* بيانات ثنائية عشوائية */
#define PM_TYPE_AGGREGATE_STATIC 8    /* مؤشر ثابت إلى التجميع */
#define PM_TYPE_EVENT            9    /* pmEventArray معبأة */
#define PM_TYPE_UNKNOWN          255  /* يُستخدم في pmValueBlock، وليس pmDesc */
/* pmDesc.sem -- دلالات/تفسير قيم المقياس */
#define PM_SEM_COUNTER  1  /* عدّاد تراكمي (زيادة رتيبة) */
#define PM_SEM_INSTANT  3  /* قيمة لحظية بنطاق مستمر */
#define PM_SEM_DISCRETE 4  /* قيمة لحظية بنطاق منفصل */

يصف حقل type في pmDesc ترميزات (أو تنسيقات) مختلفة لقيمة المقياس.

إذا تم عدّ قيمة في الأجهزة الأساسية بمستوى دقة عدد صحيح أقل من 32 بت، فإنه تقع على عاتق وكيل نطاق مقاييس الأداء (PMDA) مسؤولية ترقية القيمة إلى عدد صحيح 32 بت قبل تصديرها إلى نظام جمع مقاييس الأداء (PMCS)؛ أي أن التطبيقات فوق PMAPI لا تضطر أبدًا للتعامل مع عدّادات 8 بت و16 بت.

إذا كانت قيمة مقياس الأداء من النوع PM_TYPE_AGGREGATE أو PM_TYPE_AGGREGATE_STATIC، أو PM_TYPE_EVENT أو PM_TYPE_STRING، فإن تفسير القيمة غير معروف لـ PMCS. في هذه الحالات، يجب أن يكون لدى التطبيق الذي يستخدم القيمة وPMDA الذي يوفر القيمة فهم مشترك حول كيفية بنية القيمة وتفسيرها.

يُفترض أن كل قيمة لمقياس أداء مأخوذة من مجموعة من القيم التي يمكن وصفها من حيث أبعادها ومقياسها بترميز مضغوط كما يلي. يُعرّف البُعد بواسطة قوة، أو أس، في كل من 3 أبعاد متعامدة، وهي المساحة والزمن والعدد (أو الأحداث، وهي بلا أبعاد). على سبيل المثال، يمكن تمثيل إنتاجية الإدخال/الإخراج كـ           -1 المساحة.الزمن
بينما الإجمالي الجاري لاستدعاءات النظام هو العدد، وتخصيص الذاكرة هو المساحة ومتوسط زمن الخدمة هو           -1 الزمن.العدد
في كل بُعد هناك عدد من قيم المقياس الشائعة التي يمكن استخدامها لترميز النطاقات بشكل أفضل والتي قد تستنفد خلاف ذلك دقة قيمة 32 بت. تُرمّز هذه المعلومات في بنية pmUnits المضمّنة في بنية pmDesc.

/*
 * ترميز الوحدات (أبعاد الزمن والمساحة) والمقياس
 * لقيم مقاييس الأداء
 *
 * على سبيل المثال، بنية pmUnits بقيمة
 *      { 1, -1, 0, PM_SPACE_MBYTE, PM_TIME_SEC, 0 }
 * تمثل ميغابايت/ثانية، بينما
 *      { 0, 1, -1, 0, PM_TIME_HOUR, 6 }
 * تمثل ساعات/مليون حدث
 */
typedef struct {
    int dimSpace:4;             /* بُعد المساحة */
    int dimTime:4;              /* بُعد الزمن */
    int dimCount:4;             /* بُعد الحدث */
    unsigned int scaleSpace:4;  /* أحد PM_SPACE_* أدناه */
    unsigned int scaleTime:4;   /* أحد PM_TIME_* أدناه */
    int scaleCount:4;           /* أحد PM_COUNT_* أدناه */
} pmUnits;                      /* الوحدات البُعدية ومقياس القيمة */
/* pmUnits.scaleSpace */
#define PM_SPACE_BYTE   0       /* بايتات */
#define PM_SPACE_KBYTE  1       /* كيلوبايت (1024) */
#define PM_SPACE_MBYTE  2       /* ميغابايت (1024^2) */
#define PM_SPACE_GBYTE  3       /* غيغابايت (1024^3) */
#define PM_SPACE_TBYTE  4       /* تيرابايت (1024^4) */
#define PM_SPACE_PBYTE	5	/* بيبي بايت (1024^5) */
#define PM_SPACE_EBYTE	6	/* إكسبي بايت (1024^6) */
#define PM_SPACE_ZBYTE	7	/* زيبي بايت (1024^7) */
#define PM_SPACE_YBYTE	8	/* يوبي بايت (1024^8) */
/* pmUnits.scaleTime */
#define PM_TIME_NSEC    0       /* نانوثانية */
#define PM_TIME_USEC    1       /* ميكروثانية */
#define PM_TIME_MSEC    2       /* ميلي ثانية */
#define PM_TIME_SEC     3       /* ثانية */
#define PM_TIME_MIN     4       /* دقيقة */
#define PM_TIME_HOUR    5       /* ساعة */
/*
 * pmUnits.scaleCount (مثل عدّ الأحداث، استدعاءات النظام، المقاطعات،
 * إلخ) هذه ببساطة قوى للعدد 10، وليست مُعدّدة هنا،
 * مثل 6 لـ 10^6، أو -3 لـ 10^-3
 */
#define PM_COUNT_ONE    0       /* 1 */

تُوفَّر روتينات خاصة (مثل pmExtractValue(3)، pmConvScale(3)) لمعالجة القيم بالاقتران مع بنية pmUnits التي تحدد بُعد ومقياس القيم لمقياس أداء معين.

أسفل PMAPI، تُجلب المعلومات المطلوبة لإكمال بنية pmDesc من وكلاء PMDAs، وبهذه الطريقة يمكن أن يتغير تنسيق ومقياس مقاييس الأداء ديناميكيًا، مع تطور وكلاء PMDAs وأجهزتهم الأساسية مع مرور الوقت. على وجه الخصوص، عندما تصبح بعض المقاييس فجأة بطول 64 بت، أو تغيّر وحداتها من ميغابايت إلى غيغابايت، ستستمر التطبيقات المكتوبة جيدًا التي تستخدم الخدمات التي توفرها PMAPI في العمل بشكل صحيح.

التشخيصات

تُرجع هذه الروتينات رمز خطأ سالب للإشارة إلى الفشل.

PM_ERR_PMID
PMID المطلوب غير معروف لـ PMCS
PM_ERR_NOAGENT
وكيل PMDA المسؤول عن توفير المقياس غير متاح حاليًا

تُرجع pmLookupDesc صفرًا للإشارة إلى النجاح.

تعتمد النتيجة من pmLookupDescs على وجود أي حالات فشل في البحث وخطورتها وعدد المقاييس التي يتم البحث عنها.

1.
إذا لم تكن هناك حالات فشل في البحث، ستكون القيمة المُرجعة numpmid.
2.
إذا تمت مواجهة خطأ قاتل، ستكون القيمة المُرجعة أقل من 0. على سبيل المثال PM_ERR_IPC.
3.
إذا كانت numpmid أكبر من واحد وتمت مواجهة خطأ (أخطاء) غير قاتلة، فإن القيمة المُرجعة هي عدد واصفات المقاييس التي تم البحث عنها بنجاح (أكبر من أو يساوي صفر وأقل من أو يساوي numpmid).
4.
إذا كانت numpmid تساوي واحدًا وتمت مواجهة خطأ غير قاتل، فإن القيمة المُرجعة هي رمز الخطأ (أقل من صفر).

عند مواجهة أخطاء، تؤدي أي مقاييس لا يمكن البحث عنها إلى تعيين حقل pmid في عنصر الواصف المقابل من descs إلى PM_ID_NULL. يسمح بروتوكول الخطأ المعقد قليلاً بعمليات البحث المجمعة، ثم التحقق من مزيد من تفاصيل الخطأ في حالة فشل معين.
