IBV_ALLOC_DM(3) دليل مبرمج Libibverbs IBV_ALLOC_DM(3)

الاسم

ibv_alloc_dm, ibv_free_dm, ibv_memcpy_to/from_dm - تخصيص أو تحرير مخزن مؤقت لذاكرة الجهاز (DMs) وإجراء نسخ الذاكرة إليه أو منها

ملخص

#include <infiniband/verbs.h>

struct ibv_dm *ibv_alloc_dm(struct ibv_context *context,
                            struct ibv_alloc_dm_attr *attr);

int ibv_free_dm(struct ibv_dm *dm);

الوصف

ibv_alloc_dm() يقوم بتخصيص مخزن مؤقت لذاكرة الجهاز لسياق جهاز RDMA context. الوسيط attr هو مؤشر إلى هيكل ibv_alloc_dm_attr، كما هو محدد في <infiniband/verbs.h>.

ibv_free_dm() يقوم بتحرير مخزن مؤقت لذاكرة الجهاز dm.

struct ibv_alloc_dm_attr {
size_t length;			/* طول مخزن ذاكرة الجهاز المطلوب */
uint32_t log_align_req;		/* سجل الأساس 2 لمتطلب محاذاة العنوان */
uint32_t comp_mask;		/* قناع التوافق الذي يحدد المتغيرات التالية صالحة */
};
قد تكون محاذاة العنوان مطلوبة في الحالات التي سيتم فيها إجراء عمليات RDMA الذرية باستخدام ذاكرة الجهاز.
في مثل هذه الحالات، قد يحدد المستخدم محاذاة عنوان بداية ذاكرة الجهاز باستخدام المعلمة log_align_req
في هيكل سمات التخصيص.

الوصول إلى ذاكرة جهاز مخصصة

لإجراء وصول كتابة/قراءة إلى ذاكرة جهاز مخصصة، يمكن للمستخدم استخدام استدعاءات ibv_memcpy_to_dm
و ibv_memcpy_from_dm على التوالي.

int ibv_memcpy_to_dm(struct ibv_dm *dm, uint64_t dm_offset,
                     void *host_addr, size_t length);

int ibv_memcpy_from_dm(void *host_addr, struct ibv_dm *dm,
			    uint64_t dm_offset, size_t length);

dm_offest
هو الإزاحة بالبايت من بداية مخزن ذاكرة الجهاز المخصص للوصول إليه.

host_addr
هو عنوان مخزن ذاكرة المضيف للوصول إليه.

length
هو طول النسخ بالبايت.

تسجيل ذاكرة الجهاز

قد يقوم المستخدم بتسجيل ذاكرة الجهاز المخصصة كمنطقة ذاكرة واستخدام lkey/rkey داخل sge عند نشر طلب استلام أو إرسال. يتم تعريف هذا النوع من MR على أنه يبدأ من الصفر، وبالتالي فإن أي مرجع إليه (خاصة في sge) يتم بإزاحة بالبايت من بداية المنطقة.

يتم هذا النوع من التسجيل باستخدام ibv_reg_dm_mr.

struct ibv_mr* ibv_reg_dm_mr(struct ibv_pd *pd, struct ibv_dm *dm, uint64_t dm_offset,
                  size_t length, uint32_t access);

pd
هو pd المرتبط بهذا التسجيل.

dm
هو dm المرتبط بهذا التسجيل.

dm_offest
هو الإزاحة بالبايت من بداية مخزن ذاكرة الجهاز المخصص للتسجيل.

length
هو طول الذاكرة المراد تسجيلها.

access
علامات وصول mr (استخدم تعداد ibv_access_flags). لهذا النوع من التسجيل، يجب على المستخدم تعيين علامة IBV_ACCESS_ZERO_BASED.

قيمة الإرجاع

ibv_alloc_dm() تعيد مؤشرًا إلى هيكل ibv_dm أو NULL إذا فشل الطلب. يحتوي dm الناتج على المقبض الذي يمكن للمستخدم استخدامه لاستيراد ذاكرة الجهاز هذه.

ibv_free_dm() تعيد 0 عند النجاح، أو قيمة errno عند الفشل (والتي تشير إلى سبب الفشل).

ibv_reg_dm_mr() تعيد مؤشرًا إلى هيكل ibv_mr عند النجاح أو NULL إذا فشل الطلب.

ibv_memcpy_to_dm()/ibv_memcpy_from_dm() تعيد 0 عند النجاح أو قيمة سبب الفشل عند الفشل.

ملاحظات

ibv_alloc_dm() قد يفشل إذا لم تعد ذاكرة الجهاز المتاحة كافية، حيث يتم توفير الحد الأقصى لمقدار الذاكرة المخصصة بواسطة السمة max_dm_size في هيكل ibv_device_attr_ex. ibv_free_dm() قد تفشل إذا كانت هناك موارد أخرى (مثل MR) لا تزال مرتبطة بـ DM الذي يتم تحريره.
