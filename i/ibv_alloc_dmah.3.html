ibv_alloc_dmah(3) دليل مبرمج Libibverbs ibv_alloc_dmah(3)

الاسم

ibv_alloc_dmah - تخصيص مقبض dma

int ibv_dealloc_dmah - إلغاء تخصيص مقبض dma

ملخص

#include <infiniband/verbs.h>
struct ibv_dmah *ibv_alloc_dmah(struct ibv_context *context, struct ibv_dmah_init_attr *attr);
int ibv_dealloc_dmah(struct ibv_dmah *dmah);

الوصف

ibv_alloc_dmah() تقوم بتخصيص كائن ibv_dmah المرتبط بـ context المعطى ومعامل attr المدخل.

يمكن استخدام المقبض المخصص لاحقًا لتحسين عمليات DMA و RDMA المرتبطة بمنطقة ذاكرة مسجلة.

بمجرد انتهاء استخدام ibv_dmah، يجب استدعاء ibv_dealloc_dmah().

سيقوم هذا الاستدعاء بتحرير الموارد التي تم تخصيصها سابقًا باستخدام الواجهة البرمجية ibv_alloc_dmah().

المعاملات

attr

enum ibv_tph_mem_type {
    IBV_TPH_MEM_TYPE_VM, /* ذاكرة متطايرة */
    IBV_TPH_MEM_TYPE_PM, /* ذاكرة دائمة */
};
enum ibv_dmah_init_attr_mask {
    IBV_DMAH_INIT_ATTR_MASK_CPU_ID = 1 << 0,
    IBV_DMAH_INIT_ATTR_MASK_PH = 1 << 1,
    IBV_DMAH_INIT_ATTR_MASK_TPH_MEM_TYPE = 1 << 2,
};
struct ibv_dmah_init_attr {
    uint32_t comp_mask; /* من ibv_dmah_init_attr_mask */
    uint32_t cpu_id;
    uint8_t ph;
    uint8_t tph_mem_type; /* من enum ibv_tph_mem_type */
};
comp_mask
قناع بت يحدد ما هي الحقول في البنية صالحة.
cpu_id
معرف وحدة المعالجة المركزية التي يشير إليها مقبض dma.
ph
تلميحات المعالجة، تستخدم للمساعدة في تحسين معالجة المعاملات عبر PCIe.
tph_mem_type
نوع الذاكرة المستهدفة، أحد أنواع enum ibv_tph_mem_type.

قيمة الإرجاع

ibv_alloc_dmah() ترجع مؤشرًا إلى كائن مقبض dma المخصص، أو NULL إذا فشل الطلب.

ibv_dealloc_dmah() ترجع 0 عند النجاح، وإلا قيمة errno.
