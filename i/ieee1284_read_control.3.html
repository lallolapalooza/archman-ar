ieee1284_read_control(3)

الحزمة: libieee1284

التراخيص: GPL-2.0-or-later

IEEE1284_CONTROL(3) الوظائف IEEE1284_CONTROL(3)

الاسم

ieee1284_read_control, ieee1284_write_control, ieee1284_frob_control, ieee1284_do_nack_handshake - معالجة خطوط التحكم

الملخص

#include <ieee1284.h>

int ieee1284_read_control(struct parport *port);

void ieee1284_write_control(struct parport *port, unsigned char ct);

void ieee1284_frob_control(struct parport *port, unsigned char mask, unsigned char val);

int ieee1284_do_nack_handshake(struct parport *port, unsigned char ct_before, unsigned char ct_after, struct timeval *timeout);

الوصف

هناك أربعة خطوط تحكم، ثلاثة منها عادة ما تكون معكوسة في المنافذ من طراز PC. حيث تختلف، تعمل libieee1284 على قيم IEEE 1284، وليس القيم المعكوسة من طراز PC. يتم تمثيل خطوط التحكم من خلال التعداد التالي:

enum ieee1284_control_bits
{
  C1284_NSTROBE   = 0x01,
  C1284_NAUTOFD   = 0x02,
  C1284_NINIT     = 0x04,
  C1284_NSELECTIN = 0x08,
  /* لتحويل تلك القيم إلى قيم مسجل من طراز PC، استخدم هذا: */
  C1284_INVERTED = (C1284_NSTROBE|
                    C1284_NAUTOFD|
                    C1284_NSELECTIN),
};

تعمل هذه الوظائف جميعها على المنفذ المتوازي المرتبط بـ port، والذي يجب أن يكون مستخدمًا.

القيم الحالية على خطوط التحكم متاحة عن طريق استدعاء ieee1284_read_control، ويمكن تعيينها عن طريق استدعاء ieee1284_write_control.

لتعديل القيم على مجموعة من خطوط التحكم، استخدم ieee1284_frob_control. يمكن التعبير عن تأثير ذلك بـ: ctr = ((ctr & ~mask) ^ val)؛ أي، يتم إلغاء تعيين البتات في mask، ثم يتم عكس البتات في val.

الوظيفة الخاصة ieee1284_do_nack_handshake مخصصة للاستجابة بسرعة فائقة في بروتوكول حيث يقوم الجهاز الطرفي بتعيين nAck ويجب على المضيف الاستجابة عن طريق تعيين خط تحكم. عملها، الذي يعتمد على معرفة آلة المضيف بالمقاطعة التي يولدها nAck، هو كما يلي:

1. تعيين خطوط التحكم كما هو موضح في ct_before.
2. انتظار مقاطعة nAck. إذا انتهت مهلة timeout، قم بإرجاع E1284_TIMEDOUT.
3. تعيين خطوط التحكم كما هو موضح في ct_after.

على نظام Linux باستخدام مشغل ppdev، يتم تنفيذ ذلك بواسطة مشغل الجهاز في النواة، وبالتالي فهو أسرع مما هو ممكن عادة في مكتبة مساحة المستخدم.

قيمة الإرجاع

قيمة إرجاع ieee1284_read_control، إذا كانت غير سالبة، هي رقم يمثل القيمة الموجودة على خطوط التحكم.

رموز الخطأ المحتملة لـ ieee1284_read_control:

E1284_NOTAVAIL

خطوط التحكم لهذا المنفذ غير قابلة للوصول بواسطة التطبيق.

E1284_INVALIDPORT

المعلمة port غير صالحة (على سبيل المثال، ربما لم يتم طلبها).

رموز الخطأ المحتملة لـ ieee1284_do_nack_handshake:

E1284_OK

تمت المصافحة بنجاح.

E1284_NOTAVAIL

هذه العملية غير متوفرة على هذا النوع من المنافذ أو النظام. قد يكون ذلك بسبب عدم توفر مقاطعات المنفذ، أو لأن مشغل الجهاز الأساسي لا يدعم العملية.

E1284_INVALIDPORT

المعلمة port غير صالحة (على سبيل المثال، ربما لم يتم طلبها).
