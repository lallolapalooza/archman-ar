integer function ilaslr (integer m, integer n, real, dimension( lda, * ) a, integer lda)

ILASLR تبحث في مصفوفة عن آخر صف غير صفري.

الغرض:

!>
!> ILASLR تبحث في A عن آخر صف غير صفري.
!>

المعاملات

M 
!>          M هو عدد صحيح (INTEGER)
!>          عدد صفوف المصفوفة A.
!>

N

!>          N هو عدد صحيح (INTEGER)
!>          عدد أعمدة المصفوفة A.
!>

A

!>          A هو مصفوفة حقيقية (REAL array)، البعد (LDA,N)
!>          المصفوفة A ذات البعد m × n.
!>

LDA

!>          LDA هو عدد صحيح (INTEGER)
!>          البعد الرئيسي للمصفوفة A. LDA >= max(1,M).
!>
