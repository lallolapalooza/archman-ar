توثيق الوظيفة/الإجراء الفرعي

integer function iladlc (integer m, integer n, double precision, dimension( lda, * ) a, integer lda)

ILADLC يبحث في المصفوفة عن آخر عمود غير صفري.

الغرض:

!>
!> ILADLC يبحث في A عن آخر عمود غير صفري.
!>

المعلمات

M 
!>          M هو عدد صحيح (INTEGER)
!>          عدد صفوف المصفوفة A.
!>

N

!>          N هو عدد صحيح (INTEGER)
!>          عدد أعمدة المصفوفة A.
!>

A

!>          A هي مصفوفة من النوع double precision، البعد (LDA,N)
!>          المصفوفة A بحجم m في n.
!>

LDA

!>          LDA هو عدد صحيح (INTEGER)
!>          البعد الرائد للمصفوفة A. LDA >= max(1,M).
!>
