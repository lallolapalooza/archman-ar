IBND_DISCOVER_FABRIC(3) دليل مبرمج OpenIB IBND_DISCOVER_FABRIC(3)

الاسم

ibnd_discover_fabric, ibnd_destroy_fabric, ibnd_debug ibnd_show_progress - تهيئة مكتبة ibnetdiscover.

الملخص

#include <infiniband/ibnetdisc.h>

ibnd_fabric_t *ibnd_discover_fabric(struct ibmad_port *ibmad_port, int timeout_ms, ib_portid_t *from, int hops)
void ibnd_destroy_fabric(ibnd_fabric_t *fabric)
void ibnd_debug(int i)
void ibnd_show_progress(int i)
int ibnd_set_max_smps_on_wire(int i)

الوصف

ibnd_discover_fabric() تكتشف الشبكة المتصلة بالمنفذ المحدد بواسطة ibmad_port، باستخدام مهلة زمنية محددة. المعلمات "from" و "hops" اختيارية وتسمح للمستخدم بمسح جزء من الشبكة عن طريق تحديد عقدة "from" وعدد القفزات "hops" بعيدًا عن تلك العقدة. هذا يمنح المستخدم "شبكة فرعية" "متمركزة" في أي مكان يختاره.

يجب فتح ibmad_port باستخدام فئات IB_SMI_CLASS و IB_SMI_DIRECT_CLASS على الأقل لكي تعمل ibnd_discover_fabric.

ibnd_destroy_fabric() تحرر جميع الذاكرة والموارد المرتبطة بالشبكة.

ibnd_debug() تحدد مستوى التصحيح الذي سيتم طباعته أثناء عمليات المكتبة.

ibnd_show_progress() تشير إلى أن المكتبة يجب أن تطبع ناتج تصحيح يوضح تقدمها عبر الشبكة.

ibnd_set_max_smps_on_wire() تحدد عدد SMPs التي سيتم إصدارها على السلك بشكل متزامن.

قيمة الإرجاع

ibnd_discover_fabric() ترجع NULL عند الفشل، وإلا ترجع كائن ibnd_fabric_t صالح.

ibnd_destory_fabric(), ibnd_debug() لا شيء.

ibnd_set_max_smps_on_wire() يتم إرجاع القيمة السابقة.

أمثلة

اكتشاف الشبكة بأكملها المتصلة بالجهاز mthca0، المنفذ 1.

int mgmt_classes[2] = {IB_SMI_CLASS, IB_SMI_DIRECT_CLASS}; struct ibmad_port *ibmad_port = mad_rpc_open_port(ca, ca_port, mgmt_classes, 2); ibnd_fabric_t *fabric = ibnd_discover_fabric(ibmad_port, 100, NULL, 0); ... ibnd_destroy_fabric(fabric); mad_rpc_close_port(ibmad_port);

اكتشاف عقدة واحدة فقط والعقد المتصلة بها.

... str2drpath(&(port_id.drpath), from, 0, 0); ... ibnd_discover_fabric(ibmad_port, 100, &port_id, 1); ...
