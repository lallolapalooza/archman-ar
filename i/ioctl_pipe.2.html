ioctl_pipe(2) دليل استدعاءات النظام ioctl_pipe(2)

الاسم

ioctl_pipe - عمليات ioctl() لآلية الإشعارات العامة

الملخص

#include <linux/watch_queue.h>  /* تعريف IOC_WATCH_QUEUE_* */
#include <sys/ioctl.h>
int ioctl(int pipefd, IOC_WATCH_QUEUE_SET_SIZE, int size);
int ioctl(int pipefd, IOC_WATCH_QUEUE_SET_FILTER,
          struct watch_notification_filter *filter);

الوصف

يتم توفير عمليات ioctl(2) التالية لإعداد معلمات طابور الإشعارات العام. يعتمد طابور الإشعارات على pipe(2) مفتوح بعلامة O_NOTIFICATION_PIPE.

IOC_WATCH_QUEUE_SET_SIZE (منذ Linux 5.8)
يقوم بتخصيص ذاكرة التخزين المؤقت للأنبوب مسبقًا بحيث يمكن أن تستوعب size رسالة إشعار. حاليًا، يجب أن يكون size بين 1 و 512.
IOC_WATCH_QUEUE_SET_FILTER (منذ Linux 5.8)
يمكن لمرشح طابور المشاهدة الحد من الأحداث المستلمة. يتم تمرير المرشحات في struct watch_notification_filter ويتم وصف كل مرشح بواسطة بنية struct watch_notification_type_filter.

struct watch_notification_filter {
        __u32   nr_filters;
        __u32   __reserved;
        struct watch_notification_type_filter filters[];
};
struct watch_notification_type_filter {
        __u32   type;
        __u32   info_filter;
        __u32   info_mask;
        __u32   subtype_filter[8];
};
