ieee1284_wait_status(3)

الحزمة: libieee1284

التراخيص: GPL-2.0-or-later

IEEE1284_STATUS(3) وظائف IEEE1284_STATUS(3)

الاسم

ieee1284_read_status, ieee1284_wait_status - تحليل خطوط الحالة

التركيب

#include <ieee1284.h>

int ieee1284_read_status(struct parport *port);

int ieee1284_wait_status(struct parport *port, unsigned char mask, unsigned char val, struct timeval *timeout);

الوصف

هناك خمسة خطوط حالة، أحدها عادة ما يكون معكوسًا على منافذ نمط PC. حيث يختلفان، تعمل مكتبة libieee1284 على قيم IEEE 1284، وليس قيم نمط PC المعكوسة. يتم تمثيل خطوط الحالة بالتعداد التالي:

enum ieee1284_status_bits
{
  S1284_NFAULT = 0x08,
  S1284_SELECT = 0x10,
  S1284_PERROR = 0x20,
  S1284_NACK   = 0x40,
  S1284_BUSY   = 0x80,
  /* لتحويل تلك القيم إلى قيم سجلات نمط PC، استخدم هذا: */
  S1284_INVERTED = S1284_BUSY,
};

تعمل كل هذه الوظائف على المنفذ المتوازي المرتبط بـ port، والذي يجب أن يكون محجوزًا.

الغرض من ieee1284_wait_status هو الانتظار حتى تحصل خطوط الحالة المعينة على القيم المحددة. قد يتم تعديل معلمة timeout عند الإرجاع.

قيمة الإرجاع

بالنسبة لـ ieee1284_read_status، القيمة المرجعة هي عدد صحيح غير سالب مع تعيين البتات حسب الاقتضاء لتمثيل خطوط الحالة. تشير النتيجة السالبة إلى خطأ.

بالنسبة لـ ieee1284_wait_status، القيمة المرجعة هي E1284_OK إذا كانت خطوط الحالة تعكس الآن القيم المطلوبة (أي status & mask هو val)، أو نتيجة سالبة تشير إلى خطأ.

رموز الخطأ الممكنة:

E1284_NOTIMPL

يفتقر port إلى القدرة المطلوبة. قد يكون هذا بسبب قيود هذا الإصدار من libieee1284، أو قيود الأجهزة.

E1284_NOTAVAIL

الوصول إلى خطوط الحالة غير متاح على هذا النوع من المنافذ.

E1284_TIMEDOUT

انتهت مهلة timeout.

E1284_INVALIDPORT

المعلمة port غير صالحة (على سبيل المثال، ربما لم يتم حجز port).

ملاحظات

غالبًا ما يكون دبوس nAck قادرًا على تشغيل مقاطعات على الجهاز المضيف. بمساعدة نظام التشغيل، قد تكون هذه المقاطعات مرئية للتطبيق عبر واصف الملف الذي تم إرجاعه بواسطة ieee1284_get_irq_fd.

في Linux، الشروط هي أن برنامج تشغيل المنفذ المتوازي يعرف سطر المقاطعة الذي يجب استخدامه ويستخدمه، وأن عقدة جهاز /dev/parport ذات الصلة يمكن الوصول إليها ومدعومة بواسطة برنامج تشغيل جهاز.
