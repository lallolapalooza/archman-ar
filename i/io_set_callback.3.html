io_set_callback(3)

الحزمة: libaio

التراخيص: LGPL-2.0-or-later

io_set_callback(3) Linux AIO io_set_callback(3)

الاسم

io_set_callback - تعيين دالة رد اتصال اكتمال io

ملخص

#include <errno.h>

#include <libaio.h>

static inline void io_set_callback(struct iocb *iocb, io_callback_t cb);

struct iocb {
	void		*data;
	unsigned	key;
	short		aio_lio_opcode;
	short		aio_reqprio;
	int		aio_fildes;
};

typedef void (*io_callback_t)(io_context_t ctx, struct iocb *iocb, long res, long res2);

الوصف

لا يتم استدعاء رد الاتصال إذا استخدم المستدعي الأحداث الخام من io_getevents(3)، بل فقط مع مساعدي المكتبة.
