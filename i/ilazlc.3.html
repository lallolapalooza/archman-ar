توثيق الوظيفة/الروتين الفرعي

integer function ilazlc (integer m, integer n, complex*16, dimension( lda, * ) a, integer lda)

ILAZLC تقوم بمسح مصفوفة بحثًا عن آخر عمود غير صفري.

الغرض:

!>
!> ILAZLC تقوم بمسح A بحثًا عن آخر عمود غير صفري.
!>

المعاملات

M 
!>          M هو عدد صحيح (INTEGER)
!>          عدد صفوف المصفوفة A.
!>

N

!>          N هو عدد صحيح (INTEGER)
!>          عدد أعمدة المصفوفة A.
!>

A

!>          A هي مصفوفة مركبة (COMPLEX*16) ، بأبعاد (LDA,N)
!>          المصفوفة A بحجم m × n.
!>

LDA

!>          LDA هو عدد صحيح (INTEGER)
!>          البعد الرائد للمصفوفة A. LDA >= max(1,M).
!>
