idmap_ldap(8)

الحزمة: samba

التراخيص: GPL-3.0-or-later

IDMAP_LDAP(8) أدوات إدارة النظام IDMAP_LDAP(8)

الاسم

idmap_ldap - الواجهة الخلفية لـ Samba idmap_ldap لـ Winbind

الوصف

توفر المكون الإضافي idmap_ldap وسيلة لـ Winbind لتخزين واسترداد جداول تعيين SID/uid/gid في خدمة دليل LDAP.

على عكس الواجهات الخلفية للقراءة فقط مثل idmap_rid، فهي واجهة خلفية مخصصة: هذا يعني أنها تحتاج إلى تخصيص معرفات مستخدمين ومجموعات جديدة لإنشاء تعيينات جديدة.

خيارات IDMAP

ldap_base_dn = DN

يحدد لاحقة قاعدة الدليل لاستخدامها لإدخالات تعيين SID/uid/gid. إذا لم يتم تحديدها، فسيفترض idmap_ldap استخدام خيار "ldap idmap suffix" من /etc/samba/smb.conf.

ldap_user_dn = DN

يحدد DN المستخدم للمصادقة. يجب تخزين سر المصادقة لهذا المستخدم باستخدام net idmap secret (انظر net(8)). إذا كان مفقودًا، يتم استخدام بيانات اعتماد LDAP من تكوين قاعدة بيانات passdb الخاصة بـ LDAP، وإذا كانت مفقودة أيضًا، فسيتم إجراء ربط مجهول كحل أخير.

ldap_url = ldap://server/

يحدد خادم LDAP لاستخدامه لإدخالات تعيين SID/uid/gid. إذا لم يتم تحديده، فسيفترض idmap_ldap استخدام ldap://localhost/.

range = low - high

يحدد النطاق المتاح لمعرفات المستخدم والمجموعة المطابقة التي تعد الواجهة الخلفية مسؤولة عنها.

أمثلة

يوضح المثال التالي كيفية استخدام دليل LDAP كواجهة خلفية افتراضية لـ idmap. كما أنه يقوم بتكوين نطاق idmap ولاحقة قاعدة الدليل. يجب تعيين السر لـ ldap_user_dn باستخدام "net idmap secret '*' password".

[global]
idmap config * : backend      = ldap
idmap config * : range        = 1000000-1999999
idmap config * : ldap_url     = ldap://localhost/
idmap config * : ldap_base_dn = ou=idmap,dc=example,dc=com
idmap config * : ldap_user_dn = cn=idmap_admin,dc=example,dc=com

يوضح هذا المثال كيف يمكن استخدام LDAP كواجهة خلفية للقراءة فقط بينما يستخدم tdb الواجهة الخلفية الافتراضية لتخزين التعيينات. يضيف تكوينًا صريحًا لبعض النطاقات DOM1، الذي يستخدم الواجهة الخلفية لـ idmap الخاصة بـ LDAP. لاحظ أنه يتم استخدام نطاق منفصل عن النطاق الافتراضي.

[global]
# "backend = tdb" زائد هنا لأنه الافتراضي
idmap config * : backend = tdb
idmap config * : range = 1000000-1999999
idmap config DOM1 : backend = ldap
idmap config DOM1 : range = 2000000-2999999
idmap config DOM1 : read only = yes
idmap config DOM1 : ldap_url = ldap://server/
idmap config DOM1 : ldap_base_dn = ou=idmap,dc=dom1,dc=example,dc=com
idmap config DOM1 : ldap_user_dn = cn=idmap_admin,dc=dom1,dc=example,dc=com

ملاحظة

لاستخدام المصادقة ضد خوادم LDAP، قد تحتاج إلى توفير DN وكلمة مرور. لتجنب كشف كلمة المرور كنص عادي في ملف التكوين، نقوم بتخزينها في مخزن أمان. يتم استخدام الأمر "net idmap" لتخزين سر لـ DN المحدد في نطاق idmap معين.
