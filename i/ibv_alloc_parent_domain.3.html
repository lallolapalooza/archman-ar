IBV_ALLOC_PARENT_DOMAIN(3) دليل مبرمج Libibverbs IBV_ALLOC_PARENT_DOMAIN(3)

الاسم

ibv_alloc_parent_domain(), ibv_dealloc_pd() - تخصيص وإلغاء تخصيص كائن النطاق الأصل

الملخص

#include <infiniband/verbs.h>

struct ibv_pd *ibv_alloc_parent_domain(struct ibv_context *context, struct ibv_parent_domain_init_attr *attr);

الوصف

ibv_alloc_parent_domain() يقوم بتخصيص كائن نطاق أصل لسياق جهاز RDMA context.

يقوم كائن النطاق الأصل بتوسيع نطاق الحماية العادي بكائنات إضافية، مثل نطاق الخيوط.

النطاق الأصل قابل للتبديل تمامًا مع struct ibv_pd المستخدم لإنشائه، ويمكن استخدامه كوسيط إدخال لأي دالة تقبل struct ibv_pd.

قد يختلف سلوك كل أمر إذا تم تمرير نطاق أصل struct ibv_pd يحتوي على مؤشر struct ibv_td. على سبيل المثال، قد يختار الأمر مشاركة الموارد بين الكائنات التي تستخدم نفس نطاق الخيوط. السلوك الدقيق يعتمد على المزود.

الوسيط attr يحدد ما يلي:

enum ibv_parent_domain_init_attr_mask {
IBV_PARENT_DOMAIN_INIT_ATTR_ALLOCATORS = 1 << 0,
IBV_PARENT_DOMAIN_INIT_ATTR_PD_CONTEXT = 1 << 1,
};
struct ibv_parent_domain_init_attr {
struct ibv_pd *pd; /* مرجع لنطاق الحماية، لا يمكن أن يكون NULL */
struct ibv_td *td; /* مرجع لنطاق الخيوط، أو NULL */
uint32_t comp_mask;
void *(*alloc)(struct ibv_pd *pd, void *pd_context, size_t size,
               size_t alignment, uint64_t resource_type);
void (*free)(struct ibv_pd *pd, void *pd_context, void *ptr,
             uint64_t resource_type);
void *pd_context;
};

ibv_dealloc_pd() سيقوم بإلغاء تخصيص النطاق الأصل حيث يتم الكشف عنه كـ pd ibv_. يجب تدمير جميع الموارد التي تم إنشاؤها باستخدام النطاق الأصل قبل إلغاء تخصيص النطاق الأصل.

الوسائط

pd مرجع لنطاق الحماية الذي يستخدمه هذا النطاق الأصل.

td نطاق خيوط اختياري يستخدمه النطاق الأصل.

comp_mask قناع بت للحقول الاختيارية في هيكل ibv_parent_domain_init_attr.

alloc وظيفة مخصصة لتخصيص الذاكرة لهذا النطاق الأصل. ستستخدم تخصيصات ذاكرة المزود هذه الوظيفة لتخصيص الذاكرة المطلوبة. يتم تمرير وظيفة التخصيص إلى النطاق الأصل pd والسياق المحدد من قبل المستخدم pd_context. بالإضافة إلى ذلك، تستقبل الاستدعاء size و alignment للمخزن المؤقت المطلوب، بالإضافة إلى resource_type خاص بالبائع، والذي يتم اشتقاقه من تعداد rdma_driver_id (أعلى 32 بت) ورمز مورد خاص بالبائع. تقوم الوظيفة بإرجاع المؤشر إلى المخزن المؤقت المخصص، أو NULL للإشارة إلى خطأ. قد تعيد أيضًا IBV_ALLOCATOR_USE_DEFAULT لتطلب من المستدعي تخصيص المخزن المؤقت باستخدام المخصص الافتراضي.

تتأكد الاستدعاء من تهيئة المخزن المؤقت المخصص بالأصفار. تقع على عاتق الاستدعاء أيضًا مسؤولية التأكد من عدم إمكانية نسخ الذاكرة (COWed)، مثل استخدام madvise(MADV_DONTFORK) أو عن طريق تخصيص ذاكرة مشتركة مجهولة.

free استدعاء لتحرير مخازن الذاكرة التي تم تخصيصها باستخدام alloc() ناجح.

pd_context مؤشر لبيانات إضافية خاصة بالمستخدم ليتم ربطها بهذا النطاق الأصل. يتم تمرير المؤشر مرة أخرى إلى وظائف المخصص المخصصة.

قيمة الإرجاع

ibv_alloc_parent_domain() ترجع مؤشرًا إلى كائن ibv_pd المخصص، أو NULL إذا فشل الطلب (وتحدد errno لبيان سبب الفشل).
