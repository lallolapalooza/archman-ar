io_cancel(3)

الحزمة: libaio

التراخيص: LGPL-2.0-or-later

io_cancel(3) Linux AIO io_cancel(3)

الاسم

io_cancel - إلغاء طلبات الإدخال/الإخراج

الملخص

#include <errno.h>

#include <libaio.h>

int io_cancel(io_context_t ctx, struct iocb *iocb);

struct iocb {
	void		*data; /* العودة في حدث اكتمال الإدخال/الإخراج */
	unsigned	key;	/* للاستخدام في تحديد طلبات الإدخال/الإخراج */
	short		aio_lio_opcode;
	short		aio_reqprio; 	/* غير مستخدم */
	int		aio_fildes;
};

الوصف

يحاول إلغاء iocb تم تمريره سابقًا إلى io_submit(3). إذا تم إلغاء العملية بنجاح، يتم نسخ الحدث الناتج إلى الذاكرة المشار إليها بواسطة result دون وضعه في قائمة الاكتمال.

عندما تتم معالجة طلب واحد أو أكثر بشكل غير متزامن، قد يكون من المفيد في بعض المواقف إلغاء عملية محددة، على سبيل المثال، إذا أصبح واضحًا أن البيانات المكتوبة لم تعد دقيقة وسيتعين الكتابة فوقها قريبًا. على سبيل المثال، افترض أن تطبيقًا يكتب بيانات في ملفات في موقف حيث يجب كتابة بيانات واردة جديدة في ملف سيتم تحديثه بواسطة طلب تم إدخاله في قائمة الانتظار.

قيم الإرجاع

يتم إرجاع 0 عند النجاح، وإلا يتم إرجاع errno.

الأخطاء

EFAULT
إذا كانت أي من هياكل البيانات المشار إليها غير صالحة.
EINVAL
إذا كان aio_context المحدد بواسطة ctx غير صالح.
EAGAIN
إذا لم يتم إلغاء iocb المحدد.
ENOSYS
إذا لم يتم التنفيذ.
