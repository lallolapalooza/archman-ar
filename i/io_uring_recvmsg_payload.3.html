الاسم

io_uring_recvmsg_out - الوصول إلى البيانات من multishot recvmsg

الصيغة

#include <liburing.h>
struct io_uring_recvmsg_out *io_uring_recvmsg_validate(void *buf,
                                                       int buf_len,
                                                       struct msghdr *msgh);
void *io_uring_recvmsg_name(struct io_uring_recvmsg_out *o);
struct cmsghdr *io_uring_recvmsg_cmsg_firsthdr(struct io_uring_recvmsg_out *o,
                                               struct msghdr *msgh);
struct cmsghdr *io_uring_recvmsg_cmsg_nexthdr(struct io_uring_recvmsg_out *o,
                                              struct msghdr *msgh,
                                              struct cmsghdr *cmsg);
void *io_uring_recvmsg_payload(struct io_uring_recvmsg_out *o,
                               struct msghdr *msgh);
unsigned int io_uring_recvmsg_payload_length(struct io_uring_recvmsg_out *o,
                                             int buf_len,
                                             struct msghdr *msgh);

الوصف

تُستخدم هذه الدوال للوصول إلى البيانات في الحمولة (payload) التي تم تسليمها بواسطة io_uring_prep_recvmsg_multishot(3).

يجب أن يشير msgh إلى struct msghdr الذي تم تقديمه مع الطلب.

io_uring_recvmsg_validate(3) سيتحقق من صحة المخزن المؤقت (buffer) الذي تم تسليمه بواسطة io_uring_prep_recvmsg_multishot(3) ويستخرج io_uring_recvmsg_out إذا كان صالحًا، ويعيد مؤشرًا إليه أو NULL بخلاف ذلك.

الهيكل معرف على النحو التالي:


struct io_uring_recvmsg_out {
        __u32 namelen;    /* عدد بايتات الاسم كما سيتم ملؤه
                           * بواسطة recvmsg(2) */
        __u32 controllen; /* عدد بايتات التحكم */
        __u32 payloadlen; /* عدد بايتات الحمولة كما سيتم إرجاعه
                           * بواسطة recvmsg(2) */
        __u32 flags;      /* نتيجة العلامات كما سيتم ملؤها
                           * بواسطة recvmsg(2) */
};
  • io_uring_recvmsg_name(3)
يعيد مؤشرًا إلى الاسم في المخزن المؤقت.
  • io_uring_recvmsg_cmsg_firsthdr(3)
يعيد مؤشرًا إلى أول cmsg في المخزن المؤقت، أو NULL.
  • io_uring_recvmsg_cmsg_nexthdr(3)
يعيد مؤشرًا إلى cmsg التالي في المخزن المؤقت، أو NULL.
  • io_uring_recvmsg_payload(3)
يعيد مؤشرًا إلى الحمولة في المخزن المؤقت.
  • io_uring_recvmsg_payload_length(3)
يحسب طول الحمولة القابل للاستخدام بالبايت.
