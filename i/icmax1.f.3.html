integer function icmax1 (n, cx, incx)

ICMAX1 يبحث عن مؤشر أول عنصر في المتجه بقيمة مطلقة قصوى.

توثيق الوظيفة/الروتين الفرعي

integer function icmax1 (integer n, complex, dimension(*) cx, integer incx)

ICMAX1 يبحث عن مؤشر أول عنصر في المتجه بقيمة مطلقة قصوى.

الغرض:

!>
!> ICMAX1 يبحث عن مؤشر أول عنصر في المتجه بقيمة مطلقة قصوى.
!>
!> استنادًا إلى ICAMAX من BLAS المستوى 1.
!> التغيير هو استخدام القيمة المطلقة 'الحقيقية'.
!>

المعاملات

N 
!>          N هو عدد صحيح (INTEGER)
!>          عدد العناصر في المتجه CX.
!>

CX

!>          CX هو مصفوفة معقدة (COMPLEX array)، أبعادها (N)
!>          المتجه CX. ترجع الوظيفة ICMAX1 مؤشر أول عنصر بقيمة مطلقة قصوى.
!>

INCX

!>          INCX هو عدد صحيح (INTEGER)
!>          التباعد بين القيم المتتالية لـ CX.  INCX >= 1.
!>
