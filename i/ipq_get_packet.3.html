ipq_get_packet(3)

الحزمة: iptables

التراخيص: GPL-2.0-only

IPQ_MESSAGE_TYPE(3) دليل مبرمج Linux IPQ_MESSAGE_TYPE(3)

الاسم

ipq_message_type, ipq_get_packet, ipq_getmsgerr — استعلام رسائل الطابور

ملخص

#include <linux/netfilter.h>
#include <libipq.h>

int ipq_message_type(const unsigned char *buf);
ipq_packet_msg_t *ipq_get_packet(const unsigned char *buf);
int ipq_get_msgerr(const unsigned char *buf);

الوصف

تقوم الدالة ipq_message_type بإرجاع نوع رسالة الطابور التي تم إرجاعها إلى مساحة المستخدم عبر ipq_read.

يجب دائمًا استدعاء ipq_message_type بعد استدعاء ناجح لـ ipq_read لتحديد ما إذا كانت الرسالة رسالة حزمة أو رسالة خطأ. يجب أن تشير المعلمة buf إلى نفس البيانات التي تم الحصول عليها من الاستدعاء السابق لـ ipq_read.

ستعيد ipq_message_type إحدى القيم التالية:

NLMSG_ERROR
رسالة خطأ تم إنشاؤها بواسطة النقل Netlink.
IPQM_PACKET
رسالة حزمة تحتوي على بيانات وصفية للحزمة وبيانات حمولة اختيارية للحزمة.

يجب استدعاء الدالة ipq_get_packet إذا أعادت ipq_message_type القيمة IPQM_PACKET. يجب أن تشير المعلمة buf إلى نفس البيانات المستخدمة في استدعاء ipq_message_type. المؤشر الذي تم إرجاعه بواسطة ipq_get_packet يشير إلى رسالة حزمة، والتي تم تعريفها على النحو التالي:

typedef struct ipq_packet_msg {
	unsigned long packet_id;        /* معرف الحزمة في الطابور */
	unsigned long mark;             /* قيمة علامة Netfilter */
	long timestamp_sec;             /* وقت وصول الحزمة (ثواني) */
	long timestamp_usec;            /* وقت وصول الحزمة (+ ميكروثواني) */
	unsigned int hook;              /* نقطة ربط Netfilter التي عبرناها */
	char indev_name[IFNAMSIZ];      /* اسم واجهة الإدخال */
	char outdev_name[IFNAMSIZ];     /* اسم واجهة الإخراج */
	unsigned short hw_protocol;     /* بروتوكول الجهاز (ترتيب الشبكة) */
	unsigned short hw_type;         /* نوع الجهاز */
	unsigned char hw_addrlen;       /* طول عنوان الجهاز */
	unsigned char hw_addr[8];       /* عنوان الجهاز */
	size_t data_len;                /* طول بيانات الحزمة */
	unsigned char payload[0];       /* بيانات الحزمة الاختيارية */
} ipq_packet_msg_t;

يمكن للتطبيق قراءة كل من هذه الحقول. إذا كان وضع الطابور هو IPQ_COPY_PACKET وكانت قيمة data_len أكبر من الصفر، فيمكن الوصول إلى محتويات حمولة الحزمة في الذاكرة التي تلي بنية ipq_packet_msg_t إلى نطاق data_len.

يحتوي الحقل packet_id على معرف حزمة لاستخدامه عند استدعاء ipq_set_verdict.

يجب استدعاء الدالة ipq_get_msgerr إذا أعادت ipq_message_type القيمة NLMSG_ERROR. يجب أن تشير المعلمة buf إلى نفس البيانات المستخدمة في استدعاء ipq_message_type. القيمة التي تم إرجاعها بواسطة ipq_get_msgerr يتم تعيينها بواسطة كود النواة عالي المستوى وتتوافق مع قيم errno القياسية.
