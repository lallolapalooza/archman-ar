توثيق الوظيفة/الروتين الفرعي

integer function ilaclc (integer m, integer n, complex, dimension( lda, * ) a, integer lda)

ILACLC تقوم بمسح مصفوفة بحثًا عن آخر عمود غير صفري.

الغرض:

!>
!> ILACLC تقوم بمسح A بحثًا عن آخر عمود غير صفري.
!>

المعلمات

M 
!>          M هو عدد صحيح (INTEGER)
!>          عدد صفوف المصفوفة A.
!>

N

!>          N هو عدد صحيح (INTEGER)
!>          عدد أعمدة المصفوفة A.
!>

A

!>          A هي مصفوفة مركبة (COMPLEX)، ذات بُعد (LDA,N)
!>          المصفوفة A ذات الأبعاد m × n.
!>

LDA

!>          LDA هو عدد صحيح (INTEGER)
!>          البعد الرئيسي للمصفوفة A. LDA >= max(1,M).
!>
