مقابس نطاق الإنترنت

#include <netinet/in.h>
struct sockaddr_in {
    sa_family_t     sin_family;     /* AF_INET */
    in_port_t       sin_port;       /* رقم المنفذ */
    struct in_addr  sin_addr;       /* عنوان IPv4 */
};
struct sockaddr_in6 {
    sa_family_t     sin6_family;    /* AF_INET6 */
    in_port_t       sin6_port;      /* رقم المنفذ */
    uint32_t        sin6_flowinfo;  /* معلومات تدفق IPv6 */
    struct in6_addr sin6_addr;      /* عنوان IPv6 */
    uint32_t        sin6_scope_id;  /* مجموعة الواجهات للنطاق */
};
struct in_addr {
    in_addr_t s_addr;
};
struct in6_addr {
    uint8_t   s6_addr[16];
};
typedef uint32_t in_addr_t;
typedef uint16_t in_port_t;

مقابس نطاق UNIX

#include <sys/un.h>
struct sockaddr_un {
    sa_family_t     sun_family;     /* عائلة العنوان */
    char            sun_path[];     /* مسار اسم المقبس */
};

الوصف

sockaddr
يصف عنوان المقبس.
sockaddr_storage
بنية بحجم كافٍ على الأقل لتتسع لأي بنية عنوان أخرى من نوع sockaddr_*. يتم محاذاتها بحيث يمكن تحويل مؤشر إليها إلى مؤشر لبنى sockaddr_* أخرى واستخدامها للوصول إلى حقولها.
socklen_t
يصف طول عنوان المقبس. هذا نوع عدد صحيح بعرض 32 بت على الأقل.
sa_family_t
يصف عائلة بروتوكول المقبس. هذا نوع عدد صحيح غير موقع.

مقابس نطاق الإنترنت

sockaddr_in
يصف عنوان مقبس نطاق الإنترنت IPv4. يتم تخزين الأعضاء sin_port و sin_addr بترتيب البايتات الشبكي.
sockaddr_in6
يصف عنوان مقبس نطاق الإنترنت IPv6. يتم استخدام المصفوفة sin6_addr.s6_addr لاحتواء عنوان IPv6 المكون من 128 بت، والمخزن بترتيب البايتات الشبكي.

مقابس نطاق UNIX

sockaddr_un
يصف عنوان مقبس نطاق UNIX.

المعايير

POSIX.1-2024.

التاريخ

POSIX.1-2001.

تم اختراع socklen_t بواسطة POSIX. انظر أيضًا accept(2).

تم اختراع هذه البنى قبل قواعد تطابق الذاكرة المؤقتة (strict-aliasing) الصارمة في ISO C الحديثة. إذا تم تطبيق قواعد تطابق الذاكرة المؤقتة بصرامة، فإن استخدام هذه البنى سيكون صعبًا للغاية دون استدعاء سلوك غير محدد. قامت POSIX.1-2024 بإصلاح ذلك عن طريق طلب أن تتأكد التطبيقات من أن هذه البنى يمكن استخدامها بأمان كما تم تصميمها.

ملاحظات

socklen_t معرف أيضًا في <netdb.h>.

sa_family_t معرف أيضًا في <netinet/in.h> و <sys/un.h>.
