subroutine alarqg (character*3 path, integer nmats, logical, dimension( * ) dotype, integer ntypes, integer nin, integer nout)

ALARQG

الغرض:

!>
!> ALARQG تتعامل مع المدخلات لبرنامج اختبار LAPACK. يتم استدعاؤها
!> لتقييم سطر الإدخال الذي طلب NMATS نوع مصفوفة لـ
!> PATH. يتدفق التحكم على النحو التالي:
!>
!> IF NMATS = NTYPES THEN
!>    DOTYPE(1:NTYPES) = .TRUE.
!> ELSE
!>    اقرأ سطر الإدخال التالي لـ NMATS نوع مصفوفة
!>    عين DOTYPE(I) = .TRUE. لكل نوع صالح I
!> END IF
!>

المعلمات

PATH 
!>          PATH هو CHARACTER*3
!>          اسم مسار LAPACK للاختبار.
!>

NMATS

!>          NMATS هو INTEGER
!>          عدد أنواع المصفوفات التي سيتم استخدامها في اختبار هذا المسار.
!>

DOTYPE

!>          DOTYPE هو LOGICAL array, dimension (NTYPES)
!>          متجه العلامات الذي يشير إلى ما إذا كان سيتم اختبار كل نوع.
!>

NTYPES

!>          NTYPES هو INTEGER
!>          الحد الأقصى لعدد أنواع المصفوفات لهذا المسار.
!>

NIN

!>          NIN هو INTEGER
!>          رقم وحدة الإدخال. NIN >= 1.
!>

NOUT

!>          NOUT هو INTEGER
!>          رقم وحدة الإخراج. NOUT >= 1.
!>
