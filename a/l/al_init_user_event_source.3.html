al_init_user_event_source(3)

الحزمة: allegro

التراخيص: Zlib, BSD-3-Clause, OFL-1.1, LicenseRef-Custom

al_init_user_event_source(3) دليل وظائف المكتبة al_init_user_event_source(3)

الاسم

al_init_user_event_source - واجهة برمجة تطبيقات Allegro 5

الملخص

#include <allegro5/allegro.h>
void al_init_user_event_source(ALLEGRO_EVENT_SOURCE *src)

الوصف

تهيئة مصدر أحداث لإصدار أحداث المستخدم. يجب أن تكون المساحة المخصصة لمصدر الأحداث قد تم تخصيصها بالفعل.

إحدى الطرق الممكنة لإنشاء مصادر أحداث مخصصة هي اشتقاق هياكل أخرى مع ALLEGRO_EVENT_SOURCE في المقدمة، على سبيل المثال:

typedef struct THING THING;
struct THING {
    ALLEGRO_EVENT_SOURCE event_source;
    int field1;
    int field2;
    /* الخ...*/
};
THING *create_thing(void)
{
    THING *thing = malloc(sizeof(THING));
    if (thing) {
        al_init_user_event_source(&thing->event_source);
        thing->field1 = 0;
        thing->field2 = 0;
    }
    return thing;
}

الميزة هنا هي أن مؤشر THING سيكون هو نفسه مؤشر ALLEGRO_EVENT_SOURCE. ستتضمن الأحداث الصادرة عن مصدر الأحداث مؤشر مصدر الأحداث كمجال source، والذي يمكنك من خلاله الحصول على مؤشر إلى THING عن طريق تحويل بسيط (بعد التأكد من التحقق من أن الحدث من النوع الصحيح).

ومع ذلك، هذه مجرد تقنية واحدة ولست ملزمًا باستخدامها.

لن يتم تدمير مصدر حدث المستخدم تلقائيًا. يجب عليك تدميره يدويًا باستخدام al_destroy_user_event_source(3).
