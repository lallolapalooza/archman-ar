alacritty(5)

الحزمة: alacritty

التراخيص: Apache-2.0, MIT

/b> = [{ index = <integer>, color = "<string>" },]

تتضمن الألوان المرقمة جميع الألوان من 16 إلى 256. عندما لا يتم تعيين هذه، يتم ملؤها بقيم افتراضية منطقية.

الافتراضي: []

transparent_background_colors = true | false

سواء كان window.opacity ينطبق على جميع خلفيات الخلايا، أو فقط على الخلفية الافتراضية. عند التعيين على true ستكون جميع الخلايا شفافة بغض النظر عن لون خلفيتها.

الافتراضي: false

draw_bold_text_with_bright_colors = true | false

عندما تكون true، يتم رسم النص العريض باستخدام متغيرات الألوان الساطعة.

الافتراضي: false

الجرس

يوثق هذا القسم جدول [bell] في ملف التكوين.

animation = "Ease" | "EaseOut" | "EaseOutSine" | "EaseOutQuad" | "EaseOutCubic" | "EaseOutQuart" | "EaseOutQuint" | "EaseOutExpo" | "EaseOutCirc" | "Linear"

تأثير رسوم متحركة للجرس المرئي لوميض الشاشة عند رنين الجرس المرئي.

الافتراضي: "Linear"

duration = <integer>

مدة وميض الجرس المرئي بالمللي ثانية. سيؤدي تعيين `duration` إلى `0` إلى تعطيل الرسوم المتحركة للجرس المرئي.

الافتراضي: 0

color = "<string>"

لون الرسوم المتحركة للجرس المرئي.

الافتراضي: "#ffffff"

command = "<string>" | { program = "<string>", args = ["<string>",] }

يتم تنفيذ هذا البرنامج في كل مرة يتم فيها رنين الجرس.

عند التعيين على "None"، لن يتم تنفيذ أي أمر.

الافتراضي: "None"

الاختيار

يوثق هذا القسم جدول [selection] في ملف التكوين.

semantic_escape_chars = "<string>"

يحتوي هذا السلسلة على جميع الأحرف التي تستخدم كفواصل لـ "الكلمات الدلالية" في Alacritty.

الافتراضي: ",│`|:\"' ()[]{{}}<>\t"

save_to_clipboard = true | false

عند التعيين على true، سيتم نسخ النص المحدد إلى الحافظة الأساسية.

الافتراضي: false

المؤشر

يوثق هذا القسم جدول [cursor] في ملف التكوين.

style = { <shape>, <blinking> }

shape = "Block" | "Underline" | "Beam"
الافتراضي: "Block"

blinking = "Never" | "Off" | "On" | "Always"

Never
منع المؤشر من الوميض أبداً
Off
تعطيل الوميض افتراضياً
On
تمكين الوميض افتراضياً
Always
فرض الوميض المستمر للمؤشر

الافتراضي: "Off"

vi_mode_style = { <shape>, <blinking> } | "None"

إذا كان نمط مؤشر وضع vi هو "None" أو لم يتم تحديده، فسيعود إلى القيمة النشطة للمؤشر العادي.

الافتراضي: "None"

blink_interval = <integer>

فاصل الوميض للمؤشر بالمللي ثانية.

الافتراضي: 750

blink_timeout = <integer>

الوقت الذي يتوقف بعده المؤشر عن الوميض، بالثواني.

تحديد 0 سيعطل انتهاء مهلة الوميض.

الافتراضي: 5

unfocused_hollow = true | false

عندما تكون هذه القيمة true، سيتم عرض المؤشر كصندوق مجوف عندما لا يكون النافذة في التركيز.

الافتراضي: true

thickness = <float>

سمك المؤشر بالنسبة لعرض الخلية كرقم عشري من 0.0 إلى 1.0.

الافتراضي: 0.15

طرفية

يوثق هذا القسم جدول [terminal] في ملف التكوين.

shell = "<string>" | { program = "<string>", args = ["<string>",] }

يمكنك تعيين shell.program إلى مسار قوقلتك المفضلة، على سبيل المثال /bin/zsh. يتم تمرير الإدخالات في shell.args كوسائط للصدفة.

الافتراضي:

Linux/BSD/macOS: $SHELL أو الصدفة الافتراضية للمستخدم، إذا كانت $SHELL غير معينة
Windows: "powershell"

مثال:

shell = { program = "/bin/zsh", args = ["-l"] }

osc52 = "Disabled" | "OnlyCopy" | "OnlyPaste" | "CopyPaste"

يتحكم في القدرة على الكتابة إلى الحافظة النظامية باستخدام تسلسل الهروب OSC 52. بينما يكون تسلسل الهروب هذا مفيدًا لنسخ المحتويات من الخادم البعيد، فإن السماح لأي تطبيق بالقراءة من الحافظة يمكن إساءة استخدامه بسهولة دون توفير فوائد كبيرة مقارنة بلصق النص صراحةً.

الافتراضي: "OnlyCopy"

الفأرة

يوثق هذا القسم جدول [mouse] في ملف التكوين.

hide_when_typing = true | false

عندما تكون هذه القيمة true، يتم إخفاء المؤشر مؤقتًا عند الكتابة.

الافتراضي: false

bindings = [{ <mouse>, <mods>, <mode>, <command> | <chars> | <action> },]

راجع keyboard.bindings للحصول على توثيق كامل لـ mods و mode و command و chars و action.

عندما يلتقط تطبيق يعمل داخل Alacritty الفأرة، يمكن استخدام المعدل `Shift` لقمع تقرير الفأرة. إذا لم يتم العثور على إجراء للحدث، يتم تشغيل إجراءات للحدث بدون المعدل `Shift` بدلاً من ذلك.

mouse = "Middle" | "Left" | "Right" | "Back" | "Forward" | <integer>

زر الفأرة الذي يجب الضغط عليه لتشغيل هذا الربط.

action = <keyboard.bindings.action> | "ExpandSelection"

ExpandSelection
توسيع الاختيار إلى موقع مؤشر الفأرة الحالي.

مثال:

[mouse]
bindings = [
{ mouse = "Right", mods = "Control", action = "Paste" },
]

تلميحات

يوثق هذا القسم جدول [hints] في ملف التكوين.

يمكن استخدام تلميحات الطرفية للعثور على النص أو الارتباطات التشعبية في الجزء المرئي من الطرفية وتمريرها إلى تطبيقات أخرى.

alphabet = "<string>"

المفاتيح المستخدمة لعناوين التلميحات.

الافتراضي: "jfkdls;ahgurieowpq"

enabled = [{ <regex>, <hyperlinks>, <post_processing>, <persist>, <action>, <command>, <binding>, <mouse> },]

مصفوفة بجميع التلميحات المتاحة.

يجب أن تحتوي كل تلميحة على واحد على الأقل من regex أو hyperlinks وإما action أو command.

regex = "<string>"
التعبير العادي الذي سيتم مقارنة كل سطر به.

hyperlinks = true | false

عندما تكون هذه القيمة true، سيتم تضمين جميع الارتباطات التشعبية لتسلسل الهروب OSC 8 في التلميحات.

post_processing = true | false

عندما تكون هذه القيمة true، سيتم استخدام الاستدلالات لتقصير المطابقة إذا كانت هناك أحرف من المحتمل ألا تكون جزءًا من التلميح (على سبيل المثال، نقطة لاحقة .). هذا مفيد بشكل خاص لمعرفات URI وينطبق فقط على مطابقات regex.

persist = true | false

عندما تكون هذه القيمة true، تظل التلميحات ثابتة بعد الاختيار.

action = "Copy" | "Paste" | "Select" | "MoveViModeCursor"

Copy
نسخ نص التلميح إلى الحافظة.
Paste
لصق نص التلميح في الطرفية أو البحث.
Select
تحديد نص التلميح.
MoveViModeCursor
نقل مؤشر وضع vi إلى بداية التلميح.

command = "<string>" | { program = "<string>", args = ["<string>",] }

الأمر الذي سيتم تنفيذه عند النقر على التلميح أو تحديده باستخدام binding.

يتم دائمًا إرفاق نص التلميح كوسيط أخير.

binding = { key = "<string>", mods = "<string>", mode = "<string>" }

راجع keyboard.bindings للحصول على توثيق للقيم المتاحة.

يتحكم هذا في ربط المفتاح المستخدم لبدء عملية تحديد تلميحات لوحة المفاتيح.

mouse = { mods = "<string>", enabled = true | false }

راجع keyboard.bindings للحصول على توثيق لـ mods المتاحة.

هذا الموقع غير مرتبط بـ Arch Linux ولا تابع له بأي شكل من الأشكال. المحتوى هو ترجمة فقط لصفحات دليل Arch المنشورة على man.archlinux.org.
