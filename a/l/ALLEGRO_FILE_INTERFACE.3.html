ALLEGRO_FILE_INTERFACE(3)

الحزمة: allegro

التراخيص: Zlib, BSD-3-Clause, OFL-1.1, LicenseRef-Custom

ALLEGRO_FILE_INTERFACE(3) دليل وظائف المكتبة ALLEGRO_FILE_INTERFACE(3)

الاسم

ALLEGRO_FILE_INTERFACE - واجهة برمجة تطبيقات Allegro 5

الملخص

#include <allegro5/allegro.h>
typedef struct ALLEGRO_FILE_INTERFACE

الوصف

هيكل يحتوي على مؤشرات وظائف للتعامل مع نوع من "الملفات"، حقيقية أو افتراضية. انظر المناقشة الكاملة في al_set_new_file_interface(3).

الحقول هي:

void*         (*fi_fopen)(const char *path, const char *mode);
bool          (*fi_fclose)(ALLEGRO_FILE *f);
size_t        (*fi_fread)(ALLEGRO_FILE *f, void *ptr, size_t size);
size_t        (*fi_fwrite)(ALLEGRO_FILE *f, const void *ptr, size_t size);
bool          (*fi_fflush)(ALLEGRO_FILE *f);
int64_t       (*fi_ftell)(ALLEGRO_FILE *f);
bool          (*fi_fseek)(ALLEGRO_FILE *f, int64_t offset, int whence);
bool          (*fi_feof)(ALLEGRO_FILE *f);
int           (*fi_ferror)(ALLEGRO_FILE *f);
const char *  (*fi_ferrmsg)(ALLEGRO_FILE *f);
void          (*fi_fclearerr)(ALLEGRO_FILE *f);
int           (*fi_fungetc)(ALLEGRO_FILE *f, int c);
off_t         (*fi_fsize)(ALLEGRO_FILE *f);

يجب على وظيفة fi_open تخصيص الذاكرة لأي هيكل بيانات مستخدم تحتاج إليه. يجب إرجاع مؤشر إلى تلك الذاكرة؛ سيتم ربطها بالملف بعد ذلك. يمكن للوظائف الأخرى الوصول إلى تلك البيانات عن طريق استدعاء al_get_file_userdata(3) على مقبض الملف. إذا أعادت fi_open القيمة NULL، فستعيد al_fopen(3) أيضًا القيمة NULL.

يجب على وظيفة fi_fclose تنظيف وتحرير بيانات المستخدم، ولكن Allegro سيحرر مقبض ALLEGRO_FILE(3).

إذا كانت fi_fungetc هي NULL، فسيتم استخدام التنفيذ الافتراضي لـ Allegro لمخزن مؤقت بطول 16 حرفًا.
