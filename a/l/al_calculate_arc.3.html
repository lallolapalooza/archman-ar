al_calculate_arc(3)

الحزمة: allegro

التراخيص: Zlib, BSD-3-Clause, OFL-1.1, LicenseRef-Custom

al_calculate_arc(3) دليل وظائف المكتبة al_calculate_arc(3)

الاسم

al_calculate_arc - واجهة برمجة تطبيقات Allegro 5

الملخص

#include <allegro5/allegro_primitives.h>
void al_calculate_arc(float* dest, int stride, float cx, float cy,
   float rx, float ry, float start_theta, float delta_theta, float thickness,
   int num_points)

الوصف

عندما تكون thickness <= 0، تقوم هذه الدالة بحساب مواقع num_points نقطة متباعدة بانتظام على قوس قطع ناقص. عندما تكون thickness > 0، تقوم هذه الدالة بحساب مجموعتين من النقاط، يتم الحصول عليهما على النحو التالي: يتم الحصول على المجموعة الأولى عن طريق أخذ النقاط المحسوبة في الحالة thickness <= 0 ونقلها بمقدار thickness / 2 إلى الخارج، في اتجاه عمودي على منحنى القوس. المجموعة الثانية هي نفسها، ولكن يتم نقلها بمقدار thickness / 2 إلى الداخل بالنسبة للقوس. يتم تشابك المجموعتين من النقاط في المخزن المؤقت الوجهة (أي أن الزوج الأول من النقاط سيكون متجانسًا مع مركز القوس، وستكون النقطة الأولى من الزوج أبعد عن المركز من النقطة الثانية؛ وسيكون الزوج التالي متجانسًا أيضًا، ولكن بزاوية مختلفة وهكذا).

يتم تفسير المخزن المؤقت الوجهة dest كمجموعة من أزواج الأرقام العائمة المتباعدة بانتظام، وكل زوج يحمل إحداثيات النقطة المقابلة على القوس. الرقمين العائمين في الزوج متجاوران، والمسافة (بالبايت) بين عناوين الرقم العائم الأول في زوجين متعاقبين هي stride. على سبيل المثال، إذا كان لديك مصفوفة متراصة من الأرقام العائمة بدون مسافات بين الأزواج، فسيكون stride بالضبط 2 * sizeof(float).

مثال مع thickness <= 0:

const int num_points = 4;
float points[num_points][2];
al_calculate_arc(&points[0][0], 2 * sizeof(float), 0, 0, 10, 10, 0, ALLEGRO_PI / 2, 0, num_points);
assert((int)points[0][0] == 10);
assert((int)points[0][1] == 0);
assert((int)points[num_points - 1][0] == 0);
assert((int)points[num_points - 1][1] == 10);

مثال مع thickness > 0:

const int num_points = 4;
float points[num_points * 2][2];
al_calculate_arc(&points[0][0], 2 * sizeof(float), 0, 0, 10, 10, 0, ALLEGRO_PI / 2, 2, num_points);
assert((int)points[0][0] == 11);
assert((int)points[0][1] == 0);
assert((int)points[1][0] == 9);
assert((int)points[1][1] == 0);
assert((int)points[(num_points - 1) * 2][0] == 0);
assert((int)points[(num_points - 1) * 2][1] == 11);
assert((int)points[(num_points - 1) * 2 + 1][0] == 0);
assert((int)points[(num_points - 1) * 2 + 1][1] == 9);

المعلمات:

•
dest - المخزن المؤقت الوجهة
•
stride - المسافة (بالبايت) بين بدايات الأزواج المتتالية للنقاط
•
cx, cy - مركز القوس
•
rx, ry - أنصاف أقطار القوس
•
start_theta - الزاوية الأولية التي يتم حساب القوس منها بالراديان
•
delta_theta - امتداد الزاوية للقوس بالراديان (مرر رقمًا سالبًا لتبديل الاتجاه)
•
thickness - سمك القوس
•
num_points - عدد النقاط المراد حسابها
