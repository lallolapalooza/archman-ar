subroutine aladhd (integer iounit, character*3 path)

ALADHD

الغرض:

!>
!> ALADHD يطبع معلومات رأسية لمسارات اختبار الإجراءات التشغيلية.
!>

المعاملات

IOUNIT 
!>          IOUNIT هو عدد صحيح
!>          رقم الوحدة الذي يجب طباعة معلومات الرأس إليه.
!>

PATH

!>          PATH هو حرف بطول 3
!>          اسم المسار الذي ستتم طباعة معلومات الرأس له. المسارات الحالية هي
!>             _GE:  المصفوفات العامة
!>             _GB:  الشريط العام
!>             _GT:  الشريط الثلاثي العام
!>             _PO:  محددة موجبة متماثلة أو هرميتية
!>             _PS:  محددة موجبة نصف محددة متماثلة أو هرميتية
!>             _PP:  محددة موجبة متماثلة أو هرميتية معبأة
!>             _PB:  محددة موجبة متماثلة أو هرميتية بشريط
!>             _PT:  محددة موجبة متماثلة أو هرميتية بشريط ثلاثي
!>             _SY:  غير محددة متماثلة،
!>                     مع تقليب جزئي (Bunch-Kaufman)
!>             _SR:  غير محددة متماثلة،
!>                     مع تقليب رخ (Bunch-Kaufman محدود)
!>             _SK:  غير محددة متماثلة،
!>                     مع تقليب رخ (Bunch-Kaufman محدود)
!>                     ( تنسيق تخزين جديد للعوامل:
!>                       L والقطر لـ D مخزنان في A،
!>                       الموازيات الفرعية لـ D مخزنة في E )
!>             _SP:  غير محددة متماثلة معبأة،
!>                     مع تقليب جزئي (Bunch-Kaufman)
!>             _HA:  (مركبة) هرميتية،
!>                     خوارزمية Assen
!>             _HE:  (مركبة) غير محددة هرميتية،
!>                     مع تقليب جزئي (Bunch-Kaufman)
!>             _HR:  (مركبة) غير محددة هرميتية،
!>                     مع تقليب رخ (Bunch-Kaufman محدود)
!>             _HK:  (مركبة) غير محددة هرميتية،
!>                     مع تقليب رخ (Bunch-Kaufman محدود)
!>                     ( تنسيق تخزين جديد للعوامل:
!>                       L والقطر لـ D مخزنان في A،
!>                       الموازيات الفرعية لـ D مخزنة في E )
!>             _HP:  (مركبة) غير محددة هرميتية معبأة،
!>                     مع تقليب جزئي (Bunch-Kaufman)
!>          يجب أن يكون الحرف الأول واحدًا من S، D، C، أو Z (C أو Z فقط
!>          إذا كانت مركبة).
!>
