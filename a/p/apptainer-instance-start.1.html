apptainer-instance-start(1)

الحزمة: apptainer

التراخيص: BSD-3-Clause, BSD-3-Clause-LBNL

apptainer(1) apptainer(1)

الاسم

apptainer-instance-start - بدء تشغيل مثيل مسمى لصورة الحاوية المعطاة

السيناريو

apptainer instance start [خيارات البدء...] [وسيطات سكربت البدء...]

الوصف

يسمح لك أمر instance start بإنشاء مثيل مسمى جديد من صورة حاوية موجودة سيبدأ تشغيلها في الخلفية. إذا تم تعريف سكربت بدء تشغيل في بيانات وصفية الحاوية، فسيتم تنفيذ الأوامر الموجودة في هذا السكربت مع الأمر instance start أيضًا. يمكنك اختياريًا تمرير وسيطات إلى سكربت البدء.

يقبل apptainer instance start تنسيقات الحاويات التالية:

*.sif تنسيق صورة Singularity (SIF). أصلي لـ Singularity (3.0+) و Apptainer (v1.0.0+)

*.sqsh تنسيق SquashFS. أصلي لـ Singularity 2.4+

*.img تنسيق ext3. أصلي لإصدارات Singularity < 2.4.

directory/ sandbox format. دليل يحتوي على نظام ملفات جذر صالح و/أو بيانات وصفية Apptainer.

instance://* مثيل محلي قيد التشغيل لحاوية. (راجع مجموعة أوامر instance.)

library://* حاوية SIF مستضافة على مكتبة (لا يوجد افتراضي)

docker://* حاوية Docker/OCI مستضافة على Docker Hub أو سجل OCI آخر.

shub://* حاوية مستضافة على Singularity Hub.

oras://* حاوية SIF مستضافة على سجل OCI يدعم مواصفات OCI Registry As Storage (ORAS).

الخيارات

--add-caps="" قائمة صلاحيات مفصولة بفواصل لإضافتها

--allow-setuid[=false] السماح بملفات setuid في الحاوية (للجذر فقط)

--app="" تعيين تطبيق ليتم تشغيله داخل حاوية

--apply-cgroups="" تطبيق cgroups من ملف لعمليات الحاوية (للجذر فقط)

--authfile="" ملف مصادقة بأسلوب Docker لاستخدامه لقراءة/كتابة بيانات اعتماد سجل OCI

-B, --bind=[] تحديد مسار ربط المستخدم. المواصفات لها التنسيق src[:dest[:opts]]، حيث src و dest هما مسارات خارجية وداخلية. إذا لم يتم تحديد dest، فسيتم تعيينه على أنه src. يمكن تحديد خيارات التحميل (opts) كـ 'ro' (للقراءة فقط) أو 'rw' (للقراءة/الكتابة، وهو الافتراضي). يمكن تحديد مسارات ربط متعددة بقائمة مفصولة بفواصل.

--blkio-weight=0 وزن نسبي لـ Block IO في النطاق 10-1000، 0 لتعطيله

--blkio-weight-device=[] وزن نسبي لـ Block IO لجهاز معين

--boot[=false] تنفيذ /sbin/init لتمهيد الحاوية (للجذر فقط)

-e, --cleanenv[=false] تنظيف البيئة قبل تشغيل الحاوية

--compat[=false] تطبيق إعدادات لزيادة التوافق مع OCI/Docker. يستدل على --containall، --no-init، --no-umask، --no-eval، --writable-tmpfs.

-c, --contain[=false] استخدام الحد الأدنى من /dev وإفراغ الدلائل الأخرى (مثل /tmp و $HOME) بدلاً من مشاركة أنظمة الملفات من المضيف الخاص بك

-C, --containall[=false] احتواء ليس فقط أنظمة الملفات، ولكن أيضًا PID، IPC، والبيئة

--cpu-shares=-1 حصة وحدة المعالجة المركزية للحاوية

--cpus="" عدد وحدات المعالجة المركزية المتاحة للحاوية

--cpuset-cpus="" قائمة وحدات المعالجة المركزية للمضيف المتاحة للحاوية

--cpuset-mems="" قائمة عقد الذاكرة للمضيف المتاحة للحاوية

--disable-cache[=false] عدم استخدام أو إنشاء ذاكرة التخزين المؤقت

--dmtcp-launch="" نقطة حفظ لـ dmtcp لحفظ حالة عملية الحاوية (تجريبي)

--dmtcp-restart="" نقطة حفظ لـ dmtcp لإعادة تشغيل عملية الحاوية (تجريبي)

--dns="" قائمة خوادم DNS مفصولة بفواصل لإضافتها في resolv.conf

--docker-host="" تحديد مضيف Docker daemon مخصص

--docker-login[=false] تسجيل الدخول إلى مستودع Docker بشكل تفاعلي

--drop-caps="" قائمة صلاحيات مفصولة بفواصل لإسقاطها

--env=[] تمرير متغير البيئة إلى العملية المحتواة

--env-file=[] تمرير متغيرات البيئة من ملف إلى العملية المحتواة

-f, --fakeroot[=false] تشغيل الحاوية بمظهر التشغيل كجذر

--fusemount=[] تحديد تحميل نظام ملفات FUSE بالشكل ': ' - حيث يكون 'container' أو 'host'، مما يحدد مكان إجراء التحميل ('container-daemon' أو 'host-daemon' سيشغل عملية FUSE بشكل منفصل). هو مسار تنفيذ FUSE، بالإضافة إلى خيارات التحميل. هو موقع داخل الحاوية حيث سيتم إرفاق تحميل FUSE. على سبيل المثال 'container:sshfs 10.0.0.1:/ /sshfs'. يستلزم --pid.

-h, --help[=false] مساعدة للبدء

-H, --home="/build" تحديد مسار دليل المنزل. يمكن أن يكون المصدر مسارًا مصدرًا أو زوجًا src:dest. src هو مسار المصدر لدليل المنزل خارج الحاوية و dest يحل محل دليل المنزل داخل الحاوية.

--hostname="" تعيين اسم مضيف الحاوية

-i, --ipc[=false] تشغيل الحاوية في مساحة اسم IPC جديدة

--keep-privs[=false] السماح لمستخدم الجذر بالاحتفاظ بالصلاحيات في الحاوية (للجذر فقط)

--memory="" حد الذاكرة بالبايت

--memory-reservation="" حد الذاكرة الناعم بالبايت

--memory-swap="" حد الذاكرة المتبادلة (swap)، استخدم -1 للذاكرة المتبادلة غير المحدودة

--mount=[] تحديد تحميل مثل 'type=bind,source=/opt,destination=/hostopt'.

-n, --net[=false] تشغيل الحاوية في مساحة شبكة جديدة (تقوم بإعداد واجهة شبكة جسر افتراضيًا)

--netns-path="" الانضمام إلى مساحة اسم الشبكة في المسار المحدد (كجذر، أو إذا سمح به في apptainer.conf)

--network="" تحديد نوع الشبكة المطلوب مفصولة بفواصل، كل شبكة ستقوم بتشغيل واجهة مخصصة داخل الحاوية

--network-args=[] تحديد وسيطات الشبكة التي سيتم تمريرها إلى ملحقات CNI

--no-eval[=false] عدم تقييم متغيرات البيئة أو أوامر OCI CMD/ENTRYPOINT/ARGS

--no-home[=false] عدم تحميل دليل منزل المستخدم إذا لم يكن /home هو دليل العمل الحالي

--no-https[=false] استخدام http بدلاً من https لـ docker:// oras:// و library:///... URIs

--no-init[=false] عدم بدء عملية shim مع --pid

--no-mount=[] تعطيل واحد أو أكثر من خيارات 'mount xxx' المحددة في apptainer.conf و/أو تحديد مسار وجهة مطلق لتعطيل إدخال مسار ربط، أو 'bind-paths' لتعطيل جميع إدخالات مسارات الربط.

--no-privs[=false] إسقاط جميع الصلاحيات من مستخدم الجذر في الحاوية)

--no-umask[=false] عدم نشر umask إلى الحاوية، تعيين umask افتراضي 0022

--nv[=false] تمكين دعم Nvidia

--nvccli[=false] استخدام nvidia-container-cli لإعداد وحدة معالجة الرسومات (تجريبي)

--oom-kill-disable[=false] تعطيل قاتل OOM

-o, --overlay=[] استخدام صورة overlayFS لتخزين البيانات الدائمة أو كطبقة قراءة فقط للحاوية

--passphrase[=false] طلب عبارة مرور تشفير

--pem-path="" إدخال مسار لمفتاح RSA بتنسيق PEM لحاوية مشفرة

--pid-file="" كتابة PID المثيل إلى الملف بالاسم المعطى

--pids-limit=0 تحديد عدد PIDs للحاوية، استخدم -1 لغير المحدود

--rocm[=false] تمكين دعم Rocm التجريبي

-S, --scratch=[] تضمين دليل خدش داخل الحاوية يتم ربطه بدليل مؤقت (استخدم -W لفرض الموقع)

--security=[] تمكين ميزات الأمان (SELinux، Apparmor، Seccomp)

--unsquash[=false] تحويل ملف SIF إلى sandbox مؤقت قبل التشغيل

-u, --userns[=false] تشغيل الحاوية في مساحة اسم مستخدم جديدة

--uts[=false] تشغيل الحاوية في مساحة اسم UTS جديدة

-W, --workdir="" دليل العمل المراد استخدامه لـ /tmp، /var/tmp و $HOME (إذا تم استخدام -c/--contain أيضًا)

-w, --writable[=false] بشكل افتراضي، تكون جميع حاويات Apptainer للقراءة فقط. يجعل هذا الخيار نظام الملفات قابلاً للوصول

هذا الموقع غير مرتبط بـ Arch Linux ولا تابع له بأي شكل من الأشكال. المحتوى هو ترجمة فقط لصفحات دليل Arch المنشورة على man.archlinux.org.
