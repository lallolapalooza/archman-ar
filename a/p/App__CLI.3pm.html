App__CLI(3pm)

الحزمة: perl-app-cli

التراخيص: GPL, PerlArtistic

App::CLI(3) وثائق Perl مساهم بها من قبل المستخدم App::CLI(3)

الاسم

App::CLI - وحدة إرسال لبرامج واجهة سطر الأوامر

الملخص

package MyApp;
use base 'App::CLI';        # المرسل لتطبيقك
                            # ليس من الضروري وضع المرسل
                            # في المستوى الأعلى لتطبيقك
package main;
MyApp->dispatch;            # استدعاء المرسل حيثما تريد
package MyApp::List;
use base qw(App::CLI::Command); # أي (أمر فرعي) لتطبيقك
use constant options => (
    "h|help"   => "help",
    "verbose"  => "verbose",
    'n|name=s'  => 'name',
);
use constant subcommands => qw(User Nickname type); # إذا كنت تريد أوامر فرعية
                                                    # يتم إرسالها تلقائيًا إلى الأوامر الفرعية
                                                    # عند استدعاء $ myapp list [user|nickname|--type]
                                                    # لاحظ أن 'type' ليست بحروف كبيرة
                                                    # إنه أمر فرعي مهمل
sub run {
    my ($self, @args) = @_;
    print "verbose" if $self->{verbose};
    my $name = $self->{name}; # الحصول على الوسيط بعد الخيار الطويل --name
    if ($self->{help}) {
        # إذا $ myapp list --help أو $ myapp list -h
        # فقط أخرج PODs
    } else {
        # افعل شيئًا عند استدعاء $ myapp list
        # بدون أمر فرعي و --help
    }
}
package MyApp::List::User;
use base qw(App::CLI::Command);
use constant options => (
    "h|help"  =>  "help",
);
sub run {
    my ($self,@args) = @_;
    # كود لسرد المستخدم
}
pakcage MyApp::List::Nickname;
use base qw(App::CLI::Command);
use constant options => (
    "sort=s"  =>  "sort",
);
sub run {
    my ($self,@args) = @_;
    # كود لسرد اللقب
}
package MyApp::List::type;   # نوع قديم من الأوامر الفرعية لا يمكن أن يتفرع بشكل لا نهائي
use base qw(MyApp::List);    # يجب أن يرث أمر الوالد
sub run {
    my ($self, @args);
    # التشغيل إلى هنا عند استدعاء $ myapp list --type
}
package MyApp::Help;
use base 'App::CLI::Command::Help';
use constant options => (
    'verbose' => 'verbose',
);
sub run {
    my ($self, @arg) = @_;
    # افعل شيئًا
    $self->SUPER(@_); # App::CLI::Command::Help سيخرج POD لكل أمر
}

الوصف

"App::CLI" يرسل أوامر واجهة سطر الأوامر (CLI) إلى فئات الأوامر. كما يدعم الأوامر الفرعية والخيارات الخاصة بالأمر.

الطرق

get_opt([@config], %opt_map)

يعطي خريطة خيارات، يتم معالجتها بواسطة Getopt::Long::Parser.

dispatch(@args)

واجهة المرسل

cmd_map($cmd)

يجد اسم الحزمة التي تنفذ الأمر المطلوب.

يتم البحث عن الأمر أولاً في "alias". إذا كان الاسم المستعار موجودًا ويشير إلى اسم حزمة يبدأ بالحرف "+"، فيتم إرجاع اسم الحزمة هذا (مطروحًا منه الـ "+"). هذا يجعل من الممكن تعيين الأوامر إلى أي حزم.

بخلاف ذلك، يتم البحث عن الحزمة في نتيجة استدعاء "commands"، ويتم بناء اسم الحزمة عن طريق تحويل الحرف الأول من اسم الأمر إلى حروف كبيرة، وإلحاقه باسم حزمة التطبيق نفسه.

إذا فشلت كلتا الطريقتين، ولم يتم تعيين الأمر إلى أي اسم حزمة، يتم إرجاع "undef" بدلاً من ذلك.

get_cmd($cmd, @arg)

إرجاع أمر فرعي من المستوى الأول عبر $ARGV[0].
