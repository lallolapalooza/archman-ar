App::CLI::Command::Commands(3) توثيق بيرل مقدم من المستخدم App::CLI::Command::Commands(3)

الاسم

App::CLI::Command::Commands - طباعة قائمة بالأوامر لتطبيقك

الملخص

package MyApp;
use base qw(App::CLI App::CLI::Command);
# اجعل تطبيقك يحصل على قائمة بالأوامر
use constant alias => (
    commands => '+App::CLI::Command::Commands',
);
1;
