aio_init(3) دليل وظائف المكتبة aio_init(3)

الاسم

aio_init - تهيئة الإدخال/الإخراج غير المتزامن

المكتبة

مكتبة الوقت الحقيقي (librt, -lrt)

الملخص

#define _GNU_SOURCE         /* انظر feature_test_macros(7) */
#include <aio.h>
void aio_init(const struct aioinit *init);

الوصف

تسمح الدالة aio_init() الخاصة بـ GNU للمستدعي بتوفير تلميحات ضبط لتطبيق POSIX AIO في glibc. استخدام هذه الدالة اختياري، ولكن لتكون فعالة، يجب استدعاؤها قبل استخدام أي وظائف أخرى في واجهة برمجة تطبيقات POSIX AIO.

يتم توفير معلومات الضبط في المخزن المؤقت الذي يشير إليه الوسيط init. هذا المخزن المؤقت هو بنية بالشكل التالي:


struct aioinit {
    int aio_threads;    /* الحد الأقصى لعدد الخيوط */
    int aio_num;        /* عدد العمليات المتزامنة المتوقعة */
    int aio_locks;      /* غير مستخدم */
    int aio_usedba;     /* غير مستخدم */
    int aio_debug;      /* غير مستخدم */
    int aio_numusers;   /* غير مستخدم */
    int aio_idle_time;  /* عدد الثواني قبل انتهاء الخيط الخامل
                           (منذ glibc 2.2) */
    int aio_reserved;
};

يتم استخدام الحقول التالية في بنية aioinit:

aio_threads
يحدد هذا الحقل الحد الأقصى لعدد خيوط العمل التي قد تستخدمها التطبيق. إذا تجاوز عدد عمليات الإدخال/الإخراج المعلقة هذا الحد، فسيتم وضع العمليات الزائدة في قائمة انتظار حتى يصبح خيط العمل متاحًا. إذا تم تحديد هذا الحقل بقيمة أقل من 1، فسيتم استخدام القيمة 1. القيمة الافتراضية هي 20.
aio_num
يجب أن يحدد هذا الحقل الحد الأقصى لعدد عمليات الإدخال/الإخراج المتزامنة التي يتوقع المستدعي إضافتها. إذا تم تحديد قيمة أقل من 32 لهذا الحقل، فستتم تقريبها إلى 32. القيمة الافتراضية هي 64.
aio_idle_time
يحدد هذا الحقل الوقت بالثواني الذي يجب أن ينتظره خيط العمل للحصول على طلبات أخرى قبل إنهائه، بعد الانتهاء من طلب سابق. القيمة الافتراضية هي 1.
