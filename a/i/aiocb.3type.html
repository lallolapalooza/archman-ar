aiocb(3type) aiocb(3type)

الاسم

aiocb — كتلة تحكم الإدخال/الإخراج غير المتزامن

المكتبة

مكتبة C القياسية (libc)

المُلخّص

#include <aio.h>
struct aiocb {
    int             aio_fildes;     /* واصف الملف */
    off_t           aio_offset;     /* إزاحة الملف */
    volatile void  *aio_buf;        /* موقع المخزن المؤقت */
    size_t          aio_nbytes;     /* طول النقل */
    int             aio_reqprio;    /* إزاحة أولوية الطلب */
    struct sigevent aio_sigevent;   /* رقم الإشارة وقيمتها */
    int             aio_lio_opcode; /* العملية المراد تنفيذها */
};
