argtable2(3)

الحزمة: argtable

التراخيص: LGPL-2.0-or-later

أنواع إدخالات جدول الوسائط

يقوم Argtable بتعريف سبعة أنواع من هياكل بيانات الوسائط التي يمكن استخدامها في إدخال جدول الوسائط.

خيارات سطر الأوامر الصحيحة

-i10, -n -5, --count=20, --values=1,2,3,4

struct arg_int
   {
   struct arg_hdr hdr;  /* رأس argtable الداخلي */
   int count;           /* عدد القيم الصحيحة المستلمة */
   int *ival;           /* مصفوفة القيم الصحيحة المحللة */
   };

تُستخدم هذه الوسائط لالتقاط القيم الصحيحة من سطر الأوامر. يمكن تمرير الوسائط المتعددة مفصولة بفواصل (على سبيل المثال، --values=1,2,3,4). تقوم الوسيطة ival بتخزين مؤشرات إلى القيم الصحيحة التي تم تحليلها. يجب تحديد حجم مصفوفة ival عند إنشاء الوسيطة باستخدام معلمة maxcount.

تلميح: من القانوني تعيين قيم افتراضية في مصفوفة ival قبل استدعاء الدالة arg_parse. لن يقوم Argtable بتعديل إدخالات ival التي لم يتم استلام وسيطة سطر أوامر لها.

تلميح: لا يُنصح بالوسائط الرقمية غير الموسومة لأن GNU getopt يخطئ في تفسير القيم السالبة (مثل -123) على أنها خيارات موسومة (مثل -1 -2 -3). الوسائط الموسومة (مثل -x -123، --tag=-123) لا تعاني من هذه المشكلة.

خيارات سطر الأوامر الحقيقية/المزدوجة

-x2.234, -y 7e-03, -z-3.3E+6, --pi=3.1415, --tolerance 1.0E-6

struct arg_dbl
   {
   struct arg_hdr hdr;  /* رأس argtable الداخلي */
   int count;           /* عدد القيم المستلمة في dval[] */
   double *dval;        /* مصفوفة القيم المزدوجة المحللة */
   };

نفس arg_int باستثناء أن القيم المحللة يتم تخزينها في dval كقيم مزدوجة.

خيارات سطر الأوامر النصية

-Dmacro, -t mytitle, -m "my message string", --title="hello world"

struct arg_str
   {
   struct arg_hdr hdr;  /* رأس argtable الداخلي */
   int count;           /* عدد السلاسل النصية المستلمة في sval[] */
   const char **sval;   /* مصفوفة مؤشرات إلى سلاسل الوسائط المحللة */
   };

نفس arg_int باستثناء أن مؤشرات إلى السلاسل النصية المحللة يتم إرجاعها في sval بدلاً من نسخة منفصلة من السلسلة النصية. في الواقع، تشير هذه المؤشرات فعليًا إلى مخازن السلسلة النصية الأصلية المخزنة في argv[]، لذلك لا ينبغي تعديل محتوياتها. ومع ذلك، من القانوني تهيئة مؤشرات السلسلة النصية في مصفوفة sval للإشارة إلى سلاسل نصية افتراضية يقدمها المستخدم قبل استدعاء arg_parse. سيقوم Argtable فقط بتعديل محتويات sval عند اكتشاف وسائط سطر الأوامر المطابقة.

خيارات سطر الأوامر للتعبيرات العادية

commit, update, --command=commit, --command=update

struct arg_rex
   {
   struct arg_hdr hdr;  /* رأس argtable الداخلي */
   int count;           /* عدد السلاسل النصية المستلمة في sval[] */
   const char **sval;   /* مصفوفة مؤشرات إلى سلاسل الوسائط المحللة */
   };

مشابه لـ arg_str باستثناء أنه يتم قبول قيم الوسائط النصية فقط إذا كانت تطابق تعبيرًا عاديًا محددًا مسبقًا. التعبيرات العادية مفيدة لمطابقة كلمات مفتاحية في سطر الأوامر، خاصة إذا كانت هناك حاجة إلى سلاسل نصية غير حساسة لحالة الأحرف أو مطابقة الأنماط. يتم تعريف التعبير العادي بواسطة معلمة pattern التي يتم تمريرها إلى مُنشئ arg_rex ويتم تقييمها باستخدام regex. يمكن التحكم في سلوكه عبر أعلام bit القياسية لـ regex. يتم تمرير هذه الأعلام إلى argtable عبر معلمة flags في مُنشئ arg_rex. ومع ذلك، فقط اثنان من أعلام regex القياسية ذات صلة بـ argtable، وهما REG_EXTENDED (استخدام تعبيرات عادية موسعة بدلاً من الأساسية) و REG_ICASE (تجاهل حالة الأحرف). يمكن دمج هذه الأعلام منطقيًا إذا رغبت في ذلك. انظر regex(3) لمزيد من التفاصيل حول مطابقة التعبيرات العادية.

قيود: لا يدعم Argtable وظائف arg_rex على منصات Microsoft Windows لأن مترجمات Microsoft لا تتضمن دعم regex الضروري كمعيار.

خيارات سطر الأوامر لأسماء الملفات

-o myfile, -Ihome/foo/bar, --input=~/doc/letter.txt, --name a.out

struct arg_file
   {
   struct arg_hdr hdr;      /* رأس argtable الداخلي */
   int count;               /* عدد سلاسل أسماء الملفات المستلمة */
   const char **filename;   /* مؤشر إلى سلسلة اسم الملف الكامل */
   const char **basename;   /* مؤشر إلى اسم الملف باستثناء المسار البادئ */
   const char **extension;  /* مؤشر إلى امتداد اسم الملف */
   };

مشابه لـ arg_str ولكن يُفترض أن تشير السلاسل النصية للوسائط إلى أسماء ملفات، وبالتالي يتم إجراء بعض التحليل الإضافي لفصل اسم الملف الأساسي والامتداد (إذا كانا موجودين). تخزن كل من المصفوفات الثلاث filename[]، basename[]، extension[] ما يصل إلى maxcount إدخال، ويشير الإدخال i'th لكل من هذه المصفوفات إلى مكونات مختلفة من نفس مخزن السلسلة النصية.

على سبيل المثال، -o /home/heitmann/mydir/foo.txt سيتم تحليلها على النحو التالي: 

filename[i]  = "/home/heitmann/mydir/foo.txt"
basename[i]  =                      "foo.txt"
extension[i] =                         ".txt"

إذا لم يكن لاسم الملف مسار بادئ، فإن اسم الملف الأساسي هو نفس اسم الملف. إذا لم يتم التعرف على امتداد، فسيتم تقديمه كـ NULL. تعتبر الامتدادات كل نص بدءًا من آخر نقطة في اسم الملف.

تلميح: يعامل Argtable أسماء الملفات كسلاسل نصية فقط ولا يحاول أبدًا فتحها كملفات أو إجراء أي بحث في الدلائل عليها.

خيارات سطر الأوامر للتاريخ/الوقت

12/31/04, -d 1982-11-28, --time 23:59

struct arg_date
   {
   struct arg_hdr hdr;  /* رأس argtable الداخلي */
   const char *format;  /* سلسلة تنسيق التاريخ المقدمة من المستخدم والتي تم تمريرها إلى المُنشئ */
   int count;           /* عدد طوابع التاريخ المستلمة في tmval[] */
   struct tm *tmval;    /* مصفوفة طوابع التاريخ */
   };

يقبل سلسلة نصية لطابع زمني من سطر الأوامر ويحولها إلى تنسيق struct tm باستخدام دالة النظام strptime. يتم تحديد تنسيق الوقت بواسطة سلسلة format التي تم تمريرها إلى مُنشئ arg_date، ويتم تمريرها مباشرة إلى strptime. انظر strptime(3) لمزيد من التفاصيل حول سلسلة التنسيق.

قيود: لا يدعم Argtable وظائف arg_date على منصات Microsoft Windows لأن مترجمات Microsoft لا تتضمن دعم strptime الضروري كمعيار.

خيارات الملاحظات

struct arg_rem
   {
   struct arg_hdr hdr;  /* رأس argtable الداخلي */
   };

هيكل arg_rem هو هيكل وهمي بمعنى أنه لا يمثل خيار سطر أوامر ليتم تحليله. بدلاً من ذلك، يوفر وسيلة لتضمين سلاسل datatype و glossary إضافية في مخرجات الدوال arg_print_syntax، arg_print_syntaxv، و arg_print_glossary. على هذا النحو، يمكن استخدام هياكل arg_rem في جدول الوسائط لإدراج سطور إضافية من النص في أوصاف المسرد أو لإدراج حقول نصية إضافية في وصف بناء الجملة.

خيار نهاية الجدول

struct arg_end
   {
   struct arg_hdr hdr;  /* رأس argtable الداخلي */
   int count;           /* عدد الأخطاء المستلمة */
   int *error;          /* مصفوفة رموز الأخطاء */
   void **parent;       /* مؤشرات إلى خيارات سطر الأوامر الخاطئة */
   const char **argval; /* مؤشرات إلى قيم وسائط سطر الأوامر الخاطئة */
   };
يجب أن يحتوي كل جدول وسائط على هيكل arg_end كآخر إدخال له.
يشير إلى نهاية جدول الوسائط ويخزن رموز الأخطاء التي تم إنشاؤها بواسطة المحلل أثناء معالجته لجدول الوسائط.
تحدد معلمة maxerrors التي تم تمريرها إلى مُنشئ arg_end الحد الأقصى لعدد الأخطاء التي يمكن للهيكل تخزينها.
يتم تجاهل أي أخطاء أخرى واستبدالها برمز الخطأ الفردي ARG_ELIMIT الذي يتم الإبلاغ عنه لاحقًا للمستخدم بالرسالة "too many errors".
حد maxerrors يبلغ 20 يعتبر معقولًا جدًا.
ستقوم الدالة arg_print_errors بطباعة الأخطاء المخزنة
في هيكل arg_end بنفس الترتيب الذي حدثت به،
لذلك لا داعي لفهم التفاصيل الداخلية لهيكل arg_end.

مرجع الوظائف


int arg_nullcheck (void **argtable)

يعيد قيمة غير صفرية إذا كان يحتوي مصفوفة argtable[] على أي إدخالات NULL حتى إدخال arg_end* المنتهي. يعيد صفرًا بخلاف ذلك.

int arg_parse (int argc, char **argv, void **argtable)

يحلل وسائط سطر الأوامر في argv[] باستخدام بناء الجملة لسطر الأوامر المحدد في argtable[]، ويعيد عدد الأخطاء التي تمت مواجهتها. يتم تسجيل تفاصيل الخطأ في هيكل arg_end الخاص بجدول الوسائط الذي يمكن عرض الأخطاء منه لاحقًا باستخدام الدالة arg_print_errors. عند التحليل الناجح، ستحتوي هياكل arg_xxx المشار إليها في argtable[] على قيم الوسائط المستخرجة من سطر الأوامر.

void arg_print_option (FILE *fp, const char *shortopts, const char *longopts, const char *datatype, const char *suffix)

تطبع هذه الدالة بناء الجملة للوسيطة، كما في -K|--scalar=<int>، حيث يتم إعطاء الخيارات القصيرة، والخيارات الطويلة، ونوع البيانات كمعلمات لهذه الدالة. تُستخدم بشكل أساسي داخل دوال errorfn الخاصة بهياكل arg_xxx كوسيلة لعرض بناء الجملة للوسيطة داخل رسائل الخطأ. ومع ذلك، يمكن استخدامه أيضًا في كود المستخدم إذا رغبت في ذلك. يتم توفير السلسلة suffix كوسيلة ملائمة لإلحاق أسطر جديدة وما شابه ذلك بنهاية العرض ويمكن تقديمها كـ NULL إذا لم تكن مطلوبة.

void arg_print_syntax (FILE *fp, void **argtable, const char *suffix)

يطبع بناء الجملة لسطر الأوامر بأسلوب GNU لجدول الوسائط المحدد، كما في: [-abcv] [--scalar=<n>] [-o myfile] <file>
[<file>]
يتم توفير السلسلة suffix كوسيلة ملائمة لإلحاق أسطر جديدة وما شابه ذلك بنهاية العرض ويمكن تقديمها كـ NULL إذا لم تكن مطلوبة.
