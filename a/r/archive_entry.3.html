archive_entry(3)

الحزمة: libarchive

التراخيص: BSD-2-Clause

archive_entry(3) دليل وظائف المكتبة archive_entry(3)

الاسم

archive_entry_clear، archive_entry_clone، archive_entry_free، archive_entry_new — وظائف لإدارة أوصاف إدخالات الأرشيف

المكتبة

مكتبة الأرشيف المتدفقة (libarchive, -larchive)

الملخص

#include <archive_entry.h>

struct archive_entry *
archive_entry_clear(struct archive_entry *);

struct archive_entry *
archive_entry_clone(struct archive_entry *);

void
archive_entry_free(struct archive_entry *);

struct archive_entry *
archive_entry_new(void);

الوصف

تقوم هذه الوظائف بإنشاء ومعالجة كائنات البيانات التي تمثل إدخالات داخل أرشيف. يمكنك التفكير في `struct archive_entry` كإصدار قوي من `struct stat`: فهو يتضمن كل شيء من `struct stat` بالإضافة إلى المسار المرتبط به، وأسماء المجموعة والمستخدم النصية، وما إلى ذلك. تُستخدم هذه الكائنات بواسطة libarchive(3) لتمثيل البيانات الوصفية المرتبطة بإدخال معين في أرشيف.

الإنشاء والتدمير

توجد وظائف لتخصيص وتدمير ومسح ونسخ كائنات archive_entry:

archive_entry_clear()
يمسح الكائن، ويعيد تعيين جميع الحقول الداخلية إلى نفس الحالة التي كانت عليها الكائنات التي تم إنشاؤها حديثًا. هذا متاح للسماح لك بإعادة استخدام الكائنات بسرعة دون إهدار الذاكرة.
archive_entry_clone()
عملية نسخ عميق؛ يتم تكرار جميع حقول النص.
archive_entry_free()
يحرر كائن `struct archive_entry`.
archive_entry_new()
يخصص ويعيد كائن `struct archive_entry` فارغ.

مجموعات الوظائف

نظرًا للعدد الكبير من الوظائف، يمكن العثور على وظائف الوصول في صفحات الدليل مجمعة حسب الغرض.

archive_entry_acl(3)
معالجة قوائم التحكم في الوصول
archive_entry_paths(3)
معالجة أسماء المسارات
archive_entry_perms(3)
معالجة المستخدم والمجموعة والوضع
archive_entry_stat(3)
الوظائف التي ليست في المجموعات الأخرى والنسخ من وإلى `struct stat`.
archive_entry_time(3)
معالجة حقول الوقت

تقوم معظم الوظائف بتعيين أو قراءة الإدخالات في كائن. هذه الوظائف لها أحد الأشكال التالية:

archive_entry_set_XXXX()
يخزن البيانات المقدمة في الكائن. على وجه الخصوص، بالنسبة للنصوص، يتم تخزين المؤشر، وليس النص المشار إليه.
archive_entry_copy_XXXX()
كما هو الحال أعلاه، باستثناء أنه يتم نسخ البيانات المشار إليها إلى الكائن.
archive_entry_XXXX()
يعيد البيانات المحددة. في حالة النصوص، يتم إرجاع مؤشر مؤهل `const` إلى النص.
يمكن تعيين بيانات النص أو الوصول إليها كسلاسل أحرف عريضة أو كسلاسل `char` عادية. الوظائف التي تستخدم سلاسل أحرف عريضة تنتهي باللاحقة _w. لاحظ أن هذه تمثيلات مختلفة لنفس البيانات: على سبيل المثال، إذا قمت بتخزين سلسلة ضيقة وقراءة السلسلة العريضة المقابلة، فسيقوم الكائن بالتحويل بشفافية بين التنسيقات باستخدام الإعدادات المحلية الحالية. وبالمثل، إذا قمت بتخزين سلسلة عريضة ثم قمت بتخزين سلسلة ضيقة لنفس البيانات، فسيتم تجاهل السلسلة العريضة التي تم تعيينها سابقًا لصالح البيانات الجديدة.
