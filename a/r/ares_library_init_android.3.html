ares_library_init_android(3)

الحزمة: c-ares

التراخيص: MIT

ARES_LIBRARY_INIT_ANDROID(3) دليل وظائف المكتبة ARES_LIBRARY_INIT_ANDROID(3)

الاسم

ares_library_init_android - تهيئة مكتبة c-ares لنظام Android

المُلخَّص

#include <ares.h>
int ares_library_init_android(jobject connectivity_manager)
int ares_library_android_initialized();
void ares_library_init_jvm(JavaVM *jvm)

الوصف

تقوم الدالة ares_library_init_android(3) بإجراء التهيئة اللازمة داخليًا لمكتبة c-ares عند استخدامها على نظام Android. يمكن القيام بذلك في أي وقت بعد ares_library_init(3). يجب أن تتم بعد ares_library_init_jvm. يجب استدعاء ares_library_init_android قبل أن تعمل وظيفة تحليل DNS على Android 8 (Oreo) أو أحدث عندما يتم تعيين targetSdkVersion إلى 26+.

اعتبارًا من Android 8 (API level 26)، أصبح الحصول على معلومات خادم DNS أكثر تقييدًا ولا يمكن الوصول إليه إلا باستخدام مدير الاتصال (Connectivity Manager). من الضروري تمرير مدير الاتصال إلى c-ares عبر JNI. بالإضافة إلى ذلك، يجب أن يكون إذن ACCESS_NETWORK_STATE موجودًا في تطبيق Android.

أنظمة Android الأقدم من 8 لا تحتاج إلى تهيئة لأنها أقل تقييدًا. ومع ذلك، هذا قيد وقت التشغيل وليس وقت الترجمة. يجب إجراء تهيئة Android المناسبة بغض النظر عن إصدار Android المستهدف.

ستتم إلغاء التهيئة من خلال ares_library_cleanup(3).

تسمح الدالة ares_library_init_jvm للمستدعي بتسجيل JVM مع c-ares. تهدف إلى استدعائها أثناء JNI_OnLoad لأنك مضمون أن يكون لديك JVM في تلك الدالة. JVM مطلوب لاستخدام مدير الاتصال المسجل باستخدام ares_library_init_android(3). يجب استدعاء هذا قبل ares_library_init_android(3).

يمكن استخدام الدالة ares_library_android_initialized للتحقق مما إذا كانت c-ares قد تم تهيئتها للاستخدام مع Android.

قيم الإرجاع

سيتم إرجاع ARES_SUCCESS عند النجاح، وإلا سيتم إرجاع رمز خطأ.

سلامة الخيوط

وظائف التهيئة هذه ليست آمنة للخيوط. يجب عليك استدعاؤها مرة واحدة بعد بدء البرنامج، ولكن يجب القيام بهذا الاستدعاء قبل أن يبدأ البرنامج أي خيوط أخرى. هذا مطلوب لتجنب احتمالات شروط السباق في تهيئة المكتبة، وأيضًا بسبب حقيقة أن هذه قد تستدعي وظائف من مكتبات أخرى غير آمنة للخيوط، ويمكن أن تتعارض مع أي خيط آخر يستخدم بالفعل هذه المكتبات الأخرى.

JNI

الوصول إلى مدير الاتصال عبر Java:

قم بتسجيل ares_library_android_init.

  static JNINativeMethod funcs[] = {
  { "initialize_native",     "(Landroid/net/ConnectivityManager;)I",
    (void *)&ares_library_init_android}
  };
  JNIEXPORT jint JNICALL JNI_OnLoad(JavaVM *vm, void *reserved)
  {
    JNIEnv *env = NULL;
    jclass  cls = NULL;
    jint    res;
    if ((*vm)->GetEnv(vm, (void **)&env, JNI_VERSION_1_6) != JNI_OK)
      return -1;
    cls = (*env)->FindClass(env, JNIT_CLASS);
    if (cls == NULL)
      return -1;
    res = (*env)->RegisterNatives(env, cls, funcs, sizeof(funcs)/sizeof(funcs[0]));
    if (res != 0)
      return -1;
    ares_library_init_jvm(vm);
    return JNI_VERSION_1_6;
  }
استدعاء الدالة المسجلة من Java: 
  public class MyObject {
    static {
      System.loadLibrary("cares");
    }
    private static native boolean initialize_native(ConnectivityManager
      connectivity_manager);
    public static boolean initialize(Context context) {
      initialize_native((ConnectivityManager)context.getSystemService(Context.CONNECTIVITY_SERVICE));
    }
  }

تهيئة مدير الاتصال في JNI مباشرة باستخدام سياق Android. يفترض أنه تم تسجيل JVM بالفعل من خلال JNI_OnLoad.

  void initialize(jobject android_context)
  {
    jclass obj_cls = jni_get_class(env, "android/content/Context");
    jmethodID obj_mid = jni_get_method_id(env, obj_cls, "getSystemService", "(Ljava/lang/String;)Ljava/lang/Object;");
    jfieldID fid = (*env)->GetStaticFieldID(env, obj_cls, "CONNECTIVITY_SERVICE", "Ljava/lang/String;");
    jstring str = (*env)->GetStaticObjectField(env, obj_cls, fid);
    connectivity_manager = (*env)->CallObjectMethod(env, android_context, obj_mid, str);
    if (connectivity_manager == NULL)
      return;
    ares_library_init_android(connectivity_manager);
  }
