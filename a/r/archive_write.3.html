الحزمة: libarchive

التراخيص: BSD-2-Clause

ARCHIVE_WRITE(3) دليل وظائف المكتبة ARCHIVE_WRITE(3)

الاسم

archive_write — وظائف لإنشاء الأرشيفات

المكتبة

مكتبة الأرشيف المتدفق (libarchive, -larchive)

الملخص

#include <archive.h>

الوصف

توفر هذه الوظائف واجهة برمجة تطبيقات كاملة لإنشاء ملفات الأرشيف المتدفقة. العملية العامة هي أولاً إنشاء كائن struct archive، وتعيين أي خيارات مرغوبة، وتهيئة الأرشيف، وإضافة الإدخالات، ثم إغلاق الأرشيف وتحرير جميع الموارد.

إنشاء كائن الأرشيف

انظر archive_write_new(3).

لكتابة أرشيف، يجب عليك أولاً الحصول على كائن struct archive مهيأ من archive_write_new().

تمكين المرشحات والتنسيقات، وتكوين حجم الكتلة والحشو

انظر archive_write_filter(3), archive_write_format(3) و archive_write_blocksize(3).

يمكنك بعد ذلك تعديل هذا الكائن للعمليات المرغوبة باستخدام وظائف archive_write_set_XXX() المختلفة. على وجه الخصوص، ستحتاج إلى استدعاء وظائف archive_write_add_XXX() و archive_write_set_XXX() المناسبة لتمكين دعم الضغط والتنسيق المقابل.

تعيين الخيارات

انظر archive_write_set_options(3).

فتح الأرشيف

انظر archive_write_open(3).

بمجرد إعداد كائن struct archive، قم باستدعاء archive_write_open() لفتح الأرشيف فعليًا وإعداده للكتابة. هناك عدة أشكال من هذه الوظيفة؛ يتوقع الأساسي منها منك توفير مؤشرات لعدة وظائف يمكنها توفير كتل بايت من الأرشيف. هناك أشكال ملائمة تسمح لك بتحديد اسم ملف، واصف ملف، كائن FILE *، أو كتلة ذاكرة للكتابة منها بيانات الأرشيف.

إنتاج الأرشيف

انظر archive_write_header(3) و archive_write_data(3).

تتم كتابة إدخالات الأرشيف الفردية في عملية من ثلاث خطوات: تقوم أولاً بتهيئة بنية struct archive_entry بمعلومات حول الإدخال الجديد. كحد أدنى، يجب عليك تعيين مسار الإدخال وتوفير struct stat مع حقل st_mode صالح، الذي يحدد نوع الكائن، وحقل st_size، الذي يحدد حجم جزء البيانات للكائن.

تحرير الموارد

انظر archive_write_free(3).

بعد كتابة جميع الإدخالات، استخدم وظيفة archive_write_free() لتحرير جميع الموارد.

أمثلة

يوضح المخطط التالي الاستخدام الأساسي للمكتبة. في هذا المثال، وظائف الاستدعاء هي مجرد أغلفة حول استدعاءات نظام open(2)، write(2)، و close(2) القياسية.

#ifdef __linux__
#define	_FILE_OFFSET_BITS 64
#endif
#include <sys/stat.h>
#include <archive.h>
#include <archive_entry.h>
#include <fcntl.h>
#include <stdlib.h>
#include <unistd.h>

struct mydata {
  const char *name;
  int fd;
};

int
myopen(struct archive *a, void *client_data)
{
  struct mydata *mydata = client_data;

  mydata->fd = open(mydata->name, O_WRONLY | O_CREAT, 0644);
  if (mydata->fd >= 0)
    return (ARCHIVE_OK);
  else
    return (ARCHIVE_FATAL);
}

la_ssize_t
mywrite(struct archive *a, void *client_data, const void *buff, size_t n)
{
  struct mydata *mydata = client_data;

  return (write(mydata->fd, buff, n));
}

int
myclose(struct archive *a, void *client_data)
{
  struct mydata *mydata = client_data;

  if (mydata->fd > 0)
    close(mydata->fd);
  return (0);
}

void
write_archive(const char *outname, const char **filename)
{
  struct mydata *mydata = malloc(sizeof(struct mydata));
  struct archive *a;
  struct archive_entry *entry;
  struct stat st;
  char buff[8192];
  int len;
  int fd;

  a = archive_write_new();
  mydata->name = outname;
  /* Set archive format and filter according to output file extension.
   * If it fails, set default format. Platform depended function.
   * See supported formats in archive_write_set_format_filter_by_ext.c */
  if (archive_write_set_format_filter_by_ext(a, outname) != ARCHIVE_OK)  {
    archive_write_add_filter_gzip(a);
    archive_write_set_format_ustar(a);
  }
  archive_write_open(a, mydata, myopen, mywrite, myclose);
  while (*filename) {
    stat(*filename, &st);
    entry = archive_entry_new();
    archive_entry_copy_stat(entry, &st);
    archive_entry_set_pathname(entry, *filename);
    archive_write_header(a, entry);
    if ((fd = open(*filename, O_RDONLY)) != -1) {
      len = read(fd, buff, sizeof(buff));
      while (len > 0) {
        archive_write_data(a, buff, len);
        len = read(fd, buff, sizeof(buff));
      }
      close(fd);
    }
    archive_entry_free(entry);
    filename++;
  }
  archive_write_free(a);
}

int main(int argc, const char **argv)
{
  const char *outname;
  argv++;
  outname = *argv++;
  write_archive(outname, argv);
  return 0;
}

DESCRIPTION

تهدف هذه الوظائف إلى توفير واجهة لمكتبة libarchive لتصدير بيانات الأرشيف. المكتبة قادرة على كتابة الأرشيفات بتنسيقات مختلفة، بما في ذلك tar، و zip، و iso9660، و mtree. يمكن أيضًا ضغط الأرشيفات باستخدام gzip، و bzip2، و lzma، و xz، و zstd. تقوم هذه الوظائف بتغليف المنطق اللازم لتحديد التنسيق، والضغط، وإدارة مساحة العمل، والتعامل مع الأخطاء.

تعتبر خاصية `STAR_ENTRY_NONCONTIG` في `struct archive_entry` مهمة. إذا تم تعيين هذه الخاصية، فهذا يعني أن السجل لم يكن مستمرًا في مساحة العمل المصدر. يجب على التطبيقات التي تستخدم هذه الخاصية التأكد من أن الوجهة النهائية تدعم مثل هذه الحالات.

يجب ملاحظة أن هذه المكتبة متوافقة مع مستندات Joerg Schilling `star` archiver. قد لا تتعرف التنفيذات الأخرى على هذه المفاتيح، وبالتالي لن تكون قادرة على استعادة عقد الأجهزة ذات أرقام الأجهزة الكبيرة بشكل صحيح من الأرشيفات التي تم إنشاؤها بواسطة هذه المكتبة.
