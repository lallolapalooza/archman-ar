auparse_feed(3)

الحزمة: audit

التراخيص: GPL-2.0-or-later, LGPL-2.0-or-later

AUPARSE_FEED(3) واجهة برمجة تطبيقات تدقيق لينكس AUPARSE_FEED(3)

الاسم

auparse_feed - تغذية البيانات إلى المحلل

الملخص

#include <auparse.h>

int auparse_feed(auparse_state_t *au,constchar*"data,size_t"data_len);"
au
حالة محلل التدقيق
data
مخزن مؤقت للبيانات لتغذيتها إلى المحلل، طوله data_len بايت. يتم نسخ البيانات إلى المحلل، عند العودة يمكن للمتصل تحرير أو إعادة استخدام مخزن البيانات المؤقت.
data_len
عدد البايتات في data

الوصف

auparse_feed يوفر بيانات جديدة للمحلل لاستهلاكها. يجب أن يكون قد تم استدعاء auparse_init() بنوع مصدر AUSOURCE_FEED ومؤشر NULL.

يستهلك المحلل أكبر قدر ممكن من البيانات، مستدعيًا دالة استدعاء مزودة من قبل المستخدم محددة باستخدام auparse_add_callback بنوع حدث cb_event_type من AUPARSE_CB_EVENT_READY في كل مرة يتعرف فيها المحلل على حدث كامل في تدفق البيانات. البيانات التي لم يتم تحليلها بالكامل ستظل قائمة وسيتم إلحاقها ببيانات التغذية التالية. بعد تغذية جميع البيانات إلى المحلل، يجب استدعاء auparse_flush_feed للإشارة إلى نهاية بيانات الإدخال وتمرير أي بيانات تحليل معلقة عبر نظام التحليل.

قيمة الإرجاع

يعيد -1 في حالة حدوث خطأ؛ بخلاف ذلك، يعيد 0 للنجاح.

مثال

void
auparse_callback(auparse_state_t *au, auparse_cb_event_t cb_event_type,
                 void *user_data)
{
    int *event_cnt = (int *)user_data;
    if (cb_event_type == AUPARSE_CB_EVENT_READY) {
        if (auparse_first_record(au) <= 0) return;
        printf("event: %d\n", *event_cnt);
        printf("records:%d\n", auparse_get_num_records(au));
        do {
            printf("fields:%d\n", auparse_get_num_fields(au));
            printf("type=%d ", auparse_get_type(au));
            const au_event_t *e = auparse_get_timestamp(au);
            if (e == NULL) return;
            printf("event time: %lu.%u:%lu\n",
                    (long unsigned)e->sec, e->milli, e->serial);
            auparse_first_field(au);
            do {
                printf("%s=%s (%s)\n", auparse_get_field_name(au),
                       auparse_get_field_str(au),
                       auparse_interpret_field(au));
            } while (auparse_next_field(au) > 0);
            printf("\n");
        } while(auparse_next_record(au) > 0);
        (*event_cnt)++;
    }
}
main(int argc, char **argv)
{
    char *filename = argv[1];
    FILE *fp;
    char buf[256];
    size_t len;
    int *event_cnt = malloc(sizeof(int));
    au = auparse_init(AUSOURCE_FEED, 0);
    auparse_set_eoe_timeout(2);
    *event_cnt = 1;
    auparse_add_callback(au, auparse_callback, event_cnt, free);
    if ((fp = fopen(filename, "r")) == NULL) {
        fprintf(stderr, "could not open '%s', %s\n", filename, strerror(errno));
        return 1;
    }
    while ((len = fread(buf, 1, sizeof(buf), fp))) {
        auparse_feed(au, buf, len);
    }
    auparse_flush_feed(au);
    auparse_destroy(au);
}
