aarch64-linux-gnu-ld(1)

الحزمة: aarch64-linux-gnu-binutils

التراخيص: GPL

"libgcc"، "libstd++"، "libmingw32"، أو "crtX.o" سيتم تصديرها. الرموز التي تبدأ أسماؤها بـ "__rtti_" أو "__builtin_" لن يتم تصديرها، للمساعدة مع DLLs بلغة C++. أخيرًا، هناك قائمة شاملة بالرموز الخاصة بـ cygwin والتي لا يتم تصديرها (من الواضح أن هذا ينطبق فقط عند بناء DLLs لأهداف cygwin). هذه الاستبعادات الخاصة بـ cygwin هي: "_cygwin_dll_entry@12"، "_cygwin_crt0_common@8"، "_cygwin_noncygwin_dll_entry@12"، "_fmode"، "_impure_ptr"، "cygwin_attach_dll"، "cygwin_premain0"، "cygwin_premain1"، "cygwin_premain2"، "cygwin_premain3"، و "environ". الملاحظات التالية تتعلق بالتنفيذ الأصلي للميزة وهي قديمة الآن لأهداف Cygwin و MinGW.

ملاحظة: سيؤدي استخدام امتداد "auto-import" إلى جعل قسم النص في ملف الصورة قابلاً للكتابة. هذا لا يتوافق مع مواصفات تنسيق PE-COFF المنشورة بواسطة Microsoft.

ملاحظة - سيؤدي استخدام امتداد "auto-import" أيضًا إلى وضع البيانات للقراءة فقط والتي عادة ما توضع في قسم .rdata في قسم .data بدلاً من ذلك. هذا من أجل الالتفاف حول مشكلة مع الثوابت الموصوفة هنا: http://www.cygwin.com/ml/cygwin/2004-09/msg01101.html

بشكل عام، "auto-import" سيعمل "بشكل جيد" - ولكن في بعض الأحيان قد ترى هذه الرسالة:

"لا يمكن استيراد المتغير '' تلقائيًا. يرجى قراءة وثائق --enable-auto-import الخاصة بـ ld لمزيد من التفاصيل."

تحدث هذه الرسالة عندما يقوم بعض (الـ)تعبير بالوصول إلى عنوان يتم إعطاؤه في النهاية بواسطة مجموع ثابتين (جداول استيراد Win32 تسمح بواحد فقط). الحالات التي قد تحدث فيها تشمل الوصول إلى حقول عضو للمتغيرات الهيكلية المستوردة من DLL، وكذلك استخدام فهرس ثابت في متغير مصفوفة مستورد من DLL. قد يؤدي أي متغير متعدد الكلمات (مصفوفات، هياكل، long long، إلخ) إلى حدوث حالة الخطأ هذه. ومع ذلك، بغض النظر عن نوع البيانات المحدد للمتغير المصدر، سيقوم ld دائمًا بالكشف عنه وإصدار التحذير والخروج.

هناك عدة طرق لمعالجة هذه الصعوبة، بغض النظر عن نوع البيانات للمتغير المصدر:

إحدى الطرق هي استخدام مفتاح --enable-runtime-pseudo-reloc. هذا يترك مهمة تعديل المراجع في رمز العميل لبيئة وقت التشغيل، لذا فإن هذه الطريقة تعمل فقط عندما تدعم بيئة وقت التشغيل هذه الميزة.

الحل الثاني هو إجبار أحد "الثوابت" على أن يكون متغيرًا - أي غير معروف ولا يمكن تحسينه في وقت الترجمة. بالنسبة للمصفوفات، هناك احتمالين: أ) جعل الـ indexee (عنوان المصفوفة) متغيرًا، أو ب) جعل الفهرس "الثابت" متغيرًا. وبالتالي:

extern type extern_array[];
extern_array[1] -->
   { volatile type *t=extern_array; t[1] }

أو

extern type extern_array[];
extern_array[1] -->
   { volatile int t=1; extern_array[t] }

بالنسبة للهياكل (ومعظم أنواع البيانات الأخرى متعددة الكلمات) الخيار الوحيد هو جعل الهيكل نفسه (أو long long، أو ...) متغيرًا:

extern struct s extern_struct;
extern_struct.field -->
   { volatile struct s *t=&extern_struct; t->field }

أو

extern long long extern_ll;
extern_ll -->
  { volatile long long * local_ll=&extern_ll; *local_ll }

الطريقة الثالثة للتعامل مع هذه الصعوبة هي التخلي عن "auto-import" للرمز المسبب للمشكلة وتمييزه بـ "__declspec(dllimport)". ومع ذلك، من الناحية العملية، يتطلب ذلك استخدام #defines في وقت الترجمة للإشارة إلى ما إذا كنت تقوم ببناء DLL، أو بناء رمز عميل سيرتبط بـ DLL، أو مجرد بناء/ربط مكتبة ثابتة. عند إنشاء
