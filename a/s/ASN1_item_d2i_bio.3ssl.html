ASN1_item_d2i_bio(3ssl)

الحزمة: openssl

التراخيص: Apache-2.0

ASN1_ITEM_D2I_BIO(3ssl) OpenSSL ASN1_ITEM_D2I_BIO(3ssl)

الاسم

ASN1_item_d2i_ex, ASN1_item_d2i, ASN1_item_d2i_bio_ex, ASN1_item_d2i_bio, ASN1_item_d2i_fp_ex, ASN1_item_d2i_fp, ASN1_item_i2d_mem_bio, ASN1_item_pack, ASN1_item_unpack_ex, ASN1_item_unpack - فك ترميز وترميز هياكل ASN.1 المشفرة بـ DER

الملخص

#include <openssl/asn1.h>
ASN1_VALUE *ASN1_item_d2i_ex(ASN1_VALUE **pval, const unsigned char **in,
                             long len, const ASN1_ITEM *it,
                             OSSL_LIB_CTX *libctx, const char *propq);
ASN1_VALUE *ASN1_item_d2i(ASN1_VALUE **pval, const unsigned char **in,
                          long len, const ASN1_ITEM *it);
void *ASN1_item_d2i_bio_ex(const ASN1_ITEM *it, BIO *in, void *x,
                           OSSL_LIB_CTX *libctx, const char *propq);
void *ASN1_item_d2i_bio(const ASN1_ITEM *it, BIO *in, void *x);
void *ASN1_item_d2i_fp_ex(const ASN1_ITEM *it, FILE *in, void *x,
                          OSSL_LIB_CTX *libctx, const char *propq);
void *ASN1_item_d2i_fp(const ASN1_ITEM *it, FILE *in, void *x);
BIO *ASN1_item_i2d_mem_bio(const ASN1_ITEM *it, const ASN1_VALUE *val);
ASN1_STRING *ASN1_item_pack(void *obj, const ASN1_ITEM *it, ASN1_STRING **oct);
void *ASN1_item_unpack(const ASN1_STRING *oct, const ASN1_ITEM *it);
void *ASN1_item_unpack_ex(const ASN1_STRING *oct, const ASN1_ITEM *it,
                         OSSL_LIB_CTX *libctx, const char *propq);

الوصف

تقوم ASN1_item_d2i_ex() بفك ترميز محتويات البيانات المخزنة في *in بطول len والتي يجب أن تكون بنية ASN.1 مشفرة بـ DER، باستخدام قالب ASN.1 it. تضع النتيجة في *pval إلا إذا كان pval فارغًا (NULL). إذا كان *pval غير فارغ عند الإدخال، فسيتم إعادة استخدام ASN1_VALUE الموجود هناك. بخلاف ذلك، سيتم تخصيص ASN1_VALUE جديد. إذا كانت هناك حاجة لجلب أي خوارزميات أثناء العملية، فستستخدم سياق مكتبة OSSL_LIB_CTX المقدم في المعلمة libctx وسلسلة استعلام الخصائص في propq. انظر "جلب الخوارزميات" في crypto(7) لمزيد من المعلومات حول جلب الخوارزميات. عند الخروج، سيتم تحديث *in للإشارة إلى البايت التالي في المخزن المؤقت بعد البنية التي تم فك ترميزها.

ASN1_item_d2i() مماثلة لـ ASN1_item_d2i_ex() باستثناء أنها تستخدم سياق مكتبة OSSL_LIB_CTX الافتراضي (أي فارغ NULL) مع سلسلة استعلام خصائص فارغة (NULL).

تقوم ASN1_item_d2i_bio_ex() بفك ترميز محتويات BIO الإدخال الخاص بها in، والذي يجب أن يكون بنية ASN.1 مشفرة بـ DER، باستخدام قالب ASN.1 it وتضع النتيجة في *pval إلا إذا كان pval فارغًا (NULL). إذا كان in فارغًا (NULL)، فترجع NULL، وإلا فإنها ترجع مؤشرًا إلى البنية التي تم تحليلها. إذا كانت هناك حاجة لجلب أي خوارزميات أثناء العملية، فستستخدم سياق مكتبة OSSL_LIB_CTX المقدم في المعلمة libctx وسلسلة استعلام الخصائص في propq. انظر "جلب الخوارزميات" في crypto(7) لمزيد من المعلومات حول جلب الخوارزميات.

ASN1_item_d2i_bio() مماثلة لـ ASN1_item_d2i_bio_ex() باستثناء أنها تستخدم سياق مكتبة OSSL_LIB_CTX الافتراضي (أي فارغ NULL) مع سلسلة استعلام خصائص فارغة (NULL).

ASN1_item_d2i_fp_ex() مماثلة لـ ASN1_item_d2i_bio_ex() باستثناء أنها توفر مؤشر ملف FILE بدلاً من BIO.

ASN1_item_d2i_fp() مماثلة لـ ASN1_item_d2i_fp_ex() باستثناء أنها تستخدم سياق مكتبة OSSL_LIB_CTX الافتراضي (أي فارغ NULL) مع سلسلة استعلام خصائص فارغة (NULL).

تقوم ASN1_item_i2d_mem_bio() بترميز قيمة ASN.1 المعطاة val باستخدام قالب ASN.1 it وترجع النتيجة في BIO ذاكرة.

تقوم ASN1_item_pack() بترميز قيمة ASN.1 المعطاة في obj باستخدام قالب ASN.1 it وترجع كائن ASN1_STRING. إذا كان *oct المقدم غير فارغ (NULL)، فسيتم استخدامه لتخزين النتيجة المرجعة، وإلا فسيتم إنشاء كائن ASN1_STRING جديد. إذا كان oct غير فارغ (NULL) و *oct فارغًا (NULL)، فسيتم أيضًا تعيين القيمة المرجعة في *oct. إذا كان هناك خطأ، فلن يتم تحرير ASN1_STRING المقدم اختياريًا، ولكن قد يتم مسح القيمة السابقة عند استدعاء ASN1_STRING_set0(*oct, NULL, 0) داخليًا.

تستخدم ASN1_item_unpack() ASN1_item_d2i() لفك ترميز ASN1_STRING المشفرة بـ DER oct باستخدام قالب ASN.1 it.

ASN1_item_unpack_ex() مشابه لـ ASN1_item_unpack()، ولكنه يستخدم ASN1_item_d2i_ex() بحيث يمكن استخدام libctx و propq عند إجراء جلب الخوارزميات.

قيم الإرجاع

ASN1_item_d2i_bio()، ASN1_item_unpack_ex() و ASN1_item_unpack() ترجع مؤشرًا إلى ASN1_VALUE أو NULL عند حدوث خطأ.

ASN1_item_i2d_mem_bio() ترجع مؤشرًا إلى BIO ذاكرة أو NULL عند حدوث خطأ.

ASN1_item_pack() ترجع مؤشرًا إلى ASN1_STRING أو NULL عند حدوث خطأ.
