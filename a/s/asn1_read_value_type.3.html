asn1_read_value_type(3)

الحزمة: libtasn1

التراخيص: GPL-3.0-or-later, LGPL-2.1-or-later, GFDL-1.3-or-later

asn1_read_value_type(3) libtasn1 asn1_read_value_type(3)

الاسم

asn1_read_value_type - استدعاء برمجي للواجهة

الملخص

#include <libtasn1.h>

int asn1_read_value_type(asn1_node_const root, const char * name, void * ivalue, int * len, unsigned int * etype);

المعاملات

asn1_node_const root
مؤشر إلى بنية.
const char * name
اسم العنصر داخل بنية تريد قراءته.
void * ivalue
متجه سيحتوي على محتوى العنصر، يجب أن يكون مؤشرًا إلى خلايا ذاكرة مخصصة بالفعل (يمكن أن يكون NULL).
int * len
عدد البايتات للقيمة: value[0]..value[len-1]. يحتوي في البداية على sizeof value.
unsigned int * etype
نوع القيمة المقروءة (ASN1_ETYPE)

الوصف

يقوم بإرجاع نوع وقيمة عنصر واحد داخل بنية. إذا كان العنصر اختياريًا (OPTIONAL) وهذا الاستدعاء يرجع ASN1_ELEMENT_NOT_FOUND، فهذا يعني أن هذا العنصر لم يكن موجودًا في ترميز DER الذي أنشأ البنية. العنصر الأول في SEQUENCE_OF أو SET_OF يسمى ""?1"". العنصر الثاني ""?2"" وهكذا. إذا كان root المقدم هو عقدة لعنصر تسلسل معين، فإن الكلمة المفتاحية ""?CURRENT"" مقبولة أيضًا وتشير إلى عنصر التسلسل الحالي لهذه العقدة.

لاحظ أنه يمكن أن تكون هناك قيم صالحة بأطوال صفرية. في هذه الحالات، ستنجح هذه الدالة وسيكون len صفرًا.

عدد صحيح (INTEGER)

سيحتوي VALUE على عدد صحيح بصيغة المتمم الثنائي.

integer=-1 -> value[0]=0xFF , len=1. integer=1 -> value[0]=0x01 , len=1.

مُعد (ENUMERATED)

مثل عدد صحيح (INTEGER) (ولكن بأرقام غير سالبة فقط).

قيمة منطقية (BOOLEAN)

سيكون VALUE هو السلسلة النصية المنتهية بـ NULL "TRUE" أو "FALSE" و LEN=5 أو LEN=6.

معرف الكائن (OBJECT IDENTIFIER): سيكون VALUE سلسلة نصية منتهية بـ NULL مع كل رقم مفصول بنقطة (على سبيل المثال "1.2.3.543.1").

LEN = strlen(VALUE)+1

وقت UTC (UTCTIME)

سيكون VALUE سلسلة نصية منتهية بـ NULL بأحد هذه التنسيقات: "YYMMDDhhmmss+hh'mm'" أو "YYMMDDhhmmss-hh'mm'". LEN=strlen(VALUE)+1.

وقت عام (GENERALIZEDTIME)

سيكون VALUE سلسلة نصية منتهية بـ NULL بنفس التنسيق المستخدم لتعيين القيمة.

سلسلة بايت (OCTET STRING): سيحتوي VALUE على سلسلة البايتات وسيكون LEN هو عدد البايتات.

سلسلة عامة (GENERALSTRING)

سيحتوي VALUE على السلسلة العامة وسيكون LEN هو عدد البايتات.

سلسلة بت (BIT STRING): سيحتوي VALUE على سلسلة البتات منظمة بالبايتات وسيكون LEN هو عدد البتات.

اختيار (CHOICE)

إذا كان NAME يشير إلى نوع اختيار، فسيوفر VALUE البديل المحدد.

أي (ANY)

إذا كان NAME يشير إلى نوع ANY، فسيوفر VALUE ترميز DER للبنية المستخدمة فعليًا.

القيمة المرجعة

ASN1_SUCCESS إذا تم إرجاع القيمة، ASN1_ELEMENT_NOT_FOUND إذا كان name ليس عنصرًا صالحًا، ASN1_VALUE_NOT_FOUND إذا لم تكن هناك قيمة للعنصر المحدد، و ASN1_MEM_ERROR إذا لم يكن متجه القيمة كبيرًا بما يكفي لتخزين النتيجة، وفي هذه الحالة سيحتوي len على عدد البايتات المطلوبة. في حال كانت البيانات المخزنة ذات طول صفري، قد ترجع هذه الدالة ASN1_SUCCESS حتى لو كان len المقدم صفرًا.
