asn1_object_id_der(3)

الحزمة: libtasn1

التراخيص: GPL-3.0-or-later, LGPL-2.1-or-later, GFDL-1.3-or-later

asn1_object_id_der(3) libtasn1 asn1_object_id_der(3)

الاسم

asn1_object_id_der - وظيفة API

الملخص

#include <libtasn1.h>

int asn1_object_id_der(const char * str, unsigned char * der, int * der_len, unsigned flags);

الوسائط

const char * str
معرّف كائن بتنسيق رقمي، مفصولة بنقاط.
unsigned char * der
مخزن مؤقت لاحتواء الترميز المُرجع (قد يكون NULL).
int * der_len
في البداية، حجم der؛ سيحتوي على الحجم النهائي.
unsigned flags
يجب أن يكون صفرًا

الوصف

ينشئ ترميز DER لمعرف الكائن المقدم.

المُرجعات

ASN1_SUCCESS إذا كان ترميز DER صحيحًا، ASN1_VALUE_NOT_VALID إذا لم يكن str معرف OID صالحًا، ASN1_MEM_ERROR إذا لم يكن متجه der كبيرًا بما فيه الكفاية وفي هذه الحالة سيحتوي der_len على الطول المطلوب.
