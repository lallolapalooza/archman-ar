_syscall(2) دليل استدعاءات النظام _syscall(2)

الاسم

_syscall - استدعاء استدعاء نظام بدون دعم مكتبة (مهمل)

الاختصار

#include <linux/unistd.h>
ماكرو _syscall
استدعاء النظام المطلوب

الوصف

الشيء المهم الذي يجب معرفته حول استدعاء النظام هو نموذجه الأولي. تحتاج إلى معرفة عدد الوسائط، وأنواعها، ونوع الإرجاع للدالة. هناك سبعة ماكرو تسهل عملية الاستدعاء الفعلية للنظام. وهي على الشكل التالي:


_syscallX(type,name,type1,arg1,type2,arg2,...)

حيث

X هو 0-6، وهو عدد الوسائط التي يأخذها استدعاء النظام
type هو نوع الإرجاع لاستدعاء النظام
name هو اسم استدعاء النظام
typeN هو نوع الوسيط N
argN هو اسم الوسيط N

تقوم هذه الماكرو بإنشاء دالة تسمى name بالوسائط التي تحددها. بمجرد تضمين _syscall() في ملف المصدر الخاص بك، يمكنك استدعاء استدعاء النظام بواسطة name.

الملفات

/usr/include/linux/unistd.h

المعايير

Linux.

التاريخ

بدءًا من حوالي Linux 2.6.18، تم إزالة ماكرو _syscall() من ملفات الرأس المقدمة لمساحة المستخدم. استخدم syscall(2) بدلاً من ذلك. (بعض البنى، أبرزها ia64، لم توفر ماكرو _syscall() مطلقًا؛ على تلك البنى، كان syscall(2) مطلوبًا دائمًا.)

ملاحظات

ماكرو _syscall() *لا* تنتج نموذجًا أوليًا. قد تحتاج إلى إنشاء واحد، خاصة لمستخدمي C++.

ليست جميع استدعاءات النظام مطلوبة لإرجاع رموز أخطاء موجبة أو سالبة فقط. تحتاج إلى قراءة المصدر للتأكد من كيفية إرجاعها للأخطاء. عادةً، يكون هو سالب رمز خطأ قياسي، على سبيل المثال، -EPERM. ستعيد ماكرو _syscall() النتيجة r لاستدعاء النظام عندما يكون r غير سالب، ولكنها ستعيد -1 وتعيين المتغير errno إلى -r عندما يكون r سالبًا. بالنسبة لرموز الأخطاء، انظر errno(3).

عند تعريف استدعاء نظام، يجب تمرير أنواع الوسائط بالقيمة أو بالمؤشر (للتراكيب مثل الهياكل).

أمثلة

#include <stdio.h>
#include <stdlib.h>
#include <errno.h>
#include <linux/unistd.h>       /* for _syscallX macros/related stuff */
#include <linux/kernel.h>       /* for struct sysinfo */
_syscall1(int, sysinfo, struct sysinfo *, info);
int
main(void)
{
    struct sysinfo s_info;
    int error;
    error = sysinfo(&s_info);
    printf("code error = %d\n", error);
    printf("Uptime = %lds\nLoad: 1 min %lu / 5 min %lu / 15 min %lu\n"
           "RAM: total %lu / free %lu / shared %lu\n"
           "Memory in buffers = %lu\nSwap: total %lu / free %lu\n"
           "Number of processes = %d\n",
           s_info.uptime, s_info.loads[0],
           s_info.loads[1], s_info.loads[2],
           s_info.totalram, s_info.freeram,
           s_info.sharedram, s_info.bufferram,
           s_info.totalswap, s_info.freeswap,
           s_info.procs);
    exit(EXIT_SUCCESS);
}
