__malloc_hook(3) دليل وظائف المكتبة __malloc_hook(3)

الاسم

__malloc_hook, __malloc_initialize_hook, __memalign_hook, __free_hook, __realloc_hook, __after_morecore_hook - متغيرات تصحيح malloc (مهمل)

المكتبة

مكتبة C القياسية (libc, -lc)

الخلاصة

#include <malloc.h>
typeof(void *(size_t size, const void *caller))
    *volatile  __malloc_hook;
typeof(void *(void *p, size_t size, const void *caller))
    *volatile  __realloc_hook;
typeof(void *(size_t align, size_t size, const void *caller))
    *volatile  __memalign_hook;
typeof(void *(void *p, const void *caller))
    *volatile  __free_hook;
typeof(void (void))            *__malloc_initialize_hook;
typeof(void (void)) *volatile   __after_mrecore_hook;

الوصف

تسمح لك مكتبة GNU C بتعديل سلوك malloc(3)، realloc(3)، و free(3) عن طريق تحديد دوال ربط مناسبة. يمكنك استخدام هذه الروابط لمساعدتك في تصحيح الأخطاء في البرامج التي تستخدم تخصيص الذاكرة الديناميكي، على سبيل المثال.

المتغير __malloc_initialize_hook يشير إلى دالة يتم استدعاؤها مرة واحدة عند تهيئة تطبيق malloc. هذا متغير ضعيف، لذلك يمكن تجاوزه في التطبيق بتعريف مثل التالي:


typeof(void (void))  *__malloc_initialize_hook = my_init_hook;

الآن يمكن للدالة my_init_hook() القيام بتهيئة جميع الروابط.

الدوال الأربع التي تشير إليها __malloc_hook، __realloc_hook، __memalign_hook، __free_hook لها نموذج أولي مشابه للدوال malloc(3)، realloc(3)، memalign(3)، free(3)، باستثناء أنها تحتوي على وسيط إضافي caller يمثل عنوان المتصل بـ malloc(3)، إلخ.

المتغير __after_morecore_hook يشير إلى دالة يتم استدعاؤها في كل مرة بعد طلب المزيد من الذاكرة بواسطة sbrk(2).

المعايير

GNU.

ملاحظات

استخدام دوال الربط هذه ليس آمنًا في البرامج متعددة الخيوط، وهي الآن مهملة. من glibc 2.24 فصاعدًا، تمت إزالة المتغير __malloc_initialize_hook من الواجهة البرمجية، ومن glibc 2.34 فصاعدًا، تمت إزالة جميع متغيرات الربط من الواجهة البرمجية. يجب على المبرمجين بدلًا من ذلك استباق استدعاءات الدوال ذات الصلة عن طريق تعريف وتصدير malloc()، free()، realloc()، و calloc().

أمثلة

إليك مثال قصير لكيفية استخدام هذه المتغيرات.

#include <stdio.h>
#include <malloc.h>
/* نماذج دوال الربط الخاصة بنا */
static void my_init_hook(void);
static void *my_malloc_hook(size_t, const void *);
/* متغيرات لحفظ الروابط الأصلية */
static typeof(void *(size_t, const void *))  *old_malloc_hook;
/* تجاوز دالة الربط التهيئة من مكتبة C */
typeof(void (void))  *__malloc_initialize_hook = my_init_hook;
static void
my_init_hook(void)
{
    old_malloc_hook = __malloc_hook;
    __malloc_hook = my_malloc_hook;
}
static void *
my_malloc_hook(size_t size, const void *caller)
{
    void *result;
    /* استعادة جميع الروابط القديمة */
    __malloc_hook = old_malloc_hook;
    /* استدعاء بشكل متكرر */
    result = malloc(size);
    /* حفظ الروابط الأساسية */
    old_malloc_hook = __malloc_hook;
    /* قد تستدعي printf() malloc()، لذا قم بحمايتها أيضًا */
    printf("malloc(%zu) called from %p returns %p\n",
            size, caller, result);
    /* استعادة الروابط الخاصة بنا */
    __malloc_hook = my_malloc_hook;
    return result;
}
