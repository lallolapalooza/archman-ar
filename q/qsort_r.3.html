qsort(3) دليل وظائف المكتبة qsort(3)

الاسم

qsort, qsort_r - فرز مصفوفة

المكتبة

مكتبة C القياسية (libc, -lc)

الملخص

#include <stdlib.h>
void qsort(size_t n, size_t size;
           void base[n * size], size_t n, size_t size,
           typeof(int (const void [size], const void [size]))
               *compar);
void qsort_r(size_t n, size_t size;
           void base[n * size], size_t n, size_t size,
           typeof(int (const void [size], const void [size], void *))
               *compar,
           void *arg);
متطلبات سمات اختبار الميزات لـ glibc (انظر feature_test_macros(7)):

qsort_r():

    _GNU_SOURCE

الوصف

تقوم وظيفة qsort() بفرز مصفوفة تحتوي على n عنصر بحجم size. يشير الوسيط base إلى بداية المصفوفة.

يتم فرز محتويات المصفوفة بترتيب تصاعدي وفقًا لوظيفة مقارنة يشير إليها compar، والتي يتم استدعاؤها مع وسيطين يشيران إلى الكائنات التي تتم مقارنتها.

يجب أن تُرجع وظيفة المقارنة عددًا صحيحًا أقل من أو يساوي أو أكبر من الصفر إذا كان الوسيط الأول يعتبر أقل من أو يساوي أو أكبر من الوسيط الثاني على التوالي. إذا تساوى عضوان في المقارنة، فإن ترتيبهما في المصفوفة المصنفة غير محدد.

وظيفة qsort_r() مطابقة لوظيفة qsort() باستثناء أن وظيفة المقارنة compar تأخذ وسيطًا ثالثًا. يتم تمرير مؤشر إلى وظيفة المقارنة عبر arg. بهذه الطريقة، لا تحتاج وظيفة المقارنة إلى استخدام متغيرات عامة لتمرير وسائط اعتباطية، وبالتالي فهي قابلة لإعادة الدخول وآمنة للاستخدام في الخيوط.

قيمة الإرجاع

لا تُرجع وظائف qsort() و qsort_r() أي قيمة.

ملاحظات

لمقارنة سلاسل C، يمكن لوظيفة المقارنة استدعاء strcmp(3)، كما هو موضح في المثال أدناه.

أمثلة

للحصول على مثال واحد للاستخدام، انظر المثال تحت bsearch(3).

مثال آخر هو البرنامج التالي، الذي يقوم بفرز السلاسل المقدمة في وسائط سطر الأوامر الخاصة به:

#include <stdio.h>
#include <stdlib.h>
#include <string.h>
static int
cmpstringp(const void *p1, const void *p2)
{
    /* الوسائط الفعلية لهذه الوظيفة هي "مؤشرات إلى مؤشرات إلى char"،
       ولكن وسائط strcmp(3) هي "مؤشرات إلى char"، ومن هنا جاء تحويل النوع التالي وإلغاء المرجع. */
    return strcmp(*(const char **) p1, *(const char **) p2);
}
int
main(int argc, char *argv[])
{
    if (argc < 2) {
        fprintf(stderr, "Usage: %s <string>...\n", argv[0]);
        exit(EXIT_FAILURE);
    }
    qsort(&argv[1], argc - 1, sizeof(char *), cmpstringp);
    for (size_t j = 1; j < argc; j++)
        puts(argv[j]);
    exit(EXIT_SUCCESS);
}
