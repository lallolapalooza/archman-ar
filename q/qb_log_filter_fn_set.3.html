qb_log_filter_fn_set(3)

الحزمة: libqb

التراخيص: LGPL-2.1-or-later

QB_LOG_FILTER_FN_SET(3) دليل مبرمج libqb QB_LOG_FILTER_FN_SET(3)

الاسم

qb_log_filter_fn_set - بدلاً من استخدام وظائف qb_log_filter_ctl() يمكنك تطبيق المرشحات يدوياً عن طريق تحديد دالة استدعاء وتعيين حقل الأهداف باستخدام qb_bit_set() و qb_bit_clear() كما هو موضح أدناه:

الملخص

#include <qb/qblog.h>

int32_t qb_log_filter_fn_set(
    qb_log_filter_fn  fn
);

الوصف

static void
m_filter(struct qb_log_callsite *cs)
{
        if ((cs->priority >= LOG_ALERT &&
             cs->priority <= LOG_DEBUG) &&
             strcmp(cs->filename, "my_c_file.c") == 0) {
                qb_bit_set(cs->targets, QB_LOG_SYSLOG);
        } else {
                qb_bit_clear(cs->targets, QB_LOG_SYSLOG);
        }
}
