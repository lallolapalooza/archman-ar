qblog.h(3)

الحزمة: libqb

التراخيص: LGPL-2.1-or-later

QBLOG.H(3) دليل مبرمج libqb QBLOG.H(3)

الاسم

qblog.h - يوفر واجهة برمجة التطبيقات للتسجيل أربعة أجزاء رئيسية (الأساسيات، التصفية، الخيوط & الصندوق الأسود).

الملخص

#include <qb/qblog.h>

الوصف

الفكرة وراء نظام التسجيل هذا ليست أن تكون وصفيًا ولكن لتوفير مجموعة من الأدوات لمساعدة المطور على تحقيق ما يريد بسرعة وسهولة.

واجهة برمجة تطبيقات التسجيل الأساسية.

استدعِ qb_log() لإنشاء رسالة سجل. ثم لكتابة الرسالة في مكان ذي معنى، استدعِ qb_log_ctl() لتكوين الأهداف.

أبسط استخدام ممكن:

main() {
     qb_log_init("simple-log", LOG_DAEMON, LOG_INFO);
        // ...
     qb_log(LOG_WARNING, "watch out");
        // ...
     qb_log_fini();
}

تكوين أهداف السجل.

يمكن أن يكون هدف السجل هو syslog، stderr، الصندوق الأسود، stdout، أو ملف نصي. افتراضيًا، يتم تمكين syslog فقط. في حين أن هذا شائع للخوادم، فإنه نادرًا ما يكون مناسبًا للبرامج العادية، والتي يجب تعطيله عند استخدام أهداف أخرى (انظر أدناه):

qb_log_ctl(B_LOG_SYSLOG, QB_LOG_CONF_ENABLED, QB_FALSE);

لتمكين هدف، قم بما يلي:

qb_log_ctl(QB_LOG_BLACKBOX, QB_LOG_CONF_ENABLED, QB_TRUE);

syslog، stderr، الصندوق الأسود، و stdout ثابتة (لا تحتاج إلى إنشائها، فقط تمكينها أو تعطيلها). ومع ذلك، يمكنك فتح ملفات سجلات متعددة (تقع ضمن النطاق الشامل QB_LOG_TARGET_DYNAMIC_START حتى QB_LOG_TARGET_DYNAMIC_END). للقيام بذلك، استخدم الكود التالي:

mytarget = qb_log_file_open("/var/log/mylogfile");
qb_log_ctl(mytarget, QB_LOG_CONF_ENABLED, QB_TRUE);

بمجرد تمكين / فتح أهدافك، يمكنك تكوينها على النحو التالي: تكوين حجم الصندوق الأسود:

qb_log_ctl(QB_LOG_BLACKBOX, QB_LOG_CONF_SIZE, 1024*10);

جعل التسجيل في الملف خيطيًا:

qb_log_ctl(mytarget, QB_LOG_CONF_THREADED, QB_TRUE);

في بعض الأحيان، يتم تكوين خوادم syslog (مسبقًا) لتصفية الرسائل التي لا تتجاوز أولوية معينة. عندما يحدث هذا ليكون هدف التسجيل، يتم تمرير الأولوية المحددة للرسالة دون تغيير، مما قد يؤدي إلى فقدان الرسالة. بالنسبة للرسائل التي تصل إلى أهمية LOG_DEBUG، يمكن حل ذلك عن طريق زيادة أولويات رسائل syslog بشكل متناسب (هنا، الخطوة بحيث يتم ترقية LOG_DEBUG إلى LOG_INFO):

qb_log_ctl(QB_LOG_SYSLOG, QB_LOG_CONF_PRIORITY_BUMP,
           LOG_INFO - LOG_DEBUG);

للتأكد من أن جميع السجلات إلى أهداف الملفات يتم مزامنتها مع نظام الملفات (يتم نقل الرسائل الجديدة بشكل صريح إلى جهاز التخزين أثناء وصولها، وإلا فإن القيمة الافتراضية هي QB_FALSE):

qb_log_ctl(mytarget, QB_LOG_CONF_FILE_SYNC, QB_TRUE);

تصفية الرسائل.

للحصول على المزيد من القوة فوق رسائل السجل التي تذهب إلى أي هدف، يمكنك تطبيق عوامل تصفية على الأهداف. ما يحدث هو أن مواقع الاستدعاء المطلوبة لديها البت الصحيح معين. ثم عندما يتم إنشاء رسالة السجل، يتم إرسالها إلى الأهداف بناءً على البت المعين في "هدف" مجموعة البتات لموقع الاستدعاء. يمكن تصفية الرسائل بناءً على:

لذلك، لجعل جميع السجلات من evil_function() تذهب إلى stderr، قم بما يلي:

qb_log_filter_ctl(QB_LOG_STDERR, QB_LOG_FILTER_ADD,
             QB_LOG_FILTER_FUNCTION, "evil_function", LOG_TRACE);

لذلك، لجعل جميع السجلات من totem* (بأولوية <= LOG_INFO) تذهب إلى stderr، قم بما يلي:

qb_log_filter_ctl(QB_LOG_STDERR, QB_LOG_FILTER_ADD,
             QB_LOG_FILTER_FILE, "totem", LOG_INFO);

لذلك، لجعل جميع السجلات التي تحتوي على السلسلة الفرعية "ringbuffer" تذهب إلى stderr، قم بما يلي:

qb_log_filter_ctl(QB_LOG_STDERR, QB_LOG_FILTER_ADD,
             QB_LOG_FILTER_FORMAT, "ringbuffer", LOG_TRACE);

التسجيل غير حاجب وخيطي آمن.

يكون التسجيل خيطيًا آمنًا فقط عند استخدام التسجيل الخيطي. إذا كنت تخطط للتسجيل من خيوط متعددة، يجب عليك تهيئة خيط مسجل libqb واستخدام qb_log_filter_ctl لتعيين علامة QB_LOG_CONF_THREADED على جميع أهداف التسجيل قيد الاستخدام.

لتحقيق التسجيل غير حاجب، بحيث لا تعيق أي استدعاءات لـ write() أو syslog() برنامجك، يمكنك استخدام التسجيل الخيطي أيضًا.

استخدام التسجيل الخيطي:

main() {
     qb_log_init("simple-log", LOG_DAEMON, LOG_INFO);
     qb_log_ctl(QB_LOG_SYSLOG, QB_LOG_CONF_THREADED, QB_TRUE);
        // ...
        daemonize();
        // اتصل بهذا بعد fork()
        qb_log_thread_start();
        // ...
     qb_log(LOG_WARNING, "watch out");
        // ...
     qb_log_fini();
}

صندوق أسود للتشخيص في الميدان.

يخزن هذا رسائل السجل في مخزن حلقي بحيث يمكن كتابتها إلى ملف إذا تعطل البرنامج (ستحتاج إلى التقاط SIGSEGV). يمكن بعد ذلك طباعتها بسهولة لاحقًا.

استخدام الصندوق الأسود:

static void sigsegv_handler(int sig)
{
        (void)signal (SIGSEGV, SIG_DFL);
        qb_log_blackbox_write_to_file("simple-log.fdata");
        qb_log_fini();
        raise(SIGSEGV);
}
main() {
     signal(SIGSEGV, sigsegv_handler);
     qb_log_init("simple-log", LOG_DAEMON, LOG_INFO);
     qb_log_filter_ctl(QB_LOG_BLACKBOX, QB_LOG_FILTER_ADD,
                  QB_LOG_FILTER_FILE, "*", LOG_DEBUG);
     qb_log_ctl(QB_LOG_BLACKBOX, QB_LOG_CONF_SIZE, 1024*10);
     qb_log_ctl(QB_LOG_BLACKBOX, QB_LOG_CONF_ENABLED, QB_TRUE);
        // ...
     qb_log(LOG_WARNING, "watch out");
        // ...
     qb_log_fini();
}

تمييز الرسائل.

يمكنك تمييز الرسائل باستخدام الوسيط الثاني لـ qb_logt() أو باستخدام qb_log_filter_ctl(). يمكن استخدام هذا لإضافة معلومات الميزات أو الأنظمة الفرعية إلى السجلات.

const char* my_tags_stringify(uint32_t tags) {
        if (qb_bit_is_set(tags, QB_LOG_TAG_LIBQB_MSG_BIT) {
                return "libqb";
        } else if (tags == 3) {
                return "three";
        } else {
                return "MAIN";
        }
}
main() {
        // ...
        qb_log_tags_stringify_fn_set(my_tags_stringify);
        qb_log_format_set(QB_LOG_STDERR, "[%5g] %p %b");
        // ...
        qb_logt(LOG_INFO, 3, "hello");
        qb_logt(LOG_INFO, 0, "hello");
}


الكود أعلاه سينتج:

[libqb] some message
[three] info hello
[MAIN ] info hello
