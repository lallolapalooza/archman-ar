qb_loop_timer_is_running(3)

الحزمة: libqb

التراخيص: LGPL-2.1-or-later

QB_LOOP_TIMER_IS_RUNNING(3) دليل مبرمج libqb QB_LOOP_TIMER_IS_RUNNING(3)

الاسم

qb_loop_timer_is_running - التحقق مما إذا كان المؤقت لا يزال معلقًا.

الملخص

#include <qb/qbloop.h>

int32_t qb_loop_timer_is_running(
    qb_loop_t            *l,   /* مؤشر إلى مثيل الحلقة */
    qb_loop_timer_handle  th   /* معرّف لحذف المؤقت إذا لزم الأمر. */
);

المعلمات

l مؤشر إلى مثيل الحلقة

th معرّف لحذف المؤقت إذا لزم الأمر.

الوصف

قيمة الإرجاع

QB_TRUE نعم، هذا المؤقت معلق

QB_FALSE هذا المؤقت غير موجود أو انتهت صلاحيته
