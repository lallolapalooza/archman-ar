qb_map_notify_del(3)

الحزمة: libqb

التراخيص: LGPL-2.1-or-later

QB_MAP_NOTIFY_DEL(3) دليل مبرمج libqb QB_MAP_NOTIFY_DEL(3)

الاسم

qb_map_notify_del - حذف إشعار من الخريطة.

ملخص

#include <qb/qbmap.h>

int32_t qb_map_notify_del(
    qb_map_t         *m,       /* نسخة الخريطة */
    /*
     * المفتاح (أو البادئة) لربط الإشعار به.
     */
    const char       *key,
    qb_map_notify_fn  fn,      /* الدالة الاستدعائية */
    int32_t           events   /* نوع الأحداث للتسجيل لها.*/
);

المعاملات

m نسخة الخريطة

key المفتاح (أو البادئة) لربط الإشعار به.

fn الدالة الاستدعائية

events نوع الأحداث للتسجيل لها.

الوصف

القيمة المرجعة

0 نجاح

-errno فشل

ملاحظة

يجب أن يتطابق المفتاح، الدالة الاستدعائية، والأحداث مع تلك التي أضفتها.
