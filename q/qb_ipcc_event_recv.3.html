qb_ipcc_event_recv(3)

الحزمة: libqb

التراخيص: LGPL-2.1-or-later

QB_IPCC_EVENT_RECV(3) دليل مبرمج libqb QB_IPCC_EVENT_RECV(3)

الاسم

qb_ipcc_event_recv - استقبال حدث.

المُلخص

#include <qb/qbipcc.h>

ssize_t qb_ipcc_event_recv(
    qb_ipcc_connection_t   *c,           /* مثيل الاتصال */
    void                   *msg_ptr,     /* مؤشر إلى مخزن مؤقت للرسالة لاستقبالها فيه */
    size_t                  msg_len,     /* حجم المخزن المؤقت */
    /*
     * الوقت بالمللي ثانية للانتظار لاستلام رسالة 0 == عدم الانتظار، سالب ==
     * منع، موجب == انتظر X مللي ثانية.
     */
    int32_t                 ms_timeout
);

المعاملات

c مثيل الاتصال

msg_ptr مؤشر إلى مخزن مؤقت للرسالة لاستقبالها فيه

msg_len حجم المخزن المؤقت

ms_timeout الوقت بالمللي ثانية للانتظار لاستلام رسالة 0 == عدم الانتظار، سالب == منع، موجب == انتظر X مللي ثانية.

الوصف

قيمة الإرجاع

حجم الرسالة أو الخطأ (-errno)

ملاحظة

أن msg_ptr سيتضمن qb_ipc_response_header في بداية الرسالة.
