qb_loop_job_add(3)

الحزمة: libqb

التراخيص: LGPL-2.1-or-later

QB_LOOP_JOB_ADD(3) دليل مبرمج libqb QB_LOOP_JOB_ADD(3)

الاسم

qb_loop_job_add - إضافة مهمة إلى الحلقة الرئيسية.

الملخص

#include <qb/qbloop.h>

int32_t qb_loop_job_add(
    qb_loop_t               *l,            /* مؤشر إلى مثيل الحلقة */
    enum qb_loop_priority    p,            /* الأولوية */
    void                    *data,         /* بيانات المستخدم المقدمة إلى دالة الإرسال */
    qb_loop_job_dispatch_fn  dispatch_fn   /* دالة الاستدعاء */
);

المعاملات

l مؤشر إلى مثيل الحلقة

p الأولوية

data بيانات المستخدم المقدمة إلى دالة الإرسال

dispatch_fn دالة الاستدعاء

الوصف

يتم تشغيل هذا في الدورة التالية للحلقة.

الهياكل


enum qb_loop_priority {


      QB_LOOP_LOW;

      QB_LOOP_MED;

      QB_LOOP_HIGH;
};

قيمة الإرجاع

الحالة (0 == جيد، -errno == فشل)

ملاحظة

إنها مهمة لمرة واحدة.
