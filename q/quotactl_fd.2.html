quotactl(2) دليل استدعاءات النظام quotactl(2)

الاسم

quotactl, quotactl_fd — معالجة حصص الأقراص

المكتبة

مكتبة C القياسية (libc, -lc)

الملخص

#include <sys/quota.h>
#include <xfs/xqm.h> /* تعريف ثوابت Q_X* و XFS_QUOTA_*
                        (أو <linux/dqblk_xfs.h>؛ انظر ملاحظات) */
int quotactl(int op, const char *_Nullable special, int id,
             caddr_t addr);
#include <sys/syscall.h>    /* تعريف ثوابت SYS_* */
#include <unistd.h>
int syscall(SYS_quotactl_fd, int fd, int op, int id, caddr_t addr);

ملاحظة: لا توفر glibc واجهة لـ quotactl_fd()، مما يستلزم استخدام syscall(2).

الوصف

يمكن استخدام نظام الحصص لتعيين حدود يومية، أسبوعية، أو حسب المشروع لكل مستخدم أو مجموعة على مساحة القرص المستخدمة في نظام الملفات. لكل مستخدم و/أو مجموعة، يمكن تعيين حد ناعم وحد صارم لكل نظام ملفات. الحد الصارم لا يمكن تجاوزه. يمكن تجاوز الحد الناعم، ولكن ستصدر تحذيرات. علاوة على ذلك، لا يمكن للمستخدم تجاوز الحد الناعم لأكثر من فترة سماح (أسبوع واحد افتراضيًا) في المرة الواحدة؛ بعد ذلك، يعتبر الحد الناعم حدًا صارمًا.

تقوم استدعاءات quotactl() و quotactl_fd() بمعالجة حصص الأقراص. يكمن الاختلاف بين هاتين الوظيفتين في الطريقة التي يتم بها تحديد نظام الملفات الذي تتم معالجته، انظر وصف الوسائط أدناه. انظر ملاحظات لمعرفة سبب تفضيل أحد المتغيرين على الآخر.

يشير الوسيط op إلى عملية سيتم تطبيقها على معرف المستخدم أو المجموعة المحدد في id. لتهيئة الوسيط op، استخدم الماكرو QCMD(subop, type). القيمة type هي إما USRQUOTA، لحصص المستخدمين، GRPQUOTA، لحصص المجموعات، أو (منذ Linux 4.1) PRJQUOTA، لحصص المشاريع. يتم وصف قيمة subop أدناه.

بالنسبة لـ quotactl()، الوسيط special هو مؤشر إلى سلسلة تنتهي بـ NULL تحتوي على مسار الجهاز الخاص بالكتلة (المركب) لنظام الملفات الذي تتم معالجته.

بالنسبة لـ quotactl_fd()، الوسيط fd هو واصف ملف (قد يكون مفتوحًا بعلامة O_PATH) يشير إلى ملف أو دليل في نظام الملفات الذي تتم معالجته.

الوسيط addr هو عنوان بنية بيانات اختيارية خاصة بالعملية يتم نسخها للداخل أو للخارج من النظام. يتم تفسير addr مع كل عملية أدناه.

القيمة subop هي إحدى العمليات التالية:

Q_QUOTAON
تشغيل الحصص لنظام ملفات. الوسيط id هو معرف تنسيق الحصة الذي سيتم استخدامه. حاليًا، هناك ثلاثة تنسيقات حصص مدعومة:
QFMT_VFS_OLD
تنسيق الحصة الأصلي.
QFMT_VFS_V0
تنسيق الحصة القياسي VFS v0، والذي يمكنه التعامل مع معرفات المستخدمين (UIDs) والمجموعات (GIDs) 32 بت وحدود حصص تصل إلى 2^42 بايت و 2^32 مجلد.
QFMT_VFS_V1
تنسيق حصة يمكنه التعامل مع معرفات المستخدمين (UIDs) والمجموعات (GIDs) 32 بت وحدود حصص تبلغ 2^63 - 1 بايت و 2^63 - 1 مجلد.
الوسيط addr يشير إلى مسار ملف يحتوي على حصص نظام الملفات. يجب أن يكون ملف الحصة موجودًا؛ يتم إنشاؤه عادةً باستخدام برنامج quotacheck(8)
يمكن أيضًا تخزين معلومات الحصص في inodes النظام المخفية لنظامي ext4 و XFS وأنظمة الملفات الأخرى إذا تم تكوين نظام الملفات وفقًا لذلك. في هذه الحالة، لا توجد ملفات حصص مرئية ولا حاجة لاستخدام quotacheck(8). يتم الحفاظ على معلومات الحصص دائمًا متسقة بواسطة نظام الملفات وتعمل عملية Q_QUOTAON فقط لتمكين تطبيق حدود الحصص. يتم الإشارة إلى وجود inodes نظام مخفية مع معلومات الحصص بواسطة العلم DQF_SYS_FILE في الحقل dqi_flags الذي يتم إرجاعه بواسطة عملية Q_GETINFO.
يتجاهل المتغير quotactl_fd() من استدعاء النظام هذا الوسائط addr و id، لذا فإن عملية Q_QUOTAON لـ quotactl_fd() مناسبة فقط للعمل مع inodes النظام المخفية.
تتطلب هذه العملية صلاحيات (CAP_SYS_ADMIN).
Q_QUOTAOFF
إيقاف الحصص لنظام ملفات. يتم تجاهل الوسائط addr و id. تتطلب هذه العملية صلاحيات (CAP_SYS_ADMIN).
Q_GETQUOTA
الحصول على حدود حصص الأقراص والاستخدام الحالي للمستخدم أو المجموعة id. الوسيط addr هو مؤشر إلى بنية dqblk معرفة في <sys/quota.h> على النحو التالي:

/* uint64_t هو عدد صحيح غير موقع 64 بت؛
   uint32_t هو عدد صحيح غير موقع 32 بت */
struct dqblk {      /* التعريف منذ Linux 2.4.22 */
    uint64_t dqb_bhardlimit;  /* الحد المطلق على كتل
                                 الحصة المخصصة للقرص */
    uint64_t dqb_bsoftlimit;  /* الحد المفضل على
                                 كتل حصة القرص */
    uint64_t dqb_curspace;    /* المساحة الحالية المشغولة
                                 (بالبايت) */
    uint64_t dqb_ihardlimit;  /* الحد الأقصى لعدد
                                 المجلدات المخصصة */
    uint64_t dqb_isoftlimit;  /* الحد المفضل للمجلدات */
    uint64_t dqb_curinodes;   /* العدد الحالي للمجلدات
                                 المخصصة */
    uint64_t dqb_btime;       /* الحد الزمني للاستخدام المفرط
                                 للديسك */
    uint64_t dqb_itime;       /* الحد الزمني للملفات المفرطة */
    uint32_t dqb_valid;       /* قناع بت لثوابت QIF_* */
};
/* الأعلام في dqb_valid التي تشير إلى الحقول في
   بنية dqblk الصالحة.  */
#define QIF_BLIMITS   1
#define QIF_SPACE     2
#define QIF_ILIMITS   4
#define QIF_INODES    8
#define QIF_BTIME     16
#define QIF_ITIME     32
#define QIF_LIMITS    (QIF_BLIMITS | QIF_ILIMITS)
#define QIF_USAGE     (QIF_SPACE | QIF_INODES)
#define QIF_TIMES     (QIF_BTIME | QIF_ITIME)
#define QIF_ALL       (QIF_LIMITS | QIF_USAGE | QIF_TIMES)

الحقل dqb_valid هو قناع بت يتم تعيينه للإشارة إلى الإدخالات الموجودة في بنية dqblk الصالحة. حاليًا، يقوم النواة بملء جميع الإدخالات في بنية dqblk ويضع علامة عليها كصالحة في الحقل dqb_valid. يمكن للمستخدمين غير المميزين استرداد حصصهم فقط؛ يمكن للمستخدم المميز (CAP_SYS_ADMIN) استرداد حصص أي مستخدم.
Q_GETNEXTQUOTA (منذ Linux 4.6)
هذه العملية هي نفسها Q_GETQUOTA، ولكنها ترجع معلومات الحصة لـ ID التالي الأكبر من أو يساوي id والذي لديه حصة مضبوطة.
الوسيط addr هو مؤشر إلى بنية nextdqblk التي تكون حقولها كما هي لبنية dqblk، باستثناء إضافة حقل dqb_id الذي يستخدم لإرجاع المعرف الذي يتم إرجاع معلومات الحصة له:

struct nextdqblk {
    uint64_t dqb_bhardlimit;
    uint64_t dqb_bsoftlimit;
    uint64_t dqb_curspace;
    uint64_t dqb_ihardlimit;
    uint64_t dqb_isoftlimit;
    uint64_t dqb_curinodes;
    uint64_t dqb_btime;
    uint64_t dqb_itime;
    uint32_t dqb_valid;
    uint32_t dqb_id;
};

Q_SETQUOTA
تعيين معلومات الحصة للمستخدم أو المجموعة id، باستخدام المعلومات المقدمة في بنية dqblk التي يشير إليها addr. يشير الحقل dqb_valid لبنية dqblk إلى الإدخالات التي قام المتصل بتعيينها في البنية. تحل هذه العملية محل العمليات Q_SETQLIM و Q_SETUSE في واجهات الحصص السابقة. تتطلب هذه العملية صلاحيات (CAP_SYS_ADMIN).
Q_GETINFO (منذ Linux 2.4.22)
الحصول على معلومات (ل

هذا الموقع غير مرتبط بـ Arch Linux ولا تابع له بأي شكل من الأشكال. المحتوى هو ترجمة فقط لصفحات دليل Arch المنشورة على man.archlinux.org.

a; /* معلومات حصص التخزين */ struct fs_qfilestatv qs_pquota; /* معلومات حصص تخزين المشاريع */ int32_t qs_btimelimit; /* حد مؤقت الكتل */ int32_t qs_itimelimit; /* حد مؤقت inodes */ int32_t qs_rtbtimelimit; /* حد مؤقت كتل RT */ uint16_t qs_bwarnlimit; /* حد عدد التحذيرات */ uint16_t qs_iwarnlimit; /* حد عدد التحذيرات */ uint64_t qs_pad2[8]; /* للاستخدام المستقبلي */ };
يجب ملء حقل qs_version في البنية بإصدار البنية الذي يدعمه المتلقي (حتى الآن، يتم دعم FS_QSTAT_VERSION1 فقط). سيقوم النواة بملء البنية وفقًا للإصدار المقدم. يتم تجاهل الوسيط id.
Q_XQUOTARM (معيب حتى Linux 3.16)
تحرير المساحة على القرص التي تشغلها حصص التخزين. يجب أن يكون الوسيط addr مؤشرًا إلى قيمة unsigned int تحتوي على علامات (نفس علامات حقل d_flags في بنية fs_disk_quota) التي تحدد أنواع حصص التخزين التي يجب إزالتها. (لاحظ أنه يتم تجاهل نوع حصة التخزين الممرر في الوسيط op، ولكنه يجب أن يظل صالحًا لتجاوز فحوصات معالج استدعاءات النظام الأولية quotactl.)
يجب أن تكون حصص التخزين قد تم إيقافها بالفعل. يتم تجاهل الوسيط id.
Q_XQUOTASYNC (منذ Linux 2.6.15؛ لا يفعل شيئًا منذ Linux 3.4)
كانت هذه العملية مكافئة XFS لـ Q_SYNC، ولكنها لا تفعل شيئًا منذ Linux 3.4، حيث أن sync(1) يكتب معلومات حصص التخزين إلى القرص الآن (بالإضافة إلى بيانات التعريف الأخرى لنظام الملفات التي يكتبها). يتم تجاهل الوسائط special و id و addr.

قيمة الإرجاع

عند النجاح، تعيد quotactl() القيمة 0؛ عند حدوث خطأ، تعيد -1، ويتم تعيين errno للإشارة إلى الخطأ.

أخطاء

EACCES
op هو Q_QUOTAON، وملف حصة التخزين المشار إليه بواسطة addr موجود، ولكنه ليس ملفًا عاديًا أو ليس موجودًا على نظام الملفات المشار إليه بواسطة special.
EBUSY
op هو Q_QUOTAON، ولكن تم إجراء Q_QUOTAON آخر بالفعل.
EFAULT
addr أو special غير صالح.
EINVAL
op أو type غير صالح.
EINVAL
op هو Q_QUOTAON، ولكن ملف حصة التخزين المحدد تالف.
EINVAL (منذ Linux 5.5)
op هو Q_XQUOTARM، ولكن addr لا يشير إلى أنواع حصص تخزين صالحة.
ENOENT
الملف المحدد بواسطة special أو addr غير موجود.
ENOSYS
لم يتم تجميع النواة مع خيار CONFIG_QUOTA.
ENOTBLK
special ليس جهاز كتلة.
EPERM
كان المتصل يفتقر إلى الامتياز المطلوب (CAP_SYS_ADMIN) للعملية المحددة.
ERANGE
op هو Q_SETQUOTA، ولكن الحدود المحددة خارج النطاق المسموح به من قبل تنسيق حصص التخزين.
ESRCH
لم يتم العثور على حصة تخزين للرقم المحدد. لم يتم تشغيل حصص التخزين لنظام الملفات هذا.
ESRCH
op هو Q_QUOTAON، ولكن تنسيق حصة التخزين المحدد لم يتم العثور عليه.
ESRCH
op هو Q_GETNEXTQUOTA أو Q_XGETNEXTQUOTA، ولكن لا يوجد معرف أكبر من أو يساوي id لديه حصة تخزين نشطة.

ملاحظات

رأس XFS بديل

بدلاً من <xfs/xqm.h> يمكن استخدام <linux/dqblk_xfs.h>، مع الأخذ في الاعتبار وجود عدة اختلافات في التسمية:

•
يتم تعريف علامات تمكين حصص التخزين (من تنسيق XFS_QUOTA_[UGP]DQ_{ACCT,ENFD}) بدون بادئة "X"، باسم FS_QUOTA_[UGP]DQ_{ACCT,ENFD}.
•
ينطبق الشيء نفسه على علامات نوع حصة التخزين XFS_{USER,GROUP,PROJ}_QUOTA، والتي يتم تعريفها باسم FS_{USER,GROUP,PROJ}_QUOTA.
•
يحدد ملف الرأس dqblk_xfs.h ثوابته الخاصة XQM_USRQUOTA و XQM_GRPQUOTA و XQM_PRJQUOTA لأنواع حصص التخزين المتاحة، ولكن قيمها هي نفسها كما هو الحال بالنسبة للثوابت بدون البادئة XQM_.

quotactl() مقابل quotactl_fd()

يتطلب المتغير الأصلي quotactl() من استدعاء النظام هذا تحديد جهاز الكتلة الذي يحتوي على نظام الملفات المراد تشغيله. هذا يجعله غير قابل للاستخدام في الحالات التي لا يحتوي فيها نظام الملفات على جهاز كتلة داعم (مثل tmpfs). حتى عندما يكون جهاز الكتلة موجودًا، قد يكون من الصعب تحديده (يتطلب فحص /proc/self/mounts وحتى بعض التحليل الخاص بنظام الملفات في حالة، على سبيل المثال، bcachefs). تعمل quotactl_fd() بدلاً من ذلك على نقطة التحميل، مما يتجنب هذا القيد ويسهل استخدامه (نظرًا لأن نظام الملفات المراد معالجته يتم تحديده عادةً بواسطة نقطة تحميله على أي حال).
