qb_hdb_handle_destroy(3)

الحزمة: libqb

التراخيص: LGPL-2.1-or-later

QB_HDB_HANDLE_DESTROY(3) دليل مبرمج libqb QB_HDB_HANDLE_DESTROY(3)

الاسم

qb_hdb_handle_destroy - طلب تدمير الكائن.

الملخص

#include <qb/qbhdb.h>

int32_t qb_hdb_handle_destroy(
    struct qb_hdb   *hdb,        /* مثيل قاعدة البيانات */
    qb_handle_t      handle_in   /* المقبض */
);

المعلمات

hdb مثيل قاعدة البيانات

handle_in المقبض

الوصف

عندما يكون عداد المراجع 0، سيتم تدميره.

الهياكل


struct qb_hdb {


    uint32_t    handle_count;

    qb_array_t  handles;

    uint32_t    iterator;

    void      (*destructor)(void *);

    uint32_t    first_run;
};

قيمة الإرجاع

(0 == نجاح، -errno فشل)
