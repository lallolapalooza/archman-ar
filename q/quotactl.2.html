quotactl(2) دليل استدعاءات النظام quotactl(2)

الاسم

quotactl, quotactl_fd - معالجة حصص القرص

المكتبة

مكتبة C القياسية (libc، -lc)

الملخص

#include <sys/quota.h>
#include <xfs/xqm.h> /* تعريف الثوابت Q_X* و XFS_QUOTA_*
                        (أو <linux/dqblk_xfs.h>؛ انظر ملاحظات) */
int quotactl(int op, const char *_Nullable special, int id,
             caddr_t addr);
#include <sys/syscall.h>    /* تعريف الثوابت SYS_* */
#include <unistd.h>
int syscall(SYS_quotactl_fd, int fd, int op, int id, caddr_t addr);

ملاحظة: لا توفر glibc واجهة لـ quotactl_fd()، مما يستلزم استخدام syscall(2).

الوصف

يمكن استخدام نظام الحصص لتعيين حدود لكل مستخدم، ولكل مجموعة، ولكل مشروع على مقدار مساحة القرص المستخدمة في نظام ملفات. لكل مستخدم و/أو مجموعة، يمكن تعيين حد ناعم وحد صارم لكل نظام ملفات. لا يمكن تجاوز الحد الصارم. يمكن تجاوز الحد الناعم، ولكن ستحدث تحذيرات. علاوة على ذلك، لا يمكن للمستخدم تجاوز الحد الناعم لأكثر من مدة فترة سماح (أسبوع واحد افتراضيًا) في المرة الواحدة؛ بعد ذلك، يعتبر الحد الناعم حدًا صارمًا.

تستدعي quotactl() و quotactl_fd() معالجة حصص القرص. الفرق بين هاتين الوظيفتين هو الطريقة التي يتم بها تحديد نظام الملفات الذي تتم معالجته، انظر وصف الوسائط أدناه. انظر ملاحظات لماذا قد يُفضل أحد المتغيرات على الآخر.

يحدد الوسيط op عملية سيتم تطبيقها على معرف المستخدم أو المجموعة المحدد في id. لتهيئة الوسيط op، استخدم ماكرو QCMD(subop, type). قيمة type هي إما USRQUOTA، لحصص المستخدم، GRPQUOTA، لحصص المجموعة، أو (منذ Linux 4.1) PRJQUOTA، لحصص المشروع. يتم وصف قيمة subop أدناه.

بالنسبة لـ quotactl()، فإن الوسيط special هو مؤشر إلى سلسلة نصية تنتهي بصفر تحتوي على مسار الجهاز الخاص بالكتلة (المُثبت) لنظام الملفات الذي تتم معالجته.

بالنسبة لـ quotactl_fd()، فإن الوسيط fd هو واصف ملف (يمكن فتحه بالعلامة O_PATH) يشير إلى ملف أو دليل على نظام الملفات الذي تتم معالجته.

الوسيط addr هو عنوان بنية بيانات اختيارية خاصة بالعملية يتم نسخها إلى النظام أو منه. يتم إعطاء تفسير addr مع كل عملية أدناه.

قيمة subop هي إحدى العمليات التالية:

Q_QUOTAON
تشغيل الحصص لنظام ملفات. الوسيط id هو رقم تعريف تنسيق الحصة المستخدم. حاليًا، هناك ثلاثة تنسيقات حصص مدعومة:
QFMT_VFS_OLD
تنسيق الحصة الأصلي.
QFMT_VFS_V0
تنسيق الحصة القياسي VFS v0، والذي يمكنه التعامل مع معرفات المستخدمين والمجموعات 32 بت وحدود الحصص تصل إلى 2^42 بايت و 2^32 أدوات (inodes).
QFMT_VFS_V1
تنسيق حصة يمكنه التعامل مع معرفات المستخدمين والمجموعات 32 بت وحدود الحصص 2^63 - 1 بايت و 2^63 - 1 أدوات.
الوسيط addr يشير إلى مسار ملف يحتوي على حصص نظام الملفات. يجب أن يوجد ملف الحصة؛ عادة ما يتم إنشاؤه باستخدام برنامج quotacheck(8)
يمكن أيضًا تخزين معلومات الحصص في أدوات النظام المخفية لـ ext4 و XFS وأنظمة الملفات الأخرى إذا تم تكوين نظام الملفات بذلك. في هذه الحالة، لا توجد ملفات حصص مرئية ولا حاجة لاستخدام quotacheck(8). يتم الاحتفاظ بمعلومات الحصص دائمًا متسقة بواسطة نظام الملفات وعملية Q_QUOTAON تعمل فقط لتمكين فرض حدود الحصص. يشير وجود أدوات النظام المخفية بمعلومات حصص إلى العلامة DQF_SYS_FILE في حقل dqi_flags الذي تم إرجاعه بواسطة عملية Q_GETINFO.
يتجاهل متغير quotactl_fd() من استدعاء النظام هذا الوسائط addr و id، لذا فإن عملية Q_QUOTAON لـ quotactl_fd() مناسبة فقط للعمل مع أدوات النظام المخفية.
تتطلب هذه العملية امتيازات (CAP_SYS_ADMIN).
Q_QUOTAOFF
إيقاف الحصص لنظام ملفات. يتم تجاهل الوسائط addr و id. تتطلب هذه العملية امتيازات (CAP_SYS_ADMIN).
Q_GETQUOTA
الحصول على حدود حصص القرص والاستخدام الحالي لمعرف المستخدم أو المجموعة id. الوسيط addr هو مؤشر إلى بنية dqblk معرفة في <sys/quota.h> على النحو التالي:

/* uint64_t هو عدد صحيح 64 بت غير موقع؛
   uint32_t هو عدد صحيح 32 بت غير موقع */
struct dqblk {      /* التعريف منذ Linux 2.4.22 */
    uint64_t dqb_bhardlimit;  /* الحد المطلق على كتل
                                 حصص القرص المخصصة */
    uint64_t dqb_bsoftlimit;  /* الحد المفضل على كتل
                                 حصص القرص */
    uint64_t dqb_curspace;    /* المساحة الحالية المشغولة
                                 (بالبايت) */
    uint64_t dqb_ihardlimit;  /* الحد الأقصى لعدد الأدوات
                                 المخصصة */
    uint64_t dqb_isoftlimit;  /* الحد المفضل للأدوات */
    uint64_t dqb_curinodes;   /* العدد الحالي للأدوات
                                 المخصصة */
    uint64_t dqb_btime;       /* الحد الزمني للاستخدام المفرط
                                 للقرص */
    uint64_t dqb_itime;       /* الحد الزمني للملفات المفرطة */
    uint32_t dqb_valid;       /* قناع بت للثوابت QIF_* */
};
/* العلامات في dqb_valid التي تشير إلى الحقول في
   بنية dqblk الصالحة.  */
#define QIF_BLIMITS   1
#define QIF_SPACE     2
#define QIF_ILIMITS   4
#define QIF_INODES    8
#define QIF_BTIME     16
#define QIF_ITIME     32
#define QIF_LIMITS    (QIF_BLIMITS | QIF_ILIMITS)
#define QIF_USAGE     (QIF_SPACE | QIF_INODES)
#define QIF_TIMES     (QIF_BTIME | QIF_ITIME)
#define QIF_ALL       (QIF_LIMITS | QIF_USAGE | QIF_TIMES)

حقل dqb_valid هو قناع بت يتم تعيينه للإشارة إلى الإدخالات في بنية dqblk الصالحة. حاليًا، يملأ النواة جميع إدخالات بنية dqblk ويضع علامة عليها على أنها صالحة في حقل dqb_valid. يمكن للمستخدمين غير المتميزين استرداد حصصهم الخاصة فقط؛ يمكن للمستخدم المتميز (CAP_SYS_ADMIN) استرداد حصص أي مستخدم.
Q_GETNEXTQUOTA (منذ Linux 4.6)
هذه العملية هي نفسها Q_GETQUOTA، ولكنها تعيد معلومات الحصة للمعرف التالي الأكبر من أو يساوي id والذي لديه حصة محددة.
الوسيط addr هو مؤشر إلى بنية nextdqblk التي حقولها كما في dqblk، باستثناء إضافة حقل dqb_id الذي يستخدم لإرجاع المعرف الذي يتم إرجاع معلومات الحصة له:

struct nextdqblk {
    uint64_t dqb_bhardlimit;
    uint64_t dqb_bsoftlimit;
    uint64_t dqb_curspace;
    uint64_t dqb_ihardlimit;
    uint64_t dqb_isoftlimit;
    uint64_t dqb_curinodes;
    uint64_t dqb_btime;
    uint64_t dqb_itime;
    uint32_t dqb_valid;
    uint32_t dqb_id;
};

Q_SETQUOTA
تعيين معلومات الحصة لمعرف المستخدم أو المجموعة id، باستخدام المعلومات المقدمة في بنية dqblk التي يشير إليها addr. يشير حقل dqb_valid في بنية dqblk إلى الحقول التي تم تعيينها من قبل المتصل. هذه العملية تحل محل العمليات Q_SETQLIM و Q_SETUSE في واجهات الحصص السابقة. تتطلب هذه العملية امتيازات (CAP_SYS_ADMIN).
Q_GETINFO (منذ لينكس 2.4.22)
الحصول على معلومات (مثل أوقات السماح) حول ملف الحصص. يجب أن يكون الوسيط addr مؤشرًا إلى بنية dqinfo. يتم تعريف هذه البنية في <sys/quota.h> على النحو التالي:

/* uint64_t هو عدد صحيح غير موقع 64 بت؛
   uint32_t هو عدد صحيح غير موقع 32 بت */
struct dqinfo {         /* معرف منذ لينكس 2.4.22 */
    uint64_t dqi_bgrace;  /* الوقت قبل أن يصبح حد الكتلة الناعم
                             حدًا قاسيًا */
    uint64_t dqi_igrace;  /* الوقت قبل أن يصبح حد الأينود الناعم
                             حدًا قاسيًا */
    uint32_t dqi_flags;   /* علامات لملف الحصص
                             (DQF_*) */
    uint32_t dqi_valid;
};
/* بتات لـ dqi_flags */
/* تنسيق الحصة QFMT_VFS_OLD */
#define DQF_ROOT_SQUASH (1 << 0) /* كبح الجذر ممكّن */
              /* قبل لينكس v4.0، تم تعريف هذا
                 بشكل خاص كـ V1_DQF_RSQUASH */
/* تنسيق الحصة QFMT_VFS_V0 / QFMT_VFS_V1 */
#define DQF_SYS_FILE    (1 << 16)   /* الحصة مخزنة في
                                       ملف نظام */
/* علامات في dqi_valid تشير إلى الحقول في
   بنية dqinfo صالحة.  */
#define IIF_BGRACE  1
#define IIF_IGRACE  2
#define IIF_FLAGS   4
#define IIF_ALL     (IIF_BGRACE | IIF_IGRACE | IIF_FLAGS)

يشير الحقل dqi_valid في بنية dqinfo إلى الإدخالات في البنية التي هي صالحة. حاليًا، يملأ النواة جميع الإدخالات في بنية dqinfo ويضع علامة عليها جميعًا كصالحة في الحقل dqi_valid. يتم تجاهل الوسيط id.
Q_SETINFO (منذ لينكس 2.4.22)
تعيين معلومات حول ملف الحصص. يجب أن يكون الوسيط addr مؤشرًا إلى بنية dqinfo. يشير الحقل dqi_valid لبنية dqinfo إلى الإدخالات في البنية التي تم تعيينها بواسطة المتصل. تتجاوز هذه العملية عمليتي Q_SETGRACE و Q_SETFLAGS في واجهات الحصص السابقة. يتم تجاهل الوسيط id. تتطلب هذه العملية صلاحية (CAP_SYS_ADMIN).
Q_GETFMT (منذ لينكس 2.4.22)
الحصول على تنسيق الحصة المستخدم في نظام الملفات المحدد. يجب أن يكون الوسيط addr مؤشرًا إلى مخزن مؤقت بحجم 4 بايت سيتم تخزين رقم التنسيق فيه.
Q_SYNC
تحديث النسخة الموجودة على القرص لاستخدامات الحصص لنظام ملفات. بالنسبة لـ quotactl()، إذا كان special هو NULL، فسيتم مزامنة جميع أنظمة الملفات ذات الحصص النشطة. (quotactl_fd() دائمًا ما تزامن نظام ملفات واحد فقط.) في كلتا الحالتين، يتم تجاهل الوسيطين addr و id.
Q_GETSTATS (مدعوم حتى لينكس 2.4.21)
الحصول على إحصائيات ومعلومات عامة أخرى حول نظام الحصص الفرعي. يجب أن يكون الوسيط addr مؤشرًا إلى بنية dqstats سيتم تخزين البيانات فيها. يتم تعريف هذه البنية في <sys/quota.h>. يتم تجاهل الوسيطين special و id.
هذه العملية قديمة وتمت إزالتها في لينكس 2.4.22. يتم حمل الملفات في /proc/sys/fs/quota/ للمعلومات بدلاً من ذلك.

بالنسبة لأنظمة ملفات XFS التي تستفيد من مدير حصص XFS (XQM)، يتم تجاوز العمليات المذكورة أعلاه واستخدام العمليات التالية:

Q_XQUOTAON
تشغيل الحصص لنظام ملفات XFS. يوفر XFS القدرة على تشغيل/إيقاف فرض حدود الحصص مع محاسبة الحصص. لذلك، يتوقع XFS أن يكون addr مؤشرًا إلى unsigned int يحتوي على مزيج ثنائي من العلامات التالية (المعرفة في <xfs/xqm.h>):

XFS_QUOTA_UDQ_ACCT  /* محاسبة حصص المستخدم */
XFS_QUOTA_UDQ_ENFD  /* فرض حدود حصص المستخدم */
XFS_QUOTA_GDQ_ACCT  /* محاسبة حصص المجموعة */
XFS_QUOTA_GDQ_ENFD  /* فرض حدود حصص المجموعة */
XFS_QUOTA_PDQ_ACCT  /* محاسبة حصص المشروع */
XFS_QUOTA_PDQ_ENFD  /* فرض حدود حصص المشروع */

تتطلب هذه العملية صلاحية (CAP_SYS_ADMIN). يتم تجاهل الوسيط id.
Q_XQUOTAOFF
إيقاف الحصص لنظام ملفات XFS. كما هو الحال مع Q_QUOTAON، تتوقع أنظمة ملفات XFS مؤشرًا إلى unsigned int يحدد ما إذا كان يجب إيقاف محاسبة الحصص و/أو فرض الحدود (باستخدام نفس العلامات كما في عملية Q_XQUOTAON). تتطلب هذه العملية صلاحية (CAP_SYS_ADMIN). يتم تجاهل الوسيط id.
Q_XGETQUOTA
الحصول على حدود الحصص على القرص والاستخدام الحالي للمستخدم id. الوسيط addr هو مؤشر إلى بنية fs_disk_quota، والتي يتم تعريفها في <xfs/xqm.h> على النحو التالي:

/* جميع وحدات الكتلة بالـ BBs (الكتل الأساسية)
   بحجم 512 بايت.  */
#define FS_DQUOT_VERSION  1  /* fs_disk_quota.d_version */
#define XFS_USER_QUOTA    (1<<0)  /* نوع حصة المستخدم */
#define XFS_PROJ_QUOTA    (1<<1)  /* نوع حصة المشروع */
#define XFS_GROUP_QUOTA   (1<<2)  /* نوع حصة المجموعة */
struct fs_disk_quota {
    int8_t   d_version;   /* إصدار هذه البنية */
    int8_t   d_flags;     /* XFS_{USER,PROJ,GROUP}_QUOTA */
    uint16_t d_fieldmask; /* محدد الحقل */
    uint32_t d_id;        /* معرف المستخدم، المشروع، أو المجموعة */
    uint64_t d_blk_hardlimit; /* الحد المطلق لـ
                                 كتل القرص */
    uint64_t d_blk_softlimit; /* الحد المفضل لـ
                                 كتل القرص */
    uint64_t d_ino_hardlimit; /* الحد الأقصى لـ
                                 الأينودات المخصصة */
    uint64_t d_ino_softlimit; /* الحد المفضل للأينودات */
    uint64_t d_bcount;    /* عدد كتل القرص المملوكة
                             للمستخدم */
    uint64_t d_icount;    /* عدد الأينودات المملوكة
                             للمستخدم */
    int32_t  d_itimer;    /* صفر إذا كان ضمن حدود الأينود */
                          /* إذا لم يكن كذلك، نرفض الخدمة */
    int32_t  d_btimer;    /* مشابه لما سبق؛ لـ
                             كتل القرص */
    uint16_t d_iwarns;    /* عدد التحذيرات الصادرة فيما
                             يتعلق بعدد الأينودات */
    uint16_t d_bwarns;    /* عدد التحذيرات الصادرة فيما
                             يتعلق بكتل القرص */
    int32_t  d_padding2;  /* حشو - للاستخدام المستقبلي */
    uint64_t d_rtb_hardlimit; /* الحد المطلق لـ
                                 كتل القرص للوقت الحقيقي
                                 (RT) */
    uint64_t d_rtb_softlimit; /* الحد المفضل لـ
                                 كتل القرص للوقت الحقيقي
                                 (RT) */
    uint64_t d_rtbcount;  /* عدد كتل الوقت الحقيقي المملوكة */
    int32_t  d_rtbtimer;  /* مشابه لما سبق؛ لـ
                             كتل الوقت الحقيقي */
    uint16_t d_rtbwarns;  /* عدد التحذيرات الصادرة فيما
                             يتعلق بكتل الوقت الحقيقي */
    int16_t  d_padding3;  /* حشو - للاستخدام المستقبلي */
    char     d_padding4[8];   /* المزيد من الحشو */
};

يمكن للمستخدمين غير المميزين استرداد حصصهم فقط؛ يمكن للمستخدم المميز (CAP_SYS_ADMIN) استرداد حصص أي مستخدم.
Q_XGETNEXTQUOTA (منذ لينكس 4.6)
هذه العملية هي نفسها Q_XGETQUOTA، ولكنها تعيد (في بنية fs_disk_quota التي يشير إليها addr) معلومات الحصص للمعرف التالي الأكبر من أو يساوي id والذي تم تعيين حصة له. لاحظ أنه نظرًا لأن fs_disk_quota لديها بالفعل حقل q_id، فلا حاجة لنوع بنية منفصل (على عكس عمليتي Q_GETQUOTA و Q_GETNEXTQUOTA)
Q_XSETQLIM
تعيين حدود الحصص على القرص للمستخدم id. الوسيط addr هو مؤشر إلى بنية fs_disk_quota. تتطلب هذه العملية صلاحية (CAP_SYS_ADMIN).
Q_XGETQSTAT
تعيد معلومات حصص نظام ملفات XFS الخاصة بنظام الملفات المحدد في بنية fs_quota_stat التي يشير إليها addr. هذا مفيد لمعرفة مقدار المساحة المستخدمة لتخزين معلومات الحصص، وأيضًا للحصول على حالة تشغيل/إيقاف الحصص لنظام ملفات XFS محلي معين. يتم تعريف بنية fs_quota_stat نفسها على النحو التالي:

#define FS_QSTAT_VERSION 1  /* fs_quota_stat.qs_version */
struct fs_qfilestat {
    uint64_t qfs_ino;       /* رقم الأينود */
    uint64_t qfs_nblks;     /* عدد الـ BBs
                               كتل 512 بايت */
    uint32_t qfs_nextents;  /* عدد الامتدادات */
};
struct fs_quota_stat {
    int8_t   qs_version; /* رقم الإصدار لـ
                            التغييرات المستقبلية */
    uint16_t qs_flags; /* XFS_QUOTA_{U,P,G}DQ_{ACCT,ENFD} */
    int8_t   qs_pad;   /* غير مستخدم */
    struct fs_qfilestat qs_uquota;  /* معلومات تخزين حصص المستخدم
                                       */
    struct fs_qfilestat qs_gquota;  /* معلومات تخزين حصص المجموعة
                                       */
    uint32_t qs_incoredqs;   /* عدد الـ dquots في الذاكرة الأساسية */
    int32_t  qs_btimelimit;  /* الحد الأقصى لمؤقت الكتل */
    int32_t  qs_itimelimit;  /* الحد الأقصى لمؤقت الأينودات */
    int32_t  qs_rtbtimelimit;/* الحد الأقصى لمؤقت كتل
                                 الوقت الحقيقي */
    uint16_t qs_bwarnlimit;  /* الحد الأقصى لعدد التحذيرات */
