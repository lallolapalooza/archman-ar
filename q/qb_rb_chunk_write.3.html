qb_rb_chunk_write(3)

الحزمة: libqb

التراخيص: LGPL-2.1-or-later

QB_RB_CHUNK_WRITE(3) دليل مبرمج libqb QB_RB_CHUNK_WRITE(3)

الاسم

qb_rb_chunk_write - كتابة قطعة إلى المخزن المؤقت الدائري.

الملخص

#include <qb/qbrb.h>

ssize_t qb_rb_chunk_write(
    qb_ringbuffer_t   *rb,    /* مثيل المخزن المؤقت الدائري */
    const void        *data,  /* (إدخال) البيانات المراد كتابتها */
    size_t             len    /* (إدخال) حجم القطعة. */
);

المعلمات

rb مثيل المخزن المؤقت الدائري

data (إدخال) البيانات المراد كتابتها

len (إدخال) حجم القطعة.

الوصف

هذه الدالة تستدعي ببساطة qb_rb_chunk_alloc() ثم qb_rb_chunk_commit().

قيمة الإرجاع

مقدار البايتات المخزنة فعليًا (إما len أو -1).
