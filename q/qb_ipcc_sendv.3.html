qb_ipcc_sendv(3)

الحزمة: libqb

التراخيص: LGPL-2.1-or-later

QB_IPCC_SENDV(3) دليل مبرمج libqb QB_IPCC_SENDV(3)

اسم

qb_ipcc_sendv - إرسال رسالة (iovec).

الملخص

#include <qb/qbipcc.h>

ssize_t qb_ipcc_sendv(
    qb_ipcc_connection_t   *c,        /* مثيل الاتصال */
    const struct iovec     *iov,      /* مؤشر إلى بنية iovec للإرسال */
    size_t                  iov_len   /* عدد iovecs المستخدمة */
);

المعاملات

c مثيل الاتصال

iov مؤشر إلى بنية iovec للإرسال

iov_len عدد iovecs المستخدمة

الوصف

قيمة الإرجاع

(الحجم المرسل، -errno == خطأ)

ملاحظة

يجب أن يكون iov[0] qb_ipc_request_header. سيقرأ الخادم الحقل size لتحديد مقدار الاستقبال.
