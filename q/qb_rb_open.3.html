qb_rb_open(3)

الحزمة: libqb

التراخيص: LGPL-2.1-or-later

QB_RB_OPEN(3) دليل مبرمج libqb QB_RB_OPEN(3)

الاسم

qb_rb_open - إنشاء مخزن دائري بالنوع المحدد.

الملخص

#include <qb/qbrb.h>

qb_ringbuffer_t * qb_rb_open(
    const char   *name,                   /* الاسم الفريد لهذا المخزن الدائري. */
    size_t        size,                   /* الحجم المطلوب. */
    uint32_t      flags,                  /* العلم أو الأعلام. */
    size_t        shared_user_data_size   /* حجم منطقة بيانات مشتركة. */
);

المعلمات

name الاسم الفريد لهذا المخزن الدائري.

size الحجم المطلوب.

flags العلم أو الأعلام.

shared_user_data_size حجم منطقة بيانات مشتركة.

الوصف

يقوم هذا بإنشاء وتخصيص مخزن دائري في الذاكرة المشتركة.

قيمة الإرجاع

مخزن دائري جديد أو NULL إذا كانت هناك مشكلة.

ملاحظة

سيتم تقريب الحجم الفعلي إلى حجم الصفحة التالي.
