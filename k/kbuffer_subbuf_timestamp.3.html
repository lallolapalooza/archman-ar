kbuffer_subbuf_timestamp(3)

الحزمة: libtraceevent

التراخيص: GPL-2.0-only, LGPL-2.1-only

LIBTRACEEVENT(3) دليل libtraceevent LIBTRACEEVENT(3)

الاسم

kbuffer_timestamp, kbuffer_subbuf_timestamp - دوال تقرأ بيانات مختلفة من واصف kbuffer

المُلَخَّص

#include <kbuffer.h>
unsigned long long kbuffer_timestamp(struct kbuffer *kbuf);
unsigned long long kbuffer_subbuf_timestamp(struct kbuffer *kbuf, void *subbuf);

الوصف

تقوم الدالة kbuffer_timestamp() بإرجاع الطابع الزمني للحدث الحالي لـ kbuf.

تقوم الدالة kbuffer_subbuf_timestamp() بإرجاع الطابع الزمني للمخزن الفرعي الذي تم تحميله في kbuf. هذا عادةً ما يكون (ولكن ليس مضمونًا) الطابع الزمني للحدث الأول في المخزن الفرعي.

تقوم الدالة kbuffer_start_of_data() بإرجاع الإزاحة حيث تبدأ البيانات

قيمة الإرجاع

kbuffer_read_event() ترجع الحدث الذي يوجد فيه واصف kbuf حاليًا، أو NULL إذا تم تجاوز الحدث الأخير (بواسطة kbuffer_next_event()).

kbuffer_next_event() ترجع الحدث التالي بعد الحدث الحالي أو NULL إذا لم تكن هناك أحداث أخرى.

kbuffer_read_at_offset() ترجع الحدث عند offset محدد من بداية المخزن الفرعي المخزن في kbuf، أو NULL إذا لم يوجد حدث. ملاحظة، offset تحتاج فقط إلى أن تكون إزاحة تقع على السجل، أو في بدايته. لا يلزم أن تكون بالضبط في بداية السجل.

kbuffer_missed_events() ترجع 0 إذا لم تكن هناك أحداث مفقودة قبل المخزن الفرعي المحمل. ترجع -1 إذا كان هناك عدد غير معروف من الأحداث المفقودة، أو إذا كان عدد الأحداث المفقودة معروفًا، فسيتم إرجاع هذا العدد.

kbuffer_event_size() ترجع حجم حمولة بيانات الحدث الحالي لـ kbuf.

kbuffer_curr_size() ترجع حجم السجل الكامل للحدث الحالي لـ kbuf. يتضمن هذا حجم البيانات الوصفية لهذا السجل.

kbuf_curr_offset() ترجع إزاحة السجل الحالي من بداية المخزن الفرعي kbuf.

kbuf_curr_index() ترجع فهرس السجل الحالي من بداية قسم بيانات kbuf.

مثال

#include <stdio.h>
#include <stdlib.h>
#include <fcntl.h>
#include <unistd.h>
#include <sys/stat.h>
#include <kbuffer.h>
int main (int argc, char **argv)
{
        unsigned long long ts;
        struct kbuffer *kbuf;
        struct stat st;
        char *buf;
        void *event;
        int save_offset = -1;
        int record_size;
        int offset;
        int index;
        int size;
        int ret;
        int fd;
        int i = 0;
        if (argc < 2) {
                printf("usage: %s raw-subbuffer-page\n", argv[0]);
                printf(" Try: dd count=1 bs=4096 if=/sys/kernel/tracing/per_cpu/cpu0/trace_pipe_raw of=/tmp/file\n");
                exit(0);
        }
        if (stat(argv[1], &st) < 0) {
                perror("stat");
                exit(-1);
        }
        buf = malloc(st.st_size);
        if (!buf) {
                perror("Allocating buffer");
                exit(-1);
        }
        fd = open(argv[1], O_RDONLY);
        if (fd < 0) {
                perror(argv[1]);
                exit(-1);
        }
        ret = read(fd, buf, st.st_size);
        if (ret < 0) {
                perror("Reading buffer");
                exit(-1);
        }
        close(fd);
        kbuf = kbuffer_alloc(KBUFFER_ENDIAN_SAME_AS_HOST,
                             KBUFFER_LSIZE_SAME_AS_HOST);
        if (!kbuf) {
                perror("Creating kbuffer");
                exit(-1);
        }
        ret = kbuffer_load_subbuffer(kbuf, buf);
        if (ret < 0) {
                perror("Loading sub bufer");
                exit(-1);
        }
        if (kbuffer_subbuffer_size(kbuf) > st.st_size) {
                fprintf(stderr, "kbuffer is bigger than raw size %d > %ld\n",
                        kbuffer_subbuffer_size(kbuf), st.st_size);
                exit(-1);
        }
        ret = kbuffer_missed_events(kbuf);
        if (ret) {
                if (ret > 0)
                        printf("Missed %d events before this buffer\n", ret);
                else
                        printf("Missed unknown number of events before this buffer\n");
        }
        do {
                event = kbuffer_read_event(kbuf, &ts);
                if (event) {
                        record_size = kbuffer_curr_size(kbuf);
                        offset = kbuffer_curr_offset(kbuf);
                        index = kbuffer_curr_index(kbuf);
                        size = kbuffer_event_size(kbuf);
                        if (i == 20)
                                save_offset = offset;
                        printf(" event %3d ts:%lld\trecord_size:%d size:%d\tindex:%d offset:%d\n",
                               i++, ts, record_size, size, index, offset);
                        event = kbuffer_next_event(kbuf, NULL);
                }
        } while (event);
        if (!event)
                printf("Finished sub buffer\n");
        if (save_offset > 0) {
                event = kbuffer_read_at_offset(kbuf, save_offset, &ts);
                if (!event) {
                        fprintf(stderr, "Funny, can't find event 20 at offset %d\n", save_offset);
                        exit(-1);
                }
                record_size = kbuffer_curr_size(kbuf);
                offset = kbuffer_curr_offset(kbuf);
                index = kbuffer_curr_index(kbuf);
                size = kbuffer_event_size(kbuf);
                printf("\n saved event 20 ts:%lld\trecord_size:%d size:%d\tindex:%d offset:%d\n\n",
                       ts, record_size, size, index, offset);
        }
        kbuffer_free(kbuf);
        return 0;
}
