keepalived.conf(5)

الحزمة: keepalived

التراخيص: GPL-2.0-or-later

لم يتم تنفيذ ملفات التتبع للخوادم الحقيقية بالكامل بعد. على وجه الخصوص، السماح بالوزن 0، والتعامل مع القيم المحسوبة السالبة، وإعادة التحميل.

بنية ملف التتبع هي:
track_file <STRING> {	# vrrp_track_file هو مرادف مهمل
    # الملف المراد تتبعه (الوزن الافتراضي 1)
    file <QUOTED-STRING>
    # وزن افتراضي اختياري
    weight <-2147483647..2147483647> [reverse]
    # إنشاء الملف و/أو تهيئة القيمة
    # هذا يؤدي إلى كتابة VALUE (الافتراضي 0) إلى
    # الملف المحدد عند بدء التشغيل إذا لم يكن الملف موجودًا،
    # ما لم يتم تحديد overwrite، وفي هذه الحالة سيتم الكتابة فوق أي محتوى موجود للملف
    # بالقيمة المحددة.
    init_file [VALUE] [overwrite]
}

عمليات تتبع VRRP

كتلة التكوين تبدو كالتالي:
    vrrp_track_process <STRING> {
        # العملية المراد مراقبتها (مع معلمات اختيارية)
        # تُعامل السلسلة النصية المقتبسة كعنصر واحد، لذا إذا كان العنصر الأول
        # بعد كلمة العملية مقتبسًا، فسيكون هذا هو اسم الأمر.
        # على سبيل المثال:
        #  process "/tmp/a b" param1 "param 2"
        # يعني عملية باسم '/tmp/a b' (تمت إزالة علامات الاقتباس) مع معلمات 2
        #  'param1' و 'param 2'.
        process <STRING>|<QUOTED-STRING> [<STRING>|<QUOTED-STRING> ...]
        # إذا كانت المعلمات متطابقة، فهذا يحدد تطابقًا جزئيًا (أي، أول
        #   n معلمات متطابقة بالضبط)، أو تطابقًا أوليًا، أي المعلمة الأخيرة
        #   قد تكون أطول من المعلمة المكونة.
        # لتحديد أن الأمر يجب ألا يحتوي على معلمات، لا تحدد
        #   أي معلمات، ولكن حدد param_match.
        param_match {initial|partial}
        # الوزن الافتراضي (الافتراضي 0). لوصف reverse، انظر track_process.
        # 'weight 0 reverse' سيؤدي إلى تعطل مثيل vrrp عندما يكون
        #   النصاب مرتفعًا، والعكس صحيح.
	# وزن غير صفري سيضبط أولوية VRRP لمثيل VRRP المتعقب،
	#   بينما سيؤدي وزن 0 إلى انتقال مثيل VRRP إلى حالة FAULT إذا كانت
	#   عملية التتبع في حالة فشل (انظر أعلاه لتأثير "reverse").
        weight <-254..254> [reverse]
        # الحد الأدنى لعدد العمليات للنجاح (الافتراضي 1)
        quorum NUM
        # الحد الأقصى لعدد العمليات للنجاح. على سبيل المثال، تحديد
        #   هذا الرقم إلى 1 سيؤدي إلى فشل إذا كانت هناك نسختان من العملية
        #   قيد التشغيل (ولكن احذر الانقسامات - انظر fork_delay أدناه).
        #   تحديد هذا الرقم إلى 0 يعني الفشل إذا كانت العملية المطابقة
        #   قيد التشغيل على الإطلاق.
	#   الافتراضي غير محدود.
        quorum_max NUM
        # الوقت للتأخير بعد اكتساب نصاب العملية بعد الانقسام قبل
        #   اعتبار العملية قيد التشغيل (بالأجزاء من الثانية)
        #   هذا لتجنب الارتداد لأعلى/لأسفل بسبب الانقسام/التنفيذ
        fork_delay SECS
        # الوقت للتأخير بعد فقدان نصاب العملية قبل
        #   اعتبار العملية متوقفة (بالأجزاء من الثانية)
        #   هذا لتجنب الارتداد لأسفل/لأعلى بعد الإنهاء/إعادة الانقسام للأصل.
        terminate_delay SECS
        # هذا يضبط fork_delay و terminate_delay
        delay SECS
        # عادةً ما تتم مطابقة سلسلة العملية مع اسم العملية،
        #   كما هو موضح في سطر Name: في /proc/PID/status، ما لم
        #   يتم تحديد معلمات.
        #   هذا الخيار يفرض مطابقة سطر الأوامر بالكامل
        full_command
    }

لتجنب الحاجة إلى تشغيل سكربت تتبع بشكل متكرر لمراقبة وجود العمليات (غالبًا haproxy أو nginx)، يمكن لـ vrrp_track_process مراقبة ما إذا كانت العمليات الأخرى قيد التشغيل.

فرق واحد عن pgrep هو أن track_process لا يقوم بمطابقة تعبير عادي لسلسلة الأوامر، ولكنه يقوم بمطابقة دقيقة. 'pgrep ssh' سيطابق عملية sshd، وهذا track_process لن يفعل ذلك (إنه يعادل pgrep "^ssh$").

إذا تم استخدام full_command (يعادل pgrep -f)، يتم استخدام /proc/PID/cmdline، ولكن لن يتم اكتشاف أي تحديثات لـ cmdline (لا ينبغي للعملية عادةً تغييرها، على الرغم من أنه من الممكن مع عناية كبيرة، على سبيل المثال systemd).

قبل Linux v3.2، لن يدعم track_process اكتشاف التغييرات في اسم العملية، حيث أن النواة لم تكن ترسل إشعارات بتغييرات اسم العملية قبل 3.2. معظم العمليات لا تغير أسماءها، ولكن، على سبيل المثال، يقوم firefox بعمل انقسامات لعمليات تغير أسماءها إلى "Web Content". اسم العملية المشار إليه هنا هو محتوى /proc/PID/comm.

النصاب هو عدد العمليات المتطابقة التي يجب تشغيلها لحالة OK.

قد يكون التأخير مفيدًا إذا كان من المتوقع أن يتم إعادة تحميل عملية (إيقافها وإعادة تشغيلها)، ولا يُرغب في إيقاف وتشغيل مثيل vrrp.

الوزن الإيجابي يعني أن حالة OK ستضيف <weight> إلى أولوية جميع مثيلات VRRP التي تراقبها. على العكس من ذلك، سيتم طرح وزن سلبي من الأولوية الأولية في حالة وجود عمليات غير كافية.

إذا كان مثيل vrrp أو مجموعة المزامنة ليس مالك العنوان وكان الناتج بين -253 و 253، فسيتم إضافة الناتج إلى الأولوية الأولية لمثيل VRRP (قيمة سالبة ستقلل الأولوية)، على الرغم من أن الأولوية الفعالة ستقتصر على النطاق [1،254].

إذا كان مثيل vrrp يستخدم track_process عضوًا في مجموعة مزامنة، ما لم يتم تعيين sync_group_tracking_weight في المجموعة، يجب تعيين وزن 0. وبالمثل، إذا كان مثيل vrrp هو مالك العنوان، فيجب أيضًا تعيين وزن 0.

المنطق لعدم استخدام pgrep/pidof/killall وما شابه ذلك:

في كل مرة يتم فيها تشغيل pgrep أو ما يعادله، فإنه يتكرر عبر الدلائل /proc/[1-9][0-9]*، ويفتح الملفات الوهمية status و cmdline في كل دليل. يتم تعيين الملف الوهمي cmdline إلى مساحة عنوان العملية، لذا إذا تم تبديل هذا الجزء من العملية، فسيتعين جلبه من مساحة التبديل. pgrep إلخ تشمل أيضًا العمليات الزومبي بينما keepalived لا يفعل، لأنها ليست قيد التشغيل.

هذا التنفيذ يتكرر فقط عبر دلائل /proc/[1-9][0-9]*/ عند بدء التشغيل، ولن يقرأ حتى الملفات الوهمية cmdline إذا لم يتم تحديد 'full_command' لأي من إدخالات vrrp_track_process. بعد بدء التشغيل، يستخدم موصل kernel <-> userspace process_events لتلقي إشعارات بتغييرات العمليات. إذا تم تحديد full_command لأي مثيل track_process، فسيتعين قراءة الملف الوهمي cmdline عند الإشعار بإنشاء عملية جديدة، ولكن في ذلك الوقت من غير المرجح جدًا أن تكون قد تم تبديلها بالفعل.

في نظام مزدحم بعدد كبير من عمليات الإنشاء/الإنهاء، قد يكون استخدام track_script مع pgrep/pidof/killall أكثر كفاءة، على الرغم من أن هذه العمليات غير فعالة مقارنة بالحد الأدنى الذي يحتاجه keepalived.

استخدام pgrep إلخ على نظام يقوم بالتبديل يمكن أن يؤثر سلبًا بشكل كبير على أداء النظام، بسبب الحاجة إلى جلب الذاكرة المبدلة من مساحة التبديل، مما يتسبب في تبديل إضافي.

تكوين BFD

هذا هو تنفيذ للمعيار RFC5880 (Bidirectional forwarding detection)، ويمكن تكوينه للعمل بين مثيلي keepalived، ولكن استخدام track_bfds غير الموزونة بين زوج من مثيلات VRRP الرئيسية/الاحتياطية يعني أن مثيل VRRP لن يتمكن من العمل إلا إذا كانت كلا مثيلي VRRP قيد التشغيل، مما يقوض إلى حد ما الغرض من VRRP.

تم اختبار هذا التنفيذ مع OpenBFDD (متوفر على https://github.com/dyninc/OpenBFDD).

بنية مثيل bfd هي:
bfd_instance <STRING> {
    # عنوان IP المجاور لـ BFD (مرادف neighbour_ip)
    neighbor_ip <IP ADDRESS>
    # عنوان IP المصدر المراد استخدامه (اختياري، باستثناء ضمان أن
    # المنفذ المحلي صالح، فهو مطلوب)
    source_ip <IP ADDRESS>
    # الحد الأدنى لوقت الاستقبال المطلوب، بالمللي ثانية (الدقة هي ميكرو ثانية، على سبيل المثال 3.312)
    # (الافتراضي 10 مللي ثانية)
    min_rx <DECIMAL>
    # الحد الأدنى لوقت الإرسال المطلوب، بالمللي ثانية (الدقة هي ميكرو ثانية)
    # (الافتراضي 10 مللي ثانية)
    min_tx <DECIMAL>
    # وقت الإرسال الخامل المطلوب، بالمللي ثانية (الدقة هي ميكرو ثانية)
    # (الافتراضي 1000 مللي ثانية)
    idle_tx <DECIMAL>
    # عدد الحزم المفقودة بعد
    # والذي يتم فيه الإعلان عن انتهاء الجلسة
    # (الافتراضي 5)
    multiplier <INTEGER>
    # العمل في وضع سلبي (الافتراضي نشط)
    passive
    # TTL IPv4 الصادر المراد استخدامه (الافتراضي 255)
    ttl <INTEGER>
    # عدد القفزات في IPv6 الصادرة المراد استخدامه (الافتراضي 64)
    hoplimit <INTEGER>
    # الحد الأقصى لتقليل TTL/hoplimit
    #  في الحزمة المستلمة (الافتراضي 0)
    #  (255 يعطل فحص عدد القفزات)
    max_hops <INTEGER>
    # يحدد RFC 5883 المنفذ 4784 يجب استخدامه لـ multihop bfd، بدلاً من
    # المنفذ 3784. تحديد multihop يتيح هذا الخيار، ولكن إذا كانت هناك قفزات متعددة
    # قيد الاستخدام، فستحتاج أيضًا إلى تكوين max_hops (انظر أعلاه).
    multihop [<BOOL>]
    # الوزن الافتراضي للتتبع
    # عادةً، يتم إضافة الأوزان الإيجابية إلى أولوية مثيل vrrp عندما
    # يكون مثيل bfd نشطًا، وتطرح الأوزان السلبية الأولوية عندما يكون غير نشط.
    # ومع ذلك، إذا تم تحديد reverse، فإن الوزن الإيجابي يقلل الأولوية عندما يكون
    # السكربت نشطًا، ويحسب الوزن السلبي الأولوية عندما يكون السكربت
    # 'weight 0 reverse' سيؤدي إلى تعطل مثيل vrrp عندما يكون
    # مثيل bfd نشطًا، والعكس صحيح.
    # الوزن  العكس  السكربت نشط   السكربت غير نشط
    #  +ve      لا	prio +		-
    #  -ve	لا	   -	     prio -
    #  +ve      نعم	prio -		-
    #  -ve	نعم	   -	     prio +
    weight <-253:253> [reverse]
    # عادةً ما يتم إرسال إشعارات أحداث bfd إلى كل من عمليتي VRRP و checker.
    # تحديد vrrp أو checker سيؤدي إلى إرسال إشعارات الأحداث لهذا bfd_instance
    # فقط إلى العملية المحددة
    vrrp
    checker
}

تكوين VRRPD

يحتوي على كتل فرعية لـ سكربت(ات) VRRP، مجموعة(ات) مزامنة VRRP، مجموعة(ات) تأخير ARP المجاني وإعلان الجوار غير المرغوب فيه لـ VRRP و مثيل(ات) VRRP

سكربت(ات) VRRP

سيتم تنفيذ السكربت بشكل دوري، كل <interval> ثانية. سيتم تسجيل رمز الخروج الخاص به لجميع مثيلات VRRP التي تراقبها. لاحظ أنه لن يتم تنفيذ السكربت إلا إذا كان هناك مثيل VRRP واحد على الأقل يراقبه.

الوزن الافتراضي يساوي 0، مما يعني أن أي مثيل VRRP يراقب السكربت سينتقل إلى حالة الفشل بعد <fall> فشل متتالي للسكربت. بعد ذلك، <rise> نجاحات متتالية ستؤدي إلى مغادرة مثيلات VRRP لحالة الفشل، ما لم تكن أيضًا في حالة الفشل بسبب سكربتات أو واجهات أخرى تتتبعها.

الوزن الإيجابي يعني أن <rise> نجاحات ستضيف

Sequence blocks

Sequence blocks are used for generating a series of lines where a variable is incremented by a specified amount. They are indicated by the ~SEQ() directive.

The format is ~SEQ(var, start, end, step). $var will be incremented by step repeatedly, terminating when it is greater than end. step may be omitted, in which case it defaults to 1 or -1, depending on whether end is greater or less than start. start may also be omitted, in which case it defaults to 1 if end > 0 or -1 if end < 0.

~SEQx(...) is the same as ~SEQ(...), except the variable $var will for formatted in hexadecimal, which would be useful for IPv6 addresses.

Note: At the moment it is necessary to use different variables for the ~SEQ block from any previously defined variable, including one used as the variable in a previous ~SEQ block. This may change in the future, so do not rely on a ~SEQ block variable being defined after the end of the block.

Examples:

    ~SEQ(SUBNET, 0, 3) ip_address 10.0.${SUBNET}.1
    would produce:
        ip_address 10.0.0.1
        ip_address 10.0.1.1
        ip_address 10.0.2.1
        ip_address 10.0.3.1
and
    ~SEQx(SUBNET, 144, 16, 192) ip_address fe80::20:${SUBNET}:1
  or better
    ~SEQx(SUBNET, 0x90, 0x10, 0xc0) ip_address fe80::20:${SUBNET}:1
    would produce:
        ip_address fe80::20:90:1
        ip_address fe80::20:a0:1
        ip_address fe80::20:b0:1
        ip_address fe80::20:c0:1
   Another example:
	virtual_ipaddress {
	    ~SEQx(AD2, 0x90, 0x10, 0xc0) ~SEQx(AD1, 0x12, -1, 0x0c) fe81::10:${AD2}:${AD1}
	}

There can be multiple ~SEQ elements on a line, so for example:

    $VI4= \
    track_file offset_instance_4.${IF}.${NUM}.${ID} { \
        file "${_PWD}/679/track_files/4.${IF}.${NUM}.${ID}" \
        weight -100 \
    } \
    vrrp_instance vrrp4.${IF}.${NUM}.${ID} { \
        interface bond${IF}.${NUM} \
        use_vmac vrrp4.${IF}.${NUM}.${ID} \
        virtual_router_id ${ID} \
        priority 130 \
        virtual_ipaddress { \
            10.${IF}.${NUM}.${ID}/24 \
        } \
        track_file { \
            offset_instance_4.${IF}.${NUM}.${ID} \
        } \
    }
    ~SEQ(IF,0,7) ~SEQ(NUM,0,31) ~SEQ(ID,1,254) $VI4
    will produce 65024 vrrp instances with names from vrrp4.0.0.1 through to
    vrrp4.7.31.254.

List blocks

List blocks are similar to sequence blocks, except that the values to substitute into the variable are listed in the ~LST specification.

A line starting ~LST(var, val1, val2, val3) will cause the remainder of the line to be processed multiple times, with the variable $var set initially to val1, and then val2, and finally val3. Any number of values can be specified, as long as at least one value is (although only one value would be pointless).

If it is desired to substitute more than one variable at a time, the variables and values need to be enclosed in {...} blocks. For example:


~LST({IP, IP1}, {10,1},{20,4},{5,6},{12,8}) 192.168.${IP}.${IP1}

would first set IP=10 and IP1=1, then IP=20 and IP1=4, etc, and produces:


192.168.10.1
192.168.20.4
192.168.5.6
192.168.12.8

List blocks can be nested, so:


~LST(IP, 1, 2, 3, 4) ~LST(IP1, 5,6,7) 192.169.${IP}.${IP1}

produces:
192.169.1.5
192.169.1.6
192.169.1.7
192.169.2.5
192.169.2.6
192.169.2.7
192.169.3.5
192.169.3.6
192.169.3.7
192.169.4.5
192.169.4.6
192.169.4.7

Finally, list blocks and sequence blocks can be combined, so:


~LST({IP, IP1}, {10,1},{20,4},{5,6},{12,8}) ~SEQ(IP2,168,2,172) 192.${IP2}.${IP}.${IP1}

produces:


192.168.10.1
192.170.10.1
192.172.10.1
192.168.20.4
192.170.20.4
192.172.20.4
192.168.5.6
192.170.5.6
192.172.5.6
192.168.12.8
192.170.12.8
192.172.12.8

KERNEL SETTINGS

It has been identified that if proxy_arp and proxy_arp_pvlan are enabled on an interface that has VIPs or eVIPs configured on it, it can cause incorrect replies to ARP requests due to the proxy replying to the ARP request as well as the keepalived host. Both need to be set to 0 to function properly.
