kbuffer_dup(3)

الحزمة: libtraceevent

التراخيص: GPL-2.0-only, LGPL-2.1-only

LIBTRACEEVENT(3) دليل libtraceevent LIBTRACEEVENT(3)

الاسم

kbuffer_alloc, kbuffer_dup, kbuffer_free, kbuffer_load_subbuffer, kbuffer_subbuffer, kbuffer_refresh, kbuffer_subbuffer_size, kbuffer_start_of_data - إنشاء عنصر kbuffer لتحليل مخزن نواة لينكس المؤقت الدائري للتتبع

الملخص

#include <kbuffer.h>
enum kbuffer_endian {
        KBUFFER_ENDIAN_BIG,
        KBUFFER_ENDIAN_LITTLE,
        KBUFFER_ENDIAN_SAME_AS_HOST,
};
enum kbuffer_long_size {
        KBUFFER_LSIZE_4,
        KBUFFER_LSIZE_8,
        KBUFFER_LSIZE_SAME_AS_HOST,
};
struct kbuffer;
struct tep_handle;
struct kbuffer *kbuffer_alloc(enum kbuffer_long_size size, enum kbuffer_endian endian);
struct kbuffer *kbuffer_dup(struct kbuffer *kbuf);
void kbuffer_free(struct kbuffer *kbuf);
int kbuffer_load_subbuffer(struct kbuffer *kbuf, void *subbuffer);
int kbuffer_subbuffer_size(struct kbuffer *kbuf);
int kbuffer_refresh(struct kbuffer *kbuf);
int kbuffer_start_of_data(struct kbuffer *kbuf);
void *kbuffer_subbuffer(struct kbuffer *_kbuf);

الوصف

تقوم هذه الدوال بإنشاء معالج kbuffer يمكن استخدامه لتحليل المخازن الفرعية الخام لمخزن نواة لينكس المؤقت الدائري للتتبع. يوجد المخزن المؤقت الدائري في دليل tracefs، ويمكن استرداده بواسطة tracefs_instance_get_file(3) في per_cpu/cpuX/trace_pipe_raw حيث يتم استبدال X برقم المعالج الخاص بالمخزن المؤقت الدائري المحدد. ينقسم المخزن المؤقت الدائري داخل النواة لكل معالج، بحيث يجب استرداد المخزن المؤقت الدائري الخام لكل معالج أيضًا.

سيقوم kbuffer_alloc() بإنشاء واصف يمكن استخدامه لإدارة مخزن فرعي تم قراءته من المخزن المؤقت الدائري. تحدد المعلمة size حجم الكلمة للمخزن المعطى (ملاحظة، هذا يعمل عند قراءة بيانات خام من آلات أخرى غير الآلة التي تستدعي هذه الوظيفة). تحدد endian ترتيب البايتات للآلة.

إذا تم تعيين endian إلى KBUFFER_ENDIAN_SAME_AS_HOST، فسيتم تعيين ترتيب البايتات إلى نفس ترتيب البايتات الخاص بالنظام المضيف، وهو أمر مفيد عندما يقرأ التطبيق بيانات المخزن المؤقت الدائري مباشرة من نفس الآلة التي يعمل عليها.

إذا تم تعيين size إلى KBUFFER_LSIZE_SAME_AS_HOST، وإذا كان حجم الكلمة 8، فسيتم تعيين واصف kbuffer إلى حجم طويل يبلغ 8. ولكن إذا كان الحجم 4، فسيقوم بعد ذلك بتنفيذ استدعاء uname(2)، وإذا كان الحقل machine يحتوي على السلسلة "64"، فسيتم تعيينه إلى حجم 8 بايت وليس 4 بايت. هذا لأن حجم كلمة المخزن المؤقت الدائري يعتمد على النواة وليس على مساحة المستخدم.

تقوم وظيفة kbuffer_dup() بنسخ هيكل kbuffer موجود مع هيكل جديد مخصص. سيكون له جميع خصائص kbuf الذي تم تمريره، بما في ذلك الإشارة إلى المخزن الفرعي نفسه الذي تم تحميله في kbuf. يجب تحريره باستخدام kbuffer_free().

تقوم وظيفة kbuffer_free() بتحرير الموارد التي تم إنشاؤها بواسطة kbuffer_alloc().

تأخذ kbuffer_load_subbuffer() subbuffer وهو مجموعة بيانات خام من ملف trace_pipe_raw في tracefs. يتم تقسيم مخزن نواة لينكس المؤقت الدائري إلى مخازن فرعية. كل مخزن فرعي هو مقطع بيانات مستقل يحتوي على جميع المعلومات لتقسيم الأحداث والطوابع الزمنية الفردية. هذا المخزن الفرعي هو ما يستخدمه kbuffer للتنقل عبر الأحداث.

تقوم kbuffer_subbuffer_size() بإرجاع موقع نهاية آخر حدث في المخزن الفرعي. لا تقوم بإرجاع حجم المخزن الفرعي نفسه.

يجب استخدام kbuffer_refresh() إذا تم إجراء المزيد من الكتابات على kbuffer المحمل حيث يحتاج حجم kbuffer إلى تحديثه لتكون قادرًا على قراءة الأحداث الجديدة التي تمت كتابتها منذ استدعاء kbuffer_load_subbuffer() الأخير عليه.

ملاحظة، لم يتم تنفيذ حواجز الذاكرة بهذه الوظيفة وأي مزامنة مع الكاتب هي مسؤولية التطبيق.

تقوم وظيفة kbuffer_start_of_data() بإرجاع الإزاحة حيث يبدأ تحميل البيانات الفعلي للمخزن الفرعي.

تقوم وظيفة kbuffer_subbuffer() بإرجاع مؤشر إلى المخزن الفرعي المحمل حاليًا. أي، آخر مخزن فرعي تم تحميله بواسطة kbuffer_load_subbuffer(). إذا لم يتم تحميل أي مخزن فرعي، يتم إرجاع NULL.

قيمة الإرجاع

kbuffer_alloc() ترجع واصف kbuffer مخصص أو NULL عند حدوث خطأ. يجب تحرير الواصف المرتجع باستخدام kbuffer_free()

kbuffer_load_subbuffer() ترجع 0 عند النجاح و -1 عند الخطأ.

kbuffer_subbuffer_size() ترجع الفهرس في المخزن الفرعي حيث يقع نهاية آخر حدث.

kbuffer_start_of_data() ترجع إزاحة بداية البيانات في المخزن الفرعي المحمل في kbuf.

kbuffer_subbuffer() ترجع آخر مخزن فرعي تم تحميله إلى kbuf بواسطة kbuffer_load_subbuffer() أو NULL إذا لم يتم تحميل أي منها.

kbuffer_refresh() ترجع 0 عند النجاح و -1 إذا كان kbuf هو NULL أو لم يتم تحميل مخزن فرعي له عبر kbuffer_load_subbuffer().

مثال

#include <stdio.h>
#include <stdlib.h>
#include <fcntl.h>
#include <unistd.h>
#include <sys/stat.h>
#include <kbuffer.h>
int main (int argc, char **argv)
{
        unsigned long long ts;
        struct kbuffer *kbuf;
        struct stat st;
        char *buf;
        void *event;
        int ret;
        int fd;
        int i = 0;
        if (argc < 2) {
                printf("usage: %s raw-subbuffer-page\n", argv[0]);
                printf(" Try: dd count=1 bs=4096 if=/sys/kernel/tracing/per_cpu/cpu0/trace_pipe_raw of=/tmp/file\n");
                exit(0);
        }
        if (stat(argv[1], &st) < 0) {
                perror("stat");
                exit(-1);
        }
        buf = malloc(st.st_size);
        if (!buf) {
                perror("Allocating buffer");
                exit(-1);
        }
        fd = open(argv[1], O_RDONLY);
        if (fd < 0) {
                perror(argv[1]);
                exit(-1);
        }
        ret = read(fd, buf, st.st_size);
        if (ret < 0) {
                perror("Reading buffer");
                exit(-1);
        }
        close(fd);
        kbuf = kbuffer_alloc(KBUFFER_ENDIAN_SAME_AS_HOST,
                             KBUFFER_LSIZE_SAME_AS_HOST);
        if (!kbuf) {
                perror("Creating kbuffer");
                exit(-1);
        }
        ret = kbuffer_load_subbuffer(kbuf, buf);
        if (ret < 0) {
                perror("Loading sub bufer");
                exit(-1);
        }
        if (kbuffer_subbuffer_size(kbuf) > st.st_size) {
                fprintf(stderr, "kbuffer is bigger than raw size %d > %ld\n",
                        kbuffer_subbuffer_size(kbuf), st.st_size);
                exit(-1);
        }
        printf("Kbuffer data starts at %d\n", kbuffer_start_of_data(kbuf));
        do {
                event = kbuffer_read_event(kbuf, &ts);
                if (event) {
                        printf(" event %3d ts:%lld\n", i++, ts);
                        event = kbuffer_next_event(kbuf, NULL);
                }
        } while (event);
        if (!event)
                printf("Finished sub buffer\n");
        kbuffer_free(kbuf);
        return 0;
}
