kbuffer_read_buffer(3)

الحزمة: libtraceevent

التراخيص: GPL-2.0-only, LGPL-2.1-only

LIBTRACEEVENT(3) دليل libtraceevent LIBTRACEEVENT(3)

الاسم

kbuffer_read_event, kbuffer_next_event, kbuffer_missed_events, kbuffer_event_size, kbuffer_curr_size, kbuffer_curr_offset, kbuffer_curr_index, kbuffer_read_buffer - وظائف للقراءة عبر مخزن kbuffer الفرعي.

الملخص

#include <kbuffer.h>
void *kbuffer_read_event(struct kbuffer *kbuf, unsigned long long *ts);
void *kbuffer_next_event(struct kbuffer *kbuf, unsigned long long *ts);
void *kbuffer_read_at_offset(struct kbuffer *kbuf, int offset, unsigned long long *ts);
int kbuffer_missed_events(struct kbuffer *kbuf);
int kbuffer_event_size(struct kbuffer *kbuf);
int kbuffer_curr_size(struct kbuffer *kbuf);
int kbuffer_curr_offset(struct kbuffer *kbuf);
int kbuffer_curr_index(struct kbuffer *kbuf);
int kbuffer_read_buffer(struct kbuffer *kbuf, void *buffer, int len);

الوصف

تقوم الدالة kbuffer_read_event() بقراءة الحدث التالي في واصف kbuf وإذا كان ts غير NULL، فستضع الطابع الزمني الخاص به فيه. هذا لا يغير واصف kbuf، واستدعاء هذه الدالة عدة مرات سيعطي نفس النتيجة.

تقوم الدالة kbuffer_next_event() بإرجاع الحدث التالي في واصف kbuf. ستقوم أيضًا بتعيين ts إلى الطابع الزمني للحدث الذي تم إرجاعه. يتم إرجاع NULL إذا لم تكن هناك المزيد من الأحداث وسيتم تعريف ts بشكل غير صحيح. لاحظ، إذا تم استدعاء هذا مباشرة بعد kbuffer_load_subbuffer() فمن المحتمل أن يعطي نتيجة غير متوقعة، لأنه سيعيد الحدث الثاني وليس الحدث الأول. عادةً ما تُستخدم هذه الدالة فقط للانتقال إلى الحدث التالي ومعرفة ما إذا كان هناك أي أحداث أخرى للقراءة، ودائماً ما يتم استدعاء kbuffer_read_event() أولاً.

تُرجع الدالة kbuffer_read_at_offset() الحدث الموجود في offset معين من بداية المخزن الفرعي. يمكن الحصول على هذا الـ offset بواسطة kbuffer_curr_offset(). إذا كان ts يشير إلى unsigned long long، فسيتم تعيينه إلى الطابع الزمني للحدث الموجود في الـ offset المعطى.

إذا كان المخزن الفرعي قد فاتته أحداث قبله، فستُرجع kbuffer_missed_events() قيمة غير صفرية. إذا أعادت -1، فهذا يعني أنه كانت هناك أحداث تم تفويتها، ولكن العدد الدقيق للأحداث التي تم تفويتها غير معروف. إذا أعادت رقمًا موجبًا، فإن عدد الأحداث التي تم تفويتها هو القيمة المرجعة.

تُرجع الدالة kbuffer_event_size() حجم جزء البيانات للحدث الحالي (الذي سيتم إرجاعه بواسطة kbuffer_read_event().

تُرجع الدالة kbuffer_curr_size() حجم السجل بأكمله للحدث الحالي (الذي سيتم إرجاعه بواسطة kbuffer_read_event(). الاختلاف هنا هو أن القيمة المرجعة تتضمن حجم البيانات الوصفية للسجل الحدث التي ليست جزءًا مما يتم إرجاعه بواسطة kbuffer_read_event().

تُرجع الدالة kbuffer_curr_offset() الـ offset من بداية المخزن الفرعي حيث تبدأ البيانات الوصفية للحدث الحالي للسجل. الحدث الأول لن يكون في الـ offset صفر. يمكن استخدام هذا الـ offset لاسترداد الحدث باستخدام kbuffer_read_at_offset().

تُرجع الدالة kbuffer_curr_index() الفهرس من بداية جزء البيانات للمخزن الفرعي حيث توجد البيانات الوصفية للحدث الحالي للسجل. الحدث الأول على الأرجح سيكون صفرًا، ولكن قد لا يكون كذلك إذا كان هناك طابع زمني مرفق به.

تقوم الدالة kbuffer_read_buffer() بملء buffer المعطى من kbuf بنفس الطريقة التي يقوم بها النواة باستدعاء نظام القراءة. أي، إذا كان طول len أقل من حجم المخزن الفرعي، أو كان مؤشر kbuffer الحالي غير صفري، فسيبدأ النسخ من الحدث الحالي لـ kbuf وإنشاء buffer كمخزن فرعي جديد (مع طابع زمني ورأس التزام) بهذا الحدث الذي تم العثور عليه وتضمين جميع الأحداث بعد ذلك التي يمكن أن تتناسب ضمن len. يجب أن يتضمن len حجم رأس المخزن الفرعي بالإضافة إلى الأحداث المراد تضمينها. أي، len هو حجم التخصيص لـ buffer الذي يمكن ملؤه. القيمة المرجعة من هذه الدالة هي مؤشر نهاية آخر حدث تم إضافته. إذا لم تكن هناك المزيد من الأحداث، يتم إرجاع صفر، وإذا لم يتمكن المخزن من نسخ أي أحداث لأن len كان صغيرًا جدًا، فسيتم إرجاع -1.

قيمة الإرجاع

تُرجع kbuffer_read_event() الحدث الذي يوجد به واصف kbuf حاليًا، أو NULL إذا تم تجاوز الحدث الأخير (بواسطة kbuffer_next_event()).

تُرجع kbuffer_next_event() الحدث التالي بعد الحدث الحالي أو NULL إذا لم تكن هناك المزيد من الأحداث.

تُرجع kbuffer_read_at_offset() الحدث في offset معين من بداية المخزن الفرعي المخزن في kbuf، أو NULL إذا لم يوجد حدث. لاحظ، offset يحتاج فقط لأن يكون offset يقع على السجل، أو في بدايته. لا يلزمه أن يكون بالضبط في بداية السجل.

تُرجع kbuffer_missed_events() 0 إذا لم تكن هناك أحداث تم تفويتها قبل المخزن الفرعي المحمل. تُرجع -1 إذا كان هناك عدد غير معروف من الأحداث التي تم تفويتها، أو إذا كان عدد الأحداث التي تم تفويتها معروفًا، فسيتم إرجاع هذا العدد.

تُرجع kbuffer_event_size() حجم حمولة البيانات للحدث الحالي لـ kbuf.

تُرجع kbuffer_curr_size() حجم السجل بأكمله للحدث الحالي لـ kbuf. هذا يشمل حجم البيانات الوصفية لهذا السجل.

تُرجع kbuf_curr_offset() offset السجل الحالي من بداية المخزن الفرعي لـ kbuf.

تُرجع kbuf_curr_index() مؤشر السجل الحالي من بداية قسم بيانات kbuf.

تُرجع kbuf_read_buffer() مؤشر نهاية آخر حدث تم ملؤه في buffer. إذا لم تكن هناك المزيد من الأحداث لنسخها من start فسيتم إرجاع 0. إذا لم يكن len كبيرًا بما يكفي لاستيعاب أي أحداث، فسيتم إرجاع -1.
