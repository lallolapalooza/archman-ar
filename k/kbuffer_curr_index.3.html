kbuffer_curr_index(3)

الحزمة: libtraceevent

التراخيص: GPL-2.0-only, LGPL-2.1-only

LIBTRACEEVENT(3) دليل libtraceevent LIBTRACEEVENT(3)

الاسم

kbuffer_read_event, kbuffer_next_event, kbuffer_missed_events, kbuffer_event_size, kbuffer_curr_size, kbuffer_curr_offset, kbuffer_curr_index, kbuffer_read_buffer - دوال للقراءة عبر المخزن المؤقت الفرعي kbuffer.

الملخص

#include <kbuffer.h>
void *kbuffer_read_event(struct kbuffer *kbuf, unsigned long long *ts);
void *kbuffer_next_event(struct kbuffer *kbuf, unsigned long long *ts);
void *kbuffer_read_at_offset(struct kbuffer *kbuf, int offset, unsigned long long *ts);
int kbuffer_missed_events(struct kbuffer *kbuf);
int kbuffer_event_size(struct kbuffer *kbuf);
int kbuffer_curr_size(struct kbuffer *kbuf);
int kbuffer_curr_offset(struct kbuffer *kbuf);
int kbuffer_curr_index(struct kbuffer *kbuf);
int kbuffer_read_buffer(struct kbuffer *kbuf, void *buffer, int len);

الوصف

تقوم الدالة kbuffer_read_event() بقراءة الحدث التالي في واصف kbuf وإذا كان ts غير فارغ، ستضع الطابع الزمني الخاص به فيه. لا يعدل هذا واصف kbuf، واستدعاء هذه الدالة عدة مرات سيعطي نفس النتيجة.

ستقوم الدالة kbuffer_next_event() بإرجاع الحدث التالي في واصف kbuf. ستقوم أيضًا بتعيين ts إلى الطابع الزمني للحدث المُعاد. يتم إرجاع NULL إذا لم تكن هناك أحداث أخرى وسيتم تعريف ts بشكل غير صحيح. ملاحظة، إذا تم استدعاء هذه الدالة مباشرة بعد kbuffer_load_subbuffer() فقد تعطي نتيجة غير متوقعة، لأنها ستعيد الحدث الثاني وليس الحدث الأول. عادةً ما تُستخدم هذه الدالة فقط للتحرك إلى الحدث التالي ولمعرفة ما إذا كانت هناك أحداث أخرى للقراءة، ودائمًا ما يتم استدعاء kbuffer_read_event() أولاً.

تعيد الدالة kbuffer_read_at_offset() الحدث الموجود في offset معين من بداية المخزن المؤقت الفرعي. يمكن استرداد هذا الـ offset بواسطة kbuffer_curr_offset(). إذا كان ts يشير إلى unsigned long long، فسيتم تعيينه إلى الطابع الزمني للحدث الموجود في الـ offset المعطى.

إذا كان المخزن المؤقت الفرعي قد فاته أحداث قبله، فستعيد kbuffer_missed_events() قيمة غير صفرية. إذا أعادت -1، فهذا يعني أنه كانت هناك أحداث مفقودة، ولكن العدد الدقيق للأحداث المفقودة غير معروف. إذا أعادت قيمة موجبة، فإن عدد الأحداث المفقودة هو القيمة المُعادة.

تعيد الدالة kbuffer_event_size() حجم جزء البيانات للحدث الحالي (الذي سيتم إرجاعه بواسطة kbuffer_read_event()).

تعيد الدالة kbuffer_curr_size() الحجم الكلي للسجل للحدث الحالي (الذي سيتم إرجاعه بواسطة kbuffer_read_event(). الفرق هنا هو أن القيمة المُعادة تشمل حجم بيانات التعريف الخاصة بسجل الحدث التي ليست جزءًا مما تم إرجاعه بواسطة kbuffer_read_event().

تعيد الدالة kbuffer_curr_offset() الـ offset من بداية المخزن المؤقت الفرعي حيث تبدأ بيانات التعريف الخاصة بسجل الحدث الحالي. لن يكون الحدث الأول عند الـ offset صفر. يمكن استخدام هذا الـ offset لاسترداد الحدث بواسطة kbuffer_read_at_offset().

تعيد الدالة kbuffer_curr_index() المؤشر من بداية جزء البيانات في المخزن المؤقت الفرعي حيث توجد بيانات التعريف الخاصة بالحدث الحالي. من المحتمل أن يكون الحدث الأول صفرًا، ولكن قد لا يكون كذلك إذا كان هناك طابع زمني مرتبط به.

ستقوم الدالة kbuffer_read_buffer() بملء buffer المعطى من kbuf بنفس الطريقة التي يقوم بها النواة باستدعاء نظام القراءة. أي، إذا كان الطول len أقل من حجم المخزن المؤقت الفرعي، أو كان مؤشر kbuffer الحالي غير صفري، فسيبدأ النسخ من الحدث الحالي لـ kbuf وإنشاء buffer كمخزن مؤقت فرعي جديد (مع طابع زمني ورأس التزام) مع هذا الحدث الذي تم العثور عليه، بما في ذلك جميع الأحداث اللاحقة التي يمكن أن تتناسب ضمن len. يجب أن يشمل len حجم رأس المخزن المؤقت الفرعي بالإضافة إلى الأحداث المراد تضمينها. أي، len هو الحجم المخصص لـ buffer الذي يمكن ملؤه. القيمة المُعادة من هذه الدالة هي مؤشر نهاية آخر حدث تم إضافته. إذا لم تكن هناك أحداث أخرى، فسيتم إرجاع صفر، وإذا لم يتمكن المخزن المؤقت من نسخ أي أحداث لأن len صغير جدًا، فسيتم إرجاع -1.

القيمة المُعادة

تعيد kbuffer_read_event() الحدث الذي يوجد فيه الواصف kbuf حاليًا، أو NULL إذا تم تجاوز الحدث الأخير (بواسطة kbuffer_next_event()).

تعيد kbuffer_next_event() الحدث التالي بعد الحدث الحالي أو NULL إذا لم تكن هناك أحداث أخرى.

تعيد kbuffer_read_at_offset() الحدث عند offset معين من بداية المخزن المؤقت الفرعي المخزن في kbuf، أو NULL إذا لم يكن هناك حدث موجود. ملاحظة، لا يحتاج offset إلا أن يكون offset يقع على السجل، أو في بدايته. لا يحتاج أن يكون بالضبط في بداية السجل.

تعيد kbuffer_missed_events() 0 إذا لم تكن هناك أحداث مفقودة قبل تحميل المخزن المؤقت الفرعي. تعيد -1 إذا كان هناك عدد غير معروف من الأحداث المفقودة، أو إذا كان عدد الأحداث المفقودة معروفًا، فسيتم إرجاع هذا العدد.

تعيد kbuffer_event_size() حجم حمولة بيانات الحدث الحالي لـ kbuf.

تعيد kbuffer_curr_size() الحجم الإجمالي لسجل الحدث الحالي لـ kbuf. هذا يشمل حجم بيانات التعريف لهذا السجل.

تعيد kbuf_curr_offset() offset السجل الحالي من بداية المخزن المؤقت الفرعي لـ kbuf.

تعيد kbuf_curr_index() مؤشر السجل الحالي من بداية قسم بيانات kbuf.

تعيد kbuf_read_buffer() مؤشر نهاية آخر حدث تم ملؤه في buffer. إذا لم تكن هناك أحداث أخرى لنسخها من start، فسيتم إرجاع 0. إذا كان len غير كبير بما يكفي لاستيعاب أي أحداث، فسيتم إرجاع -1.
