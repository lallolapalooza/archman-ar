kbuffer_free(3)

LIBTRACEEVENT(3) دليل libtraceevent LIBTRACEEVENT(3)

الاسم

kbuffer_alloc, kbuffer_dup, kbuffer_free, kbuffer_load_subbuffer, kbuffer_subbuffer, kbuffer_refresh, kbuffer_subbuffer_size, kbuffer_start_of_data - إنشاء عنصر kbuffer لتحليل مخزن الذاكرة المؤقت للحلقة الأساسية لتتبع نواة Linux

ملخص

#include <kbuffer.h>
enum kbuffer_endian {
        KBUFFER_ENDIAN_BIG,
        KBUFFER_ENDIAN_LITTLE,
        KBUFFER_ENDIAN_SAME_AS_HOST,
};
enum kbuffer_long_size {
        KBUFFER_LSIZE_4,
        KBUFFER_LSIZE_8,
        KBUFFER_LSIZE_SAME_AS_HOST,
};
struct kbuffer;
struct tep_handle;
struct kbuffer *kbuffer_alloc(enum kbuffer_long_size size, enum kbuffer_endian endian);
struct kbuffer *kbuffer_dup(struct kbuffer *kbuf);
void kbuffer_free(struct kbuffer *kbuf);
int kbuffer_load_subbuffer(struct kbuffer *kbuf, void *subbuffer);
int kbuffer_subbuffer_size(struct kbuffer *kbuf);
int kbuffer_refresh(struct kbuffer *kbuf);
int kbuffer_start_of_data(struct kbuffer *kbuf);
void *kbuffer_subbuffer(struct kbuffer *_kbuf);

الوصف

تقوم هذه الدوال بإنشاء واجهة kbuffer يمكن استخدامها لتحليل المخازن المؤقتة الفرعية الخام لمخزن الذاكرة المؤقت للحلقة الأساسية لتتبع نواة Linux. يتم العثور على مخزن الذاكرة المؤقت للحلقة الأساسية في الدليل tracefs، ويمكن استرداده بواسطة tracefs_instance_get_file(3) عند per_cpu/cpuX/trace_pipe_raw حيث يتم استبدال X برقم وحدة المعالجة المركزية لكل مخزن ذاكرة مؤقت للحلقة الأساسية المحدد. يتم تقسيم مخزن الذاكرة المؤقت للحلقة الأساسية داخل النواة لكل وحدة معالجة مركزية، بحيث يجب استرداد مخزن الذاكرة المؤقت للحلقة الأساسية الخام لكل وحدة معالجة مركزية أيضًا.

سيقوم kbuffer_alloc() بإنشاء واصف يمكن استخدامه لإدارة مخزن مؤقت فرعي تمت قراءته بواسطة مخزن الذاكرة المؤقت للحلقة الأساسية. تحدد المعلمة size حجم الكلمة للمخزن المؤقت المعطى (ملاحظة، هذا يعمل من قراءة البيانات الخام من آلات أخرى غير الآلة التي تستدعي هذه الدالة). تحدد endian ترتيب البايتات للآلة.

إذا تم تعيين endian إلى KBUFFER_ENDIAN_SAME_AS_HOST، فسيتم تعيين ترتيب البايتات إلى نفس ترتيب البايتات المضيف، وهو مفيد عندما يقرأ التطبيق بيانات مخزن الذاكرة المؤقت للحلقة الأساسية مباشرة من نفس الجهاز الذي يعمل عليه.

إذا تم تعيين size إلى KBUFFER_LSIZE_SAME_AS_HOST، فإذا كان حجم الكلمة 8، فسيقوم بتعيين واصف kbuffer إلى حجم طويل يبلغ 8. ولكن إذا كان الحجم 4، فسيقوم بعد ذلك بإجراء استدعاء uname(2)، وإذا كان الحقل machine يحتوي على السلسلة "64" فيه، فسيتم تعيينه إلى حجم 8 بايت وليس 4 بايت. هذا لأن حجم الكلمة الطويلة في مخزن الذاكرة المؤقت للحلقة الأساسية يعتمد على النواة وليس على مساحة المستخدم.

ستقوم الدالة kbuffer_dup() بتكرار بنية kbuffer موجودة مع تخصيص بنية جديدة. سيكون لديها جميع خصائص kbuf الممرر، بما في ذلك الإشارة إلى نفس المخزن المؤقت الفرعي الذي تم تحميله في kbuf. يجب تحريرها بواسطة kbuffer_free().

ستقوم الدالة kbuffer_free() بتحرير الموارد التي تم إنشاؤها بواسطة kbuffer_alloc().

ستأخذ الدالة kbuffer_load_subbuffer() subbuffer وهو عبارة عن كتلة بيانات خام من ملف trace_pipe_raw في tracefs. يتم تقسيم مخزن الذاكرة المؤقت للحلقة الأساسية لتتبع نواة Linux إلى مخازن مؤقتة فرعية. كل مخزن مؤقت فرعي هو مقطع بيانات مستقل يحتوي على جميع المعلومات اللازمة لفصل الأحداث والطوابع الزمنية الفردية. هذا المخزن المؤقت الفرعي هو ما يستخدمه kbuffer للمشي عبر الأحداث.

تُرجع الدالة kbuffer_subbuffer_size() موقع نهاية آخر حدث في المخزن المؤقت الفرعي. لا تُرجع حجم المخزن المؤقت الفرعي نفسه.

يجب استخدام kbuffer_refresh() إذا تم إجراء المزيد من الكتابات على kbuffer المحمّل حيث يلزم تحديث حجم kbuffer لقراءة الأحداث الجديدة التي تمت كتابتها منذ استدعاء kbuffer_load_subbuffer() الأخيرة عليه.

ملاحظة، لا يتم تنفيذ أي حواجز ذاكرة باستخدام هذه الدالة، وأي تزامن مع الكاتب هو مسؤولية التطبيق.

تُرجع الدالة kbuffer_start_of_data() إزاحة بداية تحميل البيانات الفعلي للمخزن المؤقت الفرعي.

تُرجع الدالة kbuffer_subbuffer() مؤشرًا إلى المخزن المؤقت الفرعي المحمّل حاليًا. وهذا يعني، آخر مخزن مؤقت فرعي تم تحميله بواسطة kbuffer_load_subbuffer(). إذا لم يتم تحميل أي مخزن مؤقت فرعي، يتم إرجاع NULL.

قيمة الإرجاع

تُرجع kbuffer_alloc() واصف kbuffer مخصصًا أو NULL عند حدوث خطأ. يجب تحرير الواصف المُرجع باستخدام kbuffer_free()

تُرجع kbuffer_load_subbuffer() 0 عند النجاح و -1 عند حدوث خطأ.

تُرجع kbuffer_subbuffer_size() الفهرس في المخزن المؤقت الفرعي حيث يقع نهاية آخر حدث.

تُرجع kbuffer_start_of_data() إزاحة بداية البيانات في المخزن المؤقت الفرعي المحمّل في kbuf.

تُرجع kbuffer_subbuffer() آخر مخزن مؤقت فرعي تم تحميله إلى kbuf بواسطة kbuffer_load_subbuffer() أو NULL إذا لم يتم تحميل أي مخزن.

تُرجع kbuffer_refresh() 0 عند النجاح و -1 إذا كان kbuf هو NULL أو لم يتم تحميل مخزن مؤقت فرعي له عبر kbuffer_load_subbuffer().

مثال

#include <stdio.h>
#include <stdlib.h>
#include <fcntl.h>
#include <unistd.h>
#include <sys/stat.h>
#include <kbuffer.h>
int main (int argc, char **argv)
{
        unsigned long long ts;
        struct kbuffer *kbuf;
        struct stat st;
        char *buf;
        void *event;
        int ret;
        int fd;
        int i = 0;
        if (argc < 2) {
                printf("usage: %s raw-subbuffer-page\n", argv[0]);
                printf(" Try: dd count=1 bs=4096 if=/sys/kernel/tracing/per_cpu/cpu0/trace_pipe_raw of=/tmp/file\n");
                exit(0);
        }
        if (stat(argv[1], &st) < 0) {
                perror("stat");
                exit(-1);
        }
        buf = malloc(st.st_size);
        if (!buf) {
                perror("Allocating buffer");
                exit(-1);
        }
        fd = open(argv[1], O_RDONLY);
        if (fd < 0) {
                perror(argv[1]);
                exit(-1);
        }
        ret = read(fd, buf, st.st_size);
        if (ret < 0) {
                perror("Reading buffer");
                exit(-1);
        }
        close(fd);
        kbuf = kbuffer_alloc(KBUFFER_ENDIAN_SAME_AS_HOST,
                             KBUFFER_LSIZE_SAME_AS_HOST);
        if (!kbuf) {
                perror("Creating kbuffer");
                exit(-1);
        }
        ret = kbuffer_load_subbuffer(kbuf, buf);
        if (ret < 0) {
                perror("Loading sub bufer");
                exit(-1);
        }
        if (kbuffer_subbuffer_size(kbuf) > st.st_size) {
                fprintf(stderr, "kbuffer is bigger than raw size %d > %ld\n",
                        kbuffer_subbuffer_size(kbuf), st.st_size);
                exit(-1);
        }
        printf("Kbuffer data starts at %d\n", kbuffer_start_of_data(kbuf));
        do {
                event = kbuffer_read_event(kbuf, &ts);
                if (event) {
                        printf(" event %3d ts:%lld\n", i++, ts);
                        event = kbuffer_next_event(kbuf, NULL);
                }
        } while (event);
        if (!event)
                printf("Finished sub buffer\n");
        kbuffer_free(kbuf);
        return 0;
}
