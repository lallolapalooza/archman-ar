الاسم

KDGETLED - الحصول على حالة مصابيح لوحة المفاتيح.

الاستخدام

#include <sys/ioctl.h>

long arg;

int fd, rc;

rc = ioctl(fd, KDGETLED, &arg);

الوصف

تُستخدم استدعاءات النظام ioctl هذه للتحكم في لوحة المفاتيح. تُستخدم KDGETLED لقراءة حالة مصابيح لوحة المفاتيح (Num Lock، Caps Lock، Scroll Lock). يعين arg مؤشرًا إلى long الذي سيتم تعيينه على أحد القيم التالية:

PIO_UNISCRNMAP
يقوم بتحميل جدول "قابل للتعريف من قبل المستخدم" (الرابع) في النواة والذي يربط البايتات بـ Unicodes، والتي تُترجم بعد ذلك إلى رموز شاشة وفقًا لمخطط Unicode إلى الخط الحالي. يمكن استخدام Unicodes الخاصة بدءًا من U+F000 للربط مباشرة برموز الخط. (منذ Linux 1.3.1.)
GIO_UNIMAP
الحصول على تعيين Unicode إلى خط من النواة. يشير argp إلى

struct unimapdesc {
    unsigned short  entry_ct;
    struct unipair *entries;
};

حيث يشير entries إلى مصفوفة من

struct unipair {
    unsigned short unicode;
    unsigned short fontpos;
};

(منذ Linux 1.1.92.)
PIO_UNIMAP
وضع تعيين Unicode إلى خط في النواة. يشير argp إلى struct unimapdesc. (منذ Linux 1.1.92)
PIO_UNIMAPCLR
مسح الجدول، ربما نصح خوارزمية التجزئة. يشير argp إلى

struct unimapinit {
    unsigned short advised_hashsize;  /* 0 if no opinion */
    unsigned short advised_hashstep;  /* 0 if no opinion */
    unsigned short advised_hashlevel; /* 0 if no opinion */
};

(منذ Linux 1.1.92.)
KDGKBMODE
الحصول على وضع لوحة المفاتيح الحالي. يشير argp إلى long يتم تعيينه على أحد هذه القيم:
K_RAW 0x00 /* وضع خام (scancode) */
K_XLATE 0x01 /* ترجمة مفاتيح الأكواد باستخدام خرائط المفاتيح */
K_MEDIUMRAW 0x02 /* وضع متوسط خام (scancode) */
K_UNICODE 0x03 /* وضع Unicode */
K_OFF 0x04 /* وضع معطل (منذ Linux 2.6.39) */
KDSKBMODE
تعيين وضع لوحة المفاتيح الحالي. argp هو long يساوي أحد القيم الموضحة لـ KDGKBMODE.
KDGKBMETA
الحصول على وضع معالجة مفتاح meta. يشير argp إلى long يتم تعيينه على أحد هذه القيم:
K_METABIT 0x03 تعيين البت الأعلى
K_ESCPREFIX 0x04 بادئة الهروب
KDSKBMETA
تعيين وضع معالجة مفتاح meta. argp هو long يساوي أحد القيم الموضحة أعلاه لـ KDGKBMETA.
KDGKBENT
الحصول على إدخال واحد في جدول ترجمة المفاتيح (رمز المفتاح إلى رمز الإجراء). يشير argp إلى

struct kbentry {
    unsigned char  kb_table;
    unsigned char  kb_index;
    unsigned short kb_value;
};

مع ملء العضوين الأولين: kb_table يحدد جدول المفاتيح (0 <= kb_table < MAX_NR_KEYMAPS)، و kb_index هو رمز المفتاح (0 <= kb_index < NR_KEYS). يتم تعيين kb_value إلى رمز الإجراء المقابل، أو K_HOLE إذا لم يكن هناك مفتاح كهذا، أو K_NOSUCHMAP إذا كان kb_table غير صالح.
KDSKBENT
تعيين إدخال واحد في جدول الترجمة. يشير argp إلى struct kbentry.
KDGKBSENT
الحصول على سلسلة مفتاح وظيفة واحدة. يشير argp إلى

struct kbsentry {
    unsigned char kb_func;
    unsigned char kb_string[512];
};

يتم تعيين kb_string إلى السلسلة (المنتهية بصفر) المقابلة لرمز الإجراء المفتاح الوظيفي ذي الترتيب kb_func.
KDSKBSENT
تعيين إدخال سلسلة مفتاح وظيفة واحد. يشير argp إلى struct kbsentry.
KDGKBDIACR
قراءة جدول علامات التشكيل (diacritic) في النواة. يشير argp إلى

struct kbdiacrs {
    unsigned int   kb_cnt;
    struct kbdiacr kbdiacr[256];
};

حيث kb_cnt هو عدد الإدخالات في المصفوفة، كل منها عبارة عن

struct kbdiacr {
    unsigned char diacr;
    unsigned char base;
    unsigned char result;
};

KDGETKEYCODE
قراءة إدخال جدول رمز المفتاح في النواة (رمز المسح إلى رمز المفتاح). يشير argp إلى

struct kbkeycode {
    unsigned int scancode;
    unsigned int keycode;
};

يتم تعيين keycode ليتوافق مع scancode المحدد. (89 <= scancode <= 255 فقط. لـ 1 <= scancode <= 88، keycode==scancode.) (منذ Linux 1.1.63.)
KDSETKEYCODE
كتابة إدخال جدول رمز المفتاح في النواة. يشير argp إلى struct kbkeycode. (منذ Linux 1.1.63.)
KDSIGACCEPT
يشير البرنامج الذي يستدعي العملية إلى استعداده لقبول الإشارة argp عند إنشائها عن طريق الضغط على تركيبة مفاتيح مناسبة. (1 <= argp <= NSIG). (انظر spawn_console() في linux/drivers/char/keyboard.c.)

قيمة الإرجاع

عند النجاح، يتم إرجاع 0 (باستثناء ما هو مشار إليه). عند الفشل، يتم إرجاع -1، ويتم تعيين errno للإشارة إلى الخطأ.

الأخطاء

EINVAL
argp غير صالح.

المعايير

Linux.

انظر أيضًا
