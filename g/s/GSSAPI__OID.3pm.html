GSSAPI__OID(3pm)

الحزمة: perl-gssapi

التراخيص: GPL, PerlArtistic

GSSAPI::OID(3) وثائق بيرل مقدمة من المستخدم GSSAPI::OID(3)

الاسم

GSSAPI::OID - طرق لمعالجة GSSAPI OIDs وبعض الثوابت

ملخص

use GSSAPI;
#$oid = GSSAPI::OID->new;             # نادرًا ما تكون مطلوبة أو مرغوبة
$status = GSSAPI::OID->from_str($oid, "{ 1 2 840 113554 1 2 1 1 }");
#
# مدعومة فقط على MIT Kerberos
#
$status = $oid->to_str($str);
#
# مدعومة فقط على MIT Kerberos
#
$status = $oid->inquire_names($oidset);
# ثوابت OID مقدمة:
$oid = gss_nt_user_name;
$oid = gss_nt_machine_uid_name;
$oid = gss_nt_string_uid_name;
$oid = gss_nt_service_name;
$oid = gss_nt_exported_name;
$oid = gss_nt_service_name_v2;
$oid = gss_nt_krb5_name;
$oid = gss_nt_krb5_principal;
$oid = gss_mech_krb5;
$oid = gss_mech_krb5_old;
$oid = gss_mech_krb5_v2;
$oid = gss_mech_spnego;
# إذا كان تنفيذ GSSAPI الخاص بك يدعم
# SPNEGO (Heimdal 0.7 على سبيل المثال
# يمكنك استخدام mechtype OID::gss_mech_spnego.
#
# use GSSAPI::indicate_mechs( $oidset );
# للحصول على أنواع الآليات التي يدعمها تنفيذك
$oid = gss_nt_hostbased_service; # GSS_C_NT_HOSTBASED_SERVICE

الوصف

تُستخدم كائنات "GSSAPI::OID" كمعرفات/ثوابت فريدة لـ "الآليات" -- البروتوكول والإصدار المحددان المستخدمان -- وللتشفيرات المستخدمة لتمثيل الأسماء. في العديد من الحالات، يمكنك طلب الآلية أو التشفير الافتراضي للتنفيذ باستخدام GSS_C_NO_OID. تحقق من وصف الروتين في RFC2743 إذا لم تكن متأكدًا.
