الحزمة: libtecla

gl_normal_io(gl)) /* Temporarily suspend editing */ return GLA_ABORT; printf("Hello world\n"); return GLA_CONTINUE; }

يجب أن تُرجع دوال الإجراء إحدى القيم التالية، لإخبار gl_get_line() بكيفية المتابعة.

  GLA_ABORT     -   تتسبب في أن تُرجع gl_get_line() القيمة NULL.
  GLA_RETURN    -   تتسبب في أن تُرجع gl_get_line() سطر الإدخال
                    المكتمل.
  GLA_CONTINUE  -   استئناف تحرير سطر الأوامر.

لاحظ أن الوسيط name الخاص بـ gl_register_action() يحدد الاسم الذي يمكن للمستخدم الإشارة إليه للإجراء في ملف التكوين الخاص به. هذا يسمح لهم بإعادة ربط الإجراء بتسلسل مفاتيح بديل. ولكي يعمل هذا، من الضروري استدعاء gl_register_action() بين استدعاء new_GetLine() وأول استدعاء لـ gl_get_line().

ملفات السجل

لحفظ محتويات مخزن السجل مؤقتًا قبل الخروج من تطبيقك، ومن ثم استعادتها عند بدء تشغيل التطبيق مرة أخرى، يتم توفير الدوال التالية.

 int gl_save_history(GetLine *gl, const char *filename,
                     const char *comment, int max_lines);
 int gl_load_history(GetLine *gl, const char *filename,
                     const char *comment);

يحدد الوسيط filename الاسم الذي سيتم إعطاؤه لملف السجل عند الحفظ، أو اسم ملف سجل موجود، عند التحميل. قد يحتوي هذا على تعبيرات دليل المنزل ومتغيرات البيئة، مثل "~/.myapp_history" أو "$HOME/.myapp_history".

بالإضافة إلى كل سطر في السجل، يتم تسجيل معلومات إضافية عنه، مثل متى تم إدخاله بواسطة المستخدم، وما هو مستوى التداخل الخاص به، كتعليق يسبق السطر في ملف السجل. كتابة هذا كتعليق تسمح لملف السجل بأن يكون بمثابة ملف أوامر، فقط في حالة رغبتك في إعادة تشغيل جلسة كاملة باستخدامه. نظرًا لأن بادئات التعليقات تختلف في لغات مختلفة، يتم توفير الوسيط comment لتحديد بادئة التعليق. على سبيل المثال، إذا كان تطبيقك عبارة عن shell من نوع unix، مثل bourne shell، فسوف تحدد "#" هنا. مهما اخترت لحرف التعليق، يجب عليك تحديد نفس البادئة لـ gl_load_history() التي استخدمتها عند استدعاء gl_save_history() لكتابة ملف السجل.

يجب أن يكون max_lines إما -1 لتحديد أنه يجب حفظ جميع الأسطر في قائمة السجل، أو رقم موجب يحدد سقفًا لعدد الأسطر الأحدث التي يجب حفظها.

كلتا الدالتين تُرجعان قيمة غير صفرية عند حدوث خطأ، بعد كتابة رسالة خطأ إلى stderr. لاحظ أن gl_load_history() لا تعتبر عدم وجود ملف خطأ.

قوائم سجل متعددة

إذا كان تطبيقك يستخدم كائن GetLine واحدًا لإدخال أنواع مختلفة من أسطر الإدخال، فقد ترغب في أن تقوم gl_get_line() بالتمييز بين الأنواع المختلفة من الأسطر في قائمة السجل، واستدعاء الأسطر التي تطابق النوع الحالي من السطر فقط. لدعم هذا المتطلب، تقوم gl_get_line() بتمييز الأسطر التي يتم تسجيلها في قائمة السجل بمعرف عددي اختارته التطبيق. في البداية، يتم تعيين هذا المعرف إلى 0 بواسطة new_GetLine()، ولكن يمكن تغييره لاحقًا عن طريق استدعاء gl_group_history().

  int gl_group_history(GetLine *gl, unsigned id);

يمكن أن يكون المعرف العددي id أي رقم يختاره التطبيق، ولكن لاحظ أن gl_save_history() و gl_load_history() تحافظان على الارتباط بين المعرفات و أسطر الإدخال التاريخية بين استدعاءات البرنامج، لذلك يجب عليك اختيار معرفات ثابتة لأنواع مختلفة من أسطر الإدخال التي يستخدمها تطبيقك.

كلما قامت gl_get_line() بإلحاق سطر إدخال جديد بقائمة السجل، يتم تسجيل المعرف الحالي للسجل معه، وعندما يُطلب منها استدعاء سطر إدخال تاريخي، فإنها تستدعي فقط الأسطر المميزة بالمعرف الحالي.

عرض السجل

يمكن عرض قائمة السجل عن طريق استدعاء gl_show_history().

  int gl_show_history(GetLine *gl, FILE *fp,
                      const char *fmt,
                      int all_groups,
                      int max_lines);

يقوم هذا بعرض المحتويات الحالية لقائمة السجل إلى تيار الإخراج القياسي fp. إذا كان الوسيط max_lines أكبر من أو يساوي صفرًا، فلن يتم عرض أكثر من عدد الأسطر الأحدث. إذا كان الوسيط all_groups غير صفري، يتم عرض الأسطر من جميع مجموعات السجل. بخلاف ذلك، يتم عرض تلك الخاصة بمجموعة السجل المحددة حاليًا فقط. تحدد سلسلة التنسيق الوسيطة fmt كيفية عرض السطر. يمكن أن تحتوي هذه على أحرف عشوائية يتم كتابتها حرفيًا، متداخلة مع أي من توجيهات التنسيق التالية:

  %D  -  التاريخ الذي تم فيه إدخال السطر
         أصلاً، بتنسيق مثل 2001-11-20.
  %T  -  الوقت من اليوم عند إدخال السطر،
         بتنسيق مثل 23:59:59.
  %N  -  رقم الإدخال التسلسلي للسطر في
         مخزن السجل مؤقتًا.
  %G  -  رقم مجموعة السجل التي ينتمي إليها
         السطر.
  %%  -  حرف % حرفي.
  %H  -  سطر السجل نفسه.

وهكذا، سلسلة تنسيق مثل "\%D \%T \%H\n" ستخرج شيئًا مثل:

  2001-11-20 10:23:34  Hello world

لاحظ تضمين حرف سطر جديد صريح في سلسلة التنسيق.

البحث في السجل

تسمح دالة gl_lookup_history() للتطبيق المستدعي بالبحث في الأسطر في قائمة السجل.

  typedef struct {
    const char *line;    /* السطر التاريخي المطلوب */
                         /*  . */
    unsigned group;      /* مجموعة السجل التي ينتمي إليها */
                         /*  السطر. */
    time_t timestamp;    /* التاريخ والوقت الذي تم فيه */
                         /*  إدخال السطر أصلاً. */
  } GlHistoryLine;
  int gl_lookup_history(GetLine *gl, unsigned long id,
                        GlHistoryLine *hline);

يشير الوسيط id إلى السطر المطلوب البحث عنه، حيث يتم تمثيل السطر الأول الذي تم إدخاله في قائمة السجل بعد استدعاء new_GetLine() بالرقم 0، ويتم تمثيل الأسطر المدخلة لاحقًا بأرقام متزايدة. لاحظ أنه يمكن الاستعلام عن نطاق الأسطر المحفوظة حاليًا في قائمة السجل عن طريق استدعاء الدالة gl_range_of_history()، الموصوفة لاحقًا. إذا كان السطر المطلوب موجودًا في قائمة السجل، يتم تسجيل تفاصيل السطر في المتغير المشار إليه بواسطة الوسيط hline، ويتم إرجاع 1. بخلاف ذلك، يتم إرجاع 0، ويظل المتغير المشار إليه بواسطة hline دون تغيير.

كن حذرًا لأن السلسلة النصية التي يتم إرجاعها في hline->line هي جزء من مخزن السجل مؤقتًا، لذلك لا يجب تعديلها بواسطة المستدعي، وسيتم إعادة استخدامها عند الاستدعاء التالي لأي دالة تأخذ gl كوسيط. لذلك، يجب عليك إنشاء نسخة خاصة من هذه السلسلة إذا كنت بحاجة إلى الاحتفاظ بها.

أرشفة السجل يدويًا

بشكل افتراضي، كلما أدخل المستخدم سطرًا، يتم إضافته تلقائيًا إلى قائمة السجل، قبل أن تقوم gl_get_line() بإرجاع السطر إلى المستدعي. هذا مريح لغالبية التطبيقات، ولكن هناك أيضًا تطبيقات تحتاج إلى تحكم أدق في ما يتم إضافته إلى قائمة السجل. في مثل هذه الحالات، يمكن إيقاف الإضافة التلقائية للأسطر المدخلة إلى قائمة السجل عن طريق استدعاء الدالة gl_automatic_history().

  int gl_automatic_history(GetLine *gl, int enable);

إذا تم استدعاء هذه الدالة مع تعيين الوسيط enable إلى 0، فلن تقوم gl_get_line() بأرشفة الأسطر المدخلة لاحقًا تلقائيًا. يمكن إعادة تمكين الأرشفة التلقائية للسجل في وقت لاحق، عن طريق استدعاء هذه الدالة مرة أخرى، مع تعيين الوسيط enable إلى 1. بينما تكون أرشفة السجل التلقائية معطلة، يمكن للتطبيق المستدعي استخدام gl_append_history() لإضافة الأسطر إلى قائمة السجل حسب الحاجة.

  int gl_append_history(GetLine *gl, const char *line);

يحدد الوسيط line السطر المراد إضافته إلى قائمة السجل. يجب أن يكون هذا سلسلة نصية عادية تنتهي بـ ' '. إذا كانت هذه السلسلة النصية تحتوي على أي أحرف سطر جديد، فإن السطر الذي سيتم أرشفته في قائمة السجل سيتم إنهاؤه بأول حرف من هذه الأحرف. بخلاف ذلك، سيتم إنهاؤه بالفاصل ' '. إذا كان السطر أطول من أقصى طول لسطر الإدخال، الذي تم تحديده عند استدعاء new_GetLine()، فعند استدعاء السطر، سيتم اقتطاعه إلى الطول الفعلي لسطر gl_get_line().

إذا نجحت، تُرجع gl_append_history() القيمة 0. بخلاف ذلك، تُرجع قيمة غير صفرية، وتعيّن errno إلى إحدى القيم التالية.

   EINVAL  -  كان أحد الوسائط التي تم تمريرها إلى
              gl_append_history() هو NULL.
   ENOMEM  -  كان السطر المحدد أطول من الحجم المخصص
              لمخزن السجل مؤقتًا (كما هو محدد عند
              استدعاء new_GetLine())، لذلك لم يكن من
              الممكن أرشفته.

يمكن الحصول على وصف نصي للخطأ اختياريًا عن طريق استدعاء gl_error_message(). لاحظ أنه بعد حدوث مثل هذا الخطأ، تظل قائمة السجل في حالة صالحة لتلقي أسطر سجل جديدة، لذلك لا يوجد ضرر كبير في تجاهل حالة الإرجاع الخاصة بـ gl_append_history().

تكوين السجل متنوع

إذا كنت ترغب في تغيير حجم مخزن السجل مؤقتًا الذي تم تحديده أصلاً في استدعاء new_GetLine()، يمكنك القيام بذلك باستخدام الدالة gl_resize_history().

  int gl_resize_history(GetLine *gl, size_t histlen);

يحدد الوسيط histlen الحجم الجديد بالبايت، وإذا حددت هذا القيمة بـ 0، فسيتم حذف المخزن المؤقت.

كما ذكر في مناقشة new_GetLine()، يعتمد عدد الأسطر التي يمكن تخزينها في مخزن السجل مؤقتًا على أطوال الأسطر الفردية. على سبيل المثال، يمكن لمخزن بحجم 1000 بايت تخزين 10 أسطر بطول متوسط 100 بايت، أو سطرين بطول متوسط 50 بايت. على الرغم من أن المخزن المؤقت لا يتوسع أبدًا عند إضافة أسطر جديدة، إلا أن قائمة بالمؤشرات داخل المخزن المؤقت تتوسع عند الحاجة لاستيعاب عدد الأسطر المخزنة حاليًا في المخزن. لوضع حد أقصى لعدد الأسطر في المخزن، وبالتالي سقف على مقدار الذاكرة المستخدمة في هذه القائمة، يمكنك استدعاء gl_limit_

يستجيب لإشارة يقوم باعتراضها بالفعل.

  int gl_trap_signal(GetLine *gl, int signo, unsigned flags,
                     GlAfterSignal after, int errno_value);

الوسيط signo هو رقم الإشارة التي ترغب في اعتراضها. الوسيط flags هو مجموعة من العلامات التي تحدد البيئة التي يتم فيها استدعاء معالج إشارات التطبيق، الوسيط after يخبر gl_get_line() بما يجب فعله بعد عودة معالج إشارات التطبيق، و errno_value يخبر gl_get_line() بما يجب تعيينه لـ errno إذا طُلب منه الإجهاض.

الوسيط flags هو OR بت الصفري أو أكثر من المُعدِّدات التالية:

  GLS_RESTORE_SIG  -  استعادة بيئة الإشارة للمتصل
                      أثناء معالجة الإشارة.
  GLS_RESTORE_TTY  -  استعادة إعدادات الطرفية للمتصل
                      أثناء معالجة الإشارة.
  GLS_RESTORE_LINE -  نقل المؤشر إلى بداية السطر
                      بعد سطر الإدخال قبل استدعاء
                      معالج إشارات التطبيق.
  GLS_REDRAW_LINE  -  إعادة رسم سطر الإدخال عند
                      عودة معالج إشارات التطبيق.
  GLS_UNBLOCK_SIG  -  بشكل طبيعي، إذا كان البرنامج المتصل
                      يحظر إشارة (man sigprocmask)،
                      فإن gl_get_line() لا يعترض تلك
                      الإشارة. تخبر هذه العلامة
                      gl_get_line() باعتراض الإشارة
                      وإلغاء حظرها لمدة استدعاء
                      gl_get_line().
  GLS_DONT_FORWARD -  إذا تم تضمين هذه العلامة، فلن يتم
                      توجيه الإشارة إلى معالج إشارات
                      البرنامج المتصل.

تم أيضًا تحديد مجموعتين مفيدتين بشكل شائع من العلامات على النحو التالي:

  GLS_RESTORE_ENV   = GLS_RESTORE_SIG | GLS_RESTORE_TTY |
                      GLS_REDRAW_LINE
  GLS_SUSPEND_INPUT = GLS_RESTORE_ENV | GLS_RESTORE_LINE

إذا كان معالج إشاراتك، أو معالج الإشارات الافتراضي للنظام لهذه الإشارة، إذا لم تكن قد تجاوزته، لم يكتب أبدًا إلى الطرفية، أو لم يوقف أو ينهي البرنامج المتصل، فيمكنك تعيين الوسيط flags بأمان إلى 0.

إذا كان معالج إشاراتك يكتب دائمًا إلى الطرفية، أو يقرأ منها، أو يوقف البرنامج أو ينهيه، فيجب عليك تحديد الوسيط flags كـ GL_SUSPEND_INPUT، بحيث:

1. لا يترك المؤشر في منتصف سطر الإدخال.
2. حتى يتمكن المستخدم من كتابة الإدخال وتمريره.
3. حتى لا تحتاج إلى إنهاء كل سطر إخراج بـ
   \r\n، بدلاً من مجرد \n.

مجموعة GL_RESTORE_ENV هي نفسها مثل GL_SUSPEND_INPUT، باستثناء أنها لا تحرك المؤشر، وإذا لم يقرأ معالج إشاراتك أي شيء أو يكتبه إلى الطرفية، فلن يرى المستخدم أي إشارة مرئية على أن إشارة قد تم التقاطها. قد يكون هذا مفيدًا إذا كان لديك معالج إشارات يكتب إلى الطرفية بشكل غير منتظم، حيث قد يتسبب استخدام GL_SUSPEND_LINE في تكرار سطر الإدخال بشكل غير ضروري عندما لم يتم كتابة شيء إلى الطرفية. يجب أن يتأكد مثل هذا المعالج، عندما يكتب إلى الطرفية، من بدء سطر جديد في بداية كتابته الأولى، عن طريق كتابة سطر جديد قبل العودة. إذا وصلت الإشارة أثناء قيام المستخدم بإدخال سطر يشغل سطر طرفية واحد فقط، أو إذا كان المؤشر على آخر سطر طرفي لسطر إدخال أطول، فسيكون لهذا تأثير مماثل لـ GL_SUSPEND_INPUT. بخلاف ذلك، سيبدأ الكتابة في سطر يحتوي بالفعل على جزء من سطر الإدخال المعروض. هذا لا يسبب أي ضرر، ولكنه يبدو قبيحًا بعض الشيء، وهذا هو السبب في أن مجموعة GL_SUSPEND_INPUT أفضل إذا كنت تعلم أنك ستكتب دائمًا إلى الطرفية.

الوسيط after، الذي يحدد ما يفعله gl_get_line() بعد عودة معالج إشارات التطبيق (إذا عاد)، يمكن أن يأخذ أي قيمة من القيم التالية:

  GLS_RETURN   - إرجاع سطر الإدخال المكتمل، تمامًا كما
                 لو أن المستخدم قد ضغط على مفتاح الإرجاع.
  GLS_ABORT    - تسبب في إجهاض gl_get_line(). عندما
                 يحدث ذلك، تُرجع
                 gl_get_line()
                 NULL، وستُرجع مكالمة لاحقة لـ
                 gl_return_status()
                 GLR_SIGNAL. لاحظ أنه إذا كان
                 التطبيق يحتاج إلى أن يكون
                 errno دائمًا ذا قيمة ذات معنى عندما
                 تُرجع gl_get_line()
                 NULL، فيجب أن تقوم دالة
                 الاستدعاء الخلفي بتعيين
                 errno بشكل مناسب.
  GLS_CONTINUE - استئناف تحرير سطر الأوامر.

الوسيط errno_value مخصص للدمج مع الخيار GLS_ABORT، لإخبار gl_get_line() بما يجب تعيينه للمتغير errno القياسي قبل إرجاع NULL إلى البرنامج المتصل. ومع ذلك، يمكن استخدامه أيضًا مع الخيار GL_RETURN، في حال كنت ترغب في طريقة للتمييز بين سطر إدخال تم إدخاله باستخدام مفتاح الإرجاع، وسطر تم إدخاله عن طريق استقبال إشارة.

معالجة إشارات موثوقة

معالجة الإشارات صعبة بشكل مدهش للقيام بها بشكل موثوق دون حالات سباق. في gl_get_line()، تم اتخاذ الكثير من العناية للسماح للتطبيقات بإجراء معالجة إشارات موثوقة حول gl_get_line(). هذا القسم يشرح كيفية الاستفادة من ذلك.

كمثال للمشاكل التي يمكن أن تنشأ إذا لم تتم كتابة التطبيق بشكل صحيح، تخيل أن تطبيقًا لديه معالج إشارة SIGINT يقوم بتعيين علامة عامة. الآن، افترض أن التطبيق يختبر هذه العلامة قبل استدعاء gl_get_line(). إذا حدث أن تم استقبال إشارة SIGINT في فترة زمنية صغيرة بين العبارة التي تختبر قيمة هذه العلامة، والعبارة التي تستدعي gl_get_line()، فلن يرى gl_get_line() الإشارة، ولن تتم مقاطعته. ونتيجة لذلك، لن يتمكن التطبيق من الاستجابة للإشارة حتى يقوم المستخدم بإنهاء إدخال سطر الإدخال وتعود gl_get_line(). اعتمادًا على التطبيق، قد يكون هذا كارثة أو لا، ولكن على الأقل سيحير المستخدم.

الطريقة لتجنب مثل هذه المشاكل هي القيام بما يلي.

1. إذا لزم الأمر، استخدم الدالة gl_trap_signal() لـ
تكوين gl_get_line() للإجهاض عند التقاط
إشارات مهمة.

2. تكوين gl_get_line() بحيث إذا تم حظر أي من
الإشارات التي يلتقطها عند استدعاء
gl_get_line()، فسيتم إلغاء حظرها
تلقائيًا خلال الأوقات التي ينتظر فيها
gl_get_line() إدخال/إخراج. يمكن القيام بذلك
إما على أساس كل إشارة، عن طريق استدعاء
الدالة gl_trap_signal()، وتحديد
السمة GLS_UNBLOCK للإشارة، أو عالميًا عن طريق
استدعاء الدالة gl_catch_blocked().

     void gl_catch_blocked(GetLine *gl);


تقوم هذه الدالة ببساطة بإضافة السمة
GLS_UNBLOCK إلى جميع الإشارات التي تم
تكوينها حاليًا لاعتراضها.

3. قبل استدعاء gl_get_line() مباشرة، قم بحظر تسليم
جميع الإشارات التي تم تكوين gl_get_line() لاعتراضها. يمكن القيام
بذلك باستخدام دالة POSIX sigprocmask() بالاقتران مع الدالة
gl_list_signals().

      int gl_list_signals(GetLine *gl, sigset_t *set);


تُرجع هذه الدالة مجموعة الإشارات التي تم
تكوينها حاليًا لاعتراضها في الوسيط set، والذي يكون بالشكل المطلوب بواسطة sigprocmask().

4. في المثال، سيختبر المرء الآن العلامة العامة التي
يقوم معالج الإشارة بتعيينها، مع العلم أنه لا يوجد
خطر الآن من تعيين هذه العلامة مرة أخرى حتى يقوم
gl_get_line() بإلغاء حظر إشاراته أثناء
تنفيذ عمليات الإدخال/الإخراج.

5. في النهاية، تعود gl_get_line()، إما لأن إشارة تم
التقاطها، أو حدث خطأ، أو أكمل المستخدم سطر
إدخاله.

6. الآن سيتحقق المرء من العلامة العامة للإشارة مرة
أخرى، وإذا تم تعيينها، فسيستجيب لها، ويسفر
العلم.

7. استخدم sigprocmask() لإلغاء حظر الإشارات التي تم
حظرها في الخطوة 3.

يمكن استخدام نفس التقنية حول دوال POSIX التي تتفهم الإشارات، مثل sigsetjmp() و sigsuspend()، وبشكل خاص، يمكن استخدام الدالة الأولى من هاتين الدالتين بالاقتران مع siglongjmp() لتنفيذ معالجة إشارات خالية من حالات السباق حول مكالمات نظام أخرى طويلة الأمد. الطريقة للقيام بذلك، سيتم شرحها بعد ذلك، من خلال إظهار كيف يقوم gl_get_line() باعتراض الإشارات بشكل موثوق حول مكالمات الدوال مثل read() و select() دون حالات سباق.

أول شيء يفعله gl_get_line()، كلما تم استدعاؤه، هو استخدام دالة POSIX sigprocmask() لحظر تسليم جميع الإشارات التي تم تكوينها حاليًا لاعتمادها. هذا زائد عن الحاجة إذا كان التطبيق قد قام بحظرها بالفعل، ولكنه لا يسبب ضررًا. يقوم بإلغاء هذه الخطوة قبل العودة مباشرة.

كلما احتاج gl_get_line() إلى استدعاء read() أو select() للانتظار للحصول على إدخال من المستخدم، فإنه يستدعي أولاً دالة POSIX sigsetjmp()، مع التأكد من تحديد قيمة غير صفرية للحجة savesigs. سيصبح سبب الحجة الأخيرة واضحًا بعد قليل.

إذا أعادت sigsetjmp() صفرًا، فسيقوم gl_get_line() بعد ذلك بما يلي.

a. يستخدم دالة POSIX
   sigaction() لتسجيل معالج إشارة مؤقت
   لجميع الإشارات التي تم تكوينها لاعتمادها. يقوم معالج
   الإشارة هذا بشيئين.
   1. يسجل رقم الإشارة التي تم استقبالها
      في متغير على مستوى الملف.
   2. ثم يستدعي دالة POSIX
      siglongjmp() باستخدام المخزن المؤقت الذي تم
      تمريره إلى
      sigsetjmp() كوسيط أول، وقيمة غير
      صفرية كوسيط ثانٍ.
   عند تسجيل معالج الإشارة هذا، يتم تكوين العضو
   sa_mask في الوسيط
   struct sigaction act لمكالمة
    sigaction() ليحتوي على جميع الإشارات التي
    يلتقطها
    gl_get_line(). هذا يضمن التقاط إشارة واحدة فقط في
   كل مرة بواسطة معالج الإشارة لدينا، وهذا بدوره يضمن أن
   مثيلات متعددة لمعالج الإشارة لدينا لا تتعدى على بعضها
   البعض.
b. الآن بعد أن تم إعداد معالج الإشارة،
   gl_get_line() يلغي حظر جميع الإشارات التي تم
   تكوينها لاعتمادها.
c. ثم يستدعي استدعاءات النظام
    read() أو select() للانتظار للحصول على
   إدخال لوحة المفاتيح.
d. إذا عادت استدعاءات النظام هذه (أي، لم يتم استقبال
   إشارة)، فإن
    gl_get_line() يحظر تسليم الإشارات موضع
   الاهتمام مرة أخرى.
e. ثم يعيد تثبيت معالجات الإشارات التي تم إزاحتها بواسطة
   المعالج الذي تم تثبيته للتو.

بدلاً من ذلك، إذا أعادت sigsetjmp() قيمة غير صفرية، فهذا يعني

> لن تكون كافية لتحديد ضغطات المفاتيح هذه.

مسح الشاشة الطرفية

يمكن للبرنامج المتصل مسح الشاشة الطرفية عن طريق استدعاء gl_erase_terminal(). في وضع خادم الإدخال/الإخراج غير الحاجب، تقوم هذه الدالة أيضًا بترتيب إعادة رسم سطر الإدخال الحالي من البداية عند استدعاء gl_get_line() في المرة التالية.

  int gl_erase_terminal(GetLine *gl);

عرض النص ديناميكيًا

بين استدعاءات gl_get_line()، توفر الدالة gl_display_text() طريقة ملائمة لعرض فقرات نصية، محاذية لليسار ومقسمة على سطر طرفي واحد أو أكثر وفقًا لقيود العرض الحالي للشاشة الطرفية. يمكن العثور على أمثلة لاستخدام هذه الدالة في برامج العرض التوضيحية، حيث يتم استخدامها لعرض مقدمات. في تلك الأمثلة، يتم أيضًا توضيح الاستخدام المتقدم للمقدمات اللاحقة الاختيارية، ولواحق، والأسطر المملوءة لرسم مربع حول النص.

  int gl_display_text(GetLine *gl, int indentation,
                      const char *prefix,
                      const char *suffix, int fill_char,
                      int def_width, int start,
                      const char *string);

إذا لم يكن gl متصلاً حاليًا بطرفية، على سبيل المثال إذا كان إخراج برنامج يستخدم gl_get_line() يتم توجيهه إلى برنامج آخر أو إعادة توجيهه إلى ملف، فحينئذٍ يتم استخدام قيمة المعلمة def_width كعرض الطرفية.

يحدد الوسيط indentation عدد الأحرف المستخدمة لإزاحة كل سطر إخراج. يحدد الوسيط fill_char الحرف الذي سيتم استخدامه لإجراء هذه الإزاحة.

يمكن أن يكون الوسيط prefix إما NULL، أو سلسلة نصية توضع في بداية كل سطر جديد (بعد أي إزاحة). وبالمثل، يمكن أن يكون الوسيط suffix إما NULL، أو سلسلة نصية توضع في نهاية كل سطر. يتم وضع اللاحقة بمحاذاة تامة مع الحافة اليمنى للطرفية، ويتم ملء أي مسافة بين حرفها الأول والكلمة الأخيرة في هذا السطر بالحرف المحدد عبر الوسيط fill_char. عادةً ما يكون حرف التعبئة مسافة.

يخبر الوسيط start الدالة gl_display_text() عدد الأحرف التي تم كتابتها بالفعل إلى سطر الطرفية الحالي، وبالتالي يحدد لها فهرس عمود البداية للمؤشر. نظرًا لأن القيمة المرجعة للدالة gl_display_text() هي فهرس عمود النهاية للمؤشر، عن طريق تمرير القيمة المرجعة لاستدعاء واحد إلى الوسيط start للاستدعاء التالي، يمكن تكوين فقرة مقسمة بين أكثر من سلسلة نصية واحدة عن طريق استدعاء gl_display_text() لكل جزء متتالي من الفقرة. لاحظ أن أحرف سطر جديد حرفية ضرورية في نهاية كل فقرة لبدء سطر جديد.

عند حدوث خطأ، تعيد الدالة gl_display_text() القيمة -1.

مرافق دوال الاستدعاء

ما لم يُذكر خلاف ذلك، لا ينبغي لدوال الاستدعاء، مثل دوال إكمال علامة الجدولة ودوال الأحداث، استدعاء أي دوال في هذه الوحدة. ومع ذلك، تم تصميم الدوال التالية خصيصًا لاستخدامها من قبل دوال الاستدعاء.

يخبر استدعاء الدالة gl_replace_prompt() من دالة استدعاء الدالة gl_get_line() بعرض مطالبة مختلفة عندما تعود دالة الاستدعاء. باستثناء وضع خادم غير الحاجب، ليس لها تأثير إذا تم استخدامها بين استدعاءات gl_get_line(). في وضع خادم غير الحاجب (انظر صفحة الدليل gl_io_mode(3)، عند استخدامها بين استدعاءين لـ gl_get_line() يعملان على نفس سطر الإدخال، سيتم إعادة رسم سطر الإدخال الحالي بالمطالبة الجديدة في الاستدعاء التالي لـ gl_get_line().

  void gl_replace_prompt(GetLine *gl, const char *prompt);

مجموعات الأحرف الدولية

منذ إصدار libtecla 1.4.0، أصبحت gl_get_line() نظيفة بـ 8 بت. هذا يعني أن جميع الأحرف ذات 8 بت القابلة للطباعة في الإعدادات المحلية الحالية للمستخدم يتم الآن عرضها حرفياً وتضمينها في سطر الإدخال المُرجع. بافتراض أن البرنامج المتصل يحتوي بشكل صحيح على استدعاء مثل التالي،

  setlocale(LC_CTYPE, "");

فحينئذٍ يتم تحديد الإعدادات المحلية الحالية بواسطة أول متغير من متغيرات البيئة LC_CTYPE، LC_ALL، و LANG، التي يتم العثور عليها تحتوي على اسم إعدادات محلية صالح. إذا لم يتم تعريف أي من هذه المتغيرات، أو إذا أهمل البرنامج استدعاء setlocale، فسيتم استخدام الإعدادات المحلية الافتراضية C، وهي US 7-bit ASCII. على معظم الأنظمة الشبيهة بـ Unix، يمكنك الحصول على قائمة بالإعدادات المحلية الصالحة عن طريق كتابة الأمر:

  locale -a

في موجه الأوامر. يمكن العثور على مزيد من الوثائق حول كيفية استخدام المستخدم لهذا لدخول الأحرف الدولية في صفحة الدليل tecla(7).

سلامة الخيوط

في برنامج متعدد الخيوط، يجب عليك استخدام إصدار libtecla_r.a من المكتبة. يستخدم هذا إصدارات قابلة لإعادة الدخول من وظائف النظام، حيثما كان ذلك متاحًا. لسوء الحظ، لم يتم تصميم أي من terminfo أو termcap ليكون قابلاً لإعادة الدخول، لذا لا يمكنك استخدام دوال وحدة getline بأمان في خيوط متعددة (يمكنك استخدام وحدات توسيع الملف وإكمال الكلمات المنفصلة في خيوط متعددة، انظر صفحات الدليل المقابلة للتفاصيل). ومع ذلك، بسبب استخدام دوال POSIX القابلة لإعادة الدخول للبحث عن الأدلة الرئيسية وما إلى ذلك، فمن الآمن استخدام هذه الوحدة من خيط واحد لبرنامج متعدد الخيوط، شريطة ألا تستخدم خيوطك الأخرى أي دوال termcap أو terminfo.
