Glib__Object__Introspection(3pm)

الحزمة: perl-glib-object-introspection

التراخيص: LGPL-2.1-or-later

Glib::Object::Introspection(3) وثائق بيرل المساهم بها Glib::Object::Introspection(3)

الاسم

Glib::Object::Introspection - إنشاء روابط لغة بيرل ديناميكيًا

الملخص

use Glib::Object::Introspection;
Glib::Object::Introspection->setup(
  basename => 'Gtk',
  version => '3.0',
  package => 'Gtk3');
# الآن GtkWindow، على سبيل المثال، متاحة كـ
# Gtk3::Window، ويمكنك استدعاء gtk_window_new كـ Gtk3::Window->new

الملخص العام

يستخدم Glib::Object::Introspection مشروعي gobject-introspection و libffi لإنشاء روابط ديناميكية للغة بيرل لمجموعة واسعة من المكتبات. الأمثلة تشمل Gtk، webkit، libsoup والعديد غيرها.

وصف لمستخدمي المكتبات

للسماح لـ Glib::Object::Introspection بإنشاء روابط لمكتبة ما، يجب أن تكون المكتبة قد قامت بتثبيت ملف typelib، على سبيل المثال "$prefix/lib/girepository-1.0/Gtk-3.0.typelib". في الكود الخاص بك، قم ببساطة باستدعاء "Glib::Object::Introspection->setup" مع أزواج المفتاح-القيمة التالية لإعداد كل شيء:

basename => $basename
الاسم الأساسي للمكتبة التي يجب تغليفها. إذا كان ملف typelib الخاص بك يسمى "Gtk-3.0.typelib"، فإن الاسم الأساسي هو 'Gtk'.
version => $version
الإصدار المحدد للمكتبة التي يجب تغليفها، في شكل سلسلة نصية. بالنسبة لـ "Gtk-3.0.typelib"، فهو '3.0'.
package => $package
اسم حزمة بيرل حيث يجب أن تكون جميع الفئات والطرق الخاصة بالمكتبة متجذرة. إذا كانت مكتبة بالاسم الأساسي 'Gtk' تحتوي على فئة 'GtkWindow'، واخترت 'Gtk3' كحزمة، فستكون هذه الفئة متاحة كـ 'Gtk3::Window'.

روابط بيرل التي تم إنشاؤها بواسطة "Glib::Object::Introspection" تتبع اصطلاحات وحدة Glib والروابط القديمة المكتوبة يدويًا مثل Gtk2. يمكنك استخدام الأداة المضمنة "perli11ndoc" لعرض وثائق جميع المكتبات المثبتة منظمة ومعروضة وفقًا لهذه الاصطلاحات. المبادئ التوجيهية الأساسية لهذه الاصطلاحات موصوفة فيما يلي.

فضاءات الأسماء والكائنات

يتم تعيين فضاءات الأسماء الخاصة بمكتبات C إلى حزم بيرل وفقًا للخيار "package" المحدد، على سبيل المثال:

gtk_ => Gtk3
gdk_ => Gtk3::Gdk
gdk_pixbuf_ => Gtk3::Gdk::Pixbuf
pango_ => Pango

تحصل الفئات والواجهات وأنواع البيانات المعبأة والأساسية على فضاءات الأسماء الخاصة بها، بطريقة ما، حيث يتم استبدال مفهوم GType تمامًا في روابط بيرل باسم حزمة بيرل.

GtkButton => Gtk3::Button
GdkPixbuf => Gtk3::Gdk::Pixbuf
GtkScrolledWindow => Gtk3::ScrolledWindow
PangoFontDescription => Pango::FontDescription

مع تعيين الحزمة هذا وبحث الطرق المضمن في بيرل، يمكن للروابط إجراء تحويل الكائنات نيابة عنك. هذا يمنحنا صيغة موجهة للكائنات بشكل مريح، باستخدام دلالات الكائنات القياسية في بيرل:

في C:
  GtkWidget * b;
  b = gtk_check_button_new_with_mnemonic ("_Something");
  gtk_toggle_button_set_active (GTK_TOGGLE_BUTTON (b), TRUE);
  gtk_widget_show (b);
في بيرل:
  my $b = Gtk3::CheckButton->new_with_mnemonic ('_Something');
  $b->set_active (1);
  $b->show;

تلاحظ من هذا أن وحدات الماكرو للتحويل ليست ضرورية وأنك لا تحتاج إلى كتابة بادئات فضاء الأسماء كثيرًا، لذا فإن الكود الخاص بك أقصر بكثير.

الأعلام والتعدادات

يتم التعامل مع الأعلام وقيم التعداد كسلاسل نصية، لأنها أكثر قابلية للقراءة بكثير من الأرقام، ولأنها سحرية بفضل نظام GType. تتم الإشارة إلى القيم بأسمائها المستعارة؛ بشكل أساسي، قم بتجريد البادئة المشتركة، اجعلها بأحرف صغيرة، وقم بتحويل '_' إلى '-' اختياريًا:

GTK_WINDOW_TOPLEVEL => 'toplevel'
GTK_BUTTONS_OK_CANCEL => 'ok-cancel' (أو 'ok_cancel')

الأعلام هي حالة خاصة. لا يمكنك (بشكل معقول) إجراء عملية OR ثنائية على هذه الثوابت النصية، لذلك تقدم مرجعًا لمصفوفة منها بدلاً من ذلك. المصفوفات المجهولة مفيدة هنا، والمصفوفة المجهولة الفارغة هي طريقة بسيطة لقول 'لا توجد أعلام'.

FOO_BAR_BAZ | FOO_BAR_QUU | FOO_BAR_QUUX => [qw/baz quu qux/]
0 => []

في بعض الحالات تحتاج إلى معرفة ما إذا كانت بتة معينة محددة في حقل بتات؛ لذلك تقوم الطرق التي ترجع الأعلام بإرجاع كائن مُحمّل بشكل زائد. انظر Glib لمزيد من التفاصيل حول العمليات المسموح بها على كائنات الأعلام هذه، ولكن إليك مثال سريع:

في C:
  /* event->state هو حقل بتات */
  if (event->state & GDK_CONTROL_MASK) g_printerr ("control was down\n");
في بيرل:
  # $event->state هو كائن خاص
  warn "control was down\n" if $event->state & "control-mask";

ولكن هذا يعمل أيضًا:

warn "control was down\n" if $event->state * "control-mask";
warn "control was down\n" if $event->state >= "control-mask";
warn "control and shift were down\n"
                          if $event->state >= ["control-mask", "shift-mask"];

إدارة الذاكرة

الوظائف الخاصة بإضافة مرجع وإزالة مرجع للكائنات وتحرير هياكل البيانات المعبأة غير متاحة في بيرل، لأن كل شيء يتم التعامل معه تلقائيًا بواسطة الروابط. سيتم الاحتفاظ بالكائنات على قيد الحياة طالما أن لديك قيمة بيرل قياسية تشير إليها أو أن الكائن مُشار إليه بطريقة أخرى، على سبيل المثال من حاوية.

الشيء الوحيد الذي عليك الانتباه إليه هو دورة حياة الهياكل غير المُدارة بالمراجع، والتي تشمل معظم الأشياء المشتقة من "Glib::Boxed". إذا جاءت من رد استدعاء إشارة، فقد تكون صالحة فقط حتى تعود، أو إذا كانت جزءًا داخليًا لكائن آخر، فقد تكون صالحة فقط طالما أن هذا الكائن موجود. إذا كنت في شك، يمكنك "copy". الهياكل من "copy" أو "new" هي ملكك وتعيش طالما تم الإشارة إليها من بيرل.

ردود الاستدعاء

استخدم خدع رد الاستدعاء / الإغلاق القياسية في بيرل مع ردود الاستدعاء. الاستخدام الأكثر شيوعًا لردود الاستدعاء سيكون مع طريقة Glib "signal_connect":

$widget->signal_connect (event => \&event_handler, $user_data);
$button->signal_connect (clicked => sub { warn "hi!\n" });

$user_data اختياري، ومع إغلاقات بيرل غالبًا ما لا تحتاج إليها (انظر "المتغيرات المستمرة مع الإغلاقات" في perlsub).

يتم الاحتفاظ ببيانات المستخدم في قيمة قياسية، تم تهيئتها من ما قدمته في "signal_connect" وما إلى ذلك. يتم تمريرها إلى رد الاستدعاء بأسلوب بيرل المعتاد "الاستدعاء عن طريق المرجع" مما يعني أن رد الاستدعاء يمكنه تعديل حجته الأخيرة، أي $_[-1]، لتعديل بيانات المستخدم المحفوظة. هذا دقيق بعض الشيء، ولكن يمكنك استخدامه لبعض "الحالة" المرتبطة بالاتصال.

$widget->signal_connect (activate => \&my_func, 1);
sub my_func {
  print "activation count: $_[-1]\n";
  $_[-1] ++;
}

نظرًا لأن بيانات المستخدم المحفوظة هي قيمة قياسية جديدة، فلا يوجد تغيير في المتغير (إلخ) الذي مررته في الأصل إلى "signal_connect".

إذا كان لديك كائن أب في بيانات المستخدم (أو الإغلاق) ، فعليك توخي الحذر بشأن المراجع الدائرية التي تمنع تدمير الأب والابن. انظر "جمع القمامة ذو المرحلتين" في perlobj حول هذا بشكل عام. تحتاج الأدوات الطرفية مثل "Gtk3::Window" دائمًا إلى "$widget->destroy" صريح، لذا فإن إشارة "destroy" الخاصة بها هي مكان جيد لكسر المراجع الدائرية. ولكن بالنسبة للأدوات الأخرى، يكون الأمر ودودًا بشكل عام لتجنب الدوائر في المقام الأول، إما عن طريق استخدام مراجع ضعيفة في بيانات المستخدم، أو ربما تحديد موقع كائن أب ديناميكيًا باستخدام "$widget->get_ancestor".
