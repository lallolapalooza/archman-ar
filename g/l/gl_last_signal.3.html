gl_last_signal(3)

الحزمة: libtecla

التراخيص: custom:X11

ctivity_timeout(GetLine *gl, GlTimeoutFn *callback, void *data, unsigned long sec, unsigned long nsec);

يتم تحديد المهلة بتنسيق عدد صحيح من الثواني وعدد صحيح من النانو ثانية، عبر الوسيطين sec و nsec على التوالي. بعد ذلك، كلما لم يتم رؤية أي نشاط لهذه الفترة الزمنية، يتم استدعاء الدالة المحددة عبر الوسيط callback. يتم تمرير الوسيط data الخاص بـ gl_inactivity_timeout() حرفيًا إلى دالة رد الاتصال هذه كلما تم استدعاؤها، ويمكن بالتالي استخدامه لتمرير معلومات خاصة بالتطبيق عشوائية إلى رد الاتصال. يتم توفير الماكرو التالي في libtecla.h لتستخدمه التطبيقات للإعلان عن وتحديد دوال رد اتصال المهلة.

  #define GL_TIMEOUT_FN(fn) \
               GlAfterTimeout (fn)(GetLine *gl, void *data)

عند العودة، يُتوقع من دالة رد اتصال التطبيق أن تُرجع أحد المُعدِّدات التالية لإخبار gl_get_line() بكيفية المتابعة بعد معالجة المهلة بواسطة رد الاتصال.

  GLTO_ABORT    -  أخبر gl_get_line() بالإلغاء. عندما يحدث
                   هذا، ستُرجع gl_get_line()
                   NULL، وسيُرجع استدعاء لاحق لـ
                   gl_return_status()
                   GLR_TIMEOUT. لاحظ أنه إذا كان التطبيق
                   يحتاج إلى أن يكون errno دائمًا له قيمة
                   ذات مغزى عندما تُرجع
                   gl_get_line()
                   NULL، فيجب على دالة رد الاتصال تعيين
                   errno بشكل مناسب.
  GLTO_REFRESH  -  إعادة رسم سطر الإدخال، ثم المتابعة
                   في انتظار الإدخال. يجب عليك إرجاع هذه
                   القيمة إذا قامت دالة رد الاتصال بالكتابة
                   إلى الطرفية دون استدعاء
                   gl_normal_io(gl) أولاً.
  GLTO_CONTINUE -  في وضع الإدخال/الإخراج المتزامن العادي،
                   استمر في انتظار الإدخال، دون إعادة رسم
                   سطر إدخال المستخدم.
                   في وضع خادم الإدخال/الإخراج غير المتزامن
                   (انظر gl_io_mode(3))، اجعل
                   gl_get_line() يتصرف كما لو كان
                   الإدخال/الإخراج قد تم حظره. هذا يعني أن
                   gl_get_line() سيعود فورًا بـ
                   NULL، وسيُرجع استدعاء لاحق لـ
                   gl_return_status()
                   GLR_BLOCKED.

لاحظ أنه قبل استدعاء دالة رد الاتصال، يقوم gl_get_line() بحظر معظم الإشارات، ويترك معالجات الإشارة الخاصة به مثبتة، لذلك إذا كنت بحاجة إلى التقاط إشارة معينة، فستحتاج إلى تثبيت معالج الإشارة الخاص بك مؤقتًا، وإلغاء حظر الإشارة. تأكد من إعادة حظر الإشارة (إذا كانت محظورة في الأصل) وإعادة تثبيت معالج الإشارة الأصلي، إن وجد، قبل العودة.

إذا احتاجت دالة رد الاتصال إلى القراءة أو الكتابة إلى الطرفية، فيجب عليها مثاليًا استدعاء gl_normal_io(gl) أولاً لتعليق تحرير السطر مؤقتًا. سيؤدي ذلك إلى استعادة الطرفية إلى وضعها الطبيعي، ووضع الإدخال/الإخراج المتزامن، ونقل المؤشر إلى بداية سطر طرفي جديد. لاحقًا، عند عودة دالة رد الاتصال، سيلاحظ gl_get_line() أنه تم استدعاء gl_normal_io()، وسيعيد عرض سطر الإدخال ويستأنف التحرير. لاحظ أنه في هذه الحالة، تكون قيمتا الإرجاع GLTO_REFRESH و GLTO_CONTINUE متكافئتين.

لدعم الحالات التي تستدعي فيها دالة رد الاتصال دالة طرف ثالث تكتب أحيانًا وبشكل غير متوقع إلى الطرفية، يتم إعادة تمكين التحويل التلقائي من "0 إلى "0 قبل استدعاء دالة رد الاتصال. إذا علمت دالة رد الاتصال أن الدالة الطرف الثالث كتبت إلى الطرفية، فيجب عليها بعد ذلك إرجاع قيمة الإرجاع GLTO_REFRESH، لإخبار gl_get_line() بإعادة عرض سطر الإدخال.

لاحظ أنه على الرغم من أن وسيط المهلة يتضمن مكون نانو ثانية، إلا أن الساعات الحاسوبية القليلة present لها دقة تتجاوز بضعة مللي ثانية، لذا فإن طلب أقل من بضعة مللي ثانية يعادل طلب صفر ثانية على العديد من الأنظمة. إذا كان هذا يمثل مشكلة، فيجب عليك استناد اختيار المهلة الخاص بك إلى الدقة الفعلية لساعة المضيف (على سبيل المثال، عن طريق استدعاء sysconf(_SC_CLK_TCK)).

لإيقاف المهلات، ببساطة قم باستدعاء gl_inactivity_timeout() مع وسيط callback بقيمة 0. يتم تجاهل الوسيط data في هذه الحالة.

إعدادات معالجة الإشارات الافتراضية

افتراضيًا، تقوم الدالة gl_get_line() باعتراض عدد من الإشارات. هذا مهم بشكل خاص للإشارات التي من شأنها افتراضيًا إنهاء العملية، لأن الطرفية تحتاج إلى استعادتها إلى حالة قابلة للاستخدام قبل حدوث ذلك. في هذا القسم، يتم وصف الإشارات التي يتم اعتراضها افتراضيًا، وكيف تستجيب لها gl_get_line(). تغيير هذه الإعدادات الافتراضية هو موضوع القسم التالي.

عند التقاط المجموعة الفرعية التالية من الإشارات، تقوم gl_get_line() أولاً باستعادة إعدادات الطرفية ومعالجة الإشارات إلى ما كانت عليه قبل استدعاء gl_get_line()، وتعيد إرسال الإشارة، للسماح لمعالجات الإشارة الخاصة بالتطبيق باستلامها، ثم إذا كانت العملية لا تزال موجودة، تُرجع gl_get_line() NULL وتضبط errno كما هو محدد أدناه.

 SIGINT  -  يتم إنشاء هذه الإشارة عن طريق مفتاح مقاطعة لوحة
            المفاتيح (عادةً ^C)، ومفتاح إيقاف لوحة
            المفاتيح.
            errno=EINTR
 SIGHUP  -  يتم إنشاء هذه الإشارة عند خروج الطرفية
            التحكمية.
            errno=ENOTTY
 SIGPIPE -  يتم إنشاء هذه الإشارة عندما يحاول برنامج
            الكتابة إلى أنبوب لا تتم قراءته من نهايته
            البعيدة بواسطة أي عملية. يمكن أن يحدث هذا
            على سبيل المثال إذا قمت باستدعاء
            gl_change_terminal() لإعادة توجيه
            الإخراج إلى أنبوب مخفي تحت طرفية زائفة.
            errno=EPIPE
 SIGQUIT -  يتم إنشاء هذه الإشارة عن طريق مفتاح إنهاء
            لوحة المفاتيح (عادةً ^\).
            errno=EINTR
 SIGABRT -  يتم إنشاء هذه الإشارة بواسطة دالة
            abort() القياسية C. افتراضيًا، تقوم بإنهاء
            العملية وتوليد تفريغ للنواة.
            errno=EINTR
 SIGTERM -  هذه هي الإشارة الافتراضية التي يرسلها أمر
            kill في UN*X إلى العمليات.
            errno=EINTR

لاحظ أنه في حالة جميع الإشارات المذكورة أعلاه، ينص POSIX على أنه افتراضيًا يتم إنهاء العملية، مع إضافة تفريغ للنواة في حالة إشارة SIGQUIT. بعبارة أخرى، إذا لم يتجاوز التطبيق المستدعي المعالج الافتراضي عن طريق توفير معالج إشارة خاص به، فإن استلام الإشارة المقابلة سيؤدي إلى إنهاء التطبيق قبل أن تُرجع gl_get_line().

إذا تم إلغاء gl_get_line() مع تعيين errno إلى EINTR، يمكنك معرفة الإشارة التي تسببت في إلغائها، عن طريق استدعاء الدالة التالية.

  int gl_last_signal(const GetLine *gl);

تقوم هذه الدالة بإرجاع الرمز الرقمي (مثل SIGINT) لآخر إشارة تم استقبالها خلال آخر استدعاء لـ gl_get_line()، أو -1 إذا لم يتم استقبال أي إشارات.

على الأنظمة التي تدعمها، عند استقبال إشارة SIGWINCH (تغيير النافذة)، تقوم gl_get_line() بالاستعلام من الطرفية لمعرفة حجمها الجديد، وإعادة رسم سطر الإدخال الحالي لاستيعاب الحجم الجديد، ثم تعود للانتظار لإدخال لوحة المفاتيح من المستخدم. على عكس الإشارات الأخرى، لا يتم إعادة إرسال هذه الإشارة إلى التطبيق.

أخيرًا، تتسبب الإشارات التالية في أن تقوم gl_get_line() أولاً باستعادة الطرفية وبيئة الإشارات إلى ما كانت عليه قبل استدعاء gl_get_line()، ثم إعادة إرسال الإشارة إلى التطبيق. إذا كانت العملية لا تزال موجودة بعد تسليم الإشارة، فإن gl_get_line() تعيد بعد ذلك تثبيت معالجات الإشارة الخاصة بها، وتحول الطرفية مرة أخرى إلى الوضع الخام، وتعرض سطر الإدخال مرة أخرى، وتعود إلى انتظار إدخال الطرفية من المستخدم.

 SIGCONT    -  يتم إنشاء هذه الإشارة عند استئناف عملية
               متوقفة.
 SIGPOLL    -  في أنظمة SVR4، تخطر هذه الإشارة العملية
               بحدث إدخال/إخراج غير متزامن. لاحظ أنه
               تحت 4.3+BSD، SIGIO و SIGPOLL متماثلان.
               في الأنظمة الأخرى، يتم تجاهل SIGIO
               افتراضيًا، لذلك لا تعترضها
               gl_get_line() افتراضيًا.
 SIGPWR     -  يتم إنشاء هذه الإشارة عند حدوث فشل في
               الطاقة (يفترض عند تشغيل النظام على
               UPS).
 SIGALRM    -  يتم إنشاء هذه الإشارة عند انتهاء مؤقت.
 SIGUSR1    -  إشارة خاصة بالتطبيق.
 SIGUSR2    -  إشارة أخرى خاصة بالتطبيق.
 SIGVTALRM  -  يتم إنشاء هذه الإشارة عند انتهاء مؤقت
               افتراضي (انظر man setitimer(2)).
 SIGXCPU    -  يتم إنشاء هذه الإشارة عندما تتجاوز العملية
               حد وقت وحدة المعالجة المركزية البرمجي الخاص
               بها.
 SIGXFSZ    -  يتم إنشاء هذه الإشارة عندما تتجاوز العملية
               حد حجم الملف البرمجي الخاص بها.
 SIGTSTP    -  يتم إنشاء هذه الإشارة بواسطة مفتاح إيقاف
               الطرفية، وهو عادةً ^Z، أو مفتاح إيقاف
               الطرفية المؤجل، وهو عادةً ^Y.
 SIGTTIN    -  يتم إنشاء هذه الإشارة إذا حاولت البرامج
               القراءة من الطرفية أثناء تشغيل البرنامج
               في الخلفية.
 SIGTTOU    -  يتم إنشاء هذه الإشارة إذا حاولت البرامج
               الكتابة إلى الطرفية أثناء تشغيل البرنامج
               في الخلفية.

من الواضح أن ليست كل الإشارات المذكورة أعلاه مدعومة على جميع الأنظمة، لذلك يتم تجميع الكود لدعمها بشكل شرطي في مكتبة tecla.

لاحظ أنه إذا تم استقبال إشارات SIGKILL أو SIGPOLL، والتي لا يمكن التقاطها بحكم تعريفها، أو أي من إشارات الاستثناء المولدة بواسطة الأجهزة، مثل SIGSEGV و SIGBUS و SIGFPE، ولم يتم التعامل معها بينما كانت gl_get_line() تضع الطرفية في الوضع الخام، فسيتم إنهاء البرنامج دون استعادة الطرفية إلى حالة قابلة للاستخدام. في الممارسة العملية، تقوم shells التحكم في المهام عادةً بإعادة تعيين إعدادات الطرفية عندما يتخلى البرنامج عن الطرفية التحكمية، لذا فإن هذه مشكلة فقط مع shells القديمة.

معالجة إشارات مخصصة

سرد القسم السابق الإشارات التي تعترضها gl_get_line() افتراضيًا، ووصف كيفية استجابتها لها. يصف هذا القسم كيفية إضافة إشارات وإزالتها من قائمة الإشارات المعترضة، وكيفية تحديد كيفية استجابة gl_get_line() لإشارة معينة.

إذا لم تكن بحاجة إلى أن تقوم gl_get_line() بأي شيء استجابةً لإشارة تعترضها عادةً، يمكنك إخبار gl_get_line() بتجاهل تلك الإشارة عن طريق استدعاء gl_ignore_signal().

  int gl_ignore_signal(GetLine *gl, int signo);

الوسيط signo هو رقم الإشارة (مثل SIGINT) التي تريد تجاهلها. إذا لم تكن الإشارة المحددة حاليًا من بين تلك التي يتم اعتراضها، فلن تقوم هذه الدالة بأي شيء.
