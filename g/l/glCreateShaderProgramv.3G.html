glCreateShaderProgramv(3G)

الحزمة: opengl-man-pages

التراخيص: SGI-B-2.0, OPL-1.0

GLCREATESHADERPROGRA(3G) [FIXME: manual] GLCREATESHADERPROGRA(3G)

الاسم

glCreateShaderProgramv - إنشاء برنامج مستقل من مصفوفة من سلاسل تعليمات مصدرية تنتهي بصفر

مواصفات C

GLuint glCreateShaderProgramv(GLenum type, GLsizei count, const char **strings);

المعاملات

type

يحدد نوع الـ shader المراد إنشاؤه.

count

يحدد عدد سلاسل تعليمات المصدر في المصفوفة strings.

strings

يحدد عنوان مصفوفة من المؤشرات إلى سلاسل تعليمات المصدر التي سيتم إنشاء كائن البرنامج منها.

الوصف

glCreateShaderProgram ينشئ كائن برنامج يحتوي على shaders مجمعة ومرتبطة لمرحلة واحدة محددة بواسطة type. strings يشير إلى مصفوفة من count سلاسل لإنشاء ملفات تنفيذية للـ shader.

glCreateShaderProgram يعادل (بافتراض عدم حدوث أخطاء):

    const GLuint shader = glCreateShader(type);
    if (shader) {
        glShaderSource(shader, count, strings, NULL);
        glCompileShader(shader);
        const GLuint program = glCreateProgram();
        if (program) {
            GLint compiled = GL_FALSE;
            glGetShaderiv(shader, GL_COMPILE_STATUS, &compiled);
            glProgramParameteri(program, GL_PROGRAM_SEPARABLE, GL_TRUE);
            if (compiled) {
                glAttachShader(program, shader);
                glLinkProgram(program);
                glDetachShader(program, shader);
            }
            /* append-shader-info-log-to-program-info-log */
        }
        glDeleteShader(shader);
        return program;
    } else {
        return 0;
    }

كائن البرنامج الذي تم إنشاؤه بواسطة glCreateShaderProgram لديه حالته GL_PROGRAM_SEPARABLE مضبوطة على GL_TRUE.

الأخطاء

يتم إنشاء GL_INVALID_ENUM إذا كان type ليس نوع shader مقبول.

يتم إنشاء GL_INVALID_VALUE إذا كان count سالباً.

يتم إنشاء أخطاء أخرى إذا فشل كود الـ shader المقدم في التجميع والربط، كما هو موضح للأوامر في تسلسل الكود الزائف أعلاه، ولكن يتم إنشاء جميع هذه الأخطاء دون أي آثار جانبية لتنفيذ تلك الأوامر.
