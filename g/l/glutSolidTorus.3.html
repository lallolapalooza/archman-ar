glutSolidTorus(3)

الحزمة: freeglut

التراخيص: X11

glutSolidTorus(3) freeglut glutSolidTorus(3)

الاسم

glutSolidTorus, glutWireTorus - عرض شكل طارة صلب أو سلكي على التوالي (دونات).

الصيغة

void glutSolidTorus(GLdouble innerRadius,
                    GLdouble outerRadius,
                    GLint nsides, GLint rings);
void glutWireTorus(GLdouble innerRadius,
                   GLdouble outerRadius,
                   GLint nsides, GLint rings);

الوسائط

innerRadius
نصف القطر الداخلي للطارة.
outerRadius
نصف القطر الخارجي للطارة.
nsides
عدد الأضلاع لكل قسم شعاعي.
rings
عدد التقسيمات الشعاعية للطارة.

الوصف

تقوم الدالتان glutSolidTorus و glutWireTorus بعرض طارة صلبة أو سلكية (دونات) على التوالي، مركزة عند أصل إحداثيات النمذجة مع محاذاة محورها مع المحور Z.

انظر أيضًا

glutSolidSphere, glutSolidCube, glutSolidCone, glutSolidDodecahedron, glutSolidOctahedron, glutSolidTetrahedron, glutSolidIcosahedron, glutSolidTeapot
