al_io(gl)) /* تعليق مؤقت للتعديل */ return GLA_ABORT; printf("Hello world\n"); return GLA_CONTINUE; }

يجب أن تعيد دوال الإجراءات إحدى القيم التالية، لإخبار gl_get_line() بكيفية المتابعة.

  GLA_ABORT     -   تسبب في أن تعيد gl_get_line() القيمة NULL.
  GLA_RETURN    -   تسبب في أن تعيد gl_get_line()
                    السطر المكتمل.
  GLA_CONTINUE  -   استئناف تعديل سطر الأوامر.

لاحظ أن الوسيط name للدالة gl_register_action() يحدد الاسم الذي يمكن للمستخدم الرجوع إليه للإجراء في ملف التكوين الخاص به. يسمح له هذا بإعادة ربط الإجراء بتسلسل مفاتيح بديل. ولكي يعمل هذا، من الضروري استدعاء gl_register_action() بين استدعاء new_GetLine() والاستدعاء الأول لـ gl_get_line().

ملفات السجل

لحفظ محتويات مخزن السجل قبل إنهاء تطبيقك، ولاحقًا استعادتها عند بدء تشغيل التطبيق مرة أخرى، تم توفير الدوال التالية.

 int gl_save_history(GetLine *gl, const char *filename,
                     const char *comment, int max_lines);
 int gl_load_history(GetLine *gl, const char *filename,
                     const char *comment);

يحدد الوسيط filename الاسم الذي يتم إعطاؤه لملف السجل عند الحفظ، أو اسم ملف سجل موجود، عند التحميل. قد يحتوي هذا على تعبيرات مسار المستخدم والمتغيرات البيئية، مثل ""~/.myapp_history"" أو ""$HOME/.myapp_history".

بالإضافة إلى كل سطر سجل، يتم تسجيل معلومات إضافية عنه، مثل وقت إدخاله بواسطة المستخدم، ومستوى تداخله، كتعليق يسبق السطر في ملف السجل. كتابة هذا كتعليق يسمح لملف السجل بأن يكون بمثابة ملف أوامر، فقط في حال كنت ترغب في إعادة تشغيل جلسة كاملة باستخدامها. نظرًا لأن بادئات التعليقات تختلف في اللغات المختلفة، يتم توفير الوسيط comment لتحديد بادئة التعليق. على سبيل المثال، إذا كان تطبيقك عبارة عن shell يونكس، مثل bourne shell، فستحدد ""#"" هنا. أي شيء تختاره لحرف التعليق، يجب عليك تحديد نفس البادئة لـ gl_load_history() التي استخدمتها عند استدعاء gl_save_history() لكتابة ملف السجل.

يجب أن يكون max_lines إما -1 لتحديد حفظ جميع الأسطر في قائمة السجل، أو رقمًا موجبًا يحدد حدًا أقصى لعدد الأسطر الأحدث التي يجب حفظها.

تعيد كلتا الدالتين قيمة غير صفرية عند حدوث خطأ، بعد كتابة رسالة خطأ إلى stderr. لاحظ أن gl_load_history() لا تعتبر عدم وجود ملف خطأ.

قوائم سجل متعددة

إذا كان تطبيقك يستخدم كائن GetLine واحدًا لإدخال العديد من أنواع أسطر الإدخال المختلفة، فقد ترغب في أن تقوم gl_get_line() بتمييز أنواع الأسطر المختلفة في قائمة السجل، واستدعاء الأسطر التي تطابق النوع الحالي للسطر فقط. لدعم هذا المطلب، تقوم gl_get_line() بتمييز الأسطر التي يتم تسجيلها في قائمة السجل بمعرف صحيح اختارته التطبيق. مبدئيًا، يتم تعيين هذا المعرف إلى 0 بواسطة new_GetLine()، ولكن يمكن تغييره لاحقًا عن طريق استدعاء gl_group_history().

  int gl_group_history(GetLine *gl, unsigned id);

يمكن أن يكون المعرف الصحيح id أي رقم اختاره التطبيق، ولكن لاحظ أن gl_save_history() و gl_load_history() تحافظان على الارتباط بين المعرفات وأسطر الإدخال التاريخية بين استدعاءات البرنامج، لذلك يجب عليك اختيار معرفات ثابتة لأنواع أسطر الإدخال المختلفة التي يستخدمها تطبيقك.

كلما قامت gl_get_line() بإلحاق سطر إدخال جديد بقائمة السجل، يتم تسجيل معرف السجل الحالي معه، وعندما يُطلب منها استدعاء سطر إدخال تاريخي، فإنها تستدعي فقط الأسطر المميزة بالمعرف الحالي.

عرض السجل

يمكن عرض قائمة السجل عن طريق استدعاء gl_show_history().

  int gl_show_history(GetLine *gl, FILE *fp,
                      const char *fmt,
                      int all_groups,
                      int max_lines);

يعرض هذا المحتويات الحالية لقائمة السجل إلى دفق الإخراج القياسي fp. إذا كان الوسيط max_lines أكبر من أو يساوي صفرًا، فلن يتم عرض أكثر من هذا العدد من الأسطر الأحدث. إذا كان الوسيط all_groups غير صفري، يتم عرض الأسطر من جميع مجموعات السجل. وإلا، يتم عرض فقط الأسطر من مجموعة السجل المحددة حاليًا. تحدد وسيطة سلسلة التنسيق، fmt، كيفية عرض السطر. يمكن أن تحتوي هذه على أحرف عشوائية يتم كتابتها حرفيًا، متداخلة مع أي من توجيهات التنسيق التالية:

  %D  -  التاريخ الذي تم فيه إدخال السطر في الأصل،
         منسقًا مثل 2001-11-20.
  %T  -  الوقت من اليوم الذي تم فيه إدخال السطر،
         منسقًا مثل 23:59:59.
  %N  -  رقم الإدخال التسلسلي للسطر في
         مخزن السجل.
  %G  -  رقم مجموعة السجل التي ينتمي إليها
         السطر.
  %%  -  حرف % حرفي.
  %H  -  سطر السجل نفسه.

وبالتالي، فإن سلسلة تنسيق مثل ""%D %T %H\n"" ستخرج شيئًا مثل:

  2001-11-20 10:23:34  Hello world

لاحظ تضمين حرف سطر جديد صريح في سلسلة التنسيق.

البحث في السجل

تسمح الدالة gl_lookup_history() لتطبيق الاستدعاء بالبحث عن الأسطر في قائمة السجل.

  typedef struct {
    const char *line;    /* السطر التاريخي المطلوب */
                         /*  . */
    unsigned group;      /* مجموعة السجل التي ينتمي إليها */
                         /*  السطر. */
    time_t timestamp;    /* التاريخ والوقت الذي تم فيه */
                         /*  إدخال السطر في الأصل. */
  } GlHistoryLine;
  int gl_lookup_history(GetLine *gl, unsigned long id,
                        GlHistoryLine *hline);

يحدد الوسيط id السطر المطلوب البحث عنه، حيث يتم تمييز السطر الأول الذي تم إدخاله في قائمة السجل بعد استدعاء new_GetLine() بالرقم 0، ويتم تمييز الأسطر المدخلة لاحقًا بأرقام أعلى تدريجيًا. لاحظ أن النطاق الحالي للأسطر المحفوظة في قائمة السجل يمكن الاستعلام عنه عن طريق استدعاء الدالة gl_range_of_history()، الموضحة لاحقًا. إذا كان السطر المطلوب موجودًا في قائمة السجل، يتم تسجيل تفاصيل السطر في المتغير المشار إليه بواسطة الوسيط hline، ويتم إرجاع 1. بخلاف ذلك، يتم إرجاع 0، ويتم ترك المتغير المشار إليه بواسطة hline دون تغيير.

احذر من أن السلسلة النصية المعادة في hline->line هي جزء من مخزن السجل، لذلك يجب ألا يتم تعديلها بواسطة المتصل، وسيتم إعادة تدويرها عند الاستدعاء التالي لأي دالة تأخذ gl كوسيط لها. لذلك، يجب عليك إنشاء نسخة خاصة من هذه السلسلة إذا كنت بحاجة إلى الاحتفاظ بها.

أرشفة السجل يدويًا

افتراضيًا، كلما أدخل المستخدم سطرًا، يتم إلحاقه تلقائيًا بقائمة السجل، قبل أن تعيد gl_get_line() السطر إلى المتصل. هذا مناسب لغالبية التطبيقات، ولكن هناك أيضًا تطبيقات تحتاج إلى تحكم أدق فيما يتم إضافته إلى قائمة السجل. في مثل هذه الحالات، يمكن إيقاف الإضافة التلقائية للأسطر المدخلة إلى قائمة السجل عن طريق استدعاء الدالة gl_automatic_history().

  int gl_automatic_history(GetLine *gl, int enable);

إذا تم استدعاء هذه الدالة مع تعيين وسيطها enable إلى 0، فلن تقوم gl_get_line() بأرشفة الأسطر المدخلة لاحقًا تلقائيًا. يمكن إعادة تمكين الأرشفة التلقائية لاحقًا، عن طريق استدعاء هذه الدالة مرة أخرى، مع تعيين وسيطها enable إلى 1. بينما تكون أرشفة السجل التلقائية معطلة، يمكن لتطبيق الاستدعاء استخدام gl_append_history() لإلحاق الأسطر بقائمة السجل حسب الحاجة.

  int gl_append_history(GetLine *gl, const char *line);

يحدد الوسيط line السطر المراد إضافته إلى قائمة السجل. يجب أن تكون سلسلة نصية عادية منتهية بـ ' '. إذا احتوت هذه السلسلة على أي أحرف سطر جديد، فسيكون السطر الذي تتم أرشفته في قائمة السجل منتهيًا بالأول منها. وإلا، فسيكون منتهيًا بالفاصل ' '. إذا كان السطر أطول من الحد الأقصى لطول الإدخال، الذي تم تحديده عند استدعاء new_GetLine()، فعند استدعائه، سيتم اقتطاعه إلى الطول الفعلي لسطر gl_get_line().

إذا نجحت، تعيد gl_append_history() القيمة 0. وإلا، فإنها تعيد قيمة غير صفرية، وتعيّن errno إلى أحد القيم التالية.

   EINVAL  -  أحد الوسائط التي تم تمريرها إلى
              gl_append_history() كان NULL.
   ENOMEM  -  كان السطر المحدد أطول من الحجم المخصص
              لمخزن السجل (كما هو محدد عند استدعاء
              new_GetLine())، لذلك لم يكن من الممكن
              أرشفته.

يمكن الحصول اختياريًا على وصف نصي للخطأ عن طريق استدعاء gl_error_message(). لاحظ أنه بعد حدوث مثل هذا الخطأ، تظل قائمة السجل في حالة صالحة لاستقبال أسطر سجل جديدة، لذلك هناك ضرر قليل في تجاهل حالة الإرجاع لـ gl_append_history().

تكوين سجل متنوع

إذا كنت ترغب في تغيير حجم مخزن السجل الذي تم تحديده في الأصل في استدعاء new_GetLine()، يمكنك القيام بذلك باستخدام الدالة gl_resize_history().

  int gl_resize_history(GetLine *gl, size_t histlen);

يحدد الوسيط histlen الحجم الجديد بالبايت، وإذا قمت بتعيين هذا إلى 0، فسيتم حذف المخزن.

كما ذكر في مناقشة new_GetLine()، يعتمد عدد الأسطر التي يمكن تخزينها في مخزن السجل على أطوال الأسطر الفردية. على سبيل المثال، يمكن لمخزن بحجم 1000 بايت تخزين 10 أسطر بطول متوسط 100 بايت، أو سطرين بطول متوسط 50 بايت. على الرغم من أن المخزن لا يتوسع أبدًا عند إضافة أسطر جديدة، إلا أن قائمة بالمؤشرات داخل المخزن تتوسع عند الحاجة لاستيعاب عدد الأسطر المخزنة حاليًا في المخزن. لوضع حد أقصى لعدد الأسطر في المخزن، وبالتالي سقف لكمية الذاكرة المستخدمة في هذه القائمة، يمكنك استدعاء الدالة gl_limit_history
