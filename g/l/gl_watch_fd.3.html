الحزمة: libtecla

التراخيص: custom:X11

vity_timeout(GetLine *gl, GlTimeoutFn *callback,
                     void *data, unsigned long sec,
                     unsigned long nsec);

يتم تحديد المهلة بالصيغة التي تتكون من عدد صحيح من الثواني وعدد صحيح من النانوثواني، عبر الوسيطين sec و nsec على التوالي. بعد ذلك، كلما لم يتم رؤية أي نشاط لهذه الفترة الزمنية، يتم استدعاء الدالة المحددة عبر الوسيط callback. يتم تمرير الوسيط data الخاص بـ gl_inactivity_timeout() كما هو إلى دالة الاستدعاء هذه كلما تم استدعاؤها، ويمكن بالتالي استخدامه لتمرير معلومات عشوائية خاصة بالتطبيق إلى دالة الاستدعاء. يتم توفير الماكرو التالي في libtecla.h لتستخدمه التطبيقات لتعريف ونماذج دوال استدعاء المهلة.

  #define GL_TIMEOUT_FN(fn) \
               GlAfterTimeout (fn)(GetLine *gl, void *data)

عند العودة، يُتوقع من دالة الاستدعاء الخاصة بالتطبيق أن تُعيد أحد المحددات التالية لإخبار gl_get_line() بكيفية المتابعة بعد معالجة المهلة بواسطة دالة الاستدعاء.

  GLTO_ABORT    -  تخبر gl_get_line() بالإجهاض. عندما
                   يحدث هذا، ستُعيد gl_get_line()
                   NULL، وسيُعيد استدعاء لاحق لـ
                   gl_return_status() القيمة
                   GLR_TIMEOUT. لاحظ أنه إذا كان التطبيق
                   يحتاج إلى أن تكون errno دائمًا ذات
                   قيمة ذات معنى عندما تُعيد gl_get_line()
                   القيمة NULL، فيجب على دالة الاستدعاء
                   تعيين errno بشكل مناسب.
  GLTO_REFRESH  -  إعادة رسم سطر الإدخال، ثم الاستمرار في
                   الانتظار للإدخال. يجب عليك إرجاع هذه
                   القيمة إذا قامت دالة الاستدعاء بالكتابة
                   على الطرفية دون استدعاء gl_normal_io(gl)
                   أولاً.
  GLTO_CONTINUE -  في وضع الإدخال/الإخراج الاعتيادي الحاجب،
                   استمر في انتظار الإدخال، دون إعادة رسم
                   سطر إدخال المستخدم.
                   في وضع خادم الإدخال/الإخراج غير الحاجب
                   (راجع gl_io_mode(3))، اجعل gl_get_line()
                   يتصرف كما لو كان الإدخال/الإخراج حاجبًا.
                   هذا يعني أن gl_get_line() ستُعيد
                   على الفور NULL، وسيُعيد استدعاء لاحق
                   لـ gl_return_status() القيمة
                   GLR_BLOCKED.

لاحظ أنه قبل استدعاء دالة الاستدعاء، تقوم gl_get_line() بحجب معظم الإشارات، وتترك معالجات الإشارات الخاصة بها مثبتة، لذا إذا كنت تحتاج إلى التقاط إشارة معينة، فستحتاج إلى تثبيت معالج الإشارات الخاص بك مؤقتًا، وإلغاء حظر الإشارة. تأكد من إعادة حظر الإشارة (إذا كانت محظورة في الأصل) وإعادة تثبيت معالج الإشارة الأصلي، إن وجد، قبل العودة.

إذا احتاجت دالة الاستدعاء إلى القراءة أو الكتابة على الطرفية، فيجب عليها مثاليًا استدعاء gl_normal_io(gl) أولاً لتعليق تحرير السطر مؤقتًا. سيؤدي هذا إلى استعادة الطرفية إلى وضعها الطبيعي، والإدخال/الإخراج الحاجب، ونقل المؤشر إلى بداية سطر طرفية جديد. لاحقًا، عند عودة دالة الاستدعاء، ستلاحظ gl_get_line() أن gl_normal_io() قد تم استدعاؤها، وستعيد عرض سطر الإدخال واستئناف التحرير. لاحظ أنه في هذه الحالة، تكون قيم العودة، GLTO_REFRESH و GLTO_CONTINUE متكافئة.

لدعم الحالات التي تستدعي فيها دالة الاستدعاء دالة طرف ثالث تكتب أحيانًا وبشكل غير متوقع إلى الطرفية، يتم إعادة تمكين التحويل التلقائي لـ "0 إلى "0 قبل استدعاء دالة الاستدعاء. إذا عرفت دالة الاستدعاء أن دالة الطرف الثالث كتبت إلى الطرفية، فيجب عليها بعد ذلك إرجاع قيمة العودة GLTO_REFRESH، لإخبار gl_get_line() بإعادة عرض سطر الإدخال.

لاحظ أنه على الرغم من أن وسيط المهلة يتضمن مكون نانو ثانية، إلا أن قليلًا من ساعات الكمبيوتر الحالية لديها دقة أدق من بضعة مللي ثانية، لذا فإن طلب أقل من بضعة مللي ثانية يعادل طلب صفر ثانية على العديد من الأنظمة. إذا كان هذا يمثل مشكلة، فيجب عليك بناء اختيار المهلة الخاص بك على الدقة الفعلية لساعة المضيف (على سبيل المثال، عن طريق استدعاء sysconf(_SC_CLK_TCK)).

لإيقاف المهل، ببساطة استدعِ gl_inactivity_timeout() مع وسيط callback بقيمة 0. يتم تجاهل الوسيط data في هذه الحالة.

الافتراضات الخاصة بمعالجة الإشارات

افتراضيًا، تعترض الدالة gl_get_line() عددًا من الإشارات. هذا مهم بشكل خاص للإشارات التي كانت ستنهي العملية افتراضيًا، حيث تحتاج الطرفية إلى استعادتها إلى حالة قابلة للاستخدام قبل حدوث ذلك. في هذا القسم، يتم وصف الإشارات التي يتم التقاطها افتراضيًا، وكيف تستجيب لها gl_get_line(). تغيير هذه الافتراضات هو موضوع القسم التالي.

عند التقاط المجموعة الفرعية التالية من الإشارات، تقوم gl_get_line() أولاً باستعادة إعدادات الطرفية ومعالجة الإشارات إلى كيف كانت قبل استدعاء gl_get_line()، وتعيد إرسال الإشارة، للسماح لمعالجات الإشارات الخاصة بالتطبيق باستلامها، ثم إذا كانت العملية لا تزال موجودة، تُعيد gl_get_line() القيمة NULL وتعين errno كما هو محدد أدناه.

 SIGINT  -  تُنشأ هذه الإشارة بكل من مفتاح مقاطعة لوحة
            المفاتيح (عادةً ^C)، ومفتاح كسر لوحة المفاتيح.
            errno=EINTR
 SIGHUP  -  تُنشأ هذه الإشارة عند خروج الطرفية المتحكمة.
            errno=ENOTTY
 SIGPIPE -  تُنشأ هذه الإشارة عندما يحاول برنامج الكتابة
            إلى أنبوب لا يتم قراءته من نهايته البعيدة من قبل
            أي عملية. يمكن أن يحدث هذا على سبيل المثال إذا
            كنت قد استدعيت gl_change_terminal()
            لإعادة توجيه الإخراج إلى أنبوب مخفي تحت طرفية
            وهمية.
            errno=EPIPE
 SIGQUIT -  تُنشأ هذه الإشارة بمفتاح إنهاء لوحة المفاتيح
            (عادةً ^\).
            errno=EINTR
 SIGABRT -  تُنشأ هذه الإشارة بواسطة دالة abort()
            القياسية في C. افتراضيًا، تقوم بإنهاء العملية
            وإنشاء تفريغ أساسي.
            errno=EINTR
 SIGTERM -  هذه هي الإشارة الافتراضية التي يرسلها أمر
            kill الخاص بـ UN*X إلى العمليات.
            errno=EINTR

لاحظ أنه في حالة جميع الإشارات المذكورة أعلاه، ينص معيار POSIX على أنه افتراضيًا يتم إنهاء العملية، مع إضافة تفريغ أساسي في حالة إشارة SIGQUIT. بعبارة أخرى، إذا لم يتجاوز التطبيق المستدعي المعالج الافتراضي عن طريق توفير معالج إشارات خاص به، فإن استلام الإشارة المقابلة سينهي التطبيق قبل عودة gl_get_line().

إذا أجهضت gl_get_line() مع تعيين errno إلى EINTR، يمكنك معرفة الإشارة التي سببت إجهاضها، عن طريق استدعاء الدالة التالية.

  int gl_last_signal(const GetLine *gl);

تُعيد هذه الدالة الرمز الرقمي (مثل SIGINT) لآخر إشارة تم استلامها أثناء أحدث استدعاء لـ gl_get_line()، أو -1 إذا لم يتم استلام أي إشارات.

على الأنظمة التي تدعم ذلك، عند استلام إشارة SIGWINCH (تغيير النافذة)، تقوم gl_get_line() بالاستعلام من الطرفية لمعرفة حجمها الجديد، وإعادة رسم سطر الإدخال الحالي لاستيعاب الحجم الجديد، ثم تعود للانتظار لإدخال لوحة المفاتيح من المستخدم. على عكس الإشارات الأخرى، لا يتم إعادة إرسال هذه الإشارة إلى التطبيق.

أخيرًا، تتسبب الإشارات التالية في قيام gl_get_line() أولاً باستعادة البيئة الطرفية والإشارية إلى ما كانت عليه قبل استدعاء gl_get_line()، ثم إعادة إرسال الإشارة إلى التطبيق. إذا كانت العملية لا تزال موجودة بعد تسليم الإشارة، فإن gl_get_line() تعيد بعد ذلك تثبيت معالجات الإشارات الخاصة بها، وتبدل الطرفية مرة أخرى إلى وضعها الخام، وتعيد عرض سطر الإدخال، وتعود لانتظار إدخال الطرفية من المستخدم.

 SIGCONT    -  تُنشأ هذه الإشارة عند استئناف عملية معلقة.
 SIGPOLL    -  على أنظمة SVR4، تخطر هذه الإشارة العملية
               بحدث إدخال/إخراج غير متزامن. لاحظ أنه في
               ظل 4.3+BSD، SIGIO و SIGPOLL متماثلان. على
               أنظمة أخرى، يتم تجاهل SIGIO افتراضيًا،
               لذلك لا تلتقطها gl_get_line()
               افتراضيًا.
 SIGPWR     -  تُنشأ هذه الإشارة عند حدوث انقطاع في التيار
               الكهربائي (يفترض عند تشغيل النظام على UPS).
 SIGALRM    -  تُنشأ هذه الإشارة عند انتهاء صلاحية مؤقت.
 SIGUSR1    -  إشارة خاصة بالتطبيق.
 SIGUSR2    -  إشارة أخرى خاصة بالتطبيق.
 SIGVTALRM  -  تُنشأ هذه الإشارة عند انتهاء صلاحية مؤقت
               افتراضي (راجع man setitimer(2)).
 SIGXCPU    -  تُنشأ هذه الإشارة عندما يتجاوز البرنامج حده
               الناعم لوقت المعالج.
 SIGXFSZ    -  تُنشأ هذه الإشارة عندما يتجاوز البرنامج حده
               الناعم لحجم الملف.
 SIGTSTP    -  تُنشأ هذه الإشارة بمفتاح تعليق الطرفية،
               والذي عادة ما يكون ^Z، أو مفتاح تعليق
               الطرفية المؤجل، والذي عادة ما يكون ^Y.
 SIGTTIN    -  تُنشأ هذه الإشارة إذا حاول البرنامج القراءة
               من الطرفية أثناء تشغيل البرنامج في الخلفية.
 SIGTTOU    -  تُنشأ هذه الإشارة إذا حاول البرنامج الكتابة
               إلى الطرفية أثناء تشغيل البرنامج في الخلفية.

من الواضح أن هذه الإشارات ليست كلها مدعومة على جميع الأنظمة، لذا يتم تجميع الكود لدعمها بشكل مشروط في مكتبة tecla.

لاحظ أنه إذا تم استقبال إشارات SIGKILL أو SIGPOLL، التي لا يمكن التقاطها بحكم تعريفها، أو أي من إشارات استثناءات الأجهزة، مثل SIGSEGV و SIGBUS و SIGFPE، وتم تجاهلها أثناء قيام gl_get_line() بجعل الطرفية في وضع خام، فسيتم إنهاء البرنامج دون استعادة الطرفية إلى حالة قابلة للاستخدام. في الممارسة العملية، تقوم قذائف التحكم في المهام عادةً بإعادة ضبط إعدادات الطرفية عندما يتخلى البرنامج عن الطرفية المتحكمة، لذا فإن هذه مشكلة فقط مع القذائف القديمة.

معالجة الإشارات المخصصة

سرد القسم السابق الإشارات التي تلتقطها gl_get_line() افتراضيًا، ووصف كيف تستجيب لها. يصف هذا القسم كيفية إضافة إشارات إلى قائمة الإشارات الملتقطة وإزالتها، وكيفية تحديد كيفية استجابة gl_get_line() لإشارة معينة.

إذا لم تكن بحاجة إلى قيام gl_get_line() بأي شيء ردًا على إشارة تلتقطها عادةً، فيمكنك إخبار gl_get_line() بتجاهل تلك الإشارة عن طريق استدعاء gl_ignore_signal().

  int gl_ignore_signal(GetLine *gl, int signo);

الوسيط signo هو رقم الإشارة (مثل SIGINT) التي تريد تجاهلها. إذا لم تكن الإشارة المحددة حاليًا من بين تلك التي يتم التقاطها، فإن هذه الدالة لا تفعل شيئًا.

تسمح الدالة gl_trap_signal() لك بإضافة إشارة جديدة إلى القائمة التي تلتقطها gl_get_line()، أو تعديل كيفية استجابتها
