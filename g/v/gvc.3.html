gvc(3)

الحزمة: graphviz

التراخيص: EPL

LIBGVC(3) دليل وظائف المكتبة LIBGVC(3)

الاسم

libgvc - مكتبة سياق Graphviz

الملخص

#include <graphviz/gvc.h>
/* إعداد سياق graphviz */
extern GVC_t *gvNEWcontext(const lt_symlist_t *builtins, int demand_loading);
/*  إعداد سياق graphviz - بديل */
/*     (يغلف الوظيفتين أعلاه باستخدام معلومات مدمجة في libgvc) */
extern GVC_t *gvContext(void);
/* تحليل وسيطات سطر الأوامر - على الأقل argv[0] يحدد محرك التخطيط */
extern int gvParseArgs(GVC_t *gvc, int argc, char **argv);
extern graph_t *gvNextInputGraph(GVC_t *gvc);
/* حساب التخطيط باستخدام محرك محدد */
extern int gvLayout(GVC_t *gvc, graph_t *g, char *engine);
/* حساب التخطيط باستخدام محرك التخطيط من وسيطات سطر الأوامر */
extern int gvLayoutJobs(GVC_t *gvc, graph_t *g);
/* التحقق مما إذا كان التخطيط قد تم */
extern bool gvLayoutDone(graph_t *g);
/* عرض التخطيط في سمات السلسلة للرسم البياني */
extern void attach_attrs(graph_t *g);
/* تحليل سلسلة HTML */
extern char *agstrdup_html(char *s);
extern int aghtmlstr(char *s);
/* عرض التخطيط بتنسيق محدد في ملف مفتوح */
extern int gvRender(GVC_t *gvc, graph_t *g, char *format, FILE *out);
/* عرض التخطيط بتنسيق محدد في ملف مفتوح */
extern int gvRenderFilename(GVC_t *gvc, graph_t *g, char *format, char *filename);
/* عرض التخطيط وفقًا لخيارات -T و -o التي تم العثور عليها بواسطة gvParseArgs */
extern int gvRenderJobs(GVC_t *gvc, graph_t *g);
/* تنظيف هياكل بيانات التخطيط - التخطيطات ليست قابلة للتداخل (حتى الآن) */
extern int gvFreeLayout(GVC_t *gvc, graph_t *g);
/* تنظيف سياق graphviz */
extern int gvFreeContext(GVC_t *gvc);
/* الاستعلام عن المكونات الإضافية المتاحة */
/* انظر التعليق في gvc.h            */
extern char** gvPluginList(GVC_t *gvc, char* kind, int* cnt, char*);

الوصف

libgvc يوفر سياقًا للتطبيقات التي ترغب في معالجة وعرض الرسوم البيانية. يوفر تحليل سطر الأوامر، وشيفرة عرض شائعة، وآلية إضافية للعارضات.

انظر أيضًا

dot(1)، neato(1)، cdt(3)، cgraph(3)
