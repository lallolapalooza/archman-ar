gearman_client_do_background(3)

الحزمة: gearmand

التراخيص: BSD-3-Clause

GEARMAN_CLIENT_DO_BACKGROUND(3) وثائق Gearmand GEARMAN_CLIENT_DO_BACKGROUND(3)

الاسم

gearman_client_do_background - وثائق Gearmand، https://gearman.org/gearmand/

الملخص

#include <libgearman/gearman.h>

النوع gearman_priority_t
gearman_return_t gearman_client_do_background(gearman_client_st *client, const char *function_name, const char *unique, const void *workload, size_t workload_size, char *job_handle)

تغير في الإصدار 0.21: لن يتم إرجاع GEARMAN_PAUSE بعد الآن. سيعمل إجراء "do" الآن حتى يتم إرساله.

gearman_return_t gearman_client_do_high_background(gearman_client_st *client, const char *function_name, const char *unique, const void *workload, size_t workload_size, gearman_job_handle_t job_handle)
gearman_return_t gearman_client_do_low_background(gearman_client_st *client, const char *function_name, const char *unique, const void *workload, size_t workload_size, gearman_job_handle_t job_handle)

الوصف

gearman_client_do_background() تنفذ طلبًا واحدًا إلى خادم gearmand وتعيد الحالة عبر gearman_return_t.

gearman_client_add_task_high_background() و gearman_client_add_task_low_background() متطابقان مع gearman_client_do_background()، فقط أنهما يضبطان gearman_priority_t إلى عالٍ أو منخفض.

إذا لم يكن job_handle فارغًا، فسيتم ملؤه باسم job_handle للمهمة التي تم إنشاؤها. يحتاج job handle إلى حجم GEARMAN_JOB_HANDLE_SIZE. يرجى الرجوع إلى gearman_job_handle_t لمزيد من المعلومات.

gearman_return_t

/*
# خادم Gearman والمكتبة
# حقوق النشر (C) 2012 Data Differential, http://datadifferential.com/
# جميع الحقوق محفوظة.
#
# الاستخدام والتوزيع مرخص بموجب ترخيص BSD. انظر
# ملف COPYING في هذا الدليل للحصول على النص الكامل.
*/
#include <string.h>
#include <stdlib.h>
#include <stdio.h>
#include <libgearman-1.0/gearman.h>
int main(void)
{
  gearman_client_st *client= gearman_client_create(NULL);
  gearman_return_t ret= gearman_client_add_server(client, "localhost", 0);
  if (gearman_failed(ret))
  {
    return EXIT_FAILURE;
  }
  gearman_job_handle_t job_handle;
  gearman_return_t rc= gearman_client_do_background(client,
                                                    "reverse_function",
                                                    "unique_value",
                                                    "my string to reverse", strlen("my string to reverse"),
                                                    job_handle);
  if (gearman_success(rc))
  {
    // استخدم القيمة
    printf("%s\n", job_handle);
  }
  gearman_client_free(client);
  return 0;
}
