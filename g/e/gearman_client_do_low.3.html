gearman_client_do_low(3)

الحزمة: gearmand

التراخيص: BSD-3-Clause

GEARMAN_CLIENT_DO_LOW(3) Gearmand GEARMAN_CLIENT_DO_LOW(3)

الاسم

gearman_client_do_low - توثيق Gearmand، https://gearman.org/gearmand/

الملخص

#include <libgearman/gearman.h>

void *gearman_client_do(gearman_client_st *client, const char *function_name, const char *unique, const void *workload, size_t workload_size, size_t *result_size, gearman_return_t *ret_ptr)

تغير في الإصدار 0.21: لن يتم إرجاع GEARMAN_PAUSE بعد الآن. سيعمل عملية do الآن حتى الاكتمال أو الخطأ.

void *gearman_client_do_high(gearman_client_st *client, const char *function_name, const char *unique, const void *workload, size_t workload_size, size_t *result_size, gearman_return_t *ret_ptr)
void *gearman_client_do_low(gearman_client_st *client, const char *function_name, const char *unique, const void *workload, size_t workload_size, size_t *result_size, gearman_return_t *ret_ptr)

الوصف

gearman_client_do() تنفذ طلبًا واحدًا إلى خادم gearmand وتنتظر ردًا.

gearman_client_do_high() و gearman_client_do_low() مطابقتان لـ gearman_client_do()، فقط أنهما تضبطان الأولوية إما إلى عالية أو منخفضة.

ستقوم جميع الدوال بحظر حتى يتم إرجاع استجابة أو خطأ.

قيمة الإرجاع

gearman_client_do() ترجع مؤشرًا إلى قيمة يجب على المستدعي تحريرها. إذا تم توفير ret_ptr، فسيتم تخزين أي أخطاء حدثت فيه. نظرًا لأن القيمة NULL/صفر هي قيمة صالحة، فستحتاج دائمًا إلى التحقق من ret_ptr إذا كنت مهتمًا بالأخطاء.

/*
# خادم ومكتبة Gearman

# الاستخدام والتوزيع مرخصان بموجب ترخيص BSD. انظر

#include <string.h>
#include <stdlib.h>
#include <libgearman/gearman.h>
int main(void)
{
  gearman_client_st *client= gearman_client_create(NULL);
  gearman_return_t ret= gearman_client_add_server(client, "localhost", 0);
  if (gearman_failed(ret))
  {
    return EXIT_FAILURE;
  }
  size_t result_size;
  gearman_return_t rc;
  void *value= gearman_client_do(client, "reverse_function", "unique_value",
                                 "my string to reverse", strlen("my string to reverse"),
                                 &result_size, &rc);
  if (gearman_success(rc))
  {
    // استخدم القيمة
  }
  free(value);
  gearman_client_free(client);
  return 0;
}
