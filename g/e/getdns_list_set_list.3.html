getdns_list_set_list(3)

الحزمة: getdns

التراخيص: BSD

getdns_list_set(3) getdns getdns_list_set(3)

الاسم

getdns_list_set, getdns_list_set_bindata, getdns_list_set_dict, getdns_list_set_int, getdns_list_set_list, -- تعيين قيمة عن طريق الفهرس من قائمة getdns

المكتبة

مكتبة محلل DNS (libgetdns, -lgetdns)

الملخص

#include <getdns.h>

getdns_return_t
getdns_list_set_bindata (getdns_list *this_list,

size_t index,
getdns_bindata *child_bindata)

getdns_return_t
getdns_list_set_dict (getdns_list *this_list,

size_t index,
getdns_dict *child_dict)

getdns_return_t
getdns_list_set_int (getdns_list *this_list,

size_t index,
uint32_t child_uint32_t)

getdns_return_t
getdns_list_set_list (getdns_list *this_list,

size_t index,
getdns_list *child_list)

الوصف

يُستخدم نوع getdns_list لإدارة قائمة غير متجانسة من القيم المفهرسة التي تشمل:


getdns_bindata
getdns_dict
getdns_list
uint32_t

this_list القائمة التي سيتم فيها إضافة/تحديث العنصر المفهرس

index الفهرس الذي سيتم تعيين القيمة المرتبطة به. إذا كان الفهرس موجودًا في القائمة، يتم استبدال القيمة المرتبطة بهذا الفهرس، وإذا لم يكن الفهرس موجودًا في القائمة، تتم إضافة عنصر جديد إلى القائمة في الفهرس المحدد.

child_bindata child_dict child_list child_uint32 القيمة المراد تعيينها للعنصر المفهرس

قيم الإرجاع

عند اكتمال العملية بنجاح، تُرجع الدوال GETDNS_RETURN_GOOD، وإلا، تُرجع قيم الخطأ التالية:

GETDNS_RETURN_NO_SUCH_LIST_ITEM إذا كان الفهرس خارج النطاق

GETDNS_RETURN_GENERIC_ERROR إذا لم تكن this_list قائمة صالحة
