الملخص

#include <libgearman/gearman.h>

النوع gearman_work_t
النوع gearman_job_priority_t
gearman_work_t gearman_work(gearman_job_priority_t priority)
gearman_work_t gearman_work_epoch(time_t epoch, gearman_job_priority_t priority)
gearman_work_t gearman_work_background(gearman_job_priority_t priority)

قم بالترجمة والربط باستخدام -lgearman

الوصف

gearman_work_t يصف العمل الخاص بـ gearman_execute().

gearman_work() ينشئ gearman_work_t مع أولوية.

gearman_work_epoch() ينشئ gearman_work_t الذي يخبر gearman_execute() بتنفيذ عبء العمل في الوقت المحدد بواسطة epoch.

gearman_work_background() ينشئ gearman_work_t الذي يخبر gearman_execute() بتنفيذ عبء العمل كعملية خلفية.
