الاسم

shadow, getspnam - روتينيات ملف كلمة المرور المشفرة

الصياغة

#include <shadow.h>

struct spwd *getspent();

struct spwd *getspnam(char *name);

void setspent();

void endspent();

struct spwd *fgetspent(FILE *fp);

struct spwd *sgetspent(char *cp);

int putspent(struct spwd *p, FILE *fp);

int lckpwdf();

int ulckpwdf();

الوصف

shadow تتعامل مع محتويات ملف كلمة المرور المشفرة، /etc/shadow. البنية في ملف #include هي:

struct spwd {
      char		*sp_namp; /* اسم تسجيل الدخول للمستخدم */
      char		*sp_pwdp; /* كلمة المرور المشفرة */
      long int		sp_lstchg; /* آخر تغيير لكلمة المرور */
      long int		sp_min; /* عدد الأيام التي لا يمكن تغيير كلمة المرور خلالها. */
      long int		sp_max; /* عدد الأيام قبل وجوب تغيير كلمة المرور */
      long int		sp_warn; /* عدد الأيام للتحذير من انتهاء صلاحية كلمة المرور */
      long int		sp_inact; /* عدد الأيام قبل أن يصبح الحساب غير نشط */
      long int		sp_expire; /* تاريخ انتهاء صلاحية الحساب */
      unsigned long int	sp_flag; /* محجوز للاستخدام المستقبلي */
}

معاني كل حقل هي:

•sp_namp - مؤشر إلى اسم المستخدم المنتهي بصفر
•sp_pwdp - مؤشر إلى كلمة المرور المنتهية بصفر
•sp_lstchg - عدد الأيام منذ 1 يناير 1970 تم تغيير كلمة المرور آخر مرة
•sp_min - عدد الأيام التي لا يمكن فيها تغيير كلمة المرور
•sp_max - عدد الأيام التي يجب بعدها تغيير كلمة المرور
•sp_warn - عدد الأيام قبل انتهاء صلاحية كلمة المرور التي يتم فيها تحذير المستخدم من انتهاء صلاحية كلمة المرور الوشيك
•sp_inact - عدد الأيام بعد انتهاء صلاحية كلمة المرور التي يعتبر فيها الحساب غير نشط ومعطلاً
•sp_expire - عدد الأيام منذ 1 يناير 1970 الذي سيتم فيه تعطيل الحساب
•sp_flag - محجوز للاستخدام المستقبلي

الوصف

getspent، getspname، fgetspent، و sgetspent تعيد كل منها مؤشرًا إلى struct spwd. getspent تعيد الإدخال التالي من الملف، و fgetspent تعيد الإدخال التالي من التدفق المعطى، والذي يفترض أن يكون ملفًا بالتنسيق الصحيح. sgetspent تعيد مؤشرًا إلى struct spwd باستخدام السلسلة الموفرة كمدخل. getspnam تبحث من الموضع الحالي في الملف عن إدخال مطابق لـ name.

setspent و endspent يمكن استخدامهما لبدء وإنهاء، على التوالي، الوصول إلى ملف كلمة المرور المشفرة.

يجب استخدام الروتينيات lckpwdf و ulckpwdf لضمان الوصول الحصري إلى ملف /etc/shadow. تحاول lckpwdf الحصول على قفل باستخدام pw_lock لمدة تصل إلى 15 ثانية. تستمر في محاولة الحصول على قفل ثانٍ باستخدام spw_lock لبقية الـ 15 ثانية الأولية. في حال فشل أي من المحاولتين بعد إجمالي 15 ثانية، تعيد lckpwdf القيمة -1. عند الحصول على كلا القفلين، يتم إرجاع 0.

التشخيصات

تعيد الروتينيات القيمة NULL إذا لم تكن هناك إدخالات أخرى متاحة أو إذا حدث خطأ أثناء المعالجة. الروتينيات التي لها int كقيمة إرجاع تعيد 0 للنجاح و -1 للفشل.

تحذيرات

يجوز فقط للمستخدم الخارق استخدام هذه الروتينيات لأن الوصول إلى ملف كلمة المرور المشفرة مقيد.
