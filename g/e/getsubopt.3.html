getsubopt(3) دليل وظائف المكتبة getsubopt(3)

الاسم

getsubopt — تحليل وسيطات الخيارات الفرعية من سلسلة نصية

المكتبة

مكتبة C القياسية (libc، -lc)

الملخص

#include <stdlib.h>
int getsubopt(char **restrict optionp, char *const *restrict tokens,
              char **restrict valuep);
متطلبات ماكرو اختبار الميزة لـ glibc (انظر feature_test_macros(7)):

getsubopt():

    _XOPEN_SOURCE >= 500
        || /* منذ glibc 2.12: */ _POSIX_C_SOURCE >= 200809L

الوصف

تقوم getsubopt() بتحليل قائمة الخيارات الفرعية المفصولة بفواصل والموجودة في optionp. (عادةً ما يتم إنتاج قائمة خيارات فرعية كهذه عند استخدام getopt(3) لتحليل سطر الأوامر؛ انظر على سبيل المثال الخيار -o الخاص بـ mount(8).) قد يتضمن كل خيار فرعي قيمة مرتبطة به، مفصولة عن اسم الخيار الفرعي بعلامة يساوي. المثال التالي هو نوع السلسلة النصية التي قد يتم تمريرها في optionp:


ro,name=xyz

الوسيط tokens هو مؤشر إلى مصفوفة ذات مؤشرات خالية (NULL-terminated) للرموز (tokens) التي ستبحث عنها getsubopt() في optionp. يجب أن تكون الرموز سلاسل نصية مميزة، ذات نهايات خالية (null-terminated)، تحتوي على حرف واحد على الأقل، بدون علامات يساوي أو فواصل متضمنة.

كل استدعاء لـ getsubopt() يعيد معلومات حول الخيار الفرعي التالي الذي لم تتم معالجته في optionp. يتم تفسير أول علامة يساوي في الخيار الفرعي (إن وجدت) كفاصل بين الاسم وقيمة ذلك الخيار الفرعي. تمتد القيمة إلى الفاصلة التالية، أو (للخيار الفرعي الأخير) إلى نهاية السلسلة النصية. إذا تطابق اسم الخيار الفرعي مع اسم معروف من tokens، وتم العثور على سلسلة قيمة، فإن getsubopt() تقوم بتعيين *valuep إلى عنوان تلك السلسلة. يتم استبدال أول فاصلة في optionp ببايت فارغ، لذا فإن *valuep هو بالضبط "سلسلة القيمة" لذلك الخيار الفرعي.

إذا تم التعرف على الخيار الفرعي، ولكن لم يتم العثور على سلسلة قيمة، يتم تعيين *valuep إلى NULL.

عندما تعود getsubopt()، فإن optionp يشير إلى الخيار الفرعي التالي، أو إلى البايت الفارغ ('\0') في نهاية السلسلة النصية إذا تم للتو معالجة الخيار الفرعي الأخير.

قيمة الإرجاع

إذا تم التعرف على الخيار الفرعي الأول في optionp، فإن getsubopt() تعيد فهرس عنصر الخيار الفرعي المطابق في tokens. بخلاف ذلك، يتم إرجاع -1 ويكون *valuep هو السلسلة النصية الكاملة name[=value].

بما أن *optionp يتم تغييره، فإن الخيار الفرعي الأول قبل الاستدعاء لـ getsubopt() ليس (بالضرورة) هو نفسه الخيار الفرعي الأول بعد getsubopt().

ملاحظات

نظرًا لأن getsubopt() تستبدل أي فواصل تجدها في السلسلة *optionp، يجب أن تكون هذه السلسلة قابلة للكتابة؛ لا يمكن أن تكون ثابت سلسلة نصية.

أمثلة

يتوقع البرنامج التالي الخيارات الفرعية التي تتبع الخيار "-o".

#define _XOPEN_SOURCE 500
#include <stdio.h>
#include <stdlib.h>
#include <assert.h>
int
main(int argc, char *argv[])
{
    enum {
        RO_OPT = 0,
        RW_OPT,
        NAME_OPT
    };
    char *const token[] = {
        [RO_OPT]   = "ro",
        [RW_OPT]   = "rw",
        [NAME_OPT] = "name",
        NULL
    };
    char *subopts;
    char *value;
    int opt;
    int readonly = 0;
    int readwrite = 0;
    char *name = NULL;
    int errfnd = 0;
    while ((opt = getopt(argc, argv, "o:")) != -1) {
        switch (opt) {
        case 'o':
            subopts = optarg;
            while (*subopts != '\0' && !errfnd) {
                switch (getsubopt(&subopts, token, &value)) {
                case RO_OPT:
                    readonly = 1;
                    break;
                case RW_OPT:
                    readwrite = 1;
                    break;
                case NAME_OPT:
                    if (value == NULL) {
                        fprintf(stderr,
                                "Missing value for suboption '%s'\n",
                                token[NAME_OPT]);
                        errfnd = 1;
                        continue;
                    }
                    name = value;
                    break;
                default:
                    fprintf(stderr,
                            "No match found for token: /%s/\n", value);
                    errfnd = 1;
                    break;
                }
            }
            if (readwrite && readonly) {
                fprintf(stderr,
                        "Only one of '%s' and '%s' can be specified\n",
                        token[RO_OPT], token[RW_OPT]);
                errfnd = 1;
            }
            break;
        default:
            errfnd = 1;
        }
    }
    if (errfnd || argc == 1) {
        fprintf(stderr, "\nUsage: %s -o <suboptstring>\n", argv[0]);
        fprintf(stderr,
                "suboptions are 'ro', 'rw', and 'name=<value>'\n");
        exit(EXIT_FAILURE);
    }
    /* Remainder of program...  */
    exit(EXIT_SUCCESS);
}
