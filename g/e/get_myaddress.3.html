void clnt_pcreateerror(const char *s);
طباعة رسالة إلى الخطأ القياسي تشير إلى سبب عدم القدرة على إنشاء مقبض عميل RPC. تتم إضافة السلسلة s ونقطتين رأسيتين إلى بداية الرسالة. تستخدم عندما تفشل استدعاءات clnt_create() أو clntraw_create() أو clnttcp_create() أو clntudp_create().
void clnt_perrno(enum clnt_stat stat);
طباعة رسالة إلى الخطأ القياسي تتوافق مع الحالة التي تشير إليها stat. تستخدم بعد callrpc().
clnt_perror(CLIENT *clnt, const char *s);
طباعة رسالة إلى الخطأ القياسي تشير إلى سبب فشل استدعاء RPC؛ clnt هو المقبض المستخدم لإجراء الاستدعاء. تتم إضافة السلسلة s ونقطتين رأسيتين إلى بداية الرسالة. تستخدم بعد clnt_call().
char *clnt_spcreateerror(const char *s);
مثل clnt_pcreateerror()، باستثناء أنها تعيد سلسلة بدلاً من الطباعة إلى الخطأ القياسي.
أخطاء: تعيد مؤشرًا إلى بيانات ثابتة يتم الكتابة فوقها في كل استدعاء.
char *clnt_sperrno(enum clnt_stat stat);
تأخذ نفس وسائط clnt_perrno()، ولكن بدلاً من إرسال رسالة إلى الخطأ القياسي تشير إلى سبب فشل استدعاء RPC، تعيد مؤشرًا إلى سلسلة تحتوي على الرسالة. تنتهي السلسلة بسطر جديد.
يتم استخدام clnt_sperrno() بدلاً من clnt_perrno() إذا كان البرنامج لا يحتوي على خطأ قياسي (كما هو الحال غالبًا مع برنامج يعمل كخادم)، أو إذا كان المبرمج لا يريد إخراج الرسالة باستخدام printf(3)، أو إذا كان تنسيق الرسالة مختلفًا عن ذلك الذي تدعمه clnt_perrno(). ملاحظة: على عكس clnt_sperror() و clnt_spcreateerror()، تعيد clnt_sperrno() مؤشرًا إلى بيانات ثابتة، ولكن النتيجة لن يتم الكتابة فوقها في كل استدعاء.
char *clnt_sperror(CLIENT *rpch, const char *s);
مثل clnt_perror()، باستثناء أنها (مثل clnt_sperrno()) تعيد سلسلة بدلاً من الطباعة إلى الخطأ القياسي.
أخطاء: تعيد مؤشرًا إلى بيانات ثابتة يتم الكتابة فوقها في كل استدعاء.
CLIENT *clntraw_create(unsigned long prognum, unsigned long versnum);
يقوم هذا الإجراء بإنشاء عميل RPC وهمي للبرنامج البعيد prognum، الإصدار versnum. النقل المستخدم لتمرير الرسائل إلى الخدمة هو في الواقع مخزن مؤقت داخل مساحة عنوان العملية، لذا يجب أن يعيش خادم RPC المقابل في نفس مساحة العنوان؛ انظر svcraw_create(). هذا يسمح بمحاكاة RPC واكتساب نفقات RPC، مثل أوقات الذهاب والإياب، دون أي تدخل من النواة. يعيد هذا الإجراء NULL إذا فشل.
CLIENT *clnttcp_create(struct sockaddr_in *addr,
                unsigned long prognum, unsigned long versnum,
                int *sockp, unsigned int sendsz, unsigned int recvsz);
يقوم هذا الإجراء بإنشاء عميل RPC للبرنامج البعيد prognum، الإصدار versnum؛ يستخدم العميل TCP/IP كوسيلة نقل. يقع البرنامج البعيد في عنوان الإنترنت *addr. إذا كان addr->sin_port صفرًا، فسيتم تعيينه على المنفذ الفعلي الذي يستمع إليه البرنامج البعيد (يتم استشارة خدمة portmap البعيدة لهذه المعلومات). المعلمة sockp هي مقبس؛ إذا كانت RPC_ANYSOCK، فسيفتح هذا الإجراء مقبسًا جديدًا ويضبط sockp. نظرًا لأن RPC المستند إلى TCP يستخدم الإدخال/الإخراج المؤقت، يمكن للمستخدم تحديد حجم المخازن المؤقتة للإرسال والاستقبال باستخدام المعلمات sendsz و recvsz؛ القيم الصفرية تختار القيم الافتراضية المناسبة. يعيد هذا الإجراء NULL إذا فشل.
CLIENT *clntudp_create(struct sockaddr_in *addr,
                unsigned long prognum, unsigned long versnum,
                struct timeval wait, int *sockp);
يقوم هذا الإجراء بإنشاء عميل RPC للبرنامج البعيد prognum، الإصدار versnum؛ يستخدم العميل UDP/IP كوسيلة نقل. يقع البرنامج البعيد في عنوان الإنترنت addr. إذا كان addr->sin_port صفرًا، فسيتم تعيينه على المنفذ الفعلي الذي يستمع إليه البرنامج البعيد (يتم استشارة خدمة portmap البعيدة لهذه المعلومات). المعلمة sockp هي مقبس؛ إذا كانت RPC_ANYSOCK، فسيفتح هذا الإجراء مقبسًا جديدًا ويضبط sockp. يعيد نقل UDP إرسال رسالة الاستدعاء بشكل متقطع بفواصل زمنية قدرها wait حتى يتم استلام استجابة أو حتى ينتهي وقت استدعاء. يتم تحديد الوقت الإجمالي لانتهاء مهلة الاستدعاء بواسطة clnt_call().
تحذير: نظرًا لأن رسائل RPC المستندة إلى UDP يمكن أن تحمل ما يصل إلى 8 كيلوبايت من البيانات المشفرة فقط، لا يمكن استخدام هذا النقل للإجراءات التي تأخذ وسائط كبيرة أو تعيد نتائج ضخمة.
CLIENT *clntudp_bufcreate(struct sockaddr_in *addr,
            unsigned long prognum, unsigned long versnum,
            struct timeval wait, int *sockp,
            unsigned int sendsize, unsigned int recosize);
يقوم هذا الإجراء بإنشاء عميل RPC للبرنامج البعيد prognum، على versnum؛ يستخدم العميل UDP/IP كوسيلة نقل. يقع البرنامج البعيد في عنوان الإنترنت addr. إذا كان addr->sin_port صفرًا، فسيتم تعيينه على المنفذ الفعلي الذي يستمع إليه البرنامج البعيد (يتم استشارة خدمة portmap البعيدة لهذه المعلومات). المعلمة sockp هي مقبس؛ إذا كانت RPC_ANYSOCK، فسيفتح هذا الإجراء مقبسًا جديدًا ويضبط sockp. يعيد نقل UDP إرسال رسالة الاستدعاء بشكل متقطع بفواصل زمنية قدرها wait حتى يتم استلام استجابة أو حتى ينتهي وقت استدعاء. يتم تحديد الوقت الإجمالي لانتهاء مهلة الاستدعاء بواسطة clnt_call().
هذا يسمح للمستخدم بتحديد الحد الأقصى لحجم الحزمة لإرسال واستقبال رسائل RPC المستندة إلى UDP.
void get_myaddress(struct sockaddr_in *addr);
ضع عنوان IP للجهاز في *addr، دون استشارة روتينات المكتبة التي تتعامل مع /etc/hosts. يتم دائمًا تعيين رقم المنفذ على htons(PMAPPORT).
struct pmaplist *pmap_getmaps(struct sockaddr_in *addr);
واجهة مستخدم لخدمة portmap، والتي تعيد قائمة خرائط البرامج RPC الحالية إلى المنافذ على المضيف الموجود في عنوان IP *addr. يمكن لهذا الإجراء إرجاع NULL. يستخدم الأمر rpcinfo -p هذا الإجراء.
unsigned short pmap_getport(struct sockaddr_in *addr,
                    unsigned long prognum, unsigned long versnum,
                    unsigned int protocol);
واجهة مستخدم لخدمة portmap، والتي تعيد رقم المنفذ الذي تنتظر عليه خدمة تدعم رقم البرنامج prognum، الإصدار versnum، وتتحدث بروتوكول النقل المرتبط بـ protocol. قيمة protocol هي على الأرجح IPPROTO_UDP أو IPPROTO_TCP. قيمة إرجاع صفر تعني أن الخريطة غير موجودة أو أن نظام RPC فشل في الاتصال بخدمة portmap البعيدة. في الحالة الأخيرة، تحتوي المتغير العام rpc_createerr على حالة RPC.
enum clnt_stat pmap_rmtcall(struct sockaddr_in *addr,
                    unsigned long prognum, unsigned long versnum,
                    unsigned long procnum,
                    xdrproc_t inproc, char *in,
                    xdrproc_t outproc, char *out,
                    struct timeval tout, unsigned long *portp);
واجهة مستخدم لخدمة portmap، والتي تطلب من portmap على المضيف في عنوان IP *addr إجراء استدعاء RPC نيابة عنك إلى إجراء على هذا المضيف. سيتم تعديل المعلمة *portp إلى رقم منفذ البرنامج إذا نجح الإجراء. تتم مناقشة تعريفات المعلمات الأخرى في callrpc() و clnt_call(). يجب استخدام هذا الإجراء من أجل "ping" ولا شيء آخر. انظر أيضًا clnt_broadcast().
bool_t pmap_set(unsigned long prognum, unsigned long versnum,
                int protocol, unsigned short port);
واجهة مستخدم لخدمة portmap، والتي تنشئ خريطة بين الثلاثي [prognum,versnum,protocol] و port على خدمة portmap للجهاز. قيمة protocol هي على الأرجح IPPROTO_UDP أو IPPROTO_TCP. يعيد هذا الإجراء واحدًا إذا نجح، وصفرًا بخلاف ذلك. يتم تلقائيًا بواسطة svc_register().
bool_t pmap_unset(unsigned long prognum, unsigned long versnum);
واجهة مستخدم لخدمة portmap، والتي تدمر جميع الخرائط بين الثلاثي [prognum,versnum,*] و ports على خدمة portmap للجهاز. يعيد هذا الإجراء واحدًا إذا نجح، وصفرًا بخلاف ذلك.
int registerrpc(unsigned long prognum, unsigned long versnum,
                unsigned long procnum, typeof(char *(char *)) *procname,
                xdrproc_t inproc, xdrproc_t outproc);
تسجيل الإجراء procname مع حزمة خدمة RPC. إذا وصل طلب للبرنامج prognum، الإصدار versnum، والإجراء procnum، يتم استدعاء procname بمؤشر إلى وسيطه (وسائطه)؛ يجب أن يعيد procname مؤشرًا إلى نتيجته (نتائجه) الثابتة؛ يتم استخدام inproc لفك ترميز الوسائط بينما يتم استخدام outproc لتشفير النتائج. يعيد هذا الإجراء صفرًا إذا نجح التسجيل، و -1 بخلاف ذلك.
