توثيق الوظائف

subroutine cgeesx (character jobvs, character sort, external select, character sense, integer n, complex, dimension( lda, * ) a, integer lda, integer sdim, complex, dimension( * ) w, complex, dimension( ldvs, * ) vs, integer ldvs, real rconde, real rcondv, complex, dimension( * ) work, integer lwork, real, dimension( * ) rwork, logical, dimension( * ) bwork, integer info)

cgeesx يحسب القيم الذاتية، شكل شور، واختيارياً، مصفوفة متجهات شور لمصفوفات GE

الغرض:

!>
!> CGEESX يحسب لمصفوفة معقدة غير متماثلة N-by-N A،
!> القيم الذاتية، شكل شور T، واختيارياً، مصفوفة متجهات شور Z.  هذا يعطي تفكيك شور A = Z*T*(Z**H).
!>
!> اختيارياً، يقوم أيضاً بترتيب القيم الذاتية على القطر لشكل شور بحيث تكون القيم الذاتية المحددة في أعلى اليسار؛
!> يحسب مقلوب رقم شرط لمتوسط القيم الذاتية المحددة (RCONDE)؛ ويحسب مقلوب رقم شرط
!> للفضاء المتجهي الأيمن الموافق للقيم الذاتية المحددة (RCONDV).  الأعمدة الرائدة لـ Z تشكل
!> أساساً متعامداً لهذا الفضاء المتجهي.
!>
!> لمزيد من التوضيح حول مقلوب أرقام الشرط RCONDE و RCONDV، انظر القسم 4.10 من دليل مستخدمي LAPACK (حيث
!> تسمى هذه الكميات s و sep على التوالي).
!>
!> تكون المصفوفة المعقدة في شكل شور إذا كانت علوية مثلثة.
!>

المعلمات

JOBVS 
!>          JOBVS هو CHARACTER*1
!>          = 'N': لا يتم حساب متجهات شور؛
!>          = 'V': يتم حساب متجهات شور.
!>

SORT

!>          SORT هو CHARACTER*1
!>          يحدد ما إذا كان سيتم ترتيب القيم الذاتية على قطر شكل شور أم لا.
!>          = 'N': لا يتم ترتيب القيم الذاتية؛
!>          = 'S': يتم ترتيب القيم الذاتية (انظر SELECT).
!>

SELECT

!>          SELECT هي وظيفة LOGICAL لحجة COMPLEX واحدة
!>          يجب تعريف SELECT كـ EXTERNAL في الروتين الفرعي المستدعي.
!>          إذا كان SORT = 'S'، يتم استخدام SELECT لتحديد القيم الذاتية لترتيبها
!>          إلى أعلى اليسار من شكل شور.
!>          إذا كان SORT = 'N'، لا يتم الاستشهاد بـ SELECT.
!>          يتم تحديد القيمة الذاتية W(j) إذا كانت SELECT(W(j)) صحيحة.
!>

SENSE

!>          SENSE هو CHARACTER*1
!>          يحدد مقلوب أرقام الشروط التي سيتم حسابها.
!>          = 'N': لا يتم حساب أي منها؛
!>          = 'E': يتم حسابه لمتوسط القيم الذاتية المحددة فقط؛
!>          = 'V': يتم حسابه للفضاء المتجهي الأيمن المحدد فقط؛
!>          = 'B': يتم حسابه لكليهما.
!>          إذا كان SENSE = 'E'، 'V' أو 'B'، يجب أن يكون SORT = 'S'.
!>

N

!>          N هو INTEGER
!>          ترتيب المصفوفة A. N >= 0.
!>

A

!>          A هي مصفوفة COMPLEX، حجمها (LDA, N)
!>          عند الدخول، المصفوفة N-by-N A.
!>          عند الخروج، يتم استبدال A بشكل شور T.
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرائد للمصفوفة A.  LDA >= max(1,N).
!>

SDIM

!>          SDIM هو INTEGER
!>          إذا كان SORT = 'N'، فإن SDIM = 0.
!>          إذا كان SORT = 'S'، فإن SDIM = عدد القيم الذاتية التي
!>                         تكون SELECT لها صحيحة.
!>

W

!>          W هي مصفوفة COMPLEX، حجمها (N)
!>          تحتوي W على القيم الذاتية المحسوبة، بنفس الترتيب
!>          الذي تظهر به على القطر لشكل شور الناتج T.
!>

VS

!>          VS هي مصفوفة COMPLEX، حجمها (LDVS,N)
!>          إذا كان JOBVS = 'V'، فإن VS تحتوي على المصفوفة الوحدوية Z لمتجهات شور.
!>          إذا كان JOBVS = 'N'، لا يتم الاستشهاد بـ VS.
!>

LDVS

!>          LDVS هو INTEGER
!>          البعد الرائد للمصفوفة VS.  LDVS >= 1، وإذا
!>          كان JOBVS = 'V'، فإن LDVS >= N.
!>

RCONDE

!>          RCONDE هو REAL
!>          إذا كان SENSE = 'E' أو 'B'، فإن RCONDE يحتوي على مقلوب
!>          رقم الشرط لمتوسط القيم الذاتية المحددة.
!>          لا يتم الاستشهاد به إذا كان SENSE = 'N' أو 'V'.
!>

RCONDV

!>          RCONDV هو REAL
!>          إذا كان SENSE = 'V' أو 'B'، فإن RCONDV يحتوي على مقلوب
!>          رقم الشرط للفضاء المتجهي الأيمن المحدد.
!>          لا يتم الاستشهاد به إذا كان SENSE = 'N' أو 'E'.
!>

WORK

!>          WORK هي مصفوفة COMPLEX، حجمها (MAX(1,LWORK))
!>          عند الخروج، إذا كان INFO = 0، فإن WORK(1) يعيد LWORK الأمثل.
!>

LWORK

!>          LWORK هو INTEGER
!>          بعد المصفوفة WORK.  LWORK >= max(1,2*N).
!>          أيضاً، إذا كان SENSE = 'E' أو 'V' أو 'B'، فإن LWORK >= 2*SDIM*(N-SDIM)،
!>          حيث SDIM هو عدد القيم الذاتية المحددة التي تم حسابها بواسطة
!>          هذا الروتين.  لاحظ أن 2*SDIM*(N-SDIM) <= N*N/2. لاحظ أيضاً
!>          أن الخطأ يتم إرجاعه فقط إذا كان LWORK < max(1,2*N)، ولكن إذا
!>          كان SENSE = 'E' أو 'V' أو 'B' فقد لا يكون هذا كافياً.
!>          لأداء جيد، يجب أن يكون LWORK أكبر بشكل عام.
!>
!>          إذا كان LWORK = -1، فيفترض أن يتم الاستعلام عن مساحة العمل؛ يقوم الروتين
!>          فقط بحساب الحد العلوي للحجم الأمثل للمصفوفة
!>          WORK، ويعيد هذه القيمة كإدخال أول للمصفوفة
!>          WORK، ولا يتم إصدار أي رسالة خطأ متعلقة بـ LWORK بواسطة
!>          XERBLA.
!>

RWORK

!>          RWORK هي مصفوفة REAL، حجمها (N)
!>

BWORK

!>          BWORK هي مصفوفة LOGICAL، حجمها (N)
!>          لا يتم الاستشهاد بها إذا كان SORT = 'N'.
!>

INFO

!>          INFO هو INTEGER
!>          = 0: خروج ناجح
!>          < 0: إذا كان INFO = -i، فإن الوسيط i كان له قيمة غير قانونية.
!>          > 0: إذا كان INFO = i، و i هو
!>             <= N: فشلت خوارزمية QR في حساب جميع
!>                   القيم الذاتية؛ العناصر 1:ILO-1 و i+1:N من W
!>                   تحتوي على تلك القيم الذاتية التي تقاربت؛ إذا
!>                   كان JOBVS = 'V'، فإن VS يحتوي على التحويل الذي
!>                   يقلل A إلى شكله الجزئي المتقارب؛
!>             = N+1: لا يمكن إعادة ترتيب القيم الذاتية لأن بعض
!>                   القيم الذاتية كانت قريبة جداً من الانفصال (المشكلة
!>                   غير مشروطة بشدة)؛
!>             = N+2: بعد إعادة الترتيب، غير التقريب قيم بعض
