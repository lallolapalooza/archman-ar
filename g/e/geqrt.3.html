subroutine cgeqrt (integer m, integer n, integer nb, complex, dimension( lda, * ) a, integer lda, complex, dimension( ldt, * ) t, integer ldt, complex, dimension( * ) work, integer info)

CGEQRT

الغرض:

!>
!> CGEQRT تحسب عامل QR محظور لمصفوفة A المركبة M-by-N
!> باستخدام تمثيل WY المدمج لـ Q.
!>

المعاملات

M 
!>          M هو عدد صحيح (INTEGER)
!>          عدد صفوف المصفوفة A.  M >= 0.
!>

N

!>          N هو عدد صحيح (INTEGER)
!>          عدد أعمدة المصفوفة A.  N >= 0.
!>

NB

!>          NB هو عدد صحيح (INTEGER)
!>          حجم الكتلة المستخدم في QR المحظور.  MIN(M,N) >= NB >= 1.
!>

A

!>          A هي مصفوفة مركبة (COMPLEX)، أبعادها (LDA,N)
!>          عند الإدخال، المصفوفة A بحجم M-by-N.
!>          عند الخروج، العناصر على القطر وما فوقه في المصفوفة
!>          تحتوي على المصفوفة شبه المنحرفة العلوية R بحجم min(M,N)-by-N (R
!>          هي مصفوفة مثلثية علوية إذا كان M >= N)؛ العناصر الموجودة أسفل القطر
!>          هي أعمدة V.
!>

LDA

!>          LDA هو عدد صحيح (INTEGER)
!>          البعد الرائد للمصفوفة A.  LDA >= max(1,M).
!>

T

!>          T هي مصفوفة مركبة (COMPLEX)، أبعادها (LDT,MIN(M,N))
!>          تعكس الكتل المثلثية العلوية مخزنة بالشكل المدمج
!>          كسلسلة من الكتل المثلثية العلوية.  انظر أدناه
!>          لمزيد من التفاصيل.
!>

LDT

!>          LDT هو عدد صحيح (INTEGER)
!>          البعد الرائد للمصفوفة T.  LDT >= NB.
!>

WORK

!>          WORK هي مصفوفة مركبة (COMPLEX)، أبعادها (NB*N)
!>

INFO

!>          INFO هو عدد صحيح (INTEGER)
!>          = 0:  الخروج بنجاح
!>          < 0:  إذا كان INFO = -i، فإن الوسيط i كان له قيمة غير قانونية
!>

تفاصيل إضافية:

!>
!>  المصفوفة V تخزن الانعكاسات الأولية H(i) في العمود i
!>  أسفل القطر. على سبيل المثال، إذا كان M=5 و N=3، فإن المصفوفة V هي
!>
!>               V = (  1       )
!>                   ( v1  1    )
!>                   ( v1 v2  1 )
!>                   ( v1 v2 v3 )
!>                   ( v1 v2 v3 )
!>
!>  حيث تمثل vi's المتجهات التي تحدد H(i)، والتي يتم إرجاعها
!>  في المصفوفة A.  القيم 1 الموجودة على طول قطر V لا يتم تخزينها في A.
!>
!>  لتكن K=MIN(M,N).  عدد الكتل هو B = سقف(K/NB)، حيث كل
!>  كتلة من الرتبة NB باستثناء الكتلة الأخيرة، والتي تكون من الرتبة
!>  IB = K - (B-1)*NB.  لكل من الكتل B، يتم حساب عامل مصفوفة مثلثية علوية
!>  خاصة بالكتلة: T1, T2, ..., TB.  يتم تخزين T's بحجم NB-by-NB (و IB-by-IB
!>  للأخيرة) في المصفوفة T بحجم NB-by-K كـ
!>
!>               T = (T1 T2 ... TB).
!>
