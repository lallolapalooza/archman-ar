getdns_dict_set_bindata(3)

الحزمة: getdns

التراخيص: BSD

getdns_dict_set(3) getdns getdns_dict_set(3)

الاسم

getdns_dict_set, getdns_dict_set_bindata, getdns_dict_set_dict, getdns_dict_set_int, getdns_dict_set_list, -- تعيين قيمة بالاسم في قاموس getdns

المكتبة

مكتبة محلل DNS (libgetdns, -lgetdns)

الملخص

#include <getdns.h>

getdns_return_t
getdns_dict_set_bindata (getdns_dict *this_dict,

char *name,
getdns_bindata *child_bindata)

getdns_return_t
getdns_dict_set_dict (getdns_dict *this_dict,

char *name,
getdns_dict *child_dict)

getdns_return_t
getdns_dict_set_int (getdns_dict *this_dict,

char *name,
uint32_t child_uint32_t)

getdns_return_t
getdns_dict_set_list (getdns_dict *this_dict,

char *name,
getdns_list *child_list)

الوصف

يُستخدم نوع getdns_dict لإدارة أزواج الاسم/القيمة التي تكون فيها الأسماء سلاسل نصية وأنواع بيانات القيم غير متجانسة وتشمل


getdns_bindata
getdns_dict
getdns_list
uint32_t

this_dict القاموس الذي سيتم فيه إضافة/تحديث زوج الاسم/القيمة

name الاسم الذي سيتم تعيين القيمة المرتبطة به. إذا كان الاسم موجودًا في القاموس، فسيتم استبدال القيمة المرتبطة بهذا الاسم. إذا لم يكن الاسم موجودًا في القاموس، فسيتم إضافة عنصر جديد إلى القاموس.

child_bindata child_dict child_list child_uint32 القيمة التي سيتم تعيينها للاسم

قيم الإرجاع

عند اكتمال التنفيذ بنجاح، تُرجع الدوال GETDNS_RETURN_GOOD، وإلا تُرجع قيم الأخطاء التالية:

GETDNS_RETURN_GENERIC_ERROR إذا لم يكن this_dict قاموسًا صالحًا
