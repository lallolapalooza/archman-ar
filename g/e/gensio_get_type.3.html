gensio_get_type(3)

الحزمة: gensio

التراخيص: GPL

الاسم

الملخص

unsigned int depth);

unsigned int depth);

bool gensio_is_reliable(struct gensio *io);
bool gensio_is_packet(struct gensio *io);
bool gensio_is_authenticated(struct gensio *io);
bool gensio_is_encrypted(struct gensio *io);

الوصف

gensio_get_type ترجع سلسلة النوع الخاصة بـ gensio (إذا كان depth يساوي 0) أو أحد أبنائها (depth > 0). ترجع NULL إذا كان العمق أكبر من عدد الأبناء.

gensio_get_child ترجع الابن المعطى لـ gensio، ترجع NULL إذا كان depth أكبر من عدد الأبناء.

gensio_is_client ترجع true إذا كان gensio عميلاً أو false إذا كان خادماً.

gensio_is_reliable ترجع true إذا كان gensio موثوقًا به (لن يفقد البيانات). الأجهزة التسلسلية لا تعتبر موثوقة، وكذلك UDP. كل شيء آخر تقريبًا موثوق به، لأنه لن يفقد البيانات ويتم التحكم في التدفق.

gensio_is_packet ترجع إذا كان gensio موجهاً للحزم. في gensio موجه للحزم، إذا كتب جانب ما قطعة من البيانات، فعندما يقوم الجانب الآخر بالقراءة، سيحصل على نفس قطعة البيانات كوحدة واحدة بافتراض أن أحجام مخزنه مؤقتاً مضبوطة بشكل صحيح.

gensio_is_authenticated ترجع true إذا تم مصادقة الطرف البعيد، false إذا لم يتم. في حالة SSL، هذا يعني أنه تم استلام الشهادة البعيدة والتحقق منها (العميل والخادم، انظر قسم SSL في gensio(5) للحصول على التفاصيل). بالنسبة لخادم certauth، هذا يعني أن المصادقة قد نجحت في مصادقة عميل certauth إما بواسطة الشهادة أو كلمة المرور.

gensio_is_encrypted ترجع true إذا كان الاتصال مشفراً، false إذا لم يكن كذلك.

gensio_is_message ترجع إذا كان gensio موجهاً للرسائل. في gensio موجه للرسائل، يمكن للكتابة أن تحدد صراحة حدود الرسائل، ويتم تسليم هذه الحدود للقارئ مع تعيين هذه الحدود. انظر gensio(5) للحصول على تفاصيل حول كيفية عمل ذلك.

gensio_is_mux ترجع true إذا كان gensio قادراً على تعدد الإرسال، مما يعني أنه يقبل استدعاء gensio_alloc_channel().

لاحظ أنه بالنسبة لطلبات نوع "gensio_is_xxx"، فإن الخاصية عادة ما تُورث من الابن. لذلك إذا قمت بتشغيل gensio telnet فوق gensio SSL، فإن gensio telnet سيرجع true لـ gensio_is_encrypted لأن SSL مشفر.
