gnutls_x509_trust_list_get_issuer_by_subject_key_id(3)

الحزمة: gnutls

التراخيص: GPL-3.0-or-later AND LGPL-2.1-or-later

gnutls_x509_trust_list_get_issuer_by_subject_key_id(3) gnutls gnutls_x509_trust_list_get_issuer_by_subject_key_id(3)

الاسم

gnutls_x509_trust_list_get_issuer_by_subject_key_id - وظيفة API

الملخص

#include <gnutls/x509.h>

int gnutls_x509_trust_list_get_issuer_by_subject_key_id(gnutls_x509_trust_list_t list, const gnutls_datum_t * dn, const gnutls_datum_t * spki, gnutls_x509_crt_t * issuer, unsigned int flags);

الحجج

gnutls_x509_trust_list_t list
القائمة
const gnutls_datum_t * dn
هو اسم جهة الإصدار (قد يكون NULL)
const gnutls_datum_t * spki
هو معرف مفتاح الموضوع
gnutls_x509_crt_t * issuer
سيحتوي على جهة الإصدار إن وجدت. يجب إلغاء تخصيصه بعد الاستخدام.
unsigned int flags
استخدم صفر

الوصف

ستجد هذه الوظيفة جهة الإصدار بالاسم ومعرف مفتاح الموضوع المحدد، وستعيد نسخة من جهة الإصدار، والتي يجب تحريرها باستخدام gnutls_x509_crt_deinit().

الإرجاع

عند النجاح، سيتم إرجاع GNUTLS_E_SUCCESS (0)، وإلا سيتم إرجاع قيمة خطأ سالبة.

منذ

3.4.2
