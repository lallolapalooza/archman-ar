gnutls_x509_crt_get_pk_gost_raw(3)

الحزمة: gnutls

gnutls_x509_crt_get_pk_gost_raw(3) gnutls gnutls_x509_crt_get_pk_gost_raw(3)

الاسم

gnutls_x509_crt_get_pk_gost_raw - دالة API

الملخص

#include <gnutls/x509.h>

int gnutls_x509_crt_get_pk_gost_raw(gnutls_x509_crt_t crt, gnutls_ecc_curve_t * curve, gnutls_digest_algorithm_t * digest, gnutls_gost_paramset_t * paramset, gnutls_datum_t * x, gnutls_datum_t * y);

المعاملات

gnutls_x509_crt_t crt
يحمل الشهادة
gnutls_ecc_curve_t * curve
سيحمل المنحنى
gnutls_digest_algorithm_t * digest
سيحمل التجزئة
gnutls_gost_paramset_t * paramset
سيحمل معرف مجموعة معلمات GOST
gnutls_datum_t * x
سيحمل الإحداثي السيني
gnutls_datum_t * y
سيحمل الإحداثي الصادي

الوصف

ستقوم هذه الدالة بتصدير معلمات المفتاح العام GOST الموجودة في الشهادة المقدمة. سيتم تخصيص المعلمات الجديدة باستخدام gnutls_malloc() وسيتم تخزينها في بيانات الاعتماد المناسبة.

القيم المعادة

GNUTLS_E_SUCCESS عند النجاح، وإلا فإن رمز خطأ سالب.

منذ

3.6.3
