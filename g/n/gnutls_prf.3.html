gnutls_prf(3)

الاسم

gnutls_prf - دالة API

الملخص

#include <gnutls/gnutls.h>

int gnutls_prf(gnutls_session_t session, size_t label_size, const char * label, int server_random_first, size_t extra_size, const char * extra, size_t outsize, char * out);

الوسائط

gnutls_session_t session
هو نوع gnutls_session_t.
size_t label_size
طول المتغير label.
const char * label
التسمية المستخدمة في حساب PRF، عادةً ما تكون سلسلة نصية قصيرة.
int server_random_first
غير صفر إذا كان حقل العشوائية الخاص بالخادم يجب أن يكون الأول في البذرة
size_t extra_size
طول المتغير extra.
const char * extra
بيانات إضافية اختيارية لتغذية PRF بها.
size_t outsize
حجم المخزن المؤقت للإخراج المخصص مسبقًا لاحتواء المخرجات.
char * out
المخزن المؤقت المخصص مسبقًا لاحتواء البيانات التي تم إنشاؤها.

الوصف

تطبق دالة TLS Pseudo-Random-Function (PRF) على السر الرئيسي والبيانات المقدمة، ويتم تغذيتها بحقول العشوائية الخاصة بالعميل والخادم. لتوسيع المفتاح المحدد في RFC5705 انظر gnutls_prf_rfc5705().

يحتوي المتغير label عادةً على سلسلة نصية تشير إلى الغرض من البيانات التي تم إنشاؤها. تشير server_random_first إلى ما إذا كان حقل العشوائية الخاص بالعميل أو حقل العشوائية الخاص بالخادم يجب أن يكون الأول في البذرة. يشير غير الصفر إلى أن حقل العشوائية الخاص بالخادم هو الأول، و 0 يشير إلى أن حقل العشوائية الخاص بالعميل هو الأول.

يمكن استخدام المتغير extra لإضافة المزيد من البيانات إلى البذرة، بعد المتغيرات العشوائية. يمكن استخدامه للتأكد من أن المخرجات التي تم إنشاؤها مرتبطة بقوة ببعض البيانات الإضافية (على سبيل المثال، سلسلة نصية مستخدمة في مصادقة المستخدم).

يتم وضع المخرجات في out، والذي يجب تخصيصه مسبقًا.

ملاحظة

تنتج هذه الدالة مخرجات متطابقة مع gnutls_prf_rfc5705() عندما يتم تعيين server_random_first إلى 0 ويكون extra هو NULL. بموجب TLS1.3، ستعمل هذه الدالة فقط عندما تكون هذه الشروط صحيحة، أو ستعيد GNUTLS_E_INVALID_REQUEST بخلاف ذلك.

المرتجعات

GNUTLS_E_SUCCESS عند النجاح، أو رمز خطأ.
