gnutls_pubkey_verify_data2(3)

gnutls_pubkey_verify_data2(3) gnutls gnutls_pubkey_verify_data2(3)

الاسم

gnutls_pubkey_verify_data2 - وظيفة API

الملخص

#include <gnutls/abstract.h>

int gnutls_pubkey_verify_data2(gnutls_pubkey_t مفتاح عام, gnutls_sign_algorithm_t خوارزمية, unsigned int علامات, const gnutls_datum_t * بيانات, const gnutls_datum_t * توقيع);

المعاملات

gnutls_pubkey_t مفتاح عام
يحتوي على المفتاح العام
gnutls_sign_algorithm_t خوارزمية
خوارزمية التوقيع المستخدمة
unsigned int علامات
صفر أو قائمة OR من gnutls_certificate_verify_flags
const gnutls_datum_t * بيانات
تحتوي على البيانات الموقعة
const gnutls_datum_t * توقيع
يحتوي على التوقيع

الوصف

ستقوم هذه الوظيفة بالتحقق من صحة البيانات الموقعة المعطاة، باستخدام المعلمات من الشهادة.

القيمة المعادة

في حالة فشل التحقق، يتم إرجاع GNUTLS_E_PK_SIG_VERIFY_FAILED، ورمز صفر أو موجب عند النجاح. بالنسبة للتوقيعات المعروفة بأنها غير آمنة، سترجع هذه الوظيفة GNUTLS_E_INSUFFICIENT_SECURITY ما لم يتم تحديد العلم GNUTLS_VERIFY_ALLOW_BROKEN.
