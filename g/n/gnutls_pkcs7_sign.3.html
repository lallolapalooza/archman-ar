gnutls_pkcs7_sign(3)

الحزمة: gnutls

التراخيص: GPL-3.0-or-later AND LGPL-2.1-or-later

gnutls_pkcs7_sign(3) gnutls gnutls_pkcs7_sign(3)

الاسم

gnutls_pkcs7_sign - دالة API

ملخص

#include <gnutls/pkcs7.h>

int gnutls_pkcs7_sign(gnutls_pkcs7_t pkcs7, gnutls_x509_crt_t signer, gnutls_privkey_t signer_key, const gnutls_datum_t * data, gnutls_pkcs7_attrs_t signed_attrs, gnutls_pkcs7_attrs_t unsigned_attrs, gnutls_digest_algorithm_t dig, unsigned flags);

الوسائط

gnutls_pkcs7_t pkcs7
يجب أن يحتوي على نوع gnutls_pkcs7_t
gnutls_x509_crt_t signer
الشهادة لتوقيع البنية
gnutls_privkey_t signer_key
المفتاح لتوقيع البنية
const gnutls_datum_t * data
البيانات المراد توقيعها أو NULL إذا كانت البيانات مضمنة بالفعل
gnutls_pkcs7_attrs_t signed_attrs
أي سمات إضافية لتضمينها في السمات الموقعة (أو NULL)
gnutls_pkcs7_attrs_t unsigned_attrs
أي سمات إضافية لتضمينها في السمات غير الموقعة (أو NULL)
gnutls_digest_algorithm_t dig
خوارزمية التجزئة المستخدمة للتوقيع
unsigned flags
يجب أن تكون صفرًا أو أحد أعلام GNUTLS_PKCS7

الوصف

ستقوم هذه الدالة بإضافة توقيع في بنية PKCS 7 المقدمة للبيانات المقدمة. يمكن إجراء توقيعات متعددة باستخدام موقعين مختلفين.

الأعلام المتاحة هي: GNUTLS_PKCS7_EMBED_DATA، GNUTLS_PKCS7_INCLUDE_TIME، GNUTLS_PKCS7_INCLUDE_CERT، و GNUTLS_PKCS7_WRITE_SPKI. يتم شرحها في تعريف gnutls_pkcs7_sign_flags.

القيم المعادة

عند النجاح، يتم إرجاع GNUTLS_E_SUCCESS (0)، وإلا يتم إرجاع قيمة خطأ سالبة.
