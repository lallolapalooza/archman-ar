gnutls_crypto_register_mac(3)

الحزمة: gnutls

التراخيص: GPL-3.0-or-later AND LGPL-2.1-or-later

gnutls_crypto_register_mac(3) gnutls gnutls_crypto_register_mac(3)

الاسم

gnutls_crypto_register_mac - وظيفة API

الملخص

#include <gnutls/crypto.h>

int gnutls_crypto_register_mac(gnutls_mac_algorithm_t algorithm, int priority, gnutls_mac_init_func init, gnutls_mac_setkey_func setkey, gnutls_mac_setnonce_func setnonce, gnutls_mac_hash_func hash, gnutls_mac_output_func output, gnutls_mac_deinit_func deinit, gnutls_mac_fast_func hash_fast);

المعاملات

gnutls_mac_algorithm_t algorithm
هو معرف MAC الخاص بـ gnutls
int priority
هو أولوية الخوارزمية
gnutls_mac_init_func init
وظيفة تقوم بتهيئة MAC
gnutls_mac_setkey_func setkey
وظيفة تقوم بتعيين مفتاح MAC
gnutls_mac_setnonce_func setnonce
وظيفة تقوم بتعيين nonce لـ mac (يمكن أن تكون NULL لخوارزميات MAC الشائعة)
gnutls_mac_hash_func hash
تنفيذ عملية التجزئة (hash)
gnutls_mac_output_func output
تقديم ناتج MAC
gnutls_mac_deinit_func deinit
وظيفة تقوم بإلغاء تهيئة MAC
gnutls_mac_fast_func hash_fast
تنفيذ عملية MAC دفعة واحدة

الوصف

ستقوم هذه الوظيفة بتسجيل خوارزمية MAC لاستخدامها بواسطة gnutls. أي خوارزمية مسجلة ستحل محل الخوارزميات المضمنة، ووفقًا للاتفاق، تتمتع الخوارزميات المطبقة في النواة بأولوية 90، والخوارزميات المدعومة بوحدة المعالجة المركزية بأولوية 80. سيتم استخدام الخوارزمية ذات الأولوية الأدنى بواسطة gnutls.

مهمل

منذ الإصدار 3.7.0، لم يعد من الممكن تجاوز تطبيق التشفير.

يعيد

GNUTLS_E_SUCCESS عند النجاح، وإلا فإن رمز خطأ سالب.

منذ

3.4.0
