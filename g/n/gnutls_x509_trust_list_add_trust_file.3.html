gnutls_x509_trust_list_add_trust_file(3) gnutls gnutls_x509_trust_list_add_trust_file(3)

اسم

gnutls_x509_trust_list_add_trust_file - وظيفة API

الملخص

#include <gnutls/x509.h>

int gnutls_x509_trust_list_add_trust_file(gnutls_x509_trust_list_t list, const char * ca_file, const char * crl_file, gnutls_x509_crt_fmt_t type, unsigned int tl_flags, unsigned int tl_vflags);

المعاملات

gnutls_x509_trust_list_t list
القائمة
const char * ca_file
ملف يحتوي على قائمة من السلطات المصدقة (اختياري)
const char * crl_file
ملف يحتوي على قائمة من قوائم إبطال الشهادات (اختياري)
gnutls_x509_crt_fmt_t type
تنسيق الشهادات
unsigned int tl_flags
علامات من gnutls_trust_list_flags_t
unsigned int tl_vflags
gnutls_certificate_verify_flags إذا حددت العلامات GNUTLS_TL_VERIFY_CRL

الوصف

ستضيف هذه الوظيفة السلطات المصدقة المعطاة إلى القائمة الموثوقة. يتم قبول عناوين URL الخاصة بـ PKCS 11 أيضًا، بدلاً من الملفات، بواسطة هذه الوظيفة. يشير عنوان URL الخاص بـ PKCS 11 إلى قاعدة بيانات موثوق بها (وحدة نمطية معلمة بشكل خاص في p11-kit) ؛ يشير عنوان URL "pkcs11:" إلى جميع قواعد البيانات الموثوق بها في النظام. يمكن تعيين عنوان URL واحد فقط يحدد قواعد البيانات الموثوق بها ؛ لا يمكن تكديسها عبر استدعاءات متعددة.

يعيد

يعاد عدد العناصر المضافة.
