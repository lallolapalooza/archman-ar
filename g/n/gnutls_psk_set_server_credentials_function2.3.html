#include <gnutls/gnutls.h>

الوسائط

gnutls_psk_server_credentials_t cred
هو نوع gnutls_psk_server_credentials_t.
gnutls_psk_server_credentials_function2 func
هي الدالة الاستدعائية

الوصف

يمكن استخدام هذه الدالة لتعيين دالة استدعائية لاسترداد بيانات اعتماد PSK الخاصة بالمستخدم. شكل الدالة الاستدعائية هو: int (*callback)(gnutls_session_t, const gnutls_datum_t* username, gnutls_datum_t* key);

هذه الدالة الاستدعائية لها نفس دلالات الدالة gnutls_psk_set_server_credentials_function()، ولكنها تسمح باستخدام أسماء مستخدمين غير سلاسل نصية.


username يحتوي على اسم المستخدم الفعلي. يجب ملء key باستخدام gnutls_malloc().

في حالة إرجاع الدالة الاستدعائية رقمًا سالبًا، ستفترض gnutls أن اسم المستخدم غير موجود.

سيتم استدعاء الدالة الاستدعائية مرة واحدة فقط لكل مصافحة. يجب أن تُرجع الدالة الاستدعائية 0 عند النجاح، بينما يشير -1 إلى خطأ.
