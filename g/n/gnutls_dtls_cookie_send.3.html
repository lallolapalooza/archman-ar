gnutls_dtls_cookie_send(3)

الحزمة: gnutls

التراخيص: GPL-3.0-or-later AND LGPL-2.1-or-later

gnutls_dtls_cookie_send(3) gnutls gnutls_dtls_cookie_send(3)

الاسم

gnutls_dtls_cookie_send - وظيفة API

الملخص

#include <gnutls/dtls.h>

int gnutls_dtls_cookie_send(gnutls_datum_t * key, void * client_data, size_t client_data_size, gnutls_dtls_prestate_st * prestate, gnutls_transport_ptr_t ptr, gnutls_push_func push_func);

الوسائط

gnutls_datum_t * key
هو مفتاح عشوائي لاستخدامه في توليد الكوكيز
void * client_data
يحتوي على بيانات تحدد العميل (على سبيل المثال، العنوان)
size_t client_data_size
حجم بيانات العميل
gnutls_dtls_prestate_st * prestate
الكوكيز السابقة التي تم إرجاعها بواسطة gnutls_dtls_cookie_verify()
gnutls_transport_ptr_t ptr
مؤشر نقل سيتم استخدامه بواسطة push_func
gnutls_push_func push_func
وظيفة سيتم استخدامها للرد

الوصف

يمكن استخدام هذه الوظيفة لمنع هجمات الحرمان من الخدمة على خادم DTLS عن طريق مطالبة العميل بالرد باستخدام الكوكيز المرسلة بواسطة هذه الوظيفة. بهذه الطريقة، يمكن التأكد من أن العميل الذي خصصنا له موارد (مثل gnutls_session_t) هو نفسه الذي نشأ منه الحزمة الواردة الأصلية.

يجب استدعاء هذه الوظيفة عند أول حزمة واردة، قبل تخصيص أي موارد، ويجب أن تتبعها gnutls_dtls_cookie_verify().
