gnutls_set_default_priority_append(3)

الحزمة: gnutls

التراخيص: GPL-3.0-or-later AND LGPL-2.1-or-later

gnutls_set_default_priority_append(3) gnutls gnutls_set_default_priority_append(3)

الاسم

gnutls_set_default_priority_append - دالة API

الملخص

#include <gnutls/gnutls.h>

int gnutls_set_default_priority_append(gnutls_session_t session, const char * add_prio, const char ** err_pos, unsigned flags);

المعاملات

gnutls_session_t session
هو نوع gnutls_session_t.
const char * add_prio
هو سلسلة نصية تصف الأولويات التي سيتم إلحاقها بالافتراضيات
const char ** err_pos
في حالة حدوث خطأ، سيحتوي هذا على موضع الخطأ في السلسلة النصية
unsigned flags
يجب أن يكون صفرًا

الوصف

تقوم بتعيين الأولوية الافتراضية على الأكواد، وطرق تبادل المفاتيح، ووحدات التشفير (macs) مع الخيارات الإضافية في add_prio. هذه هي الطريقة الموصى بها لتعيين الافتراضيات عندما يتم إضافة خيارات إضافية قليلة فقط. هذا يعزز الاتساق بين التطبيقات التي تستخدم GnuTLS، ويسمح للتطبيقات التي تستخدم GnuTLS بتحديث الإعدادات بما يتماشى مع المكتبة.

يجب أن تبدأ السلسلة add_prio كسلسلة أولوية عادية، على سبيل المثال، بـ ':'.

للسماح للمستخدم بتجاوز الافتراضيات (على سبيل المثال، عندما تكون هناك واجهة مستخدم أو ملف تكوين متاح)، يمكن استخدام الدوال gnutls_priority_set_direct() أو gnutls_priority_set().

القيمة المعادة

GNUTLS_E_SUCCESS عند النجاح، أو رمز خطأ.

منذ

3.6.3
