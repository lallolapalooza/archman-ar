gnutls_cipher_suite_info(3)

الحزمة: gnutls

التراخيص: GPL-3.0-or-later AND LGPL-2.1-or-later

gnutls_cipher_suite_info(3) gnutls gnutls_cipher_suite_info(3)

الاسم

gnutls_cipher_suite_info - واجهة برمجة التطبيقات (API)

الملخص

#include <gnutls/gnutls.h>

const char * gnutls_cipher_suite_info(size_t idx, unsigned char * cs_id, gnutls_kx_algorithm_t * kx, gnutls_cipher_algorithm_t * cipher, gnutls_mac_algorithm_t * mac, gnutls_protocol_t * min_version);

الوسائط

size_t idx
فهرس مجموعة التشفير للحصول على معلومات عنها، تبدأ من 0.
unsigned char * cs_id
مخزن مؤقت للإخراج بسعة بايتين، يشير إلى قيمة مجموعة التشفير.
gnutls_kx_algorithm_t * kx
متغير الإخراج يشير إلى خوارزمية تبادل المفاتيح، أو NULL.
gnutls_cipher_algorithm_t * cipher
متغير الإخراج يشير إلى التشفير، أو NULL.
gnutls_mac_algorithm_t * mac
متغير الإخراج يشير إلى خوارزمية MAC، أو NULL.
gnutls_protocol_t * min_version
متغير الإخراج يشير إلى إصدار بروتوكول TLS، أو NULL.

الوصف

الحصول على معلومات حول مجموعات التشفير المدعومة. استخدم الدالة بشكل متكرر للحصول على معلومات حول جميع مجموعات التشفير المدعومة. قم بالاستدعاء مع idx=0 للحصول على معلومات حول أول مجموعة تشفير، ثم idx=1 وهكذا، حتى تعود الدالة بـ NULL.

القيمة المرتجعة

اسم مجموعة التشفير idx، وتعيين المعلومات حول مجموعة التشفير في متغيرات الإخراج. إذا كان idx خارج النطاق، يتم إرجاع NULL.
