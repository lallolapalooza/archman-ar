gnutls_pkcs7_verify(3)

الحزمة: gnutls

التراخيص: GPL-3.0-or-later AND LGPL-2.1-or-later

gnutls_pkcs7_verify(3) gnutls gnutls_pkcs7_verify(3)

الاسم

gnutls_pkcs7_verify - وظيفة API

الملخص

#include <gnutls/pkcs7.h>

int gnutls_pkcs7_verify(gnutls_pkcs7_t pkcs7, gnutls_x509_trust_list_t tl, gnutls_typed_vdata_st * vdata, unsigned int vdata_size, unsigned idx, const gnutls_datum_t * data, unsigned flags);

المعاملات

gnutls_pkcs7_t pkcs7
يجب أن يحتوي على نوع gnutls_pkcs7_t
gnutls_x509_trust_list_t tl
قائمة بالشهادات الموثوقة
gnutls_typed_vdata_st * vdata
مصفوفة من البيانات المصنفة
unsigned int vdata_size
عدد عناصر البيانات
unsigned idx
فهرس معلومات التوقيع المراد التحقق منها
const gnutls_datum_t * data
البيانات المراد التحقق منها أو NULL
unsigned flags
صفر أو قائمة OR من gnutls_certificate_verify_flags

الوصف

ستقوم هذه الوظيفة بالتحقق من البيانات المقدمة مقابل التوقيع الموجود في SignedData لهيكل PKCS 7. إذا كانت البيانات المقدمة هي NULL، فسيتم استخدام البيانات الموجودة في حقل encapsulatedContent بدلاً من ذلك.

الإرجاع

عند النجاح، يتم إرجاع GNUTLS_E_SUCCESS (0)، وإلا يتم إرجاع قيمة خطأ سالبة. ينتج عن خطأ التحقق GNUTLS_E_PK_SIG_VERIFY_FAILED ونقص البيانات المغلفة للتحقق منها إلى GNUTLS_E_REQUESTED_DATA_NOT_AVAILABLE.
