الاسم

gnutls_psk_format_imported_identity - دالة API

الملخص

#include <gnutls/gnutls.h>

int gnutls_psk_format_imported_identity(const gnutls_datum_t * identity, const gnutls_datum_t * context, gnutls_protocol_t version, gnutls_digest_algorithm_t hash, gnutls_datum_t * imported_identity);

المعلمات

const gnutls_datum_t * identity
هوية خارجية
const gnutls_datum_t * context
معلومات سياقية اختيارية
gnutls_protocol_t version
إصدار البروتوكول الذي يتم استيراد PSK إليه
gnutls_digest_algorithm_t hash
خوارزمية التجزئة المستخدمة لـ KDF
gnutls_datum_t * imported_identity
المكان الذي يتم فيه تخزين الهوية المستوردة

الوصف

تقوم هذه الدالة بتنسيق هوية PSK خارجية identity إلى شكل مستورد، كما هو موضح في RFC 9258 باسم ImportedIdentity.

عند النجاح، يتم تخصيص حقل البيانات لـ imported_identity باستخدام gnutls_malloc() ويجب على المتصل تحرير الذاكرة بعد الاستخدام.

القيمة المعادة

GNUTLS_E_SUCCESS (0) عند النجاح، وإلا فإن رمز خطأ سالب.

منذ

3.8.1
