gnutls_pkcs7_import(3)

الحزمة: gnutls

التراخيص: GPL-3.0-or-later AND LGPL-2.1-or-later

gnutls_pkcs7_import(3) gnutls gnutls_pkcs7_import(3)

الاسم

gnutls_pkcs7_import - وظيفة API

الملخص

#include <gnutls/pkcs7.h>

int gnutls_pkcs7_import(gnutls_pkcs7_t pkcs7, const gnutls_datum_t * data, gnutls_x509_crt_fmt_t format);

المعاملات

gnutls_pkcs7_t pkcs7
البيانات لتخزين PKCS7 المحلل.
const gnutls_datum_t * data
PKCS7 المشفر بـ DER أو PEM.
gnutls_x509_crt_fmt_t format
أحد DER أو PEM.

الوصف

ستقوم هذه الوظيفة بتحويل PKCS7 المشفر بـ DER أو PEM المعطى إلى تنسيق gnutls_pkcs7_t الأصلي. سيتم تخزين الإخراج في
pkcs7. سيتم تحرير أي بيانات موقعة قد تكون موجودة داخل بنية pkcs7 ، مثل الشهادات التي تم تعيينها بواسطة gnutls_pkcs7_set_crt()، وسيتم استبدالها بواسطة هذه الوظيفة.

إذا كان PKCS7 مشفرًا بـ PEM ، فيجب أن يحتوي على ترويسة "PKCS7".

القيم المعادة

عند النجاح ، يتم إرجاع GNUTLS_E_SUCCESS (0) ، وإلا فإن قيمة خطأ سالبة.
