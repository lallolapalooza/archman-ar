gnutls_pubkey_import_ecc_x962(3)

الحزمة: gnutls

gnutls_pubkey_import_ecc_x962(3) gnutls gnutls_pubkey_import_ecc_x962(3)

الاسم

gnutls_pubkey_import_ecc_x962 - وظيفة API

الملخص

#include <gnutls/abstract.h>

int gnutls_pubkey_import_ecc_x962(gnutls_pubkey_t key, const gnutls_datum_t * parameters, const gnutls_datum_t * ecpoint);

المعاملات

gnutls_pubkey_t key
الهيكل لتخزين المفتاح المحلل
const gnutls_datum_t * parameters
ترميز DER لمعلمات ANSI X9.62
const gnutls_datum_t * ecpoint
ترميز DER لنقطة المنحنى الإهليلجي ANSI X9.62

الوصف

ستقوم هذه الوظيفة بتحويل معلمات المنحنى الإهليلجي المعطاة إلى gnutls_pubkey_t. سيتم تخزين الناتج في key.

القيمة المرجعة

عند النجاح، يتم إرجاع GNUTLS_E_SUCCESS (0)، وإلا فإن قيمة خطأ سالبة.

3.0
