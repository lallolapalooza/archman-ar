guestfs-ruby(3)

الحزمة: libguestfs

التراخيص: GPL-2.0-or-later, LGPL-2.1-or-later

guestfs-ruby(3) دعم المحاكاة الافتراضية guestfs-ruby(3)

الاسم

guestfs-ruby - كيفية استخدام libguestfs من Ruby

الملخص

require 'guestfs'
g = Guestfs::Guestfs.new()
g.add_drive_opts("disk.img",
                 :readonly => true, :format => "raw")
g.launch()

الوصف

توثق صفحة الدليل هذه كيفية استدعاء libguestfs من لغة برمجة Ruby. توثق هذه الصفحة فقط الاختلافات عن واجهة برمجة التطبيقات C وتقدم بعض الأمثلة. إذا لم تكن على دراية باستخدام libguestfs، يجب عليك أيضًا قراءة guestfs(3).

الاستثناءات

يتم تعيين الأخطاء من وظائف libguestfs إلى الاستثناء "Error". يحتوي هذا الاستثناء على معلمة واحدة وهي رسالة الخطأ (سلسلة نصية).

المثال 1: إنشاء صورة قرص

# مثال يوضح كيفية إنشاء صورة قرص.

require 'guestfs'

output = "disk.img"

g = Guestfs::Guestfs.new()

# إنشاء صورة قرص متفرقة بتنسيق خام، بحجم 512 ميجابايت.
g.disk_create (output, "raw", 512 * 1024 * 1024)

# تعيين علامة التتبع حتى نتمكن من رؤية كل استدعاء libguestfs.
g.set_trace(1)

# إرفاق صورة القرص بـ libguestfs.
g.add_drive_opts(output, :format => "raw")

# تشغيل الخلفية لـ libguestfs.
g.launch();

# الحصول على قائمة الأجهزة. نظرًا لأننا أرفقنا محركًا واحدًا فقط
# أعلاه، نتوقع أن تحتوي هذه القائمة على عنصر واحد.
devices = g.list_devices()
if devices.length != 1 then
  raise "خطأ: توقع جهازًا واحدًا من list-devices"
end

# تقسيم القرص كقسم واحد بتنسيق MBR.
g.part_disk(devices[0], "mbr")

# الحصول على قائمة الأقسام. نتوقع عنصرًا واحدًا، وهو
# القسم الذي أنشأناه للتو.
partitions = g.list_partitions()
if partitions.length != 1 then
  raise "خطأ: توقع قسمًا واحدًا من list-partitions"
end

# إنشاء نظام ملفات على القسم.
g.mkfs("ext4", partitions[0])

# الآن قم بتحميل نظام الملفات حتى نتمكن من إضافة ملفات.
g.mount(partitions[0], "/")

# إنشاء بعض الملفات والمجلدات.
g.touch("/empty")
message = "Hello, world\n"
g.write("/hello", message)
g.mkdir("/foo")

# يقوم هذا برفع الملف المحلي /etc/resolv.conf إلى
# صورة القرص.
g.upload("/etc/resolv.conf", "/foo/resolv.conf")

# نظرًا لأننا كتبنا إلى القرص ونريد اكتشاف أخطاء الكتابة، استدعِ g.shutdown. لن تحتاج إلى القيام بذلك:
# سيقوم g.close بذلك ضمنيًا.
g.shutdown()

# لاحظ أيضًا أنه يتم إغلاق المقابض تلقائيًا إذا تم جمعها بواسطة جامع القمامة. أنت فقط بحاجة إلى استدعاء close
# إذا كنت ترغب في إغلاق المقبض على الفور.
g.close()

المثال 2: فحص صورة قرص جهاز افتراضي

# مثال يوضح كيفية فحص قرص جهاز افتراضي.

require 'guestfs'

if ARGV.length == 0
  puts "الاستخدام: inspect_vm disk.img"
  exit 1
end
disk = ARGV[0]

g = Guestfs::Guestfs.new()

# إرفاق صورة القرص للقراءة فقط بـ libguestfs.
g.add_drive_opts(disk, :readonly => true)

# تشغيل الخلفية لـ libguestfs.
g.launch()

# اطلب من libguestfs الفحص بحثًا عن أنظمة التشغيل.
roots = g.inspect_os()
if roots.length == 0
  puts "inspect_vm: لم يتم العثور على أنظمة تشغيل"
  exit 1
end

for root in roots do
  printf("جهاز الجذر: %s\n", root)

  # طباعة معلومات أساسية حول نظام التشغيل.
  printf("  اسم المنتج: %s\n", g.inspect_get_product_name(root))
  printf("  الإصدار:      %d.%d\n",
         g.inspect_get_major_version(root),
         g.inspect_get_minor_version(root))
  printf("  النوع:         %s\n", g.inspect_get_type(root))
  printf("  التوزيع:       %s\n", g.inspect_get_distro(root))

  # تحميل الأقراص، مثل guestfish -i.
  #
  # فرز المفاتيح حسب الطول، الأقصر أولاً، حتى ننتهي
  # بتحميل أنظمة الملفات بالترتيب الصحيح.
  mps = g.inspect_get_mountpoints(root)
  mps = mps.sort {|a,b| a[0].length <=> b[0].length}
  for mp in mps do
    begin
      g.mount_ro(mp[1], mp[0])
    rescue Guestfs::Error => msg
      printf("%s (تم التجاهل)\n", msg)
    end
  end

  # إذا كان الملف /etc/issue.net موجودًا، فقم بطباعة ما يصل إلى 3 أسطر.
  filename = "/etc/issue.net"
  if g.is_file filename then
    printf("--- %s ---\n", filename)
    lines = g.head_n(3, filename)
    for line in lines do
      puts line
    end
  end

  # إلغاء تحميل كل شيء.
  g.umount_all()
end

للحصول على قائمة بالأخطاء ضد libguestfs، استخدم هذا الرابط: https://bugzilla.redhat.com/buglist.cgi?component=libguestfs&product=Virtualization+Tools

عند الإبلاغ عن خطأ، يرجى تقديم:

  • إصدار libguestfs.
  • من أين حصلت على libguestfs (على سبيل المثال، أي توزيعة Linux، تم تجميعها من المصدر، إلخ)

  • قم بتشغيل libguestfs-test-tool(1) والصق الإخراج الكامل وغير المحرر في تقرير الخطأ.
