ts_read(3)

الحزمة: tslib

التراخيص: LGPL-2.1-only

TS_READ(3) tslib TS_READ(3)

الاسم

ts_read, ts_read_raw, ts_read_mt, ts_read_raw_mt - قراءة عينات اللمس من tslib

الملخص

#include <tslib.h>

int ts_read(struct tsdev *dev, struct ts_sample *samp, int nr);

int ts_read_raw(struct tsdev *dev, struct ts_sample *samp, int nr);

int ts_read_mt(struct tsdev *dev, struct ts_sample_mt **samp, int slots, int nr);

int ts_read_raw_mt(struct tsdev *dev, struct ts_sample_mt **samp, int slots, int nr);

الوصف

تقوم الدالة ts_read() بقراءة nr من عينات الإدخال مع تطبيق عوامل تصفية tslib. يتم تعريف struct ts_sample على النحو التالي:

struct ts_sample {
        int             x;
        int             y;
        unsigned int    pressure;
        struct timeval  tv;
};

تقوم الدالة ts_read_mt() بقراءة nr * slots من عينات الإدخال مع تطبيق عوامل تصفية tslib. يتم تعريف struct ts_sample_mt على النحو التالي:

struct ts_sample_mt {
        /* رموز أحداث ABS_MT_* الأخيرة.
         * انظر linux/input.h للأوصاف */
        int             x;
        int             y;
        unsigned int    pressure;
        int             slot;
        int             tracking_id;
        int             tool_type;
        int             tool_x;
        int             tool_y;
        unsigned int    touch_major;
        unsigned int    width_major;
        unsigned int    touch_minor;
        unsigned int    width_minor;
        int             orientation;
        int             distance;
        int             blob_id;
        struct timeval  tv;
        /* حالة BTN_TOUCH */
        short           pen_down;
        /* يتم تعيين بت TSLIB_MT_VALID في valid إذا كانت هذه العينة
         * تحتوي على بيانات جديدة؛
	 * يتم تعيين valid إلى 0 بخلاف ذلك */
        short           valid;
};

يجب على المستخدم توفير كمية الذاكرة الموصوفة في nr و slots لاحتوائها.

تقوم الدالتان ts_read_raw() و ts_read_raw_mt() بنفس الشيء بدون تطبيق عوامل تصفية tslib.

قيمة الإرجاع

يتم إرجاع عدد العينات المقروءة فعليًا. بشكل خاص عند الفتح في وضع غير حاجب، انظر ts_setup()، قد يكون هذا أقل من المطلوب في الاستدعاء. عند الفشل، يتم إرجاع رقم خطأ سالب.

مثال

يقوم البرنامج التالي بقراءة عينات الإدخال متعددة اللمس من tslib باستمرار وطباعة قيم الفتحة والموضع إلى المخرج القياسي عند لمس شاشة اللمس.

#include <stdio.h>
#include <stdlib.h>
#include <fcntl.h>
#include <tslib.h>
#define READ_SAMPLES 1
#define MAX_SLOTS 5
int main(int argc, char **argv)
{
        struct tsdev *ts;
        struct ts_sample_mt **samp_mt = NULL;
        int i, j;
        int ret;
        ts = ts_setup(NULL, 0);
        if (!ts)
                return -1;
        samp_mt = malloc(READ_SAMPLES * sizeof(struct ts_sample_mt **));
        if (!samp_mt)
                return -1;
        for (i = 0; i < READ_SAMPLES; i++) {
                samp_mt[i] = calloc(MAX_SLOTS, sizeof(struct ts_sample_mt));
                if (!samp_mt[i])
                        return -1;
        }
        while(1) {
                ret = ts_read_mt(ts, samp_mt, MAX_SLOTS, READ_SAMPLES);
                for (i = 0; i < ret; i++) {
                        printf("sample nr %d0, i);
                        for (j = 0; i < MAX_SLOTS; j++) {
                                if (!(samp_mt[i][j].valid & TSLIB_MT_VALID))
                                        continue;
                                printf("slot %d: X:%d Y: %d0,
                                       samp_mt[i][j].slot,
                                       samp_mt[i][j].x,
                                       samp_mt[i][j].y);
                        }
                }
        }
}
