Types__Standard__HashRef(3pm)

الحزمة: perl-type-tiny

التراخيص: PerlArtistic, GPL

Types::Standard::HashRef(3) وثائق بيرل التي ساهم بها المستخدم Types::Standard::HashRef(3)

الاسم

Types::Standard::HashRef - أداة تصدير لقيد نوع HashRef

الملخص

use Types::Standard -types;

# الطريقة العادية للتحقق من صحة hashref للأعداد الصحيحة.
#
HashRef->of( Int )->assert_valid( { one => 1 } );

use Types::Standard::HashRef IntHash => { of => Int },

# اختصار مصدّر
#
assert_IntHash { one => 1 };

الحالة

لا تشمل هذه الوحدة سياسة استقرار Type-Tiny.

الوصف

هذه في الغالب كود داخلي، ولكن يمكنها أيضًا العمل كأداة تصدير.

تصديرات

يمكن استخدام Types::Standard::HashRef بشكل تجريبي كمصدر.

use Types::Standard 'Int';
use Types::Standard::HashRef IntHash => { of => Int };

سيؤدي هذا إلى تصدير الدوال التالية إلى مساحة الاسم الخاصة بك:

"IntHash"
is_IntHash( $value )
assert_IntHash( $value )
to_IntHash( $value )

يمكن تصدير أنواع متعددة في وقت واحد:

use Types::Standard -types;
use Types::Standard::HashRef (
  IntHash  => { of => Int },
  NumHash  => { of => Num },
  StrHash  => { of => Str },
);

assert_IntHash { two => 2 };   # لا ينبغي أن يموت

من الممكن تقييد الـ hashref بشكل أكبر باستخدام "where":

use Types::Standard::HashRef MyThing => { of => Int, where => sub { ... } };
