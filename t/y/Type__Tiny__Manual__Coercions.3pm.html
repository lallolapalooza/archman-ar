Type__Tiny__Manual__Coercions(3pm)

الحزمة: perl-type-tiny

التراخيص: PerlArtistic, GPL

Type::Tiny::Manual::Coercions(3) وثائق مساهمة المستخدم من Perl Type::Tiny::Manual::Coercions(3)

الاسم

Type::Tiny::Manual::Coercions - معلومات متقدمة حول التحويلات

دليل

يفترض هذا القسم من الدليل أنك قرأت بالفعل Type::Tiny::Manual::UsingWithMoo.

تأخذ Type::Tiny نهجًا مختلفًا قليلاً لقيود النوع مقارنة بـ Moose. في Moose، يوجد مساحة اسم مسطحة واحدة لقيود النوع. تعرف Moose قيد النوع المسمى Str للسلاسل النصية وقيد النوع المسمى ArrayRef لمراجع المصفوفات. إذا كنت ترغب في تعريف السلاسل النصية بشكل مختلف (ربما تعتقد أن السلسلة النصية الفارغة لا تُعتبر حقًا سلسلة نصية، أو ربما تعتقد أن الكائنات التي تتجاوز "q[""]" يجب أن تُعتبر سلاسل نصية) فيمكنك تسميتها Str؛ تحتاج إلى اختيار اسم مختلف.

مع Type::Tiny، يمكن لمكتبتي أنواع توفير قيد نوع سلسلة نصية بتعريفاتهما الخاصة لما يُعتبر سلسلة نصية، ويمكنك اختيار أيهما تستورد، أو استيرادهما معًا بأسماء مختلفة:

use Some::Types qw( Str );
use Other::Types "Str" => { -as => "Str2" };

قد يبدو هذا ميزة صغيرة لـ Type::Tiny، ولكن حيث تحدث هذه الفلسفة العامة مقابل المحلية فرقًا حقيقيًا هو التحويلات.

لنتخيل أنه لجزء من تطبيقك يتعامل مع قراءة بيانات اسم المستخدم وكلمة المرور، تحتاج إلى سلسلة نصية من نوع "اسم_المستخدم:كلمة_المرور". قد ترغب في قبول مرجع مصفوفة "[$username, $password]" وتحويله إلى سلسلة نصية باستخدام "join ":", @$arrayref". ولكن جزءًا آخر من تطبيقك يتعامل مع قراءة ملفات السجل، ويريد تحويل سلسلة نصية من مرجع مصفوفة باستخدام "join "\n", @$arrayref". هذه كلها طرق معقولة تمامًا لتحويل مرجع مصفوفة. في Moose، طريقة نموذجية للقيام بذلك ستكون:

package My::UserManager {
  use Moose;
  use Moose::Util::TypeConstraints;

  coerce 'Str',
    from 'ArrayRef', via { join ":", @$_ };

  ...;
}

package My::LogReader {
  use Moose;
  use Moose::Util::TypeConstraints;

  coerce 'Str',
    from 'ArrayRef', via { join "\n", @$_ };

  ...;
}

ومع ذلك، نظرًا لأن جميع الأنواع والتحويلات عالمية في Moose، إذا تم تحميل هاتين الفئتين، فستعمل واحدة منهما فقط. ستحل إحدى الفئات محل تحويل الفئة الأخرى. يعتمد الفائز على ترتيب التحميل.

من الممكن حل هذا باستخدام أنواع Moose الأصلية، ولكنه يتطلب عملًا إضافيًا. (الحل هو أن تقوم كل من My::UserManager و My::LogReader بإنشاء نوع فرعي من Str وتحديد التحويل على هذا النوع الفرعي بدلاً من Str مباشرة.)

تحل Type::Tiny هذه المشكلة بطريقتين:

1.
تجعل Type::Tiny من الممكن لمكتبات الأنواع "حماية" قيود أنواعها لمنع الكود الخارجي من إضافة تحويلات جديدة إليها. 
$type->coercion->freeze();

يمكنك تجميد التحويلات لمكتبة النوع بأكملها باستخدام:

__PACKAGE__->make_immutable;

إذا حاولت إضافة تحويلات إلى قيد نوع لديه تحويلات مجمدة، فسوف يرمي خطأ.

use Types::Standard qw( Str ArrayRef );

Str->coercion->add_type_coercions(
  ArrayRef, sub { join "\n", @$_ },
);
2.
تجعل Type::Tiny النمط المذكور أعلاه لإضافة التحويلات إلى نوع فرعي أسهل بكثير. 
use Types::Standard ( Str ArrayRef );

my $subtype = Str->plus_coercions(
  ArrayRef, sub { join "\n", @$_ },
);

تقوم طريقة "plus_coercions" بإنشاء نوع فرعي جديد، وتضيف إليه تحويلات جديدة، وتنسخ أي تحويلات موجودة من النوع الأصل، ثم تجمد التحويلات للنوع الفرعي الجديد.

النتيجة النهائية هي أن لديك الآن "نسخة" من Str يمكنها التحويل من ArrayRef ولكن النسخ الأخرى من Str لن تتأثر بتحويلك.

تحديد التحويلات داخل مكتبات الأنواع

بعض التحويلات مثل ضم مرجع مصفوفة لإنشاء سلسلة نصية لن تكون تحويلات يتفق عليها الجميع. ضم بفاصل سطر بينهما كما في الأعلى؟ أو بنقطتين رأسيتين، أو بمسافة جدولة، أو بمسافة، أو حرف آخر؟ يعتمد الأمر كثيرًا على تطبيقك.

البعض الآخر، مثل تحويل كائن Path::Tiny من سلسلة نصية، من المحتمل أن يكون واضحًا جدًا. هذا هو النوع من التحويل الذي من المنطقي تعريفه داخل المكتبة نفسها بحيث يكون متاحًا لأي حزم تستخدم المكتبة.

my $pt = __PACKAGE__->add_type(
  Type::Tiny::Class->new(
    name    => 'Path',
    class   => 'Path::Tiny',
  ),
);

$pt->coercion->add_type_coercions(
  Str, q{ Path::Tiny::path($_) },
);

$pt->coercion->freeze;

تعديل التحويلات خارج مكتبات الأنواع

تقوم طريقة "plus_coercions" بإنشاء نوع جديد مع تحويلات إضافية. إذا كان النوع الأصلي يحتوي بالفعل على تحويلات، فإن التحويلات الجديدة لها أولوية أعلى.

هناك أيضًا طريقة "plus_fallback_coercions" تقوم بنفس الشيء مثل "plus_coercions" ولكنها تضيف التحويلات الجديدة بأولوية أقل من أي تحويلات موجودة.

توفر Type::Tiny::Class طريقة "plus_constructors" كاختصار للتحويل عبر طريقة بناء. الاثنان التاليان متماثلان:

Path->plus_constructors( Str, "new" )

Path->plus_coercions( Str, q{ Path::Tiny->new($_) } )

لإنشاء قيد نوع بدون تحويلات موجودة معينة، يمكنك استخدام "minus_coercions". يستخدم ما يلي نوع Datetime المعرف في Type::Tiny::Manual::Libraries، ويزيل التحويل من Int ولكنه يحتفظ بالتحويلات من Undef و Dict.

use Types::Standard qw( Int );
use Example::Types qw( Datetime );

has start_date => (
  is      => 'ro',
  isa     => Datetime->minus_coercions( Int ),
  coerce  => 1,
);

هناك أيضًا طريقة "no_coercions" تنشئ نوعًا فرعيًا بدون أي تحويلات على الإطلاق. هذا مفيد للغاية إما لإنشاء "لوحة فارغة" لـ "plus_coercions":

my $Path = Path->no_coercions->plus_coercions( Str, sub { ... } );

أو لتعطيل التحويلات لـ Type::Params. ستقوم Type::Params دائمًا بتحويل المعلمة تلقائيًا إذا كان هناك تحويل لهذا النوع.

use Types::Standard qw( Object );
use Types::Common::String qw( UpperCaseStr );
use Type::Params qw( signature_for );

signature_for set_account_name => (
  method     => Object,
  positional => [ UpperCaseStr->no_coercions ],
);

sub set_account_name ( $self, $name ) {
  $self->_account_name( $name );
  $self->db->update( $self );
  return $self;
}

# هذا سيموت بدلاً من التحويل من حالة الأحرف الصغيرة
$robert->set_account_name( 'bob' );

تحويلات مسماة

حل وسط بين تحديد تحويل في مكتبة النوع أو تحديده في الحزمة التي تستخدم مكتبة النوع هو أن تحدد مكتبة النوع مجموعة مسماة من التحويلات التي يمكن إضافتها اختياريًا إلى قيد النوع.

{
  package MyApp::Types;
  use Type::Library
    -extends => [ 'Types::Standard' ];

  __PACKAGE__->add_coercion(
    name              => "FromLines",
    type_constraint   => ArrayRef,
    type_coercion_map => [
      Str,     q{ [split /\n/] },
      Undef,   q{ [] },
    ],
  );
}

هذه المجموعة من التحويلات لها اسم ويمكن استيرادها واستخدامها:

use MyApp::Types qw( ArrayRef FromLines );

has lines => (
  is      => 'ro',
  isa     => ArrayRef->plus_coercions( FromLines ),
  coerce  => 1,
);

تحدد Types::Standard تحويلًا مسمى MkOpt مصممًا للاستخدام مع OptList.

use Types::Standard qw( OptList MkOpt );
my $OptList = OptList->plus_coercions( MkOpt );

تحويلات معلمة

يمكن أيضًا جعل التحويلات المسماة قابلة للمعلمة.

my $ArrayOfLines = ArrayRef->plus_coercions( Split[ qr{\n} ] );

تحدد Types::Standard تحويلات معلمة Split و Join.
