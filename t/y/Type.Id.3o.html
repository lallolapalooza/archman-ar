معرفات الأنواع.

معرف النوع هو قيمة تدل على نوع. بالنظر إلى معرفين للنوع، يمكن اختبارهما باستخدام Type.Id.provably_equal لإثبات أنهما يدلان على نفس النوع. لاحظ أن:

-المعرفات غير المتساوية لا تعني أنواعًا غير متساوية: يمكن أن يدل نفس النوع على أكثر من معرف واحد.

-يمكن تسلسل معرفات الأنواع، لكنها تحصل على هوية جديدة ومميزة عند إلغاء التسلسل، لذلك تضيع المساواة.

انظر Type.Id.example مثال على الاستخدام.

معرفات الأنواع

type !'a t

النوع لمعرفات النوع 'a .

val make : unit -> 'a t

make () هو معرف نوع جديد.

val uid : 'a t -> int

uid id هو معرف فريد وقت التشغيل لـ id .

val provably_equal : 'a t -> 'b t -> ('a, 'b) Type.eq option

provably_equal i0 i1 هو Some Equal إذا كان المعرف i0 مساويًا لـ i1 و None بخلاف ذلك.

مثال

يوضح ما يلي كيف يمكن استخدام معرفات الأنواع لتنفيذ قاموس بسيط غير متجانس للمفاتيح والقيم. على عكس قيم Map التي ترتبط مفاتيحها بنوع واحد متجانس من القيم، يمكن لهذا القاموس ربط نوع مختلف من القيم بكل مفتاح.

(** قواميس غير متجانسة. *)
module Dict : sig
  type t
  (** النوع للقواميس. *)
  type 'a key
  (** النوع للمفاتيح التي تربط قيمًا من النوع ['a]. *)
  val key : unit -> 'a key
  (** [key ()] هو مفتاح قاموس جديد. *)
  val empty : t
  (** [empty] هو القاموس الفارغ. *)
  val add : 'a key -> 'a -> t -> t
  (** [add k v d] هو [d] مع ربط [k] بـ [v]. *)
  val remove : 'a key -> t -> t
  (** [remove k d] هو [d] مع إزالة آخر ربط لـ [k]. *)
  val find : 'a key -> t -> 'a option
  (** [find k d] هو ربط [k] في [d]، إن وجد. *)
end = struct
  type 'a key = 'a Type.Id.t
  type binding = B : 'a key * 'a -> binding
  type t = (int * binding) list
  let key () = Type.Id.make ()
  let empty = []
  let add k v d = (Type.Id.uid k, B (k, v)) :: d
  let remove k d = List.remove_assoc (Type.Id.uid k) d
  let find : type a. a key -> t -> a option = fun k d ->
    match List.assoc_opt (Type.Id.uid k) d with
    | None -> None
    | Some (B (k', v)) ->
        match Type.Id.provably_equal k k' with
        | Some Type.Equal -> Some v
        | None -> assert false
end
