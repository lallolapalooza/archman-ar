subroutine ctgexc (logical wantq, logical wantz, integer n, complex, dimension( lda, * ) a, integer lda, complex, dimension( ldb, * ) b, integer ldb, complex, dimension( ldq, * ) q, integer ldq, complex, dimension( ldz, * ) z, integer ldz, integer ifst, integer ilst, integer info)

CTGEXC

الغرض:

!>
!> CTGEXC يعيد ترتيب تحليل Schur العام لمصفوفة معقدة (A,B)، باستخدام تحويل
!> تكافؤ موحد
!> (A, B) := Q * (A, B) * Z**H، بحيث يتم نقل الكتلة القطرية لـ (A, B) ذات
!> مؤشر الصف IFST إلى الصف ILST.
!>
!> يجب أن تكون (A, B) في الشكل القانوني العام لـ Schur، أي أن A و B
!> كلاهما مثلثي علوي.
!>
!> اختياريًا، يتم تحديث مصفوفات متجهات Schur العامة Q و Z.
!>
!>        Q(in) * A(in) * Z(in)**H = Q(out) * A(out) * Z(out)**H
!>        Q(in) * B(in) * Z(in)**H = Q(out) * B(out) * Z(out)**H
!>

المعاملات

WANTQ 
!>          WANTQ هو LOGICAL
!>          .TRUE. : تحديث مصفوفة التحويل اليسرى Q؛
!>          .FALSE.: عدم تحديث Q.
!>

WANTZ

!>          WANTZ هو LOGICAL
!>          .TRUE. : تحديث مصفوفة التحويل اليمنى Z؛
!>          .FALSE.: عدم تحديث Z.
!>

N

!>          N هو INTEGER
!>          رتبة المصفوفتين A و B. N >= 0.
!>

A

!>          A هي مصفوفة COMPLEX، أبعادها (LDA,N)
!>          عند الدخول، المصفوفة المثلثية العليا A في الزوج (A, B).
!>          عند الخروج، المصفوفة المحدثة A.
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرئيسي للمصفوفة A. LDA >= max(1,N).
!>

B

!>          B هي مصفوفة COMPLEX، أبعادها (LDB,N)
!>          عند الدخول، المصفوفة المثلثية العليا B في الزوج (A, B).
!>          عند الخروج، المصفوفة المحدثة B.
!>

LDB

!>          LDB هو INTEGER
!>          البعد الرئيسي للمصفوفة B. LDB >= max(1,N).
!>

Q

!>          Q هي مصفوفة COMPLEX، أبعادها (LDQ,N)
!>          عند الدخول، إذا كان WANTQ = .TRUE.، فإن المصفوفة الموحدة Q.
!>          عند الخروج، المصفوفة المحدثة Q.
!>          إذا كان WANTQ = .FALSE.، فلا يتم الرجوع إلى Q.
!>

LDQ

!>          LDQ هو INTEGER
!>          البعد الرئيسي للمصفوفة Q. LDQ >= 1؛
!>          إذا كان WANTQ = .TRUE.، فإن LDQ >= N.
!>

Z

!>          Z هي مصفوفة COMPLEX، أبعادها (LDZ,N)
!>          عند الدخول، إذا كان WANTZ = .TRUE.، فإن المصفوفة الموحدة Z.
!>          عند الخروج، المصفوفة المحدثة Z.
!>          إذا كان WANTZ = .FALSE.، فلا يتم الرجوع إلى Z.
!>

LDZ

!>          LDZ هو INTEGER
!>          البعد الرئيسي للمصفوفة Z. LDZ >= 1؛
!>          إذا كان WANTZ = .TRUE.، فإن LDZ >= N.
!>

IFST

!>          IFST هو INTEGER
!>

ILST

!>          ILST هو INTEGER
!>          تحديد إعادة ترتيب الكتل القطرية لـ (A, B).
!>          يتم نقل الكتلة ذات مؤشر الصف IFST إلى الصف ILST، عن طريق
!>          تسلسل من التبادلات بين الكتل المجاورة.
!>

INFO

!>          INFO هو INTEGER
!>           =0: خروج ناجح.
!>           <0: إذا كان INFO = -i، فإن المعامل i كان له قيمة غير قانونية.
!>           =1: ستكون المصفوفة المزدوجة المحولة (A, B) بعيدة جدًا
!>                عن الشكل العام لـ Schur؛ المشكلة غير واضحة
!>                الشرط. قد تكون (A, B) قد أعيد ترتيبها جزئيًا،
!>                ويشير ILST إلى الصف الأول من الموضع الحالي
!>                للكتلة التي يتم نقلها.
!>

subroutine ztgexc (logical wantq, logical wantz, integer n, complex*, * a, integer lda, complex*, * b, integer ldb, complex*, * q, integer ldq, complex*, * z, integer ldz, integer ifst, integer ilst, complex*, * work, integer lwork, integer info)

ZTGEXC

الغرض:

!>
!> ZTGEXC يعيد ترتيب تحليل Schur العام لمصفوفة معقدة (A,B)، باستخدام تحويل
!> تكافؤ موحد
!>
!>                (A, B) = Q * (A, B) * Z**H،
!>
!> بحيث يتم نقل الكتلة القطرية لـ (A, B) ذات مؤشر الصف IFST
!> إلى الصف ILST.
!>
!> يجب أن تكون (A, B) في الشكل القانوني العام لـ Schur (كما تم إرجاعه
!> بواسطة ZGGES)، أي أن A مثلثي علوي بشكل كتلة مع كتل قطرية 1x1 و 2x2.
!> B مثلثي علوي.
!>
!> اختياريًا، يتم تحديث مصفوفات المتجهات العامة لـ Schur Q و Z.
!>
!>        Q(in) * A(in) * Z(in)**H = Q(out) * A(out) * Z(out)**H
!>        Q(in) * B(in) * Z(in)**H = Q(out) * B(out) * Z(out)**H
!>

المعاملات

WANTQ 
!>          WANTQ هو LOGICAL
!>          .TRUE. : تحديث مصفوفة التحويل اليسرى Q؛
!>          .FALSE.: عدم تحديث Q.
!>

WANTZ

!>          WANTZ هو LOGICAL
!>          .TRUE. : تحديث مصفوفة التحويل اليمنى Z؛
!>          .FALSE.: عدم تحديث Z.
!>

N

!>          N هو INTEGER
!>          رتبة المصفوفتين A و B. N >= 0.
!>

A

!>          A هي مصفوفة COMPLEX*16، أبعادها (LDA,N)
!>          عند الدخول، المصفوفة المثلثية العليا A في الزوج (A, B).
!>          عند الخروج، المصفوفة المحدثة A.
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرئيسي للمصفوفة A. LDA >= max(1,N).
!>

B

!>          B هي مصفوفة COMPLEX*16، أبعادها (LDB,N)
!>          عند الدخول، المصفوفة المثلثية العليا B في الزوج (A, B).
!>          عند الخروج، المصفوفة المحدثة B.
!>

LDB

!>          LDB هو INTEGER
!>          البعد الرئيسي للمصفوفة B. LDB >= max(1,N).
!>

Q

!>          Q هي مصفوفة COMPLEX*16، أبعادها (LDQ,N)
!>          عند الدخول، إذا كان WANTQ = .TRUE.، فإن المصفوفة الموحدة Q.
!>          عند الخروج، المصفوفة المحدثة Q.
!>          إذا كان WANTQ = .FALSE.، فلا يتم الرجوع إلى Q.
!>

LDQ

!>          LDQ هو INTEGER
!>          البعد الرئيسي للمصفوفة Q. LDQ >= 1؛
!>          إذا كان WANTQ = .TRUE.، فإن LDQ >= N.
!>

Z

!>          Z هي مصفوفة COMPLEX*16، أبعادها (LDZ,N)
!>          عند الدخول، إذا كان WANTZ = .TRUE.، فإن المصفوفة الموحدة Z.
!>          عند الخروج، المصفوفة المحدثة Z.
!>          إذا كان WANTZ = .FALSE.، فلا يتم الرجوع إلى Z.
!>

LDZ

!>          LDZ هو INTEGER
!>          البعد الرئيسي للمصفوفة Z. LDZ >= 1؛
!>          إذا كان WANTZ = .TRUE.، فإن LDZ >= N.
!>

IFST

!>          IFST هو INTEGER
!>

ILST

!>          ILST هو INTEGER
!>          تحديد إعادة ترتيب الكتل القطرية لـ (A, B).
!>          يتم نقل الكتلة ذات مؤشر الصف IFST إلى الصف ILST، عن طريق
!>          تسلسل من التبادلات بين الكتل المجاورة.
!>

WORK

!>          WORK هي مصفوفة COMPLEX*16، أبعادها (LWORK)
!>          مساحة عمل مؤقتة.
!>

LWORK

!>          LWORK هو INTEGER
!>          حجم مصفوفة العمل WORK.
!>          LWORK >= N.
!>          إذا كان LWORK = -1، فسيتم حساب الحجم الأمثل لـ WORK.
!>

INFO

!>          INFO هو INTEGER
!>           =0: خروج ناجح.
!>           <0: إذا كان INFO = -i، فإن المعامل i كان له قيمة غير قانونية.
!>           =1: لم تكن هناك مساحة عمل كافية في WORK.
!>           =2: المشكلة غير واضحة الشرط. (A, B) قد تكون أعيد ترتيبها جزئيًا،
!>                و ILST يشير إلى الصف الأول من الموضع الحالي
!>                للكتلة التي يتم نقلها.
!>
