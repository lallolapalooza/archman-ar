threads(3perl) دليل مبرمجي Perl threads(3perl)

الاسم

threads - خيوط مترجم Perl

الإصدار

يوثق هذا المستند threads الإصدار 2.43

تحذير

الـ "خيوط المترجم" التي يوفرها Perl ليست سريعة، وخفيفة الوزن، لنظام تعدد المهام كما قد يتوقع أو يأمل المرء. يتم تنفيذ الخيوط بطريقة تجعلها سهلة لسوء الاستخدام. قليل من الناس يعرفون كيفية استخدامها بشكل صحيح أو سيكونون قادرين على تقديم المساعدة.

يتم تثبيط استخدام خيوط المترجم في Perl رسميًا.

الملخص

use threads ('yield',
             'stack_size' => 64*4096,
             'exit' => 'threads_only',
             'stringify');
sub start_thread {
    my @args = @_;
    print('Thread started: ', join(' ', @args), "\n");
}
my $thr = threads->create('start_thread', 'argument');
$thr->join();
threads->create(sub { print("I am a thread\n"); })->join();
my $thr2 = async { foreach (@files) { ... } };
$thr2->join();
if (my $err = $thr2->error()) {
    warn("Thread error: $err\n");
}
# استدعاء الخيط في سياق القائمة (ضمني) بحيث يمكنه إرجاع قائمة
my ($thr) = threads->create(sub { return (qw/a b c/); });
# أو تحديد سياق القائمة بشكل صريح
my $thr = threads->create({'context' => 'list'},
                          sub { return (qw/a b c/); });
my @results = $thr->join();
$thr->detach();
# الحصول على كائن خيط
$thr = threads->self();
$thr = threads->object($tid);
# الحصول على معرف الخيط
$tid = threads->tid();
$tid = $thr->tid();
$tid = "$thr";
# إعطاء فرصة للخيوط الأخرى للعمل
threads->yield();
yield();
# قوائم الخيوط غير المنفصلة
my @threads = threads->list();
my $thread_count = threads->list();
my @running = threads->list(threads::running);
my @joinable = threads->list(threads::joinable);
# اختبار كائنات الخيوط
if ($thr1 == $thr2) {
    ...
}
# إدارة حجم مكدس الخيط
$stack_size = threads->get_stack_size();
$old_size = threads->set_stack_size(32*4096);
# إنشاء خيط بسياق وحجم مكدس محدد
my $thr = threads->create({ 'context'    => 'list',
                            'stack_size' => 32*4096,
                            'exit'       => 'thread_only' },
                          \&foo);
# الحصول على سياق الخيط
my $wantarray = $thr->wantarray();
# التحقق من حالة الخيط
if ($thr->is_running()) {
    sleep(1);
}
if ($thr->is_joinable()) {
    $thr->join();
}
# إرسال إشارة إلى خيط
$thr->kill('SIGUSR1');
# إنهاء خيط
threads->exit();

الوصف

منذ Perl 5.8، أصبح برمجة الخيوط متاحة باستخدام نموذج يسمى خيوط المترجم والذي يوفر مترجم Perl جديد لكل خيط، وبشكل افتراضي، لا يتم مشاركة أي بيانات أو معلومات حالة بين الخيوط.

(قبل Perl 5.8، كان 5005threads متاحًا عبر واجهة برمجة التطبيقات "Thread.pm". تم إهمال نموذج الخيوط هذا، وتمت إزالته اعتبارًا من Perl 5.10.0.)

كما ذكرنا للتو، جميع المتغيرات، بشكل افتراضي، محلية للخيط. لاستخدام المتغيرات المشتركة، تحتاج أيضًا إلى تحميل threads::shared:

use threads;
use threads::shared;

عند تحميل threads::shared، يجب عليك "use threads" قبل أن تقوم "use threads::shared". (سيصدر "threads" تحذيرًا إذا قمت بذلك بالطريقة الأخرى.)

يوصى بشدة بتمكين الخيوط عبر "use threads" في أقرب وقت ممكن في النص البرمجي الخاص بك.

إذا لزم الأمر، يمكن كتابة النصوص البرمجية بحيث تعمل على كل من Perl المزود بالخيوط وغير المزود بها:

my $can_use_threads = eval 'use threads; 1';
if ($can_use_threads) {
    # قم بالمعالجة باستخدام الخيوط
    ...
} else {
    # قم بذلك بدون استخدام الخيوط
    ...
}
$thr = threads->create(FUNCTION, ARGS)
سيؤدي هذا إلى إنشاء خيط جديد يبدأ التنفيذ مع نقطة الدخول المحددة، ويمنحه قائمة ARGS كمعلمات. سيعيد كائن الخيوط المقابل، أو "undef" إذا فشل إنشاء الخيط.

FUNCTION يمكن أن يكون إما اسم دالة، أو روتين فرعي مجهول، أو مرجع كود.

my $thr = threads->create('func_name', ...);
    # أو
my $thr = threads->create(sub { ... }, ...);
    # أو
my $thr = threads->create(\&func, ...);

الطريقة "->new()" هي اسم مستعار للطريقة "->create()".

$thr->join()
سينتظر هذا الخيط حتى يكمل تنفيذه. عندما ينتهي الخيط، سيعيد "->join()" قيمة (قيم) الإرجاع لدالة نقطة الدخول.

يتم تحديد السياق (فارغ، قياسي، أو قائمة) لقيمة (قيم) الإرجاع لـ "->join()" وقت إنشاء الخيط.

# إنشاء خيط في سياق قائمة (ضمني)
my ($thr1) = threads->create(sub {
                                my @results = qw(a b c);
                                return (@results);
                             });
#   أو (صريح)
my $thr1 = threads->create({'context' => 'list'},
                           sub {
                                my @results = qw(a b c);
                                return (@results);
                           });
# استرداد نتائج القائمة من الخيط
my @res1 = $thr1->join();
# إنشاء خيط في سياق قياسي (ضمني)
my $thr2 = threads->create(sub {
                                my $result = 42;
                                return ($result);
                             });
# استرداد نتيجة قياسية من الخيط
my $res2 = $thr2->join();
# إنشاء خيط في سياق فارغ (صريح)
my $thr3 = threads->create({'void' => 1},
                           sub { print("Hello, world\n"); });
# ضم الخيط في سياق فارغ (أي، لا توجد قيمة إرجاع)
$thr3->join();

إذا خرج البرنامج دون أن يتم ضم أو فصل جميع الخيوط، فسيتم إصدار تحذير.

سيؤدي استدعاء "->join()" أو "->detach()" على خيط تم ضمه بالفعل إلى إلقاء خطأ.

$thr->detach()
يجعل الخيط غير قابل للضم، ويتسبب في تجاهل أي قيمة إرجاع محتملة. عند انتهاء البرنامج، يتم إنهاء أي خيوط منفصلة لا تزال قيد التشغيل بصمت.

إذا خرج البرنامج دون أن يتم ضم أو فصل جميع الخيوط، فسيتم إصدار تحذير.

سيؤدي استدعاء "->join()" أو "->detach()" على خيط تم فصله بالفعل إلى إلقاء خطأ.

threads->detach()
طريقة فئة تسمح للخيط بفصل نفسه.
threads->self()
طريقة فئة تسمح للخيط بالحصول على كائن الخيوط الخاص به.
$thr->tid()
يعيد معرف الخيط. معرفات الخيوط هي أعداد صحيحة فريدة، حيث يكون الخيط الرئيسي في البرنامج هو 0، ويتزايد بمقدار 1 لكل خيط تم إنشاؤه.
threads->tid()
طريقة فئة تسمح للخيط بالحصول على معرفه الخاص.
"$thr"
إذا قمت بإضافة خيار الاستيراد "stringify" إلى إعلان "use threads" الخاص بك، فعند استخدام كائن خيوط في سلسلة نصية أو سياق سلسلة نصية (على سبيل المثال، كمفتاح جدول هاش) سيؤدي ذلك إلى استخدام معرفه كقيمة: 
use threads qw(stringify);
my $thr = threads->create(...);
print("Thread $thr started\n");  # يطبع: Thread 1 started
threads->object($tid)
سيعيد هذا كائن الخيوط للخيط النشط المرتبط بمعرف الخيط المحدد. إذا كان $tid هو قيمة الخيط الحالي، فإن هذا الاستدعاء يعمل بنفس طريقة "->self()". بخلاف ذلك، يعيد "undef" إذا لم يكن هناك خيط مرتبط بـ TID، أو إذا تم ضم الخيط أو فصله، أو إذا لم يتم تحديد TID، أو إذا كان TID المحدد هو undef.
threads->yield()
هذه طريقة فئة. عندما يتم استدعاؤها، فإنها تعطي الخيط الحالي فرصة للسماح للخيوط الأخرى بالعمل.
