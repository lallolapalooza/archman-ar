toxic.conf(5)

الحزمة: toxic

التراخيص: GPL-3.0-or-later

TOXIC.CONF(5) دليل توكسيك TOXIC.CONF(5)

اسم

toxic.conf - ملف إعدادات برنامج toxic

ملخص

~/.config/tox/toxic.conf

الوصف

ملف toxic.conf هو ملف الإعدادات الرئيسي لعميل toxic(1). يستخدم بناء الجملة الذي تقبله libconfig. الأسطر التي تبدأ بـ "//" هي تعليقات وسيتم تجاهلها.

مثال

// إعدادات الواجهة
ui = {
    timestamps = true;
    alerts = false;
};
// إعدادات الصوت
audio = {
    input_device = 1;
};

خيارات

ui

إعدادات متعلقة بعناصر الواجهة.

timestamps

تمكين أو تعطيل الطوابع الزمنية. true أو false

time_format

الاختيار بين الوقت بنظام 24 ساعة و 12 ساعة. حدد 24 أو 12. سيؤدي تعيين timestamp_format و log_timestamp_format إلى تجاوز هذا الإعداد.

timestamp_format

سلسلة تنسيق الوقت للواجهة محاطة بعلامتي اقتباس مزدوجتين. انظر date(1)

log_timestamp_format

سلسلة تنسيق الوقت للتسجيل محاطة بعلامتي اقتباس مزدوجتين. (ملاحظة: يجب أن يكون جزء الوقت من سلسلة التنسيق محاطًا بأقواس مربعة لكي يتم تحليله بواسطة toxic). انظر date(1)

alerts

تمكين أو تعطيل التنبيهات الصوتية عند وقوع أحداث. true أو false

show_notification_content

عرض محتوى إشعارات سطح المكتب. true أو false

native_colors

الاختيار بين ألوان الطرفية الأصلية وموضوع ألوان toxic. true أو false

color_bar_bg

تعيين لون خلفية أشرطة حالة الدردشة. (أسود، أبيض، رمادي، بني، أحمر، أخضر، أزرق، سماوي، أصفر، أرجواني، برتقالي، وردي)

color_bar_fg

تعيين لون المقدمة (النص) لأشرطة حالة الدردشة. (أسود، أبيض، رمادي، بني، أحمر، أخضر، أزرق، سماوي، أصفر، أرجواني، برتقالي، وردي)

color_bar_accent

تعيين لون التركيز الأمامي لأشرطة حالة الدردشة. (أسود، أبيض، رمادي، بني، أحمر، أخضر، أزرق، سماوي، أصفر، أرجواني، برتقالي، وردي)

color_bar_notify

تعيين لون الإشعار (والكتابة) الأمامي في شريط حالة الدردشة. (أسود، أبيض، رمادي، بني، أحمر، أخضر، أزرق، سماوي، أصفر، أرجواني، برتقالي، وردي)

autolog

تمكين أو تعطيل التسجيل التلقائي. true أو false

show_typing_other

إظهار متى يكتب الآخرون في دردشة فردية. true أو false

show_typing_self

إظهار الآخرين عندما تكتب في دردشة فردية. true أو false

show_welcome_msg

عرض رسالة الترحيب عند بدء التشغيل. true أو false

show_connection_msg

تمكين إشعارات تغيير اتصال الأصدقاء. true أو false

show_group_connection_msg

تمكين إشعارات تغيير اتصال المجموعات. true أو false

nodelist_update_freq

كم مرة بالأيام لتحديث قائمة عقد DHT. (عدد صحيح؛ 0 لتعطيل)

autosave_freq

كم مرة بالثواني للحفظ التلقائي لبيانات Tox. (عدد صحيح؛ 0 لتعطيل)

history_size

الحد الأقصى لعدد الأسطر في سجل الدردشة. قيمة عدد صحيح. (على سبيل المثال: 700)

notification_timeout

الوقت بالمللي ثانية لعرض الإشعار. قيمة عدد صحيح. (على سبيل المثال: 3000)

line_join

مؤشر عند اتصال شخص ما أو انضمامه إلى مجموعة. ثلاثة أحرف كحد أقصى لإعدادات line_.

line_quit

مؤشر عند قطع اتصال شخص ما أو مغادرته مجموعة.

line_alert

مؤشر لرسائل التنبيه.

line_normal

مؤشر للرسائل العادية.

line_padding

ملء كل سطر ملفوف من الرسالة بما يكفي من المسافات لمحاذاته مع بداية السطر الأول. true أو false

mplex_away

تعيين حالة المستخدم عند إرفاق وفصل عن GNU screen أو tmux. true أو false

mplex_away_note

رسالة الحالة التي سيتم تعيينها عندما تكون الحالة بعيدًا بسبب فصل الشاشة/tmux. عند الإرفاق، يتم تعيين رسالة الحالة مرة أخرى إلى القيمة الأصلية.

group_part_message

رسالة المغادرة التي سيتم إرسالها إلى جميع أقران الدردشة الجماعية عند مغادرتك المجموعة. 
الخيارات التالية تتحكم فيما إذا كان سيتم إخراج جرس طرفي عند أحداث معينة.
بعض الأطراف المميزة للنافذة كعاجلة عند تلقي جرس. النوافذ العاجلة عادة ما تكون مميزة في شريط المهام وبعض مديري النوافذ يوفرون اختصارات للانتقال إلى النافذة العاجلة التالية.
هذه الخيارات لا تؤثر على خيار "alerts".

bell_on_message

تمكين/تعطيل جرس الطرفية عند تلقي رسالة. true أو false

bell_on_filetrans

تمكين/تعطيل جرس الطرفية عند تلقي نقل ملف. true أو false

bell_on_filetrans_accept

تمكين/تعطيل جرس الطرفية عند قبول نقل ملف. true أو false

bell_on_invite

تمكين/تعطيل جرس الطرفية عند تلقي دعوة جماعية/مكالمة. true أو false

audio

إعدادات متعلقة بالأجهزة الصوتية.

input_device

جهاز إدخال الصوت. قيمة عدد صحيح. الرقم يقابل /lsdev في

output_device

جهاز إخراج الصوت. قيمة عدد صحيح. الرقم يقابل /lsdev خارج

VAD_threshold

عتبة اكتشاف النشاط الصوتي. قيمة عشرية. القيم الموصى بها هي 1.0-40.0

conference_audio_channels

عدد القنوات لبث الصوت الجماعي. قيمة عدد صحيح. 1 (أحادي) أو 2 (استريو)

chat_audio_channels

عدد القنوات لبث الصوت الفردي. قيمة عدد صحيح. 1 (أحادي) أو 2 (استريو)

push_to_talk

تمكين/تعطيل الضغط للتحدث (Push-To-Talk) لدردشات الصوت الجماعي (المفتاح النشط هو F2). true أو false

tox

إعدادات متعلقة بالمسارات.

download_path

المسار الافتراضي للتنزيلات. قيمة سلسلة نصية. مسار مطلق للملفات التي تم تنزيلها.

avatar_path

المسار لصورتك الرمزية (يجب أن يكون الملف .png ولا يمكن أن يتجاوز 16.3 كيلوبايت)

autorun_path

المسار لأي نصوص برمجية يجب تشغيلها عند بدء التشغيل

chatlogs_path

المسار الافتراضي لسجلات الدردشة. قيمة سلسلة نصية. مسار مطلق لملفات سجلات الدردشة.

password_eval

استبدال المطالبة بكلمة المرور عن طريق تشغيل هذا الأمر واستخدام مخرجاته ككلمة مرور.

friends

إعدادات خاصة بالأصدقاء (لها الأولوية على الإعدادات العامة)

alias

تعيين اسم مستعار لهذا الصديق. قيمة سلسلة نصية.

auto_accept_files

تمكين أو تعطيل قبول عمليات نقل الملفات الواردة تلقائيًا. true أو false

autolog

تمكين أو تعطيل التسجيل التلقائي. true أو false

show_connection_msg

تمكين إشعارات تغيير الاتصال. true أو false

tab_name_colour

لون اسم نافذة علامة تبويب الصديق. (أسود، أبيض، رمادي، بني، أحمر، أخضر، أزرق، سماوي، أصفر، أرجواني، برتقالي، وردي)

groupchats

إعدادات خاصة بالدردشات الجماعية (لها الأولوية على الإعدادات العامة)

autolog

تمكين أو تعطيل التسجيل التلقائي. true أو false

tab_name_colour

لون اسم نافذة علامة تبويب المجموعة. (أسود، أبيض، رمادي، بني، أحمر، أخضر، أزرق، سماوي، أصفر، أرجواني، برتقالي، وردي)
