subroutine ctpmqrt (character side, character trans, integer m, integer n, integer k, integer l, integer nb, complex, dimension( ldv, * ) v, integer ldv, complex, dimension( ldt, * ) t, integer ldt, complex, dimension( lda, * ) a, integer lda, complex, dimension( ldb, * ) b, integer ldb, complex, dimension( * ) work, integer info)

CTPMQRT

الغرض:

!>
!> CTPMQRT يطبق مصفوفة Q متعامدة مركبة تم الحصول عليها من
!>  عاكس كتلة مركب H على مصفوفة عامة
!> مصفوفة مركبة C، تتكون من كتلتين A و B.
!>

المعلمات

SIDE 
!>          SIDE هو CHARACTER*1
!>          = 'L': تطبيق Q أو Q**H من اليسار؛
!>          = 'R': تطبيق Q أو Q**H من اليمين.
!>

TRANS

!>          TRANS هو CHARACTER*1
!>          = 'N':  بدون تبديل، تطبيق Q؛
!>          = 'C':  تبديل مركب، تطبيق Q**H.
!>

M

!>          M هو INTEGER
!>          عدد صفوف المصفوفة B. M >= 0.
!>

N

!>          N هو INTEGER
!>          عدد أعمدة المصفوفة B. N >= 0.
!>

K

!>          K هو INTEGER
!>          عدد العواكس الأولية التي يحدد حاصل ضربها
!>          المصفوفة Q.
!>

L

!>          L هو INTEGER
!>          ترتيب الجزء شبه المنحرف من V.
!>          K >= L >= 0.  انظر التفاصيل الإضافية.
!>

NB

!>          NB هو INTEGER
!>          حجم الكتلة المستخدم لتخزين T.  K >= NB >= 1.
!>          يجب أن تكون هذه نفس قيمة NB المستخدمة لتوليد T
!>          في CTPQRT.
!>

V

!>          V هي مصفوفة COMPLEX، البعد (LDV,K)
!>          يجب أن يحتوي العمود i على المتجه الذي يحدد
!>          العاكس الأولي H(i)، لـ i = 1,2,...,k، كما تم إرجاعه بواسطة
!>          CTPQRT في B.  انظر التفاصيل الإضافية.
!>

LDV

!>          LDV هو INTEGER
!>          البعد الرائد للمصفوفة V.
!>          إذا كان SIDE = 'L'، فإن LDV >= max(1,M)؛
!>          إذا كان SIDE = 'R'، فإن LDV >= max(1,N).
!>

T

!>          T هي مصفوفة COMPLEX، البعد (LDT,K)
!>          عوامل المثلث العلوي للعواكس الكتلية
!>          كما تم إرجاعها بواسطة CTPQRT، مخزنة كمصفوفة NB × K.
!>

LDT

!>          LDT هو INTEGER
!>          البعد الرائد للمصفوفة T.  LDT >= NB.
!>

A

!>          A هي مصفوفة COMPLEX، البعد
!>          (LDA,N) إذا كان SIDE = 'L' أو
!>          (LDA,K) إذا كان SIDE = 'R'
!>          عند الدخول، المصفوفة A بحجم K × N أو M × K.
!>          عند الخروج، يتم استبدال A بالجزء المقابل من
!>          Q*C أو Q**H*C أو C*Q أو C*Q**H.  انظر التفاصيل الإضافية.
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرائد للمصفوفة A.
!>          إذا كان SIDE = 'L'، فإن LDA >= max(1,K)؛
!>          إذا كان SIDE = 'R'، فإن LDA >= max(1,M).
!>

B

!>          B هي مصفوفة COMPLEX، البعد (LDB,N)
!>          عند الدخول، المصفوفة B بحجم M × N.
!>          عند الخروج، يتم استبدال B بالجزء المقابل من
!>          Q*C أو Q**H*C أو C*Q أو C*Q**H.  انظر التفاصيل الإضافية.
!>

LDB

!>          LDB هو INTEGER
!>          البعد الرائد للمصفوفة B.
!>          LDB >= max(1,M).
!>

WORK

!>          WORK هي مصفوفة COMPLEX. بعد WORK هو
!>           N*NB إذا كان SIDE = 'L'، أو  M*NB إذا كان SIDE = 'R'.
!>

INFO

!>          INFO هو INTEGER
!>          = 0:  اكتمال النجاح
!>          < 0:  إذا كان INFO = -i، فإن الوسيط i كان له قيمة غير قانونية
!>

تفاصيل إضافية:

!>
!>  تحتوي أعمدة المصفوفة الخماسية V على العواكس الأولية
!>  H(1), H(2), ..., H(K)؛ V تتكون من كتلة مستطيلة V1 و
!>  كتلة شبه منحرف V2:
!>
!>        V = [V1]
!>            [V2].
!>
!>  يتم تحديد حجم الكتلة شبه المنحرف V2 بواسطة المعلمة L،
!>  حيث 0 <= L <= K؛ V2 شبه منحرف علوي، يتكون من أول L
!>  صفوف لمصفوفة علوية مثلثة بحجم K × K.  إذا كان L=K، فإن V2 مثلثية علوية؛
!>  إذا كان L=0، فلا توجد كتلة شبه منحرف، وبالتالي V = V1 مستطيلة.
!>
!>  إذا كان SIDE = 'L':  C = [A]  حيث A بحجم K × N،  B بحجم M × N و V بحجم M × K.
!>                      [B]
!>
!>  إذا كان SIDE = 'R':  C = [A B]  حيث A بحجم M × K، B بحجم M × N و V بحجم N × K.
!>
!>  يتم تشكيل المصفوفة المتعاقدة Q من V و T.
!>
!>  إذا كان TRANS='N' و SIDE='L'، يتم استبدال C عند الخروج بـ Q * C.
!>
!>  إذا كان TRANS='C' و SIDE='L'، يتم استبدال C عند الخروج بـ Q**H * C.
!>
!>  إذا كان TRANS='N' و SIDE='R'، يتم استبدال C عند الخروج بـ C * Q.
!>
!>  إذا كان TRANS='C' و SIDE='R'، يتم استبدال C عند الخروج بـ C * Q**H.
!>
