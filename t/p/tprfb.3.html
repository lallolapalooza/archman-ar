tprfb(3) دليل دوال المكتبة tprfb(3)

الاسم

tprfb - tprfb: يطبق Q (مثل larfb)

الملخص

الدوال


الروتين الفرعي ctprfb (side, trans, direct, storev, m, n, k, l, v, ldv, t, ldt, a, lda, b, ldb, work, ldwork)
CTPRFB يطبق عاكس كتلة مثلثي-خماسي عقدي على مصفوفة عقدية، والتي تتكون من كتلتين. الروتين الفرعي dtprfb (side, trans, direct, storev, m, n, k, l, v, ldv, t, ldt, a, lda, b, ldb, work, ldwork)
DTPRFB يطبق عاكس كتلة مثلثي-خماسي حقيقي على مصفوفة حقيقية، والتي تتكون من كتلتين. الروتين الفرعي stprfb (side, trans, direct, storev, m, n, k, l, v, ldv, t, ldt, a, lda, b, ldb, work, ldwork)
STPRFB يطبق عاكس كتلة مثلثي-خماسي حقيقي على مصفوفة حقيقية، والتي تتكون من كتلتين. الروتين الفرعي ztprfb (side, trans, direct, storev, m, n, k, l, v, ldv, t, ldt, a, lda, b, ldb, work, ldwork)
ZTPRFB يطبق عاكس كتلة مثلثي-خماسي عقدي على مصفوفة عقدية، والتي تتكون من كتلتين.

الوصف المفصل

توثيق الدوال

الروتين الفرعي ctprfb (character side, character trans, character direct, character storev, integer m, integer n, integer k, integer l, complex, dimension( ldv, * ) v, integer ldv, complex, dimension( ldt, * ) t, integer ldt, complex, dimension( lda, * ) a, integer lda, complex, dimension( ldb, * ) b, integer ldb, complex, dimension( ldwork, * ) work, integer ldwork)

CTPRFB يطبق عاكس كتلة "مثلثي-خماسي" عقدي على مصفوفة عقدية، والتي تتكون من كتلتين.

الغرض:

!>
!> CTPRFB يطبق عاكس كتلة عقدي H أو
!> منقوله المترافق H**H على مصفوفة عقدية C، والتي تتكون من كتلتين
!> A و B، إما من اليسار أو من اليمين.
!>
!>

المعاملات

SIDE 
!>          SIDE هو CHARACTER*1
!>          = 'L': تطبيق H أو H**H من اليسار
!>          = 'R': تطبيق H أو H**H من اليمين
!>

TRANS

!>          TRANS هو CHARACTER*1
!>          = 'N': تطبيق H (بدون تبديل)
!>          = 'C': تطبيق H**H (المنقول المترافق)
!>

DIRECT

!>          DIRECT هو CHARACTER*1
!>          يشير إلى كيفية تشكيل H من حاصل ضرب العواكس
!>          الأولية
!>          = 'F': H = H(1) H(2) . . . H(k) (إلى الأمام)
!>          = 'B': H = H(k) . . . H(2) H(1) (إلى الخلف)
!>

STOREV

!>          STOREV هو CHARACTER*1
!>          يشير إلى كيفية تخزين المتجهات التي تعرّف العواكس
!>          الأولية:
!>          = 'C': أعمدة
!>          = 'R': صفوف
!>

M

!>          M هو INTEGER
!>          عدد صفوف المصفوفة B.
!>          M >= 0.
!>

N

!>          N هو INTEGER
!>          عدد أعمدة المصفوفة B.
!>          N >= 0.
!>

K

!>          K هو INTEGER
!>          رتبة المصفوفة T، أي عدد العواكس الأولية
!>          التي يعرّف حاصل ضربها عاكس الكتلة.
!>          K >= 0.
!>

L

!>          L هو INTEGER
!>          رتبة الجزء شبه المنحرف من V.
!>          K >= L >= 0.  انظر التفاصيل الإضافية.
!>

V

!>          V هو مصفوفة COMPLEX، البعد
!>                                (LDV,K) إذا كان STOREV = 'C'
!>                                (LDV,M) إذا كان STOREV = 'R' و SIDE = 'L'
!>                                (LDV,N) إذا كان STOREV = 'R' و SIDE = 'R'
!>          المصفوفة الخماسية V، التي تحتوي على العواكس الأولية
!>          H(1), H(2), ..., H(K).  انظر التفاصيل الإضافية.
!>

LDV

!>          LDV هو INTEGER
!>          البعد الرئيسي للمصفوفة V.
!>          إذا كان STOREV = 'C' و SIDE = 'L'، فإن LDV >= max(1,M);
!>          إذا كان STOREV = 'C' و SIDE = 'R'، فإن LDV >= max(1,N);
!>          إذا كان STOREV = 'R'، فإن LDV >= K.
!>

T

!>          T هو مصفوفة COMPLEX، البعد (LDT,K)
!>          المصفوفة المثلثية K-by-K T في تمثيل
!>          عاكس الكتلة.
!>

LDT

!>          LDT هو INTEGER
!>          البعد الرئيسي للمصفوفة T.
!>          LDT >= K.
!>

A

!>          A هو مصفوفة COMPLEX، البعد
!>          (LDA,N) إذا كان SIDE = 'L' أو (LDA,K) إذا كان SIDE = 'R'
!>          عند الإدخال، المصفوفة K-by-N أو M-by-K A.
!>          عند الإخراج، تُستبدل A بالكتلة المقابلة من
!>          H*C أو H**H*C أو C*H أو C*H**H.  انظر التفاصيل الإضافية.
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرئيسي للمصفوفة A.
!>          إذا كان SIDE = 'L'، فإن LDA >= max(1,K);
!>          إذا كان SIDE = 'R'، فإن LDA >= max(1,M).
!>

B

!>          B هو مصفوفة COMPLEX، البعد (LDB,N)
!>          عند الإدخال، المصفوفة M-by-N B.
!>          عند الإخراج، تُستبدل B بالكتلة المقابلة من
!>          H*C أو H**H*C أو C*H أو C*H**H.  انظر التفاصيل الإضافية.
!>

LDB

!>          LDB هو INTEGER
!>          البعد الرئيسي للمصفوفة B.
!>          LDB >= max(1,M).
!>

WORK

!>          WORK هو مصفوفة COMPLEX، البعد
!>          (LDWORK,N) إذا كان SIDE = 'L'،
!>          (LDWORK,K) إذا كان SIDE = 'R'.
!>

LDWORK

!>          LDWORK هو INTEGER
!>          البعد الرئيسي للمصفوفة WORK.
!>          إذا كان SIDE = 'L'، فإن LDWORK >= K;
!>          إذا كان SIDE = 'R'، فإن LDWORK >= M.
!>

تفاصيل إضافية:

!>
!>  المصفوفة C هي مصفوفة مركبة مكونة من الكتلتين A و B.
!>  الكتلة B بحجم M-by-N؛ إذا كان SIDE = 'R'، فإن A بحجم M-by-K،
!>  وإذا كان SIDE = 'L'، فإن A بحجم K-by-N.
!>
!>  إذا كان SIDE = 'R' و DIRECT = 'F'، فإن C = [A B].
!>
!>  إذا كان SIDE = 'L' و DIRECT = 'F'، فإن C = [A]
!>                                      [B].
!>
!>  إذا كان SIDE = 'R' و DIRECT = 'B'، فإن C = [B A].
!>
!>  إذا كان SIDE = 'L' و DIRECT = 'B'، فإن C = [B]
!>                                      [A].
!>
!>  المصفوفة الخماسية V مكونة من كتلة مستطيلة V1 وكتلة
!>  شبه منحرفة V2.  يُحدَّد حجم الكتلة شبه المنحرفة بواسطة
!>  المعامل L، حيث 0<=L<=K.  إذا كان L=K، فإن كتلة V2 من V مثلثية؛
!>  إذا كان L=0، فلا توجد كتلة شبه منحرفة، وبالتالي V = V1 مستطيلة.
!>
!>  إذا كان DIRECT = 'F' و STOREV = 'C':  V = [V1]
!>                                         [V2]
!>     - V2 شبه منحرفة عليا (أول L صفوف من المثلثية العليا K-by-K)
!>
!>  إذا كان DIRECT = 'F' و STOREV = 'R':  V = [V1 V2]
!>
!>     - V2 شبه منحرفة سفلى (أول L أعمدة من المثلثية السفلى K-by-K)
!>
!>  إذا كان DIRECT = 'B' و STOREV = 'C':  V = [V2]
!>                                         [V1]
!>     - V2 شبه منحرفة سفلى (آخر L صفوف من المثلثية السفلى K-by-K)
!>
!>  إذا كان DIRECT = 'B' و STOREV = 'R':  V = [V2 V1]
!>
!>     - V2 شبه منحرفة عليا (آخر L أعمدة من المثلثية العليا K-by-K)
!>
!>  إذا كان STOREV = 'C' و SIDE = 'L'، فإن V هي M-by-K مع V2 بحجم L-by-K.
!>
!>  إذا كان STOREV = 'C' و SIDE = 'R'، فإن V هي N-by-K مع V2 بحجم L-by-K.
!>
!>  إذا كان STOREV = 'R' و SIDE = 'L'، فإن V هي K-by-M مع V2 بحجم K-by-L.
!>
!>  إذا كان STOREV = 'R' و SIDE = 'R'، فإن V هي K-by-N مع V2 بحجم K-by-L.
