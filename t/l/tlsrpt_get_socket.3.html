tlsrpt_get_socket(3)

الحزمة: libtlsrpt

التراخيص: LGPL-3.0-or-later

TLSRPT_GET_SOCKET(3) tlsrpt_get_socket TLSRPT_GET_SOCKET(3)

الاسم

tlsrpt_get_socket - يعيد واصف ملف المقبس المستخدم ضمن `tlsrpt_connection_t`

الملخص

#include <tlsrpt.h>

int tlsrpt_get_socket(truct tlsrpt_connection_t* con)

الوصف

تعيد الدالة tlsrpt_get_socket واصف ملف المقبس المستخدم ضمن tlsrpt_connection_t. يمكن أن يكون هذا مفيدًا لتعيين خيارات المقبس.

قيمة الإرجاع

تعيد الدالة tlsrpt_get_socket واصف ملف المقبس المستخدم ضمن tlsrpt_connection_t.

انظر أيضًا

tlsrpt_strerror(3), tlsrpt_error_code_is_internal(3)
