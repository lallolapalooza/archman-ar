tls_read(3)

الحزمة: libretls

التراخيص: ISC, BSD, MIT

TLS_READ(3) دليل وظائف المكتبة TLS_READ(3)

الاسم

tls_read، tls_write، tls_handshake، tls_error، tls_close، tls_reset — استخدام اتصال TLS

الملخص

#include <tls.h>

ssize_t
tls_read(struct tls *ctx، void *buf، size_t buflen).

ssize_t
tls_write(struct tls *ctx، const void *buf، size_t buflen).

int
tls_handshake(struct tls *ctx).

const char *
tls_error(struct tls *ctx).

int
tls_close(struct tls *ctx).

void
tls_reset(struct tls *ctx).

الوصف

tls_read() تقرأ buflen بايت من البيانات من المقبس إلى buf. تعيد مقدار البيانات المقروءة.

tls_write() تكتب buflen بايت من البيانات من buf إلى المقبس. تعيد مقدار البيانات المكتوبة.

tls_handshake() تنفذ مصافحة TLS بشكل صريح. هذا ضروري فقط إذا كنت تريد ضمان اكتمال المصافحة، حيث أن كل من tls_read() و tls_write() تقومان تلقائيًا بإجراء مصافحة TLS عند الضرورة.

يمكن استخدام الدالة tls_error() لجلب سلسلة نصية تحتوي على مزيد من المعلومات حول آخر خطأ متعلق بسياق.

tls_close() تغلق اتصالاً بعد الاستخدام. سيتم إغلاق طبقة TLS فقط، وسيكون المتصل مسؤولاً عن إغلاق واصفات الملفات، ما لم يتم إنشاء الاتصال باستخدام tls_connect(3) أو tls_connect_servername(3). بعد إغلاق الاتصال، يمكن تمرير ctx إلى tls_free(3).

قيم الإرجاع

tls_read() و tls_write() تعيدان حجمًا عند النجاح أو -1 عند حدوث خطأ.

tls_handshake() و tls_close() تعيدان 0 عند النجاح أو -1 عند حدوث خطأ.

الدوال tls_read()، tls_write()، tls_handshake()، و tls_close() لها أيضًا قيمتان خاصتان للإرجاع:

TLS_WANT_POLLIN
يحتاج واصف الملف المقروء الأساسي إلى أن يكون قابلاً للقراءة للاستمرار.
TLS_WANT_POLLOUT
يحتاج واصف الملف المكتوب الأساسي إلى أن يكون قابلاً للكتابة للاستمرار.

في حالة واصفات الملفات التي تحظر، يجب تكرار نفس استدعاء الدالة فورًا. في حالة واصفات الملفات التي لا تحظر، يجب تكرار نفس استدعاء الدالة عندما يتم استيفاء الشرط المطلوب.

لا يمكن للمتصلين بهذه الدوال الاعتماد على قيمة المتغير العام errno. لمنع سوء التعامل مع حالات الخطأ، تقوم كل من tls_read()، tls_write()، tls_handshake()، و tls_close() بمسح errno بشكل صريح.

tls_error() تعيد NULL إذا لم يحدث خطأ مع ctx أثناء أو منذ آخر استدعاء لـ tls_handshake()، tls_read()، tls_write()، tls_close()، أو tls_reset() التي تشمل ctx، أو إذا فشل تخصيص الذاكرة أثناء محاولة تجميع السلسلة النصية التي تصف آخر خطأ متعلق بـ ctx.

أمثلة

يوضح المثال التالي كيفية التعامل مع كتابات TLS على واصف ملف يحظر:

...
while (len > 0) {
	ssize_t ret;

	ret = tls_write(ctx, buf, len);
	if (ret == TLS_WANT_POLLIN || ret == TLS_WANT_POLLOUT)
		continue;
	if (ret == -1)
		errx(1, "tls_write: %s", tls_error(ctx));
	buf += ret;
	len -= ret;
}
...

يوضح المثال التالي كيفية التعامل مع كتابات TLS على واصف ملف لا يحظر باستخدام poll(2):

...
pfd[0].fd = fd;
pfd[0].events = POLLIN|POLLOUT;
while (len > 0) {
	nready = poll(pfd, 1, 0);
	if (nready == -1)
		err(1, "poll");
	if ((pfd[0].revents & (POLLERR|POLLNVAL)))
		errx(1, "bad fd %d", pfd[0].fd);
	if ((pfd[0].revents & (pfd[0].events|POLLHUP))) {
		ssize_t ret;

		ret = tls_write(ctx, buf, len);
		if (ret == TLS_WANT_POLLIN)
			pfd[0].events = POLLIN;
		else if (ret == TLS_WANT_POLLOUT)
			pfd[0].events = POLLOUT;
		else if (ret == -1)
			errx(1, "tls_write: %s", tls_error(ctx));
		else {
			buf += ret;
			len -= ret;
		}
	}
}
...

التاريخ

tls_read()، tls_write()، tls_error()، tls_close()، و tls_reset() ظهرت في OpenBSD 5.6 وحصلت على أسمائها النهائية في OpenBSD 5.7.

tls_handshake() ظهرت في OpenBSD 5.9.

تحذيرات

الدالة tls_error() تعيد مؤشرًا داخليًا. لا يجب تحريره بواسطة التطبيق، وإلا سيحدث خطأ تحرير مزدوج. سيصبح المؤشر غير صالح عند حدوث الخطأ التالي مع ctx. بالتالي، إذا كان التطبيق قد يحتاج الرسالة في وقت لاحق، فيجب عليه نسخ السلسلة النصية قبل استدعاء دالة libtls التالية التي تشمل ctx، وإلا فمن المحتمل حدوث خطأ تجزئة أو قراءة بيانات غير مقصودة.
