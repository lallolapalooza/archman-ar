Tk_GetItemTypes(3)

الحزمة: tk

التراخيص: TCL

Tk_CreateItemType(3) إجراءات مكتبة Tk Tk_CreateItemType(3)

الاسم

Tk_CreateItemType, Tk_GetItemTypes - تعريف نوع جديد من عناصر اللوحة

الملخص

#include <tk.h>

Tk_CreateItemType(typePtr)

Tk_ItemType *
Tk_GetItemTypes()

الوسائط

Tk_ItemType *typePtr (داخل)
هيكل يحدد النوع الجديد لعنصر اللوحة.

مقدمة

يتم استدعاء Tk_CreateItemType لتعريف نوع جديد من عناصر اللوحة يصفه الوسيط typePtr. يتوافق نوع العنصر مع قيمة معينة للوسيط type لأمر لوحة الرسم widget create، والكود الذي ينفذ نوع عنصر اللوحة يسمى مدير النوع. تحدد Tk عدة أنواع عناصر مدمجة، مثل rectangle و text و image، ولكن Tk_CreateItemType يسمح بتعريف أنواع عناصر إضافية. بمجرد أن تعود Tk_CreateItemType، يمكن استخدام نوع العنصر الجديد في لوحات الرسم widget الجديدة أو الموجودة تمامًا مثل أنواع العناصر المدمجة.

تعيد Tk_GetItemTypes مؤشرًا إلى الأول في قائمة جميع أنواع العناصر المعرفة حاليًا للوحات. ترتبط الإدخالات في القائمة معًا من خلال حقول nextPtr الخاصة بها، مع تمييز نهاية القائمة بواسطة nextPtr بقيمة NULL.

قد تجد أنه من الأسهل فهم بقية إدخال الدليل هذا من خلال النظر إلى كود نوع عنصر لوحة موجود بالفعل مثل bitmap (في الملف tkCanvBmap.c) أو text (tkCanvText.c). أسهل طريقة لإنشاء مدير نوع جديد هي نسخ الكود لنوع موجود وتعديله للنوع الجديد.

توفر Tk عددًا من الإجراءات المساعدة لاستخدام مديري أنواع اللوحات، مثل Tk_CanvasCoords و Tk_CanvasPsColor؛ يتم وصف هذه الإجراءات في إدخالات دليل منفصلة.

هياكل البيانات

يتكون مدير النوع من مجموعة من الإجراءات التي توفر مجموعة قياسية من العمليات على عناصر هذا النوع. يتعامل مدير النوع مع ثلاثة أنواع من هياكل البيانات. الهيكل البياني الأول هو Tk_ItemType؛ يحتوي على معلومات مثل اسم النوع والمؤشرات إلى الإجراءات القياسية التي ينفذها مدير النوع:

typedef struct Tk_ItemType {
    const char *name;
    int itemSize;
    Tk_ItemCreateProc *createProc;
    const Tk_ConfigSpec *configSpecs;
    Tk_ItemConfigureProc *configProc;
    Tk_ItemCoordProc *coordProc;
    Tk_ItemDeleteProc *deleteProc;
    Tk_ItemDisplayProc *displayProc;
    int alwaysRedraw;
    Tk_ItemPointProc *pointProc;
    Tk_ItemAreaProc *areaProc;
    Tk_ItemPostscriptProc *postscriptProc;
    Tk_ItemScaleProc *scaleProc;
    Tk_ItemTranslateProc *translateProc;
    Tk_ItemIndexProc *indexProc;
    Tk_ItemCursorProc *icursorProc;
    Tk_ItemSelectionProc *selectionProc;
    Tk_ItemInsertProc *insertProc;
    Tk_ItemDCharsProc *dCharsProc;
    Tk_ItemType *nextPtr;
} Tk_ItemType;

يتم وصف حقول هيكل Tk_ItemType بمزيد من التفصيل لاحقًا في إدخال الدليل هذا. عندما يتم استدعاء Tk_CreateItemType، يجب أن يشير وسيطه typePtr إلى هيكل مع تهيئة جميع الحقول باستثناء nextPtr، والذي يقوم Tk بتعيينه لربط جميع الأنواع معًا في قائمة. يجب أن يكون الهيكل في ذاكرة دائمة (إما مخصصة ثابتًا أو مخصصة ديناميكيًا ولكن لا يتم تحريرها أبدًا)؛ تحتفظ Tk بمؤشر إلى هذا الهيكل.

الهيكل البياني الثاني الذي يتعامل معه مدير النوع هو سجل العنصر. لكل عنصر في اللوحة، يوجد سجل عنصر واحد. عادةً ما تكون لجميع العناصر من نوع معين سجلات عناصر بنفس الهيكل، ولكن الأنواع المختلفة عادةً ما يكون لها تنسيقات مختلفة لسجلات العناصر الخاصة بها. الجزء الأول من كل سجل عنصر هو رأس به هيكل قياسي تحدده Tk عبر نوع Tk_Item؛ الجزء المتبقي من سجل العنصر يتم تحديده بواسطة مدير النوع. يجب على مدير النوع تعريف سجلات عناصره مع Tk_Item كحقل أول. على سبيل المثال، يتم تعريف سجل العنصر لعناصر bitmap على النحو التالي:

typedef struct BitmapItem {
    Tk_Item header;
    double x, y;
    Tk_Anchor anchor;
    Pixmap bitmap;
    XColor *fgColor;
    XColor *bgColor;
    GC gc;
} BitmapItem;

يحتوي الهيكل الفرعي header على معلومات تستخدمها Tk لإدارة العنصر، مثل معرفه، وعلاماته، ونوعه، ومربع الإحاطة الخاص به. الحقول التي تبدأ بـ x تنتمي إلى مدير النوع: لن تقرأها أو تكتبها Tk أبدًا. لا يحتاج مدير النوع إلى قراءة أو كتابة أي من الحقول في الرأس باستثناء أربعة حقول أسماؤها x1، y1، x2، و y2. تعطي هذه الحقول مربع إحاطة للعناصر باستخدام إحداثيات اللوحة بالأعداد الصحيحة: لا ينبغي للعنصر أن يغطي أي بكسلات بإحداثي x أقل من x1 أو إحداثي y أقل من y1، ولا ينبغي أن يغطي أي بكسلات بإحداثي x أكبر من أو يساوي x2 أو إحداثي y أكبر من أو يساوي y2. الأمر متروك لمدير النوع للحفاظ على مربع الإحاطة محدثًا مع نقل العنصر وإعادة تكوينه.

في كل مرة تستدعي فيها Tk إجراءً في مدير نوع، فإنها تمرر مؤشرًا إلى سجل عنصر. يتم تمرير الوسيط دائمًا كمؤشر إلى Tk_Item؛ يقوم مدير النوع عادةً بتحويل هذا إلى مؤشر إلى نوعه المحدد، مثل BitmapItem.

الهيكل البياني الثالث الذي تستخدمه مديري الأنواع له نوع Tk_Canvas؛ يعمل كمقبض غير شفاف للوحة widget ككل. لا يحتاج مديري الأنواع إلى معرفة أي شيء عن محتويات هذا الهيكل. يتم تمرير مقبض Tk_Canvas عادةً إلى إجراءات مدير النوع، ويمكن لمدير النوع تمرير المقبض مرة أخرى إلى إجراءات المكتبة مثل Tk_CanvasTkwin لجلب معلومات حول اللوحة.

حقول Tk_ItemType

الاسم

يصف هذا القسم والأقسام التالية كل حقل من حقول هيكل Tk_ItemType بتفصيل. يوفر الحقل name اسمًا نصيًا لنوع العنصر. بمجرد أن تعود Tk_CreateImageType، يمكن استخدام هذا الاسم في أوامر لوحة الرسم widget create لإنشاء عناصر من النوع الجديد. إذا كان نوع عنصر موجود بالفعل بهذا الاسم، فإن النوع الجديد يحل محل النوع القديم.

الأعلام (في ALWAYSREDRAW)

يحتوي الحقل typePtr->alwaysRedraw (يسمى بهذا الاسم لأسباب تاريخية) على مجموعة من بتات الأعلام التي تعدل كيفية تفاعل نواة اللوحة مع العنصر. يتم تعريف الأعلام التالية:

1
يشير إلى أنه يجب إعادة رسم العنصر دائمًا عند إعادة رسم أي جزء من اللوحة، بدلاً من إعادة رسمه فقط عندما تتداخل مساحة العنصر مع المنطقة التي يتم إعادة رسمها. يتم استخدام هذا لعناصر النافذة، على سبيل المثال، والتي تحتاج إلى إلغاء تعيين النوافذ الفرعية التي ليست على الشاشة.
TK_CONFIG_OBJS
يشير إلى أن العمليات التي تأخذ بخلاف ذلك سلسلة (أو مصفوفة من السلاسل) تأخذ في الواقع مرجع Tcl_Obj (أو مصفوفة من المراجع المماثلة). العمليات التي ينطبق عليها هذا هي configProc، و coordProc، و createProc، و indexProc، و insertProc.
TK_MOVABLE_POINTS
يشير إلى أن العنصر يدعم dCharsProc، و indexProc، و insertProc بنفس الدلالات مثل أنواع الخط والمضلع المدمجة في Tk، وأن نقاط الإحداثيات الفردية يمكن نقلها. يجب ألا يتم تعيين هذا إذا كان أي من الطرق المذكورة أعلاه NULL.

حجم العنصر

typePtr->itemSize يعطي الحجم بالبايت لسجلات العناصر من هذا النوع، بما في ذلك رأس Tk_Item. تستخدم Tk هذا الحجم لتخصيص مساحة الذاكرة للعناصر من هذا النوع. يجب أن تكون جميع سجلات العناصر لنوع معين بنفس الحجم. إذا كانت هناك حاجة إلى حقول متغيرة الطول لعنصر (مثل قائمة نقاط لمضلع)، فيمكن لمدير النوع تخصيص كائن منفصل متغير الطول والاحتفاظ بمؤشر إليه في سجل العنصر.

إجراء الإنشاء

typePtr->createProc هو مؤشر إلى إجراء يتم استدعاؤه عندما يتم إنشاء عنصر من هذا النوع. يجب أن يكون للإجراء التوقيع التالي:

typedef int Tk_ItemCreateProc(
    ClientData clientData,
    Tk_Canvas canvas,
    Tk_Item *itemPtr,
    int argc,
    char *argv[]);

clientData هو قيمة عامة ترتبط بنوع العنصر، يتم توفيرها عند استدعاء Tk_CreateItemType.
canvas هو مقبض اللوحة widget التي سيتم إنشاء العنصر فيها.
itemPtr هو مؤشر إلى سجل العنصر الذي سيتم تخصيصه وتكوينه. يجب أن يتلقى هذا الإجراء السجلات من النوع الصحيح (على سبيل المثال، BitmapItem *).
argc و argv هي وسائط سطر الأوامر التي تم تمريرها إلى أمر لوحة الرسم widget create.
يجب أن يقوم الإجراء بتكوين العنصر بناءً على الوسائط argc و argv، ويجب أن يقوم بتحديث مربع الإحاطة للعنصر (في حقول x1، y1، x2، y2 في رأس العنصر). يجب أن يعيد 0 في حالة النجاح، و 1 في حالة الفشل (على سبيل المثال، إذا كانت الوسائط غير صالحة).

CONFIGSPECS

typePtr->configSpecs هو مؤشر إلى صفيف ثابت من Tk_ConfigSpec هياكل، والتي تحدد خيارات التكوين التي تدعمها هذا النوع من العناصر. انظر Tk_ConfigureInfo و Tk_ConfigureValue للحصول على تفاصيل حول Tk_ConfigSpec. يجب أن يتضمن الصفيف إدخالًا لـ NULL لتمييز نهايته.

إجراء التكوين

typePtr->configProc هو مؤشر إلى إجراء يتم استدعاؤه لتغيير خيارات التكوين لعنصر موجود. يجب أن يكون للإجراء التوقيع التالي:

typedef void Tk_ItemConfigureProc(
    Tk_Item *itemPtr,
    Tk_Canvas canvas,
    int argc,
    char *argv[]);

يجب أن يقوم الإجراء بتحديث العنصر الذي يشير إليه itemPtr بناءً على الوسائط argc و argv. بعد تغيير العنصر، يجب على الإجراء تحديث مربع الإحاطة للعنصر (في الحقول x1، y1، x2، y2 في رأس العنصر).

إجراء الإحداثيات

typePtr->coordProc هو مؤشر إلى إجراء يتم استدعاؤه عندما يتم استدعاء أمر لوحة الرسم widget coords لهذا العنصر. يجب أن يكون للإجراء التوقيع التالي:

typedef int Tk_ItemCoordProc(
    Tk_Item *itemPtr,
    Tk_Canvas canvas,
    int argc,
    char *argv[]);

يجب أن يقوم الإجراء بتحديث إحداثيات العنصر بناءً على الوسائط argc و argv. بعد تغيير الإحداثيات، يجب على الإجراء تحديث مربع الإحاطة للعنصر.

إجراء الحذف

typePtr->deleteProc هو مؤشر إلى إجراء يتم استدعاؤه عندما يتم حذف العنصر. يجب أن يكون للإجراء التوقيع التالي:

typedef void Tk_ItemDeleteProc(
    Tk_Item *itemPtr,
    Tk_Canvas canvas);

يجب أن يقوم الإجراء بتحرير أي ذاكرة أو موارد أخرى تم تخصيصها بشكل خاص لهذا العنصر.

إجراء العرض

typePtr->displayProc هو مؤشر إلى إجراء يتم استدعاؤه لرسم العنصر. يجب أن يكون للإجراء التوقيع التالي:

typedef void Tk_ItemDisplayProc(
    Tk_Item *itemPtr,
    Tk_Canvas canvas,
    Display *display,
    Drawable drawable);

يجب على الإجراء رسم العنصر المحدد في drawable باستخدام display المحدد. يجب أن يقوم الإجراء بإجراء أي تحويلات ضرورية للإحداثيات (مثل تحويل إحداثيات اللوحة إلى إحداثيات الشاشة).

إجراء النقطة

typePtr->pointProc هو مؤشر إلى إجراء يتم استدعاؤه لتحديد ما إذا كانت نقطة معينة تقع داخل العنصر. يجب أن يكون للإجراء التوقيع التالي:

typedef int Tk_ItemPointProc(
    Tk_Item *itemPtr,
    int x,
    int y,
    Tk_Canvas canvas);

يجب أن يعيد الإجراء 1 إذا كانت النقطة (x، y) تقع داخل العنصر، و 0 بخلاف ذلك.

إجراء المنطقة

typePtr->areaProc هو مؤشر إلى إجراء يتم استدعاؤه لتحديد ما إذا كان مستطيل معين يتداخل مع العنصر. يجب أن يكون للإجراء التوقيع التالي:

typedef int Tk_ItemAreaProc(
    Tk_Item *itemPtr,
    double x1,
    double y1,
    double x2,
    double y2,
    Tk_Canvas canvas);

يجب أن يعيد الإجراء 1 إذا كان المستطيل المحدد بواسطة (x1، y1) و (x2، y2) يتداخل مع العنصر، و 0 بخلاف ذلك.

إجراء ما بعد النص

typePtr->postscriptProc هو مؤشر إلى إجراء يتم استدعاؤه لتوليد رمز ما بعد النص للعنصر. يجب أن يكون للإجراء التوقيع التالي:

typedef int Tk_ItemPostscriptProc(
    Tk_Item *itemPtr,
    Tk_Canvas canvas,
    int argc,
    char *argv[]);

يجب أن يقوم الإجراء بطباعة رمز ما بعد النص للعنصر إلى دفق الإخراج القياسي. يجب أن يعيد 0 في حالة النجاح، و 1 في حالة الفشل.

إجراء القياس

typePtr->scaleProc هو مؤشر إلى إجراء يتم استدعاؤه لتغيير حجم العنصر. يجب أن يكون للإجراء التوقيع التالي:

typedef void Tk_ItemScaleProc(
    Tk_Item *itemPtr,
    Tk_Canvas canvas,
    double x,
    double y,
    double scale);

يجب على الإجراء تغيير حجم العنصر حول النقطة (x، y) بواسطة العامل scale. بعد تغيير حجم العنصر، يجب على الإجراء تحديث مربع الإحاطة للعنصر.

إجراء الترجمة

typePtr->translateProc هو مؤشر إلى إجراء يتم استدعاؤه لنقل العنصر. يجب أن يكون للإجراء التوقيع التالي:

typedef void Tk_ItemTranslateProc(
    Tk_Item *itemPtr,
    Tk_Canvas canvas,
    double dx,
    double dy);

يجب على الإجراء نقل العنصر بواسطة dx في الاتجاه x و dy في الاتجاه y. بعد نقل العنصر، يجب على الإجراء تحديث مربع الإحاطة للعنصر.

إجراء الفهرس

typePtr->indexProc هو مؤشر إلى إجراء يتم استدعاؤه لتحديد فهرس حرف أو نقطة داخل العنصر. يجب أن يكون للإجراء التوقيع التالي:

typedef int Tk_ItemIndexProc(
    Tk_Item *itemPtr,
    Tk_Canvas canvas,
    int argc,
    char *argv[]);

يجب أن يفسر الإجراء الوسائط argc و argv لتحديد فهرس حرف أو نقطة داخل العنصر. على سبيل المثال، بالنسبة لعنصر نصي، قد تحدد الوسائط فهرس حرف معين. يجب أن يعيد الفهرس المطلوب. القيم الخاصة مثل "end" يجب أن يتم تفسيرها أيضًا.

إجراء المؤشر

typePtr->icursorProc هو مؤشر إلى إجراء يتم استدعاؤه لتحديد موضع مؤشر الإدراج داخل العنصر. يجب أن يكون للإجراء التوقيع التالي:

typedef int Tk_ItemCursorProc(
    Tk_Item *itemPtr,
    Tk_Canvas canvas,
    int index);

يجب أن يحسب الإجراء موضع إحداثيات الشاشة لمؤشر الإدراج عند index المحدد داخل العنصر.

إجراء التحديد

typePtr->selectionProc هو مؤشر إلى إجراء يتم استدعاؤه لتحديد ما إذا كان جزء من العنصر محددًا. يجب أن يكون للإجراء التوقيع التالي:

typedef int Tk_ItemSelectionProc(
    Tk_Item *itemPtr,
    Tk_Canvas canvas,
    int x,
    int y);

يجب أن يعيد الإجراء 1 إذا كان العنصر محددًا عند النقطة (x، y)، و 0 بخلاف ذلك.

إجراء الإدراج

typePtr->insertProc هو مؤشر إلى إجراء يتم استدعاؤه لإدراج نص أو عناصر أخرى في العنصر. يجب أن يكون للإجراء التوقيع التالي:

typedef int Tk_ItemInsertProc(
    Tk_Item *itemPtr,
    Tk_Canvas canvas,
    int index,
    char *string);

يجب أن يقوم الإجراء بإدراج string في العنصر عند index المحدد. يجب أن يعيد الفهرس الجديد للحرف بعد الإدراج، أو -1 في حالة الفشل.

إجراء الأحرف المحذوفة

typePtr->dCharsProc هو مؤشر إلى إجراء يتم استدعاؤه لحذف أحرف من العنصر. يجب أن يكون للإجراء التوقيع التالي:

typedef int Tk_ItemDCharsProc(
    Tk_Item *itemPtr,
    Tk_Canvas canvas,
    int first,
    int last);

يجب أن يقوم الإجراء بحذف الأحرف من first إلى last (شاملة) من العنصر. يجب أن يعيد 0 في حالة النجاح، و 1 في حالة الفشل.

NEXTPTR

typePtr->nextPtr هو مؤشر إلى عنصر Tk_ItemType التالي في القائمة. يجب ألا يتم تعيين هذا الحقل بواسطة المبرمج؛ يتم استخدامه بواسطة Tk لربط جميع أنواع العناصر معًا.

أمثلة

توجد أمثلة حول كيفية استخدام Tk_CreateItemType في ملفات مصدر Tk، مثل tkCanvBmap.c (لعناصر bitmap) و tkCanvText.c (لعناصر النص).

انظر أيضًا

Tk_CanvasCoords، Tk_CanvasPsColor، Tk_ConfigureInfo، Tk_ConfigureValue
