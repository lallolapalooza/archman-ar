Tk__Optionmenu(3pm)

الحزمة: perl-tk

التراخيص: PerlArtistic, GPL, custom

Optionmenu(3) توثيق مساهمات Perl Optionmenu(3)

الاسم

Tk::Optionmenu - دع المستخدم يختار إحدى قيم الخيارات المعرفة مسبقًا

الملخص

use Tk::Optionmenu;
$opt = $w->Optionmenu(
                    -options => REFERENCE_to_OPTIONLIST,
                    -command => CALLBACK,
                    -variable => SCALAR_REF,
                    );
$opt->addOptions( OPTIONLIST );
# OPTION LIST هو
#   أ)  $val1, $val2, $val3,...
#   ب)  [ $lab1=>$val1], [$lab2=>val2], ... ]
#   ج)  مزيج من أ) و ب)، على سبيل المثال،
#       val1, [$lab2=>val2], val3, val4, [...], ...

الوصف

تسمح أداة Optionmenu للمستخدم بالاختيار بين مجموعة معينة من الخيارات.

إذا كان يجب أن يكون المستخدم قادرًا على تغيير الخيارات المتاحة، فراجع Tk::BrowseEntry.

الخيارات

-options
تعيين (إعادة تعيين) قائمة الخيارات المعروضة.
-command
يحدد رد الاتصال الذي يتم استدعاؤه عند تحديد خيار جديد.
-variable
مرجع إلى قيمة العدد الصحيح التي تحتوي على القيمة الحالية لل خيار المحدد.
-textvariable
مرجع إلى قيمة العدد الصحيح التي تحتوي على تسمية النص للقيمة الحالية لل خيار المحدد.

الطرق

addOptions
يضيف OPTION_LIST إلى الخيارات المتاحة بالفعل.

مثال

use Tk;
my $mw = MainWindow->new();
my ($var, $tvar);
my $opt = $mw->Optionmenu(
       -options => [[jan=>1], [feb=>2], [mar=>3], [apr=>4]],
       -command => sub { print "got: ", shift, "\n" },
       -variable => \$var,
       -textvariable => \$tvar
      )->pack;
$opt->addOptions([may=>5],[jun=>6],[jul=>7],[aug=>8]);
my $f = $mw->Frame(-relief=>'groove', -borderwidth => 2)->pack;
$f->Label(-textvariable=>\$tvar)->pack(-side => 'left');
$f->Label(-text => " -> ")->pack(-side => 'left');
$f->Label(-textvariable=>\$var)->pack(-side => 'left');
$mw->Button(-text=>'Exit', -command=>sub{$mw->destroy})->pack;
MainLoop;
