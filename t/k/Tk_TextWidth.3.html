Tk_TextWidth(3)

الحزمة: tk

التراخيص: TCL

Tk_MeasureChars(3) إجراءات مكتبة Tk Tk_MeasureChars(3)

الاسم

Tk_MeasureChars, Tk_TextWidth, Tk_DrawChars, Tk_UnderlineChars - روتينات لقياس وعرض سلاسل نصية بسيطة أحادية السطر.

الملخص

#include <tk.h>

int
Tk_MeasureChars(tkfont, string, numBytes, maxPixels, flags, lengthPtr)

int
Tk_TextWidth(tkfont, string, numBytes)

Tk_DrawChars(display, drawable, gc, tkfont, string, numBytes, x, y)

Tk_UnderlineChars(display, drawable, gc, tkfont, string, x, y, firstByte, lastByte)

المعاملات

Tk_Font tkfont (في)
رمز للخط الذي سيتم رسم النص به أو قياسه. يجب أن يكون قد تم إرجاعه بواسطة استدعاء سابق لـ Tk_GetFont.
const char *string (في)
النص المراد قياسه أو عرضه. لا يلزم أن يكون منهياً بـ null. سيتم قياس أو عرض أي أحرف وصفية غير مطبوعة في السلسلة (مثل علامات الجدولة، الأسطر الجديدة، وأحرف التحكم الأخرى) بطريقة تعتمد على النظام الأساسي.
int numBytes (في)
الحد الأقصى لعدد البايتات المراد أخذها في الاعتبار عند قياس أو رسم string. يجب أن يكون أكبر من أو يساوي 0.
int maxPixels (في)
إذا كان maxPixels >= 0، فإنه يحدد أقصى طول مسموح به للسطر بالبكسل. يتم معالجة الأحرف من string فقط حتى يتم تغطية هذا العدد من البكسلات. إذا كان maxPixels < 0، فإن طول السطر غير محدود ويتم تجاهل معامل flags.
int flags (في)
بتات علم مختلفة يتم تجميعها معًا: TK_PARTIAL_OK يعني تضمين حرف طالما أن أي جزء منه يناسب الطول المحدد بواسطة maxPixels؛ بخلاف ذلك، يجب أن يناسب الحرف بالكامل ليتم اعتباره. TK_WHOLE_WORDS يعني التوقف عند حدود الكلمة، إن أمكن. إذا تم تعيين TK_AT_LEAST_ONE، فيعني ذلك إرجاع حرف واحد على الأقل حتى لو لم تتمكن أي أحرف من الملاءمة في الطول المحدد بواسطة maxPixels. إذا تم تعيين TK_AT_LEAST_ONE وتم تعيين TK_WHOLE_WORDS أيضًا، فهذا يعني أنه إذا لم يكن حتى كلمة واحدة مناسبة للسطر، فقم بإرجاع الأحرف القليلة الأولى من الكلمة التي ناسبت؛ إذا لم يكن حتى حرف واحد من الكلمة مناسبًا، فسيظل الحرف الأول يتم إرجاعه.
int *lengthPtr (خارج)
يتم ملؤه بعدد البكسلات التي تشغلها عدد الأحرف التي تم إرجاعها كنتيجة لـ Tk_MeasureChars.
Display *display (في)
الشاشة المراد الرسم عليها.
Drawable drawable (في)
النافذة أو بيكس ماب للرسم فيها.
GC gc (في)
سياق الرسوميات لرسم الأحرف. يجب أن يكون الخط المحدد في هذا السياق هو نفس الخط tkfont.
int x, y (في)
الإحداثيات التي سيتم وضع الحافة اليسرى للخط الأساسي عندها عند عرض string.
int firstByte (في)
فهرس أول بايت لأول حرف سيتم تسطيره في string. يبدأ التسطير عند الحافة اليسرى لهذا الحرف.
int lastByte (في)
فهرس أول بايت لآخر حرف سيتم رسم التسطير حتى عنده. الحرف المحدد بواسطة lastByte لن يتم تسطيره بنفسه.

الوصف

تُستخدم هذه الروتينات لقياس وعرض السلاسل النصية البسيطة ذات الخط الواحد والسطر الواحد. لقياس وعرض نص متعدد الأسطر، ذي خط واحد، ومبرر، ارجع إلى وثائق Tk_ComputeTextLayout. لا توجد واجهة برمجة في نواة Tk تدعم النص متعدد الخطوط ومتعدد الأسطر؛ يجب بناء الدعم لهذا السلوك على طبقات أبسط. لاحظ أن الواجهات الموضحة هنا موجهة نحو البايت وليس الأحرف، لذا فإن قيم الفهرس القادمة من نصوص Tcl تحتاج إلى تحويلها إلى إزاحات بايت باستخدام Tcl_UtfAtIndex والروتينات ذات الصلة.

الرمز (glyph) هو الصورة القابلة للعرض لحرف أو رقم أو أي رمز آخر. ليست كل رموز الأحرف في خط معين لها رمز. يتم قياس وعرض الأحرف مثل علامات الجدولة، الأسطر الجديدة/الإرجاع، وأحرف التحكم التي لا تحتوي على رمز بواسطة هذه الإجراءات بطريقة تعتمد على النظام الأساسي؛ تحت X، يتم استبدالها بتسلسلات هروب مسبوقة بعلامة مائلة للخلف، بينما تحت Windows و Macintosh قد يتم استبدالها بصناديق مجوفة أو صلبة. ارجع إلى وثائق Tk_ComputeTextLayout للحصول على واجهة برمجة تدعم التوسع المستقل للنظام الأساسي لأحرف الجدولة إلى أعمدة والأسطر الجديدة/الإرجاع إلى نصوص متعددة الأسطر.

يُستخدم Tk_MeasureChars لكل من حساب طول سلسلة معينة وحساب عدد الأحرف من سلسلة ما التي تناسب مساحة معينة. القيمة المرتجعة هي عدد البايتات من string التي تناسب المساحة المحددة بواسطة maxPixels تخضع للشروط الموضحة بواسطة flags. إذا احتوت جميع الأحرف، فستكون القيمة المرتجعة numBytes. يتم ملء *lengthPtr بالعرض المحسوب، بالبكسل، للجزء من السلسلة الذي تم قياسه. على سبيل المثال، إذا كانت القيمة المرتجعة 5، فسيتم ملء *lengthPtr بالمسافة بين الحافة اليسرى لـ string[0] والحافة اليمنى لـ string[4].

Tk_TextWidth هي دالة غلاف توفر واجهة أبسط للدالة Tk_MeasureChars. القيمة المرتجعة هي المساحة بالبكسل التي تحتاجها string المعطاة.

Tk_DrawChars ترسم string في الموقع المحدد في drawable المحدد.

Tk_UnderlineChars تسطر النطاق المحدد من الأحرف في string المحدد. لا ترسم الأحرف (التي يفترض أنه تم عرضها مسبقًا بواسطة Tk_DrawChars)؛ إنها ترسم التسطير فقط. تُستخدم هذه الإجراءات لتسطير بضعة أحرف دون الحاجة إلى إنشاء خط مسطر. لإنتاج نص مسطر أصلي، يجب إنشاء الخط المسطر المناسب واستخدامه.
