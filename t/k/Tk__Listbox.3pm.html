Tk__Listbox(3pm)

الحزمة: perl-tk

التراخيص: PerlArtistic, GPL, custom

$listbox->curselection
تُرجع قائمة تحتوي على الفهارس الرقمية لجميع العناصر المحددة حاليًا في مربع القائمة. إذا لم يتم تحديد أي عناصر، فسيتم إرجاع سلسلة فارغة.
$listbox->delete(first, ?last?)
تحذف عنصرًا واحدًا أو أكثر من مربع القائمة. `first` و `last` هما فهرسان يحددان العنصرين الأول والأخير المراد حذفهما. إذا لم يتم تحديد `last`، فسيتم استخدام قيمة `first`، أي سيتم حذف عنصر واحد فقط.
$listbox->get(first, ?last?)
إذا لم يتم حذف `last`، تُرجع محتوى العنصر المحدد بالفهرس `first` في مربع القائمة، أو سلسلة فارغة إذا كان `first` يشير إلى عنصر غير موجود. إذا تم تحديد `last`، تُرجع قائمة بجميع العناصر بين `first` و `last`، بما في ذلك كليهما. يمكن أن يكون كل من `first` و `last` أي شكل فهرس قياسي.
$listbox->index(index)
تُرجع القيمة الرقمية للفهرس المقابلة لـ `index`. إذا كان `index` هو `end`، تُرجع عدد العناصر في مربع القائمة (وليس فهرس العنصر الأخير).
$listbox->insert(index, ?element, element, ...?)
تُدرج صفرًا أو أكثر من العناصر الجديدة في مربع القائمة مباشرة قبل العنصر المحدد بواسطة `index`. إذا تم تحديد `index` كـ `end`، فستتم إضافة العناصر الجديدة إلى نهاية مربع القائمة. تُرجع سلسلة فارغة.
$listbox->itemcget(index, option)
تُرجع القيمة الحالية لخيار التكوين للعنصر المحدد بواسطة `option`. يمكن أن يأخذ `option` أي قيمة مقبولة بواسطة أمر `itemconfigure` الخاص بمربع القائمة.
$listbox->itemconfigure(index, ?option, value, option, value, ...?)
تستعلم أو تعدل خيارات تكوين عنصر في مربع القائمة. إذا لم يتم تحديد أي `option`، تُرجع قائمة تصف جميع الخيارات المتاحة للعنصر (راجع Tk_ConfigureInfo لشكل القائمة). إذا تم تحديد `option` مع قيمة، تُرجع قائمة تصف `option` المحدد (ستكون هذه القائمة متطابقة مع القائمة الفرعية المقابلة للقيمة المرجعة إذا لم يتم تحديد أي `option`). إذا تم تحديد زوج أو أكثر من `option`-`value`، فإنها تعدل خيارات الأداة (widget) المقابلة بالقيم المعطاة؛ في هذه الحالة، تُرجع سلسلة فارغة. الخيارات المدعومة حاليًا للعناصر هي:
-background => color
يحدد `color` لون الخلفية المستخدم عند عرض العنصر. يمكن أن يأخذ أي شكل مقبول بواسطة Tk_GetColor.
-foreground => color
يحدد `color` لون المقدمة المستخدم عند عرض العنصر. يمكن أن يأخذ أي شكل مقبول بواسطة Tk_GetColor.
-selectbackground => color
يحدد `color` لون الخلفية المستخدم عند تحديد العنصر. يمكن أن يأخذ أي شكل مقبول بواسطة Tk_GetColor.
-selectforeground => color
يحدد `color` لون المقدمة المستخدم عند تحديد العنصر. يمكن أن يأخذ أي شكل مقبول بواسطة Tk_GetColor.
$listbox->nearest(y)
بالنظر إلى إحداثي y داخل نافذة مربع القائمة، تُرجع فهرس أقرب عنصر (مرئي) لهذا الإحداثي y.
$listbox->scan(option, args)
يُستخدم هذا الأمر لتنفيذ المسح في مربعات القائمة. له شكلان حسب `option`:
$listbox->scanMark(x, y)
يسجل `x` و `y` وحالة العرض الحالية في نافذة مربع القائمة؛ يُستخدم مع أوامر `scan dragto` اللاحقة. عادةً ما يرتبط هذا الأمر بالضغط على زر الماوس على الأداة. يُرجع سلسلة فارغة.
$listbox->scanDragto(x, y.)
يحسب هذا الأمر الفرق بين وسيطات `x` و `y` الخاصة به ووسيطات `x` و `y` الخاصة بآخر `scan mark` للأداة. ثم يقوم بضبط العرض بمقدار 10 أضعاف الفرق في الإحداثيات. يرتبط هذا الأمر عادةً بأحداث حركة الماوس على الأداة، لإنتاج تأثير سحب مربع القائمة بسرعة عالية داخل النافذة. القيمة المرجعة هي سلسلة فارغة.
$listbox->see(index)
يضبط عرض مربع القائمة بحيث يكون العنصر المحدد بواسطة `index` مرئيًا. إذا كان العنصر مرئيًا بالفعل، فلن يكون للأمر أي تأثير؛ إذا كان العنصر قريبًا من حافة النافذة، فسيتم تمرير مربع القائمة بحيث يكون العنصر على الحافة؛ بخلاف ذلك، سيتم تمرير مربع القائمة لتركيز العنصر.
$listbox->selection(option, arg)
يُستخدم هذا الأمر لضبط التحديد داخل مربع القائمة. له أشكال متعددة حسب `option`:
$listbox->selectionAnchor(index)
يضبط نقطة تثبيت التحديد على العنصر المحدد بواسطة `index`. إذا كان `index` يشير إلى عنصر غير موجود، فسيتم استخدام أقرب عنصر. نقطة تثبيت التحديد هي الجزء الثابت من التحديد أثناء سحب التحديد بالماوس. يمكن استخدام الفهرس `anchor` للإشارة إلى نقطة تثبيت التحديد.
$listbox->selectionClear(first, ?last?)
إذا كان أي عنصر بين `first` و `last` (بما في ذلك كليهما) محددًا، فسيتم إلغاء تحديد هذا العنصر. لا يتغير حالة التحديد للعناصر خارج هذا النطاق.
$listbox->selectionIncludes(index)
تُرجع 1 إذا كان العنصر المحدد بواسطة `index` محددًا حاليًا، و 0 إذا لم يكن كذلك.
$listbox->selectionSet(first, ?last?)
يحدد جميع العناصر بين `first` و `last` (بما في ذلك كليهما)، دون التأثير على حالة تحديد العناصر خارج هذا النطاق.
$listbox->size
تُرجع سلسلة عشرية تشير إلى العدد الإجمالي للعناصر في مربع القائمة.
$listbox->xview(args)
يُستخدم هذا الأمر للاستعلام عن الموضع الأفقي للمعلومات في نافذة الأداة وتغييره. يمكن أن يأخذ أيًا من الأشكال التالية:
$listbox->xview
تُرجع قائمة بعنصرين. يمثل كل عنصر جزءًا كسريًا حقيقيًا بين 0 و 1؛ يصفان معًا النطاق الأفقي المرئي في النافذة. على سبيل المثال، إذا كان العنصر الأول هو .2 والعنصر الثاني هو .6، فإن 20٪ من نص مربع القائمة خارج العرض إلى اليسار، والـ 40٪ الوسطى مرئية في النافذة، و 40٪ من النص خارج العرض إلى اليمين. هذه القيم هي نفسها التي يتم تمريرها إلى أشرطة التمرير عبر الخيار `-xscrollcommand`.
$listbox->xview(index)
يضبط العرض في النافذة بحيث يظهر الموضع الحرفي المحدد بواسطة `index` على الحافة اليسرى للنافذة. يتم تحديد المواضع الحرفية بعرض الحرف `0`.
$listbox->xviewMoveto( fraction );
يضبط العرض في النافذة بحيث يكون `fraction` من العرض الكلي لنص مربع القائمة خارج العرض إلى اليسار. يجب أن يكون `fraction` جزءًا كسريًا بين 0 و 1.
$listbox->xviewScroll( number, what );
يقوم هذا الأمر بتمرير العرض في النافذة إلى اليسار أو اليمين بناءً على `number` و `what`. يجب أن يكون `number` عددًا صحيحًا. يجب أن يكون `what` هو `units` أو `pages` أو اختصارًا لأحدهما. إذا كان `what` هو `units`، يتم ضبط العرض إلى اليسار أو اليمين بمقدار `number` وحدة (عرض الحرف `0`)؛ إذا كان `pages`، يتم ضبط العرض بمقدار `number` صفحة كاملة. إذا كان `number` سالبًا، تظهر الأحرف الأبعد إلى اليسار؛ إذا كان موجبًا، تظهر الأحرف الأبعد إلى اليمين.
$listbox->yview(?args?)
يُستخدم هذا الأمر للاستعلام عن الموضع الرأسي للنص في نافذة الأداة وتغييره. يمكن أن يأخذ أيًا من الأشكال التالية:
$listbox->yview
تُرجع قائمة بعنصرين، كلاهما أجزاء كسرية حقيقية بين 0 و 1. الأول يعطي موضع عنصر مربع القائمة في أعلى النافذة، بالنسبة لمربع القائمة ككل (0.5 يعني أنه في منتصف مربع القائمة، على سبيل المثال). الثاني يعطي موضع عنصر مربع القائمة مباشرة بعد العنصر الأخير في النافذة، بالنسبة لمربع القائمة ككل. هذه هي نفس القيم التي يتم تمريرها إلى أشرطة التمرير عبر الخيار `-yscrollcommand`.
$listbox->yview(index)
يضبط العرض في النافذة بحيث يظهر العنصر المحدد بواسطة `index` في أعلى النافذة.
$listbox->yviewMoveto( fraction );
يضبط العرض في النافذة بحيث يظهر العنصر المحدد بواسطة `fraction` في أعلى النافذة. `fraction` هو جزء كسري بين 0 و 1؛ 0 يشير إلى العنصر الأول في مربع القائمة، 0.33 يشير إلى العنصر ثلث مربع القائمة، وهكذا.
$listbox->yviewScroll( number, what );
يقوم هذا الأمر بضبط العرض في النافذة لأعلى أو لأسفل بناءً على `number` و `what`. يجب أن يكون `number` عددًا صحيحًا. يجب أن يكون `what` هو `units` أو `pages`. إذا كان `what` هو `units`، يتم ضبط العرض لأعلى أو لأسفل بمقدار `number` سطر؛ إذا كان `pages`، يتم ضبط العرض بمقدار `number` شاشة كاملة. إذا كان `number` سالبًا، تظهر الأحرف الأبعد إلى اليسار؛ إذا كان موجبًا، تظهر الأحرف الأبعد إلى اليمين.
