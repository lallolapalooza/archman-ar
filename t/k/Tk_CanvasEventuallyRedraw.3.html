Tk_CanvasEventuallyRedraw(3)

الحزمة: tk

التراخيص: TCL

Tk_CanvasTkwin(3) إجراءات مكتبة Tk Tk_CanvasTkwin(3)

الاسم

Tk_CanvasTkwin، Tk_CanvasGetCoord، Tk_CanvasDrawableCoords، Tk_CanvasSetStippleOrigin، Tk_CanvasWindowCoords، Tk_CanvasEventuallyRedraw، Tk_CanvasTagsOption - إجراءات مساعدة لمديري أنواع الـ canvas

الملخص

#include <tk.h>

Tk_Window
Tk_CanvasTkwin(canvas)

int
Tk_CanvasGetCoord(interp, canvas, string, doublePtr)

Tk_CanvasDrawableCoords(canvas, x, y, drawableXPtr, drawableYPtr)

Tk_CanvasSetStippleOrigin(canvas, gc)

Tk_CanvasWindowCoords(canvas, x, y, screenXPtr, screenYPtr)

Tk_CanvasEventuallyRedraw(canvas, x1, y1, x2, y2)

Tk_OptionParseProc *Tk_CanvasTagsParseProc;

Tk_OptionPrintProc *Tk_CanvasTagsPrintProc;

الوصف

يتم استدعاء هذه الإجراءات من قبل مديري أنواع الـ canvas لأداء وظائف مساعدة متنوعة.

Tk_CanvasTkwin تُرجع Tk_Window المرتبطة بـ canvas معين.

Tk_CanvasGetCoord تترجم مواصفات نصية لإحداثي (مثل 2p أو 1.6c) إلى إحداثي canvas مزدوج الدقة. إذا كان string وصفًا صالحًا للإحداثيات، فإن Tk_CanvasGetCoord تخزن الإحداثي المقابل للـ canvas في *doublePtr وتعيد TCL_OK. بخلاف ذلك، تخزن رسالة خطأ في نتيجة المفسر وتعيد TCL_ERROR.

Tk_CanvasDrawableCoords يتم استدعاؤها بواسطة مديري الأنواع أثناء إعادة الرسم لحساب مكان رسم الأشياء. بالنظر إلى إحداثيات x و y في مساحة الـ canvas، تحسب Tk_CanvasDrawableCoords البكسل المقابل في العنصر القابل للرسم الذي يتم استخدامه حاليًا لإعادة الرسم؛ وتعيد تلك الإحداثيات في *drawableXPtr و *drawableYPtr. لا ينبغي استدعاء هذا الإجراء إلا أثناء إعادة الرسم.

Tk_CanvasSetStippleOrigin تستخدم أيضًا أثناء إعادة الرسم. إنها تعين أصل الـ stipple في gc بحيث تتماشى الـ stipples المرسومة باستخدام gc في الـ pixmap خارج الشاشة الحالي مع الـ stipples المرسومة بأصل (0,0) في نافذة الـ canvas الفعلية. تحتاج Tk_CanvasSetStippleOrigin لضمان محاذاة أنماط الـ stipple بشكل صحيح عند إعادة رسم الـ canvas في أجزاء صغيرة. تتم عمليات إعادة الرسم بطريقة الازدواج العازل حيث يتم إعادة رسم جزء من الـ canvas في pixmap خارج الشاشة ثم نسخه مرة أخرى إلى الشاشة. في هذا النهج، تحتاج أصول الـ stipple في السياقات الرسومية إلى التعديل أثناء كل إعادة رسم للتعويض عن موضع الـ pixmap خارج الشاشة بالنسبة للنافذة. إذا تم رسم عنصر باستخدام الـ stipples، فعادةً ما يستدعي مدير النوع الخاص به Tk_CanvasSetStippleOrigin مباشرة قبل استخدام gc لرسم شيء ما؛ بعد الانتهاء من الرسم، يستدعي مدير النوع XSetTSOrigin لاستعادة الأصل في gc مرة أخرى إلى (0,0) (الاستعادة مطلوبة لأن السياقات الرسومية مشتركة، لذلك لا يمكن تعديلها بشكل دائم).

Tk_CanvasWindowCoords مشابه لـ Tk_CanvasDrawableCoords باستثناء أنها تعيد الإحداثيات في نافذة الـ canvas على الشاشة، بدلاً من الإحداثيات في pixmap خارج الشاشة.

يمكن استدعاء Tk_CanvasEventuallyRedraw بواسطة مدير نوع لإبلاغ Tk بأن جزءًا من الـ canvas يحتاج إلى إعادة رسم. تحدد المعاملات x1 و y1 و x2 و y2 المنطقة التي تحتاج إلى إعادة رسم، بإحداثيات الـ canvas. نادرًا ما يحتاج مديرو الأنواع إلى استدعاء Tk_CanvasEventuallyRedraw، حيث يمكن لـ Tk عادةً اكتشاف متى تغير عنصر ما وإجراء طلب إعادة الرسم نيابة عنه (يحدث هذا، على سبيل المثال، كلما استدعى Tk configureProc أو scaleProc). الوقت الوحيد الذي يحتاج فيه مدير نوع إلى استدعاء Tk_CanvasEventuallyRedraw هو إذا تغير عنصر من تلقاء نفسه دون استدعائه من خلال أحد الإجراءات في Tk_ItemType الخاص به؛ قد يحدث هذا، على سبيل المثال، في عنصر صورة إذا تم تعديل الصورة باستخدام أوامر الصورة.

Tk_CanvasTagsParseProc و Tk_CanvasTagsPrintProc هي إجراءات تتعامل مع خيار -tags لعناصر الـ canvas. لن يستدعي كود مدير نوع الـ canvas هذه الإجراءات مباشرة، ولكنه سيستخدم عناوينها لإنشاء بنية Tk_CustomOption للخيار -tags. يبدو الكود عادةً كالتالي:

static const Tk_CustomOption tagsOption = {Tk_CanvasTagsParseProc,
    Tk_CanvasTagsPrintProc, NULL
};
static const Tk_ConfigSpec configSpecs[] = {
    ...
    {TK_CONFIG_CUSTOM, "-tags", NULL, NULL,
        NULL, 0, TK_CONFIG_NULL_OK, &tagsOption},
    ...
};
