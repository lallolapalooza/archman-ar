tk_popup(n)

الحزمة: tk

التراخيص: TCL

tk_popup(n) أوامر Tk المدمجة tk_popup(n)

الاسم

tk_popup - نشر قائمة منبثقة

الملخص

tk_popup menu x y ?entry?

الوصف

يقوم هذا الإجراء بنشر قائمة في موضع معين على الشاشة وتكوين Tk بحيث يمكن التنقل في القائمة وأبنائها المتشعبة باستخدام الفأرة أو لوحة المفاتيح. menu هو اسم عنصر قائمة، و x و y هما إحداثيات الجذر التي سيتم عرض القائمة عندها. إذا تم حذف entry أو كانت سلسلة نصية فارغة، فسيتم وضع الزاوية العلوية اليسرى للقائمة عند النقطة المحددة. بخلاف ذلك، يوفر entry فهرس إدخال في menu وسيتم وضع القائمة بحيث يتم وضع الإدخال فوق النقطة المحددة.

مثال

كيفية ربط قائمة منبثقة بسيطة بعنصر واجهة.

# إنشاء قائمة
set m [menu .popupMenu]
$m add command -label "مثال 1" -command bell
$m add command -label "مثال 2" -command bell
# إنشاء شيء للربط به
pack [label .l -text "انقر هنا!"]
# ترتيب ظهور القائمة عند النقر على الملصق
bind .l <1> {tk_popup .popupMenu %X %Y}

انظر أيضًا

bind(n)، menu(n)، tk_optionMenu(n)
