الاسم

Tk_ParseArgv - معالجة خيارات سطر الأوامر

الملخص

#include <tk.h>

int
Tk_ParseArgv(interp, tkwin, argcPtr, argv, argTable, flags)

الوسائط

Tcl_Interp *interp (في)
المفسر المستخدم لإرجاع رسائل الخطأ.
Tk_Window tkwin (في)
النافذة المستخدمة عندما تحدد الوسائط خيارات Tk. إذا كانت NULL، فلن تتم معالجة أي خيارات Tk.
int argcPtr (في/خارج)
مؤشر إلى عدد الوسائط في argv؛ يتم تعديله للاحتفاظ بعدد الوسائط غير المعالجة المتبقية بعد الاستدعاء.
const char **argv (في/خارج)
وسائط سطر الأوامر التي تم تمريرها إلى البرنامج الرئيسي. يتم تعديلها للاحتفاظ بالوسائط غير المعالجة المتبقية بعد الاستدعاء.
Tk_ArgvInfo *argTable (في)
مصفوفة واصفات الوسائط، تنتهي بعنصر من النوع TK_ARGV_END.
int flags (في)
إذا كانت غير صفرية، فإنها تحدد علامة واحدة أو أكثر تتحكم في تحليل الوسائط. يمكن دمج علامات مختلفة معًا (OR). العلامات المعرفة حاليًا هي TK_ARGV_DONT_SKIP_FIRST_ARG، TK_ARGV_NO_ABBREV، TK_ARGV_NO_LEFTOVERS، و TK_ARGV_NO_DEFAULTS.

الوصف

تقوم Tk_ParseArgv بمعالجة مصفوفة من وسائط سطر الأوامر وفقًا لجدول يصف أنواع الوسائط المتوقعة. تتم معالجة كل من الوسائط في argv بالتتابع: إذا تطابقت مع إحدى الإدخالات في argTable، تتم معالجة الوسيط وفقًا لهذا الإدخال ويتم تجاهله. يتم نسخ الوسائط التي لا تتطابق مع أي شيء في argTable إلى بداية argv (مع الاحتفاظ بترتيبها الأصلي) وإرجاعها إلى المتصل. في نهاية الاستدعاء، تقوم Tk_ParseArgv بتعيين *argcPtr للاحتفاظ بعدد الوسائط المتبقية في argv، وسيكون argv[*argcPtr] القيمة NULL. في العادة، تفترض Tk_ParseArgv أن argv[0] هو اسم أمر، لذا يتم التعامل معه كوسيط لا يتطابق مع argTable ويتم إرجاعه إلى المتصل؛ ومع ذلك، إذا تم تعيين بت TK_ARGV_DONT_SKIP_FIRST_ARG في flags، فسيتم معالجة argv[0] تمامًا مثل العناصر الأخرى في argv.

تقوم Tk_ParseArgv بإرجاع القيمة TCL_OK عادةً. إذا حدث خطأ أثناء تحليل الوسائط، فسيتم إرجاع TCL_ERROR وستترك Tk_ParseArgv رسالة خطأ في نتيجة المفسر interp بالطريقة القياسية لـ Tcl. في حالة إرجاع خطأ، لن يتم تعديل *argvPtr، ولكن قد يتم تعديل argv جزئيًا. يمكن شرح الأسباب المحتملة للأخطاء أدناه.

تحدد مصفوفة argTable أنواع الوسائط المتوقعة؛ كل عنصر من عناصرها له الهيكل التالي:

typedef struct {
    const char *key;
    int type;
    char *src;
    char *dst;
    const char *help;
} Tk_ArgvInfo;
الحقل key هو سلسلة نصية مثل “-display” أو “-bg” تتم مقارنتها بالقيم في argv. يحدد Type كيفية معالجة وسيط يتطابق مع key (المزيد عن هذا أدناه). Src و dst هما قيم إضافية تستخدم في معالجة الوسيط. يعتمد استخدامهما الدقيق على type، ولكن عادةً ما يشير src إلى قيمة ويشير dst إلى مكان تخزين القيمة. إعلانات char * لـ src و dst هي عناصر نائبة: قد تكون الأنواع الفعلية مختلفة. أخيرًا، help هي سلسلة نصية تعطي وصفًا موجزًا لهذا الخيار؛ تتم طباعة هذه السلسلة عند طلب المستخدمين للمساعدة بشأن خيارات سطر الأوامر.

عند معالجة وسيط في argv، تقارن Tk_ParseArgv الوسيط بكل key في argTable. تحدد Tk_ParseArgv أول مواصفات يكون key فيها مطابقًا تمامًا للوسيط، إذا وجدت مثل هذه المواصفات. بخلاف ذلك، تحدد Tk_ParseArgv مواصفات يكون الوسيط اختصارًا فريدًا لها. إذا كان الوسيط اختصارًا فريدًا لأكثر من مواصفات واحدة، فسيتم إرجاع خطأ. إذا لم يكن هناك إدخال مطابق في argTable، فسيتم تخطي الوسيط وإرجاعه إلى المتصل.

بمجرد العثور على مواصفات وسيط مطابق، تقوم Tk_ParseArgv بمعالجة الوسيط وفقًا للحقل type للمواصفات. يطلق على الوسيط الذي تطابق key اسم "الوسيط المتطابق" في الأوصاف أدناه. كجزء من المعالجة، قد تستخدم Tk_ParseArgv أيضًا الوسيط التالي في argv بعد الوسيط المتطابق، والذي يطلق عليه "الوسيط التالي". القيم القانونية لـ type، والمعالجة التي تسببها، هي كما يلي:

TK_ARGV_END
يمثل نهاية الجدول. يجب أن يكون الإدخال الأخير في argTable بهذا النوع؛ سيتم تجاهل جميع حقوله الأخرى ولن يتطابق أبدًا مع أي وسائط.
TK_ARGV_CONSTANT
يعامل Src كرقم صحيح ويعامل dst كمؤشر إلى رقم صحيح. يتم تخزين Src في *dst. يتم تجاهل الوسيط المتطابق.
TK_ARGV_INT
يجب أن يحتوي الوسيط التالي على سلسلة أرقام صحيحة بالتنسيق الذي تقبله strtol (على سبيل المثال، يمكن استخدام البادئات “0” و “0x” لتحديد الأرقام الثمانية أو الستة عشرية، على التوالي). يعامل Dst كمؤشر إلى رقم صحيح؛ يتم تحويل الوسيط التالي إلى قيمة عدد صحيح وتخزينه في *dst. يتم تجاهل Src. يتم تجاهل الوسيط المتطابق والوسيط التالي من argv.
TK_ARGV_FLOAT
يجب أن يحتوي الوسيط التالي على رقم عشري بالتنسيق الذي تقبله strtol. يعامل Dst كعنوان لقيمة فاصلة عائمة مزدوجة الدقة؛ يتم تحويل الوسيط التالي إلى قيمة مزدوجة الدقة وتخزينها في *dst. يتم تجاهل الوسيط المتطابق والوسيط التالي من argv.
TK_ARGV_STRING
في هذا الشكل، يعامل dst كمؤشر إلى (char *)؛ تخزن Tk_ParseArgv في *dst مؤشرًا إلى الوسيط التالي، وتتجاهل الوسيط المتطابق والوسيط التالي من argv. يتم تجاهل Src.
TK_ARGV_UID
هذا الشكل مشابه لـ TK_ARGV_STRING، باستثناء أنه يتم تحويل الوسيط إلى Tk_Uid عن طريق استدعاء Tk_GetUid. يعامل Dst كمؤشر إلى Tk_Uid؛ تخزن Tk_ParseArgv في *dst معرف Tk_Uid المقابل للوسيط التالي، وتتجاهل الوسيط المتطابق والوسيط التالي من argv. يتم تجاهل Src.
TK_ARGV_CONST_OPTION
يتسبب هذا الشكل في تعيين خيار Tk (كما لو تم استدعاء الأمر option). يعامل الحقل src كمؤشر إلى سلسلة نصية تعطي قيمة الخيار، ويعامل dst كمؤشر إلى اسم الخيار. يتم تجاهل الوسيط المتطابق. إذا كان tkwin هو NULL، فسيتم تجاهل مواصفات الوسائط من هذا النوع (كما لو لم تكن موجودة).
TK_ARGV_OPTION_VALUE
هذا الشكل مشابه لـ TK_ARGV_CONST_OPTION، باستثناء أن قيمة الخيار مأخوذة من الوسيط التالي بدلاً من src. يستخدم Dst كاسم للخيار. يتم تجاهل Src. يتم تجاهل الوسيط المتطابق والوسيط التالي. إذا كان tkwin هو NULL، فسيتم تجاهل مواصفات الوسائط من هذا النوع (كما لو لم تكن موجودة).
TK_ARGV_OPTION_NAME_VALUE
في هذه الحالة، يتم أخذ الوسيط التالي كاسم لخيار Tk ويتم أخذ الوسيط الذي يليه كقيمة لهذا الخيار. يتم تجاهل كل من src و dst. يتم تجاهل جميع الوسائط الثلاثة من argv. إذا كان tkwin هو NULL، فسيتم تجاهل مواصفات الوسائط من هذا النوع (كما لو لم تكن موجودة).
TK_ARGV_HELP
عندما يتم مواجهة هذا النوع من الخيارات، تستخدم Tk_ParseArgv الحقول help من argTable لطباع
Tk_ParseArgv(3)

NAME

Tk_ParseArgv - parse a command line into options and arguments

SYNOPSIS

#include <tk.h>

int
Tk_ParseArgv(interp, tkwin, argcPtr, argv, argTable, flags)
    Tcl_Interp *interp;
    Tk_Window tkwin;
    int *argcPtr;
    char **argv;
    Tk_ArgvInfo argTable[];
    int flags;

ARGUMENTS

TK_ARGV_HELP
If an argument in argv matches the key field of a TK_ARGV_HELP specifier, then Tk_ParseArgv formats a message describing all of the valid arguments. The message is placed in interpreter interp's result and Tk_ParseArgv returns TCL_ERROR. When this happens, the caller normally prints the help message and aborts. If the key field of a TK_ARGV_HELP specifier is NULL, then the specifier will never match any arguments; in this case the specifier simply provides extra documentation, which will be included when some other TK_ARGV_HELP entry causes help information to be returned.
TK_ARGV_REST
This option is used by programs or commands that allow the last several of their options to be the name and/or options for some other program. If a TK_ARGV_REST argument is found, then Tk_ParseArgv does not process any of the remaining arguments; it returns them all at the beginning of argv (along with any other unprocessed arguments). In addition, Tk_ParseArgv treats dst as the address of an integer value, and stores at *dst the index of the first of the TK_ARGV_REST options in the returned argv. This allows the program to distinguish the TK_ARGV_REST options from other unprocessed options that preceded the TK_ARGV_REST.
TK_ARGV_FUNC
For this kind of argument, src is treated as the address of a procedure, which is invoked to process the following argument. The procedure should have the following structure:
int
func(dst, key, nextArg)
    char *dst;
    char *key;
    char *nextArg;
{
}
The dst and key parameters will contain the corresponding fields from the argTable entry, and nextArg will point to the following argument from argv (or NULL if there are not any more arguments left in argv). If func uses nextArg (so that Tk_ParseArgv should discard it), then it should return 1. Otherwise it should return 0 and TkParseArgv will process the following argument in the normal fashion. In either event the matching argument is discarded.
TK_ARGV_GENFUNC
This form provides a more general procedural escape. It treats src as the address of a procedure, and passes that procedure all of the remaining arguments. The procedure should have the following form:
int
genfunc(dst, interp, key, argc, argv)
    char *dst;
    Tcl_Interp *interp;
    char *key;
    int argc;
    char **argv;
{
}
The dst and key parameters will contain the corresponding fields from the argTable entry. Interp will be the same as the interp argument to Tcl_ParseArgv. Argc and argv refer to all of the options after the matching one. Genfunc should behave in a fashion similar to Tk_ParseArgv: parse as many of the remaining arguments as it can, then return any that are left by compacting them to the beginning of argv (starting at argv[0]). Genfunc should return a count of how many arguments are left in argv; Tk_ParseArgv will process them. If genfunc encounters an error then it should leave an error message in interpreter interp's result, in the usual Tcl fashion, and return -1; when this happens Tk_ParseArgv will abort its processing and return TCL_ERROR.

FLAGS

TK_ARGV_DONT_SKIP_FIRST_ARG
Tk_ParseArgv normally treats argv[0] as a program or command name, and returns it to the caller just as if it had not matched argTable. If this flag is given, then argv[0] is not given special treatment.
TK_ARGV_NO_ABBREV
Normally, Tk_ParseArgv accepts unique abbreviations for key values in argTable. If this flag is given then only exact matches will be acceptable.
TK_ARGV_NO_LEFTOVERS
Normally, Tk_ParseArgv returns unrecognized arguments to the caller. If this bit is set in flags then Tk_ParseArgv will return an error if it encounters any argument that does not match argTable. The only exception to this rule is argv[0], which will be returned to the caller with no errors as long as TK_ARGV_DONT_SKIP_FIRST_ARG is not specified.
TK_ARGV_NO_DEFAULTS
Normally, Tk_ParseArgv searches an internal table of standard argument specifiers in addition to argTable. If this bit is set in flags, then Tk_ParseArgv will use only argTable and not its default table.

EXAMPLE

Here is an example definition of an argTable and some sample command lines that use the options. Note the effect on argc and argv; arguments processed by Tk_ParseArgv are eliminated from argv, and argc is updated to reflect reduced number of arguments.

/*
 * Define and set default values for globals.
 */
int debugFlag = 0;
int numReps = 100;
char defaultFileName[] = "out";
char *fileName = defaultFileName;
Boolean exec = FALSE;
/*
 * Define option descriptions.
 */
Tk_ArgvInfo argTable[] = {
    {"-X", TK_ARGV_CONSTANT, (char *) 1, (char *) &debugFlag,
        "Turn on debugging printfs"},
    {"-N", TK_ARGV_INT, NULL, (char *) &numReps,
        "Number of repetitions"},
    {"-of", TK_ARGV_STRING, NULL, (char *) &fileName,
        "Name of file for output"},
    {"x", TK_ARGV_REST, NULL, (char *) &exec,
        "File to exec, followed by any arguments (must be last argument)."},
    {NULL, TK_ARGV_END, NULL, NULL,
        NULL}
};
main(argc, argv)
    int argc;
    char *argv[];
{
    ...
    if (Tk_ParseArgv(interp, tkwin, &argc, argv, argTable, 0) != TCL_OK) {
        fprintf(stderr, "%s\n", Tcl_GetString(Tcl_GetObjResult(interp)));
        exit(1);
    }
    /*
     * Remainder of the program.
     */
}

Note that default values can be assigned to variables named in argTable: the variables will only be overwritten if the particular arguments are present in argv. Here are some example command lines and their effects.

prog -N 200 infile        # just sets the numReps variable to 200
prog -of out200 infile    # sets fileName to reference "out200"
prog -XN 10 infile        # sets the debug flag, also sets numReps
In all of the above examples, argc will be set by Tk_ParseArgv to 2, argv[0] will be “prog”, argv[1] will be “infile”, and argv[2] will be NULL.
