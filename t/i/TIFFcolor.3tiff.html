الوصف

يدعم TIFF مساحات ألوان مختلفة للصور المخزنة بهذا التنسيق. غالبًا ما تكون هناك مشكلة في تطبيق البيانات بشكل صحيح والتحويل بين مساحات الألوان المختلفة للعرض والطباعة. لتبسيط هذه المهمة، تطبق libtiff العديد من دوال تحويل الألوان بنفسها. على وجه الخصوص، تستخدم هذه الدوال في واجهة TIFFRGBAImage.

تستخدم الدالة TIFFYCbCrToRGBInit() لتهيئة حالة التحويل من YCbCr إلى RGB. تعود مسؤولية تخصيص وإلغاء تخصيص هيكل ycbcr للمبرمج. يتم تعريف الدالة TIFFYCbCrToRGB في tiffio.h على النحو التالي:

typedef struct {                /* YCbCr->RGB support */
        TIFFRGBValue* clamptab; /* جدول تحديد النطاق */
        int*          Cr_r_tab;
        int*          Cb_b_tab;
        int32_t*      Cr_g_tab;
        int32_t*      Cb_g_tab;
        int32_t*      Y_tab;
} TIFFYCbCrToRGB;

luma عبارة عن مصفوفة من الأعداد العشرية تحتوي على ثلاثة قيم تمثل نسب الأحمر والأخضر والأزرق في الإضاءة، Y (انظر القسم 21 من مواصفات TIFF 6.0، حيث تتم مناقشة صور YCbCr). يحتوي TIFFTAG_YCBCRCOEFFICIENTS على هذه القيم في ملف TIFF. refBlackWhite هي مصفوفة من الأعداد العشرية تحتوي على 6 قيم تحدد زوجًا من القيم الحدية والعليا لكل مكون من مكونات الصورة (انظر القسم 20 من مواصفات TIFF 6.0 حيث تتم مناقشة حقول الألوان). TIFFTAG_REFERENCEBLACKWHITE مسؤول عن تخزين هذه القيم في ملف TIFF. يجب أن يساعد الكود النموذجي التالي في فهم التقنية:

float *luma, *refBlackWhite;
uint16_t hs, vs;
/* تهيئة الهياكل */
ycbcr = (TIFFYCbCrToRGB*)
    _TIFFmalloc(TIFFroundup(sizeof(TIFFYCbCrToRGB), sizeof(long))
        + 4*256*sizeof(TIFFRGBValue)
        + 2*256*sizeof(int)
        + 3*256*sizeof(int32_t));
if (ycbcr == NULL) {
    TIFFError("YCbCr->RGB", "لا توجد مساحة لحالة تحويل YCbCr->RGB");
    exit(0);
}
TIFFGetFieldDefaulted(tif, TIFFTAG_YCBCRCOEFFICIENTS, &luma);
TIFFGetFieldDefaulted(tif, TIFFTAG_REFERENCEBLACKWHITE, &refBlackWhite);
if (TIFFYCbCrToRGBInit(ycbcr, luma, refBlackWhite) < 0) {
    exit(0);
}
/* بدء التحويل */
uint32_t r, g, b;
uint32_t Y;
int32_t Cb, Cr;
for each pixel in image {
    TIFFYCbCrtoRGB(img->ycbcr, Y, Cb, Cr, &r, &g, &b);
}
/* تحرير هيكل الحالة */
_TIFFfree(ycbcr);

تستخدم الدالة TIFFCIELabToRGBInit() لتهيئة حالة التحويل من CIE L*a*b* 1976 إلى RGB. يتم تعريف الدالة TIFFCIELabToRGB على النحو التالي:

#define CIELABTORGB_TABLE_RANGE 1500
typedef struct {                                /* CIE Lab 1976->RGB support */
    int      range;                             /* حجم جدول التحويل */
    float    rstep, gstep, bstep;
    float    X0, Y0, Z0;                        /* نقطة المرجع الأبيض */
    TIFFDisplay display;
    float    Yr2r[CIELABTORGB_TABLE_RANGE + 1]; /* تحويل Yr إلى r */
    float    Yg2g[CIELABTORGB_TABLE_RANGE + 1]; /* تحويل Yg إلى g */
    float    Yb2b[CIELABTORGB_TABLE_RANGE + 1]; /* تحويل Yb إلى b */
} TIFFCIELabToRGB;

display هو وصف لجهاز العرض، معرّف على النحو التالي:

typedef struct {
    float d_mat[3][3]; /* مصفوفة XYZ -> الإضاءة */
    float d_YCR;       /* إخراج الضوء للأبيض المرجعي */
    float d_YCG;
    float d_YCB;
    uint32_t d_Vrwr;   /* قيم البيكسل للأبيض المرجعي */
    uint32_t d_Vrwg;
    uint32_t d_Vrwb;
    float d_Y0R;       /* الإضاءة المتبقية لبيكسل الأسود */
    float d_Y0G;
    float d_Y0B;
    float d_gammaR;    /* قيم جاما للأجهزة الثلاثة */
    float d_gammaG;
    float d_gammaB;
} TIFFDisplay;

على سبيل المثال، يمكن استخدام جهاز sRGB، والذي يحتوي على المعلمات التالية:

TIFFDisplay display_sRGB = {
    {   /* مصفوفة XYZ -> الإضاءة */
        {  3.2410F, -1.5374F, -0.4986F },
        {  -0.9692F, 1.8760F, 0.0416F },
        {  0.0556F, -0.2040F, 1.0570F }
    },
    100.0F, 100.0F, 100.0F, /* إخراج الضوء للأبيض المرجعي */
    255, 255, 255,          /* قيم البيكسل للأبيض المرجعي */
    1.0F, 1.0F, 1.0F,       /* إضاءة الخرج المتبقية لبيكسل الأسود */
    2.4F, 2.4F, 2.4F,       /* قيم جاما للأجهزة الثلاثة */
};

refWhite هي درجة حرارة لون الأبيض المرجعي. يحتوي TIFFTAG_WHITEPOINT على إحداثيات اللون لنقطة البياض للصورة والتي يمكن من خلالها حساب الأبيض المرجعي باستخدام الصيغ التالية:

refWhite_Y = 100.0
refWhite_X = whitePoint_x / whitePoint_y * refWhite_Y
refWhite_Z = (1.0 - whitePoint_x - whitePoint_y) / whitePoint_y * refWhite_X

يتم التحويل في خطوتين: الأولى ستحول CIE L*a*b* 1976 إلى CIE XYZ باستخدام الروتين TIFFCIELabToXYZ()، والثانية ستحول CIE XYZ إلى RGB باستخدام TIFFXYZToRGB(). راجع مثال الكود أدناه:

float   *whitePoint;
float   refWhite[3];
/* تهيئة الهياكل */
img->cielab = (TIFFCIELabToRGB *)
    _TIFFmalloc(sizeof(TIFFCIELabToRGB));
if (!cielab) {
    TIFFError("CIE L*a*b*->RGB", "لا توجد مساحة لحالة تحويل CIE L*a*b*->RGB.");
    exit(0);
}
TIFFGetFieldDefaulted(tif, TIFFTAG_WHITEPOINT, &whitePoint);
refWhite[1] = 100.0F;
refWhite[0] = whitePoint[0] / whitePoint[1] * refWhite[1];
refWhite[2] = (1.0F - whitePoint[0] - whitePoint[1])
    / whitePoint[1] * refWhite[1];
if (TIFFCIELabToRGBInit(cielab, &display_sRGB, refWhite) < 0) {
    TIFFError("CIE L*a*b*->RGB", "فشل تهيئة حالة تحويل CIE L*a*b*->RGB.");
    _TIFFfree(cielab);
    exit(0);
}
/* الآن يمكننا البدء في التحويل */
uint32_t r, g, b;
uint32_t L;
int32_t a, b;
float X, Y, Z;
for each pixel in image {
    TIFFCIELabToXYZ(cielab, L, a, b, &X, &Y, &Z);
    TIFFXYZToRGB(cielab, X, Y, Z, &r, &g, &b);
}
/* لا تنس تحرير هيكل الحالة */
_TIFFfree(cielab);
