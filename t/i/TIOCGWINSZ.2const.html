TIOCSWINSZ(2const) TIOCSWINSZ(2const)

الاسم

TIOCGWINSZ, TIOCSWINSZ - الحصول على حجم النافذة وتعيينه

المكتبة

مكتبة C القياسية (libc، -lc)

الملخص

#include <asm/termbits.h>  /* تعريف الثوابت TIOC*WINSZ */
#include <sys/ioctl.h>
int ioctl(int fd, TIOCGWINSZ, struct winsize *argp);
int ioctl(int fd, TIOCSWINSZ, const struct winsize *argp);
#include <asm/termios.h>
struct winsize {
    unsigned short ws_row;
    unsigned short ws_col;
    unsigned short ws_xpixel;  /* غير مستخدم */
    unsigned short ws_ypixel;  /* غير مستخدم */
};

الوصف

يتم الاحتفاظ بأحجام النوافذ في النواة، ولكن لا تستخدمها النواة (باستثناء حالة وحدات التحكم الافتراضية، حيث تقوم النواة بتحديث حجم النافذة عند تغيير حجم وحدة التحكم الافتراضية، على سبيل المثال، عن طريق تحميل خط جديد).

TIOCGWINSZ
الحصول على حجم النافذة.
TIOCSWINSZ
تعيين حجم النافذة.

عندما يتغير حجم النافذة، يتم إرسال إشارة SIGWINCH إلى مجموعة العمليات الأمامية.

قيمة الإرجاع

عند النجاح، يتم إرجاع 0. عند الخطأ، يتم إرجاع -1، ويتم تعيين errno للإشارة إلى الخطأ.

انظر أيضًا

ioctl(2), ioctl_tty(2const)
