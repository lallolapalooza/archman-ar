TIOCMSET(2const) TIOCMSET(2const)

الاسم

TIOCMGET, TIOCMSET, TIOCMBIC, TIOCMBIS, TIOCMIWAIT, TIOCGICOUNT - تحكم المودم

المكتبة

مكتبة C القياسية (libc, -lc)

الملخص

#include <asm/termbits.h>  /* تعريف الثوابت TIOC* */
#include <sys/ioctl.h>
int ioctl(int fd, TIOCMGET, int *argp);
int ioctl(int fd, TIOCMSET, const int *argp);
int ioctl(int fd, TIOCMBIC, const int *argp);
int ioctl(int fd, TIOCMBIS, const int *argp);
int ioctl(int fd, TIOCMIWAIT, int arg);
int ioctl(int fd, TIOCGICOUNT, struct serial_icounter_struct *argp);
#include <linux/serial.h>
struct serial_icounter_struct;

الوصف

TIOCMGET
احصل على حالة بتات المودم.
TIOCMSET
اضبط حالة بتات المودم.
TIOCMBIC
امسح بتات المودم المشار إليها.
TIOCMBIS
اضبط بتات المودم المشار إليها.

تُستخدم البتات التالية بواسطة استدعاءات ioctl المذكورة أعلاه:

TIOCM_LE DSR (جاهزية مجموعة البيانات/تمكين الخط)
TIOCM_DTR DTR (جاهزية طرف البيانات)
TIOCM_RTS RTS (طلب الإرسال)
TIOCM_ST TXD ثانوي (إرسال)
TIOCM_SR RXD ثانوي (استقبال)
TIOCM_CTS CTS (جاهز للإرسال)
TIOCM_CAR DCD (كشف حامل البيانات)
TIOCM_CD انظر TIOCM_CAR
TIOCM_RNG RNG (رنين)
TIOCM_RI انظر TIOCM_RNG
TIOCM_DSR DSR (جاهزية مجموعة البيانات)
TIOCMIWAIT
انتظر حتى يتغير أي من بتات المودم الأربعة (DCD, RI, DSR, CTS). يتم تحديد البتات ذات الأهمية كقناع بت في arg، عن طريق ORing بين أي من قيم البتات: TIOCM_RNG, TIOCM_DSR, TIOCM_CD, و TIOCM_CTS. يجب على المتصل استخدام TIOCGICOUNT لمعرفة أي بت قد تغير.
TIOCGICOUNT
احصل على عدادات لمقاطعات خط التسلسل المدخلة (DCD, RI, DSR, CTS). يتم كتابة العدادات إلى البنية serial_icounter_struct التي يشير إليها argp.
ملاحظة: يتم عد كلا الانتقالين (1->0 و 0->1)، باستثناء RI، حيث يتم عد الانتقالات 0->1 فقط.

قيمة الإرجاع

عند النجاح، يتم إرجاع 0. عند حدوث خطأ، يتم إرجاع -1، ويتم تعيين errno للإشارة إلى الخطأ.

أمثلة

تحقق من حالة DTR على منفذ التسلسل.

#include <asm/termbits.h>
#include <fcntl.h>
#include <stdio.h>
#include <sys/ioctl.h>
#include <unistd.h>
int
main(void)
{
    int fd, serial;
    fd = open("/dev/ttyS0", O_RDONLY);
    ioctl(fd, TIOCMGET, &serial);
    if (serial & TIOCM_DTR)
        puts("TIOCM_DTR is set");
    else
        puts("TIOCM_DTR is not set");
    close(fd);
}
