Tie__Watch(3pm)

الحزمة: perl-tk

التراخيص: PerlArtistic, GPL, custom

Tie::Watch(3) توثيق perl/Tk Tie::Watch(3)

الاسم

Tie::Watch - وضع نقاط مراقبة على متغيرات Perl.

الملخص

use Tie::Watch;
$watch = Tie::Watch->new(
    -variable => \$frog,
    -debug    => 1,
    -shadow   => 0,                      
    -fetch    => [\&fetch, 'arg1', 'arg2', ..., 'argn'],
    -store    => \&store,
    -destroy  => sub {print "Final value=$frog.\n"},
}
%vinfo = $watch->Info;
$args  = $watch->Args(-fetch);
$val   = $watch->Fetch;
print "val=", $watch->Say($val), ".\n";
$watch->Store('Hello');
$watch->Unwatch;

الوصف

يقوم وحدة الفئة هذه بربط روتين فرعي واحد أو أكثر من اختراعك بمتغير Perl. يمكن لجميع المتغيرات أن تحتوي على استدعاءات FETCH، STORE، وDESTROY. بالإضافة إلى ذلك، يمكن للمصفوفات تحديد استدعاءات CLEAR، DELETE، EXISTS، EXTEND، FETCHSIZE، POP، PUSH، SHIFT، SPLICE، STORESIZE، وUNSHIFT، ويمكن للهاشات تحديد استدعاءات CLEAR، DELETE، EXISTS، FIRSTKEY، وNEXTKEY. إذا كانت هذه المصطلحات غير مألوفة لك، فإنني أقترح بشدة قراءة perltie.

مع Tie::Watch يمكنك:

. تعديل قيمة المتغير
. منع تغيير قيمة المتغير
. استدعاء روتين Perl/Tk عند تغيير متغير
. تتبع المراجع لمتغير

تتبع تنسيق الاستدعاء نمط مخطط Perl/Tk: قم بتوفير مرجع كود، أو، قم بتوفير مرجع مصفوفة ومرر مرجع كود الاستدعاء في العنصر الأول من المصفوفة، متبوعًا بمعاملات الاستدعاء. (انظر الأمثلة في الملخص، أعلاه.)

يوفر Tie::Watch استدعاءات افتراضية لأي استدعاءات تفشل في تحديدها. بخلاف التأثير السلبي على الأداء، فإنها تؤدي الإجراء القياسي الذي تتوقعه، لذلك يتصرف المتغير "بشكل طبيعي". بمجرد تجاوز استدعاء افتراضي، ربما لإدراج عبارات طباعة مثل كود التصحيح، ينتهي استدعاؤك عادةً باستدعاء الطريقة الأساسية (المتجاوزة). ولكن ليس عليك ذلك!

لربط اسم طريقة مرتبطة باسم استدعاء افتراضي، ببساطة اجعل اسم الطريقة المرتبطة صغيرًا وحرفها الأول كبيرًا. لذا FETCH تصبح Fetch، وNEXTKEY تصبح Nextkey، وهكذا.

فيما يلي استدعاءان لمتغير عددي. استدعاء FETCH (قراءة) لا يفعل شيئًا سوى توضيح حقيقة أنه يعيد القيمة المراد تعيينها للمتغير. استدعاء STORE (كتابة) يحول المتغير إلى أحرف كبيرة ويعيده. في جميع الحالات، يجب على الاستدعاء إرجاع قيمة القراءة أو الكتابة الصحيحة - عادةً ما يفعل ذلك عن طريق استدعاء الطريقة الأساسية.

my $fetch_scalar = sub {
    my($self) = @_;
    $self->Fetch;
};
my $store_scalar = sub {
    my($self, $new_val) = @_;
    $self->Store(uc $new_val);
};

فيما يلي استدعاءات FETCH وSTORE إما لمصفوفة أو هاش. إنها تفعل نفس الشيء تقريبًا مثل استدعاءات المتغيرات العددية، ولكنها توفر المزيد من المعلومات.

my $fetch = sub {
    my($self, $key) = @_;
    my $val = $self->Fetch($key);
    print "In fetch callback, key=$key, val=", $self->Say($val);
    my $args = $self->Args(-fetch);
    print ", args=('", join("', '",  @$args), "')" if $args;
    print ".\n";
    $val;
};
my $store = sub {
    my($self, $key, $new_val) = @_;
    my $val = $self->Fetch($key);
    $new_val = uc $new_val;
    $self->Store($key, $new_val);
    print "In store callback, key=$key, val=", $self->Say($val),
      ", new_val=", $self->Say($new_val);
    my $args = $self->Args(-store);
    print ", args=('", join("', '",  @$args), "')" if $args;
    print ".\n";
    $new_val;
};

في جميع الحالات، يكون المعامل الأول هو مرجع إلى كائن Watch، يستخدم لاستدعاء أساليب الفئة التالية.

الأساليب

$watch = Tie::Watch->new(-options => values);
الأسلوب الباني لنقطة المراقبة الذي يقبل أزواج الخيار/القيمة لإنشاء وتكوين كائن Watch. الخيار المطلوب الوحيد هو -variable.

-variable هو *مرجع* إلى متغير عددي، مصفوفة، أو هاش.

-debug (افتراضي 0) هو 1 لتنشيط عبارات طباعة التصحيح الداخلية لـ Tie::Watch.

-shadow (افتراضي 1) هو 0 لتعطيل تظليل المصفوفات والهاشات. لمنع التكرار اللانهائي، تحتفظ Tie::Watch بمتغيرات متوازية للمصفوفات والهاشات. عند إنشاء نقطة المراقبة، يتم تهيئة المتغير المتوازي الظلي بمحتويات المتغير المراقب، وعند حذف نقطة المراقبة، يتم نسخ المتغير الظلي إلى المتغير الأصلي. وبالتالي، لا تضيع التغييرات التي تم إجراؤها أثناء عملية المراقبة. التظليل مفعل افتراضيًا. إذا قمت بتعطيل التظليل، فسيتم فقدان أي تغييرات تم إجراؤها على مصفوفة أو هاش عند حذف نقطة المراقبة.

حدد أيًا من معلمات الاستدعاء ذات الصلة التالية، بالتنسيق الموصوف أعلاه: -fetch، -store، -destroy. بالإضافة إلى ذلك للمصفوفات: -clear، -extend، -fetchsize، -pop، -push، -shift، -splice، -storesize، و -unshift. بالإضافة إلى ذلك للهاشات: -clear، -delete، -exists، -firstkey، و -nextkey.

$args = $watch->Args(-fetch);
يعيد مرجعًا إلى قائمة بمعاملات الاستدعاء المحدد، أو غير محدد إذا لم يكن هناك شيء.
$watch->Fetch(); $watch->Fetch($key);
يعيد القيمة الحالية للمتغير. $key مطلوب لمصفوفة أو هاش.
%vinfo = $watch->Info();
يعيد هاشًا يفصل تفاصيل الأجزاء الداخلية لكائن Watch، مع المفاتيح التالية: 
%vinfo = {
    -variable =>  SCALAR(0x200737f8)
    -debug    =>  '0'
    -shadow   =>  '1'
    -value    =>  'HELLO SCALAR'
    -destroy  =>  ARRAY(0x200f86cc)
    -fetch    =>  ARRAY(0x200f8558)
    -store    =>  ARRAY(0x200f85a0)
    -legible  =>  above data formatted as a list of string, for printing
}

بالنسبة لكائنات Watch للمصفوفات والهاشات، يتم استبدال المفتاح -value بمفتاح -ptr وهو مرجع للمصفوفة أو الهاش المتوازي. بالإضافة إلى ذلك، بالنسبة للمصفوفة أو الهاش، هناك أزواج مفتاح/قيمة لجميع الاستدعاءات الخاصة بالمتغير.

$watch->Say($val);
يستخدم بشكل أساسي لتصحيح الأخطاء، ويعيد $val بين علامات اقتباس إذا لزم الأمر، أو السلسلة "undefined" للقيم غير المعرفة.
$watch->Store($new_val); $watch->Store($key, $new_val);
يخزن قيمة جديدة للمتغير. $key مطلوب لمصفوفة أو هاش.
$watch->Unwatch();
توقف عن مراقبة المتغير.

اعتبارات الكفاءة

إذا كان بإمكانك العيش باستخدام أساليب الفئة المقدمة، فيرجى القيام بذلك. يمكنك العبث بتجزئة الكائن مباشرة وتحسين أداء المراقبة، مع المخاطرة بأن يتوقف رمزك عن العمل في المستقبل.
