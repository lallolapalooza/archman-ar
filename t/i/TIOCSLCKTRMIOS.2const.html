TIOCSLCKTRMIOS(2const) TIOCSLCKTRMIOS(2const)

الاسم

TIOCGLCKTRMIOS, TIOCSLCKTRMIOS - قفل هيكل termios

المكتبة

مكتبة C القياسية (libc, -lc)

الملخص

#include <asm/termbits.h>  /* تعريف ثوابت TIOC*CLKTRMIOS */
#include <sys/ioctl.h>
int ioctl(int fd, TIOCGLCKTRMIOS, struct termios *argp);
int ioctl(int fd, TIOCSLCKTRMIOS, const struct termios *argp);
#include <asm/termbits.h>
struct termios;

الوصف

يمكن قفل هيكل termios الطرفي. القفل هو نفسه هيكل termios، مع بتات أو حقول غير صفرية تشير إلى قيمة مقفلة.

TIOCGLCKTRMIOS
يحصل على حالة قفل هيكل termios الطرفي.
TIOCSLCKTRMIOS
يضبط حالة قفل هيكل termios الطرفي. يمكن فقط لعملية ذات صلاحية CAP_SYS_ADMIN القيام بذلك.

قيمة الإرجاع

عند النجاح، يتم إرجاع 0. عند حدوث خطأ، يتم إرجاع -1، ويتم تعيين errno للإشارة إلى الخطأ.

الأخطاء

EPERM
إذن غير كاف.

تنبيهات

يرجى ملاحظة أن struct termios من <asm/termbits.h> مختلف وغير متوافق مع struct termios من <termios.h>. تستدعي استدعاءات ioctl هذه struct termios من <asm/termbits.h>.
