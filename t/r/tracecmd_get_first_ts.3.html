tracecmd_get_first_ts(3)

الحزمة: trace-cmd

التراخيص: GPL-2.0-only, LGPL-2.1-only

LIBTRACECMD(3) دليل libtracefs LIBTRACECMD(3)

الاسم

tracecmd_get_first_ts, tracecmd_add_ts_offset, tracecmd_get_tsc2nsec - التعامل مع الطوابع الزمنية من ملف تتبع.

الملخص

#include <trace-cmd.h>
unsigned long long tracecmd_get_first_ts(struct tracecmd_input *handle);
void tracecmd_add_ts_offset(struct tracecmd_input *handle, long long offset);
int tracecmd_get_tsc2nsec(struct tracecmd_input *handle, int *mult, int pass[*]shift, unsigned long long *offset);

الوصف

يمكن استخدام مجموعة واجهات برمجة التطبيقات هذه لقراءة بيانات التتبع من ملف تتبع مفتوح باستخدام tracecmd_open()(3)، tracecmd_open_fd()(3) أو tracecmd_open_head()(3).

تقوم الدالة tracecmd_get_first_ts() بإرجاع الطابع الزمني للسجل الأول في handle.

تقوم الدالة tracecmd_add_ts_offset() بإضافة إزاحة إلى كل سجل في handle يمثل ملف تتبع. هذا مفيد لربط ملفي تتبع مختلفين بإزاحتهما (على سبيل المثال، ملف تتبع من مضيف وملف تتبع من ضيف لم تتم مزامنتهما عند إنشائهما).

ترجع الدالة tracecmd_get_tsc2nsec قيم الحساب التي تحول الطابع الزمني الخام إلى نانو ثانية. المعلمات هي مؤشرات إلى المساحة لتخزين القيم، أو NULL لتجاهلها. سيتم حفظ المضاعف في mult، وسيتم حفظ قيمة الإزاحة في shift، وسيتم حفظ قيمة الإزاحة في offset، إذا كانت المعلمات المقابلة ليست NULL.

قيمة الإرجاع

ترجع الدالة tracecmd_get_first_ts() الطابع الزمني للسجل الأول في ملف تتبع للمقبض handle المحدد.

ترجع الدالة tracecmd_get_tsc2nsec() القيمة 0 إذا كانت ساعة التتبع تدعم قيم الإزاحة و -1 بخلاف ذلك. لاحظ أنه إذا كان ملف التتبع يحتوي على خيار TSC2NSEC، فقد تظل القيم المعادة في المعلمات صالحة حتى لو أعادت الدالة نفسها -1. قيمة الإرجاع تشير فقط إلى ما إذا كانت القيم ستستخدم في حسابات الساعة المحددة.

مثال

#include <stdlib.h>
#include <trace-cmd.h>
static int print_events(struct tracecmd_input *handle, struct tep_record *record, int cpu, void *data)
{
        static struct trace_seq seq;
        struct tep_handle *tep = tracecmd_get_tep(handle);
        const char *file = tracecmd_get_private(handle);
        if (!seq.buffer)
                trace_seq_init(&seq);
        trace_seq_reset(&seq);
        trace_seq_printf(&seq, "%s: ", file);
        trace_seq_printf(&seq, "%s: ", file);
        tep_print_event(tep, &seq, record, "%6.1000d [%03d] %s-%d %s: %s\n",
                        TEP_PRINT_TIME, TEP_PRINT_CPU, TEP_PRINT_COMM, TEP_PRINT_PID,
                        TEP_PRINT_NAME, TEP_PRINT_INFO);
        trace_seq_terminate(&seq);
        trace_seq_do_printf(&seq);
        return 0;
}
int main(int argc, char **argv)
{
        struct tracecmd_input **handles = NULL;
        unsigned long long ts, first_ts = 0;
        unsigned long long offset;
        int multi;
        int shift;
        int nr_handles = 0;
        int ret;
        int i;
        if (argc < 2) {
                printf("usage: %s trace.dat [trace.dat ...]\n", argv[0]);
                exit(-1);
        }
        for (i = 1; i < argc; i++) {
                handles = realloc(handles, sizeof(*handles) * (nr_handles + 1));
                if (!handles)
                        exit(-1);
                handles[nr_handles] = tracecmd_open(argv[i], 0);
                if (!handles[nr_handles])
                        exit(-1);
                ret = tracecmd_get_tsc2nsec(handles[nr_handles], &multi, &shift, &offset);
                if (!ret)
                        printf(" %s has tsc2nsec calculations of mult:%d shift:%d offset:%lld\n",
                                argv[i], multi, shift, offset);
                tracecmd_set_private(handles[nr_handles], argv[i]);
                ts = tracecmd_get_first_ts(handles[nr_handles]);
                if (!first_ts || ts < first_ts)
                        first_ts = ts;
                nr_handles++;
        }
        /* Set the time stamp to start at the first record found */
        for (i = 0; i < nr_handles; i++)
                tracecmd_add_ts_offset(handles[i], -first_ts);
        tracecmd_iterate_events_multi(handles, nr_handles, print_events, NULL);
        for (i = 0; i < nr_handles; i++)
                tracecmd_close(handles[i]);
        free(handles);
}
