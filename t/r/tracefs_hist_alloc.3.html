الملخص

#include <tracefs.h>
enum tracefs_hist_key_type {
        TRACEFS_HIST_KEY_NORMAL = 0,
        TRACEFS_HIST_KEY_HEX,
        TRACEFS_HIST_KEY_SYM,
        TRACEFS_HIST_KEY_SYM_OFFSET,
        TRACEFS_HIST_KEY_SYSCALL,
        TRACEFS_HIST_KEY_EXECNAME,
        TRACEFS_HIST_KEY_LOG,
        TRACEFS_HIST_KEY_USECS,
        TRACEFS_HIST_KEY_MAX
};
struct tracefs_hist_axis {
        const char *key;
        enum tracefs_hist_key_type type;
};
struct tracefs_hist *tracefs_hist_alloc(struct tracefs_tep * tep,
                        const char *system, const char *event,
                        const char *key, enum tracefs_hist_key_type type);
struct tracefs_hist *tracefs_hist_alloc_2d(struct tracefs_tep * tep,
                        const char *system, const char *event,
                        const char *key1, enum tracefs_hist_key_type type1,
                        const char *key2, enum tracefs_hist_key_type type2));
struct tracefs_hist *tracefs_hist_alloc_nd(struct tracefs_tep * tep,
                        const char *system, const char *event,
                        struct tracefs_hist_axis *axes);
struct tracefs_hist *tracefs_hist_alloc_nd_cnt(struct tep_handle *tep,
                          const char *system, const char *event_name,
                          struct tracefs_hist_axis_cnt *axes);
void tracefs_hist_free(struct tracefs_hist *hist);
int tracefs_hist_add_key(struct tracefs_hist *hist, const char *key,
                         enum tracefs_hist_key_type type);
int tracefs_hist_add_key_cnt(struct tracefs_hist *hist, const char *key,
                         enum tracefs_hist_key_type type, int cnt);
int tracefs_hist_add_value(struct tracefs_hist *hist, const char *value);

الوصف

يتم إنشاء رسوم بيانية للأحداث بواسطة ملف الزناد في دليل الحدث. يمكن أن يكون بناء الجملة معقدًا ويصعب الحصول عليه بشكل صحيح. تعالج واجهة برمجة التطبيقات هذه بناء الجملة، وتسهل إنشاء الرسوم البيانية للأحداث والتفاعل معها. راجع https://www.kernel.org/doc/html/latest/trace/histogram.html لمزيد من المعلومات.

تقوم الدالة tracefs_hist_alloc() بتخصيص واصف "struct tracefs_hist" لرسم بياني أحادي البعد وتعيد عنوانه. يجب تحرير هذا الواصف بواسطة tracefs_hist_free(). tep هو مقبض حدث تتبع (انظر tracefs_local_events(3))، الذي يحتوي على system و event اللذين سيتم ربط الرسم البياني بهما. system هو النظام أو مجموعة الحدث. event هو الحدث الذي سيتم ربط الرسم البياني به. key هو حقل من الحدث سيتم استخدامه كمفتاح (بعد/بعد) للرسم البياني. type هو نوع المفتاح. انظر أنواع المفاتيح أدناه.

تقوم الدالة tracefs_hist_alloc_2d() بتخصيص واصف "struct tracefs_hist" لرسم بياني ثنائي الأبعاد وتعيد عنوانه. يجب تحرير هذا الواصف بواسطة tracefs_hist_free(). tep هو مقبض حدث تتبع (انظر tracefs_local_events(3))، الذي يحتوي على system و event اللذين سيتم ربط الرسم البياني بهما. system هو النظام أو مجموعة الحدث. event هو الحدث الذي سيتم ربط الرسم البياني به. key1 هو الحقل الأول من الحدث الذي سيتم استخدامه كمفتاح (بعد/بعد) للرسم البياني. type1 هو نوع key1. انظر أنواع المفاتيح أدناه. key2 هو الحقل الثاني من الحدث الذي سيتم استخدامه كمفتاح (بعد/بعد) للرسم البياني. type2 هو نوع key2. انظر أنواع المفاتيح أدناه.

تقوم الدالة tracefs_hist_alloc_nd() بتخصيص واصف "struct tracefs_hist" لرسم بياني متعدد الأبعاد (N-dimensional) وتعيد عنوانه. يجب تحرير هذا الواصف بواسطة tracefs_hist_free(). tep هو مقبض حدث تتبع (انظر tracefs_local_events(3))، الذي يحتوي على system و event اللذين سيتم ربط الرسم البياني بهما. system هو النظام أو مجموعة الحدث. event هو الحدث الذي سيتم ربط الرسم البياني به. axes هي مصفوفة من أزواج key / type، تحدد أبعاد الرسم البياني.

ستقوم الدالة tracefs_hist_alloc_nd_cnt() بتهيئة واصف رسم بياني سيتم ربطه بـ system/event. هذا يقوم فقط بتهيئة الواصف بالمفاتيح axes المعطاة كمفاتيح أساسية. هذا يقوم فقط بتهيئة الواصف، ولا يبدأ الرسم البياني في النواة. الفرق بين هذه والدالة tracefs_hist_alloc_nd() هو أن المعامل axes من نوع struct tracefs_hist_axis_cnt وليس struct tracefs_hist_axis.

تقوم الدالة tracefs_hist_free() بتحرير واصف tracefs_hist. ملاحظة: لا توقف أو تعطل الرسم البياني قيد التشغيل إذا كان قد بدأ. يجب استدعاء tracefs_hist_destroy() للقيام بذلك.

تضيف الدالة tracefs_hist_add_key() مفتاحًا ثانويًا أو ثالثيًا إلى الرسم البياني. المفتاح الذي تم تمريره إلى tracefs_hist_alloc_nd() هو المفتاح الأساسي للرسم البياني. في المرة الأولى التي يتم فيها استدعاء هذه الدالة، ستضيف مفتاحًا ثانويًا (أو رسمًا بيانيًا ثنائي الأبعاد). إذا تم استدعاء هذه الدالة مرة أخرى على نفس الرسم البياني، فستضيف مفتاحًا ثالثيًا (أو رسمًا بيانيًا ثلاثي الأبعاد). المعامل hist هو واصف الرسم البياني لإضافة key إليه. type هو نوع المفتاح المراد إضافته (انظر أنواع المفاتيح أدناه).

الدالة tracefs_hist_add_key_cnt() هي نفسها الدالة tracefs_hist_add_key() باستثناء أنها تسمح بإضافة عداد للنوع المحدد. حاليًا، هناك نوع واحد فقط وهو buckets يتطلب عدادًا. عند إضافة مفتاح من نوع buckets، فإن الدالة tracefs_hist_add_key() غير كافية، حيث أن نوع buckets يتطلب عدادًا أو حجم دلو. استخدم tracefs_hist_add_key_cnt() عند تحديد مفتاح مقسم إلى دلاء، وقم بتمرير حجم تلك الدلاء إلى cnt.

ستضيف الدالة tracefs_hist_add_value() قيمة للتسجيل. بشكل افتراضي، القيمة هي ببساطة "عدد المرات" التي تم فيها ضرب مفتاح الرسم البياني. hist هو واصف الرسم البياني لإضافة القيمة إليه. value هو حقل في الرسم البياني للإضافة إليه عند ضرب مثيل للمفتاح.

أنواع المفاتيح

كل من الدالتين tracefs_hist_alloc_nd() و tracefs_hist_add_key() تضيف مفتاحًا وتتطلب أن يكون لهذا المفتاح نوع. قد تكون الأنواع:

TRACEFS_HIST_KEY_NORMAL أو صفر (0) وهو عدم تعديل النوع.

TRACEFS_HIST_KEY_HEX لعرض المفتاح بالنظام الست عشري.

TRACEFS_HIST_KEY_SYM لعرض المفتاح كرمز للنواة (إذا تم العثور عليه). إذا كان المفتاح عنوانًا، فهذا مفيد لأنه سيعرض أسماء الوظائف بدلاً من مجرد رقم.

TRACEFS_HIST_KEY_SYM_OFFSET مشابه لـ TRACEFS_HIST_KEY_SYM ولكنه سيشمل أيضًا إزاحة الوظيفة لمطابقة العنوان الدقيق.

TRACEFS_HIST_KEY_SYSCALL إذا كان المفتاح هو "معرف" استدعاء النظام (الرقم الذي يتم تمريره من مساحة المستخدم إلى النواة لإعلامها بنوع استدعاء النظام الذي يتم استدعاؤه)، فعندئذ يتم عرض اسم استدعاء النظام.

TRACEFS_HIST_KEY_EXECNAME إذا كان "common_pid" هو المفتاح (معرف العملية الخاصة بالمهمة المنفذة)، فبدلاً من عرض الرقم، يتم عرض اسم المهمة قيد التشغيل.

TRACEFS_HIST_KEY_LOG سيعرض المفتاح بمقياس لوغاريتمي ثنائي.

TRACEFS_HIST_KEY_USECS للاستخدام مع "common_timestamp" أو TRACEFS_HIST_TIMESTAMP، وفي هذه الحالة، سيعرض الطابع الزمني بالميكروثانية بدلاً من النانوثانية.

قيمة الإرجاع

تُرجع الدالة tracefs_hist_alloc_nd() واصف رسم بياني مخصص يجب تحريره بواسطة tracefs_hist_free() أو NULL في حالة حدوث خطأ.

جميع الدوال الأخرى تُرجع صفرًا عند النجاح أو -1 عند حدوث خطأ.

إذا أعادت الدالة tracefs_hist_start() خطأ، فقد يتم عرض رسالة في النواة يمكن استردادها بواسطة tracefs_error_last().

tracefs_hist_alloc(3) - libtracefs - Arch Linux

NAME

tracefs_hist_alloc, tracefs_hist_free, tracefs_hist_alloc_by_size - Allocation and freeing of tracefs histograms

SYNOPSIS

#include <tracefs.h>

struct tracefs_hist *
tracefs_hist_alloc(const char *name);

void
tracefs_hist_free(struct tracefs_hist *hist);

struct tracefs_hist *
tracefs_hist_alloc_by_size(const char *name, unsigned long size);

DESCRIPTION

The tracefs library provides a simple interface to tracefs histograms. A histogram is a data structure that is created on the tracefs filesystem (usually at /sys/kernel/debug/tracing/histograms/). When a trace is running, it can add data to these histograms. This data can then be read from the tracefs filesystem.

The structure tracefs_hist represents a histogram. It holds the file descriptors to the histogram's related files.

tracefs_hist_alloc allocates and initializes a tracefs_hist structure. The name argument is the name of the histogram. The library will look for the histogram file at /sys/kernel/debug/tracing/histograms/<name>/. If the histogram doesn't exist, it will be created.

tracefs_hist_free frees the memory allocated for the tracefs_hist structure. It does not remove the histogram from the tracefs filesystem. If the histogram was created by this library, and no other process is using it, it will be removed automatically when the last file descriptor to it is closed.

tracefs_hist_alloc_by_size is similar to tracefs_hist_alloc but also takes a size argument. This argument is the desired size of the histogram in bytes. Note that the actual size may be different due to system constraints.

KEY TYPES

The main data structure used is struct tracefs_hist. It contains the following members:

struct tracefs_hist {
    int fd;                 /* File descriptor to the histogram file */
    const char *name;       /* Name of the histogram */
    unsigned long size;     /* Size of the histogram */
    /* ... other members ... */
};

RETURN VALUE

tracefs_hist_alloc and tracefs_hist_alloc_by_size return a pointer to a newly allocated tracefs_hist structure on success, or NULL on failure. If an error occurs, the global variable errno is set to indicate the error.

tracefs_hist_free does not return a value.

EXAMPLE

The following example shows how to allocate a histogram named "my_histogram", write some data to it, and then free it:

#include <stdio.h>
#include <stdlib.h>
#include <tracefs.h>

int main() {
    struct tracefs_hist *hist;
    const char *data = "hello world";
    ssize_t written;

    /* Allocate a histogram named "my_histogram" */
    hist = tracefs_hist_alloc("my_histogram");
    if (!hist) {
        perror("tracefs_hist_alloc");
        return 1;
    }

    printf("Histogram '%s' allocated with fd %d and size %lu\\n",
           hist->name, hist->fd, hist->size);

    /* Write data to the histogram */
    written = write(hist->fd, data, strlen(data));
    if (written < 0) {
        perror("write");
        tracefs_hist_free(hist);
        return 1;
    }

    printf("Wrote %zd bytes to the histogram\\n", written);

    /* Free the histogram structure */
    tracefs_hist_free(hist);
    printf("Histogram freed.\\n");

    return 0;
}

FILES

The tracefs histograms are typically located in /sys/kernel/debug/tracing/histograms/. The library uses file descriptors to interact with these histogram files.

To link against the tracefs library, you must use the following linker switch:

-ltracefs
        Linker switch to add when building a program that uses the library.
