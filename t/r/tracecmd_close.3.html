tracecmd_close(3)

الحزمة: trace-cmd

التراخيص: GPL-2.0-only, LGPL-2.1-only

LIBTRACECMD(3) دليل libtracefs LIBTRACECMD(3)

الاسم

tracecmd_open, tracecmd_open_fd, tracecmd_open_head, tracecmd_init_data, tracecmd_close, tracecmd_set_private, tracecmd_get_private - فتح وإغلاق ملف تتبع.

الملخص

#include <trace-cmd.h>
struct tracecmd_input *tracecmd_open(const char *file, int flags);
struct tracecmd_input *tracecmd_open_fd(int fd, int flags);
struct tracecmd_input *tracecmd_open_head(const char *file, int flags);
int tracecmd_init_data(struct tracecmd_input *handle);
void tracecmd_close(struct tracecmd_input *handle);
void tracecmd_set_private(struct tracecmd_input *handle, void *data);
void *tracecmd_get_private(struct tracecmd_input *handle);

الوصف

يمكن استخدام هذه المجموعة من واجهات برمجة التطبيقات لفتح وإغلاق ملف تتبع تم تسجيله بواسطة trace-cmd(1) ويحتوي على معلومات تتبع من ftrace، المتعقب الرسمي لنواة لينكس. يتم تمثيل الملف المفتوح بواسطة بنية tracecmd_input، وجميع واجهات برمجة التطبيقات الأخرى التي تعمل مع الملف تتطلب مؤشرًا إلى البنية. تحتوي واجهات برمجة التطبيقات لفتح ملف تتبع على معلمة إدخال flag، والتي تتحكم في كيفية فتح الملف وتحليله. تعتبر flag مزيجًا من هذه الخيارات:

TRACECMD_FL_LOAD_NO_PLUGINS - لا تقم بتحميل أي إضافات
TRACECMD_FL_LOAD_NO_SYSTEM_PLUGINS - لا تقم بتحميل الإضافات على مستوى النظام، قم بتحميل الإضافات "المحلية فقط"
                                       من دليل المستخدم الرئيسي.

تقوم الدالة tracecmd_open() بفتح ملف تتبع file المحدد، وتحليل رؤوس البيانات الوصفية من الملف، وتخصيص و تهيئة بنية معالج tracecmd_input التي تمثل الملف. كما أنها تهيئ المعالج لقراءة بيانات التتبع من الملف. المعالج المُعاد جاهز للاستخدام مع واجهات برمجة التطبيقات tracecmd_read_.

تقوم الدالة tracecmd_open_fd() بنفس عمل tracecmd_open()، ولكنها تعمل مع واصف ملف لملف تتبع، تم فتحه للقراءة.

الدالة tracecmd_open_head() مماثلة للدالة tracecmd_open()، ولكنها لا تهيئ المعالج لقراءة بيانات التتبع. تقوم بقراءة وتحليل رؤوس البيانات الوصفية فقط. يجب استخدام tracecmd_init_data() قبل استخدام واجهات برمجة التطبيقات tracecmd_read_.

تقوم الدالة tracecmd_init_data() بتهيئة handle، الذي تم تخصيصه بواسطة tracecmd_open_head()، لقراءة بيانات التتبع من الملف المرتبط به. يجب استدعاء واجهة برمجة التطبيقات هذه قبل أي من واجهات برمجة التطبيقات tracecmd_read_.

تقوم الدالة tracecmd_close() بتحرير handle، وهو مؤشر إلى بنية tracecmd_input، والذي تم تخصيصه مسبقًا بواسطة واجهات برمجة التطبيقات tracecmd_open() أو tracecmd_open_fd() أو tracecmd_open_head().

تسمح الدالة tracecmd_set_private() بإضافة data محددة إلى handle يمكن استرجاعها لاحقًا.

ستقوم الدالة tracecmd_get_private() باسترجاع data التي تم تعيينها بواسطة tracecmd_set_private() للمعالج handle المحدد.

قيمة الإرجاع

ترجع الدوال tracecmd_open() و tracecmd_open_fd() و tracecmd_open_head() مؤشرًا إلى بنية tracecmd_input أو NULL في حالة حدوث خطأ. يجب تحرير البنية المُعادة بواسطة tracecmd_close(). لاحظ أنه إذا تم استخدام tracecmd_open_fd() لتخصيص معالج tracecmd_input، فعند استدعاء tracecmd_close() لإغلاقه، سيتم إغلاق واصف الملف هذا أيضًا.

ترجع الدالة tracecmd_init_data() القيمة -1 في حالة حدوث خطأ أو 0 بخلاف ذلك.

ترجع الدالة tracecmd_get_private() القيمة data التي تم تعيينها بواسطة tracecmd_set_private().

مثال

هناك نمطان مختلفان للاستخدام لفتح وقراءة بيانات التتبع من
ملف تتبع، والتي يمكن استخدامها اعتمادًا على حالة الاستخدام.
1. فتح وتهيئة ملف التتبع في خطوة واحدة:
#include <stdlib.h>
#include <trace-cmd.h>
static int print_events(struct tracecmd_input *handle, struct tep_record *record, int cpu, void *data)
{
        static struct trace_seq seq;
        struct tep_handle *tep = tracecmd_get_tep(handle);
        const char *file = tracecmd_get_private(handle);
        if (!seq.buffer)
                trace_seq_init(&seq);
        trace_seq_reset(&seq);
        trace_seq_printf(&seq, "%s: ", file);
        trace_seq_terminate(&seq);
        trace_seq_do_printf(&seq);
        return 0;
}
int main(int argc, char **argv)
{
        struct tracecmd_input *handle;
        if (argc < 2) {
                printf("usage: %s trace.dat\n", argv[0]);
                exit(-1);
        }
        handle = tracecmd_open(argv[i], 0);
        if (!handle)
                exit(-1);
        tracecmd_set_private(handles[nr_handles], argv[i]);
        tracecmd_iterate_events(handles, NULL, 0, print_events, NULL);
        tracecmd_close(handle);
}
2. فتح وتهيئة ملف التتبع في خطوتين. يسمح هذا بإجراء بعض المعالجة بناءً على البيانات الوصفية،
مقروءة من الملف، قبل تهيئة بيانات التتبع للقراءة. مثال على حالة الاستخدام هذه هو عند فتح ملفات تتبع متعددة
مسجلة في نفس جلسة التتبع. في هذه الحالة، يجب تعديل الطوابع الزمنية لجميع
أحداث التتبع بناءً على المعلومات المستخرجة من البيانات الوصفية للملف
وقبل قراءة بيانات التتبع.
#include <trace-cmd.h>
...
struct tracecmd_input *handle = tracecmd_open_head("trace.dat");
        if (!handle) {
                /* فشل في فتح ملف trace.dat */
        }
...
        /* قم ببعض المعالجة، قبل تهيئة بيانات التتبع للقراءة */
...
        if (tracecmd_init_data(handle) < 0) {
                /* فشل في تهيئة المعالج لقراءة بيانات التتبع */
        }
...
        /* قراءة بيانات التتبع من الملف، باستخدام المعالج */
...
        tracecmd_close(handle);
...
