tracefs_synth_destroy(3)

الحزمة: libtracefs

التراخيص: GPL-2.0-only, LGPL-2.1-only

LIBTRACEFS(3) دليل libtracefs LIBTRACEFS(3)

الاسم

tracefs_synth_create, tracefs_synth_destroy, tracefs_synth_complete, tracefs_synth_trace, tracefs_synth_snapshot, tracefs_synth_save, tracefs_synth_set_instance, - إنشاء أحداث اصطناعية

الملخص

#include <tracefs.h>
int tracefs_synth_create(struct tracefs_synth *synth);
int tracefs_synth_destroy(struct tracefs_synth *synth);
bool tracefs_synth_complete(struct tracefs_synth *synth);
int tracefs_synth_set_instance(struct tracefs_synth *synth, struct tracefs_instance *instance);
int tracefs_synth_trace(struct tracefs_synth *synth,
                        enum tracefs_synth_handler type, const char *var);
int tracefs_synth_snapshot(struct tracefs_synth *synth,
                           enum tracefs_synth_handler type, const char *var);
int tracefs_synth_save(struct tracefs_synth *synth,
                       enum tracefs_synth_handler type, const char *var,
                       char **save_fields);

الوصف

الأحداث الاصطناعية هي أحداث ديناميكية يتم إنشاؤها عن طريق مطابقة حدثين آخرين مما يؤدي إلى تشغيل حدث اصطناعي. أحد الأحداث هو حدث البداية الذي يتم تسجيل حقله، وعندما يتم تنفيذ الحدث الثاني، إذا كان يحتوي على حقل (أو حقول) يتطابق مع حقل (أو حقول) حدث البداية، فسيؤدي ذلك إلى تشغيل الحدث الاصطناعي. يمكن تمرير قيم الحقول بخلاف الحقول المتطابقة من حدث البداية إلى حدث النهاية لإجراء حسابات عليها، أو لمجرد تمريرها كمعامل للحدث الاصطناعي.

أحد حالات الاستخدام الشائعة هو تعيين "sched_waking" كحدث بداية. يتم تشغيل هذا الحدث عندما يتم إيقاظ عملية. ثم قم بتعيين "sched_switch" كحدث نهاية. يتم تشغيل هذا الحدث عندما يتم جدولة مهمة جديدة على وحدة المعالجة المركزية. عن طريق تعيين "common_pid" لكلا الحدثين كحقول مطابقة، يعتبر الوقت بين الحدثين هو زمن استجابة إيقاظ العملية. استخدم TRACEFS_TIMESTAMP كحقل لكلا الحدثين لحساب الفرق بالنانوثانية، أو استخدم TRACEFS_TIMESTAMP_USECS كحقول مقارنة لكلا الحدثين لحساب الفرق بالميكروثانية. يتم استخدام هذا كالمثال أدناه.

tracefs_synth_create() ينشئ الحدث الاصطناعي في النظام. بشكل افتراضي، يتم إنشاء مشغلات الرسم البياني في مثيل التتبع الأعلى، حيث يمكن استخدام أي حدث اصطناعي عالميًا عبر جميع المثيلات. في حال رغب تطبيق ما في الاحتفاظ بمشغلات الرسم البياني خارج المثيل الأعلى، فيمكنه استخدام tracefs_synth_set_instance() لوجعل الرسوم البيانية المستخدمة لإنشاء الحدث الاصطناعي في مثيل آخر غير المستوى الأعلى. يجب إنشاء واصف حدث اصطناعي باستخدام tracefs_synth_alloc(3) قبل أن يمكن استخدامه لإنشائه على النظام.

tracefs_synth_destroy() يدمر الحدث الاصطناعي. سيحاول إيقاف تشغيله في مثيله (المستوى الأعلى افتراضيًا)، ولكن إذا كان قيد التشغيل في مثيل آخر فقد يفشل لأنه مشغول.

tracefs_synth_complete() تعود بقيمة صحيحة إذا كان الحدث الاصطناعي synth يحتوي على حدث بداية وحدث نهاية.

tracefs_synth_trace() بدلاً من إجراء تتبع بسيط عند مطابقة أحداث البداية والنهاية، قم بتنفيذ معالج type حيث سيقوم TRACEFS_SYNTH_HANDLE_MAX بتتبع عند وصول المتغير var المحدد إلى حد أقصى جديد للمفاتيح المتطابقة. أو TRACEFS_SYNTH_HANDLE_CHANGE عندما يتغير var. يجب أن يكون var أحد عناصر name المستخدمة في tracefs_synth_add_end_field(3).

tracefs_synth_snapshot() عندما يكون المتغير var المحدد إما حدًا أقصى جديدًا إذا كان handler هو TRACEFS_SYNTH_HANDLE_MAX أو ببساطة قد تغير إذا كان TRACEFS_SYNTH_HANDLE_CHANGE، فخذ "لقطة" من المخزن المؤقت. تنقل اللقطة المخزن المؤقت "trace" العادي إلى مخزن مؤقت "snapshot"، والذي يمكن الوصول إليه عبر ملف "snapshot" في دليل tracefs عالي المستوى، أو أحد المثيلات. يتغير var. يجب أن يكون var أحد عناصر name المستخدمة في tracefs_synth_add_end_field(3).

tracefs_synth_save() عندما يكون المتغير var المحدد إما حدًا أقصى جديدًا إذا كان handler هو TRACEFS_SYNTH_HANDLE_MAX أو ببساطة قد تغير إذا كان TRACEFS_SYNTH_HANDLE_CHANGE، فقم بحفظ قائمة save_fields المحددة. سيتم تخزين الحقول في ملف "hist" للحدث، والذي يمكن استرداده باستخدام tracefs_event_file_read(3). يجب أن يكون var أحد عناصر name المستخدمة في tracefs_synth_add_end_field(3).

tracefs_synth_set_instance() قم بتعيين مثيل التتبع، حيث سيتم إنشاء مشغلات الرسم البياني التي تنشئ الحدث الاصطناعي. افتراضيًا، يتم استخدام المثيل العلوي. يجب استدعاء واجهة برمجة التطبيقات هذه قبل استدعاء tracefs_synth_create()، من أجل استخدام المثيل الجديد عند إنشاء الحدث. لاحظ أنه حتى لو تم إنشاء الحدث الاصطناعي في مثيل، فإنه لا يزال مرئيًا لجميع المثيلات الأخرى بما في ذلك المستوى الأعلى. أي، يمكن للمثيلات الأخرى تمكين الحدث الاصطناعي الذي تم إنشاؤه وجعله متتبعًا في المخازن المؤقتة التي تنتمي إلى المثيل الذي قام بتمكينه.

قيمة الإرجاع

جميع الدوال تعود بصفر عند النجاح أو -1 عند الخطأ.

الأخطاء

الأخطاء التالية تنطبق على جميع الاستدعاءات المذكورة أعلاه:

EPERM لم يتم التشغيل كمستخدم جذر عند الحاجة.

EINVAL إما أن يكون المعامل غير صالح (NULL عندما لا ينبغي أن يكون كذلك) أو أن يكون حقل غير متوافق مع العمليات الحسابية.

ENODEV لم يتم العثور على حدث أو أحد حقوله.

EBADE حقول حدثي البداية والنهاية غير متوافقة إما للمطابقة أو للمقارنة.

ENOMEM لا توجد ذاكرة كافية متاحة.

وقد تحدث أخطاء أخرى من استدعاءات النظام إلى النظام.

مثال

راجع tracefs_sql(3) لمزيد من الاستخدام المتعمق لبعض هذه التعليمات البرمجية.

#include <stdlib.h>
#include <tracefs.h>
#define start_event "sched_waking"
#define start_field "pid"
#define end_event "sched_switch"
#define end_field "next_pid"
#define match_name "pid"
static struct tracefs_synth *synth;
static void make_event(void)
{
        struct tep_handle *tep;
        /* تحميل جميع الأحداث من النظام */
        tep = tracefs_local_events(NULL);
        /* تهيئة الحدث الاصطناعي */
        synth = tracefs_synth_alloc(tep, "wakeup_lat",
                                    NULL, start_event,
                                    NULL, end_event,
                                    start_field, end_field,
                                    match_name);
        /* لم تعد هناك حاجة إلى tep */
        tep_free(tep);
        /* حفظ حقل "prio" باسم "prio" من حدث البداية */
        tracefs_synth_add_start_field(synth, "prio", NULL);
        /* حفظ حقل "next_comm" باسم "comm" من حدث النهاية */
        tracefs_synth_add_end_field(synth, "next_comm", "comm");
        /* حفظ حقل "prev_prio" باسم "prev_prio" من حدث النهاية */
        tracefs_synth_add_end_field(synth, "prev_prio", NULL);
        /*
         * خذ فرق زمني بالميكروثانية بين النهاية والبداية
         * وسجل كـ "delta"
         */
        tracefs_synth_add_compare_field(synth, TRACEFS_TIMESTAMP_USECS,
                                        TRACEFS_TIMESTAMP_USECS,
                                        TRACEFS_SYNTH_DELTA_END, "delta");
        /* سجل فقط إذا كان "prio" لحدث البداية أقل من 100 */
        tracefs_synth_append_start_filter(synth, TRACEFS_FILTER_COMPARE,
                                          "prio", TRACEFS_COMPARE_LT, "100");
        /*
         * سجل فقط إذا كان "next_prio" لحدث النهاية أقل من 50
         * أو أن أولوية المهمة السابقة لم تكن أكبر من أو تساوي 100.
         *   next_prio < 50 || !(prev_prio >= 100)
         */
        tracefs_synth_append_end_filter(synth, TRACEFS_FILTER_COMPARE,
                                        "next_prio", TRACEFS_COMPARE_LT, "50");
        tracefs_synth_append_end_filter(synth, TRACEFS_FILTER_OR, NULL, 0, NULL);
        tracefs_synth_append_end_filter(synth, TRACEFS_FILTER_NOT, NULL, 0, NULL);
