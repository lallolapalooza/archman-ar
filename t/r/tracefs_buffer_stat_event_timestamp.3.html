tracefs_buffer_stat_event_timestamp(3)

الحزمة: libtracefs

التراخيص: GPL-2.0-only, LGPL-2.1-only

LIBTRACEFS(3) دليل libtracefs LIBTRACEFS(3)

الاسم

tracefs_instance_get_stat, tracefs_instance_put_stat, tracefs_buffer_stat_entries, tracefs_buffer_stat_overrun, tracefs_buffer_stat_commit_overrun, tracefs_buffer_stat_bytes, tracefs_buffer_stat_event_timestamp, tracefs_buffer_stat_timestamp, tracefs_buffer_stat_dropped_events, tracefs_buffer_stat_read_events - التعامل مع إحصائيات مخزن التتبع المؤقت

الملخص

#include <tracefs.h>
struct tracefs_buffer_stat *tracefs_instance_get_stat(struct tracefs_instance *instance, int cpu);
void tracefs_instance_put_stat(struct tracefs_buffer_stat *tstat);
ssize_t tracefs_buffer_stat_entries(struct tracefs_buffer_stat *tstat);
ssize_t tracefs_buffer_stat_overrun(struct tracefs_buffer_stat *tstat);
ssize_t tracefs_buffer_stat_commit_overrun(struct tracefs_buffer_stat *tstat);
ssize_t tracefs_buffer_stat_bytes(struct tracefs_buffer_stat *tstat);
long long tracefs_buffer_stat_event_timestamp(struct tracefs_buffer_stat *tstat);
long long tracefs_buffer_stat_timestamp(struct tracefs_buffer_stat *tstat);
ssize_t tracefs_buffer_stat_dropped_events(struct tracefs_buffer_stat *tstat);
ssize_t tracefs_buffer_stat_read_events(struct tracefs_buffer_stat *tstat);

الوصف

تقوم هذه المجموعة من الدوال بقراءة وتحليل ملف tracefs/per_cpu/cpuX/stats. تحتفظ هذه الملفات بإحصائيات المخزن المؤقت الدائري لكل نواة، مثل عدد الأحداث الموجودة في المخزن المؤقت، وعدد الأحداث التي تمت قراءتها وما إلى ذلك.

تقوم الدالة tracefs_instance_get_stat() بقراءة وتحليل ملف الإحصائيات المحدد لمثيل instance وcpu معينين. نظرًا لأن المخزن المؤقت الدائري مقسم إلى مخازن مؤقتة لكل نواة، فإن المعلومات ترتبط فقط بـ cpu المحدد. يمكن استخدام المؤشر tracefs_buffer_stat المُرجع مع دوال tracefs_buffer_stat الأخرى ويجب تحريره باستخدام tracefs_instance_put_stat().

تقوم الدالة tracefs_instance_put_stat() بتحرير الموارد المخصصة لـ stat المحدد الذي تم إنشاؤه بواسطة tracefs_instance_get_stat().

تقوم الدالة tracefs_buffer_stat_entries() بإرجاع عدد الأحداث الموجودة حاليًا في المخزن المؤقت الدائري المرتبط بـ tstat.

تقوم الدالة tracefs_buffer_stat_overrun() بإرجاع عدد الأحداث التي فقدتها آلية كتابة المخزن المؤقت الدائري عندما تجاوزت القارئ.

تقوم الدالة tracefs_buffer_stat_commit_overrun() بإرجاع عدد الأحداث التي فقدت بسبب صغر حجم المخزن المؤقت الدائري وتقاطع مقاطعة مع حدث سياق أدنى قيد التسجيل وأضافت أحداثًا أكثر مما يمكن للمخزن المؤقت الدائري استيعابه. ملاحظة: هذا ليس حدثًا شائعًا وعندما يحدث فإنه يعني أن شيئًا ما لم يتم إعداده بشكل صحيح.

تقوم الدالة tracefs_buffer_stat_bytes() بإرجاع عدد البايتات التي تشغلها الأحداث الحالية. ملاحظة: تتضمن البيانات الوصفية للأحداث، ولكنها لا تشمل البيانات الوصفية للمخازن المؤقتة الفرعية.

تقوم الدالة tracefs_buffer_stat_event_timestamp() بإرجاع الطابع الزمني لآخر حدث في المخزن المؤقت الدائري.

تقوم الدالة tracefs_buffer_stat_timestamp() بإرجاع الطابع الزمني الحالي للمخزن المؤقت الدائري. ملاحظة: يتم قراءته فقط عند استدعاء tracefs_instance_get_stat(). سيكون لديها الطابع الزمني للمخزن المؤقت الدائري عند استدعاء تلك الدالة.

تقوم الدالة tracefs_buffer_stat_dropped_events() بإرجاع عدد الأحداث التي تم إسقاطها إذا كان وضع الكتابة فوق تعطيل. ستظهر الأحداث التي فقدت لأن الكاتب لحق بالقارئ ولم يتمكن من كتابة المزيد من الأحداث.

تقوم الدالة tracefs_buffer_stat_read_events() بإرجاع عدد الأحداث التي تم استهلاكها بواسطة القارئ.

قيمة الإرجاع

تقوم الدالة tracefs_instance_get_stat() بإرجاع بنية tracefs_buffer_stat التي يمكن استخدامها لاسترداد الإحصائيات عبر الدوال الأخرى. يجب تحريرها باستخدام tracefs_instance_put_stat().

جميع الدوال الأخرى التي تعيد قيمًا مختلفة من بنية tracefs_buffer_stat تعيد القيمة، أو -1 إذا لم يتم العثور على القيمة.

مثال

#include <stdlib.h>
#include <unistd.h>
#include <tracefs.h>
int main(int argc, char **argv)
{
        char *trace;
        char buf[1000];
        int ret;
        int i;
        for (i = 0; i < sizeof(buf) - 1; i++) {
                buf[i] = '0' + i % 10;
        }
        buf[i] = '\0';
        tracefs_instance_clear(NULL);
        for (i = 0; i < 4; i++) {
                ret = tracefs_printf(NULL, "%s\n", buf);
                if (ret < 0)
                        perror("write");
        }
        trace = tracefs_instance_file_read(NULL, "trace", NULL);
        printf("%s\n", trace);
        free(trace);
        for (i = 0; i < sysconf(_SC_NPROCESSORS_CONF); i++) {
                struct tracefs_buffer_stat *tstat;
                ssize_t entries, eread;
                tstat = tracefs_instance_get_stat(NULL, i);
                if (!tstat)
                        continue;
                entries = tracefs_buffer_stat_entries(tstat);
                eread = tracefs_buffer_stat_read_events(tstat);
                if (!entries && !eread) {
                        tracefs_instance_put_stat(tstat);
                        continue;
                }
                printf("CPU: %d\n", i);;
                printf("\tentries: %zd\n", entries);
                printf("\toverrun: %zd\n", tracefs_buffer_stat_overrun(tstat));
                printf("\tcommit_overrun: %zd\n", tracefs_buffer_stat_commit_overrun(tstat));
                printf("\tbytes: %zd\n", tracefs_buffer_stat_bytes(tstat));
                printf("\tevent_timestamp: %lld\n", tracefs_buffer_stat_event_timestamp(tstat));
                printf("\ttimestamp: %lld\n", tracefs_buffer_stat_timestamp(tstat));
                printf("\tdropped_events: %zd\n", tracefs_buffer_stat_dropped_events(tstat));
                printf("\tread_events: %zd\n", eread);
                tracefs_instance_put_stat(tstat);
        }
}
