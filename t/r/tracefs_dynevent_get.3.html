tracefs_dynevent_get(3)

الحزمة: libtracefs

التراخيص: GPL-2.0-only, LGPL-2.1-only

LIBTRACEFS(3) دليل libtracefs LIBTRACEFS(3)

الاسم

tracefs_dynevent_create, tracefs_dynevent_destroy, tracefs_dynevent_destroy_all, tracefs_dynevent_free, tracefs_dynevent_list_free, tracefs_dynevent_get, tracefs_dynevent_get_all, tracefs_dynevent_info, tracefs_dynevent_get_event - إنشاء وتدمير وتحرير والحصول على الأحداث الديناميكية.

الملخص

#include <tracefs.h>
struct tracefs_dynevent;
enum tracefs_dynevent_type;
int tracefs_dynevent_create(struct tracefs_dynevent *devent);
int tracefs_dynevent_destroy(struct tracefs_dynevent *devent, bool force);
int tracefs_dynevent_destroy_all(unsigned int types, bool force);
void tracefs_dynevent_free(struct tracefs_dynevent *devent);
void tracefs_dynevent_list_free(struct tracefs_dynevent **events);
struct tracefs_dynevent *tracefs_dynevent_get(enum tracefs_dynevent_type type, const char *system, const char *event);
struct tracefs_dynevent **tracefs_dynevent_get_all(unsigned int types, const char *system);
enum tracefs_dynevent_type tracefs_dynevent_info(struct tracefs_dynevent *dynevent, char **system, char **event, char **prefix, char **addr, char **format);
struct tep_event *tracefs_dynevent_get_event(struct tep_handle *tep, struct tracefs_dynevent *dynevent);

الوصف

تقوم الدالة tracefs_dynevent_create() بإنشاء الحدث الديناميكي devent في النظام.

تقوم الدالة tracefs_dynevent_destroy() بإزالة الحدث الديناميكي devent من النظام. إذا كانت force صحيحة، ستحاول الدالة تعطيل جميع الأحداث في جميع مثيلات التتبع، قبل إزالة الحدث الديناميكي. لا يتم تحرير سياق devent، استخدم tracefs_dynevent_free() لتحريره.

تقوم الدالة tracefs_dynevent_destroy_all() بإزالة جميع الأحداث الديناميكية من الأنواع المحددة من النظام. المعامل types هو نوع حدث ديناميكي محدد، أو قناع بت لأنواع الأحداث الديناميكية tracefs_dynevent_type، التي سيتم إزالتها. إذا كان types هو 0، سيتم إزالة الأحداث الديناميكية من جميع الأنواع. إذا كانت force صحيحة، ستحاول الدالة تعطيل جميع الأحداث في جميع مثيلات التتبع، قبل إزالة الأحداث الديناميكية.

تقوم الدالة tracefs_dynevent_get() بتخصيص وإرجاع مثيل واحد لحدث ديناميكي يطابق type و system و event المعطاة. يتم إرجاع NULL إذا لم يكن هناك تطابق. الحدث المرجع هو ما تم العثور عليه في النظام، ويجب تحريره باستخدام tracefs_dynevent_free(). إذا كان system هو NULL، فسيتم إرجاع أول event من أي نظام من النوع المحدد الذي له اسم event.

تقوم الدالة tracefs_dynevent_get_all() بتخصيص وإرجاع مصفوفة من المؤشرات إلى الأحداث الديناميكية من الأنواع المحددة الموجودة في النظام. العنصر الأخير في المصفوفة هو مؤشر NULL. يجب تحرير المصفوفة باستخدام tracefs_dynevent_list_free(). إذا لم تكن هناك أحداث، يتم إرجاع مؤشر NULL. المعامل types هو نوع حدث ديناميكي محدد، أو قناع بت لأنواع الأحداث الديناميكية tracefs_dynevent_type، التي سيتم استرجاعها. إذا كان types هو 0، سيتم استرجاع الأحداث الديناميكية من جميع الأنواع.

تقوم الدالة tracefs_dynevent_free() بتحرير سياق حدث ديناميكي devent.

تقوم الدالة tracefs_dynevent_list_free() بتحرير مصفوفة من المؤشرات إلى أحداث ديناميكية، تم إرجاعها بواسطة واجهة برمجة التطبيقات tracefs_dynevent_get_all().

تقوم الدالة tracefs_dynevent_info() بإرجاع نوع ومعلومات حدث ديناميكي معين dynevent. إذا لم تكن أي من المعاملات system، event، prefix، addr أو format هي NULL، فسيتم تخصيص سلاسل نصية وإرجاعها عبر هذه المعاملات. system و event يحتويان على النظام واسم الحدث الديناميكي. إذا كان prefix غير NULL، فسيحتوي على سلسلة نصية مخصصة تحمل الجزء البادئ للحدث الديناميكي (المحتوى حتى ":"، باستثناءه). إذا كان addr غير NULL، فسيحتوي على العنوان أو الدالة التي يرتبط بها الحدث الديناميكي، إذا كان ذلك ذا صلة بنوع الحدث هذا. إذا كان format غير NULL، فسيحتوي على سلسلة التنسيق للحدث الديناميكي. ملاحظة، يجب تحرير المحتوى في group، event، prefix، addr، و format باستخدام free(3) إذا تم تعيينها.

تقوم الدالة tracefs_dynevent_get_event() بإرجاع حدث tep، يصف الحدث الديناميكي المحدد. تكتشف واجهة برمجة التطبيقات أي أحداث ديناميكية تم إنشاؤها أو إزالتها حديثًا. لا يجب تحرير المؤشر المرجع إلى حدث tep.

قيمة الإرجاع

tracefs_dynevent_create() ترجع 0 عند النجاح، أو -1 عند حدوث خطأ. إذا حدث خطأ في التحليل، يمكن استخدام tracefs_error_last(3) لاسترداد رسالة الخطأ التي توضح مشكلة التحليل.

tracefs_dynevent_destroy() و tracefs_dynevent_destroy_all() ترجعان 0 عند النجاح، أو -1 عند حدوث خطأ. إذا تم تمكين force، قد تفشل الدوال في تعطيل الأحداث.

tracefs_dynevent_get() ترجع حدث ديناميكي مخصص من النظام يطابق النوع والنظام والحدث المعطيين.

tracefs_dynevent_get_all() ترجع مصفوفة مخصصة من المؤشرات إلى الأحداث الديناميكية، أو NULL في حالة حدوث خطأ أو في حالة عدم وجود أحداث في النظام. يجب تحرير تلك المصفوفة بواسطة tracefs_dynevent_list_free().

tracefs_dynevent_info() ترجع نوع الحدث الديناميكي المعطى أو TRACEFS_DYNEVENT_UNKNOWN عند حدوث خطأ. إذا كانت system، event، prefix، addr، أو format غير NULL، فستحتوي على سلاسل نصية مخصصة يجب تحريرها بواسطة free(3).

tracefs_dynevent_get_event() ترجع مؤشرًا إلى حدث tep أو NULL في حالة حدوث خطأ أو إذا كان الحدث الديناميكي المطلوب مفقودًا. المؤشر المرجع إلى حدث tep يتم التحكم فيه بواسطة @tep ولا يجب تحريره.

الأخطاء

الأخطاء التالية تنطبق على جميع الاستدعاءات المذكورة أعلاه:

ENODEV الأحداث الديناميكية من النوع المطلوب غير مهيأة للنواة قيد التشغيل.

ENOMEM خطأ في تخصيص الذاكرة.

tracefs_dynevent_create() يمكن أن تفشل مع الأخطاء التالية:

EINVAL على الأرجح حدث خطأ في التحليل (استخدم tracefs_error_last(3) لرؤية الخطأ المحتمل).

قد تحدث أخطاء أخرى ناتجة عن استدعاءات نظام داخلية.
