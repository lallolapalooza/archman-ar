tracefs_filter_string_verify(3)

الحزمة: libtracefs

التراخيص: GPL-2.0-only, LGPL-2.1-only

'&': compare = TRACEFS_COMPARE_AND; break; } while (isspace(filter[i])) i++; quote = 0; backslash = false; start_i = i; for (; filter[i]; i++) { if (quote) { if (backslash) backslash = false; else if (filter[i] == '\\') backslash = true; else if (filter[i] == quote) quote = 0; continue; } switch(filter[i]) { case '"': case '\'': quote = filter[i]; continue; case 'a' ... 'z': case 'A' ... 'Z': case '0' ... '9': case '_': continue; } break; } n = i - start_i; strncpy(val, filter + start_i, n); val[n] = '\0'; break; } n = tracefs_filter_string_append(event, &new_filter, type, field, compare, val); if (n < 0) { fprintf(stderr, "Failed making new filter:\n'%s'\n", new_filter ? new_filter : "(null)"); exit(-1); } } if (tracefs_event_filter_apply(NULL, event, new_filter)) fprintf(stderr, "Failed to apply filter on event"); tep_free(tep); printf("Created new filter: '%s'\n", new_filter); free(new_filter); exit(0); }
