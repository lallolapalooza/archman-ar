tracefs_dynevent_create(3)

الحزمة: libtracefs

التراخيص: GPL-2.0-only, LGPL-2.1-only

LIBTRACEFS(3) دليل libtracefs LIBTRACEFS(3)

الاسم

tracefs_dynevent_create, tracefs_dynevent_destroy, tracefs_dynevent_destroy_all, tracefs_dynevent_free, tracefs_dynevent_list_free, tracefs_dynevent_get, tracefs_dynevent_get_all, tracefs_dynevent_info, tracefs_dynevent_get_event - إنشاء وتدمير وتحرير والحصول على الأحداث الديناميكية.

الملخص

#include <tracefs.h>
struct tracefs_dynevent;
enum tracefs_dynevent_type;
int tracefs_dynevent_create(struct tracefs_dynevent *devent);
int tracefs_dynevent_destroy(struct tracefs_dynevent *devent, bool force);
int tracefs_dynevent_destroy_all(unsigned int types, bool force);
void tracefs_dynevent_free(struct tracefs_dynevent *devent);
void tracefs_dynevent_list_free(struct tracefs_dynevent **events);
struct tracefs_dynevent *tracefs_dynevent_get(enum tracefs_dynevent_type type, const char *system, const char *event);
struct tracefs_dynevent **tracefs_dynevent_get_all(unsigned int types, const char *system);
enum tracefs_dynevent_type tracefs_dynevent_info(struct tracefs_dynevent *dynevent, char **system, char **event, char **prefix, char **addr, char **format);
struct tep_event *tracefs_dynevent_get_event(struct tep_handle *tep, struct tracefs_dynevent *dynevent);

الوصف

تقوم الدالة tracefs_dynevent_create() بإنشاء الحدث الديناميكي devent في النظام.

تقوم الدالة tracefs_dynevent_destroy() بإزالة الحدث الديناميكي devent من النظام. إذا كانت force صحيحة، فستحاول الدالة تعطيل جميع الأحداث في جميع نسخ التتبع، قبل إزالة الحدث الديناميكي. لا يتم تحرير سياق devent، استخدم tracefs_dynevent_free() لتحريره.

تقوم الدالة tracefs_dynevent_destroy_all() بإزالة جميع الأحداث الديناميكية من الأنواع المحددة من النظام. المعامل types هو نوع حدث ديناميكي محدد، أو قناع بتات لأنواع الأحداث الديناميكية tracefs_dynevent_type، التي ستتم إزالتها. إذا كانت types تساوي 0، فستتم إزالة الأحداث الديناميكية من جميع الأنواع. إذا كانت force صحيحة، فستحاول الدالة تعطيل جميع الأحداث في جميع نسخ التتبع، قبل إزالة الأحداث الديناميكية.

تقوم الدالة tracefs_dynevent_get() بتخصيص وإرجاع نسخة واحدة من حدث ديناميكي يطابق type وsystem وevent المعطاة لها. يتم إرجاع NULL إذا لم يكن هناك تطابق. الحدث المُعاد هو ما تم العثور عليه في النظام، ويجب تحريره باستخدام tracefs_dynevent_free(). إذا كانت system تساوي NULL، فسيتم إرجاع أول event من أي نظام من النوع المحدد الذي يحمل اسم event.

تقوم الدالة tracefs_dynevent_get_all() بتخصيص وإرجاع مصفوفة من المؤشرات إلى الأحداث الديناميكية من الأنواع المحددة الموجودة في النظام. العنصر الأخير في المصفوفة هو مؤشر NULL. يجب تحرير المصفوفة باستخدام tracefs_dynevent_list_free(). إذا لم تكن هناك أحداث، يتم إرجاع مؤشر NULL. المعامل types هو نوع حدث ديناميكي محدد، أو قناع بتات لأنواع الأحداث الديناميكية tracefs_dynevent_type، التي سيتم استرجاعها. إذا كانت types تساوي 0، فسيتم استرجاع الأحداث الديناميكية من جميع الأنواع.

تقوم الدالة tracefs_dynevent_free() بتحرير سياق حدث ديناميكي devent.

تقوم الدالة tracefs_dynevent_list_free() بتحرير مصفوفة من المؤشرات إلى الأحداث الديناميكية، التي تم إرجاعها بواسطة واجهة tracefs_dynevent_get_all().

تقوم الدالة tracefs_dynevent_info() بإرجاع النوع و معلومات حدث ديناميكي معين dynevent. إذا كان أي من المعاملات system أو event أو prefix أو addr أو format لا يساوي NULL، فسيتم تخصيص سلاسل نصية وإرجاعها عبر هذه المعاملات. يحمل system وevent النظام و اسم الحدث الديناميكي. إذا كانت prefix لا تساوي NULL، فستحمل سلسلة نصية مخصصة تحتوي على جزء البادئة من الحدث الديناميكي (المحتوى حتى ":"، باستثنائه). إذا كانت addr لا تساوي NULL، فستحمل العنوان أو الدالة التي يرتبط بها الحدث الديناميكي، إذا كان ذلك ذا صلة بهذا النوع من الأحداث. إذا كانت format لا تساوي NULL، فستحمل سلسلة التنسيق للحدث الديناميكي. لاحظ أن المحتوى في group وevent وprefix وaddr وformat يجب تحريره باستخدام free(3) إذا تم تعيينه.

تقوم الدالة tracefs_dynevent_get_event() بإرجاع حدث tep يصف الحدث الديناميكي المعطى. تكتشف الواجهة أي أحداث ديناميكية تم إنشاؤها أو إزالتها حديثًا. المؤشر المُعاد إلى حدث tep يتم التحكم فيه بواسطة @tep ويجب عدم تحريره.

قيمة الإرجاع

تُرجع tracefs_dynevent_create() القيمة 0 عند النجاح، أو -1 عند الخطأ. إذا حدث خطأ في التحليل، فيمكن استخدام tracefs_error_last(3) لاسترداد رسالة الخطأ التي تشرح مشكلة التحليل.

تُرجع tracefs_dynevent_destroy() و tracefs_dynevent_destroy_all() القيمة 0 عند النجاح، أو -1 عند الخطأ. إذا كانت force مفعّلة، فقد تفشل الدوال في تعطيل الأحداث.

تُرجع الدالة tracefs_dynevent_get() حدثًا ديناميكيًا مخصصًا من النظام يطابق النوع والنظام والحدث المعطاة.

تُرجع الدالة tracefs_dynevent_get_all() مصفوفة مخصصة من المؤشرات إلى الأحداث الديناميكية، أو NULL في حالة خطأ أو في حالة عدم وجود أحداث في النظام. يجب تحرير تلك المصفوفة بواسطة tracefs_dynevent_list_free().

تُرجع tracefs_dynevent_info() نوع الحدث الديناميكي المعطى أو TRACEFS_DYNEVENT_UNKNOWN عند الخطأ. إذا كانت system أو event أو prefix أو addr أو format لا تساوي NULL، فستحتوي على سلاسل نصية مخصصة يجب تحريرها بواسطة free(3).

تُرجع الدالة tracefs_dynevent_get_event() مؤشرًا إلى حدث tep أو NULL في حالة خطأ أو إذا كان الحدث الديناميكي المطلوب مفقودًا. المؤشر المُعاد إلى حدث tep يتم التحكم فيه بواسطة @tep ويجب عدم تحريره.

الأخطاء

الأخطاء التالية تخص جميع الاستدعاءات أعلاه:

ENODEV الأحداث الديناميكية من النوع المطلوب غير مهيأة لنواة التشغيل الحالية.

ENOMEM خطأ في تخصيص الذاكرة.

قد تفشل tracefs_dynevent_create() بالأخطاء التالية:

EINVAL على الأرجح حدث خطأ في التحليل (استخدم tracefs_error_last(3) لربما ترى ما كان ذلك الخطأ).

قد تحدث أخطاء أخرى أيضًا بسبب استدعاءات النظام الداخلية.

مثال

#include <stdlib.h>
#include <unistd.h>
#include <sys/wait.h>
#include <tracefs.h>
static struct tep_event *open_event;
static struct tep_format_field *file_field;
static struct tep_event *openret_event;
static struct tep_format_field *ret_field;
static int callback(struct tep_event *event, struct tep_record *record,
                    int cpu, void *data)
{
        struct trace_seq seq;
        trace_seq_init(&seq);
        tep_print_event(event->tep, &seq, record, "%d-%s: ", TEP_PRINT_PID, TEP_PRINT_COMM);
        if (event->id == open_event->id) {
                trace_seq_puts(&seq, "open file='");
                tep_print_field(&seq, record->data, file_field);
                trace_seq_puts(&seq, "'\n");
        } else if (event->id == openret_event->id) {
                unsigned long long ret;
                tep_read_number_field(ret_field, record->data, &ret);
                trace_seq_printf(&seq, "open ret=%lld\n", ret);
        } else {
                goto out;
        }
        trace_seq_terminate(&seq);
        trace_seq_do_printf(&seq);
out:
        trace_seq_destroy(&seq);
        return 0;
}
static pid_t run_exec(char **argv, char **env)
{
        pid_t pid;
        pid = fork();
        if (pid)
                return pid;
        execve(argv[0], argv, env);
        perror("exec");
        exit(-1);
}
const char *mykprobe = "my_kprobes";
int main (int argc, char **argv, char **env)
{
        struct tracefs_dynevent *kprobe, *kretprobe;
        const char *sysnames[] = { mykprobe, NULL };
        struct tracefs_instance *instance;
        struct tep_handle *tep;
        pid_t pid;
        if (argc < 2) {
                printf("usage: %s command\n", argv[0]);
                exit(-1);
        }
        instance = tracefs_instance_create("exec_open");
        if (!instance) {
                perror("creating instance");
                exit(-1);
        }
        tracefs_dynevent_destroy_all(TRACEFS_DYNEVENT_KPROBE | TRACEFS_DYNEVENT_KRETPROBE, true);
        kprobe = tracefs_kprobe_alloc(mykprobe, "open", "do_sys_openat2",
                                      "file=+0($arg2):ustring flags=+0($arg3):x64 mode=+8($arg3):x64\n");
        kretprobe = tracefs_kretprobe_alloc(mykprobe, "openret", "do_sys_openat2", "ret=%ax", 0);
        if (!kprobe || !kretprobe) {
                perror("allocating dynamic events");
                exit(-1);
        }
        if (tracefs_dynevent_create(kprobe) || tracefs_dynevent_create(kretprobe)){
                char *err = tracefs_error_last(NULL);
                perror("Failed to create kprobes:");
                if (err && strlen(err))
                        fprintf(stderr, "%s\n", err);
                exit(-1);
        }
        tep = tracefs_local_events_system(NULL, sysnames);
        if (!tep) {
                perror("reading events");
                exit(-1);
        }
        open_event = tep_find_event_by_name(tep, mykprobe, "open");
        file_field = tep_find_field(open_event, "file");
        openret_event = tep_find_event_by_name(tep, mykprobe, "openret");
        ret_field = tep_find_field(openret_event, "ret");
        tracefs_event_enable(instance, mykprobe, NULL);
        pid = run_exec(&argv[1], env);
        /* Let the child start to run */
        sched_yield();
        do {
                tracefs_load_cmdlines(NULL, tep);
                tracefs_iterate_raw_events(tep, instance, NULL, 0, callback, NULL);
        } while (waitpid(pid, NULL, WNOHANG) != pid);
        /* Will disable the events */
        tracefs_dynevent_destroy_all(TRACEFS_DYNEVENT_KPROBE | TRACEFS_DYNEVENT_KRETPROBE, true);
        tracefs_dynevent_free(kprobe);
        tracefs_dynevent_free(kretprobe);
        tracefs_instance_destroy(instance);
        tep_free(tep);
        return 0;
}
