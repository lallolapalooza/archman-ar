tracecmd_get_tep(3)

الحزمة: trace-cmd

التراخيص: GPL-2.0-only, LGPL-2.1-only

LIBTRACECMD(3) دليل libtracefs LIBTRACECMD(3)

الاسم

tracecmd_read_cpu_first, tracecmd_read_data, tracecmd_read_at, tracecmd_free_record, tracecmd_get_tep - قراءة الأحداث المسجلة من ملف تتبع.

الملخص

#include <trace-cmd.h>
struct tep_record *tracecmd_read_cpu_first(struct tracecmd_input *handle, int cpu);
struct tep_record *tracecmd_read_data(struct tracecmd_input *handle, int cpu);
struct tep_record *tracecmd_read_at(struct tracecmd_input *handle, unsigned long long offset, int *cpu);
void tracecmd_free_record(struct tep_record *record);
struct tep_handle *tracecmd_get_tep(struct tracecmd_input *handle);

الوصف

يمكن استخدام مجموعة واجهات برمجة التطبيقات هذه لقراءة بيانات التتبع من ملف تتبع تم فتحه باستخدام tracecmd_open()(3) أو tracecmd_open_fd()(3) أو tracecmd_open_head()(3).

تقوم الدالة tracecmd_read_cpu_first() بقراءة سجل التتبع الأول لوحدة المعالجة المركزية cpu المعطاة من ملف تتبع مرتبط بـ handle. يجب تحرير السجل المُرجع باستخدام tracecmd_free_record().

تقوم الدالة tracecmd_read_data() بقراءة سجل التتبع التالي لوحدة المعالجة المركزية cpu المعطاة من ملف تتبع مرتبط بـ handle وتزيد مؤشر موقع القراءة، بحيث أن الاستدعاء التالي لـ tracecmd_read_data() لن يقرأ نفس السجل مرة أخرى. يجب تحرير السجل المُرجع باستخدام tracecmd_free_record().

تقوم الدالة tracecmd_read_at() بقراءة سجل تتبع من offset محدد داخل الملف المرتبط بـ handle. يتم تخزين وحدة المعالجة المركزية التي حدث عليها حدث التتبع في cpu. لا تغير الدالة مؤشر موقع القراءة الحالي. يجب تحرير السجل المُرجع باستخدام tracecmd_free_record().

تقوم الدالة tracecmd_free_record() بتحرير record مُرجع بواسطة أي من واجهات برمجة التطبيقات tracecmd_read_.

تقوم الدالة tracecmd_get_tep() بإرجاع سياق tep لـ handle معين.

قيمة الإرجاع

ترجع الدوال tracecmd_read_cpu_first() و tracecmd_read_data() و tracecmd_read_at() مؤشرًا إلى struct tep_record أو NULL في حالة حدوث خطأ. يجب تحرير السجل المُرجع باستخدام tracecmd_free_record().

ترجع الدالة tracecmd_get_tep() مؤشرًا إلى سياق tep أو NULL إذا لم يكن هناك سياق tep لـ handle المحدد. يجب **عدم** تحرير مؤشر tep المُرجع.

مثال

#include <trace-cmd.h>
...
struct tracecmd_input *handle = tracecmd_open("trace.dat");
        if (!handle) {
                /* فشل فتح ملف trace.dat */
        }
...
unsigned long long offset = 0;
struct tep_record *rec;
int cpu = 0;
        rec = tracecmd_read_cpu_first(handle, cpu);
        while (rec) {
                ...
                if ( /* تم ملاحظة سجل مثير للاهتمام */) {
                        /* تخزين موقع السجل المثير للاهتمام */
                        offset = rec->offset;
                }
                ...
                tracecmd_free_record(rec);
                rec = tracecmd_read_data(handle, cpu);
        }
        ...
        if (offset) {
                rec = tracecmd_read_at(handle, offset, &cpu);
                if (rec) {
                        /* تم الحصول على السجل في الموقع على وحدة المعالجة المركزية */
                        ...
                        tracecmd_free_record(rec);
                }
        }
...
        tracecmd_close(hadle);
