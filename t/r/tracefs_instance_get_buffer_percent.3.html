tracefs_instance_get_buffer_percent(3)

الحزمة: libtracefs

التراخيص: GPL-2.0-only, LGPL-2.1-only

LIBTRACEFS(3) libtracefs Manual LIBTRACEFS(3)

الاسم

tracefs_instance_get_name, tracefs_instance_get_trace_dir, tracefs_instances_walk, tracefs_instance_exists, tracefs_instance_get_buffer_size, tracefs_instance_set_buffer_size, tracefs_instance_get_buffer_percent, tracefs_instance_set_buffer_percent - وظائف مساعدة للعمل مع مثيلات التتبع.

الملخص

#include <tracefs.h>
const char *tracefs_instance_get_name(struct tracefs_instance *instance);
const char *tracefs_instance_get_trace_dir(struct tracefs_instance *instance);
int tracefs_instances_walk(int (*callback)(const char *, void *), void *context);
bool tracefs_instance_exists(const char *name);
size_t tracefs_instance_get_buffer_size(struct tracefs_instance *instance, int cpu);
int tracefs_instance_set_buffer_size(struct tracefs_instance *instance, size_t size, int cpu);
int tracefs_instance_get_buffer_percent(struct tracefs_instance *instance);
int tracefs_instance_set_buffer_percent(struct tracefs_instance *instance, int val);

الوصف

وظائف مساعدة للعمل مع مثيلات التتبع.

تقوم الدالة tracefs_instance_get_name() بإرجاع اسم instance المحدد. لاحظ أن المثيل العلوي ليس له اسم، والدالة ترجع NULL له.

تقوم الدالة tracefs_instance_get_trace_dir() بإرجاع دليل التتبع، حيث تم تكوين instance المحدد.

تقوم الدالة tracefs_instances_walk() بالمرور عبر جميع مثيلات التتبع المهيأة في النظام وتستدعي callback لكل منها. يتم تمرير الوسيط context إلى callback، جنبًا إلى جنب مع اسم المثيل. إذا أرجعت callback قيمة غير صفرية، تتوقف الدورة. لاحظ أنه لا يتم استدعاء callback للمثيل العلوي.

تقوم الدالة tracefs_instance_exists() بالتحقق مما إذا كان مثيل بالاسم name المحدد موجودًا في النظام.

تقوم الدالة tracefs_instace_get_buffer_size() بإرجاع حجم المخزن المؤقت الدائري. إذا كانت cpu سالبة، فإنها ترجع الحجم الإجمالي لجميع المخازن المؤقتة الدائرية لكل CPU، وإلا فإنها ترجع حجم المخزن المؤقت الدائري لكل CPU لـ cpu.

تقوم الدالة tracefs_instance_set_buffer_size() بتعيين حجم المخزن المؤقت الدائري. إذا كانت cpu سالبة، فإنها تعين جميع المخازن المؤقتة الدائرية لكل CPU إلى size (لاحظ أن الحجم الإجمالي هو عدد وحدات المعالجة المركزية * size). إذا تم تحديد cpu، فإنها تعين فقط حجم المخزن المؤقت الدائري لكل CPU.

تقوم الدالة tracefs_instance_set_buffer_percent() بتعيين قيمة نسبة المخزن المؤقت للمخزن المؤقت الدائري للتتبع للمثيل instance أو للمخزن المؤقت على المستوى الأعلى إذا كان instance هو NULL. تحدد نسبة المخزن المؤقت متى سيتم حظر القراءات على tracefs_cpu_read(3)، tracefs_cpu_buffered_read(3)، tracefs_cpu_write(3) و tracefs_cpu_pipe(3) عندما لا يتم تعيين O_NONBLOCK. يجب أن تكون قيمة val بين 0 و 100، حيث:

  0   - حظر حتى يكون هناك أي بيانات في المخزن المؤقت الدائري
  1   - حظر حتى يتم ملء 1٪ من المخازن الفرعية للمخزن المؤقت الدائري
  50  - حظر حتى يتم ملء 50٪ من المخازن الفرعية للمخزن المؤقت الدائري
  100 - حظر حتى يتم ملء المخزن المؤقت الدائري بأكمله

لاحظ أنه يمكن استخدام أي رقم من 0 إلى 100 حيث يمثل النسبة المئوية للمخزن المؤقت الدائري التي يجب ملؤها قبل أن يتم إعلام القارئ المحظور بوجود بيانات للاسترداد.

تقوم الدالة tracefs_instance_get_buffer_percent() باسترداد إعداد نسبة المخزن المؤقت الحالي للمخزن المؤقت الدائري للتتبع للمثيل instance أو للمخزن المؤقت على المستوى الأعلى إذا كان instance هو NULL.

قيمة الإرجاع

ترجع الدالة tracefs_instance_get_name() سلسلة نصية أو NULL في حالة المثيل العلوي. يجب عدم تحرير السلسلة النصية المرجعة.

ترجع الدالة tracefs_instance_get_trace_dir() سلسلة نصية أو NULL في حالة حدوث خطأ. يجب عدم تحرير السلسلة النصية المرجعة.

ترجع الدالة tracefs_instances_walk() القيمة 0، إذا تم المرور عبر جميع المثيلات، 1 إذا تم إيقاف الدورة بواسطة callback، أو -1 في حالة حدوث خطأ.

ترجع الدالة tracefs_instance_exists() القيمة true إذا كان مثيل بالاسم name المحدد موجودًا في النظام أو false بخلاف ذلك.

ترجع الدالة tracefs_instance_get_buffer_size() حجم المخزن المؤقت الدائري اعتمادًا على قيمة cpu الممررة، أو -1 عند حدوث خطأ.

ترجع الدالة tracefs_instance_set_buffer_size() القيمة صفر عند النجاح و -1 عند حدوث خطأ.

#include <tracefs.h>
struct tracefs_instance *inst;
....
char *name = tracefs_instance_get_name(inst);
        if(name) {
                /* تم الحصول على اسم المثيل */
        }
char *dir = tracefs_instance_get_trace_dir(inst);
        if(dir) {
                /* تم الحصول على دليل التتبع للمثيل */
        }
...
static int instance_walk(char *name, void *context)
{
        /* تم الحصول على مثيل بالاسم */
        return 0;
}
...
        if (tracefs_instances_walk(instance_walk, NULL) < 0) {
                /* حدث خطأ أثناء المرور عبر المثيلات */
        }
...
        if (tracefs_instance_exists("foo")) {
                /* يوجد مثيل بالاسم foo في النظام */
        } else {
                /* لا يوجد مثيل بالاسم foo في النظام */
        }
