tracefs_instance_set_buffer_percent(3)

الحزمة: libtracefs

التراخيص: GPL-2.0-only, LGPL-2.1-only

LIBTRACEFS(3) دليل libtracefs LIBTRACEFS(3)

الاسم

tracefs_instance_get_name, tracefs_instance_get_trace_dir, tracefs_instances_walk, tracefs_instance_exists, tracefs_instance_get_buffer_size, tracefs_instance_set_buffer_size, tracefs_instance_get_buffer_percent, tracefs_instance_set_buffer_percent - وظائف مساعدة للتعامل مع مثيلات التتبع.

الملخص

#include <tracefs.h>
const char *tracefs_instance_get_name(struct tracefs_instance *instance);
const char *tracefs_instance_get_trace_dir(struct tracefs_instance *instance);
int tracefs_instances_walk(int (*callback)(const char *, void *), void *context);
bool tracefs_instance_exists(const char *name);
size_t tracefs_instance_get_buffer_size(struct tracefs_instance *instance, int cpu);
int tracefs_instance_set_buffer_size(struct tracefs_instance *instance, size_t size, int cpu);
int tracefs_instance_get_buffer_percent(struct tracefs_instance *instance);
int tracefs_instance_set_buffer_percent(struct tracefs_instance *instance, int val);

الوصف

وظائف مساعدة للتعامل مع مثيلات التتبع.

تقوم الدالة tracefs_instance_get_name() بإرجاع اسم instance المحدد. لاحظ أن المثيل الأعلى ليس له اسم، لذا ترجع الدالة NULL له.

تقوم الدالة tracefs_instance_get_trace_dir() بإرجاع دليل التتبع، حيث يتم تكوين instance المحدد.

تقوم الدالة tracefs_instances_walk() بالتجول عبر جميع مثيلات التتبع المهيأة في النظام واستدعاء callback لكل واحد منها. يتم تمرير الوسيط context إلى callback، بالإضافة إلى اسم المثيل. إذا أعادت callback قيمة غير صفرية، تتوقف العملية. لاحظ، أن callback لا يتم استدعاؤها للمثيل الأعلى.

تتحقق الدالة tracefs_instance_exists() مما إذا كان مثيل بالاسم name المحدد موجودًا في النظام.

ترجع الدالة tracefs_instace_get_buffer_size() حجم المخزن المؤقت الدائري. إذا كان cpu سالبًا، فإنها ترجع الحجم الإجمالي لجميع المخازن المؤقتة الدائرية لكل CPU، وإلا فإنها ترجع حجم المخزن المؤقت الدائري لكل CPU لـ cpu.

تقوم الدالة tracefs_instance_set_buffer_size() بتعيين حجم المخزن المؤقت الدائري. إذا كان cpu سالبًا، فإنها تقوم بتعيين جميع المخازن المؤقتة الدائرية لكل CPU إلى size (لاحظ أن الحجم الإجمالي هو عدد وحدات المعالجة المركزية * size). إذا تم تحديد cpu، فإنها تقوم فقط بتعيين حجم المخزن المؤقت الدائري لكل CPU.

تقوم الدالة tracefs_instance_set_buffer_percent() بتعيين قيمة النسبة المئوية للمخزن المؤقت لمخزن التتبع الدائري لـ instance أو المخزن المؤقت الأعلى إذا كان instance هو NULL. تحدد النسبة المئوية للمخزن المؤقت متى سيتم حظر القارئات على tracefs_cpu_read(3)، tracefs_cpu_buffered_read(3)، tracefs_cpu_write(3) و tracefs_cpu_pipe(3) عندما لا يتم تعيين O_NONBLOCK. يجب أن تكون قيمة val بين 0 و 100، حيث:

  0   - حظر حتى تكون هناك أي بيانات في المخزن المؤقت الدائري
  1   - حظر حتى تمتلئ 1٪ من المخازن المؤقتة الفرعية للمخزن المؤقت الدائري
  50  - حظر حتى تمتلئ 50٪ من المخازن المؤقتة الفرعية للمخزن المؤقت الدائري
  100 - حظر حتى يمتلئ المخزن المؤقت الدائري بالكامل

لاحظ، يمكن استخدام أي رقم من 0 إلى 100 حيث تمثل النسبة المئوية للمخزن المؤقت الدائري التي يجب ملؤها قبل إشعار القارئ المحظور بوجود بيانات لاستردادها.

تقوم الدالة tracefs_instance_get_buffer_percent() باسترداد إعداد النسبة المئوية الحالية للمخزن المؤقت لمخزن التتبع الدائري لـ instance أو المخزن المؤقت الأعلى إذا كان instance هو NULL.

قيمة الإرجاع

ترجع الدالة tracefs_instance_get_name() سلسلة نصية أو NULL في حالة المثيل الأعلى. لا يجب تحرير السلسلة النصية المرجعة.

ترجع الدالة tracefs_instance_get_trace_dir() سلسلة نصية أو NULL في حالة حدوث خطأ. لا يجب تحرير السلسلة النصية المرجعة.

ترجع الدالة tracefs_instances_walk() القيمة 0، إذا تم تكرار جميع المثيلات، 1 إذا تم إيقاف التكرار بواسطة callback، أو -1 في حالة حدوث خطأ.

ترجع الدالة tracefs_instance_exists() true إذا كان مثيل بالاسم name المحدد موجودًا في النظام أو false بخلاف ذلك.

ترجع الدالة tracefs_instance_get_buffer_size() حجم المخزن المؤقت الدائري اعتمادًا على قيمة cpu الممررة، أو -1 عند حدوث خطأ.

ترجع الدالة tracefs_instance_set_buffer_size() صفر عند النجاح و -1 عند حدوث خطأ.

#include <tracefs.h>
struct tracefs_instance *inst;
....
char *name = tracefs_instance_get_name(inst);
        if(name) {
                /* تم الحصول على اسم المثيل */
        }
char *dir = tracefs_instance_get_trace_dir(inst);
        if(dir) {
                /* تم الحصول على دليل تتبع المثيل */
        }
...
static int instance_walk(char *name, void *context)
{
        /* تم الحصول على مثيل بالاسم */
        return 0;
}
...
        if (tracefs_instances_walk(instance_walk, NULL) < 0) {
                /* خطأ أثناء التجول عبر المثيلات */
        }
...
        if (tracefs_instance_exists("foo")) {
                /* يوجد مثيل بالاسم foo في النظام */
        } else {
                /* لا يوجد مثيل بالاسم foo في النظام */
        }
