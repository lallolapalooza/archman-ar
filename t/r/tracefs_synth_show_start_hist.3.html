tracefs_synth_show_start_hist(3)

LIBTRACEFS(3) دليل libtracefs LIBTRACEFS(3)

الاسم

tracefs_synth_echo_cmd, tracefs_synth_get_start_hist, tracefs_synth_get_name, tracefs_synth_raw_fmt, tracefs_synth_show_event, tracefs_synth_show_start_hist, tracefs_synth_show_end_hist, tracefs_synth_get_event - استرداد بيانات الأحداث الاصطناعية.

الملخص

#include <tracefs.h>
int tracefs_synth_echo_cmd(struct trace_seq *seq, struct tracefs_synth *synth);
struct tracefs_hist *tracefs_synth_get_start_hist(struct tracefs_synth *synth);
const char *tracefs_synth_get_name(struct tracefs_synth *synth);
int tracefs_synth_raw_fmt(struct trace_seq *seq, struct tracefs_synth *synth);
const char *tracefs_synth_show_event(struct tracefs_synth *synth);
const char *tracefs_synth_show_start_hist(struct tracefs_synth *synth);
const char *tracefs_synth_show_end_hist(struct tracefs_synth *synth);
struct tep_event *tracefs_synth_get_event(struct tep_handle *tep, struct tracefs_synth *synth);

الوصف

الأحداث الاصطناعية هي أحداث ديناميكية يتم إنشاؤها عن طريق مطابقة حدثين آخرين مما يؤدي إلى تشغيل حدث اصطناعي. أحد الأحداث هو حدث البداية حيث يتم تسجيل حقل معين، وعند تنفيذ الحدث الثاني، إذا كان يحتوي على حقل (أو حقول) يطابق حقل (أو حقول) حدث البداية، فسيؤدي ذلك إلى تشغيل الحدث الاصطناعي. قد يتم تمرير قيم الحقول بخلاف الحقول المتطابقة من حدث البداية إلى حدث النهاية لإجراء عمليات حسابية عليها، أو لمجرد تمريرها كمعامل للحدث الاصطناعي.

أحد حالات الاستخدام الشائعة هو تعيين "sched_waking" كحدث بداية. يتم تشغيل هذا الحدث عند إيقاظ عملية. ثم يتم تعيين "sched_switch" كحدث نهاية. يتم تشغيل هذا الحدث عند جدولة مهمة جديدة على وحدة المعالجة المركزية. عن طريق تعيين "common_pid" لكلا الحدثين كحقول مطابقة، يتم اعتبار الوقت بين الحدثين زمن الاستيقاظ لهذه العملية. استخدم TRACEFS_TIMESTAMP كحقل لكلا الحدثين لحساب الفرق بالنانوثانية، أو استخدم TRACEFS_TIMESTAMP_USECS كحقول مقارنة لكلا الحدثين لحساب الفرق بالميكروثانية. يستخدم هذا كالمثال أدناه.

راجع tracefs_synth_alloc(3) لتخصيص الأحداث الاصطناعية، و tracefs_synth_create() لإنشاء الحدث الاصطناعي على النظام.

تعمل tracefs_synth_echo_cmd() مثل tracefs_synth_create()، ولكن بدلاً من إنشاء الحدث الاصطناعي في النظام، ستقوم بكتابة أوامر الصدى لإنشائه يدويًا في seq المحدد.

tracefs_synth_get_start_hist() تُرجع واصفًا لـ struct tracefs_hist يصف المدرج التكراري المستخدم لإنشاء الحدث الاصطناعي.

enum tracefs_synth_handler {
        TRACEFS_SYNTH_HANDLE_MATCH,
        TRACEFS_SYNTH_HANDLE_MAX,
        TRACEFS_SYNTH_HANDLE_CHANGE,
};

tracefs_synth_get_name() تُرجع اسم الحدث الاصطناعي أو NULL في حالة حدوث خطأ. تنتمي السلسلة المُرجعة إلى كائن الحدث الاصطناعي ويتم تحريرها مع الحدث بواسطة tracefs_synth_free().

tracefs_synth_raw_fmt() تكتب سلاسل التنسيق الأولية (الحدث الديناميكي والمدرجات التكرارية) للحدث الاصطناعي في seq المحدد.

tracefs_synth_show_event() تُرجع تنسيق الحدث الديناميكي المستخدم بواسطة الحدث الاصطناعي أو NULL في حالة حدوث خطأ. تنتمي السلسلة المُرجعة إلى كائن الحدث الاصطناعي ويتم تحريرها مع الحدث بواسطة tracefs_synth_free().

tracefs_synth_show_start_hist() تُرجع تنسيق المدرج التكراري البدائي المستخدم بواسطة الحدث الاصطناعي أو NULL في حالة حدوث خطأ. تنتمي السلسلة المُرجعة إلى كائن الحدث الاصطناعي ويتم تحريرها مع الحدث بواسطة tracefs_synth_free().

tracefs_synth_show_end_hist() تُرجع تنسيق المدرج التكراري النهائي المستخدم بواسطة الحدث الاصطناعي أو NULL في حالة حدوث خطأ. تنتمي السلسلة المُرجعة إلى كائن الحدث الاصطناعي ويتم تحريرها مع الحدث بواسطة tracefs_synth_free().

تُرجع الدالة tracefs_synth_get_event() حدث tep، يصف الحدث الاصطناعي المحدد. تكتشف واجهة برمجة التطبيقات أي أحداث ديناميكية تم إنشاؤها أو إزالتها حديثًا. لا يجب تحرير المؤشر المرتجع إلى حدث tep الذي يتحكم فيه @tep.

قيمة الإرجاع

tracefs_synth_get_name() و tracefs_synth_show_event() و tracefs_synth_show_start_hist() و tracefs_synth_show_end_hist() تُرجع سلسلة مملوكة لكائن الحدث الاصطناعي.

تُرجع الدالة tracefs_synth_get_event() مؤشرًا إلى حدث tep أو NULL في حالة حدوث خطأ أو إذا كان الحدث الاصطناعي المطلوب مفقودًا. لا يجب تحرير المؤشر المرتجع إلى حدث tep الذي يتحكم فيه @tep.

تُرجع جميع الدوال الأخرى صفرًا عند النجاح أو -1 عند حدوث خطأ.

الأخطاء

الأخطاء التالية تنطبق على جميع الاستدعاءات المذكورة أعلاه:

EPERM لم يتم التشغيل كمستخدم جذر عند الحاجة.

EINVAL إما أن يكون المعامل غير صالح (NULL عندما لا ينبغي أن يكون) أو حقل غير متوافق مع العمليات الحسابية.

ENODEV لم يتم العثور على حدث أو أحد حقوله.

EBADE حقول أحداث البداية والنهاية غير متوافقة إما للمطابقة أو للمقارنة.

ENOMEM لا توجد ذاكرة كافية متاحة.

وقد تحدث أخطاء أخرى من استدعاءات النظام إلى النظام.

مثال

راجع tracefs_sql(3) لاستخدام أكثر تفصيلاً لبعض هذا الكود.

#include <stdlib.h>
#include <tracefs.h>
#define start_event "sched_waking"
#define start_field "pid"
#define end_event "sched_switch"
#define end_field "next_pid"
#define match_name "pid"
static struct tracefs_synth *synth;
static void make_event(void)
{
        struct tep_handle *tep;
        /* تحميل جميع الأحداث من النظام */
        tep = tracefs_local_events(NULL);
        /* تهيئة الحدث الاصطناعي */
        synth = tracefs_synth_alloc(tep, "wakeup_lat",
                                    NULL, start_event,
                                    NULL, end_event,
                                    start_field, end_field,
                                    match_name);
        /* لم تعد هناك حاجة لـ tep */
        tep_free(tep);
        /* حفظ حقل "prio" باسم "prio" من حدث البداية */
        tracefs_synth_add_start_field(synth, "prio", NULL);
        /* حفظ حقل "next_comm" باسم "comm" من حدث النهاية */
        tracefs_synth_add_end_field(synth, "next_comm", "comm");
        /* حفظ حقل "prev_prio" باسم "prev_prio" من حدث النهاية */
        tracefs_synth_add_end_field(synth, "prev_prio", NULL);
        /*
         * أخذ فرق زمني بالميكروثانية بين النهاية والبداية
         * وتسجيله باسم "delta"
         */
        tracefs_synth_add_compare_field(synth, TRACEFS_TIMESTAMP_USECS,
                                        TRACEFS_TIMESTAMP_USECS,
                                        TRACEFS_SYNTH_DELTA_END, "delta");
        /* تسجيل فقط إذا كان حقل "prio" لحدث البداية أقل من 100 */
        tracefs_synth_append_start_filter(synth, TRACEFS_FILTER_COMPARE,
                                          "prio", TRACEFS_COMPARE_LT, "100");
        /*
         * تسجيل فقط إذا كان حقل "next_prio" لحدث النهاية أقل من 50
         * أو إذا لم يكن أولوية المهمة السابقة أكبر من أو تساوي 100.
         *   next_prio < 50 || !(prev_prio >= 100)
         */
        tracefs_synth_append_end_filter(synth, TRACEFS_FILTER_COMPARE,
                                        "next_prio", TRACEFS_COMPARE_LT, "50");
        tracefs_synth_append_end_filter(synth, TRACEFS_FILTER_OR, NULL, 0, NULL);
        tracefs_synth_append_end_filter(synth, TRACEFS_FILTER_NOT, NULL, 0, NULL);
        tracefs_synth_append_end_filter(synth, TRACEFS_FILTER_OPEN_PAREN, NULL, 0, NULL);
        tracefs_synth_append_end_filter(synth, TRACEFS_FILTER_COMPARE,
                                        "prev_prio", TRACEFS_COMPARE_GE, "100");
        /*
         * ملاحظة، ما سبق أضاف فقط: "next_prio < 50 || !(prev_prio >= 100"
         * هذا لأن، عند تنفيذ synth، سيتم إضافة القوس الختامي المتبقي.
         * أي، ستنتهي السلسلة بـ:
         * "next_prio < 50 || !(prev_prio >= 100)" عند تشغيل أحد tracefs_sync_create()
         * أو tracefs_sync_echo_cmd().
         */
}
