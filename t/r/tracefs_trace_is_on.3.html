tracefs_trace_is_on(3)

الحزمة: libtracefs

التراخيص: GPL-2.0-only, LGPL-2.1-only

LIBTRACEFS(3) دليل libtracefs LIBTRACEFS(3)

الاسم

tracefs_trace_is_on, tracefs_trace_on, tracefs_trace_off, tracefs_trace_on_get_fd, tracefs_trace_on_fd, tracefs_trace_off_fd - وظائف لتمكين أو تعطيل التتبع.

الملخص

#include <tracefs.h>
int tracefs_trace_is_on(struct tracefs_instance *instance);
int tracefs_trace_on(struct tracefs_instance *instance);
int tracefs_trace_off(struct tracefs_instance *instance);
int tracefs_trace_on_get_fd(struct tracefs_instance *instance);
int tracefs_trace_on_fd(int fd);
int tracefs_trace_off_fd(int fd);

الوصف

يمكن استخدام هذه المجموعة من الوظائف للتحقق من التتبع وتمكينه أو تعطيله للكتابة في المخزن المؤقت الحلقي في مثيل التتبع المحدد. يتم تمكين التتبع عند تمكين الكتابة في المخزن المؤقت الحلقي.

تقوم الدالة tracefs_trace_is_on() بالتحقق مما إذا كان التتبع ممكّنًا للمثيل instance المحدد. إذا كان instance فارغًا (NULL)، يتم استخدام المثيل العلوي.

تقوم الدوال tracefs_trace_on() و tracefs_trace_off() بتعيين حالة التتبع في المثيل instance لتمكينها أو تعطيلها. إذا كان instance فارغًا (NULL)، يتم استخدام المثيل العلوي.

تقوم الدالة tracefs_trace_on_get_fd() بإرجاع واصف ملف إلى ملف "tracing_on" من المثيل instance المحدد. إذا كان instance فارغًا (NULL)، يتم استخدام مثيل التتبع العلوي. يمكن استخدام الواصف المُرجع لتمكين أو تعطيل تتبع المثيل بسرعة.

تقوم الدوال tracefs_trace_on_fd() و tracefs_trace_off_fd() بتعيين حالة التتبع للتمكين أو التعطيل باستخدام واصف الملف fd المحدد. يجب فتح واصف الملف هذا للكتابة باستخدام tracefs_trace_on_get_fd(3) لمثيل التتبع المطلوب. هذه الدوال أسرع من tracefs_trace_on و tracefs_trace_off.

قيمة الإرجاع

ترجع الدالة tracefs_trace_is_on() القيمة 0 إذا كان التتبع معطلاً، 1 إذا كان ممكّنًا، أو -1 في حالة حدوث خطأ.

ترجع الدالة tracefs_trace_on_get_fd() واصف ملف إلى ملف "tracing_on" للقراءة والكتابة، والذي يجب إغلاقه باستخدام close(). في حالة حدوث خطأ، يتم إرجاع -1.

ترجع الدوال tracefs_trace_on() و tracefs_trace_off() و tracefs_trace_on_fd() و tracefs_trace_off_fd() القيمة -1 في حالة حدوث خطأ، وإلا ترجع 0.

مثال

#include <tracefs.h>
        int ret;
        ret = tracefs_trace_is_on(NULL);
        if (ret == 0) {
                /* Tracing is disabled in the top instance */
                /* التتبع معطل في المثيل العلوي */
        } else if (ret == 1) {
                /* Tracing is enabled in the top instance */
                /* التتبع ممكّن في المثيل العلوي */
        } else {
                /* Error getting tracing state of the top instance */
                /* خطأ في الحصول على حالة تتبع المثيل العلوي */
        }
        ...
        if (!tracefs_trace_on(NULL)) {
            /* Enabled tracing in the top instance */
            /* تم تمكين التتبع في المثيل العلوي */
            ...
            if (!tracefs_trace_off(NULL)) {
                /* Disabled tracing in the top instance */
                /* تم تعطيل التتبع في المثيل العلوي */
            } else {
                /* Error disabling tracing in the top instance */
                /* خطأ في تعطيل التتبع في المثيل العلوي */
            }
        } else {
                /* Error enabling tracing in the top instance */
                /* خطأ في تمكين التتبع في المثيل العلوي */
        }
        ...
        int fd = tracefs_trace_on_get_fd(NULL);
        if (fd < 0) {
                /* Error opening tracing_on file */
                /* خطأ في فتح ملف tracing_on */
        }
        ...
        if (!tracefs_trace_on_fd(fd)) {
            /* Enabled tracing in the top instance */
            /* تم تمكين التتبع في المثيل العلوي */
            ...
            if (!tracefs_trace_off_fd(fd)) {
                /* Disabled tracing in the top instance */
                /* تم تعطيل التتبع في المثيل العلوي */
            } else {
                /* Error disabling tracing in the top instance */
                /* خطأ في تعطيل التتبع في المثيل العلوي */
            }
        } else {
                /* Error enabling tracing in the top instance */
                /* خطأ في تمكين التتبع في المثيل العلوي */
        }
        ...
        close(fd);
