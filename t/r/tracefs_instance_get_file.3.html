tracefs_instance_get_file(3)

الحزمة: libtracefs

التراخيص: GPL-2.0-only, LGPL-2.1-only

LIBTRACEFS(3) دليل libtracefs LIBTRACEFS(3)

الاسم

tracefs_file_exists, tracefs_dir_exists, tracefs_instance_get_file, tracefs_instance_get_dir - التعامل مع الملفات والدلائل في مثيلات التتبع.

الملخص

#include <tracefs.h>
bool tracefs_file_exists(struct tracefs_instance *instance, char *name);
bool tracefs_dir_exists(struct tracefs_instance *instance, char *name);
char *tracefs_instance_get_file(struct tracefs_instance *instance, const char *file);
char *tracefs_instance_get_dir(struct tracefs_instance *instance);

الوصف

يمكن استخدام مجموعة واجهات برمجة التطبيقات هذه للتعامل مع ملفات التتبع في جميع مثيلات التتبع. تأخذ كل واجهة من واجهات برمجة التطبيقات هذه وسيطة instance، والتي يمكن أن تكون NULL للعمل على المثيل الأعلى. بخلاف ذلك، فإنها تعمل على مثيل تم إنشاؤه باستخدام tracefs_insance_create(3)

يقوم tracefs_file_exists() بالتحقق مما إذا كان ملف بالاسم name موجودًا في instance.

يقوم tracefs_dir_exists() بالتحقق مما إذا كان دليل بالاسم name موجودًا في instance.

يقوم tracefs_instance_get_file() بإرجاع المسار الكامل للملف بالاسم file المحدد في instance. ملاحظة، لا يتحقق مما إذا كان الملف موجودًا في المثيل.

يقوم tracefs_instance_get_dir() بإرجاع المسار الكامل للدليل بالاسم name المحدد في instance. ملاحظة، لا يتحقق مما إذا كان الدليل موجودًا في المثيل.

قيمة الإرجاع

تقوم tracefs_file_exists() و tracefs_dir_exists() بإرجاع true إذا كان الملف / الدليل موجودًا في المثيل المحدد أو false إذا لم يكن موجودًا.

تقوم tracefs_instance_get_file() و tracefs_instance_get_dir() بإرجاع سلسلة نصية أو NULL في حالة حدوث خطأ. يجب تحرير السلسلة النصية المرجعة باستخدام tracefs_put_tracing_file().

مثال

#include <tracefs.h>
struct tracefs_instance *inst = tracefs_instance_create("foo");
        if (!inst) {
                /* خطأ في إنشاء مثيل تتبع جديد */
                ...
        }
        if (tracefs_file_exists(inst,"trace_clock")) {
                /* المثيل foo يدعم ساعة التتبع */
                char *path, *clock;
                int size;
                path =  = tracefs_instance_get_file(inst, "trace_clock")
                if (!path) {
                        /* خطأ في الحصول على المسار إلى ملف trace_clock في المثيل foo */
                        ...
                }
                ...
                tracefs_put_tracing_file(path);
                clock = tracefs_instance_file_read(inst, "trace_clock", &size);
                if (!clock) {
                        /* فشل في قراءة ملف trace_clock في المثيل foo */
                        ...
                }
                ...
                free(clock);
                if (tracefs_instance_file_write(inst, "trace_clock", "global") != strlen("global")) {
                        /* فشل في تعيين ساعة التتبع العامة في المثيل foo */
                        ...
                }
        } else {
                /* المثيل foo لا يدعم ساعة التتبع */
        }
        if (tracefs_dir_exists(inst,"options")) {
                /* المثيل foo يدعم خيارات التتبع */
                char *path = tracefs_instance_get_file(inst, "options");
                if (!path) {
                        /* خطأ في الحصول على المسار إلى دليل options في المثيل foo */
                        ...
                }
                tracefs_put_tracing_file(path);
        } else {
                /* المثيل foo لا يدعم خيارات التتبع */
        }
        ...
        if (tracefs_instance_is_new(inst))
                tracefs_instance_destroy(inst);
        else
                tracefs_instance_free(inst);
        ...
        long long int res;
        if (tracefs_instance_file_read_number(NULL, "tracing_on", &res) == 0) {
                if (res == 0) {
                        /* التتبع معطل في المثيل الأعلى */
                } else if (res == 1) {
                        /* التتبع ممكّن في المثيل الأعلى */
                } else {
                        /* حالة تتبع غير معروفة للمثيل الأعلى */
                }
        } else {
                /* فشل في قراءة عدد صحيح من ملف tracing_on */
        }
        ...
        int fd;
        fd = tracefs_instance_file_open(NULL, "tracing_on", O_WRONLY);
        if (fd >= 0) {
                /* تم الحصول على واصف الملف لملف tracing_on من المثيل الأعلى للكتابة */
                ...
                close(fd);
        }
