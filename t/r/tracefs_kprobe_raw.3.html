tracefs_kprobe_raw(3)

الحزمة: libtracefs

التراخيص: GPL-2.0-only, LGPL-2.1-only

LIBTRACEFS(3) دليل libtracefs LIBTRACEFS(3)

الاسم

tracefs_kprobe_alloc, tracefs_kretprobe_alloc, tracefs_kprobe_raw, tracefs_kretprobe_raw, tracefs_kprobe_destroy - تخصيص، الحصول، إنشاء، وإزالة نقاط kprobe

الملخص

#include <tracefs.h>
struct tracefs_dynevent *
tracefs_kprobe_alloc(const char *system, const char *event,
                       const char *addr, const char *format);
struct tracefs_dynevent *
tracefs_kretprobe_alloc(const char *system, const char *event,
                          const char *addr, const char *format, unsigned int max);
int tracefs_kprobe_raw(const char *system, const char *event,
                         const char *addr, const char *format);
int tracefs_kretprobe_raw(const char *system, const char *event,
                            const char *addr, const char *format);
int tracefs_kprobe_destroy(const char *system, const char *event,
                           const char *addr, const char *format, bool force);

الوصف

تُخصِّص tracefs_kprobe_alloc() سياق kprobe جديدًا. لا يتم تكوين kprobe في النظام. يمكن إضافة kprobe إلى النظام بتمرير الواصف المُعاد إلى tracefs_dynevent_create(3). سيكون kprobe الجديد في مجموعة system (أو kprobes إذا كانت system تساوي NULL) ويحمل اسم event (أو addr إذا كانت event تساوي NULL). سيتم إدراج kprobe عند addr (اسم الدالة، مع أو بدون إزاحة، أو عنوان)، وسيحدد format تنسيق kprobe. راجع ملف وثائق Linux تحت: Documentation/trace/kprobetrace.rst

tracefs_kretprobe_alloc() هي نفس tracefs_kprobe_alloc، لكنها تخصِّص سياقًا لـ kretprobe. لديها معامل إضافي واحد، وهو اختياري، max - عدد maxactive. راجع وصف kretprobes في ملف Documentation/trace/kprobetrace.rst.

ستُنشئ tracefs_kprobe_raw() حدث kprobe. إذا كانت system تساوي NULL، فسيُستخدم "kprobes" الافتراضي للمجموعة (نظام الحدث). وإلا، إذا تم تحديد system، فسيُنشأ kprobe تحت المجموعة بهذا الاسم. event هو اسم حدث kprobe المراد إنشاؤه. يمكن أن يكون addr دالة، أو دالة وإزاحة، أو عنوانًا في النواة. هذا هو المكان الذي سيُدرج فيه موقع kprobe في النواة. format هو تنسيق kprobe كما هو محدد كـ FETCHARGS في مصدر نواة Linux في وثيقة Documentation/trace/kprobetrace.rst.

tracefs_kretprobe_raw() هي نفس tracefs_kprobe_raw()، باستثناء أنها تنشئ kretprobe بدلاً من kprobe. الفرق موصوف أيضًا في مصدر نواة Linux في ملف Documentation/trace/kprobetrace.rst.

ستدمِّر tracefs_kprobe_destroy() kprobe أو kretprobe محددًا أنشأته tracefs_kprobe_raw() أو tracefs_kretprobe_raw() بنفس المعاملات.

القيمة المُعادة

تُعيد tracefs_kprobe_raw() و tracefs_kretprobe_raw() 0 عند النجاح، أو -1 عند الخطأ. إذا حدث خطأ في التحليل في tracefs_kprobe_raw() أو tracefs_kretprobe_raw()، فيمكن استخدام tracefs_error_last(3) لاسترداد رسالة الخطأ التي تشرح مشكلة التحليل.

تُعيد واجهات tracefs_kprobe_alloc() و tracefs_kretprobe_alloc() مؤشرًا إلى بنية tracefs_dynevent المخصصة، التي تصف المسبار. يجب تحرير هذا المؤشر بواسطة tracefs_dynevent_free(3). ملاحظة: هذا يخصص فقط واصفًا يمثل kprobe. لا يعدِّل النظام قيد التشغيل.

تُعيد tracefs_kprobe_destroy() 0 عند النجاح أو -1 عند الخطأ إذا لم تتمكن من تدمير (أو العثور على) kprobe أو kretprobe بنجاح.

الأخطاء

الأخطاء التالية تخص جميع الاستدعاءات أعلاه:

EPERM لم يتم التشغيل كمستخدم جذر

ENODEV أحداث Kprobe غير مكوَّنة للنواة قيد التشغيل.

ENOMEM خطأ في تخصيص الذاكرة.

يمكن أن تفشل tracefs_kprobe_raw()، و tracefs_kretprobe_raw()، و tracefs_kprobe_alloc()، و tracefs_kretprobe_alloc() بالأخطاء التالية:

EBADMSG إذا كانت addr تساوي NULL.

EINVAL على الأرجح حدث خطأ في التحليل (استخدم tracefs_error_last(3) لرؤية ما كان ذلك الخطأ على الأرجح).

قد تحدث أخطاء أخرى أيضًا بسبب استدعاءات النظام الداخلية.

مثال

#include <stdlib.h>
#include <unistd.h>
#include <sys/wait.h>
#include <tracefs.h>
static struct tep_event *open_event;
static struct tep_format_field *file_field;
static struct tep_event *openret_event;
static struct tep_format_field *ret_field;
static int callback(struct tep_event *event, struct tep_record *record,
                    int cpu, void *data)
{
        struct trace_seq seq;
        trace_seq_init(&seq);
        tep_print_event(event->tep, &seq, record, "%d-%s: ", TEP_PRINT_PID, TEP_PRINT_COMM);
        if (event->id == open_event->id) {
                trace_seq_puts(&seq, "open file='");
                tep_print_field(&seq, record->data, file_field);
                trace_seq_puts(&seq, "'\n");
        } else if (event->id == openret_event->id) {
                unsigned long long ret;
                tep_read_number_field(ret_field, record->data, &ret);
                trace_seq_printf(&seq, "open ret=%lld\n", ret);
        } else {
                goto out;
        }
        trace_seq_terminate(&seq);
        trace_seq_do_printf(&seq);
out:
        trace_seq_destroy(&seq);
        return 0;
}
static pid_t run_exec(char **argv, char **env)
{
        pid_t pid;
        pid = fork();
        if (pid)
                return pid;
        execve(argv[0], argv, env);
        perror("exec");
        exit(-1);
}
const char *mykprobe = "my_kprobes";
enum kprobe_type {
        KPROBE,
        KRETPROBE,
};
static void __kprobe_create(enum kprobe_type type, const char *event,
                            const char *addr, const char *fmt)
{
        char *err;
        int r;
        if (type == KPROBE)
                r = tracefs_kprobe_raw(mykprobe, event, addr, fmt);
        else
                r = tracefs_kretprobe_raw(mykprobe, event, addr, fmt);
        if (r < 0) {
                err = tracefs_error_last(NULL);
                perror("Failed to create kprobe:");
                if (err && strlen(err))
                        fprintf(stderr, "%s\n", err);
        }
}
static void kprobe_create(const char *event, const char *addr,
                          const char *fmt)
{
        __kprobe_create(KPROBE, event, addr, fmt);
}
static void kretprobe_create(const char *event, const char *addr,
                             const char *fmt)
{
        __kprobe_create(KRETPROBE, event, addr, fmt);
}
int main (int argc, char **argv, char **env)
{
        struct tracefs_instance *instance;
        struct tep_handle *tep;
        const char *sysnames[] = { mykprobe, NULL };
        pid_t pid;
        if (argc < 2) {
                printf("usage: %s command\n", argv[0]);
                exit(-1);
        }
        instance = tracefs_instance_create("exec_open");
        if (!instance) {
                perror("creating instance");
                exit(-1);
        }
        tracefs_dynevent_destroy_all(TRACEFS_DYNEVENT_KPROBE | TRACEFS_DYNEVENT_KRETPROBE, true);
        kprobe_create("open", "do_sys_openat2",
                      "file=+0($arg2):ustring flags=+0($arg3):x64 mode=+8($arg3):x64\n");
        kretprobe_create("openret", "do_sys_openat2", "ret=%ax");
        tep = tracefs_local_events_system(NULL, sysnames);
        if (!tep) {
                perror("reading events");
                exit(-1);
        }
        open_event = tep_find_event_by_name(tep, mykprobe, "open");
        file_field = tep_find_field(open_event, "file");
        openret_event = tep_find_event_by_name(tep, mykprobe, "openret");
        ret_field = tep_find_field(openret_event, "ret");
        tracefs_event_enable(instance, mykprobe, NULL);
        pid = run_exec(&argv[1], env);
        /* Let the child start to run */
        sched_yield();
        do {
                tracefs_load_cmdlines(NULL, tep);
                tracefs_iterate_raw_events(tep, instance, NULL, 0, callback, NULL);
        } while (waitpid(pid, NULL, WNOHANG) != pid);
        /* سيعطّل الأحداث */
        tracefs_dynevent_destroy_all(TRACEFS_DYNEVENT_KPROBE | TRACEFS_DYNEVENT_KRETPROBE, true);
        tracefs_instance_destroy(instance);
        tep_free(tep);
        return 0;
}
