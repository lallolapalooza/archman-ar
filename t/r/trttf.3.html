توثيق الوظيفة

subroutine ctrttf (character transr, character uplo, integer n, complex, dimension( 0: lda-1, 0: * ) a, integer lda, complex, dimension( 0: * ) arf, integer info)

CTRTTF تنسخ مصفوفة مثلثية من التنسيق الكامل القياسي (TR) إلى التنسيق المعبأ الكامل المستطيل (TF).

الغرض:

!>
!> CTRTTF تنسخ مصفوفة مثلثية A من التنسيق الكامل القياسي (TR)
!> إلى التنسيق المعبأ الكامل المستطيل (TF) .
!>

المعلمات

TRANSR 
!>          TRANSR هو CHARACTER*1
!>          = 'N':  ARF في وضع عادي مطلوب؛
!>          = 'C':  ARF في وضع منقول مترافق مطلوب؛
!>

UPLO

!>          UPLO هو CHARACTER*1
!>          = 'U':  A هي مصفوفة مثلثية عليا؛
!>          = 'L':  A هي مصفوفة مثلثية سفلى.
!>

N

!>          N هو INTEGER
!>          رتبة المصفوفة A.  N >= 0.
!>

A

!>          A هي مصفوفة COMPLEX، البعد ( LDA, N )
!>          عند الإدخال، تحتوي المصفوفة المثلثية A.  إذا كان UPLO = 'U'، فإن
!>          الجزء العلوي المثلثي الرئيسي N-by-N من المصفوفة A يحتوي على
!>          المصفوفة المثلثية العليا، والجزء المثلثي السفلي الصارم
!>          من A لا يتم الرجوع إليه.  إذا كان UPLO = 'L'، فإن
!>          الجزء السفلي المثلثي الرئيسي N-by-N من المصفوفة A يحتوي على
!>          المصفوفة المثلثية السفلى، والجزء المثلثي العلوي الصارم
!>          من A لا يتم الرجوع إليه.
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرئيسي للمصفوفة A.  LDA >= max(1,N).
!>

ARF

!>          ARF هي مصفوفة COMPLEX، البعد ( N*(N+1)/2 )،
!>          عند الخروج، يتم تخزين المصفوفة المثلثية العليا أو السفلى A في
!>          تنسيق RFP. لمزيد من المناقشة، انظر الملاحظات أدناه.
!>

INFO

!>          INFO هو INTEGER
!>          = 0:  الانتهاء بنجاح
!>          < 0:  إذا كان INFO = -i، فإن الوسيط i كان له قيمة غير قانونية
!>

!>  نعطي مثالًا حيث N = 6.
!>
!>      AP مصفوفة عليا             AP مصفوفة سفلى
!>
!>   00 01 02 03 04 05       00
!>      11 12 13 14 15       10 11
!>         22 23 24 25       20 21 22
!>            33 34 35       30 31 32 33
!>               44 45       40 41 42 43 44
!>                  55       50 51 52 53 54 55
!>
!>
!>  لنفترض TRANSR = 'N'. يحتفظ RFP بـ AP كما يلي:
!>  بالنسبة لـ UPLO = 'U'، يتكون شبه المنحرف العلوي A(0:5,0:2) من الأعمدة الثلاثة الأخيرة
!>  من AP العلوي. يتكون المثلث السفلي A(4:6,0:2) من
!>  منقول مترافق للأعمدة الثلاثة الأولى من AP العلوي.
!>  بالنسبة لـ UPLO = 'L'، يتكون شبه المنحرف السفلي A(1:6,0:2) من الأعمدة الثلاثة الأولى
!>  من AP السفلي. يتكون المثلث العلوي A(0:2,0:2) من
!>  منقول مترافق للأعمدة الثلاثة الأخيرة من AP السفلي.
!>  للإشارة إلى الترافق، نضع -- فوق العنصر. هذا يغطي
!>  الحالة N زوجي و TRANSR = 'N'.
!>
!>         RFP A                   RFP A
!>
!>                                -- -- --
!>        03 04 05                33 43 53
!>                                   -- --
!>        13 14 15                00 44 54
!>                                      --
!>        23 24 25                10 11 55
!>
!>        33 34 35                20 21 22
!>        --
!>        00 44 45                30 31 32
!>        -- --
!>        01 11 55                40 41 42
!>        -- -- --
!>        02 12 22                50 51 52
!>
!>  الآن، لنفترض TRANSR = 'C'. RFP A في كلتا حالتي UPLO هو مجرد
!>  منقول مترافق لـ RFP A أعلاه. لذلك نحصل على:
!>
!>
!>           RFP A                   RFP A
!>
!>     -- -- -- --                -- -- -- -- -- --
!>     03 13 23 33 00 01 02    33 00 10 20 30 40 50

!>  نعطي مثالًا حيث N = 5.
!>
!>     AP مصفوفة عليا             AP مصفوفة سفلى
!>
!>   00 01 02 03 04              00
!>      11 12 13 14              10 11
!>         22 23 24              20 21 22
!>            33 34              30 31 32 33
!>               44              40 41 42 43 44
!>
!>
!>  لنفترض TRANSR = 'N'. يحتفظ RFP بـ AP كما يلي:
!>  بالنسبة لـ UPLO = 'U'، يتكون شبه المنحرف العلوي A(0:4,0:2) من الأعمدة الثلاثة الأخيرة
!>  من AP العلوي. يتكون المثلث السفلي A(3:4,0:1) من
!>  منقول مترافق للأعمدة الاثنين الأوليين من AP العلوي.
!>  بالنسبة لـ UPLO = 'L'، يتكون شبه المنحرف السفلي A(0:4,0:2) من الأعمدة الثلاثة الأولى
!>  من AP السفلي. يتكون المثلث العلوي A(0:1,1:2) من
!>  منقول مترافق للأعمدة الاثنين الأخيرين من AP السفلي.
!>  للإشارة إلى الترافق، نضع -- فوق العنصر. هذا يغطي
!>  الحالة N فردي و TRANSR = 'N'.
