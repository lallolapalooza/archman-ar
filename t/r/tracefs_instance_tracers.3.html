tracefs_instance_tracers(3)

الحزمة: libtracefs

التراخيص: GPL-2.0-only, LGPL-2.1-only

LIBTRACEFS(3) دليل libtracefs LIBTRACEFS(3)

الاسم

tracefs_instance_tracers, tracefs_tracer_set, tracefs_tracer_clear - تمكين أو تعطيل متعقب في مثيل أو المستوى الأعلى

الملخص

#include <tracefs.h>
char ** tracefs_instance_tracers(struct tracefs_instance *instance);
int tracefs_tracer_set(struct tracefs_instance *instance, enum tracefs_tracers tracer);
int tracefs_tracer_set(struct tracefs_instance *instance, enum tracefs_tracers tracer, const char *name);
int tracefs_tracer_clear(struct tracefs_instance *instance);

الوصف

سيقوم tracefs_instance_tracers بإرجاع قائمة بالمتعقبات المتاحة لمثيل instance معين (لاحظ أن المثيل قد لا يحتوي على نفس مجموعة المتعقبات المتاحة مثل المستوى الأعلى). إذا كان instance فارغًا (NULL)، فإن القائمة المرجعة للمتعقبات المتاحة ستكون للمستوى الأعلى.

يقوم tracefs_tracer_set بتمكين متعقب في المثيل المحدد، المحدد بواسطة المعلمة instance. إذا كان instance فارغًا (NULL)، فسيتم تغيير المثيل الأعلى. إذا تم تعيين tracer إلى TRACFES_TRACER_CUSTOM، فيجب تمرير سلسلة name كمعلمة ثالثة، ويتم كتابة ذلك في المثيل لتمكين المتعقب بهذا الاسم. هذا مفيد للنوى الأحدث أو المخصصة التي تحتوي على متعقبات غير محددة بعد في تعداد tracefs_tracers.

يقوم tracefs_tracer_clear بتعطيل المتعقب للمثيل المحدد بواسطة المتغير instance، أو المثيل الأعلى إذا كان فارغًا (NULL). هذا يعادل استدعاء tracefs_tracer_set مع TRACEFS_TRACER_NOP كمعلمة tracer.

تعدادات TRACEFS_TRACER

التعدادات المعرفة حاليًا والتي يتم قبولها هي:

TRACEFS_TRACER_NOP: هذا هو المتعقب الخامل، والذي لا يفعل شيئًا ويستخدم لمسح أي متعقب نشط.

TRACEFS_TRACER_FUNCTION: يمكّن معظم الوظائف في النواة من التتبع.

TRACEFS_TRACER_FUNCTION_GRAPH: يمكّن معظم الوظائف في النواة من التتبع بالإضافة إلى عودة الوظيفة.

TRACEFS_TRACER_IRQSOFF: يتتبع زمن الكمون عند تعطيل المقاطعات.

TRACEFS_TRACER_PREEMPTOFF: يتتبع زمن الكمون عند تعطيل الاستباق (الوقت في النواة الذي لا يمكن فيه جدولة المهام من وحدة المعالجة المركزية).

TRACEFS_TRACER_PREEMPTIRQSOFF: يتتبع إجمالي زمن الكمون المدمج لتعطيل المقاطعات وكذلك تعطيل الاستباق.

TRACEFS_TRACER_WAKEUP: يتتبع زمن الكمون لأعلى مهمة ذات أولوية للاستيقاظ.

TRACEFS_TRACER_WAKEUP_RT: يتتبع زمن الكمون لأعلى مهمة في الوقت الفعلي للاستيقاظ. يتم تجاهل جميع المهام الأخرى.

TRACEFS_TRACER_WAKEUP_DL: يتتبع زمن الكمون لأعلى مهمة DEADLINE للاستيقاظ. يتم تجاهل جميع المهام الأخرى.

TRACEFS_TRACER_MMIOTRACE: يتتبع تفاعل الأجهزة مع النواة.

TRACEFS_TRACER_HWLAT: يكتشف زمن الكمون الناجم عن الأجهزة الخارجة عن نطاق النواة.

TRACEFS_TRACER_BRANCH: يتتبع متى تم اتخاذ فروع احتمالية أو غير احتمالية.

TRACEFS_TRACER_BLOCK: متعقب خاص لأجهزة الكتلة.

لاحظ أن المتعقبات المذكورة أعلاه قد لا تكون متاحة في النواة وسيُرجع tracefs_tracer_set() خطأ مع تعيين errno إلى ENODEV، إذا كانت النواة لا تدعم خيار tracer، أو الخيار المخصص إذا تم استخدام TRACEFS_TRACER_CUSTOM.

قيمة الإرجاع

يعيد 0 عند النجاح، أو -1 عند حدوث خطأ.

الأخطاء

يمكن أن يفشل tracefs_tracer_set() مع الأخطاء التالية:

EINVAL: معلمة tracer خارج نطاق ما هو محدد.

ENOMEM: خطأ في تخصيص الذاكرة.

ENOENT: لا تدعم النواة قيد التشغيل المتعقبات.

ENODEV: المتعقب المحدد غير مدعوم في النواة قيد التشغيل.

قد تحدث أخطاء أخرى بسبب استدعاءات النظام الداخلية.

مثال

#include <stdlib.h>
#include <stdio.h>
#include <getopt.h>
#include <errno.h>
#include <tracefs.h>
int main(int argc, char *argv[])
{
        struct tracefs_instance *inst = NULL;
        enum tracefs_tracers t = TRACEFS_TRACER_NOP;
        const char *cust = NULL;
        const char *buf = NULL;
        char **tracers;
        int ret;
        int ch;
        int i;
        while ((ch = getopt(argc, argv, "nfgiwdc:B:")) > 0) {
                switch (ch) {
                case 'f': t = TRACEFS_TRACER_FUNCTION; break;
                case 'g': t = TRACEFS_TRACER_FUNCTION_GRAPH; break;
                case 'i': t = TRACEFS_TRACER_PREEMPTIRQSOFF; break;
                case 'w': t = TRACEFS_TRACER_WAKEUP_RT; break;
                case 'd': t = TRACEFS_TRACER_WAKEUP_DL; break;
                case 'c':
                        t = TRACEFS_TRACER_CUSTOM;
                        cust = optarg;
                        break;
                case 'B':
                        buf = optarg;
                        break;
                case 'n':
                        /* nop */
                        break;
                default:
                        printf("Unknow arg %c\n", ch);
                        exit(-1);
                }
        }
        if (buf) {
                inst = tracefs_instance_create(buf);
                if (!inst) {
                        printf("failed to create instance\n");
                        exit(-1);
                }
        }
        if (t == TRACEFS_TRACER_CUSTOM)
                ret = tracefs_tracer_set(inst, t, cust);
        else
                ret = tracefs_tracer_set(inst, t);
        if (ret < 0) {
                if (errno == ENODEV) {
                        if (cust)
                                printf("Tracer '%s' not supported by kernel\n", cust);
                        else
                                printf("Tracer not supported by kernel\n");
                        tracers = tracefs_instance_tracers(inst);
                        printf("Available tracers:");
                        for (i = 0; tracers && tracers[i]; i++)
                                printf(" %s", tracers[i]);
                        tracefs_list_free(tracers);
                        printf("\n");
                } else
                        perror("Error");
                if (inst) {
                        tracefs_instance_destroy(inst);
                        tracefs_instance_free(inst);
                }
                exit(-1);
        }
        tracefs_instance_free(inst);
        exit(0);
}
