tracefs_instance_get_affinity(3)

الحزمة: libtracefs

التراخيص: GPL-2.0-only, LGPL-2.1-only

LIBTRACEFS(3) دليل libtracefs LIBTRACEFS(3)

الاسم

tracefs_instance_set_affinity, tracefs_instance_set_affinity_set, tracefs_instance_set_affinity_raw, tracefs_instance_get_affinity, tracefs_instance_get_affinity_set, tracefs_instance_get_affinity_raw - تعيين أو استرداد التقارب لمثيل أو المستوى العلوي للمعالجات التي تمكن التتبع.

المُلَخَّص

#include <tracefs.h>
int tracefs_instance_set_affinity(struct tracefs_instance *instance, const char *cpu_str);
int tracefs_instance_set_affinity_set(struct tracefs_instance *instance, cpu_set_t *set, size_t set_size);
int tracefs_instance_set_affinity_raw(struct tracefs_instance *instance, const char *mask);
char *tracefs_instance_get_affinity(struct tracefs_instance *instance);
int tracefs_instance_get_affinity_set(struct tracefs_instance *instance, cpu_set_t *set, size_t set_size);
char *tracefs_instance_get_affinity_raw(struct tracefs_instance *instance);

الوصف

تقوم هذه الدوال بتعيين أو استرداد تقارب وحدة المعالجة المركزية (CPU affinity) الذي يحد من وحدات المعالجة المركزية التي سيتم تمكين التتبع لها لمثيل معين معرف بواسطة المعلمة instance. إذا كان instance يساوي NULL، فسيتم التأثير على المثيل العلوي.

تأخذ الدالة tracefs_instance_set_affinity() سلسلة نصية cpu_str وهي قائمة بوحدات المعالجة المركزية لتعيين التقارب لها. إذا كانت cpu_str تساوي NULL، فسيتم تعيين جميع وحدات المعالجة المركزية في النظام. تنسيق cpu_str هو سلسلة نصية مفصولة بفواصل من أرقام عشرية بدون مسافات. يمكن تحديد نطاق بواسطة شرطة.

مثال: "1,4,6-8"

لا يلزم أن تكون الأرقام مرتبة باستثناء النطاقات، حيث يجب أن يكون الرقم الثاني مساويًا أو أكبر من الأول.

تأخذ الدالة tracefs_instance_set_affinity_set() مجموعة معالجات مركزية محددة بواسطة CPU_SET(3). حجم المجموعة المحدد بواسطة set_size هو حجم set بالبايت. إذا كانت set تساوي NULL، فسيتم تعيين جميع وحدات المعالجة المركزية في النظام، ويتم تجاهل set_size.

تأخذ الدالة tracefs_instance_set_affinity_raw() سلسلة نصية تحتوي على قناع بت (bitmask) سداسي عشري، حيث يتم فصل كل 32 بت بفواصل. بالنسبة لجهاز يحتوي على أكثر من 32 وحدة معالجة مركزية، لتعيين وحدات المعالجة المركزية 1-10 ووحدة المعالجة المركزية 40:

"100,000007fe"

حيث أن ما سبق هو تمثيل سداسي عشري للبتات 1-10 والبت 40.

ستقوم الدالة tracefs_instance_get_affinity() باسترداد التقارب في شكل قابل للقراءة البشرية.

مثال: "1,4,6-8"

يجب تحرير السلسلة النصية المرجعة باستخدام free(3).

ستقوم الدالة tracefs_instance_get_affinity_set() بتعيين جميع البتات في مجموعة المعالجات المركزية المقدمة (من CPU_SET(3)). لاحظ أنها لن تمسح أي بتات تم تعيينها بالفعل في المجموعة ولكن وحدات المعالجة المركزية ليست كذلك. إذا كان يجب تعيين البتات فقط لوحدات المعالجة المركزية التي تم تمكينها، فيجب إجراء CPU_ZERO_S() على المجموعة قبل استدعاء هذه الدالة.

ستقوم الدالة tracefs_instance_get_affinity_raw() ببساطة بقراءة tracefs_instance_cpumask وإرجاع تلك السلسلة النصية. يجب تحرير السلسلة النصية المرجعة باستخدام free(3).

قيمة الإرجاع

ترجع جميع دوال التعيين 0 عند النجاح و -1 عند الخطأ.

ترجع الدالتان tracefs_instance_get_affinity() و tracefs_instance_get_affinity_raw() سلسلة نصية مخصصة يجب تحريرها باستخدام free(3)، أو NULL عند الخطأ.

ترجع الدالة tracefs_instance_get_affinity_set() عدد وحدات المعالجة المركزية التي تم تعيينها، أو -1 عند الخطأ.

الأخطاء

الأخطاء التالية تنطبق على جميع الاستدعاءات المذكورة أعلاه:

EFBIG إذا تم تعيين وحدة معالجة مركزية أكبر من تلك الموجودة في النظام.

EINVAL كان أحد المعلمات غير صالح.

الأخطاء التالية خاصة بالدالتين tracefs_instance_set_affinity() و tracefs_instance_set_affinity_set():

ENOMEM خطأ في تخصيص الذاكرة.

ENODEV لم يتم تكوين الأحداث الديناميكية من النوع المطلوب لنواة التشغيل.

الأخطاء التالية خاصة بالدالة tracefs_instance_set_affinity() فقط

EACCES تم تعديل cpu_str بواسطة خيط آخر عند معالجتها.

مثال

#include <sched.h>
#include <stdio.h>
#include <stdlib.h>
#include <tracefs.h>
int main (int argc, char **argv)
{
        struct trace_seq seq;
        cpu_set_t *set;
        size_t set_size;
        char *c;
        int cpu1;
        int cpu2;
        int i;
        c = tracefs_instance_get_affinity(NULL);
        printf("The affinity was %s\n", c);
        free(c);
        if (argc < 2) {
                tracefs_instance_set_affinity(NULL, NULL);
                exit(0);
        }
        /* Show example using a set */
        if (argc == 2 && !strchr(argv[1],',')) {
                cpu1 = atoi(argv[1]);
                c = strchr(argv[1], '-');
                if (c++)
                        cpu2 = atoi(c);
                else
                        cpu2 = cpu1;
                if (cpu2 < cpu1) {
                        fprintf(stderr, "Invalid CPU range\n");
                        exit(-1);
                }
                set = CPU_ALLOC(cpu2 + 1);
                set_size = CPU_ALLOC_SIZE(cpu2 + 1);
                CPU_ZERO_S(set_size, set);
                for ( ; cpu1 <= cpu2; cpu1++)
                        CPU_SET(cpu1, set);
                tracefs_instance_set_affinity_set(NULL, set, set_size);
                CPU_FREE(set);
                exit(0);
        }
        trace_seq_init(&seq);
        for (i = 1; i < argc; i++) {
                if (i > 1)
                        trace_seq_putc(&seq, ',');
                trace_seq_puts(&seq, argv[i]);
        }
        trace_seq_terminate(&seq);
        tracefs_instance_set_affinity(NULL, seq.buffer);
        trace_seq_destroy(&seq);
        exit(0);
        return 0;
}
