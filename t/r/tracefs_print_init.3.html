tracefs_print_init(3)

الحزمة: libtracefs

التراخيص: GPL-2.0-only, LGPL-2.1-only

LIBTRACEFS(3) دليل libtracefs LIBTRACEFS(3)

الاسم

tracefs_print_init, tracefs_print_close, tracefs_printf, tracefs_vprintf - فتح، إغلاق وكتابة سلاسل نصية منسقة في مخزن التتبع.

الملخص

#include <tracefs.h>
int tracefs_print_init(struct tracefs_instance *instance);
int tracefs_printf(struct tracefs_instance *instance, const char *fmt, ...);
int tracefs_vprintf(struct tracefs_instance *instance, const char *fmt, va_list ap);
void tracefs_print_close(struct tracefs_instance *instance);

الوصف

مجموعة من الوظائف لكتابة سلاسل نصية منسقة في مخزن التتبع. راجع Documentation/trace/ftrace.rst من شجرة نواة لينكس لمزيد من المعلومات حول كتابة البيانات من مساحة المستخدم في مخزن التتبع. كل هذه الواجهات لها instance كوسيط أول. إذا تم تمرير NULL كوسيط instance، فسيتم استخدام أعلى مثيل تتبع.

تقوم وظيفة tracefs_print_init() بتهيئة المكتبة للكتابة في مخزن التتبع للمثيل المحدد instance. لا يلزم استدعاء هذه الواجهة قبل كتابة السلاسل النصية، فستقوم أي من واجهات printf باستدعائها تلقائيًا، إذا لم تكن المكتبة مهيأة بعد. ولكن استدعاء tracefs_print_init() مسبقًا سيسرع عملية الكتابة.

تقوم وظيفة tracefs_printf() بكتابة سلسلة نصية منسقة في مخزن التتبع للمثيل المحدد instance. الوسيط fmt هو سلسلة نصية بتنسيق printf، متبوعة بوسائط متغيرة ....

تقوم وظيفة tracefs_vprintf() بكتابة سلسلة نصية منسقة في مخزن التتبع للمثيل المحدد instance. الوسيط fmt هو سلسلة نصية بتنسيق printf، متبوعة بقائمة الوسائط ap.

تقوم وظيفة tracefs_print_close() بإغلاق الموارد، المستخدمة من قبل المكتبة للكتابة في مخزن التتبع للمثيل المحدد.

قيمة الإرجاع

ترجع وظائف tracefs_print_init() و tracefs_printf() و tracefs_vprintf() القيمة 0 إذا نجحت العملية، أو -1 في حالة حدوث خطأ.

مثال

#include <tracefs.h>
if (tracefs_print_init(NULL) < 0) {
 /* فشلت تهيئة المكتبة للكتابة في مخزن التتبع لأعلى مثيل تتبع */
}
void foo_print(char *format, ...)
{
        va_list ap;
        va_start(ap, format);
        if (tracefs_vprintf(NULL, format, ap) < 0) {
                /* فشلت الطباعة في مخزن التتبع */
        }
        va_end(ap);
}
void foo_print_string(char *message)
{
        if (tracefs_printf(NULL, "رسالة من مساحة المستخدم: %s", message) < 0) {
                /* فشلت الطباعة في مخزن التتبع */
        }
}
tracefs_print_close();
