tracefs_cpu_buffered_read_buf(3)

الحزمة: libtracefs

التراخيص: GPL-2.0-only, LGPL-2.1-only

LIBTRACEFS(3) دليل libtracefs LIBTRACEFS(3)

الاسم

tracefs_cpu_read_buf, tracefs_cpu_buffered_read_buf, tracefs_cpu_flush_buf - قراءة بيانات trace_pipe_raw وإرجاع kbuffer

الملخص

#include <tracefs.h>
struct kbuffer *tracefs_cpu_read_buf(struct tracefs_cpu *tcpu, bool nonblock);
struct kbuffer *tracefs_cpu_buffered_read_buf(struct tracefs_cpu *tcpu, bool nonblock);
struct kbuffer *tracefs_cpu_flush_buf(struct tracefs_cpu *tcpu);

الوصف

يمكن استخدام مجموعة الواجهات هذه لقراءة البيانات الخام من ملفات trace_pipe_raw في نظام ملفات tracefs وإرجاع بنية kbuffer لقراءتها بها.

تقوم tracefs_cpu_read_buf() بقراءة ملفات trace_pipe_raw المرتبطة بـ tcpu وإرجاع بنية kbuffer يمكن استخدامها للتكرار على الأحداث. إذا تم تعيين nonblock، ولم تتوفر بيانات، فستعود فورًا. بخلاف ذلك، اعتمادًا على كيفية فتح tcpu، ستتوقف. إذا تم فتح tcpu مع تعيين nonblock، فلن يكون لـ nonblock أي فرق.

tracefs_cpu_buffered_read_buf() هي في الأساس نفس tracefs_cpu_read_buf() باستثناء أنها تستخدم أنبوبًا من خلال splice لتخزين القراءات مؤقتًا. سيؤدي هذا إلى تجميع القراءات مع الحفاظ على القراءة من المخزن المؤقت الدائري أقل تطفلاً على النظام، حيث أن القراءة طوال الوقت يمكن أن تسبب اضطرابًا كبيرًا. لاحظ، أحد الاختلافات بين هذا و tracefs_cpu_read() هو أنه سيقرأ فقط في صفحات المخزن المؤقت الفرعي. إذا لم يملأ المخزن المؤقت الدائري صفحة، فلن يعود بأي شيء، حتى مع تعيين nonblock. يجب استدعاء tracefs_cpu_flush_buf() أو tracefs_cpu_flush() لقراءة بقية الملف في نهاية التتبع.

تقوم tracefs_cpu_flush_buf() بقراءة ملف trace_pipe_raw المرتبط بـ tcpu ووضعه في buffer، والذي يجب أن يكون بحجم المخزن المؤقت الفرعي الذي تم استرداده. يجب استدعاء هذا في نهاية التتبع للحصول على بقية البيانات. سيقوم هذا الاستدعاء بتحويل واصف الملف لـ trace_pipe_raw إلى وضع عدم الحظر.

قيمة الإرجاع

ترجع الدوال tracefs_cpu_read_buf() و tracefs_cpu_buffered_read_buf() و tracefs_cpu_flush() واصف kbuffer يمكن التكرار عليه للعثور على الأحداث. لاحظ، هذا الواصف جزء من بنية tracefs_cpu ولا ينبغي تحريره. سيتم تحريره. يرجع NULL عند الخطأ أو إذا تم تعيين nonblock ولا توجد أحداث متاحة. في حالة عدم وجود أحداث، سيتم تعيين errno مع EAGAIN.

مثال

#include <stdlib.h>
#include <ctype.h>
#include <tracefs.h>
static void read_page(struct tep_handle *tep, struct kbuffer *kbuf)
{
        static struct trace_seq seq;
        struct tep_record record;
        if (seq.buffer)
                trace_seq_reset(&seq);
        else
                trace_seq_init(&seq);
        while ((record.data = kbuffer_read_event(kbuf, &record.ts))) {
                record.size = kbuffer_event_size(kbuf);
                kbuffer_next_event(kbuf, NULL);
                tep_print_event(tep, &seq, &record,
                                "%s-%d %9d\t%s: %s\n",
                                TEP_PRINT_COMM,
                                TEP_PRINT_PID,
                                TEP_PRINT_TIME,
                                TEP_PRINT_NAME,
                                TEP_PRINT_INFO);
                trace_seq_do_printf(&seq);
                trace_seq_reset(&seq);
        }
}
int main (int argc, char **argv)
{
        struct tracefs_cpu *tcpu;
        struct tep_handle *tep;
        struct kbuffer *kbuf;
        int cpu;
        if (argc < 2 || !isdigit(argv[1][0])) {
                printf("usage: %s cpu\n\n", argv[0]);
                exit(-1);
        }
        cpu = atoi(argv[1]);
        tep = tracefs_local_events(NULL);
        if (!tep) {
                perror("Reading trace event formats");
                exit(-1);
        }
        tcpu = tracefs_cpu_open(NULL, cpu, 0);
        if (!tcpu) {
                perror("Open CPU 0 file");
                exit(-1);
        }
        while ((kbuf = tracefs_cpu_buffered_read_buf(tcpu, true))) {
                read_page(tep, kbuf);
        }
        kbuf = tracefs_cpu_flush_buf(tcpu);
        if (kbuf)
                read_page(tep, kbuf);
        tracefs_cpu_close(tcpu);
        tep_free(tep);
        return 0;
}
