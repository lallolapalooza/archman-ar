tracefs_instance_file_clear(3)

الحزمة: libtracefs

التراخيص: GPL-2.0-only, LGPL-2.1-only

LIBTRACEFS(3) دليل libtracefs LIBTRACEFS(3)

الاسم

tracefs_instance_file_open, tracefs_instance_file_write, tracefs_instance_file_write_number, tracefs_instance_file_append, tracefs_instance_file_clear, tracefs_instance_file_read, tracefs_instance_file_read_number - التعامل مع الملفات في مثيلات التتبع.

الملخص

#include <tracefs.h>
int tracefs_instance_file_open(struct tracefs_instance *instance, const char *file, int mode);
int tracefs_instance_file_write(struct tracefs_instance *instance, const char *file, const char *str);
int tracefs_instance_file_write_number(struct tracefs_instance *instance, const char *file, size_t val);
int tracefs_instance_file_append(struct tracefs_instance *instance, const char *file, const char *str);
int tracefs_instance_file_clear(struct tracefs_instance *instance, const char *file);
char *tracefs_instance_file_read(struct tracefs_instance *instance, const char *file, int *psize);
int tracefs_instance_file_read_number(struct tracefs_instance *instance, const char *file, long long int *res);

الوصف

يمكن استخدام مجموعة واجهات برمجة التطبيقات هذه للعمل مع ملفات التتبع في جميع مثيلات التتبع. كل من واجهات برمجة التطبيقات هذه تأخذ وسيطة instance، والتي يمكن أن تكون NULL للعمل على المثيل الأعلى. بخلاف ذلك، فإنه يعمل على مثيل تم إنشاؤه باستخدام tracefs_insance_create(3).

تقوم الدالة tracefs_instance_file_open() بفتح ملف file التتبع من المثيل instance المحدد وتعيد واصف ملف له. يمكن تحديد وضع الوصول إلى الملف mode، راجع open(3) لمزيد من التفاصيل. إذا تم تمرير -1 كـ mode، يتم استخدام O_RDWR الافتراضي.

تقوم الدالة tracefs_instance_file_write() بكتابة سلسلة str في ملف file من المثيل instance المحدد، بدون حرف NULL الختامي. عند فتح الملف، تحاول هذه الدالة اقتطاع حجم الملف إلى الصفر، مما يؤدي إلى مسح جميع الإعدادات الموجودة مسبقًا.

تقوم الدالة tracefs_instance_file_write_number() بتحويل val إلى سلسلة ثم كتابتها إلى الملف المحدد. هذه دالة مساعدة تقوم بتحويل الرقم إلى سلسلة ثم تستدعي tracefs_instance_file_write().

تقوم الدالة tracefs_instance_file_append() بكتابة سلسلة str في ملف file من المثيل instance المحدد، بدون حرف NULL الختامي. هذه الدالة مشابهة لـ tracefs_instance_file_write()، ولكن لا يتم مسح المحتوى الحالي للملف. وبالتالي، تتم إضافة الإعدادات الجديدة إلى الإعدادات الموجودة (إن وجدت).

تحاول الدالة tracefs_instance_file_clear() اقتطاع حجم الملف إلى الصفر، مما يؤدي إلى مسح جميع الإعدادات الموجودة مسبقًا. إذا كان للملف محتوى لا يتم مسحه بهذه الطريقة، فلن يكون لهذا أي تأثير.

تقوم الدالة tracefs_instance_file_read() بقراءة محتوى ملف file من المثيل instance المحدد.

تقوم الدالة tracefs_instance_file_read_number() بقراءة محتوى ملف file من المثيل instance المحدد وتحويله إلى عدد صحيح long long، والذي يتم تخزينه في res.

قيمة الإرجاع

تعيد الدالة tracefs_instance_file_open() واصف ملف للملف المفتوح. يجب إغلاقه باستخدام close(3). في حالة حدوث خطأ، يتم إرجاع -1.

تعيد الدالة tracefs_instance_file_write() عدد البايتات المكتوبة، أو -1 في حالة حدوث خطأ.

تعيد الدالة tracefs_instance_file_write_number() القيمة 0 عند النجاح و -1 عند حدوث خطأ.

تعيد الدالة tracefs_instance_file_append() عدد البايتات المكتوبة، أو -1 في حالة حدوث خطأ.

تعيد الدالة tracefs_instance_file_clear() القيمة 0 عند النجاح، أو -1 في حالة حدوث خطأ.

تعيد الدالة tracefs_instance_file_read() مؤشرًا إلى سلسلة منتهية بـ NULL، تم قراءتها من الملف، أو NULL في حالة حدوث خطأ. يجب تحرير السلسلة المرجعة باستخدام free().

تعيد الدالة tracefs_instance_file_read_number() القيمة 0 إذا تم قراءة عدد صحيح صالح من الملف وتخزينه في res أو -1 في حالة حدوث خطأ.

#include <tracefs.h>
struct tracefs_instance *inst = tracefs_instance_create("foo");
        if (!inst) {
                /* خطأ في إنشاء مثيل تتبع جديد */
                ...
        }
        if (tracefs_file_exists(inst,"trace_clock")) {
                /* المثيل foo يدعم ساعة التتبع */
                char *path, *clock;
                int size;
                path =  = tracefs_instance_get_file(inst, "trace_clock")
                if (!path) {
                        /* خطأ في الحصول على المسار إلى ملف trace_clock في المثيل foo */
                        ...
                }
                ...
                tracefs_put_tracing_file(path);
                clock = tracefs_instance_file_read(inst, "trace_clock", &size);
                if (!clock) {
                        /* فشل في قراءة ملف trace_clock في المثيل foo */
                        ...
                }
                ...
                free(clock);
                if (tracefs_instance_file_write(inst, "trace_clock", "global") != strlen("global")) {
                        /* فشل في تعيين ساعة التتبع الشاملة في المثيل foo */
                        ...
                }
        } else {
                /* المثيل foo لا يدعم ساعة التتبع */
        }
        if (tracefs_dir_exists(inst,"options")) {
                /* المثيل foo يدعم خيارات التتبع */
                char *path = tracefs_instance_get_file(inst, "options");
                if (!path) {
                        /* خطأ في الحصول على المسار إلى الدليل options في المثيل foo */
                        ...
                }
                tracefs_put_tracing_file(path);
        } else {
                /* المثيل foo لا يدعم خيارات التتبع */
        }
        ...
        if (tracefs_instance_is_new(inst))
                tracefs_instance_destroy(inst);
        else
                tracefs_instance_free(inst);
        ...
        long long int res;
        if (tracefs_instance_file_read_number(NULL, "tracing_on", &res) == 0) {
                if (res == 0) {
                        /* تم تعطيل التتبع في المثيل الأعلى */
                } else if (res == 1) {
                        /* تم تمكين التتبع في المثيل الأعلى */
                } else {
                        /* حالة تتبع غير معروفة للمثيل الأعلى */
                }
        } else {
                /* فشل في قراءة عدد صحيح من ملف tracing_on */
        }
        ...
        int fd;
        fd = tracefs_instance_file_open(NULL, "tracing_on", O_WRONLY);
        if (fd >= 0) {
                /* تم الحصول على واصف الملف لملف tracing_on من المثيل الأعلى للكتابة */
                ...
                close(fd);
