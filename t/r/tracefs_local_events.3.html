tracefs_local_events(3)

الحزمة: libtracefs

التراخيص: GPL-2.0-only, LGPL-2.1-only

LIBTRACEFS(3) دليل libtracefs LIBTRACEFS(3)

الاسم

tracefs_local_events, tracefs_local_events_system, tracefs_fill_local_events, tracefs_load_cmdlines, tracefs_load_headers - تهيئة معالج tep بأحداث التتبع من النظام المحلي.

الملخص

#include <tracefs.h>
struct tep_handle *tracefs_local_events(const char *tracing_dir);
struct tep_handle *tracefs_local_events_system(const char *tracing_dir, const char * const *sys_names);
int tracefs_fill_local_events(const char *tracing_dir, struct tep_handle *tep, int *parsing_failures);
int tracefs_load_cmdlines(const char *tracing_dir, struct tep_handle *tep);
int tracefs_load_headers(const char *tracing_dir, struct tep_handle *tep);

الوصف

وظائف لتهيئة معالج tep بأحداث التتبع من النظام المحلي.

تقوم الدالة tracefs_local_events() بتخصيص معالج tep جديد وتهيئته بأحداث من جميع أنظمة التتبع، الموجودة في الدليل tracing_dir المحدد. يمكن أن يكون هذا NULL أو موقع نقطة تحميل tracefs لأنظمة تتبع الجهاز المحلي، أو قد يكون مسارًا لنسخة من دليل tracefs من جهاز آخر.

تقوم الدالة tracefs_local_events_system() بتخصيص معالج tep جديد وتهيئته بأحداث من أنظمة التتبع المحددة sys_names، الموجودة في الدليل tracing_dir المحدد. يمكن أن يكون هذا NULL أو موقع نقطة تحميل tracefs لأنظمة تتبع الجهاز المحلي، أو قد يكون مسارًا لنسخة من دليل tracefs من جهاز آخر. الوسيط sys_names هو مصفوفة من أسماء أنظمة التتبع، والتي سيتم استخدامها لتهيئة المعالج tep. يجب أن يكون العنصر الأخير في هذه المصفوفة مؤشر NULL.

تقوم الدالة tracefs_fill_local_events() بتهيئة المعالج tep المخصص بالفعل بأحداث من جميع أنظمة التتبع، الموجودة في الدليل tracing_dir المحدد. يمكن أن يكون هذا NULL أو موقع نقطة تحميل tracefs لأنظمة تتبع الجهاز المحلي، أو قد يكون مسارًا لنسخة من دليل tracefs من جهاز آخر. يجب أن يكون الوسيط tep مؤشرًا إلى معالج tep مخصص بالفعل، والذي سيتم تهيئته. يمكن أن يكون الوسيط parsing_failures NULL أو مؤشرًا إلى عدد صحيح، حيث يتم إرجاع عدد حالات الفشل أثناء تحليل ملفات الأحداث.

ستقوم الوظائف المذكورة أعلاه أيضًا بتحميل التعيينات بين معرفات العمليات (pids) وأسماء أوامر العمليات. في بعض الحالات، يتم إنشاء مقبض tep باستخدام إحدى الوظائف أعلاه قبل بدء التتبع. نظرًا لأنه يتم تحديث التعيينات أثناء التتبع، قد تكون هناك حاجة لقراءة التعيينات مرة أخرى بعد التتبع. تقوم الدالة tracefs_load_cmdlines() بذلك. tracing_dir هو دليل نقطة التحميل للتحميل منه، أو NULL لاستخدام نقطة التحميل لنظام ملفات tracefs.

ستقوم الدالة tracefs_load_headers() بقراءة "header_page" من دليل الأحداث، مما سيؤدي إلى تحديث مقبض tep بمعلومات حول كيفية تحليل المخزن المؤقت الفرعي لحلقة التتبع.

قيمة الإرجاع

تقوم الدالتان tracefs_local_events() و tracefs_local_events_system() بإرجاع مؤشر إلى معالج tep مخصص ومهيأ، أو NULL في حالة حدوث خطأ. يجب تحرير معالج tep المرجع باستخدام tep_free(3).

تقوم الدالة tracefs_fill_local_events() بإرجاع -1 في حالة حدوث خطأ، أو 0 بخلاف ذلك.

تقوم الدالة tracefs_load_cmdlines() بإرجاع -1 في حالة حدوث خطأ، أو 0 بخلاف ذلك.

مثال

#include <tracefs.h>
struct tep_handle *tep;
...
        tep = tracefs_local_events(NULL);
        if (!tep) {
                /* فشل في تهيئة معالج tep بأحداث محلية من المثيل العلوي */
                ...
        }
...
        tep_free(tep);
...
        const char *systems[] = {"ftrace", "irq", NULL};
        tep = tracefs_local_events_system(NULL, systems);
        if (!tep) {
                /* فشل في تهيئة معالج tep بأحداث محلية من
                 * أنظمة ftrace و irq في المثيل العلوي.
                 */
                ...
        }
...
        tep_free(tep);
...
        int parsing_failures;
        tep = tep_alloc();
        if (!tep) {
                /* فشل في تخصيص معالج tep */
                ...
        }
        if (tracefs_fill_local_events(NULL, tep, &parsing_failures) < 0) {
                /* فشل في تهيئة معالج tep بأحداث محلية من المثيل العلوي */
        }
        tracefs_load_cmdlines(NULL, tep);
...
        tep_free(tep);
