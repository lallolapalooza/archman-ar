trace_seq_destroy(3)

الحزمة: libtraceevent

التراخيص: GPL-2.0-only, LGPL-2.1-only

LIBTRACEEVENT(3) دليل libtraceevent LIBTRACEEVENT(3)

الاسم

trace_seq_init, trace_seq_destroy, trace_seq_reset, trace_seq_terminate, trace_seq_putc, trace_seq_puts, trace_seq_printf, trace_seq_vprintf, trace_seq_do_fprintf, trace_seq_do_printf - تهيئة / تدمير تسلسل تتبع.

الملخص

#include <event-parse.h>
#include <trace-seq.h>
void trace_seq_init(struct trace_seq *s);
void trace_seq_destroy(struct trace_seq *s);
void trace_seq_reset(struct trace_seq *s);
void trace_seq_terminate(struct trace_seq *s);
int trace_seq_putc(struct trace_seq *s, unsigned char c);
int trace_seq_puts(struct trace_seq *s, const char *str);
int trace_seq_printf(struct trace_seq *s, const char *fmt, ...);
int trace_seq_vprintf(struct trace_seq *s, const char *fmt, va_list args);
int trace_seq_do_printf(struct trace_seq *s);
int trace_seq_do_fprintf(struct trace_seq *s, FILE *fp);

الوصف

تُستخدم تسلسلات التتبع للسماح للدالة باستدعاء عدة دوال أخرى لإنشاء سلسلة بيانات للاستخدام.

تقوم الدالة trace_seq_init() بتهيئة تسلسل التتبع s.

تقوم الدالة trace_seq_destroy() بتدمير تسلسل التتبع s وتحرير جميع موارده التي تم استخدامها.

تقوم الدالة trace_seq_reset() بإعادة تهيئة تسلسل التتبع s. سيتم حذف جميع الأحرف التي تم كتابتها بالفعل في s.

تقوم الدالة trace_seq_terminate() بإنهاء تسلسل التتبع s. تضع حرف null '\0' في نهاية المخزن المؤقت.

تقوم الدالة trace_seq_putc() بوضع حرف واحد c في تسلسل التتبع s.

تقوم الدالة trace_seq_puts() بوضع سلسلة منتهية بـ NULL str في تسلسل التتبع s.

تقوم الدالة trace_seq_printf() بوضع سلسلة مهيأة fmt _مع وسائط متغيرة _... في تسلسل التتبع s.

تقوم الدالة trace_seq_vprintf() بوضع سلسلة مهيأة fmt _مع قائمة وسائط _args في تسلسل التتبع s.

تقوم الدالة trace_seq_do_printf() بطباعة المخزن المؤقت لتسلسل التتبع s إلى المخرج القياسي stdout.

تقوم الدالة trace_seq_do_fprintf() بطباعة المخزن المؤقت لتسلسل التتبع s إلى الملف المحدد fp.

قيمة الإرجاع

تعيد كلتا الدالتين trace_seq_putc() و trace_seq_puts() عدد الأحرف التي تم وضعها في تسلسل التتبع، أو 0 في حالة حدوث خطأ.

تعيد كلتا الدالتين trace_seq_printf() و trace_seq_vprintf() القيمة 0 إذا تجاوز التتبع المساحة الحرة للمخزن المؤقت، أو عدد الأحرف المطبوعة، أو قيمة سالبة في حالة حدوث خطأ.

تعيد كلتا الدالتين trace_seq_do_printf() و trace_seq_do_fprintf() عدد الأحرف المطبوعة، أو -1 في حالة حدوث خطأ.

مثال

#include <event-parse.h>
#include <trace-seq.h>
...
struct trace_seq seq;
trace_seq_init(&seq);
...
void foo_seq_print(struct trace_seq *tseq, char *format, ...)
{
        va_list ap;
        va_start(ap, format);
        if (trace_seq_vprintf(tseq, format, ap) <= 0) {
                /* فشل الطباعة في تسلسل التتبع */
        }
        va_end(ap);
}
trace_seq_reset(&seq);
char *str = " مثال صفحة دليل";
if (trace_seq_puts(&seq, str) != strlen(str)) {
        /* فشل وضع str في تسلسل التتبع */
}
if (trace_seq_putc(&seq, ':') != 1) {
        /* فشل وضع ':' في تسلسل التتبع */
}
if (trace_seq_printf(&seq, " تسلسل تتبع: %d", 1) <= 0) {
        /* فشل الطباعة في تسلسل التتبع */
}
foo_seq_print( &seq, "  %d\n", 2);
trace_seq_terminate(&seq);
...
if (trace_seq_do_printf(&seq) < 0 ) {
        /* فشل طباعة المخزن المؤقت للتسلسل إلى المخرج القياسي */
}
FILE *fp = fopen("trace.txt", "w");
if (trace_seq_do_fprintf(&seq, fp) < 0 ) [
        /* فشل طباعة المخزن المؤقت للتسلسل إلى ملف trace.txt */
}
trace_seq_destroy(&seq);
...
