tracefs_buffer_stat_timestamp(3)

الحزمة: libtracefs

التراخيص: GPL-2.0-only, LGPL-2.1-only

LIBTRACEFS(3) دليل libtracefs LIBTRACEFS(3)

الاسم

tracefs_instance_get_stat, tracefs_instance_put_stat, tracefs_buffer_stat_entries, tracefs_buffer_stat_overrun, tracefs_buffer_stat_commit_overrun, tracefs_buffer_stat_bytes, tracefs_buffer_stat_event_timestamp, tracefs_buffer_stat_timestamp, tracefs_buffer_stat_dropped_events, tracefs_buffer_stat_read_events - التعامل مع إحصائيات مخزن التتبع المؤقت

الملخص

#include <tracefs.h>
struct tracefs_buffer_stat *tracefs_instance_get_stat(struct tracefs_instance *instance, int cpu);
void tracefs_instance_put_stat(struct tracefs_buffer_stat *tstat);
ssize_t tracefs_buffer_stat_entries(struct tracefs_buffer_stat *tstat);
ssize_t tracefs_buffer_stat_overrun(struct tracefs_buffer_stat *tstat);
ssize_t tracefs_buffer_stat_commit_overrun(struct tracefs_buffer_stat *tstat);
ssize_t tracefs_buffer_stat_bytes(struct tracefs_buffer_stat *tstat);
long long tracefs_buffer_stat_event_timestamp(struct tracefs_buffer_stat *tstat);
long long tracefs_buffer_stat_timestamp(struct tracefs_buffer_stat *tstat);
ssize_t tracefs_buffer_stat_dropped_events(struct tracefs_buffer_stat *tstat);
ssize_t tracefs_buffer_stat_read_events(struct tracefs_buffer_stat *tstat);

الوصف

تقوم هذه المجموعة من الدوال بقراءة وتحليل ملف tracefs/per_cpu/cpuX/stats. تحتوي هذه الملفات على إحصائيات مخزن الحلقة لكل وحدة معالجة مركزية، مثل عدد الأحداث الموجودة في المخزن المؤقت، وعدد الأحداث التي تمت قراءتها وما إلى ذلك.

تقوم الدالة tracefs_instance_get_stat() بقراءة وتحليل ملف إحصائيات معين لمثيل instance ووحدة معالجة مركزية cpu معينة. نظرًا لأن المخزن المؤقت مقسم إلى مخازن مؤقتة لكل وحدة معالجة مركزية، فإن المعلومات ترتبط فقط بوحدة المعالجة المركزية cpu المعطاة. يمكن استخدام المؤشر tracefs_buffer_stat الذي تم إرجاعه مع دوال tracefs_buffer_stat الأخرى ويجب تحريره باستخدام tracefs_instance_put_stat().

تقوم الدالة tracefs_instance_put_stat() بتحرير الموارد المخصصة لـ stat المعطاة التي تم إنشاؤها بواسطة tracefs_instance_get_stat().

ترجع الدالة tracefs_buffer_stat_entries() عدد الأحداث الموجودة حاليًا في المخزن المؤقت المرتبط بـ tstat.

ترجع الدالة tracefs_buffer_stat_overrun() عدد الأحداث التي تم فقدانها بسبب تجاوز كاتب المخزن المؤقت للقارئ.

ترجع الدالة tracefs_buffer_stat_commit_overrun() عدد الأحداث التي تم فقدانها لأن المخزن المؤقت كان صغيرًا جدًا وقاطعت مقاطعة حدث سياق أدنى قيد التسجيل وأضافت أحداثًا أكثر مما يمكن أن يحتويه المخزن المؤقت. لاحظ أن هذا ليس شيئًا شائعًا وعندما يحدث يعني أن شيئًا ما لم يتم إعداده بشكل صحيح.

ترجع الدالة tracefs_buffer_stat_bytes() عدد البايتات التي تشغلها الأحداث الحالية. ملاحظة، أنها تتضمن البيانات الوصفية للأحداث، ولكنها لا تتضمن البيانات الوصفية للمخازن المؤقتة الفرعية.

ترجع الدالة tracefs_buffer_stat_event_timestamp() الطابع الزمني لآخر حدث في المخزن المؤقت.

ترجع الدالة tracefs_buffer_stat_timestamp() الطابع الزمني الحالي للمخزن المؤقت. ملاحظة، أنه يتم قراءته فقط عند استدعاء tracefs_instance_get_stat(). سيكون له الطابع الزمني للمخزن المؤقت عندما تم استدعاء تلك الدالة.

ترجع الدالة tracefs_buffer_stat_dropped_events() عدد الأحداث التي تم إسقاطها إذا كان وضع الكتابة فوق معطلاً. ستظهر الأحداث التي تم فقدانها لأن الكاتب لحق بالقارئ ولم يتمكن من كتابة المزيد من الأحداث.

ترجع الدالة tracefs_buffer_stat_read_events() عدد الأحداث التي تم استهلاكها بواسطة قارئ.

قيمة الإرجاع

ترجع الدالة tracefs_instance_get_stat() بنية tracefs_buffer_stat التي يمكن استخدامها لاسترداد الإحصائيات عبر الدوال الأخرى. يجب تحريرها باستخدام tracefs_instance_put_stat().

الدوال الأخرى التي ترجع قيمًا مختلفة من بنية tracefs_buffer_stat ترجع جميعها القيمة، أو -1 إذا لم يتم العثور على القيمة.

مثال

#include <stdlib.h>
#include <unistd.h>
#include <tracefs.h>
int main(int argc, char **argv)
{
        char *trace;
        char buf[1000];
        int ret;
        int i;
        for (i = 0; i < sizeof(buf) - 1; i++) {
                buf[i] = '0' + i % 10;
        }
        buf[i] = '\0';
        tracefs_instance_clear(NULL);
        for (i = 0; i < 4; i++) {
                ret = tracefs_printf(NULL, "%s\n", buf);
                if (ret < 0)
                        perror("write");
        }
        trace = tracefs_instance_file_read(NULL, "trace", NULL);
        printf("%s\n", trace);
        free(trace);
        for (i = 0; i < sysconf(_SC_NPROCESSORS_CONF); i++) {
                struct tracefs_buffer_stat *tstat;
                ssize_t entries, eread;
                tstat = tracefs_instance_get_stat(NULL, i);
                if (!tstat)
                        continue;
                entries = tracefs_buffer_stat_entries(tstat);
                eread = tracefs_buffer_stat_read_events(tstat);
                if (!entries && !eread) {
                        tracefs_instance_put_stat(tstat);
                        continue;
                }
                printf("CPU: %d\n", i);;
                printf("\tentries: %zd\n", entries);
                printf("\toverrun: %zd\n", tracefs_buffer_stat_overrun(tstat));
                printf("\tcommit_overrun: %zd\n", tracefs_buffer_stat_commit_overrun(tstat));
                printf("\tbytes: %zd\n", tracefs_buffer_stat_bytes(tstat));
                printf("\tevent_timestamp: %lld\n", tracefs_buffer_stat_event_timestamp(tstat));
                printf("\ttimestamp: %lld\n", tracefs_buffer_stat_timestamp(tstat));
                printf("\tdropped_events: %zd\n", tracefs_buffer_stat_dropped_events(tstat));
                printf("\tread_events: %zd\n", eread);
                tracefs_instance_put_stat(tstat);
        }
}
