tracefs_event_systems(3)

الحزمة: libtracefs

التراخيص: GPL-2.0-only, LGPL-2.1-only

LIBTRACEFS(3) دليل libtracefs LIBTRACEFS(3)

الاسم

tracefs_event_systems, tracefs_system_events, tracefs_event_enable, tracefs_event_disable, tracefs_event_is_enabled - العمل مع أنظمة وأحداث التتبع.

مُلَخَّص

#include <tracefs.h>
enum tracefs_event_state {
        TRACEFS_ERROR = -1,
        TRACEFS_ALL_DISABLED = 0,
        TRACEFS_ALL_ENABLED = 1,
        TRACEFS_SOME_ENABLED = 2,
};
char **tracefs_event_systems(const char *tracing_dir);
char **tracefs_system_events(const char *tracing_dir, const char *system);
int tracefs_event_enable(struct tracefs_instance *instance, const char *system,
                           const char *event);
int tracefs_event_disable(struct tracefs_instance *instance, const char *system,
                            const char *event);
enum tracefs_enable_state tracefs_event_is_enabled(struct tracefs_instance *instance,
                         const char *system, const char *event);

الوصف

واجهات برمجة التطبيقات المتعلقة بأنظمة وأحداث التتبع.

تقوم الدالة tracefs_event_systems() بإرجاع مصفوفة من السلاسل النصية بأسماء جميع أنظمة التتبع المسجلة، الموجودة في الدليل tracing_dir المحدد. يمكن أن يكون هذا NULL أو موقع نقطة تحميل tracefs لأنظمة التتبع للجهاز المحلي، أو قد يكون مسارًا لنسخة من دليل tracefs من جهاز آخر. العنصر الأخير في المصفوفة هو مؤشر NULL. يجب تحرير المصفوفة باستخدام واجهة برمجة التطبيقات tracefs_list_free().

تقوم الدالة tracefs_system_events() بإرجاع مصفوفة من السلاسل النصية بأسماء جميع أحداث التتبع المسجلة لنظام التتبع المحدد بواسطة system، الموجود في الدليل tracing_dir المحدد. يمكن أن يكون هذا NULL أو موقع نقطة تحميل tracefs لأنظمة التتبع للجهاز المحلي، أو قد يكون مسارًا لنسخة من دليل tracefs من جهاز آخر. العنصر الأخير في المصفوفة هو مؤشر NULL. يجب تحرير المصفوفة باستخدام واجهة برمجة التطبيقات tracefs_list_free().

تقوم الدالة tracefs_event_enable() بتمكين حدث معين بناءً على system و event الممررين للمثيل instance المحدد. إذا كان instance هو NULL، فسيتم استخدام دليل التتبع على المستوى الأعلى. إذا كان كل من system و event هما NULL، فسيتم تمكين جميع الأحداث للمثيل instance. إذا كان event هو NULL، فسيتم تمكين جميع الأحداث داخل system. إذا كان system هو NULL، فسيتم البحث في جميع الأنظمة وسيتم تمكين أي حدث داخل نظام يتطابق مع event. يمكن أن يكون كل من system و event تعبيرات قياسية كما هو محدد في regex(3).

تقوم الدالة tracefs_event_disable() بتعطيل الأحداث التي تطابق المعلمات system و event للمثيل instance المحدد. تخضع الأحداث التي يتم تعطيلها لنفس قواعد tracefs_event_enable() لمطابقة الأحداث. أي، إذا كان instance هو NULL، فسيتم استخدام دليل التتبع على المستوى الأعلى. إذا كان كل من system و event هما NULL، فسيتم تعطيل جميع الأحداث للمثيل المحدد، وهكذا.

ترجع الدالة tracefs_event_is_enabled() ما إذا كان الحدث ممكّنًا، أو مجموعة من الأحداث ممكّنة، أو نظام ممكّن، أو جميع الأحداث ممكّنة. إذا كان كل من system و event هما NULL، فسترجع حالة تمكين جميع الأحداث. إذا كان system غير NULL و event هو NULL، فستتحقق مما إذا كانت جميع الأحداث في جميع الأنظمة التي يحددها system وترجع حالة تمكين تلك الأحداث. إذا كان system هو NULL و event غير NULL، فستطابق جميع الأحداث في جميع الأنظمة التي تطابق event وترجع حالة تمكينها. إذا كان كل من system و event غير NULL، فسترجع حالة تمكين جميع الأحداث المطابقة. حالة التمكين معرفة على النحو التالي:

TRACEFS_ERROR - حدث خطأ بما في ذلك عدم تطابق أي أحداث.

TRACEFS_ALL_DISABLED - تم تعطيل جميع الأحداث المطابقة.

TRACEFS_ALL_ENABLED - تم تمكين جميع الأحداث المطابقة.

TRACEFS_SOME_ENABLED - تم تمكين بعض الأحداث المطابقة بينما لم يتم تمكين أحداث أخرى.

قيمة الإرجاع

ترجع الدالتان tracefs_event_systems() و tracefs_system_events() مصفوفة من السلاسل النصية. العنصر الأخير في تلك المصفوفة هو مؤشر NULL. يجب تحرير المصفوفة باستخدام واجهة برمجة التطبيقات tracefs_list_free(). في حالة حدوث خطأ، يتم إرجاع NULL.

ترجع كلتا الدالتين tracefs_event_enable() و tracefs_event_disable() القيمة 0 إذا وجدتا أي أحداث مطابقة (لاحظ أنها لا تتحقق من الحالة السابقة للحدث. إذا وجدت tracefs_event_enable() حدثًا ممكّنًا بالفعل، ولا توجد أخطاء أخرى، فسترجع 0). إذا حدث خطأ، حتى لو تم العثور على أحداث أخرى، فسترجع -1 وسيتم تعيين errno. إذا لم تحدث أخطاء، ولكن لم يتم العثور على أحداث تطابق المعلمات system و event، فسيتم إرجاع -1 ولن يتم تعيين errno.

ترجع tracefs_event_is_enabled() حالة تمكين الأحداث المطابقة أو TRACEFS_ERROR عند حدوث خطأ.

#include <tracefs.h>
char **systems = tracefs_event_systems(NULL);
        if (systems) {
                int i = 0;
                /* تم الحصول على أنظمة التتبع المسجلة من مثيل التتبع العلوي */
                while (systems[i]) {
                        char **events = tracefs_system_events(NULL, systems[i]);
                        if (events) {
                                /* تم الحصول على الأحداث المسجلة في system[i] من مثيل التتبع العلوي */
                                int j = 0;
                                while (events[j]) {
                                        /* تم الحصول على الحدث[j] في system[i] من مثيل التتبع العلوي */
                                        j++;
                                }
                                tracefs_list_free(events);
                        }
                        i++;
                }
                tracefs_list_free(systems);
        }
