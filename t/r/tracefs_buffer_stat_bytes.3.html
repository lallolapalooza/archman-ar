tracefs_buffer_stat_bytes(3)

الحزمة: libtracefs

التراخيص: GPL-2.0-only, LGPL-2.1-only

LIBTRACEFS(3) دليل libtracefs LIBTRACEFS(3)

الاسم

tracefs_instance_get_stat, tracefs_instance_put_stat, tracefs_buffer_stat_entries, tracefs_buffer_stat_overrun, tracefs_buffer_stat_commit_overrun, tracefs_buffer_stat_bytes, tracefs_buffer_stat_event_timestamp, tracefs_buffer_stat_timestamp, tracefs_buffer_stat_dropped_events, tracefs_buffer_stat_read_events - التعامل مع إحصائيات ذاكرة التتبع المؤقتة

الملخص

#include <tracefs.h>
struct tracefs_buffer_stat *tracefs_instance_get_stat(struct tracefs_instance *instance, int cpu);
void tracefs_instance_put_stat(struct tracefs_buffer_stat *tstat);
ssize_t tracefs_buffer_stat_entries(struct tracefs_buffer_stat *tstat);
ssize_t tracefs_buffer_stat_overrun(struct tracefs_buffer_stat *tstat);
ssize_t tracefs_buffer_stat_commit_overrun(struct tracefs_buffer_stat *tstat);
ssize_t tracefs_buffer_stat_bytes(struct tracefs_buffer_stat *tstat);
long long tracefs_buffer_stat_event_timestamp(struct tracefs_buffer_stat *tstat);
long long tracefs_buffer_stat_timestamp(struct tracefs_buffer_stat *tstat);
ssize_t tracefs_buffer_stat_dropped_events(struct tracefs_buffer_stat *tstat);
ssize_t tracefs_buffer_stat_read_events(struct tracefs_buffer_stat *tstat);

الوصف

تقوم هذه المجموعة من الدوال بقراءة وتحليل الملف tracefs/per_cpu/cpuX/stats. تحتفظ هذه الملفات بإحصائيات ذاكرة التتبع المؤقتة الدائرية لكل وحدة معالجة مركزية، مثل عدد الأحداث الموجودة في الذاكرة المؤقتة، وعدد الأحداث التي تمت قراءتها وما إلى ذلك.

تقوم الدالة tracefs_instance_get_stat() بقراءة وتحليل ملف إحصائيات معين لوحدة instance ووحدة معالجة مركزية cpu معينة. نظرًا لأن الذاكرة المؤقتة الدائرية مقسمة إلى ذاكرات مؤقتة لكل وحدة معالجة مركزية، فإن المعلومات مرتبطة فقط بوحدة المعالجة المركزية cpu المعطاة. يمكن استخدام المؤشر tracefs_buffer_stat المُعاد مع دوال tracefs_buffer_stat الأخرى ويجب تحريره باستخدام tracefs_instance_put_stat().

تقوم الدالة tracefs_instance_put_stat() بتحرير الموارد المخصصة لـ stat المعطاة التي تم إنشاؤها بواسطة tracefs_instance_get_stat().

تعيد الدالة tracefs_buffer_stat_entries() عدد الأحداث الموجودة حاليًا في الذاكرة المؤقتة الدائرية المرتبطة بـ tstat.

تعيد الدالة tracefs_buffer_stat_overrun() عدد الأحداث التي فقدتها الذاكرة المؤقتة الدائرية بسبب تجاوز الكاتب للقارئ.

تعيد الدالة tracefs_buffer_stat_commit_overrun() عدد الأحداث التي فُقدت لأن الذاكرة المؤقتة الدائرية كانت صغيرة جدًا وقاطع مقاطعة حدث سياق أقل يتم تسجيله وأضاف أحداثًا أكثر مما يمكن أن تحتويه الذاكرة المؤقتة الدائرية. لاحظ أن هذا ليس شيئًا شائعًا وعندما يحدث فهذا يعني أن شيئًا ما لم يتم إعداده بشكل صحيح.

تعيد الدالة tracefs_buffer_stat_bytes() عدد البايتات التي تشغلها الأحداث الحالية. لاحظ أنها تشمل البيانات الوصفية للأحداث، ولكنها لا تشمل البيانات الوصفية للذاكرة المؤقتة الفرعية.

تعيد الدالة tracefs_buffer_stat_event_timestamp() الطابع الزمني لآخر حدث في الذاكرة المؤقتة الدائرية.

تعيد الدالة tracefs_buffer_stat_timestamp() الطابع الزمني الحالي للذاكرة المؤقتة الدائرية. لاحظ أنه يتم قراءتها فقط عند استدعاء tracefs_instance_get_stat(). سيكون لها الطابع الزمني للذاكرة المؤقتة الدائرية عندما تم استدعاء تلك الدالة.

تعيد الدالة tracefs_buffer_stat_dropped_events() عدد الأحداث التي تم إسقاطها إذا تم تعطيل وضع الكتابة فوق. ستظهر الأحداث التي فُقدت لأن الكاتب وصل إلى القارئ ولم يتمكن من كتابة المزيد من الأحداث.

تعيد الدالة tracefs_buffer_stat_read_events() عدد الأحداث التي استهلكها القارئ.

قيمة الإرجاع

تعيد الدالة tracefs_instance_get_stat() بنية tracefs_buffer_stat التي يمكن استخدامها لاسترداد الإحصائيات عبر الدوال الأخرى. يجب تحريرها باستخدام tracefs_instance_put_stat().

جميع الدوال الأخرى التي تعيد قيمًا مختلفة من بنية tracefs_buffer_stat تعيد القيمة، أو -1 إذا لم يتم العثور على القيمة.

مثال

#include <stdlib.h>
#include <unistd.h>
#include <tracefs.h>
int main(int argc, char **argv)
{
        char *trace;
        char buf[1000];
        int ret;
        int i;
        for (i = 0; i < sizeof(buf) - 1; i++) {
                buf[i] = '0' + i % 10;
        }
        buf[i] = '\0';
        tracefs_instance_clear(NULL);
        for (i = 0; i < 4; i++) {
                ret = tracefs_printf(NULL, "%s\n", buf);
                if (ret < 0)
                        perror("write");
        }
        trace = tracefs_instance_file_read(NULL, "trace", NULL);
        printf("%s\n", trace);
        free(trace);
        for (i = 0; i < sysconf(_SC_NPROCESSORS_CONF); i++) {
                struct tracefs_buffer_stat *tstat;
                ssize_t entries, eread;
                tstat = tracefs_instance_get_stat(NULL, i);
                if (!tstat)
                        continue;
                entries = tracefs_buffer_stat_entries(tstat);
                eread = tracefs_buffer_stat_read_events(tstat);
                if (!entries && !eread) {
                        tracefs_instance_put_stat(tstat);
                        continue;
                }
                printf("CPU: %d\n", i);;
                printf("\tentries: %zd\n", entries);
                printf("\toverrun: %zd\n", tracefs_buffer_stat_overrun(tstat));
                printf("\tcommit_overrun: %zd\n", tracefs_buffer_stat_commit_overrun(tstat));
                printf("\tbytes: %zd\n", tracefs_buffer_stat_bytes(tstat));
                printf("\tevent_timestamp: %lld\n", tracefs_buffer_stat_event_timestamp(tstat));
                printf("\ttimestamp: %lld\n", tracefs_buffer_stat_timestamp(tstat));
                printf("\tdropped_events: %zd\n", tracefs_buffer_stat_dropped_events(tstat));
                printf("\tread_events: %zd\n", eread);
                tracefs_instance_put_stat(tstat);
        }
}
