tracefs_snapshot_clear(3)

الحزمة: libtracefs

التراخيص: GPL-2.0-only, LGPL-2.1-only

LIBTRACEFS(3) دليل libtracefs LIBTRACEFS(3)

الاسم

tracefs_snapshot_snap, tracefs_snapshot_clear, tracefs_snapshot_free - واجهة برمجة تطبيقات لإنشاء، مسح، وقراءة اللقطات

الملخص

#include <tracefs.h>
int tracefs_snapshot_snap(struct tracefs_instance *instance);
int tracefs_snapshot_clear(struct tracefs_instance *instance);
int tracefs_snapshot_free(struct tracefs_instance *instance);

الوصف

يوفر تتبع نواة لينكس ميزة "اللقطة". تحتوي النواة على مخزنين دائريين. أحدهما يتم الكتابة إليه بواسطة نظام التتبع والآخر هو مخزن "اللقطة". عند حدوث لقطة، يتم تبديل المخزنين، ويصبح مخزن اللقطة الحالي هو الذي يتم الكتابة إليه، ويصبح المخزن الذي كان يتم الكتابة إليه هو اللقطة المحفوظة.

ملاحظة: يتم تخصيص مخزن اللقطة في المرة الأولى التي يتم فيها أخذه، لذا من الأفضل أخذ لقطة في البداية قبل الحاجة إليها حتى يتم تخصيصه وتكون لقطة جاهزة، ثم ستحدث اللقطة على الفور.

ستقوم tracefs_snapshot_snap() بتخصيص (إذا لم يتم تخصيصه بالفعل) مخزن اللقطة ثم تأخذ "لقطة" (تبديل المخزن الرئيسي الذي يتم الكتابة إليه مع مخزن اللقطة المخصص). ستقوم بذلك لمخزن instance المحدد أو المخزن الرئيسي إذا كان instance NULL.

ستقوم tracefs_snapshot_clear() بمسح محتويات مخزن اللقطة للمخزن instance المحدد أو المخزن الرئيسي إذا كان instance NULL.

ستقوم tracefs_snapshot_free() بتحرير اللقطة المخصصة للمخزن instance المحدد أو المخزن الرئيسي إذا كان instance NULL. وهذا يعني، إذا تم إجراء استدعاء آخر لـ tracefs_snapshot_snap() بعد هذا، فستحتاج إلى تخصيص مخزن اللقطة مرة أخرى قبل أن تتمكن من أخذ لقطة. يجب استخدام هذه الوظيفة لتحرير ذاكرة النواة التي يستخدمها مخزن اللقطة عند عدم الحاجة إليها.

قيمة الإرجاع

تُرجع tracefs_snapshot_snap()، tracefs_snapshot_clear()، و tracefs_snapshot_free() جميعها 0 عند النجاح و -1 عند الفشل.

مثال

#include <stdlib.h>
#include <stdio.h>
#include <tracefs.h>
static int callback(struct tep_event *event, struct tep_record *record, int cpu, void *data)
{
        static struct trace_seq seq;
        struct tep_handle *tep = event->tep;
        if (!seq.buffer)
                trace_seq_init(&seq);
        trace_seq_reset(&seq);
        tep_print_event(tep, &seq, record, "[%03d] %s-%d %6.1000d\t%s: %s\n",
                                TEP_PRINT_CPU,
                                TEP_PRINT_COMM,
                                TEP_PRINT_PID,
                                TEP_PRINT_TIME,
                                TEP_PRINT_NAME,
                                TEP_PRINT_INFO);
        trace_seq_do_printf(&seq);
        return 0;
}
int main (int argc, char **argv)
{
        struct tracefs_instance *instance;
        struct tep_handle *tep;
        char *line = NULL;
        size_t len = 0;
        int ret;
        instance = tracefs_instance_create("my_snapshots");
        if (!instance) {
                perror("creating instance");
                exit(-1);
        }
        tep = tracefs_local_events(NULL);
        if (!tep) {
                perror("reading event formats");
                goto out;
        }
        /* Make sure the snapshot buffer is allocated */
        ret = tracefs_snapshot_snap(instance);
        if (ret < 0)
                goto out;
        ret = tracefs_event_enable(instance, "sched", NULL);
        if (ret < 0) {
                perror("enabling event");
                goto out;
        }
        for (;;) {
                printf("Hit enter without text to take snapshot!\n");
                printf("Enter any text to display the snapshot\n");
                printf("Enter 'quit' to exit\n");
                getline(&line, &len, stdin);
                ret = tracefs_snapshot_snap(instance);
                if (ret < 0) {
                        perror("taking snapshot");
                        goto out;
                }
                if (!line)
                        break;
                if (strlen(line) < 2)
                        continue;
                if (strncmp(line, "quit", 4) == 0)
                        break;
                tracefs_iterate_snapshot_events(tep, instance, NULL, 0, callback, NULL);
        }
        free(line);
        tracefs_instance_clear(instance);
 out:
        tracefs_snapshot_free(instance);
        tracefs_event_disable(instance, "sched", NULL);
        tracefs_instance_destroy(instance);
        tracefs_instance_free(instance);
        exit(0);
}

tracefs_snapshot_clear(3)

tracefs_snapshot_clear - تمسح لقطات تتبع النظام.

الصيغة

#include <tracefs.h>
int tracefs_snapshot_clear(const char *snapshot_name);

الوصف

تقوم الدالة tracefs_snapshot_clear بمسح (حذف) لقطة تتبع نظام تم إنشاؤها مسبقًا. يتم تحديد اللقطة المراد مسحها بواسطة snapshot_name.

المعاملات

  • snapshot_name: اسم اللقطة التي سيتم مسحها.

القيمة المعادة

تُرجع الدالة 0 عند النجاح، و-1 عند الفشل. في حالة الفشل، يتم تعيين errno للإشارة إلى الخطأ.

الأخطاء

  • EINVAL: snapshot_name غير صالح.
  • ENOENT: اللقطة المسماة snapshot_name غير موجودة.
  • EACCES: لا توجد أذونات كافية لمسح اللقطة.

أمثلة

    #include <tracefs.h>
    #include <stdio.h>

    int main() {
        const char *snapshot = "my_snapshot";
        int ret;

        // افترض أن اللقطة "my_snapshot" موجودة وتم إنشاؤها
        // ...

        ret = tracefs_snapshot_clear(snapshot);
        if (ret == -1) {
            perror("tracefs_snapshot_clear");
            return 1;
        }

        printf("Snapshot '%s' cleared successfully.\n", snapshot);
        return 0;
    }
