subroutine ctrcon (character norm, character uplo, character diag, integer n, complex, dimension( lda, * ) a, integer lda, real rcond, complex, dimension( * ) work, real, dimension( * ) rwork, integer info)

CTRCON

الغرض:

!>
!> CTRCON تقدر مقلوب رقم الشرط للمصفوفة المثلثية A، سواء في المعيار 1 أو المعيار اللانهائي.
!>
!> يتم حساب معيار A ويتم الحصول على تقدير لـ
!> norm(inv(A))، ثم يتم حساب مقلوب رقم الشرط على أنه
!>    RCOND = 1 / ( norm(A) * norm(inv(A)) ).
!>

المعلمات

NORM 
!>          NORM is CHARACTER*1
!>          يحدد ما إذا كان مطلوبًا رقم الشرط 1-norm أو رقم الشرط infinity-norm:
!>          = '1' or 'O':  1-norm;
!>          = 'I':         Infinity-norm.
!>

UPLO

!>          UPLO is CHARACTER*1
!>          = 'U':  A هي مصفوفة مثلثية علوية؛
!>          = 'L':  A هي مصفوفة مثلثية سفلية.
!>

DIAG

!>          DIAG is CHARACTER*1
!>          = 'N':  A هي مصفوفة مثلثية غير وحدوية؛
!>          = 'U':  A هي مصفوفة مثلثية وحدوية.
!>

N

!>          N is INTEGER
!>          رتبة المصفوفة A.  N >= 0.
!>

A

!>          A is COMPLEX array, dimension (LDA,N)
!>          المصفوفة المثلثية A.  إذا كان UPLO = 'U'، فإن الجزء العلوي المثلثي
!>          الموجود في الجزء العلوي من المصفوفة A (بحجم N x N) يحتوي على
!>          المصفوفة المثلثية العلوية، ولا يتم الإشارة إلى الجزء المثلثي السفلي
!>          الصرف لـ A.  إذا كان UPLO = 'L'، فإن الجزء السفلي المثلثي
!>          الموجود في الجزء السفلي من المصفوفة A (بحجم N x N) يحتوي على
!>          المصفوفة المثلثية السفلية، ولا يتم الإشارة إلى الجزء المثلثي العلوي
!>          الصرف لـ A.  إذا كان DIAG = 'U'، فإن العناصر القطرية لـ A
!>          لا يتم الإشارة إليها أيضًا وتُفترض أنها 1.
!>

LDA

!>          LDA is INTEGER
!>          البعد الرائد للمصفوفة A.  LDA >= max(1,N).
!>

RCOND

!>          RCOND is REAL
!>          مقلوب رقم شرط المصفوفة A،
!>          يتم حسابه كـ RCOND = 1/(norm(A) * norm(inv(A))).
!>

WORK

!>          WORK is COMPLEX array, dimension (2*N)
!>

RWORK

!>          RWORK is REAL array, dimension (N)
!>

INFO

!>          INFO is INTEGER
!>          = 0:  انتهاء ناجح
!>          < 0:  إذا كان INFO = -i، فإن المعلمة i كانت لها قيمة غير قانونية
!>
