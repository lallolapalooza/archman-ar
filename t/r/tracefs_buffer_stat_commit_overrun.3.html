tracefs_buffer_stat_commit_overrun(3)

الحزمة: libtracefs

التراخيص: GPL-2.0-only, LGPL-2.1-only

LIBTRACEFS(3) دليل libtracefs LIBTRACEFS(3)

الاسم

tracefs_instance_get_stat, tracefs_instance_put_stat, tracefs_buffer_stat_entries, tracefs_buffer_stat_overrun, tracefs_buffer_stat_commit_overrun, tracefs_buffer_stat_bytes, tracefs_buffer_stat_event_timestamp, tracefs_buffer_stat_timestamp, tracefs_buffer_stat_dropped_events, tracefs_buffer_stat_read_events - التعامل مع إحصائيات المخزن المؤقت للتتبع

الملخص

#include <tracefs.h>
struct tracefs_buffer_stat *tracefs_instance_get_stat(struct tracefs_instance *instance, int cpu);
void tracefs_instance_put_stat(struct tracefs_buffer_stat *tstat);
ssize_t tracefs_buffer_stat_entries(struct tracefs_buffer_stat *tstat);
ssize_t tracefs_buffer_stat_overrun(struct tracefs_buffer_stat *tstat);
ssize_t tracefs_buffer_stat_commit_overrun(struct tracefs_buffer_stat *tstat);
ssize_t tracefs_buffer_stat_bytes(struct tracefs_buffer_stat *tstat);
long long tracefs_buffer_stat_event_timestamp(struct tracefs_buffer_stat *tstat);
long long tracefs_buffer_stat_timestamp(struct tracefs_buffer_stat *tstat);
ssize_t tracefs_buffer_stat_dropped_events(struct tracefs_buffer_stat *tstat);
ssize_t tracefs_buffer_stat_read_events(struct tracefs_buffer_stat *tstat);

الوصف

تقوم هذه المجموعة من الوظائف بقراءة وتحليل ملف tracefs/per_cpu/cpuX/stats. تحتفظ هذه الملفات بإحصائيات المخزن المؤقت الحلقي لكل وحدة معالجة مركزية، مثل عدد الأحداث الموجودة في المخزن المؤقت الحلقي، وعدد الأحداث التي تمت قراءتها وما إلى ذلك.

ستقوم وظيفة tracefs_instance_get_stat() بقراءة وتحليل ملف إحصائيات معين لـ instance و cpu معينة. نظرًا لأن المخزن المؤقت الحلقي مقسم إلى مخازن مؤقتة لكل وحدة معالجة مركزية، فإن المعلومات مرتبطة فقط بوحدة المعالجة المركزية cpu المعينة. يمكن استخدام المؤشر tracefs_buffer_stat المُرجع مع وظائف tracefs_buffer_stat الأخرى ويجب تحريره باستخدام tracefs_instance_put_stat().

ستقوم وظيفة tracefs_instance_put_stat() بتحرير الموارد المخصصة لـ stat المعطاة والتي تم إنشاؤها بواسطة tracefs_instance_get_stat().

تعيد وظيفة tracefs_buffer_stat_entries() عدد الأحداث الموجودة حاليًا في المخزن المؤقت الحلقي المرتبط بـ tstat.

تعيد وظيفة tracefs_buffer_stat_overrun() عدد الأحداث التي فقدها قارئ المخزن المؤقت الحلقي بسبب تجاوزه.

تعيد وظيفة tracefs_buffer_stat_commit_overrun() عدد الأحداث التي فقدت بسبب أن المخزن المؤقت الحلقي كان صغيرًا جدًا وأن مقاطعة قطعت حدث سياق أقل أثناء تسجيله وأضافت أحداثًا أكثر مما يمكن أن يحتويه المخزن المؤقت الحلقي. لاحظ أن هذا ليس حدثًا شائعًا وعندما يحدث يعني أن شيئًا ما لم يتم إعداده بشكل صحيح.

تعيد وظيفة tracefs_buffer_stat_bytes() عدد البايتات التي تشغلها الأحداث الحالية. ملاحظة، تتضمن البيانات الوصفية للأحداث، ولكنها لا تتضمن البيانات الوصفية للمخازن المؤقتة الفرعية.

تعيد وظيفة tracefs_buffer_stat_event_timestamp() الطابع الزمني لآخر حدث في المخزن المؤقت الحلقي.

تعيد وظيفة tracefs_buffer_stat_timestamp() الطابع الزمني الحالي للمخزن المؤقت الحلقي. ملاحظة، يتم قراءتها فقط عند استدعاء tracefs_instance_get_stat(). سيكون لها الطابع الزمني للمخزن المؤقت الحلقي عند استدعاء تلك الوظيفة.

تعيد وظيفة tracefs_buffer_stat_dropped_events() عدد الأحداث التي تم إسقاطها إذا تم تعطيل وضع الكتابة فوق. ستظهر الأحداث التي فقدت بسبب أن الكاتب لحق بالقارئ ولم يتمكن من كتابة المزيد من الأحداث.

تعيد وظيفة tracefs_buffer_stat_read_events() عدد الأحداث التي استهلكها القارئ.

قيمة الإرجاع

تُرجع الوظيفة tracefs_instance_get_stat() بنية tracefs_buffer_stat التي يمكن استخدامها لاسترداد الإحصائيات عبر الوظائف الأخرى. يجب تحريرها باستخدام tracefs_instance_put_stat().

الوظائف الأخرى التي تُرجع قيمًا مختلفة من بنية tracefs_buffer_stat تُرجع جميعها القيمة، أو -1 إذا لم يتم العثور على القيمة.

مثال

#include <stdlib.h>
#include <unistd.h>
#include <tracefs.h>
int main(int argc, char **argv)
{
        char *trace;
        char buf[1000];
        int ret;
        int i;
        for (i = 0; i < sizeof(buf) - 1; i++) {
                buf[i] = '0' + i % 10;
        }
        buf[i] = '\0';
        tracefs_instance_clear(NULL);
        for (i = 0; i < 4; i++) {
                ret = tracefs_printf(NULL, "%s\n", buf);
                if (ret < 0)
                        perror("write");
        }
        trace = tracefs_instance_file_read(NULL, "trace", NULL);
        printf("%s\n", trace);
        free(trace);
        for (i = 0; i < sysconf(_SC_NPROCESSORS_CONF); i++) {
                struct tracefs_buffer_stat *tstat;
                ssize_t entries, eread;
                tstat = tracefs_instance_get_stat(NULL, i);
                if (!tstat)
                        continue;
                entries = tracefs_buffer_stat_entries(tstat);
                eread = tracefs_buffer_stat_read_events(tstat);
                if (!entries && !eread) {
                        tracefs_instance_put_stat(tstat);
                        continue;
                }
                printf("CPU: %d\n", i);;
                printf("\tentries: %zd\n", entries);
                printf("\toverrun: %zd\n", tracefs_buffer_stat_overrun(tstat));
                printf("\tcommit_overrun: %zd\n", tracefs_buffer_stat_commit_overrun(tstat));
                printf("\tbytes: %zd\n", tracefs_buffer_stat_bytes(tstat));
                printf("\tevent_timestamp: %lld\n", tracefs_buffer_stat_event_timestamp(tstat));
                printf("\ttimestamp: %lld\n", tracefs_buffer_stat_timestamp(tstat));
                printf("\tdropped_events: %zd\n", tracefs_buffer_stat_dropped_events(tstat));
                printf("\tread_events: %zd\n", eread);
                tracefs_instance_put_stat(tstat);
        }
}
