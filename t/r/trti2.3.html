الوصف التفصيلي

توثيق الوظيفة

subroutine ctrti2 (character uplo, character diag, integer n, complex, dimension( lda, * ) a, integer lda, integer info)

CTRTI2 تحسب معكوس مصفوفة مثلثة (خوارزمية غير مجمعة).

الغرض:

!>
!> CTRTI2 تحسب معكوس مصفوفة مثلثة علوية أو سفلية مركبة.
!>
!> هذه هي نسخة المستوى 2 BLAS من الخوارزمية.
!>

المعاملات

UPLO 
!>          UPLO هو CHARACTER*1
!>          يحدد ما إذا كانت المصفوفة A مثلثة علوية أم سفلية.
!>          = 'U':  مثلثة علوية
!>          = 'L':  مثلثة سفلية
!>

DIAG

!>          DIAG هو CHARACTER*1
!>          يحدد ما إذا كانت المصفوفة A مثلثة الوحدة أم لا.
!>          = 'N':  غير مثلثة الوحدة
!>          = 'U':  مثلثة الوحدة
!>

N

!>          N هو INTEGER
!>          رتبة المصفوفة A.  N >= 0.
!>

A

!>          A هو مصفوفة COMPLEX، ذات بعد (LDA,N)
!>          عند الإدخال، المصفوفة المثلثة A.  إذا كان UPLO = 'U'، فإن
!>          الجزء العلوي الأيمن من المصفوفة A ذي البعد n × n يحتوي على
!>          المصفوفة المثلثة العلوية، والجزء السفلي الأدنى
!>          من المصفوفة A لا يتم الرجوع إليه.  إذا كان UPLO = 'L'، فإن
!>          الجزء السفلي الأيسر من المصفوفة A ذي البعد n × n يحتوي على
!>          المصفوفة المثلثة السفلية، والجزء العلوي الأدنى
!>          من المصفوفة A لا يتم الرجوع إليه.  إذا كان DIAG = 'U'، فإن
!>          العناصر القطرية لـ A لا يتم الرجوع إليها أيضًا ويُفترض أنها 1.
!>
!>          عند الخروج، معكوس (المصفوفة المثلثة) للمصفوفة الأصلية، بنفس
!>          تنسيق التخزين.
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرائد للمصفوفة A.  LDA >= max(1,N).
!>

INFO

!>          INFO هو INTEGER
!>          = 0: خروج ناجح
!>          < 0: إذا كان INFO = -k، فقد كان المعامل k ذي قيمة غير قانونية
!>
