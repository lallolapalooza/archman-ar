tracecmd_get_traceid(3)

الحزمة: trace-cmd

التراخيص: GPL-2.0-only, LGPL-2.1-only

LIBTRACECMD(3) دليل libtracefs LIBTRACECMD(3)

الاسم

tracecmd_get_traceid, tracecmd_get_guest_cpumap - إدارة جلسة تتبع مع نظراء تتبع متعددين، مسجلة في ملفات تتبع متعددة.

الملخص

#include <trace-cmd.h>
unsigned long long tracecmd_get_traceid(struct tracecmd_input *handle);
int tracecmd_get_guest_cpumap(struct tracecmd_input *handle, unsigned long long trace_id, const char **name, int *vcpu_count, const int **cpu_pid);

الوصف

يمكن استخدام مجموعة الواجهات البرمجية هذه لإدارة جلسة تتبع مع نظراء تتبع متعددين، على سبيل المثال، تتبع كل من المضيف وجهاز افتراضي واحد أو أكثر. يتم تسجيل بيانات التتبع لكل نظير من الجلسة في ملفات تتبع منفصلة. يتم تخزين المعلومات حول النظراء من الجلسة في البيانات الوصفية لكل ملف تتبع. تستخدم هذه الواجهات البرمجية تلك المعلومات لاستخراج بيانات التتبع ومزامنتها.

تقوم الدالة tracecmd_get_traceid() بإرجاع معرف التتبع المخزن في البيانات الوصفية لملف التتبع المرتبط بالمقبض handle. لكل نظير من جلسة تتبع معرف فريد لهذا النظير وتلك الجلسة التتبعية فقط. يُستخدم هذا المعرف لمطابقة ملفات التتبع المتعددة المسجلة في نفس جلسة التتبع.

تقوم الدالة tracecmd_get_guest_cpumap() بالحصول على تعيين وحدات المعالجة المركزية الافتراضية للضيف (VCPU) إلى العملية المضيفة التي تمثل وحدات المعالجة المركزية الافتراضية هذه ويتم تخزينها في البيانات الوصفية لملف التتبع المرتبط بالمقبض handle. يتم جمع هذه المعلومات أثناء جلسة تتبع مضيف-ضيف ويتم تخزينها في ملف التتبع المضيف. المعلمة trace_id هي معرف تتبع الضيف في جلسة التتبع هذه. إذا كان الضيف الذي يحمل هذا المعرف جزءًا من تلك الجلسة، فإن تعيين VCPU إلى عملية المضيف موجود في ملف التتبع المضيف ويتم إرجاع المعلومات في المعلمات name و vcpu_count و cpu_pid. تحتوي المعلمة name على اسم الضيف، وتحتوي المعلمة vcpu_count على عدد وحدات المعالجة المركزية الافتراضية للضيف، وتحتوي مصفوفة cpu_pid على تعيين VCPU إلى عملية المضيف. المصفوفة بحجم vcpu_count حيث يكون الفهرس هو VCPU والقيمة هي معرف العملية (PID) للعملية المضيفة التي تقوم بتشغيل VCPU هذا. يجب عدم تحرير name و vcpu_count و cpu_pid.

قيمة الإرجاع

تقوم الدالة tracecmd_get_traceid() بإرجاع معرف تتبع 64 بت.

تقوم الدالة tracecmd_get_guest_cpumap() بإرجاع -1 في حالة حدوث خطأ أو 0 بخلاف ذلك. إذا تم إرجاع 0، فإن المعلمات name و vcpu_count و cpu_pid تحتوي على المعلومات المطلوبة.

مثال

#include <trace-cmd.h>
...
struct tracecmd_input *host = tracecmd_open("trace.dat");
        if (!host) {
                /* فشل في فتح ملف التتبع المضيف */
        }
struct tracecmd_input *guest1 = tracecmd_open_head("trace-Guest1.dat");
        if (!guest1) {
                /* فشل في فتح ملف التتبع guest1 */
        }
struct tracecmd_input *guest2 = tracecmd_open_head("trace-Guest2.dat");
        if (!guest2) {
                /* فشل في فتح ملف التتبع guest2 */
        }
unsigned long long guest_id_1 = tracecmd_get_traceid(guest1);
unsigned long long guest_id_2 = tracecmd_get_traceid(guest2);
int *cpu_pid_1, *cpu_pid_2;
int vcount_1, vcount_2;
char *name_1, *name_2;
        if (!tracecmd_get_guest_cpumap(host, guest_id_1, &name_1, &vcount_1, &cpu_pid_1)) {
                /* المضيف وضيف guest1 بالاسم name_1 جزء من نفس جلسة التتبع.
                 * تم الحصول على تعيين VCPU لـ guest1 إلى PID المضيف.
                 */
        }
        if (!tracecmd_get_guest_cpumap(host, guest_id_2, &name_2, &vcount_2, &cpu_pid_2)) {
                /* المضيف وضيف guest2 بالاسم name_2 جزء من نفس جلسة التتبع.
                 * تم الحصول على تعيين VCPU لـ guest2 إلى PID المضيف.
                 */
        }
...
        tracecmd_close(guest1);
        tracecmd_close(guest2);
        tracecmd_close(handle);
