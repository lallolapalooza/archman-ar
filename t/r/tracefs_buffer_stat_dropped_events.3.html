tracefs_buffer_stat_dropped_events(3)

الحزمة: libtracefs

التراخيص: GPL-2.0-only, LGPL-2.1-only

LIBTRACEFS(3) libtracefs Manual LIBTRACEFS(3)

الاسم

tracefs_instance_get_stat, tracefs_instance_put_stat, tracefs_buffer_stat_entries, tracefs_buffer_stat_overrun, tracefs_buffer_stat_commit_overrun, tracefs_buffer_stat_bytes, tracefs_buffer_stat_event_timestamp, tracefs_buffer_stat_timestamp, tracefs_buffer_stat_dropped_events, tracefs_buffer_stat_read_events - التعامل مع إحصائيات مخزن التتبع

الملخص

#include <tracefs.h>
struct tracefs_buffer_stat *tracefs_instance_get_stat(struct tracefs_instance *instance, int cpu);
void tracefs_instance_put_stat(struct tracefs_buffer_stat *tstat);
ssize_t tracefs_buffer_stat_entries(struct tracefs_buffer_stat *tstat);
ssize_t tracefs_buffer_stat_overrun(struct tracefs_buffer_stat *tstat);
ssize_t tracefs_buffer_stat_commit_overrun(struct tracefs_buffer_stat *tstat);
ssize_t tracefs_buffer_stat_bytes(struct tracefs_buffer_stat *tstat);
long long tracefs_buffer_stat_event_timestamp(struct tracefs_buffer_stat *tstat);
long long tracefs_buffer_stat_timestamp(struct tracefs_buffer_stat *tstat);
ssize_t tracefs_buffer_stat_dropped_events(struct tracefs_buffer_stat *tstat);
ssize_t tracefs_buffer_stat_read_events(struct tracefs_buffer_stat *tstat);

الوصف

تقوم هذه المجموعة من الدوال بقراءة وتحليل ملف tracefs/per_cpu/cpuX/stats. تحتوي هذه الملفات على إحصائيات مخزن الحلقة لكل وحدة معالجة مركزية، مثل عدد الأحداث الموجودة في مخزن الحلقة، وعدد الأحداث التي تمت قراءتها وما إلى ذلك.

تقوم الدالة tracefs_instance_get_stat() بقراءة وتحليل ملف الإحصائيات المحدد لمثيل instance ووحدة المعالجة المركزية cpu المحددين. نظرًا لأن مخزن الحلقة مقسم إلى مخازن لكل وحدة معالجة مركزية، فإن المعلومات ترتبط فقط بوحدة المعالجة المركزية cpu المحددة. يمكن استخدام المؤشر tracefs_buffer_stat الذي تم إرجاعه مع الدوال الأخرى tracefs_buffer_stat ويجب تحريره باستخدام tracefs_instance_put_stat().

تقوم الدالة tracefs_instance_put_stat() بتحرير الموارد المخصصة لـ stat المحدد الذي تم إنشاؤه بواسطة tracefs_instance_get_stat().

تقوم الدالة tracefs_buffer_stat_entries() بإرجاع عدد الأحداث الموجودة حاليًا في مخزن الحلقة المرتبط بـ tstat.

تقوم الدالة tracefs_buffer_stat_overrun() بإرجاع عدد الأحداث التي فقدها كاتب مخزن الحلقة بسبب تجاوز القارئ.

تقوم الدالة tracefs_buffer_stat_commit_overrun() بإرجاع عدد الأحداث التي فقدت لأن مخزن الحلقة كان صغيرًا جدًا وتداخل مقاطعة مع تسجيل حدث سياق أقل وأضافت المزيد من الأحداث مما يمكن أن يحتويه مخزن الحلقة. لاحظ أن هذا ليس شيئًا شائعًا وعندما يحدث يعني أن شيئًا ما لم يتم إعداده بشكل صحيح.

تقوم الدالة tracefs_buffer_stat_bytes() بإرجاع عدد البايتات التي تشغلها الأحداث الحالية. لاحظ أنها تشمل البيانات الوصفية للأحداث، ولكنها لا تشمل البيانات الوصفية للمخازن الفرعية.

تقوم الدالة tracefs_buffer_stat_event_timestamp() بإرجاع الطابع الزمني لآخر حدث في مخزن الحلقة.

تقوم الدالة tracefs_buffer_stat_timestamp() بإرجاع الطابع الزمني الحالي لمخزن الحلقة. لاحظ أنه يتم قراءته فقط عند استدعاء tracefs_instance_get_stat(). سيكون لها الطابع الزمني لمخزن الحلقة عند استدعاء تلك الدالة.

تقوم الدالة tracefs_buffer_stat_dropped_events() بإرجاع عدد الأحداث التي تم إسقاطها إذا تم تعطيل وضع الكتابة فوق. ستظهر الأحداث التي فقدت بسبب وصول الكاتب إلى القارئ وعدم القدرة على كتابة المزيد من الأحداث.

تقوم الدالة tracefs_buffer_stat_read_events() بإرجاع عدد الأحداث التي تم استهلاكها بواسطة قارئ.

قيمة الإرجاع

تقوم الدالة tracefs_instance_get_stat() بإرجاع بنية tracefs_buffer_stat التي يمكن استخدامها لاسترداد الإحصائيات عبر الدوال الأخرى. يجب تحريرها باستخدام tracefs_instance_put_stat().

الدوال الأخرى التي ترجع قيمًا مختلفة من بنية tracefs_buffer_stat ترجع جميعها القيمة، أو -1 إذا لم يتم العثور على القيمة.

مثال

#include <stdlib.h>
#include <unistd.h>
#include <tracefs.h>
int main(int argc, char **argv)
{
        char *trace;
        char buf[1000];
        int ret;
        int i;
        for (i = 0; i < sizeof(buf) - 1; i++) {
                buf[i] = '0' + i % 10;
        }
        buf[i] = '\0';
        tracefs_instance_clear(NULL);
        for (i = 0; i < 4; i++) {
                ret = tracefs_printf(NULL, "%s\n", buf);
                if (ret < 0)
                        perror("write");
        }
        trace = tracefs_instance_file_read(NULL, "trace", NULL);
        printf("%s\n", trace);
        free(trace);
        for (i = 0; i < sysconf(_SC_NPROCESSORS_CONF); i++) {
                struct tracefs_buffer_stat *tstat;
                ssize_t entries, eread;
                tstat = tracefs_instance_get_stat(NULL, i);
                if (!tstat)
                        continue;
                entries = tracefs_buffer_stat_entries(tstat);
                eread = tracefs_buffer_stat_read_events(tstat);
                if (!entries && !eread) {
                        tracefs_instance_put_stat(tstat);
                        continue;
                }
                printf("CPU: %d\n", i);;
                printf("\tentries: %zd\n", entries);
                printf("\toverrun: %zd\n", tracefs_buffer_stat_overrun(tstat));
                printf("\tcommit_overrun: %zd\n", tracefs_buffer_stat_commit_overrun(tstat));
                printf("\tbytes: %zd\n", tracefs_buffer_stat_bytes(tstat));
                printf("\tevent_timestamp: %lld\n", tracefs_buffer_stat_event_timestamp(tstat));
                printf("\ttimestamp: %lld\n", tracefs_buffer_stat_timestamp(tstat));
                printf("\tdropped_events: %zd\n", tracefs_buffer_stat_dropped_events(tstat));
                printf("\tread_events: %zd\n", eread);
                tracefs_instance_put_stat(tstat);
        }
}
