الوصف

مكرر التتبع على الأحداث الأولية.

ستقوم الدالة tracefs_iterate_raw_events() بقراءة مخازن بيانات tracefs الأولية واستدعاء الدالة callback المحددة لكل حدث تصادفه. يتم تكرار الأحداث بترتيب مرتب: الأقدم أولاً. مطلوب معالج tep مهيأ (انظر tracefs_local_events(3)). إذا كان instance فارغًا (NULL)، فسيتم استخدام المخزن المؤقت tracefs الأعلى، وإلا فسيتم قراءة المخزن المؤقت للمثيل instance المقابل. لتصفية مجموعة فرعية من وحدات المعالجة المركزية فقط، يمكن تعيين cpus و cpu_size لاستدعاء callback فقط مع الأحداث التي وقعت على وحدات المعالجة المركزية المحددة، وإلا إذا كان cpus فارغًا (NULL) فسيتم استدعاء الدالة callback لجميع الأحداث، ويتم تجاهل cpu_size. سيتم استدعاء الدالة callback بالمعاملات التالية: مؤشر إلى هيكل tep_event الذي يتوافق مع نوع الحدث؛ السجل الذي يمثل الحدث؛ وحدة المعالجة المركزية التي وقع عليها الحدث؛ ومؤشر إلى callback_context محدد من قبل المستخدم. إذا أعادت callback قيمة غير صفرية، يتوقف التكرار.

تعمل الدالة tracefs_iterate_snapshot_events() بنفس طريقة عمل tracefs_iterate_raw_events() باستثناء أنها تعمل على المخزن المؤقت للقطة.

استخدم tracefs_iterate_stop() لإجبار tracefs_iterate_raw_events() قيد التنفيذ على التوقف. يمكن استدعاء هذا من رد اتصال يتم استدعاؤه بواسطة المكرر (على الرغم من أن القيمة غير الصفرية ستوقفه)، أو من خيط آخر.

تُستخدم الدالة tracefs_follow_event() مع tracefs_iterate_raw_events() ولكن بدلاً من استدعاء رد الاتصال لكل حدث، يتم استدعاؤه فقط للنظام system / اسم الحدث event_name المحدد للدالة. رد الاتصال callback هو نفسه الخاص بـ tracefs_iterate_raw_events()، وسيتم تمرير callback_context المدخل أيضًا إلى رد الاتصال callback. ملاحظة، إذا أعادت شيئًا آخر غير 0، فسوف توقف الحلقة قبل استدعاء رد الاتصال callback الخاص بـ tracefs_iterate_raw_events().

ستقوم الدالة tracefs_follow_missed_events() باستدعاء callback عند اكتشاف أحداث تم تخطيها. ستقوم بتعيين المعامل record لـ callback إلى السجل الذي جاء بعد الأحداث التي تم تخطيها، وسيكون event من نوع الحدث الذي ينتمي إليه record. سيتم تعيين cpu إلى وحدة المعالجة المركزية التي تخطت الأحداث، و callback_data سيكون المحتوى الذي تم تمريره إلى الدالة.

ستقوم الدالة tracefs_follow_event_clear() بإزالة المتابعين من instance الذين يتطابقون مع system و event_name. إذا كان system و event_name كلاهما فارغين (NULL)، فسيتم إزالة جميع المتابعين للأحداث المرتبطة بـ instance. إذا كان system فقط فارغًا (NULL)، فسيتم إزالة جميع المتابعين الذين يتابعون الأحداث التي تطابق event_name. إذا كان event_name فقط فارغًا (NULL)، فسيتم إزالة جميع المتابعين المتصلين بالأحداث التي تنتمي إلى نظام يتطابق مع system.

ستقوم الدالة tracefs_follow_missed_events_clear() بإزالة جميع المتابعين للأحداث التي تم تخطيها.

قيمة الإرجاع

ترجع الدالة tracefs_iterate_raw_events() القيمة -1 في حالة حدوث خطأ أو 0 بخلاف ذلك.

تقوم كل من tracefs_follow_event_clear() و tracefs_follow_missed_events_clear() بإرجاع 0 عند النجاح و -1 عند حدوث خطأ، أو إذا لم تجد متابعين مطابقين وكان يجب إزالتهم.

مثال

#include <unistd.h>
#include <tracefs.h>
#include <stdbool.h>
#include <signal.h>
struct my_struct {
        bool            stopped;
};
#define MAX_COUNT 500000
static int counter;
static int callback(struct tep_event *event, struct tep_record *record,
                    int cpu, void *data)
{
        struct my_struct *my_data = data;
        static struct trace_seq seq;
        if (counter++ > MAX_COUNT) {
                my_data->stopped = true;
                return 1;
        }
        if (!seq.buffer)
                trace_seq_init(&seq);
        tep_print_event(event->tep, &seq, record, "%16s-%-5d [%03d] %6.1000d %s: %s\n",
                        TEP_PRINT_COMM, TEP_PRINT_PID, TEP_PRINT_CPU,
                        TEP_PRINT_TIME, TEP_PRINT_NAME, TEP_PRINT_INFO);
        trace_seq_terminate(&seq);
        trace_seq_do_printf(&seq);
        trace_seq_reset(&seq);
        return 0;
}
static int sched_callback(struct tep_event *event, struct tep_record *record,
                          int cpu, void *data)
{
        static struct tep_format_field *prev_pid;
        static struct tep_format_field *next_pid;
        unsigned long long pid;
        int this_pid = *(int *)data;
        if (!prev_pid) {
                prev_pid = tep_find_field(event, "prev_pid");
                next_pid = tep_find_field(event, "next_pid");
                if (!prev_pid || !next_pid) {
                        fprintf(stderr, "No pid fields??\n");
                        return -1;
                }
        }
        tep_read_number_field(prev_pid, record->data, &pid);
        if (pid == this_pid)
                printf("WE ARE LEAVING!\n");
        tep_read_number_field(next_pid, record->data, &pid);
        if (pid == this_pid)
                printf("WE ARE ARRIVING!\n");
        return 0;
}
static int missed_callback(struct tep_event *event, struct tep_record *record,
                           int cpu, void *data)
{
        printf("OOPS! cpu %d dropped ", cpu);
        if (record->missed_events > 0)
                printf("%lld ", record->missed_events);
        printf("events\n");
        return 0;
}

        my_data.stopped = true;
}

        tracefs_event_enable(instance, NULL, NULL);
