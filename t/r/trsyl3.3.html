CTRSYL3

الغرض:

!>
!>  CTRSYL3 تحل معادلة مصفوفة سيليستر المعقدة:
!>
!>     op(A)*X + X*op(B) = scale*C أو
!>     op(A)*X - X*op(B) = scale*C,
!>
!>  حيث op(A) = A أو A**H، و A و B كلاهما مثلثي علوي. A
!>  هو M*M و B هو N*N؛ الطرف الأيمن C والحل X هما
!>  M*N؛ و scale هو عامل قياس إخراج، تم تعيينه <= 1 لتجنب
!>  الفيضان في X.
!>
!>  هذه هي النسخة المقطعية للخوارزمية.
!>

المعلمات

TRANA 
!>          TRANA هو CHARACTER*1
!>          يحدد الخيار op(A):
!>          = 'N': op(A) = A    (لا يوجد تبديل)
!>          = 'C': op(A) = A**H (تبديل المرافق)
!>

TRANB

!>          TRANB هو CHARACTER*1
!>          يحدد الخيار op(B):
!>          = 'N': op(B) = B    (لا يوجد تبديل)
!>          = 'C': op(B) = B**H (تبديل المرافق)
!>

ISGN

!>          ISGN هو INTEGER
!>          يحدد الإشارة في المعادلة:
!>          = +1: حل op(A)*X + X*op(B) = scale*C
!>          = -1: حل op(A)*X - X*op(B) = scale*C
!>

M

!>          M هو INTEGER
!>          ترتيب المصفوفة A، وعدد الصفوف في
!>          المصفوفتين X و C. M >= 0.
!>

N

!>          N هو INTEGER
!>          ترتيب المصفوفة B، وعدد الأعمدة في
!>          المصفوفتين X و C. N >= 0.
!>

A

!>          A هي مصفوفة COMPLEX، dimension (LDA,M)
!>          المصفوفة المثلثية العلوية A.
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرئيسي للمصفوفة A. LDA >= max(1,M).
!>

B

!>          B هي مصفوفة COMPLEX، dimension (LDB,N)
!>          المصفوفة المثلثية العلوية B.
!>

LDB

!>          LDB هو INTEGER
!>          البعد الرئيسي للمصفوفة B. LDB >= max(1,N).
!>

C

!>          C هي مصفوفة COMPLEX، dimension (LDC,N)
!>          عند الإدخال، مصفوفة الطرف الأيمن C بحجم M*N.
!>          عند الإخراج، يتم استبدال C بمصفوفة الحل X.
!>

LDC

!>          LDC هو INTEGER
!>          البعد الرئيسي للمصفوفة C. LDC >= max(1,M)
!>

SCALE

!>          SCALE هو REAL
!>          عامل القياس، scale، تم تعيينه <= 1 لتجنب الفيضان في X.
!>

SWORK

!>          SWORK هي مصفوفة REAL، dimension (MAX(2, ROWS), MAX(1,COLS)).
!>          عند الإخراج، إذا كان INFO = 0، فإن SWORK(1) يعيد قيمة ROWS
!>          المثلى و SWORK(2) يعيد قيمة COLS المثلى.
!>

LDSWORK

!>          LDSWORK هو INTEGER
!>          LDSWORK >= MAX(2,ROWS)، حيث ROWS = ((M + NB - 1) / NB + 1)
!>          و NB هو حجم المقطع الأمثل.
!>
!>          إذا كان LDSWORK = -1، فيفترض استعلام عن مساحة العمل؛
!>          يقوم الروتين فقط بحساب الأبعاد المثلى لمصفوفة SWORK،
!>          يعيد هذه القيم كأول وثاني مدخل لمصفوفة SWORK،
!>          ولا يتم إصدار أي رسالة خطأ متعلقة بـ LWORK بواسطة XERBLA.
!>

INFO

!>          INFO هو INTEGER
!>          = 0: خروج ناجح
!>          < 0: إذا كان INFO = -i، فإن الوسيط i كان له قيمة غير قانونية
!>          = 1: A و B لهما قيم ذاتية مشتركة أو قريبة جداً؛ تم استخدام
!>               قيم مضطربة لحل المعادلة (ولكن المصفوفتين
!>               A و B لم تتغير).
!>

NAME

        

            
trsyl3 — solve the Sylvester equation AX + XB = C or AX - XB = C

        

    





    

        
SYNOPSIS

        

            
SUBROUTINE ZTRSYL3( TRANS, TRANSA, TRANSB, SIDE, DIRECT, C, DC, IFFT, M, N, A, LDA, B, LDB, C, LDC, SCALE, SWORK, LDSWORK, INFO )

  CHARACTER(1) :: DIRECT, SIDE, TRANSA, TRANSB, TRANS
  COMPLEX*16 :: A( LDA, M ), B( LDB, N ), C( LDC, N )
  INTEGER :: C, DC, IFFT, INFO, LDA, LDB, LDSWORK, LDC, M, N
  INTEGER :: NB
  DOUBLE PRECISION :: SCALE
  COMPLEX*16 :: SWORK( LDSWORK, MAX( 1, COLS ) )

        

    








Description


The program ZTRSYL3 solves the Sylvester equation.




Arguments


Status








Function Documentation


M


!>          M is INTEGER
!>          The order of the matrix A, and the number of rows in the
!>          matrices X and C. M >= 0.
!> 


N


!>          N is INTEGER
!>          The order of the matrix B, and the number of columns in the
!>          matrices X and C. N >= 0.
!> 


A


!>          A is COMPLEX*16 array, dimension (LDA,M)
!>          The upper triangular matrix A.
!> 


LDA


!>          LDA is INTEGER
!>          The leading dimension of the array A. LDA >= max(1,M).
!> 


B


!>          B is COMPLEX*16 array, dimension (LDB,N)
!>          The upper triangular matrix B.
!> 


LDB


!>          LDB is INTEGER
!>          The leading dimension of the array B. LDB >= max(1,N).
!> 


C


!>          C is COMPLEX*16 array, dimension (LDC,N)
!>          On entry, the M-by-N right hand side matrix C.
!>          On exit, C is overwritten by the solution matrix X.
!> 


LDC


!>          LDC is INTEGER
!>          The leading dimension of the array C. LDC >= max(1,M)
!> 


SCALE


!>          SCALE is DOUBLE PRECISION
!>          The scale factor, scale, set <= 1 to avoid overflow in X.
!> 


SWORK


!>          SWORK is DOUBLE PRECISION array, dimension (MAX(2, ROWS),
!>          MAX(1,COLS)).
!>          On exit, if INFO = 0, SWORK(1) returns the optimal value ROWS
!>          and SWORK(2) returns the optimal COLS.
!> 


LDSWORK


!>          LDSWORK is INTEGER
!>          LDSWORK >= MAX(2,ROWS), where ROWS = ((M + NB - 1) / NB + 1)
!>          and NB is the optimal block size.
!>
!>          If LDSWORK = -1, then a workspace query is assumed; the routine
!>          only calculates the optimal dimensions of the SWORK matrix,
!>          returns these values as the first and second entry of the SWORK
!>          matrix, and no error message related LWORK is issued by XERBLA.
!> 


INFO


!>          INFO is INTEGER
!>          = 0: successful exit
!>          < 0: if INFO = -i, the i-th argument had an illegal value
!>          = 1: A and B have common or very close eigenvalues; perturbed
!>               values were used to solve the equation (but the matrices
!>               A and B are unchanged).
!>
