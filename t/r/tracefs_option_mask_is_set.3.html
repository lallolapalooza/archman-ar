tracefs_option_mask_is_set(3)

الحزمة: libtracefs

التراخيص: GPL-2.0-only, LGPL-2.1-only

LIBTRACEFS(3) دليل libtracefs LIBTRACEFS(3)

الاسم

tracefs_options_get_supported, tracefs_option_is_supported, tracefs_options_get_enabled, tracefs_option_is_enabled, tracefs_option_mask_is_set, tracefs_option_id - الحصول على خيارات ftrace والتحقق منها.

الملخص

#include <tracefs.h>
const struct tracefs_options_mask *tracefs_options_get_supported(struct tracefs_instance *instance);
bool tracefs_option_is_supported(struct tracefs_instance *instance, enum tracefs_option_id id);
const struct tracefs_options_mask *tracefs_options_get_enabled(struct tracefs_instance *instance);
bool tracefs_option_is_enabled(struct tracefs_instance *instance, enum tracefs_option_id id);
bool tracefs_option_mask_is_set(const struct tracefs_options_mask options, enum tracefs_option_id id);
enum tracefs_option_id *tracefs_option_id(const char *name);

الوصف

يمكن استخدام مجموعة واجهات برمجة التطبيقات هذه للحصول على خيارات ftrace الحالية والتحقق منها. قد تعتمد خيارات ftrace المدعومة على إصدار النواة وتكوين النواة.

تقوم الدالة tracefs_options_get_supported() بالحصول على جميع خيارات ftrace التي يدعمها النظام في instance المحدد. إذا كانت instance فارغة (NULL)، يتم إرجاع خيارات النسخة الرئيسية المدعومة. مجموعة الخيارات المدعومة هي نفسها في جميع نسخ التتبع المنشأة، ولكنها قد تختلف عن نسخة التتبع الرئيسية.

تقوم الدالة *tracefs_option_is_supported()/ بالتحقق مما إذا كان الخيار ذو id المحدد مدعومًا من قبل النظام في instance المحدد. إذا كانت instance فارغة (NULL)، يتم استخدام نسخة التتبع الرئيسية. إذا كان الخيار مدعومًا في نسخة التتبع الرئيسية، فقد لا يكون مدعومًا في نسخة تتبع تم إنشاؤها.

تقوم الدالة tracefs_options_get_enabled() بالحصول على جميع خيارات ftrace، الممكّنة حاليًا في instance المحدد. إذا كانت instance فارغة (NULL)، يتم إرجاع الخيارات الممكّنة لنسخة التتبع الرئيسية.

تقوم الدالة tracefs_option_is_enabled() بالتحقق مما إذا كان الخيار ذو id المحدد ممكّنًا في instance المحدد. إذا كانت instance فارغة (NULL)، يتم استخدام نسخة التتبع الرئيسية.

تقوم الدالة tracefs_option_mask_is_set() بالتحقق مما إذا كانت البتة، المقابلة للخيار ذي id، مضبوطة في قناع البتات options الذي تم إرجاعه من tracefs_option_get_enabled() و tracefs_option_is_supported().

تقوم الدالة tracefs_option_id() بتحويل name الخيار إلى id المقابل له، إذا تم العثور عليه. هذا يسمح بالعثور على id الخيار لاستخدامه في الدوال الأخرى إذا كان name فقط هو المعروف.

قيمة الإرجاع

تقوم الدالتان tracefs_options_get_supported() و tracefs_options_get_enabled()، عند النجاح، بإرجاع مؤشر إلى قناع البتات داخل instance، أو قناع بتات عام للمستوى الأعلى، أو NULL في حالة حدوث خطأ. نظرًا لأن قناع البتات المُرجع جزء من بنية instance (أو متغير عام) ولا يجب تحريره أو تعديله.

ترجع الدالتان tracefs_option_is_supported() و tracefs_option_is_enabled() القيمة true إذا كان الخيار مدعومًا / ممكّنًا، أو false بخلاف ذلك.

ترجع الدالة tracefs_option_mask_is_set() القيمة true إذا كانت البتة المقابلة مضبوطة في القناع أو false بخلاف ذلك.

ترجع الدالة tracefs_option_id() id المقابل المحدد بواسطة tracefs_options(3) من name المحدد. إذا لم يتم العثور على name، فسيتم إرجاع TRACEFS_OPTION_INVALID.

مثال

#include <tracefs.h>
...
const struct tracefs_options_mask *options;
...
options = tracefs_options_get_supported(NULL);
if (!options) {
        /* فشل الحصول على الخيارات المدعومة */
} else {
        ...
}
...
options = tracefs_options_get_enabled(NULL);
if (!options) {
        /* فشل الحصول على الخيارات، الممكّنة في النسخة الرئيسية */
} else {
        ...
}
if (tracefs_options_mask_is_set(options, TRACEFS_OPTION_LATENCY_FORMAT)) {
        ...
}
...
if (tracefs_option_is_supported(NULL, TRACEFS_OPTION_LATENCY_FORMAT)) {
        /* خيار تنسيق الكمون مدعوم */
}
...
if (tracefs_option_is_enabled(NULL, TRACEFS_OPTION_STACKTRACE)) {
        /* خيار تتبع المكدس ممكّن في النسخة الرئيسية */
}
