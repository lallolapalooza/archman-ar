tracefs_event_filter_apply(3)

الحزمة: libtracefs

التراخيص: GPL-2.0-only, LGPL-2.1-only

LIBTRACEFS(3) دليل libtracefs LIBTRACEFS(3)

الاسم

tracefs_filter_string_append، tracefs_filter_string_verify، tracefs_event_filter_apply، tracefs_event_filter_clear - إضافة، التحقق من، وتطبيق مرشحات الأحداث

الملخص

#include <tracefs.h>
int tracefs_filter_string_append(struct tep_event *event, char **filter,
                                 struct tracefs_filter type, const char *field,
                                 enum tracefs_synth_compare compare, const char *val);
int tracefs_filter_string_verify(struct tep_event *event, const char *filter, char **err);
int tracefs_event_filter_apply(struct tracefs_instance *instance, struct tep_event *event, const char *filter);
int tracefs_event_filter_clear(struct tracefs_instance *instance, struct tep_event *event);

الوصف

tracefs_filter_string_append() هي طريقة لإنشاء والتحقق من مرشحات الأحداث لحدث معين. ستقوم بالتحقق من أن field تنتمي إلى الحدث وأن خيار compare المستخدم صحيح لنوع الحقل، وكذلك val. بالنسبة لنوع type الذي ليس من نوع TRACEFS_FILTER_COMPARE، ستقوم ببناء سلسلة منطقية والتأكد أيضًا من أن الصيغة صحيحة. على سبيل المثال، لا يوجد عدد أقواس مغلقة أكثر من أقواس مفتوحة، ولا يتم وضع AND (&&) أو OR (||) في غير مكانه، إلخ.

tracefs_synth_append_start_filter() تنشئ مرشحًا أو تضيف إليه لحدث البدء. اعتمادًا على type، ستقوم ببناء سلسلة من الرموز للأقواس أو العبارات المنطقية، أو قد تضيف مقارنة للحقل field إلى val بناءً على compare.

إذا كان type هو: TRACEFS_FILTER_COMPARE - انظر أدناه TRACEFS_FILTER_AND - إضافة "&&" إلى المرشح TRACEFS_FILTER_OR - إضافة "||" إلى المرشح TRACEFS_FILTER_NOT - إضافة "!" إلى المرشح TRACEFS_FILTER_OPEN_PAREN - إضافة "(" إلى المرشح TRACEFS_FILTER_CLOSE_PAREN - إضافة ")" إلى المرشح

يتم تجاهل field و compare و val ما لم يكن type يساوي TRACEFS_FILTER_COMPARE، عندها سيتم استخدام _compare لما يلي:

TRACEFS_COMPARE_EQ - field == val

TRACEFS_COMPARE_NE - field != val

TRACEFS_COMPARE_GT - field > val

TRACEFS_COMPARE_GE - field >= val

TRACEFS_COMPARE_LT - field < val

TRACEFS_COMPARE_LE - field <= val

TRACEFS_COMPARE_RE - field ~ "val" : حيث field هو سلسلة نصية.

TRACEFS_COMPARE_AND - field & val : حيث field هو حقل بتات (flags).

tracefs_filter_string_verify() تقوم بتحليل filter للتأكد من أن الحقول تنتمي إلى event، وأن الصيغة صحيحة. إذا كان هناك خطأ في الصيغة، ولم يكن err فارغًا (NULL)، فسيتم تخصيصه برسالة خطأ توضح ما هو الخطأ في المرشح. يجب تحرير err باستخدام free().

tracefs_event_filter_apply() تطبق سلسلة المرشح filter المعطاة على event في المثيل instance المحدد.

tracefs_event_filter_clear() تمسح جميع المرشحات على event في المثيل instance المحدد.

قيمة الإرجاع

tracefs_filter_string_append() ترجع 0 عند النجاح و -1 عند الخطأ.

tracefs_filter_string_verify() ترجع 0 عند النجاح و -1 عند الخطأ. إذا كان هناك خطأ، ولم يكن errno هو ENOMEM، فسيتم تخصيص err وسيحمل سلسلة نصية تصف ما هو الخطأ الذي تم العثور عليه في filter. يجب تحرير err باستخدام free().

tracefs_event_filter_apply() ترجع 0 عند النجاح و -1 عند الخطأ.

tracefs_event_filter_clear() ترجع 0 عند النجاح و -1 عند الخطأ.

مثال

#include <stdlib.h>
#include <stdio.h>
#include <ctype.h>
#include <tracefs.h>
static void usage(char **argv)
{
        fprintf(stderr, "usage: %s [system] event filter\n", argv[0]);
        exit(-1);
}
int main (int argc, char **argv)
{
        struct tep_handle *tep;
        struct tep_event *event;
        const char *system = NULL;
        const char *event_name;
        const char *filter;
        char *new_filter = NULL;
        char *err = NULL;
        int i;
        if (argc < 3)
                usage(argv);
        if (argc < 4) {
                event_name = argv[1];
                filter = argv[2];
        } else {
                system = argv[1];
                event_name = argv[2];
                filter = argv[3];
        }
        /* Load all events from the system */
        tep = tracefs_local_events(NULL);
        if (!tep) {
                perror("tep");
                exit(-1);
        }
        event = tep_find_event_by_name(tep, system, event_name);
        if (!event) {
                fprintf(stderr, "Event %s%s%s not found\n",
                        system ? system : "" , system ? " " : "",
                        event_name);
                exit(-1);
        }
        if (tracefs_filter_string_verify(event, filter, &err) < 0) {
                perror("tracecfs_event_verify_filter");
                if (err)
                        fprintf(stderr, "%s", err);
                free(err);
                exit(-1);
        }
        for (i = 0; filter[i]; i++) {
                char buf[strlen(filter)];
                char *field = NULL;
                char *val = NULL;
                enum tracefs_filter type;
                enum tracefs_compare compare = 0;
                int start_i, n;
                int quote;
                bool backslash;
                while (isspace(filter[i]))
                        i++;
                switch(filter[i]) {
                case '(':
                        type = TRACEFS_FILTER_OPEN_PAREN;
                        break;
                case ')':
                        type = TRACEFS_FILTER_CLOSE_PAREN;
                        break;
                case '!':
                        type = TRACEFS_FILTER_NOT;
                        break;
                case '&':
                        type = TRACEFS_FILTER_AND;
                        i++;
                        break;
                case '|':
                        type = TRACEFS_FILTER_OR;
                        i++;
                        break;
                default:
                        type = TRACEFS_FILTER_COMPARE;
                        while (isspace(filter[i]))
                                i++;
                        start_i = i;
                        for (; filter[i]; i++) {
                                switch(filter[i]) {
                                case 'a' ... 'z':
                                case 'A' ... 'Z':
                                case '0' ... '9':
                                case '_':
                                        continue;
                                }
                                break;
                        }
                        n = i - start_i;
                        field = buf;
                        strncpy(field, filter + start_i, n);
                        field[n++] = '\0';
                        val = buf + n;
                        while (isspace(filter[i]))
                                i++;
                        start_i = i;
                        switch(filter[i++]) {
                        case '>':
                                compare = TRACEFS_COMPARE_GT;
                                if (filter[i] == '=') {
                                        i++;
                                        compare = TRACEFS_COMPARE_GE;
                                }
                                break;
                        case '<':
                                compare = TRACEFS_COMPARE_LT;
                                if (filter[i] == '=') {
                                        i++;
                                        compare = TRACEFS_COMPARE_LE;
                                }
                                break;
                        case '=':
                                compare = TRACEFS_COMPARE_EQ;
                                i++;
                                break;
                        case '!':
                                compare = TRACEFS_COMPARE_NE;
                                i++;
                                break;
                        case '~':
                                compare = TRACEFS_COMPARE_RE;
                                break;
