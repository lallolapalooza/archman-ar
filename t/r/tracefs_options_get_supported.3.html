tracefs_options_get_supported(3)

الحزمة: libtracefs

التراخيص: GPL-2.0-only, LGPL-2.1-only

LIBTRACEFS(3) دليل libtracefs LIBTRACEFS(3)

الاسم

tracefs_options_get_supported, tracefs_option_is_supported, tracefs_options_get_enabled, tracefs_option_is_enabled, tracefs_option_mask_is_set, tracefs_option_id - الحصول على خيارات ftrace والتحقق منها.

الملخص

#include <tracefs.h>
const struct tracefs_options_mask *tracefs_options_get_supported(struct tracefs_instance *instance);
bool tracefs_option_is_supported(struct tracefs_instance *instance, enum tracefs_option_id id);
const struct tracefs_options_mask *tracefs_options_get_enabled(struct tracefs_instance *instance);
bool tracefs_option_is_enabled(struct tracefs_instance *instance, enum tracefs_option_id id);
bool tracefs_option_mask_is_set(const struct tracefs_options_mask options, enum tracefs_option_id id);
enum tracefs_option_id *tracefs_option_id(const char *name);

الوصف

يمكن استخدام مجموعة واجهات برمجة التطبيقات هذه للحصول على خيارات ftrace الحالية والتحقق منها. قد تعتمد خيارات ftrace المدعومة على إصدار النواة وتكوين النواة.

تقوم الدالة tracefs_options_get_supported() بالحصول على جميع خيارات ftrace التي يدعمها النظام في instance المحدد. إذا كانت instance فارغة (NULL)، فسيتم إرجاع خيارات النسخة الأساسية (top trace instance) المدعومة. مجموعة الخيارات المدعومة هي نفسها في جميع النسخ الأساسية التي تم إنشاؤها، ولكن قد تختلف عن النسخة الأساسية.

تقوم الدالة tracefs_option_is_supported() بالتحقق مما إذا كان الخيار ذو id المحدد مدعومًا من قبل النظام في instance المحدد. إذا كانت instance فارغة (NULL)، يتم استخدام النسخة الأساسية. إذا كان خيار ما مدعومًا في النسخة الأساسية، فقد لا يكون مدعومًا في نسخة أساسية تم إنشاؤها.

تقوم الدالة tracefs_options_get_enabled() بالحصول على جميع خيارات ftrace الممكنة حاليًا في instance المحدد. إذا كانت instance فارغة (NULL)، فسيتم إرجاع خيارات النسخة الأساسية الممكنة.

تقوم الدالة tracefs_option_is_enabled() بالتحقق مما إذا كان الخيار ذو id المحدد ممكنًا في instance المحدد. إذا كانت instance فارغة (NULL)، يتم استخدام النسخة الأساسية.

تقوم الدالة tracefs_option_mask_is_set() بالتحقق مما إذا كانت البتة المقابلة للخيار ذي id محددة في قناع البتات options الذي تم إرجاعه من tracefs_option_get_enabled() و tracefs_option_is_supported().

تقوم الدالة tracefs_option_id() بتحويل name الخيار إلى id المقابل له، إذا تم العثور عليه. هذا يسمح بالعثور على id الخيار لاستخدامه في الدوال الأخرى إذا كان name فقط هو المعروف.

قيمة الإرجاع

تقوم الدوال tracefs_options_get_supported() و tracefs_options_get_enabled()، عند النجاح، بإرجاع مؤشر إلى قناع البتات داخل النسخة، أو قناع بتات عام للطبقة العليا، أو NULL في حالة حدوث خطأ. نظرًا لأن قناع البتات الذي تم إرجاعه هو جزء من هيكل النسخة (أو متغير عام) ويجب عدم تحريره أو تعديله.

تقوم الدوال tracefs_option_is_supported() و tracefs_option_is_enabled() بإرجاع true إذا كان الخيار مدعومًا / ممكنًا، أو false بخلاف ذلك.

تقوم الدالة tracefs_option_mask_is_set() بإرجاع true إذا كان الخيار المقابل محددًا في القناع أو false بخلاف ذلك.

تقوم الدالة tracefs_option_id() بإرجاع id المقابل المحدد بواسطة tracefs_options(3) من name المحدد. إذا لم يتم العثور على name، فسيتم إرجاع TRACEFS_OPTION_INVALID.

مثال

#include <tracefs.h>
...
const struct tracefs_options_mask *options;
...
options = tracefs_options_get_supported(NULL);
if (!options) {
        /* فشل في الحصول على الخيارات المدعومة */
} else {
        ...
}
...
options = tracefs_options_get_enabled(NULL);
if (!options) {
        /* فشل في الحصول على الخيارات الممكنة في النسخة الأساسية */
} else {
        ...
}
if (tracefs_options_mask_is_set(options, TRACEFS_OPTION_LATENCY_FORMAT)) {
        ...
}
...
if (tracefs_option_is_supported(NULL, TRACEFS_OPTION_LATENCY_FORMAT)) {
        /* خيار تنسيق التأخير مدعوم */
}
...
if (tracefs_option_is_enabled(NULL, TRACEFS_OPTION_STACKTRACE)) {
        /* خيار تتبع المكدس ممكن في النسخة الأساسية */
}
