tracefs_instance_alloc(3)

الحزمة: libtracefs

التراخيص: GPL-2.0-only, LGPL-2.1-only

LIBTRACEFS(3) دليل libtracefs LIBTRACEFS(3)

الاسم

tracefs_instance_create, tracefs_instance_destroy, tracefs_instance_alloc, tracefs_instance_free, tracefs_instance_is_new, tracefs_instances, tracefs_instance_clear, tracefs_instance_reset - إدارة مثيلات التتبع.

الملخص

#include <tracefs.h>
struct tracefs_instance *tracefs_instance_create(const char *name);
int tracefs_instance_destroy(struct tracefs_instance *instance);
struct tracefs_instance *tracefs_instance_alloc(const char *tracing_dir, const char *name);
void tracefs_instance_free(struct tracefs_instance *instance);
bool tracefs_instance_is_new(struct tracefs_instance *instance);
char **tracefs_instances(const char *regex);
void tracefs_instance_clear(struct tracefs_instance *instance);
void tracefs_instance_reset(struct tracefs_instance *instance);

الوصف

يمكن استخدام مجموعة الوظائف هذه لإدارة مثيلات التتبع. مثيل التتبع هو مخزن مؤقت فرعي يستخدمه نظام التتبع في لينكس. بالنظر إلى اسم فريد، فإن الأحداث الممكّنة في مثيل لا تؤثر على نظام التتبع الرئيسي، ولا على المثيلات الأخرى، حيث أن الأحداث الممكّنة في نظام التتبع الرئيسي أو المثيلات الأخرى لا تؤثر على المثيل المعطى.

تقوم الوظيفة tracefs_instance_create() بتخصيص وتهيئة بنية tracefs_instance جديدة وإعادتها. إذا كان المثيل بالاسم name غير موجود بعد في النظام، فسيتم إنشاؤه. يمكن أن يكون name فارغًا (NULL)، ثم يتم تهيئة بنية tracefs_instance الجديدة للمثيل العلوي. لاحظ أنه لا يمكن إنشاء المثيل العلوي في النظام إذا لم يكن موجودًا.

تقوم tracefs_instance_destroy() بإزالة المثيل من النظام، ولكنها لا تحرر البنية. لا يزال يتعين استدعاء tracefs_instance_free() على instance.

تقوم الوظيفة tracefs_instance_alloc()* بتخصيص بنية tracefs_instance جديدة لمثيل تتبع موجود. إذا كان المثيل غير موجود في النظام، تفشل الوظيفة. يشير المعامل tracing_dir إلى دليل تتبع النظام. يمكن أن يكون فارغًا (NULL)، ثم يتم استخدام دليل تتبع النظام الافتراضي. هذا المعامل مفيد لتخصيص المثيلات لأدلة التتبع، التي تم نسخها من جهاز آخر. name هو اسم المثيل، أو فارغ (NULL) للمثيل العلوي في tracing_dir المعطى.

تقوم الوظيفة tracefs_instance_free() بتحرير بنية tracefs_instance، دون إزالة مثيل التتبع من النظام.

تقوم الوظيفة tracefs_instance_is_new() بالتحقق مما إذا كان instance المعطى تم إنشاؤه حديثًا بواسطة tracefs_instance_create()، أم أنه كان موجودًا في النظام قبل ذلك.

تقوم الوظيفة tracefs_instances() بإرجاع قائمة بالمثيلات الموجودة في النظام والتي تتطابق مع التعبير العادي regex. إذا كان regex فارغًا (NULL)، فسيقوم بمطابقة جميع المثيلات الموجودة. يجب تحرير القائمة المرجعة باستخدام tracefs_list_free(3). لاحظ، إذا لم يتم العثور على مثيلات، يتم إرجاع قائمة فارغة ويجب تحريرها أيضًا باستخدام tracefs_list_free(3).

تقوم الوظيفة tracefs_instance_clear() بمسح المخزن المؤقت الدائري للمثيل instance المعطى، أو المخزن المؤقت الدائري للمستوى العلوي إذا كان instance فارغًا (NULL).

تقوم الوظيفة tracefs_instance_reset() بإعادة تعيين المثيل instance المعطى إلى حالته الافتراضية.

قيمة الإرجاع

تقوم الوظيفتان tracefs_instance_create() و tracefs_instance_alloc() بإرجاع مؤشر إلى بنية tracefs_instance مخصصة حديثًا. يجب تحريرها باستخدام tracefs_instance_free().

تقوم الوظيفة tracefs_instance_destroy() بإرجاع 0 إذا نجحت في إزالة المثيل، وإلا فإنها ترجع -1 إذا كان المثيل غير موجود أو فشلت في إزالته.

تقوم الوظيفة tracefs_instance_is_new() بإرجاع true إذا كان tracefs_instance_create() الذي خصص instance قد قام أيضًا بإنشاء مثيل التتبع في النظام، أو false إذا كان مثيل التتبع موجودًا بالفعل في النظام عندما تم تخصيص instance بواسطة tracefs_instance_create() أو tracefs_instance_alloc().

تقوم الوظيفة tracefs_instances() بإرجاع قائمة بأسماء المثيلات الموجودة في النظام. يجب تحرير القائمة باستخدام tracefs_list_free(3). يتم إرجاع قائمة فارغة إذا لم يكن هناك مثيل موجود يتطابق مع regex، ويجب تحرير هذه القائمة أيضًا باستخدام tracefs_list_free(3). يتم إرجاع NULL عند حدوث خطأ.

تقوم الوظيفة tracefs_instance_clear() بإرجاع 0 إذا نجحت في مسح المخزن المؤقت الدائري، أو -1 عند حدوث خطأ.

مثال

#include <tracefs.h>
struct tracefs_instance *inst = tracefs_instance_create("foo");
        if (!inst) {
                /* خطأ في إنشاء مثيل تتبع جديد */
                ...
        }
        ...
        if (tracefs_instance_is_new(inst))
                tracefs_instance_destroy(inst);
        tracefs_instance_free(inst);
...
struct tracefs_instance *inst = tracefs_instance_alloc(NULL, "bar");
        if (!inst) {
                /* خطأ في تخصيص مثيل التتبع 'bar' */
                ...
        }
        ...
        tracefs_instance_reset(inst);
        tracefs_instance_free(inst);
