tracefs_instance_put_stat(3)

الحزمة: libtracefs

التراخيص: GPL-2.0-only, LGPL-2.1-only

LIBTRACEFS(3) دليل libtracefs LIBTRACEFS(3)

الاسم

tracefs_instance_get_stat, tracefs_instance_put_stat, tracefs_buffer_stat_entries, tracefs_buffer_stat_overrun, tracefs_buffer_stat_commit_overrun, tracefs_buffer_stat_bytes, tracefs_buffer_stat_event_timestamp, tracefs_buffer_stat_timestamp, tracefs_buffer_stat_dropped_events, tracefs_buffer_stat_read_events - التعامل مع إحصائيات مخزن التتبع المؤقت

الملخص

#include <tracefs.h>
struct tracefs_buffer_stat *tracefs_instance_get_stat(struct tracefs_instance *instance, int cpu);
void tracefs_instance_put_stat(struct tracefs_buffer_stat *tstat);
ssize_t tracefs_buffer_stat_entries(struct tracefs_buffer_stat *tstat);
ssize_t tracefs_buffer_stat_overrun(struct tracefs_buffer_stat *tstat);
ssize_t tracefs_buffer_stat_commit_overrun(struct tracefs_buffer_stat *tstat);
ssize_t tracefs_buffer_stat_bytes(struct tracefs_buffer_stat *tstat);
long long tracefs_buffer_stat_event_timestamp(struct tracefs_buffer_stat *tstat);
long long tracefs_buffer_stat_timestamp(struct tracefs_buffer_stat *tstat);
ssize_t tracefs_buffer_stat_dropped_events(struct tracefs_buffer_stat *tstat);
ssize_t tracefs_buffer_stat_read_events(struct tracefs_buffer_stat *tstat);

الوصف

تقوم هذه المجموعة من الدوال بقراءة وتحليل الملف tracefs/per_cpu/cpuX/stats. تحتوي هذه الملفات على إحصائيات المخزن المؤقت للحلقة لكل وحدة معالجة مركزية، مثل عدد الأحداث الموجودة في المخزن المؤقت للحلقة، وعدد الأحداث التي تمت قراءتها، وما إلى ذلك.

تقوم الدالة tracefs_instance_get_stat() بقراءة وتحليل ملف الإحصائيات المحدد لوحدة instance ووحدة المعالجة المركزية cpu المعطاة. نظرًا لأن المخزن المؤقت للحلقة مقسم إلى مخازن مؤقتة لكل وحدة معالجة مركزية، فإن المعلومات مرتبطة فقط بوحدة المعالجة المركزية cpu المعطاة. يمكن استخدام المؤشر tracefs_buffer_stat العائد مع دوال tracefs_buffer_stat الأخرى ويجب تحريره باستخدام tracefs_instance_put_stat().

تقوم الدالة tracefs_instance_put_stat() بتحرير الموارد المخصصة لـ stat المعطاة التي تم إنشاؤها بواسطة tracefs_instance_get_stat().

تعيد الدالة tracefs_buffer_stat_entries() عدد الأحداث الموجودة حاليًا في المخزن المؤقت للحلقة المرتبط بـ tstat.

تعيد الدالة tracefs_buffer_stat_overrun() عدد الأحداث التي فقدها كاتب المخزن المؤقت للحلقة بسبب تجاوز القارئ.

تعيد الدالة tracefs_buffer_stat_commit_overrun() عدد الأحداث التي فقدت بسبب أن المخزن المؤقت للحلقة كان صغيرًا جدًا وقاطع مقاطعة حدثًا في سياق أقل أثناء تسجيله وأضاف المزيد من الأحداث مما يمكن أن يحتويه المخزن المؤقت للحلقة. لاحظ أن هذا ليس حدثًا شائعًا وعندما يحدث فهذا يعني أن شيئًا ما لم يتم إعداده بشكل صحيح.

تعيد الدالة tracefs_buffer_stat_bytes() عدد البايتات التي تشغلها الأحداث الحالية. لاحظ أنها تشمل البيانات الوصفية للأحداث، ولكنها لا تشمل البيانات الوصفية للمخازن المؤقتة الفرعية.

تعيد الدالة tracefs_buffer_stat_event_timestamp() الطابع الزمني لآخر حدث في المخزن المؤقت للحلقة.

تعيد الدالة tracefs_buffer_stat_timestamp() الطابع الزمني الحالي للمخزن المؤقت للحلقة. لاحظ أنه يتم قراءتها فقط عند استدعاء tracefs_instance_get_stat(). سيكون لديها الطابع الزمني للمخزن المؤقت للحلقة عند استدعاء تلك الدالة.

تعيد الدالة tracefs_buffer_stat_dropped_events() عدد الأحداث التي تم إسقاطها إذا تم تعطيل وضع الكتابة فوق. ستعرض الأحداث التي فقدت بسبب أن الكاتب لحق بالقارئ ولم يتمكن من كتابة المزيد من الأحداث.

تعيد الدالة tracefs_buffer_stat_read_events() عدد الأحداث التي تم استهلاكها بواسطة القارئ.

قيمة الإرجاع

تعيد الدالة tracefs_instance_get_stat() بنية tracefs_buffer_stat التي يمكن استخدامها لاسترداد الإحصائيات عبر الدوال الأخرى. يجب تحريرها باستخدام tracefs_instance_put_stat().

الدوال الأخرى التي تعيد قيمًا مختلفة من بنية tracefs_buffer_stat تعيد كلها القيمة، أو -1 إذا لم يتم العثور على القيمة.

مثال

#include <stdlib.h>
#include <unistd.h>
#include <tracefs.h>
int main(int argc, char **argv)
{
        char *trace;
        char buf[1000];
        int ret;
        int i;
        for (i = 0; i < sizeof(buf) - 1; i++) {
                buf[i] = '0' + i % 10;
        }
        buf[i] = '\0';
        tracefs_instance_clear(NULL);
        for (i = 0; i < 4; i++) {
                ret = tracefs_printf(NULL, "%s\n", buf);
                if (ret < 0)
                        perror("write");
        }
        trace = tracefs_instance_file_read(NULL, "trace", NULL);
        printf("%s\n", trace);
        free(trace);
        for (i = 0; i < sysconf(_SC_NPROCESSORS_CONF); i++) {
                struct tracefs_buffer_stat *tstat;
                ssize_t entries, eread;
                tstat = tracefs_instance_get_stat(NULL, i);
                if (!tstat)
                        continue;
                entries = tracefs_buffer_stat_entries(tstat);
                eread = tracefs_buffer_stat_read_events(tstat);
                if (!entries && !eread) {
                        tracefs_instance_put_stat(tstat);
                        continue;
                }
                printf("CPU: %d\n", i);;
                printf("\tentries: %zd\n", entries);
                printf("\toverrun: %zd\n", tracefs_buffer_stat_overrun(tstat));
                printf("\tcommit_overrun: %zd\n", tracefs_buffer_stat_commit_overrun(tstat));
                printf("\tbytes: %zd\n", tracefs_buffer_stat_bytes(tstat));
                printf("\tevent_timestamp: %lld\n", tracefs_buffer_stat_event_timestamp(tstat));
                printf("\ttimestamp: %lld\n", tracefs_buffer_stat_timestamp(tstat));
                printf("\tdropped_events: %zd\n", tracefs_buffer_stat_dropped_events(tstat));
                printf("\tread_events: %zd\n", eread);
                tracefs_instance_put_stat(tstat);
        }
}
