tracefs_trace_off_fd(3)

الحزمة: libtracefs

التراخيص: GPL-2.0-only, LGPL-2.1-only

LIBTRACEFS(3) دليل libtracefs LIBTRACEFS(3)

الاسم

tracefs_trace_is_on, tracefs_trace_on, tracefs_trace_off, tracefs_trace_on_get_fd, tracefs_trace_on_fd, tracefs_trace_off_fd - وظائف لتمكين أو تعطيل التتبع.

الملخص

#include <tracefs.h>
int tracefs_trace_is_on(struct tracefs_instance *instance);
int tracefs_trace_on(struct tracefs_instance *instance);
int tracefs_trace_off(struct tracefs_instance *instance);
int tracefs_trace_on_get_fd(struct tracefs_instance *instance);
int tracefs_trace_on_fd(int fd);
int tracefs_trace_off_fd(int fd);

الوصف

يمكن استخدام مجموعة الوظائف هذه للتحقق من أو تمكين أو تعطيل الكتابة إلى المخزن المؤقت الدائري في مثيل التتبع المحدد. يتم تمكين التتبع عندما يتم تمكين الكتابة إلى المخزن المؤقت الدائري.

تقوم الوظيفة tracefs_trace_is_on() بالتحقق مما إذا كان التتبع ممكّنًا للمثيل instance المحدد. إذا كان instance هو NULL، يتم استخدام المثيل العلوي.

تقوم الوظيفتان tracefs_trace_on() و tracefs_trace_off() بتعيين حالة التتبع في المثيل instance إلى حالة التمكين أو التعطيل. إذا كان instance هو NULL، يتم استخدام المثيل العلوي.

تقوم الوظيفة tracefs_trace_on_get_fd() بإرجاع واصف ملف إلى ملف "tracing_on" من المثيل instance المحدد. إذا كان instance هو NULL، يتم استخدام مثيل التتبع العلوي. يمكن استخدام الواصف المرجع لتمكين أو تعطيل تتبع المثيل بسرعة.

تقوم الوظيفتان tracefs_trace_on_fd() و tracefs_trace_off_fd() بتعيين حالة التتبع إلى تمكين أو تعطيل باستخدام واصف الملف fd المحدد. يجب فتح واصف الملف هذا للكتابة باستخدام tracefs_trace_on_get_fd(3) لمثيل التتبع المطلوب. هذه الوظائف أسرع من tracefs_trace_on و tracefs_trace_off.

قيمة الإرجاع

تقوم الوظيفة tracefs_trace_is_on() بإرجاع 0 إذا كان التتبع معطلاً، 1 إذا كان ممكّنًا، أو -1 في حالة حدوث خطأ.

تقوم الوظيفة tracefs_trace_on_get_fd() بإرجاع واصف ملف إلى ملف "tracing_on" للقراءة والكتابة، والذي يجب إغلاقه باستخدام close(). في حالة حدوث خطأ، يتم إرجاع -1.

تقوم الوظائف tracefs_trace_on() و tracefs_trace_off() و tracefs_trace_on_fd() و tracefs_trace_off_fd() بإرجاع -1 في حالة حدوث خطأ أو 0 بخلاف ذلك.

مثال

#include <tracefs.h>
        int ret;
        ret = tracefs_trace_is_on(NULL);
        if (ret == 0) {
                /* التتبع معطل في المثيل العلوي */
        } else if (ret == 1) {
                /* التتبع ممكّن في المثيل العلوي */
        } else {
                /* خطأ في الحصول على حالة تتبع المثيل العلوي */
        }
        ...
        if (!tracefs_trace_on(NULL)) {
            /* تم تمكين التتبع في المثيل العلوي */
            ...
            if (!tracefs_trace_off(NULL)) {
                /* تم تعطيل التتبع في المثيل العلوي */
            } else {
                /* خطأ في تعطيل التتبع في المثيل العلوي */
            }
        } else {
                /* خطأ في تمكين التتبع في المثيل العلوي */
        }
        ...
        int fd = tracefs_trace_on_get_fd(NULL);
        if (fd < 0) {
                /* خطأ في فتح ملف tracing_on */
        }
        ...
        if (!tracefs_trace_on_fd(fd)) {
            /* تم تمكين التتبع في المثيل العلوي */
            ...
            if (!tracefs_trace_off_fd(fd)) {
                /* تم تعطيل التتبع في المثيل العلوي */
            } else {
                /* خطأ في تعطيل التتبع في المثيل العلوي */
            }
        } else {
                /* خطأ في تمكين التتبع في المثيل العلوي */
        }
        ...
        close(fd);
