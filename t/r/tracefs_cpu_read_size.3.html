tracefs_cpu_read_size(3)

الحزمة: libtracefs

التراخيص: GPL-2.0-only, LGPL-2.1-only

LIBTRACEFS(3) دليل libtracefs LIBTRACEFS(3)

الاسم

tracefs_cpu_read_size, tracefs_cpu_read, tracefs_cpu_buffered_read, tracefs_cpu_write, tracefs_cpu_stop, tracefs_cpu_flush, tracefs_cpu_flush_write, tracefs_cpu_pipe - قراءة بيانات trace_pipe_raw

المُلَخَّص

#include <tracefs.h>
int tracefs_cpu_read_size(struct tracefs_cpu *tcpu);
int tracefs_cpu_read(struct tracefs_cpu *tcpu, void *buffer, bool nonblock);
int tracefs_cpu_buffered_read(struct tracefs_cpu *tcpu, void *buffer, bool nonblock);
int tracefs_cpu_write(struct tracefs_cpu *tcpu, int wfd, bool nonblock);
int tracefs_cpu_stop(struct tracefs_cpu *tcpu);
int tracefs_cpu_flush(struct tracefs_cpu *tcpu, void *buffer);
int tracefs_cpu_flush_write(struct tracefs_cpu *tcpu, int wfd);
int tracefs_cpu_pipe(struct tracefs_cpu *tcpu, int wfd, bool nonblock);

الوصف

يمكن استخدام مجموعة واجهات برمجة التطبيقات هذه لقراءة البيانات الأولية من ملفات trace_pipe_raw في نظام ملفات tracefs.

تقوم tracefs_cpu_read_size() بإرجاع حجم المخزن المؤقت الفرعي لـ trace_pipe_raw. هذا يرجع الحد الأدنى لحجم المخزن المؤقت الذي يتم تمريره إلى الدوال أدناه.

تقوم tracefs_cpu_read() بقراءة ملفات trace_pipe_raw المرتبطة بـ tcpu إلى buffer. يجب أن يكون buffer على الأقل بحجم المخزن المؤقت الفرعي للحلقة، والذي يتم إرجاعه بواسطة tracefs_cpu_read_size(). إذا تم تعيين nonblock، ولا توجد بيانات متاحة، فسوف تعود فورًا. بخلاف ذلك، اعتمادًا على كيفية فتح tcpu، ستتوقف. إذا تم فتح tcpu مع تعيين nonblock، فلن يكون لـ nonblock هذا فرق.

الدالة tracefs_cpu_buffered_read() هي في الأساس نفس الدالة tracefs_cpu_read() باستثناء أنها تستخدم أنبوبًا عبر splice لتخزين القراءات مؤقتًا. سيؤدي هذا إلى تجميع القراءات مع إبقاء القراءة من الحلقة أقل تدخلاً في النظام، حيث أن القراءة المستمرة فقط يمكن أن تسبب اضطرابًا كبيرًا. ملاحظة، أحد الاختلافات بين هذه وبين tracefs_cpu_read() هو أنها ستقرأ فقط في صفحات المخزن المؤقت الفرعي. إذا لم تملأ الحلقة صفحة، فلن تعيد شيئًا، حتى مع تعيين nonblock. يجب إجراء استدعاءات لـ tracefs_cpu_flush() لقراءة بقية الملف في نهاية التتبع.

ستقوم tracefs_cpu_write() بتوجيه البيانات من ملف trace_pipe_raw المرتبط بـ tcpu إلى واصف الملف wfd. إذا تم تعيين nonblock، فلن تتوقف إذا لم يكن هناك شيء للكتابة. ملاحظة، ستكتب فقط حجم المخزن المؤقت الفرعي من البيانات إلى wfd. هناك حاجة إلى استدعاءات لـ tracefs_cpu_flush_write() لكتابة الباقي.

ستحاول tracefs_cpu_stop() إلغاء حظر مهمة متوقفة في tcpu تقرأها. في الأنوية القديمة، قد لا تفعل شيئًا لقراءات الأنابيب، حيث أن الأنوية القديمة لا توقظ المهام التي تنتظر على الحلقة. تعود بـ 0 إذا أيقظت بالتأكيد أي منتظرين محتملين، ولكنها تعود بـ 1 إذا لم تكن متأكدة من أنها نجحت وقد يحتاج المنتظرون إلى إرسال إشارة إليهم.

تقرأ tracefs_cpu_flush() ملف trace_pipe_raw المرتبط بـ tcpu وتضعه في buffer، والذي يجب أن يكون بحجم المخزن المؤقت الفرعي الذي يتم استرداده بواسطة tracefs_cpu_read_size(). يجب استدعاء هذا في نهاية التتبع للحصول على بقية البيانات. سيحول هذا الاستدعاء واصف ملف trace_pipe_raw إلى وضع غير مانع.

tracefs_cpu_flush_write() نفس trace_cpu_flush() باستثناء أنها تأخذ واصف ملف wfd لتفريغ البيانات إليه.

tracefs_cpu_pipe() مشابه لـ tracefs_cpu_write() ولكن يجب أن يكون واصف الملف wfd أنبوبًا. هذا الاستدعاء هو تحسين لـ tracefs_cpu_write() الذي يستخدم استدعاءين لـ splice(2) لتوصيل واصف ملف trace_pipe_raw بواصف الكتابة. تتطلب splice(2) أن يكون أحد واصفات الملفات المدخلة أنبوبًا. إذا أراد التطبيق تمرير البيانات إلى أنبوب موجود، فلا يوجد سبب لوجود استدعاءين لنظام splice(2) ويمكن لـ tracefs_cpu_pipe() ببساطة استخدام استدعاء واحد لـ wfd.

قيمة الإرجاع

تعيد tracefs_cpu_open() واصف struct tracefs_cpu يمكن استخدامه بواسطة الدوال الأخرى أو NULL في حالة الخطأ.

تعيد tracefs_cpu_read_size() الحد الأدنى لحجم المخازن المؤقتة المراد استخدامها مع tracefs_cpu_read() و tracefs_cpu_buffered_read() و tracefs_cpu_flush(). تعيد قيمة سالبة في حالة الخطأ.

تعيد tracefs_cpu_read() عدد البايتات التي تم قراءتها، أو قيمة سالبة في حالة الخطأ.

تعيد tracefs_cpu_buffered_read() عدد البايتات التي تم قراءتها أو قيمة سالبة في حالة الخطأ.

تعيد tracefs_cpu_write() عدد البايتات التي تم كتابتها إلى الملف أو قيمة سالبة في حالة الخطأ.

تعيد tracefs_cpu_stop() صفر إذا تم ضمان إيقاظ أي منتظرين من انتظار الإدخال، أو تعيد واحدًا إذا كانت هذه نواة قديمة لا توفر هذا الضمان، وقد تحتاج إلى إرسال إشارة إلى أي منتظرين. تعيد قيمة سالبة في حالة الخطأ.

تعيد tracefs_cpu_flush() عدد البايتات التي تم قراءتها أو قيمة سالبة في حالة الخطأ.

تعيد tracefs_cpu_flush_write() عدد البايتات التي تم كتابتها إلى الملف أو قيمة سالبة في حالة الخطأ.

مثال

#define _LARGEFILE64_SOURCE
#include <stdlib.h>
#include <ctype.h>
#include <pthread.h>
#include <unistd.h>
#include <tracefs.h>
struct thread_data {
        struct tracefs_cpu      *tcpu;
        int                     done;
        int                     fd;
};
static void *thread_run(void *arg)
{
        struct thread_data *data = arg;
        struct tracefs_cpu *tcpu = data->tcpu;
        int fd = data->fd;
        int ret;
        while (!data->done) {
                ret = tracefs_cpu_write(tcpu, fd, false);
                printf("wrote %d\n", ret);
        }
        return NULL;
}
int main (int argc, char **argv)
{
        struct tracefs_instance *instance;
        struct thread_data data;
        pthread_t thread;
        char *file;
        int secs = 10;
        int cpu;
        int ret;
        if (argc < 3 || !isdigit(argv[1][0])) {
                printf("usage: %s cpu file_destination [sleep secs]\n\n", argv[0]);
                exit(-1);
        }
        cpu = atoi(argv[1]);
        file = argv[2];
        if (argc > 3)
                secs = atoi(argv[3]);
        instance = tracefs_instance_create("cpu_write");
        if (!instance) {
                perror("create instance");
                exit(-1);
        }
        memset(&data, 0, sizeof(data));
        data.tcpu = tracefs_cpu_open(instance, cpu, 0);
        if (!data.tcpu) {
                perror("Open instance");
                exit(-1);
        }
        data.fd = open(file, O_WRONLY | O_CREAT | O_TRUNC | O_LARGEFILE, 0644);
        if (data.fd < 0) {
                perror(file);
                exit(-1);
        }
        pthread_create(&thread, NULL, thread_run, &data);
        sleep(secs);
        data.done = 1;
        printf("stopping\n");
        ret = tracefs_cpu_stop(data.tcpu);
        printf("joining %d\n", ret);
        pthread_join(thread, NULL);
        tracefs_trace_off(instance);
        do {
                ret = tracefs_cpu_flush_write(data.tcpu, data.fd);
                printf("flushed %d\n", ret);
        } while (ret > 0);
        tracefs_trace_on(instance);
        tracefs_cpu_close(data.tcpu);
        close(data.fd);
        return 0;
}
