subroutine ctrsna (character job, character howmny, logical, dimension( * ) select, integer n, complex, dimension( ldt, * ) t, integer ldt, complex, dimension( ldvl, * ) vl, integer ldvl, complex, dimension( ldvr, * ) vr, integer ldvr, real, dimension( * ) s, real, dimension( * ) sep, integer mm, integer m, complex, dimension( ldwork, * ) work, integer ldwork, real, dimension( * ) rwork, integer info)

CTRSNA

الغرض:

!>
!> CTRSNA تقدر الأرقام الشرطية العكسية للقيم الذاتية المحددة
!> و/أو المتجهات الذاتية اليمنى لمصفوفة مثلثية علوية معقدة T (أو لأي مصفوفة Q*T*Q**H مع Q وحدوية).
!>

المعلمات

JOB 
!>          JOB is CHARACTER*1
!>          يحدد ما إذا كانت الأرقام الشرطية مطلوبة للقيم الذاتية
!>          (S) أو المتجهات الذاتية (SEP):
!>          = 'E': للقيم الذاتية فقط (S);
!>          = 'V': للمتجهات الذاتية فقط (SEP);
!>          = 'B': لكليهما (S و SEP).
!>

HOWMNY

!>          HOWMNY is CHARACTER*1
!>          = 'A': حساب الأرقام الشرطية لجميع الأزواج الذاتية;
!>          = 'S': حساب الأرقام الشرطية للأزواج الذاتية المحددة
!>                 المحددة بالمصفوفة SELECT.
!>

SELECT

!>          SELECT is LOGICAL array, dimension (N)
!>          إذا كان HOWMNY = 'S'، فإن SELECT تحدد الأزواج الذاتية التي
!>          تُطلب لها الأرقام الشرطية. لتحديد الأرقام الشرطية
!>          للزوج الذاتي j، يجب تعيين SELECT(j) إلى .TRUE..
!>          إذا كان HOWMNY = 'A'، فلا يتم الرجوع إلى SELECT.
!>

N

!>          N is INTEGER
!>          رتبة المصفوفة T. N >= 0.
!>

T

!>          T is COMPLEX array, dimension (LDT,N)
!>          المصفوفة المثلثية العلوية T.
!>

LDT

!>          LDT is INTEGER
!>          البعد الرئيسي للمصفوفة T. LDT >= max(1,N).
!>

VL

!>          VL is COMPLEX array, dimension (LDVL,M)
!>          إذا كان JOB = 'E' أو 'B'، يجب أن تحتوي VL على المتجهات الذاتية اليسرى لـ T
!>          (أو لأي Q*T*Q**H مع Q وحدوية)، المقابلة للأزواج الذاتية
!>          المحددة بواسطة HOWMNY و SELECT. يجب تخزين المتجهات الذاتية
!>          في أعمدة متتالية في VL، كما تم إرجاعها بواسطة
!>          CHSEIN أو CTREVC.
!>          إذا كان JOB = 'V'، فلا يتم الرجوع إلى VL.
!>

LDVL

!>          LDVL is INTEGER
!>          البعد الرئيسي للمصفوفة VL.
!>          LDVL >= 1; وإذا كان JOB = 'E' أو 'B'، فإن LDVL >= N.
!>

VR

!>          VR is COMPLEX array, dimension (LDVR,M)
!>          إذا كان JOB = 'E' أو 'B'، يجب أن تحتوي VR على المتجهات الذاتية اليمنى لـ T
!>          (أو لأي Q*T*Q**H مع Q وحدوية)، المقابلة للأزواج الذاتية
!>          المحددة بواسطة HOWMNY و SELECT. يجب تخزين المتجهات الذاتية
!>          في أعمدة متتالية في VR، كما تم إرجاعها بواسطة
!>          CHSEIN أو CTREVC.
!>          إذا كان JOB = 'V'، فلا يتم الرجوع إلى VR.
!>

LDVR

!>          LDVR is INTEGER
!>          البعد الرئيسي للمصفوفة VR.
!>          LDVR >= 1; وإذا كان JOB = 'E' أو 'B'، فإن LDVR >= N.
!>

S

!>          S is REAL array, dimension (MM)
!>          إذا كان JOB = 'E' أو 'B'، فإن الأرقام الشرطية العكسية للقيم
!>          الذاتية المحددة، مخزنة في عناصر متتالية للمصفوفة.
!>          وبالتالي فإن S(j)، و SEP(j)، والأعمدة j من VL و VR
!>          تقابل جميعها نفس الزوج الذاتي (ولكن ليس بالضرورة
!>          الزوج الذاتي j، إلا إذا تم تحديد جميع الأزواج الذاتية).
!>          إذا كان JOB = 'V'، فلا يتم الرجوع إلى S.
!>

SEP

!>          SEP is REAL array, dimension (MM)
!>          إذا كان JOB = 'V' أو 'B'، فإن الأرقام الشرطية العكسية المقدرة
!>          للمتجهات الذاتية المحددة، مخزنة في عناصر متتالية
!>          للمصفوفة.
!>          إذا كان JOB = 'E'، فلا يتم الرجوع إلى SEP.
!>

MM

!>          MM is INTEGER
!>          عدد العناصر في المصفوفات S (إذا كان JOB = 'E' أو 'B')
!>           و/أو SEP (إذا كان JOB = 'V' أو 'B'). MM >= M.
!>

M

!>          M is INTEGER
!>          عدد العناصر في المصفوفات S و/أو SEP المستخدمة فعليًا
!>          لتخزين الأرقام الشرطية المقدرة.
!>          إذا كان HOWMNY = 'A'، يتم تعيين M إلى N.
!>

WORK

!>          WORK is COMPLEX array, dimension (LDWORK,N+6)
!>          إذا كان JOB = 'E'، فلا يتم الرجوع إلى WORK.
!>

LDWORK

!>          LDWORK is INTEGER
!>          البعد الرئيسي للمصفوفة WORK.
!>          LDWORK >= 1; وإذا كان JOB = 'V' أو 'B'، فإن LDWORK >= N.
!>

RWORK

!>          RWORK is REAL array, dimension (N)
!>          إذا كان JOB = 'E'، فلا يتم الرجوع إلى RWORK.
!>

INFO

!>          INFO is INTEGER
!>          = 0: إنهاء ناجح
!>          < 0: إذا كان INFO = -i، فإن المعلمة i كانت لها قيمة غير قانونية
!>

تفاصيل إضافية:

!>
!>  يتم تعريف معكوس الرقم الشرطي لقيمة ذاتية lambda على أنه
!>
!>          S(lambda) = |v**H*u| / (norm(u)*norm(v))
!>
!>  حيث u و v هما المتجهان الذاتيان الأيمن والأيسر لـ T المقابلان
!>  لـ lambda؛ v**H يمثل منقولة v المترافقة، و norm(u)
!>  يمثل المعيار الإقليدي. هذه الأرقام الشرطية العكسية تقع دائمًا
!>  بين صفر (مشروط بشكل سيء للغاية) وواحد (مشروط بشكل جيد للغاية).
!>  إذا كان n = 1، يتم تعريف S(lambda) على أنها 1.
!>
!>  يتم إعطاء حد خطأ تقريبي لقيمة ذاتية محسوبة W(i) بواسطة
!>
!>                      EPS * norm(T) / S(i)
!>
!>  حيث EPS هي دقة الآلة.
!>
!>  يتم تعريف معكوس الرقم الشرطي للمتجه الذاتي الأيمن u
!>  المقابل لـ lambda على النحو التالي. افترض
!>
!>              T = ( lambda  c  )
!>                  (   0    T22 )
!>
!>  ثم يكون الرقم الشرطي العكسي هو
!>
!>          SEP( lambda, T22 ) = sigma-min( T22 - lambda*I )
!>
!>  حيث sigma-min يمثل أصغر قيمة مفردة. نقوم بتقدير
!>  أصغر قيمة مفردة عن طريق معكوس تقدير للمعيار الواحد
!>  لمعكوس T22 - lambda*I. إذا كان n = 1، يتم تعريف SEP(1)
!>  على أنها abs(T(1,1)).
!>
!>  يتم إعطاء حد خطأ تقريبي للمتجه الذاتي الأيمن المحسوب VR(i)
!>  بواسطة
!>
!>                      EPS * norm(T) / SEP(i)
!>
