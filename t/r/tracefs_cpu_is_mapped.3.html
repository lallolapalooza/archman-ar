NAME

tracefs_cpu_is_mapped - check if a CPU is mapped

SYNOPSIS

#include <libtracefs.h>

bool tracefs_cpu_is_mapped( struct tracefs_context *context, int cpu);

DESCRIPTION

The function tracefs_cpu_is_mapped() checks if the given cpu is mapped by the tracefs(3) context.

RETURN VALUE

Returns true if the CPU is mapped, false otherwise.

EXAMPLE

This example checks if CPU 0 is mapped by the default tracefs context.

#include <stdio.h>

#include <stdlib.h>

#include <tracefs.h>

int main(int argc, char **argv) {

struct tracefs_context *context;

context = tracefs_open_default();

if (!context) {

perror("tracefs_open_default");

return 1;

}

if (tracefs_cpu_is_mapped(context, 0)) {

printf("CPU 0 is mapped.\n");

} else {

printf("CPU 0 is not mapped.\n");

}

tracefs_close(context);

return 0;

}

الاسم

tracefs_cpu_is_mapped - التحقق مما إذا كانت وحدة المعالجة المركزية mapped

الملخص

#include <libtracefs.h>

bool tracefs_cpu_is_mapped( struct tracefs_context *context, int cpu);

الوصف

الدالة tracefs_cpu_is_mapped() تتحقق مما إذا كانت وحدة المعالجة المركزية cpu المعطاة mapped بواسطة سياق tracefs(3).

قيمة الإرجاع

تُرجع true إذا كانت وحدة المعالجة المركزية mapped، وإلا فإنها تُرجع false.

مثال

هذا المثال يتحقق مما إذا كانت وحدة المعالجة المركزية 0 mapped بواسطة سياق tracefs الافتراضي.

#include <stdio.h>

#include <stdlib.h>

#include <tracefs.h>

int main(int argc, char **argv) {

struct tracefs_context *context;

context = tracefs_open_default();

if (!context) {

perror("tracefs_open_default");

return 1;

}

if (tracefs_cpu_is_mapped(context, 0)) {

printf("CPU 0 is mapped.\n");

} else {

printf("CPU 0 is not mapped.\n");

}

tracefs_close(context);

return 0;

}
