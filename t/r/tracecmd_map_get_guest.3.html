tracecmd_map_get_guest(3)

الحزمة: trace-cmd

التراخيص: GPL-2.0-only, LGPL-2.1-only

LIBTRACECMD(3) دليل libtracefs LIBTRACECMD(3)

الاسم

tracecmd_map_vcpus, tracecmd_get_cpu_map, tracecmd_map_find_by_host_pid, tracecmd_map_get_host_pid, tracecmd_map_get_guest, tracecmd_map_set_private, tracecmd_map_get_private - تعيين بيانات المضيف والضيف

الموجز

#include <trace-cmd.h>
int tracecmd_map_vcpus(struct tracecmd_input **handles, int nr_handles);
struct tracecmd_cpu_map *tracecmd_get_cpu_map(struct tracecmd_input *handle, int cpu);
struct tracecmd_cpu_map *tracecmd_map_find_by_host_pid(struct tracecmd_input *handle,
                                                      int host_pid);
int tracecmd_map_get_host_pid(struct tracecmd_cpu_map *map);
struct tracecmd_input *tracecmd_map_get_guest(struct tracecmd_cpu_map *map);
void tracecmd_map_set_private(struct tracecmd_cpu_map *map, void *priv);
void *tracecmd_map_get_private(struct tracecmd_cpu_map *map);

الوصف

تُستخدم مجموعة الواجهات هذه لتعيين ملفات تتبع المضيف والضيف لتسهيل تحليل التتبع بشكل أكبر.

تأخذ الدالة tracecmd_map_vcpus() مصفوفة من handles حيث تم إنشاء كل عنصر في تلك المصفوفة بواسطة إحدى دوال tracecmd_open(3)، وعدد handles كرقم nr_handles. يُتوقع أن يكون المقبض الأول في مصفوفة handles هو الواصف لملف تتبع المضيف، والباقي هي ملفات تتبع الضيف التي تعمل على المضيف، وتم إنشاؤها بواسطة تفاعلات trace-cmd record(1) و trace-cmd agent(1). تعيد عدد الضيوف الموجودين في handles الذين تم ربطهم بالمضيف، أو قيمة سالبة عند حدوث خطأ.

تعيد الدالة tracecmd_get_cpu_map() واصفًا لوحدة معالجة مركزية (CPU) معينة لمقبض. إذا كان handle ضيفًا تم تعريفه بواسطة tracecmd_map_vcpus()، فإن التعيين الذي تم إنشاؤه من تلك الدالة والمرتبط بـ vCPU المحدد هذا (يُشار إليه بـ cpu) من handle. يمكن استخدام هذا الواصف بواسطة الدوال tarcecmd_map_get_guest() و tracecmd_map_set_private() و tracecmd_map_get_private().

ستعيد الدالة tracecmd_map_find_by_host_pid() تعيينًا لوحدة معالجة مركزية افتراضية (vCPU) للضيف يتم التعامل معها بواسطة host_pid المعطى. ملاحظة، يمكن أن يكون handle المقدم إما مقبض المضيف أو أحد مقابض الضيف لهذا المضيف الذي تم تعيينه بواسطة tracecmd_map_vcpus()، حتى لو لم يكن مقبض الضيف يحتوي على vCPU الذي يمثله host_pid.

ستعيد الدالة tracecmd_map_get_host_pid() معرف المضيف host_pid لـ map معين تم استرجاعه بواسطة إحدى الدوال أعلاه.

ستعيد الدالة tracecmd_map_get_guest() مقبض الضيف guest_handle لـ map معين تم استرجاعه بواسطة إحدى الدوال أعلاه.

تسمح الدالة tracecmd_map_set_private() للتطبيق بتعيين بيانات خاصة لتعيين خيط المضيف vCPU الضيف المحدد بواسطة map.

تسترجع الدالة tracecmd_map_get_private() بيانات priv من map التي تم تعيينها بواسطة tracecmd_map_set_private().

قيمة الإرجاع

تعيد الدالة tracecmd_map_vcpus() عدد الضيوف في مصفوفة handles الذين تم تعيينهم لمقبض المضيف الذي هو الإدخال الأول في handles. تعيد -1 عند حدوث خطأ.

تعيد الدالة tracecmd_get_cpu_map() تعيينًا تم إنشاؤه بواسطة tracecmd_map_vcpus() لـ cpu معين لـ handle معين، أو NULL إذا لم يتم العثور عليه.

تعيد الدالة tracecmd_map_find_by_host_pid() تعيينًا مرتبطًا بمهمة المضيف بالمعرف host_pid الخاص بها. يمكن أن يكون handle إما مقبض المضيف، أو أحد مقابض الضيف التي تم تعيينها للمضيف عبر tracecmd_map_vcpus()، حتى لو كان مقبض الضيف ضيفًا آخر غير الضيف الذي يخصه التعيين. تعيد NULL إذا لم يتم العثور عليه.

تعيد الدالة tracecmd_map_get_host_pid() معرف عملية المضيف للتعيين المرتبط المحدد بواسطة map.

تعيد الدالة tracecmd_map_get_guest() مقبض الضيف للتعيين المرتبط المحدد بواسطة map.

تعيد الدالة tracecmd_map_get_private() البيانات الخاصة للتعيين المحدد بواسطة map والتي تم تعيينها بواسطة tracecmd_map_set_private().

مثال

#include <stdlib.h>
#include <errno.h>
#include <trace-cmd.h>
int main(int argc, char **argv)
{
        struct tracecmd_input **handles = NULL;
        int nr_handles;
        int i;
        if (argc < 2) {
                printf("usage: host_trace.dat guest1_trace.dat [guest2_trace.dat ...]\n");
                exit(-1);
        }
        for (i = 1; i < argc; i++) {
                handles = realloc(handles, sizeof(*handles) * (nr_handles + 1));
                if (!handles)
                        exit(-1);
                handles[nr_handles] = tracecmd_open(argv[i], 0);
                if (!handles[nr_handles]) {
                        perror(argv[1]);
                        exit(-1);
                }
                tracecmd_set_private(handles[nr_handles], argv[i]);
                nr_handles++;
        }
        tracecmd_map_vcpus(handles, nr_handles);
        for (i = 1; i < nr_handles; i++) {
                struct tracecmd_cpu_map *map;
                struct tep_handle *tep;
                const char *file = tracecmd_get_private(handles[i]);
                int cpus, cpu;
                printf("Mappings for guest %s:\n", file);
                tep = tracecmd_get_tep(handles[i]);
                cpus = tep_get_cpus(tep);
                for (cpu = 0; cpu < cpus; cpu++) {
                        printf("  [%03d] ", cpu);
                        map = tracecmd_get_cpu_map(handles[i], cpu);
                        if (!map) {
                                printf("Has no mapping!\n");
                                continue;
                        }
                        printf("host_pid: %d\n", tracecmd_map_get_host_pid(map));
                }
        }
        for (i = 0; i < nr_handles; i++)
                tracecmd_close(handles[i]);
        free(handles);
        exit(0);
}
