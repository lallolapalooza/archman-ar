tracefs_synth_get_event(3)

الحزمة: libtracefs

التراخيص: GPL-2.0-only, LGPL-2.1-only

LIBTRACEFS(3) دليل libtracefs LIBTRACEFS(3)

الاسم

tracefs_synth_echo_cmd, tracefs_synth_get_start_hist, tracefs_synth_get_name, tracefs_synth_raw_fmt, tracefs_synth_show_event, tracefs_synth_show_start_hist, tracefs_synth_show_end_hist, tracefs_synth_get_event - استرداد بيانات الأحداث الاصطناعية.

الملخص

#include <tracefs.h>
int tracefs_synth_echo_cmd(struct trace_seq *seq, struct tracefs_synth *synth);
struct tracefs_hist *tracefs_synth_get_start_hist(struct tracefs_synth *synth);
const char *tracefs_synth_get_name(struct tracefs_synth *synth);
int tracefs_synth_raw_fmt(struct trace_seq *seq, struct tracefs_synth *synth);
const char *tracefs_synth_show_event(struct tracefs_synth *synth);
const char *tracefs_synth_show_start_hist(struct tracefs_synth *synth);
const char *tracefs_synth_show_end_hist(struct tracefs_synth *synth);
struct tep_event *tracefs_synth_get_event(struct tep_handle *tep, struct tracefs_synth *synth);

الوصف

الأحداث الاصطناعية هي أحداث ديناميكية يتم إنشاؤها عن طريق مطابقة حدثين آخرين يؤديان إلى تشغيل حدث اصطناعي. أحد الأحداث هو الحدث البادئ الذي يتم فيه تسجيل حقل ما، وعندما يتم تنفيذ الحدث الثاني، إذا كان يحتوي على حقل (أو حقول) يطابق حقل (أو حقول) الحدث البادئ، فسيؤدي ذلك إلى تشغيل الحدث الاصطناعي. قد يتم تمرير قيم الحقول بخلاف الحقول المطابقة من الحدث البادئ إلى الحدث الختامي لإجراء عمليات حسابية عليها، أو ببساطة لتمريرها كمعلمة للحدث الاصطناعي.

أحد حالات الاستخدام الشائعة هو تعيين "sched_waking" كالحدث البادئ. يتم تشغيل هذا الحدث عند إيقاظ عملية. ثم يتم تعيين "sched_switch" كالحدث الختامي. يتم تشغيل هذا الحدث عند جدولة مهمة جديدة على وحدة المعالجة المركزية. من خلال تعيين "common_pid" لكلا الحدثين كحقول مطابقة، يعتبر الوقت بين الحدثين زمن الاستيقاظ المتأخر لهذه العملية. استخدم TRACEFS_TIMESTAMP كحقل لكلا الحدثين لحساب الفرق بالنانوثانية، أو استخدم TRACEFS_TIMESTAMP_USECS كحقول مقارنة لكلا الحدثين لحساب الفرق بالميكروثانية. يتم استخدام هذا كمثال أدناه.

انظر tracefs_synth_alloc(3) لتخصيص الأحداث الاصطناعية، و tracefs_synth_create() لإنشاء الحدث الاصطناعي على النظام.

tracefs_synth_echo_cmd() يعمل مثل tracefs_synth_create()، ولكن بدلاً من إنشاء الحدث الاصطناعي في النظام، فإنه سيكتب أوامر الصدى لإنشائه يدوياً في seq المعطى.

tracefs_synth_get_start_hist() يعيد واصف struct tracefs_hist يصف الرسم البياني المستخدم لإنشاء الحدث الاصطناعي.

enum tracefs_synth_handler {
        TRACEFS_SYNTH_HANDLE_MATCH,
        TRACEFS_SYNTH_HANDLE_MAX,
        TRACEFS_SYNTH_HANDLE_CHANGE,
};

tracefs_synth_get_name() يعيد اسم الحدث الاصطناعي أو NULL في حالة الخطأ. تنتمي السلسلة النصية المعادة إلى كائن حدث synth ويتم تحريرها مع الحدث بواسطة tracefs_synth_free().

tracefs_synth_raw_fmt() يكتب سلاسل التنسيق الأولية (الحدث الديناميكي والرسوم البيانية) للحدث الاصطناعي في seq المعطى.

tracefs_synth_show_event() يعيد تنسيق الحدث الديناميكي المستخدم بواسطة الحدث الاصطناعي أو NULL في حالة الخطأ. تنتمي السلسلة النصية المعادة إلى كائن حدث synth ويتم تحريرها مع الحدث بواسطة tracefs_synth_free().

tracefs_synth_show_start_hist() يعيد تنسيق الرسم البياني البادئ المستخدم بواسطة الحدث الاصطناعي أو NULL في حالة الخطأ. تنتمي السلسلة النصية المعادة إلى كائن حدث synth ويتم تحريرها مع الحدث بواسطة tracefs_synth_free().

tracefs_synth_show_end_hist() يعيد تنسيق الرسم البياني الختامي المستخدم بواسطة الحدث الاصطناعي أو NULL في حالة الخطأ. تنتمي السلسلة النصية المعادة إلى كائن حدث synth ويتم تحريرها مع الحدث بواسطة tracefs_synth_free().

تقوم الدالة tracefs_synth_get_event() بإعادة حدث tep، يصف الحدث الاصطناعي المعطى. تكتشف الواجهة البرمجية أي أحداث ديناميكية تم إنشاؤها أو إزالتها حديثاً. المؤشر المعاد إلى حدث tep يتم التحكم فيه بواسطة @tep ويجب عدم تحريره.

قيمة الإرجاع

tracefs_synth_get_name() و tracefs_synth_show_event() و tracefs_synth_show_start_hist() و tracefs_synth_show_end_hist() تعيد سلسلة نصية مملوكة لكائن حدث synth.

تقوم الدالة tracefs_synth_get_event() بإعادة مؤشر إلى حدث tep أو NULL في حالة حدوث خطأ أو إذا كان الحدث الاصطناعي المطلوب مفقوداً. يتم التحكم في المؤشر المعاد إلى حدث tep بواسطة @tep ويجب عدم تحريره.

جميع الدوال الأخرى تعيد صفراً عند النجاح أو -1 عند حدوث خطأ.

الأخطاء

الأخطاء التالية لجميع الاستدعاءات المذكورة أعلاه:

EPERM لم يتم التشغيل كمستخدم جذر عند الحاجة.

EINVAL إما أن يكون المعلمة غير صالحة (NULL عندما لا يجب أن تكون) أو حقلاً غير متوافق مع العمليات الحسابية.

ENODEV لم يتم العثور على حدث أو أحد حقوله.

EBADE حقول الأحداث البادئة والختامية غير متوافقة إما للمطابقة أو للمقارنة.

ENOMEM لا تتوفر ذاكرة كافية.

وقد تحدث أخطاء أخرى من استدعاءات النظام إلى النظام.

مثال

انظر tracefs_sql(3) لاستخدام أعمق لبعض هذا الكود.

#include <stdlib.h>
#include <tracefs.h>
#define start_event "sched_waking"
#define start_field "pid"
#define end_event "sched_switch"
#define end_field "next_pid"
#define match_name "pid"
static struct tracefs_synth *synth;
static void make_event(void)
{
        struct tep_handle *tep;
        /* تحميل جميع الأحداث من النظام */
        tep = tracefs_local_events(NULL);
        /* تهيئة الحدث الاصطناعي */
        synth = tracefs_synth_alloc(tep, "wakeup_lat",
                                    NULL, start_event,
                                    NULL, end_event,
                                    start_field, end_field,
                                    match_name);
        /* لم تعد هناك حاجة إلى tep */
        tep_free(tep);
        /* حفظ حقل "prio" باسم "prio" من الحدث البادئ */
        tracefs_synth_add_start_field(synth, "prio", NULL);
        /* حفظ حقل "next_comm" باسم "comm" من الحدث الختامي */
        tracefs_synth_add_end_field(synth, "next_comm", "comm");
        /* حفظ حقل "prev_prio" باسم "prev_prio" من الحدث الختامي */
        tracefs_synth_add_end_field(synth, "prev_prio", NULL);
        /*
         * أخذ فرق زمني بالميكروثانية بين النهاية والبداية
         * وتسجيله باسم "delta"
         */
        tracefs_synth_add_compare_field(synth, TRACEFS_TIMESTAMP_USECS,
                                        TRACEFS_TIMESTAMP_USECS,
                                        TRACEFS_SYNTH_DELTA_END, "delta");
        /* تسجيل فقط إذا كان حقل الحدث البادئ "prio" أقل من 100 */
        tracefs_synth_append_start_filter(synth, TRACEFS_FILTER_COMPARE,
                                          "prio", TRACEFS_COMPARE_LT, "100");
        /*
         * تسجيل فقط إذا كان حقل الحدث الختامي "next_prio" أقل من 50
         * أو لم يكن أولوية المهمة السابقة أكبر من أو تساوي 100.
         *   next_prio < 50 || !(prev_prio >= 100)
         */
        tracefs_synth_append_end_filter(synth, TRACEFS_FILTER_COMPARE,
                                        "next_prio", TRACEFS_COMPARE_LT, "50");
        tracefs_synth_append_end_filter(synth, TRACEFS_FILTER_OR, NULL, 0, NULL);
        tracefs_synth_append_end_filter(synth, TRACEFS_FILTER_NOT, NULL, 0, NULL);
        tracefs_synth_append_end_filter(synth, TRACEFS_FILTER_OPEN_PAREN, NULL, 0, NULL);
        tracefs_synth_append_end_filter(synth, TRACEFS_FILTER_COMPARE,
                                        "prev_prio", TRACEFS_COMPARE_GE, "100");
        /*
         * ملاحظة، تم إضافة ما يلي فقط: "next_prio < 50 || !(prev_prio >= 100"
         * وذلك لأنه عند تنفيذ synth، سيتم إضافة القوس الختامي المتبقي.
         * أي، ستنتهي السلسلة النصية بـ:
         * "next_prio < 50 || !(prev_prio >= 100)" عند تشغيل أحد tracefs_sync_create()
         * أو tracefs_sync_echo_cmd().
         */
}
