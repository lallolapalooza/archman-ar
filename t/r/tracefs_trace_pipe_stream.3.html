tracefs_trace_pipe_stream(3)

الحزمة: libtracefs

التراخيص: GPL-2.0-only, LGPL-2.1-only

LIBTRACEFS(3) دليل libtracefs LIBTRACEFS(3)

الاسم

tracefs_trace_pipe_stream, tracefs_trace_pipe_print, tracefs_trace_pipe_stop - إعادة توجيه تدفق بيانات التتبع إلى مخرج أو المخرج القياسي.

الملخص

#include <tracefs.h>
ssize_t tracefs_trace_pipe_stream(int fd, struct tracefs_instance *instance, int flags);
ssize_t tracefs_trace_pipe_print(struct tracefs_instance *instance, int flags);
void tracefs_trace_pipe_stop(struct tracefs_instance *instance);

الوصف

إذا تم تمرير NULL كـ instance، يتم استخدام المثيل الأعلى للتتبع.

يمكن إيقاف قراءة ملف trace_pipe عن طريق استدعاء tracefs_trace_pipe_stop()، والذي يمكن وضعه في معالج إشارة في حال رغب التطبيق في إيقاف القراءة، على سبيل المثال، عن طريق ضغط المستخدم على Ctrl-C.

تقوم الدالة tracefs_trace_pipe_stream() بإعادة توجيه تدفق بيانات التتبع إلى ملف مخرج. يتم استخدام استدعاء النظام "splice" لنقل البيانات دون نسخ بين مساحة عنوان النواة ومساحة عنوان المستخدم. fd هو واصف الملف لملف المخرج و flags هو قناع بت من الأعلام التي سيتم تمريرها إلى استدعاء النظام open لملف trace_pipe (انظر open(2)). إذا كانت flags تحتوي على O_NONBLOCK، فسيتم تمرير ذلك أيضًا إلى استدعاءات splice التي قد تقرأ الملف إلى واصف تدفق المخرج. ملاحظة، يتم "OR" O_RDONLY إلى flags وفقط O_NONBLOCK مفيد لهذه المعلمة.

الدالة tracefs_trace_pipe_print() مشابهة لـ tracefs_trace_pipe_stream()، ولكن يتم إعادة توجيه تدفق بيانات التتبع إلى المخرج القياسي (stdout).

قيمة الإرجاع

الدوال tracefs_trace_pipe_stream() و tracefs_trace_pipe_print() تُرجع عدد البايتات المنقولة إذا كانت العملية ناجحة، أو -1 في حالة حدوث خطأ.

مثال

#include <stdio.h>
#include <stdlib.h>
#include <unistd.h>
#include <signal.h>
#include <fcntl.h>
#include <tracefs.h>
void stop(int sig)
{
        tracefs_trace_pipe_stop(NULL);
}
int main(int argc, char **argv)
{
        mode_t mode = S_IRUSR | S_IWUSR | S_IRGRP | S_IROTH;
        const char *filename;
        int fd;
        int ret;
        if (argc < 2) {
                fprintf(stderr, "usage: %s output_file\n", argv[0]);
                exit(-1);
        }
        filename = argv[1];
        fd = creat(filename, mode);
        if (fd < 0) {
                perror(filename);
                exit(-1);
        }
        signal(SIGINT, stop);
        ret = tracefs_trace_pipe_stream(fd, NULL, SPLICE_F_NONBLOCK);
        close(fd);
        return ret;
}
