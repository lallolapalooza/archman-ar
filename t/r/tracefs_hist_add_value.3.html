tracefs_hist_add_value(3)

الحزمة: libtracefs

التراخيص: GPL-2.0-only، LGPL-2.1-only

الاسم

tracefs_hist_alloc, tracefs_hist_alloc_2d, tracefs_hist_alloc_nd, tracefs_hist_alloc_nd_cnt, tracefs_hist_free, tracefs_hist_add_key, tracefs_hist_add_key_cnt, tracefs_hist_add_value - إنشاء وتدمير رسوم الأحداث البيانية

الوصف

يتم إنشاء رسوم الأحداث البيانية بواسطة ملف المشغّل في دليل الحدث. قد تكون الصيغة معقدة ويصعب الحصول عليها بشكل صحيح. تتعامل واجهة البرمجة هذه مع الصيغة، وتسهّل إنشاء رسوم الأحداث البيانية والتفاعل معها. راجع https://www.kernel.org/doc/html/latest/trace/histogram.html لمزيد من المعلومات.

تُخصّص tracefs_hist_alloc() واصف "struct tracefs_hist" لرسم بياني أحادي البعد وتعيد عنوانه. يجب تحرير هذا الواصف بواسطة tracefs_hist_free(). يمثل tep مقبض حدث تتبع (راجع tracefs_local_events(3))، الذي يحمل system وevent اللذين سيُربط بهما الرسم البياني. system هو نظام أو مجموعة الحدث. event هو الحدث الذي سيُربط به الرسم البياني. key هو حقل من الحدث سيُستخدم كمفتاح (بُعد) للرسم البياني. type هو نوع key. راجع أنواع المفاتيح أدناه.

تُخصّص tracefs_hist_alloc_2d() واصف "struct tracefs_hist" لرسم بياني ثنائي الأبعاد وتعيد عنوانه. يجب تحرير هذا الواصف بواسطة tracefs_hist_free(). يمثل tep مقبض حدث تتبع (راجع tracefs_local_events(3))، الذي يحمل system وevent اللذين سيُربط بهما الرسم البياني. system هو نظام أو مجموعة الحدث. event هو الحدث الذي سيُربط به الرسم البياني. key1 هو الحقل الأول من الحدث الذي سيُستخدم كمفتاح (بُعد) للرسم البياني. type1 هو نوع key1. راجع أنواع المفاتيح أدناه. key2 هو الحقل الثاني من الحدث الذي سيُستخدم كمفتاح (بُعد) للرسم البياني. type2 هو نوع key2. راجع أنواع المفاتيح أدناه.

تُخصّص tracefs_hist_alloc_nd() واصف "struct tracefs_hist" لرسم بياني متعدد الأبعاد (N) وتعيد عنوانه. يجب تحرير هذا الواصف بواسطة tracefs_hist_free(). يمثل tep مقبض حدث تتبع (راجع tracefs_local_events(3))، الذي يحمل system وevent اللذين سيُربط بهما الرسم البياني. system هو نظام أو مجموعة الحدث. event هو الحدث الذي سيُربط به الرسم البياني. axes هو مصفوفة من أزواج key / type، تحدد أبعاد الرسم البياني.

ستُهيّئ tracefs_hist_alloc_nd_cnt() واصف رسم بياني سيُربط بـ system/event. هذا يهيّئ الواصف فقط بمفاتيح axes المعطاة كأساسية. هذا يهيّئ الواصف فقط، ولا يبدأ الرسم البياني في النواة. الفرق بين هذا وtracefs_hist_alloc_nd() هو أن معامل axes من نوع struct tracefs_hist_axis_cnt وليس struct tracefs_hist_axis.

تُحرّر tracefs_hist_free() واصف tracefs_hist. ملاحظة، لا توقف أو تعطّل الرسم البياني قيد التشغيل إذا كان قد بدأ. يجب استدعاء tracefs_hist_destroy() للقيام بذلك.

تضيف tracefs_hist_add_key() مفتاحًا ثانويًا أو ثلاثيًا إلى الرسم البياني. المفتاح الممرر إلى tracefs_hist_alloc_nd() هو المفتاح الأساسي للرسم البياني. في المرة الأولى التي تُستدعى فيها هذه الدالة، ستضيف مفتاحًا ثانويًا (أو رسمًا بيانيًا ثنائي الأبعاد). إذا استُدعيت هذه الدالة مرة أخرى على نفس الرسم البياني، فستضيف مفتاحًا ثلاثيًا (أو رسمًا بيانيًا ثلاثي الأبعاد). معامل hist هو واصف الرسم البياني لإضافة key إليه. type هو نوع المفتاح المراد إضافته (راجع أنواع المفاتيح أدناه).

tracefs_hist_add_key_cnt() هي نفس tracefs_hist_add_key() باستثناء أنها تسمح بإضافة عدّاد للنوع المعطى. حاليًا، يوجد فقط نوع buckets الذي يتطلب عدّادًا. عند إضافة مفتاح من نوع buckets، لا تكفي tracefs_hist_add_key()، لأن نوع buckets يتطلب عدّادًا أو حجم دلو. استخدم tracefs_hist_add_key_cnt() عند تحديد مفتاح مقسّم إلى دلاء، ومرّر حجم تلك الدلاء في cnt.

ستضيف tracefs_hist_add_value() قيمة للتسجيل. افتراضيًا، القيمة هي ببساطة "hitcount" لعدد المرات التي تم فيها إصابة نسخة من مفتاح الرسم البياني. hist هو واصف الرسم البياني لإضافة القيمة إليه. value هو حقل في الرسم البياني لإضافته عند إصابة نسخة من المفتاح.

أنواع المفاتيح

تضيف كل من tracefs_hist_alloc_nd() وtracefs_hist_add_key() مفتاحًا وتتطلب أن يكون لهذا المفتاح نوع. قد تكون الأنواع:

TRACEFS_HIST_KEY_NORMAL أو صفر (0) وهو عدم تعديل النوع.

TRACEFS_HIST_KEY_HEX لعرض المفتاح بالنظام الست عشري.

TRACEFS_HIST_KEY_SYM لعرض المفتاح كرمز للنواة (إذا وُجد). إذا كان المفتاح عنوانًا، فهذا مفيد لأنه سيعرض أسماء الدوال بدلاً من مجرد رقم.

TRACEFS_HIST_KEY_SYM_OFFSET مشابه لـ TRACEFS_HIST_KEY_SYM لكنه سيتضمن أيضًا إزاحة الدالة لمطابقة العنوان الدقيق.

TRACEFS_HIST_KEY_SYSCALL إذا كان المفتاح هو "id" لاستدعاء نظام (الرقم الممرر من مساحة المستخدم إلى النواة لإخبارها باستدعاء النظام الذي تستدعيه)، فسيُعرض اسم استدعاء النظام.

TRACEFS_HIST_KEY_EXECNAME إذا كان "common_pid" هو المفتاح (معرّف العملية للمهمة المنفذة)، فبدلاً من عرض الرقم، اعرض اسم المهمة قيد التشغيل.

سيعرض TRACEFS_HIST_KEY_LOG المفتاح بمقياس لوغاريتمي ثنائي.

TRACEFS_HIST_KEY_USECS للاستخدام مع "common_timestamp" أو TRACEFS_HIST_TIMESTAMP، وفي هذه الحالة سيعرض الطابع الزمني بالميكروثانية بدلاً من النانوثانية.

قيمة الإرجاع

تُعيد tracefs_hist_alloc_nd() واصف رسم بياني مخصصًا يجب تحريره بواسطة tracefs_hist_free() أو NULL عند حدوث خطأ.

جميع الدوال الأخرى تُعيد صفرًا عند النجاح أو -1 عند حدوث خطأ.

إذا أعادت tracefs_hist_start() خطأً، فقد تُعرض رسالة في النواة يمكن استرجاعها بواسطة tracefs_error_last().

مثال

#include <stdlib.h>
#include <ctype.h>
#include <unistd.h>
#include <tracefs.h>
enum commands {
        START,
        PAUSE,
        CONT,
        RESET,
        DELETE,
        SHOW,
};
static void parse_system_event(char *group, char **system, char **event)
{
        *system = strtok(group, "/");
        *event = strtok(NULL, "/");
        if (!*event) {
                *event = *system;
                *system = NULL;
        }
}
static int parse_keys(char *keys, struct tracefs_hist_axis_cnt **axes)
{
        char *sav = NULL;
        char *key;
        int cnt = 0;
        for (key = strtok_r(keys, ",", &sav); key; key = strtok_r(NULL, ",", &sav)) {
                struct tracefs_hist_axis_cnt *ax;
                char *att;
                ax = realloc(*axes, sizeof(*ax) * (cnt + 2));
                if (!ax) {
                        perror("Failed to allocate axes");
                        exit(-1);
                }
                ax[cnt].key = key;
                ax[cnt].type = 0;
                ax[cnt + 1].key = NULL;
                ax[cnt + 1].type = 0;
                *axes = ax;
                att = strchr(key, '.');
                if (att) {
                        *att++ = '\0';
                        if (strcmp(att, "hex") == 0)
                                ax[cnt].type = TRACEFS_HIST_KEY_HEX;
                        else if (strcmp(att, "sym") == 0)
                                ax[cnt].type = TRACEFS_HIST_KEY_SYM;
                        else if (strcmp(att, "sym_offset") == 0)
                                ax[cnt].type = TRACEFS_HIST_KEY_SYM_OFFSET;
                        else if (strcmp(att, "syscall") == 0)
                                ax[cnt].type = TRACEFS_HIST_KEY_SYSCALL;
                        else if (strcmp(att, "exec") == 0)
                                ax[cnt].type = TRACEFS_HIST_KEY_EXECNAME;
                        else if (strcmp(att, "log") == 0)
                                ax[cnt].type = TRACEFS_HIST_KEY_LOG;
                        else if (strcmp(att, "usecs") == 0)
                                ax[cnt].type = TRACEFS_HIST_KEY_USECS;
                        else if (strncmp(att, "buckets", 7) == 0) {
                                if (att[7] != '=' && !isdigit(att[8])) {
                                        fprintf(stderr, "'buckets' key type requires '=<size>'\n");
                                        exit(-1);
                                }
                                ax[cnt].type = TRACEFS_HIST_KEY_BUCKETS;
                                ax[cnt].cnt = atoi(&att[8]);
                        } else {
                                fprintf(stderr, "Undefined attribute '%s'\n", att);
                                fprintf(stderr,"  Acceptable attributes:\n");
                                fprintf(stderr,"    hex, sym, sym_offset, syscall, exe, log, usecs\n");
                                exit(-1);
                        }
                }
                cnt++;
        }
        return cnt;
}
static void process_hist(enum commands cmd, const char *instance_name,
                         char *group, char *keys, char *vals, char *sort,
                         char *ascend, char *desc)
{
        struct tracefs_instance *instance = NULL;
        struct tracefs_hist *hist;
        struct tep_handle *tep;
        struct tracefs_hist_axis_cnt *axes = NULL;
        char *system;
        char *event;
        char *sav;
        char *val;
        int ret;
        int cnt;
        if (instance_name) {
                instance = tracefs_instance_create(instance_name);
                if (!instance) {
                        fprintf(stderr, "Failed instance create\n");
                        exit(-1);
                }
        }
        tep = tracefs_local_events(NULL);
        if (!tep) {
                perror("Could not read events");
                exit(-1);
        }
        parse_system_event(group, &system, &event);
        if (cmd == SHOW) {
                char *content;
                content = tracefs_event_file_read(instance, system, event,
                                                  "hist", NULL);
                if (!content) {
                        perror("Reading hist file");
                        exit(-1);
                }
                printf("%s\n", content);
                free(content);
                return;
        }
        if (!keys) {
                fprintf(stderr, "Command needs -k option\n");
                exit(-1);
        }
        cnt = parse_keys(keys, &axes);
        if (!cnt) {
                fprintf(stderr, "No keys??\n");
                exit(-1);
        }
        /* buckets require the nd_cnt function */
        switch (cnt) {
        case 2:
                if (axes[1].type == TRACEFS_HIST_KEY_BUCKETS)
                        cnt = -1;
                /* fall through */
        case 1:
                if (axes[0].type == TRACEFS_HIST_KEY_BUCKETS)
                        cnt = -1;
        }
        /* Show examples of hist1d and hist2d */
        switch (cnt) {
        case 1:
                hist = tracefs_hist_alloc(tep, system, event,
                                          axes[0].key, axes[0].type);
                break;
        case 2:
                hist = tracefs_hist_alloc_2d(tep, system, event,
                                             axes[0].key, axes[0].type,
                                             axes[1].key, axes[1].type);
                break;
        default:
                /* Really, 1 and 2 could use this too */
                hist = tracefs_hist_alloc_nd_cnt(tep, system, event, axes);
        }
        if (!hist) {
                fprintf(stderr, "Failed hist create\n");
                exit(-1);
        }
        if (vals) {
                sav = NULL;
                for (val = strtok_r(vals, ",", &sav); val; val = strtok_r(NULL, ",", &sav)) {
                        ret = tracefs_hist_add_value(hist, val);
                        if (ret) {
                                fprintf(stderr, "Failed to add value %s\n", val);
                                exit(-1);
                        }
                }
        }
        if (sort) {
                sav = NULL;
                for (val = strtok_r(sort, ",", &sav); val; val = strtok_r(NULL, ",", &sav)) {
                        ret = tracefs_hist_add_sort_key(hist, val);
                        if (ret) {
                                fprintf(stderr, "Failed to add sort key/val %s\n", val);
                                exit(-1);
                        }
                }
        }
        if (ascend) {
                sav = NULL;
                for (val = strtok_r(ascend, ",", &sav); val; val = strtok_r(NULL, ",", &sav)) {
                        ret = tracefs_hist_sort_key_direction(hist, val, TRACEFS_HIST_SORT_ASCENDING);
                        if (ret) {
                                fprintf(stderr, "Failed to add ascending key/val %s\n", val);
                                exit(-1);
                        }
                }
        }
        if (desc) {
                sav = NULL;
                for (val = strtok_r(desc, ",", &sav); val; val = strtok_r(NULL, ",", &sav)) {
                        ret = tracefs_hist_sort_key_direction(hist, val, TRACEFS_HIST_SORT_DESCENDING);
                        if (ret) {
                                fprintf(stderr, "Failed to add descending key/val %s\n", val);
                                exit(-1);
                        }
                }
        }
        tracefs_error_clear(instance);
        switch (cmd) {
        case START:
                ret = tracefs_hist_start(instance, hist);
                if (ret) {
                        char *err = tracefs_error_last(instance);
                        if (err)
                                fprintf(stderr, "\n%s\n", err);
                }
                break;
        case PAUSE:
                ret = tracefs_hist_pause(instance, hist);
                break;
        case CONT:
                ret = tracefs_hist_continue(instance, hist);
                break;
        case RESET:
                ret = tracefs_hist_reset(instance, hist);
                break;
        case DELETE:
                ret = tracefs_hist_destroy(instance, hist);
                break;
        case SHOW:
                /* Show was already done */
                break;
        }
        if (ret)
                fprintf(stderr, "Failed: command\n");
        exit(ret);
}
int main (int argc, char **argv, char **env)
{
        enum commands cmd;
        char *instance = NULL;
        char *cmd_str;
        char *event = NULL;
        char *keys = NULL;
        char *vals = NULL;
        char *sort = NULL;
        char *desc = NULL;
        char *ascend = NULL;
        if (argc < 2) {
                fprintf(stderr, "usage: %s command [-B instance][-e [system/]event][-k keys][-v vals][-s sort]\n", argv[0]);
                fprintf(stderr, "      [-a ascending][-d descending]\n");
                exit(-1);
        }
        cmd_str = argv[1];
        if (!strcmp(cmd_str, "start"))
                cmd = START;
        else if (!strcmp(cmd_str, "pause"))
                cmd = PAUSE;
        else if (!strcmp(cmd_str, "cont"))
                cmd = CONT;
        else if (!strcmp(cmd_str, "reset"))
                cmd = RESET;
        else if (!strcmp(cmd_str, "delete"))
                cmd = DELETE;
        else if (!strcmp(cmd_str, "show"))
                cmd = SHOW;
        else {
                fprintf(stderr, "Unknown command %s\n", cmd_str);
                exit(-1);
        }
        for (;;) {
                int c;
                c = getopt(argc - 1, argv + 1, "e:k:v:B:s:d:a:");
                if (c == -1)
                        break;
                switch (c) {
                case 'e':
                        event = optarg;
                        break;
                case 'k':
                        keys = optarg;
                        break;
                case 'v':
                        vals = optarg;
                        break;
                case 'B':
                        instance = optarg;
                        break;
                case 's':
                        sort = optarg;
                        break;
                case 'd':
                        desc = optarg;
                        break;
                case 'a':
                        ascend = optarg;
                        break;
                }
        }
        if (!event) {
                event = "kmem/kmalloc";
                if (!keys)
                        keys = "call_site.sym,bytes_req";
                if (!vals)
                        vals = "bytes_alloc";
                if (!sort)
                        sort = "bytes_req,bytes_alloc";
                if (!desc)
                        desc = "bytes_alloc";
        }
        process_hist(cmd, instance, event, keys, vals, sort, ascend, desc);
}
