tracefs_synth_set_instance(3)

الحزمة: libtracefs

التراخيص: GPL-2.0-only, LGPL-2.1-only

LIBTRACEFS(3) دليل libtracefs LIBTRACEFS(3)

الاسم

tracefs_synth_create, tracefs_synth_destroy, tracefs_synth_complete, tracefs_synth_trace, tracefs_synth_snapshot, tracefs_synth_save, tracefs_synth_set_instance, - إنشاء أحداث اصطناعية

الملخص

#include <tracefs.h>
int tracefs_synth_create(struct tracefs_synth *synth);
int tracefs_synth_destroy(struct tracefs_synth *synth);
bool tracefs_synth_complete(struct tracefs_synth *synth);
int tracefs_synth_set_instance(struct tracefs_synth *synth, struct tracefs_instance *instance);
int tracefs_synth_trace(struct tracefs_synth *synth,
                        enum tracefs_synth_handler type, const char *var);
int tracefs_synth_snapshot(struct tracefs_synth *synth,
                           enum tracefs_synth_handler type, const char *var);
int tracefs_synth_save(struct tracefs_synth *synth,
                       enum tracefs_synth_handler type, const char *var,
                       char **save_fields);

الوصف

الأحداث الاصطناعية هي أحداث ديناميكية يتم إنشاؤها عن طريق مطابقة حدثين آخرين يؤديان إلى حدث اصطناعي. أحد الأحداث هو الحدث البادئ الذي يتم تسجيل حقل فيه، وعند تنفيذ الحدث الثاني، إذا كان يحتوي على حقل (أو حقول) يتطابق مع حقل (أو حقول) الحدث البادئ، فسيؤدي ذلك إلى تشغيل الحدث الاصطناعي. قد يتم تمرير قيم الحقول بخلاف الحقول المتطابقة من الحدث البادئ إلى الحدث النهائي لإجراء عمليات حسابية عليها، أو ببساطة لتمريرها كمعلمة للحدث الاصطناعي.

أحد حالات الاستخدام الشائعة هو تعيين "sched_waking" كالحدث البادئ. يتم تشغيل هذا الحدث عند إيقاظ عملية. ثم تعيين "sched_switch" كالحدث المنتهي. يتم تشغيل هذا الحدث عند جدولة مهمة جديدة على وحدة المعالجة المركزية. عن طريق تعيين "common_pid" لكلا الحدثين كحقول مطابقة، يعتبر الوقت بين الحدثين هو زمن الاستجابة لإيقاظ تلك العملية. استخدم TRACEFS_TIMESTAMP كحقل لكلا الحدثين لحساب الفرق بالنانوثانية، أو استخدم TRACEFS_TIMESTAMP_USECS كحقول مقارنة لكلا الحدثين لحساب الفرق بالميكروثانية. يتم استخدام هذا كمثال أدناه.

tracefs_synth_create() ينشئ الحدث الاصطناعي في النظام. بشكل افتراضي، يتم إنشاء مشغلات الرسم البياني في أعلى مثيل تتبع، حيث يمكن استخدام أي حدث اصطناعي عالميًا عبر جميع المثيلات. في حالة رغبة تطبيق ما في إبقاء مشغلات الرسم البياني خارج المستوى الأعلى، يمكنه استخدام tracefs_synth_set_instance() لجعل الرسوم البيانية المستخدمة لإنشاء الحدث الاصطناعي في مثيل غير المستوى الأعلى. يجب إنشاء واصف الحدث الاصطناعي باستخدام tracefs_synth_alloc(3) قبل أن يمكن استخدامه لإنشائه على النظام.

tracefs_synth_destroy() يدمر الحدث الاصطناعي. سيحاول إيقاف تشغيله في مثيله (الأعلى افتراضيًا)، ولكن إذا كان يعمل في مثيل آخر، فقد يفشل لأنه مشغول.

tracefs_synth_complete() تعيد true إذا كان الحدث الاصطناعي synth يحتوي على كل من الحدث البادئ والحدث المنتهي.

tracefs_synth_trace() بدلاً من مجرد إجراء تتبع عند مطابقة الحدثين البادئ والمنتهي، قم بتنفيذ معالج type حيث يقوم TRACEFS_SYNTH_HANDLE_MAX بإجراء تتبع عند وصول المتغير var المحدد إلى الحد الأقصى الجديد للمفاتيح المطابقة. أو TRACEFS_SYNTH_HANDLE_CHANGE عندما يتغير var. يجب أن يكون var أحد عناصر name المستخدمة في tracefs_synth_add_end_field(3).

tracefs_synth_snapshot() عند أن يكون المتغير var المحدد إما حدًا أقصى جديدًا إذا كان handler هو TRACEFS_SYNTH_HANDLE_MAX أو قد تغير ببساطة إذا كان TRACEFS_SYNTH_HANDLE_CHANGE، فخذ "لقطة" من المخزن المؤقت. تنقل اللقطة المخزن المؤقت "trace" العادي إلى مخزن مؤقت "snapshot"، والذي يمكن الوصول إليه عبر ملف "snapshot" في دليل tracefs الأعلى، أو أحد المثيلات. يتغير var. يجب أن يكون var أحد عناصر name المستخدمة في tracefs_synth_add_end_field(3).

tracefs_synth_save() عند أن يكون المتغير var المحدد إما حدًا أقصى جديدًا إذا كان handler هو TRACEFS_SYNTH_HANDLE_MAX أو قد تغير ببساطة إذا كان TRACEFS_SYNTH_HANDLE_CHANGE، فقم بحفظ قائمة save_fields المحددة. سيتم تخزين الحقول في ملف "hist" للرسم البياني للحدث، والذي يمكن استعادته باستخدام tracefs_event_file_read(3). يجب أن يكون var أحد عناصر name المستخدمة في tracefs_synth_add_end_field(3).

tracefs_synth_set_instance() تعيين مثيل التتبع، حيث سيتم إنشاء مشغلات الرسم البياني التي تنشئ الحدث الاصطناعي. افتراضيًا، يتم استخدام المثيل الأعلى. يجب استدعاء هذه الواجهة البرمجية قبل استدعاء tracefs_synth_create()، من أجل استخدام المثيل الجديد عند إنشاء الحدث. لاحظ أنه حتى لو تم إنشاء الحدث الاصطناعي في مثيل، إلا أنه لا يزال مرئيًا لجميع المثيلات الأخرى بما في ذلك المستوى الأعلى. أي أن المثيلات الأخرى يمكنها تمكين الحدث الاصطناعي الذي تم إنشاؤه و تتبعه في المخازن المؤقتة التي تنتمي إلى المثيل الذي قام بتمكينه.

قيمة الإرجاع

جميع الوظائف تعيد صفر عند النجاح أو -1 عند الخطأ.

الأخطاء

الأخطاء التالية لجميع الاستدعاءات أعلاه:

EPERM لم يتم التشغيل كمستخدم جذر عند الحاجة.

EINVAL إما أن تكون معلمة غير صالحة (NULL عندما لا ينبغي) أو حقل غير متوافق مع العمليات الحسابية.

ENODEV لم يتم العثور على حدث أو أحد حقوله.

EBADE حقول الحدثين البادئ والمنتهي غير متوافقة إما للمطابقة أو للمقارنة.

ENOMEM لا توجد ذاكرة كافية متاحة.

وقد تحدث أخطاء أخرى من استدعاءات النظام إلى النظام.

مثال

انظر tracefs_sql(3) لمزيد من الاستخدام العميق لبعض هذا الكود.

#include <stdlib.h>
#include <tracefs.h>
#define start_event "sched_waking"
#define start_field "pid"
#define end_event "sched_switch"
#define end_field "next_pid"
#define match_name "pid"
static struct tracefs_synth *synth;
static void make_event(void)
{
        struct tep_handle *tep;
        /* تحميل جميع الأحداث من النظام */
        tep = tracefs_local_events(NULL);
        /* تهيئة الحدث الاصطناعي */
        synth = tracefs_synth_alloc(tep, "wakeup_lat",
                                    NULL, start_event,
                                    NULL, end_event,
                                    start_field, end_field,
                                    match_name);
        /* لم يعد tep مطلوبًا */
        tep_free(tep);
        /* حفظ حقل "prio" باسم "prio" من الحدث البادئ */
        tracefs_synth_add_start_field(synth, "prio", NULL);
        /* حفظ "next_comm" باسم "comm" من الحدث المنتهي */
        tracefs_synth_add_end_field(synth, "next_comm", "comm");
        /* حفظ "prev_prio" باسم "prev_prio" من الحدث المنتهي */
        tracefs_synth_add_end_field(synth, "prev_prio", NULL);
        /*
         * أخذ فرق زمني بالميكروثانية بين النهاية والبداية
         * وتسجيله باسم "delta"
         */
        tracefs_synth_add_compare_field(synth, TRACEFS_TIMESTAMP_USECS,
                                        TRACEFS_TIMESTAMP_USECS,
                                        TRACEFS_SYNTH_DELTA_END, "delta");
        /* تسجيل فقط إذا كان "prio" الحدث البادئ أقل من 100 */
        tracefs_synth_append_start_filter(synth, TRACEFS_FILTER_COMPARE,
                                          "prio", TRACEFS_COMPARE_LT, "100");
        /*
         * تسجيل فقط إذا كان "next_prio" الحدث المنتهي أقل من 50
         * أو إذا لم تكن قيمة prio للمهمة السابقة أكبر من أو تساوي 100.
         *   next_prio < 50 || !(prev_prio >= 100)
         */
        tracefs_synth_append_end_filter(synth, TRACEFS_FILTER_COMPARE,
                                        "next_prio", TRACEFS_COMPARE_LT, "50");
        tracefs_synth_append_end_filter(synth, TRACEFS_FILTER_OR, NULL, 0, NULL);
        tracefs_synth_append_end_filter(synth, TRACEFS_FILTER_NOT, NULL, 0, NULL)
; tracefs_synth_append_end_filter(synth, TRACEFS_FILTER_OPEN_PAREN, NULL, 0, NULL); tracefs_synth_append_end_filter(synth, TRACEFS_FILTER_COMPARE, "prev_prio", TRACEFS_COMPARE_GE, "100"); /* * ملاحظة، لقد أضافت فقط: "next_prio < 50 || !(prev_prio >= 100" * هذا لأنه، عند تنفيذ synth، سيتم إضافة القوس المغلق المتبقي. أي أن السلسلة ستنتهي لتصبح: * "next_prio < 50 || !(prev_prio >= 100)" عند تشغيل tracefs_sync_create() * أو tracefs_sync_echo_cmd(). */ } /* عرض كيفية إنشاء الحدث الاصطناعي */ static void show_event(void) { struct trace_seq s; trace_seq_init(&s); tracefs_synth_echo_cmd(&s, synth); trace_seq_terminate(&s); trace_seq_do_printf(&s); trace_seq_destroy(&s); } int main (int argc, char **argv) { make_event(); if (argc > 1) { if (!strcmp(argv[1], "create")) { /* إنشاء الحدث الاصطناعي */ tracefs_synth_create(synth); } else if (!strcmp(argv[1], "delete")) { /* حذف الحدث الاصطناعي */ tracefs_synth_destroy(synth); } else { printf("usage: %s [create|delete]\n", argv[0]); exit(-1); } } else show_event(); tracefs_synth_free(synth); return 0; }
