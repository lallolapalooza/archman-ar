tracefs_synth_save(3)

الحزمة: libtracefs

التراخيص: GPL-2.0-only, LGPL-2.1-only

LIBTRACEFS(3) دليل libtracefs LIBTRACEFS(3)

الاسم

tracefs_synth_create, tracefs_synth_destroy, tracefs_synth_complete, tracefs_synth_trace, tracefs_synth_snapshot, tracefs_synth_save, tracefs_synth_set_instance, - إنشاء أحداث اصطناعية

الملخص

#include <tracefs.h>
int tracefs_synth_create(struct tracefs_synth *synth);
int tracefs_synth_destroy(struct tracefs_synth *synth);
bool tracefs_synth_complete(struct tracefs_synth *synth);
int tracefs_synth_set_instance(struct tracefs_synth *synth, struct tracefs_instance *instance);
int tracefs_synth_trace(struct tracefs_synth *synth,
                        enum tracefs_synth_handler type, const char *var);
int tracefs_synth_snapshot(struct tracefs_synth *synth,
                           enum tracefs_synth_handler type, const char *var);
int tracefs_synth_save(struct tracefs_synth *synth,
                       enum tracefs_synth_handler type, const char *var,
                       char **save_fields);

الوصف

الأحداث الاصطناعية هي أحداث ديناميكية يتم إنشاؤها عن طريق مطابقة حدثين آخرين يؤديان إلى حدث اصطناعي. أحد الأحداث هو الحدث البادئ الذي يتم فيه تسجيل حقل، وعند تنفيذ الحدث الثاني، إذا كان يحتوي على حقل (أو حقول) يتطابق مع حقل (أو حقول) الحدث البادئ، فسيؤدي ذلك إلى تشغيل الحدث الاصطناعي. قد يتم تمرير قيم الحقول بخلاف الحقول المتطابقة من الحدث البادئ إلى الحدث النهائي لإجراء عمليات حسابية عليها، أو ببساطة لتمريرها كمعامل للحدث الاصطناعي.

يتمثل أحد حالات الاستخدام الشائعة في تعيين "sched_waking" كحدث بدء. يتم تشغيل هذا الحدث عند إيقاظ عملية. ثم تعيين "sched_switch" كحدث نهاية. يتم تشغيل هذا الحدث عند جدولة مهمة جديدة على وحدة المعالجة المركزية. عن طريق تعيين "common_pid" لكلا الحدثين كحقول مطابقة، يعتبر الوقت بين الحدثين هو زمن انتقال الاستيقاظ لهذه العملية. استخدم TRACEFS_TIMESTAMP كحقل لكلا الحدثين لحساب الفرق بالنانوثانية، أو استخدم TRACEFS_TIMESTAMP_USECS كحقول مقارنة لكلا الحدثين لحساب الفرق بالميكروثانية. يستخدم هذا كمثال أدناه.

tracefs_synth_create() يقوم بإنشاء الحدث الاصطناعي في النظام. افتراضيًا، يتم إنشاء مشغلات الرسم البياني التكراري في مثيل التتبع العلوي، حيث يمكن استخدام أي حدث اصطناعي عالميًا عبر جميع المثيلات. في حالة رغبة تطبيق في إبقاء مشغلات الرسم البياني التكراري خارج المثيل العلوي، يمكنه استخدام tracefs_synth_set_instance() لجعل الرسوم البيانية التكرارية المستخدمة لإنشاء الحدث الاصطناعي في مثيل آخر غير المثيل العلوي. يجب إنشاء واصف للحدث الاصطناعي باستخدام tracefs_synth_alloc(3) قبل استخدام ذلك لإنشائه على النظام.

tracefs_synth_destroy() يقوم بتدمير الحدث الاصطناعي. سيحاول إيقاف تشغيله في مثيله (العلوي افتراضيًا)، ولكن إذا كان يعمل في مثيل آخر فقد يفشل بسبب الاستخدام.

tracefs_synth_complete() تعيد قيمة صحيحة إذا كان الحدث الاصطناعي synth يحتوي على حدث بدء وحدث نهاية.

tracefs_synth_trace() بدلاً من مجرد إجراء تتبع عند مطابقة أحداث البداية والنهاية، قم بتنفيذ معالج type حيث يقوم TRACEFS_SYNTH_HANDLE_MAX بتتبع عندما يصل المتغير var المعطى إلى الحد الأقصى الجديد للمفاتيح المتطابقة. أو TRACEFS_SYNTH_HANDLE_CHANGE عندما يتغير var. يجب أن يكون var أحد عناصر name المستخدمة في tracefs_synth_add_end_field(3).

tracefs_synth_snapshot() عندما يكون المتغير var المعطى إما الحد الأقصى الجديد إذا كان handler هو TRACEFS_SYNTH_HANDLE_MAX أو يتغير ببساطة إذا كان TRACEFS_SYNTH_HANDLE_CHANGE، فقم بأخذ "لقطة" من المخزن المؤقت. تقوم اللقطة بنقل المخزن المؤقت "trace" العادي إلى مخزن مؤقت "snapshot"، والذي يمكن الوصول إليه عبر ملف "snapshot" في الدليل العلوي tracefs، أو أحد المثيلات. تتغير var. يجب أن يكون var أحد عناصر name المستخدمة في tracefs_synth_add_end_field(3).

tracefs_synth_save() عندما يكون المتغير var المعطى إما الحد الأقصى الجديد إذا كان handler هو TRACEFS_SYNTH_HANDLE_MAX أو يتغير ببساطة إذا كان TRACEFS_SYNTH_HANDLE_CHANGE، فقم بحفظ قائمة save_fields المعطاة. سيتم تخزين الحقول في ملف "hist" الخاص بالحدث الذي يمكن استرداده باستخدام tracefs_event_file_read(3). يجب أن يكون var أحد عناصر name المستخدمة في tracefs_synth_add_end_field(3).

tracefs_synth_set_instance() قم بتعيين مثيل التتبع، حيث سيتم إنشاء مشغلات الرسم البياني التكراري التي تنشئ الحدث الاصطناعي. افتراضيًا، يتم استخدام المثيل العلوي. يجب استدعاء واجهة برمجة التطبيقات هذه قبل استدعاء tracefs_synth_create()، من أجل استخدام المثيل الجديد عند إنشاء الحدث. لاحظ أنه حتى لو تم إنشاء الحدث الاصطناعي في مثيل، فإنه لا يزال مرئيًا لجميع المثيلات الأخرى بما في ذلك المستوى العلوي. أي أن المثيلات الأخرى يمكنها تمكين الحدث الاصطناعي الذي تم إنشاؤه وله تتبعه في المخازن المؤقتة التي تنتمي إلى المثيل الذي قام بتمكينه.

قيمة الإرجاع

جميع الوظائف تعيد صفر عند النجاح أو -1 عند الخطأ.

الأخطاء

الأخطاء التالية لجميع الاستدعاءات المذكورة أعلاه:

EPERM لم يتم التشغيل كمستخدم جذر عند الحاجة.

EINVAL إما أن يكون المعامل غير صالح (NULL عندما لا ينبغي أن يكون) أو حقل غير متوافق مع العمليات الحسابية.

ENODEV لم يتم العثور على حدث أو أحد حقوله.

EBADE حقول الحدثين البادئ والنهائي غير متوافقة إما للمطابقة أو المقارنة.

ENOMEM لا توجد ذاكرة كافية متاحة.

وقد تحدث أخطاء أخرى من استدعاءات النظام إلى النظام.

مثال

راجع tracefs_sql(3) للحصول على استخدام أعمق لبعض هذا الكود.

#include <stdlib.h>
#include <tracefs.h>
#define start_event "sched_waking"
#define start_field "pid"
#define end_event "sched_switch"
#define end_field "next_pid"
#define match_name "pid"
static struct tracefs_synth *synth;
static void make_event(void)
{
        struct tep_handle *tep;
        /* تحميل جميع الأحداث من النظام */
        tep = tracefs_local_events(NULL);
        /* تهيئة الحدث الاصطناعي */
        synth = tracefs_synth_alloc(tep, "wakeup_lat",
                                    NULL, start_event,
                                    NULL, end_event,
                                    start_field, end_field,
                                    match_name);
        /* لم تعد هناك حاجة إلى tep */
        tep_free(tep);
        /* حفظ حقل "prio" كـ "prio" من الحدث البادئ */
        tracefs_synth_add_start_field(synth, "prio", NULL);
        /* حفظ "next_comm" كـ "comm" من الحدث النهائي */
        tracefs_synth_add_end_field(synth, "next_comm", "comm");
        /* حفظ "prev_prio" كـ "prev_prio" من الحدث النهائي */
        tracefs_synth_add_end_field(synth, "prev_prio", NULL);
        /*
         * أخذ فرق زمني بالميكروثانية بين النهاية والبداية
         * وتسجيله كـ "delta"
         */
        tracefs_synth_add_compare_field(synth, TRACEFS_TIMESTAMP_USECS,
                                        TRACEFS_TIMESTAMP_USECS,
                                        TRACEFS_SYNTH_DELTA_END, "delta");
        /* تسجيل فقط إذا كان "prio" للحدث البادئ أقل من 100 */
        tracefs_synth_append_start_filter(synth, TRACEFS_FILTER_COMPARE,
                                          "prio", TRACEFS_COMPARE_LT, "100");
        /*
         * تسجيل فقط إذا كان "next_prio" للحدث النهائي أقل من 50
         * أو لم يكن الأولوية السابقة للمهمة أكبر من أو تساوي 100.
         *   next_prio < 50 || !(prev_prio >= 100)
         */
        tracefs_synth_append_end_filter(synth, TRACEFS_FILTER_COMPARE,
                                        "next_prio", TRACEFS_COMPARE_LT, "50");
        tracefs_synth_append_end_filter(synth, TRACEFS_FILTER_OR, NULL, 0, NULL);
        tracefs_synth_append_end_filter(synth, TRACEFS_FILTER_NOT, NULL, 0, NULL);
