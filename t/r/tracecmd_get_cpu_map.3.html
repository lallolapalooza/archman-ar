tracecmd_get_cpu_map(3)

الحزمة: trace-cmd

التراخيص: GPL-2.0-only, LGPL-2.1-only

LIBTRACECMD(3) دليل libtracefs LIBTRACECMD(3)

الاسم

tracecmd_map_vcpus, tracecmd_get_cpu_map, tracecmd_map_find_by_host_pid, tracecmd_map_get_host_pid, tracecmd_map_get_guest, tracecmd_map_set_private, tracecmd_map_get_private - ربط بيانات المضيف والضيف

الخلاصة

#include <trace-cmd.h>
int tracecmd_map_vcpus(struct tracecmd_input **handles, int nr_handles);
struct tracecmd_cpu_map *tracecmd_get_cpu_map(struct tracecmd_input *handle, int cpu);
struct tracecmd_cpu_map *tracecmd_map_find_by_host_pid(struct tracecmd_input *handle,
                                                      int host_pid);
int tracecmd_map_get_host_pid(struct tracecmd_cpu_map *map);
struct tracecmd_input *tracecmd_map_get_guest(struct tracecmd_cpu_map *map);
void tracecmd_map_set_private(struct tracecmd_cpu_map *map, void *priv);
void *tracecmd_map_get_private(struct tracecmd_cpu_map *map);

الوصف

تُستخدم هذه المجموعة من واجهات برمجة التطبيقات لربط ملفات تتبع المضيف والضيف لتسهيل تحليل التتبع الإضافي.

تأخذ الدالة tracecmd_map_vcpus() مصفوفة من handles حيث يتم إنشاء كل عنصر في تلك المصفوفة بواسطة إحدى وظائف tracecmd_open(3)، وعدد handles كـ nr_handles. يُتوقع أن يكون الـ handle الأول في مصفوفة handles هو الواصف لملف تتبع المضيف، والبقية هي ملفات تتبع الضيف التي تعمل على المضيف، والتي تم إنشاؤها بواسطة تفاعلات trace-cmd record(1) و trace-cmd agent(1). تُرجع عدد الضيوف الموجودين في handles الذين تم ربطهم بالمضيف، أو قيمة سالبة في حالة حدوث خطأ.

تُرجع الدالة tracecmd_get_cpu_map() واصفًا لوحدة المعالجة المركزية (CPU) معينة لـ handle. إذا كان الـ handle ضيفًا تم تعريفه من tracecmd_map_vcpus()، فإن الربط الذي تم إنشاؤه من تلك الدالة والمرتبط بوحدة المعالجة المركزية الافتراضية (vCPU) هذه (المشار إليها بـ cpu) من handle. يمكن استخدام هذا الواصف بواسطة وظائف tarcecmd_map_get_guest() و tracecmd_map_set_private() و tracecmd_map_get_private().

ستُرجع الدالة tracecmd_map_find_by_host_pid() ربطًا لوحدة معالجة مركزية افتراضية للضيف يتم التعامل معها بواسطة host_pid المعطى. ملاحظة، يمكن أن يكون الـ handle المُمرر إما handle المضيف أو أحد الـ handles الخاصة بالضيف لذلك المضيف والتي تم ربطها بواسطة tracecmd_map_vcpus()، حتى لو لم يكن handle الضيف يحتوي على وحدة المعالجة المركزية التي يمثلها host_pid.

ستُرجع الدالة tracecmd_map_get_host_pid() الـ host_pid لـ map معين تم استرداده بواسطة إحدى الوظائف أعلاه.

ستُرجع الدالة tracecmd_map_get_guest() الـ guest_handle لـ map معين تم استرداده بواسطة إحدى الوظائف أعلاه.

تسمح الدالة tracecmd_map_set_private() للتطبيق بتعيين بيانات خاصة لربط خيط المضيف بوحدة المعالجة المركزية الافتراضية للضيف الذي تم تحديده بواسطة map.

تسترد الدالة tracecmd_map_get_private() بيانات priv من map التي تم تعيينها بواسطة tracecmd_map_set_private().

قيمة الإرجاع

تُرجع الدالة tracecmd_map_vcpus() عدد الضيوف في مصفوفة handles التي تم ربطها بـ handle المضيف وهو الإدخال الأول في handles. تُرجع -1 في حالة حدوث خطأ.

تُرجع الدالة tracecmd_get_cpu_map() خريطة تم إنشاؤها بواسطة tracecmd_map_vcpus() لوحدة cpu معينة لـ handle معين، أو NULL إذا لم يتم العثور عليها.

تُرجع الدالة tracecmd_map_find_by_host_pid() ربطًا مرتبطًا بمهمة المضيف بمعرف host_pid الخاص بها. يمكن أن يكون الـ handle إما handle المضيف، أو أحد الـ handles الخاصة بالضيف التي تم ربطها بالمضيف عبر tracecmd_map_vcpus()، حتى لو كان handle الضيف مختلفًا عن الضيف الذي يتعلق به الربط. تُرجع NULL إذا لم يتم العثور عليه.

تُرجع الدالة tracecmd_map_get_host_pid() معرف عملية المضيف للربط المرتبط المحدد بواسطة map.

تُرجع الدالة tracecmd_map_get_guest() handle الضيف للربط المرتبط المحدد بواسطة map.

تُرجع الدالة tracecmd_map_get_private() البيانات الخاصة للربط المحدد بواسطة map والتي تم تعيينها بواسطة tracecmd_map_set_private().

مثال

#include <stdlib.h>
#include <errno.h>
#include <trace-cmd.h>
int main(int argc, char **argv)
{
        struct tracecmd_input **handles = NULL;
        int nr_handles;
        int i;
        if (argc < 2) {
                printf("usage: host_trace.dat guest1_trace.dat [guest2_trace.dat ...]\n");
                exit(-1);
        }
        for (i = 1; i < argc; i++) {
                handles = realloc(handles, sizeof(*handles) * (nr_handles + 1));
                if (!handles)
                        exit(-1);
                handles[nr_handles] = tracecmd_open(argv[i], 0);
                if (!handles[nr_handles]) {
                        perror(argv[1]);
                        exit(-1);
                }
                tracecmd_set_private(handles[nr_handles], argv[i]);
                nr_handles++;
        }
        tracecmd_map_vcpus(handles, nr_handles);
        for (i = 1; i < nr_handles; i++) {
                struct tracecmd_cpu_map *map;
                struct tep_handle *tep;
                const char *file = tracecmd_get_private(handles[i]);
                int cpus, cpu;
                printf("Mappings for guest %s:\n", file);
                tep = tracecmd_get_tep(handles[i]);
                cpus = tep_get_cpus(tep);
                for (cpu = 0; cpu < cpus; cpu++) {
                        printf("  [%03d] ", cpu);
                        map = tracecmd_get_cpu_map(handles[i], cpu);
                        if (!map) {
                                printf("Has no mapping!\n");
                                continue;
                        }
                        printf("host_pid: %d\n", tracecmd_map_get_host_pid(map));
                }
        }
        for (i = 0; i < nr_handles; i++)
                tracecmd_close(handles[i]);
        free(handles);
        exit(0);
}
