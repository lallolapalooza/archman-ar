tracefs_cpu_buffered_read(3)

الحزمة: libtracefs

التراخيص: GPL-2.0-only, LGPL-2.1-only

LIBTRACEFS(3) libtracefs Manual LIBTRACEFS(3)

اسم

tracefs_cpu_read_size, tracefs_cpu_read, tracefs_cpu_buffered_read, tracefs_cpu_write, tracefs_cpu_stop, tracefs_cpu_flush, tracefs_cpu_flush_write, tracefs_cpu_pipe - قراءة بيانات trace_pipe_raw

الخلاصة

#include <tracefs.h>
int tracefs_cpu_read_size(struct tracefs_cpu *tcpu);
int tracefs_cpu_read(struct tracefs_cpu *tcpu, void *buffer, bool nonblock);
int tracefs_cpu_buffered_read(struct tracefs_cpu *tcpu, void *buffer, bool nonblock);
int tracefs_cpu_write(struct tracefs_cpu *tcpu, int wfd, bool nonblock);
int tracefs_cpu_stop(struct tracefs_cpu *tcpu);
int tracefs_cpu_flush(struct tracefs_cpu *tcpu, void *buffer);
int tracefs_cpu_flush_write(struct tracefs_cpu *tcpu, int wfd);
int tracefs_cpu_pipe(struct tracefs_cpu *tcpu, int wfd, bool nonblock);

الوصف

يمكن استخدام مجموعة واجهات برمجة التطبيقات هذه لقراءة البيانات الأولية من ملفات trace_pipe_raw في نظام ملفات tracefs.

ترجع tracefs_cpu_read_size() حجم المخزن المؤقت الفرعي لـ trace_pipe_raw. يرجع هذا الحد الأدنى لحجم المخزن المؤقت الذي يتم تمريره إلى الوظائف أدناه.

تقوم tracefs_cpu_read() بقراءة ملفات trace_pipe_raw المرتبطة بـ tcpu في buffer. يجب أن يكون buffer على الأقل بحجم المخزن المؤقت الفرعي للحلقة التخزينية، والذي يتم إرجاعه بواسطة tracefs_cpu_read_size(). إذا تم تعيين nonblock، ولا تتوفر بيانات، فستعود على الفور. بخلاف ذلك، اعتمادًا على كيفية فتح tcpu، سيتم حظرها. إذا تم فتح tcpu مع تعيين nonblock، فلن يكون لـ nonblock هذا أي فرق.

tracefs_cpu_buffered_read() هي في الأساس نفس tracefs_cpu_read() باستثناء أنها تستخدم أنبوبًا عبر splice لتخزين القراءات مؤقتًا. سيؤدي ذلك إلى تجميع القراءات مع الحفاظ على القراءة من المخزن المؤقت الدائري أقل تدخلاً في النظام، حيث أن مجرد القراءة طوال الوقت يمكن أن يسبب اضطرابًا كبيرًا. لاحظ، أحد الاختلافات بين هذا و tracefs_cpu_read() هو أنه سيقرأ فقط في صفحات المخزن المؤقت الفرعي. إذا لم يملأ المخزن المؤقت الدائري صفحة، فلن يعيد أي شيء، حتى مع تعيين nonblock. يجب إجراء استدعاءات لـ tracefs_cpu_flush() لقراءة بقية الملف في نهاية التتبع.

ستقوم tracefs_cpu_write() بتوجيه البيانات من ملف trace_pipe_raw المرتبط بـ tcpu إلى واصف الملف wfd. إذا تم تعيين nonblock، فلن يتم حظرها إذا لم يكن هناك شيء للكتابة. لاحظ، أنها ستكتب فقط بحجم المخزن المؤقت الفرعي إلى wfd. هناك حاجة إلى استدعاءات لـ tracefs_cpu_flush_write() لكتابة الباقي.

تقوم tracefs_cpu_stop() بمحاولة إلغاء حظر مهمة محظورة على tcpu تقرأها. في النوى الأقدم، قد لا تفعل شيئًا لقراءات الأنابيب، حيث أن النوى الأقدم لا توقظ المهام التي تنتظر على المخزن المؤقت الدائري. تعود 0 إذا أيقظت بالتأكيد أي منتظرين محتملين، ولكنها تعود 1 إذا لم تكن متأكدة من أنها نجحت وقد يحتاج المنتظرون إلى إرسال إشارة إليهم.

تقوم tracefs_cpu_flush() بقراءة ملف trace_pipe_raw المرتبط بواسطة tcpu ووضعه في buffer، والذي يجب أن يكون بحجم المخزن المؤقت الفرعي الذي يتم استرداده. بواسطة tracefs_cpu_read_size(). يجب استدعاء هذا في نهاية التتبع للحصول على بقية البيانات. سيقوم هذا الاستدعاء بتحويل واصف ملف trace_pipe_raw إلى وضع غير حاظر.

tracefs_cpu_flush_write() نفس trace_cpu_flush() باستثناء أنها تأخذ واصف ملف wfd لتدفق البيانات إليه.

tracefs_cpu_pipe() مشابهة لـ tracefs_cpu_write() ولكن يجب أن يكون واصف الملف wfd أنبوبًا. هذا الاستدعاء هو تحسين لـ tracefs_cpu_write() الذي يستخدم استدعاءين لـ splice(2) لتوصيل واصف ملف trace_pipe_raw بواصف الملف للكتابة. يتطلب splice(2) أن يكون أحد واصفات الملفات التي تم تمريرها أنبوبًا. إذا أراد التطبيق تمرير البيانات إلى أنبوب موجود، فلا يوجد سبب لوجود استدعاءين لـ splice(2) ويمكن لـ tracefs_cpu_pipe() ببساطة استخدام استدعاء واحد لـ wfd.

قيمة الإرجاع

ترجع tracefs_cpu_open() واصف struct tracefs_cpu الذي يمكن استخدامه بواسطة الوظائف الأخرى أو NULL في حالة حدوث خطأ.

ترجع tracefs_cpu_read_size() الحد الأدنى لحجم المخازن المؤقتة للاستخدام مع tracefs_cpu_read() و tracefs_cpu_buffered_read() و tracefs_cpu_flush(). ترجع قيمة سالبة في حالة الخطأ.

ترجع tracefs_cpu_read() عدد البايتات المقروءة، أو قيمة سالبة في حالة الخطأ.

ترجع tracefs_cpu_buffered_read() عدد البايتات المقروءة أو قيمة سالبة في حالة الخطأ.

ترجع tracefs_cpu_write() عدد البايتات المكتوبة في الملف أو قيمة سالبة في حالة الخطأ.

ترجع tracefs_cpu_stop() صفرًا إذا تم ضمان إيقاظ أي منتظرين من الانتظار على الإدخال، أو ترجع واحدًا إذا كانت هذه نواة أقدم لا توفر هذا الضمان، وقد تحتاج إشارة إلى إرسالها إلى أي منتظرين. ترجع قيمة سالبة في حالة الخطأ.

ترجع tracefs_cpu_flush() عدد البايتات المقروءة أو قيمة سالبة في حالة الخطأ.

ترجع tracefs_cpu_flush_write() عدد البايتات المكتوبة في الملف أو قيمة سالبة في حالة الخطأ.

مثال

#define _LARGEFILE64_SOURCE
#include <stdlib.h>
#include <ctype.h>
#include <pthread.h>
#include <unistd.h>
#include <tracefs.h>
struct thread_data {
        struct tracefs_cpu      *tcpu;
        int                     done;
        int                     fd;
};
static void *thread_run(void *arg)
{
        struct thread_data *data = arg;
        struct tracefs_cpu *tcpu = data->tcpu;
        int fd = data->fd;
        int ret;
        while (!data->done) {
                ret = tracefs_cpu_write(tcpu, fd, false);
                printf("wrote %d\n", ret);
        }
        return NULL;
}
int main (int argc, char **argv)
{
        struct tracefs_instance *instance;
        struct thread_data data;
        pthread_t thread;
        char *file;
        int secs = 10;
        int cpu;
        int ret;
        if (argc < 3 || !isdigit(argv[1][0])) {
                printf("usage: %s cpu file_destination [sleep secs]\n\n", argv[0]);
                exit(-1);
        }
        cpu = atoi(argv[1]);
        file = argv[2];
        if (argc > 3)
                secs = atoi(argv[3]);
        instance = tracefs_instance_create("cpu_write");
        if (!instance) {
                perror("create instance");
                exit(-1);
        }
        memset(&data, 0, sizeof(data));
        data.tcpu = tracefs_cpu_open(instance, cpu, 0);
        if (!data.tcpu) {
                perror("Open instance");
                exit(-1);
        }
        data.fd = open(file, O_WRONLY | O_CREAT | O_TRUNC | O_LARGEFILE, 0644);
        if (data.fd < 0) {
                perror(file);
                exit(-1);
        }
        pthread_create(&thread, NULL, thread_run, &data);
        sleep(secs);
        data.done = 1;
        printf("stopping\n");
        ret = tracefs_cpu_stop(data.tcpu);
        printf("joining %d\n", ret);
        pthread_join(thread, NULL);
        tracefs_trace_off(instance);
        do {
                ret = tracefs_cpu_flush_write(data.tcpu, data.fd);
                printf("flushed %d\n", ret);
        } while (ret > 0);
        tracefs_trace_on(instance);
        tracefs_cpu_close(data.tcpu);
        close(data.fd);
        return 0;
}
