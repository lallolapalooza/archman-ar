الحزمة: lapack-doc

التراخيص: custom

sigma-min(C) بالمعكوس التقديري لـ !> المعيار 1- للـ inverse(C). المعكوس الحقيقي للمعيار 1- للـ inverse(C) !> لا يمكن أن يختلف عن sigma-min(C) بأكثر من عامل sqrt(n1*n2). !> !> عندما يكون SEP صغيرًا، يمكن للتغييرات الصغيرة في T أن تسبب تغييرات كبيرة في !> الفضاء الجزئي الثابت. الحد التقريبي للخطأ الزاوي الأقصى في الفضاء الجزئي الثابت الأيمن المحسوب هو !> !> EPS * norm(T) / SEP !>

التعريف في السطر 262 من الملف ctrsen.f.

subroutine dtrsen (character job, character compq, logical, dimension( * ) select, integer n, double precision, dimension( ldt, * ) t, integer ldt, double precision, dimension( ldq, * ) q, integer ldq, double precision, dimension( * ) wr, double precision, dimension( * ) wi, integer m, double precision s, double precision sep, double precision, dimension( * ) work, integer lwork, integer, dimension( * ) iwork, integer liwork, integer info)

DTRSEN

الغرض:

!>
!> تقوم DTRSEN بإعادة ترتيب تحليل شور الحقيقي للمصفوفة الحقيقية
!> A = Q*T*Q**T، بحيث تظهر مجموعة مختارة من القيم الذاتية في
!> الكتل القطرية الرائدة للمصفوفة شبه المثلثية العلوية T،
!> وتشكل الأعمدة الرائدة لـ Q أساسًا متعامدًا للمتجه الفضائي الثابت الأيمن المقابل.
!>
!> تقوم الروتين اختياريًا بحساب الأرقام الشرطية المعكوسة لـ
!> مجموعة القيم الذاتية (S) أو الفضاء الجزئي الثابت (SEP).
!>
!> يجب أن تكون T في شكل شور القياسي (كما تم إرجاعه بواسطة DHSEQR)، أي،
!> شبه علوي مصفوفة مقسمة إلى كتل بكتل قطرية 1x1 و 2x2؛ كل
!> كتلة قطرية 2x2 لها عناصرها القطرية متساوية و
!> عناصرها خارج القطر ذات إشارات متعاكسة.
!>

المعاملات

JOB 
!>          JOB هو CHARACTER*1
!>          يحدد ما إذا كانت الأرقام الشرطية مطلوبة لـ
!>          مجموعة القيم الذاتية (S) أو الفضاء الجزئي الثابت (SEP):
!>          = 'N': لا شيء؛
!>          = 'E': للقيم الذاتية فقط (S)؛
!>          = 'V': للفضاء الجزئي الثابت فقط (SEP)؛
!>          = 'B': لكل من القيم الذاتية والفضاء الجزئي الثابت (S و
!>                 SEP).
!>

COMPQ

!>          COMPQ هو CHARACTER*1
!>          = 'V': تحديث المصفوفة Q لمتجهات شور.
!>          = 'N': عدم تحديث Q.
!>

SELECT

!>          SELECT هو مصفوفة LOGICAL، أبعادها (N)
!>          يحدد SELECT القيم الذاتية في المجموعة المختارة. لتحديد
!>          قيمة ذاتية حقيقية w(j)، يجب تعيين SELECT(j) إلى
!>          .TRUE.. لتحديد زوج من القيم الذاتية المترافقة المعقدة
!>          w(j) و w(j+1)، المقابلة لكتلة قطرية 2x2،
!>          يجب تعيين SELECT(j) أو SELECT(j+1) أو كليهما إلى
!>          .TRUE.؛ يجب أن يكون زوج القيم الذاتية المترافقة المعقدة
!>          إما مضمنًا في المجموعة أو مستبعدًا منها.
!>

N

!>          N هو INTEGER
!>          رتبة المصفوفة T. N >= 0.
!>

T

!>          T هي مصفوفة DOUBLE PRECISION، أبعادها (LDT,N)
!>          عند الدخول، المصفوفة شبه المثلثية العلوية T، في شكل شور
!>          القياسي.
!>          عند الخروج، يتم استبدال T بالمصفوفة المعاد ترتيبها T، مرة أخرى في
!>          شكل شور القياسي، مع القيم الذاتية المعاد ترتيبها في
!>          الكتل القطرية الرائدة.
!>

LDT

!>          LDT هو INTEGER
!>          البعد الرائد للمصفوفة T. LDT >= max(1,N).
!>

Q

!>          Q هي مصفوفة DOUBLE PRECISION، أبعادها (LDQ,N)
!>          عند الدخول، إذا كان COMPQ = 'V'، فإن المصفوفة Q لمتجهات شور.
!>          عند الخروج، تم ضرب Q بالمصفوفة التحويل المتعامدة التي
!>          تعيد ترتيب T؛ تشكل الأعمدة الرائدة M لـ Q أساسًا متعامدًا لـ
!>          الفضاء الجزئي الثابت المحدد.
!>          إذا كان COMPQ = 'N'، لا يتم الرجوع إلى Q.
!>

LDQ

!>          LDQ هو INTEGER
!>          البعد الرائد للمصفوفة Q.
!>          LDQ >= 1؛ وإذا كان COMPQ = 'V'، فإن LDQ >= N.
!>

WR

!>          WR هي مصفوفة DOUBLE PRECISION، أبعادها (N)
!>

WI

!>          WI هي مصفوفة DOUBLE PRECISION، أبعادها (N)
!>
!>          الأجزاء الحقيقية والخيالية، على التوالي، للقيم الذاتية المعاد ترتيبها
!>          لـ T. القيم الذاتية مخزنة بنفس ترتيبها على القطر لـ T، مع WR(i) = T(i,i) و، إذا
!>          كان T(i:i+1,i:i+1) كتلة قطرية 2x2، فإن WI(i) > 0 و
!>          WI(i+1) = -WI(i). لاحظ أنه إذا كانت القيمة الذاتية المعقدة
!>          غير مستقرة بما فيه الكفاية، فقد تختلف قيمتها بشكل كبير
!>          عن قيمتها قبل إعادة الترتيب.
!>

M

!>          M هو INTEGER
!>          بعد الفضاء الجزئي الثابت المحدد.
!>          0 < = M <= N.
!>

S

!>          S هو DOUBLE PRECISION
!>          إذا كان JOB = 'E' أو 'B'، فإن S هو حد أدنى على المعكوس
!>          للرقم الشرطي للمجموعة المختارة من القيم الذاتية.
!>          لا يمكن لـ S التقليل من تقدير الرقم الشرطي المعكوس الحقيقي
!>          بأكثر من عامل sqrt(N). إذا كان M = 0 أو N، فإن S = 1.
!>          إذا كان JOB = 'N' أو 'V'، لا يتم الرجوع إلى S.
!>

SEP

!>          SEP هو DOUBLE PRECISION
!>          إذا كان JOB = 'V' أو 'B'، فإن SEP هو تقدير المعكوس
!>          للرقم الشرطي للفضاء الجزئي الثابت المحدد. إذا
!>          كان M = 0 أو N، فإن SEP = norm(T).
!>          إذا كان JOB = 'N' أو 'E'، لا يتم الرجوع إلى SEP.
!>

WORK

!>          WORK هي مصفوفة DOUBLE PRECISION، أبعادها (MAX(1,LWORK))
!>          عند الخروج، إذا كان INFO = 0، فإن WORK(1) يعيد LWORK الأمثل.
!>

LWORK

!>          LWORK هو INTEGER
!>          بعد المصفوفة WORK.
!>          إذا كان JOB = 'N'، LWORK >= max(1,N)؛
!>          إذا كان JOB = 'E'، LWORK >= max(1,M*(N-M))؛
!>          إذا كان JOB = 'V' أو 'B'، LWORK >= max(1,2*M*(N-M)).
!>
!>          إذا كان LWORK = -1، فسيتم افتراض استعلام عن مساحة العمل؛ الروتين
!>          يحسب فقط الحجم الأمثل للمصفوفة WORK، ويعيد
!>          هذه القيمة كمدخل أول للمصفوفة WORK، ولا يتم إصدار رسالة خطأ
!>          متعلقة بـ LWORK بواسطة XERBLA.
!>

IWORK

!>          IWORK هي مصفوفة INTEGER، أبعادها (MAX(1,LIWORK))
!>          عند الخروج، إذا كان INFO = 0، فإن IWORK(1) يعيد LIWORK الأمثل.
!>

LIWORK

!>          LIWORK هو INTEGER
!>          بعد المصفوفة IWORK.
!>          إذا كان JOB = 'N' أو 'E'، LIWORK >= 1؛
!>          إذا كان JOB = 'V' أو 'B'، LIWORK >= max(1,M*(N-M)).
!>
!>          إذا كان LIWORK = -1، فسيتم افتراض استعلام عن مساحة العمل؛ الـ
!>          روتين يحسب فقط الحجم الأمثل للمصفوفة IWORK،
!>          ويعيد هذه القيمة كمدخل أول للمصفوفة IWORK، ولا يتم إصدار رسالة خطأ
!>          متعلقة بـ LIWORK بواسطة XERBLA.
!>

INFO

!>          INFO هو INTEGER
!>          = 0: خروج ناجح
!>          < 0: إذا كان INFO = -i، فإن المعامل i كان لديه قيمة غير قانونية
!>          = 1: فشلت إعادة ترتيب T لأن بعض القيم الذاتية قريبة جدًا
!>               من الفصل (المشكلة غير مستقرة جدًا)؛
!>               قد تم إعادة ترتيب T جزئيًا، وتحتوي WR و WI
!>               على القيم الذاتية بنفس ترتيبها في T؛ يتم تعيين S و
!>               SEP (إذا طلبت) إلى صفر.
!>

!>
!>  تقوم DTRSEN أولاً بجمع القيم الذاتية المحددة عن طريق حساب تحويل متعامد Z لنقلها
!>  إلى الزاوية العلوية اليسرى من T.
!>  بمعنى آخر، القيم الذاتية المحددة هي القيم الذاتية لـ T11 في:
!>
!>          Z**T * T * Z = ( T11 T12 ) n1
!>                         (  0  T22 ) n2
!>                            n1  n2
!>
!>  حيث N = n1+n2 و Z**T تعني منقول Z. الأعمدة N1 الأولى
!>  لـ Z تمتد الفضاء الجزئي الثابت المحدد لـ T.
!>
!>  إذا تم الحصول على T من تحليل شور الحقيقي للمصفوفة
!>  A = Q*T*Q**T، فإن تحليل شور الحقيقي المعاد ترتيبه لـ A يُعطى بـ
!>  A = (Q*Z)*(Z**T*T*Z)*(Q*Z)**T، والأعمدة N1 الأولى لـ Q*Z تمتد
!>  الفضاء الجزئي الثابت المقابل لـ A.
!>
!>  قد يتم إرجاع الرقم الشرطي المعكوس لمتوسط القيم الذاتية لـ
!>  T11 في S. S يقع بين 0 (غير مستقر جدًا)
!>  و 1 (مستقر جدًا). يتم حسابه على النحو التالي. أولاً نحسب
!>  R بحيث
!>
!>                         P = ( I  R ) n1
!>                             ( 0  0 ) n2
!>                               n1 n2
!>
!>  هو المتروب لـ الفضاء الجزئي الثابت المرتبط بـ T11.
!>  R هو حل معادلة سيلفستر:
!>
!>                        T11*R - R*T22 = T12.
!>
!>  لنفترض أن F-norm(M) يشير إلى معيار Frobenius لـ M و 2-norm(M) يشير إلى
!>  المعيار 2 لـ M. ثم يتم حساب S كحد أدنى
!>
!>                      (1 + F-norm(R)**2)**(-1/2)
!>
!>  على المعكوس لـ 2-norm(P)، وهو الرقم الشرطي المعكوس الحقيقي.
!>  لا يمكن لـ S التقليل من تقدير 1 / 2-norm(P) بأكثر من عامل
!>  sqrt(N).
!>
!>  حد خطأ تقريبي لمتوسط القيم الذاتية المحسوبة لـ
!>  T11 هو
!>
!>                         EPS * norm(T) / S
!>
!>  حيث EPS هو دقة الآلة.
!>
!>  يتم إرجاع الرقم الشرطي المعكوس للفضاء الجزئي الثابت الأيمن
!>  الممتد بواسطة الأعمدة N1 الأولى لـ Z (أو لـ Q*Z) في SEP.
!>  يتم تعريف SEP كفصل T11 و T22:
!>
!>                     sep( T11, T22 ) = sigma-min( C )
!>
!>  حيث sigma-min(C) هو أصغر قيمة فردية لـ
!>  مصفوفة n1*n2-by-n1*n2
!>
!>     C  = kprod( I(n2), T11 ) - kprod( transpose(T22), I(n1) )
!>
!>  I(m) هي مصفوفة الوحدة m x m، و kprod تشير إلى منتج كرونكر. نقدر
!>  sigma-min(C) بالمعكوس
