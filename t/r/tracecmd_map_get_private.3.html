tracecmd_map_get_private(3)

الحزمة: trace-cmd

التراخيص: GPL-2.0-only, LGPL-2.1-only

LIBTRACECMD(3) دليل libtracefs LIBTRACECMD(3)

الاسم

tracecmd_map_vcpus, tracecmd_get_cpu_map, tracecmd_map_find_by_host_pid, tracecmd_map_get_host_pid, tracecmd_map_get_guest, tracecmd_map_set_private, tracecmd_map_get_private - ربط بيانات المضيف والضيف

الملخص

#include <trace-cmd.h>
int tracecmd_map_vcpus(struct tracecmd_input **handles, int nr_handles);
struct tracecmd_cpu_map *tracecmd_get_cpu_map(struct tracecmd_input *handle, int cpu);
struct tracecmd_cpu_map *tracecmd_map_find_by_host_pid(struct tracecmd_input *handle,
                                                      int host_pid);
int tracecmd_map_get_host_pid(struct tracecmd_cpu_map *map);
struct tracecmd_input *tracecmd_map_get_guest(struct tracecmd_cpu_map *map);
void tracecmd_map_set_private(struct tracecmd_cpu_map *map, void *priv);
void *tracecmd_map_get_private(struct tracecmd_cpu_map *map);

الوصف

تُستخدم هذه المجموعة من واجهات برمجة التطبيقات لربط ملفات تتبع المضيف والضيف لتسهيل تحليل التتبع الإضافي.

تأخذ الدالة tracecmd_map_vcpus() مصفوفة من المقابض حيث تم إنشاء كل عنصر في تلك المصفوفة بواسطة أحد وظائف tracecmd_open(3)، وعدد المقابض كـ nr_handles. يُتوقع أن يكون المقبض الأول في مصفوفة المقابض هو واصف ملف تتبع المضيف، والباقي هي ملفات تتبع الضيف التي تعمل على المضيف، والتي تم إنشاؤها بواسطة تفاعلات trace-cmd record(1) و trace-cmd agent(1). تُرجع عدد الضيوف الموجودين في المقابض الذين تم ربطهم بالمضيف، أو قيمة سالبة في حالة حدوث خطأ.

تُرجع الدالة tracecmd_get_cpu_map() واصفًا لوحدة المعالجة المركزية (CPU) معينة لمقبض. إذا كان المقبض ضيفًا تم تعريفه من tracecmd_map_vcpus()، فإن الربط الذي تم إنشاؤه بواسطة تلك الدالة والذي يرتبط بوحدة المعالجة المركزية الافتراضية (vCPU) المحددة (يشار إليها بواسطة cpu) من المقبض. يمكن استخدام هذا الواصف بواسطة وظائف tarcecmd_map_get_guest() و tracecmd_map_set_private() و tracecmd_map_get_private().

ستُرجع الدالة tracecmd_map_find_by_host_pid() ربطًا لوحدة المعالجة المركزية الافتراضية للضيف التي يتم التعامل معها بواسطة host_pid المحدد. لاحظ أن المقبض المُمرر يمكن أن يكون إما مقبض المضيف أو أحد مقبض الضيف لذلك المضيف الذي تم ربطه بواسطة tracecmd_map_vcpus()، حتى لو كان مقبض الضيف لا يحتوي على وحدة المعالجة المركزية الافتراضية التي يمثلها host_pid.

ستُرجع الدالة tracecmd_map_get_host_pid() معرف العملية (host_pid) للربط المعطى بواسطة map والذي تم استرجاعه بواسطة إحدى الوظائف المذكورة أعلاه.

ستُرجع الدالة tracecmd_map_get_guest() مقبض الضيف للربط المعطى بواسطة map والذي تم استرجاعه بواسطة إحدى الوظائف المذكورة أعلاه.

تسمح الدالة tracecmd_map_set_private() للتطبيق بتعيين بيانات خاصة لوحدة المعالجة المركزية الافتراضية للضيف مقابل ربط خيط المضيف المحدد بواسطة map.

تسترجع الدالة tracecmd_map_get_private() البيانات الخاصة (priv) من map التي تم تعيينها بواسطة tracecmd_map_set_private().

قيمة الإرجاع

تُرجع الدالة tracecmd_map_vcpus() عدد الضيوف في مصفوفة المقابض التي تم ربطها بمقبض المضيف الذي هو الإدخال الأول في المقابض. تُرجع -1 في حالة حدوث خطأ.

تُرجع الدالة tracecmd_get_cpu_map() ربطًا تم إنشاؤه بواسطة tracecmd_map_vcpus() لوحدة معالجة مركزية (CPU) معينة لـ handle معين، أو NULL إذا لم يتم العثور عليه.

تُرجع الدالة tracecmd_map_find_by_host_pid() ربطًا يرتبط بمهمة المضيف بمعرف العملية host_pid. يمكن أن يكون handle إما مقبض المضيف، أو أحد مقبض الضيف الذي تم ربطه بالمضيف عبر tracecmd_map_vcpus()، حتى لو كان مقبض الضيف ضيفًا آخر غير الضيف الذي تم إنشاء الربط من أجله. تُرجع NULL إذا لم يتم العثور عليه.

تُرجع الدالة tracecmd_map_get_host_pid() معرف عملية المضيف للربط المرتبط المحدد بواسطة map.

تُرجع الدالة tracecmd_map_get_guest() مقبض الضيف للربط المرتبط المحدد بواسطة map.

تُرجع الدالة tracecmd_map_get_private() البيانات الخاصة للربط المحدد بواسطة map والذي تم تعيينه بواسطة tracecmd_map_set_private().

مثال

#include <stdlib.h>
#include <errno.h>
#include <trace-cmd.h>
int main(int argc, char **argv)
{
        struct tracecmd_input **handles = NULL;
        int nr_handles;
        int i;
        if (argc < 2) {
                printf("usage: host_trace.dat guest1_trace.dat [guest2_trace.dat ...]\n");
                exit(-1);
        }
        for (i = 1; i < argc; i++) {
                handles = realloc(handles, sizeof(*handles) * (nr_handles + 1));
                if (!handles)
                        exit(-1);
                handles[nr_handles] = tracecmd_open(argv[i], 0);
                if (!handles[nr_handles]) {
                        perror(argv[1]);
                        exit(-1);
                }
                tracecmd_set_private(handles[nr_handles], argv[i]);
                nr_handles++;
        }
        tracecmd_map_vcpus(handles, nr_handles);
        for (i = 1; i < nr_handles; i++) {
                struct tracecmd_cpu_map *map;
                struct tep_handle *tep;
                const char *file = tracecmd_get_private(handles[i]);
                int cpus, cpu;
                printf("Mappings for guest %s:\n", file);
                tep = tracecmd_get_tep(handles[i]);
                cpus = tep_get_cpus(tep);
                for (cpu = 0; cpu < cpus; cpu++) {
                        printf("  [%03d] ", cpu);
                        map = tracecmd_get_cpu_map(handles[i], cpu);
                        if (!map) {
                                printf("Has no mapping!\n");
                                continue;
                        }
                        printf("host_pid: %d\n", tracecmd_map_get_host_pid(map));
                }
        }
        for (i = 0; i < nr_handles; i++)
                tracecmd_close(handles[i]);
        free(handles);
        exit(0);
}
