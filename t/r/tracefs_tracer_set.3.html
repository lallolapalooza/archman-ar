tracefs_tracer_set(3)

الحزمة: libtracefs

التراخيص: GPL-2.0-only, LGPL-2.1-only

LIBTRACEFS(3) دليل libtracefs LIBTRACEFS(3)

الاسم

tracefs_instance_tracers, tracefs_tracer_set, tracefs_tracer_clear - تمكين أو تعطيل متعقب في نسخة أو المستوى العلوي

الملخص

#include <tracefs.h>
char ** tracefs_instance_tracers(struct tracefs_instance *instance);
int tracefs_tracer_set(struct tracefs_instance *instance, enum tracefs_tracers tracer);
int tracefs_tracer_set(struct tracefs_instance *instance, enum tracefs_tracers tracer, const char *name);
int tracefs_tracer_clear(struct tracefs_instance *instance);

الوصف

tracefs_instance_tracers سترجع قائمة بالمتعقبات المتاحة لنسخة معينة instance (لاحظ، قد لا يكون للنسخة نفس مجموعة المتعقبات المتاحة مثل المستوى العلوي). إذا كانت instance NULL، فإن قائمة المتعقبات المتاحة المُرجعة ستكون للمستوى العلوي.

tracefs_tracer_set يمكّن متعقبًا في النسخة المعطاة، المُعرّفة بواسطة المعامل instance. إذا كانت instance NULL، فإن نسخة المستوى العلوي يتم تغييرها. إذا تم تعيين tracer إلى TRACFES_TRACER_CUSTOM فيجب تمرير سلسلة name كمعامل ثالث، وهذه تُكتب في النسخة لتمكين المتعقب بهذا الاسم. هذا مفيد للنواة الأحدث أو المخصصة التي تحتوي على متعقبات لم يتم تحديدها بعد بواسطة تعداد tracefs_tracers.

tracefs_tracer_clear يعطّل المتعقب للنسخة المعطاة المُعرّفة بالمتغير instance، أو نسخة المستوى العلوي إذا كانت NULL. هذا يعادل استدعاء tracefs_tracer_set باستخدام TRACEFS_TRACER_NOP كمعامل tracer.

تعدادات TRACEFS_TRACER

التعدادات المعرفة حاليًا والتي تم قبولها هي:

TRACEFS_TRACER_NOP : هذا هو المتعقب الخامل، الذي لا يفعل شيئًا ويستخدم لمسح أي متعقب نشط.

TRACEFS_TRACER_FUNCTION : يُمكّن معظم الوظائف في النواة للتتبع.

TRACEFS_TRACER_FUNCTION_GRAPH : يُمكّن معظم الوظائف في النواة للتتبع وكذلك عودة الوظيفة.

TRACEFS_TRACER_IRQSOFF : يتتبع زمن تأخير المقاطعات المعطلة.

TRACEFS_TRACER_PREEMPTOFF : يتتبع زمن تأخير الاستباق المعطل (الوقت في النواة الذي لا يمكن فيه جدولة المهام من وحدة المعالجة المركزية).

TRACEFS_TRACER_PREEMPTIRQSOFF : يتتبع إجمالي زمن التأخير عند تعطيل المقاطعات وكذلك عند تعطيل الاستباق.

TRACEFS_TRACER_WAKEUP : يتتبع زمن التأخير الذي تستغرقه المهمة ذات الأولوية الأعلى للاستيقاظ.

TRACEFS_TRACER_WAKEUP_RT : يتتبع زمن التأخير الذي تستغرقه المهمة ذات الأولوية الأعلى في الوقت الفعلي للاستيقاظ. يتم تجاهل جميع المهام الأخرى.

TRACEFS_TRACER_WAKEUP_DL : يتتبع زمن التأخير الذي تستغرقه المهمة ذات الأولوية الأعلى DEADLINE للاستيقاظ. يتم تجاهل جميع المهام الأخرى.

TRACEFS_TRACER_MMIOTRACE : يتتبع تفاعل الأجهزة مع النواة.

TRACEFS_TRACER_HWLAT : يكتشف زمن التأخير الناتج عن الأجهزة الخارجة عن نطاق النواة.

TRACEFS_TRACER_BRANCH : يتتبع عند اتخاذ الفروع المحتملة أو غير المحتملة.

TRACEFS_TRACER_BLOCK : متعقب خاص لأجهزة الكتل.

لاحظ أن المتعقبات المذكورة أعلاه قد لا تكون متاحة في النواة و tracefs_tracer_set() سترجع خطأ مع تعيين errno إلى ENODEV، إذا كانت النواة لا تدعم خيار tracer، أو الخيار المخصص إذا تم استخدام TRACEFS_TRACER_CUSTOM.

قيمة الإرجاع

ترجع 0 عند النجاح، أو -1 عند الخطأ.

الأخطاء

tracefs_tracer_set() قد تفشل مع الأخطاء التالية:

EINVAL المعامل tracer خارج نطاق ما تم تعريفه.

ENOMEM خطأ في تخصيص الذاكرة.

ENOENT المتعقبات غير مدعومة في النواة قيد التشغيل.

ENODEV المتعقب المحدد غير مدعوم في النواة قيد التشغيل.

قد تحدث أخطاء أخرى ناتجة عن استدعاءات نظام داخلية.

مثال

#include <stdlib.h>
#include <stdio.h>
#include <getopt.h>
#include <errno.h>
#include <tracefs.h>
int main(int argc, char *argv[])
{
        struct tracefs_instance *inst = NULL;
        enum tracefs_tracers t = TRACEFS_TRACER_NOP;
        const char *cust = NULL;
        const char *buf = NULL;
        char **tracers;
        int ret;
        int ch;
        int i;
        while ((ch = getopt(argc, argv, "nfgiwdc:B:")) > 0) {
                switch (ch) {
                case 'f': t = TRACEFS_TRACER_FUNCTION; break;
                case 'g': t = TRACEFS_TRACER_FUNCTION_GRAPH; break;
                case 'i': t = TRACEFS_TRACER_PREEMPTIRQSOFF; break;
                case 'w': t = TRACEFS_TRACER_WAKEUP_RT; break;
                case 'd': t = TRACEFS_TRACER_WAKEUP_DL; break;
                case 'c':
                        t = TRACEFS_TRACER_CUSTOM;
                        cust = optarg;
                        break;
                case 'B':
                        buf = optarg;
                        break;
                case 'n':
                        /* nop */
                        break;
                default:
                        printf("Unknow arg %c\n", ch);
                        exit(-1);
                }
        }
        if (buf) {
                inst = tracefs_instance_create(buf);
                if (!inst) {
                        printf("failed to create instance\n");
                        exit(-1);
                }
        }
        if (t == TRACEFS_TRACER_CUSTOM)
                ret = tracefs_tracer_set(inst, t, cust);
        else
                ret = tracefs_tracer_set(inst, t);
        if (ret < 0) {
                if (errno == ENODEV) {
                        if (cust)
                                printf("Tracer '%s' not supported by kernel\n", cust);
                        else
                                printf("Tracer not supported by kernel\n");
                        tracers = tracefs_instance_tracers(inst);
                        printf("Available tracers:");
                        for (i = 0; tracers && tracers[i]; i++)
                                printf(" %s", tracers[i]);
                        tracefs_list_free(tracers);
                        printf("\n");
                } else
                        perror("Error");
                if (inst) {
                        tracefs_instance_destroy(inst);
                        tracefs_instance_free(inst);
                }
                exit(-1);
        }
        tracefs_instance_free(inst);
        exit(0);
}
