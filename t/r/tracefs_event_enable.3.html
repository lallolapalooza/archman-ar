tracefs_event_enable(3)

الحزمة: libtracefs

التراخيص: GPL-2.0-only, LGPL-2.1-only

LIBTRACEFS(3) دليل libtracefs LIBTRACEFS(3)

الاسم

tracefs_event_systems, tracefs_system_events, tracefs_event_enable, tracefs_event_disable, tracefs_event_is_enabled - العمل مع أنظمة وأحداث التتبع.

ملخص

#include <tracefs.h>
enum tracefs_event_state {
        TRACEFS_ERROR = -1,
        TRACEFS_ALL_DISABLED = 0,
        TRACEFS_ALL_ENABLED = 1,
        TRACEFS_SOME_ENABLED = 2,
};
char **tracefs_event_systems(const char *tracing_dir);
char **tracefs_system_events(const char *tracing_dir, const char *system);
int tracefs_event_enable(struct tracefs_instance *instance, const char *system,
                           const char *event);
int tracefs_event_disable(struct tracefs_instance *instance, const char *system,
                            const char *event);
enum tracefs_enable_state tracefs_event_is_enabled(struct tracefs_instance *instance,
                         const char *system, const char *event);

الوصف

واجهات برمجة التطبيقات المتعلقة بأنظمة وأحداث التتبع.

تقوم الدالة tracefs_event_systems() بإرجاع مصفوفة من السلاسل النصية بأسماء جميع أنظمة التتبع المسجلة، الموجودة في الدليل tracing_dir المحدد. يمكن أن يكون هذا NULL أو موقع نقطة تحميل tracefs لأنظمة التتبع للجهاز المحلي، أو قد يكون مسارًا إلى نسخة من دليل tracefs من جهاز آخر. العنصر الأخير في المصفوفة هو مؤشر NULL. يجب تحرير المصفوفة باستخدام واجهة برمجة التطبيقات tracefs_list_free().

تقوم الدالة tracefs_system_events() بإرجاع مصفوفة من السلاسل النصية بأسماء جميع أحداث التتبع المسجلة لنظام التتبع المحدد بواسطة system، الموجودة في الدليل tracing_dir المحدد. يمكن أن يكون هذا NULL أو موقع نقطة تحميل tracefs لأنظمة التتبع للجهاز المحلي، أو قد يكون مسارًا إلى نسخة من دليل tracefs من جهاز آخر. العنصر الأخير في المصفوفة هو مؤشر NULL. يجب تحرير المصفوفة باستخدام واجهة برمجة التطبيقات tracefs_list_free().

تقوم الدالة tracefs_event_enable() بتمكين حدث محدد بناءً على system و event المقدمين للمثيل instance المحدد. إذا كان instance هو NULL، فسيتم استخدام دليل التتبع العلوي. إذا كان كل من system و event هو NULL، فسيتم تمكين جميع الأحداث للمثيل instance. إذا كان event هو NULL، فسيتم تمكين جميع الأحداث داخل النظام system. إذا كان system هو NULL، فسيتم البحث في جميع الأنظمة وسيتم تمكين أي حدث داخل نظام يطابق event. يمكن أن يكون كل من system و event تعبيرات عادية كما هو محدد في regex(3).

تقوم الدالة tracefs_event_disable() بتعطيل الأحداث التي تطابق معلمات system و event للمثيل instance المحدد. تتبع الأحداث التي يتم تعطيلها نفس قواعد tracefs_event_enable() لمطابقة الأحداث. هذا يعني، إذا كان instance هو NULL، فسيتم استخدام دليل التتبع العلوي. إذا كان كل من system و event هو NULL، فسيتم تعطيل جميع الأحداث للمثيل المحدد، وهكذا.

تقوم الدالة tracefs_event_is_enabled() بإرجاع ما إذا كان الحدث ممكّنًا، أو مجموعة من الأحداث ممكّنة، أو نظام ممكّن، أو جميع الأحداث ممكّنة. إذا كان كل من system و event هو NULL، فسترجع حالة تمكين جميع الأحداث. إذا كان system غير NULL و event هو NULL، فستقوم بفحص ما إذا كانت جميع الأحداث في جميع الأنظمة التي يطابقها system وترجع حالة تمكين تلك الأحداث. إذا كان system هو NULL و event غير NULL، فسيقوم بمطابقة جميع الأحداث في جميع الأنظمة التي تطابق event وإرجاع حالة تمكينها. إذا كان كل من system و event غير NULL، فسترجع حالة تمكين جميع الأحداث المطابقة. حالة التمكين محددة على النحو التالي:

TRACEFS_ERROR - حدث خطأ بما في ذلك عدم مطابقة أي حدث.

TRACEFS_ALL_DISABLED - تم تعطيل جميع الأحداث المطابقة.

TRACEFS_ALL_ENABLED - تم تمكين جميع الأحداث المطابقة.

TRACEFS_SOME_ENABLED - تم تمكين بعض الأحداث المطابقة بينما لم يتم تمكين البعض الآخر.

قيمة الإرجاع

تقوم الدوال tracefs_event_systems() و tracefs_system_events() بإرجاع مصفوفة من السلاسل النصية. العنصر الأخير في تلك المصفوفة هو مؤشر NULL. يجب تحرير المصفوفة بواسطة واجهة برمجة التطبيقات tracefs_list_free(). في حالة حدوث خطأ، يتم إرجاع NULL.

تقوم كل من tracefs_event_enable() و tracefs_event_disable() بإرجاع 0 إذا وجدت أي أحداث مطابقة (ملاحظة: لا تتحقق من الحالة السابقة للحدث. إذا وجدت tracefs_event_enable() حدثًا ممكّنًا بالفعل، ولا توجد أخطاء أخرى، فسترجع 0). إذا حدث خطأ، حتى لو تم العثور على أحداث أخرى، فسترجع -1 وسيتم تعيين errno. إذا لم تحدث أخطاء، ولكن لم يتم العثور على أحداث تطابق معلمات system و event، فسيتم إرجاع -1 ولن يتم تعيين errno.

تقوم الدالة tracefs_event_is_enabled() بإرجاع حالة التمكين للأحداث المطابقة أو TRACEFS_ERROR عند حدوث خطأ.

مثال

#include <tracefs.h>
char **systems = tracefs_event_systems(NULL);
        if (systems) {
                int i = 0;
                /* تم الحصول على أنظمة التتبع المسجلة من مثيل التتبع العلوي */
                while (systems[i]) {
                        char **events = tracefs_system_events(NULL, systems[i]);
                        if (events) {
                                /* تم الحصول على الأحداث المسجلة في system[i] من مثيل التتبع العلوي */
                                int j = 0;
                                while (events[j]) {
                                        /* تم الحصول على event[j] في system[i] من مثيل التتبع العلوي */
                                        j++;
                                }
                                tracefs_list_free(events);
                        }
                        i++;
                }
                tracefs_list_free(systems);
        }
