tracefs_instance_get_trace_dir(3)

الحزمة: libtracefs

التراخيص: GPL-2.0-only, LGPL-2.1-only

LIBTRACEFS(3) دليل libtracefs LIBTRACEFS(3)

الاسم

tracefs_instance_get_name, tracefs_instance_get_trace_dir, tracefs_instances_walk, tracefs_instance_exists, tracefs_instance_get_buffer_size, tracefs_instance_set_buffer_size, tracefs_instance_get_buffer_percent, tracefs_instance_set_buffer_percent - وظائف مساعدة للعمل مع مثيلات التتبع.

موجز

#include <tracefs.h>
const char *tracefs_instance_get_name(struct tracefs_instance *instance);
const char *tracefs_instance_get_trace_dir(struct tracefs_instance *instance);
int tracefs_instances_walk(int (*callback)(const char *, void *), void *context);
bool tracefs_instance_exists(const char *name);
size_t tracefs_instance_get_buffer_size(struct tracefs_instance *instance, int cpu);
int tracefs_instance_set_buffer_size(struct tracefs_instance *instance, size_t size, int cpu);
int tracefs_instance_get_buffer_percent(struct tracefs_instance *instance);
int tracefs_instance_set_buffer_percent(struct tracefs_instance *instance, int val);

الوصف

وظائف مساعدة للعمل مع مثيلات التتبع.

تقوم الدالة tracefs_instance_get_name() بإرجاع اسم instance المحدد. لاحظ أن المثيل العلوي ليس له اسم، وتقوم الدالة بإرجاع NULL له.

تقوم الدالة tracefs_instance_get_trace_dir() بإرجاع دليل التتبع، حيث تم تكوين instance المحدد.

تقوم الدالة tracefs_instances_walk() بالمرور عبر جميع مثيلات التتبع المكونة في النظام واستدعاء callback لكل واحد منها. يتم تمرير الوسيط context إلى callback، مع اسم المثيل. إذا أعاد callback قيمة غير صفرية، تتوقف العملية. ملاحظة، لا يتم استدعاء callback للمثيل العلوي.

تقوم الدالة tracefs_instance_exists() بالتحقق مما إذا كان مثيل بالاسم name المحدد موجودًا في النظام.

تقوم الدالة tracefs_instace_get_buffer_size() بإرجاع حجم المخزن المؤقت الدائري. إذا كانت cpu سالبة، فإنها ترجع الحجم الإجمالي لجميع المخازن المؤقتة الدائرية لكل وحدة معالجة مركزية، وإلا فإنها ترجع حجم المخزن المؤقت الدائري لكل وحدة معالجة مركزية لـ cpu.

تقوم الدالة tracefs_instance_set_buffer_size() بتعيين حجم المخزن المؤقت الدائري. إذا كانت cpu سالبة، فإنها تعين جميع المخازن المؤقتة الدائرية لكل وحدة معالجة مركزية إلى size (لاحظ أن الحجم الإجمالي هو عدد وحدات المعالجة المركزية * size). إذا تم تحديد cpu، فإنها تعين فقط حجم المخزن المؤقت الدائري لكل وحدة معالجة مركزية.

تقوم الدالة tracefs_instance_set_buffer_percent() بتعيين قيمة نسبة المخزن المؤقت لمخزن التتبع الدائري لـ instance أو المخزن المؤقت للمستوى الأعلى إذا كان instance هو NULL. تحدد نسبة المخزن المؤقت متى سيتم حظر القراء على tracefs_cpu_read(3)، tracefs_cpu_buffered_read(3)، tracefs_cpu_write(3) و tracefs_cpu_pipe(3) عند عدم تعيين O_NONBLOCK. يجب أن تكون قيمة val بين 0 و 100، حيث:

  0   - حظر حتى تكون هناك أي بيانات في المخزن المؤقت الدائري
  1   - حظر حتى يتم ملء 1% من المخازن الفرعية للمخزن المؤقت الدائري
  50  - حظر حتى يتم ملء 50% من المخازن الفرعية للمخزن المؤقت الدائري
  100 - حظر حتى يتم ملء المخزن المؤقت الدائري بالكامل

ملاحظة، يمكن استخدام أي رقم من 0 إلى 100 حيث تكون النسبة المئوية للمخزن المؤقت التي يجب ملؤها قبل إخطار القارئ المحظور بوجود بيانات لاستردادها.

تقوم الدالة tracefs_instance_get_buffer_percent() باسترداد إعداد نسبة المخزن المؤقت الحالي لمخزن التتبع الدائري لـ instance أو المخزن المؤقت للمستوى الأعلى إذا كان instance هو NULL.

قيمة الإرجاع

ترجع الدالة tracefs_instance_get_name() سلسلة نصية أو NULL في حالة المثيل العلوي. يجب *عدم* تحرير السلسلة النصية المرتجعة.

ترجع الدالة tracefs_instance_get_trace_dir() سلسلة نصية أو NULL في حالة حدوث خطأ. يجب *عدم* تحرير السلسلة النصية المرتجعة.

ترجع الدالة tracefs_instances_walk() القيمة 0، إذا تم التكرار على جميع المثيلات، 1 إذا تم إيقاف التكرار بواسطة callback، أو -1 في حالة حدوث خطأ.

ترجع الدالة tracefs_instance_exists() true إذا كان مثيل بالاسم name المحدد موجودًا في النظام أو false بخلاف ذلك.

ترجع الدالة tracefs_instance_get_buffer_size() حجم المخزن المؤقت الدائري اعتمادًا على قيمة cpu الممررة، أو -1 عند حدوث خطأ.

ترجع الدالة tracefs_instance_set_buffer_size() صفرًا عند النجاح و -1 عند حدوث خطأ.

مثال

#include <tracefs.h>
struct tracefs_instance *inst;
....
char *name = tracefs_instance_get_name(inst);
        if(name) {
                /* تم الحصول على اسم المثيل */
        }
char *dir = tracefs_instance_get_trace_dir(inst);
        if(dir) {
                /* تم الحصول على دليل التتبع للمثيل */
        }
...
static int instance_walk(char *name, void *context)
{
        /* تم الحصول على المثيل بالاسم */
        return 0;
}
...
        if (tracefs_instances_walk(instance_walk, NULL) < 0) {
                /* خطأ أثناء المرور عبر المثيلات */
        }
...
        if (tracefs_instance_exists("foo")) {
                /* يوجد مثيل بالاسم foo في النظام */
        } else {
                /* لا يوجد مثيل بالاسم foo في النظام */
        }
