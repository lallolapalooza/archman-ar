tracecmd_open(3)

الحزمة: trace-cmd

التراخيص: GPL-2.0-only, LGPL-2.1-only

LIBTRACECMD(3) دليل libtracefs LIBTRACECMD(3)

الاسم

tracecmd_open, tracecmd_open_fd, tracecmd_open_head, tracecmd_init_data, tracecmd_close, tracecmd_set_private, tracecmd_get_private - فتح وإغلاق ملف تتبع.

الملخص

#include <trace-cmd.h>
struct tracecmd_input *tracecmd_open(const char *file, int flags);
struct tracecmd_input *tracecmd_open_fd(int fd, int flags);
struct tracecmd_input *tracecmd_open_head(const char *file, int flags);
int tracecmd_init_data(struct tracecmd_input *handle);
void tracecmd_close(struct tracecmd_input *handle);
void tracecmd_set_private(struct tracecmd_input *handle, void *data);
void *tracecmd_get_private(struct tracecmd_input *handle);

الوصف

يمكن استخدام مجموعة الواجهات البرمجية هذه لفتح وإغلاق ملف تتبع مسجل بواسطة trace-cmd(1) ويحتوي على معلومات تتبع من ftrace، متعقب نواة لينكس الرسمي. يمثل الملف المفتوح هيكل tracecmd_input، وجميع الواجهات البرمجية الأخرى التي تعمل مع الملف تتطلب مؤشرًا إلى الهيكل. تحتوي الواجهات البرمجية لفتح ملف تتبع على معلمة flag (علم)، تتحكم في كيفية فتح الملف وتحليله. الflag هو مزيج من هذه الخيارات:

TRACECMD_FL_LOAD_NO_PLUGINS - لا تقم بتحميل أي إضافات
TRACECMD_FL_LOAD_NO_SYSTEM_PLUGINS - لا تقم بتحميل الإضافات على مستوى النظام، قم فقط بتحميل الإضافات "المحلية فقط"
                                       من الدليل الرئيسي للمستخدم.

تقوم الدالة tracecmd_open() بفتح ملف التتبع المحدد file، وتحليل رؤوس البيانات الوصفية من الملف، وتخصيص وتهيئة هيكل معالج tracecmd_input يمثل الملف. كما أنها تهيئ المعالج لقراءة بيانات التتبع من الملف. المعالج الذي تم إرجاعه جاهز للاستخدام مع واجهات tracecmd_read_.

تقوم الدالة tracecmd_open_fd() بنفس عمل tracecmd_open()، ولكنها تعمل مع واصف ملف لملف تتبع، مفتوح للقراءة.

الدالة tracecmd_open_head() مماثلة لـ tracecmd_open()، ولكنها لا تهيئ المعالج لقراءة بيانات التتبع. تقوم بقراءة وتحليل رؤوس البيانات الوصفية فقط. يجب استخدام tracecmd_init_data() قبل استخدام واجهات tracecmd_read_.

تقوم الدالة tracecmd_init_data() بتهيئة handle، تم تخصيصه بواسطة tracecmd_open_head()، لقراءة بيانات التتبع من الملف المرتبط بها. يجب استدعاء هذه الواجهة البرمجية قبل أي من واجهات tracecmd_read_.

تقوم الدالة tracecmd_close() بتحرير handle، وهو مؤشر إلى هيكل tracecmd_input، تم تخصيصه مسبقًا بواسطة واجهات tracecmd_open() أو tracecmd_open_fd() أو tracecmd_open_head().

تسمح الدالة tracecmd_set_private() بإضافة data محددة إلى handle يمكن استردادها لاحقًا.

ستقوم الدالة tracecmd_get_private() باسترداد data التي تم تعيينها بواسطة tracecmd_set_private() لـ handle المحدد.

قيمة الإرجاع

تقوم الدوال tracecmd_open() وtracecmd_open_fd() و tracecmd_open_head() بإرجاع مؤشر إلى هيكل tracecmd_input أو NULL في حالة حدوث خطأ. يجب تحرير الهيكل الذي تم إرجاعه بواسطة tracecmd_close(). لاحظ أنه إذا تم استخدام tracecmd_open_fd() لتخصيص معالج tracecmd_input، فعند استدعاء tracecmd_close() لإغلاقه، سيتم إغلاق واصف الملف هذا أيضًا.

تعيد الدالة tracecmd_init_data() -1 في حالة حدوث خطأ أو 0 بخلاف ذلك.

تعيد الدالة tracecmd_get_private() data التي تم تعيينها بواسطة tracecmd_set_private().

مثال

هناك نمطان مختلفان للاستخدام لفتح وقراءة بيانات التتبع من ملف تتبع،
والتي يمكن استخدامها اعتمادًا على حالة الاستخدام.
1. فتح وتهيئة ملف التتبع في خطوة واحدة:
#include <stdlib.h>
#include <trace-cmd.h>
static int print_events(struct tracecmd_input *handle, struct tep_record *record, int cpu, void *data)
{
        static struct trace_seq seq;
        struct tep_handle *tep = tracecmd_get_tep(handle);
        const char *file = tracecmd_get_private(handle);
        if (!seq.buffer)
                trace_seq_init(&seq);
        trace_seq_reset(&seq);
        trace_seq_printf(&seq, "%s: ", file);
        trace_seq_terminate(&seq);
        trace_seq_do_printf(&seq);
        return 0;
}
int main(int argc, char **argv)
{
        struct tracecmd_input *handle;
        if (argc < 2) {
                printf("usage: %s trace.dat\n", argv[0]);
                exit(-1);
        }
        handle = tracecmd_open(argv[i], 0);
        if (!handle)
                exit(-1);
        tracecmd_set_private(handles[nr_handles], argv[i]);
        tracecmd_iterate_events(handles, NULL, 0, print_events, NULL);
        tracecmd_close(handle);
}
2. فتح وتهيئة ملف التتبع في خطوتين. هذا يسمح بإجراء بعض المعالجة بناءً على البيانات الوصفية، المقروءة من الملف، قبل تهيئة بيانات التتبع للقراءة. مثال لحالة استخدام كهذه هو عند فتح ملفات تتبع متعددة مسجلة في نفس جلسة التتبع. في هذه الحالة، يجب تعديل الطوابع الزمنية لجميع أحداث التتبع بناءً على المعلومات الواردة من البيانات الوصفية للملف وقبل قراءة بيانات التتبع.
#include <trace-cmd.h>
...
struct tracecmd_input *handle = tracecmd_open_head("trace.dat");
        if (!handle) {
                /* فشل في فتح ملف trace.dat */
        }
...
        /* قم ببعض المعالجة، قبل تهيئة بيانات التتبع للقراءة */
...
        if (tracecmd_init_data(handle) < 0) {
                /* فشل في تهيئة المعالج لقراءة بيانات التتبع */
        }
...
        /* قراءة بيانات التتبع من الملف، باستخدام المعالج */
...
        tracecmd_close(handle);
...

trace-cmd(1)

الاسم

trace-cmd – أداة سطر أوامر لتتبع نواة لينكس.

الملخص

trace-cmd [options] [command] [args]

الوصف

trace-cmd هي واجهة سطر أوامر سهلة الاستخدام لأداة تتبع نواة لينكس (ftrace). تتيح للمستخدمين تسجيل وتصور بيانات التتبع من نواة لينكس.

قيمة الإرجاع

عند تشغيل أمر، تعيد trace-cmd رمز خروج الأمر.

مثال

لتتبع حدث ‘sched_switch’ لمدة 10 ثوانٍ:

trace-cmd record -e sched_switch -d 10

الملفات

/sys/kernel/debug/tracing/
الدليل الرئيسي لنظام ملفات ftrace.

انظر أيضاً

trace-cmd-record(1), trace-cmd-report(1), trace-cmd-list(1), ftrace(7)
