tracefs_follow_event_clear(3)

الحزمة: libtracefs

التراخيص: GPL-2.0-only, LGPL-2.1-only

LIBTRACEFS(3) دليل libtracefs LIBTRACEFS(3)

tracefs_iterate_raw_events, tracefs_iterate_stop, tracefs_follow_event, tracefs_follow_missed_events, tracefs_follow_event_clear, tracefs_follow_missed_events_clear, tracefs_iterate_snapshot_events - التكرار فوق الأحداث في المخزن المؤقت الدائري

الوصف

المكرر tracefs فوق الأحداث الخام.

تقوم الدالة tracefs_iterate_raw_events() بقراءة المخازن المؤقتة للبيانات الخام لـ tracefs واستدعاء الدالة callback المحددة لكل حدث تصادفه. يتم التكرار فوق الأحداث بترتيب مرتب: الأقدم أولاً. يتطلب معالج tep مُهيأ (انظر tracefs_local_events(3)). إذا كان instance فارغًا، فسيتم استخدام المخزن المؤقت tracefs العلوي، وإلا فسيتم قراءة المخزن المؤقت للمثيل instance المقابل. لتصفية مجموعة فرعية فقط من وحدات المعالجة المركزية (CPUs)، يمكن تعيين cpus و cpu_size لاستدعاء callback فقط مع الأحداث التي وقعت على وحدات المعالجة المركزية المحددة، وإلا إذا كان cpus فارغًا، فسيتم استدعاء الدالة callback لجميع الأحداث، وسيتم تجاهل cpu_size. سيتم استدعاء الدالة callback بالمعلمات التالية: مؤشر إلى struct tep_event الذي يتوافق مع نوع الحدث؛ السجل الذي يمثل الحدث؛ وحدة المعالجة المركزية التي وقع عليها الحدث؛ ومؤشر إلى callback_context المحدد من قبل المستخدم. إذا أعادت callback قيمة غير صفرية، يتوقف التكرار.

تعمل الدالة tracefs_iterate_snapshot_events() بنفس طريقة عمل tracefs_iterate_raw_events() باستثناء أنها تعمل على المخزن المؤقت لللقطة.

استخدم tracefs_iterate_stop() لإجبار tracefs_iterate_raw_events() قيد التنفيذ على التوقف. يمكن استدعاء هذا من رد اتصال يتم استدعاؤه بواسطة المكرر (على الرغم من أن العودة بقيمة غير صفرية ستوقفه)، أو من مؤشر ترابط آخر.

تُستخدم الدالة tracefs_follow_event() مع tracefs_iterate_raw_events() ولكن بدلاً من استدعاء رد الاتصال لكل حدث، يتم استدعاؤه فقط للنظام system / اسم الحدث event_name المحدد للدالة. رد الاتصال callback هو نفسه الخاص بـ tracefs_iterate_raw_events()، وسيتم تمرير callback_context المُدخل إلى callback أيضًا. ملاحظة، إذا أعاد شيئًا آخر غير 0، فسيؤدي ذلك إلى إيقاف الحلقة قبل استدعاء callback الخاص بـ tracefs_iterate_raw_events().

ستقوم الدالة tracefs_follow_missed_events() باستدعاء callback عند اكتشاف أحداث فائتة. ستقوم بتعيين المعلمة record الخاصة برد الاتصال إلى السجل الذي جاء بعد الأحداث الفائتة، وسيحتوي event على نوع الحدث الذي يمثله record. سيتم تعيين cpu إلى وحدة المعالجة المركزية التي فاتتها الأحداث، وسيكون callback_data هو المحتوى الذي تم تمريره إلى الدالة.

ستقوم الدالة tracefs_follow_event_clear() بإزالة المتابعين من instance الذين يتطابقون مع system و event_name. إذا كان system و event_name كلاهما فارغين، فستقوم بإزالة جميع متابعين الأحداث المرتبطين بـ instance. إذا كان system فارغًا فقط، فستقوم بإزالة جميع المتابعين الذين يتابعون الأحداث التي تطابق event_name. إذا كان event_name فارغًا فقط، فستقوم بإزالة جميع المتابعين المتصلين بالأحداث التي تنتمي إلى نظام يتطابق مع system.

ستقوم الدالة tracefs_follow_missed_events_clear() بإزالة جميع المتابعين للأحداث الفائتة.

قيمة الإرجاع

تعيد الدالة tracefs_iterate_raw_events() القيمة -1 في حالة حدوث خطأ أو 0 بخلاف ذلك.

تعيد كلتا الدالتين tracefs_follow_event_clear() و tracefs_follow_missed_events_clear() القيمة 0 عند النجاح و -1 عند الخطأ، أو إذا لم تجد أي متابعين مطابقين يجب إزالتهم.

#include <unistd.h>
#include <tracefs.h>
#include <stdbool.h>
#include <signal.h>
struct my_struct {
        bool            stopped;
};
#define MAX_COUNT 500000
static int counter;
static int callback(struct tep_event *event, struct tep_record *record,
                    int cpu, void *data)
{
        struct my_struct *my_data = data;
        static struct trace_seq seq;
        if (counter++ > MAX_COUNT) {
                my_data->stopped = true;
                return 1;
        }
        if (!seq.buffer)
                trace_seq_init(&seq);
        tep_print_event(event->tep, &seq, record, "%16s-%-5d [%03d] %6.1000d %s: %s\n",
                        TEP_PRINT_COMM, TEP_PRINT_PID, TEP_PRINT_CPU,
                        TEP_PRINT_TIME, TEP_PRINT_NAME, TEP_PRINT_INFO);
        trace_seq_terminate(&seq);
        trace_seq_do_printf(&seq);
        trace_seq_reset(&seq);
        return 0;
}
static int sched_callback(struct tep_event *event, struct tep_record *record,
                          int cpu, void *data)
{
        static struct tep_format_field *prev_pid;
        static struct tep_format_field *next_pid;
        unsigned long long pid;
        int this_pid = *(int *)data;
        if (!prev_pid) {
                prev_pid = tep_find_field(event, "prev_pid");
                next_pid = tep_find_field(event, "next_pid");
                if (!prev_pid || !next_pid) {
                        fprintf(stderr, "No pid fields??\n");
                        return -1;
                }
        }
        tep_read_number_field(prev_pid, record->data, &pid);
        if (pid == this_pid)
                printf("WE ARE LEAVING!\n");
        tep_read_number_field(next_pid, record->data, &pid);
        if (pid == this_pid)
                printf("WE ARE ARRIVING!\n");
        return 0;
}
static int missed_callback(struct tep_event *event, struct tep_record *record,
                           int cpu, void *data)
{
        printf("OOPS! cpu %d dropped ", cpu);
        if (record->missed_events > 0)
                printf("%lld ", record->missed_events);
        printf("events\n");
        return 0;
}

        tracefs_iterate_stop(instance);
        my_data.stopped = true;
}

        tracefs_event_enable(instance, NULL, NULL);
        sleep(1);

الاسم

tracefs_follow_event_clear، tracefs_follow_missed_events_clear، tracefs_instance_destroy، tracefs_instance_free - مسح الأحداث واتباعها وتدمير مثيل tracefs وتحريره

ملخص

#include <tracefs.h>

void tracefs_follow_event_clear(struct tracefs_instance *instance,
                                const char *event_name,
                                const char *subsystem_name);
void tracefs_follow_missed_events_clear(struct tracefs_instance *instance);
void tracefs_instance_destroy(struct tracefs_instance *instance);
void tracefs_instance_free(struct tracefs_instance *instance);

الوصف

تُستخدم هذه الوظائف لإدارة وتنظيف مثيل tracefs.

  • tracefs_follow_event_clear(): تقوم هذه الوظيفة بمسح جميع الأحداث التي يتم تتبعها حاليًا في المثيل المحدد. إذا تم توفير event_name و subsystem_name، فسيتم مسح الحدث المحدد فقط. إذا تم تمرير NULL لكلا الوسيطتين، فسيتم مسح جميع الأحداث التي يتم تتبعها.
  • tracefs_follow_missed_events_clear(): تقوم هذه الوظيفة بمسح العدادات والبيانات المتعلقة بالأحداث الفائتة للمثيل المحدد.
  • tracefs_instance_destroy(): تقوم هذه الوظيفة بتدمير المثيل وتحرير أي موارد متعلقة به، باستثناء الذاكرة التي تم تخصيصها للمثيل نفسه.
  • tracefs_instance_free(): تقوم هذه الوظيفة بتحرير الذاكرة التي تم تخصيصها للمثيل.

قيمة الإرجاع

لا تُرجع أي من هذه الوظائف أي قيم.

مثال

// ... (الكود السابق) ...
        /*
         * لا يلزم استدعاء tracefs_follow_event_clear() و
         * tracefs_follow_missed_events_clear() هنا لأن
         * tracefs_instance_free() ستنظفها.
         */
        tracefs_follow_event_clear(instance, NULL, NULL);
        tracefs_follow_missed_events_clear(instance);
        tracefs_instance_destroy(instance);
        tracefs_instance_free(instance);
        if (my_data.stopped) {
                if (counter > MAX_COUNT)
                        printf("انتهى الحد الأقصى للعد");
                else
                        printf("انتهى عبر الإشارة");
        }
        return 0;
}
