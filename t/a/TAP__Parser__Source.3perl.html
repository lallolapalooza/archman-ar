TAP::Parser::Source(3perl) دليل مبرمجي بيرل TAP::Parser::Source(3perl)

الاسم

TAP::Parser::Source - مصدر TAP وبيانات وصفية عنه

الإصدار

الإصدار 3.50

ملخص

use TAP::Parser::Source;
my $source = TAP::Parser::Source->new;
$source->raw( \'مرجع لمصدر TAP الخام' )
       ->config( \%config )
       ->merge( $boolean )
       ->switches( \@switches )
       ->test_args( \@args )
       ->assemble_meta;
do { ... } if $source->meta->{is_file};
# انظر assemble_meta للحصول على قائمة كاملة بالبيانات المتاحة

الوصف

مصدر TAP هو شيء ينتج تيارًا من TAP ليستهلكه المحلل، مثل ملف تنفيذي، أو ملف نصي، أو أرشيف، أو مقبض إدخال/إخراج، أو قاعدة بيانات، إلخ. تقوم "TAP::Parser::Source" بتغليف هذه المصادر الخام، وتوفر بعض البيانات الوصفية المفيدة حولها. تستخدمها TAP::Parser::SourceHandlers، والتي تقوم بكل ما هو مطلوب لإنتاج والتقاط تيار TAP من المصدر الخام، وتعبئته في TAP::Parser::Iterator ليستهلكه المحلل.

ما لم تكن تكتب معالج مصدر TAP جديدًا، أو إضافة، أو فئة فرعية من TAP::Parser، فمن المحتمل أنك لن تحتاج إلى استخدام هذه الوحدة مباشرة.

الطرق

طرق الفئة

"new"

my $source = TAP::Parser::Source->new;

يعيد كائن "TAP::Parser::Source" جديدًا.

طرق النسخة

"raw"

my $raw = $source->raw;
$source->raw( $some_value );

مُحدِّد/مُستقبِل متسلسل للمصدر TAP الخام. هذا مرجع، لأنه قد يحتوي على كميات كبيرة من البيانات (على سبيل المثال: TAP خام).

"meta"

my $meta = $source->meta;
$source->meta({ %some_value });

مُحدِّد/مُستقبِل متسلسل للبيانات الوصفية حول المصدر. هذا افتراضي إلى hashref فارغ. انظر "assemble_meta" لمزيد من المعلومات.

"has_meta"

صحيح إذا كان المصدر يحتوي على بيانات وصفية.

"config"

my $config = $source->config;
$source->config({ %some_value });

مُحدِّد/مُستقبِل متسلسل لتكوين المصدر، إن وجد أي شيء مقدم من المستخدم. كيفية استخدامه متروك لك. هذا افتراضي إلى hashref فارغ. انظر "config_for" لمزيد من المعلومات.

"merge"

my $merge = $source->merge;
$source->config( $bool );

مُحدِّد/مُستقبِل متسلسل للعلامة التي تحدد ما إذا كان يجب دمج STDOUT و STDERR (عند الاقتضاء). الافتراضي هو undef.

"switches"

my $switches = $source->switches;
$source->config([ @switches ]);

مُحدِّد/مُستقبِل متسلسل لقائمة مفاتيح سطر الأوامر التي يجب تمريرها إلى المصدر (عند الاقتضاء). الافتراضي هو undef.

"test_args"

my $test_args = $source->test_args;
$source->config([ @test_args ]);

مُحدِّد/مُستقبِل متسلسل لقائمة وسائط سطر الأوامر التي يجب تمريرها إلى المصدر (عند الاقتضاء). الافتراضي هو undef.

"assemble_meta"

my $meta = $source->assemble_meta;

تجمع البيانات الوصفية حول المصدر "الخام"، وتخزنها في "meta" وتعيدها كـ hashref. يتم ذلك حتى لا تضطر معالجات المصادر TAP::Parser::SourceHandlers إلى تكرار الشيكات الشائعة. حاليًا يشمل هذا:

  is_scalar => $bool,
  is_hash   => $bool,
  is_array  => $bool,
  # للقيم العددية:
  length => $n
  has_newlines => $bool
  # يتم القيام بذلك فقط إذا كانت القيمة العددية تبدو كاسم ملف
  is_file => $bool,
  is_dir  => $bool,
  is_symlink => $bool,
  file => {
      # يتم القيام بذلك فقط إذا كانت القيمة العددية تبدو كاسم ملف
      basename => $string, # بما في ذلك الامتداد
      dir      => $string,
      ext      => $string,
      lc_ext   => $string,
      # فحوصات النظام
      exists  => $bool,
      stat    => [ ... ], # perldoc -f stat
      empty   => $bool,
      size    => $n,
      text    => $bool,
      binary  => $bool,
      read    => $bool,
      write   => $bool,
      execute => $bool,
      setuid  => $bool,
      setgid  => $bool,
      sticky  => $bool,
      is_file => $bool,
      is_dir  => $bool,
      is_symlink => $bool,
      # يتم القيام بذلك فقط إذا كان الملف عبارة عن رابط رمزي
      lstat      => [ ... ], # perldoc -f lstat
      # يتم القيام بذلك فقط إذا كان الملف عبارة عن ملف نصي قابل للقراءة
      shebang => $first_line,
  }
# للمصفوفات:
size => $n,

"shebang"

احصل على سطر shebang لملف نصي برمجي.

my $shebang = TAP::Parser::Source->shebang( $some_script );

يمكن استدعاؤها كطريقة فئة

"config_for"

my $config = $source->config_for( $class );

يعيد "config" لـ $class المعطى. قد تكون أسماء الفئات مؤهلة بالكامل أو مختصرة، على سبيل المثال:

# هذه معادلة
$source->config_for( 'Perl' );
$source->config_for( 'TAP::Parser::SourceHandler::Perl' );

إذا تم إعطاء $class مؤهلة بالكامل، فسيتم فحص نسختها المختصرة أولاً.
