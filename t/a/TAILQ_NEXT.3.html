TAILQ(3) دليل الدوال القياسية TAILQ(3)

الاسم

TAILQ_CONCAT, TAILQ_EMPTY, TAILQ_ENTRY, TAILQ_FIRST, TAILQ_FOREACH, TAILQ_FOREACH_REVERSE, TAILQ_HEAD, TAILQ_HEAD_INITIALIZER, TAILQ_INIT, TAILQ_INSERT_AFTER, TAILQ_INSERT_BEFORE, TAILQ_INSERT_HEAD, TAILQ_INSERT_TAIL, TAILQ_LAST, TAILQ_NEXT, TAILQ_REMOVE - تنفيذ ذيل قائمة مزدوجة

المكتبة

مكتبة C القياسية (libc, -lc)

الملخص

#include <sys/queue.h>
TAILQ_ENTRY(TYPE);
TAILQ_HEAD(HEADNAME, TYPE);
TAILQ_HEAD TAILQ_HEAD_INITIALIZER(TAILQ_HEAD head);
void TAILQ_INIT(TAILQ_HEAD *head);
int TAILQ_EMPTY(TAILQ_HEAD *head);
void TAILQ_INSERT_HEAD(TAILQ_HEAD *head,
                         struct TYPE *elm, TAILQ_ENTRY NAME);
void TAILQ_INSERT_TAIL(TAILQ_HEAD *head,
                         struct TYPE *elm, TAILQ_ENTRY NAME);
void TAILQ_INSERT_BEFORE(struct TYPE *listelm,
                         struct TYPE *elm, TAILQ_ENTRY NAME);
void TAILQ_INSERT_AFTER(TAILQ_HEAD *head, struct TYPE *listelm,
                         struct TYPE *elm, TAILQ_ENTRY NAME);
struct TYPE *TAILQ_FIRST(TAILQ_HEAD *head);
struct TYPE *TAILQ_LAST(TAILQ_HEAD *head, HEADNAME);
struct TYPE *TAILQ_PREV(struct TYPE *elm, HEADNAME, TAILQ_ENTRY NAME);
struct TYPE *TAILQ_NEXT(struct TYPE *elm, TAILQ_ENTRY NAME);
TAILQ_FOREACH(struct TYPE *var, TAILQ_HEAD *head,
                         TAILQ_ENTRY NAME);
TAILQ_FOREACH_REVERSE(struct TYPE *var, TAILQ_HEAD *head, HEADNAME,
                         TAILQ_ENTRY NAME);
void TAILQ_REMOVE(TAILQ_HEAD *head, struct TYPE *elm,
                         TAILQ_ENTRY NAME);
void TAILQ_CONCAT(TAILQ_HEAD *head1, TAILQ_HEAD *head2,
                         TAILQ_ENTRY NAME);

الوصف

تحدد هذه الماكروهات وتعمل على قائمة ذيل مزدوجة.

في تعريفات الماكروهات، TYPE هو اسم بنية معرفة من قبل المستخدم، والتي يجب أن تحتوي على حقل من نوع TAILQ_ENTRY، يسمى NAME. الوسيط HEADNAME هو اسم بنية معرفة من قبل المستخدم يجب تعريفها باستخدام الماكرو TAILQ_HEAD().

الإنشاء

يتم ترأس قائمة الذيل المزدوجة بواسطة بنية معرفة باستخدام الماكرو TAILQ_HEAD(). تحتوي هذه البنية على زوج من المؤشرات، واحد إلى العنصر الأول في القائمة والآخر إلى العنصر الأخير في القائمة. يتم ربط العناصر بطريقة يمكن إزالة عنصر عشوائي دون اجتياز القائمة. يمكن إضافة عناصر جديدة إلى القائمة بعد عنصر موجود، قبل عنصر آخر، في رأس القائمة أو في نهاية القائمة. يتم تعريف بنية TAILQ_HEAD على النحو التالي:


TAILQ_HEAD(HEADNAME, TYPE) head;

حيث struct HEADNAME هي البنية التي سيتم تعريفها، و struct TYPE هو نوع العناصر المرتبطة بالقائمة. يمكن بعد ذلك تعريف مؤشر إلى رأس القائمة على النحو التالي:


struct HEADNAME *headp;

(الأسماء head و headp قابلة للاختيار من قبل المستخدم.)

TAILQ_ENTRY() تعرف بنية تربط عناصر القائمة.

TAILQ_HEAD_INITIALIZER() تقيم إلى مُهيئ للقائمة head.

TAILQ_INIT() تهيئ القائمة المشار إليها بواسطة

TAILQ_EMPTY() تعيد TRUE إذا لم تكن هناك عناصر في القائمة. head.

الإدراج

TAILQ_INSERT_HEAD() يدرج العنصر الجديد elm في رأس القائمة.

TAILQ_INSERT_TAIL() يدرج العنصر الجديد elm في نهاية القائمة.

TAILQ_INSERT_BEFORE() يدرج العنصر الجديد elm قبل العنصر listelm.

TAILQ_INSERT_AFTER() يدرج العنصر الجديد elm بعد العنصر listelm.

التجول

TAILQ_FIRST() يعيد العنصر الأول في القائمة، أو NULL إذا كانت القائمة فارغة.

TAILQ_LAST() يعيد العنصر الأخير في القائمة. إذا كانت القائمة فارغة، فإن القيمة المعادة هي NULL.

TAILQ_PREV() يعيد العنصر السابق في القائمة، أو NULL إذا كان هذا هو الأول.

TAILQ_NEXT() يعيد العنصر التالي في القائمة، أو NULL إذا كان هذا هو الأخير.

TAILQ_FOREACH() يجتاز القائمة المشار إليها بواسطة head في الاتجاه الأمامي، مع تعيين كل عنصر إلى var. يتم تعيين var إلى NULL إذا انتهى التكرار بشكل طبيعي، أو إذا لم تكن هناك عناصر.

TAILQ_FOREACH_REVERSE() يجتاز القائمة المشار إليها بواسطة head في الاتجاه العكسي، مع تعيين كل عنصر إلى var.

الإزالة

TAILQ_REMOVE() يزيل العنصر elm من القائمة.

وظائف أخرى

TAILQ_CONCAT() يربط القائمة التي ترأسها head2 بنهاية القائمة التي ترأسها head1، مزيلاً جميع العناصر من الأولى.

قيمة الإرجاع

TAILQ_EMPTY() تعيد قيمة غير صفرية إذا كانت القائمة فارغة، وصفر إذا كانت تحتوي على عنصر واحد على الأقل.

TAILQ_FIRST() و TAILQ_LAST() و TAILQ_PREV() و TAILQ_NEXT() تعيد مؤشراً إلى بنية TYPE الأولى أو الأخيرة أو التالية أو السابقة، على التوالي.

TAILQ_HEAD_INITIALIZER() تعيد مُهيئ يمكن تعيينه إلى القائمة head.

المعايير

BSD.

التاريخ

4.4BSD.

ملاحظات

TAILQ_FOREACH() و TAILQ_FOREACH_REVERSE() لا تسمحان بإزالة أو تحرير var داخل الحلقة، حيث قد يؤثر ذلك على التجول. TAILQ_FOREACH_SAFE() و TAILQ_FOREACH_REVERSE_SAFE()، المتوفرتان في BSDs ولكن ليس في glibc، تصلحان هذا القيد عن طريق السماح بإزالة وتحرير var بأمان داخل الحلقة دون التأثير على التجول.

مثال

#include <stddef.h>
#include <stdio.h>
#include <stdlib.h>
#include <sys/queue.h>
struct entry {
    int data;
    TAILQ_ENTRY(entry) entries;             /* ذيل قائمة مزدوجة */
};
TAILQ_HEAD(tailhead, entry);
int
main(void)
{
    struct entry *n1, *n2, *n3, *np;
    struct tailhead head;                   /* رأس القائمة */
    int i;
    TAILQ_INIT(&head);                      /* تهيئة القائمة */
    n1 = malloc(sizeof(struct entry));      /* الإدراج في الرأس */
    TAILQ_INSERT_HEAD(&head, n1, entries);
    n1 = malloc(sizeof(struct entry));      /* الإدراج في النهاية */
    TAILQ_INSERT_TAIL(&head, n1, entries);
    n2 = malloc(sizeof(struct entry));      /* الإدراج بعد */
    TAILQ_INSERT_AFTER(&head, n1, n2, entries);
    n3 = malloc(sizeof(struct entry));      /* الإدراج قبل */
    TAILQ_INSERT_BEFORE(n2, n3, entries);
    TAILQ_REMOVE(&head, n2, entries);       /* الإزالة */
    free(n2);
                                            /* التجول الأمامي */
    i = 0;
    TAILQ_FOREACH(np, &head, entries)
        np->data = i++;
                                            /* التجول العكسي */
    TAILQ_FOREACH_REVERSE(np, &head, tailhead, entries)
        printf("%i\n", np->data);
                                            /* إزالة القائمة */
    n1
