الوصف

تعرف هذه الماكرو وتعمل على قائمة انتظار مزدوجة الذيل.

في تعريفات الماكرو، TYPE هو اسم هيكل معرف من قبل المستخدم، والذي يجب أن يحتوي على حقل من نوع TAILQ_ENTRY، يسمى NAME. الوسيط HEADNAME هو اسم هيكل معرف من قبل المستخدم والذي يجب تعريفه باستخدام الماكرو TAILQ_HEAD().

إنشاء

تُدار قائمة الانتظار بواسطة هيكل معرف باستخدام الماكرو TAILQ_HEAD(). يحتوي هذا الهيكل على زوج من المؤشرات، أحدهما يشير إلى العنصر الأول في القائمة والآخر إلى العنصر الأخير في القائمة. العناصر مرتبطة بروابط مزدوجة، مما يسمح بإزالة أي عنصر دون الحاجة إلى اجتياز القائمة. يمكن إضافة عناصر جديدة إلى القائمة بعد عنصر موجود، أو قبل عنصر موجود، في مقدمة القائمة أو في نهايتها. يتم تعريف هيكل TAILQ_HEAD على النحو التالي:


TAILQ_HEAD(HEADNAME, TYPE) head;

حيث struct HEADNAME هو الهيكل المراد تعريفه، و struct TYPE هو نوع العناصر المراد ربطها في القائمة. يمكن بعد ذلك تعريف مؤشر إلى رأس القائمة على النحو التالي:


struct HEADNAME *headp;

(الاسمين head و headp قابلان للاختيار من قبل المستخدم.)

TAILQ_ENTRY() يعرّف هيكلاً يربط عناصر القائمة.

TAILQ_HEAD_INITIALIZER() يقيم إلى مُهيئ للقائمة head.

TAILQ_INIT() يهيئ القائمة المشار إليها بواسطة

TAILQ_EMPTY() تعيد TRUE إذا لم تكن هناك عناصر في القائمة. head.

إدراج

TAILQ_INSERT_HEAD() تُدرج العنصر الجديد elm في مقدمة القائمة.

TAILQ_INSERT_TAIL() تُدرج العنصر الجديد elm في نهاية القائمة.

TAILQ_INSERT_BEFORE() تُدرج العنصر الجديد elm قبل العنصر listelm.

TAILQ_INSERT_AFTER() تُدرج العنصر الجديد elm بعد العنصر listelm.

اجتياز

TAILQ_FIRST() تعيد العنصر الأول في القائمة، أو NULL إذا كانت القائمة فارغة.

TAILQ_LAST() تعيد العنصر الأخير في القائمة. إذا كانت القائمة فارغة، فإن القيمة المعادة هي NULL.

TAILQ_PREV() تعيد العنصر السابق في القائمة، أو NULL إذا كان هذا هو الأول.

TAILQ_NEXT() تعيد العنصر التالي في القائمة، أو NULL إذا كان هذا هو الأخير.

TAILQ_FOREACH() تجتاز القائمة المشار إليها بواسطة head في الاتجاه الأمامي، وتعين كل عنصر إلى var. يتم تعيين var إلى NULL إذا انتهت الحلقة بشكل طبيعي، أو إذا لم تكن هناك عناصر.

TAILQ_FOREACH_REVERSE() تجتاز القائمة المشار إليها بواسطة head في الاتجاه العكسي، وتعين كل عنصر إلى var.

إزالة

TAILQ_REMOVE() تزيل العنصر elm من القائمة.

ميزات أخرى

TAILQ_CONCAT() ستلحق القائمة head2 بنهاية القائمة head1، مع إزالة جميع الإدخالات من السابقة.

قيمة العائد

TAILQ_EMPTY() تعيد قيمة غير صفرية إذا كانت القائمة فارغة، وصفر إذا كان هناك عنصر واحد على الأقل.

TAILQ_FIRST() و TAILQ_LAST() و TAILQ_PREV() و TAILQ_NEXT() تعيد مؤشراً إلى هيكل TYPE الأول أو الأخير أو السابق أو التالي، على التوالي.

TAILQ_HEAD_INITIALIZER() تعيد مُهيئاً يمكن تعيينه إلى القائمة head.

المعايير

BSD.

التاريخ

4.4BSD.

تنبيهات

TAILQ_FOREACH() و TAILQ_FOREACH_REVERSE() لا تسمحان بإزالة أو تحرير var داخل الحلقة، حيث قد يتداخل ذلك مع الاجتياز. TAILQ_FOREACH_SAFE() و TAILQ_FOREACH_REVERSE_SAFE()، المتوفرتان في BSDs ولكن ليس في glibc، تعالج هذا القيد عن طريق السماح بإزالة وتحرير var داخل الحلقة دون تعطيل الاجتياز.

أمثلة

#include <stddef.h>
#include <stdio.h>
#include <stdlib.h>
#include <sys/queue.h>
struct entry {
    int data;
    TAILQ_ENTRY(entry) entries;             /* قائمة ذيل مزدوجة */
};
TAILQ_HEAD(tailhead, entry);
int
main(void)
{
    struct entry *n1, *n2, *n3, *np;
    struct tailhead head;                   /* رأس القائمة */
    int i;
    TAILQ_INIT(&head);                      /* تهيئة القائمة */
    n1 = malloc(sizeof(struct entry));      /* إدراج في المقدمة */
    TAILQ_INSERT_HEAD(&head, n1, entries);
    n1 = malloc(sizeof(struct entry));      /* إدراج في الذيل */
    TAILQ_INSERT_TAIL(&head, n1, entries);
    n2 = malloc(sizeof(struct entry));      /* إدراج بعد */
    TAILQ_INSERT_AFTER(&head, n1, n2, entries);
    n3 = malloc(sizeof(struct entry));      /* إدراج قبل */
    TAILQ_INSERT_BEFORE(n2, n3, entries);
    TAILQ_REMOVE(&head, n2, entries);       /* إزالة */
    free(n2);
                                            /* اجتياز أمامي */
    i = 0;
    TAILQ_FOREACH(np, &head, entries)
        np->data = i++;
                                            /* اجتياز عكسي */
    TAILQ_FOREACH_REVERSE(np, &head, tailhead, entries)
        printf("%i\n", np->data);
                                            /* إزالة القائمة */
    n1

هذا الموقع غير مرتبط بـ Arch Linux ولا تابع له بأي شكل من الأشكال. المحتوى هو ترجمة فقط لصفحات دليل Arch المنشورة على man.archlinux.org.


TAILQ_FIRST(&head);
while (n1 != NULL) {
    n2 = TAILQ_NEXT(n1, entries);
    free(n1);
    n1 = n2;
}
TAILQ_INIT(&head);
exit(EXIT_SUCCESS);
