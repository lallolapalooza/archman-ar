TAILQ_CONCAT, TAILQ_EMPTY, TAILQ_ENTRY, TAILQ_FIRST, TAILQ_FOREACH, TAILQ_FOREACH_REVERSE, TAILQ_HEAD, TAILQ_HEAD_INITIALIZER, TAILQ_INIT, TAILQ_INSERT_AFTER, TAILQ_INSERT_BEFORE, TAILQ_INSERT_HEAD, TAILQ_INSERT_TAIL, TAILQ_LAST, TAILQ_NEXT, TAILQ_REMOVE - تنفيذ قائمة مزدوجة الذيل

المكتبة

المكتبة القياسية (libc, -lc)

الملخص

#include <sys/queue.h>
TAILQ_ENTRY(TYPE);
TAILQ_HEAD(HEADNAME, TYPE);
TAILQ_HEAD TAILQ_HEAD_INITIALIZER(TAILQ_HEAD head);
void TAILQ_INIT(TAILQ_HEAD *head);
int TAILQ_EMPTY(TAILQ_HEAD *head);
void TAILQ_INSERT_HEAD(TAILQ_HEAD *head,
                         struct TYPE *elm, TAILQ_ENTRY NAME);
void TAILQ_INSERT_TAIL(TAILQ_HEAD *head,
                         struct TYPE *elm, TAILQ_ENTRY NAME);
void TAILQ_INSERT_BEFORE(struct TYPE *listelm,
                         struct TYPE *elm, TAILQ_ENTRY NAME);
void TAILQ_INSERT_AFTER(TAILQ_HEAD *head, struct TYPE *listelm,
                         struct TYPE *elm, TAILQ_ENTRY NAME);
struct TYPE *TAILQ_FIRST(TAILQ_HEAD *head);
struct TYPE *TAILQ_LAST(TAILQ_HEAD *head, HEADNAME);
struct TYPE *TAILQ_PREV(struct TYPE *elm, HEADNAME, TAILQ_ENTRY NAME);
struct TYPE *TAILQ_NEXT(struct TYPE *elm, TAILQ_ENTRY NAME);
TAILQ_FOREACH(struct TYPE *var, TAILQ_HEAD *head,
                         TAILQ_ENTRY NAME);
TAILQ_FOREACH_REVERSE(struct TYPE *var, TAILQ_HEAD *head, HEADNAME,
                         TAILQ_ENTRY NAME);
void TAILQ_REMOVE(TAILQ_HEAD *head, struct TYPE *elm,
                         TAILQ_ENTRY NAME);
void TAILQ_CONCAT(TAILQ_HEAD *head1, TAILQ_HEAD *head2,
                         TAILQ_ENTRY NAME);

الوصف

تعرف هذه الماكروات وتعمل على قائمة مزدوجة الذيل.

في تعريفات الماكرو، TYPE هو اسم هيكل معرف من قبل المستخدم، والذي يجب أن يحتوي على حقل من نوع TAILQ_ENTRY، يسمى NAME. الوسيط HEADNAME هو اسم هيكل معرف من قبل المستخدم، والذي يجب تعريفه باستخدام الماكرو TAILQ_HEAD().

إنشاء

تُقاد القائمة المزدوجة بهيكل معرف باستخدام الماكرو TAILQ_HEAD(). يحتوي هذا الهيكل على زوج من المؤشرات، أحدهما إلى العنصر الأول في القائمة والآخر إلى العنصر الأخير في القائمة. ترتبط العناصر بشكل مزدوج بحيث يمكن إزالة عنصر عشوائي دون اجتياز القائمة. يمكن إضافة عناصر جديدة إلى القائمة بعد عنصر موجود، قبل عنصر موجود، في مقدمة القائمة أو في نهايتها. يتم تعريف هيكل TAILQ_HEAD على النحو التالي:


TAILQ_HEAD(HEADNAME, TYPE) head;

حيث struct HEADNAME هو الهيكل المراد تعريفه و struct TYPE هو نوع العناصر التي سيتم ربطها في القائمة. يمكن بعد ذلك تعريف عنوان مؤشر إلى رأس القائمة على أنه:


struct HEADNAME *headp;

(أسماء head و headp قابلة للاختيار من قبل المستخدم.)

TAILQ_ENTRY() تعرف هيكلًا يربط عناصر القائمة.

TAILQ_HEAD_INITIALIZER() يقيم إلى مُهيئ للقائمة head.

TAILQ_INIT() يهيئ القائمة المشار إليها بواسطة

TAILQ_EMPTY() تُرجع صحيحًا إذا لم تكن هناك عناصر في القائمة. head.

إدخال

TAILQ_INSERT_HEAD() يُدخل العنصر الجديد elm في مقدمة القائمة.

TAILQ_INSERT_TAIL() يُدخل العنصر الجديد elm في نهاية القائمة.

TAILQ_INSERT_BEFORE() يُدخل العنصر الجديد elm قبل العنصر listelm.

TAILQ_INSERT_AFTER() يُدخل العنصر الجديد elm بعد العنصر listelm.

اجتياز

TAILQ_FIRST() تُرجع العنصر الأول في القائمة، أو NULL إذا كانت القائمة فارغة.

TAILQ_LAST() تُرجع العنصر الأخير في القائمة. إذا كانت القائمة فارغة، فإن القيمة المرجعة هي NULL.

TAILQ_PREV() تُرجع العنصر السابق في القائمة، أو NULL إذا كان هذا هو الأول.

TAILQ_NEXT() تُرجع العنصر التالي في القائمة، أو NULL إذا كان هذا هو الأخير.

TAILQ_FOREACH() يجتاز القائمة المشار إليها بواسطة head في اتجاه forward، ويُسند كل عنصر إلى var. يتم تعيين var إلى NULL إذا انتهت الحلقة بشكل طبيعي، أو إذا لم تكن هناك عناصر.

TAILQ_FOREACH_REVERSE() يجتاز القائمة المشار إليها بواسطة head في اتجاه reverse، ويُسند كل عنصر إلى var.

حذف

TAILQ_REMOVE() يحذف العنصر elm من القائمة.

ميزات أخرى

TAILQ_CONCAT() ستربط القائمة التي يرأسها head2 بنهاية القائمة التي يرأسها head1، وحذف جميع العناصر من القائمة السابقة.

قيمة الإرجاع

TAILQ_EMPTY() تُرجع قيمة غير صفرية إذا كانت القائمة فارغة، وصفرًا إذا كان هناك عنصر واحد على الأقل.

TAILQ_FIRST(), TAILQ_LAST(), TAILQ_PREV(), و TAILQ_NEXT() تُرجع مؤشرًا إلى هيكل TYPE الأول، الأخير، السابق، أو التالي، على التوالي.

TAILQ_HEAD_INITIALIZER() تُرجع مُهيئًا يمكن تعيينه إلى القائمة head.

المعايير

BSD.

التاريخ

4.4BSD.

ملاحظات

TAILQ_FOREACH() و TAILQ_FOREACH_REVERSE() لا تسمحان بحذف أو تحرير var داخل الحلقة، لأن هذا قد يتداخل مع الاجتياز. TAILQ_FOREACH_SAFE() و TAILQ_FOREACH_REVERSE_SAFE() (الموجودتان في BSDs ولكن ليس في glibc) تصلحان هذا القيد بالسماح بحذف var من القائمة وتحريره داخل الحلقة دون التدخل في الاجتياز.

أمثلة

#include <stddef.h>
#include <stdio.h>
#include <stdlib.h>
#include <sys/queue.h>
struct entry {
    int data;
    TAILQ_ENTRY(entry) entries;             /* قائمة مزدوجة الذيل */
};
TAILQ_HEAD(tailhead, entry);
int
main(void)
{
    struct entry *n1, *n2, *n3, *np;
    struct tailhead head;                   /* رأس القائمة */
    int i;
    TAILQ_INIT(&head);                      /* تهيئة القائمة */
    n1 = malloc(sizeof(struct entry));      /* إدخال في المقدمة */
    TAILQ_INSERT_HEAD(&head, n1, entries);
    n1 = malloc(sizeof(struct entry));      /* إدخال في النهاية */
    TAILQ_INSERT_TAIL(&head, n1, entries);
    n2 = malloc(sizeof(struct entry));      /* إدخال بعد */
    TAILQ_INSERT_AFTER(&head, n1, n2, entries);
    n3 = malloc(sizeof(struct entry));      /* إدخال قبل */
    TAILQ_INSERT_BEFORE(n2, n3, entries);
    TAILQ_REMOVE(&head, n2, entries);       /* حذف */
    free(n2);
                                            /* اجتياز forward */
    i = 0;
    TAILQ_FOREACH(np, &head, entries)
        np->data = i++;
                                            /* اجتياز reverse */
    TAILQ_FOREACH_REVERSE(np, &head, tailhead, entries)
        printf("%i\n", np->data);
                                            /* إفراغ القائمة */
    n1
TAILQ_INIT(3) - Linux manual page

NAME

TAILQ_INIT

initialize a tail queue

LIBRARY

libbsd

SYNOPSIS

#include <sys/queue.h>


void
TAILQ_INIT(tailq_head_t *head);

DESCRIPTION

The TAILQ_INIT() macro initializes the tail queue pointed to by head. An initialized tail queue is an empty queue. This macro should be used to initialize all tail queues before they are used.

RETURN VALUE

None.

STANDARDS

The TAILQ_INIT() macro is part of the Berkeley Software Distribution (BSD) queue macros, and is expected to be available on most systems. It is not part of any ANSI/ISO C standard.

HISTORY

The tail queue macros first appeared in FreeBSD 3.0.

CAVEATS

None.

EXAMPLES

To initialize a tail queue named myqueue:

struct myqueue_head {
    int data;
    struct myqueue_entry {
        struct myqueue_entry *tqe_next;
        struct myqueue_entry *tqe_prev;
    } myqueue_entries;
} myqueue;
...
TAILQ_INIT(&myqueue);

To iterate over a tail queue and free all elements:

struct myqueue_entry *n1, *n2;
TAILQ_FOREACH_SAFE(n1, &myqueue, myqueue_entries, n2) {
    free(n1);
}
TAILQ_INIT(&myqueue);
