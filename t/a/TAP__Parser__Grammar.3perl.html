TAP::Parser::Grammar(3perl) دليل مبرمجي بيرل TAP::Parser::Grammar(3perl)

الاسم

TAP::Parser::Grammar - صيغة لبروتوكول Test Anything.

الإصدار

الإصدار 3.50

الملخص

use TAP::Parser::Grammar;
my $grammar = $self->make_grammar({
  iterator => $tap_parser_iterator,
  parser   => $tap_parser,
  version  => 12,
});
my $result = $grammar->tokenize;

الوصف

"TAP::Parser::Grammar" يقوم بتقسيم الأسطر من TAP::Parser::Iterator وإنشاء فئات فرعية من TAP::Parser::Result لتمثيل الرموز.

لا تحاول استخدام هذه الفئة مباشرة. لن يكون ذلك منطقيًا. إنها موجودة بشكل أساسي لضمان قدرتنا على الحصول على صيغ قابلة للتبديل عندما يتم توسيع TAP في تاريخ مستقبلي (بالإضافة إلى أن هذه الأشياء كانت تسبب فوضى في المحلل).

الأساليب

أساليب الفئة

"new"

my $grammar = TAP::Parser::Grammar->new({
    iterator => $iterator,
    parser   => $parser,
    version  => $version,
});

يعيد كائن صيغة TAP::Parser يقوم بتحليل تدفق TAP من المكرر المحدد. كل من "iterator" و "parser" هي حجج مطلوبة. إذا لم يتم تعيين "version"، فإنها افتراضيًا إلى 12 (انظر "set_version" لمزيد من التفاصيل).

أساليب المثيل

"set_version"

$grammar->set_version(13);

يخبر الصيغة بإصدار بناء جملة TAP الذي سيتم دعمه. أدنى إصدار مدعوم هو 12. على الرغم من أن 'TAP version' ليس بناء جملة صالحًا للإصدار 12، إلا أنه مقبول حتى يمكن تحليل أرقام الإصدارات الأعلى.

"tokenize"

my $token = $grammar->tokenize;

هذه الطريقة ستعيد كائن TAP::Parser::Result يمثل السطر الحالي من TAP.

"token_types"

my @types = $grammar->token_types;

يعيد أنواع الرموز المختلفة التي يمكن لهذه الصيغة تحليلها.

"syntax_for"

my $syntax = $grammar->syntax_for($token_type);

يعيد تعبيرًا عاديًا تم تجميعه مسبقًا سيطابق جزءًا من TAP المقابل لنوع الرمز. على سبيل المثال (وليس أنه يجب عليك الاهتمام حقًا بهذا، "$grammar->syntax_for('comment')" سيعيد "qr/^#(.*)/").

"handler_for"

my $handler = $grammar->handler_for($token_type);

يعيد مرجع رمز يستدعي، عند تمرير سطر TAP المناسب إليه، الرمز الذي تم تحليله المقابل لهذا السطر. نتيجة لذلك، تبدو حلقة تحليل TAP الأساسية مشابهة للآتي:

my @tokens;
my $grammar = TAP::Grammar->new;
LINE: while ( defined( my $line = $parser->_next_chunk_of_tap ) ) {
    for my $type ( $grammar->token_types ) {
        my $syntax  = $grammar->syntax_for($type);
        if ( $line =~ $syntax ) {
            my $handler = $grammar->handler_for($type);
            push @tokens => $grammar->$handler($line);
            next LINE;
        }
    }
    push @tokens => $grammar->_make_unknown_token($line);
}

صيغة TAP

ملاحظة: هذه الصيغة قديمة قليلاً. لا يزال هناك بعض النقاش حولها وسيتم توفير صيغة جديدة عندما تكون الأمور أكثر تحديدًا.

لا يستخدم TAP::Parser صيغة رسمية لأن TAP هو في الأساس بروتوكول قائم على التدفق. في الواقع، من القانوني تمامًا وجود تدفق لا نهائي. لنفس السبب الذي لا نطبق به التعبيرات العادية على التدفقات، لا نستخدم صيغة رسمية هنا. بدلاً من ذلك، نقوم بتحليل TAP في أسطر.

لأغراض التوافق المستقبلي، يتم حاليًا الإشارة إلى أي نتيجة لا تتطابق مع الصيغة التالية باسم TAP::Parser::Result::Unknown. إنه *ليس* خطأ في التحليل.

صيغة رسمية ستبدو مشابهة لما يلي:

(*
    في الوقت الحالي، أقوم بالغش في EBNF بالسماح
    لبعض المصطلحات بتحديدها بواسطة فئات الأحرف POSIX عن طريق
    استخدام بناء الجملة التالي:
      digit ::= [:digit:]
    بقدر ما أعلم، هذا ليس EBNF صالحًا.  قاضيني.  لم
    أعرف كيف أكتب "char" بطريقة أخرى (مشاكل Unicode).
    الاقتراحات مرحب بها.
*)
tap            ::= version? comment | unknown | leading_plan lines
                   |
                   lines trailing_plan {comment}
version        ::= 'TAP version ' positiveInteger {positiveInteger} "\n"
leading_plan   ::= plan skip_directive? "\n"
trailing_plan  ::= plan "\n"
plan           ::= '1..' nonNegativeInteger
lines          ::= line {line}
line           ::= (comment | test | unknown | bailout ) "\n"
test           ::= status positiveInteger? description? directive?
status         ::= 'not '? 'ok '
description    ::= (character - (digit | '#')) {character - '#'}
directive      ::= todo_directive | skip_directive
todo_directive ::= hash_mark 'TODO' ' ' {character}
skip_directive ::= hash_mark 'SKIP' ' ' {character}
comment        ::= hash_mark {character}
hash_mark      ::= '#' {' '}
bailout        ::= 'Bail out!' {character}
unknown        ::= { (character - "\n") }
(* POSIX character classes and other terminals *)
digit              ::= [:digit:]
character          ::= ([:print:] - "\n")
positiveInteger    ::= ( digit - '0' ) {digit}
nonNegativeInteger ::= digit {digit}

الوراثة

يرجى الاطلاع على "SUBCLASSING" في TAP::Parser للحصول على نظرة عامة على الوراثة.

إذا كنت تريد حقًا وراثة صيغة TAP::Parser، فإن أفضل شيء هو قراءة الكود. لا توجد طريقة سهلة لتلخيصه هنا.
