الوصف

تُعرّف هذه وحدات الماكرو وتتعامل مع قائمة انتظار مزدوجة الارتباط.

في تعريفات وحدات الماكرو، TYPE هو اسم نوع بنية معرفة من قبل المستخدم، والتي يجب أن تحتوي على حقل من نوع TAILQ_ENTRY، يُطلق عليه NAME. الوسيط HEADNAME هو اسم بنية معرفة من قبل المستخدم، والتي يجب الإعلان عنها باستخدام وحدة الماكرو TAILQ_HEAD().

الإنشاء

يتم تعريف قائمة الانتظار المزدوجة الارتباط باستخدام بنية معرفة بواسطة وحدة الماكرو TAILQ_HEAD(). تحتوي هذه البنية على زوج من المؤشرات، أحدهما يشير إلى العنصر الأول في القائمة والآخر يشير إلى العنصر الأخير في القائمة. العناصر مرتبطة بشكل مزدوج، مما يسمح بإزالة أي عنصر دون الحاجة إلى اجتياز القائمة. يمكن إضافة عناصر جديدة إلى القائمة بعد عنصر موجود، قبله، في بداية القائمة، أو في نهايتها. يتم الإعلان عن بنية TAILQ_HEAD على النحو التالي:


TAILQ_HEAD(HEADNAME, TYPE) head;

حيث struct HEADNAME هي البنية المراد تعريفها و struct TYPE هو نوع العناصر المرتبطة في القائمة. يمكن لاحقًا الإعلان عن مرجع لرأس القائمة على النحو التالي:


struct HEADNAME *headp;

(أسماء head و headp قابلة للاختيار من قبل المستخدم.)

TAILQ_ENTRY() يُعرّف بنية لربط عناصر القائمة.

TAILQ_HEAD_INITIALIZER() يُقيّم إلى مُهيئ للقائمة head.

TAILQ_INIT() يُهيئ القائمة المشار إليها بواسطة

TAILQ_EMPTY() يُقيّم إلى صحيح إذا لم تكن هناك عناصر في القائمة. head.

الإدراج

TAILQ_INSERT_HEAD() تُدرج العنصر الجديد elm في بداية القائمة.

TAILQ_INSERT_TAIL() تُدرج العنصر الجديد elm في نهاية القائمة.

TAILQ_INSERT_BEFORE() تُدرج العنصر الجديد elm قبل العنصر listelm.

TAILQ_INSERT_AFTER() تُدرج العنصر الجديد elm بعد العنصر listelm.

التجول

TAILQ_FIRST() تُرجع العنصر الأول في القائمة، أو NULL إذا كانت القائمة فارغة.

TAILQ_LAST() تُرجع العنصر الأخير في القائمة. إذا كانت القائمة فارغة، فإن القيمة المرجعة هي NULL.

TAILQ_PREV() تُرجع العنصر السابق في القائمة، أو NULL إذا كان هذا هو الأول.

TAILQ_NEXT() تُرجع العنصر التالي في القائمة، أو NULL إذا كان هذا هو الأخير.

TAILQ_FOREACH() يجتاز القائمة المشار إليها بواسطة head في الاتجاه الأمامي، ويُعيّن كل عنصر إلى var. يتم تعيين var إلى NULL إذا اكتملت الحلقة بشكل طبيعي، أو إذا لم تكن هناك عناصر.

TAILQ_FOREACH_REVERSE() يجتاز القائمة المشار إليها بواسطة head في الاتجاه الخلفي، ويُعيّن كل عنصر إلى var.

الحذف

TAILQ_REMOVE() تزيل العنصر elm من القائمة.

ميزات أخرى

TAILQ_CONCAT() تربط قائمة الرأس head2 بنهاية قائمة الرأس head1، وتزيل جميع العناصر من الأولى.

قيمة الإرجاع

TAILQ_EMPTY() تُرجع قيمة غير صفرية إذا كانت القائمة فارغة، وصفر إذا كانت تحتوي على عنصر واحد على الأقل.

TAILQ_FIRST()، TAILQ_LAST()، TAILQ_PREV() و TAILQ_NEXT() تُرجع مؤشرًا إلى بنية TYPE الأولى، الأخيرة، السابقة، أو التالية، على التوالي.

TAILQ_HEAD_INITIALIZER() تُرجع مُهيئًا يمكن تعيينه لرأس القائمة head.

المعايير

BSD.

السجل

4.4BSD.

تحذيرات

TAILQ_FOREACH() و TAILQ_FOREACH_REVERSE() لا تسمحان بحذف أو تحرير var داخل الحلقة، لأن ذلك قد يتداخل مع عملية التجول. TAILQ_FOREACH_SAFE() و TAILQ_FOREACH_REVERSE_SAFE()، والمتوفرة في BSDs ولكن ليس في glibc، تُصحح هذا القيد بالسماح بحذف وتحرير var بأمان داخل الحلقة دون التداخل مع عملية التجول.

مثال

#include <stddef.h>
#include <stdio.h>
#include <stdlib.h>
#include <sys/queue.h>
struct entry {
    int data;
    TAILQ_ENTRY(entry) entries;             /* Tail queue */
};
TAILQ_HEAD(tailhead, entry);
int
main(void)
{
    struct entry *n1, *n2, *n3, *np;
    struct tailhead head;                   /* Tail queue head */
    int i;
    TAILQ_INIT(&head);                      /* Initialize the queue */
    n1 = malloc(sizeof(struct entry));      /* Insert at the head */
    TAILQ_INSERT_HEAD(&head, n1, entries);
    n1 = malloc(sizeof(struct entry));      /* Insert at the tail */
    TAILQ_INSERT_TAIL(&head, n1, entries);
    n2 = malloc(sizeof(struct entry));      /* Insert after */
    TAILQ_INSERT_AFTER(&head, n1, n2, entries);
    n3 = malloc(sizeof(struct entry));      /* Insert before */
    TAILQ_INSERT_BEFORE(n2, n3, entries);
    TAILQ_REMOVE(&head, n2, entries);       /* Deletion */
    free(n2);
                                            /* Forward traversal */
    i = 0;
    TAILQ_FOREACH(np, &head, entries)
        np->data = i++;
                                            /* Reverse traversal */
    TAILQ_FOREACH_REVERSE(np, &head, tailhead, entries)
        printf("%i\n", np->data);
                                            /* TailQ deletion */
