tar_extract_file(3)

الحزمة: libtar

التراخيص: BSD

tar_extract_file(3) استدعاءات مكتبة C tar_extract_file(3)

الاسم

tar_extract_file، tar_extract_regfile، tar_extract_hardlink، tar_extract_symlink، tar_extract_chardev، tar_extract_blockdev، tar_extract_dir، tar_extract_fifo، tar_skip_regfile، tar_set_file_perms - استخراج الملفات من أرشيف tar

الملخص

#include <libtar.h>

int tar_extract_file(TAR *t, char *realname);

int tar_extract_regfile(TAR *t, char *realname);

int tar_skip_regfile(TAR *t);

int tar_extract_dir(TAR *t, char *realname);

int tar_extract_hardlink(TAR *t, char *realname);

int tar_extract_symlink(TAR *t, char *realname);

int tar_extract_blockdev(TAR *t, char *realname);

int tar_extract_chardev(TAR *t, char *realname);

int tar_extract_fifo(TAR *t, char *realname);

int tar_set_file_perms(TAR *t, char *realname);

الإصدار

توثق صفحة الدليل هذه الإصدار 1.2 من libtar.

الوصف

تعمل الدالة tar_extract_file() كواجهة أمامية للدوال الأخرى tar_extract_*(). تتحقق من رأس tar الحالي المرتبط بمقبض TAR t (والذي يجب تهيئته أولاً عن طريق استدعاء th_read()) لتحديد نوع الملف الذي يشير إليه الرأس. ثم تستدعي الدالة المناسبة tar_extract_*() لاستخراج هذا النوع من الملفات.

تتجاوز الدالة tar_skip_regfile() كتل محتوى الملف وتضع مؤشر الملف في الموقع المتوقع لكتلة رأس tar التالية.

تقوم الدالة tar_set_file_perms() بتعيين سمات الملف المستخرج لمطابقة القيم المشفرة. يشمل ذلك وقت التعديل، الوضع، المالك، والمجموعة للملف. يتم استدعاء هذه الدالة تلقائيًا بواسطة tar_extract_file()، ولكن التطبيقات التي تستدعي الدوال الأخرى tar_extract_*() مباشرة ستحتاج إلى استدعاء tar_set_file_perms() يدويًا إذا كان هذا السلوك مرغوبًا فيه.

قيم الإرجاع

عند الانتهاء بنجاح، ستعيد الدوال الموثقة هنا القيمة 0. عند الفشل، ستعيد القيمة -1 وتعيين errno إلى قيمة مناسبة.

ستعيد الدالة tar_extract_dir() القيمة 1 إذا كان الدليل موجودًا بالفعل.

الأخطاء

ستفشل الدالة tar_extract_file() في الحالات التالية:

EEXIST
إذا تم تعيين علامة O_NOOVERWRITE وكان الملف موجودًا بالفعل.

ستفشل الدوال tar_extract_*() في الحالات التالية:

EINVAL
تعذر إضافة إدخال إلى جدول تجزئة الملف الداخلي.
EINVAL
تمت قراءة أقل من T_BLOCKSIZE بايت من أرشيف tar.
EINVAL
يشير رأس الملف الحالي المرتبط بـ t إلى نوع ملف آخر غير الذي تعرفه الدالة المستدعاة.

قد تفشل أيضًا إذا فشلت أي من الدوال التالية: mkdir()، write()، link()، symlink()، mknod()، mkfifo()، utime()، chown()، lchown()، chmod()، أو lstat().
