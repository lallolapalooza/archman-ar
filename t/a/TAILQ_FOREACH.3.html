TAILQ(3) دليل الدوال القياسية TAILQ(3)

الاسم

TAILQ_CONCAT, TAILQ_EMPTY, TAILQ_ENTRY, TAILQ_FIRST, TAILQ_FOREACH, TAILQ_FOREACH_REVERSE, TAILQ_HEAD, TAILQ_HEAD_INITIALIZER, TAILQ_INIT, TAILQ_INSERT_AFTER, TAILQ_INSERT_BEFORE, TAILQ_INSERT_HEAD, TAILQ_INSERT_TAIL, TAILQ_LAST, TAILQ_NEXT, TAILQ_REMOVE - تنفيذ قائمة انتظار مزدوجة الارتباط

المكتبة

مكتبة الدوال القياسية (libc، -lc)

الملخص

#include <sys/queue.h>
TAILQ_ENTRY(TYPE);
TAILQ_HEAD(HEADNAME, TYPE);

                         struct TYPE *elm, TAILQ_ENTRY NAME);
void TAILQ_INSERT_TAIL(TAILQ_HEAD *head,
                         struct TYPE *elm, TAILQ_ENTRY NAME);
void TAILQ_INSERT_BEFORE(struct TYPE *listelm,
                         struct TYPE *elm, TAILQ_ENTRY NAME);
void TAILQ_INSERT_AFTER(TAILQ_HEAD *head, struct TYPE *listelm,
                         struct TYPE *elm, TAILQ_ENTRY NAME);
struct TYPE *TAILQ_FIRST(TAILQ_HEAD *head);

                         TAILQ_ENTRY NAME);

                         TAILQ_ENTRY NAME);

الوصف

ستعرف وحدات الماكرو هذه وتعمل على قوائم الانتظار المزدوجة الارتباط.

في تعريفات وحدات الماكرو، TYPE هو اسم هيكل معرف من قبل المستخدم يجب أن يحتوي على حقل من نوع TAILQ_ENTRY، يسمى NAME. الوسيط HEADNAME هو اسم هيكل معرف من قبل المستخدم يجب تعريفه باستخدام الماكرو TAILQ_HEAD().

الإنشاء

قائمة الانتظار المزدوجة الارتباط يتم توجيهها بواسطة هيكل معرف بواسطة الماكرو TAILQ_HEAD(). يحتوي هذا الهيكل على زوج من المؤشرات، أحدهما إلى العنصر الأول في القائمة والآخر إلى العنصر الأخير في القائمة. يتم ربط العناصر بشكل مزدوج بحيث يمكن إزالة عنصر عشوائي دون الحاجة إلى المرور عبر القائمة. يمكن إضافة عناصر جديدة إلى القائمة بعد عنصر موجود، أو قبل عنصر موجود، أو في بداية القائمة، أو في نهاية القائمة. يتم تعريف هيكل TAILQ_HEAD على النحو التالي:


TAILQ_HEAD(HEADNAME, TYPE) head;

حيث struct HEADNAME هو الهيكل الذي سيتم تعريفه، و struct TYPE هو نوع العناصر التي سيتم ربطها بالقائمة. يمكن بعد ذلك تعريف مرجع لمؤشر رأس القائمة على النحو التالي:


struct HEADNAME *headp;

(الأسماء head و headp يمكن للمستخدم اختيارها.)

TAILQ_ENTRY() تعرف هيكلاً يربط عناصر القائمة.

TAILQ_HEAD_INITIALIZER() تقيّم إلى مُهيئ للقائمة head.

TAILQ_INIT() تهيئ القائمة المشار إليها بواسطة

TAILQ_EMPTY() تعيد صحيحًا إذا لم تكن هناك عناصر في القائمة. head.

الإدراج

TAILQ_INSERT_HEAD() يُدرج العنصر الجديد elm في بداية القائمة.

TAILQ_INSERT_TAIL() يُدرج العنصر الجديد elm في نهاية القائمة.

TAILQ_INSERT_BEFORE() يُدرج العنصر الجديد elm قبل العنصر listelm.

TAILQ_INSERT_AFTER() يُدرج العنصر الجديد elm بعد العنصر listelm.

التكرار

TAILQ_FIRST() تعيد العنصر الأول في القائمة، أو NULL إذا كانت القائمة فارغة.

TAILQ_LAST() تعيد العنصر الأخير في القائمة. إذا كانت القائمة فارغة، فإن القيمة المعادة هي NULL.

TAILQ_PREV() تعيد العنصر السابق في القائمة، أو NULL إذا كان هذا هو الأول.

TAILQ_NEXT() تعيد العنصر التالي في القائمة، أو NULL إذا كان هذا هو الأخير.

TAILQ_FOREACH() تكرر عبر القائمة المشار إليها بواسطة head في الاتجاه الأمامي، وتعين كل عنصر إلى var بالتناوب. يتم تعيين var إلى NULL إذا انتهت الحلقة بشكل طبيعي، أو إذا لم تكن هناك عناصر.

TAILQ_FOREACH_REVERSE() تكرر عبر القائمة المشار إليها بواسطة head في الاتجاه العكسي، وتعين كل عنصر إلى var.

الحذف

TAILQ_REMOVE() تحذف العنصر elm من القائمة.

ميزات أخرى

TAILQ_CONCAT() تربط القائمة الموجهة بواسطة head2 بنهاية القائمة الموجهة بواسطة head1، وتزيل جميع العناصر من الأولى.

قيمة الإرجاع

TAILQ_EMPTY() تعيد قيمة غير صفرية إذا كانت القائمة فارغة، وصفرًا إذا كان هناك عنصر واحد على الأقل.

TAILQ_FIRST(), TAILQ_LAST(), TAILQ_PREV(), و TAILQ_NEXT() يعيد مؤشرًا إلى هيكل TYPE الأول، الأخير، السابق، أو التالي، على التوالي.

TAILQ_HEAD_INITIALIZER() يعيد مُهيئًا يمكن تعيينه لرأس القائمة head.

المعايير

BSD.

التاريخ

4.4BSD.

تحذيرات

TAILQ_FOREACH() و TAILQ_FOREACH_REVERSE() لا يسمحان بحذف أو تحرير var داخل الحلقة، حيث قد يتداخل ذلك مع التكرار. TAILQ_FOREACH_SAFE() و TAILQ_FOREACH_REVERSE_SAFE()، الموجودة في BSD ولكن ليس في glibc، تصلح هذا القيد بالسماح بحذف var من القائمة وتحريره داخل الحلقة دون التداخل مع التكرار.

أمثلة

#include <stddef.h>
#include <stdio.h>
#include <stdlib.h>
#include <sys/queue.h>
struct entry {
    int data;
    TAILQ_ENTRY(entry) entries;             /* قائمة انتظار مزدوجة الارتباط */
};
TAILQ_HEAD(tailhead, entry);
int
main(void)
{
    struct entry *n1, *n2, *n3, *np;
    struct tailhead head;                   /* رأس القائمة */
    int i;
    TAILQ_INIT(&head);                      /* تهيئة القائمة */
    n1 = malloc(sizeof(struct entry));      /* الإدراج في البداية */
    TAILQ_INSERT_HEAD(&head, n1, entries);
    n1 = malloc(sizeof(struct entry));      /* الإدراج في النهاية */
    TAILQ_INSERT_TAIL(&head, n1, entries);
    n2 = malloc(sizeof(struct entry));      /* الإدراج بعد */
    TAILQ_INSERT_AFTER(&head, n1, n2, entries);
    n3 = malloc(sizeof(struct entry));      /* الإدراج قبل */
    TAILQ_INSERT_BEFORE(n2, n3, entries);
    TAILQ_REMOVE(&head, n2, entries);       /* الحذف */
    free(n2);
                                            /* التكرار للأمام */
    i = 0;
    TAILQ_FOREACH(np, &head, entries)
        np->data = i++;
                                            /* التكرار للخلف */
    TAILQ_FOREACH_REVERSE(np, &head, tailhead, entries)
        printf("%i\n", np->data);
                                            /* حذف القائمة */
