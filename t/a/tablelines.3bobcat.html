tablelines.3bobcat

الحزمة: libbobcat

التراخيص: GPL

FBB::TableLines(3bobcat) فئة دعم الجدول FBB::TableLines(3bobcat)

الاسم

FBB::TableLines - تخصص من TableSupport لكتابة خطوط فصل أفقية

ملخص

#include <bobcat/tableLines>
خيار الربط: -lbobcat

الوصف

هذه الفئة هي تخصص من FBB::TableSupport ويمكن استخدامها لكتابة فواصل خطوط أفقية (وعمودية) في الجداول التي تم إنشاؤها بواسطة كائنات Table أو TableBuf.

مساحة الاسم

FBB
جميع المنشئات، الأعضاء، المعاملات، والمعدلات، المذكورة في صفحة الدليل هذه، معرّفة في مساحة الاسم FBB.

ترث من

FBB::TableSupport

المنشئات

المنشئات الافتراضية ومنشئات النقل وعامل تعيين النقل متاحة.

الدوال الأعضاء

جميع الأعضاء العامة لـ TableSupport متاحة. ارجع إلى صفحة الدليل tablesupport(3bobcat) للتفاصيل.

مثال

#include <iostream>
#include <bobcat/table>
#include <bobcat/tablelines>
using namespace std;
using namespace FBB;
int main(int argc, char **argv)
{
    TableLines tablelines;
                                // عرض/فواصل الأعمدة 0، 1، و 2
    tablelines << 0 << " | " << " | ";
                                // خط أفقي فوق العمودين 1 و 2 للصف 1
    tablelines << TableLines::HLine(1, 1, 3);
    Table tab(tablelines, 3, Table::ROWWISE, Table::EQUALWIDTH);
    // أو: Table tab(tablelines, 3, Table::ROWWISE);
    tab << Align(0, std::left);     // تعيين محاذاة العمود غير الافتراضية
    tab.fill(argv + 1, argv + argc);// ملء نطاق القيم
    cout << tab << ’\n’;            // إكمال الجدول وإدراجه
    tab << "hello" << "" << "wo";   // إضافة عناصر إضافية.
    if (tab.nRows() > 2)
        tab << Align(2, 2, center); // تعيين تخطيط عنصر معين
    cout << tab << ’\n’;
}
