الوصف

تحدد هذه الماكروات وتعمل على قائمة مزدوجة ذات نهاية ذيل.

في تعريفات الماكرو، TYPE هو اسم بنية محددة من قبل المستخدم يجب أن تحتوي على حقل من النوع TAILQ_ENTRY، باسم NAME. الوسيط HEADNAME هو اسم بنية محددة من قبل المستخدم يجب تعريفها باستخدام الماكرو TAILQ_HEAD().

الإنشاء

يتم تعريف قائمة نهاية الذيل بواسطة بنية محددة بواسطة الماكرو TAILQ_HEAD(). تحتوي هذه البنية على زوج من المؤشرات، أحدهما يشير إلى العنصر الأول في القائمة والآخر إلى العنصر الأخير في القائمة. العناصر مرتبطة بشكل مزدوج، مما يسمح بإزالة أي عنصر دون المرور عبر القائمة. يمكن إضافة عناصر جديدة إلى القائمة بعد عنصر موجود، أو قبل عنصر آخر، في بداية القائمة، أو في نهايتها. يتم تعريف بنية TAILQ_HEAD على النحو التالي:


TAILQ_HEAD(HEADNAME, TYPE) head;

حيث struct HEADNAME هي البنية التي سيتم تعريفها و struct TYPE هو نوع العناصر المرتبطة في القائمة. يمكن لاحقًا تعريف مؤشر إلى رأس القائمة كالتالي:


struct HEADNAME *headp;

(الأسماء head و headp قابلة للاختيار من قبل المستخدم.)

TAILQ_ENTRY() يعرف بنية تربط عناصر القائمة.

TAILQ_HEAD_INITIALIZER() يقيم إلى مُهيئ للقائمة head.

TAILQ_INIT() يهيئ القائمة المشار إليها بواسطة head. TAILQ_EMPTY() يقيم إلى صحيح (true) إذا لم تكن هناك عناصر في القائمة.

الإدراج

TAILQ_INSERT_HEAD() تُدرج العنصر الجديد elm في بداية القائمة.

TAILQ_INSERT_TAIL() تُدرج العنصر الجديد elm في نهاية القائمة.

TAILQ_INSERT_BEFORE() تُدرج العنصر الجديد elm قبل العنصر listelm.

TAILQ_INSERT_AFTER() تُدرج العنصر الجديد elm بعد العنصر listelm.

التكرار

TAILQ_FIRST() تُرجع العنصر الأول في القائمة، أو NULL إذا كانت القائمة فارغة.

TAILQ_LAST() تُرجع العنصر الأخير في القائمة. إذا كانت القائمة فارغة، فإن القيمة المرجعة هي NULL.

TAILQ_PREV() تُرجع العنصر السابق في القائمة، أو NULL إذا كان هذا هو الأول.

TAILQ_NEXT() تُرجع العنصر التالي في القائمة، أو NULL إذا كان هذا هو الأخير.

TAILQ_FOREACH() يتكرر عبر القائمة المشار إليها بواسطة head في الاتجاه الأمامي، ويعين كل عنصر إلى var. يتم تعيين var إلى NULL إذا انتهت الحلقة بشكل طبيعي، أو إذا لم تكن هناك عناصر.

TAILQ_FOREACH_REVERSE() يتكرر عبر القائمة المشار إليها بواسطة head في الاتجاه العكسي، ويعين كل عنصر إلى var.

الإزالة

TAILQ_REMOVE() تزيل العنصر elm من القائمة.

ميزات أخرى

TAILQ_CONCAT() تقوم بدمج قائمة الرأس head2 في نهاية قائمة الرأس head1، وإزالة جميع العناصر من القائمة السابقة.

قيمة الإرجاع

TAILQ_EMPTY() تُرجع قيمة غير صفرية إذا كانت القائمة فارغة، وصفر إذا كان هناك عنصر واحد على الأقل.

TAILQ_FIRST(), TAILQ_LAST(), TAILQ_PREV(), و TAILQ_NEXT() تُرجع مؤشرًا إلى بنية TYPE الأولى، أو الأخيرة، أو السابقة، أو التالية، على التوالي.

TAILQ_HEAD_INITIALIZER() تُرجع مُهيئًا يمكن تعيينه إلى القائمة head.

المعايير

BSD.

التاريخ

4.4BSD.

محاذير

TAILQ_FOREACH() و TAILQ_FOREACH_REVERSE() لا تسمحان بإزالة أو تحرير قيمة var داخل الحلقة، لأن ذلك قد يتداخل مع التكرار. TAILQ_FOREACH_SAFE() و TAILQ_FOREACH_REVERSE_SAFE(), الموجودتان في أنظمة BSD ولكن ليس في glibc، تعالجان هذا القيد بالسماح بإزالة وتحرير var داخل الحلقة دون التداخل مع التكرار.

أمثلة

#include <stddef.h>
#include <stdio.h>
#include <stdlib.h>
#include <sys/queue.h>
struct entry {
    int data;
    TAILQ_ENTRY(entry) entries;             /* قائمة الذيل */
};
TAILQ_HEAD(tailhead, entry);
int
main(void)
{
    struct entry *n1, *n2, *n3, *np;
    struct tailhead head;                   /* رأس القائمة */
    int i;
    TAILQ_INIT(&head);                      /* تهيئة القائمة */
    n1 = malloc(sizeof(struct entry));      /* الإدراج في الرأس */
    TAILQ_INSERT_HEAD(&head, n1, entries);
    n1 = malloc(sizeof(struct entry));      /* الإدراج في الذيل */
    TAILQ_INSERT_TAIL(&head, n1, entries);
    n2 = malloc(sizeof(struct entry));      /* الإدراج بعد */
    TAILQ_INSERT_AFTER(&head, n1, n2, entries);
    n3 = malloc(sizeof(struct entry));      /* الإدراج قبل */
    TAILQ_INSERT_BEFORE(n2, n3, entries);
    TAILQ_REMOVE(&head, n2, entries);       /* الإزالة */
    free(n2);
                                            /* التكرار الأمامي */
    i = 0;
    TAILQ_FOREACH(np, &head, entries)
        np->data = i++;
                                            /* التكرار العكسي */
    TAILQ_FOREACH_REVERSE(np, &head, tailhead, entries)
        printf("%i\n", np->data);
                                            /* إزالة القائمة */
    n

NAME

TAILQ_ENTRY, TAILQ_INIT, TAILQ_INSERT_HEAD, TAILQ_INSERT_TAIL, TAILQ_INSERT_AFTER, TAILQ_INSERT_BEFORE, TAILQ_REMOVE, TAILQ_REPLACE, TAILQ_GETNEXT, TAILQ_PREV, TAILQ_NEXT, TAILQ_LAST, TAILQ_FOREACH, TAILQ_FOREACH_SAFE, TAILQ_HEAD, TAILQ_HEAD_INITIALIZER, TAILQ_EMPTY, TAILQ_FIRST, TAILQ_LAST, TAILQ_COUNT — macros for manipulating tail queues

LIBRARY

libbsd.

SYNOPSIS

#include <bsd/sys/queue.h>

struct TAILQ_ENTRY { ... };

void TAILQ_INIT(struct TAILQ_HEAD *head);

void TAILQ_INSERT_HEAD(struct TAILQ_HEAD *head, struct TAILQ_ENTRY *elm, ...);

void TAILQ_INSERT_TAIL(struct TAILQ_HEAD *head, struct TAILQ_ENTRY *elm, ...);

void TAILQ_INSERT_AFTER(struct TAILQ_HEAD *head, struct TAILQ_ENTRY *listelm, struct TAILQ_ENTRY *elm, ...);

void TAILQ_INSERT_BEFORE(struct TAILQ_HEAD *head, struct TAILQ_ENTRY *listelm, struct TAILQ_ENTRY *elm, ...);

void TAILQ_REMOVE(struct TAILQ_HEAD *head, struct TAILQ_ENTRY *elm, ...);

void TAILQ_REPLACE(struct TAILQ_HEAD *head, struct TAILQ_ENTRY *oldelm, struct TAILQ_ENTRY *newelm, ...);

struct TAILQ_ENTRY * TAILQ_GETNEXT(struct TAILQ_ENTRY *elm, ...);

struct TAILQ_ENTRY * TAILQ_PREV(struct TAILQ_ENTRY *elm, ...);

struct TAILQ_ENTRY * TAILQ_NEXT(struct TAILQ_ENTRY *elm, ...);

struct TAILQ_ENTRY * TAILQ_LAST(struct TAILQ_HEAD *head, ...);

#define TAILQ_FOREACH(var, head, ...) ...

#define TAILQ_FOREACH_SAFE(var, head, nextvar, ...) ...

struct TAILQ_HEAD { ... };

#define TAILQ_HEAD_INITIALIZER(head) ...

int TAILQ_EMPTY(struct TAILQ_HEAD *head);

struct TAILQ_ENTRY * TAILQ_FIRST(struct TAILQ_HEAD *head);

struct TAILQ_ENTRY * TAILQ_LAST(struct TAILQ_HEAD *head, ...);

int TAILQ_COUNT(struct TAILQ_HEAD *head);

DESCRIPTION

These macros provide an implementation of tail queues. Tail queues are doubly-linked lists that support efficient insertion and deletion at both ends. The macros operate on structures that contain the tail queue linkage pointers.

TAILQ_ENTRY declares the structure that holds the forward and backward pointers needed for a tail queue entry. This structure must be included in the structure that is to be part of the tail queue.

TAILQ_INIT initializes an empty tail queue represented by head.

TAILQ_INSERT_HEAD inserts the element elm at the head of the tail queue head.

TAILQ_INSERT_TAIL inserts the element elm at the tail of the tail queue head.

TAILQ_INSERT_AFTER inserts the element elm after the element listelm in the tail queue head.

TAILQ_INSERT_BEFORE inserts the element elm before the element listelm in the tail queue head.

TAILQ_REMOVE removes the element elm from the tail queue head.

TAILQ_REPLACE replaces the element oldelm with the element newelm in the tail queue head.

TAILQ_GETNEXT returns the next element in the tail queue after elm.

TAILQ_PREV returns the previous element in the tail queue before elm.

TAILQ_NEXT is a synonym for TAILQ_GETNEXT.

TAILQ_LAST returns the last element in the tail queue head.

TAILQ_FOREACH iterates over all elements in the tail queue head. The variable var will be assigned each element in turn. The third argument is the name of the tail queue entry field within the structure.

TAILQ_FOREACH_SAFE is a safe version of TAILQ_FOREACH that allows elements to be removed during iteration. The variable nextvar should be declared as a pointer to the same type as var and will hold the next element before the current element is processed.

TAILQ_HEAD declares a tail queue head structure.

TAILQ_HEAD_INITIALIZER is a macro that can be used to initialize a tail queue head statically.

TAILQ_EMPTY returns true if the tail queue head is empty, and false otherwise.

TAILQ_FIRST returns the first element in the tail queue head, or NULL if the queue is empty.

TAILQ_LAST returns the last element in the tail queue head, or NULL if the queue is empty.

TAILQ_COUNT returns the number of elements in the tail queue head.

The third and subsequent arguments to TAILQ_INSERT_HEAD, TAILQ_INSERT_TAIL, TAILQ_INSERT_AFTER, TAILQ_INSERT_BEFORE, TAILQ_REMOVE, and TAILQ_REPLACE, and the second argument to TAILQ_GETNEXT, TAILQ_PREV, TAILQ_NEXT, TAILQ_LAST, TAILQ_FOREACH, TAILQ_FOREACH_SAFE are the names of the fields within the tail queue entry structure that hold the forward and backward pointers. These macros assume that the tail queue entry structure is embedded within a larger structure, and that the field names are unique across all structures using tail queues.

RETURN VALUE

TAILQ_FIRST, TAILQ_LAST, TAILQ_NEXT, and TAILQ_PREV return a pointer to the respective tail queue entry, or NULL if the queue is empty or the element is not found. TAILQ_EMPTY returns 1 if the queue is empty, and 0 otherwise. TAILQ_COUNT returns the number of elements in the queue.

STANDARDS

These macros are based on the BSD queue.h implementation and are widely available on BSD systems and systems that emulate BSD functionality.

HISTORY

The tail queue macros first appeared in FreeBSD 2.1.

CAVEATS

The macros assume that the tail queue entry structure is embedded within a larger structure, and that the field names are unique across all structures using tail queues. This allows for efficient implementation without requiring the user to explicitly pass the structure type.

EXAMPLES

The following example demonstrates how to use the tail queue macros to manage a list of integers:

First, define a structure that includes the tail queue entry:

struct int_list_entry {
    int data;
    TAILQ_ENTRY(int_list_entry) entries;
};

Then, use the macros to manipulate the list:

#include <stdio.h>
#include <stdlib.h>
#include <bsd/sys/queue.h>

struct int_list_entry {
    int data;
    TAILQ_ENTRY(int_list_entry) entries;
};

struct int_list_head {
    TAILQ_HEAD(int_list_head_s, int_list_entry) head;
};

int
main(void)
{
    struct int_list_head list;
    struct int_list_entry *n1, *n2;

    TAILQ_INIT(&list.head);

    /* Insert elements */
    n1 = malloc(sizeof(struct int_list_entry));
    n1->data = 10;
    TAILQ_INSERT_TAIL(&list.head, n1, entries);

    n1 = malloc(sizeof(struct int_list_entry));
    n1->data = 20;
    TAILQ_INSERT_TAIL(&list.head, n1, entries);

    n1 = malloc(sizeof(struct int_list_entry));
    n1->data = 5;
    TAILQ_INSERT_HEAD(&list.head, n1, entries);

    /* Iterate and print elements */
    printf("List elements:\n");
    TAILQ_FOREACH(n1, &list.head, entries) {
        printf("%d\n", n1->data);
    }

    /* Remove and free elements */
    n1 = TAILQ_FIRST(&list.head);
    while (n1 != NULL) {
        n2 = TAILQ_NEXT(n1, entries);
        free(n1);
        n1 = n2;
    }
    TAILQ_INIT(&list.head);
    exit(EXIT_SUCCESS);
}

الوصف

توفر وحدات الماكرو هذه تطبيقًا لقوائم الذيل. قوائم الذيل هي قوائم مرتبطة مزدوجة تدعم الإدراج والحذف الفعالين في كلا الطرفين. تعمل وحدات الماكرو على الهياكل التي تحتوي على مؤشرات ربط قائمة الذيل.

TAILQ_ENTRY يعلن عن الهيكل الذي يحمل مؤشرات التقدم والخلف اللازمة لإدخال قائمة الذيل. يجب تضمين هذا الهيكل في الهيكل الذي سيتم جزءًا من قائمة الذيل.

TAILQ_INIT تهيئة قائمة ذيل فارغة ممثلة بـ head.

TAILQ_INSERT_HEAD يدرج العنصر elm في بداية قائمة الذيل head.

TAILQ_INSERT_TAIL يدرج العنصر elm في نهاية قائمة الذيل head.

TAILQ_INSERT_AFTER يدرج العنصر elm بعد العنصر listelm في قائمة الذيل head.

TAILQ_INSERT_BEFORE يدرج العنصر elm قبل العنصر listelm في قائمة الذيل head.

TAILQ_REMOVE يزيل العنصر elm من قائمة الذيل head.

TAILQ_REPLACE يستبدل العنصر oldelm بالعنصر newelm في قائمة الذيل head.

TAILQ_GETNEXT يعيد العنصر التالي في قائمة الذيل بعد elm.

TAILQ_PREV يعيد العنصر السابق في قائمة الذيل قبل elm.

TAILQ_NEXT هو مرادف لـ TAILQ_GETNEXT.

TAILQ_LAST يعيد العنصر الأخير في قائمة الذيل head.

TAILQ_FOREACH يتكرر على جميع العناصر في قائمة الذيل head. سيتم تعيين المتغير var لكل عنصر بالتناوب. الوسيط الثالث هو اسم حقل إدخال قائمة الذيل داخل الهيكل.

TAILQ_FOREACH_SAFE هو نسخة آمنة من TAILQ_FOREACH تسمح بإزالة العناصر أثناء التكرار. يجب تعريف المتغير nextvar كمؤشر إلى نفس نوع var وسيحتوي على العنصر التالي قبل معالجة العنصر الحالي.

TAILQ_HEAD يعلن عن هيكل رأس قائمة الذيل.

TAILQ_HEAD_INITIALIZER هي وحدة ماكرو يمكن استخدامها لتهيئة رأس قائمة الذيل بشكل ثابت.

TAILQ_EMPTY يعيد قيمة صحيحة إذا كانت قائمة الذيل head فارغة، وقيمة خاطئة بخلاف ذلك.

TAILQ_FIRST يعيد العنصر الأول في قائمة الذيل head، أو NULL إذا كانت القائمة فارغة.

TAILQ_LAST يعيد العنصر الأخير في قائمة الذيل head، أو NULL إذا كانت القائمة فارغة.

TAILQ_COUNT يعيد عدد العناصر في قائمة الذيل head.

الوسائط الثالثة وما بعدها لـ TAILQ_INSERT_HEAD، TAILQ_INSERT_TAIL، TAILQ_INSERT_AFTER، TAILQ_INSERT_BEFORE، TAILQ_REMOVE، و TAILQ_REPLACE، والوسيط الثاني لـ TAILQ_GETNEXT، TAILQ_PREV، TAILQ_NEXT، TAILQ_LAST، TAILQ_FOREACH، TAILQ_FOREACH_SAFE هي أسماء الحقول داخل هيكل إدخال قائمة الذيل التي تحمل مؤشرات التقدم والخلف. تفترض وحدات الماكرو هذه أن هيكل إدخال قائمة الذيل مضمن داخل هيكل أكبر، وأن أسماء الحقول فريدة عبر جميع الهياكل التي تستخدم قوائم الذيل.

قيمة الإرجاع

TAILQ_FIRST، TAILQ_LAST، TAILQ_NEXT، و TAILQ_PREV تعيد مؤشرًا إلى إدخال قائمة الذيل المعني، أو NULL إذا كانت القائمة فارغة أو لم يتم العثور على العنصر. TAILQ_EMPTY تعيد 1 إذا كانت القائمة فارغة، و 0 بخلاف ذلك. TAILQ_COUNT يعيد عدد العناصر في القائمة.

المعايير

تستند وحدات الماكرو هذه إلى تطبيق queue.h الخاص بنظام BSD وهي متاحة على نطاق واسع في أنظمة BSD والأنظمة التي تحاكي وظائف BSD.

التاريخ

ظهرت وحدات الماكرو الخاصة بقوائم الذيل لأول مرة في FreeBSD 2.1.

محاذير

تفترض وحدات الماكرو أن هيكل إدخال قائمة الذيل مضمن داخل هيكل أكبر، وأن أسماء الحقول فريدة عبر جميع الهياكل التي تستخدم قوائم الذيل. هذا يسمح بتنفيذ فعال دون مطالبة المستخدم بتمرير نوع الهيكل بشكل صريح.

أمثلة

يوضح المثال التالي كيفية استخدام وحدات الماكرو الخاصة بقوائم الذيل لإدارة قائمة من الأعداد الصحيحة:

أولاً، عرّف هيكلًا يتضمن إدخال قائمة الذيل:

struct int_list_entry {
    int data;
    TAILQ_ENTRY(int_list_entry) entries;
};

بعد ذلك، استخدم وحدات الماكرو لمعالجة القائمة:

#include <stdio.h>
#include <stdlib.h>
#include <bsd/sys/queue.h>

struct int_list_entry {
    int data;
    TAILQ_ENTRY(int_list_entry) entries;
};

struct int_list_head {
    TAILQ_HEAD(int_list_head_s, int_list_entry) head;
};

int
main(void)
{
    struct int_list_head list;
    struct int_list_entry *n1, *n2;

    TAILQ_INIT(&list.head);

    /* Insert elements */
    n1 = malloc(sizeof(struct int_list_entry));
    n1->data = 10;
    TAILQ_INSERT_TAIL(&list.head, n1, entries);

    n1 = malloc(sizeof(struct int_list_entry));
    n1->data = 20;
    TAILQ_INSERT_TAIL(&list.head, n1, entries);

    n1 = malloc(sizeof(struct int_list_entry));
    n1->data = 5;
    TAILQ_INSERT_HEAD(&list.head, n1, entries);

    /* Iterate and print elements */
    printf("List elements:\n");
    TAILQ_FOREACH(n1, &list.head, entries) {
        printf("%d\n", n1->data);
    }

    /* Remove and free elements */
    n1 = TAILQ_FIRST(&list.head);
    while (n1 != NULL) {
        n2 = TAILQ_NEXT(n1, entries);
        free(n1);
        n1 = n2;
    }
    TAILQ_INIT(&list.head);
    exit(EXIT_SUCCESS);
}
