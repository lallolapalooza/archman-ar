const Tcl_ChannelType *typePtr (in)
مؤشر إلى بنية تحتوي على نقاط الوصول إلى الوظائف التي يمكن استدعاؤها لإجراء عمليات الإدخال والإخراج على القناة.
const char *channelName (in)
اسم القناة، مثل file3؛ يجب ألا يكون قيد الاستخدام بواسطة أي قناة أخرى. يمكن أن يكون NULL، وفي هذه الحالة يتم إنشاء القناة بدون اسم. إذا تم تعيين القناة التي تم إنشاؤها إلى إحدى القنوات القياسية (stdin، stdout، أو stderr)، فسيكون الاسم المعين هو اسم القناة القياسية.
ClientData instanceData (in)
قيمة عشوائية لكلمة واحدة مرتبطة بهذه القناة. يتم تمرير هذه القيمة إلى الوظائف الموجودة في typePtr عند استدعائها.
int mask (in)
مزيج OR من TCL_READABLE و TCL_WRITABLE يشير إلى ما إذا كانت القناة قابلة للقراءة والكتابة.
Tcl_Channel channel (in)
القناة المراد العمل عليها.
int direction (in)
TCL_READABLE يعني أنك تريد مقبض الإدخال؛ TCL_WRITABLE يعني أنك تريد مقبض الإخراج.
ClientData *handlePtr (out)
المؤشر إلى الموقع الذي يجب تخزين مقبض نظام التشغيل المطلوب فيه.
int size (in)
الحجم، بالبايت، للمخازن المؤقتة التي سيتم تخصيصها في هذه القناة.
int mask (in)
مزيج OR من TCL_READABLE، TCL_WRITABLE، و TCL_EXCEPTION يشير إلى الأحداث التي وقعت على هذه القناة.
Tcl_Interp *interp (in)
المترجم الحالي. (قد يكون NULL)
const char *optionName (in)
اسم الخيار غير الصحيح.
const char *optionList (in)
قائمة خيارات محددة (كلمات مفصولة بمسافات، بدون -) لإلحاقها بقائمة الخيارات القياسية. يمكن أن تكون NULL فقط لرسائل الخطأ القياسية.

الوصف

يستخدم Tcl بنية قناة من طبقتين. يوفر طبقة عليا عامة للسماح لبرامج C و Tcl بإجراء الإدخال والإخراج باستخدام نفس واجهات برمجة التطبيقات لمختلف الملفات والأجهزة والمقابس وما إلى ذلك. يتم وصف واجهات برمجة التطبيقات C العامة في صفحة الدليل Tcl_OpenFileChannel.

توفر الطبقة السفلية برامج تشغيل قنوات محددة لكل نوع جهاز مدعوم على كل منصة. يصف إدخال الدليل هذا لواجهات برمجة التطبيقات C التي تتواصل بين الطبقة العامة وبرامج تشغيل القنوات المحددة. كما يشرح كيف يمكن إضافة أنواع قنوات جديدة عن طريق توفير برامج تشغيل قنوات جديدة.

تتكون برامج تشغيل القنوات من عدة مكونات: أولاً، يوفر كل برنامج تشغيل قناة بنية Tcl_ChannelType التي تحتوي على مؤشرات إلى وظائف تنفذ العمليات المختلفة التي تستخدمها الطبقة العامة للتواصل مع برنامج تشغيل القناة. يتم وصف بنية Tcl_ChannelType والوظائف المشار إليها بواسطةها في قسم TCL_CHANNELTYPE، أدناه.

ثانياً، عادةً ما توفر برامج تشغيل القنوات أمر Tcl لإنشاء مثيلات من هذا النوع من القنوات. على سبيل المثال، ينشئ أمر Tcl open قنوات تستخدم برامج تشغيل قنوات الملفات والأوامر، وينشئ أمر Tcl socket قنوات تستخدم بروتوكولات شبكة TCP للاتصال عبر الشبكة.

ثالثاً، توفر برامج تشغيل القنوات اختيارياً وظيفة C لإنشاء مثيلات من هذا النوع من القنوات. على سبيل المثال، تفتح Tcl_OpenFileChannel قناة تستخدم برنامج تشغيل قناة الملفات، وتفتح Tcl_OpenTcpClient قناة تستخدم بروتوكول شبكة TCP. تستخدم وظائف الإنشاء هذه عادةً Tcl_CreateChannel داخليًا لفتح القناة.

لإضافة نوع قناة جديد، يجب عليك تنفيذ واجهة برمجة تطبيقات C أو أمر Tcl يفتح قناة عن طريق استدعاء Tcl_CreateChannel. عندما يستدعي برنامج التشغيل الخاص بك Tcl_CreateChannel، فإنه يمرر بنية Tcl_ChannelType التي تصف إجراءات الإدخال / الإخراج لبرنامج التشغيل الخاص بك. ثم تستدعي الطبقة العامة الوظائف المشار إليها في تلك البنية لإجراء عمليات على القناة.

تفتح Tcl_CreateChannel قناة جديدة وتربط typePtr و instanceData بها. تفتح القناة في الوضع المحدد بواسطة mask. لمناقشة برامج تشغيل القنوات وعملياتها وبنية Tcl_ChannelType، انظر القسم TCL_CHANNELTYPE، أدناه.

تتفاعل Tcl_CreateChannel مع الكود الذي يدير القنوات القياسية. بمجرد تهيئة قناة قياسية عن طريق استدعاء Tcl_GetStdChannel أو استدعاء Tcl_SetStdChannel، سيؤدي إغلاق هذه القناة إلى جعل الاستدعاء التالي لـ Tcl_CreateChannel يعين هذه القناة كقناة قياسية جديدة أيضًا. انظر Tcl_StandardChannels لمناقشة عامة حول القنوات القياسية وسلوك مكتبة Tcl فيما يتعلق بها.

تُرجع Tcl_GetChannelInstanceData بيانات المثيل المرتبطة بالقناة في channel. هذا هو نفس وسيط instanceData في استدعاء Tcl_CreateChannel الذي أنشأ هذه القناة.

تُرجع Tcl_GetChannelType مؤشرًا إلى بنية Tcl_ChannelType التي يستخدمها القناة في الوسيط channel. هذا هو نفس وسيط typePtr في استدعاء Tcl_CreateChannel الذي أنشأ هذه القناة.

تُرجع Tcl_GetChannelName سلسلة تحتوي على الاسم المرتبط بالقناة، أو NULL إذا كان وسيط channelName لـ Tcl_CreateChannel هو NULL.

تضع Tcl_GetChannelHandle مقبض نظام التشغيل المرتبط بالقناة channel للاتجاه direction المعطى في الموقع المحدد بواسطة handlePtr.

المحدد بواسطة handlePtr ويعيد TCL_OK. إذا لم يكن للقناة مقبض جهاز للاتجاه المحدد، فسيتم إرجاع TCL_ERROR بدلاً من ذلك. ستعيد برامج تشغيل القنوات المختلفة أنواعًا مختلفة من المقابض. راجع إدخالات الدليل لكل برنامج تشغيل لتحديد نوع المقبض الذي تم إرجاعه.

Tcl_GetChannelThread يعيد معرف الخيط الذي يدير حاليًا channel المحدد. يسمح هذا لبرامج تشغيل القنوات بإرسال أحداث الملف الخاصة بها إلى طابور الأحداث الصحيح حتى بالنسبة لنواة متعددة الخيوط.

Tcl_GetChannelMode يعيد مجموعة ORed من TCL_READABLE و TCL_WRITABLE، مما يشير إلى ما إذا كانت القناة مفتوحة للإدخال والإخراج.

Tcl_GetChannelBufferSize يعيد حجم المخازن المؤقتة، بالبايت، المخصصة لتخزين الإدخال أو الإخراج في channel. إذا لم يتم تعيين القيمة بواسطة استدعاء سابق لـ Tcl_SetChannelBufferSize، الموصوفة أدناه، فسيتم إرجاع القيمة الافتراضية 4096.

Tcl_SetChannelBufferSize يعين الحجم، بالبايت، للمخازن المؤقتة التي سيتم تخصيصها في العمليات اللاحقة على القناة لتخزين الإدخال أو الإخراج. يجب أن يكون الوسيط size بين واحد ومليون، مما يسمح بمخازن مؤقتة من بايت واحد إلى مليون بايت. إذا كان size خارج هذا النطاق، فإن Tcl_SetChannelBufferSize يعين حجم المخزن المؤقت إلى 4096.

Tcl_NotifyChannel يتم استدعاؤها بواسطة برنامج تشغيل القناة للإشارة إلى الطبقة العامة أن الأحداث المحددة بواسطة mask قد وقعت على القناة. برامج تشغيل القنوات مسؤولة عن استدعاء هذه الدالة كلما احتاجت معالجات القناة إلى الاستدعاء للقناة (أو تم تنفيذ مهام معلقة أخرى مثل مسح للكتابة). انظر WATCHPROC أدناه لمزيد من التفاصيل.

Tcl_BadChannelOption يتم استدعاؤها من setOptionProc أو getOptionProc الخاص ببرنامج تشغيل معين لإنشاء رسالة خطأ كاملة.

Tcl_ChannelBuffered يعيد عدد البايتات من الإدخال المخزنة حاليًا في المخزن المؤقت الداخلي (منطقة الدفع الخلفي) للقناة نفسها. لا يقوم بالإبلاغ عن البيانات في المخازن المؤقتة الإجمالية لطبقة القنوات التي تنتمي إليها القناة المقدمة.

Tcl_IsChannelShared يتحقق من عداد المرجع لـ channel المحدد ويعيد ما إذا كانت channel مشتركة بين مترجمات متعددة (النتيجة == 1) أم لا (النتيجة == 0).

Tcl_IsChannelRegistered يتحقق مما إذا كانت channel المحددة مسجلة في interp المحدد (النتيجة == 1) أم لا (النتيجة == 0).

Tcl_IsChannelExisting يتحقق مما إذا كانت قناة بالاسم المحدد مسجلة في قائمة القنوات العالمية (على مستوى الخيط) (النتيجة == 1) أم لا (النتيجة == 0).

Tcl_CutChannel يزيل channel المحدد من قائمة القنوات العالمية (على مستوى الخيط) (للخيط الحالي). لا يُسمح بالتطبيق على قناة لا تزال مسجلة في أي مترجم. كما أنه يخطر برنامج التشغيل إذا كان إصدار Tcl_ChannelType هو TCL_CHANNEL_VERSION_4 (أو أعلى)، وتم تعريف Tcl_DriverThreadActionProc له.

Tcl_SpliceChannel يضيف channel المحدد إلى قائمة القنوات العالمية (على مستوى الخيط) (للخيط الحالي). لا يُسمح بالتطبيق على قناة مسجلة في أي مترجم. كما أنه يخطر برنامج التشغيل إذا كان إصدار Tcl_ChannelType هو TCL_CHANNEL_VERSION_4 (أو أعلى)، وتم تعريف Tcl_DriverThreadActionProc له.

Tcl_ClearChannelHandlers يزيل جميع معالجات القناة والبرامج النصية للأحداث المرتبطة بـ channel المحدد، وبالتالي يوقف جميع معالجة الأحداث لهذه القناة.

TCL_CHANNELTYPE

يوفر برنامج تشغيل القناة بنية Tcl_ChannelType تحتوي على مؤشرات إلى الدوال التي تنفذ العمليات المختلفة على القناة؛ يتم استدعاء هذه العمليات حسب الحاجة من قبل الطبقة العامة. تم ترقيم البنية بدءًا من Tcl 8.3.2/8.4 لتصحيح مشكلة مع برامج تشغيل القنوات المتراكمة. انظر قسم OLD CHANNEL TYPES أدناه للحصول على تفاصيل حول البنية القديمة.

تحتوي بنية Tcl_ChannelType على الحقول التالية:

typedef struct Tcl_ChannelType {
        const char *typeName;
        Tcl_ChannelTypeVersion version;
        Tcl_DriverCloseProc *closeProc;
        Tcl_DriverInputProc *inputProc;
        Tcl_DriverOutputProc *outputProc;
        Tcl_DriverSeekProc *seekProc;
        Tcl_DriverSetOptionProc *setOptionProc;
        Tcl_DriverGetOptionProc *getOptionProc;
        Tcl_DriverWatchProc *watchProc;
        Tcl_DriverGetHandleProc *getHandleProc;
        Tcl_DriverClose2Proc *close2Proc;
        Tcl_DriverBlockModeProc *blockModeProc;
        Tcl_DriverFlushProc *flushProc;
        Tcl_DriverHandlerProc *handlerProc;
        Tcl_DriverWideSeekProc *wideSeekProc;
        Tcl_DriverThreadActionProc *threadActionProc;
        Tcl_DriverTruncateProc *truncateProc;
} Tcl_ChannelType;

ليس من الضروري توفير تطبيقات لجميع عمليات القناة. يمكن تعيين تلك التي ليست ضرورية إلى NULL في الهيكل: blockModeProc، seekProc، setOptionProc، getOptionProc، getHandleProc، و close2Proc، بالإضافة إلى flushProc، handlerProc، threadActionProc، و truncateProc. يجب أن تعيد الدوال الأخرى التي لا يمكن تنفيذها بطريقة ذات معنى EINVAL عند استدعائها، للإشارة إلى أن العمليات التي تمثلها غير متاحة. لاحظ أيضًا أن wideSeekProc يمكن أن تكون NULL إذا كان seekProc كذلك.

يجب على المستخدم فقط استخدام الهيكل أعلاه لتهيئة Tcl_ChannelType. عند الإشارة إلى حقول في بنية Tcl_ChannelType، يجب استخدام الدوال التالية للحصول على القيم: Tcl_ChannelName، Tcl_ChannelVersion، Tcl_ChannelBlockModeProc، Tcl_ChannelCloseProc، Tcl_ChannelClose2Proc، Tcl_ChannelInputProc، Tcl_ChannelOutputProc، Tcl_ChannelSeekProc، Tcl_ChannelWideSeekProc، Tcl_ChannelThreadActionProc، Tcl_ChannelTruncateProc، Tcl_ChannelSetOptionProc، Tcl_ChannelGetOptionProc، Tcl_ChannelWatchProc، Tcl_ChannelGetHandleProc، Tcl_ChannelFlushProc، أو Tcl_ChannelHandlerProc.

تم إجراء التغيير على الهياكل بطريقة تجعل أنواع القنوات القياسية متوافقة ثنائياً. ومع ذلك، فإن أنواع القنوات التي تستخدم قنوات متراكمة (مثل TLS، Trf) لديها إصدارات جديدة لتتوافق مع التغيير المذكور أعلاه نظرًا لأن التعليمات البرمجية السابقة للقنوات المتراكمة كانت بها مشاكل.

TYPENAME

يحتوي الحقل typeName على سلسلة منتهية بـ NULL تحدد نوع الجهاز الذي ينفذه برنامج التشغيل هذا، على سبيل المثال، file أو socket.

يمكن استرداد هذه القيمة باستخدام Tcl_ChannelName، والتي تعيد مؤشرًا إلى السلسلة.

VERSION

يجب تعيين الحقل version إلى إصدار البنية الذي تحتاجه. TCL_CHANNEL_VERSION_2 هو الحد الأدنى الموصى به. يجب تعيين TCL_CHANNEL_VERSION_3 لتحديد عضو wideSeekProc. يجب تعيين TCL_CHANNEL_VERSION_4 لتحديد عضو threadActionProc (يتضمن wideSeekProc). يجب تعيين TCL_CHANNEL_VERSION_5 لتحديد أعضاء truncateProc (يتضمن wideSeekProc و threadActionProc). إذا لم يتم تعيينه إلى أي من هذه، فسيتم افتراض أن Tcl_ChannelType هذا لديه البنية الأصلية. انظر OLD CHANNEL TYPES لمزيد من التفاصيل. بينما ستتعرف Tcl على الهياكل وتعمل مع كليهما، يجب أن تكون القنوات المتراكمة على الأقل TCL_CHANNEL_VERSION_2 لتعمل بشكل صحيح.

يمكن استرداد هذه القيمة باستخدام Tcl_ChannelVersion، والتي تعيد أحد القيم التالية: TCL_CHANNEL_VERSION_5، TCL_CHANNEL_VERSION_4، TCL_CHANNEL_VERSION_3، TCL_CHANNEL_VERSION_2 أو TCL_CHANNEL_VERSION_1.

BLOCKMODEPROC

يحتوي الحقل blockModeProc على عنوان دالة يتم استدعاؤها بواسطة الطبقة العامة لتعيين وضع الحظر وغير الحظر على الجهاز. يجب أن يتوافق BlockModeProc مع النموذج الأولي التالي:

typedef int Tcl_DriverBlockModeProc(
        ClientData instanceData,
        int mode);

instanceData هو نفس القيمة التي تم تمريرها إلى Tcl_CreateChannel عند إنشاء هذه القناة. الوسيط mode هو إما TCL_MODE_BLOCKING أو TCL_MODE_NONBLOCKING لتعيين الجهاز في وضع الحظر أو غير الحظر. يجب أن تعيد الدالة صفرًا إذا نجحت العملية، أو رمز خطأ POSIX غير صفري إذا فشلت العملية.

إذا نجحت العملية، يمكن للدالة تعديل instanceData المقدمة لتسجيل أن القناة دخلت في وضع الحظر أو غير الحظر ولتنفيذ سلوك الحظر أو غير الحظر. بالنسبة لبعض أنواع الأجهزة، يمكن تنفيذ سلوك الحظر وغير الحظر بواسطة نظام التشغيل الأساسي؛ بالنسبة لأنواع الأجهزة الأخرى، يجب محاكاة السلوك في برنامج تشغيل القناة.

يمكن استرداد هذه القيمة باستخدام Tcl_ChannelBlockModeProc، والتي تعيد مؤشرًا إلى الدالة.

يجب أن يكون برنامج تشغيل القناة الذي ** لا ** يوفر blockModeProc حذرًا جدًا جدًا. يجب عليه إخبار الطبقة العامة بالضبط عن وضع الحظر الذي يقبله، ويجب عليه أيضًا توثيق ذلك للمستخدم حتى لا يتغير وضع الحظر للقناة إلى قيمة غير مقبولة. قد يؤدي أي ارتباك هنا إلى دخول المفسر في طريق مسدود (وهمي ويصعب العثور عليه).

CLOSEPROC AND CLOSE2PROC

يحتوي الحقل closeProc على عنوان دالة يتم استدعاؤها بواسطة الطبقة العامة لتنظيف المعلومات المتعلقة ببرنامج التشغيل عند إغلاق القناة. يجب أن يتوافق CloseProc مع النموذج الأولي التالي:

typedef int Tcl_DriverCloseProc(
        ClientData instanceData,
        Tcl_Interp *interp);

الوسيط instanceData هو نفس القيمة التي تم توفيرها إلى Tcl_CreateChannel عند إنشاء القناة. يجب على الدالة تحرير أي تخزين تحتفظ به برنامج تشغيل القناة لهذه القناة، وإغلاق أجهزة الإدخال والإخراج التي تغلفها هذه القناة. سيتم مسح جميع المخرجات المعلقة إلى الجهاز قبل استدعاء هذه الدالة، ولن يتم استدعاء أي عمليات أخرى لبرنامج التشغيل على هذه المثيل بعد استدعاء closeProc. إذا نجحت عملية الإغلاق، فإن الإجراء

INPUTPROC

يحتوي الحقل inputProc على عنوان دالة تستدعيها الطبقة العامة لقراءة البيانات من جهاز أو ملف وتخزينها في مخزن مؤقت داخلي. يجب أن يتوافق InputProc مع النموذج الأولي التالي:

typedef int Tcl_DriverInputProc(
        ClientData instanceData,
        char *buf,
        int bufSize,
        int *errorCodePtr);

InstanceData هو نفس القيمة التي تم تمريرها إلى Tcl_CreateChannel عند إنشاء القناة. يشير الوسيط buf إلى مصفوفة بايت سيتم تخزين إدخال الجهاز فيها، ويشير الوسيط bufSize إلى عدد البايتات المتاحة في buf.

يشير الوسيط errorCodePtr إلى متغير صحيح توفره الطبقة العامة. إذا حدث خطأ، يجب على الدالة تعيين هذه القيمة إلى رمز خطأ POSIX يحدد الخطأ.

يجب على الدالة قراءة البيانات من الجهاز المغلف بالقناة وتخزينها في buf. في حالة النجاح، يجب أن تعيد الدالة عددًا غير سالب يشير إلى عدد البايتات التي تم قراءتها من الجهاز وتخزينها في buf. في حالة حدوث خطأ، يجب أن تعيد الدالة -1. إذا حدث خطأ بعد قراءة بعض البيانات من الجهاز، فسيتم فقدانها.

إذا كان بإمكان inputProc تحديد أن الجهاز لديه بعض البيانات المتاحة ولكن أقل من المطلوب بواسطة الوسيط bufSize، فيجب على الدالة محاولة قراءة ما هو متاح فقط والعودة دون حظر. إذا لم تكن هناك بيانات متاحة في الجهاز وكانت القناة في وضع غير مانع، فيجب على الدالة العودة بخطأ EAGAIN. إذا لم تكن هناك بيانات متاحة في الجهاز وكانت القناة في وضع مانع، فيجب على الدالة أن تحظر لأقصر فترة ممكنة حتى تقرأ بايت واحد على الأقل من الجهاز، ثم تعيد أقصى ما يمكن قراءته دون حظر.

يمكن استرداد هذه القيمة باستخدام Tcl_ChannelInputProc، والتي تعيد مؤشرًا إلى الدالة.

OUTPUTPROC

يحتوي الحقل outputProc على عنوان دالة تستدعيها الطبقة العامة لنقل البيانات من المخزن المؤقت الداخلي إلى جهاز الإخراج. يجب أن يتوافق OutputProc مع النموذج الأولي التالي:

typedef int Tcl_DriverOutputProc(
        ClientData instanceData,
        const char *buf,
        int toWrite,
        int *errorCodePtr);

InstanceData هو نفس القيمة التي تم تمريرها إلى Tcl_CreateChannel عند إنشاء القناة. يحتوي الوسيط buf على مصفوفة بايت سيتم كتابتها إلى الجهاز، ويشير الوسيط toWrite إلى عدد البايتات التي سيتم كتابتها من الوسيط buf.

يشير الوسيط errorCodePtr إلى متغير صحيح توفره الطبقة العامة. إذا حدث خطأ، يجب على الدالة تعيين هذه القيمة إلى رمز خطأ POSIX يحدد الخطأ.

يجب على الدالة كتابة البيانات في buf إلى جهاز الإخراج المغلف بالقناة. في حالة النجاح، يجب أن تعيد الدالة عددًا غير سالب يشير إلى عدد البايتات التي تم كتابتها إلى جهاز الإخراج. عادة ما تكون قيمة الإرجاع هي نفسها toWrite، ولكنها قد تكون أقل في بعض الحالات، مثل عندما يتم مقاطعة عملية الإخراج بواسطة إشارة. إذا حدث خطأ، يجب أن تعيد الدالة -1. في حالة حدوث خطأ، قد تكون بعض البيانات قد تمت كتابتها إلى الجهاز.

إذا كانت القناة في وضع غير مانع ولا يمكن لجهاز الإخراج استيعاب أي بيانات، فيجب على الدالة العودة بـ -1 مع خطأ EAGAIN دون كتابة أي بيانات.

يمكن استرداد هذه القيمة باستخدام Tcl_ChannelOutputProc، والتي تعيد مؤشرًا إلى الدالة.

SEEKPROC AND WIDESEEKPROC

يحتوي الحقل seekProc على عنوان دالة تستدعيها الطبقة العامة لنقل نقطة الوصول التي سيتم تطبيق عمليات الإدخال أو الإخراج اللاحقة عليها. يجب أن يتوافق SeekProc مع النموذج الأولي التالي:

typedef int Tcl_DriverSeekProc(
        ClientData instanceData,
        long offset,
        int seekMode,
        int *errorCodePtr);

الوسيط instanceData هو نفس القيمة التي تم تمريرها إلى Tcl_CreateChannel عند إنشاء هذه القناة. Offset و seekMode لهما نفس المعنى كما في الإجراء Tcl_Seek (وصف في دليل Tcl_OpenFileChannel).

يشير الوسيط errorCodePtr إلى متغير صحيح توفره الطبقة العامة لإرجاع قيمة errno للدالة. يجب على الدالة تعيين هذه القيمة إلى رمز خطأ POSIX إذا حدث خطأ. إذا كان نوع القناة لا ينفذ تحديد الموضع، فيجب على الدالة تخزين رمز خطأ EINVAL.

قيمة الإرجاع هي نقطة الوصول الجديدة أو -1 في حالة حدوث خطأ. إذا حدث خطأ، يجب على الدالة عدم نقل نقطة الوصول.

إذا كان هناك حقل غير فارغ لـ seekProc، فقد يحتوي الحقل wideSeekProc على عنوان دالة بديلة للاستخدام، والتي تتعامل مع الإزاحات الواسعة (أي، أكبر من 32 بت)، مما يسمح بتحديد المواقع في ملفات أكبر من 2 جيجابايت. يتم استدعاء wideSeekProc تفضيلاً على seekProc، ولكن يجب تعريف كليهما إذا تم تعريف wideSeekProc. يجب أن يتوافق WideSeekProc مع النموذج الأولي التالي:

typedef Tcl_WideInt Tcl_DriverWideSeekProc(
        ClientData instanceData,
        Tcl_WideInt offset,
        int seekMode,
        int *errorCodePtr);

الوسائط وقيم الإرجاع هي نفسها كما هو الحال مع seekProc، باستثناء أن نوع الإزاحة ونوع الإرجاع مختلفان.

يمكن استرداد الحقل seekProc باستخدام Tcl_ChannelSeekProc، والتي تعيد مؤشرًا إلى الدالة، ويمكن استرداد الحقل wideSeekProc باستخدام Tcl_ChannelWideSeekProc.

SETOPTIONPROC

يحتوي الحقل setOptionProc على عنوان دالة تستدعيها الطبقة العامة لتعيين خيار محدد لنوع القناة في قناة. يجب أن يتوافق setOptionProc مع النموذج الأولي التالي:

typedef int Tcl_DriverSetOptionProc(
        ClientData instanceData,
        Tcl_Interp *interp,
        const char *optionName,
        const char *newValue);

optionName هو اسم خيار لتعيينه، و newValue هو القيمة الجديدة لهذا الخيار، كسلسلة نصية. instanceData هو نفس القيمة التي تم تمريرها إلى Tcl_CreateChannel عند إنشاء هذه القناة. يجب على الدالة تنفيذ أي إجراءات خاصة بنوع القناة ضرورية لتنفيذ القيمة الجديدة للخيار.

يتم إدارة بعض الخيارات بواسطة الكود العام ولا يتم استدعاء هذه الدالة أبدًا لتعيينها، على سبيل المثال، -blockmode. أنواع أخرى من الخيارات خاصة بكل نوع قناة ويتم استدعاء دالة setOptionProc الخاصة بالسائق لتنفيذها. قد يكون الحقل setOptionProc فارغًا، مما يشير إلى أن هذا النوع من القنوات لا يدعم خيارات محددة للنوع.

إذا تم تعديل قيمة الخيار بنجاح إلى قيمتها الجديدة، فتعيد الدالة TCL_OK. يجب أن تستدعي Tcl_BadChannelOption، والتي تعيد TCL_ERROR إذا لم يتم التعرف على اسم الخيار optionName. إذا حدد newValue قيمة غير مدعومة أو إذا حدث خطأ في استدعاء نظام، فيجب على الدالة ترك رسالة خطأ في نتيجة interp إذا لم يكن interp فارغًا. يجب على الدالة أيضًا استدعاء Tcl_SetErrno لتخزين رمز خطأ POSIX مناسب.

يمكن استرداد هذه القيمة باستخدام Tcl_ChannelSetOptionProc، والتي تعيد مؤشرًا إلى الدالة.

GETOPTIONPROC

يحتوي الحقل getOptionProc على عنوان دالة تستدعيها الطبقة العامة للحصول على قيمة خيار محدد لنوع القناة في قناة. يجب أن يتوافق getOptionProc مع النموذج الأولي التالي:

typedef int Tcl_DriverGetOptionProc(
        ClientData instanceData,
        Tcl_Interp *interp,
        const char *optionName,
        Tcl_DString *optionValue);

OptionName هو اسم خيار مدعوم من قبل هذا النوع من القنوات. إذا لم يكن اسم الخيار فارغًا، فتقوم الدالة بتخزين قيمته الحالية، كسلسلة نصية، في سلسلة Tcl الديناميكية optionValue. إذا كان optionName فارغًا، فتقوم الدالة بتخزين قائمة بديلة لجميع الخيارات المدعومة وقيمها الحالية في optionValue. في حالة النجاح، تعيد الدالة TCL_OK. يجب أن تستدعي Tcl_BadChannelOption، والتي تعيد TCL_ERROR إذا لم يتم التعرف على اسم الخيار optionName. إذا حدد newValue قيمة غير مدعومة أو إذا حدث خطأ في استدعاء نظام، فيجب على الدالة ترك رسالة خطأ في نتيجة interp إذا لم يكن interp فارغًا. يجب على الدالة أيضًا استدعاء Tcl_SetErrno لتخزين رمز خطأ POSIX مناسب.
