Tcl_NewDictObj(3)

الحزمة: tcl

التراخيص: TCL

اسم

TCL

ملخص

Tcl_DictObjGet, Tcl_DictObjPutKeyList, Tcl_DictObjRemoveKeyList

الوسائط

interp, objPtr, search, key, value, done

الوصف

استخدام Tcl_DictObjGet بشكل متكرر لبناء هذا الملف أمر سهل. باستخدام Tcl_DictObjPutKeyList، يتم إنشاء قواميس متداخلة للمفاتيح غير النهائية حيث لا توجد بعد. باستخدام Tcl_DictObjRemoveKeyList، يجب أن تكون جميع المفاتيح غير النهائية موجودة وتحتوي على قيم قواميس.

مثال

استخدام واجهة تكرار القاموس للبحث عما إذا كان هناك مفتاح يطابق نفسه:

Tcl_DictSearch search;
Tcl_Obj *key, *value;
int done;
/*
 * نفترض أن interp و objPtr هما معلمات. هذه هي الطريقة الاصطلاحية لبدء تكرار على القاموس؛ فهي تضع قفلًا على التمثيل الداخلي الذي يضمن عدم وجود مشاكل في التعديل المتزامن عند استخدام معالجة المراجع العادية أيضًا. يتم تحرير القفل تلقائيًا عند انتهاء التكرار، ولكن يجب تحريره يدويًا إذا خرجت من الحلقة بطريقة استثنائية. ومع ذلك، فمن الآمن محاولة تحرير القفل حتى لو انتهينا من التكرار عبر الحلقة.
 */
if (Tcl_DictObjFirst(interp, objPtr, &search,
        &key, &value, &done) != TCL_OK) {
    return TCL_ERROR;
}
for (; !done ; Tcl_DictObjNext(&search, &key, &value, &done)) {
    /*
     * ملاحظة: strcmp() ليست طريقة جيدة لمقارنة القيم وتستخدم هنا للتوضيح فقط.
     */
    if (!strcmp(Tcl_GetString(key), Tcl_GetString(value))) {
        break;
    }
}
Tcl_DictObjDone(&search);
Tcl_SetObjResult(interp, Tcl_NewBooleanObj(!done));
return TCL_OK;
