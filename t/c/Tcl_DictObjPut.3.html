Tcl_DictObjPut(3)

الحزمة: tcl

التراخيص: TCL

الاسم

Tcl_DictObjGet

الملخص

Tcl_DictObjGet(interp, objPtr, &search, key, value, &done)

المعاملات

interp - المفسر

objPtr - الكائن

search - الباحث

key - المفتاح

value - القيمة

done - الحالة

الوصف

يحافظ على بنية الملف. مع الاستخدام المتكرر لـ Tcl_DictObjGet، يمكن بناؤه بسهولة. مع Tcl_DictObjPutKeyList، يتم إنشاء قواميس متداخلة للمفاتيح غير النهائية حيث لا توجد بعد. مع Tcl_DictObjRemoveKeyList، يجب أن توجد جميع المفاتيح غير النهائية وأن تحتوي على قيم قاموس.

مثال

استخدام واجهة تكرار القاموس للبحث عما إذا كان المفتاح موجودًا يتطابق مع نفسه:

Tcl_DictSearch search;
Tcl_Obj *key, *value;
int done;
/*
 * نفترض أن interp و objPtr هما معلمات. هذه هي الطريقة الأسلوبية لبدء التكرار على القاموس؛ إنها تضع قفلًا على التمثيل الداخلي لضمان عدم وجود تعديلات متزامنة عندما يتم أيضًا استخدام التعامل مع المراجع من قبل الكائنات. يتم تحرير القفل تلقائيًا عند اكتمال التكرار، ولكن يجب تحريره يدويًا عند حدوث خروج استثنائي من التكرار. ومع ذلك، من الآمن محاولة تحرير القفل حتى لو انتهينا بالفعل من التكرار على التكرار.
 */
if (Tcl_DictObjFirst(interp, objPtr, &search,
        &key, &value, &done) != TCL_OK) {
    return TCL_ERROR;
}
for (; !done ; Tcl_DictObjNext(&search, &key, &value, &done)) {
    /*
     * ملاحظة: strcmp() ليست طريقة جيدة لمقارنة القيم وتستخدم فقط للعرض.
     */
    if (!strcmp(Tcl_GetString(key), Tcl_GetString(value))) {
        break;
    }
}
Tcl_DictObjDone(&search);
Tcl_SetObjResult(interp, Tcl_NewBooleanObj(!done));
return TCL_OK;
