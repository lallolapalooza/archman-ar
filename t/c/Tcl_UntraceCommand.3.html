Tcl_UntraceCommand(3)

الحزمة: tcl

التراخيص: TCL

الاسم

Tcl_CommandTraceInfo, Tcl_TraceCommand, Tcl_UntraceCommand - مراقبة إعادة تسمية وحذف الأوامر

الملخص

#include <tcl.h>

ClientData

int
Tcl_TraceCommand(interp, cmdName, flags, proc, clientData)

void
Tcl_UntraceCommand(interp, cmdName, flags, proc, clientData)

إذا كانت غير فارغة (non-NULL)، تعطي القيمة الأخيرة التي تم إرجاعها بواسطة Tcl_CommandTraceInfo، لذا سيقوم هذا الاستدعاء بإرجاع معلومات حول التتبع التالي. إذا كانت فارغة (NULL)، فسيقوم هذا الاستدعاء بإرجاع معلومات حول أول تتبع.

الوصف

Tcl_TraceCommand يسمح لإجراء C بمراقبة العمليات التي تتم على أمر Tcl، بحيث يتم استدعاء إجراء C كلما تم إعادة تسمية الأمر أو حذفه. إذا تم إنشاء التتبع بنجاح، فترجع Tcl_TraceCommand القيمة TCL_OK. إذا حدث خطأ (مثل تحديد cmdName لأمر غير موجود)، فترجع TCL_ERROR ويتم ترك رسالة خطأ في نتيجة المفسر.

يشير الوسيط flags إلى Tcl_TraceCommand متى يجب استدعاء إجراء التتبع. يتكون من تركيبة منطقية لأي من القيم التالية:

TCL_TRACE_RENAME
استدعاء proc كلما تم إعادة تسمية الأمر.
TCL_TRACE_DELETE
استدعاء proc عند حذف الأمر.

كلما حدثت إحدى العمليات المحددة على الأمر، سيتم استدعاء proc. يجب أن تتطابق وسائطها ونتيجتها مع نوع Tcl_CommandTraceProc:

typedef void Tcl_CommandTraceProc(
        ClientData clientData,
        Tcl_Interp *interp,
        const char *oldName,
        const char *newName,
        int flags);

ستكون الوسائط clientData و interp هي نفسها التي تم تمريرها إلى Tcl_TraceCommand عند إنشاء التتبع. عادةً ما تشير ClientData إلى بنية بيانات خاصة بالتطبيق تصف ما يجب فعله عند استدعاء proc. OldName يعطي اسم الأمر الذي يتم إعادة تسميته، و newName يعطي الاسم الذي يتم إعادة تسمية الأمر إليه (أو NULL عند حذف الأمر.) Flags هو تركيبة منطقية من البتات التي قد توفر عدة قطع من المعلومات. سيتم تعيين أحد البتات TCL_TRACE_RENAME و TCL_TRACE_DELETE في flags للإشارة إلى العملية التي يتم تنفيذها على الأمر. سيتم تعيين البت TCL_TRACE_DESTROYED في flags إذا كان سيتم تدمير التتبع؛ قد تكون هذه المعلومات مفيدة لـ proc حتى تتمكن من تنظيف هياكل بياناتها الداخلية (انظر القسم TCL_TRACE_DESTROYED أدناه لمزيد من التفاصيل). نظرًا لأن حذف الأوامر يمكن أن يحدث كجزء من حذف المفسر الذي يحتوي عليها، يجب أن يكون proc حذرًا بشأن التحقق مما يمكن استدعاؤه باستخدام قيمة interp التي تم تمريرها. الإجراء Tcl_InterpDeleted هو أداة مهمة لهذا الغرض. عندما يرجع Tcl_InterpDeleted القيمة 1، لن يتمكن proc من استدعاء أي نصوص برمجية في interp. تقتصر وظيفة proc في هذه الظروف على تنظيف هياكل بياناته الخاصة.

يمكن استخدام Tcl_UntraceCommand لإزالة تتبع. إذا كان الأمر المحدد بواسطة interp و cmdName و flags لديه تتبع مضبوط باستخدام flags و proc و clientData، فسيتم إزالة التتبع المقابل. إذا لم يكن هناك مثل هذا التتبع، فإن استدعاء Tcl_UntraceCommand ليس له تأثير. نفس البتات صالحة لـ flags كما في استدعاءات Tcl_TraceCommand.

يمكن استخدام Tcl_CommandTraceInfo لاسترداد معلومات حول التتبعات المضبوطة على أمر معين. القيمة المرجعة من Tcl_CommandTraceInfo هي clientData المرتبطة بتتبع معين. يجب أن يكون التتبع على الأمر المحدد بواسطة الوسائط interp و cmdName و flags (لاحظ أن الأعلام يتم تجاهلها حاليًا؛ يجب تعيين flags إلى 0 للتوافق المستقبلي) وأن يكون إجراء التتبع الخاص به هو نفسه proc الوسيط. إذا كانت الوسيط prevClientData فارغة (NULL)، فإن القيمة المرجعة تتوافق مع أول تتبع مطابق (تم إنشاؤه مؤخرًا)، أو NULL إذا لم تكن هناك تتبعات مطابقة. إذا كانت الوسيط prevClientData غير فارغة (non-NULL)، فيجب أن تكون القيمة المرجعة من استدعاء سابق لـ Tcl_CommandTraceInfo. في هذه الحالة، ستتوافق القيمة المرجعة الجديدة مع التتبع التالي المطابق بعد التتبع الذي تتوافق clientData الخاصة به مع prevClientData، أو NULL إذا لم يتطابق أي تتبع مع prevClientData أو إذا لم يعد هناك تتبعات مطابقة بعده. هذه الآلية تجعل من الممكن المرور عبر جميع تتبعات أمر معين لها نفس proc.

استدعاء الأوامر أثناء التتبعات

أثناء تتبعات إعادة التسمية، يكون الأمر الذي يتم إعادة تسميته مرئيًا بالاسمين في نفس الوقت، ويظل الأمر موجودًا أثناء تتبعات الحذف، إلا إذا كان المفسر الذي يحتويه قيد الحذف. ومع ذلك، لا توجد آلية للإشارة إلى حدوث خطأ في إجراء التتبع، لذا يجب اتخاذ الحذر الشديد لضمان عدم فقدان الأخطاء بصمت.

تتبعات متعددة

من الممكن وجود تتبعات متعددة على نفس الأمر. عند حدوث ذلك، سيتم استدعاء جميع إجراءات التتبع عند كل وصول، بترتيب من الأكثر إنشاءً إلى الأقل إنشاءً. ستفشل محاولات حذف الأمر أثناء تتبع الحذف بصمت، حيث أن الأمر مجدول بالفعل للحذف على أي حال. إذا تمت إعادة تسمية الأمر الذي تتم إعادة تسميته بواسطة أحد تتبعات إعادة التسمية الخاصة به، فإن هذه الإعادة التسمية لها الأسبقية على تلك التي أدت إلى تشغيل التتبع ولن يتم إعادة تنفيذ مجموعة التتبعات؛ إذا أعادت عدة تتبعات تسمية الأمر، فإن إعادة التسمية الأخيرة لها الأسبقية.

علم TCL_TRACE_DESTROYED

في رد استدعاء الحذف إلى proc، يتم تعيين البت TCL_TRACE_DESTROYED في flags.

Tcl_UntraceCommand(3)

الاسم

Tcl_UntraceCommand - إلغاء تتبع أمر Tcl

المكتبة

libtcl8.6

الصيغة

#include <tcl.h>

void
Tcl_UntraceCommand(interp, objv, length, traceProc, clientData)
    Tcl_Interp *interp;
    const Tcl_Obj *const objv[];
    int length;
    Tcl_Trace traceProc;
    ClientData clientData;

الوصف

تقوم الدالة Tcl_UntraceCommand بإلغاء تسجيل إجراء تتبع معين تم تسجيله سابقًا باستخدام Tcl_TraceCommand. يجب استدعاء هذه الدالة فقط بواسطة إجراءات التتبع نفسها، وليس بواسطة أي جزء آخر من Tcl. يتم استخدامها عادةً داخل الإجراءات التي يتم استدعاؤها بواسطة Tcl_TraceCommand.

تأخذ Tcl_UntraceCommand الوسيطات التالية:

  • interp: مؤشر إلى مترجم Tcl الذي تم تسجيل التتبع فيه.
  • objv: مصفوفة من كائنات Tcl تمثل وسيطات الأمر.
  • length: عدد العناصر في objv.
  • traceProc: مؤشر إلى إجراء التتبع المراد إلغاء تسجيله.
  • clientData: بيانات العميل المرتبطة بإجراء التتبع.

القيود

يجب استدعاء Tcl_UntraceCommand فقط بواسطة إجراء تتبع تم تسجيله بواسطة Tcl_TraceCommand. قد يؤدي استدعاؤها من أماكن أخرى إلى سلوك غير محدد.

ملاحظات

عندما يقوم إجراء تتبع بإلغاء تسجيل نفسه، يجب عليه استخدام Tcl_UntraceCommand لتنظيف أي موارد مرتبطة بالتتبع.

مثال

يقوم هذا المثال بتوضيح كيفية إلغاء تسجيل إجراء تتبع:

/*
 * إجراء تتبع مثال.
 */
int
MyTraceProc(ClientData clientData, Tcl_Interp *interp, int level, const char *command, Tcl_Command commandToken, int objc, Tcl_Obj *const objv[])
{
    // ... إجراءات التتبع ...

    /* إلغاء تسجيل التتبع بعد تنفيذ العمل. */
    Tcl_UntraceCommand(interp, objv, objc, MyTraceProc, clientData);

    return TCL_OK;
}

/*
 * تسجيل إجراء التتبع.
 */
void
RegisterMyTrace(Tcl_Interp *interp)
{
    ClientData clientData = NULL; /* بيانات العميل الخاصة بك */
    Tcl_TraceCommand(interp, "mycommand", MyTraceProc, clientData, TCL_TRACE_DESTROYED_FLAG);
}
