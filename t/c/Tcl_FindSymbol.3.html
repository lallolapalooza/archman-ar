Tcl_FindSymbol(3)

الحزمة: tcl

التراخيص: TCL

Load(3) إجراءات مكتبة Tcl Load(3)

الاسم

Tcl_LoadFile, Tcl_FindSymbol - تحميل مكتبات ديناميكية مستقلة عن المنصة

ملخص

#include <tcl.h>

int
Tcl_LoadFile(interp, pathPtr, symbols, flags, procPtrs, loadHandlePtr)

void *
Tcl_FindSymbol(interp, loadHandle, symbol)

الوسائط

Tcl_Interp *interp (في)
المفسر المستخدم لتقديم رسائل الخطأ.
Tcl_Obj *pathPtr (في)
اسم الملف المراد تحميله. إذا كان اسماً واحداً، فسيتم استخدام مسار البحث عن المكتبة للبيئة الحالية لحله.
const char *const symbols[] (في)
مصفوفة من أسماء الرموز (عادةً وظائف) التي يجب حلها أثناء تحميل المكتبة، أو NULL إذا لم يكن هناك رموز ليتم حلها. إذا تم تقديم مصفوفة، يجب أن يكون الإدخال الأخير في المصفوفة NULL.
int flags (في)
يجب أن تكون القيمة عادةً 0، ولكن TCL_LOAD_GLOBAL أو TCL_LOAD_LAZY أو مزيج من هذين مسموح به أيضًا.
void *procPtrs (خارج)
يشير إلى مصفوفة ستحتوي على عناوين الوظائف الموصوفة في الوسيط symbols. يجب أن يكون NULL إذا لم يكن هناك رموز ليتم حلها.
Tcl_LoadHandle *loadHandlePtr (خارج)
يشير إلى متغير سيحتوي على مقبض للرمز المجرد الذي يصف المكتبة التي تم تحميلها.
Tcl_LoadHandle loadHandle (في)
مقبض مجرد يصف المكتبة للبحث عن رمز فيها.
const char *symbol (في)
اسم الرمز المراد البحث عنه.

الوصف

تقوم Tcl_LoadFile بتحميل ملف من نظام الملفات (بما في ذلك أي نظام ملفات افتراضي قد يكون مثبتًا) وتوفر مقبضًا له يمكن استخدامه في عمليات أخرى. توفر مصفوفة symbols، إذا لم تكن NULL، مجموعة من أسماء الرموز (عادةً وظائف) التي يجب حلها من المكتبة وسيتم تخزينها في المصفوفة المشار إليها بواسطة procPtrs. إذا لم يتم حل أي من الرموز، فسيفشل تحميل الملف مع ترك رسالة خطأ في المفسر (إذا كان غير NULL). نتيجة Tcl_LoadFile هي رمز خطأ Tcl قياسي. يمكن إلغاء تحميل المكتبة باستخدام Tcl_FSUnloadFile.

تقوم Tcl_FindSymbol بتحديد موقع رمز في مكتبة محملة وتعيده. إذا تعذر العثور على الرمز، فإنه يعيد NULL ويضبط رسالة خطأ في interp المحدد (إذا كان غير NULL). لاحظ أنه من غير الآمن استخدام هذه العملية على مقبض تم تمريره إلى Tcl_FSUnloadFile.

انظر أيضاً

Tcl_FSLoadFile(3), Tcl_FSUnloadFile(3), load(n), unload(n)
