الاسم

TCGETS, TCSETS, TCSETSW, TCSETSF - الحصول على سمات الطرفية وتعيينها

مكتبة C القياسية (libc، -lc)

الوصف

TCGETS
مكافئ لـ tcgetattr(fd, argp).
الحصول على الإعدادات الحالية للطرفية.
TCSETS
مكافئ لـ tcsetattr(fd, TCSANOW, argp).
تعيين الإعدادات الحالية للطرفية.
TCSETSW
مكافئ لـ tcsetattr(fd, TCSADRAIN, argp).
السماح بتفريغ مخزن الإخراج المؤقت، ثم تعيين الإعدادات الحالية للطرفية.
TCSETSF
مكافئ لـ tcsetattr(fd, TCSAFLUSH, argp).
السماح بتفريغ مخزن الإخراج المؤقت، ثم تجاهل أي إدخال معلق، ثم تعيين الإعدادات الحالية للطرفية.

الأوامر الأربعة التالية لـ ioctl مشابهة لـ TCGETS, TCSETS, TCSETSW, TCSETSF، إلا أنها تستخدم struct termios2 بدلاً من struct termios. إذا كان الحقل c_cflag يحتوي على العلم BOTHER، فسيتم تخزين معدل الباود في الحقلين c_ispeed و c_ospeed كقيم صحيحة. هذه الأوامر ioctl غير مدعومة على جميع المعماريات.

TCSETS2

TCSETSF2

الأوامر الأربعة التالية لـ ioctl مشابهة لـ TCGETS, TCSETS, TCSETSW, TCSETSF، إلا أنها تستخدم struct termio بدلاً من struct termios.

قيمة الإرجاع

عند النجاح، يتم إرجاع 0. عند حدوث خطأ، يتم إرجاع -1 ويتم تعيين errno للإشارة إلى الخطأ.

تنبيهات

struct termios من <asm/termbits.h> تختلف وغير متوافقة مع struct termios من <termios.h>. استدعاءات ioctl هذه تتطلب struct termios من <asm/termbits.h>.

أمثلة

الحصول على أو تعيين معدل الباود العشوائي على الطرفية.

/* SPDX-License-Identifier: GPL-2.0-or-later */
#include <asm/termbits.h>
#include <fcntl.h>
#include <stdio.h>
#include <stdlib.h>
#include <sys/ioctl.h>
#include <unistd.h>
int
main(int argc, char *argv[])
{
#if !defined BOTHER
    fprintf(stderr, "BOTHER غير معروف\n");
    /* قد يسقط البرنامج في TCGETS/TCSETS مع ثوابت Bnnn */
    exit(EXIT_FAILURE);
#else
    /* تعريف هيكل tio، يعتمد نوعه على ioctl المدعوم */
# if defined TCGETS2
    struct termios2 tio;
# else
    struct termios tio;
# endif
    int fd, rc;
    if (argc != 2 && argc != 3 && argc != 4) {
        fprintf(stderr, "الاستخدام: %s جهاز [مخرج [مدخل]]\n", argv[0]);
        exit(EXIT_FAILURE);
    }
    fd = open(argv[1], O_RDWR | O_NONBLOCK | O_NOCTTY);
    if (fd < 0) {
        perror("open");
        exit(EXIT_FAILURE);
    }
    /* الحصول على القيم الحالية للطرفية عبر ioctl المدعوم */
# if defined TCGETS2
    rc = ioctl(fd, TCGETS2, &tio);
# else
    rc = ioctl(fd, TCGETS, &tio);
# endif
    if (rc) {
        perror("TCGETS");
        close(fd);
        exit(EXIT_FAILURE);
    }
    /* تغيير معدل الباود عندما يكون هناك المزيد من الوسائط */
    if (argc == 3 || argc == 4) {
        /* مسح معدل الإخراج الحالي وملء قيمة جديدة */
        tio.c_cflag &= ~CBAUD;
        tio.c_cflag |= BOTHER;
        tio.c_ospeed = atoi(argv[2]);
        /* مسح معدل الإدخال الحالي وملء قيمة جديدة */
        tio.c_cflag &= ~(CBAUD << IBSHIFT);
        tio.c_cflag |= BOTHER << IBSHIFT;
        /* عندما لا تكون الوسيطة 4 موجودة، استخدم معدل الإخراج مرة أخرى */
        tio.c_ispeed = (argc == 4) ? atoi(argv[3]) : atoi(argv[2]);
        /* تعيين قيم الطرفية عبر ioctl المدعوم */
# if defined TCSETS2
        rc = ioctl(fd, TCSETS2, &tio);
# else
        rc = ioctl(fd, TCSETS, &tio);
# endif
        if (rc) {
            perror("TCSETS");
            close(fd);
            exit(EXIT_FAILURE);
        }
        /* والحصول على القيم التي تم تعيينها بالفعل */
# if defined TCGETS2
        rc = ioctl(fd, TCGETS2, &tio);
# else
        rc = ioctl(fd, TCGETS, &tio);
# endif
        if (rc) {
            perror("TCGETS");
            close(fd);
            exit(EXIT_FAILURE);
        }
    }
    close(fd);
    printf("معدل الإخراج: %u\n", tio.c_ospeed);
    printf("معدل الإدخال: %u\n", tio.c_ispeed);
    exit(EXIT_SUCCESS);
#endif
}
