المعاملات

const Tcl_Filesystem *fsPtr (in)
يشير إلى بنية تحتوي على عناوين الإجراءات التي يمكن استدعاؤها لأداء عمليات نظام الملفات المختلفة.
Tcl_Obj *pathPtr (in)
يستخدم المسار الذي تمثله هذه القيمة لعملية معينة. إذا لم يكن للقيمة تمثيل داخلي للمسار بالفعل، فسيتم تحويلها لامتلاك واحد.
Tcl_Obj *srcPathPtr (in)
كما هو الحال مع pathPtr ، ولكن يستخدم للملف المصدر لعملية نسخ أو إعادة تسمية.
Tcl_Obj *destPathPtr (in)
كما هو الحال مع pathPtr ، ولكن يستخدم لاسم الملف الوجهة لعملية نسخ أو إعادة تسمية.
int recursive (in)
ما إذا كان سيتم إزالة الدلائل الفرعية ومحتوياتها أيضًا.
const char *encodingName (in)
ترميز البيانات المخزنة في الملف المحدد بواسطة pathPtr والذي سيتم تقييمه.
const char *pattern (in)
فقط الملفات أو الدلائل التي تطابق هذا النمط سيتم إرجاعها.
Tcl_GlobTypeData *types (in)
فقط الملفات أو الدلائل التي تطابق أوصاف النوع الموجودة في هذه البنية سيتم إرجاعها. قد تكون هذه المعاملة NULL.
Tcl_Interp *interp (in)
المترجم المراد استخدامه إما للنتائج، أو التقييم، أو الإبلاغ عن رسائل الخطأ.
void *clientData (in)
الوصف الأصلي لقيمة المسار المراد إنشاؤها.
Tcl_Obj *firstPtr (in)
القيمة الأولى من قيمتي مسار للمقارنة. قد يتم تحويل القيمة إلى نوع path.
Tcl_Obj *secondPtr (in)
القيمة الثانية من قيمتي مسار للمقارنة. قد يتم تحويل القيمة إلى نوع path.
Tcl_Obj *listObj (in)
قائمة عناصر المسار للعمل عليها بعملية join.
int elements (in)
عدد العناصر في listObj التي يجب ضمها معًا. إذا كان سالبًا، فسيتم ضم جميع العناصر.
Tcl_Obj **errorPtr (out)
في حالة حدوث خطأ، يتم ملؤه بقيمة تحتوي على اسم الملف الذي تسبب في خطأ في عمليات النسخ/إعادة التسمية المختلفة.
int index (in)
فهرس السمة المعنية.
Tcl_Obj *objPtr (in)
القيمة المراد تعيينها في العملية.
Tcl_Obj **objPtrRef (out)
يتم ملؤه بقيمة تحتوي على نتيجة العملية.
Tcl_Obj *resultPtr (out)
قيمة مخصصة مسبقًا لتخزين (باستخدام Tcl_ListObjAppendElement) قائمة الملفات أو الدلائل التي تم مطابقتها بنجاح.
int mode (in)
قناع يتكون من واحد أو أكثر من R_OK، W_OK، X_OK و F_OK. تطلب R_OK، W_OK و X_OK التحقق مما إذا كان الملف موجودًا ولديه أذونات قراءة، كتابة وتنفيذ، على التوالي. تطلب F_OK فقط التحقق من وجود الملف.
Tcl_StatBuf *statPtr (out)
البنية التي تحتوي على نتيجة عملية stat أو lstat.
const char *sym1 (in)
اسم إجراء للبحث عنه في جدول رموز الملف
const char *sym2 (in)
اسم إجراء للبحث عنه في جدول رموز الملف
Tcl_PackageInitProc **proc1Ptr (out)
يتم ملؤه بوظيفة التهيئة لهذا الكود.
Tcl_PackageInitProc **proc2Ptr (out)
يتم ملؤه بوظيفة التهيئة الآمنة لهذا الكود.
void **clientDataPtr (out)
يتم ملؤه بقيمة clientData ليتم تمريرها إلى وظيفة إلغاء تحميل هذا الكود عند استدعائها.
Tcl_LoadHandle *loadHandlePtr (out)
يتم ملؤه برمز تجريدي يمثل الملف المحمل.
Tcl_FSUnloadFileProc **unloadProcPtr (out)
يتم ملؤه بالوظيفة المراد استخدامها لتحميل هذا الجزء من الكود.
Tcl_LoadHandle loadHandle (in)
مقبض المكتبة المحملة المراد تحميلها.
utimbuf *tval (in)
يتم قراءة أوقات الوصول والتعديل في هذه البنية واستخدامها لتعيين تلك القيم لملف معين.
const char *modeString (in)
يحدد كيفية الوصول إلى الملف. قد يأخذ أيًا من القيم المسموح بها للمعامل mode لأمر Tcl open.
int permissions (in)
علامات إذن بنمط POSIX مثل 0644. إذا تم إنشاء ملف جديد، فسيتم تعيين هذه الأذونات على الإنشاء

واجهة برمجة تطبيقات نظام الملفات الافتراضي

يوفر نظام الملفات هيكل Tcl_Filesystem الذي يحتوي على مؤشرات إلى وظائف تنفذ العمليات المختلفة على نظام الملفات؛ يتم استدعاء هذه العمليات حسب الحاجة من قبل القدرة العامة، والتي تحدث عادةً من خلال الوظائف المذكورة أعلاه.

يتم التعامل مع هياكل Tcl_Filesystem بواسطة الطرق التالية.

Tcl_FSRegister يأخذ مؤشرًا إلى هيكل نظام الملفات وبيانات اختيارية مرتبطة بهذا الهيكل. عند استدعاء هذه الوظيفة، سيربط Tcl نظام الملفات بقائمة أنظمة الملفات المعروفة ويجعله يعمل على الفور. لا يتحقق Tcl مما إذا كان نفس نظام الملفات قد تم تسجيله عدة مرات (وهو بشكل عام ليس فكرة جيدة). سيتم إرجاع TCL_OK.

Tcl_FSUnregister يزيل الهيكل المحدد من قائمة أنظمة الملفات المعروفة، إذا كان مسجلاً، ويعيد TCL_OK. إذا لم يكن نظام الملفات مسجلاً حاليًا، فسيتم إرجاع TCL_ERROR.

Tcl_FSData سيعيد بيانات العميل المرتبطة بنظام الملفات المحدد، إذا كان مسجلاً. وإلا، سيعيد NULL.

Tcl_FSMountsChanged يستخدم لإبلاغ نواة Tcl بأن نقاط التحميل لنظام الملفات المحدد (المسجل بالفعل) قد تغيرت، وأن التمثيلات المخزنة مؤقتًا للملفات قد تكون غير كافية الآن.

باستثناء الحقول الثلاثة الأولى في هذا الهيكل التي تحتوي على عناصر بيانات بسيطة، تحتوي جميع الحقول الأخرى على عناوين وظائف يتم استدعاؤها بواسطة القدرة العامة لأداء مجموعة واسعة من الإجراءات المتعلقة بنظام الملفات.

تنقسم الوظائف المذكورة في هذا الهيكل إلى ثلاث فئات: وظائف البنية التحتية (ما وراءها يجب تنفيذ جميعها تقريبًا)، وظائف تشغيلية (يجب تنفيذها إذا تم توفير نظام ملفات كامل)، ووظائف الكفاءة (يجب تنفيذها فقط إذا كان يمكن إجراؤها بكفاءة، أو إذا كان لها آثار جانبية ضرورية لنظام الملفات؛ يحتوي Tcl على محاكاة أبطأ يمكن أن يلجأ إليها). من المهم ملاحظة أنه في الإصدار الحالي من Tcl، يتم استخدام معظم هذه الاستدعاءات الراجعة فقط للتعامل مع الأوامر التي تبدأ في Tcl، وليس في C. هذا يعني أنه إذا تم إصدار أمر إعادة تسمية ملف في Tcl، ولم تقم أنظمة الملفات ذات الصلة بتنفيذ Tcl_FSRenameFileProc الخاص بها، فإن نواة Tcl ستلجأ إلى مزيج من وظائف نظام الملفات الأخرى (ستستخدم Tcl_FSCopyFileProc متبوعًا بـ Tcl_FSDeleteFileProc، وإذا لم يتم تنفيذ Tcl_FSCopyFileProc، فهناك استدعاءات راجعة إضافية). ومع ذلك، إذا تم إصدار أمر Tcl_FSRenameFileProc على مستوى C، فلن تحدث هذه الاستدعاءات الراجعة. هذا صحيح باستثناء العناصر الأربعة الأخيرة في جدول نظام الملفات (lstat، load، getcwd، و chdir)، والتي يتم فيها إجراء الاستدعاءات الراجعة بالفعل على مستوى C.

ستأخذ أي وظائف تأخذ أسماء الدلائل بتنسيق Tcl_Obj هذه الأسماء بتنسيق UTF-8. تم تصميم واجهة برمجة تطبيقات البنية التحتية لنظام الملفات لدعم التحويل الفعال والمخزن مؤقتًا لهذه المسارات UTF-8 إلى تمثيلات أصلية.
