#include <tcl.h>

Tcl_Command
Tcl_NRCreateCommand(interp, cmdName, proc, nreProc, clientData,

Tcl_NREvalObjv(interp, objc, objv, flags)

Tcl_NRCmdSwap(interp, cmd, objc, objv, flags)

الوسائط

Tcl_Interp *interp (مدخل)
المفسر المعني.
const char *cmdName (مدخل)
اسم الأمر المراد إنشاؤه.
Tcl_ObjCmdProc *proc (مدخل)
يتم استدعاؤه لتقييم أمر. غالبًا ما يكون مجرد غلاف صغير يستخدم Tcl_NRCallObjProc لاستدعاء nreProc باستخدام قفزة جديدة. يتصرف بنفس الطريقة التي يتصرف بها الوسيط proc للـ Tcl_CreateObjCommand(3) (راجع).
Tcl_ObjCmdProc *nreProc (مدخل)
يتم استدعاؤه بدلاً من proc عند استخدام قفزة بالفعل.
ClientData clientData (مدخل)
قيمة اعتباطية بكلمة واحدة يتم تمريرها إلى proc، nreProc، deleteProc و objProc.
Tcl_CmdDeleteProc *deleteProc (مدخل/مخرج)
يتم استدعاؤه قبل حذف cmdName من المفسر، مما يسمح بالتنظيف الخاص بالأمر. قد يكون NULL.
int objc (مدخل)
عدد العناصر في objv.
Tcl_Obj **objv (مدخل)
الكلمات في الأمر.
Tcl_Obj *objPtr (مدخل)
نص برمجي أو تعبير لتقييمه.
int flags (مدخل)
كما هو موضح لـ Tcl_EvalObjv.
Tcl_Command cmd (مدخل)
رمز مميز لاستخدامه بدلاً من واحد مشتق من الكلمة الأولى لـ objv لتقييم أمر.
Tcl_Obj *resultPtr (مخرج)
مؤشر إلى Tcl_Obj غير مشترك حيث يتم تخزين نتيجة التقييم إذا كان رمز الإرجاع هو TCL_OK.
Tcl_NRPostProc *postProcPtr (مدخل)
دالة للدفع.
ClientData data0 (مدخل)
ClientData data1 (مدخل)
ClientData data2 (مدخل)
ClientData data3 (مدخل)
data0 إلى data3 هي أربع قيم بكلمة واحدة سيتم تمريرها إلى الدالة المعينة بواسطة postProcPtr عند استدعائها.

الوصف

توفر هذه الوظائف واجهة لمكدس الوظائف الذي يتكرر المفسر من خلاله لتقييم الأوامر. يتم تنفيذ الروتين وراء الأمر بواسطة وظيفة أولية وأي وظائف إضافية يدفعها الروتين إلى المكدس أثناء تقدمه. يقوم المفسر نفسه بدفع وظائف إلى المكدس للاستجابة لنهاية الروتين وممارسة أشكال أخرى من التحكم مثل التبديل بين المكدسات قيد التقدم وتقييم النصوص البرمجية الأخرى في مستويات إضافية دون إضافة إطارات إلى مكدس C. لتنفيذ روتين، يتم استدعاء الوظيفة الأولية للروتين ثم تقوم قطعة صغيرة من التعليمات البرمجية تسمى قفزة بسحب الوظائف بشكل متكرر من المكدس واستدعائها، باستخدام قيمة الاستدعاء الأخير كقيمة للروتين.

Tcl_NRCallObjProc يستدعي nreProc باستخدام قفزة جديدة.

Tcl_NRCreateCommand، بديل لـ Tcl_CreateObjCommand، يحل cmdName، الذي قد يحتوي على مؤهلات مساحة الاسم، نسبة إلى مساحة الاسم الحالية، وينشئ أمرًا بهذا الاسم، ويعيد رمزًا مميزًا للأمر يمكن استخدامه في استدعاءات لاحقة لـ Tcl_GetCommandName. باستثناء بعض الحالات المذكورة أدناه، يتم حذف أي أمر موجود بنفس الاسم أولاً. إذا كان interp قيد الحذف، فلن تنشئ Tcl_NRCreateCommand أي أمر، ولن تحذف أي أمر، وتعيد NULL.

Tcl_NREvalObj يدفع وظيفة تشبه Tcl_EvalObjEx ولكنها لا تستهلك مساحة على مكدس C.

Tcl_NREvalObjv يدفع وظيفة تشبه Tcl_EvalObjv ولكنها لا تستهلك مساحة على مكدس C.

Tcl_NRCmdSwap تشبه Tcl_NREvalObjv، ولكنها تستخدم cmd، رمز مميز تم إرجاعه سابقًا بواسطة Tcl_CreateObjCommand أو Tcl_GetCommandFromObj، بدلاً من حل الكلمة الأولى لـ objv.

Tcl_NRExprObj يدفع وظيفة تقيم objPtr كتعبير بنفس الطريقة التي يعمل بها Tcl_ExprObj ولكن دون استهلاك مساحة على مكدس C.

جميع الوظائف تعيد TCL_OK إذا تم جدولة تقييم النص البرمجي أو الأمر أو التعبير بنجاح. خلاف ذلك (على سبيل المثال إذا تعذر حل اسم الأمر)، فإنها تعيد TCL_ERROR وتخزن رسالة كنتيجة للمفسر.

Tcl_NRAddCallback يدفع postProcPtr. التوقيع لـ Tcl_NRPostProc هو:

typedef int
Tcl_NRPostProc(
        ClientData data[],
        Tcl_Interp *interp,
        int result);

data هو مؤشر إلى مصفوفة تحتوي على data0 إلى data3. result هي القيمة التي تم إرجاعها بواسطة الوظيفة السابقة التي تنفذ جزءًا من الروتين.

مثال

يستخدم الأمر التالي Tcl_EvalObjEx، الذي يستهلك مساحة على مكدس C، لتقييم نص برمجي:

int
TheCmdOldObjProc(
    ClientData clientData,
    Tcl_Interp *interp,
    int objc,
    Tcl_Obj *const objv[])
{
    int result;
    Tcl_Obj *objPtr;
    ... preparation ...
    result = Tcl_EvalObjEx(interp, objPtr, 0);
    ... postprocessing ...
    return result;
}
Tcl_CreateObjCommand(interp, "theCommand",
        TheCmdOldObjProc, clientData, TheCmdDeleteProc);

لتجنب استهلاك مساحة على مكدس C، يتم إعادة تسمية TheCmdOldObjProc إلى TheCmdNRObjProc ويتم تقسيم خطوة ما بعد المعالجة إلى وظيفة منفصلة، TheCmdPostProc، والتي يتم دفعها إلى مكدس الوظائف. يتم استبدال Tcl_EvalObjEx بـ Tcl_NREvalObj، الذي يستخدم قفزة بدلاً من استهلاك مساحة على مكدس C. نسخة جديدة من TheCmdOldObjProc هي مجرد غلاف يستخدم Tcl_NRCallObjProc لاستدعاء TheCmdNRObjProc:

int
TheCmdOldObjProc(
    ClientData clientData,
    Tcl_Interp *interp,
    int objc,
    Tcl_Obj *const objv[])
{
    return Tcl_NRCallObjProc(interp, TheCmdNRObjProc,
            clientData, objc, objv);
}
int
TheCmdNRObjProc
    ClientData clientData,
    Tcl_Interp *interp,
    int objc,
    Tcl_Obj *const objv[])
{
    Tcl_Obj *objPtr;
    ... preparation ...
    Tcl_NRAddCallback(interp, TheCmdPostProc,
            data0, data1, data2, data3);
    /* data0 .. data3 هي حتى أربعة عناصر بكلمة واحدة لتمريرها
     * إلى إجراء ما بعد المعالجة */
    return Tcl_NREvalObj(interp, objPtr, 0);
}
int
TheCmdNRPostProc(
    ClientData data[],
    Tcl_Interp *interp,
    int result)
{
    /* data[0] .. data[3] هي الكلمات الأربع للبيانات
     * التي تم تمريرها إلى Tcl_NRAddCallback */
    ... postprocessing ...
    return result;
}

يمكن لأي وظيفة تشكل روتينًا أن تدفع وظائف أخرى، مما يجعل من الممكن تنفيذ هياكل الحلقات والتسلسل باستخدام مكدس الوظائف.

Tcl_NRExprObj(3)

NAME

Tcl_NRExprObj - evaluate a numeric expression

SYNOPSIS 

#include <tcl.h>

int
Tcl_NRExprObj(interp, string, resultPtr)
    Tcl_Interp *interp;
    const char *string;
    Tcl_Obj **resultPtr;

ARGUMENTS

interp (in)
TCL interpreter to use for evaluation.
string (in)
String containing the expression to evaluate.
resultPtr (out)
Pointer to where the resulting Tcl_Obj will be stored. The result is a 32-bit integer.

DESCRIPTION

The function Tcl_NRExprObj evaluates the expression given by string and stores the result in resultPtr. The result is a 32-bit integer. It returns TCL_OK on success, or TCL_ERROR if an error occurred during evaluation. If TCL_ERROR is returned, the interpreter will contain an error message.

EXAMPLE 

Tcl_Interp *interp = Tcl_CreateInterp();
Tcl_Obj *result = NULL;
if (Tcl_NRExprObj(interp, "1 + 2 * 3", &result) == TCL_OK) {
    // result contains the integer 7
    printf("%s\n", Tcl_GetString(result));
}
Tcl_FreeObj(result);
Tcl_DeleteInterp(interp);

KEYWORDS

expression, evaluation, numeric, integer
