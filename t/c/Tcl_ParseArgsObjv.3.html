الحزمة: tcl

التراخيص: TCL

Tcl_ParseArgsObjv(3) إجراءات مكتبة Tcl Tcl_ParseArgsObjv(3)

الاسم

Tcl_ParseArgsObjv - تحليل الوسائط وفقًا لجدول وصفي

الصيغة

#include <tcl.h>

int
Tcl_ParseArgsObjv(interp, argTable, objcPtr, objv, remObjv)

الوسائط

Tcl_Interp *interp (out)
المكان الذي يتم فيه تخزين رسائل الخطأ.
const Tcl_ArgvInfo *argTable (in)
مؤشر إلى مجموعة من الأوصاف للوسائط.
int *objcPtr (in/out)
مؤشر إلى متغير يحتوي على عدد الوسائط في objv. سيتم تعديله لتخزين عدد الوسائط المتبقية في قائمة الوسائط غير المعالجة المخزنة في remObjv.
Tcl_Obj *const *objv (in)
مصفوفة الوسائط المطلوب تحليلها.
Tcl_Obj ***remObjv (out)
مؤشر إلى متغير سيحتوي على مصفوفة الوسائط غير المعالجة. إذا لم تكن الوسائط غير المعالجة مطلوبة، فيجب أن يكون المعامل remObjv هو NULL. إذا تم تحديث objcPtr إلى قيمة غير صفرية، فيجب تحرير المصفوفة التي تم إرجاعها عبر هذا باستخدام ckfree.

الوصف

توفر الدالة Tcl_ParseArgsObjv نظامًا لتحليل قوائم الوسائط بالصيغة “-someName someValue ...”. توجد قوائم الوسائط هذه عادةً في وسائط البرنامج وفي وسائط أوامر Tcl. يفترض محلل الوسائط هذا أن ترتيب الوسائط غير مهم، باستثناء نسخ الخيارات اللاحقة التي تستبدل النسخ السابقة.

يتم وصف مصفوفة الوسائط المطلوب تحليلها بواسطة المعامل argTable. يشير هذا إلى مجموعة من هياكل وصف الوسائط التي تنتهي بسجل يحتوي على حقل type معين على TCL_ARGV_END. للتسهيل، يتم توفير السجلات النموذجية التالية:

TCL_ARGV_AUTO_HELP
يمكّن معالج الوسائط من توفير المساعدة عند تمرير الوسيط -help.
TCL_ARGV_AUTO_REST
يشير إلى أنه يجب تجاهل الوسائط بعد --.
TCL_ARGV_TABLE_END
يحدد نهاية مجموعة أوصاف الوسائط.

إدخالات وصف الوسائط

يجب أن يكون كل إدخال في مجموعة وصف الوسائط بنية من النوع Tcl_ArgvInfo. يتم تعريف البنية على النحو التالي:

typedef struct {
    int type;
    const char *keyStr;
    void *srcPtr;
    void *dstPtr;
    const char *helpStr;
    ClientData clientData;
} Tcl_ArgvInfo;

يحتوي الحقل keyStr على اسم الخيار؛ حسب العرف، يبدأ بحرف -. تصف الحقول type و srcPtr و dstPtr و clientData تفسير قيمة الوسيط، كما هو موضح أدناه. يوفر الحقل helpStr نصًا يستخدم لمساعدة المستخدم عند الطلب.

كما ذكرنا سابقًا، يتم استخدام الحقل type لوصف تفسير قيمة الوسيط. القيم التالية مقبولة لـ type:

TCL_ARGV_CONSTANT
لا يحتوي الوسيط على قيمة لاحقة. إذا كان هذا الوسيط موجودًا، يتم نسخ القيمة (المكاملة) في الحقل srcPtr إلى المتغير المشار إليه بواسطة dstPtr. يتم تجاهل الحقل clientData.
TCL_ARGV_END
تحدد هذه القيمة نهاية جميع أوصاف الخيارات في المجموعة. يتم تجاهل جميع الحقول الأخرى.
TCL_ARGV_FLOAT
يأخذ هذا الوسيط قيمة لاحقة من نوع النقطة العائمة. يتم تخزين القيمة (بعد تحليلها بواسطة Tcl_GetDoubleFromObj) كقيمة مزدوجة الدقة في المتغير المشار إليه بواسطة dstPtr. يتم تجاهل حقلي srcPtr و clientData.
TCL_ARGV_FUNC
يأخذ هذا الوسيط اختياريًا قيمة لاحقة؛ يقرر معالج الوسائط الذي يتم تمريره عبر srcPtr ما إذا كان من الضروري أخذ الوسيط التالي. سيكون للدالة التوقيع التالي:
typedef int (Tcl_ArgvFuncProc)(
        ClientData clientData,
        Tcl_Obj *objPtr,
        void *dstPtr);

النتيجة هي قيمة منطقية تشير إلى ما إذا كان قد تم استهلاك الوسيط التالي. clientData هي قيمة إدخال الجدول، objPtr هي القيمة التي تمثل الوسيط التالي أو NULL إذا لم تكن هناك وسائط تالية. المعامل dstPtr للدالة Tcl_ArgvFuncProc هو المكان الذي يتم فيه كتابة القيمة المحللة.

TCL_ARGV_GENFUNC
يأخذ هذا الوسيط صفرًا أو أكثر من الوسائط اللاحقة؛ يعيد معالج الوسائط الذي يتم تمريره عبر srcPtr عددها (أو عددًا سالبًا للإشارة إلى خطأ، وفي هذه الحالة يجب أيضًا تعيين نتيجة المفسر). سيكون للدالة التوقيع التالي:
typedef int (Tcl_ArgvGenFuncProc)(
        ClientData clientData,
        Tcl_Interp *interp,
        int objc,
        Tcl_Obj *const *objv,
        void *dstPtr);

clientData هي قيمة إدخال الجدول، interp هو المكان الذي سيتم فيه تخزين أي رسائل خطأ، objc و objv يصفان مصفوفة لجميع الوسائط المتبقية، والمعامل dstPtr للدالة Tcl_ArgvGenFuncProc هو المكان الذي يتم فيه كتابة القيمة (أو القيم) المحللة.

TCL_ARGV_HELP
لا يأخذ هذا الوسيط الخاص أي قيمة لاحقة، ولكنه يتسبب في أن تقوم Tcl_ParseArgsObjv بإنشاء رسالة خطأ تصف الوسائط المدعومة. يتم تجاهل جميع الحقول باستثناء الحقل helpStr.
TCL_ARGV_INT
يأخذ هذا الوسيط قيمة لاحقة من نوع عدد صحيح. يتم تخزين القيمة (بعد تحليلها بواسطة Tcl_GetIntFromObj) كعدد صحيح في المتغير المشار إليه بواسطة dstPtr. يتم تجاهل الحقل srcPtr.
TCL_ARGV_REST
لا يأخذ هذا الوسيط الخاص أي قيمة لاحقة، ولكنه يحدد جميع الوسائط اللاحقة لعدم معالجتها. يتم تجاهل الحقول srcPtr و dstPtr و clientData.
TCL_ARGV_STRING
يأخذ هذا الوسيط قيمة لاحقة من نوع سلسلة نصية. يتم تخزين مؤشر إلى السلسلة في dstPtr؛ سيكون محتوى السلسلة مرتبطًا بعمر تمثيل قيمة السلسلة للوسيط. لذلك، يجب نسخه إذا كان يحتاج إلى الاحتفاظ به. يتم تجاهل حقلي srcPtr و clientData.
