Tcl_NRCreateCommand(3)

الحزمة: tcl

التراخيص: TCL

NRE(3) إجراءات مكتبة Tcl NRE(3)

الاسم

Tcl_NRCreateCommand, Tcl_NRCallObjProc, Tcl_NREvalObj, Tcl_NREvalObjv, Tcl_NRCmdSwap, Tcl_NRExprObj, Tcl_NRAddCallback - تقييم غير متكرر (بدون مكدس) لسكريبتات Tcl.

الملخص

#include <tcl.h>

Tcl_Command
Tcl_NRCreateCommand(interp, cmdName, proc, nreProc, clientData,
                    deleteProc)

int
Tcl_NRCallObjProc(interp, nreProc, clientData, objc, objv)

int
Tcl_NREvalObj(interp, objPtr, flags)

int
Tcl_NREvalObjv(interp, objc, objv, flags)

int
Tcl_NRCmdSwap(interp, cmd, objc, objv, flags)

int
Tcl_NRExprObj(interp, objPtr, resultPtr)

void
Tcl_NRAddCallback(interp, postProcPtr, data0, data1, data2, data3)

المعاملات

Tcl_Interp *interp (in)
المفسر ذو الصلة.
const char *cmdName (in)
اسم الأمر المراد إنشاؤه.
Tcl_ObjCmdProc *proc (in)
يتم استدعاؤه لتقييم أمر. عادة ما يكون مجرد غلاف صغير يستخدم Tcl_NRCallObjProc لاستدعاء nreProc بآلية قفز جديدة. يتصرف بنفس طريقة المعامل proc لـ Tcl_CreateObjCommand(3) (انظر).
Tcl_ObjCmdProc *nreProc (in)
يتم استدعاؤه بدلاً من proc عندما تكون آلية القفز قيد الاستخدام.
ClientData clientData (in)
قيمة عشوائية من بايت واحد يتم تمريرها إلى proc و nreProc و deleteProc و objProc.
Tcl_CmdDeleteProc *deleteProc (in/out)
يتم استدعاؤه قبل حذف cmdName من المفسر، مما يسمح بالتنظيف الخاص بالأمر. يمكن أن يكون NULL.
int objc (in)
عدد العناصر في objv.
Tcl_Obj **objv (in)
الكلمات في الأمر.
Tcl_Obj *objPtr (in)
سكريبت أو تعبير لتقييمه.
int flags (in)
كما هو موضح لـ Tcl_EvalObjv.
Tcl_Command cmd (in)
رمز مميز ليستخدم بدلاً من واحد مشتق من العنصر الأول من objv لتقييم أمر.
Tcl_Obj *resultPtr (out)
كائن Tcl غير مشترك حيث يتم تخزين نتيجة التقييم إذا كان رمز الإرجاع هو TCL_OK.
Tcl_NRPostProc *postProcPtr (in)
وظيفة يتم دفعها.
ClientData data0 (in)
...
ClientData data1 (in)
...
ClientData data2 (in)
...
ClientData data3 (in)
data0 إلى data3 هي أربع قيم بايت واحد يتم تمريرها إلى الوظيفة المعينة بواسطة postProcPtr عند استدعائها.

الوصف

توفر هذه الوظائف واجهة لمكدس الوظائف الذي يتكرر المفسر لتقييم الأوامر. يتم تنفيذ الوظيفة خلف الأمر بواسطة وظيفة تمهيدية وأي وظائف أخرى يتم دفعها إلى المكدس أثناء العملية. يدفع المفسر نفسه وظائف إلى المكدس للتفاعل مع نهاية الروتين وللتحكم في أنواع أخرى، مثل التبديل بين أنماط المكدس قيد التشغيل وتقييم سكريبتات إضافية دون إضافة إطارات في مكدس C. لتشغيل روتين، يتم استدعاء الوظيفة التمهيدية للروتين أولاً، ثم يتم استدعاء قطعة صغيرة من التعليمات البرمجية تسمى آلية القفز، والتي تتخذ بشكل متكرر وظائف من المكدس وتستدعيها، باستخدام قيمة الاستدعاء الأخير كقيمة للروتين.

Tcl_NRCallObjProc تستدعي nreProc باستخدام آلية قفز جديدة.

Tcl_NRCreateCommand، بديل لـ Tcl_CreateObjCommand، يقوم بحل cmdName، والذي قد يحتوي على بادئات مساحة اسم، فيما يتعلق بالمساحة الحالية، وينشئ أمرًا بهذا الاسم، ويعيد رمزًا مميزًا للأمر يمكن استخدامه في استدعاءات Tcl_GetCommandName اللاحقة. باستثناء بعض الحالات المذكورة أدناه، يتم أولاً إزالة أي أمر موجود بنفس الاسم. إذا كان interp قيد الحذف، فلن ينشئ Tcl_NRCreateCommand أي أوامر، ولن يحذف أي أوامر، ويعيد NULL.

Tcl_NREvalObj يدفع وظيفة مشابهة لـ Tcl_EvalObjEx ولكنها لا تستهلك مساحة في مكدس C.

Tcl_NREvalObjv يدفع وظيفة مشابهة لـ Tcl_EvalObjv ولكنها لا تستهلك مساحة في مكدس C.

Tcl_NRCmdSwap مشابه لـ Tcl_NREvalObjv، ولكنه يستخدم cmd، وهو رمز مميز تم إرجاعه مسبقًا بواسطة Tcl_CreateObjCommand أو Tcl_GetCommandFromObj، بدلاً من حل العنصر الأول من objv.

Tcl_NRExprObj يدفع وظيفة تقيم objPtr كتعبير بنفس طريقة Tcl_ExprObj، ولكن دون استهلاك مساحة في مكدس C.

جميع الوظائف تعيد TCL_OK إذا تمت جدولة تقييم السكريبت أو الأمر أو التعبير بنجاح. بخلاف ذلك (على سبيل المثال، إذا لم يكن من الممكن حل اسم الأمر)، فإنها تعيد TCL_ERROR وتخزن رسالة كنتيجة للمفسر.

Tcl_NRAddCallback يدفع postProcPtr. توقيع Tcl_NRPostProc هو:

typedef int
Tcl_NRPostProc(
        ClientData data[],
        Tcl_Interp *interp,
        int result);

data هو مؤشر إلى مصفوفة تحتوي على data0 إلى data3. result هي القيمة التي تم إرجاعها بواسطة الوظيفة السابقة التي تنفذ جزءًا من الروتين.

مثال

يستخدم البيان التالي Tcl_EvalObjEx، الذي يستهلك مساحة في مكدس C، لتقييم سكريبت:

int
TheCmdOldObjProc(
    ClientData clientData,
    Tcl_Interp *interp,
    int objc,
    Tcl_Obj *const objv[])
{
    int result;
    Tcl_Obj *objPtr;
    ... تحضير ...
    result = Tcl_EvalObjEx(interp, objPtr, 0);
    ... معالجة لاحقة ...
    return result;
}
Tcl_CreateObjCommand(interp, "theCommand",
        TheCmdOldObjProc, clientData, TheCmdDeleteProc);

لتجنب استهلاك مساحة في مكدس C، تتم إعادة تسمية TheCmdOldObjProc إلى TheCmdNRObjProc ويتم تقسيم خطوة المعالجة اللاحقة إلى وظيفة منفصلة، TheCmdPostProc، والتي يتم دفعها إلى مكدس الوظائف. يتم استبدال Tcl_EvalObjEx بـ Tcl_NREvalObj، الذي يستخدم آلية قفز بدلاً من استهلاك مساحة في مكدس C. الإصدار الجديد من TheCmdOldObjProc هو مجرد غلاف يستخدم Tcl_NRCallObjProc لاستدعاء TheCmdNRObjProc:

int
TheCmdOldObjProc(
    ClientData clientData,
    Tcl_Interp *interp,
    int objc,
    Tcl_Obj *const objv[])
{
    return Tcl_NRCallObjProc(interp, TheCmdNRObjProc,
            clientData, objc, objv);
}
int
TheCmdNRObjProc
    ClientData clientData,
    Tcl_Interp *interp,
    int objc,
    Tcl_Obj *const objv[])
{
    Tcl_Obj *objPtr;
    ... تحضير ...
    Tcl_NRAddCallback(interp, TheCmdPostProc,
            data0, data1, data2, data3);
    /* data0 .. data3 هي ما يصل إلى أربعة عناصر بايت واحد يتم تمريرها إلى وظيفة المعالجة اللاحقة */
    return Tcl_NREvalObj(interp, objPtr, 0);
}
int
TheCmdNRPostProc(
    ClientData data[],
    Tcl_Interp *interp,
    int result)
{
    /* data[0] .. data[3] هي العناصر الأربعة للبيانات
     * التي تم تمريرها إلى Tcl_NRAddCallback */
    ... معالجة لاحقة ...
    return result;
}

يمكن لأي وظيفة تشكل جزءًا من روتين دفع وظائف أخرى، مما يسمح بتنفيذ هياكل تحكم مثل الحلقات والتسلسلات باستخدام مكدس الوظائف.
