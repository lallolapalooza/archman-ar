Tcl_Interp *interp (out)

مؤشر إلى سلسلة نصية ذات بنية قائمة صحيحة.
int *argcPtr (out)
يتم ملؤه بعدد العناصر في list.
const char ***argvPtr (out)
سيتم ملء *argvPtr بعنوان مصفوفة من المؤشرات إلى السلاسل النصية التي هي العناصر المستخرجة من list. ستكون هناك *argcPtr إدخالات صالحة في المصفوفة، تليها إدخال NULL.
int argc (in)
عدد العناصر في argv.
const char *const *argv (in)
مصفوفة سلاسل نصية ليتم دمجها معًا في قائمة واحدة. سيصبح كل سلسلة نصية عنصرًا منفصلاً في القائمة.
const char *src (in)
السلسلة النصية التي ستصبح عنصرًا في قائمة.
int *flagsPtr (in)
مؤشر إلى كلمة لملئها بمعلومات حول src. يجب تمرير قيمة *flagsPtr إلى Tcl_ConvertElement.
int length (in)
عدد البايتات في السلسلة النصية src.
char *dst (in)

معلومات حول src. يجب أن تكون القيمة التي تم إرجاعها من استدعاء سابق لـ Tcl_ScanElement، ربما تم OR-ed مع TCL_DONT_USE_BRACES.

الوصف

يمكن استخدام هذه الإجراءات لتفكيك وإعادة تجميع قوائم Tcl. تقوم Tcl_SplitList بتقسيم قائمة إلى عناصرها المكونة، وإرجاع مصفوفة من المؤشرات إلى العناصر باستخدام argcPtr و argvPtr. أثناء استخراج الوسائط، تلتزم Tcl_SplitList بالقواعد المعتادة للاستبدالات والشرطات المائلة العكسية والأقواس. منطقة الذاكرة التي يشير إليها *argvPtr مخصصة ديناميكيًا؛ بالإضافة إلى مصفوفة المؤشرات، فإنها تحتوي أيضًا على نسخ من جميع عناصر القائمة. تقع على عاتق المتصل مسؤولية تحرير كل هذه التخزين. على سبيل المثال، افترض أنك استدعيت Tcl_SplitList بالكود التالي:

int argc, code;
char *string;
char **argv;
...
code = Tcl_SplitList(interp, string, &argc, &argv);

ثم يجب عليك تحرير التخزين في النهاية باستدعاء مثل التالي:

Tcl_Free((char *) argv);

تُرجع Tcl_SplitList عادةً TCL_OK، مما يعني أنه تم تحليل القائمة بنجاح. إذا كان هناك خطأ في بناء الجملة في list، فسيتم إرجاع TCL_ERROR وسيستقر نتيجة المفسر على رسالة خطأ تصف المشكلة (إذا لم يكن interp NULL). إذا تم إرجاع TCL_ERROR، فلن يتم تخصيص ذاكرة ولن يتم تعديل *argvPtr.

Tcl_Merge هي عكس Tcl_SplitList: تأخذ مجموعة من السلاسل النصية المقدمة بواسطة argc و argv وتنشئ سلسلة نصية ناتجة لها بنية قائمة صحيحة. هذا يعني أنه يمكن استخدام أوامر مثل index لاستخراج العناصر الأصلية مرة أخرى. بالإضافة إلى ذلك، إذا تم تمرير نتيجة Tcl_Merge إلى Tcl_Eval، فسيتم تحليلها إلى argc كلمة ستكون قيمها هي نفسها السلاسل النصية argv التي تم تمريرها إلى Tcl_Merge. ستقوم Tcl_Merge بتعديل عناصر القائمة باستخدام الأقواس و/أو الشرطات المائلة العكسية لإنتاج بنية قائمة Tcl صحيحة. يتم تخصيص السلسلة النصية الناتجة ديناميكيًا باستخدام Tcl_Alloc؛ يجب على المتصل تحرير المساحة في النهاية باستخدام Tcl_Free.

إذا تم تمرير نتيجة Tcl_Merge إلى Tcl_SplitList، فإن العناصر التي تم إرجاعها بواسطة Tcl_SplitList ستكون متطابقة مع تلك التي تم تمريرها إلى Tcl_Merge. ومع ذلك، فإن العكس ليس صحيحًا: إذا تم تمرير Tcl_SplitList إلى سلسلة نصية معينة، وتم تمرير argc و argv الناتجة إلى Tcl_Merge، فقد لا تكون السلسلة النصية الناتجة هي نفسها السلسلة النصية الأصلية التي تم تمريرها إلى Tcl_SplitList. هذا لأن Tcl_Merge قد تستخدم الشرطات المائلة العكسية والأقواس بشكل مختلف عن السلسلة النصية الأصلية.

Tcl_ScanElement و Tcl_ConvertElement هما الإجراءان اللذان يقومان بكل العمل الحقيقي لـ Tcl_Merge. تقوم Tcl_ScanElement بفحص وسيطتها src وتحدد كيفية استخدام الشرطات المائلة العكسية والأقواس عند تحويلها إلى عنصر قائمة. تُرجع تقديرًا زائدًا لعدد الأحرف المطلوبة لتمثيل src كعنصر قائمة، وتخزن معلومات في *flagsPtr المطلوبة بواسطة Tcl_ConvertElement.

Tcl_ConvertElement هو إجراء مصاحب لـ Tcl_ScanElement. يقوم بالعمل الفعلي لتحويل سلسلة نصية إلى عنصر قائمة. يجب أن تكون وسيطتها flags هي نفس القيمة التي تم إرجاعها بواسطة Tcl_ScanElement. تكتب Tcl_ConvertElement عنصر قائمة صحيحًا إلى الذاكرة بدءًا من *dst وتُرجع عدد الأحرف الإجمالي المكتوب، والذي لن يكون أكثر من النتيجة التي تم إرجاعها بواسطة Tcl_ScanElement. تكتب Tcl_ConvertElement فقط العنصر الفعلي للقائمة بدون أي مسافات بادئة أو لاحقة: يقع على عاتق المتصل تضمين مسافات بين عناصر القائمة المتجاورة.

تستخدم Tcl_ConvertElement أحد نهجين مختلفين لمعالجة الأحرف الخاصة في src. كلما أمكن، فإنها تعالج الأحرف الخاصة عن طريق إحاطة السلسلة النصية بالأقواس. هذا ينتج إخراجًا أنيقًا، ولكنه لا يمكن استخدامه في بعض المواقف، مثل عندما تحتوي src على أقواس غير متطابقة. في هذه المواقف، تعالج Tcl_ConvertElement الأحرف الخاصة عن طريق إنشاء تسلسلات مائلة عكسية لها. قد يصر المتصل على النهج الثاني عن طريق OR-ing قيمة العلامة التي تم إرجاعها بواسطة Tcl_ScanElement مع TCL_DONT_USE_BRACES. على الرغم من أن هذا سينتج نتيجة أقبح، إلا أنه مفيد في بعض المواقف الخاصة، مثل عندما يتم استخدام Tcl_ConvertElement لإنشاء جزء من وسيطة لأمر Tcl. في هذه الحالة، فإن إحاطة src بأقواس معقوفة قد يتسبب في عدم تحليل الأمر بشكل صحيح.

بشكل افتراضي، ستستخدم Tcl_ConvertElement الاقتباس في إخراجها للتأكد من أن الحرف الأول للعنصر ليس الهاش (“#”.) هذا للتأكد من أن العنصر الأول لأي قائمة تم تمريرها إلى eval لن يتم تحليله بشكل خاطئ كبداية تعليق. عندما لا يكون عنصر القائمة هو العنصر الأول في القائمة، فإن هذا الاقتباس غير ضروري. عندما يمكن للمتصل التأكد من أن العنصر ليس العنصر الأول في القائمة، يمكنه تعطيل اقتباس الهاش الرائد عن طريق OR-ing قيمة العلامة التي تم إرجاعها بواسطة Tcl_ScanElement مع TCL_DONT_QUOTE_HASH.

Tcl_ScanCountedElement و Tcl_ConvertCountedElement هي نفسها Tcl_ScanElement و Tcl_ConvertElement، باستثناء أن طول السلسلة النصية src محدد بواسطة الوسيطة length، ويمكن أن تحتوي السلسلة النصية على فراغات مضمنة.
