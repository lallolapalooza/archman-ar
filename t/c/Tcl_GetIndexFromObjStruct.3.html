Tcl_GetIndexFromObjStruct(3)

الحزمة: tcl

التراخيص: TCL

Tcl_GetIndexFromObj(3) إجراءات مكتبة Tcl Tcl_GetIndexFromObj(3)

الاسم

Tcl_GetIndexFromObj, Tcl_GetIndexFromObjStruct - البحث عن سلسلة نصية في جدول الكلمات المفتاحية

الملخص

#include <tcl.h>

int
Tcl_GetIndexFromObj(interp, objPtr, tablePtr, msg, flags,
indexPtr)

int
Tcl_GetIndexFromObjStruct(interp, objPtr, structTablePtr, offset,
                          msg, flags, indexPtr)

المعاملات

Tcl_Interp *interp (في)
المفسر المستخدم للإبلاغ عن الأخطاء؛ إذا كان NULL، فلن يتم تقديم رسالة عند حدوث أخطاء.
Tcl_Obj *objPtr (في/خارج)
تُستخدم القيمة النصية لهذا الكائن للبحث في tablePtr. يتم تعديل التمثيل الداخلي ليحمل فهرس إدخال الجدول المطابق.
const char *const *tablePtr (في)
مصفوفة من السلاسل النصية المنتهية بـ NULL. يتم تمييز نهاية المصفوفة بسلسلة نصية NULL. لاحظ أنه قد يتم الاحتفاظ بمراجع لـ tablePtr في التمثيل الداخلي لـ objPtr، لذلك يجب أن يمثل هذا عنوان مصفوفة مخصصة ثابتة.
const void *structTablePtr (في)
مصفوفة من النوع العشوائي، عادةً نوع struct معين. يجب أن يكون العضو الأول في الهيكل سلسلة نصية منتهية بـ NULL. يتم تحديد حجم الهيكل بواسطة offset. لاحظ أنه قد يتم الاحتفاظ بمراجع لـ structTablePtr في التمثيل الداخلي لـ objPtr، لذلك يجب أن يمثل هذا عنوان مصفوفة مخصصة ثابتة من الهياكل.
int offset (في)
الإزاحة التي يجب إضافتها إلى structTablePtr للوصول إلى الإدخال التالي. يتم تمييز نهاية المصفوفة بسلسلة نصية NULL.
const char *msg (في)
سلسلة نصية منتهية بـ NULL تصف ما يتم البحث عنه، مثل option. يتم تضمين هذه السلسلة النصية في رسائل الخطأ.
int flags (في)
مزيج من البتات (OR) يوفر معلومات إضافية للعملية. البت الوحيد المعرف حاليًا هو TCL_EXACT.
int *indexPtr (خارج)
يتم إرجاع فهرس السلسلة النصية في tablePtr التي تتطابق مع قيمة objPtr هنا.

الوصف

توفر هذه الإجراءات طريقة فعالة للبحث عن الكلمات المفتاحية، أسماء المفاتيح، أسماء الخيارات، والأشياء المماثلة حيث يجب اختيار القيمة الحرفية لقيمة Tcl من مجموعة محددة مسبقًا. تقارن Tcl_GetIndexFromObj objPtr بكل سلسلة نصية في tablePtr للعثور على تطابق. يحدث التطابق إذا كانت القيمة النصية لـ objPtr متطابقة مع إحدى السلاسل النصية في tablePtr، أو إذا كانت اختصارًا فريدًا غير فارغ لإحدى السلاسل النصية في tablePtr بالضبط ولم يتم تحديد العلم TCL_EXACT؛ في كلتا الحالتين، يتم تخزين فهرس الإدخال المطابق في *indexPtr ويتم إرجاع TCL_OK.

إذا لم يكن هناك إدخال مطابق، يتم إرجاع TCL_ERROR ويتم ترك رسالة خطأ في نتيجة interp إذا لم يكن interp NULL. يتم تضمين msg في رسالة الخطأ للإشارة إلى ما كان يتم البحث عنه. على سبيل المثال، إذا كان msg هو option، فستكون رسالة الخطأ بالشكل التالي: "bad option \"firt\": must be first, second, or third".

إذا أكملت Tcl_GetIndexFromObj بنجاح، فإنها تعدل التمثيل الداخلي لـ objPtr لتخزين عنوان الجدول وفهرس الإدخال المطابق. إذا تم استدعاء Tcl_GetIndexFromObj مرة أخرى بنفس وسيطات objPtr و tablePtr (على سبيل المثال، أثناء إعادة استدعاء أمر Tcl)، فإنها تعيد الفهرس المطابق فورًا دون الحاجة إلى تكرار عملية البحث. ملاحظة: تفترض Tcl_GetIndexFromObj أن إدخالات tablePtr ثابتة: لا يجب أن تتغير بين الاستدعاءات. إذا كانت قيمة objPtr سلسلة نصية فارغة، فإن Tcl_GetIndexFromObj ستعاملها كقيمة غير مطابقة وتعيد TCL_ERROR.

تعمل Tcl_GetIndexFromObjStruct تمامًا مثل Tcl_GetIndexFromObj، باستثناء أنها بدلاً من التعامل مع tablePtr كمصفوفة من مؤشرات السلاسل النصية، فإنها تتعامل معها كمؤشر إلى السلسلة النصية الأولى في سلسلة من السلاسل النصية التي تفصل بينها offset بايت (أي أن هناك مؤشرًا إلى أول مصفوفة أحرف في tablePtr، ومؤشرًا إلى ثاني مصفوفة أحرف في tablePtr + offset بايت، وهكذا) هذا مفيد بشكل خاص عند معالجة أشياء مثل Tk_ConfigurationSpec، التي تكون مفاتيحها النصية في نفس المكان في كل عنصر من عناصر المصفوفة.
