المعاملات

const Tcl_ChannelType *typePtr (in)
يشير إلى بنية تحتوي على عناوين الإجراءات التي يمكن استدعاؤها لتنفيذ عمليات الإدخال/الإخراج والوظائف الأخرى على القناة.
const char *channelName (in)
اسم هذه القناة، مثل file3؛ يجب ألا يكون قيد الاستخدام من قبل أي قناة أخرى. يمكن أن يكون NULL، وفي هذه الحالة يتم إنشاء القناة بدون اسم. إذا تم تعيين القناة التي تم إنشاؤها لواحدة من القنوات القياسية (stdin، stdout أو stderr)، فسيكون اسم القناة المعينة هو اسم القناة القياسية.
ClientData instanceData (in)
قيمة كلمة واحدة اختيارية ليتم ربطها بهذه القناة. يتم تمرير هذه القيمة إلى الإجراءات في typePtr عند استدعائها.
int mask (in)
مزيج ORed من TCL_READABLE و TCL_WRITABLE للإشارة إلى ما إذا كانت القناة قابلة للقراءة والكتابة.

القناة المراد العمل عليها.
int direction (in)
TCL_READABLE يعني أن مقبض الإدخال مطلوب؛ TCL_WRITABLE يعني أن مقبض الإخراج مطلوب.
ClientData *handlePtr (out)
يشير إلى الموقع الذي يجب تخزين المقبض الخاص بنظام التشغيل فيه.
int size (in)
الحجم، بالبايت، للمخازن المؤقتة التي سيتم تخصيصها في هذه القناة.
int mask (in)
مزيج ORed من TCL_READABLE، TCL_WRITABLE و TCL_EXCEPTION الذي يشير إلى الأحداث التي وقعت على هذه القناة.
Tcl_Interp *interp (in)
المترجم الحالي. (يمكن أن يكون NULL)
const char *optionName (in)

الوصف

تستخدم Tcl بنية قناة ثنائية الطبقات. توفر طبقة عليا عامة لتمكين برامج C و Tcl من إجراء عمليات الإدخال والإخراج باستخدام نفس واجهات برمجة التطبيقات لمجموعة متنوعة من الملفات والأجهزة والمقابس وما إلى ذلك. واجهات برمجة التطبيقات العامة لـ C موصوفة في إدخال الدليل الخاص بـ Tcl_OpenFileChannel.

توفر الطبقة السفلية برامج تشغيل قنوات خاصة بالنوع لكل نوع جهاز مدعوم على كل منصة. يصف إدخال الدليل هذا واجهات برمجة التطبيقات لـ C المستخدمة للتواصل بين الطبقة العامة وبرامج تشغيل القنوات الخاصة بالنوع. كما يشرح كيف يمكن إضافة أنواع جديدة من القنوات عن طريق توفير برامج تشغيل قنوات جديدة.

تتكون برامج تشغيل القنوات من عدد من المكونات: أولاً، يوفر كل برنامج تشغيل قناة بنية Tcl_ChannelType تحتوي على مؤشرات إلى الإجراءات التي تنفذ العمليات المختلفة التي تستخدمها الطبقة العامة للتواصل مع برنامج تشغيل القناة. يتم وصف بنية Tcl_ChannelType والإجراءات المشار إليها بها في القسم TCL_CHANNELTYPE، أدناه.

ثانيًا، توفر برامج تشغيل القنوات عادةً أمر Tcl لإنشاء مثيلات من هذا النوع من القنوات. على سبيل المثال، ينشئ أمر Tcl open قنوات تستخدم برامج تشغيل قنوات الملفات والأوامر، وينشئ أمر Tcl socket قنوات تستخدم مقابس TCP للاتصال بالشبكة.

ثالثًا، يوفر برنامج تشغيل القناة اختياريًا إجراء C لفتح مثيلات قنوات من هذا النوع. على سبيل المثال، يفتح Tcl_OpenFileChannel قناة تستخدم برنامج تشغيل قنوات الملفات، ويفتح Tcl_OpenTcpClient قناة تستخدم بروتوكول شبكة TCP. عادةً ما تستخدم وظائف الإنشاء هذه Tcl_CreateChannel داخليًا لفتح القناة.

لإضافة نوع جديد من القنوات، يجب عليك تنفيذ واجهة برمجة تطبيقات C أو أمر Tcl يفتح قناة عن طريق استدعاء Tcl_CreateChannel. عند استدعاء برنامج التشغيل الخاص بك لـ Tcl_CreateChannel، فإنه يمرر بنية Tcl_ChannelType تصف إجراءات الإدخال/الإخراج الخاصة ببرنامج التشغيل. ستقوم الطبقة العامة بعد ذلك باستدعاء الإجراءات المشار إليها في تلك البنية لتنفيذ عمليات على القناة.

يفتح Tcl_CreateChannel قناة جديدة ويربط typePtr و instanceData المقدمين بها. يتم فتح القناة في الوضع المحدد بواسطة mask. لمناقشة برامج تشغيل القنوات وعملياتها وبنية Tcl_ChannelType، راجع القسم TCL_CHANNELTYPE، أدناه.

يتفاعل Tcl_CreateChannel مع التعليمات البرمجية التي تدير القنوات القياسية. بمجرد تهيئة قناة قياسية إما من خلال استدعاء Tcl_GetStdChannel أو استدعاء Tcl_SetStdChannel، سيؤدي إغلاق هذه القناة القياسية إلى قيام الاستدعاء التالي لـ Tcl_CreateChannel بجعل القناة الجديدة هي القناة القياسية الجديدة أيضًا. راجع Tcl_StandardChannels للحصول على شرح عام للقنوات القياسية وسلوك مكتبة Tcl فيما يتعلق بها.

يعيد Tcl_GetChannelInstanceData بيانات المثيل المرتبطة بالقناة في channel. هذا هو نفس الوسيط instanceData في استدعاء Tcl_CreateChannel الذي أنشأ هذه القناة.

يعيد Tcl_GetChannelType مؤشرًا إلى بنية Tcl_ChannelType التي تستخدمها القناة في الوسيط channel. هذا هو نفس الوسيط typePtr في استدعاء Tcl_CreateChannel الذي أنشأ هذه القناة.

يعيد Tcl_GetChannelName سلسلة تحتوي على الاسم المرتبط بالقناة، أو NULL إذا كان الوسيط channelName لـ Tcl_CreateChannel هو NULL.

يقوم Tcl_GetChannelHandle بوضع المقبض الخاص بنظام التشغيل المرتبط بـ channel للاتجاه direction المحدد في الموقع

هذا الموقع غير مرتبط بـ Arch Linux ولا تابع له بأي شكل من الأشكال. المحتوى هو ترجمة فقط لصفحات دليل Arch المنشورة على man.archlinux.org.

INPUTPROC

يحتوي الحقل inputProc على عنوان دالة يتم استدعاؤها بواسطة قادر القراءة لقراءة البيانات من جهاز أو ملف وتخزينها في مخزن مؤقت داخلي. يجب أن يتوافق InputProc مع النموذج الأولي التالي:

typedef int Tcl_DriverInputProc(
        ClientData instanceData,
        char *buf,
        int bufSize,
        int *errorCodePtr);

InstanceData هو نفس القيمة التي تم تمريرها إلى Tcl_CreateChannel عند إنشاء القناة. يشير الوسيط buf إلى مصفوفة من البايتات سيتم تخزين بيانات الإدخال من الجهاز فيها، ويشير الوسيط bufSize إلى عدد البايتات المتاحة في buf.

يشير الوسيط errorCodePtr إلى متغير عدد صحيح مقدم من قادر القراءة. في حالة حدوث خطأ، يجب على الدالة تعيين هذا المتغير إلى رمز خطأ POSIX يحدد الخطأ الذي حدث.

يجب على الدالة قراءة البيانات من الجهاز المغلف بالقناة وتخزينها في buf. في حالة النجاح، يجب على الدالة إرجاع رقم غير سالب يشير إلى عدد البايتات التي تم قراءتها من الجهاز وتخزينها في buf. في حالة حدوث خطأ، يجب على الدالة إرجاع -1. إذا حدث خطأ بعد قراءة بعض البيانات من الجهاز، فسيتم فقدان تلك البيانات.

إذا كان بإمكان inputProc تحديد أن جهاز الإدخال يحتوي على بعض البيانات المتاحة ولكن أقل من المطلوب بواسطة الوسيط bufSize، فيجب على الدالة محاولة قراءة البيانات المتاحة فقط والعودة دون حظر. إذا لم تكن هناك بيانات متاحة في جهاز الإدخال وكانت القناة في وضع غير مانع، فيجب على الدالة إرجاع خطأ EAGAIN. إذا لم تكن هناك بيانات متاحة في جهاز الإدخال وكانت القناة في وضع مانع، فيجب على الدالة الانتظار لأقصر وقت ممكن حتى يمكن قراءة بايت واحد على الأقل من الجهاز، ثم إرجاع أقصى قدر من البيانات التي يمكن قراءتها دون حظر.

يمكن استرداد هذه القيمة باستخدام Tcl_ChannelInputProc، والتي ترجع مؤشرًا إلى الدالة.

OUTPUTPROC

يحتوي الحقل outputProc على عنوان دالة يتم استدعاؤها بواسطة قادر الكتابة لنقل البيانات من المخزن المؤقت الداخلي إلى جهاز إخراج. يجب أن يتوافق OutputProc مع النموذج الأولي التالي:

typedef int Tcl_DriverOutputProc(
        ClientData instanceData,
        const char *buf,
        int toWrite,
        int *errorCodePtr);

InstanceData هو نفس القيمة التي تم تمريرها إلى Tcl_CreateChannel عند إنشاء القناة. يحتوي الوسيط buf على مصفوفة من البايتات التي سيتم كتابتها إلى الجهاز، ويشير الوسيط toWrite إلى عدد البايتات التي سيتم كتابتها من الوسيط buf.

يشير الوسيط errorCodePtr إلى متغير عدد صحيح مقدم من قادر الكتابة. في حالة حدوث خطأ، يجب على الدالة تعيين هذا المتغير إلى رمز خطأ POSIX يحدد الخطأ.

يجب على الدالة كتابة البيانات في buf إلى جهاز الإخراج المغلف بالقناة. في حالة النجاح، يجب على الدالة إرجاع رقم غير سالب يشير إلى عدد البايتات التي تم كتابتها إلى جهاز الإخراج. عادة ما تكون قيمة الإرجاع هي نفسها toWrite، ولكنها قد تكون أقل في بعض الحالات مثل إذا تم مقاطعة عملية الكتابة بإشارة. في حالة حدوث خطأ، يجب على الدالة إرجاع -1. في حالة حدوث خطأ، قد تكون بعض البيانات قد تمت كتابتها إلى الجهاز.

إذا كانت القناة في وضع غير مانع ولم يتمكن جهاز الإخراج من استيعاب أي بيانات، فيجب على الدالة إرجاع -1 مع خطأ EAGAIN دون كتابة أي بيانات.

يمكن استرداد هذه القيمة باستخدام Tcl_ChannelOutputProc، والتي ترجع مؤشرًا إلى الدالة.

SEEKPROC AND WIDESEEKPROC

يحتوي الحقل seekProc على عنوان دالة يتم استدعاؤها بواسطة قادر القراءة لنقل نقطة الوصول التي سيتم تطبيق عمليات القراءة أو الكتابة التالية عليها. يجب أن يتوافق SeekProc مع النموذج الأولي التالي:

typedef int Tcl_DriverSeekProc(
        ClientData instanceData,
        long offset,
        int seekMode,
        int *errorCodePtr);

الوسيط instanceData هو نفس القيمة المقدمة إلى Tcl_CreateChannel عند إنشاء هذه القناة. Offset و seekMode لهما نفس المعنى كما في الإجراء Tcl_Seek (المذكور في دليل Tcl_OpenFileChannel).

يشير الوسيط errorCodePtr إلى متغير عدد صحيح مقدم من قادر القراءة لإرجاع قيم errno من الدالة. يجب على الدالة تعيين هذا المتغير إلى رمز خطأ POSIX في حالة حدوث خطأ. يجب على الدالة تخزين رمز EINVAL إذا كان نوع القناة لا ينفذ البحث.

قيمة الإرجاع هي نقطة الوصول الجديدة أو -1 في حالة حدوث خطأ. في حالة حدوث خطأ، لا يجب على الدالة نقل نقطة الوصول.

إذا كان هناك حقل غير فارغ لـ seekProc، فقد يحتوي الحقل wideSeekProc على عنوان دالة بديلة لاستخدامها والتي تعالج الإزاحات (أي، الإزاحات الأكبر من 32 بت)، مما يسمح بالبحث داخل ملفات أكبر من 2 جيجابايت. يتم استدعاء wideSeekProc مفضلاً على seekProc، ولكن يجب تعريف كليهما إذا تم تعريف wideSeekProc. يجب أن يتوافق WideSeekProc مع النموذج الأولي التالي:

typedef Tcl_WideInt Tcl_DriverWideSeekProc(
        ClientData instanceData,
        Tcl_WideInt offset,
        int seekMode,
        int *errorCodePtr);

الوسائط وقيم الإرجاع هي نفسها كما هو الحال مع seekProc، باستثناء أن نوع الإزاحة ونوع الإرجاع مختلفان.

يمكن استرداد الحقل seekProc باستخدام Tcl_ChannelSeekProc، والذي يرجع مؤشرًا إلى الدالة، ويمكن استرداد الحقل wideSeekProc باستخدام Tcl_ChannelWideSeekProc.

SETOPTIONPROC

يحتوي الحقل setOptionProc على عنوان دالة يتم استدعاؤها بواسطة قادر القراءة لتعيين خيار خاص بنوع القناة في قناة. يجب أن يتوافق setOptionProc مع النموذج الأولي التالي:

typedef int Tcl_DriverSetOptionProc(
        ClientData instanceData,
        Tcl_Interp *interp,
        const char *optionName,
        const char *newValue);

optionName هو اسم خيار سيتم تعيينه، و newValue هو القيمة الجديدة لهذا الخيار، كسلسلة نصية. instanceData هو نفس القيمة المقدمة إلى Tcl_CreateChannel عند إنشاء هذه القناة. يجب على الدالة تنفيذ الإجراء المحدد وفقًا لنوع القناة لتنفيذ القيمة الجديدة للخيار.

تتم معالجة بعض الخيارات بواسطة الكود العام ولا يتم استدعاء هذه الدالة أبدًا لتعيينها، على سبيل المثال -blockmode. الخيارات الأخرى خاصة بكل نوع قناة وسيتم استدعاء دالة setOptionProc الخاصة بمشغل القناة لتنفيذها. قد يكون الحقل setOptionProc فارغًا، مما يشير إلى أن هذا النوع من القنوات لا يدعم خيارات خاصة.

إذا تم تعديل قيمة الخيار بنجاح إلى قيمتها الجديدة، فإن الدالة ترجع TCL_OK. يجب عليها استدعاء Tcl_BadChannelOption، والتي ترجع TCL_ERROR إذا لم يتم التعرف على optionName. إذا حدد newValue قيمة غير مدعومة أو إذا حدث خطأ في استدعاء نظام، فيجب على الدالة ترك رسالة خطأ في نتيجة interp إذا لم يكن interp فارغًا. يجب على الدالة أيضًا استدعاء Tcl_SetErrno لتخزين رمز خطأ POSIX مناسب.

يمكن استرداد هذه القيمة باستخدام Tcl_ChannelSetOptionProc، والتي ترجع مؤشرًا إلى الدالة.

GETOPTIONPROC

يحتوي الحقل getOptionProc على عنوان دالة يتم استدعاؤها بواسطة قادر القراءة للحصول على قيمة خيار خاص بنوع القناة في قناة. يجب أن يتوافق getOptionProc مع النموذج الأولي التالي:

typedef int Tcl_DriverGetOptionProc(
        ClientData instanceData,
        Tcl_Interp *interp,
        const char *optionName,
        Tcl_DString *optionValue);

optionName هو اسم خيار مدعوم من قبل هذا النوع من القنوات. إذا لم يكن اسم الخيار NULL، فإن الدالة تخزن قيمته الحالية، كسلسلة نصية، في السلسلة الديناميكية Tcl optionValue. إذا كان optionName NULL، فإن الدالة تخزن في optionValue قائمة متناوبة من جميع الخيارات المدعومة وقيمها الحالية. في حالة النجاح، ترجع الدالة TCL_OK. يجب عليها استدعاء Tcl_BadChannelOption، والتي ترجع TCL_ERROR إذا لم يتم التعرف على optionName. إذا حدد newValue قيمة للخيار غير مدعومة أو إذا حدث خطأ في استدعاء نظام، فيجب على الدالة ترك رسالة خطأ في نتيجة interp إذا لم يكن interp فارغًا. يجب على الدالة أيضًا استدعاء Tcl_SetErrno لتخزين رمز خطأ POSIX مناسب.

يمكن استرداد هذه القيمة باستخدام Tcl_ChannelGetOptionProc، والتي ترجع مؤشرًا إلى الدالة.
