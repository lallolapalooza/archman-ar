المعاملات

Tcl_Interp *interp (in)
المفسر المستخدم للإبلاغ عن الأخطاء، أو NULL إذا لم يكن الإبلاغ عن الأخطاء مطلوبًا.
const char *name (in)
اسم الترميز المطلوب تحميله.
Tcl_Encoding encoding (in)
الترميز للاستعلام عنه، تحريره، أو استخدامه لتحويل النص. إذا كان encoding هو NULL، يتم استخدام ترميز النظام الحالي.
Tcl_Obj *objPtr (in)
اسم الترميز للحصول على رمزه.
Tcl_Encoding *encodingPtr (out)
يشير إلى المساحة حيث سيتم كتابة رمز الترميز.
const char *src (in)
بالنسبة لدوال Tcl_ExternalToUtf، مصفوفة من البايتات بالترميز المحدد التي سيتم تحويلها إلى UTF-8. بالنسبة لدوال Tcl_UtfToExternal و Tcl_WinUtfToTChar، مصفوفة من أحرف UTF-8 التي سيتم تحويلها إلى الترميز المحدد.
const TCHAR *tsrc (in)
مصفوفة من أحرف Windows TCHAR لتحويلها إلى UTF-8.
int srcLen (in)
طول src أو tsrc بالبايت. إذا كان الطول سالبًا، يتم استخدام الطول الخاص بالترميز للسلسلة.
Tcl_DString *dstPtr (out)
مؤشر إلى Tcl_DString غير مهيأ أو تم تحريره سيتم تخزين النتيجة المحولة فيه.
int flags (in)
بتات علامات مختلفة تم دمجها معًا. TCL_ENCODING_START يعني أن مخزن المصدر هو الكتلة الأولى في تيار إدخال (قد يكون متعدد الكتل)، مما يخبر روتين التحويل بإعادة التعيين إلى حالة أولية وإجراء أي تهيئة تحتاج إلى الحدوث قبل تحويل البايت الأول. TCL_ENCODING_END يعني أن مخزن المصدر هو الكتلة الأخيرة في تيار إدخال (قد يكون متعدد الكتل)، مما يخبر روتين التحويل بإجراء أي إنهاء يحتاج إلى الحدوث بعد تحويل البايت الأخير ثم إعادة التعيين إلى حالة أولية. TCL_ENCODING_STOPONERROR يعني أن روتين التحويل يجب أن يعود فورًا عند قراءة حرف مصدر غير موجود في الترميز الهدف؛ وإلا سيتم استبدال حرف احتياطي افتراضي تلقائيًا.
Tcl_EncodingState *statePtr (in/out)
يستخدم عند تحويل تيار بايت (عادةً ما يكون طويلًا أو ذا طول غير محدد) قطعة بقطعة. يقوم روتين التحويل بتخزين حالته الحالية في *statePtr بعد تحويل src (المخزن الذي يحتوي على القطعة الحالية)؛ يجب إعادة تمرير معلومات الحالة تلك عند تحويل القطعة التالية من التيار حتى يعرف روتين التحويل الحالة التي كان عليها عندما توقف في نهاية القطعة الأخيرة. قد يكون NULL، وفي هذه الحالة يتم تجاهل القيمة المحددة لـ flags ويفترض أن مخزن المصدر يحتوي على السلسلة الكاملة المراد تحويلها.
char *dst (out)
المخزن الذي سيتم تخزين النتيجة المحولة فيه. لن يتم تخزين أكثر من dstLen بايت في dst.
int dstLen (in)
الحد الأقصى لطول مخزن الإخراج dst بالبايت.
int *srcReadPtr (out)
يتم ملؤه بعدد البايتات من src التي تم تحويلها فعليًا. قد يكون هذا أقل من طول المصدر الأصلي إذا كانت هناك مشكلة في تحويل بعض أحرف المصدر. قد يكون NULL.
int *dstWrotePtr (out)
يتم ملؤه بعدد البايتات التي تم تخزينها فعليًا في مخزن الإخراج كنتيجة للتحويل. قد يكون NULL.
int *dstCharsPtr (out)
يتم ملؤه بعدد الأحرف التي تتوافق مع عدد البايتات المخزنة في مخزن الإخراج. قد يكون NULL.
Tcl_DString *bufPtr (out)
مساحة تخزين لاسم ترميز النظام المحدد.
const Tcl_EncodingType *typePtr (in)
هيكل يحدد نوع ترميز جديد.
Tcl_Obj *searchPath (in)
قائمة الدلائل في نظام الملفات التي سيتم البحث فيها عن ملفات بيانات الترميز.
const char *path (in)
مسار إلى موقع ملف الترميز.

مقدمة

تقوم هذه الروتينات بالتحويل بين تمثيل الأحرف الداخلي لـ Tcl، UTF-8، وتمثيلات الأحرف التي تستخدمها أنظمة التشغيل أو أنظمة الملفات المختلفة، مثل Unicode، ASCII، أو Shift-JIS. عند التشغيل على السلاسل النصية، مثل الحصول على أسماء الملفات أو عرض الأحرف باستخدام خطوط دولية، يجب ترجمة السلاسل النصية إلى تنسيق واحد أو ربما تنسيقات متعددة يمكن لمكالمات النظام المختلفة استخدامها. على سبيل المثال، على محطة عمل Unix يابانية، قد يحصل المستخدم على اسم ملف ممثل بترميز الملف EUC-JP ثم يقوم بترجمة الأحرف إلى ترميز الخط jisx0208 لعرض اسم الملف في عنصر واجهة Tk. الغرض من حزمة الترميز هو المساعدة في سد فجوة الترجمة. يوفر UTF-8 أرضية وسيطة لجميع الترميزات المختلفة. في المثال أعلاه، سيتم ترجمة النص إلى UTF-8 من أي ترميز ملف يستخدمه نظام التشغيل. ثم سيتم ترجمته من UTF-8 إلى أي ترميز خط تتطلبه روتينات العرض.

بعض الترميزات الأساسية مدمجة في Tcl. يمكن تعريف ترميزات أخرى بواسطة المستخدم أو تحميلها ديناميكيًا من ملفات الترميز بطريقة مستقلة عن المنصة.

الوصف

Tcl_GetEncoding تبحث عن ترميز بالنظر إلى name الخاص به. قد يشير الاسم إلى ترميز Tcl مدمج، أو ترميز معرف من قبل المستخدم مسجل عن طريق استدعاء Tcl_CreateEncoding، أو ملف ترميز قابل للتحميل ديناميكيًا. القيمة المرجعة هي رمز يمثل الترميز ويمكن استخدامه في استدعاءات لاحقة لإجراءات مثل Tcl_GetEncodingName، Tcl_FreeEncoding، و Tcl_UtfToExternal. إذا لم يشير الاسم إلى أي ترميز معروف أو قابل للتحميل، يتم إرجاع NULL ويتم إرجاع رسالة خطأ في interp.

تحتفظ حزمة الترميز بقاعدة بيانات لجميع الترميزات المستخدمة حاليًا. في المرة الأولى التي يتم فيها رؤية name، تقوم Tcl_GetEncoding بإرجاع ترميز مع عدد مراجع يساوي 1. إذا تم طلب نفس name مرات إضافية، فسيتم زيادة عدد المراجع لهذا الترميز دون تكلفة تخصيص ترميز جديد وجميع هياكل البيانات المرتبطة به.

عندما لم يعد encoding مطلوبًا، يجب استدعاء Tcl_FreeEncoding لإصداره. عندما لا يعود encoding قيد الاستخدام في أي مكان (أي، تم تحريره بقدر عدد المرات التي تم الحصول عليه فيها)، ستقوم Tcl_FreeEncoding بتحرير جميع المساحات التي كان يستخدمها الترميز وحذفه من قاعدة البيانات.

Tcl_GetEncodingFromObj تتعامل مع التمثيل النصي لـ objPtr كاسم ترميز، وتبحث عن ترميز بهذا الاسم، تمامًا كما تفعل Tcl_GetEncoding. عند العثور على ترميز، يتم تخزينه مؤقتًا داخل قيمة objPtr للإشارة المستقبلية، ويتم كتابة رمز Tcl_Encoding إلى المساحة التي يشير إليها

Tcl_ExternalToUtfDString يحول مخزنًا مصدرًا src من الترميز المحدد encoding إلى UTF-8. يتم تخزين البايتات المحولة في dstPtr، والتي يتم إنهاؤها بعد ذلك بصفر ثنائي مشفر خاص. يجب على المتصل استدعاء Tcl_DStringFree في النهاية لتحرير أي معلومات مخزنة في dstPtr. عند التحويل، إذا لم يكن من الممكن تمثيل أي حرف في المخزن المصدر بالترميز الوجهة، فسيتم استخدام حرف افتراضي احتياطي. القيمة المرتجعة هي مؤشر إلى القيمة المخزنة في DString.

Tcl_ExternalToUtf يحول مخزنًا مصدرًا src من الترميز المحدد encoding إلى UTF-8. يتم تحويل ما لا يزيد عن srcLen بايت من المخزن المصدر ويتم تخزين ما لا يزيد عن dstLen بايت في dst. في جميع الحالات، يتم ملء *srcReadPtr بعدد البايتات المحولة من src ويتم ملء *dstWrotePtr بعدد البايتات المخزنة في dst. القيمة المرتجعة هي واحدة مما يلي:

TCL_OK
تم تحويل جميع البايتات من src.
TCL_CONVERT_NOSPACE
لم يكن مخزن الوجهة كافيًا لجميع البيانات المحولة؛ تم تحويل أكبر عدد ممكن من الأحرف.
TCL_CONVERT_MULTIBYTE
كانت البايتات الأخيرة من المخزن المصدر بداية تسلسل متعدد البايتات، ولكن كانت هناك حاجة إلى المزيد من البايتات لإكمال هذا التسلسل. يجب أن تمر استدعاءات لاحقة لروتين التحويل بمخزن يحتوي على البايتات غير المحولة المتبقية في src بالإضافة إلى بعض البايتات الإضافية من تدفق المخزن المدخل لتحويل تسلسل متعدد البايتات الذي تم تقسيمه سابقًا بشكل صحيح.
TCL_CONVERT_SYNTAX
احتوى المخزن المصدر على تسلسل أحرف غير صالح. قد يحدث هذا إذا تم إتلاف تدفق الإدخال أو إذا تم تحديد طريقة ترميز الإدخال بشكل غير صحيح.
TCL_CONVERT_UNKNOWN
احتوى المخزن المصدر على حرف لم يكن من الممكن تمثيله في ترميز الوجهة، وتم تحديد TCL_ENCODING_STOPONERROR.

Tcl_UtfToExternalDString يحول مخزنًا مصدرًا src من UTF-8 إلى الترميز المحدد encoding. يتم تخزين البايتات المحولة في dstPtr، والتي يتم إنهاؤها بعد ذلك بصفر ثنائي مناسب مشفر خاص. يجب على المتصل استدعاء Tcl_DStringFree في النهاية لتحرير أي معلومات مخزنة في dstPtr. عند التحويل، إذا لم يكن من الممكن تمثيل أي حرف في المخزن المصدر بالترميز الوجهة، فسيتم استخدام حرف افتراضي احتياطي. القيمة المرتجعة هي مؤشر إلى القيمة المخزنة في DString.

Tcl_UtfToExternal يحول مخزنًا مصدرًا src من UTF-8 إلى الترميز المحدد encoding. يتم تحويل ما لا يزيد عن srcLen بايت من المخزن المصدر ويتم تخزين ما لا يزيد عن dstLen بايت في dst. في جميع الحالات، يتم ملء *srcReadPtr بعدد البايتات المحولة من src ويتم ملء *dstWrotePtr بعدد البايتات المخزنة في dst. القيم المرتجعة هي نفسها كما في Tcl_ExternalToUtf.

Tcl_WinUtfToTChar و Tcl_WinTCharToUtf هما دالتان مساعدتان للتحويل بين UTF-8 وسلاسل Windows المستندة إلى نوع TCHAR، والذي يكون بالاتفاق حرف Unicode في Windows NT.

Tcl_GetEncodingName هو عكس Tcl_GetEncoding تقريبًا. بالنظر إلى encoding، القيمة المرتجعة هي الوسيط name الذي تم استخدامه لإنشاء الترميز. لا يُضمن أن يبقى السلسلة التي ترجعها Tcl_GetEncodingName صالحة إلا حتى يتم حذف encoding. يجب على المتصل عدم تعديل هذه السلسلة.

Tcl_SetSystemEncoding يضبط الترميز الافتراضي الذي يجب استخدامه عندما يمرر المستخدم قيمة فارغة للوسيط encoding إلى أي دالة ترميز أخرى. إذا كان name فارغًا، يتم إعادة تعيين ترميز النظام إلى الترميز الافتراضي للنظام، binary. إذا كان الاسم لا يشير إلى أي ترميز معروف أو قابل للتحميل، يتم إرجاع TCL_ERROR ويتم ترك رسالة خطأ في interp. بخلاف ذلك، تزيد هذه الإجراءات من عداد المرجع للترميز النظامي الجديد، وتقلل من عداد المرجع للترميز النظامي القديم، وترجع TCL_OK.

Tcl_GetEncodingNameFromEnvironment توفر وسيلة لمكتبة Tcl للإبلاغ عن اسم الترميز الذي تعتبره صحيحًا كترميز للنظام، بناءً على استدعاءات النظام وفحص البيئة المناسبة للمنصة. تتلقى bufPtr، وهو مؤشر إلى DString تم تهيئته أو تحريره، وتكتب اسم الترميز فيه. القيمة المرتجعة هي Tcl_DStringValue.

Tcl_GetEncodingNames يضبط قيمة interp كقائمة بأسماء جميع الترميزات المعرفة أو القابلة للتحميل ديناميكيًا، مع البحث في دليل الترميزات المحدد بواسطة Tcl_SetDefaultEncodingDir. لا تضمن هذه الإجراءات أن الملفات القابلة للتحميل ديناميكيًا تحتوي على بيانات صالحة، ولكنها تضمن فقط وجودها.

Tcl_CreateEncoding يحدد ترميزًا جديدًا ويسجل الإجراءات C التي يتم استدعاؤها للتحويل بين هذا الترميز و UTF-8. تصبح الترميزات التي تم إنشاؤها بواسطة Tcl_CreateEncoding مرئية بعد ذلك في قاعدة البيانات المستخدمة بواسطة Tcl_GetEncoding. كما هو الحال مع إجراء Tcl_GetEncoding، القيمة المرتجعة هي رمز يمثل الترميز ويمكن استخدامه في استدعاءات لاحقة لوظائف الترميز الأخرى. ترجع Tcl_CreateEncoding ترميزًا بعداد مرجع 1. إذا كان هناك بالفعل ترميز بنفس الاسم المحدد، فسيتم استبدال إدخاله في قاعدة البيانات بالترميز الجديد؛ يظل الرمز للترميز القديم صالحًا ويستمر في التصرف كما كان من قبل، ولكن مستخدمي الرمز الجديد يستدعون الآن إجراءات الترميز الجديدة.

يحتوي الوسيط typePtr لـ Tcl_CreateEncoding على معلومات حول اسم الترميز والإجراءات التي يتم استدعاؤها للتحويل بين هذا الترميز و UTF-8. يتم تعريفه على النحو التالي:

typedef struct Tcl_EncodingType {
    const char *encodingName;
    Tcl_EncodingConvertProc *toUtfProc;
    Tcl_EncodingConvertProc *fromUtfProc;
    Tcl_EncodingFreeProc *freeProc;
    ClientData clientData;
    int nullSize;
} Tcl_EncodingType;

يوفر encodingName اسمًا للترميز، والذي يمكن الإشارة إليه في إجراءات أخرى مثل Tcl_GetEncoding. يشير toUtfProc إلى إجراء استدعاء لتشغيله لتحويل النص من هذا الترميز إلى UTF-8. يشير fromUtfProc إلى إجراء استدعاء لتشغيله لتحويل النص من UTF-8 إلى هذا الترميز. يشير freeProc إلى إجراء استدعاء لتشغيله عند حذف هذا الترميز. قد يكون الحقل freeProc فارغًا. يحتوي clientData على قيمة عشوائية من كلمة واحدة، يتم تمريرها إلى toUtfProc و fromUtfProc و freeProc عند تشغيلها. عادةً ما يكون مؤشرًا إلى بنية تحتوي على معلومات خاصة بالترميز يمكن استخدامها بواسطة إجراءات الاستدعاء. على سبيل المثال، قد تستخدم ترميزان متشابهان جدًا مثل ascii و macRoman نفس إجراء الاستدعاء، ولكنهما يستخدمان قيم clientData مختلفة للتحكم في سلوكهما. يحدد nullSize عدد الأصفار التي تعني نهاية السلسلة في هذا الترميز. يجب أن يكون 1 (للترميزات أحادية البايت أو متعددة البايت مثل ASCII أو Shift-JIS) أو 2 (للترميزات ثنائية البايت مثل Unicode). لا يتم قبول الترميزات ذات الحجم الثابت من 3 بايتات أو أكثر لكل حرف (مثل CNS11643).

يجب أن تتوافق إجراءات الاستدعاء toUtfProc و fromUtfProc مع النوع Tcl_EncodingConvertProc:

typedef int Tcl_EncodingConvertProc(
        ClientData clientData,
        const char *src,
        int srcLen,
        int flags,
        Tcl_EncodingState *statePtr,
        char *dst,
        int dstLen,
        int *srcReadPtr,
        int *dstWrotePtr,
        int *dstCharsPtr);

يتم استدعاء إجراءات toUtfProc و fromUtfProc بواسطة Tcl_ExternalToUtf أو Tcl_UtfToExternal لإجراء التحويل الفعلي. الوسيط clientData لهذه الإجراءات هو نفس clientData المحدد في Tcl_CreateEncoding عند إنشاء الترميز. الوسائط المتبقية لإجراءات الاستدعاء هي نفس الوسائط، الموثقة في الأعلى، لـ Tcl_ExternalToUtf أو Tcl_UtfToExternal، مع الاستثناءات التالية. إذا كان الوسيط srcLen لإحدى هذه الدوال عالية المستوى سالبًا، فإن القيمة الممررة إلى إجراء الاستدعاء ستكون الطول المناسب لـ src للترميز المحدد. إذا كان أي من الوسائط srcReadPtr أو dstWrotePtr أو dstCharsPtr لإحدى هذه الدوال عالية المستوى فارغًا، فإن القيمة المقابلة الممررة إلى إجراء الاستدعاء ستكون مكانًا غير فارغ.

يجب أن يتوافق إجراء الاستدعاء freeProc، إذا لم يكن فارغًا، مع النوع Tcl_EncodingFreeProc:

typedef void Tcl_EncodingFreeProc(
        ClientData clientData);

يتم استدعاء دالة freeProc هذه عند حذف الترميز. الوسيط clientData هو نفس clientData المحدد في Tcl_CreateEncoding عند إنشاء الترميز.

Tcl_GetEncodingSearchPath و Tcl_SetEncodingSearchPath يتم استدعاؤهما للوصول إلى وتكوين قائمة الدلائل التي يتم البحث فيها عن ملفات بيانات الترميز.

القيمة المرتجعة لـ Tcl_GetEncodingSearchPath هي القيمة التي تم تخزينها بواسطة آخر استدعاء ناجح لـ Tcl_SetEncodingSearchPath. إذا لم يتم إجراء أي استدعاءات لـ Tcl_SetEncodingSearchPath، ستقوم Tcl بحساب قيمة أولية بناءً على البيئة. هناك مسار بحث ترميز واحد للعملية بأكملها، مشترك بين جميع الخيوط في العملية.

Tcl_SetEncodingSearchPath يخزن searchPath ويرجع TCL_OK، ما لم يكن searchPath قائمة Tcl غير صالحة، وفي هذه الحالة يرجع TCL_ERROR. عناصر searchPath
