WideSeekProc debe corresponder a la siguiente prototipos: typedef Tcl_WideInt Tcl_DriverWideSeekProc( ClientData instanceData, Tcl_WideInt offset, int seekMode, int *errorCodePtr); // Los argumentos y valores de retorno son los mismos que en seekProc, excepto que el tipo de desplazamiento y el tipo de retorno son diferentes. // El valor de seekProc puede ser obtenido con Tcl_ChannelSeekProc, que devuelve un puntero a la función, y el valor de wideSeekProc puede ser obtenido con Tcl_ChannelWideSeekProc. // El campo setOptionProc contiene la dirección de una función llamada por el núcleo general para establecer una opción específica del tipo de canal en un canal. setOptionProc debe corresponder a la siguiente prototipos: typedef int Tcl_DriverSetOptionProc( ClientData instanceData, Tcl_Interp *interp, const char *optionName, const char *newValue); // optionName es el nombre de una opción a establecer, y newValue es el nuevo valor para esa opción, como una cadena. instanceData es el mismo que el valor proporcionado a Tcl_CreateChannel cuando este canal fue creado. La función debe realizar la acción específica del tipo de canal para implementar el nuevo valor de la opción. // Algunas opciones son gestionadas por el código general y esta función nunca se llama para establecerlas, por ejemplo, -blockmode. Otras opciones son específicas de cada tipo de canal y la función setOptionProc del driver del canal se llama para implementarlas. El campo setOptionProc puede ser NULL, lo cual indica que este tipo de canal no soporta opciones específicas. // Si el valor de la opción se modifica con éxito a la nueva valor, la función devuelve TCL_OK. Debe llamar a Tcl_BadChannelOption que retorna TCL_ERROR si el nombre de la opción optionName no es reconocido. Si newValue especifica un valor que no es soportado o si ocurre un error en una llamada de sistema, la función debe dejar un mensaje de error en el resultado de interp si interp no es NULL. La función debe también llamar a Tcl_SetErrno para almacenar un código de error POSIX adecuado. // Este valor puede ser obtenido con Tcl_ChannelSetOptionProc, que devuelve un puntero a la función. // El campo getOptionProc contiene la dirección de una función llamada por el núcleo general para obtener el valor de una opción específica del tipo de canal en un canal. getOptionProc debe corresponder a la siguiente prototipos: typedef int Tcl_DriverGetOptionProc( ClientData instanceData, Tcl_Interp *interp, const char *optionName, Tcl_DString *optionValue); // OptionName es el nombre de una opción soportada por este tipo de canal. Si el nombre de la opción no es NULL, la función almacena su valor actual (como una cadena) en el string dinámico optionValue. Si optionName es NULL, la función almacena en optionValue una lista alternativa de todas las opciones soportadas y sus valores actuales. En caso de éxito, la función devuelve TCL_OK. Debe llamar a Tcl_BadChannelOption que retorna TCL_ERROR si el optionName no es reconocido. Si newValue especifica un valor que no es soportado o si ocurre un error en una llamada de sistema, la función debe dejar un mensaje de error en el resultado de interp si interp no es NULL. La función debe también llamar a Tcl_SetErrno para almacenar un código de error POSIX adecuado.

WATCHPROC

يحتوي الحقل watchProc على عنوان دالة يتم استدعاؤها بواسطة المستوى العام لتهيئة آلية الإبلاغ عن الأحداث ذات الأهمية في هذه القناة. يجب أن يتطابق WatchProc مع النموذج الأولي التالي:

typedef void Tcl_DriverWatchProc(
        ClientData instanceData,
        int mask);

instanceData هو نفس القيمة التي تم تمريرها إلى Tcl_CreateChannel عند إنشاء هذه القناة. الوسيط mask هو تركيبة OR من TCL_READABLE و TCL_WRITABLE و TCL_EXCEPTION؛ يشير إلى الأحداث التي يرغب المتصل في الإبلاغ عنها في هذه القناة.

يجب أن تقوم الدالة بتهيئة آليات خاصة بنوع الجهاز للإبلاغ عند وجود حدث ذي أهمية في القناة. عند حدوث حدث ذي أهمية في القناة، يكون برنامج تشغيل القناة مسؤولاً عن استدعاء Tcl_NotifyChannel لإبلاغ وحدة القناة العامة. يجب على برنامج التشغيل تجنب استنزاف برامج تشغيل القنوات الأخرى أو مصادر الاستدعاءات العكسية عن طريق استدعاء Tcl_NotifyChannel بشكل متكرر للغاية. يمكن ضمان العدالة عن طريق استدعاء قائمة انتظار الأحداث Tcl للسماح بجدولة حدث القناة بشكل متسلسل مع الأحداث الأخرى. راجع وصف Tcl_QueueEvent للحصول على تفاصيل حول جدولة حدث.

يمكن استرداد هذه القيمة باستخدام Tcl_ChannelWatchProc، والتي تعيد عنوانًا للدالة.

GETHANDLEPROC

يحتوي الحقل getHandleProc على عنوان دالة يتم استدعاؤها بواسطة المستوى العام لاسترداد مقبض معين للقناة. يجب أن يتطابق GetHandleProc مع النموذج الأولي التالي:

typedef int Tcl_DriverGetHandleProc(
        ClientData instanceData,
        int direction,
        ClientData *handlePtr);

InstanceData هو نفس القيمة التي تم تمريرها إلى Tcl_CreateChannel عند إنشاء هذه القناة. الوسيط direction هو TCL_READABLE لاسترداد المقبض المستخدم للإدخال، أو TCL_WRITABLE لاسترداد المقبض المستخدم للإخراج.

إذا كان لتنفيذ القناة مقابض خاصة، فيجب على الدالة استرداد المقبض المناسب وفقًا للوسيط direction. يجب تخزين المقبض في المكان المشار إليه بواسطة handlePtr، ويجب إرجاع TCL_OK. إذا لم تكن القناة مفتوحة للاتجاه المحدد، أو إذا كان تنفيذ القناة لا يستخدم مقابض خاصة، فيجب على الدالة إرجاع TCL_ERROR.

يمكن استرداد هذه القيمة باستخدام Tcl_ChannelGetHandleProc، والتي تعيد عنوانًا للدالة.

FLUSHPROC

الحقل flushProc محجوز حاليًا للاستخدام المستقبلي. يجب تعيينه إلى NULL. يجب أن يتطابق FlushProc مع النموذج الأولي التالي:

typedef int Tcl_DriverFlushProc(
        ClientData instanceData);

يمكن استرداد هذه القيمة باستخدام Tcl_ChannelFlushProc، والتي تعيد عنوانًا للدالة.

HANDLERPROC

يحتوي الحقل handlerProc على عنوان دالة يتم استدعاؤها بواسطة المستوى العام لإبلاغ القناة بحدوث حدث. يجب تعريفه لبرامج تشغيل القنوات المكدسة التي ترغب في تلقي إشعارات بالأحداث في القناة الأساسية (القناة المكدسة). يجب أن يتطابق HandlerProc مع النموذج الأولي التالي:

typedef int Tcl_DriverHandlerProc(
        ClientData instanceData,
        int interestMask);

InstanceData هو نفس القيمة التي تم تمريرها إلى Tcl_CreateChannel عند إنشاء هذه القناة. الوسيط interestMask هو تركيبة OR من TCL_READABLE أو TCL_WRITABLE؛ يشير إلى نوع الحدث الذي وقع في هذه القناة.

يمكن استرداد هذه القيمة باستخدام Tcl_ChannelHandlerProc، والتي تعيد عنوانًا للدالة.

THREADACTIONPROC

يحتوي الحقل threadActionProc على عنوان الدالة التي يستدعيها المستوى العام عند إنشاء قناة، أو إغلاق قناة، أو نقلها إلى مؤشر ترابط آخر، أي عندما قد تحتاج تهيئة أو تحديث حالة مؤشر الترابط الخاصة ببرنامج التشغيل. يمكن أن تكون NULL. يستخدم الإجراء TCL_CHANNEL_THREAD_REMOVE لإعلام برنامج التشغيل بأنه يجب عليه تحديث أو إزالة أي بيانات خاصة بمؤشر الترابط قد يحتفظ بها للقناة.

يستخدم الإجراء TCL_CHANNEL_THREAD_INSERT لإعلام برنامج التشغيل بأنه يجب عليه تحديث أو تهيئة أي بيانات خاصة بمؤشر الترابط قد يحتفظ بها باستخدام مؤشر الترابط الحالي كمرتبط. راجع Tcl_CutChannel و Tcl_SpliceChannel لمزيد من التفاصيل.

typedef void Tcl_DriverThreadActionProc(
        ClientData instanceData,
        int action);

InstanceData هو نفس القيمة التي تم تمريرها إلى Tcl_CreateChannel عند إنشاء هذه القناة.

يمكن استرداد هذه القيم باستخدام Tcl_ChannelThreadActionProc، والتي تعيد عنوانًا للدالة.

TRUNCATEPROC

يحتوي الحقل truncateProc على عنوان الدالة التي يستدعيها المستوى العام عند اقتطاع قناة إلى طول معين. يمكن أن تكون NULL.

typedef int Tcl_DriverTruncateProc(
        ClientData instanceData,
        Tcl_WideInt length);

InstanceData هو نفس القيمة التي تم تمريرها إلى Tcl_CreateChannel عند إنشاء هذه القناة، و length هو الطول الجديد للملف الأساسي، والذي يجب ألا يكون سالبًا. يجب أن تكون النتيجة 0 عند النجاح أو رمز خطأ (كافٍ لاستخدامه مع Tcl_SetErrno) عند الفشل.

يمكن استرداد هذه القيم باستخدام Tcl_ChannelTruncateProc، والتي تعيد عنوانًا للدالة.

TCL_BADCHANNELOPTION

يولد هذا الإجراء رسالة خطأ "خيار سيئ" في مترجم (اختياري). يتم استخدامه بواسطة برامج تشغيل القنوات عند طلب خيار تعيين/الحصول غير صالح. هدفه هو ربط قائمة الخيارات العامة بالخيارات المحددة وتجميع رسالة خطأ الخيارات العامة.

يرجع دائمًا TCL_ERROR.

يتم إنشاء رسالة خطأ في قيمة نتيجة interp للإشارة إلى أنه تم استدعاء أمر بخيار غير صحيح. يكون تنسيق الرسالة:

    bad option "blah": should be one of
    <...generic options...>+<...specific options...>
وهكذا تحصل على سبيل المثال: 
    bad option "-blah": should be one of -blocking,
    -buffering, -buffersize, -eofchar, -translation,
    -peername, or -sockname
عند استدعائها مع optionList يساوي "peername sockname".

"blah" هو الوسيط optionName و <specific options> هي قائمة كلمات محددة مفصولة بمسافات. تعتني الدالة بإدراج واصلات سلبية قبل كل خيار، وفواصل بعده، و "or" قبل الخيار الأخير.

OLD CHANNEL TYPES

يحتوي هيكل Tcl_ChannelType الأصلي (8.3.1 والإصدارات السابقة) على الحقول التالية:

typedef struct Tcl_ChannelType {
    const char *typeName;
    Tcl_DriverBlockModeProc *blockModeProc;
    Tcl_DriverCloseProc *closeProc;
    Tcl_DriverInputProc *inputProc;
    Tcl_DriverOutputProc *outputProc;
    Tcl_DriverSeekProc *seekProc;
    Tcl_DriverSetOptionProc *setOptionProc;
    Tcl_DriverGetOptionProc *getOptionProc;
    Tcl_DriverWatchProc *watchProc;
    Tcl_DriverGetHandleProc *getHandleProc;
    Tcl_DriverClose2Proc *close2Proc;
} Tcl_ChannelType;

على الرغم من أنه من الممكن إنشاء قنوات باستخدام الهيكل السابق، فإن الكود الداخلي للقناة سيحدد الإصدار. من الضروري استخدام هيكل Tcl_ChannelType الجديد إذا كنت تنشئ برنامج تشغيل قناة مكدسًا بسبب مشكلات في التنفيذ السابق للقنوات المكدسة (8.2.0 إلى 8.3.1).

قبل 8.4.0 (أي خلال الإصدارات المتأخرة من 8.3 وجزء مبكر من دورة تطوير 8.4) احتوى هيكل Tcl_ChannelType على الحقول التالية:

typedef struct Tcl_ChannelType {
    const char *typeName;
    Tcl_ChannelTypeVersion version;
    Tcl_DriverCloseProc *closeProc;
    Tcl_DriverInputProc *inputProc;
    Tcl_DriverOutputProc *outputProc;
    Tcl_DriverSeekProc *seekProc;
    Tcl_DriverSetOptionProc *setOptionProc;
    Tcl_DriverGetOptionProc *getOptionProc;
    Tcl_DriverWatchProc *watchProc;
    Tcl_DriverGetHandleProc *getHandleProc;
    Tcl_DriverClose2Proc *close2Proc;
    Tcl_DriverBlockModeProc *blockModeProc;
    Tcl_DriverFlushProc *flushProc;
    Tcl_DriverHandlerProc *handlerProc;
    Tcl_DriverTruncateProc *truncateProc;
} Tcl_ChannelType;

عند تسجيل الهيكل السابق كنوع قناة، يجب أن يكون الحقل version دائمًا TCL_CHANNEL_VERSION_2.
