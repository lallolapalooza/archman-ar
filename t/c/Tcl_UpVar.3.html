Tcl_UpVar(3)

الحزمة: tcl

التراخيص: TCL

Tcl_UpVar(3) إجراءات مكتبة Tcl Tcl_UpVar(3)

الاسم

Tcl_UpVar, Tcl_UpVar2 - ربط متغير بآخر

الملخص

#include <tcl.h>

int
Tcl_UpVar(interp, frameName, sourceName, destName, flags)

int
Tcl_UpVar2(interp, frameName, name1, name2, destName, flags)

المعاملات

Tcl_Interp *interp (في)
المترجم الذي يحتوي على المتغيرات؛ يستخدم أيضًا للإبلاغ عن الأخطاء.
const char *frameName (في)
تحدد إطار المكدس الذي يحتوي على المتغير المصدر. قد يكون له أي من الأشكال التي يقبلها الأمر upvar، مثل #0 أو 1.
const char *sourceName (في)
اسم المتغير المصدر، في الإطار المحدد بواسطة frameName. قد يشير إلى متغير قياسي أو متغير مصفوفة له فهرس بين قوسين.
const char *destName (في)
اسم المتغير الوجهة، والذي سيتم ربطه بالمتغير المصدر بحيث تشير المراجع إلى destName إلى المتغير الآخر. يجب ألا يكون موجودًا حاليًا إلا كمتغير مرتبط عبر upvar.
int flags (في)
واحد من TCL_GLOBAL_ONLY، TCL_NAMESPACE_ONLY أو 0؛ إذا كان غير صفري، فإن destName هو متغير عام أو متغير مساحة اسم؛ خلاف ذلك، فهو محلي للإجراء الحالي (أو مساحة الاسم الحالية إذا لم يكن هناك إجراء نشط).
const char *name1 (في)
الجزء الأول من اسم المتغير المصدر (اسم قياسي، أو اسم مصفوفة بدون فهرس المصفوفة).
const char *name2 (في)
إذا كان المتغير المصدر عنصرًا في مصفوفة، فإنه يعطي فهرس العنصر. بالنسبة للمتغيرات المصدر القياسية، يكون NULL.

الوصف

Tcl_UpVar و Tcl_UpVar2 يوفران نفس الوظائف التي يوفرها الأمر upvar: فهما ينشئان رابطًا من متغير مصدر إلى متغير وجهة، بحيث يتم تمرير المراجع إلى الوجهة بشفافية إلى المصدر. يمكن تحديد اسم المتغير المصدر إما كسلسلة واحدة مثل xyx أو a(24) (عن طريق استدعاء Tcl_UpVar) أو في جزأين حيث تم فصل اسم المصفوفة عن اسم العنصر (عن طريق استدعاء Tcl_UpVar2). يتم تحديد اسم متغير الوجهة في سلسلة واحدة؛ لا يمكن أن يكون عنصر مصفوفة.

تعيد كلتا الإجراءات إما TCL_OK أو TCL_ERROR، وتتركان رسالة خطأ في نتيجة المترجم إذا حدث خطأ.

كما هو الحال مع الأمر upvar، لا يلزم أن يكون المتغير المصدر موجودًا؛ إذا كان موجودًا، فإن إلغاء تعيينه لاحقًا لا يدمر الرابط. قد يكون متغير الوجهة موجودًا في وقت الاستدعاء، ولكن إذا كان كذلك، فيجب أن يوجد كمتغير مرتبط.
