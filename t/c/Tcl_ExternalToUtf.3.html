الوسائط

Tcl_Interp *interp (in)
المفسر المستخدم للإبلاغ عن الأخطاء، أو NULL إذا لم يتم الإبلاغ عن أي شيء.
const char *name (in)
اسم الترميز المطلوب تحميله.
Tcl_Encoding encoding (in)
الترميز المطلوب الاستعلام عنه، تحريره، أو استخدامه لتحويل النص. إذا كان encoding NULL، فسيتم استخدام الترميز الحالي للنظام.
Tcl_Obj *objPtr (in)
اسم الترميز للحصول على رمزه.
Tcl_Encoding *encodingPtr (out)
سيتم كتابة رمز الترميز هنا.
const char *src (in)
بالنسبة للإجراءات Tcl_ExternalToUtf، مصفوفة من البايتات بالترميز المحدد والتي سيتم تحويلها إلى UTF-8. بالنسبة للإجراءات Tcl_UtfToExternal و Tcl_WinUtfToTChar، مصفوفة من أحرف UTF-8 التي سيتم تحويلها إلى الترميز المحدد.
const TCHAR *tsrc (in)
مصفوفة من أحرف Windows TCHAR التي سيتم تحويلها إلى UTF-8.
int srcLen (in)
طول src أو tsrc بالبايتات. إذا كان الطول سالبًا، فسيتم استخدام الطول المحدد بواسطة الترميز.
Tcl_DString *dstPtr (out)
DString غير مهيأ أو تم تحريره سيتم تخزين النتيجة المحولة فيه.
int flags (in)
بتات العلم ORed. TCL_ENCODING_START يشير إلى أن مخزن الإدخال هو أول كتلة في تدفق (ربما متعدد الكتل) ويجب إعادة تشغيل روتين التحويل إلى حالة أولية وإجراء أي تهيئة ضرورية قبل تحويل البايت الأول. TCL_ENCODING_END يشير إلى أن مخزن الإدخال هو آخر كتلة في تدفق (ربما متعدد الكتل)، مما يشير إلى أن روتين التحويل يجب أن يجري أي إنهاء ضروري بعد تحويل البايت الأخير ثم إعادة تشغيل إلى حالة أولية. TCL_ENCODING_STOPONERROR يشير إلى أن روتين التحويل يجب أن يعود فورًا عند قراءة حرف إدخال غير موجود في الترميز الهدف؛ وإلا، سيتم استبدال حرف بديل افتراضي تلقائيًا.
Tcl_EncodingState *statePtr (in/out)
يتم استخدامه عند تحويل تدفق من البايتات (عادةً ما يكون طويلًا أو غير محدد) على أساس قطعة بقطعة. يخزن روتين التحويل حالته الحالية في *statePtr بعد تحويل src (المخزن الذي يحتوي على القطعة الحالية)؛ يجب تمرير معلومات الحالة هذه مرة أخرى عند تحويل القطعة التالية من التدفق حتى يعرف روتين التحويل الحالة التي كان عليها عند تركه في نهاية القطعة الأخيرة. قد يكون NULL، وفي هذه الحالة يتم تجاهل القيمة المحددة لـ flags ويُفترض أن مخزن الإدخال يحتوي على النص الكامل المراد تحويله.
char *dst (out)
المخزن الذي سيتم تخزين نتيجة التحويل فيه. لن يتم تخزين أكثر من dstLen بايتات في dst.
int dstLen (in)
الطول الأقصى للمخزن الهدف dst بالبايتات.
int *srcReadPtr (out)
سيتم ملؤه بعدد البايتات من src التي تم تحويلها فعليًا. قد يكون هذا أقل من الطول الأصلي لأحرف الإدخال إذا كانت هناك مشكلة في تحويل بعض أحرف الإدخال. قد يكون NULL.
int *dstWrotePtr (out)
سيتم ملؤه بعدد البايتات التي تم تخزينها في المخزن الهدف كنتيجة للتحويل. قد يكون NULL.
int *dstCharsPtr (out)
سيتم ملؤه بعدد الأحرف التي تتوافق مع عدد البايتات المخزنة في المخزن الهدف. قد يكون NULL.
Tcl_DString *bufPtr (out)
مساحة تخزين لاسم نظام الترميز المحدد.
const Tcl_EncodingType *typePtr (in)
بنية تحدد نوع ترميز جديد.
Tcl_Obj *searchPath (in)
قائمة بالمجلدات التي سيتم البحث فيها عن ملفات الترميز.
const char *path (in)
مسار إلى موقع ملف الترميز.

مقدمة

تقوم هذه الإجراءات بالتحويل بين تمثيل الأحرف الداخلي لـ Tcl، UTF-8، وتمثيلات الأحرف التي تستخدمها أنظمة تشغيل أو أنظمة ملفات مختلفة، مثل Unicode أو ASCII أو Shift-JIS. عند العمل على السلاسل النصية، مثل الحصول على أسماء الملفات أو عرض الأحرف باستخدام خطوط دولية، يجب ترجمة السلاسل النصية إلى تنسيق واحد أو أكثر يمكن لاستدعاءات النظام المختلفة استخدامه. على سبيل المثال، في عمل ياباني على Unix، قد يحصل المستخدم على اسم ملف ممثل في ترميز EUC-JP ثم يترجم الأحرف إلى ترميز jisx0208 لعرض اسم الملف في عنصر واجهة Tk. الغرض من حزمة الترميز هو المساعدة في سد فجوة الترجمة. يوفر UTF-8 نقطة وسط لجميع الترميزات المختلفة. في المثال أعلاه، سيتم ترجمة النص إلى UTF-8 من الترميز الذي يستخدمه نظام التشغيل، ثم سيتم ترجمته من UTF-8 إلى الترميز الذي تتطلبه إجراءات العرض.

تم تضمين بعض الترميزات الأساسية في Tcl. يمكن تعريف ترميزات أخرى بواسطة المستخدم أو تحميلها ديناميكيًا من ملفات الترميز بشكل مستقل عن المنصة.

وصف

Tcl_GetEncoding تبحث عن ترميز معين بالنظر إلى name الخاص به. يمكن للاسم الإشارة إلى ترميز Tcl معرف مسبقًا، أو ترميز معرف بواسطة المستخدم تم تسجيله باستخدام Tcl_CreateEncoding، أو ترميز قابل للتحميل ديناميكيًا. القيمة المرجعة هي رمز يمثل الترميز ويمكن استخدامه في استدعاءات لاحقة لإجراءات مثل Tcl_GetEncodingName و Tcl_FreeEncoding و Tcl_UtfToExternal. إذا لم يشير الاسم إلى أي ترميز معروف أو قابل للتحميل، فسيتم إرجاع NULL وسيتم إرجاع رسالة خطأ في interp.

تحتفظ حزمة الترميز ببنك للترميزات المستخدمة حاليًا. في المرة الأولى التي يتم فيها رؤية name، تقوم Tcl_GetEncoding بإرجاع ترميز بعداد مرجع يساوي 1. إذا تم طلب نفس name مرة أخرى، فسيتم زيادة عداد المراجع لهذا الترميز دون الحاجة إلى تخصيص ترميز جديد وهياكله المرتبطة به.

عندما لا يكون الترميز مطلوبًا بعد الآن، يجب استدعاء Tcl_FreeEncoding لتحريره. عندما لا يكون الترميز مستخدمًا بعد الآن (أي، تم تحريره بالقدر الذي تم الحصول عليه به)، ستقوم Tcl_FreeEncoding بتحرير جميع مساحات الذاكرة التي استخدمها الترميز وإزالة الترميز من قاعدة البيانات.

Tcl_GetEncodingFromObj تعالج تمثيل السلسلة النصية لـ objPtr كاسم ترميز وتجد ترميزًا بهذا الاسم، تمامًا مثل Tcl_GetEncoding. عندما يتم العثور على ترميز، يتم تخزينه مؤقتًا في قيمة objPtr للإشارات المستقبلية، ويتم كتابة رمز Tcl_Encoding في مساحة التخزين المؤقت.

هذا الموقع غير مرتبط بـ Arch Linux ولا تابع له بأي شكل من الأشكال. المحتوى هو ترجمة فقط لصفحات دليل Arch المنشورة على man.archlinux.org.

إلى encodingPtr، ويتم إرجاع القيمة TCL_OK. إذا لم يتم العثور على هذا الترميز، يتم إرجاع القيمة TCL_ERROR، ولا تتم أي كتابة إلى *encodingPtr. تمامًا كما هو الحال مع Tcl_GetEncoding، يجب على المستدعي استدعاء Tcl_FreeEncoding على رمز الترميز الناتج عندما لا يعود ذلك الرمز قيد الاستخدام.

تحوّل Tcl_ExternalToUtfDString مخزنًا مصدريًا src من encoding المحدد إلى UTF-8. تُخزَّن البايتات المحوَّلة في dstPtr، الذي يُنهى بعد ذلك بحرف null. يجب على المستدعي في النهاية استدعاء Tcl_DStringFree لتحرير أي معلومات مخزنة في dstPtr. عند التحويل، إذا تعذّر تمثيل أي من الأحرف في المخزن المصدر في الترميز الهدف، فسيُستخدم حرف احتياطي افتراضي. القيمة المُرجعة هي مؤشر إلى القيمة المخزنة في DString.

تحوّل Tcl_ExternalToUtf مخزنًا مصدريًا src من encoding المحدد إلى UTF-8. يُحوَّل ما يصل إلى srcLen بايت من المخزن المصدر وتُخزَّن ما يصل إلى dstLen بايت محوَّل في dst. في جميع الحالات، تُملأ *srcReadPtr بعدد البايتات التي حُوِّلت بنجاح من src وتُملأ *dstWrotePtr بالعدد المقابل من البايتات التي خُزِّنت في dst. القيمة المُرجعة هي إحدى القيم التالية:

TCL_OK
حُوِّلت جميع بايتات src.
TCL_CONVERT_NOSPACE
لم يكن المخزن الهدف كبيرًا بما يكفي لجميع البيانات المحوَّلة؛ ومع ذلك حُوِّل أكبر عدد ممكن من الأحرف التي يمكن أن تتسع.
TCL_CONVERT_MULTIBYTE
كانت البايتات القليلة الأخيرة في المخزن المصدر بداية تسلسل متعدد البايتات، لكن كانت هناك حاجة إلى مزيد من البايتات لإكمال هذا التسلسل. يجب أن يمرر الاستدعاء اللاحق لروتين التحويل مخزنًا يحتوي على البايتات غير المحوَّلة التي بقيت في src بالإضافة إلى بعض البايتات الإضافية من التدفق المصدر لتحويل تسلسل البايتات المتعددة المنقسم سابقًا بشكل صحيح.
TCL_CONVERT_SYNTAX
احتوى المخزن المصدر على تسلسل أحرف غير صالح. قد يحدث هذا إذا تضرر تدفق الإدخال أو إذا أُسيء تحديد طريقة ترميز الإدخال.
TCL_CONVERT_UNKNOWN
احتوى المخزن المصدر على حرف تعذّر تمثيله في الترميز الهدف وتم تحديد TCL_ENCODING_STOPONERROR.

تحوّل Tcl_UtfToExternalDString مخزنًا مصدريًا src من UTF-8 إلى encoding المحدد. تُخزَّن البايتات المحوَّلة في dstPtr، الذي يُنهى بعد ذلك بحرف null المناسب الخاص بالترميز. يجب على المستدعي في النهاية استدعاء Tcl_DStringFree لتحرير أي معلومات مخزنة في dstPtr. عند التحويل، إذا تعذّر تمثيل أي من الأحرف في المخزن المصدر في الترميز الهدف، فسيُستخدم حرف احتياطي افتراضي. القيمة المُرجعة هي مؤشر إلى القيمة المخزنة في DString.

تحوّل Tcl_UtfToExternal مخزنًا مصدريًا src من UTF-8 إلى encoding المحدد. يُحوَّل ما يصل إلى srcLen بايت من المخزن المصدر وتُخزَّن ما يصل إلى dstLen بايت محوَّل في dst. في جميع الحالات، تُملأ *srcReadPtr بعدد البايتات التي حُوِّلت بنجاح من src وتُملأ *dstWrotePtr بالعدد المقابل من البايتات التي خُزِّنت في dst. القيم المُرجعة هي نفس القيم المُرجعة لـ Tcl_ExternalToUtf.

Tcl_WinUtfToTChar وTcl_WinTCharToUtf هما دالتان مساعدتان خاصتان بنظام Windows فقط للتحويل بين سلاسل UTF-8 وسلاسل Windows بناءً على نوع TCHAR الذي هو بحكم العرف حرف Unicode على Windows NT.

Tcl_GetEncodingName هو تقريبًا عكس Tcl_GetEncoding. بالنظر إلى encoding، فإن القيمة المُرجعة هي وسيطة name التي استُخدمت لإنشاء الترميز. السلسلة التي تُرجعها Tcl_GetEncodingName مضمونة فقط بالاستمرار حتى يُحذف encoding. يجب على المستدعي عدم تعديل هذه السلسلة.

تضبط Tcl_SetSystemEncoding الترميز الافتراضي الذي ينبغي استخدامه كلما مرر المستخدم قيمة NULL لوسيطة encoding إلى أي من دوال الترميز الأخرى. إذا كانت name هي NULL، يُعاد ضبط ترميز النظام إلى ترميز النظام الافتراضي، binary. إذا لم يكن الاسم يشير إلى أي ترميز معروف أو قابل للتحميل، تُرجع TCL_ERROR وتُترك رسالة خطأ في interp. وإلا، فإن هذا الإجراء يزيد عدد مراجع ترميز النظام الجديد، ويُنقص عدد مراجع ترميز النظام القديم، ويُرجع TCL_OK.

توفّر Tcl_GetEncodingNameFromEnvironment وسيلة لمكتبة Tcl للإبلاغ عن اسم الترميز الذي تعتقد أنه الترميز الصحيح لاستخدامه كترميز النظام، بناءً على استدعاءات النظام وفحص البيئة المناسبة للمنصة. تقبل bufPtr، وهو مؤشر إلى Tcl_DString غير مهيأ أو محرَّر، وتكتب اسم الترميز فيه. تُرجع Tcl_DStringValue.

تضبط Tcl_GetEncodingNames نتيجة interp إلى قائمة تتكون من أسماء جميع الترميزات المعرَّفة حاليًا أو التي يمكن تحميلها ديناميكيًا، بالبحث في مسار الترميز المحدد بواسطة Tcl_SetDefaultEncodingDir. لا يضمن هذا الإجراء أن ملفات الترميز القابلة للتحميل ديناميكيًا تحتوي على بيانات صالحة، بل فقط أنها موجودة.

يعرّف Tcl_CreateEncoding ترميزًا جديدًا ويسجّل إجراءات C التي تُستدعى للتحويل بين الترميز وUTF-8. تصبح الترميزات التي ينشئها Tcl_CreateEncoding بعد ذلك مرئية في قاعدة البيانات التي يستخدمها Tcl_GetEncoding. تمامًا كما هو الحال مع إجراء Tcl_GetEncoding، فإن القيمة المُرجعة هي رمز يمثل الترميز ويمكن استخدامه في الاستدعاءات اللاحقة لدوال الترميز الأخرى. تُرجع Tcl_CreateEncoding ترميزًا بعدد مراجع يساوي 1. إذا كان هناك ترميز بالاسم name المحدد موجودًا بالفعل، فسيُستبدل مدخله في قاعدة البيانات بالترميز الجديد؛ سيظل رمز الترميز القديم صالحًا ويستمر في التصرف كما كان من قبل، لكن مستخدمي الرمز الجديد سيستدعون الآن إجراءات الترميز الجديدة.

تحتوي وسيطة typePtr لـ Tcl_CreateEncoding على معلومات حول اسم الترميز والإجراءات التي ستُستدعى للتحويل بين هذا الترميز وUTF-8. وهي معرَّفة كما يلي:

typedef struct Tcl_EncodingType {
    const char *encodingName;
    Tcl_EncodingConvertProc *toUtfProc;
    Tcl_EncodingConvertProc *fromUtfProc;
    Tcl_EncodingFreeProc *freeProc;
    ClientData clientData;
    int nullSize;
} Tcl_EncodingType;

توفّر encodingName اسمًا نصيًا للترميز، يمكن الإشارة إليه به في إجراءات أخرى مثل Tcl_GetEncoding. تشير toUtfProc إلى إجراء استدعاء عكسي لاستدعائه لتحويل النص من هذا الترميز إلى UTF-8. تشير fromUtfProc إلى إجراء استدعاء عكسي لاستدعائه لتحويل النص من UTF-8 إلى هذا الترميز. تشير freeProc إلى إجراء استدعاء عكسي لاستدعائه عند حذف هذا الترميز. قد يكون حقل freeProc هو NULL. تحتوي clientData على قيمة عشوائية من كلمة واحدة تُمرَّر إلى toUtfProc وfromUtfProc وfreeProc كلما استُدعيت. عادةً، يكون هذا مؤشرًا إلى بنية بيانات تحتوي على معلومات خاصة بالترميز يمكن أن تستخدمها إجراءات الاستدعاء العكسي. على سبيل المثال، قد يستخدم ترميزان متشابهان جدًا مثل ascii وmacRoman نفس إجراء الاستدعاء العكسي، لكن يستخدمان قيمًا مختلفة من clientData للتحكم في سلوكه. تحدد nullSize عدد بايتات الصفر التي تدل على نهاية السلسلة في هذا الترميز. يجب أن تكون 1 (للترميزات أحادية البايت أو متعددة البايتات مثل ASCII أو Shift-JIS) أو 2 (للترميزات ثنائية البايت مثل Unicode). لا تُقبل الترميزات ذات الحجم الثابت التي تحتوي على 3 بايتات أو أكثر لكل حرف (مثل CNS11643).

يجب أن تطابق إجراءات الاستدعاء العكسي toUtfProc وfromUtfProc النوع Tcl_EncodingConvertProc:

typedef int Tcl_EncodingConvertProc(
        ClientData clientData,
        const char *src,
        int srcLen,
        int flags,
        Tcl_EncodingState *statePtr,
        char *dst,
        int dstLen,
        int *srcReadPtr,
        int *dstWrotePtr,
        int *dstCharsPtr);

تُستدعى إجراءات toUtfProc وfromUtfProc بواسطة عائلة دوال Tcl_ExternalToUtf أو Tcl_UtfToExternal لتنفيذ التحويل الفعلي. معامل clientData لهذه الإجراءات هو نفسه حقل clientData المحدد لـ Tcl_CreateEncoding عند إنشاء الترميز. أما الوسائط المتبقية لإجراءات الاستدعاء العكسي فهي نفس الوسائط الموثقة في الأعلى لـ Tcl_ExternalToUtf أو Tcl_UtfToExternal، مع الاستثناءات التالية. إذا كانت وسيطة srcLen لإحدى تلك الدوال عالية المستوى سالبة، فستكون القيمة الممررة إلى إجراء الاستدعاء العكسي هي طول السلسلة المناسب الخاص بالترميز لـ src. إذا كانت أي من وسائط srcReadPtr أو dstWrotePtr أو dstCharsPtr لإحدى الدوال عالية المستوى هي NULL، فستكون القيمة المقابلة الممررة إلى إجراء الاستدعاء العكسي موقعًا غير NULL.

يجب أن يطابق إجراء الاستدعاء العكسي freeProc، إذا لم يكن NULL، النوع Tcl_EncodingFreeProc:

typedef void Tcl_EncodingFreeProc(
        ClientData clientData);

تُستدعى دالة freeProc هذه عند حذف الترميز. معامل clientData هو نفسه حقل clientData المحدد لـ Tcl_CreateEncoding عند إنشاء الترميز.

تُستدعى Tcl_GetEncodingSearchPath وTcl_SetEncodingSearchPath للوصول إلى قائمة أدلة نظام الملفات التي يُبحث فيها عن ملفات بيانات الترميز وتعيينها.

القيمة التي تُرجعها Tcl_GetEncodingSearchPath هي القيمة المخزنة بواسطة آخر استدعاء ناجح لـ Tcl_SetEncodingSearchPath. إذا لم يحدث أي استدعاء لـ Tcl_SetEncodingSearchPath، فستحسب Tcl قيمة أولية بناءً على البيئة. يوجد مسار بحث واحد للترميز للعملية بأكملها، تشترك فيه جميع الخيوط في العملية.

تخزّن Tcl_SetEncodingSearchPath searchPath وتُرجع TCL_OK، ما لم تكن searchPath قائمة Tcl صالحة، مما يؤدي إلى إرجاع TCL_ERROR. عناصر searchPath
