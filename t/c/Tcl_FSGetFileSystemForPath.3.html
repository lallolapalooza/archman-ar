الحجج

const Tcl_Filesystem *fsPtr (in)
يشير إلى بنية تحتوي على عناوين الإجراءات التي يمكن استدعاؤها لأداء عمليات نظام الملفات المختلفة.
Tcl_Obj *pathPtr (in)
يتم استخدام المسار الذي تمثله هذه القيمة للعملية المعنية. إذا لم تكن للقيمة تمثيل مسار داخلي بالفعل، فسيتم تحويلها ليكون لديها واحد.
Tcl_Obj *srcPathPtr (in)
كما هو الحال بالنسبة لـ pathPtr ، ولكن يتم استخدامه للملف المصدر لعملية نسخ أو إعادة تسمية.
Tcl_Obj *destPathPtr (in)
كما هو الحال بالنسبة لـ pathPtr ، ولكن يتم استخدامه لاسم الملف الوجهة لعملية نسخ أو إعادة تسمية.
int recursive (in)
ما إذا كان سيتم إزالة الدلائل الفرعية ومحتوياتها أيضًا.
const char *encodingName (in)
ترميز البيانات المخزنة في الملف المحدد بواسطة pathPtr وسيتم تقييمه.
const char *pattern (in)
سيتم إرجاع الملفات أو الدلائل التي تطابق هذا النمط فقط.
Tcl_GlobTypeData *types (in)
سيتم إرجاع الملفات أو الدلائل التي تطابق أوصاف النوع الموجودة في هذه البنية فقط. قد تكون هذه الوسيطة NULL.
Tcl_Interp *interp (in)
المترجم لاستخدامه إما للنتائج أو التقييم أو الإبلاغ عن رسائل الخطأ.
void *clientData (in)
الوصف الأصلي لقيمة المسار المراد إنشاؤه.
Tcl_Obj *firstPtr (in)
القيمة الأولى من قيمتي مسار للمقارنة. قد يتم تحويل القيمة إلى نوع مسار.
Tcl_Obj *secondPtr (in)
القيمة الثانية من قيمتي مسار للمقارنة. قد يتم تحويل القيمة إلى نوع مسار.
Tcl_Obj *listObj (in)
قائمة عناصر المسار للعمل عليها مع عملية ضم.
int elements (in)
عدد العناصر في listObj التي يجب ضمها معًا. إذا كانت سالبة، فسيتم ضم جميع العناصر.
Tcl_Obj **errorPtr (out)
في حالة حدوث خطأ، يتم ملؤها بقيمة تحتوي على اسم الملف الذي تسبب في حدوث خطأ في عمليات النسخ/إعادة التسمية المختلفة.
int index (in)
فهرس السمة المعنية.
Tcl_Obj *objPtr (in)
القيمة المراد تعيينها في العملية.
Tcl_Obj **objPtrRef (out)
يتم ملؤها بقيمة تحتوي على نتيجة العملية.
int mode (in)
قناع يتكون من واحد أو أكثر من R_OK، W_OK، X_OK و F_OK. تطلب R_OK، W_OK و X_OK التحقق مما إذا كان الملف موجودًا وله أذونات قراءة أو كتابة أو تنفيذ، على التوالي. يطلب F_OK فقط التحقق من وجود الملف.
Tcl_StatBuf *statPtr (out)
البنية التي تحتوي على نتيجة عملية stat أو lstat.
const char *sym1 (in)
اسم إجراء للبحث عنه في جدول رموز الملف
const char *sym2 (in)
اسم إجراء للبحث عنه في جدول رموز الملف
Tcl_PackageInitProc **proc1Ptr (out)
يتم ملؤها بدالة التهيئة لهذا الكود.
Tcl_PackageInitProc **proc2Ptr (out)
يتم ملؤها بدالة التهيئة الآمنة لهذا الكود.
void **clientDataPtr (out)
يتم ملؤها بقيمة clientData التي سيتم تمريرها إلى دالة إلغاء تحميل هذا الكود عند استدعائها.
Tcl_LoadHandle *loadHandlePtr (out)
يتم ملؤها برمز مجرد يمثل الملف المحمل.
Tcl_FSUnloadFileProc **unloadProcPtr (out)
يتم ملؤها بالدالة التي يجب استخدامها لإلغاء تحميل هذا الجزء من الكود.
Tcl_LoadHandle loadHandle (in)
مقبض المكتبة المحملة المراد إلغاء تحميلها.
utimbuf *tval (in)
تتم قراءة أوقات الوصول والتعديل في هذه البنية واستخدامها لتعيين تلك القيم لملف معين.
const char *modeString (in)
يحدد كيفية الوصول إلى الملف. قد يكون له أي من القيم المسموح بها لـ mode الوسيطة لأمر Tcl open.
int permissions (in)
علامات الإذن على نمط POSIX مثل 0644. إذا تم إنشاء ملف جديد، فسيتم تعيين هذه الأذونات على c

هيكل *Tcl_StatBuf* معلومات حول الجهاز، inode (دائمًا 0 على Windows)، وضع الامتياز، nlink (دائمًا 1 على Windows)، معرف المستخدم (دائمًا 0 على Windows)، معرف المجموعة (دائمًا 0 على Windows)، rdev (نفسه مثل الجهاز على Windows)، الحجم، وقت آخر وصول، وقت آخر تعديل، ووقت آخر تغيير بيانات وصفية. راجع **PORTABLE STAT RESULT API** لوصف كيفية كتابة كود محمول لتخصيص والوصول إلى هيكل *Tcl_StatBuf*. إذا كان *path* موجودًا، فإن **Tcl_FSStat** تعيد 0 ويتم ملء هيكل الإحصائيات بالبيانات. وإلا، يتم إرجاع -1، ولا يتم تقديم أي معلومات إحصائية. **Tcl_FSOpenFileChannel** تفتح ملفًا محددًا بواسطة *pathPtr* وتعيد مؤشر قناة يمكن استخدامه لإجراء عمليات الإدخال والإخراج على الملف. تستند واجهة برمجة التطبيقات هذه إلى إجراء **fopen** لمكتبة الإدخال والإخراج القياسية في Unix. تركيب ومعنى جميع الوسائط مشابهة لتلك المقدمة في أمر Tcl **open** عند فتح ملف. إذا حدث خطأ أثناء فتح القناة، تعيد **Tcl_FSOpenFileChannel** NULL وتسجل رمز خطأ POSIX يمكن استرجاعه باستخدام **Tcl_GetErrno**. بالإضافة إلى ذلك، إذا كان *interp* غير NULL، فإن **Tcl_FSOpenFileChannel** تترك رسالة خطأ في نتيجة *interp* بعد أي خطأ. لا يتم تسجيل القناة التي تم إنشاؤها حديثًا في المفسر المقدم؛ لتسجيلها، استخدم **Tcl_RegisterChannel**. إذا تم إغلاق إحدى القنوات القياسية، **stdin** أو **stdout** أو **stderr**، في وقت سابق، فإن فعل إنشاء القناة الجديدة يعينها أيضًا كبديل للقناة القياسية. **Tcl_FSGetCwd** تستبدل الإصدار الخاص بالمكتبة من **getcwd**. هذه تعيد دليل العمل الحالي لمكتبة Tcl. قد يختلف هذا عن دليل العمل الحالي للنظام الأساسي، وهو ما يحدث عندما لا يكون دليل العمل الحالي في نظام الملفات الأصلي. النتيجة هي مؤشر إلى Tcl_Obj يحدد الدليل الحالي، أو NULL إذا تعذر تحديد الدليل الحالي. إذا تم إرجاع NULL، يتم ترك رسالة خطأ في نتيجة *interp*. تمت زيادة عدد المرجع بالفعل للنتيجة للمتصل. عندما لم تعد هناك حاجة إليها، يجب تقليل عدد المرجع هذا. هذا ضروري لأغراض الأمان فيما يتعلق بالخيوط، للسماح لعدة خيوط بالوصول إلى هذه الدوال والدوال ذات الصلة، مع ضمان أن النتائج صالحة دائمًا. **Tcl_FSChdir** تستبدل الإصدار الخاص بالمكتبة من **chdir**. يتم تطبيع المسار ثم تمريره إلى نظام الملفات الذي يدعي ذلك. إذا لم يقم نظام الملفات هذا بتنفيذ هذه الدالة، فستعود Tcl إلى مزيج من **stat** و **access** للتحقق مما إذا كان الدليل موجودًا ولديه الأذونات المناسبة. بالنسبة للنتائج، انظر وثائق **chdir**. إذا نجحت، نحتفظ بسجل للمسار الناجح في *cwdPathPtr* لاستدعاءات **Tcl_FSGetCwd** اللاحقة. **Tcl_FSPathSeparator** تعيد حرف الفاصل الذي يجب استخدامه للعنصر الأكثر تحديدًا للمسار المحدد بواسطة *pathPtr* (أي، الجزء الأخير من المسار). يتم إرجاع الفاصل كـ Tcl_Obj يحتوي على سلسلة نصية بطول 1. إذا كان المسار غير صالح، يتم إرجاع NULL. **Tcl_FSJoinPath** تأخذ Tcl_Obj المحدد، والذي يجب أن يكون قائمة صالحة (يُسمح لها بأن يكون عدد مرجعها صفرًا)، وتعيد قيمة المسار المعطاة بالنظر إلى أول *elements* من العناصر كأجزاء مسار صالحة (قد يكون كل جزء مسار مسارًا كاملاً، أو مسارًا جزئيًا، أو مجرد دليل واحد محتمل أو اسم ملف). إذا كان أي جزء مسار هو بالفعل مسار مطلق، فسيتم تجاهل جميع أجزاء المسار السابقة. إذا كان *elements* أقل من 0، نستخدم القائمة بأكملها. من الممكن أن تكون القيمة المرجعة عنصرًا فعليًا من القائمة المعطاة، لذلك يجب على المتصل توخي الحذر لزيادة عدد مرجع النتيجة قبل تحرير القائمة. القيمة المرجعة، عادةً مع عدد مرجع صفر (ولكن يمكن مشاركتها تحت بعض الظروف)، تحتوي على المسار المدمج. يجب على المتصل إضافة عدد مرجع إلى القيمة قبل استخدامها. على وجه الخصوص، يمكن أن تكون القيمة المرجعة عنصرًا من القائمة المعطاة، لذلك قد يؤدي تحرير القائمة إلى تحرير القيمة قبل الأوان إذا لم يتم أخذ عدد مرجع. إذا كان عدد العناصر صفرًا، فإن القيمة المرجعة ستكون Tcl_Obj بسلسلة نصية فارغة. **Tcl_FSSplitPath** تأخذ Tcl_Obj المحدد، والذي يجب أن يكون مسارًا صالحًا، وتعيد قيمة قائمة Tcl تحتوي على كل جزء من هذا المسار كعنصر. تعيد قائمة بقيمة مرجع صفر. إذا كان *lenPtr* المعطى غير NULL، فسيتم تحديث المتغير الذي يشير إليه ليحتوي على عدد العناصر في القائمة المرجعة. **Tcl_FSEqualPaths** تختبر ما إذا كان المساران المعطيان يمثلان نفس كائن نظام الملفات. تعيد 1 إذا كانت المسارات متساوية، و 0 إذا كانت مختلفة. إذا كان أي من المسارين NULL، يتم دائمًا إرجاع 0. **Tcl_FSGetNormalizedPath** تحاول استخراج تمثيل مسار قياسي فريد من Tcl_Obj المحدد، والذي يمكن استخدام قيمته النصية كمعرف فريد للملف. تعيد قيمة المسار القياسي، مملوكة بواسطة Tcl، أو NULL إذا كان المسار غير صالح أو تعذر تحويله بنجاح. استخراج المسارات المطلقة والقياسية فعال للغاية (لأن نظام الملفات يعمل على هذه التمثيلات داخليًا)، على الرغم من أن النتيجة عندما يحتوي نظام الملفات على العديد من الروابط الرمزية قد لا تكون النسخة الأكثر سهولة في الاستخدام للمسار. القيمة المرجعة مملوكة بواسطة Tcl ولها عمر مكافئ لعمر *pathPtr* المعطى (ما لم يكن هذا مسارًا نسبيًا، وفي هذه الحالة قد يتم تحرير قيمة المسار القياسي في أي وقت يتغير فيه cwd) - يمكن للمتصل بالطبع زيادة عدد المرجع إذا رغب في الاحتفاظ بنسخة لفترة أطول. **Tcl_FSJoinToPath** تأخذ القيمة المعطاة، والتي يجب عادةً أن تكون مسارًا صالحًا أو NULL، وتضيف إليها مصفوفة أجزاء المسار المعطاة. تعيد قيمة، عادةً مع عدد مرجع صفر (ولكن يمكن مشاركتها تحت بعض الظروف)، تحتوي على المسار المدمج. يجب على المتصل إضافة عدد مرجع إلى القيمة قبل استخدامها. إذا كان لأي من القيم المقدمة لهذه الدالة (*pathPtr* أو عناصر *path*) عدد مرجع صفر، فسيتم تحريرها عند عودة هذه الدالة. **Tcl_FSConvertToPathType** تحاول تحويل Tcl_Obj المحدد إلى نوع مسار Tcl صالح، مع مراعاة حقيقة أن cwd قد تغير حتى لو كان هذا القيمة يفترض بالفعل أنها من النوع الصحيح. قد يبدأ اسم الملف بـ "~" (للإشارة إلى الدليل الرئيسي للمستخدم الحالي) أو "~<user>" (للإشارة إلى دليل رئيسي لأي مستخدم). إذا نجح التحويل (أي، القيمة هي مسار صالح في أحد أنظمة الملفات الحالية)، فسيتم إرجاع **TCL_OK**. وإلا، سيتم إرجاع **TCL_ERROR**، وقد يتم ترك رسالة خطأ في المفسر. **Tcl_FSGetInternalRep** تستخرج التمثيل الداخلي لقيمة مسار معين، في نظام الملفات المحدد. إذا كانت قيمة المسار تنتمي إلى نظام ملفات مختلف، فإننا نعيد NULL. إذا كان التمثيل الداخلي NULL حاليًا، فإننا نحاول إنشائه، عن طريق استدعاء **Tcl_FSCreateInternalRepProc** الخاص بنظام الملفات. تعيد NULL أو تمثيل مسار داخلي صالح. يتم تخزين هذا التمثيل الداخلي مؤقتًا، بحيث لا تتطلب الاستدعاءات المتكررة لهذه الدالة تحويلات إضافية. **Tcl_FSGetTranslatedPath** تحاول استخراج المسار المترجم من Tcl_Obj المحدد. إذا نجح الترجمة (أي، القيمة هي مسار صالح)، فسيتم إرجاعها. وإلا، سيتم إرجاع NULL، وقد يتم ترك رسالة خطأ في المفسر. المسار "المترجم" هو المسار الذي لا يحتوي على تسلسلات "~" أو "~user" (تم توسيعها إلى تمثيلها الحالي في نظام الملفات). القيمة المرجعة مملوكة للمتصل، والذي يجب تخزينها أو استدعاء **Tcl_DecrRefCount** لضمان تحرير الذاكرة. هذه الدالة ذات فائدة عملية قليلة، وعادة ما تكون **Tcl_FSGetNormalizedPath** أو **Tcl_FSGetNativePath** دوال أفضل للاستخدام لمعظم الأغراض. **Tcl_FSGetTranslatedStringPath** تفعل نفس الشيء مثل **Tcl_FSGetTranslatedPath**، ولكنها تعيد سلسلة نصية أو NULL. السلسلة النصية المرجعة هي ديناميكية

البنية التحتية لنظام الملفات

تحتوي هذه الحقول على معلومات أساسية حول بنية نظام الملفات وعناوين الوظائف المستخدمة لربط نظام ملفات معين بمسار ملف، والتعامل مع التمثيل الداخلي لمسارات الملفات، على سبيل المثال، نسخ هذه التمثيلات وتحريرها.

اسم النوع

يحتوي الحقل typeName على سلسلة نصية تنتهي بحرف null تحدد نوع نظام الملفات المنفذ، على سبيل المثال، native أو zip أو vfs.

طول البنية

عادةً ما يتم تنفيذ الحقل structureLength كـ sizeof(Tcl_Filesystem)، وهو موجود للسماح بتوافق أسهل مع الإصدارات الثنائية في حالة تغيير حجم البنية في إصدار مستقبلي من Tcl.

الإصدار

يجب تعيين الحقل version على TCL_FILESYSTEM_VERSION_1.

وظيفة المسار في نظام الملفات

يحتوي الحقل pathInFilesystemProc على عنوان وظيفة يتم استدعاؤها لتحديد ما إذا كانت قيمة المسار المعطاة تنتمي إلى نظام الملفات هذا أم لا. ستستدعي Tcl فقط الوظائف المتبقية لنظام الملفات مع مسار تعيد فيه هذه الوظيفة TCL_OK. إذا كان المسار لا ينتمي، فيجب إرجاع -1 (سلوك Tcl لأي قيمة إرجاع أخرى غير محدد). إذا تم إرجاع TCL_OK، فيمكن استخدام المعلمة الاختيارية للإخراج clientDataPtr لإرجاع تمثيل داخلي (خاص بنظام الملفات) للمسار، والذي سيتم تخزينه داخل قيمة المسار ويمكن استرداده بكفاءة بواسطة وظائف أنظمة الملفات الأخرى من أنظمة أخرى. ستقوم Tcl بتخزين معلومات تفيد بأن هذا المسار ينتمي إلى نظام الملفات هذا في ذاكرة التخزين المؤقت في نفس الوقت. يتم إلغاء صلاحية ذاكرة التخزين المؤقت هذه عند إضافة أو إزالة هياكل نظام الملفات في القائمة الداخلية لأنظمة الملفات المعروفة لدى Tcl.

typedef int Tcl_FSPathInFilesystemProc(
        Tcl_Obj *pathPtr,
        ClientData *clientDataPtr);

وظيفة نسخ التمثيل الداخلي

تنشئ هذه الوظيفة نسخة من التمثيل الداخلي للمسار ويتم استدعاؤها عندما تحتاج Tcl إلى تكرار قيمة المسار. إذا كانت NULL، فلن تقوم Tcl بنسخ التمثيل الداخلي، والذي قد يحتاج إلى إعادة إنشائه لاحقًا.

typedef ClientData Tcl_FSDupInternalRepProc(
        ClientData clientData);

وظيفة تحرير التمثيل الداخلي

تحرر التمثيل الداخلي. يجب تنفيذها إذا كان تحرير التمثيلات الداخلية مطلوبًا (أي، إذا تم حجز ذاكرة عند إنشاء تمثيل داخلي)، ولكن يمكن أن تكون NULL في حالات أخرى.

typedef void Tcl_FSFreeInternalRepProc(
        ClientData clientData);

وظيفة التحويل من التمثيل الداخلي إلى التمثيل المعياري

وظيفة لتحويل تمثيل داخلي إلى مسار معياري. مطلوبة فقط إذا كان نظام الملفات ينشئ قيم مسار نقية بدون تمثيل سلسلة أو مسار. القيمة المرجعة هي قيمة Tcl التي يكون تمثيل السلسلة الخاص بها هو المسار المعياري.

typedef Tcl_Obj *Tcl_FSInternalToNormalizedProc(
        ClientData clientData);

وظيفة إنشاء التمثيل الداخلي

وظيفة لأخذ قيمة مسار وحساب تمثيل داخلي لها، وتخزين هذا التمثيل الأصلي في القيمة. يمكن أن تكون NULL إذا لم يكن هناك تمثيل داخلي للمسارات، أو إذا كانت Tcl_FSPathInFilesystemProc لنظام الملفات هذا تنشئ دائمًا تمثيلًا داخليًا فورًا للمسارات التي تقبلها.

typedef ClientData Tcl_FSCreateInternalRepProc(

وظيفة توحيد المسار

وظيفة لتوحيد المسار. يجب تنفيذها لجميع أنظمة الملفات التي قد تحتوي على تمثيلات متعددة للسلسلة لنفس المسار. في Tcl، يجب أن يكون لكل مسار تمثيل فريد واحد موحد للسلسلة. اعتمادًا على نظام الملفات، قد يكون هناك أكثر من تمثيل سلسلة واحد غير موحد يشير إلى نفس المسار (على سبيل المثال، مسار نسبي، مسار بأحرف كبيرة مختلفة إذا كان نظام الملفات غير حساس لحالة الأحرف، مسار يحتوي على مرجع لمجلد رئيسي مثل ~، مسار يحتوي على روابط رمزية، وما إلى ذلك). إذا كان المكون الأخير للمسار رابطًا رمزيًا، فلا يجب تحويله إلى القيمة التي يشير إليها (ولكن يجب أن تكون حالته أو جوانبه الأخرى فريدة). يجب تحويل جميع المكونات الأخرى للمسار من الروابط الرمزية. هذا الاستثناء مطلوب لمحاذاة سلوك Tcl لعمليات file delete و file rename و file copy مع الروابط الرمزية. يمكن استدعاء هذه الوظيفة مع nextCheckpoint في بداية المسار (أي، صفر)، أو في نهاية المسار، أو في أي نقطة فاصلة للملف في المسار. لن يتم الإشارة إلى أي موضع عشوائي آخر في المسار. في الحالة الأخيرة من الحالات الثلاث الصالحة، يمكن للمنفذ افتراض أن المسار حتى الفاصلة للملف وشاملة معروف ومعياري.

typedef int Tcl_FSNormalizePathProc(
        Tcl_Interp *interp,
        Tcl_Obj *pathPtr,
        int nextCheckpoint);

عمليات نظام الملفات

تحتوي حقول هذا القسم من البنية على عناوين الوظائف التي يتم استدعاؤها لتنفيذ عمليات نظام الملفات الأساسية. يجب على نظام الملفات الذي يتوقع استخدامه مع مجموعة الأوامر الكاملة لـ Tcl تنفيذ كل هذه العمليات. إذا لم يتم تنفيذ بعضها، فقد تفشل أوامر Tcl معينة عند العمل على مسارات داخل نظام الملفات هذا. ومع ذلك، في بعض الحالات قد يكون هذا مرغوبًا (على سبيل المثال، يجب ألا ينفذ نظام ملفات للقراءة فقط الوظائف الأربع الأخيرة، ويجب ألا ينفذ نظام ملفات لا يدعم الروابط الرمزية وظيفة readlink، إلخ. تتوقع نواة Tcl أن تتصرف أنظمة الملفات بهذه الطريقة).

وظيفة نوع مسار نظام الملفات

وظيفة لتحديد نوع المسار في نظام الملفات هذا. يمكن أن تكون NULL، وفي هذه الحالة لن تتوفر معلومات النوع لمستخدمي نظام الملفات. يستخدم type لأغراض إعلامية فقط ويجب إرجاعه كتمثيل سلسلة لقيمة Tcl_Obj المرجعة. قد تكون قيمة إرجاع نموذجية networked أو zip أو ftp. نتيجة Tcl_Obj مملوكة بالكامل لنظام الملفات، لذا ستقوم Tcl بزيادة عداد المراجع لهذه القيمة إذا رغبت في ذلك.

typedef Tcl_Obj *Tcl_FSFilesystemPathTypeProc(
        Tcl_Obj *pathPtr);

وظيفة فاصل نظام الملفات

وظيفة لإرجاع حرف الفاصل لنظام الملفات هذا. يجب تنفيذ هذا فقط إذا كان نظام الملفات يرغب في استخدام فاصل مختلف عن الفاصل القياسي /. من بين الاستخدامات الأخرى، يتم إرجاعه بواسطة الأمر file separator. يجب أن يكون للقيمة المرجعة عداد مراجع صفر.

typedef Tcl_Obj *Tcl_FSFilesystemSeparatorProc(
        Tcl_Obj *pathPtr);

وظيفة إحصائيات

وظيفة لمعالجة استدعاء Tcl_FSStat. يجب تنفيذها لأي نظام ملفات معقول، حيث تعتمد العديد من أوامر مستوى Tcl عليها بشكل حاسم (على سبيل المثال، file atime و file isdirectory و file size و glob).

typedef int Tcl_FSStatProc(
        Tcl_Obj *pathPtr,
        Tcl_StatBuf *statPtr);

تقوم وظيفة Tcl_FSStatProc بملء البنية statPtr بمعلومات حول الملف المحدد. لا يلزم حقوق الوصول إلى الملف للحصول على هذه المعلومات، ولكن يلزم حقوق البحث لجميع الدلائل المذكورة في المسار الذي يؤدي إلى الملف. تتضمن البنية stat معلومات حول الجهاز، و inode (دائمًا 0 في Windows)، ووضع الامتيازات، وعدد الروابط (دائمًا 1 في Windows)، و معرف المستخدم (دائمًا 0 في Windows)، و معرف المجموعة (دائمًا 0 في Windows)، و rdev (نفس الجهاز في Windows)، والحجم، وتاريخ آخر وصول، وتاريخ آخر تعديل، وتاريخ آخر تغيير للبيانات الوصفية.

إذا كان الملف الذي تمثله pathPtr موجودًا، فترجع Tcl_FSStatProc 0 ويتم ملء البنية statPtr بالبيانات. بخلاف ذلك، يتم إرجاع -1 ولا يتم تقديم أي بيانات إحصائية.

وظيفة الوصول

وظيفة لمعالجة استدعاء Tcl_FSAccess. يجب تنفيذها لأي نظام ملفات معقول، حيث تعتمد العديد من أوامر مستوى Tcl عليها بشكل حاسم (على سبيل المثال، file exists و file readable).

typedef int Tcl_FSAccessProc(
        Tcl_Obj *pathPtr,
        int mode);

تتحقق وظيفة Tcl_FSAccessProc مما إذا كان سيُسمح للعملية بالقراءة أو الكتابة أو اختبار وجود الملف (أو كائن نظام ملفات آخر) الذي يحمل الاسم في pathPtr. إذا كان المسار يشير إلى رابط رمزي، فيجب اختبار أذونات الملف الذي يشير إليه هذا الرابط الرمزي.

في حالة النجاح (تم منح جميع الأذونات المطلوبة)، يتم إرجاع 0. في حالة حدوث خطأ (طلب أحد البتات في mode إذنًا لا يمكن منحه، أو حدث خطأ آخر)، يتم إرجاع -1.

وظيفة فتح قناة ملف

وظيفة لمعالجة استدعاء Tcl_FSOpenFileChannel. يجب تنفيذها لأي نظام ملفات معقول، حيث تعتمد العديد من أوامر مستوى Tcl عليها بشكل حاسم (على سبيل المثال، file open و file close).

typedef int Tcl_FSOpenFileChannelProc(
        Tcl_Obj *pathPtr,
        int mode);
