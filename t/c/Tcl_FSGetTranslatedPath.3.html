Tcl_FSOpenFileChannelProc

وظيفة لمعالجة استدعاء Tcl_FSOpenFileChannel. يجب أن يتم تنفيذها لأي نظام ملفات معقول، حيث أن أي عمليات تتطلب فتح أو الوصول إلى محتويات ملف ستستخدمها (مثل open، encoding، والعديد من أوامر Tk).

typedef Tcl_Channel Tcl_FSOpenFileChannelProc(
        Tcl_Interp *interp,
        Tcl_Obj *pathPtr,
        int mode,
        int permissions);

يقوم Tcl_FSOpenFileChannelProc بفتح ملف محدد بواسطة pathPtr ويعيد مقبض قناة يمكن استخدامه لإجراء الإدخال والإخراج على الملف. هذه الواجهة تمثل نموذجًا لإجراء fopen لمكتبة الإدخال والإخراج القياسية في Unix. بناء الجملة ومعنى جميع الوسائط مشابهة لتلك المقدمة في أمر Tcl open عند فتح ملف، حيث يكون الوسيط mode مزيجًا من علامات POSIX O_RDONLY، O_WRONLY، إلخ. إذا حدث خطأ أثناء فتح القناة، فإن Tcl_FSOpenFileChannelProc يعيد NULL ويسجل رمز خطأ POSIX يمكن استرجاعه باستخدام Tcl_GetErrno. بالإضافة إلى ذلك، إذا كان interp غير NULL، فإن Tcl_FSOpenFileChannelProc يترك رسالة خطأ في نتيجة interp بعد أي خطأ.

يجب ألا يتم تسجيل القناة التي تم إنشاؤها حديثًا في المفسر المقدم بواسطة Tcl_FSOpenFileChannelProc؛ هذه المهمة تقع على عاتق المستدعي لـ Tcl_FSOpenFileChannel (إذا لزم الأمر). إذا تم إغلاق إحدى القنوات القياسية، stdin، stdout أو stderr سابقًا، فإن إجراء إنشاء القناة الجديدة يعينها أيضًا كبديل للقناة القياسية.

MATCHINDIRECTORYPROC

وظيفة لمعالجة استدعاء Tcl_FSMatchInDirectory. إذا لم يتم تنفيذها، فستفتقر وظائف glob والنسخ المتكرر في نظام الملفات (وقد يؤثر هذا على أوامر مثل encoding names التي تستخدم وظائف glob داخليًا).

typedef int Tcl_FSMatchInDirectoryProc(
        Tcl_Interp *interp,
        Tcl_Obj *resultPtr,
        Tcl_Obj *pathPtr,
        const char *pattern,
        Tcl_GlobTypeData *types);

يجب أن تعيد الوظيفة جميع الملفات أو الدلائل (أو كائنات نظام الملفات الأخرى) التي تطابق النمط المحدد وتتوافق مع مواصفات types المعطاة. هناك طريقتان يمكن استدعاء هذه الوظيفة بهما. إذا كان pattern هو NULL، فإن pathPtr هو مواصفات مسار كامل لملف أو دليل واحد يجب التحقق من وجوده ونوعه الصحيح. بخلاف ذلك، pathPtr هو دليل، والذي يجب على الوظيفة البحث في محتوياته عن ملفات أو دلائل ذات نوع صحيح. في كلتا الحالتين، يمكن افتراض أن pathPtr ليس NULL وغير فارغ. لا توثق حاليًا ما إذا كان pathPtr سيحتوي على فاصل ملف في نهايته أم لا، لذلك يجب أن يكون الكود مرنًا لكلا الاحتمالين.

قيمة الإرجاع هي نتيجة Tcl قياسية تشير إلى ما إذا كان هناك خطأ قد حدث أثناء عملية المطابقة. يتم وضع رسائل الخطأ في interp، ما لم يكن interp NULL وفي هذه الحالة لا يلزم إنشاء رسالة خطأ؛ عند نتيجة TCL_OK، يجب إضافة النتائج إلى قيمة resultPtr المعطاة (والتي يمكن افتراض أنها قائمة Tcl صالحة غير مشتركة). يجب أن تتضمن التطابقات المضافة إلى resultPtr أي بادئة مسار معطاة في pathPtr (وهذا يعني عادةً أنها ستكون مواصفات مسار مطلقة). لاحظ أنه إذا لم يتم العثور على أي تطابقات، فإن هذا يؤدي ببساطة إلى نتيجة فارغة؛ يتم الإبلاغ عن الأخطاء فقط للمشاكل الفعلية في الملفات أو نظام الملفات التي قد تحدث أثناء عملية المطابقة.

يحتوي هيكل Tcl_GlobTypeData المقدم في المعلمة types على الحقول التالية:

typedef struct Tcl_GlobTypeData {
    /* يتوافق مع bcdpfls كما في 'find -t' */
    int type;
    /* يتوافق مع أذونات الملف */
    int perm;
    /* نوع Mac مقبول */
    Tcl_Obj *macType;
    /* منشئ Mac مقبول */
    Tcl_Obj *macCreator;
} Tcl_GlobTypeData;

هناك حالتان مهمتان يجب التعامل معهما بشكل صحيح، سواء عندما يكون types غير NULL. الحالتان هما عندما يكون types->types & TCL_GLOB_TYPE_DIR أو types->types & TCL_GLOB_TYPE_MOUNT صحيحين (وبشكل خاص عندما تكون الأعلام الأخرى خاطئة). في الحالة الأولى، يجب على الوظيفة إدراج الدلائل المحتوية. تستخدم Tcl هذا لتنفيذ globing متكرر، لذلك من المهم أن تقوم أنظمة الملفات بتنفيذ مطابقة الدلائل بشكل صحيح. في الحالة الثانية، مع TCL_GLOB_TYPE_MOUNT، يجب على نظام الملفات إدراج نقاط التحميل الموجودة داخل pathPtr المعطى (وفي هذه الحالة، لا يلزم أن يقع pathPtr ضمن نفس نظام الملفات - بخلاف جميع الحالات الأخرى التي يتم فيها استدعاء هذه الوظيفة). دعم هذا أمر بالغ الأهمية إذا كانت Tcl ستتمتع بانتقالات سلسة بين نظام ملفات وآخر.

UTIMEPROC

وظيفة لمعالجة استدعاء Tcl_FSUtime. مطلوبة للسماح بتعيين (وليس قراءة) الأوقات باستخدام file mtime، file atime وتنفيذ open-r/open-w/fcopy لـ file copy.

typedef int Tcl_FSUtimeProc(
        Tcl_Obj *pathPtr,
        struct utimbuf *tval);

يجب تغيير أوقات الوصول والتعديل للملف المحدد بواسطة pathPtr إلى القيم المعطاة في هيكل tval.

يجب أن تكون قيمة الإرجاع 0 عند النجاح و -1 عند حدوث خطأ، كما هو الحال مع نظام utime.

LINKPROC

وظيفة لمعالجة استدعاء Tcl_FSLink. يجب تنفيذها فقط إذا كان نظام الملفات يدعم الروابط، وقد تكون NULL بخلاف ذلك.

typedef Tcl_Obj *Tcl_FSLinkProc(
        Tcl_Obj *linkNamePtr,
        Tcl_Obj *toPtr,
        int linkAction);

إذا كان toPtr هو NULL، يُطلب من الوظيفة قراءة محتويات رابط. النتيجة هي Tcl_Obj يحدد محتويات الرابط المعطى بواسطة linkNamePtr، أو NULL إذا تعذر قراءة الرابط. النتيجة مملوكة للمستدعي (ويجب لذلك زيادة عداد المراجع الخاص بها قبل إعادتها). يجب على أي مستدعين استدعاء Tcl_DecrRefCount لهذه النتيجة عندما لا تكون مطلوبة بعد. إذا كان toPtr غير NULL، يجب على الوظيفة محاولة إنشاء رابط. يجب أن تكون النتيجة في هذه الحالة هي toPtr إذا نجح الرابط، و NULL بخلاف ذلك. في هذه الحالة، لا تكون النتيجة مملوكة للمستدعي (أي لا يلزم إجراء أي تعديلات على عداد المراجع من أي طرف). راجع وثائق Tcl_FSLink للتفسير الصحيح لأعلام linkAction.

LISTVOLUMESPROC

وظيفة لسرد أي وحدات نظام ملفات تمت إضافتها بواسطة نظام الملفات هذا. يجب تنفيذها فقط إذا أضاف نظام الملفات وحدات في بداية نظام الملفات، بحيث يمكن إرجاعها بواسطة file volumes.

typedef Tcl_Obj *Tcl_FSListVolumesProc(void);

يجب أن تكون النتيجة قائمة بوحدات نظام الملفات التي أضافها نظام الملفات هذا، أو NULL (أو قائمة فارغة) إذا لم يتم توفير أي وحدات. تعتبر قيمة النتيجة مملوكة لنظام الملفات (وليس لنواة Tcl)، ولكن يجب إعطاؤها عداد مراجع لـ Tcl. ستستخدم Tcl محتويات القائمة ثم تقلل عداد المراجع هذا. هذا يسمح لأنظمة الملفات باختيار ما إذا كانت تريد بالفعل الاحتفاظ بـ "قائمة عامة" للوحدات أم لا (إذا لم يكن الأمر كذلك، فإنها تنشئ القائمة فورًا وتمررها إلى Tcl مع عداد مراجع قدره 1 ثم تنسى القائمة، وإذا كان الأمر كذلك، فإنها ببساطة تزيد عداد المراجع لقائمتها العامة وتمررها إلى Tcl التي ستنسخ المحتويات ثم تقلل العداد إلى حيث كان).

لذلك، تعتبر Tcl القيم المرتجعة من هذا الإجراء للقراءة فقط.

FILEATTRSTRINGSPROC

وظيفة لسرد جميع سلاسل السمات الصالحة لنظام الملفات هذا. إذا لم يتم تنفيذها، فلن يدعم نظام الملفات أمر file attributes. هذا يسمح بإرفاق معلومات إضافية تعسفية بالملفات في نظام الملفات. إذا لم يتم تنفيذها، فلا حاجة لتنفيذ طرق get و set.

typedef const char *const *Tcl_FSFileAttrStringsProc(
        Tcl_Obj *pathPtr,
        Tcl_Obj **objPtrRef);

يمكن للوظيفة المستدعاة إما إعادة مصفوفة من السلاسل، أو قد تعيد NULL بدلاً من ذلك وتضع قائمة Tcl في objPtrRef المعطاة. ستأخذ Tcl هذه القائمة وتزيد عداد المراجع الخاص بها أولاً قبل استخدامها. عند الانتهاء من هذا الاستخدام، ستقلل Tcl عداد المراجع. لذلك، إذا كان يجب التخلص من القائمة بواسطة Tcl عند الانتهاء، فيجب أن يكون لديها عداد مراجع صفر، وإذا كان يجب عدم التخلص من القائمة، فيجب على نظام الملفات التأكد من أنه يعيد قيمة بعداد مراجع لا يقل عن واحد.

FILEATTRSGETPROC

وظيفة لمعالجة استدعاء Tcl_FSFileAttrsGet، يستخدم بواسطة file attributes.

typedef int Tcl_FSFileAttrsGetProc(
        Tcl_Interp *interp,
        int index,
        Tcl_Obj *pathPtr,
        Tcl_Obj **objPtrRef);

تعيد رمز إرجاع Tcl قياسي. قيمة السمة المسترجعة، والتي تتوافق مع العنصر ذي الفهرس index في القائمة التي أعادتها Tcl_FSFileAttrStringsProc، هي Tcl_Obj يتم وضعها في objPtrRef (إذا تم إرجاع TCL_OK) ومن المحتمل أن يكون لها عداد مراجع صفر. في كلتا الحالتين، يجب علينا إما تخزينها في مكان ما (على سبيل المثال، نتيجة Tcl)، أو زيادة/تقليل عداد المراجع الخاص بها لضمان تحريرها بشكل صحيح.

FILEATTRSSETPROC

وظيفة لمعالجة استدعاء Tcl_FSFileAttrsSet، يستخدم بواسطة file attributes. إذا كان نظام الملفات للقراءة فقط، فلا حاجة لتنفيذ هذا.

typedef int Tcl_FSFileAttrsSetProc(
        Tcl_Interp *interp,
        int index,
        Tcl_Obj *pathPtr,
        Tcl_Obj *objPtr);

يجب تعيين قيمة السمة للعنصر ذي الفهرس index في القائمة التي أعادتها Tcl_FSFileAttrStringsProc إلى objPtr المعطى.

CREATEDIRECTORYPROC

إجراءات نظام الملفات

تصف هذه الوثيقة إجراءات نظام الملفات التي يوفرها Tcl. يمكن استخدام هذه الإجراءات لتنفيذ عمليات نظام الملفات المخصصة.

CREATEDIRECTORYPROC

وظيفة لمعالجة استدعاء Tcl_FSCreateDirectory. يجب تنفيذها ما لم يكن نظام الملفات للقراءة فقط.

typedef int Tcl_FSCreateDirectoryProc(
        Tcl_Obj *pathPtr);

القيمة المعادة هي نتيجة Tcl قياسية تشير إلى ما إذا كان قد حدث خطأ في العملية. إذا نجحت، يجب أن يكون قد تم إنشاء دليل جديد في نظام الملفات في الموقع المحدد بواسطة pathPtr.

REMOVEDIRECTORYPROC

وظيفة لمعالجة استدعاء Tcl_FSRemoveDirectory. يجب تنفيذها ما لم يكن نظام الملفات للقراءة فقط.

typedef int Tcl_FSRemoveDirectoryProc(
        Tcl_Obj *pathPtr,
        int recursive,
        Tcl_Obj **errorPtr);

القيمة المعادة هي نتيجة Tcl قياسية تشير إلى ما إذا كان قد حدث خطأ في العملية. إذا نجحت، يجب أن يكون قد تم حذف الدليل المحدد بواسطة pathPtr من نظام الملفات. إذا تم توفير علامة recursive، فيجب حذف دليل غير فارغ دون خطأ. إذا لم يتم توفير هذه العلامة وكان الدليل غير فارغ، فيجب الإشارة إلى خطأ POSIX "EEXIST". إذا حدث خطأ، فيجب وضع اسم الملف أو الدليل الذي تسبب في الخطأ في errorPtr.

DELETEFILEPROC

وظيفة لمعالجة استدعاء Tcl_FSDeleteFile. يجب تنفيذها ما لم يكن نظام الملفات للقراءة فقط.

typedef int Tcl_FSDeleteFileProc(
        Tcl_Obj *pathPtr);

القيمة المعادة هي نتيجة Tcl قياسية تشير إلى ما إذا كان قد حدث خطأ في العملية. إذا نجحت، يجب أن يكون قد تم حذف الملف المحدد بواسطة pathPtr من نظام الملفات. لاحظ أنه إذا كان نظام الملفات يدعم الروابط الرمزية، فسوف يستدعي Tcl دائمًا هذه الوظيفة وليس Tcl_FSRemoveDirectoryProc عند الحاجة إلى حذفها (حتى لو كانت روابط رمزية للأدلة).

كفاءة نظام الملفات

لا يلزم تنفيذ هذه الوظائف لنظام ملفات معين لأن النواة توفر تنفيذًا احتياطيًا متاحًا. انظر كل وصف فردي لمعرفة عواقب ترك الحقل NULL.

LSTATPROC

وظيفة لمعالجة استدعاء Tcl_FSLstat. إذا لم يتم تنفيذها، سيحاول Tcl استخدام statProc المعرفة أعلاه بدلاً من ذلك. لذلك، لا يلزم تنفيذها إلا إذا كان نظام الملفات يمكنه التمييز بين استدعاءات stat و lstat.

typedef int Tcl_FSLstatProc(
        Tcl_Obj *pathPtr,
        Tcl_StatBuf *statPtr);

سلوك هذه الوظيفة مشابه جدًا لسلوك Tcl_FSStatProc المعرفة أعلاه، باستثناء أنه إذا تم تطبيقها على رابط رمزي، فإنها تُرجع معلومات حول الرابط، وليس حول الملف الهدف.

COPYFILEPROC

وظيفة لمعالجة استدعاء Tcl_FSCopyFile. إذا لم يتم تنفيذها، فسوف يعتمد Tcl على آلية النسخ open-r و open-w و fcopy. لذلك، لا يلزم تنفيذها إلا إذا كان نظام الملفات يمكنه أداء هذا الإجراء بكفاءة أكبر.

typedef int Tcl_FSCopyFileProc(
        Tcl_Obj *srcPathPtr,
        Tcl_Obj *destPathPtr);

القيمة المعادة هي نتيجة Tcl قياسية تشير إلى ما إذا كان قد حدث خطأ في عملية النسخ. لاحظ أن destPathPtr هو اسم الملف الذي سيصبح نسخة من srcPathPtr. إنه ليس اسم دليل يمكن نسخ srcPathPtr إليه (أي أن الوظيفة أبسط بكثير من أمر file copy على مستوى Tcl). لاحظ أنه إذا كان نظام الملفات يدعم الروابط الرمزية، فسوف يستدعي Tcl دائمًا هذه الوظيفة وليس copyDirectoryProc عند الحاجة إلى نسخها (حتى لو كانت روابط رمزية للأدلة). أخيرًا، إذا قرر نظام الملفات أنه لا يمكنه دعم إجراء file copy، فإن استدعاء Tcl_SetErrno(EXDEV) وإرجاع نتيجة غير TCL_OK سيخبر Tcl باستخدام آلياته الاحتياطية القياسية.

RENAMEFILEPROC

وظيفة لمعالجة استدعاء Tcl_FSRenameFile. إذا لم يتم تنفيذها، فسوف يعتمد Tcl على آلية النسخ والحذف. لذلك، لا يلزم تنفيذها إلا إذا كان نظام الملفات يمكنه أداء هذا الإجراء بكفاءة أكبر.

typedef int Tcl_FSRenameFileProc(
        Tcl_Obj *srcPathPtr,
        Tcl_Obj *destPathPtr);

القيمة المعادة هي نتيجة Tcl قياسية تشير إلى ما إذا كان قد حدث خطأ في عملية إعادة التسمية. إذا قرر نظام الملفات أنه لا يمكنه دعم إجراء file rename، فإن استدعاء Tcl_SetErrno(EXDEV) وإرجاع نتيجة غير TCL_OK سيخبر Tcl باستخدام آلياته الاحتياطية القياسية.

COPYDIRECTORYPROC

وظيفة لمعالجة استدعاء Tcl_FSCopyDirectory. إذا لم يتم تنفيذها، فسوف يعتمد Tcl على آلية file mkdir و file copy بشكل متكرر. لذلك، لا يلزم تنفيذها إلا إذا كان نظام الملفات يمكنه أداء هذا الإجراء بكفاءة أكبر.

typedef int Tcl_FSCopyDirectoryProc(
        Tcl_Obj *srcPathPtr,
        Tcl_Obj *destPathPtr,
        Tcl_Obj **errorPtr);

القيمة المعادة هي نتيجة Tcl قياسية تشير إلى ما إذا كان قد حدث خطأ في عملية النسخ. إذا حدث خطأ، فيجب وضع اسم الملف أو الدليل الذي تسبب في الخطأ في errorPtr. لاحظ أن destPathPtr هو اسم الدليل الذي سيصبح صورة طبق الأصل لـ srcPathPtr. إنه ليس اسم دليل يمكن نسخ srcPathPtr إليه (أي أن الوظيفة أبسط بكثير من أمر file copy على مستوى Tcl). أخيرًا، إذا قرر نظام الملفات أنه لا يمكنه دعم إجراء نسخ الدليل، فإن استدعاء Tcl_SetErrno(EXDEV) وإرجاع نتيجة غير TCL_OK سيخبر Tcl باستخدام آلياته الاحتياطية القياسية.

LOADFILEPROC

وظيفة لمعالجة استدعاء Tcl_FSLoadFile. إذا لم يتم تنفيذها، فسوف يعتمد Tcl على آلية النسخ إلى ملف مؤقت أصلي ثم استدعاء Tcl_FSLoadFile على هذا النسخة المؤقتة. لذلك، لا يلزم تنفيذها إلا إذا كان نظام الملفات يمكنه تحميل التعليمات البرمجية مباشرة، أو يمكن تنفيذها ببساطة لإرجاع TCL_ERROR لتعطيل وظيفة التحميل في نظام الملفات هذا بالكامل.

typedef int Tcl_FSLoadFileProc(
        Tcl_Interp *interp,
        Tcl_Obj *pathPtr,
        Tcl_LoadHandle *handlePtr,
        Tcl_FSUnloadFileProc *unloadProcPtr);

تُرجع رمز إكمال Tcl قياسيًا. إذا حدث خطأ، يتم ترك رسالة خطأ في نتيجة interp. تقوم الوظيفة بتحميل ملف التعليمات البرمجية الثنائية ديناميكيًا في الذاكرة. عند التحميل الناجح، يجب ملء handlePtr برمز للملف المحمل ديناميكيًا، ويجب ملء unloadProcPtr بعنوان إجراء. سيتم استدعاء إجراء إلغاء التحميل مع Tcl_LoadHandle المعطى كمعلمته الوحيدة عندما يحتاج Tcl إلى إلغاء تحميل الملف. على سبيل المثال، لنظام الملفات الأصلي، Tcl_LoadHandle المعادة حاليًا هي رمز يمكن استخدامه في TclpFindSymbol الخاص للوصول إلى وظائف في التعليمات البرمجية الجديدة. كل نظام ملفات حر في تعريف Tcl_LoadHandle كما يتطلب. أخيرًا، إذا قرر نظام الملفات أنه لا يمكنه دعم إجراء تحميل الملف، فإن استدعاء Tcl_SetErrno(EXDEV) وإرجاع نتيجة غير TCL_OK سيخبر Tcl باستخدام آلياته الاحتياطية القياسية.

UNLOADFILEPROC

وظيفة لإلغاء تحميل ملف تم تحميله بنجاح سابقًا. إذا تم تنفيذ التحميل، فيجب أيضًا تنفيذ هذا، إذا كان هناك أي إجراء تنظيف مطلوب.

typedef void Tcl_FSUnloadFileProc(
        Tcl_LoadHandle loadHandle);

GETCWDPROC

وظيفة لمعالجة استدعاء Tcl_FSGetCwd. لا تحتاج معظم أنظمة الملفات إلى تنفيذ هذا. عادةً ما يتم استدعاؤها مرة واحدة فقط، إذا تم استدعاء getcwd قبل chdir. قد تكون NULL.

typedef Tcl_Obj *Tcl_FSGetCwdProc(
        Tcl_Interp *interp);

إذا كان نظام الملفات يدعم مفهومًا أصليًا للدليل الحالي (والذي قد يتغير بشكل مستقل عن Tcl)، فيجب أن تُرجع هذه الوظيفة دليل العمل الحالي كنتيجة، أو NULL إذا تعذر تحديد الدليل الحالي (على سبيل المثال، لا يمتلك المستخدم الأذونات المناسبة على دليل العمل الحالي). إذا تم إرجاع NULL، يتم ترك رسالة خطأ في نتيجة interp.

CHDIRPROC

وظيفة لمعالجة استدعاء Tcl_FSChdir. إذا لم تقم أنظمة الملفات بتنفيذ هذا، فسيتم محاكاته بسلسلة من عمليات التحقق من الوصول إلى الدليل. بخلاف ذلك، تحتاج أنظمة الملفات الافتراضية التي تنفذ هذا فقط إلى الاستجابة بنتيجة إرجاع إيجابية إذا كان pathPtr دليلًا صالحًا وممكن الوصول إليه في نظام ملفاتها. لا يلزمها تذكر النتيجة، حيث سيتم تذكرها تلقائيًا للاستخدام بواسطة Tcl_FSGetCwd. يجب أن تقوم أنظمة الملفات الحقيقية بتنفيذ الإجراء الصحيح (أي استدعاء واجهة برمجة تطبيقات chdir الصحيحة للنظام).

typedef int Tcl_FSChdirProc(
        Tcl_Obj *pathPtr);

يغير Tcl_FSChdirProc دليل العمل الحالي للتطبيق إلى القيمة المحددة في pathPtr. تُرجع الوظيفة -1 عند حدوث خطأ أو 0 عند النجاح.
