الوسائط

const Tcl_Filesystem *fsPtr (in)
يشير إلى بنية تحتوي على عناوين الإجراءات التي يمكن استدعاؤها لأداء عمليات نظام الملفات المختلفة.
Tcl_Obj *pathPtr (in)
المسار الممثل بهذه القيمة يستخدم للعملية المعنية. إذا لم تكن للقيمة تمثيل داخلي path بالفعل، فسيتم تحويلها للحصول عليه.
Tcl_Obj *srcPathPtr (in)
كما هو الحال بالنسبة لـ pathPtr، ولكن يستخدم للملف المصدر لعملية نسخ أو إعادة تسمية.
Tcl_Obj *destPathPtr (in)
كما هو الحال بالنسبة لـ pathPtr، ولكن يستخدم لاسم الملف الوجهة لعملية نسخ أو إعادة تسمية.
int recursive (in)
ما إذا كان سيتم إزالة الدلائل الفرعية ومحتوياتها أيضًا.
const char *encodingName (in)
ترميز البيانات المخزنة في الملف المحدد بواسطة pathPtr والتي سيتم تقييمها.
const char *pattern (in)
سيتم إرجاع الملفات أو الدلائل التي تطابق هذا النمط فقط.
Tcl_GlobTypeData *types (in)
سيتم إرجاع الملفات أو الدلائل التي تطابق أوصاف النوع الموجودة في هذه البنية فقط. قد تكون هذه الوسيطة NULL.
Tcl_Interp *interp (in)
المفسر المراد استخدامه إما للنتائج أو التقييم أو الإبلاغ عن رسائل الخطأ.
void *clientData (in)
الوصف الأصلي لقيمة المسار المراد إنشاؤها.
Tcl_Obj *firstPtr (in)
القيمة الأولى من قيمتي المسار المراد مقارنتهما. قد يتم تحويل القيمة إلى نوع path.
Tcl_Obj *secondPtr (in)
القيمة الثانية من قيمتي المسار المراد مقارنتهما. قد يتم تحويل القيمة إلى نوع path.
Tcl_Obj *listObj (in)
قائمة عناصر المسار المراد تشغيل عملية join عليها.
int elements (in)
عدد العناصر في listObj التي يجب دمجها معًا. إذا كانت سالبة، فسيتم دمج جميع العناصر.
Tcl_Obj **errorPtr (out)
في حالة حدوث خطأ، يتم ملؤها بقيمة تحتوي على اسم الملف الذي تسبب في حدوث خطأ في عمليات النسخ/إعادة التسمية المختلفة.
int index (in)
فهرس السمة المعنية.
Tcl_Obj *objPtr (in)
القيمة المراد تعيينها في العملية.
Tcl_Obj **objPtrRef (out)
يتم ملؤها بقيمة تحتوي على نتيجة العملية.
int mode (in)
قناع يتكون من واحد أو أكثر من R_OK، W_OK، X_OK و F_OK. تطلب R_OK، W_OK و X_OK التحقق مما إذا كان الملف موجودًا ولديه أذونات القراءة أو الكتابة أو التنفيذ، على التوالي. تطلب F_OK فقط التحقق من وجود الملف.
Tcl_StatBuf *statPtr (out)
البنية التي تحتوي على نتيجة عملية stat أو lstat.
const char *sym1 (in)
اسم إجراء للبحث عنه في جدول رموز الملف
const char *sym2 (in)
اسم إجراء للبحث عنه في جدول رموز الملف
Tcl_PackageInitProc **proc1Ptr (out)
يتم ملؤها بوظيفة التهيئة لهذا الكود.
Tcl_PackageInitProc **proc2Ptr (out)
يتم ملؤها بوظيفة التهيئة الآمنة لهذا الكود.
void **clientDataPtr (out)
يتم ملؤها بقيمة clientData لتمريرها إلى وظيفة إلغاء تحميل هذا الكود عند استدعائها.
Tcl_LoadHandle *loadHandlePtr (out)
يتم ملؤها برمز مجرد يمثل الملف المحمل.
Tcl_FSUnloadFileProc **unloadProcPtr (out)
يتم ملؤها بالوظيفة المراد استخدامها لإلغاء تحميل هذا الجزء من الكود.
Tcl_LoadHandle loadHandle (in)
مقبض للمكتبة المحملة المراد إلغاء تحميلها.
utimbuf *tval (in)
يتم قراءة أوقات الوصول والتعديل في هذه البنية واستخدامها لتعيين تلك القيم لملف معين.
const char *modeString (in)
يحدد كيفية الوصول إلى الملف. قد يكون له أي من القيم المسموح بها للمعامل mode لأمر Tcl open.
int permissions (in)
علامات الأذونات بنمط POSIX مثل 0644. إذا تم إنشاء ملف جديد، فسيتم تعيين هذه الأذونات على الملف المنشأ.
