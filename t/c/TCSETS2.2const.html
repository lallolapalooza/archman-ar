الاسم

TCGETS, TCSETS, TCSETSW, TCSETSF - الحصول على خصائص الطرفية وتعيينها

المكتبة

مكتبة C القياسية (libc, -lc)

الوصف

TCGETS
مكافئ لـ tcgetattr(fd, argp).
للحصول على إعدادات المنفذ التسلسلي الحالية.
TCSETS
مكافئ لـ tcsetattr(fd, TCSANOW, argp).
لتعيين إعدادات المنفذ التسلسلي الحالية.
TCSETSW
مكافئ لـ tcsetattr(fd, TCSADRAIN, argp).
للسماح بتفريغ مخزن الإخراج المؤقت وتعيين إعدادات المنفذ التسلسلي الحالية.
TCSETSF
مكافئ لـ tcsetattr(fd, TCSAFLUSH, argp).
للسماح بتفريغ مخزن الإخراج المؤقت، وتجاهل بيانات الإدخال المعلقة، وتعيين إعدادات المنفذ التسلسلي الحالية.

الأوامر الأربعة التالية لـ ioctl مشابهة لـ TCGETS, TCSETS, TCSETSW, TCSETSF، باستثناء أنها تستخدم struct termios2 بدلاً من struct termios. إذا كان العضو c_cflag في struct termios يحتوي على العلامة BOTHER، فسيتم تخزين معدل الباود في الأعضاء c_ispeed و c_ospeed كقيم صحيحة. هذه الأوامر ioctl غير مدعومة على جميع المعماريات.

TCSETS2
TCSETSW2
TCSETSF2

الأوامر الأربعة التالية لـ ioctl مشابهة لـ TCGETS, TCSETS, TCSETSW, TCSETSF، باستثناء أنها تستخدم struct termio بدلاً من struct termios.

قيمة الإرجاع

عند النجاح، يتم إرجاع 0. عند الخطأ، يتم إرجاع -1 ويتم تعيين errno للإشارة إلى الخطأ.

تحذيرات

struct termios من <asm/termbits.h> مختلفة وغير متوافقة مع struct termios من <termios.h>. تتطلب أوامر ioctl هذه struct termios من <asm/termbits.h>.

أمثلة

الحصول على أو تعيين أي معدل باود على المنفذ التسلسلي.

/* SPDX-License-Identifier: GPL-2.0-or-later */
#include <asm/termbits.h>
#include <fcntl.h>
#include <stdio.h>
#include <stdlib.h>
#include <sys/ioctl.h>
#include <unistd.h>
int
main(int argc, char *argv[])
{
#if !defined BOTHER
    fprintf(stderr, "BOTHER غير مدعوم\n");
    /* قد يفشل البرنامج في TCGETS/TCSETS مع ثوابت Bnnn */
    exit(EXIT_FAILURE);
#else
    /* أعلن عن هيكل tio، ويعتمد نوعه على ioctl المدعومة */
# if defined TCGETS2
    struct termios2 tio;
# else
    struct termios tio;
# endif
    int fd, rc;
    if (argc != 2 && argc != 3 && argc != 4) {
        fprintf(stderr, "الاستخدام: %s جهاز [إخراج [إدخال]]\n", argv[0]);
        exit(EXIT_FAILURE);
    }
    fd = open(argv[1], O_RDWR | O_NONBLOCK | O_NOCTTY);
    if (fd < 0) {
        perror("open");
        exit(EXIT_FAILURE);
    }
    /* الحصول على إعدادات المنفذ التسلسلي الحالية عبر ioctl المدعومة */
# if defined TCGETS2
    rc = ioctl(fd, TCGETS2, &tio);
# else
    rc = ioctl(fd, TCGETS, &tio);
# endif
    if (rc) {
        perror("TCGETS");
        close(fd);
        exit(EXIT_FAILURE);
    }
    /* تغيير معدل الباود عند وجود وسائط إضافية */
    if (argc == 3 || argc == 4) {
        /* امسح معدل الإخراج الحالي واملأ قيمة جديدة */
        tio.c_cflag &= ~CBAUD;
        tio.c_cflag |= BOTHER;
        tio.c_ospeed = atoi(argv[2]);
        /* امسح معدل الإدخال الحالي واملأ قيمة جديدة */
        tio.c_cflag &= ~(CBAUD << IBSHIFT);
        tio.c_cflag |= BOTHER << IBSHIFT;
        /* عندما لا يتم توفير الوسيط 4، استخدم معدل الإخراج مرة أخرى */
        tio.c_ispeed = (argc == 4) ? atoi(argv[3]) : atoi(argv[2]);
        /* تعيين إعدادات المنفذ التسلسلي عبر ioctl المدعومة */
# if defined TCSETS2
        rc = ioctl(fd, TCSETS2, &tio);
# else
        rc = ioctl(fd, TCSETS, &tio);
# endif
        if (rc) {
            perror("TCSETS");
            close(fd);
            exit(EXIT_FAILURE);
        }
        /* والحصول على القيم التي تم تكوينها بالفعل */
# if defined TCGETS2
        rc = ioctl(fd, TCGETS2, &tio);
# else
        rc = ioctl(fd, TCGETS, &tio);
# endif
        if (rc) {
            perror("TCGETS");
            close(fd);
            exit(EXIT_FAILURE);
        }
    }
    close(fd);
    printf("معدل الباود للإخراج: %u\n", tio.c_ospeed);
    printf("معدل الباود للإدخال: %u\n", tio.c_ispeed);
    exit(EXIT_SUCCESS);
#endif
}
