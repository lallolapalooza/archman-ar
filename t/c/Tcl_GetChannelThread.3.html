WATCHPROC

يحتوي الحقل watchProc على عنوان دالة تستدعيها الطبقة العامة لتهيئة آلية إعلام بالأحداث ذات الأهمية في هذه القناة. يجب أن تتطابق WatchProc مع النموذج الأولي التالي:

typedef void Tcl_DriverWatchProc(
        ClientData instanceData,
        int mask);

instanceData هو نفس القيمة التي تم تمريرها إلى Tcl_CreateChannel عند إنشاء هذه القناة. الوسيط mask هو تركيبة OR من TCL_READABLE و TCL_WRITABLE و TCL_EXCEPTION؛ يشير إلى الأحداث التي يريد المتصل إعلامه بها في هذه القناة.

يجب على الدالة تهيئة آليات خاصة بنوع الجهاز للإعلام عند وجود حدث ذي أهمية في القناة. عندما يحدث حدث ذو أهمية في القناة، يجب على مشغل القناة استدعاء Tcl_NotifyChannel لإبلاغ وحدة القناة العامة. يجب على المشغل تجنب تعطيل مشغلات القنوات الأخرى أو مصادر رد النداء عن طريق استدعاء Tcl_NotifyChannel بشكل متكرر للغاية. يمكن ضمان العدالة عن طريق استخدام قائمة انتظار أحداث Tcl للسماح بالاستجابة لحدث القناة بالتتابع مع الأحداث الأخرى. ارجع إلى وصف Tcl_QueueEvent للحصول على تفاصيل حول جدولة حدث.

يمكن استرداد هذه القيمة باستخدام Tcl_ChannelWatchProc، والذي يعيد عنوان الدالة.

GETHANDLEPROC

يحتوي الحقل getHandleProc على عنوان دالة تستدعيها الطبقة العامة لاسترداد معالج (handle) خاص بالقناة. يجب أن تتطابق GetHandleProc مع النموذج الأولي التالي:

typedef int Tcl_DriverGetHandleProc(
        ClientData instanceData,
        int direction,
        ClientData *handlePtr);

instanceData هو نفس القيمة التي تم تمريرها إلى Tcl_CreateChannel عند إنشاء هذه القناة. الوسيط direction هو TCL_READABLE لاسترداد المعالج المستخدم للإدخال، أو TCL_WRITABLE لاسترداد المعالج المستخدم للإخراج.

إذا كان تطبيق القناة يحتوي على معالجات خاصة، فيجب على الدالة استرداد المعالج المناسب بناءً على الوسيط direction. يجب تخزين المعالج في الموقع الذي يشير إليه handlePtr، ويجب إرجاع TCL_OK. إذا لم تكن القناة مفتوحة للاتجاه المحدد أو إذا كان تطبيق القناة لا يستخدم معالجات خاصة، فيجب على الدالة إرجاع TCL_ERROR.

يمكن استرداد هذه القيمة باستخدام Tcl_ChannelGetHandleProc، والذي يعيد عنوان الدالة.

FLUSHPROC

الحقل flushProc محجوز للاستخدام المستقبلي. يجب تعيينه إلى NULL. يجب أن تتطابق FlushProc مع النموذج الأولي التالي:

typedef int Tcl_DriverFlushProc(
        ClientData instanceData);

يمكن استرداد هذه القيمة باستخدام Tcl_ChannelFlushProc، والذي يعيد عنوان الدالة.

HANDLERPROC

يحتوي الحقل handlerProc على عنوان دالة تستدعيها الطبقة العامة لإعلام القناة بحدوث حدث. يجب تعريفه لمشغلات القنوات المتراصة (stacked) التي ترغب في إعلامها بالأحداث في القناة الأساسية (المتراصة). يجب أن تتطابق HandlerProc مع النموذج الأولي التالي:

typedef int Tcl_DriverHandlerProc(
        ClientData instanceData,
        int interestMask);

instanceData هو نفس القيمة التي تم تمريرها إلى Tcl_CreateChannel عند إنشاء هذه القناة. interestMask هو تركيبة OR من TCL_READABLE أو TCL_WRITABLE؛ يشير إلى نوع الحدث الذي وقع في هذه القناة.

يمكن استرداد هذه القيمة باستخدام Tcl_ChannelHandlerProc، والذي يعيد عنوان الدالة.

THREADACTIONPROC

يحتوي الحقل threadActionProc على عنوان الدالة التي تستدعيها الطبقة العامة عند إنشاء قناة، أو إغلاق قناة، أو عند الانتقال إلى مؤشر ترابط آخر، أي عندما قد تحتاج حالة مؤشر ترابط المشغل الخاصة إلى التهيئة أو التحديث. يمكن أن يكون NULL. يتم استخدام الإجراء TCL_CHANNEL_THREAD_REMOVE لإعلام المشغل بأنه يجب تحديث أو إزالة أي بيانات خاصة بالنوع قد يحتفظ بها للقناة.

يتم استخدام الإجراء TCL_CHANNEL_THREAD_INSERT لإعلام المشغل بأنه يجب تحديث أو تهيئة أي بيانات خاصة بالنوع قد يحتفظ بها باستخدام مؤشر الترابط الحالي كمعاون. ارجع إلى Tcl_CutChannel و Tcl_SpliceChannel للحصول على تفاصيل إضافية.

typedef void Tcl_DriverThreadActionProc(
        ClientData instanceData,
        int action);

instanceData هو نفس القيمة التي تم تمريرها إلى Tcl_CreateChannel عند إنشاء هذه القناة.

يمكن استرداد هذه القيم باستخدام Tcl_ChannelThreadActionProc، والذي يعيد عنوان الدالة.

TRUNCATEPROC

يحتوي الحقل truncateProc على عنوان الدالة التي تستدعيها الطبقة العامة عند اقتطاع القنوات إلى طول معين. يمكن أن يكون NULL.

typedef int Tcl_DriverTruncateProc(
        ClientData instanceData,
        Tcl_WideInt length);

instanceData هو نفس القيمة التي تم تمريرها إلى Tcl_CreateChannel عند إنشاء هذه القناة، و length هو الطول الجديد للملف الأساسي، والذي يجب ألا يكون سالبًا. يجب أن تكون النتيجة 0 في حالة النجاح، أو رمز خطأ (مناسب للاستخدام مع Tcl_SetErrno) في حالة الفشل.

يمكن استرداد هذه القيم باستخدام Tcl_ChannelTruncateProc، والذي يعيد عنوان الدالة.

TCL_BADCHANNELOPTION

تنشئ هذه الإجراءات رسالة خطأ "خيار غير صالح" في مفسر (اختياري). تستخدمها مشغلات القنوات عند طلب خيار تعيين/الحصول غير صالح. هدفها هو دمج قائمة الخيارات العامة مع قائمة الخيارات الخاصة وتجميع رسالة الخطأ الخاصة بالخيارات العامة.

تعيد دائمًا TCL_ERROR.

يتم إنشاء رسالة خطأ في قيمة نتيجة interp للإشارة إلى أنه تم استدعاء أمر بخيار غير صالح. تكون الرسالة بالشكل:

    bad option "blah": should be one of
    <...generic options...>+<...specific options...>
على سبيل المثال: 
    bad option "-blah": should be one of -blocking,
    -buffering, -buffersize, -eofchar, -translation,
    -peername, or -sockname
عند استدعائها مع optionList مساوٍ لـ "peername sockname".

"blah" هو الوسيط optionName، و "<specific options>" هي قائمة كلمات خاصة مفصولة بمسافات. تتولى الدالة إدخال واصلات (-) قبل كل خيار، وفواصل (,) بعدها، و "or" قبل الخيار الأخير.

OLD CHANNEL TYPES

يحتوي نوع القناة الأصلي (8.3.1 والإصدارات السابقة) Tcl_ChannelType على الحقول التالية:

typedef struct Tcl_ChannelType {
    const char *typeName;
    Tcl_DriverBlockModeProc *blockModeProc;
    Tcl_DriverCloseProc *closeProc;
    Tcl_DriverInputProc *inputProc;
    Tcl_DriverOutputProc *outputProc;
    Tcl_DriverSeekProc *seekProc;
    Tcl_DriverSetOptionProc *setOptionProc;
    Tcl_DriverGetOptionProc *getOptionProc;
    Tcl_DriverWatchProc *watchProc;
    Tcl_DriverGetHandleProc *getHandleProc;
    Tcl_DriverClose2Proc *close2Proc;
} Tcl_ChannelType;

على الرغم من أنه يمكن إنشاء قناة باستخدام الهيكل السابق، إلا أن الكود الداخلي للقناة سيحدد الإصدار. من الضروري استخدام الهيكل الجديد Tcl_ChannelType إذا كنت تقوم بإنشاء مشغل قناة متراص (stacked) بسبب مشاكل في التنفيذ السابق للقنوات المتراصة (8.2.0 إلى 8.3.1).

قبل 8.4.0 (أي، خلال الإصدارات اللاحقة من 8.3 والجزء المبكر من دورة تطوير 8.4) احتوى الهيكل Tcl_ChannelType على الحقول التالية:

typedef struct Tcl_ChannelType {
    const char *typeName;
    Tcl_ChannelTypeVersion version;
    Tcl_DriverCloseProc *closeProc;
    Tcl_DriverInputProc *inputProc;
    Tcl_DriverOutputProc *outputProc;
    Tcl_DriverSeekProc *seekProc;
    Tcl_DriverSetOptionProc *setOptionProc;
    Tcl_DriverGetOptionProc *getOptionProc;
    Tcl_DriverWatchProc *watchProc;
    Tcl_DriverGetHandleProc *getHandleProc;
    Tcl_DriverClose2Proc *close2Proc;
    Tcl_DriverBlockModeProc *blockModeProc;
    Tcl_DriverFlushProc *flushProc;
    Tcl_DriverHandlerProc *handlerProc;
    Tcl_DriverTruncateProc *truncateProc;
} Tcl_ChannelType;

عند تسجيل الهيكل السابق كنوع قناة، يجب أن يكون الحقل version دائمًا TCL_CHANNEL_VERSION_2.
