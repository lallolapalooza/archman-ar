الوسائط

Tcl_Interp *interp (in)
يستخدم لتقديم تقارير الأخطاء وللبحث عن قناة مسجلة فيه.
const char *fileName (in)
اسم ملف محلي أو عبر الشبكة.
const char *mode (in)
الوضع الذي يتم به الوصول إلى الملف. يمكن أن يأخذ أي قيمة مسموح بها لوسيط mode في أمر open الخاص بـ Tcl.
int permissions (in)
علامات صلاحيات POSIX مثل 0644. إذا تم إنشاء ملف جديد، سيتم تعيين هذه الصلاحيات للملف المنشأ.
int argc (in)
عدد العناصر في argv.
const char **argv (in)
وسائط لإنشاء خط أنابيب للأوامر. هذه القيم لها نفس معنى وسائط عدم التبديل في أمر exec.
int flags (in)
هو حالة مقبضات الإدخال/الإخراج القياسية في خط الأنابيب: مزيج OR من TCL_STDIN، TCL_STDOUT، TCL_STDERR، و TCL_ENFORCE_MODE. إذا تم تعيين TCL_STDIN، فإن stdin للطفل الأول في خط الأنابيب هو قناة خط الأنابيب، وإلا فهو نفس stdin للعملية المستدعية؛ وينطبق الشيء نفسه على TCL_STDOUT و TCL_STDERR. إذا لم يتم تعيين TCL_ENFORCE_MODE، يمكن لخط الأنابيب إعادة توجيه مقبضات الإدخال/الإخراج القياسية لتجاوز مقبضات الإدخال/الإخراج القياسية التي تم تعيين TCL_STDIN، TCL_STDOUT، و TCL_STDERR لها. إذا تم تعيينها، فإن هذه التوجيهات تسبب خطأ.
ClientData handle (in)
مقبض نظام التشغيل للإدخال/الإخراج إلى ملف. بالنسبة لنظام Unix، يكون واصف ملف؛ بالنسبة لنظام Windows، يكون HANDLE.
int readOrWrite (in)
مزيج OR من TCL_READABLE و TCL_WRITABLE للإشارة إلى العمليات الصالحة على handle.
const char *channelName (in)
اسم القناة.
int *modePtr (out)
مؤشر إلى متغير صحيح سيحتوي على مزيج OR من TCL_READABLE و TCL_WRITABLE الذي يشير إلى ما إذا كانت القناة مفتوحة للقراءة والكتابة.
const char *pattern (in)
النمط المراد مطابقته، يتم تمريره إلى Tcl_StringMatch، أو NULL.
Tcl_Channel channel (in)
قناة Tcl للإدخال أو الإخراج. يجب أن تكون نتيجة إجراء مثل Tcl_OpenFileChannel.
Tcl_Obj *readObjPtr (in/out)
مؤشر إلى قيمة Tcl سيتم تخزين الأحرف المقروءة من القناة فيها. إذا كان ترميز القناة binary، فهذا يعادل عدد البايتات المقروءة من القناة.
int charsToRead (in)
عدد الأحرف المراد قراءتها من القناة. إذا كان ترميز القناة binary، فهذا يعادل عدد البايتات المراد قراءتها من القناة.
int appendFlag (in)
إذا كان غير صفري، فسيتم إلحاق البيانات المقروءة من القناة بالقيمة. بخلاف ذلك، ستحل البيانات محل المحتوى الموجود في القيمة.
char *readBuf (out)
مخزن مؤقت سيتم تخزين البايتات المقروءة من القناة فيه.
int bytesToRead (in)
عدد البايتات المراد قراءتها من القناة. يجب أن يكون المخزن المؤقت readBuf كبيرًا بما يكفي لاستيعاب هذا العدد من البايتات.
Tcl_Obj *lineObjPtr (in/out)
مؤشر إلى قيمة Tcl سيتم تخزين السطر المقروء من القناة فيها. سيتم إلحاق السطر المقروء بالقيمة الحالية للقيمة.
Tcl_DString *lineRead (in/out)
مؤشر إلى سلسلة ديناميكية سيتم تخزين السطر المقروء من القناة فيها. يجب أن يكون قد تم تهيئته بواسطة المتصل. سيتم إلحاق السطر المقروء بأي بيانات موجودة بالفعل في السلسلة الديناميكية.
const char *input (in)
الإدخال المراد إضافته إلى مخزن مؤقت للقناة.
int inputLen (in)
طول الإدخال.
int addAtEnd (in)
علامة تشير إلى ما إذا كان سيتم إضافة الإدخال في نهاية المخزن المؤقت للقناة أو في بدايته.
Tcl_Obj *writeObjPtr (in)
مؤشر إلى قيمة Tcl سيتم كتابة محتوياتها إلى القناة.
const char *charBuf (in)
مخزن مؤقت يحتوي على الأحرف المراد كتابتها إلى القناة.
const char *byteBuf (in)
مخزن مؤقت يحتوي على البايتات المراد كتابتها إلى القناة.
int bytesToWrite (in)
عدد البايتات المراد استهلاكها من charBuf أو byteBuf وكتابتها إلى القناة.
Tcl_WideInt offset (in)
كم عدد البايتات التي سيتم نقل مؤشر الوصول بها في القناة للعملية التالية للإدخال أو الإخراج؟ يتم قياسها من الموضع المحدد بواسطة seekMode. يمكن أن تكون موجبة أو سالبة.
int seekMode (in)
إلى أي نقطة وصول سيتم البحث؟ يستخدم مع offset لحساب نقطة الوصول الجديدة للقناة. القيم القانونية هي SEEK_SET، SEEK_CUR، و SEEK_END.
Tcl_WideInt length (in)
الطول (غير سالب) لتقصير القناة.
const char *optionName (in)
اسم خيار قابل للتطبيق على القناة، على سبيل المثال -blocking. يمكن أن يأخذ أي قيمة مقبولة بواسطة أمر fconfigure.
Tcl_DString *optionValue (in)
المكان الذي سيتم فيه تخزين قيمة خيار أو قائمة بجميع الخيارات وقيمها. يجب أن يكون قد تم تهيئته بواسطة المتصل.
const char *newValue (in)
القيمة الجديدة للخيار المحدد بواسطة optionName.

توفر آلية قنوات Tcl آلية مستقلة عن الجهاز والمنصة لتنفيذ عمليات الإدخال/الإخراج ذات التخزين المؤقت على أنواع مختلفة من الملفات والمقابس والأجهزة. آلية القنوات قابلة للتوسيع لأنواع القنوات الجديدة، عن طريق توفير مشغل قناة منخفض المستوى للنوع الجديد؛ تم وصف واجهة مشغل القناة في دليل Tcl_CreateChannel.

TCL_READCHARS, TCL_READOBJS, AND TCL_READ

Tcl_ReadChars يقرأ العدد المطلوب من الأحرف من القناة المحددة بواسطة channel ويضعها في عنصر Tcl الذي يشير إليه readObjPtr. Tcl_ReadChars يتوقع أن يكون readObjPtr عنصر Tcl فارغًا.

يقوم Tcl_ReadChars بتحويل الأحرف من ترميز القناة إلى UTF-8 قبل إضافتها إلى readObjPtr. يعتمد هذا التحويل على الترميز المحدد للقناة. إذا كان ترميز القناة هو "binary"، فلن يتم إجراء أي تحويل، وسيتم تخزين البايتات كما هي في تمثيل Tcl الداخلي كصفيفة بايت.

قيمة bytesToRead تحدد الحد الأقصى لعدد البايتات المراد قراءتها. إذا كانت bytesToRead قيمتها -1، فسيقوم الأمر بقراءة جميع الأحرف المتاحة حاليًا (وضع غير مانع) أو كل شيء حتى نهاية الملف (وضع مانع).

قد تكون القيمة المرجعة أصغر من القيمة المطلوبة للقراءة، مما يشير إلى أن كمية بيانات أقل مما تم طلبه كانت متاحة. يُطلق على هذا "قراءة قصيرة". في الوضع المانع، يمكن أن يحدث هذا فقط عند نهاية الملف. في الوضع غير المانع، يمكن أن تحدث قراءة قصيرة أيضًا إذا لم يكن هناك مدخلات كافية متاحة حاليًا: Tcl_ReadChars تُرجع عددًا صغيرًا بدلاً من الانتظار لمزيد من البيانات.

إذا كانت القناة في الوضع المانع، فإن القيمة المرجعة صفر تشير إلى حالة نهاية الملف. إذا كانت القناة في الوضع غير المانع، فإن القيمة المرجعة صفر تشير إما إلى عدم توفر مدخلات حاليًا أو إلى حالة نهاية الملف. استخدم Tcl_Eof و Tcl_InputBlocked لمعرفة أي من هذه الحالات حدثت بالفعل.

Tcl_ReadChars يترجم تمثيلات نهاية السطر المختلفة إلى التمثيل الداخلي القياسي \n وفقًا لوضع التعرف على نهاية السطر الحالي. يتم وصف التعرف على نهاية السطر والأوضاع المختلفة الخاصة بكل منصة في إدخال الدليل لأمر Tcl fconfigure .

كتحسين للأداء، عند القراءة من قناة ذات ترميز binary ، لا يتم تحويل البايتات إلى UTF-8 عند قراءتها. بدلاً من ذلك، يتم تخزينها في التمثيل الداخلي لـ readObjPtr كقيمة مصفوفة بايت. لن يتم إنشاء التمثيل النصي لهذه القيمة إلا إذا لزم الأمر (على سبيل المثال، بسبب استدعاء Tcl_GetStringFromObj ). بهذه الطريقة، يمكن قراءة البيانات الموجهة بالبايت من قناة، ومعالجتها عن طريق استدعاء Tcl_GetByteArrayFromObj والوظائف ذات الصلة، ثم كتابتها إلى قناة دون تكلفة التحويل إلى أو من UTF-8.

Tcl_Read مشابه لـ Tcl_ReadChars باستثناء أنها لا تقوم بتحويلات الترميز، بغض النظر عن ترميز القناة. إنها مهملة وموجودة للتوافق مع الإصدارات السابقة مع امتدادات Tcl غير الدولية. تستهلك بايتات من channel وتخزنها في readBuf ، مع إجراء ترجمات نهاية السطر في الطريق. القيمة المرجعة لـ Tcl_Read هي عدد البايتات، حتى bytesToRead ، المكتوبة في readBuf . المخزن المؤقت الذي ينتجه Tcl_Read ليس له نهاية فارغة. محتوياته صالحة من الموضع الصفري حتى الموضع المشار إليه بالقيمة المرجعة، باستثناء ذلك الموضع.

Tcl_ReadRaw هو نفس Tcl_Read ولكنه لا يعوض عن التكديس. بينما Tcl_Read (والوظائف الأخرى في واجهة برمجة التطبيقات) تحصل دائمًا على بياناتها من القناة العلوية في المكدس الذي تنتمي إليه القناة المقدمة، فإن Tcl_ReadRaw لا تفعل ذلك. لذلك، هذه الوظيفة قابلة للاستخدام *فقط* لمحركات القنوات التحويلية، أي المحركات المستخدمة في منتصف مكدس من القنوات، لنقل البيانات من القناة أدناه إلى التحويل.

TCL_GETSOBJ AND TCL_GETS

Tcl_GetsObj تستهلك بايتات من channel ، وتحول البايتات إلى UTF-8 بناءً على ترميز القناة، حتى يتم رؤية سطر كامل من المدخلات. إذا كان ترميز القناة هو binary ، فيتم التعامل مع كل بايت مقروء من القناة كحرف يونيكود فردي. يتم إلحاق جميع أحرف السطر باستثناء حرف (أو أحرف) نهاية السطر المحدد بالتمثيل النصي لـ lineObjPtr . يتم قراءة أحرف نهاية السطر والتخلص منها.

إذا تم قراءة سطر بنجاح، فإن القيمة المرجعة أكبر من أو تساوي صفر وتشير إلى عدد البايتات المخزنة في lineObjPtr . إذا حدث خطأ، فإن Tcl_GetsObj تُرجع -1 وتسجل رمز خطأ POSIX يمكن استرداده باستخدام Tcl_GetErrno . ترجع Tcl_GetsObj أيضًا -1 إذا تم الوصول إلى نهاية الملف؛ يمكن استخدام الإجراء Tcl_Eof للتمييز بين خطأ وحالة نهاية الملف.

إذا كانت القناة في الوضع غير المانع، يمكن أن تكون القيمة المرجعة أيضًا -1 إذا لم تكن هناك بيانات متاحة أو أن البيانات المتاحة لم تحتوي على حرف نهاية السطر. عند إرجاع -1، يمكن استدعاء الإجراء Tcl_InputBlocked لتحديد ما إذا كانت القناة محظورة بسبب عدم توفر المدخلات.

Tcl_Gets هو نفس Tcl_GetsObj باستثناء أن الأحرف الناتجة يتم إلحاقها بالسلسلة النصية الديناميكية المعطاة بواسطة lineRead بدلاً من قيمة Tcl.

TCL_UNGETS

Tcl_Ungets تُستخدم لإضافة بيانات إلى طابور المدخلات لقناة، في مقدمة أو مؤخرة الطابور. المؤشر input يشير إلى البيانات المراد إضافتها. يتم إعطاء طول المدخلات المراد إضافتها بواسطة inputLen . تشير قيمة غير صفرية لـ addAtEnd إلى أنه سيتم إضافة البيانات في نهاية الطابور؛ بخلاف ذلك، سيتم إضافتها في مقدمة الطابور. إذا كانت channel لديها علامة EOF "ملتصقة"، فلن يتم إضافة أي بيانات إلى طابور المدخلات. Tcl_Ungets تُرجع inputLen أو -1 في حالة حدوث خطأ.

TCL_WRITECHARS, TCL_WRITEOBJ, AND TCL_WRITE

Tcl_WriteChars تقبل bytesToWrite من البايتات من بيانات الأحرف في charBuf . يتم تحويل أحرف UTF-8 في المخزن المؤقت إلى ترميز القناة وتجميعها للإخراج إلى channel . إذا كانت bytesToWrite قيمتها سالبة، فإن Tcl_WriteChars تتوقع أن يكون charBuf له نهاية فارغة وتقوم بإخراج كل شيء حتى النهاية الفارغة.

قد لا تظهر البيانات المجمعة للإخراج على جهاز الإخراج فورًا، بسبب التخزين المؤقت الداخلي. إذا كان يجب أن تظهر البيانات فورًا، اتصل بـ Tcl_Flush بعد استدعاء Tcl_WriteChars ، أو اضبط خيار -buffering على القناة على none . إذا كنت ترغب في ظهور البيانات فور قبول سطر كامل للإخراج، فاضبط خيار -buffering على القناة على وضع line .

القيمة المرجعة لـ Tcl_WriteChars هي عدد البايتات التي تم قبولها للإخراج إلى القناة. هذا إما أكبر من صفر للإشارة إلى النجاح أو -1 للإشارة إلى حدوث خطأ. في حالة حدوث خطأ، تسجل Tcl_WriteChars رمز خطأ POSIX يمكن استرداده باستخدام Tcl_GetErrno .

يتم ترجمة أحرف نهاية السطر في بيانات الإخراج إلى تسلسلات نهاية سطر خاصة بالمنصة وفقًا للخيار -translation للقناة. يتم ذلك حتى لو لم يكن للقناة ترميز.

Tcl_WriteObj مشابه لـ Tcl_WriteChars باستثناء أنها تقبل قيمة Tcl سيتم إخراج محتوياتها إلى القناة. يتم تحويل أحرف UTF-8 في التمثيل النصي لـ writeObjPtr إلى ترميز القناة وتجميعها للإخراج إلى channel . كتحسين للأداء، عند الكتابة إلى قناة ذات ترميز binary ، لا يتم تحويل أحرف UTF-8 عند كتابتها. بدلاً من ذلك، يتم كتابة البايتات في التمثيل الداخلي لـ writeObjPtr كقيمة مصفوفة بايت إلى القناة. سيتم إنشاء تمثيل مصفوفة البايت للقيمة إذا لزم الأمر. بهذه الطريقة، يمكن قراءة البيانات الموجهة بالبايت من قناة، ومعالجتها عن طريق استدعاء Tcl_GetByteArrayFromObj والوظائف ذات الصلة، ثم كتابتها إلى قناة دون تكلفة التحويل إلى أو من UTF-8.

Tcl_Write مشابه لـ Tcl_WriteChars باستثناء أنها لا تقوم بتحويلات الترميز، بغض النظر عن ترميز القناة. إنها مهملة وموجودة للتوافق مع الإصدارات السابقة مع امتدادات Tcl غير الدولية. تقبل bytesToWrite من البايتات من byteBuf وتجمعها للإخراج إلى channel . إذا كانت bytesToWrite قيمتها سالبة، فإن Tcl_Write تتوقع أن يكون byteBuf له نهاية فارغة وتقوم بإخراج كل شيء حتى النهاية الفارغة.

Tcl_WriteRaw هو نفس Tcl_Write ولكنه لا يعوض عن التكديس. بينما Tcl_Write (والوظائف الأخرى في واجهة برمجة التطبيقات) تغذي مدخلاتها دائمًا إلى القناة العلوية في المكدس الذي تنتمي إليه القناة المقدمة، فإن Tcl_WriteRaw لا تفعل ذلك. لذلك، هذه الوظيفة قابلة للاستخدام *فقط* لمحركات القنوات التحويلية، أي المحركات المستخدمة في منتصف مكدس من القنوات، لنقل البيانات من التحويل إلى القناة أسفله.

TCL_FLUSH

Tcl_Flush يتسبب في كتابة جميع بيانات الإخراج المخزنة مؤقتًا للقناة channel إلى الملف أو الجهاز الأساسي لها في أقرب وقت ممكن. إذا كانت القناة في الوضع المانع، فإن الاستدعاء لا يعود حتى يتم إرسال جميع البيانات المخزنة مؤقتًا إلى القناة أو يحدث خطأ ما. يعود الاستدعاء فورًا إذا كانت القناة غير مانعة؛ يبدأ عملية إفراغ في الخلفية ستقوم بكتابة البيانات المخزنة مؤقتًا إلى القناة في النهاية، بأسرع ما يمكن للقناة استيعابها.

القيمة المرجعة عادة ما تكون TCL_OK . في حالة حدوث خطأ، تُرجع Tcl_Flush TCL_ERROR وتسجل رمز خطأ POSIX يمكن استرداده باستخدام Tcl_GetErrno .

TCL_SEEK

Tcl_Seek ينقل نقطة الوصول في channel حيث سيتم قراءة أو كتابة البيانات اللاحقة. يتم إفراغ الإخراج المخزن مؤقتًا إلى القناة ويتم تجاهل المدخلات المخزنة مؤقتًا، قبل عملية البحث.

Tcl_Seek يعيد عادة نقطة الوصول الجديدة. في حالة حدوث خطأ، تُرجع Tcl_Seek -1 وتسجل رمز خطأ POSIX يمكن استرداده باستخدام Tcl_GetErrno . بعد الخطأ، قد يتم نقل نقطة الوصول أو قد لا يتم نقلها.

TCL_TELL

Tcl_Tell تُرجع نقطة الوصول الحالية لقناة. القيمة المرجعة هي -1 إذا كانت القناة لا تدعم البحث.

TCL_TRUNCATECHANNEL

Tcl_TruncateChannel تقوم بقص الملف الأساسي للقناة channel إلى طول length محدد من البايتات. تُرجع TCL_OK إذا نجحت العملية، و TCL_ERROR بخلاف ذلك.

TCL_GETCHANNELOPTION
