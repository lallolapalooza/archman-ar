TCGETS.2const

TCGETS(2const) TCGETS(2const)

الاسم

TCGETS, TCSETS, TCSETSW, TCSETSF, TCGETS2, TCSETS2, TCSETSW2, TCSETSF2, TCGETA, TCSETA, TCSETAW, TCSETAF - الحصول على سمات الجهاز الطرفي وتعيينها

المكتبة

مكتبة C القياسية (libc, -lc)

الوصف

TCGETS
يعادل tcgetattr(fd, argp).
الحصول على إعدادات المنفذ التسلسلي الحالية.
TCSETS
يعادل tcsetattr(fd, TCSANOW, argp).
تعيين إعدادات المنفذ التسلسلي الحالية.
TCSETSW
يعادل tcsetattr(fd, TCSADRAIN, argp).
السماح لمخزن الإخراج بالانتهاء، وتع ين إعدادات المنفذ التسلسلي الحالية.
TCSETSF
يعادل tcsetattr(fd, TCSAFLUSH, argp).
السماح لمخزن الإخراج بالانتهاء، وتجاهل المدخلات المعلقة، وتع ين إعدادات المنفذ التسلسلي الحالية.

الأربعة استدعاءات ioctl التالية هي تمامًا مثل TCGETS، TCSETS، TCSETSW، TCSETSF، باستثناء أنها تأخذ struct termios2 * بدلاً من struct termios *. إذا كان عضو الهيكل c_cflag يحتوي على العلامة BOTHER، فسيتم تخزين معدل الباود في أعضاء الهيكل c_ispeed و c_ospeed كقيم صحيحة. هذه الاستدعاءات ioctl غير مدعومة على جميع البنى.

الأربعة استدعاءات ioctl التالية هي تمامًا مثل TCGETS، TCSETS، TCSETSW، TCSETSF، باستثناء أنها تأخذ struct termio * بدلاً من struct termios *.

قيمة الإرجاع

عند النجاح، يتم إرجاع 0. عند حدوث خطأ، يتم إرجاع -1 ويتم تعيين errno لتوضيح الخطأ.

محاذير

struct termios من <asm/termbits.h> مختلف وغير متوافق مع struct termios من <termios.h>. تتطلب استدعاءات ioctl هذه struct termios من <asm/termbits.h>.

أمثلة

الحصول على أو تعيين معدل باود عشوائي على المنفذ التسلسلي.

/* SPDX-License-Identifier: GPL-2.0-or-later */
#include <asm/termbits.h>
#include <fcntl.h>
#include <stdio.h>
#include <stdlib.h>
#include <sys/ioctl.h>
#include <unistd.h>
int
main(int argc, char *argv[])
{
#if !defined BOTHER
    fprintf(stderr, "BOTHER غير مدعوم\n");
    /* قد يتراجع البرنامج إلى TCGETS/TCSETS مع ثوابت Bnnn */
    exit(EXIT_FAILURE);
#else
    /* تعريف هيكل tio، يعتمد نوعه على ioctl المدعوم */
# if defined TCGETS2
    struct termios2 tio;
# else
    struct termios tio;
# endif
    int fd, rc;
    if (argc != 2 && argc != 3 && argc != 4) {
        fprintf(stderr, "الاستخدام: %s device [output [input] ]\n", argv[0]);
        exit(EXIT_FAILURE);
    }
    fd = open(argv[1], O_RDWR | O_NONBLOCK | O_NOCTTY);
    if (fd < 0) {
        perror("open");
        exit(EXIT_FAILURE);
    }
    /* الحصول على إعدادات المنفذ التسلسلي الحالية عبر ioctl المدعوم */
# if defined TCGETS2
    rc = ioctl(fd, TCGETS2, &tio);
# else
    rc = ioctl(fd, TCGETS, &tio);
# endif
    if (rc) {
        perror("TCGETS");
        close(fd);
        exit(EXIT_FAILURE);
    }
    /* تغيير معدل الباود عندما تم تقديم وسائط إضافية */
    if (argc == 3 || argc == 4) {
        /* مسح معدل الباود الإخراج الحالي وملء قيمة جديدة */
        tio.c_cflag &= ~CBAUD;
        tio.c_cflag |= BOTHER;
        tio.c_ospeed = atoi(argv[2]);
        /* مسح معدل الباود الإدخال الحالي وملء قيمة جديدة */
        tio.c_cflag &= ~(CBAUD << IBSHIFT);
        tio.c_cflag |= BOTHER << IBSHIFT;
        /* عندما لا يتم توفير الوسيط الرابع، استخدم نفس معدل الباود للإخراج */
        tio.c_ispeed = (argc == 4) ? atoi(argv[3]) : atoi(argv[2]);
        /* تعيين إعدادات المنفذ التسلسلي الجديدة عبر ioctl المدعوم */
# if defined TCSETS2
        rc = ioctl(fd, TCSETS2, &tio);
# else
        rc = ioctl(fd, TCSETS, &tio);
# endif
        if (rc) {
            perror("TCSETS");
            close(fd);
            exit(EXIT_FAILURE);
        }
        /* والحصول على القيم الجديدة التي تم تكوينها بالفعل */
# if defined TCGETS2
        rc = ioctl(fd, TCGETS2, &tio);
# else
        rc = ioctl(fd, TCGETS, &tio);
# endif
        if (rc) {
            perror("TCGETS");
            close(fd);
            exit(EXIT_FAILURE);
        }
    }
    close(fd);
    printf("معدل الباود للإخراج: %u\n", tio.c_ospeed);
    printf("معدل الباود للإدخال: %u\n", tio.c_ispeed);
    exit(EXIT_SUCCESS);
#endif
}
