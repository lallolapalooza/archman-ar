المعاملات

Tcl_Interp *interp (مدخل)
يستخدم للإبلاغ عن الأخطاء وللبحث عن قناة مسجلة فيه.
const char *fileName (مدخل)
اسم ملف محلي أو شبكي.
const char *mode (مدخل)
يحدد كيفية الوصول إلى الملف. يمكن أن يأخذ أي من القيم المسموح بها لمعامل mode لأمر Tcl open.
int permissions (مدخل)
علامات الأذونات بأسلوب POSIX مثل 0644. إذا تم إنشاء ملف جديد، فسيتم تعيين هذه الأذونات على الملف المنشأ.
int argc (مدخل)
عدد العناصر في argv.
const char **argv (مدخل)
معاملات لإنشاء خط أنابيب أوامر. هذه القيم لها نفس معنى المعاملات غير التبديلية لأمر Tcl exec.
int flags (مدخل)
يحدد تصرف مقابض stdio في خط الأنابيب: مزيج ORed لـ TCL_STDIN، TCL_STDOUT، TCL_STDERR، و TCL_ENFORCE_MODE. إذا تم تعيين TCL_STDIN، فإن الإدخال القياسي للطفل الأول في الأنبوب هو قناة الأنبوب، وإلا فهو نفسه مثل الإدخال القياسي للعملية المستدعية؛ وبالمثل لـ TCL_STDOUT و TCL_STDERR. إذا لم يتم تعيين TCL_ENFORCE_MODE، فيمكن للأنبوب إعادة توجيه مقابض stdio لتجاوز مقابض stdio التي تم تعيين TCL_STDIN، TCL_STDOUT، و TCL_STDERR لها. إذا تم تعيينها، فإن هذه التوجيهات تسبب خطأ.
ClientData handle (مدخل)
مقبض خاص بنظام التشغيل للإدخال/الإخراج إلى ملف. بالنسبة لنظام Unix، هذا هو واصف ملف (file descriptor)، وبالنسبة لنظام Windows، فهو HANDLE.
int readOrWrite (مدخل)
مزيج ORed لـ TCL_READABLE و TCL_WRITABLE للإشارة إلى العمليات الصحيحة على handle.
const char *channelName (مدخل)
اسم القناة.
int *modePtr (مخرج)
يشير إلى متغير صحيح سيستقبل مزيج ORed من TCL_READABLE و TCL_WRITABLE لتحديد ما إذا كانت القناة مفتوحة للقراءة والكتابة.
const char *pattern (مدخل)
النمط للمطابقة، يتم تمريره إلى Tcl_StringMatch، أو NULL.
Tcl_Channel channel (مدخل)
قناة Tcl للإدخال أو الإخراج. يجب أن تكون قد كانت نتيجة استدعاء إجراء مثل Tcl_OpenFileChannel.
Tcl_Obj *readObjPtr (مدخل/مخرج)
مؤشر إلى قيمة Tcl لتخزين الأحرف المقروءة من القناة فيها.
int charsToRead (مدخل)
عدد الأحرف التي يجب قراءتها من القناة. إذا كان ترميز القناة هو binary، فهذا يعادل عدد البايتات التي يجب قراءتها من القناة.
int appendFlag (مدخل)
إذا كان غير صفري، فسيتم إلحاق البيانات المقروءة من القناة بالقيمة. بخلاف ذلك، ستحل البيانات محل المحتويات الحالية للقيمة.
char *readBuf (مخرج)
مخزن مؤقت لتخزين البايتات المقروءة من القناة.
int bytesToRead (مدخل)
عدد البايتات التي يجب قراءتها من القناة. يجب أن يكون المخزن المؤقت readBuf كبيراً بما يكفي لاستيعاب هذا العدد من البايتات.
Tcl_Obj *lineObjPtr (مدخل/مخرج)
مؤشر إلى قيمة Tcl لتخزين السطر المقروء من القناة فيها. سيتم إلحاق السطر المقروء بالقيمة الحالية للقيمة.
Tcl_DString *lineRead (مدخل/مخرج)
مؤشر إلى سلسلة Tcl ديناميكية لتخزين السطر المقروء من القناة فيها. يجب أن يكون قد تم تهيئته بواسطة المستدعي. سيتم إلحاق السطر المقروء بأي بيانات موجودة بالفعل في السلسلة الديناميكية.
const char *input (مدخل)
المدخلات لإضافتها إلى مخزن مؤقت للقناة.
int inputLen (مدخل)
طول المدخلات.
int addAtEnd (مدخل)
علامة تشير إلى ما إذا كان يجب إضافة المدخلات إلى نهاية المخزن المؤقت للقناة أو بدايته.
Tcl_Obj *writeObjPtr (مدخل)
مؤشر إلى قيمة Tcl سيتم إخراج محتوياتها إلى القناة.
const char *charBuf (مدخل)
مخزن مؤقت يحتوي على الأحرف المراد إخراجها إلى القناة.
const char *byteBuf (مدخل)
مخزن مؤقت يحتوي على البايتات المراد إخراجها إلى القناة.
int bytesToWrite (مدخل)
عدد البايتات المراد استهلاكها من charBuf أو byteBuf وإخراجها إلى القناة.
Tcl_WideInt offset (مدخل)
مدى نقل نقطة الوصول في القناة التي سيتم تطبيق عملية الإدخال أو الإخراج التالية عليها، مقاسة بالبايت من الموضع المحدد بواسطة seekMode. يمكن أن تكون موجبة أو سالبة.
int seekMode (مدخل)
النقطة التي سيتم البحث بالنسبة لها؛ تستخدم مع offset لحساب نقطة الوصول الجديدة للقناة. القيم القانونية هي SEEK_SET، SEEK_CUR، و SEEK_END.
Tcl_WideInt length (مدخل)
الطول (غير السالب) لقطع القناة.
const char *optionName (مدخل)
اسم خيار قابل للتطبيق على هذه القناة، مثل -blocking. يمكن أن يأخذ أي من القيم التي يقبلها أمر fconfigure.
Tcl_DString *optionValue (مدخل)
المكان الذي سيتم فيه تخزين قيمة خيار أو قائمة بجميع الخيارات وقيمها. يجب أن يكون قد تم تهيئته بواسطة المستدعي.
const char *newValue (مدخل)
القيمة الجديدة للخيار المحدد بواسطة optionName.

الوصف

يوفر آلية قناة Tcl آليات مستقلة عن الأجهزة والمنصات لتنفيذ عمليات الإدخال/الإخراج المخزنة مؤقتاً على أنواع مختلفة من الملفات والمقابس والأجهزة. آلية القناة قابلة للتوسيع إلى أنواع قنوات جديدة، عن طريق توفير مشغل قناة منخفض المستوى للنوع الجديد؛ واجهة مشغل القناة موصوفة في صفحة الدليل لـ Tcl_CreateChannel. توفر آلية القناة مخطط تخزين مؤقت مشابه لـ I/O القياسي في Unix، كما تسمح بتنفيذ I/O غير محظور على القنوات.

الإجراءات الموصوفة في هذا الدليل تشكل واجهات برمجة التطبيقات العامة لطبقة القناة. لوصف بنية مشغل القناة وكيفية تنفيذ مشغلات القنوات لأنواع القنوات الجديدة، استشر صفحة الدليل لـ Tcl_CreateChannel.
