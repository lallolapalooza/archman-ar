tc-ematch(8)

الحزمة: iproute2

التراخيص: GPL-2.0-or-later

ematch(8) Linux ematch(8)

الاسم

ematch - مطابقات موسعة للاستخدام مع مرشحات "basic" أو "cgroup" أو "flow"

الملخص

tc filter add .. basic match EXPR .. flowid ..

EXPR := TERM [ { and | or } EXPR ]

TERM := [ not ] { MATCH | '(' EXPR ')' }

MATCH := module '(' ARGS ')'

ARGS := ARG1 ARG2 ..

المطابقات

cmp

مطابقة مقارنة بسيطة: مقارنة حسابية لبيانات الحزمة بقيمة معينة.

cmp( ALIGN at OFFSET [ ATTRS ] { eq | lt | gt } VALUE )

ALIGN := { u8 | u16 | u32 }

ATTRS := [ layer LAYER ] [ mask MASK ] [ trans ]

LAYER := { link | network | transport | 0..2 }

meta

مطابقة البيانات الوصفية (Metadata)

meta( OBJECT { eq | lt |gt } OBJECT )

OBJECT := { META_ID | VALUE }

META_ID := id [ shift SHIFT ] [ mask MASK ]

سمات البيانات الوصفية:

random قيمة عشوائية 32 بت

loadavg_1 متوسط الحمل في آخر 5 دقائق

nf_mark علامة Netfilter

vlan علامة Vlan

sk_rcvbuf حجم مخزن الاستقبال المؤقت

sk_snd_queue طول طابور الإرسال

يمكن الحصول على قائمة كاملة بسمات البيانات الوصفية عبر

# tc filter add dev eth1 basic match 'meta(list)'

nbyte

مطابقة تسلسل بايت بيانات الحزمة

nbyte( NEEDLE at OFFSET [ layer LAYER ] )

NEEDLE := { string | c-escape-sequence }

OFFSET := int

LAYER := { link | network | transport | 0..2 }

u32

مطابقة u32

u32( ALIGN VALUE MASK at [ nexthdr+ ] OFFSET )

ALIGN := { u8 | u16 | u32 }

ipset

اختبار عضوية الحزمة في ipset

ipset( SETNAME FLAGS )

SETNAME := string

FLAGS := { FLAG [, FLAGS] }

خيارات العلامة هي نفسها تلك المستخدمة في مطابقة "set" الخاصة بـ iptables.

عند استخدام مطابقة ipset مع نوع المجموعة "ip_set_hash:net,iface"، يمكن الاستعلام عن الواجهة باستخدام الصيغة "src,dst (عنوان IP المصدر، الواجهة الخارجة) أو "src,src" (عنوان IP المصدر، الواجهة الواردة).

ipt

اختبار الحزمة مقابل مطابقات xtables

ipt( [-6] -m MATCH_NAME FLAGS )

MATCH_NAME := string

FLAGS := { FLAG [, FLAGS] }

خيارات العلامة هي نفسها تلك المستخدمة في مطابقة xtable المستخدمة.

canid

قاعدة ematch لمطابقة إطارات CAN

canid( IDLIST )

IDLIST := IDSPEC[IDLIST]

IDSPEC := { ’sff’ CANID | ’eff’ CANID }

CANID := ID[:MASK]

ID, MASK := رقم سداسي عشري (مثل 0x123)

تحذيرات

تستخدم صيغة ematch أقواس '(' و ')' لتجميع التعابير. يجب الهروب من جميع الأقواس بشكل صحيح لمنع سطر أوامر shell من تفسيرها مباشرة.

عند استخدام مطابقة ipset مع الجهاز "ifb"، سيكون الجهاز الخارج هو جهاز ifb نفسه، على سبيل المثال "ifb0". يتم التعامل مع الواجهة الأصلية (أي الجهاز الذي وصلت عليه الحزمة) على أنها الواجهة الواردة.

مثال & الاستخدام

# tc filter add .. basic match ...

# 'cmp(u16 at 3 layer 2 mask 0xff00 gt 20)'

# 'meta(nfmark gt 24)' and 'meta(tcindex mask 0xf0 eq 0xf0)'

# 'nbyte("ababa" at 12 layer 1)'

# 'u32(u16 0x1122 0xffff at nexthdr+4)'

التحقق مما إذا كان عنوان IP المصدر للحزمة عضوًا في مجموعة تسمى bulk:

# 'ipset(bulk src)'

التحقق مما إذا كان عنوان IP المصدر للحزمة والواجهة التي وصلت إليها الحزمة عضوين في مجموعة "hash:net,iface" تسمى interactive:

# 'ipset(interactive src,src)'

التحقق مما إذا كانت الحزمة تتطابق مع حالة IPSec بمعرف مطلب (reqid) 1:

# 'ipt(-m policy --dir in --pol ipsec --reqid 1)'
