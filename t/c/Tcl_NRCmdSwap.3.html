Tcl_NRCmdSwap(3)

الحزمة: tcl

التراخيص: TCL

NRE(3) إجراءات مكتبة Tcl NRE(3)

الاسم

Tcl_NRCreateCommand, Tcl_NRCallObjProc, Tcl_NREvalObj, Tcl_NREvalObjv, Tcl_NRCmdSwap, Tcl_NRExprObj, Tcl_NRAddCallback - التقييم غير العودي (بدون مكدس) لسكريبتات Tcl.

الموجز

#include <tcl.h>

Tcl_Command
Tcl_NRCreateCommand(interp, cmdName, proc, nreProc, clientData,
                    deleteProc)

int
Tcl_NRCallObjProc(interp, nreProc, clientData, objc, objv)

int
Tcl_NREvalObj(interp, objPtr, flags)

int
Tcl_NREvalObjv(interp, objc, objv, flags)

int
Tcl_NRCmdSwap(interp, cmd, objc, objv, flags)

int
Tcl_NRExprObj(interp, objPtr, resultPtr)

void
Tcl_NRAddCallback(interp, postProcPtr, data0, data1, data2, data3)

المعاملات

Tcl_Interp *interp (في)

const char *cmdName (في)
اسم الأمر المراد إنشاؤه.
Tcl_ObjCmdProc *proc (في)
يتم استدعاؤه لتقييم أمر. غالبًا ما يكون مجرد غلاف صغير يستخدم Tcl_NRCallObjProc لاستدعاء nreProc باستخدام قفزة جديدة. يتصرف بنفس الطريقة التي يتصرف بها المعامل proc في Tcl_CreateObjCommand(3) (راجع).
Tcl_ObjCmdProc *nreProc (في)
يتم استدعاؤه بدلاً من proc عند استخدام قفزة بالفعل.
ClientData clientData (في)
قيمة ذات كلمة واحدة عشوائية يتم تمريرها إلى proc و nreProc و deleteProc و objProc.
Tcl_CmdDeleteProc *deleteProc (في/خارج)
يتم استدعاؤه قبل حذف cmdName من المفسر، مما يسمح بالتنظيف الخاص بالأمر. قد يكون NULL.
int objc (في)
عدد العناصر في objv.
Tcl_Obj **objv (في)
كلمات في الأمر.
Tcl_Obj *objPtr (في)
سكريبت أو تعبير للتقييم.
int flags (في)
كما هو موضح لـ Tcl_EvalObjv.
Tcl_Command cmd (في)
رمز مميز لاستخدامه بدلاً من الرمز المشتق من الكلمة الأولى في objv لتقييم أمر.
Tcl_Obj *resultPtr (خارج)
مؤشر إلى Tcl_Obj غير مشترك حيث يتم تخزين نتيجة التقييم إذا كان رمز الإرجاع هو TCL_OK.
Tcl_NRPostProc *postProcPtr (في)
دالة لدفعها.
ClientData data0 (في)
ClientData data1 (في)
ClientData data2 (في)
ClientData data3 (في)
data0 حتى data3 هي أربع قيم بكلمة واحدة سيتم تمريرها إلى الدالة المحددة بواسطة postProcPtr عند استدعائها.

الوصف

توفر هذه الدوال واجهة لمكدس الوظائف الذي يتكرر من خلاله المفسر لتقييم الأوامر. يتم تنفيذ الروتين وراء الأمر بواسطة دالة أولية وأي دوال إضافية يدفعها الروتين على المكدس أثناء تقدمه. يقوم المفسر نفسه بدفع الدوال على المكدس للتفاعل مع نهاية الروتين وممارسة أشكال أخرى من التحكم مثل التبديل بين المكدسات قيد التقدم وتقييم سكريبتات أخرى بمستويات إضافية دون إضافة إطارات إلى مكدس C. لتنفيذ روتين، يتم استدعاء الدالة الأولية للروتين ثم تقوم قطعة صغيرة من التعليمات البرمجية تسمى قفزة بشكل تكراري بأخذ الدوال من المكدس واستدعائها، باستخدام قيمة آخر استدعاء كقيمة للروتين.

Tcl_NRCallObjProc يستدعي nreProc باستخدام قفزة جديدة.

Tcl_NRCreateCommand، بديل لـ Tcl_CreateObjCommand، يحل cmdName، الذي قد يحتوي على مؤهلات مساحة الاسم، بالنسبة للمساحة الاسمية الحالية، وينشئ أمرًا بهذا الاسم، ويعيد رمزًا مميزًا للأمر يمكن استخدامه في استدعاءات لاحقة لـ Tcl_GetCommandName. باستثناء بعض الحالات المذكورة أدناه، يتم أولاً حذف أي أمر موجود بنفس الاسم. إذا كان interp قيد الحذف، فإن Tcl_NRCreateCommand لا ينشئ أي أمر، ولا يحذف أي أمر، ويعيد NULL.

Tcl_NREvalObj يدفع دالة تشبه Tcl_EvalObjEx ولكنها لا تستهلك مساحة على مكدس C.

Tcl_NREvalObjv يدفع دالة تشبه Tcl_EvalObjv ولكنها لا تستهلك مساحة على مكدس C.

Tcl_NRCmdSwap يشبه Tcl_NREvalObjv، ولكنه يستخدم cmd، وهو رمز مميز تم إرجاعه مسبقًا بواسطة Tcl_CreateObjCommand أو Tcl_GetCommandFromObj، بدلاً من حل الكلمة الأولى في objv.

Tcl_NRExprObj يدفع دالة تقيّم objPtr كتعبير بنفس الطريقة التي تقيم بها Tcl_ExprObj ولكن دون استهلاك مساحة على مكدس C.

ترجع جميع الدوال TCL_OK إذا تم جدولة تقييم السكريبت أو الأمر أو التعبير بنجاح. خلاف ذلك (على سبيل المثال، إذا لم يكن من الممكن حل اسم الأمر)، فإنها ترجع TCL_ERROR وتخزن رسالة كنتيجة للمفسر.

Tcl_NRAddCallback يدفع postProcPtr. توقيع Tcl_NRPostProc هو:

typedef int
Tcl_NRPostProc(
        ClientData data[],
        Tcl_Interp *interp,
        int result);

data هو مؤشر إلى مصفوفة تحتوي على data0 حتى data3. result هي القيمة التي تم إرجاعها بواسطة الدالة السابقة التي تنفذ جزءًا من الروتين.

مثال

يستخدم الأمر التالي Tcl_EvalObjEx، الذي يستهلك مساحة على مكدس C، لتقييم سكريبت:

int
TheCmdOldObjProc(
    ClientData clientData,
    Tcl_Interp *interp,
    int objc,
    Tcl_Obj *const objv[])
{
    int result;
    Tcl_Obj *objPtr;
    ... preparation ...
    result = Tcl_EvalObjEx(interp, objPtr, 0);
    ... postprocessing ...
    return result;
}
Tcl_CreateObjCommand(interp, "theCommand",
        TheCmdOldObjProc, clientData, TheCmdDeleteProc);

لتجنب استهلاك مساحة على مكدس C، يتم إعادة تسمية TheCmdOldObjProc إلى TheCmdNRObjProc ويتم تقسيم خطوة ما بعد المعالجة إلى دالة منفصلة، TheCmdPostProc، التي يتم دفعها على مكدس الوظائف. يتم استبدال Tcl_EvalObjEx بـ Tcl_NREvalObj، الذي يستخدم قفزة بدلاً من استهلاك مساحة على مكدس C. الإصدار الجديد من TheCmdOldObjProc هو مجرد غلاف يستخدم Tcl_NRCallObjProc لاستدعاء TheCmdNRObjProc:

int
TheCmdOldObjProc(
    ClientData clientData,
    Tcl_Interp *interp,
    int objc,
    Tcl_Obj *const objv[])
{
    return Tcl_NRCallObjProc(interp, TheCmdNRObjProc,
            clientData, objc, objv);
}
int
TheCmdNRObjProc
    ClientData clientData,
    Tcl_Interp *interp,
    int objc,
    Tcl_Obj *const objv[])
{
    Tcl_Obj *objPtr;
    ... preparation ...
    Tcl_NRAddCallback(interp, TheCmdPostProc,
            data0, data1, data2, data3);
    /* data0 .. data3 هي ما يصل إلى أربعة عناصر بكلمة واحدة لتمريرها
     * إلى إجراء ما بعد المعالجة */
    return Tcl_NREvalObj(interp, objPtr, 0);
}
int
TheCmdNRPostProc(
    ClientData data[],
    Tcl_Interp *interp,
    int result)
{
    /* data[0] .. data[3] هي الكلمات الأربعة للبيانات
     * التي تم تمريرها إلى Tcl_NRAddCallback */
    ... postprocessing ...
    return result;
}

يمكن لأي دالة تشكل روتينًا أن تدفع دوال أخرى، مما يجعل من الممكن تنفيذ تراكيب حلقية وتسلسلية باستخدام مكدس الوظائف.
