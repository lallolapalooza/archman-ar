الوسائط

Tcl_Interp *interp (في)
مفسر Tcl لاستخدامه في الإبلاغ عن الأخطاء. يمكن أن يكون المفسر NULL إذا لم يكن مطلوبًا الإبلاغ عن الأخطاء.
Tcl_Obj *textObj (في/خارج)
يشير إلى القيمة التي سيتم الحصول على النص للبحث فيها. قد يتم تحويل التمثيل الداخلي للقيمة إلى شكل يمكن البحث فيه بكفاءة.
Tcl_Obj *patObj (في/خارج)
يشير إلى القيمة التي سيتم الحصول على تعبير عادي منها. يتم تخزين التعبير العادي المترجم مؤقتًا في القيمة.
const char *text (في)
النص للبحث عن مطابقة بتعبير عادي.
const char *pattern (في)
سلسلة نصية في شكل نمط تعبير عادي.
Tcl_RegExp regexp (في)
تعبير عادي مترجم. يجب أن يكون قد تم إرجاعه مسبقًا بواسطة Tcl_GetRegExpFromObj أو Tcl_RegExpCompile.
const char *start (في)
إذا كان text مجرد جزء من سلسلة أخرى، فهذه الوسيطة تحدد بداية السلسلة الأكبر. إذا لم تكن هي نفسها text، فلن يُسمح بأي مطابقات “^”.
int index (في)
يحدد أي نطاق مطلوب: 0 يعني نطاق المطابقة الكاملة، 1 أو أكبر يعني النطاق الذي طابق تعبيرًا فرعيًا محاطًا بأقواس.
const char **startPtr (خارج)
يتم تخزين عنوان الحرف الأول في النطاق هنا، أو NULL إذا لم يكن هناك نطاق.
const char **endPtr (خارج)
يتم تخزين عنوان الحرف الذي يلي الأخير في النطاق هنا، أو NULL إذا لم يكن هناك نطاق.
int cflags (في)
مزيج OR-ed من علامات الترجمة TCL_REG_ADVANCED، TCL_REG_EXTENDED، TCL_REG_BASIC، TCL_REG_EXPANDED، TCL_REG_QUOTE، TCL_REG_NOCASE، TCL_REG_NEWLINE، TCL_REG_NLSTOP، TCL_REG_NLANCH، TCL_REG_NOSUB، و TCL_REG_CANMATCH. انظر أدناه لمزيد من المعلومات.
int offset (في)
إزاحة الحرف في النص حيث يجب أن يبدأ المطابقة. قيمة الإزاحة ليس لها تأثير على مطابقات ^. يتم التحكم في هذا السلوك بواسطة eflags.
int nmatches (في)
عدد التعبيرات الفرعية المطابقة التي يجب تذكرها للاستخدام لاحقًا. إذا كانت هذه القيمة 0، فلن يتم حساب أي معلومات مطابقة للتعبير الفرعي. إذا كانت القيمة -1، فسيتم تذكر جميع التعبيرات الفرعية المطابقة. سيتم اعتبار أي قيمة أخرى كحد أقصى لعدد التعبيرات الفرعية التي يجب تذكرها.
int eflags (في)
مزيج OR-ed من علامات التنفيذ TCL_REG_NOTBOL و TCL_REG_NOTEOL. انظر أدناه لمزيد من المعلومات.
Tcl_RegExpInfo *infoPtr (خارج)
عنوان الموقع الذي يجب تخزين المعلومات حول مطابقة سابقة فيه بواسطة Tcl_RegExpGetInfo.

الوصف

Tcl_RegExpMatch تحدد ما إذا كانت وسيطتها pattern تطابق regexp، حيث يتم تفسير regexp كتعبير عادي باستخدام القواعد في صفحة المرجع re_syntax. إذا كانت هناك مطابقة، فإن Tcl_RegExpMatch ترجع 1. إذا لم تكن هناك مطابقة، فإن Tcl_RegExpMatch ترجع 0. إذا حدث خطأ في عملية المطابقة (على سبيل المثال، pattern ليس تعبيرًا عاديًا صالحًا)، فإن Tcl_RegExpMatch ترجع -1 وتترك رسالة خطأ في نتيجة المفسر. Tcl_RegExpMatchObj مشابهة لـ Tcl_RegExpMatch باستثناء أنها تعمل على قيم Tcl textObj و patObj بدلاً من سلاسل UTF. Tcl_RegExpMatchObj هي بشكل عام أكثر كفاءة من Tcl_RegExpMatch، لذا فهي الواجهة المفضلة.

Tcl_RegExpCompile، Tcl_RegExpExec، و Tcl_RegExpRange توفر وصولاً أقل مستوى إلى معالج أنماط التعبيرات العادية. Tcl_RegExpCompile يترجم سلسلة تعبير عادي إلى الشكل الداخلي المستخدم للمطابقة الفعالة للأنماط. القيمة المرجعة هي رمز لهذا الشكل المترجم، والذي يمكن استخدامه في استدعاءات لاحقة لـ Tcl_RegExpExec أو Tcl_RegExpRange. إذا حدث خطأ أثناء ترجمة التعبير العادي، فإن Tcl_RegExpCompile ترجع NULL وتترك رسالة خطأ في نتيجة المفسر. ملاحظة: القيمة المرجعة من Tcl_RegExpCompile صالحة فقط حتى الاستدعاء التالي لـ Tcl_RegExpCompile؛ ليس من الآمن الاحتفاظ بهذه القيم لفترات طويلة.

Tcl_RegExpExec تنفذ معالج أنماط التعبير العادي. ترجع 1 إذا كان text يحتوي على نطاق من الأحرف يطابق regexp، 0 إذا لم يتم العثور على مطابقة، و -1 إذا حدث خطأ. في حالة حدوث خطأ، تترك Tcl_RegExpExec رسالة خطأ في نتيجة المفسر. عند البحث في سلسلة عن مطابقات متعددة لنمط، من المهم التمييز بين بداية السلسلة الأصلية وبداية البحث الحالي. على سبيل المثال، عند البحث عن تكرار ثانٍ لمطابقة، قد يشير الوسيط text إلى الحرف الذي يلي المطابقة الأولى؛ ومع ذلك، من المهم لمعالج النمط معرفة أن هذه ليست بداية السلسلة بأكملها، حتى لا يسمح لذرات “^” في النمط بالمطابقة. الوسيط start يوفر هذه المعلومات بالإشارة إلى بداية السلسلة الكاملة التي تحتوي على text. سيكون Start أقل من أو يساوي text؛ إذا كان أقل من text، فلن يُسمح بمطابقات ^.

Tcl_RegExpRange يمكن استدعاؤها بعد أن ترجع Tcl_RegExpExec؛ فهي توفر معلومات مفصلة حول نطاقات السلسلة التي طابقت أجزاء النمط. Tcl_RegExpRange ترجع زوجًا من المؤشرات في *startPtr و *endPtr التي تحدد نطاقًا من الأحرف في السلسلة المصدر لآخر استدعاء لـ Tcl_RegExpExec. يشير index إلى أي من النطاقات المتعددة مرغوب فيه: إذا كان index هو 0، يتم إرجاع معلومات حول النطاق العام للأحرف الذي طابق النمط بأكمله؛ بخلاف ذلك، يتم إرجاع معلومات حول نطاق الأحرف الذي طابق التعبير الفرعي المحاط بأقواس ذي الرقم index. إذا لم يكن هناك نطاق يتوافق مع index، يتم تخزين NULL في *startPtr و *endPtr.

Tcl_GetRegExpFromObj، Tcl_RegExpExecObj، و Tcl_RegExpGetInfo هي واجهات قيم توفر التحكم الأكثر مباشرة لمكتبة التعبيرات العادية لهنري سبنسر. للمستخدمين الذين يحتاجون إلى تعديل خيارات الترجمة والتنفيذ مباشرة، يوصى باستخدام هذه الواجهات بدلاً من استدعاء وظائف التعبيرات العادية الداخلية. تتعامل هذه الواجهات مع تفاصيل ترجمات UTF إلى Unicode بالإضافة إلى توفير تحسينات في الأداء من خلال التخزين المؤقت في قيم النمط والسلسلة.

Tcl_GetRegExpFromObj تحاول إرجاع تعبير عادي مترجم من patObj. إذا لم تحتوِ القيمة بالفعل على تعبير عادي مترجم، فستحاول إنشائه من السلسلة في القيمة وتعيينه للتمثيل الداخلي لـ patObj. القيمة المرجعة لهذه الوظيفة هي من النوع Tcl_RegExp. القيمة المرجعة هي رمز لهذا الشكل المترجم، والذي يمكن استخدامه في استدعاءات لاحقة لـ Tcl_RegExpExecObj أو Tcl_RegExpGetInfo. إذا حدث خطأ أثناء ترجمة التعبير العادي، فإن Tcl_GetRegExpFromObj ترجع NULL وتترك رسالة خطأ في نتيجة المفسر. يمكن استخدام رمز التعبير العادي طالما أن التمثيل الداخلي لـ patObj يشير إلى الشكل المترجم. الوسيط cflags هو OR ثنائي من علامات واحدة أو أكثر مما يلي والتي تتحكم في ترجمة patObj:

Tcl_RegExpGetInfo(3) — Tcl

NAME

Tcl_RegExpGetInfo, Tcl_RegExpExecObj, Tcl_RegExpMatchObj — get information about regular expression matches

SYNOPSIS

#include <tcl.h>

int
Tcl_RegExpExecObj(interp, regexp, objPtr, nmatches, monitored, offset, eflags);
int
Tcl_RegExpGetInfo(interp, regexp, infoPtr);
int
Tcl_RegExpMatchObj(interp, regexp, objPtr, offset, eflags);

ARGUMENTS

Tcl_Interp *interp
مفسر Tcl.
Tcl_RegExp regexp
معرّف تعبير منتظم تم إنشاؤه بواسطة Tcl_RegExpCompile.
Tcl_Obj *objPtr
الكائن الذي يحتوي على السلسلة المراد البحث فيها.
int nmatches
عدد التعبيرات الفرعية التي سيتم تسجيل معلوماتها.
Tcl_RegExpIndices *monitored
مؤشر لمصفوفة لتخزين مؤشرات المطابقة للتعبيرات الفرعية. يجب أن يكون حجم المصفوفة nmatches+1.
long offset
يبدأ البحث عن المطابقة من هذه المؤشر (عدد الأحرف Unicode).
int eflags
أعلام لتعديل سلوك المطابقة (مثل TCL_REG_NOTBOL).
Tcl_RegExpInfo *infoPtr
مؤشر لبنية لتخزين معلومات حول المطابقة.

DESCRIPTION

تقوم هذه الدوال بتنفيذ المطابقة باستخدام التعبيرات المنتظمة في Tcl. تُستخدم Tcl_RegExpCompile لتجميع تعبير منتظم في شكل داخلي يمكن استخدامه بواسطة دوال المطابقة.

تُعدّ Tcl_RegExpExecObj هي الدالة الرئيسية للمطابقة. إنها تُرجع 1 إذا تم العثور على مطابقة، 0 إذا لم يتم العثور على مطابقة، و -1 إذا حدث خطأ. يتم تخزين معلومات المطابقة في البنية monitored.

تُستخدم Tcl_RegExpGetInfo لاسترداد معلومات حول المطابقة الأخيرة التي تم إجراؤها باستخدام تعبير منتظم معين.

تُعدّ Tcl_RegExpMatchObj نسخة مبسطة من Tcl_RegExpExecObj تُرجع فقط ما إذا كانت المطابقة ناجحة أم لا، ولا تُرجع أي معلومات حول المطابقة.

تُستخدم الأعلام التالية لتعديل سلوك التجميع والمطابقة:

TCL_REG_ADVANCED
تجميع تعبيرات منتظمة متقدمة ("ARE"s). يطابق هذا الوضع بناء الجملة العادي للتعبير المنتظم الذي تقبله أوامر regexp و regsub الخاصة بـ Tcl.
TCL_REG_EXTENDED
تجميع تعبيرات منتظمة موسعة ("ERE"s). يطابق هذا الوضع بناء الجملة للتعبير المنتظم الذي تعرفه Tcl 8.0 والإصدارات الأقدم.
TCL_REG_BASIC
تجميع تعبيرات منتظمة أساسية ("BRE"s). يطابق هذا الوضع بناء الجملة للتعبير المنتظم الذي تتعرف عليه أدوات Unix الشائعة مثل sed و grep. هذا هو الوضع الافتراضي إذا لم يتم تحديد أي أعلام.
TCL_REG_EXPANDED
تجميع التعبير المنتظم (أساسي، موسع، أو متقدم) باستخدام بناء جملة موسع يسمح بالتعليقات والمسافات البيضاء. يتجاهل هذا الوضع المسافات البيضاء غير المسبوقة والمسافات البيضاء خارج أقواس التعبيرات والتعليقات من علامة # إلى نهاية السطر.
TCL_REG_QUOTE
تجميع سلسلة حرفية، حيث تُعامل جميع الأحرف كأحرف عادية.
TCL_REG_NOCASE
التجميع للمطابقة التي تتجاهل فروق حالة الأحرف (كبيرة/صغيرة).
TCL_REG_NEWLINE
التجميع للمطابقة الحساسة لسطر جديد. بشكل افتراضي، سطر جديد هو حرف عادي تمامًا بدون معنى خاص في التعبيرات المنتظمة أو السلاسل. مع هذا العلم، لا تطابق تعبيرات الأقواس "[^...]" والرمز "." سطرًا جديدًا أبدًا. يطابق الرمز "^" سلسلة فارغة بعد أي سطر جديد بالإضافة إلى وظيفته العادية، ويطابق الرمز "$" سلسلة فارغة قبل أي سطر جديد بالإضافة إلى وظيفته العادية. REG_NEWLINE هو OR البت لـ REG_NLSTOP و REG_NLANCH.
TCL_REG_NLSTOP
التجميع للمطابقة الجزئية الحساسة لسطر جديد، مع تأثر سلوك تعبيرات الأقواس "[^...]" والرمز "."، ولكن ليس سلوك الرمز "^" والرمز "$". في هذا الوضع، لا تطابق تعبيرات الأقواس "[^...]" والرمز "." سطرًا جديدًا أبدًا.
TCL_REG_NLANCH
التجميع للمطابقة العكسية الجزئية الحساسة لسطر جديد، مع تأثر سلوك الرمز "^" والرمز "$" (الـ "مثبتات")، ولكن ليس سلوك تعبيرات الأقواس "[^...]" والرمز ".". في هذا الوضع، يطابق الرمز "^" سلسلة فارغة بعد أي سطر جديد بالإضافة إلى وظيفته العادية، ويطابق الرمز "$" سلسلة فارغة قبل أي سطر جديد بالإضافة إلى وظيفته العادية.
TCL_REG_NOSUB
التجميع للمطابقة التي تُبلغ فقط عن النجاح أو الفشل، وليس ما تمت مطابقته. هذا يقلل من عبء التجميع وقد يحسن الأداء. لن تُبلغ الاستدعاءات اللاحقة لـ Tcl_RegExpGetInfo أو Tcl_RegExpRange عن أي معلومات مطابقة.
TCL_REG_CANMATCH
التجميع للمطابقة التي تُبلغ عن احتمالية إكمال مطابقة جزئية بالنظر إلى مزيد من النص (انظر أدناه).

يمكن تحديد واحد فقط من TCL_REG_EXTENDED، TCL_REG_ADVANCED، TCL_REG_BASIC، و TCL_REG_QUOTE.

تُنفذ Tcl_RegExpExecObj المطابق للتعبير المنتظم. تُرجع 1 إذا كانت objPtr تحتوي على نطاق من الأحرف يطابق regexp، و 0 إذا لم يتم العثور على مطابقة، و -1 إذا حدث خطأ. في حالة حدوث خطأ، تترك Tcl_RegExpExecObj رسالة خطأ في نتيجة المفسر. تشير قيمة nmatches إلى المطابق عدد التعبيرات الفرعية ذات الأهمية. إذا كانت nmatches هي 0، فلن يتم تسجيل أي معلومات عن مطابقة التعبيرات الفرعية، مما قد يسمح للمطابق بإجراء تحسينات مختلفة. إذا كانت القيمة -1، فسيتم تذكر جميع التعبيرات الفرعية في النمط. إذا كانت القيمة عددًا صحيحًا موجبًا، فسيتم تذكر عدد التعبيرات الفرعية هذا فقط. تبدأ المطابقة عند مؤشر حرف Unicode المحدد بواسطة offset. على عكس Tcl_RegExpExec، لا يتأثر سلوك المثبتات بقيمة الإزاحة. بدلاً من ذلك، يتم التحكم صراحةً في سلوك المثبتات بواسطة الوسيط eflags، وهو OR بت لصفر أو أكثر من الأعلام التالية:

TCL_REG_NOTBOL
لن يُعامل الحرف البادئ كبداية سطر أو بداية السلسلة، لذلك لن يطابق الرمز "^" هناك. لاحظ أن هذا العلم ليس له تأثير على كيفية مطابقة الرمز "\A".
TCL_REG_NOTEOL
لن يُعامل الحرف الأخير في السلسلة كسطر نهاية أو نهاية السلسلة، لذلك لن يطابق الرمز "$" هناك. لاحظ أن هذا العلم ليس له تأثير على كيفية مطابقة الرمز "\Z".

تسترد Tcl_RegExpGetInfo معلومات حول المطابقة الأخيرة التي تم إجراؤها باستخدام تعبير منتظم معين regexp. يحتوي الوسيط infoPtr على مؤشر إلى بنية مُعرفة على النحو التالي:

typedef struct Tcl_RegExpInfo {
    int nsubs;
    Tcl_RegExpIndices *matches;
    long extendStart;
} Tcl_RegExpInfo;

يحتوي الحقل nsubs على عدد التعبيرات الفرعية الموضوعة بين قوسين داخل التعبير المنتظم. إذا تم استخدام TCL_REG_NOSUB، فستكون هذه القيمة صفرًا. يشير الحقل matches إلى مصفوفة من nsubs+1 قيم تشير إلى حدود كل تعبير فرعي مطابق. يشير العنصر الأول في المصفوفة إلى النطاق الذي طابقته التعبير المنتظم بأكمله، وتشير العناصر اللاحقة إلى التعبيرات الفرعية الموضوعة بين قوسين بالترتيب الذي تظهر به في النمط. كل عنصر هو بنية مُعرفة على النحو التالي:

typedef struct Tcl_RegExpIndices {
    long start;
    long end;
} Tcl_RegExpIndices;

القيم start و end هي مؤشرات أحرف Unicode بالنسبة لموقع الإزاحة داخل objPtr حيث بدأت المطابقة. يحدد المؤشر start الحرف الأول للتعبير الفرعي المطابق. يحدد المؤشر end الحرف الأول بعد التعبير الفرعي المطابق. إذا طابق التعبير الفرعي سلسلة فارغة، فستكون start و end متساويين. إذا لم يشارك التعبير الفرعي في المطابقة، فسيتم تعيين start و end إلى -1.

يتم تعيين الحقل extendStart في Tcl_RegExpInfo فقط إذا تم استخدام العلم TCL_REG_CANMATCH. يشير إلى أول حرف في السلسلة حيث يمكن حدوث مطابقة. إذا تم العثور على مطابقة، فسيكون هذا هو نفس بداية المطابقة الحالية. إذا لم يتم العثور على مطابقة، فإنه يشير إلى أقرب نقطة قد تحدث فيها مطابقة إذا تمت إضافة نص إضافي إلى السلسلة. إذا لم تكن المطابقة ممكنة حتى مع المزيد من النص، فسيتم تعيين هذا الحقل إلى -1.
