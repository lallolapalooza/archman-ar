Tcl_ExprBoolean(3)

الحزمة: tcl

التراخيص: TCL

Tcl_ExprLong(3) إجراءات مكتبة Tcl Tcl_ExprLong(3)

الاسم

Tcl_ExprLong، Tcl_ExprDouble، Tcl_ExprBoolean، Tcl_ExprString - تقييم تعبير

الملخص

#include <tcl.h>

int
Tcl_ExprLong(interp, expr, longPtr)

int
Tcl_ExprDouble(interp, expr, doublePtr)

int
Tcl_ExprBoolean(interp, expr, booleanPtr)

int
Tcl_ExprString(interp, expr)

الوسائط

Tcl_Interp *interp (في)
المفسر الذي سيتم في سياقه تقييم expr.
const char *expr (في)
التعبير المراد تقييمه.
long *longPtr (خارج)
مؤشر إلى الموقع الذي سيتم فيه تخزين القيمة الصحيحة للتعبير.
int *doublePtr (خارج)
مؤشر إلى الموقع الذي سيتم فيه تخزين قيمة النقطة العائمة للتعبير.
int *booleanPtr (خارج)
مؤشر إلى الموقع الذي سيتم فيه تخزين قيمة منطقية 0/1 للتعبير.

الوصف

تقوم هذه الإجراءات الأربعة جميعها بتقييم التعبير المعطى بواسطة الوسيط expr وإرجاع النتيجة بأحد أربعة أشكال مختلفة. يمكن أن يكون للتعبير أي من الأشكال التي يقبلها الأمر expr. لاحظ أنه تم استبدال هذه الإجراءات إلى حد كبير بالإجراءات المستندة إلى القيمة Tcl_ExprLongObj و Tcl_ExprDoubleObj و Tcl_ExprBooleanObj و Tcl_ExprObj. تقوم تلك الإجراءات المستندة إلى القيمة بتقييم تعبير محتفظ به في قيمة Tcl بدلاً من سلسلة نصية. يمكن للقيمة الوسيطة الاحتفاظ بتمثيل داخلي يكون أكثر كفاءة في التنفيذ.

يشير الوسيط interp إلى مفسر يستخدم لتقييم التعبير (مثل المتغيرات وأوامر Tcl المتداخلة) ولإرجاع معلومات الخطأ.

لجميع هذه الإجراءات، القيمة المرجعة هي نتيجة Tcl قياسية: TCL_OK تعني أنه تم تقييم التعبير بنجاح، و TCL_ERROR تعني حدوث خطأ أثناء تقييم التعبير. إذا تم إرجاع TCL_ERROR، فستحتوي نتيجة المفسر على رسالة تصف الخطأ. إذا حدث خطأ أثناء تنفيذ أمر Tcl مضمن في التعبير، فسيتم إرجاع هذا الخطأ.

إذا تم تقييم التعبير بنجاح، فسيتم إرجاع قيمته بأحد أربعة أشكال، اعتمادًا على الإجراء الذي تم استدعاؤه. تخزن Tcl_ExprLong قيمة صحيحة في *longPtr. إذا كانت القيمة الفعلية للتعبير رقمًا عائمًا، فسيتم اقتطاعه إلى عدد صحيح. إذا كانت القيمة الفعلية للتعبير سلسلة نصية غير رقمية، فسيتم إرجاع خطأ.

تخزن Tcl_ExprDouble قيمة نقطة عائمة في *doublePtr. إذا كانت القيمة الفعلية للتعبير عددًا صحيحًا، فسيتم تحويله إلى نقطة عائمة. إذا كانت القيمة الفعلية للتعبير سلسلة نصية غير رقمية، فسيتم إرجاع خطأ.

تخزن Tcl_ExprBoolean قيمة منطقية 0/1 في *booleanPtr. إذا كانت القيمة الفعلية للتعبير عددًا صحيحًا أو رقمًا عائمًا، فسيتم تخزين 0 في *booleanPtr إذا كانت القيمة صفرًا و 1 بخلاف ذلك. إذا كانت القيمة الفعلية للتعبير سلسلة نصية غير رقمية، فيجب أن تكون إحدى القيم التي تقبلها Tcl_GetBoolean مثل "yes" أو "no"، وإلا يحدث خطأ.

تُرجع Tcl_ExprString قيمة التعبير كسلسلة نصية مخزنة في نتيجة المفسر.
