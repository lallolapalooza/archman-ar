const Tcl_Filesystem *fsPtr (in)
يشير إلى بنية تحتوي على عناوين الإجراءات التي يمكن استدعاؤها لتنفيذ عمليات نظام الملفات المختلفة.
Tcl_Obj *pathPtr (in)
المسار الذي تمثله هذه القيمة يُستخدم للعملية المعنية. إذا لم تكن للقيمة تمثيل مسار داخلي بالفعل، فسيتم تحويلها لتكون كذلك.
Tcl_Obj *srcPathPtr (in)
كما هو الحال مع pathPtr، ولكن يُستخدم للملف المصدر لعملية نسخ أو إعادة تسمية.
Tcl_Obj *destPathPtr (in)
كما هو الحال مع pathPtr، ولكن يُستخدم لاسم الملف الوجهة لعملية نسخ أو إعادة تسمية.
int recursive (in)
ما إذا كان سيتم إزالة الدلائل الفرعية ومحتوياتها أيضًا.
const char *encodingName (in)
ترميز البيانات المخزنة في الملف المحدد بواسطة pathPtr والتي سيتم تقييمها.
const char *pattern (in)
فقط الملفات أو الدلائل التي تتطابق مع هذا النمط سيتم إرجاعها.
Tcl_GlobTypeData *types (in)
فقط الملفات أو الدلائل التي تتطابق مع أوصاف النوع الموجودة في هذه البنية سيتم إرجاعها. قد تكون هذه المعاملة NULL.
Tcl_Interp *interp (in)

Tcl_Obj *secondPtr (in)

Tcl_Obj *listObj (in)
قائمة عناصر المسار المراد تشغيلها مع عملية دمج.

Tcl_Obj **errorPtr (out)
في حالة حدوث خطأ، سيتم ملؤها بقيمة تحتوي على اسم الملف الذي تسبب في حدوث خطأ في عمليات النسخ/إعادة التسمية المختلفة.
int index (in)

Tcl_Obj *objPtr (in)

Tcl_Obj **objPtrRef (out)

int mode (in)

const char *modeString (in)
تحدد كيفية الوصول إلى الملف. قد يكون له أي من القيم المسموح بها لمعامل mode لأمر Tcl open.
int permissions (in)
مُعاملات الأذونات بأسلوب POSIX مثل 0644. إذا تم إنشاء ملف جديد، فسيتم تعيين هذه الأذونات على الملف المنشأ.
بالوقت، وقت آخر تعديل، ووقت آخر تغيير للبيانات الوصفية. راجع PORTABLE STAT RESULT API لوصف كيفية كتابة كود محمول لتخصيص والوصول إلى بنية Tcl_StatBuf. إذا كان path موجودًا، ترجع Tcl_FSLstat 0 ويتم ملء بنية stat بالبيانات. وإلا، يتم إرجاع -1، ولا يتم إعطاء أي معلومات stat. Tcl_FSUtime تستبدل الإصدار المكتبي للمكتبة من utime. هذه ترجع 0 عند النجاح و -1 عند الخطأ (حسب توثيق utime). إذا نجحت، ستقوم الدالة بتحديث قيم "atime" و "mtime" للملف المحدد. Tcl_FSFileAttrsGet تنفذ الوصول للقراءة للأمر الفرعي file attributes القابل للربط. سيتم استدعاء الدالة المناسبة لنظام الملفات الذي ينتمي إليه pathPtr. إذا كانت النتيجة TCL_OK، فسيتم وضع قيمة في objPtrRef، والتي ستكون صالحة مؤقتًا فقط (ما لم يتم استدعاء Tcl_IncrRefCount). Tcl_FSFileAttrsSet تنفذ الوصول للكتابة للأمر الفرعي file attributes القابل للربط. سيتم استدعاء الدالة المناسبة لنظام الملفات الذي ينتمي إليه pathPtr. Tcl_FSFileAttrStrings تنفذ جزءًا من الأمر الفرعي file attributes القابل للربط. سيتم استدعاء الدالة المناسبة لنظام الملفات الذي ينتمي إليه pathPtr. قد تقوم الإجراءية المستدعاة بإرجاع مصفوفة من السلاسل النصية، أو قد ترجع NULL بدلاً من ذلك وتضع قائمة Tcl في objPtrRef المعطاة. ستقوم Tcl بأخذ هذه القائمة وزيادة عداد المراجع الخاص بها أولاً قبل استخدامها. عند اكتمال الاستخدام، ستقوم Tcl بتقليل عداد المراجع الخاص بها. لذلك، إذا كان ينبغي التخلص من القائمة عند الانتهاء، فيجب أن يكون عداد المراجع الخاص بها صفرًا، وإذا كان ينبغي عدم التخلص من القائمة، فيجب على نظام الملفات التأكد من الاحتفاظ بعداد مراجع للقيمة. Tcl_FSAccess تتحقق مما إذا كانت العملية ستسمح بقراءة أو كتابة أو اختبار وجود الملف (أو كائن نظام ملفات آخر) الذي اسمه pathname. إذا كان pathname رابطًا رمزيًا على Unix، فسيتم اختبار أذونات الملف الذي يشير إليه هذا الرابط الرمزي. عند النجاح (تم منح جميع الأذونات المطلوبة)، يتم إرجاع صفر. عند الخطأ (تم رفض إذن واحد على الأقل في الوضع المطلوب، أو حدث خطأ آخر)، يتم إرجاع -1. Tcl_FSStat تملأ بنية Tcl_StatBuf statPtr بمعلومات حول الملف المحدد. لا تحتاج إلى أي حقوق وصول للملف للحصول على هذه المعلومات ولكنك تحتاج إلى حقوق بحث لجميع الدلائل المذكورة في المسار المؤدي إلى الملف. تتضمن بنية Tcl_StatBuf معلومات تتعلق بالجهاز، inode (دائمًا 0 على Windows)، وضع الامتياز، nlink (دائمًا 1 على Windows)، معرف المستخدم (دائمًا 0 على Windows)، معرف المجموعة (دائمًا 0 على Windows)، rdev (نفس الجهاز على Windows)، الحجم، وقت آخر وصول، وقت آخر تعديل، ووقت آخر تغيير للبيانات الوصفية. راجع PORTABLE STAT RESULT API لوصف كيفية كتابة كود محمول لتخصيص والوصول إلى بنية Tcl_StatBuf. إذا كان path موجودًا، ترجع Tcl_FSStat 0 ويتم ملء بنية stat بالبيانات. وإلا، يتم إرجاع -1، ولا يتم إعطاء أي معلومات stat. Tcl_FSOpenFileChannel تفتح ملفًا محددًا بواسطة pathPtr وترجع مقبض قناة يمكن استخدامه لإجراء الإدخال والإخراج على الملف. هذه الواجهة البرمجية مصممة على غرار إجراء fopen لمكتبة الإدخال والإخراج القياسية في Unix. تركيب ومعنى جميع الوسائط مشابهة لتلك المعطاة في أمر Tcl open عند فتح ملف. إذا حدث خطأ أثناء فتح القناة، ترجع Tcl_FSOpenFileChannel NULL وتسجل رمز خطأ POSIX يمكن استرجاعه باستخدام Tcl_GetErrno. بالإضافة إلى ذلك، إذا كان interp غير NULL، تترك Tcl_FSOpenFileChannel رسالة خطأ في نتيجة interp بعد أي خطأ. لا يتم تسجيل القناة المنشأة حديثًا في المفسر المزود؛ لتسجيلها، استخدم Tcl_RegisterChannel. إذا تم إغلاق إحدى القنوات القياسية، stdin، stdout، أو stderr سابقًا، فإن فعل إنشاء القناة الجديدة يعينها أيضًا كبديل للقناة القياسية. Tcl_FSGetCwd تستبدل الإصدار المكتبي للمكتبة من getcwd. هذه ترجع الدليل العامل الحالي لمكتبة Tcl. قد يختلف هذا عن الدليل العامل الحالي للمنصة الأصلية، وهو ما يحدث عندما لا يكون الدليل العامل الحالي في نظام الملفات الأصلي. النتيجة هي مؤشر إلى Tcl_Obj يحدد الدليل الحالي، أو NULL إذا تعذر تحديد الدليل الحالي. إذا تم إرجاع NULL، يتم ترك رسالة خطأ في نتيجة interp. تم زيادة عداد المراجع للقيمة الناتجة بالفعل للمستدعي. عندما لم تعد هناك حاجة إليها، يجب تقليل عداد المراجع هذا. هذا ضروري لأغراض أمان الخيط، للسماح لخيوط متعددة بالوصول إلى هذه الدالة والدوال ذات الصلة، مع ضمان أن النتائج صالحة دائمًا. Tcl_FSChdir تستبدل الإصدار المكتبي للمكتبة من chdir. يتم تطبيع المسار ثم تمريره إلى نظام الملفات الذي يدعيه. إذا كان نظام الملفات هذا لا ينفذ هذه الدالة، فستعود Tcl إلى مزيج من stat و access للتحقق مما إذا كان الدليل موجودًا ولديه الأذونات المناسبة. بالنسبة للنتائج، راجع توثيق chdir. إذا نجحت، نحتفظ بسجل للمسار الناجح في cwdPathPtr للمكالمات اللاحقة إلى Tcl_FSGetCwd. Tcl_FSPathSeparator ترجع حرف الفاصل الذي يجب استخدامه للعنصر الأكثر تحديدًا للمسار المحدد بواسطة pathPtr (أي الجزء الأخير من المسار). يتم إرجاع الفاصل كـ Tcl_Obj يحتوي على سلسلة نصية بطول 1. إذا كان المسار غير صالح، يتم إرجاع NULL. Tcl_FSJoinPath تأخذ Tcl_Obj المعطاة، والتي يجب أن تكون قائمة صالحة (يسمح لها بوجود عداد مراجع صفر)، وترجع قيمة المسار المعطاة عن طريق اعتبار العناصر elements الأولى كأجزاء مسار صالحة (قد يكون كل جزء مسار مسارًا كاملاً، أو مسارًا جزئيًا، أو مجرد اسم دليل أو ملف ممكن واحد). إذا كان أي جزء مسار هو في الواقع مسار مطلق، فسيتم تجاهل جميع الأجزاء السابقة. إذا كان elements أقل من 0، نستخدم القائمة بأكملها. من الممكن أن تكون القيمة المرجعة في الواقع عنصرًا من القائمة المعطاة، لذلك يجب أن يكون المستدعي حذرًا لزيادة عداد المراجع للنتيجة قبل تحرير القائمة. القيمة المرجعة، عادةً مع عداد مراجع صفر (ولكن يمكن مشاركتها في ظل بعض الظروف)، تحتوي على المسار المدمج. يجب على المستدعي إضافة عداد مراجع للقيمة قبل استخدامها. على وجه الخصوص، قد تكون القيمة المرجعة عنصرًا من القائمة المعطاة، لذا فإن تحرير القائمة قد يحرر القيمة مبكرًا إذا لم يتم أخذ عداد مراجع. إذا كان عدد العناصر صفرًا، فستكون القيمة المرجعة Tcl_Obj بسلسلة فارغة. Tcl_FSSplitPath تأخذ Tcl_Obj المعطاة، والتي يجب أن تكون مسارًا صالحًا، وترجع قيمة قائمة Tcl تحتوي على كل جزء من هذا المسار كعنصر. ترجع قيمة قائمة بعداد مراجع صفر. إذا كان lenPtr المعطى غير NULL، فسيتم تحديث المتغير الذي يشير إليه ليحتوي على عدد العناصر في القائمة المرجعة. Tcl_FSEqualPaths تختبر ما إذا كان المساران المعطيان يمثلان نفس كائن نظام الملفات. ترجع 1 إذا كانت المسارات متساوية، و 0 إذا كانت مختلفة. إذا كان أي مسار NULL، يتم دائمًا إرجاع 0. Tcl_FSGetNormalizedPath تحاول استخراج تمثيل مسار طبيعي فريد من Tcl_Obj المعطى، والتي يمكن استخدام قيمتها النصية كمعرف فريد للملف. ترجع قيمة المسار الطبيعي، مملوكة لـ Tcl، أو NULL إذا كان المسار غير صالح أو تعذر تحويله بنجاح. استخراج المسارات المطلقة والطبيعية فعال للغاية (لأن نظام الملفات يعمل على هذه التمثيلات داخليًا)، على الرغم من أن النتيجة عندما يحتوي نظام الملفات على العديد من الروابط الرمزية قد لا يكون الإصدار الأكثر سهولة في الاستخدام للمسار. القيمة المرجعة مملوكة لـ Tcl ولها عمر يعادل عمر pathPtr المعطى (ما لم يكن هذا مسارًا نسبيًا، وفي هذه الحالة يمكن تحرير قيمة المسار الطبيعي في أي وقت يتغير فيه cwd) - يمكن للمستدعي بالطبع زيادة عداد المراجع إذا رغب في الاحتفاظ بنسخة لفترة أطول. Tcl_FSJoinToPath تأخذ القيمة المعطاة، والتي عادة ما تكون مسارًا صالحًا أو NULL، وتدمج عليها مصفوفة أجزاء المسار المعطاة. ترجع قيمة، عادةً مع عداد مراجع صفر (ولكن يمكن مشاركتها في ظل بعض الظروف)، تحتوي على المسار المدمج. يجب على المستدعي إضافة عداد مراجع للقيمة قبل استخدامها. إذا كانت أي من القيم المعطاة لهذه الدالة (pathPtr أو عناصر path) لها عداد مراجع صفر، فسيتم تحريرها عند إرجاع هذه الدالة. Tcl_FSConvertToPathType تحاول تحويل Tcl_Obj المعطاة إلى نوع مسار Tcl صالح، مع مراعاة حقيقة أن cwd قد تغير حتى لو كان هذا القيمة يفترض بالفعل أنها من النوع الصحيح. قد يبدأ اسم الملف بـ "~" (للدلالة على الدليل الرئيسي للمستخدم الحالي) أو "~" (للدلالة على الدليل الرئيسي لأي مستخدم). إذا نجح التحويل (أي أن القيمة مسار صالح في أحد أنظمة الملفات الحالية)، يتم إرجاع TCL_OK. وإلا يتم إرجاع TCL_ERROR، وقد يتم ترك رسالة خطأ في المفسر. Tcl_FSGetInternalRep تستخرج التمثيل الداخلي لقيمة المسار المعطاة، في نظام الملفات المعطى. إذا كانت قيمة المسار تنتمي إلى نظام ملفات مختلف، نرجع NULL. إذا كان التمثيل الداخلي حاليًا NULL، نحاول إنشاؤه، عن طريق استدعاء Tcl_FSCreateInternalRepProc لنظام الملفات. ترجع NULL أو تمثيل مسار داخلي صالح. يتم تخزين هذا التمثيل الداخلي مؤقتًا، بحيث لا تتطلب المكالمات المتكررة لهذه الدالة تحويلات إضافية. Tcl_FSGetTranslatedPath تحاول استخراج المسار المترجم من Tcl_Obj المعطى. إذا نجح الترجمة (أي أن القيمة مسار صالح)، يتم إرجاعها. وإلا سيتم إرجاع NULL، وقد يتم ترك رسالة خطأ في المفسر. المسار "المترجم" هو مسار لا يحتوي على تسلسلات "~" أو "~user" (تم توسيعها إلى تمثيلها الحالي في نظام الملفات). القيمة المرجعة مملوكة للمستدعي، والذي يجب عليه تخزينها أو استدعاء Tcl_DecrRefCount لضمان تحرير الذاكرة. هذه الدالة ذات فائدة عملية قليلة، وعادة ما تكون Tcl_FSGetNormalizedPath أو Tcl_FSGetNativePath دوال أفضل للاستخدام لمعظم الأغراض. واجهة برمجة تطبيقات نظام الملفات الافتراضية

يقدم نظام الملفات بنية Tcl_Filesystem تحتوي على مؤشرات لوظائف تنفذ العمليات المختلفة على نظام ملفات؛ يتم استدعاء هذه العمليات حسب الحاجة بواسطة الطبقة العامة، والتي تحدث بشكل عام من خلال الوظائف المدرجة أعلاه.

يتم معالجة هياكل Tcl_Filesystem باستخدام الأساليب التالية.

تأخذ Tcl_FSRegister مؤشرًا إلى بنية نظام ملفات وقطعة اختيارية من البيانات لربطها بنظام الملفات هذا. عند استدعاء هذه الوظيفة، ستقوم Tcl بربط نظام الملفات بقائمة أنظمة الملفات المعروفة، وسيصبح وظيفيًا بالكامل على الفور. Tcl لا تتحقق مما إذا كان نفس نظام الملفات مسجلاً عدة مرات (وبشكل عام هذا ليس شيئًا جيدًا للقيام به). سيتم إرجاع TCL_OK.

تزيل Tcl_FSUnregister بنية نظام الملفات المعطاة من قائمة أنظمة الملفات المعروفة، إذا كانت معروفة، وتعيد TCL_OK. إذا لم يكن نظام الملفات مسجلاً حاليًا، يتم إرجاع TCL_ERROR.

ستعيد Tcl_FSData بيانات العميل المرتبطة بنظام الملفات المعطى، إذا كان نظام الملفات هذا مسجلاً. وإلا، ستعيد NULL.

تستخدم Tcl_FSMountsChanged لإعلام نواة Tcl بأن مجموعة نقاط التحميل لنظام الملفات المعطى (المسجل بالفعل) قد تغيرت، وأن تمثيلات الملفات المخزنة مؤقتًا قد لا تكون صحيحة بعد ذلك.

باستثناء الحقول الثلاثة الأولى في هذه البنية التي تحتوي على عناصر بيانات بسيطة، تحتوي جميع الإدخالات على عناوين وظائف تستدعيها طبقة نظام الملفات العامة لأداء النطاق الكامل للإجراءات المتعلقة بنظام الملفات.

تنقسم الوظائف العديدة في هذه البنية إلى ثلاث فئات: وظائف البنية التحتية (يجب تنفيذ معظمها)، ووظائف التشغيل (يجب تنفيذها إذا تم توفير نظام ملفات كامل)، ووظائف الكفاءة (تحتاج فقط إلى التنفيذ إذا كان يمكن القيام بها بكفاءة، أو إذا كانت لها آثار جانبية مطلوبة لنظام الملفات؛ لدى Tcl محاكيات أقل كفاءة يمكنها العودة إليها). من المهم ملاحظة أنه في الإصدار الحالي من Tcl، يتم استخدام معظم هذه التراجعات فقط لمعالجة الأوامر التي تم بدؤها في Tcl، وليس في C. ما يعنيه هذا هو أنه إذا تم إصدار أمر file rename في Tcl، ولم تقم أنظمة الملفات ذات الصلة بتنفيذ Tcl_FSRenameFileProc الخاص بها، فإن نواة Tcl ستعود بدلاً من ذلك إلى مزيج من وظائف نظام الملفات الأخرى (ستستخدم Tcl_FSCopyFileProc متبوعة بـ Tcl_FSDeleteFileProc، وإذا لم يتم تنفيذ Tcl_FSCopyFileProc فهناك تراجع إضافي). ومع ذلك، إذا تم إصدار أمر Tcl_FSRenameFileProc على مستوى C، فلن تحدث أي تراجعات من هذا القبيل. هذا ينطبق باستثناء الإدخالات الأربعة الأخيرة في جدول نظام الملفات (lstat، load، getcwd و chdir) والتي تحدث لها بالفعل تراجعات على مستوى C.

أي وظائف تأخذ أسماء مسارات في شكل Tcl_Obj تأخذ تلك الأسماء في شكل UTF-8. تم تصميم واجهة برمجة تطبيقات البنية التحتية لنظام الملفات لدعم التحويل الفعال والمخزن مؤقتًا لهذه المسارات UTF-8 إلى تمثيلات أصلية أخرى.

ENFILECHANNELPROC

الدالة لمعالجة استدعاء Tcl_FSOpenFileChannel. يجب تنفيذها لأي نظام ملفات معقول، لأن أي عمليات تتطلب فتح أو الوصول إلى محتويات ملف ستستخدمها (على سبيل المثال، open، encoding، والعديد من أوامر Tk).

typedef Tcl_Channel Tcl_FSOpenFileChannelProc(
        Tcl_Interp *interp,
        Tcl_Obj *pathPtr,
        int mode,
        int permissions);

تقوم Tcl_FSOpenFileChannelProc بفتح ملف محدد بواسطة pathPtr وإرجاع مقبض قناة يمكن استخدامه لإجراء الإدخال والإخراج على الملف. هذه الواجهة برمجة التطبيقات مستوحاة من إجراء fopen لمكتبة الإدخال / الإخراج القياسية في Unix. يشبه بناء الجملة ومعنى جميع الوسائط تلك المعطاة في أمر Tcl open عند فتح ملف، حيث يكون الوسيط mode مزيجًا من علامات POSIX O_RDONLY، O_WRONLY، إلخ. إذا حدث خطأ أثناء فتح القناة، ترجع Tcl_FSOpenFileChannelProc قيمة NULL وتسجل رمز خطأ POSIX يمكن استرداده باستخدام Tcl_GetErrno. بالإضافة إلى ذلك، إذا كان interp غير NULL، فإن Tcl_FSOpenFileChannelProc تترك رسالة خطأ في نتيجة interp بعد أي خطأ.

يجب ألا يتم تسجيل القناة التي تم إنشاؤها حديثًا في المفسر المزود بواسطة Tcl_FSOpenFileChannelProc؛ هذه المهمة تقع على عاتق المتصل بـ Tcl_FSOpenFileChannel (إذا لزم الأمر). إذا تم إغلاق إحدى القنوات القياسية، stdin، stdout أو stderr، سابقًا، فإن عملية إنشاء القناة الجديدة تقوم أيضًا بتعيينها كبديل للقناة القياسية.

MATCHINDIRECTORYPROC

الدالة لمعالجة استدعاء Tcl_FSMatchInDirectory. إذا لم يتم تنفيذها، فإن وظائف glob والنسخ المتكرر ستكون مفقودة في نظام الملفات (وقد يؤثر ذلك على الأوامر مثل encoding names التي تستخدم وظائف glob داخليًا).

typedef int Tcl_FSMatchInDirectoryProc(
        Tcl_Interp *interp,
        Tcl_Obj *resultPtr,
        Tcl_Obj *pathPtr,
        const char *pattern,
        Tcl_GlobTypeData *types);

يجب أن ترجع الدالة جميع الملفات أو الدلائل (أو كائنات نظام الملفات الأخرى) التي تتطابق مع النمط المحدد وتتوافق مع مواصفات types المعطاة. هناك طريقتان يمكن استدعاء هذه الدالة بهما. إذا كان pattern هو NULL، فإن pathPtr هو مواصفات مسار كامل لملف أو دليل واحد يجب التحقق من وجوده ونوعه الصحيح. بخلاف ذلك، pathPtr هو دليل، ويجب على الدالة البحث في محتوياته عن ملفات أو دلائل ذات نوع صحيح. في كلتا الحالتين، يمكن افتراض أن pathPtr غير NULL وغير فارغ. لا توثق حاليًا ما إذا كان pathPtr سينتهي بفاصل ملف أم لا، لذا يجب أن يكون الكود مرنًا لكلا الاحتمالين.

القيمة المرجعة هي نتيجة Tcl قياسية تشير إلى ما إذا كان هناك خطأ قد حدث في عملية المطابقة. يتم وضع رسائل الخطأ في interp، إلا إذا كان interp NULL في هذه الحالة لا يلزم إنشاء رسالة خطأ؛ عند نتيجة TCL_OK، يجب إضافة النتائج إلى قيمة resultPtr المعطاة (والتي يمكن افتراض أنها قائمة Tcl صالحة غير مشتركة). يجب أن تتضمن المطابقات المضافة إلى resultPtr أي بادئة مسار معطاة في pathPtr (يعني هذا عادةً أنها ستكون مواصفات مسار مطلق). لاحظ أنه إذا لم يتم العثور على مطابقات، فإن ذلك يؤدي ببساطة إلى نتيجة فارغة؛ يتم الإبلاغ عن الأخطاء فقط للمشاكل الفعلية في الملفات أو نظام الملفات التي قد تحدث أثناء عملية المطابقة.

يحتوي الهيكل Tcl_GlobTypeData الذي تم تمريره في المعامل types على الحقول التالية:

typedef struct Tcl_GlobTypeData {
    /* يتوافق مع bcdpfls كما في 'find -t' */
    int type;
    /* يتوافق مع أذونات الملف */
    int perm;
    /* نوع Mac مقبول */
    Tcl_Obj *macType;
    /* منشئ Mac مقبول */
    Tcl_Obj *macCreator;
} Tcl_GlobTypeData;

هناك حالتان مهمتان يجب التعامل معهما بشكل صحيح، سواء عندما يكون types غير NULL. الحالتان هما عندما يكون types->types & TCL_GLOB_TYPE_DIR أو types->types & TCL_GLOB_TYPE_MOUNT صحيحين (وخاصة عندما تكون العلامات الأخرى خاطئة). في الحالة الأولى، يجب على الدالة سرد الدلائل المحتوية. تستخدم Tcl هذا لتنفيذ glob متكرر، لذلك من الضروري أن تقوم أنظمة الملفات بتنفيذ مطابقة الدلائل بشكل صحيح. في الحالة الثانية، مع TCL_GLOB_TYPE_MOUNT، يجب على نظام الملفات سرد نقاط التحميل التي تقع ضمن pathPtr المعطى (وفي هذه الحالة، لا يلزم أن يقع pathPtr ضمن نفس نظام الملفات - بخلاف جميع الحالات الأخرى التي يتم فيها استدعاء هذه الدالة). دعم هذا أمر بالغ الأهمية إذا أرادت Tcl أن تتمتع بانتقالات سلسة بين نظام ملفات وآخر.

UTIMEPROC

الدالة لمعالجة استدعاء Tcl_FSUtime. مطلوبة للسماح بتعيين (وليس قراءة) الأوقات باستخدام file mtime، file atime، وتنفيذ open-r/open-w/fcopy لـ file copy.

typedef int Tcl_FSUtimeProc(
        Tcl_Obj *pathPtr,
        struct utimbuf *tval);

يجب تغيير أوقات الوصول والتعديل للملف المحدد بواسطة pathPtr إلى القيم المعطاة في الهيكل tval.

يجب أن تكون القيمة المرجعة 0 عند النجاح و -1 عند حدوث خطأ، كما هو الحال مع utime في النظام.

LINKPROC

الدالة لمعالجة استدعاء Tcl_FSLink. يجب تنفيذها فقط إذا كان نظام الملفات يدعم الروابط، وقد تكون NULL بخلاف ذلك.

typedef Tcl_Obj *Tcl_FSLinkProc(
        Tcl_Obj *linkNamePtr,
        Tcl_Obj *toPtr,
        int linkAction);

إذا كان toPtr هو NULL، يُطلب من الدالة قراءة محتويات رابط. النتيجة هي Tcl_Obj تحدد محتويات الرابط المعطى بواسطة linkNamePtr، أو NULL إذا لم يتمكن الرابط من القراءة. النتيجة مملوكة للمتصل (وينبغي بالتالي زيادة عداد مرجعها قبل إرجاعها). يجب على أي متصلين استدعاء Tcl_DecrRefCount على هذه النتيجة عندما لم تعد مطلوبة. إذا كان toPtr غير NULL، فيجب على الدالة محاولة إنشاء رابط. يجب أن تكون النتيجة في هذه الحالة هي toPtr إذا كان الرابط ناجحًا و NULL بخلاف ذلك. في هذه الحالة، النتيجة ليست مملوكة للمتصل (أي، لا حاجة لأي معالجة لعدد المراجع من أي طرف). انظر الوثائق الخاصة بـ Tcl_FSLink لتفسير صحيح لعلامات linkAction.

LISTVOLUMESPROC

الدالة لسرد أي وحدات نظام الملفات تمت إضافتها بواسطة نظام الملفات هذا. يجب تنفيذها فقط إذا كان نظام الملفات يضيف وحدات في بداية نظام الملفات، حتى يمكن إرجاعها بواسطة file volumes.

typedef Tcl_Obj *Tcl_FSListVolumesProc(void);

يجب أن تكون النتيجة قائمة بوحدات نظام الملفات التي تمت إضافتها بواسطة نظام الملفات هذا، أو NULL (أو قائمة فارغة) إذا لم يتم توفير أي وحدات. تعتبر قيمة النتيجة مملوكة لنظام الملفات (وليس لنواة Tcl)، ولكن يجب إعطاؤها عداد مرجع لـ Tcl. ستستخدم Tcl محتويات القائمة ثم تقلل هذا العداد. هذا يسمح لأنظمة الملفات بالاختيار ما إذا كانت تريد بالفعل الاحتفاظ بـ "قائمة عالمية" للوحدات أم لا (إذا لم يكن الأمر كذلك، فإنها تنشئ القائمة أثناء التنقل وتمررها إلى Tcl مع عداد مرجع قدره 1 ثم تنسى القائمة، وإذا كان الأمر كذلك، فإنها ببساطة تزيد عداد مرجع قائمتها العالمية وتمررها إلى Tcl التي ستنسخ المحتويات ثم تقلل العداد إلى حيث كان).

لذلك، تعتبر Tcl قيم الإرجاع من هذه الدالة للقراءة فقط.

FILEATTRSTRINGSPROC

الدالة لسرد جميع سلاسل السمات الصالحة لنظام الملفات هذا. إذا لم يتم تنفيذها، فلن يدعم نظام الملفات أمر file attributes. يتيح هذا إرفاق معلومات إضافية اعتباطية بالملفات في نظام الملفات. إذا لم يتم تنفيذه، فلا حاجة لتنفيذ أساليب get و set.

typedef const char *const *Tcl_FSFileAttrStringsProc(
        Tcl_Obj *pathPtr,
        Tcl_Obj **objPtrRef);

قد تقوم الدالة المستدعاة بإرجاع مصفوفة من السلاسل، أو قد ترجع بدلاً من ذلك NULL وتضع قائمة Tcl في objPtrRef المعطى. ستأخذ Tcl هذه القائمة وتزيد عداد مرجعها أولاً قبل استخدامها. عند اكتمال هذا الاستخدام، ستقوم Tcl بتقليل عداد مرجعها. لذا، إذا كان يجب على Tcl التخلص من القائمة عند الانتهاء، فيجب أن يكون عداد مرجعها صفرًا، وإذا كان يجب عدم التخلص من القائمة، فيجب على نظام الملفات التأكد من أنه يرجع قيمة بعداد مرجع لا يقل عن واحد.

FILEATTRSGETPROC

الدالة لمعالجة استدعاء Tcl_FSFileAttrsGet، المستخدم بواسطة file attributes.

typedef int Tcl_FSFileAttrsGetProc(
        Tcl_Interp *interp,
        int index,
        Tcl_Obj *pathPtr,
        Tcl_Obj **objPtrRef);

ترجع رمز نتيجة Tcl قياسي. قيمة السمة المستردة، والتي تتوافق مع العنصر ذي الفهرس index في القائمة التي تم إرجاعها بواسطة Tcl_FSFileAttrStringsProc، هي Tcl_Obj توضع في objPtrRef (إذا تم إرجاع TCL_OK) ومن المحتمل أن يكون عداد مرجعها صفرًا. في كلتا الحالتين، يجب علينا إما تخزينها في مكان ما (على سبيل المثال، نتيجة Tcl)، أو زيادة/تقليل عداد مرجعها لضمان تحريرها بشكل صحيح.

FILEATTRSSETPROC

الدالة لمعالجة استدعاء Tcl_FSFileAttrsSet، المستخدم بواسطة file attributes. إذا كان نظام الملفات للقراءة فقط، فلا حاجة لتنفيذ ذلك.

typedef int Tcl_FSFileAttrsSetProc(
        Tcl_Interp *interp,
        int index,
        Tcl_Obj *pathPtr,
        Tcl_Obj *objPtr);

يجب تعيين قيمة السمة للعنصر ذي الفهرس index في القائمة التي تم إرجاعها بواسطة Tcl_FSFileAttrStringsProc إلى objPtr المعطى.
