Tcl_NREvalObjv(3)

الحزمة: tcl

التراخيص: TCL

NRE(3) إجراءات مكتبة Tcl NRE(3)

الاسم

Tcl_NRCreateCommand, Tcl_NRCallObjProc, Tcl_NREvalObj, Tcl_NREvalObjv, Tcl_NRCmdSwap, Tcl_NRExprObj, Tcl_NRAddCallback - تقييم غير متكرر (بدون تعبئة) لسكريبتات Tcl.

الملخص

#include <tcl.h>

Tcl_Command
Tcl_NRCreateCommand(interp, cmdName, proc, nreProc, clientData,
                    deleteProc)

int
Tcl_NRCallObjProc(interp, nreProc, clientData, objc, objv)

int
Tcl_NREvalObj(interp, objPtr, flags)

int
Tcl_NREvalObjv(interp, objc, objv, flags)

int
Tcl_NRCmdSwap(interp, cmd, objc, objv, flags)

int
Tcl_NRExprObj(interp, objPtr, resultPtr)

void
Tcl_NRAddCallback(interp, postProcPtr, data0, data1, data2, data3)

الوسائط

Tcl_Interp *interp (in)
المترجم ذو الصلة.
const char *cmdName (in)
اسم الأمر المراد إنشاؤه.
Tcl_ObjCmdProc *proc (in)
يتم استدعاؤه لتقييم أمر. غالبًا ما يكون غلافًا صغيرًا يستخدم Tcl_NRCallObjProc لاستدعاء nreProc ببرنامج طائر جديد. يتصرف بشكل مشابه لـ proc الخاص بـ Tcl_CreateObjCommand(3) (انظر).
Tcl_ObjCmdProc *nreProc (in)
يتم استدعاؤه بدلاً من proc عندما يكون هناك برنامج طائر قيد الاستخدام بالفعل.
ClientData clientData (in)
قيمة اعتباطية بحجم كلمة واحدة يتم تمريرها إلى proc و nreProc و deleteProc و objProc.
Tcl_CmdDeleteProc *deleteProc (in/out)
يتم استدعاؤه قبل إزالة cmdName من المترجم، مما يسمح بالتنظيف الخاص بالأمر. يمكن أن يكون NULL.
int objc (in)
عدد العناصر في objv.
Tcl_Obj **objv (in)
الكلمات في الأمر.
Tcl_Obj *objPtr (in)
سكريبت أو تعبير لتقييمه.
int flags (in)
كما هو موضح لـ Tcl_EvalObjv.
Tcl_Command cmd (in)
رمز مميز لاستخدامه بدلاً من الرمز المشتق من الكلمة الأولى في objv لتقييم أمر.
Tcl_Obj *resultPtr (out)
كائن Tcl غير مشترك يتم فيه تخزين نتيجة التقييم إذا كان رمز الإرجاع هو TCL_OK.
Tcl_NRPostProc *postProcPtr (in)
دالة سيتم دفعها.
ClientData data0 (in)
ClientData data1 (in)
ClientData data2 (in)
ClientData data3 (in)
data0 إلى data3 هي أربع قيم بحجم كلمة واحدة سيتم تمريرها إلى الدالة المعينة بواسطة postProcPtr عند استدعائها.

الوصف

توفر هذه الوظائف واجهة إلى مكدس الوظائف الذي يتجول فيه المترجم لتقييم الأوامر. يتم تنفيذ الدالة وراء الأمر بواسطة دالة أولية وأي وظائف إضافية يضعها الروتين على المكدس أثناء تقدمه. يضع المترجم نفسه وظائف على المكدس للاستجابة لنهاية الروتين وممارسة أشكال التحكم مثل التبديل بين مكدسات الأوامر قيد التقدم أو تقييم السكريبتات على مستويات إضافية دون إضافة إطارات في مكدس C. لتنفيذ روتين، يتم استدعاء الدالة الأولية للروتين أولاً، ثم يتم استدعاء كتلة صغيرة من التعليمات البرمجية تسمى برنامج طائر تتخذ الوظائف من المكدس بشكل تكراري وتستدعيها، باستخدام قيمة الاستدعاء الأخير كقيمة للروتين.

Tcl_NRCallObjProc يستدعي nreProc ببرنامج طائر جديد.

Tcl_NRCreateCommand، بديل لـ Tcl_CreateObjCommand، يحل cmdName، والذي يمكن أن يحتوي على محدِّدات مسافة، فيما يتعلق بالمساحة الحالية، وينشئ أمرًا بهذا الاسم ويعيد رمزًا مميزًا للأمر يمكن استخدامه في الاستدعاءات اللاحقة لـ Tcl_GetCommandName. باستثناء بعض الحالات المذكورة أدناه، يتم أولاً إزالة أي أمر موجود بنفس الاسم. إذا كان interp قيد الإزالة، فإن Tcl_NRCreateCommand لا ينشئ أي أمر، ولا يزيل أي أمر، ويعيد NULL.

Tcl_NREvalObj يدفع دالة مشابهة لـ Tcl_EvalObjEx، ولكنه يستهلك مساحة في مكدس C.

Tcl_NREvalObjv يدفع دالة مشابهة لـ Tcl_EvalObjv، ولكنه يستهلك مساحة في مكدس C.

Tcl_NRCmdSwap مشابه لـ Tcl_NREvalObjv، ولكنه يستخدم cmd، وهو رمز مميز تم إرجاعه مسبقًا بواسطة Tcl_CreateObjCommand أو Tcl_GetCommandFromObj، بدلاً من حل الكلمة الأولى في objv.

Tcl_NRExprObj يدفع دالة تقيّم objPtr كتعبير بنفس الطريقة التي تعمل بها Tcl_ExprObj، ولكن دون استهلاك مساحة في مكدس C.

ترجع جميع الوظائف TCL_OK إذا تمت جدولة تقييم السكريبت أو الأمر أو التعبير بنجاح. بخلاف ذلك (على سبيل المثال، إذا تعذر حل اسم الأمر)، فإنها ترجع TCL_ERROR وتخزن رسالة كنتيجة للمترجم.

Tcl_NRAddCallback يدفع postProcPtr. توقيع Tcl_NRPostProc هو:

typedef int
Tcl_NRPostProc(
        ClientData data[],
        Tcl_Interp *interp,
        int result);

data هو مؤشر إلى مصفوفة تحتوي على data0 إلى data3. result هي القيمة المرجعة من الدالة السابقة التي تنفذ جزءًا من الروتين.

مثال

يستخدم الأمر التالي Tcl_EvalObjEx، الذي يستهلك مساحة في مكدس C، لتقييم سكريبت:

int
TheCmdOldObjProc(
    ClientData clientData,
    Tcl_Interp *interp,
    int objc,
    Tcl_Obj *const objv[])
{
    int result;
    Tcl_Obj *objPtr;
    ... إعداد ...
    result = Tcl_EvalObjEx(interp, objPtr, 0);
    ... معالجة لاحقة ...
    return result;
}
Tcl_CreateObjCommand(interp, "theCommand",
        TheCmdOldObjProc, clientData, TheCmdDeleteProc);

لتجنب استهلاك مساحة في مكدس C، يتم إعادة تسمية TheCmdOldObjProc إلى TheCmdNRObjProc ويتم تقسيم خطوة المعالجة اللاحقة إلى دالة منفصلة، TheCmdPostProc، التي يتم دفعها إلى مكدس الوظائف. يتم استبدال Tcl_EvalObjEx بـ Tcl_NREvalObj، الذي يستخدم برنامجًا طائرًا بدلاً من استهلاك مساحة في مكدس C. الإصدار الجديد من TheCmdOldObjProc هو مجرد غلاف يستخدم Tcl_NRCallObjProc لاستدعاء TheCmdNRObjProc:

int
TheCmdOldObjProc(
    ClientData clientData,
    Tcl_Interp *interp,
    int objc,
    Tcl_Obj *const objv[])
{
    return Tcl_NRCallObjProc(interp, TheCmdNRObjProc,
            clientData, objc, objv);
}
int
TheCmdNRObjProc
    ClientData clientData,
    Tcl_Interp *interp,
    int objc,
    Tcl_Obj *const objv[])
{
    Tcl_Obj *objPtr;
    ... إعداد ...
    Tcl_NRAddCallback(interp, TheCmdPostProc,
            data0, data1, data2, data3);
    /* data0 .. data3 هي عناصر تصل إلى أربعة عناصر بحجم كلمة واحدة سيتم تمريرها إلى دالة المعالجة اللاحقة */
    return Tcl_NREvalObj(interp, objPtr, 0);
}
int
TheCmdNRPostProc(
    ClientData data[],
    Tcl_Interp *interp,
    int result)
{
    /* data[0] .. data[3] هي البيانات الأربعة من مكدس البيانات
     * التي تم تمريرها إلى Tcl_NRAddCallback */
    ... معالجة لاحقة ...
    return result;
}

يمكن لأي دالة تشكل روتيناً دفع وظائف أخرى، مما يسمح بتنفيذ هياكل التحكم مثل الحلقات والتسلسلات باستخدام مكدس الوظائف.
