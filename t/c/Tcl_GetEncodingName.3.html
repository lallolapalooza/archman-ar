المعاملات

Tcl_Interp *interp (in)
المفسر المراد استخدامه لتقارير الأخطاء، أو NULL إذا لم تكن هناك حاجة لتقارير الأخطاء.
const char *name (in)
اسم الترميز المراد تحميله.
Tcl_Encoding encoding (in)
الترميز المراد الاستعلام عنه، تحريره، أو استخدامه لتحويل النص. إذا كان encoding هو NULL، يتم استخدام ترميز النظام الحالي.
Tcl_Obj *objPtr (in)
اسم الترميز المراد الحصول على رمزه.
Tcl_Encoding *encodingPtr (out)
يشير إلى المساحة التي سيتم فيها كتابة رمز الترميز.
const char *src (in)
بالنسبة لدوال Tcl_ExternalToUtf، مصفوفة بايتات بالترميز المحدد المراد تحويلها إلى UTF-8. بالنسبة لدوال Tcl_UtfToExternal و Tcl_WinUtfToTChar، مصفوفة أحرف UTF-8 المراد تحويلها إلى الترميز المحدد.
const TCHAR *tsrc (in)
مصفوفة من أحرف Windows TCHAR لتحويلها إلى UTF-8.
int srcLen (in)
طول src أو tsrc بالبايت. إذا كان الطول سالبًا، يتم استخدام الطول المحدد بالترميز للسلسلة.
Tcl_DString *dstPtr (out)
مؤشر إلى Tcl_DString غير مهيأ أو محرر سيتم تخزين النتيجة المحولة فيه.
int flags (in)
بتات علامات مختلفة مدمجة معًا (OR). TCL_ENCODING_START يعني أن المخزن المؤقت للمصدر هو الكتلة الأولى في تيار إدخال (ربما متعدد الكتل)، مما يخبر روتين التحويل بإعادة التعيين إلى حالة أولية وتنفيذ أي تهيئة تحتاج إلى الحدوث قبل تحويل البايت الأول. TCL_ENCODING_END يعني أن المخزن المؤقت للمصدر هو الكتلة الأخيرة في تيار إدخال (ربما متعدد الكتل)، مما يخبر روتين التحويل بتنفيذ أي إنهاءات تحتاج إلى الحدوث بعد تحويل البايت الأخير ثم إعادة التعيين إلى حالة أولية. TCL_ENCODING_STOPONERROR يعني أن روتين التحويل يجب أن يعود فورًا عند قراءة حرف مصدر غير موجود في الترميز الهدف؛ وإلا سيتم تلقائيًا استبدال حرف احتياطي افتراضي.
Tcl_EncodingState *statePtr (in/out)
يستخدم عند تحويل تيار بايتات (عادةً طويل أو ذي طول غير محدد) قطعة بقطعة. يقوم روتين التحويل بتخزين حالته الحالية في *statePtr بعد تحويل src (المخزن المؤقت الذي يحتوي على الكتلة الحالية)؛ يجب إعادة تمرير معلومات الحالة تلك عند تحويل القطعة التالية من التيار حتى يعرف روتين التحويل في أي حالة تركها في نهاية القطعة السابقة. قد يكون NULL، وفي هذه الحالة يتم تجاهل القيمة المحددة لـ flags ويُفترض أن المخزن المؤقت للمصدر يحتوي على السلسلة الكاملة للتحويل.
char *dst (out)
المخزن المؤقت الذي سيتم تخزين النتيجة المحولة فيه. لن يتم تخزين أكثر من dstLen بايت في dst.
int dstLen (in)
الحد الأقصى لطول المخزن المؤقت للإخراج dst بالبايت.
int *srcReadPtr (out)
تم ملؤه بعدد البايتات من src التي تم تحويلها فعليًا. قد يكون هذا أقل من طول المصدر الأصلي إذا كانت هناك مشكلة في تحويل بعض أحرف المصدر. قد يكون NULL.
int *dstWrotePtr (out)
تم ملؤه بعدد البايتات التي تم تخزينها فعليًا في المخزن المؤقت للإخراج كنتيجة للتحويل. قد يكون NULL.
int *dstCharsPtr (out)
تم ملؤه بعدد الأحرف التي تتوافق مع عدد البايتات المخزنة في المخزن المؤقت للإخراج. قد يكون NULL.
Tcl_DString *bufPtr (out)
مساحة لتخزين اسم ترميز النظام المحدد.
const Tcl_EncodingType *typePtr (in)
هيكل يحدد نوع ترميز جديد.
Tcl_Obj *searchPath (in)
قائمة بمجلدات نظام الملفات التي سيتم البحث فيها عن ملفات بيانات الترميز.
const char *path (in)
مسار إلى موقع ملف الترميز.

مقدمة

تقوم هذه الإجراءات بالتحويل بين تمثيل الأحرف الداخلي لـ Tcl، وهو UTF-8، وتمثيلات الأحرف التي تستخدمها أنظمة التشغيل أو أنظمة الملفات المختلفة، مثل Unicode أو ASCII أو Shift-JIS. عند العمل على السلاسل النصية، مثل الحصول على أسماء الملفات أو عرض الأحرف باستخدام خطوط دولية، يجب ترجمة السلاسل النصية إلى تنسيق واحد أو ربما تنسيقات متعددة يمكن لاستدعاءات النظام المختلفة استخدامها. على سبيل المثال، على محطة عمل Unix يابانية، قد يحصل المستخدم على اسم ملف ممثل في ترميز الملف EUC-JP ثم يترجم الأحرف إلى ترميز الخط jisx0208 لعرض اسم الملف في عنصر واجهة Tk. الغرض من حزمة الترميز هو المساعدة في سد فجوة الترجمة. يوفر UTF-8 أرضية وسيطة لجميع الترميزات المختلفة. في المثال أعلاه، سيتم ترجمة النص إلى UTF-8 من أي ترميز ملف يستخدمه نظام التشغيل. ثم سيتم ترجمته من UTF-8 إلى أي ترميز خط تتطلبه روتينات العرض.

بعض الترميزات الأساسية مدمجة في Tcl. يمكن تعريف ترميزات أخرى من قبل المستخدم أو تحميلها ديناميكيًا من ملفات الترميز بطريقة مستقلة عن المنصة.

الوصف

Tcl_GetEncoding يبحث عن ترميز بالنظر إلى name. قد يشير الاسم إلى ترميز Tcl مدمج، أو ترميز معرف من قبل المستخدم تم تسجيله عن طريق استدعاء Tcl_CreateEncoding، أو ملف ترميز قابل للتحميل ديناميكيًا. القيمة المرتجعة هي رمز يمثل الترميز ويمكن استخدامه في استدعاءات لاحقة لإجراءات مثل Tcl_GetEncodingName و Tcl_FreeEncoding و Tcl_UtfToExternal. إذا لم يشير الاسم إلى أي ترميز معروف أو قابل للتحميل، يتم إرجاع NULL ويتم إرجاع رسالة خطأ في interp.

تحتفظ حزمة الترميز بقاعدة بيانات لجميع الترميزات قيد الاستخدام حاليًا. في المرة الأولى التي تتم فيها رؤية name، تُرجع Tcl_GetEncoding ترميزًا مع عدد مرجعي قدره 1. إذا تم طلب نفس name مرات أخرى، فسيتم زيادة عدد المراجع لهذا الترميز دون تكلفة تخصيص ترميز جديد وجميع هياكله البيانات المرتبطة به.

عندما لا يعود encoding مطلوبًا، يجب استدعاء Tcl_FreeEncoding لتحريره. عندما لا يعود encoding قيد الاستخدام في أي مكان (أي، تم تحريره بنفس عدد المرات التي تم الحصول عليه فيها)، ستقوم Tcl_FreeEncoding بتحرير جميع المساحات التي كان يستخدمها الترميز وحذفه من قاعدة البيانات.

Tcl_GetEncodingFromObj يعامل التمثيل النصي لـ objPtr كاسم ترميز، ويجد ترميزًا بهذا الاسم، تمامًا كما تفعل Tcl_GetEncoding. عند العثور على ترميز، يتم تخزينه مؤقتًا داخل قيمة objPtr للرجوع المستقبلي، ويتم كتابة رمز Tcl_Encoding إلى مساحة التخزين

مُعَرَّف بواسطة encodingPtr، ويتم إرجاع القيمة TCL_OK. إذا لم يتم العثور على ترميز كهذا، يتم إرجاع القيمة TCL_ERROR، ولا يتم إجراء أي كتابة إلى *encodingPtr. تمامًا مثل Tcl_GetEncoding، يجب على المتصل استدعاء Tcl_FreeEncoding على رمز الترميز الناتج عندما لم يعد هذا الرمز قيد الاستخدام.

تحول Tcl_ExternalToUtfDString المخزن المؤقت للمصدر src من الترميز المحدد encoding إلى UTF-8. يتم تخزين البايتات المحولة في dstPtr، والتي يتم إنهاؤها بعد ذلك بحرف NULL. يجب على المتصل استدعاء Tcl_DStringFree في النهاية لتحرير أي معلومات مخزنة في dstPtr. عند التحويل، إذا لم يكن من الممكن تمثيل أي من الأحرف في المخزن المؤقت للمصدر في الترميز الهدف، فسيتم استخدام حرف احتياطي افتراضي. قيمة الإرجاع هي مؤشر إلى القيمة المخزنة في DString.

تحول Tcl_ExternalToUtf المخزن المؤقت للمصدر src من الترميز المحدد encoding إلى UTF-8. يتم تحويل ما يصل إلى srcLen بايت من المخزن المؤقت للمصدر ويتم تخزين ما يصل إلى dstLen بايت محول في dst. في جميع الحالات، يتم ملء *srcReadPtr بعدد البايتات التي تم تحويلها بنجاح من src ويتم ملء *dstWrotePtr بالعدد المقابل للبايتات التي تم تخزينها في dst. قيمة الإرجاع هي واحدة مما يلي:

TCL_OK
تم تحويل جميع بايتات src.
TCL_CONVERT_NOSPACE
لم يكن المخزن المؤقت الوجهة كبيرًا بما يكفي لجميع البيانات المحولة؛ تم تحويل أكبر عدد ممكن من الأحرف.
TCL_CONVERT_MULTIBYTE
كانت البايتات القليلة الأخيرة في المخزن المؤقت للمصدر بداية تسلسل متعدد البايتات، ولكن كانت هناك حاجة لمزيد من البايتات لإكمال هذا التسلسل. يجب على استدعاء لاحق لروتين التحويل تمرير مخزن مؤقت يحتوي على البايتات غير المحولة المتبقية في src بالإضافة إلى بعض البايتات الإضافية من دفق المصدر لتحويل تسلسل متعدد البايتات المقسم سابقًا بشكل صحيح.
TCL_CONVERT_SYNTAX
احتوى المخزن المؤقت للمصدر على تسلسل أحرف غير صالح. قد يحدث هذا إذا تضررت دفق الإدخال أو إذا تم تحديد طريقة ترميز الإدخال بشكل غير صحيح.
TCL_CONVERT_UNKNOWN
احتوى المخزن المؤقت للمصدر على حرف لم يكن من الممكن تمثيله في الترميز الهدف وتم تحديد TCL_ENCODING_STOPONERROR.

تحول Tcl_UtfToExternalDString المخزن المؤقت للمصدر src من UTF-8 إلى الترميز المحدد encoding. يتم تخزين البايتات المحولة في dstPtr، والتي يتم إنهاؤها بعد ذلك بـ NULL الخاص بالترميز المحدد. يجب على المتصل استدعاء Tcl_DStringFree في النهاية لتحرير أي معلومات مخزنة في dstPtr. عند التحويل، إذا لم يكن من الممكن تمثيل أي من الأحرف في المخزن المؤقت للمصدر في الترميز الهدف، فسيتم استخدام حرف احتياطي افتراضي. قيمة الإرجاع هي مؤشر إلى القيمة المخزنة في DString.

تحول Tcl_UtfToExternal المخزن المؤقت للمصدر src من UTF-8 إلى الترميز المحدد encoding. يتم تحويل ما يصل إلى srcLen بايت من المخزن المؤقت للمصدر ويتم تخزين ما يصل إلى dstLen بايت محول في dst. في جميع الحالات، يتم ملء *srcReadPtr بعدد البايتات التي تم تحويلها بنجاح من src ويتم ملء *dstWrotePtr بالعدد المقابل للبايتات التي تم تخزينها في dst. قيم الإرجاع هي نفسها قيم الإرجاع لـ Tcl_ExternalToUtf.

Tcl_WinUtfToTChar و Tcl_WinTCharToUtf هي دوال مساعدة خاصة بنظام Windows فقط للتحويل بين UTF-8 وسلاسل Windows بناءً على نوع TCHAR الذي هو بالاتفاقية حرف Unicode على نظام Windows NT.

Tcl_GetEncodingName هو عكس Tcl_GetEncoding تقريبًا. بالنظر إلى encoding، فإن قيمة الإرجاع هي الوسيط name الذي تم استخدامه لإنشاء الترميز. يضمن أن السلسلة التي ترجعها Tcl_GetEncodingName تستمر فقط حتى يتم حذف encoding. يجب على المتصل عدم تعديل هذه السلسلة.

تقوم Tcl_SetSystemEncoding بتعيين الترميز الافتراضي الذي يجب استخدامه كلما مرر المستخدم قيمة NULL للوسيط encoding لأي من دوال الترميز الأخرى. إذا كان name NULL، تتم إعادة تعيين ترميز النظام إلى ترميز النظام الافتراضي، binary. إذا لم يشير الاسم إلى أي ترميز معروف أو قابل للتحميل، يتم إرجاع TCL_ERROR ويتم ترك رسالة خطأ في interp. بخلاف ذلك، تقوم هذه الإجراءات بزيادة عدد المراجع للترميز النظامي الجديد، وتقليل عدد المراجع للترميز النظامي القديم، وإرجاع TCL_OK.

توفر Tcl_GetEncodingNameFromEnvironment وسيلة لمكتبة Tcl للإبلاغ عن اسم الترميز الذي تعتقد أنه الصحيح للاستخدام كترميز نظامي، بناءً على استدعاءات النظام وفحص البيئة المناسبة للمنصة. تقبل bufPtr، وهو مؤشر إلى Tcl_DString غير مهيأ أو محرّر، وتكتب اسم الترميز إليه. يتم إرجاع Tcl_DStringValue.

تقوم Tcl_GetEncodingNames بتعيين نتيجة interp إلى قائمة تتكون من أسماء جميع الترميزات المعرفة حاليًا أو التي يمكن تحميلها ديناميكيًا، مع البحث في مسار الترميز المحدد بواسطة Tcl_SetDefaultEncodingDir. لا يضمن هذا الإجراء أن تحتوي ملفات الترميز القابلة للتحميل ديناميكيًا على بيانات صالحة، بل تضمن وجودها فقط.

تعرف Tcl_CreateEncoding ترميزًا جديدًا وتسجل إجراءات C التي يتم استدعاؤها للتحويل بين الترميز و UTF-8. الترميزات التي تم إنشاؤها بواسطة Tcl_CreateEncoding تكون مرئية بعد ذلك في قاعدة البيانات المستخدمة بواسطة Tcl_GetEncoding. تمامًا مثل إجراء Tcl_GetEncoding، فإن قيمة الإرجاع هي رمز يمثل الترميز ويمكن استخدامه في الاستدعاءات اللاحقة لدوال الترميز الأخرى. ترجع Tcl_CreateEncoding ترميزًا بعدد مراجع 1. إذا كان الترميز بالاسم المحدد name موجودًا بالفعل، فسيتم استبدال إدخاله في قاعدة البيانات بالترميز الجديد؛ سيظل الرمز الخاص بالترميز القديم صالحًا ويستمر في العمل كما كان من قبل، ولكن مستخدمي الرمز الجديد سيستدعون الآن إجراءات الترميز الجديدة.

يحتوي الوسيط typePtr إلى Tcl_CreateEncoding على معلومات حول اسم الترميز والإجراءات التي سيتم استدعاؤها للتحويل بين هذا الترميز و UTF-8. يتم تعريفه على النحو التالي:

typedef struct Tcl_EncodingType {
    const char *encodingName;
    Tcl_EncodingConvertProc *toUtfProc;
    Tcl_EncodingConvertProc *fromUtfProc;
    Tcl_EncodingFreeProc *freeProc;
    ClientData clientData;
    int nullSize;
} Tcl_EncodingType;

يوفر encodingName اسم سلسلة للترميز، والذي يمكن الرجوع إليه في إجراءات أخرى مثل Tcl_GetEncoding. يشير toUtfProc إلى إجراء استدعاء لتنفيذ التحويل من هذا الترميز إلى UTF-8. يشير fromUtfProc إلى إجراء استدعاء لتنفيذ التحويل من UTF-8 إلى هذا الترميز. يشير freeProc إلى إجراء استدعاء لتنفيذه عند حذف الترميز. قد يكون حقل freeProc NULL. يحتوي clientData على قيمة كلمة واحدة اعتباطية يتم تمريرها إلى toUtfProc و fromUtfProc و freeProc كلما تم استدعاؤها. عادةً، يكون هذا مؤشرًا إلى هيكل بيانات يحتوي على معلومات خاصة بالترميز يمكن استخدامها بواسطة إجراءات الاستدعاء. على سبيل المثال، قد يستخدم ترميزان متشابهان جدًا مثل ascii و macRoman نفس إجراء الاستدعاء، ولكنهما يستخدمان قيمًا مختلفة لـ clientData للتحكم في سلوكه. يحدد nullSize عدد بايتات الصفر التي تمثل نهاية السلسلة في هذا الترميز. يجب أن يكون 1 (لترميزات أحادية البايت أو متعددة البايتات مثل ASCII أو Shift-JIS) أو 2 (لترميزات مزدوجة البايت مثل Unicode). لا يتم قبول الترميزات ذات الحجم الثابت بثلاثة بايتات أو أكثر لكل حرف (مثل CNS11643).

يجب أن تتطابق إجراءات الاستدعاء toUtfProc و fromUtfProc مع نوع Tcl_EncodingConvertProc:

typedef int Tcl_EncodingConvertProc(
        ClientData clientData,
        const char *src,
        int srcLen,
        int flags,
        Tcl_EncodingState *statePtr,
        char *dst,
        int dstLen,
        int *srcReadPtr,
        int *dstWrotePtr,
        int *dstCharsPtr);

يتم استدعاء إجراءات toUtfProc و fromUtfProc بواسطة عائلة الدوال Tcl_ExternalToUtf أو Tcl_UtfToExternal لتنفيذ التحويل الفعلي. معلمة clientData لهذه الإجراءات هي نفسها حقل clientData المحدد لـ Tcl_CreateEncoding عند إنشاء الترميز. الوسائط المتبقية لإجراءات الاستدعاء هي نفس الوسائط، الموثقة في الأعلى، لـ Tcl_ExternalToUtf أو Tcl_UtfToExternal، مع الاستثناءات التالية. إذا كان الوسيط srcLen لأحد هذه الدوال عالية المستوى سالبًا، فإن القيمة الممررة إلى إجراء الاستدعاء ستكون طول السلسلة المحدد بالترميز لـ src. إذا كان أي من الوسائط srcReadPtr أو dstWrotePtr أو dstCharsPtr لأحد الدوال عالية المستوى NULL، فإن القيمة المقابلة الممررة إلى إجراء الاستدعاء ستكون موقعًا غير NULL.

يجب أن يطابق إجراء الاستدعاء freeProc، إذا كان غير NULL، نوع Tcl_EncodingFreeProc:

typedef void Tcl_EncodingFreeProc(
        ClientData clientData);

يتم استدعاء دالة freeProc هذه عند حذف الترميز. معلمة clientData هي نفسها حقل clientData المحدد لـ Tcl_CreateEncoding عند إنشاء الترميز.

يتم استدعاء Tcl_GetEncodingSearchPath و Tcl_SetEncodingSearchPath للوصول إلى قائمة الدلائل في نظام الملفات التي يتم البحث فيها عن ملفات بيانات الترميز وتعيينها.

قيمة الإرجاع من Tcl_GetEncodingSearchPath هي القيمة التي تم تخزينها بواسطة آخر استدعاء ناجح لـ Tcl_SetEncodingSearchPath. إذا لم تحدث استدعاءات لـ Tcl_SetEncodingSearchPath، فسوف تحسب Tcl قيمة أولية بناءً على البيئة. يوجد مسار بحث ترميز واحد للعملية بأكملها، مشترك بين جميع الخيوط في العملية.

تقوم Tcl_SetEncodingSearchPath بتخزين searchPath وترجع TCL_OK، ما لم يكن searchPath قائمة Tcl غير صالحة، مما يتسبب في إرجاع TCL_ERROR. عناصر searchPath/i> لم يتم التحقق منها كدليل موجود في نظام الملفات وقابل للقراءة. عند البحث عن ملفات بيانات الترميز، يتم تجاهل أدلة نظام الملفات غير الموجودة أو غير القابلة للقراءة في searchPath بصمت.

Tcl_GetDefaultEncodingDir و Tcl_SetDefaultEncodingDir هي واجهات قديمة يُفضل استبدالها باستدعاءات لـ Tcl_GetEncodingSearchPath و Tcl_SetEncodingSearchPath. يتم استدعاؤها للوصول إلى وتعيين العنصر الأول في قائمة searchPath. نظرًا لأن Tcl يبحث في searchPath عن ملفات بيانات الترميز بترتيب القائمة، فإن هذه الروتينات تحدد الدليل "الافتراضي" الذي يمكن العثور فيه على ملفات بيانات الترميز.

ملفات الترميز

سيحد المساحة من تجميع كل خوارزمية ترميز ممكنة مسبقًا في Tcl، لذلك يتم تخزين العديد من الترميزات على القرص كملفات ترميز قابلة للتحميل ديناميكيًا. هذا السلوك يسمح أيضًا للمستخدم بإنشاء ملفات ترميز إضافية يمكن تحميلها باستخدام نفس الآلية. تحتوي ملفات الترميز هذه على معلومات حول الجداول و/أو تسلسلات الهروب المستخدمة للمعاينة بين ترميز خارجي و Unicode. قد يتكون الترميز الخارجي من أحرف ذات بايت واحد، أو بايتات متعددة، أو بايتات مزدوجة.

يتم تمثيل كل ترميز قابل للتحميل ديناميكيًا كملف نصي. السطر الأول من الملف، الذي يبدأ بالرمز "#"، هو تعليق يوفر وصفًا للملف يمكن قراءته بواسطة الإنسان. يحدد السطر التالي نوع ملف الترميز. يمكن أن يكون أحد الأحرف التالية:

[1] S
ترميز أحادي البايت، حيث يكون الحرف الواحد دائمًا بطول بايت واحد في الترميز. مثال على ذلك هو iso8859-1، المستخدم للعديد من اللغات الأوروبية.
[2] D
ترميز مزدوج البايت، حيث يكون الحرف الواحد دائمًا بطول بايتين في الترميز. مثال على ذلك هو big5، المستخدم للنص الصيني.
[3] M
ترميز متعدد البايتات، حيث قد يكون الحرف الواحد بطول بايت واحد أو بايتين. بعض البايتات هي بايتات رائدة، تشير إلى أن بايتًا آخر يجب أن يتبع وأن البايتات معًا تمثل حرفًا واحدًا. بايتات أخرى ليست بايتات رائدة وتمثل نفسها. مثال على ذلك هو shiftjis، المستخدم من قبل العديد من أجهزة الكمبيوتر اليابانية.
[4] E
ملف ترميز تسلسلات الهروب، يحدد أن تسلسلات معينة من البايتات لا تمثل أحرفًا، بل أوامر تصف كيفية تفسير البايتات التالية.

السطر الثالث من الملف هو ثلاثة أرقام. الرقم الأول هو حرف السقوط (بالنظام السداسي عشري) لاستخدامه عند التحويل من UTF-8 إلى هذا الترميز. الرقم الثاني هو 1 إذا كان هذا الملف يمثل ترميزًا لخط رموز، أو 0 بخلاف ذلك. الرقم الأخير (بالنظام العشري) هو عدد الصفحات التالية.

الأسطر اللاحقة في المثال أعلاه هي صفحات تصف كيفية المعاينة من الترميز إلى Unicode ذي البايتين. يحدد السطر الأول في الصفحة رقم الصفحة. يتبعه 256 رقمًا ثنائي البايت، مرتبة كـ 16 صفًا من 16 رقمًا. بالنظر إلى حرف في الترميز، يتم استخدام البايت الأعلى لهذا الحرف لتحديد الصفحة، ويتم استخدام البايت الأدنى لهذا الحرف كفهرس لاختيار أحد الأرقام ثنائية البايت في تلك الصفحة - القيمة التي تم الحصول عليها هي حرف Unicode المقابل. من خلال فحص المثال أعلاه، يمكن للمرء أن يرى أن الأحرف 0x7E و 0x8163 في shiftjis تتوافق مع 203E و 2026 في Unicode، على التوالي.

بعد الصفحة الأولى ستأتي جميع الصفحات الأخرى، كل منها بنفس تنسيق الصفحة الأولى: رقم واحد يحدد الصفحة متبوعًا بـ 256 حرف Unicode ثنائي البايت. إذا كان حرف في الترميز يتوافق مع حرف Unicode 0000، فهذا يعني أن الحرف غير موجود فعليًا. إذا كانت جميع الأحرف في صفحة تتوافق مع 0000، يمكن حذف تلك الصفحة.

الحالة [4] هي ملف ترميز تسلسلات الهروب. الأسطر في هذا النوع من الملفات لها نفس تنسيق هذا المثال المأخوذ من ترميز iso2022-jp:

# Encoding file: iso2022-jp, escape-driven
E
init		{}
final		{}
iso8859-1	\x1b(B
jis0201		\x1b(J
jis0208		\x1b$@
jis0208		\x1b$B
jis0212		\x1b$(D
gb2312		\x1b$A
ksc5601		\x1b$(C

في الملف، يمثل العمود الأول خيارًا والعمود الثاني هو القيمة المرتبطة به. init هو سلسلة لإصدارها أو توقعها قبل تحويل الحرف الأول، بينما final هو سلسلة لإصدارها أو توقعها بعد الحرف الأخير. جميع الخيارات الأخرى هي أسماء ترميزات قائمة على الجداول؛ القيمة المرتبطة بها هي تسلسل الهروب الذي يحدد ذلك الترميز. يتم استخدام بناء جملة Tcl للقيم؛ في المثال أعلاه، على سبيل المثال، يمثل "{}" السلسلة الفارغة ويمثل "\x1b" الحرف 27.

عندما تواجه Tcl_GetEncoding اسم ترميز لم يتم تحميله، فإنها تحاول تحميل ملف ترميز يسمى name.enc من الدليل الفرعي encoding لكل دليل يبحث فيه Tcl عن مكتبة البرامج النصية الخاصة به. إذا كان ملف الترميز موجودًا، ولكنه معطوب، فسيتم ترك رسالة خطأ في interp.
