الوسائط

const Tcl_ChannelType *typePtr (in)
بنية تحتوي على نقاط الدخول للإجراءات التي يمكن استدعاؤها لأداء عمليات الإدخال/الإخراج ووظائف أخرى على القناة.
const char *channelName (in)
اسم القناة، على سبيل المثال، file3؛ لا يمكن أن يكون قيد الاستخدام من قبل أي قناة أخرى. يمكن أن يكون NULL، في هذه الحالة يتم إنشاء القناة بدون اسم. إذا تم تعيين القناة التي تم إنشاؤها إلى إحدى القنوات القياسية (stdin، stdout، أو stderr)، فسيكون الاسم المعين هو اسم القناة القياسية.
ClientData instanceData (in)
قيمة عشوائية ذات كلمة واحدة مرتبطة بهذه القناة. يتم تمرير هذه القيمة إلى الإجراءات في typePtr عند استدعائها.
int mask (in)
مزيج OR من TCL_READABLE و TCL_WRITABLE يشير إلى ما إذا كانت القناة قابلة للقراءة والكتابة.
Tcl_Channel channel (in)
القناة المراد العمل عليها.
int direction (in)
TCL_READABLE يعني أنك تريد مقبض الإدخال؛ TCL_WRITABLE يعني أنك تريد مقبض الإخراج.
ClientData *handlePtr (out)
المكان الذي يجب تخزين مقبض نظام التشغيل المطلوب فيه.
int size (in)
الحجم، بالبايت، للمخازن المؤقتة المراد تخصيصها في هذه القناة.
int mask (in)
مزيج OR من TCL_READABLE، TCL_WRITABLE، و TCL_EXCEPTION يشير إلى الأحداث التي وقعت في هذه القناة.
Tcl_Interp *interp (in)
المفسر الحالي. (يمكن أن يكون NULL)
const char *optionName (in)
اسم الخيار غير الصالح.
const char *optionList (in)
قائمة بالخيارات المحددة (كلمات مفصولة بمسافات، بدون -) لإضافتها إلى قائمة الخيارات العامة القياسية. يمكن أن يكون NULL لرسائل خطأ الخيارات العامة.

الوصف

يستخدم Tcl بنية قنوات ذات طبقتين. يوفر مستوى أعلى عامًا حتى تتمكن برامج C و Tcl من إجراء الإدخال والإخراج باستخدام نفس واجهات برمجة التطبيقات (APIs) لمجموعة متنوعة من الملفات والأجهزة والمقابس وما إلى ذلك. يتم وصف واجهات برمجة التطبيقات C العامة في صفحة الدليل الخاصة بـ Tcl_OpenFileChannel.

توفر الطبقة السفلية برامج تشغيل قنوات خاصة بكل نوع جهاز مدعوم على كل منصة. تصف صفحة الدليل هذه واجهات برمجة التطبيقات C التي تتواصل بين الطبقة العامة وبرامج تشغيل القنوات الخاصة. كما تشرح كيفية إضافة أنواع قنوات جديدة عن طريق توفير برامج تشغيل قنوات جديدة.

تتكون برامج تشغيل القنوات من عدة مكونات: أولاً، يوفر كل برنامج تشغيل قناة بنية Tcl_ChannelType تحتوي على مؤشرات إلى الإجراءات التي تنفذ العمليات المتنوعة التي تستخدمها الطبقة العامة للتواصل مع برنامج تشغيل القناة. يتم وصف بنية Tcl_ChannelType والإجراءات التي تشير إليها في قسم TCL_CHANNELTYPE أدناه.

ثانيًا، توفر برامج تشغيل القنوات عادةً أمر Tcl لإنشاء مثيلات لهذا النوع من القنوات. على سبيل المثال، ينشئ أمر Tcl open قنوات تستخدم برامج تشغيل قنوات الملفات والأوامر، وينشئ أمر Tcl socket قنوات تستخدم بروتوكولات شبكة TCP للاتصال بالشبكة.

ثالثًا، يوفر برنامج تشغيل القناة اختياريًا دالة C لفتح مثيلات هذا النوع من القنوات. على سبيل المثال، يفتح Tcl_OpenFileChannel قناة تستخدم برنامج تشغيل قناة الملفات، ويفتح Tcl_OpenTcpClient قناة تستخدم بروتوكول شبكة TCP. عادةً ما تستدعي دوال الإنشاء هذه Tcl_CreateChannel داخليًا لفتح القناة.

لإضافة نوع قناة جديد، يجب عليك تنفيذ واجهة برمجة تطبيقات C أو أمر Tcl يفتح قناة عن طريق استدعاء Tcl_CreateChannel. عندما يستدعي برنامج التشغيل الخاص بك Tcl_CreateChannel، فإنه يمرر بنية Tcl_ChannelType التي تصف وظائف الإدخال/الإخراج لبرنامج التشغيل الخاص بك. ستقوم الطبقة العامة بعد ذلك باستدعاء الإجراءات المشار إليها في هذه البنية لإجراء عمليات على القناة.

Tcl_CreateChannel يفتح قناة جديدة ويربط typePtr و instanceData بها. يتم فتح القناة في الوضع المحدد بواسطة mask. لمناقشة برامج تشغيل القنوات وعملياتها وبنية Tcl_ChannelType، راجع قسم TCL_CHANNELTYPE أدناه.

Tcl_CreateChannel يتفاعل مع الكود الذي يدير القنوات القياسية. بمجرد تهيئة قناة قياسية من خلال استدعاء Tcl_GetStdChannel أو استدعاء Tcl_SetStdChannel، سيؤدي إغلاق هذه القناة إلى جعل الاستدعاء التالي لـ Tcl_CreateChannel القناة القياسية الجديدة. راجع Tcl_StandardChannels لمعالجة عامة للقنوات القياسية وسلوك مكتبة Tcl فيما يتعلق بها.

Tcl_GetChannelInstanceData يعيد بيانات المثيل المرتبطة بالقناة في channel. هذا هو نفس الوسيط instanceData في استدعاء Tcl_CreateChannel الذي أنشأ هذه القناة.

Tcl_GetChannelType يعيد مؤشرًا إلى بنية Tcl_ChannelType المستخدمة بواسطة القناة في الوسيط channel. هذا هو نفس الوسيط typePtr في استدعاء Tcl_CreateChannel الذي أنشأ هذه القناة.

Tcl_GetChannelName يعيد سلسلة نصية تحتوي على الاسم المرتبط بالقناة، أو NULL إذا كان الوسيط channelName لـ Tcl_CreateChannel هو NULL.

Tcl_GetChannelHandle يضع مقبض نظام التشغيل المرتبط بـ channel للاتجاه direction المحدد في handlePtr.

أمثلة

يوضح المثال التالي كيف يتم استخدام Tcl_CreateChannel:

int
Tcl_CreateChannel(
    (const Tcl_ChannelType *) "fileChannelType",
    "file3",
    0,
    TCL_READABLE | TCL_WRITABLE
);

TCL_CHANNELTYPE

يحتوي حقل Tcl_ChannelType على معلومات حول نوع قناة Tcl. يتم استخدامه عند إنشاء قناة جديدة.

typedef struct Tcl_ChannelType {
    const char *typeName;
    Tcl_DriverCloseProc *closeProc;
    Tcl_DriverInputProc *inputProc;
    Tcl_DriverOutputProc *outputProc;
    Tcl_DriverSeekProc *seekProc;
    Tcl_DriverSetOptionProc *setOptionProc;
    Tcl_DriverGetOptionProc *getOptionProc;
    Tcl_DriverWatchProc *watchProc;
    Tcl_DriverGetHandleProc *getHandleProc;
    Tcl_DriverClose2Proc *close2Proc;
    Tcl_DriverBlockModeProc *blockModeProc;
    Tcl_DriverFlushProc *flushProc;
    Tcl_DriverHandlerProc *handlerProc;
    Tcl_DriverThreadActionProc *threadActionProc;
    Tcl_DriverTruncateProc *truncateProc;
} Tcl_ChannelType;

The fields are:

typeName
اسم هذا النوع من القنوات. يتم استخدامه في رسائل الخطأ.
closeProc
يشير إلى إجراء يتم استدعاؤه عند إغلاق القناة.
inputProc
يشير إلى إجراء يتم استدعاؤه لقراءة البيانات من القناة.
outputProc
يشير إلى إجراء يتم استدعاؤه لكتابة البيانات إلى القناة.
seekProc
يشير إلى إجراء يتم استدعاؤه لتغيير موضع القراءة/الكتابة في القناة.
setOptionProc
يشير إلى إجراء يتم استدعاؤه لتعيين خيارات القناة.
getOptionProc
يشير إلى إجراء يتم استدعاؤه للحصول على خيارات القناة.
watchProc
يشير إلى إجراء يتم استدعاؤه لمراقبة أحداث القناة (قابل للقراءة، قابل للكتابة، استثناء).
getHandleProc
يشير إلى إجراء يتم استدعاؤه للحصول على المقبض الأساسي للقناة.
close2Proc
إجراء إغلاق بديل يسمح بإعادة استخدام المقبض.
blockModeProc
يشير إلى إجراء يتم استدعاؤه لتغيير وضع الحظر للقناة.
flushProc
يشير إلى إجراء يتم استدعاؤه لتفريغ المخزن المؤقت للقناة.
handlerProc
يشير إلى إجراء يتم استدعاؤه عند وقوع حدث في القناة.
threadActionProc
يشير إلى إجراء يتم استدعاؤه عند تغيير خيوط القناة.
truncateProc
يشير إلى إجراء يتم استدعاؤه لقطع طول القناة.

يمكن استرداد هذا الهيكل باستخدام Tcl_GetChannelType، الذي يعيد مؤشرًا إلى الهيكل.

PROCEDURES

TCL_CREATECHANNEL

ينشئ قناة Tcl جديدة. يجب أن يكون channelTypePtr هو المؤشر إلى بنية Tcl_ChannelType المناسبة للقناة التي يتم إنشاؤها.

TCL_GETCHANNELTYPE

يعيد مؤشراً إلى بنية Tcl_ChannelType للقناة المحددة.

TCL_REGISTERSHANNELTYPE

يسجل نوع قناة جديد. يجب أن يتم استدعاء هذا قبل استخدام نوع القناة.

TCL_UNREGISTERSHANNELTYPE

يلغي تسجيل نوع قناة. يجب عدم استدعاء هذا إذا كانت هناك قنوات لا تزال تستخدم نوع القناة هذا.

TCL_BADCHANNELOPTION

يولد رسالة خطأ "خيار سيء" في مترجم (اختياري). يستخدم من قبل برامج تشغيل القنوات عندما يتم طلب خيار تعيين/الحصول غير صالح.

CHANNEL DRIVER PROCEDURES

البروتوكولات التالية هي جزء من واجهة برنامج تشغيل القناة. يتم استدعاؤها بواسطة طبقة Tcl العامة.

CLOSEPROC

يجب على closeProc تحرير أي موارد مرتبطة بالقناة. إذا كانت القناة لا تزال تحتوي على بيانات للكتابة، فيجب على closeProc محاولة كتابتها قبل التحرير. إذا حدث خطأ في استدعاء النظام، فيجب على الوظيفة ترك رسالة خطأ في نتيجة interp إذا لم يكن interp فارغاً. يجب على الوظيفة أيضًا استدعاء Tcl_SetErrno لتخزين رمز خطأ POSIX مناسب.

typedef int Tcl_DriverCloseProc(
        ClientData instanceData,
        Tcl_Interp *interp);

يمكن استرداد هذه القيمة باستخدام Tcl_ChannelCloseProc، التي تعيد مؤشراً إلى الوظيفة.

INPUTPROC

يجب على inputProc محاولة قراءة أكبر عدد ممكن من البايتات من القناة، ووضعها في المخزن المؤقت المحدد بواسطة bufPtr. يجب ألا تقرأ الوظيفة أكثر من bufSize بايت. يجب أن تعيد الوظيفة عدد البايتات المقروءة، أو -1 إذا حدث خطأ. إذا حدث خطأ في استدعاء النظام، فيجب على الوظيفة ترك رسالة خطأ في نتيجة interp إذا لم يكن interp فارغاً. يجب على الوظيفة أيضًا استدعاء Tcl_SetErrno لتخزين رمز خطأ POSIX مناسب.

typedef int Tcl_DriverInputProc(
        ClientData instanceData,
        char *bufPtr,
        int bufSize,
        Tcl_Interp *interp);

يمكن استرداد هذه القيمة باستخدام Tcl_ChannelInputProc، التي تعيد مؤشراً إلى الوظيفة.

OUTPUTPROC

يجب على outputProc محاولة كتابة جميع البايتات من المخزن المؤقت المحدد بواسطة bufPtr إلى القناة. يجب أن تعيد الوظيفة عدد البايتات المكتوبة، أو -1 إذا حدث خطأ. إذا حدث خطأ في استدعاء النظام، فيجب على الوظيفة ترك رسالة خطأ في نتيجة interp إذا لم يكن interp فارغاً. يجب على الوظيفة أيضًا استدعاء Tcl_SetErrno لتخزين رمز خطأ POSIX مناسب.

typedef int Tcl_DriverOutputProc(
        ClientData instanceData,
        const char *bufPtr,
        int bufSize,
        Tcl_Interp *interp);

يمكن استرداد هذه القيمة باستخدام Tcl_ChannelOutputProc، التي تعيد مؤشراً إلى الوظيفة.

SEEKPROC

يجب على seekProc تغيير موضع القراءة/الكتابة للقناة. يجب أن تعيد الوظيفة موضع القراءة/الكتابة الجديد، أو -1 إذا حدث خطأ. إذا حدث خطأ في استدعاء النظام، فيجب على الوظيفة ترك رسالة خطأ في نتيجة interp إذا لم يكن interp فارغاً. يجب على الوظيفة أيضًا استدعاء Tcl_SetErrno لتخزين رمز خطأ POSIX مناسب.

typedef Tcl_WideInt Tcl_DriverSeekProc(
        ClientData instanceData,
        Tcl_WideInt offset,
        int seekMode,
        Tcl_Interp *interp);

يمكن استرداد هذه القيمة باستخدام Tcl_ChannelSeekProc، التي تعيد مؤشراً إلى الوظيفة.

SETOPTIONPROC

يجب على setOptionProc تعيين خيار للقناة. يجب أن تعيد الوظيفة 0 عند النجاح، أو رمز خطأ POSIX (مناسب للاستخدام مع Tcl_SetErrno) عند الفشل. إذا فشلت الوظيفة بسبب استدعاء نظام، فيجب عليها أيضًا ترك رسالة خطأ في نتيجة interp إذا لم يكن interp فارغاً.

typedef int Tcl_DriverSetOptionProc(
        ClientData instanceData,
        Tcl_Interp *interp,
        const char *optionName,
        const Tcl_Obj *valuePtr);

يمكن استرداد هذه القيمة باستخدام Tcl_ChannelSetOptionProc، التي تعيد مؤشراً إلى الوظيفة.

GETOPTIONPROC

يجب على getOptionProc الحصول على قيمة خيار للقناة. يجب أن تعيد الوظيفة 0 عند النجاح، أو رمز خطأ POSIX (مناسب للاستخدام مع Tcl_SetErrno) عند الفشل. إذا فشلت الوظيفة بسبب استدعاء نظام، فيجب عليها أيضًا ترك رسالة خطأ في نتيجة interp إذا لم يكن interp فارغاً.

typedef int Tcl_DriverGetOptionProc(
        ClientData instanceData,
        Tcl_Interp *interp,
        const char *optionName,
        Tcl_Obj *valuePtr);

يمكن استرداد هذه القيمة باستخدام Tcl_ChannelGetOptionProc، التي تعيد مؤشراً إلى الوظيفة.

بعض الخيارات يتم التعامل معها بواسطة الكود العام ولا يتم استدعاء هذه الوظيفة أبداً لاسترداد قيمتها، على سبيل المثال -blockmode. الخيارات الأخرى خاصة بكل نوع قناة وسيتم استدعاء إجراء getOptionProc الخاص ببرنامج تشغيل القناة لتنفيذها. يمكن أن يكون حقل getOptionProc فارغاً، مما يشير إلى أن نوع القناة هذا لا يدعم أي خيارات خاصة بالنوع.

WATCHPROC

يحتوي حقل watchProc على عنوان وظيفة يتم استدعاؤها بواسطة الطبقة العامة لتهيئة آلية إشعار الأحداث لملاحظة الأحداث ذات الأهمية على هذه القناة. يجب أن يتطابق WatchProc مع النموذج الأولي التالي:

typedef void Tcl_DriverWatchProc(
        ClientData instanceData,
        int mask);

instanceData هو نفس القيمة التي تم تمريرها إلى Tcl_CreateChannel عند إنشاء هذه القناة. الوسيط mask هو مزيج ORed من TCL_READABLE و TCL_WRITABLE و TCL_EXCEPTION؛ إنه يشير إلى الأحداث التي يهتم المتصل بملاحظتها على هذه القناة.

يجب على الوظيفة تهيئة آليات محددة لنوع الجهاز لملاحظة الأحداث ذات الأهمية على القناة. عندما تحدث واحدة أو أكثر من الأحداث المحددة على القناة، يكون برنامج تشغيل القناة مسؤولاً عن استدعاء Tcl_NotifyChannel لإعلام وحدة القناة العامة. يجب على برنامج التشغيل توخي الحذر لعدم تجويع برامج تشغيل القنوات الأخرى أو مصادر الاستدعاءات عن طريق استدعاء Tcl_NotifyChannel بشكل متكرر جداً. يمكن ضمان العدالة باستخدام قائمة انتظار أحداث Tcl للسماح بجدولة حدث القناة بالتتابع مع الأحداث الأخرى. انظر وصف Tcl_QueueEvent للحصول على تفاصيل حول كيفية وضع حدث في قائمة الانتظار.

يمكن استرداد هذه القيمة باستخدام Tcl_ChannelWatchProc، التي تعيد مؤشراً إلى الوظيفة.

GETHANDLEPROC

يحتوي حقل getHandleProc على عنوان وظيفة يتم استدعاؤها بواسطة الطبقة العامة لاسترداد مقبض خاص بالجهاز من القناة. يجب أن يتطابق GetHandleProc مع النموذج الأولي التالي:

typedef int Tcl_DriverGetHandleProc(
        ClientData instanceData,
        int direction,
        ClientData *handlePtr);

InstanceData هو نفس القيمة التي تم تمريرها إلى Tcl_CreateChannel عند إنشاء هذه القناة. الوسيط direction هو إما TCL_READABLE لاسترداد المقبض المستخدم للإدخال، أو TCL_WRITABLE لاسترداد المقبض المستخدم للإخراج.

إذا كان تنفيذ القناة يحتوي على مقابض خاصة بالجهاز، فيجب على الوظيفة استرداد المقبض المناسب المرتبط بالقناة، وفقًا للوسيط direction. يجب تخزين المقبض في الموقع المشار إليه بواسطة handlePtr، ويجب إعادة TCL_OK. إذا لم تكن القناة مفتوحة للاتجاه المحدد، أو إذا لم يستخدم تنفيذ القناة مقابض الأجهزة، فيجب على الوظيفة إعادة TCL_ERROR.

يمكن استرداد هذه القيمة باستخدام Tcl_ChannelGetHandleProc، التي تعيد مؤشراً إلى الوظيفة.

FLUSHPROC

حقل flushProc محجوز حاليًا للاستخدام المستقبلي. يجب تعيينه إلى NULL. يجب أن يتطابق FlushProc مع النموذج الأولي التالي:

typedef int Tcl_DriverFlushProc(
        ClientData instanceData);

يمكن استرداد هذه القيمة باستخدام Tcl_ChannelFlushProc، التي تعيد مؤشراً إلى الوظيفة.

HANDLERPROC

يحتوي حقل handlerProc على عنوان وظيفة يتم استدعاؤها بواسطة الطبقة العامة لإعلام القناة بوقوع حدث. يجب أن تكون معرفة لبرامج تشغيل القنوات المكدسة التي ترغب في تلقي إشعارات بالأحداث التي تحدث على القناة الأساسية (المكدسة). يجب أن يتطابق HandlerProc مع النموذج الأولي التالي:

typedef int Tcl_DriverHandlerProc(
        ClientData instanceData,
        int interestMask);

InstanceData هو نفس القيمة التي تم تمريرها إلى Tcl_CreateChannel عند إنشاء هذه القناة. interestMask هو مزيج ORed من TCL_READABLE أو TCL_WRITABLE؛ إنه يشير إلى نوع الحدث الذي وقع على هذه القناة.

يمكن استرداد هذه القيمة باستخدام Tcl_ChannelHandlerProc، التي تعيد مؤشراً إلى الوظيفة.

THREADACTIONPROC

يحتوي حقل threadActionProc على عنوان الوظيفة التي يتم استدعاؤها بواسطة الطبقة العامة عند إنشاء قناة، أو إغلاقها، أو نقلها إلى خيط آخر، أي في أي وقت قد يلزم تهيئة أو تحديث حالة برنامج تشغيل خاصة بالخيط. يمكن أن يكون NULL. يستخدم الإجراء TCL_CHANNEL_THREAD_REMOVE لإعلام برنامج التشغيل بأنه يجب عليه تحديث أو إزالة أي بيانات خاصة بالخيط قد يحتفظ بها للقناة.

يستخدم الإجراء TCL_CHANNEL_THREAD_INSERT لإعلام برنامج التشغيل بأنه يجب عليه تحديث أو تهيئة أي بيانات خاصة بالخيط قد يحتفظ بها باستخدام الخيط الحالي كمنظم. انظر Tcl_CutChannel و Tcl_SpliceChannel لمزيد من التفاصيل.

typedef void Tcl_DriverThreadActionProc(
        ClientData instanceData,
        int action);

InstanceData هو نفس القيمة التي تم تمريرها إلى Tcl_CreateChannel عند إنشاء هذه القناة.

يمكن استرداد هذه القيم باستخدام Tcl_ChannelThreadActionProc، التي تعيد مؤشراً إلى الوظيفة.

TRUNCATEPROC

يحتوي حقل truncateProc على عنوان الوظيفة التي يتم استدعاؤها بواسطة الطبقة العامة عند قطع طول القناة إلى طول معين. يمكن أن يكون NULL.

typedef int Tcl_DriverTruncateProc(
        ClientData instanceData,
        Tcl_WideInt length);

InstanceData هو نفس القيمة التي تم تمريرها إلى Tcl_CreateChannel عند إنشاء هذه القناة، و length هو الطول الجديد للملف الأساسي، والذي يجب ألا يكون سالباً. يجب أن تكون النتيجة 0 عند النجاح أو رمز errno (مناسب للاستخدام مع Tcl_SetErrno) عند الفشل.

يمكن استرداد هذه القيم باستخدام Tcl_ChannelTruncateProc، التي تعيد مؤشراً إلى الوظيفة.

TCL_BADCHANNELOPTION

تولد هذه الإجراءات رسالة خطأ "خيار سيء" في مترجم (اختياري). يتم استخدامها بواسطة برامج تشغيل القنوات عندما يتم طلب خيار تعيين/الحصول غير صالح. الغرض منها هو ربط قائمة الخيارات العامة بالخيارات المحددة وتجزئة سلسلة رسائل خطأ الخيارات العامة.

تعيد دائمًا TCL_ERROR

يتم إنشاء رسالة خطأ في قيمة نتيجة interp للإشارة إلى أنه تم استدعاء أمر بخيار سيء. يكون تنسيق الرسالة

    bad option "blah": should be one of
    <...generic options...>+<...specific options...>
لذلك تحصل على على سبيل المثال: 
    bad option "-blah": should be one of -blocking,
    -buffering, -buffersize, -eofchar, -translation,
    -peername, or -sockname
عند استدعائها مع optionList مساوية لـ "peername sockname"

"blah" هو الوسيط optionName و "<specific options>" هي قائمة منفصلة بمسافات من الكلمات الخيارات المحددة. تأخذ الوظيفة عناية جيدة بإدراج علامات الطرح قبل كل خيار، وفاصلات بعد، و "or" قبل الخيار الأخير.

OLD CHANNEL TYPES

يحتوي هيكل Tcl_ChannelType الأصلي (8.3.1 وما دون) على الحقول التالية:

typedef struct Tcl_ChannelType {
    const char *typeName;
    Tcl_DriverBlockModeProc *blockModeProc;
    Tcl_DriverCloseProc *closeProc;
    Tcl_DriverInputProc *inputProc;
    Tcl_DriverOutputProc *outputProc;
    Tcl_DriverSeekProc *seekProc;
    Tcl_DriverSetOptionProc *setOptionProc;
    Tcl_DriverGetOptionProc *getOptionProc;
    Tcl_DriverWatchProc *watchProc;
    Tcl_DriverGetHandleProc *getHandleProc;
    Tcl_DriverClose2Proc *close2Proc;
} Tcl_ChannelType;

لا يزال من الممكن إنشاء قنوات باستخدام الهيكل أعلاه. سيحدد كود القناة الداخلي الإصدار. من الضروري استخدام هيكل Tcl_ChannelType الجديد إذا كنت تنشئ برنامج تشغيل قناة مكدس، بسبب المشاكل في التنفيذ السابق للقناة المكدسة (في 8.2.0 إلى 8.3.1).

قبل 8.4.0 (أي خلال الإصدارات اللاحقة من 8.3 والجزء المبكر من دورة تطوير 8.4) احتوى هيكل Tcl_ChannelType على الحقول التالية:

typedef struct Tcl_ChannelType {
    const char *typeName;
    Tcl_ChannelTypeVersion version;
    Tcl_DriverCloseProc *closeProc;
    Tcl_DriverInputProc *inputProc;
    Tcl_DriverOutputProc *outputProc;
    Tcl_DriverSeekProc *seekProc;
    Tcl_DriverSetOptionProc *setOptionProc;
    Tcl_DriverGetOptionProc *getOptionProc;
    Tcl_DriverWatchProc *watchProc;
    Tcl_DriverGetHandleProc *getHandleProc;
    Tcl_DriverClose2Proc *close2Proc;
    Tcl_DriverBlockModeProc *blockModeProc;
    Tcl_DriverFlushProc *flushProc;
    Tcl_DriverHandlerProc *handlerProc;
    Tcl_DriverTruncateProc *truncateProc;
} Tcl_ChannelType;

عند تسجيل الهيكل أعلاه كنوع قناة، يجب أن يكون حقل version دائمًا TCL_CHANNEL_VERSION_2.
