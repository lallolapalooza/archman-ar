Tcl_FSCopyFile(3)

int *lenPtr (out)
Si no es NULL، se llena con el número de elementos en el camino separado.
Tcl_Obj *basePtr (in)
El camino base desde el cual unir los elementos dados. Puede ser NULL.
int objc (in)
El número de elementos en objv.
Tcl_Obj *const objv[] (in)
Los elementos a unir al camino base dado.
Tcl_Obj *linkNamePtr (in)
El nombre del enlace a crear o leer.
Tcl_Obj *toPtr (in)
A qué debe vincularse el nombre del enlace linkNamePtr، o NULL si el enlace especificado por linkNamePtr debe ser leído.
int linkAction (in)
Una combinación OR de flags indicando qué tipo de enlace se debe crear (se ignorará si toPtr es NULL). Los bits válidos a establecer son TCL_CREATE_SYMBOLIC_LINK y TCL_CREATE_HARD_LINK. Cuando ambos flags están activos y el sistema de archivos subyacente puede realizar ambos tipos de enlaces, se prefieren los enlaces simbólicos.

الوصف

هناك العديد من الأسباب لاستدعاء وظائف واجهة برمجة التطبيقات Tcl_FS (على سبيل المثال، Tcl_FSAccess و Tcl_FSStat) بدلاً من استدعاء وظائف على مستوى النظام مثل access و stat مباشرة. أولاً، تعمل هذه الوظائف عبر الأنظمة الأساسية، لذا فإن أي امتداد يستدعي هذه الوظائف يجب أن يعمل دون تعديلات على كل من Unix و Windows. ثانيًا، يقوم تنفيذ بعض هذه الوظائف على Windows بتصحيح بعض الأخطاء في استدعاءات مستوى النظام. ثالثًا، تعالج استدعاءات الوظائف هذه أي تحويلات مسار من نوع "UTF إلى المنصة الأصلية" قد تكون ضرورية (والتي يمكنها تخزين نتائج هذه التحويلات مؤقتًا لتحسين الكفاءة في الاستدعاءات اللاحقة). رابعًا، وربما الأهم من ذلك، فإن جميع هذه الوظائف "واعية بأنظمة الملفات الافتراضية". يمكن لأي نظام ملفات افتراضي (VFS، للاختصار) تم تسجيله (عبر Tcl_FSRegister) إعادة توجيه الوصول إلى الملفات إلى وسائط بديلة أو طرق وصول. هذا يعني أن جميع هذه الوظائف (وبالتالي، الأوامر Tcl المقابلة file و glob و pwd و cd و open وما إلى ذلك) يمكن أن تعمل على "ملفات" ليست ملفات أصلية في نظام الملفات الأصلي. هذا يعني أيضًا أن أي امتداد Tcl يصل إلى نظام الملفات (FS، للاختصار) عبر واجهة برمجة التطبيقات هذه سيعمل تلقائيًا على أنه "واعي بأنظمة الملفات الافتراضية". من الواضح أنه إذا وصل امتداد إلى نظام الملفات الأصلي مباشرة (عبر واجهات برمجة التطبيقات الخاصة بالنظام، على سبيل المثال)، فلا يمكن لـ Tcl اعتراض مثل هذه الاستدعاءات.

إذا تم تسجيل أنظمة الملفات الافتراضية بشكل صحيح، فيمكن أن تكون "الملفات"، على سبيل المثال، بعيدة (كما هو موجود على خادم FTP بعيد) أو مؤرشفة (كما هو موجود داخل ملف .zip). توفر أنظمة الملفات المسجلة هذه جدول وظائف بحث لتنفيذ جزء أو كل الوظائف المذكورة هنا. أخيرًا، تقوم استدعاءات Tcl_FSStat و Tcl_FSLstat بتجريد أي ذاكرة تخزين مؤقت struct stat يتم الإعلان عنها بالفعل، مما يسمح باستخدام نفس الكود على الأنظمة التي تدعم الملفات الأكبر من 2 جيجابايت وتلك التي لا تدعمها.

تمت هيكلة واجهة برمجة التطبيقات Tcl_FS باستخدام Tcl_Obj ويمكنها تخزين التمثيلات الداخلية وسلاسل النصوص الأخرى المتعلقة بالمسارات (على سبيل المثال، الدليل الحالي) مؤقتًا. أحد الآثار الجانبية لهذا هو أنه لا يجب تمرير قيم ذات عداد مرجعي صفر إلى هذه الوظائف. قد يؤدي محاولة استدعاء مثل هذه الاستدعاءات إلى تسرب الذاكرة (في بعض الحالات، قد يقوم رمز نظام الملفات بإسقاط مرجع للقيمة الممررة، وبالتالي لا تفترض أنه بعد أي من هذه الاستدعاءات، لا تزال القيمة ذات عداد مرجعي صفر - قد تم زيادته). أو، في حالات تسرب الذاكرة المباشرة، قد يؤدي إلى خطأ تجزئة (أو خطأ وصول آخر للذاكرة) بسبب تحرير القيمة أثناء المعالجة المعقدة للقيم لضمان تطبيع المسار وكونه مطلقًا لنظام الملفات. الدرس العملي الذي يجب تعلمه هو:

Tcl_Obj *path = Tcl_NewStringObj(...);
Tcl_FSWhatever(path);
Tcl_DecrRefCount(path);

هذا غير صحيح وقد يتسبب في أخطاء في الذاكرة. يجب زيادة عداد المراجع للمسار path قبل تمريره، أو إنقاصه. لهذا السبب، تعتبر القيم ذات عداد مرجعي صفر غير صالحة كمسارات لنظام الملفات وأي استدعاء لوظيفة واجهة برمجة تطبيقات Tcl_FS مع مثل هذه القيم لن يؤدي إلى أي إجراء.

وظائف واجهة برمجة التطبيقات Tcl_FS

Tcl_FSCopyFile تحاول نسخ الملف المحدد بواسطة srcPathPtr إلى المسار المحدد بواسطة destPathPtr. إذا كان المساران المحددان على نفس نظام الملفات (وفقًا لـ Tcl_FSGetFileSystemForPath)، فسيتم استدعاء وظيفة نسخ الملف لنظام الملفات. بخلاف ذلك، تُرجع الوظيفة -1 وتضبط المتغير العام errno على EXDEV (رمز POSIX يعني رابطًا متقاطعًا).

Tcl_FSCopyDirectory تحاول نسخ الدليل المحدد بواسطة srcPathPtr إلى المسار المحدد بواسطة destPathPtr. إذا كان المساران المحددان على نفس نظام الملفات (وفقًا لـ Tcl_FSGetFileSystemForPath)، فسيتم استدعاء وظيفة نسخ الدليل لنظام الملفات. بخلاف ذلك، تُرجع الوظيفة -1 وتضبط المتغير العام errno على EXDEV (رمز POSIX يعني رابطًا متقاطعًا).

Tcl_FSCreateDirectory تحاول إنشاء الدليل المحدد بواسطة pathPtr عن طريق استدعاء وظيفة إنشاء الدليل لنظام الملفات الذي يمتلك الدليل.

Tcl_FSDeleteFile تحاول حذف الملف المحدد بواسطة pathPtr عن طريق استدعاء وظيفة حذف الملف لنظام الملفات.

Tcl_FSRemoveDirectory تحاول حذف الدليل المحدد بواسطة pathPtr عن طريق استدعاء وظيفة حذف الدليل لنظام الملفات.

Tcl_FSRenameFile تحاول إعادة تسمية الملف أو الدليل المحدد بواسطة srcPathPtr إلى المسار المحدد بواسطة destPathPtr. إذا كان المساران المحددان على نفس نظام الملفات (وفقًا لـ Tcl_FSGetFileSystemForPath)، فسيتم استدعاء وظيفة إعادة تسمية الملف لنظام الملفات (إذا لم تكن NULL). بخلاف ذلك، تُرجع الوظيفة -1 وتضبط المتغير العام errno على EXDEV (رمز POSIX يعني رابطًا متقاطعًا).

Tcl_FSListVolumes يستدعي كل نظام ملفات لديه وظيفة list volumes غير NULL ويطلب من هذه الأنظمة إرجاع قائمة وحداتها الجذرية. تقوم بتجميع القيم المرجعة في قائمة يتم إرجاعها إلى المتصل (مع عداد مرجعي صفر).

Tcl_FSEvalFileEx يقرأ محتويات ملف محدد بواسطة pathPtr باستخدام ترميز التشفير المحدد بواسطة encodingName ويقوم بتقييم محتوياته كبرنامج نصي Tcl. تُرجع نفس المعلومات التي تُرجعها Tcl_EvalObjEx. إذا كان encodingName NULL، فسيتم استخدام ترميز التشفير الافتراضي للنظام لقراءة محتويات الملف. إذا لم يكن من الممكن قراءة الملف، فسيتم إرجاع خطأ Tcl لوصف سبب عدم إمكانية قراءة الملف. حرف نهاية الملف (EOF) لجميع الأنظمة هو \x1A (^Z). إذا كنت تريد أن يكون لديك \x1A في شفرة المصدر في برنامجك، يمكنك استخدام \x1A، والذي سيستبدله مترجم Tcl على الأرجح بـ \x1A. Tcl_FSEvalFile هو نسخة أبسط من Tcl_FSEvalFileEx والتي تستخدم دائمًا ترميز التشفير الافتراضي للنظام عند قراءة محتويات الملف.

Tcl_FSLoadFile يقوم بتحميل ملف ثنائي ديناميكيًا في الذاكرة ويُرجع عناوين إجراءين داخل هذا الملف، إذا تم تعريفهما. سيتم استدعاء الوظيفة المناسبة لنظام الملفات الذي ينتمي إليه pathPtr. إذا لم ينفذ نظام الملفات هذا وظيفة (لن تنفذ معظم أنظمة الملفات الافتراضية ذلك بسبب قيود النظام على تحميل التعليمات البرمجية الثنائية ديناميكيًا)، فسيحاول Tcl نسخ الملف إلى دليل مؤقت وتحميل هذا الملف المؤقت. Tcl_FSUnloadFile يعكس العملية، ويطلب من العملية إزالة المكتبة المحددة بواسطة loadHandle من التنفيذ. لاحظ أنه، كما هو الحال مع الأمر unload، لا يمنح هذا المكتبة فرصة للتنظيف.

تُرجع كلتا الوظيفتين أعلاه رمز إكمال Tcl قياسيًا. إذا حدث خطأ، فسيتم وضع رسالة خطأ في نتيجة interp (ما لم يكن interp NULL، وهو أمر مسموح به). يمكن استخدام الرمز المميز المقدم بواسطة المتغير المشار إليه بواسطة loadHandlePtr مع Tcl_FindSymbol.

Tcl_FSMatchInDirectory يستخدمها رمز globbing للعثور على جميع الملفات التي تتطابق مع نمط معين في دليل. سيتم استدعاء الوظيفة المناسبة لنظام الملفات الذي ينتمي إليه pathPtr.

القيمة المرجعة هي نتيجة Tcl قياسية تشير إلى ما إذا كان قد حدث خطأ في globbing. يتم وضع رسائل الخطأ في interp (ما لم يكن interp NULL، وهو أمر مسموح به)، ولكن يتم وضع النتائج الجيدة في resultPtr المحدد.

ملاحظة: يقوم رمز glob بتنفيذ الأنماط المتكررة داخليًا، لذلك سيتم تمرير الأنماط البسيطة فقط إلى هذه الوظيفة، والتي يمكن مطابقتها باستخدام منطق string match. لمعالجة التكرار، ستستدعي Tcl هذه الوظيفة بشكل متكرر طالبةً فقط إرجاع الدلائل. حالة خاصة من استدعائها بنمط NULL تعني أن المسار يحتاج فقط إلى التحقق من النوع الصحيح.

Tcl_FSLink يحل محل إصدار المكتبة لـ readlink ويوسعها لدعم إنشاء الروابط. سيتم استدعاء الوظيفة المناسبة لنظام الملفات الذي ينتمي إليه linkNamePtr.

إذا كان toPtr NULL، يتم إجراء إجراء "قراءة رابط". النتيجة هي كائن Tcl يحدد محتويات الرابط الرمزي المحدد بواسطة linkNamePtr، أو NULL إذا لم يكن من الممكن قراءة الرابط. النتيجة هي ملكية كاملة للمتصل، والذي يجب عليه استدعاء Tcl_DecrRefCount عندما لا تكون النتيجة مطلوبة بعد. إذا كان toPtr غير NULL، فيجب على Tcl إنشاء رابط من أحد الأنواع الممررة في linkAction. هذا العلم هو مزيج OR من TCL_CREATE_SYMBOLIC_LINK و TCL_CREATE_HARD_LINK. إذا كان هناك خيار (أي، تم تمرير أكثر من علم واحد)، فإن اتفاقية Tcl هي تفضيل الروابط الرمزية. عندما يتم إنشاء رابط بنجاح، يجب أن تكون القيمة المرجعة هي toPtr (لذلك فهي بالفعل ملكية كاملة للمتصل). إذا لم ينجح، يتم إرجاع NULL.

Tcl_FSLstat يملأ البنية Tcl_StatBuf statPtr بمعلومات حول الملف المحدد. لا تحتاج إلى أذونات وصول للملف للحصول على هذه المعلومات، ولكنك تحتاج إلى أذونات بحث لجميع الدلائل المذكورة في المسار الذي يؤدي إلى الملف. تتضمن البنية Tcl_StatBuf معلومات حول الجهاز، و inode (دائمًا 0 على Windows)، ووضع الامتيازات، و nlink (دائمًا 1 على Windows)، و user ID (دائمًا 0 على Windows)، و group ID (دائمًا 0 على Windows)، و rdev (نفس الجهاز على Windows)، والحجم، وآخر وصول

إذا كان path موجودًا، فإن Tcl_FSLstat يُرجع 0 ويتم ملء بنية Tcl_StatBuf بالبيانات. بخلاف ذلك، يتم إرجاع -1 ولا يتم توفير بيانات الإحصائيات.

يحل Tcl_FSUtime محل إصدار المكتبة من utime.

يُرجع هذا 0 عند النجاح و -1 عند حدوث خطأ (وفقًا لتوثيق utime). إذا نجحت، ستقوم الدالة بتحديث قيم atime و mtime للملف المحدد.

يطبق Tcl_FSFileAttrsGet الوصول للقراءة للأمر الفرعي file attributes من الأمر file. سيتم استدعاء الدالة المناسبة لنظام الملفات الذي ينتمي إليه pathPtr.

إذا كانت النتيجة TCL_OK، فقد تم تعيين قيمة لـ objPtrRef، والتي ستكون صالحة مؤقتًا فقط (ما لم يتم استدعاء Tcl_IncrRefCount).

يطبق Tcl_FSFileAttrsSet الوصول للكتابة للأمر الفرعي file attributes من الأمر file. سيتم استدعاء الدالة المناسبة لنظام الملفات الذي ينتمي إليه pathPtr.

يطبق Tcl_FSFileAttrStrings جزءًا من الأمر الفرعي file attributes من الأمر file. سيتم استدعاء الدالة المناسبة لنظام الملفات الذي ينتمي إليه pathPtr.

قد تُرجع الإجراءية المسماة مصفوفة من السلاسل النصية، أو قد تُرجع NULL وتضع قائمة Tcl في objPtrRef. ستأخذ Tcl هذه القائمة وتزيد عدد مرجعها قبل استخدامها. بمجرد الانتهاء من استخدامها، ستقلل Tcl عدد مرجعها. لذلك، إذا كان يجب حذف القائمة بواسطة Tcl عند الانتهاء، فيجب أن يكون لها عدد مرجع صفر، وإذا كان لا يجب حذف القائمة، فيجب على نظام الملفات التأكد من أنها تحافظ على عدد مرجع للقيمة.

يتحقق Tcl_FSAccess مما إذا كان سيُسمح للعملية بقراءة أو كتابة أو اختبار وجود الملف (أو كائن نظام الملفات الآخر) الذي يحمل الاسم pathname. إذا كان pathname رابطًا رمزيًا في Unix، فسيتم اختبار حقوق الملف الذي يشير إليه هذا الرابط الرمزي.

عند النجاح (تم منح جميع الأذونات المطلوبة)، يتم إرجاع 0. عند حدوث خطأ (على الأقل بت واحد في الوضع المطلوب يمثل رفض إذن، أو حدث خطأ آخر)، يتم إرجاع -1.

يكمل Tcl_FSStat بنية Tcl_StatBuf بمعلومات حول الملف المحدد. لا يلزم وجود حقوق وصول للملف للحصول على هذه المعلومات، ولكن حقوق البحث مطلوبة في جميع الدلائل المذكورة في المسار إلى الملف. تتضمن بنية Tcl_StatBuf معلومات حول الجهاز، و inode (دائمًا 0 في Windows)، ووضع الامتياز، و nlink (دائمًا 1 في Windows)، و UID (دائمًا 0 في Windows)، و GID (دائمًا 0 في Windows)، و rdev (نفس الجهاز في Windows)، والحجم، ووقت آخر وصول، ووقت آخر تعديل، ووقت آخر تغيير للبيانات الوصفية. انظر PORTABLE STAT RESULT API لوصف كيفية كتابة كود محمول لاستيعاب بنية Tcl_StatBuf والوصول إليها.

إذا كان path موجودًا، فإن Tcl_FSStat يُرجع 0 ويتم ملء بنية statPtr بالبيانات. بخلاف ذلك، يتم إرجاع -1 ولا يتم توفير بيانات الإحصائيات.

يفتح Tcl_FSOpenFileChannel ملفًا محددًا بواسطة pathPtr ويُرجع مقبض قناة يمكن استخدامه للقراءة والكتابة في الملف. تم تصميم واجهة برمجة التطبيقات هذه على غرار الدالة fopen في مكتبة الإدخال/الإخراج القياسية في Unix. بناء الجملة ومعنى جميع الوسائط مشابهة لتلك الواردة في الأمر open بلغة Tcl عند فتح ملف. إذا حدث خطأ أثناء فتح القناة، فإن Tcl_FSOpenFileChannel يُرجع NULL ويسجل رمز خطأ POSIX يمكن استرداده باستخدام Tcl_GetErrno. بالإضافة إلى ذلك، إذا كان interp ليس NULL، فإن Tcl_FSOpenFileChannel يترك رسالة خطأ في نتيجة interp بعد أي خطأ.

القناة التي تم إنشاؤها حديثًا غير مسجلة في المفسر المقدم؛ لتسجيلها، استخدم Tcl_RegisterChannel. إذا تم إغلاق إحدى القنوات القياسية، stdin أو stdout أو stderr سابقًا، فإن إجراء إنشاء القناة الجديدة يعينها أيضًا كبديل للقنوات القياسية.

يحل Tcl_FSGetCwd محل إصدار المكتبة من getcwd.

يُرجع الدليل الحالي العمل لمكتبة Tcl. قد يختلف هذا عن الدليل الحالي العمل الأصلي، وهو ما يحدث عندما لا يكون الدليل الحالي العمل في نظام الملفات الأصلي.

النتيجة هي مؤشر إلى Tcl_Obj يحدد الدليل الحالي، أو NULL إذا لم يكن من الممكن تحديد الدليل الحالي. إذا تم إرجاع NULL، فسيتم ترك رسالة خطأ في نتيجة interp.

تم بالفعل زيادة عدد مرجع النتيجة للمتصل. عندما لا تكون هناك حاجة إليها، يجب تقليل عدد المرجع هذا. هذا ضروري لأمان الخيوط المتعددة، بحيث يمكن لخيوط متعددة الوصول إلى هذه الوظيفة والوظائف ذات الصلة، مما يضمن أن النتائج صالحة دائمًا.

يحل Tcl_FSChdir محل إصدار المكتبة من chdir. يتم تطبيع المسار ثم تمريره إلى نظام الملفات الذي يطالب به. إذا كان نظام الملفات هذا لا يطبق هذه الدالة، فإن Tcl تحول إلى مزيج من stat و access للتحقق مما إذا كان الدليل موجودًا ولديه الأذونات المناسبة.

بالنسبة للنتائج، انظر إلى توثيق chdir. إذا نجحت، يتم الاحتفاظ بسجل للمسار الناجح في cwdPathPtr للاستدعاءات اللاحقة لـ Tcl_FSGetCwd.

يُرجع Tcl_FSPathSeparator حرف الفاصل الذي سيتم استخدامه للعنصر الأكثر تحديدًا في المسار المحدد بواسطة pathPtr (أي، الجزء الأخير من المسار).

يتم إرجاع الفاصل كـ Tcl_Obj يحتوي على سلسلة نصية بطول 1. إذا كان المسار غير صالح، يتم إرجاع NULL.

يأخذ Tcl_FSJoinPath قيمة Tcl_Obj المعطاة، والتي يجب أن تكون قائمة صالحة (تسمح بعدد مرجع صفر)، ويُرجع قيمة المسار المعطاة من خلال اعتبار أول elements كقطع مسار صالحة (يمكن أن تكون كل قطعة مسارًا كاملاً، أو مسارًا جزئيًا، أو مجرد اسم محتمل لدليل أو ملف). إذا كان أي جزء من المسار هو بالفعل مسار مطلق، فسيتم تجاهل جميع الأجزاء السابقة. إذا كان elements أقل من 0، فسيتم استخدام القائمة بأكملها.

من الممكن أن تكون القيمة المرجعة في الواقع عنصرًا من القائمة المعطاة، لذلك يجب على المتصل توخي الحذر في زيادة عدد مرجع النتيجة قبل تحرير القائمة.

القيمة المرجعة، عادةً مع عدد مرجع صفر (ولكن قد تشارك تحت ظروف معينة)، تحتوي على المسار المدمج. يجب على المتصل إضافة عدد مرجع إلى القيمة قبل استخدامها. على وجه الخصوص، قد تكون القيمة المرجعة عنصرًا من القائمة المعطاة، لذلك إذا لم يتم أخذ عدد مرجع، فقد يؤدي تحرير القائمة إلى تحرير القيمة مبكرًا. إذا كان عدد العناصر صفرًا، فستكون القيمة المرجعة Tcl_Obj سلسلة فارغة.

يأخذ Tcl_FSSplitPath قيمة Tcl_Obj المعطاة، والتي يجب أن تكون مسارًا صالحًا، ويُرجع قيمة قائمة تحتوي على كل قطعة من هذا المسار كعنصر. يُرجع قيمة قائمة بعدد مرجع صفر. إذا كان lenPtr المقدم ليس NULL، فسيحتوي المتغير الذي يشير إليه على عدد العناصر في القائمة المرجعة.

يختبر Tcl_FSEqualPaths ما إذا كان المساران المعطيان يمثلان نفس كائن نظام الملفات. يُرجع 1 إذا كانا متساويين، و 0 إذا كانا مختلفين. إذا كان أي من المسارين NULL، فسيتم إرجاع 0 دائمًا.

يحاول Tcl_FSGetNormalizedPath استخراج تمثيل عادي فريد للمسار المعطى، والتي يمكن استخدام قيمتها النصية كمعرف فريد للملف.

يُرجع المسار الموحد، المملوك بواسطة Tcl، أو NULL إذا كان المسار غير صالح أو لم يكن من الممكن تحويله بشكل صحيح. استخراج المسارات المطلقة والموحدة فعال للغاية (لأن نظام الملفات يعمل بهذه التمثيلات داخليًا)، على الرغم من أن النتيجة عندما يحتوي نظام الملفات على العديد من الروابط الرمزية قد لا تكون النسخة الأكثر سهولة للمستخدم للمسار. القيمة المرجعة مملوكة بواسطة Tcl ولها عمر مكافئ لعمر pathPtr (ما لم يكن مسارًا نسبيًا، وفي هذه الحالة يمكن تحرير القيمة الموحدة في أي وقت يتغير فيه cwd) - يمكن للمتصل زيادة عدد المرجع إذا رغب في الاحتفاظ بنسخة أطول.

يأخذ Tcl_FSJoinToPath القيمة المعطاة، والتي يجب عادةً أن تكون مسارًا صالحًا أو NULL، ويدمج قطع المسار المعطاة.

يُرجع قيمة، عادةً مع عدد مرجع صفر (ولكن قد تشارك تحت ظروف معينة)، تحتوي على المسار المدمج. يجب على المتصل إضافة عدد مرجع إلى القيمة قبل استخدامها. إذا كان لأي من القيم التي تم تمريرها إلى هذه الدالة (pathPtr أو عناصر path) عدد مرجع صفر، فسيتم تحريرها عند إرجاع هذه الدالة.

يحاول Tcl_FSConvertToPathType تحويل Tcl_Obj المعطى إلى نوع مسار Tcl صالح، مع الأخذ في الاعتبار أنه قد يكون قد تم تغيير الدليل الحالي العمل حتى لو كان هذا القيمة من المفترض بالفعل أن تكون من النوع الصحيح. يمكن أن يبدأ اسم الملف بـ ~ (للإشارة إلى دليل المستخدم الحالي) أو ~ (للإشارة إلى دليل أي مستخدم).

إذا نجح التحويل (أي، القيمة مسار صالح في أحد أنظمة الملفات الحالية)، يتم إرجاع TCL_OK. بخلاف ذلك، يتم إرجاع TCL_ERROR، وقد تكون هناك رسالة خطأ في المفسر.

يستخرج Tcl_FSGetInternalRep التمثيل الداخلي لقيمة المسار المعطاة، في نظام الملفات المعطى. إذا كانت قيمة المسار تنتمي إلى نظام ملفات مختلف، فإنها تُرجع NULL. إذا كان التمثيل الداخلي NULL حاليًا، فإنه يحاول إنشائه عن طريق استدعاء Tcl_FSCreateInternalRepProc لنظام الملفات.

typedef int Tcl_FSDeleteFileProc(

كفاءة نظام الملفات

لا يلزم تنفيذ هذه الدوال لنظام ملفات معين لأن النواة لديها تنفيذ احتياطي متاح. راجع كل وصف فردي للعواقب المترتبة على ترك الحقل NULL.

LSTATPROC

معالج استدعاء Tcl_FSLstat. إذا لم يتم تنفيذه، ستحاول Tcl استخدام statProc المعرّف أعلاه بدلاً من ذلك. لذلك، لا يلزم تنفيذه إلا إذا كان نظام الملفات يمكنه التمييز بين استدعاءات stat و lstat.

typedef int Tcl_FSLstatProc(
        Tcl_Obj *pathPtr,
        Tcl_StatBuf *statPtr);

سلوك هذه الدالة مشابه جدًا لسلوك Tcl_FSStatProc المعرّف أعلاه، باستثناء أنه إذا تم تطبيقها على رابط رمزي، فإنها تعيد معلومات حول الرابط، وليس حول الملف الهدف.

COPYFILEPROC

معالج استدعاء Tcl_FSCopyFile. إذا لم يتم تنفيذه، ستعود Tcl إلى آلية نسخ تعتمد على open-r و open-w و fcopy. لذلك، لا يلزم تنفيذه إلا إذا كان نظام الملفات يمكنه أداء هذا الإجراء بكفاءة أكبر.

typedef int Tcl_FSCopyFileProc(
        Tcl_Obj *srcPathPtr,
        Tcl_Obj *destPathPtr);

القيمة المرتجعة هي نتيجة Tcl قياسية تشير إلى ما إذا كان قد حدث خطأ في عملية النسخ. لاحظ أن destPathPtr هو اسم الملف الذي يجب أن يصبح نسخة من srcPathPtr. إنه ليس أبدًا اسم دليل يمكن نسخ srcPathPtr إليه (أي أن الدالة أبسط بكثير من أمر Tcl الفرعي file copy). لاحظ أنه إذا كان نظام الملفات يدعم الروابط الرمزية، فإن Tcl ستستدعي دائمًا هذه الدالة ولن تستدعي copyDirectoryProc عند الحاجة لنسخها (حتى لو كانت روابط رمزية لدلائل). أخيرًا، إذا قرر نظام الملفات أنه لا يمكنه دعم إجراء file copy، فإن استدعاء Tcl_SetErrno(EXDEV) وإعادة نتيجة غير TCL_OK سيخبر Tcl باستخدام آلياتها الاحتياطية القياسية.

RENAMEFILEPROC

معالج استدعاء Tcl_FSRenameFile. إذا لم يتم تنفيذه، ستعود Tcl إلى آلية نسخ وحذف. لذلك، لا يلزم تنفيذه إلا إذا كان نظام الملفات يمكنه أداء هذا الإجراء بكفاءة أكبر.

typedef int Tcl_FSRenameFileProc(

typedef int Tcl_FSCopyDirectoryProc(
        Tcl_Obj *srcPathPtr,

القيمة المرتجعة هي نتيجة Tcl قياسية تشير إلى ما إذا كان قد حدث خطأ في عملية النسخ. في حالة حدوث خطأ، يجب وضع اسم الملف أو الدليل الذي تسبب في الخطأ في errorPtr. لاحظ أن destPathPtr هو اسم الدليل الذي يجب أن يصبح صورة مرآة لـ srcPathPtr. إنه ليس اسم دليل يمكن نسخ srcPathPtr إليه (أي أن الدالة أبسط بكثير من أمر Tcl الفرعي file copy). أخيرًا، إذا قرر نظام الملفات أنه لا يمكنه دعم إجراء نسخ الدليل، فإن استدعاء Tcl_SetErrno(EXDEV) وإعادة نتيجة غير TCL_OK سيخبر Tcl باستخدام آلياتها الاحتياطية القياسية.

LOADFILEPROC

معالج استدعاء Tcl_FSLoadFile. إذا لم يتم تنفيذه، ستعود Tcl إلى آلية النسخ إلى ملف مؤقت أصلي ثم استدعاء Tcl_FSLoadFile على هذا النسخة المؤقتة. لذلك، لا يلزم تنفيذه إلا إذا كان نظام الملفات يمكنه تحميل الشفرة مباشرة، أو يمكن تنفيذه ببساطة لإعادة TCL_ERROR لتعطيل وظيفة التحميل بالكامل في نظام الملفات هذا.

typedef int Tcl_FSLoadFileProc(
        Tcl_Interp *interp,
        Tcl_Obj *pathPtr,
        Tcl_LoadHandle *handlePtr,
        Tcl_FSUnloadFileProc *unloadProcPtr);

يعيد رمز اكتمال Tcl قياسي. في حالة حدوث خطأ، يتم ترك رسالة خطأ في نتيجة interp. تقوم الدالة بتحميل ملف شفرة ثنائي ديناميكيًا في الذاكرة. عند التحميل بنجاح، يجب ملء handlePtr برمز للملف الذي تم تحميله ديناميكيًا، ويجب ملء unloadProcPtr بعنوان إجراء. سيتم استدعاء إجراء التفريغ باستخدام Tcl_LoadHandle المعطى كمعلمته الوحيدة عندما تحتاج Tcl إلى تفريغ الملف. على سبيل المثال، بالنسبة لنظام الملفات الأصلي، Tcl_LoadHandle المرتجع حاليًا هو رمز يمكن استخدامه في TclpFindSymbol الخاص للوصول إلى الدوال في الشفرة الجديدة. كل نظام ملفات حر في تعريف Tcl_LoadHandle كما يتطلب. أخيرًا، إذا قرر نظام الملفات أنه لا يمكنه دعم إجراء تحميل الملف، فإن استدعاء Tcl_SetErrno(EXDEV) وإعادة نتيجة غير TCL_OK سيخبر Tcl باستخدام آلياتها الاحتياطية القياسية.

GETCWDPROC

معالج استدعاء Tcl_FSGetCwd. لا تحتاج معظم أنظمة الملفات إلى تنفيذ هذا. عادةً ما يتم استدعاؤه مرة واحدة فقط، إذا تم استدعاء getcwd قبل chdir. يمكن أن يكون NULL.

typedef Tcl_Obj *Tcl_FSGetCwdProc(
        Tcl_Interp *interp);

إذا كان نظام الملفات يدعم مفهومًا أصليًا للدليل الحالي (والذي قد يتغير بشكل مستقل عن Tcl)، فيجب أن تعيد هذه الدالة هذا الدليل الحالي كنتيجة، أو NULL إذا تعذر تحديد الدليل الحالي (على سبيل المثال، لا يملك المستخدم الأذونات المناسبة على الدليل الحالي). إذا تمت إعادة NULL، يتم ترك رسالة خطأ في نتيجة interp.

CHDIRPROC

معالج استدعاء Tcl_FSChdir. إذا لم تقم أنظمة الملفات بتنفيذ هذا، فسيتم محاكاته بواسطة سلسلة من عمليات التحقق من الوصول إلى الدليل. خلاف ذلك، تحتاج أنظمة الملفات الافتراضية التي تنفذه فقط إلى الاستجابة بنتيجة إرجاع إيجابية إذا كان pathPtr دليلًا صالحًا ويمكن الوصول إليه في نظام ملفاتها. لا تحتاج إلى تذكر النتيجة، حيث سيتم تذكرها تلقائيًا للاستخدام بواسطة Tcl_FSGetCwd. يجب على أنظمة الملفات الحقيقية تنفيذ الإجراء الصحيح (أي استدعاء واجهة برمجة تطبيقات النظام chdir الصحيحة).

typedef int Tcl_FSChdirProc(
        Tcl_Obj *pathPtr);

تغير Tcl_FSChdirProc دليل العمل الحالي للتطبيق إلى القيمة المحددة في pathPtr. تعيد الدالة -1 عند حدوث خطأ أو 0 عند النجاح.
