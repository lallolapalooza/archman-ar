Tcl_SetVar(3)

الحزمة: tcl

التراخيص: TCL

الاسم

Tcl_SetVar, Tcl_SetVar2, Tcl_GetVar, Tcl_GetVar2, Tcl_UnsetVar, Tcl_UnsetVar2 — ضبط وتعيين وقراءة وإلغاء تعيين متغيرات Tcl

الملخص

#include <tcl.h>

int
Tcl_SetVar(interp, varName, newValue, flags);
int
Tcl_SetVar2(interp, part1, part2, newValue, flags);
char *
Tcl_GetVar(interp, varName, flags);
char *
Tcl_GetVar2(interp, part1, part2, flags);
int
Tcl_UnsetVar(interp, varName, flags);
int
Tcl_UnsetVar2(interp, part1, part2, flags);

الوسائط

Tcl_Interp *interp (إدخال):
مفسر Tcl الذي سيتم استخدامه.
const char *varName (إدخال):
اسم المتغير المراد ضبطه أو إلغاء تعيينه أو قراءته. إذا كان المتغير مصفوفة، فسيحتوي الاسم على اسم المصفوفة والفهرس، مفصولين بحرف '('.
const char *newValue (إدخال):
القيمة الجديدة للمتغير.
int flags (إدخال):
مجموعة من الأعلام التي تتحكم في كيفية معالجة المتغير. يمكن أن تكون هذه القيم هي:
TCL_GLOBAL_ONLY:
يتم البحث عن المتغير في النطاق العام فقط. إذا لم يتم تعيين هذا العلم، فسيتم البحث عن المتغير في النطاق الحالي؛ إذا لم يتم العثور عليه، فسيتم البحث عنه في النطاق العام.
TCL_LEAVE_ERR_MSG:
إذا حدث خطأ، فلن يتم تعيين رسالة الخطأ في المفسر. هذا مفيد عند استدعاء وظائف Tcl من الكود المكتوب بلغة C، حيث لا تريد أن تؤدي الأخطاء إلى تعيين رسائل خطأ مكررة.
TCL_APPEND_VALUE:
للتطبيق. إذا تم تعيين هذا العلم، فإن قيمة المتغير الحالية (المحتوية على سلسلة نصية) سيتم إلحاقها بالقيمة الجديدة. يجب أن تكون القيمة الحالية للمتغير سلسلة نصية صالحة. عند الإلحاق، يجب أن تكون القيمة الأصلية للمتغير قائمة صالحة، بحيث تكون العملية عبارة عن إلحاق عنصر قائمة جديد بقائمة.
TCL_LIST_ELEMENT:
للتطبيق. إذا تم تعيين هذا العلم، فإن القيمة الجديدة سيتم التعامل معها كعنصر قائمة فردي وإلحاقها بالقيمة الحالية للمتغير. إذا لم تكن القيمة الحالية للمتغير قائمة، فسيتم تحويلها إلى قائمة وإلحاق العنصر الجديد بها. يجب ملاحظة أن هذا العلم يجب استخدامه بحذر، لأنه قد يؤدي إلى تعيينات غير متوقعة إذا لم يكن المتغير متوقعًا أن يكون قائمة. لن يتم إضافة عنصر إلى القائمة إذا كان عنصر القائمة الحالي هو العنصر الأول في القائمة أو القائمة الفرعية (أي قيمة المتغير الحالية فارغة، أو تحتوي على الحرف الوحيد '{'، أو تنتهي بـ '}'). عند الإلحاق، يجب أن تكون القيمة الأصلية للمتغير قائمة صالحة، بحيث تكون العملية عبارة عن إلحاق عنصر قائمة جديد بقائمة.
const char *part1 (إدخال):
الجزء الأول من اسم المتغير. بالنسبة للمتغيرات العادية، هذا هو اسم المتغير. بالنسبة لمصفوفات Tcl، هذا هو اسم المصفوفة.
const char *part2 (إدخال):
الجزء الثاني من اسم المتغير. بالنسبة للمتغيرات العادية، يكون هذا NULL. بالنسبة لمصفوفات Tcl، هذا هو الفهرس.

الوصف

تقوم إجراءات Tcl_SetVar و Tcl_SetVar2 بتعيين قيمة متغير Tcl. يتم معالجة الوسائط بنفس الطريقة في كلتا الإجرائيتين: Tcl_SetVar يأخذ اسم المتغير كاملاً كسلسلة نصية واحدة، بينما يقسم Tcl_SetVar2 اسم المتغير إلى جزأين (الجزء الأول والجزء الثاني) يمكن أن يكونا اسم مصفوفة وفهرسها.

إذا تم تعيين العلم TCL_APPEND_VALUE، فسيتم إلحاق newValue بقيمة المتغير الحالية. يجب أن تكون القيمة الحالية للمتغير سلسلة نصية. إذا تم تعيين العلم TCL_LIST_ELEMENT، فسيتم التعامل مع newValue كعنصر قائمة فردي وسيتم إلحاقه بقيمة المتغير الحالية. إذا لم تكن القيمة الحالية للمتغير قائمة، فسيتم تحويلها إلى قائمة وإلحاق العنصر الجديد بها. لن يتم إضافة عنصر إلى القائمة إلا إذا كان العنصر الأول في القائمة أو القائمة الفرعية (أي أن القيمة الحالية للمتغير فارغة، أو تحتوي على الحرف الوحيد '{'، أو تنتهي بـ '}'). عند الإلحاق، يجب أن تكون القيمة الأصلية للمتغير قائمة صالحة، بحيث تكون العملية عبارة عن إلحاق عنصر قائمة جديد بقائمة.

تُرجع Tcl_SetVar و Tcl_SetVar2 القيمة الحالية للمتغير بعد تعيينه، أو NULL في حالة حدوث خطأ. يتم إرجاع TCL_OK في حالة النجاح، و TCL_ERROR في حالة الفشل.

تقوم إجراءات Tcl_GetVar و Tcl_GetVar2 بإرجاع القيمة الحالية لمتغير. يتم معالجة وسائط هذه الإجراءات بنفس طريقة وسائط Tcl_SetVar و Tcl_SetVar2. في الظروف العادية، تكون القيمة المرجعة مؤشرًا إلى قيمة المتغير (المخزنة في بنية متغير Tcl ولن تتغير قبل المكالمة التالية لـ Tcl_SetVar أو Tcl_SetVar2). تستخدم Tcl_GetVar و Tcl_GetVar2 بتات العلم TCL_GLOBAL_ONLY و TCL_LEAVE_ERR_MSG، وكلاهما لهما نفس معنى Tcl_SetVar. إذا حدث خطأ في قراءة المتغير (مثل عدم وجود المتغير أو تحديد عنصر مصفوفة لمتغير قياسي)، فسيتم إرجاع NULL.

يمكن استخدام Tcl_UnsetVar و Tcl_UnsetVar2 لإزالة متغير، بحيث تُرجع الاستدعاءات المستقبلية لـ Tcl_GetVar أو Tcl_GetVar2 للمتغير خطأ. يتم معالجة وسائط هذه الإجراءات بنفس طريقة وسائط Tcl_GetVar و Tcl_GetVar2. إذا تمت إزالة المتغير بنجاح، فسيتم إرجاع TCL_OK. إذا تعذر إزالة المتغير لأنه غير موجود، فسيتم إرجاع TCL_ERROR. إذا تم تحديد عنصر مصفوفة، فسيتم إزالة العنصر المحدد ولكن تبقى المصفوفة. إذا تم تحديد اسم مصفوفة بدون فهرس، فسيتم إزالة المصفوفة بأكملها.
