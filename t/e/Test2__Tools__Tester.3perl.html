Test2::Tools::Tester(3perl) دليل مبرمجي بيرل Test2::Tools::Tester(3perl)

الاسم

Test2::Tools::Tester - أدوات للمساعدة في اختبار أدوات الاختبار الأخرى.

الوصف

هذه مجموعة من الأدوات المفيدة عند اختبار أدوات الاختبار الأخرى.

الملخص

use Test2::Tools::Tester qw/event_groups filter_events facets/;
use Test2::Tools::Basic qw/plan pass ok/;
use Test2::Tools::Compare qw/is like/;
my $events = intercept {
    plan 11;
    pass('pass');
    ok(1, 'pass');
    is(1, 1, "pass");
    like(1, 1, "pass");
};
# التقاط الأحداث التي تم إنشاؤها بواسطة أدوات في Test2::Tools::Basic
my $basic = filter $events => 'Test2::Tools::Basic';
# التقاط الأحداث التي تم إنشاؤها بواسطة Test2::Tools::Basic;
my $compare = filter $events => 'Test2::Tools::Compare';
# التقاط الأحداث التي تم إنشاؤها بواسطة أدوات تسمى 'ok'.
my $oks = filter $events => qr/.*::ok$/;
my $grouped = group_events $events;
# تقسيم الأحداث إلى هذا الهيكل:
{
    '__NA__' => [ ... ],
    'Test2::Tools::Basic' => {
        '__ALL__' => [ $events->[0], $events->[1], $events->[2] ],
        plan => [ $events->[0] ],
        pass => [ $events->[1] ],
        ok => [ $events->[2] ],
    },
    Test2::Tools::Compare => { ... },
}
# الحصول على قائمة بجميع جوانب التأكيد (assert facets) من قائمة الأحداث.
my $assert_facets = facets assert => $events;
# [
#   bless({ details => 'pass', pass => 1}, 'Test2::EventFacet::Assert'),
#   bless({ details => 'pass', pass => 1}, 'Test2::EventFacet::Assert'),
# ]
# نفس الشيء، لجوانب المعلومات (info facets)
my $info_facets = facets info => $events;

التصدير

لا يتم تصدير أي وظائف فرعية افتراضيًا.

$array_ref = filter $events => $PACKAGE
$array_ref = filter $events => $PACKAGE1, $PACKAGE2
$array_ref = filter $events => qr/match/
$array_ref = filter $events => qr/match/, $PACKAGE
تأخذ هذه الوظيفة مصفوفة أحداث كوسيط أول. يجب أن تكون جميع الوسائط الإضافية إما اسم حزمة، أو تعبير عادي. سيتم إرجاع أي حدث يتم إنشاؤه بواسطة أداة في أي من الحزم، أو بواسطة أداة تطابق أيًا من التعبيرات العادية، في مصفوفة.
$grouped = group_events($events)
تقوم هذه الوظيفة بالتكرار على جميع الأحداث في مصفوفة الوسيط وتقسيمها إلى مجموعات. هيكل البيانات الناتج هو: 
{ PACKAGE => { SUBNAME => [ $EVENT1, $EVENT2, ... ]}}

إذا لم تكن حزمة الحدث معروفة، فسيتم وضعها في مصفوفة تحت مفتاح '__NA__' في الجذر. إذا لم يكن اسم الوظيفة الفرعية معروفًا، فعادةً ما يذهب تحت مفتاح '__ANON__' داخل اسم الحزمة.

بالإضافة إلى ذلك، يوجد مفتاح '__ALL__' تحت كل حزمة يخزن جميع الأحداث المصنفة في تلك المجموعة.

مثال أكثر اكتمالاً:

{
    '__NA__' => [ $event->[3] ],
    'Test2::Tools::Basic' => {
        '__ALL__' => [ $events->[0], $events->[1], $events->[2] ],
        plan => [ $events->[0] ],
        pass => [ $events->[1] ],
        ok => [ $events->[2] ],
    },
}
$arrayref = facets TYPE => $events
ستقوم هذه الوظيفة بتجميع قائمة بجميع جوانب النوع المحدد التي تم العثور عليها في مصفوفة الأحداث. إذا كان للجانب فئة "Test2::EventFacet::TYPE" متاحة، فسيتم بناء الجانب في مثيل للفئة، وإلا فسيتم تركه كـ hashref. يتم الحفاظ على ترتيب الجوانب. 
my $assert_facets = facets assert => $events;
# [
#   bless({ details => 'pass', pass => 1}, 'Test2::EventFacet::Assert'),
#   bless({ details => 'pass', pass => 1}, 'Test2::EventFacet::Assert'),
# ]
