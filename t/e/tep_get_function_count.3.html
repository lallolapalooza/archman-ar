tep_get_function_count(3)

الحزمة: libtraceevent

التراخيص: GPL-2.0-only, LGPL-2.1-only

LIBTRACEEVENT(3) دليل libtraceevent LIBTRACEEVENT(3)

الاسم

tep_set_function_resolver, tep_reset_function_resolver, tep_register_function, tep_register_print_string, tep_get_function_count - واجهات برمجة التطبيقات ذات الصلة بالوظائف tep

الملخص

#include <event-parse.h>
typedef char *(tep_func_resolver_t)(void *priv, unsigned long long *addrp, char **modp);
int tep_set_function_resolver(struct tep_handle *tep, tep_func_resolver_t *func, void *priv);
void tep_reset_function_resolver(struct tep_handle *tep);
int tep_register_function(struct tep_handle *tep, char *name, unsigned long long addr, char *mod);
int tep_register_print_string(struct tep_handle *tep, const char *fmt, unsigned long long addr);
int tep_get_function_count(struct tep_handle *tep)

الوصف

قد تمتلك بعض الأدوات بالفعل طريقة لحل وظائف النواة. تتيح لها واجهات برمجة التطبيقات هذه الاستمرار في استخدامها بدلاً من تكرار جميع الإدخالات بداخلها.

النوع tep_func_resolver_t هو النموذج الأولي لـ محلل وظائف النواة البديل. تتلقى هذه الدالة مؤشرًا إلى سياقها المخصص (تم تعيينه باستخدام استدعاء tep_set_function_resolver()) وعنوان وظيفة نواة، والتي يجب حلها. في حالة النجاح، يجب أن تعيد اسم الوظيفة ووحدتها (إن وجدت) في modp.

تقوم الدالة tep_set_function_resolver() بتسجيل func كمحلل وظائف نواة بديل. الوسيط tep هو سياق محلل أحداث التتبع. الوسيط priv هو سياق مخصص للدالة func. يتم استخدام محلل الوظائف بواسطة واجهات برمجة التطبيقات tep_find_function(3)، tep_find_function_address(3)، و tep_print_func_field() لحل عنوان الوظيفة إلى اسم الوظيفة.

تقوم الدالة tep_reset_function_resolver() بإعادة تعيين محلل وظائف النواة إلى الوظيفة الافتراضية. الوسيط tep هو سياق محلل أحداث التتبع.

يمكن استخدام واجهات برمجة التطبيقات هذه للعثور على اسم الوظيفة وعنوان البدء، عن طريق العنوان المحدد. لا يلزم أن يكون العنوان المحدد دقيقًا، وسيتم تحديد الوظيفة التي تحتويه.

تقوم الدالة tep_register_function() بتسجيل اسم وظيفة مقابل لعنوان (اختياري) ووحدة. يتم استخدام هذا التعيين في حالة أن متعقب الوظائف أو الأحداث لديه "%pS" معلمة في سلسلة التنسيق الخاصة به. من الشائع تمرير أسماء وظائف kallsyms مع عناوينها المقابلة باستخدام هذه الوظيفة. الوسيط tep هو سياق محلل أحداث التتبع. name هو اسم الوظيفة، يتم نسخ السلسلة داخليًا. addr هو عنوان بدء الوظيفة. mod هي وحدة النواة التي قد تكون الوظيفة فيها (NULL إن لم تكن).

تقوم الدالة tep_register_print_string() بتسجيل سلسلة بالعنوان الذي تم تخزينها فيه في النواة. يتم تمرير بعض السلاسل الداخلية للنواة بعنوان ثابت إلى أحداث معينة. "%s" في حقل تنسيق الحدث الذي يحتوي على عنوان يحتاج إلى معرفة السلسلة التي ستكون في هذا العنوان. توفر tep_register_print_string() التحليل مع التعيين بين عناوين النواة وهذه السلاسل. الوسيط tep هو سياق محلل أحداث التتبع. fmt هي السلسلة التي سيتم تسجيلها، ويتم نسخها داخليًا. addr هو العنوان الذي تقع فيه السلسلة.

tep_get_function_count() تعيد عدد الوظائف المسجلة في معالج tep.

قيمة الإرجاع

تعيد الدالة tep_set_function_resolver() القيمة 0 في حالة النجاح، أو -1 في حالة حدوث خطأ.

تعيد الدالة tep_register_function() القيمة 0 في حالة النجاح. في حالة حدوث خطأ، يتم إرجاع -1، ويتم تعيين errno إلى رقم الخطأ المناسب.

تعيد الدالة tep_register_print_string() القيمة 0 في حالة النجاح. في حالة حدوث خطأ، يتم إرجاع -1، ويتم تعيين errno إلى رقم الخطأ المناسب.

char *my_resolve_kernel_addr(void *context,

{
        struct db *function_database = context;
        struct symbol *sym = sql_lookup(function_database, *addrp);
        if (!sym)
                return NULL;
        *modp = sym->module_name;
        return sym->name;
}
void show_function( unsigned long long addr)
{
        unsigned long long fstart;
        const char *fname;
        if (tep_set_function_resolver(tep, my_resolve_kernel_addr,
                                      function_database) != 0) {
                /* فشل في تسجيل my_resolve_kernel_addr */
        }
        /* تستخدم واجهات برمجة التطبيقات هذه my_resolve_kernel_addr() لحل العنوان */
        fname = tep_find_function(tep, addr);
        fstart = tep_find_function_address(tep, addr);
        /*
           العنوان موجود في وظيفة اسمها fname، تبدأ عند عنوان fstart،
           في الإزاحة (addr - fstart)
        */
        tep_reset_function_resolver(tep);
}
