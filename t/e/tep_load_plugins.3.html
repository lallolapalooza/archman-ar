TEP_LOAD_PLUGINS(3)

اسم

tep_load_plugins - تحميل المكونات الإضافية

المكتبة

libtep - مكتبة التتبع والأحداث

الصيغة

                    #include <tep.h>

int tep_load_plugins(const char *path);

الوصف

تقوم الدالة tep_load_plugins بتحميل المكونات الإضافية من المسار المحدد.

يجب أن يتم تهيئة libtep أولاً باستخدام tep_init().

الوسائط

path
المسار الذي يحتوي على المكونات الإضافية.

القيمة المرجعة

ترجع الدالة 0 عند النجاح، و -1 عند الفشل.

أمثلة

                    #include <tep.h>
                    int main() {
                        if (tep_init() < 0) {
                            return 1;
                        }
                        if (tep_load_plugins("/usr/lib/tep") < 0) {
                            return 1;
                        }
                        // Use TEP functionality
                        tep_shutdown();
                        return 0;
                    }

انظر أيضًا

tep_init(3), tep_shutdown(3)
