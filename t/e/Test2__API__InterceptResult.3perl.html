Test2::API::InterceptResult(3perl) دليل مبرمجي Perl Test2::API::InterceptResult(3perl)

الاسم

Test2::API::InterceptResult - تمثيل قائمة بالأحداث.

الوصف

يمثل هذا الصنف قائمة بالأحداث، والتي يتم الحصول عليها عادةً باستخدام intercept() من Test2::API.

هذا الصنف موجه للأشخاص الذين يرغبون في التحقق من نتائج أدوات الاختبار التي يكتبونها.

يوفر هذا الصنف طرقًا لتطبيع وتلخيص أو تعيين قائمة الأحداث. يجعل ناتج هذه العمليات التحقق من أدوات الاختبار الخاصة بك والأحداث التي تولدها أسهل بكثير. في معظم الحالات، هذا يغنيك عن الحاجة إلى فهم عميق لنموذج الحدث/الجانب.

ملخص

عادةً ما تحصل على مثيل لهذا الصنف عند استخدام intercept() من Test2::API.

use Test2::V0;
use Test2::API qw/intercept/;
my $events = intercept {
    ok(1, "pass");
    ok(0, "fail");
    todo "broken" => sub { ok(0, "fixme") };
    plan 3;
};
# هذا هو عادةً البناء الأكثر فائدة
# squash_info() يدمج التأكيدات والتشخيصات المرتبطة
#   (ويعيد مثيلًا جديدًا مع التعديلات)
# flatten() يكثف بيانات الجانب إلى التفاصيل الرئيسية لكل حدث
#   (ويعيد تلك الهياكل في مرجع مصفوفة)
is(
    $events->squash_info->flatten(),
    [
        {
            causes_failure => 0,
            name => 'pass',
            pass => 1,
            trace_file => 'xxx.t',
            trace_line => 5,
        },
        {
            causes_failure => 1,
            name => 'fail',
            pass => 0,
            trace_file => 'xxx.t',
            trace_line => 6,
            # يمكن أن تكون هناك أكثر من رسالة تشخيص واحدة، لذا فإن هذا
            # دائمًا مصفوفة عند وجودها.
            diag => ["Failed test 'fail'\nat xxx.t line 6."],
        },
        {
            causes_failure => 0,
            name => 'fixme',
            pass => 0,
            trace_file => 'xxx.t',
            trace_line => 7,
            # يمكن أن تكون هناك أكثر من رسالة تشخيص واحدة أو سبب tod
            # ، لذا هذه دائمًا مصفوفة عند وجودها.
            todo => ['broken'],
            # تم تحويل رسالة Diag إلى ملاحظة نظرًا لأن التأكيد كان
            # TODO
            note => ["Failed test 'fixme'\nat xxx.t line 7."],
        },
        {
            causes_failure => 0,
            plan => 3,
            trace_file => 'xxx.t',
            trace_line => 8,
        },
    ],
    "الأحداث المسطحة تبدو كما نتوقع"
);

راجع Test2::API::InterceptResult::Event للحصول على وصف كامل لما يقدمه flatten() لكل حدث.

الطرق

يرجى ملاحظة أن أي طرق لا تعدل المثيل الأصلي ما لم يُطلب منها ذلك.

البناء

$events = Test2::API::InterceptResult->new(@EVENTS)
$events = Test2::API::InterceptResult->new_from_ref(\@EVENTS)
تنشئ هذه الطريقة مثيلاً جديدًا من Test2::API::InterceptResult من الأحداث المعطاة.

في الشكل الأول، يتم إرجاع مرجع مصفوفة جديد مبارك. في شكل 'new_from_ref'، يتم مباركة المرجع الذي تمرره مباشرة.

سيؤدي كل من هذين إلى إطلاق استثناء إذا تم استدعاؤهما في سياق فارغ. هذا مهم بشكل أساسي لطرق "التصفية" المدرجة أدناه والتي تعيد عادةً مثيلاً جديدًا، فإنها تطلق استثناءً في مثل هذه الحالات لأنه من المحتمل أن شخصًا ما قصد تصفية الأصل في مكانه.

$clone = $events->clone()
إنشاء نسخة مستنسخة من الأحداث الأصلية. لاحظ أن هذه نسخة عميقة، يتم تكرار الهيكل بأكمله. يستخدم هذا "dclone" من Storable لتحقيق النسخ العميق.

التطبيع

@events = $events->event_list
يعيد هذا جميع الأحداث في شكل قائمة.
$hub = $events->hub
يعيد هذا مثيلاً جديدًا من Test2::Hub قام بمعالجة جميع الأحداث الموجودة في المثيل. يمنحك هذا طريقة بسيطة لفحص تغييرات الحالة التي تسببها أحداثك.
$state = $events->state
يعيد هذا ملخصًا لحالة المحور بعد معالجة جميع الأحداث. 
{
    count        => 2,      # عدد التأكيدات التي تم إجراؤها
    failed       => 1,      # عدد حالات فشل الاختبار التي شوهدت
    is_passing   => 0,      # منطقي، صحيح إذا كان الاختبار سيمر
                            # بعد معالجة الأحداث.
    plan         => 2,      # خطة، إما رقم، أو غير معرف، 'SKIP'، أو 'NO PLAN'
    follows_plan => 1,      # صحيح إذا كانت هناك خطة وتم اتباعها.
                            # خطأ إذا لم تتطابق الخطة والتأكيدات،
                            # غير معرف إذا لم تكن هناك خطة في
                            # قائمة الأحداث.
    bailed_out   => undef,  # غير معرف ما لم يكن هناك خروج مبكر في
                            # الأحداث، وفي هذه الحالة ستكون سلسلة نصية
                            # تشرح سبب الخروج المبكر، إذا لم يكن هناك
                            # سبب معطى، فسيتم تعيينه ببساطة إلى
                            # صحيح (1).
    skip_reason  => undef,  # إذا كان هناك skip_all، فستعطي هذا
                            # السبب.
}
$new = $events->upgrade
$events->upgrade(in_place => $BOOL)
ملاحظة: هذا يعيد عادةً مثيلاً جديدًا، تاركًا الأصل دون تغيير. إذا قمت باستدعائه في سياق فارغ، فسوف يطلق استثناءً. إذا كنت ترغب في تعديل الأصل، فيجب عليك تمرير الخيار "in_place => 1". يمكنك استدعاء هذا في سياق فارغ عندما تطلب تعديله في مكانه. يعيد الشكل في مكانه المثيل الذي تم تعديله حتى تتمكن من ربط الطرق.

سيؤدي هذا إلى إنشاء نسخة من القائمة حيث تم تحويل جميع الأحداث إلى مثيلات Test2::API::InterceptResult::Event. هذا مفيد للغاية لأن Test2::API::InterceptResult::Event يوفر واجهة أفضل بكثير للعمل مع الأحداث. هذا يسمح لك بتجنب التفكير في أنواع الأحداث القديمة.

هذا يعني أيضًا أن اختباراتك ضد القائمة ليست هشة إذا تغيرت الأداة التي تختبرها عشوائيًا نوع الأحداث التي تولدها (أي التغيير من Test2::Event::Ok إلى Test2::Event::Pass، كلاهما يقوم بتأكيدات وكلاهما سيتم تطبيعهما إلى مثيلات Test2::API::InterceptResult::Event متطابقة (أو قريبة بما فيه الكفاية)).

في الواقع، أنت تريد هذا دائمًا تقريبًا، والسبب الوحيد لعدم القيام بذلك تلقائيًا هو التأكد من أن أداة intercept() متوافقة مع الإصدارات السابقة.

$new = $events->squash_info
$events->squash_info(in_place => $BOOL)
ملاحظة: هذا يعيد عادةً مثيلاً جديدًا، تاركًا الأصل دون تغيير. إذا قمت باستدعائه في سياق فارغ، فسوف يطلق استثناءً. إذا كنت ترغب في تعديل الأصل، فيجب عليك تمرير الخيار "in_place => 1". يمكنك استدعاء هذا في سياق فارغ عندما تطلب تعديله في مكانه. يعيد الشكل في مكانه المثيل الذي تم تعديله حتى تتمكن من ربط الطرق.

ملاحظة: سيتم تحويل جميع الأحداث في المثيل الجديد أو المعدل إلى مثيلات Test2::API::InterceptResult::Event. لا توجد طريقة لتجنب ذلك، عملية الضغط تتطلب فئة الحدث المحدثة.

كان Test::More والعديد من الأدوات القديمة الأخرى ترسل ملاحظات، وتشخيصات، وتأكيدات كأحداث منفصلة. كان الاختبار الفرعي في Test::More يرسل ملاحظة باسم الاختبار الفرعي، وتأكيد الاختبار الفرعي، وأخيرًا حدث تشخيص إذا فشل الاختبار الفرعي. هذه الطريقة ستطبع الأشياء عن طريق ضغط الملاحظة والتشخيص في نفس الحدث مثل الاختبار الفرعي (هذا يختلف عن وضعها في الاختبار الفرعي، وهو ما لا يحدث).

التصفية

ملاحظة: هذه عادةً ما تعيد مثيلات جديدة، تاركة الأصل دون تغيير. إذا قمت باستدعائها في سياق فارغ، فسوف تطلق استثناءات. إذا كنت ترغب في تعديل الأصول، فيجب عليك تمرير الخيار "in_place => 1". يمكنك استدعاء هذه في سياق فارغ عندما تطلب تعديلها في مكانها. تعيد الأشكال في مكانها المثيل الذي تم تعديله حتى تتمكن من ربط الطرق.
