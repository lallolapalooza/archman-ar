tep_set_loglevel(3)

الحزمة: libtraceevent

التراخيص: GPL-2.0-only, LGPL-2.1-only

LIBTRACEEVENT(3) دليل libtraceevent LIBTRACEEVENT(3)

الاسم

tep_set_loglevel - تعيين مستوى تسجيل المكتبة

ملخص

#include <event-parse.h>
enum tep_loglevel {
        TEP_LOG_NONE = 0,
        TEP_LOG_CRITICAL,
        TEP_LOG_ERROR,
        TEP_LOG_WARNING,
        TEP_LOG_INFO,
        TEP_LOG_DEBUG,
        TEP_LOG_ALL
};
int tep_set_loglevel(enum tep_loglevel level);

الوصف

تقوم الدالة tep_set_loglevel() بتعيين مستوى تسجيلات المكتبة التي ستتم طباعتها على وحدة التحكم. مستويات تسجيل المكتبة هي:

        TEP_LOG_NONE - لا تطبع أي تسجيلات.
        TEP_LOG_CRITICAL - طباعة التسجيلات الحرجة، مشكلة قد تسبب تعطلًا.
        TEP_LOG_ERROR - طباعة تسجيلات الأخطاء، مشكلة قد تعطل المنطق الأساسي لواجهة برمجة التطبيقات.
        TEP_LOG_WARNING - طباعة التحذيرات، مشكلة قد تحد من نتيجة واجهة برمجة التطبيقات.
        TEP_LOG_INFO - طباعة معلومات حول التنفيذ الطبيعي لواجهة برمجة التطبيقات.
        TEP_LOG_DEBUG - طباعة معلومات التصحيح.
        TEP_LOG_ALL - طباعة التسجيلات من جميع المستويات.

يعني تعيين مستوى التسجيل إلى قيمة معينة أنه سيتم أيضًا طباعة التسجيلات من المستويات السابقة. على سبيل المثال، TEP_LOG_WARNING سيطبع أي تسجيلات ذات خطورة TEP_LOG_WARNING و TEP_LOG_ERROR و TEP_LOG_CRITICAL. مستوى التسجيل الافتراضي هو TEP_LOG_CRITICAL.

مثال

#include <event-parse.h>
tep_set_loglevel(TEP_LOG_ALL);
...
/* استدعاء واجهات برمجة تطبيقات libtraceevent وملاحظة أي تسجيلات تنتجها */
...
tep_set_loglevel(TEP_LOG_CRITICAL);
