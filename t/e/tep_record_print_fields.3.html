tep_record_print_fields(3)

الحزمة: libtraceevent

التراخيص: GPL-2.0-only, LGPL-2.1-only

LIBTRACEEVENT(3) دليل libtraceevent LIBTRACEEVENT(3)

الاسم

tep_print_field_content, tep_print_fields, tep_print_num_field, tep_print_func_field, tep_record_print_fields, tep_record_print_selected_fields - طباعة محتوى الحقل.

الملخص

#include <event-parse.h>
#include <trace-seq.h>
void tep_print_field_content(struct trace_seq *s, void *data, int size, struct tep_format_field *field);
void tep_print_fields(struct trace_seq *s, void *data, int size, struct tep_event *event);
int tep_print_num_field(struct trace_seq *s, const char *fmt, struct tep_event *event, const char *name, struct tep_record *record, int err);
int tep_print_func_field(struct trace_seq *s, const char *fmt, struct tep_event *event, const char *name, struct tep_record *record, int err);
void tep_record_print_fields(struct trace_seq *s, struct tep_record *record, struct tep_event *event);
void tep_record_print_selected_fields(struct trace_seq *s, struct tep_record *record, struct tep_event *event, int select_mask);

الوصف

تقوم هذه الدوال بطباعة بيانات الحقول المسجلة، وفقًا لنوع الحقل.

تقوم الدالة tep_print_field_content() باستخراج القيمة الخام المسجلة data للحقل field وطباعتها في s، وفقًا لنوع الحقل.

تقوم الدالة tep_print_fields() بطباعة اسم كل حقل متبوعًا بقيمة الحقل المسجل وفقًا لنوع الحقل:

"اسم_الحقل1=قيمة_الحقل1 اسم_الحقل2=قيمة_الحقل2 ..."

تقوم بالتكرار على جميع حقول event، وتستدعي tep_print_field_content() لكل منها.

تقوم الدالة tep_print_num_field() بطباعة حقل رقمي بسلسلة تنسيق معينة. يتم البحث في event عن حقل بالاسم name. إذا تم العثور على مثل هذا الحقل، يتم استخراج قيمته من record وطباعتها في s، وفقًا لسلسلة التنسيق fmt المعطاة. إذا كانت المعلمة err غير صفرية، وعند حدوث خطأ - يتم طباعته في s.

تقوم الدالة tep_print_func_field() بطباعة حقل دالة بسلسلة تنسيق معينة. يتم البحث في event عن حقل بالاسم name. إذا تم العثور على مثل هذا الحقل، يتم استخراج قيمته من record. يُفترض أن تكون القيمة عنوان دالة، ويتم إجراء بحث للعثور على اسم هذه الدالة. يتم طباعة اسم الدالة (إذا تم العثور عليها) وعنوانها في s، وفقًا لسلسلة التنسيق fmt المعطاة. إذا كانت المعلمة err غير صفرية، وعند حدوث خطأ - يتم طباعته في s.

تقوم الدالة tep_record_print_fields() بطباعة اسم الحقل متبوعًا بقيمته لجميع حقول السجل.

تقوم الدالة tep_record_print_selected_fields() بطباعة اسم الحقل متبوعًا بقيمته لمجموعة فرعية محددة من حقول السجل. يتم تحديد الحقول المراد طباعتها بواسطة قناع البت select_mask.

قيمة الإرجاع

تُرجع الدالتان tep_print_num_field() و tep_print_func_field() القيمة 1 عند النجاح، -1 في حالة حدوث خطأ، أو 0 إذا كان مخزن الطباعة s ممتلئًا.

مثال

#include <event-parse.h>
#include <trace-seq.h>
...
struct tep_handle *tep = tep_alloc();
...
struct trace_seq seq;
trace_seq_init(&seq);
struct tep_event *event = tep_find_event_by_name(tep, "timer", "hrtimer_start");
...
void process_record(struct tep_record *record)
{
        struct tep_format_field *field_pid = tep_find_common_field(event, "common_pid");
        trace_seq_reset(&seq);
        /* طباعة قيمة "common_pid" */
        tep_print_field_content(&seq, record->data, record->size, field_pid);
        /* طباعة جميع حقول حدث "hrtimer_start" */
        tep_print_fields(&seq, record->data, record->size, event);
        /* طباعة قيمة حقل "expires" بتنسيق مخصص */
        tep_print_num_field(&seq, " timer expires in %llu ", event, "expires", record, 0);
        /* طباعة عنوان واسم حقل "function" بتنسيق مخصص */
        tep_print_func_field(&seq, " timer function is %s ", event, "function", record, 0);
 }

trace-seq.h
        ملف الرأس لتضمينه للوصول إلى واجهات برمجة تطبيقات تسلسلات التتبع.
        تُستخدم تسلسلات التتبع للسماح للدالة باستدعاء عدة دوال أخرى لإنشاء سلسلة بيانات للاستخدام.

        مفتاح الرابط للإضافة عند بناء برنامج يستخدم المكتبة.
