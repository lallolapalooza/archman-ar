tep_load_plugins_hook(3)

الحزمة: libtraceevent

التراخيص: GPL-2.0-only, LGPL-2.1-only

LIBTRACEEVENT(3) دليل libtraceevent LIBTRACEEVENT(3)

الاسم

tep_load_plugins, tep_unload_plugins, tep_load_plugins_hook, tep_add_plugin_path, tep_plugin_add_option - تحميل / إلغاء تحميل مكونات traceevent.

الملخص

#include <event-parse.h>
struct tep_plugin_list *tep_load_plugins(struct tep_handle *tep);
void tep_unload_plugins(struct tep_plugin_list *plugin_list, struct tep_handle *tep);
void tep_load_plugins_hook(struct tep_handle *tep, const char *suffix,
                           void (*load_plugin)(struct tep_handle *tep,
                                               const char *path,
                                               const char *name,
                                               void *data),
                           void *data);
int tep_add_plugin_path(struct tep_handle *tep, char *path,
                          enum tep_plugin_load_priority prio);
int tep_plugin_add_option(const char *name, const char *val);

الوصف

تقوم الدالة tep_load_plugins() بتحميل جميع المكونات، الموجودة في مجلدات المكونات. الوسيط tep هو سياق محلل أحداث التتبع. مجلدات المكونات هي:

        - المجلدات، المحددة في tep→plugins_dir ذات الأولوية TEP_PLUGIN_FIRST
        - مجلد المكونات الخاص بالنظام، المحدد عند وقت تجميع المكتبة. يعتمد هذا على بادئة تثبيت المكتبة وعادة ما يكون
          (install_preffix)/lib/traceevent/plugins
        - المجلد، المحدد بواسطة متغير البيئة TRACEEVENT_PLUGIN_DIR
        - مجلد المكونات الخاص بالمستخدم، الموجود في ~/.local/lib/traceevent/plugins
        - المجلدات، المحددة في tep→plugins_dir ذات الأولوية TEP_PLUGIN_LAST

يمكن التحكم في تحميل المكونات بواسطة tep_flags، باستخدام واجهة برمجة التطبيقات tep_set_flag():

        TEP_DISABLE_SYS_PLUGINS       - لا تقم بتحميل المكونات الموجودة في
                                        مجلد مكونات النظام.
        TEP_DISABLE_PLUGINS           - لا تقم بتحميل أي مكونات.

يجب استدعاء واجهة برمجة التطبيقات tep_set_flag() قبل tep_load_plugins()، إذا لم يكن تحميل جميع المكونات هو الحالة المرغوبة.

تقوم الدالة tep_unload_plugins() بإلغاء تحميل المكونات، التي تم تحميلها سابقًا بواسطة tep_load_plugins(). الوسيط tep هو سياق محلل أحداث التتبع. plugin_list هي قائمة المكونات المحملة، التي تم إرجاعها بواسطة الدالة tep_load_plugins().

تقوم الدالة tep_load_plugins_hook() بالمرور عبر جميع المجلدات التي تحتوي على مكونات وتستدعي خطاف load_plugin() المحدد من قبل المستخدم لكل ملف مكون. فقط الملفات ذات الامتداد suffix المحدد تعتبر مكونات. data هو سياق محدد من قبل المستخدم، يتم تمريره إلى load_plugin(). المجلدات وترتيب المرور هي نفسها كما في واجهة برمجة التطبيقات tep_load_plugins().

تضيف الدالة tep_add_plugin_path() مجلدات إضافية تحتوي على مكونات إلى قائمة tep→plugins_dir. يجب استدعاؤها قبل tep_load_plugins() لكي يتم تحميل المكونات من المجلدات الجديدة. الوسيط tep هو سياق محلل أحداث التتبع. path هو المسار الكامل لمجلد المكونات الجديد. يحدد الوسيط prio ترتيب أولوية التحميل للمجلد الجديد للمكونات. أولوية التحميل مهمة في حالة وجود إصدارات مختلفة من نفس المكون الموجودة في مجلدات مكونات متعددة. يفوز المكون الأخير الذي تم تحميله. يمكن أن تكون الأولوية:

        TEP_PLUGIN_FIRST      - تحميل المكونات من هذا المجلد أولاً
        TEP_PLUGIN_LAST       - تحميل المكونات من هذا المجلد أخيرًا

حيث أن المكونات في TEP_PLUGIN_LAST" ستحظى بالأسبقية على المكونات في المجلدات الأخرى.

تقوم الدالة tep_plugin_add_option() بتعيين الخيارات التي يحددها مكون. name هو اسم الخيار الذي سيتم تعيينه للقيمة val. يمكن للمكونات إضافة خيارات لتغيير سلوكها وتقوم tep_plugin_add_option() باستخدامها من قبل التطبيق لإجراء هذه التعديلات.

قيمة الإرجاع

تقوم الدالة tep_load_plugins() بإرجاع قائمة بالمكونات التي تم تحميلها بنجاح، أو NULL في حالة عدم تحميل أي مكونات. الدالة tep_add_plugin_path() ترجع -1 في حالة حدوث خطأ، و 0 بخلاف ذلك.

مثال

#include <event-parse.h>
...
struct tep_handle *tep = tep_alloc();
...
tep_add_plugin_path(tep, "~/dev_plugins", TEP_PLUGIN_LAST);
...
struct tep_plugin_list *plugins = tep_load_plugins(tep);
if (plugins == NULL) {
        /* no plugins are loaded */
}
...
tep_unload_plugins(plugins, tep);
...
void print_plugin(struct tep_handle *tep, const char *path,
                  const char *name, void *data)
{
        pritnf("Found libtraceevent plugin %s/%s\n", path, name);
}
...
tep_load_plugins_hook(tep, ".so", print_plugin, NULL);
...
