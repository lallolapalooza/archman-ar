tep_add_plugin_path(3)

الحزمة: libtraceevent

التراخيص: GPL-2.0-only, LGPL-2.1-only

LIBTRACEEVENT(3) دليل libtraceevent LIBTRACEEVENT(3)

الاسم

tep_load_plugins, tep_unload_plugins, tep_load_plugins_hook, tep_add_plugin_path, tep_plugin_add_option - تحميل / إلغاء تحميل إضافات traceevent.

الملخص

#include <event-parse.h>
struct tep_plugin_list *tep_load_plugins(struct tep_handle *tep);
void tep_unload_plugins(struct tep_plugin_list *plugin_list, struct tep_handle *tep);
void tep_load_plugins_hook(struct tep_handle *tep, const char *suffix,
                           void (*load_plugin)(struct tep_handle *tep,
                                               const char *path,
                                               const char *name,
                                               void *data),
                           void *data);
int tep_add_plugin_path(struct tep_handle *tep, char *path,
                          enum tep_plugin_load_priority prio);
int tep_plugin_add_option(const char *name, const char *val);

الوصف

تقوم الدالة tep_load_plugins() بتحميل جميع الإضافات الموجودة في أدلة الإضافات. الوسيط tep هو سياق محلل أحداث التتبع. أدلة الإضافات هي:

        - الأدلة المحددة في tep→plugins_dir ذات الأولوية TEP_PLUGIN_FIRST
        - دليل الإضافات الخاص بالنظام، المحدد وقت تجميع المكتبة. يعتمد
          على بادئة تثبيت المكتبة وعادة ما يكون
          (install_preffix)/lib/traceevent/plugins
        - الدليل المحدد بواسطة متغير البيئة TRACEEVENT_PLUGIN_DIR
        - دليل الإضافات الخاص بالمستخدم، الموجود في ~/.local/lib/traceevent/plugins
        - الأدلة المحددة في tep→plugins_dir ذات الأولوية TEP_PLUGIN_LAST

يمكن التحكم في تحميل الإضافات بواسطة tep_flags، باستخدام واجهة برمجة التطبيقات tep_set_flag():

        TEP_DISABLE_SYS_PLUGINS       - لا تقم بتحميل الإضافات الموجودة في
                                        دليل الإضافات الخاص بالنظام.
        TEP_DISABLE_PLUGINS           - لا تقم بتحميل أي إضافات.

يجب استدعاء واجهة برمجة التطبيقات tep_set_flag() قبل tep_load_plugins()، إذا لم يكن تحميل جميع الإضافات هو المطلوب.

تقوم الدالة tep_unload_plugins() بإلغاء تحميل الإضافات التي تم تحميلها مسبقًا بواسطة tep_load_plugins(). الوسيط tep هو سياق محلل أحداث التتبع. plugin_list هي قائمة الإضافات المحملة، التي تم إرجاعها بواسطة الدالة tep_load_plugins().

تقوم الدالة tep_load_plugins_hook() بالمرور عبر جميع الأدلة التي تحتوي على إضافات وتستدعي خطاف load_plugin() الذي يحدده المستخدم لكل ملف إضافة. فقط الملفات ذات اللاحقة suffix المحددة تعتبر إضافات. data هو سياق يحدده المستخدم، يتم تمريره إلى load_plugin(). الأدلة وترتيب المرور هي نفسها الموجودة في واجهة برمجة التطبيقات tep_load_plugins().

تضيف الدالة tep_add_plugin_path() أدلة إضافية تحتوي على إضافات إلى القائمة tep→plugins_dir. يجب استدعاؤها قبل tep_load_plugins() حتى يتم تحميل الإضافات من الأدلة الجديدة. الوسيط tep هو سياق محلل أحداث التتبع. path هو المسار الكامل لدليل الإضافات الجديد. يحدد الوسيط prio ترتيب أولوية التحميل لدليل الإضافات الجديد. أولوية التحميل مهمة في حالة وجود إصدارات مختلفة من نفس الإضافة موجودة في أدلة إضافات متعددة. الإضافة الأخيرة المحملة هي التي تفوز. يمكن أن تكون الأولوية:

        TEP_PLUGIN_FIRST      - تحميل الإضافات من هذا الدليل أولاً
        TEP_PLUGIN_LAST       - تحميل الإضافات من هذا الدليل أخيراً

حيث تفوز الإضافات في TEP_PLUGIN_LAST" على الإضافات الموجودة في الأدلة الأخرى.

تضبط الدالة tep_plugin_add_option() الخيارات التي تحددها إضافة. name هو اسم الخيار المراد تعيينه إلى val. يمكن للإضافات إضافة خيارات لتغيير سلوكها وتستخدم tep_plugin_add_option() من قبل التطبيق لإجراء تلك التعديلات.

قيمة الإرجاع

تقوم الدالة tep_load_plugins() بإرجاع قائمة بالإضافات التي تم تحميلها بنجاح، أو NULL في حالة عدم تحميل أي إضافات. تقوم الدالة tep_add_plugin_path() بإرجاع -1 في حالة حدوث خطأ، و 0 بخلاف ذلك.

مثال

#include <event-parse.h>
...
struct tep_handle *tep = tep_alloc();
...
tep_add_plugin_path(tep, "~/dev_plugins", TEP_PLUGIN_LAST);
...
struct tep_plugin_list *plugins = tep_load_plugins(tep);
if (plugins == NULL) {
        /* لا توجد إضافات محملة */
}
...
tep_unload_plugins(plugins, tep);
...
void print_plugin(struct tep_handle *tep, const char *path,
                  const char *name, void *data)
{
        pritnf("تم العثور على إضافة libtraceevent %s/%s\n", path, name);
}
...
tep_load_plugins_hook(tep, ".so", print_plugin, NULL);
...
