Test__MockTime(3pm)

الحزمة: perl-test-mocktime

التراخيص: PerlArtistic, GPL

Test::MockTime(3pm) وثائق Perl من مساهمي المستخدم Test::MockTime(3pm)

الاسم

Test::MockTime - يستبدل الوقت الفعلي بالوقت المحاكى

الإصدار

الإصدار 0.17

الملخص

use Test::MockTime qw( :all );
set_relative_time(-600);
# قم ببعض الاختبارات التي تعتمد على زيادة الوقت من 600 ثانية مضت
set_absolute_time(0);
# قم ببعض الاختبارات الأخرى التي تعتمد على بدء الوقت من الحقبة الزمنية (epoch)
# قد تختلف الحقبة الزمنية حسب المنصة. راجع perlport.
set_fixed_time(CORE::time());
# قم ببعض الاختبارات الأخرى التي تعتمد على بقاء الوقت عند الوقت الفعلي الحالي
set_absolute_time('1970-01-01T00:00:00Z');
# قم ببعض الاختبارات التي تعتمد على بدء الوقت من وقت الحقبة الزمنية ليونكس
set_fixed_time('01/01/1970 00:00:00', '%m/%d/%Y %H:%M:%S');
# قم ببعض الاختبارات التي تعتمد على بقاء الوقت عند وقت الحقبة الزمنية ليونكس
restore_time();
# استئناف الخدمة العادية

الوصف

تم إنشاء هذه الوحدة لتمكين مجموعات الاختبار من اختبار الكود في نقاط زمنية محددة. على وجه التحديد، تقوم باستبدال localtime و gmtime و time في وقت الترجمة ثم تعتمد على المستخدم لتوفير وقت وهمي عبر set_relative_time أو set_absolute_time أو set_fixed_time لتغيير الاستدعاءات المستقبلية لـ gmtime أو time أو localtime.

الإجراءات الفرعية/الطرق

set_absolute_time
إذا تم تمرير وسيطة رقمية واحدة، فإن الوسيطة تمثل وقتاً مطلقاً (على سبيل المثال، إذا تم تمرير 0، فسيكون الوقت المطلق هو الحقبة الزمنية)، ويقوم بحساب الإزاحة للسماح للاستدعاءات اللاحقة لـ time و gmtime و localtime بعكس ذلك.

على سبيل المثال، في الكود التالي

Test::MockTime::set_absolute_time(0);
my ($start) = time;
sleep 2;
my ($end) = time;

يجب أن تحتوي المتغير $end على ثانيتين بعد الحقبة الزمنية؛

إذا تم تمرير وسيطتين، يتم اعتبار الوسيطة الأولى وقتاً مطلقاً بتنسيق سلسلة معين (على سبيل المثال، "01/01/1970 00:00:00"). يتم اعتبار الوسيطة الثانية كسلسلة تنسيق "strptime" (على سبيل المثال، "%m/%d/%Y %H:%M:%S"). إذا تم تمرير وسيطة واحدة، ولكن هذه الوسيطة ليست رقمية، فسيتم افتراض سلسلة تنسيق "strptime" هي "%Y-%m-%dT%H:%M:%SZ".

على سبيل المثال، في الكود التالي

Test::MockTime::set_absolute_time('1970-01-01T00:00:00Z');
my ($start) = time;
sleep 2;
my ($end) = time;

يجب أن تحتوي المتغير $end على ثانيتين بعد الحقبة الزمنية ليونكس؛

set_relative_time($relative)
تأخذ كوسيطة قيمة نسبية من الوقت الحالي (على سبيل المثال، إذا تم تمرير -10، سيتم تحويل الوقت الحالي إلى وقت الآلة الفعلي - 10 ثوانٍ) وتقوم بحساب الإزاحة للسماح للاستدعاءات اللاحقة لـ time و gmtime و localtime بعكس ذلك.

على سبيل المثال، في الكود التالي

my ($start) = time;
Test::MockTime::set_relative_time(-600);
sleep 600;
my ($end) = time;

يجب أن تحتوي المتغير $end على قيم متطابقة أو متشابهة جداً مع المتغير $start.

set_fixed_time
إذا تم تمرير وسيطة رقمية واحدة، فإن الوسيطة تمثل وقتاً مطلقاً (على سبيل المثال، إذا تم تمرير 0، فسيكون الوقت المطلق هو الحقبة الزمنية). ستعيد جميع الاستدعاءات اللاحقة لـ gmtime و localtime و time هذه القيمة.

على سبيل المثال، في الكود التالي

Test::MockTime::set_fixed_time(time)
my ($start) = time;
sleep 3;
my ($end) = time;

ستحتوي المتغير $end والمتغير $start على نفس النتائج

إذا تم تمرير وسيطتين، يتم اعتبار الوسيطة الأولى وقتاً مطلقاً بتنسيق سلسلة معين (على سبيل المثال، "01/01/1970 00:00:00"). يتم اعتبار الوسيطة الثانية كسلسلة تنسيق "strptime" (على سبيل المثال، "%m/%d/%Y %H:%M:%S"). إذا تم تمرير وسيطة واحدة، ولكن هذه الوسيطة ليست رقمية، فسيتم افتراض سلسلة تنسيق "strptime" هي "%Y-%m-%dT%H:%M:%SZ".

restore()
استعادة قيم معالجة الوقت الافتراضية. "restore_time" هو اسم مستعار. عند تصديره بعلامة 'all'، يتم تصدير هذا الإجراء الفرعي باسم "restore_time".

التكوين والبيئة

لا يتطلب Test::MockTime أي ملفات تكوين أو متغيرات بيئة.
