tep_plugin_kvm_get_func(3)

الحزمة: libtraceevent

التراخيص: GPL-2.0-only, LGPL-2.1-only

LIBTRACEEVENT(3) دليل libtraceevent LIBTRACEEVENT(3)

الاسم

tep_plugin_kvm_get_func, tep_plugin_kvm_put_func - إضافة اسم دالة لمؤشر التعليمات الخاص بمكون KVM الإضافي

المُلَخَّص

#include <event-parse.h>
const char *tep_plugin_kvm_get_func(struct tep_event *event,
                                    struct tep_record *record,
                                    unsigned long long *paddr);
void tep_plugin_kvm_put_func(const char *func);

الوصف

لا ينبغي استدعاء الدالتين tep_plugin_kvm_get_func() و tep_plugin_kvm_put_func() بواسطة التطبيق، بل يجب تعريفهما بواسطة التطبيق.

تحتوي أحداث معينة (مثل kvm_exit و kvm_entry) على مؤشر التعليمات حيث تغير السياق من الضيف إلى المضيف. نظرًا لأن المضيف يعرف فقط مؤشر التعليمات ولا يمتلك معلومات حول الدالة التي ينتمي إليها مؤشر التعليمات هذا في الضيف، فإنه يمكنه فقط طباعة العنوان.

لكن قد يمتلك التطبيق المزيد من المعلومات حول الضيف، ويعرف أين كان الضيف عند حدوث الخروج، بل وقد يعرف اسم الدالة لهذا العنوان.

يتم استدعاء مكون KVM الإضافي لـ libtraceevent عند هذه الأحداث، ثم يستدعي tep_plugin_kvm_get_func() لمعرفة ما إذا كانت الدالة يمكنها تحويل عنوان مؤشر التعليمات إلى اسم دالة حقيقي. إذا كان الإرجاع غير NULL، فستقوم بطباعة الدالة في المخرجات لهذا الحدث.

تم تعريف هذه الدوال حاليًا كدوال ضعيفة داخل المكون الإضافي، حتى لا تتطلب تعريفها في مكان آخر. لكي يقوم التطبيق بتجاوز الدالة الضعيفة، سيحتاج إلى تعريف الدالة في ملف يتم تجميعه مع -rdynamic. سيؤدي ذلك إلى إخبار الرابط الديناميكي بفحص ملف الكائن هذا واستخدام أسماء الدوال لحل الدوال الضعيفة في كائنات مشتركة أخرى (في هذه الحالة، كائن مشاركة مكون KVM الإضافي).

إذا قام التطبيق بتعريف tep_plugin_kvm_get_func()، فيجب عليه استخدام النموذج الأولي أعلاه. سيحتوي event على حدث KVM الذي يحتوي على حقل مؤشر التعليمات. سيكون record هو مثيل هذا الحدث. لا تحتاج دالة التطبيق إلى استخدام هذه المعلمات، ولكن قد تكون مفيدة للعثور على اسم الدالة للعنوان. paddr هو مؤشر إلى قيمة 64 بت (حيث قد يتم استخدام 32 بت فقط في الأجهزة 32 بت). هذه القيمة هي مؤشر التعليمات للبحث عنه. إذا عرف التطبيق عنوان بدء الدالة أيضًا، يمكنه تعيين paddr إلى هذا العنوان، وسيضيف مكون KVM الإضافي أيضًا " +offset" إلى اسم الدالة حيث يكون الإزاحة هي القيمة الأصلية في paddr مطروحًا منها القيمة في paddr عند استدعائها. أخيرًا، يجب على التطبيق إرجاع اسم الدالة كسلسلة نصية منتهية بـ nul إذا تم العثور عليها.

إذا تم تخصيص السلسلة النصية التي تم إرجاعها بواسطة tep_plugin_kvm_get_func()، فسوف يقوم مكون KVM الإضافي باستدعاء tep_plugin_kvm_put_func() عندما ينتهي منها، ويمرر القيمة التي تم إرجاعها بواسطة tep_plugin_kvm_get_func() كـ func. يتيح هذا للتطبيق تحريرها إذا لزم الأمر.

قيمة الإرجاع

لا ينبغي استدعاء tep_plugin_kvm_get_func() بواسطة التطبيق، بل يجب تعريفه بواسطة التطبيق. يجب أن ترجع سلسلة نصية منتهية بـ nul تمثل الدالة لمؤشر التعليمات المحدد الذي تم تمريره إليها بالمرجع في paddr. يمكنها بعد ذلك تحديث paddr اختياريًا إلى قيمة تحتوي على بداية الدالة. يمكن تحرير السلسلة النصية التي تم إرجاعها بواسطة tep_plugin_kvm_put_func() والتي يجب على التطبيق تعريفها لتنظيف السلسلة النصية.

المثال أدناه يحتاج إلى تجميعه باستخدام العلامة -rdynamic حتى يتمكن الرابط الديناميكي من حل دالات tep_plugin_kvm_get_func() و tep_plugin_kvm_put_func().

عند تشغيله مقابل ملف trace.dat تم إنتاجه بواسطة trace-cmd(1) يسجل أحداث kvm_exit و kvm_entry على ضيف، ثم تم تمرير ملف /proc/kallsyms الخاص بالضيف كمعلمة ثانية، فإن المخرجات المنتجة ستبدو شيئًا مثل:

CPU 0/KVM-20407 83156.177626 [000] kvm_exit     reason APIC_ACCESS rip 0xffffffffb0056ee2 exit native_apic_mem_write+0x2 info 10b0 0
CPU 0/KVM-20407 83156.177632 [000] kvm_entry     vcpu 0 rip 0xffffffffb0056ee8 enter native_apic_mem_write+0x8

ولكن بدون تلك الاستدعاءات، ستبدو هكذا:

CPU 0/KVM-20407 83156.177626 [000] kvm_exit     reason APIC_ACCESS rip 0xffffffffb0056ee2 info 10b0 0
CPU 0/KVM-20407 83156.177632 [000] kvm_entry     vcpu 0 rip 0xffffffffb0056ee8

مثال

#include <stdio.h>
#include <stdlib.h>
#include <event-parse.h>
#include <trace-cmd.h>
#include <sys/stat.h>
static struct tep_handle *tep;
const char *tep_plugin_kvm_get_func(struct tep_event *event, struct tep_record *record,
                                    unsigned long long *paddr)
{
        const char *func;
        char *event_func;
        char *ename;
        func = tep_find_function(tep, *paddr);
        if (!func)
                return NULL;
        if (strcmp(event->name, "kvm_exit") == 0)
                ename = "exit";
        else
                ename = "enter";
        /*
         * Normally, passing back func directly is sufficient and then
         * tep_plugin_kvm_put_func() would not be required. But this example
         * is showing how to handle allocation of the returned string.
         */
        event_func = malloc(strlen(ename) + strlen(func) + 2);
        if (!event_func)
                return NULL;
        sprintf(event_func, "%s %s", ename, func);
        *paddr = tep_find_function_address(tep, *paddr);
        return event_func;
}
void tep_plugin_kvm_put_func(const char *func)
{
        char *f = (char *)func;
        free(f);
}
static int show_event(struct tracecmd_input *handle, struct tep_event *event,
                      struct tep_record *record, int cpu, void *data)
{
        static struct trace_seq seq;
        tep = data;
        if (!seq.buffer)
                trace_seq_init(&seq);
        trace_seq_reset(&seq);
        tep_print_event(tracecmd_get_tep(handle), &seq, record,
                        "%s-%d\t%6.1000d [%03d] %s\t%s\n",
                        TEP_PRINT_COMM, TEP_PRINT_PID,
                        TEP_PRINT_TIME, TEP_PRINT_CPU,
                        TEP_PRINT_NAME, TEP_PRINT_INFO);
        trace_seq_terminate(&seq);
        trace_seq_do_printf(&seq);
        return 0;
}
int main(int argc, char **argv)
{
        struct tracecmd_input *handle;
        struct tep_handle *guest_tep;
        struct stat st;
        FILE *fp;
        char *buf;
        if (argc < 3) {
                printf("usage: trace.dat guest_kallsyms_file\n");
                exit(-1);
        }
        handle = tracecmd_open(argv[1], 0);
        if (!handle) {
                perror(argv[1]);
                exit(-1);
        }
        /* Just for kallsyms parsing */
        guest_tep = tep_alloc();
        if (!guest_tep)
                exit(-1);
        if (stat(argv[2], &st) < 0) {
                perror(argv[2]);
                exit(-1);
        }
        buf = malloc(st.st_size + 1);
        if (!buf)
                exit(-1);
        fp = fopen(argv[2], "r");
        if (!fp) {
                perror(argv[2]);
                exit(-1);
        }
        if (fread(buf, st.st_size, 1, fp) < 0) {
                perror(argv[2]);
                exit(-1);
        }
        buf[st.st_size] = '\0';
        if (tep_parse_kallsyms(guest_tep, buf) < 0) {
                printf("Failed to parse %s\n", argv[2]);
                exit(-1);
        }
        free(buf);
        tracecmd_follow_event(handle, "kvm", "kvm_exit", show_event, guest_tep);
        tracecmd_follow_event(handle, "kvm", "kvm_entry", show_event, guest_tep);
        tracecmd_iterate_events(handle, NULL, 0, NULL, NULL);
        tep_free(guest_tep);
        tracecmd_close(handle);
}
