tep_is_pid_registered

tep_is_pid_registered — check if a process ID is registered

SYNOPSIS

#include <tep.h>

bool
tep_is_pid_registered(tep_handle_t *handle, pid_t pid);

DESCRIPTION

The tep_is_pid_registered() function checks if a process ID (pid) is currently registered within the trace session managed by handle.

ARGUMENTS

handle
A pointer to a tep_handle_t structure that represents the trace session.
pid
The process ID to check for registration.

RETURN VALUE

Returns true if the process ID is registered, and false otherwise.

EXAMPLE

#include <stdio.h>
#include <stdlib.h>
#include <unistd.h>
#include <tep.h>

int main() {
    tep_handle_t *handle = NULL;
    pid_t target_pid = getpid(); // Get the current process ID

    // Initialize the TEP handle (replace with your trace file or setup)
    handle = tep_open("trace.dat"); // Example: opens a trace file
    if (!handle) {
        perror("tep_open");
        return 1;
    }

    // Check if the target PID is registered
    if (tep_is_pid_registered(handle, target_pid)) {
        printf("Process ID %d is registered.\n", target_pid);
    } else {
        printf("Process ID %d is not registered.\n", target_pid);
    }

    // Close the TEP handle
    tep_close(handle);

    return 0;
}
