Test::utf8(3) وثائق Perl المساهمة من المستخدم Test::utf8(3)

الاسم

Test::utf8 - اختبارات utf8 مفيدة

الملخص

# التحقق من أن السلسلة النصية جيدة
is_valid_string($string);   # التحقق من أن السلسلة النصية صالحة
is_sane_utf8($string);      # التحقق من عدم الترميز المزدوج
# التحقق من أن السلسلة النصية لها سمات معينة
is_flagged_utf8($string1);   # تم تعيين علامة utf8
is_within_ascii($string2);   # تحتوي فقط على أحرف ASCII
isnt_within_ascii($string3); # تحتوي على أحرف خارج نطاق ASCII
is_within_latin_1($string4); # تحتوي فقط على أحرف latin-1
isnt_within_ascii($string5); # تحتوي على أحرف خارج نطاق latin-1

الوصف

توفر هذه الوحدة مجموعة من الاختبارات المفيدة للتعامل مع سلاسل utf8 النصية في Perl.

تتضمن هذه الوحدة نوعين من الاختبارات: اختبارات الصلاحية التي تتحقق مما إذا كانت السلسلة النصية صالحة وغير تالفة، بينما تختبر الاختبارات المميزة ما إذا كانت السلسلة النصية لها مجموعة معينة من الخصائص.

اختبارات الصلاحية

is_valid_string($string, $testname)
تتحقق مما إذا كانت السلسلة النصية "صالحة"، أي أنها تمر وتعطي قيمة صحيحة ما لم يتم تعيين علامة utf8 الداخلية على مقياس غير مكون من تسلسل بايت utf-8 صالح.

لا ينبغي أن يحدث هذا أبدًا، ومن الناحية النظرية، يجب أن ينجح هذا الاختبار دائمًا. ما لم تقم أنت (أو وحدة تستخدمها) بالتلاعب داخل مقياس باستخدام وظائف Encode الخاصة أو كود XS، فلا يجب أن تجد نفسك في وضع تكون فيه لديك مقياس تالف. ولكن إذا حدث ذلك، وكان كذلك، فيجب أن يساعدك هذا الدالة في اكتشاف المشكلة.

لتوضيح الأمر، إليك مثال لحالة الخطأ التي يمكن لهذا الاختبار اكتشافها:

my $mark = "Mark";
my $leon = "L\x{e9}on";
is_valid_string($mark);  # ينجح، ليس utf-8
is_valid_string($leon);  # ينجح، ليس utf-8
my $iloveny = "I \x{2665} NY";
is_valid_string($iloveny);      # ينجح، utf-8 صحيح
my $acme = "L\x{c3}\x{a9}on";
Encode::_utf8_on($acme);      # (من فضلك لا تفعل أشياء كهذه)
is_valid_string($acme);       # ينجح، تسلسل بايت utf-8 صحيح تمت ترقيته
Encode::_utf8_on($leon);      # (هذا هو سبب عدم فعل أشياء كهذه)
is_valid_string($leon);       # يفشل! البايت \x{e9} ليس utf-8 صالحًا
is_sane_utf8($string, $name)
يفشل هذا الاختبار إذا احتوت السلسلة النصية على شيء يبدو أنه utf8 مشكوك فيه، أي أنه يحتوي على شيء يبدو كتسلسل متعدد البايت لحرف latin-1 ولكن لم يتم توجيه Perl لمعاملته على هذا النحو. السلاسل النصية التي ليست utf8 تفشل دائمًا تلقائيًا.

قد تساعد بعض الأمثلة:

# سينجح هذا لأنه سلسلة نصية latin-1 عادية
is_sane_utf8("Hello L\x{e9}eon");
# سيفشل هذا لأن \x{c3}\x{a9} يبدو كتسلسل بايت utf8 لحرف e-acute
my $string = "Hello L\x{c3}\x{a9}on";
is_sane_utf8($string);
# سينجح هذا لأن utf8 تم تفسيره بشكل صحيح على أنه utf8
Encode::_utf8_on($string)
is_sane_utf8($string);

من الواضح أن هذا ليس موثوقًا بنسبة مائة بالمائة. الحالة الهامشية التي سيفشل فيها هذا هي حيث يكون لديك "\x{c2}" (وهو "LATIN CAPITAL LETTER WITH CIRCUMFLEX") أو "\x{c3}" (وهو "LATIN CAPITAL LETTER WITH TILDE") متبوعًا بأحد رموز علامات الترقيم latin-1.

# حرف كبير A مع علامة مدّ محاط بعلامات اقتباس ذكية
# سيفشل هذا لأنه سيرى "\x{c2}\x{94}" ويعتقد
# أنه في الواقع تسلسل utf8 لعلامة الاقتباس النهائية
is_sane_utf8("\x{93}\x{c2}\x{94}");

ومع ذلك، نظرًا لأن هذا نادرًا ما يحدث، فإن هذا الاختبار موثوق به بشكل معقول في معظم الحالات. لا يزال، يجب توخي الحذر في الحالات التي يتم فيها وضع البيانات الديناميكية بجوار علامات ترقيم latin-1 لتجنب النتائج السلبية الخاطئة.

هناك حالتان لتسببا في فشل هذا الاختبار؛ تحتوي السلسلة النصية على تسلسلات بايت utf8 ولم يتم تمييز السلسلة النصية كـ utf8 (يعني هذا عادةً أنك حصلت عليها من مصدر خارجي مثل مكتبة C؛ عندما يحتاج Perl إلى تخزين سلسلة نصية داخليًا كـ utf8، فإنه يقوم بتشفيرها وتمييزها بشفافية) أو تحتوي سلسلة نصية مميزة كـ utf8 على تسلسلات بايت تبدو عند ترجمتها إلى أحرف كتسلسل بايت utf8. تشخيصات الاختبار تخبرك بأي حالة.

اختبارات خصائص السلسلة النصية

تسمح لك هذه الروتينات بالتحقق من نطاق الأحرف في سلسلة نصية. لاحظ أن هذه الروتينات لا تهتم بالترميز الفعلي الذي يستخدمه Perl داخليًا لتخزين الأحرف، بل تتحقق فقط مما إذا كانت السلسلة النصية تحتوي فقط على أحرف يمكن تمثيلها في الترميز المسمى:

is_within_ascii
يختبر أن السلسلة النصية تحتوي فقط على أحرف موجودة في مجموعة أحرف ASCII.
is_within_latin_1
يختبر أن السلسلة النصية تحتوي فقط على أحرف موجودة في latin-1.

ببساطة تحقق مما إذا كان مقياس معين مميزًا كـ utf8 أم لا بواسطة داخليات Perl:

is_flagged_utf8($string, $name)
ينجح إذا تم تمييز السلسلة النصية بواسطة داخليات Perl كـ utf8، يفشل إذا لم تكن كذلك.
isnt_flagged_utf8($string,$name)
عكس "is_flagged_utf8"، ينجح فقط إذا لم يتم تمييز السلسلة النصية كـ utf8 بواسطة داخليات Perl.

ملاحظة: يمكنك الإشارة إلى هذه الدالة باسم "isn't_flagged_utf8" إذا كنت ترغب حقًا في ذلك.
