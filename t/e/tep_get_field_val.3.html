tep_get_field_val(3)

الحزمة: libtraceevent

التراخيص: GPL-2.0-only, LGPL-2.1-only

LIBTRACEEVENT(3) دليل libtraceevent LIBTRACEEVENT(3)

الاسم

tep_get_any_field_val, tep_get_common_field_val, tep_get_field_val, tep_get_field_raw - الحصول على قيمة حقل.

الملخص

#include <event-parse.h>
#include <trace-seq.h>
int tep_get_any_field_val(struct trace_seq *s, struct tep_event *event, const char *name, struct tep_record *record, unsigned long long *val, int err);
int tep_get_common_field_val(struct trace_seq *s, struct tep_event *event, const char *name, struct tep_record *record, unsigned long long *val, int err);
int tep_get_field_val(struct trace_seq *s, struct tep_event *event, const char *name, struct tep_record *record, unsigned long long *val, int err);
void *tep_get_field_raw(struct trace_seq *s, struct tep_event *event, const char *name, struct tep_record *record, int *len, int err);

الوصف

يمكن استخدام هذه الدوال للبحث عن حقل واسترداد قيمته.

تبحث الدالة tep_get_any_field_val() في record عن حقل بالاسم name، وهو جزء من event. إذا تم العثور على الحقل، يتم تخزين قيمته في val. إذا حدث خطأ ولم يكن err صفرًا، فسيتم كتابة سلسلة خطأ في s.

تقوم الدالة tep_get_common_field_val() بنفس عمل tep_get_any_field_val()، ولكنها تبحث فقط في الحقول المشتركة. يعمل هذا مع أي حدث لأن جميع الأحداث تتضمن الحقول المشتركة.

تقوم الدالة tep_get_field_val() بنفس عمل tep_get_any_field_val()، ولكنها تبحث فقط في الحقول الخاصة بالحدث.

تبحث الدالة tep_get_field_raw() في record عن حقل بالاسم name، وهو جزء من event. إذا تم العثور على الحقل، يتم إرجاع مؤشر إلى مكان وجود الحقل في بيانات record الأولية. يتم تخزين حجم البيانات في len. إذا حدث خطأ ولم يكن err صفرًا، فسيتم كتابة سلسلة خطأ في s.

قيمة الإرجاع

تُرجع الدوال tep_get_any_field_val() و tep_get_common_field_val() و tep_get_field_val() القيمة 0 عند النجاح، أو -1 في حالة حدوث خطأ.

تُرجع الدالة tep_get_field_raw() مؤشرًا إلى البيانات الأولية للحقل، وتضع طول هذه البيانات في len. في حالة حدوث خطأ، يتم إرجاع NULL.

مثال

#include <event-parse.h>
#include <trace-seq.h>
...
struct tep_handle *tep = tep_alloc();
...
struct tep_event *event = tep_find_event_by_name(tep, "kvm", "kvm_exit");
...
void process_record(struct tep_record *record)
{
        int len;
        char *comm;
        struct tep_event *event;
        unsigned long long val;
        event = tep_find_event_by_record(tep, record);
        if (event != NULL) {
                if (tep_get_common_field_val(NULL, event, "common_type",
                                             record, &val, 0) == 0) {
                        /* Got the value of common type field */
                }
                if (tep_get_field_val(NULL, event, "pid", record, &val, 0) == 0) {
                        /* Got the value of pid specific field */
                }
                comm = tep_get_field_raw(NULL, event, "comm", record, &len, 0);
                if (comm != NULL) {
                        /* Got a pointer to the comm event specific field */
                }
        }
}
