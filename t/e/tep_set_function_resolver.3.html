tep_set_function_resolver(3)

الحزمة: libtraceevent

التراخيص: GPL-2.0-only, LGPL-2.1-only

LIBTRACEEVENT(3) libtraceevent Manual LIBTRACEEVENT(3)

الاسم

tep_set_function_resolver, tep_reset_function_resolver, tep_register_function, tep_register_print_string, tep_get_function_count - واجهات برمجة تطبيقات tep المتعلقة بالدوال

الملخص

#include <event-parse.h>
typedef char *(tep_func_resolver_t)(void *priv, unsigned long long *addrp, char **modp);
int tep_set_function_resolver(struct tep_handle *tep, tep_func_resolver_t *func, void *priv);
void tep_reset_function_resolver(struct tep_handle *tep);
int tep_register_function(struct tep_handle *tep, char *name, unsigned long long addr, char *mod);
int tep_register_print_string(struct tep_handle *tep, const char *fmt, unsigned long long addr);
int tep_get_function_count(struct tep_handle *tep)

الوصف

قد تمتلك بعض الأدوات بالفعل طريقة لحل دوال النواة. تسمح لها واجهات برمجة التطبيقات هذه بالاستمرار في استخدامها بدلاً من تكرار جميع الإدخالات بداخلها.

النوع tep_func_resolver_t هو النموذج الأولي لحل دوال النواة البديل. تستقبل هذه الدالة مؤشرًا إلى سياقها المخصص (تم تعيينه باستخدام استدعاء tep_set_function_resolver()) وعنوان دالة نواة، والتي يجب حلها. في حالة النجاح، يجب أن تعيد اسم الدالة ووحدتها (إن وجدت) في modp.

تسجل الدالة tep_set_function_resolver() func كمحلل دوال نواة بديل. الوسيط tep هو سياق محلل أحداث التتبع. الوسيط priv هو سياق مخصص للدالة func. يتم استخدام محلل الدوال بواسطة واجهات برمجة التطبيقات tep_find_function(3) و tep_find_function_address(3) و tep_print_func_field() لحل عنوان دالة إلى اسم دالة.

تعيد الدالة tep_reset_function_resolver() محلل دوال النواة إلى الدالة الافتراضية. الوسيط tep هو سياق محلل أحداث التتبع.

يمكن استخدام واجهات برمجة التطبيقات هذه للعثور على اسم الدالة وعنوان بدايتها، بواسطة عنوان معين. لا يلزم أن يكون العنوان المعطى دقيقًا، بل سيحدد الدالة التي تحتوي عليه.

تسجل الدالة tep_register_function() اسم دالة يتم تعيينه إلى عنوان و (اختياري) وحدة. يتم استخدام هذا التعيين في حالة أن متعقب الدوال أو الأحداث يحتوي على معلمة "%pS" في سلسلة التنسيق الخاصة به. من الشائع تمرير أسماء دوال kallsyms مع عناوينها المقابلة مع هذه الدالة. الوسيط tep هو سياق محلل أحداث التتبع. name هو اسم الدالة، ويتم نسخ السلسلة داخليًا. addr هو عنوان بداية الدالة. mod هي وحدة النواة التي قد تكون الدالة فيها (NULL إن لم تكن).

تسجل الدالة tep_register_print_string() سلسلة نصية بواسطة العنوان الذي تم تخزينها فيه في النواة. يتم تمرير بعض السلاسل النصية الداخلية للنواة ذات العنوان الثابت إلى أحداث معينة. "%s" في حقل تنسيق الحدث الذي يحتوي على عنوان يحتاج إلى معرفة ما هي السلسلة النصية الموجودة في ذلك العنوان. توفر tep_register_print_string() التحليل مع تعيين بين عناوين النواة وتلك السلاسل النصية. الوسيط tep هو سياق محلل أحداث التتبع. fmt هي السلسلة النصية التي سيتم تسجيلها، ويتم نسخها داخليًا. addr هو العنوان الذي كانت السلسلة النصية موجودة فيه.

تعيد الدالة tep_get_function_count() عدد الدوال المسجلة في معالج tep.

قيمة الإرجاع

تعيد الدالة tep_set_function_resolver() القيمة 0 في حالة النجاح، أو -1 في حالة حدوث خطأ.

تعيد الدالة tep_register_function() القيمة 0 في حالة النجاح. في حالة حدوث خطأ، يتم إرجاع -1، ويتم تعيين errno إلى رقم الخطأ المناسب.

تعيد الدالة tep_register_print_string() القيمة 0 في حالة النجاح. في حالة حدوث خطأ، يتم إرجاع -1، ويتم تعيين errno إلى رقم الخطأ المناسب.

مثال

#include <event-parse.h>
...
struct tep_handle *tep = tep_alloc();
...
char *my_resolve_kernel_addr(void *context,
                             unsigned long long *addrp, char **modp)
{
        struct db *function_database = context;
        struct symbol *sym = sql_lookup(function_database, *addrp);
        if (!sym)
                return NULL;
        *modp = sym->module_name;
        return sym->name;
}
void show_function( unsigned long long addr)
{
        unsigned long long fstart;
        const char *fname;
        if (tep_set_function_resolver(tep, my_resolve_kernel_addr,
                                      function_database) != 0) {
                /* فشل تسجيل my_resolve_kernel_addr */
        }
        /* تستخدم واجهات برمجة التطبيقات هذه my_resolve_kernel_addr() لحل العنوان */
        fname = tep_find_function(tep, addr);
        fstart = tep_find_function_address(tep, addr);
        /*
           العنوان addr موجود في الدالة المسماة fname، وتبدأ من العنوان fstart،
           في الإزاحة (addr - fstart)
        */
        tep_reset_function_resolver(tep);
}

        if (tep_register_function(tep, "kvm_exit",
                                (unsigned long long) 0x12345678, "kvm") != 0) {
                /* فشل تسجيل تعيين عنوان kvm_exit */
        }
...
        if (tep_register_print_string(tep, "print string",
                                (unsigned long long) 0x87654321, NULL) != 0) {
                /* فشل تسجيل تعيين عنوان "print string" */
        }
