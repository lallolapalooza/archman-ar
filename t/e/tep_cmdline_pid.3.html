tep_cmdline_pid(3)

الحزمة: libtraceevent

التراخيص: GPL-2.0-only, LGPL-2.1-only

LIBTRACEEVENT(3) دليل libtraceevent LIBTRACEEVENT(3)

الاسم

tep_register_comm, tep_override_comm, tep_is_pid_registered, tep_data_comm_from_pid, tep_data_pid_from_comm, tep_cmdline_pid - إدارة تعيينات معرف العملية (pid) إلى اسم العملية.

الملخص

#include <event-parse.h>
int tep_register_comm(struct tep_handle *tep, const char *comm, int pid);
int tep_override_comm(struct tep_handle *tep, const char *comm, int pid);
bool tep_is_pid_registered(struct tep_handle *tep, int pid);
const char *tep_data_comm_from_pid(struct tep_handle *pevent, int pid);
struct cmdline *tep_data_pid_from_comm(struct tep_handle *pevent, const char *comm, struct cmdline *next);
int tep_cmdline_pid(struct tep_handle *pevent, struct cmdline *cmdline);

الوصف

يمكن استخدام هذه الدوال للتعامل مع تعيين معرف العملية (pid) إلى اسم العملية. تبني المكتبة ذاكرة تخزين مؤقت لهذه التعيينات، والتي تستخدم لعرض اسم العملية، بدلاً من معرفها. يمكن استرداد هذه المعلومات من ملف tracefs/saved_cmdlines.

تقوم الدالة tep_register_comm() بتسجيل تعيين pid / اسم العملية. إذا كانت هناك عملية بنفس pid مسجلة بالفعل، يتم إرجاع خطأ. الوسيط pid هو معرف العملية، والوسيط comm هو اسم العملية، و tep هو سياق الحدث. يتم نسخ comm داخليًا.

تقوم الدالة tep_override_comm() بتسجيل تعيين pid / اسم العملية. إذا كانت هناك عملية بنفس معرف العملية مسجلة بالفعل، يتم تحديث سلسلة اسم العملية بالجديدة. الوسيط pid هو معرف العملية، والوسيط comm هو اسم العملية، و tep هو سياق الحدث. يتم نسخ comm داخليًا.

تقوم الدالة tep_is_pid_registered() بالتحقق مما إذا كان معرف العملية (pid) لديه تعيين اسم عملية مسجل. الوسيط pid هو معرف العملية، و tep هو سياق الحدث.

تقوم الدالة tep_data_comm_from_pid() بإرجاع اسم العملية لمعرف عملية (pid) معين. الوسيط pid هو معرف العملية، و tep هو سياق الحدث. لا يجب تحرير السلسلة المرجعة، ولكن سيتم تحريرها عند إغلاق معالج tep.

تقوم الدالة tep_data_pid_from_comm() بإرجاع معرف عملية (pid) لاسم عملية معين. الوسيط comm هو اسم العملية، و tep هو سياق الحدث. الوسيط next هو بنية cmdline للبحث عن معرف العملية التالي. نظرًا لأنه قد يكون هناك أكثر من معرف عملية واحد لعملية معينة، يمكن تمرير نتيجة هذه المكالمة مرة أخرى في مكالمة متكررة في المعلمة next، للبحث عن معرف العملية التالي. إذا كان next هو NULL، فسوف ترجع معرف العملية الأول المرتبط بـ comm. تقوم الدالة بإجراء بحث خطي، لذلك قد تكون بطيئة.

تقوم الدالة tep_cmdline_pid() بإرجاع معرف العملية (pid) المرتبط بـ cmdline معين. الوسيط tep هو سياق الحدث.

قيمة الإرجاع

ترجع الدالة tep_register_comm() القيمة 0 عند النجاح. في حالة حدوث خطأ، يتم إرجاع -1 ويتم تعيين errno للإشارة إلى سبب المشكلة: ENOMEM، إذا لم تكن هناك ذاكرة كافية لنسخ comm أو EEXIST إذا كان تعيين لـ pid هذا مسجلًا بالفعل.

ترجع الدالة tep_override_comm() القيمة 0 عند النجاح. في حالة حدوث خطأ، يتم إرجاع -1 ويتم تعيين errno للإشارة إلى سبب المشكلة: ENOMEM، إذا لم تكن هناك ذاكرة كافية لنسخ comm.

ترجع الدالة tep_is_pid_registered() القيمة true إذا كان pid لديه اسم عملية معين له، والقيمة false بخلاف ذلك.

ترجع الدالة tep_data_comm_from_pid() اسم العملية كسلسلة نصية، أو السلسلة النصية "<...>" إذا لم يكن هناك تعيين لـ pid المحدد.

ترجع الدالة tep_data_pid_from_comm() مؤشرًا إلى بنية cmdline، والتي تحمل معرف عملية (pid) لعملية معينة، أو NULL إذا لم يتم العثور على شيء. يمكن تمرير هذه النتيجة مرة أخرى في مكالمة متكررة كالمعلمة next للدالة.

ترجع الدالة tep_cmdline_pid() معرف العملية (pid) لـ cmdline المحدد. إذا كان cmdline هو NULL، فسيتم إرجاع -1.

مثال

يسجل المثال التالي معرف العملية (pid) للأمر "ls"، في سياق الحدث tep ويجري عمليات بحث مختلفة لتعيينات معرف العملية (pid) / اسم العملية:

#include <event-parse.h>
...
int ret;
int ls_pid = 1021;
struct tep_handle *tep = tep_alloc();
...
        ret = tep_register_comm(tep, "ls", ls_pid);
        if (ret != 0 && errno == EEXIST)
                ret = tep_override_comm(tep, "ls", ls_pid);
        if (ret != 0) {
                /* فشل تسجيل تعيين معرف العملية (pid) / الأمر */
        }
...
        if (tep_is_pid_registered(tep, ls_pid) == 0) {
                /* تعيين الأمر لـ ls_pid غير مسجل */
        }
...
        const char *comm = tep_data_comm_from_pid(tep, ls_pid);
        if (comm) {
                /* تم العثور على اسم العملية لـ ls_pid */
        }
...
        int pid;
        struct cmdline *cmd = tep_data_pid_from_comm(tep, "ls", NULL);
        while (cmd) {
                pid = tep_cmdline_pid(tep, cmd);
                /* تم العثور على معرف العملية (pid) للأمر "ls" */
                cmd = tep_data_pid_from_comm(tep, "ls", cmd);
        }
