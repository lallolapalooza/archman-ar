tep_clear_flag(3)

الحزمة: libtraceevent

التراخيص: GPL-2.0-only, LGPL-2.1-only

LIBTRACEEVENT(3) دليل libtraceevent LIBTRACEEVENT(3)

الاسم

tep_set_flag, tep_clear_flag, tep_test_flag - إدارة الأعلام لسياق محلل أحداث التتبع.

ملخص

#include <event-parse.h>
enum tep_flag {
        TEP_NSEC_OUTPUT,
        TEP_DISABLE_SYS_PLUGINS,
        TEP_DISABLE_PLUGINS
};
void tep_set_flag(struct tep_handle *tep, enum tep_flag flag);
void tep_clear_flag(struct tep_handle *tep, enum tep_flag flag);
bool tep_test_flag(struct tep_handle *tep, enum tep_flag flag);

الوصف

تُعرّف أعلام سياق محلل أحداث التتبع في enum tep_flag:

TEP_NSEC_OUTPUT - طباعة الطابع الزمني للحدث بالنانو ثانية، بدلاً من الميكرو ثانية.
TEP_DISABLE_SYS_PLUGINS - تعطيل الإضافات الموجودة في دليل الإضافات الخاص بالنظام.
                        يتم تعريف هذا الدليل وقت بناء المكتبة، وعادة ما يعتمد على
                        بادئة تثبيت المكتبة: (install_preffix)/lib/traceevent/plugins
TEP_DISABLE_PLUGINS - تعطيل جميع إضافات المكتبة:
                        - في دليل الإضافات الخاص بالنظام
                        - في الدليل المحدد بواسطة متغير البيئة TRACEEVENT_PLUGIN_DIR
                        - في دليل المستخدم الرئيسي، ~/.traceevent/plugins

ملاحظة: يجب تعيين الأعلام المتعلقة بالإضافات قبل استدعاء واجهة برمجة التطبيقات tep_load_plugins().

تقوم الدالة tep_set_flag() بتعيين flag لسياق tep.

تقوم الدالة tep_clear_flag() بإلغاء تعيين flag من سياق tep.

تقوم الدالة tep_test_flag() بالتحقق مما إذا كان flag معينًا لسياق tep.

قيمة الإرجاع

تقوم الدالة tep_test_flag() بإرجاع true إذا كان flag معينًا، وإلا فإنها ترجع false.

مثال

#include <event-parse.h>
...
struct tep_handle *tep = tep_alloc();
...
/* طباعة الطوابع الزمنية بالنانو ثانية */
tep_set_flag(tep,  TEP_NSEC_OUTPUT);
...
if (tep_test_flag(tep, TEP_NSEC_OUTPUT)) {
        /* طباعة الطوابع الزمنية بالنانو ثانية */
} else {
        /* طباعة الطوابع الزمنية بالميكرو ثانية */
}
...
/* طباعة الطوابع الزمنية بالميكرو ثانية */
tep_clear_flag(tep, TEP_NSEC_OUTPUT);
...
