الحزمة: perl-template-toolkit

التراخيص: PerlArtistic

Template::Test(3) وثائق مساهمة Perl Template::Test(3)

NAME

Template::Test - وحدة لأتمتة نصوص اختبار TT2

SYNOPSIS

use Template::Test;
$Template::Test::DEBUG = 0;   # قم بتعيين هذا إلى 1 لرؤية كل اختبار أثناء تشغيله
$Template::Test::EXTRA = 2;   # اختباريْن إضافيين يتبعان test_expect()...
# يمكن استدعاء ok() أي عدد من المرات قبل test_expect
ok( $true_or_false )
# يقسم test_expect() المدخل إلى اختبارات متعددة، ويعالج كل منها باستخدام TT2 ويقارن المخرج المُنشأ مع المخرج المتوقع
test_expect($input, $template, \%replace );
# $input هو نص أو مقبض ملف (مثال: قسم DATA بعد __END__)
test_expect( $text );
test_expect( \*DATA );
# $template هو كائن Template أو تجزئة إعدادات
my $template_cfg = { ... };
test_expect( $input, $template_cfg );
my $template_obj = Template->new($template_cfg);
test_expect( $input, $template_obj );
# $replace هو مرجع تجزئة لمتغيرات القالب
my $replace = {
    a => 'alpha',
    b => 'bravo'
};
test_expect( $input, $template, $replace );
# يجب الإعلان عن ok() الذي يتم استدعاؤه بعد test_expect في $EXTRA (2)
ok( $true_or_false )
ok( $true_or_false )

DESCRIPTION

تعرّف الوحدة Template::Test الدالة الفرعية test_expect() والدوال الفرعية الأخرى ذات الصلة التي يمكن استخدامها لأتمتة نصوص الاختبار الخاصة بـ Template Toolkit. انظر الأمثلة على الاستخدام في مجلد "t" الخاص بالتوزيع.

SUBRUTINAS DE PAQUETE

text_expect()

تقوم الدالة الفرعية test_expect() بتقسيم المستند المدخل إلى اختبارات منفصلة متعددة، وتعالج كل منها باستخدام Template Toolkit، ثم تقارن المخرج المُنشأ بالمخرج المتوقع، والذي تم تحديده أيضًا في المستند المدخل. تقوم بإنشاء مخرج "ok" / "not ok" المألوف المتوافق مع Test::Harness.

يجب تحديد مدخل الاختبار كنص أو كمرجع لمقبض ملف (مثال: GLOB أو IO::Handle) يمكن القراءة منه. على وجه الخصوص، يسمح هذا لمدخل الاختبار بأن يكون بعد العلامة __END__ وقراءته باستخدام المقبض DATA.

use Template::Test;
test_expect(\*DATA);
__END__
# هذا هو الاختبار الأول (هذا تعليق)
-- test --
blah blah blah [% foo %]
-- expect --
blah blah blah value_of_foo
# هذا هو الاختبار الثاني (لا مفاجآت، هذا هو الحال)
-- test --
more blah blah [% bar %]
-- expect --
more blah blah value_of_bar

يتم تجاهل الأسطر بين الاختبارات بشكل عام. أي سطر يبدأ بـ # يُعامل كتعليق ويتم تجاهله.

المعاملان الثاني والثالث لـ test_expect() اختياريان. يمكن أن يكون الثاني مرجعًا لكائن Template يُستخدم لمعالجة أجزاء القالب، أو مرجعًا لتجزئة قيم الإعدادات التي تُستخدم لإنشاء كائن Template جديد.

# تمرير مرجع لتجزئة الإعدادات
my $config = {
    INCLUDE_PATH => '/here/there:/every/where',
    POST_CHOMP   => 1,
};
test_expect(\*DATA, $config);
# أو إنشاء كائن Template صراحةً
my $template = Template->new($config);
test_expect(\*DATA, $template);

يمكن استخدام المعامل الثالث للإشارة إلى تجزئة لمتغيرات القالب التي يتم تعريفها أثناء معالجة الاختبارات. يتم تمرير هذا إلى طريقة معالجة Template process().

my $replace = {
    a => 'alpha',
    b => 'bravo',
};
test_expect(\*DATA, $config, $replace);

يمكن ترك المعامل الثاني بدون تعريف لتحديد إعداد افتراضي لـ Template.

test_expect(\*DATA, undef, $replace);

لاختبار نتائج إعدادات Template المختلفة، يمكنك أيضًا تمرير مرجع لقائمة من كائنات Template المسماة كمعامل ثانٍ.

my $tt1 = Template->new({ ... });
my $tt2 = Template->new({ ... });
my @tts = [ one => $tt1, two => $tt1 ];

يُستخدم الكائن الأول في القائمة بشكل افتراضي. يمكن تبديل الكائنات الأخرى باستخدام علامة "-- use $name --". يجب أن يتبع هذا فورًا سطر "-- test --". سيتم استخدام هذا الكائن لبقية الاختبار، أو حتى يتم تحديد كائن آخر.

-- test --
-- use one --
[% blah %]
-- expect --
blah, blah
-- test --
still using one...
-- expect --
...
-- test --
-- use two --
[% blah %]
-- expect --
blah, blah, more blah

تقوم الدالة الفرعية test_expect() بحساب عدد الاختبارات ثم تستدعي ntests() لإنشاء سطر الاختبار المألوف ""1..$ntests\n"". يولد كل اختبار مُعرَّف رقمي اختبار. يشير الأول إلى أنه تم معالجة المدخل بدون أخطاء، والثاني إلى أن المخرج يطابق المتوقع.

يمكن تشغيل اختبارات إضافية قبل test_expect() عن طريق استدعاء ok(). يتم تخزين نتائج هذه الاختبارات مؤقتًا حتى يتم استدعاء ntests() ويمكن حساب العدد الإجمالي للاختبارات. بعد ذلك، يتم إنشاء سطر ""1..$ntests"" جنبًا إلى جنب مع أسطر ""ok $n""/""not ok $n"" لكل نتيجة اختبار مخزنة مؤقتًا. بعد ذلك، تقوم استدعاءات ok() بإنشاء سطر إخراج على الفور.

my $something = SomeObject->new();
ok( $something );
my $other = AnotherThing->new();
ok( $other );
test_expect(\*DATA);

إذا كنت ترغب في تجنب استدعاء test_expect()، يمكنك استدعاء ntests($n) لتحديد عدد الاختبارات وإنشاء سطر الرأس. بعد ذلك، ما عليك سوى استدعاء ok() لكل اختبار بقيمة صحيحة أو خاطئة للإشارة إلى اجتياز الاختبار أو فشله.

ntests(2);
ok(1);
ok(0);

إذا كنت كسولًا حقًا، يمكنك ببساطة استدعاء ok() وعدم القلق بشأن تحديد عدد الاختبارات. سيتم تخزين جميع نتائج الاختبارات مؤقتًا حتى نهاية البرنامج النصي وسيتم طباعتها في كتلة واحدة قبل انتهاء البرنامج.

ok( $x );
ok( $y );

يمكنك تحديد جزء معين فقط من ملف الإدخال للاختبار باستخدام علامتي "-- start --" و "-- stop --". ما قبل الظهور الأول لـ "-- start --" يتم تجاهله، وكذلك ما بعد الظهور التالي لعلامة "-- stop --".

-- test --
this is test 1 (not performed)
-- expect --
this is test 1 (not performed)
-- start --
-- test --
this is test 2
-- expect --
this is test 2
-- stop --
...

ntests()

دالة فرعية تستخدم لتحديد عدد الاختبارات المتوقع تنفيذها.

ok($test)

تنتج رسالة ""ok $n"" أو ""not ok $n"" إذا كانت $test صحيحة أو خاطئة.

not_ok($test)

العكس المنطقي لـ ok(). تطبع رسالة ""ok $n"" إذا كانت $test خاطئة والعكس صحيح.

callsign()

لأسباب تاريخية وفائدة عامة، تحدد الوحدة أيضًا دالة فرعية callsign() تُرجع تجزئة تربط الأحرف "a" إلى "z" بما يعادلها الأبجدي باللغة العربية (مثال: رموز النداء اللاسلكي). يُستخدم هذا كمصدر معروف للعديد من نصوص الاختبار.

هذا الموقع غير مرتبط بـ Arch Linux ولا تابع له بأي شكل من الأشكال. المحتوى هو ترجمة فقط لصفحات دليل Arch المنشورة على man.archlinux.org.
