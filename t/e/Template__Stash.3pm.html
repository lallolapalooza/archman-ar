Template__Stash(3pm)

الحزمة: perl-template-toolkit

التراخيص: PerlArtistic

Template::Stash(3) توثيق Perl مساهم من المستخدم Template::Stash(3)

الاسم

Template::Stash - تخزين سحري لمتغيرات القالب

الملخص

use Template::Stash;
my $stash = Template::Stash->new(\%vars);
# الحصول على قيم المتغيرات
$value = $stash->get($variable);
$value = $stash->get(\@compound);
# تعيين قيمة متغير
$stash->set($variable, $value);
$stash->set(\@compound, $value);
# قيمة متغير افتراضية
$stash->set($variable, $value, 1);
$stash->set(\@compound, $value, 1);
# تعيين قيم متغيرات دفعة واحدة
$stash->update(\%new_vars)
# دوال لـ (إلغاء) توطين المتغيرات
$stash = $stash->clone(\%new_vars);
$stash = $stash->declone();

الوصف

الوحدة "Template::Stash" تعرف فئة كائن تستخدم لتخزين قيم المتغيرات للاستخدام في وقت التشغيل لمعالج القوالب. يتم تخزين قيم المتغيرات داخليًا في مرجع هاش (والذي يتم مباركته لإنشاء الكائن) وهي متاحة عبر الدوال get() و set().

قد تشير المتغيرات إلى مصفوفات الهاش، القوائم، الإجراءات الفرعية، والكائنات بالإضافة إلى القيم البسيطة. يؤدي التخزين مؤقتًا تلقائيًا إلى تنفيذ السحر الصحيح عند التعامل مع المتغيرات، استدعاء الدوال أو طرق الكائنات، الفهرسة في القوائم، الهاش، إلخ.

يحتوي التخزين المؤقت على دوال clone() و declone() التي يستخدمها معالج القوالب لإنشاء نسخ مؤقتة للتخزين المؤقت لتوطين التغييرات التي تم إجراؤها على المتغيرات.

الدوال العامة

new(\%params)

تقوم الدالة البانية new() بإنشاء وإرجاع مرجع لكائن "Template::Stash" جديد.

my $stash = Template::Stash->new();

يمكن تمرير مرجع هاش لتوفير المتغيرات والقيم التي يجب استخدامها لتهيئة التخزين المؤقت.

my $stash = Template::Stash->new({ var1 => 'value1',
                                   var2 => 'value2' });

get($variable)

تقوم الدالة get() باسترداد المتغير المسمى بالمعلمة الأولى.

$value = $stash->get('var1');

يمكن استرداد المتغيرات المركبة المنقوطة عن طريق تحديد عناصر المتغير بمرجع إلى قائمة. يحتل كل عقدة في المتغير مدخلين في القائمة. الأول يعطي اسم عنصر المتغير، والثاني هو مرجع إلى قائمة وسائط لذلك العنصر، أو 0 إذا لم يكن هناك.

[% foo.bar(10).baz(20) %]
$stash->get([ 'foo', 0, 'bar', [ 10 ], 'baz', [ 20 ] ]);

set($variable, $value, $default)

تقوم الدالة set() بتعيين اسم المتغير في المعلمة الأولى إلى القيمة المحددة في الثانية.

$stash->set('var1', 'value1');

إذا كانت المعلمة الثالثة تقييمًا لقيمة صحيحة، فسيتم تعيين المتغير فقط إذا لم يكن له قيمة صحيحة من قبل.

$stash->set('var2', 'default_value', 1);

يمكن تحديد المتغيرات المركبة المنقوطة كما في get() أعلاه.

[% foo.bar = 30 %]
$stash->set([ 'foo', 0, 'bar', 0 ], 30);

يمكن تحديد المتغير السحري '"IMPORT"' والذي يجب أن تكون قيمته المقابلة مرجع هاش. يتم نسخ محتويات مصفوفة الهاش (أي استيرادها) إلى مساحة الأسماء الحالية.

# foo.bar = baz, foo.wiz = waz
$stash->set('foo', { 'bar' => 'baz', 'wiz' => 'waz' });
# استيراد 'foo' إلى مساحة الأسماء الرئيسية: bar = baz, wiz = waz
$stash->set('IMPORT', $stash->get('foo'));

update($variables)

يمكن استخدام هذه الدالة لتعيين أو تحديث عدة متغيرات في وقت واحد.

$stash->update({
    foo => 10,
    bar => 20,
});

getref($variable)

هذه الميزة غير الموثقة ترجع إغلاقًا يمكن استدعاؤه للحصول على قيمة متغير. يتم استخدامه لتنفيذ مراجع المتغيرات التي يتم تقييمها بكسل.

[% x = \foo.bar.baz %]          # x هو مرجع لـ foo.bar.baz
[% x %]                         # تقييم foo.bar.baz

clone(\%params)

تنشئ الدالة clone() وترجع كائن "Template::Stash" جديد يمثل نسخة محلية من التخزين المؤقت الأصل. يمكن تحديث المتغيرات بحرية في التخزين المؤقت المستنسخ وعند استدعاء declone()، يتم إرجاع التخزين المؤقت الأصلي بجميع أعضائه سليمة وفي نفس الحالة التي كانت عليها قبل استدعاء clone().

لتسهيل الأمر، يمكن تمرير هاش من المعلمات إلى clone() والتي تستخدم لتحديث أي متغيرات بسيطة (أي تلك التي لا تحتوي على أي عناصر مساحة أسماء مثل "foo" و "bar" ولكن ليس "foo.bar") أثناء استنساخ التخزين المؤقت. لإضافة وتحديث المتغيرات المعقدة، يجب استخدام الدالة set() بعد استدعاء "clone()." سيؤدي هذا بشكل صحيح إلى حل و/أو إنشاء أي هياكل مساحة أسماء ضرورية.

يحافظ التخزين المؤقت المستنسخ على مرجع للتخزين المؤقت الذي تم نسخه منه في عضوه "_PARENT".

declone()

ترجع الدالة declone() المرجع "_PARENT" ويمكن استخدامها لاستعادة حالة التخزين المؤقت كما هو موضح أعلاه.

define_vmethod($type, $name, $code)

يمكن استخدام هذه الدالة لتعريف دوال افتراضية جديدة. يجب أن تكون المعلمة الأولى إما "scalar" أو "item" لتعريف دالة افتراضية قياسية، "hash" لتعريف دوال هاش افتراضية، أو إما "array" أو "list" لدوال قائمة افتراضية. يجب أن تكون المعلمة الثانية اسم الدالة الجديدة. يجب أن تكون المعلمة الثالثة مرجعًا لإجراء فرعي ينفذ الدالة. يتم تمرير عنصر البيانات الذي يتم استدعاء الدالة الافتراضية عليه إلى الإجراء الفرعي كوسيط أول.

$stash->define_vmethod(
    item => ucfirst => sub {
        my $text = shift;
        return ucfirst $text
    }
);

الدوال الداخلية

dotop($root, $item, \@args, $lvalue)

هذه هي دالة عملية "النقطة" الأساسية التي تقيّم عناصر المتغيرات مقابل جذرها.

undefined($ident, $args)

يتم استدعاء هذه الدالة عندما يصادف get() قيمة غير معرفة. إذا كان خيار STRICT ساري المفعول، فسيتم طرح استثناء يشير إلى استخدام قيمة غير معرفة. بخلاف ذلك، ستقوم بإرجاع سلسلة فارغة بصمت.

يمكن إعادة تعريف الدالة في فئة فرعية لتنفيذ معالجة بديلة للقيم غير المعرفة.
