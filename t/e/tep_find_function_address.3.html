tep_find_function_address(3)

الحزمة: libtraceevent

التراخيص: GPL-2.0-only, LGPL-2.1-only

LIBTRACEEVENT(3) دليل libtraceevent LIBTRACEEVENT(3)

الاسم

tep_find_function, tep_find_function_address, tep_find_function_info - البحث عن اسم الدالة / عنوان البداية.

الملخص

#include <event-parse.h>
const char *tep_find_function(struct tep_handle *tep, unsigned long long addr);
unsigned long long tep_find_function_address(struct tep_handle *tep, unsigned long long addr);
int tep_find_function_info(struct tep_handle *tep, unsigned long long addr, const char **name,
                           unsigned long long *start, unsigned long *size);

الوصف

يمكن استخدام هذه الدوال للبحث عن اسم الدالة وعنوان البداية، بناءً على عنوان معطى. لا يشترط أن يكون العنوان المعطى دقيقاً، بل سيتم تحديد الدالة التي تحتويه.

تقوم الدالة tep_find_function() بإرجاع اسم الدالة التي تحتوي العنوان المعطى addr. الوسيط tep هو سياق محلل أحداث التتبع.

تقوم الدالة tep_find_function_address() بإرجاع عنوان بداية الدالة، بناءً على العنوان المعطى addr. لا يشترط أن يكون العنوان المعطى دقيقاً، بل سيتم تحديد الدالة التي تحتويه. الوسيط tep هو سياق محلل أحداث التتبع.

تقوم الدالة tep_find_function_info() باسترداد name (الاسم)، عنوان البداية (start)، و size (حجم) نص الدالة عند address، إذا تم العثور عليها. ملاحظة: إذا كان مقبض tep يحتوي على محلل دوال (يستخدمه perf)، فسيتم تعيين size إلى صفر.

قيمة الإرجاع

تقوم الدالة tep_find_function() بإرجاع اسم الدالة، أو NULL في حالة عدم العثور عليها.

تقوم الدالة tep_find_function_address() بإرجاع عنوان بداية الدالة، أو 0 في حالة عدم العثور عليها.

تقوم الدالة tep_find_function_info() بإرجاع 1 إذا تم العثور على دالة للعنوان المعطى، أو 0 إذا لم يتم العثور عليها.

مثال

#include <event-parse.h>
...
struct tep_handle *tep = tep_alloc();
...
void show_function_name(unsigned long long addr)
{
        const char *fname = tep_find_function(tep, addr);
        if (fname)
                printf("تم العثور على الدالة %s عند 0x%0llx\n", fname, addr);
        else
                printf("لم يتم العثور على دالة عند 0x%0llx\n", addr);
}
void show_function_start_addr(unsigned long long addr)
{
        const char *fname = tep_find_function(tep, addr);
        unsigned long long fstart;
        if (!fname) {
                printf("لم يتم العثور على دالة عند 0x%0llx\n", addr);
                return;
        }
        fstart = tep_find_function_address(tep, addr);
        printf("الدالة %s عند 0x%llx تبدأ عند 0x%0llx\n",
               fname, addr, fstart);
}
void show_function_info(unsigned long long addr)
{
        const char *fname;
        unsigned long long fstart;
        unsigned long size;
        ret = tep_find_function_info(tep, addr, &fname, &fstart, &size);
        if (!ret) {
                printf("لم يتم العثور على دالة عند 0x%0lx\n", addr);
                return;
        }
        printf("الدالة %s عند 0x%lx تبدأ عند 0x%0lx وحجمها %ld\n",
               fname, addr, fstart, size);
}
