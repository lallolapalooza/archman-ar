tep_filter_match(3)

الحزمة: libtraceevent

التراخيص: GPL-2.0-only, LGPL-2.1-only

LIBTRACEEVENT(3) دليل libtraceevent LIBTRACEEVENT(3)

الاسم

tep_filter_alloc, tep_filter_free, tep_filter_reset, tep_filter_make_string, tep_filter_copy, tep_filter_compare, tep_filter_match, tep_event_filtered, tep_filter_remove_event, tep_filter_strerror, tep_filter_add_filter_str - واجهات برمجة تطبيقات متعلقة بتصفية الأحداث.

الملخص

#include <event-parse.h>
struct tep_event_filter *tep_filter_alloc(struct tep_handle *tep);
void tep_filter_free(struct tep_event_filter *filter);
void tep_filter_reset(struct tep_event_filter *filter);
enum tep_errno tep_filter_add_filter_str(struct tep_event_filter *filter, const char *filter_str);
int tep_event_filtered(struct tep_event_filter *filter, int event_id);
int tep_filter_remove_event(struct tep_event_filter *filter, int event_id);
enum tep_errno tep_filter_match(struct tep_event_filter *filter, struct tep_record *record);
int tep_filter_copy(struct tep_event_filter *dest, struct tep_event_filter *source);
int tep_filter_compare(struct tep_event_filter *filter1, struct tep_event_filter *filter2);
char *tep_filter_make_string(struct tep_event_filter *filter, int event_id);
int tep_filter_strerror(struct tep_event_filter *filter, enum tep_errno err, char *buf, size_t buflen);

الوصف

يمكن ربط المرشحات بالأحداث المتتبعة. يمكن استخدامها لتصفية الأحداث المختلفة عند إخراجها. يمكن تصفية كل حدث بناءً على معلماته، الموصوفة في ملف تنسيق الحدث. يمكن استخدام هذه المجموعة من الدوال لإنشاء، حذف، تعديل وربط مرشحات الأحداث.

تقوم الدالة tep_filter_alloc() بإنشاء مرشح أحداث جديد. الوسيط tep هو سياق محلل أحداث التتبع.

تقوم الدالة tep_filter_free() بتحرير مرشح الأحداث وجميع الموارد التي استخدمها.

تقوم الدالة tep_filter_reset() بإزالة جميع القواعد من مرشح الأحداث وإعادة تعيينه.

تقوم الدالة tep_filter_add_filter_str() بإضافة قاعدة جديدة إلى filter. الوسيط filter_str هو سلسلة المرشح، التي تحتوي على القاعدة.

تقوم الدالة tep_event_filtered() بالتحقق مما إذا كان الحدث ذو event_id لديه filter.

تقوم الدالة tep_filter_remove_event() بإزالة filter لحدث ذي event_id.

تقوم الدالة tep_filter_match() باختبار ما إذا كان record يطابق filter المحدد.

تقوم الدالة tep_filter_copy() بنسخ مرشح source إلى مرشح dest.

تقوم الدالة tep_filter_compare() بمقارنة مرشحين - filter1 و filter2.

تقوم الدالة tep_filter_make_string() بإنشاء سلسلة نصية، تعرض محتويات filter لـ event_id محدد.

تقوم الدالة tep_filter_strerror() بنسخ المخزن المؤقت لأخطاء filter إلى buf المحدد بالحجم buflen. إذا كان المخزن المؤقت للأخطاء فارغًا، فسيتم نسخ سلسلة تصف الخطأ err في buf.

قيمة الإرجاع

تقوم الدالة tep_filter_alloc() بإرجاع مؤشر إلى مرشح الأحداث الذي تم إنشاؤه حديثًا، أو NULL في حالة حدوث خطأ.

تقوم الدالة tep_filter_add_filter_str() بإرجاع 0 إذا تمت إضافة القاعدة بنجاح أو رمز خطأ سالب. استخدم tep_filter_strerror() لرؤية رسالة الخطأ الفعلية في حالة حدوث خطأ.

تقوم الدالة tep_event_filtered() بإرجاع 1 إذا تم العثور على المرشح للحدث المحدد، أو 0 بخلاف ذلك.

تقوم الدالة tep_filter_remove_event() بإرجاع 1 إذا تمت إزالة الحدث، أو 0 إذا لم يتم العثور على الحدث.

تقوم الدالة tep_filter_match() بإرجاع tep_errno، وفقًا للنتيجة:

TEP_ERRNO__FILTER_MATCH        - تم العثور على مرشح للحدث، السجل مطابق.
TEP_ERRNO__FILTER_MISS         - تم العثور على مرشح للحدث، السجل غير مطابق.
TEP_ERRNO__FILTER_NOT_FOUND    - لم يتم العثور على مرشح لحدث السجل.
TEP_ERRNO__NO_FILTER           - لا توجد قواعد في المرشح.

أو أي tep_errno آخر، إذا حدث خطأ أثناء الاختبار.

تقوم الدالة tep_filter_copy() بإرجاع 0 عند النجاح أو -1 إذا لم يتم نسخ جميع القواعد.

تقوم الدالة tep_filter_compare() بإرجاع 1 إذا كانت المرشحات متطابقة في المحتوى، أو 0 إذا لم تكن كذلك.

تقوم الدالة tep_filter_make_string() بإرجاع سلسلة نصية، والتي يجب تحريرها باستخدام free()، أو NULL في حالة حدوث خطأ.

تقوم الدالة tep_filter_strerror() بإرجاع 0 إذا تم ملء الرسالة بنجاح، أو -1 في حالة حدوث خطأ.

مثال

#include <event-parse.h>
...
struct tep_handle *tep = tep_alloc();
...
char errstr[200];
int ret;
struct tep_event_filter *filter = tep_filter_alloc(tep);
struct tep_event_filter *filter1 = tep_filter_alloc(tep);
ret = tep_filter_add_filter_str(filter, "sched/sched_wakeup:target_cpu==1");
if(ret < 0) {
        tep_filter_strerror(filter, ret, errstr, sizeof(errstr));
        /* فشل في إضافة قاعدة جديدة إلى المرشح، سلسلة الخطأ موجودة في errstr */
}
if (tep_filter_copy(filter1, filter) != 0) {
        /* فشل في نسخ المرشح إلى filter1 */
}
...
if (tep_filter_compare(filter, filter1) != 1) {
        /* المرشحان مختلفان */
}
...
void process_record(struct tep_handle *tep, struct tep_record *record)
{
        struct tep_event *event;
        char *fstring;
        event = tep_find_event_by_record(tep, record);
        if (tep_event_filtered(filter, event->id) == 1) {
                /* الحدث لديه مرشح */
                fstring = tep_filter_make_string(filter, event->id);
                if (fstring != NULL) {
                        /* المرشح للحدث موجود في fstring */
                        free(fstring);
                }
        }
        switch (tep_filter_match(filter, record)) {
        case TEP_ERRNO__FILTER_MATCH:
                /* المرشح يطابق السجل */
                break;
        case TEP_ERRNO__FILTER_MISS:
                /* المرشح لا يطابق السجل */
                break;
        case TEP_ERRNO__FILTER_NOT_FOUND:
                /* لم يتم العثور على مرشح لحدث السجل */
                break;
        case TEP_ERRNO__NO_FILTER:
                /* لا توجد قواعد في المرشح */
                break
        default:
                /* حدث خطأ أثناء الاختبار */
                break;
        }
        if (tep_filter_remove_event(filter, event->id) == 1) {
                /* تمت إزالة الحدث من المرشح */
        }
}
...
tep_filter_reset(filter);
...
tep_filter_free(filter);
tep_filter_free(filter1);
...
