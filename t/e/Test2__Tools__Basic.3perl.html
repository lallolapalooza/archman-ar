Test2::Tools::Basic(3perl) دليل مبرمجي Perl Test2::Tools::Basic(3perl)

الاسم

Test2::Tools::Basic - تطبيق Test2 لأدوات الاختبار الأساسية.

الوصف

هذا تطبيق Test2 للأدوات الأساسية التي تم توفيرها في الأصل بواسطة Test::More. لا توفر هذه الحزمة جميع أدوات Test::More، فقط الأدوات الأساسية/البسيطة. تم تعديل بعض الأدوات لتحسين قدرات التشخيص.

الملخص

use Test2::Tools::Basic;
ok($x, "اختبار بسيط");
if ($passing) {
    pass('اختبار ناجح');
}
else {
    fail('اختبار فاشل');
}
diag "هذه رسالة تشخيص على STDERR";
note "هذه رسالة تشخيص على STDOUT";
{
    my $todo = todo "سبب لـ todo";
    ok(0, "هذا الاختبار لـ todo");
}
ok(1, "هذا الاختبار ليس لـ todo");
todo "سبب" => sub {
    ok(0, "هذا الاختبار لـ todo");
};
ok(1, "هذا الاختبار ليس لـ todo");
SKIP: {
    skip "هذا سيمحو محرك الأقراص الخاص بك";
    # هذا لا يتم تشغيله أبداً:
    ok(!system('sudo rm -rf /'), "محو محرك الأقراص");
}
done_testing;

التصدير

يتم تصدير جميع الوظائف بشكل افتراضي.

التخطيط

plan($num)
plan('tests' => $num)
plan('skip_all' => $reason)
تعيين عدد الاختبارات المتوقعة. يجب القيام بذلك أولاً أو أخيراً، أبداً في منتصف الاختبار.

للتوافق مع الإصدارات القديمة، يمكنك تحديد 'tests' كوسيط أول قبل العدد. يمكنك أيضاً استخدام هذا لتخطي الكل باستخدام البادئة 'skip_all'، متبوعًا بسبب التخطي.

skip_all($reason)
تعيين الخطة إلى 0 مع سبب، ثم الخروج بقيمة صحيحة. يجب استخدام هذا قبل تشغيل أي اختبارات.
done_testing
يستخدم لتمييز نهاية الاختبار. هذه طريقة آمنة للحصول على عدد ديناميكي أو غير معروف من الاختبارات.
bail_out($reason)
يتم استدعاؤه عندما تسوء الأمور بشكل فظيع: أوقف كل شيء، واقتل جميع المسارات والعمليات، وأنهِ العملية بحالة خروج خاطئة.

التأكيدات

ok($bool)
ok($bool, $name)
ok($bool, $name, @diag)
تأكيد بسيط. إذا كانت $bool صحيحة، ينجح الاختبار، وإذا كانت خاطئة، يفشل الاختبار. اسم الاختبار اختياري، وجميع الوسائط بعد الاسم تُضاف كرسائل تشخيص فقط إذا فشل الاختبار. إذا نجح الاختبار، سيتم تجاهل جميع وسائط التشخيص.
pass()
pass($name)
إطلاق اختبار ناجح (حدث Ok واحد). الاسم اختياري.
fail()
fail($name)
fail($name, @diag)
إطلاق اختبار فاشل (حدث Ok واحد). الاسم والتشخيصات اختيارية.

التشخيصات

diag(@messages)
كتابة رسائل التشخيص. سيتم ربط جميع العناصر في @messages في سلسلة واحدة بدون فاصل. عند استخدام TAP، يتم إرسال التشخيصات إلى STDERR.

ترجع قيمة خاطئة، للحفاظ على الفشل.

note(@messages)
كتابة رسائل ملاحظات التشخيص. سيتم ربط جميع العناصر في @messages في سلسلة واحدة بدون فاصل. عند استخدام TAP، يتم إرسال الملاحظات إلى STDOUT.

البيانات الوصفية

$todo = todo($reason)
todo $reason => sub { ... }
يستخدم هذا لوضع علامة على بعض النتائج بأنها TODO. TODO يعني أن الاختبار قد يفشل، ولكنه لن يتسبب في فشل مجموعة الاختبارات بأكملها.

هناك طريقتان لاستخدام هذا. الأولى هي استخدام كتلة تعليمات برمجية، وسيتم تطبيق TODO فقط على كتلة التعليمات البرمجية.

ok(1, "قبل"); # ليس TODO
todo 'هذا سيفشل' => sub {
    # هذا هو TODO، وكذلك أي اختبار آخر في هذه الكتلة.
    ok(0, "بلح");
};
ok(1, "بعد"); # ليس TODO

الطريقة الأخرى هي استخدام متغير نطاق. سينتهي TODO عندما يتم تدمير المتغير أو تعيينه إلى undef.

ok(1, "قبل"); # ليس TODO
{
    my $todo = todo 'هذا سيفشل';
    # هذا هو TODO، وكذلك أي اختبار آخر في هذه الكتلة.
    ok(0, "بلح");
};
ok(1, "بعد"); # ليس TODO

هذا هو نفس الشيء، ولكن بدون نطاق "{...}".

ok(1, "قبل"); # ليس TODO
my $todo = todo 'هذا سيفشل';
ok(0, "بلح"); # TODO
$todo = undef;
ok(1, "بعد"); # ليس TODO
skip($why)
skip($why, $count)
يستخدم هذا لتخطي بعض الاختبارات. يتطلب هذا تغليف اختباراتك في كتلة مسماة "SKIP:". هذا سحري إلى حد ما. إذا لم يتم تحديد $count، فسيصدر نتيجة واحدة. إذا قمت بتحديد $count، فسيصدر هذا العدد من النتائج. 
SKIP: {
    skip "هذا سيمحو محرك الأقراص الخاص بك";
    # هذا لا يتم تشغيله أبداً:
    ok(!system('sudo rm -rf /'), "محو محرك الأقراص");
}
