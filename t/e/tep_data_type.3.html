tep_data_type(3)

الحزمة: libtraceevent

التراخيص: GPL-2.0-only, LGPL-2.1-only

LIBTRACEEVENT(3) دليل libtraceevent LIBTRACEEVENT(3)

الاسم

tep_data_type, tep_data_pid, tep_data_preempt_count, tep_data_flags - استخراج الحقول المشتركة من سجل.

الملخص

#include <event-parse.h>
enum trace_flag_type {
        TRACE_FLAG_IRQS_OFF,
        TRACE_FLAG_IRQS_NOSUPPORT,
        TRACE_FLAG_NEED_RESCHED,
        TRACE_FLAG_HARDIRQ,
        TRACE_FLAG_SOFTIRQ,
};
int tep_data_type(struct tep_handle *tep, struct tep_record *rec);
int tep_data_pid(struct tep_handle *tep, struct tep_record *rec);
int tep_data_preempt_count(struct tep_handle *tep, struct tep_record *rec);
int tep_data_flags(struct tep_handle *tep, struct tep_record *rec);

الوصف

يمكن استخدام هذه المجموعة من الدوال لاستخراج الحقول المشتركة من سجل.

تقوم الدالة tep_data_type() بالحصول على معرف الحدث من السجل rec. تقرأ حقل "common_type". الوسيط tep هو سياق محلل حدث التتبع.

تقوم الدالة tep_data_pid() بالحصول على معرف العملية من السجل rec. تقرأ حقل "common_pid". الوسيط tep هو سياق محلل حدث التتبع.

تقوم الدالة tep_data_preempt_count() بالحصول على عداد الاستباق من السجل rec. تقرأ حقل "common_preempt_count". الوسيط tep هو سياق محلل حدث التتبع.

تقوم الدالة tep_data_flags() بالحصول على علامات التأخير من السجل rec. تقرأ حقل "common_flags". الوسيط tep هو سياق محلل حدث التتبع. علامات التأخير المدعومة هي:

        TRACE_FLAG_IRQS_OFF,          المقاطعات معطلة.
        TRACE_FLAG_IRQS_NOSUPPORT,    قراءة علامة IRQ غير مدعومة من قبل البنية.
        TRACE_FLAG_NEED_RESCHED,      تحتاج المهمة إلى إعادة جدولة.
        TRACE_FLAG_HARDIRQ,           يعمل Hard IRQ.
        TRACE_FLAG_SOFTIRQ,           يعمل Soft IRQ.

قيمة الإرجاع

تقوم الدالة tep_data_type() بإرجاع عدد صحيح، يمثل معرف الحدث.

تقوم الدالة tep_data_pid() بإرجاع عدد صحيح، يمثل معرف العملية.

تقوم الدالة tep_data_preempt_count() بإرجاع عدد صحيح، يمثل عداد الاستباق.

تقوم الدالة tep_data_flags() بإرجاع عدد صحيح، يمثل علامات التأخير. انظر إلى التعداد trace_flag_type لمعرفة العلامات المدعومة.

ترجع جميع هذه الدوال في حالة حدوث خطأ عددًا صحيحًا سالبًا.

مثال

#include <event-parse.h>
...
struct tep_handle *tep = tep_alloc();
...
void process_record(struct tep_record *record)
{
        int data;
        data = tep_data_type(tep, record);
        if (data >= 0) {
                /* حصلت على معرف الحدث */
        }
        data = tep_data_pid(tep, record);
        if (data >= 0) {
                /* حصلت على معرف العملية */
        }
        data = tep_data_preempt_count(tep, record);
        if (data >= 0) {
                /* حصلت على عداد الاستباق */
        }
        data = tep_data_flags(tep, record);
        if (data >= 0) {
                /* حصلت على علامات التأخير */
        }
}
...
