Test__More.3perl

الحزمة: perl

التراخيص: Artistic-1.0-Perl OR GPL-1.0-or-later

'waffle'; my $bar = 'yarblokos'; is( $foo, $bar, 'هل foo هو نفسه bar?' );

سينتج شيئًا كهذا:

not ok 17 - هل foo هو نفسه bar؟
#   فشل الاختبار 'هل foo هو نفسه bar؟'
#   في foo.t في السطر 139.
#          حصلت على: 'waffle'
#     متوقع: 'yarblokos'

لذا يمكنك معرفة ما الخطأ الذي حدث دون إعادة تشغيل الاختبار.

يتم تشجيعك على استخدام is() و isnt() بدلاً من ok() حيثما أمكن، ولكن لا تغتر باستخدامها لمعرفة ما إذا كان شيء ما صحيحًا أم خاطئًا!

# XXX سيء!
is( exists $brooklyn{tree}, 1, 'شجرة تنمو في بروكلين' );

هذا لا يتحقق مما إذا كان "exists $brooklyn{tree}" صحيحًا، بل يتحقق مما إذا كان يُرجع 1. هذا مختلف جدًا. توجد تحذيرات مماثلة للقيم الخاطئة و 0. في هذه الحالات، استخدم ok().

ok( exists $brooklyn{tree},    'شجرة تنمو في بروكلين' );

عادةً لا يوفر الاستدعاء البسيط لـ isnt() اختبارًا قويًا، ولكن هناك حالات لا يمكنك فيها قول الكثير عن قيمة ما سوى أنها تختلف عن قيمة أخرى:

new_ok $obj, "Foo";
my $clone = $obj->clone;
isa_ok $obj, "Foo", "Foo->clone";
isnt $obj, $clone, "clone() تنتج كائنًا مختلفًا";

تاريخيًا، دعمنا دالة "isn't()" كاسم مستعار لـ isnt()، ولكن في Perl 5.37.9 تم إهمال دعم استخدام الفاصلة العليا كفاصل للحزم، وبحلول Perl 5.42.0 سيتم إزالة الدعم لها تمامًا. وفقًا لذلك، تم أيضًا إهمال استخدام "isn't()"، وسينتج تحذيرات عند استخدامه ما لم يتم تعطيل تحذيرات 'deprecated' بشكل خاص في النطاق الذي يتم استخدامه فيه. يُنصح بشدة بالانتقال إلى استخدام isnt() بدلاً من ذلك.

like
like( $got, qr/expected/, $test_name );

على غرار ok()، تقوم like() بمطابقة $got مع التعبير العادي "qr/expected/".

إذًا هذا:

like($got, qr/expected/, 'هذا مثل ذلك');

مشابه لـ:

ok( $got =~ m/expected/, 'هذا مثل ذلك');

(تذكير "هذا مثل ذلك".)

الوسيط الثاني هو تعبير عادي. يمكن تقديمه كمرجع للتعبير العادي (أي "qr//") أو (للتوافق الأفضل مع إصدارات Perl الأقدم) كسلسلة نصية تبدو كتعبير عادي (المحددات البديلة غير مدعومة حاليًا):

like( $got, '/expected/', 'هذا مثل ذلك' );

يمكن وضع خيارات التعبير العادي في النهاية ('/expected/i').

مزاياها على ok() مشابهة لمزايا is() و isnt(). تشخيصات أفضل عند الفشل.

unlike
unlike( $got, qr/expected/, $test_name );

تعمل تمامًا مثل like()، فقط إنها تتحقق مما إذا كان $got لا يطابق النمط المحدد.

cmp_ok
cmp_ok( $got, $op, $expected, $test_name );

في منتصف الطريق بين ok() و is() تقع cmp_ok(). يتيح لك هذا مقارنة وسيطين باستخدام أي عامل ثنائي في Perl. ينجح الاختبار إذا كانت المقارنة صحيحة وتفشل بخلاف ذلك.

# ok( $got eq $expected );
cmp_ok( $got, 'eq', $expected, 'هذا eq ذاك' );
# ok( $got == $expected );
cmp_ok( $got, '==', $expected, 'هذا == ذاك' );
# ok( $got && $expected );
cmp_ok( $got, '&&', $expected, 'هذا && ذاك' );
...إلخ...

ميزتها على ok() هي أنه عند فشل الاختبار ستعرف ما هي قيم $got و $expected:

not ok 1
#   فشل الاختبار في foo.t في السطر 12.
#     '23'
#         &&
#     undef

إنها مفيدة أيضًا في الحالات التي تقارن فيها الأرقام و استخدام is() لـ "eq" سيتداخل:

cmp_ok( $big_hairy_number, '==', $another_big_hairy_number );

إنها مفيدة بشكل خاص عند مقارنة العلاقة أكبر من أو أصغر من بين القيم:

cmp_ok( $some_value, '<=', $upper_limit );
can_ok
can_ok($module, @methods);
can_ok($object, @methods);

تتحقق من أن $module أو $object يمكنه القيام بهذه @methods (يعمل مع الدوال أيضًا).

can_ok('Foo', qw(this that whatever));

هو تقريبًا نفس قول:

ok( Foo->can('this') && 
    Foo->can('that') && 
    Foo->can('whatever') 
  );

فقط دون كل الكتابة وواجهة أفضل. مفيد للاختبار السريع للواجهة.

بغض النظر عن عدد @methods التي تتحقق منها، فإن استدعاء can_ok() واحد يعتبر اختبارًا واحدًا. إذا رغبت في خلاف ذلك، استخدم:

foreach my $meth (@methods) {
    can_ok('Foo', $meth);
}
isa_ok
isa_ok($object,   $class, $object_name);
isa_ok($subclass, $class, $object_name);
isa_ok($ref,      $type,  $ref_name);

تتحقق مما إذا كان "$object->isa($class)" المحدد. تتحقق أيضًا من أن الكائن تم تعريفه في المقام الأول. مفيد لهذا النوع من الأشياء:

my $obj = Some::Module->new;
isa_ok( $obj, 'Some::Module' );

حيث ستحتاج بخلاف ذلك إلى كتابة:

my $obj = Some::Module->new;
ok( defined $obj && $obj->isa('Some::Module') );

للحماية من انفجار نص الاختبار الخاص بك.

يمكنك أيضًا اختبار فئة، للتأكد من أنها لديها السلف الصحيح:

isa_ok( 'Vole', 'Rodent' );

يعمل على المراجع أيضًا:

isa_ok( $array_ref, 'ARRAY' );

التشخيصات لهذا الاختبار تشير عادةً فقط إلى "الكائن". إذا كنت ترغب في أن تكون أكثر تحديدًا، يمكنك توفير $object_name (على سبيل المثال، "Test customer").

new_ok
my $obj = new_ok( $class );
my $obj = new_ok( $class => \@args );
my $obj = new_ok( $class => \@args, $object_name );

وظيفة راحة تجمع بين إنشاء كائن واستدعاء isa_ok() على هذا الكائن.

إنها أساسًا مكافئة لـ:

my $obj = $class->new(@args);
isa_ok $obj, $class, $object_name;

إذا لم يتم توفير @args، فسيتم استخدام قائمة فارغة.

تعمل هذه الوظيفة فقط على new() وتفترض أن new() ستُرجع كائنًا واحدًا فقط وهو $class.

subtest
subtest $name => \&code, @args;

subtest() تشغل &code كاختبارها الصغير الخاص بها بخطة خاصة بها ونتيجة خاصة بها. يحسب الاختبار الرئيسي هذا كاختبار واحد باستخدام نتيجة الاختبار الفرعي بالكامل لتحديد ما إذا كان صحيحًا أم خاطئًا.

على سبيل المثال...

use Test::More tests => 3;

pass("الاختبار الأول");
subtest 'اختبار فرعي مثال' => sub {
    plan tests => 2;
    pass("هذا اختبار فرعي");
    pass("وهذا أيضًا");
};
pass("الاختبار الثالث");

هذا سينتج.

1..3
ok 1 - الاختبار الأول
    # اختبار فرعي: اختبار فرعي مثال
    1..2
    ok 1 - هذا اختبار فرعي
    ok 2 - وهذا أيضًا
ok 2 - اختبار فرعي مثال
ok 3 - الاختبار الثالث

يمكن لـ subtest استدعاء "skip_all". لن يتم تشغيل أي اختبارات، ولكن يعتبر الاختبار الفرعي تخطيًا.

subtest 'skippy' => sub {
    plan skip_all => 'لأنني قلت ذلك';
    pass('هذا الاختبار لن يتم تشغيله أبدًا');
};

تُرجع صحيحة إذا نجح الاختبار الفرعي، وخاطئة بخلاف ذلك.

نظرًا لكيفية عمل الاختبارات الفرعية، يمكنك حذف خطة إذا رغبت في ذلك. يضيف هذا done_testing() ضمنيًا إلى نهاية الاختبار الفرعي الخاص بك. الاختباران الفرعيان التاليان متكافئان:

subtest 'اختبار فرعي مع done_testing() ضمني', sub {
    ok 1, 'يجب أن تعمل الاختبارات الفرعية مع اختبار منطقي ضمني';
    ok 1, '... وتدعم أكثر من اختبار واحد';
    ok 1, '... بغض النظر عن عدد الاختبارات التي تم تشغيلها';
};
subtest 'اختبار فرعي مع done_testing() صريح', sub {
    ok 1, 'يجب أن تعمل الاختبارات الفرعية مع اختبار منطقي صريح';
    ok 1, '... وتدعم أكثر من اختبار واحد';
    ok 1, '... بغض النظر عن عدد الاختبارات التي تم تشغيلها';
    done_testing();
};

الوسائط الإضافية المعطاة لـ "subtest" يتم تمريرها إلى الاستدعاء. على سبيل المثال:

sub my_subtest {
    my $range = shift;
    ...
}
for my $range (1, 10, 100, 1000) {
    subtest "اختبار النطاق $range", \&my_subtest, $range;
}
pass
fail
pass($test_name);
fail($test_name);

أحيانًا تريد ببساطة القول بأن الاختبارات قد نجحت. عادةً ما يكون هذا هو الحال عندما يكون لديك شرط معقد يصعب وضعه في ok(). في هذه الحالة، يمكنك ببساطة استخدام pass() (لإعلان نجاح الاختبار) أو fail (للفشل). إنهما مرادفان لـ ok(1) و ok(0).

استخدم هذه بحذر شديد جدًا.

اختبارات الوحدات (Modules)

في بعض الأحيان تريد التحقق مما إذا كانت وحدة، أو قائمة من الوحدات، يمكن تحميلها بنجاح. على سبيل المثال، غالبًا ما تريد اختبارًا أولًا يقوم ببساطة بتحميل جميع الوحدات في التوزيع للتأكد من أنها تعمل قبل المتابعة في اختبارات أكثر تعقيدًا.

SYNOPSIS

    use Test::More tests => 1;
    ok 1, "This is a test";

Or

    use Test::More tests => 2;

Or

    use Test::More;

Then after all your code, you must end your test file with:

    done_testing();

Or in newer versions of perl:

    use Test::More;

Or

    use Test::More;

Or

    use Test::More;

DESCRIPTION

Test::More is a module designed to be the main testing module for most Perl programs. It provides a simple interface to the testing infrastructure.

For more advanced testing needs, consider Test::Deep, Test::Builder::Module, or modules from the Test2 ecosystem.

For simple tests, Test::More is the best choice. For tests that are a little more complex, but still simple, Test::More is still the best choice. If you're not sure if you should be using Test::More, you should be using Test::More.

If you are completely new to testing in Perl, you should look at Test::Simple first. It provides a subset of Test::More's interface, and is easier to learn.

EXIT CODES

Test::More will exit with status 0 if all tests pass, and a non-zero status otherwise. This is compatible with Test::Harness and "make test".

COMPATIBILITY

Test::More is forwards compatible with Test::Simple. You can change your tests from:

    use Test::Simple tests => 1;

to

    use Test::More tests => 1;

and nothing else will need to change.

CAVEATS and NOTES

Complex data structures
Test::More is not designed to test complex data structures. For that, you should use Test::Deep which has more flexible testing functions for complex data structures.
Threads
Test::More will only be aware of threads if "use threads" has been done before Test::More is loaded. This is ok: 
use threads;
use Test::More;

This may cause problems:

use Test::More
use threads;

Perl 5.8.1 and above are supported. Anything below that has too many bugs.

HISTORY

This is a case of convergent evolution with Joshua Pritikin's Test module. I was largely unaware of its existence when I'd first written my own ok() routines. This module exists because I can't figure out how to easily wedge test names into Test's interface (along with a few other problems).

The goal here is to have a testing utility that's simple to learn, quick to use and difficult to trip yourself up with while still providing more flexibility than the existing Test.pm. As such, the names of the most common routines are kept tiny, special cases and magic side-effects are kept to a minimum. WYSIWYG.

ALTERNATIVES

Test2::Suite is the most recent and modern set of tools for testing.

Test::Simple if all this confuses you and you just want to write some tests. You can upgrade to Test::More later (it's forward compatible).

Test::Legacy tests written with Test.pm, the original testing module, do not play well with other testing libraries. Test::Legacy emulates the Test.pm interface and does play well with others.

ADDITIONAL LIBRARIES

Test::Differences for more ways to test complex data structures. And it plays well with Test::More.

Test::Class is like xUnit but more perlish.

Test::Deep gives you more powerful complex data structure testing.

Test::Inline shows the idea of embedded testing.

Mock::Quick The ultimate mocking library. Easily spawn objects defined on the fly. Can also override, block, or reimplement packages as needed.

Test::FixtureBuilder Quickly define fixture data for unit tests.

OTHER COMPONENTS

Test::Harness is the test runner and output interpreter for Perl. It's the thing that powers "make test" and where the "prove" utility comes from.
