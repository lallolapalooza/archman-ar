الحزمة: perl-template-toolkit

التراخيص: PerlArtistic

Template::Plugin(3) توثيق حزم Perl المساهمة Template::Plugin(3)

الاسم

Template::Plugin - فئة أساسية لإضافات Template Toolkit

ملخص

package MiOrg::Template::Plugin::MiPlugin;
use base qw( Template::Plugin );
use Template::Plugin;
use MiModulo;
sub new {
    my $class   = shift;
    my $context = shift;
    bless {
        ...
    }, $class;
}

الوصف

الإضافة (plugin) لـ Template Toolkit هي ببساطة وحدة Perl موجودة في مكان معروف بالتثبيت (على سبيل المثال، "Template::Plugin::*") وتلتزم بمعيار عادي، مما يسمح بتحميلها واستخدامها تلقائيًا.

تحدد الوحدة "Template::Plugin" فئة أساسية يمكن أن تشتق منها وحدات الإضافات الأخرى. لا تحتاج الإضافة إلى الاشتقاق من Template::Plugin، ولكن يجب أن تلتزم بواجهتها الموجهة للكائنات على الأقل.

يُنصح بإنشاء الإضافات في مساحة اسم الحزمة الخاصة بها لتجنب التعارض مع إضافات Toolkit. على سبيل المثال:

package MiOrg::Template::Plugin::FooBar;

استخدم خيار PLUGIN_BASE لتحديد مساحة الاسم التي تستخدمها. على سبيل المثال:

use Template;
my $template = Template->new({
    PLUGIN_BASE => 'MiOrg::Template::Plugin',
});

الطرق

تشكل الطرق التالية الواجهة الأساسية بين Template Toolkit ووحدات الإضافات.

load($context)

يتم استدعاء هذه الطريقة بواسطة Toolkit عند تحميل وحدة الإضافة لأول مرة. يتم استدعاؤها كطريقة حزمة وبالتالي تتلقى ضمنيًا اسم الفئة كمعامل أول. كما يتم تمرير مرجع إلى كائن Template::Context الذي يقوم بتحميل الإضافة. السلوك الافتراضي للطريقة load() هو ببساطة إرجاع اسم الفئة. ثم يستخدم السياق المتصل اسم الفئة هذا لاستدعاء الطريقة new() للفئة.

package MiPlugin;
sub load {               # يتم استدعاؤها كـ MiPlugin->load($context)
    my ($class, $context) = @_;
    return $class;       # ترجع 'MiPlugin'
}

new($context, @params)

يتم استدعاء هذه الطريقة لإنشاء كائن إضافة جديد لتوجيه USE. يتم استدعاؤها كطريقة حزمة مقابل اسم الفئة الذي تم إرجاعه بواسطة load(). يتم تمرير مرجع إلى كائن Template::Context الذي ينشئ الإضافة، بالإضافة إلى أي معلمات إضافية محددة في توجيه USE.

sub new {                # يتم استدعاؤها كـ MiPlugin->new($context)
    my ($class, $context, @params) = @_;
    bless {
        _CONTEXT => $context,
    }, $class;           # ترجع كائن MiPlugin المبارك
}

error($error)

تُستخدم هذه الطريقة، الموروثة من الوحدة Template::Base، للإبلاغ عن الأخطاء وإرجاعها. يمكن استدعاؤها كطريقة حزمة لتعيين/إرجاع المتغير $ERROR، أو كطريقة كائن لتعيين/إرجاع خاصية _ERROR للكائن. عند استدعائها بمعامل، تقوم بتعيين المتغير المقابل وإرجاع undef. عند عدم استدعائها بمعامل، ترجع قيمة المتغير.

package MiPlugin;
use base 'Template::Plugin';
sub new {
    my ($class, $context, $dsn) = @_;
    return $class->error('No data source specified')
        unless $dsn;
    bless {
        _DSN => $dsn,
    }, $class;
}
package main;
my $something = MiPlugin->new()
    || die MiPlugin->error(), "\n";
$something->do_something()
    || die $something->error(), "\n";

سحر أعمق

كائن Template::Context الذي يتعامل مع تحميل الإضافات واستخدامها يستدعي الطريقتين new() و error() مقابل اسم الفئة الذي تم إرجاعه بواسطة الطريقة load(). بعبارات شبه كود، يبدو الأمر كالتالي:

$class  = MiPlugin->load($context);       # ترجع 'MiPlugin'
$object = $class->new($context, @params)  # MiPlugin->new(...)
    || die $class->error();               # MiPlugin->error()

يمكن لطريقة load() بدلاً من ذلك إرجاع مرجع لكائن تم إنشاؤه. في هذه الحالة، يتم استدعاء الطريقتين new() و error() كطرق كائن مقابل الكائن الذي تم إنشاؤه.

package YourPlugin;
sub load {
    my ($class, $context) = @_;
    bless {
        _CONTEXT => $context,
    }, $class;
}
sub new {
    my ($self, $context, @params) = @_;
    return $self;
}

في هذا المثال، تم تنفيذ إضافة "Singleton". يتم إنشاء كائن واحد عند استدعاء load()، وهو يقوم بإرجاع نفسه لكل استدعاء لـ new().

يمكن أن يتطلب تطبيق آخر إنشاء كائن فردي لكل استدعاء لـ new()، ولكن مع مشاركة كل كائن مرجعًا لكائن آخر للاحتفاظ بالبيانات المخزنة مؤقتًا، واتصالات قاعدة البيانات، وما إلى ذلك. يوضح هذا المثال شبه الكود المبدأ.

package MiServidor;
sub load {
    my ($class, $context) = @_;
    bless {
        _CONTEXT => $context,
        _CACHE   => { },
    }, $class;
}
sub new {
    my ($self, $context, @params) = @_;
    MiCliente->new($self, @params);
}
sub add_to_cache   { ... }
sub get_from_cache { ... }
package MiCliente;
sub new {
    my ($class, $server, $blah) = @_;
    bless {
        _SERVER => $server,
        _BLAH   => $blah,
    }, $class;
}
sub get {
    my $self = shift;
    $self->{ _SERVER }->get_from_cache(@_);
}
sub put {
    my $self = shift;
    $self->{ _SERVER }->add_to_cache(@_);
}

عند تحميل الإضافة، يتم إنشاء مثيل "MiServidor". يتم استدعاء الطريقة new() مقابل هذا المثيل، والذي يقوم بإنشاء وإرجاع كائن "MiCliente" جاهز للتواصل مع منشئ هذه الإضافة.
