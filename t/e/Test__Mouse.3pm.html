Test__Mouse(3pm)

الحزمة: perl-mouse

التراخيص: PerlArtistic, GPL

Test::Mouse(3pm) توثيق Perl مساهمة من المستخدم Test::Mouse(3pm)

الاسم

Test::Mouse - وظائف اختبار لميزات Mouse الخاصة

الملخص

use Test::More plan => 1;
use Test::Mouse;
meta_ok($class_or_obj, "... Foo has a ->meta");
does_ok($class_or_obj, $role, "... Foo does the Baz role");
has_attribute_ok($class_or_obj, $attr_name, "... Foo has the 'bar' attribute");

الوصف

توفر هذه الوحدة وظائف اختبار مفيدة للفئات المستندة إلى Mouse. إنها إصدار أول تجريبي، لذا فإن التعليقات والاقتراحات مرحب بها جدًا.

الوظائف المصدرة

meta_ok ($class_or_object)
يختبر ما إذا كانت الفئة أو الكائن لديه فئة وصفية.
does_ok ($class_or_object, $role, ?$message)
يختبر ما إذا كانت الفئة أو الكائن تؤدي دورًا معينًا، على غرار ما تفعله "isa_ok" للطريقة "isa".
has_attribute_ok($class_or_object, $attr_name, ?$message)
يختبر ما إذا كانت الفئة أو الكائن لديه سمة معينة، على غرار ما تفعله "can_ok" للطرق.
with_immutable { CODE } @class_names
يقوم بتشغيل CODE (والذي يجب أن يحتوي على اختبارات عادية) مرتين، ويجعل كل فئة في @class_names غير قابلة للتغيير بين التشغيلين.
