tep_override_comm(3)

الحزمة: libtraceevent

التراخيص: GPL-2.0-only, LGPL-2.1-only

LIBTRACEEVENT(3) دليل libtraceevent LIBTRACEEVENT(3)

الاسم

tep_register_comm, tep_override_comm, tep_is_pid_registered, tep_data_comm_from_pid, tep_data_pid_from_comm, tep_cmdline_pid - إدارة تعيينات معرف العملية (pid) إلى اسم العملية.

الملخص

#include <event-parse.h>
int tep_register_comm(struct tep_handle *tep, const char *comm, int pid);
int tep_override_comm(struct tep_handle *tep, const char *comm, int pid);
bool tep_is_pid_registered(struct tep_handle *tep, int pid);
const char *tep_data_comm_from_pid(struct tep_handle *pevent, int pid);
struct cmdline *tep_data_pid_from_comm(struct tep_handle *pevent, const char *comm, struct cmdline *next);
int tep_cmdline_pid(struct tep_handle *pevent, struct cmdline *cmdline);

الوصف

يمكن استخدام هذه الدوال للتعامل مع تعيين معرف العملية (pid) إلى اسم العملية. يقوم المكتب ببناء ذاكرة تخزين مؤقت لهذه التعيينات، والتي تستخدم لعرض اسم العملية، بدلاً من معرفها. يمكن استرداد هذه المعلومات من ملف tracefs/saved_cmdlines.

تقوم الدالة tep_register_comm() بتسجيل تعيين pid / اسم عملية. إذا كانت هناك عملية بنفس pid مسجلة بالفعل، يتم إرجاع خطأ. الوسيط pid هو معرف العملية، الوسيط comm هو اسم العملية، tep هو سياق الحدث. يتم نسخ comm داخليًا.

تقوم الدالة tep_override_comm() بتسجيل تعيين pid / اسم عملية. إذا كانت هناك عملية بنفس معرف العملية مسجلة بالفعل، يتم تحديث سلسلة اسم العملية بواحدة جديدة. الوسيط pid هو معرف العملية، الوسيط comm هو اسم العملية، tep هو سياق الحدث. يتم نسخ comm داخليًا.

تقوم الدالة tep_is_pid_registered() بالتحقق مما إذا كان pid لديه تعيين اسم عملية مسجل. الوسيط pid هو معرف العملية، tep هو سياق الحدث.

تقوم الدالة tep_data_comm_from_pid() بإرجاع اسم العملية لمعرف عملية معين. الوسيط pid هو معرف العملية، tep هو سياق الحدث. لا ينبغي تحرير السلسلة المرجعة، ولكن سيتم تحريرها عند إغلاق المعالج tep.

تقوم الدالة tep_data_pid_from_comm() بإرجاع pid لاسم عملية معين. الوسيط comm هو اسم العملية، tep هو سياق الحدث. الوسيط next هو بنية cmdline للبحث عن pid التالي. نظرًا لأنه قد يكون هناك أكثر من pid واحد لاسم عملية معين، يمكن تمرير نتيجة هذا الاستدعاء مرة أخرى في استدعاء متكرر في المعلمة next، للبحث عن pid التالي. إذا كان next فارغًا (NULL)، فسيعيد أول pid مرتبط بـ comm. تقوم الدالة بإجراء بحث خطي، لذلك قد تكون بطيئة.

تقوم الدالة tep_cmdline_pid() بإرجاع pid المرتبط بـ cmdline معين. الوسيط tep هو سياق الحدث.

قيمة الإرجاع

تعيد الدالة tep_register_comm() القيمة 0 عند النجاح. في حالة حدوث خطأ، يتم إرجاع -1 ويتم تعيين errno للإشارة إلى سبب المشكلة: ENOMEM، إذا لم تكن هناك ذاكرة كافية لنسخ comm أو EEXIST إذا كان هناك تعيين لـ pid هذا مسجل بالفعل.

تعيد الدالة tep_override_comm() القيمة 0 عند النجاح. في حالة حدوث خطأ، يتم إرجاع -1 ويتم تعيين errno للإشارة إلى سبب المشكلة: ENOMEM، إذا لم تكن هناك ذاكرة كافية لنسخ comm.

تعيد الدالة tep_is_pid_registered() القيمة true إذا كان pid لديه اسم عملية معين له، و false خلاف ذلك.

تعيد الدالة tep_data_comm_from_pid() اسم العملية كسلسلة، أو السلسلة "<...>" إذا لم يكن هناك تعيين لـ pid المعطى.

تعيد الدالة tep_data_pid_from_comm() مؤشرًا إلى بنية cmdline، والتي تحتوي على pid لعملية معينة، أو NULL إذا لم يتم العثور على أي منها. يمكن تمرير هذه النتيجة مرة أخرى في استدعاء متكرر كمعلمة next للدالة.

تعيد الدالة tep_cmdline_pid() القيمة pid لـ cmdline المعطى. إذا كان cmdline فارغًا (NULL)، فسيتم إرجاع -1.

مثال

يسجل المثال التالي pid لأمر "ls"، في سياق الحدث tep ويقوم بإجراء عمليات بحث مختلفة لتعيينات pid / اسم العملية:

#include <event-parse.h>
...
int ret;
int ls_pid = 1021;
struct tep_handle *tep = tep_alloc();
...
        ret = tep_register_comm(tep, "ls", ls_pid);
        if (ret != 0 && errno == EEXIST)
                ret = tep_override_comm(tep, "ls", ls_pid);
        if (ret != 0) {
                /* فشل تسجيل تعيين pid / الأمر */
        }
...
        if (tep_is_pid_registered(tep, ls_pid) == 0) {
                /* تعيين الأمر لـ ls_pid غير مسجل */
        }
...
        const char *comm = tep_data_comm_from_pid(tep, ls_pid);
        if (comm) {
                /* تم العثور على اسم العملية لـ ls_pid */
        }
...
        int pid;
        struct cmdline *cmd = tep_data_pid_from_comm(tep, "ls", NULL);
        while (cmd) {
                pid = tep_cmdline_pid(tep, cmd);
                /* تم العثور على pid للعملية "ls" */
                cmd = tep_data_pid_from_comm(tep, "ls", cmd);
        }
