Test2__Tools__Process(3pm)

الحزمة: perl-test2-tools-process

التراخيص: PerlArtistic

Test2::Tools::Process(3) وثائق Perl مقدمة من المستخدم Test2::Tools::Process(3)

الاسم

Test2::Tools::Process - اختبارات الوحدة للكود الذي يستدعي exit، exec، system أو qx()

الإصدار

الإصدار 0.07

الملخص

use Test2::V0 -no_srand => 1;
use Test2::Tools::Process;

process {
  system 'foo', 'bar';
} [
  # تحقق من أن أول نظام تم استدعاؤه هو أمر foo مع أي وسيط
  proc_event(system => array {
    item 'foo';
    etc;
  }, sub {
    # محاكاة أمر foo
    my($proc, @args) = @_;
    note "faux bar command: @args";
    # محاكاة إنهاء طبيعي
    $proc->exit(0);
  }),
];

process {
  exit 2;
  note 'لا يتم التنفيذ';
} [
  # يمكن استخدام أي تحقق من Test2 على حالة الخروج
  proc_event(exit => match qr/^[2-3]$/),
];

process {
  exit 4;
} [
  # أو يمكنك التحقق من تطابق حالة الخروج رقميًا
  proc_event(exit => 4),
];

process {
  exit 5;
} [
  # أو ببساطة التحقق من استدعاء exit.
  proc_event('exit'),
];

process {
  exec 'foo bar';
  exec 'baz';
  note 'لا يتم التنفيذ';
} [
  # محاكاة أول exec كفاشل
  proc_event(exec => match qr/^foo\b/, sub {
    my($return, @command) = @_;
    $! = 2;
    return 0;
  }),
  # سيتم محاكاة ثاني exec بنجاح
  proc_event('exec'),
];

# اعتراض `exit`
is intercept_exit { exit 10 }, 10;

# اعتراض `exec`
is intercept_exec { exec 'foo', 'bar', 'baz' }, ['foo','bar','baz'];

done_testing;

الوصف

تُستخدم مجموعة أدوات الاختبار هذه لكتابة اختبارات الوحدة للكود الذي يتفاعل مع العمليات دون استخدام عمليات حقيقية قد يكون لها آثار جانبية غير مرغوبة. كما أنها تسمح باختبار الكود الذي يخرج من تدفق البرنامج دون إنهاء الاختبار فعليًا. حتى الآن، يسمح باختبار و/أو محاكاة "exit"، "exec"، "system"، "readpipe" و "qx//". ستتم إضافة اختبارات عمليات أخرى في المستقبل.

تحذف هذه الوحدة بعض الأفكار من Test::Exit. على وجه الخصوص، لا تستخدم استثناءات لمحاكاة "exit" أو "exec"، لذلك يمكنك اختبار الكود الذي يستدعيها في "eval" بحرية.

الدوال

process

my $ok = process { ... } \@events, $test_name;
my $ok = process { ... } \@events;
my $ok = process { ... } $test_name;
my $ok = process { ... };

ينفذ الكتلة، لاعتراض استدعاءات "exit"، "exec"، "system"، "readpipe" و "qx//". تتم مقارنة الاستدعاءات مع "\@events" كأحداث عملية متوقعة. انظر "proc_event" أدناه لتعريف الأحداث الفردية، والملخص أعلاه للأمثلة.

named_signal

my $signame = named_signal $name;

بالنظر إلى اسم إشارة مثل "KILL"، فإنها تعيد رقم الإشارة الصحيح. تثير استثناء إذا كان $name غير صالح.

intercept_exit

my $status = intercept_exit { ... };

يعترض أي استدعاء "exit" داخل الكتلة ويعيد قيمة حالة الخروج. يعيد "undef" إذا لم تكن هناك استدعاءات "exec".

intercept_exec

my $arrayref = intercept_exec { ... };

يعترض أي استدعاء "exec" داخل الكتلة ويعيد سطر الأوامر الذي تم تمريره إليه. يعيد "undef" إذا لم تكن هناك استدعاءات "exec".

التحققات

proc_event

process { ... } [
  proc_event($type => $check, $callback),
  proc_event($type => $check),
  proc_event($type => $callback),
  proc_event($type),

  # تحقق إضافية لأحداث النظام
  proc_event('system' => $check, \%result_check, $callback),
  proc_event('system' => \%result_check, $callback),
  proc_event('system' => $check, \%result_check),
  proc_event('system' => \%result_check),
];

تنشئ الدالة "proc_event" حدث عملية، مع تحقق واختياري ورد اتصال. كيف يعمل $check يعتمد على $type. إذا لم يتم توفير $check، فسيتحقق فقط من تطابق $type. نظرًا لطبيعتها، يتم محاكاة أحداث "exit" و "exec". تتم استدعاءات النظام الفعلية لأحداث "system"، إلا إذا تم توفير $callback.

exit
حدث عملية لاستدعاء "exit". يتم التحقق مقابل القيمة الحالة التي تم تمريرها إلى "exit". ستكون هذه القيمة دائمًا عددًا صحيحًا. إذا لم يتم تمرير أي قيمة حالة إلى "exit"، فسيتم استخدام 0 كقيمة الحالة.

إذا لم يتم توفير رد اتصال، فسيتم محاكاة استدعاء "exit" (ناجح) عن طريق إنهاء كتلة العملية دون تنفيذ المزيد من الكود. ستستمر بقية الاختبار.

proc_event( exit => sub {
  my($proc, $status) = @_;
  $proc->terminate;
});

يأخذ رد الاتصال كائن $proc وقيمة $status لاستدعاء "exit". بشكل طبيعي، لا تعيد "exit"، لذا ما تريد فعله هو استدعاء الطريقة "terminate" على الكائن $proc.

exec
حدث عملية لاستدعاء "exec". يتم التحقق مقابل الأمر الذي تم تمريره إلى "exec". إذا تم استدعاء "exec" بوسيط واحد، فسيكون سلسلة، وإلا فسيكون مرجع مصفوفة. بهذه الطريقة يمكنك التمييز بين وضعي SCALAR و LIST لـ "exec".

إذا لم يتم توفير رد اتصال، فسيتم محاكاة استدعاء "exec" (ناجح) عن طريق إنهاء كتلة العملية دون تنفيذ المزيد من الكود. ستستمر بقية الاختبار.

proc_event( exec => sub {
  my($proc, @command) = @_;
  ...;
});

يأخذ رد الاتصال كائن $proc والوسائط التي تم تمريرها إلى "exec" كـ @command. يمكنك محاكاة استدعاء "exec" فاشل باستخدام الطريقة "errno" للكائن $proc:

proc_event( exec => sub {
  my($proc, @command) = @_;
  $proc->errno(2); # هذه هي قيمة errno
});

لمحاكاة استدعاء "exec" ناجح، فقط تذكر استدعاء الطريقة "terminate" على الكائن $proc.

proc_event( exec => sub {
  my($proc, @command) = @_;
  $proc->terminate;
});
system
حدث عملية لـ "system"، "piperead" و "qx//". التحقق الأول (كما هو الحال مع "exec") هو مقابل سلسلة الأوامر التي تم تمريرها إلى "system". الثاني هو مرجع تجزئة مع تحقق من النتائج.
status
حدث عملية لاستدعاء "status". يتم التحقق مقابل القيمة المرجعة لـ "status". ستكون هذه القيمة دائمًا عددًا صحيحًا. إذا لم يتم تمرير أي قيمة مرجعة، فسيتم استخدام 0 كقيمة مرجعة.
