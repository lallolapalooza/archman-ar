Test__Requires(3pm)

الحزمة: perl-test-requires

التراخيص: PerlArtistic, GPL

Test::Requires(3) وثائق Perl المساهمة من قبل المستخدم Test::Requires(3)

الاسم

Test::Requires - التحقق مما إذا كان يمكن تحميل الوحدة

الملخص

# في Makefile.PL الخاص بك
use inc::Module::Install;
test_requires 'Test::Requires';
# في الاختبار الخاص بك
use Test::More tests => 10;
use Test::Requires {
    'HTTP::MobileAttribute' => 0.01, # تخطي الكل إذا لم يتم تثبيت HTTP::MobileAttribute
};
isa_ok HTTP::MobileAttribute->new, 'HTTP::MobileAttribute::NonMobile';
# أو
use Test::More tests => 10;
use Test::Requires qw( 
    HTTP::MobileAttribute
);
isa_ok HTTP::MobileAttribute->new, 'HTTP::MobileAttribute::NonMobile';
# أو
use Test::More tests => 10;
use Test::Requires;
test_requires 'Some::Optional::Test::Required::Modules';
isa_ok HTTP::MobileAttribute->new, 'HTTP::MobileAttribute::NonMobile';

الوصف

تتحقق Test::Requires مما إذا كان يمكن تحميل الوحدة.

إذا فشل هذا بدلاً من فشل الاختبارات، فإنه **يتخطى جميع الاختبارات**.

يمكن استخدام Test::Requires أيضًا لطلب حد أدنى من إصدار Perl:

use Test::Requires "5.010";  # الاقتباس ضروري!!

# أو
use Test::Requires "v5.10";
