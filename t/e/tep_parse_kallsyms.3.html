tep_parse_kallsyms(3)

الحزمة: libtraceevent

التراخيص: GPL-2.0-only, LGPL-2.1-only

LIBTRACEEVENT(3) دليل libtraceevent LIBTRACEEVENT(3)

الاسم

tep_parse_saved_cmdlines, tep_parse_printk_formats, tep_parse_kallsyms - وظائف تحليل لتحميل التعيينات

الملخص

#include <event-parse.h>
int tep_parse_saved_cmdlines(struct tep_handle *tep, const char *buf);
int tep_parse_printk_formats(struct tep_handle *tep, const char *buf);
int tep_parse_kallsyms(struct tep_handle *tep, const char *buf);

الوصف

tep_parse_saved_cmdlines() هي وظيفة مساعدة لتحليل المحتوى في نظام ملفات tracefs الخاص بملف "saved_cmdlines" (مخزن في مخزن مؤقت للسلسلة يتم تمريره عبر buf) وتحميل تعيين معرفات العمليات (pid) إلى أسماء الأوامر (comm names) في معالج tep. تخزن الأحداث معرف العملية (pid) ويستخدم هذا لعرض أسماء العمليات المرتبطة بمعرفات العمليات هذه. تقوم بتحليل السلسلة buf التي تحتوي على محتوى saved_cmdlines وتنتهي بحرف فارغ (\0).

tep_parse_printk_formats() هي وظيفة مساعدة لتحليل المحتوى في نظام ملفات tracefs الخاص بملف "printk_formats" (مخزن في مخزن مؤقت للسلسلة يتم تمريره عبر buf) وتحميل تعيين العناوين للسلاسل التي قد يتم الرجوع إليها بواسطة الأحداث. تخزن الأحداث عنوان السلاسل الثابتة فقط في النواة، ويتم تصدير تعيين عناوينها إلى مساحة المستخدم في ملف printk_formats. تقوم بتحليل السلسلة buf التي تحتوي على محتوى printk_formats وتنتهي بحرف فارغ (\0).

tep_parse_kallsyms() هي وظيفة مساعدة لتحليل تنسيق Linux kernel /proc/kallsyms (مخزن في مخزن مؤقت للسلسلة يتم تمريره عبر buf) وتحميل الوظائف في معالج tep بحيث يمكن تعيين عناوين IP للوظائف إلى أسمائها عند تحليل الأحداث باستخدام %pS في حقل تنسيق الطباعة. تقوم بتحليل السلسلة buf التي تحتوي على محتوى /proc/kallsyms وتنتهي بحرف فارغ (\0).

قيمة الإرجاع

ترجع الوظيفة tep_parse_saved_cmdlines() القيمة 0 في حالة النجاح، أو -1 في حالة حدوث خطأ.

ترجع الوظيفة tep_parse_printk_formats() القيمة 0 في حالة النجاح، أو -1 في حالة حدوث خطأ.

ترجع الوظيفة tep_parse_kallsyms() القيمة 0 في حالة النجاح، أو -1 في حالة حدوث خطأ.

مثال

...
#include <event-parse.h>
#include <tracefs.h>
#include <stdlib.h>
int load_cmdlines(struct tep_handle *tep)
{
        char *buf = NULL;
        int r;
        buf = tracefs_instance_file_read(NULL, "saved_cmdlines", NULL);
        if (!buf)
                return -1;
        r = tep_parse_saved_cmdlines(tep, buf);
        free(buf);
        return r;
}
int load_print_strings(struct tep_handle *tep)
{
        char *buf = NULL;
        int r;
        buf = tracefs_instance_file_read(NULL, "printk_formats", NULL);
        if (!buf)
                return -1;
        r = tep_parse_printk_formats(tep, buf);
        free(buf);
        return r;
}
int load_kallsyms(struct tep_handle *tep)
{
        char *line = NULL;
        char *buf = NULL;
        size_t sz = 0;
        FILE *fp;
        int len = 0;
        int r;
        fp = fopen("/proc/kallsyms", "r");
        while ((r = getline(&line, &sz, fp)) >= 0) {
                buf = realloc(buf, len + r + 1);
                memcpy(buf + len, line, r);
                len += r;
        }
        free(line);
        fclose(fp);
        if (!buf)
                return -1;
        buf[len] = 0;
        r = tep_parse_kallsyms(tep, buf);
        free(buf);
        return r;
}
