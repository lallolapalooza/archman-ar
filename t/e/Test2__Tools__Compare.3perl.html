Test2::Tools::Compare(3perl) دليل مبرمجي Perl Test2::Tools::Compare(3perl)

الاسم

Test2::Tools::Compare - أدوات لمقارنة هياكل البيانات العميقة.

الوصف

كان لدى Test::More الدالة is_deeply(). هذه المكتبة هي إصدار Test2 التي يمكن استخدامها لمقارنة هياكل البيانات، ولكنها تذهب أبعد من ذلك حيث توفر أدوات لبناء مواصفات لهياكل البيانات التي يمكنك من خلالها التحقق من بياناتك. هناك نسخ 'صارمة' و 'مرنة' للأدوات.

الملخص

use Test2::Tools::Compare;
# Hash لأغراض العرض
my $some_hash = {a => 1, b => 2, c => 3};
# فحص صارم، يجب أن يتطابق كل شيء
is(
    $some_hash,
    {a => 1, b => 2, c => 3},
    "الـ hash الذي حصلنا عليه يطابق توقعاتنا"
);
# فحص مرن، يتم التحقق فقط من الحقول التي نهتم بها، ويمكننا استخدام
# تعبير عادي لتقريب حقل.
like(
    $some_hash,
    {a => 1, b => qr/\A[0-9]+\z/},
    "'a' هو 1، 'b' هو عدد صحيح، لا نهتم بـ 'c'."
);

متقدم

تتوفر أدوات بناء Hash، و Array، والكائنات التعريفية التي تسمح لك بإنشاء مواصفات. هذه الأدوات أكثر تفصيلاً من مجرد توفير Hash، ولكنها تتميز بأن كل مكون تحدده له رقم سطر مرتبط به. هذا مفيد لتصحيح الأخطاء حيث أن إخراج الفشل سيخبرك ليس فقط بالحقل الذي كان غير صحيح، ولكن أيضًا بالسطر الذي أعلنت فيه عن الحقل.

use Test2::Tools::Compare qw{
    is like isnt unlike
    match mismatch validator
    hash array bag object meta number float rounded within string subset bool
    in_set not_in_set check_set
    item field call call_list call_hash prop check all_items all_keys all_vals all_values
    etc end filter_items
    T F D DF E DNE FDNE U L
    event fail_events
    exact_ref
};
is(
    $some_hash,
    hash {
        field a => 1;
        field b => 2;
        field c => 3;
    },
    "Hash يطابق المواصفات"
);

أدوات المقارنة

$bool = is($got, $expect)
$bool = is($got, $expect, $name)
$bool = is($got, $expect, $name, @diag)
$got هو هيكل البيانات الذي تريد التحقق منه. $expect هو ما تريد أن يبدو عليه $got. $name هو اسم اختياري للاختبار. @diag هي رسائل تشخيصية اختيارية سيتم طباعتها إلى STDERR في حالة الفشل، ولن يتم عرضها عند نجاح المقارنة. يتم إرجاع نتيجة المقارنة المنطقية صح/خطأ.

هذا هو المدقق الصارم. يتطلب المدقق الصارم تطابقًا مثاليًا بين $got و $expect. يجب تحديد جميع حقول الـ hash، ووجود جميع عناصر الـ array، وما إلى ذلك. يجب أن تكون جميع المراجع غير العددية/الـ hash/الـ array/التعبيرات العادية متطابقة (نفس عنوان الذاكرة). سيتم اجتياز مراجع العدديات والـ hash والـ array ومقارنتها. سيتم مقارنة مراجع التعبيرات العادية لمعرفة ما إذا كانت تحتوي على نفس النمط.

is(
    $some_hash,
    {a => 1, b => 2, c => 3},
    "الـ hash الذي حصلنا عليه يطابق توقعاتنا"
);

الاستثناء الوحيد للصرامة هو عند إعطائها كائن $expect تم بناؤه من مواصفات، وفي هذه الحالة تحدد المواصفات الصرامة. تنطبق الصرامة فقط على القيم/المراجع الحرفية المقدمة وتحويلها إلى مواصفات لك.

is(
    $some_hash,
    hash {    # ملاحظة: الدالة hash غير مصدّرة افتراضيًا
        field a => 1;
        field b => match(qr/\A[0-9]+\z/);    # ملاحظة: الدالة match غير مصدّرة افتراضيًا
        # لا نهتم بالحقول الأخرى.
    },
    "مقارنة الـ hash ليست صارمة"
);

يعمل هذا مع الهياكل العميقة والسطحية على حد سواء. على سبيل المثال، يمكنك استخدام هذا لمقارنة سلسلتين نصيتين:

is('foo', 'foo', "السلاسل النصية متطابقة");

ملاحظة: هذه ليست الأداة المستخدمة إذا كنت تريد التحقق مما إذا كان مرجعان متطابقين تمامًا، استخدم ref_is() من المكون الإضافي Test2::Tools::Ref بدلاً من ذلك. في معظم الأوقات، سيعمل هذا أيضًا، ومع ذلك توجد مشاكل إذا كان مرجعك يحتوي على دورة ويشير إلى نفسه في مرحلة ما. إذا حدث هذا، سيتم طرح استثناء لكسر تكرار لا نهائي.

ملاحظة: سيتم مقارنة القيم غير المرجعية كسلاسل نصية باستخدام "eq"، وهذا يعني أن السلاسل النصية '2.0' و '2' لن تتطابق، ولكن الأرقام 2.0 و 2 ستتطابق، حيث يتم تحويلهما إلى سلاسل نصية '2'.

$bool = isnt($got, $expect)
$bool = isnt($got, $expect, $name)
$bool = isnt($got, $expect, $name, @diag)
عكس is(). يقوم بجميع نفس عمليات التحقق، ولكنه ينجح عند وجود عدم تطابق.
$bool = like($got, $expect)
$bool = like($got, $expect, $name)
$bool = like($got, $expect, $name, @diag)
$got هو هيكل البيانات الذي تريد التحقق منه. $expect هو ما تريد أن يبدو عليه $got. $name هو اسم اختياري للاختبار. @diag هي رسائل تشخيصية اختيارية سيتم طباعتها إلى STDERR في حالة الفشل، ولن يتم عرضها عند نجاح المقارنة. يتم إرجاع نتيجة المقارنة المنطقية صح/خطأ.

هذا هو المدقق المرن. سيتجاهل مفاتيح الـ hash أو مؤشرات الـ array التي لا تحددها بالفعل في هيكل $expect الخاص بك. بالإضافة إلى ذلك، سيتم استخدام التعبيرات العادية والمراجع الفرعية كمدققين. إذا قمت بتوفير تعبير عادي باستخدام "qr/.../"، فسيتم استخدام التعبير العادي نفسه للتحقق من القيمة المقابلة في هيكل $got. وينطبق الشيء نفسه على مراجع الكود، حيث يتم تمرير القيمة كوسيط أول (وفي $_) ويجب أن يعيد الفرع قيمة منطقية. في هذه الأداة، تقوم التعبيرات العادية بتحويل الشيء الذي تتحقق منه إلى سلسلة نصية.

like(
    $some_hash,
    {a => 1, b => qr/\A[0-9]+\z/},
    "'a' هو 1، 'b' هو عدد صحيح، لا نهتم بالحقول الأخرى"
);

يعمل هذا مع الهياكل العميقة والسطحية على حد سواء. على سبيل المثال، يمكنك استخدام هذا لمقارنة سلسلتين نصيتين:

like('foo bar', qr/^foo/, "السلسلة النصية تطابق النمط");
$bool = unlike($got, $expect)
$bool = unlike($got, $expect, $name)
$bool = unlike($got, $expect, $name, @diag)
عكس like(). يقوم بجميع نفس عمليات التحقق، ولكنه ينجح عند وجود عدم تطابق.

يمكن جعل الدوال is()، isnt()، like()، و unlike() تطبع $got باستخدام Data::Dumper عند فشل الاختبارات عن طريق تعيين متغير البيئة "T2_AUTO_DUMP" إلى "1". (بدلاً من ذلك، يمكن تعيين "T2_AUTO_DUMP" إلى اسم وحدة Perl توفر طريقة Dump() متوافقة.) يمكن استخدام متغير البيئة "T2_AUTO_DEPARSE" لتمكين إلغاء التحليل الخاص بـ Data::Dumper لمراجع الكود.

فحوصات سريعة

ملاحظة: لا يتم تصدير أي من هذه افتراضيًا. تحتاج إلى طلبها.

الفحوصات السريعة هي طريقة لإنشاء مواصفات قيم شائعة بسرعة. يمكن استخدام هذه في الهياكل التي يتم تمريرها إلى "is" و "like" من خلال وسيط $expect.

مثال:

is($foo, T(), '$foo لديه قيمة صحيحة');
$check = T()
يتحقق هذا من أن القيمة في هيكل $got المقابل صحيحة، أي قيمة صحيحة ستفي بالغرض. 
is($foo, T(), '$foo لديه قيمة صحيحة');
is(
    { a => 'xxx' },
    { a => T() },
    "المفتاح 'a' صحيح"
);
$check = F()
يتحقق هذا من أن القيمة في هيكل $got المقابل خاطئة، أي قيمة خاطئة ستفي بالغرض، ولكن يجب أن تكون القيمة موجودة. 
is($foo, F(), '$foo لديه قيمة خاطئة');

    { a => 0

أداة إنشاء الـ HASH

ملاحظة: لا يتم تصدير أي من هذه بشكل افتراضي. يجب عليك طلبها.

$check = hash {
    field foo => 1;
    field bar => 2;
    # تأكد من أن الفهرس 'baz' غير موجود في الـ hash.
    field baz => DNE();
    # تأكد من وجود المفتاح ولكنه تم تعيينه كـ undef
    field bat => undef;
    # يمكن استخدام أي فحص
    field boo => $check;
    # قم بتعيين فحوصات تنطبق على جميع المفاتيح أو القيم.
    all_vals match qr/a/, match qr/b/;    # يجب أن تحتوي جميع القيم على 'a' و 'b'
    all_keys match qr/x/;                 # يجب أن تحتوي جميع الفهارس على 'x'
    ...
    end(); # اختياري، يضمن عدم وجود فهارس أخرى.
};
$check = hash { ... }
يُستخدم هذا لتعريف فحص للـ hash.
field $NAME => $VAL
field $NAME => $CHECK
يعرّف فحص حقل. سيتحقق هذا من مفتاح الـ hash المحدد بواسطة $NAME ويضمن مطابقته للقيمة في $VAL. يمكنك تضمين أي فحص صالح في $VAL، مثل نتيجة استدعاء آخر لـ "array { ... }"، DNE()، إلخ.

ملاحظة: لا يمكن استخدام هذه الوظيفة إلا داخل أداة إنشاء فرعية للـ hash ويجب استدعاؤها في سياق void.

all_keys($CHECK1, $CHECK2, ...)
يضيف فحوصات تنطبق على جميع المفاتيح. يمكنك وضع هذا في أي مكان داخل كتلة الـ hash ويمكن استدعاؤه أي عدد من المرات مع أي عدد من الوسائط.
all_vals($CHECK1, $CHECK2, ...)
all_values($CHECK1, $CHECK2, ...)
يضيف فحوصات تنطبق على جميع القيم. يمكنك وضع هذا في أي مكان داخل كتلة الـ hash ويمكن استدعاؤه أي عدد من المرات مع أي عدد من الوسائط.
end()
يضمن عدم وجود فهارس فريدة في الـ hash، بخلاف تلك المحددة. هذا هو أساسًا "use strict" لفحص الـ hash. يمكنك استخدامه في أي مكان داخل أداة الإنشاء الفرعية للـ hash، على الرغم من أنه يتم وضعه عادةً في النهاية.
etc()
يتجاهل أي فهارس إضافية تم العثور عليها في الـ hash. هذا هو عكس end(). يمكنك استخدامه في أي مكان داخل أداة الإنشاء الفرعية للـ hash، على الرغم من أنه يتم وضعه عادةً في النهاية.
DNE()
هذا فحص مفيد يمكن استخدامه مع field() لضمان عدم وجود مفتاح (D) (N) موجود (E). 
field foo => DNE();

أداة إنشاء الـ ARRAY

ملاحظة: لا يتم تصدير أي من هذه بشكل افتراضي. يجب عليك طلبها.

$check = array {
    # استخدم الفهرس التالي، في هذه الحالة الفهرس 0؛
    item 'a';
    # احصل على الفهرس 1 تلقائيًا
    item 'b';
    # حدد الفهرس
    item 2 => 'c';
    # نتجاهل الفهرس 3، مما يعني أنه لا يهم ما هو.
    item 4 => 'e';
    # احصل على الفهرس 5.
    item 'f';
    # قم بتصفية أي عناصر متبقية تحتوي على 0-9.
    filter_items { grep {!m/[0-9]/} @_ };
    # فحوصات تنطبق على جميع العناصر. يمكنك استخدامها في أي مكان داخل كتلة الـ array ويمكن استدعاؤها أي عدد من المرات مع أي عدد من الوسائط.
    all_items match qr/a/, match qr/b/;
    all_items match qr/x/;
    # من العناصر المتبقية (بعد التصفية) العنصر التالي (الآن الفهرس 6) يجب أن يكون 'g'.
    item 6 => 'g';
    item 7 => DNE; # تأكد من عدم وجود الفهرس 7.
    end(); # تأكد من عدم وجود فهارس إضافية.
};
$check = array { ... }
item $VAL
item $CHECK
item $IDX, $VAL
item $IDX, $CHECK
يضيف عنصرًا متوقعًا إلى الـ array. إذا لم يتم تحديد $IDX، فسيتم حسابه تلقائيًا بناءً على آخر عنصر تم إضافته. يمكنك تخطي الفهارس، مما يعني أنها لن يتم التحقق منها.

يمكنك توفير أي قيمة للتحقق منها في $VAL، أو يمكنك توفير أي كائن فحص صالح.

ملاحظة: يجب إضافة العناصر بالترتيب.

ملاحظة: لا يمكن استخدام هذه الوظيفة إلا داخل أداة إنشاء فرعية للـ array، أو bag، أو subset ويجب استدعاؤها في سياق void.

filter_items { my @remaining = @_; ...; return @filtered }

ملاحظة: لا يمكن استخدام هذه الوظيفة إلا داخل أداة إنشاء فرعية للـ array، أو bag، أو subset ويجب استدعاؤها في سياق void.

يضيف فحوصات تنطبق على جميع العناصر. يمكنك وضع هذا في أي مكان داخل كتلة الـ array ويمكن استدعاؤه أي عدد من المرات مع أي عدد من الوسائط.
