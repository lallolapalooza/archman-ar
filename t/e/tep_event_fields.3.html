tep_event_fields(3)

الحزمة: libtraceevent

التراخيص: GPL-2.0-only, LGPL-2.1-only

LIBTRACEEVENT(3) دليل libtraceevent LIBTRACEEVENT(3)

الاسم

tep_event_common_fields, tep_event_fields - الحصول على قائمة الحقول لحدث.

ملخص

#include <event-parse.h>
struct tep_format_field **tep_event_common_fields(struct tep_event *event);
struct tep_format_field **tep_event_fields(struct tep_event *event);

الوصف

تقوم الدالة tep_event_common_fields() بإرجاع مصفوفة من المؤشرات إلى الحقول الشائعة للحدث event. يتم تخصيص المصفوفة داخل الدالة ويجب تحريرها باستخدام free(). العنصر الأخير في المصفوفة هو NULL.

تقوم الدالة tep_event_fields() بإرجاع مصفوفة من المؤشرات إلى الحقول الخاصة بالحدث event. يتم تخصيص المصفوفة داخل الدالة ويجب تحريرها باستخدام free(). العنصر الأخير في المصفوفة هو NULL.

قيمة الإرجاع

ترجع كلتا الدالتين tep_event_common_fields() و tep_event_fields() مصفوفة من المؤشرات إلى هياكل tep_format_field في حالة النجاح، أو NULL في حالة حدوث خطأ.

مثال

#include <event-parse.h>
...
struct tep_handle *tep = tep_alloc();
...
int i;
struct tep_format_field **fields;
struct tep_event *event = tep_find_event_by_name(tep, "kvm", "kvm_exit");
if (event != NULL) {
        fields = tep_event_common_fields(event);
        if (fields != NULL) {
                i = 0;
                while (fields[i]) {
                        /*
                          المشي عبر قائمة الحقول المشتركة
                          لحدث kvm_exit
                        */
                        i++;
                }
                free(fields);
        }
        fields = tep_event_fields(event);
        if (fields != NULL) {
                i = 0;
                while (fields[i]) {
                        /*
                          المشي عبر قائمة الحقول الخاصة
                          بالحدث kvm_exit
                        */
                        i++;
                }
                free(fields);
        }
}
...
