الحزمة: perl-template-gd

Template::Plugin::GD::Graph::bars3d(3) توثيق مساهمة المستخدم لـ Perl Template::Plugin::GD::Graph::bars3d(3)

الاسم

Template::Plugin::GD::Graph::bars3d - إنشاء رسوم بيانية شريطية ثلاثية الأبعاد مع محاور ومفتاح

الملخص

[% USE g = GD.Graph.bars3d(x_size, y_size); %]

أمثلة

[% FILTER null;
    data = [
        ["Jan", "Feb", "Mar", "Apr", "May", "Jun", "Jul", "Aug",
                                     "Sep", "Oct", "Nov", "Dec", ],
        [-5, -4, -3, -3, -1,  0,  2,  1,  3,  4,  6,  7],
        [4,   3,  5,  6,  3,1.5, -1, -3, -4, -6, -7, -8],
        [1,   2,  2,  3,  4,  3,  1, -1,  0,  2,  3,  2],
    ];

    USE my_graph = GD.Graph.bars3d();

    my_graph.set( 
            x_label => 'Month',
            y_label => 'Measure of success',
            title => 'A 3d Bar Chart',
            y_max_value => 8,
            y_min_value => -8,
            y_tick_number => 16,
            y_label_skip => 2,
            box_axis => 0,
            line_width => 3,
            zero_axis_only => 1,
            x_label_position => 1,  
            y_label_position => 1,  

            x_label_skip => 3,
            x_tick_offset => 2,

            transparent => 0,
    );      
    my_graph.set_legend("Us", "Them", "Others");
    my_graph.plot(data).png | stdout(1);
   END;
-%]

الوصف

توفر الإضافة GD.Graph.bars3d واجهة لفئة GD::Graph::bars3d المعرفة بواسطة الوحدة GD::Graph3d. تسمح برسم مجموعة بيانات واحدة أو أكثر (x، y) كأشرطة y مقابل x بمظهر ثلاثي الأبعاد، جنبًا إلى جنب مع المحاور والمفاتيح.

راجع GD::Graph3d لمزيد من التفاصيل.
