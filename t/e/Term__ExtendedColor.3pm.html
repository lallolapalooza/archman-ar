Term__ExtendedColor(3pm)

الحزمة: perl-term-extendedcolor

التراخيص: PerlArtistic, GPL

Term::ExtendedColor(3) وثائق Perl من مساهمة المستخدم Term::ExtendedColor(3)

الاسم

Term::ExtendedColor - تلوين إخراج الشاشة باستخدام 256 لونًا

الملخص

use Term::ExtendedColor qw(:all);
# أو استخدم الوسم 'attributes' لاستيراد الوظائف لضبط
# السمات فقط.
# سيؤدي هذا إلى استيراد الوظائف التالية:
# fg(), bg(), bold(), underline(), inverse(), italic(), clear()
use Term::ExtendedColor ':attributes';
## ألوان المقدمة
my $red_text = fg('red2', 'هذا باللون الأحمر');
my $spring   = fg('springgreen3', 'هذا أخضر');
## ألوان الخلفية
print bg('red5', "نص المقدمة الافتراضي على خلفية حمراء داكنة"), "\n";
my $red_on_green = fg('red3', bg('green12', 'نص أحمر على خلفية خضراء'));
print "$red_on_green\n";
## سمات السقوط
Term::ExtendedColor::autoreset(0);
my $bold  = fg('bold', 'هذا غامق');
my $red   = fg('red2', 'هذا أحمر... وغامق');
my $green = bg('green28', 'هذا أحمر غامق على خلفية خضراء');
# تأكد من مسح جميع السمات عندما يكون autoreset متوقفًا،
# وإلا فإن واجهة المستخدم الخاصة بالمستخدم ستتعطل
my $clear = clear();
print "$bold\n";
print "$red\n";
print "$green $clear\n";
## سمات غير ملونة
# قم بتشغيل autoreset مرة أخرى
Term::ExtendedColor::autoreset(1);
for(qw(italic underline blink reverse bold)) {
  print fg($_, $_), "\n";
}
# للراحة
my $bolded = bold("نص غامق!");
my $italic = italic("نص مائل!");
## إزالة جميع السمات من بيانات الإدخال
my @colored;
push(@colored, fg('bold', fg('red2', 'غامق وأحمر')));
push(@colored, fg('green13', fg('italic', 'أخضر، مائل')));
print "$_\n" for @colored;
print "$_\n" for uncolor(@colored);

الوصف

توفر Term::ExtendedColor وظائف لإرسال ما يسمى بتسلسلات الهروب الموسعة إلى الطرفية. يجب استخدام هذا مع طرفية متوافقة مع 256 لونًا؛ راجع قسم ملاحظات للحصول على مصفوفة من محاكيات الطرفية التي تدعم هذا حاليًا.

التصدير

لا شيء افتراضيًا.

يتم توفير وسمين للراحة:

# استيراد جميع الوظائف
use Term::ExtendedColor qw(:all);
# استيراد وظائف لضبط السمات
# fg(), bg(), bold(), italic(), underline(), inverse(), clear()
use Term::ExtendedColor qw(:attributes);

الوظائف

fg($color, $string)

my $green = fg('green2', 'مقدمة خضراء');
my @blue  = fg('blue4',  ['يمكنها أخذ مراجع المصفوفات أيضًا']);
my $x_color = fg('mediumorchid1', 'استخدام تعيينات من rgb.txt الخاصة بـ X11');
my $arbitary_color = fg(4, 'هذا ملون باللون الخامس من ألوان ANSI');
my $raw_seq = fg('38;5;197;48;5;53;1;3;4;5;7', 'هذا يعمل أيضًا');

ضبط ألوان وسمات المقدمة.

انظر "الألوان والسمات" للحجج الأولى الصالحة. بالإضافة إلى ذلك، يمكن تحديد الألوان باستخدام قيم فهرسها:

my $yellow = fg(220, 'أصفر');

إذا كانت المتغير الداخلي $AUTORESET غير صفري (افتراضي)، فسيتم تغليف كل عنصر في قائمة السلاسل النصية بتسلسلات الهروب بطريقة سيتم فيها تعيين السمات المطلوبة قبل السلسلة النصية وإعادة تعيينها إلى القيم الافتراضية بعد السلسلة النصية.

يمكن تمكين سمات السقوط عن طريق تعيين $AUTORESET إلى قيمة خاطئة.

Term::ExtendedColor::autoreset( 0 );
my $red   = fg('red1', 'أحمر');
my $green = fg('green1', 'أخضر');
print "النص بعد $red أحمر حتى $green\n";
print "النص لا يزال أخضر، ", bold('والآن غامق أيضًا!');
# إذا خرجت الآن دون إعادة تعيين الألوان والسمات، فمن المحتمل أن
# يتم تعطيل المطالبة الخاصة بك.
clear(); # كل شيء عاد إلى طبيعته

إذا تم تمرير سمة غير صالحة، فسيتم إرجاع البيانات الأصلية دون تعديل.

إذا لم يتم تمرير أي سمة، فسيتم طرح استثناء.

bg($color, $string)

my $green_bg = bg('green4', 'خلفية خضراء');
my @blue_bg  = bg('blue6',  ['خلفية زرقاء']);

مثل fg()، لكنه يضبط لون الخلفية.

uncolor($string) | uncolour($string)

my $stripped = uncolor($colored_data);
my @no_color = uncolor(\@colored);
my @no_color = uncolor(@colored);

إزالة جميع سمات السمات وتسلسلات الهروب اللونية من الإدخال.

انظر uncolor للحصول على أداة سطر الأوامر التي تستخدم هذه الوظيفة.

get_colors() | get_colours()

my $colors = get_colors();

ترجع مرجع تجزئة بجميع السمات والألوان المتاحة.

clear()

عند استدعائها في سياق قياسي، ترجع تسلسل الهروب الذي يعيد تعيين جميع السمات إلى قيمها الافتراضية.

عند استدعائها في سياق فارغ، تطبعها مباشرة.

autoreset()

تشغيل/إيقاف إعادة التعيين التلقائي. مفعل افتراضيًا.

Term::ExtendedColor::autoreset( 0 ); # إيقاف إعادة التعيين التلقائي

bold(\@data)

وظيفة مساعدة يمكن استخدامها بدلاً من "fg('bold', \@data)"؛

عند استدعائها بدون وسائط، ترجع سلسلة نصية تقوم بإيقاف سمة الغامق.

italic(\@data)

وظيفة مساعدة يمكن استخدامها بدلاً من "fg('italic', \@data)"؛

عند استدعائها بدون وسائط، ترجع سلسلة نصية تقوم بإيقاف سمة المائل.

underline(\@data)

وظيفة مساعدة يمكن استخدامها بدلاً من "fg('underline', \@data)"؛

عند استدعائها بدون وسائط، ترجع سلسلة نصية تقوم بإيقاف سمة الخط السفلي.

inverse(\@data)

عكس الفيديو / معكوس. وظيفة مساعدة يمكن استخدامها بدلاً من "fg('inverse', \@data)"؛

عند استدعائها بدون وسائط، ترجع سلسلة نصية تقوم بإيقاف سمة المعكوس.

ملاحظات

تلتزم الرموز التي تنشئها هذه الوحدة بتوسيع معيار ألوان ANSI الذي تم تطبيقه لأول مرة في xterm في عام 1999. الفهارس الـ 16 الأولى من الألوان (0 - 15) هي ألوان ANSI العادية، بينما الفهرس 16 - 255 هو الامتداد. لا تدعم جميع محاكيات الطرفيات هذا الامتداد، على الرغم من أنني واجهت صعوبة في العثور على واحدة لا تفعل ذلك. :)

لا توجد طريقة لإعطاء هذه الألوان الموسعة أسماء ذات معنى.

كان تفكيرنا الأول هو مطابقتها مع بعض أسماء الألوان القياسية، مثل تلك الموجودة في مواصفات HTML 4.0 أو SVG. لم تتطابق.

لذلك، تم تسميتها حسب لونها الأساسي (أحمر، أخضر، أرجواني) بالإضافة إلى الفهرس؛ الفهرس الأول (دائمًا 1) هو ألمع ظل لهذا اللون المحدد، بينما الفهرس الأخير هو الأغمق.

من الممكن أيضًا استخدام بعض أسماء ألوان X، كما هو محدد في "rgb.txt". لاحظ أن قيم الألوان لا تتطابق تمامًا؛ إنها مجرد تقريب.

يمكن استرداد قائمة كاملة بالألوان المتاحة باستخدام get_colors(). انظر "الألوان والسمات" للحصول على القائمة الكاملة. يمكن أيضًا استرداد جميع الألوان المعينة برمجيًا باستخدام get_colors().

الألوان والسمات

NAME

Term::ExtendedColor - colored terminal output, with extended color support

SYNOPSIS

    use Term::ExtendedColor qw(:all);

    print fg('red'), "This is red text\n";
    print bg('blue'), "This has a blue background\n";
    print style('bold'), "This is bold\n";
    print fg('red'), bg('yellow'), style('bold'), "Bold red on yellow\n";

    # Using 256 colors
    print fg(208), "This is orange text\n"; # 256 color index 208
    print bg(12), "This has a dark blue background\n"; # 256 color index 12

    # Using the standard color names (if supported by terminal)
    print fg('darkcyan'), "This is dark cyan text\n";

    # Using X11 color names (if supported by terminal)
    print fg('aquamarine3'), "This is aquamarine text\n";

    # Combine functions
    print fg(fg('red'), style('bold')), "Bold red text\n";
    print fg(style('bold'), 'red'), "Bold red text\n";

    # Combine multiple styles and colors
    print fg(bg('red'), style('bold', 'italic')), "Bold, italic red text on default background\n";

    # Reset attributes
    print reset(), "Back to default\n";
    print fg('red'), "Red text", reset(), "and then default\n";

    # Print with attributes
    print style('bold', 'underline'), "Bold and underlined\n";
    print style('bold', 'italic', 'underline'), "Bold, italic, and underlined\n";

    # Reset specific attributes
    print style('bold'), "Bold text ", fg('red'), "red text", reset(qw(bold)), " normal text\n";

    # Set background and foreground using ANSI escape codes directly
    # SGR 0;38;5;197;48;5;89;1;3;4;7 m
    print "\e[38;5;197;48;5;89;1;3;4;7mThis text is foreground 197, background 89, bold, italic, underline, reverse\e[0m\n";

    # Using the function
    my $foo = fg('48;5;89;38;5;197;1;3;4;7', 'foo');
    print $foo;

DESCRIPTION

Term::ExtendedColor is a Perl module that provides functions to generate ANSI escape sequences for colored and styled terminal output. It supports basic ANSI colors, extended 256-color palettes, and standard X11 color names.

EXPORTS

The following functions are exported by default:

    fg, bg, style, reset

The following tags are available for import:

    :all    Imports all of the above.
