Template__View(3pm)

الحزمة: perl-template-toolkit

التراخيص: PerlArtistic

Template::View(3) توثيق مساهمة المستخدم في Perl Template::View(3)

الاسم

Template::View - عرض مخصص لسياق معالجة القالب

الملخص

# تعريف عرض
[% VIEW view
        # بعض الوسائط القياسية
        prefix        => 'my_',
        suffix        => '.tt2',
        notfound      => 'no_such_file'
        ...
        # أي بيانات أخرى
        title         => 'My View title'
        other_item    => 'Joe Random Data'
        ...
%]
   # إضافة تعريفات بيانات جديدة، عبر الإشارة الذاتية 'my'
   [% my.author = "$abw.name <$abw.email>" %]
   [% my.copy   = "&copy; Copyright 2000 $my.author" %]
   # تعريف كتلة محلية
   [% BLOCK header %]
   This is the header block, title: [% title or my.title %]
   [% END %]
[% END %]
# الوصول إلى عناصر البيانات للعرض
[% view.title %]
[% view.other_item %]
# الوصول إلى الكتل مباشرة ('include_naked' option, set by default)
[% view.header %]
[% view.header(title => 'New Title') %]
# القوالب غير المحلية لها بادئة/لاحقة مرفقة
[% view.footer %]           # => [% INCLUDE my_footer.tt2 %]
# شكل أكثر تفصيلاً للوصول إلى الكتل
[% view.include( 'header', title => 'The Header Title' ) %]
[% view.include_header( title => 'The Header Title' ) %]
# شكل قصير جداً لما سبق ('include_naked' option, set by default)
[% view.header( title => 'The Header Title' ) %]
# القوالب غير المحلية لها بادئة/لاحقة مرفقة
[% view.footer %]           # => [% INCLUDE my_footer.tt2 %]
# السقوط على قالب 'notfound' ('my_no_such_file.tt2')
# إذا لم يتم العثور على القالب
[% view.include('missing') %]
[% view.include_missing %]
[% view.missing %]
# print() يتضمن قالبًا ذا صلة بنوع الوسيط
[% view.print("some text") %]     # type=TEXT, template='text'
[% BLOCK my_text.tt2 %]           # 'text' مع بادئة/لاحقة
   Text: [% item %]
[% END %]
# الآن print() يطبع مرجع هش، تم تعيينه إلى قالب 'hash'
[% view.print(some_hash_ref) %]   # type=HASH, template='hash'
[% BLOCK my_hash.tt2 %]           # 'hash' مع بادئة/لاحقة
   hash keys: [% item.keys.sort.join(', ')
[% END %]
# الآن print() يطبع مرجع قائمة، تم تعيينه إلى قالب 'list'
[% view.print(my_list_ref) %]     # type=ARRAY, template='list'
[% BLOCK my_list.tt2 %]           # 'list' مع بادئة/لاحقة
   list: [% item.join(', ') %]
[% END %]
# print() يعين 'My::Object' إلى 'My_Object'
[% view.print(myobj) %]
[% BLOCK my_My_Object.tt2 %]
   [% item.this %], [% item.that %]
[% END %]
# تحديث جدول التعيين
[% view.map.ARRAY = 'my_list_template' %]
[% view.map.TEXT  = 'my_text_block'    %]
# تغيير البادئة، اللاحقة، اسم العنصر، إلخ.
[% view.prefix = 'your_' %]
[% view.default = 'anyobj' %]
...

الأساليب

new($context, \%config)

ينشئ عرضًا جديدًا لـ Template::View يعرض عرضًا مخصصًا للكائن $context المحدد.

يمكن تمرير مرجع إلى مصفوفة تجزئة من خيارات التكوين كوسيط ثانٍ.

prefix
بادئة تضاف إلى جميع أسماء القوالب. 
[% USE view(prefix => 'my_') %]
[% view.view('foo', a => 20) %]     # => my_foo
suffix
لاحقة تضاف إلى جميع أسماء القوالب. 
[% USE view(suffix => '.tt2') %]
[% view.view('foo', a => 20) %]     # => foo.tt2
map
مصفوفة تجزئة تعين أنواع المراجع إلى أسماء القوالب. تستخدم طريقة print() هذا لتحديد القالب الذي سيتم استخدامه لعرض أي عنصر معين. العناصر TEXT و HASH و ARRAY افتراضيًا إلى 'test' و 'hash' و 'list' وفقًا لذلك. 
[% USE view(map => { ARRAY   => 'my_list',
                     HASH    => 'your_hash',
                     My::Foo => 'my_foo', } ) %]
[% view.print(some_text) %]         # => text
[% view.print(a_list) %]            # => my_list
[% view.print(a_hash) %]            # => your_hash
[% view.print(a_foo) %]             # => my_foo
[% BLOCK text %]
   Text: [% item %]
[% END %]
[% BLOCK my_list %]
   list: [% item.join(', ') %]
[% END %]
[% BLOCK your_hash %]
   hash keys: [% item.keys.sort.join(', ')
[% END %]
[% BLOCK my_foo %]
   Foo: [% item.this %], [% item.that %]
[% END %]
method
اسم طريقة قد توفرها الكائنات التي تم تمريرها إلى print() لتقديم أنفسهم للعرض. إذا تعذر العثور على إدخال خريطة محدد لكائن مرجعي وكان يدعم الطريقة (الافتراضي: 'present') فسيتم استدعاء الطريقة، مع تمرير العرض كوسيط. يمكن للكائن بعد ذلك إجراء استدعاءات عكسية مقابل العرض لتقديم نفسه. 
package Foo;
sub present {
    my ($self, $view) = @_;
    return "a regular view of a Foo\n";
}
sub debug {
    my ($self, $view) = @_;
    return "a debug view of a Foo\n";
}

في قالب:

[% USE view %]
[% view.print(my_foo_object) %]     # a regular view of a Foo
[% USE view(method => 'debug') %]
[% view.print(my_foo_object) %]     # a debug view of a Foo
default
القالب الافتراضي للاستخدام إذا لم يتم العثور على إدخال خريطة محدد لعنصر. 
[% USE view(default => 'my_object') %]
[% view.print(objref) %]            # => my_object

إذا لم يتم توفير إدخال خريطة أو افتراضي، فسوف يحاول العرض إنشاء اسم قالب من فئة الكائن، واستبدال أي تسلسل من الأحرف غير الكلمات بشرطات سفلية مفردة، على سبيل المثال:

# 'fubar' هو كائن من الفئة Foo::Bar
[% view.print(fubar) %]             # => Foo_Bar

ستتم إضافة أي بادئة ولاحقة حالية إلى اسم القالب الافتراضي وأي اسم تم إنشاؤه من فئة الكائن.

notfound
القالب الاحتياطي للاستخدام إذا لم يتم العثور على أي قالب آخر.
item
اسم متغير القالب الذي تقوم طريقة print() بتعيين العنصر الحالي إليه. الافتراضي هو 'item'. 
[% USE view %]
[% BLOCK list %]
   [% item.join(', ') %]
[% END %]
[% view.print(a_list) %]
[% USE view(item => 'thing') %]
[% BLOCK list %]
   [% thing.join(', ') %]
[% END %]
[% view.print(a_list) %]
view_prefix
بادئة الطرق التي يجب تعيينها إلى view() بواسطة AUTOLOAD. الافتراضي هو 'view_'. 
[% USE view %]
[% view.view_header() %]                    # => view('header')
[% USE view(view_prefix => 'show_me_the_' %]
[% view.show_me_the_header() %]             # => view('header')
view_naked
علامة للإشارة إلى ما إذا كان يجب إجراء أي محاولة لتعيين أسماء الطرق إلى أسماء القوالب حيث لا تتطابق مع view_prefix. الافتراضي هو 0. 
[% USE view(view_naked => 1) %]
[% view.header() %]                 # => view('header')
