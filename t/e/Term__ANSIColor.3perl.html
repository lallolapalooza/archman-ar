Term::ANSIColor(3perl) دليل مبرمجي بيرل Term::ANSIColor(3perl)

الاسم

Term::ANSIColor - تلوين إخراج الشاشة باستخدام تسلسلات هروب ANSI

الملخص

use Term::ANSIColor;
print color('bold blue');
print "This text is bold blue.\n";
print color('reset');
print "This text is normal.\n";
print colored("Yellow on magenta.", 'yellow on_magenta'), "\n";
print "This text is normal.\n";
print colored(['yellow on_magenta'], 'Yellow on magenta.', "\n");
print colored(['red on_bright_yellow'], 'Red on bright yellow.', "\n");
print colored(['bright_red on_black'], 'Bright red on black.', "\n");
print "\n";
# Map escape sequences back to color names.
use Term::ANSIColor 1.04 qw(uncolor);
my @names = uncolor('01;31');
print join(q{ }, @names), "\n";
# Strip all color escape sequences.
use Term::ANSIColor 2.01 qw(colorstrip);
print colorstrip("\e[1mThis is bold\e[0m"), "\n";
# Determine whether a color is valid.
use Term::ANSIColor 2.02 qw(colorvalid);
my $valid = colorvalid('blue bold', 'on_magenta');
print "Color string is ", $valid ? "valid\n" : "invalid\n";
# Create new aliases for colors.
use Term::ANSIColor 4.00 qw(coloralias);
coloralias('alert', 'red');
print "Alert is ", coloralias('alert'), "\n";
print colored("This is in red.", 'alert'), "\n";
use Term::ANSIColor qw(:constants);
print BOLD, BLUE, "This text is in bold blue.\n", RESET;
use Term::ANSIColor qw(:constants);
{
    local $Term::ANSIColor::AUTORESET = 1;
    print BOLD BLUE "This text is in bold blue.\n";
    print "This text is normal.\n";
}
use Term::ANSIColor 2.00 qw(:pushpop);
print PUSHCOLOR RED ON_GREEN "This text is red on green.\n";
print PUSHCOLOR BRIGHT_BLUE "This text is bright blue on green.\n";
print RESET BRIGHT_BLUE "This text is just bright blue.\n";
print POPCOLOR "Back to red on green.\n";
print LOCALCOLOR GREEN ON_BLUE "This text is green on blue.\n";
print "This text is red on green.\n";
{
    local $Term::ANSIColor::AUTOLOCAL = 1;
    print ON_BLUE "This text is red on blue.\n";
    print "This text is red on green.\n";
}
print POPCOLOR "Back to whatever we started as.\n";

الوصف

يقدم هذا الموديول واجهتين، إحداهما عبر color() و colored() والأخرى عبر الثوابت. كما أنه يوفر وظائف مساعدة uncolor() و colorstrip() و colorvalid() و coloralias()، والتي يجب استيرادها صراحةً لاستخدامها (راجع "الملخص").

إذا كنت تستخدم Term::ANSIColor في أمر طرفي، ففكر في دعم معيار CLICOLOR. راجع "دعم CLICOLOR" لمزيد من المعلومات.

راجع "التوافق" لمعرفة إصدارات Term::ANSIColor التي قدمت ميزات معينة وإصدارات Perl التي تضمنت هذه الميزات.

الألوان المدعومة

تقسم محاكيات الطرفيات التي تدعم الألوان إلى أربعة أنواع: تلك التي تدعم ثمانية ألوان فقط، وتلك التي تدعم ستة عشر لونًا، وتلك التي تدعم 256 لونًا، وتلك التي تدعم ألوان 24 بت. يوفر هذا الموديول رموز هروب ANSI لجميع هذه الأنواع. يشار إلى هذه الألوان بالألوان ANSI من 0 إلى 7 (عادي)، ومن 8 إلى 15 (16 لونًا)، ومن 16 إلى 255 (256 لونًا)، ولون حقيقي (يطلق عليه direct-color بواسطة xterm).

للأسف، غالبًا ما يعتمد تفسير الألوان من 0 إلى 7 على ما إذا كانت المحاكي يدعم ثمانية ألوان أم ستة عشر لونًا. المحاكيات التي تدعم ثمانية ألوان فقط (مثل طرفية لينكس) ستعرض الألوان من 0 إلى 7 بسطوع طبيعي وتتجاهل الألوان من 8 إلى 15، وتتعامل معها بنفس طريقة اللون الأبيض. المحاكيات التي تدعم 16 لونًا، مثل gnome-terminal، تعرض عادةً الألوان من 0 إلى 7 كإصدارات خافتة أو داكنة والألوان من 8 إلى 15 بسطوع طبيعي. في مثل هذه المحاكيات، عادةً ما يُعرض اللون الأبيض العادي (اللون 7) كرمادي باهت، مما يتطلب استخدام اللون الأبيض الساطع (15) للحصول على لون أبيض حقيقي. عادةً ما يكون اللون الأسود الساطع لونًا رماديًا داكنًا، على الرغم من أن بعض الطرفيات تعرضه باللون الأسود الخالص. تعامل بعض محاكيات الطرفيات ذات الـ 16 لونًا أيضًا اللون الأصفر العادي (اللون 3) كبرتقالي أو بني، واللون الأصفر الساطع (اللون 11) كأصفر.

باتباع الاصطلاح القياسي للمحاكيات ذات الـ 16 لونًا، يوفر هذا الموديول زوجًا من السمات لكل لون. لكل لون عادي (من 0 إلى 7)، يتم الحصول على اللون الساطع المقابل (من 8 إلى 15) عن طريق إضافة البادئة "bright_" إلى اسم اللون العادي. على سبيل المثال، "red" هو اللون 1 و "bright_red" هو اللون 9. وينطبق الشيء نفسه على ألوان الخلفية: "on_red" هو اللون العادي و "on_bright_red" هو اللون الساطع. قم بتمييز هذه السلاسل لواجهة الثوابت.

للأسف، لا توجد طريقة لمعرفة ما إذا كانت المحاكي الحالي يدعم أكثر من ثمانية ألوان، مما يجعل اختيار الألوان صعبًا. الخيار الأكثر تحفظًا هو استخدام الألوان العادية فقط، والتي يتم عرضها على الأقل على جميع المحاكيات. ومع ذلك، ستظهر داكنة في محاكيات الطرفيات ذات الـ 16 لونًا، بما في ذلك معظم المحاكيات الشائعة في بيئات X UNIX. إذا كنت تعرف أن العرض هو أحد هذه المحاكيات، فقد ترغب في استخدام البدائل الساطعة بدلاً من ذلك. والأفضل من ذلك، قدم للمستخدم طريقة لتكوين الألوان لتطبيق معين لتناسب محاكي الطرفية الخاص بهم.

بالنسبة للمحاكيات ذات 256 لونًا، يوفر هذا الموديول بالإضافة إلى ذلك "ansi0" إلى "ansi15"، وهي نفس ألوان 0 إلى 15 في محاكيات الـ 16 لونًا ولكنها تستخدم صيغة هروب الـ 256 لونًا، و "grey0" إلى "grey23" تتراوح من الأسود تقريبًا إلى الأبيض تقريبًا، ومجموعة من ألوان RGB. ألوان RGB هي من الشكل "rgbRGB" حيث R و G و B هي أرقام من 0 إلى 5 تعطي شدة الأحمر والأخضر والأزرق. الألوان الرمادية وألوان RGB متاحة أيضًا كـ "ansi16" إلى "ansi255" إذا كنت تريد أسماء بسيطة لجميع الألوان الـ 256. يتم أيضًا توفير المتغيرات "on_" لجميع هذه الألوان. قد يتم تجاهل هذه الألوان تمامًا على الطرفيات التي لا تدعم 256 لونًا أو قد يتم تفسيرها بشكل خاطئ وتنتج سلوكًا عشوائيًا. السمات الإضافية مثل الوميض أو الخط المائل أو الخط العريض قد لا تعمل مع لوحة الـ 256 لونًا.

بالنسبة للمحاكيات ذات الألوان الحقيقية، يدعم هذا الموديول سمات من الشكل "rNNNgNNNbNNN" و "on_rNNNgNNNbNNN" لجميع القيم من NNN بين 0 و 255. تمثل هذه الألوان الأمامية والخلفية، على التوالي، مع قيم RGB المعطاة بواسطة الأرقام NNN. قد يتم تجاهل هذه الألوان تمامًا على الطرفيات التي لا تدعم الألوان الحقيقية أو قد يتم تفسيرها بشكل خاطئ وتنتج سلوكًا عشوائيًا.

واجهة الوظائف

تستخدم واجهة الوظائف سلاسل سمات لوصف الألوان وسمات النص التي سيتم تعيينها للنص. السمات غير اللونية المعترف بها هي clear (مسح)، reset (إعادة تعيين)، bold (غامق)، dark (داكن)، faint (خافت)، italic (مائل)، underline (تحته خط)، underscore (تحته خط)، blink (وميض)، reverse (عكسي)، و concealed (مخفي). Clear و reset (إعادة تعيين إلى السمات الافتراضية)، dark و faint (خافت ومشبع)، و underline و underscore (تحته خط) متكافئة، لذا استخدم ما هو أكثر بديهية بالنسبة لك.

لاحظ أن ليست كل السمات مدعومة من قبل جميع أنواع الطرفيات، وقد لا تدعم بعض الطرفيات أيًا من هذه التسلسلات. Dark و faint، و italic، و blink، و concealed بشكل خاص غالبًا ما لا يتم تنفيذها.

سمات الألوان الأمامية العادية المعترف بها (الألوان 0 إلى 7) هي:

black  red  green  yellow  blue  magenta  cyan  white

سمات الألوان الأمامية الساطعة المقابلة (الألوان 8 إلى 15) هي:

bright_black  bright_red      bright_green  bright_yellow
bright_blue   bright_magenta  bright_cyan   bright_white

سمات ألوان الخلفية العادية المعترف بها (الألوان 0 إلى 7) هي:

on_black  on_red      on_green  on yellow
on_blue   on_magenta  on_cyan   on_white

سمات ألوان الخلفية الساطعة المعترف بها (الألوان 8 إلى 15) هي:

on_br

بالنسبة لمحطات الألوان 256، فإن ألوان المقدمة المعترف بها هي:

ansi0 .. ansi255
grey0 .. grey23

بالإضافة إلى "rgbRGB" بالنسبة لقيم R و G و B من 0 إلى 5، مثل "rgb000" أو "rgb515". وبالمثل، فإن ألوان الخلفية المعترف بها هي:

on_ansi0 .. on_ansi255
on_grey0 .. on_grey23

بالإضافة إلى "on_rgbRGB" بالنسبة لقيم R و G و B من 0 إلى 5.

بالنسبة لمحطات الألوان الحقيقية، فإن ألوان المقدمة المعترف بها هي "rRRRgGGGbBBB" بالنسب لقيم RRR و GGG و BBB بين 0 و 255. وبالمثل، فإن ألوان الخلفية المعترف بها هي "on_rRRRgGGGbBBB" بالنسب لقيم RRR و GGG و BBB بين 0 و 255.

لأي من السمات المذكورة أعلاه، لا يكون للتبديل أهمية.

السمات، بمجرد تعيينها، تستمر حتى يتم إلغاؤها (عن طريق طباعة السمة "clear" أو "reset"). كن حذرًا للقيام بذلك، وإلا فإن السمة ستستمر بعد انتهاء تشغيل البرنامج النصي الخاص بك، ويشعر الناس بالانزعاج الشديد من تغيير موجه الأوامر والكتابة الخاصة بهم إلى ألوان غريبة.

color(ATTR[, ATTR ...])
color() تأخذ أي عدد من السلاسل النصية كمعاملات وتعتبرها قوائم مفصولة بمسافات للسمات. ثم تشكل وتعيد التسلسل المباشر لتعيين تلك السمات. لا تطبعه، بل تعيده فقط، لذلك سيتعين عليك طباعته بنفسك إذا كنت تريد ذلك. هذا يتيح لك حفظه كسلسلة نصية، أو تمريره إلى شيء آخر، أو إرساله إلى مقبض ملف، أو القيام بأي شيء آخر قد تهتم به. color() يرمي استثناء إذا تم إعطاؤه سمة غير صالحة.
colored(STRING, ATTR[, ATTR ...])
colored(ATTR-REF, STRING[, STRING...])
كوسيلة مساعدة لإعادة تعيين الألوان، تأخذ colored() قيمة قياسية كوسيط أول وأي عدد من سلاسل السمات كوسيط ثانٍ وتعيد القيمة القياسية ملفوفة في رموز هروب بحيث يتم تعيين السمات حسب الطلب قبل السلسلة وإعادة تعيينها إلى الوضع الطبيعي بعد السلسلة. بدلاً من ذلك، يمكنك تمرير مرجع إلى مصفوفة كوسيط أول، ثم سيتم اعتبار محتويات تلك المصفوفة كسمات ورموز ألوان والباقي من الوسائط كنص لتلوينه.

عادةً، تقوم colored() فقط بوضع رموز السمات في بداية ونهاية السلسلة، ولكن إذا قمت بتعيين $Term::ANSIColor::EACHLINE إلى سلسلة نصية ما، فسيتم اعتبار تلك السلسلة فاصل السطر وسيتم تعيين السمة في بداية كل سطر من السلسلة المقدمة وإعادة تعيينها في نهاية كل سطر. غالبًا ما يكون هذا مرغوبًا فيه إذا كان الإخراج يحتوي على أسطر جديدة وتستخدم ألوان خلفية، حيث غالبًا ما يتم تفسير لون الخلفية الذي يستمر عبر سطر جديد بواسطة المحطة الطرفية على أنه يوفر لون الخلفية الافتراضي للسطر التالي. البرامج مثل pagers يمكن أن يتم خداعها أيضًا بواسطة السمات التي تمتد عبر الأسطر. عادةً ما ترغب في تعيين $Term::ANSIColor::EACHLINE إلى "\n" لاستخدام هذه الميزة.

ضع في اعتبارك بشكل خاص تعيين $Term::ANSIColor::EACHLINE إذا كنت تقوم بتبديل الإخراج إلى المخرج القياسي ومخرج الخطأ القياسي ولا تقوم بتفريغ المخرج القياسي (عبر autoflush() أو تعيين $|). إذا لم تفعل ذلك، فقد تظهر رموز إعادة تعيين اللون بشكل غير متوقع في المخزن المؤقت للإخراج القياسي بدلاً من الذهاب إلى العرض، مما يتسبب في ظهور إخراج الخطأ القياسي بلون خاطئ.

uncolor(ESCAPE)
uncolor() تؤدي الترجمة العكسية لـ color()، وتحويل تسلسلات الهروب إلى قائمة من السلاسل النصية المقابلة للسمات التي يتم تعيينها بواسطة تلك التسلسلات. uncolor() لن تعيد أبدًا "ansi16" إلى "ansi255"، وبدلاً من ذلك تفضل أسماء "grey" و "rgb" (وبالمثل لـ "on_ansi16" إلى "on_ansi255").
colorstrip(STRING[, STRING ...])
colorstrip() تزيل جميع تسلسلات هروب الألوان من السلاسل النصية المقدمة، وتعيد السلاسل النصية المعدلة بشكل منفصل في سياق مصفوفة أو مدمجة معًا في سياق قياسي. لا يتم تعديل وسائطها.
colorvalid(ATTR[, ATTR ...])
colorvalid() تأخذ سلاسل السمات بنفس طريقة color() وتعيد صحيحًا إذا كانت جميع السمات معروفة وخطأ بخلاف ذلك.
coloralias(ALIAS[, ATTR ...])
إذا تم تحديد ATTR، فيتم تفسيره كقائمة من السلاسل النصية المفصولة بمسافات تسمي سمات أو أسماء مستعارة موجودة. في هذه الحالة، تقوم coloralias() بإعداد اسم مستعار لـ ALIAS لمجموعة السمات المعطاة بواسطة ATTR. من تلك النقطة فصاعدًا، يمكن تمرير ALIAS إلى color() و colored() و colorvalid() وسيكون لها نفس المعنى مثل تسلسل السمات المعطى في ATTR. أحد الاستخدامات الممكنة لهذه المنشأة هو إعطاء أسماء أكثر معنى لألوان RGB ذات 256 لونًا. فقط الأحرف الأبجدية الرقمية ASCII، "."، "_"، و "-" مسموح بها في أسماء الأسماء المستعارة.

إذا تضمن ATTR أسماء مستعارة، فسيتم توسيع تلك الأسماء المستعارة في وقت التعريف وسيتم استخدام قيمها لتعريف الاسم المستعار الجديد. هذا يعني أنه إذا قمت بتعريف اسم مستعار A بدلالة اسم مستعار آخر B، ثم أعدت تعريف الاسم المستعار B لاحقًا، فلن تتغير قيمة الاسم المستعار A.

إذا لم يتم تحديد ATTR، فإن coloralias() تعيد السمة القياسية أو السمات التي يتم تسمية ALIAS بها، إن وجدت، أو undef إذا لم يكن ALIAS موجودًا. إذا تم تسميته لسمات متعددة، فستكون القيمة المعادة سلسلة نصية واحدة وسيتم فصل السمات بمسافات.

هذه هي نفس المنشأة التي تستخدمها متغير البيئة ANSI_COLORS_ALIASES (انظر "البيئة" أدناه) ولكن يمكن استخدامها في وقت التشغيل، وليس فقط عند تحميل الوحدة.

الاستدعاءات اللاحقة لـ coloralias() بنفس ALIAS ستتجاوز الأسماء المستعارة السابقة. لا توجد طريقة لإزالة اسم مستعار.

الأسماء المستعارة ليس لها تأثير على القيمة المعادة من uncolor().

تحذير: الأسماء المستعارة عامة وتؤثر على جميع المتصلين في نفس العملية. لا توجد طريقة لتعيين اسم مستعار مقصور على كتلة معينة من التعليمات البرمجية أو كائن معين.

واجهة الثوابت

بدلاً من ذلك، إذا قمت باستيراد ":constants"، يمكنك استخدام الثوابت التالية مباشرة:

CLEAR           RESET             BOLD            DARK
FAINT           ITALIC            UNDERLINE       UNDERSCORE
BLINK           REVERSE           CONCEALED
BLACK           RED               GREEN           YELLOW
BLUE            MAGENTA           CYAN            WHITE
BRIGHT_BLACK    BRIGHT_RED        BRIGHT_GREEN    BRIGHT_YELLOW
BRIGHT_BLUE     BRIGHT_MAGENTA    BRIGHT_CYAN     BRIGHT_WHITE
ON_BLACK        ON_RED            ON_GREEN        ON_YELLOW
ON_BLUE         ON_MAGENTA        ON_CYAN         ON_WHITE
ON_BRIGHT_BLACK ON_BRIGHT_RED     ON_BRIGHT_GREEN ON_BRIGHT_YELLOW
ON_BRIGHT_BLUE  ON_BRIGHT_MAGENTA ON_BRIGHT_CYAN  ON_BRIGHT_WHITE

هذه هي نفسها color('attribute') ويمكن استخدامها إذا كنت تفضل الكتابة:

print BOLD BLUE ON_WHITE "Text", RESET, "\n";

إلى

print colored ("Text", 'bold blue on_white'), "\n";

(لاحظ أن سطر النهاية يبقى منفصلاً لتجنب إرباك المحطة الطرفية كما هو موضح أعلاه نظرًا لاستخدام لون خلفية.)

إذا قمت باستيراد ":constants256"، يمكنك استخدام الثوابت التالية مباشرة:

ANSI0 .. ANSI255
GREY0 .. GREY23
RGBXYZ (لقيم X و Y و Z من 0 إلى 5، مثل RGB000 أو RGB515)
ON_ANSI0 .. ON_ANSI255
ON_GREY0 .. ON_GREY23
ON_RGBXYZ (لقيم X و Y و Z من 0 إلى 5)

لاحظ أن ":constants256" لا يشمل الثوابت الأخرى، لذلك إذا كنت تريد مزج الاثنين، فستحتاج إلى تضمين ":constants" أيضًا. قد ترغب في استيراد "RESET" على الأقل بشكل صريح، كما في:

use Term::ANSIColor 4.00 qw(RESET :constants256);

لا تدعم واجهة الثوابت الألوان الحقيقية والأسماء المستعارة.

عند استخدام الثوابت، إذا كنت لا ترغب في الاضطرار إلى تذكر إضافة ", RESET" في نهاية كل سطر طباعة، يمكنك تعيين $Term::ANSIColor::AUTORESET إلى قيمة صحيحة. بعد ذلك، سيتم إعادة تعيين وضع العرض تلقائيًا إذا لم تكن هناك فاصلة بعد الثابت. بعبارة أخرى، مع تعيين هذا المتغير:

print BOLD BLUE "Text\n";

سيعيد تعيين وضع العرض بعد ذلك، بينما:

print BOLD, BLUE, "Text\n";

لن يفعل ذلك. إذا كنت تستخدم ألوان خلفية، فمن المحتمل أن ترغب إما في استخدام say() (في الإصدارات الأحدث من Perl) أو طباعة سطر النهاية باستخدام عبارة طباعة منفصلة لتجنب إرباك المحطة الطرفية.

إذا تم تعيين $Term::ANSIColor::AUTOLOCAL (انظر أدناه)، فإنه يحتل الأسبقية على $Term::ANSIColor::AUTORESET، ويتم تجاهل الأخير.

تتميز واجهة الإجراء الفرعي بميزة على واجهة الثوابت في أن إجراءين فرعيين فقط يتم تصديرهما إلى مساحة الأسماء الخاصة بك، مقابل ثمانية وثلاثين في واجهة الثوابت، وتدعم الأسماء المستعارة وسمات الألوان الحقيقية. على الجانب الآخر، تتمتع واجهة الثوابت بميزة التحقق من الأخطاء بشكل أفضل في وقت التجميع، نظرًا لأن الأسماء المكتوبة بشكل خاطئ لـ
