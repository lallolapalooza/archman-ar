Template(3pm)

الحزمة: perl-template-toolkit

التراخيص: PerlArtistic

Template(3pm) وثائق Perl المساهمة من المستخدم Template(3pm)

الاسم

Template - وحدة واجهة أمامية لـ Template Toolkit

الملخص

use Template;
# بعض الخيارات المفيدة (انظر أدناه للقائمة الكاملة)
my $config = {
    INCLUDE_PATH => '/search/path',  # أو قائمة مرجعية
    INTERPOLATE  => 1,               # توسيع "$var" في النص العادي
    POST_CHOMP   => 1,               # تنظيف المسافات البيضاء
    PRE_PROCESS  => 'header',        # إضافة بادئة لكل قالب
    EVAL_PERL    => 1,               # تقييم كتل كود Perl
};
# إنشاء كائن Template
my $template = Template->new($config);
# تعريف متغيرات القالب للاستبدال
my $vars = {
    var1  => $value,
    var2  => \%hash,
    var3  => \@list,
    var4  => \&code,
    var5  => $object,
};
# تحديد اسم ملف الإدخال، أو مقبض ملف، أو مرجع نص، إلخ.
my $input = 'myfile.html';
# معالجة قالب الإدخال، مع استبدال المتغيرات
$template->process($input, $vars)
    || die $template->error();

الوصف

تصف هذه الوثيقة وحدة Template وهي الواجهة المباشرة في Perl لـ Template Toolkit. تغطي استخدام الوحدة وتعطي ملخصًا موجزًا لخيارات التكوين وتوجيهات القوالب. يرجى الاطلاع على Template::Manual للحصول على دليل مرجعي كامل يتعمق أكثر في ميزات واستخدامات Template Toolkit. يتوفر أيضًا Template::Tutorial كدليل تمهيدي لاستخدام Template Toolkit.

الأساليب

new(\%config)

يقوم أسلوب البناء new() (المطبق بواسطة فئة الأساس Template::Base) بإنشاء مثيل جديد لكائن "Template". يمكن تمرير مرجع لمصفوفة هاش لعناصر التكوين كمعامل.

my $tt = Template->new({
    INCLUDE_PATH => '/usr/local/templates',
    EVAL_PERL    => 1,
}) || die $Template::ERROR, "\n";

يتم إرجاع مرجع لكائن "Template" جديد، أو undef في حالة حدوث خطأ. في الحالة الأخيرة، يمكن استرداد رسالة الخطأ عن طريق استدعاء error() كأسلوب فئة أو عن طريق فحص متغير الحزمة $Template::ERROR مباشرة.

my $tt = Template->new(\%config)
    || die Template->error(), "\n";
my $tt = Template->new(\%config)
    || die $Template::ERROR, "\n";

للتسهيل، يمكن أيضًا تحديد عناصر التكوين كقائمة عناصر بدلاً من مرجع مصفوفة هاش. يتم طيها تلقائيًا في مصفوفة هاش بواسطة الباني.

my $tt = Template->new(INCLUDE_PATH => '/tmp', POST_CHOMP => 1)
    || die $Template::ERROR, "\n";

process($template, \%vars, $output, %options)

يتم استدعاء الأسلوب process() لمعالجة قالب. يحدد المعامل الأول قالب الإدخال كواحد مما يلي:

  • اسم ملف نسبي لـ "INCLUDE_PATH"، إذا تم تحديده
  • مرجع لنص يحتوي على نص القالب
  • مرجع لمقبض ملف (على سبيل المثال، "IO::Handle" أو فئة فرعية) أو "GLOB" (على سبيل المثال، "\*STDIN")، يمكن قراءة القالب منه.

يمكن تمرير مرجع لمصفوفة هاش كمعامل ثانٍ، يحتوي على تعريفات لمتغيرات القالب.

# اسم الملف
$tt->process('welcome.tt2')
    || die $tt->error(), "\n";
# مرجع نص
$text = "[% INCLUDE header %]\nHello world!\n[% INCLUDE footer %]";
$tt->process(\$text)
    || die $tt->error(), "\n";
# مقبض ملف (GLOB)
$tt->process(\*DATA)
    || die $tt->error(), "\n";
__END__
[% INCLUDE header %]
This is a template defined in the __END__ section which is
accessible via the DATA "file handle".
[% INCLUDE footer %]

افتراضيًا، تتم طباعة ناتج القالب المعالج إلى "STDOUT". يعيد الأسلوب process() بعد ذلك القيمة 1 للإشارة إلى النجاح. يمكن تمرير معامل ثالث لأسلوب process() لتحديد موقع إخراج مختلف. يمكن أن يكون هذا القيمة واحدًا مما يلي:

  • سلسلة نصية عادية تشير إلى اسم ملف سيتم فتحه (نسبيًا لـ "OUTPUT_PATH"، إذا تم تحديده) وسيتم كتابة الإخراج إليه
  • ملف GLOB مفتوح جاهز للإخراج
  • مرجع لمقياس (على سبيل المثال، سلسلة نصية) يتم إلحاق الإخراج/الخطأ به
  • مرجع لدالة فرعية يتم استدعاؤها، مع تمرير الإخراج كمعامل
  • أي مرجع كائن ينفذ أسلوب print() (على سبيل المثال، "IO::Handle"، "Apache::Request"، إلخ) والذي سيتم استدعاؤه، مع تمرير الإخراج الناتج كمعامل.

أمثلة:

# اسم ملف الإخراج
$tt->process('welcome.tt2', $vars, 'welcome.html')
    || die $tt->error(), "\n";
# مرجع لدالة إخراج
sub myout {
    my $output = shift;
    ...
}
$tt->process('welcome.tt2', $vars, \&myout)
    || die $tt->error(), "\n";
# مرجع لسلسلة نصية للإخراج
my $output = '';
$tt->process('welcome.tt2', $vars, \$output)
    || die $tt->error(), "\n";
print "output: $output\n";

في معالج Apache/mod_perl:

sub handler {
    my $req = shift;
    # ... الكود الخاص بك هنا ...
    # توجيه الإخراج إلى Apache::Request عبر $req->print($output)
    $tt->process($file, $vars, $req) || do {
        $req->log_reason($tt->error());
        return SERVER_ERROR;
    };
    return OK;
}

بعد معامل الإخراج الاختياري الثالث، يمكن أن يأتي مرجع اختياري لمصفوفة هاش أو قائمة من أزواج "(name, value)" لتوفير خيارات إضافية للإخراج. الخيار الوحيد المدعوم حاليًا هو "binmode"، والذي، عند تعيينه إلى أي قيمة حقيقية، سيضمن تعيين الملفات التي تم إنشاؤها (ولكن ليس أي مقابض ملفات موجودة تم تمريرها) إلى الوضع الثنائي.

# إما: مرجع مصفوفة هاش للخيارات
$tt->process($infile, $vars, $outfile, { binmode => 1 })
    || die $tt->error(), "\n";
# أو: قائمة من أزواج اسم، قيمة
$tt->process($infile, $vars, $outfile, binmode => 1)
    || die $tt->error(), "\n";

بدلاً من ذلك، يمكن لمعامل "binmode" تحديد طبقة IO معينة مثل ":utf8".

$tt->process($infile, $vars, $outfile, binmode => ':utf8')
    || die $tt->error(), "\n";

يمكن استخدام عنصر التكوين "OUTPUT" لتحديد موقع إخراج افتراضي بخلاف "\*STDOUT". يحدد "OUTPUT_PATH" دليلاً يجب أن يتم وضعه كبادئة لجميع مواقع الإخراج المحددة كملفات.

my $tt = Template->new({
    OUTPUT      => sub { ... },       # افتراضي
    OUTPUT_PATH => '/tmp',
...
}) || die Template->error(), "\n";
# استخدام OUTPUT الافتراضي (يتم استدعاء الدالة الفرعية)
$tt->process('welcome.tt2', $vars)
    || die $tt->error(), "\n";
# كتابة الملف إلى '/tmp/welcome.html'
$tt->process('welcome.tt2', $vars, 'welcome.html')
    || die $tt->error(), "\n";

يعيد الأسلوب process() القيمة 1 عند النجاح أو "undef" عند حدوث خطأ. يمكن استرداد رسالة الخطأ التي تم إنشاؤها في الحالة الأخيرة عن طريق استدعاء الأسلوب error(). انظر أيضًا "ملخص التكوين" الذي يصف كيف يمكن تخصيص معالجة الأخطاء بشكل أكبر.

error()

عند استدعائه كأسلوب فئة، فإنه يعيد قيمة متغير الحزمة $ERROR. وبالتالي، فإن ما يلي مكافئ.

my $tt = Template->new()
    || die Template->error(), "\n";
my $tt = Template->new()
    || die $Template::ERROR, "\n";

عند استدعائه كأسلوب كائن، فإنه يعيد قيمة المتغير الداخلي "_ERROR"، كما تم تعيينه بواسطة شرط خطأ في استدعاء سابق لـ process().

$tt->process('welcome.tt2')
    || die $tt->error(), "\n";

يتم تمثيل الأخطاء في Template Toolkit بواسطة كائنات من فئة Template::Exception. إذا أعاد الأسلوب process() قيمة خاطئة، فإن الـ

BLOCK

Define a named template block for INCLUDE, PROCESS and WRAPPER to use.

[% BLOCK hello %]
   Hello World
[% END %]
[% INCLUDE hello %]

FOREACH

Repeat the enclosed "FOREACH" ... "END" block for each value in the list.

[% FOREACH variable IN [ val, val, val ] %]    # either
[% FOREACH variable IN list %]                 # or
   The variable is set to [% variable %]
[% END %]

WHILE

The block enclosed between "WHILE" and "END" block is processed while the specified condition is true.

[% WHILE condition %]
   content
[% END %]

IF / UNLESS / ELSIF / ELSE

The enclosed block is processed if the condition is true / false.

[% IF condition %]
   content
[% ELSIF condition %]
 content
[% ELSE %]
 content
[% END %]
[% UNLESS condition %]
   content
[% # ELSIF/ELSE as per IF, above %]
   content
[% END %]

SWITCH / CASE

Multi-way switch/case statement.

[% SWITCH variable %]
[%   CASE val1 %]
       content
[%   CASE [ val2, val3 ] %]
       content
[%   CASE %]         # or [% CASE DEFAULT %]
       content
[% END %]

MACRO

Define a named macro.

[% MACRO name <directive> %]
[% MACRO name(arg1, arg2) <directive> %]
...
[% name %]
[% name(val1, val2) %]

FILTER

Process enclosed "FILTER" ... "END" block then pipe through a filter.

[% FILTER name %]                       # either
[% FILTER name( params ) %]             # or
[% FILTER alias = name( params ) %]     # or
   content
[% END %]

USE

Load a plugin module (see "Template::<Manual::Plugins"), or any regular Perl module when the "LOAD_PERL" option is set.

[% USE name %]                      # either
[% USE name( params ) %]            # or
[% USE var = name( params ) %]      # or
...
[% name.method %]
[% var.method %]

PERL / RAWPERL

Evaluate enclosed blocks as Perl code (requires the "EVAL_PERL" option to be set).

[% PERL %]
 # perl code goes here
 $stash->set('foo', 10);
 print "set 'foo' to ", $stash->get('foo'), "\n";
 print $context->include('footer', { var => $val });
[% END %]
[% RAWPERL %]
   # raw perl code goes here, no magic but fast.
   $output .= 'some output';
[% END %]

TRY / THROW / CATCH / FINAL

Exception handling.

[% TRY %]
 content
   [% THROW type info %]
[% CATCH type %]
 catch content
   [% error.type %] [% error.info %]
[% CATCH %] # or [% CATCH DEFAULT %]
 content
[% FINAL %]
   this block is always processed
[% END %]

NEXT

Jump straight to the next item in a "FOREACH" or "WHILE" loop.

[% NEXT %]

LAST

Break out of "FOREACH" or "WHILE" loop.

[% LAST %]

RETURN

Stop processing current template and return to including templates.

[% RETURN %]

STOP

Stop processing all templates and return to caller.

[% STOP %]

TAGS

Define new tag style or characters (default: "[%" "%]").

[% TAGS html %]
[% TAGS <!-- --> %]

COMMENTS

Ignored and deleted.

[% # this is a comment to the end of line
   foo = 'bar'
%]
[%# placing the '#' immediately inside the directive
    tag comments out the entire directive
%]
