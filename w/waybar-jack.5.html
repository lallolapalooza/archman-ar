waybar-jack(5)

الحزمة: waybar

التراخيص: MIT

waybar-jack(5) دليل تنسيقات الملفات waybar-jack(5)

الاسم

waybar - وحدة JACK

الوصف

تعرض وحدة jack الحالة الحالية لخادم JACK.

الإعداد

معنون بواسطة jack

format:
typeof: string
default: {load}%
التنسيق، كيف يجب عرض المعلومات. يستخدم هذا التنسيق عندما لا يتم تحديد تنسيقات أخرى.

format-connected:
typeof: string
يستخدم هذا التنسيق عندما تكون الوحدة متصلة بخادم JACK.

format-disconnected:
typeof: string
يستخدم هذا التنسيق عندما لا تكون الوحدة متصلة بخادم JACK.

format-xrun:
typeof: string
يستخدم هذا التنسيق لفترة استطلاع واحدة عندما يبلغ خادم JACK عن xrun.

realtime:
typeof: bool
default: true
خيار إسقاط صلاحيات الوقت الحقيقي لعميل JACK الذي افتتحه Waybar.

tooltip:
typeof: bool
default: true
خيار لتعطيل تلميح الأداة عند التحويم.

tooltip-format:
typeof: string
default: {bufsize}/{samplerate} {latency}ms
تنسيق المعلومات المعروضة في تلميح الأداة.

interval:
typeof: integer
default: 1
الفاصل الزمني الذي يتم فيه استطلاع المعلومات.

rotate:
typeof: integer
قيمة موجبة لتدوير تسمية النص (بزيادات 90 درجة).

max-length:
typeof: integer
الحد الأقصى للطول بالحروف الذي يجب أن تعرضه الوحدة.

min-length:
typeof: integer
الحد الأدنى للطول بالحروف الذي يجب أن تقبله الوحدة.

align:
typeof: float
محاذاة التسمية داخل الوحدة، حيث 0 هو محاذاة لليسار و 1 هو محاذاة لليمين. إذا تم تدوير الوحدة، فستتبع تدفق النص.

justify:
typeof: string
محاذاة النص داخل تسمية الوحدة، مما يسمح بخيارات 'left'، 'right'، أو 'center' لتحديد الموضع.

on-click:
typeof: string
الأمر الذي سيتم تنفيذه عند النقر على الوحدة.

on-click-middle:
typeof: string
الأمر الذي سيتم تنفيذه عند النقر بزر عجلة الماوس على الوحدة.

on-click-right:
typeof: string
الأمر الذي سيتم تنفيذه عند النقر بزر الماوس الأيمن على الوحدة.

on-update:
typeof: string
الأمر الذي سيتم تنفيذه عند تحديث الوحدة.

menu:
typeof: string
الإجراء الذي يعرض القائمة.

menu-file:
typeof: string
موقع ملف واصف القائمة. يجب أن يكون هناك عنصر من نوع

GtkMenu بالمعرف menu

menu-actions:
typeof: array
الإجراءات المقابلة لأزرار القائمة.

expand:
typeof: bool
default: false
يمكّن هذه الوحدة من استهلاك كل المساحة المتبقية ديناميكيًا.

بدائل التنسيق

{load}: تحميل وحدة المعالجة المركزية الحالي المقدر بواسطة JACK.

{bufsize}: حجم مخزن JACK المؤقت.

{samplerate}: معدل العينة الذي يعمل به خادم JACK.

{latency}: المدة، بالمللي ثانية، لحجم المخزن المؤقت الحالي.

{xruns}: عدد xruns التي أبلغ عنها خادم JACK منذ بدء Waybar.

أمثلة

"jack": {
	"format": "DSP {}%",
	"format-xrun": "{xruns} xruns",
	"format-disconnected": "DSP off",
	"realtime": true
}
