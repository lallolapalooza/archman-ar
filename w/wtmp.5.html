utmp(5) دليل تنسيقات الملفات utmp(5)

الاسم

utmp, wtmp - سجلات تسجيل الدخول

ملخص

#include <utmp.h>

الوصف

يسمح ملف utmp باكتشاف معلومات حول من يستخدم النظام حاليًا. قد يكون هناك مستخدمون آخرون يستخدمون النظام حاليًا، لأن ليس كل البرامج تستخدم تسجيل utmp.

تحذير: يجب ألا يكون ملف utmp قابلاً للكتابة من قبل فئة المستخدم "other"، لأن العديد من برامج النظام (بشكل أحمق) تعتمد على سلامته. أنت تخاطر بتزييف ملفات سجل النظام وتعديل ملفات النظام إذا تركت ملف utmp قابلاً للكتابة لأي مستخدم بخلاف المالك والمجموعة المالكة للملف.

الملف عبارة عن سلسلة من الهياكل utmp، معرّفة على النحو التالي في <utmp.h> (لاحظ أن هذا مجرد واحد من عدة تعريفات؛ تعتمد التفاصيل على إصدار libc):


/* القيم للحقل ut_type، أدناه */
#define EMPTY         0 /* السجل لا يحتوي على معلومات صالحة
                           (كان يُعرف سابقاً باسم UT_UNKNOWN في Linux) */
#define RUN_LVL       1 /* تغيير في مستوى تشغيل النظام (انظر
init(1)) */
#define BOOT_TIME     2 /* وقت تمهيد النظام (في ut_tv) */
#define NEW_TIME      3 /* الوقت بعد تغيير ساعة النظام
                           (في ut_tv) */
#define OLD_TIME      4 /* الوقت قبل تغيير ساعة النظام
                           (في ut_tv) */
#define INIT_PROCESS  5 /* عملية تم إنشاؤها بواسطة
init(1) */
#define LOGIN_PROCESS 6 /* عملية قائد الجلسة لتسجيل دخول المستخدم */
#define USER_PROCESS  7 /* عملية عادية */
#define DEAD_PROCESS  8 /* عملية منتهية */
#define ACCOUNTING    9 /* غير منفذ */
#define UT_LINESIZE      32
#define UT_NAMESIZE      32
#define UT_HOSTSIZE     256
struct exit_status {              /* نوع لـ ut_exit، أدناه */
    short e_termination;          /* حالة إنهاء العملية */
    short e_exit;                 /* حالة خروج العملية */
};
struct utmp {
    short   ut_type;              /* نوع السجل */
    pid_t   ut_pid;               /* معرف عملية تسجيل الدخول */
    char    ut_line[UT_LINESIZE]; /* اسم جهاز tty - "/dev/" */
    char    ut_id[4];             /* لاحقة اسم الطرفية،
                                     أو معرف inittab(5) */
    char    ut_user[UT_NAMESIZE]; /* اسم المستخدم */
    char    ut_host[UT_HOSTSIZE]; /* اسم المضيف لتسجيل الدخول عن بعد، أو
                                     إصدار النواة لرسائل مستوى التشغيل */
    struct  exit_status ut_exit;  /* حالة خروج العملية
                                     المعرفة بـ DEAD_PROCESS؛ لا
                                     تستخدمها Linux
init(1) */
    /* يجب أن يكون الحقلان ut_session و ut_tv بنفس الحجم عند
       الترجمة 32- و 64-بت.  يتيح ذلك مشاركة ملفات البيانات والذاكرة المشتركة
       بين تطبيقات 32- و 64-بت.  */
#if __WORDSIZE == 64 && defined __WORDSIZE_COMPAT32
    int32_t ut_session;           /* معرف الجلسة (
getsid(2))،
                                     تستخدم للنوافذ */
    struct {
        int32_t tv_sec;           /* الثواني */
        int32_t tv_usec;          /* الميكروثانية */
    } ut_tv;                      /* الوقت الذي تم فيه إنشاء السجل */
#else
     long   ut_session;           /* معرف الجلسة */
     struct timeval ut_tv;        /* الوقت الذي تم فيه إنشاء السجل */
#endif
    int32_t ut_addr_v6[4];        /* عنوان الإنترنت للمضيف
                                     البعيد؛ يستخدم عنوان IPv4
                                     فقط ut_addr_v6[0] */
    char __unused[20];            /* محجوز للاستخدام المستقبلي */
};
/* تراجعات للتوافق */
#define ut_name ut_user
#ifndef _NO_UT_TIME
#define ut_time ut_tv.tv_sec
#endif
#define ut_xtime ut_tv.tv_sec
#define ut_addr ut_addr_v6[0]

يمنح هذا الهيكل اسم الملف الخاص المرتبط بطرفية المستخدم، واسم تسجيل دخول المستخدم، ووقت تسجيل الدخول في شكل time(2). تنتهي حقول السلسلة بحرف فارغ ('\0') إذا كانت أقصر من حجم الحقل.

نتجت الإدخالات الأولى التي تم إنشاؤها عن معالجة init(1) لملف inittab(5). ومع ذلك، قبل معالجة الإدخال، يقوم init(1) بتنظيف utmp عن طريق تعيين ut_type إلى DEAD_PROCESS، ومسح ut_user و ut_host و ut_time بأحرف فارغة لكل سجل يكون فيه ut_type ليس DEAD_PROCESS أو RUN_LVL ولا توجد عملية بالمعرف ut_pid. إذا لم يتم العثور على سجل فارغ بالمعرف ut_id المطلوب، يقوم init(1) بإنشاء واحد جديد. يقوم بتعيين ut_id من inittab، و ut_pid و ut_time إلى القيم الحالية، و ut_type إلى INIT_PROCESS.

يقوم mingetty(8) (أو agetty(8)) بتحديد موقع الإدخال بواسطة معرف العملية، ويغير ut_type إلى LOGIN_PROCESS، ويغير ut_time، ويحدد ut_line، وينتظر حتى يتم إنشاء الاتصال. يقوم login(1)، بعد مصادقة المستخدم، بتغيير ut_type إلى USER_PROCESS، ويغير ut_time، ويحدد ut_host و ut_addr. اعتمادًا على mingetty(8) (أو agetty(8)) و login(1)، قد يتم تحديد الإدخالات بواسطة ut_line بدلاً من ut_pid المفضلة.

عندما يكتشف init(1) أن عملية قد خرجت، فإنه يحدد موقع إدخال utmp الخاص بها بواسطة ut_pid، ويضبط ut_type إلى DEAD_PROCESS، ويمسح ut_user و ut_host و ut_time بأحرف فارغة.

يقوم xterm(1) ومحاكيات الطرفية الأخرى بإنشاء سجل USER_PROCESS مباشرة وتوليد ut_id باستخدام السلسلة التي هي الجزء اللاحق من اسم الطرفية (الأحرف التي تلي /dev/[pt]ty). إذا وجدوا DEAD_PROCESS لهذا المعرف، فإنهم يعيدون استخدامه، وإلا فإنهم ينشئون إدخالًا جديدًا. إذا تمكنوا، فإنهم سيضعون علامة DEAD_PROCESS عليه عند الخروج، ويوصى بمسح ut_line و ut_time و ut_user و ut_host أيضًا.

يقوم telnetd(8) بإعداد إدخال LOGIN_PROCESS ويترك الباقي لـ login(1) كالمعتاد. بعد انتهاء جلسة telnet، يقوم telnetd(8) بتنظيف utmp بالطريقة الموضحة.

يسجل ملف wtmp جميع عمليات تسجيل الدخول وتسجيل الخروج. تنسيقه مطابق تمامًا لـ utmp باستثناء أن اسم المستخدم الفارغ يشير إلى تسجيل الخروج من الطرفية المرتبطة. علاوة على ذلك، يشير اسم الطرفية ~ باسم المستخدم shutdown أو reboot إلى إيقاف تشغيل النظام أو إعادة تشغيله، وزوج أسماء الطرفيات |/} يسجل وقت النظام القديم/الجديد عندما يقوم date(1) بتغييره. يتم الاحتفاظ بـ wtmp بواسطة login(1) و init(1) وبعض إصدارات getty(8) (مثل mingetty(8) أو agetty(8)). لا يقوم أي من هذه البرامج بإنشاء الملف، لذلك إذا تمت إزالته، يتم إيقاف تسجيل السجلات.

الملفات

/var/run/utmp
/var/log/wtmp

الإصدارات

لا يحدد POSIX.1 بنية utmp، بل بنية تسمى utmpx (كجزء من امتداد XSI)، مع مواصفات للحقول ut_type و ut_pid و ut_line و ut_id و ut_user و ut_tv. لا يحدد POSIX.1 أطوال الحقول ut_line و ut_user.

يحدد Linux بنية utmpx لتكون نفس بنية utmp.

المعايير

Linux.

التاريخ

إدخالات utmp في Linux لا تتوافق مع v7/BSD ولا مع System V؛ إنها مزيج من الاثنين.

يحتوي v7/BSD على حقول أقل؛ والأهم من ذلك أنه يفتقر إلى ut_type، مما يتسبب في عرض البرامج الأصلية الشبيهة بـ v7/BSD (على سبيل المثال) الإدخالات المنتهية أو إدخالات تسجيل الدخول. علاوة على ذلك، لا يوجد ملف تكوين يقوم بتخصيص فتحات للجلسات. تقوم BSD بذلك لأنها تفتقر إلى حقول ut_id.

في Linux (كما في System V)، لن يتغير حقل ut_id للسجل أبدًا بمجرد تعيينه، مما يحجز هذا الفتحة دون الحاجة إلى ملف تكوين. قد يؤدي مسح ut_id إلى ظروف سباق تؤدي إلى إدخالات utmp تالفة وثغرات أمنية محتملة. مسح الحقول المذكورة أعلاه عن طريق الملء

wtmp(5) - Linux manual pages

NAME

wtmp - login record file

SYNOPSIS

/var/run/wtmp

DESCRIPTION

The file wtmp records logins and logouts. It is used by the commands who(1) and last(1).

Each entry in the file is a structure of type struct utmp as defined in <utmp.h>.

The structure contains at least the following fields:

char ut_user[32]; Username.

char ut_id[4]; Identifier for the session.

char ut_type; Type of entry.

pid_t ut_pid; PID of the process associated with the entry.

char ut_line[32]; The device name of the terminal (e.g. tty1).

char ut_host[256]; The name of the remote host (used for login entries from network daemons).

struct exit_status This contains the exit status and signal information about the process that was started in this utmp entry.

struct timeval ut_tv; Time the entry was recorded.

char ut_addr_v6[4][4]; This field is used to store the Internet address of the remote host. The field is defined as a 2D array for compatibility with IPv4 addresses.

The valid entries for ut_type are:

EMPTY
(0) No username.
RUN_LVL
(1) Change in runlevel.
BOOT_TIME
(2) System boot.
OLD_TIME
(3) Time was changed.
NEW_TIME
(4) Time was changed.
INIT_PROCESS
(5) Process started by init.
LOGIN_PROCESS
(6) A login process.
USER_PROCESS
(7) A user process.
DEAD_PROCESS
(8) A process has died.
LOGIN_DATETIME
(9) New login time.
UT_UNKNOWN
(10) Indicates an unknown entry type.
AC_SCREEN
(11) Used by ac(1) to signal screen switching.
AC_OFF
(12) Used by ac(1) to signal system shutdown.

The field ut_id is used to identify sessions. When a login process is started, it is given a unique ut_id. This identifier is then used in subsequent entries related to that session, such as when the process dies.

The fields ut_user, ut_line, ut_id, and ut_host are null-padded if the data does not fill the field.

System V semantics require that ut_user and ut_line are null-terminated. Filling the remaining bytes with null bytes is not required by System V semantics, but makes it possible to run many programs which assume BSD semantics and which do not modify utmp. Linux uses the BSD conventions for line contents, as documented above.

System V has no ut_host or ut_addr_v6 fields.

NOTES

Unlike various other systems, where utmp logging can be disabled by removing the file, utmp must always exist on Linux. If you want to disable who(1), then do not make utmp world readable.

The file format is machine-dependent, so it is recommended that it be processed only on the machine architecture where it was created.

Note that on biarch platforms, that is, systems which can run both 32-bit and 64-bit applications (x86-64, ppc64, s390x, etc.), ut_tv is the same size in 32-bit mode as in 64-bit mode. The same goes for ut_session and ut_time if they are present. This allows data files and shared memory to be shared between 32-bit and 64-bit applications. This is achieved by changing the type of ut_session to int32_t, and that of ut_tv to a struct with two int32_t fields tv_sec and tv_usec. Since ut_tv may not be the same as struct timeval, then instead of the call:


gettimeofday((struct timeval *) &ut.ut_tv, NULL);

the following method of setting this field is recommended:


struct utmp ut;
struct timeval tv;
gettimeofday(&tv, NULL);
ut.ut_tv.tv_sec = tv.tv_sec;
ut.ut_tv.tv_usec = tv.tv_usec;

الاسم

wtmp - سجل تسجيل الدخول

الملخص

/var/run/wtmp

الوصف

يسجل الملف wtmp عمليات تسجيل الدخول والخروج. يتم استخدامه بواسطة الأوامر who(1) و last(1).

كل إدخال في الملف هو هيكل من النوع struct utmp كما هو محدد في <utmp.h>.

يحتوي الهيكل على الحقول التالية على الأقل:

char ut_user[32]; اسم المستخدم.

char ut_id[4]; معرف الجلسة.

char ut_type; نوع الإدخال.

pid_t ut_pid; معرف العملية (PID) المرتبط بالإدخال.

char ut_line[32]; اسم الجهاز الطرفي (على سبيل المثال، tty1).

char ut_host[256]; اسم المضيف البعيد (يستخدم لإدخالات تسجيل الدخول من خوادم الشبكة).

struct exit_status يحتوي هذا على معلومات حالة الخروج والإشارة الخاصة بالعملية التي بدأت في هذا الإدخال utmp.

struct timeval ut_tv; وقت تسجيل الإدخال.

char ut_addr_v6[4][4]; يستخدم هذا الحقل لتخزين عنوان الإنترنت للمضيف البعيد. تم تعريف الحقل كمصفوفة ثنائية الأبعاد لتوافقه مع عناوين IPv4.

الإدخالات الصالحة لـ ut_type هي:

EMPTY
(0) لا يوجد اسم مستخدم.
RUN_LVL
(1) تغيير في مستوى التشغيل.
BOOT_TIME
(2) إقلاع النظام.
OLD_TIME
(3) تم تغيير الوقت.
NEW_TIME
(4) تم تغيير الوقت.
INIT_PROCESS
(5) عملية بدأها init.
LOGIN_PROCESS
(6) عملية تسجيل دخول.
USER_PROCESS
(7) عملية مستخدم.
DEAD_PROCESS
(8) ماتت عملية.
LOGIN_DATETIME
(9) وقت تسجيل دخول جديد.
UT_UNKNOWN
(10) يشير إلى نوع إدخال غير معروف.
AC_SCREEN
(11) تستخدم بواسطة ac(1) للإشارة إلى تبديل الشاشة.
AC_OFF
(12) تستخدم بواسطة ac(1) للإشارة إلى إيقاف تشغيل النظام.

يستخدم الحقل ut_id لتحديد الجلسات. عند بدء عملية تسجيل دخول، يتم منحها ut_id فريد. ثم يتم استخدام هذا المعرف في الإدخالات اللاحقة المتعلقة بتلك الجلسة، مثل عندما تموت العملية.

يتم ملء الحقول ut_user و ut_line و ut_id و ut_host بالبايتات الفارغة إذا لم تملأ البيانات الحقل. لا يتطلب النظام V ملء البايتات المتبقية ببايتات فارغة، ولكنه يجعل من الممكن تشغيل العديد من البرامج التي تفترض دلالات BSD والتي لا تعدل utmp. يستخدم لينكس اصطلاحات BSD لمحتويات الخط، كما هو موضح أعلاه.

لا يحتوي النظام V على حقلي ut_host أو ut_addr_v6.

ملاحظات

على عكس الأنظمة الأخرى المختلفة، حيث يمكن تعطيل تسجيل utmp عن طريق إزالة الملف، يجب أن يوجد utmp دائمًا في لينكس. إذا كنت ترغب في تعطيل who(1)، فلا تجعل utmp قابلاً للقراءة عالميًا.

تنسيق الملف يعتمد على الآلة، لذلك يوصى بمعالجته فقط على بنية الآلة التي تم إنشاؤه عليها.

لاحظ أنه على منصات biarch، أي الأنظمة التي يمكنها تشغيل تطبيقات 32 بت و 64 بت (x86-64، ppc64، s390x، إلخ)، فإن ut_tv بنفس حجم وضع 32 بت كما في وضع 64 بت. الشيء نفسه ينطبق على ut_session و ut_time إذا كانت موجودة. هذا يسمح بمشاركة ملفات البيانات والذاكرة المشتركة بين تطبيقات 32 بت و 64 بت. يتم تحقيق ذلك عن طريق تغيير نوع ut_session إلى int32_t، ونوع ut_tv إلى هيكل بحقلين int32_t هما tv_sec و tv_usec. نظرًا لأن ut_tv قد لا يكون هو نفسه struct timeval، فبدلاً من استدعاء:


gettimeofday((struct timeval *) &ut.ut_tv, NULL);

يتم التوصية بالطريقة التالية لتعيين هذا الحقل:


struct utmp ut;
struct timeval tv;
gettimeofday(&tv, NULL);
ut.ut_tv.tv_sec = tv.tv_sec;
ut.ut_tv.tv_usec = tv.tv_usec;
