WWW::SMS

احصل على أحدث إصدار من Www::Sms من http://search.cpan.org/author/GIULIENK/
قم بتحديث SMS.pm بالمزود الجديد إذا رغبت في ذلك
انسخ o2UK.pm إلى دليل مكتبة Perl الخاصة بك /WWW/SMS/، على سبيل المثال إلى:
linux:/usr/lib/perl5/site_perl/5.6.1/WWW/SMS>
يجب عليك التسجيل في O2 http://www.o2.co.uk/ (genie سابقًا)
وهو مجاني في الوقت الحالي ويمكنك إرسال 100 رسالة نصية شهريًا مجانًا
يجب عليك تمرير اسم المستخدم / كلمة المرور أو الملف الذي يحتوي عليهما
لا أعرف جميع بطاقات SIM المتوفرة بعد: o2 و vodafone تعملان
واختبره :-)
use WWW::SMS;
# بادئة دولية، بادئة مشغل، رقم هاتف، نص الرسالة
my $sms = WWW::SMS->new('44', '7777', '123456', 'This is a test.','username' => 'your_o2_username', 'password' => 'your_o2_password');
for ( $sms->gateways() ) {
    print "Trying $_...\n";
    if ( $sms->send($_) ) {               # محاولة إرسال رسالة نصية...
        last;                             # حتى تنجح ;)
    } else {
        print $WWW::SMS::Error;           # للأسف!
    }
}
