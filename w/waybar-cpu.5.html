waybar-cpu(5)

الحزمة: waybar

التراخيص: MIT

waybar-cpu(5) دليل تنسيقات الملفات waybar-cpu(5)

الاسم

waybar - وحدة وحدة المعالجة المركزية (CPU)

الوصف

وحدة cpu تعرض استخدام وحدة المعالجة المركزية الحالي.

التكوين

interval:
النوع: عدد صحيح (integer)
الافتراضي: 10
الفاصل الزمني الذي يتم فيه استقصاء المعلومات.

format:
النوع: سلسلة نصية (string)
الافتراضي: {usage}%
التنسيق، كيف يجب عرض المعلومات. يتم إدراج البيانات في {}.

format-icons:
النوع: مصفوفة/كائن (array/object)
بناءً على الاستخدام الحالي، يتم اختيار الأيقونة المقابلة.
الترتيب من منخفض إلى مرتفع. أو حسب الحالة إذا كان كائناً.

max-length:
النوع: عدد صحيح (integer)
الحد الأقصى للطول بالبايت الذي يجب أن تعرضه الوحدة.

min-length:
النوع: عدد صحيح (integer)
الحد الأدنى للطول بالبايت الذي يجب أن تقبله الوحدة.

align:
النوع: عدد عشري (float)
محاذاة التسمية داخل الوحدة، حيث 0 هو المحاذاة إلى اليسار و 1 هو المحاذاة إلى اليمين. إذا تم تدوير الوحدة، فستتبع تدفق النص.

justify:
النوع: سلسلة نصية (string)
محاذاة النص داخل تسمية الوحدة، تسمح بخيارات 'left'، 'right'، أو 'center' لتحديد الموضع.

rotate:
النوع: عدد صحيح (integer)
قيمة موجبة لتدوير تسمية النص (بزيادات 90 درجة).

states:
النوع: كائن (object)
عدد من حالات استخدام وحدة المعالجة المركزية التي يتم تنشيطها عند مستويات استخدام معينة. انظر waybar-states(5).

on-click:
النوع: سلسلة نصية (string)
الأمر الذي سيتم تنفيذه عند النقر على الوحدة.

on-click-middle:
النوع: سلسلة نصية (string)
الأمر الذي سيتم تنفيذه عند النقر بزر الماوس الأوسط على الوحدة.

on-click-right:
النوع: سلسلة نصية (string)
الأمر الذي سيتم تنفيذه عند النقر بزر الماوس الأيمن على الوحدة.

on-update:
النوع: سلسلة نصية (string)
الأمر الذي سيتم تنفيذه عند تحديث الوحدة.

on-scroll-up:
النوع: سلسلة نصية (string)
الأمر الذي سيتم تنفيذه عند التمرير لأعلى على الوحدة.

on-scroll-down:
النوع: سلسلة نصية (string)
الأمر الذي سيتم تنفيذه عند التمرير لأسفل على الوحدة.

smooth-scrolling-threshold:
النوع: عدد عشري (double)
العتبة التي سيتم استخدامها عند التمرير.

tooltip:
النوع: قيمة منطقية (bool)
الافتراضي: true
خيار لتعطيل تلميح الأداة عند التحويم.

expand:
النوع: قيمة منطقية (bool)
الافتراضي: false
يمكّن هذه الوحدة من استهلاك كل المساحة المتبقية ديناميكياً.

بدائل التنسيق

{load}: تحميل وحدة المعالجة المركزية الحالي.

{usage}: استخدام وحدة المعالجة المركزية الإجمالي الحالي.

{usage{n}}: استخدام النواة n من وحدة المعالجة المركزية الحالي. يتم ترقيم النوى بدءًا من الصفر، لذا ستكون النواة الأولى {usage0} والرابعة {usage3}.

{avg_frequency}: متوسط تردد وحدة المعالجة المركزية الحالي (بناءً على جميع النوى) بالجيجاهرتز.

{max_frequency}: أقصى تردد لوحدة المعالجة المركزية الحالي (بناءً على النواة ذات التردد الأعلى) بالجيجاهرتز.

{min_frequency}: أدنى تردد لوحدة المعالجة المركزية الحالي (بناءً على النواة ذات التردد الأدنى) بالجيجاهرتز.

{icon}: أيقونة لاستخدام وحدة المعالجة المركزية الإجمالي.

{icon{n}}: أيقونة لاستخدام النواة n من وحدة المعالجة المركزية. استخدمها مثل {icon0}.

أمثلة

التكوين الأساسي:

"cpu": {
	"interval": 10,
	"format": "{}% ",
	"max-length": 10
}

استخدام وحدة المعالجة المركزية لكل نواة يتم عرضه كأيقونات:

"cpu": {
	"interval": 1,
	"format": "{icon0}{icon1}{icon2}{icon3} {usage:>2}% ",
	"format-icons": ["▁", "▂", "▃", "▄", "▅", "▆", "▇", "█"],
}
