الحزمة: postgresql

الاسم

SELECT, TABLE, WITH - استرداد الصفوف من جدول أو عرض

الملخص

[ WITH [ RECURSIVE ] with_query [, ...] ]
SELECT [ ALL | DISTINCT [ ON ( expression [, ...] ) ] ]
    [ { * | expression [ [ AS ] output_name ] } [, ...] ]
    [ FROM from_item [, ...] ]
    [ WHERE condition ]
    [ GROUP BY [ ALL | DISTINCT ] grouping_element [, ...] ]
    [ HAVING condition ]
    [ WINDOW window_name AS ( window_definition ) [, ...] ]
    [ { UNION | INTERSECT | EXCEPT } [ ALL | DISTINCT ] select ]
    [ ORDER BY expression [ ASC | DESC | USING operator ] [ NULLS { FIRST | LAST } ] [, ...] ]
    [ LIMIT { count | ALL } ]
    [ OFFSET start [ ROW | ROWS ] ]
    [ FETCH { FIRST | NEXT } [ count ] { ROW | ROWS } { ONLY | WITH TIES } ]
    [ FOR { UPDATE | NO KEY UPDATE | SHARE | KEY SHARE } [ OF from_reference [, ...] ] [ NOWAIT | SKIP LOCKED ] [...] ]
where from_item can be one of:
    [ ONLY ] table_name [ * ] [ [ AS ] alias [ ( column_alias [, ...] ) ] ]
                [ TABLESAMPLE sampling_method ( argument [, ...] ) [ REPEATABLE ( seed ) ] ]
    [ LATERAL ] ( select ) [ [ AS ] alias [ ( column_alias [, ...] ) ] ]
    with_query_name [ [ AS ] alias [ ( column_alias [, ...] ) ] ]
    [ LATERAL ] function_name ( [ argument [, ...] ] )
                [ WITH ORDINALITY ] [ [ AS ] alias [ ( column_alias [, ...] ) ] ]
    [ LATERAL ] function_name ( [ argument [, ...] ] ) AS alias ( column_definition [, ...] )
    [ LATERAL ] function_name ( [ argument [, ...] ] ) AS ( column_definition [, ...] )
    [ LATERAL ] ROWS FROM( function_name ( [ argument [, ...] ] ) [ AS ( column_definition [, ...] ) ] [, ...] )
                [ WITH ORDINALITY ] [ [ AS ] alias [ ( column_alias [, ...] ) ] ]
    from_item join_type from_item { ON join_condition | USING ( join_column [, ...] ) [ AS join_using_alias ] }
    from_item NATURAL join_type from_item
    from_item CROSS JOIN from_item
and grouping_element can be one of:
    ( )
    expression
    ( expression [, ...] )
    ROLLUP ( { expression | ( expression [, ...] ) } [, ...] )
    CUBE ( { expression | ( expression [, ...] ) } [, ...] )
    GROUPING SETS ( grouping_element [, ...] )
and with_query is:
    with_query_name [ ( column_name [, ...] ) ] AS [ [ NOT ] MATERIALIZED ] ( select | values | insert | update | delete | merge )
        [ SEARCH { BREADTH | DEPTH } FIRST BY column_name [, ...] SET search_seq_col_name ]
        [ CYCLE column_name [, ...] SET cycle_mark_col_name [ TO cycle_mark_value DEFAULT cycle_mark_default ] USING cycle_path_col_name ]
TABLE [ ONLY ] table_name [ * ]

الوصف

SELECT يسترد الصفوف من جدول واحد أو أكثر. المعالجة العامة لـ SELECT هي كما يلي:

1.يتم حساب جميع الاستعلامات في قائمة WITH. هذه تعمل بشكل فعال كجداول مؤقتة يمكن الرجوع إليها في قائمة FROM. يتم حساب استعلام WITH الذي يتم الرجوع إليه أكثر من مرة في FROM مرة واحدة فقط، ما لم يتم تحديد خلاف ذلك باستخدام NOT MATERIALIZED. (انظر قسم WITH Clause أدناه.)
2.يتم حساب جميع العناصر في قائمة FROM. (كل عنصر في قائمة FROM هو جدول حقيقي أو افتراضي.) إذا تم تحديد أكثر من عنصر واحد في قائمة FROM، يتم ربطها معًا (cross-joined). (انظر قسم FROM Clause أدناه.)
3.إذا تم تحديد عبارة WHERE، يتم استبعاد جميع الصفوف التي لا تحقق الشرط من الإخراج. (انظر قسم WHERE Clause أدناه.)
4.إذا تم تحديد عبارة GROUP BY، أو إذا كانت هناك استدعاءات لوظائف تجميعية، يتم دمج الإخراج في مجموعات من الصفوف التي تتطابق مع قيمة واحدة أو أكثر، ويتم حساب نتائج الوظائف التجميعية. إذا كانت عبارة HAVING موجودة، فإنها تستبعد المجموعات التي لا تحقق الشرط المحدد. (انظر قسم GROUP BY Clause و HAVING Clause أدناه.) على الرغم من أن أعمدة إخراج الاستعلام يتم حسابها اسميًا في الخطوة التالية، إلا أنه يمكن أيضًا الإشارة إليها (بالاسم أو الرقم التسلسلي) في عبارة GROUP BY.
5.يتم حساب صفوف الإخراج الفعلية باستخدام تعبيرات إخراج SELECT لكل صف أو مجموعة صفوف محددة. (انظر قسم SELECT List أدناه.)
6.SELECT DISTINCT يستبعد الصفوف المكررة من النتيجة. SELECT DISTINCT ON يستبعد الصفوف التي تتطابق مع جميع التعبيرات المحددة. SELECT ALL (الافتراضي) سيعيد جميع الصفوف المرشحة، بما في ذلك التكرارات. (انظر قسم DISTINCT Clause أدناه.)
7.باستخدام عوامل التشغيل UNION، INTERSECT، و EXCEPT، يمكن دمج إخراج أكثر من عبارة SELECT واحدة لتكوين مجموعة نتائج واحدة. تعيد عامل التشغيل UNION جميع الصفوف الموجودة في مجموعة نتائج واحدة أو كليهما. عامل التشغيل INTERSECT يعيد جميع الصفوف الموجودة حصريًا في كلتا المجموعتين. عامل التشغيل EXCEPT يعيد الصفوف الموجودة في المجموعة الأولى ولكن ليس في الثانية. في جميع الحالات الثلاث، يتم استبعاد الصفوف المكررة ما لم يتم تحديد ALL. يمكن إضافة الكلمة الضوضائية DISTINCT لتحديد استبعاد الصفوف المكررة بشكل صريح. لاحظ أن DISTINCT هو السلوك الافتراضي هنا، حتى لو كان ALL هو الافتراضي لـ SELECT نفسه. (انظر قسم UNION Clause، INTERSECT Clause، و EXCEPT Clause أدناه.)
8.إذا تم تحديد عبارة ORDER BY، يتم فرز الصفوف المعادة بالترتيب المحدد. إذا لم يتم إعطاء ORDER BY، يتم إرجاع الصفوف بأي ترتيب يراه النظام أسرع في الإنتاج. (انظر قسم ORDER BY Clause أدناه.)
9.إذا تم تحديد عبارة LIMIT (أو FETCH FIRST) أو OFFSET، فإن عبارة SELECT تعيد فقط مجموعة فرعية من صفوف النتائج. (انظر قسم LIMIT Clause أدناه.)
10.إذا تم تحديد FOR UPDATE، FOR NO KEY UPDATE، FOR SHARE أو FOR KEY SHARE، فإن عبارة SELECT تقفل الصفوف المحددة ضد التحديثات المتزامنة. (انظر قسم The Locking Clause أدناه.)

يجب أن يكون لديك امتياز SELECT على كل عمود مستخدم في أمر SELECT. استخدام FOR NO KEY UPDATE، FOR UPDATE، FOR SHARE أو FOR KEY SHARE يتطلب أيضًا امتياز UPDATE (على عمود واحد على الأقل من كل جدول تم تحديده).

المعلمات

عبارة WITH

تسمح عبارة WITH بتحديد استعلام فرعي واحد أو أكثر يمكن الرجوع إليه بالاسم في الاستعلام الرئيسي. تعمل الاستعلامات الفرعية بشكل فعال كجداول مؤقتة أو عروض لمدة الاستعلام الرئيسي. يمكن أن يكون كل استعلام فرعي عبارة SELECT، TABLE، VALUES، INSERT، UPDATE، DELETE، أو MERGE. عند كتابة عبارة تعديل البيانات (INSERT، UPDATE، DELETE، أو MERGE) في WITH، من المعتاد تضمين عبارة RETURNING. إنها نتيجة RETURNING، وليس الجدول الأساسي الذي تقوم العبارة بتعديله، التي تشكل الجدول المؤقت الذي تقرأه الاستعلام الرئيسي. إذا تم حذف RETURNING، فسيتم تنفيذ العبارة، ولكنها لن تنتج أي مخرجات ولذلك لا يمكن الرجوع إليها كجدول بواسطة الاستعلام الرئيسي.

يجب تحديد اسم (بدون مؤهل مخطط) لكل استعلام WITH. اختياريًا، يمكن تحديد قائمة بأسماء الأعمدة؛ إذا تم حذف ذلك، يتم استنتاج أسماء الأعمدة من الاستعلام الفرعي.

إذا تم تحديد RECURSIVE، فإنه يسمح لاستعلام فرعي SELECT بالإشارة إلى نفسه بالاسم. يجب أن يكون هذا الاستعلام الفرعي بالشكل

non_recursive_term UNION [ ALL | DISTINCT ] recursive_term

حيث يجب أن يظهر الإسناد الذاتي التكراري على الجانب الأيمن من UNION. يُسمح بإسناد ذاتي تكراري واحد فقط لكل استعلام. لا يتم دعم عبارات تعديل البيانات التكرارية، ولكن يمكنك استخدام نتائج استعلام SELECT تكراري في عبارة تعديل البيانات. انظر القسم 7.8 للحصول على مثال.

تأثير آخر لـ RECURSIVE هو أن استعلامات WITH لا تحتاج إلى ترتيب: يمكن لاستعلام أن يشير إلى استعلام آخر يكون لاحقًا في القائمة. (ومع ذلك، لا يتم تنفيذ الإسنادات الدائرية، أو التكرار المتبادل.) بدون RECURSIVE، يمكن لاستعلامات WITH فقط الإشارة إلى استعلامات WITH شقيقة تكون أقدم في قائمة WITH.

عندما يكون هناك استعلامات متعددة في قائمة WITH، يجب كتابة RECURSIVE مرة واحدة فقط، مباشرة بعد WITH. تنطبق على جميع الاستعلامات في قائمة WITH، على الرغم من أنها لا تؤثر على الاستعلامات التي لا تستخدم التكرار أو الإحالات الأمامية.

Clauses

A SELECT statement can have the following clauses:

The FROM clause specifies one or more source tables for the data. For example: 

FROM items,
     manufacturers
or 
FROM items
     JOIN manufacturers ON items.manufacturer_id = manufacturers.id
The FROM clause must appear before the WHERE and GROUP BY clauses, but can appear before or after the WHERE clause if the join is between multiple tables.

For further details see FROM.

Join Constructs

A FROM clause item can be a table name, a joined table, or a sub-SELECT. When multiple items are listed separated by commas, they are implicitly cross-joined. When the keyword JOIN is used, various join types can be specified:

INNER JOIN

INNER JOIN is the default join type. It returns only rows where the join condition is met in both tables.

LEFT OUTER JOIN

LEFT OUTER JOIN returns all rows from the left table, and the matched rows from the right table. If there is no match, the result is NULL from the right side. It is useful for finding rows in one table that don't have matches in another.

RIGHT OUTER JOIN

RIGHT OUTER JOIN is the opposite of LEFT OUTER JOIN. It returns all rows from the right table, and the matched rows from the left table. If there is no match, the result is NULL from the left side.

FULL OUTER JOIN

FULL OUTER JOIN returns all rows when there is a match in either the left or the right table. It combines the results of both LEFT and RIGHT OUTER JOINs. NULLs are used where there is no match.

NATURAL JOIN

NATURAL JOIN is a join that automatically joins based on common column names in both tables. If there are no common column names, it acts like a CROSS JOIN.

USING ( column_list )

USING is similar to NATURAL JOIN but allows you to specify the columns to join on. It's shorthand for joining on equality of the named columns.

CROSS JOIN

CROSS JOIN produces the Cartesian product of the two tables, meaning it combines every row from the first table with every row from the second table. It's equivalent to an INNER JOIN without a WHERE clause.

LATERAL JOIN

LATERAL JOIN allows a subquery to reference columns from previous tables in the FROM clause. This is useful for applying table-valued functions or complex subqueries row-by-row.

NOTE: The terms LEFT JOIN, RIGHT JOIN, and FULL JOIN are synonyms for LEFT OUTER JOIN, RIGHT OUTER JOIN, and FULL OUTER JOIN, respectively.

Outer joins are useful for finding rows in one table that don't have matches in another. For example, to find all customers and any orders they might have, you could use:

SELECT ...
FROM customers LEFT JOIN orders ON customers.id = orders.customer_id;

If a table is listed on the right side of a LEFT OUTER JOIN, it is only returned if its join condition is met. Otherwise, NULL values are returned for columns from that table. This is a notational convenience, since they do nothing you couldn't do with plain FROM and WHERE.

LEFT OUTER JOIN returns all rows in the qualified Cartesian product (i.e., all combined rows that pass its join condition), plus one copy of each row in the left-hand table for which there was no right-hand row that passed the join condition. This left-hand row is extended to the full width of the joined table by inserting null values for the right-hand columns. Note that only the JOIN clause's own condition is considered while deciding which rows have matches. Outer conditions are applied afterwards.

Conversely, RIGHT OUTER JOIN returns all the joined rows, plus one row for each unmatched right-hand row (extended with nulls on the left). This is just a notational convenience, since you could convert it to a LEFT OUTER JOIN by switching the left and right tables.

FULL OUTER JOIN returns all the joined rows, plus one row for each unmatched left-hand row (extended with nulls on the right), plus one row for each unmatched right-hand row (extended with nulls on the left).

ON join_condition

join_condition is an expression resulting in a value of type boolean (similar to a WHERE clause) that specifies which rows in a join are considered to match.

USING ( join_column [, ...] ) [ AS join_using_alias ]

A clause of the form USING ( a, b, ... ) is shorthand for ON left_table.a = right_table.a AND left_table.b = right_table.b .... Also, USING implies that only one of each pair of equivalent columns will be included in the join output, not both.

If a join_using_alias name is specified, it provides a table alias for the join columns. Only the join columns listed in the USING clause are addressable by this name. Unlike a regular alias, this does not hide the names of the joined tables from the rest of the query. Also unlike a regular alias, you cannot write a column alias list — the output names of the join columns are the same as they appear in the USING list.

NATURAL

NATURAL is shorthand for a USING list that mentions all columns in the two tables that have matching names. If there are no common column names, NATURAL is equivalent to ON TRUE.

CROSS JOIN

CROSS JOIN is equivalent to INNER JOIN ON (TRUE), that is, no rows are removed by qualification. They produce a simple Cartesian product, the same result as you get from listing the two tables at the top level of FROM, but restricted by the join condition (if any).

LATERAL

The LATERAL key word can precede a sub-SELECT FROM item. This allows the sub-SELECT to refer to columns of FROM items that appear before it in the FROM list. (Without LATERAL, each sub-SELECT is evaluated independently and so cannot cross-reference any other FROM item.)

LATERAL can also precede a function-call FROM item, but in this case it is a noise word, because the function expression can refer to earlier FROM items in any case.

A LATERAL item can appear at top level in the FROM list, or within a JOIN tree. In the latter case it can also refer to any items that are on the left-hand side of a JOIN that it is on the right-hand side of.

When a FROM item contains LATERAL cross-references, evaluation proceeds as follows: for each row of the FROM item providing the cross-referenced column(s), or set of rows of multiple FROM items providing the columns, the LATERAL item is evaluated using that row or row set's values of the columns. The resulting row(s) are joined as usual with the rows they were computed from. This is repeated for each row or set of rows from the column source table(s).

The column source table(s) must be INNER or LEFT joined to the LATERAL item, else there would not be a well-defined set of rows from which to compute each set of rows for the LATERAL item. Thus, although a construct such as X RIGHT JOIN LATERAL Y is syntactically valid, it is not actually allowed for Y to reference X.

WHERE Clause

The optional WHERE clause has the general form

WHERE condition

where condition is any expression that evaluates to a result of type boolean. Any row that does not satisfy this condition will be eliminated from the output. A row satisfies the condition if it returns true when the actual row values are substituted for any variable references.

GROUP BY Clause

The optional GROUP BY clause has the general form

GROUP BY [ ALL | DISTINCT ] grouping_element [, ...]

GROUP BY will condense into a single row all selected rows that share the same values for the grouped expressions. An expression used inside a grouping_element can be an input column name, or the name or ordinal number of an output column (SELECT list item), or an arbitrary expression formed from input-column values. In case of ambiguity, a GROUP BY name will be interpreted as an input-column name rather than an output column name.

If any of GROUPING SETS, ROLLUP or CUBE are present as grouping elements, then the GROUP BY clause as a whole defines some number of independent grouping sets. The effect of this is equivalent to constructing a UNION ALL between subqueries with the individual grouping sets as their GROUP BY clauses. The optional DISTINCT clause removes duplicate sets before processing; it does not transform the UNION ALL into a UNION DISTINCT. For further details on the handling of grouping sets see Section 7.2.4.

Aggregate functions, if any are used, are computed across all rows making up each group, producing a separate value for each group. (If there are aggregate functions but no GROUP BY clause, the query is treated as having a single group comprising all the selected rows.) The set of rows fed to each aggregate function can be further filtered by attaching a FILTER clause to the aggregate function call; see Section 4.2.7 for more information. When a FILTER clause is present, only those rows matching it are included in the input to that aggregate function.

When GROUP BY is present, or any aggregate functions are present, it is not valid for the SELECT list expressions to refer to ungrouped columns except within aggregate functions or when the ungrouped column is functionally dependent on the grouped columns, since there would otherwise be more than one possible value to return for an ungrouped column. A functional dependency exists if the grouped columns (or a subset thereof) are the primary key of the table containing the ungrouped column.

Keep in mind that all aggregate functions are evaluated before evaluating any “scalar” expressions in the HAVING clause or SELECT list. This means that, for example, a CASE expression cannot be used to skip evaluation of an aggregate function; see Section 4.2.14.

Currently, FOR NO KEY UPDATE, FOR UPDATE, FOR SHARE and FOR KEY SHARE cannot be specified with GROUP BY.

HAVING Clause

The optional HAVING clause has the general form

HAVING condition

where condition is the same as specified for the WHERE clause.

HAVING eliminates group rows that do not satisfy the condition. HAVING is different from WHERE: WHERE filters individual rows before the application of GROUP BY, while HAVING filters group rows created by GROUP BY. Each column referenced in condition must unambiguously reference a grouping column, unless the reference appears within an aggregate function or the ungrouped column is functionally dependent on the grouping columns.

The presence of HAVING turns a query into a grouped query even if there is no GROUP BY clause. This is the same as what happens when the query contains aggregate functions but no GROUP BY clause. All the selected rows are considered to form a single group, and the SELECT list and HAVING clause can only reference table columns from within aggregate functions. Such a query will emit a single row if the HAVING condition is true, zero rows if it is not true.

Currently, FOR NO KEY UPDATE, FOR UPDATE, FOR SHARE and FOR KEY SHARE cannot be specified with HAVING.

WINDOW Clause

The optional WINDOW clause has the general form

WINDOW window_name AS ( window_definition ) [, ...]

where window_name is a name that can be referenced from OVER clauses or subsequent window definitions, and window_definition is

[ existing_window_name ]
[ PARTITION BY expression [, ...] ]
[ ORDER BY expression [ ASC | DESC | USING operator ] [ NULLS { FIRST | LAST } ] [, ...] ]
[ frame_clause ]

If an existing_window_name is specified it must refer to an earlier entry in the WINDOW list; the new window copies its partitioning clause from that entry, as well as its ordering clause if any. In this case the new window cannot specify its own PARTITION BY clause, and it can specify ORDER BY only if the copied window does not have one. The new window always uses its own frame clause; the copied window must not specify a frame clause.

The elements of the PARTITION BY list are interpreted in much the same fashion as elements of a GROUP BY clause, except that they are always simple expressions and never the name or number of an output column. Another difference is that these expressions can contain aggregate function calls, which are not allowed in a regular GROUP BY clause. They are allowed here because windowing occurs after grouping and aggregation.

Similarly, the elements of the ORDER BY list are interpreted in much the same fashion as elements of a statement-level ORDER BY clause, except that the expressions are always taken as simple expressions and never the name or number of an output column.

The optional frame_clause defines the window frame for window functions that depend on the frame (not all do). The window frame is a set of related rows for each row of the query (called the current row). The frame_clause can be one of

{ RANGE | ROWS | GROUPS } frame_start [ frame_exclusion ]
{ RANGE | ROWS | GROUPS } BETWEEN frame_start AND frame_end [ frame_exclusion ]

where frame_start and frame_end can be one of

UNB
ORDER BY، LIMIT، FOR NO KEY UPDATE، FOR UPDATE، FOR SHARE، أو FOR KEY SHARE عبارة.

يقوم عامل التشغيل INTERSECT بحساب تقاطع مجموعات الصفوف التي تعيدها عبارات SELECT المعنية. يكون الصف في تقاطع مجموعتي نتائج إذا ظهر في كلتا مجموعتي النتائج.

لا تحتوي نتيجة INTERSECT على صفوف مكررة ما لم يتم تحديد الخيار ALL. مع ALL، سيظهر الصف الذي يحتوي على m تكرار في الجدول الأيسر و n تكرار في الجدول الأيمن min(m,n) مرة في مجموعة النتائج. يمكن كتابة DISTINCT لتحديد السلوك الافتراضي لإزالة الصفوف المكررة بشكل صريح.

يتم تقييم عوامل تشغيل INTERSECT المتعددة في نفس عبارة SELECT من اليسار إلى اليمين، ما لم تحدد الأقواس خلاف ذلك. ترتبط INTERSECT بقوة أكبر من UNION. أي أن A UNION B INTERSECT C ستُقرأ على أنها A UNION (B INTERSECT C).

حاليًا، لا يمكن تحديد FOR NO KEY UPDATE، FOR UPDATE، FOR SHARE، و FOR KEY SHARE إما لنتيجة INTERSECT أو لأي مدخل لـ INTERSECT.

عبارة EXCEPT

تحتوي عبارة EXCEPT على هذا الشكل العام:

select_statement EXCEPT [ ALL | DISTINCT ] select_statement

select_statement هي أي عبارة SELECT بدون عبارة ORDER BY، LIMIT، FOR NO KEY UPDATE، FOR UPDATE، FOR SHARE، أو FOR KEY SHARE.

يقوم عامل التشغيل EXCEPT بحساب مجموعة الصفوف الموجودة في نتيجة عبارة SELECT اليسرى ولكن ليس في نتيجة العبارة اليمنى.

لا تحتوي نتيجة EXCEPT على صفوف مكررة ما لم يتم تحديد الخيار ALL. مع ALL، سيظهر الصف الذي يحتوي على m تكرار في الجدول الأيسر و n تكرار في الجدول الأيمن max(m-n,0) مرة في مجموعة النتائج. يمكن كتابة DISTINCT لتحديد السلوك الافتراضي لإزالة الصفوف المكررة بشكل صريح.

يتم تقييم عوامل تشغيل EXCEPT المتعددة في نفس عبارة SELECT من اليسار إلى اليمين، ما لم تحدد الأقواس خلاف ذلك. ترتبط EXCEPT بنفس المستوى مثل UNION.

حاليًا، لا يمكن تحديد FOR NO KEY UPDATE، FOR UPDATE، FOR SHARE، و FOR KEY SHARE إما لنتيجة EXCEPT أو لأي مدخل لـ EXCEPT.

عبارة ORDER BY

تحتوي عبارة ORDER BY الاختيارية على هذا الشكل العام:

ORDER BY expression [ ASC | DESC | USING operator ] [ NULLS { FIRST | LAST } ] [, ...]

تتسبب عبارة ORDER BY في فرز صفوف النتائج وفقًا للتعبير (التعبيرات) المحددة. إذا كان صفان متساويين وفقًا للتعبير الأقصى لليسار، تتم مقارنتهما وفقًا للتعبير التالي وهكذا. إذا كانا متساويين وفقًا لجميع التعبيرات المحددة، يتم إرجاعهما بترتيب يعتمد على التنفيذ.

يمكن أن يكون كل expression اسم أو رقم ترتيبي لعمود الإخراج (عنصر قائمة SELECT)، أو يمكن أن يكون تعبيرًا عشوائيًا تم تكوينه من قيم أعمدة الإدخال.

يشير الرقم الترتيبي إلى الموضع الترتيبي (من اليسار إلى اليمين) لعمود الإخراج. هذه الميزة تجعل من الممكن تحديد ترتيب بناءً على عمود ليس له اسم فريد. هذا ليس ضروريًا على الإطلاق لأنه من الممكن دائمًا تعيين اسم لعمود إخراج باستخدام عبارة AS.

من الممكن أيضًا استخدام تعبيرات عشوائية في عبارة ORDER BY، بما في ذلك الأعمدة التي لا تظهر في قائمة إخراج SELECT. وبالتالي، تكون العبارة التالية صالحة:

SELECT name FROM distributors ORDER BY code;

أحد قيود هذه الميزة هو أن عبارة ORDER BY التي تنطبق على نتيجة عبارة UNION، INTERSECT، أو EXCEPT يمكنها فقط تحديد اسم عمود الإخراج أو رقمه، وليس تعبيرًا.

إذا كان تعبير ORDER BY اسمًا بسيطًا يتطابق مع كل من اسم عمود الإخراج واسم عمود الإدخال، فستقوم ORDER BY بتفسيره على أنه اسم عمود الإخراج. هذا هو عكس الاختيار الذي ستتخذه GROUP BY في نفس الموقف. هذا التناقض تم إجراؤه ليكون متوافقًا مع معيار SQL.

اختياريًا، يمكنك إضافة الكلمة الرئيسية ASC (تصاعدي) أو DESC (تنازلي) بعد أي تعبير في عبارة ORDER BY. إذا لم يتم تحديده، فسيتم افتراض ASC افتراضيًا. بدلاً من ذلك، يمكن تحديد اسم عامل تشغيل ترتيب معين في عبارة USING. يجب أن يكون عامل تشغيل الترتيب عضوًا أقل من أو أكبر من في عائلة عوامل تشغيل B-tree. ASC يعادل عادةً USING < و DESC يعادل عادةً USING >. (ولكن يمكن لمنشئ نوع بيانات معرف من قبل المستخدم تحديد بالضبط ما هو الترتيب الافتراضي للفرز، وقد يتوافق مع عوامل تشغيل بأسماء أخرى.)

إذا تم تحديد NULLS LAST، فإن القيم الفارغة يتم فرزها بعد جميع القيم غير الفارغة؛ إذا تم تحديد NULLS FIRST، فإن القيم الفارغة يتم فرزها قبل جميع القيم غير الفارغة. إذا لم يتم تحديد أي منهما، فإن السلوك الافتراضي هو NULLS LAST عند تحديد ASC أو ضمنيًا، و NULLS FIRST عند تحديد DESC (وبالتالي، الافتراضي هو التصرف كما لو كانت القيم الفارغة أكبر من القيم غير الفارغة). عند تحديد USING، يعتمد الترتيب الافتراضي للقيم الفارغة على ما إذا كان عامل التشغيل عامل تشغيل أقل من أو أكبر من.

لاحظ أن خيارات الترتيب تنطبق فقط على التعبير الذي تليه؛ على سبيل المثال، ORDER BY x، y DESC لا يعني نفس الشيء مثل ORDER BY x DESC، y DESC.

يتم فرز بيانات السلسلة الحرفية وفقًا للترتيب اللغوي الذي ينطبق على العمود الذي يتم فرزه. يمكن تجاوز ذلك عند الحاجة عن طريق تضمين عبارة COLLATE في expression، على سبيل المثال ORDER BY mycolumn COLLATE "en_US". لمزيد من المعلومات، راجع القسم 4.2.10 والقسم 23.2.

عبارة LIMIT

تتكون عبارة LIMIT من قسمين فرعيين مستقلين:

LIMIT { count | ALL }
OFFSET start

يحدد المعامل count الحد الأقصى لعدد الصفوف التي سيتم إرجاعها، بينما يحدد start عدد الصفوف التي سيتم تخطيها قبل البدء في إرجاع الصفوف. عند تحديد كليهما، يتم تخطي start صفوف قبل البدء في عد count صفوف سيتم إرجاعها.

إذا تم تقييم التعبير count إلى NULL، فسيتم التعامل معه على أنه LIMIT ALL، أي لا يوجد حد. إذا تم تقييم start إلى NULL، فسيتم التعامل معه بنفس طريقة OFFSET 0.

قدم SQL:2008 صيغة مختلفة لتحقيق نفس النتيجة، والتي تدعمها PostgreSQL أيضًا. وهي:

OFFSET start { ROW | ROWS }
FETCH { FIRST | NEXT } [ count ] { ROW | ROWS } { ONLY | WITH TIES }

في هذه الصيغة، يتطلب المعيار أن تكون قيمة start أو count ثابتًا حرفيًا، أو معلمة، أو اسم متغير؛ كامتداد لـ PostgreSQL، يُسمح بتعبيرات أخرى، ولكنها ستحتاج بشكل عام إلى تضمينها في أقواس لتجنب الغموض. إذا تم حذف count في عبارة FETCH، فسيتم افتراض القيمة 1. يتم استخدام الخيار WITH TIES لإرجاع أي صفوف إضافية تتساوى في المكان الأخير في مجموعة النتائج وفقًا لعبارة ORDER BY؛ يعد ORDER BY إلزاميًا في هذه الحالة، ولا يُسمح بـ SKIP LOCKED. ROW و ROWS بالإضافة إلى FIRST و NEXT هي كلمات غير مؤثرة لا تؤثر على تأثيرات هذه العبارات. وفقًا للمعيار، يجب أن تأتي عبارة OFFSET قبل عبارة FETCH إذا تم تقديمهما؛ لكن PostgreSQL أكثر تساهلاً وتسمح بأي ترتيب.

عند استخدام LIMIT، من الجيد استخدام عبارة ORDER BY تقيد صفوف النتائج بترتيب فريد. وإلا ستحصل على مجموعة جزئية غير متوقعة من صفوف الاستعلام — قد تطلب الصف العاشر إلى العشرين، ولكن العاشر إلى العشرين في أي ترتيب؟ لا تعرف أي ترتيب ما لم تحدد ORDER BY.

يأخذ مخطط الاستعلام في الاعتبار LIMIT عند إنشاء مخطط استعلام، لذلك من المحتمل جدًا أن تحصل على خطط مختلفة (تؤدي إلى ترتيبات صفوف مختلفة) اعتمادًا على ما تستخدمه لـ LIMIT و OFFSET. وبالتالي، فإن استخدام قيم LIMIT و OFFSET مختلفة لتحديد مجموعات جزئية مختلفة من نتيجة الاستعلام *سيعطي نتائج غير متسقة* ما لم تفرض ترتيب نتائج يمكن التنبؤ به باستخدام ORDER BY. هذا ليس خطأ؛ إنه نتيجة متأصلة لحقيقة أن SQL لا يعد بتقديم نتائج الاستعلام بأي ترتيب معين ما لم يتم استخدام ORDER BY لتقييد الترتيب.

من الممكن حتى أن تعيد عمليات التنفيذ المتكررة لنفس استعلام LIMIT مجموعات جزئية مختلفة من صفوف الجدول، إذا لم يكن هناك ORDER BY لفرض اختيار مجموعة جزئية محددة. مرة أخرى، هذا ليس خطأ؛ يتم ببساطة عدم ضمان تحديد النتائج في هذه الحالة.

عبارة القفل

FOR UPDATE، FOR NO KEY UPDATE، FOR SHARE، و FOR KEY SHARE هي عبارات قفل؛ فهي تؤثر على كيفية قيام SELECT بقفل الصفوف أثناء الحصول عليها من الجدول.

عبارة القفل لها الشكل العام

FOR lock_strength [ OF from_reference [, ...] ] [ NOWAIT | SKIP LOCKED ]

حيث يمكن أن يكون lock_strength أحد

UPDATE
NO KEY UPDATE
SHARE
KEY SHARE

يجب أن يكون from_reference *اسم مستعار* للجدول أو *اسم الجدول* غير المخفي المشار إليه في عبارة FROM. لمزيد من المعلومات حول كل وضع قفل على مستوى الصف، راجع القسم 13.3.2.

لمنع العملية من الانتظار حتى تنتهي المعاملات الأخرى من الالتزام، استخدم إما الخيار NOWAIT أو SKIP LOCKED. مع NOWAIT، تقوم العبارة بالإبلاغ عن خطأ، بدلاً من الانتظار، إذا لم يكن من الممكن قفل صف محدد على الفور. مع SKIP LOCKED، يتم تخطي أي صفوف محددة لا يمكن قفلها على الفور. يوفر تخطي الصفوف المقفلة رؤية غير متسقة للبيانات، لذلك هذا ليس مناسبًا للعمل العام، ولكنه يمكن استخدامه لتجنب تنازع القفل مع العديد من المستهلكين الذين يصلون إلى جدول شبيه بالصفوف. لاحظ أن NOWAIT و SKIP LOCKED تنطبق فقط على الأقفال على مستوى الصف — القفل المطلوب ROW SHARE على مستوى الجدول يتم أخذه بالطريقة العادية (انظر الفصل 13). يمكنك استخدام LOCK مع خيار NOWAIT أولاً، إذا كنت بحاجة إلى الحصول على قفل على مستوى الجدول دون انتظار.

إذا تم تسمية جداول محددة في عبارة قفل، فعندئذٍ يتم قفل الصفوف التي تأتي من تلك الجداول فقط؛ أي جداول أخرى مستخدمة في SELECT يتم قراءتها ببساطة كالمعتاد. عبارة قفل بدون قائمة جداول تؤثر على جميع الجداول المستخدمة في العبارة. إذا تم تطبيق عبارة قفل على عرض أو استعلام فرعي، فإنها تؤثر على جميع الجداول المستخدمة في العرض أو الاستعلام الفرعي. ومع ذلك، لا تنطبق هذه العبارات على استعلامات WITH المشار إليها بواسطة الاستعلام الأساسي. إذا كنت تريد حدوث قفل للصفوف داخل استعلام WITH، حدد عبارة قفل داخل استعلام WITH.

يمكن كتابة عبارات قفل متعددة

حتى لو كان من الضروري تحديد سلوك قفل مختلف للجداول المختلفة. إذا تم ذكر نفس الجدول (أو تأثره ضمنيًا) بأكثر من بند قفل واحد، فإنه يتم معالجته كما لو كان محددًا فقط بأقوى بند. وبالمثل، يتم معالجة الجدول بـ NOWAIT إذا تم تحديده في أي من البنود التي تؤثر عليه. بخلاف ذلك، يتم معالجته بـ SKIP LOCKED إذا تم تحديده في أي من البنود التي تؤثر عليه.

لا يمكن استخدام بنود القفل في سياقات لا يمكن فيها تحديد الصفوف التي تم إرجاعها بشكل واضح مع صفوف جدول فردية؛ على سبيل المثال، لا يمكن استخدامها مع التجميع.

عندما يظهر بند قفل في المستوى الأعلى للاستعلام SELECT ، تكون الصفوف المقفلة هي بالضبط تلك التي يعيدها الاستعلام؛ في حالة استعلام الربط، تكون الصفوف المقفلة هي تلك التي تساهم في صفوف الربط التي تم إرجاعها. بالإضافة إلى ذلك، سيتم قفل الصفوف التي استوفت شروط الاستعلام اعتبارًا من لقطة الاستعلام، على الرغم من أنها لن يتم إرجاعها إذا تم تحديثها بعد اللقطة ولم تعد تستوفي شروط الاستعلام. إذا تم استخدام LIMIT، يتوقف القفل بمجرد إرجاع عدد كافٍ من الصفوف لتلبية الحد (ولكن لاحظ أنه سيتم قفل الصفوف التي تم تجاوزها بواسطة OFFSET). وبالمثل، إذا تم استخدام بند قفل في استعلام مؤشر، فلن يتم قفل سوى الصفوف التي تم جلبها فعليًا أو تجاوزها بواسطة المؤشر.

عندما يظهر بند قفل في استعلام فرعي SELECT ، تكون الصفوف المقفلة هي تلك التي يعيدها الاستعلام الفرعي إلى الاستعلام الخارجي. قد يتضمن ذلك عددًا أقل من الصفوف مما قد تشير إليه معاينة الاستعلام الفرعي وحده، حيث يمكن استخدام شروط من الاستعلام الخارجي لتحسين تنفيذ الاستعلام الفرعي. على سبيل المثال،

SELECT * FROM (SELECT * FROM mytable FOR UPDATE) ss WHERE col1 = 5;

سيؤدي ذلك إلى قفل الصفوف التي لها col1 = 5 فقط، حتى لو لم يكن هذا الشرط موجودًا نصيًا داخل الاستعلام الفرعي.

فشلت الإصدارات السابقة في الحفاظ على قفل يتم ترقيته بواسطة نقطة حفظ لاحقة. على سبيل المثال، هذا الكود:

BEGIN;
SELECT * FROM mytable WHERE key = 1 FOR UPDATE;
SAVEPOINT s;
UPDATE mytable SET ... WHERE key = 1;
ROLLBACK TO s;

لم ينجح في الحفاظ على قفل FOR UPDATE بعد ROLLBACK TO. تم إصلاح ذلك في الإصدار 9.3.


تنبيه

من الممكن أن يقوم أمر SELECT الذي يتم تشغيله في مستوى عزل المعاملة READ COMMITTED ويستخدم ORDER BY وبند قفل بإرجاع صفوف بترتيب خاطئ. وذلك لأن ORDER BY يتم تطبيقه أولاً. يقوم الأمر بفرز النتيجة، ولكنه قد يحظر بعد ذلك محاولة الحصول على قفل على صف واحد أو أكثر من الصفوف. بمجرد أن يتم إلغاء حظر SELECT، قد تكون قد تم تعديل بعض قيم الأعمدة المستخدمة في الترتيب، مما يؤدي إلى ظهور هذه الصفوف بترتيب خاطئ (على الرغم من أنها مرتبة من حيث قيم الأعمدة الأصلية). يمكن التغلب على ذلك عند الحاجة عن طريق وضع بند FOR UPDATE/SHARE في استعلام فرعي، على سبيل المثال

SELECT * FROM (SELECT * FROM mytable FOR UPDATE) ss ORDER BY column1;

لاحظ أن هذا سيؤدي إلى قفل جميع صفوف mytable، بينما سيؤدي FOR UPDATE في المستوى الأعلى إلى قفل الصفوف التي تم إرجاعها فعليًا فقط. قد يؤدي ذلك إلى اختلاف كبير في الأداء، خاصة إذا تم دمج ORDER BY مع LIMIT أو قيود أخرى. لذا يوصى بهذه التقنية فقط إذا كان من المتوقع حدوث تحديثات متزامنة لأعمدة الترتيب وكان مطلوبًا نتيجة مرتبة بدقة.

في مستوى عزل المعاملة REPEATABLE READ أو SERIALIZABLE، سيؤدي ذلك إلى فشل التسلسل (مع SQLSTATE '40001')، لذلك لا توجد إمكانية لاستلام صفوف بترتيب خاطئ تحت مستويات العزل هذه.

أمر TABLE

الأمر

TABLE name

يعادل

SELECT * FROM name

يمكن استخدامه كأمر على المستوى الأعلى أو كمتغير بناء جملة لتوفير المساحة في أجزاء من الاستعلامات المعقدة. فقط بنود WITH، UNION، INTERSECT، EXCEPT، ORDER BY، LIMIT، OFFSET، FETCH، و FOR يمكن استخدامها مع TABLE؛ لا يمكن استخدام بند WHERE وأي شكل من أشكال التجميع.

لاحظ أن استعلام WITH تم تقييمه مرة واحدة فقط، لذلك حصلنا على مجموعتين من نفس القيم العشوائية الثلاث.

يستخدم هذا المثال WITH RECURSIVE للعثور على جميع المرؤوسين (المباشرين أو غير المباشرين) للموظفة Mary، ومستوى عدم مباشرتهم، من جدول يظهر فقط المرؤوسين المباشرين:

WITH RECURSIVE employee_recursive(distance, employee_name, manager_name) AS (
    SELECT 1, employee_name, manager_name
    FROM employee
    WHERE manager_name = 'Mary'
  UNION ALL
    SELECT er.distance + 1, e.employee_name, e.manager_name
    FROM employee_recursive er, employee e
    WHERE er.employee_name = e.manager_name
  )
SELECT distance, employee_name FROM employee_recursive;

لاحظ الشكل النموذجي للاستعلامات المتكررة: شرط مبدئي، يليه UNION، ثم الجزء المتكرر من الاستعلام. تأكد من أن الجزء المتكرر من الاستعلام سيعيد في النهاية لا شيء، وإلا فسوف يتكرر الاستعلام بلا نهاية. (انظر القسم 7.8 لمزيد من الأمثلة.)

يستخدم هذا المثال LATERAL لتطبيق دالة إرجاع مجموعة get_product_names() لكل صف من جدول manufacturers:

SELECT m.name AS mname, pname
FROM manufacturers m, LATERAL get_product_names(m.id) pname;

لن تظهر الشركات المصنعة التي لا تحتوي حاليًا على أي منتجات في النتيجة، نظرًا لأنها ربط داخلي. إذا كنا نرغب في تضمين أسماء هذه الشركات المصنعة في النتيجة، فيمكننا القيام بما يلي:

SELECT m.name AS mname, pname
FROM manufacturers m LEFT JOIN LATERAL get_product_names(m.id) pname ON true;

التوافق

بالطبع، بيان SELECT متوافق مع معيار SQL. ولكن هناك بعض الامتدادات وبعض الميزات المفقودة.

بنود FROM المحذوفة

يتيح PostgreSQL حذف بند FROM. له استخدام مباشر لحساب نتائج التعبيرات البسيطة:

SELECT 2+2;
 ?column?
----------
        4

بعض قواعد بيانات SQL الأخرى لا تستطيع فعل ذلك إلا بإدخال جدول وهمي من صف واحد للاختيار منه.

قوائم SELECT الفارغة

يمكن أن تكون قائمة تعبيرات الإخراج بعد SELECT فارغة، مما ينتج عنه جدول نتائج بعمود صفر. هذه ليست صيغة صالحة وفقًا لمعيار SQL. تسمح PostgreSQL بذلك لتكون متوافقة مع السماح بالجداول ذات الأعمدة الصفرية. ومع ذلك، لا يُسمح بقائمة فارغة عند استخدام DISTINCT.
