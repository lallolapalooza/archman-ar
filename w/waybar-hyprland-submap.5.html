waybar-hyprland-submap(5)

الحزمة: waybar

التراخيص: MIT

waybar-hyprland-submap(5) دليل تنسيقات الملفات waybar-hyprland-submap(5)

الاسم

waybar - وحدة hyprland الفرعية

الوصف

تعرض الوحدة الفرعية submap الخريطة الفرعية النشطة حاليًا بشكل مشابه لـ sway/mode.

التكوين

يتم تناولها بواسطة hyprland/submap

format:
النوع: string
الافتراضي: {}
التنسيق، كيف يجب عرض المعلومات. في {} يتم عرض الخريطة الفرعية النشطة حاليًا.

rotate:
النوع: integer
قيمة موجبة لتدوير تسمية النص (بزيادات 90 درجة).

max-length:
النوع: integer
الحد الأقصى لطول الوحدة الذي يجب عرضه.

min-length:
النوع: integer
الحد الأدنى لطول الوحدة الذي يجب قبوله.

align:
النوع: float
محاذاة التسمية داخل الوحدة، حيث 0 هو المحاذاة إلى اليسار و 1 هو المحاذاة إلى اليمين. إذا تم تدوير الوحدة، فسوف تتبع تدفق النص.

justify:
النوع: string
محاذاة النص داخل تسمية الوحدة، مما يسمح بخيارات 'left'، 'right'، أو 'center' لتحديد الموضع.

on-click:
النوع: string
الأمر الذي يجب تنفيذه عند النقر على الوحدة.

on-click-middle:
النوع: string
الأمر الذي يجب تنفيذه عند النقر بزر الماوس الأوسط على الوحدة.

on-click-right:
النوع: string
الأمر الذي يجب تنفيذه عند النقر بزر الماوس الأيمن على الوحدة.

on-update:
النوع: string
الأمر الذي يجب تنفيذه عند تحديث الوحدة.

on-scroll-up:
النوع: string
الأمر الذي يجب تنفيذه عند التمرير لأعلى على الوحدة.

on-scroll-down:
النوع: string
الأمر الذي يجب تنفيذه عند التمرير لأسفل على الوحدة.

smooth-scrolling-threshold:
النوع: double
العتبة التي يجب استخدامها عند التمرير.

tooltip:
النوع: bool
الافتراضي: true
خيار لتعطيل التلميح عند المرور فوقه.

always-on:
النوع: bool
الافتراضي: false
خيار لعرض الأداة حتى عندما لا تكون هناك خريطة فرعية نشطة.

default-submap
النوع: string
الافتراضي: Default
خيار لتعيين اسم الخريطة الفرعية للعرض عندما لا تكون في خريطة فرعية نشطة.

menu:
النوع: string
الإجراء الذي يظهر القائمة.

menu-file:
النوع: string
موقع ملف وصف القائمة. يجب أن يكون هناك عنصر من النوع

GtkMenu مع المعرف menu

menu-actions:
النوع: array
الإجراءات المقابلة لأزرار القائمة.

expand:
النوع: bool
الافتراضي: false
يمكن هذه الوحدة من استهلاك كل المساحة المتبقية ديناميكيًا.

أمثلة

"hyprland/submap": {
	"format": "️ {}",
	"max-length": 8,
	"tooltip": false
}
