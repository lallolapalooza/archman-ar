waybar-sway-mode(5)

الحزمة: waybar

التراخيص: MIT

waybar-sway-mode(5) دليل تنسيقات الملفات waybar-sway-mode(5)

الاسم

waybar - وحدة وضع sway

الوصف

تعرض الوحدة mode وضع الربط الحالي لـ Sway

التكوين

يتم التعامل معها بواسطة sway/mode

format:
type: string
default: {}
التنسيق، كيف يجب عرض المعلومات. في {} يتم إدخال البيانات.

rotate:
type: integer
قيمة موجبة لتدوير تسمية النص (بزيادات 90 درجة).

max-length:
type: integer
الحد الأقصى لطول الوحدة التي يجب عرضها (بالأحرف).

min-length:
type: integer
الحد الأدنى لطول الوحدة التي يجب قبولها (بالأحرف).

align:
type: float
محاذاة التسمية داخل الوحدة، حيث 0 هي المحاذاة لليسار و 1 هي المحاذاة لليمين. إذا تم تدوير الوحدة، فستتبع تدفق النص.

justify:
type: string
محاذاة النص داخل تسمية الوحدة، مما يسمح بخيارات 'left'، 'right'، أو 'center' لتحديد الموضع.

on-click:
type: string
الأمر الذي سيتم تنفيذه عند النقر على الوحدة.

on-click-middle:
type: string
الأمر الذي سيتم تنفيذه عند النقر بزر الوسط (عجلة الماوس) على الوحدة.

on-click-right:
type: string
الأمر الذي سيتم تنفيذه عند النقر بزر اليمين على الوحدة.

on-update:
type: string
الأمر الذي سيتم تنفيذه عند تحديث الوحدة.

on-scroll-up:
type: string
الأمر الذي سيتم تنفيذه عند التمرير للأعلى على الوحدة.

on-scroll-down:
type: string
الأمر الذي سيتم تنفيذه عند التمرير للأسفل على الوحدة.

smooth-scrolling-threshold:
type: double
العتبة التي سيتم استخدامها عند التمرير.

tooltip:
type: bool
default: true
خيار لتعطيل تلميح الشاشة عند المرور فوقه.

menu:
type: string
الإجراء الذي يفتح القائمة.

menu-file:
type: string
موقع ملف وصف القائمة. يجب أن يكون هناك عنصر من النوع

GtkMenu بالمعرف menu

menu-actions:
type: array
الإجراءات المقابلة لأزرار القائمة.

expand:
type: bool
default: false
يمكّن هذه الوحدة من استهلاك كل المساحة المتبقية ديناميكيًا.

أمثلة

"sway/mode": {
	"format": " {}",
	"max-length": 50
}
