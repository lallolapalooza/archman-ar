waybar-custom(5)

الحزمة: waybar

التراخيص: MIT

waybar-custom(5) دليل تنسيقات الملفات waybar-custom(5)

الاسم

waybar - وحدة مخصصة

الوصف

تعرض وحدة custom إما مخرجات سكربت أو نصًا ثابتًا. لعرض نص ثابت، حدد حقل format فقط.

الإعدادات

يُشار إليها بواسطة custom/<name>

exec:
النوع: string
مسار السكربت الذي يجب تنفيذه.

exec-if:
النوع: string
مسار سكربت يحدد ما إذا كان السكربت في exec يجب تنفيذه.
سيتم تنفيذ exec إذا كان رمز الخروج لـ exec-if يساوي 0.

hide-empty-text:
النوع: bool
يعطّل الوحدة عندما تكون المخرجات فارغة، لكن قد يحتوي التنسيق على محتوى ثابت إضافي.

exec-on-event:
النوع: bool
الافتراضي: true
إذا تم تعيين أمر حدث (مثل on-click أو on-scroll-up) فسيُعاد تنفيذ السكربت بعد تنفيذ أمر الحدث.

return-type:
النوع: string
انظر return-type

interval:
النوع: integer
الفاصل الزمني (بالثواني) الذي يتم فيه استقصاء المعلومات.
استخدم once إذا كنت تريد تنفيذ الوحدة فقط عند بدء التشغيل.
يمكنك تحديثها يدويًا بإشارة. إذا لم يتم تعريف interval أو signal، فمن المفترض أن سكربت المخرجات يكرر نفسه.
إذا تم تعريف signal فسيتم تشغيل السكربت مرة واحدة عند بدء التشغيل ولن يتم تحديثه إلا بإشارة.

restart-interval:
النوع: integer
فاصل إعادة التشغيل (بالثواني).
لا يمكن استخدامه مع خيار interval، لذا فقط مع السكربتات المستمرة.
بمجرد خروج السكربت، ستُعاد إعادة تنفيذه بعد restart-interval.

signal:
النوع: integer
رقم الإشارة المستخدم لتحديث الوحدة.
الرقم صالح بين 1 و N، حيث SIGRTMIN+N = SIGRTMAX.
إذا لم يتم تعريف فاصل زمني فستكون الإشارة هي الطريقة الوحيدة لتحديث الوحدة.

format:
النوع: string
الافتراضي: {text}
التنسيق، وكيف يجب عرض المعلومات. على {text} تُدرج البيانات.

format-icons:
النوع: array
بناءً على النسبة المئوية المحددة، يتم اختيار الأيقونة المقابلة. الترتيب من low إلى high.

rotate:
النوع: integer
قيمة موجبة لتدوير تسمية النص (بزيادات 90 درجة).

max-length:
النوع: integer
الحد الأقصى للطول بالأحرف الذي يجب أن تعرضه الوحدة.

min-length:
النوع: integer
الحد الأدنى للطول بالأحرف الذي يجب أن تقبله الوحدة.

align:
النوع: float
محاذاة التسمية داخل الوحدة، حيث 0 محاذاة لليسار و1 محاذاة لليمين. إذا تم تدوير الوحدة، فستتبع اتجاه النص.

justify:
النوع: string
محاذاة النص داخل تسمية الوحدة، مع السماح بخيارات 'left' أو 'right' أو 'center' لتحديد الموضع.

on-click:
النوع: string
الأمر الذي يُنفذ عند النقر على الوحدة.

on-click-middle:
النوع: string
الأمر الذي يُنفذ عند النقر الأوسط على الوحدة باستخدام عجلة الفأرة.

on-click-right:
النوع: string
الأمر الذي يُنفذ عند النقر بزر الفأرة الأيمن على الوحدة.

on-update:
النوع: string
الأمر الذي يُنفذ عند تحديث الوحدة.

on-scroll-up:
النوع: string
الأمر الذي يُنفذ عند التمرير لأعلى على الوحدة.

on-scroll-down:
النوع: string
الأمر الذي يُنفذ عند التمرير لأسفل على الوحدة.

smooth-scrolling-threshold:
النوع: double
العتبة المستخدمة عند التمرير.

tooltip:
النوع: bool
الافتراضي: true
خيار لتعطيل التلميح عند التحويم.

tooltip-format:
النوع: string
تنسيق التلميح. إذا تم تحديده، فإنه يتجاوز أي مخرجات تلميح من السكربت في exec.
يستخدم نفس بدائل التنسيق مثل format.

escape:
النوع: bool
الافتراضي: false
خيار لتمكين تهريب مخرجات السكربت.

menu:
النوع: string
الإجراء الذي ينبثق القائمة.

menu-file:
النوع: string
موقع ملف وصف القائمة. يجب أن يكون هناك عنصر من النوع

GtkMenu بالمعرف menu

menu-actions:
النوع: array
الإجراءات المقابلة لأزرار القائمة.

expand:
النوع: bool
الافتراضي: false
يمكّن هذه الوحدة من استهلاك كل المساحة المتبقية ديناميكيًا.

نوع الإرجاع

عندما يتم تعيين return-type إلى json، تتوقع Waybar أن يُخرج سكربت exec بياناته بتنسيق JSON. يجب أن يبدو هكذا:

{"text": "$text", "tooltip": "$tooltip", "class": "$class", "percentage": $percentage }

تقبل معلمة class أيضًا مصفوفة من السلاسل النصية.

إذا لم يتم تحديد شيء أو تم تحديد خيار غير صالح، تتوقع Waybar مخرجات بنمط i3blocks. القيم مفصولة بـ سطر جديد. يجب أن يبدو هكذا:

$text\n$tooltip\n$class*

class هو صنف CSS، لتطبيق أنماط مختلفة في style.css

بدائل التنسيق

{text}: مخرجات السكربت.

{percentage} النسبة المئوية التي يمكن تعيينها عبر نوع إرجاع json.

{icon}: أيقونة من 'format-icons' وفقًا للنسبة المئوية.

أمثلة

Spotify:

"custom/spotify": {
	"format": " {text}",
	"max-length": 40,
	"interval": 30, // أزل هذا إذا كان سكربتك لا نهائيًا ويكتب في حلقة
	"exec": "$HOME/.config/waybar/mediaplayer.sh 2> /dev/null", // سكربت في مجلد resources
	"exec-if": "pgrep spotify",
	"return-type": "json"
}

mpd:

"custom/mpd": {
	"format": "♪ {text}",
	//"max-length": 15,
	"interval": 10,
	"exec": "mpc current",
	"exec-if": "pgrep mpd",
	"on-click": "mpc toggle",
	"on-click-right": "sonata"
}

cmus:

"custom/cmus": {
	"format": "♪ {text}",
	//"max-length": 15,
	"interval": 10,
	"exec": "cmus-remote -C "format_print '%a - %t'"", // الفنان - العنوان
	"exec-if": "pgrep cmus",
	"on-click": "cmus-remote -u",                        //تبديل الإيقاف المؤقت
	"escape": true                                       //التعامل مع كيانات الترميز
}

Pacman

"custom/pacman": {
	"format": "{text}  ",
	"interval": "once",
	"exec": "pacman_packages",
	"on-click": "update-system",
	"signal": 8
}

Pacman بديل

"custom/pacman": {
	"format": "{text}  ",
	"interval": 3600,                     // كل ساعة
	"exec": "checkupdates | wc -l",       // عدد التحديثات
	"exec-if": "exit 0",                  // يعمل دائمًا؛ فكر في شروط تشغيل متقدمة
	"on-click": "termite -e 'sudo pacman -Syu'; pkill -SIGRTMIN+8
waybar"، // تحديث النظام "signal": 8 }

يمكنك استخدام الإشارة وتحديث عدد الحزم المتاحة باستخدام pkill -RTMIN+8 waybar.

STYLE

  • #custom-<name>
  • #custom-<name>.<class>
  • يمكن تعيين <class> بواسطة السكربت. لمزيد من المعلومات راجع return-type
