wl_display(3)

الحزمة: wayland-docs

التراخيص: MIT

من جانب العميل.

سيتم إخطار المستمعين المضافين باستخدام wl_display_add_client_created_listener() بواسطة هذه الوظيفة بعد اكتمال بناء العميل.

عند الفشل، تقوم هذه الوظيفة بتعيينerrno وفقًا لذلك وترجع NULL.

عند النجاح، يتولى كائن العميل الجديد ملكية واصف الملف. عند الفشل، تظل ملكية واصف ملف نقطة نهاية المقبس دون تغيير، وتقع على عاتق المتصل مسؤولية التنظيف، على سبيل المثال، استدعاء close().

struct wl_protocol_logger * wl_display_add_protocol_logger (struct wl_display * display, wl_protocol_logger_func_t func, void * user_data)

يضيف مسجل بروتوكول جديد.

عند وصول رسالة بروتوكول جديدة أو إرسالها من الخادم، سيتم استدعاء جميع وظائف مسجل البروتوكول، مع تمرير مؤشر user_data، ونوع الرسالة (طلب أو حدث)، والرسالة الفعلية. تنتهي مدة صلاحية الرسائل التي يتم تمريرها إلى وظيفة المسجل عند عودتها، لذلك لا يمكن تخزين الرسائل والوصول إليها لاحقًا.

errno يتم تعيينه عند حدوث خطأ.

المعلمات

display كائن العرض
func الوظيفة التي سيتم استدعاؤها لتسجيل رسالة بروتوكول جديدة
user_data مؤشر بيانات المستخدم لتمريره إلى func

القيم المرجعة

كائن مسجل البروتوكول عند النجاح، NULL عند الفشل.

uint32_t * wl_display_add_shm_format (struct wl_display * display, uint32_t format)

إضافة دعم لتنسيق بكسل wl_shm

المعلمات

display كائن العرض
format تنسيق بكسل wl_shm المراد الإعلان عنه

القيم المرجعة

مؤشر إلى تنسيق wl_shm الذي تمت إضافته إلى القائمة أو NULL إذا فشلت إضافته إلى القائمة.

يضيف تنسيق wl_shm المحدد إلى قائمة التنسيقات التي يعلن عنها كائن wl_shm عند ربط عميل به. إضافة تنسيق إلى القائمة يعني أن العملاء سيعرفون أن المجمع يدعم هذا التنسيق وقد يستخدمه لإنشاء مخازن مؤقتة wl_shm. يجب أن يكون المجمع قادرًا على التعامل مع تنسيق البكسل عند طلبه من قبل عميل.

يدعم المجمع افتراضيًا WL_SHM_FORMAT_ARGB8888 و WL_SHM_FORMAT_XRGB8888.

int wl_display_add_socket (struct wl_display * display, const char * name)

إضافة مقبس إلى عرض Wayland لكي تتصل به العملاء.

المعلمات

display عرض Wayland الذي يجب إضافة المقبس إليه.
name اسم مقبس Unix.

القيم المرجعة

0 في حالة النجاح. -1 في حالة الفشل.

هذا يضيف مقبس Unix إلى عرض Wayland والذي يمكن للعملاء استخدامه للاتصال بعرض Wayland.

إذا تم تمرير NULL كاسم، فسيتم البحث عن متغير البيئة WAYLAND_DISPLAY لاسم المقبس. إذا لم يتم تعيين WAYLAND_DISPLAY، فسيتم استخدام wayland-0 الافتراضي.

إذا كان اسم المقبس مسارًا نسبيًا، فسيتم إنشاء مقبس Unix في الدليل الذي يشير إليه متغير البيئة XDG_RUNTIME_DIR. إذا كان XDG_RUNTIME_DIR غير صالح أو لم يتم تعيينه، تفشل هذه الوظيفة وتُرجع -1.

إذا كان اسم المقبس مسارًا مطلقًا، فسيتم استخدامه كما هو لمقبس Unix.

يجب ألا يتجاوز طول مسار المقبس المحسوب الحد الأقصى لطول مسار مقبس Unix. تفشل الوظيفة أيضًا إذا لم يكن لدى المستخدم إذن الكتابة في الدليل أو إذا كان المسار مستخدمًا بالفعل.

struct wl_display * wl_display_connect (const char * name)

الاتصال بعرض Wayland

المعلمات

name اسم عرض Wayland للاتصال به

القيم المرجعة

كائن wl_display أو NULL عند الفشل

الاتصال بعرض Wayland المسمى name. إذا كان name هو NULL، فسيتم استبدال قيمته بمتغير البيئة WAYLAND_DISPLAY إذا تم تعيينه، وإلا فسيتم استخدام العرض "wayland-0".

إذا تم تعيين WAYLAND_SOCKET، فسيتم تفسيره كرقم واصف ملف يشير إلى مقبس مفتوح بالفعل. في هذه الحالة، يتم استخدام المقبس كما هو ويتم تجاهل name.

إذا كان name مسارًا نسبيًا، فسيتم فتح المقبس بالنسبة إلى دليل XDG_RUNTIME_DIR.

إذا كان name مسارًا مطلقًا، فسيتم استخدام هذا المسار كما هو لموقع المقبس الذي يستمع إليه خادم Wayland؛ لا يتم إجراء أي تأهيل داخل XDG_RUNTIME_DIR.

إذا كان name هو NULL وتم تعيين متغير البيئة WAYLAND_DISPLAY إلى اسم مسار مطلق، فسيتم استخدام اسم المسار هذا كما هو للمقبس بنفس طريقة استخدام name كاسم مسار مطلق. دعم المسارات المطلقة في name و WAYLAND_DISPLAY موجود منذ إصدار Wayland 1.15.

المعلمات

display كائن سياق العرض

القيمة المعادة

عدد الأحداث التي تم توزيعها بنجاح أو -1 في حالة الفشل

توزيع الأحداث على طابور الأحداث الافتراضي.

إذا كان طابور الأحداث الافتراضي فارغًا، فستتوقف هذه الدالة عن العمل حتى تكون هناك أحداث للقراءة من واجهة عرض العرض. يتم قراءة الأحداث وتجميعها في طوابير الأحداث المناسبة. أخيرًا، يتم توزيع الأحداث الموجودة في طابور الأحداث الافتراضي. في حالة الفشل، يتم إرجاع -1 وتعيين errno بشكل مناسب.

في بيئة متعددة الخيوط، لا تنتظر يدويًا باستخدام poll() (أو ما يعادله) قبل استدعاء هذه الدالة، فقد يؤدي ذلك إلى حدوث توقف تام. إذا كان الاعتماد الخارجي على poll() (أو ما يعادله) مطلوبًا، فراجع wl_display_prepare_read_queue() لمعرفة كيفية القيام بذلك.

هذه الدالة آمنة للخيوط طالما أنها توزع الطابور الصحيح على الخيط الصحيح. وهي متوافقة أيضًا مع واجهة برمجة التطبيقات لإعداد قراءة الأحداث متعددة الخيوط (راجع wl_display_prepare_read_queue())، وتستخدم وظائف معادلة داخليًا. لا يُسمح باستدعاء هذه الدالة أثناء إعداد الخيط لقراءة الأحداث، وسيؤدي القيام بذلك إلى حدوث توقف تام.

ملاحظة

لا يمكن التحقق مما إذا كانت هناك أحداث في الطابور أم لا. لتوزيع أحداث الطابور الافتراضي دون حظر، راجع wl_display_dispatch_pending().

int wl_display_flush (struct wl_display * display)

إرسال جميع الطلبات المخزنة مؤقتًا على العرض إلى الخادم

المعلمات

display كائن سياق العرض

القيمة المعادة

عدد البايتات المرسلة بنجاح أو -1 في حالة الفشل

إرسال جميع البيانات المخزنة مؤقتًا على جانب العميل إلى الخادم. يجب على العملاء دائمًا استدعاء هذه الدالة قبل الحظر عند انتظار إدخال من واجهة عرض العرض. عند النجاح، يتم إرجاع عدد البايتات المرسلة إلى الخادم. عند الفشل، تعيد هذه الدالة -1 ويتم تعيين errno بشكل مناسب.

wl_display_flush() لا تتوقف أبدًا. ستقوم بكتابة أكبر قدر ممكن من البيانات، ولكن إذا لم يكن بالإمكان كتابة جميع البيانات، فسيتم تعيين errno إلى EAGAIN وسيتم إرجاع -1. في هذه الحالة، استخدم poll على واجهة عرض العرض للانتظار حتى تصبح قابلة للكتابة مرة أخرى.

void wl_display_set_global_filter (struct wl_display * display, wl_display_global_filter_func_t filter, void * data)

تعيين وظيفة تصفية للكائنات العامة

المعلمات

display كائن عرض Wayland.
filter وظيفة التصفية العامة.
data بيانات المستخدم المراد ربطها بتصفية الكائنات العامة.

تعيين تصفية لـ wl_display للإعلان عن كائنات عامة أو إخفائها عن العملاء. ستُستخدم التصفية المحددة أثناء الإعلان عن wl_global لتحديد ما إذا كان ينبغي الإعلان عن كائن عام لعميل معين، وأثناء ربط wl_global لتحديد ما إذا كان ينبغي السماح لعميل معين بالربط بكائن عام.

سيتم رفع خطأ للعملاء الذين يحاولون الربط بكائن عام تم تصفيته.

سيؤدي تعيين التصفية إلى NULL إلى الإعلان عن جميع الكائنات العامة لجميع العملاء. الافتراضي هو عدم وجود تصفية.

يجب تثبيت التصفية قبل اتصال أي عميل ويجب دائمًا اتخاذ نفس القرار بالنظر إلى عميل وكائن عام. عدم القيام بذلك سيؤدي إلى تصفية غير متسقة وتسلسل أحداث wl_registry معطل.

void wl_display_set_max_buffer_size (struct wl_display * display, size_t max_buffer_size)

تعديل الحد الأقصى لحجم مخازن اتصال العملاء

المعلمات

display سياق العرض.
max_buffer_size الحد الأقصى لحجم مخازن الاتصال.

مخازن العملاء غير محدودة افتراضيًا. تحدد هذه الوظيفة حدًا لحجم مخازن الاتصال.

القيمة 0 لـ max_buffer_size تطلب أن تكون المخازن غير محدودة.

الحجم الفعلي لمخازن الاتصال هو قوة العدد اثنين، ولذلك يتم تقريب max_buffer_size المطلوب إلى أقرب قوة للعدد اثنين.

قد لا يؤخذ خفض الحد الأقصى للحجم في الاعتبار على الفور إذا كان المحتوى الحالي للمخزن المؤقت لا يتناسب مع حد الحجم الجديد.
