fido_bio_enroll_new(3)

الاسم

fido_bio_enroll_new، fido_bio_enroll_free، fido_bio_enroll_last_status، fido_bio_enroll_remaining_samples — واجهة برمجة تطبيقات FIDO2 للتسجيل البيومتري

الملخص

#include <fido.h>
#include <fido/bio.h>

#define FIDO_BIO_ENROLL_FP_GOOD				0x00
#define FIDO_BIO_ENROLL_FP_TOO_HIGH			0x01
#define FIDO_BIO_ENROLL_FP_TOO_LOW			0x02
#define FIDO_BIO_ENROLL_FP_TOO_LEFT			0x03
#define FIDO_BIO_ENROLL_FP_TOO_RIGHT			0x04
#define FIDO_BIO_ENROLL_FP_TOO_FAST			0x05
#define FIDO_BIO_ENROLL_FP_TOO_SLOW			0x06
#define FIDO_BIO_ENROLL_FP_POOR_QUALITY			0x07
#define FIDO_BIO_ENROLL_FP_TOO_SKEWED			0x08
#define FIDO_BIO_ENROLL_FP_TOO_SHORT			0x09
#define FIDO_BIO_ENROLL_FP_MERGE_FAILURE		0x0a
#define FIDO_BIO_ENROLL_FP_EXISTS			0x0b
#define FIDO_BIO_ENROLL_FP_DATABASE_FULL		0x0c
#define FIDO_BIO_ENROLL_NO_USER_ACTIVITY		0x0d
#define FIDO_BIO_ENROLL_NO_USER_PRESENCE_TRANSITION	0x0e

الوصف

تُجرى عمليات تسجيل FIDO2 البيومترية المستمرة وتُختصر في libfido2 بواسطة النوع fido_bio_enroll_t.

الوظائف الموصوفة في هذه الصفحة تسمح بتخصيص نوع fido_bio_enroll_t، وتحريره، وفحصه. لعمليات الأجهزة على fido_bio_enroll_t، يرجى الرجوع إلى fido_bio_dev_get_info(3).

تُرجع الدالة fido_bio_enroll_new() مؤشرًا إلى نوع fido_bio_enroll_t جديد مخصص وفارغ. إذا تعذر تخصيص الذاكرة، يتم إرجاع NULL.

تقوم الدالة fido_bio_enroll_free() بتحرير الذاكرة التي تدعم *enroll_p، حيث يجب أن يكون *enroll_p قد تم تخصيصه مسبقًا بواسطة fido_bio_enroll_new(). عند العودة، يتم تعيين *enroll_p إلى NULL. يمكن أن يكون أي من enroll_p أو *enroll_p هو NULL، وفي هذه الحالة تكون fido_bio_enroll_free() عملية فارغة (NOP).

تُرجع الدالة fido_bio_enroll_last_status() حالة التسجيل الخاصة بـ enroll.

تُرجع الدالة fido_bio_enroll_remaining_samples() عدد العينات المتبقية لإكمال enroll.
