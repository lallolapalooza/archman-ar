FAT_IOCTL_GET_VOLUME_ID(2const) FAT_IOCTL_GET_VOLUME_ID(2const)

الاسم

FAT_IOCTL_GET_VOLUME_ID - قراءة معرف الوحدة في نظام ملفات FAT

المكتبة

مكتبة C القياسية (libc، -lc)

الملخص

#include <linux/msdos_fs.h>  /* تعريف الثوابت FAT_* */
#include <sys/ioctl.h>
int ioctl(int fd, FAT_IOCTL_GET_VOLUME_ID, uint32_t *id);

الوصف

تُعرف أنظمة ملفات FAT بمعرف وحدة. يمكن قراءة معرف الوحدة باستخدام FAT_IOCTL_GET_VOLUME_ID.

يمكن أن يكون الوسيط fd واصف ملف لأي ملف أو دليل في نظام الملفات. يكفي إنشاء واصف الملف عن طريق استدعاء open(2) مع العلامة O_RDONLY.

الوسيط id هو مؤشر إلى الحقل الذي سيتم ملؤه بمعرف الوحدة. عادةً ما يتم عرض معرف الوحدة للمستخدم كمجموعة من حقلين بحجم 16 بت:


printf("Volume ID %04x-%04x\n", id >> 16, id & 0xFFFF);

قيمة الإرجاع

عند النجاح، يتم إرجاع 0. عند الخطأ، يتم إرجاع -1، ويتم تعيين errno للإشارة إلى الخطأ.

المعايير

Linux.

التاريخ

Linux 3.11.

أمثلة

يوضح البرنامج التالي استخدام ioctl(2) لعرض معرف وحدة نظام ملفات FAT.

تم تسجيل الإخراج التالي عند تطبيق البرنامج للدليل /mnt/user:


$ ./display_fat_volume_id /mnt/user
Volume ID 6443-6241

مصدر البرنامج (display_fat_volume_id.c)

#include <fcntl.h>
#include <linux/msdos_fs.h>
#include <stdint.h>
#include <stdio.h>
#include <stdlib.h>
#include <sys/ioctl.h>
#include <unistd.h>
int
main(int argc, char *argv[])
{
    int       fd;
    int       ret;
    uint32_t  id;
    if (argc != 2) {
        printf("Usage: %s FILENAME\n", argv[0]);
        exit(EXIT_FAILURE);
    }
    fd = open(argv[1], O_RDONLY);
    if (fd == -1) {
        perror("open");
        exit(EXIT_FAILURE);
    }
    /*
     * Read volume ID.
     */
    ret = ioctl(fd, FAT_IOCTL_GET_VOLUME_ID, &id);
    if (ret == -1) {
        perror("ioctl");
        exit(EXIT_FAILURE);
    }
    /*
     * Format the output as two groups of 16 bits each.
     */
    printf("Volume ID %04x-%04x\n", id >> 16, id & 0xFFFF);
    close(fd);
    exit(EXIT_SUCCESS);
}
