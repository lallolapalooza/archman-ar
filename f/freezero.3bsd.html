freezero(3bsd)

الحزمة: libbsd

التراخيص: custom

reallocarray(3bsd) 3bsd reallocarray(3bsd)

الاسم

reallocarray، recallocarray، freezero — تخصيص وإلغاء تخصيص الذاكرة

المكتبة

المكتبة "libbsd"

الملخص

#include <stdlib.h> (انظر libbsd(7) لاستخدام التضمين.)
void *
reallocarray(void *ptr, size_t nmemb, size_t size);

void *
recallocarray(void *ptr, size_t oldnmemb, size_t nmemb, size_t size);

void
freezero(void *ptr, size_t size);

الوصف

تم تصميم الدالة reallocarray() للتخصيص الآمن للمصفوفات، وهي تشبه الدالة realloc() باستثناء أنها تعمل على nmemb عضو بحجم size وتتحقق من تجاوز الأعداد الصحيحة في الحساب nmemb * size.

تم تصميم الدالة recallocarray() لتخصيص الذاكرة التي تحتوي على بيانات حساسة، وهي تضمن مسح الذاكرة التي لم تعد مخصصة بشكل صريح . هذا يعني أن الكائنات المجانية المخزنة مؤقتًا يتم مسحها باستخدام explicit_bzero(3).

الدالة recallocarray() تشبه الدالة reallocarray() باستثناء أنها تضمن مسح الذاكرة المخصصة حديثًا بشكل مشابه للدالة calloc(). إذا كان ptr يساوي NULL، يتم تجاهل oldnmemb وتكون المكالمة مكافئة للدالة calloc(). إذا لم يكن ptr يساوي NULL، فيجب أن تكون oldnmemb قيمة بحيث يكون oldnmemb * size هو حجم التخصيص السابق الذي أعاد ptr، وإلا فإن السلوك غير محدد. الدالة freezero() تشبه الدالة free() باستثناء أنها تضمن مسح الذاكرة بشكل صريح. إذا كان ptr يساوي NULL، فلا يحدث أي إجراء. إذا لم يكن ptrNULL، يجب أن يكون الوسيط size مساويًا أو أصغر من حجم التخصيص السابق الذي أعاد ptr. تضمن الدالة freezero() مسح نطاق الذاكرة بدءًا من ptrsize أثناء إلغاء تخصيص الكائن بأكمله الذي تم تخصيصه في الأصل.

قيم الإرجاع

تعيد الدالتان reallocarray() و recallocarray() مؤشرًا إلى المساحة المخصصة إذا نجحت؛ وإلا، يتم إرجاع مؤشر فارغ ويتم تعيين errno إلى ENOMEM.

إذا كان ضرب nmemb و size يؤدي إلى تجاوز الأعداد الصحيحة، فإن reallocarray() و recallocarray() ترجع NULL وتعين errno إلى ENOMEM.

إذا لم يكن ptr يساوي NULL وكان ضرب oldnmemb و size يؤدي إلى تجاوز الأعداد الصحيحة، فإن recallocarray() ترجع NULL وتعين errno إلى EINVAL.

أنماط شائعة

فكر في calloc() أو الامتدادات reallocarray() و recallocarray() عندما يكون هناك ضرب في الوسيط size للدالة malloc() أو realloc(). على سبيل المثال، تجنب هذا النمط الشائع لأنه قد يؤدي إلى تجاوز الأعداد الصحيحة:

if ((p = malloc(num * size)) == NULL)
	err(1, NULL);

بديل مباشر هو reallocarray():

if ((p = reallocarray(NULL, num, size)) == NULL)
	err(1, NULL);

بدلاً من ذلك، يمكن استخدام calloc() بتكلفة حمل التهيئة الزائد.

عند استخدام realloc()، كن حذرًا لتجنب النمط التالي:

size += 50;
if ((p = realloc(p, size)) == NULL)
	return (NULL);

لا تقم بتعديل المتغير الذي يصف مقدار الذاكرة التي تم تخصيصها حتى ينجح التخصيص. يمكن أن يؤدي هذا إلى سلوك شاذ للبرنامج إذا تم استخدام قيمة حجم غير صحيحة. في معظم الحالات، ستؤدي العينة أعلاه أيضًا إلى تسرب الذاكرة. كما ذكرنا سابقًا، يشير إلغاء الإرجاع NULL إلى أن الكائن القديم لا يزال مخصصًا. الكود الأفضل يبدو كالتالي:

newsize = size + 50;
if ((newp = realloc(p, newsize)) == NULL) {
	free(p);
	p = NULL;
	size = 0;
	return (NULL);
}
p = newp;
size = newsize;

كما هو الحال مع malloc()، من المهم التأكد من أن قيمة الحجم الجديدة لن تتجاوز؛ أي تجنب التخصيصات مثل التالية:

if ((newp = realloc(p, num * size)) == NULL) {
	...

بدلاً من ذلك، استخدم reallocarray():

if ((newp = reallocarray(p, num, size)) == NULL) {
	...

تعد استدعاءات realloc() باستخدام NULL ptr مكافئة لاستدعاء malloc(). بدلاً من هذا النمط:

if (p == NULL)
	newp = malloc(newsize);
else
	newp = realloc(p, newsize);

استخدم ما يلي:

newp = realloc(p, newsize);

يجب استخدام الدالة recallocarray() لتغيير حجم الكائنات التي تحتوي على بيانات حساسة مثل المفاتيح. لتجنب تسرب المعلومات، تضمن مسح الذاكرة قبل وضعها في قائمة الكائنات المجانية الداخلية. يجب أن يتم إلغاء تخصيص كائن كهذا عن طريق استدعاء freezero().

انظر أيضًا

malloc(3)، calloc(3)، alloca(3)
