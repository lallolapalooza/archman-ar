الاسم

getpwent_r، fgetpwent_r - الحصول على إدخال ملف passwd بطريقة قابلة لإعادة الدخول

المكتبة

مكتبة C القياسية (libc،  -lc)

الملخص

#include <pwd.h>
int getpwent_r(size_t size;
               struct passwd *restrict pwbuf,
               char buf[restrict size], size_t size,
               struct passwd **restrict pwbufp);
int fgetpwent_r(size_t size;
               FILE *restrict stream, struct passwd *restrict pwbuf,
               char buf[restrict size], size_t size,
               struct passwd **restrict pwbufp);
متطلبات ماكرو اختبار الميزات لـ glibc (انظر feature_test_macros(7)):

الوصف

الدالتان getpwent_r() و fgetpwent_r() هما النسختان القابلتان لإعادة الدخول من getpwent(3) و fgetpwent(3). الأولى تقرأ إدخال passwd التالي من التدفق الذي تم تهيئته بواسطة setpwent(3). الثانية تقرأ إدخال passwd التالي من stream.

تم تعريف بنية passwd في <pwd.h> على النحو التالي:


struct passwd {
    char    *pw_name;      /* اسم المستخدم */
    char    *pw_passwd;    /* كلمة مرور المستخدم */
    uid_t    pw_uid;       /* معرف المستخدم */
    gid_t    pw_gid;       /* معرف المجموعة */
    char    *pw_gecos;     /* معلومات المستخدم */
    char    *pw_dir;       /* دليل المنزل */
    char    *pw_shell;     /* برنامج الصدفة */
};

لمزيد من المعلومات حول حقول هذه البنية، راجع passwd(5).

الدوال غير القابلة لإعادة الدخول تعيد مؤشرًا إلى مساحة تخزين ثابتة، حيث تحتوي هذه المساحة الثابتة على مؤشرات إضافية لاسم المستخدم، كلمة المرور، حقل gecos، دليل المنزل، والصدفة. الدوال القابلة لإعادة الدخول الموضحة هنا تعيد كل ذلك في مخازن مؤقتة مقدمة من المتصل. أولاً، هناك المخزن المؤقت pwbuf الذي يمكنه استيعاب struct passwd. وبعد ذلك، المخزن المؤقت buf بحجم size يمكنه استيعاب سلاسل نصية إضافية. يتم تخزين نتيجة هذه الدوال، struct passwd المقروءة من التدفق، في المخزن المؤقت المقدم *pwbuf، ويتم إعادة مؤشر إلى هذا struct passwd في *pwbufp.

قيمة الإرجاع

عند النجاح، تعيد هذه الدوال 0 ويكون *pwbufp مؤشرًا إلى struct passwd. عند حدوث خطأ، تعيد هذه الدوال قيمة خطأ ويكون *pwbufp هو NULL.

الأخطاء

ENOENT
لا توجد إدخالات أخرى.
ERANGE
مساحة المخزن المؤقت المقدمة غير كافية. حاول مرة أخرى بمخزن مؤقت أكبر.

السمات

لشرح المصطلحات المستخدمة في هذا القسم، انظر attributes(7).

الواجهة السمة القيمة
getpwent_r () سلامة الخيوط MT-Unsafe race:pwent locale
fgetpwent_r () سلامة الخيوط MT-Safe

في الجدول أعلاه، يشير pwent في race:pwent إلى أنه إذا تم استخدام أي من الدوال setpwent() أو getpwent() أو endpwent() أو getpwent_r() بالتوازي في خيوط مختلفة من البرنامج، فقد تحدث سباقات بيانات.

الإصدارات

تستخدم أنظمة أخرى النموذج الأولي


struct passwd *
getpwent_r(struct passwd *pwd, char buf[.size], int size);

أو، بشكل أفضل،


int
getpwent_r(struct passwd *pwd, char buf[.size], int size,
           FILE **pw_fp);

المعايير

لا شيء.

التاريخ

تم تنفيذ هذه الدوال بأسلوب يشبه إصدار POSIX للدوال مثل getpwnam_r(3).

ملاحظات

الدالة getpwent_r() ليست قابلة لإعادة الدخول حقًا لأنها تشترك في موضع القراءة في التدفق مع جميع الخيوط الأخرى.

أمثلة

#define _GNU_SOURCE
#include <pwd.h>
#include <stdint.h>
#include <stdio.h>
#include <stdlib.h>
#define BUFLEN 4096
int
main(void)
{
    struct passwd pw;
    struct passwd *pwp;
    char buf[BUFLEN];
    int i;
    setpwent();
    while (1) {
        i = getpwent_r(&pw, buf, sizeof(buf), &pwp);
        if (i)
            break;
        printf("%s (%jd)\tHOME %s\tSHELL %s\n", pwp->pw_name,
               (intmax_t) pwp->pw_uid, pwp->pw_dir, pwp->pw_shell);
    }
    endpwent();
    exit(EXIT_SUCCESS);
}
