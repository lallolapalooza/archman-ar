الاسم

fi_domain - فتح نطاق وصول إلى النسيج

ملخص

#include <rdma/fabric.h>
#include <rdma/fi_domain.h>
int fi_domain(struct fid_fabric *fabric, struct fi_info *info,
    struct fid_domain **domain, void *context);
int fi_domain2(struct fid_fabric *fabric, struct fi_info *info,
    struct fid_domain **domain, uint64_t flags, void *context);
int fi_close(struct fid *domain);
int fi_domain_bind(struct fid_domain *domain, struct fid *eq,
    uint64_t flags);
int fi_open_ops(struct fid *domain, const char *name, uint64_t flags,
    void **ops, void *context);
int fi_set_ops(struct fid *domain, const char *name, uint64_t flags,
    void *ops, void *context);

الوسائط

fabric
نطاق النسيج
info
معلومات النسيج، بما في ذلك إمكانيات النطاق وسماته. يجب أن يكون الهيكل fi_info قد تم الحصول عليه باستخدام fi_getinfo() أو fi_dupinfo().
domain
نطاق وصول مفتوح.
context
سياق محدد من قبل المستخدم مرتبط بالنطاق. يتم إرجاع هذا السياق كجزء من أي حدث غير متزامن مرتبط بالنطاق.
eq
قائمة انتظار الأحداث للعمليات غير المتزامنة التي بدأت على النطاق.
name
اسم مرتبط بواجهة.
ops
عمليات واجهة النسيج.

الوصف

يشير نطاق الوصول عادةً إلى بطاقة شبكة فعلية أو افتراضية أو منفذ جهاز؛ ومع ذلك، قد يمتد نطاق عبر مكونات أجهزة متعددة لتحقيق تجاوز الفشل أو تقسيم البيانات. يحدد النطاق حدود ربط الموارد المختلفة معًا. قد تتشارك موارد النسيج التي تنتمي إلى نفس النطاق في الموارد.

fi_domain

يفتح نطاق وصول إلى النسيج، يُشار إليه أيضًا باسم نطاق الموارد. يتم تحديد نطاقات النسيج بواسطة اسم. يتم تحديد خصائص النطاق المفتوح باستخدام المعلمة info.

fi_domain2

على غرار fi_domain، ولكنه يقبل معلمة إضافية flags. يستخدم بشكل أساسي لفتح نطاق نظير. انظر fi_peer(3).

fi_open_ops

يستخدم fi_open_ops لفتح واجهات خاصة بالمزود. يمكن استخدام واجهات المزود للوصول إلى موارد وعمليات منخفضة المستوى خاصة بنطاق الموارد المفتوح. تفاصيل واجهات النطاق خارج نطاق هذه الوثائق.

fi_set_ops

يقوم fi_set_ops بتعيين ردود الاتصال التي يجب على المزود استدعاؤها بدلاً من تنفيذ مهام محددة. يسمح هذا للمستخدمين بتعديل أو التحكم في السلوك الافتراضي للمزود. من الناحية المفاهيمية، يسمح للمستخدم بربط وظائف محددة يستخدمها المزود واستبدالها بوظائفه الخاصة.

يتم تحديد العمليات التي يتم تعديلها باستخدام سلسلة أحرف معروفة، يتم تمريرها كمعلمة name. يعتمد تنسيق المعلمة ops على قيمة name. ستشير المعلمة ops إلى هيكل يحتوي على ردود الاتصال والحقول الأخرى التي يحتاجها المزود لاستدعاء وظائف المستخدم.

إذا قبل المزود الاستبدال، فسيعيد FI_SUCCESS. إذا كان الاستبدال غير معروف أو غير مدعوم، فسيعيد المزود -FI_ENOSYS. يجب تعيين عمليات الاستبدال قبل تخصيص الموارد على النطاق.

تم تعريف عمليات fi_set_ops وهياكل ردود الاتصال المقابلة التالية.

FI_SET_OPS_HMEM_OVERRIDE – استبدالات الذاكرة غير المتجانسة

يسمح استبدال HMEM للمستخدمين بتجاوز عمليات HMEM التي قد يقوم بها المزود. حاليًا، يقتصر نطاق استبدال HMEM على السماح للمستخدم بتعريف وظائف نقل الذاكرة التي يجب على المزود استخدامها عند الوصول إلى مخزن مؤقت للمستخدم. تحتاج وظائف نقل الذاكرة المعرفة من قبل المستخدم إلى حساب جميع أنواع واجهات HMEM المختلفة التي قد يواجهها المزود.

ستكتسب جميع الكائنات المخصصة مقابل النطاق هذا الاستبدال.

اسم هيكل الاستبدال HMEM وردود الاتصال هو كما يلي.

#define FI_SET_OPS_HMEM_OVERRIDE "hmem_override_ops"
struct fi_hmem_override_ops {
    size_t  size;
    ssize_t (*copy_from_hmem_iov)(void *dest, size_t size,
        enum fi_hmem_iface iface, uint64_t device, const struct iovec *hmem_iov,
        size_t hmem_iov_count, uint64_t hmem_iov_offset);
    ssize_t (*copy_to_hmem_iov)(enum fi_hmem_iface iface, uint64_t device,
    const struct iovec *hmem_iov, size_t hmem_iov_count,
        uint64_t hmem_iov_offset, const void *src, size_t size);
};

يجب تعيين جميع الحقول في هيكل fi_hmem_override_ops (غير فارغة) إلى قيمة صالحة.

size
يجب تعيين هذا إلى sizeof(struct fi_hmem_override_ops). يستخدم الحقل size لأغراض التوافق الأمامي والخلفي.
copy_from_hmem_iov
انسخ البيانات من الجهاز/hmem إلى ذاكرة المضيف. يجب أن تعيد هذه الوظيفة fi_errno سالبًا عند الخطأ، أو عدد البايتات المنسوخة عند النجاح.
copy_to_hmem_iov
انسخ البيانات من ذاكرة المضيف إلى الجهاز/hmem. يجب أن تعيد هذه الوظيفة fi_errno سالبًا عند الخطأ، أو عدد البايتات المنسوخة عند النجاح.

fi_domain_bind

يربط قائمة انتظار الأحداث بالنطاق. ستكون قائمة انتظار الأحداث المرتبطة بالنطاق هي قائمة الانتظار الافتراضية للتحكم في الأحداث غير المتزامنة التي تحدث على النطاق أو نقاط النهاية النشطة المخصصة على نطاق. يشمل ذلك أحداث CM. يمكن لنقاط النهاية توجيه أحداث التحكم الخاصة بها إلى قوائم انتظار بديلة عن طريق الارتباط مباشرة بقائمة الانتظار.

مهمل: ربط قائمة انتظار الأحداث بالنطاق باستخدام علامة FI_REG_MR يشير إلى أن المزود يجب أن يقوم بجميع عمليات تسجيل الذاكرة بشكل غير متزامن، مع الإبلاغ عن الاكتمال من خلال قائمة انتظار الأحداث. إذا لم يتم ربط قائمة انتظار الأحداث بالنطاق باستخدام علامة FI_REG_MR، فإن طلبات تسجيل الذاكرة تكتمل بشكل متزامن.

fi_close

يستخدم استدعاء fi_close لتحرير جميع الموارد المرتبطة بالنطاق أو الواجهة. يجب تحرير جميع الكائنات المرتبطة بالنطاق المفتوح قبل استدعاء fi_close، وإلا فسيعيد الاستدعاء -FI_EBUSY.
