mlx5dv_create_steering_anchor / mlx5dv_destroy_steering_anchor(3) دليل وظائف المكتبة mlx5dv_create_steering_anchor / mlx5dv_destroy_steering_anchor(3)

الاسم

mlx5dv_create_steering_anchor - إنشاء مرساة توجيه

mlx5dv_destroy_steering_anchor - تدمير مرساة توجيه

الملخص

#include <infiniband/mlx5dv.h>
struct mlx5dv_steering_anchor *
mlx5dv_create_steering_anchor(struct ibv_context *context,
                              struct mlx5dv_steering_anchor_attr *attr);
int mlx5dv_destroy_steering_anchor(struct mlx5dv_steering_anchor *sa);

الوصف

يمكن للمستخدم استلام الحزم إلى صندوق رمل تم تكوينه بواسطة المستخدم ويقوم بمعالجة الحزم في نهايتها يتم اتخاذ قرار توجيه خط أنابيب حول ما يجب فعله بالحزمة.

تسمح مرساة التوجيه للمستخدم بإعادة إدخال الحزمة إلى النواة لمزيد من المعالجة.

mlx5dv_create_steering_anchor() ينشئ مرساة تسمح بإدخال الحزمة مرة أخرى إلى خط أنابيب التوجيه الخاص بالنواة.

mlx5dv_destroy_steering_anchor() يدمر مرساة توجيه.

المعاملات

context

سياق الجهاز لربطه بمرساة التوجيه.

attr

تحدد سمات المرساة الأولوية ونوع جدول التدفق الذي ستشير إليه المرساة.

struct mlx5dv_steering_anchor_attr {
        enum mlx5dv_flow_table_type ft_type;
        uint16_t priority;
        uint64_t comp_mask;
};
ft_type
نوع جدول التدفق الذي ستشير إليه المرساة.
priority
الأولوية داخل ft_type التي ستشير إليها المرساة المنشأة.
comp_mask
محجوز للتوسيع المستقبلي، يجب أن يكون 0 الآن.

mlx5dv_steering_anchor

struct mlx5dv_steering_anchor {
    uint32_t id;
};
id
معرف جدول التدفق لاستخدامه كوجهة عند إنشاء إدخال جدول التدفق.

قيمة الإرجاع

mlx5dv_create_steering_anchor() ترجع مؤشرًا إلى mlx5dv_steering_anchor جديد عند النجاح. عند حدوث خطأ، يتم إرجاع NULL وتعيين errno.

mlx5dv_destroy_steering_anchor() ترجع 0 عند النجاح وقيمة errno عند حدوث خطأ.
