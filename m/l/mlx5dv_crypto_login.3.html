الاسم

mlx5dv_crypto_login - ينشئ جلسة تسجيل دخول للتشفير

mlx5dv_crypto_login_query_state - يستعلم عن حالة جلسة تسجيل الدخول الحالية للتشفير

mlx5dv_crypto_logout - يسجل الخروج من جلسة تسجيل الدخول الحالية للتشفير

المُلَخَّص

#include <infiniband/mlx5dv.h>
int mlx5dv_crypto_login(struct ibv_context *context,
            struct mlx5dv_crypto_login_attr *login_attr);
int mlx5dv_crypto_login_query_state(struct ibv_context *context,
                    enum mlx5dv_crypto_login_state *state);
int mlx5dv_crypto_logout(struct ibv_context *context);

الوصف

عند استخدام محرك تشفير بطريقة الاستيراد المغلفة، يجب أن تكون هناك جلسة تسجيل دخول تشفير نشطة لإنشاء واستعلام مفاتيح تشفير البيانات (DEKs).

mlx5dv_crypto_login() ينشئ جلسة تسجيل دخول تشفير باستخدام بيانات الاعتماد المعطاة في login_attr ويربطها بـ context. يمكن ربط جلسة تسجيل دخول تشفير نشطة واحدة فقط لكل سياق جهاز.

mlx5dv_crypto_login_query_state() يستعلم عن حالة جلسة تسجيل دخول التشفير المرتبطة بـ context ويعيد الحالة في state، والتي تشير إلى ما إذا كانت صالحة، غير صالحة، أو غير موجودة. يمكن أن تصبح جلسة تسجيل دخول التشفير الصالحة غير صالحة إذا تم حذف بيانات الاعتماد أو مفتاح KEK للاستيراد المستخدم في جلسة تسجيل دخول التشفير أثناء جلسة التسجيل (على سبيل المثال بواسطة مسؤول التشفير). في هذه الحالة، يجب استدعاء mlx5dv_crypto_logout() لتدمير جلسة تسجيل دخول التشفير الحالية غير الصالحة، وإذا كان الأمر لا يزال ضروريًا، يجب استدعاء mlx5dv_crypto_login() لإنشاء جلسة تسجيل دخول تشفير جديدة ببيانات اعتماد صالحة ومفتاح KEK للاستيراد.

mlx5dv_crypto_logout() يسجل الخروج من جلسة تسجيل دخول التشفير الحالية المرتبطة بـ context.

مفاتيح DEKs الموجودة التي تم تحميلها مسبقًا إلى الجهاز أثناء جلسة تسجيل دخول التشفير لا تحتاج إلى جلسة تسجيل دخول تشفير نشطة لاستخدامها (في MKey أو أثناء حركة المرور).

المُعاملات

context

سياق الجهاز لربط جلسة تسجيل دخول التشفير به.

login_attr

سمات تسجيل دخول التشفير تحدد بيانات الاعتماد لتسجيل الدخول بها ومفتاح KEK للاستيراد الذي سيتم استخدامه للاتصالات الآمنة أثناء جلسة تسجيل دخول التشفير.

struct mlx5dv_crypto_login_attr {
    uint32_t credential_id;
    uint32_t import_kek_id;
    char credential[48];
    uint64_t comp_mask;
};
credential_id
معرف لبيانات اعتماد، من بيانات الاعتماد المخزنة على الجهاز، يشير إلى بيانات الاعتماد التي يجب التحقق منها مقابل بيانات الاعتماد المقدمة في credential.
import_kek_id
معرف لمفتاح KEK للاستيراد، من مفاتيح KEK للاستيراد المخزنة على الجهاز، يشير إلى مفتاح KEK للاستيراد الذي سيتم استخدامه لفك تغليف بيانات الاعتماد المقدمة في credential وكذلك لجميع الاتصالات الآمنة الأخرى أثناء جلسة تسجيل دخول التشفير.
credential
بيانات الاعتماد لتسجيل الدخول بها. يجب تقديمها مغلفة باستخدام خوارزمية تغليف المفتاح AES باستخدام مفتاح KEK للاستيراد المحدد بواسطة import_kek_id.
comp_mask
محجوز للتوسع المستقبلي، يجب أن يكون 0 الآن.

state

يشير إلى حالة جلسة تسجيل دخول التشفير الحالية. يمكن أن تكون واحدة من MLX5DV_CRYPTO_LOGIN_STATE_VALID، MLX5DV_CRYPTO_LOGIN_STATE_NO_LOGIN، و MLX5DV_CRYPTO_LOGIN_STATE_INVALID.

قيمة الإرجاع

mlx5dv_crypto_login() تعيد 0 عند النجاح وقيمة errno عند الخطأ.

mlx5dv_crypto_login_query_state() تعيد 0 عند النجاح وتحدث state بالحالة المستعلم عنها. عند الخطأ، يتم إرجاع قيمة errno.

mlx5dv_crypto_logout() تعيد 0 عند النجاح وقيمة errno عند الخطأ.

الأخطاء

EEXIST
جلسة تسجيل دخول تشفير موجودة بالفعل.
EINVAL
تم تقديم سمات غير صالحة، أو أن واحدًا أو أكثر من credential، credential_id، و import_kek_id غير صالحة.
ENOENT
لا توجد جلسة تسجيل دخول تشفير.
