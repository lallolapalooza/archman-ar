الوصف

توفر مكتبة libmaxminddb وظائف للتعامل مع ملفات MaxMind DB. راجع https://maxmind.github.io/MaxMind-DB/ لمواصفات تنسيق MaxMind DB. يتم تمثيل قاعدة البيانات والنتائج بواسطة هياكل بيانات مختلفة. يتم فتح قواعد البيانات عن طريق استدعاء MMDB_open(). يمكنك البحث عن عناوين IP كسلسلة نصية باستخدام MMDB_lookup_string() أو كمؤشر إلى هيكل sockaddr باستخدام MMDB_lookup_sockaddr().

إذا وجدت عملية البحث عنوان IP في قاعدة البيانات، فإنها ترجع هيكل MMDB_lookup_result_s. إذا أشار هذا الهيكل إلى أن قاعدة البيانات تحتوي على بيانات لعنوان IP، فهناك عدد من الوظائف التي يمكن استخدامها لجلب هذه البيانات. وتشمل هذه MMDB_get_value() و MMDB_get_entry_data_list(). راجع وثائق الوظيفة أدناه لمزيد من التفاصيل.

عندما تنتهي من مقبض قاعدة البيانات، يجب عليك استدعاء MMDB_close().

جميع الوظائف والهياكل والماكروات المرئية للعامة تبدأ بـ "MMDB_".

هياكل البيانات

جميع هياكل البيانات المصدرة بواسطة رأس maxminddb.h لهذه المكتبة معرفة مسبقًا بالشكل typedef struct foo_s { ... } foo_s بحيث يمكنك الإشارة إليها بدون البادئة struct.

توفر هذه المكتبة هياكل البيانات التالية:

MMDB_s

هذا هو مقبض ملف MaxMind DB. نوثق فقط بعض حقول هذا الهيكل المخصصة للاستخدام العام. تخضع جميع الحقول الأخرى للتغيير وهي مخصصة للاستخدام الداخلي فقط.

typedef struct MMDB_s {
    uint32_t flags;
    const char *filename;
    ...
    MMDB_metadata_s metadata;
} MMDB_s;
  • uint32_t flags - الأعلام التي تم فتح قاعدة البيانات بها. راجع وثائق MMDB_open() لمزيد من التفاصيل.
  • const char *filename - اسم الملف الذي تم فتحه، كما تم تمريره إلى MMDB_open().
  • MMDB_metadata_s metadata - البيانات الوصفية لقاعدة البيانات.

MMDB_metadata_s و MMDB_description_s

يمكن استرداد هذا الهيكل من هيكل MMDB_s. يحتوي على البيانات الوصفية المقروءة من ملف قاعدة البيانات. لاحظ أنه قد تجد أنه من الأنسب الوصول إلى هذه البيانات الوصفية عن طريق استدعاء MMDB_get_metadata_as_entry_data_list() بدلاً من ذلك.

typedef struct MMDB_metadata_s {
    uint32_t node_count;
    uint16_t record_size;
    uint16_t ip_version;
    const char *database_type;
    struct {
        size_t count;
        const char **names;
    } languages;
    uint16_t binary_format_major_version;
    uint16_t binary_format_minor_version;
    uint64_t build_epoch;
    struct {
        size_t count;
        MMDB_description_s **descriptions;
    } description;
} MMDB_metadata_s;
typedef struct MMDB_description_s {
    const char *language;
    const char *description;
} MMDB_description_s;

يجب أن تكون هذه الهياكل واضحة بذاتها إلى حد كبير.

يجب أن يكون العضو ip_version دائمًا 4 أو 6. يجب أن يكون binary_format_major_version دائمًا 2.

لا يوجد شرط بأن تحتوي البيانات الوصفية لقاعدة البيانات على لغات أو أوصاف، لذلك يمكن أن يكون count لهذه الأجزاء من البيانات الوصفية صفرًا. يجب تعبئة جميع حقول MMDB_metadata_s الأخرى.

MMDB_lookup_result_s

يتم إرجاع هذا الهيكل كنتيجة للبحث عن عنوان IP.

typedef struct MMDB_lookup_result_s {
    bool found_entry;
    MMDB_entry_s entry;
    uint16_t netmask;
} MMDB_lookup_result_s;

إذا كان العضو found_entry خاطئًا، فإن الأعضاء الأخرى في هذا الهيكل لا تحتوي على قيم ذات معنى. تحقق دائمًا من أن found_entry صحيح أولاً.

يتم استخدام العضو entry للبحث عن البيانات المرتبطة بعنوان IP.

يخبرك العضو netmask عن الشبكة الفرعية التي ينتمي إليها عنوان IP في قاعدة البيانات هذه. على سبيل المثال، إذا قمت بالبحث عن العنوان 1.1.1.1 في قاعدة بيانات IPv4 وكان netmask المُرجع هو 16، فإن العنوان جزء من الشبكة الفرعية 1.1.0.0/16.

إذا كانت قاعدة البيانات قاعدة بيانات IPv6، فإن قناع الشبكة المرجع هو دائمًا طول بادئة IPv6 (من 0-128)، حتى لو كانت قاعدة البيانات هذه تحتوي أيضًا على شبكات IPv4. إذا قمت بالبحث عن عنوان IPv4 وترغب في تحويل قناع الشبكة إلى قيمة قناع شبكة IPv4، يمكنك ببساطة طرح 96 من القيمة.

MMDB_result_s

لا تحتاج حقًا إلى التعمق في هذا الهيكل. ستحصل عليه من هيكل MMDB_lookup_result_s وتمرره إلى وظائف مختلفة.

MMDB_entry_data_s

يستخدم هذا الهيكل لإرجاع إدخال قسم بيانات واحد لعنوان IP. يمكن لهذه الإدخالات بدورها الإشارة إلى إدخالات أخرى، كما هو الحال في أشياء مثل الخرائط والمصفوفات. بعض حقول هذا الهيكل غير موثقة لأنها مخصصة للاستخدام الداخلي فقط.

typedef struct MMDB_entry_data_s {
    bool has_data;
    union {
        uint32_t pointer;
        const char *utf8_string;
        double double_value;
        const uint8_t *bytes;
        uint16_t uint16;
        uint32_t uint32;
        int32_t int32;
        uint64_t uint64;
        {mmdb_uint128_t or uint8_t[16]} uint128;
        bool boolean;
        float float_value;
    };
    ...
    uint32_t data_size;
    uint32_t type;
} MMDB_entry_data_s;

العضو has_data صحيح إذا تم العثور على بيانات لعملية بحث معينة. راجع MMDB_get_value() لمزيد من التفاصيل. إذا كان هذا العضو خاطئًا، فإن أيًا من القيم الأخرى في الهيكل ليست ذات معنى.
