الوصف

توفر مكتبة libmaxminddb دوالاً للعمل مع ملفات MaxMind DB. راجع https://maxmind.github.io/MaxMind-DB/ للحصول على مواصفات تنسيق MaxMind DB. يتم تمثيل قاعدة البيانات والنتائج جميعها بهياكل بيانات مختلفة. يتم فتح قواعد البيانات عن طريق استدعاء MMDB_open(). يمكنك البحث عن عناوين IP كنص باستخدام MMDB_lookup_string() أو كمؤشر إلى هيكل sockaddr باستخدام MMDB_lookup_sockaddr().

إذا وجد البحث عنوان IP في قاعدة البيانات، فإنه يُرجع هيكل MMDB_lookup_result_s. إذا أشار ذلك الهيكل إلى أن قاعدة البيانات تحتوي على بيانات لعنوان IP، فهناك عدد من الدوال التي يمكن استخدامها لجلب تلك البيانات. وتشمل هذه MMDB_get_value() و MMDB_get_entry_data_list(). راجع توثيق الدوال أدناه لمزيد من التفاصيل.

عند الانتهاء من التعامل مع مقبض قاعدة البيانات، يجب عليك استدعاء MMDB_close().

جميع الدوال والهياكل والماكروهات المرئية للعموم تبدأ بـ "MMDB_".

هياكل البيانات

جميع هياكل البيانات التي يصدرها ملف الترويسة maxminddb.h الخاص بهذه المكتبة معرّفة بـ typedef بالشكل typedef struct foo_s { ... } foo_s بحيث يمكنك الإشارة إليها دون البادئة struct.

توفر هذه المكتبة هياكل البيانات التالية:

MMDB_s

هذا هو المقبض لملف MaxMind DB. نوثّق فقط بعض حقول هذا الهيكل المخصصة للاستخدام العام. جميع الحقول الأخرى عرضة للتغيير ومخصصة للاستخدام الداخلي فقط.

typedef struct MMDB_s {
    uint32_t flags;
    const char *filename;
    ...
    MMDB_metadata_s metadata;
} MMDB_s;
  • uint32_t flags - الأعلام التي فُتحت بها قاعدة البيانات هذه. راجع توثيق MMDB_open() لمزيد من التفاصيل.
  • const char *filename - اسم الملف الذي تم فتحه، كما تم تمريره إلى MMDB_open().
  • MMDB_metadata_s metadata - البيانات الوصفية لقاعدة البيانات.

MMDB_metadata_s و MMDB_description_s

يمكن استرداد هذا الهيكل من هيكل MMDB_s. يحتوي على البيانات الوصفية المقروءة من ملف قاعدة البيانات. لاحظ أنك قد تجد أنه أكثر ملاءمة للوصول إلى هذه البيانات الوصفية عن طريق استدعاء MMDB_get_metadata_as_entry_data_list() بدلاً من ذلك.

typedef struct MMDB_metadata_s {
    uint32_t node_count;
    uint16_t record_size;
    uint16_t ip_version;
    const char *database_type;
    struct {
        size_t count;
        const char **names;
    } languages;
    uint16_t binary_format_major_version;
    uint16_t binary_format_minor_version;
    uint64_t build_epoch;
    struct {
        size_t count;
        MMDB_description_s **descriptions;
    } description;
} MMDB_metadata_s;
typedef struct MMDB_description_s {
    const char *language;
    const char *description;
} MMDB_description_s;

يجب أن تكون هذه الهياكل واضحة بذاتها في الغالب.

يجب أن يكون العضو ip_version دائمًا 4 أو 6. ويجب أن يكون binary_format_major_version دائمًا 2.

لا يوجد شرط بأن تتضمن البيانات الوصفية لقاعدة البيانات لغات أو أوصافًا، لذا يمكن أن يكون count لهذه الأجزاء من البيانات الوصفية صفرًا. يجب أن تكون جميع حقول MMDB_metadata_s الأخرى مملوءة.

MMDB_lookup_result_s

يُعاد هذا الهيكل كنتيجة للبحث عن عنوان IP.

typedef struct MMDB_lookup_result_s {
    bool found_entry;
    MMDB_entry_s entry;
    uint16_t netmask;
} MMDB_lookup_result_s;

إذا كان العضو found_entry خطأً فإن الأعضاء الآخرين في هذا الهيكل لا يحتويان على قيم ذات معنى. تحقق دائمًا من أن found_entry صحيح أولاً.

يُستخدم العضو entry للبحث عن البيانات المرتبطة بعنوان IP.

يخبرك العضو netmask بالشبكة الفرعية التي ينتمي إليها عنوان IP في قاعدة البيانات هذه. على سبيل المثال، إذا بحثت عن العنوان 1.1.1.1 في قاعدة بيانات IPv4 وكان netmask المُعاد هو 16، فإن العنوان جزء من الشبكة الفرعية 1.1.0.0/16.

إذا كانت قاعدة البيانات قاعدة بيانات IPv6، فإن netmask المُعاد يكون دائمًا طول بادئة IPv6 (من 0 إلى 128)، حتى لو كانت تلك قاعدة البيانات تحتوي أيضًا على شبكات IPv4. إذا بحثت عن عنوان IPv4 وأردت تحويل netmask إلى قيمة netmask لـ IPv4، يمكنك ببساطة طرح 96 من القيمة.

MMDB_result_s

لا تحتاج حقًا إلى التنقيب في هذا الهيكل. ستحصل عليه من هيكل MMDB_lookup_result_s وتمريره إلى دوال مختلفة.

MMDB_entry_data_s

يُستخدم هذا الهيكل لإرجاع إدخال قسم بيانات واحد لعنوان IP. يمكن لهذه الإدخالات بدورها الإشارة إلى إدخالات أخرى، كما هو الحال في أشياء مثل الخرائط والمصفوفات. بعض أعضاء هذا الهيكل غير موثقة لأنها مخصصة للاستخدام الداخلي فقط.

typedef struct MMDB_entry_data_s {
    bool has_data;
    union {
        uint32_t pointer;
        const char *utf8_string;
        double double_value;
        const uint8_t *bytes;
        uint16_t uint16;
        uint32_t uint32;
        int32_t int32;
        uint64_t uint64;
        {mmdb_uint128_t or uint8_t[16]} uint128;
        bool boolean;
        float float_value;
    };
    ...
    uint32_t data_size;
    uint32_t type;
} MMDB_entry_data_s;

يكون العضو has_data صحيحًا إذا تم العثور على بيانات لعملية بحث معينة. راجع MMDB_get_value() لمزيد من التفاصيل. إذا كان هذا العضو خطأً فإن أيًا من القيم الأخرى في الهيكل لا يكون ذا معنى.

يحدد الاتحاد في بداية الهيكل البيانات الفعلية. لتحديد أي اتحاد

إذا كان العضو مملوءًا، يجب عليك فحص العضو type. يجب ألا يُملأ العضو pointer في الاتحاد أبدًا في أي بيانات تُرجعها واجهة برمجة التطبيقات. يجب حل المؤشرات داخليًا.

العضو data_size ذو صلة فقط بالبيانات من النوع utf8_string وbytes. لا ينتهي utf8_string بـ NULL ويجب استخدام data_size لتحديد طوله.

يمكن مقارنة العضو type بأحد الماكروز MMDB_DATA_TYPE_*.

الأعداد الصحيحة 128-بت

تعتمد معالجة البيانات من النوع uint128 على كيفية دعم منصتك للأعداد الصحيحة 128-بت، سواء كانت تدعمها أم لا. مع GCC 4.4 و4.5 يمكننا كتابة unsigned int __attribute__ ((__mode__ (TI))). مع إصدارات أحدث من GCC (4.6+) وClang (3.2+) يمكننا ببساطة كتابة "unsigned __int128".

للتعامل مع هذه الاختلافات، تعرّف هذه المكتبة نوعًا mmdb_uint128_t. هذا النوع معرّف في الترويسة maxminddb.h حتى تتمكن من استخدامه في الكود الخاص بك.

مع المترجمات الأقدم، لا يمكننا استخدام عدد صحيح، لذلك نستخدم مصفوفة من 16 بايت من uint8_t. هذه هي البيانات المباشرة من قاعدة البيانات.

توفر هذه المكتبة ماكرو عامًا MMDB_UINT128_IS_BYTE_ARRAY. إذا كان هذا صحيحًا (1)، تُرجع قيم uint128 كمصفوفة بايت، وإلا كعدد صحيح mmdb_uint128_t.

ماكروز أنواع البيانات

توفر هذه المكتبة ماكرو لكل نوع بيانات محدد بواسطة مواصفات MaxMind DB.

  • MMDB_DATA_TYPE_UTF8_STRING
  • MMDB_DATA_TYPE_DOUBLE
  • MMDB_DATA_TYPE_BYTES
  • MMDB_DATA_TYPE_UINT16
  • MMDB_DATA_TYPE_UINT32
  • MMDB_DATA_TYPE_MAP
  • MMDB_DATA_TYPE_INT32
  • MMDB_DATA_TYPE_UINT64
  • MMDB_DATA_TYPE_UINT128
  • MMDB_DATA_TYPE_ARRAY
  • MMDB_DATA_TYPE_BOOLEAN
  • MMDB_DATA_TYPE_FLOAT

هناك أيضًا بعض الأنواع المخصصة للاستخدام الداخلي فقط:

  • MMDB_DATA_TYPE_EXTENDED
  • MMDB_DATA_TYPE_POINTER
  • MMDB_DATA_TYPE_CONTAINER
  • MMDB_DATA_TYPE_END_MARKER

إذا رأيت أحد هذه في البيانات المُرجعة، فقد حدث خطأ فادح. قاعدة البيانات تالفة أو تم إنشاؤها بشكل غير صحيح أو هناك خطأ في كود libmaxminddb.

قيم المؤشرات وMMDB_close()

الأعضاء utf8_string وbytes و(ربما) العضو uint128 في هذه البنية كلها مؤشرات مباشرة إلى قسم البيانات في قاعدة البيانات. قد يكون هذا كتلة ذاكرة calloc أو mmap. في كلتا الحالتين، تصبح هذه المؤشرات غير صالحة بعد استدعاء MMDB_close().

إذا كنت بحاجة إلى الرجوع إلى هذه البيانات بعد ذلك الوقت، يجب عليك نسخ البيانات باستخدام دالة مناسبة (strdup، memcpy، إلخ).

MMDB_entry_data_list_s

هذه البنية تغلف قائمة مترابطة من بنى MMDB_entry_data_s.

typedef struct MMDB_entry_data_list_s {  
    MMDB_entry_data_s entry_data;  
    struct MMDB_entry_data_list_s *next;  
} MMDB_entry_data_list_s;

تتيح لك هذه البنية فحص بيانات الخريطة أو المصفوفة عبر التكرار على القائمة المترابطة.

MMDB_search_node_s

هذه البنية تغلف السجلين لعقدة بحث. هذا مفيد حقًا إذا كنت تريد كتابة كود يتكرر على شجرة البحث بأكملها بدلاً من البحث عن IP محدد.

typedef struct MMDB_search_node_s {  
    uint64_t left_record;  
    uint64_t right_record;  
    uint8_t left_record_type;  
    uint8_t right_record_type;  
    MMDB_entry_s left_record_entry;  
    MMDB_entry_s right_record_entry;  
} MMDB_search_node_s;

ستكون أنواع السجلات أحد القيم التالية:

  • MMDB_RECORD_TYPE_SEARCH_NODE - يشير السجل إلى عقدة البحث التالية.
  • MMDB_RECORD_TYPE_EMPTY - السجل هو عنصر نائب يشير إلى عدم وجود بيانات لـ IP. يجب أن ينتهي البحث هنا.
  • MMDB_RECORD_TYPE_DATA - يحتوي السجل على بيانات في قسم البيانات في قاعدة البيانات. استخدم السجل عند البحث عن بيانات السجل.
  • MMDB_RECORD_TYPE_INVALID - السجل غير صالح. قد يكون السبب عقدة غير صالحة تم البحث عنها أو قاعدة بيانات تالفة.

يكون MMDB_entry_s للسجل صالحًا فقط إذا كان النوع MMDB_RECORD_TYPE_DATA. استخدام سجل من أنواع أخرى ينتج خطأً أو بيانات غير صالحة.

رموز الحالة

تُرجع هذه المكتبة (أو تملأ) رموز حالة للعديد من الدوال. رموز الحالة هذه هي:

  • MMDB_SUCCESS - كل شيء نجح.
  • MMDB_FILE_OPEN_ERROR - حدث خطأ عند فتح ملف MaxMind DB.
  • MMDB_IO_ERROR - فشلت عملية إدخال/إخراج. تحقق من errno لمزيد من التفاصيل.
  • MMDB_CORRUPT_SEARCH_TREE_ERROR - عند البحث عن IP في شجرة البحث تم الحصول على نتيجة مستحيلة. قاعدة البيانات تالفة أو تم إنشاؤها بشكل غير صحيح أو هناك خطأ في كود libmaxminddb.
  • MMDB_INVALID_METADATA_ERROR - هناك خطأ في شيء ما في قاعدة البيانات. يشمل ذلك مفاتيح بيانات وصفية مفقودة أو قيم مستحيلة (مثل ip_version = 7).
  • MMDB_UNKNOWN_DATABASE_FORMAT_ERROR - تشير البيانات الوصفية لقاعدة البيانات إلى أن إصدارها الرئيسي ليس 2. يمكن لهذه المكتبة التعامل فقط مع الإصدار الرئيسي 2.
  • MMDB_OUT_OF_MEMORY_ERROR - فشل استدعاء تخصيص الذاكرة (malloc، إلخ).
  • MMDB_INVALID_DATA_ERROR - يحتوي حقل بيانات على بيانات غير صالحة. على سبيل المثال، حقل uint16 يدّعي أنه أكثر من 2 بايت. قاعدة البيانات على الأرجح تالفة أو تم إنشاؤها بشكل غير صحيح.
  • MMDB_INVALID_LOOKUP_PATH_ERROR - مسار البحث الممرر إلى MMDB_get_value أو MMDB_vget_value أو MMDB_aget_value يحتوي على إزاحة مصفوفة أكبر من LONG_MAX أو أصغر من LONG_MIN.
  • MMDB_LOOKUP_PATH_DOES_NOT_MATCH_DATA_ERROR - مسار البحث الممرر إلى MMDB_get_value أو MMDB_vget_value أو MMDB_aget_value لا يتطابق مع بنية بيانات الإدخال. قد تحدث عدة أسباب. قد يتضمن مسار البحث مفتاحًا غير موجود في خريطة. قد يتضمن مسار البحث فهرس مصفوفة أكبر من مصفوفة أو أصغر من الإزاحة الدنيا من نهاية مصفوفة. قد يحدث أيضًا عندما يتوقع المسار العثور على خريطة أو مصفوفة حيث لا توجد أي منهما.

يجب التعامل مع جميع رموز الحالة كقيم من النوع int.

MMDB_strerror()

const char *MMDB_strerror(int error_code)

تأخذ هذه الدالة رمز حالة وتُرجع سلسلة بالإنجليزية تشرح الحالة.

الدوال

توفر هذه المكتبة الدوال المُصدَّرة التالية:

MMDB_open()

int MMDB_open(  
    const char *const filename,  
    uint32_t flags,  
    MMDB_s *const mmdb);

تفتح هذه الدالة مقبضًا لملف MaxMind DB. القيمة المُرجعة هي رمز حالة كما هو محدد أعلاه. يجب عليك التحقق من القيمة المُرجعة لهذا الاستدعاء.

MMDB_s mmdb;  
int status =  
    MMDB_open("/path/to/file.mmdb", MMDB_MODE_MMAP, &mmdb);  
if (MMDB_SUCCESS != status) { ... }  
...  
MMDB_close(&mmdb);

يجب ترميز filename كـ UTF-8 على Windows.

يمكن أن تكون بنية MMDB_s التي تمررها على المكدس أو محجوزة في ذاكرة الكومة. ومع ذلك، إذا نجح الفتح، فستحتوي على بيانات محجوزة في ذاكرة الكومة، لذلك يجب عليك إغلاقها باستخدام MMDB_close(). إذا لم يكن رمز الحالة MMDB_SUCCESS، فستضمن هذه المكتبة تحرير جميع الذاكرة المحجوزة قبل الإرجاع.

الأعلام الحالية هي:

•
MMDB_MODE_MMAP - يفتح الملف باستخدام mmap().

يمكنك تمرير 0 كقيمة لـ flags لفتح الملف بالأعلام الافتراضية. ومع ذلك، قد تتغير هذه القيم في الإصدارات المستقبلية. القيمة الافتراضية هي MMDB_MODE_MMAP.

MMDB_close()

void MMDB_close(MMDB_s *const mmdb);

تحرر هذه المكتبة أي ذاكرة محجوزة أو معمّمة بـ mmap مرتبطة ببنية MMDB_s. لا تحرر الذاكرة الخاصة بقاعدة البيانات.

y المخصصة للبنية نفسها! إذا قمت بتخصيص البنية من الكومة (heap) فأنت مسؤول عن تحريرها.

MMDB_lookup_string()

MMDB_lookup_result_s MMDB_lookup_string(
    MMDB_s *const mmdb,
    const char *const ipstr,
    int *const gai_error,
    int *const mmdb_error);

تبحث هذه الدالة عن عنوان IP يتم تمريره كسلسلة منتهية بقيمة null. داخليًا تستدعي getaddrinfo() لتحويل العنوان إلى صيغة ثنائية. ثم تستدعي MMDB_lookup_sockaddr() للبحث عن العنوان في قاعدة البيانات. إذا كنت قد حوّلت عنوانًا بالفعل يمكنك استدعاء MMDB_lookup_sockaddr() مباشرةً، بدلاً من تحويل العنوان مرتين.

int gai_error, mmdb_error;
MMDB_lookup_result_s result =
    MMDB_lookup_string(&mmdb, "1.2.3.4", &gai_error, &mmdb_error);
if (0 != gai_error) { ... }
if (MMDB_SUCCESS != mmdb_error) { ... }
if (result.found_entry) { ... }

تُرجع هذه الدالة دائمًا بنية MMDB_lookup_result_s، ولكن يجب عليك أيضًا التحقق من المعاملين gai_error و mmdb_error. إذا أشار أي منهما إلى خطأ فإن البنية المُرجعة لا معنى لها.

إذا لم يحدث أي خطأ فلا تزال بحاجة إلى التأكد من أن العضو found_entry في النتيجة المُرجعة صحيح. إذا لم يكن كذلك، فهذا يعني أن عنوان IP ليس له إدخال في قاعدة البيانات.

ستعمل هذه الدالة مع عناوين IPv4 حتى عندما تحتوي قاعدة البيانات على بيانات لكل من عناوين IPv4 و IPv6. سيتم البحث عن عنوان IPv4 بصيغة '::xxx.xxx.xxx.xxx' بدلاً من إعادة تعيينه إلى الكتلة ::ffff:xxx.xxx.xxx.xxx المخصصة لعناوين IPv6 المعيّنة من IPv4.

إذا مررت عنوان IPv6 إلى قاعدة بيانات تحتوي على بيانات IPv4 فقط فإن العضو found_entry سيكون خاطئًا، لكن حالة mmdb_error ستظل MMDB_SUCCESS.

MMDB_lookup_sockaddr()

MMDB_lookup_result_s MMDB_lookup_sockaddr(
    MMDB_s *const mmdb,
    const struct sockaddr *const sockaddr,
    int *const mmdb_error);

تبحث هذه الدالة عن عنوان IP تم تحويله بالفعل بواسطة getaddrinfo().

بخلاف عدم استدعاء getaddrinfo() نفسها، فإن هذه الدالة مطابقة لدالة MMDB_lookup_string().

int mmdb_error;
MMDB_lookup_result_s result =
    MMDB_lookup_sockaddr(&mmdb, address->ai_addr, &mmdb_error);
if (MMDB_SUCCESS != mmdb_error) { ... }
if (result.found_entry) { ... }

دوال البحث عن البيانات

هناك ثلاث دوال للبحث عن البيانات المرتبطة بعنوان IP.

int MMDB_get_value(
    MMDB_entry_s *const start,
    MMDB_entry_data_s *const entry_data,
    ...);
int MMDB_vget_value(
    MMDB_entry_s *const start,
    MMDB_entry_data_s *const entry_data,
    va_list va_path);
int MMDB_aget_value(
    MMDB_entry_s *const start,
    MMDB_entry_data_s *const entry_data,
    const char *const *const path);

تسمح الدوال الثلاث بثلاثة أنماط استدعاء مختلفة قليلاً، لكنها جميعًا تفعل الشيء نفسه.

المعامل الأول هو قيمة MMDB_entry_s. في معظم الحالات ستأتي هذه من قيمة MMDB_lookup_result_s التي تُرجعها MMDB_lookup_string() أو MMDB_lookup_sockaddr().

المعامل الثاني هو مرجع إلى بنية MMDB_entry_data_s. سيتم تعبئتها بالبيانات التي يتم البحث عنها، إن وُجدت. إذا لم يتم العثور على شيء، فإن العضو has_data في هذه البنية سيكون خاطئًا. إذا كان has_data صحيحًا فيمكنك النظر إلى العضو data_type.

المعامل الأخير هو مسار البحث. يتكون المسار من مجموعة من السلاسل النصية التي تمثل إما مفاتيح خريطة (مثل "city") أو فهارس مصفوفة (مثل "0"، "1"، "-1") لاستخدامها في البحث.

ستُعامَل فهارس المصفوفة السالبة كإزاحة من نهاية المصفوفة. على سبيل المثال، يشير "-1" إلى العنصر الأخير من المصفوفة.

يسمح لك مسار البحث بالتنقل في بنية بيانات معقدة. على سبيل المثال، بالنظر إلى هذه البيانات:

{
    "names": {
        "en": "Germany",
        "de": "Deutschland"
    },
    "cities": [ "Berlin", "Frankfurt" ]
}

يمكننا البحث عن الاسم الإنجليزي بهذا الكود:

MMDB_lookup_result_s result =
    MMDB_lookup_sockaddr(&mmdb, address->ai_addr, &mmdb_error);
MMDB_entry_data_s entry_data;
int status =
    MMDB_get_value(&result.entry, &entry_data,
                   "names", "en", NULL);
if (MMDB_SUCCESS != status) { ... }
if (entry_data.has_data) { ... }

إذا أردنا العثور على المدينة الأولى فسيكون مسار البحث "cities", "0". إذا لم توفر مسار بحث على الإطلاق، فستحصل على الإدخال الذي يتوافق مع الخريطة ذات المستوى الأعلى. يجب أن ينتهي مسار البحث دائمًا بـ NULL، بغض النظر عن الدالة التي تستدعيها.

تأخذ دالة MMDB_get_value عددًا متغيرًا من المعاملات. جميع المعاملات بعد مؤشر البنية MMDB_entry_data_s * هي مسار البحث. يجب أن يكون المعامل الأخير NULL.

تقبل دالة MMDB_vget_value va_list كمسار بحث. يجب أن يكون العنصر الأخير الذي يتم استرجاعه بواسطة va_arg() NULL.

أخيرًا، تقبل MMDB_aget_value مصفوفة من السلاسل النصية كمسار بحث. يجب أن يكون العضو الأخير في هذه المصفوفة NULL.

إذا أردت الحصول على جميع بيانات الإدخال دفعة واحدة يمكنك استدعاء MMDB_get_entry_data_list() بدلاً من ذلك.

لكل من الدوال الثلاث، القيمة المُرجعة هي رمز حالة كما هو محدد أعلاه.

MMDB_get_entry_data_list()

int MMDB_get_entry_data_list(
    MMDB_entry_s *start,
    MMDB_entry_data_list_s **const entry_data_list);

تسمح لك هذه الدالة بالحصول على جميع بيانات بنية بيانات معقدة دفعة واحدة، بدلاً من البحث عن كل جزء باستخدام استدعاءات متكررة لـ MMDB_get_value().

MMDB_lookup_result_s result =
    MMDB_lookup_sockaddr(&mmdb, address->ai_addr, &mmdb_error);
MMDB_entry_data_list_s *entry_data_list, *first;
int status =
    MMDB_get_entry_data_list(&result.entry, &entry_data_list);
if (MMDB_SUCCESS != status) { ... }
// احفظ هذا حتى نتمكن من تحرير هذه البيانات لاحقًا
first = entry_data_list;
while (1) {
    MMDB_entry_data_list_s *next = entry_data_list = entry_data_list->next;
    if (NULL == next) {
        break;
    }
    switch (next->entry_data.type) {
        case MMDB_DATA_TYPE_MAP: { ... }
        case MMDB_DATA_TYPE_UTF8_STRING: { ... }
        ...
    }
}
MMDB_free_entry_data_list(first);

يعود إليك تفسير بنية بيانات entry_data_list. القائمة مرتبطة باجتياز العمق أولاً. لنستخدم هذه البنية كمثال:

{
    "names": {
        "en": "Germany",
        "de": "Deutschland"
    },
    "cities": [ "Berlin", "Frankfurt" ]
}

ستتكون القائمة من العناصر التالية:

1.
MAP - الخريطة ذات المستوى الأعلى
2.
UTF8_STRING - مفتاح "names"
3.
MAP - خريطة مفتاح "names"
4.
UTF8_STRING - مفتاح "en"
5.
UTF8_STRING - قيمة مفتاح "en"
6.
UTF8_STRING - مفتاح "de"
7.
UTF8_STRING - قيمة مفتاح "de"
8.
UTF8_STRING - مفتاح "cities"
9.
ARRAY - قيمة مفتاح "cities"
10.
UTF8_STRING - array[0]
11.
UTF8_STRING - array[1]

القيمة المُرجعة للدالة هي رمز حالة كما هو محدد أعلاه.

MMDB_free_entry_data_list()

void MMDB_free_entry_data_list(
    MMDB_entry_data_list_s *const entry_data_list);

ستقوم دالتا MMDB_get_entry_data_list() و MMDB_get_metadata_as_entry_data_list() بتخصيص بنية القائمة المرتبطة من الكومة. استدعِ هذه الدالة لتحرير بنية MMDB_entry_data_list_s.

MMDB_get_metadata_as_entry_data_list()

int MMDB_get_metadata_as_entry_data_list(
    MMDB_s *const mmdb,
    MMDB_entry_data_list_s **const entry_data_list);

تسمح لك هذه الدالة باسترجاع بيانات التعريف الخاصة بقاعدة البيانات كـ قائمة مرتبطة من بنى MMDB_entry_data_list_s. يمكن أن تكون هذه طريقة أكثر ملاءمة للتعامل مع بيانات التعريف بدلاً من استخدام بنية بيانات التعريف مباشرةً.

    MMDB_entry_data_list_s *entry_data_list, *first;
    int status =
        MMDB_get_metadata_as_entry_data_list(&mmdb, &entry_data_list);
    if (MMDB_SUCCESS != status) { ... }
    first = entry_data_list;
    ... // افعل شيئًا بالبيانات
    MMDB_free_entry_data_list(first);

القيمة المُرجعة للدالة هي رمز حالة كما هو محدد أعلاه.

MMDB_dump_entry_data_list()

int MMDB_dump_entry_data_list(
    FILE *const stream,
    MMDB_entry_data_list_s *const entry_data_list,
    int indent);

تأخذ هذه الدالة قائمة مرتبطة من بنى MMDB_entry_data_list_s وتحولها إلى سلسلة نصية إلى stream المحدد. المعامل indent هو مستوى الإزاحة الابتدائي للمخرجات المُنشأة. يتم زيادته لبنى البيانات المتداخلة (الخرائط، المصفوفات، إلخ).

يجب أن يكون stream مقبض ملف (stdout، إلخ). إذا كانت منصتك توفر شيئًا مثل open_memstream() من GNU فيمكنك استخدامه لالتقاط المخرجات كسلسلة نصية.

يتم تنسيق المخرجات بأسلوب يشبه JSON، لكن القيم يتم تمييزها بنوع بياناتها (باستثناء الخرائط والمصفوفات التي تظهر بـ "{}" و "[]" على التوالي).

قد يتغير تنسيق المخرجات المحددفي الإصدارات المستقبلية، لذا يجب ألا تعتمد على التنسيق المحدد الذي تنتجه هذه الدالة. الغرض منها هو استخدامها لعرض البيانات للمستخدمين بطريقة قابلة للقراءة ولأغراض تصحيح الأخطاء.

القيمة المُرجعة من الدالة هي رمز حالة كما هو معرّف أعلاه.

MMDB_read_node()

int MMDB_read_node(
    MMDB_s *const mmdb,
    uint32_t node_number,
    MMDB_search_node_s *const node);

تقرأ هذه الدالة عقدة محددة في شجرة البحث. الوسيط الثالث هو مرجع إلى بنية MMDB_search_node_s التي ستُملأ بواسطة هذه الدالة.

القيمة المُرجعة هي رمز حالة. إذا مرّرت node_number أكبر من عدد العقد في قاعدة البيانات، فستُرجع هذه الدالة MMDB_INVALID_NODE_NUMBER_ERROR، وإلا فستُرجع MMDB_SUCCESS.

العقدة الأولى في شجرة البحث هي دائمًا العقدة 0. إذا أردت التكرار عبر شجرة البحث بأكملها، فستبدأ بقراءة العقدة 0 ثم تتبع السجلات التي تتكوّن منها هذه العقدة، بناءً على نوع كل سجل. إذا كان النوع MMDB_RECORD_TYPE_SEARCH_NODE فإن السجل يحتوي على عدد صحيح يمثّل العقدة التالية التي يجب البحث عنها.

MMDB_lib_version()

const char *MMDB_lib_version(void)

تُرجع هذه الدالة إصدار المكتبة كسلسلة نصية، شيء مثل "2.0.0".

مثال

#include <errno.h>
#include <maxminddb.h>
#include <stdlib.h>
#include <string.h>
int main(int argc, char **argv)
{
    char *filename = argv[1];
    char *ip_address = argv[2];
    MMDB_s mmdb;
    int status = MMDB_open(filename, MMDB_MODE_MMAP, &mmdb);
    if (MMDB_SUCCESS != status) {
        fprintf(stderr, "\n  Can't open %s - %s\n",
                filename, MMDB_strerror(status));
        if (MMDB_IO_ERROR == status) {
            fprintf(stderr, "    IO error: %s\n", strerror(errno));
        }
        exit(1);
    }
    int gai_error, mmdb_error;
    MMDB_lookup_result_s result =
        MMDB_lookup_string(&mmdb, ip_address, &gai_error, &mmdb_error);
    if (0 != gai_error) {
        fprintf(stderr,
                "\n  Error from getaddrinfo for %s - %s\n\n",
                ip_address, gai_strerror(gai_error));
        exit(2);
    }
    if (MMDB_SUCCESS != mmdb_error) {
        fprintf(stderr,
                "\n  Got an error from libmaxminddb: %s\n\n",
                MMDB_strerror(mmdb_error));
        exit(3);
    }
    MMDB_entry_data_list_s *entry_data_list = NULL;
    int exit_code = 0;
    if (result.found_entry) {
        int status = MMDB_get_entry_data_list(&result.entry,
                                              &entry_data_list);
        if (MMDB_SUCCESS != status) {
            fprintf(
                stderr,
                "Got an error looking up the entry data - %s\n",
                MMDB_strerror(status));
            exit_code = 4;
            goto end;
        }
        if (NULL != entry_data_list) {
            MMDB_dump_entry_data_list(stdout, entry_data_list, 2);
        }
    } else {
        fprintf(
            stderr,
            "\n  No entry for this IP address (%s) was found\n\n",
            ip_address);
        exit_code = 5;
    }
    end:
        MMDB_free_entry_data_list(entry_data_list);
        MMDB_close(&mmdb);
        exit(exit_code);
}
