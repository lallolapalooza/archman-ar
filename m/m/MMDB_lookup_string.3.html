الوصف

توفر مكتبة libmaxminddb وظائف للتعامل مع ملفات MaxMind DB. راجع https://maxmind.github.io/MaxMind-DB/ لمواصفات تنسيق MaxMind DB. يتم تمثيل قاعدة البيانات والنتائج بالكامل بواسطة هياكل بيانات مختلفة. يتم فتح قواعد البيانات عن طريق استدعاء MMDB_open(). يمكنك البحث عن عناوين IP كسلسلة نصية باستخدام MMDB_lookup_string() أو كمؤشر إلى هيكل sockaddr باستخدام MMDB_lookup_sockaddr().

إذا وجدت عملية البحث عنوان IP في قاعدة البيانات، فإنها تُرجع هيكل MMDB_lookup_result_s. إذا أشار هذا الهيكل إلى أن قاعدة البيانات تحتوي على بيانات لعنوان IP، فهناك عدد من الوظائف التي يمكن استخدامها لجلب هذه البيانات. تشمل هذه MMDB_get_value() و MMDB_get_entry_data_list(). راجع وثائق الوظيفة أدناه لمزيد من التفاصيل.

عندما تنتهي من التعامل مع مقبض قاعدة البيانات، يجب عليك استدعاء MMDB_close().

جميع الوظائف والهياكل والمُعَرِّفات العامة مرئية تبدأ بـ "MMDB_".

هياكل البيانات

جميع هياكل البيانات التي تصدرها ملفات الرأس maxminddb.h لهذه المكتبة تم تعريفها باسم مستعار typedef struct foo_s { ... } foo_s، لذلك يمكنك الإشارة إليها بدون البادئة struct.

توفر هذه المكتبة هياكل البيانات التالية:

MMDB_s

هذا هو المقبض لملف MaxMind DB. نوثق فقط بعض حقول هذا الهيكل المخصصة للاستخدام العام. تخضع جميع الحقول الأخرى للتغيير وهي مخصصة للاستخدام الداخلي فقط.

typedef struct MMDB_s {
    uint32_t flags;
    const char *filename;
    ...
    MMDB_metadata_s metadata;
} MMDB_s;
  • uint32_t flags - العلامات التي تم فتح قاعدة البيانات بها. راجع وثائق MMDB_open() لمزيد من التفاصيل.
  • const char *filename - اسم الملف الذي تم فتحه، كما تم تمريره إلى MMDB_open().
  • MMDB_metadata_s metadata - البيانات الوصفية لقاعدة البيانات.

MMDB_metadata_s و MMDB_description_s

يمكن استرداد هذا الهيكل من الهيكل MMDB_s. يحتوي على البيانات الوصفية التي تمت قراءتها من ملف قاعدة البيانات. لاحظ أنك قد تجد أنه من الأسهل الوصول إلى هذه البيانات الوصفية عن طريق استدعاء MMDB_get_metadata_as_entry_data_list() بدلاً من ذلك.

typedef struct MMDB_metadata_s {
    uint32_t node_count;
    uint16_t record_size;
    uint16_t ip_version;
    const char *database_type;
    struct {
        size_t count;
        const char **names;
    } languages;
    uint16_t binary_format_major_version;
    uint16_t binary_format_minor_version;
    uint64_t build_epoch;
    struct {
        size_t count;
        MMDB_description_s **descriptions;
    } description;
} MMDB_metadata_s;
typedef struct MMDB_description_s {
    const char *language;
    const char *description;
} MMDB_description_s;

يجب أن تكون هذه الهياكل واضحة بذاتها إلى حد كبير.

يجب أن يكون العضو ip_version دائمًا 4 أو 6. يجب أن يكون binary_format_major_version دائمًا 2.

لا يوجد شرط بأن تتضمن البيانات الوصفية لقاعدة البيانات لغات أو أوصافًا، لذا يمكن أن يكون count لهذه الأجزاء من البيانات الوصفية صفرًا. يجب ملء جميع حقول MMDB_metadata_s الأخرى.

MMDB_lookup_result_s

يتم إرجاع هذا الهيكل كنتيجة للبحث عن عنوان IP.

typedef struct MMDB_lookup_result_s {
    bool found_entry;
    MMDB_entry_s entry;
    uint16_t netmask;
} MMDB_lookup_result_s;

إذا كان العضو found_entry خاطئًا، فإن الأعضاء الأخرى في هذا الهيكل لا تحتوي على قيم ذات معنى. تحقق دائمًا من أن found_entry صحيح أولاً.

يُستخدم العضو entry للبحث عن البيانات المرتبطة بعنوان IP.

يخبرك العضو netmask بالشبكة الفرعية التي ينتمي إليها عنوان IP في قاعدة البيانات هذه. على سبيل المثال، إذا قمت بالبحث عن العنوان 1.1.1.1 في قاعدة بيانات IPv4 وكان netmask المُعاد هو 16، فإن العنوان جزء من الشبكة الفرعية 1.1.0.0/16.

إذا كانت قاعدة البيانات هي قاعدة بيانات IPv6، فإن قناع الشبكة المُعاد هو دائمًا طول بادئة IPv6 (من 0-128)، حتى لو كانت قاعدة البيانات هذه تحتوي أيضًا على شبكات IPv4. إذا قمت بالبحث عن عنوان IPv4 وترغب في تحويل قناع الشبكة إلى قيمة قناع شبكة IPv4، يمكنك ببساطة طرح 96 من القيمة.

MMDB_result_s

لست بحاجة حقًا إلى التعمق في هذا الهيكل. ستحصل عليه من الهيكل MMDB_lookup_result_s وتمرره إلى وظائف مختلفة.

MMDB_entry_data_s

يُستخدم هذا الهيكل لإرجاع إدخال قسم بيانات واحد لعنوان IP. يمكن أن تشير هذه الإدخالات بدورها إلى إدخالات أخرى، كما هو الحال بالنسبة لأشياء مثل الخرائط والمصفوفات. لم يتم توثيق بعض أعضاء هذا الهيكل لأنها مخصصة للاستخدام الداخلي فقط.

typedef struct MMDB_entry_data_s {
    bool has_data;
    union {
        uint32_t pointer;
        const char *utf8_string;
        double double_value;
        const uint8_t *bytes;
        uint16_t uint16;
        uint32_t uint32;
        int32_t int32;
        uint64_t uint64;
        {mmdb_uint128_t or uint8_t[16]} uint128;
        bool boolean;
        float float_value;
    };
    ...
    uint32_t data_size;
    uint32_t type;
} MMDB_entry_data_s;

العضو has_data صحيح إذا تم العثور على بيانات لعملية بحث معينة. راجع MMDB_get_value() لمزيد من التفاصيل. إذا كان هذا العضو خاطئًا، فإن أيًا من القيم الأخرى في الهيكل ليست ذات معنى.

يحدد الاتحاد في بداية الهيكل البيانات الفعلية. لتحديد أي عضو من أعضاء الاتحاد

MMDB_lookup_string()

MMDB_lookup_result_s MMDB_lookup_string(
    MMDB_s *const mmdb,
    const char *const ipstr,
    int *const gai_error,
    int *const mmdb_error);

تبحث هذه الدالة عن عنوان IP يتم تمريره كسلسلة نصية منتهية بالصفر. تستدعي داخليًا الدالة getaddrinfo() لحل العنوان إلى شكله الثنائي. ثم تستدعي الدالة MMDB_lookup_sockaddr() للبحث عن العنوان في قاعدة البيانات. إذا كنت قد حللت عنوانًا بالفعل، يمكنك استدعاء الدالة MMDB_lookup_sockaddr() مباشرة، دون الحاجة إلى حله مرتين.

int gai_error, mmdb_error;
MMDB_lookup_result_s result =
    MMDB_lookup_string(&mmdb, "1.2.3.4", &gai_error, &mmdb_error);
if (0 != gai_error) { ... }
if (MMDB_SUCCESS != mmdb_error) { ... }
if (result.found_entry) { ... }

تعيد هذه الدالة دائمًا بنية MMDB_lookup_result_s، ولكن يجب عليك أيضًا التحقق من معلمتي gai_error و mmdb_error. إذا أشارت أي منهما إلى خطأ، فإن البنية المعاد تكون عديمة الفائدة.

إذا لم يحدث أي خطأ، يجب عليك أيضًا التأكد من أن عضو found_entry في البنية المعاد صحيح. إذا لم يكن كذلك، فهذا يعني أن عنوان IP ليس له سجل في قاعدة البيانات.

تعمل هذه الدالة مع عناوين IPv4 حتى عندما تحتوي قاعدة البيانات على بيانات لكل من IPv4 و IPv6. سيتم البحث عن عنوان IPv4 باعتباره "::xxx.xxx.xxx.xxx" بدلاً من إعادة تعيينه إلى نطاق ::ffff:xxx.xxx.xxx.xxx المخصص لعناوين IPv6 المكتظة بـ IPv4.

إذا مررت عنوان IPv6 إلى قاعدة بيانات تحتوي فقط على بيانات IPv4، فسيكون العضو found_entry خاطئًا، ولكن سيظل حالة mmdb_error هي MMDB_SUCCESS.

MMDB_lookup_sockaddr()

MMDB_lookup_result_s MMDB_lookup_sockaddr(
    MMDB_s *const mmdb,
    const struct sockaddr *const sockaddr,
    int *const mmdb_error);

تبحث هذه الدالة عن عنوان IP تم حله بالفعل باستخدام getaddrinfo().

باستثناء عدم استدعاء getaddrinfo()، فإن هذه الدالة متطابقة مع الدالة MMDB_lookup_string().

int mmdb_error;
MMDB_lookup_result_s result =
    MMDB_lookup_sockaddr(&mmdb, address->ai_addr, &mmdb_error);
if (MMDB_SUCCESS != mmdb_error) { ... }
if (result.found_entry) { ... }

وظائف البحث عن البيانات

هناك ثلاث دوال للبحث عن البيانات المرتبطة بعنوان IP.

int MMDB_get_value(
    MMDB_entry_s *const start,
    MMDB_entry_data_s *const entry_data,
    ...);
int MMDB_vget_value(
    MMDB_entry_s *const start,
    MMDB_entry_data_s *const entry_data,
    va_list va_path);
int MMDB_aget_value(
    MMDB_entry_s *const start,
    MMDB_entry_data_s *const entry_data,
    const char *const *const path);

تسمح الدوال الثلاث بثلاثة أنماط استدعاء مختلفة قليلاً، ولكنها جميعها تقوم بنفس المهمة.

المعلمة الأولى هي قيمة MMDB_entry_s. في معظم الحالات، سيأتي هذا من القيمة MMDB_lookup_result_s التي أعادتها الدالة MMDB_lookup_string() أو MMDB_lookup_sockaddr().

المعلمة الثانية هي مرجع إلى بنية MMDB_entry_data_s. سيتم ملء هذه البنية بالبيانات التي يتم البحث عنها، إن وجدت. إذا لم يتم العثور على شيء، فسيكون العضو has_data لهذه البنية خاطئًا. إذا كان has_data صحيحًا، يمكنك فحص العضو data_type.

المعلمة الأخيرة هي مسار بحث. يتكون المسار من سلسلة من السلاسل النصية التي تمثل مفاتيح خريطة (على سبيل المثال، "city") أو فهارس مصفوفة (على سبيل المثال، "0"، "1"، "-1") لاستخدامها في البحث.

تُعامل فهارس المصفوفة السالبة كإزاحة من نهاية المصفوفة. على سبيل المثال، يشير "-1" إلى العنصر الأخير في المصفوفة.

يسمح مسار البحث بالتنقل عبر بنية معقدة. على سبيل المثال، بالنظر إلى هذه البنية:

{
    "names": {
        "en": "Germany",
        "de": "Deutschland"
    },
    "cities": [ "Berlin", "Frankfurt" ]
}

يمكننا البحث عن الاسم باللغة الإنجليزية باستخدام هذا الرمز:

MMDB_lookup_result_s result =
    MMDB_lookup_sockaddr(&mmdb, address->ai_addr, &mmdb_error);
MMDB_entry_data_s entry_data;
int status =
    MMDB_get_value(&result.entry, &entry_data,
                   "names", "en", NULL);
if (MMDB_SUCCESS != status) { ... }
if (entry_data.has_data) { ... }

إذا أردنا العثور على المدينة الأولى، فسيكون مسار البحث هو "cities" و "0". إذا لم تحدد مسار بحث، فستحصل على السجل المقابل للخريطة العليا. يجب أن ينتهي مسار البحث دائمًا بـ NULL، بغض النظر عن الدالة التي تستخدمها.

تقبل الدالة MMDB_get_value عددًا متغيرًا من الوسائط. جميع الوسائط بعد المؤشر إلى MMDB_entry_data_s هي مسار البحث. يجب أن تكون الوسيطة الأخيرة NULL.

تقبل الدالة MMDB_vget_value قائمة va_list كمسار بحث. يجب أن تكون الوسيطة الأخيرة التي تم استردادها بواسطة va_arg() هي NULL.

أخيرًا، تقبل الدالة MMDB_aget_value مصفوفة من السلاسل النصية كمسار بحث. يجب أن يكون العنصر الأخير في هذه المصفوفة NULL.

إذا كنت تريد الحصول على جميع بيانات الإدخال دفعة واحدة، يمكنك استدعاء الدالة MMDB_get_entry_data_list().

لكل من الدوال الثلاث، القيمة المعاد هي رمز حالة كما هو محدد أعلاه.

MMDB_get_entry_data_list()

int MMDB_get_entry_data_list(
    MMDB_entry_s *start,
    MMDB_entry_data_list_s **const entry_data_list);

تسمح هذه الدالة بالحصول على جميع البيانات من بنية معقدة دفعة واحدة، بدلاً من البحث عن كل جزء عبر استدعاءات متكررة لـ MMDB_get_value().

MMDB_lookup_result_s result =
    MMDB_lookup_sockaddr(&mmdb, address->ai_addr, &mmdb_error);
MMDB_entry_data_list_s *entry_data_list, *first;
int status =
    MMDB_get_entry_data_list(&result.entry, &entry_data_list);
if (MMDB_SUCCESS != status) { ... }
// حفظ هذا ليتم تحريره لاحقًا
first = entry_data_list;
while (1) {
    MMDB_entry_data_list_s *next = entry_data_list = entry_data_list->next;
    if (NULL == next) {
        break;
    }
    switch (next->entry_data.type) {
        case MMDB_DATA_TYPE_MAP: { ... }
        case MMDB_DATA_TYPE_UTF8_STRING: { ... }
        ...
    }
}
MMDB_free_entry_data_list(first);

إنها مسألة تفسير بنية entry_data_list. القائمة مرتبطة بترتيب العمق أولاً. سأستخدم هذه البنية كمثال:

{
    "names": {
        "en": "Germany",
        "de": "Deutschland"
    },
    "cities": [ "Berlin", "Frankfurt" ]
}

ستحتوي القائمة على العناصر التالية:

1.
MAP - الخريطة الجذرية
2.
UTF8_STRING - المفتاح "names"
3.
MAP - خريطة "names"
4.
UTF8_STRING - المفتاح "en"
5.
UTF8_STRING - القيمة للمفتاح "en"
6.
UTF8_STRING - المفتاح "de"
7.
UTF8_STRING - القيمة للمفتاح "de"
8.
UTF8_STRING - المفتاح "cities"
9.
ARRAY - القيمة للمفتاح "cities"
10.
UTF8_STRING - array[0]
11.
UTF8_STRING - array[1]

القيمة المعاد من الدالة هي رمز حالة كما هو محدد أعلاه.

MMDB_free_entry_data_list()

void MMDB_free_entry_data_list(
    MMDB_entry_data_list_s *const entry_data_list);

تخصص الدوال MMDB_get_entry_data_list() و MMDB_get_metadata_as_entry_data_list() بنية قائمة مرتبطة بالذاكرة. استدعِ هذه الدالة لتحرير بنية MMDB_entry_data_list_s.

MMDB_get_metadata_as_entry_data_list()

int MMDB_get_metadata_as_entry_data_list(
    MMDB_s *const mmdb,
    MMDB_entry_data_list_s **const entry_data_list);

تسمح هذه الدالة باسترداد بيانات التعريف الخاصة بقاعدة البيانات كقائمة مرتبطة من هياكل MMDB_entry_data_list_s. قد يكون هذا أكثر ملاءمة من استخدام بنية بيانات التعريف مباشرة.

    MMDB_entry_data_list_s *entry_data_list, *first;
    int status =
        MMDB_get_metadata_as_entry_data_list(&mmdb, &entry_data_list);
    if (MMDB_SUCCESS != status) { ... }
    first = entry_data_list;
    ... // قم بعمل شيء بالبيانات
    MMDB_free_entry_data_list(first);

القيمة المعاد من الدالة هي رمز حالة كما هو محدد أعلاه.

MMDB_dump_entry_data_list()

int MMDB_dump_entry_data_list(
    FILE *const stream,
    MMDB_entry_data_list_s *const entry_data_list,
    int indent);

تأخذ هذه الدالة قائمة مرتبطة من هياكل MMDB_entry_data_list_s وتحولها إلى التسلسل في التدفق المعطى. المعلمة indent هي مستوى المسافة البادئة الأولي للإخراج الذي تم إنشاؤه. يتم زيادتها للهياكل المتداخلة (الخرائط، المصفوفات، إلخ).

يجب أن يكون التدفق مقبض ملف (stdout، إلخ). إذا وفرت منصتك شيئًا مثل open_memstream() (GNU)، يمكنك استخدامه لالتقاط الإخراج كسلسلة نصية.

الإخراج منسق بشكل مشابه لـ JSON، ولكن القيم مميزة بنوع بياناتها (ما لم يتم عرض الخرائط والمصفوفات على أنها "{" و "[]" على التوالي).

قد يتغير الشكل المحدد للإخراج في الإصدارات المستقبلية.
