MMDB_lookup_sockaddr(3)

MMDB_lookup_string()

MMDB_lookup_result_s MMDB_lookup_string(
    MMDB_s *const mmdb,
    const char *const ipstr,
    int *const gai_error,
    int *const mmdb_error);

تبحث هذه الدالة عن عنوان IP يتم تمريره كسلسلة نصية منتهية بصفر. تقوم داخليًا باستدعاء getaddrinfo() لحل العنوان إلى شكله الثنائي. ثم تقوم باستدعاء MMDB_lookup_sockaddr() للبحث عن العنوان في قاعدة البيانات. إذا كنت قد قمت بالفعل بحل عنوان، يمكنك استدعاء MMDB_lookup_sockaddr() مباشرة دون حله مرتين.

int gai_error, mmdb_error;
MMDB_lookup_result_s result =
    MMDB_lookup_string(&mmdb, "1.2.3.4", &gai_error, &mmdb_error);
if (0 != gai_error) { ... }
if (MMDB_SUCCESS != mmdb_error) { ... }
if (result.found_entry) { ... }

ترجع هذه الدالة دائمًا هيكل MMDB_lookup_result_s، ولكن يجب عليك أيضًا التحقق من معاملي gai_error و mmdb_error. إذا أشار أي منهما إلى خطأ، فإن الهيكل المرتجع يكون عديم الفائدة.

إذا لم يكن هناك خطأ، يجب عليك التأكد من أن عضو found_entry في الهيكل المرتجع صحيح. إذا لم يكن كذلك، فهذا يعني أن عنوان IP ليس له سجل في قاعدة البيانات.

تعمل هذه الدالة مع عناوين IPv4 حتى عندما تحتوي قاعدة البيانات على بيانات لكل من IPv4 و IPv6. سيتم البحث عن عنوان IPv4 كـ ::xxx.xxx.xxx.xxx، وليس كـ ::ffff:xxx.xxx.xxx.xxx.

إذا مررت عنوان IPv6 إلى قاعدة بيانات تحتوي على بيانات IPv4 فقط، فسيكون العضو found_entry خاطئًا، ولكن حالة mmdb_error ستظل MMDB_SUCCESS.

MMDB_lookup_sockaddr()

MMDB_lookup_result_s MMDB_lookup_sockaddr(
    MMDB_s *const mmdb,
    const struct sockaddr *const sockaddr,
    int *const mmdb_error);

تبحث هذه الدالة عن عنوان IP تم حله بالفعل بواسطة getaddrinfo().

ما لم تكن تقوم باستدعاء getaddrinfo() بنفسك، فإن هذه الدالة متطابقة مع الدالة MMDB_lookup_string().

int mmdb_error;
MMDB_lookup_result_s result =
    MMDB_lookup_sockaddr(&mmdb, address->ai_addr, &mmdb_error);
if (MMDB_SUCCESS != mmdb_error) { ... }
if (result.found_entry) { ... }

البحث عن البيانات المرتبطة بعنوان IP

هناك ثلاث دوال للبحث عن البيانات المرتبطة بعنوان IP.

int MMDB_get_value(
    MMDB_entry_s *const start,
    MMDB_entry_data_s *const entry_data,
    ...);
int MMDB_vget_value(
    MMDB_entry_s *const start,
    MMDB_entry_data_s *const entry_data,
    va_list va_path);
int MMDB_aget_value(
    MMDB_entry_s *const start,
    MMDB_entry_data_s *const entry_data,
    const char *const *const path);

تسمح الدوال الثلاث بأنماط استدعاء مختلفة قليلاً، ولكنها جميعًا تقوم بنفس المهمة.

المعلمة الأولى هي قيمة MMDB_entry_s. في معظم الحالات، ستأتي هذه القيمة من هيكل MMDB_lookup_result_s المرتجع من MMDB_lookup_string() أو MMDB_lookup_sockaddr().

المعلمة الثانية هي مرجع إلى هيكل MMDB_entry_data_s. سيتم ملء هذا بالبيانات التي يتم البحث عنها، إذا تم العثور على أي شيء. إذا لم يتم العثور على شيء، فسيكون العضو has_data في هذا الهيكل خاطئًا. إذا كان has_data صحيحًا، يمكنك رؤية العضو data_type.

المعلمة الأخيرة هي مسار بحث. يتكون المسار من سلسلة من السلاسل النصية التي تمثل مفاتيح الخريطة (على سبيل المثال، "city") أو فهارس المصفوفة (على سبيل المثال، "0"، "1"، "-1") لاستخدامها في البحث.

سيتم التعامل مع فهارس المصفوفات السالبة كإزاحة من نهاية المصفوفة. على سبيل المثال، "-1" يشير إلى العنصر الأخير في المصفوفة.

يسمح مسار البحث بالتنقل في هيكل معقد. على سبيل المثال، بالنظر إلى هيكل البيانات هذا:

{
    "names": {
        "en": "Germany",
        "de": "Deutschland"
    },
    "cities": [ "Berlin", "Frankfurt" ]
}

يمكننا البحث عن الاسم باللغة الإنجليزية باستخدام هذا الكود:

MMDB_lookup_result_s result =
    MMDB_lookup_sockaddr(&mmdb, address->ai_addr, &mmdb_error);
MMDB_entry_data_s entry_data;
int status =
    MMDB_get_value(&result.entry, &entry_data,
                   "names", "en", NULL);
if (MMDB_SUCCESS != status) { ... }
if (entry_data.has_data) { ... }

إذا أردنا العثور على المدينة الأولى، فسيكون مسار البحث "cities" و "0". إذا لم تقدم مسار بحث، فستحصل على السجل الذي يتوافق مع الخريطة الجذرية. يجب أن ينتهي مسار البحث دائمًا بـ NULL، بغض النظر عن الدالة التي تستخدمها.

تقبل الدالة MMDB_get_value عددًا متغيرًا من الوسائط. جميع الوسائط بعد المؤشر إلى MMDB_entry_data_s هي مسار البحث. يجب أن تكون الوسيطة الأخيرة NULL.

تقبل الدالة MMDB_vget_value قائمة va_list كمسار بحث. يجب أن يكون العنصر الأخير المسترد باستخدام va_arg() هو NULL.

أخيرًا، تقبل الدالة MMDB_aget_value مصفوفة من السلاسل النصية كمسار بحث. يجب أن يكون العنصر الأخير في هذه المصفوفة NULL.

إذا كنت ترغب في الحصول على جميع بيانات الإدخال في وقت واحد، يمكنك استدعاء MMDB_get_entry_data_list().

لكل من الدوال الثلاث، القيمة المرتجعة هي رمز حالة كما هو محدد أعلاه.

MMDB_get_entry_data_list()

int MMDB_get_entry_data_list(

MMDB_lookup_result_s result =
    MMDB_lookup_sockaddr(&mmdb, address->ai_addr, &mmdb_error);
MMDB_entry_data_list_s *entry_data_list, *first;
int status =
    MMDB_get_entry_data_list(&result.entry, &entry_data_list);
if (MMDB_SUCCESS != status) { ... }
// save this so we can free this data later
first = entry_data_list;
while (1) {

    if (NULL == next) {
        break;
    }
    switch (next->entry_data.type) {
        case MMDB_DATA_TYPE_MAP: { ... }
        case MMDB_DATA_TYPE_UTF8_STRING: { ... }
        ...
    }
}
MMDB_free_entry_data_list(first);

تسمح لك هذه الدالة باسترداد بيانات تعريف قاعدة البيانات كقائمة مترابطة من هياكل MMDB_entry_data_list_s. قد يكون هذا أكثر ملاءمة من استخدام هيكل بيانات التعريف مباشرة.

    MMDB_entry_data_list_s *entry_data_list, *first;
    int status =
        MMDB_get_metadata_as_entry_data_list(&mmdb, &entry_data_list);
    if (MMDB_SUCCESS != status) { ... }
    first = entry_data_list;
    ... // افعل شيئًا بالبيانات
    MMDB_free_entry_data_list(first);

MMDB_dump_entry_data_list()

int MMDB_dump_entry_data_list(
    FILE *const stream,
    MMDB_entry_data_list_s *const entry_data_list,
    int indent);

تستقبل هذه الدالة قائمة مترابطة من هياكل MMDB_entry_data_list_s وتقوم بتحويلها إلى تدفق محدد stream. المعلمة indent هي مستوى المسافة البادئة الأولي للإخراج الذي تم إنشاؤه. يتم زيادتها للهياكل المتداخلة (الخرائط، المصفوفات، إلخ).

يجب أن يكون stream مقبض ملف (stdout، إلخ). إذا كانت منصتك توفر شيئًا مثل open_memstream() (GNU)، يمكنك استخدامه لالتقاط الإخراج كسلسلة نصية.

الإخراج منسق بأسلوب يشبه JSON، ولكن القيم موسومة بنوع بياناتها (باستثناء الخرائط والمصفوفات التي يتم عرضها كـ "{}" و "[]" على التوالي).
