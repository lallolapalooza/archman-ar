MMDB_close(3)

الحزمة: libmaxminddb

التراخيص: Apache-2.0

للهيكل نفسه! إذا خصصت الهيكل من الكومة، فأنت مسؤول عن تحريره.

MMDB_lookup_string()

MMDB_lookup_result_s MMDB_lookup_string(
    MMDB_s *const mmdb,
    const char *const ipstr,
    int *const gai_error,
    int *const mmdb_error);

تقوم هذه الوظيفة بالبحث عن عنوان IP يتم تمريره كسلسلة نصية منتهية بالصفر. داخليًا، تستدعي getaddrinfo() لحل العنوان إلى شكل ثنائي. ثم تستدعي MMDB_lookup_sockaddr() للبحث عن العنوان في قاعدة البيانات. إذا كنت قد قمت بالفعل بحل عنوان، يمكنك استدعاء MMDB_lookup_sockaddr() مباشرة، بدلاً من حل العنوان مرتين.

int gai_error, mmdb_error;
MMDB_lookup_result_s result =
    MMDB_lookup_string(&mmdb, "1.2.3.4", &gai_error, &mmdb_error);
if (0 != gai_error) { ... }
if (MMDB_SUCCESS != mmdb_error) { ... }
if (result.found_entry) { ... }

ترجع هذه الوظيفة دائمًا بنية MMDB_lookup_result_s، ولكن يجب عليك أيضًا التحقق من معلمات gai_error و mmdb_error. إذا أشارت أي من هذه إلى خطأ، فإن البنية المرجعة لا معنى لها.

إذا لم يحدث خطأ، فلا يزال يتعين عليك التأكد من أن عضو found_entry في النتيجة المرجعة صحيح. إذا لم يكن كذلك، فهذا يعني أن عنوان IP لا يحتوي على إدخال في قاعدة البيانات.

ستعمل هذه الوظيفة مع عناوين IPv4 حتى عندما تحتوي قاعدة البيانات على بيانات لكل من عناوين IPv4 و IPv6. سيتم البحث عن عنوان IPv4 على أنه '::xxx.xxx.xxx.xxx' بدلاً من إعادة تعيينه إلى الكتلة ::ffff:xxx.xxx.xxx.xxx المخصصة لعناوين IPv6 المعينة من IPv4.

إذا قمت بتمرير عنوان IPv6 إلى قاعدة بيانات تحتوي فقط على بيانات IPv4، فسيكون عضو found_entry خاطئًا، ولكن حالة mmdb_error ستظل MMDB_SUCCESS.

MMDB_lookup_sockaddr()

MMDB_lookup_result_s MMDB_lookup_sockaddr(
    MMDB_s *const mmdb,
    const struct sockaddr *const sockaddr,
    int *const mmdb_error);

تبحث هذه الوظيفة عن عنوان IP تم حله بالفعل بواسطة getaddrinfo().

بخلاف عدم استدعاء getaddrinfo() بنفسها، هذه الوظيفة مطابقة تمامًا لوظيفة MMDB_lookup_string().

int mmdb_error;
MMDB_lookup_result_s result =
    MMDB_lookup_sockaddr(&mmdb, address->ai_addr, &mmdb_error);
if (MMDB_SUCCESS != mmdb_error) { ... }
if (result.found_entry) { ... }

وظائف البحث عن البيانات

هناك ثلاث وظائف للبحث عن البيانات المرتبطة بعنوان IP.

int MMDB_get_value(
    MMDB_entry_s *const start,
    MMDB_entry_data_s *const entry_data,
    ...);
int MMDB_vget_value(
    MMDB_entry_s *const start,
    MMDB_entry_data_s *const entry_data,
    va_list va_path);
int MMDB_aget_value(
    MMDB_entry_s *const start,
    MMDB_entry_data_s *const entry_data,
    const char *const *const path);

تسمح الوظائف الثلاث بثلاثة أساليب استدعاء مختلفة قليلاً، لكنها جميعًا تفعل الشيء نفسه.

المعلمة الأولى هي قيمة MMDB_entry_s. في معظم الحالات، ستأتي من قيمة MMDB_lookup_result_s التي تم إرجاعها بواسطة MMDB_lookup_string() أو MMDB_lookup_sockaddr().

المعلمة الثانية هي مرجع إلى بنية MMDB_entry_data_s. سيتم ملؤها بالبيانات التي يتم البحث عنها، إن وجدت. إذا لم يتم العثور على شيء، فسيتم تعيين عضو has_data في هذه البنية إلى false. إذا كان has_data صحيحًا، فيمكنك النظر إلى عضو data_type.

المعلمة الأخيرة هي مسار بحث. يتكون المسار من مجموعة من السلاسل النصية التي تمثل إما مفاتيح الخريطة (مثل، "city") أو فهارس المصفوفة (مثل، "0"، "1"، "-1") لاستخدامها في البحث.

سيتم التعامل مع فهارس المصفوفة السالبة كإزاحة من نهاية المصفوفة. على سبيل المثال، يشير "-1" إلى العنصر الأخير في المصفوفة.

يسمح لك مسار البحث بالتنقل في بنية بيانات معقدة. على سبيل المثال، بالنظر إلى هذه البيانات:

{
    "names": {
        "en": "Germany",
        "de": "Deutschland"
    },
    "cities": [ "Berlin", "Frankfurt" ]
}

يمكننا البحث عن الاسم الإنجليزي باستخدام هذا الكود:

MMDB_lookup_result_s result =
    MMDB_lookup_sockaddr(&mmdb, address->ai_addr, &mmdb_error);
MMDB_entry_data_s entry_data;
int status =
    MMDB_get_value(&result.entry, &entry_data,
                   "names", "en", NULL);
if (MMDB_SUCCESS != status) { ... }
if (entry_data.has_data) { ... }

إذا أردنا العثور على المدينة الأولى، فسيكون مسار البحث "cities", "0". إذا لم تقم بتوفير مسار بحث على الإطلاق، فستحصل على الإدخال الذي يتوافق مع الخريطة على المستوى الأعلى. يجب أن ينتهي مسار البحث دائمًا بـ NULL، بغض النظر عن الوظيفة التي تستدعيها.

تأخذ وظيفة MMDB_get_value عددًا متغيرًا من الوسائط. جميع الوسائط بعد مؤشر بنية MMDB_entry_data_s * هي مسار البحث. يجب أن تكون الوسيطة الأخيرة NULL.

تقبل وظيفة MMDB_vget_value قائمة va_list كمسار بحث. يجب أن تكون العنصر الأخير الذي تم استرداده بواسطة va_arg() هو NULL.

أخيرًا، تقبل وظيفة MMDB_aget_value مصفوفة من السلاسل النصية كمسار بحث. يجب أن يكون العضو الأخير في هذه المصفوفة NULL.

إذا كنت ترغب في الحصول على جميع بيانات الإدخال دفعة واحدة، يمكنك استدعاء MMDB_get_entry_data_list() بدلاً من ذلك.

لكل من الوظائف الثلاث، القيمة المرجعة هي رمز حالة كما هو محدد أعلاه.

MMDB_get_entry_data_list()

int MMDB_get_entry_data_list(
    MMDB_entry_s *start,
    MMDB_entry_data_list_s **const entry_data_list);

تسمح لك هذه الوظيفة بالحصول على جميع البيانات لهيكل بيانات معقد دفعة واحدة، بدلاً من البحث عن كل جزء باستخدام استدعاءات متكررة لـ MMDB_get_value().

MMDB_lookup_result_s result =
    MMDB_lookup_sockaddr(&mmdb, address->ai_addr, &mmdb_error);
MMDB_entry_data_list_s *entry_data_list, *first;
int status =
    MMDB_get_entry_data_list(&result.entry, &entry_data_list);
if (MMDB_SUCCESS != status) { ... }
// حفظ هذا حتى نتمكن من تحرير هذه البيانات لاحقًا
first = entry_data_list;
while (1) {
    MMDB_entry_data_list_s *next = entry_data_list = entry_data_list->next;
    if (NULL == next) {
        break;
    }
    switch (next->entry_data.type) {
        case MMDB_DATA_TYPE_MAP: { ... }
        case MMDB_DATA_TYPE_UTF8_STRING: { ... }
        ...
    }
}
MMDB_free_entry_data_list(first);

يعتمد عليك تفسير بنية بيانات entry_data_list. القائمة مرتبطة في اجتياز عمق أول. دعنا نستخدم هذه البنية كمثال:

{
    "names": {
        "en": "Germany",
        "de": "Deutschland"
    },
    "cities": [ "Berlin", "Frankfurt" ]
}

ستتكون القائمة من العناصر التالية:

1.
MAP - الخريطة على المستوى الأعلى
2.
UTF8_STRING - المفتاح "names"
3.
MAP - خريطة للمفتاح "names"
4.
UTF8_STRING - المفتاح "en"
5.
UTF8_STRING - القيمة للمفتاح "en"
6.
UTF8_STRING - المفتاح "de"
7.
UTF8_STRING - القيمة للمفتاح "de"
8.
UTF8_STRING - المفتاح "cities"
9.
ARRAY - القيمة للمفتاح "cities"
10.
UTF8_STRING - array[0]
11.
UTF8_STRING - array[1]

القيمة المرجعة للوظيفة هي رمز حالة كما هو محدد أعلاه.

MMDB_free_entry_data_list()

void MMDB_free_entry_data_list(
    MMDB_entry_data_list_s *const entry_data_list);

ستقوم وظائف MMDB_get_entry_data_list() و MMDB_get_metadata_as_entry_data_list() بتخصيص بنية القائمة المرتبطة من الكومة. استدعِ هذه الوظيفة لتحرير بنية MMDB_entry_data_list_s.

MMDB_get_metadata_as_entry_data_list()

int MMDB_get_metadata_as_entry_data_list(
    MMDB_s *const mmdb,
    MMDB_entry_data_list_s **const entry_data_list);

تسمح لك هذه الوظيفة باسترداد بيانات تعريف قاعدة البيانات كقائمة مرتبطة من هياكل MMDB_entry_data_list_s. يمكن أن يكون هذا طريقة أكثر ملاءمة للتعامل مع بيانات التعريف من استخدام بنية بيانات التعريف مباشرة.

    MMDB_entry_data_list_s *entry_data_list, *first;
    int status =
        MMDB_get_metadata_as_entry_data_list(&mmdb, &entry_data_list);
    if (MMDB_SUCCESS != status) { ... }
    first = entry_data_list;
    ... // قم بشيء ما بالبيانات
    MMDB_free_entry_data_list(first);

القيمة المرجعة للوظيفة هي رمز حالة كما هو محدد أعلاه.

MMDB_dump_entry_data_list()

int MMDB_dump_entry_data_list(
    FILE *const stream,
    MMDB_entry_data_list_s *const entry_data_list,
    int indent);

تأخذ هذه الوظيفة قائمة مرتبطة من هياكل MMDB_entry_data_list_s وتحولها إلى سلسلة نصية إلى stream المحدد. المعلمة indent هي مستوى المسافة البادئة الأولي للإخراج الذي تم إنشاؤه. يتم زيادتها لهياكل البيانات المتداخلة (الخرائط، المصفوفات، إلخ).

يجب أن يكون stream مقبض ملف ( stdout، إلخ). إذا كانت منصتك توفر شيئًا مثل GNU open_memstream()، يمكنك استخدامه لالتقاط الإخراج كسلسلة نصية.

يتم تنسيق الإخراج بطريقة تشبه JSON، ولكن يتم تمييز القيم بأنواع بياناتها (باستثناء الخرائط والمصفوفات التي يتم عرضها مع ""{}"" و ""[]"" على التوالي).

قد يتغير تنسيق الإخراج المحدد في الإصدارات المستقبلية.
