MMDB_strerror(3)

الحزمة: libmaxminddb

التراخيص: Apache-2.0

MMDB_lookup_string()

MMDB_lookup_result_s MMDB_lookup_string(
    MMDB_s *const mmdb,
    const char *const ipstr,
    int *const gai_error,
    int *const mmdb_error);

يبحث هذا التابع عن عنوان IP يتم تمريره كسلسلة نصية منتهية بالصفر. يقوم داخليًا باستدعاء getaddrinfo() لحل العنوان إلى شكله الثنائي. ثم يقوم باستدعاء MMDB_lookup_sockaddr() للبحث عن العنوان في قاعدة البيانات. إذا كنت قد قمت بالفعل بحل عنوان، يمكنك استدعاء MMDB_lookup_sockaddr() مباشرة دون حلها مرتين.

int gai_error, mmdb_error;
MMDB_lookup_result_s result =
    MMDB_lookup_string(&mmdb, "1.2.3.4", &gai_error, &mmdb_error);
if (0 != gai_error) { ... }
if (MMDB_SUCCESS != mmdb_error) { ... }
if (result.found_entry) { ... }

يعيد هذا التابع دائمًا هيكل MMDB_lookup_result_s، ولكن يجب عليك أيضًا التحقق من معاملي gai_error و mmdb_error. إذا أشار أي منهما إلى خطأ، فإن الهيكل المعاد يكون غير فعال.

إذا لم يحدث أي خطأ، فلا يزال يتعين عليك التأكد من أن العضو found_entry في النتيجة المعاد صحيح. إذا لم يكن كذلك، فهذا يعني أن عنوان IP ليس له سجل في قاعدة البيانات.

يعمل هذا التابع مع عناوين IPv4 حتى لو كانت قاعدة البيانات تحتوي على بيانات لكل من IPv4 و IPv6. سيتم البحث عن عنوان IPv4 كـ ::xxx.xxx.xxx.xxx، وليس كـ ::ffff:xxx.xxx.xxx.xxx.

إذا قمت بتمرير عنوان IPv6 إلى قاعدة بيانات تحتوي فقط على بيانات IPv4، فسيكون العضو found_entry خاطئًا، ولكن سيظل حالة mmdb_error هي MMDB_SUCCESS.

MMDB_lookup_sockaddr()

MMDB_lookup_result_s MMDB_lookup_sockaddr(
    MMDB_s *const mmdb,
    const struct sockaddr *const sockaddr,
    int *const mmdb_error);

يبحث هذا التابع عن عنوان IP تم حله بالفعل بواسطة getaddrinfo().

بدون استدعاء getaddrinfo() بنفسه، فإن هذا التابع مطابق للتابع MMDB_lookup_string().

int mmdb_error;
MMDB_lookup_result_s result =
    MMDB_lookup_sockaddr(&mmdb, address->ai_addr, &mmdb_error);
if (MMDB_SUCCESS != mmdb_error) { ... }
if (result.found_entry) { ... }

وظائف البحث عن البيانات

هناك ثلاثة توابع للبحث عن البيانات المرتبطة بعنوان IP.

int MMDB_get_value(
    MMDB_entry_s *const start,
    MMDB_entry_data_s *const entry_data,
    ...);
int MMDB_vget_value(
    MMDB_entry_s *const start,
    MMDB_entry_data_s *const entry_data,
    va_list va_path);
int MMDB_aget_value(
    MMDB_entry_s *const start,
    MMDB_entry_data_s *const entry_data,
    const char *const *const path);

تسمح التوابع الثلاثة بثلاثة أنماط استدعاء مختلفة قليلاً، ولكنها جميعًا تقوم بنفس المهمة.

المعلمة الأولى هي قيمة من MMDB_entry_s. في معظم الحالات، ستأتي هذه من قيمة MMDB_lookup_result_s التي أعادها MMDB_lookup_string() أو MMDB_lookup_sockaddr().

المعلمة الثانية هي مؤشر إلى هيكل MMDB_entry_data_s. سيتم ملء هذا بالبيانات التي يتم البحث عنها، إذا تم العثور على أي بيانات. إذا لم تكن هناك بيانات، فسيكون العضو has_data لهذا الهيكل خاطئًا. إذا كان has_data صحيحًا، يمكنك فحص العضو data_type.

المعلمة الأخيرة هي مسار بحث. يتكون المسار من سلسلة من السلاسل النصية التي تمثل مفاتيح الخريطة (على سبيل المثال، "city") أو فهارس المصفوفة (على سبيل المثال، "0"، "1"، "-1") لاستخدامها في البحث.

تعامل فهارس المصفوفة السالبة كإزاحة من نهاية المصفوفة. على سبيل المثال، "-1" يشير إلى العنصر الأخير في المصفوفة.

يسمح مسار البحث بالتنقل عبر هيكل معقد. على سبيل المثال، بالنظر إلى هذه البيانات:

{
    "names": {
        "en": "Germany",
        "de": "Deutschland"
    },
    "cities": [ "Berlin", "Frankfurt" ]
}

يمكننا البحث عن الاسم باللغة الإنجليزية باستخدام هذا الكود:

MMDB_lookup_result_s result =
    MMDB_lookup_sockaddr(&mmdb, address->ai_addr, &mmdb_error);
MMDB_entry_data_s entry_data;
int status =
    MMDB_get_value(&result.entry, &entry_data,
                   "names", "en", NULL);
if (MMDB_SUCCESS != status) { ... }
if (entry_data.has_data) { ... }

إذا أردنا العثور على المدينة الأولى، فسيكون مسار البحث هو "cities"، "0". إذا لم تقدم مسار بحث، فستحصل على السجل المقابل للخريطة الجذرية. يجب أن ينتهي مسار البحث دائمًا بـ NULL، بغض النظر عن التابع الذي تستدعيه.

يأخذ التابع MMDB_get_value عددًا متغيرًا من المعاملات. جميع المعاملات بعد المؤشر إلى MMDB_entry_data_s هي مسار البحث. يجب أن تكون المعاملة الأخيرة NULL.

يستقبل التابع MMDB_vget_value قائمة va_list كمسار بحث. يجب أن تكون العنصر الأخير المسترد بواسطة va_arg() هو NULL.

أخيرًا، يستقبل التابع MMDB_aget_value مصفوفة من السلاسل النصية كمسار بحث. يجب أن يكون العنصر الأخير في المصفوفة NULL.

إذا كنت ترغب في الحصول على جميع البيانات من الإدخال دفعة واحدة، يمكنك استدعاء MMDB_get_entry_data_list().

لكل من التوابع الثلاثة، القيمة المعاد هي رمز حالة كما هو معرف أعلاه.

MMDB_get_entry_data_list()

int MMDB_get_entry_data_list(
    MMDB_entry_s *start,
    MMDB_entry_data_list_s **const entry_data_list);

يسمح لك هذا التابع بالحصول على جميع البيانات من هيكل معقد دفعة واحدة، بدلاً من البحث عن كل جزء عبر استدعاءات متكررة لـ MMDB_get_value().

MMDB_lookup_result_s result =
    MMDB_lookup_sockaddr(&mmdb, address->ai_addr, &mmdb_error);
MMDB_entry_data_list_s *entry_data_list, *first;
int status =
    MMDB_get_entry_data_list(&result.entry, &entry_data_list);
if (MMDB_SUCCESS != status) { ... }
// احفظ هذا حتى تتمكن من تحرير هذه البيانات لاحقًا
first = entry_data_list;
while (1) {
    MMDB_entry_data_list_s *next = entry_data_list = entry_data_list->next;
    if (NULL == next) {
        break;
    }
    switch (next->entry_data.type) {
        case MMDB_DATA_TYPE_MAP: { ... }
        case MMDB_DATA_TYPE_UTF8_STRING: { ... }
        ...
    }
}
MMDB_free_entry_data_list(first);

تقع على عاتقك مسؤولية تفسير هيكل entry_data_list. القائمة مرتبطة بعمق. سنستخدم هذا الهيكل كمثال:

{
    "names": {
        "en": "Germany",
        "de": "Deutschland"
    },
    "cities": [ "Berlin", "Frankfurt" ]
}

ستحتوي القائمة على العناصر التالية:

1.
MAP - الخريطة الجذرية
2.
UTF8_STRING - المفتاح "names"
3.
MAP - الخريطة للمفتاح "names"
4.
UTF8_STRING - المفتاح "en"
5.
UTF8_STRING - قيمة المفتاح "en"
6.
UTF8_STRING - المفتاح "de"
7.
UTF8_STRING - قيمة المفتاح "de"
8.
UTF8_STRING - المفتاح "cities"
9.
ARRAY - قيمة المفتاح "cities"
10.
UTF8_STRING - array[0]
11.
UTF8_STRING - array[1]

القيمة المعاد للتابع هي رمز حالة كما هو معرف أعلاه.

MMDB_free_entry_data_list()

void MMDB_free_entry_data_list(
    MMDB_entry_data_list_s *const entry_data_list);

تخصص التوابع MMDB_get_entry_data_list() و MMDB_get_metadata_as_entry_data_list() هيكل القائمة المرتبطة من ذاكرة الكومة. استدعِ هذا التابع لتحرير هيكل MMDB_entry_data_list_s.

MMDB_get_metadata_as_entry_data_list()

int MMDB_get_metadata_as_entry_data_list(
    MMDB_s *const mmdb,
    MMDB_entry_data_list_s **const entry_data_list);

يسمح لك هذا التابع بالحصول على البيانات الوصفية لقاعدة البيانات كقائمة من هياكل MMDB_entry_data_list_s. قد يكون هذا أكثر ملاءمة من استخدام هيكل البيانات الوصفية مباشرة.

    MMDB_entry_data_list_s *entry_data_list, *first;
    int status =
        MMDB_get_metadata_as_entry_data_list(&mmdb, &entry_data_list);
    if (MMDB_SUCCESS != status) { ... }
    first = entry_data_list;
    ... // فعل شيء بالبيانات
    MMDB_free_entry_data_list(first);

القيمة المعاد للتابع هي رمز حالة كما هو معرف أعلاه.

MMDB_dump_entry_data_list()

int MMDB_dump_entry_data_list(
    FILE *const stream,
    MMDB_entry_data_list_s *const entry_data_list,
    int indent);

يأخذ هذا التابع قائمة من هياكل MMDB_entry_data_list_s ويقوم بتسلسلها إلى التدفق المحدد. المعلمة indent هي مستوى المسافة البادئة الأولي للمخرجات المنتجة. يتم زيادتها للهياكل المتداخلة (الخرائط، المصفوفات، إلخ).

يجب أن يكون stream مقبض ملف (stdout، إلخ). إذا كانت منصتك توفر شيئًا مثل open_memstream() من GNU، يمكنك استخدامه لالتقاط الإخراج كسلسلة نصية.

المخرجات بتنسيق شبيه بـ JSON، ولكن القيم تمييز بنوع بياناتها (ما لم يتم عرض الخرائط والمصفوفات كـ "{}" و "[]" على التوالي).

قد يتغير التنسيق المحدد للمخرجات في الإصدارات المستقبلية.

اسم

MMDB_read_node()

ملخص

int MMDB_read_node(
    MMDB_s *const mmdb,
    uint32_t node_number,
    MMDB_search_node_s *const node);

وصف

تقوم هذه الدالة بقراءة عقدة محددة في شجرة البحث. الوسيط الثالث هو مرجع إلى بنية MMDB_search_node_s التي سيتم ملؤها بواسطة هذه الدالة.

هياكل البيانات

القيمة المرجعة هي رمز حالة. إذا قمت بتمرير node_number أكبر من عدد العقد في قاعدة البيانات، فسترجع هذه الدالة MMDB_INVALID_NODE_NUMBER_ERROR، وإلا فإنها سترجع MMDB_SUCCESS.

رموز الحالة

العقدة الأولى في شجرة البحث هي دائمًا العقدة 0. إذا كنت ترغب في المرور عبر شجرة البحث بأكملها، فستبدأ بقراءة العقدة 0 ثم تتبع الإدخالات التي تشكل هذه العقدة، بناءً على نوع كل إدخال. إذا كان النوع هو MMDB_RECORD_TYPE_SEARCH_NODE، فسوف يحتوي السجل على عدد صحيح للعقدة التالية للبحث عنها.

الدوال

العقدة الأولى في شجرة البحث هي دائمًا العقدة 0. إذا كنت ترغب في المرور عبر شجرة البحث بأكملها، فستبدأ بقراءة العقدة 0 ثم تتبع الإدخالات التي تشكل هذه العقدة، بناءً على نوع كل إدخال. إذا كان النوع هو MMDB_RECORD_TYPE_SEARCH_NODE، فسوف يحتوي السجل على عدد صحيح للعقدة التالية للبحث عنها.

مثال

#include <errno.h>
#include <maxminddb.h>
#include <stdlib.h>
#include <string.h>
int main(int argc, char **argv)
{
    char *filename = argv[1];
    char *ip_address = argv[2];
    MMDB_s mmdb;
    int status = MMDB_open(filename, MMDB_MODE_MMAP, &mmdb);
    if (MMDB_SUCCESS != status) {
        fprintf(stderr, "\n  Can't open %s - %s\n",
                filename, MMDB_strerror(status));
        if (MMDB_IO_ERROR == status) {
            fprintf(stderr, "    IO error: %s\n", strerror(errno));
        }
        exit(1);
    }
    int gai_error, mmdb_error;
    MMDB_lookup_result_s result =
        MMDB_lookup_string(&mmdb, ip_address, &gai_error, &mmdb_error);
    if (0 != gai_error) {
        fprintf(stderr,
                "\n  Error from getaddrinfo for %s - %s\n\n",
                ip_address, gai_strerror(gai_error));
        exit(2);
    }
    if (MMDB_SUCCESS != mmdb_error) {
        fprintf(stderr,
                "\n  Got an error from libmaxminddb: %s\n\n",
                MMDB_strerror(mmdb_error));
        exit(3);
    }
    MMDB_entry_data_list_s *entry_data_list = NULL;
    int exit_code = 0;
    if (result.found_entry) {
        int status = MMDB_get_entry_data_list(&result.entry,
                                              &entry_data_list);
        if (MMDB_SUCCESS != status) {
            fprintf(
                stderr,
                "Got an error looking up the entry data - %s\n",
                MMDB_strerror(status));
            exit_code = 4;
            goto end;
        }
        if (NULL != entry_data_list) {
            MMDB_dump_entry_data_list(stdout, entry_data_list, 2);
        }
    } else {
        fprintf(
            stderr,
            "\n  No entry for this IP address (%s) was found\n\n",
            ip_address);
        exit_code = 5;
    }
    end:
        MMDB_free_entry_data_list(entry_data_list);
        MMDB_close(&mmdb);
        exit(exit_code);
}
