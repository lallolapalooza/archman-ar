الوصف

توفر مكتبة libmaxminddb وظائف للعمل مع ملفات MaxMind DB. راجع https://maxmind.github.io/MaxMind-DB/ لمواصفات تنسيق MaxMind DB. يتم تمثيل قاعدة البيانات والنتائج بالكامل بواسطة هياكل بيانات مختلفة. يتم فتح قواعد البيانات عن طريق استدعاء MMDB_open(). يمكنك البحث عن عناوين IP كسلسلة نصية باستخدام MMDB_lookup_string() أو كمؤشر إلى هيكل sockaddr باستخدام MMDB_lookup_sockaddr().

إذا وجدت عملية البحث عنوان IP في قاعدة البيانات، فإنها تُرجع هيكل MMDB_lookup_result_s. إذا أشار هذا الهيكل إلى أن قاعدة البيانات تحتوي على بيانات لعنوان IP، فهناك عدد من الوظائف التي يمكن استخدامها لجلب هذه البيانات. تشمل هذه الوظائف MMDB_get_value() و MMDB_get_entry_data_list(). راجع توثيق الوظائف أدناه لمزيد من التفاصيل.

عند الانتهاء من مقبض قاعدة البيانات، يجب عليك استدعاء MMDB_close().

تبدأ جميع الوظائف والهياكل والماكروات المرئية للعامة بـ "MMDB_".

هياكل البيانات

يتم تعريف جميع هياكل البيانات التي تصدرها رأس maxminddb.h لهذه المكتبة في شكل typedef struct foo_s { ... } foo_s بحيث يمكنك الإشارة إليها بدون البادئة struct.

توفر هذه المكتبة هياكل البيانات التالية:

MMDB_s

هذا هو المقبض لملف MaxMind DB. نوثق فقط بعض حقول هذا الهيكل المخصصة للاستخدام العام. جميع الحقول الأخرى تخضع للتغيير ومخصصة للاستخدام الداخلي فقط.

typedef struct MMDB_s {
    uint32_t flags;
    const char *filename;
    ...
    MMDB_metadata_s metadata;
} MMDB_s;
  • uint32_t flags - العلامات التي تم فتح قاعدة البيانات بها. راجع توثيق MMDB_open() لمزيد من التفاصيل.
  • const char *filename - اسم الملف الذي تم فتحه، كما تم تمريره إلى MMDB_open().
  • MMDB_metadata_s metadata - البيانات الوصفية لقاعدة البيانات.

MMDB_metadata_s و MMDB_description_s

يمكن استرداد هذا الهيكل من هيكل MMDB_s. يحتوي على البيانات الوصفية التي تمت قراءتها من ملف قاعدة البيانات. لاحظ أنه قد تجد أنه من المريح أكثر الوصول إلى هذه البيانات الوصفية عن طريق استدعاء MMDB_get_metadata_as_entry_data_list() بدلاً من ذلك.

typedef struct MMDB_metadata_s {
    uint32_t node_count;
    uint16_t record_size;
    uint16_t ip_version;
    const char *database_type;
    struct {
        size_t count;
        const char **names;
    } languages;
    uint16_t binary_format_major_version;
    uint16_t binary_format_minor_version;
    uint64_t build_epoch;
    struct {
        size_t count;
        MMDB_description_s **descriptions;
    } description;
} MMDB_metadata_s;
typedef struct MMDB_description_s {
    const char *language;
    const char *description;
} MMDB_description_s;

يجب أن تكون هذه الهياكل واضحة بذاتها إلى حد كبير.

يجب أن تكون قيمة العضو ip_version دائمًا 4 أو 6. يجب أن تكون قيمة binary_format_major_version دائمًا 2.

لا يوجد شرط بأن تتضمن البيانات الوصفية لقاعدة البيانات لغات أو أوصافًا، لذا يمكن أن يكون count لهذه الأجزاء من البيانات الوصفية صفرًا. يجب ملء جميع حقول MMDB_metadata_s الأخرى.

MMDB_lookup_result_s

يتم إرجاع هذا الهيكل كنتيجة للبحث عن عنوان IP.

typedef struct MMDB_lookup_result_s {
    bool found_entry;
    MMDB_entry_s entry;
    uint16_t netmask;
} MMDB_lookup_result_s;

إذا كان العضو found_entry خاطئًا، فإن الأعضاء الآخرين في هذا الهيكل لا تحتوي على قيم ذات معنى. تحقق دائمًا من أن found_entry صحيح أولاً.

يُستخدم العضو entry للبحث عن البيانات المرتبطة بعنوان IP.

يخبرك العضو netmask بالشبكة الفرعية التي ينتمي إليها عنوان IP في قاعدة البيانات هذه. على سبيل المثال، إذا بحثت عن العنوان 1.1.1.1 في قاعدة بيانات IPv4 وكانت netmask المُرجعة هي 16، فإن العنوان هو جزء من الشبكة الفرعية 1.1.0.0/16.

إذا كانت قاعدة البيانات قاعدة بيانات IPv6، فإن قناع الشبكة المُرجع يكون دائمًا طول بادئة IPv6 (من 0-128)، حتى لو كانت قاعدة البيانات هذه تحتوي *أيضًا* على شبكات IPv4. إذا بحثت عن عنوان IPv4 وأردت تحويل قناع الشبكة إلى قيمة قناع شبكة IPv4، يمكنك ببساطة طرح 96 من القيمة.

MMDB_result_s

لست بحاجة حقًا إلى التعمق في هذا الهيكل. ستحصل عليه من هيكل MMDB_lookup_result_s وتمرره إلى وظائف مختلفة.

MMDB_entry_data_s

يُستخدم هذا الهيكل لإرجاع إدخال قسم بيانات واحد لعنوان IP. يمكن أن تشير هذه الإدخالات بدورها إلى إدخالات أخرى، كما هو الحال في أشياء مثل الخرائط والمصفوفات. لم يتم توثيق بعض أعضاء هذا الهيكل لأنها مخصصة للاستخدام الداخلي فقط.

typedef struct MMDB_entry_data_s {
    bool has_data;
    union {
        uint32_t pointer;
        const char *utf8_string;
        double double_value;
        const uint8_t *bytes;
        uint16_t uint16;
        uint32_t uint32;
        int32_t int32;
        uint64_t uint64;
        {mmdb_uint128_t or uint8_t[16]} uint128;
        bool boolean;
        float float_value;
    };
    ...
    uint32_t data_size;
    uint32_t type;
} MMDB_entry_data_s;

العضو has_data يكون صحيحًا إذا تم العثور على بيانات لبحث معين. راجع MMDB_get_value() لمزيد من التفاصيل. إذا كان هذا العضو خاطئًا، فإن أيًا من القيم الأخرى في الهيكل ليست ذات معنى.

يحدد الاتحاد في بداية الهيكل البيانات الفعلية. لتحديد أي عضو في الاتحاد

أعداد صحيحة 128 بت

يعتمد التعامل مع نوع البيانات uint128 على كيفية دعم نظامك للأعداد الصحيحة ذات 128 بت، إن كان يدعمها أم لا. مع GCC 4.4 و 4.5 يمكننا كتابة unsigned int __attribute__ ((__mode__ (TI))). مع إصدارات أحدث من GCC (4.6+) و clang (3.2+) يمكننا ببساطة كتابة "unsigned __int128".

للتغلب على هذه الاختلافات، تحدد هذه المكتبة النوع mmdb_uint128_t. يتم تعريف هذا النوع في الملف الرأسي maxminddb.h حتى تتمكن من استخدامه في الكود الخاص بك.

مع المترجمات القديمة، لا يمكننا استخدام عدد صحيح، لذلك نستخدم مصفوفة من 16 بايت من uint8_t. هذه هي البيانات الخام لقاعدة البيانات.

توفر هذه المكتبة ماكرو عام MMDB_UINT128_IS_BYTE_ARRAY. إذا كان هذا صحيحًا (1)، فسيتم إرجاع قيم uint128 كمصفوفة بايت، وإذا كان خاطئًا فسيتم إرجاعها كعدد صحيح mmdb_uint128_t.

وحدات ماكرو لأنواع البيانات

توفر هذه المكتبة ماكرو لكل نوع بيانات معرف بواسطة مخطط MaxMind DB.

  • MMDB_DATA_TYPE_UTF8_STRING
  • MMDB_DATA_TYPE_DOUBLE
  • MMDB_DATA_TYPE_BYTES
  • MMDB_DATA_TYPE_UINT16
  • MMDB_DATA_TYPE_UINT32
  • MMDB_DATA_TYPE_MAP
  • MMDB_DATA_TYPE_INT32
  • MMDB_DATA_TYPE_UINT64
  • MMDB_DATA_TYPE_UINT128
  • MMDB_DATA_TYPE_ARRAY
  • MMDB_DATA_TYPE_BOOLEAN
  • MMDB_DATA_TYPE_FLOAT

هناك أيضًا بعض الأنواع المستخدمة داخليًا فقط:

  • MMDB_DATA_TYPE_EXTENDED
  • MMDB_DATA_TYPE_POINTER
  • MMDB_DATA_TYPE_CONTAINER
  • MMDB_DATA_TYPE_END_MARKER

إذا رأيت أحد هذه الأنواع في البيانات المرجعة، فهذا يعني أن شيئًا ما قد حدث بشكل خاطئ للغاية. قاعدة البيانات تالفة أو تم إنشاؤها بشكل غير صحيح أو هناك خطأ في كود libmaxminddb.

قيم المؤشرات و MMDB_close()

الأعضاء utf8_string و bytes و (ربما) العضو uint128 في هذا الهيكل هي جميعها مؤشرات مباشرة إلى قسم بيانات قاعدة البيانات. يمكن أن يكون هذا كتلة ذاكرة تم تخصيصها باستخدام calloc أو mmap. في كلتا الحالتين، ستصبح هذه المؤشرات غير صالحة بعد استدعاء MMDB_close().

إذا كنت بحاجة إلى الإشارة إلى هذه البيانات بعد ذلك الوقت، فيجب عليك نسخ البيانات باستخدام وظيفة مناسبة (مثل strdup أو memcpy، إلخ).

MMDB_entry_data_list_s

يغلف هذا الهيكل قائمة مرتبطة من هياكل MMDB_entry_data_s.

typedef struct MMDB_entry_data_list_s {
    MMDB_entry_data_s entry_data;
    struct MMDB_entry_data_list_s *next;
} MMDB_entry_data_list_s;

يتيح لك هذا الهيكل رؤية بيانات الخريطة أو المصفوفة الكاملة عن طريق التكرار عبر القائمة المرتبطة.

MMDB_search_node_s

يغلف هذا الهيكل سجلين لعقدة بحث. هذا مفيد حقًا فقط إذا كنت ترغب في كتابة كود يتكرر عبر شجرة البحث بأكملها بدلاً من البحث عن عنوان IP محدد.

typedef struct MMDB_search_node_s {
    uint64_t left_record;
    uint64_t right_record;
    uint8_t left_record_type;
    uint8_t right_record_type;
    MMDB_entry_s left_record_entry;
    MMDB_entry_s right_record_entry;
} MMDB_search_node_s;

يمكن أن تأخذ أنواع السجلات أحد القيم التالية:

  • MMDB_RECORD_TYPE_SEARCH_NODE - يشير السجل إلى عقدة بحث تالية.
  • MMDB_RECORD_TYPE_EMPTY - السجل هو عنصر نائب يشير إلى عدم وجود بيانات لعنوان IP. يجب أن ينتهي البحث هنا.
  • MMDB_RECORD_TYPE_DATA - السجل مخصص للبيانات في قسم بيانات قاعدة البيانات. استخدم إدخال السجل عند البحث عن بيانات السجل.
  • MMDB_RECORD_TYPE_INVALID - السجل غير صالح. إما تم العثور على عقدة غير صالحة أو أن قاعدة البيانات تالفة.

السجل MMDB_entry_s يكون صالحًا فقط إذا كان النوع هو MMDB_RECORD_TYPE_DATA. سيؤدي استخدام سجل لأنواع سجلات أخرى إلى حدوث خطأ أو بيانات غير صالحة.

رموز الحالة

تعيد هذه المكتبة (أو تملأ) رموز حالة للعديد من الوظائف. رموز الحالة هذه هي:

  • MMDB_SUCCESS - تم كل العمل بنجاح
  • MMDB_FILE_OPEN_ERROR - حدث خطأ أثناء محاولة فتح ملف MaxMind DB.
  • MMDB_IO_ERROR - فشلت عملية إدخال/إخراج. تحقق من errno لمزيد من التفاصيل.
  • MMDB_CORRUPT_SEARCH_TREE_ERROR - عند البحث عن عنوان IP في شجرة البحث، تم الحصول على نتيجة مستحيلة. قاعدة البيانات تالفة أو تم إنشاؤها بشكل غير صحيح أو هناك خطأ في كود libmaxminddb.
  • MMDB_INVALID_METADATA_ERROR - هناك خطأ في شيء ما في قاعدة البيانات. يشمل ذلك مفاتيح البيانات الوصفية المفقودة والقيم المستحيلة (مثل ip_version يساوي 7).
  • MMDB_UNKNOWN_DATABASE_FORMAT_ERROR - تشير البيانات الوصفية لقاعدة البيانات إلى أن الإصدار الرئيسي يختلف عن 2. هذه المكتبة يمكنها فقط التعامل مع الإصدار 2.
  • MMDB_OUT_OF_MEMORY_ERROR - فشل استدعاء لعملية تخصيص ذاكرة (malloc، إلخ).
  • MMDB_INVALID_DATA_ERROR - يحتوي إدخال في قسم البيانات على بيانات غير صالحة. على سبيل المثال، يدعي حقل uint16 أنه أطول من 2 بايت. من المحتمل أن تكون قاعدة البيانات تالفة أو تم إنشاؤها بشكل غير صحيح.
  • MMDB_INVALID_LOOKUP_PATH_ERROR - يحتوي مسار البحث الذي تم تمريره إلى MMDB_get_value أو MMDB_vget_value أو MMDB_aget_value على إزاحة مصفوفة أكبر من LONG_MAX أو أصغر من LONG_MIN.
  • MMDB_LOOKUP_PATH_DOES_NOT_MATCH_DATA_ERROR - مسار البحث الذي تم تمريره إلى MMDB_get_value أو MMDB_vget_value أو MMDB_aget_value لا يتطابق مع بنية البيانات للسجل. هناك عدة أسباب لحدوث ذلك. قد يتضمن مسار البحث مفتاحًا غير موجود في الخريطة. قد يتضمن مسار البحث فهرس مصفوفة أكبر من مصفوفة أو أصغر من الحد الأدنى للإزاحة من نهاية مصفوفة. قد يحدث أيضًا عندما يتوقع المسار العثور على خريطة أو مصفوفة غير موجودة.

يجب التعامل مع جميع رموز الحالة كقيم من النوع int.

MMDB_strerror()

const char *MMDB_strerror(int error_code)

تأخذ هذه الوظيفة رمز حالة وتعيد سلسلة نصية باللغة الإنجليزية تشرح الحالة.

الوظائف

توفر هذه المكتبة الوظائف المصدرة التالية:

MMDB_open()

int MMDB_open(
    const char *const filename,
    uint32_t flags,
    MMDB_s *const mmdb);

تفتح هذه الوظيفة مرجعًا لملف MaxMind DB. القيمة المرجعة هي رمز حالة كما هو محدد أعلاه. تحقق دائمًا من القيمة المرجعة لهذا الاستدعاء.

MMDB_s mmdb;
int status =
    MMDB_open("/path/to/file.mmdb", MMDB_MODE_MMAP, &mmdb);
if (MMDB_SUCCESS != status) { ... }
...
MMDB_close(&mmdb);

يجب ترميز filename كـ UTF-8 على Windows.

قد يكون الهيكل MMDB_s الذي تمرره على المكدس أو مخصصًا في الكومة. ومع ذلك، إذا كان الفتح ناجحًا، فسوف يحتوي على بيانات مخصصة في الكومة، لذا يجب عليك إغلاقه باستخدام MMDB_close(). إذا لم تكن الحالة المرجعة MMDB_SUCCESS، تضمن هذه المكتبة تحرير جميع الذاكرة المخصصة قبل العودة.

الأعلام المقدمة حاليًا هي:

•
MMDB_MODE_MMAP - يفتح الملف باستخدام mmap().

يمكن أن يؤدي تمرير قيم أخرى للأعلام إلى نتائج غير متوقعة. في المستقبل، قد نضيف أعلامًا أخرى يمكنك دمجها مع الوضع، بالإضافة إلى أوضاع أخرى.

يمكنك أيضًا تمرير 0 كقيمة للأعلام، وفي هذه الحالة سيتم فتح الملف باستخدام الأعلام الافتراضية. ومع ذلك، قد تتغير هذه القيم الافتراضية في الإصدارات المستقبلية. القيمة الافتراضية الحالية هي MMDB_MODE_MMAP.

MMDB_close()

void MMDB_close(MMDB_s *const mmdb);

تقوم هذه المكتبة بتحرير أي ذاكرة مخصصة أو ممسوحة ضوئيًا توجد في الهيكل MMDB_s. *لا تحرر ذاكرة النظام.

MMDB_lookup_string()

MMDB_lookup_result_s MMDB_lookup_string(
    MMDB_s *const mmdb,
    const char *const ipstr,
    int *const gai_error,
    int *const mmdb_error);

تبحث هذه الدالة في عنوان IP يتم تمريره كسلسلة نصية تنتهي بصفر. داخليًا، تستدعي getaddrinfo() لحل العنوان إلى شكل ثنائي. ثم تستدعي MMDB_lookup_sockaddr() للبحث عن العنوان في قاعدة البيانات. إذا كنت قد قمت بالفعل بحل عنوان، يمكنك استدعاء MMDB_lookup_sockaddr() مباشرة، بدلاً من حل العنوان مرتين.

int gai_error, mmdb_error;
MMDB_lookup_result_s result =
    MMDB_lookup_string(&mmdb, "1.2.3.4", &gai_error, &mmdb_error);
if (0 != gai_error) { ... }
if (MMDB_SUCCESS != mmdb_error) { ... }
if (result.found_entry) { ... }

تعيد هذه الدالة دائمًا بنية MMDB_lookup_result_s، ولكن يجب عليك أيضًا التحقق من معاملي gai_error و mmdb_error. إذا أشارت أي من هذه إلى خطأ، فإن البنية المعادة لا معنى لها.

إذا لم يحدث خطأ، فما زلت بحاجة إلى التأكد من أن عضو found_entry في النتيجة المعادة صحيح. إذا لم يكن كذلك، فهذا يعني أن عنوان IP ليس له إدخال في قاعدة البيانات.

ستعمل هذه الدالة مع عناوين IPv4 حتى عندما تحتوي قاعدة البيانات على بيانات لكل من عناوين IPv4 و IPv6. سيتم البحث عن عنوان IPv4 على أنه '::xxx.xxx.xxx.xxx' بدلاً من إعادة تعيينه إلى كتلة ::ffff:xxx.xxx.xxx.xxx المخصصة لعناوين IPv6 الممثلة لـ IPv4.

إذا مررت عنوان IPv6 إلى قاعدة بيانات تحتوي فقط على بيانات IPv4، فسيكون عضو found_entry خاطئًا، ولكن حالة mmdb_error ستظل MMDB_SUCCESS.

MMDB_lookup_result_s MMDB_lookup_sockaddr(
    MMDB_s *const mmdb,
    const struct sockaddr *const sockaddr,
    int *const mmdb_error);

تبحث هذه الدالة في عنوان IP الذي تم حله بالفعل بواسطة getaddrinfo().

بخلاف عدم استدعاء getaddrinfo() بنفسها، هذه الدالة مطابقة تمامًا للدالة MMDB_lookup_string().

int mmdb_error;
MMDB_lookup_result_s result =
    MMDB_lookup_sockaddr(&mmdb, address->ai_addr, &mmdb_error);
if (MMDB_SUCCESS != mmdb_error) { ... }
if (result.found_entry) { ... }

هناك ثلاث دوال للبحث عن البيانات المرتبطة بعنوان IP.

int MMDB_get_value(

تسمح الدوال الثلاث بثلاثة أنماط استدعاء مختلفة قليلاً، لكنها جميعًا تفعل الشيء نفسه.

المعلمة الأولى هي قيمة MMDB_entry_s. في معظم الحالات، ستأتي من قيمة MMDB_lookup_result_s التي أعادتها MMDB_lookup_string() أو MMDB_lookup_sockaddr().

المعلمة الثانية هي مرجع إلى بنية MMDB_entry_data_s. سيتم ملؤها بالبيانات التي يتم البحث عنها، إن وجدت. إذا لم يتم العثور على شيء، فسيتم تعيين عضو has_data لهذه البنية على خطأ. إذا كان has_data صحيحًا، يمكنك النظر إلى عضو data_type.

المعلمة النهائية هي مسار بحث. يتكون المسار من مجموعة من السلاسل النصية التي تمثل إما مفاتيح الخريطة (مثل "city") أو مؤشرات المصفوفة (مثل "0" أو "1" أو "-1") لاستخدامها في البحث.

سيتم التعامل مع مؤشرات المصفوفة السالبة كإزاحة من نهاية المصفوفة. على سبيل المثال، يشير "-1" إلى العنصر الأخير في المصفوفة.

يسمح لك مسار البحث بالتنقل في بنية بيانات معقدة. على سبيل المثال، بالنظر إلى هذه البيانات:

{
    "names": {
        "en": "Germany",
        "de": "Deutschland"
    },
    "cities": [ "Berlin", "Frankfurt" ]
}

يمكننا البحث عن الاسم الإنجليزي بهذا الرمز:

MMDB_lookup_result_s result =
    MMDB_lookup_sockaddr(&mmdb, address->ai_addr, &mmdb_error);
MMDB_entry_data_s entry_data;
int status =
    MMDB_get_value(&result.entry, &entry_data,
                   "names", "en", NULL);
if (MMDB_SUCCESS != status) { ... }
if (entry_data.has_data) { ... }

إذا أردنا العثور على المدينة الأولى، فسيكون مسار البحث "cities"، "0". إذا لم تقدم مسار بحث على الإطلاق، فستحصل على الإدخال الذي يتوافق مع الخريطة العلوية. يجب أن ينتهي مسار البحث دائمًا بـ NULL، بغض النظر عن الدالة التي تستدعيها.

تأخذ الدالة MMDB_get_value عددًا متغيرًا من المعاملات. جميع المعاملات بعد مؤشر بنية MMDB_entry_data_s * هي مسار البحث. يجب أن تكون المعاملة الأخيرة NULL.

تقبل الدالة MMDB_vget_value قائمة va_list كمسار بحث. يجب أن تكون العنصر الأخير المسترد بواسطة va_arg() هو NULL.

أخيرًا، تقبل الدالة MMDB_aget_value مصفوفة من السلاسل النصية كمسار بحث. يجب أن يكون العضو الأخير في هذه المصفوفة NULL.

إذا كنت تريد الحصول على جميع بيانات الإدخال مرة واحدة، يمكنك استدعاء MMDB_get_entry_data_list() بدلاً من ذلك.

لكل دالة من الدوال الثلاث، القيمة المعادة هي رمز حالة كما هو محدد أعلاه.

maxminddb - C library for the MaxMind DB file format

NAME

maxminddb - C library for the MaxMind DB file format

SYNOPSIS

#include <maxminddb.h>

Data Types

typedef struct MMDB_s MMDB_s;
typedef struct MMDB_entry_data_s MMDB_entry_data_s;
typedef struct MMDB_entry_data_list_s MMDB_entry_data_list_s;
typedef struct MMDB_lookup_result_s MMDB_lookup_result_s;
typedef struct MMDB_search_node_s MMDB_search_node_s;

#define MMDB_SUCCESS 0
#define MMDB_FILE_NOT_FOUND_ERROR -1
#define MMDB_INVALID_FILE_FORMAT_ERROR -2
#define MMDB_IO_ERROR -3
#define MMDB_OUT_OF_MEMORY_ERROR -4
#define MMDB_SUCCESS_UNFINDABLE_ADDRESS -5
#define MMDB_NULL_POINTER_ERROR -6
#define MMDB_INVALID_DATA_ERROR -7
#define MMDB_INVALID_NODE_NUMBER_ERROR -8
#define MMDB_INVALID_LOOKUP_PATH_ERROR -9
#define MMDB_LOOKUP_PATH_DOES_NOT_EXIST_ERROR -10

#define MMDB_MODE_MMAP 0
#define MMDB_MODE_FILE 1

int MMDB_open(const char *filename, int mode, MMDB_s *result);

MMDB_lookup_result_s MMDB_lookup_string(MMDB_s *mmdb, const char *ip_address,

                               MMDB_entry_data_list_s *list, int indent);
int MMDB_read_node(MMDB_s *const mmdb, uint32_t node_number,
                   MMDB_search_node_s *const node);
const char *MMDB_lib_version(void);

DESCRIPTION

The libmaxminddb library implements a reader for the MaxMind DB file format. This is a binary format for storing geographical location data as well as other data.

This library is not thread-safe by default. It is thread-safe if compiled with a thread-safe implementation of malloc and free.

The library can be built in three different modes:

MMDB_MODE_MMAP
Memory-map the database file. This is the default and generally the most performant mode.
MMDB_MODE_FILE
Read the database file using standard file I/O operations. This may be useful on systems that do not support mmap or for security reasons.
MMDB_MODE_SHARED_MEM
Read the database file into a shared memory segment. This allows multiple processes to share the same memory map of the database.

When calling MMDB_open, use the mode parameter to select the desired mode. If you do not specify a mode, MMDB_MODE_MMAP will be used.

DATA STRUCTURES

MMDB_s
This structure represents an open MaxMind DB database. It should not be initialized directly by the user.
MMDB_entry_data_s
This structure represents a single data entry within the database. It contains a pointer to the data, the size of the data, and the data type.
MMDB_entry_data_list_s
This structure represents a list of data entries. It is typically used to return the result of a lookup that has multiple values associated with it.
MMDB_lookup_result_s
This structure contains the result of a lookup operation. It includes a boolean flag indicating whether an entry was found, and an MMDB_entry_data_s structure representing the found entry.
MMDB_search_node_s
This structure represents a node in the search tree. It contains the data associated with the node and pointers to the left and right child nodes.

STATUS CODES

The functions in this library return integer status codes. A return value of MMDB_SUCCESS (0) indicates success. Negative values indicate errors. The most common error codes are defined in maxminddb.h. These include:

MMDB_SUCCESS
Operation completed successfully.
MMDB_FILE_NOT_FOUND_ERROR
The specified database file could not be found.
MMDB_INVALID_FILE_FORMAT_ERROR
The database file is not in a valid MaxMind DB format.
MMDB_IO_ERROR
An I/O error occurred while reading the database file.
MMDB_OUT_OF_MEMORY_ERROR
The library could not allocate sufficient memory.
MMDB_SUCCESS_UNFINDABLE_ADDRESS
The IP address was not found in the database. Note that this is considered a successful lookup, but the address was not found.
MMDB_NULL_POINTER_ERROR
A required pointer argument was NULL.
MMDB_INVALID_DATA_ERROR
The data within the database is invalid.
MMDB_INVALID_NODE_NUMBER_ERROR
An invalid node number was specified when reading from the search tree.
MMDB_INVALID_LOOKUP_PATH_ERROR
The provided lookup path is invalid.
MMDB_LOOKUP_PATH_DOES_NOT_EXIST_ERROR
The specified lookup path does not exist in the database.
MMDB_CLOSING_IS_FORBIDDEN_ERROR
An attempt was made to close a database opened in shared memory mode without proper handling.

The MMDB_strerror function can be used to convert these status codes into human-readable error messages.

FUNCTIONS

MMDB_open()

int MMDB_open(const char *filename, int mode, MMDB_s *result);

Opens a MaxMind DB database file. The filename argument is the path to the database file. The mode argument specifies how the file should be opened (e.g., MMDB_MODE_MMAP, MMDB_MODE_FILE, MMDB_MODE_SHARED_MEM). The result argument is a pointer to an MMDB_s structure that will be populated with the database information.

Returns MMDB_SUCCESS on success, or an error code otherwise.

MMDB_close()

void MMDB_close(MMDB_s *mmdb);

Closes an opened MaxMind DB database and frees any associated resources. The mmdb argument is a pointer to the MMDB_s structure representing the database.

MMDB_lookup_string()

MMDB_lookup_result_s MMDB_lookup_string(MMDB_s *mmdb, const char *ip_address,
                                       int *gai_error, int *mmdb_error);

Looks up an IP address in the database. The ip_address argument is a string representation of the IP address to look up. The gai_error and mmdb_error arguments are pointers to integers that will store any errors returned by getaddrinfo and the MaxMind DB library, respectively.

Returns an MMDB_lookup_result_s structure containing the lookup result. If the IP address is not found, the found_entry field of the returned structure will be false, and mmdb_error will be set to MMDB_SUCCESS_UNFINDABLE_ADDRESS.

MMDB_lookup_sockaddr()

MMDB_lookup_result_s MMDB_lookup_sockaddr(MMDB_s *mmdb, const struct sockaddr *sa,
                                         int *mmdb_error);

Looks up an IP address represented by a struct sockaddr in the database. The mmdb_error argument is a pointer to an integer that will store any errors returned by the MaxMind DB library.

Returns an MMDB_lookup_result_s structure containing the lookup result. If the IP address is not found, the found_entry field of the returned structure will be false, and mmdb_error will be set to MMDB_SUCCESS_UNFINDABLE_ADDRESS.

MMDB_strerror()

const char *MMDB_strerror(int status);

Returns a human-readable string representation of the given status code.

MMDB_get_entry_data_list()

int MMDB_get_entry_data_list(const MMDB_entry_data_s *entry,
                             MMDB_entry_data_list_s **list);

Retrieves a list of data entries associated with a given entry. This is useful when a lookup may return multiple values (e.g., in GeoIP2 databases).

The entry argument is a pointer to the MMDB_entry_data_s structure obtained from a lookup. The list argument is a pointer to a pointer where the newly allocated MMDB_entry_data_list_s will be stored.

Returns MMDB_SUCCESS on success, or an error code otherwise.

MMDB_free_entry_data_list()

void MMDB_free_entry_data_list(MMDB_entry_data_list_s *list);

Frees the memory associated with an MMDB_entry_data_list_s structure. This function should be called after you are finished with the list obtained from MMDB_get_entry_data_list.

MMDB_dump_entry_data_list()

void MMDB_dump_entry_data_list(FILE *file, MMDB_entry_data_list_s *list,
                               int indent);

Dumps the contents of an MMDB_entry_data_list_s to a specified file stream. The indent argument controls the indentation level for pretty-printing.

This function is primarily intended for debugging and showing data to users. The exact formatting may change between library

The return value of the function is a status code as defined above.

MMDB_read_node()

int MMDB_read_node(MMDB_s *const mmdb, uint32_t node_number,
                   MMDB_search_node_s *const node);

This reads a specific node in the search tree. The third argument is a reference to an MMDB_search_node_s structure that will be populated by this function.

The return value is a status code. If you pass a node_number that is greater than the number of nodes in the database, this function will return MMDB_INVALID_NODE_NUMBER_ERROR, otherwise it will return MMDB_SUCCESS.

The first node in the search tree is always node 0. If you wanted to iterate over the whole search tree, you would start by reading node 0 and then following the the records that make up this node, based on the type of each record. If the type is MMDB_RECORD_TYPE_SEARCH_NODE then the record contains an integer for the next node to look up.

MMDB_lib_version()

const char *MMDB_lib_version(void);

This function returns the library version as a string, something like "2.0.0".

EXAMPLE

#include <errno.h>
#include <maxminddb.h>
#include <stdlib.h>
#include <string.h>
int main(int argc, char **argv)
{
    char *filename = argv[1];
    char *ip_address = argv[2];
    MMDB_s mmdb;
    int status = MMDB_open(filename, MMDB_MODE_MMAP, &mmdb);
    if (MMDB_SUCCESS != status) {
        fprintf(stderr, "\n  Can't open %s - %s\n",
                filename, MMDB_strerror(status));
        if (MMDB_IO_ERROR == status) {
            fprintf(stderr, "    IO error: %s\n", strerror(errno));
        }
        exit(1);
    }
    int gai_error, mmdb_error;
    MMDB_lookup_result_s result =
        MMDB_lookup_string(&mmdb, ip_address, &gai_error, &mmdb_error);
    if (0 != gai_error) {
        fprintf(stderr,
                "\n  Error from getaddrinfo for %s - %s\n\n",
                ip_address, gai_strerror(gai_error));
        exit(2);
    }
    if (MMDB_SUCCESS != mmdb_error) {
        fprintf(stderr,
                "\n  Got an error from libmaxminddb: %s\n\n",
                MMDB_strerror(mmdb_error));
        exit(3);
    }
    MMDB_entry_data_list_s *entry_data_list = NULL;
    int exit_code = 0;
    if (result.found_entry) {
        int status = MMDB_get_entry_data_list(&result.entry,
                                              &entry_data_list);
        if (MMDB_SUCCESS != status) {
            fprintf(
                stderr,
                "Got an error looking up the entry data - %s\n",
                MMDB_strerror(status));
            exit_code = 4;
            goto end;
        }
        if (NULL != entry_data_list) {
            MMDB_dump_entry_data_list(stdout, entry_data_list, 2);
        }
    } else {
        fprintf(
            stderr,
            "\n  No entry for this IP address (%s) was found\n\n",
            ip_address);
        exit_code = 5;
    }
    end:
        MMDB_free_entry_data_list(entry_data_list);
        MMDB_close(&mmdb);
        exit(exit_code);
}
