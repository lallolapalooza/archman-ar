الوصف

توفر مكتبة libmaxminddb وظائف للتعامل مع ملفات MaxMind DB. انظر مواصفات تنسيق MaxMind DB على https://maxmind.github.io/MaxMind-DB/. يتم تمثيل قواعد البيانات والنتائج بواسطة هياكل مختلفة. يتم فتح قواعد البيانات عن طريق استدعاء MMDB_open(). يمكنك البحث عن عناوين IP كسلاسل نصية باستخدام MMDB_lookup_string() أو كـ مؤشرات لهياكل sockaddr باستخدام MMDB_lookup_sockaddr().

إذا وجد البحث عنوان IP في قاعدة البيانات، فإنه يعيد هيكل MMDB_lookup_result_s. إذا أشار هذا الهيكل إلى أن قاعدة البيانات تحتوي على بيانات لعنوان IP، فهناك العديد من الوظائف التي يمكنك استخدامها للحصول على تلك البيانات. وتشمل هذه MMDB_get_value() و MMDB_get_entry_data_list(). انظر وثائق الوظائف أدناه لمزيد من التفاصيل.

عند الانتهاء من التعامل مع قاعدة البيانات، قم باستدعاء MMDB_close().

جميع الوظائف والهياكل والماكروات العامة تبدأ بـ MMDB_.

هياكل البيانات

جميع الهياكل التي تصدرها هذه المكتبة maxminddb.h يتم تعريفها كـ typedef struct foo_s { ... } foo_s، لذا يمكنك الإشارة إليها دون البادئة struct.

توفر هذه المكتبة الهياكل التالية:

MMDB_s

هذه هي مقبض ملف MaxMind DB. نحن نوثق فقط بعض حقول هذا الهيكل المخصصة للاستخدام العام. تخضع جميع الحقول الأخرى للتغيير وهي مخصصة للاستخدام الداخلي فقط.

typedef struct MMDB_s {
    uint32_t flags;
    const char *filename;
    ...
    MMDB_metadata_s metadata;
} MMDB_s;
  • uint32_t flags - العلامات التي تم بها فتح قاعدة البيانات هذه. انظر وثائق MMDB_open() لمزيد من التفاصيل.
  • const char *filename - اسم الملف الذي تم فتحه، كما تم تمريره إلى MMDB_open().
  • MMDB_metadata_s metadata - البيانات الوصفية لقاعدة البيانات.

MMDB_metadata_s و MMDB_description_s

يمكن الحصول على هذا الهيكل من هيكل MMDB_s. يحتوي على البيانات الوصفية التي تم قراءتها من ملف قاعدة بيانات. لاحظ أنه قد يكون من الأنسب الوصول إلى هذه البيانات الوصفية عن طريق استدعاء MMDB_get_metadata_as_entry_data_list().

typedef struct MMDB_metadata_s {
    uint32_t node_count;
    uint16_t record_size;
    uint16_t ip_version;
    const char *database_type;
    struct {
        size_t count;
        const char **names;
    } languages;
    uint16_t binary_format_major_version;
    uint16_t binary_format_minor_version;
    uint64_t build_epoch;
    struct {
        size_t count;
        MMDB_description_s **descriptions;
    } description;
} MMDB_metadata_s;
typedef struct MMDB_description_s {
    const char *language;
    const char *description;
} MMDB_description_s;

يجب أن تكون هذه الهياكل واضحة بذاتها في الغالب.

يجب أن يكون عضو ip_version دائمًا 4 أو 6. يجب أن يكون عضو binary_format_major_version دائمًا 2.

لا توجد متطلبات محددة على البيانات الوصفية لقاعدة البيانات التي تشمل اللغات أو الأوصاف، لذلك قد يكون عدد count لهذه العناصر الوصفية صفرًا. يجب ملء جميع الأعضاء الآخرين في MMDB_metadata_s.

MMDB_lookup_result_s

يتم إرجاع هذا الهيكل كنتيجة للبحث عن عنوان IP.

typedef struct MMDB_lookup_result_s {
    bool found_entry;
    MMDB_entry_s entry;
    uint16_t netmask;
} MMDB_lookup_result_s;

إذا كان العضو found_entry خاطئًا، فإن الأعضاء الأخرى في هذا الهيكل لا تحتوي على قيم ذات معنى. تحقق دائمًا من أن found_entry صحيح أولاً.

يستخدم العضو entry للبحث عن البيانات المرتبطة بعنوان IP.

يشير العضو netmask إلى الشبكة الفرعية التي ينتمي إليها عنوان IP في قاعدة البيانات هذه. على سبيل المثال، إذا قمت بالبحث عن العنوان 1.1.1.1 في قاعدة بيانات IPv4 وكانت القيمة المرجعة 16، فإن العنوان ينتمي إلى الشبكة الفرعية 1.1.0.0/16.

إذا كانت قاعدة البيانات قاعدة بيانات IPv6، فإن القيمة المرجعة لـ netmask هي دائمًا طول بادئة IPv6 (من 0 إلى 128)، حتى لو كانت قاعدة البيانات هذه تحتوي أيضًا على شبكات IPv4. إذا كنت تبحث عن عنوان IPv4 وتريد تحويل قناع الشبكة إلى قيمة شبكة فرعية IPv4، فما عليك سوى طرح 96 من القيمة.

MMDB_result_s

لا تحتاج إلى البحث في هذا الهيكل. ستحصل على هذا الهيكل من هيكل MMDB_lookup_result_s وتمرره إلى وظائف مختلفة.

MMDB_entry_data_s

يستخدم هذا الهيكل لإرجاع كتلة بيانات واحدة لعنوان IP. يمكن أن تشير هذه الكتل إلى كتل أخرى، كما هو الحال مع الخرائط والمصفوفات. بعض أعضاء هذا الهيكل غير موثقة، لأنها فريدة للاستخدام الداخلي.

typedef struct MMDB_entry_data_s {
    bool has_data;
    union {
        uint32_t pointer;
        const char *utf8_string;
        double double_value;
        const uint8_t *bytes;
        uint16_t uint16;
        uint32_t uint32;
        int32_t int32;
        uint64_t uint64;
        {mmdb_uint128_t or uint8_t[16]} uint128;
        bool boolean;
        float float_value;
    };
    ...
    uint32_t data_size;
    uint32_t type;
} MMDB_entry_data_s;

العضو has_data صحيح إذا تم العثور على بيانات لاستعلام معين. انظر MMDB_get_value() لمزيد من التفاصيل. إذا كان هذا العضو خاطئًا، فإن أيًا من القيم المتبقية في الهيكل ليست ذات معنى.
