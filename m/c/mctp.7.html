mctp(7) معلومات متفرقة يدوية mctp(7)

الاسم

mctp - بروتوكول نقل المكونات الإدارية

الملخص

#include <linux/mctp.h>  /* عناوين MCTP وأنواع الثوابت البروتوكول */
#include <sys/socket.h>
mctp_socket = socket(AF_MCTP, SOCK_DGRAM, 0);

الوصف

تنفذ لينكس بروتوكول نقل المكونات الإدارية (MCTP)، المحدد بمواصفات DMTF DSP0236، حاليًا الإصدار 1. هذا بروتوكول غير متصل، يستخدم عادة بين الأجهزة داخل نظام خادم. لا يتم ضمان موثوقية الرسائل وترتيبها، ولكن حدود الرسائل محفوظة.

واجهة برمجة التطبيقات (API) لمراسلة MCTP تستخدم واجهة مقابس قياسية، باستخدام فئات استدعاءات النظام sendto(2) و recvfrom(2) لنقل الرسائل. يمكن تجزئة الرسائل إلى حزم قبل الإرسال، وإعادة تجميعها عند نقطة النهاية البعيدة. هذا التجزئة وإعادة التجميع شفافان لمساحة المستخدم.

تنسيق العنوان

عناوين MCTP (المشار إليها أيضًا باسم EIDs، أو معرفات نقطة النهاية) هي قيم من بايت واحد، من النوع mctp_eid_t. تحدد الحزم بين نقطة نهاية محلية وبعيدة بواسطة EIDs المصدر والوجهة، بالإضافة إلى قيمة علامة ثلاثية البت.

يتم تمرير بيانات العنونة في استدعاءات نظام المقابس من خلال هيكل sockaddr_mctp، المعرف على النحو التالي:


typedef uint8_t        mctp_eid_t;
struct mctp_addr {
    mctp_eid_t         s_addr;
};
struct sockaddr_mctp {
    unsigned short     smctp_family;  /* = AF_MCTP */
    uint16_t           __smctp_pad0;
    int                smctp_network; /* معرف شبكة محلي */
    struct mctp_addr   smctp_addr;    /* EID */
    uint8_t            smctp_type;    /* بايت نوع الرسالة */
    uint8_t            smctp_tag;     /* قيمة العلامة والمالك */
    uint8_t            __smctp_pad1;
};

إرسال الرسائل

يمكن إرسال الرسائل باستخدام استدعاءات النظام sendto(2) و sendmsg(2)، عن طريق توفير هيكل sockaddr_mctp يصف معلمات العنونة.


ssize_t               n;
const char            *buf;
struct sockaddr_mctp  addr;
/* يجب أن تكون الحقول غير المستخدمة صفرًا */
memset(&addr, 0, sizeof(addr));
/* تحديد وجهة الرسالة */
addr.smctp_family = AF_MCTP;
addr.smctp_network = 0;
addr.smctp_addr.s_addr = 8;  /* EID بعيد */
addr.smctp_tag = MCTP_TAG_OWNER;
addr.smctp_type = MYPROGRAM_MCTP_TYPE_ECHO;  /* مثال على النوع */
buf = "hello, world!"
n = sendto(sd, buf, 13, 0,
           (struct sockaddr *) &addr, sizeof(addr));

هنا، يمتلك المرسل علامة الرسالة؛ لذا يتم استخدام MCTP_TAG_OWNER كبيانات العلامة. سيقوم النواة بتخصيص قيمة علامة محددة لهذه الرسالة. إذا لم تكن هناك علامة متاحة، فسوف تقوم sendto(2) بإرجاع خطأ، مع ضبط errno على EBUSY. تظل هذه العلامة المخصصة مرتبطة بالمقبس، بحيث يتم استقبال أي ردود على الرسالة المرسلة بواسطة نفس المقبس.

يتطلب إرسال رسالة MCTP القدرة CAP_NET_RAW.

استلام الرسائل

يمكن استلام الرسائل باستخدام استدعاءات النظام recvfrom(2) و recvmsg(2).


char                  buf[13];
socklen_t             addrlen;
struct sockaddr_mctp  addr;
addrlen = sizeof(addr);
n = recvfrom(sd, buf, sizeof(buf), 0,
             (struct sockaddr *) &addr, &addrlen);

لاستقبال رسالة واردة، سيحتاج المستلم إما إلى إرسال رسالة مسبقًا إلى نفس نقطة النهاية، أو إجراء bind(2) لاستقبال جميع الرسائل من نوع معين:


struct sockaddr_mctp  addr;
addr.smctp_family = AF_MCTP;
addr.smctp_network = MCTP_NET_ANY;
addr.smctp_addr.s_addr = MCTP_ADDR_ANY;
addr.smctp_type = MYPROGRAM_MCTP_TYPE_ECHO;  /* نوع "الصدى" الخاص بنا */
int rc = bind(sd, (struct sockaddr *) &addr, sizeof(addr));

يتطلب هذا الاستدعاء القدرة CAP_NET_BIND_SERVICE، وسينتج عنه استقبال المقبس لجميع الرسائل المرسلة إلى EIDs المعينة محليًا، لنوع الرسالة المحدد.

بعد أن تقوم recvfrom(2) أو recvmsg(2) بإرجاع حالة نجاح، سيتم ملء وسيطة العنوان المقدمة بشبكة المرسل و EID، بالإضافة إلى قيمة العلامة المستخدمة للرسالة. يجب أن يمر أي رد على هذه الرسالة نفس العنوان وقيمة العلامة (مع مسح بت TO) للإشارة إلى أنها موجهة إلى نفس نقطة النهاية البعيدة.
