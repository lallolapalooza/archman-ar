الوصف

int mcrypt_get_key_size( char* algorithm, char* directory);

تُرجع الحد الأقصى لحجم المفتاح المدعوم للخوارزمية.

int* mcrypt_module_get_algo_supported_key_sizes( char* algorithm, char* directory, int* sizes);

تُرجع أحجام المفاتيح التي تدعمها الخوارزمية. إذا كان sizes صفرًا وتُرجع NULL، فسيتم دعم جميع أحجام المفاتيح بين 1 و mcrypt_get_key_size() بواسطة الخوارزمية. إذا كانت 1، فسيتم دعم حجم mcrypt_get_key_size() فقط وسيكون sizes[0] مساويًا له. إذا كان أكبر من 1، فإن هذا العدد يحدد عدد العناصر في sizes وهي أحجام المفاتيح التي تدعمها الخوارزمية. تختلف هذه الدالة عن mcrypt_enc_get_supported_key_sizes()، لأن القيمة المرجعة هنا مخصصة (ليست ثابتة)، وبالتالي يجب تحريرها.

char** mcrypt_list_algorithms ( char* libdir, int* size);

تُرجع مؤشرًا إلى مصفوفة أحرف تحتوي على جميع خوارزميات mcrypt الموجودة في libdir، أو إذا كانت NULL، في الدليل الافتراضي. الحجم هو عدد مصفوفات الأحرف. يتم تخصيص المصفوفات داخليًا ويجب تحريرها باستخدام mcrypt_free_p().

char** mcrypt_list_modes ( char* libdir, int *size);

تُرجع مؤشرًا إلى مصفوفة أحرف تحتوي على جميع أوضاع mcrypt الموجودة في libdir، أو إذا كانت NULL، في الدليل الافتراضي. الحجم هو عدد مصفوفات الأحرف. يجب تحرير المصفوفات باستخدام mcrypt_free_p().

void mcrypt_free_p (char **p, int size);

تحرر المؤشر إلى المصفوفة التي تم إرجاعها بواسطة الدوال السابقة.

void mcrypt_free (void *ptr);

تحرر الذاكرة المستخدمة بواسطة المؤشر.

void mcrypt_perror(int err);

تقوم هذه الدالة بطباعة وصف يمكن للبشر فهمه للخطأ 'err' في stderr. يجب أن يكون err قيمة تم إرجاعها بواسطة mcrypt_generic_init().

const char* mcrypt_strerror(int err);

تُرجع هذه الدالة وصفًا يمكن للبشر فهمه للخطأ 'err'. يجب أن يكون err قيمة تم إرجاعها بواسطة mcrypt_generic_init().

int mcrypt_mutex_register ( void (*mutex_lock)(void) , void (*mutex_unlock)(void) );

تُستخدم هذه الدالة فقط في التطبيقات متعددة الخيوط وفقط إذا تم ترجمتها مع دعم تحميل الوحدات الديناميكي. يتم استخدام هذا فعليًا داخليًا في libltdl. باستثناء تحميل الوحدات الديناميكي، فإن libmcrypt آمنة للاستخدام مع خيوط متعددة.

بعض البرامج التجريبية تتبع هنا. قم بترجمتها باستخدام "cc prog.c -lmcrypt" أو "cc prog.c -lmcrypt -lltdl" اعتمادًا على تثبيتك. يتم استخدام Libltdl لفتح المكتبات الديناميكية (الوحدات).

/* أول مثال: يقوم بتشفير stdin إلى stdout باستخدام TWOFISH بمفتاح 128 بت و CFB */
#include <mcrypt.h>
#include <stdio.h>
#include <stdlib.h>
/* #include <mhash.h> */
main() {
  MCRYPT td;
  int i;
  char *key;
  char password[20];
  char block_buffer;
  char *IV;
  int keysize=16; /* 128 بت */
  key=calloc(1, keysize);
  strcpy(password, "A_large_key");
/* إنشاء المفتاح باستخدام كلمة المرور */
/*  mhash_keygen( KEYGEN_MCRYPT, MHASH_MD5, key, keysize, NULL, 0, password, strlen(password));
 */
  memmove( key, password, strlen(password));
  td = mcrypt_module_open("twofish", NULL, "cfb", NULL);
  if (td==MCRYPT_FAILED) {
     return 1;
  }
  IV = malloc(mcrypt_enc_get_iv_size(td));
/* ضع بيانات عشوائية في IV. لاحظ أن هذه ليست بيانات عشوائية حقيقية،
 * فكر في استخدام /dev/random أو /dev/urandom.
 */
  /*  srand(time(0)); */
  for (i=0; i< mcrypt_enc_get_iv_size( td); i++) {
    IV[i]=rand();
  }
  i=mcrypt_generic_init( td, key, keysize, IV);
  if (i<0) {
     mcrypt_perror(i);
     return 1;
  }
  /* يتم إجراء التشفير في CFB بالبايت */
  while ( fread (&block_buffer, 1, 1, stdin) == 1 ) {
      mcrypt_generic (td, &block_buffer, 1);
/* علق أعلاه وفك التعليق عن هذا لفك التشفير */
/*    mdecrypt_generic (td, &block_buffer, 1);  */
      fwrite ( &block_buffer, 1, 1, stdout);
  }
/* إنهاء مهمة التشفير، وتحميل الوحدة */
  mcrypt_generic_end(td);
  return 0;
}
/* المثال الثاني: يقوم بالتشفير باستخدام CBC و SAFER+ بمفتاح 192 بت */
#include <mcrypt.h>
#include <stdio.h>
#include <stdlib.h>
main() {
  MCRYPT td;
  int i;
  char *key; /* تم إنشاؤه باستخدام mcrypt_gen_key */
  char *block_buffer;
  char *IV;
  int blocksize;
  int keysize = 24; /* 192 بت == 24 بايت */
  key = calloc(1, keysize);
  strcpy(key, "A_large_and_random_key");
  td = mcrypt_module_open("saferplus", NULL, "cbc", NULL);
  blocksize = mcrypt_enc_get_block_size(td);
  block_buffer = malloc(blocksize);
/* ولكن لسوء الحظ هذا لا يملأ كل المفتاح لذا فإن البايت المتبقية
 * يتم حشوها بالأصفار. حاول استخدام مفاتيح كبيرة أو تحويلها باستخدام mcrypt_gen_key().
 */
  IV=malloc(mcrypt_enc_get_iv_size(td));
/* ضع بيانات عشوائية في IV. لاحظ أن هذه ليست بيانات عشوائية حقيقية،
 * فكر في استخدام /dev/random أو /dev/urandom.
 */
/* srand(time(0)); */
  for (i=0; i < mcrypt_enc_get_iv_size(td); i++) {
    IV[i]=rand();
  }
  mcrypt_generic_init( td, key, keysize, IV);
  /* يتم إجراء التشفير في CBC بالكتل */
  while ( fread (block_buffer, 1, blocksize, stdin) == blocksize ) {
      mcrypt_generic (td, block_buffer, blocksize);
/*      mdecrypt_generic (td, block_buffer, blocksize); */
      fwrite ( block_buffer, 1, blocksize, stdout);
  }
/* إنهاء مهمة التشفير */
  mcrypt_generic_deinit (td);
/* إلغاء تحميل الوحدة المحملة */
  mcrypt_module_close(td);
  return 0;
}
