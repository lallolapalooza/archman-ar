الاسم

msgrcv, msgsnd - عمليات طوابير رسائل System V

المكتبة

مكتبة C القياسية (libc, -lc)

الملخص

#include <sys/msg.h>
ssize_t msgrcv(size_t msgsz;
               int msqid, void msgp[msgsz], size_t msgsz, long msgtyp,
               int msgflg);
int msgsnd(size_t msgsz;
               int msqid, const void msgp[msgsz], size_t msgsz,
               int msgflg);

الوصف

يتم استخدام استدعاءات النظام msgsnd() و msgrcv() لإرسال الرسائل إلى، واستقبال الرسائل من، طابور رسائل System V. يجب أن تكون لدى العملية المستدعية صلاحية الكتابة على طابور الرسائل لإرسال رسالة، وصلاحية القراءة لاستقبال رسالة.

الوسيط msgp هو مؤشر إلى بنية معرفة من قبل المتصل بالشكل العام التالي:


struct msgbuf {
    long mtype;       /* نوع الرسالة، يجب أن يكون > 0 */
    char mtext[1];    /* بيانات الرسالة */
};

حقل mtext هو مصفوفة (أو بنية أخرى) يحدد حجمها بواسطة msgsz، وهي قيمة صحيحة غير سالبة. يُسمح برسائل ذات طول صفر (أي، لا يوجد حقل mtext). يجب أن يكون لحقل mtype قيمة عدد صحيح موجب تمامًا. يمكن استخدام هذه القيمة بواسطة العملية المتلقية لاختيار الرسالة (انظر وصف msgrcv() أدناه).

msgsnd()

يقوم استدعاء النظام msgsnd() بإلحاق نسخة من الرسالة المشار إليها بواسطة msgp بطابور الرسائل الذي يحدده المعرف msqid.

إذا كانت هناك مساحة كافية في الطابور، تنجح msgsnd() على الفور. يتم التحكم في سعة الطابور بواسطة الحقل msg_qbytes في البنية البيانية المرتبطة بطابور الرسائل. أثناء إنشاء الطابور، يتم تهيئة هذا الحقل إلى MSGMNB بايت، ولكن يمكن تعديل هذا الحد باستخدام msgctl(2). يعتبر طابور الرسائل ممتلئًا إذا كان أي من الشروط التالية صحيحًا:

•
إضافة رسالة جديدة إلى الطابور ستتسبب في تجاوز الحجم الإجمالي للبايتات في الطابور الحد الأقصى لحجم الطابور (الحقل msg_qbytes).
•
إضافة رسالة أخرى إلى الطابور ستتسبب في تجاوز العدد الإجمالي للرسائل في الطابور الحد الأقصى لحجم الطابور (الحقل msg_qbytes). هذا الفحص ضروري لمنع وضع عدد غير محدود من الرسائل ذات الطول الصفري في الطابور. على الرغم من أن هذه الرسائل لا تحتوي على بيانات، إلا أنها تستهلك ذاكرة نواة (مقفل).

إذا كانت المساحة غير كافية في الطابور، فإن السلوك الافتراضي لـ msgsnd() هو الحظر حتى تصبح المساحة متاحة. إذا تم تحديد IPC_NOWAIT في msgflg، فستفشل المكالمة بدلاً من ذلك مع الخطأ EAGAIN.

قد تفشل مكالمة msgsnd() المحظورة أيضًا في الحالات التالية:

•
تمت إزالة الطابور، وفي هذه الحالة تفشل استدعاء النظام مع تعيين errno إلى EIDRM؛ أو
•
تم التقاط إشارة، وفي هذه الحالة تفشل استدعاء النظام مع تعيين errno إلى EINTR؛ انظر signal(7). (msgsnd() لا تتم إعادة تشغيلها تلقائيًا أبدًا بعد مقاطعتها بواسطة معالج إشارة، بغض النظر عن إعداد العلم SA_RESTART عند إنشاء معالج إشارة.)

عند اكتمالها بنجاح، يتم تحديث بنية بيانات طابور الرسائل على النحو التالي:

•
يتم تعيين msg_lspid إلى معرف العملية للعملية المستدعية.
•
تتم زيادة msg_qnum بمقدار 1.
•
يتم تعيين msg_stime إلى الوقت الحالي.

msgrcv()

يقوم استدعاء النظام msgrcv() بإزالة رسالة من الطابور المحدد بواسطة msqid ووضعها في المخزن المؤقت المشار إليه بواسطة msgp.

الوسيط msgsz يحدد الحد الأقصى للحجم بالبايت للعضو mtext في البنية المشار إليها بواسطة الوسيط msgp. إذا كان طول نص الرسالة أكبر من msgsz، فإن السلوك يعتمد على ما إذا كان MSG_NOERROR محددًا في msgflg. إذا تم تحديد MSG_NOERROR، فسيتم اقتطاع نص الرسالة (وسيتم فقدان الجزء المقتطع)؛ إذا لم يتم تحديد MSG_NOERROR، فلن تتم إزالة الرسالة من الطابور ويفشل استدعاء النظام ويعيد -1 مع تعيين errno إلى E2BIG.

ما لم يتم تحديد MSG_COPY في msgflg (انظر أدناه)، يحدد الوسيط msgtyp نوع الرسالة المطلوبة، على النحو التالي:

•
إذا كان msgtyp يساوي 0، تتم قراءة الرسالة الأولى في الطابور.
•
إذا كان msgtyp أكبر من 0، تتم قراءة الرسالة الأولى في الطابور من النوع msgtyp، ما لم يتم تحديد MSG_EXCEPT في msgflg، وفي هذه الحالة تتم قراءة الرسالة الأولى في الطابور من النوع الذي لا يساوي msgtyp.
•
إذا كان msgtyp أقل من 0، تتم قراءة الرسالة الأولى في الطابور بأقل نوع أقل من أو يساوي القيمة المطلقة لـ msgtyp.

الوسيط msgflg هو قناع بت تم إنشاؤه عن طريق ORing صفرًا أو أكثر من الأعلام التالية:

IPC_NOWAIT
ارجع على الفور إذا لم تكن هناك رسالة من النوع المطلوب في الطابور. تفشل استدعاء النظام مع تعيين errno إلى ENOMSG.
MSG_COPY (منذ Linux 3.8)
جلب نسخة غير مدمرة من الرسالة في الموضع الترتيبي في الطابور المحدد بواسطة msgtyp (تعتبر الرسائل مرقمة بدءًا من 0).
يجب تحديد هذا العلم بالاقتران مع IPC_NOWAIT، مع النتيجة أنه إذا لم تكن هناك رسالة متاحة في الموضع المحدد، تفشل المكالمة على الفور مع الخطأ ENOMSG. نظرًا لأنها تغير معنى msgtyp بطرق متعامدة، لا يمكن تحديد MSG_COPY و MSG_EXCEPT معًا في msgflg.
تمت إضافة العلم MSG_COPY لتنفيذ آلية التحقق من نقاط الاستعادة للنواة وهو متاح فقط إذا تم بناء النواة مع الخيار CONFIG_CHECKPOINT_RESTORE.
MSG_EXCEPT
يستخدم مع msgtyp أكبر من 0 لقراءة الرسالة الأولى في الطابور بنوع رسالة يختلف عن msgtyp.
MSG_NOERROR
لاقتطاع نص الرسالة إذا كان أطول من msgsz بايت.

إذا لم تكن هناك رسالة من النوع المطلوب متاحة ولم يتم تحديد IPC_NOWAIT في msgflg، فستتم مقاطعة العملية المستدعية حتى يحدث أي من الشروط التالية:

•
يتم وضع رسالة من النوع المطلوب في الطابور.
•
يتم إزالة طابور الرسائل من النظام. في هذه الحالة، تفشل استدعاء النظام مع تعيين errno إلى EIDRM.
•
تلتقط العملية المستدعية إشارة. في هذه الحالة، تفشل استدعاء النظام مع تعيين errno إلى EINTR. (msgrcv() لا تتم إعادة تشغيلها تلقائيًا أبدًا بعد مقاطعتها بواسطة معالج إشارة، بغض النظر عن إعداد العلم SA_RESTART عند إنشاء معالج إشارة.)

عند اكتمالها بنجاح، يتم تحديث بنية بيانات طابور الرسائل على النحو التالي:

يتم تعيين msg_lrpid إلى معرف العملية للعملية المستدعية.
يتم إنقاص msg_qnum بمقدار 1.
يتم تعيين msg_rtime إلى الوقت الحالي.

قيمة الإرجاع

عند النجاح، تعيد msgsnd() القيمة 0 وتعيد msgrcv() عدد البايتات التي تم نسخها فعليًا إلى المصفوفة mtext. عند الفشل، تعيد كلتا الدالتين -1، وتقوم بتعيين errno للإشارة إلى الخطأ.

الأخطاء

قد تفشل استدعاءات النظام msgsnd() و msgrcv() وتعين errno للإشارة إلى الخطأ في الحالات التالية:

EACCES
لا يمتلك الاستدعاء صلاحية الوصول المطلوبة (كتابة لـ msgsnd()، قراءة لـ msgrcv()) أو تم تغيير إذن الوصول الخاص بتطبيق MSG_SLM.
EAGAIN
IPC_NOWAIT تم تحديده في msgflg وكان الطابور ممتلئًا (لـ msgsnd()) أو لم تكن هناك رسالة من النوع المطلوب متاحة (لـ msgrcv()).
E2BIG
كان حجم الرسالة أكبر من msgsz ولم يتم تحديد MSG_NOERROR (لـ msgrcv()).
EFAULT
تم الوصول إلى عنوان ذاكرة غير صالح.
EINTR
تم مقاطعة استدعاء النظام بواسطة إشارة.
EINVAL
msqid غير صالح، أو أن msgsz أكبر من الحد الأقصى المسموح به (MSGMAX)، أو أن msgtyp غير صالح.
ENOMSG
لم تكن هناك رسالة من النوع المطلوب متاحة، وتم تحديد IPC_NOWAIT (لـ msgrcv()).
ENOSPC
كان الطابور ممتلئًا (بالنسبة لحجم البايتات) ولم يتم تحديد IPC_NOWAIT (لـ msgsnd()).
ENOSPC
حدث نقص في مساحة الذاكرة في النظام (على سبيل المثال، تم الوصول إلى الحد الأقصى لعدد الطوابير المسموح به).
ENOMSG
MSG_COPY تم تحديده، ولكن لم تكن هناك رسالة في الموضع المحدد في الطابور.
EIDRM
تمت إزالة الطابور.

ERRORS

msgsnd() can fail with the following errors:

EACCES
The calling process does not have write permission on the message queue, and does not have the CAP_IPC_OWNER capability in the user namespace that governs its IPC namespace.
EAGAIN
The message can't be sent due to the msg_qbytes limit for the queue and IPC_NOWAIT was specified in msgflg.
EFAULT
The address pointed to by msgp isn't accessible.
EIDRM
The message queue was removed.
EINTR
Sleeping on a full message queue condition, the process caught a signal.
EINVAL
Invalid msqid value, or nonpositive mtype value, or invalid msgsz value (less than 0 or greater than the system value MSGMAX).
ENOMEM
The system does not have enough memory to make a copy of the message pointed to by msgp.

msgrcv() can fail with the following errors:

E2BIG
The message text length is greater than msgsz and MSG_NOERROR isn't specified in msgflg.
EACCES
The calling process does not have read permission on the message queue, and does not have the CAP_IPC_OWNER capability in the user namespace that governs its IPC namespace.
EFAULT
The address pointed to by msgp isn't accessible.
EIDRM
While the process was sleeping to receive a message, the message queue was removed.
EINTR
While the process was sleeping to receive a message, the process caught a signal; see signal(7).
EINVAL
msqid was invalid, or msgsz was less than 0.
EINVAL (since Linux 3.14)
msgflg specified MSG_COPY, but not IPC_NOWAIT.
EINVAL (since Linux 3.14)
msgflg specified both MSG_COPY and MSG_EXCEPT.
ENOMSG
IPC_NOWAIT was specified in msgflg and no message of the requested type existed on the message queue.
ENOMSG
IPC_NOWAIT and MSG_COPY were specified in msgflg and the queue contains fewer than msgtyp messages.
ENOSYS (since Linux 3.8)
Both MSG_COPY and IPC_NOWAIT were specified in msgflg, and this kernel was configured without CONFIG_CHECKPOINT_RESTORE.

STANDARDS

POSIX.1-2024.

The MSG_EXCEPT and MSG_COPY flags are Linux-specific; their definitions can be obtained by defining the _GNU_SOURCE feature test macro.

HISTORY

POSIX.1-2001, SVr4.

The msgp argument is declared as struct msgbuf * in glibc 2.0 and 2.1. It is declared as void * in glibc 2.2 and later, as required by SUSv2 and SUSv3.

NOTES

The following limits on message queue resources affect the msgsnd() call:

MSGMAX
Maximum size of a message text, in bytes (default value: 8192 bytes). On Linux, this limit can be read and modified via /proc/sys/kernel/msgmax.
MSGMNB
Maximum number of bytes that can be held in a message queue (default value: 16384 bytes). On Linux, this limit can be read and modified via /proc/sys/kernel/msgmnb. A privileged process (Linux: a process with the CAP_SYS_RESOURCE capability) can increase the size of a message queue beyond MSGMNB using the msgctl(2) IPC_SET operation.

The implementation has no intrinsic system-wide limits on the number of message headers (MSGTQL) and the number of bytes in the message pool (MSGPOOL).

BUGS

In Linux 3.13 and earlier, if msgrcv() was called with the MSG_COPY flag, but without IPC_NOWAIT, and the message queue contained less than msgtyp messages, then the call would block until the next message is written to the queue. At that point, the call would return a copy of the message, regardless of whether that message was at the ordinal position msgtyp. This bug is fixed in Linux 3.14.

Specifying both MSG_COPY and MSC_EXCEPT in msgflg is a logical error (since these flags impose different interpretations on msgtyp). In Linux 3.13 and earlier, this error was not diagnosed by msgrcv(). This bug is fixed in Linux 3.14.

EXAMPLES

The program below demonstrates the use of msgsnd() and msgrcv().

The example program is first run with the -s option to send a message and then run again with the -r option to receive a message.

The following shell session shows a sample run of the program:


$ ./a.out -s
sent: a message at Wed Mar  4 16:25:45 2015
$ ./a.out -r
message received: a message at Wed Mar  4 16:25:45 2015

Program source

#include <errno.h>
#include <stdio.h>
#include <stdlib.h>
#include <sys/ipc.h>
#include <sys/msg.h>
#include <time.h>
#include <unistd.h>
struct msgbuf {
    long mtype;
    char mtext[80];
};
static void
usage(char *prog_name, char *msg)
{
    if (msg != NULL)
        fputs(msg, stderr);
    fprintf(stderr, "Usage: %s [options]\n", prog_name);
    fprintf(stderr, "Options are:\n");
    fprintf(stderr, "-s        send message using msgsnd()\n");
    fprintf(stderr, "-r        read message using msgrcv()\n");
    fprintf(stderr, "-t        message type (default is 1)\n");
    fprintf(stderr, "-k        message queue key (default is 1234)\n");
    exit(EXIT_FAILURE);
}
static void
send_msg(int qid, int msgtype)
{
    time_t         t;
    struct msgbuf  msg;
    msg.mtype = msgtype;
    time(&t);
    snprintf(msg.mtext, sizeof(msg.mtext), "a message at %s",
             ctime(&t));
    if (msgsnd(qid, &msg, sizeof(msg.mtext),
               IPC_NOWAIT) == -1)
    {
        perror("msgsnd error");
        exit(EXIT_FAILURE);
    }
    printf("sent: %s\n", msg.mtext);
}
static void
get_msg(int qid, int msgtype)
{
    struct msgbuf msg;
    if (msgrcv(qid, &msg, sizeof(msg.mtext), msgtype,
               MSG_NOERROR | IPC_NOWAIT) == -1) {
        if (errno != ENOMSG) {
            perror("msgrcv");
            exit(EXIT_FAILURE);
        }
        printf("No message available for msgrcv()\n");
    } else {
        printf("message received: %s\n", msg.mtext);
    }
}
int
main(int argc, char *argv[])
{
    int  qid, opt;
    int  mode = 0;               /* 1 = send, 2 = receive */
    int  msgtype = 1;
    int  msgkey = 1234;
    while ((opt = getopt(argc, argv, "srt:k:")) != -1) {
        switch (opt) {
        case 's':
            mode = 1;
            break;
        case 'r':
            mode = 2;
            break;
        case 't':
            msgtype = atoi(optarg);
            if (msgtype <= 0)
                usage(argv[0], "-t option must be greater than 0\n");
            break;
        case 'k':
            msgkey = atoi(optarg);
            break;
        default:
            usage(argv[0], "Unrecognized option\n");
        }
    }
    if (mode == 0)
        usage(argv[0], "must use either -s or -r option\n");
    qid = msgget(msgkey, IPC_CREAT | 0666);
    if (qid == -1) {
        perror("msgget");
        exit(EXIT_FAILURE);
    }
    if (mode == 2)
        get_msg(qid, msgtype);
    else
        send_msg(qid, msgtype);
    exit(EXIT_SUCCESS);
}
