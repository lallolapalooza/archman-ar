MIME__Type(3pm)

الحزمة: perl-mime-types

التراخيص: PerlArtistic

MIME::Type(3) وثائق Perl مساهم بها من المستخدم MIME::Type(3)

الاسم

MIME::Type - وصف لنوع MIME واحد

ملخص

use MIME::Types;
my $mimetypes = MIME::Types->new;
my MIME::Type $plaintext = $mimetypes->type('text/plain');
print $plaintext->mediaType;   # text
print $plaintext->subType;     # plain
my @ext = $plaintext->extensions;
print "@ext"                   # txt asc c cc h hh cpp
print $plaintext->encoding     # 8bit
if($plaintext->isBinary)       # false
if($plaintext->isText)         # true
if($plaintext->equals('text/plain') {...}
if($plaintext eq 'text/plain') # نفس الشيء
print MIME::Type->simplified('x-appl/x-zip') #  'appl/zip'

الوصف

تُستخدم أنواع MIME في كيانات MIME، على سبيل المثال كجزء من البريد الإلكتروني وحركة مرور HTTP. في بعض الأحيان تكون هناك حاجة لمعرفة حقيقية حول نوع MIME. تخزن الكائنات من "MIME::Type" المعلومات حول أحد هذه الأنواع.

محملة بشكل زائد

تحميل زائد: "" تحويل إلى سلسلة نصية
سيؤدي التحويل إلى سلسلة نصية (استخدام الكائن في مكان يتطلب سلسلة نصية) إلى اسم النوع، وهو نفس ما ترجعه type().

مثال: استخدام التحويل إلى سلسلة نصية

my $mime = MIME::Type->new('text/html');
print "$mime\n";   # تحويل صريح إلى سلسلة نصية
print $mime;       # تحويل ضمني إلى سلسلة نصية
تحميل زائد: == مقارنة رقمية
يقارن ما إذا كان type() للكائنات هو نفسه.
تحميل زائد: cmp مقارنة السلاسل النصية
عند مقارنة كائن MIME::Type بسلسلة نصية أو كائن MIME::Type آخر، يتم استدعاء طريقة equals(). المقارنة ذكية، مما يعني أنها توسع المقارنة النصية الشائعة ببعض الميزات المحددة في مواصفات RFC ذات الصلة.

طرق

التهيئة

$class->new(%options)
إنشاء (تهيئة) كائن MIME::Type جديد يدير نوع MIME واحد. 
-Option    --Default
 charset     undef
 encoding    <يعتمد على النوع>
 extensions  []
 simplified  <مشتق من النوع>
 system      undef
 type        <مطلوب>
charset => $charset
تحديد مجموعة الأحرف الافتراضية لهذا النوع.
encoding => '7bit'|'8bit'|'base64'|'quoted-printable'
كيف يجب تشفير هذه البيانات لنقلها بأمان. يعتمد الافتراضي على النوع: أنواع MIME التي يبدأ نوعها الرئيسي بـ "text/" ستكون افتراضياً "quoted-printable" وجميع الأنواع الأخرى ستكون "base64".
extensions => REF-ARRAY
مصفوفة من الامتدادات التي تستخدم هذا MIME.
simplified => STRING
يمكن أن يبدأ كل من التسمية الرئيسية والتسمية الفرعية لأنواع MIME بـ "x-"، للإشارة إلى أنه اسم غير مسجل. بالطبع، بعد التسجيل، يمكن أن تختفي هذه العلامة مما يزيد الارتباك. السلسلة النصية المبسطة لها علامات "x-" محذوفة ومترجمة إلى أحرف صغيرة.
system => REGEX
تعبير نمطي يحدد الأنظمة التي تكون فيها هذه القاعدة صالحة. يتم مطابقة التعبير النمطي على $^O.
type => STRING
النوع المحدد هنا. يتكون من نوع و نوع فرعي، كلاهما غير حساس لحالة الأحرف. ستعيد هذه الوحدة أحرفًا صغيرة، لكنها ستقبل أحرفًا كبيرة.

سمات

$obj->charset()
[2.28] تنص RFC6657 على أن تسجيلات IANA لأنواع الفئة النصية يجب أن تذكر صراحةً مجموعة الأحرف الافتراضية الخاصة بها. يحتوي MIME-Types على قائمة معدة يدويًا لهذه الافتراضيات.

قد ترجع هذه الطريقة أيضًا "_REQUIRED"، عندما لا يكون هناك افتراضي، أو "_FRAMED" عندما يتم تحديد مجموعة الأحرف بواسطة المحتوى.

$obj->encoding()
ترجع نوع الترميز المطلوب لنقل بيانات هذا النوع بأمان.
$obj->extensions()
ترجع قائمة بالامتدادات المعروف أنها مستخدمة لهذا النوع MIME.
$any->simplified( [$string] )
ترجع نوع MIME المبسط لهذا الكائن أو السلسلة النصية المحددة. يمكن تسجيل أسماء أنواع MIME رسميًا. حتى ذلك الحين، يجب أن تحمل بادئة "x-" للإشارة إلى ذلك. بالطبع، بعد التعرف عليها، يمكن أن تختفي بادئة "x-". في العديد من الحالات، نفضل الإصدار المبسط للنوع.

مثال: نتائج simplified()

my $mime = MIME::Type->new(type => 'x-appl/x-zip');
print $mime->simplified;                     # 'appl/zip'
print $mime->simplified('text/PLAIN');       # 'text/plain'
print MIME::Type->simplified('x-xyz/x-abc'); # 'xyz/abc'
$obj->system()
ترجع التعبير النمطي الذي يمكن استخدامه لتحديد ما إذا كان هذا النوع نشطًا على النظام الذي تعمل عليه.
$obj->type()
ترجع النوع الطويل لهذا الكائن، على سبيل المثال 'text/plain'

المعرفة

$obj->defaultCharset()
[2.29] وفقًا لـ RFC6657، يجب على جميع أنواع "text/*" إما تحديد مجموعة أحرف افتراضية في تسجيل IANA الخاص بها، أو طلب معلمة مجموعة الأحرف. قد تتطلب الأنواع غير النصية أيضًا مجموعة أحرف.

من الصعب استخراج هذه المعلومات من ملفات تسجيل IANA تلقائيًا، لذا فهي تتم صيانتها يدويًا.

مثال: استخدام مجموعة الأحرف الافتراضية

$charset //= $type->defaultCharset // 'utf-8';
$obj->equals($string|$mime)
يقارن كائن نوع MIME هذا بسلسلة نصية أو كائن آخر. في حالة السلسلة النصية، سيحدث التبسيط.
$obj->isAscii()
اسم قديم لـ isText().
$obj->isBinary()
ترجع "true" عندما لا يُعرف النوع بأنه نصي. انظر isText().
$obj->isExperimental()
[2.00] ترجع "true" عندما يتم تعريف النوع للاستخدام التجريبي؛ يبدأ النوع الفرعي بـ "x."
$obj->isPersonal()
[2.00] ترجع "true" عندما يتم تعريف النوع بواسطة شخص للاستخدام الخاص؛ يبدأ النوع الفرعي بـ "prs."
$obj->isRegistered()
يجب أن تبدأ أنواع MIME التي لم تسجلها IANA ولم يتم تعريفها في RFCs بـ "x-". ينطبق هذا على كل من نوع الوسائط والنوع الفرعي. في حالة بدء أي من النوعين بـ "x-"، سترجع هذه الطريقة "false".
$obj->isSignature()
ترجع "true" عندما يكون النوع في قائمة التواقيع المعروفة.
$obj->isText()
[2.05] جميع الأنواع التي قد تحتوي على سمة مجموعة الأحرف هي أنواع نصية. ومع ذلك، لا يوجد حاليًا سجل للسمات في هذه الوحدة ... لذا نخمن.
$obj->isVendor()
[2.00] ترجع "true" عندما يتم تعريف النوع بواسطة بائع؛ يبدأ النوع الفرعي بـ "vnd."
$obj->mediaType()
نوع وسائط MIME المبسط. بالنسبة لـ 'text/plain'، سترجع 'text'.

لأسباب تاريخية، لا تزال طريقة 'mainType' تستخدم لاسترداد نفس القيمة. ومع ذلك، تم إيقاف هذه الطريقة.
