: sig end

type t = string

val normalize : string -> (t, t) Result.t

يقوم بتطبيع السلسلة النصية المشفرة بـ UTF-8. التسلسلات غير الصالحة لـ UTF-8 تؤدي إلى خطأ ويتم استبدالها بـ U+FFFD. يتم وضع أحرف المعرفات في شكل تطبيع NFC. تبقى أحرف Unicode الأخرى دون تغيير.

val capitalize : string -> (t, t) Result.t

مثل normalize ، ولكن إذا بدأت السلسلة بحرف معرف صغير، يتم استبداله بالحرف المقابل الكبير. لا تتغير الأحرف اللاحقة.

val uncapitalize : string -> (t, t) Result.t

مثل normalize ، ولكن إذا بدأت السلسلة بحرف معرف كبير، يتم استبداله بالحرف المقابل الصغير. لا تتغير الأحرف اللاحقة.

val is_capitalized : t -> bool

ترجع true إذا بدأت السلسلة النصية المطبعة بحرف معرف كبير، و false بخلاف ذلك. قد ترجع نتائج غير صحيحة إذا لم تكن السلسلة النصية مطبعة.

val is_valid_identifier : t -> bool

تحقق مما إذا كانت السلسلة النصية المطبعة صالحة كمعرف OCaml:

-جميع الأحرف هي أحرف معرفات

-لا تبدأ برقم أو علامة اقتباس مفردة

val is_lowercase : t -> bool

ترجع true إذا كانت السلسلة النصية المطبعة تحتوي فقط على أحرف معرفات صغيرة، و false بخلاف ذلك. قد ترجع نتائج غير صحيحة إذا لم تكن السلسلة النصية مطبعة.

type validation_result =
| Valid
| Invalid_character of Uchar.t
(* حرف غير مسموح به
*)
| Invalid_beginning of Uchar.t
(* حرف غير مسموح به كحرف أول
*)

val validate_identifier : ?with_dot:bool -> t -> validation_result

مثل is_valid_identifier ، ولكنها ترجع رمز خطأ أكثر تفصيلاً. يمكن السماح بالنقاط لتوسيع الدعم لمعرفات تشبه المسارات.

val starts_like_a_valid_identifier : t -> bool

يتحقق مما إذا كانت السلسلة النصية المطبعة تبدأ بحرف معرف غير الرقم أو علامة الاقتباس المفردة. لا يتم التحقق من الأحرف اللاحقة.
