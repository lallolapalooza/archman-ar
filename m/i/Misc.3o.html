val find_first_mono : (int -> bool) -> int

find_first_mono p يأخذ شرطًا p : int -> bool نفترض أنه: 1. أحادي على الأعداد الطبيعية: إذا كان a <= b فإن p a يستلزم p b، 2. صحيح لبعض الأعداد الطبيعية في النطاق 0; max_int (هذا يعادل p max_int = true).

find_first_mono p هو أصغر عدد طبيعي N يحقق p، محسوب في O(log(N)) استدعاءات لـ p.

يدعم تطبيقنا حالتين لا يتم فيهما احترام الشروط المسبقة لـ p:

- إذا كان p دائمًا False، فإننا نرجع max_int دون تكرار أو تعطل.

- إذا كان p غير أحادي ولكنه صحيح في النهاية، فإننا نرجع قيمة صحيحة.

عمليات السلاسل النصية

val search_substring : string -> string -> int -> int

search_substring pat str start ترجع موضع أول ظهور للسلسلة النصية pat في السلسلة النصية str. يبدأ البحث عند الإزاحة start من str. تثير Not_found إذا لم يتم العثور على pat.

val replace_substring : before:string -> after:string -> string -> string

replace_substring ~before ~after str تستبدل جميع ظهورات before بـ after في str وترجع السلسلة الناتجة.

val rev_split_words : string -> string list

rev_split_words s تقسم s إلى كلمات مفصولة بمسافات بيضاء وترجع قائمة الكلمات بترتيب عكسي.

val cut_at : string -> char -> string * string

String.cut_at s c ترجع زوجًا يحتوي على السلسلة الفرعية قبل أول ظهور لـ c في s، والسلسلة الفرعية بعد هذا الظهور. let (before, after) = String.cut_at s c in
before ^ String.make 1 c ^ after هو الهوية إذا كانت s تحتوي على c.

تثير Not_found إذا لم يظهر الحرف في السلسلة.

val ordinal_suffix : int -> string

ordinal_suffix n هي اللاحقة المناسبة لإلحاقها بالرقم n كرقم ترتيبي: 1 -> "st"، 2 -> "nd"، 3 -> "rd"، 4 -> "th"، وهكذا. تتعامل مع الأرقام الأكبر (على سبيل المثال، 42 -> "nd") والأرقام 11-13 (التي تحصل جميعها على "th") بشكل صحيح.

val normalise_eol : string -> string

normalise_eol s ترجع نسخة جديدة من s مع إزالة أي '\r'. القصد هو معالجة النص الذي سيتم طباعته لاحقًا على قناة تقوم بتحويلات نهاية السطر (مثل Windows).

val delete_eol_spaces : string -> string

delete_eol_spaces s ترجع نسخة جديدة من s مع إزالة أي مسافات نهاية السطر. القصد هو تطبيع خرج المستوي الأعلى للاختبارات.

type ref_and_value =

val protect_refs : ref_and_value list -> (unit -> 'a) -> 'a

protect_refs l f يقوم مؤقتًا بتعيين r إلى v لكل R (r, v) في l أثناء تنفيذ f. تتم استعادة المحتوى السابق للمراجع حتى لو أثارت f استثناءً، دون تغيير تتبع الاستثناء.

val get_ref : 'a list ref -> 'a list

get_ref lr ترجع محتوى مرجع القائمة lr وتعيده إلى قائمة فارغة.

val set_or_ignore : ('a -> 'b option) -> 'b option ref -> 'a -> unit

set_or_ignore f opt x تعين opt إلى f x إذا رجعت Some _، أو تترك opt دون تغيير إذا رجعت None.

val edit_distance : string -> string -> int -> int option

edit_distance a b cutoff تحسب مسافة التحرير بين السلسلتين a و b. للمساعدة في الكفاءة، تستخدم حدًا أقصى: إذا كانت المسافة d أقل من cutoff، ترجع Some d، وإلا ترجع None.

خوارزمية المسافة الحالية هي Damerau-Levenshtein: تحسب عدد الإدراجات، الحذف، استبدال الأحرف، أو تبديل الأحرف لتحويل كلمة إلى أخرى. قد تتغير الخوارزمية المحددة في المستقبل.

val spellcheck : string list -> string -> string list

spellcheck env name يأخذ قائمة بأسماء env موجودة في البيئة الحالية واسمًا خاطئًا name، ويرجع قائمة اقتراحات مأخوذة من env قريبة من name وقد تكون خطأ مطبعيًا لأحدها.

val align_hint : prefix:string -> main:Format_doc.t -> hint:Format_doc.t -> Format_doc.t * Format_doc.t

aligned_hint main hint تقوم بمحاذاة الرسالة main ورسالة الاقتراح hint عموديًا. يتم التحكم في المحاذاة الرأسية باستخدام صناديق @{<ralign> ...: بداية الصندوق، سواء كانت رسالة الاقتراح أو الرسالة الرئيسية، يتم إزاحتها إلى اليسار لضمان محاذاة نهايات الصندوقين عموديًا، مع الأخذ في الاعتبار بادئة مسبقة prefix قبل الرسالة الرئيسية. على سبيل المثال،

let main, sub =
  align_hint
    ~prefix:"Error: "
    (doc_printf "@{<ralign>The value @}%a is not an instance variable"
      Style.inline_code "foobar"
    )
    (doc_printf
      "@{<ralign>Did you mean @}%a" Style.inline_code "foobaz"
    ) in
   printf "Error: %a%a" pp_doc main pp_doc sub

ينتج النص التالي:

Error:   The value "foobaz" is not an instance variable
Hint: Did you mean "foobar"?

حيث يتم إزاحة الرسالة الرئيسية إلى اليسار لمحاذاة "foobaz" و "foobar".
