MIME__Entity(3pm)

الحزمة: perl-mime-tools

التراخيص: GPL, PerlArtistic

MIME::Entity(3) وثائق Perl المساهمة من المستخدم MIME::Entity(3)

الاسم

MIME::Entity - فئة للرسائل MIME المحللة والمفككة

الملخص

قبل المتابعة، يجب عليك الاطلاع على MIME::Tools للتأكد من فهمك لمكان هذا المودول في الخطط الكبرى للأمور. هيا، افعل ذلك الآن. سأنتظر.

هل أنت مستعد؟ حسناً...

### إنشاء كيان:
$top = MIME::Entity->build(From    => 'me@myhost.com',
                           To      => 'you@yourhost.com',
                           Subject => "Hello, nurse!",
                           Data    => \@my_message);
### إرفاق أشياء به:
$top->attach(Path     => $gif_path,
             Type     => "image/gif",
             Encoding => "base64");
### توقيعه:
$top->sign;
### إخراجه:
$top->print(\*STDOUT);

الوصف

فئة فرعية من Mail::Internet.

توفر هذه الحزمة فئة لتمثيل كيانات رسائل MIME، كما هو محدد في RFCs 2045، 2046، 2047، 2048 و 2049.

أمثلة

أمثلة الإنشاء

إنشاء مستند لملف نصي عادي بترميز ASCII ذي 7 بت (الكثير من الأشياء افتراضية لنا):

$ent = MIME::Entity->build(Path=>"english-msg.txt");

إنشاء مستند لملف نصي يحتوي على أحرف 8 بت (Latin-1):

$ent = MIME::Entity->build(Path     =>"french-msg.txt",
                           Encoding =>"quoted-printable",
                           From     =>'jean.luc@inria.fr',
                           Subject  =>"C'est bon!");

إنشاء مستند لملف GIF (الوصف اختياري تمامًا؛ لاحظ أننا بحاجة إلى تحديد نوع المحتوى والترميز نظرًا لأنهما ليسا القيم الافتراضية):

$ent = MIME::Entity->build(Description => "A pretty picture",
                           Path        => "./docs/mime-sm.gif",
                           Type        => "image/gif",
                           Encoding    => "base64");

إنشاء مستند لديك النص الخاص به بالفعل، باستخدام "Data":

$ent = MIME::Entity->build(Type        => "text/plain",
                           Encoding    => "quoted-printable",
                           Data        => ["First line.\n",
                                          "Second line.\n",
                                          "Last line.\n"]);

إنشاء رسالة متعددة الأجزاء، مع إعطاء الهيكل بأكمله بشكل صريح:

### إنشاء المستوى الأعلى، وإعداد رؤوس البريد:
$top = MIME::Entity->build(Type     => "multipart/mixed",
                           From     => 'me@myhost.com',
                           To       => 'you@yourhost.com',
                           Subject  => "Hello, nurse!");
### المرفق #1: مستند نصي بسيط:
$top->attach(Path=>"./testin/short.txt");
### المرفق #2: ملف GIF:
$top->attach(Path        => "./docs/mime-sm.gif",
             Type        => "image/gif",
             Encoding    => "base64");
### المرفق #3: نص سنقوم بإنشائه بالنص المتوفر لدينا:
$top->attach(Data => $contents);

بافتراض أنك لا تعرف مسبقًا أنك ستحصل على مرفقات؟ لا مشكلة: يمكنك "الإرفاق" بأجزاء فردية أيضًا:

$top = MIME::Entity->build(From    => 'me@myhost.com',
                           To      => 'you@yourhost.com',
                           Subject => "Hello, nurse!",
                           Data    => \@my_message);
if ($GIF_path) {
    $top->attach(Path     => $GIF_path,
                 Type     => 'image/gif');
}

نسخ كيان (رؤوس، أجزاء... كل شيء ما عدا بيانات الجسم الخارجية):

my $deepcopy = $top->dup;

أمثلة الوصول

### الحصول على الرأس، وهو MIME::Head:
$head = $ent->head;
### الحصول على الجسم، كـ MIME::Body;
$bodyh = $ent->bodyhandle;
### الحصول على نوع MIME المقصود (كما هو معلن في الرأس):
$type = $ent->mime_type;
### الحصول على النوع الفعلي لـ MIME (في حال فشل فك الترميز):
$eff_type = $ent->effective_type;
### الحصول على المقدمة، الأجزاء، والخاتمة:
$preamble   = $ent->preamble;          ### مرجع إلى مصفوفة من الأسطر
$num_parts  = $ent->parts;
$first_part = $ent->parts(0);          ### كيان
$epilogue   = $ent->epilogue;          ### مرجع إلى مصفوفة من الأسطر

أمثلة التلاعب

العبث ببيانات الجسم:

### قراءة بيانات الجسم (غير المشفرة):
if ($io = $ent->open("r")) {
    while (defined($_ = $io->getline)) { print $_ }
    $io->close;
}
### كتابة بيانات الجسم (غير المشفرة):
if ($io = $ent->open("w")) {
    foreach (@lines) { $io->print($_) }
    $io->close;
}
### حذف ملفات أي بيانات خارجية (على القرص):
$ent->purge;

العبث بالتوقيع:

### توقيعه (يزيل تلقائيًا أي توقيع موجود):
$top->sign(File=>"$ENV{HOME}/.signature");
### إزالة أي توقيع ضمن 15 سطرًا من النهاية:
$top->remove_sig(15);

العبث بالرؤوس:

### حساب أطوال المحتوى للأجزاء الفردية بناءً على الأجسام:
###   (افعل هذا قبل الطباعة مباشرة!)
$entity->sync_headers(Length=>'COMPUTE');

العبث بالهيكل:

### إذا كان متعدد الأجزاء بجزء 0 أو 1، فقم بدمجه في جزء واحد:
$top->make_singlepart;
### إذا كان جزءًا واحدًا، فقم بتوسيعه إلى متعدد الأجزاء بجزء واحد:
$top->make_multipart;

حذف الأجزاء:

### حذف بعض أجزاء رسالة متعددة الأجزاء:
my @keep = grep { keep_part($_) } $msg->parts;
$msg->parts(\@keep);

أمثلة الإخراج

الطباعة إلى مقابض الملفات:

### طباعة الرسالة بأكملها:
$top->print(\*STDOUT);
### طباعة الرأس فقط:
$top->print_header(\*STDOUT);
### طباعة الجسم (المشفر) فقط... يشمل الأجزاء أيضًا!
$top->print_body(\*STDOUT);

التحويل إلى سلسلة نصية... لاحظ أنه يمكن كتابة "stringify_xx" أيضًا كـ "xx_as_string"؛ الدوال مرادفة، ولن يتم إيقاف أي من الشكلين.

إذا قمت بتعيين المتغير $MIME::Entity::BOUNDARY_DELIMITER إلى سلسلة نصية، فسيتم استخدام تلك السلسلة كمحدد لنهاية السطر عند الإخراج. إذا لم يتم تعيينه، فسيكون إنهاء السطر حرف سطر جديد (\n)

لاحظ أن $MIME::Entity::BOUNDARY_DELIMITER ينطبق فقط على الأجزاء الهيكلية لبيانات MIME التي تم إنشاؤها بواسطة هذه الحزمة وعلى الإخراج المشفر بـ Base64؛ إذا استخدم جزء داخليًا محدد نهاية سطر مختلف وتم إخراجه كما هو، فلن يتم تغيير إنهاء السطر ليتطابق مع $MIME::Entity::BOUNDARY_DELIMITER.

### تحويل الرسالة بأكملها إلى سلسلة نصية:
print $top->stringify;              ### أو $top->as_string
### تحويل الرأس فقط إلى سلسلة نصية:
print $top->stringify_header;       ### أو $top->header_as_string
### تحويل الجسم (المشفر) فقط إلى سلسلة نصية... يشمل الأجزاء أيضًا!
print $top->stringify_body;         ### أو $top->body_as_string

التصحيح:

### طباعة معلومات التصحيح:
$entity->dump_skeleton(\*STDERR);

الواجهة العامة

الإنشاء

new [SOURCE]
طريقة فئة. إنشاء كيان MIME جديد وفارغ. في الأساس، هذا يستخدم مُنشئ Mail::Internet...

إذا كان SOURCE عبارة عن ARRAYREF، فمن المفترض أن يكون مصفوفة من الأسطر التي سيتم استخدامها لإنشاء كل من الرأس وجسم في الذاكرة.

بخلاف ذلك، إذا كان SOURCE معرفًا، فمن المفترض أن يكون مقبض ملف سيتم قراءة الرأس والجسم في الذاكرة منه.

ملاحظة: في كلتا الحالتين، لن يتم تحليل الجسم: سيتم قراءته ببساطة!

ambiguous_content
طريقة مثيل.

تُرجع صحيحًا إذا كان هذا الكيان أو أي من أجزائه، بشكل متكرر له "MIME::Head" يشير إلى محتوى غامض.

لاحظ بعناية الفرق بين:

$entity->head

هذا الموقع غير مرتبط بـ Arch Linux ولا تابع له بأي شكل من الأشكال. المحتوى هو ترجمة فقط لصفحات دليل Arch المنشورة على man.archlinux.org.

توقيع

توقيع PARAMHASH
طريقة مثيل، تم تجاوزها. يضيف توقيعًا إلى الرسالة. الوسائط هي:
مرفق
بدلاً من إضافة النص، يضيف المحتوى كمرفق بالرسالة. ستكون الحالة "inline"، وسيشير الوصف إلى أنه توقيع. السلوك الافتراضي هو إضافة التوقيع إلى نص الرسالة (أو إلى نص الجزء الأول إذا كان متعدد الأجزاء). خاص بـ MIME؛ جديد في هذه الفئة الفرعية.
ملف
يستخدم محتويات هذا الملف كتوقيع. يثير خطأ إذا لم يتمكن من قراءته. مثل طريقة الفئة الأعلى.
فرض
يوقع حتى لو لم يكن نوع المحتوى "text/*". يمكن استخدامه لأنواع غير قياسية مثل "x-foobar"، ولكن افعل ذلك بحذر! خاص بـ MIME؛ جديد في هذه الفئة الفرعية.
إزالة
عادةً، نحاول إزالة أي توقيع موجود. إذا كان صحيحًا، فإنه يعطينا معلمة NLINES من استدعاء remove_sig. إذا كانت صفرًا ولكن محددة، فهذا يشير إلى أنهم لن يزيلوا أي توقيع موجود. إذا لم يتم تحديده، يتم إجراء الإزالة بالقيمة الافتراضية 10 أسطر. جديد في هذه الفئة الفرعية.
توقيع
يستخدم هذا النص كتوقيع. يمكنك توفيره كعدد مفرد أو كمرجع لمصفوفة من الأسطر المنتهية بسطر جديد. مثل طريقة الفئة الأعلى.

هذه الطريقة حذرة جدًا مع رسائل MIME: ستحاول فقط توقيع رسالة بنوع محتوى "text/*"، ما لم يتم تحديد "Force".

إذا أرسلت هذه الرسالة إلى كيان متعدد الأجزاء، فسيتم إعادة توجيهها إلى الجزء الأول (عادةً ما تكون الأجزاء الأخرى "مرفقات").

انتباه: حاليًا، تقوم بتحميل جزء الرسالة بأكمله كمصفوفة من الأسطر في الذاكرة، لذلك من المحتمل أنك لا ترغب في استخدامه مع رسائل طويلة جدًا.

يعيد قيمة صحيحة عند النجاح، وقيمة خاطئة عند الفشل.

suggest_encoding
طريقة مثيل. تعيد اقتراح ترميز مناسب بناءً على نوع المحتوى الفعال.

"text" و "message" يتم تحليل أجسام الأنواع سطرًا بسطر بحثًا عن أحرف 8 بت وأسطر طويلة؛ يشير عدم وجود أي منهما إلى أن الرسالة 7bit-ok. يتم اختيار الأنواع الأخرى بشكل مستقل عن أجسامها:

نوع رئيسي:      7bit ok؟    الترميز المقترح:
-----------------------------------------------------------
text              نعم         7bit
text              لا          quoted-printable
message           نعم         7bit
message           لا          binary
multipart         *          binary (في حالة وجود بعض الأجزاء السيئة)
image, etc...    *          base64
sync_headers OPTIONS
طريقة مثيل. تجري مجموعة متنوعة من المهام التي تضمن مزامنة رؤوس MIME لكيان "شجرة" مع الأجزاء التي تصفها. يمكن أن تكون مكلفة مثل الطباعة إذا كانت تتضمن الترميز المسبق للأجزاء؛ ومع ذلك، فإن الهدف هو إنتاج تنسيق MIME نظيف. عادةً ما تستخدمها فقط إذا كنت تعالج وتعيد توجيه MIME من مصدر خارجي.

OPTIONS هو تجزئة تصف ما سيتم القيام به.

Length
أحد حقول MIME الرسمية غير الرسمية هو "Content-Length". عادةً، لا تهتم بهذا الحقل؛ ومع ذلك، إذا كنت تستعد لإخراج مخصص لـ HTTP، فقد تفعل ذلك. يشير قيمة هذا الخيار إلى ما سيتم القيام به:

COMPUTE يعني تعيين حقل "Content-Length" لكل جزء غير متعدد الأجزاء في الكيان، وتركه فارغًا لكل جزء متعدد الأجزاء في الكيان.

ERASE يعني أن جميع حقول "Content-Length" سيتم تركها فارغة. هذا سريع، غير مؤلم، وآمن.

أي قيمة خاطئة (الافتراضي) تعني أنه لن يتم اتخاذ أي إجراء.

Nonstandard
كل حقل رأس يبدأ بـ "Content-" هو، وفقًا لمعايير RFC، حقل MIME. ومع ذلك، فإن بعضها غير قياسي وقد يسبب مشاكل مع بعض قراء MIME الذين يفسرونها بشكل مختلف.

ERASE يعني أن جميع هذه الحقول سيتم تركها فارغة. يتم ذلك قبل Length (انظر) ويتم التصرف بناءً عليه.

أي قيمة خاطئة (الافتراضي) تعني أنه لن يتم اتخاذ أي إجراء.

يعيد قيمة صحيحة إذا سار كل شيء على ما يرام، وقيمة خاطئة إذا لم يكن كذلك.

tidy_body
طريقة مثيل، تم تجاوزها. حاليًا غير مطبقة لرسائل MIME. لا تفعل شيئًا وتعيد false.

إخراج

dump_skeleton [FILEHANDLE]
طريقة مثيل. تطبع هيكل الكيان في مقبض الملف FILEHANDLE، أو في المقبض المحدد حاليًا إذا لم يتم توفيره.

يتم طباعة كل كيان بمستوى مسافة بادئة مناسب، وتستخدم السمات التالية:

Content-type: multipart/mixed
Effective-type: multipart/mixed
Body-file: NONE
Subject: Hey there!
Num-parts: 2

هذا مفيد حقًا لتصحيح الأخطاء؛ لا أقدم ضمانات بشأن اتساق تنسيق الإخراج مع مرور الوقت.

print [OUTSTREAM]
طريقة مثيل، تم تجاوزها. تطبع الكيان في OUTSTREAM، أو في ملف الإخراج الحالي إذا لم يتم توفيره. يمكن أن يكون OUTSTREAM ملفًا أو أي كائن يستجيب لرسالة print().

يتم طباعة الكيان كتدفق MIME صالح! هذا يعني أن الرأس دائمًا ما يتم طباعته أولاً، وسيتم ترميز الجسم (إذا كان موجودًا) إذا أشار الرأس إلى أنه يجب ذلك. على سبيل المثال، قد يبدو إخراجك كالتالي:

Subject: Greetings
Content-transfer-encoding: base64
SGkgdGhlcmUhCkJ5ZSB0aGVyZSEK

إذا كان لهذا الكيان نوع MIME "multipart/*"، فسيتم طباعة المقدمات والأجزاء والخاتمة مع حدود مناسبة بين كل منها. يتم تجاهل أي جسم جزء:

Content-type: multipart/mixed; boundary="*----*"
Content-transfer-encoding: 7bit
[Preamble]
--*----*
[Entity: Part 0]
--*----*
[Entity: Part 1]
--*----*--
[Epilogue]

إذا كان لهذا الكيان نوع MIME لجزء واحد فقط ولا توجد أجزاء مرفقة، فنحن ننظر إلى كيان عادي لجزء واحد فقط: يتم طباعة الجسم وفقًا للرأس. إذا لم يكن هناك جسم، يتم إصدار تحذير ويعتبر الجسم فارغًا:

Content-type: image/gif
Content-transfer-encoding: base64
[Encoded body]

إذا كان لهذا الكيان نوع MIME لجزء واحد فقط ولكنه يحتوي أيضًا على أجزاء، فنحن ننظر على الأرجح إلى جزء واحد "معاد تحليله"، عادةً من نوع "message/*" (يمكنك الحصول على كيانات كهذه إذا قمت بتعيين الخيار parse_nested_messages(NEST) على true في المحلل). في هذه الحالة، سيتم طباعة الأجزاء مع فواصل أسطر فارغة، وسيتم تجاهل أي جسم جزء:

Content-type: message/rfc822
Content-transfer-encoding: 7bit
[Entity: Part 0]
[Entity: Part 1]

في جميع الحالات، عند طباعة "جزء" من الكيان، يتم استدعاء هذه الطريقة بشكل متكرر.

ملاحظة: من المحتمل جدًا أن يكون الإخراج غير متطابق مع أي إدخال قمت بتحليله للحصول على هذا الكيان. إذا كنت تقوم ببناء معالج بريد، فمن مسؤوليتك الاحتفاظ بهذه المعلومات.

print_body [OUTSTREAM]
طريقة مثيل، تم تجاوزها. تطبع جسم الكيان في OUTSTREAM، أو في ملف الإخراج الحالي إذا لم يتم توفيره. يمكن أن يكون OUTSTREAM ملفًا أو أي كائن يستجيب لرسالة print().

يتم طباعة الجسم ليتم تضمينه في تدفق MIME صالح؛ هذا يعني أنه سيتم ترميز الجسم إذا أشار الرأس إلى أنه يجب ذلك.

انتباه: "الجسم" يعني "ما يأتي بعد الرأس". يتضمن الجسم متعدد الأجزاء التمثيلات المطبوعة لأجزائه الفرعية.

انتباه: يتم تخزين الجسم غير مرمّز؛ ومع ذلك، فإن الهدف هو استخدام الترميز لتحديد كيفية طباعة الجسم. هذا يعني أن طريقة print() تضمن دائمًا أن تحصل على تدفق جاهز للإرسال حيث يكون الجسم متسقًا مع رأسه. إذا كنت ترغب في الحصول على نص الجسم مباشرة، يمكنك قراءته من مقبض الجسم أو استخدام:

$ent->bodyhandle->print($outstream);

الذي يستخدم استدعاءات read() لاستخراج المعلومات، وبالتالي سيعمل مع كل من الأجسام النصية والملف الثنائي.

انتباه: يرجى توفير OUTSTREAM. تختلف طريقة التجاوز هذه عن Mail::Internet، التي تخرج إلى STDOUT إذا لم يتم توفير ملف إخراج: قد يسبب هذا ارتباكًا.

print_header [OUTSTREAM]
طريقة مثيل، موروثة. تطبع الرأس في OUTSTREAM. يجب عليك توفير OUTSTREAM!
stringify
طريقة مثيل. تعيد الكيان كسلسلة، تمامًا كما ستطبع "print". سيتم ترميز الجسم بالضرورة وسيتضمن أي أجزاء فرعية. يمكنك أيضًا استخدام as_string().
stringify_body
طريقة مثيل. تعيد جسم الرسالة مرمّزًا كسلسلة، تمامًا كما ستطبع "print_body". يمكنك أيضًا استخدام body_as_string().

إذا كنت تريد الجسم غير المرمّز، وتعمل مع رسالة لجزء واحد فقط (مثل image/gif)، يمكنك قراءتها من مقبض الجسم أو استخدام:

$ent->bodyhandle->print($outstream);

الذي يستخدم استدعاءات read() لاستخراج المعلومات، وبالتالي سيعمل مع كل من الأجسام النصية والملف الثنائي.
