mxml(3)

الحزمة: mxml

التراخيص: LGPL

mxmlIndexFind

ابحث عن العقدة المطابقة التالية.

mxml_node_t * mxmlIndexFind (
    mxml_index_t *ind,
    const char *element,
    const char *value
);

يجب عليك استدعاء mxmlIndexReset قبل استخدام هذه الدالة لأول مرة مع مجموعة معينة من سلاسل "element" و "value". يمثل تمرير NULL لكل من "element" و "value" ما يعادل استدعاء mxmlIndexEnum.

mxmlIndexGetCount

الحصول على عدد العقد في الفهرس.

int  mxmlIndexGetCount (
    mxml_index_t *ind
);

mxmlIndexNew

إنشاء فهرس جديد.

mxml_index_t * mxmlIndexNew (
    mxml_node_t *node,
    const char *element,
    const char *attr
);

سيحتوي الفهرس على جميع العقد التي تحتوي على العنصر و/أو السمة المسماة. إذا كان كل من "element" و "attr" هو NULL، فسوف يحتوي الفهرس على قائمة مرتبة من العناصر في شجرة العقد. يتم فرز العقد حسب اسم العنصر واختياريًا حسب قيمة السمة إذا كانت الوسيطة "attr" ليست NULL.

mxmlIndexReset

إعادة تعيين مؤشر التعداد/البحث في الفهرس وإرجاع العقدة الأولى في الفهرس.

mxml_node_t * mxmlIndexReset (
    mxml_index_t *ind
);

يجب استدعاء هذه الدالة قبل استخدام mxmlIndexEnum أو mxmlIndexFind لأول مرة.

mxmlLoadFd

تحميل واصف ملف في شجرة عقد XML.

mxml_node_t * mxmlLoadFd (
    mxml_node_t *top,
    int fd,
    mxml_load_cb_t cb
);

تتم إضافة العقد في الملف المحدد إلى العقدة العلوية المحددة. إذا لم يتم توفير عقدة علوية، يجب أن يكون ملف XML صالحًا مع عقدة أصل واحدة مثل
للملف بأكمله. تُرجع دالة الاستدعاء العكسي نوع القيمة الذي يجب استخدامه للعقد الفرعية. الثوابت MXML_INTEGER_CALLBACK، MXML_OPAQUE_CALLBACK، MXML_REAL_CALLBACK، و MXML_TEXT_CALLBACK معرّفة لتحميل العقد الفرعية (البيانات) من النوع المحدد.

ملاحظة: الخطأ البرمجي الأكثر شيوعًا عند استخدام مكتبة Mini-XML هو تحميل ملف XML باستخدام MXML_TEXT_CALLBACK، الذي يُرجع النص المتضمن كسلسلة من الكلمات مفصولة بمسافات بيضاء، بدلاً من استخدام MXML_OPAQUE_CALLBACK الذي يُرجع النص المتضمن كسلسلة واحدة (بما في ذلك المسافات البيضاء).

mxmlNewCDATA

إنشاء عقدة CDATA جديدة.

mxml_node_t * mxmlNewCDATA (
    mxml_node_t *parent,
    const char *data
);

تتم إضافة عقدة CDATA الجديدة إلى نهاية قائمة العقد الفرعية للعقدة الأصل المحددة. يمكن استخدام الثابت MXML_NO_PARENT لتحديد أن عقدة CDATA الجديدة ليس لها أصل. يجب أن تنتهي السلسلة النصية للبيانات بـ null ويتم نسخها إلى العقدة الجديدة. تستخدم عقد CDATA حاليًا النوع MXML_ELEMENT.

تتم إضافة العقدة المخصصة الجديدة إلى نهاية قائمة العقد الفرعية للعقدة الأصل المحددة. يمكن استخدام الثابت MXML_NO_PARENT لتحديد أن عقدة العنصر الجديدة ليس لها أصل. يمكن تمرير NULL عندما لا يتم تخصيص البيانات في العقدة ديناميكيًا أو تتم إدارتها بشكل منفصل.

mxmlNewXML

إنشاء شجرة مستند XML جديدة.

    const char *version
);

تحدد الوسيطة "version" رقم الإصدار الذي سيتم وضعه في عقدة العنصر ?xml. إذا كانت NULL، يتم افتراض الإصدار "1.0".

mxmlRelease

إصدار عقدة.

int  mxmlRelease (
    mxml_node_t *node
);

عندما يصل عدد المراجع إلى الصفر، يتم حذف العقدة (وأي عقد فرعية) عبر mxmlDelete.

mxmlRemove

إزالة عقدة من أصلها.

void mxmlRemove (
    mxml_node_t *node
);

لا تقوم هذه الدالة بتحرير الذاكرة المستخدمة بواسطة العقدة - استخدم mxmlDelete لذلك. لا تقوم هذه الدالة بأي شيء إذا لم تكن للعقدة أصل.
