MeasureChar(3)

الحزمة: tk

التراخيص: TCL

Tk_MeasureChars(3) إجراءات مكتبة Tk Tk_MeasureChars(3)

الاسم

Tk_MeasureChars, Tk_TextWidth, Tk_DrawChars, Tk_UnderlineChars - إجراءات لقياس وعرض سلاسل نصية بسيطة أحادية السطر.

الملخص

#include <tk.h>

int
Tk_MeasureChars(tkfont, string, numBytes, maxPixels, flags, lengthPtr)

int
Tk_TextWidth(tkfont, string, numBytes)

Tk_DrawChars(display, drawable, gc, tkfont, string, numBytes, x, y)

Tk_UnderlineChars(display, drawable, gc, tkfont, string, x, y, firstByte, lastByte)

الوسائط

Tk_Font tkfont (في)
رمز الخط الذي سيتم رسم النص به أو قياسه. يجب أن يكون قد تم إرجاعه بواسطة استدعاء سابق لـ Tk_GetFont.
const char *string (في)
النص المراد قياسه أو عرضه. لا يلزم أن يكون منتهيًا بصفر. سيتم قياس أي أحرف وصفية غير قابلة للطباعة في السلسلة (مثل علامات الجدولة، وفواصل الأسطر، وأحرف التحكم الأخرى) أو عرضها بطريقة تعتمد على النظام الأساسي.
int numBytes (في)
الحد الأقصى لعدد البايتات التي سيتم أخذها في الاعتبار عند قياس أو رسم string. يجب أن يكون أكبر من أو يساوي 0.
int maxPixels (في)
إذا كان maxPixels >= 0، فإنه يحدد أقصى طول للسطر بالبكسل. تتم معالجة الأحرف من string فقط حتى يتم تغطية هذا العدد من البكسلات. إذا كان maxPixels < 0، فإن طول السطر غير محدود ويتم تجاهل الوسيط flags.
int flags (في)
بتات علم مختلفة مدمجة معًا: TK_PARTIAL_OK يعني تضمين حرف طالما أن أي جزء منه يناسب الطول المحدد بواسطة maxPixels؛ وإلا، يجب أن يناسب الحرف بالكامل ليتم أخذه في الاعتبار. TK_WHOLE_WORDS يعني التوقف عند حدود الكلمة، إن أمكن. إذا تم تعيين TK_AT_LEAST_ONE، فهذا يعني إرجاع حرف واحد على الأقل حتى لو لم تتمكن أي أحرف من الملاءمة في الطول المحدد بواسطة maxPixels. إذا تم تعيين TK_AT_LEAST_ONE وتم تعيين TK_WHOLE_WORDS أيضًا، فهذا يعني أنه إذا لم تتناسب كلمة واحدة على السطر، فقم بإرجاع الأحرف القليلة الأولى من الكلمة التي تناسبت؛ إذا لم تتناسب حتى حرف واحد من الكلمة، فسيتم إرجاع الحرف الأول.
int *lengthPtr (خارج)
يتم ملؤه بعدد البكسلات التي تشغلها عدد الأحرف التي تم إرجاعها كنتيجة لـ Tk_MeasureChars.
Display *display (في)
الشاشة التي سيتم الرسم عليها.
Drawable drawable (في)
النافذة أو الرسم النقطي الذي سيتم الرسم فيه.
GC gc (في)
سياق الرسوميات لرسم الأحرف. يجب أن يكون الخط المحدد في هذا GC هو نفسه tkfont.
int x, y (في)
الإحداثيات التي سيتم وضع الحافة اليسرى للخط الأساسي عندها عند عرض string.
int firstByte (في)
فهرس البايت الأول للحرف الأول المراد تسطيره في string. يبدأ التسطير عند الحافة اليسرى لهذا الحرف.
int lastByte (في)
فهرس البايت الأول لآخر حرف سيتم رسم التسطير حتى عنده. الحرف المحدد بواسطة lastByte لن يتم تسطيره بنفسه.

الوصف

هذه الإجراءات مخصصة لقياس وعرض السلاسل النصية البسيطة أحادية الخط. لقياس وعرض نص أحادي الخط متعدد الأسطر ومبرر، ارجع إلى وثائق Tk_ComputeTextLayout. لا توجد واجهة برمجة في القلب من Tk تدعم النص متعدد الخطوط ومتعدد الأسطر؛ يجب بناء دعم لهذا السلوك على طبقات أبسط. لاحظ أن الواجهات الموصوفة هنا تستند إلى البايتات وليس الأحرف، لذا فإن قيم الفهارس الواردة من نصوص Tcl تحتاج إلى تحويلها إلى إزاحات بايت باستخدام Tcl_UtfAtIndex والإجراءات ذات الصلة.

الرمز (glyph) هو الصورة القابلة للعرض لحرف أو رقم أو رمز آخر. ليس لكل رموز الأحرف في خط معين رمز (glyph). الأحرف مثل علامات الجدولة، وفواصل الأسطر/الإرجاع، وأحرف التحكم التي ليس لها رمز (glyph) يتم قياسها وعرضها بواسطة هذه الإجراءات بطريقة تعتمد على النظام الأساسي؛ تحت X، يتم استبدالها بتسلسلات هروب مسبوقة بشرطة مائلة، بينما تحت Windows و Macintosh قد يتم استبدالها بمربعات فارغة أو صلبة. ارجع إلى وثائق Tk_ComputeTextLayout للحصول على واجهة برمجة تدعم التوسع المستقل عن النظام الأساسي لأحرف الجدولة إلى أعمدة وفواصل الأسطر/الإرجاع إلى نصوص متعددة الأسطر.

يتم استخدام Tk_MeasureChars على حد سواء لحساب طول سلسلة نصية معينة وحساب عدد الأحرف من سلسلة نصية تناسب مساحة معينة. القيمة المُرجعة هي عدد البايتات من string التي تناسب المساحة المحددة بواسطة maxPixels وفقًا للشروط الموصوفة بواسطة flags. إذا تناسبت جميع الأحرف، فإن القيمة المُرجعة ستكون numBytes. يتم ملء *lengthPtr بالعرض المحسوب، بالبكسلات، للجزء من السلسلة النصية الذي تم قياسه. على سبيل المثال، إذا كانت القيمة المُرجعة 5، فسيتم ملء *lengthPtr بالمسافة بين الحافة اليسرى لـ string[0] والحافة اليمنى لـ string[4].

Tk_TextWidth هي دالة تغليف توفر واجهة أبسط للدالة Tk_MeasureChars. القيمة المُرجعة هي مقدار المساحة بالبكسلات التي تحتاجها string المعطاة.

يقوم Tk_DrawChars برسم string في الموقع المحدد في drawable المعطاة.

يقوم Tk_UnderlineChars بتسطير النطاق المحدد من الأحرف في string المعطاة. لا يرسم الأحرف (التي يفترض أنه تم عرضها سابقًا بواسطة Tk_DrawChars)؛ بل يرسم التسطير فقط. يتم استخدام هذا الإجراء لتسطير عدد قليل من الأحرف دون الحاجة إلى إنشاء خط مسطر. لإنتاج نص مسطر أصلي، يجب إنشاء واستخدام الخط المسطر المناسب.
