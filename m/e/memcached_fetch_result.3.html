memcached_fetch_result(3)

الحزمة: libmemcached-awesome

التراخيص: BSD

MEMCACHED_FETCH_RESULT(3) libmemcached-awesome MEMCACHED_FETCH_RESULT(3)

الاسم

memcached_fetch_result - استرجاع البيانات من الخادم

الملخص

#include <libmemcached/memcached.h>
قم بالترجمة والربط مع -lmemcached
memcached_result_st *memcached_fetch_result(memcached_st *ptr, memcached_result_st *result, memcached_return_t *error)
char *memcached_get(memcached_st *ptr, const char *key, size_t key_length, size_t *value_length, uint32_t *flags, memcached_return_t *error)
memcached_return_t memcached_mget(memcached_st *ptr, const char *const *keys, const size_t *key_length, size_t number_of_keys)
char *memcached_get_by_key(memcached_st *ptr, const char *group_key, size_t group_key_length, const char *key, size_t key_length, size_t *value_length, uint32_t *flags, memcached_return_t *error)
memcached_return_t memcached_mget_by_key(memcached_st *ptr, const char *group_key, size_t group_key_length, const char *const *keys, const size_t *key_length, size_t number_of_keys)
memcached_return_t memcached_fetch_execute(memcached_st *ptr, memcached_execute_fn *callback, void *context, uint32_t number_of_callbacks)
memcached_return_t memcached_mget_execute(memcached_st *ptr, const char *const *keys, const size_t *key_length, size_t number_of_keys, memcached_execute_fn *callback, void *context, uint32_t number_of_callbacks)
memcached_return_t memcached_mget_execute_by_key(memcached_st *ptr, const char *group_key, size_t group_key_length, const char *const *keys, const size_t *key_length, size_t number_of_keys, memcached_execute_fn *callback, void *context, uint32_t number_of_callbacks)
typedef memcached_return_t (*memcached_execute_fn)(const memcached_st *ptr, memcached_result_st *result, void *context)

الوصف

تُستخدم memcached_get() لجلب قيمة فردية من الخادم. يجب عليك تمرير مفتاح وطوله لجلب الكائن. يجب عليك توفير ثلاثة متغيرات مؤشر والتي ستعطيك حالة الكائن المُعاد. مؤشر uint32_t لاحتواء أية أعلام قمت بتخزينها مع القيمة، ومؤشر size_t والذي سيُملأ بحجم الكائن، ومؤشر memcached_return_t لاحتواء أي خطأ. سيُعاد الكائن عند النجاح وسيُعاد NULL عند الفشل. أي كائن يُعاد بواسطة memcached_get() يجب أن يُحرر بواسطة التطبيق المستدعي.

تُستخدم memcached_mget() لتحديد مفاتيح متعددة في وقت واحد. بالنسبة لعمليات المفاتيح المتعددة، يكون استخدام هذه الدالة أسرع دائمًا. تعمل هذه الدالة دائمًا بشكل غير متزامن.

لاسترجاع البيانات بعد تنفيذ ناجح لـ memcached_mget()، ستحتاج إلى استدعاء memcached_fetch_result(). يجب عليك الاستمرار في استدعاء هذه الدالة حتى تُعيد NULL (أي لا مزيد من القيم). إذا كنت بحاجة إلى الإنهاء في منتصف استدعاء memcached_mget()، يمكنك تنفيذ memcached_quit()، على الرغم من أن ذلك ليس مطلوبًا.

تُستخدم memcached_fetch_result() لجلب قيمة فردية من الخادم. يجب دائمًا استدعاء memcached_mget() قبل استخدام هذه الطريقة. يجب عليك تمرير مفتاح وطوله لجلب الكائن. يجب عليك توفير ثلاثة متغيرات مؤشر والتي ستعطيك حالة الكائن المُعاد. مؤشر uint32_t لاحتواء أية أعلام قمت بتخزينها مع القيمة، ومؤشر size_t والذي سيُملأ بحجم الكائن، ومؤشر memcached_return_t لاحتواء أي خطأ. سيُعاد الكائن عند النجاح وسيُعاد NULL عند الفشل. تُعاد MEMCACHED_END بواسطة قيمة *error عندما تكون جميع الكائنات التي تم العثور عليها قد أُعيدت. القيمة النهائية عند MEMCACHED_END هي null.

تُستخدم memcached_fetch_result() لإعادة بنية memcached_result_st من خادم memcached. كائن النتيجة متوافق مع التغييرات المستقبلية على الخادم. لمزيد من المعلومات يرجى الرجوع إلى مساعدة memcached_result_st. ستخصص هذه الدالة بنية نتيجة ديناميكيًا لك إذا لم تمرر واحدة إلى الدالة.

memcached_fetch_execute() هي دالة استدعاء عكسي لمجموعات النتائج. بدلاً من إعادة النتائج إليك للمعالجة، فإنها تمرر كل مجموعة من مجموعات النتائج إلى قائمة الدوال التي توفرها. تمرر إلى الدالة memcached_st يمكن استنساخها للاستخدام في الدالة المستدعاة (لا يمكن استخدامها مباشرة). كما تمرر مجموعة نتيجة لا تحتاج إلى تحرير. أخيرًا تمرر "سياقًا". هذا مجرد مؤشر إلى مرجع ذاكرة توفره للدالة المستدعاة. حاليًا يتم تمرير قيمة واحدة فقط إلى كل استدعاء دالة. في المستقبل سيكون هناك خيار للسماح بأن يكون هذا مصفوفة.

memcached_mget_execute() و memcached_mget_execute_by_key() مشابهتان لـ memcached_mget()، لكنها قد تُشغّل دوال الاستدعاء العكسي الموفرة مع مجموعات النتائج أثناء إرسال الاستعلامات. إذا حاولت تنفيذ multiget كبير جدًا باستخدام memcached_mget() فقد تواجه حالة جمود في نواة نظام التشغيل (ستفشل في كتابة البيانات إلى المقبس لأن مخزن الإدخال المؤقت ممتلئ). تحل memcached_mget_execute() هذه المشكلة بمعالجة بعض النتائج قبل الاستمرار في إرسال الطلبات. يرجى ملاحظة أن هذه الدالة متاحة فقط في البروتوكول الثنائي.

تتصرف memcached_get_by_key() و memcached_mget_by_key() بطبيعة مشابهة لـ memcached_get() و memcached_mget(). الفرق هو أنها تأخذ مفتاحًا رئيسيًا يُستخدم لتحديد الخادم الذي تم تخزين كائن فيه إذا تم استخدام تقسيم المفاتيح للتخزين.

جميع الدوال المذكورة أعلاه لم يتم اختبارها عند تعيين MEMCACHED_BEHAVIOR_USE_UDP. سيؤدي تنفيذ أي من هذه الدوال مع تفعيل هذا السلوك إلى إعادة MEMCACHED_NOT_SUPPORTED، أو بالنسبة لتلك الدوال التي لا تُعيد memcached_return_t، سيتم تعيين معامل الخطأ إلى MEMCACHED_NOT_SUPPORTED.

القيمة المُعادة

ستُعيد memcached_get() NULL عند الخطأ. يجب عليك النظر إلى قيمة الخطأ لتحديد ما كان الخطأ الفعلي.

تُعيد memcached_fetch_execute() MEMCACHED_SUCCESS إذا كانت جميع المفاتيح ناجحة. ستُعاد MEMCACHED_NOTFOUND إذا لم يتم العثور على أي مفاتيح على الإطلاق.

تُعيّن memcached_fetch_result() الخطأ إلى MEMCACHED_END عند الانتهاء الناجح. ستُعاد MEMCACHED_NOTFOUND إذا لم يتم العثور على أي مفاتيح على الإطلاق.
