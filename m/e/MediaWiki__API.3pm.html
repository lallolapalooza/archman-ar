MediaWiki__API(3pm)

الحزمة: perl-mediawiki-api

التراخيص: GPL3

MediaWiki::API(3) User Contributed Perl Documentation MediaWiki::API(3)

الاسم

MediaWiki::API - يوفر واجهة Perl لواجهة برمجة تطبيقات MediaWiki (https://www.mediawiki.org/wiki/API)

الإصدار

الإصدار 0.52

الملخص

توفر هذه الوحدة واجهة بين Perl وواجهة برمجة تطبيقات MediaWiki (https://www.mediawiki.org/wiki/API) مما يسمح بإنشاء نصوص برمجية لأتمتة التحرير واستخراج البيانات من مواقع مدعومة بـ MediaWiki مثل Wikipedia.

use MediaWiki::API;
my $mw = MediaWiki::API->new();
$mw->{config}->{api_url} = 'https://en.wikipedia.org/w/api.php';
# تسجيل الدخول إلى الويكي
$mw->login( { lgname => 'username', lgpassword => 'password' } )
  || die $mw->{error}->{code} . ': ' . $mw->{error}->{details};
# الحصول على قائمة المقالات في فئة
my $articles = $mw->list ( {
  action => 'query',
  list => 'categorymembers',
  cmtitle => 'Category:Perl',
  cmlimit => 'max' } )
  || die $mw->{error}->{code} . ': ' . $mw->{error}->{details};
# وطباعة عناوين المقالات
foreach (@{$articles}) {
    print "$_->{title}\n";
}
# الحصول على معلومات المستخدم
my $userinfo = $mw->api( {
  action => 'query',
  meta => 'userinfo',
  uiprop => 'blockinfo|hasmsg|groups|rights|options|editcount|ratelimits' } );
  ...

الوظائف

MediaWiki::API->new( $config_hashref )

يعيد كائن واجهة برمجة تطبيقات MediaWiki. يمكنك تمرير تكوين كـ hashref عند استدعاء new، أو تعيين التكوين لاحقًا. عند إنشاء كائن جديد، يتم تعيين القيم الافتراضية لأقصى تأخير وإعادة المحاولة.

my $mw = MediaWiki::API->new( { api_url => 'https://en.wikipedia.org/w/api.php' }  );

خيارات التكوين هي:

  • api_url = 'مسار api.php الخاص بـ mediawiki'.
  • files_url = 'عنوان URL الأساسي للملفات'. (مطلوب إذا أعادت الواجهة عنوان URL نسبي للصور مثل /images/a/picture.jpg)
  • upload_url = 'https://en.wikipedia.org/wiki/Special:Upload'. (مسار صفحة التحميل الخاصة المطلوبة إذا كنت ترغب في تحميل الصور)
  • on_error = مرجع دالة للاتصال به إذا حدث خطأ في الوحدة.
  • use_http_get = منطقي 0 أو 1 (الافتراضي 0). إذا تم تعيينه إلى 1، ستستخدم وحدة Perl طريقة http GET للوصول إلى الواجهة. افتراضيًا، تستخدم طريقة POST. لاحظ أن الوحدة ستظل تستخدم POST لمكالمات الواجهة التي تتطلب POST بغض النظر عن قيمة خيار التكوين هذا. حاليًا، تعمل الإجراءات التالية مع GET: query، logout، paraminfo - انظر تكوين get_actions أدناه.
  • get_actions = Hashref (الافتراضي { 'query' => 1, 'logout' => 1, 'paraminfo' => 1 } ). يحتوي هذا على إجراءات الواجهة التي تدعمها طريقة http GET إذا تم تمكينها. قد تحتوي بعض الويكيات على امتدادات تضيف المزيد من الوظائف التي تعمل مع طلب http GET. إذا كان الأمر كذلك، يمكنك إضافة إجراءات حسب الحاجة.
  • retries = قيمة عدد صحيح؛ عدد مرات إعادة إرسال طلب واجهة برمجة التطبيقات إذا حدث خطأ في HTTP أو خطأ في فك ترميز JSON. الافتراضي 0 (حاول مرة واحدة فقط - لا تعيد المحاولة). إذا تم تعيين max_retries إلى 4، وكان الويكي معطلاً، فلن يتم الإبلاغ عن الخطأ إلا بعد محاولة الاتصال الخامسة.
  • retry_delay = قيمة عدد صحيح بالثواني؛ مقدار الوقت للانتظار قبل إعادة محاولة الطلب إذا حدث خطأ في HTTP أو خطأ في فك ترميز JSON.
  • max_lag = قيمة عدد صحيح بالثواني؛ تعمل Wikipedia على مجموعة من قواعد البيانات وبالتالي فإن معدلات التحرير العالية تتسبب في تأخر الخوادم التابعة. إذا تم تعيين خيار التكوين هذا، فإذا كان التأخير أكثر من قيمة max_lag، فستنتظر الواجهة قبل إعادة محاولة الطلب. 5 هي قيمة موصى بها. يمكن العثور على مزيد من المعلومات حول هذا الموضوع على https://www.mediawiki.org/wiki/Manual:Maxlag_parameter. لاحظ أن خيار التكوين يتضمن شرطة سفلية لذا قم بمطابقة مخطط تسمية خيارات التكوين الأخرى.
  • max_lag_delay = قيمة عدد صحيح بالثواني؛ يحدد خيار التكوين هذا التأخير للانتظار قبل إعادة محاولة الطلب عندما يبلغ الخادم عن تأخير يزيد عن قيمة max_lag. القيمة الافتراضية هي 5 إذا كنت تستخدم تكوين max_lag.
  • max_lag_retries = قيمة عدد صحيح؛ عدد مرات إعادة إرسال طلب واجهة برمجة التطبيقات إذا أبلغ الخادم عن تأخير يزيد عن قيمة max_lag. إذا تم الوصول إلى الحد الأقصى لعدد مرات إعادة المحاولة، يتم إرجاع خطأ. تعيين هذا إلى قيمة سالبة مثل -1 يعني إعادة إرسال الطلب حتى يكون max_lag الخاص بالخوادم أقل من الحد أو يحدث خطأ آخر. الافتراضي 4.
  • no_proxy = منطقي؛ تم التعيين إلى 1 لتعطيل استخدام أي وكيل تم تعيينه في البيئة. لاحظ افتراضيًا إذا قمت بتعيين متغيرات بيئة الوكيل، فستحاول الوحدة استخدامها. تمت إضافة هذه الميزة في الإصدار 0.29. الإصدارات الأقدم من هذا تتجاهل أي إعدادات وكيل، ولكن يمكنك تعيينها بنفسك عن طريق استدعاء MediaWiki::API->{ua}->env_proxy() بعد إنشاء مثيل جديد لفئة API. يمكن العثور على مزيد من المعلومات حول env_proxy على http://search.cpan.org/~gaas/libwww-perl-5.834/lib/LWP/UserAgent.pm#Proxy_attributes

مثال لتكوين on_error يمكن أن يكون شيئًا كهذا:

$mw->{config}->{on_error} = \&on_error;
sub on_error {
  print "Error code: " . $mw->{error}->{code} . "\n";
  print $mw->{error}->{stacktrace}."\n";
  die;
}

يتم تخزين الأخطاء في $mw->{error}->{code} مع مزيد من المعلومات في $mw->{error}->{details}. $mw->{error}->{stacktrace} تتضمن التفاصيل وتتبع المكدس لتحديد مكان نشوء أي مشاكل (في بعض التعليمات البرمجية التي تستخدم هذه الوحدة على سبيل المثال).

رموز الأخطاء هي كما يلي:

  • ERR_NO_ERROR = 0 (لا يوجد خطأ)
  • ERR_CONFIG = 1 (خطأ في التكوين)
  • ERR_HTTP = 2 (خطأ اتصال متعلق بـ http)
  • ERR_API = 3 (خطأ تم إرجاعه بواسطة واجهة برمجة تطبيقات MediaWiki)
  • ERR_LOGIN = 4 (خطأ في تسجيل الدخول إلى MediaWiki)
  • ERR_EDIT = 5 (خطأ في وظيفة التحرير)
  • ERR_PARAMS = 6 (خطأ في المعلمات التي تم تمريرها إلى دالة مساعدة)
  • ERR_UPLOAD = 7 (خطأ في مرافق تحميل الملفات)
  • ERR_DOWNLOAD = 8 (خطأ في تنزيل ملف)

معلمات وكائنات أخرى مفيدة في كائن MediaWiki::API هي:

  • MediaWiki::API->{ua} = كائن LWP::UserAgent. يمكنك تعديل هذا للحصول على أو تعديل ملفات تعريف الارتباط (MediaWiki::API->{ua}->cookie_jar) أو لتغيير سلسلة UserAgent التي ترسلها وحدة Perl هذه (MediaWiki::API->{ua}->agent)
  • MediaWiki::API->{response} = آخر كائن استجابة تم إرجاعه بواسطة LWP::UserAgent بعد طلب الواجهة.

MediaWiki::API->login( $query_hashref )

يسجل الدخول إلى MediaWiki. المعلمات هي تلك المستخدمة بواسطة واجهة برمجة تطبيقات MediaWiki (https://www.mediawiki.org/wiki/API:Login). يعيد hashref مع بعض تفاصيل تسجيل الدخول، أو undef عند فشل تسجيل الدخول. إذا قامت MediaWiki بإرسال طلب LoginToken، تتم محاولة تسجيل الدخول مرة أخرى، ولكن مع الرمز المميز المرسل من تسجيل الدخول الأولي. يتم تخزين الأخطاء في MediaWiki::API->{error}->{code} و MediaWiki::API->{error}->{details}.

my $mw = MediaWiki::API->new( { api_url => 'https://en.wikipedia.org/w/api.php' }  );
# تسجيل الدخول إلى الويكي
$mw->login( {lgname => 'username', lgpassword => 'password' } )
  || die $mw->{error}->{code} . ': ' . $mw->{error}->{details};

MediaWiki::API->api( $query_hashref, $options_hashref )

استدعاء واجهة برمجة تطبيقات MediaWiki. يتم تمرير المعلمات كـ hashref والتي يتم وصفها في صفحة واجهة برمجة تطبيقات MediaWiki (https://www.mediawiki.org/wiki/API). يعيد hashref مع نتائج المكالمة أو undef عند الفشل مع رمز الخطأ والتفاصيل المخزنة في MediaWiki::API->{error}->{code} و MediaWiki::API->{error}->{details}. تستخدم MediaWiki::API الوحدة LWP::UserAgent لإرسال طلبات http إلى واجهة برمجة تطبيقات MediaWiki. بعد أي مكالمة واجهة برمجة تطبيقات، يكون كائن الاستجابة الذي تم إرجاعه بواسطة LWP::UserAgent متاحًا في $mw->{response}. لن تقوم هذه الوظيفة بتعديل hashref الاستعلام المدخل بأي شكل من الأشكال.

binmode STDOUT, ':utf8';
# الحصول على اسم الموقع
if ( my $ref = $mw->api( { action => 'query', meta => 'siteinfo' } ) ) {
  print $ref->{query}->{general}->{sitename};
}
# قائمة عناوين "ألبرت أينشتاين" بلغات مختلفة.
my $titles
= $mw->api( { action => 'query', titles => 'Albert Einstein', prop => 'langlinks', lllimit => 'max' } ) || die $mw->{error}->{code} . ': ' . $mw->{error}->{details}; my ($pageid,$langlinks) = each ( %{ $titles->{query}->{pages} } ); foreach ( @{ $langlinks->{langlinks} } ) { print "$_->{'*'}\n"; }

تستخدم واجهة برمجة تطبيقات MediaWiki ترميز UTF-8 ويتم ترميز أي معلمات نصية ذات 8 بت تلقائيًا بواسطة استدعاء واجهة برمجة التطبيقات. إذا كانت المعلمات الخاصة بك بالفعل بتنسيق UTF-8، فسيتم اكتشاف ذلك وسيتم تخطي الترميز. إذا كانت معلماتك تحتوي لسبب ما على بيانات UTF-8 ولكن لم يتم تعيين علامة UTF-8 (أي لم تستخدم `use utf8;` pragma)، فيجب عليك منع إعادة الترميز عن طريق تمرير الخيار `skip_encoding => 1` في `<span class="Li">$options_hash</span>`. على سبيل المثال:

my $mw = MediaWiki::API->new();
$mw->{config}->{api_url} = 'https://fr.wiktionary.org/w/api.php';
my $query = {action => 'query',
  list => 'categorymembers',
  cmlimit => 'max'};
$query->{cmtitle} ="Cat\x{e9}gorie:moyen_fran\x{e7}ais"; # latin1 string
$mw->list ( $query ); # ok 
$query->{cmtitle} = "Cat". pack("U", 0xe9)."gorie:moyen_fran".pack("U",0xe7)."ais"; # unicode string
$mw->list ( $query ); # ok
$query->{cmtitle} ="Cat\x{c3}\x{a9}gorie:moyen_fran\x{c3}\x{a7}ais";  # unicode data without utf-8 flag
# $mw->list ( $query ); # NOT OK
$mw->list ( $query, {skip_encoding => 1} ); # ok

إذا كنت تقوم باستدعاء وظيفة واجهة برمجة تطبيقات تتطلب تحميل ملف، على سبيل المثال `import` أو `upload`، فحدد الملف المراد تحميله كمصفوفة تحتوي على اسم الملف المحلي. قد تقوم واجهة برمجة التطبيقات بإرجاع تحذير، على سبيل المثال للقول بأن الملف مكرر. لتجاهل التحذيرات وفرض التحميل، استخدم `ignorewarnings => 1`. يمكن العثور على جميع المعلمات كما هو الحال مع كل شيء آخر في صفحة واجهة برمجة تطبيقات MediaWiki.

$mw->api( {
  action => 'import',
  xml => ['wiki_dump.xml']
 } );
$mw->api( {
  action => 'upload',
  filename => 'test.png',
  comment => 'a test image',
  file => ['test.png'],
 } );

يمكنك أيضًا توفير البيانات المراد تحميلها مباشرة، إذا كنت ترغب في قراءة البيانات بنفسك. في هذه الحالة، قم بتوفير مصفوفة بثلاث معلمات، تبدأ بـ `undef`، متبوعة باسم الملف، ثم زوج `Content => $data` الذي يحتوي على البيانات.

$mw->api( {
  action => 'import',
  xml => [ undef, 'wiki_dump.xml', Content => $data ]
 } );
$mw->api( {
  action => 'upload',
  filename => 'test.png',
  comment => 'a test image',
  file => [ undef, 'test.png', Content => $data ],
 } );

MediaWiki::API->logout()

تسجيل خروج المستخدم الحالي ومسح ملفات تعريف الارتباط ورموز التعديل المرتبطة.

MediaWiki::API->edit( $query_hashref, $options_hashref )

وظيفة مساعدة لإجراء التعديلات باستخدام واجهة برمجة تطبيقات MediaWiki. يتم تمرير المعلمات كـ hashref والتي تم وصفها في صفحة تعديل واجهة برمجة تطبيقات MediaWiki (https://www.mediawiki.org/wiki/API:Changing_wiki_content). لاحظ أنك بحاجة إلى `$wgEnableWriteAPI = true` في LocalSettings.php الخاص بك لاستخدام هذه الميزات. ستقوم هذه الوظيفة بتعديل hashref المدخل.

حاليا

  • إنشاء/تعديل الصفحات (Mediawiki >= 1.13)
  • نقل الصفحات (Mediawiki >= 1.12)
  • التراجع (Mediawiki >= 1.12)
  • حذف الصفحات (Mediawiki >= 1.12)
  • تحميل الصور (Mediawiki >= 1.16)
  • استيراد الصفحات (Mediawiki >= 1.15)
  • (إلغاء) حماية الصفحات (Mediawiki >= 1.12)
  • (إلغاء) حظر المستخدمين (Mediawiki >= 1.12)
  • (إلغاء) مراقبة صفحة (Mediawiki >= 1.18)
  • إرسال بريد إلكتروني للمستخدم (Mediawiki >= 1.14)
  • الدوريات التغييرات (Mediawiki >= 1.14)

مدعومة عبر هذا الاستدعاء. استخدم هذا الاستدعاء لتعديل الصفحات دون الحاجة إلى القلق بشأن الحصول على رمز تعديل من واجهة برمجة التطبيقات أولاً. ستقوم الوظيفة بتخزين رموز التعديل مؤقتًا لتسريع التعديلات المستقبلية.

يعيد hashref مع نتائج الاستدعاء أو `undef` عند الفشل مع تخزين رمز الخطأ والتفاصيل في `MediaWiki::API->{error}->{code}` و `MediaWiki::API->{error}->{details}`.

يحتوي hashref الخيارات حاليًا على معلمة اختيارية واحدة (`skip_encoding => 1`). تم وصف ذلك أعلاه في وثائق استدعاء `MediaWiki::API->api`.

فيما يلي بعض مقتطفات الكود كمثال. المثال الأول لإضافة بعض النصوص إلى صفحة موجودة (إذا لم تكن الصفحة موجودة فلن يحدث شيء). لاحظ أنه تم حفظ الطابع الزمني للمراجعة التي نغيرها. يسمح لنا هذا بتجنب تعارضات التعديل. يتم تمرير القيمة مرة أخرى إلى وظيفة التعديل، وإذا قام شخص ما بتحرير الصفحة في هذه الأثناء، فسيتم إرجاع خطأ.

my $pagename = "Wikipedia:Sandbox";
my $ref = $mw->get_page( { title => $pagename } );
unless ( $ref->{missing} ) {
  my $timestamp = $ref->{timestamp};
  $mw->edit( {
    action => 'edit',
    title => $pagename,
    basetimestamp => $timestamp, # to avoid edit conflicts
    text => $ref->{'*'} . "\nAdditional text" } )
    || die $mw->{error}->{code} . ': ' . $mw->{error}->{details};
}

يحذف الكود التالي صفحة بالاسم `"DeleteMe"`. يمكنك تحديد سبب للحذف، وإلا فسيتم استخدام سبب تم إنشاؤه.

# delete a page
$mw->edit( {
  action => 'delete', title => 'DeleteMe', reason => 'no longer needed' } ) 
  || die $mw->{error}->{code} . ': ' . $mw->{error}->{details};

ينقل هذا الكود صفحة من `MoveMe` إلى `MoveMe2`.

# move a page
$mw->edit( {
  action => 'move', from => 'MoveMe', to => 'MoveMe2' } )
  || die $mw->{error}->{code} . ': ' . $mw->{error}->{details};

تقوم هذه المقتطفات البرمجية التالية بالتراجع عن تعديل واحد أو أكثر من قبل المستخدم `MrVandal`. إذا لم يكن المستخدم هو آخر محرر للصفحة، فسيتم إرجاع خطأ. إذا لم يتم تمرير مستخدم، فسيتم التراجع عن تعديلات آخر من قام بتغيير الصفحة.

$mw->edit( {
  action => 'rollback', title => 'Sandbox', user => 'MrVandal' } )
  || die $mw->{error}->{code} . ': ' . $mw->{error}->{details};

MediaWiki::API->get_page( $params_hashref )

وظيفة مساعدة للحصول على أحدث محتويات الصفحة (وغيرها من البيانات الوصفية) لصفحة. تستدعي وظيفة واجهة برمجة التطبيقات ذات المستوى الأدنى باستعلام المراجعات للحصول على أحدث مراجعة.

# get some page contents
my $page = $mw->get_page( { title => 'Main Page' } );
# print page contents
print $page->{'*'};

يعيد hashref يحتوي على المفاتيح التالية أو `undef` عند حدوث خطأ. إذا كانت الصفحة مفقودة، فسيحتوي hashref الذي تم إرجاعه فقط على `ns` و `title` ومفتاح يسمى `"missing"`.

  • `'*'` - محتويات الصفحة
  • `'pageid'` - معرف الصفحة
  • `'revid'` - معرف المراجعة
  • `'timestamp'` - الطابع الزمني للمراجعة
  • `'user'` - المستخدم الذي قام بالمراجعة
  • `'title'` - عنوان الصفحة
  • `'ns'` - مساحة الاسم التي توجد بها الصفحة
  • `'size'` - حجم الصفحة بالبايت

يمكن قراءة المعلومات الكاملة حول هذه على (https://www.mediawiki.org/wiki/API:Query_-_Properties#revisions_.2F_rv)

MediaWiki::API->list( $query_hashref, $options_hashref )

وظيفة مساعدة للحصول على قوائم باستخدام واجهة برمجة تطبيقات MediaWiki. يتم تمرير المعلمات كـ hashref والتي تم وصفها في صفحة تعديل واجهة برمجة تطبيقات MediaWiki (https://www.mediawiki.org/wiki/API:Query_-_Lists). تقوم هذه الوظيفة بتعديل query_hashref المدخل.

ستعيد هذه الوظيفة مرجعًا لمصفوفة من الـ hashes أو `undef` عند الفشل. تتعامل مع الحصول على قوائم من بيانات واجهة برمجة تطبيقات MediaWiki، وتستمر في الطلب باستخدام اتصال آخر إذا لزم الأمر. يحتوي hashref الخيارات حاليًا على ثلاث معلمات:

  • `max => value`
  • `hook => \&function_hook`
  • `skip_encoding => 1`

تحدد قيمة `max` الحد الأقصى لـ "الاستعلامات" التي سيتم استخدامها لسحب البيانات. على سبيل المثال، الحد الافتراضي لكل استعلام هو 10 عناصر، ولكن يمكن زيادته إلى 500 للمستخدمين العاديين وإلى أعلى للمسؤولين والروبوتات. إذا تم رفع الحد إلى 500 وتم تعيين `max` إلى 2، فسيتم إرجاع 1000 نتيجة كحد أقصى.

إذا كنت ترغب في معالجة قوائم كبيرة، على سبيل المثال المقالات في فئة كبيرة، يمكنك تمرير وظيفة ربط، والتي سيتم تمرير مرجع لمصفوفة من النتائج لكل اتصال استعلام إليها.

تعمل معلمة `skip_encoding` كما هو موضح أعلاه في وثائق استدعاء `MediaWiki::API->api`.

binmode STDOUT, ':utf8';
# process the first 400 articles in the main namespace in the category "Surnames".
# get 100 at a time, with a max of 4 and pass each 100 to our hook.
$mw->list ( { action => 'query',
              list => 'categorymembers',
              cmtitle => 'Category:Surnames',
              cmnamespace => 0,
              cmlimit=>'100' },
            { max => 4, hook => \&print_articles } )
|| die $mw->{error}->{code} . ': ' . $mw->{error}->{details};
# print the name of each article
sub print_articles {
  my ($ref) = @_;
  foreach (@$ref) {
    print "$_->{title}\n";
  }
}

MediaWiki::API->upload( $params_hashref )

تم إيقاف هذه الوظيفة. للتحميل على إصدارات mediawiki 1.16 أو أحدث، يوصى باستخدام `MediaWiki::API->edit` أو `MediaWiki::API->api` مباشرة، والتي تتمتع بمعالجة أفضل للأخطاء، وتدعم تحميل الملفات عن طريق تمرير اسم ملف فقط.

وظيفة لتحميل الملفات إلى MediaWiki. لا تستخدم هذه الوظيفة واجهة برمجة تطبيقات MediaWiki حاليًا حيث لم يتم بعد تنفيذ دعم تحميل الملفات. بدلاً من ذلك، تقوم بالتحميل باستخدام صفحة `Special:Upload`، وبذلك يلزم إعداد تكوين إضافي.

my $mw = MediaWiki::API->new( {
 api_url => 'https://en.wikipedia.org/w/api.php' }  );
# configure the special upload location.
$mw->{config}->{upload_url} = 'https://en.wikipedia.org/wiki/Special:Upload';

ثم يتم استدعاء وظيفة التحميل على النحو التالي

# upload a file to MediaWiki
open FILE, "myfile.jpg" or die $!;
binmode FILE;
my ($buffer, $data);
while ( read(FILE, $buffer, 65536) )  {
  $data .= $buffer;
}
close(FILE);
$mw->upload( { title => 'file.jpg',
               summary => 'This is the summary to go on the Image:file.jpg page',
               data => $data } ) || die $mw->{error}->{code} . ': ' . $mw->{error}->{details};

فحص الأخطاء محدود. لاحظ أيضًا أن الوحدة ستفرض تحميل الملف، متجاهلة أي تحذيرات لحجم الملف أو الكتابة فوق ملف قديم.

الاسم

MediaWiki::API->download( $params_hashref )

الإصدار

وظيفة لتنزيل الصور/الملفات من MediaWiki. قد تحتاج إلى تكوين عنوان URL للملف إذا أعادت واجهة برمجة التطبيقات عنوان URL نسبيًا.

my $mw = MediaWiki::API->new( {
  api_url => 'https://www.exotica.org.uk/mediawiki/api.php' }  );
# تكوين عنوان URL للملف. لا تحتاج ويكيبيديا إلى هذا ولكن تحتاج ويكي ExoticA إليه.
$mw->{config}->{files_url} = 'https://www.exotica.org.uk';

يتم بعد ذلك استدعاء وظيفة التنزيل على النحو التالي

my $file = $mw->download( { title => 'Image:Mythic-Beasts_Logo.png'} )
  || die $mw->{error}->{code} . ': ' . $mw->{error}->{details};

إذا لم يكن الملف موجودًا (على الويكي) يتم إرجاع سلسلة فارغة. إذا لم يكن من الممكن تنزيل الملف يتم إرجاع undef.
