Module__Install(3pm)

الحزمة: perl-module-install

التراخيص: PerlArtistic, GPL

version

version '0.01';

تُستخدم "version" لتحديد إصدار التوزيع، مما يميزه عن إصدار أي وحدة داخل التوزيع.

ومع ذلك، في معظم الحالات، تريد أن يتطابق مع إصدار الوحدة الرئيسية للتوزيع، وهو ما يمكنك القيام به باستخدام "version_from".

version_from

version_from 'lib/My/Module.pm';

يحصل "version_from" على إصدار التوزيع من ملف محدد موجود في حزمة التوزيع. في معظم الأحيان، يتم ذلك من الوحدة الرئيسية.

"version_from" سيبحث عن أول تعيين لـ $VERSION ويستخدم نفس القيمة، باستخدام تقنية متوافقة مع أدوات مسح إصدارات الوحدات الأخرى.

license

license 'perl';

يحدد "license" الترخيص للتوزيع.

في معظم الحالات، ستكون هذه القيمة 'perl'، مما يعني "نفس ترخيص Perl نفسه". تشمل القيم الأخرى المسموح بها 'gpl'، 'lgpl'، 'bsd'، 'MIT'، و 'artistic'.

تُعتبر هذه القيمة دائمًا ملخصًا، ومن الطبيعي تضمين ملف LICENSE في التوزيع، والذي يحتوي على الترخيص الكامل للتوزيع.

كما يُذكر أنه إذا كان التوزيع مخططًا لرفعه إلى CPAN، فيجب أن يكون بترخيص مفتوح وفقًا لـ OSI. البرامج التجارية غير مسموح بها في CPAN.

license_from

license_from 'lib/My/Module.pm';

يحصل "license_from" على ترخيص التوزيع من ملف محدد موجود في حزمة التوزيع. في معظم الأحيان، يتم ذلك من الوحدة الرئيسية.

"license_from" سيبحث في الملف المحدد عن قسم ترخيص أو حقوق نشر ويبحث عن سلاسل مختلفة لتحديد النوع العام للترخيص.

حاليًا، يدعم فقط القيم الست المذكورة في ملخص "license".

perl_version

perl_version '5.006';

يُستخدم "perl_version" لتحديد الحد الأدنى لإصدار تفسير Perl الذي يتطلبه التوزيع الخاص بك.

عند تحديد الإصدار، يجب عليك محاولة استخدام سلسلة الإصدار القياسية. أجزاء إصدار Perl هي بثلاثة أرقام، لذا فإن الاعتماد على Perl 5.6 سيتم تحويله إلى '5.006' وإصدار Perl 5.10.2 سيتم تحويله إلى '5.010002'.

perl_version_from

perl_version_from 'lib/My/Module.pm'

يحصل "perl_version_from" على الحد الأدنى لإصدار تفسير Perl من ملف محدد موجود في حزمة التوزيع. في معظم الأحيان، يتم ذلك من الوحدة الرئيسية.

يتم اكتشاف الحد الأدنى للإصدار بالبحث عن استدعاءات ماكرو "use 5.xxx" في ملف الوحدة.

recommends

recommends 'Text::CSV_XS' => '0.50'

يشير "recommends" إلى وحدة تشغيل اختيارية توفر وظائف إضافية. التبعيات الموصى بها ليست ضرورية لبناء أو اختبار التوزيع الخاص بك، ولكنها تعتبر "مفيدة جدًا".

كما هو الحال مع "requires"، تعتمد التبعية على وحدة، وليس على توزيع. يشير الإصدار صفر إلى أن أي إصدار من الوحدة موصى به.

requires

requires 'List::Util' => 0;
requires 'LWP'        => '5.69';

يشير "requires" إلى تبعية تشغيل عادية لتوزيعك على وحدة أخرى. ستتضمن معظم التوزيعات أحد هذه التعليمات، مشيرة إلى الوحدات اللازمة لـ CPAN (أو غيرها).

يمكن التعبير عن تبعية "requires" على أنها "إذا كنت تريد تثبيت واستخدام هذا التوزيع، فيجب عليك أولاً تثبيت هذه الوحدات.".

لاحظ أن التبعية تستند إلى وحدة، وليس على توزيع. هذا يضمن أن تبعيتك صحيحة دائمًا، حتى لو انتقلت الوحدة أو اختلطت بتوزيع آخر، وهو أمر ممكن في بعض الأحيان.

تشير تبعية الإصدار صفر إلى أن أي إصدار من الوحدة كافٍ. يجب عمومًا اقتباس الإصدارات للوضوح.

test_requires

test_requires 'Test::More' => '0.47';

يشير "test_requires" إلى تبعية نص اختبار لتوزيعك. تنسيق المواصفات مطابق لتنسيق "requires".

يختلف "test_requires" عن "requires" في أنه يشير إلى وحدة مطلوبة فقط أثناء اختبار التوزيع (غالبًا فترة زمنية قصيرة) ولكنها غير مطلوبة بعد تثبيت التوزيع.

يُستخدم "testrequires" للسماح للمثبت بالمرونة في كيفية توفير الوحدة، وللسماح للحزم (Debian، FreeBSD، ActivePerl إلخ) بالاحتفاظ فقط بالتبعيات اللازمة للتشغيل في الوقت الحقيقي.

يُستخدم "include" أحيانًا مع "test_requires" لتضمين وحدة تم اختبارها جيدًا في حزمة التوزيع نفسها حتى يقوم المستخدم بتثبيتها مباشرة من CPAN.

configure_requires

configure_requires 'File::Spec' => '0.80';

يشير "configure_requires" إلى تبعية تهيئة لتوزيعك. تنسيق المواصفات مطابق لتنسيق "requires".

يُستخدم "configure_requires" لتجنب معضلة كيفية استخدام وحدة CPAN في Makefile.PL الخاص بك، عندما تحتاج إلى تحميل Makefile.PL (وبالتالي وحدة CPAN) لمعرفة أنك بحاجة إليها.

تقليديًا، لم يكن من الممكن كسر هذا، وبالتالي اضطرت نصوص Makefile.PL إلى الاعتماد على مناهج القاسم المشترك، أو توفير تلك التبعيات مع الأمر "include".

ينشئ "configure_requires" سجلًا في المفتاح الخاص "configure_requires: في ملف META.yml للتوزيع.

على الرغم من أن جزءًا كبيرًا من META.yml يعتبر استشاريًا، إلا أن عميل CPAN سيعامل محتويات "configure_requires: بشكل قاطع وسيتم تثبيت الوحدات المذكورة قبل تشغيل "Makefile.PL" (الذي يحدد منه التبعيات الأخرى لاحقًا).

يرجى ملاحظة أن الدعم لـ "configure_requires: في عملاء CPAN ليس مكتملًا بنسبة 100٪ عند كتابة هذا ولا يمكن الاعتماد عليه بعد.

نظرًا لأن Module::Install نفسه يدعم فقط 5.005، فسيتم إضافة تعليمات "configure_requires( perl => '5.005' ); " مكافئة بصمت إلى توزيعك.

requires_external_bin

requires_external_bin 'cvs';

كجزء من دوره كلغة ربط رئيسية، تقوم العديد من وحدات Perl بتشغيل أوامر أو برامج على نظام المضيف.

يُستخدم "requires_external_bin" للتحقق من توفر أمر معين على نظام المضيف.

بينما تكون وحدة Perl المفقودة قابلة للحل في وقت الإنشاء، فإن البرنامج الثنائي الخارجي المفقود ليس كذلك، وبالتالي سيتوقف تشغيل "Makefile.PL" بنتيجة 'NA' (غير قابل للتطبيق).

في المستقبل، سيضيف هذا الأمر أيضًا معلومات إضافية إلى بيانات التعريف الخاصة بالتوزيع، مما يسمح لبرامج التعبئة التلقائية الخاصة بأنظمة التشغيل المحددة باكتشاف حزم غير Perl المطلوبة تلقائيًا كتبعيات.

install_script

# The following are equivalent
install_script 'script/scriptname'

يوفر "install_script" دعمًا لتثبيت البرامج النصية التي سيتم تحويلها إلى أوامر في الطرفية على كل من Unix و Windows (في الحالة الأخيرة، عبر ملف .bat).

لاحظ أنه من الشائع عدم وضع .pl في نهاية هذه النصوص، حتى تشعر بأنها طبيعية أكثر عند استخدامها.

في المثال أعلاه، يمكن تشغيل البرنامج script/scriptname بعد التثبيت ببساطة عن طريق القيام بما يلي:

> scriptname
Running scriptname 0.01...
>

حسب العرف، يجب وضع البرامج النصية في مجلد /script داخل التوزيع الخاص بك. لدعم عدد أقل من أنواع الإدخال، إذا كان النص موجودًا في مجلد script، فيجب الإشارة إليه بالاسم فقط.

# The following are equivalent
install_script 'foo';
install_script 'script/foo';

no_index

no_index directory => 'examples';
no_index package   => 'DB';

غالبًا ما يوفر التوزيع نصوصًا توضيحية أو وحدات اختبار (.pm) بالإضافة إلى وحدات المكتبة الفعلية.

في معظم الحالات، لا ترغب في فهرستها في فهرس CPAN، حزم Perl الرئيسية.

sections: - title: Commands subsections: - title: no_index, black_list, remove content: |

The "no_index" command is used to indicate directories or files where there might be non-library .pm files or other files that the CPAN indexer and websites such as https://metacpan.org/ or http://search.cpan.org/ should explicitly ignore.

The most common situation is to ignore example or demo directories, but a variety of different situations may require a "no_index" entry.

Another common use for "no_index" is to prevent the PAUSE indexer complaining when your module makes changes inside a "package DB" block. This is used to interact with the debugger in some specific ways.

See the META.yml documentation for more details on what "no_index" values are allowed.

The inc, t and share (if "install_share" is used) directories are automatically "no_index"'ed for you if found and do not require an explicit command.

To summarize, if you can see it on https://metacpan.org/ or http://search.cpan.org/ and you shouldn't be able to, you need a "no_index" entry to remove it.

- title: 'installdirs, install_as_*' code: | installdirs 'site'; # the default install_as_core; # alias for installdirs 'perl' install_as_cpan; # alias for installdirs 'site' install_as_site; # alias for installdirs 'site' install_as_vendor; # alias for installdirs 'vendor' content: |

The "installdirs" and "install_as" commands specify the location where the module should be installed; this is the equivalent to ExtUtils::MakeMaker's "INSTALLDIRS" option. For almost all regular modules, the default is recommended, and need not be changed. Dual-life (core and CPAN) modules, as well as vendor-specific modules, may need to use the other options.

If unsure, do not use this option.

- title: WriteAll content: |

The "WriteAll" command is generally the last command in the file; it writes out META.yml and Makefile so the user can run the "make", "make test", "make install" install sequence.

- title: EXTENSIONS content: |

All extensions belong to the Module::Install::* namespace, and inherit from Module::Install::Base. There are three categories of extensions:

subsections: - title: Standard Extensions content: |

Methods defined by a standard extension may be called as plain functions inside Makefile.PL; a corresponding singleton object will be spawned automatically. Other extensions may also invoke its methods just like their own methods:

# delegates to $other_extension_obj->method_name(@args)
          $self->method_name(@args);

At the first time an extension's method is invoked, a POD-stripped version of it will be included under the inc/Module/Install/ directory, and becomes fixed -- i.e., even if the user had installed a different version of the same extension, the included one will still be used instead.

If the author wish to upgrade extensions in inc/ with installed ones, simply run "perl Makefile.PL" again; Module::Install determines whether you are an author by the existence of the inc/.author/ directory. End-users can reinitialize everything and become the author by typing "make realclean" and "perl Makefile.PL".

- title: Private Extensions content: |

Those extensions take the form of Module::Install::PRIVATE and Module::Install::PRIVATE::*.

Authors are encouraged to put all existing Makefile.PL magics into such extensions (e.g. Module::Install::PRIVATE for common bits; Module::Install::PRIVATE::DISTNAME for functions specific to a distribution).

Private extensions should not to be released on CPAN; simply put them somewhere in your @INC, under the "Module/Install/" directory, and start using their functions in Makefile.PL. Like standard extensions, they will never be installed on the end-user's machine, and therefore never conflict with other people's private extensions.

- title: Administrative Extensions content: |

Extensions under the Module::Install::Admin::* namespace are never included with the distribution. Their methods are not directly accessible from Makefile.PL or other extensions; they are invoked like this:

# delegates to $other_admin_extension_obj->method_name(@args)
          $self->admin->method_name(@args);

These methods only take effect during the initialization run, when inc/ is being populated; they are ignored for end-users. Again, to re-initialize everything, just run "perl Makefile.PL" as the author.

Scripts (usually one-liners in Makefile) that wish to dispatch AUTOLOAD functions into administrative extensions (instead of standard extensions) should use the Module::Install::Admin module directly. See Module::Install::Admin for details.

- title: Module::Install::WriteAll content: |

Provides the "WriteAll", which writes all the requires files, such as META.yml and Makefile.

"WriteAll" takes four optional named parameters:

"check_nmake" (defaults to true)
If true, invokes functions with the same name.

The use of this param is no longer recommended.

"inline" (defaults to false)
If true, invokes "&Inline->write" Inline modules.
"meta" (defaults to true)
If true, writes a "META.yml" file.
"sign" (defaults to false)
If true, invokes "sign" command to digitally sign erm... something.
