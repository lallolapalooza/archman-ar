mysql_select_db(3)

الحزمة: mariadb

التراخيص: GPL-2.0-only

mysql_select_db(3) MariaDB Connector/C mysql_select_db(3)

الاسم

mysql_select_db - يختار قاعدة بيانات كقاعدة افتراضية

الملخص

#include <mysql.h>
int mysql_select_db(MYSQL * mysql,
                    const char * db);

الوصف

يختار قاعدة بيانات كقاعدة افتراضية. يعيد صفر عند النجاح، وقيمة غير صفرية عند الفشل

المعلمات

•
mysql هو معرف اتصال، تم تخصيصه مسبقًا بواسطة mysql_init(3) وتم توصيله بواسطة mysql_real_connect(3).
•
db - اسم قاعدة البيانات الافتراضية

ملاحظات

•
لاسترداد اسم قاعدة البيانات الافتراضية، إما قم بتنفيذ أمر SQL SELECT DATABASE() أو استرداد القيمة عبر الدالة mariadb_get_infov(3) API.
•
يمكن أيضًا تعيين قاعدة البيانات الافتراضية بواسطة المعلمة db في mysql_real_connect(3).

أمثلة

SQL

# التبديل إلى قاعدة البيانات الافتراضية test
USE test;
# التحقق من قاعدة البيانات الافتراضية
SELECT DATABASE();
+------------+
| database() |
+------------+
| test       |
+------------+

MariaDB Connector/C

static int set_default_db(MYSQL *mysql)
{
  int rc;
  char *default_db;
  /* تغيير قاعدة البيانات الافتراضية إلى test */
  rc= mysql_select_db(mysql, "test");
  if (rc)
    return rc;  /* خطأ */
  /* الحصول على قاعدة البيانات الافتراضية */
  rc= mariadb_get_infov(mysql, MARIADB_CONNECTION_SCHEMA, &default_db);
  if (rc)
    return rc; /* خطأ */
  if (strcmp("test", default_db) != NULL)
  {
    printf("Wrong default database\n");
    return 1;
  }
  printf("Default database: %s", default_db);
  return 0;
}
