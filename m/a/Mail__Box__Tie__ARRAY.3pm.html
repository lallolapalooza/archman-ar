Mail__Box__Tie__ARRAY(3pm)

الحزمة: perl-mail-box

التراخيص: Artistic-1.0-Perl, GPL-1.0-or-later

Mail::Box::Tie::ARRAY(3) توثيق Perl من مساهمة المستخدم Mail::Box::Tie::ARRAY(3)

الاسم

Mail::Box::Tie::ARRAY - الوصول إلى مجلد رسائل موجود كمصفوفة

الوراثة

Mail::Box::Tie::ARRAY
  هو Mail::Box::Tie

الملخص

use Mail::Box::Manager;
my $mgr    = Mail::Box::Manager->new;
my $folder = $mgr->open(folder => 'inbox');
use Mail::Box::Tie::ARRAY;
tie my(@inbox), 'Mail::Box::Tie::ARRAY', $folder;
# مهمل، ولكنه يعمل أيضًا
use Mail::Box::Tie;
tie my(@inbox), 'Mail::Box::Tie', $folder;
foreach (@inbox) {print $_->short}
print $_->print foreach @inbox;
my $emails = @inbox;
print $inbox[3];
print scalar @inbox;
push @inbox, Mail::Box::Message->new(...);
delete $inbox[6];
print $inbox[0]->head->get('status');
my $folder = tied @inbox;
untie @inbox;

الوصف

بالتأكيد عندما تنظر إلى مجلد كقائمة رسائل، فمن المنطقي الوصول إلى المجلد عبر مصفوفة.

لا يتم دعم جميع عمليات المصفوفات. في الواقع، يتم تعديل معظم الوظائف التي من شأنها تقليل حجم المصفوفة بدلاً من ذلك لوضع علامة على الرسائل للحذف.

أمثلة على ما لا يمكنك فعله:

shift/unshift/pop/splice @inbox;

يمتد "الوصف" في Mail::Box::Tie.

tie @array, 'Mail::Box::Tie::ARRAY', FOLDER

» مثال: ربط مصفوفة بمجلد

my $mgr   = Mail::Box::Manager->new;
my $inbox = $mgr->new(folder => $ENV{MAIL});
tie my(@inbox), 'Mail::Box::Tie::Array', ref $inbox, $inbox;
$class->new($folder, $type)

واجهة الربط

$obj->DELETE()
وضع علامة على رسالة لإزالتها. كن حذرًا لأن الرسالة تبقى في المجلد، ولا تتم إزالتها قبل كتابة المجلد.

» مثال:

delete $inbox[5];
$inbox[5]->delete;   # نفس الشيء
$obj->FETCH($index)
الحصول على الرسالة الموجودة في الموقع المحدد في قائمة الرسائل الموجودة في هذا المجلد. سيتم إرجاع الرسائل المحذوفة كـ "undef".

» مثال:

print $inbox[3];     # الرسالة الرابعة في المجلد
print @inbox[3,0];   # الرابعة والأولى في المجلد
print $inbox[-1];    # آخر رسالة
$obj->FETCHSIZE()
إرجاع العدد الإجمالي للرسائل في المجلد. يتم استدعاء هذا عندما يتم استخدام مصفوفة المجلد في سياق قياسي، على سبيل المثال.

» مثال:

if(@inbox > 10)    # هل يحتوي على أكثر من 10 رسائل؟
my $nrmsgs = @inbox;
$obj->PUSH(@messages)
إضافة @messages إلى نهاية المجلد.

» مثال:

push @inbox, $newmsg;
$obj->STORE($index, $message)
استبدال الرسالة العشوائية غير مسموح به - القيام بذلك من شأنه أن يزعج الخيوط وما إلى ذلك. يحدث خطأ إذا حاولت القيام بذلك. الشيء الوحيد المسموح به هو تخزين رسالة في أول فهرس فارغ في نهاية المجلد (وهو ما يمكن تحقيقه أيضًا باستخدام PUSH()).

» مثال:

$inbox[8]  = $add;
$inbox[-1] = $add;
push @inbox, $add;
$obj->STORESIZE($length)
يتم تعيين جميع الرسائل خلف $length إلى نهاية المجلد ليتم حذفها.

التفاصيل

مجلد مرتبط كمصفوفة

القيود

تنفيذ هذه الوحدة "TIEARRAY", "FETCH", "STORE", "FETCHSIZE", "STORESIZE", "DELETE", "PUSH", و "DESTROY".

هذه الوحدة لا تنفذ جميع الطرق الأخرى كما هو موضح في توثيق Tie::Array، لأن مصفوفة الرسائل الحقيقية غير مسموح لها بالانكماش أو التشويه.
