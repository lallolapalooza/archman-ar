Mail__Box__Manager(3pm)

الحزمة: perl-mail-box

التراخيص: Artistic-1.0-Perl, GPL-1.0-or-later

type => $foldertypename|$foldertype
يحدد نوع المجلد. إذا لم تحدد هذا الخيار عند فتح مجلد للقراءة، يتحقق المدير من جميع أنواع المجلدات المسجلة بالترتيب لفتح المجلد. إذا فتحت مجلدًا جديدًا للكتابة، فسيكون الافتراضي هو النوع المسجل الأخير (إذا تم تحديد أكثر من نوع، فسيتم استخدام الأول في القائمة).

حالياً، الأنواع هي:

  • "mbox"
  • "mh"
  • "maildir"
  • "pop3"
  • "pop3s"
  • "imap4"
  • "imap4s"

يمكنك أيضًا استخدام أسماء مثل "pop" و "pops" و "imap" و "imaps".

» مثال: فتح مجلد عبر المدير

my $jack  = $manager->open(folder => '=jack', type => 'mbox');
my $rcvd  = $manager->open('myMail',
   type => 'Mail::Box::Mbox', access => 'rw');
my $inbox = $manager->open('Inbox')
   or die "Cannot open Inbox.\n";
my $pop   = 'pop3://myself:secret@pop3.server.com:120/x';
my $send  = $manager->open($url);
my $send  = $manager->open(folder => '/x',
  type => 'pop3', username => 'myself', password => 'secret'
  server_name => 'pop3.server.com', server_port => '120');
$obj->openFolders()
يعيد قائمة بجميع المجلدات المفتوحة.

إدارة المجلدات الموجودة

$obj->delete($foldername, %options)
يحذف المجلد المسمى. %options هي نفس الخيارات في open().

قد يستغرق حذف المجلد بعض الوقت. اعتمادًا على نوع المجلد، يجب قراءة المجلد أولاً. بالنسبة لبعض أنواع المجلدات، سيكون هذا سريعًا.

-Option   --Default
 recursive  <folder's default>
recursive => BOOLEAN
يمكن حذف بعض المجلدات بشكل متكرر فقط، بينما تتمتع مجلدات أخرى بمرونة أكبر.

نقل الرسائل إلى المجلدات

$obj->appendMessage( [$folder|$foldername], $messages, %options )
يضيف رسالة واحدة أو أكثر إلى مجلد (وبالتالي، يُعرّف على أنه appendMessages()). يمكنك تحديد $foldername أو مجلد مفتوح كوسيط أول. عندما يكون الاسم اسم مجلد مفتوح، يتم التعامل معه كما لو تم تحديد كائن المجلد، ولا يتم الوصول مباشرة إلى ملفات المجلد. يمكنك أيضًا تحديد اسم المجلد كجزء من قائمة الخيارات.

إذا تمت إضافة رسالة إلى مجلد مفتوح بالفعل، فسيتم إضافتها داخليًا فقط إلى هيكل البرنامج. لن يتم كتابة البيانات على القرص حتى يتم تنفيذ كتابة لهذا المجلد. عند إعطاء اسم مجلد غير مفتوح، يتم فتح المجلد، ويتم تخزين الرسائل على القرص، ثم يتم إغلاق المجلد.

يجب أن تكون الرسالة نسخة من Mail::Message. لا يلزم أن يتطابق نوع الرسالة الفعلي مع نوع المجلد - سيحاول المجلد حل الاختلافات بأقل قدر من فقدان المعلومات. يتم إرجاع الرسائل المحولة (كيف تم كتابتها بالفعل) كقائمة.

%options هي قائمة من المفاتيح/القيم، والتي يتم إضافتها إلى (أو الكتابة فوق) الخيارات الافتراضية لنوع المجلد المكتشف.

» مثال:

$mgr->appendMessage('=send', $message, folderdir => '/');
$mgr->appendMessage($received, $inbox->messages);
my @appended = $mgr->appendMessages($inbox->messages, folder => 'Drafts');
$_->label(seen => 1) for @appended;
$obj->copyMessage( [$folder|$foldername], @messages, %options )
ينسخ @messages من مجلد إلى آخر. إذا كان المجلد الوجهة مفتوحًا بالفعل، فسيتم استخدام Mail::Box::copyTo(). بخلاف ذلك، سيتم استدعاء Mail::Box::appendMessages().

يجب عليك تحديد اسم مجلد أو كائن مجلد كوسيط أول، أو في قائمة الخيارات. الخيارات هي نفسها التي يمكن تحديدها عند فتح مجلد.

-Option--Default
 share   false
share => BOOLEAN
محاولة مشاركة التخزين الفعلي للرسائل. قد تكون أنواع المجلدات مختلفة، ولكن هذا يعتمد على نوع المجلد الذي يتم نسخ الرسالة إليه. يتم تجاهله بصمت إذا كانت المشاركة غير ممكنة.

» مثال:

my $drafts = $mgr->open(folder => 'Drafts');
my $outbox = $mgr->open(folder => 'Outbox');
$mgr->copyMessage($outbox, $drafts->message(0));
my @messages = $drafts->message(1,2);
$mgr->copyMessage('=Trash', @messages,
   folderdir => '/tmp', create => 1);
$mgr->copyMessage($drafts->message(1), create => 1,
   folder => '=Drafts' folderdir => '/tmp');
$obj->moveMessage( [$folder|$foldername], @messages, %options )
ينقل @message من $folder إلى مجلد آخر.

تحذير: لا يحدث حذف الرسالة إلا عند إغلاق المجلد، لذلك يتم تمييز الرسالة فقط على أنها محذوفة في المجلد المصدر المفتوح.

تحذير: قد تُفقد تسميات الرسائل عند نقل رسالة من مجلد إلى آخر. يتم محاولة ترجمة التسميات، ولكن هناك اختلافات في التفسير حسب التطبيق.

$mgr->moveMessage($received, $inbox->message(1))

يعادل

 $mgr->copyMessage($received, $inbox->message(1), share => 1);
 $inbox->message(1)->delete;
-Option--Default
 share   true
share => BOOLEAN

إدارة سلاسل الرسائل

$obj->discoverThreads( [@folders], %options )
[3.012] ينشئ كائن Mail::Box::Thread::Manager يجمع سلاسل الرسائل الموجودة في مجموعة المجلدات المحددة.

سيتم تمرير جميع %options الأخرى إلى Mail::Box::Thread::Manager::new().

-Option --Default
 folders  <positional folder arguments>
folders => \@folders
كبديل للمعلمات الموضعية لتحديد @folders، يمكنك تمريرها عبر هذه المعلمة المسماة.

» مثال:

my $t2 = $mgr->discoverThreads($inbox);
my $t3 = $mgr->discoverThreads($inbox, $send);
my $t1 = $mgr->discoverThreads(folders => [ $inbox, $send ]);

الداخليات

$obj->decodeFolderURL($url)
يحاول تفكيك اسم مجلد محدد كـ $url (انظر open()) إلى خيارات منفصلة. يجب تمرير الأحرف الخاصة مثل علامة @، والنقطتين، والشرطة المائلة المستخدمة في أجزاء المستخدم أو كلمة المرور كـ $url-encoded.
$obj->toBeThreaded($folder, $messages)
يشير إلى المدير بأن جميع مديري السلاسل الذين يستخدمون المجلد المحدد يجب أن يكونوا على علم بوصول رسائل جديدة.
$obj->toBeUnthreaded($folder, $messages)
يشير إلى المدير بأن جميع مديري السلاسل الذين يستخدمون المجلد المحدد يجب أن يكونوا على علم بخروج رسائل جديدة أو على وشك الخروج.
