Mail__DKIM__Signature(3pm)

الحزمة: perl-mail-dkim

التراخيص: PerlArtistic, GPL

Mail::DKIM::Signature(3) وثائق مساهم بها من مستخدمي Perl Mail::DKIM::Signature(3)

الاسم

Mail::DKIM::Signature - يمثل عنوان DKIM-Signature

الإصدار

الإصدار 1.20240923

المُنشِئات

new() - إنشاء توقيع جديد من المعلمات

my $signature = Mail::DKIM::Signature->new(
                    [ Algoritmo => 'rsa-sha1', ]
                    [ Firma => $base64, ]
                    [ Método => 'relaxed', ]
                    [ Dominio => 'example.org', ]
                    [ Identidad => 'user@example.org', ]
                    [ Cabeceras => 'from:subject:date:message-id', ]
                    [ Consulta => 'dns', ]
                    [ Selector => 'alpha', ]
                    [ Timestamp => time(), ]
                    [ Expiración => time() + 86400, ]
                );

parse() - إنشاء توقيع جديد من عنوان DKIM-Signature

my $sig = Mail::DKIM::Signature->parse(
                'DKIM-Signature: a=rsa-sha1; b=yluiJ7+0=; c=relaxed'
          );

ينشئ توقيعًا عن طريق تحليل محتوى عنوان DKIM-Signature المقدم. لا يلزم تضمين اسم العنوان (أي "DKIM-Signature:") ولكنه موصى به، بحيث يمكن الاحتفاظ باسم العنوان وإرجاعه بنفس الطريقة في as_string().

ملاحظة: الإدخال إلى المُنشئ بنفس تنسيق إخراج الطريقة as_string().

الطرق

algorithm() - الحصول على أو تعيين حقل الخوارزمية (a=)

الخوارزمية المستخدمة لإنشاء التوقيع. يجب أن تكون واحدة مما يلي: - "rsa-sha1"، خوارزمية تجزئة RSA-1 SHA-1 - "rsa-sha256"، خوارزمية تجزئة RSA-256 SHA-256 - "ed25519-sha256"، خوارزمية تجزئة Ed25519-256 SHA-256

انظر أيضًا hash_algorithm().

as_string() - رأس التوقيع كسلسلة

print $signature->as_string . "\n";

مخرج

DKIM-Signature: a=rsa-sha1; b=yluiJ7+0=; c=relaxed

كما هو موضح في المثال، يمكن استخدام الطريقة as_string لإنشاء DKIM-Signature الذي يتم إلحاقه بالرسالة الموقعة.

as_string_without_data() - التوقيع بدون حقل البيانات

print $signature->as_string_without_data . "\n";

مخرج

DKIM-Signature: a=rsa-sha1; b=; c=relaxed

يشبه الطريقة as_string()، ولكنه يستبعد دائمًا حقل البيانات. هذا يستخدم بواسطة طرق توحيد DKIM، التي تتطلب تضمين هذا الجزء من التوقيع في الرسالة الموقعة.

body_count() - الحصول على أو تعيين عدد البايتات في حقل الجسم (l=)

my $i = $signature->body_count;

يبلغ المُحقق بعدد البايتات في جسم البريد الإلكتروني المتضمن في التجزئة التشفيرية، بدءًا من 0 مباشرة بعد CRLF قبل الجسم. يُعرف أيضًا باسم علامة l=.

عند إنشاء توقيع، يمكن تخطي هذه العلامة، أو تعيينها بعد أن يتلقى نظام التوحيد المختار الجسم الكامل للرسالة (ولكن قبل إجراء توحيد DKIM-Signature).

body_hash() - الحصول على أو تعيين تجزئة الجسم (bh=)

my $bh = $signature->body_hash;

جزء تجزئة الجسم من الرسالة. يتم تجاهل المسافات البيضاء في هذه القيمة. هذه العلامة إلزامية.

عند الوصول إلى هذه القيمة، يتم إزالة المسافات البيضاء من العلامة نيابة عنك.

canonicalization() - الحصول على أو تعيين حقل التوحيد (c=)

$signature->canonicalization('relaxed', 'simple');
($header, $body) = $signature->canonicalization;

توحيد الرسالة (الافتراضي هو "simple/simple"). هذا يبلغ المُحقق بنوع التوحيد المستخدم لإعداد الرسالة للتوقيع.

في سياق عددي، يعيد توحيد الرأس/الجسم كسلسلة مفصولة بـ /. في سياق قائمة، يعيد مصفوفة من عنصرين، تحتوي أولاً على توحيد الرأس، ثم الجسم.

data() - الحصول على أو تعيين حقل البيانات (b=)

my $base64 = $signature->data;
$signature->data($base64);

بيانات التوقيع. يتم تلقائيًا إزالة المسافات البيضاء من القيمة المرجعة. البيانات مشفرة بـ Base64.

domain() - الحصول على أو تعيين حقل النطاق (d=)

my $d = $signature->domain;          # يحصل على قيمة النطاق
$signature->domain('example.org');   # يعين قيمة النطاق

نطاق الكيان الموقع، كما هو محدد في التوقيع. هذا هو النطاق الذي سيتم استخدامه للبحث عن المفتاح العام.

إذا كنت تستخدم اسم نطاق دولي (IDN)، فيجب تحويل اسم النطاق إلى ASCII (وفقًا للقسم 4.1 من RFC 3490) قبل تمريره إلى هذه الطريقة.

expiration() - الحصول على أو تعيين حقل انتهاء صلاحية التوقيع (x=)

انتهاء صلاحية التوقيع (الافتراضي هو undef، مما يعني عدم وجود انتهاء صلاحية). انتهاء صلاحية التوقيع، إذا تم تعريفه، هو عدد صحيح غير موقع يحدد الثواني في نظام Unix من عام 1970 حتى انتهاء صلاحية التوقيع.

get_public_key() - الحصول على المفتاح العام المشار إليه بهذا التوقيع

my $pubkey = $signature->get_public_key;

يتم تحديد المفتاح العام المراد الحصول عليه بواسطة البروتوكول والمحدد والنطاق لهذا التوقيع.

تقوم هذه الطريقة بتخزين نتيجة الاسترداد مؤقتًا، لذلك لا تتطلب الاستدعاءات اللاحقة استعلامات DNS إضافية.

ستقوم هذه الطريقة بإطلاق خطأ في حالة حدوث خطأ.

get_tag() - الوصول إلى القيمة الخام لعلامة في هذا التوقيع

my $raw_identity = $signature->get_tag('i');

استخدم هذه الطريقة للوصول إلى علامة لا يدعمها Mail::DKIM بالفعل، أو إذا كنت ترغب في تجاوز فك تشفير القيمة بواسطة Mail::DKIM.

على سبيل المثال، قيمة علامة i= (الهوية) مشفرة بتنسيق quoted-printable. إذا كنت تستخدم الطريقة identity()، فسيقوم Mail::DKIM بفك تشفير القيمة بتنسيق quoted-printable قبل إرجاعها. ومع ذلك، إذا كنت تستخدم get_tag('i')، يمكنك الوصول إلى القيمة المشفرة بتنسيق quoted-printable للقيمة.

hash_algorithm() - الوصول إلى خوارزمية التجزئة المحددة في هذا التوقيع

my $hash = $signature->hash_algorithm;

يحدد خوارزمية التجزئة المستخدمة كجزء من الخوارزمية المحددة في التوقيع.

بالنسبة للخوارزمية "rsa-sha1"، فإن خوارزمية التجزئة هي "sha1". وبالمثل، بالنسبة للخوارزمية "rsa-sha256"، فإن خوارزمية التجزئة هي "sha256". إذا لم تكن الخوارزمية معروفة، فسيتم إرجاع undef.

headerlist() - الحصول على أو تعيين رؤوس الحقول الموقعة (h=)

$signature->headerlist('a:b:c');
my $headerlist = $signature->headerlist;
my @headers = $signature->headerlist;

حقول الرأس الموقعة. قائمة مفصولة بنقطتين بأسماء حقول الرأس التي تحدد حقول الرأس المقدمة لخوارزمية التوقيع.

في سياق عددي، يتم إرجاع قائمة أسماء حقول الرأس كسلسلة واحدة، مع ربط الأسماء بنقطتين. في سياق قائمة، يتم إرجاعها كقائمة.

identity() - الحصول على أو تعيين حقل هوية التوقيع (i=)

my $i = $signature->identity;

هوية المستخدم أو الوكيل الذي يوقع هذه الرسالة. تحتوي الهوية على جزء محلي اختياري متبوعًا بـ "@" واسم نطاق. يجب أن يكون اسم النطاق هو نفسه اسم النطاق المرجع من الطريقة domain() أو نطاق فرعي منه.

من الناحية المثالية، يجب أن تتطابق الهوية مع الهوية المدرجة في حقل From: أو حقل Sender: ولكن هذا ليس إلزاميًا. إذا لم تكن الهوية المستخدمة مصرحًا لها بتوقيع الرسالة المعطاة، فلن يتم تحديد ذلك هنا.

إذا كنت تستخدم اسم نطاق دولي (IDN)، فيجب تحويل اسم النطاق إلى ASCII (وفقًا للقسم 4.1 من RFC 3490) قبل تمريره إلى هذه الطريقة.

يتم تشفير قيم الهوية بتنسيق quoted-printable في التوقيع. استخدام هذه الطريقة يترجم إلى /quoted-printable/ حسب الضرورة. إذا كنت تريد القيمة الخام بتنسيق quoted-printable للهوية، فاستخدم $signature->get_t...
