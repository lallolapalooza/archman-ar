Math::BigRat(3perl) دليل مبرمجي Perl Math::BigRat(3perl)

الاسم

Math::BigRat - حزمة حساب أعداد كسرية بأي حجم

الملخص

# طريقة بناء عامة (تُرجع دائمًا كائنًا جديدًا)

$x = Math::BigRat->bone();                # إنشاء +1
$x = Math::BigRat->bone('-');             # إنشاء -1
$x = Math::BigRat->binf();                # إنشاء +inf
$x = Math::BigRat->binf('-');             # إنشاء -inf
$x = Math::BigRat->bnan();                # إنشاء ليس رقمًا (NaN)
$x = Math::BigRat->bpi();                 # يُرجع قيمة باي (π)

$y = $x->as_float();    # يُرجع كـ Math::BigFloat

# طرق منطقية (لا تُعدّل المتلقي)
$x->is_zero();          # صحيح إذا كان $x هو 0
$x->is_one();           # صحيح إذا كان $x هو +1
$x->is_one("+");        # صحيح إذا كان $x هو +1
$x->is_one("-");        # صحيح إذا كان $x هو -1
$x->is_inf();           # صحيح إذا كان $x هو +inf أو -inf
$x->is_inf("+");        # صحيح إذا كان $x هو +inf
$x->is_inf("-");        # صحيح إذا كان $x هو -inf
$x->is_nan();           # صحيح إذا كان $x هو NaN
$x->is_finite();        # صحيح إذا كان -inf < $x < inf
$x->is_positive();      # صحيح إذا كان $x > 0
$x->is_pos();           # صحيح إذا كان $x > 0
$x->is_negative();      # صحيح إذا كان $x < 0
$x->is_neg();           # صحيح إذا كان $x < 0
$x->is_non_positive()   # صحيح إذا كان $x <= 0
$x->is_non_negative()   # صحيح إذا كان $x >= 0

$x->bneg();             # النفي
$x->babs();             # القيمة المطلقة
$x->bsgn();             # دالة الإشارة (-1, 0, 1, أو NaN)
$x->bdigitsum();        # مجموع الأرقام العشرية
$x->binc();             # زيادة $x بمقدار 1
$x->bdec();             # إنقاص $x بمقدار 1
$x->badd($y);           # الجمع (إضافة $y إلى $x)

$x->bmul($y);           # الضرب (ضرب $x في $y)
$x->bmuladd($y, $z);    # $x = $x * $y + $z
$x->bdiv($y);           # القسمة (مقربة للأسفل)
$x->bmod($y);           # باقي القسمة (x % y)
$x->bmodinv($mod);      # معكوس الضرب النمطي
$x->bmodpow($y, $mod);  # الأس النمطي (($x ** $y) % $mod)
$x->btdiv($y);          # القسمة (مقطوعة)، تعيين $x إلى الناتج
$x->btmod($y);          # باقي القسمة (مقطوع)
$x->binv()              # معكوس (1/$x)
$x->bpow($y);           # قوة الوسطاء (x ** y)
$x->blog();             # لوغاريتم $x للأساس e (عدد أويلر)
$x->blog($base);        # لوغاريتم $x للأساس $base (مثلًا، الأساس 2)
$x->bexp();             # حساب e ** $x حيث e هو عدد أويلر
$x->bilog2();           # log2($x) مقربًا للأسفل لأقرب عدد صحيح
$x->bilog10();          # log10($x) مقربًا للأسفل لأقرب عدد صحيح
$x->bclog2();           # log2($x) مقربًا للأعلى لأقرب عدد صحيح
$x->bclog10();          # log10($x) مقربًا للأعلى لأقرب عدد صحيح
$x->bnok($y);           # توافيق (معامل ذو الحدين n فوق k)
$x->bperm($y);          # تباديل
$x->buparrow($n, $y);   # ترميز Knuth السهمي للأعلى
$x->bhyperop($n, $y);   # العامل الخارق من الدرجة n
$x->backermann($y);     # دالة أكرمان
$x->bsin();             # الجيب
$x->bcos();             # جيب التمام
$x->batan();            # الظل العكسي
$x->batan2($y);         # الظل العكسي بوسيطين
$x->bsqrt();            # حساب الجذر التربيعي
$x->broot($y);          # الجذر الـ $y للـ $x (مثلًا $y == 3 => جذر تكعيبي)
$x->bfac();             # مضروب $x (1*2*3*4*..$x)
$x->bdfac();            # مضروب مزدوج لـ $x ($x*($x-2)*($x-4)*...)
$x->btfac();            # مضروب ثلاثي لـ $x ($x*($x-3)*($x-6)*...)
$x->bmfac($k);          # مضروب متعدد من الدرجة $k للـ $x ($x*($x-$k)*...)
$x->bfib($k);           # عدد فيبوناتشي الـ $k
$x->blucas($k);         # عدد لوكاس الـ $k
$x->blsft($n);          # إزاحة لليسار بمقدار $n موضع في القاعدة 2
$x->blsft($n, $b);      # إزاحة لليسار بمقدار $n موضع في القاعدة $b
$x->brsft($n);          # إزاحة لليمين بمقدار $n موضع في القاعدة 2
$x->brsft($n, $b);      # إزاحة لليمين بمقدار $n موضع في القاعدة $b
# طرق بت (تُعدّل المتلقي)
$x->bblsft($y);         # إزاحة لليسار بت
$x->bbrsft($y);         # إزاحة لليمين بت
$x->band($y);           # و بت
$x->bior($y);           # أو بت الشامل
$x->bxor($y);           # أو بت الحصري
$x->bnot();             # نفي بت (متمم الاثنين)
# طرق التقريب (تُعدّل المتلقي)
$x->round($A, $P, $R);  # تقريب إلى دقة أو دقة باستخدام
                        #   وضع التقريب $R
$x->bround($n);         # الدقة: الحفاظ على $n من الأرقام
$x->bfround($n);        # $n > 0: التقريب إلى المنزلة الـ $n قبل النقطة العشرية
                        # $n < 0: التقريب إلى المنزلة الـ $n بعد النقطة العشرية
$x->bfloor();           # التقريب نحو سالب اللانهاية
$x->bceil();            # التقريب نحو موجب اللانهاية
$x->bint();             # التقريب نحو الصفر
# طرق رياضية أخرى (لا تُعدّل المتلقي)
$x->bgcd($y);           # القاسم المشترك الأكبر
$x->blcm($y);           # المضاعف المشترك الأصغر
# طرق خصائص الكائن (لا تُعدّل المتلقي)
$x->sign();             # الإشارة، إما + أو - أو NaN
$x->digit($n);          # الرقم الـ $n، بدءًا من اليمين
$x->digit(-$n);         # الرقم الـ $n، بدءًا من اليسار
$x->digitsum();         # مجموع الأرقام العشرية
$x->length();           # يُرجع عدد الأرقام في العدد
$x->mantissa();         # يُرجع (الموقع) الجزء العشري كـ Math::BigInt
$x->exponent();         # يُرجع الأس كـ Math::BigInt

$x->numerator();        # البسط
$x->denominator();      # المقام

$x->bstr();             # التدوين العشري (ربما مع حشو بالأصفار)
$x->bnstr();            # سلسلة نصية بتدوين مُطَبَّع
$x->bestr();            # سلسلة نصية بتدوين هندسي
$x->bdstr();            # سلسلة نصية بتدوين عشري (بدون حشو)
$x->bfstr();            # سلسلة نصية بتدوين كسري

$x->as_oct();           # كسلسلة نصية ثمانية موقعة مع بادئة "0"

الوصف

Math::BigRat يكمل Math::Bi

con $y وخذ بعين الاعتبار الإشارة. تُرجع -1، 0، 1 أو غير محدد.

bacmp()
$x->bacmp($y);

قارن $x مع $y مع تجاهل إشارتهما. تُرجع -1، 0، 1 أو غير محدد.

beq()
$x -> beq($y);

تُرجع true إذا وفقط إذا كان $x يساوي $y، وfalse بخلاف ذلك.

bne()
$x -> bne($y);

تُرجع true إذا وفقط إذا كان $x لا يساوي $y، وfalse بخلاف ذلك.

blt()
$x -> blt($y);

تُرجع true إذا وفقط إذا كان $x أصغر من $y، وfalse بخلاف ذلك.

ble()
$x -> ble($y);

تُرجع true إذا وفقط إذا كان $x أصغر من أو يساوي $y، وfalse بخلاف ذلك.

bgt()
$x -> bgt($y);

تُرجع true إذا وفقط إذا كان $x أكبر من $y، وfalse بخلاف ذلك.

bge()
$x -> bge($y);

تُرجع true إذا وفقط إذا كان $x أكبر من أو يساوي $y، وfalse بخلاف ذلك.

blsft()/brsft()
تُستخدم لإزاحة الأرقام إلى اليسار أو اليمين.

يرجى الرجوع إلى الوثائق في Math::BigInt للحصول على تفاصيل إضافية.

bneg()
$x->bneg();

تُستخدم لعكس قيمة الكائن بدلاً من.

bnorm()
$x->bnorm();

تقليل الرقم إلى أقصر شكل ممكن. يتم استدعاء هذا الروتين تلقائيًا عند الحاجة.

binc()
$x->binc();

تزيد $x بمقدار 1 وتُرجع النتيجة.

bdec()
$x->bdec();

تنقص $x بمقدار 1 وتُرجع النتيجة.

badd()
$x->badd($y);

تجمع $y مع $x وتُرجع النتيجة.

bsub()
$x->bsub($y);

تطرح $y من $x وتُرجع النتيجة.

bmul()
$x->bmul($y);

تضرب $y في $x وتُرجع النتيجة.

bdiv()
$q = $x->bdiv($y);
($q, $r) = $x->bdiv($y);

في سياق قياسي، تقسم $x على $y وتُرجع النتيجة. في سياق قائمة، تجري قسمة صحيحة (F-division)، وتُرجع عددًا صحيحًا $q والباقي $r بحيث يكون $x = $q * $y + $r. الباقي (modulo) يساوي نتيجة "$x->bmod($y)".

bmod()
$x->bmod($y);

تُرجع $x modulo $y. عندما يكون $x عددًا محدودًا و $y عددًا محدودًا وغير صفري، تكون النتيجة مطابقة تمامًا للباقي بعد القسمة الصحيحة (F-division). إذا كان كلاهما $x و $y عددين صحيحين، فإن النتيجة تكون مطابقة تمامًا لنتيجة المعامل % في Perl.

binv()
$x->binv();

معكوس $x.

bsqrt()
$x->bsqrt();

يحسب الجذر التربيعي لـ $x.

bpow()
$x->bpow($y);

يحسب $x ** $y.

يرجى الرجوع إلى الوثائق في Math::BigInt للحصول على تفاصيل إضافية.

broot()
$x->broot($n);

يحسب الجذر النوني لـ $x.

bmodpow()
$num->bmodpow($exp,$mod);           # الأس المعياري
                                    # ($num**$exp % $mod)

تُرجع قيمة $num مرفوعة للقوة $exp في المعيار $mod باستخدام الأس الثنائي. "bmodpow" أسرع بكثير من كتابة

$num ** $exp % $mod

لأنها تقلل المتغيرات الداخلية في المعيار عندما يكون ذلك ممكنًا، وتعمل على أرقام أصغر.

"bmodpow" تدعم أيضًا الأسس السالبة.

bmodpow($num, -1, $mod)

تكافئ تمامًا

bmodinv($num, $mod)
bmodinv()
$x->bmodinv($mod);          # معكوس معياري

تُرجع المعكوس الضربي لـ $x معياري $mod. إذا

$y = $x -> copy() -> bmodinv($mod)

فإن $y هو أقرب رقم إلى الصفر، مع نفس إشارة $mod، والذي يحقق

($x * $y) % $mod = 1 % $mod

إذا كان $x و $y غير صفريين، فيجب أن يكونا أوليين نسبيًا، أي "bgcd($y, $mod)==1". تُرجع "NaN" عندما لا يوجد معكوس ضربي معياري.

blog()
$x->blog($base, $accuracy);         # لوغاريتم x للأساس $base

إذا كان $base غير معرف، يتم استخدام الرقم أويلر (e):

print $x->blog(undef, 100);         # لوغاريتم (x) بـ 100 رقم عشري
bexp()
$x->bexp($accuracy);        # حساب e ** X

يحسب عددين صحيحين A و B بحيث A/B يساوي "e ** $x"، حيث "e" هو رقم أويلر.

تمت إضافة هذه الطريقة في الإصدار v0.20 من Math::BigRat (مايو 2007).

انظر أيضًا "blog()".

bnok()
انظر "bnok()" في Math::BigInt.
bperm()
انظر "bperm()" في Math::BigInt.
bfac()
$x->bfac();

يحسب مضروب $x. على سبيل المثال:

print Math::BigRat->new('3/1')->bfac(), "\n";   # 1*2*3
print Math::BigRat->new('5/1')->bfac(), "\n";   # 1*2*3*4*5

تعمل حاليًا فقط مع الأعداد الصحيحة.

band()
$x->band($y);               # AND بتويز
bior()
$x->bior($y);               # OR شامل
bxor()
$x->bxor($y);               # XOR حصري
bnot()
$x->bnot();                 # NOT بتويز (المكمل الثنائي)
bfloor()
$x->bfloor();

يقرب $x إلى ما لا نهاية السلبية، أي يضبط $x على أكبر عدد صحيح أصغر من أو يساوي $x.

bceil()
$x->bceil();

يقرب $x إلى ما لا نهاية الإيجابية، أي يضبط $x على أصغر عدد صحيح أكبر من أو يساوي $x.

bint()
$x->bint();

يقرب $x باتجاه الصفر.

bgcd()
$x -> bgcd($y);             # القاسم المشترك الأكبر لـ $x و $y
$x -> bgcd($y, $z, ...);    # القاسم المشترك الأكبر لـ $x، $y، $z، ...

تُرجع القاسم المشترك الأكبر (GCD)، وهو الرقم ذو القيمة المطلقة الأكبر بحيث تكون $x / gcd، $y / gcd، ... عددًا صحيحًا. على سبيل المثال، عندما تكون المعاملات 4/5 و 6/5، يكون القاسم المشترك الأكبر هو 2/5. هذا تعميم للقاسم المشترك الأكبر العادي للأعداد الصحيحة. انظر "gcd()" في Math::BigInt.

digit()
print Math::BigRat->new('123/1')->digit(1);     # 1
print Math::BigRat->new('123/1')->digit(-1);    # 3

يُرجع الرقم N-th من X عندما يكون X قيمة عدد صحيح.

$len = $x->length();

يُرجع طول $x بالأرقام للقيم الصحيحة.

($n, $d) = $x->parts();

يُرجع قائمة بـ (عدد صحيح) و (مقام غير موقع) كـ BigInts.

$x->dparts();

يُرجع الجزء الصحيح والجزء الكسري.

$x->fparts();

يُرجع أصغر بسط ومقام، بحيث يعطي البسط مقسومًا على المقام القيمة الأصلية مرة أخرى. بالنسبة للأرقام المحدودة، كلا القيمتين عددين صحيحين. تذكير: كسر.

$x->numerator();

يُرجع نسخة من البسط (الجزء العلوي من الخط) كـ BigInt.

$x->denominator();

يُرجع نسخة من المقام (الجزء السفلي من الخط) كـ BigInt موجب.

bstr()
my $x = Math::BigRat->new('8/4');
print $x->bstr(), "\n";             # يطبع 1/2

يُرجع BigRat كسلسلة ثمانية. يعمل فقط مع الأعداد الصحيحة.

print $x->as_bin(), "\n";               # '0b1101'

يُرجع BigRat كسلسلة ثنائية. يعمل فقط مع الأعداد الصحيحة.

numify()
my $y = $x->numify();

يُرجع الكائن كمتغير قياسي. قد يحدث فقدان للبيانات إذا لم يتمكن الكائن من تمثيل نفسه كمتغير قياسي عادي (عدد صحيح أو عشري). لذلك، استخدم "as_int() أو "as_float() بدلاً من ذلك.

يتم استخدام هذا الروتين تلقائيًا عند الحاجة إلى متغير قياسي.

my $x = Math::BigRat->new('3/1');
@array = (0, 1, 2, 3);
$y = $array[$x];                # تعيين $y إلى 3
config()

حرفيات الأرقام

بعد "use Math::BigRat ':constant'" يتم تحويل جميع الحرفيات الرقمية في النطاق المحدد إلى كائنات "Math::BigRat". يحدث هذا التحويل في وقت الترجمة. يتم تحويل كل رقم غير صحيح إلى NaN.

على سبيل المثال،

perl -MMath::BigRat=:constant -le 'print 2**150'

يطبع القيمة الدقيقة لـ "2**150". لاحظ أنه بدون تحويل الحرفيات إلى كائنات، يتم حساب التعبير "2**150" باستخدام متغيرات Perl القياسية، مما قد يؤدي إلى نتيجة غير دقيقة.

ملاحظة: لا تتأثر السلاسل النصية، لذلك

use Math::BigRat qw/:constant/;
$x = "1234567890123456789012345678901234567890"
        + "123456789123456789";

يعطي ما تتوقعه. يجب عليك استدعاء "Math::BigRat->new() صراحةً على واحد على الأقل من المعاملين. يجب عليك أيضًا اقتباس الثوابت الكبيرة لتجنب فقدان الدقة:

use Math::BigRat;
$x = Math::BigRat->new("1234567889123456789123456789123456789");

بدون علامات اقتباس، يقوم Perl أولاً بتحويل الثابت الكبير إلى قيمة عشرية في وقت الترجمة، ثم يحولها إلى كائن Math::BigRat في وقت التشغيل، مما يؤدي إلى نتيجة غير دقيقة.
