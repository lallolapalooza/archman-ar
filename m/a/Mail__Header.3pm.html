Mail__Header(3pm)

الحزمة: perl-mailtools

التراخيص: PerlArtistic, GPL

Mail::Header(3) وثائق Perl المساهمة من المستخدم Mail::Header(3)

الاسم

Mail::Header - معالجة رؤوس MIME

الملخص

use Mail::Header;

my $head = Mail::Header->new;
my $head = Mail::Header->new( \*STDIN );
my $head = Mail::Header->new( [<>], Modify => 0);

الوصف

قراءة، كتابة، إنشاء، ومعالجة رؤوس MIME، الجزء الأمامي من كل رسالة بريد إلكتروني حديثة، ولكنها تستخدم أيضًا في بروتوكولات أخرى مثل HTTP. يتم الاحتفاظ بالحقول في كائنات Mail::Field.

كن على علم بأن حقول الرأس لكل منها جزء اسم، والذي يجب التعامل معه بشكل غير حساس لحالة الأحرف، وجزء محتوى، والذي قد يتم طيه على عدة أسطر.

Mail::Header لا تتبع دائمًا معايير RFC بما يكفي، ولا تساعدك في ترميزات الأحرف. لا تستخدم مراجع ضعيفة حيث يمكنها (لأنها لم تكن موجودة عندما كُتبت الوحدة) مما يكلف بعض الأداء ويجعل التنفيذ أكثر تعقيدًا قليلاً. تنفيذ Mail::Message::Head أحدث بكثير وبالتالي أفضل.

الطرق

المنشئات

$obj->dup()
إنشاء نسخة مكررة من الكائن الحالي.
$obj->new( [$source], [%options] )
Mail::Header->new( [$source], [%options] )
قد يكون $source واصف ملف (مرجع إلى GLOB) أو مرجع إلى مصفوفة. إذا تم إعطاؤه، فسيتم تهيئة الكائن الجديد برؤوس إما من المصفوفة التي تمت قراءتها من واصف الملف.

%options هي قائمة خيارات معطاة في شكل أزواج مفتاح-قيمة، تمامًا مثل جدول التجزئة. الخيارات الصالحة هي

-Option    --Default
 FoldLength  79
 MailFrom    'KEEP'
 Modify      false
FoldLength => INTEGER
طول السطر الافتراضي الذي سيتم استخدامه عند طي أسطر الرأس. انظر fold_length().
MailFrom => 'IGNORE'|'COERCE'|'KEEP'|'ERROR'
انظر الطريقة mail_from().
Modify => BOOLEAN
إذا كانت هذه القيمة *حقيقية*، فسيتم إعادة تنسيق الرؤوس، وإلا فإن تنسيق أسطر الرأس سيبقى دون تغيير.

المنشئات "الزائفة"

كن على علم بأن المنشئات التالية تتطلب جميعها كائن رأس تم إنشاؤه بالفعل، وسيتم تدمير محتواه الأصلي.

$obj->empty()
إفراغ كائن "Mail::Header" موجود من جميع الأسطر.
$obj->extract(ARRAY)
استخراج رأس من المصفوفة المعطاة إلى كائن Mail::Header موجود. "extract" سيقوم بتعديل هذه المصفوفة. يعيد الكائن الذي تم استدعاء الطريقة عليه.
$obj->header( [ARRAY] )
"header" يقوم بعمليات متعددة. أولاً، سيستخرج رأسًا من ARRAY، إذا تم توفيره. ثم سيقوم بإعادة تنسيق الرأس (إذا كان إعادة التنسيق مسموحًا به)، وأخيرًا سيعيد مرجعًا إلى مصفوفة تحتوي على الرأس في شكل قابل للطباعة.
$obj->header_hashref( [HASH] )
مثل header()، ولكنه سيقوم في النهاية بتعيين رؤوس من مرجع تجزئة، وسيعيد الرؤوس كمرجع تجزئة.

مثال:

$fields->{From} = 'Tobias Brox <tobix@cpan.org>';
$fields->{To}   = ['you@somewhere', 'me@localhost'];
$head->header_hashref($fields);
$obj->read($fh)
قراءة رأس من واصف الملف المعطى إلى كائن Mail::Header موجود.

المُوصلات

$obj->fold_length( [$tag], [$length] )
تعيين طول الطي الافتراضي لجميع العلامات أو لعلامة واحدة فقط. بدون وسيطات، يتم إرجاع طول الطي الافتراضي. مع وسيطتين، يقوم بتعيين طول الطي للعلامة المعطاة ويعيد القيمة السابقة. إذا تم توفير $length فقط، فسيتم تعيين طول الطي الافتراضي للكائن الحالي.

في النموذج ذي الوسيطتين، يمكن استدعاء "fold_length" كطريقة ثابتة، مما يحدد أطوال الطي الافتراضية للعلامات التي ستستخدمها جميع كائنات "Mail::Header". انظر الطريقة "fold" لوصف كيفية استخدام "Mail::Header" لهذه القيم.

$obj->mail_from('IGNORE'|'COERCE'|'KEEP'|'ERROR')
يحدد هذا ما يجب فعله عند مواجهة سطر `From '. القيم الصالحة هي "IGNORE" - تجاهل الرأس وحذفه، "ERROR" - استدعاء خطأ (استدعاء die)، "COERCE" - إعادة تسميتها إلى Mail-From، و "KEEP" - الاحتفاظ بها.
$obj->modify( [$value] )
إذا كانت $value *خاطئة*، فلن تقوم "Mail::Header" بإعادة تنسيق تلقائي للرؤوس، بخلاف التأكد من أن السطر يبدأ بالعلامات المعطاة.

المعالجة

$obj->add( $tag, $line [, $index] )
إضافة سطر جديد إلى الرأس. إذا كان $tag هو "undef"، فسيتم استخراج العلامة من بداية السطر المعطى. إذا تم توفير $index، فسيتم إدراج السطر الجديد في الرأس في النقطة المعطاة، وإلا فسيتم إلحاق السطر الجديد بنهاية الرأس.
$obj->as_string()
إرجاع الرأس كسلسلة نصية واحدة.
$obj->cleanup()
إزالة أي سطر رأس يحتوي، بخلاف العلامة، على مسافات بيضاء فقط
$obj->combine( $tag [, $with] )
دمج جميع مثيلات $tag في واحد. سيتم ربط الأسطر معًا بواسطة $with، أو مسافة واحدة إذا لم يتم توفيرها. سيتم وضع العنصر الجديد في الرأس حيث كان المثيل الأول، وسيتم إزالة جميع المثيلات الأخرى لـ $tag.
$obj->count($tag)
إرجاع عدد المرات التي تظهر فيها العلامة المعطاة في الرأس
$obj->delete( $tag [, $index ] )
حذف علامة من الرأس. إذا تم توفير $index، فسيتم إزالة المثيل N للعلامة. إذا لم يتم توفير $index، فسيتم إزالة جميع مثيلات العلامة.
$obj->fold( [$length] )
طي الرأس. إذا لم يتم توفير $length، فستستخدم "Mail::Header" القواعد التالية لتحديد الطول الذي يجب طي السطر به.
$obj->get( $tag [, $index] )
الحصول على النص من سطر. إذا تم توفير $index، فسيتم إرجاع نص المثيل N. إذا لم يتم توفيره، فإن القيمة المعادة تعتمد على السياق الذي تم فيه استدعاء "get". في سياق مصفوفة، سيتم إرجاع قائمة بجميع النصوص من جميع مثيلات $tag. في سياق قياسي، سيتم إرجاع النص للمثيل الأول.

تم فك طي الأسطر، ولكن لا تزال تنتهي بسطر جديد (انظر "chomp")

$obj->print( [$fh] )
طباعة الرأس إلى واصف الملف المعطى، أو "STDOUT" إذا لم يتم توفير واصف ملف.
$obj->replace( $tag, $line [, $index ] )
استبدال سطر في الرأس. إذا كان $tag هو "undef"، فسيتم استخراج العلامة من بداية السطر المعطى. إذا تم توفير $index، فسيحل السطر الجديد محل المثيل N لتلك العلامة، وإلا فسيتم استبدال المثيل الأول للعلامة. إذا لم تظهر العلامة في الرأس، فسيتم إلحاق سطر جديد بالرأس.
$obj->tags()
