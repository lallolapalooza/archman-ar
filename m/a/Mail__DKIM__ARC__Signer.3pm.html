Mail__DKIM__ARC__Signer(3pm)

الحزمة: perl-mail-dkim

التراخيص: PerlArtistic, GPL

Mail::DKIM::ARC::Signer(3) وثائق Perl المقدمة من المستخدم Mail::DKIM::ARC::Signer(3)

الاسم

Mail::DKIM::ARC::Signer - يولد توقيع DKIM لرسالة

الإصدار

version 1.20240923

ملخص

use Mail::DKIM::ARC::Signer;
use Mail::DKIM::TextWrap;  #موصى به
# إنشاء كائن موقع
my $signer = Mail::DKIM::ARC::Signer->new(
                Algorithm => 'rsa-sha256',
                Chain => 'none',    # أو pass|fail|ar
                Domain => 'example.org',
                SrvId => 'example.org',
                Selector => 'selector1',
                KeyFile => 'private.key',
                Headers => 'x-header:x-header2',
           );
# قراءة بريد إلكتروني من مقبض ملف
$signer->load(*STDIN);
# ملاحظة: يجب أن يحتوي أي بريد إلكتروني يتم توقيعه بـ ARC على رأس Authentication-Results
# حتى يمكن أن يغطي ختم ARC تلك النتائج المنسوخة إلى
# رأس ARC-Authentication-Results.
# أو قراءة بريد إلكتروني وتمريره إلى الموقع، سطرًا واحدًا في كل مرة
while (<STDIN>)
{
    # إزالة محددات الأسطر المحلية
    chomp;
    s/\015$//;
    # استخدام محددات أسطر SMTP
    $signer->PRINT("$_\015\012");
}
$signer->CLOSE;
die 'فشل' $signer->result_detail() unless $signer->result() eq 'sealed';
# الحصول على جميع رؤوس التوقيع لإضافتها في بداية الرسالة
# ARC-Seal, ARC-Message-Signature و ARC-Authentication-Results
# بهذا الترتيب.
print $signer->as_string;

الوصف

هذه الفئة هي الجزء من Mail::DKIM المسؤول عن إنشاء أختام ARC لرسالة معينة. تقوم بإنشاء كائن من هذه الفئة، مع تحديد معلمات ARC-Message-Signature التي ترغب في إنشائها.

تحتاج أيضًا إلى تمرير قيمة 'Chain' (pass أو fail) من التحقق من أختام ARC السابقة على الرسالة.

بعد ذلك، تقوم بتغذيتها بالرسالة بأكملها باستخدام "PRINT()"، مع الانتهاء بـ "CLOSE()".

أخيرًا، استخدم طريقة "as_string()" للحصول على رؤوس ARC الجديدة.

ملاحظة: لا يمكنك ختم رسالة إلا إذا تم بالفعل إضافة رأس Authentication-Results إليها، إما باستخدام "PRINT()" لتغذيتها مسبقًا في هذه الوحدة، أو عن طريق إضافة رسالة تم مصادقتها بالفعل بواسطة آليات الفحص الواردة لديك.

ليس من الضروري عمل ARC-Seal لرسالة تحتوي بالفعل على توقيعات DKIM إذا لم تقم بتعديل الرسالة وبالتالي كسر توقيع DKIM الحالي أو توقيع ARC-Message العلوي على البريد الإلكتروني.

توقيعات جميلة

يتضمن Mail::DKIM وحدة تغليف التوقيع (تقوم بإدراج فواصل الأسطر في التوقيع المنشأ ليبدو أجمل في الرسالة الناتجة. لتمكين هذه الوحدة، قم ببساطة باستدعاء

use Mail::DKIM::TextWrap;

في برنامجك قبل إنشاء التوقيع.

منشئ

new()

إنشاء توقيع موجه للكائنات.

# إنشاء موقع باستخدام السياسة الافتراضية
my $signer = Mail::DKIM::ARC::Signer->new(
                Algorithm => 'rsa-sha256',
                Chain => 'none',    # أو pass|fail|ar
                Domain => 'example.org',
                SrvId => 'example.org',
                Selector => 'selector1',
                KeyFile => 'private.key',
                Headers => 'x-header:x-header2',
           );
مفتاح
بدلاً من استخدام "KeyFile"، استخدم "Key" لاستخدام كائن Mail::DKIM::PrivateKey تم تحميله بالفعل.
سلسلة
قيمة cv= لرأس Arc-Seal. "ar" يعني نسخها من رأس Authentication-Results، أو استخدام none إذا لم يكن هناك واحد.
SrvId
معرف authserv-id في رؤوس Authentication-Results، افتراضيًا هو Domain.
رؤوس
قائمة مفصولة بنقطتين للرؤوس المراد توقيعها، يتم إضافتها إلى قائمة الرؤوس الافتراضية كما هو موضح في مواصفات DKIM.

لكل رأس محدد، سيتم توقيع جميع الرؤوس من هذا النوع الموجودة في الرسالة، ولكننا لن نوقع بشكل زائد أو نوقع رؤوسًا غير موجودة.

إذا كنت بحاجة إلى تحكم أكبر في الرؤوس الموقعة، يرجى استخدام طريقة extended_headers() بدلاً من ذلك.

قائمة الرؤوس الموقعة افتراضيًا هي كما يلي:

From Sender Reply-To Subject Date
Message-ID To Cc MIME-Version
Content-Type Content-Transfer-Encoding Content-ID Content-Description
Resent-Date Resent-From Resent-Sender Resent-To Resent-cc
Resent-Message-ID
In-Reply-To References
List-Id List-Help List-Unsubscribe List-Subscribe
List-Post List-Owner List-Archive
علامات
هاش ريف اختياري من العلامات الإضافية التي سيتم إضافتها إلى توقيع ARC الذي تم إنشاؤه.
SealTags
هاش ريف اختياري من العلامات الإضافية التي سيتم إضافتها إلى ختم ARC الذي تم إنشاؤه.

طرق

PRINT()

تغذية جزء من الرسالة إلى الموقع.

$signer->PRINT("سطر من الرسالة\015\012");

تغذي محتوى الرسالة التي يتم توقيعها في الموقع. تم تصميم واجهة برمجة التطبيقات بهذه الطريقة حتى لا يلزم قراءة الرسالة بأكملها في الذاكرة دفعة واحدة.

يرجى ملاحظة أنه على الرغم من أن طريقة PRINT() تتوقع منك استخدام أحرف إنهاء الأسطر بأسلوب SMTP، يجب عليك عدم استخدام تقنية "dot-stuffing" بأسلوب SMTP الموضحة في RFC 2821 القسم 4.5.2. ولا يجب عليك استخدام تسلسل <CR><LF>.<CR><LF> لإنهاء الرسالة.

CLOSE()

استدعِ هذا عند الانتهاء من تغذية الرسالة.

$signer->CLOSE;

تكمل هذه الطريقة عملية التوحيد القياسي، وتحسب تجزئة، وتنشئ توقيعًا.

extended_headers()

تتجاوز هذه الطريقة الرؤوس التي سيتم توقيعها وتسمح بتحكم أكبر مما هو ممكن مع خاصية Headers في المنشئ.

تتوقع الطريقة تمرير HashRef.

المفاتيح هي الرؤوس التي سيتم توقيعها، والقيم هي إما عدد الرؤوس من هذا النوع المراد توقيعها، أو القيم الخاصة '*' و '+'.

* سيوقع جميع الرؤوس من هذا النوع الموجودة في الرسالة.

+ سيوقع جميع الرؤوس + 1 من هذا النوع الموجودة في الرسالة لمنع إضافة رؤوس إضافية.

يمكنك تجاوز أي من الرؤوس الافتراضية عن طريق تضمينها في الهاش ريف، وتعطيلها عن طريق إعطائها قيمة 0.

المفاتيح غير حساسة لحالة الأحرف مع إضافة القيم حتى أعلى قيمة.

Headers => {
    'X-test'  => '*',
    'x-test'  => '1',
    'Subject' => '+',
    'Sender'  => 0,
},

add_signature()

تستخدم بواسطة سياسة الموقع لإنشاء توقيع جديد.

$signer->add_signature(new Mail::DKIM::Signature(...));

يمكن لسياسات الموقع استخدام هذه الطريقة لتحديد معلمات كاملة للتوقيع المضاف، بما في ذلك نوع التوقيع. لمزيد من المعلومات، راجع Mail::DKIM::SignerPolicy.

algorithm()

الحصول على الخوارزمية المحددة أو تعيينها.

$alg = $signer->algorithm;
$signer->algorithm('rsa-sha256');

domain()

الحصول على النطاق المحدد أو تعيينه.

$alg = $signer->domain;
$signer->domain('example.org');

load()

تحميل الرسالة بأكملها من مقبض ملف.

$signer->load($file_handle);

يقرأ رسالة كاملة من مقبض الملف المحدد، ويغذيها في الموقع. يجب أن تستخدم الرسالة محددات أسطر <CRLF> (نفس بروتوكول SMTP).

headers()

تحديد الرؤوس التي يجب وضعها في التوقيع.

my $headers = $signer->headers;

هذا سلسلة تحتوي على أسماء حقول الرأس التي سيتم توقيعها، مفصولة بنقطتين.

key()

الحصول على المفتاح أو تعيينه.
