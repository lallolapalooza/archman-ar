manadv_init_obj(3) دليل وظائف المكتبة manadv_init_obj(3)

الاسم

manadv_init_obj - تهيئة كائن الأوامر المباشرة mana من هياكل ibv_xxx

ملخص”

#include <infiniband/manadv.h>
int manadv_init_obj(struct manadv_obj *obj, uint64_t obj_type);

الوصف

manadv_init_obj() ستقوم هذه الوظيفة بتهيئة هياكل manadv_xxx بناءً على النوع المقدم. يتم أخذ معلومات التهيئة من هياكل ibv_xx المقدمة كجزء من المدخلات.

المعلمات

obj
هياكل manadv_xxx التي سيتم إرجاعها.
struct manadv_qp {
    void        *sq_buf;
    uint32_t    sq_count;
    uint32_t    sq_size;
    uint32_t    sq_id;
    uint32_t    tx_vp_offset;
    void        *db_page;
};
struct manadv_cq {
    void        *buf;
    uint32_t    count;
    uint32_t    cq_id;
};
struct manadv_rwq {
    void        *buf;
    uint32_t    count;
    uint32_t    size;
    uint32_t    wq_id;
    void        *db_page;
};
struct manadv_obj {
    struct {
        struct ibv_qp       *in;
        struct manadv_qp    *out;
    } qp;
    struct {
        struct ibv_cq       *in;
        struct manadv_cq    *out;
    } cq;
    struct {
        struct ibv_wq       *in;
        struct manadv_rwq   *out;
    } rwq;
};
obj_type
أنواع هياكل manadv_xxx التي سيتم إرجاعها.
enum manadv_obj_type {
    MANADV_OBJ_QP   = 1 << 0,
    MANADV_OBJ_CQ   = 1 << 1,
    MANADV_OBJ_RWQ  = 1 << 2,
};

قيمة الإرجاع

0 عند النجاح أو قيمة errno عند الفشل (والتي تشير إلى سبب الفشل).
