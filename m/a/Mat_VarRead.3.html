Mat_VarRead(3)

الحزمة: libmatio

التراخيص: BSD-2-Clause

MAT_VARREAD(3) دليل وظائف المكتبة MAT_VARREAD(3)

الاسم

Mat_VarRead — يقرأ المعلومات والبيانات لمتغير معين في ملف MAT الخاص بـ MATLAB.

الملخص

#include <matio.h>

matvar_t *
Mat_VarRead(mat_t *matfp, const char *name);

الوصف

الدالة Mat_VarRead() تقرأ المعلومات والبيانات للمتغير name في ملف MAT المفتوح.

قيم الإرجاع

إذا تم العثور على المتغير في ملف MAT وتمت قراءته بنجاح، يتم إرجاع مؤشر إلى بنية متغير MATLAB. إذا لم يتم العثور على المتغير، أو حدث خطأ أثناء قراءة المتغير، يتم إرجاع NULL.

أمثلة

برنامج المثال هذا يفتح ملف MAT المسمى بالوسيط الأول للبرنامج، ويقرأ متغيرًا يسمى x من الملف. تستخدم دالة matio المسماة Mat_VarPrint لطباعة المعلومات والبيانات من الملف.

#include "matio.h"

int
main(int argc, char **argv)
{
    mat_t    *matfp;
    matvar_t *matvar;

    matfp = Mat_Open(argv[1], MAT_ACC_RDONLY);
    if ( NULL == matfp ) {
        fprintf(stderr, "Error opening MAT file %0، argv[1]);
        return EXIT_FAILURE;
    }

    matvar = Mat_VarRead(matfp, "x");
    if ( NULL != matvar ) {
        Mat_VarPrint(matvar);
        Mat_VarFree(matvar);
    }

    Mat_Close(matfp);
    return EXIT_SUCCESS;
}
