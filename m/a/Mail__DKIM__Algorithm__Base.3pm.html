Mail__DKIM__Algorithm__Base(3pm)

الحزمة: perl-mail-dkim

التراخيص: PerlArtistic, GPL

Mail::DKIM::Algorithm::Base(3) وثائق Perl المساهمة من المستخدم Mail::DKIM::Algorithm::Base(3)

الاسم

Mail::DKIM::Algorithm::Base - الفئة الأساسية لـ "خوارزميات" DKIM

الإصدار

version 1.20240923

الملخص

my $algorithm = new Mail::DKIM::Algorithm::rsa_sha1(
                    Signature => $dkim_signature
                );
# إضافة رؤوس
$algorithm->add_header("Subject: this is the subject\015\012");
$algorithm->finish_header;
# إضافة جسم
$algorithm->add_body("This is the body.\015\012");
$algorithm->add_body("Another line of the body.\015\012");
$algorithm->finish_body;
# الآن قم بالتوقيع أو التحقق...
# TODO...

المُنشئ

لا ينبغي عليك إنشاء كائن من هذه الفئة مباشرة. بدلاً من ذلك، استخدم إحدى فئات تنفيذ خوارزميات DKIM، مثل rsa_sha1:

my $algorithm = new Mail::DKIM::Algorithm::rsa_sha1(
                    Signature => $dkim_signature
                );

الأساليب

add_body() - تغذية جزء من الجسم إلى الخوارزمية/التقنين

$algorithm->add_body("This is the body.\015\012");
$algorithm->add_body("Another line of the body.\015\012");

يجب تغذية الجسم "سطر" واحد في كل مرة.

add_header() - تغذية حقل رأس إلى الخوارزمية/التقنين

$algorithm->add_header("Subject: this is the subject\015\012");

يجب أن يبدأ الرأس باسم حقل الرأس ويستمر عبر أي أسطر مطوية (بما في ذلك تسلسلات <CRLF> المضمنة). ينتهي بتسلسل <CRLF> في نهاية حقل الرأس.

finish_body() - الإشارة إلى نهاية جسم الرسالة

$algorithm->finish_body

استدعِ هذه الطريقة عندما يتم تقديم جميع الأسطر من الجسم. بعد استدعاء هذه الطريقة، استخدم sign() أو verify() للحصول على النتائج من الخوارزمية.

finish_header() - الإشارة إلى نهاية كتلة حقول الرأس

$algorithm->finish_header;

استدعِ هذه الطريقة عندما يتم تقديم جميع الرؤوس.

sign() - إنشاء توقيع باستخدام مفتاح خاص

$base64 = $algorithm->sign($private_key);

signature() - الحصول على/تعيين التوقيع الذي تعمل عليه هذه الخوارزمية

my $old_signature = $algorithm->signature;
$algorithm->signature($new_signature);

verify() - التحقق من توقيع

$result = $algorithm->verify();

يجب استدعاؤها بعد finish_body().

النتيجة هي قيمة صحيحة/خاطئة: تشير القيمة الصحيحة إلى أن بيانات التوقيع صالحة، وتشير القيمة الخاطئة إلى أنها غير صالحة.

بالنسبة للتوقيع غير الصالح، يمكن الحصول على التفاصيل من $algorithm->{verification_details} أو $@.
