Mail__Internet(3pm)

الحزمة: perl-mailtools

التراخيص: PerlArtistic, GPL

Mail::Internet(3) توثيق مساهمي Perl Mail::Internet(3)

الاسم

Mail::Internet - التعامل مع رسائل البريد الإلكتروني

الموجز

use Mail::Internet;
my $msg = Mail::Internet->new(\*STDIN);

الوصف

هذه الحزمة تطبق قراءة، إنشاء، معالجة، وكتابة رسائل البريد الإلكتروني. في بعض الأحيان، تحاول التطبيق أن يكون ذكيًا جدًا، لكن في الحالة العامة يعمل كما هو متوقع.

إذا كنت تكتب تطبيقًا جديدًا، يجب عليك استخدام توزيع Mail::Box، والذي يحتوي على المزيد من الميزات ويتعامل مع الرسائل بشكل أفضل وفقًا لمعايير RFCs. انظر إلى http://perl.overmeer.net/mailbox/. يمكنك أيضًا اختيار MIME::Entity، للحصول على دعم متعدد الأجزاء واحد على الأقل في تطبيقك.

الطرق

المُنشِئات

$obj->dup()
ينسخ الرسالة بالكامل. يتم نسخ كل من الرأس والجسم بعمق: يتم إرجاع كائن Mail::Internet جديد.
$obj->extract(\@lines)
يستخرج الرأس والجسم من مصفوفة من أسطر الرسالة. يتطلب كائنًا تم إنشاؤه بالفعل باستخدام new()، وسيتم الكتابة فوق محتوياته.
$obj->new( [$arg], [%options] )
Mail::Internet->new( [$arg], [%options] )
$arg اختياري ويمكن أن يكون واصف ملف (مرجع GLOB) أو مرجع لمصفوفة. إذا تم تقديمه، فسيتم تهيئة الكائن الجديد بالرؤوس والأجسام من قائمة الأسطر المقروءة من واصف الملف.

يمكن أيضًا تقديم معلمات Mail::Header::new() %options "Modify"، "MailFrom" و "FoldLength".

-Option--Default
 Body    []
 Header  undef
Body => ARRAY-of-LINES
يجب أن تكون قيمة هذا الخيار مرجعًا لمصفوفة تحتوي على أسطر جسم الرسالة. يجب أن ينتهي كل سطر بـ "\n" (LF). إذا تم توفير Body، فلن يحاول "Mail::Internet" قراءة الجسم من $arg (حتى لو تم تحديده).
Header => Mail::Header
يجب أن تكون قيمة هذا الخيار كائن Mail::Header. إذا تم تقديمه، فلن يحاول "Mail::Internet" قراءة رأس البريد من $arg، إذا تم تحديده.
$obj->read($fh)
يقرأ رسالة من $fh في كائن رسالة موجود بالفعل. يوصى باستخدام new() مع $fh كحجة أولى.

المُوصِلات

$obj->body( [$body] )
يعيد جسم الرسالة. هذا مرجع لمصفوفة. يمثل كل إدخال في المصفوفة سطرًا من الرسالة.

إذا تم تقديم $body، فيمكن أن يكون مرجعًا لمصفوفة أو مصفوفة، وفي هذه الحالة سيتم استبدال الجسم. إذا تم تمرير مرجع، فسيتم استخدامه مباشرة ولن يتم نسخه، لذلك فإن أي تغييرات لاحقة لهذه المصفوفة تؤثر على محتوى الجسم.

$obj->head()
يعيد كائن "Mail::Header" الذي يحتوي على رؤوس الرسالة الحالية.

معالجة الرسالة ككل

$obj->as_mbox_string( [$already_escaped] )
يعيد الرسالة كسلسلة نصية بتنسيق mbox. $already_escaped، إذا كان صحيحًا، يشير إلى أنه تم استدعاء escape_from() على هذا الكائن.
$obj->as_string()
يعيد الرسالة كسلسلة نصية واحدة.
$obj->print( [$fh] )
يطبع الرأس أو الجسم أو الرسالة الكاملة إلى واصف الملف $fh. $fd يجب أن يكون مرجعًا لـ GLOB. إذا لم يتم تقديم $fh، فسيتم إرسال الإخراج إلى STDOUT.

مثال:

$mail->print( \*STDOUT );  # يطبع الرسالة إلى STDOUT
$obj->print_body( [$fh] )
يطبع الجسم فقط إلى $fh (الافتراضي هو STDOUT).
$obj->print_header( [$fh] )
يطبع الرأس فقط إلى $fh (الافتراضي هو STDOUT).

معالجة الرأس

العديد من هذه الطرق هي مجرد تغليف حول الطرق التي توفرها Mail::Header.

$obj->add(PAIRS)
PAIRS هي الحقل ومحتوى الحقل. لكل PAIR، يتم استدعاء Mail::Header::add(). تتم إضافة جميع الحقول بعد الحقول الموجودة. يتم إرجاع الأخير الذي تم إضافته.
$obj->combine( $tag, [$with] )
انظر Mail::Header::combine().
$obj->delete( $tag, [$tags] )
يحذف جميع الحقول التي تحمل الاسم $tag. Mail::Header::delete() يقوم بالمهمة.
$obj->fold( [$length] )
انظر Mail::Header::fold().
$obj->fold_length( [$tag], [$length] )
انظر Mail::Header::fold_length().
$obj->get( $tag, [$tags] )
في سياق قائمة، يتم إرجاع جميع الحقول التي تحمل الاسم $tag. في سياق عددي، يتم إرجاع أول حقل مطابق لأول $tag تم العثور عليه فقط. يتم استدعاء Mail::Header::get() لجمع البيانات.
$obj->header(\@lines)
انظر Mail::Header::header().
$obj->replace(PAIRS)
PAIRS هي الحقل ومحتوى الحقل. لكل PAIR، يتم استدعاء Mail::Header::replace() مع الفهرس 0. إذا كان الحقل $field موجودًا بالفعل في الرأس، فسيتم حذفه أولاً. لا يجب تحديد نفس اسم الحقل مرتين.

معالجة الجسم

$obj->remove_sig( [$nlines] )
يحاول إزالة توقيع المستخدم من جسم الرسالة. يتم ذلك عن طريق البحث عن سطر مطابق لـ '-- ' في آخر $nlines من الرسالة. إذا تم العثور عليه، فسيتم حذف هذا السطر وجميع ما بعده. إذا لم يتم تحديد $nlines، فسيتم استخدام قيمة افتراضية تبلغ 10. سيكون هذا أكثر فائدة في نصوص الرد التلقائي.
$obj->sign(%options)
يضيف توقيعك إلى الجسم. ستقوم remove_sig() بإزالة التوقيعات الموجودة أولاً. 
-Option--Default
 File       undef
 Signature  []
File => FILEHANDLE
يأخذ من سطر الملف $FILEHANDLE جميع الأسطر بدءًا من أول "--".
Signature => STRING|ARRAY-of-LINES
$obj->tidy_body()
يزيل جميع الأسطر في بداية ونهاية الجسم التي تحتوي فقط على مسافات بيضاء.

وظائف عالية المستوى

$obj->escape_from()
يمكن أن يسبب مشاكل في بعض التطبيقات إذا كانت الرسالة تحتوي على سطر يبدأ بـ `From '، خاصة عند محاولة تقسيم مجلد. تقوم هذه الطريقة بإدراج "`" في بداية أي سطر يطابق النمط المنتظم "/^"*From/>
$obj->nntppost( [%options] )
ينشر مقالًا باستخدام NNTP. يتطلب تثبيت Net::NNTP. 
-Option--Default
 Debug   <false>
 Host    <required>
 Port    119
>
Debug => BOOLEAN
قيمة التصحيح (Debug) لتمريرها إلى Net::NNTP، انظر Net::NNTP
Host => HOSTNAME|Net::NNTP object
اسم خادم NNTP للاتصال به، أو كائن Net::NNTP للاستخدام.
Port => INTEGER
رقم المنفذ للاتصال به على المضيف البعيد
$obj->reply(%options)
إنشاء كائن جديد مع تهيئة الرأس للرد على الكائن الحالي. وسيكون الجسم نسخة من الرسالة الحالية مع إزاحة.

سيتم قراءة ملف ".mailhdr" في دليلك الرئيسي (إذا كان موجودًا) أولاً، لتوفير القيم الافتراضية.

-Option  --Default
 Exclude   []
 Indent    '>'
 Keep      []
 ReplyAll  false
Exclude => ARRAY-of-FIELDS
إزالة الحقول المدرجة من الرسالة المنتجة.
Indent => STRING
استخدام كسلسلة إزاحة. قد تحتوي السلسلة على "%%" للحصول على "%" واحد، %f للحصول على الاسم الأول، %F هو الحرف الأول من %f، %l هو الاسم الأخير، %L هو حرفه الأول، %n هو السلسلة الكاملة من، و %I هو الحرف الأول من كل اسم في السلسلة الكاملة.
Keep => ARRAY-of-FIELDS
نسخ الحقول المدرجة من الرسالة الأصلية.
ReplyAll => BOOLEAN
تضمين تلقائي لجميع عناوين "إلى" و "نسخة" في البريد الأصلي، باستثناء تلك المذكورة في قائمة "نسخة مخفية".
$obj->send( [$type, [$args...]] )
إرسال رسالة Mail::Internet باستخدام Mail::Mailer. يتم تمرير $type و $args إلى Mail::Mailer::new().
$obj->smtpsend( [%options] )
إرسال رسالة Mail::Internet باستخدام SMTP مباشر إلى العناوين المحددة، يمكن أن يكون كل منها سلسلة نصية أو مرجعًا لقائمة عناوين البريد الإلكتروني. إذا لم يتم تحديد أي من "To" أو "Cc" أو "Bcc"، فسيتم استخراج العناوين من الرسالة التي يتم إرسالها.

ستكون القيمة المرجعة قائمة بعناوين البريد الإلكتروني التي تم إرسال الرسالة إليها. إذا لم يتم إرسال الرسالة، فستكون القائمة فارغة.

يتطلب تثبيت Net::SMTP و Net::Domain.

-Option  --Default
 Bcc       undef
 Cc        undef
 Debug     <false>
 Hello     localhost.localdomain
 Host      $ENV{SMTPHOSTS}
 MailFrom  Mail::Util::mailaddress()
 Port      25
 To        undef
Bcc => ADDRESSES
Cc => ADDRESSES
Debug => BOOLEAN
قيمة التصحيح (Debug) لتمريرها إلى Net::SMTP، انظر <Net::SMTP>
Hello => STRING
إرسال أمر HELO (أو EHLO) إلى الخادم بالاسم المحدد.
Host => HOSTNAME
اسم خادم SMTP للاتصال به، أو كائن Net::SMTP للاستخدام.

إذا لم يتم تحديد "Host"، فسيتم العثور على مضيف SMTP عن طريق محاولة الاتصال أولاً بالمضيفين المحددين في $ENV{SMTPHOSTS}، وهو قائمة مفصولة بنقطتين، ثم "mailhost" و "localhost".

MailFrom => ADDRESS
عنوان البريد الإلكتروني المستخدم كمرسل. افتراضيًا، يوفر Mail::Util::mailaddress() عنوان المرسل.
Port => INTEGER
رقم المنفذ للاتصال به على المضيف البعيد
To => ADDRESSES
$obj->unescape_from(())
إزالة إلغاء التشفير الذي أضافته escape_from().
