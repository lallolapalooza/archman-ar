Mail__DKIM__ARC__Verifier(3pm)

الحزمة: perl-mail-dkim

التراخيص: PerlArtistic, GPL

Mail::DKIM::ARC::Verifier(3) توثيق Perl مقدم من المستخدم Mail::DKIM::ARC::Verifier(3)

الاسم

Mail::DKIM::ARC::Verifier - يتحقق من صحة رسالة مختومة بـ ARC

الإصدار

version 1.20240923

الملخص

use Mail::DKIM::ARC::Verifier;
# إنشاء كائن مدقق
my $arc = Mail::DKIM::ARC::Verifier->new();
# قراءة بريد إلكتروني من مقبض ملف
$arc->load(*STDIN);
# أو قراءة بريد إلكتروني وتمريره إلى المدقق، بشكل تدريجي
while (<STDIN>)
{
    # إزالة نهايات الأسطر المحلية
    chomp;
    s/\015$//;
    # استخدام نهايات أسطر SMTP
    $arc->PRINT("$_\015\012");
}
$arc->CLOSE;
# ما هي نتيجة التحقق؟
my $result = $arc->result;
# طباعة النتائج لجميع تواقيع الرسالة والأختام الموجودة على الرسالة
foreach my $signature ($arc->signatures)
{
    print $signature->prefix() . ' v=' . $signature->instance .
                                   ' ' . $signature->result_detail . "\n";
}
# مثال على المخرجات. لاحظ أنه لتمرير التحقق، يجب أن يتطابق فقط آخر
# توقيع ARC-Message-Signature، لأن خطوات أخرى قد تكون قد عدلت التوقيع. ما يهم
# هو أن جميع ARC-Seals تمر، وأن آخر ARC-Message-Signature يمر.

الوصف

يسمح كائن المدقق بفحص رسالة بريد إلكتروني بحثًا عن أختام ARC وتحقق من صحة تواقيعها المرتبطة. يتتبع المدقق حالة الرسالة أثناء قراءتها في الذاكرة. عند قراءة الرسالة بالكامل، يتم التحقق من صحة التواقيع ويمكن الوصول إلى نتائج التحقق.

لاستخدام المدقق، قم أولاً بإنشاء كائن المدقق. ثم ابدأ في "تغذية" الرسالة المراد التحقق منها. عند قراءة جميع _الترويسات_، يقوم المدقق بما يلي:

1. التحقق مما إذا تم العثور على أي تواقيع ARC
2. الاستعلام عن المفاتيح العامة اللازمة للتحقق من صحة التواقيع
3. إعداد الخوارزميات وكائنات التوحيد القياسي المناسبة
4. توحيد الترويسات وحساب تجزئة الترويسة

ثم، عند تغذية _جسم_ الرسالة بالكامل إلى المدقق، يتم حساب تجزئة الجسم والتحقق من صحة التواقيع.

يمكن التحقق من نتائج التحقق باستخدام "result()" أو "signatures()".

يتم حساب النتيجة النهائية بواسطة الخوارزمية الموضحة في https://tools.ietf.org/html/draft-ietf-dmarc-arc-protocol-06 - إذا مرت جميع ترويسات ARC-Seal وكان توقيع ARC-Message-Signature الأعلى فهرسًا (i=) يمر، فإن الختم سليم.

المُنشئ

new()

ينشئ مدققًا كائني التوجه.

my $arc = Mail::DKIM::ARC::Verifier->new();
my $arc = Mail::DKIM::ARC::Verifier->new(%options);

الخيارات الوحيدة المدعومة في هذا الوقت هي:

AS_Canonicalization
إذا تم تحديده، يتم كتابة الرسالة الموحدة لـ ARC-Seal إلى السلسلة المرجعية أو مقبض الملف.
AMA_Canonicalization
إذا تم تحديده، يتم كتابة الرسالة الموحدة لـ ARC-Message-Signature إلى السلسلة المرجعية أو مقبض الملف.
Strict
إذا كانت القيمة صحيحة، يتم رفض تجزئات sha1 ومفاتيح التوقيع الأقصر من 1024 بت.

الأساليب

PRINT()

يغذي جزءًا من الرسالة إلى المدقق.

$arc->PRINT("a line of the message\015\012");
$arc->PRINT('more of');
$arc->PRINT(" the message\015\012bye\015\012");

يغذي محتوى الرسالة التي يتم التحقق منها في المدقق. تم تصميم واجهة برمجة التطبيقات بهذه الطريقة بحيث لا يلزم قراءة الرسالة بأكملها في الذاكرة مرة واحدة.

يرجى ملاحظة أنه على الرغم من أن طريقة PRINT() تتوقع منك استخدام أحرف إنهاء الأسطر بأسلوب SMTP، إلا أنه لا ينبغي عليك استخدام تقنية حشو النقاط بأسلوب SMTP الموضحة في RFC 2821 القسم 4.5.2. ولا يجب عليك استخدام تسلسل <CR><LF>.<CR><LF> لإنهاء الرسالة.

CLOSE()

استدعِ هذا عند الانتهاء من تغذية الرسالة.

$arc->CLOSE;

تنهي هذه الطريقة عملية التوحيد القياسي، وتحسب التجزئة، وتتحقق من صحة التوقيع.

load()

تحميل الرسالة بأكملها من مقبض ملف.

$arc->load($file_handle);

يقرأ رسالة كاملة من مقبض الملف المحدد، ويغذيها في المدقق. يجب أن تستخدم الرسالة نهايات أسطر <CRLF> (نفس بروتوكول SMTP).

message_originator()

الوصول إلى ترويسة "From".

my $address = $arc->message_originator;

يعيد "عنوان المرسل" الموجود في الرسالة، ككائن Mail::Address. هذا عادةً هو الاسم (الأول) وعنوان البريد الإلكتروني الموجود في ترويسة From:. إذا لم تكن هناك ترويسة From:، يتم إرجاع كائن Mail::Address فارغ.

للحصول على جزء عنوان البريد الإلكتروني فقط، قم بما يلي:

my $email = $arc->message_originator->address;

انظر أيضًا "message_sender()".

message_sender()

الوصول إلى ترويسة "From" أو "Sender".

my $address = $arc->message_sender;

يعيد "المرسل" الموجود في الرسالة، ككائن Mail::Address. هذا عادةً هو الاسم (الأول) وعنوان البريد الإلكتروني الموجود في ترويسة Sender:. إذا لم تكن هناك ترويسة Sender:، فإنه يكون الاسم (الأول) وعنوان البريد الإلكتروني في ترويسة From:. إذا لم تكن أي من الترويسات موجودة، فسيتم إرجاع كائن Mail::Address فارغ.

للحصول على جزء عنوان البريد الإلكتروني فقط، قم بما يلي:

my $email = $arc->message_sender->address;

"المرسل" هو صندوق البريد الخاص بالوكيل المسؤول عن الإرسال الفعلي للرسالة. على سبيل المثال، إذا قامت سكرتيرة بإرسال رسالة لشخص آخر، فسيكون "المرسل" هو السكرتيرة وسيكون "المرسل الأصلي" هو المؤلف الفعلي.

result()

الوصول إلى نتيجة التحقق.

my $result = $arc->result;

يعطي نتيجة التحقق. القيم التالية ممكنة:

pass
يتم إرجاعه إذا تم العثور على سلسلة ARC صالحة، مع مرور جميع ARC-Seals، وأن آخر (أعلى فهرس) ARC-Message-Signature يمر.
fail
يتم إرجاعه إذا فشل أي ARC-Seal، أو إذا فشل ARC-Message-Signature. سيتم أيضًا فشل إذا كان هناك فشل مؤقت في DNS، وهو عيب معروف في هذا الإصدار من ARC::Verifier. قد ترفض الإصدارات المستقبلية هذه الرسالة بشكل قاطع (4xx) وتطلب من المرسل محاولة التسليم لاحقًا لتجنب إنشاء سلسلة معطلة. لا يوجد خطأ مؤقت لـ ARC، لأنه لا معنى لتوقيع سلسلة تحتوي على خطأ مؤقت أو أن كل مرسل بريد عشوائي سيستخدم أحد هذه الأخطاء.
invalid
يتم إرجاعه إذا تعذر التحقق من ARC-Seal بسبب مشكلة في التوقيع نفسه أو سجل المفتاح العام. أي أنه تعذر معالجة التوقيع.
none
يتم إرجاعه إذا لم يتم العثور على ترويسات ARC-*.

result_detail()

الوصول إلى النتيجة، بالإضافة إلى التفاصيل إذا كانت متاحة.

my $detail = $dkim->result_detail;

يتم بناء التفاصيل عن طريق أخذ النتيجة (مثل "pass"، "fail"، "invalid" أو "none") وإلحاق أي تفاصيل قدمتها عملية التحقق لأعلى ARC-Seal في أقواس.

النتائج التالية ممكنة من طريقة result_detail():

pass
fail (bad RSA signature)
fail (OpenSSL error: ...)
fail (message has been altered)
fail (body has been altered)
invalid (bad instance)
invalid (invalid domain in d tag)
invalid (missing q tag)
invalid (missing d tag)
invalid (missing s tag)
invalid (unsupported version 0.1)
invalid (unsupported algorithm ...)
invalid (unsupported canonicalization ...)
invalid (unsupported query protocol ...)
invalid (signature is expired)
invalid (public key: not available)
invalid (public key: u
