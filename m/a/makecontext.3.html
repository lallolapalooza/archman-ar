makecontext(3) دليل دوال المكتبة makecontext(3)

الاسم

makecontext, swapcontext - معالجة سياق المستخدم

المكتبة

مكتبة C القياسية (libc, -lc)

الملخص

#include <ucontext.h>
void makecontext(ucontext_t *ucp, typeof(void (int arg0, ...)) *func,
                 int argc, ...);
int swapcontext(ucontext_t *restrict oucp,
                 const ucontext_t *restrict ucp);

الوصف

في بيئة شبيهة بـ System V، توجد الأنواع ucontext_t (المعرفة في <ucontext.h> والموصوفة في getcontext(3)) والدوال الأربع getcontext(3)، setcontext(3)، makecontext()، و swapcontext() التي تسمح بالتبديل بين خيوط التحكم المتعددة داخل العملية على مستوى المستخدم.

تقوم الدالة makecontext() بتعديل السياق الذي يشير إليه ucp (والذي تم الحصول عليه من استدعاء getcontext(3)). قبل استدعاء makecontext()، يجب على المستدعي تخصيص مكدس جديد لهذا السياق وتعيين عنوانه إلى ucp->uc_stack، وتحديد سياق لاحق وتعيين عنوانه إلى ucp->uc_link.

عندما يتم تنشيط هذا السياق لاحقًا (باستخدام setcontext(3) أو swapcontext())، يتم استدعاء الدالة func، وتمرير سلسلة من الوسائط الصحيحة (int) التي تتبع argc؛ يجب على المستدعي تحديد عدد هذه الوسائط في argc. عند عودة هذه الدالة، يتم تنشيط السياق اللاحق. إذا كان مؤشر السياق اللاحق NULL، تنتهي الخيط.

تقوم الدالة swapcontext() بحفظ السياق الحالي في الهيكل الذي يشير إليه oucp، ثم تنشط السياق الذي يشير إليه ucp.

قيمة الإرجاع

عند النجاح، لا تعود الدالة swapcontext(). (ولكن قد نعود لاحقًا، في حالة تنشيط oucp، وفي هذه الحالة تبدو الدالة swapcontext() وكأنها تعود بالقيمة 0). عند حدوث خطأ، تعود الدالة swapcontext() بالقيمة -1 وتقوم بتعيين errno للإشارة إلى الخطأ.

الأخطاء

ENOMEM
مساحة مكدس غير كافية متبقية.

السمات

للحصول على شرح للمصطلحات المستخدمة في هذا القسم، راجع attributes(7).

الواجهة السمة القيمة
makecontext () خيط السلامة MT-Safe race:ucp
swapcontext () خيط السلامة MT-Safe race:oucp race:ucp

المعايير

لا شيء.

التاريخ

glibc 2.1. SUSv2, POSIX.1-2001. تمت إزالته في POSIX.1-2008، مشيرًا إلى مشاكل قابلية النقل، ويوصي بإعادة كتابة التطبيقات لاستخدام خيوط POSIX بدلاً من ذلك.

ملاحظات

تفسير ucp->uc_stack هو كما هو الحال في sigaltstack(2)، أي أن هذا الهيكل يحتوي على بداية وطول منطقة ذاكرة لاستخدامها كمكدس، بغض النظر عن اتجاه نمو المكدس. وبالتالي، لا يلزم أن يقلق برنامج المستخدم بشأن هذا الاتجاه.

على المعماريات التي تكون فيها أنواع int والمؤشرات بنفس الحجم (مثل x86-32، حيث كلا النوعين 32 بت)، قد تتمكن من تمرير مؤشرات كوسائط إلى makecontext() بعد argc. ومع ذلك، فإن القيام بذلك غير مضمون ليكون قابلًا للنقل، وغير محدد وفقًا للمعايير، ولن يعمل على المعماريات التي تكون فيها المؤشرات أكبر من ints. ومع ذلك، بدءًا من glibc 2.8، أجرت glibc بعض التغييرات على makecontext()، للسماح بذلك على بعض المعماريات 64 بت (مثل x86-64).

أمثلة

يوضح برنامج المثال التالي استخدام getcontext(3)، makecontext()، و swapcontext(). ينتج عن تشغيل البرنامج الإخراج التالي:


$ ./a.out
main: swapcontext(&uctx_main, &uctx_func2)
func2: started
func2: swapcontext(&uctx_func2, &uctx_func1)
func1: started
func1: swapcontext(&uctx_func1, &uctx_func2)
func2: returning
func1: returning
main: exiting

مصدر البرنامج

#include <err.h>
#include <stdio.h>
#include <stdlib.h>
#include <ucontext.h>
static ucontext_t uctx_main, uctx_func1, uctx_func2;
static void
func1(void)
{
    printf("%s: started\n", __func__);
    printf("%s: swapcontext(&uctx_func1, &uctx_func2)\n", __func__);
    if (swapcontext(&uctx_func1, &uctx_func2) == -1)
        err(EXIT_FAILURE, "swapcontext");
    printf("%s: returning\n", __func__);
}
static void
func2(void)
{
    printf("%s: started\n", __func__);
    printf("%s: swapcontext(&uctx_func2, &uctx_func1)\n", __func__);
    if (swapcontext(&uctx_func2, &uctx_func1) == -1)
        err(EXIT_FAILURE, "swapcontext");
    printf("%s: returning\n", __func__);
}
int
main(int argc, char *argv[])
{
    char func1_stack[16384];
    char func2_stack[16384];
    if (getcontext(&uctx_func1) == -1)
        err(EXIT_FAILURE, "getcontext");
    uctx_func1.uc_stack.ss_sp = func1_stack;
    uctx_func1.uc_stack.ss_size = sizeof(func1_stack);
    uctx_func1.uc_link = &uctx_main;
    makecontext(&uctx_func1, func1, 0);
    if (getcontext(&uctx_func2) == -1)
        err(EXIT_FAILURE, "getcontext");
    uctx_func2.uc_stack.ss_sp = func2_stack;
    uctx_func2.uc_stack.ss_size = sizeof(func2_stack);
    /* Successor context is f1(), unless argc > 1 */
    uctx_func2.uc_link = (argc > 1) ? NULL : &uctx_func1;
    makecontext(&uctx_func2, func2, 0);
    printf("%s: swapcontext(&uctx_main, &uctx_func2)\n", __func__);
    if (swapcontext(&uctx_main, &uctx_func2) == -1)
        err(EXIT_FAILURE, "swapcontext");
    printf("%s: exiting\n", __func__);
    exit(EXIT_SUCCESS);
}
