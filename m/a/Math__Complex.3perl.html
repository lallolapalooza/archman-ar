Math__Complex.3perl

Math::Complex(3perl) دليل مبرمجي Perl Math::Complex(3perl)

الاسم

Math::Complex — الأعداد المركبة وما يرتبط بها من دوال رياضية

الملخص

use Math::Complex;
$z = Math::Complex->make(5, 6);
$t = 4 - 3*i + $z;
$j = cplxe(1, 2*pi/3);

الوصف

تتيح لك هذه الحزمة إنشاء ومعالجة الأعداد المركبة. افتراضيًا، تقتصر Perl على الأعداد الحقيقية، ولكن عبارة "use" إضافية تجلب دعمًا كاملاً للأعداد المركبة، إلى جانب مجموعة كاملة من الدوال الرياضية المرتبطة عادةً بالأعداد المركبة و/أو الممتدة إليها.

إذا كنت تتساءل ما هي الأعداد المركبة، فقد تم اختراعها لتكون قادرة على حل المعادلة التالية:

x*x = -1

وبحكم التعريف، الحل هو i (يستخدم المهندسون j بدلاً من ذلك لأن i تشير عادةً إلى شدة التيار، ولكن الاسم لا يهم). الرقم i هو عدد خيالي بحت.

الحسابات مع الأعداد الخيالية البحتة تعمل تمامًا كما تتوقعها مع الأعداد الحقيقية... عليك فقط أن تتذكر أن

i*i = -1

لذلك لديك:

5i + 7i = i * (5 + 7) = 12i
4i - 3i = i * (4 - 3) = i
4i * 2i = -8
6i / 2i = 3
1 / i = -i

الأعداد المركبة هي أعداد لها جزء حقيقي وجزء خيالي، وعادة ما يتم ترميزها:

a + bi

حيث "a" هو الجزء الحقيقي و "b" هو الجزء الخيالي. الحسابات مع الأعداد المركبة مباشرة. عليك تتبع الأجزاء الحقيقية والخيالية، ولكن بخلاف ذلك، تنطبق القواعد المستخدمة للأعداد الحقيقية:

(4 + 3i) + (5 - 2i) = (4 + 5) + i(3 - 2) = 9 + i
(2 + i) * (4 - i) = 2*4 + 4i -2i -i*i = 8 + 2i + 1 = 9 + 2i

يمكن تمثيل الأعداد المركبة بيانيًا في مستوى (يسمى أيضًا المستوى المركب، ولكنه في الواقع مستوى ثنائي الأبعاد). العدد

z = a + bi

هو النقطة التي إحداثياتها (a, b). في الواقع، سيكون المتجه الذي ينشأ من (0، 0) إلى (a، b). يتبع ذلك أن إضافة عددين مركبين هي إضافة متجهية.

نظرًا لوجود تطابق واحد لواحد بين نقطة في المستوى ثنائي الأبعاد وعدد مركب (أي أن التعيين فريد ومتبادل)، يمكن أيضًا تحديد عدد مركب بشكل فريد باستخدام الإحداثيات القطبية:

[rho, theta]

حيث "rho" هي المسافة إلى نقطة الأصل، و "theta" هي الزاوية بين المتجه والمحور x. هناك تدوين لهذا باستخدام الشكل الأسي، وهو:

rho * exp(i * theta)

حيث i هو الرقم الخيالي الشهير الذي تم تقديمه أعلاه. التحويل بين هذا الشكل والشكل الديكارتي "a + bi" فوري:

a = rho * cos(theta)
b = rho * sin(theta)

وهو ما يتم التعبير عنه أيضًا بهذه الصيغة:

z = rho * exp(i * theta) = rho * (cos theta + i * sin theta)

بمعنى آخر، هو إسقاط المتجه على المحورين x و y. يسمي علماء الرياضيات rho المعيار أو القيمة المطلقة و theta الحجة للعدد المركب. معيار "z" مُرمز هنا كـ abs(z).

التدوين القطبي (المعروف أيضًا بالتمثيل المثلثي) أكثر ملاءمة لإجراء عمليات الضرب والقسمة للأعداد المركبة، بينما التدوين الديكارتي أكثر ملاءمة للجمع والطرح. الأعداد الحقيقية موجودة على المحور x، وبالتالي فإن y أو theta تساوي صفرًا أو pi.

تم تعريف جميع العمليات الشائعة التي يمكن إجراؤها على عدد حقيقي للعمل على الأعداد المركبة أيضًا، وهي مجرد امتدادات للعمليات المعرفة على الأعداد الحقيقية. هذا يعني أنها تحافظ على معناها الطبيعي عندما لا يوجد جزء خيالي، بشرط أن يكون العدد ضمن مجموعة تعريفها.

على سبيل المثال، يتم تعريف الروتين "sqrt" الذي يحسب الجذر التربيعي لوسيطه فقط للأعداد الحقيقية غير السالبة ويؤدي إلى عدد حقيقي غير سالب (إنه تطبيق من R+ إلى R+). إذا سمحنا له بإرجاع عدد مركب، فيمكن تمديده إلى الأعداد الحقيقية السالبة ليصبح تطبيقًا من R إلى C (مجموعة الأعداد المركبة):

sqrt(x) = x >= 0 ? sqrt(x) : sqrt(-x)*i

يمكن أيضًا تمديده ليصبح تطبيقًا من C إلى C، بينما يتصرف تقييده على R كما هو محدد أعلاه باستخدام التعريف التالي:

sqrt(z = [r,t]) = sqrt(r) * exp(i * t/2)

بالفعل، يمكن ترميز العدد الحقيقي السالب كـ "[x,pi]" (المعيار x غير سالب دائمًا، لذا فإن "[x,pi]" هو حقًا "-x"، وهو عدد سالب) والتعريف أعلاه يذكر أن

sqrt([x,pi]) = sqrt(x) * exp(i*pi/2) = [sqrt(x),pi/2] = sqrt(x)*i

وهو بالضبط ما حددناه للأعداد الحقيقية السالبة أعلاه. ترجع "sqrt" حلاً واحدًا فقط: إذا كنت تريد كلا الحلين، استخدم الدالة "root".

جميع الدوال الرياضية الشائعة المعرفة على الأعداد الحقيقية والتي تم تمديدها إلى الأعداد المركبة تشترك في نفس الخاصية وهي العمل كما هو معتاد عندما يكون الجزء الخيالي صفرًا (وإلا، فلن يطلق عليه امتداد، أليس كذلك؟).

عملية جديدة ممكنة على عدد مركب وهي الهوية للأعداد الحقيقية تسمى المرافق، ويتم ترميزها بشريط أفقي فوق العدد، أو "~z" هنا.

 z = a + bi
~z = a - bi

بسيط... الآن انظر:

z * ~z = (a + bi) * (a - bi) = a*a + b*b

لقد رأينا أن معيار "z" تم ترميزه كـ abs(z) وتم تعريفه على أنه المسافة إلى نقطة الأصل، المعروفة أيضًا باسم:

rho = abs(z) = sqrt(a*a + b*b)

لذا

z * ~z = abs(z) ** 2

إذا كان z عددًا حقيقيًا بحتًا (أي "b == 0")، فإن ما سبق يعطي:

a * a = abs(a) ** 2

وهو صحيح ( "abs" له المعنى العادي للعدد الحقيقي، أي يرمز إلى القيمة المطلقة). يوضح هذا المثال لماذا يتم ترميز معيار "z" بـ abs(z): إنه يمدد الدالة "abs" إلى الأعداد المركبة، ولكنه لا يزال "abs" العادي الذي نعرفه عندما لا يكون للعدد المركب جزء خيالي بالفعل... هذا يبرر بعد الواقعة استخدامنا لتدوين "abs" للمعيار.

العمليات

بالنظر إلى التدوينات التالية:

z1 = a + bi = r1 * exp(i * t1)
z2 = c + di = r2 * exp(i * t2)
z = <any complex or real number>

العمليات التالية (المُحمّلة بشكل زائد) مدعومة على الأعداد المركبة:

z1 + z2 = (a + c) + i(b + d)
z1 - z2 = (a - c) + i(b - d)
z1 * z2 = (r1 * r2) * exp(i * (t1 + t2))
z1 / z2 = (r1 / r2) * exp(i * (t1 - t2))
z1 ** z2 = exp(z2 * log z1)
~z = a - bi
abs(z) = r1 = sqrt(a*a + b*b)
sqrt(z) = sqrt(r1) * exp(i * t/2)
exp(z) = exp(a) * exp(i * b)
log(z) = log(r1) + i*t
sin(z) = 1/2i (exp(i * z1) - exp(-i * z))
cos(z) = 1/2 (exp(i * z1) + exp(-i * z))
atan2(y, x) = atan(y / x) # مع مراعاة الربع الصحيح، لاحظ الترتيب.

التعريف المستخدم للوسائط المركبة لـ atan2() هو

-i log((x + iy)/sqrt(x*x+y*y))

لاحظ أن atan2(0, 0) غير معرف بشكل جيد.

العمليات الإضافية التالية مدعومة على كل من الأعداد الحقيقية والمركبة:

Re(z) = a
Im(z) = b
arg(z) = t
abs(z) = r
cbrt(z) = z ** (1/3)
log10(z) = log(z) / log(10)
logn(z, n) = log(z) / log(n)
tan(z) = sin(z) / cos(z)
csc(z) = 1 / sin(z)
sec(z) = 1 / cos(z)
cot(z) = 1 / tan(z)
asin(z) = -i * log(i*z + sqrt(1-z*z))
acos(z) = -i * log(z + i*sqrt(1-z*z))
atan(z) = i/2 * log((i+z) / (i-z))
acsc(z) = asin(1 / z)
asec(z) = acos(1 / z)
acot(z) = atan(1 / z) = -i/2 * log((i+z) / (z-i))
sinh(z) = 1/2 (exp(z) - exp(-z))
cosh(z) = 1/2 (exp(z) + exp(-z))
tanh(z) = sinh(z) / cosh(z) = (exp(z) - exp(-z)) / (exp(z) + exp(-z))
csch(z) = 1 / sinh(z)
sech(z) = 1 / cosh(z)
coth(z) = 1 / tanh(z)
asinh(z) = log(z + sqrt(z*z+1))
acosh(z) = log(z + sqrt(z*z-1))
atanh(z) = 1/2 * log((1+z) / (1-z))
acsch(z) = asinh(1 / z)
asech(z) = acosh(1 / z)
acoth(z) = atanh(1 / z) = 1/2 * log((1+z) / (z-1))

acotanh، ولها أسماء مستعارة rho، theta، ln، cosec، cotan، acosec، acotan، cosech، cotanh، acosech، acotanh، على التوالي. "Re"، "Im"، "arg"، "abs"، "rho"، و "theta" يمكن استخدامها أيضًا كـ mutators. "cbrt" تُرجع حلاً واحدًا فقط: إذا كنت تريد الحلول الثلاثة، استخدم وظيفة "root".

وظيفة root متاحة لحساب جميع الجذور n لبعض الأعداد المركبة، حيث n عدد صحيح موجب تمامًا. هناك بالضبط n جذور، تُرجع كقائمة. الحصول على العدد الذي يسميه علماء الرياضيات "j" بحيث:

1 + j + j*j = 0;

هو أمر بسيط بكتابة:

$j = (root(1, 3))[1];

يُعطى الجذر k لـ "z = [r,t]" بواسطة:

(root(z, n))[k] = r**(1/n) * exp(i * (t + 2*k*pi)/n)

يمكنك إرجاع الجذر k مباشرة عن طريق "root(z, n, k)"، مع بدء الفهرسة من صفر وتنتهي عند n - 1.

تم تعريف عامل مقارنة الأرقام spaceship، <=>، أيضًا. لضمان توافق تقييده على الأعداد الحقيقية مع ما تتوقعه، يتم إجراء المقارنة على الجزء الحقيقي من العدد المركب أولاً، ويتم مقارنة الأجزاء التخيلية فقط عندما تتطابق الأجزاء الحقيقية.

CREATION

لإنشاء عدد مركب، استخدم أحد الخيارين:

$z = Math::Complex->make(3, 4);
$z = cplx(3, 4);

إذا كنت تعرف الشكل الديكارتي للعدد، أو

$z = 3 + 4*i;

إذا كنت تفضل ذلك. لإنشاء عدد باستخدام الشكل القطبي، استخدم إما:

$z = Math::Complex->emake(5, pi/3);
$x = cplxe(5, pi/3);

بدلاً من ذلك. الوسيط الأول هو المعيار، والثاني هو الزاوية (بالراديان، الدائرة الكاملة هي 2*pi). (تلميح: "e" يُستخدم كترميز للأعداد المركبة في الشكل القطبي).

من الممكن كتابة:

$x = cplxe(-3, pi/4);

ولكن سيتم تحويل ذلك بصمت إلى "[3,-3pi/4]"، نظرًا لأن المعيار يجب أن يكون غير سالب (يمثل المسافة إلى نقطة الأصل في المستوى المركب).

من الممكن أيضًا أن يكون العدد المركب وسيطًا في "make"، "emake"، "cplx"، و "cplxe": سيتم استخدام المكون المناسب للوسيط.

$z1 = cplx(-2,  1);
$z2 = cplx($z1, 4);

ستفهم أيضًا "new"، "make"، "emake"، "cplx"، و "cplxe" وسيطًا واحدًا (كسلسلة نصية) من الأشكال

2-3i
-3i
[2,3]
[2,-3pi/4]
[2]

في هذه الحالة، سيتم تحليل المكونات الديكارتية والأسية المناسبة من السلسلة النصية واستخدامها لإنشاء أعداد مركبة جديدة. سيتم تعيين المكون التخيلي والزاوية، على التوالي، إلى الصفر افتراضيًا.

ستفهم أيضًا "new"، "make"، "emake"، "cplx"، و "cplxe" حالة عدم وجود وسائط: هذا يعني صفرًا عاديًا أو (0، 0).

DISPLAYING

عند طباعة عدد مركب، يتم عرضه عادةً على نمط شكله الديكارتي a+bi، ولكن هناك حالات مشروعة يكون فيها نمط الشكل القطبي [r,t] أكثر ملاءمة. تُعرف عملية تحويل العدد المركب إلى سلسلة نصية قابلة للعرض باسم stringification.

عن طريق استدعاء طريقة الفئة "Math::Complex::display_format" وتوفير "polar" أو "cartesian" كوسيط، يمكنك تجاوز نمط العرض الافتراضي، وهو "cartesian". عدم توفير أي وسيط يُرجع الإعدادات الحالية.

يمكن تجاوز هذا الافتراضي لكل رقم على حدة عن طريق استدعاء طريقة "display_format" بدلاً من ذلك. كما في السابق، عدم توفير أي وسيط يُرجع نمط العرض الحالي لهذا الرقم. بخلاف ذلك، سيكون أي شيء تحدده هو نمط العرض الجديد لهذا الرقم المحدد.

على سبيل المثال:

use Math::Complex;
Math::Complex::display_format('polar');
$j = (root(1, 3))[1];
print "j = $j\n";               # Prints "j = [1,2pi/3]"
$j->display_format('cartesian');
print "j = $j\n";               # Prints "j = -0.5+0.866025403784439i"

يحاول نمط القطب التأكيد على الزوايا مثل k*pi/n (حيث n عدد صحيح موجب و k عدد صحيح ضمن [-9، +9])، وهذا يسمى polar pretty-printing.

لعكس عملية الـ stringifying، انظر "make" و "emake".

CHANGED IN PERL 5.6

يمكن الآن استدعاء طريقة فئة "display_format" وطريقة الكائن المقابلة لها "display_format" باستخدام تجزئة وسيطة بدلاً من وسيط واحد فقط.

يمكن تغيير نمط تنسيق العرض القديم، والذي يمكن أن يكون له قيم "cartesian" أو "polar"، باستخدام الوسيط "style".

$j->display_format(style => "polar");

الاتصال ذو الوسيط الواحد يعمل أيضًا.

$j->display_format("polar");

هناك معلمان جديدان للعرض.

الأول هو "format"، وهو سلسلة تنسيق بنمط sprintf() لاستخدامها لكلا الجزأين الرقميين للعدد المركب (الأعداد المركبة). هذا يعتمد إلى حد ما على النظام ولكن في أغلب الأحيان يتوافق مع "%.15g". يمكنك العودة إلى الافتراضي عن طريق تعيين "format" إلى "undef".

# $j من المثال أعلاه
$j->display_format('format' => '%.5f');
print "j = $j\n";               # Prints "j = -0.50000+0.86603i"
$j->display_format('format' => undef);
print "j = $j\n";               # Prints "j = -0.5+0.86603i"

لاحظ أن هذا يؤثر أيضًا على القيم المرجعة لطرق "display_format": في سياق القائمة، سيتم إرجاع تجزئة الوسيط بأكملها، على عكس قيمة الوسيط فقط. هذا تنافر محتمل مع الإصدارات السابقة إذا كنت قد استدعيت طريقة "display_format" في سياق القائمة.

الوسيط الثاني الجديد للعرض هو "polar_pretty_print"، والذي يمكن تعيينه إلى صحيح أو خطأ، والافتراضي هو صحيح. انظر القسم السابق لمعرفة معنى ذلك.

USAGE

بفضل التحميل الزائد، فإن التعامل مع العمليات الحسابية للأعداد المركبة بسيط وشفاف تقريبًا.

إليك بعض الأمثلة:

use Math::Complex;
$j = cplxe(1, 2*pi/3);  # $j ** 3 == 1
print "j = $j, j**3 = ", $j ** 3, "\n";
print "1 + j + j**2 = ", 1 + $j + $j**2, "\n";
$z = -16 + 0*i;                 # Force it to be a complex
print "sqrt($z) = ", sqrt($z), "\n";
$k = exp(i * 2*pi/3);
print "$j - $k = ", $j - $k, "\n";
$z->Re(3);                      # Re, Im, arg, abs,
$j->arg(2);                     # (the last two aka rho, theta)
                                # can be used also as mutators.

CONSTANTS

PI

الثابت "pi" وبعض مضاعفاته المفيدة (pi2، pi4، و pip2 (pi/2) و pip4 (pi/4)) متاحة أيضًا إذا تم تصديرها بشكل منفصل:

use Math::Complex ':pi';
$third_of_circle = pi2 / 3;

Inf

يمكن تصدير لا نهاية النقطة العائمة كروتين فرعي Inf():

use Math::Complex qw(Inf sinh);
my $AlsoInf = Inf() + 42;
my $AnotherInf = sinh(1e42);
print "$AlsoInf is $AnotherInf\n" if $AlsoInf == $AnotherInf;

لاحظ أن الشكل النصي للا نهاية يختلف بين المنصات: يمكن أن يكون على سبيل المثال أيًا من

inf
infinity
INF
1.#INF

أو قد يكون شيئًا آخر.

لاحظ أيضًا أنه في بعض المنصات، قد يؤدي محاولة استخدام اللانهاية في عمليات حسابية إلى تعطل Perl لأن استخدام اللانهاية يتسبب في إرسال SIGFPE أو ما يعادله. الطريقة لتجاهل ذلك هي

local $SIG{FPE} = sub { };
