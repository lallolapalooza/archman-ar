munge_enum(3)

الحزمة: munge

التراخيص: GPL-3.0-or-later, LGPL-3.0-or-later

MUNGE_ENUM(3) MUNGE Uid 'N' Gid Emporium MUNGE_ENUM(3)

الاسم

munge_enum_is_valid, munge_enum_int_to_str, munge_enum_str_to_int - دوال تعداد MUNGE

الملخص

#include <munge.h>

int munge_enum_is_valid (munge_enum_t type, int val);

const char * munge_enum_int_to_str (munge_enum_t type, int val);

int munge_enum_str_to_int (munge_enum_t type, const char *str);

cc `pkg-config --cflags --libs munge` -o foo foo.c

الوصف

تقوم الدالة munge_enum_is_valid() بالتحقق مما إذا كانت القيمة val المعطاة هي تعداد MUNGE صالح من النوع المحدد type في تكوين البرنامج كما تم تجميعه حاليًا. بعض التعدادات تتوافق مع خيارات لا يمكن تمكينها إلا وقت الترجمة.

تقوم الدالة munge_enum_int_to_str() بتحويل تعداد MUNGE val من النوع المحدد type إلى سلسلة نصية.

تقوم الدالة munge_enum_str_to_int() بتحويل السلسلة النصية str غير الحساسة لحالة الأحرف والمختتمة بـ NULL إلى تعداد MUNGE المقابل من النوع المحدد type.

القيمة المعادة

تُرجع الدالة munge_enum_is_valid() قيمة غير صفرية إذا كانت القيمة val المعطاة تعدادًا صالحًا.

تُرجع الدالة munge_enum_int_to_str() سلسلة نصية ثابتة مختتمة بـ NULL، أو NULL عند حدوث خطأ؛ يجب ألا يقوم المستدعي بتحرير هذه السلسلة أو تعديلها.

تُرجع الدالة munge_enum_str_to_int() تعداد MUNGE عند النجاح (أي >= 0)، أو -1 عند حدوث خطأ.

أنواع التعداد

يمكن تحديد أنواع التعداد التالية.

MUNGE_ENUM_CIPHER
تحديد تعدادات لأنواع التشفير المتاحة.
MUNGE_ENUM_MAC
تحديد تعدادات لأنواع MAC المتاحة.
MUNGE_ENUM_ZIP
تحديد تعدادات لأنواع الضغط المتاحة.

الأخطاء

ارجع إلى munge(3) للحصول على قائمة كاملة بالأخطاء.

مثال

يوضح مثال البرنامج التالي كيفية الاستعلام عن قائمة بأنواع التشفير المتاحة.

#include <stdio.h>
#include <stdlib.h>
#include <munge.h>

int
main (int argc, char *argv[])
{
    int i;
    const char *p;
    munge_enum_t t = MUNGE_ENUM_CIPHER;
    for (i = 0; (p = munge_enum_int_to_str (t, i)) != NULL; i++) {
        if (munge_enum_is_valid (t, i)) {
            printf ("%2d = %s\n", i, p);
        }
    }
    exit (EXIT_SUCCESS);
}
