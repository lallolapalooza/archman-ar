munge(3)

الحزمة: munge

التراخيص: GPL-3.0-or-later, LGPL-3.0-or-later

MUNGE(3) MUNGE Uid 'N' Gid Emporium MUNGE(3)

الاسم

munge_encode, munge_decode, munge_strerror - وظائف MUNGE الأساسية

ملخص

#include <munge.h>

munge_err_t munge_encode (char **cred, munge_ctx_t ctx,
                          const void *buf, int len);

munge_err_t munge_decode (const char *cred, munge_ctx_t ctx,
                          void **buf, int *len, uid_t *uid, gid_t *gid);

const char * munge_strerror (munge_err_t e);

cc `pkg-config --cflags --libs munge` -o foo foo.c

الوصف

تقوم الدالة munge_encode() بإنشاء بيانات اعتماد (credential) تحتوي على سلسلة نصية مشفرة بـ base64 وتنتهي بحرف فارغ. يمكن أيضاً تغليف حمولة (payload) محددة بواسطة مخزن مؤقت buf بطول len. إذا كان سياق MUNGE ctx هو NULL، فسيتم استخدام السياق الافتراضي. يتم إرجاع مؤشر إلى بيانات الاعتماد الناتجة عبر cred؛ عند حدوث خطأ، يتم تعيينه إلى NULL. يتحمل المستدعي مسؤولية تحرير الذاكرة التي يشير إليها cred.

تقوم الدالة munge_decode() بالتحقق من صحة بيانات الاعتماد cred المنتهية بحرف فارغ. إذا لم يكن سياق MUNGE ctx هو NULL، فسيتم تعيينه إلى السياق المستخدم لترميز بيانات الاعتماد. إذا لم يكن buf و len هما NULL، فسيتم تخصيص ذاكرة للحمولة المغلفة (تصل إلى 1 ميغابايت)، وسيتم تعيين buf للإشارة إلى هذه البيانات، وسيتم تعيين len إلى طولها. سيتم إضافة بايت فارغ إضافي إلى بيانات الحمولة هذه ولكن لن يتم تضمينه في طولها. إذا لم تكن هناك حمولة، فسيتم تعيين buf إلى NULL وسيتم تعيين len إلى 0. بالنسبة لأخطاء معينة (على سبيل المثال، EMUNGE_CRED_EXPIRED، EMUNGE_CRED_REWOUND، EMUNGE_CRED_REPLAYED)، قد يتم تخصيص ذاكرة الحمولة إذا لزم الأمر. يتحمل المستدعي مسؤولية تحرير الذاكرة التي يشير إليها buf. إذا لم يكن uid أو gid هو NULL، فسيتم تعيينهما إلى معرف المستخدم (UID) / معرف المجموعة (GID) للعملية التي أنشأت بيانات الاعتماد.

تقوم الدالة munge_strerror() بإرجاع سلسلة نصية وصفية تصف رقم خطأ MUNGE e.

قيمة الإرجاع

ترجع الدالتان munge_encode() و munge_decode() EMUNGE_SUCCESS عند النجاح، أو خطأ MUNGE بخلاف ذلك. إذا تم استخدام سياق MUNGE، فقد يحتوي على رسالة خطأ أكثر تفصيلاً يمكن الوصول إليها عبر munge_ctx_strerror().

ترجع الدالة munge_strerror() مؤشراً إلى سلسلة نصية ثابتة تنتهي بحرف فارغ؛ لا ينبغي تحرير هذه السلسلة أو تعديلها بواسطة المستدعي.

الأخطاء

EMUNGE_SUCCESS
نجاح.
EMUNGE_SNAFU
خطأ داخلي.
EMUNGE_BAD_ARG
وسيط غير صالح.
EMUNGE_BAD_LENGTH
تجاوز الحد الأقصى لطول الرسالة كما هو محدد في إعدادات munged.
EMUNGE_OVERFLOW
تجاوز الحد الأقصى لطول المخزن المؤقت.
EMUNGE_NO_MEMORY
غير قادر على تخصيص الذاكرة المطلوبة.
EMUNGE_SOCKET
غير قادر على التواصل مع الخادم (daemon) على مقبس النطاق.
EMUNGE_BAD_CRED
بيانات الاعتماد لا تتطابق مع التنسيق المحدد.
EMUNGE_BAD_VERSION
تحتوي بيانات الاعتماد على رقم إصدار غير مدعوم.
EMUNGE_BAD_CIPHER
تحتوي بيانات الاعتماد على نوع تشفير غير مدعوم.
EMUNGE_BAD_MAC
تحتوي بيانات الاعتماد على نوع MAC غير مدعوم.
EMUNGE_BAD_ZIP
تحتوي بيانات الاعتماد على نوع ضغط غير مدعوم.
EMUNGE_BAD_REALM
تحتوي بيانات الاعتماد على نطاق أمان غير معروف.
EMUNGE_CRED_INVALID
بيانات الاعتماد غير صالحة. هذا يعني أنه لم يكن من الممكن فك تشفير بيانات الاعتماد بنجاح. والأكثر احتمالاً هو أن المفاتيح السرية على المضيفين المشفرين والمفككين لا تتطابق. احتمال آخر هو أن بيانات الاعتماد قد تم تعديلها منذ تشفيرها.
EMUNGE_CRED_EXPIRED
تم فك تشفير بيانات الاعتماد بنجاح، ولكن وقت فك التشفير الخاص بها لاحق لوقت تشفيرها بأكثر من وقت حياتها (TTL). قد تكون الساعات على المضيفين المشفرين والمفككين غير متزامنة.
EMUNGE_CRED_REWOUND
تم فك تشفير بيانات الاعتماد بنجاح، ولكن وقت فك التشفير الخاص بها سابق لوقت تشفيرها بأكثر من وقت حياتها (TTL). الساعات على المضيفين المشفرين والمفككين غير متزامنة.
EMUNGE_CRED_REPLAYED
تم فك تشفير بيانات الاعتماد بنجاح، ولكن تم فك تشفيرها مسبقًا على هذا المضيف ضمن فترة انحراف وقت حياتها (TTL skew).
EMUNGE_CRED_UNAUTHORIZED
العميل غير مصرح له بفك تشفير بيانات الاعتماد بناءً على معرف المستخدم الفعال و/أو معرف المجموعة الفعال للعملية.

مثال

يوضح برنامج المثال التالي استخدام بيانات اعتماد MUNGE للتحقق من معرف المستخدم الفعال ومعرف المجموعة للعملية المشفرة.

#include <stdio.h>
#include <stdlib.h>
#include <unistd.h>
#include <munge.h>

int
main (int argc, char *argv[])
{
    char *cred;
    munge_err_t err;
    uid_t uid;
    gid_t gid;
    err = munge_encode (&cred, NULL, NULL, 0);
    if (err != EMUNGE_SUCCESS) {
        fprintf (stderr, "Error: Failed to encode credential: %s\n",
                munge_strerror (err));
        exit (EXIT_FAILURE);
    }
    err = munge_decode (cred, NULL, NULL, NULL, &uid, &gid);
    if (err != EMUNGE_SUCCESS) {
        fprintf (stderr, "Error: Failed to decode credential: %s\n",
                munge_strerror (err));
        exit (EXIT_FAILURE);
    }
    printf ("uid=%d gid=%d\n", uid, gid);
    free (cred);
    exit (EXIT_SUCCESS);
}

ملاحظات

قد تقوم كل من munge_encode() و munge_decode() بتخصيص ذاكرة يتحمل المستدعي مسؤولية تحريرها. الفشل في القيام بذلك سيؤدي إلى تسرب الذاكرة.
