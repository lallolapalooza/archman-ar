OSSL_LIB_CTX(3ssl)

الحزمة: openssl

الاسم

OSSL_LIB_CTX, OSSL_LIB_CTX_get_data, OSSL_LIB_CTX_new, OSSL_LIB_CTX_new_from_dispatch, OSSL_LIB_CTX_new_child, OSSL_LIB_CTX_free, OSSL_LIB_CTX_load_config, OSSL_LIB_CTX_get0_global_default, OSSL_LIB_CTX_set0_default - سياق مكتبة OpenSSL

الملخص

#include <openssl/crypto.h>
typedef struct ossl_lib_ctx_st OSSL_LIB_CTX;
OSSL_LIB_CTX *OSSL_LIB_CTX_new(void);
OSSL_LIB_CTX *OSSL_LIB_CTX_new_from_dispatch(const OSSL_CORE_HANDLE *handle,
                                             const OSSL_DISPATCH *in);
OSSL_LIB_CTX *OSSL_LIB_CTX_new_child(const OSSL_CORE_HANDLE *handle,
                                     const OSSL_DISPATCH *in);
int OSSL_LIB_CTX_load_config(OSSL_LIB_CTX *ctx, const char *config_file);
void OSSL_LIB_CTX_free(OSSL_LIB_CTX *ctx);
OSSL_LIB_CTX *OSSL_LIB_CTX_get0_global_default(void);
OSSL_LIB_CTX *OSSL_LIB_CTX_set0_default(OSSL_LIB_CTX *ctx);
void *OSSL_LIB_CTX_get_data(OSSL_LIB_CTX *ctx, int index);

الوصف

OSSL_LIB_CTX هو نوع سياق مكتبة داخلي لـ OpenSSL. يمكن للتطبيقات إنشاء سياقات خاصة بها، ولكن يمكنها أيضًا استخدام NULL لاستخدام السياق الافتراضي للدوال التي تقبل OSSL_LIB_CTX كوسيط.

عند استخدام سياق مكتبة غير افتراضي، يجب توخي الحذر في التطبيقات متعددة الخيوط لتنظيف موارد الخيوط المحلية قبل تحرير OSSL_LIB_CTX. راجع OPENSSL_thread_stop_ex(3) لمزيد من المعلومات.

OSSL_LIB_CTX_new() ينشئ سياق مكتبة OpenSSL جديدًا.

OSSL_LIB_CTX_new_from_dispatch() ينشئ سياق مكتبة OpenSSL جديدًا مهيئًا مع ردود استدعاء من بنية OSSL_DISPATCH. هذا مفيد بشكل أساسي لمؤلفي الموفرين. يجب أن تكون الوسيطات handle و dispatch هي نفسها التي يتم تمريرها إلى دالة OSSL_provider_init الخاصة بالموفر. تتطلب بعض دوال OpenSSL، مثل BIO_new_from_core_bio(3)، إنشاء سياق المكتبة بهذه الطريقة لتعمل.

OSSL_LIB_CTX_new_child() مفيد فقط لمؤلفي الموفرين ويقوم بنفس الشيء الذي تقوم به OSSL_LIB_CTX_new_from_dispatch()، ولكنه يربط سياق المكتبة الجديد بسياق مكتبة التطبيق. سياق المكتبة الجديد هو سياق مكتبة كامل بحد ذاته، ولكنه سيحتوي على جميع الموفرين المتاحين بنفس الطريقة كما هو متاح في سياق مكتبة التطبيق (دون الحاجة إلى إعادة تحميلهم). إذا قام التطبيق بتحميل أو إلغاء تحميل موفرين من سياق مكتبة التطبيق، فسيتم عكس ذلك تلقائيًا في سياق المكتبة الفرعي.

بالإضافة إلى الموفرين، ينظر سياق المكتبة الفرعي أيضًا إلى الخصائص الافتراضية (المحددة عبر EVP_set_default_properties(3)) لمكتبة التطبيق الأصلية. إذا تم استدعاء EVP_set_default_properties(3) مباشرة على سياق المكتبة الفرعي، فإن الخصائص الجديدة ستحل محل خصائص مكتبة التطبيق الأصلية وسيتوقف النظر إلى الخصائص.

عند استدعاء OSSL_LIB_CTX_new_child() ضمن نطاق دالة OSSL_provider_init الخاصة بالموفر، فإن الموفر الحالي لن يكون متاحًا بعد في سياق مكتبة التطبيق وبالتالي لن يكون متاحًا في سياق المكتبة الفرعي الجديد. بمجرد عودة OSSL_provider_init، سيصبح الموفر الجديد متاحًا في سياق مكتبة التطبيق وسيتم عكس ذلك في سياق المكتبة الفرعي الجديد.

OSSL_LIB_CTX_load_config() يقوم بتحميل ملف تكوين باستخدام السياق ctx. يمكن استخدام هذا لربط سياق مكتبة مع موفرين تم تحميلهم من تكوين. لا ينبغي استدعاء هذه الدالة بشكل متزامن من خيوط متعددة على نفس ctx.

OSSL_LIB_CTX_free() يحرر السياق ctx، ما لم يكن السياق الافتراضي لـ OpenSSL. إذا كانت الوسيطة NULL، فلا يتم فعل شيء.

OSSL_LIB_CTX_get0_global_default() يعيد مرجعًا ملموسًا (غير NULL) إلى سياق المكتبة العام الافتراضي.

OSSL_LIB_CTX_set0_default() يضبط سياق مكتبة OpenSSL الافتراضي في الخيط الحالي إلى ctx. يتم إرجاع السياق الافتراضي السابق. يجب على المتصل توخي الحذر لاستعادة السياق الافتراضي السابق باستدعاء لاحق لهذه الدالة. إذا كان ctx هو NULL، فلن يتم إجراء أي تغيير على السياق الافتراضي، ولكن يتم إرجاع مؤشر إلى السياق الحالي. في حالة نجاح استدعاء هذه الدالة، فإن القيمة المرجعة ستكون دائمًا سياق مكتبة ملموسًا (غير NULL).

نظرًا لتغيير السياق الافتراضي، يجب توخي الحذر بشأن بدء المهام غير المتزامنة (راجع ASYNC_start_job(3))، لأنه عند بدء مهمة غير متزامنة، سيتم استخدام سياق المكتبة الافتراضي طوال مدة المهمة، بغض النظر عن كيفية إجراء تغييرات لاحقة في خيط التنفيذ الخاص بالمتصل. هذا يعني أن خيط التنفيذ الخاص بالمتصل يجب أن يتجنب تحرير سياق المكتبة الذي كان افتراضيًا عند بدء المهمة غير المتزامنة، حتى تنتهي المهمة.

OSSL_LIB_CTX_get_data() يعيد مؤشرًا إلى موقع في الذاكرة يعتمد تفسيره على الفهرس. تشير وسيطة الفهرس إلى عضو في السياق المراد استرداده. قيم الفهرس خاصة بـ OpenSSL حاليًا، وبالتالي يجب على التطبيقات عدم استدعاء هذه الدالة. إذا كان السياق NULL، تعمل الدالة على السياق الافتراضي. OSSL_LIB_CTX_get_data() يعيد مؤشرًا إلى موقع في الذاكرة يعتمد تفسيره على الفهرس.

قيم الإرجاع

OSSL_LIB_CTX_new() و OSSL_LIB_CTX_get0_global_default() و OSSL_LIB_CTX_set0_default() تعيد مؤشرًا إلى سياق المكتبة في حالة النجاح، أو NULL في حالة الخطأ.

OSSL_LIB_CTX_free() لا تعيد أي قيمة.

OSSL_LIB_CTX_load_config() تعيد 1 في حالة النجاح، 0 في حالة الخطأ.

OSSL_LIB_CTX_get_data() تعيد مؤشرًا إلى موقع في الذاكرة يعتمد تفسيره على الفهرس.
