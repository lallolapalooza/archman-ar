التراخيص: GPL

FBB::OSymCryptStream(3bobcat) التشفير وفك التشفير المتماثل FBB::OSymCryptStream(3bobcat)

الاسم

FBB::OSymCryptStream - std::ostream يقوم بالتشفير/فك التشفير المتماثل

ملخص

#include <bobcat/osymcryptstream>
خيار الربط: -lbobcat -lcrypto

الوصف

يمكن استخدام كائنات FBB::OSymCryptStream لتشفير أو فك تشفير المعلومات المتوفرة على تدفقات std::istream منفصلة.

الفئة OSymCryptStream هي قالب فئة، تستخدم معلمة غير نوعية FBB::CryptType. تقوم كائنات الفئة FBB::OSymCryptStream<FBB::ENCRYPT> بتشفير المعلومات التي تتلقاها، وتقوم كائنات الفئة FBB::OSymCryptStream<FBB::DECRYPT> بفك تشفير المعلومات التي تتلقاها.

يمكن استخدام جميع طرق التشفير المتماثل التي تحددها مكتبة OpenSSL والتي يمكن اختيارها بالاسم لتشفير/فك تشفير المعلومات. لاختيار طريقة تشفير معينة، يتم تمرير معرف إلى المنشئ. على سبيل المثال، "aes-256-gcm". للحصول على خوارزميات التشفير المدعومة حاليًا، قم بتنفيذ الأمر

    openssl list -cipher-algorithms

تقوم كائنات OSymCryptStream بقراءة المعلومات المراد تشفيرها أو فك تشفيرها من مصدر خارجي (على سبيل المثال، من كائنات std::istream). تتم كتابة الأحرف التي يتم تشفيرها أو فك تشفيرها بواسطة كائنات OSymCryptStream إلى كائنات std::ostream التي يتم تحديدها في وقت الإنشاء كمرجع ostream أو بواسطة اسم الملف.

المنشئات

o
OSymCryptStream<CryptType>(std::ostream &outStream, std::string const &cipherName, std::string const &key, std::string const &iv, size_t inBufSize = 100):
يقوم هذا المنشئ بتعريف std::streambuf يقوم بتشفير أو فك تشفير الأحرف التي يتلقاها.
- تقوم كائنات OSymCryptStream<FBB::ENCRYPT> بإجراء التشفير؛
-تقوم كائنات OSymCryptStream<FBB::DECRYPT> بإجراء فك التشفير؛
- تقوم كائنات OSymCryptStream<CryptType> بكتابة الأحرف المشفرة أو المفكوكة التشفير إلى outStream؛
- يتم تحديد طريقة التشفير المراد استخدامها بواسطة المعلمة cipherName. على سبيل المثال، "AES-256-GCM"؛
- يتم تحديد المفتاح المتماثل المراد استخدامه بواسطة المعلمة key؛
- يتم تحديد متجه التهيئة بواسطة المعلمة iv؛
- المخزن المؤقت المستخدم داخليًا لـ FBB::OSymCryptStreambuf سيحتوي على inBufSize حرفًا. القيمة الافتراضية هي أصغر قيمة مستخدمة. عند تحديد قيمة bufSize أصغر، يتم استخدام القيمة الافتراضية؛
o
OSymCryptStream<CryptType>(std::string const &outStreamName, std::string const &cipherName, std::string const &key, std::string const &iv, size_t inBufSize = 100):
نفس المنشئ السابق، ولكن المعلمة الأولى لهذا المنشئ تحدد اسم الملف لتلقي الأحرف المشفرة أو المفكوكة التشفير.

إذا فشل الإنشاء، يتم طرح استثناء، يذكر دالة openssl التي فشلت في الاكتمال (انظر أيضًا errorMsg أدناه).

منشئ النقل متاح، ومنشئ النسخ وعوامل التعيين غير متاحين.

الأعضاء الموروثون

نظرًا لأن الفئة مشتقة علنًا من std::ostream، يمكن استخدام جميع أعضاء std::ostream.

دوال الأعضاء

o
static std::string errorMsg():
إذا فشلت دالة openssl، يتم طرح استثناء يذكر اسم الدالة الفاشلة. في تلك الحالات، يمكن استدعاء الدالة errorMsg لإرجاع std::string تحتوي على رمز خطأ openssl (الذي تم إرجاعه بواسطة ERR_get_error) وتمثيله النصي (الذي تم إرجاعه بواسطة ERR_error_string). إذا كان رمز الخطأ المبلغ عنه صفرًا، فهذا يعني في الواقع أنه لم يحدث خطأ، وتم الإبلاغ عن الاستثناء بشكل خاطئ؛
o
static size_t keyLength(std::string const &cipherName):
يعيد طول المفتاح الأدنى المطلوب للخوارزمية cipherName؛
o
static size_t ivLength(std::sting const &cipherName):
يعيد طول متجه التهيئة الأدنى المطلوب للخوارزمية cipherName.

تطرح الدالتان الأخيرتان استثناءات إذا كانت cipherName لا تحتوي على اسم خوارزمية تشفير مدعومة.

المعدل

o
std::ostream &FBB::eoi(std::ostream &out):
عن طريق إدراج المعدل eoi في كائن OSymCryptStream، يعتبر إدخال المعلومات مكتملًا. يكمل إدخال المعلومات في std::ostream المحدد في وقت الإنشاء كالمعلمة الأولى لكائن OSymCryptStream. بهذه الطريقة، لا يتعين عليك الانتظار حتى يقوم المدمر لكائن OSymCryptStream بتفريغ هذا التدفق. يوضح المثال التالي استخدام المعدل eoi (الأسطر المعلقة بالقرب من السطر 40): 
    //in.seekg(0);              // عندما يتم تنشيط هذا، لن يتم
    //enc << in.rdbuf();        // معالجته بسبب ’<< eoi’
تنشيط تلك الأسطر لن يؤدي إلى معالجة تدفق in مرتين.

مثال

#include <iostream>
#include <fstream>
#include <string>
#include <bobcat/osymcryptstream>
using namespace std;
using namespace FBB;
int main(int argc, char **argv)
try
{
    if (argc == 1)
    {
        cout << "arg[1]: e - encrypt, d - decrypt,\n"
                "arg[2]: file to process, arg[3]: processed file\n";
        return 0;
    }
    string key = "0123456789abcdef0123456789abcdef";
    cout << "encryption key ? ";
    cin >> key;
    while (key.size() < 32)
        key += key;
    string iv = " 0123456789ab" "456";
    char cipherName[] =
        "AES-256-GCM"
        //"AES-256-CBC"
    ;
    ifstream in{ argv[2] };
    ofstream out{ argv[3] };
    if (*argv[1] == ’e’)
    {
        OSymCryptStream<ENCRYPT> enc{ out, cipherName, key, iv, 100 };
            // comment out the above statement and uncomment the next
            // to use the constructor expecting a string as 1st arg:
//      OSymCryptStream<ENCRYPT> enc{ argv[3], cipherName, key, 
//                                    iv, 100 };
        enc << in.rdbuf() << eoi;
        //in.seekg(0);              // when activated, this won’t
        //enc << in.rdbuf();        // be processed due to ’<< eoi’
    }
    else
    {
        OSymCryptStream<DECRYPT> decrypt{ out, cipherName, key, 
                                                           iv, 100 };
            // comment out the above statement and uncomment the next
            // to use the constructor expecting a string as 1st arg:
//      OSymCryptStream<DECRYPT> decrypt{ argv[3], cipherName, key, 
//                                        iv, 100 };
        decrypt << in.rdbuf() << eoi;
    }
}
catch (exception const &exc)
{
    cerr << exc.what() << ’\n’;
}
