OSSL_PARAM_BLD(3ssl)

الحزمة: openssl

التراخيص: Apache-2.0

OSSL_PARAM_BLD(3ssl) OpenSSL OSSL_PARAM_BLD(3ssl)

الاسم

OSSL_PARAM_BLD, OSSL_PARAM_BLD_new, OSSL_PARAM_BLD_to_param, OSSL_PARAM_BLD_free, OSSL_PARAM_BLD_push_int, OSSL_PARAM_BLD_push_uint, OSSL_PARAM_BLD_push_long, OSSL_PARAM_BLD_push_ulong, OSSL_PARAM_BLD_push_int32, OSSL_PARAM_BLD_push_uint32, OSSL_PARAM_BLD_push_int64, OSSL_PARAM_BLD_push_uint64, OSSL_PARAM_BLD_push_size_t, OSSL_PARAM_BLD_push_time_t, OSSL_PARAM_BLD_push_double, OSSL_PARAM_BLD_push_BN, OSSL_PARAM_BLD_push_BN_pad, OSSL_PARAM_BLD_push_utf8_string, OSSL_PARAM_BLD_push_utf8_ptr, OSSL_PARAM_BLD_push_octet_string, OSSL_PARAM_BLD_push_octet_ptr - وظائف للمساعدة في إنشاء مصفوفات OSSL_PARAM

الخلاصة

#include <openssl/param_build.h>

void OSSL_PARAM_BLD_free(OSSL_PARAM_BLD *bld);
int OSSL_PARAM_BLD_push_TYPE(OSSL_PARAM_BLD *bld, const char *key, TYPE val);
int OSSL_PARAM_BLD_push_BN(OSSL_PARAM_BLD *bld, const char *key,
                           const BIGNUM *bn);
int OSSL_PARAM_BLD_push_BN_pad(OSSL_PARAM_BLD *bld, const char *key,
                               const BIGNUM *bn, size_t sz);
int OSSL_PARAM_BLD_push_utf8_string(OSSL_PARAM_BLD *bld, const char *key,
                                    const char *buf, size_t bsize);

                                 char *buf, size_t bsize);

                                  void *buf, size_t bsize);

الوصف

مجموعة من الوظائف المساعدة التي تبسط إنشاء مصفوفات OSSL_PARAM. أسماء TYPE تأتي وفقًا لـ OSSL_PARAM_int(3).

OSSL_PARAM_BLD_new() تقوم بتخصيص وتهيئة بنية OSSL_PARAM_BLD جديدة حتى يمكن إضافة القيم. يتم مسح أي قيم موجودة.

OSSL_PARAM_BLD_free() تقوم بإلغاء تخصيص الذاكرة التي تم تخصيصها بواسطة OSSL_PARAM_BLD_new(). إذا كان الوسيط NULL، فلا يتم فعل شيء.

OSSL_PARAM_BLD_to_param() تحول بنية OSSL_PARAM_BLD المبنية bld إلى مصفوفة OSSL_PARAM مخصصة. يجب تحرير مصفوفة OSSL_PARAM وجميع التخزين المرتبط بها عن طريق استدعاء OSSL_PARAM_free() مع القيمة الراجعة للوظيفة. يمكن استدعاء OSSL_PARAM_BLD_free() بأمان في أي وقت بعد هذه الوظيفة.

OSSL_PARAM_BLD_push_TYPE() هي سلسلة من الوظائف التي ستنشئ كائنات OSSL_PARAM بالحجم المحدد والنوع الصحيح للوسيط val. يتم تخزين val بالقيمة ويمكن استخدام تعبير أو متغير تلقائي.

عندما يشير TYPE إلى نوع عددي صحيح، فإن أنواع الأعداد الصحيحة ذات الإشارة عادةً ما تحصل على معلمات نوع OSSL_PARAM OSSL_PARAM_INTEGER. عندما يشير TYPE إلى نوع عددي صحيح غير موقع، فإنه يحصل على نوع OSSL_PARAM OSSL_PARAM_UNSIGNED_INTEGER.

OSSL_PARAM_BLD_push_BN() هي وظيفة ستنشئ كائن OSSL_PARAM يحتوي على BIGNUM المحدد bn. عندما يكون bn صفرًا أو موجبًا، يصبح نوع OSSL_PARAM الخاص به OSSL_PARAM_UNSIGNED_INTEGER. عندما يكون bn سالبًا، يصبح نوع OSSL_PARAM الخاص به OSSL_PARAM_INTEGER. إذا تم تمييز bn على أنه مخصص بشكل آمن، فسيتم تخصيص تمثيله في OSSL_PARAM بشكل آمن أيضًا. يتم تخزين الوسيط bn بالمرجع ويجب أن يظل كائن BIGNUM الأساسي موجودًا حتى بعد استدعاء OSSL_PARAM_BLD_to_param().

OSSL_PARAM_BLD_push_BN_pad() هي وظيفة ستنشئ كائن OSSL_PARAM يحتوي على BIGNUM المحدد bn. سيتم حشو الكائن ليحتل بالضبط sz بايت، إذا تم تحديد مساحة غير كافية ينتج عنه خطأ. عندما يكون bn صفرًا أو موجبًا، يصبح نوع OSSL_PARAM الخاص به OSSL_PARAM_UNSIGNED_INTEGER. عندما يكون bn سالبًا، يصبح نوع OSSL_PARAM الخاص به OSSL_PARAM_INTEGER. إذا تم تمييز bn على أنه مخصص بشكل آمن، فسيكون تمثيله في OSSL_PARAM مخصصًا بشكل آمن أيضًا. يتم تخزين الوسيط bn بالمرجع ويجب أن يظل كائن BIGNUM الأساسي موجودًا حتى بعد استدعاء OSSL_PARAM_BLD_to_param().

OSSL_PARAM_BLD_push_utf8_string() هي وظيفة ستنشئ كائن OSSL_PARAM يشير إلى سلسلة UTF8 المحددة بواسطة buf. لا ينبغي أن يشمل طول السلسلة bsize البايت الفارغ الختامي. إذا كان صفرًا، فسيتم حسابه. يتم تخزين السلسلة التي تشير إليها buf بالمرجع ويجب أن تظل ضمن النطاق حتى بعد استدعاء OSSL_PARAM_BLD_to_param().

OSSL_PARAM_BLD_push_octet_string() هي وظيفة ستنشئ كائن OSSL_PARAM يشير إلى سلسلة البايت المحددة بواسطة buf و <bsize>. يتم تخزين الذاكرة التي تشير إليها buf بالمرجع ويجب أن تظل ضمن النطاق حتى بعد استدعاء OSSL_PARAM_BLD_to_param().

OSSL_PARAM_BLD_push_utf8_ptr() هي وظيفة ستنشئ كائن OSSL_PARAM يشير إلى سلسلة UTF8 المحددة بواسطة buf. لا ينبغي أن يشمل طول السلسلة bsize البايت الفارغ الختامي. إذا كان صفرًا، فسيتم حسابه. يتم تخزين السلسلة التي تشير إليها buf بالمرجع ويجب أن تظل ضمن النطاق حتى يتم تحرير مصفوفة OSSL_PARAM.

OSSL_PARAM_BLD_push_octet_ptr() هي وظيفة ستنشئ كائن OSSL_PARAM يشير إلى سلسلة البايت المحددة بواسطة buf. يتم تخزين الذاكرة التي تشير إليها buf بالمرجع ويجب أن تظل ضمن النطاق حتى يتم تحرير مصفوفة OSSL_PARAM.

قيم الإرجاع

OSSL_PARAM_BLD_new() تعيد بنية OSSL_PARAM_BLD المخصصة، أو NULL في حالة الخطأ.

OSSL_PARAM_BLD_to_param() تعيد مصفوفة OSSL_PARAM المخصصة، أو NULL في حالة الخطأ.

جميع وظائف OSSL_PARAM_BLD_push_TYPE تعيد 1 عند النجاح و 0 عند الخطأ.

ملاحظات

OSSL_PARAM_BLD_push_BN() و OSSL_PARAM_BLD_push_BN_pad() تدعمان فقط BIGNUMs غير السالبة. تعيدان خطأ في BIGNUMs السالبة. لتمرير BIGNUMs ذات الإشارة، استخدم OSSL_PARAM_BLD_push_signed_BN().

أمثلة

كلا المثالين ينشئان مصفوفة OSSL_PARAM تحتوي على مفتاح RSA. لكليهما، متغيرات المفاتيح المعرفة مسبقًا هي:

BIGNUM *n;           /* modulus */
unsigned int e;      /* public exponent */
BIGNUM *d;           /* private exponent */
BIGNUM *p, *q;       /* first two prime factors */
BIGNUM *dmp1, *dmq1; /* first two CRT exponents */
BIGNUM *iqmp;        /* first CRT coefficient */

المثال 1

يوضح هذا المثال كيفية إنشاء مصفوفة OSSL_PARAM تحتوي على مفتاح RSA خاص.

OSSL_PARAM_BLD *bld = OSSL_PARAM_BLD_new();
OSSL_PARAM *params = NULL;
if (bld == NULL
    || !OSSL_PARAM_BLD_push_BN(bld, "n", n)
    || !OSSL_PARAM_BLD_push_uint(bld, "e", e)
    || !OSSL_PARAM_BLD_push_BN(bld, "d", d)
    || !OSSL_PARAM_BLD_push_BN(bld, "rsa-factor1", p)
    || !OSSL_PARAM_BLD_push_BN(bld, "rsa-factor2", q)
    || !OSSL_PARAM_BLD_push_BN(bld, "rsa-exponent1", dmp1)
    || !OSSL_PARAM_BLD_push_BN(bld, "rsa-exponent2", dmq1)
    || !OSSL_PARAM_BLD_push_BN(bld, "rsa-coefficient1", iqmp)
    || (params = OSSL_PARAM_BLD_to_param(bld)) == NULL)
    goto err;
OSSL_PARAM_BLD_free(bld);
/* Use params */
...
OSSL_PARAM_free(params);
