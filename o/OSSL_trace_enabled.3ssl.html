OSSL_trace_enabled(3ssl)

الحزمة: openssl

التراخيص: Apache-2.0

OSSL_TRACE_ENABLED, OSSL_trace_begin, OSSL_trace_end, OSSL_TRACE_BEGIN, OSSL_TRACE_END, OSSL_TRACE_CANCEL, OSSL_TRACE, OSSL_TRACE1, OSSL_TRACE2, OSSL_TRACE3, OSSL_TRACE4, OSSL_TRACE5, OSSL_TRACE6, OSSL_TRACE7, OSSL_TRACE8, OSSL_TRACE9, OSSL_TRACEV, OSSL_TRACE_STRING, OSSL_TRACE_STRING_MAX, OSSL_trace_string, OSSL_trace_enabled - واجهة برمجة تطبيقات تتبع OpenSSL

الوصف

الوظائف الموصوفة هنا تهم بشكل أساسي أولئك الذين يقدمون وظائف OpenSSL، سواء في OpenSSL نفسه أو في وحدات المحرك أو ما شابه ذلك.

إذا تم تمكين مرفق التتبع (انظر "تكوين التتبع" أدناه)، تُستخدم هذه الوظائف لإنشاء مخرجات تتبع نصية مجانية.

تنقسم مخرجات التتبع إلى أنواع يتم تمكينها بشكل فردي بواسطة التطبيق. تم وصف أنواع التتبع بالتفصيل في "أنواع التتبع" في OSSL_trace_set_callback(3). يجب عدم استخدام نوع الاحتياط OSSL_TRACE_CATEGORY_ALL مع الوظائف الموصوفة هنا.

يتم تمكين التتبع لفئة معينة في وقت التشغيل إذا تم ربط ما يسمى قناة تتبع بها. قناة التتبع هي ببساطة كائن BIO يمكن للتطبيق الكتابة إليه بمخرجات التتبع الخاصة به.

لدى التطبيق طريقتان مختلفتان لتسجيل قناة تتبع، إما عن طريق تقديم كائن BIO مباشرة باستخدام OSSL_trace_set_channel(3)، أو عن طريق تقديم روتين استدعاء باستخدام OSSL_trace_set_callback(3). يتم تغليف الأخير داخليًا بواسطة كائن BIO مخصص، لذلك بالنسبة لتعليمات التتبع، فإن كلا نوعي القنوات غير قابلين للتمييز فعليًا. نسميها قناة تتبع بسيطة و قناة تتبع استدعاء، على التوالي.

لإنتاج مخرجات التتبع، من الضروري الحصول على مؤشر إلى قناة التتبع (أي، كائن BIO) باستخدام OSSL_trace_begin()، والكتابة إليه باستخدام روتينات إخراج BIO عشوائية، وأخيرًا تحرير القناة باستخدام OSSL_trace_end(). تقوم مكالمات OSSL_trace_begin()/OSSL_trace_end() التي تحيط بمخرجات التتبع بإنشاء مجموعة، والتي تعمل كقسم حرج (محمي بواسطة قفل) لضمان عدم خلط مخرجات التتبع من مؤشرات الترابط المختلفة.

عادةً لا تقوم تعليمات التتبع باستدعاء OSSL_trace_{begin,end}() مباشرة، بل تستخدم مجموعة من وحدات الماكرو المريحة، انظر قسم "الوحدات" أدناه.

الوظائف

يمكن استخدام OSSL_trace_enabled() للتحقق مما إذا كان التتبع لـ category المحدد ممكّنًا، أي ما إذا كان مرفق التتبع قد تم تمكينه بشكل ثابت (انظر "تكوين التتبع" أدناه) وتم تسجيل قناة تتبع باستخدام OSSL_trace_set_channel(3) أو OSSL_trace_set_callback(3).

OSSL_trace_begin() تُستخدم لبدء قسم تتبع، والحصول على القناة لـ category المحدد في شكل BIO. يمكن استخدام هذا BIO للإخراج فقط. المؤشر المرتجع هو NULL إذا كانت الفئة غير صالحة أو غير ممكّنة.

OSSL_trace_end() تُستخدم لإنهاء قسم تتبع.

يُعد استخدام OSSL_trace_begin() و OSSL_trace_end() لتغليف أقسام التتبع إلزاميًا. نتيجة محاولة إنتاج مخرجات تتبع خارج هذه الأقسام غير محددة.

OSSL_trace_string() تُخرج data بطول size كسلسلة نصية على BIO out. إذا كان text هو 0، تقوم الدالة بإخفاء أي أحرف تحكم مضمنة بخلاف الأسطر الجديدة وتضمن للمدخلات غير الفارغة أن المخرجات تنتهي بسطر جديد. ما لم يكن full غير صفري، يقتصر الطول (مع تحذير مناسب) على OSSL_TRACE_STRING_MAX حرفًا، والذي هو حاليًا 80.

الوحدات

هناك عدد من وحدات الماكرو المريحة المعرفة، لجعل التتبع سهلاً ومتسقًا.

OSSL_TRACE_BEGIN() و OSSL_TRACE_END() تحجز BIO "trc_out" وتُستخدم على النحو التالي لتغليف قسم تتبع:

OSSL_TRACE_BEGIN(TLS) {
    BIO_printf(trc_out, ... );
} OSSL_TRACE_END(TLS);

هذا سيتوسع عادةً إلى:

do {
    BIO *trc_out = OSSL_trace_begin(OSSL_TRACE_CATEGORY_TLS);
    if (trc_out != NULL) {
        ...
        BIO_printf(trc_out, ...);
    }
    OSSL_trace_end(OSSL_TRACE_CATEGORY_TLS, trc_out);
} while (0);

يجب استخدام OSSL_TRACE_CANCEL() قبل العودة من قسم تتبع أو القفز للخروج منه:

OSSL_TRACE_BEGIN(TLS) {
    if (some_error) {
        OSSL_TRACE_CANCEL(TLS);
        goto err;
    }
    BIO_printf(trc_out, ... );
} OSSL_TRACE_END(TLS);

هذا سيتوسع عادةً إلى:

do {
    BIO *trc_out = OSSL_trace_begin(OSSL_TRACE_CATEGORY_TLS);
    if (trc_out != NULL) {
        if (some_error) {
            OSSL_trace_end(OSSL_TRACE_CATEGORY_TLS, trc_out);
            goto err;
        }
        BIO_printf(trc_out, ... );
    }
    OSSL_trace_end(OSSL_TRACE_CATEGORY_TLS, trc_out);
} while (0);

OSSL_TRACE() و OSSL_TRACE1()، OSSL_TRACE2()، ... OSSL_TRACE9() هي ما تسمى بوحدات الماكرو لمرة واحدة:

استدعاء الماكرو "OSSL_TRACE(category, text)"، ينتج عنه مخرجات تتبع نصية حرفية.

استدعاء الماكرو "OSSL_TRACEn(category, format, arg1, ..., argn)" ينتج عنه مخرجات تتبع بأسلوب printf مع n وسيط صيغة (n=1,...,9). يتوسع إلى:

OSSL_TRACE_BEGIN(category) {
    BIO_printf(trc_out, format, arg1, ..., argN);
} OSSL_TRACE_END(category)

داخليًا، يتم تنفيذ جميع وحدات الماكرو لمرة واحدة باستخدام وحدة ماكرو عامة OSSL_TRACEV()، نظرًا لأن C90 لا يدعم وحدات الماكرو المتغيرة (variadic macros). هذه الوحدة المساعدة لها ملخص غريب نوعًا ما ويجب عدم استخدامها مباشرة.

استدعاء الماكرو "OSSL_TRACE_STRING(category, text, full, data, len)" يخرج data بطول size كسلسلة نصية إذا كان التتبع للفئة category المحدد ممكّنًا. يتوسع إلى:

OSSL_TRACE_BEGIN(category) {
    OSSL_trace_string(trc_out, text, full, data, len);
} OSSL_TRACE_END(category)

يمكن استخدام الماكرو OSSL_TRACE_ENABLED() لتنفيذ بعض التعليمات البرمجية بشكل شرطي فقط إذا كانت فئة تتبع معينة ممكّنة. في بعض الحالات، يكون هذا أبسط من الدخول إلى قسم تتبع باستخدام OSSL_TRACE_BEGIN() و OSSL_TRACE_END(). على سبيل المثال، التعليمات البرمجية

if (OSSL_TRACE_ENABLED(TLS)) {
    ...
}

تتوسع إلى

if (OSSL_trace_enabled(OSSL_TRACE_CATEGORY_TLS) {
    ...
}

ملاحظات

ليس من الضروري حماية استدعاءات وظائف مخرجات التتبع مثل OSSL_TRACE(category, ...) بواسطة OSSL_TRACE_ENABLED(category).

إذا كان إنتاج مخرجات التتبع يتطلب إجراء حسابات مساعدة، فيجب وضع هذه التعليمات البرمجية المساعدة داخل كتلة شرطية يتم تنفيذها فقط إذا كانت فئة التتبع ممكّنة.

الطريقة الأكثر طبيعية للقيام بذلك هي وضع التعليمات البرمجية داخل قسم التتبع نفسه لأنه يقدم بالفعل كتلة شرطية كهذه.

OSSL_TRACE_BEGIN(TLS) {
    int var = do_some_auxiliary_calculation();
    BIO_printf(trc_out, "var = %d\n", var);
} OSSL_TRACE_END(TLS);

في بعض الحالات، يكون من الأفضل استخدام مجموعة شرطية بسيطة بدلاً من قسم تتبع. هذه هي الحالة إذا كانت الحسابات والتتبع تحدثان في مواقع مختلفة من التعليمات البرمجية، أو إذا كانت الحسابات تستغرق وقتًا طويلاً جدًا بحيث أن وضعها داخل قسم تتبع (حرج) سيخلق الكثير من التنافس.
