OSSL_ENCODER_to_bio(3ssl)

الحزمة: openssl

التراخيص: Apache-2.0

OSSL_ENCODER_TO_BIO(3ssl) OpenSSL OSSL_ENCODER_TO_BIO(3ssl)

الاسم

OSSL_ENCODER_to_data, OSSL_ENCODER_to_bio, OSSL_ENCODER_to_fp - روتينات لتنفيذ الترميز

الملخص

#include <openssl/encoder.h>
int OSSL_ENCODER_to_data(OSSL_ENCODER_CTX *ctx, unsigned char **pdata,
                         size_t *pdata_len);
int OSSL_ENCODER_to_bio(OSSL_ENCODER_CTX *ctx, BIO *out);
int OSSL_ENCODER_to_fp(OSSL_ENCODER_CTX *ctx, FILE *fp);

وحدات ماكرو توفر الميزات:

OSSL_ENCODER_to_fp() متاحة فقط عندما يكون OPENSSL_NO_STDIO غير معرف.

الوصف

OSSL_ENCODER_to_data() يقوم بتشغيل عملية الترميز للسياق ctx، مع توجيه الإخراج إلى *pdata و *pdata_len. إذا كان *pdata فارغًا (NULL) عند استدعاء OSSL_ENCODER_to_data()، فسيتم تخصيص مخزن مؤقت باستخدام OPENSSL_zalloc(3)، وسيتم تعيين *pdata للإشارة إلى بداية هذا المخزن المؤقت، وسيتم تعيين *pdata_len لطوله عند عودة OSSL_ENCODER_to_data(). إذا كان *pdata غير فارغ (non-NULL) عند استدعاء OSSL_ENCODER_to_data()، فمن المفترض أن *pdata_len يحتوي على حجمه. في هذه الحالة، سيتم تعيين *pdata للإشارة بعد البايتات المرمزة، وسيتم تعيين *pdata_len لعدد البايتات المتبقية.

OSSL_ENCODER_to_bio() يقوم بتشغيل عملية الترميز للسياق ctx، مع توجيه الإخراج إلى BIO out.

OSSL_ENCODER_to_fp() يفعل نفس الشيء مثل OSSL_ENCODER_to_bio()، باستثناء أن الإخراج يذهب إلى FILE fp.

بالنسبة لـ OSSL_ENCODER_to_bio() و OSSL_ENCODER_to_fp()، يجب على التطبيق إعداد BIO أو FILE بشكل صحيح، على سبيل المثال لجعله في وضع نصي أو ثنائي حسب ما هو مناسب لنوع إخراج المرمز.

قيم الإرجاع

OSSL_ENCODER_to_bio() و OSSL_ENCODER_to_fp() و OSSL_ENCODER_to_data() ترجع 1 عند النجاح، أو 0 عند الفشل.

أمثلة

لترميز مفتاح خاص (pkey) كـ PKCS#8 بتنسيق PEM إلى bio:

OSSL_ENCODER_CTX *ectx;
const char *format = "PEM";
const char *structure = "PrivateKeyInfo"; /* PKCS#8 structure */
const unsigned char *pass = "my password";
ectx = OSSL_ENCODER_CTX_new_for_pkey(pkey,
                                     OSSL_KEYMGMT_SELECT_KEYPAIR
                                     | OSSL_KEYMGMT_SELECT_DOMAIN_PARAMETERS,
                                     format, structure,
                                     NULL);
if (ectx == NULL) {
    /* خطأ: لم يتم العثور على مرمّزات محتملة مناسبة */
}
if (pass != NULL)
    OSSL_ENCODER_CTX_set_passphrase(ectx, pass, strlen(pass));
if (OSSL_ENCODER_to_bio(ectx, bio)) {
    /* تم ترميز المفتاح الخاص بنجاح في bio */
} else {
    /* فشل الترميز */
}
OSSL_ENCODER_CTX_free(ectx);

لترميز مفتاح خاص (pkey) كـ PKCS#8 بتنسيق DER مشفر بـ AES-256-CBC إلى مخزن مؤقت:

OSSL_ENCODER_CTX *ectx;
const char *format = "DER";
const char *structure = "PrivateKeyInfo"; /* PKCS#8 structure */
const unsigned char *pass = "my password";
unsigned char *data = NULL;
size_t datalen;
ectx = OSSL_ENCODER_CTX_new_for_pkey(pkey,
                                     OSSL_KEYMGMT_SELECT_KEYPAIR
                                     | OSSL_KEYMGMT_SELECT_DOMAIN_PARAMETERS,
                                     format, structure,
                                     NULL);
if (ectx == NULL) {
    /* خطأ: لم يتم العثور على مرمّزات محتملة مناسبة */
}
if (pass != NULL) {
    OSSL_ENCODER_CTX_set_passphrase(ectx, pass, strlen(pass));
    OSSL_ENCODER_CTX_set_cipher(ctx, "AES-256-CBC", NULL);
}
if (OSSL_ENCODER_to_data(ectx, &data, &datalen)) {
    /*
     * تم ترميز المفتاح الخاص بنجاح في مخزن بيانات مخصص حديثًا
     */
} else {
    /* فشل الترميز */
}
OSSL_ENCODER_CTX_free(ectx);
