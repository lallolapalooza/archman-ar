الاسم

oath_totp_validate3 - دالة API

الملخص

#include <oath.h>

int oath_totp_validate3(const char * secret, size_t secret_length, time_t now, unsigned time_step_size, time_t start_offset, size_t window, int * otp_pos, uint64_t * otp_counter, const char * otp);

المعاملات

const char * secret
سلسلة السر المشتركة
size_t secret_length
طول secret
time_t now
قيمة وقت Unix للتحقق من صحة TOTP
unsigned time_step_size
معامل نظام خطوات الوقت (عادة 30)
time_t start_offset
وقت Unix لبدء عد خطوات الوقت (عادة 0)
size_t window
عدد الأكواد لمرة واحدة التي يجب اختبارها بعد/قبل الكود الأولي
int * otp_pos
موضع البحث الناتج في نافذة الكود لمرة واحدة (يمكن أن يكون NULL).
uint64_t * otp_counter
قيمة العداد المستخدمة لحساب قيمة الكود لمرة واحدة (يمكن أن تكون NULL).
const char * otp
الكود لمرة واحدة المراد التحقق من صحته.

الوصف

التحقق من صحة الكود لمرة واحدة وفقًا لخوارزمية OATH TOTP لكل RFC 6238.

حالياً، يتم دعم أطوال الكود لمرة واحدة المكونة من 6 أو 7 أو 8 أرقام فقط. قد يتم رفع هذه القيود في الإصدارات المستقبلية، على الرغم من أن بعض القيود متأصلة في البروتوكول.

يعيد

يعيد القيمة المطلقة للموضع في نافذة الكود لمرة واحدة (الصفر هو الموضع الأول)، أو OATH_INVALID_OTP إذا لم يتم العثور على أي كود لمرة واحدة في نافذة الكود لمرة واحدة، أو رمز خطأ.
