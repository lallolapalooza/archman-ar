الاسم

getopt - تحليل الخيارات في سطر الأوامر

المكتبة

مكتبة C القياسية (libc)

الملخص

#include <unistd.h>
int getopt(int argc, char * const argv[], const char *optstring);
extern char *optarg;
extern int optind, opterr;
#define _GNU_SOURCE
#include <getopt.h>
int getopt_long(int argc, char * const argv[], const char *optstring,
                const struct option *longopts, int *longindex);
int getopt_long_only(int argc, char * const argv[], const char *optstring,
                     const struct option *longopts, int *longindex);

الوصف

تقوم هذه الدوال بتحليل المعلمات في سطر الأوامر. يتم استدعاء getopt() بشكل متكرر لتحديد الخيارات التالية في مصفوفة argv.

يتم فحص المتغير العام optind، والذي يتم تهيئته إلى 1 (للإشارة إلى أن أول معلمة في argv هي اسم البرنامج)، في كل استدعاء. يتم تحديث optind بواسطة getopt() للإشارة إلى المعلمة التالية التي يجب معالجتها. عندما لا تكون هناك خيارات متبقية، يتم إرجاع getopt() بقيمة -1.

يحتوي المتغير العام optarg على مؤشر إلى قيمة الخيار (إذا كان الخيار يتطلب قيمة).

يتحكم المتغير العام opterr في سلوك طباعة رسائل الخطأ. إذا كان opterr غير صفر، فستقوم getopt() بطباعة رسالة إلى الخطأ القياسي لجميع أخطاء تحليل الخيارات.

تتوقع optstring سلسلة من أحرف الخيارات. إذا كان حرف الخيار متبوعًا بنقطتين رأسيتين (':')، فيجب أن يأخذ الخيار وسيطة. إذا كان حرف الخيار متبوعًا بثلاث نقاط رأسية ("::")، فإنه يأخذ وسيطة اختيارية.

getopt_long() و getopt_long_only() تدعمان الخيارات الطويلة (التي تبدأ بـ "--").

قيمة الإرجاع

تعتمد قيمة الإرجاع على الحرف الأول في optstring: إذا كان ':'، يتم إرجاع ':'؛ وإلا، يتم إرجاع '؟'.

getopt_long() و getopt_long_only() ترجعان أيضًا حرف الخيار عند التعرف على خيار قصير. بالنسبة للخيار الطويل، ترجعان val إذا كان flag NULL، و 0 بخلاف ذلك. إرجاع الأخطاء و -1 هي نفسها كما في getopt()، بالإضافة إلى '؟' لمطابقة غامضة أو معلمة إضافية.

البيئة

POSIXLY_CORRECT
إذا تم تعيين هذا المتغير، فإن معالجة الخيارات تتوقف بمجرد مواجهة معلمة غير خيار.

السمات

للحصول على شرح للمصطلحات المستخدمة في هذا القسم، راجع attributes(7).

الواجهة السمة القيمة
getopt ()، getopt_long ()، getopt_long_only () سلامة الترابط MT-Unsafe race:getopt env

الإصدارات

يحدد POSIX أن وسيطة مصفوفة argv يجب أن تكون const، لكن هذه الدوال تعيد ترتيب عناصرها ما لم يتم تعيين متغير البيئة POSIXLY_CORRECT. يتم استخدام const في النموذج الأولي الفعلي ليكون متوافقًا مع الأنظمة الأخرى؛ ومع ذلك، لا تعرض هذه الصفحة المؤهل لتجنب إرباك القراء.

المعايير

getopt()
POSIX.1-2008.
getopt_long()
getopt_long_only()
GNU.
استخدام '+' و '-' في بداية optstring هو امتداد لـ GNU.

التاريخ

getopt()
POSIX.1-2001، و POSIX.2.

في بعض التنفيذات القديمة، تم الإعلان عن getopt() في <stdio.h>. سمحت SUSv1 بالإعلان في <unistd.h> أو <stdio.h>. علّمت POSIX.1-1996 استخدام <stdio.h> لهذا الغرض كـ LEGACY. لا تتطلب POSIX.1-2001 الإعلان في <stdio.h>.

تأثرت الإصدارات القديمة جدًا من glibc بمتغير البيئة _PID_GNU_nonoption_argv_flags_.

ملاحظات

البرنامج الذي يفحص متجهات وسائط متعددة، أو يعيد فحص نفس المتجه أكثر من مرة، ويريد الاستفادة من امتدادات GNU مثل '+' و '-' في بداية optstring، أو يغير قيمة POSIXLY_CORRECT بين الفحوصات، يجب عليه إعادة تهيئة getopt() عن طريق إعادة تعيين optind إلى 0، بدلاً من القيمة التقليدية 1. (إعادة التعيين إلى 0 يجبر استدعاء روتين تهيئة داخلي يعيد فحص POSIXLY_CORRECT ويتحقق من امتدادات GNU في optstring.)

يتم تحليل معلمات سطر الأوامر بترتيب صارم مما يعني أن خيارًا يتطلب وسيطة سيستهلك المعلمة التالية، بغض النظر عما إذا كانت تلك المعلمة هي وسيطة الخيار المحددة بشكل صحيح أو ببساطة الخيار التالي (في سيناريو يسيء المستخدم فيه تحديد سطر الأوامر). على سبيل المثال، إذا تم تحديد optstring كـ "1n:"؛ وحدد المستخدم معلمات سطر الأوامر بشكل غير صحيح كـ prog -n -1، فسيتم إعطاء الخيار -n قيمة optarg "-1"؛، وسيتم اعتبار الخيار -1 على أنه لم يتم تحديده.

أمثلة

getopt()

يوضح مثال البرنامج البسيط التالي استخدام getopt() للتعامل مع خيارين للبرنامج: -n، بدون قيمة مرتبطة؛ و -t val، والذي يتوقع قيمة مرتبطة.

#include <stdio.h>
#include <stdlib.h>
#include <unistd.h>
int
main(int argc, char *argv[])
{
    int flags, opt;
    int nsecs, tfnd;
    nsecs = 0;
    tfnd = 0;
    flags = 0;
    while ((opt = getopt(argc, argv, "nt:")) != -1) {
        switch (opt) {
        case 'n':
            flags = 1;
            break;
        case 't':
            nsecs = atoi(optarg);
            tfnd = 1;
            break;
        default: /* '?' */
            fprintf(stderr, "Usage: %s [-t nsecs] [-n] name\n",
                    argv[0]);
            exit(EXIT_FAILURE);
        }
    }
    printf("flags=%d; tfnd=%d; nsecs=%d; optind=%d\n",
           flags, tfnd, nsecs, optind);
    if (optind >= argc) {
        fprintf(stderr, "Expected argument after options\n");
        exit(EXIT_FAILURE);
    }
    printf("name argument = %s\n", argv[optind]);
    /* Other code omitted */
    exit(EXIT_SUCCESS);
}

getopt_long()

يوضح مثال البرنامج التالي استخدام getopt_long() مع معظم ميزاته.

#include <getopt.h>
#include <stdio.h>     /* for printf */
#include <stdlib.h>    /* for exit */
int
main(int argc, char *argv[])
{
    int c;
    int digit_optind = 0;
    while (1) {
        int this_option_optind = optind ? optind : 1;
        int option_index = 0;
        static struct option long_options[] = {
            {"add",     required_argument, 0,  0 },
            {"append",  no_argument,       0,  0 },
            {"delete",  required_argument, 0,  0 },
            {"verbose", no_argument,       0,  0 },
            {"create",  required_argument, 0, 'c'},
            {"file",    required_argument, 0,  0 },
            {0,         0,                 0,  0 }
        };
        c = getopt_long(argc, argv, "abc:d:012",
                        long_options, &option_index);
        if (c == -1)
            break;
        switch (c) {
        case 0:
            printf("option %s", long_options[option_index].name);
            if (optarg)
                printf(" with arg %s", optarg);
            printf("\n");
            break;
        case '0':
        case '1':
        case '2':
            if (digit_optind != 0 && digit_optind != this_option_optind)
              printf("digits occur in two different argv-elements.\n");
            digit_optind = this_option_optind;
            printf("option %c\n", c);
            break;
        case 'a':
            printf("option a\n");
            break;
        case 'b':
            printf("option b\n");
            break;
        case 'c':
            printf("option c with value '%s'\n", optarg);
            break;
        case 'd':
            printf("option d with value '%s'\n", optarg);
            break;
        case '?':
            break;
        default:
            printf("?? getopt returned character code 0%o ??\n", c);
        }
    }
    if (optind < argc) {
        printf("non-option ARGV-elements: ");
        while (optind < argc)
            printf("%s ", argv[optind++]);
        printf("\n");
    }
    exit(EXIT_SUCCESS);
}
