عند الكتابة، لا نتوقع رؤية استجابة EOF، لذا نعامل تلك الحالة بنفس طريقة الخطأ الفادح.

قراءة استجابة من الخادم مشابهة:

do {
    /*
     * الحصول على ما يصل إلى sizeof(buf) بايت من الاستجابة. نستمر في القراءة حتى
     * يقوم الخادم بإغلاق الاتصال.
     */
    while (!eof && !SSL_read_ex(ssl, buf, sizeof(buf), &readbytes)) {
        switch (handle_io_failure(ssl, 0)) {
        case 1:
            continue; /* إعادة المحاولة */
        case 0:
            eof = 1;
            continue;
        case -1:
        default:
            printf("فشل قراءة البيانات المتبقية\n");
            goto end; /* لا يمكن إعادة المحاولة: خطأ */
        }
    }
    /*
     * لا تضمن OpenSSL أن البيانات التي تم إرجاعها هي سلسلة نصية أو
     * أنها منتهية بـ NUL، لذا نستخدم fwrite() لكتابة العدد الدقيق
     * للبايتات التي قرأناها. يمكن أن تكون البيانات غير قابلة للطباعة أو
     * تحتوي على أحرف NUL في منتصفها. لهذا المثال البسيط،
     * سنقوم بطباعتها إلى stdout على أي حال.
     */
    if (!eof)
        fwrite(buf, 1, readbytes, stdout);
} while (!eof);
/* في حال لم تنته الاستجابة بسطر جديد، نضيف واحدًا الآن */
printf("\n");

الفرق الرئيسي هذه المرة هو أنه من الصالح لنا تلقي استجابة EOF عند محاولة قراءة البيانات من الخادم. سيحدث هذا عندما يغلق الخادم الاتصال بعد إرسال جميع البيانات في استجابته.

في هذا العرض التوضيحي، نقوم فقط بطباعة جميع البيانات التي تلقيناها في الاستجابة من الخادم. نستمر في الدوران في الحلقة حتى نواجه خطأ فادحًا، أو نتلقى EOF (مما يشير إلى إنهاء سلس).

إغلاق الاتصال

كما هو الحال في مثال عميل QUIC الحظر، يجب علينا إغلاق الاتصال عندما ننتهي منه.

على الرغم من أننا تلقينا EOF في الدفق الذي كنا نقرأ منه أعلاه، إلا أن هذا لا يخبرنا شيئًا عن حالة الاتصال الأساسي. سيقوم تطبيق العرض التوضيحي الخاص بنا ببدء عملية إغلاق الاتصال عبر SSL_shutdown(3).

نظرًا لأن تطبيقنا يبدأ عملية الإغلاق، فقد نتوقع رؤية SSL_shutdown(3) يعطي قيمة إرجاع 0، ثم يجب أن نستمر في استدعائه حتى نتلقى قيمة إرجاع 1 (مما يعني أننا أكملنا الإغلاق بنجاح). نظرًا لأننا نستخدم كائن SSL غير حاظر، فقد نتوقع الاضطرار إلى إعادة محاولة هذه العملية عدة مرات. إذا أعادت SSL_shutdown(3) نتيجة سالبة، فيجب علينا استدعاء SSL_get_error(3) لمعرفة ما يجب فعله بعد ذلك. نستخدم دالة handle_io_failure() التي طورناها سابقًا لهذا الغرض:

/*
 * استدعاء SSL_shutdown() بشكل متكرر حتى يتم إغلاق الاتصال بالكامل.
 */
while ((ret = SSL_shutdown(ssl)) != 1) {
    if (ret < 0 && handle_io_failure(ssl, ret) == 1)
        continue; /* إعادة المحاولة */
}

التنظيف النهائي

كما هو الحال مع مثال عميل QUIC الحظر، بمجرد الانتهاء من الاتصال، يجب علينا تحريره. خطوات القيام بذلك لهذا المثال هي نفسها كما هو الحال بالنسبة للمثال الحظر، لذا لن نكررها هنا.
