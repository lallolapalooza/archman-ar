ossl-guide-quic-server-non-block - دليل OpenSSL: كتابة خادم QUIC بسيط غير متزامن

مثال بسيط لخادم QUIC غير متزامن

تقدم هذه الصفحة عينات مختلفة من شفرة المصدر توضح كيفية كتابة تطبيق خادم QUIC بسيط وغير متزامن يقوم بإعادة توجيه "الصدى"، يقبل اتصال عميل واحد في كل مرة، ويعيد إرسال المدخلات من العميل إلى نفس العميل. بمجرد قطع اتصال العميل الحالي، يتم قبول اتصال العميل التالي.

يقبل الخادم فقط بروتوكولات ALPN "http/1.0" و "hq-interop" ولا يقوم بتنفيذ HTTP فعليًا، بل يقوم فقط بإعادة توجيه بسيطة. هذا غير قياسي ولن تدعمه الخوادم الواقعية. هذا لأغراض توضيحية فقط.

هناك طرق مختلفة لاختبار هذا الخادم: quic-client-block.c و quic-client-non-block.c سترسل طلب HTTP/1.0 أساسي، والذي سيعيده الخادم. يمكنك أيضًا اختبار هذا الخادم عن طريق تشغيل "openssl s_client -connect localhost:4443 -4 -quic -alpn http/1.0" وإدخال نص سيقوم الخادم بإعادته.

كل من مقبس الاستماع والمقبس المتصل هما "غير متزامنين". ومع ذلك، نستخدم select() لجعل مقبس الاستماع يقوم بعملية حظر عندما لا يمكنه القراءة/الكتابة. بدلاً من التوقف والانتظار، قد تحتاج إلى تطبيقك إلى القيام بمهام أخرى أثناء عدم قدرة كائن SSL على القراءة/الكتابة. على سبيل المثال: تحديث واجهة المستخدم الرسومية أو إجراء عمليات على اتصال أو دفق آخر.

الكود المصدري الكامل لمثال خادم QUIC غير المتزامن هذا متاح في الدليل demos/guide في توزيعة OpenSSL المصدرية في الملف quic-server-non-block.c. وهو متاح أيضًا عبر الإنترنت على https://github.com/openssl/openssl/blob/master/demos/guide/quic-server-non-block.c.

نفترض أن لديك OpenSSL مثبتًا بالفعل على نظامك؛ وأن لديك بالفعل فهمًا أساسيًا لمفاهيم OpenSSL و QUIC (راجع ossl-guide-libraries-introduction(7) و ossl-guide-quic-introduction(7))؛ وأنك تعرف كيفية كتابة وبناء كود C وربطه بمكتبات libcrypto و libssl التي توفرها OpenSSL. كما يفترض أن لديك فهمًا أساسيًا لـ UDP/IP والمقابس.

إنشاء كائنات SSL_CTX و SSL

الخطوة الأولى هي إنشاء كائن SSL_CTX لخادمنا. نستخدم الدالة SSL_CTX_new(3) لهذا الغرض. نمرر كوسيطة قيمة الإرجاع للدالة OSSL_QUIC_server_method(3). يجب عليك استخدام هذه الطريقة كلما كنت تكتب خادم QUIC.

/*
 * يحتوي SSL_CTX على معلومات تكوين مشتركة لعدة اتصالات SSL لاحقة لكل عميل.
 * نقوم بتحميل طريقة QUIC الخاصة بالخادم هنا بشكل خاص.
 */
ctx = SSL_CTX_new(OSSL_QUIC_server_method());
if (ctx == NULL)
    goto err;

تحتاج الخوادم إلى مفتاح خاص وشهادة. غالبًا ما تكون شهادات CA للمصدر الوسيط مطلوبة، ويتم تكوين كل من شهادة الخادم (الكيان النهائي أو EE) وشهادات المصدر ("السلسلة") بسهولة في "ملف سلسلة" واحد. أدناه نقوم بتحميل ملف سلسلة من هذا القبيل (يجب أن تظهر شهادة EE أولاً)، ثم نقوم بتحميل المفتاح الخاص المقابل، ونتحقق من أنه يطابق شهادة الخادم. لا يتم إجراء أي فحوصات للتحقق من سلامة السلسلة (توقيعات CA أو تواريخ انتهاء صلاحية الشهادات، على سبيل المثال)، ولكننا نتحقق من اتساق المفتاح الخاص مع الشهادة المقابلة.

/*
 * قم بتحميل ملف *سلسلة* شهادة الخادم (بتنسيق PEM)، والذي يتضمن
 * ليس فقط شهادة الخادم الطرفية (الكيان النهائي)، ولكن أيضًا أي
 * شهادات CA الوسيطة للمصدر. يجب أن تكون شهادة الطرفية هي
 * الشهادة الأولى في الملف.
 *
 * في حالات الاستخدام المتقدمة، يمكن استدعاء هذا عدة مرات، مرة لكل خوارزمية مفتاح عام
 * لديها الخادم شهادة مقابلة لها.
 * ومع ذلك، يجب تحميل المفتاح الخاص المقابل (انظر أدناه) أولاً،
 * * قبل * الانتقال إلى ملف السلسلة التالي.
 */
if (SSL_CTX_use_certificate_chain_file(ctx, cert_path) <= 0) {
    fprintf(stderr, "لم نتمكن من تحميل ملف الشهادة: %s\n", cert_path);
    goto err;
}
/*
 * قم بتحميل المفتاح الخاص المقابل، وهذا يتحقق أيضًا من أن المفتاح الخاص
 * يتطابق مع شهادة الكيان النهائي المحملة للتو. لا يتحقق
 * مما إذا كانت سلسلة الشهادات صالحة، يمكن أن تكون الشهادات
 * منتهية الصلاحية، أو قد تفشل بطريقة أخرى في تشكيل سلسلة يمكن للعميل
 * التحقق منها.
 */
if (SSL_CTX_use_PrivateKey_file(ctx, key_path, SSL_FILETYPE_PEM) <= 0) {
    fprintf(stderr, "لم نتمكن من تحميل ملف المفتاح: %s\n", key_path);
    goto err;
}

معظم الخوادم، بما في ذلك هذا الخادم، لا تطلب شهادات العملاء. لذلك، لا نحتاج إلى "متجر موثوق" ونسمح بإكمال المصافحة حتى عندما لا يقدم العميل شهادة. ملاحظة: حتى لو قدم العميل شهادة موثوقة، لكي تكون مفيدة، سيحتاج تطبيق الخادم إلى شفرة مخصصة لاستخدام الهوية التي تم التحقق منها لمنح وصول غير افتراضي لهذا العميل المحدد. تمنح بعض الخوادم الوصول لجميع العملاء بشهادات من CA خاص، وهذا يتطلب معالجة قوائم إبطال الشهادات لسحب تفويض عميل. غالبًا ما يكون من الأسهل وأكثر أمانًا الاحتفاظ بقائمة من المفاتيح العامة المصرح بها بدلاً من ذلك.

على الرغم من أن هذا هو الإعداد الافتراضي، إلا أننا نستدعي الدالة SSL_CTX_set_verify(3) بشكل صريح ونمرر القيمة SSL_VERIFY_NONE إليها. الوسيط النهائي لهذه الدالة هو دالة رد اتصال يمكنك توفيرها اختياريًا لتجاوز المعالجة الافتراضية للتحقق من الشهادة. معظم التطبيقات لا تحتاج إلى القيام بذلك، لذلك يمكن تعيينها بأمان إلى NULL للحصول على المعالجة الافتراضية.

/*
 * نادرًا ما يستخدم العملاء المصادقة المستندة إلى الشهادات، ولذلك لا نطلب
 * مصادقة TLS "متبادلة" (في الواقع لا توجد طريقة لمعرفة
 * ما إذا كان العميل قد صادق على الخادم وكيف، لذلك فإن مصطلح "متبادل"
 * قد يكون مضللاً).
 *
 * بما أننا لا نطلب أو نعالج شهادات العملاء، فإننا لا نحتاج إلى
 * تكوين مخزن شهادات موثوق به، لذلك لا يلزم استدعاء
 * SSL_CTX_set_default_verify_paths(). يُفترض أن سلسلة شهادات الخادم
 * الخاصة صحيحة.
 */
SSL_CTX_set_verify(ctx, SSL_VERIFY_NONE, NULL);

يفرض QUIC أيضًا استخدام التفاوض على بروتوكول طبقة التطبيق (ALPN) لاختيار بروتوكول تطبيق. نستخدم SSL_CTX_set_alpn_select_cb(3) لهذا الغرض. يمكننا تمرير دالة رد اتصال سيتم استدعاؤها لكل اتصال لاختيار ALPN يعتبره الخادم مقبولًا.

/* إعداد دالة رد اتصال التفاوض ALPN لتحديد ALPN المقبولة. */
SSL_CTX_set_alpn_select_cb(ctx, select_alpn, NULL);

في هذه الحالة، نقبل فقط "http/1.0" و "hq-interop".

/*
* سلاسل ALPN لمصافحة TLS. فقط 'http/1.0' و 'hq-interop'
* مقبولة.
*/
static const unsigned char alpn_ossltest[] = {
    8,  'h', 't', 't', 'p', '/', '1', '.', '0',
    10, 'h', 'q', '-', 'i', 'n', 't', 'e', 'r', 'o', 'p',
};
static int select_alpn(SSL *ssl, const unsigned char **out,
                       unsigned char *out_len, const unsigned char *in,
                       unsigned int in_len, void *arg)
{
    if (SSL_select_next_proto((unsigned char **)out, out_len, alpn_ossltest,
                              sizeof(alpn_ossltest), in,
                              in_len) == OPENSSL_NPN_NEGOTIATED)
        return SSL_TLSEXT_ERR_OK;
    return SSL_TLSEXT_ERR_ALERT_FATAL;
}

هذا كل الإعداد الذي نحتاج إلى القيام به لـ SSL_CTX. بعد ذلك، نقوم بإنشاء مقبس UDP وربطه على localhost.

/* الحصول على واصف الملف لمقبس UDP جديد */
if ((fd = socket(AF_INET, SOCK_DGRAM, IPPROTO_UDP)) < 0) {
    fprintf(stderr, "لا يمكن إنشاء المقبس");
    return -1;
}
sa.sin_family = AF_INET;
sa.sin_port = htons(port);
/* الربط بمقبس UDP الجديد على localhost */
if (bind(fd, (const struct sockaddr *)&sa, sizeof(sa)) < 0) {
    fprintf(stderr, "لا يمكن الربط بـ %u\n", port);
    BIO_closesocket(fd);
    return -1;
}
/* ضبط المنفذ على وضع غير متزامن */
if (BIO_socket_nbio(fd, 1) <= 0) {
    fprintf(stderr, "غير قادر على ضبط المنفذ على وضع غير متزامن");
    BIO_closesocket(fd);
    return -1;
}

حلقة الخادم

يدخل الخادم الآن في حلقة "إلى الأبد"، يتعامل مع اتصال عميل واحد في كل مرة. قبل كل اتصال، نمسح مكدس أخطاء OpenSSL بحيث تكون تقارير الأخطاء متعلقة بالاتصال الجديد فقط.

/* Pristine error stack for each new connection */
ERR_clear_error();

ثم ننتظر حتى يصبح الاتصال جاهزًا للقراءة. يستخدم وظيفة select للانتظار حتى يصبح المقبس قابلاً للقراءة أو للكتابة، اعتمادًا على ما يتطلبه اتصال SSL.

ثم نقبل اتصالًا جديدًا تم فيه إجراء المصافحة بالفعل. ومع ذلك، نظرًا لأننا في وضع غير مانع، ستعود SSL_accept_connection(3) فورًا. لذلك، نستخدم وظيفة مساعدة لمنع الانتظار فعليًا حتى يتم إنشاء اتصال.

printf("Waiting for connection\n");
while ((conn = SSL_accept_connection(listener, 0)) == NULL) {
    wait_for_activity(listener);
}
printf("Accepted new connection\n");

تستخدم الوظيفة المساعدة wait_for_activity select() لمنع الانتظار حتى يصبح واصف الملف الخاص بكائن SSL المقدم قابلاً للقراءة. كما ذكرنا سابقًا، من المحتمل أن يستخدم تطبيق أكثر واقعية هذا الوقت للقيام بمهام أخرى.

/* Initialize the fd_set structure */
FD_ZERO(&read_fd);
FD_ZERO(&write_fd);
/*
 * Determine if we would like to write to the socket, read from it, or both.
 */
if (SSL_net_write_desired(ssl))
    FD_SET(sock, &write_fd);
if (SSL_net_read_desired(ssl))
    FD_SET(sock, &read_fd);
/*
 * Find out when OpenSSL would next like to be called, regardless of
 * whether the state of the underlying socket has changed or not.
 */
if (SSL_get_event_timeout(ssl, &tv, &isinfinite) && !isinfinite)
    tvp = &tv;
/*
 * Wait until the socket is writeable or readable. We use select here
 * for the sake of simplicity and portability, but you could equally use
 * poll/epoll or similar functions
 *
 * NOTE: For the purposes of this demonstration code this effectively
 * makes this demo block until it has something more useful to do. In a
 * real application you probably want to go and do other work here (e.g.
 * update a GUI, or service other connections).
 *
 * Let's say for example that you want to update the progress counter on
 * a GUI every 100ms. One way to do that would be to use the timeout in
 * the last parameter to "select" below. If the tvp value is greater
 * than 100ms then use 100ms instead. Then, when select returns, you
 * check if it did so because of activity on the file descriptors or
 * because of the timeout. If the 100ms GUI timeout has expired but the
 * tvp timeout has not then go and update the GUI and then restart the
 * "select" (with updated timeouts).
 */
select(sock + 1, &read_fd, &write_fd, NULL, tvp);

بعد اكتمال المصافحة، يقرأ الخادم كل مدخلات العميل.

/* Read from client until the client sends a end of stream packet */
while (!eof) {
    ret = SSL_read_ex(conn, buf + total_read, sizeof(buf) - total_read,
                      &nread);
    total_read += nread;
    if (total_read >= 8192) {
        fprintf(stderr, "Could not fit all data into buffer\n");
        goto err;
    }
    switch (handle_io_failure(conn, ret)) {
    case 1:
        continue; /* Retry */
    case 0:
        /* Reached end of stream */
        if (!SSL_has_pending(conn))
            eof = 1;
        break;
    default:
        fprintf(stderr, "Failed reading remaining data\n");
        goto err;
    }
}

أخيرًا، نعيد إرسال البيانات المستلمة إلى العميل. يمكننا استخدام SSL_write_ex2(3) لتمرير علامة خاصة SSL_WRITE_FLAG_CONCLUDE التي سترسل حزمة FIN بمجرد انتهاء الكتابة بنجاح من كتابة جميع البيانات إلى الطرف المقابل.

/* Echo client input */
while (!SSL_write_ex2(conn, buf,
                      total_read,
                      SSL_WRITE_FLAG_CONCLUDE, &total_written)) {
    if (handle_io_failure(conn, 0) == 1)
        continue;
    fprintf(stderr, "Failed to write data\n");
    goto err;
}

ثم نقوم بإغلاق الاتصال باستخدام SSL_shutdown(3)، والتي قد تحتاج إلى استدعائها عدة مرات لضمان إغلاق الاتصال بالكامل.

/*
 * Shut down the connection. We may need to call this multiple times
 * to ensure the connection is shutdown completely.
 */
while ((ret = SSL_shutdown(conn)) != 1) {
    if (ret < 0 && handle_io_failure(conn, ret) == 1)
        continue; /* Retry */
}

أخيرًا، نحرر اتصال SSL، ويكون الخادم الآن جاهزًا لقبول اتصال العميل التالي.

SSL_free(conn);

التنظيف النهائي

إذا تمكن الخادم بطريقة ما من الخروج من الحلقة اللانهائية وكان مستعدًا للخروج، فسيقوم بإلغاء تخصيص SSL المنشأ.

SSL_free(listener);

وفي الدالة الرئيسية، سيقوم بإلغاء تخصيص SSL_CTX المنشأ.

SSL_CTX_free(ctx);
BIO_closesocket(fd);
