مسار الكائن
قابل للتحديد بحرية (دور الهدف) [بادئة متغيرة]/{hci0,hci1,...}/dev_{BDRADDR}/player#/item# (دور المتحكم)

org.bluez.Error.Failed

void AddtoNowPlaying()

org.bluez.Error.NotSupported
org.bluez.Error.Failed

الخصائص

object Player [للقراءة فقط]

string Name [للقراءة فقط]

اسم العنصر القابل للعرض

string Type [للقراءة فقط]

نوع العنصر

القيم الممكنة:

"video"
"audio"
"folder"

string FolderType [للقراءة فقط، اختياري]

نوع المجلد.

متوفر إذا كانت الخاصية Type هي "Folder"

القيم الممكنة:

"mixed"
"titles"
"albums"
"artists"

boolean Playable [للقراءة فقط، اختياري]

يشير إلى ما إذا كان يمكن تشغيل العنصر

متوفر إذا كانت الخاصية Type هي "folder"

dict Metadata [للقراءة فقط]

بيانات وصفية للعنصر.

القيم الممكنة:

string Title
اسم عنوان العنصر

متوفر إذا كانت الخاصية Type هي "audio" أو "video"

string Artist
اسم الفنان للعنصر

متوفر إذا كانت الخاصية Type هي "audio" أو "video"

string Album
اسم الألبوم للعنصر

متوفر إذا كانت الخاصية Type هي "audio" أو "video"

string Genre
اسم النوع للعنصر

متوفر إذا كانت الخاصية Type هي "audio" أو "video"

uint32 NumberOfTracks
إجمالي عدد المقطوعات في الألبوم للعنصر

متوفر إذا كانت الخاصية Type هي "audio" أو "video"

uint32 Number
رقم الألبوم للعنصر

متوفر إذا كانت الخاصية Type هي "audio" أو "video"

uint32 Duration

متوفر إذا كانت الخاصية Type هي "audio" أو "video"
