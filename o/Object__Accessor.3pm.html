Object__Accessor(3pm)

الحزمة: perl-object-accessor

التراخيص: PerlArtistic, GPL

Object::Accessor(3) وثائق Perl مقدمة من المستخدم Object::Accessor(3)

الاسم

Object::Accessor - واجهة لإنشاء وصولات خاصة بالكائنات

ملخص

### استخدام الكائن
$obj = Object::Accessor->new;        # إنشاء كائن
$obj = Object::Accessor->new(@list); # إنشاء كائن مع وصولات
$obj = Object::Accessor->new(\%h);   # إنشاء كائن مع وصولات
                                     # ومعالجات السماح الخاصة بها
$bool   = $obj->mk_accessors('foo'); # إنشاء وصولات
$bool   = $obj->mk_accessors(        # إنشاء وصولات مع التحقق من الصحة
           {foo => ALLOW_HANDLER} ); # المدخلات
$bool   = $obj->mk_aliases(          # إنشاء اسم مستعار لاسم طريقة موجود
            alias_name => 'method'); #
$clone  = $obj->mk_clone;            # إنشاء نسخة من الكائن الأصلي
                                     # بدون بيانات
$bool   = $obj->mk_flush;            # مسح جميع البيانات
@list   = $obj->ls_accessors;        # استرداد قائمة بجميع
                                     # الوصولات لهذا الكائن
$bar    = $obj->foo('bar');          # تعيين 'foo' إلى 'bar'
$bar    = $obj->foo();               # استرداد 'bar' مرة أخرى
$sub    = $obj->can('foo');          # استرداد مرجع الدالة لـ
                                     # الوصول 'foo'
$bar    = $sub->('bar');             # تعيين 'foo' عبر مرجع الدالة
$bar    = $sub->();                  # استرداد 'bar' عبر مرجع الدالة
### استخدام الكائن كفئة أساسية
package My::Class;
use base 'Object::Accessor';
$obj    = My::Class->new;               # إنشاء كائن أساسي
$bool   = $obj->mk_accessors('foo');    # إنشاء وصولات، إلخ...
### جعل جميع محاولات الوصول إلى وصولات غير موجودة قاتلة
### (الافتراضي هو false)
$Object::Accessor::FATAL = 1;
### تمكين التصحيح
$Object::Accessor::DEBUG = 1;
### استخدام متقدم - استدعاءات رد الاتصال
{   my $obj = Object::Accessor->new('foo');
    $obj->register_callback( sub { ... } );
    $obj->foo( 1 ); # هذه الاستدعاءات تستدعي رد الاتصال الذي سجلته
    $obj->foo()     # مما يسمح لك بتغيير سلوك الحصول/التعيين
                    # وما يتم إرجاعه للمتصل.
}
### استخدام متقدم - سمات lvalue
{   my $obj = Object::Accessor::Lvalue->new('foo');
    print $obj->foo = 1;            # سيطبع 1
}
### استخدام متقدم - قيم السمات المقيدة بالنطاق
{   my $obj = Object::Accessor->new('foo');
    $obj->foo( 1 );
    print $obj->foo;                # سيطبع 1
    ### ربط نطاق قيمة السمة 'foo'
    ### بنطاق '$x' - عندما يخرج '$x' من
    ### النطاق، سيتم استعادة القيمة السابقة لـ 'foo'
    {   $obj->foo( 2 => \my $x );
        print $obj->foo, ' ', $x;   # سيطبع '2 2'
    }
    print $obj->foo;                # سيطبع 1
}

الوصف

"Object::Accessor" يوفر واجهة لإنشاء وصولات خاصة بالكائن (على عكس الوصولات الخاصة بـ "Class"، كما توفرها، على سبيل المثال، "Class::Accessor").

يمكنك اختيار إما أن ترث من هذه الوحدة، وبالتالي استخدام وصولاتها في وحدتك الخاصة، أو تخزين كائن "Object::Accessor" داخل الكائن الخاص بك، والوصول إلى الوصولات من هناك. انظر "ملخص" لأمثلة.

الطرق

$object = Object::Accessor->new( [ARGS] );

ينشئ كائن "Object::Accessor" جديد (وفارغ). هذه الطريقة قابلة للوراثة.

سيتم تمرير أي وسائط معطاة إلى "new" مباشرة إلى "mk_accessors".

إذا كنت تريد أن تكون قادرًا على تعيين قيم لـ وصولاتك كما لو كانت "lvalue"، فيجب عليك إنشاء الكائن الخاص بك في فضاء الأسماء "Object::Accessor::Lvalue" بدلاً من ذلك. انظر القسم الخاص بـ "وصولات LVALUE" أدناه.

$bool = $object->mk_accessors( @ACCESSORS | \%ACCESSOR_MAP );

ينشئ قائمة بالوصولات لهذا الكائن (وليس لـ "ليس" الكائنات الأخرى في نفس الفئة!). لن يقوم بالكتابة فوق البيانات الموجودة، لذلك إذا كان الوصول موجودًا بالفعل، فإن طلب إنشائه مرة أخرى هو فعليًا "لا شيء".

عند تقديم "hashref" كوسيط، بدلاً من قائمة عادية، يمكنك تحديد قائمة أزواج مفتاح/قيمة للوصولات ومدققاتها المدخلات. يمكن أن تكون المدققات أي شيء تقبله وظيفة "allow" الخاصة بـ "Params::Check". يرجى الاطلاع على صفحة الدليل الخاصة بها للحصول على التفاصيل.

على سبيل المثال:

$object->mk_accessors( {
    foo     => qr/^\d+$/,       # أرقام فقط
    bar     => [0,1],           # قيم منطقية
    zot     => \&my_sub         # دالة تحقق مخصصة
} );

يعيد صحيحًا عند النجاح، وخاطئًا عند الفشل.

الوصولات التي يتم استدعاؤها على كائن، والتي لا توجد، تعيد "undef" افتراضيًا، ولكن يمكنك جعل هذا خطأ قاتلاً عن طريق تعيين المتغير العام $FATAL إلى صحيح. انظر القسم الخاص بـ "المتغيرات العامة" للتفاصيل.

لاحظ أنه يمكنك ربط قيم السمات بنطاق. هذا يسمح لك بتغيير قيمة سمة "مؤقتًا"، وجعل قيمتها الأصلية تستعاد عند انتهاء نطاق متغيرها المرتبط؛

على سبيل المثال، في هذا المقتطف من التعليمات البرمجية، سيتم تعيين السمة "foo" مؤقتًا إلى 2، حتى نهاية نطاق $x، وعندها سيتم استعادة القيمة الأصلية 1.

my $obj = Object::Accessor->new;
$obj->mk_accessors('foo');
$obj->foo( 1 );
print $obj->foo;                # سيطبع 1
### ربط نطاق قيمة السمة 'foo'
### بنطاق '$x' - عندما يخرج '$x' من
### النطاق، سيتم استعادة القيمة السابقة لـ 'foo'
{   $obj->foo( 2 => \my $x );
    print $obj->foo, ' ', $x;   # سيطبع '2 2'
}
print $obj->foo;                # سيطبع 1

لاحظ أن جميع الوصولات قابلة للقراءة/الكتابة للجميع. انظر قسم "TODO" للتفاصيل.

@list = $self->ls_accessors;

يعيد قائمة بالوصولات التي يدعمها الكائن الحالي. يمكن استرداد مراجع الدوال المقابلة عن طريق تمرير هذه القائمة واحدة تلو الأخرى إلى طريقة "can".

$ref = $self->ls_allow(KEY)

يعيد معالج السماح للمفتاح المعطى، والذي يمكن استخدامه مع معالج allow() الخاص بـ "Params::Check". إذا لم يتم تحديد معالج سماح، فسيتم إرجاع معالج سماح يعيد دائمًا صحيحًا.

$bool = $self->mk_aliases( alias => method, [alias2 => method2, ...] );

ينشئ اسمًا مستعارًا لاسم طريقة معطى. لجميع الأغراض والغايات، هاتان الوصولتان متطابقتان الآن لهذا الكائن. هذا يشبه القيام بما يلي على مستوى جدول الرموز:

*alias = *method

هذا يسمح لك بالقيام بما يلي:

$self->mk_accessors('foo');
$self->mk_aliases( bar => 'foo' );
$self->bar( 42 );
print $self->foo;     # سيطبع 42

$clone = $self->mk_clone;

ينشئ نسخة من الكائن الحالي، والتي سيكون لها نفس الوصولات تمامًا مثل الكائن الحالي، ولكن بدون البيانات المخزنة فيها.

$bool = $self->mk_flush;

يمسح جميع البيانات من الكائن الحالي؛ سيتم إعادة تعيين جميع الوصولات إلى حالتها الافتراضية وهي "undef".

يعيد صحيحًا عند النجاح وخاطئًا عند الفشل.

$bool = $self->mk_verify;

يتحقق مما إذا كانت جميع القيم في الكائن الحالي متوافقة
