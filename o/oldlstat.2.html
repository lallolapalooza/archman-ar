stat(2) دليل استدعاءات النظام stat(2)

الاسم

stat, fstat, lstat, fstatat - الحصول على حالة الملف

المكتبة

مكتبة C القياسية (libc، -lc)

الملخص

#include <sys/stat.h>
int stat(const char *restrict path,
         struct stat *restrict statbuf);
int fstat(int fd, struct stat *statbuf);
int lstat(const char *restrict path,
         struct stat *restrict statbuf);
#include <fcntl.h>           /* تعريف الثوابت AT_* */
#include <sys/stat.h>
int fstatat(int dirfd, const char *restrict path,
         struct stat *restrict statbuf, int flags);

الوصف

تقوم استدعاءات النظام هذه بإرجاع معلومات حول ملف، في المخزن المؤقت الذي يشير إليه statbuf. لا يلزم وجود أذونات على الملف نفسه، ولكن — في حالة stat() و fstatat() و lstat() — يلزم إذن التنفيذ (البحث) على جميع الدلائل في path التي تؤدي إلى الملف.

stat() و fstatat() تستردان معلومات حول الملف الذي يشير إليه path؛ تختلف الاختلافات الخاصة بـ fstatat() فيما يلي.

lstat() مطابقة لـ stat()، إلا إذا كان path رابطًا رمزيًا، فإنه تُرجع معلومات حول الرابط نفسه، وليس الملف الذي يشير إليه الرابط.

fstat() مطابقة لـ stat()، باستثناء أن الملف الذي يجب الحصول على معلومات حوله يتم تحديده بواسطة واصف الملف fd.

بنية stat

جميع استدعاءات النظام هذه ترجع بنية stat (انظر stat(3type)).

ملاحظة: لأسباب تتعلق بالأداء والبساطة، قد تحتوي حقول مختلفة في بنية stat على معلومات الحالة من لحظات مختلفة أثناء تنفيذ استدعاء النظام. على سبيل المثال، إذا تم تغيير st_mode أو st_uid بواسطة عملية أخرى عن طريق استدعاء chmod(2) أو chown(2)، فقد تُرجع stat() الـ st_mode القديم مع الـ st_uid الجديد، أو الـ st_uid القديم مع الـ st_mode الجديد.

fstatat()

استدعاء النظام fstatat() هو واجهة أكثر عمومية للوصول إلى معلومات الملف والتي لا تزال قادرة على توفير سلوك دقيق لكل من stat() و lstat() و fstat().

إذا كان path نسبيًا، فسيتم تفسيره بالنسبة للدليل الذي يشير إليه واصف الملف dirfd (بدلاً من الدليل الحالي للعمل لعملية الاستدعاء، كما يتم القيام به بواسطة stat() و lstat() لمسار نسبي).

إذا كان path نسبيًا و dirfd هي القيمة الخاصة AT_FDCWD، فسيتم تفسير path بالنسبة للدليل الحالي للعمل لعملية الاستدعاء (مثل stat() و lstat()).

إذا كان path مطلقًا، فسيتم تجاهل dirfd.

يمكن أن تكون flags إما 0، أو تتضمن واحدًا أو أكثر من الأعلام التالية ORed:

AT_EMPTY_PATH (منذ Linux 2.6.39)
إذا كان path سلسلة فارغة (أو NULL، منذ Linux 6.11) فقم بالعمل على الملف الذي يشير إليه dirfd (الذي ربما تم الحصول عليه باستخدام open(2) مع العلم O_PATH). في هذه الحالة، يمكن أن يشير dirfd إلى أي نوع من الملفات، وليس مجرد دليل، وسيكون سلوك fstatat() مشابهًا لسلوك fstat(). إذا كان dirfd هو AT_FDCWD، فإن الاستدعاء يعمل على الدليل الحالي للعمل. هذا العلم خاص بـ Linux؛ قم بتعريف _GNU_SOURCE للحصول على تعريفه.
AT_NO_AUTOMOUNT (منذ Linux 2.6.38)
لا تقم بتثبيت المكون النهائي ("اسم الملف الأساسي") لـ path تلقائيًا. (تم تجاهل هذا العلم منذ Linux 3.1). منذ Linux 4.11، هذا العلم ضمني.
AT_SYMLINK_NOFOLLOW
إذا كان path رابطًا رمزيًا، فلا تقم بإلغاء مرجعيته: بدلاً من ذلك، قم بإرجاع معلومات حول الرابط نفسه، مثل lstat(). (بشكل افتراضي، تقوم fstatat() بإلغاء مرجعية الروابط الرمزية، مثل stat().)

انظر openat(2) لشرح الحاجة إلى fstatat().

قيمة الإرجاع

عند النجاح، يتم إرجاع صفر. عند حدوث خطأ، يتم إرجاع -1، ويتم تعيين errno للإشارة إلى الخطأ.

الأخطاء

EACCES
يُرفض إذن البحث لأحد الدلائل في بادئة المسار لـ path. (انظر أيضًا path_resolution(7)).
EBADF
fd ليس واصف ملف مفتوح صالحًا.
EBADF
(fstatat()) path نسبي ولكن dirfd ليس AT_FDCWD ولا واصف ملف صالحًا.
EFAULT
عنوان سيء.
EINVAL
(fstatat()) تم تحديد علم غير صالح في flags.
ELOOP
تم العثور على عدد كبير جدًا من الروابط الرمزية أثناء اجتياز المسار.
ENAMETOOLONG
path طويل جدًا.
ENOENT
أحد مكونات path غير موجود أو رابط رمزي معلق.
ENOENT
path سلسلة فارغة ولم يتم تحديد AT_EMPTY_PATH في flags.
ENOMEM
نفاد الذاكرة (أي، ذاكرة النواة).
ENOTDIR
أحد مكونات بادئة المسار لـ path ليس دليلًا.
ENOTDIR
(fstatat()) path نسبي و dirfd هو واصف ملف يشير إلى ملف غير دليل.
EOVERFLOW
path أو fd يشير إلى ملف لا يمكن تمثيل حجمه أو رقم inode الخاص به أو عدد الكتل الخاصة به في، على التوالي، أنواع off_t أو ino_t أو blkcnt_t. يمكن أن يحدث هذا الخطأ عندما، على سبيل المثال، تستدعي عملية تم تجميعها على منصة 32 بت بدون -D_FILE_OFFSET_BITS=64 استدعاء stat() على ملف يتجاوز حجمه (1<<31)-1 بايت.

فروق المكتبة/النواة

مع مرور الوقت، أدت الزيادات في حجم هيكل stat إلى ثلاث إصدارات متتالية من stat(): sys_stat() (الفتحة __NR_oldstat)، و sys_newstat() (الفتحة __NR_stat)، و sys_stat64() (الفتحة __NR_stat64) على المنصات 32 بت مثل i386. كان الإصداران الأولان موجودين بالفعل في Linux 1.0 (وإن كان بأسماء مختلفة)؛ تمت إضافة الأخير في Linux 2.4. تنطبق ملاحظات مماثلة على fstat() و lstat().

الهياكل الداخلية للنواة stat التي تتعامل معها الإصدارات المختلفة هي على التوالي:

__old_kernel_stat
الهيكل الأصلي، بحقول ضيقة إلى حد ما، ولا توجد حشو.
stat
حقل st_ino أكبر وحشو أضيف إلى أجزاء مختلفة من الهيكل للسماح بالتوسع المستقبلي.
stat64
حقل st_ino أكبر، حقول st_uid و st_gid أكبر لاستيعاب توسيع Linux-2.4 لمعرفات المستخدمين (UIDs) ومعرفات المجموعات (GIDs) إلى 32 بت، وحقول أخرى مختلفة تم تكبيرها ومزيد من الحشو في الهيكل. (تم استهلاك بايتات حشو مختلفة في النهاية في Linux 2.6، مع ظهور معرفات الأجهزة 32 بت ومكونات النانو ثانية لحقول الطوابع الزمنية.)

تقوم دالة الغلاف glibc stat() بإخفاء هذه التفاصيل عن التطبيقات، واستدعاء أحدث إصدار من استدعاء النظام الذي توفره النواة، وإعادة تجميع المعلومات المرتجعة إذا لزم الأمر للثنائيات القديمة.

على الأنظمة الحديثة 64 بت، الحياة أبسط: يوجد استدعاء نظام stat() واحد، وتتعامل النواة مع هيكل stat يحتوي على حقول ذات حجم كافٍ.

استدعاء النظام الأساسي الذي تستخدمه دالة الغلاف glibc fstatat() هو في الواقع يسمى fstatat64() أو، في بعض البنى، newfstatat().

أمثلة

يستدعي البرنامج التالي lstat() ويعرض الحقول المحددة في هيكل stat المرتجع.

#include <stdint.h>
#include <stdio.h>
#include <stdlib.h>
#include <sys/stat.h>
#include <sys/sysmacros.h>
#include <time.h>
int
main(int argc, char *argv[])
{
    struct stat sb;
    if (argc != 2) {
        fprintf(stderr, "Usage: %s <path>\n", argv[0]);
        exit(EXIT_FAILURE);
    }
    if (lstat(argv[1], &sb) == -1) {
        perror("lstat");
        exit(EXIT_FAILURE);
    }
    printf("ID of containing device:  [%x,%x]\n",
           major(sb.st_dev),
           minor(sb.st_dev));
    printf("File type:                ");
    switch (sb.st_mode & S_IFMT) {
    case S_IFBLK:  printf("block device\n");            break;
    case S_IFCHR:  printf("character device\n");        break;
    case S_IFDIR:  printf("directory\n");               break;
    case S_IFIFO:  printf("FIFO/pipe\n");               break;
    case S_IFLNK:  printf("symlink\n");                 break;
    case S_IFREG:  printf("regular file\n");            break;
    case S_IFSOCK: printf("socket\n");                  break;
    default:       printf("unknown?\n");                break;
    }
    printf("I-node number:            %ju\n", (uintmax_t) sb.st_ino);
    printf("Mode:                     %jo (octal)\n",
           (uintmax_t) sb.st_mode);
    printf("Link count:               %ju\n", (uintmax_t) sb.st_nlink);
    printf("Ownership:                UID=%ju   GID=%ju\n",
           (uintmax_t) sb.st_uid, (uintmax_t) sb.st_gid);
    printf("Preferred I/O block size: %jd bytes\n",
           (intmax_t) sb.st_blksize);
    printf("File size:                %jd bytes\n",
           (intmax_t) sb.st_size);
    printf("Blocks allocated:         %jd\n",
           (intmax_t) sb.st_blocks);
    printf("Last status change:       %s", ctime(&sb.st_ctime));
    printf("Last file access:         %s", ctime(&sb.st_atime));
    printf("Last file modification:   %s", ctime(&sb.st_mtime));
    exit(EXIT_SUCCESS);
}
