Object__Realize__Later(3pm)

الحزمة: perl-object-realize-later

التراخيص: GPL, PerlArtistic

Object::Realize::Later(3) توثيق Perl المقدم من المجتمع Object::Realize::Later(3)

الاسم

Object::Realize::Later - إنشاء الكائنات عند الطلب

الملخص

package MyLazyObject;
use Object::Realize::Later
   becomes => 'MyRealObject',
   realize => 'load';

الوصف

تساعد الفئة "Object::Realize::Later" في تنفيذ التطوير الشفاف للكائنات في الوقت الفعلي. يرتبط هذا بحيل التحميل التلقائي للبيانات، وهو الشقيق الأصغر للتحميل التلقائي للوظائف.

يتعلق التحقيق عند الطلب بكسب الأداء. لماذا يجب أن تنفق وقتًا مكلفًا في تحقيق كائن عندما لا تُستخدم بيانات الكائن أبدًا (أو ليس بعد)؟ في البرامج التفاعلية، يمكن أن يحسّن التحقيق عند الطلب بدء التشغيل: يتم تفعيل تحقيق الكائنات عند استخدامها، مما يوزعها على الوقت.

الطرق

الإنشاء

use(Object::Realize::Later %options)
عند استدعاء ("use") الفئة "Object::Realize::Later"، ستضيف مجموعة من الطرق إلى حزمتك (انظر قسم "الإضافات إلى فئتك"). 
-Option            --Default
 becomes             <required>
 believe_caller      <false>
 realize             <required>
 source_module       <becomes>
 warn_realization    <false>
 warn_realize_again  <false>
becomes => الفئة
ما سيكون نوع الكائن بعد التحقيق؟
believe_caller => BOOLEAN
عند استدعاء طريقة على كائن غير محقق، يتحقق AUTOLOAD مما إذا كانت هذه الطريقة تلبي الحاجة. إذا لم تكن كذلك، لا يتم إجراء التحقيق. ومع ذلك، عندما يمكن أن يؤدي التحقيق إلى كائن يوسّع وظائف الفئة المحددة بـ "becomes"، يجب عليك تعطيل هذا الفحص. في هذه الحالة، حدد true لهذا الخيار.
realize => الطريقة|CODA
كيف سيتم التحويل. إذا حددت مرجع كود، فسيتم استدعاؤه مع الكائن البطيء كوسيطة أولى، والطريقة المطلوبة كوسيطة ثانية.

بعد التحقيق، قد لا يزال لديك الكائن البطيء في أماكن مختلفة. تأكد من أن طريقة التحقيق الخاصة بك تتعامل معه، على سبيل المثال باستخدام Memoize. انظر الأمثلة أدناه.

source_module => الفئة
إذا كانت الفئة (حزمة) مضمنة في ملف (وحدة) باسم مختلف، استخدم هذه الوسيطة لتحديد اسم الملف. من المتوقع أن يتطابق الاسم مع اسم استدعاء "require".
warn_realization => BOOLEAN
يطبع رسالة تحذير عند بدء التحقيق. هذا لأغراض التصحيح.
warn_realize_again => BOOLEAN
عند تحقيق كائن، يُعاد بناء الكائن الأصلي - الذي كان يعمل كبديل - ليعمل كوكيل للكائن الحقيقي. سيصدر هذا الخيار تحذيرًا عند استخدام هذا الوكيل، مما يعني أن هناك متغيرًا في مكان ما في برنامجك لا يزال يحتفظ بمرجع إلى البديل. هذا الأخير ليس مشكلة، رغم أنه قد يبطئ كل استدعاء طريقة.

الإضافات إلى فئتك

$obj->AUTOLOAD()
عند استدعاء طريقة غير متاحة في الكائن البطيء، يتم استدعاء AUTOLOAD.
$obj->can($method)
Object::Realize::Later->can($method)
هل الطريقة المحددة $method متاحة في النسخة البطيئة أو في النسخة المحققة من هذا الكائن؟ سيعيد المرجع إلى الكود.

مثال:

MyLazyObject->can('lazyWork')      # true
MyLazyObject->can('realWork')      # true
my $lazy = MyLazyObject->new;
$lazy->can('lazyWork');            # true
$lazy->can('realWork');            # true
$obj->forceRealize()
يمكنك فرض التحميل باستدعاء هذه الطريقة على كائنك. يعيد الكائن المحقق.
Object::Realize::Later->isa($class)
هل هذا الكائن فئة فرعية من الفئة المحددة ($class) أو يمكن أن يكون فئة فرعية منها؟

مثال:

MyLazyObject->isa('MyRealObject')      # true
MyLazyObject->isa('SuperClassOfLazy'); # true
MyLazyObject->isa('SuperClassOfReal'); # true
my $lazy = MyLazyObject->new;
$lazy->isa('MyRealObject');            # true
$lazy->isa('SuperClassOfLazy');        # true
$lazy->isa('SuperClassOfReal');        # true
$obj->willRealize()
يعيد الفئة التي ستكون التالية في التحقيق.

البنى الداخلية لـ Object::Realize::Later

الطرق التالية لا يتم تصديرها إلى الفئة التي تم فيها تنفيذ "use". هذه الطرق تنفذ التحقيق الفعلي.

Object::Realize::Later->import(%options)
يتم استخدام %options لـ "import" وهي القيم بعد اسم الفئة مع "use". تنفذ هذه الروتينية تحليل الخيارات الفعلي. تولد الكود ديناميكيًا، والذي يتم تقييمه في مساحة أسماء المستدعي.
Object::Realize::Later->realizationOf( $object, [$realized] )
يعيد النسخة المحققة من $object، اختياريًا بعد تعيينها. عندما تعيد الطريقة "undef"، فإن التحقيق لم يحدث بعد أو أن الكائن المحقق قد تم حذفه بالفعل.
Object::Realize::Later->realize(%options)
يتم استدعاء هذه الطريقة عند استدعاء "$object-"forceRealize()>. تتحقق مما إذا كان التحقيق قد تم بالفعل (وفي هذه الحالة يتم إرجاع الكائن المحقق).

التفاصيل

حول التحميل عند الطلب

هناك طريقتان لتنفيذ سلوك التحميل عند الطلب: يمكنك اختيار التحقق مما إذا كنت قد حققت البيانات في كل طريقة تصل إلى البيانات، أو استخدام حيلة التحميل التلقائي للبيانات.

تنفيذ الحل الأول هو:

sub realize {
    my $self = shift;
    return $self unless $self->{_is_realized};
    # قراءة البيانات من الملف، أو ما شابه
    $self->{data} = ....;
    $self->{_is_realized} = 1;
    $self;
}
sub getData() {
    my $self = shift;
    return $self->realize->{data};
}

التنفيذ أعلاه محفوف بالمخاطر، حيث يمكنك بسهولة نسيان استدعاء realize(). لا يمكن للاختبارات تغطية جميع ترتيبات استدعاءات الطرق لاكتشاف الأخطاء.

يستخدم النهج الثاني التحميل التلقائي، وهو مدعوم من هذه الحزمة. أولاً ننشئ كائنًا بديلاً، سيتم تحويله إلى كائن محقق لاحقًا. يتم تفعيل هذا التحويل عبر AUTOLOAD.

قد يحتوي هذا الكائن البديل على بعض طرق النسخة المحققة من الكائن، لتقليل الحاجة إلى التحقيق. سيحتوي أيضًا على بعض المعلومات اللازمة لإنشاء الكائن الحقيقي.

تحل الفئة "Object::Realize::Later" مشاكل الوراثة (خاصة الطرق isa() و can()) وتوفر طريقة AUTOLOAD. طرق الفئة التي لا

هذا الموقع غير مرتبط بـ Arch Linux ولا تابع له بأي شكل من الأشكال. المحتوى هو ترجمة فقط لصفحات دليل Arch المنشورة على man.archlinux.org.

t المحددة في كائن stub يتم تمريرها كطرق صنف دون تحقيق.

المصائد

كن على دراية بالمصائد الخطيرة في التنفيذ الحالي. تظهر هذه المشاكل عند وجود مراجع متعددة لنفس الكائن المؤجل. اعتمادًا على كيفية تنفيذ التحقيق، يمكن أن تحدث أشياء فظيعة.

نسختا التحقيق:

  • عن طريق إعادة البركة (reblessing)

    هذه هي النسخة الآمنة. الكائن المحقق هو نفس الكائن المؤجل، لكنه أعيدت بركته في حزمة مختلفة. عندما توجد مراجع متعددة للكائن المؤجل، سيتم تحديثها جميعًا في الوقت نفسه، لأن معلومات البركة مخزنة داخل المتغير المُشار إليه.

  • عن طريق نسخة جديدة

    هذه هي أجمل طريقة للتحقيق، لكنها أيضًا أكثر خطورة بكثير. تأمل هذا:

    package Delayed;
use Object::Realize::Later
     becomes => 'Realized',
     realize => 'load';
sub new($)      {my($class,$v)=@_; bless {label=>$v}, $class}
sub setLabel($) {my $self = shift; $self->{label} = shift}
sub load()      {$_[0] = Realized->new($_[0]->{label}) }
package Realized;  # file Realized.pm or use use(source_module)
sub new($)      {my($class,$v)=@_; bless {label=>$v}, $class}
sub setLabel($) {my $self = shift; $self->{label} = shift}
sub getLabel()  {my $self = shift; $self->{label}}
package main;
my $original = Delayed->new('original');
my $copy     = $original;
print $original->getLabel;     # prints 'original'
print ref $original;           # prints 'Realized'
print ref $copy;               # prints 'Delayed'
$original->setLabel('changed');
print $original->getLabel;     # prints 'changed'
print $copy->getLabel;         # prints 'original'

أمثلة

المثال 1

في المثال الأول، نقوم بتحميل رسالة بشكل مؤجل. في لحظة تعريف الرسالة، نأخذ الموقع فقط. عندما يتم أخذ بيانات الرسالة (الترويسة أو الجسم)، يتم تحميل البيانات تلقائيًا.

package Mail::Message::Delayed;
use Object::Realize::Later
  ( becomes => 'Mail::Message::Real'
  , realize => 'loadMessage'
  );
sub new($) {
    my ($class, $file) = @_;
    bless { filename => $file }, $class;
}
sub loadMessage() {
    my $self = shift;
    Mail::Message::Real->new($self->{filename});
}

في البرنامج الرئيسي:

package main;
use Mail::Message::Delayed;
my $msg    = Mail::Message::Delayed->new('/home/user/mh/1');
$msg->body->print;     # this will trigger autoload.

المثال 2

قد يتم تحقيقك أيضًا عن طريق إعادة البركة. في هذه الحالة لتغيير نوع كائنك إلى نوع مختلف يخزن نفس المعلومات. هل هذا صحيح؟ هل أنت متأكد؟ للحالات البسيطة، قد يكون هذا ممكنًا:

package Alive;
use Object::Realize::Later
     becomes => 'Dead',
     realize => 'kill';
sub new()         {my $class = shift; bless {@_}, $class}
sub jump()        {print "Jump!\n"}
sub showAntlers() {print "Fight!\n"}
sub kill()        {bless(shift, 'Dead')}
package Dead;
sub takeAntlers() {...}

في البرنامج الرئيسي:

my $deer   = Alive->new(Animal => 'deer');
my $trophy = $deer->takeAntlers();

في هذه الحالة، لا يتم تغيير الكائن (المرجع) بل يتم إعادة بركته. لا يوجد خطر من الاحتفاظ بالنسخة غير المحققة من الكائن في مكان ما: جميع المتغيرات التي تعرف عن هذا الغزال المعين ترى التغيير.

المثال 3

هذه الوحدة مفيدة بشكل خاص للمشاريع الأكبر، حيث تكون هناك حاجة للسرعة أو تقليل الذاكرة. في هذه الحالة، قد يكون لديك نظرة عامة إضافية على الكائنات التي تم تحقيقها (تحويلها)، وأيها لم يتم. هذا المثال مأخوذ من وحدات MailBox:

تحاول وحدة Mail::Box تعزيز وقت الوصول إلى مجلد. إذا كنت تحتاج فقط إلى رسائل اليوم الأخير، فلماذا يجب قراءتها جميعًا؟ لذا، ينشئ MailBox جردًا للرسائل في البداية فقط. يأخذ ترويسات جميع الرسائل، لكنه يترك جسم (محتوى) الرسالة في الملف.

في حالة MailBox، يكون لكائن Mail::Message الخيار بين عدد من Mail::Message::Body، أحدها تم إعداده فقط لقراءة الجسم عند الحاجة. مقتطف كود:

package Mail::Message;
sub new($$)
{   my ($class, $head, $body) = @_;
    my $self = bless {head => $head, body => $body}, $class;
    $body->message($self);          # tell body about the message
}
sub head()     { shift->{head} }
sub body()     { shift->{body} }
sub loadBody()
{   my $self = shift;
    my $body = $self->body;
    # Catch re-invocations of the loading.  If anywhere was still
    # a reference to the old (unrealized) body of this message, we
    # return the new-one directly.
    return $body unless $body->can('forceRealize');
    # Load the body (change it to anything which really is of
    # the promised type, or a sub-class of it.
    my ($lines, $size) = .......;    # get the data
    $self->{body} = Mail::Message::Body::Lines
                         ->new($lines, $size, $self);
    # Return the realized object.
    return $self->{body};
}
package Mail::Message::Body::Lines;
use base 'Mail::Message::Body';
sub new($$$)
{   my ($class, $lines, $size, $message) = @_;
    bless { lines => $lines, size => $size
          , message => $message }, $class;
}
sub size()    { shift->{size} }
sub lines()   { shift->{lines} }
sub message() { shift->{message);
package Mail::Message::Body::Delayed;
use Object::Realize::Later
    becomes => 'Mail::Message::Body',
    realize => sub {shift->message->loadBody};
sub new($)
{   my ($class, $size) = @_;
    bless {size => $size}, $class;
}
sub size() { shift->{size} }
sub message(;$)
{   my $self = shift;
    @_ ? ($self->{message} = shift) : $self->{messages};
}
package main;
use Mail::Message;
use Mail::Message::Body::Delayed;
my $body    = Mail::Message::Body::Delayed->new(42);
my $message = Mail::Message->new($head, $body);
print $message->size;         # will not trigger realization!
print $message->can('lines'); # true, but no realization yet.
print $message->lines;        # realizes automatically.
