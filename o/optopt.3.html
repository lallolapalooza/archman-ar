الاسم

getopt, getopt_long, getopt_long_only, optarg, optind, opterr, optopt - تحليل خيارات سطر الأوامر

المكتبة

مكتبة C القياسية (libc، -lc)

الملخص

#include <unistd.h>
int getopt(int argc, char *argv[],
           const char *optstring);
extern char *optarg;
extern int optind, opterr, optopt;
#include <getopt.h>
int getopt_long(int argc, char *argv[],
           const char *optstring,
           const struct option *longopts, int *longindex);
int getopt_long_only(int argc, char *argv[],
           const char *optstring,
           const struct option *longopts, int *longindex);

الوصف

تقوم الدالة getopt() بتحليل وسائط سطر الأوامر. وسائطها argc و argv هما عدد الوسائط والمصفوفة كما تم تمريرهما إلى الدالة main() عند استدعاء البرنامج. العنصر الذي يبدأ بـ '-' (وليس بالضبط "-" أو "--") هو عنصر خيار. أحرف هذا العنصر (بخلاف '-' الأولي) هي أحرف خيارات. إذا تم استدعاء getopt() بشكل متكرر، فإنها تعيد بالتتابع كل حرف خيار من كل عنصر خيار.

المتغير optind هو فهرس العنصر التالي الذي سيتم معالجته في argv. يقوم النظام بتهيئة هذه القيمة إلى 1. يمكن للمستدعي إعادة تعيينها إلى 1 لإعادة بدء مسح argv نفسه، أو عند مسح متجه وسائط جديد.

إذا وجدت getopt() حرف خيار آخر، فإنها تعيده، وتحديث المتغير الخارجي optind والمتغير الثابت nextchar بحيث يمكن للاستدعاء التالي لـ getopt() استئناف المسح بالحرف التالي للخيار أو عنصر argv.

إذا لم تكن هناك أحرف خيارات أخرى، فإن getopt() تعيد -1. عندئذٍ يكون optind هو الفهرس في argv لأول عنصر argv ليس خيارًا.

optstring هو سلسلة تحتوي على أحرف الخيارات المشروعة. حرف الخيار المشروع هو أي حرف واحد مرئي بايت من ascii(7) (الذي تعيد الدالة isgraph(3) قيمة غير صفرية له) وليس '-' أو ':' أو ';'. إذا تبع هذا الحرف نقطتان رأسيتان، فإن الخيار يتطلب وسيطًا، لذا تضع getopt() مؤشرًا إلى النص التالي في نفس عنصر argv، أو نص عنصر argv التالي، في optarg. نقطتان رأسيتان تعنيان أن الخيار يأخذ وسيطًا اختياريًا؛ إذا كان هناك نص في عنصر argv الحالي (أي، في نفس كلمة اسم الخيار نفسه، على سبيل المثال، "-oarg")، فسيتم إعادته في optarg، وإلا فسيتم تعيين optarg إلى صفر. هذا امتداد GNU. إذا احتوت optstring على W تبعها فاصلة منقوطة، فسيتم التعامل مع -W foo على أنه الخيار الطويل --foo. (الخيار -W محجوز من قبل POSIX.2 للامتدادات الخاصة بالتنفيذ.) هذا السلوك هو امتداد GNU، وهو غير متاح مع المكتبات قبل glibc 2.

افتراضيًا، تقوم getopt() بتبديل محتويات argv أثناء مسحها، بحيث تكون جميع العناصر غير الخيارات في النهاية. يتم تنفيذ وضعين آخرين للمسح. إذا كان الحرف الأول من optstring هو '+' أو تم تعيين متغير البيئة POSIXLY_CORRECT، فإن معالجة الخيارات تتوقف فورًا عند مواجهة وسيط غير خيار. إذا لم يكن '+' هو الحرف الأول من optstring، فسيتم التعامل معه كخيار عادي. إذا كان سلوك POSIXLY_CORRECT مطلوبًا في هذه الحالة، فستحتوي optstring على رمزين '+'. إذا كان الحرف الأول من optstring هو '-'، فسيتم التعامل مع كل عنصر argv غير خيار كما لو كان وسيطًا لخيار له رمز خيار 1. (يستخدم هذا بواسطة البرامج التي تم تكوينها لتوقع الخيارات وعناصر argv الأخرى بأي ترتيب وتهتم بترتيب الاثنين.) الوسيط الخاص "--" يجبر على إنهاء مسح الخيارات بغض النظر عن وضع المسح.

أثناء معالجة قائمة الخيارات، يمكن لـ getopt() اكتشاف نوعين من الأخطاء: (1) حرف خيار لم يتم تحديده في optstring و (2) وسيط خيار مفقود (أي، خيار في نهاية سطر الأوامر بدون وسيط متوقع). يتم التعامل مع هذه الأخطاء والإبلاغ عنها على النحو التالي:

•
افتراضيًا، تطبع getopt() رسالة خطأ على الخطأ القياسي، وتضع حرف الخيار الخاطئ في optopt، وتعيد '؟' كناتج للدالة.
•
إذا قام المستدعي بتعيين المتغير العام opterr إلى صفر، فإن getopt() لا تطبع رسالة خطأ. يمكن للمستدعي تحديد حدوث خطأ عن طريق اختبار ما إذا كانت قيمة الإرجاع للدالة هي '؟'. (افتراضيًا، opterr لها قيمة غير صفرية.)
•
إذا كان الحرف الأول (بعد أي '+' أو '-' اختياري موصوف أعلاه) من optstring هو نقطتان رأسيتان (':')، فإن getopt() لا تطبع رسالة خطأ أيضًا. بالإضافة إلى ذلك، فإنها تعيد ':' بدلاً من '؟' للإشارة إلى وسيط خيار مفقود. هذا يسمح للمستدعي بالتمييز بين نوعي الأخطاء المختلفين.

getopt_long() و getopt_long_only()

تعمل الدالة getopt_long() مثل getopt() باستثناء أنها تقبل أيضًا خيارات طويلة، تبدأ بشرطتين. (إذا كان البرنامج يقبل فقط الخيارات الطويلة، فيجب تحديد optstring كسلسلة فارغة ("")، وليس NULL.) قد يتم اختصار أسماء الخيارات الطويلة إذا كان الاختصار فريدًا أو تطابقًا تامًا لبعض الخيارات المحددة. قد يأخذ الخيار الطويل معلمة، بصيغة --arg=param أو --arg param.

longopts هو مؤشر إلى العنصر الأول من مصفوفة من struct option معرفة في <getopt.h> بالشكل التالي:


struct option {
    const char *name;
    int         has_arg;
    int        *flag;
    int         val;
};

معاني الحقول المختلفة هي:

name
هو اسم الخيار الطويل.
has_arg
هو: no_argument (أو 0) إذا كان الخيار لا يأخذ وسيطًا؛ required_argument (أو 1) إذا كان الخيار يتطلب وسيطًا؛ أو optional_argument (أو 2) إذا كان الخيار يأخذ وسيطًا اختياريًا.
flag
يحدد كيفية إرجاع النتائج للخيار الطويل. إذا كان flag هو NULL، فإن getopt_long() تعيد val. (على سبيل المثال، قد يقوم البرنامج المستدعي بتعيين val إلى حرف الخيار القصير المكافئ.) بخلاف ذلك، تعيد getopt_long() 0، و flag يشير إلى متغير يتم تعيينه إلى val إذا تم العثور على الخيار، ولكنه يظل دون تغيير إذا لم يتم العثور على الخيار.
val
هي القيمة التي سيتم إرجاعها، أو تحميلها في المتغير المشار إليه بواسطة flag.

يجب ملء العنصر الأخير من المصفوفة بالأصفار.

إذا لم يكن longindex NULL، فإنه يشير إلى متغير يتم تعيينه إلى فهرس الخيار الطويل بالنسبة إلى longopts.

getopt_long_only() تشبه getopt_long()، ولكن يمكن استخدام '-' وكذلك "--" للإشارة إلى خيار طويل. إذا كان خيار يبدأ بـ '-' (ليس "--") لا يتطابق مع خيار طويل، ولكنه يتطابق مع خيار قصير، فسيتم تحليله كخيار قصير بدلاً من ذلك.

قيمة الإرجاع

إذا تم العثور على خيار بنجاح، فإن getopt() تعيد حرف الخيار. إذا تم تحليل جميع خيارات سطر الأوامر، فإن getopt() تعيد -1. إذا واجهت getopt() حرف خيار لم يكن موجودًا في optstring، فسيتم إرجاع '؟'. إذا واجهت getopt() خيارًا بوسيط مفقود، فسيتم إرجاع الـ

قيمة الإرجاع

تعتمد قيمة الإرجاع على الحرف الأول في optstring: إذا كان ':'، يتم إرجاع ':'؛ وإلا يتم إرجاع '؟'.

getopt_long() و getopt_long_only() ترجعان أيضًا حرف الخيار عند التعرف على خيار قصير. بالنسبة للخيار الطويل، فإنهما يرجعان val إذا كان flag هو NULL، وصفر بخلاف ذلك. أخطاء وإرجاع -1 هي نفسها لـ getopt()، بالإضافة إلى '؟' لمطابقة غامضة أو معلمة إضافية.

بيئة التشغيل

POSIXLY_CORRECT
إذا تم تعيين هذا المتغير، يتوقف معالجة الخيارات فورًا عند مواجهة وسيط غير خياري.

الإصدارات

ينص POSIX على أن الوسيط argv يجب أن يكون const، لكن هذه الدوال تعيد ترتيب عناصره ما لم يتم تعيين متغير البيئة POSIXLY_CORRECT. يتم استخدام const في النموذج الفعلي ليكون متوافقًا مع الأنظمة الأخرى؛ ومع ذلك، لا تعرض هذه الصفحة المعدل، لتجنب إرباك القراء.

التاريخ

POSIX.1-2001, و POSIX.2.

في بعض التطبيقات القديمة، تم تعريف getopt() في <stdio.h>. سمح SUSv1 بتعريف يظهر إما في <unistd.h> أو <stdio.h>. وضع POSIX.1-1996 استخدام <stdio.h> لهذا الغرض كـ LEGACY. لا يتطلب POSIX.1-2001 ظهور التعريف في <stdio.h>.

تأثرت الإصدارات القديمة جدًا من glibc بمتغير البيئة _PID_GNU_nonoption_argv_flags_.

ملاحظات

البرنامج الذي يقوم بفحص متجهات وسائط متعددة، أو يعيد فحص نفس المتجه أكثر من مرة، ويريد الاستفادة من امتدادات GNU مثل '+' و '-' في بداية optstring، أو يغير قيمة POSIXLY_CORRECT بين الفحوصات، يجب عليه إعادة تهيئة getopt() عن طريق إعادة تعيين optind إلى 0، بدلاً من القيمة التقليدية 1. (إعادة التعيين إلى 0 يجبر استدعاء روتين تهيئة داخلي يعيد التحقق من POSIXLY_CORRECT ويتحقق من امتدادات GNU في optstring.)

يتم تحليل وسائط سطر الأوامر بترتيب صارم مما يعني أن الخيار الذي يتطلب وسيطًا سيستهلك الوسيط التالي، بغض النظر عما إذا كان هذا الوسيط هو وسيط الخيار المحدد بشكل صحيح أو مجرد الخيار التالي (في سيناريو يسيء المستخدم فيه تحديد سطر الأوامر). على سبيل المثال، إذا تم تحديد optstring كـ "1n:" وقام المستخدم بتحديد وسائط سطر الأوامر بشكل غير صحيح كـ prog -n -1، سيتم إعطاء الخيار -n القيمة optarg "-1"، وسيتم اعتبار الخيار -1 على أنه لم يتم تحديده.

أمثلة

getopt()

يقوم مثال البرنامج البسيط التالي باستخدام getopt() لمعالجة خيارين للبرنامج: -n، بدون قيمة مرتبطة؛ و -t val، الذي يتوقع قيمة مرتبطة.

#include <stdio.h>
#include <stdlib.h>
#include <unistd.h>
int
main(int argc, char *argv[])
{
    int flags, opt;
    int nsecs, tfnd;
    nsecs = 0;
    tfnd = 0;
    flags = 0;
    while ((opt = getopt(argc, argv, "nt:")) != -1) {
        switch (opt) {
        case 'n':
            flags = 1;
            break;
        case 't':
            nsecs = atoi(optarg);
            tfnd = 1;
            break;
        default: /* '?' */
            fprintf(stderr, "Usage: %s [-t nsecs] [-n] name\n",
                    argv[0]);
            exit(EXIT_FAILURE);
        }
    }
    printf("flags=%d; tfnd=%d; nsecs=%d; optind=%d\n",
           flags, tfnd, nsecs, optind);
    if (optind >= argc) {
        fprintf(stderr, "Expected argument after options\n");
        exit(EXIT_FAILURE);
    }
    printf("name argument = %s\n", argv[optind]);
    /* Other code omitted */
    exit(EXIT_SUCCESS);
}

getopt_long()

يوضح مثال البرنامج التالي استخدام getopt_long() مع معظم ميزاته.

#include <getopt.h>
#include <stdio.h>     /* for printf */
#include <stdlib.h>    /* for exit */
int
main(int argc, char *argv[])
{
    int c;
    int digit_optind = 0;
    while (1) {
        int this_option_optind = optind ? optind : 1;
        int option_index = 0;
        static struct option long_options[] = {
            {"add",     required_argument, 0,  0 },
            {"append",  no_argument,       0,  0 },
            {"delete",  required_argument, 0,  0 },
            {"verbose", no_argument,       0,  0 },
            {"create",  required_argument, 0, 'c'},
            {"file",    required_argument, 0,  0 },
            {0,         0,                 0,  0 }
        };
        c = getopt_long(argc, argv, "abc:d:012",
                        long_options, &option_index);
        if (c == -1)
            break;
        switch (c) {
        case 0:
            printf("option %s", long_options[option_index].name);
            if (optarg)
                printf(" with arg %s", optarg);
            printf("\n");
            break;
        case '0':
        case '1':
        case '2':
            if (digit_optind != 0 && digit_optind != this_option_optind)
              printf("digits occur in two different argv-elements.\n");
            digit_optind = this_option_optind;
            printf("option %c\n", c);
            break;
        case 'a':
            printf("option a\n");
            break;
        case 'b':
            printf("option b\n");
            break;
        case 'c':
            printf("option c with value '%s'\n", optarg);
            break;
        case 'd':
            printf("option d with value '%s'\n", optarg);
            break;
        case '?':
            break;
        default:
            printf("?? getopt returned character code 0%o ??\n", c);
        }
    }
    if (optind < argc) {
        printf("non-option ARGV-elements: ");
        while (optind < argc)
            printf("%s ", argv[optind++]);
        printf("\n");
    }
    exit(EXIT_SUCCESS);
}
