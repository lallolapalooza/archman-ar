OSSL_STORE_SEARCH(3ssl)

الحزمة: openssl

التراخيص: Apache-2.0

OSSL_STORE_SEARCH(3ssl) OpenSSL OSSL_STORE_SEARCH(3ssl)

الاسم

OSSL_STORE_SEARCH, OSSL_STORE_SEARCH_by_name, OSSL_STORE_SEARCH_by_issuer_serial, OSSL_STORE_SEARCH_by_key_fingerprint, OSSL_STORE_SEARCH_by_alias, OSSL_STORE_SEARCH_free, OSSL_STORE_SEARCH_get_type, OSSL_STORE_SEARCH_get0_name, OSSL_STORE_SEARCH_get0_serial, OSSL_STORE_SEARCH_get0_bytes, OSSL_STORE_SEARCH_get0_string, OSSL_STORE_SEARCH_get0_digest - نوع ووظائف لإنشاء معايير بحث OSSL_STORE

الملخص

#include <openssl/store.h>
typedef struct ossl_store_search_st OSSL_STORE_SEARCH;
OSSL_STORE_SEARCH *OSSL_STORE_SEARCH_by_name(X509_NAME *name);
OSSL_STORE_SEARCH *OSSL_STORE_SEARCH_by_issuer_serial(X509_NAME *name,
                                                      const ASN1_INTEGER
                                                      *serial);
OSSL_STORE_SEARCH *OSSL_STORE_SEARCH_by_key_fingerprint(const EVP_MD *digest,
                                                        const unsigned char
                                                        *bytes, int len);
OSSL_STORE_SEARCH *OSSL_STORE_SEARCH_by_alias(const char *alias);
void OSSL_STORE_SEARCH_free(OSSL_STORE_SEARCH *search);
int OSSL_STORE_SEARCH_get_type(const OSSL_STORE_SEARCH *criterion);
X509_NAME *OSSL_STORE_SEARCH_get0_name(OSSL_STORE_SEARCH *criterion);
const ASN1_INTEGER *OSSL_STORE_SEARCH_get0_serial(const OSSL_STORE_SEARCH
                                                  *criterion);
const unsigned char *OSSL_STORE_SEARCH_get0_bytes(const OSSL_STORE_SEARCH
                                                  *criterion, size_t *length);
const char *OSSL_STORE_SEARCH_get0_string(const OSSL_STORE_SEARCH *criterion);
const EVP_MD *OSSL_STORE_SEARCH_get0_digest(const OSSL_STORE_SEARCH
                                            *criterion);

الوصف

تُستخدم هذه الوظائف لتحديد معايير البحث للمساعدة في البحث عن كائنات محددة عبر أسماء أخرى غير URI الذي يتم توفيره إلى OSSL_STORE_open(). على سبيل المثال، يمكن أن يكون هذا مفيدًا لتطبيق تلقى URI ويرغب بعد ذلك في إضافة معايير بحث بطريقة موحدة ومدعومة.

الأنواع

OSSL_STORE_SEARCH هو نوع غامض يحتفظ بمعيار البحث المُنشأ، ويمكن تقديمه إلى سياق OSSL_STORE باستخدام OSSL_STORE_find().

تطبيق المتصل يمتلك تخصيص OSSL_STORE_SEARCH في جميع الأوقات، ويجب أن يكون حذرًا من إلغاء تخصيصه قبل استدعاء OSSL_STORE_close() لسياق OSSL_STORE الذي تم تقديمه إليه.

وظائف التطبيق

OSSL_STORE_SEARCH_by_name()، OSSL_STORE_SEARCH_by_issuer_serial()، OSSL_STORE_SEARCH_by_key_fingerprint()، و OSSL_STORE_SEARCH_by_alias() تُستخدم لإنشاء OSSL_STORE_SEARCH من اسم موضوع، أو زوج اسم مُصدر ورقم تسلسلي، أو بصمة مفتاح، أو اسم مستعار (على سبيل المثال، اسم ودود). لا يتم نسخ المعلمات المقدمة، بل يتم الإشارة إليها فقط في معيار، لذا يجب أن يكون لها على الأقل نفس العمر الافتراضي لمعيار OSSL_STORE_SEARCH الذي تم إنشاؤه.

OSSL_STORE_SEARCH_free() تُستخدم لتحرير OSSL_STORE_SEARCH. إذا كان الوسيط NULL، فلن يتم فعل شيء.

وظائف المحمل

OSSL_STORE_SEARCH_get_type() تُرجع نوع المعيار للمُعطى OSSL_STORE_SEARCH.

OSSL_STORE_SEARCH_get0_name()، OSSL_STORE_SEARCH_get0_serial()، OSSL_STORE_SEARCH_get0_bytes()، OSSL_STORE_SEARCH_get0_string()، و OSSL_STORE_SEARCH_get0_digest() تُستخدم لاسترداد بيانات مختلفة من OSSL_STORE_SEARCH، كما هو متاح لكل نوع. لمزيد من المعلومات، انظر "أنواع المعايير المدعومة" أدناه.

أنواع المعايير المدعومة

تشمل أنواع المعايير المدعومة حاليًا:

OSSL_STORE_SEARCH_BY_NAME
يدعم هذا المعيار البحث بالمطابقة الدقيقة لاسم الموضوع. اسم الموضوع نفسه هو مؤشر X509_NAME. يتم إنشاء معيار من هذا النوع باستخدام OSSL_STORE_SEARCH_by_name()، ويتم استرداد اسم الموضوع الفعلي باستخدام OSSL_STORE_SEARCH_get0_name().
OSSL_STORE_SEARCH_BY_ISSUER_SERIAL
يدعم هذا المعيار البحث بالمطابقة الدقيقة لكل من اسم المُصدر والرقم التسلسلي. اسم المُصدر نفسه هو مؤشر X509_NAME، والرقم التسلسلي هو مؤشر ASN1_INTEGER. يتم إنشاء معيار من هذا النوع باستخدام OSSL_STORE_SEARCH_by_issuer_serial() ويتم استرداد اسم المُصدر الفعلي والرقم التسلسلي باستخدام OSSL_STORE_SEARCH_get0_name() و OSSL_STORE_SEARCH_get0_serial().
OSSL_STORE_SEARCH_BY_KEY_FINGERPRINT
يدعم هذا المعيار البحث بالمطابقة الدقيقة لبصمة المفتاح. بصمة المفتاح نفسها هي سلسلة من البايتات وطولها، بالإضافة إلى الخوارزمية التي تم استخدامها لحساب البصمة. يمكن ترك خوارزمية التجزئة (digest) غير محددة (NULL)، وفي هذه الحالة، يتعين على المحمل تحديد خوارزمية تجزئة افتراضية ومقارنة البصمات وفقًا لذلك. يتم إنشاء معيار من هذا النوع باستخدام OSSL_STORE_SEARCH_by_key_fingerprint() ويمكن استرداد البصمة الفعلية وطولها باستخدام OSSL_STORE_SEARCH_get0_bytes(). يمكن استرداد خوارزمية التجزئة باستخدام OSSL_STORE_SEARCH_get0_digest().
OSSL_STORE_SEARCH_BY_ALIAS
يدعم هذا المعيار البحث بمطابقة اسم مستعار من نوع ما. الاسم المستعار نفسه هو سلسلة C بسيطة. يتم إنشاء معيار من هذا النوع باستخدام OSSL_STORE_SEARCH_by_alias() ويتم استرداد الاسم المستعار الفعلي باستخدام OSSL_STORE_SEARCH_get0_string().

قيم الإرجاع

OSSL_STORE_SEARCH_by_name()، OSSL_STORE_SEARCH_by_issuer_serial()، OSSL_STORE_SEARCH_by_key_fingerprint()، و OSSL_STORE_SEARCH_by_alias() تُرجع مؤشر OSSL_STORE_SEARCH عند النجاح، أو NULL عند الفشل.

OSSL_STORE_SEARCH_get_type() تُرجع نوع المعيار للمُعطى OSSL_STORE_SEARCH. لا توجد قيمة خطأ.

OSSL_STORE_SEARCH_get0_name() تُرجع مؤشر X509_NAME عند النجاح، أو NULL عندما يكون OSSL_STORE_SEARCH المُعطى من نوع مختلف.

OSSL_STORE_SEARCH_get0_serial() تُرجع مؤشر ASN1_INTEGER عند النجاح، أو NULL عندما يكون OSSL_STORE_SEARCH المُعطى من نوع مختلف.

OSSL_STORE_SEARCH_get0_bytes() تُرجع مؤشر const unsigned char وتُعيّن *length لطول السلسلة عند النجاح، أو NULL عندما يكون OSSL_STORE_SEARCH المُعطى من نوع مختلف.

OSSL_STORE_SEARCH_get0_string() تُرجع مؤشر const char عند النجاح، أو NULL عندما يكون OSSL_STORE_SEARCH المُعطى من نوع مختلف.

OSSL_STORE_SEARCH_get0_digest() تُرجع مؤشر const EVP_MD. NULL هي قيمة صالحة وتعني أنه سيتم استخدام افتراضي محمل المتجر عند الاقتضاء.
