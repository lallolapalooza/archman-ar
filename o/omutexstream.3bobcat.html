omutexstream.3bobcat

الحزمة: libbobcat

التراخيص: GPL

FBB::OMutexStream(3bobcat) مجرى إخراج محمي بقفل FBB::OMutexStream(3bobcat)

الاسم

FBB::OMutexStream - يوفر مرافق مجرى إخراج std::ostream محمي بقفل

الملخص

#include <bobcat/omutexstream>
خيار الربط: -lbobcat

الوصف

في البرامج متعددة الخيوط، قد يتم تداخل الإخراج الذي تكتبه خيوط مختلفة إلى نفس مجرى الإخراج بسبب تبديل السياقات. يتم منع ذلك باستخدام الأقفال: قبل الكتابة إلى مجرى الإخراج، يتم الحصول على قفل، ويتم تحرير القفل بمجرد اكتمال الإخراج.

OMutexStream يدعم الكتابة إلى مجاري الإخراج مع معالجة أقفال القفل. يتم تهيئة كائن OMutexStream نفسه بكائن std::ostream لتلقي المعلومات المكتوبة إلى كائن OMutexStream. هذا الكائن، مثل std::cout و std::err، عادة ما يتم تعريفه ككائن يمكن الوصول إليه بشكل عام. عند إدراج المعلومات في كائن OMutexStream، يعيد OMutexStream أولاً كائن OMutexStream::Out، يقوم المُنشئ الخاص به بقفل قفل OMutexStream::Out. ينتهي عمر كائن OMutexStream::Out في نهاية عبارة الإدراج، وفي ذلك الوقت يقوم المُدمر الخاص به بتحرير القفل.

في كثير من الحالات يكون هذا مناسبًا. خاصة عندما تنتهي العبارات بأسطر جديدة (’\n’ أو endl) ينتج عن هذا إخراج نظيف. في بعض الحالات، لا تعمل هذه الطريقة. ضع في اعتبارك الموقف الذي يتم فيه إنتاج الإخراج عن طريق سلسلة من التكرارات في عبارة for. بالنسبة لهذه الحالات، يوفر OMutexStream العضو ostream الذي يعيد كائن OMutexStream::Out. نظرًا لأن هذا الكائن يقفل القفل أيضًا، يظل القفل نشطًا أثناء وجوده. أثناء وجوده، يمكن تنفيذ عبارات إدراج OMutexStream منفصلة. على سبيل المثال، سيكمل جزء التعليمات البرمجية التالي جميع المخرجات، مما يجبر الخيوط المتنافسة على الانتظار:

    void fun()
    {
        OMutexStream mout{ std::cout }; // إنشاء كائن OMutexStream
        {
            auto out{ mout.ostream() }  // يقفل القفل (القفل #1)
            mout << "Hello ";           // يقفل أيضًا (القفل #2، عند ؛
                                        //             العودة إلى القفل #1)
            out << "world\n";
        }                               // 'out' يحرر القفل
    }
كن حذرًا عند تقييد عمر الكائن الذي تم إرجاعه بواسطة OMutexStream::ostream().

(عبر OMutexStream::Out) std::ostream

المُنشئات

o
OMutexStream(std::ostream &out):
يتم تهيئة كائن OMutexStream بكائن std::ostream.

مُنشئات النسخ والنقل (وعوامل التعيين) متاحة.

المعاملات المُحمّلة

o
OMutexStream::Out operator<<(OMutexStream const &mstr, Tp &&tp):
هذا العضو هو قالب دالة. يتم تمرير مرجع إعادة التوجيه الخاص به إلى كائن OMutexStream::Out الذي تم إنشاؤه بواسطة mstr. نظرًا لأن مُنشئات OMutexStream::Out تقفل قفل الفئة، وبما أن OMutexStream::Out ورثت من std::ostream، فيمكن أيضًا استخدام جميع عمليات الإدراج والعمليات الأخرى المسموح بها لـ std::ostream لكائنات OMutexStream::Out.
o
OMutexStream::Out operator<<(OMutexStream const &mstr, Ret &(*manip)(Ret &os)):
هذا العضو أيضًا هو قالب دالة. يستخدم لإدراج المُعدّلات بدون وسائط في كائنات OMutexStream::Out.

دوال الأعضاء

o
OMutexStream::Out ostream() const:
يتم إرجاع كائن OMutexStream::Out الذي قام بقفل قفله، وسيحافظ على القفل أثناء وجوده. يمكن أيضًا استخدام جميع وظائف فئة std::ostream لكائن OMutexStream::Out الذي تم إرجاعه بواسطة ostream.
كن حذرًا عند تقييد عمر الكائن الذي تم إرجاعه بواسطة OMutexStream::ostream() لتجنب أقفال القفل الطويلة غير الضرورية.

مثال

#include <iostream>
#include <thread>
#include <chrono>
#include <cstdlib>
#include <cmath>
#include <ctime>
#include <bobcat/omutexstream>
using namespace std;
using namespace FBB;
OMutexStream mout(cout);
void run(int nr)
{
    for (size_t idx = 0; idx != 3; ++idx)
    {
        mout << "hello world 1 from thread " << nr << ": " <<
                log(rand()) << endl;
        this_thread::sleep_for(
                chrono::milliseconds(200 + rand() % 800));
       mout << "hello world 2 from thread " << nr << ": " <<
                log(rand()) << ’\n’;
        this_thread::sleep_for(
                chrono::milliseconds(200 + rand() % 800));
    }
    auto out{ mout.ostream() };
    cout << nr << ": " << out.tellp() << ’\n’;
}
int main()
{
    srand(time(0));
    thread t1(run, 1);
    thread t2(run, 2);
    thread t3(run, 3);
    thread t4(run, 4);
    t1.join();
    t2.join();
    t3.join();
    t4.join();
}
