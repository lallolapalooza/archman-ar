OSSL_PARAM(3ssl)

الحزمة: openssl

التراخيص: Apache-2.0

OSSL_PARAM(3ssl) OpenSSL OSSL_PARAM(3ssl)

الاسم

OSSL_PARAM — هيكل لتمرير أو طلب معلمات الكائن

الملخص

#include <openssl/core.h>
typedef struct ossl_param_st OSSL_PARAM;
struct ossl_param_st {
    const char *key;             /* اسم المعلمة */
    unsigned int data_type;      /* يشير إلى نوع المحتوى الموجود في البيانات */
    void *data;                  /* القيمة التي يتم تمريرها أو استلامها */
    size_t data_size;            /* حجم البيانات */
    size_t return_size;          /* الحجم المُرجع */
};

الوصف

OSSL_PARAM هو نوع يسمح بتمرير بيانات عشوائية بين طرفين لا يملكان، أو لديهما القليل جداً من المعرفة المتبادلة حول الهياكل الداخلية للكائنات المعنية.

كمثال نموذجي للاستخدام، قد يكون هناك برنامج يريد إعداد بعض المعلمات لكائن معين، أو يريد العثور على بعض المعلمات لكائن معين.

يمكن استخدام مصفوفات من هذا النوع للأغراض التالية:

  • إعداد معلمات لكائن ما

    يقوم المستدعي بإعداد مصفوفة OSSL_PARAM ويستدعي دالة (دالة الضبط) لديها معرفة عميقة بالكائن، والتي يمكنها أخذ البيانات من مصفوفة OSSL_PARAM وتعيينها بالشكل المناسب للهيكل الداخلي للكائن.

  • طلب معلمات لكائن ما

    يقوم المستدعي (طالب الخدمة) بإعداد مصفوفة OSSL_PARAM ويستدعي دالة (دالة الاستجابة) لديها معرفة عميقة بالكائن، والتي يمكنها أخذ البيانات الداخلية للكائن ونسخها (ربما تحويلها) إلى الذاكرة التي أعدها طالب الخدمة والمشار إليها بواسطة OSSL_PARAM data.

  • طلب أوصاف المعلمات

    يحصل المستدعي على مصفوفة من الثوابت OSSL_PARAM تصف المعلمات المتاحة وبعض خصائصها؛ الاسم، نوع البيانات، والحجم المتوقع. لوصف تفصيلي لكل حقل لهذا الاستخدام، انظر أوصاف الحقول أدناه.

    يمكن للمستدعي بعد ذلك استخدام المعلومات من مصفوفة الوصف هذه لبناء مصفوفة OSSL_PARAM الخاصة به لتمريرها إلى دالة ضبط أو استجابة.

عادةً، لا يكون ترتيب العناصر في مصفوفة OSSL_PARAM ذا صلة. ومع ذلك، إذا كانت دالة الاستجابة قادرة على التعامل مع عناصر متعددة بنفس المفتاح، فيجب التعامل معها بالترتيب الذي تظهر به.

يجب أن تحتوي مصفوفة OSSL_PARAM على عنصر نهائي حيث يكون key هو NULL. القالب النهائي الكامل المعتاد هو:

{ NULL, 0, NULL, 0, 0 }

يمكن تحديد ذلك أيضًا باستخدام OSSL_PARAM_END(3).

الدعم الوظيفي

توفر مكتبة libcrypto مجموعة محدودة من الدوال المساعدة للتعامل مع OSSL_PARAM ومصفوفاتها، يرجى الاطلاع على OSSL_PARAM_get_int(3). يمكن للمطورين توسيع هذه الدوال أو استبدالها حسب رغبتهم.

حقول OSSL_PARAM

key
معرف المعلمة كسلسلة نصية.

في مصفوفة OSSL_PARAM، يعتبر العنصر الذي يحتوي على هذا الحقل NULL بمثابة عنصر نهائي.

data_type
data_type هو قيمة تصف نوع وتنظيم البيانات. انظر قسم "الأنواع المدعومة" أدناه لوصف الأنواع.
data
data_size
data هو مؤشر إلى الذاكرة حيث توجد بيانات المعلمة (عند ضبط المعلمات) أو سيتم تخزينها (عند طلب المعلمات)، و data_size هو حجمها بالبايت. يعتمد تنظيم البيانات على نوع المعلمة والـ flag.

يجب إيلاء اهتمام خاص للنوع OSSL_PARAM_UTF8_STRING فيما يتعلق بسلاسل C. عند ضبط المعلمات، يجب أن يكون الحجم هو طول السلسلة، باستثناء بايت NULL. عند طلب المعلمات، يجب أن يكون الحجم هو حجم المخزن المؤقت الذي سيتم ملؤه، والذي يجب أن يحتوي على مساحة لبايت NULL.

عند طلب المعلمات، يمكن أن يكون data مساوياً لـ NULL. يمكن استخدام هذا من قبل طالب الخدمة لتحديد مقدار مساحة المخزن المؤقت المطلوبة ديناميكيًا لتخزين بيانات المعلمة. في هذه الحالة، يتم تجاهل data_size.

عند استخدام OSSL_PARAM كواصف للمعامِلات، يجب تجاهل data. إذا كان data_size صفراً، فهذا يعني أن أي حجم معقول مقبول، وإلا فإنه يحدد الحد الأقصى للحجم المسموح به.

return_size
عند استخدام مصفوفة OSSL_PARAM لطلب البيانات، يجب على المستجيب تعيين هذا الحقل للإشارة إلى حجم بيانات المعلمة، بما في ذلك الحشو إذا كان ذلك منطبقًا. في حالة عدم كون data_size حجمًا كافيًا للبيانات، يجب على المستجيب مع ذلك تعيين هذا الحقل للإشارة إلى الحد الأدنى للحجم المطلوب. (ملاحظات إضافية في "ملاحظات" أدناه).

عند استخدام OSSL_PARAM كواصف للمعلمات، يجب تجاهل return_size.

ملاحظة:

يتم تحديد أسماء المفاتيح والأنواع المرتبطة بها بواسطة الكيان الذي يوفر هذه المعلمات، أي أن أسماء المعلمات المقدمة من مكتبات OpenSSL تحددها هذه المكتبات، وأسماء المعلمات المقدمة من الموفرين تحددها الموفرون، باستثناء نماذج المؤشرات (انظر أوصاف الأنواع أدناه). الكيانات التي ترغب في ضبط أو طلب المعلمات يجب أن تعرف ما هي هذه المفاتيح وما هو نوعها، ويجب أن تكون أي وظيفة بين هذين الكيانين محصنة ويجب عليها فقط تمرير مصفوفة OSSL_PARAM.

الأنواع المدعومة

يمكن أن يكون الحقل data_type أحد الأنواع التالية:

OSSL_PARAM_INTEGER
OSSL_PARAM_UNSIGNED_INTEGER
بيانات المعلمة هي عدد صحيح (صحيح أو صحيح غير سالب) بطول عشوائي، منظم بالشكل الأصلي، أي أن البايت الأكثر أهمية أولاً في أنظمة Endian الكبيرة، والبايت الأقل أهمية أولاً في أنظمة Endian الصغيرة.
OSSL_PARAM_REAL
بيانات المعلمة هي قيمة نقطة عائمة بالشكل الأصلي.
OSSL_PARAM_UTF8_STRING
بيانات المعلمة هي سلسلة نصية قابلة للطباعة.
OSSL_PARAM_OCTET_STRING
بيانات المعلمة هي سلسلة عشوائية من البايتات.
OSSL_PARAM_UTF8_PTR
بيانات المعلمة هي مؤشر إلى سلسلة نصية قابلة للطباعة.

الفرق بين هذا و OSSL_PARAM_UTF8_STRING هو أن data لا يشير مباشرة إلى البيانات، بل إلى مؤشر يشير إلى البيانات.

إذا كان هناك أي شك حول ما يجب استخدامه، فإن OSSL_PARAM_UTF8_STRING هو دائمًا الخيار الصحيح تقريبًا.

يتم استخدامه للإشارة إلى أن البيانات الثابتة سيتم تمريرها أو تمريرها، وبالتالي لا يلزم نسخ البيانات التي يتم تمريرها، فقط المؤشر إليها.

يجب تعيين data_size إلى حجم البيانات، وليس إلى حجم المؤشر إلى البيانات. إذا تم استخدامه في طلب معلمة، فإن data_size غير ذي صلة. ومع ذلك، سيقوم المستجيب بتعيين return_size إلى حجم البيانات.

ملاحظة: استخدام هذا النوع **هش** ولا يمكن استخدامه بأمان إلا للبيانات التي تظل ثابتة وفي موقع ثابت لفترة كافية (مثل دورة حياة الكيان الذي يقدم هذه المعلمات).

OSSL_PARAM_OCTET_PTR
بيانات المعلمة هي مؤشر إلى سلسلة عشوائية من البايتات.

الفرق بين هذا و OSSL_PARAM_OCTET_STRING هو أن data يشير إلى مجموعة من البايتات العشوائية، بينما يشير OSSL_PARAM_UTF8_PTR إلى مؤشر يشير إلى البيانات.

إذا كان هناك أي شك حول ما يجب استخدامه، فإن OSSL_PARAM_UTF8_STRING هو دائمًا الخيار الصحيح تقريبًا.

يتم استخدامه للإشارة إلى أن البيانات الثابتة سيتم تمريرها أو تمريرها، وبالتالي لا يلزم نسخ البيانات التي يتم تمريرها، فقط المؤشر إليها.

يجب تعيين data_size إلى حجم البيانات، وليس إلى حجم المؤشر إلى البيانات. إذا تم استخدامه في طلب معلمة، فإن data_size غير ذي صلة. ومع ذلك، سيقوم المستجيب بتعيين return_size إلى حجم البيانات.

ملاحظة: استخدام هذا النوع **هش** ولا يمكن استخدامه بأمان إلا للبيانات التي تظل ثابتة وفي موقع ثابت لفترة كافية (مثل دورة حياة الكيان الذي يقدم هذه المعلمات).

ملاحظة:

يتم تحديد أسماء المفاتيح والأنواع المرتبطة بها بواسطة الكيان الذي يوفر هذه المعلمات، أي أن أسماء المعلمات المقدمة من مكتبات OpenSSL تحددها هذه المكتبات، وأسماء المعلمات المقدمة من الموفرين تحددها الموفرون، باستثناء نماذج المؤشرات (انظر أوصاف الأنواع أدناه). الكيانات التي ترغب في ضبط أو طلب المعلمات يجب أن تعرف ما هي هذه المفاتيح وما هو نوعها، ويجب أن تكون أي وظيفة بين هذين الكيانين محصنة ويجب عليها فقط تمرير مصفوفة OSSL_PARAM.

هذا الموقع غير مرتبط بـ Arch Linux ولا تابع له بأي شكل من الأشكال. المحتوى هو ترجمة فقط لصفحات دليل Arch المنشورة على man.archlinux.org.

الاسم

L_PARAM_OCTET_STRING

هل data لا تشير مباشرة إلى البيانات، ولكن إلى مؤشر يشير إلى البيانات.

إذا كان هناك أي شك حول ما يجب استخدامه، فإن OSSL_PARAM_OCTET_STRING هو دائمًا الاختيار الصحيح تقريبًا.

يتم استخدامه للإشارة إلى أن البيانات الثابتة سيتم تمريرها أو يتم تمريرها، وبالتالي لا يلزم نسخ البيانات التي سيتم تمريرها، فقط المؤشر إليها.

يجب تعيين data_size إلى طول البيانات، وليس طول المؤشر إلى البيانات. إذا تم استخدامه في طلب معلمات، فإن data_size غير ذي صلة. ومع ذلك، سيقوم responder بتعيين return_size إلى طول البيانات.

ملاحظة: استخدام هذا النوع ضعيف ولا يمكن استخدامه إلا بأمان للبيانات التي تظل ثابتة وفي موقع ثابت لفترة كافية (مثل دورة حياة الكيان الذي يقدم هذه المعلمات).

ملاحظات

عند تعيين المعلمات وطلبها، يجب على الوظائف المسماة أن تقرر ما هو الخطأ وما ليس خطأ. السلوك الموصى به هو:

  • يجب ببساطة تجاهل المفاتيح التي لا يتعرف عليها setter أو responder. هذا ليس خطأ بحد ذاته.
  • إذا كانت المفاتيح التي يتعرف عليها setter تشكل مجموعة بيانات متسقة، فيجب أن تخرج هذه المكالمة.
  • بخلاف return_size، لا يجب أن يغير responder حقول OSSL_PARAM أبدًا. لإرجاع قيمة، يجب عليه تغيير محتويات الذاكرة التي يشير إليها data.
  • إذا كان نوع البيانات المرتبط بمفتاح غير صحيح، فيمكن للوظيفة المسماة إرجاع خطأ.

    يمكن للوظيفة المسماة أيضًا محاولة تحويل البيانات إلى تنسيق مناسب (على سبيل المثال، من المعقول تمرير رقم كبير كـ OSSL_PARAM_OCTET_STRING، لذا حتى لو تم تعريف مفتاح كـ OSSL_PARAM_UNSIGNED_INTEGER، يمكن تمرير القيمة كـ OSSL_PARAM_OCTET_STRING

  • إذا كان data لـ OSSL_PARAM_OCTET_STRING أو OSSL_PARAM_UTF8_STRING هو NULL، فيجب على responder تعيين return_size إلى طول العنصر المراد إرجاعه وإرجاع النجاح. بعد ذلك، سيتم استدعاء responder مرة أخرى مع data يشير إلى موقع القيمة المراد وضعها.

  • إذا وجد responder أن بعض أبعاد البيانات صغيرة جدًا بالنسبة للبيانات المطلوبة، فيجب عليه تعيين return_size لكل OSSL_PARAM من هذا القبيل إلى الحد الأدنى للطول المطلوب وفي النهاية إرجاع خطأ.
  • بالنسبة لمعلمات نوع العدد الصحيح (OSSL_PARAM_UNSIGNED_INTEGER و OSSL_PARAM_INTEGER)، يمكن لـ responder اختيار إرجاع خطأ إذا لم يكن data_size حجمًا مناسبًا (حتى لو كان data_size أكبر من اللازم). إذا وجد responder الحجم المناسب، فيجب عليه ملء جميع بايتات data_size والتأكد من الحشو الصحيح لترتيب البايت الأصلي، وتعيين return_size إلى نفس القيمة مثل data_size.

أمثلة

زوج من الأمثلة لإظهار كيفية تكوين مصفوفات OSSL_PARAM.

مثال 1

هذا المثال لتعيين معلمات لكائن ما:

#include <openssl/core.h>
const char *foo = "some string";
size_t foo_l = strlen(foo);
const char bar[] = "some other string";
OSSL_PARAM set[] = {
    { "foo", OSSL_PARAM_UTF8_PTR, &foo, foo_l, 0 },
    { "bar", OSSL_PARAM_UTF8_STRING, (void *)&bar, sizeof(bar) - 1, 0 },
    { NULL, 0, NULL, 0, 0 }
};

مثال 2

هذا المثال لطلب معلمات من كائن ما:

const char *foo = NULL;
size_t foo_l;
char bar[1024];
size_t bar_l;
OSSL_PARAM request[] = {
    { "foo", OSSL_PARAM_UTF8_PTR, &foo, 0 /*irrelevant*/, 0 },
    { "bar", OSSL_PARAM_UTF8_STRING, &bar, sizeof(bar), 0 },
    { NULL, 0, NULL, 0, 0 }
};

يمكن لـ responder الذي يتلقى هذه المصفوفة (مثل params في هذا المثال) ملء المعلمات على النحو التالي:

/* OSSL_PARAM *params */
int i;
for (i = 0; params[i].key != NULL; i++) {
    if (strcmp(params[i].key, "foo") == 0) {
        *(char **)params[i].data = "foo value";
        params[i].return_size = 9; /* length of "foo value" */
    } else if (strcmp(params[i].key, "bar") == 0) {
        memcpy(params[i].data, "bar value", 10);
        params[i].return_size = 9; /* length of "bar value" */
    }
    /* Ignore what is not known */
}
