#include <openssl/decoder.h>

int OSSL_DECODER_is_a(const OSSL_DECODER *decoder, const char *name);
const char *OSSL_DECODER_get0_name(const OSSL_DECODER *decoder);
const char *OSSL_DECODER_get0_description(const OSSL_DECODER *decoder);

الوصف

OSSL_DECODER هي طريقة لأدوات فك التشفير، والتي تعرف كيفية فك تشفير البيانات المشفرة إلى كائن من نوع ما تعرف بقية OpenSSL كيفية التعامل معه.

OSSL_DECODER_fetch() تبحث عن خوارزمية ضمن الموفر الذي تم تحميله في OSSL_LIB_CTX المعطى بواسطة ctx، والتي تحمل الاسم المعطى بواسطة name والخصائص المعطاة بواسطة properties. يحدد name نوع الكائن الذي يُتوقع أن تكون طريقة فك التشفير المسترجعة قادرة على فك تشفيره، وتستخدم الخصائص لتحديد نوع الإخراج المتوقع. بالنسبة للخصائص المعروفة والقيم التي قد تمتلكها، يرجى الاطلاع على "الأسماء والخصائص" في provider-encoder(7).

OSSL_DECODER_up_ref() تزيد عدد مرات الإشارة للكائن decoder المعطى.

OSSL_DECODER_free() تقلل عدد مرات الإشارة للكائن decoder المعطى، وعندما يصل العدد إلى الصفر، تقوم بتحريره. إذا كان الوسيط NULL، فلا يتم فعل شيء.

OSSL_DECODER_get0_provider() تعيد الموفر للكائن decoder المعطى.

OSSL_DECODER_get0_properties() تعيد تعريف الخاصية المرتبط بالكائن decoder المعطى.

OSSL_DECODER_is_a() تتحقق مما إذا كان decoder تطبيقًا لخوارزمية يمكن تحديد هويتها بالاسم name.

OSSL_DECODER_get0_name() تعيد الاسم المستخدم لاسترجاع الكائن decoder المعطى.

OSSL_DECODER_get0_description() تعيد وصفًا للكائن decoder، مخصص للعرض والاستهلاك البشري. الوصف يعود لتقدير تطبيق decoder.

OSSL_DECODER_names_do_all() تتنقل عبر جميع الأسماء للكائن decoder المعطى، وتستدعي fn مع كل اسم و data كوسائط.

OSSL_DECODER_do_all_provided() تتنقل عبر جميع تطبيقات فك التشفير من قبل جميع الموفرين النشطين في سياق المكتبة libctx، ولكل تطبيق، تستدعي fn مع طريقة التطبيق و arg كوسائط.

OSSL_DECODER_gettable_params() تعيد مصفوفة OSSL_PARAM(3) من واصفات المعلمات.

OSSL_DECODER_get_params() تحاول الحصول على المعلمات المحددة بمصفوفة OSSL_PARAM(3) params. يجب تجاهل المعلمات التي لا تتعرف عليها التطبيق.

قيم الإرجاع

OSSL_DECODER_fetch() تعيد مؤشرًا إلى كائن OSSL_DECODER، أو NULL في حالة حدوث خطأ.

OSSL_DECODER_up_ref() تعيد 1 عند النجاح، أو 0 عند حدوث خطأ.

OSSL_DECODER_free() لا تعيد أي قيمة.

OSSL_DECODER_get0_provider() تعيد مؤشرًا إلى كائن موفر، أو NULL في حالة حدوث خطأ.

OSSL_DECODER_get0_properties() تعيد مؤشرًا إلى سلسلة تعريف الخاصية، أو NULL في حالة حدوث خطأ.

OSSL_DECODER_is_a() تعيد 1 إذا كان decoder قابلاً للتحديد، وإلا 0.

OSSL_DECODER_get0_name() تعيد اسم الخوارزمية من التطبيق المقدم للكائن decoder المعطى. لاحظ أن decoder قد يكون له مرادفات متعددة مرتبطة به. في هذه الحالة، يتم إرجاع الاسم الأول من تعريف الخوارزمية. يتم الاحتفاظ بملكية السلسلة المعادة بواسطة كائن decoder ولا يجب تحريرها بواسطة المتصل.

OSSL_DECODER_get0_description() تعيد مؤشرًا إلى وصف، أو NULL إذا لم يكن هناك وصف.

OSSL_DECODER_names_do_all() تعيد 1 إذا تم استدعاء الدالة الإرجاعية لجميع الأسماء. القيمة 0 تعني أن الدالة الإرجاعية لم يتم استدعاؤها لأي أسماء.

ملاحظات

OSSL_DECODER_fetch() قد يتم استدعاؤها ضمنيًا بواسطة وظائف استرجاع أخرى، باستخدام نفس سياق المكتبة والخصائص. أي واجهة برمجة تطبيقات أخرى تستخدم مفاتيح ستقوم بذلك عادة.

أمثلة

لسرد جميع أدوات فك التشفير في موفر إلى bio_out:

static void collect_decoders(OSSL_DECODER *decoder, void *stack)
{
    STACK_OF(OSSL_DECODER) *decoder_stack = stack;
    sk_OSSL_DECODER_push(decoder_stack, decoder);
    OSSL_DECODER_up_ref(decoder);
}
void print_name(const char *name, void *vdata)
{
    BIO *bio = vdata;
    BIO_printf(bio, "%s ", name);
}
STACK_OF(OSSL_DECODER) *decoders;
int i;
decoders = sk_OSSL_DECODER_new_null();
BIO_printf(bio_out, "DECODERs provided by %s:\n", provider);
OSSL_DECODER_do_all_provided(NULL, collect_decoders,
                             decoders);
for (i = 0; i < sk_OSSL_DECODER_num(decoders); i++) {
    OSSL_DECODER *decoder = sk_OSSL_DECODER_value(decoders, i);
    if (strcmp(OSSL_PROVIDER_get0_name(OSSL_DECODER_get0_provider(decoder)),
               provider) != 0)
        continue;
    if (OSSL_DECODER_names_do_all(decoder, print_name, bio_out))
           BIO_printf(bio_out, "\n");
}
sk_OSSL_DECODER_pop_free(decoders, OSSL_DECODER_free);

NAME

OSSL_DECODER - A high-level API for decoding arbitrary data

SYNOPSIS

This is a conceptual overview. Actual usage requires including appropriate headers and linking to the OpenSSL library.

#include <openssl/decoder.h>

OSSL_DECODER *OSSL_DECODER_CTX_new(void);
int OSL_DECODER_CTX_free(OSSL_DECODER *ctx);
int OSL_DECODER_CTX_set_input(OSSL_DECODER *ctx, const OSSL_PARAM *params);
int OSL_DECODER_CTX_set_output_type(OSSL_DECODER *ctx, const char *name, const char *version);
int OSL_DECODER_CTX_decode(OSSL_DECODER *ctx, void **out, const OSSL_PARAM *params);

DESCRIPTION

The OSSL_DECODER functions provide a high-level interface for decoding arbitrary data. This is useful for applications that need to parse and interpret data in various formats, such as configuration files, certificates, keys, and more. The API is designed to be flexible and extensible, allowing new decoding algorithms and formats to be added easily.

The core of the API revolves around the OSSL_DECODER context, which holds the state and configuration for the decoding process. You typically create a decoder context using OSSL_DECODER_CTX_new(), configure it with input data and desired output types, and then perform the decoding operation.

OSSL_DECODER_CTX_set_input() is used to provide the encoded data to the decoder. The input is usually provided as an array of OSSL_PARAM structures, which can describe the source and format of the data.

OSSL_DECODER_CTX_set_output_type() specifies the desired format of the decoded output. This allows the application to request a specific data structure or type.

OSSL_DECODER_CTX_decode() performs the actual decoding operation. It takes the decoder context and an array of parameters, and returns the decoded data in the out parameter. The caller is responsible for freeing the memory allocated for the decoded data.

The OSSL_PARAM structure is a key component used for passing parameters and data between the application and the OpenSSL library. It's a general-purpose structure for representing key-value pairs.

RETURN VALUES

Most of the OSSL_DECODER functions return 1 on success and 0 or -1 on failure, depending on the specific function. Error details can typically be retrieved using OpenSSL's error handling functions, such as ERR_get_error().

NOTES

This API is part of the OpenSSL 3.x effort to provide a more modern and flexible interface. It's designed to be used in conjunction with other OpenSSL components, such as providers, which implement the actual decoding logic for various data formats.

Memory management is crucial when using this API. The caller is responsible for freeing any memory allocated by the OpenSSL library for the decoded output.

Error handling should be implemented carefully by checking return values and using OpenSSL's error reporting mechanisms.

EXAMPLES

This is a simplified conceptual example. Actual implementation would involve more detailed setup and error handling.

#include <openssl/decoder.h>
#include <openssl/evp.h>
#include <openssl/pem.h>
#include <openssl/err.h>

// ...

OSSL_DECODER *decoder = NULL;
void *decoded_data = NULL;
OSSL_PARAM params[2];
OSSL_DECODER_CTX *ctx = NULL;

// Initialize decoder context
ctx = OSSL_DECODER_CTX_new();
if (ctx == NULL) {
    // Handle error
    goto err;
}

// Set input data (e.g., a PEM encoded private key string)
const char *pem_data = "-----BEGIN PRIVATE KEY-----\n...\n-----END PRIVATE KEY-----\n";
params[0] = OSSL_PARAM_construct_const_byte_ ("data", pem_data, strlen(pem_data));
params[1] = OSSL_PARAM_construct_end();

if (OSSL_DECODER_CTX_set_input(ctx, params) == 0) {
    // Handle error
    goto err;
}

// Set desired output type (e.g., EVP_PKEY)
if (OSSL_DECODER_CTX_set_output_type(ctx, "EVPPKEY", NULL) == 0) {
    // Handle error
    goto err;
}

// Perform decoding
OSSL_PARAM out_params[2];
out_params[0] = OSSL_PARAM_construct_ptr_("out", &decoded_data);
out_params[1] = OSSL_PARAM_construct_end();

if (OSSL_DECODER_CTX_decode(ctx, NULL, out_params) == 0) {
    // Handle error
    goto err;
}

// decoded_data now points to the decoded data (e.g., an EVP_PKEY object)
// Use decoded_data...

// Clean up
if (decoded_data) {
    // Free decoded_data appropriately based on its type
    // e.g., EVP_PKEY_free(decoded_data);
}
OSSL_DECODER_CTX_free(ctx);
return 0;

err:
    // Handle errors and clean up
    if (ctx) OSSL_DECODER_CTX_free(ctx);
    // Clean up decoded_data if allocated
    return 1;

الاسم

OSSL_DECODER - واجهة برمجة تطبيقات عالية المستوى لفك ترميز البيانات العشوائية

الملخص

هذا نظرة عامة مفاهيمية. الاستخدام الفعلي يتطلب تضمين رؤوس مناسبة والربط بمكتبة OpenSSL.

#include <openssl/decoder.h>

OSSL_DECODER *OSSL_DECODER_CTX_new(void);
int OSL_DECODER_CTX_free(OSSL_DECODER *ctx);
int OSL_DECODER_CTX_set_input(OSSL_DECODER *ctx, const OSSL_PARAM *params);
int OSL_DECODER_CTX_set_output_type(OSSL_DECODER *ctx, const char *name, const char *version);
int OSL_DECODER_CTX_decode(OSSL_DECODER *ctx, void **out, const OSSL_PARAM *params);

الوصف

توفر وظائف OSSL_DECODER واجهة عالية المستوى لفك ترميز البيانات العشوائية. هذا مفيد للتطبيقات التي تحتاج إلى تحليل وتفسير البيانات بتنسيقات مختلفة، مثل ملفات التكوين، الشهادات، المفاتيح، والمزيد. تم تصميم الواجهة لتكون مرنة وقابلة للتوسيع، مما يسمح بإضافة خوارزميات وتنسيقات ترميز جديدة بسهولة.

يدور جوهر الواجهة حول سياق OSSL_DECODER، الذي يحتفظ بالحالة والتكوين لعملية فك الترميز. عادةً ما تقوم بإنشاء سياق فك ترميز باستخدام OSSL_DECODER_CTX_new()، وتكوينه ببيانات الإدخال وأنواع الإخراج المرغوبة، ثم إجراء عملية فك الترميز.

يتم استخدام OSSL_DECODER_CTX_set_input() لتوفير البيانات المشفرة لبرنامج فك الترميز. يتم عادةً توفير الإدخال كمصفوفة من هياكل OSSL_PARAM، والتي يمكن أن تصف مصدر البيانات وتنسيقها.

OSSL_DECODER_CTX_set_output_type() يحدد التنسيق المطلوب للإخراج بعد فك الترميز. هذا يسمح للتطبيق بطلب بنية بيانات أو نوع معين.

OSSL_DECODER_CTX_decode() يقوم بعملية فك الترميز الفعلية. يأخذ سياق فك الترميز ومصفوفة من المعلمات، ويعيد البيانات بعد فك الترميز في المعلمة out. يكون المستدعي مسؤولاً عن تحرير الذاكرة المخصصة للبيانات بعد فك الترميز.

هيكل OSSL_PARAM هو مكون رئيسي يستخدم لتمرير المعلمات والبيانات بين التطبيق ومكتبة OpenSSL. إنه هيكل عام الغرض لتمثيل أزواج المفاتيح والقيم.

قيم الإرجاع

تعيد معظم وظائف OSSL_DECODER القيمة 1 عند النجاح و 0 أو -1 عند الفشل، اعتمادًا على الوظيفة المحددة. يمكن عادةً استرداد تفاصيل الخطأ باستخدام وظائف معالجة الأخطاء في OpenSSL، مثل ERR_get_error().

ملاحظات

هذه الواجهة البرمجية هي جزء من جهود OpenSSL 3.x لتوفير واجهة أكثر حداثة ومرونة. تم تصميمها للاستخدام بالاشتراك مع مكونات OpenSSL الأخرى، مثل المزودين (providers)، الذين ينفذون منطق فك الترميز الفعلي لتنسيقات البيانات المختلفة.

إدارة الذاكرة أمر بالغ الأهمية عند استخدام هذه الواجهة. يكون المستدعي مسؤولاً عن تحرير أي ذاكرة مخصصة من قبل مكتبة OpenSSL للإخراج بعد فك الترميز.

يجب تنفيذ معالجة الأخطاء بعناية عن طريق التحقق من قيم الإرجاع واستخدام آليات الإبلاغ عن الأخطاء في OpenSSL.

أمثلة

هذا مثال مفاهيمي مبسط. سيتضمن التنفيذ الفعلي إعدادًا أكثر تفصيلاً ومعالجة للأخطاء.

#include <openssl/decoder.h>
#include <openssl/evp.h>
#include <openssl/pem.h>
#include <openssl/err.h>

// ...

OSSL_DECODER *decoder = NULL;
void *decoded_data = NULL;
OSSL_PARAM params[2];
OSSL_DECODER_CTX *ctx = NULL;

// تهيئة سياق فك الترميز
ctx = OSSL_DECODER_CTX_new();
if (ctx == NULL) {
    // معالجة الخطأ
    goto err;
}

// تعيين بيانات الإدخال (على سبيل المثال، سلسلة مفتاح خاص مشفر بتنسيق PEM)
const char *pem_data = "-----BEGIN PRIVATE KEY-----\n...\n-----END PRIVATE KEY-----\n";
params[0] = OSSL_PARAM_construct_const_byte_ ("data", pem_data, strlen(pem_data));
params[1] = OSSL_PARAM_construct_end();

if (OSSL_DECODER_CTX_set_input(ctx, params) == 0) {
    // معالجة الخطأ
    goto err;
}

// تعيين نوع الإخراج المطلوب (على سبيل المثال، EVP_PKEY)
if (OSSL_DECODER_CTX_set_output_type(ctx, "EVPPKEY", NULL) == 0) {
    // معالجة الخطأ
    goto err;
}

// إجراء فك الترميز
OSSL_PARAM out_params[2];
out_params[0] = OSSL_PARAM_construct_ptr_("out", &decoded_data);
out_params[1] = OSSL_PARAM_construct_end();

if (OSSL_DECODER_CTX_decode(ctx, NULL, out_params) == 0) {
    // معالجة الخطأ
    goto err;
}

// decoded_data يشير الآن إلى البيانات بعد فك الترميز (على سبيل المثال، كائن EVP_PKEY)
// استخدم decoded_data...

// التنظيف
if (decoded_data) {
    // حرر decoded_data بشكل مناسب بناءً على نوعه
    // على سبيل المثال، EVP_PKEY_free(decoded_data);
}
OSSL_DECODER_CTX_free(ctx);
return 0;

err:
    // معالجة الأخطاء والتنظيف
    if (ctx) OSSL_DECODER_CTX_free(ctx);
    // تنظيف decoded_data إذا تم تخصيصه
    return 1;
