OSSL_SELF_TEST_new(3ssl)

الحزمة: openssl

التراخيص: Apache-2.0

OSSL_SELF_TEST_NEW(3ssl) OpenSSL OSSL_SELF_TEST_NEW(3ssl)

الاسم

OSSL_SELF_TEST_new, OSSL_SELF_TEST_free, OSSL_SELF_TEST_onbegin, OSSL_SELF_TEST_oncorrupt_byte, OSSL_SELF_TEST_onend - وظائف لتشغيل رد اتصال أثناء اختبار ذاتي

الملخص

#include <openssl/self_test.h>
OSSL_SELF_TEST *OSSL_SELF_TEST_new(OSSL_CALLBACK *cb, void *cbarg);
void OSSL_SELF_TEST_free(OSSL_SELF_TEST *st);
void OSSL_SELF_TEST_onbegin(OSSL_SELF_TEST *st, const char *type,
                            const char *desc);
int OSSL_SELF_TEST_oncorrupt_byte(OSSL_SELF_TEST *st, unsigned char *bytes);
void OSSL_SELF_TEST_onend(OSSL_SELF_TEST *st, int ret);

الوصف

هذه الوظائف مخصصة للاستخدام من قبل منفذي المزودين، لعرض معلومات تشخيصية أثناء الاختبار الذاتي.

OSSL_SELF_TEST_new() تقوم بتخصيص كائن OSSL_SELF_TEST غامض مرتبط برد اتصال ووسيط لرد الاتصال.

قد يتم تشغيل رد الاتصال cb عدة مرات بواسطة اختبار ذاتي للإشارة إلى مراحل مختلفة.

OSSL_SELF_TEST_free() تحرر المساحة التي خصصتها OSSL_SELF_TEST_new(). إذا كان الوسيط NULL، فلن يتم فعل شيء.

يمكن إدراج OSSL_SELF_TEST_onbegin() في بداية كتلة من كود الاختبار الذاتي. يمكن استخدامها لأغراض تشخيصية. إذا تم استدعاء هذه الوظيفة، فسيتلقى رد الاتصال cb كائن OSSL_PARAM(3) التالي.

"st-phase" (OSSL_PROV_PARAM_SELF_TEST_PHASE) <سلسلة UTF8>
القيمة هي السلسلة "Start"

يمكن إدراج OSSL_SELF_TEST_oncorrupt_byte() مباشرة بعد حساب الإجابة المعروفة، ولكن قبل أن يقارن الاختبار الذاتي النتيجة. سيتم إفساد البايت الأول في المصفوفة المقدمة bytes إذا أعاد رد الاتصال 0، وإلا فإنه يترك المصفوفة دون تغيير. يمكن استخدامه لاختبار الفشل. يمكن استخدام type و desc لتحديد اختبار ذاتي فردي لاستهداف اختبار الفشل. إذا تم استدعاء هذه الوظيفة، فسيتلقى رد الاتصال cb كائن OSSL_PARAM(3) التالي.

"st-phase" (OSSL_PROV_PARAM_SELF_TEST_PHASE) <سلسلة UTF8>
القيمة هي السلسلة "Corrupt"

يمكن إدراج OSSL_SELF_TEST_onend() في نهاية كتلة من كود الاختبار الذاتي قبل التنظيف مباشرة للإشارة إلى ما إذا كان الاختبار قد نجح أو فشل. يمكن استخدامه لأغراض تشخيصية. إذا تم استدعاء هذه الوظيفة، فسيتلقى رد الاتصال cb كائن OSSL_PARAM(3) التالي.

"st-phase" (OSSL_PROV_PARAM_SELF_TEST_PHASE) <سلسلة UTF8>
قيمة السلسلة هي "Pass" إذا كان ret غير صفري، وإلا فإن قيمته "Fail".

بعد استدعاء رد الاتصال cb، يتم تعيين القيم التي تم تعيينها بواسطة OSSL_SELF_TEST_onbegin() لـ type و desc إلى القيمة "None".

إذا تم استدعاء OSSL_SELF_TEST_onbegin() أو OSSL_SELF_TEST_oncorrupt_byte() أو OSSL_SELF_TEST_onend()، فسيتم تمرير OSSL_PARAM(3) الإضافية التالية إلى رد الاتصال.

"st-type" (OSSL_PROV_PARAM_SELF_TEST_TYPE) <سلسلة UTF8>
يتم تعيين القيمة بواسطة type الذي تم تمريره إلى OSSL_SELF_TEST_onbegin(). هذا يسمح لرد الاتصال بتحديد نوع الاختبار الذي يتم تشغيله.
"st-desc" (OSSL_PROV_PARAM_SELF_TEST_DESC) <سلسلة UTF8>
يتم تعيين القيمة بواسطة type الذي تم تمريره إلى OSSL_SELF_TEST_onbegin(). هذا يسمح لرد الاتصال بتحديد الفئة الفرعية للاختبار الذي يتم تشغيله.

قيم الإرجاع

OSSL_SELF_TEST_new() ترجع كائن OSSL_SELF_TEST المخصص، أو NULL إذا فشلت.

OSSL_SELF_TEST_oncorrupt_byte() ترجع 1 إذا حدث إفساد، وإلا فإنها ترجع 0.

أمثلة

يمكن إعداد اختبار ذاتي واحد بالطريقة التالية:

  OSSL_SELF_TEST *st = NULL;
  OSSL_CALLBACK *cb;
  void *cbarg;
  int ok = 0;
  unsigned char out[EVP_MAX_MD_SIZE];
  unsigned int out_len = 0;
  EVP_MD_CTX *ctx = EVP_MD_CTX_new();
  EVP_MD *md = EVP_MD_fetch(libctx, t->algorithm, NULL);
  /*
   * استرداد رد الاتصال - سيكون NULL إذا لم يتم تعيينه بواسطة التطبيق عبر
   * OSSL_SELF_TEST_set_callback().
   */
  OSSL_SELF_TEST_get_callback(libctx, &cb, &cbarg);
  st = OSSL_SELF_TEST_new(cb, cb_arg);
  /* تشغيل رد الاتصال الاختياري */
  OSSL_SELF_TEST_onbegin(st, OSSL_SELF_TEST_TYPE_KAT_DIGEST,
                         OSSL_SELF_TEST_DESC_MD_SHA2);
  if (!EVP_DigestInit_ex(ctx, md, NULL)
      || !EVP_DigestUpdate(ctx, pt, pt_len)
      || !EVP_DigestFinal(ctx, out, &out_len))
      goto err;
  /* إفساد اختياري - إذا أعاد رد اتصال التطبيق 0 */
  OSSL_SELF_TEST_oncorrupt_byte(st, out);
  if (out_len != t->expected_len
      || memcmp(out, t->expected, out_len) != 0)
      goto err;
  ok = 1;
err:
  OSSL_SELF_TEST_onend(st, ok);
  EVP_MD_free(md);
  EVP_MD_CTX_free(ctx);

يمكن إعداد اختبارات ذاتية متعددة بطريقة مماثلة عن طريق تكرار نمط OSSL_SELF_TEST_onbegin() و OSSL_SELF_TEST_oncorrupt_byte() و OSSL_SELF_TEST_onend() لكل اختبار.
