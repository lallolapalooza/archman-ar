type open_flag = open_flag =
| Open_rdonly (* فتح للقراءة.
*)
| Open_wronly (* فتح للكتابة.
*)
| Open_append (* فتح للإلحاق: الكتابة دائمًا في نهاية الملف.
*)
| Open_creat (* إنشاء الملف إذا لم يكن موجودًا.
*)
| Open_trunc (* تفريغ الملف إذا كان موجودًا بالفعل.
*)
| Open_excl (* فشل إذا كان Open_creat والملف موجودًا بالفعل.
*)
| Open_binary (* فتح في وضع ثنائي (لا تحويل).
*)
| Open_text (* فتح في وضع نصي (قد يقوم ببعض التحويلات).
*)
| Open_nonblock (* فتح في وضع غير حاظر.
*)

أوضاع الفتح لـ Out_channel.open_gen .

val open_bin : string -> t

فتح الملف المسمى للكتابة، وإرجاع قناة إخراج جديدة على هذا الملف، موضوعة في بداية الملف. يتم اقتطاع الملف إلى طول صفر إذا كان موجودًا بالفعل. يتم إنشاؤه إذا لم يكن موجودًا بالفعل.

val open_text : string -> t

مثل Out_channel.open_bin ، ولكن يتم فتح الملف في وضع نصي، بحيث تتم ترجمة الأسطر الجديدة أثناء عمليات الكتابة. على أنظمة التشغيل التي لا تميز بين الوضع النصي والوضع الثنائي، تعمل هذه الدالة مثل Out_channel.open_bin .

val open_gen : open_flag list -> int -> string -> t

open_gen mode perm filename يفتح الملف المسمى للكتابة، كما هو موضح أعلاه. تحدد الوسيطة الإضافية mode وضع الفتح. تحدد الوسيطة الإضافية perm أذونات الملف، في حالة الحاجة إلى إنشاء الملف. Out_channel.open_text و Out_channel.open_bin حالات خاصة لهذه الدالة.

val with_open_bin : string -> (t -> 'a) -> 'a

with_open_bin fn f يفتح قناة oc على الملف fn ويرجع f
oc . بعد عودة f، سواء بقيمة أو عن طريق إثارة استثناء، يتم ضمان إغلاق oc.

val with_open_text : string -> (t -> 'a) -> 'a

مثل Out_channel.with_open_bin ، ولكن يتم فتح القناة في الوضع النصي (انظر Out_channel.open_text ).

val with_open_gen : open_flag list -> int -> string -> (t -> 'a) -> 'a

مثل Out_channel.with_open_bin ، ولكن يمكن تحديد وضع الفتح وإذن الملف، في حالة الحاجة إلى إنشاء الملف (انظر Out_channel.open_gen ).

val close : t -> unit

إغلاق القناة المعطاة، مع تفريغ جميع عمليات الكتابة المخزنة مؤقتًا. تثير دوال الإخراج استثناء Sys_error عند تطبيقها على قناة إخراج مغلقة، باستثناء Out_channel.close و Out_channel.flush ، والتي لا تفعل شيئًا عند تطبيقها على قناة مغلقة بالفعل. لاحظ أن Out_channel.close قد تثير Sys_error إذا أبلغ نظام التشغيل عن خطأ عند التفريغ أو الإغلاق.

val close_noerr : t -> unit

نفس Out_channel.close ، ولكن تجاهل جميع الأخطاء.

الإخراج

val output_char : t -> char -> unit

كتابة الحرف على قناة الإخراج المعطاة.

val output_byte : t -> int -> unit

كتابة عدد صحيح واحد بحجم 8 بت (كحرف واحد بهذا الرمز) على قناة الإخراج المعطاة. يتم أخذ العدد الصحيح المعطى كباقي القسمة على 256.

val output_string : t -> string -> unit

كتابة السلسلة النصية على قناة الإخراج المعطاة.

val output_bytes : t -> bytes -> unit

كتابة تسلسل البايتات على قناة الإخراج المعطاة.

الإخراج المتقدم

val output : t -> bytes -> int -> int -> unit

output oc buf pos len يكتب len حرف من تسلسل البايتات buf ، بدءًا من الموضع pos ، إلى قناة الإخراج oc المعطاة.

يثير Invalid_argument إذا لم يحدد pos و len نطاقًا صالحًا لـ buf.

val output_substring : t -> string -> int -> int -> unit

نفس Out_channel.output ولكن تأخذ سلسلة نصية كوسيطة بدلاً من تسلسل بايتات.

val output_bigarray : t -> ('a, Bigarray.int8_unsigned_elt, Bigarray.c_layout) Bigarray.Array1.t -> int -> int -> unit

نفس Out_channel.output ولكن تأخذ البيانات من مصفوفة كبيرة.

التفريغ

val flush : t -> unit

تفريغ المخزن المؤقت المرتبط بقناة الإخراج المعطاة، وتنفيذ جميع عمليات الكتابة المعلقة على تلك القناة. يجب أن تكون البرامج التفاعلية حذرة بشأن تفريغ الإخراج القياسي وخطأ الإخراج القياسي في الوقت المناسب.

val flush_all : unit -> unit

تفريغ جميع قنوات الإخراج المفتوحة؛ تجاهل الأخطاء.

التحديد

val seek : t -> int64 -> unit

seek chan pos يحدد موضع الكتابة الحالي إلى pos للقناة chan. يعمل هذا فقط للملفات العادية. على الملفات من أنواع أخرى (مثل المحطات الطرفية، والقنوات، والمقابس)، يكون السلوك غير محدد.

val pos : t -> int64

إرجاع موضع الكتابة الحالي للقناة المعطاة. لا يعمل على القنوات المفتوحة بعلم Open_append (يرجع نتائج غير محددة).

بالنسبة للملفات المفتوحة في وضع نصي تحت ويندوز، يكون الموضع المرتجع تقريبيًا (بسبب تحويل نهاية الأسطر)؛ بشكل خاص، حفظ الموضع الحالي باستخدام Out_channel.pos، ثم العودة إلى هذا الموضع باستخدام Out_channel.seek لن يعمل. لكي تعمل هذه الصيغة البرمجية بشكل موثوق وقابل للنقل، يجب فتح الملف في وضع ثنائي.

السمات

val length : t -> int64

إرجاع حجم (عدد الأحرف) للملف العادي الذي تم فتح القناة عليه. إذا تم فتح القناة على ملف ليس ملفًا عاديًا، فإن النتيجة لا معنى لها.

val set_binary_mode : t -> bool -> unit

set_binary_mode oc true يضبط القناة oc على الوضع الثنائي: لا تحدث ترجمات أثناء الإخراج.

set_binary_mode oc false يضبط القناة oc على الوضع النصي: اعتمادًا على نظام التشغيل، قد تحدث بعض الترجمات أثناء الإخراج. على سبيل المثال، تحت ويندوز، سيتم ترجمة نهايات الأسطر من \n إلى \r\n.

ليس لهذه الدالة أي تأثير على أنظمة التشغيل التي لا تميز بين الوضع النصي والوضع الثنائي.

val is_binary_mode : t -> bool

is_binary_mode oc يرجع ما إذا كانت القناة oc في الوضع الثنائي (انظر Out_channel.set_binary_mode ).

Out_channel(3o) — OCamldoc

NAME

Out_channel - Output channel operations.

Module

module Out_channel : sig ... end

Documentation

This module provides access to output channels. An output channel is a destination for sequential output of data. Typically, this destination is a file, but it can also be a standard output or any other destination provided by the operating system.

The type t represents an output channel.

Output Channel Operations

val output_char : t -> char -> unit

output_char oc c writes the character c on the output channel oc.

val output_byte : t -> int -> unit

output_byte oc b writes the 8-bit integer b (0 to 255) on the output channel oc.

val output_string : t -> string -> unit

output_string oc s writes the string s on the output channel oc.

val output_bytes : t -> bytes -> unit

output_bytes oc b writes the byte sequence b on the output channel oc.

val output : t -> bytes -> int -> int -> unit

output oc s ofs len writes len bytes from the byte sequence s, starting at offset ofs, on the output channel oc.

val output_binary_string : t -> string -> unit

output_binary_string oc s writes the string s on the output channel oc. This operation is intended for binary output; the string s is not modified, and no character encoding is performed.

val write_line : t -> string -> unit

write_line oc s writes the string s followed by a newline character on the output channel oc.

val flush : t -> unit

flush oc forces the output buffer associated with oc to be written to the underlying output device.

val close : t -> unit

close oc closes the output channel oc, freeing any associated resources. Any subsequent operations on oc will raise an exception.

val out_channel_length : t -> int64

out_channel_length oc returns the number of characters written so far on the output channel oc.

val set_buffered : t -> bool -> unit

Set the buffering mode for the channel oc. If true, the channel is set to buffered mode. If false, it is set to unbuffered mode. In unbuffered mode, data output on oc will be sent to the output device immediately.

All channels are open in buffered mode by default.

val is_buffered : t -> bool

is_buffered oc returns whether the channel oc is buffered (see Out_channel.set_buffered ).

val isatty : t -> bool

isatty oc is true if oc refers to a terminal or console window, false otherwise.

Since 5.1

Examples

Writing the contents of a file:

      let write_file file s =
        Out_channel.with_open_bin file
          (fun oc -> Out_channel.output_string oc s))
