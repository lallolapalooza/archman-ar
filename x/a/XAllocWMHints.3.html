XAllocWMHints(3)

الحزمة: libx11

التراخيص: MIT AND X11

XAllocWMHints(3) دوال XLIB XAllocWMHints(3)

الاسم

XAllocWMHints, XSetWMHints, XGetWMHints, XWMHints - تخصيص بنية تلميحات مدير النوافذ وتعيين أو قراءة خاصية WM_HINTS لنافذة

الصيغة

XWMHints *XAllocWMHints(void);

int XSetWMHints(Display *display, Window w, XWMHints *wmhints);

XWMHints *XGetWMHints(Display *display, Window w);

الوسائط

display
يحدد الاتصال بخادم X.
w
يحدد النافذة.
wmhints
يحدد بنية XWMHints المراد استخدامها.

الوصف

تقوم الدالة XAllocWMHints بتخصيص وإرجاع مؤشر إلى بنية XWMHints. لاحظ أنه تم تعيين جميع الحقول في بنية XWMHints مبدئيًا إلى صفر. إذا لم تكن هناك ذاكرة كافية متاحة، فإن XAllocWMHints ترجع NULL. لتحرير الذاكرة المخصصة لهذه البنية، استخدم XFree.

تقوم الدالة XSetWMHints بتعيين تلميحات مدير النوافذ التي تتضمن معلومات وموقع الأيقونة، والحالة الأولية للنافذة، وما إذا كان التطبيق يعتمد على مدير النوافذ للحصول على مدخلات لوحة المفاتيح.

يمكن أن تولد XSetWMHints أخطاء BadAlloc و BadWindow.

تقوم الدالة XGetWMHints بقراءة تلميحات مدير النوافذ وترجع NULL إذا لم يتم تعيين خاصية WM_HINTS على النافذة أو ترجع مؤشرًا إلى بنية XWMHints إذا نجحت. عند الانتهاء من البيانات، قم بتحرير المساحة المستخدمة لها عن طريق استدعاء XFree.

يمكن أن تولد XGetWMHints خطأ BadWindow.

الخصائص

WM_HINTS
تلميحات إضافية يعينها العميل لاستخدامها من قبل مدير النوافذ. نوع C لهذه الخاصية هو XWMHints.

البنى

تحتوي بنية XWMHints على:

/* أقنعة تلميحات مدير النوافذ */

#define InputHint (1L << 0)
#define StateHint (1L << 1)
#define IconPixmapHint (1L << 2)
#define IconWindowHint (1L << 3)
#define IconPositionHint (1L << 4)
#define IconMaskHint (1L << 5)
#define WindowGroupHint (1L << 6)
#define XUrgencyHint (1L << 8)
#define AllHints (InputHint| StateHint| IconPixmapHint| IconWindowHint| IconPositionHint| IconMaskHint| WindowGroupHint) 
/* القيم */
typedef struct {
        long flags;     /* علامات تحدد الحقول المعرفة في هذه البنية */
        Bool input;     /* هل يعتمد هذا التطبيق على مدير النوافذ للحصول على
                        مدخلات لوحة المفاتيح؟ */
        int initial_state;      /* انظر أدناه */
        Pixmap icon_pixmap;     /* بيكسل ماب لاستخدامه كأيقونة */
        Window icon_window;     /* نافذة لاستخدامها كأيقونة */
        int icon_x, icon_y;     /* الموضع الأولي للأيقونة */
        Pixmap icon_mask;       /* بيكسل ماب لاستخدامه كقناع لـ icon_pixmap */
        XID window_group;       /* معرف لمجموعة النوافذ ذات الصلة */
        /* قد يتم توسيع هذه البنية في المستقبل */
} XWMHints;

يستخدم الحقل input لتوصيل نموذج التركيز المدخل الذي يستخدمه التطبيق إلى مدير النوافذ. يجب على التطبيقات التي تتوقع مدخلات ولكنها لا تعين أبدًا التركيز بشكل صريح لأي من نوافذها الفرعية (أي، تستخدم نموذج الدفع لإدارة التركيز)، مثل تطبيقات نمط X الإصدار 10 التي تستخدم التركيز المدفوع بالعقارات، تعيين هذا الحقل إلى True. وبالمثل، يجب على التطبيقات التي تعين تركيز المدخلات لنوافذها الفرعية فقط عندما يتم تمريره إلى نافذتها العلوية بواسطة مدير النوافذ أيضًا تعيين هذا الحقل إلى True. يجب على التطبيقات التي لا تتوقع أبدًا أي مدخلات لوحة مفاتيح تعيين هذا الحقل إلى False.

يجب على مديري النوافذ بنموذج السحب توفير إمكانية للتطبيقات بنموذج الدفع للحصول على مدخلات عن طريق تعيين تركيز المدخلات إلى النوافذ العلوية للتطبيقات التي يكون حقل المدخلات فيها True. يجب على مديري النوافذ بنموذج الدفع التأكد من أن تطبيقات نموذج السحب لا تكسرهم عن طريق إعادة تعيين تركيز المدخلات إلى PointerRoot عندما يكون ذلك مناسبًا (على سبيل المثال، كلما قام تطبيق يكون حقل المدخلات فيه False بتعيين تركيز المدخلات إلى أحد نوافذه الفرعية).

تعريفات علم initial_state هي:

#define WithdrawnState 0
#define NormalState 1 /* تبدأ معظم التطبيقات بهذه الطريقة */
#define IconicState 3 /* التطبيق يريد البدء كأيقونة */
يحدد icon_mask وحدات البكسل في icon_pixmap التي يجب استخدامها كأيقونة. هذا يسمح بأيقونات غير مستطيلة. يجب أن تكون كل من icon_pixmap و icon_mask خرائط بت. يسمح icon_window للتطبيق بتوفير نافذة لاستخدامها كأيقونة لمديري النوافذ الذين يدعمون هذا الاستخدام. يسمح لك window_group بتحديد أن هذه النافذة تنتمي إلى مجموعة من النوافذ الأخرى. على سبيل المثال، إذا كان تطبيق واحد يعالج نوافذ علوية متعددة، فإن هذا يسمح لك بتوفير معلومات كافية لمدير النوافذ لأيقنة جميع النوافذ بدلاً من نافذة واحدة فقط.

يشير العلم UrgencyHint، إذا تم تعيينه في حقل flags، إلى أن العميل يعتبر محتويات النافذة عاجلة، وتتطلب استجابة المستخدم في الوقت المناسب. سيبذل مدير النوافذ بعض الجهد لجذب انتباه المستخدم إلى هذه النافذة أثناء تعيين هذا العلم. يجب على العميل توفير وسيلة يمكن للمستخدم من خلالها جعل علم الاستعجال يزال (إما تخفيف الشرط الذي جعل النافذة عاجلة أو مجرد إيقاف الإنذار) أو سحب النافذة.

التشخيصات

BadAlloc
فشل الخادم في تخصيص المورد المطلوب أو ذاكرة الخادم.
BadWindow
قيمة لمُعامل Window لا تشير إلى نافذة معرفة.
