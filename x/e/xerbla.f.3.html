subroutine xerbla (character*(*) srname, integer info)

XERBLA

الغرض:

!>
!> هذه نسخة خاصة من XERBLA لاستخدامها فقط كجزء من برنامج الاختبار
!> لاختبار مخارج الخطأ من روتينات LAPACK.
!> يتم طباعة رسائل الخطأ إذا كانت INFO.NE.INFOT أو إذا كانت SRNAME.NE.SRNAMT،
!> حيث تكون INFOT و SRNAMT قيمًا مخزنة في COMMON.
!>

المعلمات

SRNAME 
!>          SRNAME هو CHARACTER*(*)
!>          اسم الروتين الفرعي الذي يستدعي XERBLA. يجب أن يتطابق هذا الاسم
!>          مع المتغير COMMON SRNAMT.
!>

INFO

!>          INFO هو INTEGER
!>          رمز العودة من الخطأ من الروتين الفرعي المستدعي. يجب أن يساوي INFO
!>          المتغير COMMON INFOT.
!>

!>
!>  يتم تمرير المتغيرات التالية عبر الكتل المشتركة INFOC و
!>  SRNAMC:
!>
!>  INFOT   INTEGER      رمز العودة المتوقع
!>  NOUT    INTEGER      رقم الوحدة لطباعة رسائل الخطأ
!>  OK      LOGICAL      تم تعيينه إلى .TRUE. إذا كان INFO = INFOT و
!>                       SRNAME = SRNAMT، وإلا يتم تعيينه إلى .FALSE.
!>  LERR    LOGICAL      تم تعيينه إلى .TRUE.، مما يشير إلى أنه تم استدعاء XERBLA
!>  SRNAMT  CHARACTER*(*) اسم الروتين الفرعي المستدعي المتوقع
!>
