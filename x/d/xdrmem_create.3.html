xdr(3) دليل وظائف المكتبة xdr(3)

الاسم

xdr - روتينات مكتبة لتمثيل البيانات الخارجية

المكتبة

مكتبة C القياسية (libc، -lc)

الملخص والوصف

تسمح هذه الروتينات لبرامج C بوصف هياكل بيانات اعتباطية بطريقة مستقلة عن الجهاز. يتم نقل البيانات لطلبات الإجراءات البعيدة باستخدام هذه الروتينات.

النماذج الأولية أدناه معلنة في <rpc/xdr.h> وتستخدم الأنواع التالية:

typedef int bool_t;
typedef typeof(bool_t (XDR *, void *, ...))  *xdrproc_t;

لإعلان نوع XDR، انظر <rpc/xdr.h>.

bool_t xdr_array(XDR *xdrs, char **arrp, unsigned int *sizep,
                 unsigned int maxsize, unsigned int elsize,
                 xdrproc_t elproc);
بديل مرشح يقوم بالترجمة بين المصفوفات ذات الطول المتغير وتمثيلاتها الخارجية المقابلة. الوسيط arrp هو عنوان المؤشر إلى المصفوفة، بينما sizep هو عنوان عداد عناصر المصفوفة؛ لا يمكن أن يتجاوز عدد العناصر هذا maxsize. الوسيط elsize هو sizeof كل عنصر من عناصر المصفوفة، و elproc هو مرشح XDR يقوم بالترجمة بين شكل C لعناصر المصفوفة وتمثيلها الخارجي. تعيد هذه الروتينات واحدًا إذا نجحت، وصفرًا بخلاف ذلك.
bool_t xdr_bool(XDR *xdrs, bool_t *bp);
بديل مرشح يقوم بالترجمة بين القيم المنطقية (أعداد صحيحة C) وتمثيلاتها الخارجية. عند ترميز البيانات، ينتج هذا المرشح قيمًا تساوي واحدًا أو صفرًا. تعيد هذه الروتينات واحدًا إذا نجحت، وصفرًا بخلاف ذلك.
bool_t xdr_bytes(XDR *xdrs, char **sp, unsigned int *sizep,
                 unsigned int maxsize);
بديل مرشح يقوم بالترجمة بين سلاسل البايت المعدودة وتمثيلاتها الخارجية. الوسيط sp هو عنوان مؤشر السلسلة. يقع طول السلسلة في العنوان sizep؛ لا يمكن أن تكون السلاسل أطول من maxsize. تعيد هذه الروتينات واحدًا إذا نجحت، وصفرًا بخلاف ذلك.
bool_t xdr_char(XDR *xdrs, char *cp);
بديل مرشح يقوم بالترجمة بين أحرف C وتمثيلاتها الخارجية. تعيد هذه الروتينات واحدًا إذا نجحت، وصفرًا بخلاف ذلك. ملاحظة: الأحرف المشفرة ليست مجمعة، وتشغل 4 بايت لكل منها. بالنسبة لمصفوفات الأحرف، من المفيد التفكير في xdr_bytes() أو xdr_opaque() أو xdr_string().
void xdr_destroy(XDR *xdrs);
ماكرو يستدعي روتين التدمير المرتبط بتيار XDR، xdrs. يشمل التدمير عادةً تحرير هياكل البيانات الخاصة المرتبطة بالتيار. استخدام xdrs بعد استدعاء xdr_destroy() غير محدد.
bool_t xdr_double(XDR *xdrs, double *dp);
بديل مرشح يقوم بالترجمة بين أرقام double C ذات الدقة المزدوجة وتمثيلاتها الخارجية. تعيد هذه الروتينات واحدًا إذا نجحت، وصفرًا بخلاف ذلك.
bool_t xdr_enum(XDR *xdrs, enum_t *ep);
بديل مرشح يقوم بالترجمة بين enum C (أعداد صحيحة فعليًا) وتمثيلاتها الخارجية. تعيد هذه الروتينات واحدًا إذا نجحت، وصفرًا بخلاف ذلك.
bool_t xdr_float(XDR *xdrs, float *fp);
بديل مرشح يقوم بالترجمة بين float C وتمثيلاتها الخارجية. تعيد هذه الروتينات واحدًا إذا نجحت، وصفرًا بخلاف ذلك.
void xdr_free(xdrproc_t proc, char *objp);
روتين تحرير عام. الوسيط الأول هو روتين XDR للكائن الذي يتم تحريره. الوسيط الثاني هو مؤشر إلى الكائن نفسه. ملاحظة: المؤشر الذي تم تمريره إلى هذا الروتين لا يتم تحريره، ولكن ما يشير إليه يتم تحريره (بشكل متكرر).
unsigned int xdr_getpos(XDR *xdrs);
ماكرو يستدعي روتين الحصول على الموضع المرتبط بتيار XDR، xdrs. تعيد الروتين عددًا صحيحًا غير موقع، والذي يشير إلى موضع تيار بايت XDR. ميزة مرغوبة لتيارات XDR هي أن العمليات الحسابية البسيطة تعمل مع هذا الرقم، على الرغم من أن مثيلات تيار XDR قد لا تضمن ذلك.
long *xdr_inline(XDR *xdrs, int len);
ماكرو يستدعي روتين الاستدعاء المضمن المرتبط بتيار XDR، xdrs. تعيد الروتين مؤشرًا إلى قطعة متجاورة من مخزن مؤقت للتيار؛ len هو طول البايت للمخزن المؤقت المطلوب. ملاحظة: يتم تحويل المؤشر إلى long *.
تحذير: قد تعيد xdr_inline() قيمة NULL (0) إذا لم تتمكن من تخصيص قطعة متجاورة من المخزن المؤقت. لذلك قد يختلف السلوك بين مثيلات التيار؛ وهو موجود من أجل الكفاءة.
bool_t xdr_int(XDR *xdrs, int *ip);
بديل مرشح يقوم بالترجمة بين أعداد C الصحيحة وتمثيلاتها الخارجية. تعيد هذه الروتينات واحدًا إذا نجحت، وصفرًا بخلاف ذلك.
bool_t xdr_long(XDR *xdrs, long *lp);
بديل مرشح يقوم بالترجمة بين أعداد long C الصحيحة وتمثيلاتها الخارجية. تعيد هذه الروتينات واحدًا إذا نجحت، وصفرًا بخلاف ذلك.
void xdrmem_create(XDR *xdrs, char *addr, unsigned int size,
                   enum xdr_op op);
يقوم هذا الروتين بتهيئة كائن تيار XDR الذي يشير إليه xdrs. تتم كتابة بيانات التيار إلى، أو قراءتها من، جزء من الذاكرة في الموقع addr الذي لا يتجاوز طوله size بايت. تحدد op اتجاه تيار XDR (إما XDR_ENCODE أو XDR_DECODE أو XDR_FREE).
bool_t xdr_opaque(XDR *xdrs, char *cp, unsigned int cnt);
بديل مرشح يقوم بالترجمة بين البيانات المعتمة ذات الحجم الثابت وتمثيلها الخارجي. الوسيط cp هو عنوان الكائن المعتم، و cnt هو حجمه بالبايت. تعيد هذه الروتينات واحدًا إذا نجحت، وصفرًا بخلاف ذلك.
bool_t xdr_pointer(XDR *xdrs, char **objpp,
                   unsigned int objsize, xdrproc_t xdrobj);
مثل xdr_reference() باستثناء أنها تسلسل المؤشرات الفارغة، بينما xdr_reference() لا تفعل ذلك. وبالتالي، يمكن لـ xdr_pointer() تمثيل هياكل البيانات المتكررة، مثل الأشجار الثنائية أو القوائم المترابطة.
void xdrrec_create(XDR *xdrs, unsigned int sendsize,
                   unsigned int recvsize, char *handle,
                   typeof(int (char *, char *, int)) *readit,
                   typeof(int (char *, char *, int)) *writeit);
يقوم هذا الروتين بتهيئة كائن تيار XDR الذي يشير إليه xdrs. تتم كتابة بيانات التيار إلى مخزن مؤقت بحجم sendsize؛ قيمة صفر تشير إلى أن النظام يجب أن يستخدم قيمة افتراضية مناسبة. تتم قراءة بيانات التيار من مخزن مؤقت بحجم recvsize؛ يمكن أيضًا تعيينه افتراضيًا مناسبًا بتمرير قيمة صفر. عندما يكون المخزن المؤقت للإخراج للتيار ممتلئًا، يتم استدعاء writeit. وبالمثل، عندما يكون المخزن المؤقت للإدخال للتيار فارغًا، يتم استدعاء readit. سلوك هذين الروتينين مشابه لاستدعاءات النظام read(2) و write(2)، باستثناء أن handle يتم تمريره إلى الروتينات السابقة كوسيط أول. ملاحظة: يجب على المتصل تعيين حقل op لتيار XDR.
تحذير: لقراءة تيار XDR تم إنشاؤه بواسطة واجهة برمجة التطبيقات هذه، ستحتاج إلى استدعاء xdrrec_skiprecord() أولاً قبل استدعاء أي واجهات برمجة تطبيقات XDR أخرى. هذا يدرج بايتات إضافية في التيار لتوفير حدود السجل.
