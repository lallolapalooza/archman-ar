الاسم

xdr - روتينات مكتبة لتمثيل البيانات الخارجي

المكتبة

مكتبة C القياسية (libc, -lc)

الملخص والوصف

تسمح هذه الروتينات لمبرمجي C بوصف هياكل البيانات العشوائية بطريقة مستقلة عن الآلة. يتم إرسال البيانات لاستدعاءات الإجراءات عن بعد باستخدام هذه الروتينات.

يتم الإعلان عن النماذج الأولية أدناه في <rpc/xdr.h> وتستخدم الأنواع التالية:

typedef int bool_t;
typedef typeof(bool_t (XDR *, void *, ...))  *xdrproc_t;

لإعلان النوع XDR، راجع <rpc/xdr.h>.

bool_t xdr_array(XDR *xdrs, char **arrp, unsigned int *sizep,
                 unsigned int maxsize, unsigned int elsize,
                 xdrproc_t elproc);
مرشح بدائي يترجم بين المصفوفات متغيرة الطول وتمثيلها الخارجي المقابل. الوسيط arrp هو عنوان المؤشر إلى المصفوفة، بينما sizep هو عنوان عداد عناصر المصفوفة؛ لا يمكن أن يتجاوز عدد العناصر هذا maxsize. الوسيط elsize هو sizeof كل عنصر من عناصر المصفوفة، و elproc هو مرشح XDR يترجم بين شكل C لعناصر المصفوفة وتمثيلها الخارجي. تعيد هذه الروتينات واحدًا إذا نجحت، وصفرًا إذا فشلت.
bool_t xdr_bool(XDR *xdrs, bool_t *bp);
مرشح بدائي يترجم بين القيم المنطقية (أعداد صحيحة C) وتمثيلها الخارجي. عند ترميز البيانات، ينتج هذا المرشح قيمًا إما واحدًا أو صفرًا. تعيد هذه الروتينات واحدًا إذا نجحت، وصفرًا إذا فشلت.
bool_t xdr_bytes(XDR *xdrs, char **sp, unsigned int *sizep,
                 unsigned int maxsize);
مرشح بدائي يترجم بين سلاسل البايتات المرقمة وتمثيلها الخارجي. الوسيط sp هو عنوان مؤشر السلسلة. يقع طول السلسلة في العنوان sizep؛ لا يمكن أن تكون السلاسل أطول من maxsize. تعيد هذه الروتينات واحدًا إذا نجحت، وصفرًا إذا فشلت.
bool_t xdr_char(XDR *xdrs, char *cp);
مرشح بدائي يترجم بين أحرف C وتمثيلها الخارجي. تعيد هذه الروتينات واحدًا إذا نجحت، وصفرًا إذا فشلت. ملاحظة: الأحرف المرمزة ليست مجمعة، وتشغل 4 بايت لكل منها. بالنسبة لمصفوفات الأحرف، يجدر النظر في xdr_bytes() أو xdr_opaque() أو xdr_string().
void xdr_destroy(XDR *xdrs);
ماكرو يستدعي روتين التدمير المرتبط بتيار XDR، xdrs. عادة ما يتضمن التدمير تحرير هياكل البيانات الخاصة المرتبطة بالتيار. استخدام xdrs بعد استدعاء xdr_destroy() غير محدد.
bool_t xdr_double(XDR *xdrs, double *dp);
مرشح بدائي يترجم بين أرقام double C دقة مزدوجة وتمثيلها الخارجي. تعيد هذه الروتينات واحدًا إذا نجحت، وصفرًا إذا فشلت.
bool_t xdr_enum(XDR *xdrs, enum_t *ep);
مرشح بدائي يترجم بين enums C (في الواقع أعداد صحيحة) وتمثيلها الخارجي. تعيد هذه الروتينات واحدًا إذا نجحت، وصفرًا إذا فشلت.
bool_t xdr_float(XDR *xdrs, float *fp);
مرشح بدائي يترجم بين floats C وتمثيلها الخارجي. تعيد هذه الروتينات واحدًا إذا نجحت، وصفرًا إذا فشلت.
void xdr_free(xdrproc_t proc, char *objp);
روتين تحرير عام. الوسيط الأول هو روتين XDR للكائن الذي يتم تحريره. الوسيط الثاني هو مؤشر إلى الكائن نفسه. ملاحظة: المؤشر الذي تم تمريره إلى هذا الروتين لا يتم تحريره، ولكن ما يشير إليه يتم تحريره (بشكل متكرر).
unsigned int xdr_getpos(XDR *xdrs);
ماكرو يستدعي روتين الحصول على الموضع المرتبط بتيار XDR، xdrs. يعيد الروتين عددًا صحيحًا غير موقع، والذي يشير إلى موضع تيار بايتات XDR. ميزة مرغوبة لتيارات XDR هي أن الحساب البسيط يعمل مع هذا الرقم، على الرغم من أن مثيلات تيار XDR لا يلزم أن تضمن ذلك.
long *xdr_inline(XDR *xdrs, int len);
ماكرو يستدعي روتين الخط المضمن المرتبط بتيار XDR، xdrs. يعيد الروتين مؤشرًا إلى جزء متصل من مخزن XDR المؤقت؛ len هو طول البايت للمخزن المؤقت المطلوب. ملاحظة: يتم تحويل المؤشر إلى long *.
تحذير: قد يعيد xdr_inline() القيمة NULL (0) إذا لم يتمكن من تخصيص جزء متصل من المخزن المؤقت. لذلك قد يختلف السلوك بين مثيلات التيار؛ إنه موجود من أجل الكفاءة.
bool_t xdr_int(XDR *xdrs, int *ip);
مرشح بدائي يترجم بين أعداد C الصحيحة وتمثيلها الخارجي. تعيد هذه الروتينات واحدًا إذا نجحت، وصفرًا إذا فشلت.
bool_t xdr_long(XDR *xdrs, long *lp);
مرشح بدائي يترجم بين أعداد long C الصحيحة وتمثيلها الخارجي. تعيد هذه الروتينات واحدًا إذا نجحت، وصفرًا إذا فشلت.
void xdrmem_create(XDR *xdrs, char *addr, unsigned int size,
                   enum xdr_op op);
تقوم هذه الروتينات بتهيئة كائن تيار XDR الذي يشير إليه xdrs. تتم كتابة بيانات التيار إلى، أو قراءتها من، قطعة من الذاكرة في الموقع addr التي لا يزيد طولها عن size بايت. تحدد op اتجاه تيار XDR (إما XDR_ENCODE أو XDR_DECODE أو XDR_FREE).
bool_t xdr_opaque(XDR *xdrs, char *cp, unsigned int cnt);
مرشح بدائي يترجم بين البيانات غير الشفافة ذات الحجم الثابت وتمثيلها الخارجي. الوسيط cp هو عنوان الكائن غير الشفاف، و cnt هو حجمه بالبايت. تعيد هذه الروتينات واحدًا إذا نجحت، وصفرًا إذا فشلت.
bool_t xdr_pointer(XDR *xdrs, char **objpp,
                   unsigned int objsize, xdrproc_t xdrobj);
مثل xdr_reference() باستثناء أنها تسلسل مؤشرات فارغة، بينما xdr_reference() لا تفعل ذلك. وبالتالي، يمكن لـ xdr_pointer() تمثيل هياكل البيانات المتكررة، مثل الأشجار الثنائية أو القوائم المرتبطة.
void xdrrec_create(XDR *xdrs, unsigned int sendsize,
                   unsigned int recvsize, char *handle,
                   typeof(int (char *, char *, int)) *readit,
                   typeof(int (char *, char *, int)) *writeit);
تقوم هذه الروتينات بتهيئة كائن تيار XDR الذي يشير إليه xdrs. تتم كتابة بيانات التيار إلى مخزن مؤقت بحجم sendsize؛ قيمة الصفر تشير إلى أن النظام يجب أن يستخدم قيمة افتراضية مناسبة. تتم قراءة بيانات التيار من مخزن مؤقت بحجم recvsize؛ يمكن أيضًا تعيينه إلى قيمة افتراضية مناسبة عن طريق تمرير قيمة صفر. عندما يمتلئ المخزن المؤقت للإخراج للتيار، يتم استدعاء writeit. وبالمثل، عندما يكون المخزن المؤقت للإدخال للتيار فارغًا، يتم استدعاء readit. سلوك هذين الروتينين مشابه لاستدعاءات النظام read(2) و write(2)، باستثناء أن handle يتم تمريره إلى الروتينات السابقة كوسيط أول. ملاحظة: يجب على المتصل تعيين حقل op لتيار XDR.
تحذير: لقراءة تيار XDR تم إنشاؤه بواسطة واجهة برمجة التطبيقات هذه، ستحتاج إلى استدعاء xdrrec_skiprecord() أولاً قبل استدعاء أي واجهات برمجة تطبيقات XDR أخرى. هذا يدرج بايتات إضافية في التيار لتوفير حدود السجل.
bool_t xdrrec_endofrecord(XDR *xdrs, bool_t senders);
يحدد نهاية السجل الحالي. إذا كان senders صحيحًا، فيجب أن يرسل xdrrec_endofrecord() السجل الحالي. إذا كان senders خاطئًا، فيجب على xdrrec_endofrecord() أن يتجاوز السجل الحالي.
إذا كان senders صحيحًا، وكان السجل فارغًا، فلن يتم إرسال أي شيء. إذا كان senders صحيحًا، وكان السجل غير فارغ، فسيتم إرسال السجل. إذا كان senders خاطئًا، فسيتم تجاوز السجل بغض النظر عن ما إذا كان فارغًا أم لا.
يعيد هذا الروتين واحدًا إذا نجح، وصفرًا إذا فشل.
bool_t xdrrec_skiprecord(XDR *xdrs);
يسمح هذا الروتين للمتصل بالتجاوز على السجل الذي تم إرساله بالفعل، ولكن لم يتم دفعه بالكامل بعد. هذا هو ما يجب عليك استدعاؤه قبل استدعاء أي روتين XDR آخر عند استخدام واجهة برمجة التطبيقات هذه لقراءة بيانات تم إرسالها عبر شبكة.
يعيد هذا الروتين واحدًا إذا نجح، وصفرًا إذا فشل.
bool_t xdr_reference(XDR *xdrs, char **pp, unsigned int size,
                     xdrproc_t proc);
فلتر بدائي لتمثيل هياكل البيانات المرجعية (على سبيل المثال، المؤشرات). pp هو مؤشر إلى المؤشر إلى الكائن الذي سيتم ت serialize أو deserialize. size هو حجم الكائن الذي يشير إليه المؤشر. proc هو روتين XDR الذي سيتم استخدامه لت serialize أو deserialize الكائن.
ملاحظة: لا يمكن لهذا الروتين تمثيل مؤشرات فارغة. يستخدم xdr_pointer() بدلاً منه إذا كنت بحاجة إلى تمثيل المؤشرات الفارغة.
bool_t xdr_setpos(XDR *xdrs, unsigned int pos);
ماكرو يستدعي روتين تعيين الموضع المرتبط بتيار XDR، xdrs. يضبط الموضع الحالي لتيار XDR إلى pos. يعيد الروتين واحدًا إذا نجح، وصفرًا إذا فشل.
bool_t xdr_short(XDR *xdrs, short *sh);
مرشح بدائي يترجم بين shorts C وتمثيلها الخارجي. تعيد هذه الروتينات واحدًا إذا نجحت، وصفرًا إذا فشلت.
bool_t xdr_sizeof(xdrproc_t proc, char *data);
يعيد حجم الكائن الذي تم وصفه بواسطة proc. data هو مؤشر إلى الكائن.
bool_t xdr_string(XDR *xdrs, char **sp, unsigned int maxsize);
مرشح بدائي يترجم بين سلاسل C NULL-terminated وتمثيلها الخارجي. sp هو عنوان مؤشر السلسلة. لا يمكن أن تتجاوز السلاسل maxsize بايت في الطول، بما في ذلك NULL terminatior. تعيد هذه الروتينات واحدًا إذا نجحت، وصفرًا إذا فشلت.
bool_t xdr_u_int(XDR *xdrs, unsigned int *uip);
مرشح بدائي يترجم بين أعداد C الصحيحة غير الموقعة وتمثيلها الخارجي. تعيد هذه الروتينات واحدًا إذا نجحت، وصفرًا إذا فشلت.
bool_t xdr_u_long(XDR *xdrs, unsigned long *ulp);
مرشح بدائي يترجم بين أعداد unsigned long C الصحيحة وتمثيلها الخارجي. تعيد هذه الروتينات واحدًا إذا نجحت، وصفرًا إذا فشلت.
bool_t xdr_union(XDR *xdrs, enum_t *dest, char *source,
                 const struct xdr_discrim *choices,
                 xdrproc_t xdrsz);
مرشح بدائي لتمثيل اتحادات C. dest هو مؤشر إلى المتغير الذي سيحتوي على المميز. source هو مؤشر إلى الاتحاد. choices هو مصفوفة من xdr_discrim الهياكل، والتي تربط قيم المميز بروتينات XDR. xdrsz هو روتين XDR الذي سيتم استخدامه ل serialize أو deserialize الاتحاد.
تتم كتابة المرجع نفسه إلى dest، ويتم استدعاء روتين XDR المناسب بناءً على المرجع.
إذا فشل أحد المرشحات، فستفشل xdr_union().
bool_t xdr_vector(XDR *xdrs, char *arr, unsigned int size,
                  unsigned int maxsize, unsigned int elsize,
                  xdrproc_t elproc);
مرشح بدائي يترجم بين المصفوفات ذات الطول الثابت وتمثيلها الخارجي. الوسيط arr هو عنوان المصفوفة، و size هو عدد العناصر في المصفوفة. لا يمكن أن يتجاوز عدد العناصر maxsize. الوسيط elsize هو sizeof كل عنصر، و elproc هو روتين XDR الذي سيتم استخدامه ل serialize أو deserialize عناصر المصفوفة.
تعيد هذه الروتينات واحدًا إذا نجحت، وصفرًا إذا فشلت.
bool_t xdr_void(void);
روتين XDR فارغ. لا يفعل شيئًا، ويعيد دائمًا واحدًا. يستخدم هذا في بعض الأحيان كعنصر نائب لروتينات XDR.
bool_t xdr_write(XDR *xdrs, char *buf);
يكتب buf إلى المخزن المؤقت للإخراج لتيار XDR. لا يضمن إرسال البيانات إلى وجهتها النهائية.
