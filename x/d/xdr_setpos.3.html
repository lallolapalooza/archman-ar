الاسم

xdr - روتينيات مكتبة لتمثيل البيانات الخارجي

المكتبة

مكتبة C القياسية (libc، -lc)

الملخص والوصف

تسمح هذه الروتينيات لمبرمجي C بوصف هياكل بيانات تعسفية بطريقة مستقلة عن الجهاز. يتم إرسال البيانات لاستدعاءات الإجراءات عن بعد باستخدام هذه الروتينيات.

النماذج الأولية أدناه معرّفة في <rpc/xdr.h> وتستخدم الأنواع التالية:

typedef int bool_t;
typedef typeof(bool_t (XDR *, void *, ...))  *xdrproc_t;

لتعريف النوع XDR، انظر <rpc/xdr.h>.

bool_t xdr_array(XDR *xdrs, char **arrp, unsigned int *sizep,
                 unsigned int maxsize, unsigned int elsize,
                 xdrproc_t elproc);
مرشح بدائي يترجم بين المصفوفات متغيرة الطول وتمثيلاتها الخارجية المقابلة. الوسيط arrp هو عنوان المؤشر إلى المصفوفة، بينما sizep هو عنوان عداد عناصر المصفوفة؛ لا يمكن أن يتجاوز هذا العداد maxsize. الوسيط elsize هو sizeof كل عنصر من عناصر المصفوفة، و elproc هو مرشح XDR يترجم بين شكل C لعناصر المصفوفة وتمثيلها الخارجي. تعيد هذه الروتينية واحدًا إذا نجحت، وصفرًا بخلاف ذلك.
bool_t xdr_bool(XDR *xdrs, bool_t *bp);
مرشح بدائي يترجم بين القيم المنطقية (أعداد C الصحيحة) وتمثيلاتها الخارجية. عند ترميز البيانات، تنتج هذه المرشحة قيمًا إما واحدًا أو صفرًا. تعيد هذه الروتينية واحدًا إذا نجحت، وصفرًا بخلاف ذلك.
bool_t xdr_bytes(XDR *xdrs, char **sp, unsigned int *sizep,
                 unsigned int maxsize);
مرشح بدائي يترجم بين سلاسل البايت المعدودة وتمثيلاتها الخارجية. الوسيط sp هو عنوان مؤشر السلسلة. يقع طول السلسلة في العنوان sizep؛ لا يمكن أن تكون السلاسل أطول من maxsize. تعيد هذه الروتينية واحدًا إذا نجحت، وصفرًا بخلاف ذلك.
bool_t xdr_char(XDR *xdrs, char *cp);
مرشح بدائي يترجم بين أحرف C وتمثيلاتها الخارجية. تعيد هذه الروتينية واحدًا إذا نجحت، وصفرًا بخلاف ذلك. ملاحظة: الأحرف المشفرة غير مضغوطة، وتشغل 4 بايت لكل منها. بالنسبة لمصفوفات الأحرف، يجدر النظر في xdr_bytes() أو xdr_opaque() أو xdr_string().
void xdr_destroy(XDR *xdrs);
وحدة ماكرو تستدعي روتينية التدمير المرتبطة بتيار XDR، xdrs. يتضمن التدمير عادةً تحرير هياكل البيانات الخاصة المرتبطة بالتيار. استخدام xdrs بعد استدعاء xdr_destroy() غير محدد.
bool_t xdr_double(XDR *xdrs, double *dp);
مرشح بدائي يترجم بين أرقام double الدقة في C وتمثيلاتها الخارجية. تعيد هذه الروتينية واحدًا إذا نجحت، وصفرًا بخلاف ذلك.
bool_t xdr_enum(XDR *xdrs, enum_t *ep);
مرشح بدائي يترجم بين enums C (في الواقع أعداد صحيحة) وتمثيلاتها الخارجية. تعيد هذه الروتينية واحدًا إذا نجحت، وصفرًا بخلاف ذلك.
bool_t xdr_float(XDR *xdrs, float *fp);
مرشح بدائي يترجم بين floats C وتمثيلاتها الخارجية. تعيد هذه الروتينية واحدًا إذا نجحت، وصفرًا بخلاف ذلك.
void xdr_free(xdrproc_t proc, char *objp);
روتينية تحرير عامة. الوسيط الأول هو روتينية XDR للكائن الذي يتم تحريره. الوسيط الثاني هو مؤشر إلى الكائن نفسه. ملاحظة: المؤشر المُمرر إلى هذه الروتينية *لا* يتم تحريره، ولكن ما يشير إليه *يتم* تحريره (بشكل متكرر).
unsigned int xdr_getpos(XDR *xdrs);
وحدة ماكرو تستدعي روتينية الحصول على الموضع المرتبطة بتيار XDR، xdrs. تعيد الروتينية عددًا صحيحًا غير مُشار إليه، يشير إلى موضع تيار بايت XDR. ميزة مرغوبة لتيارات XDR هي أن العمليات الحسابية البسيطة تعمل مع هذا الرقم، على الرغم من أن مثيلات تيار XDR قد لا تضمن ذلك.
long *xdr_inline(XDR *xdrs, int len);
وحدة ماكرو تستدعي روتينية الخط المباشر المرتبطة بتيار XDR، xdrs. تعيد الروتينية مؤشرًا إلى قطعة متجاورة من مخزن تيار XDR المؤقت؛ len هو طول البايت للمخزن المطلوب. ملاحظة: يتم تحويل المؤشر إلى long *.
تحذير: قد تعيد xdr_inline() قيمة NULL (0) إذا لم تتمكن من تخصيص قطعة متجاورة من المخزن المؤقت. لذلك قد يختلف السلوك بين مثيلات التيار؛ وهي موجودة لغرض الكفاءة.
bool_t xdr_int(XDR *xdrs, int *ip);
مرشح بدائي يترجم بين أعداد C الصحيحة وتمثيلاتها الخارجية. تعيد هذه الروتينية واحدًا إذا نجحت، وصفرًا بخلاف ذلك.
bool_t xdr_long(XDR *xdrs, long *lp);
مرشح بدائي يترجم بين أعداد long الصحيحة في C وتمثيلاتها الخارجية. تعيد هذه الروتينية واحدًا إذا نجحت، وصفرًا بخلاف ذلك.
void xdrmem_create(XDR *xdrs, char *addr, unsigned int size,
                   enum xdr_op op);
تهيئة هذه الروتينية كائن تيار XDR الذي يشير إليه xdrs. تتم كتابة بيانات التيار، أو قراءتها، من جزء من الذاكرة في الموقع addr الذي لا يزيد طوله عن size بايت. يحدد op اتجاه تيار XDR (إما XDR_ENCODE، أو XDR_DECODE، أو XDR_FREE).
bool_t xdr_opaque(XDR *xdrs, char *cp, unsigned int cnt);
مرشح بدائي يترجم بين البيانات المعتمة ذات الحجم الثابت وتمثيلها الخارجي. الوسيط cp هو عنوان الكائن المعتم، و cnt هو حجمه بالبايت. تعيد هذه الروتينية واحدًا إذا نجحت، وصفرًا بخلاف ذلك.
bool_t xdr_pointer(XDR *xdrs, char **objpp,
                   unsigned int objsize, xdrproc_t xdrobj);
مثل xdr_reference() باستثناء أنها تقوم بتسلسل المؤشرات الفارغة، بينما xdr_reference() لا تفعل ذلك. وبالتالي، يمكن لـ xdr_pointer() تمثيل هياكل البيانات المتكررة، مثل الأشجار الثنائية أو القوائم المترابطة.
void xdrrec_create(XDR *xdrs, unsigned int sendsize,
                   unsigned int recvsize, char *handle,
                   typeof(int (char *, char *, int)) *readit,
                   typeof(int (char *, char *, int)) *writeit);
تهيئة هذه الروتينية كائن تيار XDR الذي يشير إليه xdrs. تتم كتابة بيانات التيار إلى مخزن مؤقت بحجم sendsize؛ القيمة صفر تشير إلى أن النظام يجب أن يستخدم قيمة افتراضية مناسبة. تتم قراءة بيانات التيار من مخزن مؤقت بحجم recvsize؛ يمكن أيضًا تعيين هذا إلى قيمة افتراضية مناسبة عن طريق تمرير قيمة صفر. عندما يكون المخزن المؤقت للإخراج للتيار ممتلئًا، يتم استدعاء writeit. وبالمثل، عندما يكون المخزن المؤقت للإدخال للتيار فارغًا، يتم استدعاء readit. سلوك هاتين الروتينتين مشابه لاستدعاءات النظام read(2) و write(2)، باستثناء أن handle يتم تمريره إلى الروتينتين السابقتين كوسيط أول. ملاحظة: يجب على المتصل تعيين حقل op لتيار XDR.
تحذير: لقراءة تيار XDR تم إنشاؤه بواسطة واجهة برمجة التطبيقات هذه، ستحتاج إلى استدعاء xdrrec_skiprecord() أولاً قبل استدعاء أي واجهات برمجة تطبيقات XDR أخرى. هذا يضيف بايتات إضافية في التيار لتوفير حدود السجل.

هذا الموقع غير مرتبط بـ Arch Linux ولا تابع له بأي شكل من الأشكال. المحتوى هو ترجمة فقط لصفحات دليل Arch المنشورة على man.archlinux.org.

معلومات. أيضًا، تدفقات XDR التي تم إنشاؤها باستخدام واجهات برمجة التطبيقات xdr*_create المختلفة غير متوافقة لنفس السبب. 
bool_t xdrrec_endofrecord(XDR *xdrs, int sendnow);
يمكن استدعاء هذا الروتين فقط على التدفقات التي تم إنشاؤها بواسطة xdrrec_create(). يتم تمييز البيانات في المخزن المؤقت للإخراج كسجل مكتمل، ويتم كتابة المخزن المؤقت للإخراج اختياريًا إذا كان sendnow غير صفري. يعيد هذا الروتين واحدًا إذا نجح، وصفرًا بخلاف ذلك.
bool_t xdrrec_eof(XDR *xdrs);
يمكن استدعاء هذا الروتين فقط على التدفقات التي تم إنشاؤها بواسطة xdrrec_create(). بعد استهلاك بقية السجل الحالي في التدفق، يعيد هذا الروتين واحدًا إذا لم يكن لدى التدفق المزيد من الإدخال، وصفرًا بخلاف ذلك.
bool_t xdrrec_skiprecord(XDR *xdrs);
يمكن استدعاء هذا الروتين فقط على التدفقات التي تم إنشاؤها بواسطة xdrrec_create(). يخبر تنفيذ XDR أنه يجب تجاهل بقية السجل الحالي في المخزن المؤقت للإدخال للتدفق. يعيد هذا الروتين واحدًا إذا نجح، وصفرًا بخلاف ذلك.
bool_t xdr_reference(XDR *xdrs, char **pp, unsigned int size,
                     xdrproc_t proc);
بدائي يوفر تتبع المؤشرات داخل الهياكل. الوسيط pp هو عنوان المؤشر؛ size هو sizeof الهيكل الذي يشير إليه *pp؛ و proc هو إجراء XDR يقوم بتصفية الهيكل بين شكله C وتمثيله الخارجي. يعيد هذا الروتين واحدًا إذا نجح، وصفرًا بخلاف ذلك.
تحذير: لا يفهم هذا الروتين المؤشرات الفارغة. استخدم xdr_pointer() بدلاً من ذلك.
xdr_setpos(XDR *xdrs, unsigned int pos);
ماكرو يستدعي إجراء تعيين الموضع المرتبط بتدفق XDR xdrs. الوسيط pos هو قيمة موضع تم الحصول عليها من xdr_getpos(). يعيد هذا الروتين واحدًا إذا كان يمكن إعادة تحديد موضع تدفق XDR، وصفرًا بخلاف ذلك.
تحذير: من الصعب إعادة تحديد موضع بعض أنواع تدفقات XDR، لذلك قد يفشل هذا الروتين مع نوع واحد من التدفقات وينجح مع تدفق آخر.
bool_t xdr_short(XDR *xdrs, short *sp);
بدائي تصفية يقوم بترجمة الأعداد الصحيحة short C وتمثيلاتها الخارجية. يعيد هذا الروتين واحدًا إذا نجح، وصفرًا بخلاف ذلك.
void xdrstdio_create(XDR *xdrs, FILE *file, enum xdr_op op);
يقوم هذا الروتين بتهيئة كائن تدفق XDR الذي يشير إليه xdrs. تتم كتابة بيانات تدفق XDR، أو قراءتها، من تدفق stdio file. يحدد الوسيط op اتجاه تدفق XDR (إما XDR_ENCODE، XDR_DECODE، أو XDR_FREE).
تحذير: يقوم إجراء التدمير المرتبط بتدفقات XDR هذه باستدعاء fflush(3) على تدفق file، ولكنه لا يستدعي fclose(3) أبدًا.
bool_t xdr_string(XDR *xdrs, char **sp, unsigned int maxsize);
بدائي تصفية يقوم بترجمة السلاسل النصية C وتمثيلاتها الخارجية المقابلة. لا يمكن أن تكون السلاسل النصية أطول من maxsize. ملاحظة: sp هو عنوان مؤشر السلسلة النصية. يعيد هذا الروتين واحدًا إذا نجح، وصفرًا بخلاف ذلك.
bool_t xdr_u_char(XDR *xdrs, unsigned char *ucp);
بدائي تصفية يقوم بترجمة الأحرف C unsigned وتمثيلاتها الخارجية. يعيد هذا الروتين واحدًا إذا نجح، وصفرًا بخلاف ذلك.
bool_t xdr_u_int(XDR *xdrs, unsigned int *up);
بدائي تصفية يقوم بترجمة الأعداد الصحيحة unsigned C وتمثيلاتها الخارجية. يعيد هذا الروتين واحدًا إذا نجح، وصفرًا بخلاف ذلك.
bool_t xdr_u_long(XDR *xdrs, unsigned long *ulp);
بدائي تصفية يقوم بترجمة الأعداد الصحيحة unsigned long C وتمثيلاتها الخارجية. يعيد هذا الروتين واحدًا إذا نجح، وصفرًا بخلاف ذلك.
bool_t xdr_u_short(XDR *xdrs, unsigned short *usp);
بدائي تصفية يقوم بترجمة الأعداد الصحيحة unsigned short C وتمثيلاتها الخارجية. يعيد هذا الروتين واحدًا إذا نجح، وصفرًا بخلاف ذلك.
bool_t xdr_union(XDR *xdrs, enum_t *dscmp, char *unp,
                 const struct xdr_discrim *choices,
                 xdrproc_t defaultarm);     /* قد تساوي NULL */
بدائي تصفية يقوم بترجمة union C المميز وتمثيله الخارجي المقابل. يقوم أولاً بترجمة المميز للاتحاد الموجود في dscmp. هذا المميز هو دائمًا enum_t. بعد ذلك، تتم ترجمة الاتحاد الموجود في unp. الوسيط choices هو مؤشر إلى مصفوفة من هياكل xdr_discrim(). يحتوي كل هيكل على زوج مرتب من [value,proc]. إذا كان مميز الاتحاد يساوي value المرتبط، فسيتم استدعاء proc لترجمة الاتحاد. يتم الإشارة إلى نهاية مصفوفة هيكل xdr_discrim() بإجراء بقيمة NULL. إذا لم يتم العثور على المميز في مصفوفة choices، فسيتم استدعاء الإجراء defaultarm (إذا لم يكن NULL). يعيد واحدًا إذا نجح، وصفرًا بخلاف ذلك.
bool_t xdr_vector(XDR *xdrs, char *arrp, unsigned int size,
                  unsigned int elsize, xdrproc_t elproc);
بدائي تصفية يقوم بترجمة المصفوفات ذات الطول الثابت وتمثيلاتها الخارجية المقابلة. الوسيط arrp هو عنوان المؤشر إلى المصفوفة، بينما size هو عدد عناصر المصفوفة. الوسيط elsize هو sizeof لكل عنصر من عناصر المصفوفة، و elproc هو مرشح XDR يقوم بالترجمة بين شكل C لعناصر المصفوفة، وتمثيلها الخارجي. يعيد هذا الروتين واحدًا إذا نجح، وصفرًا بخلاف ذلك.
bool_t xdr_void(void);
يعيد هذا الروتين دائمًا واحدًا. يمكن تمريره إلى إجراءات RPC التي تتطلب وسيطًا للدالة، حيث لا يلزم القيام بأي شيء.
bool_t xdr_wrapstring(XDR *xdrs, char **sp);
بدائي يستدعي xdr_string(xdrs, sp,MAXUN.UNSIGNED ); حيث MAXUN.UNSIGNED هي القيمة القصوى لعدد صحيح غير موقّع. xdr_wrapstring() مفيد لأن حزمة RPC تمرر بحد أقصى روتينين XDR كوسائط، و xdr_string()، وهو أحد البدائيات الأكثر استخدامًا، يتطلب ثلاثة. يعيد واحدًا إذا نجح، وصفرًا بخلاف ذلك.
