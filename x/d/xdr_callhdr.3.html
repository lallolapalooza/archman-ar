متغير عام يتم تعيين قيمته بواسطة أي روتين لإنشاء عميل RPC لا ينجح. استخدم الروتين clnt_pcreateerror() لطباعة السبب.
void svc_destroy(SVCXPRT *xprt);
وحدة ماكرو تدمر مقبض خدمة RPC، xprt. عادة ما يشمل التدمير إلغاء تخصيص هياكل البيانات الخاصة، بما في ذلك xprt نفسه. استخدام xprt غير محدد بعد استدعاء هذا الروتين.
fd_set svc_fdset;
متغير عام يعكس قناع بت واصف ملف القراءة الخاص بجانب خدمة RPC؛ وهو مناسب كمعامل لاستدعاء النظام select(2). هذا يثير الاهتمام فقط إذا قام منفذ الخدمة بمعالجة الأحداث غير المتزامنة الخاصة به، بدلاً من استدعاء svc_run(). هذا المتغير للقراءة فقط (لا تمرر عنوانه إلى select(2)!)، ومع ذلك قد يتغير بعد استدعاءات svc_getreqset() أو أي من روتينات الإنشاء.
int svc_fds;
مشابه لـ svc_fdset، ولكنه محدود بـ 32 واصف ملف. تم إهمال هذه الواجهة بواسطة svc_fdset.
svc_freeargs(SVCXPRT *xprt, xdrproc_t inproc, char *in);
وحدة ماكرو تحرر أي بيانات تم تخصيصها بواسطة نظام RPC/XDR عند فك ترميز الوسائط لإجراء خدمة باستخدام svc_getargs(). يعيد هذا الروتين 1 إذا تم تحرير النتائج بنجاح، وصفر بخلاف ذلك.
svc_getargs(SVCXPRT *xprt, xdrproc_t inproc, char *in);
وحدة ماكرو تقوم بفك ترميز وسائط طلب RPC المرتبط بمقبض خدمة RPC، xprt. الوسيط in هو العنوان الذي سيتم وضع الوسائط فيه؛ inproc هو روتين XDR المستخدم لفك ترميز الوسائط. يعيد هذا الروتين واحدًا إذا نجح فك الترميز، وصفرًا بخلاف ذلك.
struct sockaddr_in *svc_getcaller(SVCXPRT *xprt);
الطريقة المعتمدة للحصول على عنوان الشبكة للمتصل بإجراء مرتبط بمقبض خدمة RPC، xprt.
void svc_getreqset(fd_set *rdfds);
هذا الروتين يثير الاهتمام فقط إذا كان منفذ الخدمة لا يستدعي svc_run()، ولكنه ينفذ بدلاً من ذلك معالجة أحداث غير متزامنة مخصصة. يتم استدعاؤه عندما حدد استدعاء النظام select(2) أن طلب RPC قد وصل على بعض مقابس RPC؛ rdfds هو قناع بت واصف ملف القراءة الناتج. يعود الروتين عند الانتهاء من خدمة جميع المقابس المرتبطة بقيمة rdfds.
void svc_getreq(int rdfds);
مشابه لـ svc_getreqset()، ولكنه محدود بـ 32 واصف ملف. تم إهمال هذه الواجهة بواسطة svc_getreqset().
bool_t svc_register(SVCXPRT *xprt, unsigned long prognum,
                    unsigned long versnum,
                    typeof(void (struct svc_req *, SVCXPRT *)) *dispatch,
                    unsigned long protocol);
يربط prognum و versnum بإجراء التوزيع للخدمة، dispatch. إذا كان protocol صفرًا، فلن يتم تسجيل الخدمة مع خدمة portmap. إذا كان protocol غير صفري، فسيتم إنشاء تعيين للثلاثي [prognum,versnum,protocol] إلى xprt->xp_port مع خدمة portmap المحلية (عادة ما يكون protocol صفرًا، IPPROTO_UDP أو IPPROTO_TCP). الإجراء dispatch له الشكل التالي:

dispatch(struct svc_req *request, SVCXPRT *xprt);

يعيد الروتين svc_register() واحدًا إذا نجح، وصفرًا بخلاف ذلك.
void svc_run(void);
هذا الروتين لا يعود أبدًا. ينتظر وصول طلبات RPC، ويستدعي إجراء الخدمة المناسب باستخدام svc_getreq() عند وصوله. عادة ما ينتظر هذا الروتين استدعاء النظام select(2) للعودة.
bool_t svc_sendreply(SVCXPRT *xprt, xdrproc_t outproc, char *out);
يتم استدعاؤه بواسطة إجراء توزيع خدمة RPC لإرسال نتائج استدعاء إجراء بعيد. الوسيط xprt هو مقبض النقل المرتبط بالطلب؛ outproc هو روتين XDR الذي يستخدم لتشفير النتائج؛ و out هو عنوان النتائج. يعيد هذا الروتين واحدًا إذا نجح، وصفرًا بخلاف ذلك.
void svc_unregister(unsigned long prognum, unsigned long versnum);
إزالة جميع تعيينات الزوج [prognum,versnum] إلى إجراءات التوزيع، والزوج [prognum,versnum,*] إلى رقم المنفذ.
void svcerr_auth(SVCXPRT *xprt, enum auth_stat why);
يتم استدعاؤه بواسطة إجراء توزيع خدمة يرفض أداء استدعاء إجراء بعيد بسبب خطأ في المصادقة.
void svcerr_decode(SVCXPRT *xprt);
يتم استدعاؤه بواسطة إجراء توزيع خدمة لا يمكنه فك ترميز وسائطه بنجاح. راجع أيضًا svc_getargs().
void svcerr_noproc(SVCXPRT *xprt);
يتم استدعاؤه بواسطة إجراء توزيع خدمة لا ينفذ رقم الإجراء الذي يطلبه المتصل.
void svcerr_noprog(SVCXPRT *xprt);
يتم استدعاؤه عندما لا يتم تسجيل البرنامج المطلوب مع حزمة RPC. عادة ما لا يحتاج منفذو الخدمة إلى هذا الروتين.
void svcerr_progvers(SVCXPRT *xprt, unsigned long low_vers,
                     unsigned long high_vers);
يتم استدعاؤه عندما لا يتم تسجيل الإصدار المطلوب لبرنامج مع حزمة RPC. عادة ما لا يحتاج منفذو الخدمة إلى هذا الروتين.
void svcerr_systemerr(SVCXPRT *xprt);
يتم استدعاؤه بواسطة إجراء توزيع خدمة عند اكتشاف خطأ في النظام لا يغطيه بروتوكول معين. على سبيل المثال، إذا لم يعد بإمكان الخدمة تخصيص مساحة تخزين، فقد تستدعي هذا الروتين.
void svcerr_weakauth(SVCXPRT *xprt);
يتم استدعاؤه بواسطة إجراء توزيع خدمة يرفض أداء استدعاء إجراء بعيد بسبب عدم كفاية معلمات المصادقة. يستدعي الروتين svcerr_auth(xprt, AUTH_TOOWEAK).
SVCXPRT *svcfd_create(int fd, unsigned int sendsize,
                      unsigned int recvsize);
إنشاء خدمة فوق أي واصف ملف مفتوح. عادة، يكون واصف الملف هذا مقبسًا متصلاً لبروتوكول دفق مثل TCP. يشير sendsize و recvsize إلى أحجام مخازن الإرسال والاستقبال. إذا كانا صفرين، يتم اختيار قيمة افتراضية معقولة.
SVCXPRT *svcraw_create(void);
ينشئ هذا الروتين مقبس خدمة RPC وهمي، ويعيد مؤشرًا إليه. المقبس هو في الواقع مخزن مؤقت داخل مساحة عنوان العملية، لذا يجب أن يكون عميل RPC المقابل في نفس مساحة العنوان؛ راجع clntraw_create(). يسمح هذا الروتين بمحاكاة RPC واكتساب نفقات RPC (مثل أوقات الذهاب والإياب)، دون أي تدخل من النواة. يعيد هذا الروتين NULL إذا فشل.
SVCXPRT *svctcp_create(int sock, unsigned int send_buf_size,
                       unsigned int recv_buf_size);
ينشئ هذا الروتين مقبس خدمة RPC يعتمد على TCP/IP، ويعيد مؤشرًا إليه. يرتبط المقبس بالمقبس sock، والذي قد يكون RPC_ANYSOCK، وفي هذه الحالة يتم إنشاء مقبس جديد. إذا لم يكن المقبس مرتبطًا بمنفذ TCP محلي، فسيقوم هذا الروتين بربطه بمنفذ عشوائي. عند الاكتمال، يكون xprt->xp_sock هو واصف مقبس النقل، و xprt->xp_port هو رقم منفذ النقل. يعيد هذا الروتين NULL إذا فشل. نظرًا لأن RPC المستند إلى TCP يستخدم I/O المخزن مؤقتًا، يمكن للمستخدمين تحديد حجم المخازن المؤقتة؛ القيم الصفرية تختار القيم الافتراضية المناسبة.
SVCXPRT *svcudp_bufcreate(int sock, unsigned int sendsize,
                          unsigned int recosize);
ينشئ هذا الروتين مقبس خدمة RPC يعتمد على UDP/IP، ويعيد مؤشرًا إليه. يرتبط المقبس بالمقبس sock، والذي قد يكون RPC_ANYSOCK، وفي هذه الحالة يتم إنشاء مقبس جديد. إذا لم يكن المقبس مرتبطًا بمنفذ UDP محلي، فسيقوم هذا الروتين بربطه بمنفذ عشوائي. عند الاكتمال، يكون xprt->xp_sock هو واصف مقبس النقل، و xprt->xp_port هو رقم منفذ النقل. يعيد هذا الروتين NULL إذا فشل.
هذا يسمح للمستخدم بتحديد الحد الأقصى لحجم الحزمة لإرسال واستقبال رسائل RPC المستندة إلى UDP.
SVCXPRT *svcudp_create(int sock);
هذا الاستدعاء مكافئ لـ svcudp_bufcreate(sock,SZ,SZ) لحجم افتراضي ما SZ.
bool_t xdr_accepted_reply(XDR *xdrs, struct accepted_reply *ar);
يستخدم لتشفير رسائل رد RPC. هذا الروتين مفيد للمستخدمين الذين يرغبون في إنشاء رسائل بأسلوب RPC دون استخدام حزمة RPC.
bool_t xdr_authunix_parms(XDR *xdrs, struct authunix_parms *aupp);
يستخدم لوصف بيانات اعتماد UNIX. هذا الروتين مفيد للمستخدمين الذين يرغبون في إنشاء بيانات الاعتماد هذه دون استخدام حزمة مصادقة RPC.
void xdr_callhdr(XDR *xdrs, struct rpc_msg *chdr);
