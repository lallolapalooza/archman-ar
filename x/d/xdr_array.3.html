xdr(3) دليل وظائف المكتبة xdr(3)

الاسم

xdr — روتينات مكتبة لتمثيل البيانات الخارجي

المكتبة

مكتبة C القياسية (libc, -lc)

الملخص والوصف

تسمح هذه الروتينات لمبرمجي C بوصف هياكل بيانات تعسفية بطريقة مستقلة عن الجهاز. يتم إرسال البيانات لاستدعاءات الإجراءات البعيدة باستخدام هذه الروتينات.

النماذج الأولية أدناه معلنة في <rpc/xdr.h> وتستخدم الأنواع التالية:

typedef int bool_t;
typedef typeof(bool_t (XDR *, void *, ...))  *xdrproc_t;

لإعلان النوع XDR، انظر <rpc/xdr.h>.

bool_t xdr_array(XDR *xdrs, char **arrp, unsigned int *sizep,
                 unsigned int maxsize, unsigned int elsize,
                 xdrproc_t elproc);
عامل تصفية بدائي يترجم بين المصفوفات متغيرة الطول وتمثيلاتها الخارجية المقابلة. الوسيط arrp هو عنوان المؤشر إلى المصفوفة، بينما sizep هو عنوان عدد العناصر في المصفوفة؛ لا يمكن أن يتجاوز عدد العناصر هذا maxsize. الوسيط elsize هو sizeof لكل عنصر من عناصر المصفوفة، و elproc هو مرشح XDR يترجم بين شكل C لعناصر المصفوفة وتمثيلها الخارجي. تعيد هذه الدالة واحدًا إذا نجحت، وصفرًا بخلاف ذلك.
bool_t xdr_bool(XDR *xdrs, bool_t *bp);
عامل تصفية بدائي يترجم بين القيم المنطقية (أعداد صحيحة C) وتمثيلاتها الخارجية. عند ترميز البيانات، ينتج هذا المرشح قيمًا إما واحدًا أو صفرًا. تعيد هذه الدالة واحدًا إذا نجحت، وصفرًا بخلاف ذلك.
bool_t xdr_bytes(XDR *xdrs, char **sp, unsigned int *sizep,
                 unsigned int maxsize);
عامل تصفية بدائي يترجم بين سلاسل البايت المرقمة وتمثيلاتها الخارجية. الوسيط sp هو عنوان مؤشر السلسلة. يقع طول السلسلة في العنوان sizep؛ لا يمكن أن تكون السلاسل أطول من maxsize. تعيد هذه الدالة واحدًا إذا نجحت، وصفرًا بخلاف ذلك.
bool_t xdr_char(XDR *xdrs, char *cp);
عامل تصفية بدائي يترجم بين أحرف C وتمثيلاتها الخارجية. تعيد هذه الدالة واحدًا إذا نجحت، وصفرًا بخلاف ذلك. ملاحظة: الأحرف المشفرة ليست مجمعة، وتشغل 4 بايت لكل منها. بالنسبة لمصفوفات الأحرف، من المفيد النظر في xdr_bytes() أو xdr_opaque() أو xdr_string().
void xdr_destroy(XDR *xdrs);
ماكرو يستدعي روتين التدمير المرتبط بتيار XDR، xdrs. يتضمن التدمير عادةً تحرير هياكل البيانات الخاصة المرتبطة بالتيار. استخدام xdrs بعد استدعاء xdr_destroy() غير محدد.
bool_t xdr_double(XDR *xdrs, double *dp);
عامل تصفية بدائي يترجم بين أرقام double C مزدوجة الدقة وتمثيلاتها الخارجية. تعيد هذه الدالة واحدًا إذا نجحت، وصفرًا بخلاف ذلك.
bool_t xdr_enum(XDR *xdrs, enum_t *ep);
عامل تصفية بدائي يترجم بين enum C (أعداد صحيحة فعليًا) وتمثيلاتها الخارجية. تعيد هذه الدالة واحدًا إذا نجحت، وصفرًا بخلاف ذلك.
bool_t xdr_float(XDR *xdrs, float *fp);
عامل تصفية بدائي يترجم بين float C وتمثيلاتها الخارجية. تعيد هذه الدالة واحدًا إذا نجحت، وصفرًا بخلاف ذلك.
void xdr_free(xdrproc_t proc, char *objp);
روتين تحرير عام. الوسيط الأول هو روتين XDR للكائن الذي يتم تحريره. الوسيط الثاني هو مؤشر إلى الكائن نفسه. ملاحظة: المؤشر الذي تم تمريره إلى هذا الروتين لا يتم تحريره، ولكن ما يشير إليه يتم تحريره (بشكل متكرر).
unsigned int xdr_getpos(XDR *xdrs);
ماكرو يستدعي روتين الحصول على الموضع المرتبط بتيار XDR، xdrs. تعيد الدالة عددًا صحيحًا غير موقّع، والذي يشير إلى موضع تدفق بايت XDR. ميزة مرغوبة لتيارات XDR هي أن العمليات الحسابية البسيطة تعمل مع هذا الرقم، على الرغم من أن مثيلات تدفق XDR ليست ملزمة بضمان ذلك.
long *xdr_inline(XDR *xdrs, int len);
ماكرو يستدعي روتين السطر المرتبط بتيار XDR، xdrs. تعيد الدالة مؤشرًا إلى جزء متصل من مخزن مؤقت للتيار؛ len هو طول البايت للمخزن المؤقت المطلوب. ملاحظة: يتم تحويل المؤشر إلى long *.
تحذير: قد تعيد xdr_inline() القيمة NULL (0) إذا لم تتمكن من تخصيص جزء متصل من المخزن المؤقت. لذلك قد يختلف السلوك بين مثيلات التدفق؛ وهو موجود من أجل الكفاءة.
bool_t xdr_int(XDR *xdrs, int *ip);
عامل تصفية بدائي يترجم بين أعداد C الصحيحة وتمثيلاتها الخارجية. تعيد هذه الدالة واحدًا إذا نجحت، وصفرًا بخلاف ذلك.
bool_t xdr_long(XDR *xdrs, long *lp);
عامل تصفية بدائي يترجم بين long C الصحيحة وتمثيلاتها الخارجية. تعيد هذه الدالة واحدًا إذا نجحت، وصفرًا بخلاف ذلك.
void xdrmem_create(XDR *xdrs, char *addr, unsigned int size,
                   enum xdr_op op);
تقوم هذه الدالة بتهيئة كائن تدفق XDR المشار إليه بواسطة xdrs. تتم كتابة بيانات التدفق، أو قراءتها، من جزء من الذاكرة في الموقع addr الذي لا يزيد طوله عن size بايت. يحدد op اتجاه تدفق XDR (إما XDR_ENCODE أو XDR_DECODE أو XDR_FREE).
bool_t xdr_opaque(XDR *xdrs, char *cp, unsigned int cnt);
عامل تصفية بدائي يترجم بين البيانات المعتمة ذات الحجم الثابت وتمثيلها الخارجي. الوسيط cp هو عنوان الكائن المعتم، و cnt هو حجمه بالبايت. تعيد هذه الدالة واحدًا إذا نجحت، وصفرًا بخلاف ذلك.
bool_t xdr_pointer(XDR *xdrs, char **objpp,
                   unsigned int objsize, xdrproc_t xdrobj);
مثل xdr_reference() باستثناء أنها تسلسل مؤشرات فارغة، بينما xdr_reference() لا تفعل ذلك. وبالتالي، يمكن لـ xdr_pointer() تمثيل هياكل البيانات المتكررة، مثل الأشجار الثنائية أو القوائم المرتبطة.
void xdrrec_create(XDR *xdrs, unsigned int sendsize,
                   unsigned int recvsize, char *handle,
                   typeof(int (char *, char *, int)) *readit,
                   typeof(int (char *, char *, int)) *writeit);
تقوم هذه الدالة بتهيئة كائن تدفق XDR المشار إليه بواسطة xdrs. تتم كتابة بيانات التدفق إلى مخزن مؤقت بحجم sendsize؛ قيمة صفر تشير إلى أن النظام يجب أن يستخدم قيمة افتراضية مناسبة. تتم قراءة بيانات التدفق من مخزن مؤقت بحجم recvsize؛ يمكن أيضًا تعيينه إلى قيمة افتراضية مناسبة عن طريق تمرير قيمة صفر. عندما يكون مخزن الإخراج المؤقت للتيار ممتلئًا، يتم استدعاء writeit. وبالمثل، عندما يكون مخزن الإدخال المؤقت للتيار فارغًا، يتم استدعاء readit. سلوك هاتين الدالتين مشابه لاستدعاءات النظام read(2) و write(2)، باستثناء أن handle يتم تمريره إلى الروتينات السابقة كوسيط أول. ملاحظة: يجب على المتصل تعيين حقل op لتدفق XDR.
تحذير: لقراءة تدفق XDR تم إنشاؤه بواسطة واجهة برمجة التطبيقات هذه، ستحتاج إلى استدعاء xdrrec_skiprecord() أولاً قبل استدعاء أي واجهات برمجة تطبيقات XDR أخرى. هذا يدرج بايتات إضافية في التدفق لتوفير حدود السجل
