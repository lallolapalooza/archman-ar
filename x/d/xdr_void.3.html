الملخص والوصف

تسمح هذه الإجراءات لمبرمجي C بوصف هياكل بيانات عشوائية بطريقة مستقلة عن الآلة. يتم إرسال البيانات لاستدعاءات الإجراءات البعيدة باستخدام هذه الإجراءات.

تم الإعلان عن النماذج الأولية أدناه في <rpc/xdr.h> وتستخدم الأنواع التالية:

typedef int bool_t;
typedef typeof(bool_t (XDR *, void *, ...))  *xdrproc_t;

لإعلان نوع XDR، انظر <rpc/xdr.h>.

bool_t xdr_array(XDR *xdrs, char **arrp, unsigned int *sizep,
                 unsigned int maxsize, unsigned int elsize,
                 xdrproc_t elproc);
بدائية مرشحة تقوم بالترجمة بين المصفوفات متغيرة الطول وتمثيلاتها الخارجية المقابلة. الوسيط arrp هو عنوان المؤشر إلى المصفوفة، بينما sizep هو عنوان عدد عناصر المصفوفة؛ لا يمكن أن يتجاوز عدد العناصر هذا maxsize. الوسيط elsize هو sizeof لكل عنصر من عناصر المصفوفة، و elproc هو مرشح XDR يترجم بين شكل C لعناصر المصفوفة، وتمثيلها الخارجي. تعيد هذه الدالة واحدًا إذا نجحت، وصفرًا إذا فشلت.
bool_t xdr_bool(XDR *xdrs, bool_t *bp);
بدائية مرشحة تقوم بالترجمة بين القيم المنطقية (أعداد C الصحيحة) وتمثيلاتها الخارجية. عند ترميز البيانات، تنتج هذه المرشحة قيمًا إما واحدًا أو صفرًا. تعيد هذه الدالة واحدًا إذا نجحت، وصفرًا إذا فشلت.
bool_t xdr_bytes(XDR *xdrs, char **sp, unsigned int *sizep,
                 unsigned int maxsize);
بدائية مرشحة تقوم بالترجمة بين سلاسل البايت المعدودة وتمثيلاتها الخارجية. الوسيط sp هو عنوان مؤشر السلسلة. يقع طول السلسلة في العنوان sizep؛ لا يمكن أن تكون السلاسل أطول من maxsize. تعيد هذه الدالة واحدًا إذا نجحت، وصفرًا إذا فشلت.
bool_t xdr_char(XDR *xdrs, char *cp);
بدائية مرشحة تقوم بالترجمة بين أحرف C وتمثيلاتها الخارجية. تعيد هذه الدالة واحدًا إذا نجحت، وصفرًا إذا فشلت. ملاحظة: الأحرف المشفرة غير مجمعة، وتشغل 4 بايت لكل منها. بالنسبة لمصفوفات الأحرف، يجدر النظر في xdr_bytes() أو xdr_opaque() أو xdr_string().
void xdr_destroy(XDR *xdrs);
ماكرو يستدعي إجراء التدمير المرتبط بدفق XDR، xdrs. يتضمن التدمير عادةً تحرير هياكل البيانات الخاصة المرتبطة بالتدفق. استخدام xdrs بعد استدعاء xdr_destroy() غير معرف.
bool_t xdr_double(XDR *xdrs, double *dp);
بدائية مرشحة تقوم بالترجمة بين أرقام double C للأرقام ذات الدقة المزدوجة وتمثيلاتها الخارجية. تعيد هذه الدالة واحدًا إذا نجحت، وصفرًا إذا فشلت.

bool_t xdr_float(XDR *xdrs, float *fp);
بدائية مرشحة تقوم بالترجمة بين float C وتمثيلاتها الخارجية. تعيد هذه الدالة واحدًا إذا نجحت، وصفرًا إذا فشلت.
void xdr_free(xdrproc_t proc, char *objp);
إجراء تحرير عام. الوسيط الأول هو إجراء XDR للكائن الذي يتم تحريره. الوسيط الثاني هو مؤشر إلى الكائن نفسه. ملاحظة: المؤشر الذي تم تمريره إلى هذه الدالة *لا* يتم تحريره، ولكن ما يشير إليه *يتم* تحريره (بشكل متكرر).
unsigned int xdr_getpos(XDR *xdrs);
ماكرو يستدعي إجراء الحصول على الموضع المرتبط بدفق XDR، xdrs. تعيد الدالة عددًا صحيحًا غير موقع، يشير إلى موضع تدفق بايت XDR. ميزة مرغوبة لدفقات XDR هي أن العمليات الحسابية البسيطة تعمل مع هذا الرقم، على الرغم من أن مثيلات تدفق XDR قد لا تضمن ذلك.
long *xdr_inline(XDR *xdrs, int len);
ماكرو يستدعي إجراء الاستدعاء المضمن المرتبط بدفق XDR، xdrs. تعيد الدالة مؤشرًا إلى قطعة متجاورة من مخزن XDR المؤقت؛ len هو طول البايت المطلوب للمخزن المؤقت. ملاحظة: يتم تحويل المؤشر إلى long *.
تحذير: قد يعيد xdr_inline() قيمة NULL (0) إذا لم يتمكن من تخصيص قطعة متجاورة من المخزن المؤقت. لذلك قد يختلف السلوك بين مثيلات التدفق؛ إنه موجود من أجل الكفاءة.
bool_t xdr_int(XDR *xdrs, int *ip);
بدائية مرشحة تقوم بالترجمة بين أعداد C الصحيحة وتمثيلاتها الخارجية. تعيد هذه الدالة واحدًا إذا نجحت، وصفرًا إذا فشلت.
bool_t xdr_long(XDR *xdrs, long *lp);
بدائية مرشحة تقوم بالترجمة بين أعداد long الصحيحة من C وتمثيلاتها الخارجية. تعيد هذه الدالة واحدًا إذا نجحت، وصفرًا إذا فشلت.
void xdrmem_create(XDR *xdrs, char *addr, unsigned int size,
                   enum xdr_op op);
تقوم هذه الدالة بتهيئة كائن تدفق XDR المشار إليه بواسطة xdrs. تتم كتابة بيانات التدفق، أو قراءتها، من جزء من الذاكرة في الموقع addr والذي لا يزيد طوله عن size بايت. تحدد op اتجاه تدفق XDR (إما XDR_ENCODE أو XDR_DECODE أو XDR_FREE).
bool_t xdr_opaque(XDR *xdrs, char *cp, unsigned int cnt);
بدائية مرشحة تقوم بالترجمة بين البيانات المعتمة ذات الحجم الثابت وتمثيلها الخارجي. الوسيط cp هو عنوان الكائن المعتم، و cnt هو حجمه بالبايت. تعيد هذه الدالة واحدًا إذا نجحت، وصفرًا إذا فشلت.
bool_t xdr_pointer(XDR *xdrs, char **objpp,
                   unsigned int objsize, xdrproc_t xdrobj);
مثل xdr_reference() باستثناء أنها تقوم بتسلسل المؤشرات الفارغة، بينما xdr_reference() لا تفعل ذلك. وبالتالي، يمكن لـ xdr_pointer() تمثيل هياكل البيانات المتكررة، مثل الأشجار الثنائية أو القوائم المرتبطة.
void xdrrec_create(XDR *xdrs, unsigned int sendsize,
                   unsigned int recvsize, char *handle,
                   typeof(int (char *, char *, int)) *readit,
                   typeof(int (char *, char *, int)) *writeit);
تقوم هذه الدالة بتهيئة كائن تدفق XDR المشار إليه بواسطة xdrs. تتم كتابة بيانات التدفق إلى مخزن مؤقت بحجم sendsize؛ يشير القيمة صفر إلى أن النظام يجب أن يستخدم قيمة افتراضية مناسبة. تتم قراءة بيانات التدفق من مخزن مؤقت بحجم recvsize؛ يمكن أيضًا تعيينه إلى قيمة افتراضية مناسبة عن طريق تمرير قيمة صفر. عندما يكون المخزن المؤقت الناتج للتدفق ممتلئًا، يتم استدعاء writeit. وبالمثل، عندما يكون المخزن المؤقت للإدخال للتدفق فارغًا، يتم استدعاء readit. سلوك هاتين الدالتين مشابه لاستدعاءات النظام read(2) و write(2)، باستثناء أن handle يتم تمريره إلى الدوال السابقة كوسيط أول. ملاحظة: يجب على المتصل تعيين حقل op لدفق XDR.
تحذير: لقراءة تدفق XDR تم إنشاؤه بواسطة واجهة برمجة التطبيقات هذه، ستحتاج إلى استدعاء xdrrec_skiprecord() أولاً قبل استدعاء أي واجهات برمجة تطبيقات XDR أخرى. هذا يضيف بايتات إضافية في التدفق لتوفير حدود السجلات.
bool_t xdr_reference(XDR *xdrs, char **pp, unsigned int len,
                   xdrproc_t proc);
تترجم بين مرجع إلى كائن C (مؤشر) وتمثيله الخارجي. الوسيط pp هو عنوان المؤشر للكائن، و len هو حجم الكائن بالبايت، و proc هو إجراء XDR الخاص بالكائن. تعيد هذه الدالة واحدًا إذا نجحت، وصفرًا إذا فشلت.
bool_t xdr_setpos(XDR *xdrs, unsigned int pos);
ماكرو يستدعي إجراء تعيين الموضع المرتبط بدفق XDR، xdrs. يتم تعيين الموضع الحالي لتدفق بايت XDR إلى pos. تعيد الدالة واحدًا إذا نجحت، وصفرًا إذا فشلت.
bool_t xdr_short(XDR *xdrs, short *sp);
بدائية مرشحة تقوم بالترجمة بين أعداد short القصيرة من C وتمثيلاتها الخارجية. تعيد هذه الدالة واحدًا إذا نجحت، وصفرًا إذا فشلت.
bool_t xdr_skiprecord(XDR *xdrs);
هذه الدالة تقرأ من تدفق XDR المسجل وتهمل البيانات حتى نهاية السجل الحالي. يتم استخدامها لتخطي رأس السجل.
char *xdr_stream_create(
    typeof(int (char *, char *, int)) *readit,
    typeof(int (char *, char *, int)) *writeit,
    char *handle,
    unsigned int flags
تنشئ هذه الدالة تدفق XDR مسجلًا باستخدام readit و writeit و handle. يتم استخدام flags لتهيئة التدفق.
bool_t xdr_string(XDR *xdrs, char **sp, unsigned int maxsize);

bool_t xdr_union(XDR *xdrs, enum_t *discr, char *		hndlptr,
                 const struct xdr_discmu *choices,
                 typeof(bool_t (XDR *, void *, void *)) *defaultvert);
تترجم هذه الدالة بين أنواع الاتحاد في C وتمثيلاتها الخارجية. الوسيط discr هو مؤشر إلى المتغير المميز للاتحاد. الوسيط hndlptr هو مؤشر إلى الاتحاد. الوسيط choices هو مصفوفة من هياكل xdr_discmu التي تحدد إجراءات XDR التي سيتم استخدامها لكل قيمة مميزة. الوسيط defaultvert هو مؤشر إلى إجراء XDR الذي سيتم استخدامه إذا لم يتم العثور على قيمة مميزة في choices. تعيد الدالة واحدًا إذا نجحت، وصفرًا إذا فشلت.
bool_t xdr_void(void);
لا تفعل شيئًا. تعيد دائمًا واحدًا. تستخدم لتمثيل استدعاءات RPC التي لا تتطلب وسيطًا أو لا ترجع قيمة.
bool_t xdr_wrapstring(XDR *xdrs, char **sp);
تترجم بين سلاسل C القياسية (سلاسل حرفية تنتهي بـ NULL) وتمثيلاتها الخارجية. الوسيط sp هو عنوان مؤشر السلسلة. لا توجد حدود للحجم، ولكن يتم استخدام xdr_string() إذا كان الطول غير معروف.

هذا الموقع غير مرتبط بـ Arch Linux ولا تابع له بأي شكل من الأشكال. المحتوى هو ترجمة فقط لصفحات دليل Arch المنشورة على man.archlinux.org.

void xdrstdio_create(XDR *xdrs, FILE *file, enum xdr_op op);

                 const struct xdr_discrim *choices,
                 xdrproc_t defaultarm);     /* may equal NULL */
بدائية مرشح تترجم بين union مُميَّز بلغة C وتمثيله الخارجي المقابل. تقوم أولاً بترجمة المميز للـ union الموجود في dscmp. هذا المميز هو دائمًا enum_t. بعد ذلك، يتم ترجمة الـ union الموجود في unp. الوسيط choices هو مؤشر إلى مصفوفة من هياكل xdr_discrim(). يحتوي كل هيكل على زوج مرتب من [value,proc]. إذا كان مميز الـ union يساوي value المرتبط، فسيتم استدعاء proc لترجمة الـ union. يتم تحديد نهاية مصفوفة هياكل xdr_discrim() بإجراء بقيمة NULL. إذا لم يتم العثور على المميز في مصفوفة choices، فسيتم استدعاء إجراء defaultarm (إذا لم يكن NULL). يعيد واحدًا إذا نجح، وصفرًا بخلاف ذلك.
bool_t xdr_vector(XDR *xdrs, char *arrp, unsigned int size,
                  unsigned int elsize, xdrproc_t elproc);
بدائية مرشح تترجم بين المصفوفات ذات الطول الثابت وتمثيلاتها الخارجية المقابلة. الوسيط arrp هو عنوان المؤشر إلى المصفوفة، بينما size هو عدد عناصر المصفوفة. الوسيط elsize هو sizeof كل عنصر من عناصر المصفوفة، و elproc هو مرشح XDR يترجم بين شكل عناصر المصفوفة بلغة C وتمثيلها الخارجي. يعيد هذا الإجراء واحدًا إذا نجح، وصفرًا بخلاف ذلك.
bool_t xdr_void(void);
يعيد هذا الإجراء دائمًا واحدًا. يمكن تمريره إلى إجراءات RPC التي تتطلب وسيطًا وظيفيًا، حيث لا يلزم القيام بأي شيء.
bool_t xdr_wrapstring(XDR *xdrs, char **sp);
بدائية تستدعي xdr_string(xdrs, sp,MAXUN.UNSIGNED ); حيث MAXUN.UNSIGNED هي القيمة القصوى لعدد صحيح غير موقّع. xdr_wrapstring() مفيدة لأن حزمة RPC تمرر بحد أقصى إجراءي XDR كوسائط، و xdr_string()، وهي واحدة من البدائيات الأكثر استخدامًا، تتطلب ثلاثة. تعيد واحدًا إذا نجحت، وصفرًا بخلاف ذلك.
