الاسم

xdr - إجراءات مكتبة للتمثيل الخارجي للبيانات

المكتبة

مكتبة C القياسية (libc، -lc)

الملخص والوصف

تسمح هذه الإجراءات لبرامج C بوصف هياكل بيانات اعتباطية بطريقة مستقلة عن الآلة. يتم إرسال البيانات لطلبات الإجراءات البعيدة باستخدام هذه الإجراءات.

النماذج الأولية أدناه معرّفة في <rpc/xdr.h> وتستخدم الأنواع التالية:

typedef int bool_t;
typedef typeof(bool_t (XDR *, void *, ...))  *xdrproc_t;

لتعريف النوع XDR، انظر <rpc/xdr.h>.

bool_t xdr_array(XDR *xdrs, char **arrp, unsigned int *sizep,
                 unsigned int maxsize, unsigned int elsize,
                 xdrproc_t elproc);
بدائية مرشح تترجم بين المصفوفات ذات الطول المتغير وتمثيلاتها الخارجية المقابلة. الوسيط arrp هو عنوان المؤشر إلى المصفوفة، بينما sizep هو عنوان عدد العناصر في المصفوفة؛ لا يمكن أن يتجاوز عدد العناصر هذا maxsize. الوسيط elsize هو sizeof كل عنصر من عناصر المصفوفة، و elproc هو مرشح XDR يترجم بين شكل C لعناصر المصفوفة وتمثيلها الخارجي. تعيد هذه الدالة واحدًا إذا نجحت، وصفرًا إذا فشلت.
bool_t xdr_bool(XDR *xdrs, bool_t *bp);
بدائية مرشح تترجم بين القيم المنطقية (أعداد صحيحة C) وتمثيلاتها الخارجية. عند ترميز البيانات، تنتج هذه الدالة قيمًا إما واحدًا أو صفرًا. تعيد هذه الدالة واحدًا إذا نجحت، وصفرًا إذا فشلت.
bool_t xdr_bytes(XDR *xdrs, char **sp, unsigned int *sizep,
                 unsigned int maxsize);

bool_t xdr_char(XDR *xdrs, char *cp);
بدائية مرشح تترجم بين أحرف C وتمثيلاتها الخارجية. تعيد هذه الدالة واحدًا إذا نجحت، وصفرًا إذا فشلت. ملاحظة: الأحرف المشفرة ليست مضغوطة، وتشغل 4 بايت لكل منها. بالنسبة لمصفوفات الأحرف، يجدر النظر في xdr_bytes() أو xdr_opaque() أو xdr_string().
void xdr_destroy(XDR *xdrs);
ماكرو يستدعي إجراء التدمير المرتبط بتيار XDR، xdrs. يتضمن التدمير عادةً تحرير هياكل البيانات الخاصة المرتبطة بالتيار. استخدام xdrs بعد استدعاء xdr_destroy() غير محدد.

void xdr_free(xdrproc_t proc, char *objp);
إجراء تحرير عام. الوسيط الأول هو إجراء XDR للكائن الذي يتم تحريره. الوسيط الثاني هو مؤشر إلى الكائن نفسه. ملاحظة: المؤشر الممرر إلى هذه الدالة لا يتم تحريره، ولكن ما يشير إليه يتم تحريره (بشكل متكرر).
unsigned int xdr_getpos(XDR *xdrs);
ماكرو يستدعي إجراء الحصول على الموضع المرتبط بتيار XDR، xdrs. تعيد الدالة عددًا صحيحًا غير موقّع، والذي يشير إلى موضع تيار بايت XDR. ميزة مرغوبة لتيارات XDR هي أن العمليات الحسابية البسيطة تعمل مع هذا الرقم، على الرغم من أن مثيلات تيار XDR قد لا تضمن ذلك.
long *xdr_inline(XDR *xdrs, int len);
ماكرو يستدعي إجراء التضمين المرتبط بتيار XDR، xdrs. تعيد الدالة مؤشرًا إلى جزء متجاور من مخزن XDR المؤقت؛ len هو طول البايت للمخزن المطلوب. ملاحظة: يتم تحويل المؤشر إلى long *.
تحذير: قد تعيد xdr_inline() قيمة NULL (0) إذا لم تتمكن من تخصيص جزء متجاور من المخزن المؤقت. لذلك قد يختلف السلوك بين مثيلات التيارات؛ إنه موجود من أجل الكفاءة.
bool_t xdr_int(XDR *xdrs, int *ip);

void xdrmem_create(XDR *xdrs, char *addr, unsigned int size,
                   enum xdr_op op);
تقوم هذه الدالة بتهيئة كائن تيار XDR الذي يشير إليه xdrs. تتم كتابة بيانات التيار، أو قراءتها، من جزء من الذاكرة في الموقع addr بطول لا يتجاوز size بايت. يحدد op اتجاه تيار XDR (إما XDR_ENCODE أو XDR_DECODE أو XDR_FREE).
bool_t xdr_opaque(XDR *xdrs, char *cp, unsigned int cnt);
بدائية مرشح تترجم بين البيانات المعتمة ذات الحجم الثابت وتمثيلها الخارجي. الوسيط cp هو عنوان الكائن المعتم، و cnt هو حجمه بالبايت. تعيد هذه الدالة واحدًا إذا نجحت، وصفرًا إذا فشلت.
bool_t xdr_pointer(XDR *xdrs, char **objpp,
                   unsigned int objsize, xdrproc_t xdrobj);
مثل xdr_reference() باستثناء أنها تقوم بتسلسل المؤشرات الفارغة، بينما xdr_reference() لا تفعل ذلك. وبالتالي، يمكن لـ xdr_pointer() تمثيل هياكل البيانات المتكررة، مثل الأشجار الثنائية أو قوائم الربط.
void xdrrec_create(XDR *xdrs, unsigned int sendsize,
                   unsigned int recvsize, char *handle,
                   typeof(int (char *, char *, int)) *readit,
                   typeof(int (char *, char *, int)) *writeit);
تقوم هذه الدالة بتهيئة كائن تيار XDR الذي يشير إليه xdrs. تتم كتابة بيانات التيار إلى مخزن مؤقت بحجم sendsize؛ قيمة الصفر تشير إلى أن النظام يجب أن يستخدم قيمة افتراضية مناسبة. تتم قراءة بيانات التيار من مخزن مؤقت بحجم recvsize؛ يمكن أيضًا تعيينه إلى قيمة افتراضية مناسبة عن طريق تمرير قيمة صفر. عندما يكون المخزن المؤقت للإخراج للتيار ممتلئًا، يتم استدعاء writeit. وبالمثل، عندما يكون المخزن المؤقت للإدخال للتيار فارغًا، يتم استدعاء readit. سلوك هاتين الدالتين مشابه لاستدعاءات النظام read(2) و write(2)، باستثناء أن handle يتم تمريره إلى الدالتين السابقتين كوسيط أول. ملاحظة: يجب على المتصل تعيين حقل op لتيار XDR.
تحذير: لقراءة تيار XDR تم إنشاؤه بواسطة واجهة برمجة التطبيقات هذه، ستحتاج إلى استدعاء xdrrec_skiprecord() أولاً قبل استدعاء أي واجهات برمجة تطبيقات XDR أخرى. هذا يضيف بايتات إضافية في التيار لتوفير حدود السجل.
bool_t xdrrec_skiprecord(XDR *xdrs);
هذه الدالة تتجاهل السجل الحالي في تيار XDR. يعيد هذا الإجراء واحدًا إذا نجح، وصفرًا إذا فشل.
bool_t xdr_reference(XDR *xdrs, char **pp, unsigned int size,
                     xdrproc_t proc);
هذه الدالة مشابهة لـ xdr_array()، ولكنها تتعامل مع مراجع للكائنات بدلاً من الكائنات نفسها. الوسيط pp هو مؤشر إلى مؤشر الكائن؛ size هو حجم الكائن؛ و proc هو إجراء XDR لترجمة الكائن. تعيد الدالة واحدًا إذا نجحت، وصفرًا إذا فشلت.
bool_t xdr_register(XDR *xdrs, void *objpp,
                    xdrproc_t proc);
لا تتوفر معلومات حول هذه الدالة.
bool_t xdr_setpos(XDR *xdrs, unsigned int pos);

bool_t xdr_void(XDR *xdrs);
بدائية مرشح لا تفعل شيئًا. غالبًا ما تستخدم كعنصر نائب لتمثيل عدم وجود بيانات.
bool_t xdr_union(XDR *xdrs, enum_t *discr, char *ptr,
                 const struct xdr_disc_byname *choices,
                 xdrproc_t defaultix);
