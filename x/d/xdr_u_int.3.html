xdr(3) دليل وظائف المكتبة xdr(3)

الاسم

xdr — روتينات مكتبة لتمثيل البيانات الخارجي

المكتبة

مكتبة C القياسية (libc، -lc)

ملخص ووصف

تسمح هذه الروتينات لمبرمجي C بوصف هياكل بيانات اعتباطية بطريقة مستقلة عن الآلة. يتم نقل البيانات لاستدعاءات الإجراءات عن بعد باستخدام هذه الروتينات.

النماذج الأولية أدناه معلنة في <rpc/xdr.h> وتستخدم الأنواع التالية:

typedef int bool_t;
typedef typeof(bool_t (XDR *, void *, ...))  *xdrproc_t;

لإعلان نوع XDR، انظر <rpc/xdr.h>.

bool_t xdr_array(XDR *xdrs, char **arrp, unsigned int *sizep,
                 unsigned int maxsize, unsigned int elsize,
                 xdrproc_t elproc);
مرشح بدائي يقوم بالترجمة بين المصفوفات متغيرة الطول وتمثيلها الخارجي المقابل. الوسيط arrp هو عنوان المؤشر إلى المصفوفة، بينما sizep هو عنوان عداد العناصر في المصفوفة؛ لا يمكن أن يتجاوز هذا العداد maxsize. الوسيط elsize هو sizeof كل عنصر من عناصر المصفوفة، و elproc هو مرشح XDR يقوم بالترجمة بين صيغة C لعناصر المصفوفة وتمثيلها الخارجي. تعيد هذه الروتينات واحدًا إذا نجحت، وصفرًا بخلاف ذلك.
bool_t xdr_bool(XDR *xdrs, bool_t *bp);
مرشح بدائي يقوم بالترجمة بين القيم المنطقية (أعداد صحيحة C) وتمثيلها الخارجي. عند ترميز البيانات، تنتج هذه المرشحة قيمًا إما واحدًا أو صفرًا. تعيد هذه الروتينات واحدًا إذا نجحت، وصفرًا بخلاف ذلك.
bool_t xdr_bytes(XDR *xdrs, char **sp, unsigned int *sizep,
                 unsigned int maxsize);
مرشح بدائي يقوم بالترجمة بين سلاسل البايت المعدودة وتمثيلها الخارجي. الوسيط sp هو عنوان مؤشر السلسلة. يوجد طول السلسلة في العنوان sizep؛ لا يمكن أن تكون السلاسل أطول من maxsize. تعيد هذه الروتينات واحدًا إذا نجحت، وصفرًا بخلاف ذلك.
bool_t xdr_char(XDR *xdrs, char *cp);
مرشح بدائي يقوم بالترجمة بين أحرف C وتمثيلها الخارجي. تعيد هذه الروتينات واحدًا إذا نجحت، وصفرًا بخلاف ذلك. ملاحظة: الأحرف المشفرة غير مضغوطة، وتشغل 4 بايتات لكل منها. بالنسبة لمصفوفات الأحرف، يجدر التفكير في xdr_bytes() أو xdr_opaque() أو xdr_string().
void xdr_destroy(XDR *xdrs);
ماكرو يستدعي روتين التدمير المرتبط بتيار XDR، xdrs. يتضمن التدمير عادةً تحرير هياكل البيانات الخاصة المرتبطة بالتيار. استخدام xdrs بعد استدعاء xdr_destroy() غير محدد.
bool_t xdr_double(XDR *xdrs, double *dp);
مرشح بدائي يقوم بالترجمة بين أعداد double الدقة في C وتمثيلها الخارجي. تعيد هذه الروتينات واحدًا إذا نجحت، وصفرًا بخلاف ذلك.
bool_t xdr_enum(XDR *xdrs, enum_t *ep);
مرشح بدائي يقوم بالترجمة بين enums في C (في الواقع أعداد صحيحة) وتمثيلها الخارجي. تعيد هذه الروتينات واحدًا إذا نجحت، وصفرًا بخلاف ذلك.
bool_t xdr_float(XDR *xdrs, float *fp);
مرشح بدائي يقوم بالترجمة بين floats في C وتمثيلها الخارجي. تعيد هذه الروتينات واحدًا إذا نجحت، وصفرًا بخلاف ذلك.
void xdr_free(xdrproc_t proc, char *objp);
روتين تحرير عام. الوسيط الأول هو روتين XDR للكائن الذي يتم تحريره. الوسيط الثاني هو مؤشر إلى الكائن نفسه. ملاحظة: المؤشر الذي تم تمريره إلى هذا الروتين لا يتم تحريره، ولكن ما يشير إليه يتم تحريره (بشكل متكرر).
unsigned int xdr_getpos(XDR *xdrs);
ماكرو يستدعي روتين الحصول على الموضع المرتبط بتيار XDR، xdrs. يعيد الروتين عددًا صحيحًا غير موقع، والذي يشير إلى موضع تيار بايت XDR. ميزة مرغوبة لتيارات XDR هي أن العمليات الحسابية البسيطة تعمل مع هذا الرقم، على الرغم من أن مثيلات تيار XDR لا تحتاج إلى ضمان ذلك.
long *xdr_inline(XDR *xdrs, int len);
ماكرو يستدعي روتين الاستدعاء الخطي المرتبط بتيار XDR، xdrs. يعيد الروتين مؤشرًا إلى قطعة متجاورة من مخزن تيار XDR؛ len هو طول البايت المطلوب للمخزن. ملاحظة: يتم تحويل المؤشر إلى long *.
تحذير: قد يعيد xdr_inline() قيمة NULL (0) إذا لم يتمكن من تخصيص قطعة متجاورة من المخزن. لذلك قد يختلف السلوك بين مثيلات التيار؛ وهو موجود من أجل الكفاءة.
bool_t xdr_int(XDR *xdrs, int *ip);
مرشح بدائي يقوم بالترجمة بين الأعداد الصحيحة في C وتمثيلها الخارجي. تعيد هذه الروتينات واحدًا إذا نجحت، وصفرًا بخلاف ذلك.
bool_t xdr_long(XDR *xdrs, long *lp);
مرشح بدائي يقوم بالترجمة بين long الأعداد الصحيحة في C وتمثيلها الخارجي. تعيد هذه الروتينات واحدًا إذا نجحت، وصفرًا بخلاف ذلك.
void xdrmem_create(XDR *xdrs, char *addr, unsigned int size,
                   enum xdr_op op);
يقوم هذا الروتين بتهيئة كائن تيار XDR الذي يشير إليه xdrs. يتم كتابة بيانات التيار، أو قراءتها، من جزء من الذاكرة في الموقع addr والذي لا يتجاوز طوله size بايت. يحدد op اتجاه تيار XDR (إما XDR_ENCODE أو XDR_DECODE أو XDR_FREE).
bool_t xdr_opaque(XDR *xdrs, char *cp, unsigned int cnt);
مرشح بدائي يقوم بالترجمة بين البيانات الغامضة ذات الحجم الثابت وتمثيلها الخارجي. الوسيط cp هو عنوان الكائن الغامض، و cnt هو حجمه بالبايت. تعيد هذه الروتينات واحدًا إذا نجحت، وصفرًا بخلاف ذلك.
bool_t xdr_pointer(XDR *xdrs, char **objpp,
                   unsigned int objsize, xdrproc_t xdrobj);
مثل xdr_reference() باستثناء أنه يقوم بتسلسل المؤشرات الفارغة، بينما xdr_reference() لا يفعل ذلك. وبالتالي، يمكن لـ xdr_pointer() تمثيل هياكل البيانات المتكررة، مثل الأشجار الثنائية أو القوائم المرتبطة.
void xdrrec_create(XDR *xdrs, unsigned int sendsize,
                   unsigned int recvsize, char *handle,
                   typeof(int (char *, char *, int)) *readit,
                   typeof(int (char *, char *, int)) *writeit);
يقوم هذا الروتين بتهيئة كائن تيار XDR الذي يشير إليه xdrs. يتم كتابة بيانات التيار إلى مخزن بحجم sendsize؛ تشير قيمة الصفر إلى أن النظام يجب أن يستخدم قيمة افتراضية مناسبة. يتم قراءة بيانات التيار من مخزن بحجم recvsize؛ يمكن أيضًا تعيينه إلى قيمة افتراضية مناسبة عن طريق تمرير قيمة صفر. عندما يكون مخزن الإخراج للتيار ممتلئًا، يتم استدعاء writeit. وبالمثل، عندما يكون مخزن الإدخال للتيار فارغًا، يتم استدعاء readit. سلوك هذين الروتينين مشابه لاستدعاءات النظام read(2) و write(2)، باستثناء أن handle يتم تمريره إلى الروتينات السابقة كوسيط أول. ملاحظة: يجب على المتصل تعيين حقل op لتيار XDR.
تحذير: لقراءة تيار XDR تم إنشاؤه بواسطة واجهة برمجة التطبيقات هذه، ستحتاج إلى استدعاء xdrrec_skiprecord() أولاً قبل استدعاء أي واجهات برمجة تطبيقات XDR أخرى. هذا يضيف بايتات إضافية في التيار لتوفير حدود السجل.

NAME

xdr_u_int, xdr_u_long, xdr_u_short, xdr_vector, xdr_void, xdr_wrapstring, xdrrec_endofrecord, xdrrec_eof, xdrrec_skiprecord, xdr_reference, xdr_setpos, xdr_short, xdrstdio_create, xdr_string, xdr_u_char, xdr_union - XDR filter primitives

LIBRARY

libtirpc (or libc)

SYNOPSIS AND DESCRIPTION

This manual page describes a number of XDR (External Data Representation) filter primitives. These routines are used by the RPC (Remote Procedure Call) package to translate C data types between their in-memory representation and their standard external representation. These routines are part of the XDR library.

bool_t xdr_array(XDR *xdrs, char **cpp, unsigned int *freelp, unsigned int size, unsigned int elsize, xdrproc_t elproc);
A filter primitive that translates between C arrays and their corresponding external representations. The argument cpp is the address of the pointer to the array, while freelp is the address of the number of elements in the array. The argument size is the maximum number of elements the array can hold, elsize is the sizeof each element, and elproc is an XDR filter that translates between the array elements' C form, and their external representation. This routine returns one if it succeeds, zero otherwise.
bool_t xdr_bool(XDR *xdrs, bool_t *bp);
A filter primitive that translates between C bool_t values and their external representations. This routine returns one if it succeeds, zero otherwise.
bool_t xdr_bytes(XDR *xdrs, char **sp, unsigned int *lenp, unsigned int maxsize);
A filter primitive that translates between C sequences of bytes (character arrays) and their corresponding external representations. The argument sp is the address of the pointer to the byte sequence, lenp is the address of the length of the byte sequence, and maxsize is the maximum length the byte sequence can have. This routine returns one if it succeeds, zero otherwise.
bool_t xdr_char(XDR *xdrs, char *cp);
A filter primitive that translates between C characters and their external representations. This routine returns one if it succeeds, zero otherwise.
void xdr_destroy(XDR *xdrs);
This routine deallocates the XDR stream structure pointed to by xdrs, and any associated buffers or resources. It should be called when an XDR stream is no longer needed.
bool_t xdr_double(XDR *xdrs, double *dp);
A filter primitive that translates between C double precision floating-point numbers and their external representations. This routine returns one if it succeeds, zero otherwise.
bool_t xdr_enum(XDR *xdrs, enum_t *ep);
A filter primitive that translates between C enumerated types (enum_t) and their external representations. This routine returns one if it succeeds, zero otherwise.
bool_t xdr_float(XDR *xdrs, float *fp);
A filter primitive that translates between C float precision floating-point numbers and their external representations. This routine returns one if it succeeds, zero otherwise.
void xdr_free(xdrproc_t proc, char *objp);
This routine calls the specified XDR procedure proc to free the memory associated with the XDR object objp. This is typically used after decoding data that was dynamically allocated.
unsigned int xdr_getpos(XDR *xdrs);
A macro that returns the current position in the XDR stream xdrs. This position value can be used later with xdr_setpos() to reposition the stream.
Warning: not all XDR stream types support positioning. This routine may return 0 or have other undefined behavior for unsupported stream types.
int xdr_inline(XDR *xdrs, unsigned int len);
A primitive that returns a pointer to a buffer of length len within the XDR stream xdrs. This buffer can be used to directly read or write data. Note that the buffer is only guaranteed to be valid until the next XDR operation. This routine is primarily used for optimizing the transfer of large data blocks. It returns a pointer to the buffer if successful, and NULL otherwise.
bool_t xdr_int(XDR *xdrs, int *ip);
A filter primitive that translates between C integers and their external representations. This routine returns one if it succeeds, zero otherwise.
bool_t xdr_long(XDR *xdrs, long *lp);
A filter primitive that translates between C long integers and their external representations. This routine returns one if it succeeds, zero otherwise.
void xdrmem_create(XDR *xdrs, const char *addr, unsigned int len, enum xdr_op op);
This routine initializes the XDR stream object pointed to by xdrs to operate on a memory buffer. The memory buffer is specified by addr and len. The argument op determines the direction of the XDR stream (either XDR_ENCODE, XDR_DECODE, or XDR_FREE).
bool_t xdr_opaque(XDR *xdrs, char *cp, unsigned int cnt);
A filter primitive that translates between arbitrary data buffers (opaque data) and their external representations. The argument cp is the buffer, and cnt is the number of bytes to translate. This routine returns one if it succeeds, zero otherwise.
bool_t xdr_pointer(XDR *xdrs, char **pp, unsigned int size,
                   xdrproc_t proc);
A primitive that handles XDR filtering of pointers. The argument pp is the address of the pointer; size is the size of the object pointed to; and proc is an XDR procedure that filters the object between its C form and its external representation. This routine handles NULL pointers correctly. It returns one if it succeeds, zero otherwise.
XDR *xdrrec_create(XDR *xdrs, unsigned int buflen, char *morecore,
                   char *basep, int (*readit)(), int (*writeit)());
This routine initializes the XDR stream structure pointed to by xdrs to use record-oriented stream semantics. The stream operates on a buffer of size buflen. The morecore argument is a pointer to a routine that is called to get more buffer space when needed. The basep argument is the base pointer of the buffer. The readit and writeit arguments are pointers to routines that handle reading from and writing to the underlying communication channel, respectively. This routine returns a pointer to the initialized XDR stream structure, or NULL on failure.
Warning: XDR streams created with xdrrec_create() are not compatible with streams created with other XDR creation routines (e.g., xdrstdio_create(), xdrmem_create()) because they use different internal buffering and record-marking mechanisms. All XDR operations on a record-oriented stream must be performed using the record-oriented XDR routines described below. Also, XDR streams created with different xdr*_create APIs are not compatible for the same reason.
bool_t xdrrec_endofrecord(XDR *xdrs, int sendnow);
This routine can be invoked only on streams created by xdrrec_create(). The data in the output buffer is marked as a completed record, and the output buffer is optionally written out if sendnow is nonzero. This routine returns one if it succeeds, zero otherwise.
bool_t xdrrec_eof(XDR *xdrs);
This routine can be invoked only on streams created by xdrrec_create(). After consuming the rest of the current record in the stream, this routine returns one if the stream has no more input, zero otherwise.
bool_t xdrrec_skiprecord(XDR *xdrs);
This routine can be invoked only on streams created by xdrrec_create(). It tells the XDR implementation that the rest of the current record in the stream's input buffer should be discarded. This routine returns one if it succeeds, zero otherwise.
bool_t xdr_reference(XDR *xdrs, char **pp, unsigned int size,
                     xdrproc_t proc);
A primitive that provides pointer chasing within structures. The argument pp is the address of the pointer; size is the sizeof the structure that *pp points to; and proc is an XDR procedure that filters the structure between its C form and its external representation. This routine returns one if it succeeds, zero otherwise.
Warning: this routine does not understand null pointers. Use xdr_pointer() instead.
xdr_setpos(XDR *xdrs, unsigned int pos);
A macro that invokes the set position routine associated with the XDR stream xdrs. The argument pos is a position value obtained from xdr_getpos(). This routine returns one if the XDR stream could be repositioned, and zero otherwise.
Warning: it is difficult to reposition some types of XDR streams, so this routine may fail with one type of stream and succeed with another.
bool_t xdr_short(XDR *xdrs, short *sp);
A filter primitive that translates between C short integers and their external representations. This routine returns one if it succeeds, zero otherwise.
void xdrstdio_create(XDR *xdrs, FILE *file, enum xdr_op op);
This routine initializes the XDR stream object pointed to by xdrs. The XDR stream data is written to, or read from, the stdio stream file. The argument op determines the direction of the XDR stream (either XDR_ENCODE, XDR_DECODE, or XDR_FREE).
Warning: the destroy routine associated with such XDR streams calls fflush(3) on the file stream, but never fclose(3).
bool_t xdr_string(XDR *xdrs, char **sp, unsigned int maxsize);
A filter primitive that translates between C strings and their corresponding external representations. Strings cannot be longer than maxsize. Note: sp is the address of the string's pointer. This routine returns one if it succeeds, zero otherwise.
bool_t xdr_u_char(XDR *xdrs, unsigned char *ucp);
A filter primitive that translates between unsigned C characters and their external representations. This routine returns one if it succeeds, zero otherwise.
bool_t xdr_u_int(XDR *xdrs, unsigned int *up);
A filter primitive that translates between C unsigned integers and their external representations. This routine returns one if it succeeds, zero otherwise.
bool_t xdr_u_long(XDR *xdrs, unsigned long *ulp);
A filter primitive that translates between C unsigned long integers and their external representations. This routine returns one if it succeeds, zero otherwise.
bool_t xdr_u_short(XDR *xdrs, unsigned short *usp);
A filter primitive that translates between C unsigned short integers and their external representations. This routine returns one if it succeeds, zero otherwise.
bool_t xdr_union(XDR *xdrs, enum_t *dscmp, char *unp,
                 const struct xdr_discrim *choices,
                 xdrproc_t defaultarm);     /* may equal NULL */
A filter primitive that translates between a discriminated C union and its corresponding external representation. It first translates the discriminant of the union located at dscmp. This discriminant is always an enum_t. Next the union located at unp is translated. The argument choices is a pointer to an array of xdr_discrim() structures. Each structure contains an ordered pair of [value,proc]. If the union's discriminant is equal to the associated value, then the proc is called to translate the union. The end of the xdr_discrim() structure array is denoted by a routine of value NULL. If the discriminant is not found in the choices array, then the defaultarm procedure is called (if it is not NULL). Returns one if it succeeds, zero otherwise.
bool_t xdr_vector(XDR *xdrs, char *arrp, unsigned int size,
                  unsigned int elsize, xdrproc_t elproc);
A filter primitive that translates between fixed-length arrays and their corresponding external representations. The argument arrp is the address of the pointer to the array, while size is the element count of the array. The argument elsize is the sizeof each of the array's elements, and elproc is an XDR filter that translates between the array elements' C form, and their external representation. This routine returns one if it succeeds, zero otherwise.
bool_t xdr_void(void);
This routine always returns one. It may be passed to RPC routines that require a function argument, where nothing is to be done.
bool_t xdr_wrapstring(XDR *xdrs, char **sp);
A primitive that calls xdr_string(xdrs, sp,MAXUN.UNSIGNED ); where MAXUN.UNSIGNED is the maximum value of an unsigned integer. xdr_wrapstring() is handy because the RPC package passes a maximum of two XDR routines as arguments, and xdr_string(), one of the most frequently used primitives, requires three. Returns one if it succeeds, zero otherwise.
