rpc(3) دليل وظائف المكتبة rpc(3)

الاسم

rpc - روتينيات مكتبة لطلبات الإجراءات عن بعد

المكتبة

مكتبة C القياسية (libc، -lc)

الملخص والوصف

تسمح هذه الروتينيات لبرامج C بإجراء مكالمات إجراءات على أجهزة أخرى عبر الشبكة. أولاً، يستدعي العميل إجراءً لإرسال حزمة بيانات إلى الخادم. عند استلام الحزمة، يستدعي الخادم روتين إرسال لأداء الخدمة المطلوبة، ثم يرسل رداً. أخيرًا، يعود طلب الإجراء إلى العميل.

للاستفادة من هذه الروتينيات، قم بتضمين ملف الرأس <rpc/rpc.h>.

تستخدم التعريفات أدناه الأنواع التالية:

typedef int bool_t;
typedef typeof(bool_t (XDR *, void *, ...))  *xdrproc_t;
typedef typeof(bool_t (caddr_t resp, struct sockaddr_in *raddr)
    *resultproc_t;

راجع ملفات الرأس لتصريحات أنواع AUTH، CLIENT، SVCXPRT، و XDR.

void auth_destroy(AUTH *auth);
وحدة ماكرو تدمر معلومات المصادقة المرتبطة بـ auth. عادةً ما يتضمن التدمير تخصيص هياكل البيانات الخاصة. استخدام auth غير معرف بعد استدعاء auth_destroy().
AUTH *authnone_create(void);
ينشئ ويعيد مقبض مصادقة RPC يمرر معلومات مصادقة غير قابلة للاستخدام مع كل طلب إجراء عن بعد. هذه هي المصادقة الافتراضية التي تستخدمها RPC.
AUTH *authunix_create(char *host, uid_t uid, gid_t gid,
                      int n, gid_t aup_gids[n]);
ينشئ ويعيد مقبض مصادقة RPC يحتوي على معلومات المصادقة. المعلمة host هي اسم الجهاز الذي تم إنشاء المعلومات عليه؛ uid هو معرف المستخدم؛ gid هو معرف المجموعة الحالي للمستخدم؛ n و aup_gids تشير إلى مصفوفة محسوبة بالمجموعات التي ينتمي إليها المستخدم. من السهل انتحال شخصية مستخدم.
AUTH *authunix_create_default(void);
يستدعي authunix_create() بالمعلمات المناسبة.
int callrpc(char *host, unsigned long prognum,
            unsigned long versnum, unsigned long procnum,
            xdrproc_t inproc, const char *in,
            xdrproc_t outproc, char *out);
يستدعي الإجراء البعيد المرتبط بـ prognum و versnum و procnum على الجهاز host. المعلمة in هي عنوان الوسيط (الوسائط) للإجراء، و out هي العنوان الذي سيتم فيه وضع النتائج؛ inproc يستخدم لتشفير معاملات الإجراء، و outproc يستخدم لفك تشفير نتائج الإجراء. يعيد هذا الروتين صفرًا إذا نجح، أو قيمة enum clnt_stat محولة إلى عدد صحيح إذا فشل. الروتين clnt_perrno() مفيد لترجمة حالات الفشل إلى رسائل.
تحذير: استخدام الإجراءات البعيدة بهذا الروتين يستخدم UDP/IP كنقل؛ راجع clntudp_create() للقيود. ليس لديك تحكم في المهلات أو المصادقة باستخدام هذا الروتين.
enum clnt_stat clnt_broadcast(unsigned long prognum,
                     unsigned long versnum, unsigned long procnum,
                     xdrproc_t inproc, char *in,
                     xdrproc_t outproc, char *out,
                     resultproc_t eachresult);
مثل callrpc()، باستثناء أن رسالة الاستدعاء يتم بثها إلى جميع شبكات البث المتصلة محليًا. في كل مرة يتلقى فيها رداً، يستدعي هذا الروتين eachresult()، والذي يكون شكله:

eachresult(char *out, struct sockaddr_in *addr);

حيث out هو نفس out الذي تم تمريره إلى clnt_broadcast()، باستثناء أن نتائج الإجراء البعيد يتم فك تشفيرها هناك؛ addr يشير إلى عنوان الجهاز الذي أرسل النتائج. إذا أعادت eachresult() صفرًا، ينتظر clnt_broadcast() لمزيد من الردود؛ وإلا فإنه يعود بحالة مناسبة.
تحذير: مقابس البث محدودة الحجم كحد أقصى لوحدة النقل للارتباط البيانات. بالنسبة لـ ethernet، هذه القيمة هي 1500 بايت.
enum clnt_stat clnt_call(CLIENT *clnt, unsigned long procnum,
                    xdrproc_t inproc, char *in,
                    xdrproc_t outproc, char *out,
                    struct timeval tout);
وحدة ماكرو تستدعي الإجراء البعيد procnum المرتبط بمقبض العميل clnt، والذي يتم الحصول عليه باستخدام روتين إنشاء عميل RPC مثل clnt_create(). المعلمة in هي عنوان الوسيط (الوسائط) للإجراء، و out هي العنوان الذي سيتم فيه وضع النتائج؛ inproc يستخدم لتشفير معاملات الإجراء، و outproc يستخدم لفك تشفير نتائج الإجراء؛ tout هو الوقت المسموح به لعودة النتائج.
clnt_destroy(CLIENT *clnt);
وحدة ماكرو تدمر مقبض RPC الخاص بالعميل. يتضمن التدمير عادةً تخصيص هياكل البيانات الخاصة، بما في ذلك clnt نفسه. استخدام clnt غير معرف بعد استدعاء clnt_destroy(). إذا فتحت مكتبة RPC المقبس المرتبط، فستقوم بإغلاقه أيضًا. وإلا، يظل المقبس مفتوحًا.
CLIENT *clnt_create(const char *host, unsigned long prog,
                    unsigned long vers, const char *proto);
روتين إنشاء عميل عام. host يحدد اسم المضيف البعيد حيث يوجد الخادم. proto يشير إلى نوع بروتوكول النقل الذي سيتم استخدامه. القيم المدعومة حاليًا لهذا الحقل هي "udp" و "tcp". يتم تعيين مهلات افتراضية، ولكن يمكن تعديلها باستخدام clnt_control().
تحذير: استخدام UDP له عيوبه. نظرًا لأن رسائل RPC المستندة إلى UDP يمكن أن تحتوي على 8 كيلوبايت فقط من البيانات المشفرة، لا يمكن استخدام هذا النقل للإجراءات التي تأخذ وسائط كبيرة أو تعيد نتائج ضخمة.
bool_t clnt_control(CLIENT *cl, int req, char *info);
وحدة ماكرو تستخدم لتغيير أو استرداد معلومات مختلفة حول كائن عميل. req يشير إلى نوع العملية، و info هو مؤشر إلى المعلومات. لكل من UDP و TCP، القيم المدعومة لـ req وأنواع وسائطها وماذا تفعل هي:

CLSET_TIMEOUT  struct timeval // تعيين مهلة إجمالية
CLGET_TIMEOUT  struct timeval // الحصول على مهلة إجمالية

ملاحظة: إذا قمت بتعيين المهلة باستخدام clnt_control()، فسيتم تجاهل المعلمة tout التي تم تمريرها إلى clnt_call() في جميع الاستدعاءات المستقبلية.

CLGET_SERVER_ADDR  struct sockaddr_in
                // الحصول على عنوان الخادم

العمليات التالية صالحة لـ UDP فقط:

CLSET_RETRY_TIMEOUT  struct timeval // تعيين مهلة إعادة المحاولة
CLGET_RETRY_TIMEOUT  struct timeval // الحصول على مهلة إعادة المحاولة

مهلة إعادة المحاولة هي الوقت الذي ينتظره "UDP RPC" من الخادم للرد قبل إعادة إرسال الطلب.
clnt_freeres(CLIENT * clnt, xdrproc_t outproc, char *out);
وحدة ماكرو تقوم بتحرير أي بيانات تم تخصيصها بواسطة نظام RPC/XDR عند فك تشفير نتائج استدعاء RPC. المعلمة out هي عنوان النتائج، و outproc هو روتين XDR الذي يصف النتائج. يعيد هذا الروتين واحدًا إذا تم تحرير النتائج بنجاح، وصفرًا بخلاف ذلك.
void clnt_geterr(CLIENT *clnt, struct rpc_err
