الاسم

xdr — روتينات مكتبة لتمثيل البيانات الخارجية

المكتبة

مكتبة C القياسية (libc، -lc)

الملخص والوصف

تسمح هذه الروتينات لمبرمجي C بوصف هياكل بيانات اعتباطية بطريقة مستقلة عن الجهاز. يتم نقل البيانات لاستدعاءات الإجراءات البعيدة باستخدام هذه الروتينات.

النماذج الأولية أدناه معلنة في <rpc/xdr.h> وتستخدم الأنواع التالية:

typedef int bool_t;
typedef typeof(bool_t (XDR *, void *, ...))  *xdrproc_t;

لإعلان نوع XDR، راجع <rpc/xdr.h>.

bool_t xdr_array(XDR *xdrs, char **arrp, unsigned int *sizep,
                 unsigned int maxsize, unsigned int elsize,
                 xdrproc_t elproc);
مرشح أساسي يترجم بين المصفوفات ذات الطول المتغير وتمثيلاتها الخارجية المقابلة. الوسيط arrp هو عنوان المؤشر إلى المصفوفة، بينما sizep هو عنوان عدد العناصر في المصفوفة؛ لا يمكن أن يتجاوز عدد العناصر هذا maxsize. الوسيط elsize هو sizeof كل عنصر من عناصر المصفوفة، و elproc هو مرشح XDR يترجم بين شكل C لعناصر المصفوفة وتمثيلها الخارجي. تعيد هذه الروتينات واحدًا إذا نجحت، وصفرًا بخلاف ذلك.
bool_t xdr_bool(XDR *xdrs, bool_t *bp);
مرشح أساسي يترجم بين القيم المنطقية (أعداد صحيحة C) وتمثيلاتها الخارجية. عند ترميز البيانات، تنتج هذه المرشحات قيمًا إما واحدًا أو صفرًا. تعيد هذه الروتينات واحدًا إذا نجحت، وصفرًا بخلاف ذلك.
bool_t xdr_bytes(XDR *xdrs, char **sp, unsigned int *sizep,
                 unsigned int maxsize);
مرشح أساسي يترجم بين سلاسل البايت المعدودة وتمثيلاتها الخارجية. الوسيط sp هو عنوان مؤشر السلسلة. يقع طول السلسلة في العنوان sizep؛ لا يمكن أن تكون السلاسل أطول من maxsize. تعيد هذه الروتينات واحدًا إذا نجحت، وصفرًا بخلاف ذلك.
bool_t xdr_char(XDR *xdrs, char *cp);
مرشح أساسي يترجم بين أحرف C وتمثيلاتها الخارجية. تعيد هذه الروتينات واحدًا إذا نجحت، وصفرًا بخلاف ذلك. ملاحظة: الأحرف المرمزة ليست مضغوطة، وتشغل 4 بايت لكل منها. بالنسبة لمصفوفات الأحرف، يجدر النظر في xdr_bytes() أو xdr_opaque() أو xdr_string().
void xdr_destroy(XDR *xdrs);
وحدة ماكرو تستدعي روتين الإتلاف المرتبط بتيار XDR، xdrs. يشمل الإتلاف عادةً تحرير هياكل البيانات الخاصة المرتبطة بالتيار. استخدام xdrs بعد استدعاء xdr_destroy() غير محدد.
bool_t xdr_double(XDR *xdrs, double *dp);
مرشح أساسي يترجم بين أرقام double C ودقتها وتمثيلاتها الخارجية. تعيد هذه الروتينات واحدًا إذا نجحت، وصفرًا بخلاف ذلك.
bool_t xdr_enum(XDR *xdrs, enum_t *ep);

bool_t xdr_float(XDR *xdrs, float *fp);
مرشح أساسي يترجم بين float C وتمثيلاتها الخارجية. تعيد هذه الروتينات واحدًا إذا نجحت، وصفرًا بخلاف ذلك.
void xdr_free(xdrproc_t proc, char *objp);
روتين تحرير عام. الوسيط الأول هو روتين XDR للكائن الذي يتم تحريره. الوسيط الثاني هو مؤشر إلى الكائن نفسه. ملاحظة: المؤشر الذي يتم تمريره إلى هذا الروتين *لا* يتم تحريره، ولكن ما يشير إليه *يتم* تحريره (بشكل متكرر).
unsigned int xdr_getpos(XDR *xdrs);
وحدة ماكرو تستدعي روتين الحصول على الموضع المرتبط بتيار XDR، xdrs. يعيد الروتين عددًا صحيحًا غير موقع، والذي يشير إلى موضع تيار بايت XDR. ميزة مرغوبة لتيارات XDR هي أن العمليات الحسابية البسيطة تعمل مع هذا الرقم، على الرغم من أن مثيلات تيار XDR قد لا تضمن ذلك.
long *xdr_inline(XDR *xdrs, int len);
وحدة ماكرو تستدعي روتين الاستدعاء المتوازي المرتبط بتيار XDR، xdrs. يعيد الروتين مؤشرًا إلى قطعة متجاورة من مخزن مؤقت للتيار؛ len هو طول البايت للمخزن المؤقت المطلوب. ملاحظة: يتم تحويل المؤشر إلى long *.
تحذير: قد يعيد xdr_inline() القيمة NULL (0) إذا لم يتمكن من تخصيص قطعة متجاورة من مخزن مؤقت. لذلك قد يختلف السلوك بين مثيلات التيار؛ وهو موجود لتحقيق الكفاءة.
bool_t xdr_int(XDR *xdrs, int *ip);
مرشح أساسي يترجم بين أعداد C الصحيحة وتمثيلاتها الخارجية. تعيد هذه الروتينات واحدًا إذا نجحت، وصفرًا بخلاف ذلك.
bool_t xdr_long(XDR *xdrs, long *lp);
مرشح أساسي يترجم بين long C الصحيحة وتمثيلاتها الخارجية. تعيد هذه الروتينات واحدًا إذا نجحت، وصفرًا بخلاف ذلك.
void xdrmem_create(XDR *xdrs, char *addr, unsigned int size,
                   enum xdr_op op);
يقوم هذا الروتين بتهيئة كائن تيار XDR الذي يشير إليه xdrs. تتم كتابة بيانات التيار، أو قراءتها، من جزء من الذاكرة في الموقع addr والذي لا يتجاوز طوله size بايت. تحدد op اتجاه تيار XDR (إما XDR_ENCODE أو XDR_DECODE أو XDR_FREE).
bool_t xdr_opaque(XDR *xdrs, char *cp, unsigned int cnt);
مرشح أساسي يترجم بين البيانات المعتمة ذات الحجم الثابت وتمثيلها الخارجي. الوسيط cp هو عنوان الكائن المعتم، و cnt هو حجمه بالبايت. تعيد هذه الروتينات واحدًا إذا نجحت، وصفرًا بخلاف ذلك.
bool_t xdr_pointer(XDR *xdrs, char **objpp,
                   unsigned int objsize, xdrproc_t xdrobj);
مثل xdr_reference() باستثناء أنه يقوم بتسلسل المؤشرات الفارغة، بينما xdr_reference() لا يفعل ذلك. وبالتالي، يمكن لـ xdr_pointer() تمثيل هياكل البيانات المتكررة، مثل الأشجار الثنائية أو القوائم المترابطة.
void xdrrec_create(XDR *xdrs, unsigned int sendsize,
                   unsigned int recvsize, char *handle,
                   typeof(int (char *, char *, int)) *readit,
                   typeof(int (char *, char *, int)) *writeit);
يقوم هذا الروتين بتهيئة كائن تيار XDR الذي يشير إليه xdrs. تتم كتابة بيانات التيار إلى مخزن مؤقت بحجم sendsize؛ تشير القيمة صفر إلى أن النظام يجب أن يستخدم افتراضيًا مناسبًا. تتم قراءة بيانات التيار من مخزن مؤقت بحجم recvsize؛ يمكن أيضًا تعيين هذا افتراضيًا مناسبًا عن طريق تمرير قيمة صفر. عندما يكون مخزن الإخراج المؤقت للتيار ممتلئًا، يتم استدعاء writeit. وبالمثل، عندما يكون مخزن الإدخال المؤقت للتيار فارغًا، يتم استدعاء readit. سلوك هذين الروتينين مشابه لاستدعاءات النظام read(2) و write(2)، باستثناء أن handle يتم تمريره إلى الروتينات السابقة كوسيط أول. ملاحظة: يجب على المتصل تعيين حقل op لتيار XDR.
تحذير: لقراءة تيار XDR تم إنشاؤه بواسطة واجهة برمجة التطبيقات هذه، ستحتاج إلى استدعاء xdrrec_skiprecord() أولاً قبل استدعاء أي واجهات برمجة تطبيقات XDR أخرى. هذا يدرج بايتات إضافية في التيار لتوفير حدود السجل
bool_t xdr_reference(XDR *xdrs, char **pp, unsigned int size,
                    xdrproc_t proc);
يستخدم هذا المرشح لترجمة هياكل البيانات غير الحجمية. الوسيط pp هو عنوان المؤشر إلى الكائن المراد ترجمته. الوسيط size هو حجم الكائن بالبايت. الوسيط proc هو روتين XDR الذي يترجم بين تمثيل C للكائن وتمثيله الخارجي. تعيد هذه الروتينات واحدًا إذا نجحت، وصفرًا بخلاف ذلك.
bool_t xdr_setpos(XDR *xdrs, unsigned int pos);
وحدة ماكرو تستدعي روتين تعيين الموضع المرتبط بتيار XDR، xdrs. تعيد الروتينات واحدًا إذا نجحت، وصفرًا بخلاف ذلك. يتم تعيين موضع تيار البايت XDR إلى pos، والذي تم الحصول عليه مسبقًا من استدعاء سابق لـ xdr_getpos().
bool_t xdr_short(XDR *xdrs, short *sp);
مرشح أساسي يترجم بين short C الصحيحة وتمثيلاتها الخارجية. تعيد هذه الروتينات واحدًا إذا نجحت، وصفرًا بخلاف ذلك.
bool_t xdr_skiprecord(XDR *xdrs);
تستهلك هذه الروتينات سجل XDR بأكمله. إذا كان التيار يعمل في وضع التسجيل، فإنه يتجاهل كل البايتات حتى نهاية السجل. إذا كان التيار يعمل في وضع البايت، فلا تفعل شيئًا. تعيد هذه الروتينات واحدًا إذا نجحت، وصفرًا بخلاف ذلك. ملاحظة: يجب استخدام هذا الروتين على التيارات التي تم إنشاؤها بواسطة xdrrec_create().
bool_t xdr_string(XDR *xdrs, char **sp, unsigned int maxsize);
مرشح أساسي يترجم بين سلاسل الأحرف الموجهة NULL وتمثيلاتها الخارجية. الوسيط sp هو عنوان مؤشر السلسلة، و maxsize هو أقصى طول مسموح به للسلسلة (باستثناء الحرف NULL النهائي). تعيد هذه الروتينات واحدًا إذا نجحت، وصفرًا بخلاف ذلك.
bool_t xdr_u_char(XDR *xdrs, unsigned char *ucp);
مرشح أساسي يترجم بين أحرف C غير الموقعة وتمثيلاتها الخارجية. تعيد هذه الروتينات واحدًا إذا نجحت، وصفرًا بخلاف ذلك.
bool_t xdr_u_int(XDR *xdrs, unsigned int *up);
مرشح أساسي يترجم بين الأعداد الصحيحة غير الموقعة C وتمثيلاتها الخارجية. تعيد هذه الروتينات واحدًا إذا نجحت، وصفرًا بخلاف ذلك.
bool_t xdr_u_long(XDR *xdrs, unsigned long *ulp);
مرشح أساسي يترجم بين long C الصحيحة غير الموقعة وتمثيلاتها الخارجية. تعيد هذه الروتينات واحدًا إذا نجحت، وصفرًا بخلاف ذلك.
bool_t xdr_union(XDR *xdrs, enum_t *decide_ptr, char *handle,
                 const struct xdr_discrim *choices,
                 typeof(bool_t (XDR *, void *, enum_t)) *default
هذا المرشح يترجم بين هياكل C المتحدة. الوسيط decide_ptr هو مؤشر إلى التمييز الذي يحدد أي جزء من الاتحاد سيتم ترجمته. الوسيط handle هو مؤشر إلى بنية الاتحاد. الوسيط choices هو مصفوفة من هياكل xdr_discrim التي تربط قيم التمييز بروتينات XDR الخاصة بها. الوسيط default هو مؤشر إلى روتين XDR الذي يتم استدعاؤه إذا لم تتطابق قيمة التمييز مع أي من الخيارات. تعيد هذه الروتينات واحدًا إذا نجحت، وصفرًا بخلاف ذلك.
bool_t xdr_vector(XDR *xdrs, char *arrp, unsigned int maxsize,
                  unsigned int len, xdrproc_t elementproc);
مشابه لـ xdr_array()، ولكنه يترجم بين مصفوفات ذات طول ثابت بدلاً من مصفوفات ذات طول متغير. الوسيط arrp هو عنوان المصفوفة، و maxsize هو الحد الأقصى لحجم المصفوفة. الوسيط len هو الطول الفعلي للمصفوفة. الوسيط elementproc هو روتين XDR لترجمة عناصر المصفوفة. تعيد هذه الروتينات واحدًا إذا نجحت، وصفرًا بخلاف ذلك.
bool_t xdr_void(void);
روتين XDR فارغ. يتم استدعاؤه دائمًا عند استدعاء xdr_free() لإلغاء تخصيص الكائن الذي تمت إزالته. يعيد هذا الروتين واحدًا دائمًا.
bool_t xdr_write(XDR *xdrs, char *data, unsigned int len);
