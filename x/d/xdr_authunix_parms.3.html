struct rpc_createerr rpc_createerr;
متغير عام يتم تعيين قيمته بواسطة أي روتين إنشاء عميل RPC لا ينجح. استخدم الروتين clnt_pcreateerror() لطباعة السبب.
void svc_destroy(SVCXPRT *xprt);
وحدة ماكرو تقوم بتدمير مقبض نقل خدمة RPC، xprt. عادة ما يتضمن التدمير تخصيص هياكل بيانات خاصة، بما في ذلك xprt نفسه. استخدام xprt غير محدد بعد استدعاء هذا الروتين.
fd_set svc_fdset;
متغير عام يعكس قناع بت واصف الملف للقراءة من جانب خدمة RPC؛ وهو مناسب كمعامل لاستدعاء نظام select(2). هذا يهم فقط إذا قام منفذ الخدمة بمعالجة الأحداث غير المتزامنة الخاصة به، بدلاً من استدعاء svc_run(). هذا المتغير للقراءة فقط (لا تمرر عنوانه إلى select(2)!)، ومع ذلك قد يتغير بعد استدعاء svc_getreqset() أو أي روتينات إنشاء.
int svc_fds;
على غرار svc_fdset، ولكن محدود بـ 32 واصف ملف. تم إهمال هذه الواجهة بواسطة svc_fdset.
svc_freeargs(SVCXPRT *xprt, xdrproc_t inproc, char *in);
وحدة ماكرو تقوم بتحرير أي بيانات تم تخصيصها بواسطة نظام RPC/XDR عند فك تشفير وسائط إجراء خدمة باستخدام svc_getargs(). يعيد هذا الروتين 1 إذا تم تحرير النتائج بنجاح، وصفر بخلاف ذلك.
svc_getargs(SVCXPRT *xprt, xdrproc_t inproc, char *in);
وحدة ماكرو تقوم بفك تشفير وسائط طلب RPC المرتبط بمقبض نقل خدمة RPC، xprt. الوسيط in هو العنوان حيث سيتم وضع الوسائط؛ inproc هو روتين XDR المستخدم لفك تشفير الوسائط. يعيد هذا الروتين واحدًا إذا نجح فك التشفير، وصفر بخلاف ذلك.
struct sockaddr_in *svc_getcaller(SVCXPRT *xprt);
الطريقة المعتمدة للحصول على عنوان الشبكة للمتصل بإجراء مرتبط بمقبض نقل خدمة RPC، xprt.
void svc_getreqset(fd_set *rdfds);
هذا الروتين يهم فقط إذا لم يستدعي منفذ الخدمة svc_run()، وبدلاً من ذلك ينفذ معالجة أحداث غير متزامنة مخصصة. يتم استدعاؤه عندما يكون استدعاء نظام select(2) قد حدد أن طلب RPC قد وصل على بعض مآخذ (sockets) RPC؛ rdfds هو قناع بت واصف الملف للقراءة الناتج. يعود الروتين عندما يتم خدمة جميع المآخذ المرتبطة بقيمة rdfds.
void svc_getreq(int rdfds);
على غرار svc_getreqset()، ولكن محدود بـ 32 واصف ملف. تم إهمال هذه الواجهة بواسطة svc_getreqset().
bool_t svc_register(SVCXPRT *xprt, unsigned long prognum,
                    unsigned long versnum,
                    typeof(void (struct svc_req *, SVCXPRT *)) *dispatch,
                    unsigned long protocol);
يربط prognum و versnum بإجراء توزيع الخدمة، dispatch. إذا كان protocol صفرًا، فلا يتم تسجيل الخدمة مع خدمة portmap. إذا كان protocol غير صفري، فسيتم إنشاء تعيين للثلاثي [prognum,versnum,protocol] إلى xprt->xp_port مع خدمة portmap المحلية (عادة ما يكون protocol صفرًا، IPPROTO_UDP أو IPPROTO_TCP). الإجراء dispatch له الشكل التالي:

dispatch(struct svc_req *request, SVCXPRT *xprt);

يعيد الروتين svc_register() واحدًا إذا نجح، وصفرًا بخلاف ذلك.
void svc_run(void);
هذا الروتين لا يعود أبداً. ينتظر وصول طلبات RPC، ويستدعي إجراء الخدمة المناسب باستخدام svc_getreq() عند وصول أحدها. عادة ما ينتظر هذا الإجراء استدعاء نظام select(2) للعودة.
bool_t svc_sendreply(SVCXPRT *xprt, xdrproc_t outproc, char *out);
يتم استدعاؤه بواسطة إجراء توزيع خدمة RPC لإرسال نتائج استدعاء إجراء عن بعد. الوسيط xprt هو مقبض النقل المرتبط بالطلب؛ outproc هو روتين XDR المستخدم لتشفير النتائج؛ و out هو عنوان النتائج. يعيد هذا الروتين واحدًا إذا نجح، وصفرًا بخلاف ذلك.
void svc_unregister(unsigned long prognum, unsigned long versnum);
إزالة جميع تعيينات الزوج [prognum,versnum] لإجراءات التوزيع، والزوج [prognum,versnum,*] لرقم المنفذ.
void svcerr_auth(SVCXPRT *xprt, enum auth_stat why);
يتم استدعاؤه بواسطة إجراء توزيع خدمة يرفض أداء استدعاء إجراء عن بعد بسبب خطأ في المصادقة.
void svcerr_decode(SVCXPRT *xprt);
يتم استدعاؤه بواسطة إجراء توزيع خدمة لا يمكنه فك تشفير وسائطه بنجاح. انظر أيضًا svc_getargs().
void svcerr_noproc(SVCXPRT *xprt);
يتم استدعاؤه بواسطة إجراء توزيع خدمة لا ينفذ رقم الإجراء الذي يطلبه المتصل.
void svcerr_noprog(SVCXPRT *xprt);
يتم استدعاؤه عندما لا يتم تسجيل البرنامج المطلوب مع حزمة RPC. عادة لا يحتاج منفذو الخدمة إلى هذا الروتين.
void svcerr_progvers(SVCXPRT *xprt, unsigned long low_vers,
                     unsigned long high_vers);
يتم استدعاؤه عندما لا يتم تسجيل الإصدار المطلوب من برنامج مع حزمة RPC. عادة لا يحتاج منفذو الخدمة إلى هذا الروتين.
void svcerr_systemerr(SVCXPRT *xprt);
يتم استدعاؤه بواسطة إجراء توزيع خدمة عند اكتشاف خطأ نظام لا يغطيه بروتوكول معين. على سبيل المثال، إذا لم يعد بإمكان الخدمة تخصيص مساحة تخزين، فقد تستدعي هذا الروتين.
void svcerr_weakauth(SVCXPRT *xprt);
يتم استدعاؤه بواسطة إجراء توزيع خدمة يرفض أداء استدعاء إجراء عن بعد بسبب عدم كفاية معلمات المصادقة. يستدعي الروتين svcerr_auth(xprt, AUTH_TOOWEAK).
SVCXPRT *svcfd_create(int fd, unsigned int sendsize,
                      unsigned int recvsize);
إنشاء خدمة فوق أي واصف ملف مفتوح. عادة، يكون واصف الملف هذا هو مقبس متصل لبروتوكول تدفق مثل TCP. sendsize و recvsize تشير إلى أحجام مخازن الإرسال والاستقبال. إذا كانت صفرًا، يتم اختيار قيمة افتراضية معقولة.
SVCXPRT *svcraw_create(void);
يقوم هذا الروتين بإنشاء مقبض نقل خدمة RPC وهمي، ويرجع إليه مؤشرًا. النقل هو في الواقع مخزن مؤقت داخل مساحة عنوان العملية، لذلك يجب أن يكون عميل RPC المقابل في نفس مساحة العنوان؛ انظر clntraw_create(). يسمح هذا الروتين بمحاكاة RPC واكتساب تكاليف RPC (مثل أوقات الرحلة ذهابًا وإيابًا)، دون أي تدخل من النواة. يعيد هذا الروتين NULL إذا فشل.
SVCXPRT *svctcp_create(int sock, unsigned int send_buf_size,
                       unsigned int recv_buf_size);
يقوم هذا الروتين بإنشاء مقبض نقل خدمة RPC قائم على TCP/IP، ويرجع إليه مؤشرًا. يرتبط النقل بالمقبس sock، والذي قد يكون RPC_ANYSOCK، وفي هذه الحالة يتم إنشاء مقبس جديد. إذا لم يكن المقبس مرتبطًا بمنفذ TCP محلي، فسيقوم هذا الروتين بربطه بمنفذ عشوائي. عند الانتهاء، يكون xprt->xp_sock هو واصف مقبس النقل، و xprt->xp_port هو رقم منفذ النقل. يعيد هذا الروتين NULL إذا فشل. نظرًا لأن RPC القائم على TCP يستخدم إدخال/إخراج مخزن مؤقتًا، يمكن للمستخدمين تحديد حجم المخازن المؤقتة؛ تختار القيم الصفرية افتراضيات مناسبة.
SVCXPRT *svcudp_bufcreate(int sock, unsigned int sendsize,
                          unsigned int recosize);
يقوم هذا الروتين بإنشاء مقبض نقل خدمة RPC قائم على UDP/IP، ويرجع إليه مؤشرًا. يرتبط النقل بالمقبس sock، والذي قد يكون RPC_ANYSOCK، وفي هذه الحالة يتم إنشاء مقبس جديد. إذا لم يكن المقبس مرتبطًا بمنفذ UDP محلي، فسيقوم هذا الروتين بربطه بمنفذ عشوائي. عند الانتهاء، يكون xprt->xp_sock هو واصف مقبس النقل، و xprt->xp_port هو رقم منفذ النقل. يعيد هذا الروتين NULL إذا فشل.
يسمح هذا للمستخدم بتحديد الحد الأقصى لحجم الحزمة لإرسال واستقبال رسائل RPC المستندة إلى UDP.
SVCXPRT *svcudp_create(int sock);

الملخص والوصف


هذه الوظائف هي جزء من مكتبة XDR (External Data Representation) و RPC (Remote Procedure Call) في نظام Unix.


bool_t xdr_accepted_reply(XDR *xdrs, struct accepted_reply *ar);



  

  
يستخدم لوصف رسائل الرد المقبولة لـ RPC. هذه الوظيفة مفيدة للمستخدمين
      الذين يرغبون في إنشاء رسائل بأسلوب RPC دون استخدام حزمة RPC.




bool_t xdr_authunix_parms(XDR *xdrs, struct authunix_parms *aup);



  

  
يستخدم لوصف معلمات المصادقة الخاصة بـ AUTH_UNIX. هذه الوظيفة مفيدة
      للمستخدمين الذين يرغبون في إنشاء رسائل بأسلوب RPC دون استخدام حزمة RPC.

      الذين يرغبون في إنشاء رسائل بأسلوب RPC دون استخدام حزمة RPC.




bool_t xdr_callmsg(XDR *xdrs, struct rpc_msg *cmsg);

      الذين يرغبون في إنشاء رسائل بأسلوب RPC دون استخدام حزمة RPC.




bool_t xdr_opaque_auth(XDR *xdrs, struct opaque_auth *ap);



  

  
يستخدم لوصف رسائل معلومات المصادقة لـ RPC. هذه الوظيفة مفيدة
      للمستخدمين الذين يرغبون في إنشاء رسائل بأسلوب RPC دون استخدام
      حزمة RPC.




bool_t xdr_pmap(XDR *xdrs, struct pmap *regs);

      pmap.




bool_t xdr_rejected_reply(XDR *xdrs, struct rejected_reply *rr);

bool_t xdr_replymsg(XDR *xdrs, struct rpc_msg *rmsg);

      الذين يرغبون في إنشاء رسائل بأسلوب RPC دون استخدام حزمة RPC.
