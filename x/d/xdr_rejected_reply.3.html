rpc(3) دليل دوال المكتبة rpc(3)

الاسم

rpc - روتينات مكتبية لاستدعاءات الإجراءات عن بعد

المكتبة

مكتبة C القياسية (libc, -lc)

الملخص والوصف

تسمح هذه الروتينات لبرامج C بإجراء استدعاءات إجراءات على أجهزة أخرى عبر الشبكة. أولاً، يستدعي العميل إجراءً لإرسال حزمة بيانات إلى الخادم. عند استلام الحزمة، يستدعي الخادم روتين توزيع لأداء الخدمة المطلوبة، ثم يرسل رداً. أخيرًا، يعود استدعاء الإجراء إلى العميل.

للاستفادة من هذه الروتينات، قم بتضمين ملف الرأس

<rpc/rpc.h>.

تستخدم التعريفات النموذجية أدناه الأنواع التالية:

typedef int bool_t;
typedef typeof(bool_t (XDR *, void *, ...))  *xdrproc_t;
typedef typeof(bool_t (caddr_t resp, struct sockaddr_in *raddr)
    *resultproc_t;

راجع ملفات الرأس الخاصة بتعريفات أنواع AUTH، و CLIENT، و SVCXPRT، و XDR.

void auth_destroy(AUTH *auth);
ماكرو يدمر معلومات المصادقة المرتبطة بـ auth. يتضمن التدمير عادةً تخصيص هياكل البيانات الخاصة. استخدام auth غير محدد بعد استدعاء auth_destroy().
AUTH *authnone_create(void);
إنشاء وإرجاع مقبض مصادقة RPC يمرر معلومات مصادقة غير قابلة للاستخدام مع كل استدعاء إجراء عن بعد. هذه هي المصادقة الافتراضية التي تستخدمها RPC.
AUTH *authunix_create(char *host, uid_t uid, gid_t gid,
                      int n, gid_t aup_gids[n]);
إنشاء وإرجاع مقبض مصادقة RPC يحتوي على معلومات المصادقة. المعلمة host هي اسم الجهاز الذي تم إنشاء المعلومات عليه؛ uid هو معرف المستخدم؛ gid هو معرف المجموعة الحالي للمستخدم؛ n و aup_gids يشيران إلى مصفوفة محسوبة للمجموعات التي ينتمي إليها المستخدم. من السهل انتحال هوية مستخدم.
AUTH *authunix_create_default(void);
يستدعي authunix_create() بالمعلمات المناسبة.
int callrpc(char *host, unsigned long prognum,
            unsigned long versnum, unsigned long procnum,
            xdrproc_t inproc, const char *in,
            xdrproc_t outproc, char *out);
استدعاء الإجراء البعيد المرتبط بـ prognum، و versnum، و procnum على الجهاز host. المعلمة in هي عنوان وسيط (وسائط) الإجراء، و out هو العنوان الذي سيتم فيه وضع النتائج؛ inproc يستخدم لتشفير معلمات الإجراء، و outproc يستخدم لفك تشفير نتائج الإجراء؛ تعود هذه الروتينات بالصفر إذا نجحت، أو بقيمة enum clnt_stat المحولة إلى عدد صحيح إذا فشلت. الروتين clnt_perrno() مفيد لترجمة حالات الفشل إلى رسائل.
تحذير: استدعاء الإجراءات البعيدة باستخدام هذا الروتين يستخدم UDP/IP كوسيلة نقل؛ انظر clntudp_create() للقيود. ليس لديك تحكم في المهلات أو المصادقة باستخدام هذا الروتين.
enum clnt_stat clnt_broadcast(unsigned long prognum,
                     unsigned long versnum, unsigned long procnum,
                     xdrproc_t inproc, char *in,
                     xdrproc_t outproc, char *out,
                     resultproc_t eachresult);
مثل callrpc()، باستثناء أن رسالة الاستدعاء يتم بثها إلى جميع شبكات البث المتصلة محليًا. في كل مرة تتلقى فيها استجابة، يستدعي هذا الروتين eachresult()، والذي يكون شكله:

eachresult(char *out, struct sockaddr_in *addr);

حيث out هو نفس out الذي تم تمريره إلى clnt_broadcast()، باستثناء أن مخرجات الإجراء البعيد يتم فك تشفيرها هناك؛ addr يشير إلى عنوان الجهاز الذي أرسل النتائج. إذا أعادت eachresult() صفراً، فإن clnt_broadcast() تنتظر المزيد من الردود؛ وإلا فإنها تعود بالحالة المناسبة.
تحذير: مقابس البث محدودة الحجم بأقصى وحدة نقل لطبقة ربط البيانات. بالنسبة لـ ethernet، هذه القيمة هي 1500 بايت.
enum clnt_stat clnt_call(CLIENT *clnt, unsigned long procnum,
                    xdrproc_t inproc, char *in,
                    xdrproc_t outproc, char *out,
                    struct timeval tout);
ماكرو يستدعي الإجراء البعيد procnum المرتبط بمقبض العميل clnt، والذي يتم الحصول عليه باستخدام روتين إنشاء عميل RPC مثل clnt_create(). المعلمة in هي عنوان وسيط (وسائط) الإجراء، و out هو العنوان الذي سيتم فيه وضع النتائج؛ inproc يستخدم لتشفير معلمات الإجراء، و outproc يستخدم لفك تشفير نتائج الإجراء؛ tout هو الوقت المسموح به لعودة النتائج.
clnt_destroy(CLIENT *clnt);
ماكرو يدمر مقبض RPC الخاص بالعميل. يتضمن التدمير عادةً تخصيص هياكل البيانات الخاصة، بما في ذلك clnt نفسه. استخدام clnt غير محدد بعد استدعاء clnt_destroy(). إذا فتحت مكتبة RPC المقبس المرتبط، فستقوم بإغلاقه أيضًا. وإلا، يبقى المقبس مفتوحًا.
CLIENT *clnt_create(const char *host, unsigned long prog,
                    unsigned long vers, const char *proto);
روتين إنشاء عميل عام. host يحدد اسم المضيف البعيد حيث يوجد الخادم. proto يحدد نوع بروتوكول النقل الذي سيتم استخدامه. القيم المدعومة حاليًا لهذا الحقل هي “udp” و “tcp”. يتم تعيين مهلات افتراضية، ولكن يمكن تعديلها باستخدام clnt_control().
تحذير: استخدام UDP له عيوبه. نظرًا لأن رسائل RPC المستندة إلى UDP يمكن أن تحتوي فقط على ما يصل إلى 8 كيلوبايتات من البيانات المشفرة، فإن هذا الناقل لا يمكن استخدامه للإجراءات التي تأخذ وسائط كبيرة أو تعيد نتائج ضخمة.
bool_t clnt_control(CLIENT *cl, int req, char *info);
ماكرو يستخدم لتغيير أو استرداد معلومات مختلفة حول كائن العميل. req يشير إلى نوع العملية، و info هو مؤشر إلى المعلومات. لكل من UDP و TCP، القيم المدعومة لـ req وأنواع وسائطها وماذا تفعل هي:

CLSET_TIMEOUT  struct timeval // تعيين المهلة الإجمالية
CLGET_TIMEOUT  struct timeval // الحصول على المهلة الإجمالية

ملاحظة: إذا قمت بتعيين المهلة باستخدام clnt_control()، فسيتم تجاهل المعلمة timeout التي تم تمريرها إلى clnt_call() في جميع الاستدعاءات المستقبلية.

CLGET_SERVER_ADDR  struct sockaddr_in
                // الحصول على عنوان الخادم

العمليات التالية صالحة لـ UDP فقط:

CLSET_RETRY_TIMEOUT  struct timeval // تعيين مهلة إعادة المحاولة
CLGET_RETRY_TIMEOUT  struct timeval // الحصول على مهلة إعادة المحاولة

مهلة إعادة المحاولة هي الوقت الذي تنتظره "UDP RPC" حتى يرد الخادم قبل إعادة إرسال الطلب.
clnt_freeres(CLIENT * clnt, xdrproc_t outproc, char *out);
ماكرو يحرر أي بيانات تم تخصيصها بواسطة نظام RPC/XDR عند فك تشفير نتائج استدعاء RPC. المعلمة out هي عنوان النتائج، و outproc هو روتين XDR الذي يصف النتائج. يعود هذا الروتين بواحد إذا تم تحرير النتائج بنجاح، وصفر بخلاف ذلك.
void clnt_geterr(CLIENT *clnt, struct r
pc_err *errp);
وحدة ماكرو تقوم بنسخ بنية الخطأ من مقبض العميل إلى البنية الموجودة في العنوان errp.
void clnt_pcreateerror(const char *s);
طباعة رسالة إلى الخطأ القياسي تشير إلى سبب عدم إنشاء مقبض عميل RPC. تسبق الرسالة السلسلة s ونقطتين. تستخدم عند فشل استدعاء clnt_create() أو clntraw_create() أو clnttcp_create() أو clntudp_create().
void clnt_perrno(enum clnt_stat stat);
طباعة رسالة إلى الخطأ القياسي تتوافق مع الحالة التي تشير إليها stat. تستخدم بعد callrpc().
clnt_perror(CLIENT *clnt, const char *s);
طباعة رسالة إلى الخطأ القياسي تشير إلى سبب فشل استدعاء RPC؛ clnt هو المقبض المستخدم لإجراء الاستدعاء. تسبق الرسالة السلسلة s ونقطتين. تستخدم بعد clnt_call().
char *clnt_spcreateerror(const char *s);
مثل clnt_pcreateerror()، باستثناء أنها تُرجع سلسلة نصية بدلاً من الطباعة إلى الخطأ القياسي.
أخطاء: تُرجع مؤشرًا إلى بيانات ثابتة يتم الكتابة فوقها في كل استدعاء.
char *clnt_sperrno(enum clnt_stat stat);
تأخذ نفس وسائط clnt_perrno()، ولكن بدلاً من إرسال رسالة إلى الخطأ القياسي تشير إلى سبب فشل استدعاء RPC، تُرجع مؤشرًا إلى سلسلة تحتوي على الرسالة. تنتهي السلسلة بحرف سطر جديد (NEWLINE).
يتم استخدام clnt_sperrno() بدلاً من clnt_perrno() إذا لم يكن لدى البرنامج خطأ قياسي (كما هو الحال غالبًا في برنامج يعمل كخادم)، أو إذا لم يرغب المبرمج في إخراج الرسالة باستخدام printf(3)، أو إذا كان سيتم استخدام تنسيق رسالة مختلف عن ذلك الذي تدعمه clnt_perrno(). ملاحظة: على عكس clnt_sperror() و clnt_spcreateerror()، تُرجع clnt_sperrno() مؤشرًا إلى بيانات ثابتة، ولكن لن يتم الكتابة فوق النتيجة في كل استدعاء.
char *clnt_sperror(CLIENT *rpch, const char *s);
مثل clnt_perror()، باستثناء أنها (مثل clnt_sperrno()) تُرجع سلسلة نصية بدلاً من الطباعة إلى الخطأ القياسي.
أخطاء: تُرجع مؤشرًا إلى بيانات ثابتة يتم الكتابة فوقها في كل استدعاء.
CLIENT *clntraw_create(unsigned long prognum, unsigned long versnum);
يقوم هذا الروتين بإنشاء عميل RPC وهمي للبرنامج البعيد prognum، الإصدار versnum. النقل المستخدم لتمرير الرسائل إلى الخدمة هو في الواقع مخزن مؤقت داخل مساحة عنوان العملية، لذلك يجب أن يعيش خادم RPC المقابل في نفس مساحة العنوان؛ انظر svcraw_create(). هذا يسمح بمحاكاة RPC واكتساب النفقات العامة لـ RPC، مثل أوقات الذهاب والإياب، دون أي تدخل من النواة. يُرجع هذا الروتين NULL إذا فشل.
CLIENT *clnttcp_create(struct sockaddr_in *addr,
                unsigned long prognum, unsigned long versnum,
                int *sockp, unsigned int sendsz, unsigned int recvsz);
يقوم هذا الروتين بإنشاء عميل RPC للبرنامج البعيد prognum، الإصدار versnum؛ يستخدم العميل TCP/IP كوسيلة نقل. يقع البرنامج البعيد في عنوان الإنترنت *addr. إذا كان addr->sin_port صفرًا، فسيتم تعيينه إلى المنفذ الفعلي الذي يستمع إليه البرنامج البعيد (يتم استشارة خدمة portmap البعيدة لهذه المعلومات). المعلمة sockp هي مقبس؛ إذا كانت RPC_ANYSOCK، فسيقوم هذا الروتين بفتح مقبس جديد وتعيين sockp. نظرًا لأن RPC المستند إلى TCP يستخدم إدخال/إخراج مخزّن مؤقتًا، يمكن للمستخدم تحديد حجم المخازن المؤقتة للإرسال والاستقبال باستخدام المعلمات sendsz و recvsz؛ القيم الصفرية تختار قيمًا افتراضية مناسبة. يُرجع هذا الروتين NULL إذا فشل.
CLIENT *clntudp_create(struct sockaddr_in *addr,
                unsigned long prognum, unsigned long versnum,
                struct timeval wait, int *sockp);
يقوم هذا الروتين بإنشاء عميل RPC للبرنامج البعيد prognum، الإصدار versnum؛ يستخدم العميل UDP/IP كوسيلة نقل. يقع البرنامج البعيد في عنوان الإنترنت addr. إذا كان addr->sin_port صفرًا، فسيتم تعيينه إلى المنفذ الفعلي الذي يستمع إليه البرنامج البعيد (يتم استشارة خدمة portmap البعيدة لهذه المعلومات). المعلمة sockp هي مقبس؛ إذا كانت RPC_ANYSOCK، فسيقوم هذا الروتين بفتح مقبس جديد وتعيين sockp. تعيد وسيلة نقل UDP إرسال رسالة الاستدعاء بفواصل زمنية قدرها wait حتى يتم استلام رد أو حتى ينتهي وقت استدعاء الاستدعاء. يتم تحديد الوقت الإجمالي لانتهاء صلاحية الاستدعاء بواسطة clnt_call().

هذا يسمح للمستخدم بتحديد الحد الأقصى لحجم الحزمة لإرسال واستقبال رسائل RPC المستندة إلى UDP.
void get_myaddress(struct sockaddr_in *addr);
يضع عنوان IP الخاص بالآلة في *addr، دون استشارة روتينات المكتبة التي تتعامل مع /etc/hosts. يتم دائمًا تعيين رقم المنفذ إلى htons(PMAPPORT).
struct pmaplist *pmap_getmaps(struct sockaddr_in *addr);
واجهة مستخدم لخدمة portmap، والتي تُرجع قائمة بتعيينات برنامج RPC الحالية إلى المنافذ على المضيف الموجود في عنوان IP *addr. يمكن أن تُرجع هذه الوظيفة NULL. يستخدم الأمر rpcinfo -p هذه الوظيفة.
unsigned short pmap_getport(struct sockaddr_in *addr,
                    unsigned long prognum, unsigned long versnum,
                    unsigned int protocol);
واجهة مستخدم لخدمة portmap، والتي تُرجع رقم المنفذ الذي تنتظر عليه الخدمة التي تدعم رقم البرنامج prognum، الإصدار versnum، وتتحدث بروتوكول النقل المرتبط بـ protocol. القيمة protocol غالبًا ما تكون IPPROTO_UDP أو IPPROTO_TCP. القيمة المرجعة صفر تعني أن التعيين غير موجود أو أن نظام RPC فشل في الاتصال بخدمة portmap البعيدة. في الحالة الأخيرة، تحتوي المتغير العام rpc_createerr على حالة RPC.
enum clnt_stat pmap_rmtcall(struct sockaddr_in *addr,

                    unsigned long procnum,
                    xdrproc_t inproc, char *in,
                    xdrproc_t outproc, char *out,
                    struct timeval tout, unsigned long *portp);
واجهة مستخدم لخدمة portmap، والتي تطلب من portmap على المضيف في عنوان IP *addr إجراء استدعاء RPC نيابة عنك إلى إجراء على هذا المضيف. سيتم تعديل المعلمة *portp إلى رقم منفذ البرنامج إذا نجح الإجراء. تمت مناقشة تعريفات المعلمات الأخرى في callrpc() و clnt_call(). يجب استخدام هذا الإجراء "للتنبيه" ولا شيء آخر. انظر أيضًا clnt_broadcast().
bool_t pmap_set(unsigned long prognum, unsigned long versnum,

واجهة مستخدم لخدمة portmap، والتي تنشئ تعيينًا بين الثلاثي [prognum,versnum,protocol] و port على خدمة portmap الخاصة بالآلة. قيمة protocol غالبًا ما تكون IPPROTO_UDP أو IPPROTO_TCP. تُرجع هذه الوظيفة واحدًا إذا نجحت، وصفرًا بخلاف ذلك. يتم ذلك تلقائيًا بواسطة svc_register().
bool_t pmap_unset(unsigned long prognum, unsigned long versnum);
واجهة مستخدم لخدمة portmap، والتي تدمر جميع التعيينات بين الثلاثي [prognum,versnum,*] و ports على خدمة portmap الخاصة بالآلة. تُرجع هذه الوظيفة واحدًا إذا نجحت، وصفرًا بخلاف ذلك.
int registerrpc(unsigned long prognum, unsigned long versnum,
                unsigned long procnum, typeof(char *(char *)) *procname,
                xdrproc_t inproc, xdrproc_t outproc);
تسجيل الإجراء procname مع حزمة خدمة RPC. إذا وصل طلب للبرنامج prognum، والإصدار versnum، والإجراء procnum، يتم استدعاء procname مع مؤشر إلى وسائطه؛ يجب أن تُرجع procname مؤشرًا إلى نتائجه الثابتة؛ يتم استخدام inproc لفك تشفير الوسائط بينما يتم استخدام outproc لتشفير النتائج. تُرجع هذه الوظيفة صفرًا إذا نجح التسجيل، و -1 بخلاف ذلك.
