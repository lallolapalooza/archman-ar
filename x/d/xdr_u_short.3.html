xdr(3) دليل وظائف المكتبة xdr(3)

الاسم

xdr — روتينات المكتبة لتمثيل البيانات الخارجي

المكتبة

مكتبة C القياسية (libc, -lc)

الملخص والوصف

تسمح هذه الروتينات لبرامج C بوصف هياكل بيانات عشوائية بطريقة مستقلة عن الجهاز. يتم إرسال البيانات لاستدعاءات الإجراءات البعيدة باستخدام هذه الروتينات.

النماذج الأولية أدناه معلنة في <rpc/xdr.h> وتستخدم الأنواع التالية:

typedef int bool_t;
typedef typeof(bool_t (XDR *, void *, ...))  *xdrproc_t;

لإعلان نوع XDR، راجع <rpc/xdr.h>.

bool_t xdr_array(XDR *xdrs, char **arrp, unsigned int *sizep,
                 unsigned int maxsize, unsigned int elsize,
                 xdrproc_t elproc);
مرشح بدائي يترجم بين المصفوفات ذات الطول المتغير وتمثيلاتها الخارجية المقابلة. الوسيط arrp هو عنوان المؤشر إلى المصفوفة، بينما sizep هو عنوان عدد عناصر المصفوفة؛ لا يمكن أن يتجاوز عدد العناصر هذا maxsize. الوسيط elsize هو sizeof كل عنصر من عناصر المصفوفة، و elproc هو مرشح XDR يترجم بين شكل C لعناصر المصفوفة وتمثيلها الخارجي. تعيد هذه الروتينية واحدًا إذا نجحت، وصفرًا إذا فشلت.
bool_t xdr_bool(XDR *xdrs, bool_t *bp);
مرشح بدائي يترجم بين القيم المنطقية (أعداد صحيحة C) وتمثيلاتها الخارجية. عند ترميز البيانات، تنتج هذه المرشحة قيمًا من واحد أو صفر. تعيد هذه الروتينية واحدًا إذا نجحت، وصفرًا إذا فشلت.
bool_t xdr_bytes(XDR *xdrs, char **sp, unsigned int *sizep,
                 unsigned int maxsize);
مرشح بدائي يترجم بين السلاسل النصية ذات البايتات المعدودة وتمثيلاتها الخارجية. الوسيط sp هو عنوان مؤشر السلسلة النصية. يقع طول السلسلة النصية في العنوان sizep؛ لا يمكن أن تكون السلاسل النصية أطول من maxsize. تعيد هذه الروتينية واحدًا إذا نجحت، وصفرًا إذا فشلت.
bool_t xdr_char(XDR *xdrs, char *cp);
مرشح بدائي يترجم بين أحرف C وتمثيلاتها الخارجية. تعيد هذه الروتينية واحدًا إذا نجحت، وصفرًا إذا فشلت. ملاحظة: الأحرف المشفرة ليست مضغوطة، وتشغل 4 بايت لكل منها. بالنسبة لمصفوفات الأحرف، يجدر النظر في xdr_bytes() أو xdr_opaque() أو xdr_string().
void xdr_destroy(XDR *xdrs);
ماكرو يستدعي روتينية التدمير المرتبطة بتيار XDR، xdrs. يتضمن التدمير عادةً تحرير هياكل البيانات الخاصة المرتبطة بالتيار. استخدام xdrs بعد استدعاء xdr_destroy() غير معرف.
bool_t xdr_double(XDR *xdrs, double *dp);
مرشح بدائي يترجم بين أرقام double (مزدوجة الدقة) في C وتمثيلاتها الخارجية. تعيد هذه الروتينية واحدًا إذا نجحت، وصفرًا إذا فشلت.
bool_t xdr_enum(XDR *xdrs, enum_t *ep);
مرشح بدائي يترجم بين enums في C (في الواقع أعداد صحيحة) وتمثيلاتها الخارجية. تعيد هذه الروتينية واحدًا إذا نجحت، وصفرًا إذا فشلت.
bool_t xdr_float(XDR *xdrs, float *fp);
مرشح بدائي يترجم بين floats في C وتمثيلاتها الخارجية. تعيد هذه الروتينية واحدًا إذا نجحت، وصفرًا إذا فشلت.
void xdr_free(xdrproc_t proc, char *objp);
روتينية تحرير عامة. الوسيط الأول هو روتينية XDR للكائن الذي يتم تحريره. الوسيط الثاني هو مؤشر إلى الكائن نفسه. ملاحظة: المؤشر الذي تم تمريره إلى هذه الروتينية لا يتم تحريره، ولكن ما يشير إليه يتم تحريره (بشكل متكرر).
unsigned int xdr_getpos(XDR *xdrs);
ماكرو يستدعي روتينية الحصول على الموضع المرتبطة بتيار XDR، xdrs. تعيد الروتينية عددًا صحيحًا غير موقع، والذي يشير إلى موضع تيار بايتات XDR. ميزة مرغوبة لتيارات XDR هي أن العمليات الحسابية البسيطة تعمل مع هذا الرقم، على الرغم من أن مثيلات تيار XDR قد لا تضمن ذلك.
long *xdr_inline(XDR *xdrs, int len);
ماكرو يستدعي روتينية المضمن المرتبطة بتيار XDR، xdrs. تعيد الروتينية مؤشرًا إلى قطعة متجاورة من مخزن التيار المؤقت؛ len هو طول البايت المطلوب للمخزن المؤقت. ملاحظة: يتم تحويل المؤشر إلى long *.
تحذير: قد تعيد xdr_inline() قيمة NULL (0) إذا لم تتمكن من تخصيص قطعة متجاورة من المخزن المؤقت. لذلك قد يختلف السلوك بين مثيلات التيار؛ وهي موجودة من أجل الكفاءة.
bool_t xdr_int(XDR *xdrs, int *ip);
مرشح بدائي يترجم بين الأعداد الصحيحة C وتمثيلاتها الخارجية. تعيد هذه الروتينية واحدًا إذا نجحت، وصفرًا بخلاف ذلك.
bool_t xdr_long(XDR *xdrs, long *lp);
مرشح بدائي يترجم بين أعداد long الصحيحة في C وتمثيلاتها الخارجية. تعيد هذه الروتينية واحدًا إذا نجحت، وصفرًا بخلاف ذلك.
void xdrmem_create(XDR *xdrs, char *addr, unsigned int size,
                   enum xdr_op op);
تقوم هذه الروتينية بتهيئة كائن تيار XDR الذي يشير إليه xdrs. يتم كتابة بيانات التيار، أو قراءتها، من جزء من الذاكرة في الموقع addr الذي لا يتجاوز طوله size بايت. تحدد op اتجاه تيار XDR (إما XDR_ENCODE أو XDR_DECODE أو XDR_FREE).
bool_t xdr_opaque(XDR *xdrs, char *cp, unsigned int cnt);
مرشح بدائي يترجم بين البيانات المعتمة ذات الحجم الثابت وتمثيلها الخارجي. الوسيط cp هو عنوان الكائن المعتم، و cnt هو حجمه بالبايت. تعيد هذه الروتينية واحدًا إذا نجحت، وصفرًا إذا فشلت.
bool_t xdr_pointer(XDR *xdrs, char **objpp,
                   unsigned int objsize, xdrproc_t xdrobj);
مثل xdr_reference() باستثناء أنها تقوم بتسلسل المؤشرات الفارغة، بينما xdr_reference() لا تفعل ذلك. وبالتالي، يمكن لـ xdr_pointer() تمثيل هياكل البيانات المتكررة، مثل الأشجار الثنائية أو القوائم المرتبطة.
void xdrrec_create(XDR *xdrs, unsigned int sendsize,
                   unsigned int recvsize, char *handle,
                   typeof(int (char *, char *, int)) *readit,
                   typeof(int (char *, char *, int)) *writeit);
تقوم هذه الروتينية بتهيئة كائن تيار XDR الذي يشير إليه xdrs. يتم كتابة بيانات التيار إلى مخزن مؤقت بحجم sendsize؛ القيمة الصفرية تشير إلى أن النظام يجب أن يستخدم قيمة افتراضية مناسبة. يتم قراءة بيانات التيار من مخزن مؤقت بحجم recvsize؛ يمكن أيضًا تعيين هذا المخزن المؤقت إلى قيمة افتراضية مناسبة عن طريق تمرير قيمة صفرية. عندما يكون المخزن المؤقت للإخراج للتيار ممتلئًا، يتم استدعاء writeit. وبالمثل، عندما يكون المخزن المؤقت للإدخال للتيار فارغًا، يتم استدعاء readit. سلوك هاتين الروتينيتين مشابه لاستدعاءات النظام read(2) و write(2)، باستثناء أن handle يتم تمريره إلى الروتينيات السابقة كوسيط أول. ملاحظة: يجب تعيين حقل op لتيار XDR بواسطة المتصل.
تحذير: للقراءة من تيار XDR تم إنشاؤه بواسطة واجهة برمجة التطبيقات هذه، ستحتاج إلى استدعاء xdrrec_skiprecord() أولاً قبل استدعاء أي واجهات برمجة تطبيقات XDR أخرى. هذا يدرج بايتات إضافية في التيار لتوفير حدود السجل.
