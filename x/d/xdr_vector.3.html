الاسم

xdr - مكتبة روتينات لتمثيل البيانات الخارجي

المكتبة

مكتبة C القياسية (libc، -lc)

الملخص والوصف

تسمح هذه الروتينات لمبرمجي C بوصف هياكل بيانات عشوائية بطريقة مستقلة عن الجهاز. يتم نقل البيانات لاستدعاءات الإجراءات عن بعد باستخدام هذه الروتينات.

النماذج الأولية أدناه معلنة في <rpc/xdr.h> وتستخدم الأنواع التالية:

typedef int bool_t;
typedef typeof(bool_t (XDR *, void *, ...))  *xdrproc_t;

لإعلان النوع XDR، انظر <rpc/xdr.h>.

bool_t xdr_array(XDR *xdrs, char **arrp, unsigned int *sizep,
                 unsigned int maxsize, unsigned int elsize,
                 xdrproc_t elproc);
فلتر بدائي يترجم بين المصفوفات متغيرة الطول وتمثيلاتها الخارجية المقابلة. الوسيط arrp هو عنوان المؤشر إلى المصفوفة، بينما sizep هو عنوان عدد العناصر في المصفوفة؛ لا يمكن أن يتجاوز عدد العناصر هذا maxsize. الوسيط elsize هو sizeof كل عنصر من عناصر المصفوفة، و elproc هو فلتر XDR يترجم بين صيغة C لعناصر المصفوفة وتمثيلها الخارجي. تعيد هذه الروتينات واحدًا إذا نجحت، وصفرًا بخلاف ذلك.
bool_t xdr_bool(XDR *xdrs, bool_t *bp);

bool_t xdr_bytes(XDR *xdrs, char **sp, unsigned int *sizep,
                 unsigned int maxsize);

bool_t xdr_char(XDR *xdrs, char *cp);
فلتر بدائي يترجم بين أحرف C وتمثيلاتها الخارجية. تعيد هذه الروتينات واحدًا إذا نجحت، وصفرًا بخلاف ذلك. ملاحظة: الأحرف المشفرة ليست مجمعة، وتشغل 4 بايتات لكل منها. بالنسبة لمصفوفات الأحرف، يجدر النظر في xdr_bytes() أو xdr_opaque() أو xdr_string().
void xdr_destroy(XDR *xdrs);
ماكرو يستدعي روتين التدمير المرتبط بتيار XDR، xdrs. عادة ما يتضمن التدمير تحرير هياكل البيانات الخاصة المرتبطة بالتيار. استخدام xdrs بعد استدعاء xdr_destroy() غير محدد.

void xdr_free(xdrproc_t proc, char *objp);
روتين تحرير عام. الوسيط الأول هو روتين XDR للكائن الذي يتم تحريره. الوسيط الثاني هو مؤشر إلى الكائن نفسه. ملاحظة: المؤشر الممرر إلى هذا الروتين لا يتم تحريره، ولكن ما يشير إليه يتم تحريره (بشكل متكرر).
unsigned int xdr_getpos(XDR *xdrs);
ماكرو يستدعي روتين الحصول على الموضع المرتبط بتيار XDR، xdrs. تعيد الروتين عددًا صحيحًا غير موقّع، والذي يشير إلى موضع تيار بايتات XDR. ميزة مرغوبة لتيارات XDR هي أن العمليات الحسابية البسيطة تعمل مع هذا الرقم، على الرغم من أن مثيلات تيار XDR قد لا تضمن ذلك.
long *xdr_inline(XDR *xdrs, int len);
ماكرو يستدعي روتين inline المرتبط بتيار XDR، xdrs. تعيد الروتين مؤشرًا إلى قطعة متجاورة من مخزن التيار المؤقت؛ len هو طول البايت للمخزن المطلوب. ملاحظة: يتم تحويل المؤشر إلى long *.
تحذير: xdr_inline() قد تعيد NULL (0) إذا لم تتمكن من تخصيص قطعة متجاورة من المخزن المؤقت. لذلك قد يختلف السلوك بين مثيلات التيار؛ إنه موجود من أجل الكفاءة.

void xdrmem_create(XDR *xdrs, char *addr, unsigned int size,
                   enum xdr_op op);
يقوم هذا الروتين بتهيئة كائن تيار XDR الذي يشير إليه xdrs. يتم كتابة بيانات التيار إلى، أو قراءتها من، جزء من الذاكرة في الموقع addr والذي لا يتجاوز طوله size بايت. يحدد op اتجاه تيار XDR (إما XDR_ENCODE أو XDR_DECODE أو XDR_FREE).
bool_t xdr_opaque(XDR *xdrs, char *cp, unsigned int cnt);
فلتر بدائي يترجم بين البيانات المعتمة ذات الحجم الثابت وتمثيلها الخارجي. الوسيط cp هو عنوان الكائن المعتم، و cnt هو حجمه بالبايت. تعيد هذه الروتينات واحدًا إذا نجحت، وصفرًا بخلاف ذلك.
bool_t xdr_pointer(XDR *xdrs, char **objpp,
                   unsigned int objsize, xdrproc_t xdrobj);
مثل xdr_reference() باستثناء أنها تسلسل المؤشرات الفارغة، بينما xdr_reference() لا تفعل ذلك. وبالتالي، يمكن لـ xdr_pointer() تمثيل هياكل البيانات المتكررة، مثل الأشجار الثنائية أو القوائم المرتبطة.
void xdrrec_create(XDR *xdrs, unsigned int sendsize,
                   unsigned int recvsize, char *handle,
                   typeof(int (char *, char *, int)) *readit,
                   typeof(int (char *, char *, int)) *writeit);
يقوم هذا الروتين بتهيئة كائن تيار XDR الذي يشير إليه xdrs. يتم كتابة بيانات التيار إلى مخزن مؤقت بحجم sendsize؛ قيمة صفر تشير إلى أن النظام يجب أن يستخدم قيمة افتراضية مناسبة. يتم قراءة بيانات التيار من مخزن مؤقت بحجم recvsize؛ يمكن أيضًا تعيينها إلى قيمة افتراضية مناسبة عن طريق تمرير قيمة صفر. عندما يكون المخزن المؤقت للإخراج للتيار ممتلئًا، يتم استدعاء writeit. وبالمثل، عندما يكون المخزن المؤقت للإدخال للتيار فارغًا، يتم استدعاء readit. سلوك هذين الروتينين مشابه لاستدعاءات النظام read(2) و write(2)، باستثناء أن handle يتم تمريره إلى الروتينات السابقة كوسيط أول. ملاحظة: يجب على المتصل تعيين حقل op لتيار XDR.
تحذير: لقراءة تيار XDR تم إنشاؤه بواسطة واجهة برمجة التطبيقات هذه، ستحتاج إلى استدعاء xdrrec_skiprecord() أولاً قبل استدعاء أي واجهات برمجة تطبيقات XDR أخرى. هذا يدرج بايتات إضافية في التيار لتوفير حدود السجل.
bool_t xdrrec_skiprecord(XDR *xdrs);
هذا الروتين مفيد فقط عند استخدام تيارات XDR المسجلة. يستخدم لتمييز نهاية السجل الحالي أو لتجاوز سجل غير مرغوب فيه. عند استدعائه، فإنه يتجاهل أي بيانات متبقية في السجل الحالي، ويضع مؤشر قراءة XDR في بداية السجل التالي. إذا كان التيار لا يزال في بداية سجل جديد، فإن استدعاء هذا الروتين لن يكون له أي تأثير. تعيد هذه الروتينات واحدًا إذا نجحت، وصفرًا بخلاف ذلك.
bool_t xdr_reference(XDR *xdrs, char **pp,
                     unsigned int size, xdrproc_t proc);
فلتر بدائي يستخدم لترجمة المؤشرات. الوسيط pp هو عنوان مؤشر الكائن، و size هو حجم الكائن الذي يشير إليه المؤشر، و proc هو فلتر XDR المستخدم لترجمة الكائن. تعيد هذه الروتينات واحدًا إذا نجحت، وصفرًا بخلاف ذلك. ملاحظة: لا يتم التعامل مع المؤشرات الفارغة بشكل خاص.
bool_t xdr_setpos(XDR *xdrs, unsigned int pos);

فلتر بدائي لا يقوم بأي شيء. يتم استخدامه في حالات مثل تمرير مؤشرات إلى void.
bool_t xdr_union(XDR *xdrs, enum_t *discr, char *h,
                 const struct xdr_disc_array *choices,
                 typeof(bool_t (XDR *, void *, enum_t)) *default_func);
فلتر بدائي يترجم بين اتحادات C وتمثيلاتها الخارجية. الوسيط discr هو مؤشر إلى المتغير المكتشف (discriminant) الذي يحدد أي عضو في الاتحاد سيتم ترجيحه. الوسيط h هو مؤشر إلى الاتحاد نفسه. الوسيط choices هو مصفوفة من الأوصاف للاحتمالات الممكنة للمتغير المكتشف. الوسيط default_func هو روتين يتم استدعاؤه إذا لم يكن المتغير المكتشف موجودًا في قائمة الخيارات. تعيد هذه الروتينات واحدًا إذا نجحت، وصفرًا بخلاف ذلك.
bool_t xdr_vector(XDR *xdrs, char *arr, unsigned int len,
                  unsigned int maxsize, xdrproc_t elementproc);
مثل xdr_array()، ولكن مع وسيطات مختلفة قليلاً. الوسيط arr هو مؤشر إلى بداية المصفوفة. الوسيط len هو عدد العناصر في المصفوفة (الذي سيتم تعديله). الوسيط maxsize هو الحد الأقصى لعدد العناصر المسموح به. الوسيط elementproc هو روتين XDR لترجمة عناصر المصفوفة.
