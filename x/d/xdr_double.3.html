xdr(3) دليل دوال المكتبة xdr(3)

الاسم

xdr - روتينات مكتبية لتمثيل البيانات الخارجي

المكتبة

مكتبة C القياسية (libc، -lc)

الملخص والوصف

تسمح هذه الروتينات لمبرمجي C بوصف هياكل بيانات عشوائية بطريقة مستقلة عن الآلة. تُنقل بيانات استدعاءات الإجراءات البعيدة باستخدام هذه الروتينات.

النماذج الأولية أدناه معلنة في <rpc/xdr.h> وتستخدم الأنواع التالية:

typedef int bool_t;
typedef typeof(bool_t (XDR *, void *, ...))  *xdrproc_t;

للاطلاع على إعلان النوع XDR، راجع <rpc/xdr.h>.

bool_t xdr_array(XDR *xdrs, char **arrp, unsigned int *sizep,
                 unsigned int maxsize, unsigned int elsize,
                 xdrproc_t elproc);
بدائية مرشّح تترجم بين المصفوفات متغيرة الطول وتمثيلاتها الخارجية المقابلة. الوسيط arrp هو عنوان المؤشر إلى المصفوفة، بينما sizep هو عنوان عدد عناصر المصفوفة؛ لا يمكن أن يتجاوز عدد العناصر هذا maxsize. الوسيط elsize هو sizeof لكل عنصر من عناصر المصفوفة، وelproc هو مرشّح XDR يترجم بين الشكل C لعناصر المصفوفة وتمثيلها الخارجي. تُرجع هذه الروتينة واحدًا إذا نجحت، وصفرًا خلاف ذلك.
bool_t xdr_bool(XDR *xdrs, bool_t *bp);
بدائية مرشّح تترجم بين القيم المنطقية (أعداد C الصحيحة) وتمثيلاتها الخارجية. عند ترميز البيانات، ينتج هذا المرشّح قيمًا إما واحدًا أو صفرًا. تُرجع هذه الروتينة واحدًا إذا نجحت، وصفرًا خلاف ذلك.
bool_t xdr_bytes(XDR *xdrs, char **sp, unsigned int *sizep,
                 unsigned int maxsize);
بدائية مرشّح تترجم بين سلاسل البايتات المعدودة وتمثيلاتها الخارجية. الوسيط sp هو عنوان مؤشر السلسلة. يقع طول السلسلة في العنوان sizep؛ لا يمكن أن تكون السلاسل أطول من maxsize. تُرجع هذه الروتينة واحدًا إذا نجحت، وصفرًا خلاف ذلك.
bool_t xdr_char(XDR *xdrs, char *cp);
بدائية مرشّح تترجم بين محارف C وتمثيلاتها الخارجية. تُرجع هذه الروتينة واحدًا إذا نجحت، وصفرًا خلاف ذلك. ملاحظة: المحارف المرمّزة غير محزومة، وتشغل 4 بايتات لكل منها. بالنسبة لمصفوفات المحارف، يجدر التفكير في xdr_bytes() أو xdr_opaque() أو xdr_string().
void xdr_destroy(XDR *xdrs);
ماكرو يستدعي روتينة التدمير المرتبطة بتدفق XDR، xdrs. يتضمن التدمير عادةً تحرير هياكل البيانات الخاصة المرتبطة بالتدفق. استخدام xdrs بعد استدعاء xdr_destroy() غير معرّف.
bool_t xdr_double(XDR *xdrs, double *dp);
بدائية مرشّح تترجم بين أعداد C ذات الدقة المزدوجة وتمثيلاتها الخارجية. تُرجع هذه الروتينة واحدًا إذا نجحت، وصفرًا خلاف ذلك.
bool_t xdr_enum(XDR *xdrs, enum_t *ep);
بدائية مرشّح تترجم بين التعدادات في C (وهي في الواقع أعداد صحيحة) وتمثيلاتها الخارجية. تُرجع هذه الروتينة واحدًا إذا نجحت، وصفرًا خلاف ذلك.
bool_t xdr_float(XDR *xdrs, float *fp);
بدائية مرشّح تترجم بين أعداد C العائمة وتمثيلاتها الخارجية. تُرجع هذه الروتينة واحدًا إذا نجحت، وصفرًا خلاف ذلك.
void xdr_free(xdrproc_t proc, char *objp);
روتينة تحرير عامة. الوسيط الأول هو روتينة XDR للكائن الذي يتم تحريره. الوسيط الثاني هو مؤشر إلى الكائن نفسه. ملاحظة: المؤشر الممرَّر إلى هذه الروتينة لا يتم تحريره، لكن ما يشير إليه يتم تحريره (بشكل تكراري).
unsigned int xdr_getpos(XDR *xdrs);
ماكرو يستدعي روتينة الحصول على الموضع المرتبطة بتدفق XDR، xdrs. تُرجع الروتينة عددًا صحيحًا غير موقّع، يشير إلى موضع تدفق بايتات XDR. من الخصائص المرغوبة في تدفقات XDR أن الحساب البسيط يعمل مع هذا الرقم، رغم أن نسخ تدفق XDR ليست مضطرة لضمان ذلك.
long *xdr_inline(XDR *xdrs, int len);
ماكرو يستدعي الروتينة المضمّنة المرتبطة بتدفق XDR، xdrs. تُرجع الروتينة مؤشرًا إلى قطعة متجاورة من مخزن التدفق المؤقت؛ len هو طول البايتات للمخزن المطلوب. ملاحظة: يُحوَّل المؤشر إلى long *.
تحذير: قد تُرجع xdr_inline() القيمة NULL (0) إذا لم تتمكن من تخصيص قطعة متجاورة من المخزن المؤقت. لذلك قد يختلف السلوك بين نسخ التدفق؛ وهي موجودة من أجل الكفاءة.
bool_t xdr_int(XDR *xdrs, int *ip);
بدائية مرشّح تترجم بين أعداد C الصحيحة وتمثيلاتها الخارجية. تُرجع هذه الروتينة واحدًا إذا نجحت، وصفرًا خلاف ذلك.
bool_t xdr_long(XDR *xdrs, long *lp);
بدائية مرشّح تترجم بين أعداد C الصحيحة الطويلة وتمثيلاتها الخارجية. تُرجع هذه الروتينة واحدًا إذا نجحت، وصفرًا خلاف ذلك.
void xdrmem_create(XDR *xdrs, char *addr, unsigned int size,
                   enum xdr_op op);
تهيّئ هذه الروتينة كائن تدفق XDR المشار إليه بـ xdrs. تُكتب بيانات التدفق إلى قطعة من الذاكرة في الموقع addr لا يتجاوز طولها size بايت، أو تُقرأ منها. يحدد op اتجاه تدفق XDR (إما XDR_ENCODE أو XDR_DECODE أو XDR_FREE).
bool_t xdr_opaque(XDR *xdrs, char *cp, unsigned int cnt);
بدائية مرشّح تترجم بين البيانات المعتمة ثابتة الحجم وتمثيلها الخارجي. الوسيط cp هو عنوان الكائن المعتم، وcnt هو حجمه بالبايتات. تُرجع هذه الروتينة واحدًا إذا نجحت، وصفرًا خلاف ذلك.
bool_t xdr_pointer(XDR *xdrs, char **objpp,
                   unsigned int objsize, xdrproc_t xdrobj);
مثل xdr_reference() باستثناء أنه يُسلسل المؤشرات الفارغة، بينما xdr_reference() لا يفعل ذلك. وبالتالي، يمكن لـ xdr_pointer() تمثيل هياكل بيانات تكرارية، مثل الأشجار الثنائية أو القوائم المترابطة.
void xdrrec_create(XDR *xdrs, unsigned int sendsize,
                   unsigned int recvsize, char *handle,
                   typeof(int (char *, char *, int)) *readit,
                   typeof(int (char *, char *, int)) *writeit);
تهيّئ هذه الروتينة كائن تدفق XDR المشار إليه بـ xdrs. تُكتب بيانات التدفق إلى مخزن مؤقت بحجم sendsize؛ تشير القيمة صفر إلى أنه ينبغي للنظام استخدام قيمة افتراضية مناسبة. تُقرأ بيانات التدفق من مخزن مؤقت بحجم recvsize؛ ويمكن أيضًا تعيينه إلى قيمة افتراضية مناسبة بتمرير القيمة صفر. عندما يمتلئ مخزن الإخراج المؤقت للتدفق، يُستدعى writeit. وبالمثل، عندما يفرغ مخزن الإدخال المؤقت للتدفق، يُستدعى readit. سلوك هاتين الروتينتين مشابه لاستدعاءات النظام read(2) وwrite(2)، باستثناء أن handle يُمرَّر إلى الروتينتين السابقتين كوسيط أول. ملاحظة: يجب أن يضبط المستدعي حقل op لتدفق XDR.
تحذير: للقراءة من تدفق XDR أُنشئ بواجهة API هذه، ستحتاج إلى استدعاء xdrrec_skiprecord() أولًا قبل استدعاء أي واجهات XDR أخرى. يؤدي هذا إلى إدراج بايتات إضافية في التدفق لتوفير حدود السجل

هذا الموقع غير مرتبط بـ Arch Linux ولا تابع له بأي شكل من الأشكال. المحتوى هو ترجمة فقط لصفحات دليل Arch المنشورة على man.archlinux.org.

المعلومات. أيضًا، تدفقات XDR التي تُنشأ باستخدام واجهات xdr*_create المختلفة غير متوافقة للسبب نفسه. 
bool_t xdrrec_endofrecord(XDR *xdrs, int sendnow);
لا يمكن استدعاء هذا الروتين إلا على التدفقات التي تُنشأ بواسطة xdrrec_create(). تُعلَّم البيانات الموجودة في مخزن الإخراج المؤقت كسجل مكتمل، ويُكتب مخزن الإخراج المؤقت اختياريًا إذا كانت sendnow غير صفرية. يُرجع هذا الروتين واحدًا إذا نجح، وصفرًا بخلاف ذلك.
bool_t xdrrec_eof(XDR *xdrs);
لا يمكن استدعاء هذا الروتين إلا على التدفقات التي تُنشأ بواسطة xdrrec_create(). بعد استهلاك بقية السجل الحالي في التدفق، يُرجع هذا الروتين واحدًا إذا لم يكن للتدفق مزيد من المدخلات، وصفرًا بخلاف ذلك.
bool_t xdrrec_skiprecord(XDR *xdrs);
لا يمكن استدعاء هذا الروتين إلا على التدفقات التي تُنشأ بواسطة xdrrec_create(). يخبر تنفيذ XDR بأن بقية السجل الحالي في مخزن المدخلات المؤقت للتدفق ينبغي التخلص منها. يُرجع هذا الروتين واحدًا إذا نجح، وصفرًا بخلاف ذلك.
bool_t xdr_reference(XDR *xdrs, char **pp, unsigned int size,
                     xdrproc_t proc);
عنصر أولي يوفر تتبع المؤشرات داخل البنى. الوسيط pp هو عنوان المؤشر؛ وsize هو sizeof البنية التي يشير إليها *pp؛ وproc هو إجراء XDR يرشّح البنية بين صيغتها بلغة C وتمثيلها الخارجي. يُرجع هذا الروتين واحدًا إذا نجح، وصفرًا بخلاف ذلك.
تحذير: هذا الروتين لا يفهم المؤشرات الفارغة. استخدم xdr_pointer() بدلًا من ذلك.
xdr_setpos(XDR *xdrs, unsigned int pos);
ماكرو يستدعي روتين تعيين الموضع المرتبط بتدفق XDR xdrs. الوسيط pos هو قيمة موضع تم الحصول عليها من xdr_getpos(). يُرجع هذا الروتين واحدًا إذا أمكن إعادة تحديد موضع تدفق XDR، وصفرًا بخلاف ذلك.
تحذير: من الصعب إعادة تحديد موضع بعض أنواع تدفقات XDR، لذا قد يفشل هذا الروتين مع نوع من التدفقات وينجح مع نوع آخر.
bool_t xdr_short(XDR *xdrs, short *sp);
عنصر ترشيح أولي يترجم بين أعداد C الصحيحة من نوع short و تمثيلاتها الخارجية. يُرجع هذا الروتين واحدًا إذا نجح، وصفرًا بخلاف ذلك.
void xdrstdio_create(XDR *xdrs, FILE *file, enum xdr_op op);
يهيّئ هذا الروتين كائن تدفق XDR الذي يشير إليه xdrs. تُكتب بيانات تدفق XDR إلى تدفق stdio المسمى file أو تُقرأ منه. يحدد الوسيط op اتجاه تدفق XDR (إما XDR_ENCODE أو XDR_DECODE أو XDR_FREE).
تحذير: روتين التدمير المرتبط بمثل هذه التدفقات XDR يستدعي fflush(3) على تدفق file، لكنه لا يستدعي أبدًا fclose(3).
bool_t xdr_string(XDR *xdrs, char **sp, unsigned int maxsize);
عنصر ترشيح أولي يترجم بين سلاسل C و تمثيلاتها الخارجية المقابلة. لا يمكن أن تكون السلاسل أطول من maxsize. ملاحظة: sp هو عنوان مؤشر السلسلة. يُرجع هذا الروتين واحدًا إذا نجح، وصفرًا بخلاف ذلك.
bool_t xdr_u_char(XDR *xdrs, unsigned char *ucp);
عنصر ترشيح أولي يترجم بين أحرف C غير الموقعة وتمثيلاتها الخارجية. يُرجع هذا الروتين واحدًا إذا نجح، وصفرًا بخلاف ذلك.
bool_t xdr_u_int(XDR *xdrs, unsigned int *up);
عنصر ترشيح أولي يترجم بين أعداد C الصحيحة غير الموقعة و تمثيلاتها الخارجية. يُرجع هذا الروتين واحدًا إذا نجح، وصفرًا بخلاف ذلك.
bool_t xdr_u_long(XDR *xdrs, unsigned long *ulp);
عنصر ترشيح أولي يترجم بين أعداد C الصحيحة غير الموقعة من نوع long وتمثيلاتها الخارجية. يُرجع هذا الروتين واحدًا إذا نجح، وصفرًا بخلاف ذلك.
bool_t xdr_u_short(XDR *xdrs, unsigned short *usp);
عنصر ترشيح أولي يترجم بين أعداد C الصحيحة غير الموقعة من نوع short وتمثيلاتها الخارجية. يُرجع هذا الروتين واحدًا إذا نجح، وصفرًا بخلاف ذلك.
bool_t xdr_union(XDR *xdrs, enum_t *dscmp, char *unp,
                 const struct xdr_discrim *choices,
                 xdrproc_t defaultarm);     /* may equal NULL */
عنصر ترشيح أولي يترجم بين اتحاد C مميّز وتمثيله الخارجي المقابل. يترجم أولًا مميّز الاتحاد الموجود عند dscmp. هذا المميّز دائمًا من نوع enum_t. بعد ذلك يُترجم الاتحاد الموجود عند unp. الوسيط choices هو مؤشر إلى مصفوفة من بنى xdr_discrim(). تحتوي كل بنية على زوج مرتب من [value,proc]. إذا كان مميّز الاتحاد مساويًا لـ value المرتبطة، فسيُستدعى proc لترجمة الاتحاد. يُشار إلى نهاية مصفوفة بنى xdr_discrim() بروتين قيمته NULL. إذا لم يُعثر على المميّز في مصفوفة choices، فسيُستدعى إجراء defaultarm (إذا لم يكن NULL). يُرجع واحدًا إذا نجح، وصفرًا بخلاف ذلك.
bool_t xdr_vector(XDR *xdrs, char *arrp, unsigned int size,
                  unsigned int elsize, xdrproc_t elproc);
عنصر ترشيح أولي يترجم بين المصفوفات ذات الطول الثابت و تمثيلاتها الخارجية المقابلة. الوسيط arrp هو عنوان المؤشر إلى المصفوفة، بينما size هو عدد عناصر المصفوفة. الوسيط elsize هو sizeof لكل عنصر من عناصر المصفوفة، وelproc هو مرشح XDR يترجم بين صيغة C لعناصر المصفوفة وتمثيلها الخارجي. يُرجع هذا الروتين واحدًا إذا نجح، وصفرًا بخلاف ذلك.
bool_t xdr_void(void);
يُرجع هذا الروتين دائمًا واحدًا. يمكن تمريره إلى روتينات RPC التي تتطلب وسيط دالة، حيث لا شيء يُفعل.
bool_t xdr_wrapstring(XDR *xdrs, char **sp);
عنصر أولي يستدعي xdr_string(xdrs, sp,MAXUN.UNSIGNED ); حيث MAXUN.UNSIGNED هو القيمة العظمى لعدد صحيح غير موقّع. xdr_wrapstring() مفيد لأن حزمة RPC تمرر حدًا أقصى من روتيني XDR كوسائط، وxdr_string()، أحد أكثر العناصر الأولية استخدامًا، يتطلب ثلاثة. يُرجع واحدًا إذا نجح، وصفرًا بخلاف ذلك.
