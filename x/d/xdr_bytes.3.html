الاسم

xdr - روتينات مكتبة لتمثيل البيانات الخارجي

المكتبة

مكتبة C القياسية (libc, -lc)

الملخص والوصف

تسمح هذه الروتينات لمبرمجي C بوصف هياكل البيانات العشوائية بطريقة مستقلة عن الآلة. يتم نقل البيانات لاستدعاء الإجراءات عن بعد باستخدام هذه الروتينات.

النماذج الأولية أدناه معلنة في <rpc/xdr.h> وتستخدم الأنواع التالية:

typedef int bool_t;
typedef typeof(bool_t (XDR *, void *, ...))  *xdrproc_t;

لإعلان النوع XDR، انظر <rpc/xdr.h>.

bool_t xdr_array(XDR *xdrs, char **arrp, unsigned int *sizep,
                 unsigned int maxsize, unsigned int elsize,
                 xdrproc_t elproc);
بدائية مرشح تترجم بين المصفوفات ذات الطول المتغير وتمثيلاتها الخارجية المقابلة. الوسيط arrp هو عنوان المؤشر إلى المصفوفة، بينما sizep هو عنوان عداد عناصر المصفوفة؛ لا يمكن أن يتجاوز هذا العداد maxsize. الوسيط elsize هو sizeof كل عنصر من عناصر المصفوفة، و elproc هو مرشح XDR يترجم بين شكل C لعناصر المصفوفة وتمثيلها الخارجي. تعيد هذه الدالة واحدًا إذا نجحت، وصفرًا إذا فشلت.
bool_t xdr_bool(XDR *xdrs, bool_t *bp);
بدائية مرشح تترجم بين القيم المنطقية (أعداد صحيحة C) وتمثيلاتها الخارجية. عند ترميز البيانات، تنتج هذه المرشحة قيمًا إما واحدًا أو صفرًا. تعيد هذه الدالة واحدًا إذا نجحت، وصفرًا إذا فشلت.
bool_t xdr_bytes(XDR *xdrs, char **sp, unsigned int *sizep,
                 unsigned int maxsize);
بدائية مرشح تترجم بين سلاسل البايتات المعدودة وتمثيلاتها الخارجية. الوسيط sp هو عنوان مؤشر السلسلة. طول السلسلة موجود في العنوان sizep؛ لا يمكن أن تكون السلاسل أطول من maxsize. تعيد هذه الدالة واحدًا إذا نجحت، وصفرًا إذا فشلت.
bool_t xdr_char(XDR *xdrs, char *cp);
بدائية مرشح تترجم بين أحرف C وتمثيلاتها الخارجية. تعيد هذه الدالة واحدًا إذا نجحت، وصفرًا إذا فشلت. ملاحظة: الأحرف المرمزة ليست مكدسة، وتشغل 4 بايتات لكل منها. بالنسبة لمصفوفات الأحرف، يجدر النظر في xdr_bytes() أو xdr_opaque() أو xdr_string().
void xdr_destroy(XDR *xdrs);
ماكرو يستدعي روتين الإتلاف المرتبط بتيار XDR، xdrs. يشمل الإتلاف عادةً تحرير هياكل البيانات الخاصة المرتبطة بالتيار. استخدام xdrs بعد استدعاء xdr_destroy() غير محدد.
bool_t xdr_double(XDR *xdrs, double *dp);
بدائية مرشح تترجم بين أرقام double الدقة في C وتمثيلاتها الخارجية. تعيد هذه الدالة واحدًا إذا نجحت، وصفرًا إذا فشلت.
bool_t xdr_enum(XDR *xdrs, enum_t *ep);
بدائية مرشح تترجم بين enums في C (في الواقع أعداد صحيحة) وتمثيلاتها الخارجية. تعيد هذه الدالة واحدًا إذا نجحت، وصفرًا إذا فشلت.
bool_t xdr_float(XDR *xdrs, float *fp);
بدائية مرشح تترجم بين floats في C وتمثيلاتها الخارجية. تعيد هذه الدالة واحدًا إذا نجحت، وصفرًا إذا فشلت.
void xdr_free(xdrproc_t proc, char *objp);
روتين تحرير عام. الوسيط الأول هو روتين XDR للكائن الذي يتم تحريره. الوسيط الثاني هو مؤشر إلى الكائن نفسه. ملاحظة: المؤشر المقدم إلى هذا الروتين *لا* يتم تحريره، ولكن ما يشير إليه *يتم* تحريره (بشكل متكرر).
unsigned int xdr_getpos(XDR *xdrs);
ماكرو يستدعي روتين الحصول على الموضع المرتبط بتيار XDR، xdrs. يعيد الروتين عددًا صحيحًا غير موقّع، والذي يشير إلى موضع تيار بايتات XDR. ميزة مرغوبة لتيارات XDR هي أن العمليات الحسابية البسيطة تعمل مع هذا الرقم، على الرغم من أن مثيلات تيار XDR قد لا تضمن ذلك.
long *xdr_inline(XDR *xdrs, int len);
ماكرو يستدعي الروتين المضمن المرتبط بتيار XDR، xdrs. يعيد الروتين مؤشرًا إلى قطعة متجاورة من مخزن XDR المؤقت؛ len هو طول البايت للمخزن المطلوب. ملاحظة: يتم تحويل المؤشر إلى long *.
تحذير: قد يعيد xdr_inline() قيمة NULL (0) إذا لم يتمكن من تخصيص قطعة متجاورة من مخزن مؤقت. لذلك قد يختلف السلوك بين مثيلات التيار؛ وهو موجود من أجل الكفاءة.
bool_t xdr_int(XDR *xdrs, int *ip);
بدائية مرشح تترجم بين الأعداد الصحيحة في C وتمثيلاتها الخارجية. تعيد هذه الدالة واحدًا إذا نجحت، وصفرًا إذا فشلت.
bool_t xdr_long(XDR *xdrs, long *lp);
بدائية مرشح تترجم بين الأعداد الصحيحة long في C وتمثيلاتها الخارجية. تعيد هذه الدالة واحدًا إذا نجحت، وصفرًا إذا فشلت.
void xdrmem_create(XDR *xdrs, char *addr, unsigned int size,
                   enum xdr_op op);
هذا الروتين يقوم بتهيئة كائن تيار XDR الذي يشير إليه xdrs. تتم كتابة بيانات التيار، أو قراءتها، من جزء من الذاكرة في الموقع addr والذي لا يزيد طوله عن size بايت. يحدد op اتجاه تيار XDR (إما XDR_ENCODE أو XDR_DECODE أو XDR_FREE).
bool_t xdr_opaque(XDR *xdrs, char *cp, unsigned int cnt);
بدائية مرشح تترجم بين البيانات المعتمة ذات الحجم الثابت وتمثيلها الخارجي. الوسيط cp هو عنوان الكائن المعتم، و cnt هو حجمه بالبايت. تعيد هذه الدالة واحدًا إذا نجحت، وصفرًا إذا فشلت.
bool_t xdr_pointer(XDR *xdrs, char **objpp,
                   unsigned int objsize, xdrproc_t xdrobj);
مثل xdr_reference() باستثناء أنها تسلسل مؤشرات فارغة، بينما xdr_reference() لا تفعل ذلك. وبالتالي، يمكن لـ xdr_pointer() تمثيل هياكل البيانات المتكررة، مثل الأشجار الثنائية أو القوائم المرتبطة.
void xdrrec_create(XDR *xdrs, unsigned int sendsize,
                   unsigned int recvsize, char *handle,
                   typeof(int (char *, char *, int)) *readit,
                   typeof(int (char *, char *, int)) *writeit);
هذا الروتين يقوم بتهيئة كائن تيار XDR الذي يشير إليه xdrs. تتم كتابة بيانات التيار إلى مخزن مؤقت بحجم sendsize؛ قيمة صفر تشير إلى أن النظام يجب أن يستخدم قيمة افتراضية مناسبة. تتم قراءة بيانات التيار من مخزن مؤقت بحجم recvsize؛ يمكن أيضًا تعيينه على قيمة افتراضية مناسبة عن طريق تمرير قيمة صفر. عندما يكون المخزن المؤقت للإخراج للتيار ممتلئًا، يتم استدعاء writeit. وبالمثل، عندما يكون المخزن المؤقت للإدخال للتيار فارغًا، يتم استدعاء readit. سلوك هذين الروتينين مشابه لاستدعاءات النظام read(2) و write(2)، باستثناء أنه يتم تمرير handle إلى الروتينات السابقة كوسيط أول. ملاحظة: يجب على المتصل تعيين حقل op لتيار XDR.
تحذير: لقراءة تيار XDR تم إنشاؤه بواسطة واجهة برمجة التطبيقات هذه، ستحتاج إلى استدعاء xdrrec_skiprecord() أولاً قبل استدعاء أي واجهات برمجة تطبيقات XDR أخرى. هذا يدرج بايتات إضافية في التيار لتوفير حدود السجل.
bool_t xdr_reference(XDR *xdrs, char **pp, unsigned int size,
                     xdrproc_t proc);
هذه الدالة تشبه xdr_array() ولكنها تسلسل مرجعًا بدلاً من مصفوفة. الوسيط pp هو عنوان المؤشر إلى الكائن، size هو حجم الكائن بالبايت، و proc هو روتين XDR المستخدم لتسلسل الكائن. تعيد هذه الدالة واحدًا إذا نجحت، وصفرًا إذا فشلت.
bool_t xdr_register(XDR *xdrs, void *obj, xdrproc_t encode,
                    xdrproc_t decode);
غير مستخدم.
bool_t xdr_replymsg(XDR *xdrs, struct rpc_msg *rmsg);
هذه الدالة تسلسل رسالة رد RPC. الوسيط xdrs هو مؤشر إلى بنية XDR، و rmsg هو مؤشر إلى رسالة الرد.
void xdrrec_skiprecord(XDR *xdrs);
يتجاهل هذا الروتين بيانات غير محددة حتى نهاية السجل الحالي. هذا مفيد لتجاوز سجلات XDR غير المطلوبة. يمكن استدعاؤه فقط عند قراءة تيار XDR تم إنشاؤه بواسطة xdrrec_create().
bool_t xdr_setpos(XDR *xdrs, unsigned int pos);
ماكرو يستدعي روتين تعيين الموضع المرتبط بتيار XDR، xdrs. يقوم هذا الروتين بتعيين الموضع الحالي لتيار XDR إلى pos. تعيد هذه الدالة واحدًا إذا نجحت، وصفرًا إذا فشلت.

bool_t xdr_union(XDR *xdrs, enum_t *discr, char *h,
                 const struct xdr_disc_byname *choices,
                 xdrproc_t default);
بدائية مرشح تترجم بين اتحادات C وتمثيلاتها الخارجية. الوسيط discr هو مؤشر إلى محدد الاتحاد، h هو مؤشر إلى الاتحاد، choices هو قائمة بالخيارات، و default هو روتين XDR الافتراضي. تعيد هذه الدالة واحدًا إذا نجحت، وصفرًا إذا فشلت.

bool_t xdr_void(XDR *xdrs);
لا تفعل شيئًا، وتعيد دائمًا واحدًا. مفيدة كعنصر نائب لـ xdrproc_t عند عدم الحاجة إلى تسلسل أي شيء.
bool_t xdr_wrapstring(XDR *xdrs, char **sp);
متماثلة مع xdr_string() ولكنها تقوم بتخصيص الذاكرة اللازمة للسلسلة. هذا يعني أنه يجب تحرير الذاكرة باستخدام xdr_free() لاحقًا.
