xdr — إجراءات مكتبة لتمثيل البيانات الخارجي

المكتبة

مكتبة C القياسية (libc، -lc)

الملخص والوصف

تسمح هذه الإجراءات لبرامج C بوصف هياكل بيانات عشوائية بطريقة مستقلة عن الآلة. يتم نقل البيانات لاستدعاءات الإجراءات عن بعد باستخدام هذه الإجراءات.

الإعلانات النموذجية أدناه معرفة في <rpc/xdr.h> وتستخدم الأنواع التالية:

typedef int bool_t;
typedef typeof(bool_t (XDR *, void *, ...))  *xdrproc_t;

لإعلان نوع XDR، انظر <rpc/xdr.h>.

bool_t xdr_array(XDR *xdrs, char **arrp, unsigned int *sizep,
                 unsigned int maxsize, unsigned int elsize,
                 xdrproc_t elproc);
مرشح بدائي يترجم بين المصفوفات متغيرة الطول وتمثيلياتها الخارجية المقابلة. الوسيط arrp هو عنوان المؤشر إلى المصفوفة، بينما sizep هو عنوان عدد العناصر في المصفوفة؛ لا يمكن أن يتجاوز عدد العناصر هذا maxsize. الوسيط elsize هو sizeof لكل عنصر من عناصر المصفوفة، و elproc هو مرشح XDR يترجم بين شكل C لعناصر المصفوفة وتمثيلها الخارجي. تعيد هذه الوظيفة واحدًا إذا نجحت، وصفرًا بخلاف ذلك.
bool_t xdr_bool(XDR *xdrs, bool_t *bp);
مرشح بدائي يترجم بين القيم المنطقية (أعداد صحيحة C) وتمثيلاتها الخارجية. عند ترميز البيانات، ينتج هذا المرشح قيمًا إما واحد أو صفر. تعيد هذه الوظيفة واحدًا إذا نجحت، وصفرًا بخلاف ذلك.
bool_t xdr_bytes(XDR *xdrs, char **sp, unsigned int *sizep,
                 unsigned int maxsize);
مرشح بدائي يترجم بين سلاسل البايت المعدودة وتمثيلاتها الخارجية. الوسيط sp هو عنوان مؤشر السلسلة. يقع طول السلسلة في العنوان sizep؛ لا يمكن أن تكون السلاسل أطول من maxsize. تعيد هذه الوظيفة واحدًا إذا نجحت، وصفرًا بخلاف ذلك.
bool_t xdr_char(XDR *xdrs, char *cp);
مرشح بدائي يترجم بين أحرف C وتمثيلاتها الخارجية. تعيد هذه الوظيفة واحدًا إذا نجحت، وصفرًا بخلاف ذلك. ملاحظة: لا يتم حزم الأحرف المرمزة، وتشغل 4 بايت لكل منها. بالنسبة لمصفوفات الأحرف، يجدر النظر في xdr_bytes() أو xdr_opaque() أو xdr_string().
void xdr_destroy(XDR *xdrs);
وحدة ماكرو تستدعي إجراء التدمير المرتبط بتيار XDR، xdrs. يشمل التدمير عادةً تحرير هياكل البيانات الخاصة المرتبطة بالتيار. استخدام xdrs بعد استدعاء xdr_destroy() غير محدد.

bool_t xdr_float(XDR *xdrs, float *fp);

void xdr_free(xdrproc_t proc, char *objp);
إجراء تحرير عام. الوسيط الأول هو إجراء XDR للكائن الذي يتم تحريره. الوسيط الثاني هو مؤشر إلى الكائن نفسه. ملاحظة: المؤشر الممرر إلى هذه الوظيفة *لا* يتم تحريره، ولكن ما يشير إليه *يتم* تحريره (بشكل متكرر).
unsigned int xdr_getpos(XDR *xdrs);
وحدة ماكرو تستدعي إجراء الحصول على الموضع المرتبط بتيار XDR، xdrs. تعيد الوظيفة عددًا صحيحًا غير موقع، والذي يشير إلى موضع تيار بايت XDR. ميزة مرغوبة في تيارات XDR هي أن العمليات الحسابية البسيطة تعمل مع هذا الرقم، على الرغم من أن مثيلات تيار XDR قد لا تضمن ذلك.
long *xdr_inline(XDR *xdrs, int len);
وحدة ماكرو تستدعي إجراء الاستدعاء المضمن المرتبط بتيار XDR، xdrs. تعيد الوظيفة مؤشرًا إلى جزء متجاور من مخزن مؤقت للتيار؛ len هو طول البايت للمخزن المؤقت المطلوب. ملاحظة: يتم تحويل المؤشر إلى long *.
تحذير: قد تعيد xdr_inline() القيمة NULL (0) إذا لم تتمكن من تخصيص جزء متجاور من المخزن المؤقت. لذلك قد يختلف السلوك بين مثيلات التيار؛ موجود من أجل الكفاءة.
bool_t xdr_int(XDR *xdrs, int *ip);
مرشح بدائي يترجم بين الأعداد الصحيحة C وتمثيلاتها الخارجية. تعيد هذه الوظيفة واحدًا إذا نجحت، وصفرًا بخلاف ذلك.
bool_t xdr_long(XDR *xdrs, long *lp);
مرشح بدائي يترجم بين الأعداد الصحيحة الطويلة C وتمثيلاتها الخارجية. تعيد هذه الوظيفة واحدًا إذا نجحت، وصفرًا بخلاف ذلك.
void xdrmem_create(XDR *xdrs, char *addr, unsigned int size,
                   enum xdr_op op);
تقوم هذه الوظيفة بتهيئة كائن تيار XDR المشار إليه بواسطة xdrs. يتم كتابة بيانات التيار، أو قراءتها، من جزء من الذاكرة في الموقع addr الذي لا يزيد طوله عن size بايت. يحدد op اتجاه تيار XDR (إما XDR_ENCODE أو XDR_DECODE أو XDR_FREE).
bool_t xdr_opaque(XDR *xdrs, char *cp, unsigned int cnt);
مرشح بدائي يترجم بين البيانات غير الشفافة ذات الحجم الثابت وتمثيلها الخارجي. الوسيط cp هو عنوان الكائن غير الشفاف، و cnt هو حجمه بالبايت. تعيد هذه الوظيفة واحدًا إذا نجحت، وصفرًا بخلاف ذلك.
bool_t xdr_pointer(XDR *xdrs, char **objpp,
                   unsigned int objsize, xdrproc_t xdrobj);
مثل xdr_reference() باستثناء أنها تسلسل مؤشرات فارغة، بينما xdr_reference() لا تفعل ذلك. وبالتالي، يمكن لـ xdr_pointer() تمثيل هياكل البيانات المتكررة، مثل الأشجار الثنائية أو القوائم المترابطة.
void xdrrec_create(XDR *xdrs, unsigned int sendsize,
                   unsigned int recvsize, char *handle,
                   typeof(int (char *, char *, int)) *readit,
                   typeof(int (char *, char *, int)) *writeit);
تقوم هذه الوظيفة بتهيئة كائن تيار XDR المشار إليه بواسطة xdrs. يتم كتابة بيانات التيار إلى مخزن مؤقت بحجم sendsize؛ القيمة الصفرية تشير إلى أن النظام يجب أن يستخدم قيمة افتراضية مناسبة. يتم قراءة بيانات التيار من مخزن مؤقت بحجم recvsize؛ يمكن أيضًا تعيين هذا إلى قيمة افتراضية مناسبة عن طريق تمرير قيمة صفرية. عندما يمتلئ المخزن المؤقت للإخراج للتيار، يتم استدعاء writeit. وبالمثل، عندما يكون المخزن المؤقت للإدخال للتيار فارغًا، يتم استدعاء readit. سلوك هاتين الوظيفتين مشابه لاستدعاءات النظام read(2) و write(2)، باستثناء أن handle يتم تمريره إلى الروتينات السابقة كوسيط أول. ملاحظة: يجب على المستدعي تعيين حقل op لتيار XDR.
تحذير: لقراءة تيار XDR تم إنشاؤه بواسطة واجهة برمجة التطبيقات هذه، ستحتاج إلى استدعاء xdrrec_skiprecord() أولاً قبل استدعاء أي واجهات برمجة تطبيقات XDR أخرى. هذا يضيف بايتات إضافية في التيار لتوفير حدود السجل.
bool_t xdr_reference(XDR *xdrs, char **pp, unsigned int size,
                     xdrproc_t proc);
مثل xdr_pointer() باستثناء أنها لا تسلسل مؤشرات فارغة. الوسيط pp هو عنوان المؤشر إلى الكائن، و size هو حجم الكائن بالبايت، و proc هو إجراء XDR للكائن. تعيد هذه الوظيفة واحدًا إذا نجحت، وصفرًا بخلاف ذلك.
bool_t xdr_setpos(XDR *xdrs, unsigned int pos);
وحدة ماكرو تستدعي إجراء تعيين الموضع المرتبط بتيار XDR، xdrs. يتم تعيين موضع تيار بايت XDR إلى pos. تعيد هذه الوظيفة واحدًا إذا نجحت، وصفرًا بخلاف ذلك.
bool_t xdr_short(XDR *xdrs, short *sp);
مرشح بدائي يترجم بين الأعداد الصحيحة القصيرة C وتمثيلاتها الخارجية. تعيد هذه الوظيفة واحدًا إذا نجحت، وصفرًا بخلاف ذلك.
char *xdr_sizeof(xdrproc_t proc, char *objp);
إجراء يقوم بإرجاع حجم تمثيل XDR للكائن المشار إليه بواسطة objp، باستخدام proc كإجراء XDR للكائن. تعيد الوظيفة حجم الكائن بالبايت، أو -1 إذا لم يكن بالإمكان تحديد الحجم.
bool_t xdr_string(XDR *xdrs, char **sp, unsigned int maxsize);

bool_t xdr_u_int32_t(XDR *xdrs, u_int32_t *uip);
مرشح بدائي يترجم بين الأعداد الصحيحة غير الموقعة ذات 32 بت C وتمثيلاتها الخارجية. تعيد هذه الوظيفة واحدًا إذا نجحت، وصفرًا بخلاف ذلك.

bool_t xdr_u_quad_t(XDR *xdrs, u_quad_t *uq);

مرشح بدائي يترجم بين الأعداد الصحيحة ذات 32 بت غير الموقعة C وتمثيلاتها الخارجية. تعيد هذه الوظيفة واحدًا إذا نجحت، وصفرًا بخلاف ذلك.

bool_t xdr_union(XDR *xdrs, enum_t *discr, char *fmtp,
                 const struct xdr_discunion *table,
                 typeof(bool_t (XDR *, void *, xdrproc_t)) *defaultops);
مرشح بدائي يترجم بين اتحاد C وتمثيله الخارجي. الوسيط discr هو مؤشر إلى مميز الاتحاد. الوسيط fmtp هو مؤشر إلى نموذج الاتحاد. table هو جدول يبين إجراءات XDR التي سيتم استخدامها لتمثيل كل حالة من حالات الاتحاد. defaultops هو مؤشر إلى إجراءات XDR الافتراضية التي سيتم استخدامها إذا لم يكن هناك مطابق في table.
bool_t xdr_void(void);
لا تفعل شيئًا؛ تعيد دائمًا واحدًا. هذه الوظيفة مفيدة كعنصر نائب عندما لا يكون هناك شيء لترميزه أو فك ترميزه.
bool_t xdr_wrappedstring(XDR *xdrs, char **sp);

bool_t xdr_write_bytes(XDR *xdrs, char *cp, unsigned int count);

يمكن استدعاء هذا الروتين فقط على السجلات التي تم إنشاؤها بواسطة xdrrec_create(). يتم تمييز البيانات الموجودة في المخزن المؤقت للإخراج كسجل مكتمل، ويتم كتابة المخزن المؤقت للإخراج اختياريًا إذا كانت sendnow غير صفرية. يعيد هذا الروتين واحدًا إذا نجح، وصفرًا إذا فشل.

bool_t xdrrec_skiprecord(XDR *xdrs);

bool_t xdr_reference(XDR *xdrs, char **pp, unsigned int size,
                     xdrproc_t proc);
بدائية توفر مطاردة المؤشرات داخل الهياكل. الوسيط pp هو عنوان المؤشر؛ size هو sizeof الهيكل الذي تشير إليه *pp؛ و proc هو إجراء XDR يقوم بتصفية الهيكل بين شكله C وتمثيله الخارجي. يعيد هذا الروتين واحدًا إذا نجح، وصفرًا بخلاف ذلك.

تحذير: من الصعب إعادة تعيين بعض أنواع سجلات XDR، لذلك قد يفشل هذا الروتين مع نوع واحد من السجلات وينجح مع نوع آخر.

void xdrstdio_create(XDR *xdrs, FILE *file, enum xdr_op op);

                 const struct xdr_discrim *choices,
                 xdrproc_t defaultarm);     /* قد تكون NULL */
بدائية تصفية تترجم بين union C المميز وتمثيله الخارجي المقابل. تقوم أولاً بترجمة المميز للاتحاد الواقع في dscmp. هذا المميز هو دائمًا enum_t. بعد ذلك، يتم ترجمة الاتحاد الواقع في unp. الوسيط choices هو مؤشر إلى مصفوفة من هياكل xdr_discrim(). يحتوي كل هيكل على زوج مرتب من [value،proc]. إذا كان مميز الاتحاد يساوي value المرتبط، يتم استدعاء proc لترجمة الاتحاد. يتم تحديد نهاية مصفوفة هيكل xdr_discrim() بواسطة روتين بقيمة NULL. إذا لم يتم العثور على المميز في مصفوفة choices، يتم استدعاء إجراء defaultarm (إذا لم يكن NULL). يعيد واحدًا إذا نجح، وصفرًا بخلاف ذلك.
bool_t xdr_vector(XDR *xdrs, char *arrp, unsigned int size,
                  unsigned int elsize, xdrproc_t elproc);
بدائية تصفية تترجم بين المصفوفات ذات الطول الثابت وتمثيلاتها الخارجية المقابلة. الوسيط arrp هو عنوان المؤشر إلى المصفوفة، بينما size هو عدد عناصر المصفوفة. الوسيط elsize هو sizeof لكل عنصر من عناصر المصفوفة، و elproc هو مرشح XDR يترجم بين شكل C لعناصر المصفوفة وتمثيلها الخارجي. يعيد هذا الروتين واحدًا إذا نجح، وصفرًا بخلاف ذلك.
bool_t xdr_void(void);
يعيد هذا الروتين واحدًا دائمًا. يمكن تمريره إلى روتينات RPC التي تتطلب وسيط دالة، حيث لا يلزم القيام بأي شيء.
bool_t xdr_wrapstring(XDR *xdrs, char **sp);
بدائية تستدعي xdr_string(xdrs, sp,MAXUN.UNSIGNED ); حيث MAXUN.UNSIGNED هي القيمة القصوى لعدد صحيح غير موقع. xdr_wrapstring() مفيدة لأن حزمة RPC تمرر ما يصل إلى روتينين XDR كوسائط، و xdr_string()، وهي إحدى البدائيات الأكثر استخدامًا، تتطلب ثلاثة. يعيد واحدًا إذا نجح، وصفرًا بخلاف ذلك.
