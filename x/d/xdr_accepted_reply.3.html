الاسم

rpc - روتينات مكتبة لاستدعاءات الإجراءات عن بعد

المكتبة

مكتبة C القياسية (libc، -lc)

ملخص الوصف

تسمح هذه الروتينات لبرامج C بإجراء استدعاءات إجراءات على آلات أخرى عبر الشبكة. أولاً، يستدعي العميل إجراءً لإرسال حزمة بيانات إلى الخادم. عند استلام الحزمة، يستدعي الخادم روتين توزيع لتنفيذ الخدمة المطلوبة، ثم يرسل رداً. أخيراً، تعود استدعاء الإجراء إلى العميل.

للاستفادة من هذه الروتينات، قم بتضمين ملف الرأس

<rpc/rpc.h>.

تستخدم النماذج الأولية أدناه الأنواع التالية:

typedef int bool_t;
typedef typeof(bool_t (XDR *, void *, ...))  *xdrproc_t;
typedef typeof(bool_t (caddr_t resp, struct sockaddr_in *raddr)
    *resultproc_t;

راجع ملفات الرأس للإعلانات الخاصة بأنواع AUTH وCLIENT وSVCXPRT وXDR.

void auth_destroy(AUTH *auth);
وحدة ماكرو تدمر معلومات المصادقة المرتبطة بـ auth. عادةً ما يتضمن التدمير إلغاء تخصيص هياكل البيانات الخاصة. استخدام auth غير محدد بعد استدعاء auth_destroy().
AUTH *authnone_create(void);
ينشئ ويعيد مقبض مصادقة RPC الذي يمرر معلومات مصادقة غير قابلة للاستخدام مع كل استدعاء إجراء عن بعد. هذه هي المصادقة الافتراضية التي يستخدمها RPC.
AUTH *authunix_create(char *host, uid_t uid, gid_t gid,
                      int n, gid_t aup_gids[n]);
ينشئ ويعيد مقبض مصادقة RPC يحتوي على معلومات المصادقة. المعلمة host هي اسم الجهاز الذي تم إنشاء المعلومات عليه؛ uid هو معرف المستخدم؛ gid هو معرف المجموعة الحالي للمستخدم؛ تشير n وaup_gids إلى مصفوفة محسوبة للمجموعات التي ينتمي إليها المستخدم. من السهل انتحال شخصية مستخدم.
AUTH *authunix_create_default(void);
يستدعي authunix_create() بالمعلمات المناسبة.
int callrpc(char *host, unsigned long prognum,
            unsigned long versnum, unsigned long procnum,
            xdrproc_t inproc, const char *in,
            xdrproc_t outproc, char *out);
استدعاء الإجراء عن بعد المرتبط بـ prognum وversnum وprocnum على الجهاز host. المعلمة in هي عنوان الوسيط (الوسائط) للإجراء، وout هي العنوان الذي سيتم وضع النتائج فيه؛ يستخدم inproc لتشفير معاملات الإجراء، ويستخدم outproc لفك تشفير نتائج الإجراء؛ تعيد هذه الوظيفة صفرًا في حالة النجاح، أو قيمة enum clnt_stat محسوبة إلى عدد صحيح في حالة الفشل. الوظيفة clnt_perrno() مفيدة لترجمة حالات الفشل إلى رسائل.
تحذير: استدعاء الإجراءات عن بعد باستخدام هذه الوظيفة يستخدم UDP/IP كوسيلة نقل؛ راجع clntudp_create() للقيود. ليس لديك سيطرة على المهلات أو المصادقة باستخدام هذه الوظيفة.
enum clnt_stat clnt_broadcast(unsigned long prognum,

                     xdrproc_t outproc, char *out,
                     resultproc_t eachresult);
مثل callrpc()، باستثناء أن رسالة الاستدعاء يتم بثها إلى جميع شبكات البث المتصلة محليًا. في كل مرة تتلقى فيها استجابة، تستدعي هذه الوظيفة eachresult()، والتي يكون شكلها:

eachresult(char *out, struct sockaddr_in *addr);

حيث out هو نفس out الذي تم تمريره إلى clnt_broadcast()، باستثناء أنه يتم فك تشفير نتيجة الإجراء عن بعد هناك؛ يشير addr إلى عنوان الجهاز الذي أرسل النتائج. إذا أعادت eachresult() صفرًا، تنتظر clnt_broadcast() المزيد من الردود؛ وإلا فإنها تعود بحالة مناسبة.
تحذير: مقابس البث محدودة في الحجم بحد أقصى لنقل البيانات. بالنسبة للإيثرنت، تبلغ هذه القيمة 1500 بايت.

وحدة ماكرو تستدعي الإجراء عن بعد procnum المرتبط بمقبض العميل clnt، والذي يتم الحصول عليه باستخدام روتين إنشاء عميل RPC مثل clnt_create(). المعلمة in هي عنوان الوسيط (الوسائط) للإجراء، وout هو العنوان الذي سيتم وضع النتائج فيه؛ يستخدم inproc لتشفير معاملات الإجراء، ويستخدم outproc لفك تشفير نتائج الإجراء؛ tout هو الوقت المسموح به لعودة النتائج.
clnt_destroy(CLIENT *clnt);
وحدة ماكرو تدمر مقبض RPC الخاص بالعميل. عادةً ما يتضمن التدمير إلغاء تخصيص هياكل البيانات الخاصة، بما في ذلك clnt نفسه. استخدام clnt غير محدد بعد استدعاء clnt_destroy(). إذا قامت مكتبة RPC بفتح المقبس المرتبط، فستقوم بإغلاقه أيضًا. بخلاف ذلك، يظل المقبس مفتوحًا.
CLIENT *clnt_create(const char *host, unsigned long prog,
                    unsigned long vers, const char *proto);
روتين عام لإنشاء عميل. يحدد host اسم الجهاز البعيد الذي يقع عليه الخادم. يشير proto إلى نوع بروتوكول النقل المراد استخدامه. القيم المدعومة حاليًا لهذا الحقل هي "udp" و"tcp". يتم تعيين مهلات افتراضية، ولكن يمكن تعديلها باستخدام clnt_control().
تحذير: استخدام UDP له عيوبه. نظرًا لأن رسائل RPC المستندة إلى UDP يمكن أن تحتوي على 8 كيلوبايت فقط من البيانات المشفرة، فلا يمكن استخدام وسيلة النقل هذه للإجراءات التي تأخذ وسائط كبيرة أو تعيد نتائج ضخمة.
bool_t clnt_control(CLIENT *cl, int req, char *info);
وحدة ماكرو تستخدم لتغيير أو استرداد معلومات مختلفة حول كائن العميل. يشير req إلى نوع العملية، وinfo هو مؤشر إلى المعلومات. لكل من UDP وTCP، القيم المدعومة لـ req وأنواع وسائطها وما تفعلونه هي:

CLSET_TIMEOUT  struct timeval // تعيين المهلة الإجمالية
CLGET_TIMEOUT  struct timeval // الحصول على المهلة الإجمالية

CLGET_SERVER_ADDR  struct sockaddr_in
                // الحصول على عنوان الخادم

CLSET_RETRY_TIMEOUT  struct timeval // تعيين مهلة إعادة المحاولة

مهلة إعادة المحاولة هي الوقت الذي ينتظره "UDP RPC" حتى يرد الخادم قبل إعادة إرسال الطلب.
clnt_freeres(CLIENT * clnt, xdrproc_t outproc, char *out);

void clnt_pcreateerror(const char *s);
طباعة رسالة إلى الخطأ القياسي تشير إلى سبب عدم إمكانية إنشاء مقبض عميل RPC. يتم إضافة السلسلة s ونقطتين إلى الرسالة. تستخدم عندما تفشل استدعاءات clnt_create() أو clntraw_create() أو clnttcp_create() أو clntudp_create().
void clnt_perrno(enum clnt_stat stat);
طباعة رسالة إلى الخطأ القياسي تتوافق مع الحالة التي تشير إليها stat. تستخدم بعد callrpc().

طباعة رسالة إلى الخطأ القياسي تشير إلى سبب فشل استدعاء RPC؛ clnt هو المقبض المستخدم لإجراء الاستدعاء. يتم إضافة السلسلة s ونقطتين إلى الرسالة. تستخدم بعد clnt_call().
char *clnt_spcreateerror(const char *s);
مثل clnt_pcreateerror()، باستثناء أنها تُرجع سلسلة بدلاً من الطباعة إلى الخطأ القياسي.
أخطاء: ترجع مؤشرًا إلى بيانات ثابتة يتم الكتابة فوقها في كل استدعاء.
char *clnt_sperrno(enum clnt_stat stat);
تأخذ نفس وسائط clnt_perrno()، ولكن بدلاً من إرسال رسالة إلى الخطأ القياسي تشير إلى سبب فشل استدعاء RPC، تُرجع مؤشرًا إلى سلسلة تحتوي على الرسالة. تنتهي السلسلة بحرف سطر جديد.
clnt_sperrno() تستخدم بدلاً من clnt_perrno() إذا لم يكن لدى البرنامج خطأ قياسي (كما هو الحال في برنامج يعمل كخادم)، أو إذا كان المبرمج لا يريد إخراج الرسالة باستخدام printf(3)، أو إذا كان هناك تنسيق رسالة مختلف عن ذلك الذي تدعمه clnt_perrno(). ملاحظة: بخلاف clnt_sperror() و clnt_spcreateerror()، تُرجع clnt_sperrno() مؤشرًا إلى بيانات ثابتة، ولكن لن يتم الكتابة فوق النتيجة في كل استدعاء.
char *clnt_sperror(CLIENT *rpch, const char *s);
مثل clnt_perror()، باستثناء أنها (مثل clnt_sperrno()) تُرجع سلسلة بدلاً من الطباعة إلى الخطأ القياسي.
أخطاء: ترجع مؤشرًا إلى بيانات ثابتة يتم الكتابة فوقها في كل استدعاء.
CLIENT *clntraw_create(unsigned long prognum, unsigned long versnum);
تقوم هذه الدالة بإنشاء عميل RPC وهمي للبرنامج البعيد prognum، الإصدار versnum. النقل المستخدم لتمرير الرسائل إلى الخدمة هو في الواقع مخزن مؤقت داخل مساحة عنوان العملية، لذا يجب أن يعيش خادم RPC المقابل في نفس مساحة العنوان؛ انظر svcraw_create(). هذا يسمح بمحاكاة RPC واكتساب تكاليف RPC، مثل أوقات الذهاب والإياب، دون أي تدخل من النواة. تُرجع هذه الدالة NULL إذا فشلت.
CLIENT *clnttcp_create(struct sockaddr_in *addr,
                unsigned long prognum, unsigned long versnum,
                int *sockp, unsigned int sendsz, unsigned int recvsz);
تقوم هذه الدالة بإنشاء عميل RPC للبرنامج البعيد prognum، الإصدار versnum؛ يستخدم العميل TCP/IP كوسيلة نقل. يقع البرنامج البعيد في عنوان الإنترنت *addr. إذا كان addr->sin_port صفرًا، فسيتم تعيينه إلى المنفذ الفعلي الذي يستمع إليه البرنامج البعيد (يتم استشارة خدمة portmap البعيدة لهذه المعلومات). المعامل sockp هو مقبس؛ إذا كان RPC_ANYSOCK، فستقوم هذه الدالة بفتح مقبس جديد وتعيين sockp. نظرًا لأن RPC المستند إلى TCP يستخدم I/O المخزن مؤقتًا، يمكن للمستخدم تحديد حجم مخازن الإرسال والاستقبال باستخدام المعاملين sendsz و recvsz؛ القيم الصفرية تختار الافتراضيات المناسبة. تُرجع هذه الدالة NULL إذا فشلت.
CLIENT *clntudp_create(struct sockaddr_in *addr,

                struct timeval wait, int *sockp);
تقوم هذه الدالة بإنشاء عميل RPC للبرنامج البعيد prognum، الإصدار versnum؛ يستخدم العميل UDP/IP كوسيلة نقل. يقع البرنامج البعيد في عنوان الإنترنت addr. إذا كان addr->sin_port صفرًا، فسيتم تعيينه إلى المنفذ الفعلي الذي يستمع إليه البرنامج البعيد (يتم استشارة خدمة portmap البعيدة لهذه المعلومات). المعامل sockp هو مقبس؛ إذا كان RPC_ANYSOCK، فستقوم هذه الدالة بفتح مقبس جديد وتعيين sockp. تقوم وسيلة نقل UDP بإعادة إرسال رسالة الاستدعاء بفواصل زمنية قدرها wait حتى يتم استلام استجابة أو حتى ينتهي وقت الاستدعاء. يتم تحديد الوقت الإجمالي لانتهاء وقت الاستدعاء بواسطة clnt_call().

تُستخدم لوصف رسائل استدعاء RPC. هذا الروتين مفيد للمستخدمين الذين يرغبون في إنشاء رسائل بأسلوب RPC دون استخدام حزمة RPC.
bool_t xdr_opaque_auth(XDR *xdrs, struct opaque_auth *ap);
تُستخدم لوصف رسائل معلومات مصادقة RPC. هذا الروتين مفيد للمستخدمين الذين يرغبون في إنشاء رسائل بأسلوب RPC دون استخدام حزمة RPC.
bool_t xdr_pmap(XDR *xdrs, struct pmap *regs);
تُستخدم لوصف معلمات لمختلف إجراءات portmap، خارجيًا. هذا الروتين مفيد للمستخدمين الذين يرغبون في إنشاء هذه المعلمات دون استخدام واجهة pmap.
bool_t xdr_pmaplist(XDR *xdrs, struct pmaplist **rp);
تُستخدم لوصف قائمة تعيينات المنافذ، خارجيًا. هذا الروتين مفيد للمستخدمين الذين يرغبون في إنشاء هذه المعلمات دون استخدام واجهة pmap.
bool_t xdr_rejected_reply(XDR *xdrs, struct rejected_reply *rr);
تُستخدم لوصف رسائل رد RPC. هذا الروتين مفيد للمستخدمين الذين يرغبون في إنشاء رسائل بأسلوب RPC دون استخدام حزمة RPC.
bool_t xdr_replymsg(XDR *xdrs, struct rpc_msg *rmsg);
تُستخدم لوصف رسائل رد RPC. هذا الروتين مفيد للمستخدمين الذين يرغبون في إنشاء رسائل بأسلوب RPC دون استخدام حزمة RPC.
void xprt_register(SVCXPRT *xprt);
بعد إنشاء معالجات نقل خدمة RPC، يجب عليها تسجيل نفسها لدى حزمة خدمة RPC. هذا الروتين يقوم بتعديل المتغير العام svc_fds. عادةً لا يحتاج منفذو الخدمة إلى هذا الروتين.
void xprt_unregister(SVCXPRT *xprt);
قبل تدمير معالج نقل خدمة RPC، يجب عليه إلغاء تسجيل نفسه لدى حزمة خدمة RPC. هذا الروتين يقوم بتعديل المتغير العام svc_fds. عادةً لا يحتاج منفذو الخدمة إلى هذا الروتين.
