xdr(3) دليل وظائف المكتبة xdr(3)

الاسم

xdr - روتينات مكتبية لتمثيل البيانات الخارجية

المكتبة

مكتبة C القياسية (libc، -lc)

الملخص والوصف

تسمح هذه الروتينات لبرامج C بوصف هياكل بيانات عشوائية بطريقة مستقلة عن الآلة. يتم نقل البيانات الخاصة باستدعاءات الإجراءات البعيدة باستخدام هذه الروتينات.

النماذج الأولية أدناه معرّفة في <rpc/xdr.h> وتستخدم الأنواع التالية:

typedef int bool_t;
typedef typeof(bool_t (XDR *, void *, ...))  *xdrproc_t;

لتعريف النوع XDR، انظر <rpc/xdr.h>.

bool_t xdr_array(XDR *xdrs, char **arrp, unsigned int *sizep,
                 unsigned int maxsize, unsigned int elsize,
                 xdrproc_t elproc);
مرشح بدائي يترجم بين المصفوفات متغيرة الطول وتمثيلاتها الخارجية المقابلة. الوسيط arrp هو عنوان المؤشر إلى المصفوفة، بينما sizep هو عنوان عداد عناصر المصفوفة؛ لا يمكن أن يتجاوز هذا العداد maxsize. الوسيط elsize هو sizeof كل عنصر من عناصر المصفوفة، و elproc هو مرشح XDR يترجم بين شكل C لعناصر المصفوفة وتمثيلها الخارجي. تعيد هذه الروتينية واحدًا إذا نجحت، وصفرًا إذا فشلت.
bool_t xdr_bool(XDR *xdrs, bool_t *bp);
مرشح بدائي يترجم بين القيم المنطقية (أعداد صحيحة C) وتمثيلاتها الخارجية. عند ترميز البيانات، تنتج هذه المرشحة قيمًا إما واحدًا أو صفرًا. تعيد هذه الروتينية واحدًا إذا نجحت، وصفرًا إذا فشلت.
bool_t xdr_bytes(XDR *xdrs, char **sp, unsigned int *sizep,
                 unsigned int maxsize);
مرشح بدائي يترجم بين سلاسل البايتات المرقمة وتمثيلاتها الخارجية. الوسيط sp هو عنوان مؤشر السلسلة. طول السلسلة موجود في العنوان sizep؛ لا يمكن أن تكون السلاسل أطول من maxsize. تعيد هذه الروتينية واحدًا إذا نجحت، وصفرًا إذا فشلت.
bool_t xdr_char(XDR *xdrs, char *cp);
مرشح بدائي يترجم بين أحرف C وتمثيلاتها الخارجية. تعيد هذه الروتينية واحدًا إذا نجحت، وصفرًا إذا فشلت. ملاحظة: الأحرف المشفرة ليست مجمعة، وتشغل 4 بايتات لكل منها. بالنسبة لمصفوفات الأحرف، يجدر النظر في xdr_bytes()، xdr_opaque()، أو xdr_string().
void xdr_destroy(XDR *xdrs);
وحدة ماكرو تستدعي روتينية التدمير المرتبطة بتيار XDR، xdrs. يشمل التدمير عادةً تحرير هياكل البيانات الخاصة المرتبطة بالتيار. استخدام xdrs بعد استدعاء xdr_destroy() غير محدد.
bool_t xdr_double(XDR *xdrs, double *dp);
مرشح بدائي يترجم بين أرقام double C ذات الدقة المزدوجة وتمثيلاتها الخارجية. تعيد هذه الروتينية واحدًا إذا نجحت، وصفرًا إذا فشلت.
bool_t xdr_enum(XDR *xdrs, enum_t *ep);
مرشح بدائي يترجم بين enums C (أعداد صحيحة في الواقع) وتمثيلاتها الخارجية. تعيد هذه الروتينية واحدًا إذا نجحت، وصفرًا إذا فشلت.
bool_t xdr_float(XDR *xdrs, float *fp);
مرشح بدائي يترجم بين floats C وتمثيلاتها الخارجية. تعيد هذه الروتينية واحدًا إذا نجحت، وصفرًا إذا فشلت.
void xdr_free(xdrproc_t proc, char *objp);
روتينية تحرير عامة. الوسيط الأول هو روتينية XDR للكائن الذي يتم تحريره. الوسيط الثاني هو مؤشر إلى الكائن نفسه. ملاحظة: المؤشر الممرر إلى هذه الروتينية *لا* يتم تحريره، ولكن ما يشير إليه يتم تحريره (بشكل متكرر).
unsigned int xdr_getpos(XDR *xdrs);
وحدة ماكرو تستدعي روتينية الحصول على الموضع المرتبطة بتيار XDR، xdrs. تعيد الروتينية عددًا صحيحًا غير موقع، يشير إلى موضع تيار بايتات XDR. ميزة مرغوبة في تيارات XDR هي أن العمليات الحسابية البسيطة تعمل مع هذا الرقم، على الرغم من أن مثيلات تيار XDR قد لا تضمن ذلك.
long *xdr_inline(XDR *xdrs, int len);
وحدة ماكرو تستدعي روتينية الاستدعاء المضمنة المرتبطة بتيار XDR، xdrs. تعيد الروتينية مؤشرًا إلى قطعة متجاورة من مخزن المؤقت للتيار؛ len هو طول البايتات المطلوب للمخزن المؤقت. ملاحظة: يتم تحويل المؤشر إلى long *.
تحذير: قد تعيد xdr_inline() القيمة NULL (0) إذا لم تتمكن من تخصيص قطعة متجاورة من المخزن المؤقت. لذلك قد يختلف السلوك بين مثيلات التيار؛ إنها موجودة من أجل الكفاءة.
bool_t xdr_int(XDR *xdrs, int *ip);
مرشح بدائي يترجم بين أعداد C الصحيحة وتمثيلاتها الخارجية. تعيد هذه الروتينية واحدًا إذا نجحت، وصفرًا إذا فشلت.
bool_t xdr_long(XDR *xdrs, long *lp);
مرشح بدائي يترجم بين أعداد long C الصحيحة وتمثيلاتها الخارجية. تعيد هذه الروتينية واحدًا إذا نجحت، وصفرًا إذا فشلت.
void xdrmem_create(XDR *xdrs, char *addr, unsigned int size,
                   enum xdr_op op);
تقوم هذه الروتينية بتهيئة كائن تيار XDR الذي يشير إليه xdrs. تتم كتابة بيانات التيار، أو قراءتها، من جزء من الذاكرة في الموقع addr والذي لا يتجاوز طوله size بايت. تحدد op اتجاه تيار XDR (إما XDR_ENCODE، XDR_DECODE، أو XDR_FREE).
bool_t xdr_opaque(XDR *xdrs, char *cp, unsigned int cnt);
مرشح بدائي يترجم بين البيانات غير الشفافة ذات الحجم الثابت وتمثيلها الخارجي. الوسيط cp هو عنوان الكائن غير الشفاف، و cnt هو حجمه بالبايت. تعيد هذه الروتينية واحدًا إذا نجحت، وصفرًا إذا فشلت.
bool_t xdr_pointer(XDR *xdrs, char **objpp,
                   unsigned int objsize, xdrproc_t xdrobj);
مثل xdr_reference() باستثناء أنها تسلسل المؤشرات الفارغة، بينما xdr_reference() لا تفعل ذلك. وبالتالي، يمكن لـ xdr_pointer() تمثيل هياكل البيانات المتكررة، مثل الأشجار الثنائية أو القوائم المرتبطة.
void xdrrec_create(XDR *xdrs, unsigned int sendsize,
                   unsigned int recvsize, char *handle,
                   typeof(int (char *, char *, int)) *readit,
                   typeof(int (char *, char *, int)) *writeit);
تقوم هذه الروتينية بتهيئة كائن تيار XDR الذي يشير إليه xdrs. تتم كتابة بيانات التيار إلى مخزن مؤقت بحجم sendsize؛ تشير القيمة صفر إلى أن النظام يجب أن يستخدم قيمة افتراضية مناسبة. تتم قراءة بيانات التيار من مخزن مؤقت بحجم recvsize؛ يمكن أيضًا تعيينها إلى قيمة افتراضية مناسبة عن طريق تمرير قيمة صفر. عندما يكون مخزن الإخراج المؤقت للتيار ممتلئًا، يتم استدعاء writeit. وبالمثل، عندما يكون مخزن الإدخال المؤقت للتيار فارغًا، يتم استدعاء readit. سلوك هاتين الروتينيتين مشابه لاستدعاءات النظام read(2) و write(2)، باستثناء أن handle يتم تمريره إلى الروتينيات السابقة كوسيط أول. ملاحظة: يجب على المتصل تعيين حقل op لتيار XDR.
تحذير: لقراءة تيار XDR تم إنشاؤه بواسطة واجهة برمجة التطبيقات هذه، ستحتاج إلى استدعاء xdrrec_skiprecord() أولاً قبل استدعاء أي واجهات برمجة تطبيقات XDR أخرى. هذا يدرج بايتات إضافية في التيار لتوفير حدود السجل
