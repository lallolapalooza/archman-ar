الاسم

rpc - روتينات مكتبة لطلبات الإجراءات عن بعد

المكتبة

مكتبة C القياسية (libc، -lc)

الملخص والوصف

تسمح هذه الروتينات لبرامج C بإجراء استدعاءات إجراءات على أجهزة أخرى عبر الشبكة. أولاً، يستدعي العميل إجراءً لإرسال حزمة بيانات إلى الخادم. عند استلام الحزمة، يستدعي الخادم روتين توزيع لأداء الخدمة المطلوبة، ثم يرسل ردًا. أخيرًا، يعود استدعاء الإجراء إلى العميل.

للاستفادة من هذه الروتينات، قم بتضمين ملف الرأس <rpc/rpc.h>.

تستخدم النماذج الأولية أدناه الأنواع التالية:

typedef int bool_t;
typedef typeof(bool_t (XDR *, void *, ...))  *xdrproc_t;
typedef typeof(bool_t (caddr_t resp, struct sockaddr_in *raddr)
    *resultproc_t;

راجع ملفات الرأس لإعلانات أنواع AUTH، CLIENT، SVCXPRT، و XDR.

void auth_destroy(AUTH *auth);
وحدة ماكرو تدمر معلومات المصادقة المرتبطة بـ auth. عادةً ما يتضمن التدمير إلغاء تخصيص هياكل البيانات الخاصة. استخدام auth غير محدد بعد استدعاء auth_destroy().
AUTH *authnone_create(void);
ينشئ ويعيد مؤشرًا إلى مقبض مصادقة RPC يمرر معلومات مصادقة غير قابلة للاستخدام مع كل طلب إجراء عن بعد. هذه هي المصادقة الافتراضية التي تستخدمها RPC.
AUTH *authunix_create(char *host, uid_t uid, gid_t gid,
                      int n, gid_t aup_gids[n]);
ينشئ ويعيد مؤشرًا إلى مقبض مصادقة RPC يحتوي على معلومات المصادقة. المعلمة host هي اسم الجهاز الذي تم إنشاء المعلومات عليه؛ uid هو معرف المستخدم؛ gid هو معرف المجموعة الحالي للمستخدم؛ n و aup_gids تشير إلى مصفوفة معدودة للمجموعات التي ينتمي إليها المستخدم. من السهل انتحال هوية مستخدم.
AUTH *authunix_create_default(void);
يستدعي authunix_create() بالمعلمات المناسبة.
int callrpc(char *host, unsigned long prognum,
            unsigned long versnum, unsigned long procnum,
            xdrproc_t inproc, const char *in,
            xdrproc_t outproc, char *out);
استدعاء الإجراء البعيد المرتبط بـ prognum و versnum و procnum على الجهاز host. المعلمة in هي عنوان وسيط (وسائط) الإجراء، و out هو العنوان الذي سيتم وضع النتائج فيه؛ يستخدم inproc لتشفير معاملات الإجراء، ويستخدم outproc لفك تشفير نتائج الإجراء؛ تعيد هذه الروتين صفر إذا نجحت، أو قيمة enum clnt_stat مضروبة في عدد صحيح إذا فشلت. الروتين clnt_perrno() مفيد لترجمة حالات الفشل إلى رسائل.
تحذير: يؤدي استدعاء الإجراءات البعيدة باستخدام هذا الروتين إلى استخدام UDP/IP كوسيلة نقل؛ راجع clntudp_create() للقيود. لا تملك تحكمًا في المهلات أو المصادقة باستخدام هذا الروتين.
enum clnt_stat clnt_broadcast(unsigned long prognum,
                     unsigned long versnum, unsigned long procnum,
                     xdrproc_t inproc, char *in,
                     xdrproc_t outproc, char *out,
                     resultproc_t eachresult);
مثل callrpc()، باستثناء أن رسالة الاستدعاء يتم بثها إلى جميع شبكات البث المتصلة محليًا. في كل مرة تتلقى فيها استجابة، يستدعي هذا الروتين eachresult()، والتي يكون شكلها:

eachresult(char *out, struct sockaddr_in *addr);

حيث out هو نفس out الذي تم تمريره إلى clnt_broadcast()، باستثناء أنه يتم فك تشفير مخرجات الإجراء البعيد هناك؛ addr يشير إلى عنوان الجهاز الذي أرسل النتائج. إذا أعاد eachresult() صفرًا، فإن clnt_broadcast() ينتظر المزيد من الردود؛ وإلا فإنه يعود بالحالة المناسبة.
تحذير: مقابس البث محدودة في الحجم كحد أقصى لوحدة نقل البيانات في رابط البيانات. بالنسبة للإيثرنت، تبلغ هذه القيمة 1500 بايت.
enum clnt_stat clnt_call(CLIENT *clnt, unsigned long procnum,
                    xdrproc_t inproc, char *in,
                    xdrproc_t outproc, char *out,
                    struct timeval tout);
وحدة ماكرو تستدعي الإجراء البعيد procnum المرتبط بمقبض العميل clnt، والذي تم الحصول عليه باستخدام روتين إنشاء عميل RPC مثل clnt_create(). المعلمة in هي عنوان وسيط (وسائط) الإجراء، و out هو العنوان الذي سيتم وضع النتائج فيه؛ يستخدم inproc لتشفير معاملات الإجراء، ويستخدم outproc لفك تشفير نتائج الإجراء؛ tout هو الوقت المسموح به لعودة النتائج.
clnt_destroy(CLIENT *clnt);
وحدة ماكرو تدمر مقبض RPC الخاص بالعميل. عادةً ما يتضمن التدمير إلغاء تخصيص هياكل البيانات الخاصة، بما في ذلك clnt نفسه. استخدام clnt غير محدد بعد استدعاء clnt_destroy(). إذا فتحت مكتبة RPC المقبس المرتبط، فستقوم بإغلاقه أيضًا. بخلاف ذلك، يبقى المقبس مفتوحًا.
CLIENT *clnt_create(const char *host, unsigned long prog,
                    unsigned long vers, const char *proto);
روتين إنشاء عميل عام. host يحدد اسم الجهاز البعيد حيث يوجد الخادم. proto يحدد نوع بروتوكول النقل الذي يجب استخدامه. القيم المدعومة حاليًا لهذا الحقل هي “udp” و “tcp”. يتم تعيين مهلات افتراضية، ولكن يمكن تعديلها باستخدام clnt_control().
تحذير: استخدام UDP له عيوبه. نظرًا لأن رسائل RPC المستندة إلى UDP يمكن أن تحتوي فقط على ما يصل إلى 8 كيلوبايت من البيانات المشفرة، لا يمكن استخدام هذا الناقل للإجراءات التي تأخذ وسائط كبيرة أو تعيد نتائج ضخمة.
bool_t clnt_control(CLIENT *cl, int req, char *info);
وحدة ماكرو تستخدم لتغيير أو استرداد معلومات مختلفة حول كائن العميل. req تشير إلى نوع العملية، و info هو مؤشر إلى المعلومات. لكل من UDP و TCP، القيم المدعومة لـ req وأنواع وسائطها وما تفعله هي:

CLSET_TIMEOUT  struct timeval // تعيين المهلة الإجمالية
CLGET_TIMEOUT  struct timeval // الحصول على المهلة الإجمالية

ملاحظة: إذا قمت بتعيين المهلة باستخدام clnt_control()، فسيتم تجاهل المعلمة timeout التي تم تمريرها إلى clnt_call() في جميع الاستدعاءات المستقبلية.

CLGET_SERVER_ADDR  struct sockaddr_in
                // الحصول على عنوان الخادم

العمليات التالية صالحة لـ UDP فقط:

CLSET_RETRY_TIMEOUT  struct timeval // تعيين مهلة إعادة المحاولة
CLGET_RETRY_TIMEOUT  struct timeval // الحصول على مهلة إعادة المحاولة

مهلة إعادة المحاولة هي الوقت الذي ينتظره "UDP RPC" حتى يرد الخادم قبل إعادة إرسال الطلب.
clnt_freeres(CLIENT * clnt, xdrproc_t outproc, char *out);
وحدة ماكرو تقوم بتحرير أي بيانات تم تخصيصها بواسطة نظام RPC/XDR عند فك تشفير نتائج استدعاء RPC. المعلمة out هي عنوان النتائج، و outproc هو روتين XDR الذي يصف النتائج. يعيد هذا الروتين واحدًا إذا تم تحرير النتائج بنجاح، وصفرًا بخلاف ذلك.
void clnt_geterr(CLIENT *clnt, struct rpc_err

void svc_destroy(SVCXPRT *xprt);
ماكرو يدمر مؤشر نقل خدمة RPC، xprt. يتضمن التدمير عادةً إلغاء تخصيص هياكل البيانات الخاصة، بما في ذلك xprt نفسه. استخدام xprt غير محدد بعد استدعاء هذا الروتين.
fd_set svc_fdset;
متغير عام يعكس قناع بت واصف ملف القراءة الخاص بجانب خدمة RPC؛ وهو مناسب كمعلمة لاستدعاء النظام select(2). هذا ذو أهمية فقط إذا قام منفذ الخدمة بمعالجة الأحداث غير المتزامنة الخاصة به، بدلاً من استدعاء svc_run(). هذا المتغير للقراءة فقط (لا تمرر عنوانه إلى select(2)!)، ومع ذلك قد يتغير بعد استدعاءات svc_getreqset() أو أي روتينات إنشاء.
int svc_fds;
مشابه لـ svc_fdset، ولكنه محدود بـ 32 واصف ملف. تم إهمال هذه الواجهة بواسطة svc_fdset.
svc_freeargs(SVCXPRT *xprt, xdrproc_t inproc, char *in);
ماكرو يحرر أي بيانات تم تخصيصها بواسطة نظام RPC/XDR عند فك تشفير وسيطات إجراء خدمة باستخدام svc_getargs(). يعيد هذا الروتين 1 إذا تم تحرير النتائج بنجاح، وصفر بخلاف ذلك.
svc_getargs(SVCXPRT *xprt, xdrproc_t inproc, char *in);
ماكرو يقوم بفك تشفير وسيطات طلب RPC المرتبط بمؤشر نقل خدمة RPC، xprt. المعلمة in هي العنوان الذي سيتم وضع الوسيطات فيه؛ inproc هو روتين XDR المستخدم لفك تشفير الوسيطات. يعيد هذا الروتين واحدًا إذا نجح فك التشفير، وصفر بخلاف ذلك.
struct sockaddr_in *svc_getcaller(SVCXPRT *xprt);
الطريقة المعتمدة للحصول على عنوان الشبكة للمتصل بإجراء مرتبط بمؤشر نقل خدمة RPC، xprt.
void svc_getreqset(fd_set *rdfds);
هذا الروتين ذو أهمية فقط إذا لم يقم منفذ الخدمة باستدعاء svc_run()، ولكنه بدلاً من ذلك ينفذ معالجة أحداث غير متزامنة مخصصة. يتم استدعاؤه عندما تحدد استدعاءات النظام select(2) أن طلب RPC قد وصل على بعض مقابس RPC؛ rdfds هو قناع بت واصف ملف القراءة الناتج. يعيد الروتين عندما تتم خدمة جميع المقابس المرتبطة بقيمة rdfds.
void svc_getreq(int rdfds);
مشابه لـ svc_getreqset()، ولكنه محدود بـ 32 واصف ملف. تم إهمال هذه الواجهة بواسطة svc_getreqset().
bool_t svc_register(SVCXPRT *xprt, unsigned long prognum,
                    unsigned long versnum,
                    typeof(void (struct svc_req *, SVCXPRT *)) *dispatch,
                    unsigned long protocol);
يربط prognum و versnum بإجراء التوزيع الخاص بالخدمة، dispatch. إذا كان protocol صفرًا، فلن يتم تسجيل الخدمة مع خدمة portmap. إذا كان protocol غير صفري، فسيتم إنشاء تعيين للمجموعة الثلاثية [prognum,versnum,protocol] إلى xprt->xp_port مع خدمة portmap المحلية (بشكل عام protocol هو صفر، IPPROTO_UDP أو IPPROTO_TCP). الإجراء dispatch له الشكل التالي:

dispatch(struct svc_req *request, SVCXPRT *xprt);

يعيد الروتين svc_register() واحدًا إذا نجح، وصفر بخلاف ذلك.
void svc_run(void);
هذا الروتين لا يعود أبدًا. ينتظر وصول طلبات RPC، ويستدعي إجراء الخدمة المناسب باستخدام svc_getreq() عند وصول أحدها. عادة ما ينتظر هذا الروتين استدعاء النظام select(2) للعودة.
bool_t svc_sendreply(SVCXPRT *xprt, xdrproc_t outproc, char *out);
يتم استدعاؤه بواسطة إجراء التوزيع الخاص بخدمة RPC لإرسال نتائج استدعاء الإجراء عن بعد. المعلمة xprt هي مؤشر النقل المرتبط بالطلب؛ outproc هو روتين XDR المستخدم لتشفير النتائج؛ و out هو عنوان النتائج. يعيد هذا الروتين واحدًا إذا نجح، وصفر بخلاف ذلك.
void svc_unregister(unsigned long prognum, unsigned long versnum);
إزالة جميع تعيينات الزوج [prognum,versnum] إلى إجراءات التوزيع، والزوج الثلاثي [prognum,versnum,*] إلى رقم المنفذ.
void svcerr_auth(SVCXPRT *xprt, enum auth_stat why);
يتم استدعاؤه بواسطة إجراء توزيع الخدمة الذي يرفض أداء استدعاء إجراء عن بعد بسبب خطأ في المصادقة.
void svcerr_decode(SVCXPRT *xprt);
يتم استدعاؤه بواسطة إجراء توزيع الخدمة الذي لا يمكنه فك تشفير وسيطاته بنجاح. انظر أيضًا svc_getargs().
void svcerr_noproc(SVCXPRT *xprt);
يتم استدعاؤه بواسطة إجراء توزيع الخدمة الذي لا ينفذ رقم الإجراء الذي يطلبه المتصل.
void svcerr_noprog(SVCXPRT *xprt);
يتم استدعاؤه عندما لا يتم تسجيل البرنامج المطلوب مع حزمة RPC. عادة لا يحتاج منفذو الخدمة إلى هذا الروتين.
void svcerr_progvers(SVCXPRT *xprt, unsigned long low_vers,
                     unsigned long high_vers);
يتم استدعاؤه عندما لا يتم تسجيل الإصدار المطلوب من برنامج مع حزمة RPC. عادة لا يحتاج منفذو الخدمة إلى هذا الروتين.
void svcerr_systemerr(SVCXPRT *xprt);
يتم استدعاؤه بواسطة إجراء توزيع الخدمة عندما يكتشف خطأ نظام لا تغطيه أي بروتوكول معين. على سبيل المثال، إذا لم يعد بإمكان الخدمة تخصيص مساحة تخزين، فقد تستدعي هذا الروتين.
void svcerr_weakauth(SVCXPRT *xprt);
يتم استدعاؤه بواسطة إجراء توزيع الخدمة الذي يرفض أداء استدعاء إجراء عن بعد بسبب عدم كفاية معلمات المصادقة. يستدعي الروتين svcerr_auth(xprt, AUTH_TOOWEAK).
SVCXPRT *svcfd_create(int fd, unsigned int sendsize,

إنشاء خدمة فوق أي واصف ملف مفتوح. عادةً، يكون واصف الملف هذا عبارة عن مقبس متصل لبروتوكول تدفق مثل TCP. تشير sendsize و recvsize إلى أحجام مخازن الإرسال والاستقبال. إذا كانت صفرًا، يتم اختيار افتراضي معقول.

ينشئ هذا الروتين نقل خدمة RPC وهمي، والذي يعيد مؤشرًا إليه. النقل هو في الواقع مخزن مؤقت داخل مساحة عنوان العملية، لذلك يجب أن يكون عميل RPC المقابل موجودًا في نفس مساحة العنوان؛ انظر clntraw_create(). يسمح هذا الروتين بمحاكاة RPC واكتساب تكاليف RPC (مثل أوقات الذهاب والإياب)، دون أي تدخل من النواة. يعيد هذا الروتين NULL إذا فشل.
SVCXPRT *svctcp_create(int sock, unsigned int send_buf_size,

ينشئ هذا الروتين نقل خدمة RPC قائم على TCP/IP، والذي يعيد مؤشرًا إليه. يرتبط النقل بالمقبس sock، والذي يمكن أن يكون RPC_ANYSOCK، وفي هذه الحالة يتم إنشاء مقبس جديد. إذا لم يكن المقبس مرتبطًا بمنفذ TCP محلي، يقوم هذا الروتين بربطه بمنفذ عشوائي. عند الانتهاء، يكون xprt->xp_sock هو واصف مقبس النقل، و xprt->xp_port هو رقم منفذ النقل. يعيد هذا الروتين NULL إذا فشل. نظرًا لأن RPC القائم على TCP يستخدم إدخال/إخراج مخزن مؤقت، يمكن للمستخدمين تحديد حجم المخازن المؤقتة؛ القيم الصفرية تختار افتراضيات مناسبة.
