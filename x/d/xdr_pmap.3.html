الاسم

rpc - روتينات مكتبية لطلبات الإجراءات عن بعد

المكتبة

مكتبة C القياسية (libc، -lc)

الموجز والوصف

تسمح هذه الروتينات لبرامج C بإجراء استدعاءات إجراءات على أجهزة أخرى عبر الشبكة. أولاً، يقوم العميل باستدعاء إجراء لإرسال حزمة بيانات إلى الخادم. عند استلام الحزمة، يقوم الخادم باستدعاء إجراء موزع لتنفيذ الخدمة المطلوبة، ثم يرسل رداً. أخيرًا، تعود استدعاء العمليات إلى العميل.

للاستفادة من هذه الروتينات، قم بتضمين ملف الرأس <rpc/rpc.h>.

تستخدم النماذج الأولية أدناه الأنواع التالية:

typedef int bool_t;
typedef typeof(bool_t (XDR *, void *, ...))  *xdrproc_t;
typedef typeof(bool_t (caddr_t resp, struct sockaddr_in *raddr)
    *resultproc_t;

راجع ملفات الرأس لتصريحات أنواع AUTH و CLIENT و SVCXPRT و XDR.

void auth_destroy(AUTH *auth);
ماكرو يدمر معلومات المصادقة المرتبطة بـ auth. عادة ما يتضمن التدمير إلغاء تخصيص هياكل البيانات الخاصة. استخدام auth غير محدد بعد استدعاء auth_destroy().
AUTH *authnone_create(void);
إنشاء وإرجاع مقبض مصادقة RPC يمرر معلومات مصادقة غير قابلة للاستخدام مع كل طلب إجراء عن بعد. هذه هي المصادقة الافتراضية التي يستخدمها RPC.
AUTH *authunix_create(char *host, uid_t uid, gid_t gid,
                      int n, gid_t aup_gids[n]);
إنشاء وإرجاع مقبض مصادقة RPC يحتوي على معلومات المصادقة. المعلمة host هي اسم الجهاز الذي تم فيه إنشاء المعلومات؛ uid هو معرف المستخدم؛ gid هو معرف المجموعة الحالي للمستخدم؛ تشير n و aup_gids إلى مصفوفة محسوبة للمجموعات التي ينتمي إليها المستخدم. من السهل انتحال شخصية مستخدم.
AUTH *authunix_create_default(void);
يستدعي authunix_create() مع المعلمات المناسبة.
int callrpc(char *host, unsigned long prognum,
            unsigned long versnum, unsigned long procnum,
            xdrproc_t inproc, const char *in,
            xdrproc_t outproc, char *out);
استدعاء الإجراء البعيد المرتبط بـ prognum و versnum و procnum على الجهاز host. المعلمة in هي عنوان وسيط (وسائط) الإجراء، و out هي العنوان الذي سيتم فيه وضع النتائج؛ inproc يستخدم لتشفير معلمات الإجراء، و outproc يستخدم لفك تشفير نتائج الإجراء. ترجع هذه الروتينات صفرًا إذا نجحت، أو قيمة enum clnt_stat محولة إلى عدد صحيح إذا فشلت. الروتين clnt_perrno() مفيد لترجمة حالات الفشل إلى رسائل.
تحذير: استدعاء الإجراءات البعيدة باستخدام هذا الروتين يستخدم UDP/IP كوسيلة نقل؛ راجع clntudp_create() للقيود. لا يمكنك التحكم في المهلات أو المصادقة باستخدام هذا الروتين.
enum clnt_stat clnt_broadcast(unsigned long prognum,
                     unsigned long versnum, unsigned long procnum,
                     xdrproc_t inproc, char *in,
                     xdrproc_t outproc, char *out,
                     resultproc_t eachresult);
مثل callrpc()، باستثناء أن رسالة الاستدعاء يتم بثها إلى جميع شبكات البث المتصلة محليًا. في كل مرة تتلقى فيها ردًا، يستدعي هذا الروتين eachresult()، والذي يكون شكله:

eachresult(char *out, struct sockaddr_in *addr);

حيث out هو نفس out الذي تم تمريره إلى clnt_broadcast()، باستثناء أنه يتم فك تشفير ناتج الإجراء البعيد هناك؛ addr يشير إلى عنوان الجهاز الذي أرسل النتائج. إذا أعادت eachresult() صفرًا، فستنتظر clnt_broadcast() المزيد من الردود؛ وإلا فإنها تعود مع الحالة المناسبة.
تحذير: مقابس البث محدودة الحجم بحد أقصى لوحدة نقل البيانات للرابط. بالنسبة للإيثرنت، هذه القيمة هي 1500 بايت.
enum clnt_stat clnt_call(CLIENT *clnt, unsigned long procnum,
                    xdrproc_t inproc, char *in,
                    xdrproc_t outproc, char *out,
                    struct timeval tout);
ماكرو يستدعي الإجراء البعيد procnum المرتبط بمقبض العميل clnt، والذي تم الحصول عليه باستخدام روتين إنشاء عميل RPC مثل clnt_create(). المعلمة in هي عنوان وسيط (وسائط) الإجراء، و out هي العنوان الذي سيتم فيه وضع النتائج؛ inproc يستخدم لتشفير معلمات الإجراء، و outproc يستخدم لفك تشفير نتائج الإجراء؛ tout هو الوقت المسموح به لعودة النتائج.
clnt_destroy(CLIENT *clnt);
ماكرو يدمر مقبض RPC الخاص بالعميل. عادة ما يتضمن التدمير إلغاء تخصيص هياكل البيانات الخاصة، بما في ذلك clnt نفسه. استخدام clnt غير محدد بعد استدعاء clnt_destroy(). إذا فتحت مكتبة RPC المقبس المرتبط، فستقوم بإغلاقه أيضًا. بخلاف ذلك، يبقى المقبس مفتوحًا.
CLIENT *clnt_create(const char *host, unsigned long prog,
                    unsigned long vers, const char *proto);
روتين إنشاء عميل عام. host يحدد اسم المضيف البعيد حيث يوجد الخادم. proto يشير إلى نوع بروتوكول النقل الذي يجب استخدامه. القيم المدعومة حاليًا لهذا الحقل هي “udp” و “tcp”. يتم تعيين مهلات افتراضية، ولكن يمكن تعديلها باستخدام clnt_control().
تحذير: استخدام UDP له عيوبه. نظرًا لأن رسائل RPC المستندة إلى UDP يمكن أن تحتوي فقط على ما يصل إلى 8 كيلوبايت من البيانات المشفرة، لا يمكن استخدام هذا النقل للإجراءات التي تأخذ وسائط كبيرة أو تعيد نتائج ضخمة.
bool_t clnt_control(CLIENT *cl, int req, char *info);
ماكرو يستخدم لتغيير أو استرداد معلومات مختلفة حول كائن عميل. req يشير إلى نوع العملية، و info مؤشر إلى المعلومات. لكل من UDP و TCP، القيم المدعومة لـ req وأنواع وسائطها وما تفعله هي:

CLSET_TIMEOUT  struct timeval // تعيين المهلة الإجمالية
CLGET_TIMEOUT  struct timeval // الحصول على المهلة الإجمالية

ملاحظة: إذا قمت بتعيين المهلة باستخدام clnt_control()، فسيتم تجاهل المعلمة timeout التي تم تمريرها إلى clnt_call() في جميع الاستدعاءات المستقبلية.

CLGET_SERVER_ADDR  struct sockaddr_in
                // الحصول على عنوان الخادم

العمليات التالية صالحة لـ UDP فقط:

CLSET_RETRY_TIMEOUT  struct timeval // تعيين مهلة إعادة المحاولة
CLGET_RETRY_TIMEOUT  struct timeval // الحصول على مهلة إعادة المحاولة

مهلة إعادة المحاولة هي الوقت الذي ينتظره "UDP RPC" حتى يرد الخادم قبل إعادة إرسال الطلب.
clnt_freeres(CLIENT * clnt, xdrproc_t outproc, char *out);
ماكرو يقوم بتحرير أي بيانات تم تخصيصها بواسطة نظام RPC/XDR عند فك تشفير نتائج استدعاء RPC. المعلمة out هي عنوان النتائج، و outproc هو روتين XDR الذي يصف النتائج. ترجع هذه الروتينات واحدًا إذا تم تحرير النتائج بنجاح، وصفرًا بخلاف ذلك.
void clnt_geterr(CLIENT *clnt, struct rpc_err *
t; انظر svcudp_create() للقيود. 
struct rpc_createerr rpc_createerr;
متغير عام يتم تعيين قيمته بواسطة أي إجراء لإنشاء عميل RPC لا ينجح. استخدم الإجراء clnt_pcreateerror() لطباعة السبب.
void svc_destroy(SVCXPRT *xprt);
ماكرو يدمر مقبض نقل خدمة RPC، xprt. يشمل التدمير عادةً تخصيص هياكل البيانات الخاصة، بما في ذلك xprt نفسه. استخدام xprt غير محدد بعد استدعاء هذا الإجراء.
fd_set svc_fdset;
متغير عام يعكس قناع بت واصفات ملف القراءة لجانب خدمة RPC؛ وهو مناسب كمعلمة لاستدعاء النظام select(2). هذا يثير الاهتمام فقط إذا كان منفذ الخدمة يقوم بمعالجة الأحداث غير المتزامنة الخاصة به، بدلاً من استدعاء svc_run(). هذا المتغير للقراءة فقط (لا تمرر عنوانه إلى select(2)!)، ولكنه قد يتغير بعد استدعاءات svc_getreqset() أو أي إجراءات إنشاء.
int svc_fds;
يشبه svc_fdset، ولكنه محدود بـ 32 واصف ملف. تم إهمال هذه الواجهة بواسطة svc_fdset.
svc_freeargs(SVCXPRT *xprt, xdrproc_t inproc, char *in);
ماكرو يحرر أي بيانات تم تخصيصها بواسطة نظام RPC/XDR عند فك ترميز وسيطات إجراء الخدمة باستخدام svc_getargs(). يعيد هذا الإجراء 1 إذا تم تحرير النتائج بنجاح، وصفر بخلاف ذلك.
svc_getargs(SVCXPRT *xprt, xdrproc_t inproc, char *in);
ماكرو يفك ترميز وسيطات طلب RPC المرتبط بمقبض نقل خدمة RPC، xprt. المعلمة in هي العنوان الذي سيتم فيه وضع الوسيطات؛ inproc هو إجراء XDR المستخدم لفك ترميز الوسيطات. يعيد هذا الإجراء واحدًا إذا نجح فك الترميز، وصفرًا بخلاف ذلك.
struct sockaddr_in *svc_getcaller(SVCXPRT *xprt);
الطريقة المعتمدة للحصول على عنوان الشبكة للمتصل بإجراء مرتبط بمقبض نقل خدمة RPC، xprt.
void svc_getreqset(fd_set *rdfds);
هذا الإجراء يثير الاهتمام فقط إذا كان منفذ الخدمة لا يستدعي svc_run()، ولكنه بدلاً من ذلك ينفذ معالجة أحداث مخصصة غير متزامنة. يتم استدعاؤه عندما يحدد استدعاء النظام select(2) أن طلب RPC قد وصل على بعض مقابس RPC؛ rdfds هو قناع بت واصفات ملف القراءة الناتج. يعيد الإجراء عندما يتم التعامل مع جميع المقابس المرتبطة بقيمة rdfds.
void svc_getreq(int rdfds);
يشبه svc_getreqset()، ولكنه محدود بـ 32 واصف ملف. تم إهمال هذه الواجهة بواسطة svc_getreqset().
bool_t svc_register(SVCXPRT *xprt, unsigned long prognum,
                    unsigned long versnum,
                    typeof(void (struct svc_req *, SVCXPRT *)) *dispatch,
                    unsigned long protocol);
يربط prognum و versnum بإجراء توزيع الخدمة، dispatch. إذا كان protocol صفرًا، فلن يتم تسجيل الخدمة مع خدمة portmap. إذا كان protocol غير صفري، فسيتم إنشاء تعيين للثلاثي [prognum,versnum,protocol] إلى xprt->xp_port مع خدمة portmap المحلية (بشكل عام protocol هو صفر، IPPROTO_UDP أو IPPROTO_TCP). الإجراء dispatch له الشكل التالي:

dispatch(struct svc_req *request, SVCXPRT *xprt);

يعيد الإجراء svc_register() واحدًا إذا نجح، وصفرًا بخلاف ذلك.
void svc_run(void);
هذا الإجراء لا يعود أبداً. ينتظر وصول طلبات RPC، ويستدعي إجراء الخدمة المناسب باستخدام svc_getreq() عند وصول أحدها. عادة ما ينتظر هذا الإجراء استدعاء النظام select(2) للعودة.
bool_t svc_sendreply(SVCXPRT *xprt, xdrproc_t outproc, char *out);
يتم استدعاؤه بواسطة إجراء توزيع خدمة RPC لإرسال نتائج استدعاء إجراء عن بعد. المعلمة xprt هي مقبض النقل المرتبط بالطلب؛ outproc هو إجراء XDR المستخدم لترميز النتائج؛ و out هو عنوان النتائج. يعيد هذا الإجراء واحدًا إذا نجح، وصفرًا بخلاف ذلك.
void svc_unregister(unsigned long prognum, unsigned long versnum);
إزالة جميع تعيينات الثنائي [prognum,versnum] إلى إجراءات التوزيع، والثلاثي [prognum,versnum,*] إلى رقم المنفذ.
void svcerr_auth(SVCXPRT *xprt, enum auth_stat why);
يتم استدعاؤه بواسطة إجراء توزيع خدمة يرفض أداء استدعاء إجراء عن بعد بسبب خطأ في المصادقة.
void svcerr_decode(SVCXPRT *xprt);
يتم استدعاؤه بواسطة إجراء توزيع خدمة لا يمكنه فك ترميز وسيطاته بنجاح. انظر أيضًا svc_getargs().
void svcerr_noproc(SVCXPRT *xprt);
يتم استدعاؤه بواسطة إجراء توزيع خدمة لا ينفذ رقم الإجراء الذي يطلبه المتصل.
void svcerr_noprog(SVCXPRT *xprt);
يتم استدعاؤه عندما لا يكون البرنامج المطلوب مسجلًا مع حزمة RPC. عادة لا يحتاج منفذو الخدمة إلى هذا الإجراء.
void svcerr_progvers(SVCXPRT *xprt, unsigned long low_vers,
                     unsigned long high_vers);
يتم استدعاؤه عندما لا يكون الإصدار المطلوب من برنامج مسجلًا مع حزمة RPC. عادة لا يحتاج منفذو الخدمة إلى هذا الإجراء.
void svcerr_systemerr(SVCXPRT *xprt);
يتم استدعاؤه بواسطة إجراء توزيع خدمة عند اكتشاف خطأ في النظام لا يغطيه أي بروتوكول معين. على سبيل المثال، إذا لم يعد بإمكان الخدمة تخصيص مساحة تخزين، فقد تستدعي هذا الإجراء.
void svcerr_weakauth(SVCXPRT *xprt);
يتم استدعاؤه بواسطة إجراء توزيع خدمة يرفض أداء استدعاء إجراء عن بعد بسبب عدم كفاية معلمات المصادقة. يستدعي الإجراء svcerr_auth(xprt, AUTH_TOOWEAK).
SVCXPRT *svcfd_create(int fd, unsigned int sendsize,
                      unsigned int recvsize);
إنشاء خدمة فوق أي واصف ملف مفتوح. عادة، يكون واصف الملف هذا مقبسًا متصلًا لبروتوكول تدفق مثل TCP. تشير sendsize و recvsize إلى أحجام مخازن الإرسال والاستقبال. إذا كانت صفرًا، يتم اختيار افتراضي معقول.
SVCXPRT *svcraw_create(void);
هذا الإجراء ينشئ مقبض نقل خدمة RPC وهمي، والذي يعيد مؤشرًا له. النقل هو في الواقع مخزن مؤقت داخل مساحة عنوان العملية، لذلك يجب أن يعيش عميل RPC المقابل في نفس مساحة العنوان؛ انظر clntraw_create(). يسمح هذا الإجراء بمحاكاة RPC واكتساب نفقات RPC (مثل أوقات الذهاب والعودة)، دون أي تدخل من النواة. يعيد هذا الإجراء NULL إذا فشل.
SVCXPRT *svctcp_create(int sock, unsigned int send_buf_size,
                       unsigned int recv_buf_size);
هذا الإجراء ينشئ مقبض نقل خدمة RPC يستند إلى TCP/IP، والذي يعيد مؤشرًا له. يرتبط النقل بالمقبس sock، والذي يمكن أن يكون RPC_ANYSOCK، وفي هذه الحالة يتم إنشاء مقبس جديد. إذا لم يكن المقبس مرتبطًا بمنفذ TCP محلي، فسيقوم هذا الإجراء بربطه بمنفذ عشوائي. عند الانتهاء، xprt->xp_sock هو واصف مقبس النقل، و xprt->xp_port هو رقم منفذ النقل. يعيد هذا الإجراء NULL إذا فشل. نظرًا لأن RPC المستند إلى TCP يستخدم I/O المخزن مؤقتًا، يمكن للمستخدمين تحديد حجم المخازن المؤقتة؛ تختار القيم الصفرية افتراضيات مناسبة.
SVCXPRT *svcudp_bufcreate(int sock, unsigned int sendsize,
                          unsigned int recosize);
هذا الإجراء ينشئ مقبض نقل خدمة RPC يستند إلى UDP/IP، والذي يعيد مؤشرًا له. يرتبط النقل بالمقبس sock، والذي يمكن أن يكون RPC_ANYSOCK، وفي هذه الحالة يتم إنشاء مقبس جديد. إذا لم يكن المقبس مرتبطًا بمنفذ UDP محلي، فسيقوم هذا الإجراء بربطه بمنفذ عشوائي. عند الانتهاء، xprt->xp_sock هو واصف مقبس النقل، و xprt->xp_port هو رقم منفذ النقل. يعيد هذا الإجراء NULL إذا فشل.
يسمح هذا للمستخدم بتحديد الحد الأقصى لحجم الحزمة لإرسال واستقبال رسائل RPC المستندة إلى UDP.
SVCXPRT *svcudp_create(int sock);

bool_t xdr_callmsg(XDR *xdrs
