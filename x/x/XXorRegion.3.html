XXorRegion(3)

الحزمة: libx11

التراخيص: MIT AND X11

XIntersectRegion(3) دوال XLIB XIntersectRegion(3)

الاسم

XIntersectRegion, XUnionRegion, XUnionRectWithRegion, XSubtractRegion, XXorRegion, XOffsetRegion, XShrinkRegion - حسابات المنطقة

الصيغة

int XIntersectRegion(Region sra, Region srb, Region dr_return);

int XUnionRegion(Region sra, Region srb, Region dr_return);

int XUnionRectWithRegion(XRectangle *rectangle, Region src_region, Region dest_region_return);

int XSubtractRegion(Region sra, Region srb, Region dr_return);

int XXorRegion(Region sra, Region srb, Region dr_return);

int XOffsetRegion(Region r, int dx, int dy);

int XShrinkRegion(Region r, int dx, int dy);

الوسائط

dest_region_return
يعيد المنطقة الوجهة.
dr_return
يعيد نتيجة الحساب.
dx

dy
حدد إحداثيات x و y، التي تحدد مقدار الحركة أو التصغير للمنطقة المحددة.
r
يحدد المنطقة.
rectangle
يحدد المستطيل.
sra

srb
حدد المنطقتين اللتين تريد إجراء الحساب عليهما.
src_region
يحدد المنطقة المصدر التي سيتم استخدامها.

الوصف

تقوم الدالة XIntersectRegion بحساب تقاطع منطقتين.

تقوم الدالة XUnionRegion بحساب اتحاد منطقتين.

تقوم الدالة XUnionRectWithRegion بتحديث المنطقة الوجهة من اتحاد المستطيل المحدد والمنطقة المصدر المحددة.

تقوم الدالة XSubtractRegion بطرح srb من sra وتخزين النتائج في dr_return.

تقوم الدالة XXorRegion بحساب الفرق بين اتحاد منطقتين وتقاطعهما.

تقوم الدالة XOffsetRegion بنقل المنطقة المحددة بمقدار محدد.

تقوم الدالة XShrinkRegion بتقليص المنطقة المحددة بمقدار محدد. القيم الموجبة تقلص حجم المنطقة، والقيم السالبة توسع المنطقة.
