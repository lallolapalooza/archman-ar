XScreenSaverQueryExtension(3)

الحزمة: libxss

التراخيص: X11

XScreenSaver(3) إصدار X 11 XScreenSaver(3)

الاسم

XScreenSaver - مكتبة عميل لملحق شاشة التوقف X11

المُلَخَّص

#include <X11/extensions/scrnsaver.h>

typedef struct {
    Window window;                /* نافذة شاشة التوقف */
    int state;                    /* ScreenSaver{Off,On,Disabled} */
    int kind;                     /* ScreenSaver{Blanked,Internal,External} */
    unsigned long til_or_since;   /* بالمللي ثانية */
    unsigned long idle;           /* بالمللي ثانية */
    unsigned long eventMask;      /* الأحداث */
} XScreenSaverInfo;

typedef struct {
    int type;               /* نوع الحدث */
    unsigned long serial;   /* رقم آخر طلب تمت معالجته بواسطة الخادم */
    Bool send_event;        /* صحيح إذا جاء هذا من طلب SendEvent */
    Display *display;       /* العرض الذي تمت قراءة الحدث منه */
    Window window;          /* نافذة شاشة التوقف */
    Window root;            /* النافذة الجذرية لشاشة الحدث */
    int state;              /* ScreenSaver{Off,On,Cycle} */
    int kind;               /* ScreenSaver{Blanked,Internal,External} */
    Bool forced;            /* حدود المنطقة الجديدة */
    Time time;              /* الطابع الزمني للحدث */
} XScreenSaverNotifyEvent;
Bool XScreenSaverQueryExtension(Display *dpy,
    int *event_base_return, int *error_base_return);
Status XScreenSaverQueryVersion(Display *dpy,
    int *major_version_return, int *minor_version_return);
XScreenSaverInfo *XScreenSaverAllocInfo(void);
Status XScreenSaverQueryInfo(Display *dpy, Drawable drawable,
    XScreenSaverInfo *saver_info);
void XScreenSaverSelectInput(Display *dpy, Drawable drawable,
    unsigned long mask);
void XScreenSaverSetAttributes(Display *dpy, Drawable drawable,
    int x, int y, unsigned int width, unsigned int height,
    unsigned int border_width, int depth, unsigned int class,
    Visual *visual, unsigned long valuemask,
    XSetWindowAttributes *attributes);
void XScreenSaverUnsetAttributes(Display *dpy, Drawable drawable);
void XScreenSaverRegister(Display *dpy, int screen, XID xid, Atom type);
Status XScreenSaverUnregister(Display *dpy, int screen);
Status XScreenSaverGetRegistered(Display *dpy, int screen,
    XID *xid, Atom *type);
void XScreenSaverSuspend(Display *dpy, Bool suspend);

الوصف

يوفر نظام نوافذ X دعمًا لتغيير الصورة على شاشة العرض بعد فترة خمول قابلة للضبط من قبل المستخدم لتجنب حرق فسفور أنبوب الأشعة الكاثودية. ومع ذلك، لا توجد واجهات متاحة للمستخدم للتحكم في الصورة التي يتم رسمها. يسمح هذا الامتداد لعميل "شاشة توقف" خارجي باكتشاف وقت عرض الصورة البديلة وتوفير الرسومات.

عادةً ما توفر تطبيقات خادم X الحالية شكلاً واحدًا على الأقل من صور "شاشة التوقف". تاريخيًا، كان هذا نسخة من شعار X مرسومة على نمط خلفية الجذر. ومع ذلك، طلب العديد من المستخدمين آلية للسماح لهم بكتابة برامج شاشة توقف توفر إمكانيات مشابهة لتلك التي توفرها أنظمة النوافذ الأخرى. على وجه الخصوص، غالبًا ما يرغب هؤلاء المستخدمون في أن يكونوا قادرين على عرض الشعارات التجارية، وتعليمات حول كيفية إعادة تنشيط الشاشة، وأدوات القفل التلقائي للشاشة. يوفر هذا الامتداد وسيلة لكتابة هذه العملاء.

افتراضات

يقوم هذا الامتداد بتصدير مفهوم نافذة شاشة توقف خاصة يتم رسمها فوق جميع النوافذ الأخرى على الشاشة. تحتوي هذه النافذة على السمة override-redirect مضبوطة بحيث لا تخضع للتلاعب من قبل مدير النوافذ. علاوة على ذلك، لا يتم إرجاع معرف X للنافذة مطلقًا بواسطة طلبات QueryTree على النافذة الجذرية، لذلك فهي غير مرئية عادةً للعملاء الآخرين.

XScreenSaverQueryExtension يُرجع True إذا كان امتداد XScreenSaver متاحًا على العرض المحدد. يجب على العميل استدعاء XScreenSaverQueryExtension قبل استدعاء أي وظيفة XScreenSaver أخرى للتفاوض على إصدار بروتوكول متوافق؛ وإلا سيحصل العميل على سلوك غير محدد (قد تعمل XScreenSaver أو لا تعمل).

إذا كان الامتداد مدعومًا، يتم إرجاع رقم الحدث لأحداث ScreenSaverNotify في القيمة المشار إليها بواسطة event_base. نظرًا لعدم تعريف أخطاء إضافية بواسطة هذا الامتداد، فإن نتائج error_base غير محددة.

XScreenSaverQueryVersion يُرجع True إذا نجح الطلب؛ يتم إرجاع قيم إصدارات البروتوكول الرئيسية والثانوية المدعومة بواسطة الخادم في major_version_return و minor_version_return.

XScreenSaverAllocInfo يخصص ويُرجع بنية XScreenSaverInfo للاستخدام في استدعاءات XScreenSaverQueryInfo. تتم تهيئة جميع الحقول في البنية إلى الصفر. إذا كانت الذاكرة غير كافية، يتم إرجاع NULL. يمكن تحرير نتائج هذا الروتين باستخدام XFree.

XScreenSaverQueryInfo يُرجع معلومات حول الحالة الحالية لخادم الشاشة في saver_info ويتم إرجاع قيمة غير صفرية. إذا لم يكن الامتداد مدعومًا، لا يتم تغيير saver_info ويتم إرجاع 0.

يحدد الحقل state ما إذا كانت شاشة التوقف نشطة حاليًا أم لا وكيف يجب تفسير قيمة til-or-since:

Off
الشاشة ليست قيد الحفظ حاليًا؛ til-or-since تحدد عدد المللي ثانية حتى من المتوقع تنشيط شاشة التوقف.
On
الشاشة قيد الحفظ حاليًا؛ til-or-since تحدد عدد المللي ثانية منذ تنشيط شاشة التوقف.
Disabled
شاشة التوقف معطلة حاليًا؛ til-or-since تساوي صفر.

يحدد الحقل kind الآلية التي يتم استخدامها حاليًا أو التي كان سيتم استخدامها لو تم حفظ الشاشة:

Blanked
تم تعطيل إشارة الفيديو لشاشة العرض.
Internal
تم عرض صورة شاشة توقف مدمجة يعتمد على الخادم؛ إما لم يقم أي عميل بضبط سمات نافذة شاشة التوقف أو كان لدى عميل آخر استيلاء على الخادم عند تنشيط شاشة التوقف.
External
تم رسم نافذة شاشة التوقف بسمات تم ضبطها بواسطة عميل يستخدم طلب ScreenSaverSetAttributes.

يحدد الحقل idle عدد المللي ثانية منذ آخر مدخل تم تلقيه من المستخدم على أي من أجهزة الإدخال.
يحدد الحقل event-mask أي أحداث شاشة توقف، إن وجدت، طلبها هذا العميل باستخدام ScreenSaverSelectInput.

XScreenSaverSelectInput يطلب إنشاء أحداث متعلقة بشاشة التوقف لهذا العميل. إذا لم يتم تعيين أي بتات في event-mask، فلن يتم إنشاء أي أحداث. بخلاف ذلك، يمكن تعيين أي مزيج من البتات التالية:

ScreenSaverNotify
إذا تم تعيين هذا البت، يتم إنشاء أحداث ScreenSaverNotify في كل مرة يتم فيها تنشيط شاشة التوقف أو إلغاء تنشيطها.
ScreenSaverCycle
إذا تم تعيين هذا البت، يتم إنشاء أحداث ScreenSaverNotify في كل مرة تمر فيها فترة دورة شاشة التوقف.

XScreenSaverSetAttributes يضبط السمات التي سيتم استخدامها في المرة القادمة التي يتم فيها تنشيط شاشة التوقف الخارجية. إذا كان لدى عميل آخر السمات مضبوطة حاليًا، يتم إنشاء خطأ BadAccess ويتجاهل الطلب.
بخلاف ذلك، يتم التحقق من سمات النافذة المحددة كما لو تم استخدامها في طلب CreateWindow أساسي يكون والده هو الجذر. يتم تجاهل حقل override-redirect لأنه تم تعيينه ضمنيًا إلى True. إذا أدت سمات النافذة إلى خطأ وفقًا لقواعد CreateWindow، يتم تجاهل الطلب.
بخلاف ذلك، يتم تخزين السمات وستدخل حيز التنفيذ عند التنشيط التالي الذي يحدث عندما لا يكون الخادم مستولى عليه بواسطة عميل آخر. قد يتم تحرير أي موارد محددة لسمات background-pixmap أو cursor على الفور

هذا الموقع غير مرتبط بـ Arch Linux ولا تابع له بأي شكل من الأشكال. المحتوى هو ترجمة فقط لصفحات دليل Arch المنشورة على man.archlinux.org.

. الخادم حر في نسخ موارد background-pixmap أو cursor أو استخدامها في مكانها؛ لذلك، فإن تأثير تغيير محتويات تلك الموارد غير محدد. إذا لم يعد colormap المحدد موجودًا عند تفعيل موفر شاشة التوقف، فسيتم استخدام colormap الأصل بدلاً منه. إذا لم يتم إنشاء أي أخطاء بواسطة هذا الطلب، فسيتم تحرير سمات نافذة موفر الشاشة السابقة التي تم تعيينها بواسطة هذا العميل.
عندما يتم تفعيل موفر الشاشة التالي، ولم يكن الخادم محجوزًا بواسطة عميل آخر، فسيتم إنشاء نافذة موفر الشاشة، إذا لزم الأمر، وتعيينها إلى السمات المحددة، وسيتم إنشاء الأحداث كالمعتاد. سيتم تثبيت colormap المرتبط بنافذة موفر الشاشة. أخيرًا، سيتم رسم نافذة موفر الشاشة.
تظل النافذة مرسومة وفي أعلى ترتيب التكديس حتى يتم إلغاء تنشيط موفر الشاشة استجابةً لنشاط على أي من أجهزة إدخال المستخدم، أو طلب ForceScreenSaver بقيمة Reset، أو أي طلب قد يتسبب في إلغاء رسم النافذة.
إذا تم تفعيل موفر الشاشة أثناء حجز الخادم بواسطة عميل آخر، فسيتم استخدام آلية الموفر الداخلية. يمكن استخدام طلب ForceScreenSaver بقيمة Active لإلغاء تنشيط الموفر الداخلي وتفعيل الموفر الخارجي.
إذا انقطع اتصال عميل موفر الشاشة بالخادم أثناء تفعيل موفر الشاشة، ولم يتم تعيين وضع إنهاء العميل على RetainPermanent أو RetainTemporary، فسيتم إلغاء تنشيط موفر الشاشة الحالي وسيتم تفعيل موفر الشاشة الداخلي فورًا.
عند إلغاء تنشيط موفر الشاشة، سيتم إلغاء تثبيت colormap الخاص بنافذة موفر الشاشة وسيتم إلغاء رسم النافذة (باستثناء ما هو موضح أدناه). سيتم فصل معرف موفر الشاشة XID عن النافذة، وقد يقوم الخادم، ولكنه غير ملزم، بتدمير النافذة مع أي أطفال.
عندما يتم إلغاء تنشيط موفر الشاشة ثم إعادة تنشيطه فورًا (مثل عند تبديل موفري الشاشة)، قد يترك الخادم نافذة موفر الشاشة مرسومة (عادةً لتجنب إنشاء تعرضات).

XScreenSaverUnsetAttributes يوجه الخادم إلى تجاهل أي سمات سابقة لنافذة موفر الشاشة تم تعيينها بواسطة هذا العميل.

XScreenSaverRegister يخزن XID المعطى في خاصية _SCREEN_SAVER_ID (من type المعطى) على النافذة الجذرية للشاشة المحددة screen. يعيد صفر إذا واجه خطأ ولم يتم تغيير الخاصية، وإلا فإنه يعيد قيمة غير صفرية.

XScreenSaverUnregister يزيل أي _SCREEN_SAVER_ID من النافذة الجذرية للشاشة المحددة screen. يعيد صفر إذا واجه خطأ وتم تغيير الخاصية، وإلا فإنه يعيد قيمة غير صفرية.

XScreenSaverGetRegistered يعيد XID و type المخزنين في خاصية _SCREEN_SAVER_ID على النافذة الجذرية للشاشة المحددة screen. يعيد صفر إذا واجه خطأ أو إذا كانت الخاصية غير موجودة أو ليست بالتنسيق الصحيح؛ وإلا فإنه يعيد قيمة غير صفرية.

XScreenSaverSuspend يعلق مؤقت موفر الشاشة و DPMS مؤقتًا إذا كان suspend هو 'True'، ويعيد تشغيل المؤقت إذا كان suspend هو 'False'.
يجب استخدام هذه الوظيفة بواسطة التطبيقات التي لا تريد أن يتم تفعيل موفر الشاشة أو DPMS أثناء قيامها، على سبيل المثال، بعملية تشغيل تسلسل وسائط، أو تقدم معلومات مرئية بشكل مستمر للمستخدم أثناء حالة غير تفاعلية. هذه الوظيفة غير مخصصة ليتم استدعاؤها بواسطة تطبيق موفر شاشة خارجي.
إذا تم استدعاء XScreenSaverSuspend عدة مرات مع تعيين suspend إلى 'True'، فيجب استدعاؤها عددًا متساويًا من المرات مع تعيين suspend إلى 'False' لإعادة تشغيل مؤقت موفر الشاشة. ليس لهذا الطلب أي تأثير إذا حاول عميل استئناف موفر الشاشة دون تعليقه أولاً. وبالتالي، لا يمكن استخدام XScreenSaverSuspend بواسطة عميل واحد لاستئناف موفر الشاشة إذا تم تعليقه بواسطة عميل آخر.
إذا أصبح عميل قام بتعليق موفر الشاشة غير متصل بخادم X، فسيتم إعادة تشغيل مؤقت موفر الشاشة تلقائيًا، ما لم يكن لا يزال معلقًا بواسطة عميل آخر. تعليق مؤقت موفر الشاشة لا يمنع التفعيل القسري لموفر الشاشة باستخدام طلب ForceScreenSaver، أو تعيين وضع DPMS باستخدام طلب DPMSForceLevel.
XScreenSaverSuspend لا يلغي تنشيط موفر الشاشة أو DPMS إذا كان أحدهما نشطًا في وقت تلقي الخادم X للطلب لتعليقهما. ولكن بمجرد إلغاء تنشيطهما، لن يتم تنشيطهما تلقائيًا مرة أخرى، حتى يقوم العميل بإلغاء التعليق.

الأخطاء

XScreenSaverSelectInput، و XScreenSaverQueryInfo، و XScreenSaverSetAttributes، و XScreenSaverUnsetAttributes ستقوم بإنشاء خطأ BadDrawable إذا لم يكن drawable معرفًا صالحًا. إذا تم تعيين أي بتات غير محددة في event-mask، فسيتم إنشاء خطأ BadValue بواسطة XScreenSaverSelectInput.

التوفر

XScreenSaverSuspend متاح في الإصدار 1.1 والإصدارات الأحدث من امتداد X Screen Saver. تم إصدار الإصدار 1.1 لأول مرة مع X11R7.1.
