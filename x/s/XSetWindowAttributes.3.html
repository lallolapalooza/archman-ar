XSetWindowAttributes(3)

الحزمة: libx11

التراخيص: MIT AND X11

XCreateWindow(3) دوال XLIB XCreateWindow(3)

الاسم

XCreateWindow, XCreateSimpleWindow, XSetWindowAttributes - إنشاء النوافذ وبنية سمات النافذة

الصيغة

Window XCreateWindow(Display *display, Window parent, int x, int y, unsigned int width, unsigned int height, unsigned int border_width, int depth, unsigned int class, Visual *visual, unsigned long valuemask, XSetWindowAttributes *attributes);

Window XCreateSimpleWindow(Display *display, Window parent, int x, int y, unsigned int width, unsigned int height, unsigned int border_width, unsigned long border, unsigned long background);

الوسائط

attributes
يحدد البنية التي تُؤخذ منها القيم (كما هو محدد بواسطة قناع القيم). يجب أن يحتوي قناع القيم على البتات المناسبة المضبوطة للإشارة إلى السمات التي تم تعيينها في البنية.
background
يحدد قيمة بكسل الخلفية للنافذة.
border
يحدد قيمة بكسل الحدود للنافذة.
border_width
يحدد عرض حدود النافذة المنشأة بالبكسل.
class
يحدد فئة النافذة المنشأة. يمكنك تمرير InputOutput أو InputOnly أو CopyFromParent. فئة CopyFromParent تعني أن الفئة مأخوذة من الأب.
depth
يحدد عمق النافذة. عمق CopyFromParent يعني أن العمق مأخوذ من الأب.
display
يحدد الاتصال بخادم X.
parent
يحدد النافذة الأب.
valuemask
يحدد سمات النافذة المعرفة في وسيطة السمات. هذا القناع هو الناتج المنطقي OR الشامل لبتات قناع السمات الصالحة. إذا كان valuemask صفراً، يتم تجاهل السمات ولا يُشار إليها.
visual
يحدد نوع المرئي. المرئي CopyFromParent يعني أن نوع المرئي مأخوذ من الأب.
width

height
يحددان العرض والارتفاع، وهما الأبعاد الداخلية للنافذة المنشأة ولا يشملان حدود النافذة المنشأة.
x

y
يحددان إحداثيات x و y، وهما الزاوية الخارجية العلوية اليسرى لحدود النافذة وتكون نسبية إلى داخل حدود النافذة الأب.

الوصف

تنشئ الدالة XCreateWindow نافذة فرعية غير معيّنة لنافذة أب محددة، وتعيد معرف النافذة للنافذة المنشأة، وتتسبب في أن يولّد خادم X حدث CreateNotify. تُوضع النافذة المنشأة في الأعلى في ترتيب التكديس بالنسبة للأشقاء.

نظام الإحداثيات له المحور X أفقي والمحور Y عمودي مع الأصل [0, 0] في الزاوية العلوية اليسرى. الإحداثيات صحيحة، من حيث البكسل، وتتطابق مع مراكز البكسل. لكل نافذة وخريطة بكسل نظام إحداثيات خاص بها. بالنسبة للنافذة، يكون الأصل داخل الحدود عند الزاوية الداخلية العلوية اليسرى.

يجب أن يكون border_width لنافذة InputOnly صفراً، وإلا ينتج خطأ BadMatch. بالنسبة للفئة InputOutput، يجب أن يكون نوع المرئي والعمق مزيجاً مدعوماً للشاشة، وإلا ينتج خطأ BadMatch. لا يلزم أن يكون العمق نفس عمق الأب، لكن يجب ألا يكون الأب نافذة من الفئة InputOnly، وإلا ينتج خطأ BadMatch. بالنسبة لنافذة InputOnly، يجب أن يكون العمق صفراً، ويجب أن يكون المرئي مدعوماً من الشاشة. إذا لم يتحقق أي من الشرطين، ينتج خطأ BadMatch. ومع ذلك، يمكن أن تكون النافذة الأب بأي عمق وفئة. إذا حددت أي سمة نافذة غير صالحة لنافذة، ينتج خطأ BadMatch.

النافذة المنشأة غير معروضة (معيّنة) بعد على شاشة المستخدم. لعرض النافذة، استدعِ XMapWindow. تستخدم النافذة الجديدة في البداية نفس المؤشر الخاص بأبيها. يمكن تعريف مؤشر جديد للنافذة الجديدة باستدعاء XDefineCursor. لن تكون النافذة مرئية على الشاشة إلا إذا كانت هي وجميع أسلافها معيّنة ولم تكن محجوبة بواسطة أي من أسلافها.

يمكن أن تولّد XCreateWindow أخطاء BadAlloc وBadColor وBadCursor وBadMatch وBadPixmap وBadValue وBadWindow.

تنشئ الدالة XCreateSimpleWindow نافذة فرعية InputOutput غير معيّنة لنافذة أب محددة، وتعيد معرف النافذة للنافذة المنشأة، وتتسبب في أن يولّد خادم X حدث CreateNotify. تُوضع النافذة المنشأة في الأعلى في ترتيب التكديس بالنسبة للأشقاء. أي جزء من النافذة يمتد خارج نافذة أبيها يتم قصه. يجب أن يكون border_width لنافذة InputOnly صفراً، وإلا ينتج خطأ BadMatch. ترث XCreateSimpleWindow عمقها وفئتها ومرئيها من أبيها. جميع سمات النافذة الأخرى، باستثناء الخلفية والحدود، لها قيمها الافتراضية.

يمكن أن تولّد XCreateSimpleWindow أخطاء BadAlloc وBadMatch وBadValue وBadWindow.

البنى

تحتوي بنية XSetWindowAttributes على:

/* بتات قناع قيمة سمة النافذة */

#define CWBackPixmap (1L<<0)
#define CWBackPixel (1L<<1)
#define CWBorderPixmap (1L<<2)
#define CWBorderPixel (1L<<3)
#define CWBitGravity (1L<<4)
#define CWWinGravity (1L<<5)
#define CWBackingStore (1L<<6)
#define CWBackingPlanes (1L<<7)
#define CWBackingPixel (1L<<8)
#define CWOverrideRedirect (1L<<9)
#define CWSaveUnder (1L<<10)
#define CWEventMask (1L<<11)
#define CWDontPropagate (1L<<12)
#define CWColormap (1L<<13)
#define CWCursor (1L<<14) 
/* القيم */
typedef struct {
        Pixmap background_pixmap;       /* الخلفية، أو None، أو ParentRelative */
        unsigned long background_pixel; /* بكسل الخلفية */
        Pixmap border_pixmap;           /* حدود النافذة أو CopyFromParent */
        unsigned long border_pixel;     /* قيمة بكسل الحدود */
        int bit_gravity;        /* إحدى قيم جذب البت */
        int win_gravity;        /* إحدى قيم جذب النافذة */
        int backing_store;      /* NotUseful, WhenMapped, Always */
        unsigned long backing_planes;   /* المستويات التي يجب الحفاظ عليها إن أمكن */
        unsigned long backing_pixel;    /* القيمة المستخدمة في استعادة المستويات */
        Bool save_under;        /* هل يجب حفظ البتات تحتها؟ (النوافذ المنبثقة) */
        long event_mask;        /* مجموعة الأحداث التي يجب حفظها */
        long do_not_propagate_mask;     /* مجموعة الأحداث التي يجب ألا تنتشر */
        Bool override_redirect; /* قيمة منطقية لـ override_redirect */
        Colormap colormap;      /* خريطة الألوان المرتبطة بالنافذة */
        Cursor cursor;          /* المؤشر المراد عرضه (أو None) */
} XSetWindowAttributes;

للحصول على شرح مفصل لأعضاء هذه البنية، راجع Xlib - C Language X Interface.

التشخيصات
BadAlloc
فشل الخادم في تخصيص المورد المطلوب أو ذاكرة الخادم.
BadColor
قيمة وسيطة Colormap لا تشير إلى Colormap معرّف.
BadCursor
قيمة وسيطة Cursor لا تشير إلى Cursor معرّف.
BadMatch
القيم غير موجودة لنافذة InputOnly.
BadMatch
بعض الوسائط أو زوج من الوسائط له النوع والنطاق الصحيحان لكنه يفشل في المطابقة بطريقة أخرى مطلوبة من قبل الطلب.
BadPixmap
قيمة وسيطة Pixmap لا تشير إلى Pixmap معرّف.
BadValue
بعض القيم الرقمية تقع خارج نطاق القيم المقبولة من قبل الطلب. ما لم يتم تحديد نطاق محدد لوسيطة، يتم قبول النطاق الكامل المحدد بواسطة نوع الوسيطة. أي وسيطة معرّفة كمجموعة من البدائل يمكن أن تولّد هذا الخطأ.
BadWindow
قيمة وسيطة Window لا تشير إلى Window معرّف.
