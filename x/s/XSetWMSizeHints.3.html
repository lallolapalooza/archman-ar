XSetWMSizeHints(3)

الحزمة: libx11

التراخيص: MIT AND X11

XAllocSizeHints(3) وظائف XLIB XAllocSizeHints(3)

الاسم

XAllocSizeHints, XSetWMNormalHints, XGetWMNormalHints, XSetWMSizeHints, XGetWMSizeHints, XSizeHints - تخصيص هيكل تلميحات الحجم وتعيين أو قراءة خاصية WM_NORMAL_HINTS لنافذة

الصياغة

XSizeHints *XAllocSizeHints(void);

void XSetWMNormalHints(Display *display, Window w, XSizeHints *hints);

Status XGetWMNormalHints(Display *display, Window w, XSizeHints *hints_return, long *supplied_return);

void XSetWMSizeHints(Display *display, Window w, XSizeHints *hints, Atom property);

Status XGetWMSizeHints(Display *display, Window w, XSizeHints *hints_return, long *supplied_return, Atom property);

hints

الوصف

تقوم الدالة XAllocSizeHints بتخصيص وإعادة مؤشر إلى هيكل XSizeHints. لاحظ أن جميع الحقول في هيكل XSizeHints تم تعيينها مبدئيًا إلى الصفر. إذا لم يكن هناك ذاكرة كافية متاحة، تعيد XAllocSizeHints قيمة NULL. لتحرير الذاكرة المخصصة لهذا الهيكل، استخدم XFree.

تقوم الدالة XSetWMNormalHints باستبدال تلميحات الحجم لخاصية WM_NORMAL_HINTS في النافذة المحددة. إذا لم تكن الخاصية موجودة بالفعل، تقوم XSetWMNormalHints بتعيين تلميحات الحجم لخاصية WM_NORMAL_HINTS في النافذة المحددة. يتم تخزين الخاصية بنوع WM_SIZE_HINTS وبصيغة 32.

يمكن لـ XSetWMNormalHints إنشاء أخطاء BadAlloc و BadWindow.

تقوم الدالة XGetWMNormalHints بإعادة تلميحات الحجم المخزنة في خاصية WM_NORMAL_HINTS في النافذة المحددة. إذا كانت الخاصية من نوع WM_SIZE_HINTS، وبصيغة 32، وكانت طويلة بما يكفي لاحتواء هيكل تلميحات الحجم القديم (قبل ICCCM) أو الجديد، تقوم XGetWMNormalHints بتعيين الحقول المختلفة لهيكل XSizeHints، وتعيين الوسيط supplied_return إلى قائمة الحقول التي قدمها المستخدم (سواء كانت تحتوي على قيم محددة أم لا)، وتعيد قيمة حالة غير صفرية. بخلاف ذلك، تعيد قيمة حالة صفرية.

إذا أعادت XGetWMNormalHints بنجاح وتم قراءة خاصية تلميحات الحجم القديمة (قبل ICCCM)، فإن الوسيط supplied_return سيحتوي على البتات التالية:

(USPosition|USSize|PPosition|PSize|PMinSize|
 PMaxSize|PResizeInc|PAspect)

إذا كانت الخاصية طويلة بما يكفي لاحتواء الحجم الأساسي وحقول الجاذبية للنافذة أيضًا، فإن الوسيط supplied_return سيحتوي أيضًا على البتات التالية:

PBaseSize|PWinGravity

يمكن لـ XGetWMNormalHints إنشاء خطأ PN BadWindow.

تقوم الدالة XSetWMSizeHints باستبدال تلميحات الحجم للخاصية المحددة في النافذة المحددة. إذا لم تكن الخاصية المحددة موجودة بالفعل، تقوم XSetWMSizeHints بتعيين تلميحات الحجم للخاصية المحددة في النافذة المحددة. يتم تخزين الخاصية بنوع WM_SIZE_HINTS وبصيغة 32. لتعيين تلميحات الحجم الطبيعية للنافذة، يمكنك استخدام الدالة XSetWMNormalHints.

يمكن لـ XSetWMSizeHints إنشاء أخطاء BadAlloc، BadAtom، و BadWindow.

تقوم الدالة XGetWMSizeHints بإعادة تلميحات الحجم المخزنة في الخاصية المحددة في النافذة المحددة. إذا كانت الخاصية من نوع WM_SIZE_HINTS، وبصيغة 32، وكانت طويلة بما يكفي لاحتواء هيكل تلميحات الحجم القديم (قبل ICCCM) أو الجديد، تقوم XGetWMSizeHints بتعيين الحقول المختلفة لهيكل XSizeHints، وتعيين الوسيط supplied_return إلى قائمة الحقول التي قدمها المستخدم (سواء كانت تحتوي على قيم محددة أم لا)، وتعيد قيمة حالة غير صفرية. بخلاف ذلك، تعيد قيمة حالة صفرية. للحصول على تلميحات الحجم الطبيعية للنافذة، يمكنك استخدام الدالة XGetWMNormalHints.

إذا أعادت XGetWMSizeHints بنجاح وتم قراءة خاصية تلميحات الحجم القديمة (قبل ICCCM)، فإن الوسيط supplied_return سيحتوي على البتات التالية:

(USPosition|USSize|PPosition|PSize|PMinSize|
 PMaxSize|PResizeInc|PAspect)

إذا كانت الخاصية طويلة بما يكفي لاحتواء الحجم الأساسي وحقول الجاذبية للنافذة أيضًا، فإن الوسيط supplied_return سيحتوي أيضًا على البتات التالية:

PBaseSize|PWinGravity

يمكن لـ XGetWMSizeHints إنشاء أخطاء BadAtom و BadWindow.

الخصائص

WM_NORMAL_HINTS
تلميحات الحجم لنافذة في حالتها الطبيعية. النوع C لهذه الخاصية هو XSizeHints.

الهياكل

يحتوي هيكل XSizeHints على:

/* بتات قناع تلميحات الحجم */

#define USPosition (1L << 0) /* حدد المستخدم x، y */
#define USSize (1L << 1) /* حدد المستخدم العرض، الارتفاع */
#define PPosition (1L << 2) /* حدد البرنامج الموضع */
#define PSize (1L << 3) /* حدد البرنامج الحجم */
#define PMinSize (1L << 4) /* حدد البرنامج الحد الأدنى للحجم */
#define PMaxSize (1L << 5) /* حدد البرنامج الحد الأقصى للحجم */
#define PResizeInc (1L << 6) /* حدد البرنامج زيادة تغيير الحجم */
#define PAspect (1L << 7) /* حدد البرنامج الحد الأدنى والأقصى لنسب العرض إلى الارتفاع */
#define PBaseSize (1L << 8)
#define PWinGravity (1L << 9)
#define PAllHints (PPosition| PSize| PMinSize| PMaxSize| PResizeInc| PAspect) 
/* القيم */
typedef struct {
        long flags;     /* علامات تشير إلى الحقول المحددة في هذا الهيكل */
        int x, y;       /* مهمل */
        int width, height;      /* مهمل */
        int min_width, min_height;
        int max_width, max_height;
        int width_inc, height_inc;
        struct {
               int x;   /* البسط */
               int y;   /* المقام */
        } min_aspect, max_aspect;
        int base_width, base_height;
        int win_gravity;
        /* قد يتم توسيع هذا الهيكل في المستقبل */
} XSizeHints;

أعضاء x و y و width و height مهملة الآن، وهي متبقية فقط لأسباب التوافق. يحدد الأعضاء min_width و min_height الحد الأدنى لحجم النافذة الذي لا يزال يسمح للتطبيق بأن يكون مفيدًا. يحدد الأعضاء max_width و max_height الحد الأقصى لحجم النافذة. تحدد الأعضاء width_inc و height_inc تقدمًا حسابيًا للأحجام (من الحد الأدنى إلى الحد الأقصى) التي تفضل النافذة إعادة تغيير حجمها إليها. يتم التعبير عن الأعضاء min_aspect و max_aspect كنسب من x و y، وتسمح للتطبيق بتحديد نطاق نسب العرض إلى الارتفاع التي يفضلها. تحدد الأعضاء base_width و base_height الحجم المرغوب للنافذة. سيفسر مدير النوافذ موضع النافذة وعرض حدودها لوضع نقطة المستطيل الخارجي للنافذة الكاملة المحددة بواسطة عضو win_gravity. يشمل المستطيل الخارجي للنافذة أي حدود أو زخارف يوفرها مدير النوافذ. بمعنى آخر، إذا قرر مدير النوافذ وضع النافذة حيث يكون العميل
