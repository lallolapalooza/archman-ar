الهياكل

typedef struct {
    int type;	/* نوع الحدث */
    unsigned long serial;	/* رقم آخر طلب تمت معالجته بواسطة الخادم */
    Bool send_event;	/* صحيح إذا جاء هذا من طلب SendEvent */
    Display *display;	/* العرض الذي تمت قراءة الحدث منه */
    Window window;	/* نافذة الحدث */
    int kind;	/* ShapeBounding أو ShapeClip */
    int x, y;	/* حدود المنطقة الجديدة */
    unsigned width, height;
    Time time;	/* الطابع الزمني للخادم عند تغيير المنطقة */
    Bool shaped;	/* صحيح إذا كانت المنطقة موجودة */
} XShapeEvent;

الوصف

امتداد شكل النافذة غير المستطيلة X11 يضيف نوافذ غير مستطيلة إلى نظام نوافذ X.
