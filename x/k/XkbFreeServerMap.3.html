XkbFreeServerMap(3)

الحزمة: libx11

التراخيص: MIT AND X11

XkbFreeServerMap(3) وظائف Xkb XkbFreeServerMap(3)

الاسم

XkbFreeServerMap - تحرير الذاكرة المستخدمة بواسطة عضو الخادم في بنية XkbDescRec

الخلاصة

void XkbFreeServerMap (XkbDescPtr xkb, unsigned int which, Bool free_all);

الوسائط

xkb
وصف لوحة المفاتيح الذي يحتوي على خريطة الخادم للتحرير.
which
قناع يحدد مكونات الخريطة للتحرير.
free_all
True => تحرير جميع مكونات خريطة الخادم والخادم نفسه.

الوصف

تقوم الدالة XkbFreeServerMap بتحرير المكونات المحددة لخريطة الخادم في بنية XkbDescRec المحددة بواسطة المعلمة xkb وتعيين قيم مكونات البنية المقابلة إلى NULL. تحدد المعلمة which مجموعة من أقنعة الخريطة للخادم وهي OR شامل بت (bitwise inclusive OR) للأقنعة المدرجة في الجدول 1. إذا كانت free_all صحيحة (True)، يتم تجاهل which وتقوم XkbFreeServerMap بتحرير كل مكون بنية غير فارغة (non-NULL) في خريطة الخادم، وتحرير بنية XkbServerMapRec التي تشير إليها العضو server للمعلمة xkb، وتعيين العضو server إلى NULL.

الجدول 1 أقنعة XkbAllocServerMap
القناع التأثير
XkbExplicitComponentsMask تُستخدم حقلا min_key_code و max_key_code للمعلمة xkb لتخصيص الحقل explicit لخريطة الخادم.
XkbKeyActionsMask تُستخدم حقلا min_key_code و max_key_code للمعلمة xkb لتخصيص الحقل key_acts لخريطة الخادم. تُستخدم المعلمة count_acts لتخصيص الحقل acts لخريطة الخادم.
XkbKeyBehaviorsMask تُستخدم حقلا min_key_code و max_key_code للمعلمة xkb لتخصيص الحقل behaviors لخريطة الخادم.
XkbVirtualModMapMask تُستخدم حقلا min_key_code و max_key_code للمعلمة xkb لتخصيص الحقل vmodmap لخريطة الخادم.

الهياكل

يتم تقديم الوصف الكامل للوحة مفاتيح Xkb بواسطة XkbDescRec. تمثل مكونات البنية في XkbDescRec المكونات الرئيسية لـ Xkb.

typedef struct {
   struct _XDisplay * display;      /* اتصال بخادم X */
   unsigned short     flags;        /* خاص بـ Xkb، لا تعدله */
   unsigned short     device_spec;  /* جهاز الاهتمام */
   KeyCode            min_key_code; /* أقل رمز مفتاح للجهاز */
   KeyCode            max_key_code; /* أقصى رمز مفتاح للجهاز */
   XkbControlsPtr     ctrls;        /* عناصر التحكم */
   XkbServerMapPtr    server;       /* خريطة مفاتيح الخادم */
   XkbClientMapPtr    map;          /* خريطة مفاتيح العميل */
   XkbIndicatorPtr    indicators;   /* خريطة المؤشرات */
   XkbNamesPtr        names;        /* أسماء لجميع المكونات */
   XkbCompatMapPtr    compat;       /* خريطة التوافق */
   XkbGeometryPtr     geom;         /* الهندسة المادية للوحة المفاتيح */
} XkbDescRec, *XkbDescPtr;
يشير الحقل display إلى بنية عرض X. الحقل flags خاص بالمكتبة: تعديل flags قد يؤدي إلى نتائج غير متوقعة. يحدد الحقل device_spec معرف الجهاز لجهاز إدخال لوحة المفاتيح، أو XkbUseCoreKeyboard، الذي يحدد جهاز لوحة المفاتيح الأساسي. يحدد حقلا min_key_code و max_key_code أقل وأعلى رمز مفتاح يمكن إرجاعه بواسطة لوحة المفاتيح.

كل مكون بنية له بت قناع مطابق يُستخدم في استدعاءات الدوال للإشارة إلى أنه يجب معالجة البنية بطريقة ما، مثل تخصيصها أو تحريرها. تظهر هذه الأقنعة وعلاقاتها بالحقول في XkbDescRec في الجدول 2.

الجدول 2 بتات القناع لـ XkbDescRec
بت القناع حقل XkbDescRec القيمة
XkbControlsMask ctrls (1L<<0)
XkbServerMapMask server (1L<<1)
XkbIClientMapMask map (1L<<2)
XkbIndicatorMapMask indicators (1L<<3)
XkbNamesMask names (1L<<4)
XkbCompatMapMask compat (1L<<5)
XkbGeometryMask geom (1L<<6)
XkbAllComponentsMask جميع الحقول (0x7f)

تحتوي خريطة الخادم Xkb على المعلومات التي يحتاجها الخادم لتفسير أحداث المفاتيح وهي من نوع XkbServerMapRec:

#define XkbNumVirtualMods          16
typedef struct {                    /* خريطة الخادم */
    unsigned short    num_acts;     /* # من الإدخالات المستخدمة في acts */
    unsigned short    size_acts;    /* # من الإدخالات في acts */
    XkbAction *       acts;         /* جداول ثنائية الأبعاد خطية لإجراءات المفاتيح، 1 لكل رمز مفتاح */
    XkbBehavior *     behaviors;    /* سلوكيات المفاتيح، 1 لكل رمز مفتاح */
    unsigned short *  key_acts;     /* فهرس إلى acts، 1 لكل رمز مفتاح */
    unsigned char *   explicit;     /* تجاوزات صريحة لإعادة التعيين الأساسي، 1 لكل مفتاح */
    unsigned char     vmods[XkbNumVirtualMods]; /* تعديلات حقيقية مرتبطة بالتعديلات الافتراضية */
    unsigned short *  vmodmap;      /* تعديلات افتراضية مرتبطة بالمفتاح، 1 لكل رمز مفتاح */
} XkbServerMapRec, *XkbServerMapPtr;
