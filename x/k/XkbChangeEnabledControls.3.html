الحزمة: libx11

XkbChangeEnabledControls(3) وظائف XKB XkbChangeEnabledControls(3)

الاسم

XkbChangeEnabledControls - معالجة التحكم EnabledControls

الخلاصة

Bool XkbChangeEnabledControls (Display *dpy, unsigned int device_spec, unsigned int mask, unsigned int values);

الوسائط

dpy
اتصال بخادم X
device_spec
جهاز لوحة المفاتيح المراد تعديله
mask
بت واحد -> عناصر التحكم المراد تمكينها / تعطيلها
values
بت واحد => تمكين، بت 0 => تعطيل

الوصف

التحكم EnabledControls هو قناع بت حيث يعني كل بت يتم تشغيله أن التحكم المقابل ممكّن، وعند إيقاف تشغيله، يتم تعطيله. وهو يتوافق مع حقل enabled_ctrls من هيكل XkbControlsRec (انظر الهياكل). بتات وصف أي عناصر تحكم قيد التشغيل أو متوقفة عن العمل محددة في الجدول 1.

يعرض الجدول 1 القيم الفعلية لبتات القناع الفردية المستخدمة لتحديد عناصر التحكم للتعديل ولتمكين وتعطيل التحكم. لاحظ أن نفس بت القناع يستخدم لتحديد التعديلات العامة على المعلمات المستخدمة لتكوين التحكم (أي)، ولتمكين وتعطيل التحكم (enabled_ctrls). الشذوذ في الجدول (لا توجد "موافق" في العمود) للعناصر التي لا تحتوي على سمات قابلة للتكوين؛ وللعناصر التي ليست عناصر منطقية وبالتالي لا يمكن تمكينها أو تعطيلها.

يحدد الوسيط mask عناصر التحكم المنطقية المراد تمكينها أو تعطيلها، ويحدد قناع values الحالة الجديدة لتلك العناصر. القيم الصالحة لكلا القناعين تتكون من OR شامل للبتات مأخوذة من مجموعة بتات القناع في الجدول 1، باستخدام فقط تلك الأقنعة التي تحتوي على "موافق" في عمود enabled_ctrls.

إذا كان خادم X لا يدعم إصدارًا متوافقًا من Xkb أو لم يتم تهيئة امتداد Xkb بشكل صحيح، فإن XkbChangeEnabledControls يعيد False؛ وإلا، فإنه يرسل الطلب إلى خادم X ويعيد True.

لاحظ أن التحكم EnabledControls يقوم فقط بتمكين وتعطيل عناصر التحكم؛ لا يقوم بتكوينها. بعض عناصر التحكم، مثل التحكم AudibleBell، لا تحتوي على سمات تكوين وبالتالي يتم التعامل معها فقط عن طريق تمكينها وتعطيلها. ومع ذلك، يمتلك الآخرون سمات إضافية لتكوين سلوكهم. على سبيل المثال، يستخدم التحكم RepeatControl حقلي repeat_delay و repeat_interval لوصف السلوك الزمني للمفاتيح التي تتكرر. يتم تشغيل أو إيقاف تشغيل سلوك RepeatControl اعتمادًا على قيمة بت XkbRepeatKeysMask، ولكن يجب عليك استخدام وسائل أخرى، كما هو موضح في هذا الفصل، لتكوين سلوكه بالتفصيل.

قيم الإرجاع

True
يدعم خادم X إصدارًا متوافقًا من Xkb.
False
خادم X لا يدعم إصدارًا متوافقًا من Xkb أو لم يتم تهيئة امتداد Xkb بشكل صحيح.

الهياكل

يتم تعريف هيكل XkbControlsRec على النحو التالي:
    #define XkbMaxLegalKeyCode     255
    #define XkbPerKeyBitArraySize  ((XkbMaxLegalKeyCode+1)/8)
    typedef struct {
        unsigned char   mk_dflt_btn;      /* الزر الافتراضي للماوس الذي يقوده لوحة المفاتيح */
        unsigned char   num_groups;       /* عدد مجموعات لوحة المفاتيح */
        unsigned char   groups_wrap;      /* كيفية التفاف المجموعات خارج النطاق */
        XkbModsRec      internal;         /* يحدد معدلات الخادم الداخلية */
        XkbModsRec      ignore_lock;      /* المعدلات التي يجب تجاهلها عند التحقق من الالتقاط */
        unsigned int    enabled_ctrls;    /* 1 بت => التحكم المنطقي المقابل ممكّن */
        unsigned short  repeat_delay;     /* تأخير بالمللي ثانية حتى التكرار الأول */
        unsigned short  repeat_interval;  /* تأخير بالمللي ثانية بين التكرارات */
        unsigned short  slow_keys_delay;  /* الحد الأدنى للوقت بالمللي ثانية الذي يجب أن يكون فيه المفتاح مضغوطًا ليكون صحيحًا */
        unsigned short  debounce_delay;   /* تأخير بالمللي ثانية قبل إعادة تنشيط المفتاح */
        unsigned short  mk_delay;         /* تأخير بالمللي ثانية لحدث حركة الماوس الثاني */
        unsigned short  mk_interval;      /* تأخير بالمللي ثانية بين أحداث تكرار الماوس */
        unsigned short  mk_time_to_max;   /* عدد الفواصل حتى حركة الماوس المستمرة */
        unsigned short  mk_max_speed;     /* مضاعف لأقصى سرعة للماوس */
        short           mk_curve;         /* يحدد نوع منحنى حركة الماوس */
        unsigned short  ax_options;       /* 1 بت => خيارات Access X ممكّنة */
        unsigned short  ax_timeout;       /* ثوانٍ حتى يتم تعطيل Access X */
        unsigned short  axt_opts_mask;    /* 1 بت => خيارات لإعادة التعيين عند انتهاء مهلة Access X */
        unsigned short  axt_opts_values;  /* 1 بت => تشغيل الخيار، 0 => إيقاف */
        unsigned int    axt_ctrls_mask;   /* أي بتات في enabled_ctrls للتعديل */
        unsigned int    axt_ctrls_values; /* قيم للبتات الجديدة في enabled_ctrls */
        unsigned char   per_key_repeat[XkbPerKeyBitArraySize];  /* تكرار تلقائي لكل مفتاح */
     } XkbControlsRec, *XkbControlsPtr;
