XKeyEvent(3)

الحزمة: libx11

التراخيص: MIT AND X11

XButtonEvent(3) وظائف XLIB XButtonEvent(3)

الاسم

XButtonEvent, XKeyEvent, XMotionEvent - هياكل أحداث KeyPress, KeyRelease, ButtonPress, ButtonRelease, و MotionNotify

هياكل

تحتوي هياكل أحداث KeyPress, KeyRelease, ButtonPress, ButtonRelease, و MotionNotify على:

typedef struct {
        int type;       /* ButtonPress أو ButtonRelease */
        unsigned long serial;   /* # آخر طلب تمت معالجته بواسطة الخادم */
        Bool send_event;        /* صحيح إذا جاء هذا من طلب SendEvent */
        Display *display;       /* العرض الذي تمت قراءة الحدث منه */
        Window window;  /* نافذة "الحدث" التي يتم الإبلاغ عنها بالنسبة إليها */
        Window root;    /* النافذة الجذرية التي وقع عليها الحدث */
        Window subwindow;       /* نافذة فرعية */
        Time time;      /* بالمللي ثانية */
        int x, y;       /* إحداثيات المؤشر x، y في نافذة الحدث */
        int x_root, y_root;     /* الإحداثيات بالنسبة للنافذة الجذرية */
        unsigned int state;     /* قناع المفتاح أو الزر */
        unsigned int button;    /* التفاصيل */
        Bool same_screen;       /* علم الشاشة نفسها */
} XButtonEvent;
typedef XButtonEvent XButtonPressedEvent;
typedef XButtonEvent XButtonReleasedEvent;
typedef struct {
        int type;       /* KeyPress أو KeyRelease */
        unsigned long serial;   /* # آخر طلب تمت معالجته بواسطة الخادم */
        Bool send_event;        /* صحيح إذا جاء هذا من طلب SendEvent */
        Display *display;       /* العرض الذي تمت قراءة الحدث منه */
        Window window;  /* نافذة "الحدث" التي يتم الإبلاغ عنها بالنسبة إليها */
        Window root;    /* النافذة الجذرية التي وقع عليها الحدث */
        Window subwindow;       /* نافذة فرعية */
        Time time;      /* بالمللي ثانية */
        int x, y;       /* إحداثيات المؤشر x، y في نافذة الحدث */
        int x_root, y_root;     /* الإحداثيات بالنسبة للنافذة الجذرية */
        unsigned int state;     /* قناع المفتاح أو الزر */
        unsigned int keycode;   /* التفاصيل */
        Bool same_screen;       /* علم الشاشة نفسها */
} XKeyEvent;
typedef XKeyEvent XKeyPressedEvent;
typedef XKeyEvent XKeyReleasedEvent;
typedef struct {
        int type;       /* MotionNotify */
        unsigned long serial;   /* # آخر طلب تمت معالجته بواسطة الخادم */
        Bool send_event;        /* صحيح إذا جاء هذا من طلب SendEvent */
        Display *display;       /* العرض الذي تمت قراءة الحدث منه */
        Window window;  /* نافذة "الحدث" التي يتم الإبلاغ عنها بالنسبة إليها */
        Window root;    /* النافذة الجذرية التي وقع عليها الحدث */
        Window subwindow;       /* نافذة فرعية */
        Time time;      /* بالمللي ثانية */
        int x, y;       /* إحداثيات المؤشر x، y في نافذة الحدث */
        int x_root, y_root;     /* الإحداثيات بالنسبة للنافذة الجذرية */
        unsigned int state;     /* قناع المفتاح أو الزر */
        char is_hint;   /* التفاصيل */
        Bool same_screen;       /* علم الشاشة نفسها */
} XMotionEvent;
typedef XMotionEvent XPointerMovedEvent;

عندما تتلقى هذه الأحداث، يتم تعيين أعضاء هيكلها على النحو التالي.

يتم تعيين عضو النوع (type) إلى اسم ثابت نوع الحدث الذي يحدده بشكل فريد. على سبيل المثال، عندما يقوم خادم X بالإبلاغ عن حدث GraphicsExpose لتطبيق عميل، فإنه يرسل بنية XGraphicsExposeEvent مع تعيين عضو النوع إلى GraphicsExpose. يتم تعيين عضو العرض (display) إلى مؤشر إلى العرض الذي تمت قراءة الحدث عليه. يتم تعيين عضو إرسال_الحدث (send_event) إلى True إذا جاء الحدث من طلب بروتوكول SendEvent. يتم تعيين عضو المسلسل (serial) من رقم المسلسل المبلغ عنه في البروتوكول ولكن تم توسيعه من 16 بت أقل أهمية إلى قيمة كاملة 32 بت. يتم تعيين عضو النافذة (window) إلى النافذة الأكثر فائدة لموزعي الأدوات (toolkit dispatchers).

تحتوي هذه الهياكل على الأعضاء المشتركة التالية: window، root، subwindow، time، x، y، x_root، y_root، state، و same_screen. يتم تعيين عضو النافذة (window) إلى النافذة التي تم إنشاء الحدث عليها ويشار إليها باسم نافذة الحدث. طالما أن الشروط المذكورة سابقًا قد تم استيفاؤها، فهذه هي النافذة التي يستخدمها خادم X للإبلاغ عن الحدث. يتم تعيين عضو النافذة الجذرية (root) إلى النافذة الجذرية لنافذة المصدر. يتم تعيين الأعضاء x_root و y_root إلى إحداثيات المؤشر بالنسبة لأصل النافذة الجذرية في وقت الحدث.

يتم تعيين عضو الشاشة نفسها (same_screen) للإشارة إلى ما إذا كانت نافذة الحدث على نفس الشاشة مثل النافذة الجذرية ويمكن أن تكون إما True أو False. إذا كانت True، فإن نافذة الحدث والنافذة الجذرية على نفس الشاشة. إذا كانت False، فإن نافذة الحدث والنافذة الجذرية ليستا على نفس الشاشة.

إذا كانت نافذة المصدر سفلى (inferior) لنافذة الحدث، يتم تعيين عضو النافذة الفرعية (subwindow) في الهيكل إلى النافذة الفرعية لنافذة الحدث التي هي نافذة المصدر أو النافذة الفرعية لنافذة الحدث التي هي سلف لنافذة المصدر. بخلاف ذلك، يقوم خادم X بتعيين عضو النافذة الفرعية (subwindow) إلى None. يتم تعيين عضو الوقت (time) إلى الوقت الذي تم فيه إنشاء الحدث ويتم التعبير عنه بالمللي ثانية.

إذا كانت نافذة الحدث على نفس الشاشة مثل النافذة الجذرية، يتم تعيين الأعضاء x و y إلى الإحداثيات بالنسبة لأصل نافذة الحدث. بخلاف ذلك، يتم تعيين هذه الأعضاء إلى الصفر.

يتم تعيين عضو الحالة (state) للإشارة إلى الحالة المنطقية لأزرار المؤشر والمفاتيح المعدلة قبل الحدث مباشرة، وهي OR الشامل للبتات لواحدة أو أكثر من أقنعة الأزرار أو المفاتيح المعدلة: Button1Mask, Button2Mask, Button3Mask, Button4Mask, Button5Mask, ShiftMask, LockMask, ControlMask, Mod1Mask, Mod2Mask, Mod3Mask, Mod4Mask, و Mod5Mask.

لكل من هذه الهياكل عضو يشير إلى التفاصيل. بالنسبة لهيكلي XKeyPressedEvent و XKeyReleasedEvent، يسمى هذا العضو keycode. يتم تعيينه إلى رقم يمثل مفتاحًا فعليًا على لوحة المفاتيح. keycode هو تمثيل عشوائي لأي مفتاح على لوحة المفاتيح (انظر الأقسام 12.7 و 16.1).

بالنسبة لهيكلي XButtonPressedEvent و XButtonReleasedEvent، يسمى هذا العضو button. يمثل زر المؤشر الذي تغيرت حالته ويمكن أن يكون قيمة Button1, Button2, Button3, Button4, أو Button5. بالنسبة لهيكل XPointerMovedEvent، يسمى هذا العضو is_hint. يمكن تعيينه إلى NotifyNormal أو NotifyHint.
