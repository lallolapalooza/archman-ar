XkbFreeCompatMap(3)

الحزمة: libx11

التراخيص: MIT AND X11

XkbFreeCompatMap(3) وظائف XKB XkbFreeCompatMap(3)

الاسم

XkbFreeCompatMap - تحرير خريطة توافق كاملة أو أجزاء محددة منها

الملخص

void XkbFreeCompatMap (XkbDescPtr xkb, unsigned int which, Bool free_map);

الوسائط

xkb
وصف Xkb الذي سيتم فيه تحرير خريطة التوافق
which
قناع مكونات خريطة التوافق المراد تحريرها
free_map
True => تحرير بنية XkbCompatMap نفسها

الوصف

يحدد which مكونات خريطة التوافق المراد تحريرها (انظر XkbGetCompatMap). which هو OR شامل للبتات المعروضة في الجدول 1.

جدول 1 أقنعة مكونات خريطة التوافق
القناع القيمة التأثير
XkbSymInterpMask (1<<0) تفسيرات الرموز
XkbGroupCompatMask (1<<1) خرائط المجموعات
XkbAllCompatMask (0x3) جميع مكونات خريطة التوافق

يشير free_map إلى ما إذا كان سيتم تحرير بنية XkbCompatMap نفسها. إذا كان free_map صحيحًا (True)، يتم تجاهل which، ويتم تحرير جميع مكونات خريطة التوافق غير NULL، ويتم تعيين الحقل compat في XkbDescRec الذي تشير إليه xkb إلى NULL.

البنى

    typedef struct _XkbCompatMapRec {
        XkbSymInterpretPtr    sym_interpret;            /* تفسيرات الرموز للمفاتيح */
        XkbModsRec            groups[XkbNumKbdGroups];  /* المجموعة => خريطة المعدل */
        unsigned short        num_si;                   /* عدد البنى المستخدمة في sym_interpret */
        unsigned short        size_si;                  /* عدد البنى المخصصة في sym_interpret */
    } XkbCompatMapRec, *XkbCompatMapPtr;

الوصف الكامل للوحة مفاتيح Xkb يتم توفيره بواسطة XkbDescRec. تمثل البنى المكونة في XkbDescRec المكونات الرئيسية لـ Xkb الموضحة في الشكل 1.1.

typedef struct {
   struct _XDisplay * display;      /* اتصال بخادم X */
   unsigned short     flags;        /* خاص بـ Xkb، لا تقم بالتعديل */
   unsigned short     device_spec;  /* الجهاز محل الاهتمام */
   KeyCode            min_key_code; /* أقل مفتاح لجهاز الإدخال */
   KeyCode            max_key_code; /* أعلى مفتاح لجهاز الإدخال */
   XkbControlsPtr     ctrls;        /* عناصر التحكم */
   XkbServerMapPtr    server;       /* خريطة مفاتيح الخادم */
   XkbClientMapPtr    map;          /* خريطة مفاتيح العميل */
   XkbIndicatorPtr    indicators;   /* خريطة المؤشرات */
   XkbNamesPtr        names;        /* أسماء جميع المكونات */
   XkbCompatMapPtr    compat;       /* خريطة التوافق */
   XkbGeometryPtr     geom;         /* الهندسة الفيزيائية للوحة المفاتيح */
} XkbDescRec, *XkbDescPtr;
يشير الحقل display إلى بنية عرض X. حقل flags خاص بالمكتبة: قد يؤدي تعديل flags إلى نتائج غير متوقعة. يحدد الحقل device_spec معرف الجهاز للوحة مفاتيح الإدخال، أو XkbUseCoreKeyboard، الذي يحدد جهاز لوحة المفاتيح الأساسي. تحدد حقلا min_key_code و max_key_code أقل وأكبر مفتاح يمكن إرجاعه بواسطة لوحة المفاتيح. تحدد الحقول الأخرى مكونات بنية وصف لوحة المفاتيح ويتم وصفها بالتفصيل في أقسام أخرى من هذا المستند. يحدد الجدول 2 الأقسام اللاحقة من هذا المستند التي تناقش المكونات الفردية لـ XkbDescRec.
جدول 2 إشارات مكونات XkbDescRec
حقل XkbDescRec لمزيد من المعلومات
ctrls الفصل 10
server الفصل 16
map الفصل 15
indicators الفصل 8
names الفصل 18
compat الفصل 17
geom الفصل 13

لكل مكون بنية، يوجد بت قناع مقابل يستخدم في استدعاءات الوظائف للإشارة إلى أنه يجب معالجة البنية بطريقة ما، مثل تخصيصها أو تحريرها. تظهر هذه الأقنعة وعلاقاتها بالحقول في XkbDescRec في الجدول 3.

جدول 3 بتات القناع لـ XkbDescRec
بت القناع حقل XkbDescRec القيمة
XkbControlsMask ctrls (1L<<0)
XkbServerMapMask server (1L<<1)
XkbIClientMapMask map (1L<<2)
XkbIndicatorMapMask indicators (1L<<3)
XkbNamesMask names (1L<<4)
XkbCompatMapMask compat (1L<<5)
XkbGeometryMask geom (1L<<6)
XkbAllComponentsMask جميع الحقول (0x7f)
