XkbResizeDeviceButtonActions(3)

الحزمة: libx11

التراخيص: MIT AND X11

XkbResizeDeviceButtonActions(3) XKB FUNCTIONS XkbResizeDeviceButtonActions(3)

الاسم

XkbResizeDeviceButtonActions - تخصيص مساحة إضافية لإجراءات أزرار في بنية XkbDeviceInfoRec

الملخص

Status XkbResizeDeviceButtonActions (XkbDeviceInfoPtr device_info, unsigned int new_total);

الوسائط

device_info
البنية التي يتم فيها تخصيص إجراءات الأزرار
new_total
العدد الإجمالي الجديد لإجراءات الأزرار المطلوبة

الوصف

تقوم XkbResizeDeviceButtonActions بإعادة تخصيص المساحة، إذا لزم الأمر، لضمان وجود مساحة لإجمالي new_total من إجراءات الأزرار في بنية device_info. يتم تعيين أي إدخالات جديدة تم تخصيصها إلى الصفر. إذا نجحت، تعيد XkbResizeDeviceButtonActions القيمة Success. إذا كان new_total صفرًا، فسيتم حذف جميع إجراءات الأزرار، وسيتم تعيين device_info->num_btns إلى صفر، وسيتم تعيين device_info->btn_acts إلى NULL. إذا كانت device_info غير صالحة أو تجاوز new_total القيمة 255، يتم إرجاع BadValue. إذا حدث فشل في تخصيص الذاكرة، يتم إرجاع BadAlloc.

لتحرير بنية XkbDeviceInfoRec، استخدم XkbFreeDeviceInfo.

البنيات

يتم نقل المعلومات حول أجهزة امتداد إدخال X بين برنامج عميل وامتداد Xkb في بنية XkbDeviceInfoRec:

    typedef struct {
        char *               name;          /* اسم الجهاز */
        Atom                 type;          /* اسم لفئة الأجهزة */
        unsigned short       device_spec;   /* الجهاز محل الاهتمام */
        Bool                 has_own_state; /* True=>هذا الجهاز له حالته الخاصة */
        unsigned short       supported;     /* بتات تشير إلى القدرات المدعومة */
        unsigned short       unsupported;   /* بتات تشير إلى القدرات غير المدعومة */
        unsigned short       num_btns;      /* عدد الإدخالات في btn_acts */
        XkbAction *          btn_acts;      /* إجراءات الأزرار */
        unsigned short       sz_leds;       /* العدد الإجمالي للإدخالات في متجه LEDs */
        unsigned short       num_leds;      /* عدد الإدخالات الصالحة في متجه LEDs */
        unsigned short       dflt_kbd_fb;   /* معرف امتداد الإدخال للمؤشر الافتراضي (لوحة المفاتيح الأساسية) */
        unsigned short       dflt_led_fb;   /* معرف امتداد الإدخال للمؤشر الافتراضي */
        XkbDeviceLedInfoPtr  leds;          /* أوصاف LEDs */
    } XkbDeviceInfoRec, *XkbDeviceInfoPtr;

التشخيصات

BadAlloc
تعذر تخصيص مساحة
BadValue
وسيط خارج النطاق
