XkbListComponents(3)

الحزمة: libx11

التراخيص: MIT AND X11

XkbListComponents(3) XKB FUNCTIONS XkbListComponents(3)

الاسم

XkbListComponents - قائمة مكونات لأنواع مكونات واحدة أو أكثر

الملخص

XkbComponentListPtr XkbListComponents (Display *dpy, unsigned int device_spec, XkbComponentNamesPtr ptrns, int *max_inout);

المعاملات

dpy
اتصال بخادم X
device_spec
معرف الجهاز، أو XkbUseCoreKbd
ptrns
قائمة أسماء للمكونات ذات الأهمية
max_inout
الحد الأقصى لعدد الأسماء المعادة، عدد المتبقي

الوصف

يمكنك طلب قائمة المكونات من الخادم لأنواع مكونات واحدة أو أكثر. يأخذ الطلب شكل مجموعة من الأنماط، نمط واحد لكل نوع من أنواع المكونات، بما في ذلك نمط لوصف لوحة المفاتيح الكاملة. للحصول على هذه القائمة، استخدم XkbListComponents.

XkbListComponents يستعلم الخادم للحصول على قائمة بأسماء المكونات التي تطابق الأنماط المحددة في ptrns. ينتظر الرد ويعيد الأسماء المتطابقة للمكونات في بنية XkbComponentListRec. عندما تنتهي من استخدام البنية، يجب عليك تحريرها باستخدام XkbFreeComponentList. يشير device_spec إلى جهاز معين يهتم به المتصل. يُسمح للخادم (ولكنه غير ملزم) بتقييد رده على أجزاء من قاعدة البيانات ذات الصلة بهذا الجهاز المحدد.

ptrns هو مؤشر إلى XkbComponentNamesRec. يحتوي كل حقل في ptrns على نمط يحدد المكونات ذات الأهمية. يتكون كل نمط من أحرف من ترميز ISO Latin1، ولكنه يمكن أن يحتوي فقط على الأقواس، وأحرف البدل `?' و `*', والأحرف المسموح بها في اسم فئة المكون أو عضوه. قد يكون النمط NULL، وفي هذه الحالة لا يتم إرجاع أي مكونات لهذا النوع. تتطابق الأنماط مع أسماء المكونات حساسة لحالة الأحرف. يطابق حرف البدل `?' أي حرف واحد، باستثناء قوس الفتح أو الإغلاق؛ يطابق حرف البدل `*' أي عدد من الأحرف، باستثناء قوس الفتح أو الإغلاق. إذا سمح التنفيذ بأحرف إضافية في اسم فئة المكون أو عضوه بخلاف تلك التي يتطلبها امتداد Xkb، فإن نتيجة مقارنة أحد الأحرف الإضافية بأي من أحرف البدل تعتمد على التنفيذ.

إذا احتوى النمط على أحرف غير قانونية، يتم تجاهل الأحرف غير القانونية. تتم عملية المطابقة كما لو تم حذف الأحرف غير القانونية من النمط.

يستخدم max_inout للتحكم في كمية البيانات التي يتم نقلها من وإلى الخادم. عند الإدخال، يحدد الحد الأقصى لعدد الأسماء التي سيتم إرجاعها (العدد الإجمالي للأسماء في جميع فئات المكونات). عند العودة من XkbListComponents، يحتوي max_inout على عدد الأسماء التي تطابقت مع الطلب ولكن لم يتم إرجاعها بسبب الحد.

أسماء المكونات

أسماء المكونات لها الشكل class(member) حيث تصف class مجموعة فرعية من المكونات المتاحة لنوع معين، و member الاختياري يحدد مكونًا معينًا من هذه المجموعة الفرعية. على سبيل المثال، قد يحدد الاسم "atlantis(acme)" لمكون رموز الرموز المستخدمة لتخطيط لوحة المفاتيح الوطنية atlantis بواسطة المورد "acme."؛ لكل فئة عضو افتراضي اختياري - تشير الإشارات التي تحدد فئة ولكن ليس عضوًا إلى العضو الافتراضي للفئة، إذا كان موجودًا. لا يضع Xkb قيودًا على تفسير أسماء الفئات والأعضاء المستخدمة في أسماء المكونات.

يتم تحديد أسماء class و member باستخدام أحرف من مجموعة أحرف Latin-1. يجب أن تقبل تطبيقات Xkb جميع الأحرف الأبجدية الرقمية، والناقص (`-') والشرطة السفلية (`_') في أسماء الفئات أو الأعضاء، ويجب ألا تقبل الأقواس، والزائد، والخط العمودي، وعلامة النسبة المئوية، والنجمة، وعلامة الاستفهام، أو المسافات البيضاء. استخدام الأحرف الأخرى يعتمد على التنفيذ.

البنى

يتم تمرير الأنماط المسماة للمكونات المستخدمة لوصف الطلب إلى XkbListComponents باستخدام بنية XkbComponentNamesRec. ليس لهذه البنية قيود تخصيص خاصة أو علاقات متبادلة مع هياكل أخرى؛ قم بتخصيص وتحرير هذه البنية باستخدام استدعاءات malloc و free القياسية أو ما يعادلها:

    typedef struct _XkbComponentNames {
        char *         keymap;           /* أسماء خرائط المفاتيح */
        char *         keycodes;         /* أسماء رموز المفاتيح */
        char *         types;            /* أسماء الأنواع */
        char *         compat;           /* أسماء خرائط التوافق */
        char *         symbols;          /* أسماء الرموز */
        char *         geometry;         /* أسماء التخطيط */
    } XkbComponentNamesRec, *XkbComponentNamesPtr;

XkbListComponents يعيد مؤشرًا إلى XkbComponentListRec:

    typedef struct _XkbComponentList {
        int                      num_keymaps;  /* عدد الإدخالات في خريطة المفاتيح */
        int                      num_keycodes; /* عدد الإدخالات في رموز المفاتيح */
        int                      num_types;    /* عدد الإدخالات في الأنواع */
        int                      num_compat;   /* عدد الإدخالات في التوافق */
        int                      num_symbols;  /* عدد الإدخالات في الرموز */
        int                      num_geometry; /* عدد الإدخالات في التخطيط */
        XkbComponentNamePtr      keymap;       /* أسماء خرائط المفاتيح */
        XkbComponentNamePtr      keycodes;     /* أسماء رموز المفاتيح */
        XkbComponentNamePtr      types;        /* أسماء الأنواع */
        XkbComponentNamePtr      compat;       /* أسماء خرائط التوافق */
        XkbComponentNamePtr      symbols;      /* أسماء الرموز */
        XkbComponentNamePtr      geometry;     /* أسماء التخطيط */
    } XkbComponentListRec, *XkbComponentListPtr;
    typedef struct _XkbComponentName {
        unsigned short           flags;        /* تلميحات بخصوص اسم المكون */
        char *                   name;         /* اسم المكون */
    } XkbComponentNameRec, *XkbComponentNamePtr;

ملاحظات

لاحظ أن البنية المستخدمة لتحديد الأنماط عند الإدخال هي XkbComponentNamesRec، وأن البنية المستخدمة للاحتفاظ بأسماء المكونات الفردية عند الإرجاع هي XkbComponentNameRec (بدون 's' لاحقة في Name).
