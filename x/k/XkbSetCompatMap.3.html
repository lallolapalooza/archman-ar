XkbSetCompatMap(3) وظائف Xkb XkbSetCompatMap(3)

الاسم

XkbSetCompatMap - تعديل خريطة التوافق الخاصة بالخادم

الملخص

Bool XkbSetCompatMap (Display *display, unsigned int which, XkbDescPtr xkb, Bool update_actions);

المعاملات

display
اتصال بالخادم
which
قناع مكونات خريطة التوافق المراد تعيينها
xkb
مصدر مكونات خريطة التوافق
update_actions
صحيح => تطبيق على خريطة لوحة المفاتيح الخاصة بالخادم

الوصف

لتعديل خريطة التوافق الخاصة بالخادم، قم أولاً بتعديل نسخة محلية من خريطة توافق Xkb، ثم قم باستدعاء XkbSetCompatMap. يمكنك تخصيص خريطة توافق جديدة لهذا الغرض باستخدام XkbAllocCompatMap. يمكنك أيضًا استخدام خريطة توافق من خادم آخر، على الرغم من أنك تحتاج إلى ضبط حقل device_spec في XkbDescRec وفقًا لذلك. لاحظ أن تفسيرات الرموز في خريطة التوافق ( sym_interpret، متجه هياكل XkbSymInterpretRec) يتم تخصيصها أيضًا باستخدام نفس هذه الوظيفة. XkbSetCompatMap تنسخ معلومات خريطة التوافق من وصف لوحة المفاتيح في xkb إلى الخادم المحدد في خريطة توافق display للجهاز المحدد بواسطة حقل device_spec الخاص بـ xkb. ما لم تكن قد قمت بتعديل هذا الحقل بشكل خاص، فهو جهاز لوحة المفاتيح الافتراضي. which تحدد مكونات خريطة التوافق المراد تعيينها، وهي OR شامل للبتات المعروضة في الجدول 1.

الجدول 1 أقنعة مكونات خريطة التوافق
القناع القيمة التأثير
XkbSymInterpMask (1<<0) تفسيرات الرموز
XkbGroupCompatMask (1<<1) خرائط المجموعات
XkbAllCompatMask (0x3) جميع مكونات خريطة التوافق

بعد تحديث خريطة التوافق الخاصة به للجهاز المحدد، إذا كان update_actions صحيحًا، يقوم الخادم بتطبيق خريطة التوافق الجديدة على لوحة المفاتيح بأكملها للجهاز لتوليد مجموعة جديدة من دلالات المفاتيح، وحالة التوافق، وخريطة لوحة مفاتيح أساسية جديدة. إذا كان update_actions خاطئًا، فلن يتم استخدام خريطة التوافق الجديدة لتوليد أي تعديلات على دلالات الجهاز الحالية أو الحالة أو خريطة لوحة المفاتيح الأساسية. أحد أسباب عدم تطبيق خريطة التوافق على الفور قد يكون إذا كان يتم تكوين خادم لمطابقة خادم آخر على أساس جزء بجزء؛ لا ينبغي تطبيق الخريطة حتى يتم تحديث كل شيء. لفرض تحديث في وقت لاحق، استخدم XkbSetCompatMap مع تعيين which إلى صفر و update_actions إلى صحيح. XkbSetCompatMap تعيد صحيحًا إذا نجحت وخاطئًا إذا فشلت. قد يقوم الخادم بالإبلاغ عن المشكلات التي يواجهها عند معالجة الطلب لاحقًا عبر أخطاء البروتوكول.

قيم الإرجاع

صحيح
تعيد الدالة XkbSetCompatMap صحيحًا إذا نجحت.
خاطئ
تعيد الدالة XkbSetCompatMap خاطئًا إذا فشلت.

الهياكل

الوصف الكامل للوحة مفاتيح Xkb يعطيه هيكل XkbDescRec. تمثل الهياكل المكونة في XkbDescRec المكونات الرئيسية لـ Xkb.

typedef struct {
   struct _XDisplay  *display;      /* اتصال بخادم X */
   unsigned short     flags;        /* خاص بـ Xkb، لا تقم بالتعديل */
   unsigned short     device_spec;  /* الجهاز المعني */
   KeyCode            min_key_code; /* الحد الأدنى لمفتاح الجهاز */
   KeyCode            max_key_code; /* الحد الأقصى لمفتاح الجهاز */
   XkbControlsPtr     ctrls;        /* ضوابط */
   XkbServerMapPtr    server;       /* خريطة مفاتيح الخادم */
   XkbClientMapPtr    map;          /* خريطة مفاتيح العميل */
   XkbIndicatorPtr    indicators;   /* خريطة المؤشرات */
   XkbNamesPtr        names;        /* أسماء لجميع المكونات */
   XkbCompatMapPtr    compat;       /* خريطة التوافق */
   XkbGeometryPtr     geom;         /* هندسة لوحة المفاتيح المادية */
} XkbDescRec, *XkbDescPtr;

يشير حقل display إلى هيكل عرض X. حقل flags خاص بالمكتبة: قد يؤدي تعديل flags إلى نتائج غير متوقعة. يحدد حقل device_spec معرف الجهاز الخاص بجهاز إدخال لوحة المفاتيح، أو XkbUseCoreKeyboard، الذي يحدد جهاز لوحة المفاتيح الأساسي. يحدد حقلا min_key_code و max_key_code أقل وأكبر رمز مفتاح يمكن إرجاعه بواسطة لوحة المفاتيح.

لكل مكون هيكل بت قناع مطابق يستخدم في استدعاءات الوظائف للإشارة إلى أنه يجب معالجة الهيكل بطريقة ما، مثل تخصيصه أو تحريره. توضح هذه الأقنعة وعلاقاتها بالحقول في XkbDescRec في الجدول 2.

الجدول 2 بتات القناع لـ XkbDescRec
بت القناع حقل XkbDescRec القيمة
XkbControlsMask ctrls (1L<<0)
XkbServerMapMask server (1L<<1)
XkbIClientMapMask map (1L<<2)
XkbIndicatorMapMask indicators (1L<<3)
XkbNamesMask names (1L<<4)
XkbCompatMapMask compat (1L<<5)
XkbGeometryMask geom (1L<<6)
XkbAllComponentsMask جميع الحقول (0x7f)

يحدد هيكل XkbSymInterpretRec تفسير رمز:

    typedef struct {
        KeySym        sym;         /* الرمز محل الاهتمام أو NULL */
        unsigned char flags;       /* XkbSI_AutoRepeat, XkbSI_LockingKey */
        unsigned char match;       /* يحدد كيفية تفسير تعديلات */
        unsigned char mods;        /* بتات التعديل، تتوافق مع ثمانية معدلات حقيقية */
        unsigned char virtual_mod; /* معدل واحد لإضافته إلى خريطة المعدلات الافتراضية للمفتاح */
        XkbAnyAction  act;         /* الإجراء لربطه بموضع الرمز على المفتاح */
    } XkbSymInterpretRec,*XkbSymInterpretPtr;
