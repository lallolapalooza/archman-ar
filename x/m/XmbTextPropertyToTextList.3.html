XmbTextPropertyToTextList(3)

الحزمة: libx11

التراخيص: MIT AND X11

الدوال Xutf8TextListToTextProperty و Xutf8TextPropertyToTextList هي امتدادات قدمها مشروع XFree86، Inc.، في الإصدار 4.0.2 الخاص بهم. يتم الإشارة إلى وجودها بواسطة الماكرو X_HAVE_UTF8_STRING.

الهياكل

يحتوي الهيكل XTextProperty على:

typedef struct {
        unsigned char *value;   /* بيانات الخاصية */
        Atom encoding;  /* نوع الخاصية */
        int format;     /* 8، 16، أو 32 */
        unsigned long nitems;   /* عدد العناصر في value */
} XTextProperty;

يحتوي الهيكل XICCEncodingStyle على:

#define XNoMemory -1
#define XLocaleNotSupported -2
#define XConverterNotFound -3 
typedef enum {
        XStringStyle,           /* STRING */
        XCompoundTextStyle,     /* COMPOUND_TEXT */
        XTextStyle,             /* نص بترميز المالك (النطاق المحلي الحالي) */
        XStdICCTextStyle,       /* STRING، وإلا COMPOUND_TEXT */
        XUTF8StringStyle        /* UTF8_STRING */
} XICCEncodingStyle;
