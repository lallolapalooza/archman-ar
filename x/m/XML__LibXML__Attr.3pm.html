XML__LibXML__Attr(3pm)

الحزمة: perl-xml-libxml

التراخيص: GPL

XML::LibXML::Attr(3) وثائق Perl المساهمة من المستخدم XML::LibXML::Attr(3)

الاسم

XML::LibXML::Attr - فئة سمة XML::LibXML

الملخص

use XML::LibXML;
# فقط الدوال الخاصة بعقد السمات مدرجة هنا،
# راجع صفحة دليل XML::LibXML::Node للدوال الأخرى
$attr = XML::LibXML::Attr->new($name [,$value]);
$string = $attr->getValue();
$string = $attr->value;
$attr->setValue( $string );
$node = $attr->getOwnerElement();
$attr->setNamespace($nsURI, $prefix);
$bool = $attr->isId;
$string = $attr->serializeContent;

الوصف

هذه هي الواجهة للتعامل مع السمات كعقد عادية. يعتمد تسمية الفئة على وثائق W3C DOM.

الدوال

ترث الفئة من XML::LibXML::Node. وثائق الدوال الموروثة غير مدرجة هنا.

العديد من الدوال المدرجة هنا موثقة بشكل مكثف في مواصفات DOM المستوى 3 (http://www.w3.org/TR/DOM-Level-3-Core/). يرجى الرجوع إلى المواصفات للحصول على وثائق شاملة.

new
$attr = XML::LibXML::Attr->new($name [,$value]);

منشئ الفئة. إذا كنت بحاجة إلى العمل مع سلاسل مشفرة ISO، فيجب عليك دائمًا استخدام "createAttribute" من XML::LibXML::Document.

getValue
$string = $attr->getValue();

تُرجع القيمة المخزنة للسمة. إذا تم إرجاع undef، فإن السمة ليس لها قيمة، وهذا يختلف عن كونها "غير محددة".

value
$string = $attr->value;

اسم مستعار لـ getValue()

setValue
$attr->setValue( $string );

هذا ضروري لتعيين قيمة سمة جديدة. إذا تم تمرير سلاسل مشفرة ISO كمعلمة، فيجب ربط العقدة بوثيقة، وإلا فقد يتم التشفير بشكل غير صحيح.

getOwnerElement
$node = $attr->getOwnerElement();

تُرجع العقدة التي تنتمي إليها السمة. إذا لم تكن السمة مرتبطة بعقدة، فسيتم إرجاع undef. الكتابة فوق التنفيذ الأساسي، ستُرجع دالة parentNode القيمة undef، بدلاً من العنصر المالك.

setNamespace
$attr->setNamespace($nsURI, $prefix);

تحاول هذه الدالة ربط السمة بمساحة اسم معينة. إذا كان $nsURI غير محدد أو فارغ، تقوم الدالة بإلغاء أي ارتباط سابق للسمة بمساحة اسم. إذا لم تكن مساحة الاسم معلنة مسبقًا في سياق السمة، فسوف تفشل هذه الدالة. في هذه الحالة، قد ترغب في استدعاء setNamespace() على العنصر المالك. إذا كانت مساحة الاسم غير فارغة ومعلنة في سياق السمة، ولكن مع بادئة مختلفة (غير فارغة) فقط، فلا يزال يتم ربط السمة بمساحة الاسم ولكن تحصل على بادئة مختلفة عن $prefix. تفشل الدالة أيضًا إذا كانت البادئة فارغة ولكن مساحة الاسم غير فارغة (لأن السمات غير المسبوقة يجب بحكم تعريفها ألا تنتمي إلى أي مساحة اسم). تُرجع هذه الدالة 1 عند النجاح، و 0 بخلاف ذلك.

isId
$bool = $attr->isId;

تحديد ما إذا كانت السمة من نوع ID. بالنسبة للوثائق التي تحتوي على DTD، هذه المعلومات متاحة فقط إذا تم طلب تحميل/التحقق من DTD. بالنسبة لوثائق HTML التي تم تحليلها باستخدام محلل HTML، يتم اكتشاف ID تلقائيًا. في وثائق XML، تعتبر جميع سمات "xml:id" من نوع ID.

serializeContent($docencoding)
$string = $attr->serializeContent;

هذه الدالة ليست جزءًا من واجهة برمجة تطبيقات DOM. تُرجع محتوى السمة بالشكل الذي يتم فيه تسلسلها إلى XML، أي مع جميع الأحرف الوصفية المقتبسة بشكل صحيح ومع إشارات الكيان الخام (باستثناء الكيانات التي تم توسيعها أثناء وقت التحليل). يؤدي تعيين العلامة الاختيارية $docencoding إلى 1 إلى فرض ترميز المستند للسلسلة الناتجة (والتي يتم بعد ذلك تمريرها إلى Perl كسلسلة بايت). بخلاف ذلك، يتم تمرير السلسلة إلى Perl كأحرف (مشفرة UTF-8).
