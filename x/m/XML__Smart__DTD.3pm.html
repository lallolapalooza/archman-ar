XML__Smart__DTD(3pm)

الحزمة: perl-xml-smart

التراخيص: GPL, PerlArtistic

XML::Smart::DTD(3) توثيق Perl المساهم به من المستخدم XML::Smart::DTD(3)

الاسم

XML::Smart::DTD - محلل DTD لـ XML::Smart.

الوصف

يقوم هذا بتحليل DTD ويوفر طرقًا للوصول إلى المعلومات المخزنة في DTD.

الاستخدام

use XML::Smart::DTD ;
my $dtd = XML::Smart::DTD->new('some.dtd') ;

if ( $dtd->child_exists('tag1','subtag1') ) {
...
}
use Data::Dumper ;
print Dumper( $dtd->tree ) ;

جديد

الطرق

attr_exists ( TAG , ATTR )

ترجع صحيح إذا كان السمة موجودة في العنصر TAG.

child_exists ( TAG , CHILD )

ترجع صحيح إذا كان الابن موجودًا في العنصر TAG.

elem_exists ( TAG )

ترجع صحيح إذا كان العنصر TAG موجودًا.

خطأ

ترجع قائمة الأخطاء.

get_attr_def ( TAG , ATTR )

ترجع القيمة الافتراضية لسمة.

get_attr_type ( TAG , ATTR )

ترجع نوع السمة.

get_attr_values ( TAG , ATTR )

ترجع القيم المعرفة لسمة.

get_attrs ( TAG )

ترجع قائمة السمات لعنصر.

get_attrs_req ( TAG )

ترجع قائمة السمات المطلوبة لعنصر.

get_childs ( TAG )

ترجع قائمة الأبناء لعنصر.

get_childs_req ( TAG )

ترجع قائمة الأبناء المطلوبة لعنصر.

get_elem_opt ( TAG )

ترجع خيار التكرار لعنصر:

!  مطلوب وتطابق واحد فقط
+  1 أو أكثر
*  0 أو أكثر
?  0 أو 1

get_elem_child_opt ( TAG , CHILD )

مثل get_elem_opt() ولكن هذا العنصر كابن لعنصر.

is_attr_fix ( TAG , ATTR )

ترجع صحيح إذا كانت السمة ثابتة.

is_attr_req ( TAG , ATTR )

ترجع صحيح إذا كانت السمة مطلوبة.

is_elem_any ( TAG )

ترجع صحيح إذا كان العنصر ANY.

is_elem_child_multi ( TAG , CHILD )

ترجع صحيح إذا كان العنصر يمكن أن يكون له تكرارات متعددة كابن لـ TAG.

is_elem_child_opt ( TAG , CHILD )

ترجع صحيح إذا كان العنصر اختياريًا كابن لـ TAG.

is_elem_child_req ( TAG , CHILD )

ترجع صحيح إذا كان العنصر اختياريًا كابن لـ TAG.

is_elem_child_uniq ( TAG , CHILD )

ترجع صحيح إذا كان العنصر مطلوبًا وفريدًا كابن لـ TAG.

is_elem_pcdata ( TAG )

ترجع صحيح إذا كان العنصر PCDATA (له محتوى).

is_elem_empty ( TAG )

ترجع صحيح إذا كان العنصر فارغًا (ليس له سمات أو محتوى أو أبناء).

is_elem_multi ( TAG )

ترجع صحيح إذا كان العنصر يمكن أن يكون له تكرارات متعددة عالميًا.

is_elem_opt ( TAG )

ترجع صحيح إذا كان العنصر اختياريًا عالميًا.

is_elem_parent ( TAG , @PARENTS )

ترجع صحيح إذا كانت قائمة @PARENTS يمكن أن تكون أصلًا للعنصر TAG.

is_elem_req

ترجع صحيح إذا كان العنصر مطلوبًا عالميًا.

is_elem_uniq

ترجع صحيح إذا كان العنصر فريدًا ومطلوبًا عالميًا.
