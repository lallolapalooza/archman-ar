XGetFontProperty(3)

الحزمة: libx11

التراخيص: MIT AND X11

XLoadFont(3) وظائف XLIB XLoadFont(3)

الاسم

XLoadFont, XQueryFont, XLoadQueryFont, XFreeFont, XGetFontProperty, XUnloadFont, XCharStruct, XFontProp, XChar2b, XFontStruct - تحميل أو إلغاء تحميل الخطوط وهياكل مقاييس الخطوط

الصيغة

Font XLoadFont(Display *display, _Xconst char *name);

XFontStruct *XQueryFont(Display *display, XID font_ID);

XFontStruct *XLoadQueryFont(Display *display, _Xconst char *name);

int XFreeFont(Display *display, XFontStruct *font_struct);

Bool XGetFontProperty(XFontStruct *font_struct, Atom atom, unsigned long *value_return);

int XUnloadFont(Display *display, Font font);

المعاملات

atom
يحدد الذرة (atom) لاسم الخاصية التي تريد إرجاعها.
display
يحدد الاتصال بخادم X.
font
يحدد الخط.
font_ID
يحدد معرّف الخط أو معرّف GContext.
font_struct
يحدد مساحة التخزين المرتبطة بالخط.
gc
يحدد GC (سياق الرسم).
name
يحدد اسم الخط، وهو عبارة عن سلسلة نصية منتهية بحرف فارغ (null-terminated).
value_return
يُرجع قيمة خاصية الخط.

الوصف

تقوم الدالة XLoadFont بتحميل الخط المحدد وترجع معرّف الخط المرتبط به. إذا لم يكن اسم الخط بتشفير الأحرف المحمول المضيف (Host Portable Character Encoding)، فإن النتيجة تعتمد على التنفيذ. استخدام الأحرف الكبيرة أو الصغيرة لا يهم. عندما تُستخدم الأحرف "؟" و "*" في اسم الخط، يتم إجراء مطابقة للنمط ويتم استخدام أي خط مطابق. في النمط، سيطابق الحرف "؟" أي حرف واحد، وسيُطابق الحرف "*" أي عدد من الأحرف. يتم تحديد تنسيق منظم لأسماء الخطوط في معيار اتحاد X X Logical Font Description Conventions. إذا فشلت XLoadFont في تحميل الخط المحدد، ينتج خطأ BadName. لا ترتبط الخطوط بشاشة معينة ويمكن تخزينها كمكون لأي GC. عندما لا تعود هناك حاجة للخط، قم باستدعاء XUnloadFont.

يمكن أن تنتج XLoadFont أخطاء BadAlloc و BadName.

تُرجع الدالة XQueryFont مؤشرًا إلى الهيكل XFontStruct، والذي يحتوي على معلومات مرتبطة بالخط. يمكنك الاستعلام عن خط أو الخط المخزن في GC. سيكون معرّف الخط المخزن في الهيكل XFontStruct هو معرّف GContext، ويجب أن تكون حذرًا عند استخدام هذا المعرّف في دوال أخرى (انظر XGContextFromGC). إذا لم يكن الخط موجودًا، تُرجع XQueryFont قيمة NULL. لتحرير هذه البيانات، استخدم XFreeFontInfo.

يمكن أن تنتج XLoadQueryFont خطأ BadAlloc.

توفر الدالة XLoadQueryFont الطريقة الأكثر شيوعًا للوصول إلى خط. تقوم XLoadQueryFont بفتح (تحميل) الخط المحدد وإرجاع مؤشر إلى الهيكل XFontStruct المناسب. إذا لم يكن اسم الخط بتشفير الأحرف المحمول المضيف (Host Portable Character Encoding)، فإن النتيجة تعتمد على التنفيذ. إذا لم يكن الخط موجودًا، تُرجع XLoadQueryFont قيمة NULL.

تقوم الدالة XFreeFont بحذف الارتباط بين معرّف مورد الخط والخط المحدد وتحرير الهيكل XFontStruct. سيتم تحرير الخط نفسه عندما لا تشير أي موارد أخرى إليه. لا ينبغي الإشارة إلى البيانات والخط مرة أخرى.

يمكن أن تنتج XFreeFont خطأ BadFont.

بمعامل الذرة (atom) لتلك الخاصية، تُرجع الدالة XGetFontProperty قيمة خاصية الخط المحددة. تُرجع XGetFontProperty أيضًا False إذا لم تكن الخاصية معرّفة أو True إذا كانت معرّفة. توجد مجموعة من الذرات المعرفة مسبقًا لخصائص الخطوط، والتي يمكن العثور عليها في X11/Xatom.h. تحتوي هذه المجموعة على الخصائص القياسية المرتبطة بالخط. على الرغم من أنه غير مضمون، فمن المحتمل أن تكون خصائص الخطوط المعرفة مسبقًا موجودة.

تقوم الدالة XUnloadFont بحذف الارتباط بين معرّف مورد الخط والخط المحدد. سيتم تحرير الخط نفسه عندما لا تشير أي موارد أخرى إليه. لا ينبغي الإشارة إلى الخط مرة أخرى.

يمكن أن تنتج XUnloadFont خطأ BadFont.

الهياكل

يحتوي الهيكل XFontStruct على جميع المعلومات الخاصة بالخط ويتكون من المعلومات الخاصة بالخط بالإضافة إلى مؤشر إلى مصفوفة من الهياكل XCharStruct للأحرف الموجودة في الخط. تحتوي الهياكل XFontStruct و XFontProp و XCharStruct على:

typedef struct {
	short lbearing;	/* origin to left edge of raster */
	short rbearing;	/* origin to right edge of raster */
	short width;	/* advance to next char's origin */
	short ascent;	/* baseline to top edge of raster */
	short descent;	/* baseline to bottom edge of raster */
	unsigned short attributes;	/* per char flags (not predefined) */
} XCharStruct;
typedef struct {
	Atom	name;
	unsigned long card32;
} XFontProp;
typedef struct {	/* normal 16 bit characters are two bytes */
    unsigned char byte1;
    unsigned char byte2;
} XChar2b;
typedef struct {
	XExtData *ext_data;	/* hook for extension to hang data */
	Font fid;	/* Font id for this font */
	unsigned direction;	/* hint about the direction font is painted */
	unsigned min_char_or_byte2;	/* first character */
	unsigned max_char_or_byte2;	/* last character */
	unsigned min_byte1;	/* first row that exists */
	unsigned max_byte1;	/* last row that exists */
	Bool all_chars_exist;	/* flag if all characters have nonzero size */
	unsigned default_char;	/* char to print for undefined character */
	int n_properties;	/* how many properties there are */
	XFontProp *properties;	/* pointer to array of additional properties */
	XCharStruct min_bounds;	/* minimum bounds over all existing char */
	XCharStruct max_bounds;	/* maximum bounds over all existing char */
	XCharStruct *per_char;	/* first_char to last_char information */
	int ascent;	/* logical extent above baseline for spacing */
	int descent;	/* logical decent below baseline for spacing */
} XFontStruct;

يدعم X عمليات النص أحادية البايت/الحرف، وثنائية البايت/الحرف في مصفوفة، وتشغيل الأحرف 16 بت. لاحظ أنه يمكن استخدام أي من هذه الأشكال مع خط، ولكن طلب نص أحادي البايت/الحرف يمكن أن يحدد بايت واحد فقط (أي، الصف الأول من خط ثنائي البايت). يجب أن تنظر إلى الخطوط ثنائية البايت كمصفوفة ثنائية الأبعاد من الأحرف المعرفة: يحدد البايت الأول نطاق الصفوف المعرفة، ويحدد البايت الثاني نطاق الأعمدة المعرفة للخط. تحتوي الخطوط أحادية البايت/الحرف على صف واحد معرف، ويحدد نطاق البايت الثاني المحدد في الهيكل نطاق الأحرف.

يتم تعريف الصندوق المحيط بالحرف بواسطة XCharStruct الخاص بذلك الحرف. عندما تكون الأحرف غائبة عن الخط، يتم استخدام default_char. عندما تكون الخطوط تحتوي على جميع الأحرف بنفس الحجم، يتم استخدام المعلومات الموجودة في حدود الحد الأدنى والحد الأقصى لـ XFontStruct فقط.

الأعضاء في XFontStruct لها الدلالات التالية:

  • يمكن أن يكون العضو direction إما FontLeftToRight أو FontRightToLeft. إنه مجرد تلميح حول ما إذا كانت معظم عناصر XCharStruct لديها مقياس عرض حرف موجب (FontLeftToRight) أو سالب (FontRightToLeft). البروتوكول الأساسي لا يُعرّف دعمًا للنص الرأسي.
  • إذا كان العضوان min_byte1 و max_byte1 كلاهما صفرًا، فإن min_char_or_byte2 يحدد فهرس الحرف الخطي المقابل للعنصر الأول في مصفوفة per_char، و max_char_or_byte2 يحدد فهرس الحرف الخطي للعنصر الأخير.
إذا كان أي من min_byte1 أو max_byte1 غير صفري، فإن كلا من min_char_or_byte2 و max_char_or_byte2 أقل من 256، وقيم فهرس الأحرف ثنائية البايت المقابلة لعنصر مصفوفة per_char N (عد من 0) هي:
byte1 = N/D + min_byte1
byte2 = N\D + min_char_or_byte2
حيث: 
        D = max_char_or_byte2 - min_char_or_byte2 + 1
        / = قسمة صحيحة
        \
