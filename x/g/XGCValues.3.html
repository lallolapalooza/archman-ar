XGCValues(3)

الحزمة: libx11

التراخيص: MIT AND X11

XCreateGC(3) وظائف XLIB XCreateGC(3)

الاسم

XCreateGC, XCopyGC, XChangeGC, XGetGCValues, XFreeGC, XGContextFromGC, XGCValues - إنشاء أو تحرير سياقات الرسوم أو هيكل سياق الرسم

الصيغة

GC XCreateGC(Display *display, Drawable d, unsigned long valuemask, XGCValues *values);

int XCopyGC(Display *display, GC src, unsigned long valuemask, GC dest);

int XChangeGC(Display *display, GC gc, unsigned long valuemask, XGCValues *values);

Status XGetGCValues(Display *display, GC gc, unsigned long valuemask, XGCValues *values_return);

int XFreeGC(Display *display, GC gc);

GContext XGContextFromGC(GC gc);

المعاملات

d
يحدد الرسم (drawable).
dest
يحدد سياق الرسم الوجهة.
display
يحدد الاتصال بخادم X.
gc
يحدد سياق الرسم.
src
يحدد مكونات سياق الرسم المصدر.
valuemask
يحدد أي المكونات في سياق الرسم سيتم تعيينها، نسخها، تغييرها، أو إرجاعها. هذه المعامل هو OR المنطقي الشامل (bitwise inclusive OR) لصفر أو أكثر من بتات قناع مكونات سياق الرسم الصالحة.
values
يحدد أي قيم كما هو محدد بواسطة valuemask.
values_return
تُرجع قيم سياق الرسم في هيكل XGCValues المحدد.

الوصف

تقوم الدالة XCreateGC بإنشاء سياق رسم وتعيد GC. يمكن استخدام GC مع أي رسم وجهة له نفس الجذر والعمق مثل الرسم المحدد. استخدامها مع رسومات أخرى يؤدي إلى خطأ BadMatch.

يمكن لـ XCreateGC إنشاء أخطاء BadAlloc، BadDrawable، BadFont، BadMatch، BadPixmap، و BadValue.

تقوم الدالة XCopyGC بنسخ المكونات المحددة من سياق الرسم المصدر إلى سياق الرسم الوجهة. يجب أن يكون لـ GC المصدر والوجهة نفس الجذر والعمق، وإلا يحدث خطأ BadMatch. يحدد valuemask المكون الذي سيتم نسخه، كما هو الحال في XCreateGC.

يمكن لـ XCopyGC إنشاء أخطاء BadAlloc، BadGC، و BadMatch.

تقوم الدالة XChangeGC بتغيير المكونات المحددة بواسطة valuemask لسياق الرسم المحدد. يحتوي معامل القيم على القيم التي سيتم تعيينها. القيم والقيود هي نفسها كما في XCreateGC. تغيير قناع القطع (clip-mask) يتجاوز أي طلب سابق لـ XSetClipRectangles على السياق. تغيير إزاحة الشرطة (dash-offset) أو قائمة الشرطات (dash-list) يتجاوز أي طلب سابق لـ XSetDashes على السياق. ترتيب التحقق من المكونات وتعديلها يعتمد على الخادم. إذا تم إنشاء خطأ، فقد تم تعديل مجموعة فرعية من المكونات.

يمكن لـ XChangeGC إنشاء أخطاء BadAlloc، BadFont، BadGC، BadMatch، BadPixmap، و BadValue.

تقوم الدالة XGetGCValues بإرجاع المكونات المحددة بواسطة valuemask لسياق الرسم المحدد. إذا كان valuemask يحتوي على مجموعة صالحة من بتات قناع GC (GCFunction، GCPlaneMask، GCForeground، GCBackground، GCLineWidth، GCLineStyle، GCCapStyle، GCJoinStyle، GCFillStyle، GCFillRule، GCTile، GCStipple، GCTileStipXOrigin، GCTileStipYOrigin، GCFont، GCSubwindowMode، GCGraphicsExposures، GCClipXOrigin، GCClipYOrigin، GCDashOffset، أو GCArcMode) ولم يحدث أي خطأ، تقوم XGetGCValues بتعيين المكونات المطلوبة في values_return وتعيد حالة غير صفرية. بخلاف ذلك، فإنها تعيد حالة صفرية. لاحظ أنه لا يمكن طلب قناع القطع وقائمة الشرطات (الممثلة بـ GCClipMask و GCDashList، على التوالي، في valuemask). لاحظ أيضًا أنه سيتم إرجاع معرّف مورد غير صالح (مع واحد أو أكثر من أهم ثلاثة بتات مضبوطة على 1) لـ GCFont، GCTile، و GCStipple إذا لم يتم تعيين المكون بشكل صريح من قبل العميل.

تقوم الدالة XFreeGC بتدمير سياق الرسم المحدد بالإضافة إلى جميع التخزين المرتبط به.

يمكن لـ XFreeGC إنشاء خطأ BadGC.

الهياكل

يحتوي هيكل XGCValues على:

/* بتات قناع سمات GC */

#define GCFunction (1L<<0)
#define GCPlaneMask (1L<<1)
#define GCForeground (1L<<2)
#define GCBackground (1L<<3)
#define GCLineWidth (1L<<4)
#define GCLineStyle (1L<<5)
#define GCCapStyle (1L<<6)
#define GCJoinStyle (1L<<7)
#define GCFillStyle (1L<<8)
#define GCFillRule (1L<<9)
#define GCTile (1L<<10)
#define GCStipple (1L<<11)
#define GCTileStipXOrigin (1L<<12)
#define GCTileStipYOrigin (1L<<13)
#define GCFont (1L<<14)
#define GCSubwindowMode (1L<<15)
#define GCGraphicsExposures (1L<<16)
#define GCClipXOrigin (1L<<17)
#define GCClipYOrigin (1L<<18)
#define GCClipMask (1L<<19)
#define GCDashOffset (1L<<20)
#define GCDashList (1L<<21)
#define GCArcMode (1L<<22) 
/* القيم */
typedef struct {
        int function;   /* العملية المنطقية */
        unsigned long plane_mask;       /* قناع المستوى */
        unsigned long foreground;       /* بكسل المقدمة */
        unsigned long background;       /* بكسل الخلفية */
        int line_width; /* عرض الخط (بالبكسل) */
        int line_style; /* LineSolid, LineOnOffDash, LineDoubleDash */
        int cap_style;  /* CapNotLast, CapButt, CapRound, CapProjecting */
        int join_style; /* JoinMiter, JoinRound, JoinBevel */
        int fill_style; /* FillSolid, FillTiled, FillStippled FillOpaqueStippled*/
        int fill_rule;  /* EvenOddRule, WindingRule */
        int arc_mode;   /* ArcChord, ArcPieSlice */
        Pixmap tile;    /* بكسل القرميد لعمليات التبليط */
        Pixmap stipple; /* بكسل القرميد ذو المستوى 1 للتبقيع */
        int ts_x_origin;        /* إزاحة لعمليات القرميد أو التبقيع */
        int ts_y_origin;
        Font font;      /* الخط الافتراضي للنص لعمليات النص */
        int subwindow_mode;     /* ClipByChildren, IncludeInferiors */
        Bool graphics_exposures;        /* قيمة منطقية، هل يجب إنشاء التعرضات */
        int clip_x_origin;      /* أصل القطع */
        int clip_y_origin;
        Pixmap clip_mask;       /* قطع Bitmap؛ استدعاءات أخرى للمستطيلات */
        int dash_offset;        /* معلومات الخط المخطط/المتقطع */
        char dashes;
} XGCValues;

تُستخدم سمات الوظيفة لسياق الرسم عند تحديث قسم من الرسم (الوجهة) ببتات من مكان آخر (المصدر). تحدد الوظيفة في GC كيفية حساب بتات الوجهة الجديدة من بتات المصدر وبتات الوجهة القديمة. GXcopy عادة ما تكون الأكثر فائدة لأنها ستعمل على شاشة ملونة، ولكن التطبيقات الخاصة قد تستخدم وظائف أخرى، خاصة بالتزامن مع مستويات معينة من شاشة ملونة. وظائف GC الـ 16

عند رسم خطوط بنمط الخط LineDoubleDash، يتم التحكم في الشرطات الفردية بواسطة نمط الملء بالطريقة التالية:

FillSolid الخلفية
FillTiled نفس الشيء كما هو الحال بالنسبة للشرطات الزوجية
FillOpaqueStippled نفس الشيء كما هو الحال بالنسبة للشرطات الزوجية
FillStippled الخلفية مقنعة بواسطة النقاط

قد يؤدي تخزين بيكسماب في GC إلى إنشاء نسخة أو قد لا يؤدي إلى ذلك. إذا تم استخدام البيكسما لاحقًا كوجهة لطلب رسومات، فقد ينعكس التغيير أو قد لا ينعكس في GC. إذا تم استخدام البيكسما بشكل متزامن في طلب رسومات كوجهة وكبلاط أو نقط، فإن النتائج غير محددة.

للحصول على أفضل أداء، يجب عليك الرسم قدر الإمكان باستخدام نفس GC (دون تغيير مكوناته). تعتمد تكاليف تغيير مكونات GC مقارنة باستخدام GCs مختلفة على أجهزة العرض وتنفيذ الخادم. من المحتمل جدًا أن يتم تخزين بعض معلومات GC مؤقتًا في أجهزة العرض وأن مثل هذه الأجهزة يمكنها تخزين عدد صغير فقط من GCs مؤقتًا.

القيمة dashs هي في الواقع شكل مبسط للأنماط الأكثر عمومية التي يمكن تعيينها باستخدام XSetDashes. تحديد قيمة N يعادل تحديد القائمة المكونة من عنصرين [N، N] في XSetDashes. يجب أن تكون القيمة غير صفرية، وإلا ستحدث خطأ BadValue.

تقوم قناع الاقتصاص بتقييد الكتابة إلى رسم الوجهة. إذا تم تعيين قناع الاقتصاص إلى بيكسماب، فيجب أن يكون له عمق واحد وله نفس الجذر مثل GC، وإلا ستحدث خطأ BadMatch. إذا تم تعيين قناع الاقتصاص إلى None، يتم دائمًا رسم البكسلات بغض النظر عن أصل الاقتصاص. يمكن أيضًا تعيين قناع الاقتصاص عن طريق استدعاء الوظائف XSetClipRectangles أو XSetRegion. يتم رسم البكسلات فقط حيث يحتوي قناع الاقتصاص على بت تم تعيينه إلى 1. لا يتم رسم البكسلات خارج المنطقة التي يغطيها قناع الاقتصاص أو حيث يحتوي قناع الاقتصاص على بت تم تعيينه إلى 0. يؤثر قناع الاقتصاص على جميع طلبات الرسومات. قناع الاقتصاص لا يقص المصادر. يتم تفسير أصل قناع الاقتصاص بالنسبة إلى أصل أي رسم وجهة محدد في طلب رسومات.

يمكنك تعيين وضع النافذة الفرعية إلى ClipByChildren أو IncludeInferiors. بالنسبة لـ ClipByChildren، يتم أيضًا قص كل من نوافذ المصدر والوجهة بواسطة جميع الأطفال المرئيين من نوع InputOutput. بالنسبة لـ IncludeInferiors، لا يتم قص أي نافذة مصدر أو وجهة بواسطة العناصر السفلية. سينتج عن ذلك تضمين محتويات النافذة الفرعية في المصدر والرسم عبر حدود النوافذ الفرعية للوجهة. استخدام IncludeInferiors على نافذة بعمق واحد مع عناصر سفلية معروضة بعمق مختلف ليس غير قانوني، ولكن البروتوكول الأساسي لا يحدد المعاني.

يحدد قاعدة الملء ما هي البكسلات الموجودة بالداخل (مرسومة) للمسارات المعطاة في طلبات XFillPolygon ويمكن تعيينها إلى EvenOddRule أو WindingRule. بالنسبة لـ EvenOddRule، تكون النقطة بالداخل إذا قام شعاع لا نهائي مع النقطة كأصل بتقاطع المسار عددًا فرديًا من المرات. بالنسبة لـ WindingRule، تكون النقطة بالداخل إذا قام شعاع لا نهائي مع النقطة كأصل بتقاطع عدد غير متساوٍ من أجزاء المسار الموجهة باتجاه عقارب الساعة وعكس اتجاه عقارب الساعة. جزء المسار الموجه باتجاه عقارب الساعة هو جزء يتقاطع مع الشعاع من اليسار إلى اليمين كما لوحظ من النقطة. الجزء المعاكس لاتجاه عقارب الساعة هو جزء يتقاطع مع الشعاع من اليمين إلى اليسار كما لوحظ من النقطة. الحالة التي يكون فيها جزء خط موجه متطابقًا مع الشعاع غير مهمة لأنه يمكنك ببساطة اختيار شعاع مختلف غير متطابق مع جزء.

لكل من EvenOddRule و WindingRule، تكون النقطة صغيرة بلا حدود، والمسار خط رفيع بلا حدود. البكسل يكون بالداخل إذا كان مركز البكسل بالداخل ولم يكن مركز البكسل على الحدود. إذا كان مركز البكسل على الحدود، يكون البكسل بالداخل إذا وفقط إذا كان داخل المضلع على يمينه مباشرة (اتجاه زيادة x). البكسلات ذات المراكز على حافة أفقية هي حالة خاصة وتكون بالداخل إذا وفقط إذا كان داخل المضلع أسفلها مباشرة (اتجاه زيادة y).

يتحكم وضع القوس في الملء في وظيفة XFillArcs ويمكن تعيينه إلى ArcPieSlice أو ArcChord. بالنسبة لـ ArcPieSlice، يتم ملء الأقواس كشريحة فطيرة. بالنسبة لـ ArcChord، يتم ملء الأقواس بالوتر.

يتحكم علم تعرض الرسومات في إنشاء أحداث GraphicsExpose لطلبات XCopyArea و XCopyPlane (وأي طلبات مماثلة محددة بواسطة ملحقات).

التشخيصات

BadAlloc
فشل الخادم في تخصيص المورد المطلوب أو ذاكرة الخادم.
BadDrawable
قيمة وسيطة قابلة للرسم لا تسمي نافذة أو بيكسماب محددة.
BadFont
قيمة وسيطة للخط أو سياق الرسومات لا تسمي خطًا محددًا.
BadGC
قيمة وسيطة لسياق الرسومات لا تسمي سياق رسومات محددًا.
BadMatch
يتم استخدام نافذة InputOnly كعنصر قابل للرسم.
BadMatch
بعض الوسائط أو زوج من الوسائط لهما النوع الصحيح والنطاق ولكن يفشلان في المطابقة بطريقة أخرى مطلوبة بواسطة الطلب.
BadPixmap
قيمة وسيطة بيكسماب لا تسمي بيكسماب محددًا.
BadValue
تقع بعض القيم الرقمية خارج نطاق القيم التي يقبلها الطلب. ما لم يتم تحديد نطاق معين لوسيطة، يتم قبول النطاق الكامل الذي يحدده نوع الوسيطة. يمكن لأي وسيطة معرفة كمجموعة من البدائل إنشاء هذا الخطأ.
