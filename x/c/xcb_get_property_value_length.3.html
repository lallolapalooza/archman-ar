xcb_get_property_value_length(3)

الحزمة: libxcb

التراخيص: X11

xcb_get_property(3) طلبات XCB xcb_get_property(3)

الاسم

xcb_get_property - الحصول على خاصية نافذة

المُلَخَّص

#include <xcb/xproto.h>

الوظيفة المطلوبة

xcb_get_property_cookie_t xcb_get_property(xcb_connection_t *conn, uint8_t _delete, xcb_window_t window, xcb_atom_t property, xcb_atom_t type, uint32_t long_offset, uint32_t long_length);

بنية بيانات الرد

typedef struct xcb_get_property_reply_t {
    uint8_t    response_type;
    uint8_t    format;
    uint16_t   sequence;
    uint32_t   length;
    xcb_atom_t type;
    uint32_t   bytes_after;
    uint32_t   value_len;
    uint8_t    pad0[12];
} xcb_get_property_reply_t;

وظيفة الرد

xcb_get_property_reply_t *xcb_get_property_reply(xcb_connection_t *conn, xcb_get_property_cookie_t cookie, xcb_generic_error_t **e);

ملحقات الرد

void *xcb_get_property_value(const xcb_get_property_request_t *reply);

xcb_generic_iterator_t xcb_get_property_value_end(const xcb_get_property_reply_t *reply);

وسائط الطلب

conn
اتصال XCB بـ X11.
_delete
ما إذا كان يجب حذف الخاصية بالفعل. لحذف خاصية، يجب أن يتطابق type المحدد مع نوع الخاصية الفعلي.
window
النافذة التي تريد الحصول على خاصيتها.
property
الخاصية التي تريد الحصول عليها (ذرة).
type
نوع الخاصية التي تريد الحصول عليها (ذرة).
long_offset
يحدد الإزاحة (بمضاعفات 32 بت) في الخاصية المحددة حيث يجب استرداد البيانات.
long_length
يحدد عدد مضاعفات البيانات بحجم 32 بت التي يجب استردادها (على سبيل المثال، إذا قمت بتعيين long_length إلى 4، فستتلقى 16 بايت من البيانات).

حقول الرد

response_type
نوع هذا الرد، وفي هذه الحالة XCB_GET_PROPERTY. يوجد هذا الحقل أيضًا في xcb_generic_reply_t ويمكن استخدامه للتمييز بين الردود.
sequence
رقم التسلسل لآخر طلب تم معالجته بواسطة خادم X11.
length
طول الرد، بالكلمات (الكلمة هي 4 بايت).
format
يحدد ما إذا كان يجب عرض البيانات كقائمة كميات بحجم 8 بت أو 16 بت أو 32 بت. القيم الممكنة هي 8 و 16 و 32. تتيح هذه المعلومات لخادم X إجراء عمليات تبديل البايت بشكل صحيح حسب الضرورة.
type
النوع الفعلي للخاصية (ذرة).
bytes_after
عدد البايتات المتبقية للقراءة في الخاصية إذا تم إجراء قراءة جزئية.
value_len
طول القيمة. يجب عليك استخدام الملحق المقابل بدلاً من هذا الحقل.

الوصف

الحصول على property المحدد من window المحدد. الخصائص هي على سبيل المثال عنوان النافذة (WM_NAME) أو حجمها الأدنى (WM_NORMAL_HINTS). تستخدم بروتوكولات مثل EWMH أيضًا الخصائص - على سبيل المثال، يحدد EWMH عنوان النافذة، المشفر كسلسلة UTF-8، في الخاصية _NET_WM_NAME.

القيمة المرتجعة

يعيد xcb_get_property_cookie_t. يجب معالجة الأخطاء عند استدعاء وظيفة الرد xcb_get_property_reply.

إذا كنت ترغب في معالجة الأخطاء في حلقة الأحداث بدلاً من ذلك، استخدم xcb_get_property_unchecked. انظر xcb-requests(3) للتفاصيل.

الأخطاء

xcb_atom_error_t
property أو type لا تشير إلى ذرة صالحة.
xcb_value_error_t
long_offset المحدد يتجاوز طول الخاصية الفعلي (على سبيل المثال، الخاصية لها طول 3 بايت وتقوم بتعيين long_offset إلى 1، مما يؤدي إلى إزاحة بايت تبلغ 4).
xcb_window_error_t
النافذة المحددة window غير موجودة.

مثال

/*
 * يطبع خاصية WM_NAME للنافذة.
 *
 */
void my_example(xcb_connection_t *c, xcb_window_t window) {
    xcb_get_property_cookie_t cookie;
    xcb_get_property_reply_t *reply;
    /* هذه الذرات محددة مسبقًا في بروتوكول X11. */
    xcb_atom_t property = XCB_ATOM_WM_NAME;
    xcb_atom_t type = XCB_ATOM_STRING;
    // TODO: long_length معقول لـ WM_NAME؟
    cookie = xcb_get_property(c, 0, window, property, type, 0, 0);
    if ((reply = xcb_get_property_reply(c, cookie, NULL))) {
        int len = xcb_get_property_value_length(reply);
        if (len == 0) {
            printf("TODO\n");
            free(reply);
            return;
        }
        printf("WM_NAME هو %.*s\n", len,
               (char*)xcb_get_property_value(reply));
    }
    free(reply);
}
