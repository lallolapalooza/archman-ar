xcb_send_event_checked(3)

الحزمة: libxcb

التراخيص: X11

xcb_send_event(3) طلبات XCB xcb_send_event(3)

الاسم

xcb_send_event - إرسال حدث

الملخص

#include <xcb/xproto.h>

وظيفة الطلب

xcb_void_cookie_t xcb_send_event(xcb_connection_t *conn, uint8_t propagate, xcb_window_t destination, uint32_t event_mask, const char *event);

وسائط الطلب

conn
اتصال XCB بـ X11.
propagate
إذا كانت propagate صحيحة ولم تقم أي عملاء بتحديد أي حدث على destination، فسيتم استبدال الوجهة بأقرب سلف لـ destination الذي حدد عليه بعض العملاء نوعًا ما في event_mask والذي لا توجد نافذة فاصلة له في قناع عدم التمرير الخاص به. إذا لم توجد مثل هذه النافذة أو إذا كانت النافذة سلفًا لنافذة التركيز وتم تحديد InputFocus في الأصل كوجهة، فلن يتم إرسال الحدث إلى أي عملاء. بخلاف ذلك، يتم الإبلاغ عن الحدث إلى كل عميل يحدد على الوجهة النهائية أي من الأنواع المحددة في event_mask.
destination
النافذة التي سيرسل إليها هذا الحدث. سيحصل كل عميل يحدد أي حدث ضمن event_mask على النافذة على الحدث.

تشير القيمة الخاصة XCB_SEND_EVENT_DEST_POINTER_WINDOW إلى النافذة التي تحتوي على مؤشر الفأرة.

تشير القيمة الخاصة XCB_SEND_EVENT_DEST_ITEM_FOCUS إلى النافذة التي لديها تركيز لوحة المفاتيح.

event_mask
قناع الحدث لتحديد العملاء الذين يجب أن يتلقوا الحدث المحدد. انظر destination و propagate.
event
الحدث المراد إرساله إلى destination المحدد.

الوصف

يحدد النافذة destination، ويحدد العملاء الذين يجب أن يتلقوا الحدث المحدد، ويتجاهل أي عمليات استيلاء نشطة.

يجب أن يكون event أحد الأحداث الأساسية أو حدثًا معرفًا بواسطة ملحق، بحيث يمكن لخادم X تبديل البايت بشكل صحيح للمحتويات حسب الحاجة. تظل محتويات event دون تغيير وفحصها باستثناء حقل send_event الذي يتم إجباره على 'true'.

قيمة الإرجاع

تُرجع xcb_void_cookie_t. يجب معالجة الأخطاء (إن وجدت) في حلقة الأحداث.

إذا كنت ترغب في معالجة الأخطاء مباشرة باستخدام xcb_request_check بدلاً من ذلك، فاستخدم xcb_send_event_checked. انظر xcb-requests(3) للحصول على التفاصيل.

الأخطاء

xcb_value_error_t
الحدث event المحدد ليس حدثًا أساسيًا ولا حدثًا معرفًا بواسطة ملحق.
xcb_window_error_t
النافذة destination المحددة غير موجودة.

مثال

/*
 * أخبر النافذة المعطاة بأنها تم تكوينها بحجم 800x600 بكسل.
 *
 */
void my_example(xcb_connection_t *conn, xcb_window_t window) {
    /* كل حدث X11 طوله 32 بايت. لذلك، سيقوم XCB بنسخ 32 بايت.
     * من أجل تهيئة هذه البايتات بشكل صحيح، نقوم بتخصيص 32 بايت حتى
     * لو احتجنا أقل من ذلك لـ xcb_configure_notify_event_t */
    xcb_configure_notify_event_t *event = calloc(32, 1);
    event->event = window;
    event->window = window;
    event->response_type = XCB_CONFIGURE_NOTIFY;
    event->x = 0;
    event->y = 0;
    event->width = 800;
    event->height = 600;
    event->border_width = 0;
    event->above_sibling = XCB_NONE;
    event->override_redirect = false;
    xcb_send_event(conn, false, window, XCB_EVENT_MASK_STRUCTURE_NOTIFY,
                   (char*)event);
    xcb_flush(conn);
    free(event);
}
