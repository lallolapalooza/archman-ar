xcb_change_property_checked(3)

الحزمة: libxcb

التراخيص: X11

xcb_change_property(3) طلبات XCB xcb_change_property(3)

الاسم

xcb_change_property - تغيير خاصية النافذة

الملخص

#include <xcb/xproto.h>

دالة الطلب

xcb_void_cookie_t xcb_change_property(xcb_connection_t *conn, uint8_t mode, xcb_window_t window, xcb_atom_t property, xcb_atom_t type, uint8_t format, uint32_t data_len, const void *data);

وسيطات الطلب

conn
اتصال XCB بـ X11.
mode
إحدى القيم التالية:
XCB_PROP_MODE_REPLACE
تجاهل قيمة الخاصية السابقة وتخزين البيانات الجديدة.
XCB_PROP_MODE_PREPEND
إدراج البيانات الجديدة قبل بداية البيانات الموجودة. يجب أن تتطابق format مع قيمة الخاصية الموجودة. إذا كانت الخاصية غير معرفة، يتم معاملتها على أنها معرفة بالنوع والتنسيق الصحيحين مع بيانات بطول صفر.
XCB_PROP_MODE_APPEND
إدراج البيانات الجديدة بعد بداية البيانات الموجودة. يجب أن تتطابق format مع قيمة الخاصية الموجودة. إذا كانت الخاصية غير معرفة، يتم معاملتها على أنها معرفة بالنوع والتنسيق الصحيحين مع بيانات بطول صفر.
window
النافذة التي تريد تغيير خاصيتها.
property
الخاصية التي تريد تغييرها (ذرة).
type
نوع الخاصية التي تريد تغييرها (ذرة).
format
يحدد ما إذا كان يجب عرض البيانات كقائمة من الكميات 8 بت أو 16 بت أو 32 بت. القيم الممكنة هي 8، 16، و 32. تسمح هذه المعلومات لخادم X بإجراء عمليات تبديل البايت بشكل صحيح حسب الحاجة.
data_len
يحدد عدد العناصر (انظر format).
data
بيانات الخاصية.

الوصف

يضبط أو يحدث خاصية على window المحدد. الخصائص هي على سبيل المثال عنوان النافذة (WM_NAME) أو حجمها الأدنى (WM_NORMAL_HINTS). تستخدم البروتوكولات مثل EWMH الخصائص أيضًا - على سبيل المثال، يحدد EWMH عنوان النافذة، مشفر كسلسلة نصية UTF-8، في خاصية _NET_WM_NAME.

قيمة الإرجاع

يعيد xcb_void_cookie_t. يجب التعامل مع الأخطاء (إن وجدت) في حلقة الأحداث.

إذا كنت تريد التعامل مع الأخطاء مباشرة باستخدام xcb_request_check بدلاً من ذلك، فاستخدم xcb_change_property_checked. انظر xcb-requests(3) للتفاصيل.

xcb_alloc_error_t

xcb_match_error_t

مثال

/*
 * يضبط خاصية WM_NAME للنافذة على "XCB Example".
 *
 */
void my_example(xcb_connection_t *conn, xcb_window_t window) {
    xcb_change_property(conn,
        XCB_PROP_MODE_REPLACE,
        window,
        XCB_ATOM_WM_NAME,
        XCB_ATOM_STRING,
        8,
        strlen("XCB Example"),
        "XCB Example");
    xcb_flush(conn);
}
