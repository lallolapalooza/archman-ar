NAME

Xcursor - Cursor loading library

SYNOPSIS

#include <X11/Xcursor/Xcursor.h>

DESCRIPTION

The Xcursor library loads Xcursor images from image files, converting them to the X11 Cursor data type.

Cursor Loading APIs

XcursorCursors * XcursorCursorsCreate (
	Display 	*dpy,
	const char 	*name)

Creates a cursor object from the given cursor name. The name may be one of the standard X cursor names, or it may be a name of a cursor in a theme.

XcursorCursors * XcursorCursorsCreateFont (
	Display 	*dpy,
	unsigned int 	 font)

Creates a cursor object from the given X font cursor.

XcursorCursors * XcursorCursorsCreateGlyph (
	Display 	*dpy,
	Font 	 font,
	unsigned int 	 glyph)

Creates a cursor object from the given X glyph cursor.

void XcursorCursorsDestroy (
	XcursorCursors 	*cursors)

Destroys the given cursor object, freeing all associated resources.

Cursor XcursorCursorsLoadCursor (
	XcursorCursors 	*cursors)

Loads the cursor object into the X server, returning a valid X11 Cursor identifier.

Cursor XcursorCursorsLoadHotSpotCursor (
	XcursorCursors 	*cursors,
	unsigned int 	*hotX,
	unsigned int 	*hotY)

Loads the cursor object into the X server, returning a valid X11 Cursor identifier. The hot-spot coordinates are returned in hotX and hotY.

XcursorImages * XcursorImagesCreate (
	int 	 width,
	int 	 height)

Creates an empty cursor image structure, which can be filled in later.

void XcursorImagesDestroy (
	XcursorImages 	*images)

Destroys the given cursor image structure, freeing all associated resources.

int XcursorImagesLoadFile (
	XcursorImages 	*images,
	const char 	*file)

Loads cursor images from the specified file.

int XcursorImagesLoad (
	XcursorImages 	*images,
	FILE 	*file)

Loads cursor images from the specified file stream.

int XcursorImagesGetName (
	XcursorImages 	*images,
	const char 	*name)

Loads cursor images named name from the theme.

int XcursorImagesGet (
	XcursorImages 	*images,
	const char 	*name)

Loads the cursor named name, if it exists.

int XcursorImagesLoadCursor (
	XcursorImages 	*images,
	Cursor 	 cursor)

Loads the given cursor into the images structure.

XcursorImage * XcursorImagesTake (
	XcursorImages 	*images,
	int 	 index)

Removes the image at the given index from the images structure and returns it. The caller is responsible for freeing the returned XcursorImage.

XcursorBool XcursorImagesSet (
	XcursorImages 	*images,
	int 	 index,
	XcursorImage 	*image)

Sets the image at the given index. The caller is responsible for freeing the old image if it exists.

XcursorImage * XcursorImageCreate (
	int 	 width,
	int 	 height)

Creates a cursor image structure.

void XcursorImageDestroy (
	XcursorImage 	*image)

Destroys the given cursor image structure.

void XcursorImageHash (
	XcursorImage 	*image,
	unsigned char 	 hash[XCURSOR_BITMAP_HASH_SIZE])

Computes a hash value for the given cursor image. This hash value is used to identify cursors when the XCURSOR_DISCOVER environment variable is set.

unsigned int XcursorImageGetX (
	XcursorImage 	*image)

Returns the x-coordinate of the hot spot of the cursor image.

unsigned int XcursorImageGetY (
	XcursorImage 	*image)

Returns the y-coordinate of the hot spot of the cursor image.

void XcursorImageSetX (
	XcursorImage 	*image,
	unsigned int 	 x)

Sets the x-coordinate of the hot spot of the cursor image.

void XcursorImageSetY (
	XcursorImage 	*image,
	unsigned int 	 y)

Sets the y-coordinate of the hot spot of the cursor image.

void XcursorImageSetName (
	XcursorImage 	*image,
	const char 	*name)

Sets the name of the cursor image.

const char * XcursorImageGetName (
	XcursorImage 	*image)

Returns the name of the cursor image.

void XcursorImageSetComment (
	XcursorImage 	*image,
	const char 	*comment)

Sets a comment for the cursor image.

const char * XcursorImageGetComment (
	XcursorImage 	*image)

Returns the comment for the cursor image.

unsigned char * XcursorImageGetPixel (
	XcursorImage 	*image,
	int 	 x,
	int 	 y)

Returns a pointer to the pixel data at the given coordinates. The pixel data is in RGBA format.

XcursorBool XcursorImageSetPixel (
	XcursorImage 	*image,
	int 	 x,
	int 	 y,
	unsigned char 	 r,
	unsigned char 	 g,
	unsigned char 	 b,
	unsigned char 	 a)

Sets the pixel data at the given coordinates.

XcursorBool XcursorImageLoadFile (
	XcursorImage 	*image,
	const char 	*file)

Loads the image data from the specified file into the given image structure.

int XcursorGetDefaultSize (
	Display 	*dpy)

Returns the default cursor size for the specified display.

XcursorBool XcursorSetDefaultSize (
	Display 	*dpy,
	int 	 size)

Sets the default cursor size for the specified display.

const char * XcursorGetTheme (
	Display 	*dpy)

Returns the name of the default cursor theme for the specified display.

XcursorBool XcursorSetTheme (
	Display 	*dpy,
	const char 	*theme)

Sets the default cursor theme for the specified display.

XcursorBool XcursorGetResizable (
	Display 	*dpy)

Returns whether the specified display supports resizable cursors.

XcursorBool XcursorSetResizable (
	Display 	*dpy,
	XcursorBool 	 resizable)

Sets whether the specified display supports resizable cursors.

XcursorBool XcursorSupportsARGB (
	Display 	*dpy)

Returns whether the specified display supports ARGB cursors.

XcursorBool XcursorSupportsAnim (
	Display 	*dpy)

Returns whether the specified display supports animated cursors.

void XcursorShapeLoadImages (
	Display 	*dpy,
	XcursorShape 	 shape,
	XcursorImages 	**images_return)

Loads the images for the given cursor shape. The images are returned in the images_return argument.

Cursor XcursorShapeLoadCursor (
	Display 	*dpy,
	XcursorShape 	 shape)

Loads the cursor for the given shape and returns it as an X11 Cursor.

void XcursorCommentsCreate (
	XcursorComments 	**comments_return)

Creates an empty XcursorComments structure.

void XcursorCommentsAdd (
	XcursorComments 	*comments,
	const char 	*comment)

Adds a comment to the given XcursorComments structure. The pointer

XcursorComment * XcursorCommentsGet (
	XcursorComments 	*comments,
	int 	 index)

Returns the comment at the given index. This function will return a pointer to a comment structure. If the index is out of bounds, NULL will be returned. The caller MUST NOT free this pointer; it is owned by the comments structure. The pointer is only valid until the comments structure is destroyed or modified. The caller is responsible for counting how many pointers will be in the index.

void XcursorCommentsDestroy (
	XcursorComments 	*comments)

Deallocates the given XcursorComments structure as well as the XcursorComment structures which it points to.

Animated Cursors

XcursorAnimate * XcursorAnimateCreate (
	XcursorCursors 	*cursors)

Wrap the given array of cursors in a newly allocated XcursorAnimate structure, which adds a sequence number used in XcursorAnimateNext.

void XcursorAnimateDestroy (
	XcursorAnimate 	*animate)

Discards the given animate data, freeing both the XcursorCursors array of cursors as well as the XcursorAnimate structure.

Cursor XcursorAnimateNext (
	XcursorAnimate 	*animate)

Cyclically returns the next Cursor in the array, incrementing the sequence number to prepare for the next call.

The caller is responsible for displaying the series of Cursor images. Xcursor does not do that.

The X11 XCreateFontCursor and XCreateGlyphCursor functions use this part of the API to extend the X core cursors feature to use themes.

void XcursorImageHash (

Compute a hash of the image, to display when the environment variable XCURSOR_DISCOVER is set.

Associates the given name with the images.

void XcursorNoticeCreateBitmap (
	Display 	*dpy,
	Pixmap 	 pid,
	unsigned int 	 width,
	unsigned int 	 height)

Check if the display supports either ARGB or themes, and also if the image size fits within the maximum cursor size (64 pixels). If so, create a bitmap of the specified size, and cache the result in Xcursor, identifying it with the Pixmap-id (pid) value.

void XcursorNoticePutBitmap (
	Display 	*dpy,
	Drawable 	 draw,
	XImage 	*image)

Update the image contents in the bitmap specified by the draw value (a Pixmap-id). The bitmap must have been created by XcursorNoticeCreateBitmap.

Cursor XcursorTryShapeBitmapCursor (
	Display 	*dpy,
	Pixmap 	 source,
	Pixmap 	 mask,
	XColor 	*foreground,
	XColor 	*background,
	unsigned int 	 x,
	unsigned int 	 y)

If the display supports either ARGB or themes, try to load a cursor into Xcursor's cache using the source parameter as a Pixmap-id. The source may no longer be in the cache. Xcursor uses the hash value to identify the desired image.

Cursor XcursorTryShapeCursor (
	Display 	*dpy,
	Font 	 source_font,
	Font 	 mask_font,
	unsigned int 	 source_char,
	unsigned int 	 mask_char,
	XColor _Xconst 	*foreground,
	XColor _Xconst 	*background)

If the display supports either ARGB or themes, try to load a cursor into Xcursor's cache using the source_char parameter as a shape. Using

  • the default size from XcursorGetDefaultSize,
  • the default theme from XcursorGetTheme, and
  • the source_char parameter as a shape,
Xcursor calls XcursorShapeLoadImages to load the cursor images. If successful, Xcursor uses XcursorImagesLoadCursor to load the cursor information.

ENVIRONMENT

Environment variables can be used to override resource settings, which in turn override compiled-in default values.

Some of the environment variables recognized by Xcursor are booleans, specified as follows:

true for “t”, “1”, “y” or “on”
false for “f”, “0”, “n” or “off”

Xcursor ignores other values for these booleans.

HOME
Xcursor interprets “~” in the search list as the home directory, using this variable rather than the password database.
XCURSOR_ANIM
If the display supports the Render CreateCursor request, and the Render feature is enabled, disable animated cursors if the environment variable is false.
If the environment variable is not given, Xcursor uses the resource Xcursor.anim.
XCURSOR_CORE
If the display supports the Render CreateCursor request disable the Render feature if the environment variable is false.
If the environment variable is not given, Xcursor uses the resource Xcursor.core.
XCURSOR_DISCOVER
If the variable is set, Xcursor turns on a logging feature. It displays the hash value and the image so that users can see which cursor name is associated with each image.
There is no corresponding resource setting.
XCURSOR_DITHER
This variable sets the desired dither.
If the environment variable is not given, Xcursor uses the resource Xcursor.dither.
If neither environment variable or resource is found, Xcursor uses “threshold”
These are the recognized values:
diffuse
median
ordered
threshold
XCURSOR_PATH
This variable sets the list of paths in which to search for cursors, rather than the compiled-in default list.
Directories in this path are separated by colons (:).
XCURSOR_RESIZED
Enables automatic resizing of cursors to improve their displayed size if the environment variable is true.
If the environment variable is not given, Xcursor tries the Xcursor.resized resource.
XCURSOR_SIZE
This variable sets the desired cursor size, in pixels.
If the environment variable is not given, Xcursor tries the Xcursor.size resource.
If no size is given, whether by environment variable or resource setting, Xcursor next tries the Xft.dpi resource setting to guess the size of a 16-point cursor.
Finally, if Xft.dpi is not set, Xcursor uses the display height, dividing by 48 (assuming that the height is 768).
XCURSOR_THEME
This variable selects the desired theme.
If the environment variable is not given, Xcursor tries the Xcursor.theme resource.
If neither environment variable or resource is found, Xcursor uses the default theme.
XCURSOR_THEME_CORE
Enable or disable the use of themes for core X cursors. If the environment variable is true, themes will be used for core cursors. Otherwise, they will not.
If the environment variable is not given, Xcursor uses the resource Xcursor.theme_core.

XcursorCursorsCreate(3)

الحزمة: libxcursor

التراخيص: HPND-sell-variant

NAME

XcursorCursorsCreate, XcursorCursorsDestroy, XcursorCursorsNext, XcursorCursorsSet->"XcursorCursorsCreate(3)"

SYNOPSIS

#include <Xcursor/Xcursor.h>

XcursorCursors XcursorCursorsCreate(Display *display, const char *name);
void XcursorCursorsDestroy(XcursorCursors cursors);
int XcursorCursorsNext(XcursorCursors cursors, XcursorCursor **cursor_ret);
void XcursorCursorsSet(Display *display, const char *name);

DESCRIPTION

The Xcursor library provides functions for loading and manipulating cursor themes.

XcursorCursorsCreate() creates a cursor object for the given cursor name. The name is typically the name of a cursor theme directory in the user's theme search path.

XcursorCursorsDestroy() destroys the given cursor object and frees all associated resources.

XcursorCursorsNext() returns the next cursor in the cursor object. If there are no more cursors, it returns 0. The returned cursor pointer is valid until the next call to XcursorCursorsNext() or until XcursorCursorsDestroy() is called.

XcursorCursorsSet() sets the default cursor theme for the given display to the specified name. This function is intended to be called by applications to allow users to override the system-wide default cursor theme.

FUNCTIONAL OVERVIEW

The Xcursor library allows applications to load and display cursor themes. Cursor themes are collections of cursor images that can be used to customize the appearance of the mouse pointer.

The library provides functions for creating and destroying cursor objects, iterating through the cursors within a theme, and setting the default cursor theme.

DATATYPES

XcursorCursors
An opaque type representing a collection of cursors.
XcursorCursor
A structure representing a single cursor. This structure is opaque and should not be accessed directly.

FUNCTIONS

XcursorCursorsCreate()
Creates a cursor object for the specified cursor theme name.
XcursorCursorsDestroy()
Destroys a cursor object and frees associated resources.
XcursorCursorsNext()
Retrieves the next cursor from a cursor object.
XcursorCursorsSet()
Sets the default cursor theme for the display.

ENVIRONMENT

Xcursor.theme
Specifies the cursor theme to use. If not set, the system default is used.
Xcursor.theme_core
Enables themes for core cursors if the environment variable is true.
If the environment variable is not given, Xcursor tries the Xcursor.theme_core resource.
An application can enable or disable themes using XcursorSetThemeCore.

CAVEATS

Xcursor will probably change radically in the future; weak attempts will be made to retain some level of source-file compatibility.

#include <Xcursor/Xcursor.h>

XcursorCursors XcursorCursorsCreate(Display *display, const char *name);

الوصف

توفر مكتبة Xcursor وظائف لتحميل ومعالجة سمات المؤشر.

تقوم XcursorCursorsCreate() بإنشاء كائن مؤشر لاسم المؤشر المعطى name. الاسم هو عادةً اسم دليل سمة المؤشر في مسار البحث عن السمات الخاص بالمستخدم.

تقوم XcursorCursorsDestroy() بتدمير كائن المؤشر المعطى وتحرير جميع الموارد المرتبطة به.

تقوم XcursorCursorsNext() بإرجاع المؤشر التالي في كائن المؤشر. إذا لم تكن هناك مؤشرات أخرى، فإنها ترجع 0. يشير المؤشر الذي تم إرجاعه صالحًا حتى المكالمة التالية لـ XcursorCursorsNext() أو حتى يتم استدعاء XcursorCursorsDestroy().

تقوم XcursorCursorsSet() بتعيين سمة المؤشر الافتراضية لـ display المعطى إلى الاسم المحدد name. هذه الوظيفة مخصصة ليتم استدعاؤها بواسطة التطبيقات للسماح للمستخدمين بتجاوز سمة المؤشر الافتراضية على مستوى النظام.

نظرة عامة وظيفية

تسمح مكتبة Xcursor للتطبيقات بتحميل وعرض سمات المؤشر. سمات المؤشر هي مجموعات من صور المؤشر التي يمكن استخدامها لتخصيص مظهر مؤشر الماوس.

توفر المكتبة وظائف لإنشاء وتدمير كائنات المؤشر، والتكرار عبر المؤشرات داخل السمة، وتعيين سمة المؤشر الافتراضية.

أنواع البيانات

XcursorCursors
نوع غامض يمثل مجموعة من المؤشرات.
XcursorCursor
هيكل يمثل مؤشرًا فرديًا. هذا الهيكل غامض ولا ينبغي الوصول إليه مباشرة.

الوظائف

XcursorCursorsCreate()
ينشئ كائن مؤشر لاسم سمة المؤشر المحدد.
XcursorCursorsDestroy()
يدمر كائن مؤشر ويحرر الموارد المرتبطة به.
XcursorCursorsNext()
يسترجع المؤشر التالي من كائن المؤشر.
XcursorCursorsSet()
يعين سمة المؤشر الافتراضية للشاشة.

البيئة

Xcursor.theme
يحدد سمة المؤشر المراد استخدامها. إذا لم يتم تعيينها، يتم استخدام الافتراضي النظام.
Xcursor.theme_core
يمكّن السمات للمؤشرات الأساسية إذا كان متغير البيئة true.
إذا لم يتم إعطاء متغير البيئة، تحاول Xcursor المورد Xcursor.theme_core.
يمكن للتطبيق تمكين أو تعطيل السمات باستخدام XcursorSetThemeCore.

تحذيرات

Xcursor من المحتمل أن يتغير بشكل جذري في المستقبل؛ سيتم بذل محاولات ضعيفة للحفاظ على مستوى معين من التوافق في ملفات المصدر.
