الاسم

Xcursor - مكتبة إدارة المؤشرات

الملخص

#include <X11/Xcursor/Xcursor.h>

الوصف

Xcursor مكتبة بسيطة مصممة للمساعدة في تحديد موقع المؤشرات وتحميلها. يمكن تحميل المؤشرات من الملفات أو من الذاكرة. توجد مكتبة من المؤشرات الشائعة التي تُطابق أسماء مؤشرات X القياسية. يمكن أن توجد المؤشرات بعدة أحجام وتختار المكتبة تلقائيًا أفضل حجم.

نظرة عامة وظيفية

بُنيت Xcursor في عدة طبقات؛ في الطبقة السفلى توجد شيفرة يمكنها تحميل صور المؤشرات من الملفات. فوق ذلك توجد طبقة تحدد موقع ملفات المؤشرات بناءً على مسار المكتبة والسمة. في الأعلى توجد طبقة تبني المؤشرات إما من صورة محمّلة من ملف أو من أحد مؤشرات X القياسية. عند استخدام صور محمّلة من الملفات، تفضّل Xcursor استخدام طلب CreateCursor من امتداد Render إذا كان مدعومًا من خادم X. حيث لا يكون مدعومًا، تُطابق Xcursor صورة المؤشر مع مؤشر X قياسي وتستخدم طلب CreateCursor الأساسي.

ملفات المؤشرات

تُعرّف Xcursor تنسيقًا جديدًا للمؤشرات على القرص. يحمل كل ملف صورة مؤشر واحدة أو أكثر. تُوسم كل صورة مؤشر بحجم اسمي بحيث يمكن اختيار أفضل حجم تلقائيًا. يمكن تحميل مؤشرات متعددة من الحجم الاسمي نفسه معًا؛ ويُتوقع من التطبيقات استخدامها كتسلسل متحرك.

تُخزّن ملفات المؤشرات كترويسة تحتوي على جدول محتويات يتبعه تسلسل من الكتل. يشير جدول المحتويات إلى النوع والنوع الفرعي وموضع كل كتلة في الملف. تبدو ترويسة الملف هكذا:


magic:	CARD32 "Xcur" (0x58, 0x63, 0x75, 0x72)
header:	CARD32 بايت في هذه الترويسة
version:	CARD32 رقم إصدار الملف
ntoc:	CARD32 عدد مدخلات جدول المحتويات
toc:	LISTofTOC جدول المحتويات

يبدو كل مدخل في جدول المحتويات هكذا:


type:	CARD32 نوع المدخل
subtype:	CARD32 تسمية خاصة بالنوع - الحجم للصور
position:	CARD32 الموضع المطلق بالبايت للجدول في الملف

تحتوي كل كتلة في الملف على مجموعة من حقول الترويسة المشتركة تتبعها حقول إضافية خاصة بالنوع:


header:	CARD32 بايت في ترويسة الكتلة (بما في ذلك الحقول الخاصة بالنوع)
type:	CARD32 يجب أن يطابق النوع في جدول المحتويات لهذه الكتلة
subtype:	CARD32 يجب أن يطابق النوع الفرعي في جدول المحتويات لهذه الكتلة
version:	CARD32 رقم الإصدار لنوع هذه الكتلة

يوجد حاليًا نوعان من الكتل معرّفان لملفات المؤشرات؛ التعليقات والصور. تبدو التعليقات هكذا:


header:	20 ترويسات التعليقات هي 20 بايت
type:	0xfffe0001 نوع التعليق هو 0xfffe0001
subtype:	{ 1 (COPYRIGHT), 2 (LICENSE), 3 (OTHER) }
version:	1
length:	CARD32 طول سلسلة UTF-8 بالبايت
string:	LISTofCARD8 سلسلة UTF-8

تبدو الصور هكذا:


header:	36 ترويسات الصور هي 36 بايت
type:	0xfffd0002 نوع الصورة هو 0xfffd0002
subtype:	CARD32 النوع الفرعي للصورة هو الحجم الاسمي
version:	1
width:	CARD32 يجب أن يكون أقل من أو يساوي 0x7fff
height:	CARD32 يجب أن يكون أقل من أو يساوي 0x7fff
xhot:	CARD32 يجب أن يكون أقل من أو يساوي العرض
yhot:	CARD32 يجب أن يكون أقل من أو يساوي الارتفاع
delay:	CARD32 التأخير بين إطارات الحركة بالمللي ثانية
pixels:	LISTofCARD32 بكسلات بتنسيق ARGB المعبأ

السمات

تتبع Xcursor (في الغالب) مواصفات freedesktop.org لسمات الأيقونات. مسار البحث الافتراضي الذي تستخدمه هو

~/.local/share/icons, ~/.icons, /usr/share/icons, /usr/share/pixmaps

داخل كل من هذه الأدلة، تبحث عن دليل باستخدام اسم السمة:

•
داخل دليل السمة، تبحث عن ملفات المؤشرات في الدليل الفرعي "cursors".
تبحث Xcursor عن ملف محدد، والذي يجب أن يكون أحد أسماء شكل المؤشر، على سبيل المثال، كما يُستخدم في XcursorLibraryLoadImage أو XcursorLibraryShape.
•
إذا لم تجد أي ملف مؤشر مطابق في الدليل الفرعي "cursors"، تبحث Xcursor بعد ذلك عن ملف "index.theme" في كل دليل سمة للبحث عن السمات الموروثة. تلك هي أسطر بهذا التنسيق:
Inherits = theme-name

تستخدم Xcursor أول theme-name موروث، متجاهلةً الأخرى التي قد توجد في ملف "index.theme" معين. إذا وجدت سمة موروثة، تبحث Xcursor على طول المسار لاستخدامها أيضًا. تتجاهل Xcursor المفاتيح الأخرى في ملف "index.theme"، بما في ذلك "Name" (أي الاسم الذي قد يستخدمه تطبيق رسومي كـاسم العرض).

قد يُدرج أكثر من theme-name واحد في سطر Inherits=. تنص مواصفات freedesktop.org على أن عناصر القائمة مفصولة بفواصل. تقبل Xcursor أيضًا الفاصلة المنقوطة، لكنها تترجم كليهما إلى نقطتين عند البحث في المسار. تتوقع Xcursor سطر Inherits= واحدًا فقط؛ مواصفات freedesktop.org غير واضحة بشأن ما إذا كانت المفاتيح المتعددة مسموحًا بها.

إذا لم تُضبط أي سمة، أو إذا لم يُعثر على أي مؤشر للسمة المحددة في أي مكان على طول المسار، تتحقق Xcursor من سمة "default".

عندما تجد Xcursor ملف مؤشر، تتوقف عن البحث. تستخدم دائمًا أول ملف مؤشر يُعثر عليه أثناء البحث على طول المسار.

أنواع البيانات

XcursorImage
يحمل صورة مؤشر واحدة في الذاكرة. كل بكسل في المؤشر هو قيمة 32 بت تحتوي على ARGB مع A في البايت العالي. 


typedef struct _XcursorImage {
    XcursorDim 	 size; 	/* الحجم الاسمي للمطابقة */
    XcursorDim 	 width; 	/* العرض الفعلي */
    XcursorDim 	 height; 	/* الارتفاع الفعلي */
    XcursorDim 	 xhot; 	/* النقطة الساخنة x (يجب أن تكون داخل الصورة) */
    XcursorDim 	 yhot; 	/* النقطة الساخنة y (يجب أن تكون داخل الصورة) */
    XcursorPixel 	*pixels; 	/* مؤشر إلى البكسلات */
} XcursorImage;
XcursorImages
يحمل بنى XcursorImage متعددة. يتم تحريرها جميعًا عندما يتم تحرير XcursorImages في XcursorImagesDestroy. 


typedef struct _XcursorImages {
    int 	 nimage; 	/* عدد الصور */
    XcursorImage 	**images; 	/* مصفوفة من مؤشرات XcursorImage */
} XcursorImages;
XcursorCursors
يحمل كائنات Cursor متعددة. يتم تحريرها جميعًا عندما يتم تحرير XcursorCursors. هذه تُعدّ مرجعيًا بحيث يمكن لبنى XcursorAnimate المتعددة استخدام نفس XcursorCursors. 


typedef struct _XcursorCursors {
    Display 	*dpy; 	/* العرض الذي يحمل المؤشرات */
    int 	 ref; 	/* عدد المراجع */
    int 	 ncursor; 	/* عدد المؤشرات */
    Cursor 	*cursors; 	/* مصفوفة من المؤشرات */
} XcursorCursors;
XcursorAnimate
يشير إلى مجموعة من المؤشرات وتسلسل داخل تلك المجموعة. يمكن لبنى XcursorAnimate المتعددة الإشارة إلى نفس XcursorCursors؛ كل واحدة تحمل مرجعًا يُزال عند تحرير XcursorAnimate. 


typedef struct _XcursorAnimate {
    XcursorCursors 	*cursors; 	/* قائمة المؤشرات المراد استخدامها */
    int 	 sequence; 	/* أي مؤشر هو التالي */
} XcursorAnimate;
XcursorFile
توفر Xcursor واجهة برمجية مجردة للوصول إلى بيانات الملف. توفر Xcursor تنفيذ stdio لهذه الواجهة المجردة؛ التطبيقات حرة في إنشاء تنفيذات إضافية. تتوازى هذه الدوال معتتطابق دوال stdio في القيمة المُرجعة وقيم الوسائط المتوقعة؛ أما دالتا read وwrite فتبدّلان الوسائط لتتوافقا مع نسخ POSIX. 


typedef struct _XcursorFile {
    void *closure;
    int (*read)  (XcursorFile *file, unsigned char *buf, int len);
    int (*write) (XcursorFile *file, unsigned char *buf, int len);
    int (*seek)  (XcursorFile *file, long offset, int whence);
};

تبحث هذه الدوال في مسار المكتبة، وتحمّل أول ملف يُعثر عليه بالحجم size المطلوب، باستخدام دالة خاصة (XcursorScanTheme) للعثور على السمة المناسبة:

  • إذا لم تكن theme مساوية لـ NULL، فاستخدمها.
  • إذا كانت theme مساوية لـ NULL، أو لم يكن هناك تطابق للسمة المطلوبة، فاستخدم "default" كاسم للسمة.
  • إذا فشل البحثان، تُرجع هذه الدوال NULL.
تختلف الدالتان بأكثر من عدد الصور المحمّلة:
  • تستدعي XcursorLibraryLoadImage الدالة XcursorFileLoadImage لكن
  • تستدعي XcursorLibraryLoadImages الدالة XcursorFileLoadImages، و(عند النجاح) تستدعي XcursorImagesSetName لربط name بالنتيجة.

البيئة


يمكن استخدام متغيرات البيئة لتجاوز إعدادات الموارد،
    والتي بدورها تتجاوز القيم الافتراضية المضمنة في الترجمة.


بعض متغيرات البيئة التي يتعرف عليها Xcursor هي
    قيم منطقية، تُحدد كما يلي:





  
true لـ
    “t”، “1”، “y” أو
    “on”

  

  
false لـ
    “f”، “0”، “n” أو
    “off”

  






يتجاهل Xcursor القيم الأخرى لهذه القيم المنطقية.



  
HOME

  
يفسّر Xcursor الرمز “~” في قائمة البحث على أنه دليل
      المنزل، باستخدام هذا المتغير بدلاً من قاعدة بيانات كلمات المرور.

  
XCURSOR_ANIM

  
إذا كان العرض يدعم طلب Render CreateCursor، وكانت ميزة Render
      ممكّنة، فسيتم تعطيل المؤشرات المتحركة إذا كان متغير
      البيئة false.





  

  
إذا لم يُعطَ متغير البيئة، يستخدم Xcursor المورد
      Xcursor.anim.





  
XCURSOR_CORE

  
إذا كان العرض يدعم طلب Render CreateCursor، فسيتم تعطيل ميزة Render
      إذا كان متغير البيئة false.





  

  
إذا لم يُعطَ متغير البيئة، يستخدم Xcursor المورد
      Xcursor.core.





  
XCURSOR_DISCOVER

  
إذا كان المتغير مضبوطًا، يفعّل Xcursor ميزة تسجيل. يعرض
      قيمة التجزئة والصورة حتى يتمكن المستخدمون من رؤية اسم المؤشر المرتبط
      بكل صورة.





  

  
لا يوجد إعداد مورد مقابل.





  
XCURSOR_DITHER

  
يضبط هذا المتغير التنقيط المطلوب.





  

  
إذا لم يُعطَ متغير البيئة، يستخدم Xcursor المورد
      Xcursor.dither.

  

  
إذا لم يُعثر على متغير بيئة أو مورد، يستخدم Xcursor
      “threshold”

  

  
هذه هي القيم المعترف بها:







  
diffuse

  

  
median

  

  
ordered

  

  
threshold

  







  
XCURSOR_PATH

  
يضبط هذا المتغير قائمة المسارات التي يُبحث فيها عن المؤشرات،
      بدلاً من القائمة الافتراضية المضمنة في الترجمة.





  

  
تُفصل الأدلة في هذا المسار بنقطتين (:).





  
XCURSOR_RESIZED

  
يمكّن تغيير الحجم التلقائي للمؤشرات لتحسين حجمها المعروض إذا
      كان متغير البيئة true.





  

  
إذا لم يُعطَ متغير البيئة، يحاول Xcursor المورد
      Xcursor.resized.





  
XCURSOR_SIZE

  
يضبط هذا المتغير حجم المؤشر المطلوب، بالبكسل.





  

  
إذا لم يُعطَ متغير البيئة، يحاول Xcursor المورد
      Xcursor.size.

  

  
إذا لم يُعطَ أي حجم، سواء بواسطة متغير البيئة أو إعداد المورد،
      يحاول Xcursor بعد ذلك إعداد المورد Xft.dpi لتخمين حجم
      مؤشر بحجم 16 نقطة.

  

  
أخيرًا، إذا لم يكن Xft.dpi مضبوطًا، يستخدم Xcursor ارتفاع العرض،
      مقسومًا على 48 (بافتراض أن الارتفاع هو 768).





  
XCURSOR_THEME

  
يختار هذا المتغير السمة المطلوبة.





  

  
إذا لم يُعطَ متغير البيئة، يحاول Xcursor المورد
      Xcursor.theme.

  

  
إذا لم يُعثر على متغير بيئة أو مورد، يستخدم Xcursor السمة
      الافتراضية.





  
XC

  
يمكّن السمات لمؤشرات النواة إذا كانت متغيرات البيئة
      true.





  

  
إذا لم يتم إعطاء متغير البيئة، يحاول Xcursor استخدام
      مورد Xcursor.theme_core.

  

  
يمكن للتطبيق تمكين أو تعطيل السمات باستخدام
    XcursorSetThemeCore.
