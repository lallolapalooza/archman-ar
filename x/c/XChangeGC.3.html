XChangeGC(3)

الحزمة: libx11

التراخيص: MIT AND X11

XCreateGC(3) وظائف XLIB XCreateGC(3)

الاسم

XCreateGC، XCopyGC، XChangeGC، XGetGCValues، XFreeGC، XGContextFromGC، XGCValues - إنشاء أو تحرير سياقات الرسوميات وهياكل سياق الرسوميات

الصيغة

GC XCreateGC(Display *display, Drawable d, unsigned long valuemask, XGCValues *values);

int XCopyGC(Display *display, GC src, unsigned long valuemask, GC dest);

int XChangeGC(Display *display, GC gc, unsigned long valuemask, XGCValues *values);

Status XGetGCValues(Display *display, GC gc, unsigned long valuemask, XGCValues *values_return);

int XFreeGC(Display *display, GC gc);

GContext XGContextFromGC(GC gc);

الوسائط

d
يحدد الرسم القابل للرسم.
dest
يحدد GC الوجهة.
display
يحدد الاتصال بخادم X.
gc
يحدد GC.
src
يحدد مكونات GC المصدر.
valuemask
يحدد أي مكونات في GC سيتم تعيينها أو نسخها أو تغييرها أو إرجاعها. هذه الوسيطة هي OR الشامل (bitwise inclusive OR) لصفر أو أكثر من بتات قناع مكونات GC الصالحة.
values
يحدد أي قيم كما هو محدد بواسطة قناع القيمة.
values_return
يعيد قيم GC في هيكل XGCValues المحدد.

الوصف

تقوم الدالة XCreateGC بإنشاء سياق رسوميات وتعيد GC. يمكن استخدام GC مع أي رسم قابل للرسم وجهته له نفس الجذر والعمق مثل الرسم المحدد. يؤدي الاستخدام مع رسومات قابلة للرسم أخرى إلى خطأ BadMatch.

يمكن لـ XCreateGC إنشاء أخطاء BadAlloc، BadDrawable، BadFont، BadMatch، BadPixmap، و BadValue.

تقوم الدالة XCopyGC بنسخ المكونات المحددة من GC المصدر إلى GC الوجهة. يجب أن يكون لـ GC المصدر والوجهة نفس الجذر والعمق، وإلا فسيحدث خطأ BadMatch. يحدد قناع القيمة المكون المراد نسخه، كما هو الحال في XCreateGC.

يمكن لـ XCopyGC إنشاء أخطاء BadAlloc، BadGC، و BadMatch.

تقوم الدالة XChangeGC بتغيير المكونات المحددة بواسطة قناع القيمة لـ GC المحدد. تحتوي وسيطة القيم على القيم المراد تعيينها. القيم والقيود هي نفسها كما في XCreateGC. تغيير قناع القطع (clip-mask) يلغي أي طلب XSetClipRectangles سابق على السياق. تغيير إزاحة الخط (dash-offset) أو قائمة الخطوط (dash-list) يلغي أي طلب XSetDashes سابق على السياق. ترتيب التحقق من المكونات وتغييرها يعتمد على الخادم. إذا تم إنشاء خطأ، فقد تم تعديل مجموعة فرعية من المكونات.

يمكن لـ XChangeGC إنشاء أخطاء BadAlloc، BadFont، BadGC، BadMatch، BadPixmap، و BadValue.

تقوم الدالة XGetGCValues بإعادة المكونات المحددة بواسطة قناع القيمة لـ GC المحدد. إذا كان قناع القيمة يحتوي على مجموعة صالحة من بتات قناع GC (GCFunction، GCPlaneMask، GCForeground، GCBackground، GCLineWidth، GCLineStyle، GCCapStyle، GCJoinStyle، GCFillStyle، GCFillRule، GCTile، GCStipple، GCTileStipXOrigin، GCTileStipYOrigin، GCFont، GCSubwindowMode، GCGraphicsExposures، GCClipXOrigin، GCClipYOrigin، GCDashOffset، أو GCArcMode) ولم يحدث أي خطأ، فإن XGetGCValues تعين المكونات المطلوبة في values_return وتعيد حالة غير صفرية. بخلاف ذلك، فإنها تعيد حالة صفرية. لاحظ أنه لا يمكن طلب قناع القطع وقائمة الخطوط (الممثلة بالبتات GCClipMask و GCDashList على التوالي في قناع القيمة). لاحظ أيضًا أنه سيتم إرجاع معرف مورد غير صالح (مع واحد أو أكثر من أهم ثلاث بتات مضبوطة على 1) لـ GCFont، GCTile، و GCStipple إذا لم يتم تعيين المكون بشكل صريح بواسطة العميل.

تقوم الدالة XFreeGC بتدمير GC المحدد بالإضافة إلى جميع التخزين المرتبط به.

يمكن لـ XFreeGC إنشاء خطأ BadGC.

الهياكل

يحتوي هيكل XGCValues على:

/* بتات قناع سمات GC */

#define GCFunction (1L<<0)
#define GCPlaneMask (1L<<1)
#define GCForeground (1L<<2)
#define GCBackground (1L<<3)
#define GCLineWidth (1L<<4)
#define GCLineStyle (1L<<5)
#define GCCapStyle (1L<<6)
#define GCJoinStyle (1L<<7)
#define GCFillStyle (1L<<8)
#define GCFillRule (1L<<9)
#define GCTile (1L<<10)
#define GCStipple (1L<<11)
#define GCTileStipXOrigin (1L<<12)
#define GCTileStipYOrigin (1L<<13)
#define GCFont (1L<<14)
#define GCSubwindowMode (1L<<15)
#define GCGraphicsExposures (1L<<16)
#define GCClipXOrigin (1L<<17)
#define GCClipYOrigin (1L<<18)
#define GCClipMask (1L<<19)
#define GCDashOffset (1L<<20)
#define GCDashList (1L<<21)
#define GCArcMode (1L<<22) 
/* Values */
typedef struct {
        int function;   /* logical operation */
        unsigned long plane_mask;       /* plane mask */
        unsigned long foreground;       /* foreground pixel */
        unsigned long background;       /* background pixel */
        int line_width; /* line width (in pixels) */
        int line_style; /* LineSolid, LineOnOffDash, LineDoubleDash */
        int cap_style;  /* CapNotLast, CapButt, CapRound, CapProjecting */
        int join_style; /* JoinMiter, JoinRound, JoinBevel */
        int fill_style; /* FillSolid, FillTiled, FillStippled FillOpaqueStippled*/
        int fill_rule;  /* EvenOddRule, WindingRule */
        int arc_mode;   /* ArcChord, ArcPieSlice */
        Pixmap tile;    /* tile pixmap for tiling operations */
        Pixmap stipple; /* stipple 1 plane pixmap for stippling */
        int ts_x_origin;        /* offset for tile or stipple operations */
        int ts_y_origin;
        Font font;      /* default text font for text operations */
        int subwindow_mode;     /* ClipByChildren, IncludeInferiors */
        Bool graphics_exposures;        /* boolean, should exposures be generated */
        int clip_x_origin;      /* origin for clipping */
        int clip_y_origin;
        Pixmap clip_mask;       /* bitmap clipping; other calls for rects */
        int dash_offset;        /* patterned/dashed line information */
        char dashes;
} XGCValues;

تُستخدم سمات الوظيفة لـ GC عند تحديث قسم من الرسم القابل للرسم (الوجهة) باستخدام بتات من مكان آخر (المصدر). تحدد الوظيفة في GC كيفية حساب بتات الوجهة الجديدة من بتات المصدر وبتات الوجهة القديمة. عادةً ما تكون GXcopy الأكثر فائدة لأنها ستعمل على شاشة ملونة، ولكن التطبيقات الخاصة قد تستخدم وظائف أخرى، خاصة بالتنسيق مع مستويات معينة من شاشة ملونة. وظائف GC الـ 16
