Xcursor(3) إصدار X 11 Xcursor(3)

الاسم

Xcursor - مكتبة إدارة المؤشر

ملخص

#include <X11/Xcursor/Xcursor.h>

الوصف

Xcursor هي مكتبة بسيطة مصممة للمساعدة في تحديد موقع وتحميل المؤشرات. يمكن تحميل المؤشرات من الملفات أو الذاكرة. توجد مكتبة من المؤشرات الشائعة التي ترتبط بأسماء مؤشرات X القياسية. يمكن أن توجد المؤشرات بأحجام متعددة وتقوم المكتبة تلقائيًا باختيار أفضل حجم.

نظرة عامة وظيفية

تم بناء Xcursor على عدة طبقات؛ في الطبقة السفلية يوجد الكود الذي يمكنه تحميل صور المؤشرات من الملفات. فوق ذلك توجد طبقة تحدد ملفات المؤشر بناءً على مسار المكتبة والموضوع. في القمة توجد طبقة تبني المؤشرات إما من صورة تم تحميلها من ملف أو أحد مؤشرات X القياسية. عند استخدام صور تم تحميلها من الملفات، تفضل Xcursor استخدام طلب إنشاء المؤشر (Render extension CreateCursor request) إذا كان خادم X يدعمه. حيثما لم يكن مدعومًا، تقوم Xcursor بربط صورة المؤشر بمؤشر X قياسي وتستخدم طلب الإنشاء الأساسي (core CreateCursor request).

ملفات المؤشرات

تحدد Xcursor تنسيقًا جديدًا للمؤشرات على القرص. يحتوي كل ملف على صورة مؤشر واحدة أو أكثر. يتم تمييز كل صورة مؤشر بحجم اسمي بحيث يمكن اختيار أفضل حجم تلقائيًا. يمكن تحميل مؤشرات متعددة بنفس الحجم الاسمي معًا؛ من المتوقع أن تستخدمها التطبيقات كتسلسل متحرك.

تُخزن ملفات المؤشرات كرأس يحتوي على جدول محتويات متبوعًا بسلسلة من الأجزاء. يشير رأس الملف إلى نوع، ونوع فرعي، وموضع في الملف لكل جزء. رأس الملف يبدو كالتالي:


magic:	CARD32 "Xcur" (0x58, 0x63, 0x75, 0x72)
header:	CARD32 بايت في هذا الرأس
version:	CARD32 رقم إصدار الملف
ntoc:	CARD32 عدد إدخالات جدول المحتويات
toc:	LISTofTOC جدول المحتويات

كل إدخال في جدول المحتويات يبدو كالتالي:


type:	CARD32 نوع الإدخال
subtype:	CARD32 تسمية خاصة بالنوع - حجم للصور
position:	CARD32 الموضع البايتي المطلق للجدول في الملف

لكل جزء في الملف مجموعة من حقول الرأس المشتركة متبوعة بحقول إضافية خاصة بالنوع:


header:	CARD32 بايت في رأس الجزء (بما في ذلك الحقول الخاصة بالنوع)
type:	CARD32 يجب أن يتطابق مع النوع في جدول المحتويات لهذا الجزء
subtype:	CARD32 يجب أن يتطابق مع النوع الفرعي في جدول المحتويات لهذا الجزء
version:	CARD32 رقم إصدار لهذا النوع من الجزء

يوجد حاليًا نوعان من الأجزاء محددين لملفات المؤشرات؛ التعليقات والصور. تبدو التعليقات كالتالي:


header:	20 رؤوس التعليقات تكون 20 بايت
type:	0xfffe0001 نوع التعليق هو 0xfffe0001
subtype:	{ 1 (حقوق النشر), 2 (الترخيص), 3 (أخرى) }
version:	1
length:	CARD32 طول البايت للسلسلة UTF-8
string:	LISTofCARD8 سلسلة UTF-8

تبدو الصور كالتالي:


header:	36 رؤوس الصور تكون 36 بايت
type:	0xfffd0002 نوع الصورة هو 0xfffd0002
subtype:	CARD32 النوع الفرعي للصورة هو الحجم الاسمي
version:	1
width:	CARD32 يجب أن يكون أقل من أو يساوي 0x7fff
height:	CARD32 يجب أن يكون أقل من أو يساوي 0x7fff
xhot:	CARD32 يجب أن يكون أقل من أو يساوي العرض
yhot:	CARD32 يجب أن يكون أقل من أو يساوي الارتفاع
delay:	CARD32 التأخير بين إطارات الرسوم المتحركة بالمللي ثانية
pixels:	LISTofCARD32 بكسلات بتنسيق ARGB مضغوط

المواضيع

تتبع Xcursor (في الغالب) مواصفات freedesktop.org لمواضيع الأيقونات. المسار الافتراضي للبحث الذي تستخدمه هو

~/.local/share/icons, ~/.icons, /usr/share/icons, /usr/share/pixmaps

داخل كل من هذه الأدلة، تبحث عن دليل باستخدام اسم الموضوع:

•
داخل دليل الموضوع، تبحث عن ملفات المؤشر في الدليل الفرعي "cursors".
تبحث Xcursor عن ملف محدد، والذي يجب أن يكون أحد أسماء شكل المؤشر، على سبيل المثال، كما هو مستخدم في XcursorLibraryLoadImage أو XcursorLibraryShape.
•
إذا لم تجد ملف مؤشر مطابق في الدليل الفرعي "cursors"، فإن Xcursor تبحث بعد ذلك عن ملف "index.theme" في كل دليل موضوع للبحث عن المواضيع الموروثة. هذه عبارة عن أسطر بهذا الشكل:
Inherits = اسم-الموضوع

تستخدم Xcursor أول اسم-الموضوع موروث، متجاهلة الآخرين الذين قد يكونون موجودين في ملف "index.theme" معين. إذا وجدت موضوعًا موروثًا، فإن Xcursor تبحث على طول المسار لاستخدامه أيضًا. تتجاهل Xcursor المفاتيح الأخرى في ملف "index.theme"، بما في ذلك "Name" (أي، الاسم الذي قد يستخدمه تطبيق رسومي كـ اسم العرض).

قد يتم إدراج أكثر من اسم-الموضوع واحد في سطر Inherits=. تنص مواصفات freedesktop.org على أن عناصر القائمة مفصولة بفواصل. تقبل Xcursor أيضًا الفاصلة المنقوطة، ولكنها تترجم كليهما إلى نقطتين رأسيتين عند البحث في المسار. تتوقع Xcursor سطر Inherits= واحد فقط؛ مواصفات freedesktop.org غير واضحة ما إذا كان يمكن السماح بمفاتيح متعددة.

إذا لم يتم تعيين موضوع، أو إذا لم يتم العثور على مؤشر للموضوع المحدد في أي مكان على طول المسار، فإن Xcursor تتحقق من موضوع "الافتراضي".

عندما تجد Xcursor ملف مؤشر، فإنها تتوقف عن البحث. تستخدم دائمًا ملف المؤشر الأول الذي تم العثور عليه أثناء البحث على طول المسار.

أنواع البيانات

XcursorImage
يحمل صورة مؤشر واحدة في الذاكرة. كل بكسل في المؤشر هو قيمة 32 بت تحتوي على ARGB مع A في البايت الأعلى. 


typedef struct _XcursorImage {
    XcursorDim 	 size; 	/* حجم اسمي للمطابقة */
    XcursorDim 	 width; 	/* العرض الفعلي */
    XcursorDim 	 height; 	/* الارتفاع الفعلي */
    XcursorDim 	 xhot; 	/* نقطة التنشيط x (يجب أن تكون داخل الصورة) */
    XcursorDim 	 yhot; 	/* نقطة التنشيط y (يجب أن تكون داخل الصورة) */
    XcursorPixel 	*pixels; 	/* مؤشر إلى البكسلات */
} XcursorImage;
XcursorImages
تحمل هياكل XcursorImage متعددة. يتم تحريرها جميعًا عند تحرير XcursorImages في XcursorImagesDestroy. 


typedef struct _XcursorImages {
    int 	 nimage; 	/* عدد الصور */
    XcursorImage 	**images; 	/* مصفوفة مؤشرات XcursorImage */
} XcursorImages;
XcursorCursors
تحمل كائنات المؤشر المتعددة. يتم تحريرها جميعًا عند تحرير XcursorCursors. هذه ذات مرجع عددي بحيث يمكن لهياكل XcursorAnimate متعددة استخدام نفس XcursorCursors. 


typedef struct _XcursorCursors {
    Display 	*dpy; 	/* شاشة العرض التي تحتوي على المؤشرات */
    int 	 ref; 	/* عدد المراجع */
    int 	 ncursor; 	/* عدد المؤشرات */
    Cursor 	*cursors; 	/* مصفوفة المؤشرات */
} XcursorCursors;
XcursorAnimate
تشير إلى مجموعة من المؤشرات وتسلسل ضمن هذه المجموعة. قد تشير هياكل XcursorAnimate متعددة إلى نفس XcursorCursors؛ يحتفظ كل منها بمرجع يتم إزالته عند تحرير XcursorAnimate. 


typedef struct _XcursorAnimate {
    XcursorCursors 	*cursors; 	/* قائمة المؤشرات المراد استخدامها */
    int 	 sequence; 	/* أي مؤشر هو التالي */
} XcursorAnimate;
XcursorFile
توفر Xcursor واجهة برمجة تطبيقات مجردة للوصول إلى بيانات الملف. توفر Xcursor تنفيذًا قياسيًا لهذه الواجهة المجردة؛ التطبيقات حرة في إنشاء تطبيقات إضافية. هذه الوظائف توازي معيار

تبحث هذه الوظائف في مسار المكتبة، وتحمل أول ملف يتم العثور عليه بالحجم المطلوب، باستخدام وظيفة خاصة (XcursorScanTheme) للعثور على السمة المناسبة:

  • إذا لم يكن theme NULL، فاستخدمه.
  • إذا كان theme NULL، أو إذا لم يكن هناك تطابق للسمة المطلوبة، فاستخدم "default" لاسم السمة.
  • إذا فشل كلا البحثين، فإن هذه الوظائف ترجع NULL.
تختلف الوظيفتان بأكثر من عدد الصور المحملة:
  • XcursorLibraryLoadImage تستدعي XcursorFileLoadImage ولكن
  • XcursorLibraryLoadImages تستدعي XcursorFileLoadImages، وتقوم (عند النجاح) باستدعاء XcursorImagesSetName لربط name بالنتيجة.

These functions load cursor images from various sources. XcursorFilenameLoadImages loads images from the specified filename. XcursorLibraryLoadImages loads images from the Xcursor library search path. XcursorFileLoadImages loads images from the Xcursor file format. XcursorThemeLoadImages loads images from the specified theme. XcursorThemeLoadCursor loads a specific cursor from a theme.


XcursorInit initializes the Xcursor library. XcursorReset cleans up the Xcursor library. XcursorSetTheme sets the cursor theme to use. XcursorGetTheme retrieves the current cursor theme. XcursorGetDefaultTheme retrieves the default cursor theme for a given screen. XcursorGetThemeInfo retrieves information about a theme. XcursorGetDefaultThemeInfo retrieves information about the default theme for a given screen.


XcursorSetThemeCore enables or disables themes for core cursors. XcursorGetThemeCore retrieves the current setting for core cursor theming.






FUNCTIONAL OVERVIEW


The Xcursor library provides functions for loading and managing cursor
    images. It supports various image formats and theme-based loading. The library
    allows applications to dynamically change cursor themes and manage core cursor
    theming.






DATATYPES


The following data types are used by the Xcursor functions:



  
XcursorPoint

  
Represents a point with x and y coordinates.

  
XcursorImageList

  
Represents a list of cursor images, including size, dimensions, colors,
      and pixel data.

  
XcursorCursor

  
Represents a cursor, containing an array of images, number of images,
      size, and resize mode.

  
XcursorThemeInfo

  
Represents information about a cursor theme, including name, comment,
      vendor, license, screen size, screen resolution, inheritance, and
      directories.








FUNCTIONS



  
XcursorFilenameLoadImages

  
Loads cursor images from a file specified by filename.

  
XcursorLibraryLoadImages

  
Loads cursor images from the Xcursor library search path.

  
XcursorFileLoadImages

  
Loads cursor images from the Xcursor file format.

  
XcursorThemeLoadImages

  
Loads cursor images from the specified theme_name.

  
XcursorThemeLoadCursor

  
Loads a specific cursor named cursor_name from the specified
      theme_name.

  
XcursorInit

  
Initializes the Xcursor library. Returns True on success,
      False otherwise.

  
XcursorReset

  
Cleans up the Xcursor library resources.

  
XcursorSetTheme

  
Sets the cursor theme to be used to theme_name. Returns
      True on success, False otherwise.

  
XcursorGetTheme

  
Returns the name of the currently set cursor theme.

  
XcursorGetDefaultTheme

  
Returns the name of the default cursor theme for the specified screen.
      The screen argument is the screen number.

  
XcursorGetThemeInfo

  
Retrieves information about the specified cursor theme theme_name.
      The retrieved information is stored in the location pointed to by info.
      The caller is responsible for freeing the memory allocated for the
      information.

  
XcursorGetDefaultThemeInfo

  
Retrieves information about the default cursor theme for the specified
      screen. The retrieved information is stored in the location pointed to by
      info. The caller is responsible for freeing the memory allocated
      for the information.

  
XcursorSetThemeCore

  
Enables or disables theming for core X cursors. If enable is
      True, core cursors will be themed according to the current theme.
      If enable is False, core cursors will use their default
      appearance.

  
XcursorGetThemeCore

  
Returns True if theming for core X cursors is enabled,
      False otherwise.








ENVIRONMENT



  
XCURSOR_THEME

  
Sets the default cursor theme name.

  
XCURSOR_SIZE

  
Sets the default cursor size.

  
XCURSOR_THEME_CORE

  
Enables themes for core cursors if the environment variable is
      true.





  

  
If the environment variable is not given, Xcursor tries the
      Xcursor.theme_core resource.

  

  
An application can enable or disable themes using
    XcursorSetThemeCore.

    attempts will be made to retain some level of source-file compatibility.

الوصف


تقوم هذه الدوال بتحميل صور المؤشر من مصادر مختلفة. تقوم XcursorFilenameLoadImages بتحميل الصور من filename المحدد. تقوم XcursorLibraryLoadImages بتحميل الصور من مسار البحث الخاص بمكتبة Xcursor. تقوم XcursorFileLoadImages بتحميل الصور من تنسيق ملف Xcursor. تقوم XcursorThemeLoadImages بتحميل الصور من السمة المحددة. تقوم XcursorThemeLoadCursor بتحميل مؤشر معين من سمة.


تقوم XcursorInit بتهيئة مكتبة Xcursor. تقوم XcursorReset بتنظيف موارد مكتبة Xcursor. تقوم XcursorSetTheme بتعيين سمة المؤشر المراد استخدامها. تسترجع XcursorGetTheme سمة المؤشر الحالية. تسترجع XcursorGetDefaultTheme سمة المؤشر الافتراضية لشاشة معينة. تسترجع XcursorGetThemeInfo معلومات حول سمة معينة. تسترجع XcursorGetDefaultThemeInfo معلومات حول السمة الافتراضية لشاشة معينة.


تقوم XcursorSetThemeCore بتمكين أو تعطيل السمات للمؤشرات الأساسية. تسترجع XcursorGetThemeCore الإعداد الحالي لسمات المؤشرات الأساسية.






نظرة عامة وظيفية


توفر مكتبة Xcursor وظائف لتحميل وإدارة صور المؤشر. وهي تدعم تنسيقات صور مختلفة والتحميل المستند إلى السمات. تتيح المكتبة للتطبيقات تغيير سمات المؤشر ديناميكيًا وإدارة سمات المؤشرات الأساسية.






أنواع البيانات


تُستخدم أنواع البيانات التالية بواسطة دوال Xcursor:



  
XcursorPoint

  
يمثل نقطة بإحداثيات x و y.

  
XcursorImageList

  
يمثل قائمة صور المؤشر، بما في ذلك الحجم والأبعاد والألوان وبيانات البكسل.

  
XcursorCursor

  
يمثل مؤشرًا، يحتوي على مصفوفة من الصور، وعدد الصور، والحجم، ووضع تغيير الحجم.

  
XcursorThemeInfo

  
يمثل معلومات حول سمة مؤشر، بما في ذلك الاسم والتعليق والمورد والترخيص وحجم الشاشة ودقة الشاشة والوراثة والدلائل.








الدوال



  
XcursorFilenameLoadImages

  
يقوم بتحميل صور المؤشر من ملف محدد بواسطة filename.

  
XcursorLibraryLoadImages

  
يقوم بتحميل صور المؤشر من مسار البحث الخاص بمكتبة Xcursor.

  
XcursorFileLoadImages

  
يقوم بتحميل صور المؤشر من تنسيق ملف Xcursor.

  
XcursorThemeLoadImages

  
يقوم بتحميل صور المؤشر من theme_name المحدد.

  
XcursorThemeLoadCursor

  
يقوم بتحميل مؤشر معين باسم cursor_name من theme_name المحدد.

  
XcursorInit

  
يقوم بتهيئة مكتبة Xcursor. يعيد True عند النجاح، و False عند الفشل.

  
XcursorReset

  
يقوم بتنظيف موارد مكتبة Xcursor.

  
XcursorSetTheme

  
يقوم بتعيين سمة المؤشر المراد استخدامها إلى theme_name. يعيد True عند النجاح، و False عند الفشل.

  
XcursorGetTheme

  
يعيد اسم سمة المؤشر المعينة حاليًا.

  
XcursorGetDefaultTheme

  
يعيد اسم سمة المؤشر الافتراضية للشاشة المحددة. الوسيط screen هو رقم الشاشة.

  
XcursorGetThemeInfo

  
يسترجع معلومات حول سمة المؤشر المحددة theme_name. يتم تخزين المعلومات المسترجعة في الموقع الذي يشير إليه info. يكون المتصل مسؤولاً عن تحرير الذاكرة المخصصة للمعلومات.

  
XcursorGetDefaultThemeInfo

  
يسترجع معلومات حول سمة المؤشر الافتراضية للشاشة المحددة. يتم تخزين المعلومات المسترجعة في الموقع الذي يشير إليه info. يكون المتصل مسؤولاً عن تحرير الذاكرة المخصصة للمعلومات.

  
XcursorSetThemeCore

  
يقوم بتمكين أو تعطيل السمات للمؤشرات الأساسية لنظام X. إذا كان enable هو True، فسيتم تطبيق سمات على المؤشرات الأساسية وفقًا للسمة الحالية. إذا كان enable هو False، فستستخدم المؤشرات الأساسية مظهرها الافتراضي.

  
XcursorGetThemeCore

  
يعيد True إذا تم تمكين السمات للمؤشرات الأساسية لنظام X، و False بخلاف ذلك.








البيئة



  
XCURSOR_THEME

  
يحدد اسم سمة المؤشر الافتراضية.

  
XCURSOR_SIZE

  
يحدد حجم المؤشر الافتراضي.

  
XCURSOR_THEME_CORE

  
يمكّن السمات للمؤشرات الأساسية إذا كانت متغير البيئة
      true.





  

  
إذا لم يتم إعطاء متغير البيئة، تحاول Xcursor استخدام المورد
      Xcursor.theme_core.

  

  
يمكن للتطبيق تمكين أو تعطيل السمات باستخدام
    XcursorSetThemeCore.
