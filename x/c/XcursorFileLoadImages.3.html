XcursorFileLoadImages(3)

الحزمة: libxcursor

التراخيص: HPND-sell-variant

XCURSOR(3) X Version 11 XCURSOR(3)

الاسم

Xcursor - مكتبة إدارة المؤشر

الملخص

#include <X11/Xcursor/Xcursor.h>

الوصف

Xcursor هي مكتبة بسيطة مصممة للمساعدة في تحديد مواقع المؤشرات وتحميلها. يمكن تحميل المؤشرات من الملفات أو الذاكرة. توجد مكتبة للمؤشرات الشائعة التي ترتبط بأسماء مؤشرات X القياسية. يمكن أن توجد المؤشرات بأحجام متعددة وتقوم المكتبة تلقائيًا باختيار أفضل حجم.

نظرة عامة وظيفية

تم بناء Xcursor في طبقات قليلة؛ في الطبقة السفلية يوجد كود يمكنه تحميل صور المؤشر من الملفات. فوق ذلك توجد طبقة تحدد مواقع ملفات المؤشر بناءً على مسار المكتبة والموضوع (theme). في الأعلى توجد طبقة تبني المؤشرات إما من صورة تم تحميلها من ملف أو أحد مؤشرات X القياسية. عند استخدام صور تم تحميلها من ملفات، تفضل Xcursor استخدام طلب CreateCursor الخاص بامتداد Render إذا كان مدعومًا من خادم X. في الأماكن التي لا يتم فيها الدعم، تقوم Xcursor بربط صورة المؤشر بمؤشر X قياسي وتستخدم طلب CreateCursor الأساسي.

ملفات المؤشر

تحدد Xcursor تنسيقًا جديدًا للمؤشرات على القرص. كل ملف يحتوي على صورة مؤشر واحدة أو أكثر. كل صورة مؤشر تم تمييزها بحجم اسمي بحيث يمكن اختيار أفضل حجم تلقائيًا. يمكن تحميل مؤشرات متعددة بنفس الحجم الاسمي معًا؛ من المتوقع أن تستخدمها التطبيقات كسلسلة رسوم متحركة.

يتم تخزين ملفات المؤشر كرأس يحتوي على جدول محتويات متبوعًا بسلسلة من الكتل. يشير جدول المحتويات إلى نوع، ونوع فرعي، وموقع في الملف لكل كتلة. رأس الملف يبدو كالتالي:


magic:	CARD32 "Xcur" (0x58, 0x63, 0x75, 0x72)
header:	CARD32 بايتات في هذا الرأس
version:	CARD32 رقم إصدار الملف
ntoc:	CARD32 عدد إدخالات جدول المحتويات
toc:	LISTofTOC جدول المحتويات

كل إدخال في جدول المحتويات يبدو كالتالي:


type:	CARD32 نوع الإدخال
subtype:	CARD32 تسمية خاصة بالنوع - حجم للصور
position:	CARD32 موقع البايت المطلق للجدول في الملف

كل كتلة في الملف تحتوي على مجموعة من حقول الرأس الشائعة متبوعة بحقول إضافية خاصة بالنوع:


header:	CARD32 بايتات في رأس الكتلة (بما في ذلك الحقول الخاصة بالنوع)
type:	CARD32 يجب أن يتطابق مع النوع في جدول المحتويات لهذه الكتلة
subtype:	CARD32 يجب أن يتطابق مع النوع الفرعي في جدول المحتويات لهذه الكتلة
version:	CARD32 رقم إصدار لهذا النوع من الكتل

يوجد حاليًا نوعان من الكتل محددتان لملفات المؤشر؛ التعليقات والصور. تبدو التعليقات كالتالي:


header:	20 رؤوس التعليقات هي 20 بايت
type:	0xfffe0001 نوع التعليق هو 0xfffe0001
subtype:	{ 1 (COPYRIGHT), 2 (LICENSE), 3 (OTHER) }
version:	1
length:	CARD32 طول البايت للسلسلة UTF-8
string:	LISTofCARD8 سلسلة UTF-8

تبدو الصور كالتالي:


header:	36 رؤوس الصور هي 36 بايت
type:	0xfffd0002 نوع الصورة هو 0xfffd0002
subtype:	CARD32 نوع الصورة الفرعي هو الحجم الاسمي
version:	1
width:	CARD32 يجب أن يكون أقل من أو يساوي 0x7fff
height:	CARD32 يجب أن يكون أقل من أو يساوي 0x7fff
xhot:	CARD32 يجب أن يكون أقل من أو يساوي العرض
yhot:	CARD32 يجب أن يكون أقل من أو يساوي الارتفاع
delay:	CARD32 التأخير بين إطارات الرسوم المتحركة بالمللي ثانية
pixels:	LISTofCARD32 بكسلات بتنسيق ARGB مكدسة

المواضيع (Themes)

تتبع Xcursor (في الغالب) مواصفات freedesktop.org لمواضيع الأيقونات. المسار الافتراضي للبحث الذي تستخدمه هو

~/.local/share/icons, ~/.icons, /usr/share/icons, /usr/share/pixmaps

داخل كل من هذه الدلائل، تبحث عن دليل باستخدام اسم الموضوع:

•
داخل دليل الموضوع، تبحث عن ملفات المؤشر في الدليل الفرعي "cursors".
تبحث Xcursor عن ملف محدد، يجب أن يكون أحد أسماء شكل المؤشر، على سبيل المثال، كما هو مستخدم في XcursorLibraryLoadImage أو XcursorLibraryShape.
•
إذا لم تجد ملف مؤشر مطابق في الدليل الفرعي "cursors"، فإن Xcursor تبحث بعد ذلك عن ملف "index.theme" في كل دليل موضوع للبحث عن مواضيع موروثة. تلك هي الأسطر بهذا التنسيق:
Inherits = theme-name

تستخدم Xcursor أول theme-name موروث، متجاهلة الآخرين الذين قد يكونون موجودين في ملف "index.theme" معين. إذا وجدت موضوعًا موروثًا، فإن Xcursor تبحث على طول المسار لاستخدامه أيضًا. تتجاهل Xcursor مفاتيح أخرى في ملف "index.theme"، بما في ذلك "Name" (أي، الاسم الذي قد يستخدمه تطبيق رسومي كـ اسم عرض).

يمكن سرد أكثر من theme-name واحد في سطر Inherits=. تنص مواصفات freedesktop.org على أن عناصر القائمة مفصولة بفواصل. تقبل Xcursor الفاصلة المنقوطة أيضًا، ولكنها تترجم كليهما إلى نقطتين رأسيتين (colon) عند البحث في المسار. تتوقع Xcursor سطر Inherits= واحد فقط؛ مواصفات freedesktop.org غير واضحة ما إذا كان يمكن السماح بمفاتيح متعددة.

إذا لم يتم تعيين موضوع، أو إذا لم يتم العثور على مؤشر للموضوع المحدد في أي مكان على طول المسار، تتحقق Xcursor من موضوع "default".

عندما تجد Xcursor ملف مؤشر، فإنها تتوقف عن البحث. إنها تستخدم دائمًا ملف المؤشر الأول الذي تم العثور عليه أثناء البحث على طول المسار.

أنواع البيانات

XcursorImage
يحتوي على صورة مؤشر واحدة في الذاكرة. كل بكسل في المؤشر هو قيمة 32 بت تحتوي على ARGB مع A في البايت الأعلى. 


typedef struct _XcursorImage {
    XcursorDim 	 size; 	/* حجم اسمي للمطابقة */
    XcursorDim 	 width; 	/* العرض الفعلي */
    XcursorDim 	 height; 	/* الارتفاع الفعلي */
    XcursorDim 	 xhot; 	/* موقع نقطة الاتصال x (يجب أن يكون داخل الصورة) */
    XcursorDim 	 yhot; 	/* موقع نقطة الاتصال y (يجب أن يكون داخل الصورة) */
    XcursorPixel 	*pixels; 	/* مؤشر إلى البكسلات */
} XcursorImage;
XcursorImages
يحتوي على هياكل XcursorImage متعددة. يتم تحريرها جميعًا عند تحرير XcursorImages في XcursorImagesDestroy. 


typedef struct _XcursorImages {
    int 	 nimage; 	/* عدد الصور */
    XcursorImage 	**images; 	/* مصفوفة من مؤشرات XcursorImage */
} XcursorImages;
XcursorCursors
يحتوي على كائنات مؤشر متعددة. يتم تحريرها جميعًا عند تحرير XcursorCursors. يتم حساب هذه المراجع بحيث يمكن لهياكل XcursorAnimate متعددة استخدام نفس XcursorCursors. 


typedef struct _XcursorCursors {
    Display 	*dpy; 	/* شاشة تحتوي على المؤشرات */
    int 	 ref; 	/* عداد المراجع */
    int 	 ncursor; 	/* عدد المؤشرات */
    Cursor 	*cursors; 	/* مصفوفة المؤشرات */
} XcursorCursors;
XcursorAnimate
يشير إلى مجموعة من المؤشرات وتسلسل داخل تلك المجموعة. قد تشير هياكل XcursorAnimate متعددة إلى نفس XcursorCursors؛ كل منها يحتفظ بمرجع يتم إزالته عند تحرير XcursorAnimate. 


typedef struct _XcursorAnimate {
    XcursorCursors 	*cursors; 	/* قائمة المؤشرات المراد استخدامها */
    int 	 sequence; 	/* المؤشر التالي */
} XcursorAnimate;
XcursorFile
توفر Xcursor واجهة برمجة تطبيقات مجردة للوصول إلى بيانات الملف. توفر Xcursor تنفيذًا لـ stdio لهذه الواجهة المجردة؛ التطبيقات حرة في إنشاء تطبيقات إضافية. توازي هذه الدوال دوال f الخاصة بـ stdio.

هذا الموقع غير مرتبط بـ Arch Linux ولا تابع له بأي شكل من الأشكال. المحتوى هو ترجمة فقط لصفحات دليل Arch المنشورة على man.archlinux.org.

بالقيم المتوقعة للحجج؛ تقوم دوال القراءة والكتابة بتبديل الحجج لكي تتطابق مع إصدارات POSIX. 


typedef struct _XcursorFile {
    void *closure;
    int (*read)  (XcursorFile *file, unsigned char *buf, int len);
    int (*write) (XcursorFile *file, unsigned char *buf, int len);
    int (*seek)  (XcursorFile *file, long offset, int whence);
};

	const char 	*name,

تبحث هذه الدوال في مسار المكتبة، وتحمل أول ملف تم العثور عليه بالحجم المطلوب size، باستخدام دالة خاصة (XcursorScanTheme) للعثور على السمة المناسبة:

  • إذا كان theme غير NULL، استخدمه.
  • إذا كان theme NULL، أو إذا لم يكن هناك تطابق للسمة المطلوبة، فاستخدم "default" كاسم للسمة.
  • إذا فشل كلا البحثين، فإن هذه الدوال ترجع NULL.
تختلف الدالتان بأكثر من عدد الصور المحملة:
  • XcursorLibraryLoadImage تستدعي XcursorFileLoadImage ولكن
  • XcursorLibraryLoadImages تستدعي XcursorFileLoadImages، وتقوم (في حالة النجاح) باستدعاء XcursorImagesSetName لربط name بالنتيجة.

HEME_CORE



  
Xcursor.theme_core

  
يُمكّن الثيمات لمؤشرات النواة إذا كانت المتغير البيئي
      true.





  

  
إذا لم يتم توفير المتغير البيئي، فإن Xcursor يحاول استخدام
      الخاصية Xcursor.theme_core.

  

  
يمكن للتطبيق تمكين أو تعطيل الثيمات باستخدام
    XcursorSetThemeCore.
