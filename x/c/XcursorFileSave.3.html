الاسم

Xcursor — مكتبة إدارة المؤشرات

الملخص

#include <X11/Xcursor/Xcursor.h>

الوصف

Xcursor هي مكتبة بسيطة مصممة للمساعدة في تحديد مواقع وتحميل المؤشرات. يمكن تحميل المؤشرات من ملفات أو ذاكرة. توجد مكتبة للمؤشرات الشائعة التي تقابل أسماء مؤشرات X القياسية. يمكن أن توجد المؤشرات بأحجام مختلفة وتقوم المكتبة تلقائيًا باختيار أفضل حجم.

نظرة عامة على الوظائف

تم بناء Xcursor على عدة طبقات؛ في الطبقة السفلية توجد التعليمات البرمجية التي يمكنها تحميل صور المؤشر من الملفات. فوق ذلك توجد طبقة تحدد مواقع ملفات المؤشر بناءً على مسار المكتبة والموضوع. في الأعلى توجد طبقة تبني المؤشرات إما من صورة تم تحميلها من ملف أو واحدة من مؤشرات X القياسية. عند استخدام الصور المحملة من ملفات، تفضل Xcursor استخدام طلب CreateCursor الخاص بامتداد Render إذا كان مدعومًا من خادم X. حيثما لم يكن مدعومًا، تقوم Xcursor بتعيين صورة المؤشر إلى مؤشر X قياسي وتستخدم طلب CreateCursor الأساسي.

ملفات المؤشرات

تحدد Xcursor تنسيقًا جديدًا للمؤشرات على القرص. كل ملف يحتوي على صورة مؤشر واحدة أو أكثر. كل صورة مؤشر موسومة بحجم اسمي حتى يمكن اختيار أفضل حجم تلقائيًا. يمكن تحميل مؤشرات متعددة بنفس الحجم الاسمي معًا؛ من المتوقع أن تستخدمها التطبيقات كسلسلة رسوم متحركة.

يتم تخزين ملفات المؤشرات كرأس يحتوي على جدول محتويات متبوعًا بسلسلة من الأجزاء. يشير رأس الملف إلى نوع، ونوع فرعي، وموضع في الملف لكل جزء. يبدو رأس الملف كالتالي:


magic:	CARD32 "Xcur" (0x58, 0x63, 0x75, 0x72)
header:	CARD32 بايت في هذا الرأس
version:	CARD32 رقم إصدار الملف
ntoc:	CARD32 عدد إدخالات جدول المحتويات
toc:	LISTofTOC جدول المحتويات

كل إدخال في جدول المحتويات يبدو كالتالي:


type:	CARD32 نوع الإدخال
subtype:	CARD32 تسمية خاصة بالنوع - حجم للصور
position:	CARD32 الموضع المطلق بالبايت لجدول المحتويات في الملف

كل جزء في الملف يحتوي على مجموعة من حقول الرأس الشائعة متبوعة بحقول إضافية خاصة بالنوع:


header:	CARD32 بايت في رأس الجزء (بما في ذلك الحقول الخاصة بالنوع)
type:	CARD32 يجب أن يتطابق مع النوع في TOC لهذا الجزء
subtype:	CARD32 يجب أن يتطابق مع النوع الفرعي في TOC لهذا الجزء
version:	CARD32 رقم إصدار لنوع هذا الجزء

يوجد حاليًا نوعان محددين لأجزاء ملفات المؤشر؛ التعليقات والصور. تبدو التعليقات كالتالي:


header:	20 رأس التعليقات هو 20 بايت
type:	0xfffe0001 نوع التعليق هو 0xfffe0001
subtype:	{ 1 (حقوق النشر)، 2 (الترخيص)، 3 (آخر)}
version:	1
length:	CARD32 طول البايت للسلسلة UTF-8
string:	LISTofCARD8 سلسلة UTF-8

تبدو الصور كالتالي:


header:	36 رؤوس الصور هي 36 بايت
type:	0xfffd0002 نوع الصورة هو 0xfffd0002
subtype:	CARD32 النوع الفرعي للصورة هو الحجم الاسمي
version:	1
width:	CARD32 يجب أن يكون أقل من أو يساوي 0x7fff
height:	CARD32 يجب أن يكون أقل من أو يساوي 0x7fff
xhot:	CARD32 يجب أن يكون أقل من أو يساوي العرض
yhot:	CARD32 يجب أن يكون أقل من أو يساوي الارتفاع
delay:	CARD32 التأخير بين إطارات الرسوم المتحركة بالمللي ثانية
pixels:	LISTofCARD32 بكسلات بتنسيق ARGB المعبأ

الموضوعات

تتبع Xcursor (في الغالب) مواصفات freedesktop.org لسمات الأيقونات. المسار الافتراضي للبحث الذي تستخدمه هو

~/.local/share/icons, ~/.icons, /usr/share/icons, /usr/share/pixmaps

داخل كل من هذه الدلائل، تبحث عن دليل باستخدام اسم الموضوع:

•
داخل دليل الموضوع، تبحث عن ملفات المؤشر في الدليل الفرعي "cursors".
تبحث Xcursor عن ملف محدد، يجب أن يكون أحد أسماء شكل المؤشر (shape names)، على سبيل المثال، كما هو مستخدم في XcursorLibraryLoadImage أو XcursorLibraryShape.
•
إذا لم تجد ملف مؤشر مطابق في الدليل الفرعي "cursors"، تبحث Xcursor بعد ذلك عن ملف "index.theme" في كل دليل موضوع للبحث عن الموضوعات الموروثة. تلك هي أسطر بهذا التنسيق:
Inherits = اسم-الموضوع

تستخدم Xcursor أول اسم-موضوع موروث، متجاهلة الآخرين الذين قد يكونون موجودين في ملف "index.theme" معين. إذا وجدت موضوعًا موروثًا، فإن Xcursor تبحث على طول المسار لاستخدامه أيضًا. تتجاهل Xcursor المفاتيح الأخرى في ملف "index.theme"، بما في ذلك "Name" (أي، الاسم الذي قد يستخدمه التطبيق الرسومي كـ اسم العرض).

يمكن سرد أكثر من اسم-موضوع واحد في سطر Inherits=. تنص مواصفات freedesktop.org على أن عناصر القائمة مفصولة بفواصل. تقبل Xcursor أيضًا فاصلة منقوطة، ولكنها تترجم كلاهما إلى نقطتين رأسيتين عند البحث في المسار. تتوقع Xcursor سطر Inherits= واحد فقط؛ مواصفات freedesktop.org غير واضحة ما إذا كانت المفاتيح المتعددة مسموح بها.

إذا لم يتم تعيين موضوع، أو إذا لم يتم العثور على مؤشر للموضوع المحدد في أي مكان على طول المسار، تتحقق Xcursor من الموضوع "default".

عندما تجد Xcursor ملف مؤشر، فإنها تتوقف عن البحث. تستخدم دائمًا أول ملف مؤشر تم العثور عليه أثناء البحث على طول المسار.

أنواع البيانات

XcursorImage
يحتوي على صورة مؤشر واحدة في الذاكرة. كل بكسل في المؤشر هو قيمة 32 بت تحتوي على ARGB مع A في البايت الأعلى. 


typedef struct _XcursorImage {
    XcursorDim 	 size; 	/* حجم اسمي للمطابقة */
    XcursorDim 	 width; 	/* العرض الفعلي */
    XcursorDim 	 height; 	/* الارتفاع الفعلي */
    XcursorDim 	 xhot; 	/* نقطة الاتصال x (يجب أن تكون داخل الصورة) */
    XcursorDim 	 yhot; 	/* نقطة الاتصال y (يجب أن تكون داخل الصورة) */
    XcursorPixel 	*pixels; 	/* مؤشر إلى البكسلات */
} XcursorImage;
XcursorImages
يحتوي على هياكل XcursorImage متعددة. يتم تحريرها جميعًا عند تحرير XcursorImages في XcursorImagesDestroy. 


typedef struct _XcursorImages {
    int 	 nimage; 	/* عدد الصور */
    XcursorImage 	**images; 	/* مصفوفة من مؤشرات XcursorImage */
} XcursorImages;
XcursorCursors
يحتوي على كائنات مؤشر متعددة. يتم تحريرها جميعًا عند تحرير XcursorCursors. هذه لديها عداد مراجع بحيث يمكن لهياكل XcursorAnimate متعددة استخدام نفس XcursorCursors. 


typedef struct _XcursorCursors {
    Display 	*dpy; 	/* شاشة تحتفظ بالمؤشرات */
    int 	 ref; 	/* عداد المراجع */
    int 	 ncursor; 	/* عدد المؤشرات */
    Cursor 	*cursors; 	/* مصفوفة من المؤشرات */
} XcursorCursors;
XcursorAnimate
يشير إلى مجموعة من المؤشرات وتسلسل داخل تلك المجموعة. قد تشير هياكل XcursorAnimate متعددة إلى نفس XcursorCursors؛ كل منها يحتفظ بمرجع يتم إزالته عند تحرير XcursorAnimate. 


typedef struct _XcursorAnimate {
    XcursorCursors 	*cursors; 	/* قائمة المؤشرات المراد استخدامها */
    int 	 sequence; 	/* أي مؤشر هو التالي */
} XcursorAnimate;
XcursorFile
توفر Xcursor واجهة برمجة تطبيقات مجردة للوصول إلى بيانات الملف. توفر Xcursor تنفيذًا قياسيًا لواجهة برمجة التطبيقات المجردة هذه؛ التطبيقات حرة في إنشاء تطبيقات إضافية. هذه الوظائف توازي وظائف stdio.
ns في القيمة المرجعة وقيم الوسائط المتوقعة؛ تقلب دالتا القراءة والكتابة الوسائط لكي تتوافق مع إصدارات POSIX. 


typedef struct _XcursorFile {
    void *closure;
    int (*read)  (XcursorFile *file, unsigned char *buf, int len);
    int (*write) (XcursorFile *file, unsigned char *buf, int len);
    int (*seek)  (XcursorFile *file, long offset, int whence);
};

قراءة صور المكتبة

const char 	*name,

int 	 size)
XcursorImages *XcursorLibraryLoadImages (
	const char 	*name,
	const char 	*theme,
	int 	 size)

تبحث هذه الدوال في مسار المكتبة، وتقوم بتحميل أول ملف يتم العثور عليه بالحجم المطلوب، باستخدام دالة خاصة (XcursorScanTheme) للعثور على النسق المناسب:

  • إذا كان theme ليس NULL، استخدمه.
  • إذا كان theme هو NULL، أو إذا لم يكن هناك تطابق للنسق المطلوب، استخدم "default" لاسم النسق.
  • إذا فشل كلا البحثين، فإن هذه الدوال ترجع NULL.
تختلف الدالتان عن عدد الصور المحملة:
  • XcursorLibraryLoadImage تستدعي XcursorFileLoadImage ولكن
  • XcursorLibraryLoadImages تستدعي XcursorFileLoadImages، وتقوم (عند النجاح) باستدعاء XcursorImagesSetName لربط name بالنتيجة.
