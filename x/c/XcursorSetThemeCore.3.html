XcursorSetThemeCore(3)

الحزمة: libxcursor

التراخيص: HPND-sell-variant

XCURSOR(3) X Version 11 XCURSOR(3)

الاسم

Xcursor - مكتبة إدارة المؤشرات

الملخص

#include <X11/Xcursor/Xcursor.h>

الوصف

Xcursor هي مكتبة بسيطة مصممة للمساعدة في تحديد مواقع المؤشرات وتحميلها. يمكن تحميل المؤشرات من الملفات أو الذاكرة. توجد مكتبة للمؤشرات الشائعة والتي تتوافق مع أسماء مؤشرات X القياسية. يمكن أن توجد المؤشرات بأحجام متعددة وتقوم المكتبة تلقائيًا باختيار أفضل حجم.

نظرة عامة وظيفية

تم بناء Xcursor في طبقات قليلة؛ في الطبقة السفلية يوجد كود يمكنه تحميل صور المؤشرات من الملفات. فوق ذلك توجد طبقة تحدد مواقع ملفات المؤشرات بناءً على مسار المكتبة والموضوع. في الأعلى توجد طبقة تبني المؤشرات إما من صورة تم تحميلها من ملف أو أحد مؤشرات X القياسية. عند استخدام صور تم تحميلها من ملفات، تفضل Xcursor استخدام طلب الإنشاء CreateCursor لامتداد Render إذا كان مدعومًا من خادم X. في الحالات التي لا يكون فيها مدعومًا، تقوم Xcursor بربط صورة المؤشر بمؤشر X قياسي وتستخدم طلب الإنشاء CreateCursor الأساسي.

ملفات المؤشرات

تحدد Xcursor تنسيقًا جديدًا للمؤشرات على القرص. يحتوي كل ملف على صورة مؤشر واحدة أو أكثر. يتم تمييز كل صورة مؤشر بحجم اسمي بحيث يمكن تحديد أفضل حجم تلقائيًا. يمكن تحميل مؤشرات متعددة بنفس الحجم الاسمي معًا؛ من المتوقع أن تستخدم التطبيقات هذه كمتتالية رسوم متحركة.

يتم تخزين ملفات المؤشرات كرأس يحتوي على جدول محتويات يليه تسلسل من المقاطع. يشير جدول المحتويات إلى النوع والنوع الفرعي والموضع في الملف لكل مقطع. يبدو رأس الملف كالتالي:


magic:	CARD32 "Xcur" (0x58, 0x63, 0x75, 0x72)
header:	CARD32 عدد البايتات في هذا الرأس
version:	CARD32 رقم إصدار الملف
ntoc:	CARD32 عدد إدخالات جدول المحتويات
toc:	LISTofTOC جدول المحتويات

كل إدخال في جدول المحتويات يبدو كالتالي:


type:	CARD32 نوع الإدخال
subtype:	CARD32 تسمية خاصة بالنوع - الحجم للصور
position:	CARD32 الموضع المطلق بالبايت للمقطع في الملف

كل مقطع في الملف يحتوي على مجموعة من حقول الرأس الشائعة تليها حقول إضافية خاصة بالنوع:


header:	CARD32 عدد البايتات في رأس المقطع (بما في ذلك الحقول الخاصة بالنوع)
type:	CARD32 يجب أن يتطابق مع النوع في TOC لهذا المقطع
subtype:	CARD32 يجب أن يتطابق مع النوع الفرعي في TOC لهذا المقطع
version:	CARD32 رقم إصدار لهذا النوع من المقطع

يوجد حاليًا نوعان من المقاطع معرفان لملفات المؤشرات؛ التعليقات والصور. تبدو التعليقات كالتالي:


header:	20 رؤوس التعليقات بعرض 20 بايت
type:	0xfffe0001 نوع التعليق هو 0xfffe0001
subtype:	{ 1 (COPYRIGHT), 2 (LICENSE), 3 (OTHER) }
version:	1
length:	CARD32 طول السلسلة النصية UTF-8 بالبايت
string:	LISTofCARD8 سلسلة نصية UTF-8

تبدو الصور كالتالي:


header:	36 رؤوس الصور بعرض 36 بايت
type:	0xfffd0002 نوع الصورة هو 0xfffd0002
subtype:	CARD32 النوع الفرعي للصورة هو الحجم الاسمي
version:	1
width:	CARD32 يجب أن يكون أقل من أو يساوي 0x7fff
height:	CARD32 يجب أن يكون أقل من أو يساوي 0x7fff
xhot:	CARD32 يجب أن يكون أقل من أو يساوي العرض
yhot:	CARD32 يجب أن يكون أقل من أو يساوي الارتفاع
delay:	CARD32 التأخير بين إطارات الرسوم المتحركة بالمللي ثانية
pixels:	LISTofCARD32 بكسلات بصيغة ARGB المعبأة

المواضيع

تتبع Xcursor (في الغالب) مواصفات freedesktop.org لموضوعات الأيقونات. المسار الافتراضي للبحث الذي تستخدمه هو

~/.local/share/icons, ~/.icons, /usr/share/icons, /usr/share/pixmaps

داخل كل من هذه الأدلة، تبحث عن دليل باستخدام اسم الموضوع:

•
داخل دليل الموضوع، تبحث عن ملفات المؤشرات في الدليل الفرعي "cursors".
تبحث Xcursor عن ملف محدد، يجب أن يكون أحد أسماء shape الخاصة بالمؤشر، على سبيل المثال، كما هو مستخدم في XcursorLibraryLoadImage أو XcursorLibraryShape.
•
إذا لم تجد ملف مؤشر مطابق في الدليل الفرعي "cursors"، تبحث Xcursor بعد ذلك عن ملف "index.theme" في كل دليل موضوع للبحث عن مواضيع موروثة. هذه هي أسطر بهذا الشكل:
Inherits = اسم-الموضوع

تستخدم Xcursor أول اسم-الموضوع موروث، وتتجاهل الآخرين الذين قد يكونون موجودين في ملف "index.theme" معين. إذا وجدت موضوعًا موروثًا، فإن Xcursor تبحث على طول المسار لاستخدامه أيضًا. تتجاهل Xcursor مفاتيح أخرى في ملف "index.theme"، بما في ذلك "Name" (أي، الاسم الذي قد يستخدمه تطبيق رسومي بمثابة اسم العرض).

قد يتم سرد أكثر من اسم-الموضوع واحد في سطر Inherits=. تنص مواصفات freedesktop.org على أن عناصر القائمة مفصولة بفواصل. تقبل Xcursor أيضًا الفاصلة المنقوطة، ولكنها تترجم كليهما إلى نقطتين رأسيتين عند البحث في المسار. تتوقع Xcursor سطر Inherits= واحد فقط؛ مواصفات freedesktop.org غير واضحة ما إذا كان يمكن السماح بمفاتيح متعددة.

إذا لم يتم تعيين موضوع، أو إذا لم يتم العثور على مؤشر للموضوع المحدد في أي مكان على طول المسار، فإن Xcursor تتحقق من الموضوع "افتراضي".

عندما تجد Xcursor ملف مؤشر، تتوقف عن البحث. تستخدم دائمًا ملف المؤشر الأول الذي تم العثور عليه أثناء البحث على طول المسار.

أنواع البيانات

XcursorImage
يحتوي على صورة مؤشر واحدة في الذاكرة. كل بكسل في المؤشر هو قيمة 32 بت تحتوي على ARGB مع A في البايت العالي. 


typedef struct _XcursorImage {
    XcursorDim 	 size; 	/* حجم اسمي للمطابقة */
    XcursorDim 	 width; 	/* العرض الفعلي */
    XcursorDim 	 height; 	/* الارتفاع الفعلي */
    XcursorDim 	 xhot; 	/* نقطة الاهتمام x (يجب أن تكون داخل الصورة) */
    XcursorDim 	 yhot; 	/* نقطة الاهتمام y (يجب أن تكون داخل الصورة) */
    XcursorPixel 	*pixels; 	/* مؤشر للبكسلات */
} XcursorImage;
XcursorImages
يحتوي على هياكل XcursorImage متعددة. يتم تحريرها جميعًا عند تحرير XcursorImages في XcursorImagesDestroy. 


typedef struct _XcursorImages {
    int 	 nimage; 	/* عدد الصور */
    XcursorImage 	**images; 	/* مصفوفة من مؤشرات XcursorImage */
} XcursorImages;
XcursorCursors
يحتوي على كائنات مؤشر متعددة. يتم تحريرها جميعًا عند تحرير XcursorCursors. هذه تحتوي على عداد مراجع بحيث يمكن لهياكل XcursorAnimate متعددة استخدام نفس XcursorCursors. 


typedef struct _XcursorCursors {
    Display 	*dpy; 	/* العرض الذي يحتوي على المؤشرات */
    int 	 ref; 	/* عداد المراجع */
    int 	 ncursor; 	/* عدد المؤشرات */
    Cursor 	*cursors; 	/* مصفوفة المؤشرات */
} XcursorCursors;
XcursorAnimate
يشير إلى مجموعة من المؤشرات وتسلسل ضمن هذه المجموعة. قد تشير هياكل XcursorAnimate متعددة إلى نفس XcursorCursors؛ كل منها يحتفظ بمرجع يتم إزالته عند تحرير XcursorAnimate. 


typedef struct _XcursorAnimate {
    XcursorCursors 	*cursors; 	/* قائمة المؤشرات المراد استخدامها */
    int 	 sequence; 	/* أي مؤشر هو التالي */
} XcursorAnimate;
XcursorFile
توفر Xcursor واجهة برمجة تطبيقات مجردة للوصول إلى بيانات الملف. توفر Xcursor تطبيقًا قياسيًا لهذه الواجهة المجردة؛ التطبيقات حرة في إنشاء تطبيقات إضافية. توازي هذه الوظائف وظائف stdio. 


typedef struct _XcursorFile {
    void 	*closure; 	/* بيانات خاصة بالتطبيق */
    int 	(*read)(XcursorFile *file, char *buf, int len);
    int 	(*write)(XcursorFile *file, const char *buf, int len);
    int 	(*seek)(XcursorFile *file, long offset, int whence);
    int 	(*close)(XcursorFile *file);
} XcursorFile;
