الاسم

Xcursor - مكتبة إدارة المؤشرات

الملخص

#include <X11/Xcursor/Xcursor.h>

الوصف

Xcursor هي مكتبة بسيطة مصممة للمساعدة في تحديد مواقع المؤشرات وتحميلها. يمكن تحميل المؤشرات من ملفات أو ذاكرة. توجد مكتبة للمؤشرات الشائعة التي ترتبط بأسماء المؤشرات القياسية لـ X. يمكن أن توجد المؤشرات بأحجام متعددة وتقوم المكتبة تلقائيًا باختيار أفضل حجم.

نظرة عامة وظيفية

تم بناء Xcursor على طبقات قليلة؛ في الطبقة السفلية، يوجد رمز يمكنه تحميل صور المؤشرات من الملفات. فوق ذلك، توجد طبقة تحدد مواقع ملفات المؤشرات بناءً على مسار المكتبة والموضوع (theme). في القمة، توجد طبقة تنشئ مؤشرات إما من صورة تم تحميلها من ملف أو أحد مؤشرات X القياسية. عند استخدام الصور المحملة من الملفات، تفضل Xcursor استخدام طلب إنشاء مؤشر امتداد Render إذا كان مدعومًا من خادم X. حيثما لم يكن مدعومًا، تقوم Xcursor بربط صورة المؤشر بمؤشر X قياسي وتستخدم طلب إنشاء مؤشر أساسي.

ملفات المؤشرات

تحدد Xcursor تنسيقًا جديدًا للمؤشرات على القرص. يحتوي كل ملف على صورة مؤشر واحدة أو أكثر. يتم وضع علامة على كل صورة مؤشر بحجم اسمي بحيث يمكن اختيار أفضل حجم تلقائيًا. يمكن تحميل مؤشرات متعددة بنفس الحجم الاسمي معًا؛ من المتوقع أن تستخدمها التطبيقات كسلسلة رسوم متحركة.

يتم تخزين ملفات المؤشرات كرأس يحتوي على جدول محتويات متبوعًا بسلسلة من الأجزاء (chunks). يشير جدول المحتويات إلى النوع، والنوع الفرعي، والموضع في الملف لكل جزء. يبدو رأس الملف كالتالي:


magic:	CARD32 "Xcur" (0x58, 0x63, 0x75, 0x72)
header:	CARD32 عدد البايتات في هذا الرأس
version:	CARD32 رقم إصدار الملف
ntoc:	CARD32 عدد إدخالات جدول المحتويات
toc:	LISTofTOC جدول المحتويات

كل إدخال في جدول المحتويات يبدو كالتالي:


type:	CARD32 نوع الإدخال
subtype:	CARD32 تسمية خاصة بالنوع - حجم للصور
position:	CARD32 الموضع المطلق للبايت لجدول في الملف

كل جزء في الملف له مجموعة من حقول الرأس المشتركة متبوعة بحقول إضافية خاصة بالنوع:


header:	CARD32 عدد البايتات في رأس الجزء (بما في ذلك الحقول الخاصة بالنوع)
type:	CARD32 يجب أن يتطابق مع النوع في TOC لهذا الجزء
subtype:	CARD32 يجب أن يتطابق مع النوع الفرعي في TOC لهذا الجزء
version:	CARD32 رقم إصدار لهذا النوع من الأجزاء

هناك حاليًا نوعان محددين للأجزاء في ملفات المؤشرات؛ التعليقات والصور. تبدو التعليقات كالتالي:


header:	20 رؤوس التعليقات بحجم 20 بايت
type:	0xfffe0001 نوع التعليق هو 0xfffe0001
subtype:	{ 1 (حقوق النشر)، 2 (الترخيص)، 3 (أخرى) }
version:	1
length:	CARD32 طول البايت للسلسلة UTF-8
string:	LISTofCARD8 سلسلة UTF-8

تبدو الصور كالتالي:


header:	36 رؤوس الصور بحجم 36 بايت
type:	0xfffd0002 نوع الصورة هو 0xfffd0002
subtype:	CARD32 النوع الفرعي للصورة هو الحجم الاسمي
version:	1
width:	CARD32 يجب أن يكون أقل من أو يساوي 0x7fff
height:	CARD32 يجب أن يكون أقل من أو يساوي 0x7fff
xhot:	CARD32 يجب أن يكون أقل من أو يساوي العرض
yhot:	CARD32 يجب أن يكون أقل من أو يساوي الارتفاع
delay:	CARD32 التأخير بين إطارات الرسوم المتحركة بالمللي ثانية
pixels:	LISTofCARD32 وحدات بكسل بصيغة ARGB مضغوطة

المواضيع (Themes)

تتبع Xcursor (في الغالب) مواصفات freedesktop.org لمواضيع الأيقونات. المسار الافتراضي للبحث الذي تستخدمه هو

~/.local/share/icons, ~/.icons, /usr/share/icons, /usr/share/pixmaps

داخل كل من هذه الدلائل، تبحث عن دليل باستخدام اسم الموضوع:

•
داخل دليل الموضوع، تبحث عن ملفات المؤشرات في الدليل الفرعي "cursors".
تبحث Xcursor عن ملف محدد، والذي يجب أن يكون أحد أسماء شكل المؤشر، على سبيل المثال، كما هو مستخدم في XcursorLibraryLoadImage أو XcursorLibraryShape.
•
إذا لم تجد ملف مؤشر مطابق في الدليل الفرعي "cursors"، تبحث Xcursor بعد ذلك عن ملف "index.theme" في كل دليل موضوع للبحث عن مواضيع موروثة. هذه هي الأسطر بهذا الشكل:
Inherits = theme-name

تستخدم Xcursor أول theme-name موروث، متجاهلة الآخرين الذين قد يكونون موجودين في ملف "index.theme" معين. إذا وجدت موضوعًا موروثًا، فإن Xcursor تبحث على طول المسار لاستخدامه أيضًا. تتجاهل Xcursor مفاتيح أخرى في ملف "index.theme"، بما في ذلك "Name" (أي، الاسم الذي قد يستخدمه التطبيق الرسومي كاسم عرض).

يمكن سرد أكثر من theme-name واحد في سطر Inherits=. تنص مواصفات freedesktop.org على أن عناصر القائمة مفصولة بفواصل. تقبل Xcursor أيضًا الفاصلة المنقوطة، لكنها تترجم كليهما إلى نقطتين رأسيتين عند البحث في المسار. تتوقع Xcursor سطرًا واحدًا فقط لـ Inherits=؛ مواصفات freedesktop.org غير واضحة بشأن ما إذا كان يمكن السماح بمفاتيح متعددة.

إذا لم يتم تعيين أي موضوع، أو إذا لم يتم العثور على مؤشر للموضوع المحدد في أي مكان على طول المسار، فإن Xcursor تتحقق من موضوع "default".

عندما تجد Xcursor ملف مؤشر، تتوقف عن البحث. إنها تستخدم دائمًا أول ملف مؤشر تم العثور عليه أثناء البحث على طول المسار.

أنواع البيانات

XcursorImage
تحتوي على صورة مؤشر واحدة في الذاكرة. كل بكسل في المؤشر هو قيمة 32 بت تحتوي على ARGB مع A في البايت الأعلى. 


typedef struct _XcursorImage {
    XcursorDim 	 size; 	/* حجم اسمي للمطابقة */
    XcursorDim 	 width; 	/* العرض الفعلي */
    XcursorDim 	 height; 	/* الارتفاع الفعلي */
    XcursorDim 	 xhot; 	/* نقطة ساخنة x (يجب أن تكون داخل الصورة) */
    XcursorDim 	 yhot; 	/* نقطة ساخنة y (يجب أن تكون داخل الصورة) */
    XcursorPixel 	*pixels; 	/* مؤشر إلى وحدات البكسل */
} XcursorImage;
XcursorImages
تحتوي على هياكل XcursorImage متعددة. يتم تحريرها جميعًا عند تحرير XcursorImages في XcursorImagesDestroy. 


typedef struct _XcursorImages {
    int 	 nimage; 	/* عدد الصور */
    XcursorImage 	**images; 	/* مصفوفة من مؤشرات XcursorImage */
} XcursorImages;
XcursorCursors
تحتوي على كائنات Cursor متعددة. يتم تحريرها جميعًا عند تحرير XcursorCursors. هذه مرجعية العد (reference counted) بحيث يمكن لهياكل XcursorAnimate متعددة استخدام نفس XcursorCursors. 


typedef struct _XcursorCursors {
    Display 	*dpy; 	/* العرض الذي يحتوي على المؤشرات */
    int 	 ref; 	/* عداد المراجع */
    int 	 ncursor; 	/* عدد المؤشرات */
    Cursor 	*cursors; 	/* مصفوفة المؤشرات */
} XcursorCursors;
XcursorAnimate
تشير إلى مجموعة من المؤشرات وتسلسل داخل تلك المجموعة. قد تشير هياكل XcursorAnimate متعددة إلى نفس XcursorCursors؛ يحتفظ كل منها بمرجع يتم إزالته عند تحرير XcursorAnimate. 


typedef struct _XcursorAnimate {
    XcursorCursors 	*cursors; 	/* قائمة المؤشرات المراد استخدامها */
    int 	 sequence; 	/* أي مؤشر هو التالي */
} XcursorAnimate;
XcursorFile
توفر Xcursor واجهة برمجة تطبيقات مجردة للوصول إلى بيانات الملف. توفر Xcursor تطبيق stdio لهذه الواجهة المجردة؛ التطبيقات حرة في إنشاء تطبيقات إضافية. هذه الوظائف موازية لـ stdio

هذا الموقع غير مرتبط بـ Arch Linux ولا تابع له بأي شكل من الأشكال. المحتوى هو ترجمة فقط لصفحات دليل Arch المنشورة على man.archlinux.org.

وظائف في قيمة الإرجاع وقيم الوسائط المتوقعة؛ تقوم وظائف القراءة والكتابة بتبديل الوسائط لكي تتطابق مع إصدارات POSIX. 


typedef struct _XcursorFile {
    void *closure;
    int (*read)  (XcursorFile *file, unsigned char *buf, int len);
    int (*write) (XcursorFile *file, unsigned char *buf, int len);
    int (*seek)  (XcursorFile *file, long offset, int whence);
};

تبحث هذه الوظائف في مسار المكتبة، وتحمّل أول ملف تم العثور عليه بالحجم المطلوب size، باستخدام وظيفة خاصة (XcursorScanTheme) للعثور على السمة المناسبة:

  • إذا كان theme ليس NULL، فاستخدمه.
  • إذا كان theme NULL، أو إذا لم يكن هناك تطابق للسمة المطلوبة، فاستخدم "default" لاسم السمة.
  • إذا فشل كلا البحثين، فإن هذه الوظائف ترجع NULL.
تختلف الوظيفتان بأكثر من عدد الصور المحملة:
  • XcursorLibraryLoadImage تستدعي XcursorFileLoadImage ولكن
  • XcursorLibraryLoadImages تستدعي XcursorFileLoadImages، وتقوم (عند النجاح) باستدعاء XcursorImagesSetName لربط name بالنتيجة.
