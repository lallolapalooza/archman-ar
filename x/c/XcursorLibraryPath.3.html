XCURSOR(3) الإصدار 11 من X XCURSOR(3)

الاسم

Xcursor - مكتبة إدارة المؤشرات

الملخص

#include <X11/Xcursor/Xcursor.h>

الوصف

Xcursor هي مكتبة بسيطة مصممة للمساعدة في تحديد وتحميل المؤشرات. يمكن تحميل المؤشرات من ملفات أو ذاكرة. توجد مكتبة من المؤشرات الشائعة التي تتوافق مع أسماء مؤشرات X القياسية. يمكن أن توجد المؤشرات بأحجام متعددة والمكتبة تختار تلقائيًا أفضل حجم.

نظرة عامة وظيفية

تم بناء Xcursor على طبقات قليلة؛ في الطبقة السفلية يوجد كود يمكنه تحميل صور المؤشرات من الملفات. فوق ذلك توجد طبقة تحدد ملفات المؤشرات بناءً على مسار المكتبة والموضوع. في الأعلى توجد طبقة تبني المؤشرات إما من صورة تم تحميلها من ملف أو أحد مؤشرات X القياسية. عند استخدام الصور المحملة من الملفات، تفضل Xcursor استخدام طلب إنشاء المؤشر لامتداد Render إذا كان مدعومًا بواسطة خادم X. حيث لا يكون مدعومًا، تقوم Xcursor بتعيين صورة المؤشر إلى مؤشر X قياسي وتستخدم طلب إنشاء المؤشر الأساسي.

ملفات المؤشرات

تحدد Xcursor تنسيقًا جديدًا للمؤشرات على القرص. كل ملف يحتوي على صورة مؤشر واحدة أو أكثر. كل صورة مؤشر موسومة بحجم اسمي بحيث يمكن اختيار أفضل حجم تلقائيًا. يمكن تحميل مؤشرات متعددة بنفس الحجم الاسمي معًا؛ يتوقع من التطبيقات استخدامها كسلسلة رسوم متحركة.

تُخزن ملفات المؤشرات كتنسيق رأس يحتوي على جدول محتويات متبوعًا بسلسلة من الكتل. يشير جدول المحتويات إلى النوع والنوع الفرعي والموضع في الملف لكل كتلة. رأس الملف يبدو كالتالي:


magic:	CARD32 "Xcur" (0x58, 0x63, 0x75, 0x72)
header:	CARD32 بايت في هذا الرأس
version:	CARD32 رقم إصدار الملف
ntoc:	CARD32 عدد إدخالات جدول المحتويات
toc:	LISTofTOC جدول المحتويات

كل إدخال في جدول المحتويات يبدو كالتالي:


type:	CARD32 نوع الإدخال
subtype:	CARD32 تسمية خاصة بالنوع - حجم للصور
position:	CARD32 موضع البايت المطلق للجدول في الملف

كل كتلة في الملف لها مجموعة من حقول الرأس المشتركة متبوعة بحقول إضافية خاصة بالنوع:


header:	CARD32 بايت في رأس الكتلة (بما في ذلك الحقول الخاصة بالنوع)
type:	CARD32 يجب أن تتطابق مع النوع في جدول المحتويات لهذه الكتلة
subtype:	CARD32 يجب أن تتطابق مع النوع الفرعي في جدول المحتويات لهذه الكتلة
version:	CARD32 رقم إصدار لنوع هذه الكتلة

يوجد حاليًا نوعان من الكتل محددين لملفات المؤشرات؛ التعليقات والصور. تبدو التعليقات كالتالي:


header:	20 رؤوس التعليقات هي 20 بايت
type:	0xfffe0001 نوع التعليق هو 0xfffe0001
subtype:	{ 1 (حقوق النشر)، 2 (الترخيص)، 3 (آخر)}
version:	1
length:	CARD32 طول البايت للسلسلة UTF-8
string:	LISTofCARD8 سلسلة UTF-8

تبدو الصور كالتالي:


header:	36 رؤوس الصور هي 36 بايت
type:	0xfffd0002 نوع الصورة هو 0xfffd0002
subtype:	CARD32 نوع الصورة الفرعي هو الحجم الاسمي
version:	1
width:	CARD32 يجب أن يكون أقل من أو يساوي 0x7fff
height:	CARD32 يجب أن يكون أقل من أو يساوي 0x7fff
xhot:	CARD32 يجب أن يكون أقل من أو يساوي العرض
yhot:	CARD32 يجب أن يكون أقل من أو يساوي الارتفاع
delay:	CARD32 التأخير بين إطارات الرسوم المتحركة بالمللي ثانية
pixels:	LISTofCARD32 بكسلات بتنسيق ARGB مجمعة

المواضيع

تتبع Xcursor (في الغالب) مواصفات freedesktop.org لرموز المواضيع. المسار الافتراضي للبحث الذي تستخدمه هو

~/.local/share/icons, ~/.icons, /usr/share/icons, /usr/share/pixmaps

داخل كل من هذه الدلائل، تبحث عن دليل باستخدام اسم الموضوع:

•
داخل دليل الموضوع، تبحث عن ملفات المؤشرات في الدليل الفرعي "cursors".
تبحث Xcursor عن ملف معين، والذي يجب أن يكون أحد أسماء شكل المؤشر، على سبيل المثال، كما هو مستخدم في XcursorLibraryLoadImage أو XcursorLibraryShape.
•
إذا لم تجد ملف مؤشر مطابق في الدليل الفرعي "cursors"، فإن Xcursor تبحث بعد ذلك عن ملف "index.theme" في كل دليل موضوع للبحث عن المواضيع الموروثة. تلك هي أسطر بهذا التنسيق:
Inherits = اسم-الموضوع

تستخدم Xcursor اسم الموضوع الموروث الأول، متجاهلة الآخرين الذين قد يكونون موجودين في ملف "index.theme" معين. إذا وجدت موضوعًا موروثًا، فإن Xcursor تبحث على طول المسار لاستخدامه أيضًا. تتجاهل Xcursor مفاتيح أخرى في ملف "index.theme"، بما في ذلك "Name" (أي، الاسم الذي قد يستخدمه التطبيق الرسومي كـ اسم العرض).

يمكن سرد أكثر من اسم-الموضوع واحد في سطر Inherits=. تنص مواصفات freedesktop.org على أن عناصر القائمة مفصولة بفواصل. تقبل Xcursor أيضًا الفاصلة المنقوطة، ولكنها تترجم كلاهما إلى نقطتين رأسيتين عند البحث في المسار. تتوقع Xcursor سطر Inherits= واحد فقط؛ مواصفات freedesktop.org غير واضحة بشأن ما إذا كانت المفاتيح المتعددة مسموح بها.

إذا لم يتم تعيين موضوع، أو إذا لم يتم العثور على مؤشر للموضوع المحدد في أي مكان على طول المسار، فإن Xcursor تتحقق من موضوع "الافتراضي".

عندما تجد Xcursor ملف مؤشر، فإنها تتوقف عن البحث. إنها تستخدم دائمًا ملف المؤشر الأول الذي تم العثور عليه أثناء البحث على طول المسار.

أنواع البيانات

XcursorImage
يحمل صورة مؤشر واحدة في الذاكرة. كل بكسل في المؤشر هو قيمة 32 بت تحتوي على ARGB مع A في البايت العالي. 


typedef struct _XcursorImage {
    XcursorDim 	 size; 	/* الحجم الاسمي للمطابقة */
    XcursorDim 	 width; 	/* العرض الفعلي */
    XcursorDim 	 height; 	/* الارتفاع الفعلي */
    XcursorDim 	 xhot; 	/* نقطة الاتصال x (يجب أن تكون داخل الصورة) */
    XcursorDim 	 yhot; 	/* نقطة الاتصال y (يجب أن تكون داخل الصورة) */
    XcursorPixel 	*pixels; 	/* مؤشر إلى البكسلات */
} XcursorImage;
XcursorImages
يحمل هياكل XcursorImage متعددة. يتم تحريرها جميعًا عند تحرير XcursorImages في XcursorImagesDestroy. 


typedef struct _XcursorImages {
    int 	 nimage; 	/* عدد الصور */
    XcursorImage 	**images; 	/* مصفوفة مؤشرات XcursorImage */
} XcursorImages;
XcursorCursors
يحمل كائنات المؤشر المتعددة. يتم تحريرها جميعًا عند تحرير XcursorCursors. هذه مرجع مرقم بحيث يمكن لهياكل XcursorAnimate المتعددة استخدام نفس XcursorCursors. 


typedef struct _XcursorCursors {
    Display 	*dpy; 	/* العرض الذي يحمل المؤشرات */
    int 	 ref; 	/* عدد المراجع */
    int 	 ncursor; 	/* عدد المؤشرات */
    Cursor 	*cursors; 	/* مصفوفة المؤشرات */
} XcursorCursors;
XcursorAnimate
يشير إلى مجموعة من المؤشرات وتسلسل داخل تلك المجموعة. قد تشير هياكل XcursorAnimate متعددة إلى نفس XcursorCursors؛ كل منها يحتفظ بمرجع يتم إزالته عند تحرير XcursorAnimate. 


typedef struct _XcursorAnimate {
    XcursorCursors 	*cursors; 	/* قائمة المؤشرات للاستخدام */
    int 	 sequence; 	/* أي مؤشر هو التالي */
} XcursorAnimate;
XcursorFile
توفر Xcursor واجهة برمجة تطبيقات مجردة للوصول إلى بيانات الملف. توفر Xcursor تطبيق stdio لهذه الواجهة المجردة؛ التطبيقات حرة في إنشاء تطبيقات إضافية. هذه الوظائف توازي وظائف stdio.

هذا الموقع غير مرتبط بـ Arch Linux ولا تابع له بأي شكل من الأشكال. المحتوى هو ترجمة فقط لصفحات دليل Arch المنشورة على man.archlinux.org.

سيحتوي العديد من المؤشرات على الفهرس.

	XcursorComments 	*comments)

البيئة

يمكن استخدام متغيرات البيئة لتجاوز إعدادات الموارد، والتي بدورها تتجاوز القيم الافتراضية المضمنة.

بعض متغيرات البيئة التي تتعرف عليها Xcursor هي قيم منطقية، يتم تحديدها على النحو التالي:

صحيح لـ "t" أو "1" أو "y" أو "on"
خطأ لـ "f" أو "0" أو "n" أو "off"

تتجاهل Xcursor القيم الأخرى لهذه القيم المنطقية.

HOME
تفسر Xcursor "~" في قائمة البحث على أنها الدليل الرئيسي، مع استخدام هذا المتغير بدلاً من قاعدة بيانات كلمات المرور.
XCURSOR_ANIM
إذا كان العرض يدعم طلب Render CreateCursor، وتم تمكين ميزة Render، فقم بتعطيل المؤشرات المتحركة إذا كان متغير البيئة false.
إذا لم يتم توفير متغير البيئة، تستخدم Xcursor المورد Xcursor.anim.
XCURSOR_CORE
إذا كان العرض يدعم طلب Render CreateCursor، فقم بتعطيل ميزة Render إذا كان متغير البيئة false.
إذا لم يتم توفير متغير البيئة، تستخدم Xcursor المورد Xcursor.core.
XCURSOR_DISCOVER
إذا تم تعيين المتغير، تقوم Xcursor بتشغيل ميزة تسجيل. تعرض قيمة التجزئة والصورة حتى يتمكن المستخدمون من رؤية اسم المؤشر المرتبط بكل صورة.
لا يوجد إعداد مورد مقابل.
XCURSOR_DITHER
يقوم هذا المتغير بتعيين التمويه المطلوب.
إذا لم يتم توفير متغير البيئة، تستخدم Xcursor المورد Xcursor.dither.
إذا لم يتم العثور على متغير البيئة أو المورد، تستخدم Xcursor "threshold"
هذه هي القيم المعترف بها:
diffuse
median
ordered
threshold
XCURSOR_PATH
يقوم هذا المتغير بتعيين قائمة المسارات التي سيتم البحث فيها عن المؤشرات، بدلاً من القائمة الافتراضية المضمنة.
يتم فصل الدلائل في هذا المسار بنقطتين رأسيتين (:).
XCURSOR_RESIZED
يمكّن تغيير حجم المؤشرات تلقائيًا لتحسين حجم عرضها إذا كان متغير البيئة true.
إذا لم يتم توفير متغير البيئة، تحاول Xcursor استخدام المورد Xcursor.resized.
XCURSOR_SIZE
يحدد هذا المتغير حجم المؤشر المطلوب، بالبكسل.
إذا لم يتم توفير متغير البيئة، تحاول Xcursor استخدام المورد Xcursor.size.
إذا لم يتم تحديد حجم، سواء عن طريق متغير البيئة أو إعداد المورد، تحاول Xcursor بعد ذلك إعداد المورد Xft.dpi لتخمين حجم مؤشر بحجم 16 نقطة.
أخيرًا، إذا لم يتم تعيين Xft.dpi، تستخدم Xcursor ارتفاع العرض، مقسومًا على 48 (بافتراض أن الارتفاع هو 768).
XCURSOR_THEME
يحدد هذا المتغير السمة المطلوبة.
إذا لم يتم توفير متغير البيئة، تحاول Xcursor استخدام المورد Xcursor.theme.
إذا لم يتم العثور على متغير البيئة أو المورد، تستخدم Xcursor السمة الافتراضية.

DESCRIPTION

The Xcursor library provides functions for loading and managing X cursor images. It allows applications to load cursor images from theme directories, set default cursor themes, and dynamically change the cursor theme. The library supports various cursor image formats and provides a way to associate cursor names with specific cursor images.

FUNCTIONAL OVERVIEW

The library can be used to:

Get the library path for cursor themes.
Set and get the default cursor theme for the X server.
Get the current cursor theme name.
Set the current cursor theme dynamically.
Load a specific cursor image by name from the current theme.
Set the cursor for the current pointer.
Enable or disable themes for core cursors.

DATATYPES

XcursorIcon
Represents a single cursor image, including its name, size, and the actual pixmap data.
XcursorCursorList
A structure used to hold a list of cursor images, typically loaded from a theme file.

Returns the default path where Xcursor looks for themes.

Sets the default cursor theme for the display. If `themeName` is NULL, it attempts to get the theme name from the Xcursor.defaultTheme resource or the Xcursor_DEFAULT_THEME environment variable.
const char * XcursorGetDefaultTheme(Display *display);
Returns the name of the default cursor theme for the display.
const char * XcursorGetTheme(Display *display);
Returns the name of the currently active cursor theme.
Bool XcursorSetTheme(Display *display, const char *themeName);
Sets the current cursor theme. If `themeName` is NULL, it resets the theme to the default. This function also attempts to load the theme specified by Xcursor.theme resource or the XCURSOR_THEME environment variable if XcursorSetDefaultTheme was not called or if the theme name is NULL.
XcursorIcon * XcursorGetIcon(Display *display, const char *name, XcursorCursorList *list);
Retrieves a specific cursor image from the loaded theme based on its name. If `list` is provided, it searches within that list; otherwise, it uses the currently active theme.
Bool XcursorSetCursor(Display *display, Cursor cursor);
Sets the specified X cursor for the current pointer. This is a low-level Xlib function that Xcursor might use internally or indirectly.
Bool XcursorSetThemeCore(Display *display, const char *themeName);
Sets the cursor theme specifically for core pointer types (e.g., the default arrow cursor). If `themeName` is NULL, it uses the default theme. The actual theme used may depend on the Xcursor.theme_core resource or the XCURSOR_THEME_CORE environment variable.

ENVIRONMENT

XCURSOR_PATH
Specifies additional directories to search for cursor themes.
XCURSOR_THEME
Sets the name of the cursor theme to use.
XCURSOR_THEME_CORE

Enables themes for core cursors if the environment variable is true.

If the environment variable is not given, Xcursor tries the Xcursor.theme_core resource.
An application can enable or disable themes using XcursorSetThemeCore.

CAVEATS

Xcursor will probably change radically in the future; weak attempts will be made to retain some level of source-file compatibility.

الوصف

توفر مكتبة Xcursor وظائف لتحميل وإدارة صور مؤشر X. تسمح للتطبيقات بتحميل صور المؤشر من مجلدات المواضيع، وتعيين مواضيع مؤشر افتراضية، وتغيير موضوع المؤشر ديناميكيًا. تدعم المكتبة تنسيقات صور المؤشر المختلفة وتوفر طريقة لربط أسماء المؤشرات بصور مؤشر محددة.

نظرة عامة وظيفية

يمكن استخدام المكتبة من أجل:

الحصول على مسار المكتبة لمواضيع المؤشر.
تعيين والحصول على موضوع المؤشر الافتراضي لخادم X.
الحصول على اسم موضوع المؤشر الحالي.
تعيين موضوع المؤشر الحالي ديناميكيًا.
تحميل صورة مؤشر محددة بالاسم من الموضوع الحالي.
تعيين المؤشر للمؤشر الحالي.
تمكين أو تعطيل المواضيع للمؤشرات الأساسية.

أنواع البيانات

XcursorIcon
يمثل صورة مؤشر فردية، بما في ذلك اسمها وحجمها وبيانات الصورة النقطية الفعلية.
XcursorCursorList
بنية تستخدم للاحتفاظ بقائمة صور المؤشر، يتم تحميلها عادةً من ملف الموضوع.

الوظائف

const char * XcursorLibraryPath(void);
ترجع المسار الافتراضي حيث يبحث Xcursor عن المواضيع.
Bool XcursorSetDefaultTheme(Display *display, const char *themeName);
يعين موضوع المؤشر الافتراضي للعرض. إذا كان `themeName` فارغًا (NULL)، فإنه يحاول الحصول على اسم الموضوع من مورد Xcursor.defaultTheme أو متغير البيئة Xcursor_DEFAULT_THEME.
const char * XcursorGetDefaultTheme(Display *display);
ترجع اسم موضوع المؤشر الافتراضي للعرض.
const char * XcursorGetTheme(Display *display);
ترجع اسم موضوع المؤشر النشط حاليًا.
Bool XcursorSetTheme(Display *display, const char *themeName);
يعين موضوع المؤشر الحالي. إذا كان `themeName` فارغًا (NULL)، فإنه يعيد تعيين الموضوع إلى الافتراضي. تحاول هذه الوظيفة أيضًا تحميل الموضوع المحدد بواسطة مورد Xcursor.theme أو متغير البيئة XCURSOR_THEME إذا لم يتم استدعاء XcursorSetDefaultTheme أو إذا كان اسم الموضوع فارغًا (NULL).
XcursorIcon * XcursorGetIcon(Display *display, const char *name, XcursorCursorList *list);
يسترد صورة مؤشر محددة من الموضوع المحمل بناءً على اسمها. إذا تم توفير `list`، فإنه يبحث داخل تلك القائمة؛ وإلا، فإنه يستخدم الموضوع النشط حاليًا.
Bool XcursorSetCursor(Display *display, Cursor cursor);
يعين مؤشر X المحدد للمؤشر الحالي. هذه وظيفة Xlib منخفضة المستوى قد يستخدمها Xcursor داخليًا أو بشكل غير مباشر.
Bool XcursorSetThemeCore(Display *display, const char *themeName);
يعين موضوع المؤشر خصيصًا لأنواع المؤشرات الأساسية (مثل المؤشر السهم الافتراضي). إذا كان `themeName` فارغًا (NULL)، فإنه يستخدم الموضوع الافتراضي. قد يعتمد الموضوع الفعلي المستخدم على مورد Xcursor.theme_core أو متغير البيئة XCURSOR_THEME_CORE.

البيئة

XCURSOR_PATH
يحدد الدلائل الإضافية للبحث عن مواضيع المؤشر.
XCURSOR_THEME
يعين اسم موضوع المؤشر المراد استخدامه.
XCURSOR_THEME_CORE

يمكّن المواضيع للمؤشرات الأساسية إذا كان متغير البيئة هو true.

إذا لم يتم تقديم متغير البيئة، فإن Xcursor يحاول استخدام مورد Xcursor.theme_core.
يمكن للتطبيق تمكين أو تعطيل المواضيع باستخدام XcursorSetThemeCore.
