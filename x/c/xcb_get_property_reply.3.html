xcb_get_property_reply(3)

الحزمة: libxcb

التراخيص: X11

xcb_get_property(3) طلبات XCB xcb_get_property(3)

الاسم

xcb_get_property - الحصول على خاصية النافذة

ملخص

#include <xcb/xproto.h>

الوظيفة الطلبية

xcb_get_property_cookie_t xcb_get_property(xcb_connection_t *conn, uint8_t _delete, xcb_window_t window, xcb_atom_t property, xcb_atom_t type, uint32_t long_offset, uint32_t long_length);

هيكل بيانات الرد

typedef struct xcb_get_property_reply_t {
    uint8_t    response_type;
    uint8_t    format;
    uint16_t   sequence;
    uint32_t   length;
    xcb_atom_t type;
    uint32_t   bytes_after;
    uint32_t   value_len;
    uint8_t    pad0[12];
} xcb_get_property_reply_t;

وظيفة الرد

xcb_get_property_reply_t *xcb_get_property_reply(xcb_connection_t *conn, xcb_get_property_cookie_t cookie, xcb_generic_error_t **e);

مُوصِلات الرد

void *xcb_get_property_value(const xcb_get_property_request_t *reply);

int xcb_get_property_value_length(const xcb_get_property_reply_t *reply);

xcb_generic_iterator_t xcb_get_property_value_end(const xcb_get_property_reply_t *reply);

وسائط الطلب

conn
اتصال XCB بـ X11.
_delete
ما إذا كان يجب حذف الخاصية فعليًا. لحذف خاصية، يجب أن يتطابق النوع المحدد type مع نوع الخاصية الفعلي.
window
النافذة التي تريد الحصول على خاصيتها.
property
الخاصية التي تريد الحصول عليها (رمز).
type
نوع الخاصية التي تريد الحصول عليها (رمز).
long_offset
يحدد الإزاحة (بمضاعفات 32 بت) في الخاصية المحددة حيث سيتم استرداد البيانات.
long_length
يحدد عدد مضاعفات 32 بت من البيانات التي يجب استردادها (على سبيل المثال، إذا قمت بتعيين long_length إلى 4، فستتلقى 16 بايت من البيانات).

حقول الرد

response_type
نوع هذا الرد، في هذه الحالة XCB_GET_PROPERTY. هذا الحقل موجود أيضًا في xcb_generic_reply_t ويمكن استخدامه لتمييز الردود عن بعضها البعض.
sequence
رقم تسلسل آخر طلب تمت معالجته بواسطة خادم X11.
length
طول الرد، بالكلمات (الكلمة هي 4 بايت).
format
يحدد ما إذا كان يجب عرض البيانات كقائمة من الكميات 8 بت، 16 بت، أو 32 بت. القيم الممكنة هي 8، 16، و 32. تسمح هذه المعلومات لخادم X بإجراء عمليات تبديل البايت بشكل صحيح حسب الضرورة.
type
النوع الفعلي للخاصية (رمز).
bytes_after
عدد البايتات المتبقية للقراءة في الخاصية إذا تم إجراء قراءة جزئية.
value_len
طول القيمة. يجب عليك استخدام المُوصِل المقابل بدلاً من هذا الحقل.

الوصف

يحصل على property المحدد من window المحدد. الخصائص هي على سبيل المثال عنوان النافذة (WM_NAME) أو الحد الأدنى للحجم (WM_NORMAL_HINTS). تستخدم البروتوكولات مثل EWMH أيضًا الخصائص - على سبيل المثال، يحدد EWMH عنوان النافذة، المشفر كسلسلة UTF-8، في خاصية _NET_WM_NAME.

القيمة المعادة

يعيد xcb_get_property_cookie_t. يجب معالجة الأخطاء عند استدعاء وظيفة الرد xcb_get_property_reply.

إذا كنت ترغب في معالجة الأخطاء في حلقة الأحداث بدلاً من ذلك، فاستخدم xcb_get_property_unchecked. انظر xcb-requests(3) للتفاصيل.

الأخطاء

xcb_atom_error_t
property أو type لا تشير إلى رمز صالح.
xcb_value_error_t
long_offset المحدد يتجاوز طول الخاصية الفعلي (على سبيل المثال، طول الخاصية هو 3 بايت وتقوم بتعيين long_offset إلى 1، مما ينتج عنه إزاحة بايت تبلغ 4).
xcb_window_error_t
النافذة المحددة window غير موجودة.

مثال

/*
 * يطبع خاصية WM_NAME للنافذة.
 *
 */
void my_example(xcb_connection_t *c, xcb_window_t window) {
    xcb_get_property_cookie_t cookie;
    xcb_get_property_reply_t *reply;
    /* هذه الرموز معرفة مسبقًا في بروتوكول X11. */
    xcb_atom_t property = XCB_ATOM_WM_NAME;
    xcb_atom_t type = XCB_ATOM_STRING;
    // TODO: long_length معقولة لـ WM_NAME؟
    cookie = xcb_get_property(c, 0, window, property, type, 0, 0);
    if ((reply = xcb_get_property_reply(c, cookie, NULL))) {
        int len = xcb_get_property_value_length(reply);
        if (len == 0) {
            printf("TODO\n");
            free(reply);
            return;
        }
        printf("WM_NAME is %.*s\n", len,
               (char*)xcb_get_property_value(reply));
    }
    free(reply);
}
