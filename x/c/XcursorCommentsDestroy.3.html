الاسم

Xcursor - مكتبة إدارة المؤشرات

الملخص

#include <X11/Xcursor/Xcursor.h>

الوصف

Xcursor هي مكتبة بسيطة مصممة للمساعدة في تحديد وتحميل المؤشرات. يمكن تحميل المؤشرات من ملفات أو من الذاكرة. توجد مكتبة من المؤشرات الشائعة التي ترتبط بأسماء مؤشرات X القياسية. يمكن أن توجد المؤشرات بأحجام متعددة، وتقوم المكتبة تلقائيًا باختيار أفضل حجم.

نظرة عامة وظيفية

تم بناء Xcursor على عدة طبقات؛ في الطبقة السفلية يوجد الكود الذي يمكنه تحميل صور المؤشرات من الملفات. فوق ذلك توجد طبقة تحدد مواقع ملفات المؤشرات بناءً على مسار المكتبة والموضوع. في الأعلى توجد طبقة تقوم ببناء المؤشرات إما من صورة تم تحميلها من ملف أو من أحد مؤشرات X القياسية. عند استخدام صور تم تحميلها من ملفات، تفضل Xcursor استخدام طلب إنشاء المؤشر الخاص بامتداد Render إذا كان مدعومًا من خادم X. حيثما لم يكن مدعومًا، تقوم Xcursor بربط صورة المؤشر بمؤشر X قياسي وتستخدم طلب إنشاء المؤشر الأساسي.

ملفات المؤشر

تحدد Xcursor تنسيقًا جديدًا للمؤشرات على القرص. يحتوي كل ملف على صورة مؤشر واحدة أو أكثر. يتم تمييز كل صورة مؤشر بحجم اسمي بحيث يمكن تحديد أفضل حجم تلقائيًا. يمكن تحميل مؤشرات متعددة بنفس الحجم الاسمي معًا؛ من المتوقع أن تستخدم التطبيقات هذه كم تسلسل متحرك.

يتم تخزين ملفات المؤشر كرأس يحتوي على جدول محتويات متبوعًا بسلسلة من الأجزاء. يشير جدول المحتويات إلى النوع، النوع الفرعي، والموضع في الملف لكل جزء. يبدو رأس الملف كالتالي:


magic:	CARD32 "Xcur" (0x58, 0x63, 0x75, 0x72)
header:	CARD32 بايت في هذا الرأس
version:	CARD32 رقم إصدار الملف
ntoc:	CARD32 عدد إدخالات جدول المحتويات
toc:	LISTofTOC جدول المحتويات

كل إدخال في جدول المحتويات يبدو كالتالي:


type:	CARD32 نوع الإدخال
subtype:	CARD32 تسمية خاصة بالنوع - حجم للصور
position:	CARD32 موضع البايت المطلق للجدول في الملف

كل جزء في الملف لديه مجموعة من حقول الرأس المشتركة متبوعة بحقول إضافية خاصة بالنوع:


header:	CARD32 بايت في رأس الجزء (بما في ذلك الحقول الخاصة بالنوع)
type:	CARD32 يجب أن يتطابق مع النوع في TOC لهذا الجزء
subtype:	CARD32 يجب أن يتطابق مع النوع الفرعي في TOC لهذا الجزء
version:	CARD32 رقم إصدار لهذا النوع من الأجزاء

يوجد حاليًا نوعان محددات للأجزاء لملفات المؤشرات؛ التعليقات والصور. تبدو التعليقات كالتالي:


header:	20 رؤوس التعليقات هي 20 بايت
type:	0xfffe0001 نوع التعليق هو 0xfffe0001
subtype:	{ 1 (حقوق النشر)، 2 (الترخيص)، 3 (أخرى) }
version:	1
length:	CARD32 طول البايت للسلسلة UTF-8
string:	LISTofCARD8 سلسلة UTF-8

تبدو الصور كالتالي:


header:	36 رؤوس الصور هي 36 بايت
type:	0xfffd0002 نوع الصورة هو 0xfffd0002
subtype:	CARD32 النوع الفرعي للصورة هو الحجم الاسمي
version:	1
width:	CARD32 يجب أن يكون أقل من أو يساوي 0x7fff
height:	CARD32 يجب أن يكون أقل من أو يساوي 0x7fff
xhot:	CARD32 يجب أن يكون أقل من أو يساوي العرض
yhot:	CARD32 يجب أن يكون أقل من أو يساوي الارتفاع
delay:	CARD32 التأخير بين إطارات الرسوم المتحركة بالمللي ثانية
pixels:	LISTofCARD32 بكسلات بصيغة ARGB مضغوطة

المواضيع

Xcursor تتبع (في الغالب) مواصفات freedesktop.org لمواضيع الأيقونات. المسار الافتراضي للبحث الذي تستخدمه هو

~/.local/share/icons, ~/.icons, /usr/share/icons, /usr/share/pixmaps

داخل كل من هذه الأدلة، تبحث عن دليل باستخدام اسم الموضوع:

•
داخل دليل الموضوع، تبحث عن ملفات المؤشرات في الدليل الفرعي "cursors".
تبحث Xcursor عن ملف معين، يجب أن يكون أحد أسماء شكل المؤشر، على سبيل المثال، كما هو مستخدم في XcursorLibraryLoadImage أو XcursorLibraryShape.
•
إذا لم تجد ملف مؤشر مطابق في الدليل الفرعي "cursors"، تبحث Xcursor بعد ذلك عن ملف "index.theme" في كل دليل موضوع للبحث عن مواضيع موروثة. هذه هي أسطر بهذا التنسيق:
Inherits = اسم-الموضوع

تستخدم Xcursor أول اسم-موضوع موروث، متجاهلة الآخرين الذين قد يوجدون في ملف "index.theme" معين. إذا وجدت موضوعًا موروثًا، فتبحث Xcursor على طول المسار لاستخدامه أيضًا. تتجاهل Xcursor مفاتيح أخرى في ملف "index.theme"، بما في ذلك "Name" (أي، الاسم الذي قد يستخدمه التطبيق الرسومي كـ اسم عرض).

يمكن سرد أكثر من اسم-موضوع واحد في سطر Inherits=. تنص مواصفات freedesktop.org على أن عناصر القائمة مفصولة بفواصل. تقبل Xcursor أيضًا الفاصلة المنقوطة، ولكنها تحول كلاهما إلى نقطتين عند البحث في المسار. تتوقع Xcursor سطر Inherits= واحد فقط؛ مواصفات freedesktop.org غير واضحة بشأن ما إذا كانت مفاتيح متعددة مسموح بها.

إذا لم يتم تعيين موضوع، أو إذا لم يتم العثور على مؤشر للموضوع المحدد في أي مكان على طول المسار، تتحقق Xcursor من موضوع "default".

عندما تجد Xcursor ملف مؤشر، تتوقف عن البحث. تستخدم دائمًا أول ملف مؤشر يتم العثور عليه أثناء البحث على طول المسار.

أنواع البيانات

XcursorImage
يحتوي على صورة مؤشر واحدة في الذاكرة. كل بكسل في المؤشر هو قيمة 32 بت تحتوي على ARGB مع A في البايت الأعلى. 


typedef struct _XcursorImage {
    XcursorDim 	 size; 	/* حجم اسمي للمطابقة */
    XcursorDim 	 width; 	/* العرض الفعلي */
    XcursorDim 	 height; 	/* الارتفاع الفعلي */
    XcursorDim 	 xhot; 	/* نقطة السخونة x (يجب أن تكون داخل الصورة) */
    XcursorDim 	 yhot; 	/* نقطة السخونة y (يجب أن تكون داخل الصورة) */
    XcursorPixel 	*pixels; 	/* مؤشر إلى البكسلات */
} XcursorImage;
XcursorImages
يحتوي على هياكل XcursorImage متعددة. يتم تحريرها جميعًا عند تحرير XcursorImages في XcursorImagesDestroy. 


typedef struct _XcursorImages {
    int 	 nimage; 	/* عدد الصور */
    XcursorImage 	**images; 	/* مصفوفة مؤشرات XcursorImage */
} XcursorImages;
XcursorCursors
يحتوي على كائنات المؤشر المتعددة. يتم تحريرها جميعًا عند تحرير XcursorCursors. هذه ذات عد مراجع بحيث يمكن لهياكل XcursorAnimate متعددة استخدام نفس XcursorCursors. 


typedef struct _XcursorCursors {
    Display 	*dpy; 	/* شاشة عرض تحتوي على المؤشرات */
    int 	 ref; 	/* عداد المراجع */
    int 	 ncursor; 	/* عدد المؤشرات */
    Cursor 	*cursors; 	/* مصفوفة المؤشرات */
} XcursorCursors;
XcursorAnimate
يشير إلى مجموعة من المؤشرات وتسلسل داخل تلك المجموعة. قد تشير هياكل XcursorAnimate متعددة إلى نفس XcursorCursors؛ يحتفظ كل منها بمرجع يتم إزالته عند تحرير XcursorAnimate. 


typedef struct _XcursorAnimate {
    XcursorCursors 	*cursors; 	/* قائمة المؤشرات المراد استخدامها */
    int 	 sequence; 	/* المؤشر التالي */
} XcursorAnimate;
XcursorFile
توفر Xcursor واجهة برمجة تطبيقات مجردة للوصول إلى بيانات الملف. توفر Xcursor تطبيقًا قياسيًا لهذه الواجهة المجردة؛ التطبيقات حرة في إنشاء تطبيقات إضافية. توازي هذه الدوال stdio

هذا الموقع غير مرتبط بـ Arch Linux ولا تابع له بأي شكل من الأشكال. المحتوى هو ترجمة فقط لصفحات دليل Arch المنشورة على man.archlinux.org.

functions in return value and expected argument values; the read and write functions flip the arguments around to match the POSIX versions. 


typedef struct _XcursorFile {
    void *closure;
    int (*read)  (XcursorFile *file, unsigned char *buf, int len);
    int (*write) (XcursorFile *file, unsigned char *buf, int len);
    int (*seek)  (XcursorFile *file, long offset, int whence);
};

تبحث هذه الوظائف في مسار البحث الخاص بالمكتبة، وتقوم بتحميل أول ملف يتم العثور عليه بالحجم المطلوب، باستخدام وظيفة خاصة (XcursorScanTheme) لتحديد السمة المناسبة:

  • إذا كانت theme ليست NULL، فاستخدمها.
  • إذا كانت theme NULL، أو إذا لم يكن هناك تطابق للسمة المطلوبة، فاستخدم "default" لاسم السمة.
  • إذا لم ينجح أي من البحثين، فإن هذه الوظائف تعيد NULL.
تختلف الوظيفتان أكثر من عدد الصور المحملة:
  • تستدعي XcursorLibraryLoadImage الوظيفة XcursorFileLoadImage ولكن
  • تستدعي XcursorLibraryLoadImages الوظيفة XcursorFileLoadImages، (عند النجاح) تستدعي XcursorImagesSetName لربط name بالنتيجة.

NAME


XcursorCommentsDestroy – destroy comment data






SYNOPSIS


#include <Xcursor/Xcursor.h>
void
XcursorCommentsDestroy(XcursorComments *comments);







DESCRIPTION


The function XcursorCommentsDestroy frees the memory
    allocated for the comment data structure.






ENVIRONMENT



  
THEME_CORE

  
Enables themes for core cursors if the environment variable is
      true.





  

  
If the environment variable is not given, Xcursor tries the
      Xcursor.theme_core resource.

  

  
An application can enable or disable themes using
    XcursorSetThemeCore.








CAVEATS


Xcursor will probably change radically in the future; weak
    attempts will be made to retain some level of source-file compatibility.
