الحزمة: libxcursor

التراخيص: HPND-sell-variant

Xcursor(3) إصدار X 11 Xcursor(3)

اسم

Xcursor - مكتبة إدارة المؤشرات

ملخص

#include <X11/Xcursor/Xcursor.h>

الوصف

Xcursor هي مكتبة بسيطة مصممة للمساعدة في تحديد وتحميل المؤشرات. يمكن تحميل المؤشرات من الملفات أو الذاكرة. توجد مكتبة للمؤشرات الشائعة التي تتطابق مع أسماء مؤشرات X القياسية. يمكن أن توجد المؤشرات بأحجام متعددة والمكتبة تختار تلقائيًا أفضل حجم.

نظرة عامة وظيفية

تم بناء Xcursor على بضع طبقات؛ في الطبقة السفلية يوجد كود يمكنه تحميل صور المؤشرات من الملفات. فوق ذلك توجد طبقة تحدد ملفات المؤشر بناءً على مسار المكتبة والموضوع. في الأعلى توجد طبقة تبني مؤشرات إما من صورة تم تحميلها من ملف أو أحد مؤشرات X القياسية. عند استخدام الصور المحملة من الملفات، تفضل Xcursor استخدام طلب إنشاء المؤشر الخاص بامتداد Render إذا كان مدعومًا من خادم X. حيث لا يكون مدعومًا، تقوم Xcursor بمطابقة صورة المؤشر مع مؤشر X قياسي وتستخدم طلب الإنشاء الأساسي.

ملفات المؤشرات

تحدد Xcursor تنسيقًا جديدًا للمؤشرات على القرص. يحتوي كل ملف على صورة مؤشر واحدة أو أكثر. يتم تمييز كل صورة مؤشر بحجم اسمي بحيث يمكن تحديد أفضل حجم تلقائيًا. يمكن تحميل مؤشرات متعددة من نفس الحجم الاسمي معًا؛ يتوقع من التطبيقات استخدامها كتسلسل متحرك.

يتم تخزين ملفات المؤشر كرأس يحتوي على جدول محتويات متبوعًا بتسلسل من المقاطع. يشير جدول المحتويات إلى النوع والنوع الفرعي والموضع في الملف لكل مقطع. رأس الملف يبدو كالتالي:


magic:	CARD32 "Xcur" (0x58, 0x63, 0x75, 0x72)
header:	CARD32 بايت في هذا الرأس
version:	CARD32 رقم إصدار الملف
ntoc:	CARD32 عدد إدخالات جدول المحتويات
toc:	LISTofTOC جدول المحتويات

كل إدخال في جدول المحتويات يبدو كالتالي:


type:	CARD32 نوع الإدخال
subtype:	CARD32 تسمية خاصة بالنوع - حجم للصور
position:	CARD32 الموضع المطلق بالبايت لجدول المحتويات في الملف

كل مقطع في الملف لديه مجموعة من حقول الرأس الشائعة متبوعة بحقول إضافية خاصة بالنوع:


header:	CARD32 بايت في رأس المقطع (بما في ذلك الحقول الخاصة بالنوع)
type:	CARD32 يجب أن يتطابق مع النوع في TOC لهذا المقطع
subtype:	CARD32 يجب أن يتطابق مع النوع الفرعي في TOC لهذا المقطع
version:	CARD32 رقم إصدار لهذا النوع من المقاطع

يوجد حاليًا نوعان من المقاطع محددين لملفات المؤشرات؛ التعليقات والصور. التعليقات تبدو كالتالي:


header:	20 رؤوس التعليقات هي 20 بايت
type:	0xfffe0001 نوع التعليق هو 0xfffe0001
subtype:	{ 1 (حقوق النشر)، 2 (الترخيص)، 3 (أخرى) }
version:	1
length:	CARD32 طول البايت للسلسلة النصية UTF-8
string:	LISTofCARD8 سلسلة نصية UTF-8

الصور تبدو كالتالي:


header:	36 رؤوس الصور هي 36 بايت
type:	0xfffd0002 نوع الصورة هو 0xfffd0002
subtype:	CARD32 النوع الفرعي للصورة هو الحجم الاسمي
version:	1
width:	CARD32 يجب أن يكون أقل من أو يساوي 0x7fff
height:	CARD32 يجب أن يكون أقل من أو يساوي 0x7fff
xhot:	CARD32 يجب أن يكون أقل من أو يساوي العرض
yhot:	CARD32 يجب أن يكون أقل من أو يساوي الارتفاع
delay:	CARD32 التأخير بين إطارات الرسوم المتحركة بالمللي ثانية
pixels:	LISTofCARD32 بكسلات بتنسيق ARGB المعبأ

الموضوعات

تتبع Xcursor (في الغالب) مواصفات freedesktop.org لموضوعات الأيقونات. المسار الافتراضي للبحث الذي تستخدمه هو

~/.local/share/icons, ~/.icons, /usr/share/icons, /usr/share/pixmaps

داخل كل من هذه الدلائل، تبحث عن دليل باستخدام اسم الموضوع:

•
داخل دليل الموضوع، تبحث عن ملفات المؤشر في الدليل الفرعي "cursors".
تبحث Xcursor عن ملف معين، والذي يجب أن يكون أحد أسماء *shape* للمؤشر، على سبيل المثال، كما هو مستخدم في XcursorLibraryLoadImage أو XcursorLibraryShape.
•
إذا لم تجد ملف مؤشر مطابق في الدليل الفرعي "cursors"، فإن Xcursor تبحث بعد ذلك عن ملف "index.theme" في كل دليل موضوع للبحث عن الموضوعات الموروثة. هذه هي أسطر بهذا التنسيق:
Inherits = اسم-الموضوع

تستخدم Xcursor أول اسم-موضوع موروث، متجاهلة الآخرين الذين قد يكونون موجودين في ملف "index.theme" معين. إذا وجدت موضوعًا موروثًا، فإن Xcursor تبحث على طول المسار لاستخدامه أيضًا. تتجاهل Xcursor مفاتيح أخرى في ملف "index.theme"، بما في ذلك "Name" (أي، الاسم الذي قد يستخدمه التطبيق الرسومي كـ اسم العرض).

قد يتم سرد أكثر من اسم-موضوع واحد في سطر Inherits=. تنص مواصفات freedesktop.org على أن عناصر القائمة مفصولة بفواصل. تقبل Xcursor أيضًا الفاصلة المنقوطة، ولكنها تترجم كلاهما إلى نقطتين عند البحث في المسار. تتوقع Xcursor سطرًا واحدًا فقط من Inherits=؛ مواصفات freedesktop.org غير واضحة ما إذا كانت مفاتيح متعددة مسموح بها.

إذا لم يتم تعيين موضوع، أو إذا لم يتم العثور على مؤشر للموضوع المحدد في أي مكان على طول المسار، فإن Xcursor تتحقق من موضوع "default".

عندما تجد Xcursor ملف مؤشر، تتوقف عن البحث. إنها تستخدم دائمًا أول ملف مؤشر يتم العثور عليه أثناء البحث على طول المسار.

أنواع البيانات

XcursorImage
تحتوي على صورة مؤشر واحدة في الذاكرة. كل بكسل في المؤشر هو قيمة 32 بت تحتوي على ARGB مع A في البايت العالي. 


typedef struct _XcursorImage {
    XcursorDim 	 size; 	/* حجم اسمي للمطابقة */
    XcursorDim 	 width; 	/* العرض الفعلي */
    XcursorDim 	 height; 	/* الارتفاع الفعلي */
    XcursorDim 	 xhot; 	/* نقطة التركيز x (يجب أن تكون داخل الصورة) */
    XcursorDim 	 yhot; 	/* نقطة التركيز y (يجب أن تكون داخل الصورة) */
    XcursorPixel 	*pixels; 	/* مؤشر إلى البكسلات */
} XcursorImage;
XcursorImages
تحتوي على هياكل XcursorImage متعددة. يتم تحريرها جميعًا عند تحرير XcursorImages في XcursorImagesDestroy. 


typedef struct _XcursorImages {
    int 	 nimage; 	/* عدد الصور */
    XcursorImage 	**images; 	/* مصفوفة مؤشرات XcursorImage */
} XcursorImages;
XcursorCursors
تحتوي على كائنات مؤشر متعددة. يتم تحريرها جميعًا عند تحرير XcursorCursors. هذه ذات عد مرجعي بحيث يمكن لهياكل XcursorAnimate متعددة استخدام نفس XcursorCursors. 


typedef struct _XcursorCursors {
    Display 	*dpy; 	/* شاشة X تحتفظ بالمؤشرات */
    int 	 ref; 	/* عداد المراجع */
    int 	 ncursor; 	/* عدد المؤشرات */
    Cursor 	*cursors; 	/* مصفوفة المؤشرات */
} XcursorCursors;
XcursorAnimate
تشير إلى مجموعة من المؤشرات وتسلسل داخل تلك المجموعة. قد تشير هياكل XcursorAnimate متعددة إلى نفس XcursorCursors؛ يحتفظ كل منها بمرجع يتم إزالته عند تحرير XcursorAnimate. 


typedef struct _XcursorAnimate {
    XcursorCursors 	*cursors; 	/* قائمة المؤشرات المراد استخدامها */
    int 	 sequence; 	/* أي مؤشر هو التالي */
} XcursorAnimate;
XcursorFile
توفر Xcursor واجهة برمجة تطبيقات مجردة للوصول إلى بيانات الملف. توفر Xcursor تطبيقًا قياسيًا لهذه الواجهة المجردة؛ التطبيقات حرة في إنشاء تطبيقات إضافية. هذه الدوال توازي دوال stdio

NAME

libXcursor - cursor loading and management library

SYNOPSIS

#include <X11/Xcursor/Xcursor.h>

DESCRIPTION

The Xcursor library provides functions for loading and managing Xcursor images. It provides APIs for animated cursors, glyph cursors, and display information.

Loading and Management APIs

XcursorImages * XcursorImagesCreate (
	int 	 width)

Allocates and returns a new XcursorImages structure, which can hold cursors of the given width.

void XcursorImagesDestroy (
	XcursorImages 	*images)

Deallocates the given XcursorImages structure as well as all the XcursorImage structures it points to.

XcursorImage * XcursorImageCreate (
	int 	 width,
	int 	 height)

Allocates and returns a new XcursorImage structure with the given width and height. The structure is initialized to zeros.

void XcursorImageDestroy (
	XcursorImage 	*image)

Deallocates the given XcursorImage structure. Note that this function does not free the image data itself, but rather the structure containing the pointers to the data.

void XcursorImageSet (
	XcursorImage 	*image,
	unsigned int 	 name,
	unsigned int 	 width,
	unsigned int 	 height,
	unsigned int 	 xhot,
	unsigned int 	 yhot,
	const char 	*data,
	const char 	*mask)

Sets the fields of an XcursorImage structure. The data and mask are copied.

void XcursorImageSetName (
	XcursorImage 	*image,
	unsigned int 	 name)

Sets the name field of the given XcursorImage structure.

void XcursorImageSetHotSpot (
	XcursorImage 	*image,
	unsigned int 	 xhot,
	unsigned int 	 yhot)

Sets the hotspot fields of the given XcursorImage structure.

void XcursorImageLoad (
	XcursorImages 	*images,
	XcursorFile 	*file)

Loads a cursor image from the given XcursorFile into the XcursorImages structure.

XcursorFile * XcursorFileOpen (
	const char 	*filename)

Opens the specified file for reading cursor data.

void XcursorFileClose (
	XcursorFile 	*file)

Closes the given XcursorFile.

Cursor XcursorShapeLoadCursor (
	Display 	*dpy,
	const char 	*shape)

Loads a cursor of the specified shape. If the shape is not found, Xcursor will try to load the cursor using the default size and theme.

Cursor XcursorShapeLoadNamedCursor (
	Display 	*dpy,
	const char 	*shape,
	const char 	*nickname)

Loads a cursor of the specified shape and nickname.

Cursor XcursorNameLoadCursor (
	Display 	*dpy,
	const char 	*name)

Loads a cursor with the given name. This is often used for application-specific cursors.

Cursor XcursorLoadCursor (
	Display 	*dpy,
	const char 	*file)

Loads a cursor from the specified file.

void XcursorSetDefaultTheme (
	Display 	*dpy,
	const char 	*theme)

Sets the default theme to be used for loading cursors.

void XcursorSetTheme (
	Display 	*dpy,
	const char 	*theme)

Sets the current theme to be used for loading cursors.

char * XcursorGetTheme (
	Display 	*dpy)

Returns the name of the current theme.

void XcursorSetDefaultSize (
	Display 	*dpy,
	int 	 size)

Sets the default size for cursors on the specified display.

int XcursorGetDefaultSize (
	Display 	*dpy)

Returns the default cursor size.

void XcursorSetSize (
	Display 	*dpy,
	int 	 size)

Sets the size for cursors on the specified display.

int XcursorGetSize (
	Display 	*dpy)

Returns the current cursor size.

void XcursorSetClipping (
	Display 	*dpy,
	XRectangle 	*rect)

Sets the clipping rectangle for cursors.

void XcursorSetSubpixel (
	Display 	*dpy,
	XcursorBool 	 subpixel)

Sets the subpixel rendering mode for cursors.

XcursorBool XcursorGetSubpixel (
	Display 	*dpy)

Returns the current subpixel rendering mode.

void XcursorSetRgba (
	Display 	*dpy,
	XcursorBool 	 rgba)

Sets the RGBA mode for cursors. If enabled, cursors with alpha channels will be displayed correctly.

XcursorBool XcursorGetRgba (
	Display 	*dpy)

Returns the current RGBA mode.

void XcursorSetFill (
	Display 	*dpy,
	XcursorBool 	 fill)

Sets the fill mode for cursors. If enabled, cursors will be filled with the background color.

XcursorBool XcursorGetFill (

void XcursorSetDither (

Sets the dither mode for cursors. Recognized values are "threshold", "ordered", "median", and "diffuse".

Animated Cursors

XcursorAnimate * XcursorAnimateCreate (
	XcursorCursors 	*cursors)

Wrap the given array of cursors in a newly allocated XcursorAnimate structure, which adds a sequence number used in XcursorAnimateNext.

void XcursorAnimateDestroy (
	XcursorAnimate 	*animate)

Discards the given animate data, freeing both the XcursorCursors array of cursors as well as the XcursorAnimate structure.

Cursor XcursorAnimateNext (
	XcursorAnimate 	*animate)

Cyclically returns the next Cursor in the array, incrementing the sequence number to prepare for the next call.

The caller is responsible for displaying the series of Cursor images. Xcursor does not do that.

Glyph Cursor APIs

The X11 XCreateFontCursor and XCreateGlyphCursor functions use this part of the API to extend the X core cursors feature to use themes.

void XcursorImageHash (
	XImage 	*image,
	unsigned char 	 hash[XCURSOR_BITMAP_HASH_SIZE])

Compute a hash of the image, to display when the environment variable XCURSOR_DISCOVER is set.

void XcursorImagesSetName (
	XcursorImages 	*images,
	const char 	*name)

Associates the given name with the images.

void XcursorNoticeCreateBitmap (
	Display 	*dpy,
	Pixmap 	 pid,
	unsigned int 	 width,
	unsigned int 	 height)

Check if the display supports either ARGB or themes, and also if the image size fits within the maximum cursor size (64 pixels). If so, create a bitmap of the specified size, and cache the result in Xcursor, identifying it with the Pixmap-id (pid) value.

void XcursorNoticePutBitmap (
	Display 	*dpy,
	Drawable 	 draw,
	XImage 	*image)

Update the image contents in the bitmap specified by the draw value (a Pixmap-id). The bitmap must have been created by XcursorNoticeCreateBitmap.

Cursor XcursorTryShapeBitmapCursor (
	Display 	*dpy,
	Pixmap 	 source,
	Pixmap 	 mask,
	XColor 	*foreground,
	XColor 	*background,
	unsigned int 	 x,
	unsigned int 	 y)

If the display supports either ARGB or themes, try to load a cursor into Xcursor's cache using the source parameter as a Pixmap-id. The source may no longer be in the cache. Xcursor uses the hash value to identify the desired image.

Cursor XcursorTryShapeCursor (
	Display 	*dpy,
	Font 	 source_font,
	Font 	 mask_font,
	unsigned int 	 source_char,
	unsigned int 	 mask_char,
	XColor _Xconst 	*foreground,
	XColor _Xconst 	*background)

If the display supports either ARGB or themes, try to load a cursor into Xcursor's cache using the source_char parameter as a shape. Using

  • the default size from XcursorGetDefaultSize,
  • the default theme from XcursorGetTheme, and
  • the source_char parameter as a shape,
Xcursor calls XcursorShapeLoadImages to load the cursor images. If successful, Xcursor uses XcursorImagesLoadCursor to load the cursor information.

ENVIRONMENT

Environment variables can be used to override resource settings, which in turn override compiled-in default values.

Some of the environment variables recognized by Xcursor are booleans, specified as follows:

true for “t”, “1”, “y” or “on”
false for “f”, “0”, “n” or “off”

Xcursor ignores other values for these booleans.

HOME
Xcursor interprets “~” in the search list as the home directory, using this variable rather than the password database.
XCURSOR_ANIM
If the display supports the Render CreateCursor request, and the Render feature is enabled, disable animated cursors if the environment variable is false.
If the environment variable is not given, Xcursor uses the resource Xcursor.anim.
XCURSOR_CORE
If the display supports the Render CreateCursor request disable the Render feature if the environment variable is false.
If the environment variable is not given, Xcursor uses the resource Xcursor.core.
XCURSOR_DISCOVER
If the variable is set, Xcursor turns on a logging feature. It displays the hash value and the image so that users can see which cursor name is associated with each image.
There is no corresponding resource setting.
XCURSOR_DITHER
This variable sets the desired dither.
If the environment variable is not given, Xcursor uses the resource Xcursor.dither.
If neither environment variable or resource is found, Xcursor uses “threshold”
These are the recognized values:
diffuse
median
ordered
threshold
XCURSOR_PATH
This variable sets the list of paths in which to search for cursors, rather than the compiled-in default list.
Directories in this path are separated by colons (:).
XCURSOR_RESIZED
Enables automatic resizing of cursors to improve their displayed size if the environment variable is true.
If the environment variable is not given, Xcursor tries the Xcursor.resized resource.
XCURSOR_SIZE
This variable sets the desired cursor size, in pixels.
If the environment variable is not given, Xcursor tries the Xcursor.size resource.
If no size is given, whether by environment variable or resource setting, Xcursor next tries the Xft.dpi resource setting to guess the size of a 16-point cursor.
Finally, if Xft.dpi is not set, Xcursor uses the display height, dividing by 48 (assuming that the height is 768).
XCURSOR_THEME
This variable selects the desired theme.
If the environment variable is not given, Xcursor tries the Xcursor.theme resource.
If neither environment variable or resource is found, Xcursor uses the default theme.
XCURSOR_THEME_CORE
يُمكّن السمات لمؤشرات النواة إذا كانت متغير البيئة صحيح.
إذا لم يتم تقديم متغير البيئة، يحاول Xcursor المورد Xcursor.theme_core.
يمكن للتطبيق تمكين أو تعطيل السمات باستخدام XcursorSetThemeCore.
