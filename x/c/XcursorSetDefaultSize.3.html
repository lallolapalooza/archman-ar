الاسم

Xcursor - مكتبة إدارة المؤشرات

الملخص

#include <X11/Xcursor/Xcursor.h>

الوصف

Xcursor هي مكتبة بسيطة مصممة للمساعدة في تحديد وتحميل المؤشرات. يمكن تحميل المؤشرات من ملفات أو الذاكرة. توجد مكتبة للمؤشرات الشائعة التي تتوافق مع أسماء مؤشرات X القياسية. يمكن أن توجد المؤشرات بأحجام مختلفة وتختار المكتبة تلقائيًا أفضل حجم.

نظرة عامة وظيفية

تم بناء Xcursor على طبقات قليلة؛ في الطبقة السفلية يوجد رمز يمكنه تحميل صور المؤشرات من الملفات. فوق ذلك توجد طبقة تحدد مواقع ملفات المؤشرات بناءً على مسار المكتبة والموضوع. في الأعلى توجد طبقة تبني المؤشرات إما من صورة تم تحميلها من ملف أو من أحد مؤشرات X القياسية. عند استخدام الصور المحملة من الملفات، تفضل Xcursor استخدام طلب CreateCursor الخاص بامتداد Render إذا كان مدعومًا من خادم X. حيثما لم يكن مدعومًا، تقوم Xcursor بتعيين صورة المؤشر إلى مؤشر X قياسي وتستخدم طلب CreateCursor الأساسي.

ملفات المؤشرات

تحدد Xcursor تنسيقًا جديدًا للمؤشرات على القرص. كل ملف يحتوي على صورة مؤشر واحدة أو أكثر. كل صورة مؤشر موسومة بحجم اسمي بحيث يمكن اختيار أفضل حجم تلقائيًا. يمكن تحميل مؤشرات متعددة بنفس الحجم الاسمي معًا؛ من المتوقع أن تستخدم التطبيقاتها كسلسلة رسوم متحركة.

يتم تخزين ملفات المؤشرات كرأس يحتوي على جدول محتويات متبوعًا بسلسلة من القطع. يشير جدول المحتويات إلى النوع، النوع الفرعي، والموقع في الملف لكل قطعة. يبدو رأس الملف كالتالي:


magic:	CARD32 "Xcur" (0x58, 0x63, 0x75, 0x72)
header:	CARD32 بايت في هذا الرأس
version:	CARD32 رقم إصدار الملف
ntoc:	CARD32 عدد إدخالات جدول المحتويات
toc:	LISTofTOC جدول المحتويات

يبدو كل إدخال في جدول المحتويات كالتالي:


type:	CARD32 نوع الإدخال
subtype:	CARD32 تسمية خاصة بالنوع - حجم للصور
position:	CARD32 موقع البايت المطلق للجدول في الملف

تحتوي كل قطعة في الملف على مجموعة من الحقول الرأسية الشائعة متبوعة بحقول إضافية خاصة بالنوع:


header:	CARD32 بايت في رأس القطعة (بما في ذلك الحقول الخاصة بالنوع)
type:	CARD32 يجب أن يتطابق مع النوع في جدول المحتويات لهذه القطعة
subtype:	CARD32 يجب أن يتطابق مع النوع الفرعي في جدول المحتويات لهذه القطعة
version:	CARD32 رقم إصدار لنوع القطعة هذا

يوجد حاليًا نوعان من القطع معرفان لملفات المؤشرات؛ التعليقات والصور. تبدو التعليقات كالتالي:


header:	20 رؤوس التعليقات بحجم 20 بايت
type:	0xfffe0001 نوع التعليق هو 0xfffe0001
subtype:	{ 1 (حقوق النشر), 2 (الترخيص), 3 (آخر) }
version:	1
length:	CARD32 طول البايت لسلسلة UTF-8
string:	LISTofCARD8 سلسلة UTF-8

تبدو الصور كالتالي:


header:	36 رؤوس الصور بحجم 36 بايت
type:	0xfffd0002 نوع الصورة هو 0xfffd0002
subtype:	CARD32 نوع فرعي للصورة هو الحجم الاسمي
version:	1
width:	CARD32 يجب أن يكون أقل من أو يساوي 0x7fff
height:	CARD32 يجب أن يكون أقل من أو يساوي 0x7fff
xhot:	CARD32 يجب أن يكون أقل من أو يساوي العرض
yhot:	CARD32 يجب أن يكون أقل من أو يساوي الارتفاع
delay:	CARD32 التأخير بين إطارات الرسوم المتحركة بالمللي ثانية
pixels:	LISTofCARD32 بكسلات بتنسيق ARGB مجمعة

المواضيع

تتبع Xcursor (في الغالب) مواصفات freedesktop.org لمواضيع الأيقونات. مسار البحث الافتراضي الذي تستخدمه هو

~/.local/share/icons, ~/.icons, /usr/share/icons, /usr/share/pixmaps

داخل كل من هذه الدلائل، تبحث عن دليل باستخدام اسم الموضوع:

•
داخل دليل الموضوع، تبحث عن ملفات المؤشرات في الدليل الفرعي "cursors".
تبحث Xcursor عن ملف محدد، والذي يجب أن يكون أحد أسماء شكل المؤشر، على سبيل المثال، كما هو مستخدم في XcursorLibraryLoadImage أو XcursorLibraryShape.
•
إذا لم تجد ملف مؤشر مطابق في الدليل الفرعي "cursors"، تبحث Xcursor بعد ذلك عن ملف "index.theme" في كل دليل موضوع للبحث عن المواضيع الموروثة. تلك هي أسطر بهذا التنسيق:
Inherits = theme-name

تستخدم Xcursor أول theme-name موروث، متجاهلة الآخرين الذين قد يكونون موجودين في ملف "index.theme" معين. إذا وجدت موضوعًا موروثًا، فإن Xcursor تبحث على طول المسار لاستخدامه أيضًا. تتجاهل Xcursor المفاتيح الأخرى في ملف "index.theme"، بما في ذلك "Name" (أي، الاسم الذي قد يستخدمه التطبيق الرسومي كـ اسم عرض).

يمكن إدراج أكثر من theme-name واحد في سطر Inherits=. تنص مواصفات freedesktop.org على أن عناصر القائمة مفصولة بفواصل. تقبل Xcursor أيضًا الفاصلة المنقوطة، ولكنها تترجم كلاهما إلى نقطتين رأسيتين عند البحث في المسار. تتوقع Xcursor سطرًا واحدًا فقط من Inherits=؛ مواصفات freedesktop.org غير واضحة فيما إذا كانت المفاتيح المتعددة مسموح بها.

إذا لم يتم تعيين موضوع، أو إذا لم يتم العثور على مؤشر للموضوع المحدد في أي مكان على طول المسار، فإن Xcursor تتحقق من موضوع "default".

عندما تجد Xcursor ملف مؤشر، فإنها تتوقف عن البحث. تستخدم دائمًا ملف المؤشر الأول الذي تم العثور عليه أثناء البحث على طول المسار.

أنواع البيانات

XcursorImage
يحتوي على صورة مؤشر واحدة في الذاكرة. كل بكسل في المؤشر هو قيمة 32 بت تحتوي على ARGB مع A في البايت الأعلى. 


typedef struct _XcursorImage {
    XcursorDim 	 size; 	/* حجم اسمي للمطابقة */
    XcursorDim 	 width; 	/* العرض الفعلي */
    XcursorDim 	 height; 	/* الارتفاع الفعلي */
    XcursorDim 	 xhot; 	/* نقطة التفعيل x (يجب أن تكون داخل الصورة) */
    XcursorDim 	 yhot; 	/* نقطة التفعيل y (يجب أن تكون داخل الصورة) */
    XcursorPixel 	*pixels; 	/* مؤشر إلى البكسلات */
} XcursorImage;
XcursorImages
يحتوي على هياكل XcursorImage متعددة. يتم تحريرها جميعًا عند تحرير XcursorImages في XcursorImagesDestroy. 


typedef struct _XcursorImages {
    int 	 nimage; 	/* عدد الصور */
    XcursorImage 	**images; 	/* مصفوفة من مؤشرات XcursorImage */
} XcursorImages;
XcursorCursors
يحتوي على كائنات مؤشر متعددة. يتم تحريرها جميعًا عند تحرير XcursorCursors. هذه مرقمة بالمرجع بحيث يمكن لهياكل XcursorAnimate متعددة استخدام نفس XcursorCursors. 


typedef struct _XcursorCursors {
    Display 	*dpy; 	/* الشاشة التي تحتوي على المؤشرات */
    int 	 ref; 	/* عدد المراجع */
    int 	 ncursor; 	/* عدد المؤشرات */
    Cursor 	*cursors; 	/* مصفوفة من المؤشرات */
} XcursorCursors;
XcursorAnimate
يشير إلى مجموعة من المؤشرات وتسلسل داخل تلك المجموعة. قد تشير هياكل XcursorAnimate متعددة إلى نفس XcursorCursors؛ يحتفظ كل منها بمرجع يتم إزالته عند تحرير XcursorAnimate. 


typedef struct _XcursorAnimate {
    XcursorCursors 	*cursors; 	/* قائمة المؤشرات المراد استخدامها */
    int 	 sequence; 	/* أي مؤشر هو التالي */
} XcursorAnimate;
XcursorFile
توفر Xcursor واجهة برمجة تطبيقات مجردة للوصول إلى بيانات الملف. توفر Xcursor تنفيذًا قياسيًا لواجهة برمجة التطبيقات المجردة هذه؛ التطبيقات حرة في إنشاء تطبيقات إضافية. هذه الوظائف موازية لواجهة برمجة التطبيقات القياسية.

هذا الموقع غير مرتبط بـ Arch Linux ولا تابع له بأي شكل من الأشكال. المحتوى هو ترجمة فقط لصفحات دليل Arch المنشورة على man.archlinux.org.

XcursorFile

A structure used for reading and writing cursor files. The read and write functions are expected to flip the arguments around to match the POSIX versions; the read and write functions flip the arguments around to match the POSIX versions.


typedef struct _XcursorFile {
    void *closure;
    int (*read)  (XcursorFile *file, unsigned char *buf, int len);
    int (*write) (XcursorFile *file, unsigned char *buf, int len);
    int (*seek)  (XcursorFile *file, long offset, int whence);
};

تبحث هذه الوظائف في مسار المكتبة، وتحمل أول ملف يتم العثور عليه بالحجم المطلوب، باستخدام وظيفة خاصة (XcursorScanTheme) للعثور على السمة المناسبة:

  • إذا كان theme ليس NULL، استخدمه.
  • إذا كان theme NULL، أو إذا لم يكن هناك تطابق للسمة المطلوبة، استخدم "default" لاسم السمة.
  • إذا فشل كلا البحثين، تُرجع هذه الوظائف NULL.
تختلف الوظيفتان بأكثر من عدد الصور المحملة:
  • XcursorLibraryLoadImage تستدعي XcursorFileLoadImage ولكن
  • XcursorLibraryLoadImages تستدعي XcursorFileLoadImages، و (عند النجاح) تستدعي XcursorImagesSetName لربط name بالنتيجة.
