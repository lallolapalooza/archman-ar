الحزمة: libxcursor

التراخيص: HPND-sell-variant

XCURSOR(3) إصدار X 11 XCURSOR(3)

الاسم

Xcursor - مكتبة إدارة المؤشرات

ملخص

#include <X11/Xcursor/Xcursor.h>

الوصف

Xcursor هي مكتبة بسيطة مصممة للمساعدة في تحديد مواقع وتحميل المؤشرات. يمكن تحميل المؤشرات من الملفات أو الذاكرة. توجد مكتبة للمؤشرات الشائعة التي ترتبط بأسماء مؤشرات X القياسية. يمكن أن توجد المؤشرات بأحجام متعددة والمكتبة تختار تلقائيًا أفضل حجم.

نظرة عامة وظيفية

تم بناء Xcursor على عدة طبقات؛ في الطبقة السفلية توجد الشيفرة التي يمكنها تحميل صور المؤشرات من الملفات. فوق ذلك توجد طبقة تحدد مواقع ملفات المؤشرات بناءً على مسار المكتبة والموضوع. في الأعلى توجد طبقة تبني المؤشرات إما من صورة محملة من ملف أو أحد مؤشرات X القياسية. عند استخدام الصور المحملة من ملفات، تفضل Xcursor استخدام طلب إنشاء المؤشر الخاص بامتداد Render إذا كان مدعومًا بواسطة خادم X. حيث لا يكون مدعومًا، تقوم Xcursor بربط صورة المؤشر بمؤشر X قياسي وتستخدم طلب إنشاء المؤشر الأساسي.

ملفات المؤشرات

تحدد Xcursor تنسيقًا جديدًا للمؤشرات على القرص. كل ملف يحمل صورة مؤشر واحدة أو أكثر. كل صورة مؤشر تمييز بحجم اسمي بحيث يمكن اختيار أفضل حجم تلقائيًا. يمكن تحميل مؤشرات متعددة بنفس الحجم الاسمي معًا؛ من المتوقع أن تستخدمها التطبيقات كسلسلة رسوم متحركة.

تُخزن ملفات المؤشرات كرأس يحتوي على جدول محتويات متبوعًا بتسلسل من المقاطع (chunks). يشير جدول المحتويات إلى النوع، والنوع الفرعي، والموضع في الملف لكل مقطع. يبدو رأس الملف كالتالي:


magic:	CARD32 "Xcur" (0x58, 0x63, 0x75, 0x72)
header:	CARD32 بايت في هذا الرأس
version:	CARD32 رقم إصدار الملف
ntoc:	CARD32 عدد إدخالات جدول المحتويات
toc:	LISTofTOC جدول المحتويات

كل إدخال في جدول المحتويات يبدو كالتالي:


type:	CARD32 نوع الإدخال
subtype:	CARD32 تسمية خاصة بالنوع - حجم للصور
position:	CARD32 الموضع المطلق بالبايت للمحتوى في الملف

كل مقطع في الملف لديه مجموعة من حقول الرأس الشائعة متبوعة بحقول إضافية خاصة بالنوع:


header:	CARD32 بايت في رأس المقطع (بما في ذلك الحقول الخاصة بالنوع)
type:	CARD32 يجب أن يتطابق مع النوع في TOC لهذا المقطع
subtype:	CARD32 يجب أن يتطابق مع النوع الفرعي في TOC لهذا المقطع
version:	CARD32 رقم إصدار لهذا النوع من المقاطع

يوجد حاليًا نوعان من المقاطع معرفان لملفات المؤشرات؛ التعليقات والصور. التعليقات تبدو كالتالي:


header:	20 رؤوس التعليقات بحجم 20 بايت
type:	0xfffe0001 نوع التعليق هو 0xfffe0001
subtype:	{ 1 (COPYRIGHT), 2 (LICENSE), 3 (OTHER) }
version:	1
length:	CARD32 طول البايت للسلسلة UTF-8
string:	LISTofCARD8 سلسلة UTF-8

الصور تبدو كالتالي:


header:	36 رؤوس الصور بحجم 36 بايت
type:	0xfffd0002 نوع الصورة هو 0xfffd0002
subtype:	CARD32 النوع الفرعي للصورة هو الحجم الاسمي
version:	1
width:	CARD32 يجب أن يكون أقل من أو يساوي 0x7fff
height:	CARD32 يجب أن يكون أقل من أو يساوي 0x7fff
xhot:	CARD32 يجب أن يكون أقل من أو يساوي العرض
yhot:	CARD32 يجب أن يكون أقل من أو يساوي الارتفاع
delay:	CARD32 التأخير بين إطارات الرسوم المتحركة بالمللي ثانية
pixels:	LISTofCARD32 بكسلات بتنسيق ARGB معبأة

الموضوعات

تتبع Xcursor (في الغالب) مواصفات freedesktop.org لتنظيم أيقونات الموضوعات. مسار البحث الافتراضي الذي تستخدمه هو

~/.local/share/icons, ~/.icons, /usr/share/icons, /usr/share/pixmaps

داخل كل من هذه الدلائل، تبحث عن دليل باستخدام اسم الموضوع:

•
داخل دليل الموضوع، تبحث عن ملفات المؤشرات في الدليل الفرعي "cursors".
تبحث Xcursor عن ملف محدد، يجب أن يكون أحد أسماء *شكل* المؤشر، على سبيل المثال، كما هو مستخدم في XcursorLibraryLoadImage أو XcursorLibraryShape.
•
إذا لم تجد ملف مؤشر مطابق في الدليل الفرعي "cursors"، تبحث Xcursor بعد ذلك عن ملف "index.theme" في كل دليل موضوع للبحث عن الموضوعات الموروثة. هذه هي أسطر بهذا التنسيق:
Inherits = اسم-الموضوع

تستخدم Xcursor أول *اسم-موضوع* موروث، متجاهلة العناصر الأخرى التي قد تكون موجودة في ملف "index.theme" معين. إذا وجدت موضوعًا موروثًا، تبحث Xcursor على طول المسار لاستخدامه أيضًا. تتجاهل Xcursor المفاتيح * الأخرى* في ملف "index.theme"، بما في ذلك "Name" (أي، الاسم الذي قد يستخدمه التطبيق الرسومي *اسم العرض*).

يمكن سرد أكثر من *اسم-موضوع* واحد في سطر Inherits=. تنص مواصفات freedesktop.org على أن عناصر القائمة مفصولة بفواصل. تقبل Xcursor أيضًا الفاصلة المنقوطة، ولكنها تترجم كلاهما إلى نقطتين عند البحث في المسار. تتوقع Xcursor سطر Inherits= واحد فقط؛ مواصفات freedesktop.org غير واضحة فيما إذا كانت المفاتيح المتعددة مسموح بها.

إذا لم يتم تعيين موضوع، أو إذا لم يتم العثور على مؤشر للموضوع المحدد في أي مكان على طول المسار، تتحقق Xcursor من موضوع "default".

عندما تجد Xcursor ملف مؤشر، تتوقف عن البحث. تستخدم دائمًا أول ملف مؤشر يتم العثور عليه أثناء البحث على طول المسار.

أنواع البيانات

XcursorImage
يحتوي على صورة مؤشر واحدة في الذاكرة. كل بكسل في المؤشر هو قيمة 32 بت تحتوي على ARGB مع A في البايت الأعلى. 


typedef struct _XcursorImage {
    XcursorDim 	 size; 	/* حجم اسمي للمطابقة */
    XcursorDim 	 width; 	/* العرض الفعلي */
    XcursorDim 	 height; 	/* الارتفاع الفعلي */
    XcursorDim 	 xhot; 	/* نقطة x الساخنة (يجب أن تكون داخل الصورة) */
    XcursorDim 	 yhot; 	/* نقطة y الساخنة (يجب أن تكون داخل الصورة) */
    XcursorPixel 	*pixels; 	/* مؤشر إلى البكسلات */
} XcursorImage;
XcursorImages
يحتوي على هياكل XcursorImage متعددة. يتم تحريرها جميعًا عند تحرير XcursorImages في XcursorImagesDestroy. 


typedef struct _XcursorImages {
    int 	 nimage; 	/* عدد الصور */
    XcursorImage 	**images; 	/* مصفوفة من مؤشرات XcursorImage */
} XcursorImages;
XcursorCursors
يحتوي على كائنات مؤشر متعددة. يتم تحريرها جميعًا عند تحرير XcursorCursors. هذه مع عد مراجع بحيث يمكن لهياكل XcursorAnimate متعددة استخدام نفس XcursorCursors. 


typedef struct _XcursorCursors {
    Display 	*dpy; 	/* شاشة تحتوي على المؤشرات */
    int 	 ref; 	/* عد المراجع */
    int 	 ncursor; 	/* عدد المؤشرات */
    Cursor 	*cursors; 	/* مصفوفة المؤشرات */
} XcursorCursors;
XcursorAnimate
يشير إلى مجموعة من المؤشرات وتسلسل داخل تلك المجموعة. قد تشير هياكل XcursorAnimate متعددة إلى نفس XcursorCursors؛ كل منها يحتفظ بمرجع يتم إزالته عند تحرير XcursorAnimate. 


typedef struct _XcursorAnimate {
    XcursorCursors 	*cursors; 	/* قائمة المؤشرات للاستخدام */
    int 	 sequence; 	/* أي مؤشر هو التالي */
} XcursorAnimate;
XcursorFile
توفر Xcursor واجهة برمجة تطبيقات مجردة للوصول إلى بيانات الملف. توفر Xcursor تطبيق stdio لهذه الواجهة المجردة؛ التطبيقات حرة في إنشاء تطبيقات إضافية. هذه الوظائف توازي stdio

هذا الموقع غير مرتبط بـ Arch Linux ولا تابع له بأي شكل من الأشكال. المحتوى هو ترجمة فقط لصفحات دليل Arch المنشورة على man.archlinux.org.
