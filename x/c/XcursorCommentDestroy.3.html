XCURSOR(3) إصدار X 11 XCURSOR(3)

الاسم

Xcursor - مكتبة إدارة المؤشرات

الملخص

#include <X11/Xcursor/Xcursor.h>

الوصف

Xcursor هي مكتبة بسيطة مصممة للمساعدة في تحديد وتحميل المؤشرات. يمكن تحميل المؤشرات من الملفات أو الذاكرة. توجد مكتبة للمؤشرات الشائعة التي تتوافق مع أسماء مؤشرات X القياسية. يمكن أن توجد المؤشرات بأحجام مختلفة والمكتبة تختار تلقائيًا أفضل حجم.

نظرة عامة وظيفية

تم بناء Xcursor على طبقات؛ في الطبقة السفلية توجد التعليمات البرمجية التي يمكنها تحميل صور المؤشرات من الملفات. فوق ذلك توجد طبقة تحدد مواقع ملفات المؤشرات بناءً على مسار المكتبة والموضوع. في القمة توجد طبقة تبني المؤشرات إما من صورة تم تحميلها من ملف أو من أحد مؤشرات X القياسية. عند استخدام الصور التي تم تحميلها من ملفات، تفضل Xcursor استخدام طلب CreateCursor الخاص بامتداد Render إذا كان مدعومًا من خادم X. حيث لا يكون مدعومًا، تقوم Xcursor بتعيين صورة المؤشر إلى مؤشر X قياسي وتستخدم طلب CreateCursor الأساسي.

ملفات المؤشرات

تحدد Xcursor تنسيقًا جديدًا للمؤشرات على القرص. يحتوي كل ملف على صورة مؤشر واحدة أو أكثر. يتم تمييز كل صورة مؤشر بحجم اسمي بحيث يمكن اختيار أفضل حجم تلقائيًا. يمكن تحميل مؤشرات متعددة بنفس الحجم الاسمي معًا؛ من المتوقع أن تستخدمها التطبيقات كتسلسل رسوم متحركة.

يتم تخزين ملفات المؤشرات كرأس يحتوي على جدول محتويات متبوعًا بتسلسل من الأجزاء. يشير جدول المحتويات إلى النوع والنوع الفرعي والموضع في الملف لكل جزء. يبدو رأس الملف كما يلي:


magic:	CARD32 "Xcur" (0x58, 0x63, 0x75, 0x72)
header:	CARD32 بايت في هذا الرأس
version:	CARD32 رقم إصدار الملف
ntoc:	CARD32 عدد إدخالات جدول المحتويات
toc:	LISTofTOC جدول المحتويات

كل إدخال في جدول المحتويات يبدو كما يلي:


type:	CARD32 نوع الإدخال
subtype:	CARD32 تسمية خاصة بالنوع - حجم للصور
position:	CARD32 موضع البايت المطلق للجدول في الملف

كل جزء في الملف يحتوي على مجموعة من رؤوس الحقول المشتركة متبوعة بحقول إضافية خاصة بالنوع:


header:	CARD32 بايت في رأس الجزء (بما في ذلك الحقول الخاصة بالنوع)
type:	CARD32 يجب أن يتطابق مع النوع في جدول المحتويات لهذا الجزء
subtype:	CARD32 يجب أن يتطابق مع النوع الفرعي في جدول المحتويات لهذا الجزء
version:	CARD32 رقم إصدار لنوع هذا الجزء

يوجد حاليًا نوعان من الأجزاء معرفان لملفات المؤشرات؛ التعليقات والصور. تبدو التعليقات كما يلي:


header:	20 رؤوس التعليقات بحجم 20 بايت
type:	0xfffe0001 نوع التعليق هو 0xfffe0001
subtype:	{ 1 (حقوق النشر)، 2 (الترخيص)، 3 (أخرى) }
version:	1
length:	CARD32 طول البايت للسلسلة النصية UTF-8
string:	LISTofCARD8 سلسلة نصية UTF-8

تبدو الصور كما يلي:


header:	36 رؤوس الصور بحجم 36 بايت
type:	0xfffd0002 نوع الصورة هو 0xfffd0002
subtype:	CARD32 نوع الصورة الفرعي هو الحجم الاسمي
version:	1
width:	CARD32 يجب أن يكون أقل من أو يساوي 0x7fff
height:	CARD32 يجب أن يكون أقل من أو يساوي 0x7fff
xhot:	CARD32 يجب أن يكون أقل من أو يساوي العرض
yhot:	CARD32 يجب أن يكون أقل من أو يساوي الارتفاع
delay:	CARD32 التأخير بين إطارات الرسوم المتحركة بالمللي ثانية
pixels:	LISTofCARD32 بكسلات بتنسيق ARGB مضغوط

المواضيع

تتبع Xcursor (في الغالب) مواصفات freedesktop.org لموضوعات الأيقونات. مسار البحث الافتراضي الذي تستخدمه هو

~/.local/share/icons, ~/.icons, /usr/share/icons, /usr/share/pixmaps

داخل كل من هذه الدلائل، تبحث عن دليل باستخدام اسم الموضوع:

•
داخل دليل الموضوع، تبحث عن ملفات المؤشرات في الدليل الفرعي "cursors".
تبحث Xcursor عن ملف معين، يجب أن يكون أحد أسماء shape للمؤشر، كما هو مستخدم في XcursorLibraryLoadImage أو XcursorLibraryShape.
•
إذا لم تجد ملف مؤشر مطابق في الدليل الفرعي "cursors"، فإن Xcursor تبحث بعد ذلك عن ملف "index.theme" في كل دليل موضوع للبحث عن مواضيع موروثة. هذه أسطر بهذا التنسيق:
Inherits = اسم-الموضوع

تستخدم Xcursor أول اسم-الموضوع موروث، متجاهلة الآخرين الذين قد يكونون موجودين في ملف "index.theme" معين. إذا وجدت موضوعًا موروثًا، فإن Xcursor تبحث على طول المسار لاستخدامه أيضًا. تتجاهل Xcursor مفاتيح أخرى في ملف "index.theme"، بما في ذلك "Name" (أي، الاسم الذي قد يستخدمه التطبيق الرسومي كـ اسم عرض).

قد يتم إدراج أكثر من اسم-الموضوع واحد في سطر Inherits=. تنص مواصفات freedesktop.org على أن عناصر القائمة مفصولة بفواصل. تقبل Xcursor أيضًا الفاصلة المنقوطة، ولكنها تترجم كليهما إلى نقطتين عند البحث في المسار. تتوقع Xcursor سطر Inherits= واحد فقط؛ المواصفات freedesktop.org غير واضحة ما إذا كانت المفاتيح المتعددة مسموح بها.

إذا لم يتم تعيين موضوع، أو إذا لم يتم العثور على مؤشر للموضوع المحدد في أي مكان على طول المسار، تتحقق Xcursor من موضوع "default".

عندما تجد Xcursor ملف مؤشر، فإنها تتوقف عن البحث. تستخدم دائمًا أول ملف مؤشر تم العثور عليه أثناء البحث على طول المسار.

أنواع البيانات

XcursorImage
يحتوي على صورة مؤشر واحدة في الذاكرة. كل بكسل في المؤشر هو قيمة 32 بت تحتوي على ARGB مع A في البايت الأعلى. 


typedef struct _XcursorImage {
    XcursorDim 	 size; 	/* الحجم الاسمي للمطابقة */
    XcursorDim 	 width; 	/* العرض الفعلي */
    XcursorDim 	 height; 	/* الارتفاع الفعلي */
    XcursorDim 	 xhot; 	/* نقطة الاتصال x (يجب أن تكون داخل الصورة) */
    XcursorDim 	 yhot; 	/* نقطة الاتصال y (يجب أن تكون داخل الصورة) */
    XcursorPixel 	*pixels; 	/* مؤشر إلى وحدات البكسل */
} XcursorImage;
XcursorImages
يحتوي على هياكل XcursorImage متعددة. يتم تحريرها جميعًا عند تحرير XcursorImages في XcursorImagesDestroy. 


typedef struct _XcursorImages {
    int 	 nimage; 	/* عدد الصور */
    XcursorImage 	**images; 	/* مصفوفة من مؤشرات XcursorImage */
} XcursorImages;
XcursorCursors
يحتوي على كائنات مؤشر متعددة. يتم تحريرها جميعًا عند تحرير XcursorCursors. هذه محسوبة المرجع بحيث يمكن لهياكل XcursorAnimate متعددة استخدام نفس XcursorCursors. 


typedef struct _XcursorCursors {
    Display 	*dpy; 	/* شاشة عرض تحتوي على المؤشرات */
    int 	 ref; 	/* عداد المرجع */
    int 	 ncursor; 	/* عدد المؤشرات */
    Cursor 	*cursors; 	/* مصفوفة المؤشرات */
} XcursorCursors;
XcursorAnimate
يشير إلى مجموعة من المؤشرات وتسلسل ضمن تلك المجموعة. قد تشير هياكل XcursorAnimate متعددة إلى نفس XcursorCursors؛ كل منها يحتفظ بمرجع يتم إزالته عند تحرير XcursorAnimate. 


typedef struct _XcursorAnimate {
    XcursorCursors 	*cursors; 	/* قائمة المؤشرات للاستخدام */
    int 	 sequence; 	/* أي مؤشر هو التالي */
} XcursorAnimate;
