xcb_query_tree_unchecked(3)

الحزمة: libxcb

التراخيص: X11

xcb_query_tree(3) طلبات XCB xcb_query_tree(3)

الاسم

xcb_query_tree - الاستعلام عن شجرة النافذة

ملخص

#include <xcb/xproto.h>

دالة الطلب

xcb_query_tree_cookie_t xcb_query_tree(xcb_connection_t *conn, xcb_window_t window);

هيكل بيانات الرد

typedef struct xcb_query_tree_reply_t {
    uint8_t      response_type;
    uint8_t      pad0;
    uint16_t     sequence;
    uint32_t     length;
    xcb_window_t root;
    xcb_window_t parent;
    uint16_t     children_len;
    uint8_t      pad1[14];
} xcb_query_tree_reply_t;

دالة الرد

xcb_query_tree_reply_t *xcb_query_tree_reply(xcb_connection_t *conn, xcb_query_tree_cookie_t cookie, xcb_generic_error_t **e);

مُستكشفات الرد

xcb_window_t *xcb_query_tree_children(const xcb_query_tree_request_t *reply);

int xcb_query_tree_children_length(const xcb_query_tree_reply_t *reply);

xcb_generic_iterator_t xcb_query_tree_children_end(const xcb_query_tree_reply_t *reply);

وسيطات الطلب

conn
اتصال XCB بـ X11.
window
النافذة window للاستعلام عنها.

حقول الرد

response_type
نوع هذا الرد، في هذه الحالة XCB_QUERY_TREE. هذا الحقل موجود أيضًا في xcb_generic_reply_t ويمكن استخدامه للتمييز بين الردود.
sequence
رقم تسلسل آخر طلب تم معالجته بواسطة خادم X11.
length
طول الرد، بالكلمات (الكلمة 4 بايت).
root
النافذة الجذرية window.
parent
النافذة الأم window.
children_len
عدد النوافذ الفرعية.

الوصف

يحصل على معرف النافذة الجذرية، ومعرف النافذة الأم، وقائمة النوافذ الفرعية للنافذة المحددة window. يتم سرد النوافذ الفرعية بترتيب التكديس من الأسفل إلى الأعلى.

قيمة الإرجاع

يرجع xcb_query_tree_cookie_t. يجب التعامل مع الأخطاء عند استدعاء دالة الرد xcb_query_tree_reply.

إذا كنت ترغب في التعامل مع الأخطاء في حلقة الأحداث بدلاً من ذلك، استخدم xcb_query_tree_unchecked. انظر xcb-requests(3) للتفاصيل.

الأخطاء

هذا الطلب لا يولد أي أخطاء على الإطلاق.

مثال

/*
 * يعرض النافذة الجذرية، النافذة الأم، والنوافذ الفرعية للنافذة المحددة.
 *
 */
void my_example(xcb_connection_t *conn, xcb_window_t window) {
    xcb_query_tree_cookie_t cookie;
    xcb_query_tree_reply_t *reply;
    cookie = xcb_query_tree(conn, window);
    if ((reply = xcb_query_tree_reply(conn, cookie, NULL))) {
        printf("root = 0x%08x\n", reply->root);
        printf("parent = 0x%08x\n", reply->parent);
        xcb_window_t *children = xcb_query_tree_children(reply);
        for (int i = 0; i < xcb_query_tree_children_length(reply); i++)
            printf("child window = 0x%08x\n", children[i]);
        free(reply);
    }
}
