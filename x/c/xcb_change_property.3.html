xcb_change_property(3)

الحزمة: libxcb

التراخيص: X11

xcb_change_property(3) طلبات XCB xcb_change_property(3)

الاسم

xcb_change_property - يغير خاصية النافذة

الملخص

#include <xcb/xproto.h>

وظيفة الطلب

xcb_void_cookie_t xcb_change_property(xcb_connection_t *conn, uint8_t mode, xcb_window_t window, xcb_atom_t property, xcb_atom_t type, uint8_t format, uint32_t data_len, const void *data);

وسيطات الطلب

conn
اتصال XCB بـ X11.
mode
أحد القيم التالية:
XCB_PROP_MODE_REPLACE
تجاهل قيمة الخاصية السابقة وتخزين البيانات الجديدة.
XCB_PROP_MODE_PREPEND
أدرج البيانات الجديدة قبل بداية البيانات الموجودة. يجب أن يتطابق format مع قيمة الخاصية الموجودة. إذا كانت الخاصية غير معرفة، يتم التعامل معها على أنها معرفة بنوع وتنسيق صحيحين مع بيانات بطول صفر.
XCB_PROP_MODE_APPEND
أدرج البيانات الجديدة بعد بداية البيانات الموجودة. يجب أن يتطابق format مع قيمة الخاصية الموجودة. إذا كانت الخاصية غير معرفة، يتم التعامل معها على أنها معرفة بنوع وتنسيق صحيحين مع بيانات بطول صفر.
window
النافذة التي تريد تغيير خاصيتها.
property
الخاصية التي تريد تغييرها (ذرة).
type
نوع الخاصية التي تريد تغييرها (ذرة).
format
يحدد ما إذا كان يجب النظر إلى البيانات كقائمة من الكميات 8 بت، 16 بت أو 32 بت. القيم الممكنة هي 8، 16 و 32. هذه المعلومات تسمح لخادم X بتنفيذ عمليات تبديل البايت بشكل صحيح عند الضرورة.
data_len
يحدد عدد العناصر (انظر format).
data
بيانات الخاصية.

الوصف

يضبط أو يحدث خاصية على window المحدد. الخصائص هي على سبيل المثال عنوان النافذة (WM_NAME) أو حجمها الأدنى (WM_NORMAL_HINTS). تستخدم البروتوكولات مثل EWMH أيضًا خصائص - على سبيل المثال، يحدد EWMH عنوان النافذة، مشفر كسلسلة UTF-8، في الخاصية _NET_WM_NAME.

القيمة المرجعة

يعيد xcb_void_cookie_t. يجب معالجة الأخطاء (إن وجدت) في حلقة الأحداث.

إذا كنت ترغب في معالجة الأخطاء مباشرة باستخدام xcb_request_check بدلاً من ذلك، استخدم xcb_change_property_checked. انظر xcb-requests(3) للتفاصيل.

الأخطاء

xcb_alloc_error_t
لم يتمكن خادم X من تخزين الخاصية (لا توجد ذاكرة؟).
xcb_atom_error_t
property أو type لا تشير إلى ذرة صالحة.
xcb_match_error_t

xcb_value_error_t
TODO: أسباب؟
xcb_window_error_t
النافذة window المحددة غير موجودة.

مثال

/*
 * يضبط خاصية WM_NAME للنافذة إلى "XCB Example".
 *
 */
void my_example(xcb_connection_t *conn, xcb_window_t window) {
    xcb_change_property(conn,
        XCB_PROP_MODE_REPLACE,
        window,
        XCB_ATOM_WM_NAME,
        XCB_ATOM_STRING,
        8,
        strlen("XCB Example"),
        "XCB Example");
    xcb_flush(conn);
}
