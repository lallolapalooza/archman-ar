XcmsRGB(3)

الحزمة: libx11

التراخيص: MIT AND X11

XcmsColor(3) وظائف XLIB XcmsColor(3)

الاسم

XcmsColor, XcmsRGB, XcmsRGBi, XcmsCIEXYZ, XcmsCIEuvY, XcmsCIExyY, XcmsCIELab, XcmsCIELuv, XcmsTekHVC, XcmsPad - هيكل لون Xcms

الهياكل

يحتوي هيكل XcmsColor على:

typedef unsigned long XcmsColorFormat;			/* تنسيق مواصفات اللون */
typedef struct {
	union {
		XcmsRGB RGB;
		XcmsRGBi RGBi;
		XcmsCIEXYZ CIEXYZ;
		XcmsCIEuvY CIEuvY;
		XcmsCIExyY CIExyY;
		XcmsCIELab CIELab;
		XcmsCIELuv CIELuv;
		XcmsTekHVC TekHVC;
		XcmsPad Pad;
	} spec;
	unsigned long pixel;
	XcmsColorFormat format;
} XcmsColor;			/* هيكل لون Xcms */
typedef double XcmsFloat;
typedef struct {
	unsigned short red;	/* 0x0000 إلى 0xffff */
	unsigned short green;	/* 0x0000 إلى 0xffff */
	unsigned short blue;	/* 0x0000 إلى 0xffff */
} XcmsRGB;		/* جهاز RGB */
typedef struct {
	XcmsFloat red;	/* 0.0 إلى 1.0 */
	XcmsFloat green;	/* 0.0 إلى 1.0 */
	XcmsFloat blue;	/* 0.0 إلى 1.0 */
} XcmsRGBi;		/* شدة RGB */
typedef struct {
	XcmsFloat X;
	XcmsFloat Y;	/* 0.0 إلى 1.0 */
	XcmsFloat Z;
} XcmsCIEXYZ;		/* CIE XYZ */
typedef struct {
	XcmsFloat u_prime;	/* 0.0 إلى ~0.6 */
	XcmsFloat v_prime;	/* 0.0 إلى ~0.6 */
	XcmsFloat Y; 	/* 0.0 إلى 1.0 */
} XcmsCIEuvY;		/* CIE u'v'Y */
typedef struct {
	XcmsFloat x; 	/* 0.0 إلى ~.75 */
	XcmsFloat y; 	/* 0.0 إلى ~.85 */
	XcmsFloat Y; 	/* 0.0 إلى 1.0 */
} XcmsCIExyY;		/* CIE xyY */
typedef struct {
	XcmsFloat L_star; 	/* 0.0 إلى 100.0 */
	XcmsFloat a_star;
	XcmsFloat b_star;
} XcmsCIELab;		/* CIE L*a*b* */
typedef struct {
	XcmsFloat L_star; 	/* 0.0 إلى 100.0 */
	XcmsFloat u_star;
	XcmsFloat v_star;
} XcmsCIELuv;		/* CIE L*u*v* */
typedef struct {
	XcmsFloat H; 	/* 0.0 إلى 360.0 */
	XcmsFloat V; 	/* 0.0 إلى 100.0 */
	XcmsFloat C; 	/* 0.0 إلى 100.0 */
} XcmsTekHVC;		/* TekHVC */
typedef struct {
	XcmsFloat pad0;
	XcmsFloat pad1;
	XcmsFloat pad2;
	XcmsFloat pad3;
} XcmsPad;		/* أربعة أعداد عشرية مزدوجة */

الوصف

يحتوي هيكل XcmsColor على اتحاد من الهياكل الفرعية، يدعم كل منها ترميز مواصفات اللون لمساحة لون معينة.
