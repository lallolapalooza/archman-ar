XChangeFeedbackControl(3)

الحزمة: libxi

التراخيص: MIT-open-group, SMLNJ, MIT

XGETFEEDBACKCONTROL(3)   XGETFEEDBACKCONTROL(3)

الاسم

XGetFeedbackControl, XChangeFeedbackControl - الاستعلام عن ردود فعل جهاز الإدخال وتغييرها

الملخص

#include <X11/extensions/XInput.h>
XFeedbackState *XGetFeedbackControl( Display *display,
                                     XDevice *device,
                                     int *num_feedbacks);
int XChangeFeedbackControl( Display *display,
                            XDevice *device,
                            Mask mask,
                            XFeedbackControl *control);
display
       يحدد الاتصال بخادم X.
device
       يحدد الجهاز الذي سيتم الاستعلام عن ردود فعله
       أو تعديلها.
num_feedbacks
       يحدد عنوانًا سيتم إرجاع عدد ردود الفعل
       المدعومة بواسطة الجهاز إليه.
mask
       يحدد قناعًا خاصًا بكل نوع من ردود الفعل يصف كيفية
       تعديل رد الفعل.
control
       يحدد عنوان بنية XFeedbackControl التي تحتوي على القيم الجديدة
       لرد الفعل.

الوصف

تم توفير هذه الطلبات لمعالجة أجهزة الإدخال التي تدعم ردود الفعل. سيتم إنشاء خطأ BadMatch إذا كان الجهاز المطلوب لا يدعم ردود الفعل. ما إذا كان جهاز معين يدعم ردود الفعل أم لا يمكن تحديده بفحص المعلومات التي تم إرجاعها بواسطة طلب XOpenDevice. بالنسبة للأجهزة التي تدعم ردود الفعل، سيعيد XOpenDevice بنية XInputClassInfo مع حقل input_class مساويًا للثابت FeedbackClass (المعرف في الملف XI.h).
يعيد طلب XGetFeedbackControl مؤشرًا إلى قائمة من هياكل XFeedbackState. يصف كل عنصر في هذه القائمة إحدى ردود الفعل المدعومة بواسطة الجهاز. العناصر متغيرة الطول، لذا يحتوي كل منها على طوله للسماح بالانتقال إلى العنصر التالي في القائمة.
فئات ردود الفعل المعرفة حاليًا هي: KbdFeedbackClass، PtrFeedbackClass، StringFeedbackClass، IntegerFeedbackClass، LedFeedbackClass، و BellFeedbackClass. هذه الثوابت معرفة في الملف XI.h. قد يدعم جهاز الإدخال صفرًا أو أكثر من فئات ردود الفعل، وقد يدعم ردود فعل متعددة من نفس الفئة. كل رد فعل يحتوي على معرف فئة ومعرف فريد ضمن تلك الفئة لجهاز الإدخال هذا. يتم استخدام المعرف لتحديد رد الفعل عند إجراء طلب XChangeFeedbackControl.
يمكن لـ XGetFeedbackControl إنشاء خطأ BadDevice أو BadMatch.
يعدل طلب XChangeFeedbackControl قيم رد فعل واحد على الجهاز المحدد. يتم تحديد رد الفعل بواسطة حقل المعرف لبنية XFeedbackControl التي يتم تمريرها مع الطلب. يتم تحديد حقول رد الفعل التي سيتم تعديلها بواسطة بتات القناع الذي يتم تمريره مع الطلب.
يمكن لـ XChangeFeedbackControl إنشاء خطأ BadDevice أو BadMatch أو BadValue.

الهياكل

يتم وصف كل فئة من ردود الفعل ببنية خاصة بتلك الفئة. هذه الهياكل معرفة في الملف XInput.h. XFeedbackState و XFeedbackControl هما هياكل عامة تحتوي على ثلاثة حقول في بداية كل فئة رد فعل:
typedef struct {
    XID class;
    int length;
    XID id;
} XFeedbackState, XFeedbackControl;
تحدد بنية XKbdFeedbackState السمات التي يتم إرجاعها لردود الفعل المكافئة لتلك الموجودة على لوحة مفاتيح X.
typedef struct {
    XID class;
    int length;
    XID id;
    int click;
    int percent;
    int pitch;
    int duration;
    int led_mask;
    int global_auto_repeat;
    char auto_repeats[32];
} XKbdFeedbackState;
تحدد بنية XPtrFeedbackState السمات التي يتم إرجاعها لردود الفعل المكافئة لتلك الموجودة على مؤشر X.
typedef struct {
    XID class;
    int length;
    XID id;
    int accelNum;
    int accelDenom;
    int threshold;
} XPtrFeedbackState;
تحدد بنية XIntegerFeedbackState السمات التي يتم إرجاعها لردود الفعل الرقمية.
typedef struct {
    XID class;
    int length;
    XID id;
    int resolution;
    int minVal;
    int maxVal;
} XIntegerFeedbackState;
تحدد بنية XStringFeedbackState السمات التي يتم إرجاعها لردود فعل السلاسل.
typedef struct {
    XID class;
    int length;
    XID id;
    int max_symbols;
    int num_syms_supported;
    KeySym *syms_supported;
} XStringFeedbackState;
تحدد بنية XBellFeedbackState السمات التي يتم إرجاعها لردود فعل الجرس.
typedef struct {
    XID class;
    int length;
    XID id;
    int percent;
    int pitch;
    int duration;
} XBellFeedbackState;
تحدد بنية XLedFeedbackState السمات التي يتم إرجاعها لردود فعل LED.
typedef struct {
    XID class;
    int length;
    XID id;
    int led_values;
} XLedFeedbackState;
تحدد بنية XPrtFeedbackControl السمات التي يمكن التحكم بها لردود فعل المؤشر.
typedef struct {
    XID class;
    int length;
    XID id;
    int accelNum;
    int accelDenom;
    int threshold;
} XPtrFeedbackControl;
تحدد بنية XKbdFeedbackControl السمات التي يمكن التحكم بها لردود فعل لوحة المفاتيح.
typedef struct {
    XID class;
    int length;
    XID id;
    int click;
    int percent;
    int pitch;
    int duration;
    int led_mask;
    int led_value;
    int key;
    int auto_repeat_mode;
} XKbdFeedbackControl;
تحدد بنية XStringFeedbackControl السمات التي يمكن التحكم بها لردود فعل السلاسل.
typedef struct {
    XID class;
    int length;
    XID id;
    int num_keysyms;
    KeySym  *syms_to_display;
} XStringFeedbackControl;
تحدد بنية XIntegerFeedbackControl السمات التي يمكن التحكم بها لردود الفعل الرقمية.
typedef struct {
    XID class;
    int length;
    XID id;

التشخيص

BadDevice
       تم تحديد جهاز غير صالح. الجهاز المحدد غير موجود أو لم يتم فتحه بواسطة هذا العميل عبر XOpenInputDevice. قد يحدث هذا الخطأ أيضًا إذا تسبب عميل آخر في أن يصبح الجهاز المحدد هو جهاز لوحة المفاتيح X أو جهاز المؤشر X عبر طلبات XChangeKeyboardDevice أو XChangePointerDevice.
BadMatch
       قد يحدث هذا الخطأ إذا تم إجراء طلب XGetFeedbackControl يحدد جهازًا لا يحتوي على ردود فعل، أو تم إجراء طلب XChangeFeedbackControl ببنية XFeedbackControl تحتوي على نوع رد فعل غير صالح. قد يحدث أيضًا إذا تم تحديد مجموعة غير صالحة من بتات القناع (DvKey ولكن لا يوجد DvAutoRepeatMode لردود فعل لوحة المفاتيح)، أو إذا تم تحديد KeySym غير صالح لرد فعل سلسلة.
BadValue
       تقع بعض القيم الرقمية خارج نطاق القيم التي يقبلها طلب XChangeFeedbackControl. ما لم يتم تحديد نطاق معين لمعامل، يتم قبول النطاق الكامل الذي تحدده نوع المعامل. يمكن لأي معامل تم تعريفه كمجموعة من البدائل إنشاء هذا الخطأ.
