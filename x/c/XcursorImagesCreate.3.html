الحزمة: libxcursor

التراخيص: HPND-sell-variant

XCURSOR(3) إصدار X 11 XCURSOR(3)

الاسم

Xcursor - مكتبة إدارة المؤشرات

الملخص

#include <X11/Xcursor/Xcursor.h>

الوصف

Xcursor هي مكتبة بسيطة مصممة للمساعدة في تحديد وتحميل المؤشرات. يمكن تحميل المؤشرات من ملفات أو من الذاكرة. توجد مكتبة للمؤشرات الشائعة التي تتوافق مع أسماء مؤشرات X القياسية. يمكن أن توجد المؤشرات بأحجام مختلفة والمكتبة تختار تلقائيًا أفضل حجم.

نظرة عامة على الوظائف

تم بناء Xcursor على عدة طبقات؛ في الطبقة السفلية يوجد الكود الذي يمكنه تحميل صور المؤشرات من الملفات. فوق ذلك توجد طبقة تحدد موقع ملفات المؤشرات بناءً على مسار المكتبة والموضوع. في الأعلى توجد طبقة تنشئ المؤشرات إما من صورة تم تحميلها من ملف أو أحد مؤشرات X القياسية. عند استخدام الصور التي تم تحميلها من ملفات، تفضل Xcursor استخدام طلب إنشاء المؤشر الخاص بامتداد Render إذا كان مدعومًا من خادم X. حيثما لم يكن مدعومًا، تقوم Xcursor بربط صورة المؤشر بمؤشر X قياسي وتستخدم طلب إنشاء المؤشر الأساسي.

ملفات المؤشرات

تحدد Xcursor تنسيقًا جديدًا للمؤشرات على القرص. يحتوي كل ملف على صورة مؤشر واحدة أو أكثر. يتم وضع علامة على كل صورة مؤشر بحجم اسمي بحيث يمكن اختيار أفضل حجم تلقائيًا. يمكن تحميل مؤشرات متعددة بنفس الحجم الاسمي معًا؛ من المتوقع أن تستخدم التطبيقاتها كتسلسل متحرك.

يتم تخزين ملفات المؤشرات كرأس يحتوي على جدول محتويات متبوعًا بتسلسل من الأجزاء. يشير جدول المحتويات إلى النوع، والنوع الفرعي، والموضع في الملف لكل جزء. رأس الملف يبدو كالتالي:


magic:	CARD32 "Xcur" (0x58, 0x63, 0x75, 0x72)
header:	CARD32 بايت في هذا الرأس
version:	CARD32 رقم إصدار الملف
ntoc:	CARD32 عدد إدخالات جدول المحتويات
toc:	LISTofTOC جدول المحتويات

كل إدخال في جدول المحتويات يبدو كالتالي:


type:	CARD32 نوع الإدخال
subtype:	CARD32 تسمية خاصة بالنوع - حجم الصور
position:	CARD32 موضع البايت المطلق لجدول المحتويات في الملف

كل جزء في الملف يحتوي على مجموعة من حقول الرأس المشتركة متبوعة بحقول إضافية خاصة بالنوع:


header:	CARD32 بايت في رأس الجزء (بما في ذلك الحقول الخاصة بالنوع)
type:	CARD32 يجب أن يتطابق مع النوع في جدول المحتويات لهذا الجزء
subtype:	CARD32 يجب أن يتطابق مع النوع الفرعي في جدول المحتويات لهذا الجزء
version:	CARD32 رقم إصدار لهذا النوع من الأجزاء

يوجد حاليًا نوعان من الأجزاء معرفان لملفات المؤشرات؛ التعليقات والصور. تبدو التعليقات كالتالي:


header:	20 رؤوس التعليقات بحجم 20 بايت
type:	0xfffe0001 نوع التعليق هو 0xfffe0001
subtype:	{ 1 (COPYRIGHT), 2 (LICENSE), 3 (OTHER) }
version:	1
length:	CARD32 طول البايت للسلسلة UTF-8
string:	LISTofCARD8 سلسلة UTF-8

تبدو الصور كالتالي:


header:	36 رؤوس الصور بحجم 36 بايت
type:	0xfffd0002 نوع الصورة هو 0xfffd0002
subtype:	CARD32 النوع الفرعي للصورة هو الحجم الاسمي
version:	1
width:	CARD32 يجب أن يكون أقل من أو يساوي 0x7fff
height:	CARD32 يجب أن يكون أقل من أو يساوي 0x7fff
xhot:	CARD32 يجب أن يكون أقل من أو يساوي العرض
yhot:	CARD32 يجب أن يكون أقل من أو يساوي الارتفاع
delay:	CARD32 التأخير بين إطارات الرسوم المتحركة بالمللي ثانية
pixels:	LISTofCARD32 بكسلات بتنسيق ARGB مضغوط

المواضيع

تتبع Xcursor (في الغالب) مواصفات freedesktop.org لمواضيع الأيقونات. المسار الافتراضي للبحث الذي تستخدمه هو

~/.local/share/icons, ~/.icons, /usr/share/icons, /usr/share/pixmaps

داخل كل من هذه الدلائل، تبحث عن دليل باستخدام اسم الموضوع:

•
داخل دليل الموضوع، تبحث عن ملفات المؤشرات في الدليل الفرعي "cursors".
تبحث Xcursor عن ملف محدد، والذي يجب أن يكون أحد أسماء الشكل للمؤشر، على سبيل المثال، كما هو مستخدم في XcursorLibraryLoadImage أو XcursorLibraryShape.
•
إذا لم تجد ملف مؤشر مطابق في الدليل الفرعي "cursors"، تبحث Xcursor بعد ذلك عن ملف "index.theme" في كل دليل موضوع للبحث عن المواضيع الموروثة. تلك هي سطور بهذا التنسيق:
Inherits = اسم-الموضوع

تستخدم Xcursor أول اسم-موضوع موروث، متجاهلة الآخرين الذين قد يكونون موجودين في ملف "index.theme" معين. إذا وجدت موضوعًا موروثًا، فإن Xcursor تبحث على طول المسار لاستخدامه أيضًا. تتجاهل Xcursor مفاتيح أخرى في ملف "index.theme"، بما في ذلك "Name" (أي، الاسم الذي قد يستخدمه التطبيق الرسومي كـ اسم العرض).

يمكن إدراج أكثر من اسم-موضوع واحد في سطر Inherits=. تنص مواصفات freedesktop.org على أن عناصر القائمة مفصولة بفواصل. تقبل Xcursor أيضًا الفاصلة المنقوطة، لكنها تترجم كليهما إلى نقطتين عند البحث في المسار. تتوقع Xcursor سطرًا واحدًا فقط Inherits=؛ مواصفات freedesktop.org غير واضحة ما إذا كانت المفاتيح المتعددة مسموحة.

إذا لم يتم تعيين موضوع، أو إذا لم يتم العثور على مؤشر للموضوع المحدد في أي مكان على طول المسار، فإن Xcursor تتحقق من موضوع "default".

عندما تجد Xcursor ملف مؤشر، فإنها تتوقف عن البحث. تستخدم دائمًا أول ملف مؤشر يتم العثور عليه أثناء البحث على طول المسار.

أنواع البيانات

XcursorImage
يحتوي على صورة مؤشر واحدة في الذاكرة. كل بكسل في المؤشر هو قيمة 32 بت تحتوي على ARGB مع A في البايت الأعلى. 


typedef struct _XcursorImage {
    XcursorDim 	 size; 	/* حجم اسمي للمطابقة */
    XcursorDim 	 width; 	/* عرض فعلي */
    XcursorDim 	 height; 	/* ارتفاع فعلي */
    XcursorDim 	 xhot; 	/* نقطة ساخنة x (يجب أن تكون داخل الصورة) */
    XcursorDim 	 yhot; 	/* نقطة ساخنة y (يجب أن تكون داخل الصورة) */
    XcursorPixel 	*pixels; 	/* مؤشر إلى البكسلات */
} XcursorImage;
XcursorImages
يحتوي على هياكل XcursorImage متعددة. يتم تحريرها جميعًا عند تحرير XcursorImages في XcursorImagesDestroy. 


typedef struct _XcursorImages {
    int 	 nimage; 	/* عدد الصور */
    XcursorImage 	**images; 	/* مصفوفة من مؤشرات XcursorImage */
} XcursorImages;
XcursorCursors
يحتوي على كائنات مؤشر متعددة. يتم تحريرها جميعًا عند تحرير XcursorCursors. هذه الكائنات لها عداد مرجعي بحيث يمكن لهياكل XcursorAnimate متعددة استخدام نفس XcursorCursors. 


typedef struct _XcursorCursors {
    Display 	*dpy; 	/* الشاشة التي تحتوي على المؤشرات */
    int 	 ref; 	/* عداد المراجع */
    int 	 ncursor; 	/* عدد المؤشرات */
    Cursor 	*cursors; 	/* مصفوفة من المؤشرات */
} XcursorCursors;
XcursorAnimate
يشير إلى مجموعة من المؤشرات وتسلسل داخل تلك المجموعة. قد تشير هياكل XcursorAnimate متعددة إلى نفس XcursorCursors؛ يحتفظ كل منها بمرجع يتم إزالته عند تحرير XcursorAnimate. 


typedef struct _XcursorAnimate {
    XcursorCursors 	*cursors; 	/* قائمة المؤشرات المراد استخدامها */
    int 	 sequence; 	/* أي مؤشر هو التالي */
} XcursorAnimate;
XcursorFile
توفر Xcursor واجهة برمجة تطبيقات مجردة للوصول إلى بيانات الملف. توفر Xcursor تنفيذًا قياسيًا لواجهة برمجة التطبيقات المجردة هذه؛ التطبيقات حرة في إنشاء تطبيقات إضافية. هذه الوظائف توازي وظائف stdio.

هذا الموقع غير مرتبط بـ Arch Linux ولا تابع له بأي شكل من الأشكال. المحتوى هو ترجمة فقط لصفحات دليل Arch المنشورة على man.archlinux.org.

ctions in return value and expected argument values; the read and write functions flip the arguments around to match the POSIX versions. 


typedef struct _XcursorFile {
    void *closure;
    int (*read)  (XcursorFile *file, unsigned char *buf, int len);
    int (*write) (XcursorFile *file, unsigned char *buf, int len);
    int (*seek)  (XcursorFile *file, long offset, int whence);
};

تقوم هذه الوظائف بالبحث في مسار المكتبة، وتحميل أول ملف يتم العثور عليه بالحجم المطلوب (size)، باستخدام وظيفة خاصة (XcursorScanTheme) للعثور على السمة المناسبة:

  • إذا كان theme ليس NULL، يتم استخدام ذلك.
  • إذا كان theme NULL، أو إذا لم يكن هناك تطابق للسمة المطلوبة، يتم استخدام "default" لاسم السمة.
  • إذا فشل كلا البحثين، تعيد هذه الوظائف NULL.
تختلف الوظيفتان عن بعضهما البعض بأكثر من عدد الصور التي تم تحميلها:
  • XcursorLibraryLoadImage تستدعي XcursorFileLoadImage ولكن
  • XcursorLibraryLoadImages تستدعي XcursorFileLoadImages، وتقوم (عند النجاح) باستدعاء XcursorImagesSetName لربط name بالنتيجة.

ENVIRONMENT


Environment variables can be used to override resource settings,
    which in turn override compiled-in default values.


Some of the environment variables recognized by Xcursor are
    booleans, specified as follows:






true for
    “t”, “1”, “y” or
    “on”




false for
    “f”, “0”, “n” or
    “off”








Xcursor ignores other values for these booleans.




HOME


Xcursor interprets “~” in the search list as the home
    directory, using this variable rather than the password database.


XCURSOR_ANIM


If the display supports the Render CreateCursor request, and the Render
    feature is enabled, disable animated cursors if the environment
    variable is false.








If the environment variable is not given, Xcursor uses the resource
    Xcursor.anim.






XCURSOR_CORE


If the display supports the Render CreateCursor request disable the Render
    feature if the environment variable is false.








If the environment variable is not given, Xcursor uses the resource
    Xcursor.core.






XCURSOR_DISCOVER


If the variable is set, Xcursor turns on a logging feature. It displays
    the hash value and the image so that users can see which cursor name is
    associated with each image.








There is no corresponding resource setting.






XCURSOR_DITHER


This variable sets the desired dither.








If the environment variable is not given, Xcursor uses the resource
    Xcursor.dither.




If neither environment variable or resource is found, Xcursor uses
    “threshold”




These are the recognized values:








diffuse




median




ordered




threshold










XCURSOR_PATH


This variable sets the list of paths in which to search for cursors,
    rather than the compiled-in default list.








Directories in this path are separated by colons (:).






XCURSOR_RESIZED


Enables automatic resizing of cursors to improve their displayed size if
    the environment variable is true.








If the environment variable is not given, Xcursor tries the
    Xcursor.resized resource.






XCURSOR_SIZE


This variable sets the desired cursor size, in pixels.








If the environment variable is not given, Xcursor tries the
    Xcursor.size resource.




If no size is given, whether by environment variable or resource setting,
    Xcursor next tries the Xft.dpi resource setting to guess the size
    of a 16-point cursor.




Finally, if Xft.dpi is not set, Xcursor uses the display height,
    dividing by 48 (assuming that the height is 768).






XCURSOR_THEME


This variable selects the desired theme.








If the environment variable is not given, Xcursor tries the
    Xcursor.theme resource.




If neither environment variable or resource is found, Xcursor uses the
    default theme.






XCURSOR_THEME_CORE


If the display supports the Render CreateCursor request, enable the
    Render feature if the environment variable is true.








If the environment variable is not given, Xcursor uses the resource
    Xcursor.theme_core.

NAME


XcursorGetTheme, XcursorGetThemeCore – get the cursor theme name






SYNOPSIS


#include <Xcursor/Xcursor.h>

char *
XcursorGetTheme(
    Display *dpy
);

char *
XcursorGetThemeCore(
    Display *dpy
);







DESCRIPTION


The Xcursor library provides functions for loading and managing
    cursor themes.


The XcursorGetTheme function returns the
    name of the cursor theme that is currently active for the display dpy.


The XcursorGetThemeCore function returns
    the name of the core cursor theme that is currently active for the display
    dpy.


Note that the strings returned by these functions are allocated by
    the Xcursor library and should be freed by the caller using
    XFree.






FUNCTIONAL OVERVIEW


The Xcursor library supports cursor themes, which are collections
    of cursor images that can be used to customize the appearance of the mouse
    pointer. Themes can be specified by applications or by the user through
    system-wide configuration.


The library provides functions to get and set the current cursor
    theme. It also supports core cursors, which are a set of predefined cursor
    shapes that are always available.






DATATYPES



  
Display

  
Represents a connection to an X server.

  
char *

  
A pointer to a null-terminated string, representing the name of a cursor
      theme.








FUNCTIONS



  
XcursorGetTheme(3)

  
Returns the name of the currently active cursor theme.

  
XcursorGetThemeCore(3)

  
Returns the name of the currently active core cursor theme.

  
XcursorSetTheme(3)

  
Sets the currently active cursor theme.

  
XcursorSetThemeCore(3)

  
Sets the currently active core cursor theme.

  
XcursorDefaultTheme(3)

  
Returns the name of the default cursor theme.

  
XcursorDefaultThemeCore(3)

  
Returns the name of the default core cursor theme.








ENVIRONMENT



  
XCURSOR_THEME

  
If this environment variable is set to a non-empty string, its value is
      used as the cursor theme name. If the value is empty, the default theme
      is used.

  
XCURSOR_THEME_CORE

  
If this environment variable is set to a non-empty string, its value is
      used as the core cursor theme name. If the value is empty, the default core
      theme is used. If this variable is not set, the value of
      Xcursor.theme_core resource is used.





  
Xcursor.theme (resource)

  
If the environment variable XCURSOR_THEME is not set, Xcursor
      tries this resource.

  
Xcursor.theme_core (resource)

  
If the environment variable XCURSOR_THEME_CORE is not set,
      Xcursor tries this resource.





  
XcursorSetThemeCore

  
Enables themes for core cursors if the environment variable is
      true.





  

  
If the environment variable is not given, Xcursor tries the
      Xcursor.theme_core resource.

  

  
An application can enable or disable themes using
    XcursorSetThemeCore.








CAVEATS


Xcursor will probably change radically in the future; weak
    attempts will be made to retain some level of source-file compatibility.

الاسم


XcursorGetTheme, XcursorGetThemeCore – الحصول على اسم سمة المؤشر






ملخص


#include <Xcursor/Xcursor.h>

char *
XcursorGetTheme(
    Display *dpy
);

char *
XcursorGetThemeCore(
    Display *dpy
);







الوصف


توفر مكتبة Xcursor وظائف لتحميل وإدارة سمات المؤشر.


تعرض الدالة XcursorGetTheme اسم سمة المؤشر النشطة حاليًا للعرض dpy.


تعرض الدالة XcursorGetThemeCore اسم سمة المؤشر الأساسية النشطة حاليًا للعرض dpy.


لاحظ أن السلاسل النصية التي تعرضها هاتان الدالتان تم تخصيصها بواسطة مكتبة Xcursor ويجب على المتصل تحريرها باستخدام XFree.






نظرة عامة وظيفية


تدعم مكتبة Xcursor سمات المؤشر، وهي مجموعات من صور المؤشر التي يمكن استخدامها لتخصيص مظهر مؤشر الفأرة. يمكن تحديد السمات بواسطة التطبيقات أو بواسطة المستخدم من خلال التكوين على مستوى النظام.


توفر المكتبة دوال للحصول على سمة المؤشر الحالية وتعيينها. كما تدعم المؤشرات الأساسية، وهي مجموعة من أشكال المؤشرات المحددة مسبقًا والتي تكون متاحة دائمًا.






أنواع البيانات



  
Display

  
يمثل اتصالاً بخادم X.

  
char *

  
مؤشر إلى سلسلة نصية منتهية بصفر، تمثل اسم سمة المؤشر.








الوظائف



  
XcursorGetTheme(3)

  
يعرض اسم سمة المؤشر النشطة حاليًا.

  
XcursorGetThemeCore(3)

  
يعرض اسم سمة المؤشر الأساسية النشطة حاليًا.

  
XcursorSetTheme(3)

  
يعين سمة المؤشر النشطة حاليًا.

  
XcursorSetThemeCore(3)

  
يعين سمة المؤشر الأساسية النشطة حاليًا.

  
XcursorDefaultTheme(3)

  
يعرض اسم سمة المؤشر الافتراضية.

  
XcursorDefaultThemeCore(3)

  
يعرض اسم سمة المؤشر الأساسية الافتراضية.








البيئة



  
XCURSOR_THEME

  
إذا تم تعيين متغير البيئة هذا إلى سلسلة نصية غير فارغة، يتم استخدام قيمته كاسم لسمة المؤشر. إذا كانت القيمة فارغة، يتم استخدام السمة الافتراضية.

  
XCURSOR_THEME_CORE

  
إذا تم تعيين متغير البيئة هذا إلى سلسلة نصية غير فارغة، يتم استخدام قيمته كاسم لسمة المؤشر الأساسية. إذا كانت القيمة فارغة، يتم استخدام السمة الأساسية الافتراضية. إذا لم يتم تعيين هذا المتغير، يتم استخدام قيمة مورد
      Xcursor.theme_core.





  
Xcursor.theme (resource)

  
إذا لم يتم تعيين متغير البيئة XCURSOR_THEME، تحاول Xcursor هذا المورد.

  
Xcursor.theme_core (resource)

  
إذا لم يتم تعيين متغير البيئة XCURSOR_THEME_CORE، تحاول Xcursor هذا المورد.





  
XcursorSetThemeCore

  
يمكّن السمات للمؤشرات الأساسية إذا كان متغير البيئة
      true.





  

  
إذا لم يتم توفير متغير البيئة، تحاول Xcursor المورد
      Xcursor.theme_core.

  

  
يمكن للتطبيق تمكين أو تعطيل السمات باستخدام
    XcursorSetThemeCore.
