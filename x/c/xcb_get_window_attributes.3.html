الحزمة: libxcb

التراخيص: X11

xcb_get_window_attributes(3) طلبات XCB xcb_get_window_attributes(3)

الاسم

xcb_get_window_attributes - الحصول على سمات النافذة

الملخص

#include <xcb/xproto.h>

دالة الطلب

xcb_get_window_attributes_cookie_t xcb_get_window_attributes(xcb_connection_t *conn, xcb_window_t window);

هيكل بيانات الرد

typedef struct xcb_get_window_attributes_reply_t {
    uint8_t        response_type;
    uint8_t        backing_store;
    uint16_t       sequence;
    uint32_t       length;
    xcb_visualid_t visual;
    uint16_t       _class;
    uint8_t        bit_gravity;
    uint8_t        win_gravity;
    uint32_t       backing_planes;
    uint32_t       backing_pixel;
    uint8_t        save_under;
    uint8_t        map_is_installed;
    uint8_t        map_state;
    uint8_t        override_redirect;
    xcb_colormap_t colormap;
    uint32_t       all_event_masks;
    uint32_t       your_event_mask;
    uint16_t       do_not_propagate_mask;
    uint8_t        pad0[2];
} xcb_get_window_attributes_reply_t;

دالة الرد

xcb_get_window_attributes_reply_t *xcb_get_window_attributes_reply(xcb_connection_t *conn, xcb_get_window_attributes_cookie_t cookie, xcb_generic_error_t **e);

وسائط الطلب

conn
اتصال XCB بـ X11.
window
النافذة التي سيتم الحصول على سماتها منها.

حقول الرد

رقم التسلسل لآخر طلب تم معالجته بواسطة خادم X11.
length
