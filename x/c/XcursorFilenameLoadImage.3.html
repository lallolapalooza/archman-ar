الحزمة: libxcursor

التراخيص: HPND-sell-variant

Xcursor(3) إصدار X 11 Xcursor(3)

الاسم

Xcursor - مكتبة إدارة المؤشرات

ملخص

#include <X11/Xcursor/Xcursor.h>

الوصف

Xcursor هي مكتبة بسيطة مصممة للمساعدة في تحديد وتحميل المؤشرات. يمكن تحميل المؤشرات من ملفات أو ذاكرة. توجد مكتبة للمؤشرات الشائعة التي ترتبط بأسماء مؤشرات X القياسية. يمكن أن توجد المؤشرات بأحجام مختلفة وتقوم المكتبة تلقائيًا باختيار أفضل حجم.

نظرة عامة وظيفية

تم بناء Xcursor في طبقات قليلة؛ في الطبقة السفلية يوجد رمز يمكنه تحميل صور المؤشرات من الملفات. فوق ذلك توجد طبقة تحدد مواقع ملفات المؤشرات بناءً على مسار المكتبة والموضوع. في الأعلى توجد طبقة تبني المؤشرات إما من صورة تم تحميلها من ملف أو أحد مؤشرات X القياسية. عند استخدام الصور المحملة من ملفات، تفضل Xcursor استخدام طلب إنشاء المؤشر لامتداد Render إذا كان مدعومًا بواسطة خادم X. حيثما لم يتم دعمه، ترسم Xcursor صورة المؤشر إلى مؤشر X قياسي وتستخدم طلب إنشاء المؤشر الأساسي.

ملفات المؤشرات

تحدد Xcursor تنسيقًا جديدًا للمؤشرات على القرص. كل ملف يحتوي على صورة مؤشر واحدة أو أكثر. كل صورة مؤشر موسومة بحجم اسمي بحيث يمكن اختيار أفضل حجم تلقائيًا. يمكن تحميل مؤشرات متعددة بنفس الحجم الاسمي معًا؛ من المتوقع أن تستخدمها التطبيقات كتسلسل متحرك.

تُخزن ملفات المؤشرات كرأس يحتوي على جدول محتويات متبوعًا بسلسلة من المقاطع. يشير رأس الملف إلى النوع، النوع الفرعي، والموضع في الملف لكل مقطع. رأس الملف يبدو كالتالي:


magic:	CARD32 "Xcur" (0x58, 0x63, 0x75, 0x72)
header:	CARD32 بايت في هذا الرأس
version:	CARD32 رقم إصدار الملف
ntoc:	CARD32 عدد إدخالات جدول المحتويات
toc:	LISTofTOC جدول المحتويات

كل إدخال في جدول المحتويات يبدو كالتالي:


type:	CARD32 نوع الإدخال
subtype:	CARD32 تسمية خاصة بالنوع - حجم للصور
position:	CARD32 الموضع المطلق بالبايت لجدول المحتويات في الملف

كل مقطع في الملف له مجموعة من حقول الرأس المشتركة متبوعة بحقول إضافية خاصة بالنوع:


header:	CARD32 بايت في رأس المقطع (بما في ذلك الحقول الخاصة بالنوع)
type:	CARD32 يجب أن يتطابق مع النوع في جدول المحتويات لهذا المقطع
subtype:	CARD32 يجب أن يتطابق مع النوع الفرعي في جدول المحتويات لهذا المقطع
version:	CARD32 رقم الإصدار لهذا النوع من المقاطع

يوجد حاليًا نوعان من المقاطع معرفان لملفات المؤشرات؛ التعليقات والصور. التعليقات تبدو كالتالي:


header:	20 رؤوس التعليقات هي 20 بايت
type:	0xfffe0001 نوع التعليق هو 0xfffe0001
subtype:	{ 1 (حقوق النشر)، 2 (الترخيص)، 3 (أخرى) }
version:	1
length:	CARD32 طول سلسلة UTF-8 بالبايت
string:	LISTofCARD8 سلسلة UTF-8

الصور تبدو كالتالي:


header:	36 رؤوس الصور هي 36 بايت
type:	0xfffd0002 نوع الصورة هو 0xfffd0002
subtype:	CARD32 النوع الفرعي للصورة هو الحجم الاسمي
version:	1
width:	CARD32 يجب أن يكون أقل من أو يساوي 0x7fff
height:	CARD32 يجب أن يكون أقل من أو يساوي 0x7fff
xhot:	CARD32 يجب أن يكون أقل من أو يساوي العرض
yhot:	CARD32 يجب أن يكون أقل من أو يساوي الارتفاع
delay:	CARD32 التأخير بين إطارات الرسوم المتحركة بالمللي ثانية
pixels:	LISTofCARD32 وحدات بكسل بتنسيق ARGB معبأة

المواضيع

تتبع Xcursor (في الغالب) مواصفات freedesktop.org لترميز الأيقونات. مسار البحث الافتراضي الذي تستخدمه هو

~/.local/share/icons, ~/.icons, /usr/share/icons, /usr/share/pixmaps

داخل كل من هذه الدلائل، تبحث عن دليل باستخدام اسم الموضوع:

•
داخل دليل الموضوع، تبحث عن ملفات المؤشرات في الدليل الفرعي "cursors".
تبحث Xcursor عن ملف محدد، يجب أن يكون أحد أسماء shape للمؤشر، على سبيل المثال، كما هو مستخدم في XcursorLibraryLoadImage أو XcursorLibraryShape.
•
إذا لم تجد ملف مؤشر مطابق في الدليل الفرعي "cursors"، تبحث Xcursor بعد ذلك عن ملف "index.theme" في كل دليل موضوع للبحث عن المواضيع الموروثة. هذه هي أسطر بهذا التنسيق:
Inherits = اسم-الموضوع

تستخدم Xcursor أول اسم-موضوع موروث، متجاهلة الآخرين الذين قد يكونون موجودين في ملف "index.theme" معين. إذا وجدت موضوعًا موروثًا، فتبحث Xcursor على طول المسار لاستخدامه أيضًا. تتجاهل Xcursor مفاتيح أخرى في ملف "index.theme"، بما في ذلك "Name" (أي، الاسم الذي قد يستخدمه التطبيق الرسومي كـ اسم العرض).

يمكن إدراج أكثر من اسم-موضوع واحد في سطر Inherits=. تنص مواصفات freedesktop.org على أن عناصر القائمة مفصولة بفواصل. تقبل Xcursor أيضًا الفاصلة المنقوطة، ولكنها تترجم كليهما إلى نقطتين عند البحث في المسار. تتوقع Xcursor سطر Inherits= واحد فقط؛ مواصفات freedesktop.org غير واضحة بشأن ما إذا كان يمكن السماح بمفاتيح متعددة.

إذا لم يتم تعيين موضوع، أو إذا لم يتم العثور على مؤشر للموضوع المحدد في أي مكان على طول المسار، تتحقق Xcursor من الموضوع "default".

عندما تجد Xcursor ملف مؤشر، فإنها تتوقف عن البحث. تستخدم دائمًا أول ملف مؤشر يتم العثور عليه أثناء البحث على طول المسار.

أنواع البيانات

XcursorImage
يحتوي على صورة مؤشر واحدة في الذاكرة. كل بكسل في المؤشر هو قيمة 32 بت تحتوي على ARGB مع A في البايت الأعلى. 


typedef struct _XcursorImage {
    XcursorDim 	 size; 	/* حجم اسمي للمطابقة */
    XcursorDim 	 width; 	/* عرض فعلي */
    XcursorDim 	 height; 	/* ارتفاع فعلي */
    XcursorDim 	 xhot; 	/* نقطة ساخنة x (يجب أن تكون داخل الصورة) */
    XcursorDim 	 yhot; 	/* نقطة ساخنة y (يجب أن تكون داخل الصورة) */
    XcursorPixel 	*pixels; 	/* مؤشر إلى وحدات البكسل */
} XcursorImage;
XcursorImages
يحتوي على هياكل XcursorImage متعددة. يتم تحريرها كلها عند تحرير XcursorImages في XcursorImagesDestroy. 


typedef struct _XcursorImages {
    int 	 nimage; 	/* عدد الصور */
    XcursorImage 	**images; 	/* مصفوفة مؤشرات XcursorImage */
} XcursorImages;
XcursorCursors
يحتوي على كائنات مؤشر متعددة. يتم تحريرها كلها عند تحرير XcursorCursors. يتم عد هذه المراجع بحيث يمكن لهياكل XcursorAnimate متعددة استخدام نفس XcursorCursors. 


typedef struct _XcursorCursors {
    Display 	*dpy; 	/* الشاشة التي تحتوي على المؤشرات */
    int 	 ref; 	/* عدد المراجع */
    int 	 ncursor; 	/* عدد المؤشرات */
    Cursor 	*cursors; 	/* مصفوفة المؤشرات */
} XcursorCursors;
XcursorAnimate
يشير إلى مجموعة من المؤشرات وتسلسل داخل تلك المجموعة. قد تشير هياكل XcursorAnimate متعددة إلى نفس XcursorCursors؛ كل منها يحمل مرجعًا يتم إزالته عند تحرير XcursorAnimate. 


typedef struct _XcursorAnimate {
    XcursorCursors 	*cursors; 	/* قائمة المؤشرات المراد استخدامها */
    int 	 sequence; 	/* المؤشر التالي */
} XcursorAnimate;
XcursorFile
توفر Xcursor واجهة برمجة تطبيقات مجردة للوصول إلى بيانات الملف. توفر Xcursor تطبيق stdio لهذه الواجهة المجردة؛ التطبيقات حرة في إنشاء تطبيقات إضافية. تتوازى هذه الوظائف مع stdio

هذا الموقع غير مرتبط بـ Arch Linux ولا تابع له بأي شكل من الأشكال. المحتوى هو ترجمة فقط لصفحات دليل Arch المنشورة على man.archlinux.org.

o في قيمة الإرجاع والقيم المتوقعة للمعاملات؛ تقوم دوال القراءة والكتابة بتبديل الوسائط لتتوافق مع إصدارات POSIX. 


typedef struct _XcursorFile {
    void *closure;
    int (*read)  (XcursorFile *file, unsigned char *buf, int len);
    int (*write) (XcursorFile *file, unsigned char *buf, int len);
    int (*seek)  (XcursorFile *file, long offset, int whence);
};

	const char 	*name,

تقوم هذه بالبحث في مسار المكتبة، وتحميل أول ملف تم العثور عليه بالحجم المطلوب size، باستخدام دالة خاصة (XcursorScanTheme) للعثور على السمة المناسبة:

  • إذا كان theme ليس NULL، استخدمه.
  • إذا كان theme NULL، أو إذا لم يكن هناك تطابق للسمة المطلوبة، استخدم "default" لاسم السمة.
  • إذا فشل كلا البحثين، تعيد هذه الدوال NULL.
تختلف الدالتان بما يتجاوز عدد الصور المحملة:
  • XcursorLibraryLoadImage يستدعي XcursorFileLoadImage ولكن
  • XcursorLibraryLoadImages يستدعي XcursorFileLoadImages، (عند النجاح) يستدعي XcursorImagesSetName لربط name بالنتيجة.

NAME


XcursorFilenameLoadImage, XcursorFileLoadImage, XcursorLibraryLoadImage, XcursorLibraryLoadImages - load cursor images from files or the cursor library






LIBRARY


libXcursor






SYNOPSIS


#include <Xcursor.h>


XcursorImage *
    XcursorFilenameLoadImage(const char *filename, unsigned int size);


XcursorImage *
    XcursorFileLoadImage(FILE *file, unsigned int size);


XcursorImage *
    XcursorLibraryLoadImage(const char *name, unsigned int size);


XcursorImages *
    XcursorLibraryLoadImages(const char *name, unsigned int size);






DESCRIPTION


The Xcursor library provides functions to load cursor images from
    various sources. These functions are useful for applications that need to
    display custom cursors.


XcursorFilenameLoadImage loads a cursor image from the specified
    filename. The image will be scaled to the given size.


XcursorFileLoadImage loads a cursor image from the given FILE
    pointer. The image will be scaled to the given size.


XcursorLibraryLoadImage loads a single cursor image with the
    specified name from the Xcursor cursor library. The image will be
    scaled to the given size.


XcursorLibraryLoadImages loads all cursor images with the
    specified name from the Xcursor cursor library. The images will be
    scaled to the given size. The result is an XcursorImages
    structure which contains an array of XcursorImage pointers.


If the requested size is zero, the image is loaded with its
    native size. Otherwise, the image is scaled to the requested size.


The loaded XcursorImage or XcursorImages structures
    must be freed using XcursorImageDestroy or XcursorImagesDestroy
    respectively when no longer needed.






RETURN VALUE


All functions return a pointer to the loaded XcursorImage or
    XcursorImages structure on success, and NULL on failure.

        XcursorFilenameLoadImage(3) — libXcursor — X Version 11
        
        
    
    
        
        

            

                


الاسم


XcursorFilenameLoadImage, XcursorFileLoadImage, XcursorLibraryLoadImage, XcursorLibraryLoadImages - تحميل صور المؤشر من الملفات أو مكتبة المؤشرات






المكتبة


libXcursor






الملخص


#include <Xcursor.h>


XcursorImage *
    XcursorFilenameLoadImage(const char *filename, unsigned int size);


XcursorImage *
    XcursorFileLoadImage(FILE *file, unsigned int size);


XcursorImage *
    XcursorLibraryLoadImage(const char *name, unsigned int size);


XcursorImages *
    XcursorLibraryLoadImages(const char *name, unsigned int size);






الوصف


توفر مكتبة Xcursor دوال لتحميل صور المؤشرات من مصادر مختلفة. هذه الدوال مفيدة للتطبيقات التي تحتاج إلى عرض مؤشرات مخصصة.


تقوم XcursorFilenameLoadImage بتحميل صورة مؤشر من الملف المحدد filename. سيتم تغيير حجم الصورة لتناسب الحجم size المحدد.


تقوم XcursorFileLoadImage بتحميل صورة مؤشر من مؤشر الملف FILE المحدد. سيتم تغيير حجم الصورة لتناسب الحجم size المحدد.


تقوم XcursorLibraryLoadImage بتحميل صورة مؤشر واحدة بالاسم المحدد name من مكتبة مؤشرات Xcursor. سيتم تغيير حجم الصورة لتناسب الحجم size المحدد.


تقوم XcursorLibraryLoadImages بتحميل جميع صور المؤشرات بالاسم المحدد name من مكتبة مؤشرات Xcursor. سيتم تغيير حجم الصور لتناسب الحجم size المحدد. تكون النتيجة هيكل XcursorImages الذي يحتوي على مصفوفة من مؤشرات XcursorImage.


إذا كان الحجم المطلوب صفرًا، يتم تحميل الصورة بحجمها الأصلي. بخلاف ذلك، يتم تغيير حجم الصورة إلى الحجم المطلوب.


يجب تحرير هياكل XcursorImage أو XcursorImages التي تم تحميلها باستخدام XcursorImageDestroy أو XcursorImagesDestroy على التوالي عندما لا تكون هناك حاجة إليها.






قيمة الإرجاع


ترجع جميع الدوال مؤشرًا إلى هيكل XcursorImage أو XcursorImages الذي تم تحميله عند النجاح، و NULL عند الفشل.






انظر أيضًا


XcursorImageCreate(3), XcursorImageDestroy(3), XcursorImagesDestroy(3),
    XcursorCreateRenderCursor(3)






البيئة



  
XCURSOR_THEME

  
يحدد سمة المؤشر المراد استخدامها.

  
XCURSOR_SIZE

  
يحدد حجم المؤشر الافتراضي.






R_THEME_CORE


يُمكّن السمات لمؤشرات النواة إذا كان متغير البيئة
    true.




  

  
إذا لم يتم تقديم متغير البيئة، تحاول Xcursor استخدام
      المورد Xcursor.theme_core.

  

  
يمكن للتطبيق تمكين أو تعطيل السمات باستخدام
    XcursorSetThemeCore.








محاذير


من المحتمل أن تتغير Xcursor بشكل جذري في المستقبل؛ سيتم بذل محاولات ضعيفة للحفاظ على مستوى معين من التوافق مع ملفات المصدر.
