الاسم

Xcursor - مكتبة إدارة المؤشرات

الملخص

#include <X11/Xcursor/Xcursor.h>

الوصف

Xcursor هي مكتبة بسيطة مصممة للمساعدة في تحديد وتحميل المؤشرات. يمكن تحميل المؤشرات من ملفات أو من الذاكرة. توجد مكتبة للمؤشرات الشائعة التي تتوافق مع أسماء مؤشرات X القياسية. يمكن أن توجد المؤشرات بأحجام مختلفة وتقوم المكتبة تلقائيًا باختيار أفضل حجم.

نظرة عامة وظيفية

تم بناء Xcursor في طبقتين؛ في الطبقة السفلية توجد التعليمات البرمجية التي يمكنها تحميل صور المؤشرات من الملفات. فوق ذلك توجد طبقة تحدد مواقع ملفات المؤشرات بناءً على مسار المكتبة والموضوع. في القمة توجد طبقة تبني المؤشرات إما من صورة تم تحميلها من ملف أو أحد مؤشرات X القياسية. عند استخدام الصور المحملة من الملفات، تفضل Xcursor استخدام طلب CreateCursor لامتداد Render إذا كان مدعومًا من خادم X. حيثما لم يكن مدعومًا، تقوم Xcursor بتعيين صورة المؤشر إلى مؤشر X قياسي وتستخدم طلب CreateCursor الأساسي.

ملفات المؤشرات

تحدد Xcursor تنسيقًا جديدًا للمؤشرات على القرص. يحتوي كل ملف على صورة مؤشر واحدة أو أكثر. يتم تمييز كل صورة مؤشر بحجم اسمي بحيث يمكن اختيار أفضل حجم تلقائيًا. يمكن تحميل مؤشرات متعددة بنفس الحجم الاسمي معًا؛ ومن المتوقع أن تستخدمها التطبيقات كسلسلة رسوم متحركة.

يتم تخزين ملفات المؤشرات كتذييل يحتوي على جدول محتويات متبوعًا بسلسلة من المقطوعات. يشير جدول المحتويات إلى النوع والنوع الفرعي والموقع في الملف لكل مقطوعة. يبدو تذييل الملف كما يلي:


magic:	CARD32 "Xcur" (0x58, 0x63, 0x75, 0x72)
header:	CARD32 بايت في هذا التذييل
version:	CARD32 رقم إصدار الملف
ntoc:	CARD32 عدد إدخالات جدول المحتويات
toc:	LISTofTOC جدول المحتويات

كل إدخال في جدول المحتويات يبدو كما يلي:


type:	CARD32 نوع الإدخال
subtype:	CARD32 تسمية خاصة بالنوع - حجم للصور
position:	CARD32 موقع البايت المطلق للجدول في الملف

كل مقطوعة في الملف لديها مجموعة من حقول التذييل الشائعة متبوعة بحقول إضافية خاصة بالنوع:


header:	CARD32 بايت في تذييل المقطوعة (بما في ذلك الحقول الخاصة بالنوع)
type:	CARD32 يجب أن يتطابق مع النوع في TOC لهذه المقطوعة
subtype:	CARD32 يجب أن يتطابق مع النوع الفرعي في TOC لهذه المقطوعة
version:	CARD32 رقم إصدار لنوع هذه المقطوعة

يوجد حاليًا نوعان من المقطوعات محددتان لملفات المؤشرات؛ التعليقات والصور. تبدو التعليقات كما يلي:


header:	20 تذييلات التعليقات هي 20 بايت
type:	0xfffe0001 نوع التعليق هو 0xfffe0001
subtype:	{ 1 (حقوق النشر), 2 (الترخيص), 3 (أخرى) }
version:	1
length:	CARD32 طول البايت للسلسلة UTF-8
string:	LISTofCARD8 سلسلة UTF-8

تبدو الصور كما يلي:


header:	36 تذييلات الصور هي 36 بايت
type:	0xfffd0002 نوع الصورة هو 0xfffd0002
subtype:	CARD32 نوع الصورة الفرعي هو الحجم الاسمي
version:	1
width:	CARD32 يجب أن يكون أقل من أو يساوي 0x7fff
height:	CARD32 يجب أن يكون أقل من أو يساوي 0x7fff
xhot:	CARD32 يجب أن يكون أقل من أو يساوي العرض
yhot:	CARD32 يجب أن يكون أقل من أو يساوي الارتفاع
delay:	CARD32 التأخير بين إطارات الرسوم المتحركة بالمللي ثانية
pixels:	LISTofCARD32 بكسلات بتنسيق ARGB المعبأ

المواضيع

تتبع Xcursor (في الغالب) مواصفات freedesktop.org لتسمية الأيقونات. مسار البحث الافتراضي الذي تستخدمه هو

~/.local/share/icons, ~/.icons, /usr/share/icons, /usr/share/pixmaps

داخل كل من هذه الدلائل، تبحث عن دليل باستخدام اسم الموضوع:

•
داخل دليل الموضوع، تبحث عن ملفات المؤشرات في الدليل الفرعي "cursors".
تبحث Xcursor عن ملف محدد، والذي يجب أن يكون أحد أسماء الشكل للمؤشر، على سبيل المثال، كما هو مستخدم في XcursorLibraryLoadImage أو XcursorLibraryShape.
•
إذا لم تجد ملف مؤشر مطابق في الدليل الفرعي "cursors"، تبحث Xcursor بعد ذلك عن ملف "index.theme" في كل دليل موضوع للبحث عن مواضيع موروثة. هذه هي الأسطر بهذا التنسيق:
Inherits = اسم-الموضوع

تستخدم Xcursor أول اسم-موضوع موروث، متجاهلة الآخرين الذين قد يكونون موجودين في ملف "index.theme" معين. إذا وجدت موضوعًا موروثًا، فإن Xcursor تبحث على طول المسار لاستخدامه أيضًا. تتجاهل Xcursor مفاتيح أخرى في ملف "index.theme"، بما في ذلك "Name" (أي، الاسم الذي قد يستخدمه التطبيق الرسومي كـ اسم عرض).

يمكن سرد أكثر من اسم-موضوع واحد في سطر Inherits=. تنص مواصفات freedesktop.org على أن عناصر القائمة مفصولة بفواصل. تقبل Xcursor أيضًا فاصلة منقوطة، ولكنها تترجم كلاهما إلى نقطتين عند البحث في المسار. تتوقع Xcursor سطر Inherits= واحد فقط؛ مواصفات freedesktop.org غير واضحة بشأن ما إذا كانت مفاتيح متعددة مسموح بها.

إذا لم يتم تعيين موضوع، أو إذا لم يتم العثور على مؤشر للموضوع المحدد في أي مكان على طول المسار، فإن Xcursor تتحقق من موضوع "default".

عندما تجد Xcursor ملف مؤشر، فإنها تتوقف عن البحث. إنها تستخدم دائمًا أول ملف مؤشر يتم العثور عليه أثناء البحث على طول المسار.

أنواع البيانات

XcursorImage
يحتوي على صورة مؤشر واحدة في الذاكرة. كل بكسل في المؤشر هو قيمة 32 بت تحتوي على ARGB مع A في البايت العالي. 


typedef struct _XcursorImage {
    XcursorDim 	 size; 	/* حجم اسمي للمطابقة */
    XcursorDim 	 width; 	/* العرض الفعلي */
    XcursorDim 	 height; 	/* الارتفاع الفعلي */
    XcursorDim 	 xhot; 	/* نقطة الاتصال x (يجب أن تكون داخل الصورة) */
    XcursorDim 	 yhot; 	/* نقطة الاتصال y (يجب أن تكون داخل الصورة) */
    XcursorPixel 	*pixels; 	/* مؤشر إلى البكسلات */
} XcursorImage;
XcursorImages
يحتوي على هياكل XcursorImage متعددة. يتم تحريرها جميعًا عند تحرير XcursorImages في XcursorImagesDestroy. 


typedef struct _XcursorImages {
    int 	 nimage; 	/* عدد الصور */
    XcursorImage 	**images; 	/* مصفوفة من مؤشرات XcursorImage */
} XcursorImages;
XcursorCursors
يحتوي على كائنات مؤشر متعددة. يتم تحريرها جميعًا عند تحرير XcursorCursors. هذه مرجعية العدد حتى تتمكن هياكل XcursorAnimate المتعددة من استخدام نفس XcursorCursors. 


typedef struct _XcursorCursors {
    Display 	*dpy; 	/* العرض الذي يحمل المؤشرات */
    int 	 ref; 	/* عدد المراجع */
    int 	 ncursor; 	/* عدد المؤشرات */
    Cursor 	*cursors; 	/* مصفوفة المؤشرات */
} XcursorCursors;
XcursorAnimate
يشير إلى مجموعة من المؤشرات وتسلسل داخل تلك المجموعة. قد تشير هياكل XcursorAnimate المتعددة إلى نفس XcursorCursors؛ كل منها يحمل مرجعًا يتم إزالته عند تحرير XcursorAnimate. 


typedef struct _XcursorAnimate {
    XcursorCursors 	*cursors; 	/* قائمة المؤشرات للاستخدام */
    int 	 sequence; 	/* أي مؤشر هو التالي */
} XcursorAnimate;
XcursorFile
توفر Xcursor واجهة برمجة تطبيقات مجردة للوصول إلى بيانات الملف. توفر Xcursor تطبيق stdio لهذه الواجهة البرمجية المجردة؛ التطبيقات حرة في إنشاء تطبيقات إضافية. توازي هذه الوظائف f stdio

XcursorFile

تُخزّن هذه البنية المؤشر، أو مجموعة من المؤشرات، أو مجموعة من المؤشرات، أو مجموعة من الصور، أو تعليقات، وما إلى ذلك. ولكن لا يتم تهيئة العديد من الحقول. 


typedef struct _XcursorFile {
    void *closure;
    int (*read)  (XcursorFile *file, unsigned char *buf, int len);
    int (*write) (XcursorFile *file, unsigned char *buf, int len);
    int (*seek)  (XcursorFile *file, long offset, int whence);
};

تبحث هذه الوظائف في مسار المكتبة، وتقوم بتحميل أول ملف تم العثور عليه بالحجم المطلوب، باستخدام وظيفة خاصة (XcursorScanTheme) للعثور على السمة المناسبة:

  • إذا كانت theme ليست NULL، فاستخدمها.
  • إذا كانت theme NULL، أو إذا لم يتم العثور على تطابق للسمة المطلوبة، فاستخدم "default" لاسم السمة.
  • إذا فشل كلا البحثين، ترجع هذه الوظائف NULL.
تختلف الوظيفتان بأكثر من عدد الصور التي تم تحميلها:
  • تستدعي XcursorLibraryLoadImage الدالة XcursorFileLoadImage ولكن
  • تستدعي XcursorLibraryLoadImages الدالة XcursorFileLoadImages، وتقوم (عند النجاح) باستدعاء XcursorImagesSetName لربط name بالنتيجة.

DESCRIPTION

                    
These functions load cursor images and cursors.

                

                

                    
FUNCTIONAL OVERVIEW

                    

                        
XcursorImageCreate

                        
Creates an empty XcursorImage structure with the given width and height.

                        
XcursorImageDestroy

                        
Destroys an XcursorImage structure and frees its memory.

                        
XcursorImageLoadFromBuffer

                        
Loads an XcursorImage from a buffer.

                        
XcursorShapeLoadImage

                        
Loads an XcursorImage from a file. The file is searched for in the
                            directories specified by the XcursorHome environment variable,
                            or in the default themes directories.

                        
XcursorShapeLoadFromBuffer

                        
Loads an XcursorImage from a buffer.

                        
XcursorShapeLoadCursor

                        
Loads a cursor from a file. The file is searched for in the
                            directories specified by the XcursorHome environment variable,
                            or in the default themes directories.

                        
XcursorShapeLoadFromBufferCursor

                        
Loads a cursor from a buffer.

                        
XcursorShapeParse

                        
Parses a cursor file and returns an XcursorImage. This function is deprecated.

                        
XcursorShapeParseFile

                        
Parses a cursor file and returns an XcursorImage. This function is deprecated.

                        
XcursorShapeRenderCursor

                        
Renders an XcursorImage as a cursor.

                        
XcursorShapeRenderImage

                        
Renders an XcursorImage as a cursor, allowing the hotspot to be specified.

                        
XcursorShapeSetThemeCursor

                        
Sets the current cursor theme and the name of the cursor within that theme.

                        
XcursorShapeSetThemeCore

                        
Sets the current core cursor theme and the name of the cursor within that theme.

                        
XcursorGetThemeCore

                        
Gets the current core cursor theme and name.

                    

                

                

                    
DATATYPES

                    
The XcursorImage structure is defined as follows:

                    
typedef struct {
    int width, height;
    int xhot, yhot;
    unsigned long *pixels;
    unsigned char *mask;
    int size; /* size of pixel data in bytes */
    int ncolors; /* number of colors in the color map */
    XcursorColor *colors; /* color map */
} XcursorImage;
                    

                    
The XcursorColor structure is defined as follows:

                    
typedef struct {
    unsigned short red, green, blue;
} XcursorColor;
                    

                

                

                    
FUNCTIONS

                    
The functions listed in the SYNOPSIS section are the interface to the
                        Xcursor library.

                

                

                    
ENVIRONMENT

                    

                        
XCURSOR_HOME

                        
Specifies the directory to search for cursor themes. If not set, the
                            default themes directories are used.

                        
Xcursor.theme_core

                        
Specifies the core cursor theme. This is a resource that can be set
                            in the X server's resource database.

                    

                    
The setting of the Xcursor.theme_core resource is controlled
                        by the HEME_CORE tag.

                    

                        
HEME_CORE

                        
Enables themes for core cursors if the environment variable is
                            true.

                    

                    

                        

                        
If the environment variable is not given, Xcursor tries the
                            Xcursor.theme_core resource.

                        

                        
An application can enable or disable themes using
                            XcursorSetThemeCore.

                    

                

                

                    
CAVEATS

                    
Xcursor will probably change radically in the future; weak
                        attempts will be made to retain some level of source-file compatibility.

البيئة

                            Xcursor.theme_core.

                            XcursorSetThemeCore.
