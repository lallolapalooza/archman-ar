xcb_grab_keyboard - الاستيلاء على لوحة المفاتيح

الملخص

#include <xcb/xproto.h>

دالة الطلب

xcb_grab_keyboard_cookie_t xcb_grab_keyboard(xcb_connection_t *conn, uint8_t owner_events, xcb_window_t grab_window, xcb_timestamp_t time, uint8_t pointer_mode, uint8_t keyboard_mode);

    uint8_t  status;

    uint32_t length;

وسائط الطلب

conn
اتصال XCB بـ X11.
owner_events
إذا كانت 1، فستظل النافذة grab_window تتلقى أحداث المؤشر. إذا كانت 0، فلن يتم الإبلاغ عن الأحداث للنافذة grab_window.
grab_window
يحدد النافذة التي يجب الاستيلاء على المؤشر عليها.
time
طابع زمني لتجنب حالات السباق عند تشغيل X عبر الشبكة.

سيتم استبدال القيمة الخاصة XCB_CURRENT_TIME بالوقت الحالي للخادم.

pointer_mode
إحدى القيم التالية:
XCB_GRAB_MODE_SYNC
يبدو أن حالة لوحة المفاتيح مجمدة: لا يتم إنشاء أحداث لوحة مفاتيح أخرى بواسطة الخادم حتى يصدر العميل المستولي طلب AllowEvents للإفراج أو حتى يتم تحرير الاستيلاء على لوحة المفاتيح.
XCB_GRAB_MODE_ASYNC
تستمر معالجة أحداث لوحة المفاتيح بشكل طبيعي.
keyboard_mode
إحدى القيم التالية:
XCB_GRAB_MODE_SYNC
يبدو أن حالة لوحة المفاتيح مجمدة: لا يتم إنشاء أحداث لوحة مفاتيح أخرى بواسطة الخادم حتى يصدر العميل المستولي طلب AllowEvents للإفراج أو حتى يتم تحرير الاستيلاء على لوحة المفاتيح.
XCB_GRAB_MODE_ASYNC
تستمر معالجة أحداث لوحة المفاتيح بشكل طبيعي.

الوصف

يستولي بنشاط على التحكم في لوحة المفاتيح ويولد أحداث FocusIn و FocusOut. يتم الإبلاغ عن أحداث المفاتيح الأخرى فقط للعميل المستولي.

يتم تجاوز أي استيلاء نشط على لوحة المفاتيح بواسطة هذا العميل. إذا تم الاستيلاء بنشاط على لوحة المفاتيح بواسطة عميل آخر، يتم إرجاع AlreadyGrabbed. إذا لم تكن النافذة grab_window قابلة للعرض، يتم إرجاع GrabNotViewable. إذا تم تجميد لوحة المفاتيح بواسطة استيلاء نشط لعميل آخر، يتم إرجاع GrabFrozen. إذا كان الطابع الزمني المحدد سابقًا لوقت الاستيلاء الأخير على لوحة المفاتيح أو لاحقًا للوقت الحالي لخادم X، يتم إرجاع GrabInvalidTime. بخلاف ذلك، يتم تعيين وقت الاستيلاء الأخير على لوحة المفاتيح إلى الطابع الزمني المحدد.

قيمة الإرجاع

يعيد xcb_grab_keyboard_cookie_t. يجب التعامل مع الأخطاء عند استدعاء دالة الرد xcb_grab_keyboard_reply.

إذا كنت ترغب في معالجة الأخطاء في حلقة الأحداث بدلاً من ذلك، استخدم xcb_grab_keyboard_unchecked. انظر xcb-requests(3) للتفاصيل.

مثال

/*
 * يستولي على لوحة المفاتيح بنشاط
 *
 */
void my_example(xcb_connection_t *conn, xcb_screen_t *screen) {
    xcb_grab_keyboard_cookie_t cookie;
    xcb_grab_keyboard_reply_t *reply;
    cookie = xcb_grab_keyboard(
        conn,
        true,                /* الإبلاغ عن الأحداث */
        screen->root,        /* الاستيلاء على النافذة الجذر */
        XCB_CURRENT_TIME,
        XCB_GRAB_MODE_ASYNC, /* معالجة الأحداث كالمعتاد، لا تتطلب المزامنة */
        XCB_GRAB_MODE_ASYNC
    );
    if ((reply = xcb_grab_keyboard_reply(conn, cookie, NULL))) {
        if (reply->status == XCB_GRAB_STATUS_SUCCESS)
            printf("تم الاستيلاء بنجاح على لوحة المفاتيح\n");
        free(reply);
    }
}
