XcursorFileLoadImage(3)

الحزمة: libxcursor

XCURSOR(3) X الإصدار 11 XCURSOR(3)

الاسم

Xcursor - مكتبة إدارة المؤشرات

الملخص

#include <X11/Xcursor/Xcursor.h>

الوصف

Xcursor مكتبة بسيطة مصممة للمساعدة في تحديد موقع المؤشرات وتحميلها. يمكن تحميل المؤشرات من الملفات أو من الذاكرة. توجد مكتبة من المؤشرات الشائعة التي تُطابق أسماء مؤشرات X القياسية. يمكن أن توجد المؤشرات بعدة أحجام، وتختار المكتبة تلقائيًا أفضل حجم.

نظرة عامة وظيفية

بُنيت Xcursor على عدة طبقات؛ في الطبقة السفلى توجد شيفرة يمكنها تحميل صور المؤشرات من الملفات. وفوق ذلك طبقة تحدد مواقع ملفات المؤشرات بناءً على مسار المكتبة والسمة. وفي الأعلى طبقة تبني المؤشرات إما من صورة محمّلة من ملف أو من أحد مؤشرات X القياسية. عند استخدام صور محمّلة من ملفات، تفضّل Xcursor استخدام طلب CreateCursor من امتداد Render إذا كان مدعومًا من خادم X. وحيث لا يكون مدعومًا، تُطابق Xcursor صورة المؤشر مع مؤشر X قياسي وتستخدم طلب CreateCursor الأساسي.

ملفات المؤشرات

تُعرّف Xcursor تنسيقًا جديدًا للمؤشرات على القرص. يحمل كل ملف صورة مؤشر واحدة أو أكثر. تُوسم كل صورة مؤشر بحجم اسمي بحيث يمكن اختيار أفضل حجم تلقائيًا. يمكن تحميل عدة مؤشرات من الحجم الاسمي نفسه معًا؛ ويُتوقع من التطبيقات استخدامها كتسلسل متحرك.

تُخزّن ملفات المؤشرات كترويسة تحتوي على جدول محتويات يتبعه تسلسل من الكتل. يشير جدول المحتويات إلى النوع والنوع الفرعي والموضع في الملف لكل كتلة. تبدو ترويسة الملف كما يلي:


magic:	CARD32 "Xcur" (0x58, 0x63, 0x75, 0x72)
header:	CARD32 عدد البايتات في هذه الترويسة
version:	CARD32 رقم إصدار الملف
ntoc:	CARD32 عدد مدخلات جدول المحتويات
toc:	LISTofTOC جدول المحتويات

يبدو كل مدخل في جدول المحتويات كما يلي:


type:	CARD32 نوع المدخل
subtype:	CARD32 تسمية خاصة بالنوع - الحجم بالنسبة للصور
position:	CARD32 موضع البايت المطلق للجدول في الملف

تحتوي كل كتلة في الملف على مجموعة من حقول الترويسة المشتركة تتبعها حقول إضافية خاصة بالنوع:


header:	CARD32 عدد البايتات في ترويسة الكتلة (بما في ذلك الحقول الخاصة بالنوع)
type:	CARD32 يجب أن يطابق النوع في جدول المحتويات لهذه الكتلة
subtype:	CARD32 يجب أن يطابق النوع الفرعي في جدول المحتويات لهذه الكتلة
version:	CARD32 رقم الإصدار لنوع هذه الكتلة

يوجد حاليًا نوعان من الكتل معرّفان لملفات المؤشرات؛ التعليقات والصور. تبدو التعليقات كما يلي:


header:	20 ترويسات التعليقات هي 20 بايت
type:	0xfffe0001 نوع التعليق هو 0xfffe0001
subtype:	{ 1 (COPYRIGHT), 2 (LICENSE), 3 (OTHER) }
version:	1
length:	CARD32 طول البايت لسلسلة UTF-8
string:	LISTofCARD8 سلسلة UTF-8

تبدو الصور كما يلي:


header:	36 ترويسات الصور هي 36 بايت
type:	0xfffd0002 نوع الصورة هو 0xfffd0002
subtype:	CARD32 النوع الفرعي للصورة هو الحجم الاسمي
version:	1
width:	CARD32 يجب أن يكون أقل من أو يساوي 0x7fff
height:	CARD32 يجب أن يكون أقل من أو يساوي 0x7fff
xhot:	CARD32 يجب أن يكون أقل من أو يساوي العرض
yhot:	CARD32 يجب أن يكون أقل من أو يساوي الارتفاع
delay:	CARD32 التأخير بين إطارات الحركة بالمللي ثانية
pixels:	LISTofCARD32 بكسلات بتنسيق ARGB المعبأ

السمات

تتبع Xcursor (في الغالب) مواصفات freedesktop.org لسمات الأيقونات. مسار البحث الافتراضي الذي تستخدمه هو

~/.local/share/icons, ~/.icons, /usr/share/icons, /usr/share/pixmaps

داخل كل من هذه الأدلة، تبحث عن دليل باستخدام اسم السمة:

•
داخل دليل السمة، تبحث عن ملفات المؤشرات في الدليل الفرعي “cursors”.
تبحث Xcursor عن ملف محدد، يجب أن يكون أحد أسماء شكل المؤشر، على سبيل المثال كما تُستخدم في XcursorLibraryLoadImage أو XcursorLibraryShape.
•
إذا لم تجد أي ملف مؤشر مطابق في الدليل الفرعي “cursors”، فستبحث Xcursor بعد ذلك عن ملف “index.theme” في كل دليل سمة للبحث عن السمات الموروثة. تلك أسطر بهذا التنسيق:
Inherits = theme-name

تستخدم Xcursor أول theme-name موروث، متجاهلةً غيرها مما قد يوجد في ملف “index.theme” معين. إذا وجدت سمة موروثة، تبحث Xcursor على طول المسار لاستخدامها أيضًا. تتجاهل Xcursor المفاتيح الأخرى في ملف “index.theme”، بما في ذلك “Name” (أي الاسم الذي قد يستخدمه تطبيق رسومي كـاسم العرض).

قد يُدرج أكثر من theme-name واحد في سطر Inherits=. تنص مواصفات freedesktop.org على أن عناصر القائمة مفصولة بفواصل. تقبل Xcursor أيضًا الفاصلة المنقوطة، لكنها تحوّل كليهما إلى نقطتين عند البحث في المسار. تتوقع Xcursor سطر Inherits= واحدًا فقط؛ مواصفات freedesktop.org غير واضحة بشأن ما إذا كانت المفاتيح المتعددة مسموحًا بها.

إذا لم تُضبط أي سمة، أو إذا لم يُعثر على أي مؤشر للسمة المحددة في أي مكان على طول المسار، تتحقق Xcursor من السمة “default”.

عندما تجد Xcursor ملف مؤشر، تتوقف عن البحث. تستخدم دائمًا أول ملف مؤشر يُعثر عليه أثناء البحث على طول المسار.

أنواع البيانات

XcursorImage
تحمل صورة مؤشر واحدة في الذاكرة. كل بكسل في المؤشر هو قيمة 32-بت تحتوي على ARGB مع A في البايت العالي. 


typedef struct _XcursorImage {
    XcursorDim 	 size; 	/* الحجم الاسمي للمطابقة */
    XcursorDim 	 width; 	/* العرض الفعلي */
    XcursorDim 	 height; 	/* الارتفاع الفعلي */
    XcursorDim 	 xhot; 	/* النقطة الساخنة x (يجب أن تكون داخل الصورة) */
    XcursorDim 	 yhot; 	/* النقطة الساخنة y (يجب أن تكون داخل الصورة) */
    XcursorPixel 	*pixels; 	/* مؤشر إلى البكسلات */
} XcursorImage;
XcursorImages
تحمل عدة بنى XcursorImage. يتم تحريرها جميعًا عند تحرير XcursorImages في XcursorImagesDestroy. 


typedef struct _XcursorImages {
    int 	 nimage; 	/* عدد الصور */
    XcursorImage 	**images; 	/* مصفوفة من مؤشرات XcursorImage */
} XcursorImages;
XcursorCursors
تحمل عدة كائنات Cursor. يتم تحريرها جميعًا عند تحرير XcursorCursors. هذه تُعدّ مرجعيًا بحيث يمكن لعدة بنى XcursorAnimate استخدام نفس XcursorCursors. 


typedef struct _XcursorCursors {
    Display 	*dpy; 	/* العرض الذي يحمل المؤشرات */
    int 	 ref; 	/* عدد المراجع */
    int 	 ncursor; 	/* عدد المؤشرات */
    Cursor 	*cursors; 	/* مصفوفة من المؤشرات */
} XcursorCursors;
XcursorAnimate
تشير إلى مجموعة من المؤشرات وتسلسل داخل تلك المجموعة. قد تشير عدة بنى XcursorAnimate إلى نفس XcursorCursors؛ كل واحدة تحمل مرجعًا يُزال عند تحرير XcursorAnimate. 


typedef struct _XcursorAnimate {
    XcursorCursors 	*cursors; 	/* قائمة المؤشرات المراد استخدامها */
    int 	 sequence; 	/* أي مؤشر هو التالي */
} XcursorAnimate;
XcursorFile
توفر Xcursor واجهة برمجية مجردة للوصول إلى بيانات الملف. توفر Xcursor تنفيذًا بـ stdio لهذه الواجهة المجردة؛ والتطبيقات حرة في إنشاء تنفيذات إضافية. تتوازى هذه الدوال مع دوال stdioوظائف في القيمة المُرجعة وقيم الوسائط المتوقعة؛ تقوم دالّتا القراءة والكتابة بتبديل الوسائط لتتوافق مع إصدارات POSIX. 


typedef struct _XcursorFile {
    void *closure;
    int (*read)  (XcursorFile *file, unsigned char *buf, int len);
    int (*write) (XcursorFile *file, unsigned char *buf, int len);
    int (*seek)  (XcursorFile *file, long offset, int whence);
};

تبحث هذه الدوال في مسار المكتبة، وتحمّل أول ملف يُعثر عليه بالحجم size المطلوب، باستخدام دالة خاصة (XcursorScanTheme) للعثور على السمة المناسبة:

  • إذا كانت theme ليست NULL، فاستخدمها.
  • إذا كانت theme هي NULL، أو إذا لم يكن هناك تطابق للسمة المطلوبة، فاستخدم "default" كاسم للسمة.
  • إذا فشل كلا البحثين، تُرجع هذه الدوال NULL.
تختلف الدالتان بأكثر من عدد الصور المحمّلة:
  • تستدعي XcursorLibraryLoadImage الدالة XcursorFileLoadImage لكن
  • تستدعي XcursorLibraryLoadImages الدالة XcursorFileLoadImages، و(عند النجاح) تستدعي XcursorImagesSetName لربط name بالنتيجة.
