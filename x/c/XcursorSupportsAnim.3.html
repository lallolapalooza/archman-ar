XcursorSupportsAnim(3)

الحزمة: libxcursor

التراخيص: HPND-sell-variant

سيحتوي الفهرس على العديد من المؤشرات.

void XcursorCommentsDestroy (
	XcursorComments 	*comments)

يقوم بتحرير بنية XcursorComments المعطاة بالإضافة إلى هياكل XcursorComment التي تشير إليها.

المؤشرات المتحركة

XcursorAnimate * XcursorAnimateCreate (
	XcursorCursors 	*cursors)

يغلف مصفوفة المؤشرات المعطاة في بنية XcursorAnimate مخصصة حديثًا، والتي تضيف رقم تسلسل يستخدم في XcursorAnimateNext.

void XcursorAnimateDestroy (
	XcursorAnimate 	*animate)

يتجاهل بيانات animate المعطاة، ويحرر كل من مصفوفة المؤشرات XcursorCursors وبنية XcursorAnimate.

Cursor XcursorAnimateNext (
	XcursorAnimate 	*animate)

يعيد بشكل دوري المؤشر التالي في المصفوفة، ويزيد رقم التسلسل للتحضير للاستدعاء التالي.

المتصل مسؤول عن عرض سلسلة صور المؤشرات. Xcursor لا يقوم بذلك.

واجهات برمجة تطبيقات مؤشرات الرموز

تستخدم دوال X11 XCreateFontCursor و XCreateGlyphCursor هذا الجزء من واجهة برمجة التطبيقات لتوسيع ميزة مؤشرات X الأساسية لاستخدام السمات.

void XcursorImageHash (
	XImage 	*image,
	unsigned char 	 hash[XCURSOR_BITMAP_HASH_SIZE])

يحسب تجزئة للصورة، لعرضها عندما يكون متغير البيئة XCURSOR_DISCOVER مضبوطًا.

void XcursorImagesSetName (
	XcursorImages 	*images,
	const char 	*name)

يربط الاسم المعطى بالصور.

void XcursorNoticeCreateBitmap (
	Display 	*dpy,
	Pixmap 	 pid,
	unsigned int 	 width,
	unsigned int 	 height)

يتحقق مما إذا كان العرض يدعم ARGB أو السمات، وما إذا كان حجم الصورة يتناسب مع الحد الأقصى لحجم المؤشر (64 بكسل). إذا كان الأمر كذلك، ينشئ خريطة بت بالحجم المحدد، ويخزن النتيجة مؤقتًا في Xcursor، مع تحديدها بواسطة معرف خريطة البت (pid).

void XcursorNoticePutBitmap (
	Display 	*dpy,
	Drawable 	 draw,
	XImage 	*image)

يحدث محتويات الصورة في خريطة البت المحددة بواسطة قيمة الرسم (معرف خريطة بت). يجب أن تكون خريطة البت قد تم إنشاؤها بواسطة XcursorNoticeCreateBitmap.

Cursor XcursorTryShapeBitmapCursor (
	Display 	*dpy,
	Pixmap 	 source,
	Pixmap 	 mask,
	XColor 	*foreground,
	XColor 	*background,
	unsigned int 	 x,
	unsigned int 	 y)

إذا كان العرض يدعم ARGB أو السمات، يحاول تحميل مؤشر في ذاكرة التخزين المؤقت لـ Xcursor باستخدام المعامل source كمعرف خريطة بت. قد لا تكون المصدر موجودة في ذاكرة التخزين المؤقت بعد الآن. يستخدم Xcursor قيمة التجزئة لتحديد الصورة المطلوبة.

Cursor XcursorTryShapeCursor (
	Display 	*dpy,
	Font 	 source_font,
	Font 	 mask_font,
	unsigned int 	 source_char,
	unsigned int 	 mask_char,
	XColor _Xconst 	*foreground,
	XColor _Xconst 	*background)

إذا كان العرض يدعم ARGB أو السمات، يحاول تحميل مؤشر في ذاكرة التخزين المؤقت لـ Xcursor باستخدام المعامل source_char كشكل. باستخدام

  • الحجم الافتراضي من XcursorGetDefaultSize،
  • السمة الافتراضية من XcursorGetTheme، و
  • المعامل source_char كشكل،
يقوم Xcursor باستدعاء XcursorShapeLoadImages لتحميل صور المؤشرات. في حالة النجاح، يستخدم Xcursor XcursorImagesLoadCursor لتحميل معلومات المؤشر.

واجهات برمجة تطبيقات معلومات العرض

XcursorBool XcursorSupportsARGB (
	Display 	*dpy)

يعيد صحيحًا إذا كان العرض يدعم مؤشرات ARGB. بخلاف ذلك، سيتم تعيين المؤشرات إلى مؤشر X أساسي.

XcursorBool XcursorSupportsAnim (
	Display 	*dpy)

يعيد صحيحًا إذا كان العرض يدعم المؤشرات المتحركة. بخلاف ذلك، سيتم تعيين المؤشرات إلى مؤشر X أساسي.

XcursorBool XcursorSetDefaultSize (
	Display 	*dpy,
	int 	 size)

يضبط الحجم الافتراضي للمؤشرات على العرض المحدد. عند تحميل المؤشرات، ستكون المفضلة هي تلك التي يكون حجمها الاسمي الأقرب إلى هذا الحجم.

int XcursorGetDefaultSize (
	Display 	*dpy)

يحصل على حجم المؤشر الافتراضي.

	Display 	*dpy)

يحصل على حالة المؤشرات القابلة للتغيير حاليًا.

	Display 	*dpy,

يضبط اسم السمة الحالية.

البيئة

يمكن استخدام متغيرات البيئة لتجاوز إعدادات الموارد، والتي بدورها تتجاوز القيم الافتراضية المضمنة.

بعض متغيرات البيئة التي يتعرف عليها Xcursor هي قيم منطقية، يتم تحديدها على النحو التالي:

true لـ "t" أو "1" أو "y" أو "on"
false لـ "f" أو "0" أو "n" أو "off"

يتجاهل Xcursor القيم الأخرى لهذه المتغيرات المنطقية.

HOME
يفسر Xcursor "~" في قائمة البحث على أنه دليل المنزل، مستخدمًا هذا المتغير بدلاً من قاعدة بيانات كلمات المرور.
XCURSOR_ANIM
إذا كان العرض يدعم طلب Render CreateCursor، وتم تمكين ميزة Render، فتعطيل المؤشرات المتحركة إذا كان متغير البيئة false.
إذا لم يتم توفير متغير البيئة، يستخدم Xcursor المورد Xcursor.anim.
XCURSOR_CORE
إذا كان العرض يدعم طلب Render CreateCursor، فقم بتعطيل ميزة Render إذا كان متغير البيئة false.
إذا لم يتم توفير متغير البيئة، يستخدم Xcursor المورد Xcursor.core.
XCURSOR_DISCOVER
إذا تم تعيين المتغير، يقوم Xcursor بتشغيل ميزة تسجيل. يعرض قيمة التجزئة والصورة حتى يتمكن المستخدمون من رؤية اسم المؤشر المرتبط بكل صورة.
لا يوجد إعداد مورد مقابل.
XCURSOR_DITHER
يضبط هذا المتغير التعتيم المطلوب.
إذا لم يتم توفير متغير البيئة، يستخدم Xcursor المورد Xcursor.dither.
إذا لم يتم العثور على متغير البيئة أو المورد، يستخدم Xcursor "threshold".
هذه هي القيم المعترف بها:
diffuse
median
ordered
threshold
XCURSOR_PATH
يحدد هذا المتغير قائمة المسارات التي يجب البحث فيها عن المؤشرات، بدلاً من القائمة الافتراضية المضمنة.
يتم فصل الدلائل في هذا المسار بنقطتين رأسيتين (:).
XCURSOR_RESIZED
يمكّن تغيير حجم المؤشرات تلقائيًا لتحسين حجم عرضها إذا كان متغير البيئة true.
إذا لم يتم توفير متغير البيئة، يحاول Xcursor المورد Xcursor.resized.
XCURSOR_SIZE
يحدد هذا المتغير حجم المؤشر المطلوب، بالبكسل.
إذا لم يتم توفير متغير البيئة، يحاول Xcursor المورد Xcursor.size.
إذا لم يتم تحديد حجم، سواء بواسطة متغير البيئة أو إعداد المورد، يحاول Xcursor بعد ذلك إعداد المورد Xft.dpi لتخمين حجم مؤشر بحجم 16 نقطة.
أخيرًا، إذا لم يتم تعيين Xft.dpi، يستخدم Xcursor ارتفاع العرض، مقسومًا على 48 (بافتراض أن الارتفاع هو 768).
XCURSOR_THEME
يحدد هذا المتغير السمة المطلوبة.
إذا لم يتم توفير متغير البيئة، يحاول Xcursor المورد Xcursor.theme.
إذا لم يتم العثور على متغير البيئة أو المورد، يستخدم Xcursor السمة الافتراضية.
XCURSOR_THE

ENVIRONMENT

Xcursor.theme_core
يمكّن السمات لمؤشرات الـ core إذا كانت متغير البيئة true.
إذا لم يتم توفير متغير البيئة، يحاول Xcursor استخدام المورد Xcursor.theme_core.
يمكن للتطبيق تمكين أو تعطيل السمات باستخدام XcursorSetThemeCore.
