XcursorGetThemeCore(3)

الحزمة: libxcursor

التراخيص: HPND-sell-variant

XCURSOR(3) إصدار X 11 XCURSOR(3)

الاسم

Xcursor - مكتبة إدارة المؤشرات

الخلاصة

#include <X11/Xcursor/Xcursor.h>

الوصف

Xcursor هي مكتبة بسيطة مصممة للمساعدة في تحديد وتحميل المؤشرات. يمكن تحميل المؤشرات من الملفات أو الذاكرة. توجد مكتبة للمؤشرات الشائعة التي ترتبط بأسماء مؤشرات X القياسية. يمكن أن توجد المؤشرات بأحجام متعددة وتختار المكتبة تلقائيًا أفضل حجم.

نظرة عامة وظيفية

تم بناء Xcursor على عدة طبقات؛ في الطبقة السفلية توجد الشيفرة التي يمكنها تحميل صور المؤشرات من الملفات. فوق ذلك توجد طبقة تحدد ملفات المؤشرات بناءً على مسار المكتبة والموضوع. في الأعلى توجد طبقة تبني المؤشرات إما من صورة تم تحميلها من ملف أو أحد مؤشرات X القياسية. عند استخدام الصور المحملة من الملفات، تفضل Xcursor استخدام طلب إنشاء مؤشر امتداد التقديم إذا كان مدعومًا من خادم X. حيثما لم يكن مدعومًا، تقوم Xcursor بربط صورة المؤشر بمؤشر X قياسي وتستخدم طلب إنشاء المؤشر الأساسي.

ملفات المؤشرات

تحدد Xcursor تنسيقًا جديدًا للمؤشرات على القرص. كل ملف يحتوي على صورة واحدة أو أكثر للمؤشر. يتم تمييز كل صورة مؤشر بحجم اسمي بحيث يمكن اختيار أفضل حجم تلقائيًا. يمكن تحميل مؤشرات متعددة بنفس الحجم الاسمي معًا؛ من المتوقع أن تستخدم التطبيقات هذه كرسوم متحركة متسلسلة.

يتم تخزين ملفات المؤشرات كرأس يحتوي على جدول محتويات متبوعًا بتسلسل من القطع. يشير رأس الملف إلى نوع المؤشر، والنوع الفرعي، والموضع في الملف لكل قطعة. رأس الملف يبدو كالتالي:


magic:	CARD32 "Xcur" (0x58, 0x63, 0x75, 0x72)
header:	CARD32 بايتات في هذا الرأس
version:	CARD32 رقم إصدار الملف
ntoc:	CARD32 عدد إدخالات جدول المحتويات
toc:	LISTofTOC جدول المحتويات

كل إدخال في جدول المحتويات يبدو كالتالي:


type:	CARD32 نوع الإدخال
subtype:	CARD32 تسمية خاصة بالنوع - الحجم للصور
position:	CARD32 الموضع المطلق بالبايت لجدول المحتويات في الملف

كل قطعة في الملف لها مجموعة من حقول الرأس المشتركة متبوعة بحقول إضافية خاصة بالنوع:


header:	CARD32 بايتات في رأس القطعة (بما في ذلك الحقول الخاصة بالنوع)
type:	CARD32 يجب أن تتطابق مع النوع في جدول المحتويات لهذه القطعة
subtype:	CARD32 يجب أن تتطابق مع النوع الفرعي في جدول المحتويات لهذه القطعة
version:	CARD32 رقم إصدار لهذا النوع من القطع

يوجد حاليًا نوعان من القطع معرفان لملفات المؤشرات؛ التعليقات والصور. تبدو التعليقات كالتالي:


header:	20 رؤوس التعليقات هي 20 بايت
type:	0xfffe0001 نوع التعليق هو 0xfffe0001
subtype:	{ 1 (حقوق النشر)، 2 (الترخيص)، 3 (أخرى) }
version:	1
length:	CARD32 طول البايت للسلسلة UTF-8
string:	LISTofCARD8 سلسلة UTF-8

تبدو الصور كالتالي:


header:	36 رؤوس الصور هي 36 بايت
type:	0xfffd0002 نوع الصورة هو 0xfffd0002
subtype:	CARD32 نوع الصورة الفرعي هو الحجم الاسمي
version:	1
width:	CARD32 يجب أن يكون أقل من أو يساوي 0x7fff
height:	CARD32 يجب أن يكون أقل من أو يساوي 0x7fff
xhot:	CARD32 يجب أن يكون أقل من أو يساوي العرض
yhot:	CARD32 يجب أن يكون أقل من أو يساوي الارتفاع
delay:	CARD32 التأخير بين إطارات الرسوم المتحركة بالمللي ثانية
pixels:	LISTofCARD32 وحدات البكسل بتنسيق ARGB المضغوط

المواضيع

تتبع Xcursor (في الغالب) مواصفات freedesktop.org لمواضيع الأيقونات. مسار البحث الافتراضي الذي تستخدمه هو

~/.local/share/icons, ~/.icons, /usr/share/icons, /usr/share/pixmaps

داخل كل من هذه الدلائل، تبحث عن دليل باستخدام اسم الموضوع:

•
داخل دليل الموضوع، تبحث عن ملفات المؤشرات في الدليل الفرعي "cursors".
تبحث Xcursor عن ملف محدد، والذي يجب أن يكون أحد أسماء شكل المؤشر، على سبيل المثال كما هو مستخدم في XcursorLibraryLoadImage أو XcursorLibraryShape.
•
إذا لم تجد ملف مؤشر مطابق في الدليل الفرعي "cursors"، تبحث Xcursor بعد ذلك عن ملف "index.theme" في كل دليل موضوع للبحث عن مواضيع موروثة. هذه هي سطور بهذا التنسيق:
Inherits = اسم-الموضوع

تستخدم Xcursor أول اسم-موضوع موروث، متجاهلة الآخرين الذين قد يكونون موجودين في ملف "index.theme" معين. إذا وجدت موضوعًا موروثًا، فإن Xcursor تبحث على طول المسار لاستخدامه أيضًا. تتجاهل Xcursor مفاتيح أخرى في ملف "index.theme"، بما في ذلك "Name" (أي، الاسم الذي قد يستخدمه تطبيق رسومي كاسم عرض).

يمكن إدراج أكثر من اسم-موضوع واحد في سطر Inherits=. تنص مواصفات freedesktop.org على أن عناصر القائمة مفصولة بفواصل. تقبل Xcursor أيضًا الفواصل المنقوطة، ولكنها تترجم كلاهما إلى نقطتين عند البحث في المسار. تتوقع Xcursor سطر Inherits= واحد فقط؛ مواصفات freedesktop.org غير واضحة ما إذا كان مسموحًا بمفاتيح متعددة.

إذا لم يتم تعيين موضوع، أو إذا لم يتم العثور على مؤشر للموضوع المحدد في أي مكان على طول المسار، تتحقق Xcursor من موضوع "default".

عندما تجد Xcursor ملف مؤشر، فإنها تتوقف عن البحث. إنها تستخدم دائمًا ملف المؤشر الأول الذي تم العثور عليه أثناء البحث على طول المسار.

أنواع البيانات

XcursorImage
يحمل صورة مؤشر واحدة في الذاكرة. كل بكسل في المؤشر هو قيمة 32 بت تحتوي على ARGB مع A في البايت الأعلى. 


typedef struct _XcursorImage {
    XcursorDim 	 size; 	/* حجم اسمي للمطابقة */
    XcursorDim 	 width; 	/* العرض الفعلي */
    XcursorDim 	 height; 	/* الارتفاع الفعلي */
    XcursorDim 	 xhot; 	/* نقطة ساخنة x (يجب أن تكون داخل الصورة) */
    XcursorDim 	 yhot; 	/* نقطة ساخنة y (يجب أن تكون داخل الصورة) */
    XcursorPixel 	*pixels; 	/* مؤشر إلى وحدات البكسل */
} XcursorImage;
XcursorImages
يحمل هياكل XcursorImage متعددة. يتم تحريرها جميعًا عند تحرير XcursorImages في XcursorImagesDestroy. 


typedef struct _XcursorImages {
    int 	 nimage; 	/* عدد الصور */
    XcursorImage 	**images; 	/* مصفوفة من مؤشرات XcursorImage */
} XcursorImages;
XcursorCursors
يحتوي على كائنات مؤشر متعددة. يتم تحريرها جميعًا عند تحرير XcursorCursors. هذه ذات عدد مرجعي حتى تتمكن هياكل XcursorAnimate المتعددة من استخدام نفس XcursorCursors. 


typedef struct _XcursorCursors {
    Display 	*dpy; 	/* العرض الذي يحتوي على المؤشرات */
    int 	 ref; 	/* عدد المراجع */
    int 	 ncursor; 	/* عدد المؤشرات */
    Cursor 	*cursors; 	/* مصفوفة المؤشرات */
} XcursorCursors;
XcursorAnimate
يشير إلى مجموعة من المؤشرات وتسلسل ضمن تلك المجموعة. قد تشير هياكل XcursorAnimate متعددة إلى نفس XcursorCursors؛ كل منها يحمل مرجعًا يتم إزالته عند تحرير XcursorAnimate. 


typedef struct _XcursorAnimate {
    XcursorCursors 	*cursors; 	/* قائمة المؤشرات المراد استخدامها */
    int 	 sequence; 	/* أي مؤشر هو التالي */
} XcursorAnimate;
XcursorFile
توفر Xcursor واجهة برمجة تطبيقات مجردة للوصول إلى بيانات الملف. توفر Xcursor تطبيقًا قياسيًا لهذه الواجهة المجردة؛ التطبيقات حرة في إنشاء تطبيقات إضافية. هذه الدوال توازي وظائف stdio

NAME


XcursorGetThemeCore, XcursorSetThemeCore - get or set the core cursor theme






SYNOPSIS


#include <Xcursor/Xcursor.h>
Bool
XcursorGetThemeCore(
    Display *dpy,
    const char **theme_name_return
);
Bool
XcursorSetThemeCore(
    Display *dpy,
    const char *theme_name
);







DESCRIPTION


The Xcursor library provides functions for loading and using
    cursor themes. Themes are sets of cursor images that can be used to change the
    appearance of the mouse pointer.


XcursorGetThemeCore retrieves the name of the core cursor
    theme currently in use. The core theme is a special theme that is always
    available and is used as a fallback if the requested theme cannot be loaded.


XcursorSetThemeCore sets the name of the core cursor theme.
    This function should only be called by applications that need to explicitly
    control the core theme, such as desktop environments or window managers.






FUNCTIONAL OVERVIEW


The Xcursor library allows applications to specify a cursor
    theme to be used for displaying cursors. This theme is typically loaded from
    a set of configuration files and image files that define the cursor
    appearance.


The core cursor theme is a fallback mechanism. If an application
    requests a cursor theme that does not exist or cannot be loaded, the core
    theme is used instead. This ensures that the cursor is always visible, even
    if the preferred theme is not available.






DATATYPES


The library uses the following data types:



  
Display

  
Represents a connection to an X server.

  
const char **

  
A pointer to a pointer to a constant character string, used to return the
      theme name.

  
const char *

  
A pointer to a constant character string, used to specify the theme name.








FUNCTIONS



  
XcursorGetThemeCore

  
Retrieves the name of the core cursor theme.

  
XcursorSetThemeCore

  
Sets the name of the core cursor theme.








ENVIRONMENT



  
Xcursor.theme_core

  
A resource that specifies the name of the core cursor theme. If this
      resource is set, it overrides the value specified by the
      XCURSOR_THEME_CORE environment variable.





  
XCURSOR_THEME_CORE

  
An environment variable that specifies the name of the core cursor theme.
      If this variable is set to true, themes for core cursors are
      enabled.

  

  
If the environment variable is not given, Xcursor tries the
      Xcursor.theme_core resource.

  

  
An application can enable or disable themes using
    XcursorSetThemeCore.








CAVEATS


Xcursor will probably change radically in the future; weak
    attempts will be made to retain some level of source-file compatibility.

الاسم


XcursorGetThemeCore, XcursorSetThemeCore - الحصول على أو تعيين سمة المؤشر الأساسية






الملخص


#include <Xcursor/Xcursor.h>
Bool
XcursorGetThemeCore(
    Display *dpy,
    const char **theme_name_return
);
Bool
XcursorSetThemeCore(
    Display *dpy,
    const char *theme_name
);







الوصف


توفر مكتبة Xcursor وظائف لتحميل واستخدام سمات المؤشر. السمات هي مجموعات من صور المؤشر التي يمكن استخدامها لتغيير مظهر مؤشر الماوس.


XcursorGetThemeCore يسترد اسم سمة المؤشر الأساسية المستخدمة حاليًا. السمة الأساسية هي سمة خاصة متاحة دائمًا وتستخدم كحل بديل إذا تعذر تحميل السمة المطلوبة.


XcursorSetThemeCore يعين اسم سمة المؤشر الأساسية. يجب استدعاء هذه الوظيفة فقط من قبل التطبيقات التي تحتاج إلى التحكم صراحة في السمة الأساسية ، مثل بيئات سطح المكتب أو مديري النوافذ.






نظرة عامة وظيفية


تسمح مكتبة Xcursor للتطبيقات بتحديد سمة مؤشر لاستخدامها في عرض المؤشرات. يتم تحميل هذه السمة عادةً من مجموعة من ملفات التكوين وملفات الصور التي تحدد مظهر المؤشر.


السمة الأساسية للمؤشر هي آلية احتياطية. إذا طلب تطبيق سمة مؤشر غير موجودة أو تعذر تحميلها ، يتم استخدام السمة الأساسية بدلاً من ذلك. هذا يضمن أن المؤشر مرئي دائمًا ، حتى لو لم تكن السمة المفضلة متاحة.






أنواع البيانات


تستخدم المكتبة أنواع البيانات التالية:



  
Display

  
يمثل اتصالًا بخادم X.

  
const char **

  
مؤشر إلى مؤشر لسلسلة أحرف ثابتة ، يستخدم لإرجاع اسم السمة.

  
const char *

  
مؤشر لسلسلة أحرف ثابتة ، يستخدم لتحديد اسم السمة.








الوظائف



  
XcursorGetThemeCore

  
يسترد اسم سمة المؤشر الأساسية.

  
XcursorSetThemeCore

  
يعين اسم سمة المؤشر الأساسية.








البيئة



  
Xcursor.theme_core

  
 مورد يحدد اسم سمة المؤشر الأساسية. إذا تم تعيين هذا المورد ، فإنه يتجاوز القيمة المحددة بواسطة متغير البيئة
      XCURSOR_THEME_CORE.





  
XCURSOR_THEME_CORE

  
متغير بيئة يحدد اسم سمة المؤشر الأساسية. إذا تم تعيين هذا المتغير على true، فسيتم تمكين السمات للمؤشرات الأساسية.

  

  
إذا لم يتم تقديم متغير البيئة ، تحاول Xcursor المورد
      Xcursor.theme_core.

  

  
يمكن للتطبيق تمكين أو تعطيل السمات باستخدام
    XcursorSetThemeCore.








محاذير


Xcursor من المحتمل أن يتغير بشكل جذري في المستقبل ؛ ستُبذل محاولات ضعيفة للحفاظ على مستوى ما من التوافق مع ملفات المصدر.
