XIQueryDevice(3)

الحزمة: libxi

التراخيص: MIT-open-group, SMLNJ, MIT

XIQUERYDEVICE(3)   XIQUERYDEVICE(3)

الاسم

XIQueryDevice, XIFreeDeviceInfo - الحصول على معلومات حول الأجهزة.

الملخص

#include <X11/extensions/XInput2.h>
XIDeviceInfo* XIQueryDevice( Display *display,
                             int deviceid,
                             int *ndevices_return);
XIFreeDeviceInfo( XIDeviceInfo *info);
deviceid
       يحدد الجهاز المراد الاستعلام عنه أو XIAllDevices أو
       XIAllMasterDevices.
display
       يحدد الاتصال بخادم X.
ndevices_return
       يعيد عدد الأجهزة المعادة.
info
       قائمة هياكل XIDeviceInfo للأجهزة المراد تحريرها.

الوصف

تقوم XIQueryDevice بإعادة معلومات حول جهاز إدخال واحد أو أكثر. إذا حدد
deviceid جهازًا، فإن ndevices_return تكون 1 والمعلومات المعادية تصف فقط الجهاز المطلوب.
إذا كان deviceid هو XIAllDevices أو XIAllMasterDevices، فإن ndevices_return هو عدد الأجهزة
أو الأجهزة الرئيسية، على التوالي، وتمثل المعلومات المعادية جميع الأجهزة أو جميع الأجهزة
الرئيسية، على التوالي.
لتحرير مصفوفة XIDeviceInfo المعاد بواسطة XIQueryDevice، استخدم XIFreeDeviceInfo.
لكل جهاز إدخال مطلوب، تعيد XIQueryDevice هيكل XIDeviceInfo. يحتوي كل هيكل
على معلومات حول قدرات جهاز إدخال واحد متاح للخادم.
typedef struct
{
    int                 deviceid;
    char                *name;
    int                 use;
    int                 attachment;
    Bool                enabled;
    int                 num_classes;
    XIAnyClassInfo      **classes;
} XIDeviceInfo;
deviceid هو المعرف الرقمي الفريد للجهاز. يكون deviceid فريدًا
طوال فترة حياة الجهاز، ولكن يمكن للخادم إعادة استخدام المعرف بمجرد إزالة
الجهاز.
يشير name إلى سلسلة نصية تنتهي بحرف null تحدد معرّف الجهاز.
تحدد حقلا use و attachment نوع الجهاز والحقيبة أو الاقتران الحالي.
- إذا كان use هو XIMasterPointer، فإن الجهاز هو مؤشر رئيسي ويحدد attachment
  معرّف الجهاز الخاص بلوحة المفاتيح الرئيسية المقترنة.
- إذا كان use هو XIMasterKeyboard، فإن الجهاز هو لوحة مفاتيح رئيسية، ويحدد
  حقل attachment الجهاز المقترن الرئيسي.
- إذا كان use هو XISlavePointer، فإن الجهاز هو جهاز تابع ومتصل حاليًا
  بالمؤشر الرئيسي المحدد في attachment.
- إذا كان use هو XISlaveKeyboard، فإن الجهاز هو جهاز تابع ومتصل حاليًا
  بلوحة المفاتيح الرئيسية المحددة في attachment.
- إذا كان use هو XIFloatingSlave، فإن الجهاز هو جهاز تابع غير متصل حاليًا
  بأي جهاز رئيسي. قيمة حقل attachment للأجهزة التابعة العائمة غير محددة.
يحدد الحقل enabled ما إذا كان الجهاز ممكّنًا حاليًا ويمكنه إرسال أحداث.
الأجهزة المعطلة لن ترسل أحداثًا.
يحدد الحقل num_classes عدد فئات الإدخال التي تشير إليها classes.
typedef struct
{
    int         type;
    int         sourceid;
} XIAnyClassInfo;
يحدد الحقل type نوع فئة الإدخال.
حالياً، الأنواع التالية محددة:
    XIKeyClass, XIButtonClass, XIValuatorClass, XIScrollClass,
    XITouchClass, XIGestureClass.
في المستقبل، يمكن إضافة أنواع إضافية. يُطلب من العملاء تجاهل فئات الإدخال
غير المعروفة.
sourceid هو deviceid الذي نشأت منه هذه الفئة. بالنسبة للأجهزة الرئيسية،
يكون sourceid عادةً هو معرّف الجهاز التابع الذي يرسل الأحداث حاليًا. بالنسبة
للأجهزة التابعة، يكون sourceid عادةً هو معرّف الجهاز.
قد يحتوي الجهاز على صفر أو واحد XIButtonClass، مما يشير إلى قدرة الجهاز
على إرسال أحداث الأزرار.
typedef struct {
    int           mask_len;
    unsigned char *mask;
} XIButtonState;
typedef struct
{
    int         type;
    int         sourceid;
    int         num_buttons;
    Atom        *labels;
    XIButtonState state;
} XIButtonClassInfo;
يحدد الحقل num_buttons عدد الأزرار المتاحة على هذا الجهاز. يجب أن يحتوي
الجهاز الذي يحتوي على XIButtonClass على زر واحد على الأقل.
labels هي قائمة من num_buttons Atoms تحدد تسميات الأزرار لهذا الجهاز.
إذا لم تكن التسمية None، فإن التسمية تحدد نوع الزر بالترتيب المادي للجهاز
(أي كما يتم ترقيم الأزرار على جهاز الإدخال المادي).
state هو حالة الزر الحالية كما يراها العملاء (أي بعد تطبيق تعيين الأزرار).
يحدد الحقل mask_len طول القناع بالبايت. لكل زر على الجهاز، يتم تعيين البت
المقابل في mask إذا كان الزر مضغوطًا منطقيًا حاليًا.
قد يحتوي الجهاز على صفر أو واحد XIKeyClass، مما يشير إلى قدرة الجهاز
على إرسال أحداث المفاتيح.
typedef struct
{
    int         type;
    int         sourceid;
    int         num_keycodes;
    int         *keycodes;
} XIKeyClassInfo;
يحدد الحقل num_keycodes عدد رموز المفاتيح المتاحة على هذا الجهاز. يجب
أن يحتوي الجهاز الذي يحتوي على XIKeyClass على رمز مفتاح واحد على الأقل.
keycodes هي قائمة من num_keycodes رموز المفاتيح التي يمكن للجهاز إرسالها.
قد يحتوي الجهاز على صفر أو أكثر من XIValuatorClass، مما يشير إلى قدرة
الجهاز على إرسال الإحداثيات.
typedef struct
{
    int         type;
    int         sourceid;
    int         number;
    Atom        label;
    double      min;
    double      max;
    double      value;
    int         resolution;
    int         mode;
} XIValuatorClassInfo;
يحدد الحقل number رقم المحور على الجهاز المادي.
إذا لم يكن الحقل label هو None، فإن قيمة label هي Atom تصف المحور.
min و max هما الحد الأدنى والأقصى للقيم المسموح بها على هذا المحور.
إذا كان كلاهما صفرًا، فلا توجد قيم دنيا أو قصوى محددة على هذا الجهاز.
value هي القيمة الحالية لهذا المحور.
يحدد الحقل resolution دقة الجهاز بوحدات/م.
يحدد mode وضع هذا المحور. إذا كان mode هو XIModeAbsolute، يرسل هذا
المحور إحداثيات مطلقة. إذا كان mode هو XIModeRelative، يرسل هذا الجهاز
إحداثيات نسبية.
typedef struct
{
    int         type;
    int         sourceid;
    int         number;
    int         scroll_type;
    double      increment;
    int         flags;
} XIScrollClassInfo;
تصف هذه الفئة قدرة التمرير على مقيّم. لكل XIScrollClassInfo، يوجد
XIValuatorClassInfo بنفس الرقم على الجهاز.
يحدد الحقل number رقم المقيّم على الجهاز المادي الذي تنطبق عليه
معلومات التمرير هذه. ارجع إلى XIValuatorClassInfo المناسب للحصول على معلومات
مفصلة حول هذا المقيّم.
يحدد الحقل scroll_type نوع التمرير، إما XIScrollTypeVertical أو
XIScrollTypeHorizontal.
يحدد increment قيمة التغيير التي تعتبر وحدة واحدة للتمرير لأسفل.
يحدد الحقل flags العلامات التي تنطبق على معلومات التمرير هذه:
إذا تم تعيين XIScrollFlagNoEmulation، فلن يقوم الخادم بمحاكاة أحداث
الأزرار القديمة لتغييرات المقيّم على هذا المقيّم.
إذا تم تعيين XIScrollFlagPreferred، فإن هذا المحور هو المحور المفضل
لنوع التمرير هذا وسيتم استخدامه لمحاكاة أحداث XI_Motion عند إرسال
برنامج التشغيل لأحداث أزرار التمرير القديمة.
typedef struct
{
    int         type;
    int         sourceid;
    int         mode;
    int         num_touches;
} XITouchClassInfo;
قد يحتوي الجهاز على صفر أو واحد XITouchClassInfo، مما يشير إلى قدرة
اللمس المتعدد على الجهاز. قد يرسل الجهاز الذي يحتوي على XITouchClassInfo
أحداث TouchBegin و TouchUpdate و TouchEnd و TouchOwnership.
الحقل mode هو إما XIDirectTouch لأجهزة اللمس ذات الإدخال المباشر مثل
شاشات اللمس أو XIDependentTouch لأجهزة الإدخال غير المباشرة مثل لوحات اللمس.
بالنسبة لأجهزة XIDirectTouch، يتم إرسال أحداث اللمس إلى النافذة في الموضع
الذي حدث فيه اللمس. بالنسبة لأجهزة XIDependentTouch، يتم إرسال أحداث
اللمس إلى النافذة في موضع صورة الجهاز.
