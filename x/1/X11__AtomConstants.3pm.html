X11__AtomConstants(3pm)

الحزمة: perl-x11-protocol-other

التراخيص: GPL

X11::AtomConstants(3) وثائق Perl مساهمة من المستخدم X11::AtomConstants(3)

الاسم

X11::AtomConstants -- ذرات X11 معرفة مسبقًا

المُلَخَّص

use X11::AtomConstants;
$X->ChangeProperty($window,
                   X11::AtomConstants::WM_NAME,  # الخاصية
                   X11::AtomConstants::PIXMAP,   # النوع
                   8,                            # الصيغة
                   'Replace',
                   'My Window Title');

الوصف

هذه هي الذرات المعرفة مسبقًا في X11 كثوابت Perl. على سبيل المثال "PIXMAP" هي 20، وهي ذرة PIXMAP.

my $atom_id = X11::AtomConstants::PIXMAP;  # $atom_id هو 20
my $name = $X->GetAtomName($atom_id);      # $name هو "PIXMAP"

الثابت "LAST_PREDEFINED" ليس ذرة بحد ذاتها، ولكنه آخر قيمة معرفة مسبقًا، وهو "WM_TRANSIENT_FOR".

التصدير

لا يتم تصدير أي شيء افتراضيًا، ولكن يمكن طلب الثوابت بأسلوب "Exporter" المعتاد،

use X11::AtomConstants 'PIXMAP';
blah (atom => PIXMAP);

العلامة ":all" تقوم باستيراد كل شيء،

use X11::AtomConstants ':all';
blah (atom1 => POINT,
      atom2 => RECTANGLE);

العديد من الذرات مثل "STRING" أو "WEIGHT" لها أسماء عامة إلى حد ما وقد يكون استيرادها غير مرغوب فيه. يمكن دائمًا استخدامها مع اسم الحزمة الكامل،

use X11::AtomConstants;
blah (atom => X11::AtomConstants::PIXMAP);
