X11__Protocol__Ext__RENDER(3pm)

الحزمة: perl-x11-protocol

التراخيص: GPL, PerlArtistic

Protocol::Ext::RENDER(3) وثائق بيرل مساهمة من المستخدم Protocol::Ext::RENDER(3)

اسم

X11::Protocol::Ext::RENDER - وحدة بيرل لملحق X Rendering

ملخص

use X11::Protocol;
$x = X11::Protocol->new($ENV{'DISPLAY'});
$x->init_extension('RENDER') or die;

الوصف

يضيف ملحق RENDER مجموعة جديدة من الأشكال الرسمية التي تمثل فعليًا بديلاً لروتينات الرسم في البروتوكول الأساسي، وتم إعادة تصميمها بناءً على احتياجات العملاء الأكثر حداثة. تضيف ميزات مرغوبة منذ فترة طويلة مثل تحديد البكسل الفرعي، والتكويم الشفاف (alpha compositing)، والتحديد المباشر للألوان، والمؤشرات متعددة الألوان أو المتحركة. من ناحية أخرى، فإنه يتجاهل الميزات التي لم تعد شائعة الاستخدام: الخطوط العريضة، المضلعات العشوائية (يتم دعم المثلثات وشبه المنحرفات الأفقية فقط)، الأشكال البيضاوية، عمليات الرسم بالبت، والخطوط من جانب الخادم (بدلاً من "الأحرف" التي يتم رسمها من جانب العميل ونقلها مرة واحدة).

حتى وقت كتابة هذا التقرير (أوائل عام 2004)، لا تزال المواصفات والتنفيذ تحتوي على بعض العيوب، ولكن هناك بدائل قليلة نسبيًا لتفريغ معالجة الرسوميات المعقدة إلى الخادم، كما هو مطلوب عبر الروابط البطيئة أو إذا كان العميل مكتوبًا بلغة بطيئة. احتمال آخر قد تفكر فيه هو المجموعة الفرعية ثنائية الأبعاد من OpenGL، على الرغم من أنها لا تحتوي بعد على واجهة متوافقة مع X11::Protocol.

الثوابت الرمزية

يضيف هذا الملحق أنواع الثوابت 'PictType' و 'PictOp' و 'SubPixel' و 'PolyEdge' و 'PolyMode'، مع القيم كما هو محدد في المعيار.

الطلبات

يضيف هذا الملحق عدة طلبات، يتم استدعاؤها كما هو موضح أدناه:

$x->RenderQueryVersion($major, $minor)
=>
($major, $minor)
$x->RenderQueryPictFormats()
=>
([[$id, $type, $depth,
   $red, $red_m, $green, $green_m, $blue, $blue_m,
   $alpha, $alpha_m, $cmap], ...],
 [[$fallback, [$depth, [$visual, $format], ...], ...], ...],
 [$subpixel, ...])
$x->RenderQueryPictIndexValues($pict_format)
=>
([$index, $red, $green, $blue, $alpha], ...)
$x->RenderQueryFilters($drawable)
=>
([@filters], [@aliases])
$x->RenderCreatePicture($picture, $drawable, $format,
                        'attribute' => $value, ...)
$x->RenderChangePicture($picture, 'attribute' => $value, ...)
$x->RenderSetPictureClipRectangles($pic, $x_origin, $y_origin,
                                   [$x, $y, $width, $height], ...)
$x->RenderSetPictureTransform($pict, $m11, $m12, $m13,
                                     $m21, $m22, $m23,
                                     $m31, $m32, $m33);
$x->RenderSetPictureFilter($pict, $filter, @args)
$x->RenderComposite($op, $src, $mask, $dst, $src_x, $src_y,
                    $mask_x, $mask_y, $dst_x, $dst_y,
                    $width, $height)
$x->RenderFillRectangles($op, $dst, [$red, $green, $blue, $alpha],
                         [$x, $y, $width, $height], ...)
$x->RenderTrapezoids($op, $src, $src_x, $src_y, $dst, $mask_format,
                     [$top, $bottom, $lx1, $ly1, $lx2, $ly2,
                                     $rx1, $ry1, $rx2, $ry2] ,...)
$x->RenderTriangles($op, $src, $src_x, $src_y, $dst, $mask_format,
                    [$x1, $y1, $x2, $y2, $x3, $y3])
$x->RenderTriStrip($op, $src, $src_x, $src_y, $dst, $mask_format,
                    [$x, $y], [$x, $y], [$x, $y], [$x, $y], ...)
$x->RenderTriFan($op, $src, $src_x, $src_y, $dst, $mask_format,
                 [$x, $y], [$x, $y], [$x, $y], [$x, $y], ...)
$x->RenderCreateGlyphSet($gsid, $format)
$x->RenderReferenceGlyphSet($gsid, $existing)
$x->RenderFreeGlyphSet($gsid)
$x->RenderAddGlyphs($gsid, [$glyph, $width, $height,
                            $x, $y, $x_off, $y_off, $data], ...)

تحذير: مع بعض تطبيقات الخادم (بما في ذلك XFree86 حتى الإصدار 4.4) قد يؤدي تمرير أكثر من حرف واحد إلى AddGlyphs إلى تعليق الخادم أو تعطيله. لذا، لا تفعل ذلك.

$x->RenderFreeGylphs($gsid, @glyphs)
$x->RenderCompositeGlyphs8($op, $src, $dst, $mask_format, $gsid,
                           $src_x, $src_y,
                           [$delta_x, $delta_y, $str], ...)
$x->RenderCompositeGlyphs16($op, $src, $dst, $mask_format, $gsid,
                            $src_x, $src_y,
                            [$delta_x, $delta_y, $str], ...)
$x->RenderCompositeGlyphs32($op, $src, $dst, $mask_format, $gsid,
                            $src_x, $src_y,
                            [$delta_x, $delta_y, $str], ...)

في هذه الطلبات الثلاثة، يمكن أيضًا إدخال GlyphSetIDs جديدة بين مراجع المصفوفات.

$x->RenderCreateCursor($cid, $source, $hot_x, $hot_y)
$x->RenderCreateAnimCursor($cid, [$cursor, $delay], ...)
