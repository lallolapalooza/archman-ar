XtAsprintf(3)

الحزمة: libxt

التراخيص: MIT, HPND-sell-variant, SMLNJ, MIT-open-group, X11

XtMalloc(3) XT FUNCTIONS XtMalloc(3)

اسم

XtMalloc, XtCalloc, XtRealloc, XtReallocArray, XtFree, XtNew, XtNewString - وظائف إدارة الذاكرة

الصيغة

#include <X11/Intrinsic.h>

char *XtMalloc(Cardinal size);

char *XtCalloc(Cardinal num, Cardinal size);

char *XtRealloc(char *ptr, Cardinal size);

void *XtReallocArray(void *ptr, Cardinal num, Cardinal size);

void XtFree(char *ptr);

type *XtNew(type);

String XtNewString(String string);

Cardinal XtAsprintf(char **new_string, const char *format, ...);

الوسائط

num
يحدد عدد عناصر المصفوفة.
ptr
يحدد مؤشرًا إلى التخزين القديم أو إلى كتلة التخزين التي سيتم تحريرها.
size
يحدد حجم عنصر المصفوفة (بالبايت) أو عدد البايتات المطلوب.
string
يحدد سلسلة نصية معرّفة مسبقًا.
type
يحدد نوع بيانات معرّف مسبقًا.
new_string
يحدد مؤشرًا لكتابة سلسلة نصية مخصصة حديثًا إليه.
format
يحدد سلسلة تنسيق كما هو محدد بواسطة sprintf(3c)

الوصف

تُرجع وظائف XtMalloc مؤشرًا إلى كتلة تخزين بحجم لا يقل عن البايتات المحددة. إذا كانت الذاكرة غير كافية لتخصيص الكتلة الجديدة، تستدعي XtMalloc XtErrorMsg.

تقوم وظيفة XtCalloc بتخصيص مساحة لعدد محدد من عناصر المصفوفة بالحجم المحدد وتهيئ المساحة إلى الصفر. إذا كانت الذاكرة غير كافية لتخصيص الكتلة الجديدة، تستدعي XtCalloc XtErrorMsg.

تقوم وظائف XtRealloc و XtReallocArray بتغيير حجم كتلة التخزين (مما قد يؤدي إلى نقلها). ثم تقوم بنسخ المحتويات القديمة (أو القدر الذي يتسع) إلى الكتلة الجديدة وتحرير الكتلة القديمة. إذا كانت الذاكرة غير كافية لتخصيص الكتلة الجديدة، أو إذا كانت العمليات الحسابية لحجم الكتلة الجديدة ستؤدي إلى تجاوز سعة عدد صحيح، فإن هذه الوظائف تستدعي XtErrorMsg. إذا كانت ptr هي NULL، فإن هذه الوظائف تقوم بتخصيص التخزين الجديد دون نسخ المحتويات القديمة؛ أي أنها تستدعي ببساطة XtMalloc.

تقوم وظيفة XtFree بإعادة التخزين والسماح بإعادة استخدامه. إذا كانت ptr هي NULL، فإن XtFree تعود فورًا.

تُرجع XtNew مؤشرًا إلى التخزين المخصص. إذا كانت الذاكرة غير كافية لتخصيص الكتلة الجديدة، تستدعي XtNew XtErrorMsg. XtNew هي ماكرو مساعد يستدعي XtMalloc مع الوسائط التالية المحددة:

((type *) XtMalloc((unsigned) sizeof(type))

تُرجع XtNewString مؤشرًا إلى سلسلة نصية جديدة وهي نسخة مكررة من string. إذا كانت الذاكرة غير كافية لتخصيص الكتلة الجديدة، أو إذا كان الوسيط NULL، فإن XtNewString تعود NULL. يمكن تحرير الذاكرة باستخدام XtFree.

تقوم وظيفة XtAsprintf بتخصيص مساحة لسلسلة نصية كبيرة بما يكفي لاستيعاب السلسلة النصية المحددة بنمط تنسيق sprintf(3c) عند استخدامها مع الوسائط المتبقية، وتملأها بالنتائج المنسقة. يتم وضع عنوان السلسلة النصية المخصصة في المؤشر الممرر كـ ret. يتم إرجاع طول السلسلة النصية (باستثناء البايت الصفري النهائي). إذا كانت الذاكرة غير كافية لتخصيص الكتلة الجديدة، تستدعي XtAsprintf XtErrorMsg.
