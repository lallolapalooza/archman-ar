XFontProp(3)

الحزمة: libx11

التراخيص: MIT AND X11

XLoadFont(3) وظائف XLIB XLoadFont(3)

الاسم

XLoadFont, XQueryFont, XLoadQueryFont, XFreeFont, XGetFontProperty, XUnloadFont, XCharStruct, XFontProp, XChar2b, XFontStruct - تحميل أو إلغاء تحميل الخطوط وهياكل مقاييس الخطوط

الصيغة

Font XLoadFont(Display *display, _Xconst char *name);

XFontStruct *XQueryFont(Display *display, XID font_ID);

XFontStruct *XLoadQueryFont(Display *display, _Xconst char *name);

int XFreeFont(Display *display, XFontStruct *font_struct);

Bool XGetFontProperty(XFontStruct *font_struct, Atom atom, unsigned long *value_return);

int XUnloadFont(Display *display, Font font);

المعاملات

atom
يحدد الذرة (atom) لاسم الخاصية التي تريد إرجاعها.
display
يحدد الاتصال بخادم X.
font
يحدد الخط.
font_ID
يحدد معرف الخط أو معرف GContext.
font_struct
يحدد التخزين المرتبط بالخط.
gc
يحدد GC (رسم سياق).
name
يحدد اسم الخط، وهو سلسلة نصية منتهية بصفر (null-terminated string).
value_return
يعيد قيمة خاصية الخط.

الوصف

تقوم الدالة XLoadFont بتحميل الخط المحدد وإرجاع معرف الخط المرتبط به. إذا لم يكن اسم الخط موجودًا في ترميز الأحرف القياسي المضيف (Host Portable Character Encoding)، فإن النتيجة تعتمد على التنفيذ. استخدام الأحرف الكبيرة أو الصغيرة لا يهم. عند استخدام الأحرف "؟" و "*" في اسم الخط، يتم إجراء مطابقة نمط ويتم استخدام أي خط مطابق. في النمط، يطابق الحرف "?" أي حرف فردي، ويطابق الحرف "*" أي عدد من الأحرف. يتم تحديد تنسيق منظم لأسماء الخطوط في معيار اتحاد X X Logical Font Description Conventions. إذا فشلت XLoadFont في تحميل الخط المحدد، ينتج عن ذلك خطأ BadName. الخطوط غير مرتبطة بشاشة معينة ويمكن تخزينها كمكون لأي GC. عندما لا يعود الخط مطلوبًا، قم باستدعاء XUnloadFont.

يمكن لـ XLoadFont إنشاء أخطاء BadAlloc و BadName.

تقوم الدالة XQueryFont بإرجاع مؤشر إلى البنية XFontStruct، والتي تحتوي على معلومات مرتبطة بالخط. يمكنك الاستعلام عن خط أو الخط المخزن في GC. سيكون معرف الخط المخزن في البنية XFontStruct هو معرف GContext، وتحتاج إلى توخي الحذر عند استخدام هذا المعرف في دوال أخرى (انظر XGContextFromGC). إذا لم يكن الخط موجودًا، تعيد XQueryFont القيمة NULL. لتحرير هذه البيانات، استخدم XFreeFontInfo.

يمكن لـ XLoadQueryFont إنشاء خطأ BadAlloc.

توفر الدالة XLoadQueryFont الطريقة الأكثر شيوعًا للوصول إلى الخط. تقوم XLoadQueryFont بفتح (تحميل) الخط المحدد وإرجاع مؤشر إلى البنية XFontStruct المناسبة. إذا لم يكن اسم الخط في ترميز الأحرف القياسي المضيف، فإن النتيجة تعتمد على التنفيذ. إذا لم يكن الخط موجودًا، تعيد XLoadQueryFont القيمة NULL.

تقوم الدالة XFreeFont بحذف الارتباط بين معرف مورد الخط والخط المحدد وتحرير البنية XFontStruct. سيتم تحرير الخط نفسه عندما لا تشير إليه أي موارد أخرى. لا ينبغي الإشارة إلى البيانات والخط مرة أخرى.

يمكن لـ XFreeFont إنشاء خطأ BadFont.

بالنظر إلى الذرة (atom) لتلك الخاصية، تعيد الدالة XGetFontProperty قيمة خاصية الخط المحددة. تعيد XGetFontProperty أيضًا False إذا لم يتم تعريف الخاصية أو True إذا تم تعريفها. توجد مجموعة من الذرات المعرفة مسبقًا لخصائص الخطوط، والتي يمكن العثور عليها في X11/Xatom.h. تحتوي هذه المجموعة على الخصائص القياسية المرتبطة بالخط. على الرغم من أنه غير مضمون، فمن المحتمل أن تكون خصائص الخطوط المعرفة مسبقًا موجودة.

تقوم الدالة XUnloadFont بحذف الارتباط بين معرف مورد الخط والخط المحدد. سيتم تحرير الخط نفسه عندما لا تشير إليه أي موارد أخرى. لا ينبغي الإشارة إلى الخط مرة أخرى.

يمكن لـ XUnloadFont إنشاء خطأ BadFont.

الهياكل

تحتوي البنية XFontStruct على جميع المعلومات الخاصة بالخط وتتكون من معلومات خاصة بالخط بالإضافة إلى مؤشر إلى مصفوفة من هياكل XCharStruct للأحرف الموجودة في الخط. تحتوي هياكل XFontStruct و XFontProp و XCharStruct على:

typedef struct {
	short lbearing;	/* origin to left edge of raster */
	short rbearing;	/* origin to right edge of raster */
	short width;	/* advance to next char's origin */
	short ascent;	/* baseline to top edge of raster */
	short descent;	/* baseline to bottom edge of raster */
	unsigned short attributes;	/* per char flags (not predefined) */
} XCharStruct;
typedef struct {
	Atom	name;
	unsigned long card32;
} XFontProp;
typedef struct {	/* normal 16 bit characters are two bytes */
    unsigned char byte1;
    unsigned char byte2;
} XChar2b;
typedef struct {
	XExtData *ext_data;	/* hook for extension to hang data */
	Font fid;	/* Font id for this font */
	unsigned direction;	/* hint about the direction font is painted */
	unsigned min_char_or_byte2;	/* first character */
	unsigned max_char_or_byte2;	/* last character */
	unsigned min_byte1;	/* first row that exists */
	unsigned max_byte1;	/* last row that exists */
	Bool all_chars_exist;	/* flag if all characters have nonzero size */
	unsigned default_char;	/* char to print for undefined character */
	int n_properties;	/* how many properties there are */
	XFontProp *properties;	/* pointer to array of additional properties */
	XCharStruct min_bounds;	/* minimum bounds over all existing char */
	XCharStruct max_bounds;	/* maximum bounds over all existing char */
	XCharStruct *per_char;	/* first_char to last_char information */
	int ascent;	/* logical extent above baseline for spacing */
	int descent;	/* logical decent below baseline for spacing */
} XFontStruct;

يدعم X عمليات النص أحادية البايت/الحرف، وبايتين/الحرف المصفوفة، والنص ذي 16 بت. لاحظ أنه يمكن استخدام أي من هذه الأشكال مع خط، ولكن طلب النص أحادي البايت/الحرف يمكن أن يحدد بايتًا واحدًا فقط (أي، الصف الأول لخط من بايتين). يجب عليك عرض الخطوط ذات البايتين كمصفوفة ثنائية الأبعاد من الأحرف المعرفة: يحدد byte1 نطاق الصفوف المعرفة ويحدد byte2 نطاق الأعمدة المعرفة للخط. الخطوط أحادية البايت/الحرف لها صف واحد معرف، ويحدد نطاق byte2 المحدد في البنية نطاقًا من الأحرف.

يتم تعريف الصندوق المحيط بالحرف بواسطة XCharStruct الخاص بهذا الحرف. عندما تكون الأحرف غائبة عن الخط، يتم استخدام default_char. عندما تكون الخطوط جميعها بنفس الحجم، يتم استخدام المعلومات الموجودة في الحدود الدنيا والعليا للبنية XFontStruct فقط.

تتمتع أعضاء البنية XFontStruct بالدلالات التالية:

  • يمكن أن يكون العضو direction إما FontLeftToRight أو FontRightToLeft. إنه مجرد تلميح حول ما إذا كانت معظم عناصر XCharStruct لها مقياس عرض حرف موجب (FontLeftToRight) أو سالب (FontRightToLeft). البروتوكول الأساسي لا يدعم النص العمودي.
  • إذا كان العضوان min_byte1 و max_byte1 كلاهما صفرًا، فإن min_char_or_byte2 يحدد فهرس الحرف الخطي المقابل للعنصر الأول في مصفوفة per_char، و max_char_or_byte2 يحدد فهرس الحرف الخطي للعنصر الأخير.
إذا كان أي من min_byte1 أو max_byte1 غير صفري، فإن كل من min_char_or_byte2 و max_char_or_byte2 أقل من 256، وفهرس الحرف ذي البايتين المقابل لعنصر مصفوفة per_char N (عد من 0) هو:
byte1 = N/D + min_byte1
byte2 = N%D + min_char_or_byte2
حيث: 
        D = max_char_or_byte2 - min_char_or_byte2 + 1
        / = قسمة صحيحة
        % = باقي القسمة
