XftFontOpen، باستثناء أنه يأخذ سلسلة نمط Fontconfig (التي يتم تمريرها إلى مكتبة Fontconfig FcNameParse()).

XftFont *XftFontOpenXlfd (

XftFontOpenXlfd يعمل مثل XftFontOpen، باستثناء أنه يأخذ سلسلة تحتوي على وصف خط X المنطقي (XLFD)، ويستخدم الدالة XftXlfdParse().

FcPattern *XftFontMatch (

تُستخدم أيضًا داخليًا بواسطة دوال XftFontOpen*، ويمكن استخدام XftFontMatch مباشرة لتحديد نمط خط Fontconfig الناتج عن طلب فتح خط Xft.

FcPattern *XftXlfdParse (
	_Xconst char	*xlfd_orig،
	Bool	ignore_scalable،
	Bool	complete)؛

تقوم XftXlfdParse بتحليل المعامل xlfd_orig وفقًا لوثيقة X Logical Font Description Conventions، ولكنها تتجاهل بعض الحقول: setwidth_name، add_style_name، spacing، و average_width.

تقوم XftXlfdParse بإنشاء نمط Fontconfig، وتعيين الخاصية XFT_XLFD إلى قيمة xlfd_orig، وتربط المعلومات المجمعة بخصائص Fontconfig. يتم تجاهل الحقول الفارغة أو التي تحتوي على " \* ":
FC_FOUNDRY
من foundry.
FC_FAMILY
من family.
FC_WEIGHT
weight_name، مع قيمة افتراضية FC_WEIGHT_MEDIUM.
FC_SLANT
من slant، مع قيمة افتراضية FC_SLANT_ROMAN.
FC_SIZE
من point_size.
FC_PIXEL_SIZE
من pixel_size. إذا تم تعيين point_size، بالإضافة إلى resolution_x و resolution_y، فإن القيمة يتم قياسها لتحويل ارتفاع الخط إلى نقاط.

توفر Xft مجموعات من دوال الرسم التي تختلف حسب طريقة تشفير البيانات، مثل 8 بت (ISO-8859-1) مقابل UTF-8. فيما يلي ملخص لهذه الدوال ذات الصلة:

تحديد مدى النص

توفر Xft عدة دوال لتحديد الارتفاع والعرض المطلوبين لعرض سلسلة نصية. بعد حساب الإزاحة، في الحالات التي سيتم فيها تحويل السلسلة لأعلى أو لأسفل أو لليسار أو لليمين، يشار إلى هذه الأرقام باسم مدى النص.

void XftTextExtents8 (
	Display	*dpy،
	XftFont	*font،
	_Xconst FcChar8	*string،
	int	 len،
	XGlyphInfo	*extents)؛

تقوم XftTextExtents8 بحساب مدى البكسل على الشاشة dpy لعدد لا يزيد عن len حرفًا من string المكونة من أحرف ثمانية البت عند رسمها بالخط font، وتخزينها في extents.

void XftTextExtents16 (

تقوم XftTextExtents16 بحساب مدى البكسل على الشاشة dpy لعدد لا يزيد عن len حرفًا من string المكونة من أحرف ستة عشر بت عند رسمها بالخط font، وتخزينها في extents.

void XftTextExtents32 (

تقوم XftTextExtents32 بحساب مدى البكسل على الشاشة dpy لعدد لا يزيد عن len حرفًا من string المكونة من أحرف اثنان وثلاثون بت عند رسمها بالخط font، وتخزينها في extents.

void XftTextExtentsUtf8 (

تقوم XftTextExtentsUtf8 بحساب مدى البكسل على الشاشة dpy لعدد لا يزيد عن len بايت من string المشفرة بـ UTF-8 عند رسمها بالخط font، وتخزينها في extents.

void XftTextExtentsUtf16 (
	Display	*dpy،
	XftFont	*font،
	_Xconst FcChar8	*string،
	FcEndian	 endian،
	int	 len،
	XGlyphInfo	*extents)؛

تقوم XftTextExtentsUtf16 بحساب مدى البكسل على الشاشة dpy لعدد لا يزيد عن len بايت من string المشفرة بـ UTF-16LE أو UTF-16BE عند رسمها بالخط font، وتخزينها في extents. يجب تحديد نهاية ترتيب البايتات (endianness) لـ string في endian.

void XftGlyphExtents (
	Display	*dpy،
	XftFont	*font،
	_Xconst FT_UInt	*glyphs،
	int	 nglyphs،
	XGlyphInfo	*extents)؛

تُستخدم أيضًا داخليًا بواسطة دوال XftTextExtents*()، تقوم XftGlyphExtents بحساب مدى البكسل على الشاشة dpy لعدد لا يزيد عن nglyphs في المصفوفة glyphs مرسومة بالخط font، وتخزينها في extents.

إذا كانت أي من glyphs مفقودة (يتم تحديد ذلك عن طريق فحص باستخدام XftFontCheckGlyph())، فسيتم ملء الإدخال المقابل في extents بالأصفار.

إدارة XftColor

Bool XftColorAllocName (
	Display	*dpy،
	_Xconst Visual	*visual،
	Colormap	 cmap،
	_Xconst char	*name،
	XftColor	*result)؛

استخدم XAllocNamedColor() للبحث عن اللون المسمى name للشاشة المرتبطة بخريطة الألوان cmap.

  • إذا أعادت XAllocNamedColor() قيمة غير صفرية، فإن XftColorAllocName() تملأ حقل البكسل لـ XftColor الناتج بأقرب لون مدعوم على الشاشة، بالإضافة إلى الحقول الحمراء والخضراء والزرقاء الدقيقة من قاعدة البيانات، وتعيد True.
  • إذا أعادت XAllocNamedColor() قيمة صفر، فإن XftColorAllocName() تعيد False، ولا تقوم بتحديث XftColor المشار إليه بواسطة result.
لم يتم استخدام المعامل visual.
Bool XftColorAllocValue (
	Display	*dpy،
	Visual	*visual،
	Colormap	 cmap،
	_Xconst XRenderColor	*color،
	XftColor	*result)؛

تخصيص قيمة لون:

  • إذا كان فئة visual هي TrueColor، فإن XftColorAllocValue() تضبط حقل pixel في XftColor المشار إليه بواسطة result باستخدام الحقول الحمراء والخضراء والزرقاء من المعامل color.
  • إذا كانت فئة visual ليست TrueColor، فإن XftColorAllocValue() تستدعي XAllocColor() لتخصيص إدخال في خريطة الألوان cmap. التي تعيد قيمة البكسل لأقرب لون للعناصر RGB المحددة التي تدعمها الأجهزة.
إذا نجحت XAllocColor()، فإن XftColorAllocValue() تخزن قيمة البكسل تلك في XRenderColor المشار إليه بواسطة result وتعيد True.
إذا فشلت XAllocColor()، فإن XftColorAllocValue() تعيد False ولا تعدل النتيجة.
void XftColorFree (
	Display	*dpy،
	Visual	*visual،
	Colormap	 cmap،
	XftColor	*color)؛

إذا لم تكن فئة visual هي TrueColor، فإن Xft تستدعي XFreeColors() لتحرير الإدخال من خريطة الألوان cmap التي تم تخصيص قيمة البكسل الخاصة بها في المعامل color بواسطة XftColorAllocName().

	XftDraw	*draw);

XftDrawPicture يعيد الصورة المرتبطة بكائن الرسم Xft draw.

إذا لم يدعم خادم X امتداد X Rendering، فسيتم إرجاع 0.
Picture XftDrawSrcPicture (
	XftDraw	*draw,
	_Xconst XftColor	*color);

إعادة كائن صورة X Render، والذي يستخدم لرسم الحروف، على سبيل المثال، مع XftGlyphRender() أو XftGlyphSpecRender() أو XftGlyphFontSpecRender()، بواسطة XftDrawGlyphs() أو XftDrawGlyphSpec() أو XftDrawGlyphFontSpec()، على التوالي.

إذا لم يدعم خادم X امتداد X Render، فإن هذه الوظائف تستخدم XftGlyphCore() أو XftGlyphSpecCore() أو XftGlyphFontSpecCore().
void XftDrawDestroy (
	XftDraw	*draw);

XftDrawDestroy يدمر draw (الذي تم إنشاؤه بواسطة إحدى وظائف XftDrawCreate*()) ويحرر الذاكرة التي تم تخصيصها له.

Bool XftDrawSetClip (
	XftDraw	*draw,
	Region	 r);

إعداد الاقتصاص لمعلمة XftDraw المعطاة draw بدءًا من Region:

  • إذا كانت معلمة Region r ليست فارغة، فإن Xft تنشئ Region جديدة (لنسخ المعلمة)،
  • Xft تدمر أي منطقة اقتصاص موجودة.
  • Xft تقوم بتعيين clip_type للمعلمة draw إلى XftClipTypeRegion إذا لم تكن المعلمة r فارغة. بخلاف ذلك، فإنها تقوم بتعيين clip_type إلى XftClipTypeNone.
  • أخيرًا، يقوم Xft بتحديث الاقتصاص للكائنات الموجودة، وتحديث clip_mask لكائن Picture الخاص بـ X Render، وتعيين قناع الاقتصاص في سياق الرسومات (GC) المرتبط بمعلمة XftDraw.
XftDrawSetClip() يعيد True إذا لم يكن هناك تغيير ضروري، أو إذا نجحت العملية. يعيد False إذا لم يتمكن من إنشاء Region() جديدة.
Bool XftDrawSetClipRectangles (
	XftDraw	*draw,
	int	 xOrigin,
	int	 yOrigin,
	_Xconst XRectangle	*rects,
	int	 n);

مثل XftDrawSetClip()، يقوم XftDrawSetClipRectangles() بإعداد الاقتصاص لمعلمة XftDraw المعطاة draw ولكن باستخدام مجموعة من n المستطيلات (المعلمة rects) التي يمكن استخدامها لبناء Region.

يقوم Xft بتعيين clip_type لـ draw إلى XftClipTypeRectangles ويستخدم XSetClipRectangles() للاقتصاص الأساسي (X11) و XRenderSetPictureClipRectangles() للاقتصاص X Render.
void XftDrawSetSubwindowMode (
	XftDraw	*draw,
	int	 mode);

يقوم بتعيين وضع النافذة الفرعية لمعلمة XftDraw المعطاة draw. يمكن أن يكون الوضع إما ClipByChildren (الافتراضي) أو IncludeInferiors:

  • بالنسبة لـ ClipByChildren، يتم اقتصاص كل من نوافذ المصدر والوجهة إضافيًا بواسطة جميع الأطفال المرئيين من نوع InputOutput.
  • بالنسبة لـ IncludeInferiors، لا يتم اقتصاص أي من نافذة المصدر أو الوجهة بواسطة العناصر التابعة. سيؤدي هذا إلى تضمين محتويات النوافذ الفرعية في المصدر والرسم عبر حدود النوافذ الفرعية للوجهة.
بالإضافة إلى وضع النافذة الفرعية الذي تحتفظ به Xft، فإنه يقوم بتحديث وضع النافذة الفرعية لأي سياق رسومات (GC) مرتبط باستخدام XSetSubwindowMode() وكذلك لصورة X Render باستخدام XRenderChangePicture().

رسم السلاسل النصية

void XftDrawString8 (
	XftDraw	*d,
	_Xconst XftColor	*color,
	XftFont	*font,
	int	 x,
	int	 y,
	_Xconst FcChar8	*string,
	int	 len);

XftDrawString8 يرسم ما لا يزيد عن len حرفًا من string إلى عنصر الرسم Xft d باستخدام font باللون color في الموضع x, y.

رسم أشياء أخرى

void XftDrawRect (
	XftDraw	*d,
	_Xconst XftColor	*color,
	int	 x,
	int	 y,
	unsigned int	 width,
	unsigned int	 height);

XftDrawRect يرسم مستطيلًا صلبًا باللون color والعرض width والارتفاع height المحددين في الموضع x, y إلى عنصر الرسم Xft d.

void XftCharFontSpecRender (
	Display	*dpy,
	int	 op,
	Picture	 src,
	Picture	 dst,
	int	 srcx,
	int	 srcy,
	_Xconst XftCharFontSpec	*chars,
	int	 len);

XftCharFontSpecRender() يحول المعلمة chars من XftCharFontSpec إلى XftGlyphFontSpec، ويمرر المصفوفة المحولة مع المعلمات الأخرى دون تغيير إلى XftGlyphFontSpecRender() لرسم البيانات.

void XftDrawGlyphFontSpec (
	XftDraw	*draw,
	_Xconst XftColor	*color,
	_Xconst XftGlyphFontSpec	*glyphs,
	int	 len);

رسم قائمة من الحروف المرتبطة بالخطوط في إحداثيات محددة، يتم تمريرها كمصفوفة من len هياكل XftGlyphFontSpec عبر المعلمة glyphs. يتم رسم جميع الحروف باستخدام اللون المحدد في المعلمة color.

لكل إدخال في مصفوفة glyphs:
  • إذا كان الخط المرتبط يستخدم امتداد X Render، فإن Xft يستخدم XftGlyphFontSpecRender() لرسم الحرف، باستخدام Picture تم الحصول عليه من استدعاء XftDrawSrcPicture () مع معلمتي draw و color.
  • خلاف ذلك، توفر Xft ميزة مماثلة باستخدام Xlib.
void XftGlyphFontSpecRender (
	Display	*dpy,
	int	 op,
	Picture	 src,
	Picture	 dst,
	int	 srcx,
	int	 srcy,
	_Xconst XftGlyphFontSpec	*glyphs,
	int	 nglyphs);

يتضمن هذا عدة خطوات:

  • أولاً، يتأكد Xft من أن مصفوفة glyphs كاملة باستخدام XftFontCheckGlyph() وتحميل أي حروف مفقودة باستخدام XftFontLoadGlyphs().
  • ثم يفحص Xft الحروف للعثور على أقصى قيمة Unicode. هذا يحدد حجم الترميز، أي width.
  • بعد ذلك، لكل حرف، يتحقق Xft مما إذا كانت قيمة Unicode الخاصة به تتم معالجتها بواسطة الخط المقابل. إذا لم يكن الأمر كذلك، يتم تعيين القيمة إلى صفر (0)، لتجاهلها. أثناء القيام بذلك، يتذكر Xft أيضًا أي خط تم استخدامه أولاً، ويحسب الموضع الذي سيتم فيه رسم كل حرف.
  • ثم يبني Xft مصفوفة مكافئة من الحروف بالتنسيق المتوقع من قبل مكتبة X Render.
  • أخيرًا، يستخدم XftGlyphFontSpecRender() امتداد X Render لرسم الحروف، باستخدام وظيفة XRenderCompositeText*() المناسبة وفقًا للـ width، ويتخلص من البيانات المؤقتة.
تستخدم المعلمات op و dst و src و srcx و srcy كما هو موضح في وثائق مكتبة X Render.

معالجة بيانات الخطوط

void XftFontLoadGlyphs (
	Display	*dpy,
	XftFont	*pub,
	FcBool	 need_bitmaps,
	_Xconst FT_UInt	*glyphs,
	int	 nglyph);

باستخدام FT_Load_Glyph()، يتم تحميل nglyphs لمؤشرات الأحرف (قيم Unicode) المدرجة في المصفوفة glyphs من الخط pub المحدد.

يتضمن تحميل الحرف أكثر من مجرد قراءة البيانات في الذاكرة. تستخدم Xft نمط Fontconfig المرتبط بالعرض dpy (على سبيل المثال، عبر XftDefaultSet()) لتحديد ما إذا كان سيتم استخدام مربع حدودي، أو قص الأحرف ضمن تلك الحدود، أو قياس الأحرف، أو حساب مقاييس الخط، وإضافتها إلى امتداد X Render باستخدام XRenderAddGlyphs().
void XftFontUnloadGlyphs (
	Display	*dpy,
	XftFont	*pub,
	_Xconst FT_UInt	*glyphs,
	int	 nglyph);

يتم تجاهل البيانات لما يصل إلى nglyph من الأحرف التي تتضمن مؤشرات الأحرف (قيم Unicode) المدرجة في المصفوفة glyphs للخط pub المحدد. إذا تمت إضافة الأحرف إلى امتداد X Render، فإن Xft تزيلها باستخدام XRenderFreeGlyphs(). تتتبع Xft كمية الذاكرة المستخدمة للأحرف، وتقوم بتحديث الاستخدام المرتبط بالعرض dpy.

FT_Face XftLockFace (
	XftFont	*pub);

إذا لم يتم إنشاء أي "وجه" (تمثيل في الذاكرة لنوع خط معين بنمط معين) من FreeType 2 للخط pub، فقم بإنشاء واحد باستخدام FT_New_Face(). تُستخدم أقفال الوجوه كعدد مرجعي تستخدمه Xft لضمان إنشاء وجه واحد فقط، والاحتفاظ به حتى لا يعد الخط مستخدمًا.

يتم استخدام أقفال الوجوه مباشرة في XftCharIndex() و XftFontLoadGlyphs()، والتي بدورها تستخدم في العديد من وظائف Xft.
تم تقديم أقفال الوجوه في الإصدار 1 (أكتوبر 2002). اعتمدت بعض التطبيقات، مثل Gdk/Gimp، على هذه الوظائف. في الإصدار 2.1.9 (يونيو 2006)، تم الاحتفاظ بأقفال الوجوه كجزء من واجهة برمجة التطبيقات العامة عند توفير تكوينات مكتبات مشاركة محسنة لإخفاء الرموز الخاصة.

	XftFont	*pub);

يقلل عدد المراجع لـ "وجه" FreeType 2 المرتبط بالخط.

تتم إزالة كائنات FT_Face باستخدام FT_Done_Face(). تقوم Xft بذلك في XftFontInfoDestroy() وعند التنظيف عند الفشل في XftFontInfoCreate() و XftFontOpenPattern().

التوافق

اعتبارًا من الإصدار 2 (مايو 2002)، أصبحت Xft مستقرة نسبيًا. من المتوقع أن تحتفظ بالتوافق المصدر والثنائي في الإصدارات المستقبلية.

توفر Xft واجهة توافق مع إصدارها الرئيسي السابق، Xft 1.x، الموصوفة أدناه.

ملف رأس Xft 1.x

#include <X11/Xft/XftCompat.h>

أنواع بيانات Xft 1.x

XftPattern
تحتوي على مجموعة من الأسماء مع قوائم قيم مرتبطة؛ كل اسم يشير إلى خاصية لخط. تستخدم XftPatterns كمدخلات لرمز المطابقة وكذلك لتخزين المعلومات حول خطوط محددة.
XftFontSet
تحتوي على قائمة بـ XftPatterns. داخليًا، تستخدم Xft هذا الهيكل البياني لتخزين مجموعات من الخطوط. خارجيًا، تعيد Xft نتائج قائمة الخطوط بهذا التنسيق.
XftObjectSet
تحتوي على مجموعة من الأسماء وتستخدم لتحديد الحقول التي يتم استخراجها من الخطوط ووضعها في قائمة الأنماط التي تم إرجاعها عند قائمة الخطوط.
