XftTextRender16LE(3)

الحزمة: libxft

التراخيص: HPND-sell-variant

Xft(3) إصدار X 11 Xft(3)

الاسم

Xft - مكتبة واجهة X FreeType

الملخص

#include <X11/Xft/Xft.h>

الوصف

Xft هي مكتبة بسيطة تقوم برسم النصوص والرسومات:

  • باستخدام المعلومات المقدمة من مكتبة Fontconfig،
  • تقوم Xft بتحويل حروف الخطوط باستخدام مُحَوِّل FreeType، و
  • تعرض بيانات الخط المحولة باستخدام امتداد X Rendering.

هذه الصفحة اليدوية تكاد لا تخدش سطح هذه المكتبة.

الثوابت

XFT_MAJOR
هو رقم الإصدار الرئيسي لـ Xft.
XFT_MINOR
هو رقم الإصدار الثانوي لـ Xft.
XFT_REVISION
هو رقم المراجعة لـ Xft.
XFT_VERSION
هو XFT_MAJOR مضروبًا في 10000 (عشرة آلاف)، زائد XFT_MINOR مضروبًا في 100، زائد XFT_REVISION.
XftVersion
هو اسم مستعار لـ XFT_VERSION.

يوضح المثال التالي كيف يمكن استخدام ثوابت الإصدار الخاصة بـ Xft: 

    #if (XFT_VERSION >= 20107)
    (void) puts("الإصدار 2.1.7 أو أحدث من مكتبة Xft قيد الاستخدام.");
    #else
    (void) printf("إصدار غير كافٍ من Xft (%d.%d.%d) مثبت؛
                  " تحتاج إلى إصدار 2.1.7 على الأقل.\n", XFT_MAJOR,
                  XFT_MINOR,
                  XFT_REVISION);
    #endif

أنواع البيانات

تتبع أنواع بيانات Xft اصطلاح تسمية، حيث تسبق جميع الأسماء بـ "Xft"

من Xlib...

لا تتبع أنواع بيانات Xlib اصطلاح تسمية. يتم توثيقها في Xlib - واجهة لغة C.

تستخدم Xft هذه الأسماء: Bool، Colormap، Display، Drawable، Pixmap، Region، Visual، و XRectangle.

بعض أنواع البيانات ذات أهمية خاصة لـ Xft:

Drawable
معرّف مورد X، على سبيل المثال، نافذة مرتبطة بالشاشة. يمكن استخدام كل من النوافذ والصور (pixmaps) كمصادر ووجهات في عمليات الرسوم. تُعرف هذه النوافذ والصور مجتمعة باسم drawables. ومع ذلك، لا يمكن استخدام نافذة InputOnly كمصدر أو وجهة في عملية رسومات.
Region
مجموعة اعتباطية من مواقع البكسل التي يمكن لـ Xlib معالجتها. داخليًا، يتم تمثيل Region باتحاد عدد اعتباطي من المستطيلات. تحتفظ Xlib بمستطيل يخبرها بمدى هذا الاتحاد.
Visual
يحتوي هذا الهيكل على معلومات حول تعيين الألوان الممكن.

من XRender...

مستوحاة أنواع بيانات مكتبة X Render Extension من Xlib، ولكنها تفتقر إلى الوثائق المقابلة. راجع ملف الرأس للتفاصيل:

<X11/extensions/Xrender.h>

أو اقرأ الكود المصدري (الذي يوفر مخططًا عامًا لـ libXrender).

تستخدم Xft هذه الأسماء: Picture، XGlyphInfo، XRenderColor.

من Fontconfig...

تتبع أنواع بيانات Fontconfig اصطلاح تسمية، حيث تسبق جميع الأسماء بـ "Fc"، والتي يتم توثيقها في مرجع مطوري Fontconfig.

تستخدم Xft أسماء Fontconfig هذه: FcBool، FcChar8، FcChar16، FcChar32، FcCharSet، FcEndian، FcFontSet، FcPattern، و FcResult.

من FreeType 2...

تتبع أنواع بيانات FreeType 2 اصطلاح تسمية، حيث تسبق جميع الأسماء بـ "FT_"، والتي يتم توثيقها في مرجع واجهة برمجة تطبيقات FreeType.

تستخدم Xft هذه الأسماء: FT_Face، FT_Library، و FT_UInt.

يستخدم FreeType 2 FT_UInt (عدد صحيح غير موقع) لتمثيل الحروف.

XftFont


typedef struct _XftFont {
	int	 ascent;
	int	 descent;
	int	 height;
	int	 max_advance_width;
	FcCharSet	*charset;
	FcPattern	*pattern;
} XftFont;

XftFont هو هيكل البيانات الرئيسي الذي يهم المبرمجين الذين يستخدمون Xft؛ فهو يحتوي على مقاييس الخط العامة ومؤشرات إلى مجموعة الأحرف ونمط Fontconfig المرتبط بالخط.

يتم ملء XftFonts باستخدام أي من XftFontOpen()، XftFontOpenName()، XftFontOpenXlfd()، XftFontOpenInfo()، أو XftFontOpenPattern(). يتم استخدام XftFontCopy() لنسخ XftFonts، ويتم استخدام XftFontClose() لتمييز XftFont على أنه غير مستخدم. يتم تخصيص XftFonts داخليًا، وحساب مراجعها، وتحريرها بواسطة Xft؛ لا يحتاج المبرمج عادةً إلى تخصيص أو تحرير مساحة تخزين لها.

تستخدم XftDrawGlyphs()، وعائلة XftDrawString*()، XftDrawCharSpec()، و XftDrawGlyphSpec() XftFonts لرسم النص إلى كائن XftDraw، والذي قد يتوافق مع كل من قابل للرسم أساسي في X أو قابل للرسم لامتداد X Rendering.

تُستخدم XftGlyphExtents() وعائلة XftTextExtents*() لتحديد امتدادات (الأبعاد القصوى) لـ XftFont.

يمكن تحديد تغطية الحروف أو الحروف الخاصة بـ XftFont باستخدام XftFontCheckGlyph() أو XftCharExists(). XftCharIndex() تُرجع الفهرس المحدد للحرف الخاص بـ XftFont المقابل لنقطة ترميز Unicode معطاة.

تستخدم XftGlyphRender()، XftGlyphSpecRender()، XftCharSpecRender()، وعائلة XftTextRender*() XftFonts للرسم في هياكل Picture الخاصة بـ X Rendering Extension.

ملاحظة: توفر XftDrawGlyphs()، وعائلة XftDrawString*()، XftDrawCharSpec()، و XftDrawGlyphSpec() وسيلة لرسم الخطوط مستقلة عن توفر امتداد X Rendering على خادم X.

XftFontInfo

هو كائن غير شفاف يخزن معلومات حول خط. يتم إنشاء هياكل XftFontInfo باستخدام XftFontInfoCreate()، وتحريرها باستخدام XftFontInfoDestroy()، ومقارنتها باستخدام XftFontInfoEqual(). يتم تخصيص كائنات XftFontInfo داخليًا وتحريرها بواسطة Xft؛ لا يحتاج المبرمج عادةً إلى تخصيص أو تحرير مساحة تخزين لها.

يرتبط كل هيكل XftFontInfo قيد الاستخدام بمعرف فريد، يمكن استرجاعه باستخدام XftFontInfoHash(). يمكن فتح XftFont بناءً على بيانات XftFontInfo باستخدام XftFontOpenInfo().

XftColor


typedef struct _XftColor {
	unsigned long	 pixel;
	XRenderColor	 color;
} XftColor;

يسمح كائن XftColor برسم النصوص والعناصر الأخرى بلون معين (أو أقرب تقريب يقدمه الـ X visual قيد الاستخدام).

تقوم XftColorAllocName() و XftColorAllocValue() بطلب تخصيص لون من خادم X (إذا لزم الأمر) وتهيئة أعضاء XftColor. تُعلم XftColorFree() خادم X بتحرير اللون المخصص حاليًا لـ XftColor.

بمجرد تهيئة XftColor، يمكن استخدام XftDrawSrcPicture()، XftDrawGlyphs()، عائلة XftDrawString*()، XftDrawCharSpec()، XftDrawCharFontSpec()، XftDrawGlyphSpec()، XftDrawGlyphFontSpec()، و XftDrawRect() لرسم عناصر مختلفة باستخدام

هذا الموقع غير مرتبط بـ Arch Linux ولا تابع له بأي شكل من الأشكال. المحتوى هو ترجمة فقط لصفحات دليل Arch المنشورة على man.archlinux.org.

XftDraw

هو كائن غير شفاف يحتفظ بمعلومات تُستخدم للرسم على قابل للرسم (drawable) في X باستخدام بروتوكول النواة أو امتداد X Rendering.

يتم إنشاء كائنات XftDraw باستخدام أي من الدوال التالية: XftDrawCreate() (التي تربط XftDraw بقابل رسم X موجود)، XftDrawCreateBitmap()، أو XftDrawCreateAlpha()، ويتم تدميرها باستخدام XftDrawDestroy(). يمكن تغيير قابل رسم X المرتبط بـ XftDraw باستخدام XftDrawChange(). يتم تخصيص وتحرير كائنات XftDraw داخليًا بواسطة Xft؛ لا يحتاج المبرمج عادةً إلى تخصيص أو تحرير مساحة تخزين لها.

يمكن الاستعلام عن خصائص X Display، Drawable، Colormap، و Visual لكائن XftDraw باستخدام XftDrawDisplay()، XftDrawDrawable()، XftDrawColormap()، و XftDrawVisual() على التوالي.

تستخدم العديد من الدوال كائنات XftDraw: XftDrawCharFontSpec()، XftDrawCharSpec()، XftDrawGlyphFontSpec()، XftDrawGlyphSpec()، XftDrawGlyphs()، XftDrawRect()، XftDrawSetClip()، XftDrawSetClipRectangles()، XftDrawSetSubwindowMode()، وعائلة XftDrawString*().

يتم إرجاع Picture امتداد X Rendering المرتبط بـ XftDraw بواسطة XftDrawPicture() و XftDrawSrcPicture(). يستخدم في XftCharFontSpecRender()، XftCharSpecRender()، XftGlyphFontSpecRender()، XftGlyphRender()، XftGlyphSpecRender()، وعائلة XftTextRender*().

الوظائف

التهيئة

التطبيق النموذجي الذي يستخدم Xft لا يقوم بتهيئة المكتبة بشكل صريح. يتم ذلك عادةً كنتيجة جانبية لفتح خط.

عندما تهيئة Xft، تجمع معلومات حول الشاشة، وتخزن بعض تلك المعلومات في نمط Fontconfig (وهو أساسًا مجموعة من الخصائص ذات القيم المكتوبة. يمكن للتطبيق المتصل تعديل هذا النمط لتغيير سلوك المكتبة.

لا يتم تخزين كل المعلومات المجمعة في نمط. يتم تخزين الباقي في هياكل بيانات داخلية. تجعل Xft بعض هذه المعلومات متاحة للتطبيق عبر وظائف.

Bool XftDefaultHasRender (
	Display	*dpy);

الحصول على معلومات حول الشاشة dpy إذا لم يتم ذلك بالفعل، وإرجاع صحيح إذا وجدت Xft أن الشاشة تدعم امتداد X Render، وإذا كان بإمكانها العثور على XRenderPictFormat مناسب (نوع بيانات X Render الذي يعادل Visual في Xlib) على الشاشة.

Bool XftDefaultSet (
	Display	*dpy,
	FcPattern	*defaults);

الحصول على معلومات حول الشاشة dpy إذا لم يتم ذلك بالفعل، وتعيين نمط Fontconfig الذي يحتوي على الخصائص الافتراضية التي ستستخدمها Xft لهذه الشاشة.

تستخدم Xft هذه الخصائص مبدئيًا للحصول على هذه الحدود:
XFT_MAX_GLYPH_MEMORY
(maxglyphmemory). هذا هو الحد الأقصى لذاكرة الأحرف لجميع الخطوط التي تستخدمها Xft (الافتراضي: 4 * 1024 * 1024).
XFT_MAX_UNREF_FONTS
(maxunreffonts). هذا هو الحد الأقصى لعدد الخطوط غير المشار إليها (الافتراضي: 16).
XFT_TRACK_MEM_USAGE
(trackmemusage). عندما يكون صحيحًا، تتعقب Xft استخدام ذاكرة الأحرف لتحسين الأداء عند تحديد أي منها سيتم إلغاء تحميله عند الوصول إلى الحد الأقصى لذاكرة الأحرف (الافتراضي: خطأ).
تستخدم Xft أيضًا هذه الخصائص الافتراضية في XftDefaultSubstitute().
void XftDefaultSubstitute (
	Display	*dpy,
	int	 screen,
	FcPattern	*pattern);

تقوم Xft بملء الخصائص المفقودة في pattern المحدد باستخدام الخصائص الافتراضية للشاشة المحددة dpy، على سبيل المثال، كما تم تعيينها في XftDefaultSet().

تستخدم تطبيقات Xft النموذجية هذه الدالة للمساعدة في اختيار Fontconfig لخط مناسب. يتم استبدال هذه الخصائص قبل استدعاء FcDefaultSubstitute():
FC_ANTIALIAS
صحيح إذا كان يجب على FreeType استخدام تمويه الحواف (الافتراضي: خطأ). (الافتراضي: صحيح).
FC_AUTOHINT
صحيح إذا كان يجب على FreeType استخدام التلميح التلقائي (الافتراضي: خطأ).
FC_DPI
النقاط/البوصة المستخدمة للدقة (الافتراضي: محسوب من ارتفاع الشاشة).
FC_EMBOLDEN
صحيح إذا كان يجب استخدام FT_GlyphSlot_Embolden() لجعل الخط أثقل (الافتراضي: خطأ).
FC_HINTING
صحيح إذا كان يجب استخدام التلميح عند ملء الخصائص لفتح خط (الافتراضي: صحيح).
FC_HINT_STYLE
أسلوب التلميح المستخدم عند ملء الخصائص لفتح خط (الافتراضي: FC_HINT_FULL).
FC_LCD_FILTER
المعلمة التي يتم تمريرها إلى FT_Library_SetLcdFilter() عند تحميل الأحرف (الافتراضي: FC_LCD_DEFAULT).
FC_MINSPACE
الحد الأدنى لمساحة الخط المستخدمة عند ملء الخصائص لفتح خط (الافتراضي: خطأ).
FC_RGBA
قيمة RGBA المستخدمة عند ملء الخصائص لفتح خط (الافتراضي: محسوب عن طريق استدعاء XRenderQuerySubpixelOrder()).
FC_SCALE
المقياس المستخدم في Fontconfig (الافتراضي: 1.0).
XFT_MAX_GLYPH_MEMORY
الحد الأقصى للذاكرة لخط واحد (الافتراضي: 1024 * 1024).
XFT_RENDER
صحيح إذا كانت الشاشة تدعم امتداد X Render (الافتراضي: نتيجة من XftDefaultHasRender()).
FcBool XftInit (

المعلمة config غير مستخدمة.
Xft لا تقوم بإلغاء تهيئة مكتبة Fontconfig عند الانتهاء.

XftFont *XftFontOpen (

  

    
	Display	*dpy,
	int	 screen,
	...);


    
تأخذ XftFontOpen قائمة من ثلاثيات عناصر النمط بالشكل field، type، value (تنتهي بـ NULL)، تطابق هذا النمط مع الخطوط المتاحة، وتفتح الخط المتطابق، وتضبط حجمه بشكل صحيح للشاشة رقم screen على الشاشة dpy. ترجع الخط المتطابق، أو NULL إذا لم يتم العثور على تطابق.

  




مثال:
  




    font = XftFontOpen (dpy, screen,
                        XFT_FAMILY, XftTypeString, "charter",
                        XFT_SIZE, XftTypeDouble, 12.0,
                        NULL);




  

  
يفتح هذا خط “charter” بحجم 12 نقطة. يتم تحويل حجم النقطة تلقائيًا إلى حجم البكسل الصحيح بناءً على دقة الشاشة.
