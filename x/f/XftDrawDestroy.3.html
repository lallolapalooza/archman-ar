XftDrawDestroy(3)

الحزمة: libxft

التراخيص: HPND-sell-variant

Xft(3) إصدار X 11 Xft(3)

الاسم

Xft - مكتبة واجهة X FreeType

الملخص

#include <X11/Xft/Xft.h>

الوصف

Xft هي مكتبة بسيطة تقوم برسم النصوص والرسومات:

  • باستخدام المعلومات المقدمة من مكتبة Fontconfig،
  • تقوم Xft بتحويل رموز الخطوط باستخدام مُحرك FreeType، و
  • تعرض بيانات الخط المحولة باستخدام امتداد X للتقديم.

هذه الصفحة الدليلية بالكاد تلامس سطح هذه المكتبة.

الثوابت

XFT_MAJOR
هو رقم الإصدار الرئيسي لـ Xft.
XFT_MINOR
هو رقم الإصدار الثانوي لـ Xft.
XFT_REVISION
هو رقم المراجعة لـ Xft.
XFT_VERSION
هو XFT_MAJOR مضروبًا في 10000 (عشرة آلاف)، زائد XFT_MINOR مضروبًا في 100، زائد XFT_REVISION.
XftVersion
هو اسم مستعار لـ XFT_VERSION.

يوضح المثال التالي كيف يمكن استخدام ثوابت إصدار Xft: 

    #if (XFT_VERSION >= 20107)
    (void) puts("الإصدار 2.1.7 أو أحدث من مكتبة Xft قيد الاستخدام.");
    #else
    (void) printf("إصدار غير كافٍ من Xft (%d.%d.%d) مثبت؛
                  " تحتاج إلى إصدار 2.1.7 على الأقل.\n", XFT_MAJOR,
                  XFT_MINOR,
                  XFT_REVISION);
    #endif

أنواع البيانات

تتبع أنواع بيانات Xft اصطلاح تسمية، حيث يتم تسمية جميع الأسماء مسبوقة بـ "Xft"

من Xlib...

لا تتبع أنواع بيانات Xlib اصطلاح تسمية. يتم توثيقها في Xlib - واجهة لغة C.

تستخدم Xft هذه الأسماء: Bool، Colormap، Display، Drawable، Pixmap، Region، Visual، و XRectangle.

بعض أنواع البيانات ذات أهمية خاصة لـ Xft:

Drawable
معرّف مورد X، على سبيل المثال، نافذة مرتبطة بالعرض. يمكن استخدام كل من النوافذ والبيكس مابس (pixmaps) كمصادر ووجهات في عمليات الرسومات. تُعرف هذه النوافذ والبيكس مابس بشكل جماعي باسم "drawables". ومع ذلك، لا يمكن استخدام نافذة InputOnly كمصدر أو وجهة في عملية رسومات.
Region
مجموعة عشوائية من مواقع البكسل التي يمكن لـ Xlib معالجتها. داخليًا، يتم تمثيل Region باتحاد عدد عشوائي من المستطيلات. تحتفظ Xlib بمستطيل يحدد امتداد هذا الاتحاد.
Visual
يحتوي هذا الهيكل على معلومات حول تعيين الألوان الممكنة.

من XRender...

أنواع بيانات مكتبة امتداد X للتقديم مستوحاة من Xlib، لكنها تفتقر إلى التوثيق المقابل. ارجع إلى ملف الرأس للتفاصيل:

<X11/extensions/Xrender.h>

أو اقرأ الكود المصدري (الذي يوفر مخططًا لمكتبة libXrender).

تستخدم Xft هذه الأسماء: Picture، XGlyphInfo، XRenderColor.

من Fontconfig...

تتبع أنواع بيانات Fontconfig اصطلاح تسمية، حيث يتم تسمية جميع الأسماء مسبوقة بـ "Fc" وهي موثقة في مرجع مطوري Fontconfig.

تستخدم Xft هذه الأسماء من Fontconfig: FcBool، FcChar8، FcChar16، FcChar32، FcCharSet، FcEndian، FcFontSet، FcPattern، و FcResult.

من FreeType 2...

تتبع أنواع بيانات FreeType 2 اصطلاح تسمية، حيث يتم تسمية جميع الأسماء مسبوقة بـ "FT_" وهي موثقة في مرجع واجهة برمجة تطبيقات FreeType.

تستخدم Xft هذه الأسماء: FT_Face، FT_Library، و FT_UInt.

يستخدم FreeType 2 FT_UInt (عدد صحيح غير موقع) لتمثيل *الرموز (glyphs)*.

XftFont


typedef struct _XftFont {
	int	 ascent;
	int	 descent;
	int	 height;
	int	 max_advance_width;
	FcCharSet	*charset;
	FcPattern	*pattern;
} XftFont;

XftFont هو هيكل البيانات الأساسي الذي يهم المبرمجين الذين يستخدمون Xft؛ فهو يحتوي على مقاييس الخط العامة ومؤشرات إلى مجموعة أحرف Fontconfig والنمط المرتبط بالخط.

يتم ملء XftFonts باستخدام أي من XftFontOpen()، XftFontOpenName()، XftFontOpenXlfd()، XftFontOpenInfo()، أو XftFontOpenPattern(). يتم استخدام XftFontCopy() لتكرار XftFonts، ويتم استخدام XftFontClose() لتمييز XftFont على أنه غير مستخدم. يتم تخصيص XftFonts وإدارتها بواسطة عداد المراجع وتحريرها داخليًا بواسطة Xft؛ لا يحتاج المبرمج عادةً إلى تخصيص أو تحرير مساحة لها.

تستخدم XftDrawGlyphs()، وعائلة XftDrawString*()، و XftDrawCharSpec()، و XftDrawGlyphSpec() XftFonts لرسم النصوص إلى كائن XftDraw، والذي قد يتوافق مع رسم X أساسي أو رسم امتداد X للتقديم.

تُستخدم XftGlyphExtents() وعائلة XftTextExtents*() لتحديد امتدادات (الأبعاد القصوى) لـ XftFont.

يمكن تحديد تغطية الرموز (glyphs) أو الأحرف لـ XftFont باستخدام XftFontCheckGlyph() أو XftCharExists(). ترجع XftCharIndex() فهرس الحرف الخاص بـ XftFont المقابل لنقطة رمز Unicode معينة.

تستخدم XftGlyphRender()، و XftGlyphSpecRender()، و XftCharSpecRender()، وعائلة XftTextRender*() XftFonts للرسم في هياكل Picture لامتداد X للتقديم.

ملاحظة: توفر XftDrawGlyphs()، وعائلة XftDrawString*()، و XftDrawCharSpec()، و XftDrawGlyphSpec() وسيلة لرسم الخطوط مستقلة عن توفر امتداد X للتقديم على خادم X.

XftFontInfo

هو كائن غير شفاف يخزن معلومات حول خط. يتم إنشاء هياكل XftFontInfo باستخدام XftFontInfoCreate()، وتحريرها باستخدام XftFontInfoDestroy()، ومقارنتها باستخدام XftFontInfoEqual(). يتم تخصيص كائنات XftFontInfo وتحريرها داخليًا بواسطة Xft؛ لا يحتاج المبرمج عادةً إلى تخصيص أو تحرير مساحة لها.

يرتبط كل هيكل XftFontInfo قيد الاستخدام بمعرف فريد، والذي يمكن استعادته باستخدام XftFontInfoHash(). يمكن فتح XftFont بناءً على بيانات XftFontInfo باستخدام XftFontOpenInfo().

XftColor


typedef struct _XftColor {
	unsigned long	 pixel;
	XRenderColor	 color;
} XftColor;

يسمح كائن XftColor برسم النصوص والعناصر الأخرى بلون معين (أو أقرب تقريب يقدمه العرض المرئي لـ X قيد الاستخدام).

تقوم XftColorAllocName() و XftColorAllocValue() بطلب تخصيص لون من خادم X (إذا لزم الأمر) وتهيئة أعضاء XftColor. تُعلم XftColorFree() خادم X بتحرير اللون المخصص حاليًا لـ XftColor.

بمجرد تهيئة XftColor، يمكن استخدام XftDrawSrcPicture()، و XftDrawGlyphs()، وعائلة XftDrawString*()، و XftDrawCharSpec()، و XftDrawCharFontSpec()، و XftDrawGlyphSpec()، و XftDrawGlyphFontSpec()، و XftDrawRect() لرسم كائنات مختلفة باستخدامها.

هذا الموقع غير مرتبط بـ Arch Linux ولا تابع له بأي شكل من الأشكال. المحتوى هو ترجمة فقط لصفحات دليل Arch المنشورة على man.archlinux.org.

>XftFontOpen، باستثناء أنه يأخذ سلسلة نمط Fontconfig (التي يتم تمريرها إلى مكتبة Fontconfig FcNameParse()).

XftFont *XftFontOpenXlfd (
	Display	*dpy,
	int	 screen,
	_Xconst char	*xlfd)

XftFontOpenXlfd يعمل بنفس طريقة XftFontOpen، باستثناء أنه يأخذ سلسلة تحتوي على وصف خط X المنطقي (XLFD)، ويستخدم الدالة XftXlfdParse().

FcPattern *XftFontMatch (
	Display	*dpy,
	int	 screen,
	_Xconst FcPattern	*pattern,
	FcResult	*result);

تُستخدم أيضًا داخليًا بواسطة الدوال XftFontOpen*، يمكن استخدام XftFontMatch مباشرة لتحديد نمط خط Xft الناتج عن طلب فتح خط Xft.

FcPattern *XftXlfdParse (
	_Xconst char	*xlfd_orig,
	Bool	ignore_scalable,
	Bool	complete);

تقوم XftXlfdParse بتحليل المعامل xlfd_orig وفقًا لوثيقة X Logical Font Description Conventions، ولكنها تتجاهل بعض الحقول: setwidth_name، add_style_name، spacing، و average_width.

تقوم XftXlfdParse بإنشاء نمط Fontconfig، وتعيين الخاصية XFT_XLFD إلى القيمة xlfd_orig، وتربط المعلومات المجمعة بخصائص Fontconfig. يتم تجاهل الحقول الفارغة أو "*":
FC_FOUNDRY
من foundry.
FC_FAMILY
من family.
FC_WEIGHT
weight_name، افتراضيًا FC_WEIGHT_MEDIUM.
FC_SLANT
من slant، افتراضيًا FC_SLANT_ROMAN.
FC_SIZE
من point_size.
FC_PIXEL_SIZE
من pixel_size. إذا تم تعيين point_size، وكذلك resolution_x و resolution_y، فإن القيمة يتم قياسها لتحويل ارتفاع الخط إلى نقاط.

تحديد أبعاد النص

توفر Xft العديد من الدوال لتحديد الارتفاع والعرض المطلوبين لعرض سلسلة نصية. بعد حساب الإزاحة، في الحالات التي سيتم فيها تحريك السلسلة لأعلى أو لأسفل أو لليسار أو لليمين، يشار إلى هذه الأرقام باسم أبعاد النص.

void XftTextExtents8 (
	Display	*dpy,
	XftFont	*font,
	_Xconst FcChar8	*string,
	int	 len,
	XGlyphInfo	*extents);

تحسب XftTextExtents8 أبعاد البكسل في العرض dpy لما لا يزيد عن len حرفًا من string المكون من أحرف ثنائية البايت عند رسمها باستخدام font، وتخزنها في extents.

void XftTextExtents16 (
	Display	*dpy,
	XftFont	*font,
	_Xconst FcChar16	*string,
	int	 len,
	XGlyphInfo	*extents);

تحسب XftTextExtents16 أبعاد البكسل في العرض dpy لما لا يزيد عن len حرفًا من string المكون من أحرف ستة عشرية البايت عند رسمها باستخدام font، وتخزنها في extents.

void XftTextExtents32 (
	Display	*dpy,
	XftFont	*font,
	_Xconst FcChar32	*string,
	int	 len,
	XGlyphInfo	*extents);

تحسب XftTextExtents32 أبعاد البكسل في العرض dpy لما لا يزيد عن len حرفًا من string المكون من أحرف اثنان وثلاثون بت عند رسمها باستخدام font، وتخزنها في extents.

void XftTextExtentsUtf8 (
	Display	*dpy,
	XftFont	*font,
	_Xconst FcChar8	*string,
	int	 len,
	XGlyphInfo	*extents);

تحسب XftTextExtentsUtf8 أبعاد البكسل في العرض dpy لما لا يزيد عن len بايت من string المشفر بـ UTF-8 عند رسمها باستخدام font، وتخزنها في extents.

void XftTextExtentsUtf16 (
	Display	*dpy,
	XftFont	*font,
	_Xconst FcChar8	*string,
	FcEndian	 endian,
	int	 len,
	XGlyphInfo	*extents);

تحسب XftTextExtentsUtf16 أبعاد البكسل في العرض dpy لما لا يزيد عن len بايت من string المشفر بـ UTF-16LE أو UTF-16BE عند رسمها باستخدام font، وتخزنها في extents. يجب تحديد ترتيب البايتات للسلسلة string في endian.

void XftGlyphExtents (
	Display	*dpy,
	XftFont	*font,
	_Xconst FT_UInt	*glyphs,
	int	 nglyphs,
	XGlyphInfo	*extents);

تُستخدم أيضًا داخليًا بواسطة الدوال XftTextExtents*()، تقوم XftGlyphExtents بحساب أبعاد البكسل في العرض dpy لما لا يزيد عن nglyphs في المصفوفة glyphs المرسومة باستخدام font، وتخزنها في extents.

إذا كان أي من glyphs مفقودًا (يتم تحديده عن طريق فحص باستخدام XftFontCheckGlyph())، فإن الإدخال المقابل في extents يتم ملؤه بالأصفار.

إدارة XftColor

Bool XftColorAllocName (
	Display	*dpy,
	_Xconst Visual	*visual,
	Colormap	 cmap,
	_Xconst char	*name,
	XftColor	*result);

استخدم XAllocNamedColor() للبحث عن اللون المسمى name للشاشة المرتبطة بـ colormap cmap.

  • إذا أعادت XAllocNamedColor() قيمة غير صفرية، فإن XftColorAllocName() تملأ حقل البكسل الناتج في XftColor بأقرب لون مدعوم من الشاشة، بالإضافة إلى حقول الأحمر والأخضر والأزرق الدقيقة من قاعدة البيانات، وتعيد True.
  • إذا أعادت XAllocNamedColor() قيمة صفرية، فإن XftColorAllocName() تعيد False، ولا تحدث XftColor المشار إليها بواسطة result.
المعامل visual غير مستخدم.
Bool XftColorAllocValue (
	Display	*dpy,
	Visual	*visual,
	Colormap	 cmap,
	_Xconst XRenderColor	*color,
	XftColor	*result);

تخصيص قيمة لون:

  • إذا كان فئة visual هي TrueColor، فإن XftColorAllocValue() تعين حقل pixel في XftColor المشار إليها بواسطة result باستخدام حقول الأحمر والأخضر والأزرق من المعامل color.
  • إذا لم تكن فئة visual هي TrueColor، فإن XftColorAllocValue() تستدعي XAllocColor() لتخصيص إدخال في colormap cmap، والتي تعيد قيمة البكسل للون الأقرب للعناصر RGB المحددة المدعومة بواسطة الأجهزة.
إذا نجحت XAllocColor()، فإن XftColorAllocValue() تخزن قيمة البكسل تلك في XRenderColor المشار إليها بواسطة result وتعيد True.
إذا فشلت XAllocColor()، فإن XftColorAllocValue() تعيد False ولا تعدل النتيجة.
void XftColorFree (
	Display	*dpy,
	Visual	*visual,
	Colormap	 cmap,
	XftColor	*color);

إذا لم تكن فئة visual هي TrueColor، فإن Xft تستدعي XFreeColors() لتحرير الإدخال من colormap cmap التي تم تخصيص قيمة البكسل الخاصة بها في المعامل color بواسطة XftColorAllocName().

إدارة XftDraw

XftDraw *XftDrawCreate (
	Display	*dpy,
	Drawable	 drawable,
	Visual	*visual,
	Colormap	 colormap);

تنشئ XftDrawCreate بنية يمكن استخدامها لرسم النصوص والمستطيلات باستخدام drawable و visual و colormap المحددة في display.

XftDraw *XftDrawCreateBitmap (
	Display	*dpy,
	Pixmap	 bitmap);

XftDrawCreateBitmap يتصرف مثل XftDrawCreate()، باستثناء أنه يستخدم خريطة بكسل X بعمق لون 1 بدلاً من كائن رسم X.

XftDraw * XftDrawCreateAlpha (
	Display	*dpy,
	Pixmap	 pixmap,
	int	 depth);

XftDrawCreateAlpha يتصرف مثل XftDrawCreate()، باستثناء أنه يستخدم خريطة بكسل X بعمق لون depth بدلاً من كائن رسم X.

void XftDrawChange (
	XftDraw	*draw,
	Drawable	 drawable);

XftDrawChange يغير كائن الرسم X المرتبط بكائن الرسم Xft الموجود draw من قيمته الحالية إلى drawable.

Display *XftDrawDisplay (

XftDrawDisplay يعيد مؤشرًا إلى العرض المرتبط بكائن الرسم Xft draw.

Drawable XftDrawDrawable (
	XftDraw	*draw);

XftDrawDrawable يعيد كائن الرسم X المرتبط بكائن الرسم Xft draw.

Colormap XftDrawColormap (

XftDrawColormap() تعيد خريطة الألوان المرتبطة بكائن الرسم Xft draw.

Visual *XftDrawVisual (
	XftDraw	*draw);

XftDrawVisual يعيد مؤشرًا إلى العرض المرتبط بكائن الرسم Xft draw.

Picture XftDrawPicture (
	XftDraw	*draw);

XftDrawPicture يعيد الصورة المرتبطة بكائن الرسم Xft draw.

إذا كان خادم X لا يدعم امتداد X Rendering، يتم إرجاع 0.
Picture XftDrawSrcPicture (
	XftDraw	*draw,
	_Xconst XftColor	*color);

تعيد كائن صورة X Render، والذي يستخدم لرسم الحروف، على سبيل المثال، مع XftGlyphRender() أو XftGlyphSpecRender() أو XftGlyphFontSpecRender()، بواسطة XftDrawGlyphs() أو XftDrawGlyphSpec() أو XftDrawGlyphFontSpec()، على التوالي.

إذا كان خادم X لا يدعم امتداد X Render، فإن تلك الوظائف تستخدم XftGlyphCore() أو XftGlyphSpecCore() أو XftGlyphFontSpecCore().
void XftDrawDestroy (
	XftDraw	*draw);

XftDrawDestroy يدمر draw (الذي تم إنشاؤه بواسطة إحدى وظائف XftDrawCreate*()) ويحرر الذاكرة التي تم تخصيصها له.

Bool XftDrawSetClip (
	XftDraw	*draw,
	Region	 r);

يقوم بتعيين قص للمحتوى لكائن XftDraw المحدد draw بدءًا من Region:

  • إذا كان المعامل Region r ليس فارغًا، فإن Xft ينشئ Region جديدًا (لنسخ المعامل)،
  • يقوم Xft بتدمير أي منطقة قص موجودة.
  • يقوم Xft بتعيين نوع القص (clip_type) للمعامل draw إلى XftClipTypeRegion إذا لم يكن المعامل r فارغًا. وإلا، فإنه يعين نوع القص إلى XftClipTypeNone.
  • أخيرًا، يقوم Xft بتحديث القص للكائنات الموجودة، وتحديث قناع القص (clip_mask) لكائن صورة X Render الخاص به، وتعيين قناع القص في سياق الرسومات (GC) المرتبط بكائن XftDraw.
XftDrawSetClip() تعيد True إذا لم يكن هناك تغيير ضروري، أو إذا نجحت العملية. وتعيد False إذا لم تتمكن من إنشاء Region جديد.
Bool XftDrawSetClipRectangles (
	XftDraw	*draw,
	int	 xOrigin,
	int	 yOrigin,
	_Xconst XRectangle	*rects,
	int	 n);

مثل XftDrawSetClip()، فإن XftDrawSetClipRectangles() يقوم بتعيين قص للمحتوى لكائن XftDraw المحدد draw ولكنه يستخدم مجموعة من n مستطيلات (المعامل rects) والتي يمكن استخدامها لإنشاء Region.

يقوم Xft بتعيين نوع القص (clip_type) للمعامل draw إلى XftClipTypeRectangles ويستخدم XSetClipRectangles() للقص الأساسي (X11) و XRenderSetPictureClipRectangles() لقص X Render.
void XftDrawSetSubwindowMode (
	XftDraw	*draw,
	int	 mode);

يقوم بتعيين وضع النافذة الفرعية (subwindow-mode) لكائن XftDraw المحدد draw. يمكن أن يكون الوضع إما ClipByChildren (الافتراضي) أو IncludeInferiors:

  • بالنسبة لـ ClipByChildren، يتم قص كل من النوافذ المصدر والوجهة إضافيًا بواسطة جميع الأبناء المرئيين من نوع InputOutput.
  • بالنسبة لـ IncludeInferiors، لا يتم قص أي من النافذة المصدر ولا النافذة الوجهة بواسطة الأبناء. سيؤدي هذا إلى تضمين محتويات النوافذ الفرعية في المصدر والرسم عبر حدود النوافذ الفرعية للوجهة.
بالإضافة إلى وضع النافذة الفرعية الذي تحتفظ به Xft، فإنه يقوم بتحديث وضع النافذة الفرعية لأي سياق رسومات (GC) مرتبط باستخدام XSetSubwindowMode() بالإضافة إلى صورة X Render باستخدام XRenderChangePicture().

void XftCharFontSpecRender (
	Display	*dpy,
	int	 op,
	Picture	 src,
	Picture	 dst,
	int	 srcx,
	int	 srcy,
	_Xconst XftCharFontSpec	*chars,
	int	 len);

XftCharFontSpecRender() تحول المعامل chars من XftCharFontSpec إلى XftGlyphFontSpec، وتمرر المصفوفة المحولة مع المعاملات الأخرى دون تغيير إلى XftGlyphFontSpecRender() لرسم البيانات.

void XftDrawGlyphFontSpec (
	XftDraw	*draw,
	_Xconst XftColor	*color,
	_Xconst XftGlyphFontSpec	*glyphs,
	int	 len);

ترسم قائمة من الحروف المرتبطة بالخطوط في إحداثيات محددة، يتم تمريرها كمصفوفة من len هياكل XftGlyphFontSpec عبر المعامل glyphs. يتم رسم جميع الحروف باستخدام اللون المحدد في المعامل color.

لكل إدخال في مصفوفة glyphs:
  • إذا كان الخط المرتبط يستخدم امتداد X Render، فإن Xft يستخدم XftGlyphFontSpecRender() لرسم الحرف، باستخدام Picture تم الحصول عليها عن طريق استدعاء XftDrawSrcPicture () مع معاملي draw و color.
  • وإلا، فإن Xft توفر ميزة مماثلة باستخدام Xlib.
void XftGlyphFontSpecRender (
	Display	*dpy,
	int	 op,
	Picture	 src,
	Picture	 dst,
	int	 srcx,
	int	 srcy,
	_Xconst XftGlyphFontSpec	*glyphs,
	int	 nglyphs);

يتضمن هذا عدة خطوات:

  • أولاً، يضمن Xft أن مصفوفة glyphs كاملة باستخدام XftFontCheckGlyph() وتحميل أي حروف مفقودة باستخدام XftFontLoadGlyphs().
  • ثم يفحص Xft الحروف لتحديد أقصى قيمة Unicode. هذا يحدد حجم الترميز، أي width.
  • بعد ذلك، لكل حرف، يتحقق Xft مما إذا كانت قيمة Unicode الخاصة به مدعومة من قبل الخط المقابل. إذا لم يكن كذلك، يتم تعيين القيمة إلى صفر (0)، لتجاهلها. أثناء القيام بذلك، يتذكر Xft أيضًا أي هو الخط الأول المستخدم، ويحسب الموضع الذي سيتم فيه رسم كل حرف.
  • بعد ذلك، ينشئ Xft مصفوفة مكافئة من الحروف بالتنسيق المتوقع من قبل مكتبة X Render.
  • أخيرًا، يستخدم XftGlyphFontSpecRender() امتداد X Render لرسم الحروف، باستخدام دالة XRenderCompositeText*() المناسبة وفقًا لـ width، ويتخلص من البيانات المؤقتة.
تستخدم المعاملات op و dst و src و srcx و srcy كما هو موضح في وثائق مكتبة X Render.
