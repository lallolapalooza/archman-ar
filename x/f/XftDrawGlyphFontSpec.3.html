XftDrawGlyphFontSpec(3)

الحزمة: libxft

التراخيص: HPND-sell-variant

XftFontOpen، باستثناء أنها تأخذ سلسلة نمط Fontconfig (التي يتم تمريرها إلى مكتبة Fontconfig FcNameParse()).

XftFont *XftFontOpenXlfd (
	Display	*dpy,
	int	 screen,
	_Xconst char	*xlfd)

XftFontOpenXlfd يعمل مثل XftFontOpen، باستثناء أنه يأخذ سلسلة تحتوي على وصف خط X المنطقي (XLFD)، ويستخدم الدالة XftXlfdParse().

FcPattern *XftFontMatch (
	Display	*dpy,
	int	 screen,
	_Xconst FcPattern	*pattern,
	FcResult	*result);

يتم استخدامه أيضًا داخليًا بواسطة الدوال XftFontOpen*، يمكن أيضًا استخدام XftFontMatch مباشرة لتحديد نمط خط Fontconfig الناتج عن طلب فتح خط Xft.

FcPattern *XftXlfdParse (
	_Xconst char	*xlfd_orig,
	Bool	ignore_scalable,
	Bool	complete);

يقوم XftXlfdParse بتحليل المعلمة xlfd_orig وفقًا لوثيقة اتفاقيات وصف خط X المنطقي، ولكنه يتجاهل بعض الحقول: setwidth_name، add_style_name، spacing، و average_width.

يقوم XftXlfdParse بإنشاء نمط Fontconfig، وتعيين الخاصية XFT_XLFD إلى قيمة xlfd_orig، ويربط المعلومات المجمعة بخصائص Fontconfig. يتم تجاهل الحقول الفارغة أو التي تحتوي على " * ":
FC_FOUNDRY
من foundry.
FC_FAMILY
من family.
FC_WEIGHT
weight_name، القيمة الافتراضية هي FC_WEIGHT_MEDIUM.
FC_SLANT
من slant، القيمة الافتراضية هي FC_SLANT_ROMAN.
FC_SIZE
من point_size.
FC_PIXEL_SIZE
من pixel_size. إذا تم تعيين point_size، وكذلك resolution_x و resolution_y، فسيتم تحجيم القيمة لتحويل ارتفاع الخط إلى نقاط.

عائلات الوظائف

توفر Xft مجموعات من وظائف الرسم تختلف حسب كيفية ترميز البيانات، على سبيل المثال، 8 بت (ISO-8859-1) مقابل UTF-8. إليك ملخص لهذه الوظائف ذات الصلة:

تحديد امتدادات النص

توفر Xft عدة وظائف لتحديد الارتفاع والعرض المطلوبين لعرض سلسلة نصية. بعد حساب الإزاحة، في الحالات التي سيتم فيها تحريك السلسلة لأعلى أو لأسفل أو لليسار أو لليمين، يشار إلى هذه الأرقام باسم امتدادات النص.

void XftTextExtents8 (
	Display	*dpy,
	XftFont	*font,
	_Xconst FcChar8	*string,
	int	 len,
	XGlyphInfo	*extents);

تحسب XftTextExtents8 امتدادات البكسل على الشاشة dpy لما لا يزيد عن len حرفًا من سلسلة تتكون من أحرف ثمانية البت عند رسمها بالخط font، وتخزينها في extents.

void XftTextExtents16 (
	Display	*dpy,
	XftFont	*font,
	_Xconst FcChar16	*string,
	int	 len,
	XGlyphInfo	*extents);

تحسب XftTextExtents16 امتدادات البكسل على الشاشة dpy لما لا يزيد عن len حرفًا من سلسلة تتكون من أحرف ستة عشر بت عند رسمها بالخط font، وتخزينها في extents.

void XftTextExtents32 (
	Display	*dpy,
	XftFont	*font,
	_Xconst FcChar32	*string,
	int	 len,
	XGlyphInfo	*extents);

تحسب XftTextExtents32 امتدادات البكسل على الشاشة dpy لما لا يزيد عن len حرفًا من سلسلة تتكون من أحرف اثنين وثلاثين بت عند رسمها بالخط font، وتخزينها في extents.

void XftTextExtentsUtf8 (
	Display	*dpy,
	XftFont	*font,
	_Xconst FcChar8	*string,
	int	 len,
	XGlyphInfo	*extents);

تحسب XftTextExtentsUtf8 امتدادات البكسل على الشاشة dpy لما لا يزيد عن len بايت من سلسلة مشفرة بـ UTF-8 عند رسمها بالخط font، وتخزينها في extents.

void XftTextExtentsUtf16 (
	Display	*dpy,
	XftFont	*font,
	_Xconst FcChar8	*string,
	FcEndian	 endian,
	int	 len,
	XGlyphInfo	*extents);

تحسب XftTextExtentsUtf16 امتدادات البكسل على الشاشة dpy لما لا يزيد عن len بايت من سلسلة مشفرة بـ UTF-16LE أو UTF-16BE عند رسمها بالخط font، وتخزينها في extents. يجب تحديد ترتيب البايتات للسلسلة string في endian.

void XftGlyphExtents (
	Display	*dpy,
	XftFont	*font,
	_Xconst FT_UInt	*glyphs,
	int	 nglyphs,
	XGlyphInfo	*extents);

يتم استخدامه أيضًا داخليًا بواسطة وظائف XftTextExtents*()، تقوم XftGlyphExtents بحساب امتدادات البكسل على الشاشة dpy لما لا يزيد عن nglyphs في المصفوفة glyphs المرسومة بالخط font، وتخزينها في extents.

إذا كانت أي من glyphs مفقودة (يتم تحديد ذلك عن طريق التحقق باستخدام XftFontCheckGlyph())، يتم ملء الإدخال المقابل في extents بالأصفار.

إدارة XftColor

Bool XftColorAllocName (
	Display	*dpy,
	_Xconst Visual	*visual,
	Colormap	 cmap,
	_Xconst char	*name,
	XftColor	*result);

استخدم XAllocNamedColor() للبحث عن اللون المسمى name للشاشة المرتبطة بخريطة الألوان cmap.

  • إذا أعادت XAllocNamedColor() قيمة غير صفرية، فإن XftColorAllocName() تملأ حقل البكسل XftColor الناتج بأقرب لون تدعمه الشاشة، بالإضافة إلى حقول الأحمر والأخضر والأزرق الدقيقة من قاعدة البيانات، وتعيد True.
  • إذا أعادت XAllocNamedColor() قيمة صفرية، فإن XftColorAllocName() تعيد False، ولا تقوم بتحديث XftColor المشار إليه بواسطة result.
المعلمة visual غير مستخدمة.
Bool XftColorAllocValue (
	Display	*dpy,
	Visual	*visual,
	Colormap	 cmap,
	_Xconst XRenderColor	*color,
	XftColor	*result);

تخصيص قيمة لون:

  • إذا كان فئة visual هي TrueColor، فإن XftColorAllocValue() تعين حقل pixel في XftColor المشار إليه بواسطة result باستخدام حقول الأحمر والأخضر والأزرق من المعلمة color.
  • إذا لم تكن فئة visual هي TrueColor، فإن XftColorAllocValue() تستدعي XAllocColor() لتخصيص إدخال في خريطة الألوان cmap. والتي تعيد قيمة البكسل للون الأقرب للعناصر RGB المحددة المدعومة بالأجهزة.
إذا نجحت XAllocColor()، تقوم XftColorAllocValue() بتخزين قيمة البكسل هذه في XRenderColor المشار إليه بواسطة result وتعيد True.
إذا فشلت XAllocColor()، فإن XftColorAllocValue() تعيد False ولا تعدل النتيجة.
void XftColorFree (
	Display	*dpy,
	Visual	*visual,
	Colormap	 cmap,
	XftColor	*color);

إذا لم تكن فئة visual هي TrueColor، فإن Xft يستدعي XFreeColors() لتحرير الإدخال من خريطة الألوان cmap التي تم تخصيص قيمة البكسل الخاصة بها في المعلمة color بواسطة XftColorAllocName().
