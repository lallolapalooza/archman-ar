XftTextExtents8(3)

الحزمة: libxft

التراخيص: HPND-sell-variant

b>XftFontOpen، باستثناء أنها تأخذ سلسلة نمط Fontconfig (والتي يتم تمريرها إلى مكتبة Fontconfig FcNameParse()).

XftFont *XftFontOpenXlfd (
	Display	*dpy,
	int	 screen,
	_Xconst char	*xlfd)

XftFontOpenXlfd يعمل مثل XftFontOpen، باستثناء أنه يأخذ سلسلة تحتوي على وصف الخط المنطقي X (XLFD)، ويستخدم الدالة XftXlfdParse().

FcPattern *XftFontMatch (
	Display	*dpy,
	int	 screen,
	_Xconst FcPattern	*pattern,
	FcResult	*result);

يُستخدم أيضًا داخليًا بواسطة الدوال XftFontOpen*، يمكن استخدام XftFontMatch مباشرة لتحديد نمط خط Fontconfig الناتج عن طلب فتح خط Xft.

FcPattern *XftXlfdParse (
	_Xconst char	*xlfd_orig,
	Bool	ignore_scalable,
	Bool	complete);

تقوم XftXlfdParse بتحليل المعلمة xlfd_orig وفقًا لوثيقة X Logical Font Description Conventions، ولكنها تتجاهل بعض الحقول: setwidth_name، add_style_name، spacing، و average_width.

تقوم XftXlfdParse بإنشاء نمط Fontconfig، مع ضبط الخاصية XFT_XLFD بقيمة xlfd_orig، وربط المعلومات المجمعة بخصائص Fontconfig. يتم تجاهل الحقول الفارغة أو التي تحتوي على "" :
FC_FOUNDRY
من foundry.
FC_FAMILY
من family.
FC_WEIGHT
weight_name، افتراضيًا FC_WEIGHT_MEDIUM.
FC_SLANT
من slant، افتراضيًا FC_SLANT_ROMAN.
FC_SIZE
من point_size.
FC_PIXEL_SIZE
من pixel_size. إذا تم تعيين point_size، بالإضافة إلى resolution_x و resolution_y، فسيتم تحجيم القيمة لتحويل ارتفاع الخط إلى نقاط.

عائلات الوظائف

توفر Xft مجموعات من وظائف الرسم التي تختلف وفقًا لطريقة ترميز البيانات، على سبيل المثال، 8 بت (ISO-8859-1) مقابل UTF-8. إليك ملخص لهذه الوظائف ذات الصلة:

تحديد امتدادات النص

توفر Xft عدة دوال لتحديد الارتفاع والعرض المطلوبين لعرض سلسلة نصية. بعد حساب الإزاحة، في الحالات التي سيتم فيها إزاحة السلسلة لأعلى أو لأسفل أو لليسار أو لليمين، يشار إلى هذه الأرقام باسم امتدادات النص.

void XftTextExtents8 (
	Display	*dpy,
	XftFont	*font,
	_Xconst FcChar8	*string,
	int	 len,
	XGlyphInfo	*extents);

تحسب XftTextExtents8 امتدادات البكسل على الشاشة dpy لعدد لا يزيد عن len من الأحرف الرسومية لسلسلة تتكون من أحرف ثمانية البت عند رسمها بالخط font، وتخزنها في extents.

void XftTextExtents16 (
	Display	*dpy,
	XftFont	*font,
	_Xconst FcChar16	*string,
	int	 len,
	XGlyphInfo	*extents);

تحسب XftTextExtents16 امتدادات البكسل على الشاشة dpy لعدد لا يزيد عن len من الأحرف الرسومية لسلسلة تتكون من أحرف ستة عشر بت عند رسمها بالخط font، وتخزنها في extents.

void XftTextExtents32 (
	Display	*dpy,
	XftFont	*font,
	_Xconst FcChar32	*string,
	int	 len,
	XGlyphInfo	*extents);

تحسب XftTextExtents32 امتدادات البكسل على الشاشة dpy لعدد لا يزيد عن len من الأحرف الرسومية لسلسلة تتكون من أحرف اثنان وثلاثون بت عند رسمها بالخط font، وتخزنها في extents.

void XftTextExtentsUtf8 (
	Display	*dpy,
	XftFont	*font,
	_Xconst FcChar8	*string,
	int	 len,
	XGlyphInfo	*extents);

تحسب XftTextExtentsUtf8 امتدادات البكسل على الشاشة dpy لعدد لا يزيد عن len من البايتات لسلسلة مرمزة بـ UTF-8 عند رسمها بالخط font، وتخزنها في extents.

void XftTextExtentsUtf16 (
	Display	*dpy,
	XftFont	*font,
	_Xconst FcChar8	*string,
	FcEndian	 endian,
	int	 len,
	XGlyphInfo	*extents);

تحسب XftTextExtentsUtf16 امتدادات البكسل على الشاشة dpy لعدد لا يزيد عن len من البايتات لسلسلة مرمزة بـ UTF-16LE أو UTF-16BE عند رسمها بالخط font، وتخزنها في extents. يجب تحديد ترتيب البايتات للسلسلة في endian.

void XftGlyphExtents (
	Display	*dpy,
	XftFont	*font,
	_Xconst FT_UInt	*glyphs,
	int	 nglyphs,
	XGlyphInfo	*extents);

تُستخدم أيضًا داخليًا بواسطة الدوال XftTextExtents*()، تحسب XftGlyphExtents امتدادات البكسل على الشاشة dpy لعدد لا يزيد عن nglyphs في المصفوفة glyphs عند رسمها بالخط font، وتخزنها في extents.

إذا كانت أي من glyphs مفقودة (يتم تحديدها بفحص باستخدام XftFontCheckGlyph())، فسيتم ملء الإدخال المقابل في extents بالأصفار.

إدارة XftColor

Bool XftColorAllocName (
	Display	*dpy,
	_Xconst Visual	*visual,
	Colormap	 cmap,
	_Xconst char	*name,
	XftColor	*result);

استخدم XAllocNamedColor() للبحث عن اللون المسمى name للشاشة المرتبطة بخريطة الألوان cmap.

  • إذا أعادت XAllocNamedColor() قيمة غير صفرية، فتقوم XftColorAllocName() بملء حقل البكسل XftColor الناتج بأقرب لون تدعمه الشاشة، بالإضافة إلى حقول الأحمر والأخضر والأزرق الدقيقة من قاعدة البيانات، وتعيد True.
  • إذا أعادت XAllocNamedColor() صفرًا، فتعيد XftColorAllocName() False، ولا تقوم بتحديث XftColor المشار إليها بواسطة result.
المعلمة visual غير مستخدمة.
Bool XftColorAllocValue (
	Display	*dpy,
	Visual	*visual,
	Colormap	 cmap,
	_Xconst XRenderColor	*color,
	XftColor	*result);

تخصيص قيمة لون:

  • إذا كان فئة visual هي TrueColor، فتقوم XftColorAllocValue() بتعيين حقل pixel في XftColor المشار إليه بواسطة result باستخدام حقول الأحمر والأخضر والأزرق من المعلمة color.
  • إذا لم تكن فئة visual هي TrueColor، فتقوم XftColorAllocValue() باستدعاء XAllocColor() لتخصيص إدخال في خريطة الألوان cmap. والتي تعيد قيمة البكسل للون الأقرب للعناصر RGB المحددة المدعومة بواسطة الأجهزة.
إذا نجحت XAllocColor()، تقوم XftColorAllocValue() بتخزين قيمة البكسل هذه في XRenderColor المشار إليها بواسطة result وتعيد True.
إذا فشلت XAllocColor()، فتعيد XftColorAllocValue() False ولا تعدل النتيجة.
void XftColorFree (
	Display	*dpy,
	Visual	*visual,
	Colormap	 cmap,
	XftColor	*color);

إذا لم تكن فئة visual هي TrueColor، فإن Xft تستدعي XFreeColors() لتحرير الإدخال من خريطة الألوان cmap التي تم تخصيص قيمة البكسل الخاصة بها في المعلمة color بواسطة XftColorAllocName().

التعامل مع بيانات الخط

void XftFontLoadGlyphs (
	Display	*dpy,
	XftFont	*pub,
	FcBool	 need_bitmaps,
	_Xconst FT_UInt	*glyphs,
	int	 nglyph);

باستخدام FT_Load_Glyph()، يقوم بتحميل nglyphs لمؤشرات الرموز الرسومية (قيم Unicode) المدرجة في المصفوفة glyphs من الخط المحدد pub.

تحميل الرمز الرسومي يتطلب أكثر من مجرد قراءة البيانات في الذاكرة. يستخدم Xft نمط Fontconfig المرتبط بالعرض dpy (على سبيل المثال، عبر XftDefaultSet()) لتحديد ما إذا كان سيتم استخدام مربع محيط، أو قص الرموز الرسومية ضمن هذه الحدود، أو قياس الرموز الرسومية، أو حساب مقاييس الخط، وإضافتها إلى امتداد X Render باستخدام XRenderAddGlyphs().
void XftFontUnloadGlyphs (
	Display	*dpy,
	XftFont	*pub,
	_Xconst FT_UInt	*glyphs,
	int	 nglyph);

يتخلص من البيانات لما يصل إلى nglyph من الرموز الرسومية التي تظهر مؤشرات الرموز الرسومية (قيم Unicode) الخاصة بها في المصفوفة glyphs للخط المحدد pub. إذا تمت إضافة الرموز الرسومية إلى امتداد X Render، فإن Xft يزيلها باستخدام XRenderFreeGlyphs(). يحتفظ Xft بسجل لكمية الذاكرة المستخدمة للرموز الرسومية، ويقوم بتحديث الاستخدام المرتبط بالعرض dpy.

FT_Face XftLockFace (
	XftFont	*pub);

إذا لم يتم إنشاء "وجه" FreeType 2 (تمثيل في الذاكرة لنوع خط معين بأسلوب معين) للخط pub، فقم بإنشاء واحد باستخدام FT_New_Face(). أقفال الوجه هي عداد مرجعي تستخدمه Xft لضمان إنشاء وجه واحد فقط، وأن يتم الاحتفاظ به حتى لا يعود الخط قيد الاستخدام.

يتم استخدام أقفال الوجه مباشرة في XftCharIndex() و XftFontLoadGlyphs()، والتي بدورها تستخدم في العديد من وظائف Xft.
تم تقديم أقفال الوجه في الإصدار 1 (أكتوبر 2002). اعتمدت بعض التطبيقات، مثل Gdk/Gimp، على هذه الوظائف. في الإصدار 2.1.9 (يونيو 2006)، تم الاحتفاظ بأقفال الوجه كجزء من واجهة برمجة التطبيقات العامة عندما وفرت تكوينات مكتبات المشاركة المحسنة إخفاء الرموز الخاصة.
void XftUnlockFace (
	XftFont	*pub);

يقلل من عدد المراجع لـ "وجه" FreeType 2 المرتبط بالخط.

يتم إلغاء تخصيص كائنات FT_Face باستخدام FT_Done_Face(). يقوم Xft بذلك في XftFontInfoDestroy() وعند التنظيف عند الفشل في XftFontInfoCreate() و XftFontOpenPattern().

يعرض السلسلة النصية التي سيقوم XftDrawString8() برسمها.

يعرض معرفات الخطوط التي تمت مطابقتها.

يعرض معلومات مفيدة حول الرموز الرسومية التي سيتم رسمها

التوافق

اعتبارًا من الإصدار 2 (مايو 2002)، أصبح Xft مستقرًا نسبيًا. من المتوقع أن يحتفظ بالتوافق المصدر والثنائي في الإصدارات المستقبلية.

يوفر Xft واجهة توافق مع الإصدار الرئيسي السابق، Xft 1.x، الموصوف أدناه.

ملف رأس Xft 1.x

#include <X11/Xft/XftCompat.h>

أنواع بيانات Xft 1.x

XftPattern
يحمل مجموعة من الأسماء مع قوائم القيم المرتبطة بها؛ يشير كل اسم إلى خاصية للخط. تستخدم XftPattern كمدخلات لتعليمات المطابقة وكذلك للاحتفاظ بمعلومات حول خطوط معينة.
XftFontSet
يحتوي على قائمة من XftPattern. داخليًا، يستخدم Xft هيكل البيانات هذا للاحتفاظ بمجموعات الخطوط. خارجيًا، يعيد Xft نتائج قائمة الخطوط بهذا التنسيق.
XftObjectSet
يحمل مجموعة من الأسماء ويستخدم لتحديد الحقول التي سيتم وضعها من الخطوط في قائمة الأنماط المعادة عند سرد الخطوط.
