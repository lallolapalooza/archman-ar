XftFontOpenName(3)

الحزمة: libxft

التراخيص: HPND-sell-variant

DESCRIPTION

XftDraw

هو كائن غير شفاف يحتوي على معلومات تُستخدم للرسم على كائن رسومي X باستخدام البروتوكول الأساسي أو امتداد X Rendering.

يتم إنشاء كائنات XftDraw باستخدام أي من الوظائف التالية: XftDrawCreate() (التي تربط XftDraw بكائن رسومي X موجود)، XftDrawCreateBitmap()، أو XftDrawCreateAlpha()، ويتم تدميرها باستخدام XftDrawDestroy(). يمكن تغيير الكائن الرسومي X المرتبط بـ XftDraw باستخدام XftDrawChange(). يتم تخصيص كائنات XftDraw وتحريرها داخليًا بواسطة Xft؛ لا يحتاج المبرمج عادةً إلى تخصيص مساحة تخزين لها أو تحريرها.

يمكن الاستعلام عن خصائص X Display، Drawable، Colormap، و Visual لكائن XftDraw باستخدام XftDrawDisplay()، XftDrawDrawable()، XftDrawColormap()، و XftDrawVisual()، على التوالي.

تستخدم العديد من الوظائف كائنات XftDraw: XftDrawCharFontSpec()، XftDrawCharSpec()، XftDrawGlyphFontSpec()، XftDrawGlyphSpec()، XftDrawGlyphs()، XftDrawRect()، XftDrawSetClip()، XftDrawSetClipRectangles()، XftDrawSetSubwindowMode()، وعائلة الوظائف XftDrawString*().

يتم إرجاع امتداد X Rendering Picture المرتبط بـ XftDraw بواسطة XftDrawPicture() و XftDrawSrcPicture(). يتم استخدامه بواسطة XftCharFontSpecRender()، XftCharSpecRender()، XftGlyphFontSpecRender()، XftGlyphRender()، XftGlyphSpecRender()، وعائلة الوظائف XftTextRender*().

FUNCTIONS

Initialization

عادةً، لا يقوم التطبيق النموذجي الذي يستخدم Xft بتهيئة المكتبة بشكل صريح. يتم ذلك عادةً كنتيجة جانبية لفتح خط.

عند تهيئة Xft، فإنه يجمع معلومات حول الشاشة، ويخزن بعض هذه المعلومات في نمط Fontconfig (وهو أساسًا مجموعة من الخصائص ذات القيم المعرفة بنوع. يمكن للتطبيق المستدعي تعديل هذا النمط لتغيير سلوك المكتبة.

لا يتم تخزين جميع المعلومات المجمعة في نمط. يتم تخزين الباقي في هياكل بيانات داخلية. يوفر Xft بعضًا من تلك المعلومات للتطبيق عبر الوظائف.

Bool XftDefaultHasRender (
	Display	*dpy);

الحصول على معلومات حول الشاشة dpy إذا لم يتم ذلك بالفعل، وإرجاع true إذا وجد Xft أن الشاشة تدعم امتداد X Render، وإذا كان قادرًا على العثور على XRenderPictFormat مناسب (نوع بيانات X Render الذي يشبه Visual الخاص بـ Xlib) على الشاشة.

Bool XftDefaultSet (
	Display	*dpy,
	FcPattern	*defaults);

الحصول على معلومات حول الشاشة dpy إذا لم يتم ذلك بالفعل، وتعيين نمط Fontconfig الذي يحمل الخصائص الافتراضية التي سيستخدمها Xft لهذه الشاشة.

يستخدم Xft تلك الخصائص مبدئيًا للحصول على هذه الحدود:
XFT_MAX_GLYPH_MEMORY
(maxglyphmemory). هذا هو الحد الأقصى لذاكرة الأحرف لجميع الخطوط التي يستخدمها Xft (الافتراضي: 4*1024*1024).
XFT_MAX_UNREF_FONTS
(maxunreffonts). هذا هو الحد الأقصى لعدد الخطوط غير المشار إليها (الافتراضي: 16).
XFT_TRACK_MEM_USAGE
(trackmemusage). عندما تكون صحيحة، يتتبع Xft استخدام ذاكرة الأحرف لتحسين الأداء عند تحديد أي منها سيتم إلغاء تحميله عند الوصول إلى الحد الأقصى لذاكرة الأحرف (الافتراضي: false).
يستخدم Xft أيضًا هذه الخصائص الافتراضية في XftDefaultSubstitute().
void XftDefaultSubstitute (
	Display	*dpy,
	int	 screen,
	FcPattern	*pattern);

يقوم Xft بملء الخصائص المفقودة في pattern المعطى باستخدام الخصائص الافتراضية للشاشة المحددة dpy، على سبيل المثال، كما تم تعيينها في XftDefaultSet().

عادةً ما تستخدم تطبيقات Xft هذه الوظيفة لمساعدة Fontconfig في اختيار خط مناسب. يتم استبدال هذه الخصائص قبل استدعاء FcDefaultSubstitute():
FC_ANTIALIAS
صحيح إذا كان يجب على FreeType استخدام التنعيم المضاد (الافتراضي: False). (الافتراضي: True).
FC_AUTOHINT
صحيح إذا كان يجب على FreeType استخدام التلميحات التلقائية (الافتراضي: False).
FC_DPI
نقاط/بوصة المستخدمة للدقة (الافتراضي: محسوبة من ارتفاع الشاشة).
FC_EMBOLDEN
صحيح إذا كان يجب استخدام FT_GlyphSlot_Embolden() لجعل الخط سميكًا (الافتراضي: False).
FC_HINTING
صحيح إذا كان يجب استخدام التلميحات عند ملء الخصائص لفتح خط (الافتراضي: True).
FC_HINT_STYLE
أسلوب التلميحات المستخدم عند ملء الخصائص لفتح خط (الافتراضي: FC_HINT_FULL).
FC_LCD_FILTER
المعلمة التي يتم تمريرها إلى FT_Library_SetLcdFilter() عند تحميل الأحرف (الافتراضي: FC_LCD_DEFAULT).
FC_MINSPACE
الحد الأدنى للمسافة المستخدمة عند ملء الخصائص لفتح خط (الافتراضي: False).
FC_RGBA
قيمة RGBA المستخدمة عند ملء الخصائص لفتح خط (الافتراضي: محسوبة عن طريق استدعاء XRenderQuerySubpixelOrder()).
FC_SCALE
المقياس المستخدم في Fontconfig (الافتراضي: 1.0).
XFT_MAX_GLYPH_MEMORY
الحد الأقصى للذاكرة لخط واحد (الافتراضي: 1024*1024).
XFT_RENDER
صحيح إذا كانت الشاشة تدعم امتداد X Render (الافتراضي: نتيجة XftDefaultHasRender()).
FcBool XftInit (

المعلمة config غير مستخدمة.

XftFont *XftFontOpen (

  

    
	Display	*dpy,
	int	 screen,
	...);


    
تأخذ XftFontOpen قائمة من ثلاثيات عناصر النمط بالشكل
        field، type، value (تنتهي بـ NULL)، تطابق هذا
        النمط مع الخطوط المتاحة، وتفتح الخط المطابق، وتضبط حجمه بشكل صحيح

مثال:
  




    font = XftFontOpen (dpy, screen,
                        XFT_FAMILY, XftTypeString, "charter",
                        XFT_SIZE, XftTypeDouble, 12.0,
                        NULL);




  

  
هذا يفتح الخط "charter" بحجم 12 نقطة. يتم تحويل حجم النقطة تلقائيًا
      إلى حجم البكسل الصحيح بناءً على دقة الشاشة.

Bool XftDrawSetClip (

  

    
	XftDraw	*draw,
	Region	 r);


    
يقوم بإعداد قص المنطقة لكائن XftDraw المعطى draw
        بدءًا بمنطقة Region:

  






    
  
  • إذا كانت معلمة Region r ليست فارغة، يقوم Xft بإنشاء
      Region جديد (لنسخ المعلمة)،
    • 
  
  • يقوم Xft بتدمير أي منطقة قص موجودة.
    • 
  
  • يقوم Xft بتعيين clip_type للمعلمة draw إلى
        XftClipTypeRegion إذا لم تكن المعلمة r فارغة. وإلا، فإنه
        يعين clip_type إلى XftClipTypeNone.
    • 
  
  • أخيرًا، يقوم Xft بتحديث قص المناطق للكائنات الموجودة، ويقوم بتحديث
        clip_mask لكائن صورة X Render الخاص به، ويعين قناع القص في سياق الرسوميات
        (GC) المرتبط بمعلمة XftDraw.
    • 






  

  
XftDrawSetClip() يعيد True إذا لم يكن هناك تغيير ضروري، أو
      إذا نجحت العملية. ويعيد False إذا لم يتمكن من إنشاء Region
      جديد.





  
Bool XftDrawSetClipRectangles (

  

    
	XftDraw	*draw,
	int	 xOrigin,
	int	 yOrigin,
	_Xconst XRectangle	*rects,
	int	 n);


    
مثل XftDrawSetClip()، يقوم XftDrawSetClipRectangles()
        بإعداد قص المنطقة لكائن XftDraw المعطى draw ولكنه يستخدم
        مجموعة من n مستطيلات (المعلمة rects) والتي يمكن استخدامها
        لبناء Region.

  





  

  
يقوم Xft بتعيين clip_type للمعلمة draw إلى
        XftClipTypeRectangles ويستخدم XSetClipRectangles() للقص الأساسي
        (X11) و XRenderSetPictureClipRectangles() لقص X Render.





  
void XftDrawSetSubwindowMode (

  

    
	XftDraw	*draw,
	int	 mode);


    
يقوم بتعيين وضع النافذة الفرعية لكائن XftDraw المعطى
        draw. يمكن أن يكون الوضع إما ClipByChildren (الافتراضي) أو
        IncludeInferiors:

  






    
  
  • بالنسبة لـ ClipByChildren، يتم قص كل من نوافذ المصدر والوجهة
      بالإضافة إلى جميع الأطفال المرئيين من نوع InputOutput.
    • 
  
  • بالنسبة لـ IncludeInferiors، لا يتم قص أي من نوافذ المصدر أو الوجهة
      بواسطة النوافذ الفرعية. سيؤدي هذا إلى تضمين محتويات النافذة الفرعية في
      المصدر والرسم من خلال حدود النافذة الفرعية للوجهة.
    • 






  

  
بالإضافة إلى وضع النافذة الفرعية الذي يحتفظ به Xft، فإنه يقوم بتحديث
        وضع النافذة الفرعية لأي سياق رسوميات GC مرتبط باستخدام
        XSetSubwindowMode() وكذلك لصورة X Render باستخدام
        XRenderChangePicture().

void XftDrawGlyphFontSpec (

  

    
	XftDraw	*draw,
	_Xconst XftColor	*color,
	_Xconst XftGlyphFontSpec	*glyphs,
	int	 len);


    
يرسم قائمة من الأحرف المرتبطة بالخطوط في إحداثيات محددة، يتم
        تمريرها كمصفوفة من هياكل XftGlyphFontSpec بطول len عبر

لكل إدخال في مصفوفة glyphs:






    
  
  • إذا كان الخط المرتبط يستخدم امتداد X Render، فإن Xft يستخدم
        XftGlyphFontSpecRender() لرسم الحرف، باستخدام صورة Picture
        تم الحصول عليها عن طريق استدعاء XftDrawSrcPicture () مع المعلمات
        draw و color.
    • 
  
  • بخلاف ذلك، يوفر Xft ميزة مماثلة باستخدام Xlib.
    • 






  
void XftGlyphFontSpecRender (

  

    
	Display	*dpy,
	int	 op,
	Picture	 src,
	Picture	 dst,
	int	 srcx,
	int	 srcy,
	_Xconst XftGlyphFontSpec	*glyphs,
	int	 nglyphs);


    
يتضمن هذا عدة خطوات:

  






    
  
  • أولاً، يضمن Xft أن مصفوفة glyphs مكتملة باستخدام
        XftFontCheckGlyph() وتحميل أي أحرف مفقودة باستخدام
        XftFontLoadGlyphs().
    • 
  
  • ثم يقوم Xft بفحص الأحرف للعثور على أقصى قيمة Unicode. هذا يحدد حجم
        الترميز، أي width.
    • 
  
  • بعد ذلك، لكل حرف، يتحقق Xft مما إذا كانت قيمة Unicode الخاصة به
        مدعومة بواسطة الخط المقابل. إذا لم يكن كذلك، فسيتم تعيين القيمة إلى صفر
        (0)، لتجاهلها. أثناء القيام بذلك، يتذكر Xft أيضًا أي خط تم استخدامه
        أولاً، ويحسب الموضع الذي سيتم فيه رسم كل حرف.
    • 
  
  • يقوم Xft بعد ذلك ببناء مصفوفة معادلة للأحرف بالتنسيق المتوقع من
        مكتبة X Render.
    • 
  
  • أخيرًا، تستخدم XftGlyphFontSpecRender() امتداد X Render لرسم
        الأحرف، باستخدام الدالة XRenderCompositeText*() المناسبة وفقًا
        لـ width، وتقوم بتجاهل البيانات المؤقتة.
    • 




  

  
تستخدم المعلمات op و dst و src و srcx و
        srcy كما هو موضح في وثائق مكتبة X Render.
