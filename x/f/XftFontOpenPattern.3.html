XftFontOpenPattern(3)

الحزمة: libxft

Xft(3) إصدار X 11 Xft(3)

الاسم

Xft - مكتبة واجهة X FreeType

الملخص

#include <X11/Xft/Xft.h>

الوصف

Xft مكتبة بسيطة ترسم النصوص والرسومات:

  • باستخدام المعلومات التي توفرها مكتبة Fontconfig،
  • تحوّل Xft محارف الخطوط باستخدام مُنقّط FreeType، و
  • تعرض بيانات الخطوط المحوّلة باستخدام امتداد العرض X Rendering Extension.

هذه الصفحة من الدليل بالكاد تخدش سطح هذه المكتبة.

الثوابت

XFT_MAJOR
هو رقم الإصدار الرئيسي لـ Xft.
XFT_MINOR
هو رقم الإصدار الفرعي لـ Xft.
XFT_REVISION
هو رقم المراجعة لـ Xft.
XFT_VERSION
هو XFT_MAJOR مضروبًا في 10000 (عشرة آلاف)، زائد XFT_MINOR مضروبًا في 100، زائد XFT_REVISION.
XftVersion
هو اسم بديل لـ XFT_VERSION.

يوضح المثال التالي كيف يمكن استخدام ثوابت إصدار Xft: 

    #if (XFT_VERSION >= 20107)
    (void) puts("الإصدار 2.1.7 أو أحدث من مكتبة Xft قيد"
                " الاستخدام.");
    #else
    (void) printf("إصدار غير كافٍ من Xft (%d.%d.%d) مثبّت؛
                  " يلزم الإصدار 2.1.7 على الأقل.\n", XFT_MAJOR,
                  XFT_MINOR,
                  XFT_REVISION);
    #endif

أنواع البيانات

تتبع أنواع بيانات Xft اصطلاح تسمية، حيث تُسبق جميع الأسماء بـ “Xft”

من Xlib...

لا تتبع أنواع بيانات Xlib اصطلاح تسمية. وهي موثقة في Xlib - واجهة لغة C.

يستخدم Xft هذه الأسماء: Bool، وColormap، وDisplay، وDrawable، وPixmap، وRegion، وVisual، وXRectangle.

بعض أنواع البيانات ذات صلة خاصة بـ Xft:

Drawable
مُعرّف في <X11/X.h>، وهو معرّف مورد X، مثل نافذة مرتبطة بشاشة عرض. يمكن استخدام كل من النوافذ والصور النقطية كمصادر ووجهات في عمليات الرسومات. تُعرف هذه النوافذ والصور النقطية مجتمعة باسم العناصر القابلة للرسم (drawables). ومع ذلك، لا يمكن استخدام نافذة InputOnly كمصدر أو وجهة في عملية رسومات.
Region
مُعرّف في <X11/Xutil.h> و<X11/Xregion.h>، Region هو مجموعة اعتباطية من مواقع البكسل التي يمكن لـ Xlib معالجتها. داخليًا، يُمثَّل Region باتحاد عدد اعتباطي من المستطيلات. يحتفظ Xlib بمستطيل يخبره بمدى هذا الاتحاد.
Visual
مُعرّف في <X11/Xutil.h>، يحتوي هذا البنية على معلومات حول تعيين الألوان المحتمل.

من XRender...

أنواع بيانات مكتبة امتداد العرض X Render مستوحاة من Xlib، لكنها تفتقر إلى الوثائق المقابلة. راجع ملف الترويسة للتفاصيل:

<X11/extensions/Xrender.h>

أو اقرأ الشيفرة المصدرية (التي توفر مخططًا عامًا لـ libXrender).

يستخدم Xft هذه الأسماء: Picture، وXGlyphInfo، وXRenderColor.

من Fontconfig...

تتبع أنواع بيانات Fontconfig اصطلاح تسمية، حيث تُسبق جميع الأسماء بـ “Fc” وهي موثقة في مرجع مطوري Fontconfig.

يستخدم Xft هذه الأسماء من Fontconfig: FcBool، وFcChar8، وFcChar16، وFcChar32، وFcCharSet، وFcEndian، وFcFontSet، وFcPattern، وFcResult.

من FreeType 2...

تتبع أنواع بيانات FreeType 2 اصطلاح تسمية، حيث تُسبق جميع الأسماء بـ “FT_” وهي موثقة في مرجع واجهة برمجة تطبيقات FreeType.

يستخدم Xft هذه الأسماء: FT_Face، وFT_Library، وFT_UInt.

يستخدم FreeType 2 النوع FT_UInt (عدد صحيح غير مُوقّع) لتمثيل المحارف.

XftFont


typedef struct _XftFont {
	int	 ascent;
	int	 descent;
	int	 height;
	int	 max_advance_width;
	FcCharSet	*charset;
	FcPattern	*pattern;
} XftFont;

XftFont هو بنية البيانات الأساسية التي تهم المبرمجين الذين يستخدمون Xft؛ فهو يحتوي على مقاييس الخط العامة ومؤشرات إلى مجموعة محارف Fontconfig والنمط المرتبط بالخط.

تُملأ كائنات XftFont بأي من XftFontOpen()، أو XftFontOpenName()، أو XftFontOpenXlfd()، أو XftFontOpenInfo()، أو XftFontOpenPattern(). تُستخدم XftFontCopy() لنسخ كائنات XftFont، وتُستخدم XftFontClose() لتعليم كائن XftFont على أنه غير مستخدم. تُخصَّص كائنات XftFont داخليًا، وتُعدّ مرجعيًا، وتُحرَّر بواسطة Xft؛ لا يحتاج المبرمج عادةً إلى تخصيص أو تحرير مساحة تخزين لها.

تستخدم XftDrawGlyphs()، وعائلة XftDrawString*()، وXftDrawCharSpec()، وXftDrawGlyphSpec() كائنات XftFont لعرض النص إلى كائن XftDraw، والذي قد يتوافق مع إما عنصر قابل للرسم أساسي في X أو عنصر قابل للرسم في امتداد العرض X Rendering.

تُستخدم XftGlyphExtents() وعائلة XftTextExtents*() لتحديد الأبعاد (الأبعاد القصوى) لـ XftFont.

يمكن تحديد تغطية المحارف أو الأحرف لـ XftFont باستخدام XftFontCheckGlyph() أو XftCharExists(). تُرجع XftCharIndex() فهرس الحرف الخاص بـ XftFont المقابل لنقطة ترميز يونيكود معينة.

تستخدم XftGlyphRender()، وXftGlyphSpecRender()، وXftCharSpecRender()، وعائلة XftTextRender*() كائنات XftFont للرسم في بنى Picture الخاصة بامتداد العرض X Rendering.

ملاحظة: توفر XftDrawGlyphs()، وعائلة XftDrawString*()، وXftDrawCharSpec()، وXftDrawGlyphSpec() وسيلة لعرض الخطوط مستقلة عن توفر امتداد العرض X Rendering على خادم X.

XftFontInfo

هو كائن معتم يخزّن معلومات حول خط. تُنشأ بنى XftFontInfo باستخدام XftFontInfoCreate()، وتُحرَّر باستخدام XftFontInfoDestroy()، وتُقارن باستخدام XftFontInfoEqual(). تُخصَّص كائنات XftFontInfo داخليًا وتُحرَّر بواسطة Xft؛ لا يحتاج المبرمج عادةً إلى تخصيص أو تحرير مساحة تخزين لها.

يرتبط كل بنية XftFontInfo قيد الاستخدام بمعرّف فريد، يمكن استرجاعه باستخدام XftFontInfoHash(). يمكن فتح XftFont بناءً على بيانات XftFontInfo باستخدام XftFontOpenInfo().

XftColor


typedef struct _XftColor {
	unsigned long	 pixel;
	XRenderColor	 color;
} XftColor;

يسمح كائن XftColor بعرض النص وعناصر أخرى بلون معين (أو أقرب تقريب يقدمه العرض المرئي X قيد الاستخدام).

تطلب XftColorAllocName() وXftColorAllocValue() تخصيص لون من خادم X (إذا لزم الأمر) وتهيّئ أعضاء XftColor. تُعلم XftColorFree() خادم X بتحرير اللون المخصص حاليًا لـ XftColor.

بمجرد تهيئة XftColor، يمكن استخدام XftDrawSrcPicture()، وXftDrawGlyphs()، وعائلة XftDrawString*()، وXftDrawCharSpec()، وXftDrawCharFontSpec()، وXftDrawGlyphSpec()، وXftDrawGlyphFontSpec()، وXftDrawRect() لرسم عناصر مختلفة باستخدام g it.

XftDraw

هو كائن معتم يحتفظ بالمعلومات المستخدمة للرسم إلى عنصر رسم X إما باستخدام البروتوكول الأساسي أو امتداد X Rendering.

يتم إنشاء كائنات XftDraw باستخدام أي من XftDrawCreate() (التي تربط XftDraw بعنصر رسم X موجود)، أو XftDrawCreateBitmap()، أو XftDrawCreateAlpha()، ويتم تدميرها باستخدام XftDrawDestroy(). يمكن تغيير عنصر رسم X المرتبط بـ XftDraw باستخدام XftDrawChange(). يتم تخصيص كائنات XftDraw وتحريرها داخليًا بواسطة Xft؛ لا يحتاج المبرمج عادةً إلى تخصيص أو تحرير مساحة تخزين لها.

يمكن الاستعلام عن خصائص Display وDrawable وColormap وVisual الخاصة بـ XftDraw باستخدام XftDrawDisplay() وXftDrawDrawable() وXftDrawColormap() وXftDrawVisual() على التوالي.

تستخدم عدة دوال كائنات XftDraw: XftDrawCharFontSpec() وXftDrawCharSpec() وXftDrawGlyphFontSpec() وXftDrawGlyphSpec() وXftDrawGlyphs() وXftDrawRect() وXftDrawSetClip() وXftDrawSetClipRectangles() وXftDrawSetSubwindowMode() وعائلة XftDrawString*().

يتم إرجاع Picture الخاص بامتداد X Rendering المرتبط بـ XftDraw بواسطة XftDrawPicture() وXftDrawSrcPicture(). يتم استخدامه بواسطة XftCharFontSpecRender() وXftCharSpecRender() وXftGlyphFontSpecRender() وXftGlyphRender() وXftGlyphSpecRender() وعائلة XftTextRender*().

الدوال

التهيئة

التطبيق النموذجي الذي يستخدم Xft لا يهيئ المكتبة صراحةً. يتم ذلك عادةً كأثر جانبي لفتح خط.

عندما يتم تهيئة Xft، فإنه يجمع معلومات حول العرض ويخزن بعضًا من تلك المعلومات في نمط Fontconfig (وهو أساسًا مجموعة من الخصائص ذات قيم مُصنّفة). يمكن للتطبيق المستدعي تعديل ذلك النمط لتغيير سلوك المكتبة.

ليست كل المعلومات المجمعة مخزنة في نمط. يتم تخزين الباقي في هياكل بيانات داخلية. يجعل Xft بعضًا من ذلك متاحًا للتطبيق عبر الدوال.

Bool XftDefaultHasRender (
	Display	*dpy);

الحصول على معلومات حول العرض dpy إن لم يتم ذلك بالفعل، وإرجاع true إذا وجد Xft أن العرض يدعم امتداد X Render، وإذا كان قادرًا على إيجاد XRenderPictFormat مناسب (نوع بيانات X Render الذي يماثل Visual في Xlib) على العرض.

Bool XftDefaultSet (
	Display	*dpy,
	FcPattern	*defaults);

الحصول على معلومات حول العرض dpy إن لم يتم ذلك بالفعل، وتعيين نمط Fontconfig الذي يحمل الخصائص الافتراضية التي سيستخدمها Xft لهذا العرض.

يستخدم Xft تلك الخصائص في البداية للحصول على هذه الحدود:
XFT_MAX_GLYPH_MEMORY
(maxglyphmemory). هذا هو الحد الأقصى لمقدار ذاكرة الحروف الرسومية لجميع الخطوط المستخدمة بواسطة Xft (الافتراضي: 4*1024*1024).
XFT_MAX_UNREF_FONTS
(maxunreffonts). هذا هو الحد الأقصى لعدد الخطوط غير المُشار إليها (الافتراضي: 16).
XFT_TRACK_MEM_USAGE
(trackmemusage). عند true، يتتبع Xft استخدام ذاكرة الحروف الرسومية لتحسين الأداء عند تحديد أيها يجب تفريغه عند الوصول إلى الحد الأقصى لمقدار ذاكرة الحروف الرسومية (الافتراضي: false).
يستخدم Xft أيضًا هذه الخصائص الافتراضية في XftDefaultSubstitute().
void XftDefaultSubstitute (
	Display	*dpy,
	int	 screen,
	FcPattern	*pattern);

يملأ Xft الخصائص المفقودة في pattern المحدد باستخدام الخصائص الافتراضية للعرض المحدد dpy، على سبيل المثال، كما تم تعيينها في XftDefaultSet().

تستخدم تطبيقات Xft النموذجية هذه الدالة لمساعدة Fontconfig في اختيار خط مناسب. يتم استبدال هذه الخصائص قبل استدعاء FcDefaultSubstitute():
FC_ANTIALIAS
True إذا كان يجب على FreeType استخدام التنعيم (الافتراضي: False). (الافتراضي: True).
FC_AUTOHINT
True إذا كان يجب على FreeType استخدام التلميح التلقائي (الافتراضي: False).
FC_DPI
النقاط/البوصة المستخدمة للدقة (الافتراضي: محسوبة من ارتفاع العرض).
FC_EMBOLDEN
True إذا كان يجب استخدام FT_GlyphSlot_Embolden() لتثخين خط (الافتراضي: False).
FC_HINTING
True إذا كان يجب استخدام التلميح عند ملء الخصائص لفتح خط (الافتراضي: True).
FC_HINT_STYLE
نمط التلميح المستخدم عند ملء الخصائص لفتح خط (الافتراضي: FC_HINT_FULL).
FC_LCD_FILTER
المعامل الذي يتم تمريره إلى FT_Library_SetLcdFilter() عند تحميل الحروف الرسومية (الافتراضي: FC_LCD_DEFAULT).
FC_MINSPACE
قيمة المسافة الدنيا المستخدمة عند ملء الخصائص لفتح خط (الافتراضي: False).
FC_RGBA
قيمة RGBA المستخدمة عند ملء الخصائص لفتح خط (الافتراضي: محسوبة باستدعاء XRenderQuerySubpixelOrder()).
FC_SCALE
المقياس المستخدم في Fontconfig (الافتراضي: 1.0).
XFT_MAX_GLYPH_MEMORY
الحد الأقصى للذاكرة لخط واحد (الافتراضي: 1024*1024).
XFT_RENDER
True إذا كان العرض يدعم امتداد X Render (الافتراضي: النتيجة من XftDefaultHasRender()).
FcBool XftInit (
	_Xconst char	*config);

تهيئ مكتبة Fontconfig (باستدعاء FcInit()).

المعامل config غير مستخدم.
لا يقوم Xft بإلغاء تهيئة مكتبة Fontconfig عند الانتهاء.
FcBool XftInitFtLibrary (void);

    
تهيئ مكتبة FreeType (باستدعاء FT_Init_FreeType() لإنشاء كائن مكتبة) إذا لم تكن قد تمت تهيئتها بالفعل. هذا مطلوب قبل استخدام مكتبة FreeType لقراءة بيانات الخط من ملف.

  





  

  
يستدعي Xft الدالة XftInitFtLibrary() داخليًا عبر XftFontInfoCreate() وXftFontOpenPattern().

  

  
لا يتخلص Xft من كائن المكتبة (على سبيل المثال، باستخدام FT_Done_FreeType()) عند الانتهاء.





  
int XftGetVersion (void);

فتح ومطابقة الخطوط



  
XftFont *XftFontOpen (

  

    
	Display	*dpy,
	int	 screen,
	...);


    
تأخذ XftFontOpen قائمة من ثلاثيات عناصر النمط بالشكل field، type، value (منتهية بـ NULL)، وتطابق ذلك النمط مع الخطوط المتاحة، وتفتح الخط المطابق، وتضبط حجمه بشكل صحيح لرقم الشاشة screen على العرض dpy. إرجاع الخط المطابق، أو NULL إذا لم يتم العثور على تطابق.

  




مثال:
  




    font = XftFontOpen (dpy, screen,
                        XFT_FAMILY, XftTypeString, "charter",
                        XFT_SIZE, XftTypeDouble, 12.0,
                        NULL);




  

  
هذا يفتح خط "charter" بحجم 12 نقطة. يتم تحويل حجم النقطة تلقائيًا إلى حجم البكسل الصحيح بناءً على دقة الشاشة.





  
XftFont *XftFontOpenName (

  

    
	Display	*dpy,
	int	 screen,
	_Xconst char	*name);


    
تتصرف XftFontOpenName بكما يفعل XftFontOpen،
        باستثناء أنه يأخذ سلسلة نمط Fontconfig (والتي تُمرَّر إلى
        دالة FcNameParse() في مكتبة Fontconfig).

  

  
XftFont *XftFontOpenXlfd (

  

    
	Display	*dpy,
	int	 screen,
	_Xconst char	*xlfd)


    
تتصرف XftFontOpenXlfd كما تفعل XftFontOpen،
        باستثناء أنها تأخذ سلسلة تحتوي على وصف خط منطقي X
        (XLFD)، وتستخدم دالة XftXlfdParse().

  

  
FcPattern *XftFontMatch (

  

    
	Display	*dpy,
	int	 screen,
	_Xconst FcPattern	*pattern,
	FcResult	*result);


    
تُستخدم أيضاً داخلياً بواسطة دوال XftFontOpen*،
        ويمكن أيضاً استخدام XftFontMatch مباشرةً لتحديد
        نمط خط Fontconfig الناتج عن طلب فتح خط Xft.

  

  
FcPattern *XftXlfdParse (

  

    
	_Xconst char	*xlfd_orig,
	Bool	ignore_scalable,
	Bool	complete);


    
تحلل XftXlfdParse المعامل xlfd_orig
        وفقاً لوثيقة X Logical Font Description Conventions،
        لكنها تتجاهل بعض الحقول: setwidth_name،
        add_style_name، spacing، وaverage_width.

  





  

  
تُنشئ XftXlfdParse نمط Fontconfig، وتضبط الخاصية
      XFT_XLFD على قيمة xlfd_orig، وتُعيّن المعلومات
      المجمّعة إلى خصائص Fontconfig. يتم تجاهل الحقول الفارغة أو حقول “*”:







  
FC_FOUNDRY

  
من foundry.

  
FC_FAMILY

  
من family.

  
FC_WEIGHT

  
weight_name، مع القيمة الافتراضية FC_WEIGHT_MEDIUM.

  
FC_SLANT

  
من slant، مع القيمة الافتراضية FC_SLANT_ROMAN.

  
FC_SIZE

  
من point_size.

  
FC_PIXEL_SIZE

  
من pixel_size. إذا تم تعيين point_size، بالإضافة إلى
      resolution_x وresolution_y، فسيتم قياس القيمة
      لتحويل ارتفاع الخط إلى نقاط.

يتضمن ذلك عدة خطوات:

      XftFontCheckGlyph() وتحميل أي حروف رسومية مفقودة باستخدام
      XftFontLoadGlyphs().

      وهذا يحدد حجم الترميز، أي width.

      وتحسب الموضع الذي سيتم رسم كل حرف رسومي عنده.
    • 
  
  • ثم تنشئ Xft مصفوفة مكافئة من الحروف الرسومية بالتنسيق
      المتوقع من مكتبة X Render.

    •

التوافق


اعتبارًا من الإصدار 2 (مايو 2002)، أصبح Xft مستقرًا نسبيًا.
    ومن المتوقع أن يحافظ على التوافق المصدري والثنائي في الإصدارات
  المستقبلية.


يوفر Xft واجهة توافق مع إصداره الرئيسي السابق، Xft 1.x،
    الموصوف أدناه.




ملف   الترويسة Xft 1.x


#include <X11/Xft/XftCompat.h>






أنواع   بيانات Xft 1.x



  
XftPattern

  
يحمل مجموعة من الأسماء مع قوائم قيم مرتبطة بها؛ يشير كل اسم إلى
      خاصية من خصائص الخط. تُستخدم XftPatterns كمدخلات لكود
      المطابقة بالإضافة إلى حمل معلومات حول خطوط محددة.

  
XftFontSet

  
يحتوي على قائمة من XftPatterns. داخليًا، يستخدم Xft بنية
      البيانات هذه لحمل مجموعات من الخطوط. خارجيًا، يعيد Xft
      نتائج سرد الخطوط بهذا التنسيق.

  
XftObjectSet

  
يحمل مجموعة من الأسماء ويُستخدم لتحديد الحقول التي تُؤخذ من الخطوط
      وتُوضع في قائمة الأنماط المعادة عند سرد الخطوط.
