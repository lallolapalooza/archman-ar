XftXlfdParse(3)

الحزمة: libxft

التراخيص: HPND-sell-variant

Xft(3) إصدار X 11 Xft(3)

الاسم

Xft - مكتبة واجهة X FreeType

الملخص

#include <X11/Xft/Xft.h>

الوصف

Xft هي مكتبة بسيطة ترسم النصوص والرسومات:

  • باستخدام المعلومات المقدمة من مكتبة Fontconfig،
  • تقوم Xft بتحويل صور الخطوط باستخدام مرسمة FreeType، و
  • تعرض بيانات الخط المحولة باستخدام امتداد X للتقديم.

صفحة الدليل هذه لا تخدش سطح هذه المكتبة.

الثوابت

XFT_MAJOR
هو رقم الإصدار الرئيسي لـ Xft.
XFT_MINOR
هو رقم الإصدار الفرعي لـ Xft.
XFT_REVISION
هو رقم المراجعة لـ Xft.
XFT_VERSION
هو XFT_MAJOR مضروبًا في 10000 (عشرة آلاف)، زائد XFT_MINOR مضروبًا في 100، زائد XFT_REVISION.
XftVersion
هو اسم مستعار لـ XFT_VERSION.

يوضح المثال التالي كيف يمكن استخدام ثوابت إصدار Xft: 

    #if (XFT_VERSION >= 20107)
    (void) puts("الإصدار 2.1.7 أو أحدث من مكتبة Xft في"
                " الاستخدام.");
    #else
    (void) printf("إصدار غير كافٍ من Xft (%d.%d.%d) مثبت؛"
                  " نحتاج إلى الإصدار 2.1.7 على الأقل.\n", XFT_MAJOR,
                  XFT_MINOR,
                  XFT_REVISION);
    #endif

أنواع البيانات

تتبع أنواع بيانات Xft اصطلاح تسمية، حيث تسبق جميع الأسماء بـ "Xft"

من Xlib...

لا تتبع أنواع بيانات Xlib اصطلاح تسمية. يتم توثيقها في Xlib - واجهة لغة C.

تستخدم Xft هذه الأسماء: Bool, Colormap, Display, Drawable, Pixmap, Region, Visual, و XRectangle.

بعض أنواع البيانات ذات أهمية خاصة لـ Xft:

Drawable
معلن عنه في <X11/X.h>، وهو معرف مورد X، على سبيل المثال، نافذة مرتبطة بعرض. يمكن استخدام كل من النوافذ وصور الرسوم (pixmaps) كمصادر ووجهات في عمليات الرسوم. تُعرف هذه النوافذ وصور الرسوم بشكل جماعي باسم "drawables". ومع ذلك، لا يمكن استخدام نافذة InputOnly كمصدر أو وجهة في عملية رسومات.
Region
معلن عنه في <X11/Xutil.h> و <X11/Xregion.h>، Region هي مجموعة عشوائية من مواقع البكسل التي يمكن لـ Xlib التلاعب بها. داخليًا، يتم تمثيل Region باتحاد عدد عشوائي من المستطيلات. تحتفظ Xlib بمستطيل يخبرنا بمدى هذا الاتحاد.
Visual
معلن عنه في <X11/Xutil.h>، يحتوي هذا الهيكل على معلومات حول تعيين الألوان الممكنة.

من XRender...

مستوحاة من مكتبة امتداد X للتقديم (X Render Extension) من Xlib، ولكنها تفتقر إلى الوثائق المقابلة. ارجع إلى ملف الرأس للحصول على التفاصيل:

<X11/extensions/Xrender.h>

أو اقرأ الكود المصدري (الذي يوفر مخططًا لمكتبة libXrender).

تستخدم Xft هذه الأسماء: Picture, XGlyphInfo, XRenderColor.

من Fontconfig...

تتبع أنواع بيانات Fontconfig اصطلاح تسمية، حيث تسبق جميع الأسماء بـ "Fc" والتي تم توثيقها في مرجع مطوري Fontconfig.

تستخدم Xft هذه الأسماء من Fontconfig: FcBool, FcChar8, FcChar16, FcChar32, FcCharSet, FcEndian, FcFontSet, FcPattern, و FcResult.

من FreeType 2...

تتبع أنواع بيانات FreeType 2 اصطلاح تسمية، حيث تسبق جميع الأسماء بـ "FT_" والتي تم توثيقها في مرجع واجهة برمجة تطبيقات FreeType.

تستخدم Xft هذه الأسماء: FT_Face, FT_Library, و FT_UInt.

تستخدم FreeType 2 FT_UInt (عدد صحيح غير موقع) لتمثيل صور glyphs.

XftFont


typedef struct _XftFont {
	int	 ascent;
	int	 descent;
	int	 height;
	int	 max_advance_width;
	FcCharSet	*charset;
	FcPattern	*pattern;
} XftFont;

XftFont هو هيكل البيانات الأساسي ذي الأهمية لبرامج استخدام Xft؛ فهو يحتوي على مقاييس الخط العامة ومؤشرات إلى مجموعة أحرف Fontconfig والنمط المرتبطين بالخط.

يتم ملء XftFonts باستخدام أي من XftFontOpen(), XftFontOpenName(), XftFontOpenXlfd(), XftFontOpenInfo(), أو XftFontOpenPattern(). يتم استخدام XftFontCopy() لتكرار XftFonts، ويتم استخدام XftFontClose() لوضع علامة على XftFont بأنه غير مستخدم. يتم تخصيص XftFonts داخليًا، و عد مرجعها، وتحريرها بواسطة Xft؛ لا يحتاج المبرمج عادةً إلى تخصيص أو تحرير مساحة لها.

تستخدم XftDrawGlyphs()، وعائلة XftDrawString*()، و XftDrawCharSpec()، و XftDrawGlyphSpec() XftFonts لرسم النصوص إلى كائن XftDraw، والذي قد يتوافق مع رسم X أساسي أو رسم امتداد X للتقديم.

تستخدم XftGlyphExtents() وعائلة XftTextExtents*() لتحديد حدود (الأبعاد القصوى) لـ XftFont.

يمكن تحديد تغطية صور glyphs أو أحرف XftFont باستخدام XftFontCheckGlyph() أو XftCharExists(). ترجع XftCharIndex() فهرس الحرف الخاص بـ XftFont المقابل لنقطة رمز Unicode معينة.

تستخدم XftGlyphRender()، و XftGlyphSpecRender()، و XftCharSpecRender()، وعائلة XftTextRender*() XftFonts للرسم في هياكل Picture لامتداد X للتقديم.

ملاحظة: توفر XftDrawGlyphs()، وعائلة XftDrawString*()، و XftDrawCharSpec()، و XftDrawGlyphSpec() وسيلة لتقديم الخطوط مستقلة عن توفر امتداد X للتقديم على خادم X.

XftFontInfo

هو كائن غير شفاف يخزن معلومات حول خط. يتم إنشاء هياكل XftFontInfo باستخدام XftFontInfoCreate()، وتحريرها باستخدام XftFontInfoDestroy()، ومقارنتها باستخدام XftFontInfoEqual(). يتم تخصيص XftFontInfo داخليًا وتحريرها بواسطة Xft؛ لا يحتاج المبرمج عادةً إلى تخصيص أو تحرير مساحة لها.

كل هيكل XftFontInfo قيد الاستخدام مرتبط بمعرف فريد، والذي يمكن استرداده باستخدام XftFontInfoHash(). يمكن فتح XftFont بناءً على بيانات XftFontInfo باستخدام XftFontOpenInfo().

XftColor


typedef struct _XftColor {
	unsigned long	 pixel;
	XRenderColor	 color;
} XftColor;

يسمح كائن XftColor برسم النصوص والعناصر الأخرى بلون معين (أو أقرب تقريب يقدمه تصور X المستخدم).

تقوم XftColorAllocName() و XftColorAllocValue() بطلب تخصيص لون من خادم X (إذا لزم الأمر) وتهيئة أعضاء XftColor. تطلب XftColorFree() من خادم X تحرير اللون المخصص حاليا لـ XftColor.

بمجرد تهيئة XftColor، يمكن استخدام XftDrawSrcPicture()، و XftDrawGlyphs()، وعائلة XftDrawString*()، و XftDrawCharSpec()، و XftDrawCharFontSpec()، و XftDrawGlyphSpec()، و XftDrawGlyphFontSpec()، و XftDrawRect() لرسم عناصر مختلفة باستخدامها.

هذا الموقع غير مرتبط بـ Arch Linux ولا تابع له بأي شكل من الأشكال. المحتوى هو ترجمة فقط لصفحات دليل Arch المنشورة على man.archlinux.org.

ftFontOpen، باستثناء أنها تأخذ سلسلة نمط Fontconfig (والتي يتم تمريرها إلى وظيفة FcNameParse() الخاصة بمكتبة Fontconfig).

XftFont *XftFontOpenXlfd (
	Display	*dpy,
	int	 screen,
	_Xconst char	*xlfd)

XftFontOpenXlfd يعمل كما يعمل XftFontOpen، باستثناء أنه يأخذ سلسلة تحتوي على وصف خط X المنطقي (XLFD)، ويستخدم وظيفة XftXlfdParse().

FcPattern *XftFontMatch (
	Display	*dpy,
	int	 screen,
	_Xconst FcPattern	*pattern,
	FcResult	*result);

تُستخدم أيضًا داخليًا بواسطة وظائف XftFontOpen*، يمكن استخدام XftFontMatch مباشرة لتحديد نمط خط Fontconfig الناتج عن طلب فتح خط Xft.

FcPattern *XftXlfdParse (
	_Xconst char	*xlfd_orig,
	Bool	ignore_scalable,
	Bool	complete);

تقوم XftXlfdParse بتحليل المعلمة xlfd_orig وفقًا لمستند *اتفاقيات وصف خط X المنطقي*، ولكن تتجاهل بعض الحقول: setwidth_name، add_style_name، spacing، و average_width.

تقوم XftXlfdParse بإنشاء نمط Fontconfig، مع تعيين الخاصية XFT_XLFD إلى قيمة xlfd_orig، وتقوم بتعيين المعلومات التي تم جمعها إلى خصائص Fontconfig. يتم تجاهل الحقول الفارغة أو التي تحتوي على "*" :
FC_FOUNDRY
من foundry.
FC_FAMILY
من family.
FC_WEIGHT
weight_name، الافتراضي هو FC_WEIGHT_MEDIUM.
FC_SLANT
من slant، الافتراضي هو FC_SLANT_ROMAN.
FC_SIZE
من point_size.
FC_PIXEL_SIZE
من pixel_size. إذا تم تعيين point_size، وكذلك resolution_x و resolution_y، فإن القيمة يتم قياسها لتحويل ارتفاع الخط إلى نقاط.

تحديد امتدادات النص

يوفر Xft عدة وظائف لتحديد الارتفاع والعرض المطلوبين لعرض سلسلة نصية. بعد حساب الإزاحة، في الحالات التي سيتم فيها إزاحة السلسلة لأعلى، لأسفل، لليسار أو لليمين، يشار إلى هذه الأرقام باسم امتدادات النص.

void XftTextExtents8 (
	Display	*dpy,
	XftFont	*font,
	_Xconst FcChar8	*string,
	int	 len,
	XGlyphInfo	*extents);

تحسب XftTextExtents8 امتدادات البكسل على الشاشة dpy لما لا يزيد عن len حرفًا من سلسلة تتكون من أحرف ثمانية البتات عند رسمها بالخط font، وتخزينها في extents.

void XftTextExtents16 (
	Display	*dpy,
	XftFont	*font,
	_Xconst FcChar16	*string,
	int	 len,
	XGlyphInfo	*extents);

تحسب XftTextExtents16 امتدادات البكسل على الشاشة dpy لما لا يزيد عن len حرفًا من سلسلة تتكون من أحرف ستة عشر بت عند رسمها بالخط font، وتخزينها في extents.

void XftTextExtents32 (
	Display	*dpy,
	XftFont	*font,
	_Xconst FcChar32	*string,
	int	 len,
	XGlyphInfo	*extents);

تحسب XftTextExtents32 امتدادات البكسل على الشاشة dpy لما لا يزيد عن len حرفًا من سلسلة تتكون من أحرف اثنان وثلاثون بت عند رسمها بالخط font، وتخزينها في extents.

void XftTextExtentsUtf8 (
	Display	*dpy,
	XftFont	*font,
	_Xconst FcChar8	*string,
	int	 len,
	XGlyphInfo	*extents);

تحسب XftTextExtentsUtf8 امتدادات البكسل على الشاشة dpy لما لا يزيد عن len بايت من سلسلة مشفرة بـ UTF-8 عند رسمها بالخط font، وتخزينها في extents.

void XftTextExtentsUtf16 (
	Display	*dpy,
	XftFont	*font,
	_Xconst FcChar8	*string,
	FcEndian	 endian,
	int	 len,
	XGlyphInfo	*extents);

تحسب XftTextExtentsUtf16 امتدادات البكسل على الشاشة dpy لما لا يزيد عن len بايت من سلسلة مشفرة بـ UTF-16LE أو UTF-16BE عند رسمها بالخط font، وتخزينها في extents. يجب تحديد ترتيب البايتات للسلسلة في endian.

void XftGlyphExtents (
	Display	*dpy,
	XftFont	*font,
	_Xconst FT_UInt	*glyphs,
	int	 nglyphs,
	XGlyphInfo	*extents);

تُستخدم أيضًا داخليًا بواسطة وظائف XftTextExtents*()، تحسب XftGlyphExtents امتدادات البكسل على الشاشة dpy لما لا يزيد عن nglyphs في المصفوفة glyphs المرسومة بالخط font، وتخزينها في extents.

إذا كانت أي من glyphs مفقودة (يتم تحديدها عن طريق التحقق باستخدام XftFontCheckGlyph())، فإن الإدخال المقابل في extents يتم ملؤه بالأصفار.

إدارة XftColor

Bool XftColorAllocName (
	Display	*dpy,
	_Xconst Visual	*visual,
	Colormap	 cmap,
	_Xconst char	*name,
	XftColor	*result);

استخدم XAllocNamedColor() للبحث عن اللون المسمى name للشاشة المرتبطة بـ cmap.

  • إذا أعادت XAllocNamedColor() قيمة غير صفرية، فإن XftColorAllocName() تملأ حقل البكسل الناتج لـ XftColor بأقرب لون مدعوم من الشاشة، بالإضافة إلى الحقول الحمراء والخضراء والزرقاء الدقيقة من قاعدة البيانات، وتعيد True.
  • إذا أعادت XAllocNamedColor() قيمة صفر، فإن XftColorAllocName() تعيد False، ولا تحدث XftColor المشار إليها بواسطة result.
المعلمة visual غير مستخدمة.
Bool XftColorAllocValue (
	Display	*dpy,
	Visual	*visual,
	Colormap	 cmap,
	_Xconst XRenderColor	*color,
	XftColor	*result);

تخصيص قيمة لون:

  • إذا كان فئة visual هي TrueColor، فإن XftColorAllocValue() تعين حقل pixel في XftColor المشار إليه بواسطة result باستخدام الحقول الحمراء والخضراء والزرقاء من المعلمة color.
  • إذا لم تكن فئة visual هي TrueColor، فإن XftColorAllocValue() تستدعي XAllocColor() لتخصيص إدخال في cmap، والذي يعيد قيمة البكسل للون الأقرب إلى عناصر RGB المحددة المدعومة بواسطة الأجهزة.
إذا نجحت XAllocColor()، تقوم XftColorAllocValue() بتخزين قيمة البكسل هذه في XRenderColor المشار إليها بواسطة result وتعيد True.
إذا فشلت XAllocColor()، فإن XftColorAllocValue() تعيد False ولا تعدل النتيجة.
void XftColorFree (
	Display	*dpy,
	Visual	*visual,
	Colormap	 cmap,
	XftColor	*color);

إذا لم تكن فئة visual هي TrueColor، فإن Xft تستدعي XFreeColors() لتحرير الإدخال من cmap التي تم تخصيص قيمة البكسل الخاصة بها في المعلمة color بواسطة XftColorAllocName().
