XftTextExtents32(3)

الحزمة: libxft

التراخيص: HPND-sell-variant

Xft(3) إصدار X 11 Xft(3)

الاسم

Xft - مكتبة واجهة X FreeType

المُلخَّص

#include <X11/Xft/Xft.h>

الوصف

Xft هي مكتبة بسيطة ترسم النصوص والرسومات:

  • باستخدام المعلومات المقدمة من مكتبة Fontconfig،
  • تقوم Xft بتحويل رموز الخطوط باستخدام مُصَفِّي FreeType، و
  • تعرض بيانات الخطوط المحوّلة باستخدام امتداد X Rendering Extension.

هذه الصفحة الدليلية بالكاد تخدش سطح هذه المكتبة.

الثوابت

XFT_MAJOR
هو رقم الإصدار الرئيسي لـ Xft.
XFT_MINOR
هو رقم الإصدار الفرعي لـ Xft.
XFT_REVISION
هو رقم المراجعة لـ Xft.
XFT_VERSION
هو XFT_MAJOR مضروبًا في 10000 (عشرة آلاف)، زائد XFT_MINOR مضروبًا في 100، زائد XFT_REVISION.
XftVersion
هو اسم مستعار لـ XFT_VERSION.

يوضح المثال التالي كيف يمكن استخدام ثوابت إصدار Xft: 

    #if (XFT_VERSION >= 20107)
    (void) puts("الإصدار 2.1.7 أو أحدث من مكتبة Xft قيد الاستخدام.");
    #else
    (void) printf("إصدار غير كافٍ من Xft (%d.%d.%d) مثبت؛"
                  " يلزم إصدار 2.1.7 على الأقل.\n", XFT_MAJOR,
                  XFT_MINOR,
                  XFT_REVISION);
    #endif

أنواع البيانات

تتبع أنواع بيانات Xft اتفاقية تسمية، حيث تسبق جميع الأسماء بـ "Xft"

من Xlib...

لا تتبع أنواع بيانات Xlib اتفاقية تسمية. يتم توثيقها في Xlib - واجهة لغة C.

تستخدم Xft هذه الأسماء: Bool، Colormap، Display، Drawable، Pixmap، Region، Visual، و XRectangle.

بعض أنواع البيانات ذات أهمية خاصة لـ Xft:

Drawable
معرّف في <X11/X.h>، هذا معرف مورد X، على سبيل المثال، نافذة مرتبطة بشاشة. يمكن استخدام كل من النوافذ والصور (pixmaps) كمصادر ووجهات في عمليات الرسومات. تُعرف هذه النوافذ والصور مجتمعة باسم drawables. ومع ذلك، لا يمكن استخدام نافذة InputOnly كمصدر أو وجهة في عملية رسومات.
Region
معرّف في <X11/Xutil.h> و <X11/Xregion.h>، Region هي مجموعة اعتباطية من مواقع البكسل التي يمكن لـ Xlib معالجتها. داخليًا، يتم تمثيل Region باتحاد عدد اعتباطي من المستطيلات. تحتفظ Xlib بمستطيل يبلغها بنطاق هذا الاتحاد.
Visual
معرّف في <X11/Xutil.h>، يحتوي هذا الهيكل على معلومات حول تعيين الألوان الممكن.

من XRender...

مستوحاة أنواع بيانات مكتبة X Render Extension من Xlib، ولكنها تفتقر إلى التوثيق المقابل. ارجع إلى ملف الرأس للحصول على التفاصيل:

<X11/extensions/Xrender.h>

أو اقرأ الكود المصدري (الذي يوفر مخططًا عامًا لمكتبة libXrender).

تستخدم Xft هذه الأسماء: Picture، XGlyphInfo، XRenderColor.

من Fontconfig...

تتبع أنواع بيانات Fontconfig اتفاقية تسمية، حيث تسبق جميع الأسماء بـ "Fc" وهي موثقة في مرجع مطوري Fontconfig.

تستخدم Xft أسماء Fontconfig هذه: FcBool، FcChar8، FcChar16، FcChar32، FcCharSet، FcEndian، FcFontSet، FcPattern، و FcResult.

من FreeType 2...

تتبع أنواع بيانات FreeType 2 اتفاقية تسمية، حيث تسبق جميع الأسماء بـ "FT_" وهي موثقة في مرجع واجهة برمجة تطبيقات FreeType.

تستخدم Xft هذه الأسماء: FT_Face، FT_Library، و FT_UInt.

يستخدم FreeType 2 FT_UInt (عدد صحيح غير مُشار إليه) لتمثيل *رموز (glyphs)*.

XftFont


typedef struct _XftFont {
	int	 ascent;
	int	 descent;
	int	 height;
	int	 max_advance_width;
	FcCharSet	*charset;
	FcPattern	*pattern;
} XftFont;

XftFont هو الهيكل البياني الرئيسي ذو الأهمية لبرمجيي Xft؛ فهو يحتوي على مقاييس الخطوط العامة ومؤشرات إلى مجموعة أحرف Fontconfig والنمط المرتبط بالخط.

يتم ملء XftFonts باستخدام أي من XftFontOpen()، XftFontOpenName()، XftFontOpenXlfd()، XftFontOpenInfo()، أو XftFontOpenPattern(). يتم استخدام XftFontCopy() لإنشاء نسخة مكررة من XftFonts، ويتم استخدام XftFontClose() لوضع علامة على XftFont على أنه غير مستخدم. يتم تخصيص XftFonts داخليًا، وإدارة عدد المراجع، وتحريرها بواسطة Xft؛ لا يحتاج المبرمج عادةً إلى تخصيص أو تحرير مساحة لها.

تستخدم XftDrawGlyphs()، وعائلة XftDrawString*()، و XftDrawCharSpec()، و XftDrawGlyphSpec() XftFonts لرسم النصوص إلى كائن XftDraw، والذي قد يتوافق مع كل من كائن رسومات X الأساسي أو كائن رسومات امتداد X Rendering Extension.

تُستخدم XftGlyphExtents() وعائلة XftTextExtents*() لتحديد امتدادات (الأبعاد القصوى) لـ XftFont.

يمكن تحديد تغطية الرموز (glyph) أو الأحرف لـ XftFont باستخدام XftFontCheckGlyph() أو XftCharExists(). تُرجع XftCharIndex() فهرس الحرف الخاص بـ XftFont المقابل لنقطة ترميز Unicode معطاة.

تستخدم XftGlyphRender()، و XftGlyphSpecRender()، و XftCharSpecRender()، وعائلة XftTextRender*() XftFonts للرسم في هياكل Picture الخاصة بـ X Rendering Extension.

ملاحظة: توفر XftDrawGlyphs()، وعائلة XftDrawString*()، و XftDrawCharSpec()، و XftDrawGlyphSpec() وسيلة لرسم الخطوط تكون مستقلة عن توفر امتداد X Rendering Extension على خادم X.

XftFontInfo

هو كائن غير شفاف يخزن معلومات حول خط. يتم إنشاء هياكل XftFontInfo باستخدام XftFontInfoCreate()، وتحريرها باستخدام XftFontInfoDestroy()، ومقارنتها باستخدام XftFontInfoEqual(). يتم تخصيص كائنات XftFontInfo وتحريرها داخليًا بواسطة Xft؛ لا يحتاج المبرمج عادةً إلى تخصيص أو تحرير مساحة لها.

كل هيكل XftFontInfo قيد الاستخدام مرتبط بمعرف فريد، يمكن استرجاعه باستخدام XftFontInfoHash(). يمكن فتح XftFont بناءً على بيانات XftFontInfo باستخدام XftFontOpenInfo().

XftColor


typedef struct _XftColor {
	unsigned long	 pixel;
	XRenderColor	 color;
} XftColor;

يسمح كائن XftColor برسم النصوص والعناصر الأخرى بلون معين (أو أقرب تقريب تقدمه رؤية X المستخدمة).

تقوم XftColorAllocName() و XftColorAllocValue() بطلب تخصيص لون من خادم X (إذا لزم الأمر) وتهيئة أعضاء XftColor. تُعلم XftColorFree() خادم X بتحرير اللون المخصص حاليًا لـ XftColor.

بمجرد تهيئة XftColor، يمكن استخدام XftDrawSrcPicture()، و XftDrawGlyphs()، وعائلة XftDrawString*()، و XftDrawCharSpec()، و XftDrawCharFontSpec()، و XftDrawGlyphSpec()، و XftDrawGlyphFontSpec()، و XftDrawRect() لرسم كائنات مختلفة باستخدام

);

يستخدم Xft Fontconfig لتحليل الاسم، ويمرر name إلى FcNameParse()، ويعيد النتيجة.

FcBool XftNameUnparse (
	FcPattern	*pat,
	char	*dest,
	int	 len);

مثل XfgNameParse()، يستخدم Xft Fontconfig. في هذه الحالة، يستخدم FcNameUnparse()، والذي يحول النمط pat مرة أخرى إلى سلسلة يمكن تحليلها. تخزن XftNameUnparse() النتيجة عبر مؤشر المتصل dest، ولكنها تتحقق أولاً مما إذا كانت ستناسب، باستخدام المعامل len. إذا كانت تناسب، يقوم Xft بنسخ السلسلة ويعيد FcTrue، وإلا فإنه يعيد FcFalse.

معالجة بيانات الخط

void XftFontLoadGlyphs (
	Display	*dpy,
	XftFont	*pub,
	FcBool	 need_bitmaps,
	_Xconst FT_UInt	*glyphs,
	int	 nglyph);

باستخدام FT_Load_Glyph()، قم بتحميل nglyphs لمؤشرات الأحرف (قيم Unicode) المدرجة في المصفوفة glyphs من الخط المعطى pub.

يتضمن تحميل الحرف أكثر من مجرد قراءة البيانات في الذاكرة. يستخدم Xft نمط Fontconfig المرتبط بالعرض dpy (على سبيل المثال، عبر XftDefaultSet()) لتحديد ما إذا كان سيتم استخدام مربع محيط، أو قص الأحرف ضمن تلك الحدود، أو تحجيم الأحرف، أو حساب مقاييس الخط، وإضافته إلى امتداد X Render باستخدام XRenderAddGlyphs().
void XftFontUnloadGlyphs (
	Display	*dpy,
	XftFont	*pub,
	_Xconst FT_UInt	*glyphs,
	int	 nglyph);

يتجاهل البيانات لما يصل إلى nglyph من الأحرف التي تم إدراج مؤشرات الأحرف الخاصة بها (قيم Unicode) في المصفوفة glyphs للخط المعطى pub. إذا تمت إضافة الأحرف إلى امتداد X Render، فإن Xft يزيلها باستخدام XRenderFreeGlyphs(). يتتبع Xft مقدار الذاكرة المستخدمة للأحرف، ويحدّث الاستخدام المرتبط بالعرض dpy.

FT_Face XftLockFace (
	XftFont	*pub);

إذا لم يتم إنشاء "وجه" FreeType 2 (تمثيل في الذاكرة لنوع خط معين بأسلوب معين) للخط pub، فقم بإنشاء واحد باستخدام FT_New_Face(). أقفال الوجه هي عدد مراجع تستخدمه Xft لضمان إنشاء وجه واحد فقط، وأنه يتم الاحتفاظ به حتى لا يعود الخط مستخدماً.

يتم استخدام أقفال الوجه مباشرة في XftCharIndex() و XftFontLoadGlyphs()، والتي بدورها تستخدم في العديد من دوال Xft.
تم تقديم أقفال الوجه في الإصدار 1 (أكتوبر 2002). اعتمد عدد قليل من التطبيقات، مثل Gdk/Gimp، على هذه الدوال. في الإصدار 2.1.9 (يونيو 2006)، تم الاحتفاظ بأقفال الوجه كجزء من واجهة برمجة التطبيقات العامة عند توفير تكوينات مكتبة مشتركة محسنة لإخفاء الرموز الخاصة.
void XftUnlockFace (
	XftFont	*pub);

يقلل عدد المراجع للوجه FreeType 2 المرتبط بالخط.

يتم إلغاء تخصيص كائنات FT_Face باستخدام FT_Done_Face(). يقوم Xft بذلك في XftFontInfoDestroy() وعند التنظيف عند الفشل في XftFontInfoCreate() و XftFontOpenPattern().

التصحيح

يقرأ Xft متغير البيئة XFT_DEBUG ويحوله إلى عدد صحيح. كل بت في القيمة الناتجة يخبر Xft بطباعة معلومات تصحيح الأخطاء إلى الإخراج القياسي:

1
فتح الخط
2
مطابقة وفتح خط إضافي (تفصيلي)
4
يعرض تفاصيل حول XRenderPictFormat الذي سيتم استخدامه.
8
يعرض السلسلة التي سترسمها XftDrawString8().
16
يعرض معرفات الخط التي تتم مطابقتها.
32
يعرض معلومات مفيدة حول الأحرف التي سيتم رسمها
64
يعرض تمثيل فن ASCII للأحرف.
128
يعرض تفاصيل حول إدارة ذاكرة التخزين المؤقت للأحرف
256
يعرض تفاصيل حول إدارة الذاكرة المؤقتة للأحرف
512
يعرض تقريرًا عن استخدام الذاكرة
1024
يعرض تفاصيل حول الإدارة الموسعة للذاكرة المؤقتة للأحرف

التوافق

اعتبارًا من الإصدار 2 (مايو 2002)، أصبح Xft مستقرًا نسبيًا. من المتوقع أن يحتفظ بالتوافق المصدر والثنائي في الإصدارات المستقبلية.

يوفر Xft واجهة توافق مع إصدارها الرئيسي السابق، Xft 1.x، الموضح أدناه.

ملف رأس Xft 1.x

#include <X11/Xft/XftCompat.h>

أنواع بيانات Xft 1.x

XftPattern
يحتوي على مجموعة من الأسماء مع قوائم قيم مرتبطة؛ يشير كل اسم إلى خاصية لخط. تستخدم XftPattern كمدخلات لرمز المطابقة وكذلك للاحتفاظ بمعلومات حول خطوط معينة.
XftFontSet
يحتوي على قائمة من XftPatterns. داخليًا، يستخدم Xft هيكل البيانات هذا للاحتفاظ بمجموعات من الخطوط. خارجيًا، يعيد Xft نتائج سرد الخطوط بهذا التنسيق.
XftObjectSet
يحتوي على مجموعة من الأسماء ويستخدم لتحديد أي حقول من الخطوط توضع في قائمة الأنماط المعادة عند سرد الخطوط.
