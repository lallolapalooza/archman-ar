XftNameParse(3)

الحزمة: libxft

التراخيص: HPND-sell-variant

Xft(3) X Version 11 Xft(3)

الاسم

Xft - مكتبة واجهة X FreeType

الملخص

#include <X11/Xft/Xft.h>

الوصف

Xft هي مكتبة بسيطة تقوم برسم النص والرسومات:

  • باستخدام المعلومات التي توفرها مكتبة Fontconfig،
  • تقوم Xft بتحويل رموز الخطوط باستخدام مولد الخطوط FreeType، و
  • تعرض بيانات الخط المحولة باستخدام امتداد X للتقديم (X Rendering Extension).

صفحة الدليل هذه بالكاد تخدش سطح هذه المكتبة.

الثوابت

XFT_MAJOR
هو رقم الإصدار الرئيسي لـ Xft.
XFT_MINOR
هو رقم الإصدار الفرعي لـ Xft.
XFT_REVISION
هو رقم المراجعة لـ Xft.
XFT_VERSION
هو XFT_MAJOR مضروبًا في 10000 (عشرة آلاف)، بالإضافة إلى XFT_MINOR مضروبًا في 100، بالإضافة إلى XFT_REVISION.
XftVersion
هو اسم مستعار لـ XFT_VERSION.

يوضح المثال التالي كيف يمكن استخدام ثوابت إصدار Xft: 

    #if (XFT_VERSION >= 20107)
    (void) puts("الإصدار 2.1.7 أو أحدث من مكتبة Xft قيد الاستخدام.");
    #else
    (void) printf("إصدار غير كافٍ من Xft (%d.%d.%d) مثبت؛
                  " نحتاج إلى إصدار 2.1.7 على الأقل.\n", XFT_MAJOR,
                  XFT_MINOR,
                  XFT_REVISION);
    #endif

أنواع البيانات

تتبع أنواع بيانات Xft اصطلاح تسمية، حيث يتم وضع البادئة "Xft" على جميع الأسماء.

من Xlib...

لا تتبع أنواع بيانات Xlib اصطلاح تسمية. وهي موثقة في Xlib - واجهة لغة C.

تستخدم Xft هذه الأسماء: Bool، Colormap، Display، Drawable، Pixmap، Region، Visual، و XRectangle.

بعض أنواع البيانات ذات أهمية خاصة لـ Xft:

Drawable
معرّف في <X11/X.h>، هذا هو معرف مورد X، على سبيل المثال، نافذة مرتبطة بشاشة. يمكن استخدام كل من النوافذ والصور (pixmaps) كمصادر ووجهات في عمليات الرسومات. تُعرف هذه النوافذ والصور مجتمعة باسم الـ "drawables". ومع ذلك، لا يمكن استخدام نافذة من نوع InputOnly كمصدر أو وجهة في عملية رسومات.
Region
معرّف في <X11/Xutil.h> و <X11/Xregion.h>، الـ Region هي مجموعة تعسفية من مواقع البكسل التي يمكن لـ Xlib معالجتها. داخليًا، يتم تمثيل الـ Region باتحاد عدد تعسفي من المستطيلات. تحتفظ Xlib بمستطيل يوضح مدى هذا الاتحاد.
Visual
معرّف في <X11/Xutil.h>، يحتوي هذا الهيكل على معلومات حول رسم الخرائط الممكنة للألوان.

من XRender...

تستلهم أنواع بيانات مكتبة امتداد X للتقديم (X Render Extension) من Xlib، ولكنها تفتقر إلى الوثائق المقابلة. ارجع إلى ملف الرأس للحصول على التفاصيل:

<X11/extensions/Xrender.h>

أو اقرأ الكود المصدري (الذي يوفر مخططًا لمكتبة libXrender).

تستخدم Xft هذه الأسماء: Picture، XGlyphInfo، XRenderColor.

من Fontconfig...

تتبع أنواع بيانات Fontconfig اصطلاح تسمية، حيث يتم وضع البادئة "Fc" على جميع الأسماء، وهي موثقة في مرجع مطوري Fontconfig.

تستخدم Xft أسماء Fontconfig هذه: FcBool، FcChar8، FcChar16، FcChar32، FcCharSet، FcEndian، FcFontSet، FcPattern، و FcResult.

من FreeType 2...

تتبع أنواع بيانات FreeType 2 اصطلاح تسمية، حيث يتم وضع البادئة "FT_" على جميع الأسماء، وهي موثقة في مرجع واجهة برمجة تطبيقات FreeType.

تستخدم Xft هذه الأسماء: FT_Face، FT_Library، و FT_UInt.

تستخدم FreeType 2 FT_UInt (عدد صحيح غير موقّع) لتمثيل رموز الخطوط.

XftFont


typedef struct _XftFont {
	int	 ascent;
	int	 descent;
	int	 height;
	int	 max_advance_width;
	FcCharSet	*charset;
	FcPattern	*pattern;
} XftFont;

الـ XftFont هو هيكل البيانات الأساسي ذو الأهمية للمبرمجين الذين يستخدمون Xft؛ فهو يحتوي على مقاييس خط عامة ومؤشرات إلى مجموعة أحرف Fontconfig والنمط المرتبط بالخط.

يتم ملء الـ XftFonts باستخدام أي من الوظائف: XftFontOpen()، XftFontOpenName()، XftFontOpenXlfd()، XftFontOpenInfo()، أو XftFontOpenPattern(). تُستخدم XftFontCopy() لنسخ الـ XftFonts، وتُستخدم XftFontClose() لوضع علامة على XftFont كغير مستخدم. يتم تخصيص الـ XftFonts داخليًا، ويتم عد مراجعها، وتحريرها بواسطة Xft؛ لا يحتاج المبرمج عادةً إلى تخصيص أو تحرير مساحة تخزين لها.

تستخدم الوظائف XftDrawGlyphs()، وعائلة الوظائف XftDrawString*()، و XftDrawCharSpec()، و XftDrawGlyphSpec() الـ XftFonts لرسم النص إلى كائن XftDraw، والذي قد يتوافق مع قابلية رسم X أساسية أو قابلية رسم امتداد X للتقديم.

تُستخدم XftGlyphExtents() وعائلة الوظائف XftTextExtents*() لتحديد الامتدادات (الأبعاد القصوى) لـ XftFont.

يمكن تحديد تغطية رموز الخطوط أو الأحرف لـ XftFont باستخدام XftFontCheckGlyph() أو XftCharExists(). تُرجع XftCharIndex() فهرس الحرف الخاص بـ XftFont المقابل لنقطة ترميز Unicode معينة.

تستخدم الوظائف XftGlyphRender()، و XftGlyphSpecRender()، و XftCharSpecRender()، وعائلة الوظائف XftTextRender*() الـ XftFonts للرسم في هياكل Picture الخاصة بامتداد X للتقديم.

ملاحظة: توفر الوظائف XftDrawGlyphs()، وعائلة الوظائف XftDrawString*()، و XftDrawCharSpec()، و XftDrawGlyphSpec() وسيلة لتقديم الخطوط تكون مستقلة عن توفر امتداد X للتقديم على خادم X.

XftFontInfo

هو كائن غير شفاف يخزن المعلومات حول خط. يتم إنشاء هياكل XftFontInfo باستخدام XftFontInfoCreate()، وتحريرها باستخدام XftFontInfoDestroy()، ومقارنتها باستخدام XftFontInfoEqual(). يتم تخصيص الـ XftFontInfo objects داخليًا وتحريرها بواسطة Xft؛ لا يحتاج المبرمج عادةً إلى تخصيص أو تحرير مساحة تخزين لها.

كل هيكل XftFontInfo قيد الاستخدام مرتبط بمعرف فريد، والذي يمكن استرداده باستخدام XftFontInfoHash(). يمكن فتح XftFont بناءً على بيانات XftFontInfo باستخدام XftFontOpenInfo().

XftColor


typedef struct _XftColor {
	unsigned long	 pixel;
	XRenderColor	 color;
} XftColor;

يسمح كائن XftColor برسم النص والعناصر الأخرى بلون معين (أو أقرب تقريب يقدمه العرض المرئي لـ X قيد الاستخدام).

تقوم XftColorAllocName() و XftColorAllocValue() بطلب تخصيص لون من خادم X (إذا لزم الأمر) وتهيئة أعضاء XftColor. تقوم XftColorFree() بإبلاغ خادم X بتحرير اللون المخصص حاليًا لـ XftColor.

بمجرد تهيئة XftColor، يمكن استخدام XftDrawSrcPicture()، و XftDrawGlyphs()، وعائلة الوظائف XftDrawString*()، و XftDrawCharSpec()، و XftDrawCharFontSpec()، و XftDrawGlyphSpec()، و XftDrawGlyphFontSpec()، و XftDrawRect() لرسم كائنات مختلفة باستخدامها.
