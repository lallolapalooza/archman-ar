XftDrawCharSpec(3)

الحزمة: libxft

التراخيص: HPND-sell-variant

Xft(3) إصدار X 11 Xft(3)

الاسم

Xft - مكتبة واجهة X FreeType

الملخص

#include <X11/Xft/Xft.h>

الوصف

Xft هي مكتبة بسيطة تقوم برسم النصوص والرسومات:

  • باستخدام المعلومات المقدمة من مكتبة Fontconfig،
  • يقوم Xft بتحويل رموز الخطوط باستخدام برنامج Fontconfig، و
  • يعرض بيانات الخط المحولة باستخدام امتداد X Rendering Extension.

صفحة الدليل هذه تكاد لا تلامس سطح هذه المكتبة.

ثوابت

XFT_MAJOR
هو رقم الإصدار الرئيسي لـ Xft.
XFT_MINOR
هو رقم الإصدار الفرعي لـ Xft.
XFT_REVISION
هو رقم المراجعة لـ Xft.
XFT_VERSION
هو XFT_MAJOR مضروبًا في 10000 (عشرة آلاف)، بالإضافة إلى XFT_MINOR مضروبًا في 100، بالإضافة إلى XFT_REVISION.
XftVersion
هو اسم مستعار لـ XFT_VERSION.

يوضح المثال التالي كيف يمكن استخدام ثوابت إصدار Xft: 

    #if (XFT_VERSION >= 20107)
    (void) puts("الإصدار 2.1.7 أو أحدث من مكتبة Xft قيد الاستخدام.");
    #else
    (void) printf("إصدار غير كافٍ من Xft (%d.%d.%d) مثبت؛
                  " تحتاج على الأقل إلى الإصدار 2.1.7.\n", XFT_MAJOR,
                  XFT_MINOR,
                  XFT_REVISION);
    #endif

أنواع البيانات

تتبع أنواع بيانات Xft اصطلاح تسمية، حيث تسبق جميع الأسماء بـ "Xft"

من Xlib...

لا تتبع أنواع بيانات Xlib اصطلاح تسمية. وهي موثقة في Xlib - واجهة لغة C.

يستخدم Xft هذه الأسماء: Bool، Colormap، Display، Drawable، Pixmap، Region، Visual، و XRectangle.

بعض أنواع البيانات ذات أهمية خاصة لـ Xft:

Drawable
معرّف مورد X، على سبيل المثال، نافذة مرتبطة بالعرض. يمكن استخدام كل من النوافذ و pixmaps كمصادر ووجهات في عمليات الرسومات. تُعرف هذه النوافذ و pixmaps مجتمعة باسم drawables. ومع ذلك، لا يمكن استخدام نافذة InputOnly كمصدر أو وجهة في عملية رسومات.
Region
المعرفة في <X11/Xutil.h> و <X11/Xregion.h>، Region هي مجموعة اعتباطية من مواقع البكسل التي يمكن لـ Xlib معالجتها. داخليًا، يتم تمثيل Region باتحاد عدد اعتباطي من المستطيلات. تحتفظ Xlib بمستطيل يبلغها بمدى هذا الاتحاد.
Visual
المعرف في <X11/Xutil.h>، يحتوي هذا الهيكل على معلومات حول تعيين الألوان المحتمل.

من XRender...

مستوحاة أنواع بيانات مكتبة X Render Extension من Xlib، ولكنها تفتقر إلى التوثيق المقابل. راجع ملف الرأس للحصول على التفاصيل:

<X11/extensions/Xrender.h>

أو اقرأ الكود المصدري (الذي يوفر مخططًا تفصيليًا لـ libXrender).

يستخدم Xft هذه الأسماء: Picture، XGlyphInfo، XRenderColor.

من Fontconfig...

تتبع أنواع بيانات Fontconfig اصطلاح تسمية، حيث تسبق جميع الأسماء بـ "Fc" وهي موثقة في Fontconfig Developers Reference.

يستخدم Xft هذه الأسماء من Fontconfig: FcBool، FcChar8، FcChar16، FcChar32، FcCharSet، FcEndian، FcFontSet، FcPattern، و FcResult.

من FreeType 2...

تتبع أنواع بيانات FreeType 2 اصطلاح تسمية، حيث تسبق جميع الأسماء بـ "FT_" وهي موثقة في FreeType API Reference.

يستخدم Xft هذه الأسماء: FT_Face، FT_Library، و FT_UInt.

يستخدم FreeType 2 FT_UInt (عدد صحيح غير موقع) لتمثيل الرموز.

XftFont


typedef struct _XftFont {
	int	 ascent;
	int	 descent;
	int	 height;
	int	 max_advance_width;
	FcCharSet	*charset;
	FcPattern	*pattern;
} XftFont;

XftFont هو هيكل البيانات الأساسي الذي يهم المبرمجين الذين يستخدمون Xft؛ يحتوي على مقاييس الخط العامة ومؤشرات إلى مجموعة أحرف Fontconfig والنمط المرتبط بالخط.

يتم ملء XftFonts باستخدام أي من XftFontOpen()، XftFontOpenName()، XftFontOpenXlfd()، XftFontOpenInfo()، أو XftFontOpenPattern(). يتم استخدام XftFontCopy() لمضاعفة XftFonts، ويتم استخدام XftFontClose() لوضع علامة على XftFont على أنه غير مستخدم. يتم تخصيص XftFonts داخليًا، وتتبع المراجع، وتحريرها بواسطة Xft؛ لا يحتاج المبرمج عادةً إلى تخصيص أو تحرير مساحة لها.

تستخدم XftDrawGlyphs()، وعائلة XftDrawString*()، XftDrawCharSpec()، و XftDrawGlyphSpec() XftFonts لرسم النصوص إلى كائن XftDraw، والذي قد يتوافق مع كل من رسم X الأساسي أو رسم X Rendering Extension.

تستخدم XftGlyphExtents() وعائلة XftTextExtents*() لتحديد امتدادات (الأبعاد القصوى) لـ XftFont.

يمكن تحديد تغطية الرموز أو الأحرف لـ XftFont باستخدام XftFontCheckGlyph() أو XftCharExists(). تُرجع XftCharIndex() فهرس الحرف الخاص بـ XftFont المقابل لنقطة رمز Unicode معينة.

تستخدم XftGlyphRender()، XftGlyphSpecRender()، XftCharSpecRender()، وعائلة XftTextRender*() XftFonts للرسم في هياكل Picture لـ X Rendering Extension.

ملاحظة: توفر XftDrawGlyphs()، وعائلة XftDrawString*()، XftDrawCharSpec()، و XftDrawGlyphSpec() وسيلة لرسم الخطوط مستقلة عن توافر امتداد X Rendering Extension على خادم X.

XftFontInfo

هو كائن مبهم يخزن معلومات حول خط. يتم إنشاء هياكل XftFontInfo باستخدام XftFontInfoCreate()، وتحريرها باستخدام XftFontInfoDestroy()، ومقارنتها باستخدام XftFontInfoEqual(). يتم تخصيص كائنات XftFontInfo داخليًا وتحريرها بواسطة Xft؛ لا يحتاج المبرمج عادةً إلى تخصيص أو تحرير مساحة لها.

يرتبط كل هيكل XftFontInfo قيد الاستخدام بمعرف فريد، والذي يمكن استرداده باستخدام XftFontInfoHash(). يمكن فتح XftFont بناءً على بيانات XftFontInfo باستخدام XftFontOpenInfo().

XftColor


typedef struct _XftColor {
	unsigned long	 pixel;
	XRenderColor	 color;
} XftColor;

يسمح كائن XftColor برسم النصوص والعناصر الأخرى بلون معين (أو أقرب تقريب تقدمه رؤية X المستخدمة).

تقوم XftColorAllocName() و XftColorAllocValue() بطلب تخصيص لون من خادم X (إذا لزم الأمر) وتهيئة أعضاء XftColor. تُعلم XftColorFree() خادم X بتحرير اللون المخصص حاليًا لـ XftColor.

بمجرد تهيئة XftColor، يمكن استخدام XftDrawSrcPicture()، XftDrawGlyphs()، عائلة XftDrawString*()، XftDrawCharSpec()، XftDrawCharFontSpec()، XftDrawGlyphSpec()، XftDrawGlyphFontSpec()، و XftDrawRect() لرسم كائنات مختلفة باستخدام i

XftFontOpenXlfd يعمل بنفس طريقة عمل XftFontOpen، باستثناء أنه يأخذ سلسلة تحتوي على وصف خط X المنطقي (XLFD)، ويستخدم الدالة XftXlfdParse().

FcPattern *XftFontMatch (
	Display	*dpy،
	int	 screen،
	_Xconst FcPattern	*pattern،
	FcResult	*result);

تُستخدم أيضًا داخليًا بواسطة دوال XftFontOpen*، يمكن استخدام XftFontMatch مباشرة لتحديد نمط خط Xft الناتج عن طلب فتح خط Xft.

FcPattern *XftXlfdParse (
	_Xconst char	*xlfd_orig،
	Bool	ignore_scalable،
	Bool	complete);

XftXlfdParse يحلل المعلمة xlfd_orig وفقًا لمستند X Logical Font Description Conventions، لكنه يتجاهل بعض الحقول: setwidth_name، add_style_name، spacing، و average_width.

XftXlfdParse ينشئ نمط Fontconfig، ويقوم بتعيين الخاصية XFT_XLFD إلى قيمة xlfd_orig، ويربط المعلومات المجمعة بخصائص Fontconfig. يتم تجاهل الحقول الفارغة أو التي تحتوي على “*”:
FC_FOUNDRY
من foundry.
FC_FAMILY
من family.
FC_WEIGHT
weight_name، الافتراضي هو FC_WEIGHT_MEDIUM.
FC_SLANT
من slant، الافتراضي هو FC_SLANT_ROMAN.
FC_SIZE
من point_size.
FC_PIXEL_SIZE
من pixel_size. إذا تم تعيين point_size، بالإضافة إلى resolution_x و resolution_y، فسيتم تحجيم القيمة لتحويل ارتفاع الخط إلى نقاط.

تحديد امتدادات النص

Xft يوفر عدة دوال لتحديد الارتفاع والعرض المطلوبين لعرض سلسلة نصية. بعد حساب offset، في الحالات التي سيتم فيها إزاحة السلسلة لأعلى أو لأسفل أو لليسار أو لليمين، يشار إلى هذه الأرقام باسم امتدادات النص.

void XftTextExtents8 (
	Display	*dpy،
	XftFont	*font،
	_Xconst FcChar8	*string،
	int	 len،
	XGlyphInfo	*extents);

XftTextExtents8 تحسب امتدادات البكسل على الشاشة dpy لعدد لا يزيد عن len حرفًا لسلسلة مكونة من أحرف ثمانية البتات عند رسمها بالخط font، وتخزنها في extents.

XftTextExtentsUtf16 تحسب امتدادات البكسل على الشاشة dpy لعدد لا يزيد عن len بايت لسلسلة مرمزة بـ UTF-16LE أو UTF-16BE عند رسمها بالخط font، وتخزنها في extents. يجب تحديد ترتيب البايتات (endianness) للسلسلة string في endian.

void XftGlyphExtents (
	Display	*dpy،
	XftFont	*font،
	_Xconst FT_UInt	*glyphs،
	int	 nglyphs،
	XGlyphInfo	*extents);

تُستخدم أيضًا داخليًا بواسطة دوال XftTextExtents*()، XftGlyphExtents تحسب امتدادات البكسل على الشاشة dpy لعدد لا يزيد عن nglyphs في المصفوفة glyphs عند رسمها بالخط font، وتخزنها في extents.

إذا كانت أي من glyphs مفقودة (يتم تحديد ذلك بالتحقق باستخدام XftFontCheckGlyph())، فسيتم ملء الإدخال المقابل في extents بالأصفار.

إدارة XftColor

Bool XftColorAllocName (
	Display	*dpy،
	_Xconst Visual	*visual،
	Colormap	 cmap،

	XftColor	*result);

استخدم XAllocNamedColor() للبحث عن اللون المسمى name للشاشة المرتبطة بخريطة الألوان cmap.

  • إذا أعادت XAllocNamedColor() قيمة غير صفرية، فسيقوم XftColorAllocName() بملء حقل البكسل الخاص بـ XftColor الناتج باللون الأقرب الذي تدعمه الشاشة، بالإضافة إلى حقول الأحمر والأخضر والأزرق الدقيقة من قاعدة البيانات، ويعيد True.
  • إذا أعادت XAllocNamedColor() صفرًا، فسيقوم XftColorAllocName() بإعادة False، ولن يقوم بتحديث XftColor الذي تشير إليه result.
المعلمة visual غير مستخدمة.
Bool XftColorAllocValue (
	Display	*dpy،
	Visual	*visual،
	Colormap	 cmap،

تخصيص قيمة لون:

  • إذا كان فئة visual هي TrueColor، فسيقوم XftColorAllocValue() بتعيين حقل pixel في XftColor الذي تشير إليه result باستخدام الحقول الحمراء والخضراء والزرقاء من المعلمة color.
  • إذا لم تكن فئة visual هي TrueColor، فسيقوم XftColorAllocValue() باستدعاء XAllocColor() لتخصيص إدخال في خريطة الألوان cmap. الذي يعيد قيمة البكسل للون الأقرب إلى عناصر RGB المحددة التي تدعمها الأجهزة.
إذا نجحت XAllocColor()، فسيقوم XftColorAllocValue() بتخزين قيمة البكسل هذه في XRenderColor الذي تشير إليه result ويعيد True.
إذا فشلت XAllocColor()، فسيقوم XftColorAllocValue() بإعادة False ولا يقوم بتعديل النتيجة.
void XftColorFree (

معالجة بيانات الخط

void XftFontLoadGlyphs (
	Display	*dpy,
	XftFont	*pub,
	FcBool	 need_bitmaps,
	_Xconst FT_UInt	*glyphs,
	int	 nglyph);

باستخدام FT_Load_Glyph()، قم بتحميل nglyphs لمؤشرات الرموز (قيم Unicode) المدرجة في المصفوفة glyphs من الخط المعطى pub.

يتضمن تحميل الرمز أكثر من مجرد قراءة البيانات في الذاكرة. يستخدم Xft نمط Fontconfig المرتبط بالعرض dpy (على سبيل المثال، عبر XftDefaultSet()) لتحديد ما إذا كان سيتم استخدام مربع حدودي، أو قص الرموز ضمن هذه الحدود، أو قياس الرموز، أو حساب مقاييس الخط، أو إضافتها إلى امتداد X Render باستخدام XRenderAddGlyphs().
void XftFontUnloadGlyphs (
	Display	*dpy,
	XftFont	*pub,
	_Xconst FT_UInt	*glyphs,
	int	 nglyph);

تجاهل البيانات لما يصل إلى nglyph من الرموز التي تكون مؤشرات الرموز (قيم Unicode) لها مدرجة في المصفوفة glyphs للخط المعطى pub. إذا تمت إضافة الرموز إلى امتداد X Render، يقوم Xft بإزالتها باستخدام XRenderFreeGlyphs(). يتتبع Xft مقدار الذاكرة المستخدمة للرموز، ويحدث الاستخدام المرتبط بالعرض dpy.

FT_Face XftLockFace (
	XftFont	*pub);

إذا لم يتم إنشاء أي "وجه" FreeType 2 (تمثيل في الذاكرة لخط معين بنمط معين) للخط pub، فقم بإنشاء واحد باستخدام FT_New_Face(). تُستخدم أقفال الوجوه كعدد مراجع يستخدمه Xft لضمان إنشاء وجه واحد فقط، وأن يتم الاحتفاظ به حتى لا يعود الخط مستخدمًا.

يتم استخدام أقفال الوجوه مباشرة في XftCharIndex() و XftFontLoadGlyphs()، والتي بدورها تستخدم في العديد من وظائف Xft.
تم تقديم أقفال الوجوه في الإصدار 1 (أكتوبر 2002). اعتمدت بعض التطبيقات، مثل Gdk/Gimp، على هذه الوظائف. في الإصدار 2.1.9 (يونيو 2006)، تم الاحتفاظ بأقفال الوجوه كجزء من واجهة برمجة التطبيقات العامة عندما وفرت تكوينات المكتبات المشتركة المحسنة إخفاء الرموز الخاصة.
void XftUnlockFace (
	XftFont	*pub);

يقلل من عدد مراجع "وجه" FreeType 2 المرتبط بالخط.

يتم إلغاء تخصيص كائنات FT_Face باستخدام FT_Done_Face(). يقوم Xft بذلك في XftFontInfoDestroy() وعند التنظيف عند الفشل في XftFontInfoCreate() و XftFontOpenPattern().

التوافق

اعتبارًا من الإصدار 2 (مايو 2002)، أصبح Xft مستقرًا نسبيًا. ومن المتوقع أن يحتفظ بتوافق المصدر والثنائي في الإصدارات المستقبلية.

يوفر Xft واجهة توافق لإصداره الرئيسي السابق، Xft 1.x، الموصوف أدناه.

ملف رأس Xft 1.x

#include <X11/Xft/XftCompat.h>

أنواع بيانات Xft 1.x

XftPattern
يحتوي على مجموعة من الأسماء مع قوائم قيم مرتبطة؛ يشير كل اسم إلى خاصية للخط. تُستخدم XftPattern كمدخلات لرمز المطابقة وكذلك للاحتفاظ بمعلومات حول خطوط معينة.
XftFontSet
يحتوي على قائمة من XftPatterns. داخليًا، يستخدم Xft بنية البيانات هذه للاحتفاظ بمجموعات من الخطوط. خارجيًا، يعيد Xft نتائج قائمة الخطوط بهذا التنسيق.
XftObjectSet
يحتوي على مجموعة من الأسماء ويستخدم لتحديد الحقول من الخطوط التي يتم وضعها في قائمة الأنماط التي تم إرجاعها عند سرد الخطوط.
