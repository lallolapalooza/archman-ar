XftDrawString16(3)

الحزمة: libxft

التراخيص: HPND-sell-variant

Xft(3) X Version 11 Xft(3)

الاسم

Xft - مكتبة واجهة X FreeType

الموجز

#include <X11/Xft/Xft.h>

الوصف

Xft هي مكتبة بسيطة ترسم النصوص والرسومات:

  • باستخدام المعلومات المقدمة من مكتبة Fontconfig،
  • يقوم Xft بتحويل رموز الخطوط باستخدام مُنشئ الخطوط FreeType، و
  • يعرض بيانات الخط المحولة باستخدام امتداد X Rendering Extension.

صفحة الدليل هذه بالكاد تخدش سطح هذه المكتبة.

الثوابت

XFT_MAJOR
هو رقم الإصدار الرئيسي لـ Xft.
XFT_MINOR
هو رقم الإصدار الثانوي لـ Xft.
XFT_REVISION
هو رقم المراجعة لـ Xft.
XFT_VERSION
هو XFT_MAJOR مضروبًا في 10000 (عشرة آلاف)، زائد XFT_MINOR مضروبًا في 100، زائد XFT_REVISION.
XftVersion
هو اسم مستعار لـ XFT_VERSION.

يوضح المثال التالي كيف يمكن استخدام ثوابت إصدار Xft: 

    #if (XFT_VERSION >= 20107)
    (void) puts("Version 2.1.7 or later of the Xft library is in"
                " use.");
    #else
    (void) printf("Insufficient version of Xft (%d.%d.%d) installed;
                  " need at least version 2.1.7.\n", XFT_MAJOR,
                  XFT_MINOR,
                  XFT_REVISION);
    #endif

أنواع البيانات

تتبع أنواع بيانات Xft اتفاقية تسمية، بادئة جميع الأسماء بـ "Xft"

من Xlib...

لا تتبع أنواع بيانات Xlib اتفاقية تسمية. وهي موثقة في Xlib - C Language Interface.

يستخدم Xft هذه الأسماء: Bool، Colormap، Display، Drawable، Pixmap، Region، Visual، و XRectangle.

بعض أنواع البيانات ذات أهمية خاصة لـ Xft:

Drawable
معلن عنه في <X11/X.h>، هذا هو معرف مورد X، على سبيل المثال، نافذة مرتبطة بالعرض. يمكن استخدام كل من النوافذ والصور النقطية كمصادر ووجهات في عمليات الرسومات. تُعرف هذه النوافذ والصور النقطية مجتمعة باسم الـ drawables. ومع ذلك، لا يمكن استخدام نافذة InputOnly كمصدر أو وجهة في عملية رسومات.
Region
معلن عنه في <X11/Xutil.h> و <X11/Xregion.h>، Region هي مجموعة اعتباطية من مواقع البكسل التي يمكن لـ Xlib معالجتها. داخليًا، يتم تمثيل Region باتحاد عدد اعتباطي من المستطيلات. تحتفظ Xlib بمستطيل يخبرنا بمدى هذا الاتحاد.
Visual
معلن عنه في <X11/Xutil.h>، يحتوي هذا الهيكل على معلومات حول تعيين الألوان الممكن.

من XRender...

أنواع بيانات مكتبة امتداد X Rendering Extension مستوحاة من Xlib، ولكنها تفتقر إلى الوثائق المقابلة. ارجع إلى ملف الرأس للحصول على التفاصيل:

<X11/extensions/Xrender.h>

أو اقرأ الكود المصدري (الذي يوفر مخططًا عامًا لـ libXrender).

يستخدم Xft هذه الأسماء: Picture، XGlyphInfo، XRenderColor.

من Fontconfig...

تتبع أنواع بيانات Fontconfig اتفاقية تسمية، بادئة جميع الأسماء بـ "Fc" وهي موثقة في Fontconfig Developers Reference.

يستخدم Xft هذه الأسماء Fontconfig: FcBool، FcChar8، FcChar16، FcChar32، FcCharSet، FcEndian، FcFontSet، FcPattern، و FcResult.

من FreeType 2...

تتبع أنواع بيانات FreeType 2 اتفاقية تسمية، بادئة جميع الأسماء بـ "FT_" وهي موثقة في FreeType API Reference.

يستخدم Xft هذه الأسماء: FT_Face، FT_Library، و FT_UInt.

يستخدم FreeType 2 FT_UInt (عدد صحيح غير موقع) لتمثيل الرموز.

XftFont


typedef struct _XftFont {
	int	 ascent;
	int	 descent;
	int	 height;
	int	 max_advance_width;
	FcCharSet	*charset;
	FcPattern	*pattern;
} XftFont;

XftFont هو هيكل البيانات الأساسي الذي يهم المبرمجين الذين يستخدمون Xft؛ فهو يحتوي على مقاييس خط عامة ومؤشرات إلى مجموعة أحرف Fontconfig والنمط المرتبط بالخط.

يتم ملء XftFonts بأي من XftFontOpen()، XftFontOpenName()، XftFontOpenXlfd()، XftFontOpenInfo()، أو XftFontOpenPattern(). يستخدم XftFontCopy() لمضاعفة XftFonts، ويستخدم XftFontClose() لوضع علامة على XftFont على أنه غير مستخدم. يتم تخصيص XftFonts داخليًا، وعدها بالإشارة، وتحريرها بواسطة Xft؛ لا يحتاج المبرمج عادةً إلى تخصيص أو تحرير مساحة لها.

تستخدم XftDrawGlyphs()، وعائلة XftDrawString*()، XftDrawCharSpec()، و XftDrawGlyphSpec() XftFonts لرسم النص إلى كائن XftDraw، والذي قد يتوافق مع رسم X أساسي أو رسم امتداد X Rendering Extension.

تستخدم XftGlyphExtents() وعائلة XftTextExtents*() لتحديد الامتدادات (الأبعاد القصوى) لـ XftFont.

يمكن تحديد تغطية الرمز أو الحرف لـ XftFont باستخدام XftFontCheckGlyph() أو XftCharExists(). تُرجع XftCharIndex() فهرس الحرف الخاص بـ XftFont المقابل لنقطة رمز Unicode معينة.

تستخدم XftGlyphRender()، XftGlyphSpecRender()، XftCharSpecRender()، وعائلة XftTextRender*() XftFonts للرسم في هياكل Picture الخاصة بـ X Rendering Extension.

ملاحظة: توفر XftDrawGlyphs()، وعائلة XftDrawString*()، XftDrawCharSpec()، و XftDrawGlyphSpec() وسيلة لرسم الخطوط مستقلة عن توفر امتداد X Rendering Extension على خادم X.

XftFontInfo

هو كائن غامض يخزن معلومات حول خط. يتم إنشاء هياكل XftFontInfo باستخدام XftFontInfoCreate()، وتحريرها باستخدام XftFontInfoDestroy()، ومقارنتها باستخدام XftFontInfoEqual(). يتم تخصيص كائنات XftFontInfo وتحريرها داخليًا بواسطة Xft؛ لا يحتاج المبرمج عادةً إلى تخصيص أو تحرير مساحة لها.

يرتبط كل هيكل XftFontInfo قيد الاستخدام بمعرف فريد، والذي يمكن استرداده باستخدام XftFontInfoHash(). يمكن فتح XftFont بناءً على بيانات XftFontInfo باستخدام XftFontOpenInfo().

XftColor


typedef struct _XftColor {
	unsigned long	 pixel;
	XRenderColor	 color;
} XftColor;

يسمح كائن XftColor برسم النصوص والعناصر الأخرى بلون معين (أو أقرب تقريب يقدمه مرئي X المستخدم).

تقوم XftColorAllocName() و XftColorAllocValue() بطلب تخصيص لون من خادم X (إذا لزم الأمر) وتهيئة أعضاء XftColor. تُعلم XftColorFree() خادم X بتحرير اللون المخصص حاليًا لـ XftColor.

يعيد كائن صورة X Render، والذي يستخدم لرسم الحروف، على سبيل المثال، مع XftGlyphRender() أو XftGlyphSpecRender() أو XftGlyphFontSpecRender()، بواسطة XftDrawGlyphs() أو XftDrawGlyphSpec() أو XftDrawGlyphFontSpec()، على التوالي.

void XftDrawDestroy (
	XftDraw	*draw);

XftDrawDestroy يدمر draw (الذي تم إنشاؤه بواسطة إحدى وظائف XftDrawCreate*()) ويحرر الذاكرة التي تم تخصيصها له.

Bool XftDrawSetClip (
	XftDraw	*draw,
	Region	 r);

يقوم بتعيين القطع للمعلّم XftDraw draw بدءًا بمنطقة Region:

  • إذا لم تكن المنطقة r معلمة null، فإن Xft ينشئ منطقة Region جديدة (لنسخ المعلّمة)،
  • يقوم Xft بتدمير أي منطقة قطع موجودة.
  • يقوم Xft بتعيين clip_type للمعلم draw إلى XftClipTypeRegion إذا لم تكن المعلمة r null. وإلا، يقوم بتعيين clip_type إلى XftClipTypeNone.
  • أخيرًا، يقوم Xft بتحديث القطع للكائنات الموجودة، وتحديث clip_mask لصورة X Render الخاصة به Picture، وتعيين قناع القطع في سياق الرسومات (GC) المرتبط بالمعلم XftDraw.
XftDrawSetClip() يعيد True إذا لم يكن هناك تغيير ضروري، أو إذا نجحت العملية. يعيد False إذا لم يتمكن من إنشاء المنطقة الجديدة Region().
Bool XftDrawSetClipRectangles (
	XftDraw	*draw,
	int	 xOrigin,
	int	 yOrigin,
	_Xconst XRectangle	*rects,
	int	 n);

مثل XftDrawSetClip()، يقوم XftDrawSetClipRectangles() بإعداد القطع للمعلم XftDraw draw ولكن يستخدم مجموعة من n مستطيلات (المعلم rects) التي يمكن استخدامها لبناء منطقة Region.

يقوم Xft بتعيين clip_type للمعلم draw إلى XftClipTypeRectangles ويستخدم XSetClipRectangles() للقطع الأساسي (X11) و XRenderSetPictureClipRectangles() لقطع X Render.
void XftDrawSetSubwindowMode (
	XftDraw	*draw,
	int	 mode);

يقوم بتعيين وضع النافذة الفرعية للمعلم XftDraw draw. يمكن أن يكون الوضع إما ClipByChildren (الافتراضي) أو IncludeInferiors:

  • بالنسبة لـ ClipByChildren، يتم قطع كل من نوافذ المصدر والوجهة بالإضافة إلى جميع الأطفال المرئيين من نوع InputOutput.
  • بالنسبة لـ IncludeInferiors، لا يتم قطع أي من نافذة المصدر أو الوجهة بواسطة النوافذ الفرعية. سيؤدي هذا إلى تضمين محتويات النافذة الفرعية في المصدر والرسم عبر حدود النافذة الفرعية للوجهة.
بالإضافة إلى وضع النافذة الفرعية الذي يحتفظ به Xft، فإنه يقوم بتحديث وضع النافذة الفرعية لأي سياق رسومات GC مرتبط باستخدام XSetSubwindowMode() وكذلك لصورة X Render Picture باستخدام XRenderChangePicture().

void XftDrawString8 (

XftDrawString8 يرسم ما لا يزيد عن len من الحروف من string إلى كائن الرسم Xft d باستخدام font باللون color في الموضع x، y.

XftDrawRect يرسم مستطيلاً مصمتاً باللون color والعرض width والارتفاع height المحددين في الموضع x، y إلى كائن الرسم Xft d.

void XftCharFontSpecRender (
	Display	*dpy,
	int	 op,
	Picture	 src,
	Picture	 dst,
	int	 srcx,
	int	 srcy,
	_Xconst XftCharFontSpec	*chars,
	int	 len);

XftCharFontSpecRender() يقوم بتحويل المعلم chars من XftCharFontSpec إلى XftGlyphFontSpec، ويمرر المصفوفة المحولة مع المعلمات الأخرى دون تغيير إلى XftGlyphFontSpecRender() لرسم البيانات.

void XftDrawGlyphFontSpec (
	XftDraw	*draw,
	_Xconst XftColor	*color,
	_Xconst XftGlyphFontSpec	*glyphs,
	int	 len);

يرسم قائمة من الحروف المرتبطة بالخطوط في إحداثيات محددة، والتي يتم تمريرها كمصفوفة من len هياكل XftGlyphFontSpec عبر المعلم glyphs. يتم رسم جميع الحروف باستخدام اللون المحدد في المعلم color.

لكل إدخال في مصفوفة glyphs:
  • إذا كان الخط المرتبط يستخدم امتداد X Render، فإن Xft يستخدم XftGlyphFontSpecRender() لرسم الحرف، باستخدام صورة Picture تم الحصول عليها من استدعاء XftDrawSrcPicture () مع معلمات draw و color.
  • بخلاف ذلك، يوفر Xft ميزة مماثلة باستخدام Xlib.
void XftGlyphFontSpecRender (
	Display	*dpy,
	int	 op,
	Picture	 src,
	Picture	 dst,
	int	 srcx,
	int	 srcy,
	_Xconst XftGlyphFontSpec	*glyphs,
	int	 nglyphs);

يتضمن هذا عدة خطوات:

  • أولاً، يتأكد Xft من أن مصفوفة glyphs كاملة باستخدام XftFontCheckGlyph() وتحميل أي حروف مفقودة باستخدام XftFontLoadGlyphs().
  • ثم يفحص Xft الحروف للعثور على أقصى قيمة Unicode. هذا يحدد حجم الترميز، أي width.
  • بعد ذلك، لكل حرف، يتحقق Xft مما إذا كانت قيمة Unicode الخاصة به تتم معالجتها بواسطة الخط المقابل. إذا لم يكن الأمر كذلك، يتم تعيين القيمة إلى صفر (0)، لتجاهلها. أثناء القيام بذلك، يتذكر Xft أيضًا الخط الأول المستخدم، ويحسب الموضع الذي سيتم فيه رسم كل حرف.
  • يقوم Xft ببناء مصفوفة مكافئة من الحروف بالتنسيق المتوقع من مكتبة X Render.
  • أخيرًا، يستخدم XftGlyphFontSpecRender() امتداد X Render لرسم الحروف، باستخدام دالة XRenderCompositeText*() المناسبة وفقًا لـ width، ويتخلص من البيانات المؤقتة.
تستخدم المعلمات op و dst و src و srcx و srcy كما هو موضح في توثيق مكتبة X Render.

معالجة بيانات الخط

void XftFontLoadGlyphs (
	Display	*dpy,
	XftFont	*pub,
	FcBool	 need_bitmaps,
	_Xconst FT_UInt	*glyphs,
	int	 nglyph);

باستخدام FT_Load_Glyph()، قم بتحميل nglyphs لمؤشرات الأحرف (قيم Unicode) المدرجة في المصفوفة glyphs من الخط المعطى pub.

تحميل الحرف يتضمن أكثر من مجرد قراءة البيانات في الذاكرة. يستخدم Xft نمط Fontconfig المرتبط بالعرض dpy (على سبيل المثال، عبر XftDefaultSet()) لتحديد ما إذا كان سيتم استخدام مربع الإحاطة، أو قص الأحرف ضمن تلك الحدود، أو قياس الأحرف، أو حساب مقاييس الخط، وإضافتها إلى امتداد X Render باستخدام XRenderAddGlyphs().
void XftFontUnloadGlyphs (
	Display	*dpy,
	XftFont	*pub,
	_Xconst FT_UInt	*glyphs,
	int	 nglyph);

تخلص من البيانات لما يصل إلى nglyph من الأحرف التي تم إدراج مؤشرات الأحرف الخاصة بها (قيم Unicode) في المصفوفة glyphs للخط المعطى pub. إذا تمت إضافة الأحرف إلى امتداد X Render، يقوم Xft بإزالتها باستخدام XRenderFreeGlyphs(). يتتبع Xft مقدار الذاكرة المستخدمة للأحرف، ويحدث الاستخدام المرتبط بالعرض dpy.

FT_Face XftLockFace (
	XftFont	*pub);

إذا لم يتم إنشاء أي "وجه" (تمثيل في الذاكرة لنمط خط معين بنمط معين) من FreeType 2 للخط pub، قم بإنشاء واحد باستخدام FT_New_Face(). أقفال الوجه هي عداد مراجع تستخدمه Xft لضمان إنشاء وجه واحد فقط، وأن يتم الاحتفاظ به حتى لم يعد الخط مستخدمًا.

يتم استخدام أقفال الوجه مباشرة في XftCharIndex() و XftFontLoadGlyphs()، والتي بدورها تستخدم في العديد من وظائف Xft.
تم تقديم أقفال الوجه في الإصدار 1 (أكتوبر 2002). اعتمدت بعض التطبيقات، مثل Gdk/Gimp، على هذه الوظائف. في الإصدار 2.1.9 (يونيو 2006)، تم الاحتفاظ بأقفال الوجه كجزء من واجهة برمجة التطبيقات العامة عندما وفرت تكوينات مكتبات المشاركة المحسنة لإخفاء الرموز الخاصة.
void XftUnlockFace (
	XftFont	*pub);

يقلل عداد المراجع لـ "وجه" FreeType 2 المرتبط بالخط.

يتم إلغاء تخصيص كائنات FT_Face باستخدام FT_Done_Face(). يقوم Xft بذلك في XftFontInfoDestroy() وعند التنظيف عند الفشل في XftFontInfoCreate() و XftFontOpenPattern().

التوافق

اعتبارًا من الإصدار 2 (مايو 2002)، أصبح Xft مستقرًا نسبيًا. من المتوقع أن يحتفظ بالتوافق المصدر والثنائي في الإصدارات المستقبلية.

يوفر Xft واجهة توافق لإصداره الرئيسي السابق، Xft 1.x، الموضح أدناه.

ملف رأس Xft 1.x

#include <X11/Xft/XftCompat.h>

أنواع بيانات Xft 1.x

XftPattern
يحتوي على مجموعة من الأسماء مع قوائم قيم مرتبطة؛ يشير كل اسم إلى خاصية لخط. يتم استخدام XftPattern كمدخلات لرمز المطابقة وكذلك لتخزين معلومات حول خطوط محددة.
XftFontSet
يحتوي على قائمة من XftPatterns. داخليًا، يستخدم Xft بنية البيانات هذه لتخزين مجموعات من الخطوط. خارجيًا، يعيد Xft نتائج سرد الخطوط بهذا التنسيق.
XftObjectSet
يحتوي على مجموعة من الأسماء ويستخدم لتحديد الحقول التي من الخطوط التي يتم وضعها في قائمة الأنماط المعادة عند سرد الخطوط.
