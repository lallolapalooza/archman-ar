XRemoveHosts(3)

الحزمة: libx11

التراخيص: MIT AND X11

XAddHost(3) وظائف XLIB XAddHost(3)

الاسم

XAddHost, XAddHosts, XListHosts, XRemoveHost, XRemoveHosts, XSetAccessControl, XEnableAccessControl, XDisableAccessControl, XHostAddress, XServerInterpretedAddress - التحكم في الوصول إلى المضيف وهيكل التحكم في المضيف

الصيغة

int XAddHost(Display *display, XHostAddress *host);

int XAddHosts(Display *display, XHostAddress *hosts, int num_hosts);

XHostAddress *XListHosts(Display *display, int *nhosts_return, Bool *state_return);

int XRemoveHost(Display *display, XHostAddress *host);

int XRemoveHosts(Display *display, XHostAddress *hosts, int num_hosts);

int XSetAccessControl(Display *display, int mode);

int XEnableAccessControl(Display *display);

int XDisableAccessControl(Display *display);

المعاملات

display
يحدد الاتصال بخادم X.
host
يحدد المضيف المراد إضافته أو إزالته.
hosts
يحدد كل مضيف سيتم إضافته أو إزالته.
mode
يحدد الوضع. يمكنك تمرير EnableAccess أو DisableAccess.
nhosts_return
يعيد عدد المضيفين الموجودين حاليًا في قائمة التحكم في الوصول.
num_hosts
يحدد عدد المضيفين.
state_return
يعيد حالة التحكم في الوصول.

الوصف

تقوم الدالة XAddHost بإضافة المضيف المحدد إلى قائمة التحكم في الوصول لهذا العرض. يجب أن يكون الخادم على نفس المضيف مثل العميل الذي يصدر الأمر، وإلا فستنتج خطأ BadAccess.

يمكن لـ XAddHost إنشاء أخطاء BadAccess و BadValue.

تقوم الدالة XAddHosts بإضافة كل مضيف محدد إلى قائمة التحكم في الوصول لهذا العرض. يجب أن يكون الخادم على نفس المضيف مثل العميل الذي يصدر الأمر، وإلا فستنتج خطأ BadAccess.

يمكن لـ XAddHosts إنشاء أخطاء BadAccess و BadValue.

تعيد الدالة XListHosts قائمة التحكم في الوصول الحالية بالإضافة إلى ما إذا كان استخدام القائمة عند إعداد الاتصال قد تم تمكينه أو تعطيله. تسمح XListHosts للبرنامج بمعرفة الآلات التي يمكنها إجراء الاتصالات. كما تعيد مؤشرًا إلى قائمة هياكل المضيف التي تم تخصيصها بواسطة الدالة. عند عدم الحاجة إليها بعد الآن، يجب تحرير هذه الذاكرة عن طريق استدعاء XFree.

تقوم الدالة XRemoveHost بإزالة المضيف المحدد من قائمة التحكم في الوصول لهذا العرض. يجب أن يكون الخادم على نفس المضيف مثل عملية العميل، وإلا فستنتج خطأ BadAccess. إذا قمت بإزالة جهازك من قائمة الوصول، فلن تتمكن بعد ذلك من الاتصال بهذا الخادم، ولا يمكن التراجع عن هذه العملية ما لم تقم بإعادة تعيين الخادم.

يمكن لـ XRemoveHost إنشاء أخطاء BadAccess و BadValue.

تقوم الدالة XRemoveHosts بإزالة كل مضيف محدد من قائمة التحكم في الوصول لهذا العرض. يجب أن يكون خادم X على نفس المضيف مثل عملية العميل، وإلا فستنتج خطأ BadAccess. إذا قمت بإزالة جهازك من قائمة الوصول، فلن تتمكن بعد ذلك من الاتصال بهذا الخادم، ولا يمكن التراجع عن هذه العملية ما لم تقم بإعادة تعيين الخادم.

يمكن لـ XRemoveHosts إنشاء أخطاء BadAccess و BadValue.

تقوم الدالة XSetAccessControl إما بتمكين أو تعطيل استخدام قائمة التحكم في الوصول عند كل إعداد اتصال.

يمكن لـ XSetAccessControl إنشاء أخطاء BadAccess و BadValue.

تقوم الدالة XEnableAccessControl بتمكين استخدام قائمة التحكم في الوصول عند كل إعداد اتصال.

يمكن لـ XEnableAccessControl إنشاء خطأ BadAccess.

تقوم الدالة XDisableAccessControl بتعطيل استخدام قائمة التحكم في الوصول عند كل إعداد اتصال.

يمكن لـ XDisableAccessControl إنشاء خطأ BadAccess.

الهياكل

يحتوي هيكل XHostAddress على:

typedef struct {
        int family;     /* على سبيل المثال FamilyInternet */
        int length;     /* طول العنوان، بالبايت */
        char *address;  /* مؤشر إلى مكان العثور على العنوان */
} XHostAddress;

يحدد عضو العائلة أي عائلة عنوان بروتوكول سيتم استخدامها (على سبيل المثال، TCP/IP أو DECnet) ويمكن أن تكون FamilyInternet، FamilyInternet6، FamilyServerInterpreted، FamilyDECnet، أو FamilyChaos. يحدد عضو الطول طول العنوان بالبايت. يحدد عضو العنوان مؤشرًا إلى العنوان.

بالنسبة لعائلة ServerInterpreted، يتم تجاهل الطول ويشير عضو العنوان إلى هيكل XServerInterpretedAddress الذي يحتوي على:

typedef struct {
        int typelength;         /* طول سلسلة النوع، بالبايت */
        int valuelength;        /* طول سلسلة القيمة، بالبايت */
        char *type;             /* مؤشر إلى مكان العثور على سلسلة النوع */
        char *value;            /* مؤشر إلى مكان العثور على العنوان */
} XServerInterpretedAddress;

يشير عضوان النوع والقيمة إلى سلاسل تمثل نوع وقيمة إدخال الخادم المفسر. قد لا تنتهي هذه السلاسل بـ NULL، لذا يجب توخي الحذر عند الوصول إليها. يحدد عضوان typelength و valuelength الطول بالبايت لسلاسل النوع والقيمة.

التشخيصات

BadAccess
حاول عميل تعديل قائمة التحكم في الوصول من مضيف محلي (أو مصرح به بشكل آخر) غير ذلك.
BadValue
تقع قيمة رقمية ما خارج نطاق القيم التي يقبلها الطلب. ما لم يتم تحديد نطاق معين لمعامل، يتم قبول النطاق الكامل الذي تحدده بنية المعامل. يمكن لأي معامل محدد كمجموعة من البدائل إنشاء هذا الخطأ.
