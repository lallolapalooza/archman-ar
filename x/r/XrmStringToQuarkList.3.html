XrmStringToQuarkList(3)

الحزمة: libx11

التراخيص: MIT AND X11

XrmUniqueQuark(3) وظائف XLIB XrmUniqueQuark(3)

الاسم

XrmUniqueQuark, XrmStringToQuark, XrmPermStringToQuark, XrmQuarkToString, XrmStringToQuarkList, XrmStringToBindingQuarkList - التعامل مع الأكواد المميزة للموارد

الملخص

#include <X11/Xresource.h>

XrmQuark XrmUniqueQuark(void);

#define XrmStringToName(string) XrmStringToQuark(string)
#define XrmStringToClass(string) XrmStringToQuark(string)
#define XrmStringToRepresentation(string) XrmStringToQuark(string)

XrmQuark XrmStringToQuark(char *string);

XrmQuark XrmPermStringToQuark(_Xconst char *string);

#define XrmStringToName(string) XrmStringToQuark(string)
#define XrmStringToClass(string) XrmStringToQuark(string)
#define XrmStringToRepresentation(string) XrmStringToQuark(string)

XrmQuark XrmStringToQuark(char *string);

XrmQuark XrmPermStringToQuark(_Xconst char *string);

#define XrmNameToString(name) XrmQuarkToString(name)
#define XrmClassToString(class) XrmQuarkToString(class)
#define XrmRepresentationToString(type) XrmQuarkToString(type)

char *XrmQuarkToString(XrmQuark quark);

#define XrmStringToNameList(str, name) XrmStringToQuarkList((str), (name))
#define XrmStringToClassList(str,class) XrmStringToQuarkList((str), (class))

void XrmStringToQuarkList(char *string, XrmQuarkList quarks_return);

void XrmStringToBindingQuarkList(_Xconst char *string, XrmBindingList bindings_return, XrmQuarkList quarks_return);

الوسائط

bindings_return
يُرجع قائمة الارتباطات.
quark
يحدد الكود المميز الذي يُراد الحصول على النص المكافئ له.
quarks_return
يُرجع قائمة الأكواد المميزة.
string
يحدد النص الذي سيتم تخصيص كود مميز أو قائمة أكواد مميزة له.

الوصف

تقوم الدالة XrmUniqueQuark بتخصيص كود مميز مضمون عدم تمثيله لأي نص معروف لمدير الموارد.

يمكن استخدام هذه الدوال للتحويل من النص إلى تمثيل الكود المميز. إذا لم يكن النص بترميز الأحرف المحمول (Host Portable Character Encoding)، يكون التحويل معتمدًا على التنفيذ. ليس بالضرورة أن يكون النص المُمرر إلى XrmStringToQuark مخزنًا دائمًا. XrmPermStringToQuark تشبه XrmStringToQuark، باستثناء أن Xlib يُسمح لها بافتراض أن النص المُمرر مخزن دائمًا، وبالتالي يمكن استخدامه كقيمة تُرجعها XrmQuarkToString.

لأي كود مميز معين، إذا أعادت XrmStringToQuark قيمة غير NULL، فإن جميع الاستدعاءات المستقبلية ستعيد نفس القيمة (نفس العنوان).

يمكن استخدام هذه الدوال للتحويل من تمثيل الكود المميز إلى النص. يجب عدم تعديل أو تحرير النص الذي تشير إليه القيمة المرجعة. النص المرجع يكون مطابقًا حرفًا بحرف للنص الأصلي الذي تم تمريره إلى إحدى روتينات تحويل النص إلى كود مميز. إذا لم يوجد نص لهذا الكود المميز، تُرجع XrmQuarkToString NULL. لأي كود مميز معين، إذا أعادت XrmQuarkToString قيمة غير NULL، فإن جميع الاستدعاءات المستقبلية ستعيد نفس القيمة (نفس العنوان).

يمكن استخدام هذه الدوال للتحويل من النص إلى تمثيل الكود المميز. إذا لم يكن النص بترميز الأحرف المحمول (Host Portable Character Encoding)، يكون التحويل معتمدًا على التنفيذ. ليس بالضرورة أن يكون النص المُمرر إلى XrmStringToQuark مخزنًا دائمًا. XrmPermStringToQuark تشبه XrmStringToQuark، باستثناء أن Xlib يُسمح لها بافتراض أن النص المُمرر مخزن دائمًا، وبالتالي يمكن استخدامه كقيمة تُرجعها XrmQuarkToString.

لأي كود مميز معين، إذا أعادت XrmStringToQuark قيمة غير NULL، فإن جميع الاستدعاءات المستقبلية ستعيد نفس القيمة (نفس العنوان).

تقوم الدالة XrmStringToQuarkList بتحويل النص المنتهي بـ NULL (عادةً اسم مؤهل بالكامل) إلى قائمة من الأكواد المميزة. يجب على المتصل تخصيص مساحة كافية لقائمة الأكواد المميزة قبل استدعاء XrmStringToQuarkList. لاحظ أن النص يجب أن يكون بتنسيق اسم المورد الصحيح (انظر القسم 15.1). إذا لم يكن النص بترميز الأحرف المحمول (Host Portable Character Encoding)، يكون التحويل معتمدًا على التنفيذ.

قائمة الارتباطات هي قائمة من النوع XrmBindingList وتشير إلى ما إذا كانت مكونات قوائم الأسماء أو الفئات مرتبطة بإحكام أو بشكل فضفاض (أي، ما إذا كان تجريد المكونات الوسيطة محددًا).

typedef enum {XrmBindTightly, XrmBindLoosely} XrmBinding, *XrmBindingList;

XrmBindTightly تشير إلى أن نقطة تفصل المكونات، و XrmBindLoosely تشير إلى أن علامة النجمة (*) تفصل المكونات.

تقوم الدالة XrmStringToBindingQuarkList بتحويل النص المحدد إلى قائمة ارتباطات وقائمة أكواد مميزة. يجب على المتصل تخصيص مساحة كافية لقائمة الأكواد المميزة وقائمة الارتباطات قبل استدعاء XrmStringToBindingQuarkList. إذا لم يكن النص بترميز الأحرف المحمول (Host Portable Character Encoding) يكون التحويل معتمدًا على التنفيذ. يتم فصل أسماء المكونات في القائمة بواسطة نقطة أو علامة نجمة (*). إذا لم يبدأ النص بنقطة أو علامة نجمة، يتم افتراض نقطة. على سبيل المثال، “*a.b*c” تصبح:

الأكواد المميزة a b c
الارتباطات فضفاض محكم فضفاض
