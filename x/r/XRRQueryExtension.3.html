أنواع البيانات

التدويرات/الانعكاسات

يمكن أن يكون أيًا مما يلي:

#define RR_Rotate_0             1
#define RR_Rotate_90            2
#define RR_Rotate_180           4
#define RR_Rotate_270           8
/∗ جديد في بروتوكول 1.0، للسماح بانعكاس الشاشة */
/∗ يتم تطبيق الانعكاس بعد التدوير */
#define RR_Reflect_X            16
#define RR_Reflect_Y            32
typedef struct {
    int	width, height;
    int	mwidth, mheight;
} XRRScreenSize;
typedef struct {
    int type;				/∗ قاعدة الحدث */
    unsigned long serial;	/∗ # آخر طلب تمت معالجته بواسطة الخادم */
    Bool send_event;		/∗ صحيح إذا كان هذا جاء من طلب SendEvent */
    Display *display;		/∗ العرض الذي تمت قراءة الحدث منه */
    Window window;			/∗ النافذة التي اختارت لهذا الحدث */
    Window root;			/∗ النافذة الجذر للشاشة المتغيرة */
    Time timestamp;			/∗ وقت حدوث تغيير الشاشة */
    Time config_timestamp;	/∗ وقت آخر تغيير تكوين */
    SizeID size_index;
    SubpixelOrder subpixel_order;
    Rotation rotation;
    int width;
    int height;
    int mwidth;
    int mheight;
} XRRScreenChangeNotifyEvent;
يحتوي الهيكل XRRScreenSize على حجم جذر محتمل بالبكسل وبالميليمتر.

يتم إرسال حدث XRRScreenChangeNotifyEvent إلى العميل الذي طلب الإشعار كلما تم تغيير تكوين الشاشة. يمكن للعميل إجراء هذا الطلب عن طريق استدعاء XRRSelectInput، مع تمرير العرض، والنافذة الجذر، وقناع RRScreenChangeNotifyMask.

XRRScreenConfiguration هو نوع بيانات غير شفاف يحتوي على معلومات تكوين الشاشة.

الطوابع الزمنية

يتم تضمين الطوابع الزمنية ويجب استخدامها لضمان أن العميل يلعب بمجموعة كاملة: قد تغير الشاشة الخصائص أثناء التنقل وهذا يضمن أن معرفتها بالتكوين محدثة. هذا للمساعدة في المشكلات عندما قد تصبح الشاشات قابلة للتوصيل والتشغيل في المستقبل.

الوصف

Xrandr هي مكتبة بسيطة مصممة للواجهة مع امتداد تغيير حجم وتدوير X. هذا يسمح للعملاء بتغيير حجم وتدوير النافذة الجذر للشاشة، إلى جانب القدرة على عكس الشاشة حول أي من المحورين (إذا كانت مدعومة من قبل التنفيذ). قد يتم تنفيذ التدوير والانعكاس عن طريق البرامج وقد ينتج عنها أداء أبطأ إذا تم تنفيذ التدوير والانعكاس بهذه الطريقة (كما هو الحال في جميع التنفيذات اعتبارًا من أكتوبر 2002).

تقوم مكتبة Xrandr ببعض التخزين المؤقت الأدنى لتجنب الرحلات ذهابًا وإيابًا لتزويد العملاء بمعلومات مستخدمة بشكل متكرر. انظر "امتداد تغيير حجم وتدوير X" لوصف تفصيلي؛ لاحظ أيضًا أن تبديل العمق، كما هو موصوف في المستند، لم يتم تنفيذه، وقد (أو قد لا) يتم تنفيذه أبدًا، حيث أن ذاكرة العرض تنمو بسرعة، وبدأت أطر العمل بالفعل في دعم الهجرة، مما يخفف من الحاجة إلى تبديل العمق. إذا تم تنفيذه في المستقبل، نتوقع القيام بذلك عبر امتداد متوافق تصاعديًا للمكتبة/البروتوكول الحالي؛ يجب أن تستمر الوظائف الموصوفة هنا في العمل.

يعد التدوير والانعكاس وكيفية تفاعلهما أمرًا مربكًا. في Randr، يتم تدوير نظام الإحداثيات في اتجاه عكس عقارب الساعة بالنسبة للاتجاه الطبيعي. الانعكاس يكون على طول نظام إحداثيات نظام النوافذ، وليس محاور X و Y للشاشة المادية، بحيث لا يتفاعل التدوير والانعكاس. الطريقة الأخرى للنظر في الانعكاس هي معالجته كما هو محدد في الاتجاه "الطبيعي"، قبل التدوير.

حدث XRRScreenChangeNotify يتم إرساله إلى العملاء الذين يطلبون إبلاغهم كلما تغير تكوين النافذة الجذر. قد تشمل تغييرات التكوين الدقة، الحجم المادي، ترتيب البكسلات الفرعية (انظر XRender(3))، والتدوير. لاحظ أن التغييرات على أي من هذه أو كلها قد تحدث بسبب أحداث خارجية (تحكم المستخدم في خادم X، توصيل شاشة / لوحة مسطحة مختلفة قابلة للتوصيل والتشغيل) ولا تنتج فقط عن طلب بروتوكول/مكتبة إلى خادم X.

بالإضافة إلى ذلك، للقضاء على شرط سباق محتمل، قد يتم إنشاء هذا الحدث فور اختيار الإشعار إذا تغيرت الشاشة منذ أن قام العميل Xrandr بالاتصال بخادم X، لتمكين تغيير دقة الشاشة بشكل موثوق عندما قد يقوم مستخدم بتسجيل الدخول وتغيير التكوين أثناء بدء تشغيل عميل واحد أو العديد من العملاء.

إشعار Xlib

يجب على العملاء استدعاء Xlib مرة أخرى باستخدام XRRUpdateConfiguration عند إنشاء أحداث إشعار تغيير تكوين الشاشة (أو حدوث تغييرات في تكوين النافذة الجذر، لتحديث رؤية Xlib للدقة، الحجم، التدوير، الانعكاس، أو ترتيب البكسلات الفرعية. بشكل عام، ستقوم أطر العمل بتنفيذ هذه العملية نيابة عن التطبيقات؛ لم نكن نريد تغيير بيانات بنية العرض خلف ظهور أطر العمل، حيث أنه في العملاء متعددين الخيوط، قد تحدث ظروف سباق مختلفة. يجب أن توفر أطر العمل للعملاء آلية ما للإشعار بتغيير الشاشة، بالطبع.
