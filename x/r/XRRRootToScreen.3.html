الحزمة: libxrandr

التراخيص: custom

الاسم


Xrandr - امتداد X لتغيير الحجم، الدوران، والانعكاس.

أنواع البيانات

الدورانات/الانعكاسات

يمكن أن تكون أيًا مما يلي:

#define RR_Rotate_0             1
#define RR_Rotate_90            2
#define RR_Rotate_180           4
#define RR_Rotate_270           8
/∗ جديد في بروتوكول 1.0، للسماح بانعكاس الشاشة */
/∗ يتم تطبيق الانعكاس بعد الدوران */
#define RR_Reflect_X            16
#define RR_Reflect_Y            32
typedef struct {
    int	width, height;
    int	mwidth, mheight;
} XRRScreenSize;
typedef struct {
    int type;				/∗ قاعدة الحدث */
    unsigned long serial;	/∗ # آخر طلب تمت معالجته بواسطة الخادم */
    Bool send_event;		/∗ صحيح إذا جاء هذا من طلب SendEvent */
    Display *display;		/∗ الشاشة التي تمت قراءة الحدث منها */
    Window window;			/∗ النافذة التي اختارت لهذا الحدث */
    Window root;			/∗ النافذة الجذر للشاشة المتغيرة */
    Time timestamp;			/∗ وقت حدوث تغيير الشاشة */
    Time config_timestamp;	/∗ وقت آخر تغيير في التكوين */
    SizeID size_index;
    SubpixelOrder subpixel_order;
    Rotation rotation;
    int width;
    int height;
    int mwidth;
    int mheight;
} XRRScreenChangeNotifyEvent;
يحتوي هيكل XRRScreenSize على حجم جذر محتمل بالبكسل وبالمليمتر.

XRRScreenChangeNotifyEvent يتم إرسالها إلى العميل الذي طلب الإخطار عند تغيير تكوين الشاشة. يمكن للعميل إجراء هذا الطلب عن طريق استدعاء XRRSelectInput، وتمرير العرض، والنافذة الجذر، وقناع RRScreenChangeNotifyMask.

XRRScreenConfiguration هو نوع بيانات مبهم يحتوي على معلومات تكوين الشاشة.

الطوابع الزمنية

تتضمن الطوابع الزمنية ويجب استخدامها لضمان أن العميل يلعب بمجموعة كاملة: قد تقوم الشاشة بتغيير الخصائص بشكل فوري، وهذا يضمن أن معرفتها بالتكوين حديثة. هذا للمساعدة في المشكلات عندما قد تصبح الشاشات قابلة للتوصيل الساخن في المستقبل.

الوصف

Xrandr هي مكتبة بسيطة مصممة للربط بامتداد X لتغيير الحجم والدوران. يسمح هذا للعملاء بتغيير حجم ودوران النافذة الجذر للشاشة، مع القدرة على عكس الشاشة حول أي محور (إذا كان مدعومًا من قبل التنفيذ). قد يتم تنفيذ الدوران والانعكاس بواسطة البرامج وقد ينتج عنه أداء أبطأ إذا تم تنفيذ الدوران والانعكاس بهذه الطريقة (كما هو الحال مع جميع التطبيقات اعتبارًا من أكتوبر 2002).

مكتبة Xrandr تقوم ببعض التخزين المؤقت الأدنى لتجنب الرحلات ذهابًا وإيابًا لتوفير معلومات مستخدمة بشكل متكرر للعملاء. انظر "امتداد X لتغيير الحجم والدوران" للحصول على وصف مفصل؛ لاحظ أيضًا أن تبديل العمق، كما هو موضح في الوثيقة، غير مطبق، وقد (أو قد لا) يتم تنفيذه أبدًا، نظرًا لأن ذاكرة العرض تنمو بسرعة، وبدأت أدوات العمل بالفعل في دعم الترحيل، مما يخفف الحاجة إلى تبديل العمق. إذا تم تنفيذه في المستقبل، نتوقع القيام بذلك عبر امتداد متوافق للأعلى مع المكتبة/البروتوكول الحالي؛ يجب أن تستمر الوظائف الموضحة هنا في العمل.

الدوران والانعكاس وكيفية تفاعلهما يمكن أن يكون مربكًا. في Randr، يتم تدوير نظام الإحداثيات في اتجاه عكس عقارب الساعة بالنسبة للاتجاه الطبيعي. يتم الانعكاس على طول نظام إحداثيات نظام النوافذ، وليس محاور X و Y الفعلية للشاشة، بحيث لا يتفاعل الدوران والانعكاس. الطريقة الأخرى للتفكير في الانعكاس هي معاملته كما هو محدد في الاتجاه "الطبيعي"، قبل الدوران.

يتم إرسال الحدث XRRScreenChangeNotify إلى العملاء الذين يطلبون الإعلام عند تغيير تكوين النافذة الجذر. قد تشمل تغييرات التكوين الدقة، الحجم الفعلي، ترتيب البيكسل الفرعي (انظر XRender(3))، والدوران. لاحظ أن التغييرات على أي من هذه أو كلها قد تحدث بسبب أحداث خارجية (تحكم المستخدم في خادم X، توصيل/فصل شاشة عرض مختلفة/لوحة مسطحة) وليست فقط نتيجة لطلب بروتوكول/مكتبة إلى خادم X.

بالإضافة إلى ذلك، للقضاء على مشكلة سباق محتملة، يمكن إنشاء هذا الحدث فور تحديد الإخطار إذا تغيرت الشاشة منذ أن قام العميل Xrandr بالاتصال بخادم X، لتمكين تغيير دقة الشاشة بشكل موثوق عندما قد يقوم المستخدم بتسجيل الدخول وتغيير التكوين بينما يبدأ عميل واحد أو أكثر في العمل.

إخطار Xlib

يجب على العملاء استدعاء Xlib مرة أخرى باستخدام XRRUpdateConfiguration عند إنشاء أحداث إشعار تغيير تكوين الشاشة (أو حدوث تغييرات في تكوين النافذة الجذر، لتحديث رؤية Xlib للدقة، الحجم، الدوران، الانعكاس، أو ترتيب البيكسل الفرعي. بشكل عام، ستقوم أدوات العمل بإجراء هذه العملية نيابة عن التطبيقات؛ لم نكن نرغب في تغيير بيانات هيكل العرض خلف ظهر أدوات العمل، حيث قد تحدث ظروف سباق مختلفة في العملاء متعددة الخيوط. يجب على أدوات العمل توفير آلية للعملاء للإبلاغ عن تغيير الشاشة، بالطبع.
