XrmStringToBindingQuarkList(3)

الحزمة: libx11

التراخيص: MIT AND X11

XrmUniqueQuark(3) دوال XLIB XrmUniqueQuark(3)

الاسم

XrmUniqueQuark, XrmStringToQuark, XrmPermStringToQuark, XrmQuarkToString, XrmStringToQuarkList, XrmStringToBindingQuarkList - التعامل مع الكواركات (Quarks) للموارد

ملخص

#include <X11/Xresource.h>

XrmQuark XrmUniqueQuark(void);

#define XrmStringToName(string) XrmStringToQuark(string)
#define XrmStringToClass(string) XrmStringToQuark(string)
#define XrmStringToRepresentation(string) XrmStringToQuark(string)

XrmQuark XrmStringToQuark(char *string);

XrmQuark XrmPermStringToQuark(_Xconst char *string);

#define XrmStringToName(string) XrmStringToQuark(string)
#define XrmStringToClass(string) XrmStringToQuark(string)
#define XrmStringToRepresentation(string) XrmStringToQuark(string)

XrmQuark XrmStringToQuark(char *string);

XrmQuark XrmPermStringToQuark(_Xconst char *string);

#define XrmNameToString(name) XrmQuarkToString(name)
#define XrmClassToString(class) XrmQuarkToString(class)
#define XrmRepresentationToString(type) XrmQuarkToString(type)

char *XrmQuarkToString(XrmQuark quark);

#define XrmStringToNameList(str, name) XrmStringToQuarkList((str), (name))
#define XrmStringToClassList(str,class) XrmStringToQuarkList((str), (class))

void XrmStringToQuarkList(char *string, XrmQuarkList quarks_return);

void XrmStringToBindingQuarkList(_Xconst char *string, XrmBindingList bindings_return, XrmQuarkList quarks_return);

المعاملات

bindings_return
يعيد قائمة الربط (binding list).
quark
يحدد الكوارك الذي يرغب في الحصول على السلسلة النصية المكافئة له.
quarks_return
يعيد قائمة الكواركات.
string
يحدد السلسلة النصية التي سيتم تخصيص كوارك أو قائمة كواركات لها.

الوصف

تقوم الدالة XrmUniqueQuark بتخصيص كوارك يضمن أنه لا يمثل أي سلسلة نصية معروفة لمدير الموارد.

يمكن استخدام هذه الدوال للتحويل من سلسلة نصية إلى تمثيل كوارك. إذا لم تكن السلسلة النصية في ترميز الحروف المحمول المضيف (Host Portable Character Encoding)، فإن التحويل يعتمد على التنفيذ. السلسلة النصية المعاملة إلى XrmStringToQuark لا تحتاج إلى أن تكون تخزينًا مخصصًا بشكل دائم. XrmPermStringToQuark هي مثل XrmStringToQuark، باستثناء أن Xlib مسموح لها بالافتراض بأن السلسلة النصية المعاملة مخصصة بشكل دائم، وبالتالي، يمكن استخدامها كقيمة سيتم إرجاعها بواسطة XrmQuarkToString.

لأي كوارك معين، إذا أعادت XrmStringToQuark قيمة غير NULL، فإن جميع الاستدعاءات المستقبلية ستعيد نفس القيمة (عنوان مطابق).

يمكن استخدام هذه الدوال للتحويل من تمثيل كوارك إلى سلسلة نصية. يجب عدم تعديل السلسلة النصية التي تشير إليها القيمة المرجعة أو تحريرها. السلسلة النصية المرجعة متطابقة حرفًا بحرف مع السلسلة النصية الأصلية التي تم تمريرها إلى إحدى روتين تحويل السلسلة إلى كوارك. إذا لم توجد سلسلة نصية لذلك الكوارك، فإن XrmQuarkToString تعيد NULL. لأي كوارك معين، إذا أعادت XrmQuarkToString قيمة غير NULL، فإن جميع الاستدعاءات المستقبلية ستعيد نفس القيمة (عنوان مطابق).

يمكن استخدام هذه الدوال للتحويل من سلسلة نصية إلى تمثيل كوارك. إذا لم تكن السلسلة النصية في ترميز الحروف المحمول المضيف (Host Portable Character Encoding)، فإن التحويل يعتمد على التنفيذ. السلسلة النصية المعاملة إلى XrmStringToQuark لا تحتاج إلى أن تكون تخزينًا مخصصًا بشكل دائم. XrmPermStringToQuark هي مثل XrmStringToQuark، باستثناء أن Xlib مسموح لها بالافتراض بأن السلسلة النصية المعاملة مخصصة بشكل دائم، وبالتالي، يمكن استخدامها كقيمة سيتم إرجاعها بواسطة XrmQuarkToString.

لأي كوارك معين، إذا أعادت XrmStringToQuark قيمة غير NULL، فإن جميع الاستدعاءات المستقبلية ستعيد نفس القيمة (عنوان مطابق).

تقوم الدالة XrmStringToQuarkList بتحويل السلسلة النصية المنتهية بـ NULL (عادةً اسم مؤهل بالكامل) إلى قائمة من الكواركات. يجب على المستدعي تخصيص مساحة كافية لقائمة الكواركات قبل استدعاء XrmStringToQuarkList. لاحظ أن السلسلة النصية يجب أن تكون بتنسيق اسم المورد الصحيح (راجع القسم 15.1). إذا لم تكن السلسلة النصية في ترميز الحروف المحمول المضيف (Host Portable Character Encoding)، فإن التحويل يعتمد على التنفيذ.

قائمة الربط هي قائمة من النوع XrmBindingList وتشير ما إذا كانت مكونات قائمة الاسم أو الفئة مرتبطة بإحكام أو بشكل فضفاض (أي، ما إذا كان الاستخدام الجامح للمكونات الوسيطة محددًا).

typedef enum {XrmBindTightly, XrmBindLoosely} XrmBinding, *XrmBindingList;

XrmBindTightly تشير إلى أن فاصلة تفصل بين المكونات، و XrmBindLoosely تشير إلى أن نجمة تفصل بين المكونات.

تقوم الدالة XrmStringToBindingQuarkList بتحويل السلسلة النصية المحددة إلى قائمة ربط وقائمة كواركات. يجب على المستدعي تخصيص مساحة كافية لقائمة الكواركات وقائمة الربط قبل استدعاء XrmStringToBindingQuarkList. إذا لم تكن السلسلة النصية في ترميز الحروف المحمول المضيف (Host Portable Character Encoding)، فإن التحويل يعتمد على التنفيذ. يتم فصل أسماء المكونات في القائمة بواسطة فاصلة أو حرف نجمة. إذا لم تبدأ السلسلة النصية بفاصلة أو نجمة، فسيتم افتراض فاصلة. على سبيل المثال، "“*a.b*c”" تصبح:

الكواركات a b c
الروابط فضفاض محكم فضفاض
