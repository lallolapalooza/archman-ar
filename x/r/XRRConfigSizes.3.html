أنواع البيانات

التدويرات/الانعكاسات

يمكن أن تكون أيًا مما يلي:

#define RR_Rotate_0             1
#define RR_Rotate_90            2
#define RR_Rotate_180           4
#define RR_Rotate_270           8
/* new in 1.0 protocol, to allow reflection of screen */
/* reflection is applied after rotation */
#define RR_Reflect_X            16
#define RR_Reflect_Y            32
typedef struct {
    int	width, height;
    int	mwidth, mheight;
} XRRScreenSize;
typedef struct {
    int type;				/* event base */
    unsigned long serial;	/* # of last request processed by server */
    Bool send_event;		/* true if this came from a SendEvent request */
    Display *display;		/* Display the event was read from */
    Window window;			/* window which selected for this event */
    Window root;			/* Root window for changed screen */
    Time timestamp;			/* when the screen change occurred */
    Time config_timestamp;	/* when the last configuration change */
    SizeID size_index;
    SubpixelOrder subpixel_order;
    Rotation rotation;
    int width;
    int height;
    int mwidth;
    int mheight;
} XRRScreenChangeNotifyEvent;
يحتوي الهيكل XRRScreenSize على حجم جذر محتمل بالبكسل وبالميليمتر.

يتم إرسال حدث XRRScreenChangeNotifyEvent إلى العملاء الذين طلبوا الإشعار كلما تم تغيير تكوين الشاشة. يمكن للعميل إجراء هذا الطلب عن طريق استدعاء XRRSelectInput، مع تمرير العرض، ونافذة الجذر، وقناع RRScreenChangeNotifyMask.

XRRScreenConfiguration هو نوع بيانات غير شفاف يحتوي على معلومات التكوين للشاشة.

الطوابع الزمنية

يتم تضمين الطوابع الزمنية ويجب استخدامها للتأكد من أن العميل يلعب بجميع الأوراق: قد تغير الشاشة خصائصها أثناء التنقل، وهذا يضمن أن معرفتها بالتكوين محدثة. هذا للمساعدة في معالجة المشكلات عندما قد تصبح الشاشات قابلة للتوصيل السريع في المستقبل.

الوصف

Xrandr هي مكتبة بسيطة مصممة للتعامل مع امتداد X لتغيير الحجم والتدوير. يتيح هذا للعملاء تغيير حجم وتدوير النافذة الجذر للشاشة، إلى جانب القدرة على عكس الشاشة حول أي محور (إذا كان مدعومًا من قبل التنفيذ). قد يتم تنفيذ التدوير والانعكاس بواسطة البرامج وقد يؤدي إلى أداء أبطأ إذا تم تنفيذ التدوير والانعكاس بهذه الطريقة (كما هو الحال مع جميع التنفيذات اعتبارًا من أكتوبر 2002).

تقوم مكتبة Xrandr بتخزين مؤقت بسيط لتجنب الرحلات ذهابًا وإيابًا لتزويد العملاء بمعلومات مستخدمة بشكل متكرر. راجع "امتداد X لتغيير الحجم والتدوير" لوصف مفصل؛ لاحظ أيضًا أن تبديل العمق، كما هو موضح في المستند، غير مطبق، وقد (أو قد لا) يتم تنفيذه أبدًا، نظرًا لتزايد ذاكرة العرض بسرعة، وبدأت مجموعات الأدوات بالفعل في دعم الترحيل، مما يخفف من الحاجة إلى تبديل العمق. إذا تم تنفيذه في المستقبل، نتوقع القيام بذلك عبر امتداد متوافق للخلف للمكتبة/البروتوكول الحالي؛ يجب أن تستمر الوظائف الموضحة هنا في العمل.

التدوير والانعكاس وكيفية تفاعلهما يمكن أن يكون مربكًا. في Randr، يتم تدوير نظام الإحداثيات في اتجاه عكس عقارب الساعة بالنسبة للاتجاه الطبيعي. يتم الانعكاس على طول نظام إحداثيات نظام النوافذ، وليس محاور X و Y للشاشة المادية، بحيث لا يتفاعل التدوير والانعكاس. الطريقة الأخرى للنظر في الانعكاس هي معاملته كما هو محدد في الاتجاه "الطبيعي"، قبل التدوير.

يتم إرسال حدث XRRScreenChangeNotify إلى العملاء الذين يطلبون الإبلاغ كلما تغير تكوين النافذة الجذر. قد تشمل تغييرات التكوين الدقة، والحجم المادي، وترتيب البكسل الفرعي (انظر XRender(3))، والتدوير. لاحظ أن التغييرات في أي من هذه أو كلها يمكن أن تحدث بسبب أحداث خارجية (تحكم المستخدم في خادم X، توصيل شاشة / لوحة عرض مختلفة مباشرة) وليست فقط نتيجة لطلب بروتوكول/مكتبة إلى خادم X.

بالإضافة إلى ذلك، للقضاء على مشكلة سباق محتملة، يمكن إنشاء هذا الحدث فور تحديد الإشعار إذا تغيرت الشاشة منذ اتصال العميل بـ Xrandr بخادم X، لتمكين تغيير دقة الشاشة بشكل موثوق عندما قد يقوم المستخدم بتسجيل الدخول وتغيير التكوين بينما يبدأ تشغيل عميل واحد أو أكثر.

إشعار Xlib

يجب على العملاء الاستدعاء مرة أخرى إلى Xlib باستخدام XRRUpdateConfiguration عند إنشاء أحداث إشعار تغيير تكوين الشاشة (أو حدوث تغييرات تكوين النافذة الجذر، لتحديث رؤية Xlib للدقة، والحجم، والتدوير، والانعكاس، أو ترتيب البكسل الفرعي. بشكل عام، ستقوم مجموعات الأدوات بتنفيذ هذه العملية نيابة عن التطبيقات؛ لم نرغب في تغيير بيانات بنية العرض خلف كواليس مجموعات الأدوات، حيث قد تحدث ظروف سباق مختلفة في العملاء متعددين الخيوط. يجب أن توفر مجموعات الأدوات للعملاء آلية ما للإشعار بتغيير الشاشة، بالطبع.
