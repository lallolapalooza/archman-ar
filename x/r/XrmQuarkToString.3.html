XrmQuarkToString(3)

الحزمة: libx11

التراخيص: MIT AND X11

XrmUniqueQuark(3) وظائف XLIB XrmUniqueQuark(3)

الاسم

XrmUniqueQuark, XrmStringToQuark, XrmPermStringToQuark, XrmQuarkToString, XrmStringToQuarkList, XrmStringToBindingQuarkList - التعامل مع موارد الكواركات (quarks)

الملخص

#include <X11/Xresource.h>

XrmQuark XrmUniqueQuark(void);

#define XrmStringToName(string) XrmStringToQuark(string)
#define XrmStringToClass(string) XrmStringToQuark(string)
#define XrmStringToRepresentation(string) XrmStringToQuark(string)

XrmQuark XrmStringToQuark(char *string);

XrmQuark XrmPermStringToQuark(_Xconst char *string);

#define XrmStringToName(string) XrmStringToQuark(string)
#define XrmStringToClass(string) XrmStringToQuark(string)
#define XrmStringToRepresentation(string) XrmStringToQuark(string)

XrmQuark XrmStringToQuark(char *string);

XrmQuark XrmPermStringToQuark(_Xconst char *string);

#define XrmNameToString(name) XrmQuarkToString(name)
#define XrmClassToString(class) XrmQuarkToString(class)
#define XrmRepresentationToString(type) XrmQuarkToString(type)

char *XrmQuarkToString(XrmQuark quark);

#define XrmStringToNameList(str, name) XrmStringToQuarkList((str), (name))
#define XrmStringToClassList(str,class) XrmStringToQuarkList((str), (class))

void XrmStringToQuarkList(char *string, XrmQuarkList quarks_return);

void XrmStringToBindingQuarkList(_Xconst char *string, XrmBindingList bindings_return, XrmQuarkList quarks_return);

المعاملات

quarks_return

الوصف

تقوم الدالة XrmUniqueQuark بتخصيص كوارك مضمون لعدم تمثيل أي سلسلة معروفة لمدير الموارد.

يمكن استخدام هذه الدوال للتحويل من سلسلة إلى تمثيل الكوارك. إذا لم تكن السلسلة في ترميز الأحرف المحمول للنظام (Host Portable Character Encoding)، فإن التحويل يعتمد على التنفيذ. لا تحتاج السلسلة الوسيطة إلى XrmStringToQuark إلى مساحة تخزين مخصصة بشكل دائم. XrmPermStringToQuark تشبه XrmStringToQuark، باستثناء أن Xlib مسموح لها بافتراض أن الوسيط السلسلة مخصص بشكل دائم، وبالتالي، يمكن استخدامه كقيمة يتم إرجاعها بواسطة XrmQuarkToString.

لأي كوارك معين، إذا أعادت XrmStringToQuark قيمة غير NULL، فإن جميع الاستدعاءات المستقبلية ستعيد نفس القيمة (عنوان مطابق).

يمكن استخدام هذه الدوال للتحويل من تمثيل الكوارك إلى سلسلة. يجب عدم تعديل السلسلة التي تشير إليها القيمة المرجعة أو تحريرها. تكون السلسلة المرجعة مطابقة حرفًا بحرف للسلسلة الأصلية التي تم تمريرها إلى إحدى روتينيات تحويل السلسلة إلى كوارك. إذا لم توجد سلسلة لهذا الكوارك، فإن XrmQuarkToString تعيد NULL. لأي كوارك معين، إذا أعادت XrmQuarkToString قيمة غير NULL، فإن جميع الاستدعاءات المستقبلية ستعيد نفس القيمة (عنوان مطابق).

يمكن استخدام هذه الدوال للتحويل من سلسلة إلى تمثيل الكوارك. إذا لم تكن السلسلة في ترميز الأحرف المحمول للنظام (Host Portable Character Encoding)، فإن التحويل يعتمد على التنفيذ. لا تحتاج السلسلة الوسيطة إلى XrmStringToQuark إلى مساحة تخزين مخصصة بشكل دائم. XrmPermStringToQuark تشبه XrmStringToQuark، باستثناء أن Xlib مسموح لها بافتراض أن الوسيط السلسلة مخصص بشكل دائم، وبالتالي، يمكن استخدامه كقيمة يتم إرجاعها بواسطة XrmQuarkToString.

لأي كوارك معين، إذا أعادت XrmStringToQuark قيمة غير NULL، فإن جميع الاستدعاءات المستقبلية ستعيد نفس القيمة (عنوان مطابق).

تقوم الدالة XrmStringToQuarkList بتحويل السلسلة المنتهية بـ NULL (عادةً اسم مؤهل بالكامل) إلى قائمة من الكواركات. يجب على المتصل تخصيص مساحة كافية لقائمة الكواركات قبل استدعاء XrmStringToQuarkList. لاحظ أن السلسلة يجب أن تكون بتنسيق اسم المورد الصحيح (انظر القسم 15.1). إذا لم تكن السلسلة في ترميز الأحرف المحمول للنظام (Host Portable Character Encoding)، فإن التحويل يعتمد على التنفيذ.

قائمة الربط هي قائمة من النوع XrmBindingList وتشير ما إذا كانت مكونات قوائم الاسم أو الفئة مرتبطة بإحكام أو بشكل فضفاض (أي، إذا تم تحديد استخدام علامة البدل للمكونات الوسيطة).

typedef enum {XrmBindTightly, XrmBindLoosely} XrmBinding, *XrmBindingList;

XrmBindTightly تشير إلى أن نقطة تفصل بين المكونات، و XrmBindLoosely تشير إلى أن نجمة تفصل بين المكونات.

تقوم الدالة XrmStringToBindingQuarkList بتحويل السلسلة المحددة إلى قائمة ربط وقائمة كواركات. يجب على المتصل تخصيص مساحة كافية لقائمة الكواركات وقائمة الربط قبل استدعاء XrmStringToBindingQuarkList. إذا لم تكن السلسلة في ترميز الأحرف المحمول للنظام (Host Portable Character Encoding) فإن التحويل يعتمد على التنفيذ. يتم فصل أسماء المكونات في القائمة بواسطة نقطة أو علامة نجمة. إذا لم تبدأ السلسلة بنقطة أو نجمة، فيتم افتراض نقطة. على سبيل المثال، "“*a.b*c”" تصبح:

الكواركات a b c
الروابط فضفاض محكم فضفاض
