Xrandr - امتداد تغيير حجم وتدوير وعكس الشاشة في X.

أنواع البيانات

التدويرات/الانعكاسات

يمكن أن تكون أيًا مما يلي:

#define RR_Rotate_0             1
#define RR_Rotate_90            2
#define RR_Rotate_180           4
#define RR_Rotate_270           8
/∗ جديد في بروتوكول 1.0، للسماح بانعكاس الشاشة */
/∗ يتم تطبيق الانعكاس بعد التدوير */
#define RR_Reflect_X            16
#define RR_Reflect_Y            32
typedef struct {
    int	width, height;
    int	mwidth, mheight;
} XRRScreenSize;
typedef struct {
    int type;				/∗ قاعدة الأحداث */
    unsigned long serial;	/∗ # آخر طلب تمت معالجته بواسطة الخادم */
    Bool send_event;		/∗ صحيح إذا جاء هذا من طلب SendEvent */
    Display *display;		/∗ الشاشة التي تمت قراءة الحدث منها */
    Window window;			/∗ النافذة التي اختارت لهذا الحدث */
    Window root;			/∗ نافذة الجذر للشاشة المتغيرة */
    Time timestamp;			/∗ وقت حدوث تغيير الشاشة */
    Time config_timestamp;	/∗ وقت آخر تغيير في التكوين */
    SizeID size_index;
    SubpixelOrder subpixel_order;
    Rotation rotation;
    int width;
    int height;
    int mwidth;
    int mheight;
} XRRScreenChangeNotifyEvent;
يحتوي الهيكل XRRScreenSize على حجم جذر محتمل بالبكسل وبالملمتر.

XRRScreenChangeNotifyEvent يتم إرسالها إلى العملاء الذين طلبوا إعلامًا عند تغيير تكوين الشاشة. يمكن للعميل إجراء هذا الطلب عن طريق استدعاء XRRSelectInput، وتمرير العرض، ونافذة الجذر، وقناع RRScreenChangeNotifyMask.

XRRScreenConfiguration هو نوع بيانات غير شفاف يحتوي على معلومات تكوين الشاشة.

الأوقات

يتم تضمين الأوقات ويجب استخدامها للتأكد من أن العميل يلعب بأوراق كاملة: قد تغير الشاشة الخصائص أثناء التنقل ويضمن ذلك أن معرفته بالتكوين محدثة. هذا للمساعدة في المشكلات عندما قد تصبح الشاشات قابلة للتوصيل الساخن في المستقبل.

الوصف

Xrandr هي مكتبة بسيطة مصممة للربط بامتداد تغيير حجم وتدوير X. يسمح هذا للعملاء بتغيير حجم وتدوير نافذة الجذر للشاشة، جنبًا إلى جنب مع القدرة على عكس الشاشة حول أي محور (إذا كان مدعومًا من قبل التنفيذ). قد يتم تنفيذ التدوير والانعكاس بواسطة البرامج وقد ينتج عن ذلك أداء أبطأ إذا تم تنفيذ التدوير والانعكاس بهذه الطريقة (كما هو الحال مع جميع التطبيقات اعتبارًا من أكتوبر 2002).

لا تقوم مكتبة Xrandr بإجراء تخزين مؤقت بسيط لتجنب رحلات الذهاب والإياب لتزويد العملاء بمعلومات مستخدمة بشكل متكرر. راجع "امتداد تغيير حجم وتدوير X" للحصول على وصف تفصيلي؛ لاحظ أيضًا أن تبديل العمق، كما هو موضح في الوثيقة، لم يتم تنفيذه، وقد (أو قد لا) يتم تنفيذه على الإطلاق، حيث تتزايد ذاكرة العرض بسرعة، وبدأت أدوات التطوير بالفعل في دعم الترحيل، مما يقلل من الحاجة إلى تبديل العمق. إذا تم تنفيذه في المستقبل، نتوقع القيام بذلك عبر امتداد متوافق صعوديًا للمكتبة/البروتوكول الحالي؛ يجب أن تستمر الوظائف الموضحة هنا في العمل.

التدوير والانعكاس وكيفية تفاعلهما قد يكون مربكًا. في Randr، يتم تدوير نظام الإحداثيات في اتجاه عكس عقارب الساعة بالنسبة للاتجاه الطبيعي. يتم الانعكاس على طول نظام إحداثيات نظام النوافذ، وليس محور X و Y للشاشة الفعلية، بحيث لا يتفاعل التدوير والانعكاس. الطريقة الأخرى للنظر في الانعكاس هي معاملته كما هو محدد في الاتجاه "الطبيعي"، قبل التدوير.

يتم إرسال حدث XRRScreenChangeNotify إلى العملاء الذين يطلبون إعلامهم كلما تغير تكوين نافذة الجذر. قد تشمل تغييرات التكوين الدقة، والحجم الفعلي، وترتيب البكسلات الفرعية (انظر XRender(3))، والتدوير. لاحظ أن التغييرات على أي من هذه أو كلها قد تحدث بسبب أحداث خارجية (تحكم المستخدم في خادم X، توصيل/فصل شاشة/لوحة مسطحة خارجية) وليست فقط نتيجة لطلب بروتوكول/مكتبة إلى خادم X.

بالإضافة إلى ذلك، للقضاء على شرط سباق محتمل، يمكن إنشاء هذا الحدث فور اختيار الإخطار إذا تغيرت الشاشة منذ أن اتصل عميل Xrandr بخادم X، لتمكين تغيير دقة الشاشة بشكل موثوق عندما قد يقوم المستخدم بتسجيل الدخول وتغيير التكوين أثناء بدء تشغيل عميل واحد أو أكثر.

إشعار Xlib

يجب على العملاء استدعاء Xlib مرة أخرى باستخدام XRRUpdateConfiguration عند إنشاء أحداث إعلام تغيير تكوين الشاشة (أو حدوث تغييرات في تكوين نافذة الجذر، لتحديث رؤية Xlib للدقة، والحجم، والتدوير، والانعكاس، أو ترتيب البكسلات الفرعية. بشكل عام، ستقوم أدوات التطوير بإجراء هذه العملية نيابة عن التطبيقات؛ لم نكن نرغب في تغيير بيانات هيكل العرض خلف ظهور أدوات التطوير، حيث قد تحدث ظروف سباق مختلفة في العملاء متعددي الخيوط. يجب أن توفر أدوات التطوير للعملاء آلية للإخطار بتغيير الشاشة، بالطبع.
