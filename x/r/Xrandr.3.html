Xrandr - امتداد X لتغيير الحجم والدوران والانعكاس.

أنواع البيانات

الدورانات/الانعكاسات

يمكن أن يكون أيًا مما يلي:

#define RR_Rotate_0             1
#define RR_Rotate_90            2
#define RR_Rotate_180           4
#define RR_Rotate_270           8
/∗ جديد في بروتوكول 1.0، للسماح بانعكاس الشاشة */
/∗ يتم تطبيق الانعكاس بعد الدوران */
#define RR_Reflect_X            16
#define RR_Reflect_Y            32
typedef struct {
    int	width, height;
    int	mwidth, mheight;
} XRRScreenSize;
typedef struct {
    int type;				/∗ قاعدة الحدث */
    unsigned long serial;	/∗ # آخر طلب تمت معالجته بواسطة الخادم */
    Bool send_event;		/∗ صحيح إذا جاء هذا من طلب SendEvent */
    Display *display;		/∗ العرض الذي تمت قراءة الحدث منه */
    Window window;			/∗ النافذة التي تم اختيار هذا الحدث لها */
    Window root;			/∗ النافذة الجذر للشاشة المتغيرة */
    Time timestamp;			/∗ متى حدث تغيير الشاشة */
    Time config_timestamp;	/∗ متى تم آخر تغيير تكوين */
    SizeID size_index;
    SubpixelOrder subpixel_order;
    Rotation rotation;
    int width;
    int height;
    int mwidth;
    int mheight;
} XRRScreenChangeNotifyEvent;
يحتوي الهيكل XRRScreenSize على حجم جذر محتمل بالبكسل وبالمليمتر.

يتم إرسال حدث XRRScreenChangeNotifyEvent إلى العملاء الذين طلبوا الإخطار عند تغيير تكوين الشاشة. يمكن للعميل إجراء هذا الطلب عن طريق استدعاء XRRSelectInput، وتمرير العرض، والنافذة الجذر، وقناع RRScreenChangeNotifyMask.

XRRScreenConfiguration هو نوع بيانات غير شفاف يحتوي على معلومات التكوين للشاشة.

الطوابع الزمنية

يتم تضمين الطوابع الزمنية ويجب استخدامها لضمان أن العميل يعمل بمعلومات كاملة: قد تتغير خصائص الشاشة أثناء التشغيل وهذا يضمن أن معرفتها بالتكوين محدثة. هذا للمساعدة في المشكلات عندما قد تكون الشاشات قابلة للتبديل السريع في المستقبل.

الوصف

Xrandr هي مكتبة بسيطة مصممة للتفاعل مع امتداد X لتغيير الحجم والدوران. يسمح هذا للعملاء بتغيير حجم ودوران النافذة الجذر للشاشة، بالإضافة إلى القدرة على عكس الشاشة حول أي من المحورين (إذا كان مدعومًا من قبل التنفيذ). قد يتم تنفيذ الدوران والانعكاس بواسطة البرامج وقد يؤدي ذلك إلى أداء أبطأ إذا تم تنفيذ الدوران والانعكاس بهذه الطريقة (كما هو الحال مع جميع عمليات التنفيذ اعتبارًا من أكتوبر 2002).

لا تقوم مكتبة Xrandr بإجراء تخزين مؤقت بسيط لتجنب الرحلات ذهابًا وإيابًا لتوفير معلومات شائعة الاستخدام للعملاء. انظر "امتداد X لتغيير الحجم والدوران" للحصول على وصف مفصل؛ لاحظ أيضًا أن تبديل العمق، كما هو موضح في الوثيقة، لم يتم تنفيذه، وقد لا يتم تنفيذه أبدًا (أو قد لا يتم)، حيث تزداد ذاكرة العرض بسرعة، وبدأت مجموعات الأدوات بالفعل في دعم الترحيل، مما يقلل من الحاجة إلى تبديل العمق. إذا تم تنفيذه في المستقبل، نتوقع القيام بذلك عبر امتداد متوافق مع الإصدارات السابقة للمكتبة/البروتوكول الحالي؛ يجب أن تستمر الوظائف الموضحة هنا في العمل.

يمكن أن يكون الدوران والانعكاس وكيفية تفاعلهما مربكًا. في Randr، يتم تدوير نظام الإحداثيات في اتجاه عكس عقارب الساعة بالنسبة للاتجاه الطبيعي. يحدث الانعكاس على طول نظام إحداثيات نظام النوافذ، وليس على المحورين X و Y للشاشة الفعلية، بحيث لا يتفاعل الدوران والانعكاس. الطريقة الأخرى للنظر في الانعكاس هي معاملته كما هو محدد في الاتجاه "الطبيعي"، قبل الدوران.

يتم إرسال حدث XRRScreenChangeNotify إلى العملاء الذين يطلبون إبلاغهم متى يتغير تكوين النافذة الجذر. قد تشمل تغييرات التكوين الدقة، الحجم الفعلي، ترتيب البكسلات الفرعية (انظر XRender(3))، والدوران. لاحظ أن التغييرات في أي من هذه أو كلها قد تحدث بسبب أحداث خارجية (تحكم المستخدم في خادم X، توصيل/فصل شاشة/لوحة مسطحة أخرى) وليست فقط نتيجة لطلب بروتوكول/مكتبة إلى خادم X.

بالإضافة إلى ذلك، للقضاء على حالة سباق محتملة، يمكن إنشاء هذا الحدث فور اختيار الإخطار إذا تغيرت الشاشة منذ أن اتصل عميل Xrandr بخادم X، لتمكين تغيير دقة الشاشة بشكل موثوق عندما قد يقوم المستخدم بتسجيل الدخول وتغيير التكوين أثناء بدء تشغيل عميل واحد أو أكثر.

إخطار Xlib

يجب على العملاء العودة إلى Xlib باستخدام XRRUpdateConfiguration عند إنشاء أحداث الإخطار بتغيير تكوين الشاشة (أو حدوث تغييرات في تكوين النافذة الجذر، لتحديث عرض Xlib للدقة، الحجم، الدوران، الانعكاس، أو ترتيب البكسلات الفرعية. بشكل عام، ستقوم مجموعات الأدوات بأداء هذه العملية نيابة عن التطبيقات؛ لم نرغب في تغيير بيانات بنية العرض خلف ظهور مجموعات الأدوات، حيث قد تحدث ظروف سباق مختلفة في العملاء متعددي الخيوط. يجب أن توفر مجموعات الأدوات للعملاء بعض آلية للإخطار بتغيير الشاشة، بالطبع.
