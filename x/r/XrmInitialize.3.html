XrmInitialize(3)

الحزمة: libx11

التراخيص: MIT AND X11

XrmInitialize(3) وظائف XLIB XrmInitialize(3)

الاسم

XrmInitialize, XrmParseCommand, XrmValue, XrmOptionKind, XrmOptionDescRec - تهيئة مدير الموارد، هياكل مدير الموارد، وتحليل سطر الأوامر

الصيغة

#include <X11/Xresource.h>

void XrmInitialize(void);

void XrmParseCommand(XrmDatabase *database, XrmOptionDescList table, int table_count, _Xconst char *name, int *argc_in_out, char **argv_in_out);

الوسائط

argc_in_out
يحدد عدد الوسائط ويعيد عدد الوسائط المتبقية.
argv_in_out
يحدد وسائط سطر الأوامر ويعيد الوسائط المتبقية.
database
يحدد قاعدة بيانات الموارد.
name
يحدد اسم التطبيق.
table
يحدد جدول وسائط سطر الأوامر المراد تحليلها.
table_count
يحدد عدد الإدخالات في الجدول.

الوصف

تقوم الدالة XrmInitialize بتهيئة مدير الموارد. يجب استدعاؤها قبل استخدام أي دوال Xrm أخرى.

تقوم الدالة XrmParseCommand بتحليل زوج (argc, argv) وفقًا لجدول الخيارات المحدد، وتحميل الخيارات المعترف بها في قاعدة البيانات المحددة بنوع "String"، وتعديل زوج (argc, argv) لإزالة جميع الخيارات المعترف بها. إذا كانت قاعدة البيانات تحتوي على NULL، تقوم XrmParseCommand بإنشاء قاعدة بيانات جديدة وتعيد مؤشرًا إليها. بخلاف ذلك، تتم إضافة الإدخالات إلى قاعدة البيانات المحددة. إذا تم إنشاء قاعدة بيانات، فسيتم إنشاؤها في البيئة المحلية الحالية.

يتم استخدام الجدول المحدد لتحليل سطر الأوامر. تتم إزالة الخيارات المعترف بها في الجدول من argv، وتتم إضافة الإدخالات إلى قاعدة بيانات الموارد المحددة بالترتيب الذي تظهر به في argv. تحتوي إدخالات الجدول على معلومات حول سلسلة الخيار، اسم الخيار، نمط الخيار، وقيمة لتوفيرها إذا كان نوع الخيار هو XrmoptionNoArg. تتم مقارنة أسماء الخيارات بايت ببايت مع الوسائط في argv، بشكل مستقل عن أي بيئة محلية. يتم تخزين قيم الموارد المعطاة في الجدول في قاعدة بيانات الموارد دون تعديل. يتم إنشاء جميع إدخالات قاعدة بيانات الموارد باستخدام تمثيل من النوع "String". تحدد وسيطة argc عدد الوسائط في argv ويتم تعيينها عند العودة إلى العدد المتبقي من الوسائط التي لم يتم تحليلها. يجب أن تكون الوسيطة name هي اسم تطبيقك لاستخدامها في بناء إدخال قاعدة البيانات. يتم إضافة الوسيطة name إلى resourceName في جدول الخيارات قبل تخزين إدخال قاعدة البيانات. يتم التعامل مع الوسيطة name كمكون واحد، حتى لو كانت تحتوي على نقاط مضمنة. لا يتم إدراج حرف فاصل (ربط)، لذلك يجب أن يحتوي الجدول على نقطة (.) أو علامة نجمة (*) كأول حرف في كل إدخال resourceName. لتحديد اسم مورد مؤهل بشكل كامل، يمكن أن يحتوي إدخال resourceName على مكونات متعددة. إذا لم تكن الوسيطة name و resourceNames بتشفير الأحرف المحمولة للمضيف (Host Portable Character Encoding)، فإن النتيجة تعتمد على التنفيذ.

الهياكل

تحتوي الهياكل XrmValue و XrmOptionKind و XrmOptionDescRec على:

typedef struct {
        unsigned int size;
        XPointer addr;
} XrmValue, *XrmValuePtr;
typedef enum {
        XrmoptionNoArg, /* القيمة محددة في XrmOptionDescRec.value */
        XrmoptionIsArg, /* القيمة هي سلسلة الخيار نفسها */
        XrmoptionStickyArg,     /* القيمة هي الأحرف التي تلي الخيار مباشرة */
        XrmoptionSepArg,        /* القيمة هي الوسيطة التالية في argv */
        XrmoptionResArg,        /* المورد والقيمة في الوسيطة التالية في argv */
        XrmoptionSkipArg,       /* تجاهل هذا الخيار والوسيطة التالية في argv */
        XrmoptionSkipLine,      /* تجاهل هذا الخيار وبقية argv */
        XrmoptionSkipNArgs      /* تجاهل هذا الخيار والوسائط
                   XrmOptionDescRec.value التالية في argv */
} XrmOptionKind;
typedef struct {
        char *option;   /* سلسلة تحديد الخيار في argv                    */
        char *specifier;        /* الربط واسم المورد (بدون اسم التطبيق)    */
        XrmOptionKind argKind;  /* أي نمط من أنماط الخيارات هو            */
        XPointer value; /* القيمة المقدمة إذا كان XrmoptionNoArg أو
                   XrmoptionSkipNArgs   */
} XrmOptionDescRec, *XrmOptionDescList;
