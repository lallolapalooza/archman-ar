الاسم

hspell - مدقق إملائي للعبرية (واجهة برمجة التطبيقات C)

الملخص

#include <hspell.h>

int hspell_init(struct dict_radix **dictp, int flags);

void hspell_uninit(struct dict_radix *dictp);

int hspell_check_word(struct dict_radix *dict, const char *word, int *preflen);

void hspell_trycorrect(struct dict_radix *dict, const char *word, struct corlist *cl);

int corlist_init(struct corlist *cl);

int corlist_free(struct corlist *cl);

int corlist_n(struct corlist *cl);

char *corlist_str(struct corlist *cl, int i);

unsigned int hspell_is_canonic_gimatria(const char *word);

typedef int hspell_word_split_callback_func(const char *word, const char *baseword, int preflen, int prefspec);

int hspell_enum_splits(struct dict_radix *dict, const char *word, hspell_word_split_callback_func *enumf);

void hspell_set_dictionary_path(const char *path);

const char *hspell_get_dictionary_path(void);

الوصف

يصف دليل الاستخدام هذا واجهة برمجة تطبيقات C لمدقق الهجاء العبري Hspell. يرجى الرجوع إلى hspell(1) للحصول على وصف لمشروع Hspell، ومعيار الهجاء الخاص به، وكيف يعمل.

يجب استدعاء الدالة hspell_init() أولاً لتهيئة مكتبة Hspell. تقوم بإعداد بعض الهياكل العامة (انظر قسم الملاحظات) ثم تقرأ ملفات القاموس الضرورية (التي تكون أماكنها ثابتة عند بناء المكتبة). المعلمة 'dictp' هي مؤشر إلى كائن struct dict_radix*، والذي يتم تعديله ليشير إلى قاموس مخصص حديثًا. لذلك، فإن استدعاء hspell_init() نموذجي يبدو كالتالي:


struct dict_radix *dict;
hspell_init(&dict, flags);

لاحظ أن النوع (struct dict_radix*) هو مؤشر غير شفاف - لا يمتلك مستخدم المكتبة وصولاً إلى الحقول المنفصلة في هذا الهيكل.

يمكن للمعلمة 'flags' أن تحتوي على تشغيل بتات من عدة علامات تعدل سلوك Hspell الافتراضي؛ تشغيل HSPELL_OPT_HE_SHEELA يسمح لـ Hspell بالتعرف على البادئة الاستفهامية He (he ha-she'ela). HSPELL_OPT_DEFAULT هو مرادف لتشغيل أي علامة خاصة، أي أنه يساوي 0.

hspell_init() ترجع 0 عند النجاح، أو أرقامًا سالبة عند حدوث أخطاء. حاليًا، الخطأ الوحيد هو -1، مما يعني أنه تعذر قراءة ملفات القاموس.

تقوم الدالة hspell_uninit() بإلغاء آثار hspell_init()، وتحرير أي ذاكرة تم تخصيصها أثناء التهيئة.

تقوم الدالة hspell_check_word() بالتحقق مما إذا كانت كلمة معينة كلمة عبرية صحيحة (ربما مع جزيئات بادئة ملحقة بطريقة صحيحة نحويًا). يتم إرجاع 1 إذا كانت الكلمة صحيحة، أو 0 إذا كانت غير صحيحة.

يجب أن تكون المعلمة 'word' كلمة عبرية واحدة، بترميز iso8859-8، ربما تحتوي على أحرف الاقتباس الأحادية أو المزدوجة ASCII (التي تمثل geresh و gershayim المستخدمين في العبرية للاختصارات، والاختصارات، وبعض الأصوات الأجنبية). إذا كان البرنامج المستدعي يعمل مع ترميزات أخرى، فيجب عليه تحويل الكلمة إلى iso8859-8 أولاً. على وجه الخصوص، لا يتم حاليًا التعرف على امتدادات cp1255 (ترميز العبرية في MS-Windows) إلى iso8859-8 مثل أحرف niqqud، أو geresh أو gershayim، ويجب إزالتها من الكلمة قبل استدعاء hspell_check_word().

إلى المعلمة 'preflen'، تكتب الدالة مرة أخرى عدد الأحرف التي تم التعرف عليها كجزيء بادئة - الباقي من 'word' هو كلمة قائمة بذاتها. نظرًا لأنه يمكن قراءة الكلمات العبرية عادةً بطرق مختلفة متعددة، فإن هذه الميزة (الحصول على بادئة واحدة من قراءة واحدة ممكنة) ليست مفيدة جدًا عادةً، ومن المرجح إزالتها في إصدار مستقبلي.

توفر الدالة hspell_enum_splits() طريقة للحصول على جميع تقسيمات الكلمة 'word' الممكنة إلى بادئة اختيارية وكلمة قائمة بذاتها. لكل تقسيم ممكن (وقانوني، حيث لا يمكن لبعض الكلمات قبول بادئات معينة)، يتم استدعاء دالة رد اتصال (callback function) معرفة من قبل المستخدم. يتم تزويد دالة رد الاتصال هذه بالكلمة الكاملة، وطول البادئة، والكلمة القائمة بذاتها، وحقل بت يصف أنواع الكلمات التي يمكن أن تقبلها هذه البادئة. لاحظ أنه في بعض الحالات، الكلمة التي تبدأ بالحرف waw يتم مضاعفة هذا الحرف قبل البادئة، لذا في بعض الأحيان strlen(word)!=strlen(baseword)+preflen.

تحاول الدالة hspell_trycorrect() العثور على قائمة تصحيحات ممكنة لكلمة غير صحيحة. نظرًا لأن كثافة الكلمات في العبرية عالية (سلسلة عشوائية من الأحرف، خاصة إذا كانت قصيرة، لديها احتمال كبير بأن تكون كلمة صحيحة)، تحاول هذه الدالة تجربة التصحيحات بناءً على افتراض خطأ في الهجاء (استبدال أحرف تبدو متشابهة، فقدان أو وجود حرف qri'a)، وليس خطأ كتابي (إصبع انزلق على لوحة المفاتيح، إلخ) - انظر أيضًا قسم الملاحظات.

hspell_trycorrect() ترجع قائمة التصحيحات في هيكل من نوع struct corlist. يجب تخصيص هذا الهيكل أولاً باستدعاء corlist_init() ثم تحريره لاحقًا باستدعاء corlist_free(). الماكرو corlist_n() يرجع عدد الكلمات الموجودة في قائمة تصحيحات مخصصة، و corlist_str() يرجع الكلمة i. وفقًا لذلك، إليك مثال استخدام لـ hspell_trycorrect():


struct corlist cl;
printf ("Found misspelled word %s. Possible corrections:\n", w);
corlist_init (&cl);
hspell_trycorrect (dict, w, &cl);
for (i=0; i<corlist_n(&cl); i++) {
printf ("%s\n", corlist_str(&cl, i));
}

تتحقق الدالة hspell_is_canonic_gimatria() مما إذا كانت الكلمة المقدمة هي canonic gimatria - أي، الطريقة الصحيحة لكتابة العدد الذي تمثله في gimatria. قد يرغب المتصل في قبول canonic gimatria كلمات عبرية صحيحة، حتى لو أفادت hspell_check_word() سابقًا أن الكلمة غير موجودة. hspell_is_canonic_gimatria() ترجع الرقم الممثل كـ gimatria في 'word' إذا كانت بالفعل gimatria صحيحة (في شكل canonic)، أو 0 بخلاف ذلك.

تقوم hspell_init() عادةً بقراءة ملفات القاموس من مسار تم تجميعه في المكتبة. هذا منطقي عندما يتم توزيع كود المكتبة والقواميس معًا، ولكن في بعض السيناريوهات، قد يرغب مستخدم المكتبة في استخدام قواميس Hspell الموجودة بالفعل على النظام في مسار عشوائي. يمكن استخدام الدالة hspell_set_dictionary_path() لتعيين هذا المسار، ويجب استخدامها قبل استدعاء hspell_init(). المسار المقدم هو مسار قائمة الكلمات، والملفات الإدخالية الأخرى لها هذا المسار مع إضافة بادئة. يمكن استخدام hspell_get_dictionary_path() للعثور على المسار الحالي. في العديد من التثبيتات، يكون الافتراضي هو "/usr/local/share/hspell/hebrew.wgz".

الربط

على معظم الأنظمة، يتم تجميع مكتبة Hspell لاستخدام مكتبة Zlib لقراءة القواميس المضغوطة. لذلك، يجب على البرنامج الذي يرتبط بمكتبة Hspell أن يرتبط أيضًا بمكتبة Zlib (عادةً، عن طريق إضافة "-lz" إلى سطر التصريف).

يجب على البرامج التي تستخدم autoconf للبحث عن مكتبة Hspell أن تتذكر إخبار AC_CHECK_LIB بالربط أيضًا مع مكتبة -lz عند التحقق من -lhspell.

ملاحظات

بينما كانت واجهة برمجة التطبيقات الموصوفة هنا مستقرة لسنوات، فقد تتغير في المستقبل. يتم تشجيع المستخدمين على مقارنة قيم وحدات الماكرو الصحيحة HSPELL_VERSION_MAJOR و HSPELL_VERSION_MINOR مع تلك التي يتوقعها كاتب البرنامج. وحدة ماكرو ثالثة، HSPELL_VERSION_EXTRA تحتوي على سلسلة يمكن أن تصف تعديلات الإصدار الفرعي (على سبيل المثال، إصدارات تجريبية).

واجهة برمجة تطبيقات C الحالية لـ Hspell منخفضة المستوى للغاية، بمعنى أنها تترك للمستخدم مهمة تنفيذ العديد من الميزات التي يعتاد عليها بعض المستخدمين من مدقق الهجاء. على سبيل المثال، لا توفر أي مرافق لقاموس شخصي يمكن للمستخدم تعريفه. كما أن لديها وظائف منفصلة للتحقق من الكلمات العبرية الصحيحة و gimatria الصحيحة، ولا توجد دالة للقيام بكليهما. يُفترض أن المستدعي - مكتبة تدقيق هجاء أكبر أو معالج نصوص (على سبيل المثال) - لديه بالفعل هذه المرافق. إذا لم يكن الأمر كذلك، فقد ترغب في الاطلاع على مصادر hspell(1) للحصول على مثال تنفيذي.
