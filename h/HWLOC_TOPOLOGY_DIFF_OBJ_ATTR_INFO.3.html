HWLOC_TOPOLOGY_DIFF_OBJ_ATTR_INFO(3)

الحزمة: hwloc

التراخيص: BSD-3-Clause

الاسم

hwloc-topology-diff - التعامل مع اختلافات الطوبولوجيا

المقدمة

#include <hwloc.h>

typedef struct hwloc_topology_diff *hwloc_topology_diff_t;

وصف مفصل

تسمح وظائف hwloc-topology-diff بفحص الطوبولوجيات المختلفة بين مفاهيم مختلفة، وتصديرها إلى XML، وتحميلها من XML. يمكن استخدام هذا لتتبع التغييرات في الطوبولوجيا بمرور الوقت، أو لمقارنة الطوبولوجيات بين الآلات المختلفة.

int hwloc_topology_diff_build(hwloc_topology_t topology, hwloc_topology_t newtopology, unsigned long flags, hwloc_topology_diff_t *diff)

حساب الفرق بين طوبولوجيتين. يتم تخزين الفرق كقائمة من إدخالات hwloc_topology_diff_t بدءًا من diff. يتم حسابه عن طريق إجراء مسح عميق للأشجار الطوبولوجية لكليهما في وقت واحد.

إذا كان الفرق بين كائنين معقدًا جدًا بحيث لا يمكن تمثيله (على سبيل المثال، إذا كانت بعض الكائنات لها أنواع مختلفة، أو أعداد مختلفة من الأبناء)، يتم وضع إدخال فرق خاص من نوع HWLOC_TOPOLOGY_DIFF_TOO_COMPLEX في قائمة الانتظار. لا يستمر حساب الفرق أسفل هذه الكائنات. لذلك، يعني كل إدخال فرق من هذا القبيل أنه لم يكن من الممكن حساب الفرق بين الشجرتين الفرعيتين.

الإرجاع

0 إذا كان يمكن تمثيل الفرق بشكل صحيح.

0 مع diff يشير إلى NULL إذا لم يكن هناك فرق بين الطوبولوجيتين.

1 إذا كان الفرق معقدًا جدًا (انظر أعلاه). ستكون بعض الإدخالات في القائمة من نوع HWLOC_TOPOLOGY_DIFF_TOO_COMPLEX.

-1 عند حدوث أي خطأ آخر.

ملاحظة

flags غير مستخدم حاليًا. يجب أن يكون 0.

يجب تحرير الناتج diff لاحقًا باستخدام hwloc_topology_diff_destroy().

يمكن فقط تصدير الناتج diff إلى XML أو تمريره إلى hwloc_topology_diff_apply() إذا تم إرجاع 0، أي إذا لم يتم إدراج أي إدخال من نوع HWLOC_TOPOLOGY_DIFF_TOO_COMPLEX.

قد يتم تعديل الناتج diff عن طريق إزالة بعض الإدخالات من القائمة. يجب تحرير الإدخالات المحذوفة عن طريق تمريرها إلى hwloc_topology_diff_destroy() (ممكن كقائمة أخرى).

int hwloc_topology_diff_destroy(hwloc_topology_diff_t diff)

تدمير قائمة اختلافات الطوبولوجيا.

الإرجاع

0.

int hwloc_topology_diff_export_xml(hwloc_topology_diff_t diff, const char *refname, const char *xmlpath)

تصدير قائمة اختلافات الطوبولوجيا إلى ملف XML. إذا لم يكن NULL، فإن refname يحدد سلسلة معرف للطوبولوجيا المرجعية التي تم استخدامها كقاعدة عند حساب هذا الاختلاف. يتم تقديم هذا المعرف عادةً باسم ملف XML الآخر الذي يحتوي على الطوبولوجيا المرجعية. يتم تقديم هذه السمة عند قراءة الاختلاف من XML.

الإرجاع

0 عند النجاح، -1 عند حدوث خطأ.

int hwloc_topology_diff_export_xmlbuffer(hwloc_topology_diff_t diff, const char *refname, char **xmlbuffer, int *buflen)

تصدير قائمة اختلافات الطوبولوجيا إلى مخزن مؤقت XML. إذا لم يكن NULL، فإن refname يحدد سلسلة معرف للطوبولوجيا المرجعية التي تم استخدامها كقاعدة عند حساب هذا الاختلاف. يتم تقديم هذا المعرف عادةً باسم ملف XML الآخر الذي يحتوي على الطوبولوجيا المرجعية.

ينتهي المخزن المؤقت المُرجع بـ \0 المدرج في الطول المُرجع.

الإرجاع

0 عند النجاح، -1 عند حدوث خطأ.

ملاحظة

يجب تحرير مخزن XML المؤقت لاحقًا باستخدام hwloc_free_xmlbuffer().

int hwloc_topology_diff_load_xml(const char *xmlpath, hwloc_topology_diff_t *diff, char **refname)

تحميل قائمة اختلافات الطوبولوجيا من ملف XML. إذا لم يكن NULL، فسيتم ملء refname بسلسلة المعرف للطوبولوجيا المرجعية لملف الاختلاف، إذا تم تحديدها في ملف XML. هذا المعرف هو عادةً اسم ملف XML الآخر الذي يحتوي على الطوبولوجيا المرجعية.

الإرجاع

0 عند النجاح، -1 عند حدوث خطأ.

ملاحظة

يجب على المتصل تحرير المؤشر المُرجع في refname لاحقًا.

int hwloc_topology_diff_load_xmlbuffer(const char *xmlbuffer, int buflen, hwloc_topology_diff_t *diff, char **refname)

تحميل قائمة اختلافات الطوبولوجيا من مخزن مؤقت XML. بناء قائمة من الاختلافات من مخزن مؤقت الذاكرة XML المُقدم في xmlbuffer وطوله buflen (بما في ذلك \0 النهائي). قد يكون هذا المخزن المؤقت قد تم ملؤه سابقًا بواسطة hwloc_topology_diff_export_xmlbuffer().

إذا لم يكن NULL، فسيكون refname مملوءًا بسلسلة المعرف للطوبولوجيا المرجعية لملف الاختلاف، إذا تم تحديدها في ملف XML. هذا المعرف هو عادةً اسم ملف XML الآخر الذي يحتوي على الطوبولوجيا المرجعية.

الإرجاع

0 عند النجاح، -1 عند حدوث خطأ.

ملاحظة

يجب على المتصل تحرير المؤشر المُرجع في refname لاحقًا.
