الوصف التفصيلي

تأكد من الاطلاع على الرسم البياني في المصطلحات والتعريفات الذي يوضح شجرة طوبولوجيا كاملة، بما في ذلك الأعماق، علاقات الأبناء/الأشقاء/الأبناء الأشقاء، ومثال على طوبولوجيا غير متماثلة حيث تحتوي حزمة واحدة على ذاكرة تخزين مؤقت أقل من أقرانها.

توثيق نوع التعداد

enum hwloc_get_type_depth_e

المُعدِّد

HWLOC_TYPE_DEPTH_UNKNOWN
لا يوجد كائن من النوع المحدد في الطوبولوجيا.
HWLOC_TYPE_DEPTH_MULTIPLE
توجد كائنات من النوع المحدد في أعماق مختلفة في الطوبولوجيا (فقط للمجموعات).
HWLOC_TYPE_DEPTH_NUMANODE
عمق افتراضي لعقد NUMA.
HWLOC_TYPE_DEPTH_BRIDGE
عمق افتراضي لمستوى كائن الجسر.
HWLOC_TYPE_DEPTH_PCI_DEVICE
عمق افتراضي لمستوى كائن جهاز PCI.
HWLOC_TYPE_DEPTH_OS_DEVICE
عمق افتراضي لمستوى كائن جهاز النظام.
HWLOC_TYPE_DEPTH_MISC
عمق افتراضي للكائن المتنوع.
HWLOC_TYPE_DEPTH_MEMCACHE
عمق افتراضي لكائن ذاكرة التخزين المؤقت.

توثيق الوظائف

hwloc_obj_type_t hwloc_get_depth_type (hwloc_topology_t topology, int depth)

تُرجع نوع الكائنات في العمق depth. يجب أن يكون depth بين 0 و hwloc_topology_get_depth()-1، أو عمق افتراضي مثل HWLOC_TYPE_DEPTH_NUMANODE.

القيمة المرجعة

نوع الكائنات في العمق depth.

(hwloc_obj_type_t)-1 إذا كان العمق depth غير موجود.

int hwloc_get_memory_parents_depth (hwloc_topology_t topology)

تُرجع عمق الآباء الذين ترتبط بهم كائنات الذاكرة. تتمتع كائنات الذاكرة بأعماق افتراضية سالبة لأنها ليست جزءًا من التسلسل الهرمي الرئيسي للكائنات الموجودة على جانب المعالج. لا ينبغي مقارنة هذا العمق بأعماق المستويات الأخرى.

إذا كانت جميع كائنات الذاكرة مرتبطة بالآباء العاديين بنفس العمق، يمكن مقارنة عمق الأب هذا بالآخرين كالمعتاد، على سبيل المثال لمعرفة ما إذا كانت عقد NUMA مرتبطة فوق أو تحت الحزم.

القيمة المرجعة

عمق الآباء العاديين لجميع أبناء الذاكرة إذا كان لجميع هؤلاء الآباء نفس العمق. على سبيل المثال، عمق مستوى الحزمة إذا كانت جميع عقد NUMA مرتبطة بكائنات الحزمة.

HWLOC_TYPE_DEPTH_MULTIPLE إذا لم يكن لدى الآباء العاديين لجميع أبناء الذاكرة نفس العمق. على سبيل المثال، إذا كانت بعض عقد NUMA مرتبطة بالحزم بينما يرتبط البعض الآخر بالمجموعات.

unsigned hwloc_get_nbobjs_by_depth (hwloc_topology_t topology, int depth)

تُرجع عرض المستوى في العمق depth.

القيمة المرجعة

عدد الكائنات في عمق الطوبولوجيا depth.

0 إذا لم تكن هناك كائنات في العمق depth.

int hwloc_get_nbobjs_by_type (hwloc_topology_t topology, hwloc_obj_type_t type) [inline]

تُرجع عرض المستوى type type.

القيمة المرجعة

عدد الكائنات من النوع type.

-1 إذا كانت هناك مستويات متعددة تحتوي على كائنات من هذا النوع، مثل HWLOC_OBJ_GROUP.

0 إذا لم تكن هناك كائنات في العمق depth.

hwloc_obj_t hwloc_get_next_obj_by_depth (hwloc_topology_t topology, int depth, hwloc_obj_t prev) [inline]

تُرجع الكائن التالي في العمق depth.

القيمة المرجعة

الكائن الأول في العمق depth إذا كان prev هو NULL.

الكائن الذي يلي prev في العمق depth إذا كان prev ليس NULL.

NULL إذا لم يكن هناك مثل هذا الكائن.

hwloc_obj_t hwloc_get_next_obj_by_type (hwloc_topology_t topology, hwloc_obj_type_t type, hwloc_obj_t prev) [inline]

تُرجع الكائن التالي من النوع type.

القيمة المرجعة

الكائن الأول من النوع type إذا كان prev هو NULL.

الكائن الذي يلي prev من النوع type إذا كان prev ليس NULL.

NULL إذا لم يكن هناك مثل هذا الكائن.

NULL إذا كانت هناك مستويات متعددة تحتوي على كائنات من هذا النوع (على سبيل المثال HWLOC_OBJ_GROUP)، قد يعود المتصل إلى hwloc_get_obj_by_depth().

hwloc_obj_t

هذا الموقع غير مرتبط بـ Arch Linux ولا تابع له بأي شكل من الأشكال. المحتوى هو ترجمة فقط لصفحات دليل Arch المنشورة على man.archlinux.org.

hwloc_get_obj_by_depth (hwloc_topology_t topology, int depth, unsigned idx)

يُرجع كائن الطوبولوجيا في الفهرس المنطقي idx من العمق depth.

المرجع

الكائن إذا كان موجودًا.

NULL إذا لم يكن هناك كائن بهذا الفهرس والعمق.

hwloc_obj_t hwloc_get_obj_by_type (hwloc_topology_t topology, hwloc_obj_type_t type, unsigned idx) [inline]

يُرجع كائن الطوبولوجيا في الفهرس المنطقي idx بالنوع type.

المرجع

الكائن إذا كان موجودًا.

NULL إذا لم يكن هناك كائن بهذا الفهرس والنوع.

NULL إذا كان هناك مستويات متعددة تحتوي على كائنات من هذا النوع (مثل HWLOC_OBJ_GROUP)، يمكن للمستدعي الرجوع إلى hwloc_get_obj_by_depth().

hwloc_obj_t hwloc_get_root_obj (hwloc_topology_t topology) [inline]

يُرجع الكائن الأعلى في شجرة الطوبولوجيا. نوعه هو HWLOC_OBJ_MACHINE.

لا يمكن لهذه الدالة أن تُرجع NULL.

int hwloc_get_type_depth (hwloc_topology_t topology, hwloc_obj_type_t type)

يُرجع عمق الكائنات من النوع type.

المرجع

عمق الكائنات من النوع type.

عمق افتراضي سالب إذا تم إعطاء نوع عقدة NUMA أو I/O أو نوع عام. يتم تخزين هذه الكائنات في مستويات خاصة ليست متعلقة بوحدة المعالجة المركزية. يمكن تمرير هذا العمق الافتراضي إلى دوال hwloc أخرى مثل hwloc_get_obj_by_depth() ولكن لا ينبغي اعتباره عمقًا فعليًا من قبل التطبيق. على وجه الخصوص، لا ينبغي مقارنته بأي عمق كائن آخر أو بعمق الطوبولوجيا بأكملها.

HWLOC_TYPE_DEPTH_UNKNOWN إذا لم يكن هناك كائن من هذا النوع موجودًا في البنية التحتية، أو إذا كان نظام التشغيل لا يوفر هذا النوع من المعلومات.

HWLOC_TYPE_DEPTH_MULTIPLE إذا تم إعطاء النوع HWLOC_OBJ_GROUP وكان هناك مستويات متعددة من المجموعات.

ملاحظة

إذا كان النوع غائبًا ولكن يمكن قبول نوع مشابه، انظر أيضًا hwloc_get_type_or_below_depth() و hwloc_get_type_or_above_depth().

انظر أيضًا

hwloc_get_memory_parents_depth() لإدارة عمق كائنات الذاكرة.

hwloc_type_sscanf_as_depth() لإرجاع عمق الكائنات التي يتم إعطاء نوعها كسلسلة نصية.

int hwloc_get_type_or_above_depth (hwloc_topology_t topology, hwloc_obj_type_t type) [inline]

يُرجع عمق الكائنات من النوع type أو أعلاه. إذا لم يكن هناك كائن من هذا النوع موجودًا في البنية التحتية، تُرجع الدالة عمق أول كائن "موجود" يحتوي عادةً على type.

هذه الدالة ذات معنى فقط لأنواع الكائنات العادية. إذا تم إعطاء نوع كائن ذاكرة أو I/O أو عام، فسيتم دائمًا إرجاع العمق الافتراضي المقابل (راجع hwloc_get_type_depth()).

قد تُرجع HWLOC_TYPE_DEPTH_MULTIPLE لـ HWLOC_OBJ_GROUP تمامًا مثل hwloc_get_type_depth().

int hwloc_get_type_or_below_depth (hwloc_topology_t topology, hwloc_obj_type_t type) [inline]

يُرجع عمق الكائنات من النوع type أو أدناه. إذا لم يكن هناك كائن من هذا النوع موجودًا في البنية التحتية، تُرجع الدالة عمق أول كائن "موجود" يوجد عادةً داخل type.

هذه الدالة ذات معنى فقط لأنواع الكائنات العادية. إذا تم إعطاء نوع كائن ذاكرة أو I/O أو عام، فسيتم دائمًا إرجاع العمق الافتراضي المقابل (راجع hwloc_get_type_depth()).

قد تُرجع HWLOC_TYPE_DEPTH_MULTIPLE لـ HWLOC_OBJ_GROUP تمامًا مثل hwloc_get_type_depth().

int hwloc_topology_get_depth (hwloc_topology_t restrict topology)

الحصول على عمق الشجرة الهرمية للكائنات. هذا هو عمق كائنات HWLOC_OBJ_PU زائد واحد.

المرجع

عمق شجرة الكائنات.

ملاحظة

يتم تجاهل عقد NUMA وكائنات I/O والكائنات العامة عند حساب عمق الشجرة (يتم وضعها في مستويات خاصة).
