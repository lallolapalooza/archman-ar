int hwloc_bitmap_intersects (hwloc_const_bitmap_t bitmap1, hwloc_const_bitmap_t bitmap2)

اختبر ما إذا كانت خرائط البت bitmap1 و bitmap2 متقاطعة.

الإرجاع

1 إذا كانت خرائط البت متقاطعة، 0 بخلاف ذلك.

ملاحظة

خريطة البت الفارغة لا تتقاطع مع أي خريطة بت أخرى.

int hwloc_bitmap_isequal (hwloc_const_bitmap_t bitmap1, hwloc_const_bitmap_t bitmap2)

اختبر ما إذا كانت خريطة البت bitmap1 مساوية لخريطة البت bitmap2.

الإرجاع

1 إذا كانت خرائط البت متساوية، 0 بخلاف ذلك.

int hwloc_bitmap_isfull (hwloc_const_bitmap_t bitmap)

اختبر ما إذا كانت خريطة البت bitmap ممتلئة بالكامل.

الإرجاع

1 إذا كانت خريطة البت ممتلئة، 0 بخلاف ذلك.

ملاحظة

خريطة البت الممتلئة تكون دائمًا مضبوطة إلى ما لا نهاية.

int hwloc_bitmap_isincluded (hwloc_const_bitmap_t sub_bitmap, hwloc_const_bitmap_t super_bitmap)

اختبر ما إذا كانت خريطة البت sub_bitmap جزءًا من خريطة البت super_bitmap.

الإرجاع

1 إذا كانت sub_bitmap مضمنة في super_bitmap، 0 بخلاف ذلك.

ملاحظة

تعتبر خريطة البت الفارغة مضمنة في أي خريطة بت أخرى.

int hwloc_bitmap_isset (hwloc_const_bitmap_t bitmap, unsigned id)

اختبر ما إذا كان الفهرس id جزءًا من خريطة البت bitmap.

الإرجاع

1 إذا كانت البت عند الفهرس id مضبوطة في خريطة البت bitmap، 0 بخلاف ذلك.

int hwloc_bitmap_iszero (hwloc_const_bitmap_t bitmap)

اختبر ما إذا كانت خريطة البت bitmap فارغة.

الإرجاع

1 إذا كانت خريطة البت فارغة، 0 بخلاف ذلك.

int hwloc_bitmap_last (hwloc_const_bitmap_t bitmap)

احسب آخر فهرس (البت الأكثر أهمية) في خريطة البت bitmap.

الإرجاع

آخر فهرس مضبوط في bitmap.

-1 إذا كانت bitmap فارغة، أو إذا كانت bitmap مضبوطة إلى ما لا نهاية.

int hwloc_bitmap_last_unset (hwloc_const_bitmap_t bitmap)

احسب آخر فهرس غير مضبوط (البت الأكثر أهمية) في خريطة البت bitmap.

الإرجاع

آخر فهرس غير مضبوط في bitmap.

-1 إذا كانت bitmap ممتلئة، أو إذا لم تكن bitmap مضبوطة إلى ما لا نهاية.

int hwloc_bitmap_list_asprintf (char ** strp, hwloc_const_bitmap_t bitmap)

قم بتحويل خريطة البت إلى سلسلة نصية قابلة للتخصيص في قائمة جديدة. **لاحظ أنه إذا كانت خريطة البت عبارة عن مجموعة معالج أو مجموعة عقد، فإنها تحتوي على فهارس فيزيائية.**

القوائم هي فهارس مفصولة بفواصل أو نطاقات. النطاقات هي فهارس مفصولة بشرطة. خريطة البت التي تحتوي على البتات 1، 33، 34، وجميع البتات من 64 إلى 95 تتم طباعتها على النحو التالي: '1,33-34,64-95'. قد لا يحتوي النطاق الأخير على فهرس نهاية إذا كانت خريطة البت مضبوطة إلى ما لا نهاية.

الإرجاع

عدد الأحرف التي تمت كتابتها (باستثناء \0 النهائي).

-1 في حالة حدوث خطأ، على سبيل المثال مع ضبط errno على ENOMEM عند فشل تخصيص السلسلة النصية الناتجة.

int hwloc_bitmap_list_snprintf (char *restrict buf, size_t buflen, hwloc_const_bitmap_t bitmap)

قم بتحويل خريطة البت إلى تنسيق القائمة. **لاحظ أنه إذا كانت خريطة البت عبارة عن مجموعة معالج أو مجموعة عقد، فإنها تحتوي على فهارس فيزيائية.**

القوائم هي فهارس مفصولة بفواصل أو نطاقات. النطاقات هي فهارس مفصولة بشرطة. خريطة البت التي تحتوي على البتات 1، 33، 34، وجميع البتات من 64 إلى 95 تتم طباعتها على النحو التالي: '1,33-34,64-95'. قد لا يحتوي النطاق الأخير على فهرس نهاية إذا كانت خريطة البت مضبوطة إلى ما لا نهاية.

يمكن كتابة ما يصل إلى buflen حرفًا في المخزن المؤقت buf.

إذا كان buflen يساوي 0، يمكن أن يكون buf آمنًا NULL.

الإرجاع

عدد الأحرف التي تمت كتابتها فعليًا إذا لم يتم الاقتطاع، أو التي كان سيتم كتابتها (باستثناء \0 النهائي).

-1 في حالة حدوث خطأ.

int hwloc_bitmap_list_sscanf (hwloc_bitmap_t bitmap, const char *restrict string)

قم بتحليل سلسلة نصية للقائمة وتخزينها في خريطة البت bitmap. **لاحظ أنه إذا كانت خريطة البت عبارة عن مجموعة معالج أو مجموعة عقد، فيجب أن تحتوي السلسلة النصية المدخلة على فهارس فيزيائية.**

القوائم هي فهارس مفصولة بفواصل أو نطاقات. النطاقات هي فهارس مفصولة بشرطة. السلسلة النصية '1,33-34,64-95' يتم تحليلها كخريطة بت تحتوي على البتات 1، 33، 34، وجميع البتات من 64 إلى 95. قد لا يحتوي النطاق الأخير على فهرس نهاية إذا كانت خريطة البت مضبوطة إلى ما لا نهاية.

الإرجاع

0 عند النجاح، -1 عند حدوث خطأ.

int hwloc_bitmap_next (hwloc_const_bitmap_t bitmap, int prev)

احسب الفهرس التالي في خريطة البت bitmap الذي يأتي بعد الفهرس prev.

الإرجاع

أول فهرس مضبوط في bitmap إذا كان prev يساوي -1.

الفهرس التالي المضبوط في bitmap إذا كان prev لا يساوي -1.

-1 إذا لم يكن هناك فهرس بفهرس أعلى مضبوط في bitmap.

int hwloc_bitmap_next_unset (hwloc_const_bitmap_t bitmap, int prev)

احسب الفهرس التالي غير المضبوط في خريطة البت bitmap الذي يأتي بعد الفهرس prev.

الإرجاع

أول فهرس غير مضبوط في bitmap إذا كان prev يساوي -1.

الفهرس التالي غير المضبوط في bitmap إذا كان prev لا يساوي -1.

-1 إذا لم يكن هناك فهرس بفهرس أعلى غير مضبوط في bitmap.

int hwloc_bitmap_not (hwloc_bitmap_t res, hwloc_const_bitmap_t bitmap)

اعكس خريطة البت bitmap وخزن النتيجة في خريطة البت res. يمكن أن تكون res هي نفسها bitmap.

int hwloc_bitmap_nr_ulongs (hwloc_const_bitmap_t bitmap)

أرجع عدد قيم unsigned longs المطلوبة لتخزين خريطة البت bitmap بالكامل. هذا هو عدد قيم unsigned longs المتجاورة من البت الأول لخريطة البت (حتى لو كانت غير مضبوطة) حتى آخر بت مضبوط. هذا مفيد لمعرفة قيمة nr التي يجب تمريرها إلى hwloc_bitmap_to_ulongs() (أو أي استدعاءات لـ hwloc_bitmap_to_ith_ulong() مطلوبة) لتحويل خريطة بت بالكامل إلى عدة قيم unsigned longs.

عند استدعائها على مخرج hwloc_topology_get_topology_cpuset()، يكون الرقم الذي تم إرجاعه كبيرًا بما يكفي لجميع مجموعات المعالجات في التوبولوجيا.

الإرجاع

عدد قيم unsigned longs المطلوبة.

-1 إذا كانت bitmap لا نهائية.

int hwloc_bitmap_only (hwloc_bitmap_t bitmap, unsigned id)

قم بإفراغ خريطة البت bitmap وأضف البت id.

int hwloc_bitmap_or (hwloc_bitmap_t res, hwloc_const_bitmap_t bitmap1, hwloc_const_bitmap_t bitmap2)

قم بعمل عملية OR المنطقية بين خريطة البت bitmap1 و bitmap2 وخزن النتيجة في خريطة البت res. يمكن أن تكون res هي نفسها bitmap1 أو bitmap2.
