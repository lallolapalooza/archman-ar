توثيق الدالة

subroutine chetrd_he2hb (character uplo, integer n, integer kd, complex, dimension( lda, * ) a, integer lda, complex, dimension( ldab, * ) ab, integer ldab, complex, dimension( * ) tau, complex, dimension( * ) work, integer lwork, integer info)

CHETRD_HE2HB

الغرض:

!>
!> CHETRD_HE2HB تقلل المصفوفة الهرميتية المركبة A إلى شكل شريطي هرميتي مركب AB عن طريق تحويل تشابه وحدوي:
!> Q**H * A * Q = AB.
!>

المعاملات

UPLO 
!>          UPLO هو CHARACTER*1
!>          = 'U':  المثلث العلوي من A مخزن؛
!>          = 'L':  المثلث السفلي من A مخزن.
!>

N

!>          N هو INTEGER
!>          رتبة المصفوفة A.  N >= 0.
!>

KD

!>          KD هو INTEGER
!>          عدد الأقطار الفائقة للمصفوفة المختزلة إذا كانت UPLO = 'U'،
!>          أو عدد الأقطار الفرعية إذا كانت UPLO = 'L'.  KD >= 0.
!>          يتم تخزين المصفوفة المختزلة في المصفوفة AB.
!>

A

!>          A هي مصفوفة COMPLEX، حجمها (LDA,N)
!>          عند الدخول، المصفوفة الهرميتية A.  إذا كانت UPLO = 'U'، فإن الجزء العلوي المثلثي
!>          من A بحجم N*N يحتوي على الجزء العلوي المثلثي من المصفوفة A، والجزء السفلي المثلثي الصارم
!>          من A لا تتم الإشارة إليه.  إذا كانت UPLO = 'L'، فإن الجزء السفلي المثلثي
!>          من A بحجم N*N يحتوي على الجزء السفلي المثلثي من المصفوفة A، والجزء العلوي المثلثي الصارم
!>          من A لا تتم الإشارة إليه.
!>          عند الخروج، إذا كانت UPLO = 'U'، فإن القطر والقطر الفائق الأول
!>          من A يتم استبدالها بالعناصر المقابلة للمصفوفة الثلاثية القطر T، والعناصر فوق القطر الفائق الأول،
!>          مع المصفوفة TAU، تمثل المصفوفة الوحدوية Q كحاصل ضرب للمنعكسات الأولية؛ إذا كانت UPLO
!>          = 'L'، فإن القطر والقطر الفرعي الأول من A يتم استبدالها
!>          بالعناصر المقابلة للمصفوفة الثلاثية القطر T، والعناصر تحت القطر الفرعي الأول،
!>          مع المصفوفة TAU، تمثل المصفوفة الوحدوية Q كحاصل ضرب للمنعكسات الأولية. انظر التفاصيل الإضافية.
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرائد للمصفوفة A.  LDA >= max(1,N).
!>

AB

!>          AB هي مصفوفة COMPLEX، حجمها (LDAB,N)
!>          عند الخروج، المثلث العلوي أو السفلي للمصفوفة الشريطية الهرميتية A، مخزنة في أول KD+1 صفوف من المصفوفة.
!>          العمود j من A مخزن في العمود j من المصفوفة AB
!>          على النحو التالي:
!>          إذا كانت UPLO = 'U'، AB(kd+1+i-j,j) = A(i,j) لـ max(1,j-kd)<=i<=j؛
!>          إذا كانت UPLO = 'L'، AB(1+i-j,j) = A(i,j) لـ j<=i<=min(n,j+kd).
!>

LDAB

!>          LDAB هو INTEGER
!>          البعد الرائد للمصفوفة AB.  LDAB >= KD+1.
!>

TAU

!>          TAU هي مصفوفة COMPLEX، حجمها (N-KD)
!>          عوامل القياس للمنعكسات الأولية (انظر التفاصيل الإضافية).
!>

WORK

!>          WORK هي مصفوفة COMPLEX، حجمها (MAX(1,LWORK))
!>          عند الخروج، إذا كان INFO = 0، أو إذا كان LWORK = -1،
!>          يعيد WORK(1) حجم LWORK.
!>

LWORK

!>          LWORK هو INTEGER
!>          بعد المصفوفة WORK الذي يجب حسابه
!>          عن طريق استعلام مساحة العمل.
!>          إذا كان N <= KD+1، فإن LWORK >= 1، وإلا فإن LWORK = MAX(1, LWORK_QUERY).
!>
!>          إذا كان LWORK = -1، فإنه يُفترض استعلام مساحة العمل؛ الروتين
!>          يحسب فقط الحجم الأمثل للمصفوفة WORK، ويعيد
!>          هذه القيمة كالإدخال الأول للمصفوفة WORK، ولا يتم إصدار رسالة خطأ
!>          متعلقة بـ LWORK بواسطة XERBLA.
!>          LWORK_QUERY = N*KD + N*max(KD,FACTOPTNB) + 2*KD*KD
!>          حيث FACTOPTNB هو الحجب المستخدم بواسطة خوارزمية QR أو LQ
!>          عادةً ما يكون FACTOPTNB=128 خيارًا جيدًا، بخلاف ذلك
!>          سيؤدي وضع LWORK=-1 إلى توفير حجم WORK.
!>

INFO

!>          INFO هو INTEGER
!>          = 0:  انتهاء ناجح
!>          < 0:  إذا كان INFO = -i، كان المعامل i غير قانوني
!>

!>
!>  إذا كانت UPLO = 'U'، يتم تمثيل المصفوفة Q كحاصل ضرب للمنعكسات الأولية
!>
!>     Q = H(k)**H . . . H(2)**H H(1)**H، حيث k = n-kd.
!>
!>  كل H(i) له الشكل
!>
!>     H(i) = I - tau * v * v**H
!>
!>  حيث tau هو عامل قياس مركب، و v هو متجه مركب مع
!>  v(1:i+kd-1) = 0 و v(i+kd) = 1؛ conjg(v(i+kd+1:n)) مخزن عند الخروج في
!>  A(i,i+kd+1:n)، و tau في TAU(i).
!>
!>  إذا كانت UPLO = 'L'، يتم تمثيل المصفوفة Q كحاصل ضرب للمنعكسات الأولية
!>
!>     Q = H(1) H(2) . . . H(k)، حيث k = n-kd.
!>
!>  كل H(i) له الشكل
!>
!>     H(i) = I - tau * v * v**H
!>
!>  حيث tau هو عامل قياس مركب، و v هو متجه مركب مع
!>  v(kd+1:i) = 0 و v(i+kd+1) = 1؛ v(i+kd+2:n) مخزن عند الخروج في
!>  A(i+kd+2:n,i)، و tau في TAU(i).
!>
!>  محتويات A عند الخروج موضحة بالأمثلة التالية
!>  مع n = 5:
!>
!>  إذا كانت UPLO = 'U':                       إذا كانت UPLO = 'L':
!>
!>    (  ab  ab/v1  v1      v1     v1    )              (  ab                            )
!>    (      ab     ab/v2   v2     v2    )              (  ab/v1  ab                     )
!>    (             ab      ab/v3  v3    )              (  v1     ab/v2  ab              )
!>    (                     ab     ab/v4 )              (  v1     v2     ab/v3  ab       )
!>    (                            ab    )              (  v1     v2     v3     ab/v4 ab )
!>
!>  حيث d و e تمثل العناصر القطرية وغير القطرية لـ T، و vi
!>  تمثل
