hwloc_obj_t hwloc_get_non_io_ancestor_obj (hwloc_topology_t topology, hwloc_obj_t ioobj) [inline]

احصل على أول كائن سلف غير إدخال/إخراج. بالنظر إلى كائن الإدخال/الإخراج ioobj، ابحث عن أصغر كائن سلف غير إدخال/إخراج. هذا الكائن (عادي أو ذاكرة) يمكن استخدامه للربط لأنه يحتوي على مجموعات وحدات معالجة مركزية وعقد غير فارغة ولأن محليته هي نفسها مثل ioobj.

المرتجعات

كائن غير إدخال/إخراج.

ملاحظة

لا يمكن لهذه الوظيفة أن ترجع NULL.

الكائن الناتج هو عادة كائن عادي ولكنه قد يكون أيضًا كائن ذاكرة (مثل عقدة NUMA) للمنصات المستقبلية إذا تم ربط كائنات الإدخال/الإخراج بالذاكرة بدلاً من وحدات المعالجة المركزية.

hwloc_obj_t hwloc_get_pcidev_by_busid (hwloc_topology_t topology, unsigned domain, unsigned bus, unsigned dev, unsigned func) [inline]

ابحث عن كائن جهاز PCI الذي يطابق معرف ناقل PCI المعطى بالمجال والناقل والجهاز ووظيفة معرف ناقل PCI.

المرتجعات

كائن جهاز PCI مطابق إذا كان موجودًا، NULL بخلاف ذلك.

hwloc_obj_t hwloc_get_pcidev_by_busidstring (hwloc_topology_t topology, const char * busid) [inline]

ابحث عن كائن جهاز PCI الذي يطابق معرف ناقل PCI المعطى كسلسلة xxxx:yy:zz.t أو yy:zz.t.

المرتجعات

كائن جهاز PCI مطابق إذا كان موجودًا، NULL بخلاف ذلك.
