الاسم

htobe16, htole16, be16toh, le16toh, htobe32, htole32, be32toh, le32toh, htobe64, htole64, be64toh, le64toh - تحويل القيم بين ترتيب البايت للمضيف وترتيب البايت الكبير/الصغير

المكتبة

مكتبة C القياسية (libc, -lc)

الخلاصة

#include <endian.h>
uint16_t htobe16(uint16_t host_16bits);
uint16_t htole16(uint16_t host_16bits);
uint16_t be16toh(uint16_t big_endian_16bits);
uint16_t le16toh(uint16_t little_endian_16bits);
uint32_t htobe32(uint32_t host_32bits);
uint32_t htole32(uint32_t host_32bits);
uint32_t be32toh(uint32_t big_endian_32bits);
uint32_t le32toh(uint32_t little_endian_32bits);
uint64_t htobe64(uint64_t host_64bits);
uint64_t htole64(uint64_t host_64bits);
uint64_t be64toh(uint64_t big_endian_64bits);
uint64_t le64toh(uint64_t little_endian_64bits);
متطلبات ماكرو اختبار الميزة لـ glibc (انظر feature_test_macros(7)):

htobe16(), htole16(), be16toh(), le16toh(), htobe32(), htole32(), be32toh(), le32toh(), htobe64(), htole64(), be64toh(), le64toh():

    منذ glibc 2.19:
        _DEFAULT_SOURCE
    في glibc حتى 2.19 بما في ذلك:
        _BSD_SOURCE

الوصف

تقوم هذه الوظائف بتحويل ترميز البايت للقيم الصحيحة من ترتيب البايت الذي تستخدمه وحدة المعالجة المركزية الحالية (المضيف) ، إلى وترتيب البايت الكبير والصغير.

الرقم ، nn ، في اسم كل وظيفة يشير إلى حجم العدد الصحيح الذي تتعامل معه الوظيفة ، إما 16 أو 32 أو 64 بت.

الوظائف التي تحمل أسماء على شكل "htobenn" تحول من ترتيب بايت المضيف إلى ترتيب البايت الكبير.

الوظائف التي تحمل أسماء على شكل "htolenn" تحول من ترتيب بايت المضيف إلى ترتيب البايت الصغير.

الوظائف التي تحمل أسماء على شكل "benntoh" تحول من ترتيب البايت الكبير إلى ترتيب بايت المضيف.

الوظائف التي تحمل أسماء على شكل "lenntoh" تحول من ترتيب البايت الصغير إلى ترتيب بايت المضيف.

الإصدارات

توجد وظائف مشابهة في BSDs ، حيث يكون ملف الرأس المطلوب هو <sys/endian.h> بدلاً من <endian.h>. للأسف ، لم تتبع NetBSD و FreeBSD و glibc اتفاقية تسمية OpenBSD الأصلية لهذه الوظائف ، حيث يظهر المكون nn دائمًا في نهاية اسم الوظيفة (وبالتالي ، على سبيل المثال ، في NetBSD و FreeBSD و glibc ، ما يعادل "betoh32" في OpenBSD هو "be32toh").

المعايير

لا يوجد.

التاريخ

glibc 2.9.

هذه الوظائف مشابهة لوظائف عائلة byteorder(3) القديمة. على سبيل المثال ، be32toh() مطابق لـ ntohl().

ميزة وظائف byteorder(3) هي أنها وظائف قياسية متاحة على جميع أنظمة UNIX. من ناحية أخرى ، حقيقة أنها صممت للاستخدام في سياق TCP / IP تعني أنها تفتقر إلى المتغيرات 64 بت والصغيرة الموصوفة في هذه الصفحة.

أمثلة

يعرض البرنامج أدناه نتائج تحويل عدد صحيح من ترتيب بايت المضيف إلى ترتيب بايت صغير وكبير. نظرًا لأن ترتيب بايت المضيف هو إما صغير أو كبير ، فإن أحد هذه التحويلات فقط سيكون له تأثير. عندما نقوم بتشغيل هذا البرنامج على نظام صغير مثل x86-32 ، نرى ما يلي:


$ ./a.out;
x.u32 = 0x44332211
htole32(x.u32) = 0x44332211
htobe32(x.u32) = 0x11223344

مصدر البرنامج

#include <endian.h>
#include <stdint.h>
#include <stdio.h>
#include <stdlib.h>
int
main(void)
{
    union {
        uint32_t u32;
        uint8_t arr[4];
    } x;
    x.arr[0] = 0x11;	/* Lowest-address byte */
    x.arr[1] = 0x22;
    x.arr[2] = 0x33;
    x.arr[3] = 0x44;	/* Highest-address byte */
    printf("x.u32 = %#x\n", x.u32);
    printf("htole32(x.u32) = %#x\n", htole32(x.u32));
    printf("htobe32(x.u32) = %#x\n", htobe32(x.u32));
    exit(EXIT_SUCCESS);
}
