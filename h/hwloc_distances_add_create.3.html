الاسم

hwloc_distances_add - إضافة مسافات بين الكائنات

الموجز

الأنواع المعرفة


typedef void * hwloc_distances_add_handle_t

التعدادات


enum hwloc_distances_add_flag_e { HWLOC_DISTANCES_ADD_FLAG_GROUP, HWLOC_DISTANCES_ADD_FLAG_GROUP_INACCURATE }

الوظائف


hwloc_distances_add_handle_t hwloc_distances_add_create (hwloc_topology_t topology, const char *name, unsigned long kind, unsigned long flags)
int hwloc_distances_add_values (hwloc_topology_t topology, hwloc_distances_add_handle_t handle, unsigned nbobjs, hwloc_obj_t *objs, hwloc_uint64_t *values, unsigned long flags)
int hwloc_distances_add_commit (hwloc_topology_t topology, hwloc_distances_add_handle_t handle, unsigned long flags)

الوصف المفصل

الطريقة المعتادة لإضافة المسافات هي:

hwloc_distances_add_handle_t handle;
int err = -1;
handle = hwloc_distances_add_create(topology, "name", kind, 0);
if (handle) {
  err = hwloc_distances_add_values(topology, handle, nbobjs, objs, values, 0);
  if (!err)
    err = hwloc_distances_add_commit(topology, handle, flags);
}


إذا كان err يساوي 0 في النهاية، فإن الإضافة كانت ناجحة.

توثيق النوع المعرف

typedef void* hwloc_distances_add_handle_t

مقبض لهيكل مسافات جديد أثناء إضافته إلى الطوبولوجيا.

توثيق نوع التعداد

enum hwloc_distances_add_flag_e

علامات لإضافة مسافات جديدة إلى طوبولوجيا.

المُعدِّد

HWLOC_DISTANCES_ADD_FLAG_GROUP
محاولة تجميع الكائنات بناءً على معلومات المسافة المقدمة حديثًا. يتم إجراء التجميع فقط عندما يحتوي هيكل المسافات على أوقات التأخير، وعندما تكون جميع الكائنات من نفس النوع.
HWLOC_DISTANCES_ADD_FLAG_GROUP_INACCURATE
إذا تم التجميع، اعتبر قيم المسافة غير دقيقة وقم بتخفيف المقارنات أثناء خوارزميات التجميع. قد يتم تعديل الدقة الفعلية من خلال متغير البيئة HWLOC_GROUPING_ACCURACY (انظر متغيرات البيئة).

توثيق الوظائف

int hwloc_distances_add_commit (hwloc_topology_t topology, hwloc_distances_add_handle_t handle, unsigned long flags)

تأكيد هيكل مسافات جديد. تقوم هذه الوظيفة بإنهاء هيكل المسافات وإدراجه في الطوبولوجيا.

تم إرجاع المعامل handle سابقًا بواسطة hwloc_distances_add_create(). ثم تم تحديد الكائنات والقيم باستخدام hwloc_distances_add_values().

يقوم flags بتكوين سلوك الوظيفة باستخدام مجموعة اختيارية OR'ed من hwloc_distances_add_flag_e. يمكن استخدامه لطلب تجميع الكائنات الموجودة بناءً على المسافات.

عند حدوث خطأ، يتم تدمير هيكل المسافات المؤقت ومحتوياته.

الإرجاع

0 عند النجاح.

-1 عند حدوث خطأ.

hwloc_distances_add_handle_t hwloc_distances_add_create (hwloc_topology_t topology, const char * name, unsigned long kind, unsigned long flags)

إنشاء هيكل مسافات فارغ جديد. إنشاء هيكل مسافات فارغ ليتم ملؤه بواسطة hwloc_distances_add_values() ثم تأكيده بواسطة hwloc_distances_add_commit().

المعامل name اختياري، يمكن أن يكون NULL. بخلاف ذلك، سيتم نسخه داخليًا وقد يتم تحريره لاحقًا بواسطة المتصل.

يحدد kind نوع المسافة كمجموعة OR'ed من hwloc_distances_kind_e. يمكن إعطاء نوع معنى واحد فقط ونوع مصدر واحد فقط إذا كان ذلك مناسبًا (مثل HWLOC_DISTANCES_KIND_MEANS_BANDWIDTH و HWLOC_DISTANCES_KIND_FROM_USER). سيتم تعيين النوع HWLOC_DISTANCES_KIND_HETEROGENEOUS_TYPES تلقائيًا وفقًا للكائنات ذات الأنواع المختلفة في hwloc_distances_add_values().

flags يجب أن تكون 0 في الوقت الحالي.

الإرجاع

مقبض hwloc_distances_add_handle_t يجب تمريره بعد ذلك إلى hwloc_distances_add_values() و hwloc_distances_add_commit().

NULL عند حدوث خطأ.

int hwloc_distances_add_values (hwloc_topology_t topology, hwloc_distances_add_handle_t handle, unsigned nbobjs, hwloc_obj_t * objs, hwloc_uint64_t * values, unsigned long flags)

تحديد الكائنات والقيم في هيكل مسافات فارغ جديد. تحديد الكائنات والقيم لهيكل مسافات جديد تم إرجاعه كمقبض بواسطة hwloc_distances_add_create(). يجب بعد ذلك تأكيد الهيكل بواسطة hwloc_distances_add_commit().

عدد الكائنات هو nbobjs ومصفوفة الكائنات هي objs. يتم تخزين قيم المسافات كمصفوفة أحادية البعد في values. المسافة من الكائن i إلى الكائن j موجودة في الخانة i*nbobjs+j.

nbobjs يجب أن يكون على الأقل 2.

سيتم نسخ المصفوفتين objs و values داخليًا، وقد يتم تحريرهما لاحقًا بواسطة المتصل.

عند حدوث خطأ، يتم تدمير هيكل المسافات المؤقت ومحتوياته.

flags يجب أن تكون 0 في الوقت الحالي.

الإرجاع

0 عند النجاح.

-1 عند حدوث خطأ.
