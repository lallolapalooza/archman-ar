الاسم

hwlocality_linux_libnuma_bitmask - التشغيل البيني مع قناع بتات libnuma في لينكس

الملخص

الوظائف


struct bitmask * hwloc_cpuset_to_linux_libnuma_bitmask (hwloc_topology_t topology, hwloc_const_cpuset_t cpuset)
struct bitmask * hwloc_nodeset_to_linux_libnuma_bitmask (hwloc_topology_t topology, hwloc_const_nodeset_t nodeset)
int hwloc_cpuset_from_linux_libnuma_bitmask (hwloc_topology_t topology, hwloc_cpuset_t cpuset, const struct bitmask *bitmask)
int hwloc_nodeset_from_linux_libnuma_bitmask (hwloc_topology_t topology, hwloc_nodeset_t nodeset, const struct bitmask *bitmask)

الوصف التفصيلي

تساعد هذه الواجهة في التحويل بين أقنعة بتات libnuma في لينكس ومجموعات وحدات المعالجة المركزية (cpusets) ومجموعات العقد (nodesets) الخاصة بـ hwloc.

ملاحظة

يجب أن تتطابق طبولوجيا topology مع الجهاز الحالي.

سلوك libnuma غير معرف إذا كان النواة (kernel) لا تدعم NUMA. (عندما لا يتم تعيين CONFIG_NUMA في تكوين النواة). قد لا تكون هذه المساعدة و libnuma متوافقتين تمامًا في هذه الحالة، والتي يمكن اكتشافها عن طريق التحقق مما إذا كانت numa_available() تُرجع -1.

وثائق الوظائف

int hwloc_cpuset_from_linux_libnuma_bitmask (hwloc_topology_t topology, hwloc_cpuset_t cpuset, const struct bitmask * bitmask) [inline]

تحويل قناع بتات libnuma bitmask إلى مجموعة وحدات المعالجة المركزية (cpuset) الخاصة بـ hwloc cpuset. يمكن استخدام هذه الوظيفة بعد استدعاء العديد من وظائف numa_ التي تستخدم struct bitmask كمعامل إخراج.

القيم المعادة

0 عند النجاح.

-1 مع تعيين errno إلى ENOMEM إذا فشلت أي إعادة تخصيص داخلية.

struct bitmask * hwloc_cpuset_to_linux_libnuma_bitmask (hwloc_topology_t topology, hwloc_const_cpuset_t cpuset) [inline]

تحويل مجموعة وحدات المعالجة المركزية (cpuset) الخاصة بـ hwloc cpuset إلى قناع بتات libnuma bitmask المعاد. يجب تحرير قناع البتات المعاد لاحقًا باستخدام numa_bitmask_free.

يمكن استخدام هذه الوظيفة قبل استدعاء العديد من وظائف numa_ التي تستخدم struct bitmask كمعامل إدخال.

القيم المعادة

struct bitmask جديد تم تخصيصه، أو NULL عند حدوث خطأ.

int hwloc_nodeset_from_linux_libnuma_bitmask (hwloc_topology_t topology, hwloc_nodeset_t nodeset, const struct bitmask * bitmask) [inline]

تحويل قناع بتات libnuma bitmask إلى مجموعة عقد NUMA الخاصة بـ hwloc nodeset. يمكن استخدام هذه الوظيفة بعد استدعاء العديد من وظائف numa_ التي تستخدم struct bitmask كمعامل إخراج.

القيم المعادة

0 عند النجاح.

-1 مع تعيين errno إلى ENOMEM إذا فشلت أي إعادة تخصيص داخلية.

struct bitmask * hwloc_nodeset_to_linux_libnuma_bitmask (hwloc_topology_t topology, hwloc_const_nodeset_t nodeset) [inline]

تحويل مجموعة عقد NUMA الخاصة بـ hwloc nodeset إلى قناع بتات libnuma bitmask المعاد. يجب تحرير قناع البتات المعاد لاحقًا باستخدام numa_bitmask_free.

يمكن استخدام هذه الوظيفة قبل استدعاء العديد من وظائف numa_ التي تستخدم struct bitmask كمعامل إدخال.
