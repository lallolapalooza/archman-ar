HWLOC_LOCATION_TYPE_OBJECT(3)

الحزمة: hwloc

التراخيص: BSD-3-Clause

Note

يجب أن يكون البادئ initiator من النوع HWLOC_LOCATION_TYPE_CPUSET عند الإشارة إلى عمليات الوصول التي تقوم بها نوى وحدة المعالجة المركزية. HWLOC_LOCATION_TYPE_OBJECT غير مستخدم حاليًا داخليًا بواسطة hwloc، ولكن يمكن للمستخدمين استخدامه على سبيل المثال لتقديم معلومات مخصصة حول عمليات الوصول إلى الذاكرة التي تقوم بها وحدات معالجة الرسومات.

int hwloc_memattr_get_by_name (hwloc_topology_t topology, const char * name, hwloc_memattr_id_t * id)

إرجاع معرف سمة الذاكرة بالاسم المحدد.

Returns

0 عند النجاح.

-1 مع تعيين errno إلى EINVAL إذا لم تكن هناك مثل هذه السمة موجودة.

int hwloc_memattr_get_initiators (hwloc_topology_t topology, hwloc_memattr_id_t attribute, hwloc_obj_t target_node, unsigned long flags, unsigned * nr, struct hwloc_location * initiators, hwloc_uint64_t * values)

إرجاع البادئات التي لها قيم لسمة معينة لعقدة NUMA مستهدفة محددة. إرجاع البادئات للسمة المعطاة والعقدة المستهدفة في مصفوفة initiators. إذا كانت values ليست NULL، يتم تخزين قيم السمة المقابلة في المصفوفة التي تشير إليها.

عند الإدخال، تشير nr إلى عدد البادئات التي يمكن تخزينها في المصفوفة initiators (و values). عند الإخراج، تشير nr إلى عدد البادئات (والقيم) التي تم العثور عليها فعليًا، حتى لو لم يكن من الممكن تخزين بعضها في المصفوفة. يتم تجاهل البادئات التي لم يكن من الممكن تخزينها، ولكن لا يزال الإرجاع ناجحًا (0). قد يكتشف المتصل ذلك عن طريق مقارنة القيمة المشار إليها بواسطة nr قبل وبعد استدعاء الوظيفة.

لا ينبغي تعديل البادئات المرجعة أو تحريرها، فهي تنتمي إلى الطوبولوجيا.

target_node لا يمكن أن تكون NULL.

flags يجب أن تكون 0 في الوقت الحالي.

إذا لم تكن السمة متعلقة ببادئ معين (ليس لها علامة HWLOC_MEMATTR_FLAG_NEED_INITIATOR)، فلن يتم إرجاع أي بادئ.

Returns

0 عند النجاح أو -1 عند الخطأ.

Note

هذه الوظيفة مخصصة للأدوات والتصحيح (قائمة المعلومات الداخلية) بدلاً من استعلامات التطبيقات. يجب على التطبيقات بدلاً من ذلك تحديد عقد NUMA المفيدة باستخدام hwloc_get_local_numanode_objs() ثم النظر في قيم سماتها لبعض البادئات ذات الصلة.

int hwloc_memattr_get_targets (hwloc_topology_t topology, hwloc_memattr_id_t attribute, struct hwloc_location * initiator, unsigned long flags, unsigned * nr, hwloc_obj_t * targets, hwloc_uint64_t * values)

إرجاع عقد NUMA المستهدفة التي لها بعض القيم لسمة معينة. إرجاع الأهداف للسمة المعطاة في مصفوفة targets (للبادئ المعطى إن وجد). إذا كانت values ليست NULL، يتم تخزين قيم السمة المقابلة في المصفوفة التي تشير إليها.

عند الإدخال، تشير nr إلى عدد الأهداف التي يمكن تخزينها في المصفوفة targets (و values). عند الإخراج، تشير nr إلى عدد الأهداف (والقيم) التي تم العثور عليها فعليًا، حتى لو لم يكن من الممكن تخزين بعضها في المصفوفة. يتم تجاهل الأهداف التي لم يكن من الممكن تخزينها، ولكن لا يزال الإرجاع ناجحًا (0). قد يكتشف المتصل ذلك عن طريق مقارنة القيمة المشار إليها بواسطة nr قبل وبعد استدعاء الوظيفة.

لا ينبغي تعديل الأهداف المرجعة أو تحريرها، فهي تنتمي إلى الطوبولوجيا.

يتم تجاهل المعلمة initiator إذا لم تكن السمة متعلقة ببادئ معين (ليس لها علامة HWLOC_MEMATTR_FLAG_NEED_INITIATOR). بخلاف ذلك، قد تكون initiator غير NULL للإبلاغ فقط عن الأهداف التي لها قيمة لهذا البادئ.

flags يجب أن تكون 0 في الوقت الحالي.

Note

هذه الوظيفة مخصصة للأدوات والتصحيح (قائمة المعلومات الداخلية) بدلاً من استعلامات التطبيقات. يجب على التطبيقات بدلاً من ذلك تحديد عقد NUMA المفيدة باستخدام hwloc_get_local_numanode_objs() ثم النظر في قيم سماتها.

Returns

0 عند النجاح أو -1 عند الخطأ.

Note

يجب أن يكون البادئ initiator من النوع HWLOC_LOCATION_TYPE_CPUSET عند الإشارة إلى عمليات الوصول التي تقوم بها نوى وحدة المعالجة المركزية. HWLOC_LOCATION_TYPE_OBJECT غير مستخدم حاليًا داخليًا بواسطة hwloc، ولكن يمكن للمستخدمين استخدامه على سبيل المثال لتقديم معلومات مخصصة حول عمليات الوصول إلى الذاكرة التي تقوم بها وحدات معالجة الرسومات.

int hwloc_memattr_get_value (hwloc_topology_t topology, hwloc_memattr_id_t attribute, hwloc_obj_t target_node, struct hwloc_location * initiator, unsigned long flags, hwloc_uint64_t * value)

إرجاع قيمة سمة لعقدة NUMA مستهدفة محددة. إذا لم تكن السمة متعلقة ببادئ معين (ليس لها علامة HWLOC_MEMATTR_FLAG_NEED_INITIATOR)، يتم تجاهل الموقع initiator وقد يكون NULL.

target_node لا يمكن أن تكون NULL. إذا كانت attribute هي HWLOC_MEMATTR_ID_CAPACITY، فيجب أن تكون target_node عقدة NUMA. إذا كانت HWLOC_MEMATTR_ID_LOCALITY، فيجب أن يكون لدى target_node مجموعة CPU.

flags يجب أن تكون 0 في الوقت الحالي.

Returns

0 عند النجاح.

-1 عند الخطأ، على سبيل المثال مع تعيين errno إلى EINVAL إذا كانت العلامات غير صالحة أو لم تكن هناك مثل هذه السمة موجودة.

Note

يجب أن يكون البادئ initiator من النوع HWLOC_LOCATION_TYPE_CPUSET عند الإشارة إلى عمليات الوصول التي تقوم بها نوى وحدة المعالجة المركزية. HWLOC_LOCATION_TYPE_OBJECT غير مستخدم حاليًا داخليًا بواسطة hwloc، ولكن يمكن للمستخدمين استخدامه على سبيل المثال لتقديم معلومات مخصصة حول عمليات الوصول إلى الذاكرة التي تقوم بها وحدات معالجة الرسومات.

int hwloc_topology_get_default_nodeset (hwloc_topology_t topology, hwloc_nodeset_t nodeset, unsigned long flags)

إرجاع مجموعة عقد NUMA الافتراضية. في الآلات ذات الذاكرة غير المتجانسة، تعتبر بعض عقد NUMA هي الافتراضية، أي حيث يجب إجراء التخصيصات الأساسية. هذه عادة ما تكون عقد DRAM.

قد تكون العقد الأخرى محفوظة لاستخدام محدد (ذاكرة أجهزة الإدخال / الإخراج، مثل ذاكرة GPU)، ذاكرة صغيرة ولكن عالية الأداء (HBM)، ذاكرة كبيرة ولكن بطيئة (NVM)، وما إلى ذلك. لا ينبغي عادةً تخصيص المخازن المؤقتة من هناك إلا إذا تطلب الأمر صراحةً.

تقوم هذه الوظيفة بملء nodeset ببِتات عقد NUMA التي تعتبر افتراضية.

من المضمون أن يكون لهذه العقد مجموعات CPU غير متقاطعة، أي قد لا تمتلك النوى عقد NUMA محلية متعددة بعد الآن. وبالتالي يمكن استخدام هذا للتكرار عبر المنصة مقسمة إلى مجالات NUMA منفصلة، على سبيل المثال لربط مهمة واحدة لكل نطاق NUMA.

يجب أن تحتفظ أي نواة كانت لديها بعض عقد NUMA المحلية في الطوبولوجيا الأولية، بواحدة في مجموعة العقد الافتراضية. الحالات الطرفية التي قد يكون فيها هذا خاطئًا تتكون من آلات غير متماثلة مع عقد DRAM مفقودة، أو طوبولوجيات تم تقييدها بالفعل إلى عدد أقل من عقد NUMA.

يمكن تمرير مجموعة العقد المرجعة إلى hwloc_topology_restrict() مع HWLOC_RESTRICT_FLAG_BYNODESET لإزالة جميع العقد غير الافتراضية من الطوبولوجيا. ستكون الطوبولوجيا الناتجة أسهل في الاستخدام عند التكرار عبر عقد NUMA (المتجانسة الآن).

تعتمد الاستدلالات للعثور على العقد الافتراضية على طبقات الذاكرة وأنواعها الفرعية (انظر الذاكرة غير المتجانسة) بالإضافة إلى الافتراض بأن موردي الأجهزة يسردون العقد الافتراضية أولاً في جداول الأجهزة.

flags يجب أن تكون 0 في الوقت الحالي.

Returns

0 عند النجاح.

-1 عند الخطأ.

Note

عادةً ما تحتوي مجموعة العقد المرجعة على جميع العقد من طبقة ذاكرة واحدة، غالبًا طبقة DRAM.

يتم تضمين مجموعة العقد المرجعة في قائمة العقد المتاحة التي تم إرجاعها بواسطة hwloc_topology_get_topology_nodeset(). وهي أصغر تمامًا إذا كانت الآلة تحتوي على ذاكرة غير متجانسة.

قد تُرجع الاستدلالات مجموعة فرعية من العقد إذا لم يتمكن hwloc من تخمين أنواع الذاكرة و/أو إذا تمت إزالة بعض العقد الافتراضية من الطوبولوجيا سابقًا (على سبيل المثال باستخدام hwloc_topology_restrict()).
