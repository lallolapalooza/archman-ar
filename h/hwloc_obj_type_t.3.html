enum hwloc_obj_bridge_type_e

نوع جانب واحد (علوي أو سفلي) لجسر I/O.

المُعدِّد

HWLOC_OBJ_BRIDGE_HOST
جانب المضيف للجسر، ممكن فقط علويًا.
HWLOC_OBJ_BRIDGE_PCI
جانب PCI للجسر.

enum hwloc_obj_cache_type_e

نوع الذاكرة المخبأة (الكاش).

المُعدِّد

HWLOC_OBJ_CACHE_UNIFIED
ذاكرة مخبأة موحدة.
HWLOC_OBJ_CACHE_DATA
ذاكرة مخبأة للبيانات.
HWLOC_OBJ_CACHE_INSTRUCTION
ذاكرة مخبأة للتعليمات (يتم تصفيتها افتراضيًا).

enum hwloc_obj_osdev_type_e

نوع جهاز نظام التشغيل.

المُعدِّد

HWLOC_OBJ_OSDEV_BLOCK
جهاز كتلة لنظام التشغيل، أو جهاز ذاكرة غير متطايرة. على سبيل المثال 'sda' أو 'dax2.0' على لينكس.
HWLOC_OBJ_OSDEV_GPU
جهاز GPU لنظام التشغيل. على سبيل المثال ':0.0' لعرض GL، 'card0' لجهاز Linux DRM.
HWLOC_OBJ_OSDEV_NETWORK
جهاز شبكة لنظام التشغيل. على سبيل المثال الواجهة 'eth0' على لينكس.
HWLOC_OBJ_OSDEV_OPENFABRICS
جهاز openfabrics لنظام التشغيل. على سبيل المثال HCA InfiniBand 'mlx4_0'، واجهة Omni-Path 'hfi1_0'، أو HCA Atos/Bull BXI 'bxi0' على لينكس.
HWLOC_OBJ_OSDEV_DMA
جهاز محرك DMA لنظام التشغيل. على سبيل المثال قناة DMA 'dma0chan0' على لينكس.
HWLOC_OBJ_OSDEV_COPROC
جهاز معالج مساعد لنظام التشغيل. على سبيل المثال 'opencl0d0' لجهاز OpenCL، 'cuda0' لجهاز CUDA.

enum hwloc_obj_type_t

نوع كائن الطوبولوجيا.

ملاحظة

لا تعتمد على ترتيب أو اكتمال القيم حيث يمكن تعريف قيم جديدة في المستقبل! إذا كنت بحاجة إلى مقارنة الأنواع، استخدم hwloc_compare_types() بدلاً من ذلك.

المُعدِّد

HWLOC_OBJ_MACHINE
آلة. مجموعة من المعالجات والذاكرة مع تماسك ذاكرة التخزين المؤقت. يتم استخدام هذا النوع دائمًا للكائن الجذري للطوبولوجيا، ولا يستخدم في أي مكان آخر. وبالتالي فإن الوالد له هو دائمًا NULL.
HWLOC_OBJ_PACKAGE
حزمة مادية. الحزمة المادية التي يتم إدخالها عادة في مقبس على اللوحة الأم. تحتوي حزمة المعالج عادة على نوى متعددة، وقد تحتوي على بعض الشرائح.
HWLOC_OBJ_CORE
نواة. وحدة حساب (قد تشاركها عدة وحدات معالجة، المعروفة أيضًا باسم المعالجات المنطقية).
HWLOC_OBJ_PU
وحدة معالجة، أو معالج (منطقي). وحدة تنفيذ (قد تشارك نواة مع وحدات معالجة منطقية أخرى، على سبيل المثال في حالة نواة SMT). هذا هو أصغر كائن يمثل موارد وحدة المعالجة المركزية، ولا يمكن أن يكون له أي أطفال باستثناء كائنات Misc.

يتم الإبلاغ عن الكائنات من هذا النوع دائمًا ويمكن استخدامها كحل بديل عندما لا تكون الكائنات الأخرى متوفرة.

HWLOC_OBJ_L1CACHE
ذاكرة التخزين المؤقت للمستوى 1 للبيانات (أو الموحدة).
HWLOC_OBJ_L2CACHE
ذاكرة التخزين المؤقت للمستوى 2 للبيانات (أو الموحدة).
HWLOC_OBJ_L3CACHE
ذاكرة التخزين المؤقت للمستوى 3 للبيانات (أو الموحدة).
HWLOC_OBJ_L4CACHE
ذاكرة التخزين المؤقت للمستوى 4 للبيانات (أو الموحدة).
HWLOC_OBJ_L5CACHE
ذاكرة التخزين المؤقت للمستوى 5 للبيانات (أو الموحدة).
