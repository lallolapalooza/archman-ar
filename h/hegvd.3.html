subroutine chegvd (integer itype, character jobz, character uplo, integer n, complex, dimension( lda, * ) a, integer lda, complex, dimension( ldb, * ) b, integer ldb, real, dimension( * ) w, complex, dimension( * ) work, integer lwork, real, dimension( * ) rwork, integer lrwork, integer, dimension( * ) iwork, integer liwork, integer info)

CHEGVD

الغرض:

!>
!> CHEGVD يحسب جميع القيم الذاتية، اختياريًا، المتجهات الذاتية
!> لمشكلة هرميتية معينة معمم، من الشكل
!> A*x=(lambda)*B*x,  A*Bx=(lambda)*x,  أو B*A*x=(lambda)*x. هنا A و
!> B مفترض أنهما هرميتيان و B موجب محدد أيضًا.
!> إذا كانت المتجهات الذاتية مطلوبة، فإنه يستخدم خوارزمية divide and conquer.
!>
!>

المعلمات

ITYPE 
!>          ITYPE هو عدد صحيح
!>          يحدد نوع المشكلة المراد حلها:
!>          = 1:  A*x = (lambda)*B*x
!>          = 2:  A*B*x = (lambda)*x
!>          = 3:  B*A*x = (lambda)*x
!>

JOBZ

!>          JOBZ هو حرف واحد
!>          = 'N':  حساب القيم الذاتية فقط؛
!>          = 'V':  حساب القيم الذاتية والمتجهات الذاتية.
!>

UPLO

!>          UPLO هو حرف واحد
!>          = 'U':  المثلثات العلوية لـ A و B مخزنة؛
!>          = 'L':  المثلثات السفلية لـ A و B مخزنة.
!>

N

!>          N هو عدد صحيح
!>          ترتيب المصفوفات A و B.  N >= 0.
!>

A

!>          A هي مصفوفة معقدة، البعد (LDA, N)
!>          عند الإدخال، المصفوفة الهرميتية A.  إذا كان UPLO = 'U'، فإن
!>          الجزء العلوي من المثلث A بأبعاد N × N يحتوي على
!>          الجزء العلوي من المصفوفة A.  إذا كان UPLO = 'L'، فإن
!>          الجزء السفلي من المثلث A بأبعاد N × N يحتوي على
!>          الجزء السفلي من المصفوفة A.
!>
!>          عند الخروج، إذا كان JOBZ = 'V'، وإذا كان INFO = 0، فإن A تحتوي على
!>          المصفوفة Z للمتجهات الذاتية.  المتجهات الذاتية تتم معايرتها
!>          على النحو التالي:
!>          إذا كان ITYPE = 1 أو 2، Z**H*B*Z = I؛
!>          إذا كان ITYPE = 3، Z**H*inv(B)*Z = I.
!>          إذا كان JOBZ = 'N'، فعند الخروج، يتم تدمير الجزء العلوي (إذا كان UPLO='U')
!>          أو الجزء السفلي (إذا كان UPLO='L') من A، بما في ذلك
!>          القطر.
!>

LDA

!>          LDA هو عدد صحيح
!>          البعد الرئيسي للمصفوفة A.  LDA >= max(1,N).
!>

B

!>          B هي مصفوفة معقدة، البعد (LDB, N)
!>          عند الإدخال، المصفوفة الهرميتية B.  إذا كان UPLO = 'U'، فإن
!>          الجزء العلوي من المثلث B بأبعاد N × N يحتوي على
!>          الجزء العلوي من المصفوفة B.  إذا كان UPLO = 'L'، فإن
!>          الجزء السفلي من المثلث B بأبعاد N × N يحتوي على
!>          الجزء السفلي من المصفوفة B.
!>
!>          عند الخروج، إذا كان INFO <= N، فإن الجزء من B الذي يحتوي على المصفوفة
!>          يتم استبداله بالعامل المثلثي U أو L من تحليل Cholesky
!>          B = U**H*U أو B = L*L**H.
!>

LDB

!>          LDB هو عدد صحيح
!>          البعد الرئيسي للمصفوفة B.  LDB >= max(1,N).
!>

W

!>          W هي مصفوفة حقيقية، البعد (N)
!>          إذا كان INFO = 0، فإن القيم الذاتية بترتيب تصاعدي.
!>

WORK

!>          WORK هي مصفوفة معقدة، البعد (MAX(1,LWORK))
!>          عند الخروج، إذا كان INFO = 0، فإن WORK(1) يعيد LWORK الأمثل.
!>

LWORK

!>          LWORK هو عدد صحيح
!>          طول المصفوفة WORK.
!>          إذا كان N <= 1،                LWORK >= 1.
!>          إذا كان JOBZ  = 'N' و N > 1، LWORK >= N + 1.
!>          إذا كان JOBZ  = 'V' و N > 1، LWORK >= 2*N + N**2.
!>
!>          إذا كان LWORK = -1، فإنه يُفترض استعلام عن مساحة العمل؛ تقوم الدالة
!>          بحساب أحجام مساحات العمل المثلى للمصفوفات WORK و RWORK و
!>          IWORK فقط، وتعيد هذه القيم كمدخلات أولى
!>          للمصفوفات WORK و RWORK و IWORK، ولا يتم إصدار أي رسالة خطأ
!>          متعلقة بـ LWORK أو LRWORK أو LIWORK بواسطة XERBLA.
!>

RWORK

!>          RWORK هي مصفوفة حقيقية، البعد (MAX(1,LRWORK))
!>          عند الخروج، إذا كان INFO = 0، فإن RWORK(1) يعيد LRWORK الأمثل.
!>

LRWORK

!>          LRWORK هو عدد صحيح
!>          بعد المصفوفة RWORK.
!>          إذا كان N <= 1،                LRWORK >= 1.
!>          إذا كان JOBZ  = 'N' و N > 1، LRWORK >= N.
!>          إذا كان JOBZ  = 'V' و N > 1، LRWORK >= 1 + 5*N + 2*N**2.
!>
!>          إذا كان LRWORK = -1، فإنه يُفترض استعلام عن مساحة العمل؛ تقوم الدالة
!>          بحساب أحجام مساحات العمل المثلى للمصفوفات WORK و RWORK و
!>          IWORK فقط، وتعيد هذه القيم كمدخلات أولى
!>          للمصفوفات WORK و RWORK و IWORK، ولا يتم إصدار أي رسالة خطأ
!>          متعلقة بـ LWORK أو LRWORK أو LIWORK بواسطة XERBLA.
!>

IWORK

!>          IWORK هي مصفوفة أعداد صحيحة، البعد (MAX(1,LIWORK))
!>          عند الخروج، إذا كان INFO = 0، فإن IWORK(1) يعيد LIWORK الأمثل.
!>

LIWORK

!>          LIWORK هو عدد صحيح
!>          بعد المصفوفة IWORK.
!>          إذا كان N <= 1،                LIWORK >= 1.
!>          إذا كان JOBZ  = 'N' و N > 1، LIWORK >= 1.
!>          إذا كان JOBZ  = 'V' و N > 1، LIWORK >= 3 + 5*N.
!>
!>          إذا كان LIWORK = -1، فإنه يُفترض استعلام عن مساحة العمل؛ تقوم الدالة
!>          بحساب أحجام مساحات العمل المثلى للمصفوفات WORK و RWORK و
!>          IWORK فقط، وتعيد هذه القيم كمدخلات أولى
!>          للمصفوفات WORK و RWORK و IWORK، ولا يتم إصدار أي رسالة خطأ
!>          متعلقة بـ LWORK أو LRWORK أو LIWORK بواسطة XERBLA.
!>

INFO

!>          INFO هو عدد صحيح
!>          = 0:  انتهاء ناجح
!>          < 0:  إذا كان INFO = -i، فإن المعلمة i كانت ذات قيمة غير قانونية
!>          > 0:  CPOTRF أو CHEEVD أعادت رمز خطأ:
!>             <= N:  إذا كان INFO = i و JOBZ = 'N'، فإن الخوارزمية
!>                    فشلت في التقارب؛ لم تتقارب i عنصرًا غير قطري في
!>                    نموذج ثلاثي قطري وسيط إلى
!>                    صفر؛
!>                    إذا كان INFO = i و JOBZ = 'V'، فإن الخوارزمية
!>                    فشلت في حساب قيمة ذاتية أثناء العمل على
!>                    المصفوفة الفرعية الواقعة في الصفوف والأعمدة INFO/(N+1)
!>                    حتى mod(INFO,N+1)؛
!>             > N:   إذا كان INFO = N + i، لـ 1 <= i <= N، فإن
!>                    الصغرى الرئيسية الأمامية من الرتبة i للمصفوفة B ليست موجبة.
!>                    لم يكن من الممكن إكمال تحليل B ولم يتم حساب أي قيم ذاتية أو متجهات ذاتية.
!>
