int hwloc_bitmap_intersects (hwloc_const_bitmap_t bitmap1, hwloc_const_bitmap_t bitmap2)

اختبر ما إذا كانت خرائط البت bitmap1 و bitmap2 متقاطعة.

القيمة المعادة

1 إذا كانت خرائط البت متقاطعة، 0 بخلاف ذلك.

ملاحظة

خريطة البت الفارغة لا تتقاطع مع أي خريطة بت أخرى.

int hwloc_bitmap_isequal (hwloc_const_bitmap_t bitmap1, hwloc_const_bitmap_t bitmap2)

اختبر ما إذا كانت خريطة البت bitmap1 تساوي خريطة البت bitmap2.

القيمة المعادة

1 إذا كانت خرائط البت متساوية، 0 بخلاف ذلك.

int hwloc_bitmap_isfull (hwloc_const_bitmap_t bitmap)

اختبر ما إذا كانت خريطة البت bitmap ممتلئة بالكامل.

القيمة المعادة

1 إذا كانت خريطة البت ممتلئة، 0 بخلاف ذلك.

ملاحظة

خريطة البت الممتلئة تكون دائمًا مضبوطة بشكل لا نهائي.

int hwloc_bitmap_isincluded (hwloc_const_bitmap_t sub_bitmap, hwloc_const_bitmap_t super_bitmap)

اختبر ما إذا كانت خريطة البت sub_bitmap جزءًا من خريطة البت super_bitmap.

القيمة المعادة

1 إذا كانت sub_bitmap مضمنة في super_bitmap، 0 بخلاف ذلك.

ملاحظة

تعتبر خريطة البت الفارغة مضمنة في أي خريطة بت أخرى.

int hwloc_bitmap_isset (hwloc_const_bitmap_t bitmap, unsigned id)

اختبر ما إذا كان الفهرس id جزءًا من خريطة البت bitmap.

القيمة المعادة

1 إذا تم ضبط البت عند الفهرس id في خريطة البت bitmap، 0 بخلاف ذلك.

int hwloc_bitmap_iszero (hwloc_const_bitmap_t bitmap)

اختبر ما إذا كانت خريطة البت bitmap فارغة.

القيمة المعادة

1 إذا كانت خريطة البت فارغة، 0 بخلاف ذلك.

int hwloc_bitmap_last (hwloc_const_bitmap_t bitmap)

احسب آخر فهرس (البت الأكثر أهمية) في خريطة البت bitmap.

القيمة المعادة

آخر فهرس تم ضبطه في bitmap.

-1 إذا كانت bitmap فارغة، أو إذا كانت bitmap مضبوطة بشكل لا نهائي.

int hwloc_bitmap_last_unset (hwloc_const_bitmap_t bitmap)

احسب آخر فهرس غير مضبوط (البت الأكثر أهمية) في خريطة البت bitmap.

القيمة المعادة

آخر فهرس غير مضبوط في bitmap.

-1 إذا كانت bitmap ممتلئة، أو إذا لم تكن bitmap مضبوطة بشكل لا نهائي.

int hwloc_bitmap_list_asprintf (char ** strp, hwloc_const_bitmap_t bitmap)

قم بتحويل خريطة البت إلى سلسلة نصية قائمة منسقة. لاحظ أنه إذا كانت خريطة البت عبارة عن مجموعة معالج أو مجموعة عقد، فإنها تحتوي على فهارس مادية.

القوائم عبارة عن فهارس أو نطاقات مفصولة بفواصل. النطاقات هي فهارس مفصولة بشرطة. خريطة البت التي تحتوي على البتات 1، 33، 34، وكل شيء من 64 إلى 95 تتم طباعتها على هيئة '1,33-34,64-95'. قد لا يكون للنطاق الأخير فهرس نهاية إذا كانت خريطة البت مضبوطة بشكل لا نهائي.

القيمة المعادة

عدد الأحرف التي تم كتابتها (لا تشمل \0 الختامية).

-1 عند حدوث خطأ، على سبيل المثال مع تعيين errno إلى ENOMEM عند فشل تخصيص السلسلة النصية الناتجة.

int hwloc_bitmap_list_snprintf (char *restrict buf, size_t buflen, hwloc_const_bitmap_t bitmap)

قم بتحويل خريطة البت إلى تنسيق قائمة. لاحظ أنه إذا كانت خريطة البت عبارة عن مجموعة معالج أو مجموعة عقد، فإنها تحتوي على فهارس مادية.

القوائم عبارة عن فهارس أو نطاقات مفصولة بفواصل. النطاقات هي فهارس مفصولة بشرطة. خريطة البت التي تحتوي على البتات 1، 33، 34، وكل شيء من 64 إلى 95 تتم طباعتها على هيئة '1,33-34,64-95'. قد لا يكون للنطاق الأخير فهرس نهاية إذا كانت خريطة البت مضبوطة بشكل لا نهائي.

يمكن كتابة ما يصل إلى buflen حرفًا في المخزن المؤقت buf.

إذا كان buflen يساوي 0، يمكن أن يكون buf آمنًا بأن يكون NULL.

القيمة المعادة

عدد الأحرف التي تمت كتابتها بالفعل إذا لم يتم الاقتطاع، أو التي كان سيتم كتابتها (لا تشمل \0 الختامية).

-1 عند حدوث خطأ.

int hwloc_bitmap_list_sscanf (hwloc_bitmap_t bitmap, const char *restrict string)

قم بتحليل سلسلة نصية قائمة وتخزينها في خريطة البت bitmap. لاحظ أنه إذا كانت خريطة البت عبارة عن مجموعة معالج أو مجموعة عقد، فيجب أن تحتوي السلسلة النصية المدخلة على فهارس مادية.

القوائم عبارة عن فهارس أو نطاقات مفصولة بفواصل. النطاقات هي فهارس مفصولة بشرطة. السلسلة النصية '1,33-34,64-95' يتم تحليلها كخريطة بت تحتوي على البتات 1، 33، 34، وكل شيء من 64 إلى 95. قد لا يكون للنطاق الأخير فهرس نهاية إذا كانت خريطة البت مضبوطة بشكل لا نهائي.

القيمة المعادة

0 عند النجاح، -1 عند حدوث خطأ.

int hwloc_bitmap_next (hwloc_const_bitmap_t bitmap, int prev)

احسب الفهرس التالي في خريطة البت bitmap الذي يأتي بعد الفهرس prev.

القيمة المعادة

أول فهرس مضبوط في bitmap إذا كانت prev هي -1.

الفهرس التالي المضبوط في bitmap إذا كانت prev ليست -1.

-1 إذا لم يتم ضبط أي فهرس بفهرس أعلى في bitmap.

int hwloc_bitmap_next_unset (hwloc_const_bitmap_t bitmap, int prev)

احسب الفهرس التالي غير المضبوط في خريطة البت bitmap الذي يأتي بعد الفهرس prev.

القيمة المعادة

أول فهرس غير مضبوط في bitmap إذا كانت prev هي -1.

الفهرس التالي غير المضبوط في bitmap إذا كانت prev ليست -1.

-1 إذا لم يتم ضبط أي فهرس بفهرس أعلى في bitmap.

int hwloc_bitmap_not (hwloc_bitmap_t res, hwloc_const_bitmap_t bitmap)

اعكس خريطة البت bitmap وخزن النتيجة في خريطة البت res. يمكن أن يكون res مطابقًا لـ bitmap

int hwloc_bitmap_nr_ulongs (hwloc_const_bitmap_t bitmap)

أرجع عدد unsigned longs المطلوبة لتخزين خريطة البت bitmap بالكامل. هذا هو عدد unsigned longs المتجاورة من البت الأول لخريطة البت (حتى لو لم يكن مضبوطًا) وصولاً إلى آخر بت مضبوط. هذا مفيد لمعرفة المعلمة nr التي يجب تمريرها إلى hwloc_bitmap_to_ulongs() (أو استدعاءات hwloc_bitmap_to_ith_ulong() المطلوبة) لتحويل خريطة بت بالكامل إلى multiple unsigned longs.

عند استدعائها على ناتج hwloc_topology_get_topology_cpuset()، يكون العدد المرجع كبيرًا بما يكفي لجميع مجموعات المعالجات في الطوبولوجيا.

القيمة المعادة

عدد unsigned longs المطلوبة.

-1 إذا كانت bitmap لا نهائية.

int hwloc_bitmap_only (hwloc_bitmap_t bitmap, unsigned id)

أفرغ خريطة البت bitmap وأضف البت id.
