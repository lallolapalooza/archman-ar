متغير عام يتم تعيين قيمته بواسطة أي روتين لإنشاء عميل RPC لا ينجح. استخدم الروتين clnt_pcreateerror() لطباعة السبب.
void svc_destroy(SVCXPRT *xprt);
ماكرو يدمر مقبض نقل خدمة RPC، xprt. يشمل التدمير عادةً تخصيص هياكل البيانات الخاصة، بما في ذلك xprt نفسها. استخدام xprt غير محدد بعد استدعاء هذا الروتين.
fd_set svc_fdset;
متغير عام يعكس قناع بت واصف الملف للقراءة من جانب خدمة RPC؛ إنه مناسب كمعلمة لاستدعاء النظام select(2). هذا يهم فقط إذا كان منفذ الخدمة يقوم بمعالجة الأحداث غير المتزامنة الخاصة به، بدلاً من استدعاء svc_run(). هذا المتغير للقراءة فقط (لا تمرر عنوانه إلى select(2)!)، ولكنه قد يتغير بعد استدعاءات svc_getreqset() أو أي روتينات إنشاء.
int svc_fds;
مشابه لـ svc_fdset، ولكنه محدود بـ 32 واصف ملف. تم إهمال هذه الواجهة بواسطة svc_fdset.
svc_freeargs(SVCXPRT *xprt, xdrproc_t inproc, char *in);
ماكرو يحرر أي بيانات تم تخصيصها بواسطة نظام RPC/XDR عند فك ترميز الوسائط لإجراء خدمة باستخدام svc_getargs(). يعيد هذا الروتين 1 إذا تم تحرير النتائج بنجاح، وصفر بخلاف ذلك.
svc_getargs(SVCXPRT *xprt, xdrproc_t inproc, char *in);
ماكرو يقوم بفك ترميز وسائط طلب RPC المرتبط بمقبض نقل خدمة RPC، xprt. المعلمة in هي العنوان الذي سيتم وضع الوسائط فيه؛ inproc هو روتين XDR المستخدم لفك ترميز الوسائط. يعيد هذا الروتين واحدًا إذا نجح فك الترميز، وصفر بخلاف ذلك.
struct sockaddr_in *svc_getcaller(SVCXPRT *xprt);
الطريقة المعتمدة للحصول على عنوان الشبكة للمتصل بإجراء مرتبط بمقبض نقل خدمة RPC، xprt.
void svc_getreqset(fd_set *rdfds);
هذا الروتين يهم فقط إذا كان منفذ الخدمة لا يستدعي svc_run()، بل ينفذ معالجة أحداث غير متزامنة مخصصة. يتم استدعاؤه عندما تكون استدعاء النظام select(2) قد حدد وصول طلب RPC على بعض مقابس RPC؛ rdfds هو قناع بت واصف الملف الناتج للقراءة. يعود الروتين عندما تتم خدمة جميع المقابس المرتبطة بالقيمة rdfds.
void svc_getreq(int rdfds);
مشابه لـ svc_getreqset()، ولكنه محدود بـ 32 واصف ملف. تم إهمال هذه الواجهة بواسطة svc_getreqset().
bool_t svc_register(SVCXPRT *xprt, unsigned long prognum,
                    unsigned long versnum,
                    typeof(void (struct svc_req *, SVCXPRT *)) *dispatch,
                    unsigned long protocol);
يربط prognum و versnum بالإجراء الفرعي للخدمة، dispatch. إذا كان protocol صفرًا، فلا يتم تسجيل الخدمة مع خدمة portmap. إذا كان protocol غير صفري، فسيتم إنشاء تعيين للثلاثي [prognum,versnum,protocol] إلى xprt->xp_port مع خدمة portmap المحلية (بشكل عام protocol يكون صفرًا، IPPROTO_UDP أو IPPROTO_TCP). الإجراء dispatch له الشكل التالي:

dispatch(struct svc_req *request, SVCXPRT *xprt);

يعيد الروتين svc_register() واحدًا إذا نجح، وصفر بخلاف ذلك.
void svc_run(void);
هذا الروتين لا يعود أبدًا. ينتظر وصول طلبات RPC، ويستدعي الإجراء الفرعي المناسب للخدمة باستخدام svc_getreq() عند وصول طلب. هذا الروتين عادة ما ينتظر استدعاء النظام select(2) للعودة.
bool_t svc_sendreply(SVCXPRT *xprt, xdrproc_t outproc, char *out);
يتم استدعاؤه بواسطة روتين الإرسال لخدمة RPC لإرسال نتائج استدعاء إجراء عن بعد. المعلمة xprt هي مقبض النقل المرتبط بالطلب؛ outproc هو روتين XDR المستخدم لتشفير النتائج؛ و out هو عنوان النتائج. يعيد هذا الروتين واحدًا إذا نجح، وصفر بخلاف ذلك.
void svc_unregister(unsigned long prognum, unsigned long versnum);
إزالة جميع تعيينات الثنائي [prognum,versnum] إلى روتينات الإرسال، والثلاثي [prognum,versnum,*] إلى رقم المنفذ.
void svcerr_auth(SVCXPRT *xprt, enum auth_stat why);
يتم استدعاؤه بواسطة روتين إرسال خدمة يرفض أداء استدعاء إجراء عن بعد بسبب خطأ في المصادقة.
void svcerr_decode(SVCXPRT *xprt);
يتم استدعاؤه بواسطة روتين إرسال خدمة لا يمكنه فك ترميز وسائطه بنجاح. راجع أيضًا svc_getargs().
void svcerr_noproc(SVCXPRT *xprt);
يتم استدعاؤه بواسطة روتين إرسال خدمة لا ينفذ رقم الإجراء الذي يطلبه المتصل.
void svcerr_noprog(SVCXPRT *xprt);
يتم استدعاؤه عندما لا يتم تسجيل البرنامج المطلوب مع حزمة RPC. عادة لا يحتاج منفذو الخدمة إلى هذا الروتين.
void svcerr_progvers(SVCXPRT *xprt, unsigned long low_vers,
                     unsigned long high_vers);
يتم استدعاؤه عندما لا يتم تسجيل الإصدار المطلوب لبرنامج مع حزمة RPC. عادة لا يحتاج منفذو الخدمة إلى هذا الروتين.
void svcerr_systemerr(SVCXPRT *xprt);
يتم استدعاؤه بواسطة روتين إرسال خدمة عندما يكتشف خطأ في النظام لا يغطيه بروتوكول معين. على سبيل المثال، إذا لم يعد بإمكان الخدمة تخصيص مساحة تخزين، فقد تستدعي هذا الروتين.
void svcerr_weakauth(SVCXPRT *xprt);
يتم استدعاؤه بواسطة روتين إرسال خدمة يرفض أداء استدعاء إجراء عن بعد بسبب معلمات مصادقة غير كافية. يستدعي الروتين svcerr_auth(xprt, AUTH_TOOWEAK).
SVCXPRT *svcfd_create(int fd, unsigned int sendsize,
                      unsigned int recvsize);
إنشاء خدمة فوق أي واصف ملف مفتوح. عادةً، يكون واصف الملف هذا مأخذ توصيل متصل لبروتوكول تدفق مثل TCP. تشير sendsize و recvsize إلى أحجام مخازن الإرسال والاستقبال. إذا كانت صفرًا، يتم اختيار قيمة افتراضية معقولة.
SVCXPRT *svcraw_create(void);
يقوم هذا الروتين بإنشاء خدمة RPC وهمية، ويعيد مؤشرًا إليها. النقل هو في الواقع مخزن مؤقت داخل مساحة عنوان العملية، لذلك يجب أن يكون عميل RPC المقابل موجودًا في نفس مساحة العنوان؛ راجع clntraw_create(). يسمح هذا الروتين بمحاكاة RPC واكتساب تكاليف RPC (مثل أوقات الذهاب والإياب)، دون أي تدخل من النواة. يعيد هذا الروتين NULL إذا فشل.
SVCXPRT *svctcp_create(int sock, unsigned int send_buf_size,
                       unsigned int recv_buf_size);
يقوم هذا الروتين بإنشاء نقل خدمة RPC قائم على TCP/IP، ويعيد مؤشرًا إليه. يرتبط النقل بالمأخذ sock، والذي قد يكون RPC_ANYSOCK، وفي هذه الحالة يتم إنشاء مأخذ جديد. إذا لم يكن المأخذ مرتبطًا بمنفذ TCP محلي، فسيقوم هذا الروتين بربطه بمنفذ عشوائي. عند الانتهاء، يكون xprt->xp_sock هو واصف مأخذ النقل، و xprt->xp_port هو رقم منفذ النقل. يعيد هذا الروتين NULL إذا فشل. نظرًا لأن RPC القائم على TCP يستخدم إدخال/إخراج مخزن مؤقت، يمكن للمستخدمين تحديد حجم المخازن المؤقتة؛ القيم الصفرية تختار قيمًا افتراضية مناسبة.
SVCXPRT *svcudp_bufcreate(int sock, unsigned int sendsize,
                          unsigned int recosize);
يقوم هذا الروتين بإنشاء نقل خدمة RPC قائم على UDP/IP، ويعيد مؤشرًا إليه. يرتبط النقل بالمأخذ sock، والذي قد يكون RPC_ANYSOCK، وفي هذه الحالة يتم إنشاء مأخذ جديد. إذا لم يكن المأخذ مرتبطًا بمنفذ UDP محلي، فسيقوم هذا الروتين بربطه بمنفذ عشوائي. عند الانتهاء، يكون xprt->xp_sock هو واصف مأخذ النقل، و xprt->xp_port هو رقم منفذ النقل. يعيد هذا الروتين NULL إذا فشل.
هذا يسمح للمستخدم بتحديد الحد الأقصى لحجم الحزمة لإرسال واستقبال رسائل RPC المستندة إلى UDP.
SVCXPRT *svcudp_create(int sock);
