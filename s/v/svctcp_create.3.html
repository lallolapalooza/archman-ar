char *clnt_sperrno(enum clnt_stat stat);
تأخذ نفس وسائط clnt_perrno()، ولكن بدلاً من إرسال رسالة إلى الخطأ القياسي تشير إلى سبب فشل استدعاء RPC، تُرجع مؤشرًا إلى سلسلة تحتوي على الرسالة. تنتهي السلسلة بسطر جديد.
يتم استخدام clnt_sperrno() بدلاً من clnt_perrno() إذا لم يكن للبرنامج خطأ قياسي (كما قد لا يكون البرنامج الذي يعمل كخادم)، أو إذا كان المبرمج لا يريد إخراج الرسالة باستخدام printf(3)، أو إذا كان تنسيق الرسالة المراد استخدامه مختلفًا عن ذلك الذي تدعمه clnt_perrno(). ملاحظة: على عكس clnt_sperror() و clnt_spcreateerror()، تُرجع clnt_sperrno() مؤشرًا إلى بيانات ثابتة، ولكن النتيجة لن يتم الكتابة فوقها عند كل استدعاء.

CLIENT *clntraw_create(unsigned long prognum, unsigned long versnum);
تقوم هذه الروتين بإنشاء عميل RPC وهمي للبرنامج البعيد prognum، الإصدار versnum. النقل المستخدم لتمرير الرسائل إلى الخدمة هو في الواقع مخزن مؤقت داخل مساحة عنوان العملية، لذلك يجب أن يعيش خادم RPC المقابل في نفس مساحة العنوان؛ انظر svcraw_create(). هذا يسمح بمحاكاة RPC واكتساب نفقات RPC، مثل أوقات الرحلة ذهابًا وإيابًا، دون أي تدخل من النواة. تُرجع هذه الروتين NULL إذا فشلت.
CLIENT *clnttcp_create(struct sockaddr_in *addr,
                unsigned long prognum, unsigned long versnum,
                int *sockp, unsigned int sendsz, unsigned int recvsz);
تقوم هذه الروتين بإنشاء عميل RPC للبرنامج البعيد prognum، الإصدار versnum؛ يستخدم العميل TCP/IP كنقل. يقع البرنامج البعيد في عنوان الإنترنت *addr. إذا كان addr->sin_port صفرًا، فسيتم تعيينه إلى المنفذ الفعلي الذي يستمع عليه البرنامج البعيد (يتم استشارة خدمة portmap البعيدة للحصول على هذه المعلومات). الوسيط sockp هو مقبس؛ إذا كان RPC_ANYSOCK، فستقوم هذه الروتين بفتح مقبس جديد وتعيين sockp. نظرًا لأن RPC المستند إلى TCP يستخدم إدخال/إخراج مخزن مؤقتًا، يمكن للمستخدم تحديد حجم مخازن الإدخال والإخراج المؤقتة للإرسال والاستقبال باستخدام الوسيطين sendsz و recvsz؛ تختار القيم الصفرية قيمًا افتراضية مناسبة. تُرجع هذه الروتين NULL إذا فشلت.
CLIENT *clntudp_create(struct sockaddr_in *addr,

                struct timeval wait, int *sockp);
تقوم هذه الروتين بإنشاء عميل RPC للبرنامج البعيد prognum، الإصدار versnum؛ يستخدم العميل UDP/IP كنقل. يقع البرنامج البعيد في عنوان الإنترنت addr. إذا كان addr->sin_port صفرًا، فسيتم تعيينه إلى المنفذ الفعلي الذي يستمع عليه البرنامج البعيد (يتم استشارة خدمة portmap البعيدة للحصول على هذه المعلومات). الوسيط sockp هو مقبس؛ إذا كان RPC_ANYSOCK، فستقوم هذه الروتين بفتح مقبس جديد وتعيين sockp. يقوم نقل UDP بإعادة إرسال رسالة الاستدعاء على فترات زمنية مدتها wait حتى يتم استلام استجابة أو حتى ينتهي وقت الاستدعاء. يتم تحديد الوقت الإجمالي لانتهاء الاستدعاء بواسطة clnt_call().

unsigned short pmap_getport(struct sockaddr_in *addr,

واجهة مستخدم لخدمة portmap، والتي تُرجع رقم المنفذ الذي تنتظر عليه خدمة تدعم رقم البرنامج prognum، الإصدار versnum، وتتحدث بروتوكول النقل المرتبط بـ protocol. القيمة protocol على الأرجح هي IPPROTO_UDP أو IPPROTO_TCP. القيمة المرجعة صفر تعني أن التعيين غير موجود أو فشلت حزمة RPC في الاتصال بخدمة portmap البعيدة. في الحالة الأخيرة، تحتوي المتغير العام rpc_createerr على حالة RPC.

int registerrpc(unsigned long prognum, unsigned long versnum,

void svc_destroy(SVCXPRT *xprt);
ماكرو يدمر مقبض نقل خدمة RPC، xprt. يتضمن التدمير عادةً تخصيص هياكل البيانات الخاصة، بما في ذلك xprt نفسها. استخدام xprt غير محدد بعد استدعاء هذا الروتين.

bool_t svc_register(SVCXPRT *xprt, unsigned long prognum,
                    unsigned long versnum,
                    typeof(void (struct svc_req *, SVCXPRT *)) *dispatch,
                    unsigned long protocol);
يربط prognum و versnum بإجراء خدمة الإرسال، dispatch. إذا كان protocol صفرًا، فلن يتم تسجيل الخدمة مع خدمة portmap. إذا كان protocol غير صفري، فسيتم إنشاء تعيين للمثلث [prognum,versnum,protocol] إلى xprt->xp_port مع خدمة portmap المحلية (بشكل عام protocol هو صفر، IPPROTO_UDP أو IPPROTO_TCP). الإجراء dispatch له الشكل التالي:

dispatch(struct svc_req *request, SVCXPRT *xprt);

يعيد الروتين svc_register() واحدًا إذا نجح، وصفر بخلاف ذلك.

هذا الروتين لا يعود أبدًا. ينتظر وصول طلبات RPC، ويستدعي إجراء الخدمة المناسب باستخدام svc_getreq() عند وصول واحد. ينتظر هذا الروتين عادةً استدعاء نظام select(2) للعودة.

SVCXPRT *svctcp_create(int sock, unsigned int send_buf_size,
                       unsigned int recv_buf_size);
يقوم هذا الروتين بإنشاء نقل خدمة RPC يعتمد على TCP/IP، والذي يعيد إليه مؤشرًا. يرتبط النقل بالمقبس sock، والذي يمكن أن يكون RPC_ANYSOCK، وفي هذه الحالة يتم إنشاء مقبس جديد. إذا لم يكن المقبس مرتبطًا بمنفذ TCP محلي، فإن هذا الروتين يربطه بمنفذ عشوائي. عند اكتمال العملية، يكون xprt->xp_sock هو واصف مقبس النقل، و xprt->xp_port هو رقم منفذ النقل. يعيد هذا الروتين NULL إذا فشل. نظرًا لأن RPC المستند إلى TCP يستخدم إدخال/إخراج مخزن مؤقت، يمكن للمستخدمين تحديد حجم المخازن المؤقتة؛ القيم الصفرية تختار افتراضيات مناسبة.
SVCXPRT *svcudp_bufcreate(int sock, unsigned int sendsize,
                          unsigned int recosize);
يقوم هذا الروتين بإنشاء نقل خدمة RPC يعتمد على UDP/IP، والذي يعيد إليه مؤشرًا. يرتبط النقل بالمقبس sock، والذي يمكن أن يكون RPC_ANYSOCK، وفي هذه الحالة يتم إنشاء مقبس جديد. إذا لم يكن المقبس مرتبطًا بمنفذ UDP محلي، فإن هذا الروتين يربطه بمنفذ عشوائي. عند اكتمال العملية، يكون xprt->xp_sock هو واصف مقبس النقل، و xprt->xp_port هو رقم منفذ النقل. يعيد هذا الروتين NULL إذا فشل.
يسمح هذا للمستخدم بتحديد الحد الأقصى لحجم الحزمة لإرسال واستقبال رسائل RPC المستندة إلى UDP.
SVCXPRT *svcudp_create(int sock);
هذا الاستدعاء يعادل svcudp_bufcreate(sock,SZ,SZ) لبعض الحجم الافتراضي SZ.
