bool_t svc_register(SVCXPRT *xprt, unsigned long prognum,
                    unsigned long versnum,
                    typeof(void (struct svc_req *, SVCXPRT *)) *dispatch,

يربط prognum و versnum بإجراء خدمة التوزيع، dispatch. إذا كان protocol صفرًا، فلن يتم تسجيل الخدمة مع خدمة portmap. إذا كان protocol غير صفر، فسيتم إنشاء تعيين للثلاثي [prognum,versnum,protocol] إلى xprt->xp_port مع خدمة portmap المحلية (عادة ما يكون protocol صفرًا، IPPROTO_UDP أو IPPROTO_TCP). الإجراء dispatch له الشكل التالي:

dispatch(struct svc_req *request, SVCXPRT *xprt);

يعيد الروتين svc_register() واحدًا إذا نجح، وصفر إذا لم يكن كذلك.

SVCXPRT *svcraw_create(void);
يقوم هذا الروتين بإنشاء خدمة RPC وهمية، والتي يعيد مؤشرًا إليها. النقل هو في الواقع مخزن مؤقت داخل مساحة عنوان العملية، لذلك يجب أن يكون عميل RPC المقابل في نفس مساحة العنوان؛ انظر clntraw_create(). يسمح هذا الروتين بمحاكاة RPC واكتساب نفقات RPC (مثل أوقات الذهاب والإياب)، دون أي تدخل من النواة. يعيد هذا الروتين NULL إذا فشل.
SVCXPRT *svctcp_create(int sock, unsigned int send_buf_size,
                       unsigned int recv_buf_size);
يقوم هذا الروتين بإنشاء نقل خدمة RPC قائم على TCP/IP، والذي يعيد مؤشرًا إليه. يرتبط النقل بالمقبس sock، والذي قد يكون RPC_ANYSOCK، وفي هذه الحالة يتم إنشاء مقبس جديد. إذا لم يكن المقبس مرتبطًا بمنفذ TCP محلي، فسيقوم هذا الروتين بربطه بمنفذ عشوائي. عند الانتهاء، يكون xprt->xp_sock هو واصف مقبس النقل، و xprt->xp_port هو رقم منفذ النقل. يعيد هذا الروتين NULL إذا فشل. نظرًا لأن RPC القائم على TCP يستخدم إدخال/إخراج مخزن مؤقت، يمكن للمستخدمين تحديد حجم المخازن المؤقتة؛ القيم الصفرية تختار افتراضيات مناسبة.
