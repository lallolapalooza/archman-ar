ماكرو يقوم بنسخ بنية الخطأ من مقبض العميل إلى البنية الموجودة في العنوان errp.
void clnt_pcreateerror(const char *s);
طباعة رسالة إلى الخطأ القياسي توضح سبب عدم إمكانية إنشاء مقبض عميل RPC. يتم إضافة السلسلة s ونقطتين رأسيتين قبل الرسالة. يتم استخدامه عند فشل استدعاء clnt_create() أو clntraw_create() أو clnttcp_create() أو clntudp_create().
void clnt_perrno(enum clnt_stat stat);
طباعة رسالة إلى الخطأ القياسي تتوافق مع الحالة التي تشير إليها stat. يتم استخدامه بعد callrpc().
clnt_perror(CLIENT *clnt, const char *s);
طباعة رسالة إلى الخطأ القياسي توضح سبب فشل استدعاء RPC؛ clnt هو المقبض المستخدم لإجراء الاستدعاء. يتم إضافة السلسلة s ونقطتين رأسيتين قبل الرسالة. يتم استخدامه بعد clnt_call().
char *clnt_spcreateerror(const char *s);
مثل clnt_pcreateerror()، باستثناء أنه يعيد سلسلة بدلاً من الطباعة إلى الخطأ القياسي.
أخطاء: يعيد مؤشرًا إلى بيانات ثابتة يتم استبدالها في كل استدعاء.
char *clnt_sperrno(enum clnt_stat stat);
يأخذ نفس المعلمات مثل clnt_perrno()، ولكنه بدلاً من إرسال رسالة إلى الخطأ القياسي توضح سبب فشل استدعاء RPC، يعيد مؤشرًا إلى سلسلة تحتوي على الرسالة. تنتهي السلسلة بسطر جديد.
يتم استخدام clnt_sperrno() بدلاً من clnt_perrno() إذا لم يكن لدى البرنامج خطأ قياسي (كما هو الحال غالبًا مع برنامج يعمل كخادم)، أو إذا كان المبرمج لا يريد أن يتم إخراج الرسالة باستخدام printf(3)، أو إذا كان سيتم استخدام تنسيق رسالة مختلف عن ذلك الذي تدعمه clnt_perrno(). ملاحظة: على عكس clnt_sperror() و clnt_spcreateerror()، يعيد clnt_sperrno() مؤشرًا إلى بيانات ثابتة، ولكن النتيجة لن يتم استبدالها في كل استدعاء.
char *clnt_sperror(CLIENT *rpch, const char *s);
مثل clnt_perror()، باستثناء أنه (مثل clnt_sperrno()) يعيد سلسلة بدلاً من الطباعة إلى الخطأ القياسي.
أخطاء: يعيد مؤشرًا إلى بيانات ثابتة يتم استبدالها في كل استدعاء.
CLIENT *clntraw_create(unsigned long prognum, unsigned long versnum);
يقوم هذا الإجراء بإنشاء عميل RPC وهمي للبرنامج البعيد prognum، الإصدار versnum. وسيلة النقل المستخدمة لتمرير الرسائل إلى الخدمة هي في الواقع مخزن مؤقت داخل مساحة عنوان العملية، لذلك يجب أن يعيش خادم RPC المقابل في نفس مساحة العنوان؛ انظر svcraw_create(). هذا يسمح بمحاكاة RPC والحصول على نفقات RPC، مثل أوقات الذهاب والإياب، دون أي تدخل من النواة. يعيد هذا الإجراء NULL إذا فشل.
CLIENT *clnttcp_create(struct sockaddr_in *addr,
                unsigned long prognum, unsigned long versnum,
                int *sockp, unsigned int sendsz, unsigned int recvsz);
يقوم هذا الإجراء بإنشاء عميل RPC للبرنامج البعيد prognum، الإصدار versnum؛ يستخدم العميل TCP/IP كوسيلة نقل. يقع البرنامج البعيد في عنوان الإنترنت *addr. إذا كان addr->sin_port صفرًا، فسيتم تعيينه على المنفذ الفعلي الذي يستمع إليه البرنامج البعيد (يتم استشارة خدمة portmap البعيدة للحصول على هذه المعلومات). المعلمة sockp هي مقبس؛ إذا كانت RPC_ANYSOCK، فسيقوم هذا الإجراء بفتح واحد جديد وتعيين sockp. نظرًا لأن RPC المستند إلى TCP يستخدم إدخال/إخراج مخزن مؤقت، يمكن للمستخدم تحديد حجم المخازن المؤقتة للإرسال والاستقبال باستخدام المعلمات sendsz و recvsz؛ القيم الصفرية تختار القيم الافتراضية المناسبة. يعيد هذا الإجراء NULL إذا فشل.
CLIENT *clntudp_create(struct sockaddr_in *addr,
                unsigned long prognum, unsigned long versnum,
                struct timeval wait, int *sockp);
يقوم هذا الإجراء بإنشاء عميل RPC للبرنامج البعيد prognum، الإصدار versnum؛ يستخدم العميل UDP/IP كوسيلة نقل. يقع البرنامج البعيد في عنوان الإنترنت addr. إذا كان addr->sin_port صفرًا، فسيتم تعيينه على المنفذ الفعلي الذي يستمع إليه البرنامج البعيد (يتم استشارة خدمة portmap البعيدة للحصول على هذه المعلومات). المعلمة sockp هي مقبس؛ إذا كانت RPC_ANYSOCK، فسيقوم هذا الإجراء بفتح واحد جديد وتعيين sockp. تعيد وسيلة نقل UDP رسالة الاستدعاء على فترات زمنية قدرها wait حتى يتم استلام استجابة أو حتى ينتهي وقت استدعاء الرسالة. يتم تحديد الوقت الإجمالي لانتهاء استدعاء الرسالة بواسطة clnt_call().
تحذير: نظرًا لأن رسائل RPC المستندة إلى UDP يمكن أن تحتوي على ما يصل إلى 8 كيلوبايت من البيانات المشفرة فقط، لا يمكن استخدام وسيلة النقل هذه للإجراءات التي تأخذ وسائط كبيرة أو تعيد نتائج ضخمة.
CLIENT *clntudp_bufcreate(struct sockaddr_in *addr,
            unsigned long prognum, unsigned long versnum,
            struct timeval wait, int *sockp,
            unsigned int sendsize, unsigned int recosize);
يقوم هذا الإجراء بإنشاء عميل RPC للبرنامج البعيد prognum، على versnum؛ يستخدم العميل UDP/IP كوسيلة نقل. يقع البرنامج البعيد في عنوان الإنترنت addr. إذا كان addr->sin_port صفرًا، فسيتم تعيينه على المنفذ الفعلي الذي يستمع إليه البرنامج البعيد (يتم استشارة خدمة portmap البعيدة للحصول على هذه المعلومات). المعلمة sockp هي مقبس؛ إذا كانت RPC_ANYSOCK، فسيقوم هذا الإجراء بفتح واحد جديد وتعيين sockp. تعيد وسيلة نقل UDP رسالة الاستدعاء على فترات زمنية قدرها wait حتى يتم استلام استجابة أو حتى ينتهي وقت استدعاء الرسالة. يتم تحديد الوقت الإجمالي لانتهاء استدعاء الرسالة بواسطة clnt_call().
هذا يسمح للمستخدم بتحديد الحد الأقصى لحجم الحزمة لإرسال واستقبال رسائل RPC المستندة إلى UDP.
void get_myaddress(struct sockaddr_in *addr);
يضع عنوان IP للجهاز في *addr، دون استشارة روتينيات المكتبة التي تتعامل مع /etc/hosts. يتم دائمًا تعيين رقم المنفذ على htons(PMAPPORT).
struct pmaplist *pmap_getmaps(struct sockaddr_in *addr);
واجهة مستخدم لخدمة portmap، والتي تعيد قائمة بخرائط البرامج RPC الحالية إلى المنافذ على المضيف الموجود في عنوان IP *addr. يمكن لهذا الإجراء أن يعيد NULL. يستخدم الأمر rpcinfo -p هذا الإجراء.
unsigned short pmap_getport(struct sockaddr_in *addr,
                    unsigned long prognum, unsigned long versnum,
                    unsigned int protocol);
واجهة مستخدم لخدمة portmap، والتي تعيد رقم المنفذ الذي تنتظر عليه خدمة تدعم رقم البرنامج prognum، الإصدار versnum، وتتحدث بروتوكول النقل المرتبط بـ protocol. قيمة protocol على الأرجح IPPROTO_UDP أو IPPROTO_TCP. قيمة إرجاع صفر تعني أن التعيين غير موجود أو أن نظام RPC فشل في الاتصال بخدمة portmap البعيدة. في الحالة الأخيرة، تحتوي المتغير العام rpc_createerr على حالة RPC.
enum clnt_stat pmap_rmtcall(struct sockaddr_in *addr,
                    unsigned long prognum, unsigned long versnum,
                    unsigned long procnum,
                    xdrproc_t inproc, char *in,
                    xdrproc_t outproc, char *out,
                    struct timeval tout, unsigned long *portp);
واجهة مستخدم لخدمة portmap، والتي تطلب من portmap على المضيف الموجود في عنوان IP *addr إجراء استدعاء RPC نيابة عنك إلى إجراء على ذلك المضيف. سيتم تعديل المعلمة *portp إلى رقم منفذ البرنامج إذا نجح الإجراء. تمت مناقشة تعريفات المعلمات الأخرى في callrpc() و clnt_call(). يجب استخدام هذا الإجراء لـ "ping" ولا شيء آخر. انظر أيضًا clnt_broadcast().
bool_t pmap_set(unsigned long prognum, unsigned long versnum,
                int protocol, unsigned short port);
واجهة مستخدم لخدمة portmap، والتي تنشئ تعيينًا بين الثلاثي [prognum، versnum، protocol] و port على خدمة portmap للجهاز. قيمة protocol على الأرجح IPPROTO_UDP أو IPPROTO_TCP. يعيد هذا الإجراء واحدًا إذا نجح، وصفرًا بخلاف ذلك. يتم تلقائيًا بواسطة svc_register().
bool_t pmap_unset(unsigned long prognum, unsigned long versnum);
واجهة مستخدم لخدمة portmap، والتي تدمر جميع التعيينات بين الثلاثي [prognum، versnum، *] و ports على خدمة portmap للجهاز. يعيد هذا الإجراء واحدًا إذا نجح، وصفرًا بخلاف ذلك.
int registerrpc(unsigned long prognum, unsigned long versnum,
                unsigned long procnum, typeof(char *(char *)) *procname,
                xdrproc_t inproc, xdrproc_t outproc);
تسجيل الإجراء procname مع حزمة خدمة RPC. إذا وصل طلب للبرنامج prognum، الإصدار versnum، والإجراء procnum، يتم استدعاء procname بمؤشر إلى معلماته؛ يجب أن يعيد procname مؤشرًا إلى نتائجه الثابتة؛ يتم استخدام inproc لفك تشفير المعلمات بينما يتم استخدام outproc لتشفير النتائج. يعيد هذا الإجراء صفرًا إذا نجح التسجيل، و -1 بخلاف ذلك.
