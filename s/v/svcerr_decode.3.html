الاسم

rpc - روتينات مكتبية لاستدعاءات الإجراءات البعيدة

المكتبة

مكتبة C القياسية (libc, -lc)

الملخص والوصف

تسمح هذه الروتينات لبرامج C بإجراء استدعاءات إجراءات على أجهزة أخرى عبر الشبكة. أولاً، يقوم العميل باستدعاء إجراء لإرسال حزمة بيانات إلى الخادم. عند استلام الحزمة، يستدعي الخادم روتين توزيع لأداء الخدمة المطلوبة، ثم يرسل رداً. أخيراً، تعود استدعاء الإجراء إلى العميل.

لاستخدام هذه الروتينات، قم بتضمين ملف الرأس <rpc/rpc.h>.

النماذج الأولية أدناه تستفيد من الأنواع التالية:

typedef int bool_t;
typedef typeof(bool_t (XDR *, void *, ...))  *xdrproc_t;
typedef typeof(bool_t (caddr_t resp, struct sockaddr_in *raddr)
    *resultproc_t;

راجع ملفات الرأس لتعريفات أنواع AUTH و CLIENT و SVCXPRT و XDR.

ماكرو يقوم بتدمير معلومات المصادقة المرتبطة بـ auth. عادة ما يتضمن التدمير إلغاء تخصيص هياكل البيانات الخاصة. استخدام auth غير محدد بعد استدعاء auth_destroy().

إنشاء وإرجاع مقبض مصادقة RPC يقوم بتمرير معلومات مصادقة غير قابلة للاستخدام مع كل استدعاء إجراء بعيد. هذه هي المصادقة الافتراضية التي تستخدمها RPC.
AUTH *authunix_create(char *host, uid_t uid, gid_t gid,

إنشاء وإرجاع مقبض مصادقة RPC يحتوي على معلومات المصادقة. المعلمة host هي اسم الجهاز الذي تم إنشاء المعلومات عليه؛ uid هو معرف المستخدم؛ gid هو معرف المجموعة الحالي للمستخدم؛ n و aup_gids تشير إلى مصفوفة مرقمة بالمجموعات التي ينتمي إليها المستخدم. من السهل انتحال شخصية مستخدم.
AUTH *authunix_create_default(void);
يستدعي authunix_create() مع المعلمات المناسبة.
int callrpc(char *host, unsigned long prognum,
            unsigned long versnum, unsigned long procnum,
            xdrproc_t inproc, const char *in,
            xdrproc_t outproc, char *out);
استدعاء الإجراء البعيد المرتبط بـ prognum و versnum و procnum على الجهاز host. المعلمة in هي عنوان وسيط (وسائط) الإجراء، و out هو العنوان الذي سيتم وضع النتيجة (النتائج) فيه؛ يستخدم inproc لتشفير معلمات الإجراء، ويستخدم outproc لفك تشفير نتائج الإجراء؛ تعيد هذه الروتينية الصفر إذا نجحت، أو قيمة enum clnt_stat محولة إلى عدد صحيح إذا فشلت. الروتينية clnt_perrno() مفيدة لترجمة حالات الفشل إلى رسائل.

مثل callrpc()، باستثناء أن رسالة الاستدعاء يتم بثها إلى جميع شبكات البث المتصلة محليًا. في كل مرة تتلقى فيها رداً، تستدعي هذه الروتينية eachresult()، التي يكون شكلها:

eachresult(char *out, struct sockaddr_in *addr);

حيث out هو نفس out الذي تم تمريره إلى clnt_broadcast()، باستثناء أن مخرجات الإجراء البعيد يتم فك تشفيرها هناك؛ addr يشير إلى عنوان الجهاز الذي أرسل النتائج. إذا أعادت eachresult() صفراً، فإن clnt_broadcast() تنتظر المزيد من الردود؛ وإلا فإنها تعود بالحالة المناسبة.

clnt_destroy(CLIENT *clnt);
ماكرو يقوم بتدمير مقبض RPC الخاص بالعميل. يتضمن التدمير عادةً إلغاء تخصيص هياكل البيانات الخاصة، بما في ذلك clnt نفسه. استخدام clnt غير محدد بعد استدعاء clnt_destroy(). إذا قامت مكتبة RPC بفتح المقبس المرتبط، فسيتم إغلاقه أيضًا. وإلا، يبقى المقبس مفتوحًا.
CLIENT *clnt_create(const char *host, unsigned long prog,
                    unsigned long vers, const char *proto);
روتين إنشاء عميل عام. يحدد host اسم المضيف البعيد حيث يوجد الخادم. يحدد proto نوع بروتوكول النقل المستخدم. القيم المدعومة حاليًا لهذا الحقل هي "udp" و "tcp". يتم تعيين مهلات افتراضية، ولكن يمكن تعديلها باستخدام clnt_control().
تحذير: استخدام UDP له عيوبه. نظرًا لأن رسائل RPC المستندة إلى UDP يمكن أن تحمل 8 كيلوبايت فقط من البيانات المشفرة، لا يمكن استخدام هذا الناقل للإجراءات التي تتلقى وسائط كبيرة أو تعيد نتائج ضخمة.
bool_t clnt_control(CLIENT *cl, int req, char *info);
ماكرو يستخدم لتغيير أو استرداد معلومات مختلفة حول كائن العميل. يشير req إلى نوع العملية، و info هو مؤشر إلى المعلومات. لكل من UDP و TCP، القيم المدعومة لـ req وأنواع وسائطها وما تفعله هي:

CLSET_TIMEOUT  struct timeval // تعيين المهلة الإجمالية
CLGET_TIMEOUT  struct timeval // الحصول على المهلة الإجمالية

CLGET_SERVER_ADDR  struct sockaddr_in
                // الحصول على عنوان الخادم

يطبع رسالة إلى الخطأ القياسي تشير إلى سبب عدم إمكانية إنشاء مقبض عميل RPC. يتم إضافة السلسلة s ونقطتين رأسيتين قبل الرسالة. تستخدم عند فشل استدعاء clnt_create() أو clntraw_create() أو clnttcp_create() أو clntudp_create().

يطبع رسالة إلى الخطأ القياسي تتوافق مع الحالة التي تشير إليها stat. تستخدم بعد callrpc().

يطبع رسالة إلى الخطأ القياسي تشير إلى سبب فشل استدعاء RPC؛ clnt هو المقبض المستخدم لإجراء الاستدعاء. يتم إضافة السلسلة s ونقطتين رأسيتين قبل الرسالة. تستخدم بعد clnt_call().
char *clnt_spcreateerror(const char *s);

أخطاء: تُرجع مؤشرًا إلى بيانات ثابتة يتم الكتابة فوقها في كل استدعاء.
char *clnt_sperrno(enum clnt_stat stat);
تأخذ نفس الوسائط مثل clnt_perrno()، ولكن بدلاً من إرسال رسالة إلى الخطأ القياسي تشير إلى سبب فشل استدعاء RPC، تُرجع مؤشرًا إلى سلسلة تحتوي على الرسالة. تنتهي السلسلة بـ NEWLINE.
يتم استخدام clnt_sperrno() بدلاً من clnt_perrno() إذا لم يكن لدى البرنامج خطأ قياسي (كما هو الحال غالبًا مع برنامج يعمل كخادم)، أو إذا كان المبرمج لا يريد إخراج الرسالة باستخدام printf(3)، أو إذا كان سيتم استخدام تنسيق رسالة مختلف عن ذلك الذي تدعمه clnt_perrno(). ملاحظة: على عكس clnt_sperror() و clnt_spcreateerror()، تُرجع clnt_sperrno() مؤشرًا إلى بيانات ثابتة، ولكن لن يتم الكتابة فوق النتيجة في كل استدعاء.

أخطاء: تُرجع مؤشرًا إلى بيانات ثابتة يتم الكتابة فوقها في كل استدعاء.
CLIENT *clntraw_create(unsigned long prognum, unsigned long versnum);
يقوم هذا الروتين بإنشاء عميل RPC وهمي للبرنامج البعيد prognum، الإصدار versnum. النقل المستخدم لتمرير الرسائل إلى الخدمة هو في الواقع مخزن مؤقت داخل مساحة عنوان العملية، لذلك يجب أن يكون خادم RPC المقابل موجودًا في نفس مساحة العنوان؛ انظر svcraw_create(). يسمح هذا بمحاكاة RPC والحصول على نفقات RPC، مثل أوقات الذهاب والإياب، دون أي تدخل من النواة. يُرجع هذا الروتين NULL إذا فشل.
CLIENT *clnttcp_create(struct sockaddr_in *addr,
                unsigned long prognum, unsigned long versnum,
                int *sockp, unsigned int sendsz, unsigned int recvsz);
يقوم هذا الروتين بإنشاء عميل RPC للبرنامج البعيد prognum، الإصدار versnum؛ يستخدم العميل TCP/IP كوسيلة نقل. يقع البرنامج البعيد في عنوان الإنترنت *addr. إذا كان addr->sin_port صفرًا، فسيتم تعيينه على المنفذ الفعلي الذي يستمع إليه البرنامج البعيد (يتم استشارة خدمة portmap البعيدة لهذه المعلومات). الوسيط sockp هو مقبس؛ إذا كان RPC_ANYSOCK، فسيفتح هذا الروتين مقبسًا جديدًا ويعين sockp. نظرًا لأن RPC المستند إلى TCP يستخدم الإدخال/الإخراج المخزن مؤقتًا، يمكن للمستخدم تحديد حجم المخازن المؤقتة للإرسال والاستقبال باستخدام الوسائط sendsz و recvsz؛ القيم الصفرية تختار القيم الافتراضية المناسبة. يُرجع هذا الروتين NULL إذا فشل.
