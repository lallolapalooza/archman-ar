rpc(3) دليل وظائف المكتبة rpc(3)

الاسم

rpc - روتينات المكتبة لاستدعاءات الإجراءات البعيدة

المكتبة

مكتبة C القياسية (libc، -lc)

الملخص والوصف

تسمح هذه الروتينات لبرامج C بإجراء استدعاءات إجراءات على أجهزة أخرى عبر الشبكة. أولاً، يستدعي العميل إجراءً لإرسال حزمة بيانات إلى الخادم. عند استلام الحزمة، يستدعي الخادم إجراء توزيع لأداء الخدمة المطلوبة، ثم يرسل ردًا. أخيرًا، يعود استدعاء الإجراء إلى العميل.

للاستفادة من هذه الروتينات، قم بتضمين ملف الرأس <rpc/rpc.h>.

النماذج الأولية أدناه تستخدم الأنواع التالية:

typedef int bool_t;
typedef typeof(bool_t (XDR *, void *, ...))  *xdrproc_t;
typedef typeof(bool_t (caddr_t resp, struct sockaddr_in *raddr)
    *resultproc_t;

راجع ملفات الرأس للإعلانات عن أنواع AUTH و CLIENT و SVCXPRT و XDR.

void auth_destroy(AUTH *auth);
ماكرو يدمر معلومات المصادقة المرتبطة بـ auth. عادةً ما يتضمن التدمير إلغاء تخصيص هياكل البيانات الخاصة. استخدام auth غير محدد بعد استدعاء auth_destroy().
AUTH *authnone_create(void);
إنشاء وإرجاع مقبض مصادقة RPC يمرر معلومات مصادقة غير قابلة للاستخدام مع كل استدعاء إجراء بعيد. هذه هي المصادقة الافتراضية التي يستخدمها RPC.
AUTH *authunix_create(char *host, uid_t uid, gid_t gid,
                      int n, gid_t aup_gids[n]);
إنشاء وإرجاع مقبض مصادقة RPC يحتوي على معلومات المصادقة. المعلمة host هي اسم الجهاز الذي تم إنشاء المعلومات عليه؛ uid هو معرف المستخدم؛ gid هو معرف المجموعة الحالي للمستخدم؛ n و aup_gids تشير إلى مصفوفة محسوبة للمجموعات التي ينتمي إليها المستخدم. من السهل انتحال هوية المستخدم.
AUTH *authunix_create_default(void);
يستدعي authunix_create() بالمعلمات المناسبة.
int callrpc(char *host, unsigned long prognum,
            unsigned long versnum, unsigned long procnum,
            xdrproc_t inproc, const char *in,
            xdrproc_t outproc, char *out);
استدعاء الإجراء البعيد المرتبط بـ prognum و versnum و procnum على الجهاز host. المعلمة in هي عنوان وسيط (وسائط) الإجراء، و out هو عنوان مكان وضع النتائج؛ inproc يستخدم لتشفير معلمات الإجراء، و outproc يستخدم لفك تشفير نتائج الإجراء؛ تعيد هذه الروتينات صفرًا إذا نجحت، أو قيمة enum clnt_stat محوّلة إلى عدد صحيح إذا فشلت. الروتين clnt_perrno() مفيد لترجمة حالات الفشل إلى رسائل.
تحذير: استدعاء الإجراءات البعيدة باستخدام هذا الروتين يستخدم UDP/IP كنقل؛ راجع clntudp_create() للقيود. ليس لديك تحكم في المهلات أو المصادقة باستخدام هذا الروتين.
enum clnt_stat clnt_broadcast(unsigned long prognum,
                     unsigned long versnum, unsigned long procnum,
                     xdrproc_t inproc, char *in,
                     xdrproc_t outproc, char *out,
                     resultproc_t eachresult);
مثل callrpc()، باستثناء أن رسالة الاستدعاء يتم بثها إلى جميع شبكات البث المتصلة محليًا. في كل مرة تتلقى فيها ردًا، يستدعي هذا الروتين eachresult()، والذي يكون شكله:

eachresult(char *out, struct sockaddr_in *addr);

حيث out هو نفس out الذي تم تمريره إلى clnt_broadcast()، باستثناء أن مخرجات الإجراء البعيد يتم فك تشفيرها هناك؛ addr يشير إلى عنوان الجهاز الذي أرسل النتائج. إذا أعادت eachresult() صفرًا، ينتظر clnt_broadcast() المزيد من الردود؛ وإلا فإنه يعود بحالة مناسبة.
تحذير: مقابس البث محدودة الحجم بأقصى وحدة نقل للوصلة البيانات. بالنسبة للإيثرنت، هذه القيمة هي 1500 بايت.
enum clnt_stat clnt_call(CLIENT *clnt, unsigned long procnum,
                    xdrproc_t inproc, char *in,
                    xdrproc_t outproc, char *out,
                    struct timeval tout);
ماكرو يستدعي الإجراء البعيد procnum المرتبط بمقبض العميل clnt، والذي يتم الحصول عليه باستخدام روتين إنشاء عميل RPC مثل clnt_create(). المعلمة in هي عنوان وسيط (وسائط) الإجراء، و out هو عنوان مكان وضع النتائج؛ inproc يستخدم لتشفير معلمات الإجراء، و outproc يستخدم لفك تشفير نتائج الإجراء؛ tout هو الوقت المسموح به لعودة النتائج.
clnt_destroy(CLIENT *clnt);
ماكرو يدمر مقبض RPC للعميل. عادةً ما يتضمن التدمير إلغاء تخصيص هياكل البيانات الخاصة، بما في ذلك clnt نفسه. استخدام clnt غير محدد بعد استدعاء clnt_destroy(). إذا فتحت مكتبة RPC المقبس المرتبط، فسيتم إغلاقه أيضًا. بخلاف ذلك، يظل المقبس مفتوحًا.
CLIENT *clnt_create(const char *host, unsigned long prog,
                    unsigned long vers, const char *proto);
روتين إنشاء عميل عام. host يحدد اسم المضيف البعيد حيث يوجد الخادم. proto يشير إلى نوع بروتوكول النقل الذي سيتم استخدامه. القيم المدعومة حاليًا لهذا الحقل هي "udp" و "tcp". يتم تعيين المهلات الافتراضية، ولكن يمكن تعديلها باستخدام clnt_control().
تحذير: استخدام UDP له عيوبه. نظرًا لأن رسائل RPC المستندة إلى UDP يمكن أن تحتوي على 8 كيلوبايت فقط من البيانات المشفرة، فلا يمكن استخدام هذا النقل للإجراءات التي تأخذ وسائط كبيرة أو تعيد نتائج ضخمة.
bool_t clnt_control(CLIENT *cl, int req, char *info);
ماكرو يستخدم لتغيير أو استرداد معلومات مختلفة حول كائن العميل. req يشير إلى نوع العملية، و info هو مؤشر إلى المعلومات. لكل من UDP و TCP، القيم المدعومة لـ req وأنواع وسائطها وماذا تفعل هي:

CLSET_TIMEOUT  struct timeval // تعيين المهلة الإجمالية
CLGET_TIMEOUT  struct timeval // الحصول على المهلة الإجمالية

ملاحظة: إذا قمت بتعيين المهلة باستخدام clnt_control()، فسيتم تجاهل المعلمة tout التي تم تمريرها إلى clnt_call() في جميع الاستدعاءات المستقبلية.

CLGET_SERVER_ADDR  struct sockaddr_in
                // الحصول على عنوان الخادم

العمليات التالية صالحة لـ UDP فقط:

CLSET_RETRY_TIMEOUT  struct timeval // تعيين مهلة إعادة المحاولة
CLGET_RETRY_TIMEOUT  struct timeval // الحصول على مهلة إعادة المحاولة

مهلة إعادة المحاولة هي الوقت الذي تنتظره "UDP RPC" رد الخادم قبل إعادة إرسال الطلب.
clnt_freeres(CLIENT * clnt, xdrproc_t outproc, char *out);
ماكرو يحرر أي بيانات تم تخصيصها بواسطة نظام RPC/XDR عند فك تشفير نتائج استدعاء RPC. المعلمة out هي عنوان النتائج، و outproc هو روتين XDR الذي يصف النتائج. تعيد هذه الروتينات واحدًا إذا تم تحرير النتائج بنجاح، وصفرًا بخلاف ذلك.
void clnt_geterr(CLIENT *clnt, struct rpc_err

*errp);
ماكرو يقوم بنسخ بنية الخطأ من معالج العميل إلى البنية الموجودة في العنوان errp.
void clnt_pcreateerror(const char *s);
يطبع رسالة إلى الخطأ القياسي تشير إلى سبب عدم إمكانية إنشاء معالج RPC للعميل. يتم إضافة السلسلة النصية s ونقطتين رأسيتين قبل الرسالة. يستخدم عند فشل استدعاء clnt_create() أو clntraw_create() أو clnttcp_create() أو clntudp_create().
void clnt_perrno(enum clnt_stat stat);
يطبع رسالة إلى الخطأ القياسي تتوافق مع الحالة التي يشير إليها stat. يستخدم بعد callrpc().
clnt_perror(CLIENT *clnt, const char *s);
يطبع رسالة إلى الخطأ القياسي تشير إلى سبب فشل استدعاء RPC؛ clnt هو المعالج المستخدم لإجراء الاستدعاء. يتم إضافة السلسلة النصية s ونقطتين رأسيتين قبل الرسالة. يستخدم بعد clnt_call().
char *clnt_spcreateerror(const char *s);
مثل clnt_pcreateerror()، إلا أنه يعيد سلسلة نصية بدلاً من طباعتها على الخطأ القياسي.
عيوب: يعيد مؤشرًا إلى بيانات ثابتة يتم الكتابة فوقها في كل استدعاء.
char *clnt_sperrno(enum clnt_stat stat);
يأخذ نفس وسائط clnt_perrno()، ولكن بدلاً من إرسال رسالة إلى الخطأ القياسي تشير إلى سبب فشل استدعاء RPC، يعيد مؤشرًا إلى سلسلة نصية تحتوي على الرسالة. تنتهي السلسلة بحرف سطر جديد.
يستخدم clnt_sperrno() بدلاً من clnt_perrno() إذا كان البرنامج لا يحتوي على خطأ قياسي (كما هو الحال في برنامج يعمل كخادم غالبًا)، أو إذا كان المبرمج لا يريد إخراج الرسالة باستخدام printf(3)، أو إذا كان سيتم استخدام تنسيق رسالة مختلف عن ذلك الذي تدعمه clnt_perrno(). ملاحظة: على عكس clnt_sperror() و clnt_spcreateerror()، يعيد clnt_sperrno() مؤشرًا إلى بيانات ثابتة، ولكن النتيجة لن يتم الكتابة فوقها في كل استدعاء.
char *clnt_sperror(CLIENT *rpch, const char *s);
مثل clnt_perror()، باستثناء أنه (مثل clnt_sperrno()) يعيد سلسلة نصية بدلاً من الطباعة على الخطأ القياسي.
عيوب: يعيد مؤشرًا إلى بيانات ثابتة يتم الكتابة فوقها في كل استدعاء.
CLIENT *clntraw_create(unsigned long prognum, unsigned long versnum);
يقوم هذا الروتين بإنشاء عميل RPC وهمي للبرنامج البعيد prognum، الإصدار versnum. الوسيلة المستخدمة لتمرير الرسائل إلى الخدمة هي في الواقع مخزن مؤقت داخل مساحة عنوان العملية، لذلك يجب أن يعيش خادم RPC المقابل في نفس مساحة العنوان؛ انظر svcraw_create(). هذا يسمح بمحاكاة RPC والحصول على تكاليف RPC، مثل أوقات الذهاب والإياب، دون أي تدخل من النواة. يعيد هذا الروتين NULL إذا فشل.
CLIENT *clnttcp_create(struct sockaddr_in *addr,
                unsigned long prognum, unsigned long versnum,
                int *sockp, unsigned int sendsz, unsigned int recvsz);
يقوم هذا الروتين بإنشاء عميل RPC للبرنامج البعيد prognum، الإصدار versnum؛ يستخدم العميل TCP/IP كوسيلة نقل. يقع البرنامج البعيد في عنوان الإنترنت *addr. إذا كان addr->sin_port صفرًا، فسيتم تعيينه إلى المنفذ الفعلي الذي يستمع إليه البرنامج البعيد (يتم استشارة خدمة portmap البعيدة للحصول على هذه المعلومات). المعلمة sockp هي مقبس؛ إذا كانت RPC_ANYSOCK، فسيقوم هذا الروتين بفتح واحد جديد وتعيين sockp. نظرًا لأن RPC المستند إلى TCP يستخدم الإدخال/الإخراج المخزن مؤقتًا، يمكن للمستخدم تحديد حجم مخازن الإرسال والاستقبال مع المعلمات sendsz و recvsz؛ القيم صفر تختار قيمًا افتراضية مناسبة. يعيد هذا الروتين NULL إذا فشل.
CLIENT *clntudp_create(struct sockaddr_in *addr,
                unsigned long prognum, unsigned long versnum,

يقوم هذا الروتين بإنشاء عميل RPC للبرنامج البعيد prognum، الإصدار versnum؛ يستخدم العميل UDP/IP كوسيلة نقل. يقع البرنامج البعيد في عنوان الإنترنت addr. إذا كان addr->sin_port صفرًا، فسيتم تعيينه إلى المنفذ الفعلي الذي يستمع إليه البرنامج البعيد (يتم استشارة خدمة portmap البعيدة للحصول على هذه المعلومات). المعلمة sockp هي مقبس؛ إذا كانت RPC_ANYSOCK، فسيقوم هذا الروتين بفتح واحد جديد وتعيين sockp. تعيد وسيلة نقل UDP رسالة الاستدعاء بشكل متكرر في فترات زمنية تبلغ wait حتى يتم تلقي استجابة أو حتى ينتهي وقت الاستدعاء. يتم تحديد الوقت الإجمالي لانتهاء صلاحية الاستدعاء بواسطة clnt_call().

متغير عام يتم تعيين قيمته بواسطة أي روتين إنشاء عميل RPC لا ينجح. استخدم الروتين clnt_pcreateerror() لطباعة السبب.
void svc_destroy(SVCXPRT *xprt);
ماكرو يدمر مؤشر نقل خدمة RPC، xprt. يتضمن التدمير عادةً إلغاء تخصيص هياكل البيانات الخاصة، بما في ذلك xprt نفسه. استخدام xprt غير محدد بعد استدعاء هذا الروتين.
fd_set svc_fdset;
متغير عام يعكس قناع بت واصف الملف للقراءة من جانب خدمة RPC؛ وهو مناسب كمعامل لاستدعاء النظام select(2). هذا يهم فقط إذا كان منفذ الخدمة يقوم بمعالجة الأحداث غير المتزامنة الخاصة به، بدلاً من استدعاء svc_run(). هذا المتغير للقراءة فقط (لا تمرر عنوانه إلى select(2)!)، ومع ذلك قد يتغير بعد استدعاءات svc_getreqset() أو أي روتينات إنشاء.
int svc_fds;
مشابه لـ svc_fdset، ولكنه محدود بـ 32 واصف ملف. تم إهمال هذه الواجهة بواسطة svc_fdset.
svc_freeargs(SVCXPRT *xprt, xdrproc_t inproc, char *in);
ماكرو يحرر أي بيانات تم تخصيصها بواسطة نظام RPC/XDR عند فك ترميز وسائط إجراء خدمة باستخدام svc_getargs(). يعيد هذا الروتين 1 إذا تم تحرير النتائج بنجاح، وصفر بخلاف ذلك.
svc_getargs(SVCXPRT *xprt, xdrproc_t inproc, char *in);
ماكرو يقوم بفك ترميز وسائط طلب RPC المرتبط بمؤشر نقل خدمة RPC، xprt. المعامل in هو العنوان الذي سيتم فيه وضع الوسائط؛ inproc هو روتين XDR المستخدم لفك ترميز الوسائط. يعيد هذا الروتين واحدًا إذا نجح فك الترميز، وصفر بخلاف ذلك.
struct sockaddr_in *svc_getcaller(SVCXPRT *xprt);
الطريقة المعتمدة للحصول على عنوان الشبكة للمتصل بإجراء مرتبط بمؤشر نقل خدمة RPC، xprt.
void svc_getreqset(fd_set *rdfds);
هذا الروتين يهم فقط إذا كان منفذ الخدمة لا يستدعي svc_run()، ولكنه بدلاً من ذلك ينفذ معالجة أحداث غير متزامنة مخصصة. يتم استدعاؤه عندما يكون استدعاء النظام select(2) قد حدد أن طلب RPC قد وصل على بعض مقابس RPC؛ rdfds هو قناع بت واصف الملف للقراءة الناتج. يعود الروتين عندما يتم خدمة جميع المقابس المرتبطة بقيمة rdfds.
void svc_getreq(int rdfds);
مشابه لـ svc_getreqset()، ولكنه محدود بـ 32 واصف ملف. تم إهمال هذه الواجهة بواسطة svc_getreqset().
bool_t svc_register(SVCXPRT *xprt, unsigned long prognum,
                    unsigned long versnum,
                    typeof(void (struct svc_req *, SVCXPRT *)) *dispatch,
                    unsigned long protocol);
يربط prognum و versnum بإجراء إرسال الخدمة، dispatch. إذا كان protocol صفرًا، فلن يتم تسجيل الخدمة مع خدمة portmap. إذا كان protocol غير صفري، فسيتم إنشاء تعيين للثلاثي [prognum,versnum,protocol] إلى xprt->xp_port مع خدمة portmap المحلية (عادةً protocol هو صفر، IPPROTO_UDP أو IPPROTO_TCP). الإجراء dispatch له الشكل التالي:

dispatch(struct svc_req *request, SVCXPRT *xprt);

يعيد الروتين svc_register() واحدًا إذا نجح، وصفرًا بخلاف ذلك.
void svc_run(void);
هذا الروتين لا يعود أبدًا. ينتظر وصول طلبات RPC، ويستدعي إجراء الخدمة المناسب باستخدام svc_getreq() عند وصول طلب. هذا الإجراء عادة ما ينتظر استدعاء النظام select(2) للعودة.
bool_t svc_sendreply(SVCXPRT *xprt, xdrproc_t outproc, char *out);
يتم استدعاؤه بواسطة إجراء إرسال لخدمة RPC لإرسال نتائج مكالمة إجراء عن بعد. المعامل xprt هو مؤشر النقل المرتبط بالطلب؛ outproc هو روتين XDR المستخدم لترميز النتائج؛ و out هو عنوان النتائج. يعيد هذا الروتين واحدًا إذا نجح، وصفرًا بخلاف ذلك.
void svc_unregister(unsigned long prognum, unsigned long versnum);
إزالة جميع تعيينات الثنائي [prognum,versnum] إلى إجراءات الإرسال، والتعيينات للثلاثي [prognum,versnum,*] إلى رقم المنفذ.
void svcerr_auth(SVCXPRT *xprt, enum auth_stat why);
يتم استدعاؤه بواسطة إجراء إرسال خدمة يرفض أداء مكالمة إجراء عن بعد بسبب خطأ في المصادقة.
void svcerr_decode(SVCXPRT *xprt);
يتم استدعاؤه بواسطة إجراء إرسال خدمة لا يمكنه فك ترميز وسائطها بنجاح. انظر أيضًا svc_getargs().
void svcerr_noproc(SVCXPRT *xprt);
يتم استدعاؤه بواسطة إجراء إرسال خدمة لا ينفذ رقم الإجراء الذي يطلبه المتصل.
void svcerr_noprog(SVCXPRT *xprt);
يتم استدعاؤه عندما لا يكون البرنامج المطلوب مسجلاً مع حزمة RPC. عادةً لا يحتاج منفذو الخدمة إلى هذا الروتين.
void svcerr_progvers(SVCXPRT *xprt, unsigned long low_vers,
                     unsigned long high_vers);
يتم استدعاؤه عندما لا يكون الإصدار المطلوب من برنامج مسجلاً مع حزمة RPC. عادةً لا يحتاج منفذو الخدمة إلى هذا الروتين.
void svcerr_systemerr(SVCXPRT *xprt);
يتم استدعاؤه بواسطة إجراء إرسال خدمة عندما يكتشف خطأ نظام لا تغطيه بروتوكول معين. على سبيل المثال، إذا لم يعد بإمكان الخدمة تخصيص مساحة تخزين، فقد تستدعي هذا الروتين.
void svcerr_weakauth(SVCXPRT *xprt);
يتم استدعاؤه بواسطة إجراء إرسال خدمة يرفض أداء مكالمة إجراء عن بعد بسبب عدم كفاية معلمات المصادقة. يستدعي الروتين svcerr_auth(xprt, AUTH_TOOWEAK).
SVCXPRT *svcfd_create(int fd, unsigned int sendsize,
                      unsigned int recvsize);
إنشاء خدمة فوق أي واصف ملف مفتوح. عادةً، يكون واصف الملف هذا مقبسًا متصلاً لبروتوكول تدفق مثل TCP. تشير sendsize و recvsize إلى أحجام مخازن الإرسال والاستقبال. إذا كانت صفرًا، يتم اختيار قيمة افتراضية معقولة.
SVCXPRT *svcraw_create(void);
ينشئ هذا الروتين نقل خدمة RPC وهمي، ويعيد إليه مؤشرًا. النقل هو في الواقع مخزن مؤقت داخل مساحة عنوان العملية، لذلك يجب أن يكون عميل RPC المقابل موجودًا في نفس مساحة العنوان؛ انظر clntraw_create(). يسمح هذا الروتين بمحاكاة RPC واكتساب نفقات RPC (مثل أوقات الذهاب والعودة)، دون أي تدخل من النواة. يعيد هذا الروتين NULL إذا فشل.
SVCXPRT *svctcp_create(int sock, unsigned int send_buf_size,
                       unsigned int recv_buf_size);
ينشئ هذا الروتين نقل خدمة RPC قائم على TCP/IP، ويعيد إليه مؤشرًا. يتم ربط النقل بالمقبس sock، والذي يمكن أن يكون RPC_ANYSOCK، وفي هذه الحالة يتم إنشاء مقبس جديد. إذا لم يكن المقبس مرتبطًا بمنفذ TCP محلي، فسيقوم هذا الروتين بربطه بمنفذ عشوائي. عند الانتهاء، يكون xprt->xp_sock هو واصف مقبس النقل، و xprt->xp_port هو رقم منفذ النقل. يعيد هذا الروتين NULL إذا فشل. نظرًا لأن RPC القائم على TCP يستخدم إدخال/إخراج مخزن مؤقت، يمكن للمستخدمين تحديد حجم المخازن المؤقتة؛ القيم الصفرية تختار قيمًا افتراضية مناسبة.
SVCXPRT *svcudp_bufcreate(int sock, unsigned int sendsize,
                          unsigned int recosize);
ينشئ هذا الروتين نقل خدمة RPC قائم على UDP/IP، ويعيد إليه مؤشرًا. يتم ربط النقل بالمقبس sock، والذي يمكن أن يكون RPC_ANYSOCK، وفي هذه الحالة يتم إنشاء مقبس جديد. إذا لم يكن المقبس مرتبطًا بمنفذ UDP محلي، فسيقوم هذا الروتين بربطه بمنفذ عشوائي. عند الانتهاء، يكون xprt->xp_sock هو واصف مقبس النقل، و xprt->xp_port هو رقم منفذ النقل. يعيد هذا الروتين NULL إذا فشل.
يسمح هذا للمستخدم بتحديد الحد الأقصى لحجم الحزمة لإرسال واستقبال رسائل RPC القائمة على UDP.
SVCXPRT *svcudp_create(int sock);
هذا الاستدعاء يعادل svcudp_bufcreate(sock,SZ,SZ) لحجم افتراضي ما SZ.
bool_t xdr_accepted_reply(XDR *xdrs, struct accepted_reply *ar);
يستخدم لترميز رسائل رد RPC. هذا الروتين مفيد للمستخدمين الذين يرغبون في إنشاء رسائل بأسلوب RPC دون استخدام حزمة RPC.
bool_t xdr_authunix_parms(XDR *xdrs, struct authunix_parms *aupp);
يستخدم لوصف بيانات اعتماد UNIX. هذا الروتين مفيد للمستخدمين الذين يرغبون في إنشاء بيانات الاعتماد هذه دون استخدام حزمة مصادقة RPC.
void xdr_callhdr(XDR *xdrs, struct rpc_msg *chdr);
يستخدم لوصف رسائل رؤوس مكالمات RPC. هذا الروتين مفيد للمستخدمين الذين يرغبون في إنشاء رسائل بأسلوب RPC دون استخدام حزمة RPC.

LIBRARY


Ports library (libpnetutil, -lnetutil)






SYNOPSIS AND DESCRIPTION


These routines are part of the Sun Microsystems RPC library.


void svc_freeargs(SVCXPRT *xprt, xdrproc_t xdr_func, char *args);



  

  
Free the arguments that were passed to an RPC procedure. The xdr_func
      parameter is the XDR routine used to deserialize the arguments. The args
      parameter is a pointer to the arguments. This routine should be called by
      service implementors after the call to svc_run(3)
      returns.




bool_t xdr_callmsg(XDR *xdrs, struct rpc_msg *rmsg);



  

  
Used for describing RPC call messages. This routine is useful for users
      who wish to generate RPC-style messages without using the RPC
    package.




bool_t xdr_opaque_auth(XDR *xdrs, struct opaque_auth *ap);



  

  
Used for describing RPC authentication information messages. This routine
      is useful for users who wish to generate RPC-style messages without using
      the RPC package.




bool_t xdr_pmap(XDR *xdrs, struct pmap *regs);



  

  
Used for describing parameters to various portmap procedures,
      externally. This routine is useful for users who wish to generate these
      parameters without using the pmap interface.




bool_t xdr_pmaplist(XDR *xdrs, struct pmaplist **rp);



  

  
Used for describing a list of port mappings, externally. This routine is
      useful for users who wish to generate these parameters without using the
      pmap interface.




bool_t xdr_rejected_reply(XDR *xdrs, struct rejected_reply *rr);



  

  
Used for describing RPC reply messages. This routine is useful for users
      who wish to generate RPC-style messages without using the RPC
    package.




bool_t xdr_replymsg(XDR *xdrs, struct rpc_msg *rmsg);



  

  
Used for describing RPC reply messages. This routine is useful for users
      who wish to generate RPC style messages without using the RPC
    package.




void xprt_register(SVCXPRT *xprt);



  

  
After RPC service transport handles are created, they should register
      themselves with the RPC service package. This routine modifies the global
      variable svc_fds. Service implementors usually do not need this
      routine.




void xprt_unregister(SVCXPRT *xprt);



  

  
Before an RPC service transport handle is destroyed, it should unregister
      itself with the RPC service package. This routine modifies the global
      variable svc_fds. Service implementors usually do not need this
      routine.
