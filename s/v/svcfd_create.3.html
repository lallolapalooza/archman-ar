rpc(3) دليل الدوال المكتبية rpc(3)

الاسم

rpc - إجراءات مكتبية لاستدعاء الإجراءات عن بعد

المكتبة

مكتبة C القياسية (libc،  -lc)

الملخص والوصف

تسمح هذه الإجراءات لبرامج C بإجراء استدعاءات إجراءات على أجهزة أخرى عبر الشبكة. أولاً، يستدعي العميل إجراءً لإرسال حزمة بيانات إلى الخادم. عند استلام الحزمة، يستدعي الخادم إجراءً لتنفيذ الخدمة المطلوبة، ثم يرسل ردًا. أخيرًا، تعود استدعاءات الإجراءات إلى العميل.

للاستفادة من هذه الإجراءات، قم بتضمين ملف الترويسة <rpc/rpc.h>.

تستخدم النماذج الأولية أدناه الأنواع التالية:

typedef int bool_t;
typedef typeof(bool_t (XDR *, void *, ...))  *xdrproc_t;
typedef typeof(bool_t (caddr_t resp, struct sockaddr_in *raddr)
    *resultproc_t;

راجع ملفات الترويسة لتعريفات الأنواع AUTH و CLIENT و SVCXPRT و XDR.

void auth_destroy(AUTH *auth);
ماكرو يدمر معلومات المصادقة المرتبطة بـ auth. عادة ما يتضمن التدمير إلغاء تخصيص هياكل البيانات الخاصة. استخدام auth غير معرف بعد استدعاء auth_destroy().
AUTH *authnone_create(void);
ينشئ ويعيد مقبض مصادقة RPC الذي يمرر معلومات مصادقة غير قابلة للاستخدام مع كل استدعاء إجراء عن بعد. هذه هي المصادقة الافتراضية التي يستخدمها RPC.
AUTH *authunix_create(char *host, uid_t uid, gid_t gid,
                      int n, gid_t aup_gids[n]);
ينشئ ويعيد مقبض مصادقة RPC يحتوي على معلومات المصادقة. المعلمة host هي اسم الجهاز الذي تم إنشاء المعلومات عليه؛ uid هو معرف المستخدم؛ gid هو معرف المجموعة الحالي للمستخدم؛ تشير n و aup_gids إلى مصفوفة محسوبة للمجموعات التي ينتمي إليها المستخدم. من السهل انتحال شخصية مستخدم.
AUTH *authunix_create_default(void);
يستدعي authunix_create() بالمعلمات المناسبة.
int callrpc(char *host, unsigned long prognum,
            unsigned long versnum, unsigned long procnum,
            xdrproc_t inproc, const char *in,
            xdrproc_t outproc, char *out);
استدعاء الإجراء البعيد المرتبط بـ prognum و versnum و procnum على الجهاز host. المعلمة in هي عنوان وسيط (وسائط) الإجراء، و out هو العنوان الذي يجب وضع النتيجة (النتائج) فيه؛ يستخدم inproc لتشفير وسائط الإجراء، ويستخدم outproc لفك تشفير نتائج الإجراء؛ تعيد هذه الدالة صفرًا إذا نجحت، أو قيمة enum clnt_stat محرّفة إلى عدد صحيح إذا فشلت. الدالة clnt_perrno() مفيدة لترجمة حالات الفشل إلى رسائل.
تحذير: استخدام الإجراءات البعيدة بهذه الدالة يستخدم UDP/IP كبروتوكول نقل؛ راجع clntudp_create() للقيود. ليس لديك تحكم في المهلات أو المصادقة باستخدام هذه الدالة.
enum clnt_stat clnt_broadcast(unsigned long prognum,
                     unsigned long versnum, unsigned long procnum,
                     xdrproc_t inproc, char *in,
                     xdrproc_t outproc, char *out,
                     resultproc_t eachresult);
مثل callrpc()، باستثناء أن رسالة الاستدعاء يتم بثها إلى جميع شبكات البث المتصلة محليًا. في كل مرة تتلقى فيها ردًا، تستدعي هذه الدالة eachresult()، والتي يكون شكلها:

eachresult(char *out, struct sockaddr_in *addr);

حيث out هو نفس out الذي تم تمريره إلى clnt_broadcast()، باستثناء أن خرج الإجراء البعيد يتم فك تشفيره هناك؛ يشير addr إلى عنوان الجهاز الذي أرسل النتائج. إذا أعادت eachresult() صفرًا، فإن clnt_broadcast() تنتظر المزيد من الردود؛ وإلا فإنها تعود بحالة مناسبة.
تحذير: مقابس البث محدودة في الحجم بحد أقصى لوحدة النقل الخاصة برابطة البيانات. بالنسبة للإيثرنت، تبلغ هذه القيمة 1500 بايت.
enum clnt_stat clnt_call(CLIENT *clnt, unsigned long procnum,
                    xdrproc_t inproc, char *in,
                    xdrproc_t outproc, char *out,
                    struct timeval tout);
ماكرو يستدعي الإجراء البعيد procnum المرتبط بمقبض العميل clnt، والذي يتم الحصول عليه باستخدام روتين إنشاء عميل RPC مثل clnt_create(). المعلمة in هي عنوان وسيط (وسائط) الإجراء، و out هو العنوان الذي يجب وضع النتيجة (النتائج) فيه؛ يستخدم inproc لتشفير وسائط الإجراء، ويستخدم outproc لفك تشفير نتائج الإجراء؛ tout هو الوقت المسموح به للردود.
clnt_destroy(CLIENT *clnt);
ماكرو يدمر مقبض RPC الخاص بالعميل. عادة ما يتضمن التدمير إلغاء تخصيص هياكل البيانات الخاصة، بما في ذلك clnt نفسه. استخدام clnt غير معرف بعد استدعاء clnt_destroy(). إذا فتحت مكتبة RPC المقبس المرتبط، فستقوم بإغلاقه أيضًا. وإلا، سيظل المقبس مفتوحًا.
CLIENT *clnt_create(const char *host, unsigned long prog,
                    unsigned long vers, const char *proto);
روتين إنشاء عميل عام. يحدد host اسم الجهاز البعيد حيث يقع الخادم. يحدد proto نوع بروتوكول النقل المراد استخدامه. القيم المدعومة حاليًا لهذا الحقل هي "udp" و "tcp". يتم تعيين المهلات الافتراضية، ولكن يمكن تعديلها باستخدام clnt_control().
تحذير: استخدام UDP له قصوره. نظرًا لأن رسائل RPC المستندة إلى UDP يمكن أن تحتوي على ما يصل إلى 8 كيلوبايتات فقط من البيانات المشفرة، فلا يمكن استخدام هذا البروتوكول للإجراءات التي تأخذ وسائط كبيرة أو تعيد نتائج ضخمة.
bool_t clnt_control(CLIENT *cl, int req, char *info);
ماكرو يستخدم لتغيير أو استرداد معلومات مختلفة حول كائن العميل. يشير req إلى نوع العملية، و info هو مؤشر إلى المعلومات. لكل من UDP و TCP، القيم المدعومة لـ req وأنواع وسائطها وما تفعله هي:

CLSET_TIMEOUT  struct timeval // تعيين المهلة الإجمالية
CLGET_TIMEOUT  struct timeval // الحصول على المهلة الإجمالية

ملاحظة: إذا قمت بتعيين المهلة باستخدام clnt_control()، فسيتم تجاهل المعلمة timeout التي تم تمريرها إلى clnt_call() في جميع الاستدعاءات المستقبلية.

CLGET_SERVER_ADDR  struct sockaddr_in
                // الحصول على عنوان الخادم

العمليات التالية صالحة لـ UDP فقط:

CLSET_RETRY_TIMEOUT  struct timeval // تعيين مهلة إعادة المحاولة
CLGET_RETRY_TIMEOUT  struct timeval // الحصول على مهلة إعادة المحاولة

مهلة إعادة المحاولة هي الوقت الذي ينتظره "UDP RPC" حتى يرد الخادم قبل إعادة إرسال الطلب.
clnt_freeres(CLIENT * clnt, xdrproc_t outproc, char *out);
ماكرو يقوم بتحرير أي بيانات تم تخصيصها بواسطة نظام RPC/XDR عند فك تشفير نتائج استدعاء RPC. المعلمة out هي عنوان النتائج، و outproc هو إجراء XDR الذي يصف النتائج. تعيد هذه الدالة واحدًا إذا تم تحرير النتائج بنجاح، وصفرًا بخلاف ذلك.
void clnt_geterr(CLIENT *clnt, struct rpc_err

هذا الموقع غير مرتبط بـ Arch Linux ولا تابع له بأي شكل من الأشكال. المحتوى هو ترجمة فقط لصفحات دليل Arch المنشورة على man.archlinux.org.

الملخص والوصف

SVCXPRT *svcfd_create(int fd, u_int sendsz, u_int recvsz);
يستخدم لإنشاء مقبض ناقل خدمة RPC الذي يتواصل عبر واجهة ملف موصوف. يجب أن يكون fd واجهة ملف موجودة. يجب ضبط sendsz و recvsz إلى الحجم المناسب. يمكن أن تكون قيمتها 0 لاستخدام الحجم الافتراضي.
SVCXPRT *svcraw_create(void);
يستخدم لإنشاء مقبض ناقل خدمة RPC "خام". يسمح هذا للمستخدم بالتعامل مع عمليات الإرسال والاستقبال بنفسه. لا ينبغي أن يستخدمه معظم المطورين.
SVCXPRT *svctcp_create(const char *host, u_int sendsz, u_int recvsz);
يستخدم لإنشاء مقبض ناقل خدمة RPC عبر TCP. إذا كان host هو NULL، فسيتم إنشاء مقبض استماع. وإلا، فسيتم إنشاء مقبض عميل للتواصل مع host. يجب ضبط sendsz و recvsz إلى الحجم المناسب. يمكن أن تكون قيمتها 0 لاستخدام الحجم الافتراضي.
SVCXPRT *svcudp_create(const char *host);
يستخدم لإنشاء مقبض ناقل خدمة RPC عبر UDP. إذا كان host هو NULL، فسيتم إنشاء مقبض استماع. وإلا، فسيتم إنشاء مقبض عميل للتواصل مع host.
SVCXPRT *svcudp_bufcreate(const char *host, u_int sendsz, u_int recvsz);
يستخدم لإنشاء مقبض ناقل خدمة RPC عبر UDP مع تحديد حجم المخزن المؤقت للإرسال والاستقبال. إذا كان host هو NULL، فسيتم إنشاء مقبض استماع. وإلا، فسيتم إنشاء مقبض عميل للتواصل مع host. يجب ضبط sendsz و recvsz إلى الحجم المناسب.
bool_t xdr_accepted_reply(XDR *xdrs, struct rpc_msg *rmsp);
تستخدم لوصف رسائل الرد المقبولة في RPC. هذا الروتين مفيد للمستخدمين الذين يرغبون في إنشاء رسائل بأسلوب RPC دون استخدام حزمة RPC.
bool_t xdr_authunix_parms(XDR *xdrs, struct authunix_parms *aup);
تستخدم لوصف معلمات المصادقة Unix في RPC. هذا الروتين مفيد للمستخدمين الذين يرغبون في إنشاء رسائل بأسلوب RPC دون استخدام حزمة RPC.
bool_t xdr_callhdr(XDR *xdrs, struct rpc_msg *chdr);
تستخدم لوصف رؤوس رسائل استدعاء RPC. هذا الروتين مفيد للمستخدمين الذين يرغبون في إنشاء رسائل بأسلوب RPC دون استخدام حزمة RPC.
bool_t xdr_callmsg(XDR *xdrs, struct rpc_msg *cmsg);
تستخدم لوصف رسائل استدعاء RPC. هذا الروتين مفيد للمستخدمين الذين يرغبون في إنشاء رسائل بأسلوب RPC دون استخدام حزمة RPC.
bool_t xdr_opaque_auth(XDR *xdrs, struct opaque_auth *ap);
تستخدم لوصف رسائل معلومات مصادقة RPC. هذا الروتين مفيد للمستخدمين الذين يرغبون في إنشاء رسائل بأسلوب RPC دون استخدام حزمة RPC.
bool_t xdr_pmap(XDR *xdrs, struct pmap *regs);
تستخدم لوصف معلمات لإجراءات portmap المختلفة، خارجيًا. هذا الروتين مفيد للمستخدمين الذين يرغبون في إنشاء هذه المعلمات دون استخدام واجهة pmap.
bool_t xdr_pmaplist(XDR *xdrs, struct pmaplist **rp);
تستخدم لوصف قائمة تعيينات المنافذ، خارجيًا. هذا الروتين مفيد للمستخدمين الذين يرغبون في إنشاء هذه المعلمات دون استخدام واجهة pmap.
bool_t xdr_rejected_reply(XDR *xdrs, struct rejected_reply *rr);
تستخدم لوصف رسائل الرد في RPC. هذا الروتين مفيد للمستخدمين الذين يرغبون في إنشاء رسائل بأسلوب RPC دون استخدام حزمة RPC.
bool_t xdr_replymsg(XDR *xdrs, struct rpc_msg *rmsg);
تستخدم لوصف رسائل الرد في RPC. هذا الروتين مفيد للمستخدمين الذين يرغبون في إنشاء رسائل بأسلوب RPC دون استخدام حزمة RPC.
void xprt_register(SVCXPRT *xprt);
بعد إنشاء مقابض نقل خدمة RPC، يجب أن تسجل نفسها مع حزمة خدمة RPC. هذا الروتين يعدل المتغير العام svc_fds. لا يحتاج منفذو الخدمة عادةً إلى هذا الروتين.
void xprt_unregister(SVCXPRT *xprt);
قبل تدمير مقبض نقل خدمة RPC، يجب أن يلغي تسجيل نفسه مع حزمة خدمة RPC. هذا الروتين يعدل المتغير العام svc_fds. لا يحتاج منفذو الخدمة عادةً إلى هذا الروتين.
