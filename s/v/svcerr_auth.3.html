الاسم

rpc - روتينات المكتبة لاستدعاءات الإجراءات عن بُعد

المكتبة

مكتبة C القياسية (libc، -lc)

الملخص والوصف

تسمح هذه الروتينات لبرامج C بإجراء استدعاءات إجراءات على أجهزة أخرى عبر الشبكة. أولاً، يستدعي العميل إجراءً لإرسال حزمة بيانات إلى الخادم. عند استلام الحزمة، يستدعي الخادم روتين إرسال لأداء الخدمة المطلوبة، ثم يرسل رداً مرة أخرى. أخيراً، يعود استدعاء الإجراء إلى العميل.

للاستفادة من هذه الروتينات، قم بتضمين ملف الترويسة <rpc/rpc.h>.

تستخدم النماذج الأولية أدناه الأنواع التالية:

typedef int bool_t;
typedef typeof(bool_t (XDR *, void *, ...))  *xdrproc_t;
typedef typeof(bool_t (caddr_t resp, struct sockaddr_in *raddr)
    *resultproc_t;

راجع ملفات الترويسة للحصول على تعريفات أنواع AUTH وCLIENT وSVCXPRT وXDR.

void auth_destroy(AUTH *auth);
ماكرو يدمّر معلومات المصادقة المرتبطة بـauth. يتضمن التدمير عادةً إلغاء تخصيص هياكل البيانات الخاصة. استخدام auth غير معرّف بعد استدعاء auth_destroy().
AUTH *authnone_create(void);
إنشاء وإرجاع مقبض مصادقة RPC يمرر معلومات مصادقة غير قابلة للاستخدام مع كل استدعاء إجراء عن بُعد. هذه هي المصادقة الافتراضية التي يستخدمها RPC.
AUTH *authunix_create(char *host, uid_t uid, gid_t gid,
                      int n, gid_t aup_gids[n]);
إنشاء وإرجاع مقبض مصادقة RPC يحتوي على معلومات المصادقة. المعامل host هو اسم الجهاز الذي أُنشئت عليه المعلومات؛ uid هو معرّف المستخدم للمستخدم؛ gid هو معرّف المجموعة الحالي للمستخدم؛ n وaup_gids يشيران إلى مصفوفة معدودة من المجموعات التي ينتمي إليها المستخدم. من السهل انتحال شخصية مستخدم.
AUTH *authunix_create_default(void);
يستدعي authunix_create() بالمعاملات المناسبة.
int callrpc(char *host, unsigned long prognum,
            unsigned long versnum, unsigned long procnum,
            xdrproc_t inproc, const char *in,
            xdrproc_t outproc, char *out);
استدعاء الإجراء البعيد المرتبط بـprognum وversnum وprocnum على الجهاز host. المعامل in هو عنوان وسيط/وسائط الإجراء، وout هو عنوان مكان وضع النتيجة/النتائج؛ يُستخدم inproc لتشفير معاملات الإجراء، ويُستخدم outproc لفك تشفير نتائج الإجراء. تُرجع هذه الروتين صفراً إذا نجحت، أو قيمة enum clnt_stat محوّلة إلى عدد صحيح إذا فشلت. الروتين clnt_perrno() مفيد لترجمة حالات الفشل إلى رسائل.
تحذير: استدعاء الإجراءات البعيدة بهذا الروتين يستخدم UDP/IP كناقل؛ راجع clntudp_create() للقيود. ليس لديك تحكم في المهلات أو المصادقة باستخدام هذا الروتين.
enum clnt_stat clnt_broadcast(unsigned long prognum,
                     unsigned long versnum, unsigned long procnum,
                     xdrproc_t inproc, char *in,
                     xdrproc_t outproc, char *out,
                     resultproc_t eachresult);
مثل callrpc()، باستثناء أن رسالة الاستدعاء تُبث إلى جميع شبكات البث المتصلة محلياً. في كل مرة يتلقى فيها استجابة، يستدعي هذا الروتين eachresult()، الذي يكون شكله:

eachresult(char *out, struct sockaddr_in *addr);

حيث out هو نفسه out الممرر إلى clnt_broadcast()، باستثناء أن مخرجات الإجراء البعيد تُفكّ تشفيرها هناك؛ addr يشير إلى عنوان الجهاز الذي أرسل النتائج. إذا أرجع eachresult() صفراً، ينتظر clnt_broadcast() المزيد من الردود؛ وإلا فإنه يعود بالحالة المناسبة.
تحذير: مقابس البث محدودة الحجم بوحدة النقل القصوى لرابط البيانات. بالنسبة لإيثرنت، هذه القيمة هي 1500 بايت.
enum clnt_stat clnt_call(CLIENT *clnt, unsigned long procnum,
                    xdrproc_t inproc, char *in,
                    xdrproc_t outproc, char *out,
                    struct timeval tout);
ماكرو يستدعي الإجراء البعيد procnum المرتبط بمقبض العميل clnt، الذي يتم الحصول عليه باستخدام روتين إنشاء عميل RPC مثل clnt_create(). المعامل in هو عنوان وسيط/وسائط الإجراء، وout هو عنوان مكان وضع النتيجة/النتائج؛ يُستخدم inproc لتشفير معاملات الإجراء، ويُستخدم outproc لفك تشفير نتائج الإجراء؛ tout هو الوقت المسموح لعودة النتائج.
clnt_destroy(CLIENT *clnt);
ماكرو يدمّر مقبض RPC الخاص بالعميل. يتضمن التدمير عادةً إلغاء تخصيص هياكل البيانات الخاصة، بما في ذلك clnt نفسه. استخدام clnt غير معرّف بعد استدعاء clnt_destroy(). إذا فتحت مكتبة RPC المقبس المرتبط، فستغلقه أيضاً. وإلا، يبقى المقبس مفتوحاً.
CLIENT *clnt_create(const char *host, unsigned long prog,
                    unsigned long vers, const char *proto);
روتين إنشاء عميل عام. يحدد host اسم المضيف البعيد حيث يوجد الخادم. يشير proto إلى نوع بروتوكول النقل المراد استخدامه. القيم المدعومة حالياً لهذا الحقل هي “udp” و“tcp”. يتم تعيين المهلات الافتراضية، ولكن يمكن تعديلها باستخدام clnt_control().
تحذير: استخدام UDP له عيوبه. نظراً لأن رسائل RPC القائمة على UDP يمكنها حمل ما يصل إلى 8 كيلوبايت فقط من البيانات المشفرة، لا يمكن استخدام هذا الناقل للإجراءات التي تأخذ وسائط كبيرة أو تُرجع نتائج ضخمة.
bool_t clnt_control(CLIENT *cl, int req, char *info);
ماكرو يُستخدم لتغيير أو استرداد معلومات متنوعة حول كائن عميل. يشير req إلى نوع العملية، وinfo هو مؤشر إلى المعلومات. لكل من UDP وTCP، القيم المدعومة لـreq وأنواع وسائطها وما تفعله هي:

CLSET_TIMEOUT  struct timeval // تعيين المهلة الإجمالية
CLGET_TIMEOUT  struct timeval // الحصول على المهلة الإجمالية

ملاحظة: إذا قمت بتعيين المهلة باستخدام clnt_control()، فسيتم تجاهل معامل المهلة الممرر إلى clnt_call() في جميع الاستدعاءات المستقبلية.

CLGET_SERVER_ADDR  struct sockaddr_in
                // الحصول على عنوان الخادم

العمليات التالية صالحة لـ UDP فقط:

CLSET_RETRY_TIMEOUT  struct timeval // تعيين مهلة إعادة المحاولة
CLGET_RETRY_TIMEOUT  struct timeval // الحصول على مهلة إعادة المحاولة

مهلة إعادة المحاولة هي الوقت الذي ينتظره "UDP RPC" لرد الخادم قبل إعادة إرسال الطلب.
clnt_freeres(CLIENT * clnt, xdrproc_t outproc, char *out);
ماكرو يحرر أي بيانات خصصها نظام RPC/XDR عندما فكّ تشفير نتائج استدعاء RPC. المعامل out هو عنوان النتائج، وoutproc هو روتين XDR الذي يصف النتائج. تُرجع هذه الروتين واحداً إذا تم تحرير النتائج بنجاح، وصفراً بخلاف ذلك.
void clnt_geterr(CLIENT *clnt, struct rpc_err*errp);
ماكرو يقوم بنسخ بنية الخطأ من مقبض العميل إلى البنية الموجودة في العنوان errp.
void clnt_pcreateerror(const char *s);
طباعة رسالة إلى الخطأ القياسي تشير إلى سبب عدم إنشاء مقبض عميل RPC. تسبق الرسالة بالسلسلة s ونقطتين. تستخدم عندما تفشل استدعاءات clnt_create() أو clntraw_create() أو clnttcp_create() أو clntudp_create().
void clnt_perrno(enum clnt_stat stat);
طباعة رسالة إلى الخطأ القياسي تتوافق مع الحالة التي يشير إليها stat. تستخدم بعد callrpc().
clnt_perror(CLIENT *clnt, const char *s);
طباعة رسالة إلى الخطأ القياسي تشير إلى سبب فشل استدعاء RPC؛ clnt هو المقبض المستخدم لإجراء الاستدعاء. تسبق الرسالة بالسلسلة s ونقطتين. تستخدم بعد clnt_call().
char *clnt_spcreateerror(const char *s);
مثل clnt_pcreateerror()، ولكنها تعيد سلسلة بدلاً من الطباعة إلى الخطأ القياسي.
عيوب: تعيد مؤشرًا إلى بيانات ثابتة يتم الكتابة فوقها في كل استدعاء.
char *clnt_sperrno(enum clnt_stat stat);
تأخذ نفس وسائط clnt_perrno()، ولكن بدلاً من إرسال رسالة إلى الخطأ القياسي تشير إلى سبب فشل استدعاء RPC، تعيد مؤشرًا إلى سلسلة تحتوي على الرسالة. تنتهي السلسلة بسطر جديد.
يتم استخدام clnt_sperrno() بدلاً من clnt_perrno() إذا لم يكن للبرنامج خطأ قياسي (كما هو الحال في برنامج يعمل كخادم)، أو إذا كان المبرمج لا يريد أن يتم إخراج الرسالة باستخدام printf(3)، أو إذا كان يجب استخدام تنسيق رسالة مختلف عن ذلك الذي تدعمه clnt_perrno(). ملاحظة: على عكس clnt_sperror() و clnt_spcreateerror()، فإن clnt_sperrno() تعيد مؤشرًا إلى بيانات ثابتة، ولكن لن يتم الكتابة فوق النتيجة في كل استدعاء.
char *clnt_sperror(CLIENT *rpch, const char *s);
مثل clnt_perror()، ولكنها (مثل clnt_sperrno()) تعيد سلسلة بدلاً من الطباعة إلى الخطأ القياسي.
عيوب: تعيد مؤشرًا إلى بيانات ثابتة يتم الكتابة فوقها في كل استدعاء.
CLIENT *clntraw_create(unsigned long prognum, unsigned long versnum);
يقوم هذا الروتين بإنشاء عميل RPC وهمي للبرنامج البعيد prognum، الإصدار versnum. الوسيلة المستخدمة لتمرير الرسائل إلى الخدمة هي في الواقع مخزن مؤقت داخل مساحة عنوان العملية، لذلك يجب أن يعيش خادم RPC المقابل في نفس مساحة العنوان؛ انظر svcraw_create(). هذا يسمح بمحاكاة RPC واكتساب نفقات RPC، مثل أوقات الذهاب والإياب، دون أي تدخل من النواة. يعيد هذا الروتين NULL إذا فشل.
CLIENT *clnttcp_create(struct sockaddr_in *addr,
                unsigned long prognum, unsigned long versnum,
                int *sockp, unsigned int sendsz, unsigned int recvsz);
يقوم هذا الروتين بإنشاء عميل RPC للبرنامج البعيد prognum، الإصدار versnum؛ يستخدم العميل TCP/IP كوسيلة نقل. يقع البرنامج البعيد في عنوان الإنترنت *addr. إذا كان addr->sin_port صفرًا، فسيتم تعيينه إلى المنفذ الفعلي الذي يستمع إليه البرنامج البعيد (يتم استشارة خدمة portmap البعيدة لهذه المعلومات). المعامل sockp هو مقبس؛ إذا كان RPC_ANYSOCK، فسيقوم هذا الروتين بفتح مقبس جديد وتعيين sockp. نظرًا لأن RPC المستند إلى TCP يستخدم إدخال/إخراج مخزن مؤقتًا، يمكن للمستخدم تحديد حجم المخازن المؤقتة للإرسال والاستقبال باستخدام المعاملين sendsz و recvsz؛ القيم صفر تختار القيم الافتراضية المناسبة. يعيد هذا الروتين NULL إذا فشل.
CLIENT *clntudp_create(struct sockaddr_in *addr,
                unsigned long prognum, unsigned long versnum,
                struct timeval wait, int *sockp);
يقوم هذا الروتين بإنشاء عميل RPC للبرنامج البعيد prognum، الإصدار versnum؛ يستخدم العميل UDP/IP كوسيلة نقل. يقع البرنامج البعيد في عنوان الإنترنت addr. إذا كان addr->sin_port صفرًا، فسيتم تعيينه إلى المنفذ الفعلي الذي يستمع إليه البرنامج البعيد (يتم استشارة خدمة portmap البعيدة لهذه المعلومات). المعامل sockp هو مقبس؛ إذا كان RPC_ANYSOCK، فسيقوم هذا الروتين بفتح مقبس جديد وتعيين sockp. تقوم وسيلة نقل UDP بإعادة إرسال رسالة الاستدعاء بفواصل زمنية قدرها wait حتى يتم استلام استجابة أو حتى تنتهي مهلة الاستدعاء. يتم تحديد الوقت الإجمالي لانتهاء مهلة الاستدعاء بواسطة clnt_call().
تحذير: نظرًا لأن رسائل RPC المستندة إلى UDP يمكن أن تحمل ما يصل إلى 8 كيلوبايت من البيانات المشفرة فقط، فلا يمكن استخدام وسيلة النقل هذه للإجراءات التي تأخذ وسائط كبيرة أو تعيد نتائج ضخمة.
CLIENT *clntudp_bufcreate(struct sockaddr_in *addr,
            unsigned long prognum, unsigned long versnum,
            struct timeval wait, int *sockp,
            unsigned int sendsize, unsigned int recosize);
يقوم هذا الروتين بإنشاء عميل RPC للبرنامج البعيد prognum، على versnum؛ يستخدم العميل UDP/IP كوسيلة نقل. يقع البرنامج البعيد في عنوان الإنترنت addr. إذا كان addr->sin_port صفرًا، فسيتم تعيينه إلى المنفذ الفعلي الذي يستمع إليه البرنامج البعيد (يتم استشارة خدمة portmap البعيدة لهذه المعلومات). المعامل sockp هو مقبس؛ إذا كان RPC_ANYSOCK، فسيقوم هذا الروتين بفتح مقبس جديد وتعيين sockp. تقوم وسيلة نقل UDP بإعادة إرسال رسالة الاستدعاء بفواصل زمنية قدرها wait حتى يتم استلام استجابة أو حتى تنتهي مهلة الاستدعاء. يتم تحديد الوقت الإجمالي لانتهاء مهلة الاستدعاء بواسطة clnt_call().
هذا يسمح للمستخدم بتحديد الحد الأقصى لحجم الحزمة لإرسال واستقبال رسائل RPC المستندة إلى UDP.
void get_myaddress(struct sockaddr_in *addr);
وضع عنوان IP للجهاز في *addr، دون استشارة روتينيات المكتبة التي تتعامل مع /etc/hosts. يتم دائمًا تعيين رقم المنفذ إلى htons(PMAPPORT).
struct pmaplist *pmap_getmaps(struct sockaddr_in *addr);
واجهة مستخدم لخدمة portmap، والتي تعيد قائمة بخرائط البرامج إلى المنافذ الحالية لـ RPC على المضيف الموجود في عنوان IP *addr. يمكن أن يعيد هذا الروتين NULL. يستخدم الأمر rpcinfo -p هذا الروتين.
unsigned short pmap_getport(struct sockaddr_in *addr,
                    unsigned long prognum, unsigned long versnum,
                    unsigned int protocol);
واجهة مستخدم لخدمة portmap، والتي تعيد رقم المنفذ الذي تنتظر عليه خدمة تدعم رقم البرنامج prognum، والإصدار versnum، وتتحدث بروتوكول النقل المرتبط بـ protocol. قيمة protocol على الأرجح IPPROTO_UDP أو IPPROTO_TCP. القيمة المرجعة صفر تعني أن التعيين غير موجود أو أن نظام RPC فشل في الاتصال بخدمة portmap البعيدة. في الحالة الأخيرة، تحتوي المتغير العام rpc_createerr على حالة RPC.
enum clnt_stat pmap_rmtcall(struct sockaddr_in *addr,
                    unsigned long prognum, unsigned long versnum,
                    unsigned long procnum,
                    xdrproc_t inproc, char *in,
                    xdrproc_t outproc, char *out,
                    struct timeval tout, unsigned long *portp);
واجهة مستخدم لخدمة portmap، والتي توجه portmap على المضيف في عنوان IP *addr لإجراء استدعاء RPC نيابة عنك إلى إجراء على ذلك المضيف. سيتم تعديل المعامل *portp إلى رقم منفذ البرنامج إذا نجح الإجراء. يتم مناقشة تعريفات المعاملات الأخرى في callrpc() و clnt_call(). يجب استخدام هذا الإجراء لـ "ping" وليس لأي شيء آخر. انظر أيضًا clnt_broadcast().
bool_t pmap_set(unsigned long prognum, unsigned long versnum,
                int protocol, unsigned short port);
واجهة مستخدم لخدمة portmap، والتي تنشئ تعيينًا بين الثلاثي [prognum,versnum,protocol] و port على خدمة portmap للجهاز. قيمة protocol على الأرجح IPPROTO_UDP أو IPPROTO_TCP. يعيد هذا الروتين واحدًا إذا نجح، وصفرًا بخلاف ذلك. يتم القيام بذلك تلقائيًا بواسطة svc_register().
bool_t pmap_unset(unsigned long prognum, unsigned long versnum);
واجهة مستخدم لخدمة portmap، والتي تدمر جميع التعيينات بين الثلاثي [prognum,versnum,*] و ports على خدمة portmap للجهاز. يعيد هذا الروتين واحدًا إذا نجح، وصفرًا بخلاف ذلك.
int registerrpc(unsigned long prognum, unsigned long versnum,
                unsigned long procnum, typeof(char *(char *)) *procname,
                xdrproc_t inproc, xdrproc_t outproc);
تسجيل الإجراء procname مع حزمة خدمة RPC. إذا وصل طلب للبرنامج prognum، والإصدار versnum، والإجراء procnum، يتم استدعاء procname بمؤشر إلى وسيطاته؛ يجب أن يعيد procname مؤشرًا إلى نتائجه الثابتة؛ يتم استخدام inproc لفك ترميز الوسائط بينما يتم استخدام outproc لترميز النتائج. يعيد هذا الروتين صفرًا إذا نجح التسجيل، -1 بخلاف ذلك.
تحذير: يتم الوصول إلى الإجراءات البعيدة المسجلة بهذا الشكل باستخدام وسيلة نقل UDP/IP
المنفذ؛ انظر svcudp_create() للاطلاع على القيود. 
struct rpc_createerr rpc_createerr;
متغير عام يتم تعيين قيمته بواسطة أي روتين لإنشاء عميل RPC لا ينجح. استخدم الروتين clnt_pcreateerror() لطباعة سبب الفشل.
void svc_destroy(SVCXPRT *xprt);
ماكرو يدمّر مقبض نقل خدمة RPC، xprt. يتضمن التدمير عادةً إلغاء تخصيص هياكل البيانات الخاصة، بما في ذلك xprt نفسه. استخدام xprt غير معرّف بعد استدعاء هذا الروتين.
fd_set svc_fdset;
متغير عام يعكس قناع بتات واصفات الملفات القابلة للقراءة على جانب خدمة RPC؛ وهو مناسب كمعامل لاستدعاء النظام select(2). هذا ذو أهمية فقط إذا كان منفّذ الخدمة يقوم بمعالجة الأحداث غير المتزامنة الخاصة به، بدلاً من استدعاء svc_run(). هذا المتغير للقراءة فقط (لا تمرّر عنوانه إلى select(2)!)، ومع ذلك قد يتغير بعد استدعاءات svc_getreqset() أو أي روتين إنشاء.
int svc_fds;
مشابه لـ svc_fdset، لكنه محدود بـ 32 واصف ملف. هذه الواجهة أصبحت مهجورة بواسطة svc_fdset.
svc_freeargs(SVCXPRT *xprt, xdrproc_t inproc, char *in);
ماكرو يحرّر أي بيانات خصصها نظام RPC/XDR عندما فكّ ترميز المعاملات لإجراء خدمة باستخدام svc_getargs(). يعيد هذا الروتين 1 إذا تم تحرير النتائج بنجاح، وصفراً بخلاف ذلك.
svc_getargs(SVCXPRT *xprt, xdrproc_t inproc, char *in);
ماكرو يفكّ ترميز معاملات طلب RPC المرتبط بمقبض نقل خدمة RPC، xprt. المعامل in هو العنوان الذي ستوضع فيه المعاملات؛ inproc هو روتين XDR المستخدم لفكّ ترميز المعاملات. يعيد هذا الروتين واحداً إذا نجح فكّ الترميز، وصفراً بخلاف ذلك.
struct sockaddr_in *svc_getcaller(SVCXPRT *xprt);
الطريقة المعتمدة للحصول على عنوان الشبكة لمتصل إجراء مرتبط بمقبض نقل خدمة RPC، xprt.
void svc_getreqset(fd_set *rdfds);
هذا الروتين ذو أهمية فقط إذا لم يستدعِ منفّذ الخدمة svc_run()، بل نفّذ بدلاً من ذلك معالجة أحداث غير متزامنة مخصصة. يُستدعى عندما يحدد استدعاء النظام select(2) أن طلب RPC قد وصل على بعض مقابس RPC؛ rdfds هو قناع بتات واصفات الملفات القابلة للقراءة الناتج. يعود الروتين عندما تتم خدمة جميع المقابس المرتبطة بقيمة rdfds.
void svc_getreq(int rdfds);
مشابه لـ svc_getreqset()، لكنه محدود بـ 32 واصف ملف. هذه الواجهة أصبحت مهجورة بواسطة svc_getreqset().
bool_t svc_register(SVCXPRT *xprt, unsigned long prognum,
                    unsigned long versnum,
                    typeof(void (struct svc_req *, SVCXPRT *)) *dispatch,
                    unsigned long protocol);
يربط prognum وversnum بإجراء إرسال الخدمة، dispatch. إذا كان protocol صفراً، فلا يتم تسجيل الخدمة مع خدمة portmap. إذا كان protocol غير صفري، فسيتم إنشاء تعيين للثلاثية [prognum,versnum,protocol] إلى xprt->xp_port مع خدمة portmap المحلية (عادةً يكون protocol صفراً، أو IPPROTO_UDP أو IPPROTO_TCP). الإجراء dispatch له الشكل التالي:

dispatch(struct svc_req *request, SVCXPRT *xprt);

يعيد الروتين svc_register() واحداً إذا نجح، وصفراً بخلاف ذلك.
void svc_run(void);
هذا الروتين لا يعود أبداً. ينتظر وصول طلبات RPC، ويستدعي إجراء الخدمة المناسب باستخدام svc_getreq() عند وصول طلب. عادةً ما ينتظر هذا الإجراء عودة استدعاء النظام select(2).
bool_t svc_sendreply(SVCXPRT *xprt, xdrproc_t outproc, char *out);
يُستدعى بواسطة روتين إرسال خدمة RPC لإرسال نتائج استدعاء إجراء عن بُعد. المعامل xprt هو مقبض النقل المرتبط بالطلب؛ outproc هو روتين XDR المستخدم لتشفير النتائج؛ وout هو عنوان النتائج. يعيد هذا الروتين واحداً إذا نجح، وصفراً بخلاف ذلك.
void svc_unregister(unsigned long prognum, unsigned long versnum);
يزيل جميع تعيينات الثنائية [prognum,versnum] إلى إجراءات الإرسال، والثلاثية [prognum,versnum,*] إلى رقم المنفذ.
void svcerr_auth(SVCXPRT *xprt, enum auth_stat why);
يُستدعى بواسطة روتين إرسال خدمة يرفض تنفيذ استدعاء إجراء عن بُعد بسبب خطأ في المصادقة.
void svcerr_decode(SVCXPRT *xprt);
يُستدعى بواسطة روتين إرسال خدمة لا يمكنه فكّ ترميز معاملاته بنجاح. انظر أيضاً svc_getargs().
void svcerr_noproc(SVCXPRT *xprt);
يُستدعى بواسطة روتين إرسال خدمة لا ينفّذ رقم الإجراء الذي يطلبه المتصل.
void svcerr_noprog(SVCXPRT *xprt);
يُستدعى عندما لا يكون البرنامج المطلوب مسجلاً مع حزمة RPC. عادةً لا يحتاج منفّذو الخدمة إلى هذا الروتين.
void svcerr_progvers(SVCXPRT *xprt, unsigned long low_vers,
                     unsigned long high_vers);
يُستدعى عندما لا تكون النسخة المطلوبة من برنامج مسجلة مع حزمة RPC. عادةً لا يحتاج منفّذو الخدمة إلى هذا الروتين.
void svcerr_systemerr(SVCXPRT *xprt);
يُستدعى بواسطة روتين إرسال خدمة عندما يكتشف خطأ نظام لا يغطيه أي بروتوكول معين. على سبيل المثال، إذا لم تعد الخدمة قادرة على تخصيص مساحة تخزين، فقد تستدعي هذا الروتين.
void svcerr_weakauth(SVCXPRT *xprt);
يُستدعى بواسطة روتين إرسال خدمة يرفض تنفيذ استدعاء إجراء عن بُعد بسبب معاملات مصادقة غير كافية. يستدعي الروتين svcerr_auth(xprt, AUTH_TOOWEAK).
SVCXPRT *svcfd_create(int fd, unsigned int sendsize,
                      unsigned int recvsize);
ينشئ خدمة فوق أي واصف ملف مفتوح. عادةً، يكون واصف الملف هذا مقبساً متصلاً لبروتوكول تدفق مثل TCP. تشير sendsize وrecvsize إلى أحجام مخازن الإرسال والاستقبال المؤقتة. إذا كانتا صفراً، يتم اختيار قيمة افتراضية معقولة.
SVCXPRT *svcraw_create(void);
ينشئ هذا الروتين نقل خدمة RPC وهمياً، ويعيد مؤشراً إليه. النقل هو في الواقع مخزن مؤقت داخل مساحة عنوان العملية، لذا يجب أن يكون عميل RPC المقابل في نفس مساحة العنوان؛ انظر clntraw_create(). يسمح هذا الروتين بمحاكاة RPC وقياس تكاليف RPC (مثل أزمنة الذهاب والإياب)، دون أي تدخل من النواة. يعيد هذا الروتين NULL إذا فشل.
SVCXPRT *svctcp_create(int sock, unsigned int send_buf_size,
                       unsigned int recv_buf_size);
ينشئ هذا الروتين نقل خدمة RPC قائماً على TCP/IP، ويعيد مؤشراً إليه. يرتبط النقل بالمقبس sock، الذي قد يكون RPC_ANYSOCK، وفي هذه الحالة يتم إنشاء مقبس جديد. إذا لم يكن المقبس مرتبطاً بمنفذ TCP محلي، فإن هذا الروتين يربطه بمنفذ عشوائي. عند الاكتمال، يكون xprt->xp_sock واصف مقبس النقل، وxprt->xp_port رقم منفذ النقل. يعيد هذا الروتين NULL إذا فشل. نظراً لأن RPC القائم على TCP يستخدم إدخال/إخراج مخزّن مؤقتاً، يمكن للمستخدمين تحديد حجم المخازن المؤقتة؛ القيم الصفرية تختار قيماً افتراضية مناسبة.
SVCXPRT *svcudp_bufcreate(int sock, unsigned int sendsize,
                          unsigned int recosize);
ينشئ هذا الروتين نقل خدمة RPC قائماً على UDP/IP، ويعيد مؤشراً إليه. يرتبط النقل بالمقبس sock، الذي قد يكون RPC_ANYSOCK، وفي هذه الحالة يتم إنشاء مقبس جديد. إذا لم يكن المقبس مرتبطاً بمنفذ UDP محلي، فإن هذا الروتين يربطه بمنفذ عشوائي. عند الاكتمال، يكون xprt->xp_sock واصف مقبس النقل، وxprt->xp_port رقم منفذ النقل. يعيد هذا الروتين NULL إذا فشل.
يسمح هذا للمستخدم بتحديد الحد الأقصى لحجم الحزمة لإرسال واستقبال رسائل RPC القائمة على UDP.
SVCXPRT *svcudp_create(int sock);
هذا الاستدعاء مكافئ لـ svcudp_bufcreate(sock,SZ,SZ) لبعض الحجم الافتراضي SZ.
bool_t xdr_accepted_reply(XDR *xdrs, struct accepted_reply *ar);
يُستخدم لتشفير رسائل رد RPC. هذا الروتين مفيد للمستخدمين الذين يرغبون في إنشاء رسائل بنمط RPC دون استخدام حزمة RPC.
bool_t xdr_authunix_parms(XDR *xdrs, struct authunix_parms *aupp);
يُستخدم لوصف بيانات اعتماد UNIX. هذا الروتين مفيد للمستخدمين الذين يرغبون في إنشاء بيانات الاعتماد هذه دون استخدام حزمة مصادقة RPC.
void xdr_callhdr(XDR *xdrs, struct rpc_msg *chdr);
يُستخدم لوصف رسائل رأس استدعاء RPC. هذا الروتين مفيد للمستخدمين الذين يرغبون في إنشاء رسائل بنمط RPC دون استخدام حزمة RPC.
bool_t xdr_callmsg(XDR *xdrs, struct rpc_msg *cmsg);
يُستخدم لوصف رسائل استدعاء RPC. هذا الإجراء مفيد للمستخدمين الذين يرغبون في إنشاء رسائل على غرار RPC دون استخدام حزمة RPC.
bool_t xdr_opaque_auth(XDR *xdrs, struct opaque_auth *ap);
يُستخدم لوصف رسائل معلومات مصادقة RPC. هذا الإجراء مفيد للمستخدمين الذين يرغبون في إنشاء رسائل على غرار RPC دون استخدام حزمة RPC.
bool_t xdr_pmap(XDR *xdrs, struct pmap *regs);
يُستخدم لوصف معلمات لإجراءات portmap مختلفة، خارجيًا. هذا الإجراء مفيد للمستخدمين الذين يرغبون في إنشاء هذه المعلمات دون استخدام واجهة pmap.
bool_t xdr_pmaplist(XDR *xdrs, struct pmaplist **rp);
يُستخدم لوصف قائمة تعيينات المنافذ، خارجيًا. هذا الإجراء مفيد للمستخدمين الذين يرغبون في إنشاء هذه المعلمات دون استخدام واجهة pmap.
bool_t xdr_rejected_reply(XDR *xdrs, struct rejected_reply *rr);
يُستخدم لوصف رسائل رد RPC. هذا الإجراء مفيد للمستخدمين الذين يرغبون في إنشاء رسائل على غرار RPC دون استخدام حزمة RPC.
bool_t xdr_replymsg(XDR *xdrs, struct rpc_msg *rmsg);
يُستخدم لوصف رسائل رد RPC. هذا الإجراء مفيد للمستخدمين الذين يرغبون في إنشاء رسائل على غرار RPC دون استخدام حزمة RPC.
void xprt_register(SVCXPRT *xprt);
بعد إنشاء معالجات نقل خدمة RPC، يجب عليها تسجيل نفسها مع حزمة خدمة RPC. هذا الإجراء يعدل المتغير العام svc_fds. عادةً ما لا يحتاج منفذو الخدمة إلى هذا الإجراء.
void xprt_unregister(SVCXPRT *xprt);
قبل تدمير معالج نقل خدمة RPC، يجب عليه إلغاء تسجيل نفسه مع حزمة خدمة RPC. هذا الإجراء يعدل المتغير العام svc_fds. عادةً ما لا يحتاج منفذو الخدمة إلى هذا الإجراء.
