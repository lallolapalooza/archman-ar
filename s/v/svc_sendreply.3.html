svc_sendreply(3)

rpc(3) دليل دوال المكتبة rpc(3)

الاسم

rpc - روتينيات المكتبة لاستدعاءات الإجراءات عن بعد

المكتبة

مكتبة C القياسية (libc, -lc)

الملخص والوصف

تسمح هذه الروتينيات لبرامج C بإجراء استدعاءات إجراءات على أجهزة أخرى عبر الشبكة. أولاً، يستدعي العميل إجراءً لإرسال حزمة بيانات إلى الخادم. عند استلام الحزمة، يستدعي الخادم روتين تفويض لأداء الخدمة المطلوبة، ثم يرسل ردًا. أخيرًا، يعود استدعاء الإجراء إلى العميل.

للاستفادة من هذه الروتينيات، قم بتضمين ملف الرأس <rpc/rpc.h>.

النماذج الأولية أدناه تستفيد من الأنواع التالية:

typedef int bool_t;
typedef typeof(bool_t (XDR *, void *, ...))  *xdrproc_t;
typedef typeof(bool_t (caddr_t resp, struct sockaddr_in *raddr)
    *resultproc_t;

راجع ملفات الرأس لتعريفات الأنواع AUTH، CLIENT، SVCXPRT، و XDR.

void auth_destroy(AUTH *auth);
ماكرو يدمر معلومات المصادقة المرتبطة بـ auth. عادة ما يتضمن التدمير إلغاء تخصيص هياكل البيانات الخاصة. استخدام auth غير محدد بعد استدعاء auth_destroy().
AUTH *authnone_create(void);
إنشاء وإرجاع مقبض مصادقة RPC يمرر معلومات مصادقة غير قابلة للاستخدام مع كل استدعاء إجراء عن بعد. هذه هي المصادقة الافتراضية التي تستخدمها RPC.
AUTH *authunix_create(char *host, uid_t uid, gid_t gid,
                      int n, gid_t aup_gids[n]);
إنشاء وإرجاع مقبض مصادقة RPC يحتوي على معلومات المصادقة. المعامل host هو اسم الجهاز الذي تم إنشاء المعلومات عليه؛ uid هو معرف المستخدم؛ gid هو معرف المجموعة الحالي للمستخدم؛ n و aup_gids يشيران إلى مصفوفة محسوبة للمجموعات التي ينتمي إليها المستخدم. من السهل انتحال هوية مستخدم.
AUTH *authunix_create_default(void);
يستدعي authunix_create() بالمعاملات المناسبة.
int callrpc(char *host, unsigned long prognum,
            unsigned long versnum, unsigned long procnum,
            xdrproc_t inproc, const char *in,
            xdrproc_t outproc, char *out);
استدعاء الإجراء البعيد المرتبط بـ prognum و versnum و procnum على الجهاز host. المعامل in هو عنوان الوسيط (الوسائط) للإجراء، و out هو العنوان الذي سيتم وضع النتائج فيه؛ inproc يستخدم لتشفير معلمات الإجراء، و outproc يستخدم لفك تشفير نتائج الإجراء. يعيد هذا الروتين صفرًا إذا نجح، أو قيمة enum clnt_stat محولة إلى عدد صحيح إذا فشل. الروتين clnt_perrno() مفيد لترجمة حالات الفشل إلى رسائل.
تحذير: استدعاء الإجراءات البعيدة باستخدام هذا الروتين يستخدم UDP/IP كـ نقل؛ راجع clntudp_create() للقيود. لا يمكنك التحكم في المهلات أو المصادقة باستخدام هذا الروتين.
enum clnt_stat clnt_broadcast(unsigned long prognum,
                     unsigned long versnum, unsigned long procnum,
                     xdrproc_t inproc, char *in,
                     xdrproc_t outproc, char *out,
                     resultproc_t eachresult);
مثل callrpc()، باستثناء أن رسالة الاستدعاء يتم بثها إلى جميع شبكات البث المتصلة محليًا. في كل مرة يتلقى فيها ردًا، يستدعي هذا الروتين eachresult()، الذي يكون شكله:

eachresult(char *out, struct sockaddr_in *addr);

حيث out هو نفس out الممرر إلى clnt_broadcast()، باستثناء أن مخرجات الإجراء البعيد يتم فك تشفيرها هناك؛ addr يشير إلى عنوان الجهاز الذي أرسل النتائج. إذا أعاد eachresult() صفرًا، فإن clnt_broadcast() ينتظر المزيد من الردود؛ وإلا فإنه يعود بحالة مناسبة.
تحذير: مقابس البث محدودة الحجم بأقصى وحدة نقل للربط بالبيانات. بالنسبة للإيثرنت، هذه القيمة هي 1500 بايت.
enum clnt_stat clnt_call(CLIENT *clnt, unsigned long procnum,
                    xdrproc_t inproc, char *in,
                    xdrproc_t outproc, char *out,
                    struct timeval tout);
ماكرو يستدعي الإجراء البعيد procnum المرتبط بمقبض العميل clnt، الذي تم الحصول عليه باستخدام روتين إنشاء عميل RPC مثل clnt_create(). المعامل in هو عنوان الوسيط (الوسائط) للإجراء، و out هو العنوان الذي سيتم وضع النتائج فيه؛ inproc يستخدم لتشفير معلمات الإجراء، و outproc يستخدم لفك تشفير نتائج الإجراء؛ tout هو الوقت المسموح به لعودة النتائج.
clnt_destroy(CLIENT *clnt);
ماكرو يدمر مقبض RPC الخاص بالعميل. عادة ما يتضمن التدمير إلغاء تخصيص هياكل البيانات الخاصة، بما في ذلك clnt نفسه. استخدام clnt غير محدد بعد استدعاء clnt_destroy(). إذا قامت مكتبة RPC بفتح المقبس المرتبط، فستقوم بإغلاقه أيضًا. وإلا، يبقى المقبس مفتوحًا.
CLIENT *clnt_create(const char *host, unsigned long prog,
                    unsigned long vers, const char *proto);
روتين إنشاء عميل عام. host يحدد اسم الجهاز البعيد حيث يوجد الخادم. proto يشير إلى نوع بروتوكول النقل المستخدم. القيم المدعومة حاليًا لهذا الحقل هي "udp" و "tcp". يتم تعيين المهلات الافتراضية، ولكن يمكن تعديلها باستخدام clnt_control().
تحذير: استخدام UDP له عيوبه. نظرًا لأن رسائل RPC المستندة إلى UDP لا يمكن أن تحتوي إلا على ما يصل إلى 8 كيلوبايتات من البيانات المشفرة، لا يمكن استخدام هذا النقل للإجراءات التي تأخذ وسائط كبيرة أو تعيد نتائج ضخمة.
bool_t clnt_control(CLIENT *cl, int req, char *info);
ماكرو يستخدم لتغيير أو استرداد معلومات مختلفة حول كائن العميل. req يشير إلى نوع العملية، و info هو مؤشر إلى المعلومات. لكل من UDP و TCP، القيم المدعومة لـ req وأنواع الوسائط الخاصة بها وماذا تفعل هي:

CLSET_TIMEOUT  struct timeval // تعيين المهلة الإجمالية
CLGET_TIMEOUT  struct timeval // الحصول على المهلة الإجمالية

ملاحظة: إذا قمت بتعيين المهلة باستخدام clnt_control()، فسيتم تجاهل المعامل المهلة الممرر إلى clnt_call() في جميع الاستدعاءات المستقبلية.

CLGET_SERVER_ADDR  struct sockaddr_in
                // الحصول على عنوان الخادم

العمليات التالية صالحة لـ UDP فقط:

CLSET_RETRY_TIMEOUT  struct timeval // تعيين مهلة إعادة المحاولة
CLGET_RETRY_TIMEOUT  struct timeval // الحصول على مهلة إعادة المحاولة

مهلة إعادة المحاولة هي الوقت الذي ينتظره "UDP RPC" رد الخادم قبل إعادة إرسال الطلب.
