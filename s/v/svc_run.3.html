rpc(3) دليل وظائف المكتبة rpc(3)

اسم

rpc - روتينات مكتبة لاستدعاءات الإجراءات عن بعد

مكتبة

مكتبة C القياسية (libc، -lc)

ملخص ووصف

تسمح هذه الروتينات لبرامج C بإجراء استدعاءات إجراءات على أجهزة أخرى عبر الشبكة. أولاً، يستدعي العميل إجراءً لإرسال حزمة بيانات إلى الخادم. عند استلام الحزمة، يستدعي الخادم روتين توزيع لأداء الخدمة المطلوبة، ثم يرسل رداً. أخيراً، تعود استدعاء الإجراء إلى العميل.

للاستفادة من هذه الروتينات، قم بتضمين ملف الرأس <rpc/rpc.h>.

تستخدم النماذج الأولية أدناه الأنواع التالية:

typedef int bool_t;
typedef typeof(bool_t (XDR *, void *, ...))  *xdrproc_t;
typedef typeof(bool_t (caddr_t resp, struct sockaddr_in *raddr)
    *resultproc_t;

راجع ملفات الرأس للإعلانات الخاصة بالأنواع AUTH و CLIENT و SVCXPRT و XDR.

void auth_destroy(AUTH *auth);
ماكرو يقوم بتدمير معلومات المصادقة المرتبطة بـ auth. يتضمن التدمير عادةً إلغاء تخصيص هياكل البيانات الخاصة. استخدام auth غير محدد بعد استدعاء auth_destroy().
AUTH *authnone_create(void);
يقوم بإنشاء وإرجاع مقبض مصادقة RPC يمرر معلومات مصادقة غير قابلة للاستخدام مع كل استدعاء إجراء عن بعد. هذه هي المصادقة الافتراضية التي يستخدمها RPC.
AUTH *authunix_create(char *host, uid_t uid, gid_t gid,
                      int n, gid_t aup_gids[n]);
يقوم بإنشاء وإرجاع مقبض مصادقة RPC يحتوي على معلومات المصادقة. المعلمة host هي اسم الجهاز الذي تم إنشاء المعلومات عليه؛ uid هو معرف المستخدم؛ gid هو معرف المجموعة الحالي للمستخدم؛ n و aup_gids تشير إلى مصفوفة معدودة للمجموعات التي ينتمي إليها المستخدم. من السهل انتحال هوية مستخدم.
AUTH *authunix_create_default(void);
يستدعي authunix_create() بالمعلمات المناسبة.
int callrpc(char *host, unsigned long prognum,
            unsigned long versnum, unsigned long procnum,
            xdrproc_t inproc, const char *in,
            xdrproc_t outproc, char *out);
يستدعي الإجراء البعيد المرتبط بـ prognum و versnum و procnum على الجهاز host. المعلمة in هي عنوان وسيط (وسائط) الإجراء، و out هو العنوان الذي سيتم فيه وضع النتيجة (النتائج)؛ inproc يستخدم لتشفير معاملات الإجراء، و outproc يستخدم لفك تشفير نتائج الإجراء؛ تعيد هذه الروتين صفر إذا نجحت، أو قيمة enum clnt_stat محولة إلى عدد صحيح إذا فشلت. الروتين clnt_perrno() مفيد لترجمة حالات الفشل إلى رسائل.
تحذير: استدعاء الإجراءات البعيدة باستخدام هذا الروتين يستخدم UDP/IP كوسيلة نقل؛ راجع clntudp_create() للقيود. ليس لديك تحكم في أوقات المهلة أو المصادقة باستخدام هذا الروتين.
enum clnt_stat clnt_broadcast(unsigned long prognum,
                     unsigned long versnum, unsigned long procnum,
                     xdrproc_t inproc, char *in,
                     xdrproc_t outproc, char *out,
                     resultproc_t eachresult);
مثل callrpc()، باستثناء أن رسالة الاستدعاء يتم بثها إلى جميع شبكات البث المتصلة محليًا. في كل مرة تتلقى فيها استجابة، يقوم هذا الروتين باستدعاء eachresult()، والتي يكون شكلها:

eachresult(char *out, struct sockaddr_in *addr);

حيث out هو نفس out الذي تم تمريره إلى clnt_broadcast()، باستثناء أن خرج الإجراء البعيد يتم فك تشفيره هناك؛ addr يشير إلى عنوان الجهاز الذي أرسل الردود. إذا أعاد eachresult() صفراً، ينتظر clnt_broadcast() المزيد من الردود؛ وإلا فإنه يعود بالحالة المناسبة.
تحذير: مقابس البث محدودة الحجم بأقصى وحدة نقل للربط البياني. بالنسبة لـ ethernet، تبلغ هذه القيمة 1500 بايت.
enum clnt_stat clnt_call(CLIENT *clnt, unsigned long procnum,
                    xdrproc_t inproc, char *in,
                    xdrproc_t outproc, char *out,
                    struct timeval tout);
ماكرو يستدعي الإجراء البعيد procnum المرتبط بمقبض العميل clnt، والذي يتم الحصول عليه باستخدام روتين إنشاء عميل RPC مثل clnt_create(). المعلمة in هي عنوان وسيط (وسائط) الإجراء، و out هو العنوان الذي سيتم فيه وضع النتيجة (النتائج)؛ inproc يستخدم لتشفير معاملات الإجراء، و outproc يستخدم لفك تشفير نتائج الإجراء؛ tout هو الوقت المسموح به لعودة النتائج.
clnt_destroy(CLIENT *clnt);
ماكرو يقوم بتدمير مقبض RPC الخاص بالعميل. يتضمن التدمير عادةً إلغاء تخصيص هياكل البيانات الخاصة، بما في ذلك clnt نفسه. استخدام clnt غير محدد بعد استدعاء clnt_destroy(). إذا فتحت مكتبة RPC المقبس المرتبط، فستقوم بإغلاقه أيضًا. بخلاف ذلك، يبقى المقبس مفتوحًا.
CLIENT *clnt_create(const char *host, unsigned long prog,
                    unsigned long vers, const char *proto);
روتين إنشاء عميل عام. host يحدد اسم المضيف البعيد حيث يوجد الخادم. proto يشير إلى نوع بروتوكول النقل الذي سيتم استخدامه. القيم المدعومة حاليًا لهذا الحقل هي "udp" و "tcp". يتم تعيين أوقات مهلة افتراضية، ولكن يمكن تعديلها باستخدام clnt_control().
تحذير: استخدام UDP له عيوبه. نظرًا لأن رسائل RPC المستندة إلى UDP يمكن أن تحتوي على 8 كيلوبايتات كحد أقصى من البيانات المشفرة، فلا يمكن استخدام هذا الناقل للإجراءات التي تأخذ وسائط كبيرة أو تعيد نتائج ضخمة.
bool_t clnt_control(CLIENT *cl, int req, char *info);
ماكرو يستخدم لتغيير أو استرداد معلومات مختلفة حول كائن عميل. req يشير إلى نوع العملية، و info هو مؤشر إلى المعلومات. لكل من UDP و TCP، القيم المدعومة لـ req وأنواع وسائطها وماذا تفعل هي:

CLSET_TIMEOUT  struct timeval // تعيين المهلة الإجمالية
CLGET_TIMEOUT  struct timeval // الحصول على المهلة الإجمالية

ملاحظة: إذا قمت بتعيين المهلة باستخدام clnt_control()، فسيتم تجاهل معلمة المهلة الممررة إلى clnt_call() في جميع الاستدعاءات المستقبلية.

CLGET_SERVER_ADDR  struct sockaddr_in
                // الحصول على عنوان الخادم

العمليات التالية صالحة لـ UDP فقط:

CLSET_RETRY_TIMEOUT  struct timeval // تعيين مهلة إعادة المحاولة
CLGET_RETRY_TIMEOUT  struct timeval // الحصول على مهلة إعادة المحاولة

مهلة إعادة المحاولة هي الوقت الذي ينتظره "UDP RPC" حتى يرد الخادم قبل إعادة إرسال الطلب.
clnt_freeres(CLIENT * clnt, xdrproc_t outproc, char *out);
ماكرو يقوم بتحرير أي بيانات تم تخصيصها بواسطة نظام RPC/XDR عند فك تشفير نتائج استدعاء RPC. المعلمة out هي عنوان النتائج، و outproc هو روتين XDR الذي يصف النتائج. يعيد هذا الروتين واحدًا إذا تم تحرير النتائج بنجاح، وصفرًا بخلاف ذلك.
void clnt_geterr(CLIENT *clnt, struct rpc_err *errp);
ماكرو يقوم بنسخ بنية الخطأ من مقبض العميل إلى البنية الموجودة في العنوان errp.
void clnt_pcreateerror(const char *s);
يطبع رسالة إلى الخطأ القياسي توضح سبب عدم القدرة على إنشاء مقبض عميل RPC. يتم إضافة السلسلة s ونقطتين رأسيتين إلى بداية الرسالة. يستخدم عند فشل استدعاء clnt_create() أو clntraw_create() أو clnttcp_create() أو clntudp_create().
void clnt_perrno(enum clnt_stat stat);
يطبع رسالة إلى الخطأ القياسي تتوافق مع الحالة التي تشير إليها stat. يستخدم بعد callrpc().
clnt_perror(CLIENT *clnt, const char *s);
يطبع رسالة إلى الخطأ القياسي توضح سبب فشل استدعاء RPC؛ clnt هو المقبض المستخدم لإجراء الاستدعاء. يتم إضافة السلسلة s ونقطتين رأسيتين إلى بداية الرسالة. يستخدم بعد clnt_call().
char *clnt_spcreateerror(const char *s);
مثل clnt_pcreateerror()، ولكنه يعيد سلسلة بدلاً من طباعتها إلى الخطأ القياسي.
أخطاء: يعيد مؤشرًا إلى بيانات ثابتة يتم استبدالها في كل استدعاء.
char *clnt_sperrno(enum clnt_stat stat);
يأخذ نفس وسائط clnt_perrno()، ولكنه بدلاً من إرسال رسالة إلى الخطأ القياسي توضح سبب فشل استدعاء RPC، يعيد مؤشرًا إلى سلسلة تحتوي على الرسالة. تنتهي السلسلة بحرف سطر جديد.
يتم استخدام clnt_sperrno() بدلاً من clnt_perrno() إذا لم يكن للبرنامج خطأ قياسي (كما هو الحال غالبًا في البرامج التي تعمل كخوادم)، أو إذا كان المبرمج لا يريد إخراج الرسالة باستخدام printf(3)، أو إذا كان تنسيق الرسالة مختلفًا عن ذلك الذي تدعمه clnt_perrno(). ملاحظة: على عكس clnt_sperror() و clnt_spcreateerror()، يعيد clnt_sperrno() مؤشرًا إلى بيانات ثابتة، ولكن النتيجة لن يتم استبدالها في كل استدعاء.
char *clnt_sperror(CLIENT *rpch, const char *s);
مثل clnt_perror()، ولكنه (مثل clnt_sperrno()) يعيد سلسلة بدلاً من طباعتها إلى الخطأ القياسي.
أخطاء: يعيد مؤشرًا إلى بيانات ثابتة يتم استبدالها في كل استدعاء.
CLIENT *clntraw_create(unsigned long prognum, unsigned long versnum);
ينشئ هذا الروتين عميل RPC وهمي للبرنامج البعيد prognum، الإصدار versnum. النقل المستخدم لتمرير الرسائل إلى الخدمة هو في الواقع مخزن مؤقت داخل مساحة عناوين العملية، لذلك يجب أن يعيش خادم RPC المقابل في نفس مساحة العناوين؛ انظر svcraw_create(). يسمح هذا بمحاكاة RPC واكتساب تكاليف RPC، مثل أوقات الذهاب والإياب، دون أي تدخل من النواة. يعيد هذا الروتين NULL إذا فشل.
CLIENT *clnttcp_create(struct sockaddr_in *addr,
                unsigned long prognum, unsigned long versnum,
                int *sockp, unsigned int sendsz, unsigned int recvsz);
ينشئ هذا الروتين عميل RPC للبرنامج البعيد prognum، الإصدار versnum؛ يستخدم العميل TCP/IP كنقل. يقع البرنامج البعيد في عنوان الإنترنت *addr. إذا كان addr->sin_port صفرًا، فسيتم تعيينه إلى المنفذ الفعلي الذي يستمع عليه البرنامج البعيد (يتم استشارة خدمة portmap البعيدة لهذه المعلومات). المعامل sockp هو مقبس؛ إذا كان RPC_ANYSOCK، فسيقوم هذا الروتين بفتح مقبس جديد وتعيين sockp. نظرًا لأن RPC المستند إلى TCP يستخدم إدخال/إخراج مخزن مؤقت، يمكن للمستخدم تحديد حجم المخازن المؤقتة للإرسال والاستقبال باستخدام المعاملات sendsz و recvsz؛ القيم الصفرية تختار قيمًا افتراضية مناسبة. يعيد هذا الروتين NULL إذا فشل.
CLIENT *clntudp_create(struct sockaddr_in *addr,
                unsigned long prognum, unsigned long versnum,
                struct timeval wait, int *sockp);
ينشئ هذا الروتين عميل RPC للبرنامج البعيد prognum، الإصدار versnum؛ يستخدم العميل UDP/IP كنقل. يقع البرنامج البعيد في عنوان الإنترنت addr. إذا كان addr->sin_port صفرًا، فسيتم تعيينه إلى المنفذ الفعلي الذي يستمع عليه البرنامج البعيد (يتم استشارة خدمة portmap البعيدة لهذه المعلومات). المعامل sockp هو مقبس؛ إذا كان RPC_ANYSOCK، فسيقوم هذا الروتين بفتح مقبس جديد وتعيين sockp. يعيد نقل UDP إرسال رسالة الاستدعاء في فترات زمنية مقدارها wait حتى يتم استلام استجابة أو حتى انتهاء صلاحية الاستدعاء. الوقت الإجمالي لانتهاء صلاحية الاستدعاء محدد بواسطة clnt_call().
تحذير: نظرًا لأن رسائل RPC المستندة إلى UDP يمكن أن تحمل ما يصل إلى 8 كيلوبايت من البيانات المشفرة، فلا يمكن استخدام هذا النقل للإجراءات التي تأخذ وسائط كبيرة أو تعيد نتائج ضخمة.
CLIENT *clntudp_bufcreate(struct sockaddr_in *addr,
            unsigned long prognum, unsigned long versnum,
            struct timeval wait, int *sockp,
            unsigned int sendsize, unsigned int recosize);
ينشئ هذا الروتين عميل RPC للبرنامج البعيد prognum، على versnum؛ يستخدم العميل UDP/IP كنقل. يقع البرنامج البعيد في عنوان الإنترنت addr. إذا كان addr->sin_port صفرًا، فسيتم تعيينه إلى المنفذ الفعلي الذي يستمع عليه البرنامج البعيد (يتم استشارة خدمة portmap البعيدة لهذه المعلومات). المعامل sockp هو مقبس؛ إذا كان RPC_ANYSOCK، فسيقوم هذا الروتين بفتح مقبس جديد وتعيين sockp. يعيد نقل UDP إرسال رسالة الاستدعاء في فترات زمنية مقدارها wait حتى يتم استلام استجابة أو حتى انتهاء صلاحية الاستدعاء. الوقت الإجمالي لانتهاء صلاحية الاستدعاء محدد بواسطة clnt_call().
يسمح هذا للمستخدم بتحديد الحد الأقصى لحجم الحزمة لإرسال واستقبال رسائل RPC المستندة إلى UDP.
void get_myaddress(struct sockaddr_in *addr);
يضع عنوان IP الخاص بالجهاز في *addr، دون استشارة روتينات المكتبة التي تتعامل مع /etc/hosts. يتم دائمًا تعيين رقم المنفذ إلى htons(PMAPPORT).
struct pmaplist *pmap_getmaps(struct sockaddr_in *addr);
واجهة مستخدم لخدمة portmap، والتي تعيد قائمة بتعيينات برامج RPC الحالية إلى المنافذ على المضيف الموجود في عنوان IP *addr. يمكن أن يعيد هذا الروتين NULL. يستخدم الأمر rpcinfo -p هذا الروتين.
unsigned short pmap_getport(struct sockaddr_in *addr,
                    unsigned long prognum, unsigned long versnum,
                    unsigned int protocol);
واجهة مستخدم لخدمة portmap، والتي تعيد رقم المنفذ الذي تنتظر عليه خدمة تدعم البرنامج رقم prognum، الإصدار versnum، وتتحدث بروتوكول النقل المرتبط بـ protocol. قيمة protocol غالبًا ما تكون IPPROTO_UDP أو IPPROTO_TCP. القيمة المرجعة صفر تعني أن التعيين غير موجود أو أن نظام RPC فشل في الاتصال بخدمة portmap البعيدة. في الحالة الأخيرة، تحتوي المتغير العام rpc_createerr على حالة RPC.
enum clnt_stat pmap_rmtcall(struct sockaddr_in *addr,
                    unsigned long prognum, unsigned long versnum,
                    unsigned long procnum,
                    xdrproc_t inproc, char *in,
                    xdrproc_t outproc, char *out,
                    struct timeval tout, unsigned long *portp);
واجهة مستخدم لخدمة portmap، والتي توجه portmap على المضيف في عنوان IP *addr لإجراء استدعاء RPC نيابة عنك إلى إجراء على هذا المضيف. سيتم تعديل المعامل *portp إلى رقم منفذ البرنامج إذا نجح الإجراء. تمت مناقشة تعريفات المعاملات الأخرى في callrpc() و clnt_call(). يجب استخدام هذا الإجراء لـ "ping" وليس لشيء آخر. انظر أيضًا clnt_broadcast().
bool_t pmap_set(unsigned long prognum, unsigned long versnum,
                int protocol, unsigned short port);
واجهة مستخدم لخدمة portmap، والتي تنشئ تعيينًا بين الثلاثي [prognum,versnum,protocol] و port على خدمة portmap للجهاز. قيمة protocol غالبًا ما تكون IPPROTO_UDP أو IPPROTO_TCP. يعيد هذا الروتين واحدًا إذا نجح، وصفرًا بخلاف ذلك. يتم تلقائيًا تنفيذه بواسطة svc_register().
bool_t pmap_unset(unsigned long prognum, unsigned long versnum);
واجهة مستخدم لخدمة portmap، والتي تدمر جميع التعيينات بين الثلاثي [prognum,versnum,*] و المنافذ على خدمة portmap للجهاز. يعيد هذا الروتين واحدًا إذا نجح، وصفرًا بخلاف ذلك.
int registerrpc(unsigned long prognum, unsigned long versnum,
                unsigned long procnum, typeof(char *(char *)) *procname,
                xdrproc_t inproc, xdrproc_t outproc);
يسجل الإجراء procname مع حزمة خدمة RPC. إذا وصل طلب للبرنامج prognum، الإصدار versnum، والإجراء procnum، فسيتم استدعاء procname بمؤشر إلى وسيطته (وسائطه)؛ يجب أن يعيد procname مؤشرًا إلى نتيجته (نتائجه) الثابتة؛ يتم استخدام inproc لفك تشفير الوسائط بينما يتم استخدام outproc لتشفير النتائج. يعيد هذا الروتين صفرًا إذا نجح التسجيل، -1 بخلاف ذلك.
تحذير: يتم الوصول إلى الإجراءات البعيدة المسجلة بهذا الشكل باستخدام نقل UDP/IP
