الاسم

rpc - روتينات المكتبة لاستدعاءات الإجراءات البعيدة

المكتبة

مكتبة C القياسية (libc، -lc)

الملخص والوصف

تسمح هذه الروتينات لبرامج C بإجراء استدعاءات إجراءات على أجهزة أخرى عبر الشبكة. أولاً، يستدعي العميل إجراءً لإرسال حزمة بيانات إلى الخادم. عند استلام الحزمة، يستدعي الخادم روتين استدعاء لأداء الخدمة المطلوبة، ثم يرسل رداً. أخيرًا، يعود استدعاء الإجراء إلى العميل.

للاستفادة من هذه الروتينات، قم بتضمين ملف الرأس <rpc/rpc.h>.

تستخدم النماذج الأولية أدناه الأنواع التالية:

typedef int bool_t;
typedef typeof(bool_t (XDR *, void *, ...))  *xdrproc_t;
typedef typeof(bool_t (caddr_t resp, struct sockaddr_in *raddr)
    *resultproc_t;

راجع ملفات الرأس للإعلانات الخاصة بأنواع AUTH و CLIENT و SVCXPRT و XDR.

void auth_destroy(AUTH *auth);
ماكرو يقوم بتدمير معلومات المصادقة المرتبطة بـ auth. يتضمن التدمير عادةً إلغاء تخصيص هياكل البيانات الخاصة. استخدام auth غير محدد بعد استدعاء auth_destroy().
AUTH *authnone_create(void);
إنشاء وإرجاع مقبض مصادقة RPC يمرر معلومات مصادقة غير قابلة للاستخدام مع كل استدعاء إجراء بعيد. هذه هي المصادقة الافتراضية التي تستخدمها RPC.
AUTH *authunix_create(char *host, uid_t uid, gid_t gid,
                      int n, gid_t aup_gids[n]);
إنشاء وإرجاع مقبض مصادقة RPC يحتوي على معلومات المصادقة. المعلمة host هي اسم الجهاز الذي تم إنشاء المعلومات عليه؛ uid هو معرف المستخدم؛ gid هو معرف المجموعة الحالي للمستخدم؛ تشير n و aup_gids إلى مصفوفة محسوبة للمجموعات التي ينتمي إليها المستخدم. من السهل انتحال شخصية مستخدم.
AUTH *authunix_create_default(void);
يستدعي authunix_create() بالمعلمات المناسبة.
int callrpc(char *host, unsigned long prognum,
            unsigned long versnum, unsigned long procnum,
            xdrproc_t inproc, const char *in,
            xdrproc_t outproc, char *out);
استدعاء الإجراء البعيد المرتبط بـ prognum و versnum و procnum على الجهاز host. المعلمة in هي عنوان وسائط الإجراء (الوسيطات)، و out هو العنوان الذي ستوضع فيه النتائج؛ يستخدم inproc لتشفير معلمات الإجراء، ويستخدم outproc لفك تشفير نتائج الإجراء. تعيد هذه الروتين صفرًا إذا نجحت، أو قيمة enum clnt_stat محولة إلى عدد صحيح إذا فشلت. الروتين clnt_perrno() مفيد لترجمة حالات الفشل إلى رسائل.
تحذير: استدعاء الإجراءات البعيدة باستخدام هذا الروتين يستخدم UDP/IP كنقل؛ راجع clntudp_create() للقيود. لا يمكنك التحكم في مهلات الانتظار أو المصادقة باستخدام هذا الروتين.
enum clnt_stat clnt_broadcast(unsigned long prognum,
                     unsigned long versnum, unsigned long procnum,
                     xdrproc_t inproc, char *in,
                     xdrproc_t outproc, char *out,
                     resultproc_t eachresult);
مثل callrpc()، باستثناء أن رسالة الاستدعاء يتم بثها إلى جميع شبكات البث المتصلة محليًا. في كل مرة تتلقى فيها رداً، يستدعي هذا الروتين eachresult()، والتي يكون شكلها:

eachresult(char *out, struct sockaddr_in *addr);

حيث out هو نفس out الذي تم تمريره إلى clnt_broadcast()، باستثناء أنه يتم فك تشفير مخرجات الإجراء البعيد هناك؛ يشير addr إلى عنوان الجهاز الذي أرسل النتائج. إذا أعادت eachresult() صفرًا، تنتظر clnt_broadcast() المزيد من الردود؛ وإلا فإنها تعود بالحالة المناسبة.
تحذير: مقابس البث محدودة الحجم بحد أقصى لوحدة النقل للارتباط البياني. بالنسبة للإيثرنت، تبلغ هذه القيمة 1500 بايت.
enum clnt_stat clnt_call(CLIENT *clnt, unsigned long procnum,
                    xdrproc_t inproc, char *in,
                    xdrproc_t outproc, char *out,
                    struct timeval tout);
ماكرو يستدعي الإجراء البعيد procnum المرتبط بمقبض العميل clnt، والذي يتم الحصول عليه باستخدام روتين إنشاء عميل RPC مثل clnt_create(). المعلمة in هي عنوان وسائط الإجراء (الوسيطات)، و out هو العنوان الذي ستوضع فيه النتائج؛ يستخدم inproc لتشفير معلمات الإجراء، ويستخدم outproc لفك تشفير نتائج الإجراء؛ tout هو الوقت المسموح به لعودة النتائج.
clnt_destroy(CLIENT *clnt);
ماكرو يقوم بتدمير مقبض RPC الخاص بالعميل. يتضمن التدمير عادةً إلغاء تخصيص هياكل البيانات الخاصة، بما في ذلك clnt نفسه. استخدام clnt غير محدد بعد استدعاء clnt_destroy(). إذا فتحت مكتبة RPC المقبس المرتبط، فستقوم بإغلاقه أيضًا. وإلا، يبقى المقبس مفتوحًا.
CLIENT *clnt_create(const char *host, unsigned long prog,
                    unsigned long vers, const char *proto);
روتين إنشاء عميل عام. يحدد host اسم المضيف البعيد حيث يقع الخادم. يشير proto إلى نوع بروتوكول النقل المراد استخدامه. القيم المدعومة حاليًا لهذا الحقل هي "udp" و "tcp". يتم تعيين مهلات افتراضية، ولكن يمكن تعديلها باستخدام clnt_control().
تحذير: استخدام UDP له قصوره. نظرًا لأن رسائل RPC المستندة إلى UDP يمكن أن تحتوي على 8 كيلوبايت فقط من البيانات المشفرة، فلا يمكن استخدام هذا النقل للإجراءات التي تأخذ وسائط كبيرة أو تعيد نتائج ضخمة.
bool_t clnt_control(CLIENT *cl, int req, char *info);
ماكرو يستخدم لتغيير أو استرداد معلومات مختلفة حول كائن العميل. يشير req إلى نوع العملية، و info هو مؤشر إلى المعلومات. لكل من UDP و TCP، القيم المدعومة لـ req وأنواع وسائطها وماذا تفعل هي:

CLSET_TIMEOUT  struct timeval // تعيين المهلة الإجمالية
CLGET_TIMEOUT  struct timeval // الحصول على المهلة الإجمالية

ملاحظة: إذا قمت بتعيين المهلة باستخدام clnt_control()، فسيتم تجاهل معلمة المهلة التي تم تمريرها إلى clnt_call() في جميع الاستدعاءات المستقبلية.

CLGET_SERVER_ADDR  struct sockaddr_in
                // الحصول على عنوان الخادم

العمليات التالية صالحة لـ UDP فقط:

CLSET_RETRY_TIMEOUT  struct timeval // تعيين مهلة إعادة المحاولة
CLGET_RETRY_TIMEOUT  struct timeval // الحصول على مهلة إعادة المحاولة

مهلة إعادة المحاولة هي الوقت الذي ينتظره "UDP RPC" رد الخادم قبل إعادة إرسال الطلب.
clnt_freeres(CLIENT * clnt, xdrproc_t outproc, char *out);
ماكرو يقوم بتحرير أي بيانات تم تخصيصها بواسطة نظام RPC/XDR عند فك تشفير نتائج استدعاء RPC. المعلمة out هي عنوان النتائج، و outproc هو روتين XDR الذي يصف النتائج. يعيد هذا الروتين واحدًا إذا تم تحرير النتائج بنجاح، وصفرًا بخلاف ذلك.
void clnt_geterr(CLIENT *clnt, struct rpc_err

*errp);
وحدة ماكرو تقوم بنسخ بنية الخطأ من مقبض العميل إلى البنية الموجودة في العنوان errp.
void clnt_pcreateerror(const char *s);
طباعة رسالة إلى الخطأ القياسي توضح سبب عدم إمكانية إنشاء مقبض عميل RPC. تسبق الرسالة بالسلسلة s ونقطتين رأسيتين. تستخدم عندما تفشل استدعاء clnt_create() أو clntraw_create() أو clnttcp_create() أو clntudp_create().
void clnt_perrno(enum clnt_stat stat);
طباعة رسالة إلى الخطأ القياسي تتوافق مع الشرط الذي تشير إليه stat. تستخدم بعد callrpc().
clnt_perror(CLIENT *clnt, const char *s);
طباعة رسالة إلى الخطأ القياسي توضح سبب فشل استدعاء RPC؛ clnt هو المقبض المستخدم لإجراء الاستدعاء. تسبق الرسالة بالسلسلة s ونقطتين رأسيتين. تستخدم بعد clnt_call().
char *clnt_spcreateerror(const char *s);
مثل clnt_pcreateerror()، إلا أنها تُرجع سلسلة نصية بدلاً من طباعتها إلى الخطأ القياسي.
أخطاء: تُرجع مؤشرًا إلى بيانات ثابتة يتم الكتابة فوقها في كل استدعاء.
char *clnt_sperrno(enum clnt_stat stat);
تأخذ نفس وسائط clnt_perrno()، ولكن بدلاً من إرسال رسالة إلى الخطأ القياسي توضح سبب فشل استدعاء RPC، تُرجع مؤشرًا إلى سلسلة نصية تحتوي على الرسالة. تنتهي السلسلة بحرف سطر جديد.
يتم استخدام clnt_sperrno() بدلاً من clnt_perrno() إذا لم يكن لدى البرنامج خطأ قياسي (كما هو الحال في برنامج يعمل كخادم)، أو إذا لم يكن المبرمج يرغب في إخراج الرسالة باستخدام printf(3)، أو إذا كان تنسيق الرسالة المراد استخدامه مختلفًا عن ذلك الذي تدعمه clnt_perrno(). ملاحظة: على عكس clnt_sperror() و clnt_spcreateerror()، تُرجع clnt_sperrno() مؤشرًا إلى بيانات ثابتة، ولكن النتيجة لن يتم الكتابة فوقها في كل استدعاء.
char *clnt_sperror(CLIENT *rpch, const char *s);
مثل clnt_perror()، إلا أنها (مثل clnt_sperrno()) تُرجع سلسلة نصية بدلاً من الطباعة إلى الخطأ القياسي.
أخطاء: تُرجع مؤشرًا إلى بيانات ثابتة يتم الكتابة فوقها في كل استدعاء.
CLIENT *clntraw_create(unsigned long prognum, unsigned long versnum);
تقوم هذه الروتين بإنشاء عميل RPC وهمي للبرنامج البعيد prognum، الإصدار versnum. الوسيلة المستخدمة لتمرير الرسائل إلى الخدمة هي في الواقع مخزن مؤقت داخل مساحة عنوان العملية، لذلك يجب أن يكون خادم RPC المقابل موجودًا في نفس مساحة العنوان؛ انظر svcraw_create(). هذا يسمح بمحاكاة RPC واكتساب نفقات RPC، مثل أوقات الذهاب والإياب، دون أي تدخل من النواة. تُرجع هذه الروتين NULL إذا فشلت.
CLIENT *clnttcp_create(struct sockaddr_in *addr,
                unsigned long prognum, unsigned long versnum,
                int *sockp, unsigned int sendsz, unsigned int recvsz);
تقوم هذه الروتين بإنشاء عميل RPC للبرنامج البعيد prognum، الإصدار versnum؛ يستخدم العميل TCP/IP كوسيلة نقل. يقع البرنامج البعيد في عنوان الإنترنت *addr. إذا كان addr->sin_port صفرًا، فسيتم تعيينه إلى المنفذ الفعلي الذي يستمع إليه البرنامج البعيد (يتم استشارة خدمة portmap البعيدة لهذه المعلومات). المعلمة sockp هي مقبس؛ إذا كانت RPC_ANYSOCK، فستقوم هذه الروتين بفتح مقبس جديد وتعيين sockp. نظرًا لأن RPC المستند إلى TCP يستخدم إدخال/إخراج مخزن مؤقتًا، يمكن للمستخدم تحديد حجم المخازن المؤقتة للإرسال والاستقبال باستخدام المعلمات sendsz و recvsz؛ تختار القيم الصفرية القيم الافتراضية المناسبة. تُرجع هذه الروتين NULL إذا فشلت.
CLIENT *clntudp_create(struct sockaddr_in *addr,

                struct timeval wait, int *sockp);
تقوم هذه الروتين بإنشاء عميل RPC للبرنامج البعيد prognum، الإصدار versnum؛ يستخدم العميل UDP/IP كوسيلة نقل. يقع البرنامج البعيد في عنوان الإنترنت addr. إذا كان addr->sin_port صفرًا، فسيتم تعيينه إلى المنفذ الفعلي الذي يستمع إليه البرنامج البعيد (يتم استشارة خدمة portmap البعيدة لهذه المعلومات). المعلمة sockp هي مقبس؛ إذا كانت RPC_ANYSOCK، فستقوم هذه الروتين بفتح مقبس جديد وتعيين sockp. تقوم وسيلة نقل UDP بإعادة إرسال رسالة الاستدعاء بفواصل زمنية قدرها wait حتى يتم استلام استجابة أو حتى ينتهي وقت الاستدعاء. يتم تحديد الوقت الإجمالي لانتهاء وقت الاستدعاء بواسطة clnt_call().

struct rpc_createerr rpc_createerr;
متغير عام يتم تعيين قيمته بواسطة أي روتين إنشاء عميل RPC لا ينجح. استخدم الروتين clnt_pcreateerror() لطباعة السبب.
void svc_destroy(SVCXPRT *xprt);
ماكرو يدمر مقبض نقل خدمة RPC، xprt. يتضمن التدمير عادةً تخصيص هياكل البيانات الخاصة، بما في ذلك xprt نفسه. استخدام xprt غير محدد بعد استدعاء هذا الروتين.
fd_set svc_fdset;
متغير عام يعكس قناع بت واصف الملف للقراءة من جانب خدمة RPC؛ وهو مناسب كمعلمة لاستدعاء النظام select(2). هذا يهم فقط إذا كان منفذ الخدمة يقوم بمعالجة الأحداث غير المتزامنة الخاصة به، بدلاً من استدعاء svc_run(). هذا المتغير للقراءة فقط (لا تمرر عنوانه إلى select(2)!)، ومع ذلك قد يتغير بعد استدعاءات svc_getreqset() أو أي روتينات إنشاء.
int svc_fds;
مشابه لـ svc_fdset، ولكنه مقيد بـ 32 واصف ملف. تم إهمال هذه الواجهة بواسطة svc_fdset.
svc_freeargs(SVCXPRT *xprt, xdrproc_t inproc, char *in);
ماكرو يحرر أي بيانات تم تخصيصها بواسطة نظام RPC/XDR عند فك ترميز الوسائط لإجراء خدمة باستخدام svc_getargs(). يعيد هذا الروتين 1 إذا تم تحرير النتائج بنجاح، وصفر بخلاف ذلك.
svc_getargs(SVCXPRT *xprt, xdrproc_t inproc, char *in);
ماكرو يقوم بفك ترميز وسائط طلب RPC المرتبط بمقبض نقل خدمة RPC، xprt. المعلمة in هي العنوان الذي سيتم فيه وضع الوسائط؛ inproc هو روتين XDR المستخدم لفك ترميز الوسائط. يعيد هذا الروتين واحدًا إذا نجح فك الترميز، وصفر بخلاف ذلك.
struct sockaddr_in *svc_getcaller(SVCXPRT *xprt);
الطريقة المعتمدة للحصول على عنوان الشبكة للمتصل بإجراء مرتبط بمقبض نقل خدمة RPC، xprt.
void svc_getreqset(fd_set *rdfds);
هذا الروتين يهم فقط إذا كان منفذ الخدمة لا يستدعي svc_run()، ولكنه بدلاً من ذلك ينفذ معالجة أحداث غير متزامنة مخصصة. يتم استدعاؤه عندما يحدد استدعاء النظام select(2) أن طلب RPC قد وصل على بعض مقابس RPC؛ rdfds هو قناع بت واصف الملف الناتج للقراءة. يعيد الروتين عندما تتم خدمة جميع المقابس المرتبطة بقيمة rdfds.
void svc_getreq(int rdfds);
مشابه لـ svc_getreqset()، ولكنه مقيد بـ 32 واصف ملف. تم إهمال هذه الواجهة بواسطة svc_getreqset().
bool_t svc_register(SVCXPRT *xprt, unsigned long prognum,
                    unsigned long versnum,
                    typeof(void (struct svc_req *, SVCXPRT *)) *dispatch,
                    unsigned long protocol);
يربط prognum و versnum بإجراء إرسال الخدمة، dispatch. إذا كان protocol صفرًا، فلا يتم تسجيل الخدمة مع خدمة portmap. إذا كان protocol غير صفري، يتم إنشاء تعيين للثلاثي [prognum,versnum,protocol] إلى xprt->xp_port مع خدمة portmap المحلية (عادةً ما يكون protocol صفرًا، IPPROTO_UDP أو IPPROTO_TCP). إجراء dispatch له الشكل التالي:

dispatch(struct svc_req *request, SVCXPRT *xprt);

يعيد الروتين svc_register() واحدًا إذا نجح، وصفر بخلاف ذلك.
void svc_run(void);
هذا الروتين لا يعود أبداً. ينتظر وصول طلبات RPC، ويستدعي إجراء الخدمة المناسب باستخدام svc_getreq() عند وصول واحد. هذا الروتين عادة ما ينتظر استدعاء النظام select(2) للعودة.
bool_t svc_sendreply(SVCXPRT *xprt, xdrproc_t outproc, char *out);
يتم استدعاؤه بواسطة إجراء إرسال خدمة RPC لإرسال نتائج استدعاء إجراء بعيد. المعلمة xprt هي مقبض النقل المرتبط بالطلب؛ outproc هو روتين XDR المستخدم لتشفير النتائج؛ و out هو عنوان النتائج. يعيد هذا الروتين واحدًا إذا نجح، وصفر بخلاف ذلك.
void svc_unregister(unsigned long prognum, unsigned long versnum);
إزالة جميع تعيينات الزوج [prognum,versnum] إلى إجراءات الإرسال، والثلاثي [prognum,versnum,*] إلى رقم المنفذ.
void svcerr_auth(SVCXPRT *xprt, enum auth_stat why);
يتم استدعاؤه بواسطة إجراء إرسال خدمة يرفض تنفيذ استدعاء إجراء بعيد بسبب خطأ في المصادقة.
void svcerr_decode(SVCXPRT *xprt);
يتم استدعاؤه بواسطة إجراء إرسال خدمة لا يمكنه فك ترميز وسائطه بنجاح. انظر أيضاً svc_getargs().
void svcerr_noproc(SVCXPRT *xprt);
يتم استدعاؤه بواسطة إجراء إرسال خدمة لا ينفذ رقم الإجراء الذي يطلبه المستدعي.
void svcerr_noprog(SVCXPRT *xprt);
يتم استدعاؤه عندما لا يكون البرنامج المطلوب مسجلاً مع حزمة RPC. عادةً لا يحتاج منفذو الخدمة إلى هذا الروتين.
void svcerr_progvers(SVCXPRT *xprt, unsigned long low_vers,
                     unsigned long high_vers);
يتم استدعاؤه عندما لا يكون الإصدار المطلوب من برنامج مسجلاً مع حزمة RPC. عادةً لا يحتاج منفذو الخدمة إلى هذا الروتين.
void svcerr_systemerr(SVCXPRT *xprt);
يتم استدعاؤه بواسطة إجراء إرسال خدمة عند اكتشاف خطأ في النظام لا يغطيه أي بروتوكول معين. على سبيل المثال، إذا لم يعد بإمكان الخدمة تخصيص مساحة، فقد تستدعي هذا الروتين.
void svcerr_weakauth(SVCXPRT *xprt);
يتم استدعاؤه بواسطة إجراء إرسال خدمة يرفض تنفيذ استدعاء إجراء بعيد بسبب عدم كفاية معلمات المصادقة. يستدعي الروتين svcerr_auth(xprt, AUTH_TOOWEAK).
SVCXPRT *svcfd_create(int fd, unsigned int sendsize,
                      unsigned int recvsize);
إنشاء خدمة فوق أي واصف ملف مفتوح. عادةً، يكون واصف الملف هذا هو مقبس متصل لبروتوكول تدفق مثل TCP. تشير sendsize و recvsize إلى أحجام مخازن الإرسال والاستقبال. إذا كانت صفرًا، يتم اختيار افتراضي معقول.
SVCXPRT *svcraw_create(void);
ينشئ هذا الروتين مقبس خدمة RPC وهمي، والذي يعيد مؤشرًا إليه. المقبس هو في الواقع مخزن مؤقت داخل مساحة عنوان العملية، لذا يجب أن يكون عميل RPC المقابل في نفس مساحة العنوان؛ انظر clntraw_create(). يسمح هذا الروتين بمحاكاة RPC واكتساب نفقات RPC (مثل أوقات الذهاب والإياب)، دون أي تدخل من النواة. يعيد هذا الروتين NULL إذا فشل.
SVCXPRT *svctcp_create(int sock, unsigned int send_buf_size,
                       unsigned int recv_buf_size);
ينشئ هذا الروتين مقبس خدمة RPC يعتمد على TCP/IP، والذي يعيد مؤشرًا إليه. يرتبط المقبس بالمقبس sock، والذي قد يكون RPC_ANYSOCK، وفي هذه الحالة يتم إنشاء مقبس جديد. إذا لم يكن المقبس مرتبطًا بمنفذ TCP محلي، فإن هذا الروتين يربطه بمنفذ عشوائي. عند اكتمال العمل، يكون xprt->xp_sock هو واصف مقبس النقل، و xprt->xp_port هو رقم منفذ النقل. يعيد هذا الروتين NULL إذا فشل. نظرًا لأن RPC المعتمد على TCP يستخدم إدخال/إخراج مخزن مؤقتًا، يمكن للمستخدمين تحديد حجم المخازن المؤقتة؛ القيم الصفرية تختار افتراضيات مناسبة.
SVCXPRT *svcudp_bufcreate(int sock, unsigned int sendsize,
                          unsigned int recosize);
ينشئ هذا الروتين مقبس خدمة RPC يعتمد على UDP/IP، والذي يعيد مؤشرًا إليه. يرتبط المقبس بالمقبس sock، والذي قد يكون RPC_ANYSOCK، وفي هذه الحالة يتم إنشاء مقبس جديد. إذا لم يكن المقبس مرتبطًا بمنفذ UDP محلي، فإن هذا الروتين يربطه بمنفذ عشوائي. عند اكتمال العمل، يكون xprt->xp_sock هو واصف مقبس النقل، و xprt->xp_port هو رقم منفذ النقل. يعيد هذا الروتين NULL إذا فشل.
يسمح هذا للمستخدم بتحديد الحد الأقصى لحجم الحزمة لإرسال واستقبال رسائل RPC المعتمدة على UDP.
SVCXPRT *svcudp_create(int sock);
هذا الاستدعاء مكافئ لـ svcudp_bufcreate(sock,SZ,SZ) لبعض الحجم الافتراضي SZ.
bool_t xdr_accepted_reply(XDR *xdrs, struct accepted_reply *ar);
يستخدم لتشفير رسائل رد RPC. هذا الروتين مفيد للمستخدمين الذين يرغبون في إنشاء رسائل بأسلوب RPC دون استخدام حزمة RPC.
bool_t xdr_authunix_parms(XDR *xdrs, struct authunix_parms *aupp);
يستخدم لوصف بيانات اعتماد UNIX. هذا الروتين مفيد للمستخدمين الذين يرغبون في إنشاء بيانات الاعتماد هذه دون استخدام حزمة مصادقة RPC.
void xdr_callhdr(XDR *xdrs, struct rpc_msg *chdr);
يستخدم لوصف رسائل رأس استدعاء RPC. هذا الروتين مفيد للمستخدمين الذين يرغبون في إنشاء رسائل بأسلوب RPC دون استخدام حزمة RPC.
