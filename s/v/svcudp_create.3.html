الاسم

rpc - إجراءات مكتبة لاستدعاءات الإجراءات عن بعد

المكتبة

مكتبة C القياسية (libc، -lc)

الملخص والوصف

تسمح هذه الإجراءات لبرامج C بإجراء استدعاءات إجراءات على أجهزة أخرى عبر الشبكة. أولاً، يستدعي العميل إجراءً لإرسال حزمة بيانات إلى الخادم. عند استلام الحزمة، يستدعي الخادم إجراءً تفويضًا لأداء الخدمة المطلوبة، ثم يرسل ردًا. أخيرًا، تعود استدعاء الإجراء إلى العميل.

للاستفادة من هذه الإجراءات، قم بتضمين ملف الرأس <rpc/rpc.h>.

تستخدم النماذج الأولية أدناه الأنواع التالية:

typedef int bool_t;
typedef typeof(bool_t (XDR *, void *, ...))  *xdrproc_t;
typedef typeof(bool_t (caddr_t resp, struct sockaddr_in *raddr)
    *resultproc_t;

راجع ملفات الرأس لتعريفات أنواع AUTH و CLIENT و SVCXPRT و XDR.

void auth_destroy(AUTH *auth);
ماكرو يقوم بتدمير معلومات المصادقة المرتبطة بـ auth. يتضمن التدمير عادةً إلغاء تخصيص هياكل البيانات الخاصة. استخدام auth غير معرف بعد استدعاء auth_destroy().
AUTH *authnone_create(void);
إنشاء وإرجاع مقبض مصادقة RPC يمرر معلومات مصادقة غير قابلة للاستخدام مع كل استدعاء إجراء عن بعد. هذه هي المصادقة الافتراضية التي يستخدمها RPC.
AUTH *authunix_create(char *host, uid_t uid, gid_t gid,
                      int n, gid_t aup_gids[n]);
إنشاء وإرجاع مقبض مصادقة RPC يحتوي على معلومات المصادقة. المعلمة host هي اسم الجهاز الذي تم إنشاء المعلومات عليه؛ uid هو معرف المستخدم؛ gid هو معرف المجموعة الحالي للمستخدم؛ تشير n و aup_gids إلى مصفوفة محسوبة للمجموعات التي ينتمي إليها المستخدم. من السهل انتحال شخصية مستخدم.
AUTH *authunix_create_default(void);
يستدعي authunix_create() بالمعلمات المناسبة.
int callrpc(char *host, unsigned long prognum,
            unsigned long versnum, unsigned long procnum,
            xdrproc_t inproc, const char *in,
            xdrproc_t outproc, char *out);
استدعاء الإجراء البعيد المرتبط بـ prognum و versnum و procnum على الجهاز host. المعلمة in هي عنوان وسيط (وسائط) الإجراء، و out هي العنوان الذي سيتم وضع النتائج فيه؛ يستخدم inproc لتشفير معلمات الإجراء، ويستخدم outproc لفك تشفير نتائج الإجراء؛ تعيد هذه الروتين صفرًا إذا نجحت، أو قيمة enum clnt_stat محوّلة إلى عدد صحيح إذا فشلت. الروتين clnt_perrno() مفيد لترجمة حالات الفشل إلى رسائل.
تحذير: استدعاء الإجراءات البعيدة بهذه الروتين تستخدم UDP/IP كبروتوكول نقل؛ راجع clntudp_create() للقيود. ليس لديك تحكم في المهلات أو المصادقة باستخدام هذه الروتين.
enum clnt_stat clnt_broadcast(unsigned long prognum,
                     unsigned long versnum, unsigned long procnum,
                     xdrproc_t inproc, char *in,
                     xdrproc_t outproc, char *out,
                     resultproc_t eachresult);
مثل callrpc()، باستثناء أن رسالة الاستدعاء يتم بثها إلى جميع شبكات البث المتصلة محليًا. في كل مرة تتلقى فيها ردًا، تستدعي هذه الروتين eachresult()، والتي يكون شكلها:

eachresult(char *out, struct sockaddr_in *addr);

حيث out هو نفس out الذي تم تمريره إلى clnt_broadcast()، باستثناء أنه يتم فك تشفير ناتج الإجراء البعيد هناك؛ addr يشير إلى عنوان الجهاز الذي أرسل النتائج. إذا أعادت eachresult() صفرًا، تنتظر clnt_broadcast() المزيد من الردود؛ بخلاف ذلك، تعود بحالة مناسبة.
تحذير: مقابس البث محدودة الحجم بأقصى وحدة نقل للارتباط البيانات. بالنسبة لشبكة الإيثرنت، تبلغ هذه القيمة 1500 بايت.
enum clnt_stat clnt_call(CLIENT *clnt, unsigned long procnum,
                    xdrproc_t inproc, char *in,
                    xdrproc_t outproc, char *out,
                    struct timeval tout);
ماكرو يستدعي الإجراء البعيد procnum المرتبط بمقبض العميل clnt، والذي يتم الحصول عليه باستخدام روتين إنشاء عميل RPC مثل clnt_create(). المعلمة in هي عنوان وسيط (وسائط) الإجراء، و out هي العنوان الذي سيتم وضع النتائج فيه؛ يستخدم inproc لتشفير معلمات الإجراء، ويستخدم outproc لفك تشفير نتائج الإجراء؛ tout هو الوقت المسموح به لعودة النتائج.
clnt_destroy(CLIENT *clnt);
ماكرو يقوم بتدمير مقبض RPC الخاص بالعميل. يتضمن التدمير عادةً إلغاء تخصيص هياكل البيانات الخاصة، بما في ذلك clnt نفسه. استخدام clnt غير معرف بعد استدعاء clnt_destroy(). إذا فتحت مكتبة RPC المقبس المرتبط، فستقوم بإغلاقه أيضًا. بخلاف ذلك، يظل المقبس مفتوحًا.
CLIENT *clnt_create(const char *host, unsigned long prog,
                    unsigned long vers, const char *proto);
روتين إنشاء عميل عام. يحدد host اسم المضيف البعيد حيث يوجد الخادم. يشير proto إلى نوع بروتوكول النقل المراد استخدامه. القيم المدعومة حاليًا لهذا الحقل هي "udp" و "tcp". يتم تعيين مهلات افتراضية، ولكن يمكن تعديلها باستخدام clnt_control().
تحذير: استخدام UDP له عيوبه. نظرًا لأن رسائل RPC المستندة إلى UDP يمكن أن تحمل 8 كيلوبايت فقط من البيانات المشفرة، فلا يمكن استخدام هذا البروتوكول للإجراءات التي تأخذ وسائط كبيرة أو تعيد نتائج ضخمة.
bool_t clnt_control(CLIENT *cl, int req, char *info);
ماكرو يستخدم لتغيير أو استرداد معلومات مختلفة حول كائن العميل. يشير req إلى نوع العملية، و info هو مؤشر إلى المعلومات. لكل من UDP و TCP، القيم المدعومة لـ req وأنواع وسائطها وما تفعله هي:

CLSET_TIMEOUT  struct timeval // تعيين المهلة الإجمالية
CLGET_TIMEOUT  struct timeval // الحصول على المهلة الإجمالية

ملاحظة: إذا قمت بتعيين المهلة باستخدام clnt_control()، فسيتم تجاهل معلمة المهلة التي تم تمريرها إلى clnt_call() في جميع الاستدعاءات المستقبلية.

CLGET_SERVER_ADDR  struct sockaddr_in
                // الحصول على عنوان الخادم

العمليات التالية صالحة لـ UDP فقط:

CLSET_RETRY_TIMEOUT  struct timeval // تعيين مهلة إعادة المحاولة
CLGET_RETRY_TIMEOUT  struct timeval // الحصول على مهلة إعادة المحاولة

مهلة إعادة المحاولة هي الوقت الذي تنتظره "UDP RPC" رد الخادم قبل إعادة إرسال الطلب.
clnt_freeres(CLIENT * clnt, xdrproc_t outproc, char *out);
ماكرو يقوم بتحرير أي بيانات تم تخصيصها بواسطة نظام RPC/XDR عند فك تشفير نتائج استدعاء RPC. المعلمة out هي عنوان النتائج، و outproc هي روتين XDR الذي يصف النتائج. تعيد هذه الروتين واحدًا إذا تم تحرير النتائج بنجاح، وصفرًا بخلاف ذلك.

ماكرو يدمر مقبض نقل خدمة RPC، xprt. يتضمن التدمير عادةً تخصيص هياكل بيانات خاصة، بما في ذلك xprt نفسه. استخدام xprt غير محدد بعد استدعاء هذا الروتين.
fd_set svc_fdset؛
متغير عام يعكس قناع بت واصف الملف للقراءة لجانب خدمة RPC؛ إنه مناسب كمعلمة لاستدعاء النظام select(2). هذا يهم فقط إذا كان منفذ الخدمة يقوم بمعالجة الأحداث غير المتزامنة الخاصة به، بدلاً من استدعاء svc_run(). هذا المتغير للقراءة فقط (لا تمرر عنوانه إلى select(2)!)، ولكنه قد يتغير بعد استدعاء svc_getreqset() أو أي روتينات إنشاء.

svc_getargs(SVCXPRT *xprt، xdrproc_t inproc، char *in)؛
ماكرو يقوم بفك تشفير وسائط طلب RPC المرتبط بمقبض نقل خدمة RPC، xprt. المعلمة in هي العنوان الذي سيتم فيه وضع الوسائط؛ inproc هو روتين XDR المستخدم لفك تشفير الوسائط. هذا الروتين يعيد واحدًا إذا نجح فك التشفير، وصفرًا بخلاف ذلك.

void svc_getreqset(fd_set *rdfds)؛
هذا الروتين يهم فقط إذا كان منفذ الخدمة لا يستدعي svc_run()، ولكنه يقوم بدلاً من ذلك بتنفيذ معالجة أحداث غير متزامنة مخصصة. يتم استدعاؤه عندما يكون استدعاء النظام select(2) قد حدد أن طلب RPC قد وصل على أحد مقابسيس RPC؛ rdfds هو قناع بت واصف الملف الناتج للقراءة. يعود الروتين عندما تتم خدمة جميع المقابسيس المرتبطة بقيمة rdfds.
void svc_getreq(int rdfds)؛
مشابه لـ svc_getreqset()، ولكنه يقتصر على 32 واصف ملف. تم إهمال هذه الواجهة بواسطة svc_getreqset().
bool_t svc_register(SVCXPRT *xprt، unsigned long prognum،
                    unsigned long versnum،
                    typeof(void (struct svc_req *، SVCXPRT *)) *dispatch،
                    unsigned long protocol)؛
يربط prognum و versnum بإجراء خدمة التوزيع، dispatch. إذا كان protocol صفرًا، فلا يتم تسجيل الخدمة مع خدمة portmap. إذا كان protocol غير صفري، فيتم إنشاء تعيين للثلاثي [prognum،versnum،protocol] إلى xprt->xp_port مع خدمة portmap المحلية (عادةً ما يكون protocol صفرًا، IPPROTO_UDP أو IPPROTO_TCP). إجراء dispatch له الشكل التالي:

dispatch(struct svc_req *request, SVCXPRT *xprt);

يعيد الروتين svc_register() واحدًا إذا نجح، وصفرًا بخلاف ذلك.
void svc_run(void)؛
هذا الروتين لا يعود أبدًا. ينتظر وصول طلبات RPC، ويستدعي إجراء الخدمة المناسب باستخدام svc_getreq() عند وصول واحد. هذا الإجراء ينتظر عادةً استدعاء النظام select(2) للعودة.
bool_t svc_sendreply(SVCXPRT *xprt، xdrproc_t outproc، char *out)؛
يتم استدعاؤه بواسطة روتين توزيع خدمة RPC لإرسال نتائج مكالمة إجراء بعيدة. المعلمة xprt هي مقبض النقل المرتبط بالطلب؛ outproc هو روتين XDR المستخدم لتشفير النتائج؛ و out هو عنوان النتائج. يعيد هذا الروتين واحدًا إذا نجح، وصفرًا بخلاف ذلك.

إزالة جميع تعيينات المزدوج [prognum،versnum] إلى روتينات التوزيع، وتعيين الثلاثي [prognum،versnum،*] إلى رقم المنفذ.

يتم استدعاؤه بواسطة روتين توزيع الخدمة الذي يرفض أداء مكالمة إجراء بعيدة بسبب خطأ مصادقة.

يتم استدعاؤه بواسطة روتين توزيع الخدمة الذي لا ينفذ رقم الإجراء الذي يطلبه المتصل.

يتم استدعاؤه عندما لا يكون البرنامج المطلوب مسجلاً مع حزمة RPC. عادةً ما لا يحتاج منفذو الخدمة إلى هذا الروتين.

هذا الروتين ينشئ نقل خدمة RPC تجريبي، والذي يعيد مؤشرًا إليه. النقل هو في الواقع مخزن مؤقت داخل مساحة عنوان العملية، لذلك يجب أن يعيش عميل RPC المقابل في نفس مساحة العنوان؛ انظر clntraw_create(). يسمح هذا الروتين بمحاكاة RPC والحصول على نفقات RPC (مثل أوقات الرحلة ذهابًا وإيابًا)، دون أي تدخل من النواة. يعيد هذا الروتين NULL إذا فشل.
SVCXPRT *svctcp_create(int sock، unsigned int send_buf_size،
                       unsigned int recv_buf_size)؛
هذا الروتين ينشئ نقل خدمة RPC قائم على TCP/IP، والذي يعيد مؤشرًا إليه. يرتبط النقل بالمقبس sock، والذي يمكن أن يكون RPC_ANYSOCK، وفي هذه الحالة يتم إنشاء مقبس جديد. إذا لم يكن المقبس مرتبطًا بمنفذ TCP محلي، فسيقوم هذا الروتين بربطه بمنفذ عشوائي. عند اكتماله، xprt->xp_sock هو واصف مقبس النقل، و xprt->xp_port هو رقم منفذ النقل. يعيد هذا الروتين NULL إذا فشل. نظرًا لأن RPC المستند إلى TCP يستخدم إدخال/إخراج مخزن مؤقتًا، يمكن للمستخدمين تحديد حجم المخازن المؤقتة؛ القيم الصفرية تختار قيمًا افتراضية مناسبة.

bool_t xdr_opaque_auth(XDR *xdrs, struct opaque_auth *ap);

bool_t xdr_pmap(XDR *xdrs, struct pmap *regs);
تُستخدم لوصف معاملات إجراءات portmap المختلفة، خارجيًا. هذا الإجراء مفيد للمستخدمين الذين يرغبون في إنشاء هذه المعاملات دون استخدام واجهة pmap.
bool_t xdr_pmaplist(XDR *xdrs, struct pmaplist **rp);
تُستخدم لوصف قائمة تعيينات المنافذ، خارجيًا. هذا الإجراء مفيد للمستخدمين الذين يرغبون في إنشاء هذه المعاملات دون استخدام واجهة pmap.
bool_t xdr_rejected_reply(XDR *xdrs, struct rejected_reply *rr);
تُستخدم لوصف رسائل الرد RPC. هذا الإجراء مفيد للمستخدمين الذين يرغبون في إنشاء رسائل بأسلوب RPC دون استخدام حزمة RPC.
bool_t xdr_replymsg(XDR *xdrs, struct rpc_msg *rmsg);
تُستخدم لوصف رسائل الرد RPC. هذا الإجراء مفيد للمستخدمين الذين يرغبون في إنشاء رسائل بأسلوب RPC دون استخدام حزمة RPC.
void xprt_register(SVCXPRT *xprt);
بعد إنشاء معالجات نقل خدمة RPC، يجب عليها تسجيل نفسها مع حزمة خدمة RPC. هذا الإجراء يعدل المتغير العام svc_fds. عادةً لا يحتاج منفذو الخدمة لهذا الإجراء.
void xprt_unregister(SVCXPRT *xprt);
قبل تدمير معالج نقل خدمة RPC، يجب عليه إلغاء تسجيل نفسه مع حزمة خدمة RPC. هذا الإجراء يعدل المتغير العام svc_fds. عادةً لا يحتاج منفذو الخدمة لهذا الإجراء.
