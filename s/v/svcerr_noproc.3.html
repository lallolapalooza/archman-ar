الاسم

rpc - روتينات مكتبة لاستدعاءات الإجراءات عن بعد

المكتبة

مكتبة C القياسية (libc، -lc)

الملخص والوصف

تسمح هذه الروتينات لبرامج C بإجراء استدعاءات إجراءات على أجهزة أخرى عبر الشبكة. أولاً، يستدعي العميل إجراءً لإرسال حزمة بيانات إلى الخادم. عند استلام الحزمة، يستدعي الخادم روتين توزيع لأداء الخدمة المطلوبة، ثم يرسل ردًا. أخيرًا، يعود استدعاء الإجراء إلى العميل.

للاستفادة من هذه الروتينات، قم بتضمين ملف الترويسة <rpc/rpc.h>.

تستخدم النماذج الأولية أدناه الأنواع التالية:

typedef int bool_t;
typedef typeof(bool_t (XDR *, void *, ...))  *xdrproc_t;
typedef typeof(bool_t (caddr_t resp, struct sockaddr_in *raddr)
    *resultproc_t;

راجع ملفات الترويسة لتعريفات أنواع AUTH، CLIENT، SVCXPRT، و XDR.

AUTH *authunix_create(char *host, uid_t uid, gid_t gid,

إنشاء وإرجاع مقبض مصادقة RPC يحتوي على معلومات المصادقة. المعلمة host هي اسم الجهاز الذي تم إنشاء المعلومات عليه؛ uid هو معرف المستخدم؛ gid هو معرف المجموعة الحالي للمستخدم؛ n و aup_gids تشير إلى مصفوفة معدودة للمجموعات التي ينتمي إليها المستخدم. من السهل انتحال شخصية مستخدم.
AUTH *authunix_create_default(void);
يستدعي authunix_create() بالمعلمات المناسبة.
int callrpc(char *host, unsigned long prognum,
            unsigned long versnum, unsigned long procnum,
            xdrproc_t inproc, const char *in,
            xdrproc_t outproc, char *out);
استدعاء الإجراء البعيد المرتبط بـ prognum، versnum، و procnum على الجهاز host. المعلمة in هي عنوان الوسائط ( الوسيطة) للإجراء، و out هو العنوان الذي سيتم وضع النتائج فيه؛ inproc يستخدم لتشفير وسائط الإجراء، و outproc يستخدم لفك تشفير نتائج الإجراء؛ تعود هذه الروتين بصفر إذا نجحت، أو بقيمة enum clnt_stat محولة إلى عدد صحيح إذا فشلت. الروتين clnt_perrno() مفيد لترجمة حالات الفشل إلى رسائل.
تحذير: استدعاء الإجراءات البعيدة باستخدام هذا الروتين يستخدم UDP/IP كبروتوكول نقل؛ راجع clntudp_create() للقيود. ليس لديك تحكم في المهلات أو المصادقة باستخدام هذا الروتين.

مثل callrpc()، باستثناء أن رسالة الاستدعاء يتم بثها إلى جميع شبكات البث المتصلة محليًا. في كل مرة يتلقى فيها استجابة، يستدعي هذا الروتين eachresult()، والذي يكون شكله:

eachresult(char *out, struct sockaddr_in *addr);

حيث out هو نفس out الذي تم تمريره إلى clnt_broadcast()، باستثناء أن مخرجات الإجراء البعيد يتم فك تشفيرها هناك؛ addr يشير إلى عنوان الجهاز الذي أرسل الردود. إذا أعاد eachresult() صفرًا، ينتظر clnt_broadcast() لمزيد من الردود؛ وإلا فإنه يعود بالحالة المناسبة.

وحدة ماكرو تدمر مقبض RPC الخاص بالعميل. يتضمن التدمير عادةً تحرير هياكل البيانات الخاصة، بما في ذلك clnt نفسه. استخدام clnt غير محدد بعد استدعاء clnt_destroy(). إذا فتحت مكتبة RPC المقبس المرتبط، فستغلقه أيضًا. وإلا، يبقى المقبس مفتوحًا.
CLIENT *clnt_create(const char *host, unsigned long prog,

روتين إنشاء عميل عام. host يحدد اسم الجهاز البعيد حيث يقع الخادم. proto يشير إلى نوع بروتوكول النقل المستخدم. القيم المدعومة حاليًا لهذا الحقل هي "udp" و "tcp". يتم تعيين مهلات افتراضية، ولكن يمكن تعديلها باستخدام clnt_control().
تحذير: استخدام UDP له عيوبه. نظرًا لأن رسائل RPC المستندة إلى UDP يمكن أن تحمل فقط ما يصل إلى 8 كيلوبايت من البيانات المشفرة، لا يمكن استخدام هذا البروتوكول للإجراءات التي تأخذ وسائط كبيرة أو تعيد نتائج ضخمة.

وحدة ماكرو تستخدم لتغيير أو استرداد معلومات مختلفة حول كائن العميل. req تشير إلى نوع العملية، و info هو مؤشر إلى المعلومات. لكل من UDP و TCP، القيم المدعومة لـ req وأنواع وسائطها وماذا تفعل هي:

CLSET_TIMEOUT  struct timeval // تعيين المهلة الإجمالية
CLGET_TIMEOUT  struct timeval // الحصول على المهلة الإجمالية

CLGET_SERVER_ADDR  struct sockaddr_in
                // الحصول على عنوان الخادم

تأخذ نفس وسائط clnt_perrno()، ولكن بدلاً من إرسال رسالة إلى الخطأ القياسي توضح سبب فشل استدعاء RPC، ترجع مؤشرًا إلى سلسلة نصية تحتوي على الرسالة. تنتهي السلسلة النصية بسطر جديد.
يتم استخدام clnt_sperrno() بدلاً من clnt_perrno() إذا لم يكن لدى البرنامج خطأ قياسي (كما قد لا يكون لدى البرنامج الذي يعمل كخادم)، أو إذا لم يرغب المبرمج في إخراج الرسالة باستخدام printf(3)، أو إذا كان سيتم استخدام تنسيق رسالة مختلف عن ذلك الذي تدعمه clnt_perrno(). ملاحظة: على عكس clnt_sperror() و clnt_spcreateerror()، فإن clnt_sperrno() ترجع مؤشرًا إلى بيانات ثابتة، ولكن النتيجة لن يتم الكتابة فوقها في كل استدعاء.

CLIENT *clntraw_create(unsigned long prognum, unsigned long versnum);
ينشئ هذا الروتين عميل RPC وهمي للبرنامج البعيد prognum، الإصدار versnum. وسيلة النقل المستخدمة لتمرير الرسائل إلى الخدمة هي في الواقع مخزن مؤقت داخل مساحة عنوان العملية، لذلك يجب أن يعيش خادم RPC المقابل في نفس مساحة العنوان؛ انظر svcraw_create(). يسمح هذا بمحاكاة RPC واكتساب نفقات RPC، مثل أوقات الذهاب والإياب، دون أي تدخل من النواة. يرجع هذا الروتين NULL إذا فشل.
CLIENT *clnttcp_create(struct sockaddr_in *addr,
                unsigned long prognum, unsigned long versnum,
                int *sockp, unsigned int sendsz, unsigned int recvsz);
ينشئ هذا الروتين عميل RPC للبرنامج البعيد prognum، الإصدار versnum؛ يستخدم العميل TCP/IP كوسيلة نقل. يقع البرنامج البعيد في عنوان الإنترنت *addr. إذا كان addr->sin_port صفرًا، فسيتم تعيينه إلى المنفذ الفعلي الذي يستمع عليه البرنامج البعيد (يتم استشارة خدمة portmap البعيدة لهذه المعلومات). الوسيط sockp هو مقبس؛ إذا كان RPC_ANYSOCK، فسيقوم هذا الروتين بفتح مقبس جديد وتعيين sockp. نظرًا لأن RPC المستند إلى TCP يستخدم إدخال/إخراج مخزن مؤقتًا، يمكن للمستخدم تحديد حجم المخازن المؤقتة للإرسال والاستقبال باستخدام الوسائط sendsz و recvsz؛ القيم الصفرية تختار القيم الافتراضية المناسبة. يرجع هذا الروتين NULL إذا فشل.

ماكرو يقوم بتدمير مقبض نقل خدمة RPC، xprt. يشمل التدمير عادةً تخصيص هياكل البيانات الخاصة، بما في ذلك xprt نفسه. استخدام xprt غير معرف بعد استدعاء هذا الروتين.
fd_set svc_fdset;
متغير عام يعكس قناع بت واصف الملف للقراءة من جانب خدمة RPC؛ وهو مناسب كمعامل لاستدعاء النظام select(2). هذا يهم فقط إذا كان منفذ الخدمة يقوم بمعالجة الأحداث غير المتزامنة الخاصة به، بدلاً من استدعاء svc_run(). هذا المتغير للقراءة فقط (لا تمرر عنوانه إلى select(2)!)، ولكنه قد يتغير بعد استدعاءات svc_getreqset() أو أي روتينات إنشاء.
int svc_fds;
مشابه لـ svc_fdset، ولكنه محدود بـ 32 واصف ملف. تم إهمال هذه الواجهة بواسطة svc_fdset.
svc_freeargs(SVCXPRT *xprt, xdrproc_t inproc, char *in);
ماكرو يقوم بتحرير أي بيانات تم تخصيصها بواسطة نظام RPC/XDR عند فك تشفير الوسائط لإجراء الخدمة باستخدام svc_getargs(). يعود هذا الروتين بالقيمة 1 إذا تم تحرير النتائج بنجاح، والصفر بخلاف ذلك.
svc_getargs(SVCXPRT *xprt, xdrproc_t inproc, char *in);
ماكرو يقوم بفك تشفير وسائط طلب RPC المرتبط بمقبض نقل خدمة RPC، xprt. المعامل in هو العنوان حيث سيتم وضع الوسائط؛ inproc هو روتين XDR المستخدم لفك تشفير الوسائط. يعود هذا الروتين بالقيمة واحد إذا نجح فك التشفير، والصفر بخلاف ذلك.
struct sockaddr_in *svc_getcaller(SVCXPRT *xprt);
الطريقة المعتمدة للحصول على عنوان الشبكة للمتصل بإجراء مرتبط بمقبض نقل خدمة RPC، xprt.
void svc_getreqset(fd_set *rdfds);
هذا الروتين يهم فقط إذا كان منفذ الخدمة لا يستدعي svc_run()، ولكنه بدلاً من ذلك ينفذ معالجة أحداث مخصصة غير متزامنة. يتم استدعاؤه عندما يحدد استدعاء النظام select(2) أن طلب RPC قد وصل على بعض مقابس RPC؛ rdfds هو قناع بت واصف الملف الناتج للقراءة. يعود الروتين عندما تتم خدمة جميع المقابس المرتبطة بقيمة rdfds.
void svc_getreq(int rdfds);
مشابه لـ svc_getreqset()، ولكنه محدود بـ 32 واصف ملف. تم إهمال هذه الواجهة بواسطة svc_getreqset().
bool_t svc_register(SVCXPRT *xprt, unsigned long prognum,
                    unsigned long versnum,
                    typeof(void (struct svc_req *, SVCXPRT *)) *dispatch,
                    unsigned long protocol);
يربط prognum و versnum بإجراء توزيع الخدمة، dispatch. إذا كان protocol صفرًا، فلن يتم تسجيل الخدمة مع خدمة portmap. إذا كان protocol غير صفري، فسيتم إنشاء تعيين للثلاثي [prognum,versnum,protocol] إلى xprt->xp_port مع خدمة portmap المحلية (عادةً ما يكون protocol صفرًا، IPPROTO_UDP أو IPPROTO_TCP). يكون لإجراء dispatch الشكل التالي:

dispatch(struct svc_req *request, SVCXPRT *xprt);

يعود الروتين svc_register() بالقيمة واحد إذا نجح، والصفر بخلاف ذلك.
void svc_run(void);
هذا الروتين لا يعود أبداً. ينتظر وصول طلبات RPC، ويستدعي إجراء الخدمة المناسب باستخدام svc_getreq() عند وصول واحد. عادة ما ينتظر هذا الروتين استدعاء النظام select(2) للعودة.
bool_t svc_sendreply(SVCXPRT *xprt, xdrproc_t outproc, char *out);
يتم استدعاؤه بواسطة إجراء توزيع خدمة RPC لإرسال نتائج استدعاء الإجراء البعيد. المعامل xprt هو مقبض النقل المرتبط بالطلب؛ outproc هو روتين XDR المستخدم لتشفير النتائج؛ و out هو عنوان النتائج. يعود هذا الروتين بالقيمة واحد إذا نجح، والصفر بخلاف ذلك.
void svc_unregister(unsigned long prognum, unsigned long versnum);
إزالة جميع تعيينات الزوج [prognum,versnum] إلى إجراءات التوزيع، وتعيينات الثلاثي [prognum,versnum,*] إلى رقم المنفذ.
void svcerr_auth(SVCXPRT *xprt, enum auth_stat why);
يتم استدعاؤه بواسطة إجراء توزيع الخدمة الذي يرفض أداء استدعاء إجراء بعيد بسبب خطأ مصادقة.
void svcerr_decode(SVCXPRT *xprt);
يتم استدعاؤه بواسطة إجراء توزيع الخدمة الذي لا يمكنه فك تشفير وسائطه بنجاح. انظر أيضًا svc_getargs().
void svcerr_noproc(SVCXPRT *xprt);
يتم استدعاؤه بواسطة إجراء توزيع الخدمة الذي لا ينفذ رقم الإجراء الذي يطلبه المتصل.
void svcerr_noprog(SVCXPRT *xprt);
يتم استدعاؤه عندما لا يكون البرنامج المطلوب مسجلاً مع حزمة RPC. عادةً لا يحتاج منفذو الخدمة إلى هذا الروتين.
void svcerr_progvers(SVCXPRT *xprt, unsigned long low_vers,
                     unsigned long high_vers);
يتم استدعاؤه عندما لا يكون الإصدار المطلوب من برنامج مسجلاً مع حزمة RPC. عادةً لا يحتاج منفذو الخدمة إلى هذا الروتين.
void svcerr_systemerr(SVCXPRT *xprt);
يتم استدعاؤه بواسطة إجراء توزيع الخدمة عند اكتشاف خطأ نظام لا يغطيه بروتوكول معين. على سبيل المثال، إذا لم يعد بإمكان الخدمة تخصيص مساحة تخزين، فقد تستدعي هذا الروتين.
void svcerr_weakauth(SVCXPRT *xprt);
يتم استدعاؤه بواسطة إجراء توزيع الخدمة الذي يرفض أداء استدعاء إجراء بعيد بسبب معلمات مصادقة غير كافية. يستدعي الروتين svcerr_auth(xprt, AUTH_TOOWEAK).
SVCXPRT *svcfd_create(int fd, unsigned int sendsize,
                      unsigned int recvsize);
إنشاء خدمة فوق أي واصف ملف مفتوح. عادةً، يكون واصف الملف هذا مقبسًا متصلاً لبروتوكول تدفق مثل TCP. تشير sendsize و recvsize إلى أحجام المخازن المؤقتة للإرسال والاستقبال. إذا كانت صفرًا، يتم اختيار قيمة افتراضية معقولة.
SVCXPRT *svcraw_create(void);
يقوم هذا الروتين بإنشاء خدمة RPC وهمية، ويعيد مؤشرًا إليها. النقل هو في الواقع مخزن مؤقت داخل مساحة عنوان العملية، لذلك يجب أن يكون عميل RPC المقابل موجودًا في نفس مساحة العنوان؛ انظر clntraw_create(). يسمح هذا الروتين بمحاكاة RPC واكتساب تكاليف RPC (مثل أوقات الذهاب والإياب)، دون أي تدخل من النواة. يعود هذا الروتين بـ NULL إذا فشل.
SVCXPRT *svctcp_create(int sock, unsigned int send_buf_size,
                       unsigned int recv_buf_size);
يقوم هذا الروتين بإنشاء نقل خدمة RPC قائم على TCP/IP، ويعيد مؤشرًا إليه. يرتبط النقل بالمقبس sock، والذي قد يكون RPC_ANYSOCK، وفي هذه الحالة يتم إنشاء مقبس جديد. إذا لم يكن المقبس مرتبطًا بمنفذ TCP محلي، فسيقوم هذا الروتين بربطه بمنفذ عشوائي. عند الانتهاء، يكون xprt->xp_sock هو واصف مقبس النقل، و xprt->xp_port هو رقم منفذ النقل. يعود هذا الروتين بـ NULL إذا فشل. نظرًا لأن RPC المستند إلى TCP يستخدم I/O المخزن مؤقتًا، يمكن للمستخدمين تحديد حجم المخازن المؤقتة؛ القيم الصفرية تختار قيمًا افتراضية مناسبة.
SVCXPRT *svcudp_bufcreate(int sock, unsigned int sendsize,
                          unsigned int recosize);
يقوم هذا الروتين بإنشاء نقل خدمة RPC قائم على UDP/IP، ويعيد مؤشرًا إليه. يرتبط النقل بالمقبس sock، والذي قد يكون RPC_ANYSOCK، وفي هذه الحالة يتم إنشاء مقبس جديد. إذا لم يكن المقبس مرتبطًا بمنفذ UDP محلي، فسيقوم هذا الروتين بربطه بمنفذ عشوائي. عند الانتهاء، يكون xprt->xp_sock هو واصف مقبس النقل، و xprt->xp_port هو رقم منفذ النقل. يعود هذا الروتين بـ NULL إذا فشل.
يسمح هذا للمستخدم بتحديد الحد الأقصى لحجم الحزمة لإرسال واستقبال رسائل RPC المستندة إلى UDP.
SVCXPRT *svcudp_create(int sock);
هذا الاستدعاء يعادل svcudp_bufcreate(sock,SZ,SZ) لبعض الحجم الافتراضي SZ.
bool_t xdr_accepted_reply(XDR *xdrs, struct accepted_reply *ar);
يستخدم لتشفير رسائل رد RPC. هذا الروتين مفيد للمستخدمين الذين يرغبون في إنشاء رسائل على طراز RPC دون استخدام حزمة RPC.
bool_t xdr_authunix_parms(XDR *xdrs, struct authunix_parms *aupp);
يستخدم لوصف بيانات اعتماد UNIX. هذا الروتين مفيد للمستخدمين الذين يرغبون في إنشاء بيانات الاعتماد هذه دون استخدام حزمة مصادقة RPC.
void xdr_callhdr(XDR *xdrs, struct rpc_msg *chdr);
يستخدم لوصف رسائل رأس استدعاء RPC. هذا الروتين مفيد للمستخدمين الذين يرغبون في إنشاء رسائل على طراز RPC دون استخدام حزمة RPC.
