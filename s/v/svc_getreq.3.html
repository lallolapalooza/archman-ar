errp);
ماكرو يقوم بنسخ بنية الخطأ من مقبض العميل إلى البنية الموجودة في العنوان errp.
void clnt_pcreateerror(const char *s);
طباعة رسالة إلى الخطأ القياسي تشير إلى سبب عدم إنشاء مقبض عميل RPC. يتم إضافة السلسلة النصية s ونقطتين إلى بداية الرسالة. تستخدم عند فشل استدعاء clnt_create(), clntraw_create(), clnttcp_create(), أو clntudp_create().
void clnt_perrno(enum clnt_stat stat);
طباعة رسالة إلى الخطأ القياسي تتوافق مع الحالة التي تشير إليها stat. تستخدم بعد callrpc().
clnt_perror(CLIENT *clnt, const char *s);
طباعة رسالة إلى الخطأ القياسي تشير إلى سبب فشل استدعاء RPC؛ clnt هو المقبض المستخدم لإجراء الاستدعاء. يتم إضافة السلسلة النصية s ونقطتين إلى بداية الرسالة. تستخدم بعد clnt_call().
char *clnt_spcreateerror(const char *s);
مثل clnt_pcreateerror()، باستثناء أنها تعيد سلسلة نصية بدلاً من طباعتها إلى الخطأ القياسي.
الأخطاء: تعيد مؤشرًا إلى بيانات ثابتة يتم الكتابة فوقها في كل استدعاء.
char *clnt_sperrno(enum clnt_stat stat);
تأخذ نفس وسيطات clnt_perrno()، ولكن بدلاً من إرسال رسالة إلى الخطأ القياسي تشير إلى سبب فشل استدعاء RPC، تعيد مؤشرًا إلى سلسلة نصية تحتوي على الرسالة. تنتهي السلسلة بـ NEWLINE.
يتم استخدام clnt_sperrno() بدلاً من clnt_perrno() إذا لم يكن لدى البرنامج خطأ قياسي (كما هو الحال غالبًا مع برنامج يعمل كخادم)، أو إذا كان المبرمج لا يريد إخراج الرسالة باستخدام printf(3)، أو إذا كان سيتم استخدام تنسيق رسالة مختلف عن ذلك الذي تدعمه clnt_perrno(). ملاحظة: على عكس clnt_sperror() و clnt_spcreateerror()، تعيد clnt_sperrno() مؤشرًا إلى بيانات ثابتة، ولكن النتيجة لن يتم الكتابة فوقها في كل استدعاء.
char *clnt_sperror(CLIENT *rpch, const char *s);
مثل clnt_perror()، باستثناء أنها (مثل clnt_sperrno()) تعيد سلسلة نصية بدلاً من الطباعة إلى الخطأ القياسي.
الأخطاء: تعيد مؤشرًا إلى بيانات ثابتة يتم الكتابة فوقها في كل استدعاء.
CLIENT *clntraw_create(unsigned long prognum, unsigned long versnum);
تقوم هذه الروتين بإنشاء عميل RPC وهمي للبرنامج البعيد prognum، الإصدار versnum. النقل المستخدم لتمرير الرسائل إلى الخدمة هو في الواقع مخزن مؤقت ضمن مساحة عنوان العملية، لذلك يجب أن يعيش خادم RPC المقابل في نفس مساحة العنوان؛ انظر svcraw_create(). هذا يسمح بمحاكاة RPC واكتساب نفقات RPC، مثل أوقات الذهاب والإياب، دون أي تدخل من النواة. تعيد هذه الروتين NULL إذا فشلت.
CLIENT *clnttcp_create(struct sockaddr_in *addr,
                unsigned long prognum, unsigned long versnum,
                int *sockp, unsigned int sendsz, unsigned int recvsz);
تقوم هذه الروتين بإنشاء عميل RPC للبرنامج البعيد prognum، الإصدار versnum؛ يستخدم العميل TCP/IP كنقل. يقع البرنامج البعيد في عنوان الإنترنت *addr. إذا كان addr->sin_port صفرًا، فسيتم تعيينه إلى المنفذ الفعلي الذي يستمع إليه البرنامج البعيد (يتم استشارة خدمة portmap البعيدة للحصول على هذه المعلومات). المعلمة sockp هي مقبس؛ إذا كانت RPC_ANYSOCK، فستفتح هذه الروتين مقبسًا جديدًا وتعيّن sockp. نظرًا لأن RPC المستند إلى TCP يستخدم إدخال/إخراج مخزن مؤقت، يمكن للمستخدم تحديد حجم مخازن الإرسال والاستقبال باستخدام المعلمات sendsz و recvsz؛ القيم الصفرية تختار الافتراضيات المناسبة. تعيد هذه الروتين NULL إذا فشلت.
CLIENT *clntudp_create(struct sockaddr_in *addr,
                unsigned long prognum, unsigned long versnum,
                struct timeval wait, int *sockp);
تقوم هذه الروتين بإنشاء عميل RPC للبرنامج البعيد prognum، الإصدار versnum؛ يستخدم العميل UDP/IP كنقل. يقع البرنامج البعيد في عنوان الإنترنت addr. إذا كان addr->sin_port صفرًا، فسيتم تعيينه إلى المنفذ الفعلي الذي يستمع إليه البرنامج البعيد (يتم استشارة خدمة portmap البعيدة للحصول على هذه المعلومات). المعلمة sockp هي مقبس؛ إذا كانت RPC_ANYSOCK، فستفتح هذه الروتين مقبسًا جديدًا وتعيّن sockp. يعيد نقل UDP إرسال رسالة الاستدعاء في فترات زمنية مدتها wait حتى يتم استلام استجابة أو حتى ينتهي وقت الاستدعاء. يتم تحديد الوقت الإجمالي لانتهاء مهلة الاستدعاء بواسطة clnt_call().
تحذير: نظرًا لأن رسائل RPC المستندة إلى UDP يمكن أن تحمل ما يصل إلى 8 كيلوبايت فقط من البيانات المشفرة، لا يمكن استخدام هذا النقل للإجراءات التي تتلقى وسيطات كبيرة أو تعيد نتائج ضخمة.
CLIENT *clntudp_bufcreate(struct sockaddr_in *addr,
            unsigned long prognum, unsigned long versnum,
            struct timeval wait, int *sockp,
            unsigned int sendsize, unsigned int recosize);
تقوم هذه الروتين بإنشاء عميل RPC للبرنامج البعيد prognum، على versnum؛ يستخدم العميل UDP/IP كنقل. يقع البرنامج البعيد في عنوان الإنترنت addr. إذا كان addr->sin_port صفرًا، فسيتم تعيينه إلى المنفذ الفعلي الذي يستمع إليه البرنامج البعيد (يتم استشارة خدمة portmap البعيدة للحصول على هذه المعلومات). المعلمة sockp هي مقبس؛ إذا كانت RPC_ANYSOCK، فستفتح هذه الروتين مقبسًا جديدًا وتعيّن sockp. يعيد نقل UDP إرسال رسالة الاستدعاء في فترات زمنية مدتها wait حتى يتم استلام استجابة أو حتى ينتهي وقت الاستدعاء. يتم تحديد الوقت الإجمالي لانتهاء مهلة الاستدعاء بواسطة clnt_call().
هذا يسمح للمستخدم بتحديد الحد الأقصى لحجم الحزمة لإرسال واستقبال رسائل RPC المستندة إلى UDP.
void get_myaddress(struct sockaddr_in *addr);
يضع عنوان IP للجهاز في *addr، دون استشارة روتينات المكتبة التي تتعامل مع /etc/hosts. يتم دائمًا تعيين رقم المنفذ إلى htons(PMAPPORT).
struct pmaplist *pmap_getmaps(struct sockaddr_in *addr);
واجهة مستخدم لخدمة portmap، والتي تعيد قائمة خرائط برامج RPC الحالية إلى المنافذ على المضيف الموجود في عنوان IP *addr. يمكن لهذه الروتين أن تعيد NULL. الأمر rpcinfo -p يستخدم هذه الروتين.
unsigned short pmap_getport(struct sockaddr_in *addr,
                    unsigned long prognum, unsigned long versnum,
                    unsigned int protocol);
واجهة مستخدم لخدمة portmap، والتي تعيد رقم المنفذ الذي تنتظر عليه خدمة تدعم رقم البرنامج prognum، الإصدار versnum، وتتحدث بروتوكول النقل المرتبط بـ protocol. من المرجح أن تكون قيمة protocol هي IPPROTO_UDP أو IPPROTO_TCP. القيمة المرجعة صفر تعني أن الخريطة غير موجودة أو أن نظام RPC فشل في الاتصال بخدمة portmap البعيدة. في الحالة الأخيرة، تحتوي المتغير العام rpc_createerr على حالة RPC.
enum clnt_stat pmap_rmtcall(struct sockaddr_in *addr,
                    unsigned long prognum, unsigned long versnum,
                    unsigned long procnum,
                    xdrproc_t inproc, char *in,
                    xdrproc_t outproc, char *out,
                    struct timeval tout, unsigned long *portp);
واجهة مستخدم لخدمة portmap، والتي تطلب من portmap على المضيف في عنوان IP *addr إجراء استدعاء RPC نيابة عنك لإجراء على هذا المضيف. سيتم تعديل المعلمة *portp إلى رقم منفذ البرنامج إذا نجح الإجراء. تم مناقشة تعريفات المعلمات الأخرى في callrpc() و clnt_call(). يجب استخدام هذا الإجراء لـ "ping" وليس لأي شيء آخر. انظر أيضًا clnt_broadcast().
bool_t pmap_set(unsigned long prognum, unsigned long versnum,
                int protocol, unsigned short port);
واجهة مستخدم لخدمة portmap، والتي تنشئ تعيينًا بين الثلاثي [prognum,versnum,protocol] و port على خدمة portmap للجهاز. من المرجح أن تكون قيمة protocol هي IPPROTO_UDP أو IPPROTO_TCP. تعيد هذه الروتين واحدًا إذا نجحت، وصفرًا بخلاف ذلك. يتم ذلك تلقائيًا بواسطة svc_register().
bool_t pmap_unset(unsigned long prognum, unsigned long versnum);
واجهة مستخدم لخدمة portmap، والتي تدمر جميع التعيينات بين الثلاثي [prognum,versnum,*] والمنافذ على خدمة portmap للجهاز. تعيد هذه الروتين واحدًا إذا نجحت، وصفرًا بخلاف ذلك.
int registerrpc(unsigned long prognum, unsigned long versnum,
                unsigned long procnum, typeof(char *(char *)) *procname,
                xdrproc_t inproc, xdrproc_t outproc);
تسجيل الإجراء procname مع حزمة خدمة RPC. إذا وصل طلب للبرنامج prognum، والإصدار versnum، والإجراء procnum، فسيتم استدعاء procname بمؤشر إلى وسيطاته (وسيطاته)؛ يجب أن يعيد procname مؤشرًا إلى نتائجه (نتائجه) الثابتة؛ يتم استخدام inproc لفك تشفير الوسيطات بينما يتم استخدام outproc لتشفير النتائج. تعيد هذه الروتين صفرًا إذا نجح التسجيل، و -1 بخلاف ذلك.
تحذير: يتم الوصول إلى الإجراءات البعيدة المسجلة بهذا الشكل باستخدام نقل UDP/IP

NAME

svc_getreq, svc_getreqset, xdr_callmsg, xdr_rejected_reply, xdr_replymsg, xdr_opaque_auth, xdr_pmap, xdr_pmaplist, xdr_accepted_reply, xdr_authunix_parms, xdr_callhdr, xprt_register, xprt_unregister — routines for the RPC service package

LIBRARY

Network Information Service Library (libnsl, -lnsl)

SYNOPSIS AND DESCRIPTION

void svc_getreq(int xprt);
Used by the RPC service package to dispatch incoming RPC calls. In the original Sun Microsystems implementation, this was called by the RPC dispatcher, as xprt was a pointer to a transport handle. The current implementation uses svc_getreqset(). Service implementors usually do not need this routine.
void svc_getreqset(struct svc_fdset *set);
Used by the RPC service package to dispatch incoming RPC calls. This routine iterates through the file descriptor set passed in set, and for each active descriptor, it calls the appropriate transport routines to read and process the RPC request. Service implementors usually do not need this routine.
bool_t xdr_accepted_reply(XDR *xdrs, struct accepted_reply *ar);
Used for describing RPC accepted reply messages. This routine is useful for users who wish to generate RPC-style messages without using the RPC package.
bool_t xdr_authunix_parms(XDR *xdrs, struct authunix_parms *aup);
Used for describing UNIX-style authentication parameters. This routine is useful for users who wish to generate RPC-style messages without using the RPC package.
bool_t xdr_callhdr(XDR *xdrs, struct rpc_msg *cmsg);
Used for describing RPC call messages. This routine is useful for users who wish to generate RPC-style messages without using the RPC package.
bool_t xdr_callmsg(XDR *xdrs, struct rpc_msg *cmsg);
Used for describing RPC call messages. This routine is useful for users who wish to generate RPC-style messages without using the RPC package.
bool_t xdr_opaque_auth(XDR *xdrs, struct opaque_auth *ap);
Used for describing RPC authentication information messages. This routine is useful for users who wish to generate RPC-style messages without using the RPC package.
bool_t xdr_pmap(XDR *xdrs, struct pmap *regs);
Used for describing parameters to various portmap procedures, externally. This routine is useful for users who wish to generate these parameters without using the pmap interface.
bool_t xdr_pmaplist(XDR *xdrs, struct pmaplist **rp);
Used for describing a list of port mappings, externally. This routine is useful for users who wish to generate these parameters without using the pmap interface.
bool_t xdr_rejected_reply(XDR *xdrs, struct rejected_reply *rr);
Used for describing RPC reply messages. This routine is useful for users who wish to generate RPC-style messages without using the RPC package.
bool_t xdr_replymsg(XDR *xdrs, struct rpc_msg *rmsg);
Used for describing RPC reply messages. This routine is useful for users who wish to generate RPC style messages without using the RPC package.
void xprt_register(SVCXPRT *xprt);
After RPC service transport handles are created, they should register themselves with the RPC service package. This routine modifies the global variable svc_fds. Service implementors usually do not need this routine.
void xprt_unregister(SVCXPRT *xprt);
Before an RPC service transport handle is destroyed, it should unregister itself with the RPC service package. This routine modifies the global variable svc_fds. Service implementors usually do not need this routine.
