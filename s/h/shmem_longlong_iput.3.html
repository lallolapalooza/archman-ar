الاسم

shmem_longlong_iput - يضع طويلة مزدوجة بشكل غير متزامن.

الملخص

#include <mpi.h>
int shmem_longlong_iput(long long *target, const long long *source, int nlong, int pe);

الوصف

تقوم الدالة shmem_longlong_iput بنسخ nlong من عناصر long long من source إلى target في pe المحدد. تعتبر هذه العملية غير متزامنة.

المعلمات

  • target: مؤشر إلى الموقع المستهدف في العملية البعيدة.
  • source: مؤشر إلى المصدر البيانات.
  • nlong: عدد عناصر long long المراد نسخها.
  • pe: رقم العملية البعيدة.

قيم الإرجاع

لا توجد قيمة إرجاع محددة.

ملاحظات

  • يجب أن تكون العمليات المستهدفة في العملية البعيدة متاحة عبر ذاكرة مشتركة.
  • يمكن أن تؤدي استدعاءات shmem_longlong_iput المتعددة إلى عمليات إرسال متعددة.

أمثلة

#include <mpi.h>
#include <stdio.h>

int main() {
    long long source_data = 12345;
    long long target_data;
    int my_pe, npes;

    MPI_Init(NULL, NULL);
    MPI_Comm_rank(MPI_COMM_WORLD, &my_pe);
    MPI_Comm_size(MPI_COMM_WORLD, &npes);

    if (npes < 2) {
        fprintf(stderr, "This example requires at least 2 processes.\n");
        MPI_Finalize();
        return 1;
    }

    int target_pe = (my_pe == 0) ? 1 : 0;

    // تهيئة البيانات المستهدفة (يمكن أن تكون جزءًا من ذاكرة مشتركة)
    if (my_pe == target_pe) {
        target_data = 0;
    }

    MPI_Barrier(MPI_COMM_WORLD);

    if (my_pe == 0) {
        shmem_longlong_iput(&target_data, &source_data, 1, target_pe);
        printf("PE %d sent %lld to PE %d\n", my_pe, source_data, target_pe);
    }

    MPI_Barrier(MPI_COMM_WORLD);

    if (my_pe == target_pe) {
        printf("PE %d received %lld\n", my_pe, target_data);
    }

    MPI_Finalize();
    return 0;
}
