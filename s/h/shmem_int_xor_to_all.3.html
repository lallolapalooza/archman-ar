shmem_int_xor_to_all(3)

الحزمة: openmpi

التراخيص: BSD-3-Clause AND LicenseRef-MPICH

SHMEM_INT_XOR_TO_ALL(3) Open MPI SHMEM_INT_XOR_TO_ALL(3)

shmem_comp4_xor_to_all(3), shmem_int_xor_to_all(3), shmem_int4_xor_to_all(3), shmem_int8_xor_to_all(3), shmem_long_xor_to_all(3), shmem_longlong_xor_to_all(3), shmem_short_xor_to_all(3) - تنفذ عملية XOR (الاستبعاد المنطقي) على المصفوفات المتماثلة عبر مجموعة PEs النشطة.

الملخص

C أو C++:

#include <mpp/shmem.h>
void shmem_int_xor_to_all(int *target, const int *source,
  int nreduce, int PE_start, int logPE_stride, int PE_size,
  int *pWrk, long *pSync)
void shmem_long_xor_to_all(long *target, const long *source,
  int nreduce, int PE_start, int logPE_stride, int PE_size,
  long *pWrk, long *pSync)
void shmem_longlong_xor_to_all(long long *target,
  const long long *source, int nreduce, int PE_start, int logPE_stride,
  int PE_size, long long *pWrk, long *pSync)
void shmem_short_xor_to_all(short *target, const short *source,
  int nreduce, int PE_start, int logPE_stride, int PE_size,
  short *pWrk, long *pSync)

Fortran:

INCLUDE "mpp/shmem.fh"
INTEGER pSync(SHMEM_REDUCE_SYNC_SIZE)
INTEGER nreduce, PE_start, logPE_stride, PE_size
CALL SHMEM_COMP4_XOR_TO_ALL(target, source, nreduce,
& PE_start, logPE_stride, PE_size, pWrk, pSync)
CALL SHMEM_INT4_XOR_TO_ALL(target, source, nreduce,
& PE_start, logPE_stride, PE_size, pWrk, pSync)
CALL SHMEM_INT8_XOR_TO_ALL(target, source, nreduce,
& PE_start, logPE_stride, PE_size, pWrk, pSync)

الوصف

تقوم روتينات الذاكرة المشتركة (SHMEM) التخفيضية بحساب تخفيض واحد أو أكثر عبر المصفوفات المتماثلة على العديد من وحدات المعالجة الافتراضية (PEs). يقوم التخفيض بإجراء عملية ثنائية تجميعية عبر مجموعة من القيم. لقائمة بروتينات تخفيض SHMEM الأخرى، راجع intro_shmem(3).

كما هو الحال مع جميع روتينات SHMEM الجماعية، تفترض كل من هذه الروتينات أن وحدات المعالجة الموجودة في المجموعة النشطة فقط هي التي تستدعي الروتين. إذا قامت وحدة معالجة ليست في المجموعة النشطة باستدعاء روتين SHMEM جماعي، فستكون النتيجة غير محددة.

يحدد الوسيط nreduce عدد عمليات التخفيض المنفصلة التي سيتم إجراؤها. توفر مصفوفة المصدر على جميع وحدات المعالجة في المجموعة النشطة عنصرًا واحدًا لكل تخفيض. يتم وضع نتائج عمليات التخفيض في مصفوفة الهدف على جميع وحدات المعالجة في المجموعة النشطة. يتم تعريف المجموعة النشطة بواسطة الثلاثي PE_start، logPE_stride، PE_size.

يمكن أن تكون مصفوفتا المصدر والهدف نفس المصفوفة، ولكنهما لا يمكن أن تكونا مصفوفتين متداخلتين.

الوسائط هي كما يلي:

target
مصفوفة متماثلة بطول nreduce عنصر لاستقبال نتائج عمليات التخفيض. يختلف نوع بيانات target حسب إصدار روتين التخفيض الذي يتم استدعاؤه واللغة المستخدمة. عند الاستدعاء من C/C++، ارجع إلى قسم الملخص للحصول على معلومات حول أنواع البيانات. عند الاستدعاء من Fortran، تكون أنواع بيانات الهدف كما يلي:
shmem_comp8_xor_to_all:** مركب، بحجم عنصر يساوي قيمتين حقيقيتين بحجم 8
بايت
shmem_comp4_xor_to_all:** مركب، بحجم عنصر يساوي قيمتين حقيقيتين بحجم 4
بايت

shmem_int8_xor_to_all:** عدد صحيح، بحجم عنصر 8 بايت

shmem_int4_xor_to_all:** عدد صحيح، بحجم عنصر 4 بايت

shmem_real8_xor_to_all:** حقيقي، بحجم عنصر 8 بايت

shmem_real4_xor_to_all:** حقيقي، بحجم عنصر 4 بايت

source
مصفوفة متماثلة، بطول nreduce عنصر، تحتوي على عنصر واحد لكل عملية تخفيض منفصلة. يجب أن يكون وسيط المصدر من نفس نوع بيانات الهدف.
nreduce
عدد العناصر في مصفوفات الهدف والمصدر. يجب أن يكون nreduce من نوع عدد صحيح. إذا كنت تستخدم Fortran، فيجب أن تكون قيمة عدد صحيح افتراضية.
PE_start
أدنى رقم PE افتراضي للمجموعة النشطة من وحدات المعالجة. يجب أن يكون PE_start من نوع عدد صحيح. إذا كنت تستخدم Fortran، فيجب أن تكون قيمة عدد صحيح افتراضية.
logPE_stride
لوغاريتم (أساس 2) للخطوة بين أرقام وحدات المعالجة الافتراضية المتتالية في المجموعة النشطة. يجب أن يكون logPE_stride من نوع عدد صحيح. إذا كنت تستخدم Fortran، فيجب أن تكون قيمة عدد صحيح افتراضية.
PE_size
عدد وحدات المعالجة في المجموعة النشطة. يجب أن يكون PE_size من نوع عدد صحيح. إذا كنت تستخدم Fortran، فيجب أن تكون قيمة عدد صحيح افتراضية.
pWrk
مصفوفة عمل متماثلة. يجب أن يكون وسيط pWrk من نفس نوع بيانات الهدف. في C/C++، تحتوي هذه على max(nreduce/2 + 1, _SHMEM_REDUCE_MIN_WRKDATA_SIZE) عنصر. في Fortran، تحتوي هذه على max(nreduce/2 + 1, SHMEM_REDUCE_MIN_WRKDATA_SIZE) عنصر.
pSync
مصفوفة عمل متماثلة. في C/C++، يكون pSync من نوع long وبحجم _SHMEM_REDUCE_SYNC_SIZE. في Fortran، يكون pSync من نوع عدد صحيح وبحجم SHMEM_REDUCE_SYNC_SIZE. إذا كنت تستخدم Fortran، فيجب أن تكون قيمة عدد صحيح افتراضية. يجب تهيئة كل عنصر في هذه المصفوفة بالقيمة _SHMEM_SYNC_VALUE (في C/C++) أو SHMEM_SYNC_VALUE (في Fortran) قبل أن تدخل أي من وحدات المعالجة في المجموعة النشطة روتين التخفيض.

يجب أن تكون قيم الوسائط nreduce، PE_start، logPE_stride، و PE_size متساوية على جميع وحدات المعالجة في المجموعة النشطة. يجب تمرير نفس مصفوفتي الهدف والمصدر، ونفس مصفوفات العمل pWrk و pSync، إلى جميع وحدات المعالجة في المجموعة النشطة.

قبل أن تقوم أي وحدة معالجة باستدعاء روتين تخفيض، يجب عليك التأكد من وجود الشروط التالية (غالبًا ما يلزم التزامن باستخدام حاجز أو طريقة أخرى للتأكد من ذلك): مصفوفات pWrk و pSync على جميع وحدات المعالجة في المجموعة النشطة لا تزال قيد الاستخدام من استدعاء سابق لروتين SHMEM جماعي. مصفوفة الهدف على جميع وحدات المعالجة في المجموعة النشطة جاهزة لاستقبال نتائج التخفيض.

عند العودة من روتين تخفيض، تكون الأمور التالية صحيحة بالنسبة لوحدة المعالجة المحلية: تم تحديث مصفوفة الهدف. تم استعادة القيم الموجودة في مصفوفة pSync إلى القيم الأصلية.

ملاحظات

يتم تعريف المصطلحات جماعي، متماثل، ومتوافق مع ذاكرة التخزين المؤقت في intro_shmem(3). تقوم جميع روتينات تخفيض SHMEM بإعادة تعيين القيم في pSync قبل أن تعود، لذلك يكفي تهيئة مخزن pSync معين مرة واحدة فقط عند استخدامه لأول مرة.

يجب عليك التأكد من أن مصفوفة pSync لا يتم تحديثها على أي وحدة معالجة في المجموعة النشطة أثناء مشاركة أي من وحدات المعالجة في معالجة روتين تخفيض SHMEM. كن حذرًا من المواقف التالية: إذا تم تهيئة مصفوفة pSync وقت التشغيل، فيلزم نوع من التزامن للتأكد من أن جميع وحدات المعالجة في مجموعة العمل قد أنهت تهيئة pSync قبل أن تدخل أي منها روتين SHMEM يتم استدعاؤه باستخدام مصفوفة المزامنة pSync. يمكن إعادة استخدام مصفوفة pSync أو pWrk في استدعاء روتين تخفيض لاحق فقط إذا لم تكن أي من وحدات المعالجة في المجموعة النشطة لا تزال تعالج روتين تخفيض سابق استخدم نفس مصفوفات pSync أو pWrk. بشكل عام، لا يمكن التأكد من ذلك إلا عن طريق إجراء نوع من التزامن. ومع ذلك، في الحالة الخاصة لروتينات التخفيض التي يتم استدعاؤها بنفس المجموعة النشطة، يمكنك تخصيص مصفوفتي pSync و pWrk والتبديل بينهما في الاستدعاءات المتتالية.

أمثلة

مثال 1: يقوم مثال Fortran هذا بتهيئة مصفوفة pSync بشكل ثابت ويحسب XOR الحصري للمتغير FOO عبر جميع وحدات المعالجة الزوجية.

INCLUDE "mpp/shmem.fh"
INTEGER PSYNC(SHMEM_REDUCE_SYNC_SIZE)
DATA PSYNC /SHMEM_REDUCE_SYNC_SIZE*SHMEM_SYNC_VALUE/
PARAMETER (NR=1)
REAL FOO, FOOXOR, PWRK(MAX(NR/2+1,SHMEM_REDUCE_MIN_WRKDATA_SIZE))
COMMON /COM/ FOO, FOOXOR, PWRK
INTRINSIC MY_PE
IF ( MOD(MY_PE(),2) .EQ. 0) THEN
  CALL SHMEM_REAL8_XOR_TO_ALL(FOOXOR, FOO, NR, 0, 1, N$PES/2,
  & PWRK, PSYNC)
  PRINT *, 'Result on PE ', MY_PE(), ' is ', FOOXOR
ENDIF
