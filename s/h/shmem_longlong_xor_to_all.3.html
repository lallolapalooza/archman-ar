shmem_longlong_xor_to_all(3)

الحزمة: openmpi

التراخيص: BSD-3-Clause AND LicenseRef-MPICH

SHMEM_LONGLONG_XOR_TO_ALL(3) Open MPI SHMEM_LONGLONG_XOR_TO_ALL(3)

shmem_comp4_xor_to_all(3), shmem_int_xor_to_all(3), shmem_int4_xor_to_all(3), shmem_int8_xor_to_all(3), shmem_long_xor_to_all(3), shmem_longlong_xor_to_all(3), shmem_short_xor_to_all(3) - ينفذ عملية XOR على مستوى البت على المصفوفات المتماثلة عبر المجموعة النشطة من PEs.

الملخص

C أو C++:

#include <mpp/shmem.h>
void shmem_int_xor_to_all(int *target, const int *source,
  int nreduce, int PE_start, int logPE_stride, int PE_size,
  int *pWrk, long *pSync)
void shmem_long_xor_to_all(long *target, const long *source,
  int nreduce, int PE_start, int logPE_stride, int PE_size,
  long *pWrk, long *pSync)
void shmem_longlong_xor_to_all(long long *target,
  const long long *source, int nreduce, int PE_start, int logPE_stride,
  int PE_size, long long *pWrk, long *pSync)
void shmem_short_xor_to_all(short *target, const short *source,
  int nreduce, int PE_start, int logPE_stride, int PE_size,
  short *pWrk, long *pSync)

Fortran:

INCLUDE "mpp/shmem.fh"
INTEGER pSync(SHMEM_REDUCE_SYNC_SIZE)
INTEGER nreduce, PE_start, logPE_stride, PE_size
CALL SHMEM_COMP4_XOR_TO_ALL(target, source, nreduce,
& PE_start, logPE_stride, PE_size, pWrk, pSync)
CALL SHMEM_INT4_XOR_TO_ALL(target, source, nreduce,
& PE_start, logPE_stride, PE_size, pWrk, pSync)
CALL SHMEM_INT8_XOR_TO_ALL(target, source, nreduce,
& PE_start, logPE_stride, PE_size, pWrk, pSync)

الوصف

تحسب روتينات الاختزال للذاكرة المشتركة (SHMEM) عملية اختزال واحدة أو أكثر عبر المصفوفات المتماثلة على عدة PEs افتراضية. ينفذ الاختزال عملية ثنائية ترابطية عبر مجموعة من القيم. للحصول على قائمة بروتينات الاختزال الأخرى في SHMEM، راجع intro_shmem(3).

كما هو الحال مع جميع روتينات SHMEM الجماعية، تفترض كل من هذه الروتينات أن PEs الموجودة في المجموعة النشطة فقط هي التي تستدعي الروتين. إذا استدعى PE غير موجود في المجموعة النشطة روتين SHMEM جماعي، فستنتج سلوكيات غير معرّفة.

تحدد الوسيطة nreduce عدد عمليات الاختزال المنفصلة التي يجب تنفيذها. توفر مصفوفة المصدر على جميع PEs في المجموعة النشطة عنصرًا واحدًا لكل عملية اختزال. توضع نتائج عمليات الاختزال في مصفوفة الهدف على جميع PEs في المجموعة النشطة. تُعرَّف المجموعة النشطة بواسطة ثلاثية PE_start وlogPE_stride وPE_size.

قد تكون مصفوفتا المصدر والهدف نفس المصفوفة، لكن لا يجوز أن تكونا مصفوفتين متداخلتين.

الوسائط هي كما يلي:

target
مصفوفة متماثلة بطول nreduce عنصرًا لاستقبال نتائج عمليات الاختزال. يختلف نوع بيانات target باختلاف إصدار روتين الاختزال الذي يتم استدعاؤه واللغة المستخدمة. عند الاستدعاء من C/C++، راجع قسم الملخص للحصول على معلومات نوع البيانات. عند الاستدعاء من Fortran، تكون أنواع بيانات target كما يلي:
shmem_comp8_xor_to_all:** مركب، بحجم عنصر يساوي قيمتين حقيقيتين بحجم 8
بايت
shmem_comp4_xor_to_all:** مركب، بحجم عنصر يساوي قيمتين حقيقيتين بحجم 4
بايت

shmem_int8_xor_to_all:** عدد صحيح، بحجم عنصر 8 بايت

shmem_int4_xor_to_all:** عدد صحيح، بحجم عنصر 4 بايت

shmem_real8_xor_to_all:** حقيقي، بحجم عنصر 8 بايت

shmem_real4_xor_to_all:** حقيقي، بحجم عنصر 4 بايت

source
مصفوفة متماثلة، بطول nreduce عنصرًا، تحتوي على عنصر واحد لكل عملية اختزال منفصلة. يجب أن تكون للوسيطة source نفس نوع بيانات target.
nreduce
عدد العناصر في مصفوفتي target وsource. يجب أن يكون nreduce من النوع integer. إذا كنت تستخدم Fortran، فيجب أن يكون قيمة integer افتراضية.
PE_start
أدنى رقم PE افتراضي في المجموعة النشطة من PEs. يجب أن يكون PE_start من النوع integer. إذا كنت تستخدم Fortran، فيجب أن يكون قيمة integer افتراضية.
logPE_stride
لوغاريتم (الأساس 2) للخطوة بين أرقام PEs الافتراضية المتتالية في المجموعة النشطة. يجب أن يكون logPE_stride من النوع integer. إذا كنت تستخدم Fortran، فيجب أن يكون قيمة integer افتراضية.
PE_size
عدد PEs في المجموعة النشطة. يجب أن يكون PE_size من النوع integer. إذا كنت تستخدم Fortran، فيجب أن يكون قيمة integer افتراضية.
pWrk
مصفوفة عمل متماثلة. يجب أن تكون للوسيطة pWrk نفس نوع بيانات target. في C/C++، تحتوي على max(nreduce/2 + 1, _SHMEM_REDUCE_MIN_WRKDATA_SIZE) عنصرًا. في Fortran، تحتوي على max(nreduce/2 + 1, SHMEM_REDUCE_MIN_WRKDATA_SIZE) عنصرًا.
pSync
مصفوفة عمل متماثلة. في C/C++، يكون pSync من النوع long وبحجم _SHMEM_REDUCE_SYNC_SIZE. في Fortran، يكون pSync من النوع integer وبحجم SHMEM_REDUCE_SYNC_SIZE. إذا كنت تستخدم Fortran، فيجب أن يكون قيمة integer افتراضية. يجب تهيئة كل عنصر من هذه المصفوفة بالقيمة _SHMEM_SYNC_VALUE (في C/C++) أو SHMEM_SYNC_VALUE (في Fortran) قبل أن يدخل أي من PEs في المجموعة النشطة إلى روتين الاختزال.

يجب أن تكون قيم الوسائط nreduce وPE_start وlogPE_stride و PE_size متساوية على جميع PEs في المجموعة النشطة. يجب تمرير نفس مصفوفتي target و source، ونفس مصفوفتي العمل pWrk وpSync، إلى جميع PEs في المجموعة النشطة.

قبل أن يستدعي أي PE روتين اختزال، يجب عليك التأكد من وجود الشروط التالية (غالبًا ما تكون المزامنة عبر حاجز أو بعض الطرق الأخرى مطلوبة لضمان ذلك): مصفوفتا pWrk وpSync على جميع PEs في المجموعة النشطة ليستا قيد الاستخدام من استدعاء سابق لروتين SHMEM جماعي. مصفوفة target على جميع PEs في المجموعة النشطة جاهزة لاستقبال نتائج الاختزال.

عند العودة من روتين الاختزال، يكون ما يلي صحيحًا بالنسبة لـ PE المحلي: يتم تحديث مصفوفة target. تتم استعادة القيم في مصفوفة pSync إلى قيمها الأصلية.

ملاحظات

تُعرَّف مصطلحات collective وsymmetric وcache aligned في intro_shmem(3). تعيد جميع روتينات الاختزال في SHMEM تعيين القيم في pSync قبل أن تعود، لذا يحتاج مخزن pSync المؤقت إلى التهيئة فقط في المرة الأولى التي يُستخدم فيها.

يجب عليك التأكد من عدم تحديث مصفوفة pSync على أي PE في المجموعة النشطة بينما يشارك أي من PEs في معالجة روتين اختزال SHMEM. احترس من المواقف التالية: إذا تمت تهيئة مصفوفة pSync في وقت التشغيل، فستكون هناك حاجة إلى نوع من المزامنة لضمان أن جميع PEs في مجموعة العمل قد قامت بتهيئة pSync قبل أن يدخل أي منها إلى روتين SHMEM يُستدعى مع مصفوفة المزامنة pSync. يمكن إعادة استخدام مصفوفة pSync أو pWrk في استدعاء روتين اختزال لاحق فقط إذا لم يكن أي من PEs في المجموعة النشطة لا يزال يعالج استدعاء روتين اختزال سابقًا استخدم نفس مصفوفتي pSync أو pWrk. بشكل عام، لا يمكن ضمان ذلك إلا من خلال القيام بنوع من المزامنة. ومع ذلك، في الحالة الخاصة لاستدعاء روتينات الاختزال بنفس المجموعة النشطة، يمكنك تخصيص مصفوفتي pSync وpWrk والتناوب بينهما في الاستدعاءات المتتالية.

أمثلة

المثال 1: يقوم مثال Fortran هذا بتهيئة مصفوفة pSync بشكل ثابت ويحسب XOR الحصري للمتغير FOO عبر جميع PEs الزوجية.

INCLUDE "mpp/shmem.fh"
INTEGER PSYNC(SHMEM_REDUCE_SYNC_SIZE)
DATA PSYNC /SHMEM_REDUCE_SYNC_SIZE*SHMEM_SYNC_VALUE/
PARAMETER (NR=1)
REAL FOO, FOOXOR, PWRK(MAX(NR/2+1,SHMEM_REDUCE_MIN_WRKDATA_SIZE))
COMMON /COM/ FOO, FOOXOR, PWRK
INTRINSIC MY_PE
IF ( MOD(MY_PE(),2) .EQ. 0) THEN
  CALL SHMEM_REAL8_XOR_TO_ALL(FOOXOR, FOO, NR, 0, 1, N$PES/2,
  & PWRK, PSYNC)
  PRINT *, 'Result on PE ', MY_PE(), ' is ', FOOXOR
ENDIF

المثال 2: خذ بعين الاعتبار استدعاء C/C++ التالي:

shmem_short_xor_to_all( target, source, 3, 0, 0, 8, pwrk, psync );

الاستدعاء السابق أكثر كفاءة، لكنه مكافئ دلاليًا لمزيج الاستدعاءات التالية:

shmem_short_xor_to_all(&(target[0]), &(source[0]), 1, 0, 0, 8,
  pwrk1, psync1);
shmem_short_xor_to_all(&(target[1]), &(source[1]), 1, 0, 0, 8,
  pwrk2, psync2);
shmem_short_xor_to_all(&(target[2]), &(source[2]), 1, 0, 0, 8,
  pwrk1, psync1);

لاحظ أنه يتم استخدام مجموعتين من مصفوفتي pWrk وpSync بالتناوب لأنه لا تتم أي مزامنة بين الاستدعاءات.
