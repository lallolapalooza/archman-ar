shadow(3)

الحزمة: shadow

التراخيص: BSD-3-Clause

SHADOW(3) استدعاءات المكتبة SHADOW(3)

الاسم

shadow, getspnam - روتينات ملف كلمة المرور المشفرة

الصيغة

#include <shadow.h>

struct spwd *getspent();

struct spwd *getspnam(char *name);

void setspent();

void endspent();

struct spwd *fgetspent(FILE *fp);

struct spwd *sgetspent(char *cp);

int putspent(struct spwd *p, FILE *fp);

int lckpwdf();

int ulckpwdf();

الوصف

shadow يتعامل مع محتويات ملف كلمة المرور الظل، /etc/shadow. البنية في ملف #include هي:

struct spwd {
      char		*sp_namp; /* اسم تسجيل الدخول للمستخدم */
      char		*sp_pwdp; /* كلمة المرور المشفرة */
      long int		sp_lstchg; /* آخر تغيير لكلمة المرور */
      long int		sp_min; /* عدد الأيام حتى يتم السماح بالتغيير */
      long int		sp_max; /* عدد الأيام قبل أن يكون التغيير مطلوبًا */
      long int		sp_warn; /* عدد الأيام للتحذير من انتهاء الصلاحية */
      long int		sp_inact; /* عدد الأيام قبل أن يصبح الحساب غير نشط */
      long int		sp_expire; /* التاريخ الذي سينتهي فيه الحساب */
      unsigned long int	sp_flag; /* محجوز للاستخدام المستقبلي */
}

معاني كل حقل هي:

•sp_namp - مؤشر إلى اسم المستخدم المنتهي بـ null
•sp_pwdp - مؤشر إلى كلمة المرور المنتهية بـ null
•sp_lstchg - عدد الأيام منذ 1 يناير 1970 تم تغيير كلمة المرور آخر مرة
•sp_min - عدد الأيام التي لا يمكن قبلها تغيير كلمة المرور
•sp_max - عدد الأيام التي يجب بعدها تغيير كلمة المرور
•sp_warn - عدد الأيام قبل انتهاء صلاحية كلمة المرور التي يتم تحذير المستخدم بها من انتهاء صلاحية كلمة المرور الوشيك
•sp_inact - عدد الأيام بعد انتهاء صلاحية كلمة المرور التي يعتبر فيها الحساب غير نشط ومعطل
•sp_expire - عدد الأيام منذ 1 يناير 1970 عندما يتم تعطيل الحساب
•sp_flag - محجوز للاستخدام المستقبلي

الوصف

getspent, getspname, fgetspent, و sgetspent كل منها تعيد مؤشرًا إلى struct spwd. getspent تعيد الإدخال التالي من الملف، وتعيد fgetspent الإدخال التالي من التدفق المعطى، والذي يفترض أنه ملف بالتنسيق الصحيح. sgetspent تعيد مؤشرًا إلى struct spwd باستخدام السلسلة المقدمة كمدخل. getspnam تبحث من الموضع الحالي في الملف عن إدخال مطابق لـ name.

setspent و endspent يمكن استخدامهما لبدء وإنهاء، على التوالي، الوصول إلى ملف كلمة المرور الظل.

يجب استخدام الروتينات lckpwdf و ulckpwdf لضمان الوصول الحصري إلى ملف /etc/shadow. lckpwdf تحاول الحصول على قفل باستخدام pw_lock لمدة تصل إلى 15 ثانية. تستمر بمحاولة الحصول على قفل ثانٍ باستخدام spw_lock لبقية الـ 15 ثانية الأولية. إذا فشلت أي محاولة بعد إجمالي 15 ثانية، فإن lckpwdf تعيد -1. عندما يتم الحصول على كلا القفلين، يتم إعادة 0.

التشخيصات

تعيد الروتينات NULL إذا لم تكن هناك إدخالات متاحة أخرى أو إذا حدث خطأ أثناء المعالجة. الروتينات التي لها int كقيمة إرجاع تعيد 0 للنجاح و -1 للفشل.

تحذيرات

قد يتم استخدام هذه الروتينات فقط من قبل المستخدم الخارق حيث أن الوصول إلى ملف كلمة المرور الظل مقيد.
