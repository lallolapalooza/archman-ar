shmem_longlong_or_to_all(3)

الحزمة: openmpi

التراخيص: BSD-3-Clause AND LicenseRef-MPICH

SHMEM_LONGLONG_OR_TO_ALL(3) Open MPI SHMEM_LONGLONG_OR_TO_ALL(3)

shmem_int_or_to_all(3)، shmem_int4_or_to_all(3)، shmem_int8_or_to_all(3)، shmem_long_or_to_all(3)، shmem_longlong_or_to_all(3)، shmem_short_or_to_all(3) - يجري عملية OR بت (bitwise OR function) عبر مجموعة من عناصر المعالجة (PEs)

SYNOPSIS

C أو C++:

#include <mpp/shmem.h>
void shmem_int_or_to_all(int *target, const int *source,
  int nreduce, int PE_start, int logPE_stride, int PE_size,
  int *pWrk, long *pSync)
void shmem_long_or_to_all(long *target, const long *source,
  int nreduce, int PE_start, int logPE_stride, int PE_size,
  long *pWrk, long *pSync)
void shmem_longlong_or_to_all(long long *target,
  const long long *source, int nreduce, int PE_start, int logPE_stride,
  int PE_size, long long *pWrk, long *pSync)
void shmem_short_or_to_all(short *target, const short *source,
  int nreduce, int PE_start, int logPE_stride, int PE_size,
  short *pWrk, long *pSync)

Fortran:

INCLUDE "mpp/shmem.fh"
INTEGER pSync(SHMEM_REDUCE_SYNC_SIZE)
INTEGER nreduce, PE_start, logPE_stride, PE_size
CALL SHMEM_INT4_OR_TO_ALL(target, source, nreduce, PE_start,
& logPE_stride, PE_size, pWrk, pSync)
CALL SHMEM_INT8_OR_TO_ALL(target, source, nreduce, PE_start,
& logPE_stride, PE_size, pWrk, pSync)

DESCRIPTION

تقوم إجراءات تقليل الذاكرة المشتركة (SHMEM) بحساب عملية تقليل واحدة أو أكثر عبر مصفوفات متماثلة على عناصر معالجة افتراضية متعددة. يقوم التقليل بإجراء عملية ثنائية ترابطية عبر مجموعة من القيم. للحصول على قائمة بإجراءات تقليل SHMEM الأخرى، راجع intro_shmem(3).

كما هو الحال مع جميع إجراءات SHMEM الجماعية، تفترض كل من هذه الإجراءات أن عناصر المعالجة في المجموعة النشطة فقط هي التي تستدعي الإجراء. إذا استدعى عنصر معالجة غير موجود في المجموعة النشطة إجراء SHMEM جماعيًا، فستكون النتيجة غير محددة.

يحدد وسيط nreduce عدد عمليات التقليل المنفصلة التي سيتم إجراؤها. توفر مصفوفة المصدر على جميع عناصر المعالجة في المجموعة النشطة عنصرًا واحدًا لكل عملية تقليل. توضع نتائج عمليات التقليل في مصفوفة الهدف على جميع عناصر المعالجة في المجموعة النشطة. يتم تعريف المجموعة النشطة بواسطة الثلاثي PE_start، logPE_stride، PE_size.

قد تكون مصفوفتي المصدر والهدف نفس المصفوفة، ولكن لا يمكن أن تكونا مصفوفتين متداخلتين.

الوسائط هي كما يلي:

target
مصفوفة متماثلة بطول nreduce عنصر لاستقبال نتائج عمليات التقليل. يختلف نوع بيانات target مع إصدار إجراء التقليل الذي يتم استدعاؤه. عند الاستدعاء من C/C++، ارجع إلى قسم SYNOPSIS للحصول على معلومات نوع البيانات. عند الاستدعاء من Fortran، تكون أنواع بيانات الهدف كما يلي:

shmem_int8_or_to_all عدد صحيح، بحجم عنصر 8 بايت.

shmem_int4_or_to_all عدد صحيح، بحجم عنصر 4 بايت.

source
مصفوفة متماثلة، بطول nreduce عنصر، تحتوي على عنصر واحد لكل عملية تقليل منفصلة. يجب أن يكون وسيط المصدر من نفس نوع بيانات target.
nreduce
عدد العناصر في مصفوفتي target و source. يجب أن يكون nreduce من نوع عدد صحيح. إذا كنت تستخدم Fortran، فيجب أن يكون قيمة عدد صحيح افتراضية.
PE_start
أقل رقم PE افتراضي للمجموعة النشطة من PEs. يجب أن يكون PE_start من نوع عدد صحيح. إذا كنت تستخدم Fortran، فيجب أن يكون قيمة عدد صحيح افتراضية.
logPE_stride
اللوغاريتم (الأساس 2) للخطوة بين أرقام PE الافتراضية المتتالية في المجموعة النشطة. يجب أن يكون logPE_stride من نوع عدد صحيح. إذا كنت تستخدم Fortran، فيجب أن يكون قيمة عدد صحيح افتراضية.
PE_size
عدد PEs في المجموعة النشطة. يجب أن يكون PE_size من نوع عدد صحيح. إذا كنت تستخدم Fortran، فيجب أن يكون قيمة عدد صحيح افتراضية.
pWrk
مصفوفة عمل متماثلة. يجب أن يكون وسيط pWrk من نفس نوع بيانات target. في C/C++، تحتوي على max(nreduce/2 + 1, _SHMEM_REDUCE_MIN_WRKDATA_SIZE) عناصر. في Fortran، تحتوي على max(nreduce/2 + 1, SHMEM_REDUCE_MIN_WRKDATA_SIZE) عناصر.
pSync
مصفوفة عمل متماثلة. في C/C++، تكون pSync من نوع long وحجم _SHMEM_REDUCE_SYNC_SIZE. في Fortran، تكون pSync من نوع عدد صحيح وحجم SHMEM_REDUCE_SYNC_SIZE. إذا كنت تستخدم Fortran، فيجب أن تكون قيمة عدد صحيح افتراضية. يجب تهيئة كل عنصر من هذه المصفوفة بالقيمة _SHMEM_SYNC_VALUE (في C/C++) أو SHMEM_SYNC_VALUE (في Fortran) قبل أن تدخل أي من PEs في المجموعة النشطة إلى إجراء التقليل.

يجب أن تكون قيم الوسائط nreduce و PE_start و logPE_stride و PE_size متساوية على جميع PEs في المجموعة النشطة. يجب تمرير نفس مصفوفات target و source، ونفس مصفوفات العمل pWrk و pSync، إلى جميع PEs في المجموعة النشطة.

قبل أن يقوم أي PE باستدعاء إجراء تقليل، يجب عليك التأكد من وجود الشروط التالية (غالبًا ما تكون هناك حاجة إلى المزامنة باستخدام حاجز أو طريقة أخرى لضمان ذلك): مصفوفات pWrk و pSync على جميع PEs في المجموعة النشطة ليست قيد الاستخدام بالفعل من استدعاء سابق لإجراء SHMEM جماعي. مصفوفة target على جميع PEs في المجموعة النشطة جاهزة لاستقبال نتائج التقليل.

عند العودة من إجراء التقليل، تكون الأمور التالية صحيحة: يتم تحديث مصفوفة target. يتم استعادة القيم في مصفوفة pSync إلى القيم الأصلية.

NOTES

يتم تعريف المصطلحات جماعي، متماثل، ومتراص في الذاكرة المؤقتة في intro_shmem(3). تعيد جميع إجراءات تقليل SHMEM تعيين القيم في pSync قبل أن تعود، لذا لا يلزم تهيئة مخزن مؤقت pSync معين إلا في المرة الأولى التي يتم استخدامه فيها.

يجب عليك التأكد من أن مصفوفة pSync لا يتم تحديثها على أي PE في المجموعة النشطة بينما تشارك أي من PEs في معالجة إجراء SHMEM تقليل. كن حذرًا لتجنب هذه المواقف: إذا تم تهيئة مصفوفة pSync في وقت التشغيل، فستكون هناك حاجة إلى نوع من المزامنة لضمان أن جميع PEs في مجموعة العمل قد قامت بتهيئة pSync قبل أن تدخل أي منها إلى إجراء SHMEM تم استدعاؤه باستخدام مصفوفة مزامنة pSync. يمكن إعادة استخدام مصفوفة pSync أو pWrk في استدعاء إجراء تقليل لاحق فقط إذا لم يكن أي من PEs في المجموعة النشطة لا يزال يعالج استدعاء إجراء تقليل سابق استخدم نفس مصفوفات pSync أو pWrk. بشكل عام، لا يمكن ضمان ذلك إلا عن طريق القيام بنوع من المزامنة. ومع ذلك، في الحالة الخاصة لإجراءات التقليل التي يتم استدعاؤها بنفس المجموعة النشطة، يمكنك تخصيص مصفوفتي pSync و pWrk والتبديل بينهما في الاستدعاءات المتتالية.

EXAMPLES

مثال 1: يقوم مثال Fortran هذا بتهيئة مصفوفة pSync بشكل ثابت وإيجاد OR المنطقي للمتغير الصحيح FOO عبر جميع PEs الزوجية.

INCLUDE "mpp/shmem.fh"
INTEGER PSYNC(SHMEM_REDUCE_SYNC_SIZE)
DATA PSYNC /SHMEM_REDUCE_SYNC_SIZE*SHMEM_SYNC_VALUE/
PARAMETER (NR=1)
REAL PWRK(MAX(NR/2+1,SHMEM_REDUCE_MIN_WRKDATA_SIZE))
INTEGER FOO, FOOOR
COMMON /COM/ FOO, FOOOR, PWRK
INTRINSIC MY_PE
IF ( MOD(MY_PE(),2) .EQ. 0) THEN
  CALL SHMEM_INT8_OR_TO_ALL(FOOOR, FOO, NR, 0, 1, N$PES/2,
  & PWRK, PSYNC)
  PRINT *,'Result on PE ',MY_PE(),' is ',FOOOR
ENDIF

مثال 2: ضع في اعتبارك استدعاء C/C++ التالي:

shmem_int_or_to_all( target, source, 3, 0, 0, 8, pwrk, psync );

الاستدعاء السابق أكثر كفاءة، ولكنه يكافئ دلاليًا، مجموعة الاستدعاءات التالية:

shmem_int_or_to_all(&(target[0]), &(source[0]), 1, 0, 0, 8,
  pwrk1, psync1);
shmem_int_or_to_all(&(target[1]), &(source[1]), 1, 0, 0, 8,
  pwrk2, psync2);
shmem_int_or_to_all(&(target[2]), &(source[2]), 1, 0, 0, 8,
  pwrk1, psync1);

لاحظ أنه يتم استخدام مجموعتين من مصفوفات pWrk و pSync بالتناوب لأنه لا يتم إجراء مزامنة بين الاستدعاءات.
