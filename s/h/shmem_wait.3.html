shmem_wait(3)

الحزمة: openmpi

التراخيص: BSD-3-Clause AND LicenseRef-MPICH

SHMEM_WAIT(3) Open MPI SHMEM_WAIT(3)

shmem_int_wait(3), shmem_int_wait(3)_until, shmem_int4_wait(3), shmem_int4_wait(3)_until, shmem_int8_wait(3), shmem_int8_wait(3)_until, shmem_long_wait(3), shmem_long_wait(3)_until, shmem_longlong_wait(3), shmem_longlong_wait(3)_until, shmem_short_wait(3), shmem_short_wait(3)_until, shmem_wait(3), shmem_wait(3)_until - ينتظر تغيير متغير على عنصر المعالجة (PE) المحلي

ملخص

C أو C++:

#include <mpp/shmem.h>
void shmem_int_wait(volatile int *var, int value)
void shmem_int_wait_until(volatile int *var, int cond, int value)
void shmem_long_wait(volatile long *var, long value)
void shmem_long_wait_until(volatile long *var, int cond, long value)
void shmem_longlong_wait(volatile long long *var, long long value)
void shmem_longlong_wait_until(volatile long long *var, int cond,
  long long value)
void shmem_short_wait(volatile short *var, short value)
void shmem_short_wait_until(volatile short *var, int cond,
  short value)
void shmem_wait(volatile long *ivar, long cmp_value)
void shmem_wait_until(volatile long *ivar, int cmp, long value)

Fortran:

INCLUDE "mpp/shmem.fh"
CALL SHMEM_INT4_WAIT(ivar, cmp_value)
CALL SHMEM_INT4_WAIT_UNTIL(ivar, cmp, cmp_value)
CALL SHMEM_INT8_WAIT(ivar, cmp_value)
CALL SHMEM_INT8_WAIT_UNTIL(ivar, cmp, cmp_value)
CALL SHMEM_WAIT(ivar, cmp_value)
CALL SHMEM_WAIT_UNTIL(ivar, cmp, cmp_value)

الوصف

shmem_wait و shmem_wait_until ينتظران تغيير ivar عبر كتابة بعيدة أو تبديل ذري صادر من معالج مختلف. يمكن استخدام هذه الروتينات للمزامنة الموجهة نقطة إلى نقطة. استدعاء shmem_wait لا يعود إلا عندما يكتب معالج آخر قيمة، ليست مساوية لـ cmp_value، في ivar على المعالج المنتظر. استدعاء shmem_wait_until لا يعود إلا عندما يغير معالج آخر ivar لتلبية الشرط الضمني في cmp و cmp_value. هذه الآلية مفيدة عندما يحتاج معالج إلى إخبار معالج آخر بأنه أكمل إجراءً ما.

الوسائط هي كالتالي:

target
كائن البيانات الصحيح الذي يمكن الوصول إليه عن بعد ليتم تحديثه على PE البعيد. إذا كنت تستخدم C/C++، يجب أن يتطابق نوع target مع النوع الضمني في قسم SYNOPSIS. إذا كنت تستخدم مترجم Fortran، يجب أن يكون من نوع integer بحجم عنصر 4 بايت لـ SHMEM_INT4_ADD و 8 بايت لـ SHMEM_INT8_ADD.
value
القيمة التي سيتم إضافتها ذريًا إلى target. إذا كنت تستخدم C/C++، يجب أن يتطابق نوع value مع النوع الضمني في قسم SYNOPSIS. إذا كنت تستخدم Fortran، يجب أن يكون من نوع integer بحجم عنصر مطابق لـ target.
pe
عدد صحيح يشير إلى رقم PE الذي سيتم تحديث target عليه. إذا كنت تستخدم Fortran، يجب أن يكون قيمة integer افتراضية.
ivar
متغير صحيح يمكن الوصول إليه عن بعد يتم تحديثه بواسطة PE آخر. إذا كنت تستخدم C/C++، يجب أن يتطابق نوع ivar مع النوع الضمني في قسم SYNOPSIS. إذا كنت تستخدم Fortran، يجب أن يكون ivar نوع integer بحجم محدد وفقًا للدالة التي يتم استدعاؤها، كما يلي:

shmem_wait, shmem_wait_until: INTEGER افتراضي

shmem_int4_wait, shmem_int4_wait_until: INTEGER*4

shmem_int8_wait, shmem_int8_wait_until: INTEGER*8

cmp
عامل المقارنة الذي يقارن ivar مع cmp_value. يجب أن يكون cmp من نوع integer. إذا كنت تستخدم Fortran، يجب أن يكون من النوع الافتراضي. إذا كنت تستخدم C/C++، يجب أن يتطابق نوع cmp مع النوع الضمني في قسم SYNOPSIS. تتم دعم قيم cmp التالية:
SHMEM_CMP_EQ
يساوي
SHMEM_CMP_NE
لا يساوي
SHMEM_CMP_GT
أكبر من
SHMEM_CMP_LE
أقل من أو يساوي
SHMEM_CMP_LT
أقل من
SHMEM_CMP_GE
أكبر من أو يساوي
cmp_value
يجب أن يكون cmp_value من نوع integer. إذا كنت تستخدم C/C++، يجب أن يتطابق نوع cmp_value مع النوع الضمني في قسم SYNOPSIS. إذا كنت تستخدم Fortran، يجب أن يكون cmp_value عددًا صحيحًا بنفس حجم ونوع ivar. تعود روتينات shmem_wait عندما لا يعود ivar مساويًا لـ cmp_value. تعود روتينات shmem_wait_until عندما يكون شرط المقارنة صحيحًا. يتم تعريف شرط المقارنة بواسطة الوسيط ivar مقارنة بـ cmp_value باستخدام عامل المقارنة، cmp.

أمثلة

مثال 1: العودة التالية تتم عندما لا يساوي المتغير ivar القيمة 100:

INTEGER*8 IVAR
CALL SHMEM_INT8_WAIT(IVAR, INT8(100))

مثال 2: الاستدعاء التالي لـ SHMEM_INT8_WAIT_UNTIL يعادل الاستدعاء لـ SHMEM_INT8_WAIT في المثال 1:

INTEGER*8 IVAR
CALL SHMEM_INT8_WAIT_UNTIL(IVAR, SHMEM_CMP_NE, INT8(100))

مثال 3: الاستدعاء التالي لـ C/C++ ينتظر حتى يتم تعيين بت الإشارة في ivar بواسطة نقل من PE بعيد:

int ivar;
shmem_int_wait_until(&ivar, SHMEM_CMP_LT, 0);

مثال 4: المثال التالي لـ Fortran في سياق روتين فرعي:

SUBROUTINE EXAMPLE()
  INTEGER FLAG_VAR
  COMMON/FLAG/FLAG_VAR
  . . .
  FLAG_VAR = FLAG_VALUE ! تهيئة متغير الحدث
  . . .
  IF (FLAG_VAR .EQ. FLAG_VALUE) THEN
    CALL SHMEM_WAIT(FLAG_VAR, FLAG_VALUE)
  ENDIF
  FLAG_VAR = FLAG_VALUE ! إعادة تعيين متغير الحدث للمرة القادمة
  . . .
END
