توثيق الوظيفة/الروتين الفرعي

subroutine shseqr (character job, character compz, integer n, integer ilo, integer ihi, real, dimension( ldh, * ) h, integer ldh, real, dimension( * ) wr, real, dimension( * ) wi, real, dimension( ldz, * ) z, integer ldz, real, dimension( * ) work, integer lwork, integer info)

SHSEQR

الغرض:

!>
!>    SHSEQR يحسب القيم الذاتية للمصفوفة هسنبرغ H
!>    وبشكل اختياري، المصفوفات T و Z من تفكيك شورش
!>    H = Z T Z**T، حيث T مصفوفة شبه مثلثية علوية (شكل شورش)،
!>    و Z هي المصفوفة المتعامدة لمتجهات شورش.
!>
!>    بشكل اختياري، يمكن ضرب Z بالمصفوفة المتعامدة Q المدخلة
!>    بحيث يمكن لهذه الروتين الفرعي إعطاء تفكيك شورش
!>    لمصفوفة A تم اختزالها إلى شكل هسنبرغ H
!>    بواسطة المصفوفة المتعامدة Q:  A = Q*H*Q**T = (QZ)*T*(QZ)**T.
!>

المعلمات

JOB 
!>          JOB هو CHARACTER*1
!>           = 'E':  حساب القيم الذاتية فقط؛
!>           = 'S':  حساب القيم الذاتية وشكل شورش T.
!>

COMPZ

!>          COMPZ هو CHARACTER*1
!>           = 'N':  لا يتم حساب متجهات شورش؛
!>           = 'I':  يتم تهيئة Z إلى مصفوفة الوحدة ويتم إرجاع المصفوفة Z
!>                   لمتجهات شورش لـ H؛
!>           = 'V':  يجب أن تحتوي Z على مصفوفة متعامدة Q عند الإدخال، ويتم
!>                   إرجاع ناتج Q*Z.
!>

N

!>          N هو INTEGER
!>           رتبة المصفوفة H.  N >= 0.
!>

ILO

!>          ILO هو INTEGER
!>

IHI

!>          IHI هو INTEGER
!>
!>           يفترض أن H هي بالفعل شبه مثلثية في الصفوف
!>           والأعمدة 1:ILO-1 و IHI+1:N. يتم عادةً تعيين ILO و IHI
!>           بواسطة استدعاء سابق لـ SGEBAL، ثم تمريرهما إلى ZGEHRD
!>           عند اختزال مخرجات المصفوفة لـ SGEBAL إلى شكل هسنبرغ.
!>           خلاف ذلك، يجب تعيين ILO و IHI إلى 1 و N على التوالي.
!>           إذا كان N > 0، فإن 1 <= ILO <= IHI <= N.
!>           إذا كان N = 0، فإن ILO = 1 و IHI = 0.
!>

H

!>          H هي مصفوفة REAL، الأبعاد (LDH,N)
!>           عند الإدخال، مصفوفة هسنبرغ العلوية H.
!>           عند الخروج، إذا كان INFO = 0 و JOB = 'S'، فإن H تحتوي على
!>           المصفوفة شبه المثلثية العلوية T من تفكيك شورش
!>           (شكل شورش)؛ يتم إرجاع الكتل القطرية 2x2 (المقابلة
!>           لأزواج القيم الذاتية المترافقة المعقدة)
!>           بالشكل القياسي، مع H(i,i) = H(i+1,i+1) و
!>           H(i+1,i)*H(i,i+1) < 0. إذا كان INFO = 0 و JOB = 'E'، فإن
!>           محتويات H غير محددة عند الخروج. (القيمة المخرجة لـ
!>           H عندما يكون INFO > 0 معطاة تحت وصف INFO أدناه.)
!>
!>           على عكس الإصدارات السابقة من SHSEQR، قد تقوم هذه الروتين الفرعي
!>           بشكل صريح H(i,j) = 0 لـ i > j و j = 1، 2، ... ILO-1
!>           أو j = IHI+1، IHI+2، ... N.
!>

LDH

!>          LDH هو INTEGER
!>           البعد الرائد للمصفوفة H. LDH >= max(1,N).
!>

WR

!>          WR هي مصفوفة REAL، الأبعاد (N)
!>

WI

!>          WI هي مصفوفة REAL، الأبعاد (N)
!>
!>           الأجزاء الحقيقية والتخيلية، على التوالي، للقيم الذاتية المحسوبة.
!>           إذا تم حساب قيمتين ذاتيتين كزوج مترافق معقد، فسيتم تخزينهما
!>           في عناصر متتالية في WR و WI، لنقل العنصر i والعنصر (i+1)،
!>           بحيث WI(i) > 0 و WI(i+1) < 0. إذا كان JOB = 'S'، يتم تخزين
!>           القيم الذاتية بنفس الترتيب الذي تظهر به على القطر الرئيسي
!>           لشكل شورش المرجع في H، مع WR(i) = H(i,i) و، إذا كانت
!>           H(i:i+1,i:i+1) كتلة قطرية 2x2، فإن WI(i) = sqrt(-H(i+1,i)*H(i,i+1))
!>           و WI(i+1) = -WI(i).
!>

Z

!>          Z هي مصفوفة REAL، الأبعاد (LDZ,N)
!>           إذا كان COMPZ = 'N'، لم يتم الرجوع إلى Z.
!>           إذا كان COMPZ = 'I'، عند الإدخال، لا يلزم تعيين Z، وعند الخروج،
!>           إذا كان INFO = 0، تحتوي Z على المصفوفة المتعامدة Z لمتجهات شورش
!>           لـ H. إذا كان COMPZ = 'V'، عند الإدخال، يجب أن تحتوي Z على
!>           مصفوفة N×N Q، والتي يفترض أنها متطابقة مع مصفوفة الوحدة
!>           باستثناء المصفوفة الفرعية Z(ILO:IHI,ILO:IHI). عند الخروج،
!>           إذا كان INFO = 0، تحتوي Z على Q*Z.
!>           عادةً ما تكون Q هي المصفوفة المتعامدة التي تم إنشاؤها بواسطة SORGHR
!>           بعد استدعاء SGEHRD الذي شكل مصفوفة هسنبرغ H. (القيمة المخرجة لـ
!>           Z عندما يكون INFO > 0 معطاة تحت وصف INFO أدناه.)
!>

LDZ

!>          LDZ هو INTEGER
!>           البعد الرائد للمصفوفة Z. إذا كان COMPZ = 'I' أو
!>           COMPZ = 'V'، فإن LDZ >= MAX(1,N).
!>           خلاف ذلك، LDZ >= 1.
!>

WORK

!>          WORK هي مصفوفة REAL، الأبعاد (LWORK)
!>           عند الخروج، إذا كان INFO = 0، فإن WORK(1) ترجع تقديرًا
!>           للقيمة المثلى لـ LWORK.
!>

LWORK

!>          LWORK هو INTEGER
!>           بعد المصفوفة WORK.  LWORK >= max(1,N)
!>           كافٍ ويحقق أداءً جيدًا جدًا وفي بعض الأحيان
!>           مثاليًا.  ومع ذلك، قد يكون LWORK بحجم يصل إلى 11*N
!>           مطلوبًا لتحقيق الأداء الأمثل.
!>           يوصى باستعلام مساحة العمل لتحديد حجم مساحة العمل الأمثل.
!>
!>           إذا كان LWORK = -1، فإن SHSEQR يقوم باستعلام مساحة العمل.
!>           في هذه الحالة، تتحقق SHSEQR من معلمات الإدخال
!>           وتقدر حجم مساحة العمل الأمثل للقيم المعطاة
!>           لـ N و ILO و IHI.  يتم إرجاع التقدير
!>           في WORK(1).  لا يتم إصدار رسالة خطأ متعلقة بـ LWORK
!>           بواسطة XERBLA.  لا يتم الوصول إلى H ولا Z.
!>

INFO

!>          INFO هو INTEGER
!>             = 0:  خروج ناجح
!>             < 0:  إذا كان INFO = -i، فإن المعلمة i كانت ذات قيمة غير قانونية
!>             > 0:  إذا كان INFO = i، فشلت SHSEQR في حساب جميع
!>                القيم الذاتية.  العناصر 1:ilo-1 و i+1:n من WR
!>                و WI تحتوي على تلك القيم الذاتية التي تم
!>                حسابها بنجاح.  (حالات الفشل نادرة.)
!>
!>                إذا كان INFO > 0 و JOB = 'E'، فإنه عند الخروج،
!>                القيم الذاتية المتبقية غير المتقاربة هي القيم الذاتية
!>                لمصفوفة هسنبرغ العلوية للصفوف
!>                والأعمدة ILO إلى INFO من القيمة النهائية، المخرجة،
!>                من H.
!>
!>                إذا كان INFO > 0 و JOB   = 'S'، فإنه عند الخروج
!>
!>           (*)  (القيمة الأولية لـ H)*U  = U*(القيمة النهائية لـ H)
!>
!>                حيث U هي مصفوفة متعامدة.  القيمة النهائية
!>                لـ H هي هسنبرغ علوية وشبه مثلثية
!>                في الصفوف والأعمدة INFO+1 إلى IHI.
!>
!>                إذا كان INFO > 0 و COMPZ = 'V'، فإنه عند الخروج
!>
!>                  (القيمة النهائية لـ Z)  =  (القيمة الأولية لـ Z)*U
!>
!>                حيث U هي المصفوفة المتعامدة في (*) (بغض النظر
!>                عن قيمة JOB.)
!>
!>                إذا كان INFO > 0 و COMPZ = 'I'، فإنه عند الخروج
!>                      (القيمة النهائية لـ Z)  = U
!>                حيث U هي المصفوفة المتعامدة في (*) (بغض النظر
!>                عن قيمة JOB.)
!>
!>                إذا كان INFO > 0 و COMPZ = 'N'، فإنه لا يتم
!>                الوصول إلى Z.
!>

!>             القيم الافتراضية التي يوفرها
!>             ILAENV(ISPEC,'SHSEQR',JOB(:1)//COMPZ(:1),N,ILO,IHI,LWORK).
!>             يُقترح تعديل هذه الافتراضيات لتحقيق أفضل أداء في كل بيئة
!>             حسابية محددة.
!>
!>            ISPEC=12: نقطة التقاطع بين SLAHQR و SLAQR0.
!>                      الافتراضي: 75. (يجب أن يكون على الأقل 11.)
!>
!>            ISPEC=13: حجم نافذة الإزالة الموصى به.
!>                      يعتمد هذا على ILO و IHI و NS. NS هو عدد
!>                      الانتقالات المتزامنة التي يتم إرجاعها بواسطة
!>                      ILAENV(ISPEC=15). (انظر ISPEC=15 أدناه.)
!>                      الافتراضي لـ (IHI-ILO+1) <= 500 هو NS.
!>                      الافتراضي لـ (IHI-ILO+1) > 500 هو 3*NS/2.
!>
!>            ISPEC=14: نقطة التقاطع. (انظر IPARMQ للتفاصيل.)
!>                      الافتراضي: 14% من حجم نافذة الإزالة.
!>
!>            ISPEC=15: عدد الانتقالات المتزامنة في تكرار QR متعدد
!>                      الانتقالات.
!>
!>                      إذا كان IHI-ILO+1 ...
!>
!>                      أكبر من ... ولكن أقل ... الافتراضي هو
!>                      أو يساوي ... من
!>
!>                           1               30          NS =   2(+)
!>                          30               60          NS =   4(+)
!>                          60              150          NS =  10(+)
!>                         150              590          NS =  **
!>                         590             3000          NS =  64
!>                        3000             6000          NS = 128
!>                        6000             infinity      NS = 256
!>
!>                  (+)  افتراضيًا، يتم تمرير بعض أو كل المصفوفات من هذا
!>                       الترتيب إلى روتين الانتقال المزدوج الضمني SLAHQR
!>                       وهذه المعلمة يتم تجاهلها. انظر ISPEC=12 أعلاه
!>                       والتعليقات في IPARMQ للتفاصيل.
!>
!>                 (**)  تشير النجوم (**) إلى دالة ad-hoc
!>                       لـ N متزايدة من 10 إلى 64.
!>
!>            ISPEC=16: حدد ضرب المصفوفة المنظمة.
!>                      إذا كان عدد الانتقالات المتزامنة (المحددة
!>                      بواسطة ISPEC=15) أقل من 14، فإن الافتراضي
!>                      لـ ISPEC=16 هو 0. بخلاف ذلك، فإن الافتراضي
!>                      لـ ISPEC=16 هو 2.
!>
