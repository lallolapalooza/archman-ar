shmem_short_wait_until(3)

الحزمة: openmpi

التراخيص: BSD-3-Clause AND LicenseRef-MPICH

SHMEM_SHORT_WAIT_UNTIL(3) Open MPI SHMEM_SHORT_WAIT_UNTIL(3)

shmem_int_wait(3), shmem_int_wait(3)_until, shmem_int4_wait(3), shmem_int4_wait(3)_until, shmem_int8_wait(3), shmem_int8_wait(3)_until, shmem_long_wait(3), shmem_long_wait(3)_until, shmem_longlong_wait(3), shmem_longlong_wait(3)_until, shmem_short_wait(3), shmem_short_wait(3)_until, shmem_wait(3), shmem_wait(3)_until - ينتظر حتى يتغير متغير على عنصر المعالجة (PE) المحلي

ملخص

C أو C++:

#include <mpp/shmem.h>
void shmem_int_wait(volatile int *var, int value)
void shmem_int_wait_until(volatile int *var, int cond, int value)
void shmem_long_wait(volatile long *var, long value)
void shmem_long_wait_until(volatile long *var, int cond, long value)
void shmem_longlong_wait(volatile long long *var, long long value)
void shmem_longlong_wait_until(volatile long long *var, int cond,
  long long value)
void shmem_short_wait(volatile short *var, short value)
void shmem_short_wait_until(volatile short *var, int cond,
  short value)
void shmem_wait(volatile long *ivar, long cmp_value)
void shmem_wait_until(volatile long *ivar, int cmp, long value)

Fortran:

INCLUDE "mpp/shmem.fh"
CALL SHMEM_INT4_WAIT(ivar, cmp_value)
CALL SHMEM_INT4_WAIT_UNTIL(ivar, cmp, cmp_value)
CALL SHMEM_INT8_WAIT(ivar, cmp_value)
CALL SHMEM_INT8_WAIT_UNTIL(ivar, cmp, cmp_value)
CALL SHMEM_WAIT(ivar, cmp_value)
CALL SHMEM_WAIT_UNTIL(ivar, cmp, cmp_value)

الوصف

shmem_wait و shmem_wait_until تنتظران حتى يتغير ivar بواسطة كتابة بعيدة أو تبديل ذري صادر من معالج مختلف. يمكن استخدام هذه الروتينات للمزامنة الموجهة بين الأقران. لا يتم إرجاع استدعاء إلى shmem_wait حتى يكتب معالج آخر قيمة، غير مساوية لـ cmp_value، إلى ivar على المعالج المنتظر. لا يتم إرجاع استدعاء إلى shmem_wait_until حتى يغير معالج آخر ivar لتلبية الشرط الضمني بواسطة cmp و cmp_value. هذه الآلية مفيدة عندما يحتاج المعالج إلى إخبار معالج آخر بأنه قد أكمل إجراءً ما.

الوسائط هي كما يلي:

target
كائن البيانات الصحيح الذي يمكن الوصول إليه عن بعد ليتم تحديثه على PE البعيد. إذا كنت تستخدم C/C++، فيجب أن يتطابق نوع الهدف مع النوع الضمني في قسم الملخص. إذا كنت تستخدم مترجم Fortran، فيجب أن يكون من نوع صحيح بحجم عنصر 4 بايت لـ SHMEM_INT4_ADD و 8 بايت لـ SHMEM_INT8_ADD.
value
القيمة التي سيتم إضافتها بشكل ذري إلى target. إذا كنت تستخدم C/C++، فيجب أن يتطابق نوع القيمة مع النوع الضمني في قسم الملخص. إذا كنت تستخدم Fortran، فيجب أن يكون من نوع صحيح بحجم عنصر مطابق لحجم target.
pe
عدد صحيح يشير إلى رقم PE الذي سيتم تحديث target عليه. إذا كنت تستخدم Fortran، فيجب أن تكون قيمة صحيحة افتراضية.
ivar
متغير صحيح يمكن الوصول إليه عن بعد يتم تحديثه بواسطة PE آخر. إذا كنت تستخدم C/C++، فيجب أن يتطابق نوع ivar مع النوع الضمني في قسم الملخص. إذا كنت تستخدم Fortran، يجب أن يكون ivar من نوع صحيح بحجم محدد وفقًا للدالة التي يتم استدعاؤها، على النحو التالي:

shmem_wait, shmem_wait_until:** صحيح افتراضي

shmem_int4_wait, shmem_int4_wait_until:** صحيح*4

shmem_int8_wait, shmem_int8_wait_until:** صحيح*8

cmp
عامل المقارنة الذي يقارن ivar بـ cmp_value. يجب أن يكون cmp من نوع صحيح. إذا كنت تستخدم Fortran، فيجب أن يكون من النوع الافتراضي. إذا كنت تستخدم C/C++، فيجب أن يتطابق نوع cmp مع النوع الضمني في قسم الملخص. يتم دعم قيم cmp التالية:
SHMEM_CMP_EQ
يساوي
SHMEM_CMP_NE
لا يساوي
SHMEM_CMP_GT
أكبر من
SHMEM_CMP_LE
أصغر من أو يساوي
SHMEM_CMP_LT
أصغر من
SHMEM_CMP_GE
أكبر من أو يساوي
cmp_value
يجب أن يكون cmp_value من نوع صحيح. إذا كنت تستخدم C/C++، فيجب أن يتطابق نوع cmp_value مع النوع الضمني في قسم الملخص. إذا كنت تستخدم Fortran، فيجب أن يكون cmp_value صحيحًا بنفس الحجم والنوع مثل ivar. تعود روتينات shmem_wait عندما لم يعد ivar يساوي cmp_value. تعود روتينات shmem_wait_until عندما يكون شرط المقارنة صحيحًا. يتم تعريف شرط المقارنة بواسطة الوسيط ivar مقارنة بـ cmp_value باستخدام عامل المقارنة، cmp.

أمثلة

مثال 1: يعود الاستدعاء التالي عندما لا يساوي المتغير ivar القيمة 100:

INTEGER*8 IVAR
CALL SHMEM_INT8_WAIT(IVAR, INT8(100))

مثال 2: الاستدعاء التالي لـ SHMEM_INT8_WAIT_UNTIL يكافئ الاستدعاء لـ SHMEM_INT8_WAIT في المثال 1:

INTEGER*8 IVAR
CALL SHMEM_INT8_WAIT_UNTIL(IVAR, SHMEM_CMP_NE, INT8(100))

مثال 3: ينتظر الاستدعاء التالي لـ C/C++ حتى يتم تعيين بت الإشارة في ivar بواسطة نقل من PE بعيد:

int ivar;
shmem_int_wait_until(&ivar, SHMEM_CMP_LT, 0);

مثال 4: المثال التالي لـ Fortran في سياق روتين فرعي:

SUBROUTINE EXAMPLE()
  INTEGER FLAG_VAR
  COMMON/FLAG/FLAG_VAR
  . . .
  FLAG_VAR = FLAG_VALUE ! تهيئة متغير الحدث
  . . .
  IF (FLAG_VAR .EQ. FLAG_VALUE) THEN
    CALL SHMEM_WAIT(FLAG_VAR, FLAG_VALUE)
  ENDIF
  FLAG_VAR = FLAG_VALUE ! إعادة تعيين متغير الحدث للمرة القادمة
  . . .
END
