shmem_short_or_to_all(3)

الحزمة: openmpi

التراخيص: BSD-3-Clause AND LicenseRef-MPICH

SHMEM_SHORT_OR_TO_ALL(3) Open MPI SHMEM_SHORT_OR_TO_ALL(3)

shmem_int_or_to_all(3), shmem_int4_or_to_all(3), shmem_int8_or_to_all(3), shmem_long_or_to_all(3), shmem_longlong_or_to_all(3), shmem_short_or_to_all(3) - يقوم بإجراء عملية OR منطقية عبر مجموعة من عناصر المعالجة (PEs)

SYNOPSIS

C أو C++:

#include <mpp/shmem.h>
void shmem_int_or_to_all(int *target, const int *source,
  int nreduce, int PE_start, int logPE_stride, int PE_size,
  int *pWrk, long *pSync)
void shmem_long_or_to_all(long *target, const long *source,
  int nreduce, int PE_start, int logPE_stride, int PE_size,
  long *pWrk, long *pSync)
void shmem_longlong_or_to_all(long long *target,
  const long long *source, int nreduce, int PE_start, int logPE_stride,
  int PE_size, long long *pWrk, long *pSync)
void shmem_short_or_to_all(short *target, const short *source,
  int nreduce, int PE_start, int logPE_stride, int PE_size,
  short *pWrk, long *pSync)

Fortran:

INCLUDE "mpp/shmem.fh"
INTEGER pSync(SHMEM_REDUCE_SYNC_SIZE)
INTEGER nreduce, PE_start, logPE_stride, PE_size
CALL SHMEM_INT4_OR_TO_ALL(target, source, nreduce, PE_start,
& logPE_stride, PE_size, pWrk, pSync)
CALL SHMEM_INT8_OR_TO_ALL(target, source, nreduce, PE_start,
& logPE_stride, PE_size, pWrk, pSync)

الوصف

تقوم روتينات الذاكرة المشتركة (SHMEM) الاختزالية بحساب عملية اختزال واحدة أو أكثر عبر مصفوفات متماثلة على العديد من PEs الافتراضية. تقوم عملية الاختزال بإجراء عملية ثنائية ترابطية عبر مجموعة من القيم. للحصول على قائمة بروتينات اختزال SHMEM الأخرى، راجع intro_shmem(3).

كما هو الحال مع جميع روتينات SHMEM الجماعية، يفترض كل من هذه الروتينات أن PEs في المجموعة النشطة فقط هي التي تستدعي الروتين. إذا استدعى PE غير موجود في المجموعة النشطة روتين SHMEM جماعي، تحدث سلوكيات غير محددة.

يحدد الوسيط nreduce عدد الاختزالات المنفصلة التي سيتم إجراؤها. توفر مصفوفة المصدر على جميع PEs في المجموعة النشطة عنصرًا واحدًا لكل اختزال. يتم وضع نتائج الاختزالات في مصفوفة الهدف على جميع PEs في المجموعة النشطة. يتم تعريف المجموعة النشطة بواسطة ثلاثي PE_start، logPE_stride، PE_size.

قد تكون مصفوفتا المصدر والهدف نفس المصفوفة، لكن لا يمكن أن تكونا مصفوفتين متداخلتين.

الوسطاء هم كالتالي:

target
مصفوفة متماثلة بطول nreduce من العناصر لاستقبال نتائج عمليات الاختزال. يختلف نوع بيانات target مع إصدار روتين الاختزال الذي يتم استدعاؤه. عند الاستدعاء من C/C++، راجع قسم SYNOPSIS للحصول على معلومات نوع البيانات. عند الاستدعاء من Fortran، تكون أنواع بيانات target كالتالي:

shmem_int8_or_to_all عدد صحيح، بحجم عنصر 8 بايت.

shmem_int4_or_to_all عدد صحيح، بحجم عنصر 4 بايت.

source
مصفوفة متماثلة، بطول nreduce من العناصر، تحتوي على عنصر واحد لكل عملية اختزال منفصلة. يجب أن يكون المصدر بنفس نوع بيانات target.
nreduce
عدد العناصر في مصفوفتي target و source. يجب أن يكون nreduce من نوع عدد صحيح. إذا كنت تستخدم Fortran، فيجب أن يكون قيمة عدد صحيح افتراضية.
PE_start
أقل رقم PE افتراضي للمجموعة النشطة من PEs. يجب أن يكون PE_start من نوع عدد صحيح. إذا كنت تستخدم Fortran، فيجب أن يكون قيمة عدد صحيح افتراضية.
logPE_stride
لوغاريتم (الأساس 2) للخطوة بين أرقام PEs الافتراضية المتتالية في المجموعة النشطة. يجب أن يكون logPE_stride من نوع عدد صحيح. إذا كنت تستخدم Fortran، فيجب أن يكون قيمة عدد صحيح افتراضية.
PE_size
عدد PEs في المجموعة النشطة. يجب أن يكون PE_size من نوع عدد صحيح. إذا كنت تستخدم Fortran، فيجب أن يكون قيمة عدد صحيح افتراضية.
pWrk
مصفوفة عمل متماثلة. يجب أن يكون نوع بيانات الوسيط pWrk هو نفس نوع بيانات target. في C/C++، تحتوي على عناصر max(nreduce/2 + 1, _SHMEM_REDUCE_MIN_WRKDATA_SIZE). في Fortran، تحتوي على عناصر max(nreduce/2 + 1, SHMEM_REDUCE_MIN_WRKDATA_SIZE).
pSync
مصفوفة عمل متماثلة. في C/C++، pSync من نوع long وبحجم _SHMEM_REDUCE_SYNC_SIZE. في Fortran، pSync من نوع عدد صحيح وبحجم SHMEM_REDUCE_SYNC_SIZE. إذا كنت تستخدم Fortran، فيجب أن يكون قيمة عدد صحيح افتراضية. يجب تهيئة كل عنصر من هذه المصفوفة بالقيمة _SHMEM_SYNC_VALUE (في C/C++) أو SHMEM_SYNC_VALUE (في Fortran) قبل أن تدخل أي من PEs في المجموعة النشطة روتين الاختزال.

يجب أن تكون قيم الوسطاء nreduce، PE_start، logPE_stride، و PE_size متساوية على جميع PEs في المجموعة النشطة. يجب تمرير نفس مصفوفات target و source، ونفس مصفوفات العمل pWrk و pSync، إلى جميع PEs في المجموعة النشطة.

قبل أن يستدعي أي PE روتين اختزال، يجب عليك التأكد من وجود الشروط التالية (غالبًا ما يكون التزامن عبر حاجز أو طريقة أخرى ضروريًا لضمان ذلك): مصفوفات pWrk و pSync على جميع PEs في المجموعة النشطة ليست قيد الاستخدام من استدعاء سابق لروتين SHMEM جماعي. مصفوفة target على جميع PEs في المجموعة النشطة جاهزة لاستقبال نتائج الاختزال.

عند العودة من روتين الاختزال، تكون الأمور التالية صحيحة: تم تحديث مصفوفة target. تم استعادة القيم في مصفوفة pSync إلى القيم الأصلية.

ملاحظات

يتم تعريف المصطلحات جماعي، متماثل، ومتوافق مع ذاكرة التخزين المؤقت في intro_shmem(3). تقوم جميع روتينات اختزال SHMEM بإعادة تعيين القيم في pSync قبل إرجاعها، لذلك تحتاج مصفوفة pSync معينة فقط إلى تهيئتها في المرة الأولى التي يتم استخدامها فيها.

يجب عليك التأكد من أن مصفوفة pSync لا يتم تحديثها على أي PE في المجموعة النشطة أثناء معالجة أي من PEs لاستدعاء روتين اختزال SHMEM. كن حذرًا لتجنب هذه المواقف: إذا تم تهيئة مصفوفة pSync وقت التشغيل، يلزم نوع من التزامن لضمان أن جميع PEs في مجموعة العمل قد قاموا بتهيئة pSync قبل أن يدخل أي منهم في روتين SHMEM يتم استدعاؤه مع مصفوفة مزامنة pSync. يمكن إعادة استخدام مصفوفة pSync أو pWrk في استدعاء روتين اختزال لاحق فقط إذا لم تكن أي من PEs في المجموعة النشطة لا تزال تعالج استدعاء روتين اختزال سابق استخدم نفس مصفوفات pSync أو pWrk. بشكل عام، لا يمكن ضمان ذلك إلا عن طريق إجراء نوع من التزامن. ومع ذلك، في الحالة الخاصة لروتينات الاختزال التي يتم استدعاؤها بنفس المجموعة النشطة، يمكنك تخصيص مصفوفتي pSync و pWrk والتبديل بينهما في الاستدعاءات المتتالية.

أمثلة

المثال 1: يقوم مثال Fortran هذا بتهيئة مصفوفة pSync بشكل ثابت وإيجاد OR المنطقي للمتغير الصحيح FOO عبر جميع PEs الزوجية.

INCLUDE "mpp/shmem.fh"
INTEGER PSYNC(SHMEM_REDUCE_SYNC_SIZE)
DATA PSYNC /SHMEM_REDUCE_SYNC_SIZE*SHMEM_SYNC_VALUE/
PARAMETER (NR=1)
REAL PWRK(MAX(NR/2+1,SHMEM_REDUCE_MIN_WRKDATA_SIZE))
INTEGER FOO, FOOOR
COMMON /COM/ FOO, FOOOR, PWRK
INTRINSIC MY_PE
IF ( MOD(MY_PE(),2) .EQ. 0) THEN
  CALL SHMEM_INT8_OR_TO_ALL(FOOOR, FOO, NR, 0, 1, N$PES/2,
  & PWRK, PSYNC)
  PRINT *,'Result on PE ',MY_PE(),' is ',FOOOR
ENDIF

المثال 2: ضع في اعتبارك استدعاء C/C++ التالي:

shmem_int_or_to_all( target, source, 3, 0, 0, 8, pwrk, psync );

الاستدعاء السابق أكثر كفاءة، ولكنه مكافئ دلاليًا، لمجموع الاستدعاءات التالية:

shmem_int_or_to_all(&(target[0]), &(source[0]), 1, 0, 0, 8,
  pwrk1, psync1);
shmem_int_or_to_all(&(target[1]), &(source[1]), 1, 0, 0, 8,
  pwrk2, psync2);
shmem_int_or_to_all(&(target[2]), &(source[2]), 1, 0, 0, 8,
  pwrk1, psync1);

لاحظ أنه يتم استخدام مجموعتين من مصفوفات pWrk و pSync بالتناوب لأنه لا يتم إجراء تزامن بين الاستدعاءات.
