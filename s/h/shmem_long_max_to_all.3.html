أمثلة

مثال 1: هذا المثال بلغة Fortran يقوم بتهيئة مصفوفة pSync بشكل ثابت ويجد القيمة القصوى للمتغير الحقيقي FOO عبر جميع وحدات المعالجة (PEs) الزوجية.

INCLUDE "mpp/shmem.fh"
INTEGER PSYNC(SHMEM_REDUCE_SYNC_SIZE)
DATA PSYNC /SHMEM_REDUCE_SYNC_SIZE*SHMEM_SYNC_VALUE/
PARAMETER (NR=1)
REAL FOO, FOOMAX, PWRK(MAX(NR/2+1,SHMEM_REDUCE_MIN_WRKDATA_SIZE))
COMMON /COM/ FOO, FOOMAX, PWRK
INTRINSIC MY_PE
IF ( MOD(MY_PE(),2) .EQ. 0) THEN
  CALL SHMEM_REAL8_MAX_TO_ALL(FOOMAX, FOO, NR, 0, 1, N$PES/2,
  & PWRK, PSYNC)
  PRINT *, 'النتيجة على PE ', MY_PE(), ' هي ', FOOMAX
ENDIF

مثال 2: بالنظر إلى استدعاء C/C++ التالي:

shmem_int_max_to_all( target, source, 3, 0, 0, 8, pwrk, psync );

الاستدعاء السابق أكثر كفاءة، ولكنه مكافئ دلالياً، لمزيج الاستدعاءات التالية:

shmem_int_max_to_all(&(target[0]), &(source[0]), 1, 0, 0, 8,
  pwrk1, psync1);
shmem_int_max_to_all(&(target[1]), &(source[1]), 1, 0, 0, 8,
  pwrk2, psync2);
shmem_int_max_to_all(&(target[2]), &(source[2]), 1, 0, 0, 8,
  pwrk1, psync1);

لاحظ أنه يتم استخدام مجموعتين من مصفوفات pWrk و pSync بالتناوب لأنه لا تتم مزامنة بين الاستدعاءات.
