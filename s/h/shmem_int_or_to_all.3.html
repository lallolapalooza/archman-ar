shmem_int_or_to_all(3)

الحزمة: openmpi

التراخيص: BSD-3-Clause AND LicenseRef-MPICH

SHMEM_INT_OR_TO_ALL(3) Open MPI SHMEM_INT_OR_TO_ALL(3)

shmem_int_or_to_all(3)، shmem_int4_or_to_all(3)، shmem_int8_or_to_all(3)، shmem_long_or_to_all(3)، shmem_longlong_or_to_all(3)، shmem_short_or_to_all(3) - تنفيذ عملية تقليل لوظيفة OR البتية عبر مجموعة من عناصر المعالجة (PEs)

الملخص

بلغة C أو C++:

#include <mpp/shmem.h>
void shmem_int_or_to_all(int *target, const int *source,
  int nreduce, int PE_start, int logPE_stride, int PE_size,
  int *pWrk, long *pSync)
void shmem_long_or_to_all(long *target, const long *source,
  int nreduce, int PE_start, int logPE_stride, int PE_size,
  long *pWrk, long *pSync)
void shmem_longlong_or_to_all(long long *target,
  const long long *source, int nreduce, int PE_start, int logPE_stride,
  int PE_size, long long *pWrk, long *pSync)
void shmem_short_or_to_all(short *target, const short *source,
  int nreduce, int PE_start, int logPE_stride, int PE_size,
  short *pWrk, long *pSync)

بلغة Fortran:

INCLUDE "mpp/shmem.fh"
INTEGER pSync(SHMEM_REDUCE_SYNC_SIZE)
INTEGER nreduce, PE_start, logPE_stride, PE_size
CALL SHMEM_INT4_OR_TO_ALL(target, source, nreduce, PE_start,
& logPE_stride, PE_size, pWrk, pSync)
CALL SHMEM_INT8_OR_TO_ALL(target, source, nreduce, PE_start,
& logPE_stride, PE_size, pWrk, pSync)

الوصف

تقوم روتينات تقليل الذاكرة المشتركة (SHMEM) بحساب عملية تقليل واحدة أو أكثر عبر مصفوفات متناظرة على PEs افتراضية متعددة. تقوم عملية التقليل بتنفيذ عملية ثنائية ترابطية عبر مجموعة من القيم. للحصول على قائمة بروتينات تقليل SHMEM الأخرى، انظر intro_shmem(3).

كما هو الحال مع جميع روتينات SHMEM الجماعية، تفترض كل من هذه الروتينات أن PEs في المجموعة النشطة فقط هي التي تستدعي الروتين. إذا استدعى PE ليس في المجموعة النشطة روتين SHMEM جماعي، فإن النتيجة تكون غير محددة.

يحدد الوسيط nreduce عدد عمليات التقليل المنفصلة التي سيتم تنفيذها. يوفر مصفوفة المصدر على جميع PEs في المجموعة النشطة عنصرًا واحدًا لكل عملية تقليل. يتم وضع نتائج عمليات التقليل في مصفوفة الهدف على جميع PEs في المجموعة النشطة. يتم تعريف المجموعة النشطة بواسطة الثلاثي PE_start، logPE_stride، PE_size.

قد تكون مصفوفتي المصدر والهدف نفس المصفوفة، ولكنهما لا يمكن أن تكونا مصفوفتين متداخلتين.

الوسائط هي كما يلي:

target
مصفوفة متناظرة بطول nreduce عنصرًا لاستقبال نتائج عمليات التقليل. يختلف نوع بيانات target مع إصدار روتين التقليل الذي يتم استدعاؤه. عند الاستدعاء من C/C++، ارجع إلى قسم الملخص للحصول على معلومات نوع البيانات. عند الاستدعاء من Fortran، أنواع بيانات الهدف هي كما يلي:

shmem_int8_or_to_all عدد صحيح، بحجم عنصر 8 بايت.

shmem_int4_or_to_all عدد صحيح، بحجم عنصر 4 بايت.

source
مصفوفة متناظرة، بطول nreduce عنصرًا، تحتوي على عنصر واحد لكل عملية تقليل منفصلة. يجب أن يكون الوسيط source بنفس نوع بيانات target.
nreduce
عدد العناصر في مصفوفتي target و source. يجب أن يكون nreduce من نوع عدد صحيح. إذا كنت تستخدم Fortran، فيجب أن يكون قيمة عدد صحيح افتراضية.
PE_start
أدنى رقم PE افتراضي للمجموعة النشطة من PEs. يجب أن يكون PE_start من نوع عدد صحيح. إذا كنت تستخدم Fortran، فيجب أن يكون قيمة عدد صحيح افتراضية.
logPE_stride
اللوغاريتم (الأساس 2) للخطوة بين أرقام PE الافتراضية المتتالية في المجموعة النشطة. يجب أن يكون logPE_stride من نوع عدد صحيح. إذا كنت تستخدم Fortran، فيجب أن يكون قيمة عدد صحيح افتراضية.
PE_size
عدد PEs في المجموعة النشطة. يجب أن يكون PE_size من نوع عدد صحيح. إذا كنت تستخدم Fortran، فيجب أن يكون قيمة عدد صحيح افتراضية.
pWrk
مصفوفة عمل متناظرة. يجب أن يكون الوسيط pWrk بنفس نوع بيانات target. في C/C++، تحتوي على max(nreduce/2 + 1, _SHMEM_REDUCE_MIN_WRKDATA_SIZE) عناصر. في Fortran، تحتوي على max(nreduce/2 + 1, SHMEM_REDUCE_MIN_WRKDATA_SIZE) عناصر.
pSync
مصفوفة عمل متناظرة. في C/C++، pSync من نوع long وبحجم _SHMEM_REDUCE_SYNC_SIZE. في Fortran، pSync من نوع integer وبحجم SHMEM_REDUCE_SYNC_SIZE. إذا كنت تستخدم Fortran، فيجب أن يكون قيمة عدد صحيح افتراضية. يجب تهيئة كل عنصر من هذه المصفوفة بالقيمة _SHMEM_SYNC_VALUE (في C/C++) أو SHMEM_SYNC_VALUE (في Fortran) قبل أن تدخل أي من PEs في المجموعة النشطة إلى روتين التقليل.

يجب أن تكون قيم الوسائط nreduce و PE_start و logPE_stride و PE_size متساوية على جميع PEs في المجموعة النشطة. يجب تمرير نفس مصفوفتي target و source، ونفس مصفوفتي العمل pWrk و pSync، إلى جميع PEs في المجموعة النشطة.

قبل أن يستدعي أي PE روتين تقليل، يجب عليك التأكد من وجود الشروط التالية (غالبًا ما تكون هناك حاجة إلى المزامنة عن طريق حاجز أو طريقة أخرى لضمان ذلك): مصفوفتي pWrk و pSync على جميع PEs في المجموعة النشطة ليست قيد الاستخدام من استدعاء سابق لروتين SHMEM جماعي. مصفوفة target على جميع PEs في المجموعة النشطة جاهزة لاستقبال نتائج التقليل.

عند العودة من روتين التقليل، تكون الأمور التالية صحيحة: تم تحديث مصفوفة target. تم استعادة القيم الموجودة في مصفوفة pSync إلى القيم الأصلية.

ملاحظات

المصطلحات جماعي (collective)، متناظر (symmetric)، ومحاذي للكاش (cache aligned) معرفة في intro_shmem(3). تقوم جميع روتينات تقليل SHMEM بإعادة تعيين القيم في pSync قبل أن تعود، لذلك تحتاج مصفوفة pSync معينة فقط إلى التهيئة في المرة الأولى التي يتم استخدامها فيها.

يجب عليك التأكد من أن مصفوفة pSync لا يتم تحديثها على أي PE في المجموعة النشطة بينما تشارك أي من PEs في معالجة روتين تقليل SHMEM. كن حذرًا لتجنب هذه المواقف: إذا تم تهيئة مصفوفة pSync في وقت التشغيل، فستكون هناك حاجة إلى نوع من المزامنة لضمان أن جميع PEs في مجموعة العمل قد قامت بتهيئة pSync قبل أن تدخل أي منها إلى روتين SHMEM تم استدعاؤه باستخدام مصفوفة مزامنة pSync. يمكن إعادة استخدام مصفوفة pSync أو pWrk في استدعاء روتين تقليل لاحق فقط إذا لم تكن أي من PEs في المجموعة النشطة لا تزال تعالج استدعاء روتين تقليل سابق استخدم نفس مصفوفات pSync أو pWrk. بشكل عام، لا يمكن ضمان ذلك إلا عن طريق القيام بنوع من المزامنة. ومع ذلك، في الحالة الخاصة لروتينات التقليل التي يتم استدعاؤها بنفس المجموعة النشطة، يمكنك تخصيص مصفوفتين pSync و pWrk والتبديل بينهما في الاستدعاءات المتتالية.

أمثلة

مثال 1: يقوم هذا المثال في Fortran بتهيئة مصفوفة pSync بشكل ثابت ويجد OR المنطقي للمتغير الصحيح FOO عبر جميع PEs الزوجية.

INCLUDE "mpp/shmem.fh"
INTEGER PSYNC(SHMEM_REDUCE_SYNC_SIZE)
DATA PSYNC /SHMEM_REDUCE_SYNC_SIZE*SHMEM_SYNC_VALUE/
PARAMETER (NR=1)
REAL PWRK(MAX(NR/2+1,SHMEM_REDUCE_MIN_WRKDATA_SIZE))
INTEGER FOO, FOOOR
COMMON /COM/ FOO, FOOOR, PWRK
INTRINSIC MY_PE
IF ( MOD(MY_PE(),2) .EQ. 0) THEN
  CALL SHMEM_INT8_OR_TO_ALL(FOOOR, FOO, NR, 0, 1, N$PES/2,
  & PWRK, PSYNC)
  PRINT *,'Result on PE ',MY_PE(),' is ',FOOOR
ENDIF

مثال 2: انظر استدعاء C/C++ التالي:

shmem_int_or_to_all( target, source, 3, 0, 0, 8, pwrk, psync );

الاستدعاء السابق أكثر كفاءة، ولكنه مكافئ دلاليًا، لمزيج الاستدعاءات التالية:

shmem_int_or_to_all(&(target[0]), &(source[0]), 1, 0, 0, 8,
  pwrk1, psync1);
shmem_int_or_to_all(&(target[1]), &(source[1]), 1, 0, 0, 8,
  pwrk2, psync2);
shmem_int_or_to_all(&(target[2]), &(source[2]), 1, 0, 0, 8,
  pwrk1, psync1);

لاحظ أنه يتم استخدام مجموعتين من مصفوفات pWrk و pSync بالتناوب لأنه لا يتم إجراء مزامنة بين الاستدعاءات.
