shmem_longlong_and_to_all(3)

الحزمة: openmpi

التراخيص: BSD-3-Clause AND LicenseRef-MPICH

SHMEM_LONGLONG_AND_TO_ALL(3) Open MPI SHMEM_LONGLONG_AND_TO_ALL(3)

shmem_int_and_to_all(3)، shmem_int4_and_to_all(3)، shmem_int8_and_to_all(3)، shmem_long_and_to_all(3)، shmem_longlong_and_to_all(3)، shmem_short_and_to_all(3) - تنفيذ عملية AND المنطقية على مصفوفات متماثلة عبر مجموعة PEs النشطة.

الملخص

C أو C++:

#include <mpp/shmem.h>
void shmem_int_and_to_all(int *target, const int *source,
  int nreduce, int PE_start, int logPE_stride, int PE_size,
  int *pWrk, long *pSync)
void shmem_long_and_to_all(long *target, const long *source,
  int nreduce, int PE_start, int  logPE_stride, int PE_size,
  long *pWrk, long *pSync)
void shmem_longlong_and_to_all(long long *target,
  const long long *source, int nreduce, int PE_start, int logPE_stride,
  int PE_size, long long *pWrk, long *pSync)
void shmem_short_and_to_all(short *target, const short *source,
  int nreduce, int PE_start, int logPE_stride, int PE_size,
  short *pWrk, long *pSync)

Fortran:

INCLUDE "mpp/shmem.fh"
INTEGER pSync(SHMEM_REDUCE_SYNC_SIZE)
INTEGER nreduce, PE_start, logPE_stride, PE_size
CALL SHMEM_INT4_AND_TO_ALL(target, source, nreduce,
& PE_start, logPE_stride, PE_size, pWrk, pSync)
CALL SHMEM_INT8_AND_TO_ALL(target, source, nreduce,
& PE_start, logPE_stride, PE_size, pWrk, pSync)

الوصف

تقوم روتينات الذاكرة المشتركة (SHMEM) الحسابية بإجراء عمليات حسابية واحدة أو أكثر عبر مصفوفات متماثلة على PEs افتراضية متعددة. يقوم الحساب بإجراء عملية ثنائية ترابطية عبر مجموعة من القيم. للحصول على قائمة بروتينات SHMEM الحسابية الأخرى، راجع intro_shmem(3).

يحدد الوسيط nreduce عدد العمليات الحسابية المنفصلة التي سيتم إجراؤها. توفر مصفوفة المصدر على جميع PEs في المجموعة النشطة عنصرًا واحدًا لكل عملية حسابية. يتم وضع نتائج العمليات الحسابية في مصفوفة الهدف على جميع PEs في المجموعة النشطة. يتم تعريف المجموعة النشطة بواسطة الثلاثي PE_start، logPE_stride، PE_size.

قد تكون مصفوفات المصدر والهدف نفس المصفوفة، ولكن لا يجوز أن تكون مصفوفات متداخلة. كما هو الحال مع جميع روتينات SHMEM الجماعية، يفترض كل من هذه الروتينات أن PEs الموجودة في المجموعة النشطة فقط هي التي تستدعي الروتين. إذا استدعت PE غير موجودة في المجموعة النشطة روتين SHMEM جماعي، فستكون النتيجة غير محددة.

الوسائط هي كما يلي:

target
مصفوفة متماثلة، بطول nreduce عنصر، لتلقي نتيجة عمليات الحساب. يختلف نوع بيانات target مع إصدار روتين الحساب الذي يتم استدعاؤه. عند الاستدعاء من C/C++، راجع قسم الملخص للحصول على معلومات نوع البيانات. عند الاستدعاء من Fortran، أنواع بيانات الهدف هي كما يلي:

shmem_int8_and_to_all: عدد صحيح، بحجم عنصر 8 بايت

shmem_int4_and_to_all: عدد صحيح، بحجم عنصر 4 بايت

source
مصفوفة متماثلة، بطول nreduce عنصر، تحتوي على عنصر واحد لكل عملية حسابية منفصلة. يجب أن يكون الوسيط source من نفس نوع بيانات target.
nreduce
عدد العناصر في مصفوفات target و source. يجب أن يكون nreduce من نوع عدد صحيح. إذا كنت تستخدم Fortran، فيجب أن تكون قيمة عدد صحيح افتراضية.
PE_start
أدنى رقم PE افتراضي للمجموعة النشطة من PEs. يجب أن يكون PE_start من نوع عدد صحيح. إذا كنت تستخدم Fortran، فيجب أن تكون قيمة عدد صحيح افتراضية.
logPE_stride
لوغاريتم (الأساس 2) للخطوة بين أرقام PE الافتراضية المتتالية في المجموعة النشطة. يجب أن يكون logPE_stride من نوع عدد صحيح. إذا كنت تستخدم Fortran، فيجب أن تكون قيمة عدد صحيح افتراضية.
PE_size
عدد PEs في المجموعة النشطة. يجب أن يكون PE_size من نوع عدد صحيح. إذا كنت تستخدم Fortran، فيجب أن تكون قيمة عدد صحيح افتراضية.
pWrk
مصفوفة عمل متماثلة. يجب أن يكون الوسيط pWrk من نفس نوع بيانات target. في C/C++، تحتوي على max(nreduce/2 + 1, _SHMEM_REDUCE_MIN_WRKDATA_SIZE) عنصرًا. في Fortran، تحتوي على max(nreduce/2 + 1, SHMEM_REDUCE_MIN_WRKDATA_SIZE) عنصرًا.
pSync
مصفوفة عمل متماثلة. في C/C++، يجب أن يكون pSync من نوع long وحجم _SHMEM_REDUCE_SYNC_SIZE. في Fortran، يجب أن يكون pSync من نوع عدد صحيح وحجم SHMEM_REDUCE_SYNC_SIZE. إذا كنت تستخدم Fortran، فيجب أن تكون قيمة عدد صحيح افتراضية. يجب تهيئة كل عنصر في هذه المصفوفة بالقيمة _SHMEM_SYNC_VALUE (في C/C++) أو SHMEM_SYNC_VALUE (في Fortran) قبل أن تدخل أي من PEs في المجموعة النشطة روتين الحساب.

يجب أن تكون قيم الوسائط nreduce، PE_start، logPE_stride، و PE_size متساوية على جميع PEs في المجموعة النشطة. يجب تمرير نفس مصفوفات target و source، ونفس مصفوفات العمل pWrk و pSync، إلى جميع PEs في المجموعة النشطة.

قبل أن يستدعي أي PE روتين حساب، يجب عليك التأكد من وجود الشروط التالية (غالبًا ما يلزم المزامنة عبر حاجز أو طريقة أخرى لضمان ذلك): مصفوفات pWrk و pSync على جميع PEs في المجموعة النشطة لم يتم استخدامها بعد من استدعاء سابق لروتين SHMEM جماعي. مصفوفة target على جميع PEs في المجموعة النشطة جاهزة لتلقي نتائج الحساب.

بعد العودة من روتين حساب، تكون الشروط التالية صحيحة بالنسبة للـ PE المحلي: تم تحديث مصفوفة target. تم استعادة القيم في مصفوفة pSync إلى القيم الأصلية.

ملاحظات

يتم تعريف المصطلحات جماعي، متماثل، ومتوافق مع الذاكرة المؤقتة في intro_shmem(3). جميع روتينات SHMEM الحسابية تعيد تعيين القيم في pSync قبل أن تعود، لذلك لا يلزم تهيئة مخزن pSync معين إلا في المرة الأولى التي يتم استخدامه فيها.

يجب عليك التأكد من أن مصفوفة pSync لا يتم تحديثها على أي PE في المجموعة النشطة بينما تشارك أي من PEs في معالجة روتين SHMEM حسابي. كن حذرًا لتجنب المواقف التالية: إذا تم تهيئة مصفوفة pSync في وقت التشغيل، فستحتاج بعض أنواع المزامنة لضمان أن جميع PEs في مجموعة العمل قد قامت بتهيئة pSync قبل أن يدخل أي منها روتين SHMEM يتم استدعاؤه مع مصفوفة مزامنة pSync. يمكن إعادة استخدام مصفوفة pSync أو pWrk في استدعاء روتين حساب لاحق فقط إذا لم تكن أي من PEs في المجموعة النشطة لا تزال تعالج روتين حساب سابق استخدم نفس مصفوفات pSync أو pWrk. بشكل عام، لا يمكن ضمان ذلك إلا عن طريق القيام بنوع ما من المزامنة. ومع ذلك، في الحالة الخاصة لروتينات الحساب التي يتم استدعاؤها مع نفس المجموعة النشطة، يمكنك تخصيص مصفوفتين pSync و pWrk والتبديل بينهما في استدعاءات متتالية.

أمثلة

المثال 1: يقوم مثال Fortran هذا بتهيئة مصفوفة pSync بشكل ثابت ويجد AND المنطقي للمتغير الصحيح FOO عبر جميع PEs الزوجية.

INCLUDE "mpp/shmem.fh"
INTEGER PSYNC(SHMEM_REDUCE_SYNC_SIZE)
DATA PSYNC /SHMEM_REDUCE_SYNC_SIZE*SHMEM_SYNC_VALUE/
PARAMETER (NR=1)
REAL PWRK(MAX(NR/2+1, SHMEM_REDUCE_MIN_WRKDATA_SIZE))
INTEGER FOO, FOOAND
COMMON /COM/ FOO, FOOAND, PWRK
INTRINSIC MY_PE
IF ( MOD(MY_PE(),2) .EQ. 0) THEN
  CALL SHMEM_INT8_AND_TO_ALL(FOOAND, FOO, NR, 0, 1, N$PES/2,
  & PWRK, PSYNC)
  PRINT *, 'Result on PE ', MY_PE(), ' is ', FOOAND
ENDIF

المثال 2: انظر استدعاء C التالي:

shmem_int_and_to_all( target, source, 3, 0, 0, 8, pwrk, psync );

الاستدعاء السابق أكثر كفاءة، ولكنه يعادل دلاليًا، مجموعة الاستدعاءات التالية:

shmem_int_and_to_all(&(target[0]), &(source[0]), 1, 0, 0, 8,
  pwrk1, psync1);
shmem_int_and_to_all(&(target[1]), &(source[1]), 1, 0, 0, 8,
  pwrk2, psync2);
shmem_int_and_to_all(&(target[2]), &(source[2]), 1, 0, 0, 8,
  pwrk1, psync1);

لاحظ أنه تم استخدام مجموعتين من مصفوفات pWrk و pSync بالتناوب لأنه لم يتم إجراء أي مزامنة بين الاستدعاءات.
