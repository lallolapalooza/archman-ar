shmem_longlong_iget(3)

shmem_longlong_iget - نقل عنصر واحد طويل جدًا من ذاكرة مشتركة إلى ذاكرة محلية.

الملخص

#include <mpe.h>

long long shmem_longlong_iget(const long long *source, size_t source_index,
                             long long *dest, size_t dest_index, size_t nelems,
                             int PE);

الوصف

دالة shmem_longlong_iget تنقل nelems عنصرًا من نوع long long من ذاكرة مشتركة بعيدة إلى ذاكرة محلية.

  • source: مؤشر إلى بداية مصفوفة long long في الذاكرة المشتركة البعيدة.
  • source_index: فهرس البداية في المصفوفة البعيدة.
  • dest: مؤشر إلى بداية مصفوفة long long في الذاكرة المحلية.
  • dest_index: فهرس البداية في المصفوفة المحلية.
  • nelems: عدد العناصر المراد نقلها.
  • PE: رقم المعالج (PE) الذي يتم استرداد البيانات منه.

تُرجع هذه الدالة القيمة 0 عند النجاح.

ملاحظات

هذه الدالة غير متزامنة، مما يعني أنها تبدأ عملية النقل وتعود فورًا. يجب استخدام shmem_quiet() لضمان اكتمال عملية النقل قبل استخدام البيانات المستلمة.

أمثلة

long long remote_array[100];
long long local_array[100];

// قم بتهيئة الذاكرة المشتركة ...

// جلب عنصر واحد من المعالج 1 إلى المعالج 0
shmem_longlong_iget(&remote_array, 10, &local_array, 20, 1, 1);
shmem_quiet(); // انتظر حتى يكتمل النقل
