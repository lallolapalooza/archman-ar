shsein(3)

الحزمة: lapack-doc

التراخيص: custom

NAME

shsein - يستعلم عن القيم الذاتية للمصفوفات الهيرميتية وشبه الهيرميتية.

SYNOPSIS

 subroutine shsein( uplo, job, eigvals,
              a, lda,
              b, ldb,
              ldv,
              work,
              rwork,
              n,
              na,
              nb,
              info )
 integer :: lda, ldb, ldv, n, na, nb, info
 real(kind=8) :: a( lda, * ), b( ldb, * ), work( * ), rwork( * )
 character(len=1) :: uplo, job, eigvals

Function/Subroutine Documentation

!> @brief
!>
!> uplo : CHARACTER*1
!>           = 'U':  A(upper) and B(upper) are the upper triangular parts
!>                 of A and B, respectively.
!>           = 'L':  A(lower) and B(lower) are the lower triangular parts
!>                 of A and B, respectively.
!>
!> job : CHARACTER*1
!>           Specifies the computation to be performed:
!>           = 'N':  Compute the eigenvalues only.
!>           = 'V':  Compute the eigenvalues and the corresponding right eigenvectors.
!>           = 'L':  Compute the eigenvalues and the corresponding left eigenvectors.
!>           = 'B':  Compute the eigenvalues and the corresponding left and
!>                 right eigenvectors.
!>
!> eigvals : CHARACTER*1
!>           Specifies the ordering of the eigenvalues:
!>           = 'S':  The eigenvalues are sorted in ascending order.
!>           = 'N':  The eigenvalues are not sorted.
!>
!> a : COMPLEX*16, array, Intent(INOUT)
!>           The n-by-n matrix A. On exit, A is overwritten by the
!>           elements of its Schur form.
!>
!> lda : INTEGER
!>           The leading dimension of the array A.  lda >= max(1,n).
!>
!> b : COMPLEX*16, array, Intent(INOUT)
!>           The n-by-n matrix B. On exit, B is overwritten by the
!>           elements of its Schur form.
!>
!> ldb : INTEGER
!>           The leading dimension of the array B.  ldb >= max(1,n).
!>
!> ldv : INTEGER
!>           The leading dimension of the array v.
!>
!> v : COMPLEX*16, array, Intent(OUT)
!>           If job = 'V', v contains the right eigenvectors of A and B.
!>           The column v(i) is the eigenvector corresponding to the eigenvalue
!>           w(i).
!>           If job = 'B', v contains the right eigenvectors of A and B.
!>           The column v(i) is the eigenvector corresponding to the eigenvalue
!>           w(i).
!>           ldv >= max(1,n).
!>
!> work : COMPLEX*16, array, Dimension(lrwork)
!>           Complex workspace array.
!>
!> rwork : DOUBLE PRECISION, array, Dimension(lrwork)
!>           Double precision workspace array.
!>
!> lrwork : INTEGER
!>           The dimension of the arrays work and rwork.
!>           lrwork >= max(1,2*n).
!>
!> n : INTEGER
!>           The order of the matrices A and B.
!>
!> na : INTEGER
!>           The number of eigenvalues of A.
!>
!> nb : INTEGER
!>           The number of eigenvalues of B.
!>
!> info : INTEGER
!>           = 0:  successful exit
!>           < 0:  if info = -i, then the i-th argument had an illegal value.
!>           > 0:  if INFO = i, i is the number of eigenvectors which
!>                failed to converge; see IFAILL and IFAILR for further
!>                details.
!>
