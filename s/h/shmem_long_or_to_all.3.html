shmem_long_or_to_all(3)

الحزمة: openmpi

التراخيص: BSD-3-Clause AND LicenseRef-MPICH

SHMEM_LONG_OR_TO_ALL(3) Open MPI SHMEM_LONG_OR_TO_ALL(3)

shmem_int_or_to_all(3)، shmem_int4_or_to_all(3)، shmem_int8_or_to_all(3)، shmem_long_or_to_all(3)، shmem_longlong_or_to_all(3)، shmem_short_or_to_all(3) - يقوم بوظيفة OR المنطقية عبر مجموعة من عناصر المعالجة (PEs)

الملخص

C أو C++:

#include <mpp/shmem.h>
void shmem_int_or_to_all(int *target, const int *source,
  int nreduce, int PE_start, int logPE_stride, int PE_size,
  int *pWrk, long *pSync)
void shmem_long_or_to_all(long *target, const long *source,
  int nreduce, int PE_start, int logPE_stride, int PE_size,
  long *pWrk, long *pSync)
void shmem_longlong_or_to_all(long long *target,
  const long long *source, int nreduce, int PE_start, int logPE_stride,
  int PE_size, long long *pWrk, long *pSync)
void shmem_short_or_to_all(short *target, const short *source,
  int nreduce, int PE_start, int logPE_stride, int PE_size,
  short *pWrk, long *pSync)

Fortran:

INCLUDE "mpp/shmem.fh"
INTEGER pSync(SHMEM_REDUCE_SYNC_SIZE)
INTEGER nreduce, PE_start, logPE_stride, PE_size
CALL SHMEM_INT4_OR_TO_ALL(target, source, nreduce, PE_start,
& logPE_stride, PE_size, pWrk, pSync)
CALL SHMEM_INT8_OR_TO_ALL(target, source, nreduce, PE_start,
& logPE_stride, PE_size, pWrk, pSync)

الوصف

تقوم إجراءات تقليل الذاكرة المشتركة (SHMEM) بحساب عملية تقليل واحدة أو أكثر عبر المصفوفات المتماثلة على وحدات معالجة افتراضية متعددة. تقوم عملية التقليل بتنفيذ عملية ثنائية ترابطية عبر مجموعة من القيم. للحصول على قائمة بإجراءات تقليل SHMEM الأخرى، راجع intro_shmem(3).

كما هو الحال مع جميع إجراءات SHMEM الجماعية، تفترض كل من هذه الإجراءات أن وحدات المعالجة في المجموعة النشطة فقط هي التي تستدعي الإجراء. إذا استدعت وحدة معالجة غير موجودة في المجموعة النشطة إجراءً جماعيًا من SHMEM، فإن السلوك يكون غير محدد.

يحدد وسيط nreduce عدد عمليات التقليل المنفصلة المراد تنفيذها. توفر مصفوفة المصدر في جميع وحدات المعالجة في المجموعة النشطة عنصرًا واحدًا لكل عملية تقليل. يتم وضع نتائج عمليات التقليل في مصفوفة الهدف في جميع وحدات المعالجة في المجموعة النشطة. يتم تحديد المجموعة النشطة بواسطة الثلاثي PE_start، logPE_stride، PE_size.

قد تكون مصفوفتا المصدر والهدف نفس المصفوفة، ولكن لا يجوز أن تكونا مصفوفتين متداخلتين.

الوسائط هي كما يلي:

target
مصفوفة متماثلة بطول nreduce عنصر لاستقبال نتائج عمليات التقليل. يختلف نوع بيانات target مع إصدار إجراء التقليل الذي يتم استدعاؤه. عند الاستدعاء من C/C++، ارجع إلى قسم الملخص للحصول على معلومات نوع البيانات. عند الاستدعاء من Fortran، أنواع بيانات الهدف هي كما يلي:

shmem_int8_or_to_all عدد صحيح، بحجم عنصر 8 بايت.

shmem_int4_or_to_all عدد صحيح، بحجم عنصر 4 بايت.

source
مصفوفة متماثلة، بطول nreduce عنصر، تحتوي على عنصر واحد لكل عملية تقليل منفصلة. يجب أن يكون وسيط المصدر بنفس نوع بيانات target.
nreduce
عدد العناصر في مصفوفتي target و source. يجب أن يكون nreduce من نوع عدد صحيح. إذا كنت تستخدم Fortran، فيجب أن يكون قيمة عدد صحيح افتراضية.
PE_start
أقل رقم PE افتراضي للمجموعة النشطة من PEs. يجب أن يكون PE_start من نوع عدد صحيح. إذا كنت تستخدم Fortran، فيجب أن يكون قيمة عدد صحيح افتراضية.
logPE_stride
اللوغاريتم (الأساس 2) للمسافة بين أرقام PEs الافتراضية المتتالية في المجموعة النشطة. يجب أن يكون logPE_stride من نوع عدد صحيح. إذا كنت تستخدم Fortran، فيجب أن يكون قيمة عدد صحيح افتراضية.
PE_size
عدد PEs في المجموعة النشطة. يجب أن يكون PE_size من نوع عدد صحيح. إذا كنت تستخدم Fortran، فيجب أن يكون قيمة عدد صحيح افتراضية.
pWrk
مصفوفة عمل متماثلة. يجب أن يكون لوسيط pWrk نفس نوع بيانات target. في C/C++، يحتوي هذا على max(nreduce/2 + 1, _SHMEM_REDUCE_MIN_WRKDATA_SIZE) عنصر. في Fortran، يحتوي هذا على max(nreduce/2 + 1, SHMEM_REDUCE_MIN_WRKDATA_SIZE) عنصر.
pSync
مصفوفة عمل متماثلة. في C/C++، يكون pSync من نوع long وحجم _SHMEM_REDUCE_SYNC_SIZE. في Fortran، يكون pSync من نوع عدد صحيح وحجم SHMEM_REDUCE_SYNC_SIZE. إذا كنت تستخدم Fortran، فيجب أن يكون قيمة عدد صحيح افتراضية. يجب تهيئة كل عنصر في هذه المصفوفة بالقيمة _SHMEM_SYNC_VALUE (في C/C++) أو SHMEM_SYNC_VALUE (في Fortran) قبل أن تدخل أي من PEs في المجموعة النشطة إلى إجراء التقليل.

يجب أن تكون قيم الوسائط nreduce و PE_start و logPE_stride و PE_size متساوية في جميع PEs في المجموعة النشطة. يجب تمرير نفس مصفوفتي target و source، ونفس مصفوفتي العمل pWrk و pSync، إلى جميع PEs في المجموعة النشطة.

قبل أن تقوم أي PE باستدعاء إجراء تقليل، يجب عليك التأكد من وجود الشروط التالية (غالبًا ما تكون هناك حاجة إلى مزامنة عبر حاجز أو طريقة أخرى لضمان ذلك): مصفوفات pWrk و pSync في جميع PEs في المجموعة النشطة لا تزال قيد الاستخدام من استدعاء سابق لإجراء SHMEM جماعي. مصفوفة target في جميع PEs في المجموعة النشطة جاهزة لاستقبال نتائج التقليل.

عند العودة من إجراء التقليل، تكون الأمور التالية صحيحة: تم تحديث مصفوفة target. تمت استعادة القيم في مصفوفة pSync إلى القيم الأصلية.

ملاحظات

تم تعريف المصطلحات جماعي، متماثل، ومتوافق مع ذاكرة التخزين المؤقت في intro_shmem(3). تقوم جميع إجراءات تقليل SHMEM بإعادة تعيين القيم في pSync قبل إرجاعها، لذا يكفي تهيئة مخزن pSync معين مرة واحدة فقط عند استخدامه لأول مرة.

يجب عليك التأكد من أن مصفوفة pSync لا يتم تحديثها على أي PE في المجموعة النشطة أثناء مشاركة أي من PEs في معالجة إجراء تقليل SHMEM. كن حذرًا لتجنب هذه المواقف: إذا تم تهيئة مصفوفة pSync وقت التشغيل، فستكون هناك حاجة إلى نوع من المزامنة لضمان أن جميع PEs في مجموعة العمل قد قامت بتهيئة pSync قبل أن تدخل أي منها إلى إجراء SHMEM يتم استدعاؤه باستخدام مصفوفة المزامنة pSync. يمكن إعادة استخدام مصفوفة pSync أو pWrk في استدعاء إجراء تقليل لاحق فقط إذا لم تكن أي من PEs في المجموعة النشطة لا تزال تعالج استدعاء إجراء تقليل سابق استخدم نفس مصفوفات pSync أو pWrk. بشكل عام، لا يمكن ضمان ذلك إلا من خلال القيام بنوع من المزامنة. ومع ذلك، في الحالة الخاصة لإجراءات التقليل التي يتم استدعاؤها بنفس المجموعة النشطة، يمكنك تخصيص مصفوفتي pSync و pWrk والتبديل بينهما في استدعاءات متتالية.

أمثلة

المثال 1: يقوم مثال Fortran هذا بتهيئة مصفوفة pSync بشكل ثابت ويجد OR المنطقي للمتغير الصحيح FOO عبر جميع PEs الزوجية.

INCLUDE "mpp/shmem.fh"
INTEGER PSYNC(SHMEM_REDUCE_SYNC_SIZE)
DATA PSYNC /SHMEM_REDUCE_SYNC_SIZE*SHMEM_SYNC_VALUE/
PARAMETER (NR=1)
REAL PWRK(MAX(NR/2+1,SHMEM_REDUCE_MIN_WRKDATA_SIZE))
INTEGER FOO, FOOOR
COMMON /COM/ FOO, FOOOR, PWRK
INTRINSIC MY_PE
IF ( MOD(MY_PE(),2) .EQ. 0) THEN
  CALL SHMEM_INT8_OR_TO_ALL(FOOOR, FOO, NR, 0, 1, N$PES/2,
  & PWRK, PSYNC)
  PRINT *,'Result on PE ',MY_PE(),' is ',FOOOR
ENDIF

المثال 2: انظر إلى استدعاء C/C++ التالي:

shmem_int_or_to_all( target, source, 3, 0, 0, 8, pwrk, psync );

الاستدعاء السابق أكثر كفاءة، ولكنه مكافئ دلاليًا، لدمج الاستدعاءات التالية:

shmem_int_or_to_all(&(target[0]), &(source[0]), 1, 0, 0, 8,
  pwrk1, psync1);
shmem_int_or_to_all(&(target[1]), &(source[1]), 1, 0, 0, 8,
  pwrk2, psync2);
shmem_int_or_to_all(&(target[2]), &(source[2]), 1, 0, 0, 8,
  pwrk1, psync1);

لاحظ أنه يتم استخدام مجموعتين من مصفوفتي pWrk و pSync بالتناوب لأنه لا تتم مزامنة بين الاستدعاءات.
