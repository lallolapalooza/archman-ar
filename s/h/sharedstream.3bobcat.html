EXAMPLE

#include <iostream>\n#include "bobcat/sharedstream"\n\nusing namespace std;\nusing namespace bobcat;\n\nint main(int argc, char **argv)\n{\n    // Create a shared stream.\n    sharedstream sharedStream(1024);\n\n    // Simulate a shared buffer.\n    char buffer[1024];\n    sharedStream.init(buffer, 1024);\n\n    char ch;\n    while (cin >> ch)\n    {\n        switch (ch)\n        {\n            case ’r’:\n            {\n                sharedStream.seekg(0);\n                cout << \"Reset stream position to 0.\\n\";\n            }\n            break;\n            case ’w’:\n            {\n                sharedStream.seekp(0, ios::end);\n                streampos pos = sharedStream.tellp();\n                string line;\n                cout << \"Enter line to write (max 1024 chars): \";\n                getline(cin, line);\n                cout << \"Writing \" << line.length() << \" bytes\\n\";\n                sharedStream.write(line.data(), line.length());\n                if (!sharedStream)\n                    cout << \"No room left to write any bytes\\n\";\n                else\n                    cout << \"Wrote \" << (sharedStream.tellp() - pos) << \" bytes\\n\";\n            }\n            break;\n            case ’q’:\n            return 0;\n            default:\n                cout << \"request not implemented: \" << ch << ’\\n’;\n            break;\n        }\n    }\n}\n

BUGS

\n

Note that by default exceptions thrown from within a\n std::stream object are caught by the stream object, setting its\n ios::failbit flag. To allow exceptions to leave a stream object, its\n exceptions member can be called, e.g., using:

\n
    myStream.exceptions(ios::failbit | ios::badbit | ios::eofbit);\n
