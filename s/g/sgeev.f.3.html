subroutine sgeev (character jobvl, character jobvr, integer n, real, dimension( lda, * ) a, integer lda, real, dimension( * ) wr, real, dimension( * ) wi, real, dimension( ldvl, * ) vl, integer ldvl, real, dimension( ldvr, * ) vr, integer ldvr, real, dimension( * ) work, integer lwork, integer info)

SGEEV تحسب القيم الذاتية، واختياريًا، المتجهات الذاتية اليسرى و/أو اليمنى لمصفوفات GE

الغرض:

!>
!> SGEEV تحسب لمصفوفة حقيقية غير متماثلة N-by-N A،
!> القيم الذاتية، واختياريًا، المتجهات الذاتية اليسرى و/أو اليمنى.
!>
!> المتجه الذاتي الأيمن v(j) للمصفوفة A يحقق
!>                  A * v(j) = lambda(j) * v(j)
!> حيث lambda(j) هي القيمة الذاتية الخاصة بها.
!> المتجه الذاتي الأيسر u(j) للمصفوفة A يحقق
!>               u(j)**H * A = lambda(j) * u(j)**H
!> حيث u(j)**H ترمز إلى المترافق المصاحب للمتجه u(j).
!>
!> يتم تسوية المتجهات الذاتية المحسوبة ليكون لها معيار إقليدي
!> يساوي 1 وأكبر مكون حقيقي.
!>

المعاملات

JOBVL 
!>          JOBVL هو CHARACTER*1
!>          = 'N': لا يتم حساب المتجهات الذاتية اليسرى للمصفوفة A؛
!>          = 'V': يتم حساب المتجهات الذاتية اليسرى للمصفوفة A.
!>

JOBVR

!>          JOBVR هو CHARACTER*1
!>          = 'N': لا يتم حساب المتجهات الذاتية اليمنى للمصفوفة A؛
!>          = 'V': يتم حساب المتجهات الذاتية اليمنى للمصفوفة A.
!>

N

!>          N هو INTEGER
!>          رتبة المصفوفة A. N >= 0.
!>

A

!>          A هو مصفوفة REAL، ذات بعد (LDA,N)
!>          عند الدخول، المصفوفة A بحجم N-by-N.
!>          عند الخروج، تم استبدال A.
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرئيسي للمصفوفة A.  LDA >= max(1,N).
!>

WR

!>          WR هي مصفوفة REAL، ذات بعد (N)
!>

WI

!>          WI هي مصفوفة REAL، ذات بعد (N)
!>          تحتوي WR و WI على الأجزاء الحقيقية والخيالية،
!>          على التوالي، للقيم الذاتية المحسوبة.  الأزواج المترافقة
!>          المتخيلة للقيم الذاتية تظهر بشكل متتالٍ
!>          مع القيمة الذاتية التي لها الجزء التخيلي الموجب
!>          أولاً.
!>

VL

!>          VL هي مصفوفة REAL، ذات بعد (LDVL,N)
!>          إذا كان JOBVL = 'V'، يتم تخزين المتجهات الذاتية اليسرى u(j)
!>          واحدة تلو الأخرى في أعمدة VL، بنفس ترتيب
!>          القيم الذاتية الخاصة بها.
!>          إذا كان JOBVL = 'N'، لا يتم استخدام VL.
!>          إذا كانت القيمة الذاتية j حقيقية، فإن u(j) = VL(:,j)،
!>          العمود j من VL.
!>          إذا كانت القيمتان الذاتيتان j و (j+1) تشكلان زوجًا
!>          مترافقًا متخيلًا، فإن u(j) = VL(:,j) + i*VL(:,j+1) و
!>          u(j+1) = VL(:,j) - i*VL(:,j+1).
!>

LDVL

!>          LDVL هو INTEGER
!>          البعد الرئيسي للمصفوفة VL.  LDVL >= 1؛ إذا
!>          كان JOBVL = 'V'، فإن LDVL >= N.
!>

VR

!>          VR هي مصفوفة REAL، ذات بعد (LDVR,N)
!>          إذا كان JOBVR = 'V'، يتم تخزين المتجهات الذاتية اليمنى v(j)
!>          واحدة تلو الأخرى في أعمدة VR، بنفس ترتيب
!>          القيم الذاتية الخاصة بها.
!>          إذا كان JOBVR = 'N'، لا يتم استخدام VR.
!>          إذا كانت القيمة الذاتية j حقيقية، فإن v(j) = VR(:,j)،
!>          العمود j من VR.
!>          إذا كانت القيمتان الذاتيتان j و (j+1) تشكلان زوجًا
!>          مترافقًا متخيلًا، فإن v(j) = VR(:,j) + i*VR(:,j+1) و
!>          v(j+1) = VR(:,j) - i*VR(:,j+1).
!>

LDVR

!>          LDVR هو INTEGER
!>          البعد الرئيسي للمصفوفة VR.  LDVR >= 1؛ إذا
!>          كان JOBVR = 'V'، فإن LDVR >= N.
!>

WORK

!>          WORK هي مصفوفة REAL، ذات بعد (MAX(1,LWORK))
!>          عند الخروج، إذا كان INFO = 0، فإن WORK(1) يحتوي على LWORK الأمثل.
!>

LWORK

!>          LWORK هو INTEGER
!>          بعد المصفوفة WORK.  LWORK >= max(1,3*N)، و
!>          إذا كان JOBVL = 'V' أو JOBVR = 'V'، فإن LWORK >= 4*N.  لأداء
!>          جيد، يجب أن يكون LWORK أكبر بشكل عام.
!>
!>          إذا كان LWORK = -1، فإنه يُفترض استعلام عن مساحة العمل؛
!>          الروتين الفرعي يحسب فقط الحجم الأمثل للمصفوفة WORK،
!>          ويعيد هذا القيمة كإدخال أول للمصفوفة WORK، ولا يتم إصدار
!>          أي رسالة خطأ متعلقة بـ LWORK بواسطة XERBLA.
!>

INFO

!>          INFO هو INTEGER
!>          = 0:  الخروج الناجح
!>          < 0:  إذا كان INFO = -i، فإن المعامل i كان له قيمة غير قانونية.
!>          > 0:  إذا كان INFO = i، فشلت خوارزمية QR في حساب جميع
!>                القيم الذاتية، ولم يتم حساب أي متجهات ذاتية؛
!>                العناصر i+1:N من WR و WI تحتوي على القيم الذاتية التي
!>                تقاربت.
!>
