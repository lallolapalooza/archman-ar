توثيق الوظيفة/الإجراء الفرعي

subroutine sgbmv (character trans, integer m, integer n, integer kl, integer ku, real alpha, real, dimension(lda,*) a, integer lda, real, dimension(*) x, integer incx, real beta, real, dimension(*) y, integer incy)

SGBMV

الغرض:

!>
!> SGBMV  يقوم بإحدى عمليات المصفوفة-المتجه التالية:
!>
!>    y := alpha*A*x + beta*y,   أو   y := alpha*A**T*x + beta*y,
!>
!> حيث alpha و beta هما قياسيان، x و y هما متجهان، و A هي مصفوفة ذات نطاق
!> بحجم m × n، مع kl تحت-أقطار و ku فوق-أقطار.
!>

المعلمات

TRANS 
!>          TRANS هو CHARACTER*1
!>           عند الإدخال، يحدد TRANS العملية المراد تنفيذها كالتالي:
!>
!>              TRANS = 'N' أو 'n'   y := alpha*A*x + beta*y.
!>
!>              TRANS = 'T' أو 't'   y := alpha*A**T*x + beta*y.
!>
!>              TRANS = 'C' أو 'c'   y := alpha*A**T*x + beta*y.
!>

M

!>          M هو INTEGER
!>           عند الإدخال، يحدد M عدد صفوف المصفوفة A.
!>           يجب أن تكون M أكبر من أو تساوي صفر.
!>

N

!>          N هو INTEGER
!>           عند الإدخال، يحدد N عدد أعمدة المصفوفة A.
!>           يجب أن تكون N أكبر من أو تساوي صفر.
!>

KL

!>          KL هو INTEGER
!>           عند الإدخال، يحدد KL عدد تحت-الأقطار للمصفوفة A.
!>           يجب أن تحقق KL:  0 .le. KL.
!>

KU

!>          KU هو INTEGER
!>           عند الإدخال، يحدد KU عدد فوق-الأقطار للمصفوفة A.
!>           يجب أن تحقق KU:  0 .le. KU.
!>

ALPHA

!>          ALPHA هو REAL
!>           عند الإدخال، يحدد القياسي ALPHA.
!>

A

!>          A هو مصفوفة REAL، ذات بُعد ( LDA, N )
!>           قبل الإدخال، يجب أن يحتوي الجزء الرئيسي ( kl + ku + 1 ) × n من
!>           المصفوفة A على معاملات المصفوفة، المقدمة عمودًا بعمود، مع وجود
!>           القطر الرئيسي للمصفوفة في الصف ( ku + 1 ) من المصفوفة، ويبدأ فوق-القطر
!>           الأول عند الموضع 2 في الصف ku، وتحت-القطر الأول يبدأ عند الموضع 1
!>           في الصف ( ku + 2 )، وهكذا.
!>           العناصر في المصفوفة A التي لا تتوافق مع عناصر في مصفوفة النطاق
!>           (مثل المثلث العلوي الأيسر بحجم ku × ku) لا يتم الرجوع إليها.
!>           مقطع البرنامج التالي سينقل مصفوفة نطاق
!>           من التخزين الكامل التقليدي إلى تخزين النطاق:
!>
!>                 DO 20, J = 1, N
!>                    K = KU + 1 - J
!>                    DO 10, I = MAX( 1, J - KU ), MIN( M, J + KL )
!>                       A( K + I, J ) = matrix( I, J )
!>              10    CONTINUE
!>              20 CONTINUE
!>

LDA

!>          LDA هو INTEGER
!>           عند الإدخال، يحدد LDA البعد الأول للمصفوفة A كما تم تعريفه
!>           في البرنامج (الفرعي) المستدعي. يجب أن يكون LDA على الأقل
!>           ( kl + ku + 1 ).
!>

X

!>          X هو مصفوفة REAL، ذات بُعد على الأقل
!>           ( 1 + ( n - 1 )*abs( INCX ) ) عندما TRANS = 'N' أو 'n'
!>           وعلى الأقل
!>           ( 1 + ( m - 1 )*abs( INCX ) ) بخلاف ذلك.
!>           قبل الإدخال، يجب أن تحتوي المصفوفة X ذات الزيادات على
!>           المتجه x.
!>

INCX

!>          INCX هو INTEGER
!>           عند الإدخال، يحدد INCX الزيادة للعناصر في
!>           X. يجب ألا يكون INCX صفرًا.
!>

BETA

!>          BETA هو REAL
!>           عند الإدخال، يحدد القياسي beta. عندما يتم
!>           توفير BETA بقيمة صفر، لا تحتاج Y إلى تعيينها عند الإدخال.
!>

Y

!>          Y هو مصفوفة REAL، ذات بُعد على الأقل
!>           ( 1 + ( m - 1 )*abs( INCY ) ) عندما TRANS = 'N' أو 'n'
!>           وعلى الأقل
!>           ( 1 + ( n - 1 )*abs( INCY ) ) بخلاف ذلك.
!>           قبل الإدخال، يجب أن تحتوي المصفوفة Y ذات الزيادات على
!>           المتجه y. عند الخروج، يتم الكتابة فوق Y بواسطة المتجه y المحدث.
!>           إذا كان m أو n صفرًا، فلن يتم الرجوع إلى Y وتقوم الدالة
!>           بالعودة السريعة.
!>

INCY

!>          INCY هو INTEGER
!>           عند الإدخال، يحدد INCY الزيادة للعناصر في
!>           Y. يجب ألا يكون INCY صفرًا.
!>
