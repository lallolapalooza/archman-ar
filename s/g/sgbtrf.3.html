subroutine sgbtrf (integer m, integer n, integer kl, integer ku, real, dimension( ldab, * ) ab, integer ldab, integer, dimension( * ) ipiv, integer info)

SGBTRF

الغرض:

!>
!> SGBTRF يحسب تحليل LU لمصفوفة A ذات النطاق الحقيقي m-by-n
!> باستخدام التجزئة الجزئية مع تبادلات الصفوف.
!>
!> هذه هي النسخة المجمعة من الخوارزمية، وتستدعي BLAS من المستوى 3.
!>

المعلمات

M 
!>          M هو عدد صحيح (INTEGER)
!>          عدد صفوف المصفوفة A.  M >= 0.
!>

N

!>          N هو عدد صحيح (INTEGER)
!>          عدد أعمدة المصفوفة A.  N >= 0.
!>

KL

!>          KL هو عدد صحيح (INTEGER)
!>          عدد الأقطار الفرعية داخل نطاق A.  KL >= 0.
!>

KU

!>          KU هو عدد صحيح (INTEGER)
!>          عدد الأقطار الفائقة داخل نطاق A.  KU >= 0.
!>

AB

!>          AB هو مصفوفة حقيقية (REAL array)، أبعادها (LDAB,N)
!>          عند الدخول، المصفوفة A بتخزين النطاق، في الصفوف KL+1 إلى
!>          2*KL+KU+1؛ لا تحتاج الصفوف 1 إلى KL من المصفوفة إلى التعيين.
!>          يتم تخزين العمود j من A في العمود j من
!>          المصفوفة AB كالتالي:
!>          AB(kl+ku+1+i-j,j) = A(i,j) for max(1,j-ku)<=i<=min(m,j+kl)
!>
!>          عند الخروج، تفاصيل التحليل: يتم تخزين U كـ
!>          مصفوفة نطاق مثلثة عليا مع KL+KU أقطار فائقة في
!>          الصفوف 1 إلى KL+KU+1، والمضاعفات المستخدمة أثناء
!>          التحليل يتم تخزينها في الصفوف KL+KU+2 إلى 2*KL+KU+1.
!>          انظر أدناه لمزيد من التفاصيل.
!>

LDAB

!>          LDAB هو عدد صحيح (INTEGER)
!>          البعد الرائد للمصفوفة AB.  LDAB >= 2*KL+KU+1.
!>

IPIV

!>          IPIV هو مصفوفة أعداد صحيحة (INTEGER array)، أبعادها (min(M,N))
!>          مؤشرات المحور؛ لـ 1 <= i <= min(M,N)، تم تبديل الصف i من
!>          المصفوفة بالصف IPIV(i).
!>

INFO

!>          INFO هو عدد صحيح (INTEGER)
!>          = 0: خروج ناجح
!>          < 0: إذا كان INFO = -i، فإن المعامل i كان له قيمة غير قانونية
!>          > 0: إذا كان INFO = +i، فإن U(i,i) يساوي صفرًا تمامًا. التحليل
!>               قد اكتمل، لكن العامل U هو بالضبط
!>               منفرد، وسيقسم على صفر إذا تم استخدامه
!>               لحساب نظام معادلات.
!>

تفاصيل إضافية:

!>
!>  مخطط تخزين النطاق موضح بالمثال التالي، عندما
!>  M = N = 6, KL = 2, KU = 1:
!>
!>  عند الدخول:                       عند الخروج:
!>
!>      *    *    *    +    +    +       *    *    *   u14  u25  u36
!>      *    *    +    +    +    +       *    *   u13  u24  u35  u46
!>      *   a12  a23  a34  a45  a56      *   u12  u23  u34  u45  u56
!>     a11  a22  a33  a44  a55  a66     u11  u22  u33  u44  u55  u66
!>     a21  a32  a43  a54  a65   *      m21  m32  m43  m54  m65   *
!>     a31  a42  a53  a64   *    *      m31  m42  m53  m64   *    *
!>
!>  عناصر المصفوفة المشار إليها بـ * لا تستخدمها الروتين؛ العناصر المشار إليها
!>  بـ + لا تحتاج إلى تعيين عند الدخول، ولكنها مطلوبة من الروتين لتخزين
!>  عناصر U بسبب الملء الناتج عن تبادلات الصفوف.
!>
