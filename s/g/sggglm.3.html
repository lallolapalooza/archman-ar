subroutine sggglm (integer n, integer m, integer p, real, dimension( lda, * ) a, integer lda, real, dimension( ldb, * ) b, integer ldb, real, dimension( * ) d, real, dimension( * ) x, real, dimension( * ) y, real, dimension( * ) work, integer lwork, integer info)

SGGGLM

الغرض:

!>
!> SGGGLM يحل مشكلة نموذج غاوس-ماركوف الخطي العام (GLM):
!>
!>         تصغير || y ||_2   مع مراعاة   d = A*x + B*y
!>             x
!>
!> حيث A مصفوفة N × M، و B مصفوفة N × P، و d متجه N معطى. يُفترض أن M <= N <= M+P، و
!>
!>            rank(A) = M    و    rank( A B ) = N.
!>
!> تحت هذه الافتراضات، يكون الشرط المقيد متسقًا دائمًا، وهناك حل فريد x وحل بأدنى معيار 2 y،
!> والذي يتم الحصول عليه باستخدام تحليل QR العام للمصفوفتين (A, B) المعطاة بواسطة
!>
!>    A = Q*(R),   B = Q*T*Z.
!>          (0)
!>
!> على وجه الخصوص، إذا كانت المصفوفة B مربعة وغير مفردة، فإن مشكلة
!> GLM تعادل مشكلة أقل المربعات الموزونة التالية
!>
!>              تصغير || inv(B)*(d-A*x) ||_2
!>                  x
!>
!> حيث يشير inv(B) إلى معكوس B.
!>
!> يجب على المستدعين لهذه الإجراءات الفرعية ملاحظة أن فحوصات التفرد/نقص الرتبة
!> المطبقة في هذه الإجراءات الفرعية بدائية. الإجراء الفرعي STRTRS الذي تستدعيه هذه
!> الإجراءات الفرعية يشير فقط إلى فشل بسبب التفرد إذا كانت المشكلة مفردة تمامًا.
!>
!> من الممكن أن يكون أحد العوامل (أو أكثر) المشاركة في التحليل QR العام
!> للزوج (A, B) قريبًا بشكل غير طبيعي من التفرد دون أن تشير هذه
!> الإجراءات الفرعية إلى خطأ. قد تكون الحلول المحسوبة لمثل هذه المشاكل شبه المفردة
!> أقل دقة بسبب فقدان الدقة العددية.
!>
!>

المعاملات

N 
!>          N هو عدد صحيح (INTEGER)
!>          عدد الصفوف في المصفوفتين A و B. N >= 0.
!>

M

!>          M هو عدد صحيح (INTEGER)
!>          عدد أعمدة المصفوفة A. 0 <= M <= N.
!>

P

!>          P هو عدد صحيح (INTEGER)
!>          عدد أعمدة المصفوفة B. P >= N-M.
!>

A

!>          A هو مصفوفة أعداد حقيقية (REAL array)، البعد (LDA,M)
!>          عند الإدخال، المصفوفة A بحجم N × M.
!>          عند الخروج، الجزء العلوي المثلثي من المصفوفة A يحتوي على
!>          المصفوفة المثلثية العليا R بحجم M × M.
!>

LDA

!>          LDA هو عدد صحيح (INTEGER)
!>          البعد الرائد للمصفوفة A. LDA >= max(1,N).
!>

B

!>          B هو مصفوفة أعداد حقيقية (REAL array)، البعد (LDB,P)
!>          عند الإدخال، المصفوفة B بحجم N × P.
!>          عند الخروج، إذا كان N <= P، فإن الجزء العلوي المثلثي من المصفوفة الفرعية
!>          B(1:N,P-N+1:P) يحتوي على المصفوفة المثلثية العليا T بحجم N × N؛
!>          إذا كان N > P، فإن العناصر على وعلى فوق القطر الفرعي (N-P)
!>          تحتوي على المصفوفة الشبه المنحرفة العليا T بحجم N × P.
!>

LDB

!>          LDB هو عدد صحيح (INTEGER)
!>          البعد الرائد للمصفوفة B. LDB >= max(1,N).
!>

D

!>          D هو مصفوفة أعداد حقيقية (REAL array)، البعد (N)
!>          عند الإدخال، D هو الطرف الأيسر لمعادلة GLM.
!>          عند الخروج، يتم تدمير D.
!>

X

!>          X هو مصفوفة أعداد حقيقية (REAL array)، البعد (M)
!>

Y

!>          Y هو مصفوفة أعداد حقيقية (REAL array)، البعد (P)
!>
!>          عند الخروج، X و Y هما حلول مشكلة GLM.
!>

WORK

!>          WORK هو مصفوفة أعداد حقيقية (REAL array)، البعد (MAX(1,LWORK))
!>          عند الخروج، إذا كان INFO = 0، فإن WORK(1) يعيد LWORK الأمثل.
!>

LWORK

!>          LWORK هو عدد صحيح (INTEGER)
!>          بعد المصفوفة WORK. LWORK >= max(1,N+M+P).
!>          لأداء أمثل، LWORK >= M+min(N,P)+max(N,P)*NB،
!>          حيث NB هو حد أعلى لأحجام القطع المثلى لـ
!>          SGEQRF, SGERQF, SORMQR و SORMRQ.
!>
!>          إذا كان LWORK = -1، فيُفترض استعلام عن مساحة العمل؛ الإجراء الفرعي
!>          يحسب فقط الحجم الأمثل للمصفوفة WORK، ويعيد
!>          هذه القيمة كإدخال أول في المصفوفة WORK، ولا يتم إصدار أي رسالة خطأ
!>          بخصوص LWORK بواسطة XERBLA.
!>

INFO

!>          INFO هو عدد صحيح (INTEGER)
!>          = 0:  انتهاء ناجح.
!>          < 0:  إذا كان INFO = -i، فإن المعامل i كان له قيمة غير قانونية.
!>          = 1:  العامل المثلثي العلوي R المرتبط بـ A في
!>                التحليل QR العام للزوج (A, B) هو مفرد تمامًا، بحيث rank(A) < M؛
!>                لم يتمكن من حساب حل أقل المربعات.
!>          = 2:  الجزء السفلي بحجم (N-M) × (N-M) من العامل الشبه منحرف العلوي
!>                T المرتبط بـ B في التحليل QR العام
!>                للزوج (A, B) هو مفرد تمامًا، بحيث
!>                rank( A B ) < N؛ لم يتمكن من حساب حل أقل المربعات.
!>
