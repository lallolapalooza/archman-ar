توثيق الوظيفة/الروتين الفرعي

subroutine sgetsqrhrt (integer m, integer n, integer mb1, integer nb1, integer nb2, real, dimension( lda, * ) a, integer lda, real, dimension( ldt, * ) t, integer ldt, real, dimension( * ) work, integer lwork, integer info)

SGETSQRHRT

الغرض:

!>
!> SGETSQRHRT يحسب تحلل QR المعبأ بعمود بحجم NB2
!> لمصفوفة معقدة M-by-N A حيث M >= N،
!>
!>    A = Q * R.
!>
!> الروتين يستخدم داخليًا تحلل TSQR المعبأ بعمود بحجم NB1
!> والمعبأ بصفوف بحجم MB1، ويقوم بإعادة بناء
!> متجهات هاوسهولدر من مخرجات TSQR. يقوم الروتين أيضًا
!> بتحويل عامل R_tsqr من مخرجات تحلل TSQR إلى
!> العامل R الذي يتوافق مع تحلل QR لهاوسهولدر،
!>
!>    A = Q_tsqr * R_tsqr = Q * R.
!>
!> يتم تخزين العوامل Q و R الناتجة بنفس التنسيق الموجود في SGEQRT
!> (Q موجود في تمثيل WY المدمج المعبأ). انظر توثيق
!> SGEQRT لمزيد من التفاصيل حول التنسيق.
!>

المعلمات

M 
!>          M هو INTEGER
!>          عدد صفوف المصفوفة A. M >= 0.
!>

N

!>          N هو INTEGER
!>          عدد أعمدة المصفوفة A. M >= N >= 0.
!>

MB1

!>          MB1 هو INTEGER
!>          حجم الكتلة الصفية المستخدم في TSQR المعبأ.
!>          MB1 > N.
!>

NB1

!>          NB1 هو INTEGER
!>          حجم الكتلة العمودية المستخدم في TSQR المعبأ.
!>          N >= NB1 >= 1.
!>

NB2

!>          NB2 هو INTEGER
!>          حجم الكتلة المستخدم في QR المعبأ الذي سيتم إخراجه.
!>          NB2 >= 1.
!>

A

!>          A هو مصفوفة REAL، البعد (LDA,N)
!>
!>          عند الإدخال: مصفوفة M-by-N A.
!>
!>          عند الإخراج:
!>           أ) العناصر على وفوق القطر
!>              للمصفوفة تحتوي على المصفوفة شبه المثلثية العلوية N-by-N
!>              R المقابلة لتحلل QR لهاوسهولدر؛
!>           ب) العناصر تحت القطر تمثل Q بواسطة
!>              أعمدة V المعبأة (تمثيل WY المدمج).
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرئيسي للمصفوفة A. LDA >= max(1,M).
!>

T

!>          T هو مصفوفة REAL، البعد (LDT,N))
!>          مثبتات الكتلة المثلثية العلوية المخزنة بتنسيق مدمج
!>          كسلسلة من الكتل المثلثية العلوية.
!>

LDT

!>          LDT هو INTEGER
!>          البعد الرئيسي للمصفوفة T. LDT >= NB2.
!>

WORK

!>          (مساحة العمل) مصفوفة REAL، البعد (MAX(1,LWORK))
!>          عند الإخراج، إذا كان INFO = 0، فإن WORK(1) يعيد LWORK الأمثل.
!>

LWORK

!>          LWORK هو INTEGER
!>          بعد مصفوفة WORK.
!>          إذا كان MIN(M,N) = 0، فإن LWORK >= 1، وإلا
!>          LWORK >= MAX( 1, LWT + LW1, MAX( LWT+N*N+LW2, LWT+N*N+N ) )،
!>          حيث
!>             NUM_ALL_ROW_BLOCKS = CEIL((M-N)/(MB1-N))،
!>             NB1LOCAL = MIN(NB1,N).
!>             LWT = NUM_ALL_ROW_BLOCKS * N * NB1LOCAL،
!>             LW1 = NB1LOCAL * N،
!>             LW2 = NB1LOCAL * MAX( NB1LOCAL, ( N - NB1LOCAL ) ).
!>
!>          إذا كان LWORK = -1، يفترض استعلام مساحة العمل.
!>          يقوم الروتين فقط بحساب الحجم الأمثل لمصفوفة WORK،
!>          ويعيد هذا الحجم كإدخال أول لمصفوفة WORK، ولا يتم إصدار
!>          أي رسالة خطأ متعلقة بـ LWORK بواسطة XERBLA.
!>

INFO

!>          INFO هو INTEGER
!>          = 0:  اكتمال ناجح
!>          < 0:  إذا كان INFO = -i، فإن المعلمة i كان لها قيمة غير قانونية
!>
