subroutine sget03 (integer n, real, dimension( lda, * ) a, integer lda, real, dimension( ldainv, * ) ainv, integer ldainv, real, dimension( ldwork, * ) work, integer ldwork, real, dimension( * ) rwork, real rcond, real resid)

SGET03

الغرض:

!>
!> SGET03 يحسب الباقي لعملية ضرب مصفوفة عامة في معكوسها:
!>    norm( I - AINV*A ) / ( N * norm(A) * norm(AINV) * EPS ),
!> حيث EPS هو خطأ الآلة.
!>

المعاملات

N 
!>          N هو عدد صحيح
!>          عدد الصفوف والأعمدة للمصفوفة A.  N >= 0.
!>

A

!>          A هي مصفوفة حقيقية، الأبعاد (LDA,N)
!>          المصفوفة الأصلية A ذات الأبعاد N x N.
!>

LDA

!>          LDA هو عدد صحيح
!>          البعد الرائد للمصفوفة A.  LDA >= max(1,N).
!>

AINV

!>          AINV هي مصفوفة حقيقية، الأبعاد (LDAINV,N)
!>          معكوس المصفوفة A.
!>

LDAINV

!>          LDAINV هو عدد صحيح
!>          البعد الرائد للمصفوفة AINV.  LDAINV >= max(1,N).
!>

WORK

!>          WORK هي مصفوفة حقيقية، الأبعاد (LDWORK,N)
!>

LDWORK

!>          LDWORK هو عدد صحيح
!>          البعد الرائد للمصفوفة WORK.  LDWORK >= max(1,N).
!>

RWORK

!>          RWORK هي مصفوفة حقيقية، الأبعاد (N)
!>

RCOND

!>          RCOND هو عدد حقيقي
!>          معكوس رقم شرط المصفوفة A، محسوب كـ
!>          ( 1/norm(A) ) / norm(AINV).
!>

RESID

!>          RESID هو عدد حقيقي
!>          norm(I - AINV*A) / ( N * norm(A) * norm(AINV) * EPS )
!>
