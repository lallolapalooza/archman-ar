توثيق الوظيفة/الروتين الفرعي

subroutine sgbequ (integer m, integer n, integer kl, integer ku, real, dimension( ldab, * ) ab, integer ldab, real, dimension( * ) r, real, dimension( * ) c, real rowcnd, real colcnd, real amax, integer info)

SGBEQU

الغرض:

!>
!> SGBEQU تحسب عوامل التحجيم للصف والأعمدة المخصصة لمصفوفة نطاق M-by-N A
!> وتقليل عدد شروطها. R تُرجع عوامل تحجيم الصف و C عوامل التحجيم العمودي،
!> المختارة لمحاولة جعل أكبر عنصر في كل صف وعمود من المصفوفة B التي عناصرها
!> B(i,j)=R(i)*A(i,j)*C(j) لها قيمة مطلقة 1.
!>
!> R(i) و C(j) مقيدة لتكون بين SMLNUM = أصغر رقم آمن و BIGNUM = أكبر رقم آمن.
!> استخدام عوامل التحجيم هذه لا يضمن تقليل عدد شروط A ولكنه يعمل بشكل جيد
!> في الممارسة العملية.
!>

المعلمات

M 
!>          M هو عدد صحيح (INTEGER)
!>          عدد صفوف المصفوفة A. M >= 0.
!>

N

!>          N هو عدد صحيح (INTEGER)
!>          عدد أعمدة المصفوفة A. N >= 0.
!>

KL

!>          KL هو عدد صحيح (INTEGER)
!>          عدد الأقطار الفرعية داخل نطاق A. KL >= 0.
!>

KU

!>          KU هو عدد صحيح (INTEGER)
!>          عدد الأقطار العلوية داخل نطاق A. KU >= 0.
!>

AB

!>          AB هي مصفوفة حقيقية (REAL array)، ذات أبعاد (LDAB,N)
!>          مصفوفة النطاق A، مخزنة في الصفوف 1 إلى KL+KU+1. يتم تخزين العمود j
!>          من A في العمود j من المصفوفة AB على النحو التالي:
!>          AB(ku+1+i-j,j) = A(i,j) لـ max(1,j-ku) <= i <= min(m,j+kl).
!>

LDAB

!>          LDAB هو عدد صحيح (INTEGER)
!>          البعد الرائد للمصفوفة AB. LDAB >= KL+KU+1.
!>

R

!>          R هي مصفوفة حقيقية (REAL array)، ذات أبعاد (M)
!>          إذا كان INFO = 0، أو INFO > M، فإن R تحتوي على عوامل تحجيم الصفوف
!>          لـ A.
!>

C

!>          C هي مصفوفة حقيقية (REAL array)، ذات أبعاد (N)
!>          إذا كان INFO = 0، فإن C تحتوي على عوامل تحجيم الأعمدة لـ A.
!>

ROWCND

!>          ROWCND هو قيمة حقيقية (REAL)
!>          إذا كان INFO = 0 أو INFO > M، فإن ROWCND تحتوي على نسبة أصغر R(i)
!>          إلى أكبر R(i). إذا كان ROWCND >= 0.1 وكان AMAX ليس كبيرًا جدًا أو
!>          صغيرًا جدًا، فلا فائدة من التحجيم بواسطة R.
!>

COLCND

!>          COLCND هو قيمة حقيقية (REAL)
!>          إذا كان INFO = 0، فإن COLCND تحتوي على نسبة أصغر C(i) إلى أكبر C(i).
!>          إذا كان COLCND >= 0.1، فلا فائدة من التحجيم بواسطة C.
!>

AMAX

!>          AMAX هو قيمة حقيقية (REAL)
!>          القيمة المطلقة لأكبر عنصر في المصفوفة. إذا كان AMAX قريبًا جدًا من
!>          الفيضان أو قريبًا جدًا من الانهيار، يجب تحجيم المصفوفة.
!>

INFO

!>          INFO هو عدد صحيح (INTEGER)
!>          = 0:  انتهاء ناجح
!>          < 0:  إذا كان INFO = -i، فإن الوسيط i كان له قيمة غير قانونية
!>          > 0:  إذا كان INFO = i، و i هو
!>                <= M:  الصف i من A هو صفر تمامًا
!>                >  M:  العمود (i-M) من A هو صفر تمامًا
!>
