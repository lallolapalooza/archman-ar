وثائق الوظيفة/الروتين الفرعي

subroutine sgesvdq (character joba, character jobp, character jobr, character jobu, character jobv, integer m, integer n, real, dimension( lda, * ) a, integer lda, real, dimension( * ) s, real, dimension( ldu, * ) u, integer ldu, real, dimension( ldv, * ) v, integer ldv, integer numrank, integer, dimension( * ) iwork, integer liwork, real, dimension( * ) work, integer lwork, real, dimension( * ) rwork, integer lrwork, integer info)

SGESVDQ يحسب تحليل القيم المفردة (SVD) باستخدام طريقة SVD QR مسبقة المعالجة لـ GE matrices

الغرض:

!>
!> SGESVDQ يحسب تحليل القيم المفردة (SVD) لمصفوفة حقيقية M-by-N A، حيث M >= N. تحليل SVD للمصفوفة A يُكتب كـ
!>                                    [++]   [xx]   [x0]   [xx]
!>              A = U * SIGMA * V^*,  [++] = [xx] * [ox] * [xx]
!>                                    [++]   [xx]
!> حيث SIGMA مصفوفة قطرية N-by-N، U مصفوفة متعامدة N-by-M، و V مصفوفة متعامدة N-by-N. العناصر القطرية لـ SIGMA هي القيم المفردة للمصفوفة A. أعمدة U و V هي المتجهات المفردة اليسرى واليمنى للمصفوفة A على التوالي.
!>

المعلمات

JOBA 
!>  JOBA هو CHARACTER*1
!>  يحدد مستوى الدقة في تحليل SVD المحسوب
!>  = 'A' الدقة المطلوبة تتوافق مع وجود خطأ خلفي محدود بـ || delta A ||_F <= f(m,n) * EPS * || A ||_F,
!>        حيث EPS = SLAMCH('Epsilon'). هذا يسمح لـ CGESVDQ بتقليم العامل المثلثي المحسوب في تحليل QR عالي التكشف الرتيب كلما كان الجزء المقلم أقل من العتبة التي هي من رتبة EPS * ||A||_F. هذا مستوى تقليم عدواني.
!>  = 'M' بالمثل كما هو الحال مع 'A'، ولكن التقليم أكثر لطفًا: يُسمح به فقط عندما يكون هناك انخفاض في القطر للعامل المثلثي في تحليل QR. هذا مستوى تقليم متوسط.
!>  = 'H' دقة عالية مطلوبة. لا يتم محاولة تحديد الرتبة الرقمية بناءً على تحليل QR عالي التكشف الرتيب.
!>  = 'E' مثل 'H'، وبالإضافة إلى ذلك يتم تقدير رقم شرط المصفوفة A ذات المقياس العمودي وإرجاعه في RWORK(1).
!>        N^(-1/4)*RWORK(1) <= ||pinv(A_scaled)||_2 <= N^(1/4)*RWORK(1)
!>

JOBP

!>  JOBP هو CHARACTER*1
!>  = 'P' يتم ترتيب صفوف A بترتيب تنازلي فيما يتعلق بـ ||A(i,:)||_\infty. هذا يعزز الدقة العددية على حساب حركة بيانات إضافية. موصى به للمتانة العددية.
!>  = 'N' لا يوجد تنقيط للصفوف.
!>

JOBR

!>          JOBR هو CHARACTER*1
!>          = 'T' بعد تحليل QR المنقح الأولي، يتم تطبيق SGESVD على
!>          مصفوفة R**T المنقولة للعامل المثلثي R المحسوب. يتضمن هذا
!>          بعض حركة البيانات الإضافية (عمليات تبديل المصفوفات). مفيد للتجارب والبحث والتطوير.
!>          = 'N' يتم إدخال العامل المثلثي R إلى SGESVD. قد يكون هذا مفضلاً لأنه يتضمن حركة بيانات أقل.
!>

JOBU

!>          JOBU هو CHARACTER*1
!>          = 'A' يتم حساب جميع المتجهات المفردة اليسرى M وإرجاعها في
!>          المصفوفة U. انظر وصف U.
!>          = 'S' أو 'U' يتم حساب N = min(M,N) من المتجهات المفردة اليسرى وإرجاعها
!>          في المصفوفة U. انظر وصف U.
!>          = 'R' يتم تحديد الرتبة العددية NUMRANK ويتم حساب وإرجاع NUMRANK فقط من المتجهات المفردة اليسرى
!>          في المصفوفة U.
!>          = 'F' يتم إرجاع المتجهات المفردة اليسرى N في شكل مُحلل كحاصل ضرب
!>          للعامل Q من تحليل QR الأولي والمتجهات المفردة اليسرى N لـ (R**T , 0)**T. إذا تم استخدام تنقيط الصفوف،
!>          فإن المعلومات الضرورية حول تنقيط الصفوف مخزنة في
!>          IWORK(N+1:N+M-1).
!>          = 'N' لا يتم حساب المتجهات المفردة اليسرى.
!>

JOBV

!>          JOBV هو CHARACTER*1
!>          = 'A', 'V' يتم حساب جميع المتجهات المفردة اليمنى N وإرجاعها في
!>          المصفوفة V.
!>          = 'R' يتم تحديد الرتبة العددية NUMRANK ويتم حساب وإرجاع NUMRANK فقط من المتجهات المفردة اليمنى
!>          في المصفوفة V. هذا الخيار مسموح به فقط إذا كان JOBU = 'R' أو JOBU = 'N'؛ وإلا فهو غير قانوني.
!>          = 'N' لا يتم حساب المتجهات المفردة اليمنى.
!>

M

!>          M هو INTEGER
!>          عدد صفوف المصفوفة المدخلة A. M >= 0.
!>

N

!>          N هو INTEGER
!>          عدد أعمدة المصفوفة المدخلة A. M >= N >= 0.
!>

A

!>          A هو مصفوفة REAL بأبعاد LDA x N
!>          عند الدخول، المصفوفة المدخلة A.
!>          عند الخروج، إذا كان JOBU .NE. 'N' أو JOBV .NE. 'N'، فإن المثلث السفلي لـ A يحتوي على
!>          متجهات هاوسهولدر المخزنة بواسطة SGEQP3. إذا كان JOBU = 'F'، يمكن استخدام متجهات هاوسهولدر هذه
!>          مع WORK(1:N) لاستعادة عوامل Q من تحليل QR المنقح الأولي لـ A. انظر وصف U.
!>

LDA

!>          LDA هو INTEGER.
!>          البعد الرائد للمصفوفة A. LDA >= max(1,M).
!>

S

!>          S هي مصفوفة REAL بأبعاد N.
!>          القيم المفردة لـ A، مرتبة بحيث S(i) >= S(i+1).
!>

U

!>          U هي مصفوفة REAL، بأبعاد
!>          LDU x M إذا كان JOBU = 'A'؛ انظر وصف LDU. في هذه الحالة،
!>          عند الخروج، تحتوي U على المتجهات المفردة اليسرى M.
!>          LDU x N إذا كان JOBU = 'S', 'U', 'R'؛ انظر وصف LDU. في هذه
!>          الحالة، تحتوي U على N أو NUMRANK من المتجهات المفردة اليسرى الرائدة.
!>          LDU x N إذا كان JOBU = 'F'؛ انظر وصف LDU. في هذه الحالة تحتوي U على
!>          مصفوفة متعامدة N x N يمكن استخدامها لتكوين المتجهات المفردة اليسرى.
!>          إذا كان JOBU = 'N'، لا يتم الإشارة إلى U.
!>

LDU

!>          LDU هو INTEGER.
!>          البعد الرائد للمصفوفة U.
!>          إذا كان JOBU = 'A', 'S', 'U', 'R', LDU >= max(1,M).
!>          إذا كان JOBU = 'F', LDU >= max(1,N).
!>          بخلاف ذلك، LDU >= 1.
!>

V

!>          V هي مصفوفة REAL، بأبعاد
!>          LDV x N إذا كان JOBV = 'A', 'V', 'R' أو إذا كان JOBA = 'E'.
!>          إذا كان JOBV = 'A', أو 'V'، تحتوي V على المصفوفة المتعامدة N-by-N V**T؛
!>          إذا كان JOBV = 'R'، تحتوي V على الصفوف NUMRANK الأولى من V**T (المتجهات المفردة اليمنى، المخزنة صفًا، للقيم المفردة الأكبر NUMRANK).
!>          إذا كان JOBV = 'N' و JOBA = 'E'، يتم استخدام V كمساحة عمل.
!>          إذا كان JOBV = 'N'، و JOBA.NE.'E'، لا يتم الإشارة إلى V.
!>

LDV

!>          LDV هو INTEGER
!>          البعد الرائد للمصفوفة V.
!>          إذا كان JOBV = 'A', 'V', 'R', أو JOBA = 'E', LDV >= max(1,N).
!>          بخلاف ذلك، LDV >= 1.
!>

NUMRANK

!>          NUMRANK هو INTEGER
!>          NUMRANK هو الرتبة الرقمية التي تم تحديدها لأول مرة بعد تحليل QR عالي التكشف الرتيب، باتباع الاستراتيجية المحددة بواسطة
!>
قيمة JOBA. إذا كان JOBV = 'R' و JOBU = 'R'، فقط NUMRANK !> يتم طلب المتجهات والقيم المفردة الرائدة في الاستدعاء !> لـ SGESVD. قد يتم تقليل قيمة NUMRANK النهائية بشكل إضافي إذا !> تم حساب بعض القيم المفردة كأصفار. !>

IWORK

!>          IWORK هو مصفوفة INTEGER، أبعادها (max(1, LIWORK)).
!>          عند الخروج، تحتوي IWORK(1:N) على تبديل الأعمدة للعامل
!>          QR للكشف عن الرتبة.
!>          إذا كان JOBP = 'P'، تحتوي IWORK(N+1:N+M-1) على مؤشرات تسلسل
!>          تبديلات الصفوف المستخدمة في تحريك الصفوف. يمكن استخدام هذه لاستعادة
!>          المتجهات المفردة اليسرى في حالة JOBU = 'F'.
!>
!>          إذا كان LIWORK أو LWORK أو LRWORK = -1، فعند الخروج، إذا كان INFO = 0،
!>          تعيد IWORK(1) الحد الأدنى لـ LIWORK.
!>

LIWORK

!>          LIWORK هو INTEGER
!>          بعد مصفوفة IWORK.
!>          LIWORK >= N + M - 1، إذا كان JOBP = 'P' و JOBA .NE. 'E'؛
!>          LIWORK >= N إذا كان JOBP = 'N' و JOBA .NE. 'E'؛
!>          LIWORK >= N + M - 1 + N، إذا كان JOBP = 'P' و JOBA = 'E'؛
!>          LIWORK >= N + N إذا كان JOBP = 'N' و JOBA = 'E'.
!>
!>          إذا كان LIWORK = -1، فيفترض استعلام مساحة العمل؛ يقوم الروتين
!>          بحساب وإرجاع الأحجام المثلى والحد الأدنى لمصفوفات
!>          WORK و IWORK و RWORK فقط، ولا يتم إصدار أي رسالة خطأ
!>          متعلقة بـ LWORK بواسطة XERBLA.
!>

WORK

!>          WORK هي مصفوفة REAL، أبعادها (max(2, LWORK))، تستخدم كمساحة عمل.
!>          عند الخروج، إذا كان LWORK.NE.-1 عند الدخول، تحتوي WORK(1:N) على معلمات
!>          ضرورية لاستعادة عامل Q من تحليل QR المحسوب بواسطة
!>          SGEQP3.
!>
!>          إذا كان LIWORK أو LWORK أو LRWORK = -1، فعند الخروج، إذا كان INFO = 0،
!>          تعيد WORK(1) الحد الأمثل لـ LWORK،
!>          وتعيد WORK(2) الحد الأدنى لـ LWORK.
!>

LWORK

!>          LWORK هو INTEGER
!>          بعد مصفوفة WORK. يتم تحديدها على النحو التالي:
!>          Let LWQP3 = 3*N+1, LWCON = 3*N, and let
!>          LWORQ = { MAX( N, 1 )، إذا كان JOBU = 'R'، 'S'، أو 'U'
!>                  { MAX( M, 1 )، إذا كان JOBU = 'A'
!>          LWSVD = MAX( 5*N, 1 )
!>          LWLQF = MAX( N/2, 1 )، LWSVD2 = MAX( 5*(N/2), 1 )، LWORLQ = MAX( N, 1 )،
!>          LWQRF = MAX( N/2, 1 )، LWORQ2 = MAX( N, 1 )
!>          إذًا، الحد الأدنى لقيمة LWORK هو:
!>          = MAX( N + LWQP3, LWSVD ) إذا كانت القيم المفردة مطلوبة فقط؛
!>          = MAX( N + LWQP3, LWCON, LWSVD ) إذا كانت القيم المفردة مطلوبة فقط،
!>                                   وتم طلب تقدير شرطي مقياس؛
!>
!>          = N + MAX( LWQP3, LWSVD, LWORQ ) إذا تم طلب القيم المفردة والمتجهات المفردة اليسرى؛
!>          = N + MAX( LWQP3, LWCON, LWSVD, LWORQ ) إذا تم طلب القيم المفردة والمتجهات المفردة اليسرى،
!>                                   وتم طلب تقدير شرطي مقياس أيضًا؛
!>
!>          = N + MAX( LWQP3, LWSVD ) إذا تم طلب القيم المفردة والمتجهات المفردة اليمنى؛
!>          = N + MAX( LWQP3, LWCON, LWSVD ) إذا تم طلب القيم المفردة والمتجهات المفردة اليمنى،
!>                                   وتم طلب تقدير شرطي مقياس أيضًا؛
!>
!>          = N + MAX( LWQP3, LWSVD, LWORQ ) إذا تم طلب SVD الكاملة بـ JOBV = 'R'؛
!>                                   بشكل مستقل عن JOBR؛
!>          = N + MAX( LWQP3, LWCON, LWSVD, LWORQ ) إذا تم طلب SVD الكاملة،
!>                                   JOBV = 'R' و، أيضًا تم طلب تقدير شرطي مقياس؛
!>                                   بشكل مستقل عن JOBR؛
!>          = MAX( N + MAX( LWQP3, LWSVD, LWORQ ),
!>         N + MAX( LWQP3, N/2+LWLQF, N/2+LWSVD2, N/2+LWORLQ, LWORQ ) ) إذا
!>                         تم طلب SVD الكاملة بـ JOBV = 'A' أو 'V'، و
!>                         JOBR ='N'
!>          = MAX( N + MAX( LWQP3, LWCON, LWSVD, LWORQ ),
!>         N + MAX( LWQP3, LWCON, N/2+LWLQF, N/2+LWSVD2, N/2+LWORLQ, LWORQ ) )
!>                         إذا تم طلب SVD الكاملة بـ JOBV = 'A' أو 'V'، و
!>                         JOBR ='N'، وتم طلب تقدير شرطي مقياس أيضًا.
!>          = MAX( N + MAX( LWQP3, LWSVD, LWORQ ),
!>         N + MAX( LWQP3, N/2+LWQRF, N/2+LWSVD2, N/2+LWORQ2, LWORQ ) ) إذا
!>                         تم طلب SVD الكاملة بـ JOBV = 'A'، 'V'، و JOBR ='T'
!>          = MAX( N + MAX( LWQP3, LWCON, LWSVD, LWORQ ),
!>         N + MAX( LWQP3, LWCON, N/2+LWQRF, N/2+LWSVD2, N/2+LWORQ2, LWORQ ) )
!>                         إذا تم طلب SVD الكاملة بـ JOBV = 'A' أو 'V'، و
!>                         JOBR ='T'، وتم طلب تقدير شرطي مقياس أيضًا.
!>          أخيرًا، يجب أن يكون LWORK على الأقل اثنين: LWORK = MAX( 2, LWORK ).
!>
!>          إذا كان LWORK = -1، فيفترض استعلام مساحة العمل؛ يقوم الروتين
!>          بحساب وإرجاع الأحجام المثلى والحد الأدنى لمصفوفات
!>          WORK و IWORK و RWORK فقط، ولا يتم إصدار أي رسالة خطأ
!>          متعلقة بـ LWORK بواسطة XERBLA.
!>

RWORK

!>          RWORK هي مصفوفة REAL، أبعادها (max(1, LRWORK)).
!>          عند الخروج،
!>          1. إذا كان JOBA = 'E'، تحتوي RWORK(1) على تقدير لعدد
!>          الشرط لمصفوفة A المقيسة بالأعمدة. إذا كانت A = C * D حيث D قطرية و C
!>          لها أعمدة وحدة في معيار الإقليدي، فعندئذ، بافتراض الرتبة الكاملة للأعمدة،
!>          N^(-1/4) * RWORK(1) <= ||pinv(C)||_2 <= N^(1/4) * RWORK(1).
!>          بخلاف ذلك، RWORK(1) = -1.
!>          2. تحتوي RWORK(2) على عدد القيم المفردة التي تم حسابها كـ
!>          أصفار دقيقة في SGESVD المطبق على المثلث العلوي أو شبه المنحرف
!>          R (من تحليل QR الأولي). في حالة الخروج المبكر (لا يوجد استدعاء لـ
!>          SGESVD، مثل حالة المصفوفة الصفرية) RWORK(2) = -1.
!>
!>          إذا كان LIWORK أو LWORK أو LRWORK = -1، فعند الخروج، إذا كان INFO = 0،
!>          تعيد RWORK(1) الحد الأدنى لـ LRWORK.
!>

LRWORK

!>          LRWORK هو INTEGER.
!>          بعد مصفوفة RWORK.
!>          إذا كان JOBP ='P'، فإن LRWORK >= MAX(2, M).
!>          بخلاف ذلك، LRWORK >= 2
!>
!>          إذا كان LRWORK = -1، فيفترض استعلام مساحة العمل؛ يقوم الروتين
!>          بحساب وإرجاع الأحجام المثلى والحد الأدنى لمصفوفات
!>          WORK و IWORK و RWORK فقط، ولا يتم إصدار أي رسالة خطأ
!>          متعلقة بـ LWORK بواسطة XERBLA.
!>

INFO

!>          INFO هو INTEGER
!>          = 0: الخروج بنجاح.
!>          < 0: إذا كان INFO = -i، فإن الوسيط i كان له قيمة غير قانونية.
!>          > 0: إذا لم يتقارب SBDSQR، يحدد INFO عدد الأقطار الفائقة
!>          لشكل ثنائي قطري وسيط B (محسوب في SGESVD) لم يتقارب
!>          إلى الصفر.
!>

تفاصيل إضافية:

!>
!>   1. يتم ترميز حركة البيانات (تبديل المصفوفة) باستخدام حلقات FOR المتداخلة البسيطة
!>   لأن BLAS و LAPACK لا يوفران إجراءات مقابلة.
!>   يتم تحديد حلقات FOR المتداخلة هذه بسهولة في كود المصدر هذا - بواسطة
!>   عبارات CONTINUE التي تحمل علامات 11**. في نسخة محسنة من هذا الرمز، يجب
!>   استبدال حلقات FOR المتداخلة باستدعاءات لإجراء فرعي محسّن.
!>   2. يقوم هذا الرمز بقياس A بواسطة 1/SQRT(M) إذا كان أكبر قيمة لـ ABS(A(i,j)) قد تسبب
!>   في تجاوز سعة معيار العمود. هذا هو الحد الأدنى من الاحتياطات، ويترك لـ
!>   روتين SVD (CGESVD) للقيام بعملية القياس الاستباقية الخاصة به إذا تم اكتشاف
!>   تجاوز سعة أو نقص سعة محتمل. لتجنب المسح المتكرر للمصفوفة A،
!>   سوف تقوم التنفيذ الأمثل بجميع عمليات القياس اللازمة قبل استدعاء
!>   CGESVD، ويمكن إيقاف القياس في CGESVD.
!>   3. تعليقات أخرى متعلقة بتحسين الكود موجودة في التعليقات في
!>   الكود، محاطة بـ [[أقواس مزدوجة]].
!>
