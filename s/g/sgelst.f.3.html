توثيق الوظيفة/الروتين الفرعي

subroutine sgelst (character trans, integer m, integer n, integer nrhs, real, dimension( lda, * ) a, integer lda, real, dimension( ldb, * ) b, integer ldb, real, dimension( * ) work, integer lwork, integer info)

SGELST يحل الأنظمة الزائدة عن الحاجة أو غير المحددة للمصفوفات GE باستخدام تفكيك QR أو LQ مع تمثيل WY المدمج لـ Q.

الغرض:

!>
!> SGELST يحل الأنظمة الخطية الحقيقية الزائدة عن الحاجة أو غير المحددة
!> التي تتضمن مصفوفة M-by-N تسمى A، أو منقولتها، باستخدام تفكيك QR أو LQ
!> للمصفوفة A مع تمثيل WY المدمج للمتجه Q.
!>
!> يُفترض أن المصفوفة A ذات رتبة كاملة، ويتم توفير حماية أساسية فقط
!> ضد المصفوفات ذات الرتبة الناقصة. هذا الروتين الفرعي يكتشف فقط
!> النقص الدقيق في الرتبة، حيث يكون عنصر قطري في العامل المثلثي
!> للمصفوفة A يساوي صفرًا بالضبط.
!>
!> من الممكن أن تكون العناصر القطرية للعامل المثلثي
!> للمصفوفة A أرقامًا صغيرة جدًا (أصغر من العتبة) دون أن يشير هذا الروتين الفرعي
!> إلى حدوث خطأ. قد تكون الحلول المحسوبة لهذه المصفوفات شبه الناقصة في الرتبة
!> أقل دقة بسبب فقدان الدقة الرقمية.
!>
!> الخيارات التالية متاحة:
!>
!> 1. إذا كان TRANS = 'N' و m >= n: ابحث عن الحل بأقل مربعات للنظام
!>    الزائد عن الحاجة، أي حل مشكلة أقل المربعات
!>                 تصغير || B - A*X ||.
!>
!> 2. إذا كان TRANS = 'N' و m < n: ابحث عن الحل بأقل معيار للنظام
!>    غير المحدد A * X = B.
!>
!> 3. إذا كان TRANS = 'T' و m >= n: ابحث عن الحل بأقل معيار للنظام
!>    غير المحدد A**T * X = B.
!>
!> 4. إذا كان TRANS = 'T' و m < n: ابحث عن الحل بأقل مربعات للنظام
!>    الزائد عن الحاجة، أي حل مشكلة أقل المربعات
!>                 تصغير || B - A**T * X ||.
!>
!> يمكن معالجة العديد من متجهات الطرف الأيمن b ومتجهات الحل x
!> في استدعاء واحد؛ يتم تخزينها كأعمدة في
!> المصفوفة B للطرف الأيمن التي حجمها M-by-NRHS ومصفوفة الحل
!> X.
!>

المعلمات

TRANS 
!>          TRANS هو CHARACTER*1
!>          = 'N': النظام الخطي يتضمن المصفوفة A؛
!>          = 'T': النظام الخطي يتضمن منقولة المصفوفة A (A**T).
!>

M

!>          M هو INTEGER
!>          عدد صفوف المصفوفة A.  M >= 0.
!>

N

!>          N هو INTEGER
!>          عدد أعمدة المصفوفة A.  N >= 0.
!>

NRHS

!>          NRHS هو INTEGER
!>          عدد الأطراف اليمنى، أي عدد
!>          أعمدة المصفوفات B و X. NRHS >=0.
!>

A

!>          A هي مصفوفة REAL، ذات بُعد (LDA,N)
!>          عند الدخول: مصفوفة A بحجم M-by-N.
!>          عند الخروج:
!>            إذا كان M >= N، يتم استبدال A بتفاصيل تفكيك QR الخاص بها
!>                       كما تم إرجاعها بواسطة SGEQRT؛
!>            إذا كان M <  N، يتم استبدال A بتفاصيل تفكيك LQ الخاص بها
!>                       كما تم إرجاعها بواسطة SGELQT.
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرائد للمصفوفة A.  LDA >= max(1,M).
!>

B

!>          B هي مصفوفة REAL، ذات بُعد (LDB,NRHS)
!>          عند الدخول: مصفوفة B للأطراف اليمنى، مخزنة
!>          كأعمدة؛ B بحجم M-by-NRHS إذا كان TRANS = 'N'، أو N-by-NRHS
!>          إذا كان TRANS = 'T'.
!>          عند الخروج: إذا كان INFO = 0، يتم استبدال B بمتجهات الحل
!>          المخزنة كأعمدة:
!>          إذا كان TRANS = 'N' و m >= n، الصفوف من 1 إلى n في B تحتوي على متجهات حل أقل المربعات؛ مجموع مربعات البواقي
!!          للحل في كل عمود يُعطى كمجموع مربعات
!>          العناصر من N+1 إلى M في ذلك العمود؛
!>          إذا كان TRANS = 'N' و m < n، الصفوف من 1 إلى N في B تحتوي على متجهات الحل بأقل معيار؛
!>          إذا كان TRANS = 'T' و m >= n، الصفوف من 1 إلى M في B تحتوي على متجهات الحل بأقل معيار؛
!>          إذا كان TRANS = 'T' و m < n، الصفوف من 1 إلى M في B تحتوي على متجهات حل أقل المربعات؛ مجموع مربعات البواقي
!>          للحل في كل عمود يُعطى كمجموع
!>          مربعات العناصر من M+1 إلى N في ذلك العمود.
!>

LDB

!>          LDB هو INTEGER
!>          البعد الرائد للمصفوفة B. LDB >= MAX(1,M,N).
!>

WORK

!>          WORK هي مصفوفة REAL، ذات بُعد (MAX(1,LWORK))
!>          عند الخروج: إذا كان INFO = 0، تُرجع WORK(1) قيمة LWORK المثلى.
!>

LWORK

!>          LWORK هو INTEGER
!>          بُعد المصفوفة WORK.
!>          LWORK >= max( 1, MN + max( MN, NRHS ) ).
!>          لأداء مثالي:
!>          LWORK >= max( 1, (MN + max( MN, NRHS ))*NB ).
!>          حيث MN = min(M,N) و NB هو حجم الكتلة الأمثل.
!>
!>          إذا كان LWORK = -1، يتم افتراض استعلام عن مساحة العمل؛ الروتين
!>          يحسب فقط الحجم الأمثل للمصفوفة WORK، ويعيد
!>          هذه القيمة كإدخال أول للمصفوفة WORK، ولا يتم إصدار أي رسالة خطأ
!>          متعلقة بـ LWORK بواسطة XERBLA.
!>

INFO

!>          INFO هو INTEGER
!>          = 0:  انتهاء ناجح
!>          < 0:  إذا كان INFO = -i، فإن المعامل i كان له قيمة غير قانونية
!>          > 0:  إذا كان INFO =  i، فإن العنصر القطري i للمصفوفة
!>                العاملة المثلثية للمصفوفة A يساوي صفرًا بالضبط، بحيث لا تمتلك A
!>                رتبة كاملة؛ لم يكن من الممكن حساب حل أقل المربعات.
!>
