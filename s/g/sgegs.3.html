subroutine sgegs (character jobvsl, character jobvsr, integer n, real, dimension( lda, * ) a, integer lda, real, dimension( ldb, * ) b, integer ldb, real, dimension( * ) alphar, real, dimension( * ) alphai, real, dimension( * ) beta, real, dimension( ldvsl, * ) vsl, integer ldvsl, real, dimension( ldvsr, * ) vsr, integer ldvsr, real, dimension( * ) work, integer lwork, integer info)

SGEGS يحسب القيم الذاتية، الشكل الحقيقي لشور، واختيارياً، متجهات شور اليسرى و/أو اليمنى لزوج المصفوفات الحقيقي (A,B)

الغرض:

!>
!> تم إهمال هذا الروتين الفرعي واستبداله بالروتين الفرعي SGGES.
!>
!> SGEGS يحسب القيم الذاتية، الشكل الحقيقي لشور، واختيارياً،
!> متجهات شور اليسرى و/أو اليمنى لزوج المصفوفات الحقيقي (A,B).
!> بالنظر إلى مصفوفتين مربعين A و B، فإن عامل تفريق شور الحقيقي المعمم له الشكل
!>
!>   A = Q*S*Z**T,  B = Q*T*Z**T
!>
!> حيث Q و Z هي مصفوفات متعامدة، T هي مصفوفة مثلثة عليا، و S
!> هي مصفوفة شبه مثلثة عليا مع كتل قطرية بحجم 1x1 و 2x2، الكتل 2x2 تتوافق مع
!> أزواج مترافقة من القيم الذاتية لـ (A,B). أعمدة Q هي متجهات شور اليسرى
!> وأعمدة Z هي متجهات شور اليمنى.
!>
!> إذا كانت القيم الذاتية لـ (A,B) فقط مطلوبة، يجب استخدام الروتين السائق
!> SGEGV بدلاً من ذلك. راجع SGEGV لوصف القيم الذاتية لمشكلة القيم الذاتية غير المتماثلة المعممة
!> (GNEP).
!>

المعلمات

JOBVSL 
!>          JOBVSL هو CHARACTER*1
!>          = 'N':  لا تحسب متجهات شور اليسرى؛
!>          = 'V':  احسب متجهات شور اليسرى (معادة في VSL).
!>

JOBVSR

!>          JOBVSR هو CHARACTER*1
!>          = 'N':  لا تحسب متجهات شور اليمنى؛
!>          = 'V':  احسب متجهات شور اليمنى (معادة في VSR).
!>

N

!>          N هو INTEGER
!>          رتبة المصفوفات A، B، VSL، و VSR.  N >= 0.
!>

A

!>          A هو مصفوفة REAL، البعد (LDA, N)
!>          عند الإدخال، المصفوفة A.
!>          عند الخروج، المصفوفة شبه المثلثية العليا S من
!>          عامل تفريق شور المعمم الحقيقي.
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرائد للمصفوفة A.  LDA >= max(1,N).
!>

B

!>          B هو مصفوفة REAL، البعد (LDB, N)
!>          عند الإدخال، المصفوفة B.
!>          عند الخروج، المصفوفة المثلثية العليا T من
!>          عامل تفريق شور المعمم الحقيقي.
!>

LDB

!>          LDB هو INTEGER
!>          البعد الرائد للمصفوفة B.  LDB >= max(1,N).
!>

ALPHAR

!>          ALPHAR هي مصفوفة REAL، البعد (N)
!>          الأجزاء الحقيقية لكل عدد ألفا يحدد قيمة ذاتية
!>          لـ GNEP.
!>

ALPHAI

!>          ALPHAI هي مصفوفة REAL، البعد (N)
!>          الأجزاء التخيلية لكل عدد ألفا يحدد قيمة ذاتية
!>          لـ GNEP.  إذا كان ALPHAI(j) صفرًا، فإن القيمة الذاتية j
!>          حقيقية؛ إذا كانت موجبة، فإن القيمة الذاتية j و (j+1)
!>          هما زوج مترافق مركب، مع
!>          ALPHAI(j+1) = -ALPHAI(j).
!>

BETA

!>          BETA هي مصفوفة REAL، البعد (N)
!>          الأعداد بيتا التي تحدد القيم الذاتية لـ GNEP.
!>          معًا، الكميات ألفا = (ALPHAR(j),ALPHAI(j)) و
!>          بيتا = BETA(j) تمثل القيمة الذاتية j لزوج المصفوفات
!>          (A,B)، في أحد الأشكال lambda = alpha/beta أو
!>          mu = beta/alpha. نظرًا لأن lambda أو mu قد تفيضان،
!>          لا ينبغي حسابهما، بشكل عام.
!>

VSL

!>          VSL هي مصفوفة REAL، البعد (LDVSL,N)
!>          إذا كان JOBVSL = 'V'، مصفوفة متجهات شور اليسرى Q.
!>          لا يتم الرجوع إليها إذا كان JOBVSL = 'N'.
!>

LDVSL

!>          LDVSL هو INTEGER
!>          البعد الرائد للمصفوفة VSL. LDVSL >=1، و
!>          إذا كان JOBVSL = 'V'، LDVSL >= N.
!>

VSR

!>          VSR هي مصفوفة REAL، البعد (LDVSR,N)
!>          إذا كان JOBVSR = 'V'، مصفوفة متجهات شور اليمنى Z.
!>          لا يتم الرجوع إليها إذا كان JOBVSR = 'N'.
!>

WORK

!>          WORK هي مصفوفة REAL، البعد (MAX(1,LWORK))
!>          عند الخروج، إذا كان INFO = 0، فإن WORK(1) يعيد LWORK الأمثل.
!>

LWORK

!>          LWORK هو INTEGER
!>          بعد المصفوفة WORK.  LWORK >= max(1,4*N).
!>          لأداء جيد، يجب أن يكون LWORK أكبر بشكل عام.
!>          لحساب القيمة المثلى لـ LWORK، استدعِ ILAENV للحصول على
!>          أحجام الكتل (لـ SGEQRF، SORMQR، و SORGQR.) ثم احسب:
!>          NB  -- MAX من أحجام الكتل لـ SGEQRF، SORMQR، و SORGQR
!>          LWORK الأمثل هو  2*N + N*(NB+1).
!>
!>          إذا كان LWORK = -1، فسيتم افتراض استعلام مساحة العمل؛ الروتين
!>          يحسب فقط الحجم الأمثل للمصفوفة WORK، ويعيد
!>          هذه القيمة كالإدخال الأول للمصفوفة WORK، ولا يتم إصدار أي رسالة خطأ
!>          متعلقة بـ LWORK بواسطة XERBLA.
!>

INFO

!>          INFO هو INTEGER
!>          = 0:  خروج ناجح
!>          < 0:  إذا كان INFO = -i، فإن المعلمة i كانت ذات قيمة غير قانونية.
!>          = 1,...,N:
!>                فشل تكرار QZ.  (A,B) ليست في شكل شور،
!>                ولكن يجب أن تكون ALPHAR(j)، ALPHAI(j)، و BETA(j)
!>                صحيحة لـ j=INFO+1,...,N.
!>          > N:  أخطاء تشير عادةً إلى مشاكل LAPACK:
!>                =N+1: خطأ عودة من SGGBAL
!>                =N+2: خطأ عودة من SGEQRF
!>                =N+3: خطأ عودة من SORMQR
!>                =N+4: خطأ عودة من SORGQR
!>                =N+5: خطأ عودة من SGGHRD
!>                =N+6: خطأ عودة من SHGEQZ (غير التكرار الفاشل)
!>                =N+7: خطأ عودة من SGGBAK (حساب VSL)
!>                =N+8: خطأ عودة من SGGBAK (حساب VSR)
!>                =N+9: خطأ عودة من SLASCL (أماكن مختلفة)
!>

sgegs(3)

الاسم

sgegs - يقوم بحساب التحلل العام لـ QZ لمصفوفة A ذات القيم الحقيقية والتحلل العام لـ QZ لمصفوفة B ذات القيم الحقيقية.

الملخص

SUBROUTINE SGEGS(A, LDA, B, LDB, ALPHA, BETA, Q, LDQ, Z, LDZ, WORK, LWORK, INFO)

توثيق الدالة/الإجراء الفرعي

هذه الدالة هي جزء من مكتبة LAPACK، وهي مكتبة برمجيات حسابية عددية لتناسب حل المشكلات الخطية.

وظيفة sgegs هي حساب التحلل العام لـ QZ لمصفوفة A ذات القيم الحقيقية والتحلل العام لـ QZ لمصفوفة B ذات القيم الحقيقية. التحلل العام لـ QZ هو:

A = Q * S * ZT

B = Q * T * ZT

حيث Q و Z مصفوفتان متعامدتان، و S و T مصفوفتان مثلثيتان علويتان.

تُستخدم هذه الدالة في العديد من التطبيقات، بما في ذلك حل مشكلات القيم الذاتية المعممة.
