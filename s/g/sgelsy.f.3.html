subroutine sgelsy (integer m, integer n, integer nrhs, real, dimension( lda, * ) a, integer lda, real, dimension( ldb, * ) b, integer ldb, integer, dimension( * ) jpvt, real rcond, integer rank, real, dimension( * ) work, integer lwork, integer info)

SGELSY يحل أنظمة مفرطة التحديد أو ناقصة التحديد لمصفوفات GE

الغرض:

!>
!> SGELSY يحسب الحل ذي المعيار الأدنى لمشكلة المربعات الصغرى الخطية الحقيقية:
!>     تصغير || A * X - B ||
!> باستخدام تحليل متعامد كامل لـ A. A هي مصفوفة M-by-N
!> والتي قد تكون ذات رتبة ناقصة.
!>
!> يمكن التعامل مع عدة متجهات جانبية يمنى b ومتجهات حل x
!> في استدعاء واحد؛ يتم تخزينها كأعمدة في
!> مصفوفة الجانب الأيمن B ذات الأبعاد M-by-NRHS ومصفوفة الحل X ذات الأبعاد N-by-NRHS.
!>
!> تقوم الروتين أولاً بحساب تحليل QR مع تثبيت الأعمدة:
!>     A * P = Q * [ R11 R12 ]
!>                 [  0  R22 ]
!> مع تعريف R11 كأكبر مصفوفة فرعية رائدة ذات تقدير
!> رقم شرطي أقل من 1/RCOND. رتبة R11، RANK،
!> هي الرتبة الفعالة لـ A.
!>
!> بعد ذلك، تعتبر R22 ضئيلة، ويتم إفناء R12
!> بواسطة تحويلات متعامدة من اليمين، والوصول إلى
!> التحليل المتعامد الكامل:
!>    A * P = Q * [ T11 0 ] * Z
!>                [  0  0 ]
!> الحل ذو المعيار الأدنى هو إذن
!>    X = P * Z**T [ inv(T11)*Q1**T*B ]
!>                 [        0         ]
!> حيث تتكون Q1 من أول RANK أعمدة من Q.
!>
!> هذا الروتين مطابق بشكل أساسي لـ xGELSX الأصلي باستثناء
!> ثلاثة اختلافات:
!>   o تم استبدال استدعاء الإجراء الفرعي xGEQPF بـ
!>     استدعاء الإجراء الفرعي xGEQP3. هذا الإجراء الفرعي هو إصدار Blas-3
!>     لتحليل QR مع تثبيت الأعمدة.
!>   o يتم تحديث المصفوفة B (الجانب الأيمن) باستخدام Blas-3.
!>   o تبديل المصفوفة B (الجانب الأيمن) أسرع وأبسط.
!>

المعلمات

M 
!>          M هو عدد صحيح (INTEGER)
!>          عدد الصفوف في المصفوفة A. M >= 0.
!>

N

!>          N هو عدد صحيح (INTEGER)
!>          عدد الأعمدة في المصفوفة A. N >= 0.
!>

NRHS

!>          NRHS هو عدد صحيح (INTEGER)
!>          عدد الجوانب اليمنى، أي عدد
!>          الأعمدة في المصفوفتين B و X. NRHS >= 0.
!>

A

!>          A هي مصفوفة حقيقية (REAL array)، ذات أبعاد (LDA,N)
!>          عند الإدخال، المصفوفة A ذات الأبعاد M-by-N.
!>          عند الخروج، تم استبدال A بتفاصيل
!>          تحليلها المتعامد الكامل.
!>

LDA

!>          LDA هو عدد صحيح (INTEGER)
!>          البعد الرائد في المصفوفة A. LDA >= max(1,M).
!>

B

!>          B هي مصفوفة حقيقية (REAL array)، ذات أبعاد (LDB,NRHS)
!>          عند الإدخال، مصفوفة الجانب الأيمن B ذات الأبعاد M-by-NRHS.
!>          عند الخروج، مصفوفة الحل X ذات الأبعاد N-by-NRHS.
!>          إذا كانت M = 0 أو N = 0، لا يتم الرجوع إلى B.
!>

LDB

!>          LDB هو عدد صحيح (INTEGER)
!>          البعد الرائد في المصفوفة B. LDB >= max(1,M,N).
!>

JPVT

!>          JPVT هي مصفوفة أعداد صحيحة (INTEGER array)، ذات أبعاد (N)
!>          عند الإدخال، إذا كان JPVT(i) .ne. 0، يتم تبديل العمود i من A
!>          إلى مقدمة AP، وإلا فإن العمود i هو عمود حر.
!>          عند الخروج، إذا كان JPVT(i) = k، فإن العمود i من AP
!>          كان العمود k من A.
!>

RCOND

!>          RCOND هو عدد حقيقي (REAL)
!>          يستخدم RCOND لتحديد الرتبة الفعالة لـ A، والتي
!>          تعرف بأنها رتبة أكبر مصفوفة فرعية مثلثة رائدة
!>          في تحليل QR مع تثبيت A،
!>          والتي رقم شرطها المقدر < 1/RCOND.
!>

RANK

!>          RANK هو عدد صحيح (INTEGER)
!>          الرتبة الفعالة لـ A، أي رتبة المصفوفة الفرعية
!>          R11. هذا هو نفس رتبة المصفوفة الفرعية
!>          T11 في التحليل المتعامد الكامل لـ A.
!>          إذا كانت NRHS = 0، فإن RANK = 0 عند الخروج.
!>

WORK

!>          WORK هي مصفوفة حقيقية (REAL array)، ذات أبعاد (MAX(1,LWORK))
!>          عند الخروج، إذا كان INFO = 0، فإن WORK(1) يعيد LWORK الأمثل.
!>

LWORK

!>          LWORK هو عدد صحيح (INTEGER)
!>          البعد في المصفوفة WORK.
!>          تتطلب استراتيجية عدم الحظر أن:
!>             LWORK >= MAX( MN+3*N+1, 2*MN+NRHS ),
!>          حيث MN = min( M, N ).
!>          تتطلب خوارزمية الكتلة أن:
!>             LWORK >= MAX( MN+2*N+NB*(N+1), 2*MN+NB*NRHS ),
!>          حيث NB هو حد علوي لحجم الكتلة الذي تم إرجاعه
!>          بواسطة ILAENV للإجراءات الفرعية SGEQP3, STZRZF, STZRQF, SORMQR,
!>          و SORMRZ.
!>
!>          إذا كان LWORK = -1، فإنه يفترض استعلام عن مساحة العمل؛ الروتين
!>          يحسب فقط الحجم الأمثل للمصفوفة WORK، ويعيد
!>          هذه القيمة كإدخال أول في المصفوفة WORK، ولا يتم إصدار
!>          أي رسالة خطأ متعلقة بـ LWORK بواسطة XERBLA.
!>

INFO

!>          INFO هو عدد صحيح (INTEGER)
!>          = 0: انتهاء ناجح
!>          < 0: إذا كان INFO = -i، فإن المعلمة i كانت لها قيمة غير قانونية.
!>
