توثيق الوظيفة/الروتين الفرعي

subroutine sgels (character trans, integer m, integer n, integer nrhs, real, dimension( lda, * ) a, integer lda, real, dimension( ldb, * ) b, integer ldb, real, dimension( * ) work, integer lwork, integer info)

SGELS يحل أنظمة GE ذات المصفوفات الزائدة أو الناقصة

الغرض:

!>
!> SGELS يحل أنظمة خطية حقيقية زائدة أو ناقصة
!> تتضمن مصفوفة A بحجم M × N، أو منقولتها، باستخدام تحليل QR أو LQ
!> للمصفوفة A.
!>
!> يُفترض أن المصفوفة A لها رتبة كاملة، ولا يتم توفير سوى حماية أولية
!> للمصفوفات ذات الرتبة الناقصة. هذا الروتين الفرعي يكتشف فقط
!> النقصان التام في الرتبة، حيث يكون عنصر قطري في العامل المثلثي
!> للمصفوفة A يساوي صفرًا تمامًا.
!>
!> من الممكن أن تكون إحدى (أو أكثر) العناصر القطرية للعامل المثلثي
!> للمصفوفة A أرقامًا صغيرة جدًا بشكل غير طبيعي دون أن يقوم هذا الروتين الفرعي بالإشارة
!> إلى خطأ. قد تكون الحلول المحسوبة لهذه المصفوفات شبه الناقصة الرتبة
!> أقل دقة بسبب فقدان الدقة الرقمية.
!>
!> الخيارات التالية متاحة:
!>
!> 1. إذا كان TRANS = 'N' و m >= n:  ابحث عن حل المربعات الصغرى لنظام
!>    زائد، أي، حل مشكلة المربعات الصغرى
!>                 تصغير || B - A*X ||.
!>
!> 2. إذا كان TRANS = 'N' و m < n:  ابحث عن حل المعيار الأدنى لنظام
!>    ناقص A * X = B.
!>
!> 3. إذا كان TRANS = 'T' و m >= n:  ابحث عن حل المعيار الأدنى لنظام
!>    ناقص A**T * X = B.
!>
!> 4. إذا كان TRANS = 'T' و m < n:  ابحث عن حل المربعات الصغرى لنظام
!>    زائد، أي، حل مشكلة المربعات الصغرى
!>                 تصغير || B - A**T * X ||.
!>
!> يمكن التعامل مع عدة متجهات للطرف الأيمن b ومتجهات حل x
!> في استدعاء واحد؛ يتم تخزينها كأعمدة في
!> المصفوفة B ذات الطرف الأيمن M × NRHS والمصفوفة X ذات الحل
!> N × NRHS.
!>

المعلمات

TRANS 
!>          TRANS هو CHARACTER*1
!>          = 'N': النظام الخطي يتضمن A؛
!>          = 'T': النظام الخطي يتضمن A**T.
!>

M

!>          M هو INTEGER
!>          عدد صفوف المصفوفة A.  M >= 0.
!>

N

!>          N هو INTEGER
!>          عدد أعمدة المصفوفة A.  N >= 0.
!>

NRHS

!>          NRHS هو INTEGER
!>          عدد الأطراف اليمنى، أي عدد
!>          أعمدة المصفوفات B و X. NRHS >=0.
!>

A

!>          A هو مصفوفة REAL، الأبعاد (LDA,N)
!>          عند الإدخال، المصفوفة A بحجم M × N.
!>          عند الخروج،
!>            إذا كان M >= N، يتم استبدال A بتفاصيل تحليل QR الخاص بها
!>                       كما تم إرجاعها بواسطة SGEQRF؛
!>            إذا كان M <  N، يتم استبدال A بتفاصيل تحليل LQ الخاص بها
!>                       كما تم إرجاعها بواسطة SGELQF.
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرئيسي للمصفوفة A.  LDA >= max(1,M).
!>

B

!>          B هو مصفوفة REAL، الأبعاد (LDB,NRHS)
!>          عند الإدخال، مصفوفة B للمتجهات الطرفية اليمنى، مخزنة
!>          حسب الأعمدة؛ B بحجم M × NRHS إذا كان TRANS = 'N'، أو N × NRHS
!>          إذا كان TRANS = 'T'.
!>          عند الخروج، إذا كان INFO = 0، يتم استبدال B بمتجهات الحل
!>          المخزنة حسب الأعمدة:
!>          إذا كان TRANS = 'N' و m >= n، فإن الصفوف من 1 إلى n في B تحتوي على متجهات
!>          حل المربعات الصغرى؛ مجموع مربعات البواقي للحل
!>          في كل عمود يُعطى بمجموع مربعات العناصر من N+1 إلى M في ذلك العمود؛
!>          إذا كان TRANS = 'N' و m < n، فإن الصفوف من 1 إلى N في B تحتوي على متجهات
!>          حل المعيار الأدنى؛
!>          إذا كان TRANS = 'T' و m >= n، فإن الصفوف من 1 إلى M في B تحتوي على متجهات
!>          حل المعيار الأدنى؛
!>          إذا كان TRANS = 'T' و m < n، فإن الصفوف من 1 إلى M في B تحتوي على متجهات
!>          حل المربعات الصغرى؛ مجموع مربعات البواقي للحل
!>          في كل عمود يُعطى بمجموع مربعات العناصر من M+1 إلى N في ذلك العمود.
!>

LDB

!>          LDB هو INTEGER
!>          البعد الرئيسي للمصفوفة B. LDB >= MAX(1,M,N).
!>

WORK

!>          WORK هي مصفوفة REAL، الأبعاد (MAX(1,LWORK))
!>          عند الخروج، إذا كان INFO = 0، فإن WORK(1) تُرجع LWORK المثلى.
!>

LWORK

!>          LWORK هو INTEGER
!>          بعد المصفوفة WORK.
!>          LWORK >= max( 1, MN + max( MN, NRHS ) ).
!>          لأداء أمثل،
!>          LWORK >= max( 1, MN + max( MN, NRHS )*NB ).
!>          حيث MN = min(M,N) و NB هو حجم الكتلة الأمثل.
!>
!>          إذا كان LWORK = -1، فإنه يُفترض استعلام مساحة العمل؛ الروتين
!>          يقوم فقط بحساب الحجم الأمثل لمصفوفة WORK، ويرجع
!>          هذه القيمة كالإدخال الأول لمصفوفة WORK، ولا يصدر رسالة خطأ
!>          متعلقة بـ LWORK بواسطة XERBLA.
!>

INFO

!>          INFO هو INTEGER
!>          = 0:  خروج ناجح
!>          < 0:  إذا كان INFO = -i، فإن الوسيط i كان له قيمة غير قانونية
!>          > 0:  إذا كان INFO =  i، فإن العنصر القطري i من
!>                العامل المثلثي للمصفوفة A يساوي صفرًا تمامًا، وبالتالي فإن A ليس لديها
!>                رتبة كاملة؛ لم يكن من الممكن حساب حل المربعات الصغرى.
!>
