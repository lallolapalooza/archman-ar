الوصف

توفر دوال sg_get_fs_stats إحصائيات لأنظمة الملفات المركبة. تأخذ كلتا الدالتين معلمة إدخالات اختيارية، والتي تشير (عند تقديمها) إلى قيمة size_t لتلقي عدد إدخالات المتجه التي تم إرجاعها.

تقوم دالتا sg_get_fs_stats() و sg_get_fs_stats_r() بتسليم إحصائيات نظام الملفات في اللحظة التي يتم فيها استدعاء الدالة. تقوم دالتا sg_get_fs_stats_diff() و sg_get_fs_stats_diff_between() بتسليم الفرق بين استدعائين لدالتي sg_get_fs_stats() أو sg_get_fs_stats_r()، على التوالي.

اختصار واجهة برمجة التطبيقات

الدالة تُرجع مالك البيانات
sg_get_fs_stats sg_fs_stats * libstatgrab (متغير محلي للخيط)
sg_get_fs_stats_r sg_fs_stats * المستدعي
sg_get_fs_stats_diff sg_fs_stats * libstatgrab (متغير محلي للخيط)
sg_get_fs_stats_diff_between sg_fs_stats * المستدعي
sg_get_valid_filesystems char ** libstatgrab (عام)

يجب تحرير متجهات sg_fs_stats المستلمة من sg_get_fs_stats_r() أو sg_get_fs_stats_diff_between() باستخدام sg_free_fs_stats() عندما لا تكون هناك حاجة إليها بعد الآن. المستدعي مسؤول عن القيام بذلك.

تأتي مكتبة statgrab مع قائمة مدمجة لأنواع أنظمة الملفات الصالحة اعتمادًا على نظام التشغيل الذي تم تجميعها من أجله. توفر بعض أنظمة التشغيل بالإضافة إلى ذلك واجهة برمجة تطبيقات لمعرفة أنواع أنظمة الملفات المعروفة أو الصحيحة لمثيل نظام التشغيل قيد التشغيل، والتي يتم استخدامها عند اكتشافها. ومع ذلك، هناك مشاكل معروفة عند جمع إحصائيات نظام الملفات: يتم تركيب أنظمة الملفات الشبكية من خوادم تم إطلاقها، ويقوم مطورو أنظمة الملفات بتشغيل برنامج تشغيل تجريبي إلخ.

لمنع العمليات من التعليق في الحصول على إحصائيات نظام الملفات أو السماح للمطورين باختبار برامج التشغيل الخاصة بهم، قد تعدل العمليات قائمة أنظمة الملفات الصالحة باستخدام sg_get_valid_filesystems() و sg_set_valid_filesystems(). قائمة المعلمات char * التي تعمل معها كلتا الدالتين تنتهي دائمًا بعنصر يشير إلى NULL.

يجب عدم تعديل القائمة التي تم إرجاعها بواسطة sg_get_valid_filesystems(). قم دائمًا بنسخ القائمة إلى هيكل خاص بك، إذا كنت تخطط لتوسيع القائمة أو تقليصها:

مثال على إزالة نظام الملفات الشبكي

int compare_fs_type(const void *va, const void *vb) {
    const char **a = (const char **)va;
    const char **b = (const char **)vb;
    return strcmp( *a, *b );
}
void filter_network_fs_types(void) {
    /* أسماء أنظمة الملفات الشبكية المعروفة على منصات مختلفة */
    const char *nfs_types[] = { "nfs", "nfs3", "nfs4", "cifs", "smbfs", "samba" };
    const size_t nfs_types_count = sizeof(nfs_types) / sizeof(nfs_types[0])
    size_t fs_entries = 0;
    const char **orig_valid_fs = sg_get_valid_filesystems(&fs_entries);
    /* نسخ إلى ذاكرة خاصة لتعديل القائمة */
    char **valid_fs = calloc( entries + 1, sizeof(valid_fs[0]) );
    memcpy( valid_fs, orig_valid_fs, (entries + 1) * sizeof(valid_fs[0]) );
    size_t i;
    for( i = 0; i < nfs_types_count; ++i ) {
        char **inv_fs = bsearch( &nfs_types[i], &valid_fs[0],
                                 fs_entries, sizeof(valid_fs[0]),
                                 compare_fs_type );
        if( NULL != inv_fs ) {
            /* نسخ بما في ذلك المؤشر الفارغ اللاحق */
            memmove( inv_fs, inv_fs + 1, fs_entries - (inv_fs - valid_fs) );
            --fs_entries;
        }
    }
    sg_set_valid_filesystems( valid_fs );
    free( valid_fs );
}

لاحظ أنه لا حاجة لنسخ السلاسل النصية الموجودة في قائمة أنظمة الملفات الصالحة في المثال أعلاه - لم يتم تعديلها.

قد تصبح القائمة التي تم إرجاعها بواسطة sg_get_valid_filesystems() غير صالحة عند استخدامها أثناء قيام العملية باستدعاءات لـ sg_set_valid_filesystems(). لا تقوم أي من دوال sg_fs_stats بحماية الوصول إلى التخزين المستخدم عالميًا حيث يتم الاحتفاظ بالنسخة الخاصة من قائمة أنظمة الملفات الصالحة. تقع على عاتق المستدعي مسؤولية عدم خلط استدعاءات التكوين مع استدعاءات جلب الإحصائيات.

بالإضافة إلى ذلك، تتوفر دالتان دعم لـ qsort(3): sg_fs_compare_device_name() و sg_fs_compare_mnt_point().

مثال على الفرز

size_t entries;
sg_fs_stats *fs_stats = NULL;
while( NULL != ( fs_stats = sg_get_fs_stats_diff(&entries) ) ) {
    /* ترتيب الإدخالات أبجديًا باستخدام نقطة التركيب */
    qsort( fs_stats, entries, sizeof(fs_stats[0]), &sg_fs_compare_mnt_point );
    show_fs_stats( fs_stats );
}

قيم الإرجاع

sg_get_fs_stats تُرجع مؤشرًا إلى بنية من النوع sg_fs_stats.

typedef enum {
        sg_fs_unknown  = 0,
        sg_fs_regular  = 1 << 0,
        sg_fs_special  = 1 << 1,
        sg_fs_loopback = 1 << 2,
        sg_fs_remote   = 1 << 3,
        sg_fs_local    = (sg_fs_regular | sg_fs_special),
        sg_fs_alltypes = (sg_fs_regular | sg_fs_special | sg_fs_loopback | sg_fs_remote)
} sg_fs_device_type;

typedef struct {
        char *device_name;
        char *fs_type;
        char *mnt_point;
        sg_fs_device_type device_type;
        unsigned long long size;
        unsigned long long used;
        unsigned long long free;
        unsigned long long avail;
        unsigned long long total_inodes;
        unsigned long long used_inodes;
        unsigned long long free_inodes;
        unsigned long long avail_inodes;
        unsigned long long io_size;
        unsigned long long block_size;
        unsigned long long total_blocks;
        unsigned long long free_blocks;
        unsigned long long used_blocks;
        unsigned long long avail_blocks;
        time_t systime;
} sg_fs_stats;
device_name
الاسم المعروف لنظام التشغيل. (على سبيل المثال، في لينكس قد يكون hda)
fs_type
نوع نظام الملفات لنظام الملفات (على سبيل المثال، hpfs أو ufs).
mnt_point
نقطة التركيب التي يتم تركيب نظام الملفات عليها.
device_type
نوع جهاز نظام الملفات، غير مملوء حاليًا ودائمًا sg_fs_unknown.
size
الحجم الإجمالي، بالبايت، لنظام الملفات.

size = used + free

used
مقدار المساحة، بالبايت، المستخدمة في نظام الملفات.
avail
مقدار المساحة، بالبايت، المتاحة في نظام الملفات لـ
