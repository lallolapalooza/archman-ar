توثيق الوظيفة/الروتين الفرعي

subroutine sgegv (character jobvl, character jobvr, integer n, real, dimension( lda, * ) a, integer lda, real, dimension( ldb, * ) b, integer ldb, real, dimension( * ) alphar, real, dimension( * ) alphai, real, dimension( * ) beta, real, dimension( ldvl, * ) vl, integer ldvl, real, dimension( ldvr, * ) vr, integer ldvr, real, dimension( * ) work, integer lwork, integer info)

SGEGV تحسب القيم الذاتية، واختياريًا، المتجهات الذاتية اليسرى و/أو اليمنى لزوج مصفوفات حقيقي (A,B).

الغرض:

!>
!> هذا الروتين مهمل وتم استبداله بالروتين SGGEV.
!>
!> SGEGV يحسب القيم الذاتية، واختياريًا، المتجهات الذاتية اليسرى و/أو اليمنى لزوج مصفوفات حقيقي (A,B).
!> بالنظر إلى مصفوفتين مربعتين A و B،
!> فإن مشكلة القيم الذاتية المعممة غير المتماثلة (GNEP) هي إيجاد القيم الذاتية lambda والمتجهات الذاتية المقابلة (غير الصفرية) x بحيث
!>
!>    A*x = lambda*B*x.
!>
!> شكل بديل هو إيجاد القيم الذاتية mu والمتجهات الذاتية المقابلة y بحيث
!>
!>    mu*A*y = B*y.
!>
!> هذان الشكلان متكافئان مع mu = 1/lambda و x = y إذا كان كل من lambda و mu غير صفريين. للتعامل مع الحالة التي يكون فيها
!> lambda أو mu صفرًا أو صغيرًا، يتم إرجاع قيمتين alpha و beta لكل قيمة ذاتية، بحيث lambda = alpha/beta و
!> mu = beta/alpha.
!>
!> المتجهات x و y في المعادلات أعلاه هي متجهات ذاتية يمنى لزوج المصفوفات (A,B). المتجهات u و v التي تحقق
!>
!>    u**H*A = lambda*u**H*B  أو  mu*v**H*A = v**H*B
!>
!> هي متجهات ذاتية يسرى لـ (A,B).
!>
!> ملاحظة: هذا الروتين يقوم على A و B
!>

المعلمات:

JOBVL 
!>          JOBVL هو CHARACTER*1
!>          = 'N':  لا تحسب المتجهات الذاتية المعممة اليسرى؛
!>          = 'V':  احسب المتجهات الذاتية المعممة اليسرى (معادة
!>                  في VL).
!>

JOBVR

!>          JOBVR هو CHARACTER*1
!>          = 'N':  لا تحسب المتجهات الذاتية المعممة اليمنى؛
!>          = 'V':  احسب المتجهات الذاتية المعممة اليمنى (معادة
!>                  في VR).
!>

N

!>          N هو INTEGER
!>          رتبة المصفوفات A، B، VL، و VR.  N >= 0.
!>

A

!>          A هو REAL array، بعد (LDA, N)
!>          عند الدخول، المصفوفة A.
!>          إذا كان JOBVL = 'V' أو JOBVR = 'V'، فإن عند الخروج A
!>          تحتوي على شكل Schur الحقيقي لـ A من التحلل Schur المعمم
!>          للزوج (A,B) بعد الموازنة.
!>          إذا لم يتم حساب أي متجهات ذاتية، فإن الكتل القطرية فقط
!>          من شكل Schur ستكون صحيحة. راجع SGGHRD و
!>          SHGEQZ للتفاصيل.
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرئيسي لـ A.  LDA >= max(1,N).
!>

B

!>          B هو REAL array، بعد (LDB, N)
!>          عند الدخول، المصفوفة B.
!>          إذا كان JOBVL = 'V' أو JOBVR = 'V'، فإن عند الخروج B تحتوي على
!>          المصفوفة المثلثية العليا التي تم الحصول عليها من B في التحلل Schur المعمم
!>          للزوج (A,B) بعد الموازنة.
!>          إذا لم يتم حساب أي متجهات ذاتية، فإن العناصر فقط من
!>          B المقابلة للكتل القطرية من شكل Schur لـ
!>          A ستكون صحيحة. راجع SGGHRD و SHGEQZ للتفاصيل.
!>

LDB

!>          LDB هو INTEGER
!>          البعد الرئيسي لـ B.  LDB >= max(1,N).
!>

ALPHAR

!>          ALPHAR هو REAL array، بعد (N)
!>          الأجزاء الحقيقية لكل عدد ألفا يحدد قيمة ذاتية لـ
!>          GNEP.
!>

ALPHAI

!>          ALPHAI هو REAL array، بعد (N)
!>          الأجزاء التخيلية لكل عدد ألفا يحدد قيمة ذاتية لـ
!>          GNEP.  إذا كان ALPHAI(j) صفرًا، فإن القيمة الذاتية j
!>          حقيقية؛ إذا كانت موجبة، فإن القيمتين الذاتيتين j و
!>          (j+1) هما زوج مترافق مركب، مع
!>          ALPHAI(j+1) = -ALPHAI(j).
!>

BETA

!>          BETA هو REAL array، بعد (N)
!>          الأعداد بيتا التي تحدد القيم الذاتية لـ GNEP.
!>
!>          معًا، الكميتان alpha = (ALPHAR(j),ALPHAI(j)) و
!>          beta = BETA(j) تمثلان القيمة الذاتية j لزوج المصفوفات
!>          (A,B)، في أحد الأشكال lambda = alpha/beta أو
!>          mu = beta/alpha. نظرًا لأن أيًا من lambda أو mu قد يحدث فيه تجاوز،
!>          لا ينبغي حسابهما بشكل عام.
!>

VL

!>          VL هو REAL array، بعد (LDVL,N)
!>          إذا كان JOBVL = 'V'، يتم تخزين المتجهات الذاتية اليسرى u(j)
!>          في أعمدة VL، بنفس ترتيب قيمها الذاتية.
!>          إذا كانت القيمة الذاتية j حقيقية، فإن u(j) = VL(:,j).
!>          إذا شكلت القيمتان الذاتيتان j و (j+1) زوجًا مترافقًا مركبًا، فإن
!>             u(j) = VL(:,j) + i*VL(:,j+1)
!>          و
!>            u(j+1) = VL(:,j) - i*VL(:,j+1).
!>
!>          يتم قياس كل متجه ذاتي بحيث يكون أكبر مكون له
!>          abs(الجزء الحقيقي) + abs(الجزء التخيلي) = 1، باستثناء المتجهات الذاتية
!>          المقابلة لقيمة ذاتية حيث alpha = beta = 0، والتي
!>          يتم تعيينها إلى الصفر.
!>          لا تتم الإشارة إليه إذا كان JOBVL = 'N'.
!>

LDVL

!>          LDVL هو INTEGER
!>          البعد الرئيسي للمصفوفة VL. LDVL >= 1، و
!>          إذا كان JOBVL = 'V'، فإن LDVL >= N.
!>

VR

!>          VR هو REAL array، بعد (LDVR,N)
!>          إذا كان JOBVR = 'V'، يتم تخزين المتجهات الذاتية اليمنى x(j)
!>          في أعمدة VR، بنفس ترتيب قيمها الذاتية.
!>          إذا كانت القيمة الذاتية j حقيقية، فإن x(j) = VR(:,j).
!>          إذا شكلت القيمتان الذاتيتان j و (j+1) زوجًا مترافقًا مركبًا، فإن
!>            x(j) = VR(:,j) + i*VR(:,j+1)
!>          و
!>            x(j+1) = VR(:,j) - i*VR(:,j+1).
!>
!>          يتم قياس كل متجه ذاتي بحيث يكون أكبر مكون له
!>          abs(الجزء الحقيقي) + abs(الجزء التخيلي) = 1، باستثناء القيم الذاتية
!>          المقابلة لقيمة ذاتية حيث alpha = beta = 0، والتي
!>          يتم تعيينها إلى الصفر.
!>          لا تتم الإشارة إليه إذا كان JOBVR = 'N'.
!>

LDVR

!>          LDVR هو INTEGER
!>          البعد الرئيسي للمصفوفة VR. LDVR >= 1، و
!>          إذا كان JOBVR = 'V'، فإن LDVR >= N.
!>

WORK

!>          WORK هو REAL array، بعد (MAX(1,LWORK))
!>          عند الخروج، إذا كان INFO = 0، فإن WORK(1) يعيد LWORK المثلى.
!>

LWORK

!>          LWORK هو INTEGER
!>          بعد المصفوفة WORK.  LWORK >= max(1,8*N).
!>          لأداء جيد، يجب أن يكون LWORK أكبر بشكل عام.
!>          لحساب القيمة المثلى لـ LWORK، اتصل بـ ILAENV للحصول على
!>          أحجام الكتل (لـ SGEQRF، SORMQR، و SORGQR). ثم احسب:
!>          NB  -- MAX لأحجام الكتل لـ SGEQRF، SORMQR، و SORGQR؛
!>          LWORK المثلى هي:
!>              2*N + MAX( 6*N, N*(NB+1) ).
!>
!>          إذا كان LWORK = -1، فسيتم افتراض استعلام مساحة عمل؛ تقوم الروتين فقط
!>          بحساب الحجم الأمثل لمصفوفة WORK، وتعيد
!>          هذه القيمة كالإدخال الأول لمصفوفة WORK، ولا يتم إصدار أي رسالة خطأ
!>          متعلقة بـ LWORK بواسطة XERBLA.
!>
