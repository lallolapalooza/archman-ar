subroutine sggsvd3 (character jobu, character jobv, character jobq, integer m, integer n, integer p, integer k, integer l, real, dimension( lda, * ) a, integer lda, real, dimension( ldb, * ) b, integer ldb, real, dimension( * ) alpha, real, dimension( * ) beta, real, dimension( ldu, * ) u, integer ldu, real, dimension( ldv, * ) v, integer ldv, real, dimension( ldq, * ) q, integer ldq, real, dimension( * ) work, integer lwork, integer, dimension( * ) iwork, integer info)

SGGSVD3 تحسب التحلل القيمة الفردية المعمم (GSVD) لمصفوفة حقيقية M-by-N A ومصفوفة حقيقية P-by-N B:

الغرض:

!>
!> SGGSVD3 تحسب التحلل القيمة الفردية المعمم (GSVD)
!> لمصفوفة حقيقية M-by-N A ومصفوفة حقيقية P-by-N B:
!>
!>       U**T*A*Q = D1*( 0 R ),    V**T*B*Q = D2*( 0 R )
!>
!> حيث U، V و Q هي مصفوفات متعامدة.
!> لتكن K+L = الرتبة العددية الفعالة للمصفوفة (A**T,B**T)**T,
!> فإن R هي مصفوفة غير شاذة مثلثة علوية بحجم K+L-by-K+L، و D1 و
!> D2 هي مصفوفات بحجم M-by-(K+L) و P-by-(K+L) على التوالي:
!>
!> إذا كان M-K-L >= 0,
!>
!>                     K  L
!>        D1 =     K ( I  0 )
!>                 L ( 0  C )
!>             M-K-L ( 0  0 )
!>
!>                   K  L
!>        D2 =   L ( 0  S )
!>             P-L ( 0  0 )
!>
!>                 N-K-L  K    L
!>   ( 0 R ) = K (  0   R11  R12 )
!>             L (  0    0   R22 )
!>
!> حيث
!>
!>   C = diag( ALPHA(K+1), ... , ALPHA(K+L) ),
!>   S = diag( BETA(K+1),  ... , BETA(K+L) ),
!>   C**2 + S**2 = I.
!>
!>   R مخزنة في A(1:K+L,N-K-L+1:N) عند الخروج.
!>
!> إذا كان M-K-L < 0,
!>
!>                   K M-K K+L-M
!>        D1 =   K ( I  0    0   )
!>             M-K ( 0  C    0   )
!>
!>                     K M-K K+L-M
!>        D2 =   M-K ( 0  S    0  )
!>             K+L-M ( 0  0    I  )
!>               P-L ( 0  0    0  )
!>
!>                    N-K-L  K   M-K  K+L-M
!>   ( 0 R ) =     K ( 0    R11  R12  R13  )
!>               M-K ( 0     0   R22  R23  )
!>             K+L-M ( 0     0    0   R33  )
!>
!> حيث
!>
!>   C = diag( ALPHA(K+1), ... , ALPHA(M) ),
!>   S = diag( BETA(K+1),  ... , BETA(M) ),
!>   C**2 + S**2 = I.
!>
!>   (R11 R12 R13 ) مخزنة في A(1:M, N-K-L+1:N), و R33 مخزنة
!>   ( 0  R22 R23 )
!>   في B(M-K+1:L,N+M-K-L+1:N) عند الخروج.
!>
!> الروتين يحسب C، S، R، وبشكل اختياري مصفوفات التحويل المتعامدة
!> U، V و Q.
!>
!> على وجه الخصوص، إذا كانت B مصفوفة غير شاذة بحجم N-by-N، فإن GSVD
!> لـ A و B تعطي ضمنيًا SVD لـ A*inv(B):
!>                      A*inv(B) = U*(D1*inv(D2))*V**T.
!> إذا كان ( A**T,B**T)**T لديه أعمدة متعامدة، فإن GSVD لـ A و B
!> يساوي أيضًا تفكيك CS لـ A و B. علاوة على ذلك، يمكن استخدام GSVD
!> لاشتقاق حل مشكلة القيمة الذاتية:
!>                      A**T*A x = lambda* B**T*B x.
!> في بعض الأدبيات، يتم تقديم GSVD لـ A و B في الشكل
!>                  U**T*A*X = ( 0 D1 ),   V**T*B*X = ( 0 D2 )
!> حيث U و V متعامدتان و X غير شاذة، D1 و D2 "قطريتان".
!> يمكن تحويل شكل GSVD السابق إلى الشكل الأخير
!> بأخذ المصفوفة غير الشاذة X على أنها
!>
!>                      X = Q*( I   0    )
!>                            ( 0 inv(R) ).
!>

المعلمات

JOBU 
!>          JOBU is CHARACTER*1
!>          = 'U':  المصفوفة المتعامدة U محسوبة؛
!>          = 'N':  U غير محسوبة.
!>

JOBV

!>          JOBV is CHARACTER*1
!>          = 'V':  المصفوفة المتعامدة V محسوبة؛
!>          = 'N':  V غير محسوبة.
!>

JOBQ

!>          JOBQ is CHARACTER*1
!>          = 'Q':  المصفوفة المتعامدة Q محسوبة؛
!>          = 'N':  Q غير محسوبة.
!>

M

!>          M is INTEGER
!>          عدد صفوف المصفوفة A.  M >= 0.
!>

N

!>          N is INTEGER
!>          عدد أعمدة المصفوفتين A و B.  N >= 0.
!>

P

!>          P is INTEGER
!>          عدد صفوف المصفوفة B.  P >= 0.
!>

K

!>          K is INTEGER
!>

L

!>          L is INTEGER
!>
!>          عند الخروج، يحدد K و L أبعاد الكتل الفرعية
!>          الموصوفة في الغرض.
!>          K + L = الرتبة العددية الفعالة للمصفوفة (A**T,B**T)**T.
!>

A

!>          A is REAL array, dimension (LDA,N)
!>          عند الإدخال، المصفوفة A بحجم M-by-N.
!>          عند الخروج، تحتوي A على المصفوفة المثلثية R، أو جزء من R.
!>          راجع الغرض للتفاصيل.
!>

LDA

!>          LDA is INTEGER
!>          البعد الرئيسي للمصفوفة A. LDA >= max(1,M).
!>

B

!>          B is REAL array, dimension (LDB,N)
!>          عند الإدخال، المصفوفة B بحجم P-by-N.
!>          عند الخروج، تحتوي B على المصفوفة المثلثية R إذا كان M-K-L < 0.
!>          راجع الغرض للتفاصيل.
!>

LDB

!>          LDB is INTEGER
!>          البعد الرئيسي للمصفوفة B. LDB >= max(1,P).
!>

ALPHA

!>          ALPHA is REAL array, dimension (N)
!>

BETA

!>          BETA is REAL array, dimension (N)
!>
!>          عند الخروج، تحتوي ALPHA و BETA على أزواج القيم الفردية المعممة
!>          لـ A و B؛
!>            ALPHA(1:K) = 1,
!>            BETA(1:K)  = 0,
!>          وإذا كان M-K-L >= 0,
!>            ALPHA(K+1:K+L) = C,
!>            BETA(K+1:K+L)  = S,
!>          أو إذا كان M-K-L < 0,
!>            ALPHA(K+1:M)=C, ALPHA(M+1:K+L)=0
!>            BETA(K+1:M) =S, BETA(M+1:K+L) =1
!>          و
!>            ALPHA(K+L+1:N) = 0
!>            BETA(K+L+1:N)  = 0
!>

U

!>          U is REAL array, dimension (LDU,M)
!>          إذا كان JOBU = 'U', تحتوي U على المصفوفة المتعامدة U بحجم M-by-M.
!>          إذا كان JOBU = 'N', لا يتم الاستعانة بـ U.
!>

LDU

!>          LDU is INTEGER
!>          البعد الرئيسي للمصفوفة U. LDU >= max(1,M) إذا كان
!>          JOBU = 'U'; LDU >= 1 خلاف ذلك.
!>

V

!>          V is REAL array, dimension (LDV,P)
!>          إذا كان JOBV = 'V', تحتوي V على المصفوفة المتعامدة V بحجم P-by-P.
!>          إذا كان JOBV = 'N', لا يتم الاستعانة بـ V.
!>

LDV

!>          LDV is INTEGER
!>          البعد الرئيسي للمصفوفة V. LDV >= max(1,P) إذا كان
!>          JOBV = 'V'; LDV >= 1 خلاف ذلك.
!>

Q

!>          Q is REAL array, dimension (LDQ,N)
!>          إذا كان JOBQ = 'Q', تحتوي Q على المصفوفة المتعامدة Q بحجم N-by-N.
!>          إذا كان JOBQ = 'N', لا يتم الاستعانة بـ Q.
!>

LDQ

!>          LDQ is INTEGER
!>          البعد الرئيسي للمصفوفة Q. LDQ >= max(1,N) إذا كان
!>          JOBQ = 'Q'; LDQ >= 1 خلاف ذلك.
!>

WORK

!>          WORK is REAL array, dimension (MAX(1,LWORK))
!>          عند الخروج، إذا كان INFO = 0, فإن WORK(1) تعيد LWORK المثلى.
!>
