subroutine sgsvj0 (character*1 jobv, integer m, integer n, real, dimension( lda, * ) a, integer lda, real, dimension( n ) d, real, dimension( n ) sva, integer mv, real, dimension( ldv, * ) v, integer ldv, real eps, real sfmin, real tol, integer nsweep, real, dimension( lwork ) work, integer lwork, integer info)

SGSVJ0 معالج مسبق للروتين sgesvj.

الغرض:

!>
!> يتم استدعاء SGSVJ0 من SGESVJ كمعالج مسبق وهذا هو غرضه الرئيسي.
!> يقوم بتطبيق دورانات جاكوبي بنفس الطريقة التي تفعلها SGESVJ، ولكنه
!> لا يتحقق من التقارب (معيار التوقف). تتوفر بعض معلمات الضبط (المميزة بـ [TP])
!> للمنفذ.
!>

المعلمات

JOBV 
!>          JOBV هو CHARACTER*1
!>          يحدد ما إذا كانت المخرجات من هذا الإجراء تستخدم
!>          لحساب المصفوفة V:
!>          = 'V': يتم تجميع حاصل ضرب دورانات جاكوبي
!>                 عن طريق ضرب المصفوفة V ذات البعد N*N بشكل لاحق.
!>                (انظر وصف V.)
!>          = 'A': يتم تجميع حاصل ضرب دورانات جاكوبي
!>                 عن طريق ضرب المصفوفة ذات البعد MV*N بشكل لاحق.
!>                (انظر أوصاف MV و V.)
!>          = 'N': لا يتم تجميع دورانات جاكوبي.
!>

M

!>          M هو INTEGER
!>          عدد الصفوف في المصفوفة المدخلة A. M >= 0.
!>

N

!>          N هو INTEGER
!>          عدد الأعمدة في المصفوفة المدخلة A.
!>          M >= N >= 0.
!>

A

!>          A هي مصفوفة REAL، أبعادها (LDA,N)
!>          عند الدخول، مصفوفة M*N A، بحيث يمثل A*diag(D)
!>          المصفوفة المدخلة.
!>          عند الخروج،
!>          A_onexit * D_onexit يمثل المصفوفة المدخلة A*diag(D)
!>          مضروبة بشكل لاحق في سلسلة من دورانات جاكوبي، حيث
!>          يتم تحديد عتبة الدوران والعدد الإجمالي للدورات في
!>          TOL و NSWEEP، على التوالي.
!>          (انظر أوصاف D و TOL و NSWEEP.)
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرائد للمصفوفة A. LDA >= max(1,M).
!>

D

!>          D هي مصفوفة REAL، أبعادها (N)
!>          تجمع المصفوفة D عوامل القياس من دورانات جاكوبي المقاسة بسرعة.
!>          عند الدخول، A*diag(D) يمثل المصفوفة المدخلة.
!>          عند الخروج، A_onexit*diag(D_onexit) يمثل المصفوفة المدخلة
!>          مضروبة بشكل لاحق في سلسلة من دورانات جاكوبي، حيث
!>          يتم تحديد عتبة الدوران والعدد الإجمالي للدورات في
!>          TOL و NSWEEP، على التوالي.
!>          (انظر أوصاف A و TOL و NSWEEP.)
!>

SVA

!>          SVA هي مصفوفة REAL، أبعادها (N)
!>          عند الدخول، تحتوي SVA على المعايير الإقليدية لأعمدة
!>          المصفوفة A*diag(D).
!>          عند الخروج، تحتوي SVA على المعايير الإقليدية لأعمدة
!>          المصفوفة onexit*diag(D_onexit).
!>

MV

!>          MV هو INTEGER
!>          إذا كان JOBV = 'A'، فسيتم ضرب MV صفوف من V بشكل لاحق
!>                           في سلسلة من دورانات جاكوبي.
!>          إذا كان JOBV = 'N'،   فلن يتم استخدام MV.
!>

V

!>          V هي مصفوفة REAL، أبعادها (LDV,N)
!>          إذا كان JOBV = 'V'، فسيتم ضرب N صفوف من V بشكل لاحق
!>                           في سلسلة من دورانات جاكوبي.
!>          إذا كان JOBV = 'A'، فسيتم ضرب MV صفوف من V بشكل لاحق
!>                           في سلسلة من دورانات جاكوبي.
!>          إذا كان JOBV = 'N'،   فلن يتم استخدام V.
!>

LDV

!>          LDV هو INTEGER
!>          البعد الرائد للمصفوفة V، LDV >= 1.
!>          إذا كان JOBV = 'V'، فإن LDV >= N.
!>          إذا كان JOBV = 'A'، فإن LDV >= MV.
!>

EPS

!>          EPS هو REAL
!>          EPS = SLAMCH('Epsilon')
!>

SFMIN

!>          SFMIN هو REAL
!>          SFMIN = SLAMCH('Safe Minimum')
!>

TOL

!>          TOL هو REAL
!>          TOL هي عتبة دورانات جاكوبي. لزوج
!>          A(:,p) و A(:,q) من أعمدة المحور، يتم تطبيق دوران جاكوبي
!>          فقط إذا كان ABS(COS(angle(A(:,p),A(:,q)))) > TOL.
!>

NSWEEP

!>          NSWEEP هو INTEGER
!>          NSWEEP هو عدد دورات دورانات جاكوبي التي سيتم
!>          إجراؤها.
!>

WORK

!>          WORK هي مصفوفة REAL، أبعادها (LWORK)
!>

LWORK

!>          LWORK هو INTEGER
!>          LWORK هو بعد WORK. LWORK >= M.
!>

INFO

!>          INFO هو INTEGER
!>          = 0:  خروج ناجح.
!>          < 0:  إذا كان INFO = -i، فإن الوسيط i كان له قيمة غير قانونية
!>
