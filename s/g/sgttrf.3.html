subroutine sgttrf (integer n, real, dimension( * ) dl, real, dimension( * ) d, real, dimension( * ) du, real, dimension( * ) du2, integer, dimension( * ) ipiv, integer info)

SGTTRF

الغرض:

!>
!> SGTTRF يحسب تحليل LU لمصفوفة ثلاثية قطرية حقيقية A
!> باستخدام الحذف مع التقسيم الجزئي وتبادلات الصفوف.
!>
!> التحليل له الشكل
!>    A = L * U
!> حيث L هو حاصل ضرب مصفوفات تبادلية ومصفوفات ثنائية قطرية سفلية واحدية،
!> و U هي مصفوفة مثلثية علوية مع عناصر غير صفرية فقط في القطر الرئيسي
!> والقطرين الفائقين الأول والثاني.
!>

المعاملات

N 
!>          N هو عدد صحيح (INTEGER)
!>          رتبة المصفوفة A.
!>

DL

!>          DL هي مصفوفة حقيقية (REAL array)، ذات بعد (N-1)
!>          عند الإدخال، يجب أن تحتوي DL على العناصر الفرعية القطرية (n-1) للمصفوفة
!>          A.
!>
!>          عند الخروج، سيتم الكتابة فوق DL بعوامل (n-1) التي
!>          تحدد المصفوفة L من تحليل LU للمصفوفة A.
!>

D

!>          D هي مصفوفة حقيقية (REAL array)، ذات بعد (N)
!>          عند الإدخال، يجب أن تحتوي D على العناصر القطرية للمصفوفة A.
!>
!>          عند الخروج، سيتم الكتابة فوق D بعناصر القطر الرئيسي (n) للمصفوفة
!>          المثلثية العلوية U من تحليل LU للمصفوفة A.
!>

DU

!>          DU هي مصفوفة حقيقية (REAL array)، ذات بعد (N-1)
!>          عند الإدخال، يجب أن تحتوي DU على العناصر الفائقة القطرية (n-1)
!>          للمصفوفة A.
!>
!>          عند الخروج، سيتم الكتابة فوق DU بالعناصر الفائقة القطرية الأولى (n-1)
!>          للمصفوفة U.
!>

DU2

!>          DU2 هي مصفوفة حقيقية (REAL array)، ذات بعد (N-2)
!>          عند الخروج، سيتم الكتابة فوق DU2 بالعناصر الفائقة القطرية الثانية (n-2)
!>          للمصفوفة U.
!>

IPIV

!>          IPIV هي مصفوفة أعداد صحيحة (INTEGER array)، ذات بعد (N)
!>          مؤشرات المحور؛ لـ 1 <= i <= n، تم تبديل الصف i للمصفوفة مع الصف IPIV(i).
!>          سيكون IPIV(i) دائمًا i أو i+1؛ IPIV(i) = i يشير إلى عدم الحاجة إلى
!>          تبديل صف.
!>

INFO

!>          INFO هو عدد صحيح (INTEGER)
!>          = 0:  خروج ناجح
!>          < 0:  إذا كان INFO = -k، فإن المعامل k كان له قيمة غير قانونية
!>          > 0:  إذا كان INFO = k، فإن U(k,k) تساوي صفرًا تمامًا. تم إكمال
!>                التحليل، ولكن العامل U مفرد تمامًا، وسيحدث قسمة على صفر إذا
!>                تم استخدامه لحل نظام من المعادلات.
!>
