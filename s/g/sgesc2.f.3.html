توثيق الوظيفة/الإجراء الفرعي

subroutine sgesc2 (integer n, real, dimension( lda, * ) a, integer lda, real, dimension( * ) rhs, integer, dimension( * ) ipiv, integer, dimension( * ) jpiv, real scale)

SGESC2 تحل نظام المعادلات الخطية باستخدام تحليل LU مع المحورية الكاملة المحسوبة بواسطة sgetc2.

الغرض:

!>
!> SGESC2 يحل نظام المعادلات الخطية
!>
!>           A * X = scale* RHS
!>
!> مع مصفوفة عامة N*N، A، باستخدام تحليل LU مع المحورية الكاملة المحسوبة بواسطة SGETC2.
!>

المعلمات

N 
!>          N هو عدد صحيح (INTEGER)
!>          رتبة المصفوفة A.
!>

A

!>          A هو مصفوفة حقيقية (REAL array)، البعد (LDA,N)
!>          عند الدخول، جزء LU من تحليل المصفوفة A ذات الأبعاد N*N المحسوبة بواسطة SGETC2:  A = P * L * U * Q
!>

LDA

!>          LDA هو عدد صحيح (INTEGER)
!>          البعد الرائد للمصفوفة A.  LDA >= max(1, N).
!>

RHS

!>          RHS هو مصفوفة حقيقية (REAL array)، البعد (N).
!>          عند الدخول، متجه الطرف الأيمن b.
!>          عند الخروج، متجه الحل X.
!>

IPIV

!>          IPIV هو مصفوفة أعداد صحيحة (INTEGER array)، البعد (N).
!>          مؤشرات المحورية؛ لكل 1 <= i <= N، تم تبديل الصف i من
!>          المصفوفة مع الصف IPIV(i).
!>

JPIV

!>          JPIV هو مصفوفة أعداد صحيحة (INTEGER array)، البعد (N).
!>          مؤشرات المحورية؛ لكل 1 <= j <= N، تم تبديل العمود j من
!>          المصفوفة مع العمود JPIV(j).
!>

SCALE

!>          SCALE هو عدد حقيقي (REAL)
!>           عند الخروج، تحتوي SCALE على عامل القياس. يتم اختيار SCALE
!>           بحيث 0 <= SCALE <= 1 لمنع تجاوز السعة في الحل.
!>
