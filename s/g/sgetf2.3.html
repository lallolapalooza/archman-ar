subroutine sgetf2 (m, n, a, lda, ipiv, info)
SGETF2 يحسب تحليل LU لمصفوفة عامة m-by-n باستخدام التقليب الجزئي مع تبادلات الصفوف (خوارزمية غير مجمعة).

وثائق الوظيفة/الإجراء الفرعي

subroutine sgetf2 (integer m, integer n, real, dimension( lda, * ) a, integer lda, integer, dimension( * ) ipiv, integer info)

SGETF2 يحسب تحليل LU لمصفوفة عامة m-by-n باستخدام التقليب الجزئي مع تبادلات الصفوف (خوارزمية غير مجمعة).

الغرض:

!>
!> SGETF2 يحسب تحليل LU لمصفوفة عامة A بحجم m × n
!> باستخدام التقليب الجزئي مع تبادلات الصفوف.
!>
!> التحليل له الشكل
!>    A = P * L * U
!> حيث P مصفوفة تباديل، L مصفوفة مثلثية سفلية بعناصر قطرية وحدات (شبه منحرف سفلي إذا كان m > n)، و U مصفوفة مثلثية علوية (شبه منحرف علوي إذا كان m < n).
!>
!> هذا هو إصدار المستوى 2 BLAS الذي يتبع نهج "النظر إلى اليمين" للخوارزمية.
!>

المعلمات

M 
!>          M هو عدد صحيح (INTEGER)
!>          عدد صفوف المصفوفة A.  M >= 0.
!>

N

!>          N هو عدد صحيح (INTEGER)
!>          عدد أعمدة المصفوفة A.  N >= 0.
!>

A

!>          A هي مصفوفة حقيقية (REAL array)، بأبعاد (LDA,N)
!>          عند الإدخال، المصفوفة m × n المراد تحليلها.
!>          عند الخروج، عوامل L و U من التحليل
!>          A = P*L*U؛ لا يتم تخزين عناصر القطر الوحدي للمصفوفة L.
!>

LDA

!>          LDA هو عدد صحيح (INTEGER)
!>          البعد الرئيسي للمصفوفة A.  LDA >= max(1,M).
!>

IPIV

!>          IPIV هي مصفوفة أعداد صحيحة (INTEGER array)، بأبعاد (min(M,N))
!>          مؤشرات التقليب؛ لكل 1 <= i <= min(M,N)، تم تبديل الصف i من
!>          المصفوفة مع الصف IPIV(i).
!>

INFO

!>          INFO هو عدد صحيح (INTEGER)
!>          = 0: تم الخروج بنجاح
!>          < 0: إذا كان INFO = -k، فإن المعلمة k كانت لها قيمة غير قانونية
!>          > 0: إذا كان INFO = k، فإن U(k,k) يساوي صفرًا تمامًا. تم
!>               إكمال التحليل، ولكن العامل U منفرد تمامًا، وسيحدث قسمة على صفر
!>               إذا تم استخدامه لحل نظام معادلات.
!>
