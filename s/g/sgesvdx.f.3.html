subroutine sgesvdx (character jobu, character jobvt, character range, integer m, integer n, real, dimension( lda, * ) a, integer lda, real vl, real vu, integer il, integer iu, integer ns, real, dimension( * ) s, real, dimension( ldu, * ) u, integer ldu, real, dimension( ldvt, * ) vt, integer ldvt, real, dimension( * ) work, integer lwork, integer, dimension( * ) iwork, integer info)

SGESVDX تحسب التحلل القيم المفردة (SVD) لمصفوفات GE

الغرض:

!>
!>  SGESVDX يحسب التحلل القيم المفردة (SVD) لمصفوفة حقيقية
!>  M-by-N A، اختياريًا يحسب المتجهات المفردة اليسرى و/أو اليمنى.  SVD مكتوبة
!>
!>      A = U * SIGMA * transpose(V)
!>
!>  حيث SIGMA مصفوفة M-by-N تكون صفرية باستثناء عناصرها القطرية
!>  min(m,n)، U مصفوفة متعامدة M-by-M، و
!>  V مصفوفة متعامدة N-by-N. العناصر القطرية لـ SIGMA
!>  هي القيم المفردة لـ A؛ وهي حقيقية وغير سالبة،
!>  وتُعاد بترتيب تنازلي.  الأعمدة min(m,n) الأولى لـ
!>  U و V هي المتجهات المفردة اليسرى واليمنى لـ A.
!>
!>  SGESVDX تستخدم مشكلة قيم ذاتية للحصول على SVD، والتي
!>  تسمح بحساب مجموعة فرعية من القيم والمتجهات المفردة.  راجع SBDSVDX للتفاصيل.
!>
!>  لاحظ أن الروتين يعيد V**T، وليس V.
!>

المعلمات

JOBU 
!>          JOBU هو CHARACTER*1
!>          يحدد خيارات لحساب كل أو جزء من المصفوفة U:
!>          = 'V':  تُعاد الأعمدة min(m,n) الأولى لـ U (المتجهات المفردة
!>                  اليسرى) أو كما تم تحديده بواسطة RANGE في
!>                  المصفوفة U؛
!>          = 'N':  لا تُحسب أي أعمدة لـ U (لا توجد متجهات مفردة يسرى).
!>

JOBVT

!>          JOBVT هو CHARACTER*1
!>           يحدد خيارات لحساب كل أو جزء من المصفوفة
!>           V**T:
!>           = 'V':  تُعاد الصفوف min(m,n) الأولى لـ V**T (المتجهات المفردة
!>                   اليمنى) أو كما تم تحديده بواسطة RANGE في
!>                   المصفوفة VT؛
!>           = 'N':  لا تُحسب أي صفوف لـ V**T (لا توجد متجهات مفردة يمنى).
!>

RANGE

!>          RANGE هو CHARACTER*1
!>          = 'A': سيتم العثور على جميع القيم المفردة.
!>          = 'V': سيتم العثور على جميع القيم المفردة في الفاصلة النصف مفتوحة (VL,VU].
!>          = 'I': سيتم العثور على القيم المفردة من IL إلى IU.
!>

M

!>          M هو INTEGER
!>          عدد صفوف المصفوفة المدخلة A.  M >= 0.
!>

N

!>          N هو INTEGER
!>          عدد أعمدة المصفوفة المدخلة A.  N >= 0.
!>

A

!>          A هو مصفوفة REAL، ذات بعد (LDA,N)
!>          عند الإدخال، المصفوفة M-by-N A.
!>          عند الخروج، يتم تدمير محتويات A.
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرائد للمصفوفة A.  LDA >= max(1,M).
!>

VL

!>          VL هو REAL
!>          إذا كانت RANGE='V'، الحد الأدنى للفاصلة التي
!>          سيتم البحث فيها عن القيم المفردة. VU > VL.
!>          لا يتم الرجوع إليها إذا كانت RANGE = 'A' أو 'I'.
!>

VU

!>          VU هو REAL
!>          إذا كانت RANGE='V'، الحد الأعلى للفاصلة التي
!>          سيتم البحث فيها عن القيم المفردة. VU > VL.
!>          لا يتم الرجوع إليها إذا كانت RANGE = 'A' أو 'I'.
!>

IL

!>          IL هو INTEGER
!>          إذا كانت RANGE='I'، فهرس
!>          أصغر قيمة مفردة سيتم إرجاعها.
!>          1 <= IL <= IU <= min(M,N)، إذا كان min(M,N) > 0.
!>          لا يتم الرجوع إليها إذا كانت RANGE = 'A' أو 'V'.
!>

IU

!>          IU هو INTEGER
!>          إذا كانت RANGE='I'، فهرس
!>          أكبر قيمة مفردة سيتم إرجاعها.
!>          1 <= IL <= IU <= min(M,N)، إذا كان min(M,N) > 0.
!>          لا يتم الرجوع إليها إذا كانت RANGE = 'A' أو 'V'.
!>

NS

!>          NS هو INTEGER
!>          العدد الإجمالي للقيم المفردة التي تم العثور عليها،
!>          0 <= NS <= min(M,N).
!>          إذا كانت RANGE = 'A'، فإن NS = min(M,N)؛ إذا كانت RANGE = 'I'، فإن NS = IU-IL+1.
!>

S

!>          S هو مصفوفة REAL، ذات بعد (min(M,N))
!>          القيم المفردة لـ A، مرتبة بحيث S(i) >= S(i+1).
!>

U

!>          U هو مصفوفة REAL، ذات بعد (LDU,UCOL)
!>          إذا كانت JOBU = 'V'، تحتوي U على أعمدة U (المتجهات المفردة
!>          اليسرى، مخزنة كأعمدة) كما تم تحديده بواسطة RANGE؛ إذا
!>          كانت JOBU = 'N'، فلا يتم الرجوع إلى U.
!>          ملاحظة: يجب على المستخدم التأكد من أن UCOL >= NS؛ إذا كانت RANGE = 'V'،
!>          فإن القيمة الدقيقة لـ NS غير معروفة مسبقًا ويجب استخدام حد أعلى.
!>

LDU

!>          LDU هو INTEGER
!>          البعد الرائد للمصفوفة U.  LDU >= 1؛ إذا
!>          كانت JOBU = 'V'، فإن LDU >= M.
!>

VT

!>          VT هو مصفوفة REAL، ذات بعد (LDVT,N)
!>          إذا كانت JOBVT = 'V'، تحتوي VT على صفوف V**T (المتجهات المفردة
!>          اليمنى، مخزنة كصفوف) كما تم تحديده بواسطة RANGE؛ إذا كانت JOBVT = 'N'،
!>          فلا يتم الرجوع إلى VT.
!>          ملاحظة: يجب على المستخدم التأكد من أن LDVT >= NS؛ إذا كانت RANGE = 'V'،
!>          فإن القيمة الدقيقة لـ NS غير معروفة مسبقًا ويجب استخدام حد أعلى.
!>

LDVT

!>          LDVT هو INTEGER
!>          البعد الرائد للمصفوفة VT.  LDVT >= 1؛ إذا
!>          كانت JOBVT = 'V'، فإن LDVT >= NS (انظر أعلاه).
!>

WORK

!>          WORK هي مصفوفة REAL، ذات بعد (MAX(1,LWORK))
!>          عند الخروج، إذا كانت INFO = 0، فإن WORK(1) تعيد LWORK الأمثل؛
!>

LWORK

!>          LWORK هو INTEGER
!>          بعد المصفوفة WORK.
!>          LWORK >= MAX(1,MIN(M,N)*(MIN(M,N)+4)) للمسارات (انظر
!>          التعليقات داخل الكود):
!>             - PATH 1  (M أكبر بكثير من N)
!>             - PATH 1t (N أكبر بكثير من M)
!>          LWORK >= MAX(1,MIN(M,N)*2+MAX(M,N)) للمسارات الأخرى.
!>          للحصول على أداء جيد، يجب أن يكون LWORK بشكل عام أكبر.
!>
!>          إذا كانت LWORK = -1، فسيتم افتراض استعلام مساحة العمل؛ الروتين
!>          يحسب فقط الحجم الأمثل لمصفوفة WORK، ويعيد
!>          هذه القيمة كإدخال أول لمصفوفة WORK، ولا يصدر رسالة خطأ
!>          متعلقة بـ LWORK بواسطة XERBLA.
!>

IWORK

!>          IWORK هي مصفوفة INTEGER، ذات بعد (12*MIN(M,N))
!>          إذا كانت INFO = 0، فإن العناصر الأولى لـ NS في IWORK تكون صفرية. إذا كانت INFO > 0،
!>          فإن IWORK تحتوي على فهارس المتجهات الذاتية التي فشلت
!>          في التقارب في SBDSVDX/SSTEVX.
!>

INFO

!>     INFO هو INTEGER
!>           = 0:  خروج ناجح
!>           < 0:  إذا كانت INFO = -i، فإن المعامل i كان له قيمة غير قانونية
!>           > 0:  إذا كانت INFO = i، فإن i متجهات ذاتية فشلت في التقارب
!>                 في SBDSVDX/SSTEVX.
!>                 إذا كانت INFO = N*2 + 1، حدث خطأ داخلي في
!>                 SBDSVDX
!>
