subroutine sgerq2 (integer m, integer n, real, dimension( lda, * ) a, integer lda, real, dimension( * ) tau, real, dimension( * ) work, integer info)

SGERQ2 تحسب تحليل RQ لمصفوفة مستطيلة عامة باستخدام خوارزمية غير محظورة.

الغرض:

!>
!> SGERQ2 تحسب تحليل RQ لمصفوفة A حقيقية بحجم m × n:
!> A = R * Q.
!>

المعاملات

M 
!>          M هو عدد صحيح (INTEGER)
!>          عدد صفوف المصفوفة A.  M >= 0.
!>

N

!>          N هو عدد صحيح (INTEGER)
!>          عدد أعمدة المصفوفة A.  N >= 0.
!>

A

!>          A هي مصفوفة حقيقية (REAL array)، البعد (LDA,N)
!>          عند الإدخال، المصفوفة A بحجم m × n.
!>          عند الخروج، إذا كان m <= n، فإن المثلث العلوي للمصفوفة الفرعية
!>          A(1:m,n-m+1:n) يحتوي على المصفوفة المثلثية العليا R بحجم m × m؛
!>          إذا كان m >= n، فإن العناصر على ومنطقة تحت القطر (m-n)
!>          تحتوي على المصفوفة شبه المنحرفة العليا R بحجم m × n؛ العناصر المتبقية،
!>          مع المصفوفة TAU، تمثل المصفوفة المتعامدة
!>          Q كناتج للانعكاسات الأولية (انظر التفاصيل الإضافية).
!>

LDA

!>          LDA هو عدد صحيح (INTEGER)
!>          البعد الرائد للمصفوفة A.  LDA >= max(1,M).
!>

TAU

!>          TAU هي مصفوفة حقيقية (REAL array)، البعد (min(M,N))
!>          العوامل القياسية للانعكاسات الأولية (انظر التفاصيل الإضافية).
!>

WORK

!>          WORK هي مصفوفة حقيقية (REAL array)، البعد (M)
!>

INFO

!>          INFO هو عدد صحيح (INTEGER)
!>          = 0: خروج ناجح
!>          < 0: إذا كان INFO = -i، فإن المعامل i كان له قيمة غير قانونية
!>

التفاصيل الإضافية:

!>
!>  يتم تمثيل المصفوفة Q كناتج للانعكاسات الأولية
!>
!>     Q = H(1) H(2) . . . H(k), حيث k = min(m,n).
!>
!>  كل H(i) لها الشكل
!>
!>     H(i) = I - tau * v * v**T
!>
!>  حيث tau هو عامل قياسي حقيقي، و v هو متجه حقيقي مع
!>  v(n-k+i+1:n) = 0 و v(n-k+i) = 1؛ v(1:n-k+i-1) يتم تخزينه عند الخروج في
!>  A(m-k+i,1:n-k+i-1)، و tau في TAU(i).
!>
