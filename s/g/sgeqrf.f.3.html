توثيق الدالة/الروتين الفرعي

subroutine sgeqrf (integer m, integer n, real, dimension( lda, * ) a, integer lda, real, dimension( * ) tau, real, dimension( * ) work, integer lwork, integer info)

SGEQRF

الغرض:

!>
!> SGEQRF يحسب تحليل QR لمصفوفة A الحقيقية من النوع M-by-N:
!>
!>    A = Q * ( R ),
!>            ( 0 )
!>
!> حيث:
!>
!>    Q هي مصفوفة متعامدة من النوع M-by-M؛
!>    R هي مصفوفة مثلثة عليا من النوع N-by-N؛
!>    0 هي مصفوفة صفرية من النوع (M-N)-by-N، إذا كان M > N.
!>
!>

المعاملات

M 
!>          M هو عدد صحيح (INTEGER)
!>          عدد صفوف المصفوفة A.  M >= 0.
!>

N

!>          N هو عدد صحيح (INTEGER)
!>          عدد أعمدة المصفوفة A.  N >= 0.
!>

A

!>          A هو مصفوفة حقيقية (REAL array)، بالبعد (LDA,N)
!>          عند الإدخال، مصفوفة A من النوع M-by-N.
!>          عند الخروج، تحتوي العناصر على القطر أو فوقه في المصفوفة
!>          على مصفوفة شبه منحرفة عليا من النوع min(M,N)-by-N (R
!>          مثلثة عليا إذا كان m >= n)؛ العناصر تحت القطر،
!>          مع المصفوفة TAU، تمثل المصفوفة المتعامدة Q كـ
!>          ناتج ضرب min(m,n) من الانعكاسات الأولية (انظر التفاصيل الإضافية).
!>

LDA

!>          LDA هو عدد صحيح (INTEGER)
!>          البعد الرائد للمصفوفة A.  LDA >= max(1,M).
!>

TAU

!>          TAU هي مصفوفة حقيقية (REAL array)، بالبعد (min(M,N))
!>          العوامل القياسية للانعكاسات الأولية (انظر التفاصيل الإضافية).
!>

WORK

!>          WORK هي مصفوفة حقيقية (REAL array)، بالبعد (MAX(1,LWORK))
!>          عند الخروج، إذا كانت INFO = 0، فإن WORK(1) تعيد LWORK المثلى.
!>

LWORK

!>          LWORK هو عدد صحيح (INTEGER)
!>          بعد المصفوفة WORK.
!>          LWORK >= 1، إذا كان MIN(M,N) = 0، و LWORK >= N، بخلاف ذلك.
!>          للأداء الأمثل LWORK >= N*NB، حيث NB
!>          هو حجم الكتلة الأمثل.
!>
!>          إذا كان LWORK = -1، فسيتم افتراض استعلام عن مساحة العمل؛ يقوم الروتين
!>          بحساب حجم مساحة العمل المثلى فقط، ويعيد
!>          هذه القيمة كالإدخال الأول في مساحة العمل، ولا يتم إصدار أي رسالة خطأ
!>          بخصوص LWORK بواسطة XERBLA.
!>

INFO

!>          INFO هو عدد صحيح (INTEGER)
!>          = 0:  انتهاء ناجح
!>          < 0:  إذا كان INFO = -i، فإن المعامل i كان له قيمة غير قانونية
!>

تفاصيل إضافية:

!>
!>  يتم تمثيل المصفوفة Q كناتج ضرب انعكاسات أولية
!>
!>     Q = H(1) H(2) . . . H(k)، حيث k = min(m,n).
!>
!>  كل H(i) لها الشكل
!>
!>     H(i) = I - tau * v * v**T
!>
!>  حيث tau هو عامل قياسي حقيقي، و v هو متجه حقيقي مع
!>  v(1:i-1) = 0 و v(i) = 1؛ يتم تخزين v(i+1:m) عند الخروج في A(i+1:m,i)،
!>  و tau في TAU(i).
!>
