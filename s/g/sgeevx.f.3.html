subroutine sgeevx (character balanc, character jobvl, character jobvr, character sense, integer n, real, dimension( lda, * ) a, integer lda, real, dimension( * ) wr, real, dimension( * ) wi, real, dimension( ldvl, * ) vl, integer ldvl, real, dimension( ldvr, * ) vr, integer ldvr, integer ilo, integer ihi, real, dimension( * ) scale, real abnrm, real, dimension( * ) rconde, real, dimension( * ) rcondv, real, dimension( * ) work, integer lwork, integer, dimension( * ) iwork, integer info)

SGEEVX تحسب القيم الذاتية، اختياريًا، المتجهات الذاتية اليسرى و/أو اليمنى لمصفوفات GE

الغرض:

!>
!> SGEEVX تحسب لمصفوفة حقيقية غير متناظرة N-by-N A،
!> القيم الذاتية، واختياريًا، المتجهات الذاتية اليسرى و/أو اليمنى.
!>
!> اختياريًا أيضًا، تحسب تحويلاً موازناً لتحسين
!> شروط القيم الذاتية والمتجهات الذاتية (ILO، IHI،
!> SCALE، و ABNRM)، مقلوبات أرقام الشروط
!> للقيم الذاتية (RCONDE)، ومقلوبات أرقام الشروط
!> للمتجهات الذاتية اليمنى
!> (RCONDV).
!>
!> المتجه الذاتي الأيمن v(j) لـ A يحقق
!>                  A * v(j) = lambda(j) * v(j)
!> حيث lambda(j) هي القيمة الذاتية الخاصة به.
!> المتجه الذاتي الأيسر u(j) لـ A يحقق
!>               u(j)**H * A = lambda(j) * u(j)**H
!> حيث u(j)**H يمثل المرافق التبادلي لـ u(j).
!>
!> المتجهات الذاتية المحسوبة تم تطبيعها لتكون لها معيار إقليدي
!> يساوي 1 وأكبر مركبة حقيقية.
!>
!> موازنة المصفوفة تعني تبديل الصفوف والأعمدة لجعلها
!> أقرب إلى المصفوفة المثلثية العليا، وتطبيق تحويل تشابه قطري
!> D * A * D**(-1)، حيث D هي مصفوفة قطرية مختارة
!> لجعل الصفوف والأعمدة في A أقرب في المعيار و
!> أرقام شروط القيم الذاتية والمتجهات الذاتية أصغر.
!> تبديل الصفوف والأعمدة لن يغير أرقام الشروط
!> (في حسابات دقيقة) ولكن القياس القطري سيغيرها.
!> لمزيد من الشرح حول الموازنة، انظر القسم 4.10.2 من
!> دليل مستخدمي LAPACK.
!>

المعلمات

BALANC 
!>          BALANC هو CHARACTER*1
!>          يشير إلى كيفية قياس المصفوفة المدخلة قطريًا
!> و/أو تبديلها لتحسين شروط القيم الذاتية الخاصة بها.
!>          = 'N': لا تقم بالقياس القطري أو التبديل؛
!>          = 'P': قم بإجراء تبديلات لجعل المصفوفة أقرب إلى
!>                 المصفوفة المثلثية العليا. لا تقم بالقياس القطري؛
!>          = 'S': قم بقياس المصفوفة قطريًا، أي استبدل A بـ
!>                 D*A*D**(-1)، حيث D هي مصفوفة قطرية مختارة
!> لجعل الصفوف والأعمدة في A متساوية تقريبًا في المعيار.
!> لا تقم بالتبديل؛
!>          = 'B': قم بقياس المصفوفة قطريًا وتبديلها.
!>
!>          ستكون مقلوبات أرقام الشروط المحسوبة للمصفوفة
!> بعد الموازنة و/أو التبديل. التبديل لا يغير
!> أرقام الشروط (في حسابات دقيقة)، ولكن الموازنة تغيرها.
!>

JOBVL

!>          JOBVL هو CHARACTER*1
!>          = 'N': لا يتم حساب المتجهات الذاتية اليسرى لـ A؛
!>          = 'V': يتم حساب المتجهات الذاتية اليسرى لـ A.
!>          إذا كان SENSE = 'E' أو 'B'، يجب أن يكون JOBVL = 'V'.
!>

JOBVR

!>          JOBVR هو CHARACTER*1
!>          = 'N': لا يتم حساب المتجهات الذاتية اليمنى لـ A؛
!>          = 'V': يتم حساب المتجهات الذاتية اليمنى لـ A.
!>          إذا كان SENSE = 'E' أو 'B'، يجب أن يكون JOBVR = 'V'.
!>

SENSE

!>          SENSE هو CHARACTER*1
!>          يحدد مقلوبات أرقام الشروط التي يتم حسابها.
!>          = 'N': لا يتم حساب أي منها؛
!>          = 'E': يتم حسابها للقيم الذاتية فقط؛
!>          = 'V': يتم حسابها للمتجهات الذاتية اليمنى فقط؛
!>          = 'B': يتم حسابها للقيم الذاتية والمتجهات الذاتية اليمنى.
!>
!>          إذا كان SENSE = 'E' أو 'B'، يجب أيضًا حساب كل من المتجهات الذاتية
!> اليسرى واليمنى (JOBVL = 'V' و JOBVR = 'V').
!>

N

!>          N هو INTEGER
!>          رتبة المصفوفة A. N >= 0.
!>

A

!>          A هو مصفوفة REAL، البعد (LDA,N)
!>          عند الدخول، المصفوفة N-by-N A.
!>          عند الخروج، تم الكتابة فوق A. إذا كان JOBVL = 'V' أو
!>          JOBVR = 'V'، تحتوي A على شكل Schur الحقيقي للمصفوفة
!> الموازنة للمصفوفة المدخلة A.
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرائد للمصفوفة A. LDA >= max(1,N).
!>

WR

!>          WR هي مصفوفة REAL، البعد (N)
!>

WI

!>          WI هي مصفوفة REAL، البعد (N)
!>          WR و WI تحتويان على الأجزاء الحقيقية والخيالية،
!>          على التوالي، للقيم الذاتية المحسوبة. الأزواج المترافقة
!>          المركبة للقيم الذاتية ستظهر بشكل متتالي
!> مع القيمة الذاتية التي لها جزء خيالي موجب
!> أولاً.
!>

VL

!>          VL هي مصفوفة REAL، البعد (LDVL,N)
!>          إذا كان JOBVL = 'V'، يتم تخزين المتجهات الذاتية اليسرى u(j)
!>          واحدة تلو الأخرى في أعمدة VL، بنفس الترتيب
!>          مثل القيم الذاتية الخاصة بها.
!>          إذا كان JOBVL = 'N'، لا يتم استخدام VL.
!>          إذا كانت القيمة الذاتية j حقيقية، فإن u(j) = VL(:,j)،
!>          العمود j من VL.
!>          إذا كانت القيمة الذاتية j و (j+1) تشكل زوجًا
!>          مترافقًا مركبًا، فإن u(j) = VL(:,j) + i*VL(:,j+1) و
!>          u(j+1) = VL(:,j) - i*VL(:,j+1).
!>

LDVL

!>          LDVL هو INTEGER
!>          البعد الرائد للمصفوفة VL. LDVL >= 1؛ إذا
!>          كان JOBVL = 'V'، فإن LDVL >= N.
!>

VR

!>          VR هي مصفوفة REAL، البعد (LDVR,N)
!>          إذا كان JOBVR = 'V'، يتم تخزين المتجهات الذاتية اليمنى v(j)
!>          واحدة تلو الأخرى في أعمدة VR، بنفس الترتيب
!>          مثل القيم الذاتية الخاصة بها.
!>          إذا كان JOBVR = 'N'، لا يتم استخدام VR.
!>          إذا كانت القيمة الذاتية j حقيقية، فإن v(j) = VR(:,j)،
!>          العمود j من VR.
!>          إذا كانت القيمة الذاتية j و (j+1) تشكل زوجًا
!>          مترافقًا مركبًا، فإن v(j) = VR(:,j) + i*VR(:,j+1) و
!>          v(j+1) = VR(:,j) - i*VR(:,j+1).
!>

LDVR

!>          LDVR هو INTEGER
!>          البعد الرائد للمصفوفة VR. LDVR >= 1، وإذا
!>          كان JOBVR = 'V'، فإن LDVR >= N.
!>

ILO

!>          ILO هو INTEGER
!>

IHI

!>          IHI هو INTEGER
!>          ILO و IHI هما قيمتان صحيحتان تم تحديدهما عند موازنة A.
!>          المصفوفة الموازنة A(i,j) = 0 إذا كان I > J و
!>          J = 1,...,ILO-1 أو I = IHI+1,...,N.
!>

SCALE

!>          SCALE هي مصفوفة REAL، البعد (N)
!>          تفاصيل التبديلات وعوامل القياس المطبقة
!>          عند موازنة A. إذا كان P(j) هو مؤشر الصف والعمود
!>          المتبادل مع الصف والعمود j، و D(j) هو عامل القياس
!>          المطبق على الصف والعمود j، فإن
!>          SCALE(J) = P(J)،    لـ J = 1,...,ILO-1
!>                   = D(J)،    لـ J = ILO,...,IHI
!>

ABNRM 

!>          ABNRM is REAL
!>          The one-norm of the balanced matrix (the maximum
!>          of the sum of absolute values of elements of any column).
!>

RCONDE 

!>          RCONDE is REAL array, dimension (N)
!>          RCONDE(j) is the reciprocal condition number of the j-th
!>          eigenvalue.
!>

RCONDV 

!>          RCONDV is REAL array, dimension (N)
!>          RCONDV(j) is the reciprocal condition number of the j-th
!>          right eigenvector.
!>

WORK 

!>          WORK is REAL array, dimension (MAX(1,LWORK))
!>          On exit, if INFO = 0, WORK(1) returns the optimal LWORK.
!>

LWORK 

!>          LWORK is INTEGER
!>          The dimension of the array WORK.   If SENSE = 'N' or 'E',
!>          LWORK >= max(1,2*N), and if JOBVL = 'V' or JOBVR = 'V',
!>          LWORK >= 3*N.  If SENSE = 'V' or 'B', LWORK >= N*(N+6).
!>          For good performance, LWORK must generally be larger.
!>
!>          If LWORK = -1, then a workspace query is assumed; the routine
!>          only calculates the optimal size of the WORK array, returns
!>          this value as the first entry of the WORK array, and no error
!>          message related to LWORK is issued by XERBLA.
!>

IWORK 

!>          IWORK is INTEGER array, dimension (2*N-2)
!>          If SENSE = 'N' or 'E', not referenced.
!>

INFO 

!>          INFO is INTEGER
!>          = 0:  successful exit
!>          < 0:  if INFO = -i, the i-th argument had an illegal value.
!>          > 0:  if INFO = i, the QR algorithm failed to compute all the
!>                eigenvalues, and no eigenvectors or condition numbers
!>                have been computed; elements 1:ILO-1 and i+1:N of WR
!>                and WI contain eigenvalues which have converged.
!>
