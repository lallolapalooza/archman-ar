توثيق الوظيفة/الإجراء الفرعي

subroutine sgeqr2p (integer m, integer n, real, dimension( lda, * ) a, integer lda, real, dimension( * ) tau, real, dimension( * ) work, integer info)

SGEQR2P تحسب تحليل QR لمصفوفة مستطيلة عامة بعناصر قطرية غير سالبة باستخدام خوارزمية غير محظورة.

الغرض:

!>
!> SGEQR2P تحسب تحليل QR لمصفوفة A الحقيقية ذات البعد m × n:
!>
!>    A = Q * ( R ),
!>            ( 0 )
!>
!> حيث:
!>
!>    Q هي مصفوفة متعامدة ذات البعد m × m؛
!>    R هي مصفوفة ثلاثية عليا ذات البعد n × n بمدخلات قطرية غير سالبة؛
!>    0 هي مصفوفة صفرية ذات البعد (m-n) × n، إذا كان m > n.
!>
!>

المعاملات

M 
!>          M هو عدد صحيح (INTEGER)
!>          عدد صفوف المصفوفة A.  M >= 0.
!>

N

!>          N هو عدد صحيح (INTEGER)
!>          عدد أعمدة المصفوفة A.  N >= 0.
!>

A

!>          A هي مصفوفة حقيقية (REAL array)، ذات البعد (LDA,N)
!>          عند الدخول، تحتوي مصفوفة A ذات البعد m × n.
!>          عند الخروج، تحتوي العناصر على ومن فوق القطر للمصفوفة على
!>          مصفوفة شبه منحرفة عليا ذات البعد min(m,n) × n (R هي
!>          مصفوفة ثلاثية عليا إذا كان m >= n). المدخلات القطرية لـ
!>          R غير سالبة؛ العناصر تحت القطر،
!>          مع المصفوفة TAU، تمثل المصفوفة المتعامدة Q كـ
!>          ناتج من معاملات انعكاس ابتدائية (انظر المزيد من التفاصيل).
!>

LDA

!>          LDA هو عدد صحيح (INTEGER)
!>          البعد الرئيسي للمصفوفة A.  LDA >= max(1,M).
!>

TAU

!>          TAU هي مصفوفة حقيقية (REAL array)، ذات البعد (min(M,N))
!>          العوامل القياسية لمعاملات الانعكاس الابتدائية (انظر المزيد من التفاصيل).
!>

WORK

!>          WORK هي مصفوفة حقيقية (REAL array)، ذات البعد (N)
!>

INFO

!>          INFO هو عدد صحيح (INTEGER)
!>          = 0: تم الخروج بنجاح
!>          < 0: إذا كان INFO = -i، فإن المعامل i كان لديه قيمة غير قانونية
!>

المزيد من التفاصيل:

!>
!>  تمثل المصفوفة Q كناتج من معاملات انعكاس ابتدائية
!>
!>     Q = H(1) H(2) . . . H(k)، حيث k = min(m,n).
!>
!>  كل H(i) لها الشكل
!>
!>     H(i) = I - tau * v * v**T
!>
!>  حيث tau هو عامل قياسي حقيقي، و v هو متجه حقيقي مع
!>  v(1:i-1) = 0 و v(i) = 1؛ v(i+1:m) مخزنة عند الخروج في A(i+1:m,i)،
!>  و tau في TAU(i).
!>
!> انظر مذكرة عمل Lapack 203 لمزيد من التفاصيل
!>
