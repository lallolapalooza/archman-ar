sg_get_network_iface_stats_r(3)

الحزمة: libstatgrab

التراخيص: LGPL, GPL

sg_get_network_iface_stats(3) sg_get_network_iface_stats(3)

اسم

sg_get_network_iface_stats, sg_get_network_iface_stats_r, sg_free_network_iface_stats, sg_network_iface_compare_name - الحصول على إحصائيات واجهة الشبكة

ملخص

#include <statgrab.h>


sg_network_iface_stats *sg_get_network_iface_stats
(size_t *entries);


sg_network_iface_stats *sg_get_network_iface_stats_r
(size_t *entries);


sg_error sg_free_network_iface_stats
(sg_network_iface_stats *data);


int sg_network_iface_compare_name
(const void *va, const void *vb);

الوصف

الدالتان sg_get_network_iface_stats() و sg_get_network_iface_stats_r() تعيدان إحصائيات حول واجهات الشبكة في الجهاز. على وجه التحديد، تعيدان سرعة الواجهة، وحالة الازدواج، وما إذا كانت نشطة حاليًا. تأخذ كلتا الدالتين معامل مدخلات اختياري، والذي يشير (عند إعطائه) إلى size_t لأخذ عدد إدخالات المتجه المعادة.

اختصار واجهة برمجة التطبيقات

الوظيفة تعيد مالك البيانات
sg_get_network_iface_stats sg_network_iface_stats * libstatgrab (محلية للمسار)
sg_get_network_iface_stats_r sg_network_iface_stats * المتصل

يجب تحرير متجهات sg_network_iface_stats المستلمة من sg_get_network_iface_stats_r() باستخدام sg_free_network_iface_stats() عند عدم الحاجة إليها بعد الآن. المتصل مسؤول عن القيام بذلك.

بالإضافة إلى ذلك، تتوفر دالة دعم لـ qsort(3): sg_network_iface_compare_name().

مثال ترتيب

size_t entries;
sg_network_iface_stats *network_iface_stats = NULL;
while( NULL != ( network_iface_stats = sg_get_network_iface_stats(&entries) ) ) {
    /* ترتيب الإدخالات أبجديًا باستخدام نقطة التحميل */
    qsort( network_iface_stats, entries, sizeof(network_iface_stats[0]), &sg_network_iface_compare_name );
    show_network_iface_stats( network_iface_stats );
}

قيم الإرجاع

تُعيد الدالة sg_get_network_iface_stats مؤشرًا إلى هيكل من النوع sg_network_iface_stats.

typedef enum{
        SG_IFACE_DUPLEX_FULL,
        SG_IFACE_DUPLEX_HALF,
        SG_IFACE_DUPLEX_UNKNOWN
}sg_iface_duplex;

ملاحظة: قد يعني القيمة SG_IFACE_DUPLEX_UNKNOWN أن الازدواج لم يتم التفاوض عليه بعد.

typedef struct {
        char *interface_name;
        int speed;
        sg_iface_duplex duplex;
        int up;
        time_t systime;
} sg_network_iface_stats;
interface_name
الاسم المعروف لنظام التشغيل. (على سبيل المثال، قد يكون على لينكس eth0)
speed
سرعة الواجهة، بالميجابت/ثانية.
duplex
حالة الازدواج التي تكون عليها الواجهة. انظر sg_iface_duplex للقيم المسموح بها.
up
ما إذا كانت الواجهة نشطة.
systime
الطابع الزمني عند جمع الإحصائيات المذكورة أعلاه بالثواني منذ بداية الحقبة

الأخطاء

قد يتطلب الحصول على تفاصيل محددة امتيازات مرتفعة، على سبيل المثال، على لينكس، سرعة الواجهة غير مرئية للعمليات غير المميزة.

توجد دوال المقارنة للتوافق مع الإصدارات السابقة أكثر من تحسينات الوظائف. المرونة المحدودة (مثل الترتيب العكسي) ونقص فرص تحسين المترجم يؤدي إلى توصية بتنفيذ روتينات المقارنة المطلوبة محليًا.
