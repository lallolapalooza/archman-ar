subroutine sgrqts (integer m, integer p, integer n, real, dimension( lda, * ) a, real, dimension( lda, * ) af, real, dimension( lda, * ) q, real, dimension( lda, * ) r, integer lda, real, dimension( * ) taua, real, dimension( ldb, * ) b, real, dimension( ldb, * ) bf, real, dimension( ldb, * ) z, real, dimension( ldb, * ) t, real, dimension( ldb, * )bwk, integer ldb, real, dimension( * ) taub, real, dimension( lwork ) work, integer lwork, real, dimension( * ) rwork, real, dimension( 4 ) result)

SGRQTS

الغرض:

!>
!> SGRQTS يختبر SGGRQF، الذي يحسب تحليل GR لمصفوفة A بحجم M-by-N ومصفوفة B بحجم P-by-N: A = R*Q و B = Z*T*Q.
!>

المعلمات

M 
!>          M هو عدد صحيح (INTEGER)
!>          عدد صفوف المصفوفة A.  M >= 0.
!>

P

!>          P هو عدد صحيح (INTEGER)
!>          عدد صفوف المصفوفة B.  P >= 0.
!>

N

!>          N هو عدد صحيح (INTEGER)
!>          عدد أعمدة المصفوفات A و B.  N >= 0.
!>

A

!>          A هي مصفوفة حقيقية (REAL array)، ذات بعد (LDA,N)
!>          المصفوفة A بحجم M-by-N.
!>

AF

!>          AF هي مصفوفة حقيقية (REAL array)، ذات بعد (LDA,N)
!>          تفاصيل تحليل GR للمصفوفتين A و B، كما تم إرجاعها
!>          بواسطة SGGRQF، انظر SGGRQF لمزيد من التفاصيل.
!>

Q

!>          Q هي مصفوفة حقيقية (REAL array)، ذات بعد (LDA,N)
!>          المصفوفة المتعامدة Q بحجم N-by-N.
!>

R

!>          R هي مصفوفة حقيقية (REAL array)، ذات بعد (LDA,MAX(M,N))
!>

LDA

!>          LDA هو عدد صحيح (INTEGER)
!>          البعد الرائد للمصفوفات A، AF، R، و Q.
!>          LDA >= max(M,N).
!>

TAUA

!>          TAUA هي مصفوفة حقيقية (REAL array)، ذات بعد (min(M,N))
!>          العوامل القياسية للمنعكسات الأولية، كما تم إرجاعها
!>          بواسطة SGGQRC.
!>

B

!>          B هي مصفوفة حقيقية (REAL array)، ذات بعد (LDB,N)
!>          عند الإدخال، المصفوفة B بحجم P-by-N.
!>

BF

!>          BF هي مصفوفة حقيقية (REAL array)، ذات بعد (LDB,N)
!>          تفاصيل تحليل GQR للمصفوفتين A و B، كما تم إرجاعها
!>          بواسطة SGGRQF، انظر SGGRQF لمزيد من التفاصيل.
!>

Z

!>          Z هي مصفوفة حقيقية (REAL array)، ذات بعد (LDB,P)
!>          المصفوفة المتعامدة Z بحجم P-by-P.
!>

T

!>          T هي مصفوفة حقيقية (REAL array)، ذات بعد (LDB,max(P,N))
!>

BWK

!>          BWK هي مصفوفة حقيقية (REAL array)، ذات بعد (LDB,N)
!>

LDB

!>          LDB هو عدد صحيح (INTEGER)
!>          البعد الرائد للمصفوفات B، BF، Z، و T.
!>          LDB >= max(P,N).
!>

TAUB

!>          TAUB هي مصفوفة حقيقية (REAL array)، ذات بعد (min(P,N))
!>          العوامل القياسية للمنعكسات الأولية، كما تم إرجاعها
!>          بواسطة SGGRQF.
!>

WORK

!>          WORK هي مصفوفة حقيقية (REAL array)، ذات بعد (LWORK)
!>

LWORK

!>          LWORK هو عدد صحيح (INTEGER)
!>          بعد المصفوفة WORK، LWORK >= max(M,P,N)**2.
!>

RWORK

!>          RWORK هي مصفوفة حقيقية (REAL array)، ذات بعد (M)
!>

RESULT

!>          RESULT هي مصفوفة حقيقية (REAL array)، ذات بعد (4)
!>          نسب الاختبار:
!>            RESULT(1) = norm( R - A*Q' ) / ( MAX(M,N)*norm(A)*ULP)
!>            RESULT(2) = norm( T*Q - Z'*B ) / (MAX(P,N)*norm(B)*ULP)
!>            RESULT(3) = norm( I - Q'*Q ) / ( N*ULP )
!>            RESULT(4) = norm( I - Z'*Z ) / ( P*ULP )
!>
