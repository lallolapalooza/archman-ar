توثيق الوظيفة/الروتين الفرعي

subroutine sggrqf (integer m, integer p, integer n, real, dimension( lda, * ) a, integer lda, real, dimension( * ) taua, real, dimension( ldb, * ) b, integer ldb, real, dimension( * ) taub, real, dimension( * ) work, integer lwork, integer info)

SGGRQF

الغرض:

!>
!> SGGRQF يحسب تحليل RQ المعمم للمصفوفة M-by-N A والمصفوفة P-by-N B:
!>
!>             A = R*Q,        B = Z*T*Q,
!>
!> حيث Q مصفوفة متعامدة N-by-N، و Z مصفوفة متعامدة P-by-P، وتأخذ R و T أحد الأشكال التالية:
!>
!> إذا كان M <= N،  R = ( 0  R12 ) M،   أو إذا كان M > N،  R = ( R11 ) M-N،
!>                  N-M  M                           ( R21 ) N
!>                                                      N
!>
!> حيث R12 أو R21 مثلثية عليا، و
!>
!> إذا كان P >= N،  T = ( T11 ) N  ،   أو إذا كان P < N،  T = ( T11  T12 ) P،
!>                 (  0  ) P-N                         P   N-P
!>                    N
!>
!> حيث T11 مثلثية عليا.
!> على وجه الخصوص، إذا كانت B مربعة وغير مفردة، فإن تحليل GRQ للمصفوفتين A و B يعطي ضمنياً تحليل RQ للمصفوفة A*inv(B):
!>
!>              A*inv(B) = (R*inv(T))*Z**T
!>
!> حيث inv(B) تمثل معكوس المصفوفة B، و Z**T تمثل منقول المصفوفة Z.
!>

المعلمات:

M 
!>          M هو عدد صحيح (INTEGER)
!>          عدد صفوف المصفوفة A.  M >= 0.
!>

P

!>          P هو عدد صحيح (INTEGER)
!>          عدد صفوف المصفوفة B.  P >= 0.
!>

N

!>          N هو عدد صحيح (INTEGER)
!>          عدد أعمدة المصفوفتين A و B. N >= 0.
!>

A

!>          A هي مصفوفة حقيقية (REAL array)، الأبعاد (LDA,N)
!>          عند الإدخال، المصفوفة A ذات الأبعاد M-by-N.
!>          عند الخروج، إذا كان M <= N، فإن الجزء المثلثي العلوي من المصفوفة الفرعية
!>          A(1:M,N-M+1:N) يحتوي على المصفوفة المثلثية العليا R ذات الأبعاد M-by-M؛
!>          إذا كان M > N، فإن العناصر الموجودة على وتحت القطر الفرعي (M-N)
!>          تحتوي على المصفوفة شبه المنحرفة العليا R ذات الأبعاد M-by-N؛ العناصر المتبقية،
!>          مع المصفوفة TAUA، تمثل المصفوفة المتعامدة Q كناتج لمركبات أساسية (انظر التفاصيل الإضافية).
!>

LDA

!>          LDA هو عدد صحيح (INTEGER)
!>          البعد الرائد للمصفوفة A. LDA >= max(1,M).
!>

TAUA

!>          TAUA هي مصفوفة حقيقية (REAL array)، الأبعاد (min(M,N))
!>          العوامل القياسية للمركبات الأساسية التي تمثل المصفوفة المتعامدة Q (انظر التفاصيل الإضافية).
!>

B

!>          B هي مصفوفة حقيقية (REAL array)، الأبعاد (LDB,N)
!>          عند الإدخال، المصفوفة B ذات الأبعاد P-by-N.
!>          عند الخروج، العناصر الموجودة على وفوق القطر في المصفوفة تمثل المصفوفة شبه المنحرفة العليا T ذات الأبعاد min(P,N)-by-N (T مثلثية عليا إذا كان P >= N)؛ العناصر الموجودة تحت القطر،
!>          مع المصفوفة TAUB، تمثل المصفوفة المتعامدة Z كناتج لمركبات أساسية (انظر التفاصيل الإضافية).
!>

LDB

!>          LDB هو عدد صحيح (INTEGER)
!>          البعد الرائد للمصفوفة B. LDB >= max(1,P).
!>

TAUB

!>          TAUB هي مصفوفة حقيقية (REAL array)، الأبعاد (min(P,N))
!>          العوامل القياسية للمركبات الأساسية التي تمثل المصفوفة المتعامدة Z (انظر التفاصيل الإضافية).
!>

WORK

!>          WORK هي مصفوفة حقيقية (REAL array)، الأبعاد (MAX(1,LWORK))
!>          عند الخروج، إذا كان INFO = 0، فإن WORK(1) يعيد حجم LWORK الأمثل.
!>

LWORK

!>          LWORK هو عدد صحيح (INTEGER)
!>          بعد المصفوفة WORK. LWORK >= max(1,N,M,P).
!>          للحصول على الأداء الأمثل LWORK >= max(N,M,P)*max(NB1,NB2,NB3)،
!>          حيث NB1 هو حجم الكتلة الأمثل لتحليل RQ
!>          لمصفوفة M-by-N، و NB2 هو حجم الكتلة الأمثل لتحليل QR
!>          لمصفوفة P-by-N، و NB3 هو حجم الكتلة الأمثل لاستدعاء SORMRQ.
!>
!>          إذا كان LWORK = -1، فسيتم افتراض استعلام عن مساحة العمل؛ الروتين
!>          يحسب فقط الحجم الأمثل للمصفوفة WORK، ويعيد هذا الحجم كإدخال أول
!>          للمصفوفة WORK، ولا يصدر رسالة خطأ متعلقة بـ LWORK بواسطة XERBLA.
!>

INFO

!>          INFO هو عدد صحيح (INTEGER)
!>          = 0:  انتهاء ناجح
!>          < 0:  إذا كان INF0= -i، فإن المعامل i كان له قيمة غير قانونية.
!>

تفاصيل إضافية:

!>
!>  يتم تمثيل المصفوفة Q كناتج لمركبات أساسية
!>
!>     Q = H(1) H(2) . . . H(k)، حيث k = min(m,n).
!>
!>  كل H(i) لها الشكل
!>
!>     H(i) = I - taua * v * v**T
!>
!>  حيث taua هو عامل حقيقي، و v هو متجه حقيقي مع
!>  v(n-k+i+1:n) = 0 و v(n-k+i) = 1؛ يتم تخزين v(1:n-k+i-1) عند الخروج في
!>  A(m-k+i,1:n-k+i-1)، و taua في TAUA(i).
!>  لتكوين Q صراحة، استخدم الروتين الفرعي LAPACK SORGRQ.
!>  لاستخدام Q لتحديث مصفوفة أخرى، استخدم الروتين الفرعي LAPACK SORMRQ.
!>
!>  يتم تمثيل المصفوفة Z كناتج لمركبات أساسية
!>
!>     Z = H(1) H(2) . . . H(k)، حيث k = min(p,n).
!>
!>  كل H(i) لها الشكل
!>
!>     H(i) = I - taub * v * v**T
!>
!>  حيث taub هو عامل حقيقي، و v هو متجه حقيقي مع
!>  v(1:i-1) = 0 و v(i) = 1؛ يتم تخزين v(i+1:p) عند الخروج في B(i+1:p,i)،
!>  و taub في TAUB(i).
!>  لتكوين Z صراحة، استخدم الروتين الفرعي LAPACK SORGQR.
!>  لاستخدام Z لتحديث مصفوفة أخرى، استخدم الروتين الفرعي LAPACK SORMQR.
!>
