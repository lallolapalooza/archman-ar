subroutine sgeqpf (integer m, integer n, real, dimension( lda, * ) a, integer lda, integer, dimension( * ) jpvt, real, dimension( * ) tau, real, dimension( * ) work, integer info)

SGEQPF

الغرض:

!>
!> تم إيقاف هذا الروتين الفرعي واستبداله بالروتين الفرعي SGEQP3.
!>
!> يحسب SGEQPF عامل QR مع تحويل الأعمدة لمصفوفة حقيقية M-by-N وهي A: A*P = Q*R.
!>

المعاملات

M 
!>          M هو عدد صحيح (INTEGER)
!>          عدد صفوف المصفوفة A. M >= 0.
!>

N

!>          N هو عدد صحيح (INTEGER)
!>          عدد أعمدة المصفوفة A. N >= 0
!>

A

!>          A هي مصفوفة حقيقية (REAL array)، حجمها (LDA,N)
!>          عند الدخول، المصفوفة A بحجم M-by-N.
!>          عند الخروج، الجزء العلوي المثلثي من المصفوفة يحتوي على المصفوفة المثلثية العلوية R بحجم min(M,N)-by-N؛ العناصر
!>          أسفل القطر، مع المصفوفة TAU، تمثل المصفوفة المتعامدة Q كناتج ضرب
!>          min(m,n) من الانعكاسات الأولية.
!>

LDA

!>          LDA هو عدد صحيح (INTEGER)
!>          البعد الرئيسي للمصفوفة A. LDA >= max(1,M).
!>

JPVT

!>          JPVT هي مصفوفة أعداد صحيحة (INTEGER array)، حجمها (N)
!>          عند الدخول، إذا كان JPVT(i) .ne. 0، فإن العمود i من A يتم تحويله
!>          إلى مقدمة A*P (عمود رائد)؛ إذا كان JPVT(i) = 0،
!>          فإن العمود i من A هو عمود حر.
!>          عند الخروج، إذا كان JPVT(i) = k، فإن العمود i من A*P
!>          كان العمود k من A.
!>

TAU

!>          TAU هي مصفوفة حقيقية (REAL array)، حجمها (min(M,N))
!>          العوامل العددية للانعكاسات الأولية.
!>

WORK

!>          WORK هي مصفوفة حقيقية (REAL array)، حجمها (3*N)
!>

INFO

!>          INFO هو عدد صحيح (INTEGER)
!>          = 0:  انتهاء ناجح
!>          < 0:  إذا كان INFO = -i، فإن المعامل i كان له قيمة غير قانونية
!>

تفاصيل إضافية:

!>
!>  المصفوفة Q ممثلة كناتج ضرب انعكاسات أولية
!>
!>     Q = H(1) H(2) . . . H(n)
!>
!>  كل H(i) لها الشكل
!>
!>     H = I - tau * v * v**T
!>
!>  حيث tau هو معامل حقيقي، و v هو متجه حقيقي مع
!>  v(1:i-1) = 0 و v(i) = 1؛ v(i+1:m) مخزن عند الخروج في A(i+1:m,i).
!>
!>  المصفوفة P ممثلة في jpvt كما يلي: إذا كان
!>     jpvt(j) = i
!>  فإن العمود j من P هو المتجه الوحدوي الأساسي i.
!>
!>  تم تعديل استراتيجية تحديث معيار الأعمدة الجزئي بواسطة
!>    Z. Drmac و Z. Bujanovic، قسم الرياضيات،
!>    جامعة زغرب، كرواتيا.
!>  -- أبريل 2011 --
!>  لمزيد من التفاصيل، انظر مذكرة عمل LAPACK رقم 176.
!>
