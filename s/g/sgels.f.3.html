subroutine sgels (character trans, integer m, integer n, integer nrhs, real, dimension( lda, * ) a, integer lda, real, dimension( ldb, * ) b, integer ldb, real, dimension( * ) work, integer lwork, integer info)

SGELS يحل أنظمة مصفوفات GE ذات الحجم الزائد أو الناقص

الغرض:

!>
!> SGELS يحل أنظمة خطية حقيقية ذات الحجم الزائد أو الناقص
!> تتضمن مصفوفة A بحجم M-by-N، أو مترافقها، باستخدام تحليل QR أو LQ
!> للمصفوفة A.
!>
!> يُفترض أن المصفوفة A لها رتبة كاملة، ويتم توفير حماية أساسية فقط
!> ضد المصفوفات ذات الرتبة الناقصة. هذا الإجراء الفرعي يكتشف فقط
!> النقص التام في الرتبة، حيث يكون عنصر قطري في العامل المثلثي
!> للمصفوفة A صفراً تمامًا.
!>
!> من الممكن أن يكون أحد (أو أكثر) من العناصر القطرية للعامل المثلثي
!> للمصفوفة A أعدادًا صغيرة جدًا دون أن يشير هذا الإجراء الفرعي إلى
!> خطأ. الحلول التي يتم حسابها لمثل هذه المصفوفات شبه الناقصة الرتبة قد
!> تكون أقل دقة بسبب فقدان الدقة العددية.
!>
!> الخيارات التالية متاحة:
!>
!> 1. إذا كان TRANS = 'N' و m >= n:  ابحث عن حل المربعات الصغرى لـ
!>    نظام زائد الحجم، أي، حل مشكلة المربعات الصغرى
!>                 minimize || B - A*X ||.
!>
!> 2. إذا كان TRANS = 'N' و m < n:  ابحث عن حل المعيار الأدنى لـ
!>    نظام ناقص الحجم A * X = B.
!>
!> 3. إذا كان TRANS = 'T' و m >= n:  ابحث عن حل المعيار الأدنى لـ
!>    نظام ناقص الحجم A**T * X = B.
!>
!> 4. إذا كان TRANS = 'T' و m < n:  ابحث عن حل المربعات الصغرى لـ
!>    نظام زائد الحجم، أي، حل مشكلة المربعات الصغرى
!>                 minimize || B - A**T * X ||.
!>
!> يمكن التعامل مع عدة متجهات يمنى b ومتجهات حل x في استدعاء واحد؛
!> يتم تخزينها كأعمدة للمصفوفة B ذات اليمين المتعددة (M-by-NRHS)
!> ومصفوفة الحل X (N-by-NRHS).
!>

المعلمات

TRANS 
!>          TRANS هو CHARACTER*1
!>          = 'N': النظام الخطي يتضمن A؛
!>          = 'T': النظام الخطي يتضمن A**T.
!>

M

!>          M هو INTEGER
!>          عدد صفوف المصفوفة A. M >= 0.
!>

N

!>          N هو INTEGER
!>          عدد أعمدة المصفوفة A. N >= 0.
!>

NRHS

!>          NRHS هو INTEGER
!>          عدد المتجهات اليمنى، أي عدد أعمدة المصفوفتين B و X. NRHS >=0.
!>

A

!>          A هي مصفوفة REAL، البعد (LDA,N)
!>          عند الدخول، المصفوفة A بحجم M-by-N.
!>          عند الخروج،
!>            إذا كان M >= N، يتم استبدال A بتفاصيل تحليل QR الخاص بها
!>                       كما تم إرجاعه بواسطة SGEQRF؛
!>            إذا كان M <  N، يتم استبدال A بتفاصيل تحليل LQ الخاص بها
!>                       كما تم إرجاعه بواسطة SGELQF.
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرائد للمصفوفة A. LDA >= max(1,M).
!>

B

!>          B هي مصفوفة REAL، البعد (LDB,NRHS)
!>          عند الدخول، المصفوفة B للمتجهات اليمنى، مخزنة
!>          كأعمدة؛ B بحجم M-by-NRHS إذا كان TRANS = 'N'، أو N-by-NRHS
!>          إذا كان TRANS = 'T'.
!>          عند الخروج، إذا كان INFO = 0، يتم استبدال B بمتجهات الحل
!>          المخزنة كأعمدة:
!>          إذا كان TRANS = 'N' و m >= n، الصفوف من 1 إلى n في B تحتوي على متجهات
!>          حل المربعات الصغرى؛ يتم إعطاء مجموع مربعات الباقي للحل
!>          في كل عمود بواسطة مجموع مربعات العناصر من N+1 إلى M في ذلك العمود؛
!>          إذا كان TRANS = 'N' و m < n، الصفوف من 1 إلى N في B تحتوي على
!>          متجهات حل المعيار الأدنى؛
!>          إذا كان TRANS = 'T' و m >= n، الصفوف من 1 إلى M في B تحتوي على
!>          متجهات حل المعيار الأدنى؛
!>          إذا كان TRANS = 'T' و m < n، الصفوف من 1 إلى M في B تحتوي على
!>          متجهات حل المربعات الصغرى؛ يتم إعطاء مجموع مربعات الباقي
!>          للحل في كل عمود بواسطة مجموع مربعات العناصر من M+1 إلى N في ذلك العمود.
!>

LDB

!>          LDB هو INTEGER
!>          البعد الرائد للمصفوفة B. LDB >= MAX(1,M,N).
!>

WORK

!>          WORK هي مصفوفة REAL، البعد (MAX(1,LWORK))
!>          عند الخروج، إذا كان INFO = 0، ترجع WORK(1) قيمة LWORK المثلى.
!>

LWORK

!>          LWORK هو INTEGER
!>          بعد المصفوفة WORK.
!>          LWORK >= max( 1, MN + max( MN, NRHS ) ).
!>          للحصول على الأداء الأمثل،
!>          LWORK >= max( 1, MN + max( MN, NRHS )*NB ).
!>          حيث MN = min(M,N) و NB هو الحجم الأمثل للكتلة.
!>
!>          إذا كان LWORK = -1، فيُفترض استعلام عن مساحة العمل؛ يقوم الإجراء الفرعي
!>          فقط بحساب الحجم الأمثل للمصفوفة WORK، ويعيد
!>          هذه القيمة كالإدخال الأول للمصفوفة WORK، ولا يصدر رسالة خطأ
!>          متعلقة بـ LWORK بواسطة XERBLA.
!>

INFO

!>          INFO هو INTEGER
!>          = 0:  انتهاء ناجح
!>          < 0:  إذا كان INFO = -i، فإن الوسيط i كان له قيمة غير قانونية
!>          > 0:  إذا كان INFO =  i، فإن العنصر القطري i للعامل
!>                المثلثي للمصفوفة A هو صفر تمامًا، بحيث أن A ليست ذات
!>                رتبة كاملة؛ لم يكن من الممكن حساب حل المربعات الصغرى.
!>
