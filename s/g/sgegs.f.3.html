subroutine sgegs (character jobvsl, character jobvsr, integer n, real, dimension( lda, * ) a, integer lda, real, dimension( ldb, * ) b, integer ldb, real, dimension( * ) alphar, real, dimension( * ) alphai, real, dimension( * ) beta, real, dimension( ldvsl, * ) vsl, integer ldvsl, real, dimension( ldvsr, * ) vsr, integer ldvsr, real, dimension( * ) work, integer lwork, integer info)

SGEGS تحسب القيم الذاتية، وشكل Schur الحقيقي، واختيارياً، متجهات Schur اليسرى و/أو اليمنى لزوج المصفوفات الحقيقي (A,B)

الغرض:

!>
!> تم إيقاف هذا الإجراء الفرعي واستبداله بالإجراء الفرعي SGGES.
!>
!> SGEGS تحسب القيم الذاتية، وشكل Schur الحقيقي، واختيارياً،
!> متجهات Schur اليسرى و/أو اليمنى لزوج المصفوفات الحقيقي (A,B).
!> بالنظر إلى مصفوفتين مربعتين A و B، فإن تحليل Schur الحقيقي المعمم له الشكل
!>
!>   A = Q*S*Z**T,  B = Q*T*Z**T
!>
!> حيث Q و Z مصفوفتان متعامدتان، T مصفوفة مثلثة علوية، و S
!> مصفوفة شبه مثلثة علوية ذات كتل قطرية بحجم 1x1 و 2x2، الكتل 2x2 تقابل أزواج القيم الذاتية المعقدة المترافقة
!> لـ (A,B). أعمدة Q هي متجهات Schur اليسرى وأعمدة Z هي متجهات Schur اليمنى.
!>
!> إذا كانت هناك حاجة فقط للقيم الذاتية لـ (A,B)، فيجب استخدام الإجراء الفرعي الموجه
!> SGEGV بدلاً من ذلك. راجع SGEGV لوصف القيم الذاتية لمشكلة القيم الذاتية غير المتناظرة المعممة
!> (GNEP).
!>

المعلمات

JOBVSL 
!>          JOBVSL هو CHARACTER*1
!>          = 'N':  لا تحسب متجهات Schur اليسرى؛
!>          = 'V':  احسب متجهات Schur اليسرى (معادة في VSL).
!>

JOBVSR

!>          JOBVSR هو CHARACTER*1
!>          = 'N':  لا تحسب متجهات Schur اليمنى؛
!>          = 'V':  احسب متجهات Schur اليمنى (معادة في VSR).
!>

N

!>          N هو INTEGER
!>          رتبة المصفوفات A، B، VSL، و VSR. N >= 0.
!>

A

!>          A هي مصفوفة REAL، ذات بعد (LDA, N)
!>          عند الإدخال، المصفوفة A.
!>          عند الخروج، المصفوفة شبه المثلثية العلوية S من
!>          تحليل Schur المعمم.
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرائد لـ A. LDA >= max(1,N).
!>

B

!>          B هي مصفوفة REAL، ذات بعد (LDB, N)
!>          عند الإدخال، المصفوفة B.
!>          عند الخروج، المصفوفة المثلثية العلوية T من
!>          تحليل Schur المعمم.
!>

LDB

!>          LDB هو INTEGER
!>          البعد الرائد لـ B. LDB >= max(1,N).
!>

ALPHAR

!>          ALPHAR هي مصفوفة REAL، ذات بعد (N)
!>          الأجزاء الحقيقية لكل قيمة عددية alpha تحدد قيمة ذاتية
!>          لـ GNEP.
!>

ALPHAI

!>          ALPHAI هي مصفوفة REAL، ذات بعد (N)
!>          الأجزاء التخيلية لكل قيمة عددية alpha تحدد قيمة ذاتية
!>          لـ GNEP. إذا كانت ALPHAI(j) صفرًا، فإن القيمة الذاتية j
!>          حقيقية؛ إذا كانت موجبة، فإن القيم الذاتية j و (j+1)
!>          عبارة عن زوج معقد مترافق، مع
!>          ALPHAI(j+1) = -ALPHAI(j).
!>

BETA

!>          BETA هي مصفوفة REAL، ذات بعد (N)
!>          القيم العددية beta التي تحدد القيم الذاتية لـ GNEP.
!>          معًا، الكميات alpha = (ALPHAR(j),ALPHAI(j)) و
!>          beta = BETA(j) تمثل القيمة الذاتية j لزوج المصفوفات
!>          (A,B)، في أحد الأشكال lambda = alpha/beta أو
!>          mu = beta/alpha. نظرًا لأن lambda أو mu قد يتجاوزان النطاق،
!>          لا ينبغي حسابهما بشكل عام.
!>

VSL

!>          VSL هي مصفوفة REAL، ذات بعد (LDVSL,N)
!>          إذا كان JOBVSL = 'V'، فإن مصفوفة متجهات Schur اليسرى Q.
!>          لا يتم الرجوع إليها إذا كان JOBVSL = 'N'.
!>

LDVSL

!>          LDVSL هو INTEGER
!>          البعد الرائد للمصفوفة VSL. LDVSL >=1، وإذا كان
!>          JOBVSL = 'V'، فإن LDVSL >= N.
!>

VSR

!>          VSR هي مصفوفة REAL، ذات بعد (LDVSR,N)
!>          إذا كان JOBVSR = 'V'، فإن مصفوفة متجهات Schur اليمنى Z.
!>          لا يتم الرجوع إليها إذا كان JOBVSR = 'N'.
!>

LDVSR

!>          LDVSR هو INTEGER
!>          البعد الرائد للمصفوفة VSR. LDVSR >= 1، وإذا كان
!>          JOBVSR = 'V'، فإن LDVSR >= N.
!>

WORK

!>          WORK هي مصفوفة REAL، ذات بعد (MAX(1,LWORK))
!>          عند الخروج، إذا كانت INFO = 0، فإن WORK(1) تعيد LWORK المثلى.
!>

LWORK

!>          LWORK هو INTEGER
!>          بعد المصفوفة WORK. LWORK >= max(1,4*N).
!>          لأداء جيد، يجب أن يكون LWORK أكبر بشكل عام.
!>          لحساب القيمة المثلى لـ LWORK، استدع ILAENV للحصول على
!>          أحجام الكتل (لـ SGEQRF، SORMQR، و SORGQR). ثم احسب:
!>          NB  -- MAX لأحجام الكتل لـ SGEQRF، SORMQR، و SORGQR
!>          LWORK المثلى هي  2*N + N*(NB+1).
!>
!>          إذا كانت LWORK = -1، فيفترض استعلام مساحة العمل؛ الإجراء الفرعي
!>          يحسب فقط الحجم الأمثل للمصفوفة WORK، ويعيد
!>          هذه القيمة كإدخال أول في المصفوفة WORK، ولا يصدر رسالة خطأ
!>          متعلقة بـ LWORK بواسطة XERBLA.
!>

INFO

!>          INFO هو INTEGER
!>          = 0:  الخروج بنجاح
!>          < 0:  إذا كانت INFO = -i، فإن الوسيط i كان له قيمة غير قانونية.
!>          = 1,...,N:
!>                فشلت دورة QZ. (A,B) ليست في شكل Schur،
!>                ولكن يجب أن تكون ALPHAR(j)، ALPHAI(j)، و BETA(j)
!>                صحيحة لـ j=INFO+1,...,N.
!>          > N:  أخطاء تشير عادةً إلى مشاكل LAPACK:
!>                =N+1: عودة الخطأ من SGGBAL
!>                =N+2: عودة الخطأ من SGEQRF
!>                =N+3: عودة الخطأ من SORMQR
!>                =N+4: عودة الخطأ من SORGQR
!>                =N+5: عودة الخطأ من SGGHRD
!>                =N+6: عودة الخطأ من SHGEQZ (بخلاف الفشل
!>                                                في الدورة)
!>                =N+7: عودة الخطأ من SGGBAK (حساب VSL)
!>                =N+8: عودة الخطأ من SGGBAK (حساب VSR)
!>                =N+9: عودة الخطأ من SLASCL (أماكن مختلفة)
!>
