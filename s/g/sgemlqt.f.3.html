subroutine sgemlqt (character side, character trans, integer m, integer n, integer k, integer mb, real, dimension( ldv, * ) v, integer ldv, real, dimension( ldt, * ) t, integer ldt, real, dimension( ldc, * ) c, integer ldc, real, dimension( * ) work, integer info)

SGEMLQT

الغرض:

!>
!> SGEMLQT تستبدل المصفوفة C الحقيقية العامة M-by-N بـ
!>
!>                 SIDE = 'L'     SIDE = 'R'
!> TRANS = 'N':      Q C            C Q
!> TRANS = 'T':   Q**T C            C Q**T
!>
!> حيث Q هي مصفوفة متعامدة حقيقية معرفة كناتج K من المعكوسات الأولية:
!>
!>       Q = H(1) H(2) . . . H(K) = I - V T V**T
!>
!> تم إنشاؤها باستخدام تمثيل WY المدمج كما تم إرجاعه بواسطة SGELQT.
!>
!> Q هي من الرتبة M إذا كانت SIDE = 'L' ومن الرتبة N إذا كانت SIDE = 'R'.
!>

المعلمات

SIDE 
!>          SIDE is CHARACTER*1
!>          = 'L': تطبيق Q أو Q**T من اليسار؛
!>          = 'R': تطبيق Q أو Q**T من اليمين.
!>

TRANS

!>          TRANS is CHARACTER*1
!>          = 'N':  بدون تبديل، تطبيق Q؛
!>          = 'C':  تبديل، تطبيق Q**T.
!>

M

!>          M is INTEGER
!>          عدد الصفوف في المصفوفة C. M >= 0.
!>

N

!>          N is INTEGER
!>          عدد الأعمدة في المصفوفة C. N >= 0.
!>

K

!>          K is INTEGER
!>          عدد المعكوسات الأولية التي يحدد حاصل ضربها
!>          المصفوفة Q.
!>          إذا كانت SIDE = 'L'، فإن M >= K >= 0؛
!>          إذا كانت SIDE = 'R'، فإن N >= K >= 0.
!>

MB

!>          MB is INTEGER
!>          حجم الكتلة المستخدم لتخزين T.  K >= MB >= 1.
!>          يجب أن تكون هذه نفس قيمة MB المستخدمة لإنشاء T
!>          في SGELQT.
!>

V

!>          V is REAL array, dimension
!>                               (LDV,M) if SIDE = 'L',
!>                               (LDV,N) if SIDE = 'R'
!>          يجب أن يحتوي الصف i على المتجه الذي يحدد
!>          المعكوس الأولي H(i)، لـ i = 1,2,...,k، كما تم إرجاعه بواسطة
!>          SGELQT في الصفوف K الأولى من مصفوفة الوسيط A.
!>

LDV

!>          LDV is INTEGER
!>          البعد الرئيسي للمصفوفة V. LDV >= max(1,K).
!>

T

!>          T is REAL array, dimension (LDT,K)
!>          العوامل المثلثية العلوية للمعكوسات الكتلية
!>          كما تم إرجاعه بواسطة SGELQT، مخزنة كمصفوفة MB-by-K.
!>

LDT

!>          LDT is INTEGER
!>          البعد الرئيسي للمصفوفة T.  LDT >= MB.
!>

C

!>          C is REAL array, dimension (LDC,N)
!>          عند الإدخال، المصفوفة C ذات الأبعاد M-by-N.
!>          عند الخروج، يتم استبدال C بـ Q C، Q**T C، C Q**T أو C Q.
!>

LDC

!>          LDC is INTEGER
!>          البعد الرئيسي للمصفوفة C. LDC >= max(1,M).
!>

WORK

!>          WORK is REAL array. The dimension of
!>          WORK is N*MB if SIDE = 'L', or  M*MB if SIDE = 'R'.
!>          (أبعاد WORK هي N*MB إذا كانت SIDE = 'L'، أو M*MB إذا كانت SIDE = 'R'.)
!>

INFO

!>          INFO is INTEGER
!>          = 0:  انتهاء ناجح
!>          < 0:  إذا كان INFO = -i، فإن الوسيط i له قيمة غير قانونية
!>
