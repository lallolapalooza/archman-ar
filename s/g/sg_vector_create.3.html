الوصف

sg_vector_create() تقوم بتخصيص وتهيئة متجه statgrab جديد مع استخدام initial_used عنصر جاهز للاستخدام. يتم تخصيص مساحة لـ alloc_count عنصر في البداية (لتجنب الكثير من استدعاءات realloc() أثناء استدعاءات sg_vector_resize() اللاحقة). يجب أن يكون قيمة block_size قوة للرقم 2، ويتم تقريبها لأعلى إلى أقرب قوة للعدد 2 عندما لا تكون كذلك. إذا لم يكن alloc_count مضاعفًا لـ block_size، فسيتم تقريبه لأعلى إلى أقرب مضاعف لـ block_size. تُرجع مؤشرًا إلى المتجه المنشأ حديثًا.

sg_vector_clear() تدمر جميع العناصر الموجودة في المتجه المعطى. على عكس sg_vector_resize( x, 0 )، يبقى حجم المتجه المخصص دون تغيير.

sg_vector_resize() يزيد أو يقلل من عدد العناصر المخصصة في المتجه المحدد. عدد العناصر المخصصة هو دائمًا مضاعف لمعلمة التهيئة block_size. في حالة خاصة، يتم استدعاء sg_vector_resize() مع 0 في الوسيط new_count، يتم تحرير المتجه بعد تدمير جميع عناصر المتجه. تُرجع مؤشرًا إلى المتجه المعاد تحجيمه.

sg_vector_free() تدمر جميع عناصر المتجه وتلغي تخصيص المساحة الخاصة بالمتجه المعطى.

sg_vector_clone() تستنسخ جميع عناصر المتجه المعطى إلى متجه جديد تم إنشاؤه بنفس المواصفات كما هو مرجع. تُرجع مؤشرًا إلى المتجه المستنسخ.

sg_vector_clone_into() تستنسخ جميع عناصر متجه المصدر المعطى إلى متجه الوجهة المعطى. يجب أن يكون متجه الوجهة قد تم إنشاؤه لنفس نوع بيانات العنصر مثل متجه المصدر. تُرجع رمز خطأ != SG_ERROR_NONE إذا حدث خطأ ما.

sg_vector_compute_diff() تحسب متجه فرق بين المتجه الذي يحتوي على الإحصائيات الحالية ومتجه آخر يحتوي على إحصائيات أقدم. إذا كان عنصر موجودًا في المتجه الحالي ولكن ليس في المتجه المقابل، فسيتم استنساخه في متجه النتيجة. إذا كان العنصر موجودًا فقط في المتجه المقابل، فلن يظهر في متجه الوجهة. تُرجع sg_vector_compute_diff() رمز خطأ != SG_ERROR_NONE إذا حدث خطأ ما.

sg_prove_vector() تثبت ما إذا كان مؤشر إلى متجه يشير حقًا إلى متجه. في حالة ما إذا كان مؤشر المتجه المعطى يشير إلى بيانات تالفة، يتم إنهاء البرنامج. عندما تُرجع sg_prove_vector()، فإنها تُرجع SG_ERROR_NONE.

sg_get_nelements() تُرجع عدد العناصر التي تحتوي عليها منطقة البيانات المعطاة، والتي يحيط بها متجه statgrab. يتم حساب رأس المتجه داخليًا من المؤشر المعطى لأول عنصر في المتجه.

sg_free_stats_buf() تحرر المتجه الذي يحيط بمنطقة البيانات المعطاة.

ملاحظات

باستثناء sg_get_nelements() و sg_free_stats_buf()، لا يمكن استدعاء أي من الوظائف المذكورة أعلاه من خارج مصادر libstatgrab. قد تتغير الهياكل وواجهات برمجة التطبيقات الموثقة دون سابق إنذار. وصف واجهة برمجة التطبيقات الأخرى مخصص للقراءة من مطوري libstatgrab فقط.

كل متجه تم إنشاؤه من عنصرين: معلومات المتجه وقائمة العناصر:

template <class T, class Impl>
struct sg_vector {
        size_t used_count;
        size_t alloc_count;
        size_t block_shift;
        Impl vector_implementation;
        T elements[alloc_count];
};

بالطبع، هذا ليس C صالحًا، لذا كان الحل هو الحيلة:

typedef struct sg_vector {
	size_t used_count;
	size_t alloc_count;
	size_t block_shift;
	struct sg_vector_init_info info;
} sg_vector;
struct sg_vector_size_helper {
	struct sg_vector v;
	long long ll;
};
#define VECTOR_SIZE offsetof(struct sg_vector_size_helper,ll)
/* Return the data ptr of a vector */
#define VECTOR_DATA(vector) \
	(vector ? (void *)(((char *)vector)+VECTOR_SIZE) : NULL)
#define VECTOR_ADDR_ARITH(ptr) \
	(sg_vector *)(((char *)(ptr))-VECTOR_SIZE)
/* Return the vector for a data */
#define VECTOR_ADDRESS(ptr) \
	((ptr) ? (SG_ERROR_NONE == sg_prove_vector(VECTOR_ADDR_ARITH(ptr)) ? VECTOR_ADDR_ARITH(ptr) : NULL ) : NULL)

هذا يسمح أيضًا للوظائف المستخدمة مثل sg_get_nelements() و sg_free_stats_buf() بالتبديل بسهولة بين بنية المتجه والمحتوى.

بنية تخصص المتجه

كما ذكرنا، تنفذ آلية المتجه استراتيجيات من مفهوم البرمجة كائنية التوجه المسمى "تعدد الأشكال". يتم وصف المتجه بكائن صغير يحتوي على سمات متأصلة مثل حجم العنصر ومجموعة من "الطرق الافتراضية" لأداء مهام متعلقة بالعنصر مثل تهيئة العناصر أو تدميرها.

typedef void (*vector_init_function)(void *item);
typedef sg_error (*vector_copy_function)(const void *src, void *dst);
typedef sg_error (*vector_compute_diff_function)(void *dst, const void *src);
typedef int (*vector_compare_function)(const void *a, const void *b);
typedef void (*vector_destroy_function)(void *item);
struct sg_vector_init_info {
        size_t item_size;
        vector_init_function init_fn;
        vector_copy_function copy_fn;
        vector_compute_diff_function compute_diff_fn;
        vector_compare_function compare_fn;
        vector_destroy_function destroy_fn;
};

يتم تهيئة مثيلات struct sg_vector_init_info مفاهيميًا بشكل ثابت عن طريق استخدام ماكرو المعالج المسبق VECTOR_INIT_INFO_FULL_INIT(type) أو VECTOR_INIT_INFO_EMPTY_INIT(type). إليك بعض الأمثلة لتوضيح كيفية القيام بذلك:

تهيئة وصف متجه إحصائيات وحدة المعالجة المركزية

VECTOR_INIT_INFO_EMPTY_INIT(sg_cpu_stats);

تهيئة وصف متجه إحصائيات معلومات المضيف

static void sg_os_stats_item_init(sg_os_stats *d);
static void sg_os_stats_item_destroy(sg_os_stats *d);
#define sg_os_stats_item_copy NULL
#define sg_os_stats_item_compute_diff NULL
#define sg_os_stats_item_compare NULL
VECTOR_INIT_INFO_FULL_INIT(sg_os_stats);

تهيئة وصف متجه إحصائيات إدخال/إخراج القرص

static void sg_disk_io_stats_item_init(sg_disk_io_stats *d);
static sg_error sg_disk_io_stats_item_copy(sg_disk_io_stats *d, const sg_disk_io_stats *s);
static sg_error sg_disk_io_stats_item_compute_diff(const sg_disk_io_stats *s, sg_disk_io_stats *d);
static int sg_disk_io_stats_item_compare(const sg_disk_io_stats *a, const sg_disk_io_stats *b);
static void sg_disk_io_stats_item_destroy(sg_disk_io_stats *d);
VECTOR_INIT_INFO_FULL_INIT(sg_disk_io_stats);
