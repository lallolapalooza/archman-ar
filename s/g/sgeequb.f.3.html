subroutine sgeequb (integer m, integer n, real, dimension( lda, * ) a, integer lda, real, dimension( * ) r, real, dimension( * ) c, real rowcnd, real colcnd, real amax, integer info)

SGEEQUB

الغرض:

!>
!> SGEEQUB يحسب عوامل قياس الصفوف والأعمدة التي تهدف إلى موازنة مصفوفة
!> M×N (A) وتقليل عدد الشرط الخاص بها. R تُرجع عوامل قياس الصفوف و C عوامل
!> قياس الأعمدة، التي تم اختيارها لمحاولة جعل أكبر عنصر في كل صف وعمود
!> في المصفوفة B ذات العناصر B(i,j)=R(i)*A(i,j)*C(j) لها قيمة مطلقة لا تتجاوز
!> الأساس (radix).
!>
!> R(i) و C(j) مقيدة لتكون قوة للأساس بين SMLNUM = أصغر رقم آمن و BIGNUM =
!> أكبر رقم آمن. استخدام عوامل القياس هذه لا يضمن تقليل عدد الشرط لـ A
!> ولكنه يعمل بشكل جيد في الممارسة.
!>
!> هذا الإجراء يختلف عن SGEEQU بتقييد عوامل القياس إلى قوة للأساس.
!> بصرف النظر عن تجاوزات الفائض والنقصان، فإن القياس بهذه العوامل لا يقدم
!> أخطاء تقريب إضافية. ومع ذلك، فإن قيم العناصر المقاسة لم تعد تقريبًا 1
!> ولكنها تقع بين sqrt(radix) و 1/sqrt(radix).
!>

المعلمات

M 
!>          M هو INTEGER
!>          عدد صفوف المصفوفة A. M >= 0.
!>

N

!>          N هو INTEGER
!>          عدد أعمدة المصفوفة A. N >= 0.
!>

A

!>          A هي مصفوفة REAL، ذات بعد (LDA,N)
!>          المصفوفة M×N التي سيتم حساب عوامل الموازنة الخاصة بها.
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرئيسي للمصفوفة A. LDA >= max(1,M).
!>

R

!>          R هي مصفوفة REAL، ذات بعد (M)
!>          إذا كانت INFO = 0 أو INFO > M، فإن R تحتوي على عوامل قياس الصفوف
!>          لـ A.
!>

C

!>          C هي مصفوفة REAL، ذات بعد (N)
!>          إذا كانت INFO = 0، فإن C تحتوي على عوامل قياس الأعمدة لـ A.
!>

ROWCND

!>          ROWCND هي REAL
!>          إذا كانت INFO = 0 أو INFO > M، فإن ROWCND تحتوي على نسبة أصغر R(i)
!>          إلى أكبر R(i). إذا كانت ROWCND >= 0.1 و AMAX ليست كبيرة جدًا أو صغيرة
!>          جدًا، فلا فائدة من القياس بـ R.
!>

COLCND

!>          COLCND هي REAL
!>          إذا كانت INFO = 0، فإن COLCND تحتوي على نسبة أصغر C(i) إلى أكبر
!>          C(i). إذا كانت COLCND >= 0.1، فلا فائدة من القياس بـ C.
!>

AMAX

!>          AMAX هي REAL
!>          القيمة المطلقة لأكبر عنصر في المصفوفة. إذا كان AMAX قريبًا جدًا
!>          من تجاوز السعة أو قريبًا جدًا من النقصان، فيجب قياس المصفوفة.
!>

INFO

!>          INFO هو INTEGER
!>          = 0: انتهاء ناجح
!>          < 0: إذا كانت INFO = -i، فإن المعلمة i كانت قيمة غير قانونية
!>          > 0: إذا كانت INFO = i، و i
!>                <= M: الصف i من A هو صفر تمامًا
!>                >  M: العمود (i-M) من A هو صفر تمامًا
!>
