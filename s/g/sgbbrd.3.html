توثيق الوظيفة/الروتين الفرعي

subroutine sgbbrd (character vect, integer m, integer n, integer ncc, integer kl, integer ku, real, dimension( ldab, * ) ab, integer ldab, real, dimension( * ) d, real, dimension( * ) e, real, dimension( ldq, * ) q, integer ldq, real, dimension( ldpt, * ) pt, integer ldpt, real, dimension( ldc, * ) c, integer ldc, real, dimension( * ) work, integer info)

SGBBRD

الغرض:

!>
!> يقوم SGBBRD بتقليل مصفوفة A العامة ذات النطاق MxN الحقيقية إلى شكل ثنائي الأقطاب B عن طريق تحويل متعامد: Q**T * A * P = B.
!>
!> تقوم الروتين بحساب B، وتشكيل Q أو P**T اختياريًا، أو حساب
!> Q**T*C لمصفوفة C معطاة.
!>

المعلمات

VECT 
!>          VECT هو CHARACTER*1
!>          يحدد ما إذا كانت سيتم تشكيل المصفوفتين Q و P**T أم لا.
!>          = 'N': لا تشكل Q أو P**T؛
!>          = 'Q': شكل Q فقط؛
!>          = 'P': شكل P**T فقط؛
!>          = 'B': شكل كليهما.
!>

M

!>          M هو INTEGER
!>          عدد صفوف المصفوفة A. M >= 0.
!>

N

!>          N هو INTEGER
!>          عدد أعمدة المصفوفة A. N >= 0.
!>

NCC

!>          NCC هو INTEGER
!>          عدد أعمدة المصفوفة C. NCC >= 0.
!>

KL

!>          KL هو INTEGER
!>          عدد الأقطار الفرعية للمصفوفة A. KL >= 0.
!>

KU

!>          KU هو INTEGER
!>          عدد الأقطار الفائقة للمصفوفة A. KU >= 0.
!>

AB

!>          AB هي مصفوفة REAL، ذات أبعاد (LDAB,N)
!>          عند الإدخال، مصفوفة النطاق A ذات الأبعاد MxN، مخزنة في الصفوف من 1 إلى
!>          KL+KU+1. يتم تخزين العمود j من A في العمود j من
!>          المصفوفة AB كالتالي:
!>          AB(ku+1+i-j,j) = A(i,j) لـ max(1,j-ku) <= i <= min(m,j+kl).
!>          عند الخروج، يتم الكتابة فوق A بقيم تم إنشاؤها أثناء
!>          الاختزال.
!>

LDAB

!>          LDAB هو INTEGER
!>          البعد الرائد للمصفوفة A. LDAB >= KL+KU+1.
!>

D

!>          D هي مصفوفة REAL، ذات أبعاد (min(M,N))
!>          العناصر القطرية للمصفوفة ثنائية الأقطاب B.
!>

E

!>          E هي مصفوفة REAL، ذات أبعاد (min(M,N)-1)
!>          العناصر القطرية الفائقة للمصفوفة ثنائية الأقطاب B.
!>

Q

!>          Q هي مصفوفة REAL، ذات أبعاد (LDQ,M)
!>          إذا كان VECT = 'Q' أو 'B'، فإن المصفوفة المتعامدة Q ذات الأبعاد MxM.
!>          إذا كان VECT = 'N' أو 'P'، فلا يتم الاستفادة من المصفوفة Q.
!>

LDQ

!>          LDQ هو INTEGER
!>          البعد الرائد للمصفوفة Q.
!>          LDQ >= max(1,M) إذا كان VECT = 'Q' أو 'B'؛ LDQ >= 1 بخلاف ذلك.
!>

PT

!>          PT هي مصفوفة REAL، ذات أبعاد (LDPT,N)
!>          إذا كان VECT = 'P' أو 'B'، فإن المصفوفة المتعامدة P' ذات الأبعاد NxN.
!>          إذا كان VECT = 'N' أو 'Q'، فلا يتم الاستفادة من المصفوفة PT.
!>

LDPT

!>          LDPT هو INTEGER
!>          البعد الرائد للمصفوفة PT.
!>          LDPT >= max(1,N) إذا كان VECT = 'P' أو 'B'؛ LDPT >= 1 بخلاف ذلك.
!>

C

!>          C هي مصفوفة REAL، ذات أبعاد (LDC,NCC)
!>          عند الإدخال، مصفوفة C ذات الأبعاد MxNCC.
!>          عند الخروج، يتم الكتابة فوق C بـ Q**T*C.
!>          لا يتم الاستفادة من C إذا كان NCC = 0.
!>

LDC

!>          LDC هو INTEGER
!>          البعد الرائد للمصفوفة C.
!>          LDC >= max(1,M) إذا كان NCC > 0؛ LDC >= 1 إذا كان NCC = 0.
!>

WORK

!>          WORK هي مصفوفة REAL، ذات أبعاد (2*max(M,N))
!>

INFO

!>          INFO هو INTEGER
!>          = 0: الخروج بنجاح.
!>          < 0: إذا كان INFO = -i، فإن المعلمة i كانت لها قيمة غير قانونية.
!>
