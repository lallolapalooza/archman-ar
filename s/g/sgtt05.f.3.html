توثيق الدالة/الروتين الفرعي

subroutine sgtt05 (character trans, integer n, integer nrhs, real, dimension( * ) dl, real, dimension( * ) d, real, dimension( * ) du, real, dimension( ldb, * ) b, integer ldb, real, dimension( ldx, * ) x, integer ldx, real, dimension( ldxact, * ) xact, integer ldxact, real, dimension( * ) ferr, real, dimension( * ) berr, real, dimension( * ) reslts)

SGTT05

الغرض:

!>
!> SGTT05 تختبر حدود الخطأ من التنقيح التكراري للحل المحسوب
!> لنظام المعادلات A*X = B، حيث A مصفوفة ثلاثية قطرية عامة
!> من الرتبة n و op(A) = A أو A**T، اعتمادًا على TRANS.
!>
!> RESLTS(1) = اختبار حد الخطأ
!>           = norm(X - XACT) / ( norm(X) * FERR )
!>
!> قيمة كبيرة تُعاد إذا كانت هذه النسبة ليست أقل من واحد.
!>
!> RESLTS(2) = الخطأ المتبقي من روتين التنقيح التكراري
!>           = الحد الأقصى لـ BERR / ( NZ*EPS + (*) )، حيث
!>             (*) = NZ*UNFL / (min_i (abs(op(A))*abs(X) +abs(b))_i )
!>             و NZ = الحد الأقصى لعدد العناصر غير الصفرية في أي صف من A، زائد 1
!>

المعاملات:

TRANS 
!>          TRANS هو CHARACTER*1
!>          يحدد شكل نظام المعادلات.
!>          = 'N':  A * X = B     (بدون تبديل)
!>          = 'T':  A**T * X = B  (تبديل)
!>          = 'C':  A**H * X = B  (تبديل مترافق = تبديل)
!>

N

!>          N هو INTEGER
!>          عدد صفوف المصفوفات X و XACT.  N >= 0.
!>

NRHS

!>          NRHS هو INTEGER
!>          عدد أعمدة المصفوفات X و XACT.  NRHS >= 0.
!>

DL

!>          DL هو مصفوفة REAL، البعد (N-1)
!>          العناصر الفرعية القطرية (n-1) للمصفوفة A.
!>

D

!>          D هو مصفوفة REAL، البعد (N)
!>          العناصر القطرية للمصفوفة A.
!>

DU

!>          DU هو مصفوفة REAL، البعد (N-1)
!>          العناصر الفائقة القطرية (n-1) للمصفوفة A.
!>

B

!>          B هو مصفوفة REAL، البعد (LDB,NRHS)
!>          متجهات الطرف الأيمن لنظام المعادلات الخطية.
!>

LDB

!>          LDB هو INTEGER
!>          البعد الرئيسي للمصفوفة B.  LDB >= max(1,N).
!>

X

!>          X هو مصفوفة REAL، البعد (LDX,NRHS)
!>          متجهات الحل المحسوبة.  يتم تخزين كل متجه كعمود في المصفوفة X.
!>

LDX

!>          LDX هو INTEGER
!>          البعد الرئيسي للمصفوفة X.  LDX >= max(1,N).
!>

XACT

!>          XACT هو مصفوفة REAL، البعد (LDX,NRHS)
!>          متجهات الحل الدقيقة.  يتم تخزين كل متجه كعمود في المصفوفة XACT.
!>

LDXACT

!>          LDXACT هو INTEGER
!>          البعد الرئيسي للمصفوفة XACT.  LDXACT >= max(1,N).
!>

FERR

!>          FERR هو مصفوفة REAL، البعد (NRHS)
!>          تقديرات حدود الخطأ الأمامي لكل متجه حل X.  إذا كانت XTRUE هي الحل الحقيقي،
!>          فإن FERR تحد من حجم أكبر مدخل في (X - XTRUE) مقسومًا على حجم
!>          أكبر مدخل في X.
!>

BERR

!>          BERR هو مصفوفة REAL، البعد (NRHS)
!>          الخطأ الخلفي النسبي المكون لكل متجه حل
!>          (أي، أصغر تغيير نسبي في أي مدخل في A أو B يجعل X حلاً دقيقًا).
!>

RESLTS

!>          RESLTS هو مصفوفة REAL، البعد (2)
!>          الحد الأقصى على متجهات الحل NRHS للنسب:
!>          RESLTS(1) = norm(X - XACT) / ( norm(X) * FERR )
!>          RESLTS(2) = BERR / ( NZ*EPS + (*) )
!>
