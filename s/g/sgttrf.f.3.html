subroutine sgttrf (integer n, real, dimension( * ) dl, real, dimension( * ) d, real, dimension( * ) du, real, dimension( * ) du2, integer, dimension( * ) ipiv, integer info)

SGTTRF

الغرض:

!>
!> SGTTRF يحسب تحليل LU لمصفوفة ثلاثية قطرية حقيقية A
!> باستخدام الحذف مع التقسيم الجزئي وتبديلات الصفوف.
!>
!> التحليل له الشكل
!>    A = L * U
!> حيث L هي حاصل ضرب مصفوفات تبديلية سفلية ثنائية القطر الوحدة
!> و U هي مصفوفة مثلثية عليا مع عناصر غير صفرية فقط في القطر الرئيسي
!> والقطرين الخارجيين الفائقين.
!>

المعاملات

N 
!>          N هو عدد صحيح (INTEGER)
!>          رتبة المصفوفة A.
!>

DL

!>          DL هو مصفوفة حقيقية (REAL array)، البعد (N-1)
!>          عند الإدخال، يجب أن تحتوي DL على عناصر القطر الفرعي (n-1) للمصفوفة
!>          A.
!>
!>          عند الخروج، يتم الكتابة فوق DL بالعوامل (n-1) التي
!>          تحدد المصفوفة L من تحليل LU للمصفوفة A.
!>

D

!>          D هو مصفوفة حقيقية (REAL array)، البعد (N)
!>          عند الإدخال، يجب أن تحتوي D على العناصر القطرية للمصفوفة A.
!>
!>          عند الخروج، يتم الكتابة فوق D بالعناصر القطرية n للمصفوفة
!>          المثلثية العليا U من تحليل LU للمصفوفة A.
!>

DU

!>          DU هو مصفوفة حقيقية (REAL array)، البعد (N-1)
!>          عند الإدخال، يجب أن تحتوي DU على عناصر القطر الفائق (n-1)
!>          للمصفوفة A.
!>
!>          عند الخروج، يتم الكتابة فوق DU بالعناصر (n-1) للقطر الفائق الأول
!>          للمصفوفة U.
!>

DU2

!>          DU2 هو مصفوفة حقيقية (REAL array)، البعد (N-2)
!>          عند الخروج، يتم الكتابة فوق DU2 بالعناصر (n-2) للقطر الفائق الثاني
!>          للمصفوفة U.
!>

IPIV

!>          IPIV هو مصفوفة أعداد صحيحة (INTEGER array)، البعد (N)
!>          فهارس المحاور؛ بالنسبة لـ 1 <= i <= n، تم تبديل الصف i للمصفوفة
!>          مع الصف IPIV(i). سيكون IPIV(i) دائمًا إما
!>          i أو i+1؛ يشير IPIV(i) = i إلى عدم الحاجة لتبديل صف.
!>

INFO

!>          INFO هو عدد صحيح (INTEGER)
!>          = 0:  إنهاء ناجح
!>          < 0:  إذا كان INFO = -k، فقد كان المعامل k غير قانوني
!>          > 0:  إذا كان INFO = k، فإن U(k,k) يساوي صفرًا بالضبط. تم
!>                إكمال التحليل، ولكن العامل U هو بالضبط
!>                منفرد، وسيحدث قسمة على صفر إذا تم استخدامه
!>                لحل نظام معادلات.
!>
