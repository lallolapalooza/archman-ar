subroutine sgeqrt2 (m, n, a, lda, t, ldt, info)

SGEQRT2 تحسب تحليل QR لمصفوفة حقيقية أو مركبة عامة باستخدام تمثيل WY المدمج للمصفوفة Q.

توثيق الوظيفة/الإجراء الفرعي

subroutine sgeqrt2 (integer m, integer n, real, dimension( lda, * ) a, integer lda, real, dimension( ldt, * ) t, integer ldt, integer info)

SGEQRT2 تحسب تحليل QR لمصفوفة حقيقية عامة M-by-N A، باستخدام تمثيل WY المدمج للمصفوفة Q.

الغرض:

!>
!> SGEQRT2 تحسب تحليل QR لمصفوفة حقيقية عامة A بحجم M-by-N،
!> باستخدام تمثيل WY المدمج للمصفوفة Q.
!>

المعلمات

M 
!>          M هو عدد صحيح (INTEGER)
!>          عدد صفوف المصفوفة A.  يجب أن يكون M >= N.
!>

N

!>          N هو عدد صحيح (INTEGER)
!>          عدد أعمدة المصفوفة A.  يجب أن يكون N >= 0.
!>

A

!>          A هي مصفوفة أعداد حقيقية (REAL array)، أبعادها (LDA,N)
!>          عند الدخول، تحتوي العناصر الموجودة على القطر وفوقه على المصفوفة المثلثية العلوية R بحجم N-by-N؛
!>          العناصر الموجودة تحت القطر تحتوي على أعمدة V. انظر أدناه لمزيد من التفاصيل.
!>

LDA

!>          LDA هو عدد صحيح (INTEGER)
!>          البعد الرائد للمصفوفة A.  يجب أن يكون LDA >= max(1,M).
!>

T

!>          T هي مصفوفة أعداد حقيقية (REAL array)، أبعادها (LDT,N)
!>          المصفوفة المثلثية العلوية بحجم N-by-N للعامل المنعكس المدمج.
!>          العناصر الموجودة على القطر وفوقه تحتوي على العامل المنعكس المدمج T؛
!>          العناصر الموجودة تحت القطر غير مستخدمة.
!>          انظر أدناه لمزيد من التفاصيل.
!>

LDT

!>          LDT هو عدد صحيح (INTEGER)
!>          البعد الرائد للمصفوفة T.  يجب أن يكون LDT >= max(1,N).
!>

INFO

!>          INFO هو عدد صحيح (INTEGER)
!>          = 0: انتهاء ناجح
!>          < 0: إذا كان INFO = -i، فإن المعلمة i كانت لها قيمة غير قانونية
!>

تفاصيل إضافية:

!>
!>  المصفوفة V تخزن العوامل المنعكسة الأولية H(i) في العمود i
!>  تحت القطر. على سبيل المثال، إذا كان M=5 و N=3، فإن المصفوفة V هي
!>
!>               V = (  1       )
!>                   ( v1  1    )
!>                   ( v1 v2  1 )
!>                   ( v1 v2 v3 )
!>                   ( v1 v2 v3 )
!>
!>  حيث تمثل الـ vi's المتجهات التي تحدد H(i)، والتي يتم إرجاعها
!>  في المصفوفة A.  الـ 1's على طول قطر V غير مخزنة في A.  العامل المنعكس المدمج H يُعطى بعد ذلك بـ
!>
!>               H = I - V * T * V**T
!>
!>  حيث V**T هو منقول المصفوفة V.
!>
