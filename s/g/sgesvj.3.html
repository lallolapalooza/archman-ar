sgesvj(3)

الحزمة: lapack-doc

التراخيص: custom

الاسم

sgesvj - حساب تحلل القيم المفردة (SVD) لمصفوفة حقيقية عامة.

الملخص

SUBROUTINE SGESVJ(
   JOBV, JOBR, JOBU, M, N, A, LDA, SVA, U, LDU, V, LDV, WORK, LWORK,
   INFO
)
CHARACTER(LEN=1) :: JOBV, JOBR, JOBU
INTEGER :: M, N, LDA, LDU, LDV, LWORK, INFO
REAL :: A( LDA, * ), SVA( * ), U( LDU, * ), V( LDV, * ), WORK( * )

وثائق الوظيفة/الروتين الفرعي

SGESVJ يحسب تحلل القيم المفردة (SVD) لمصفوفة حقيقية عامة.

المعلمات

JOBV

CHARACTER*1. يحدد حساب المتجهات المفردة اليمنى (V).

= 'N': لا يتم حساب V.
= 'V': يتم حساب V.
JOBR

CHARACTER*1. يحدد حساب الصفائف المتعامدة (R).

= 'N': لا يتم حساب R.
= 'R': يتم حساب R.
JOBU

CHARACTER*1. يحدد حساب المتجهات المفردة اليسرى (U).

= 'N': لا يتم حساب U.
= 'A': يتم حساب U.
= 'S': يتم حساب U، ولكن لا يتم إرجاع إلا الأعمدة الأولى K، حيث K هو عدد القيم المفردة غير الصفرية.
= 'C': يتم حساب U، ولكن لا يتم إرجاع إلا الأعمدة الأولى K، حيث K هو عدد القيم المفردة غير الصفرية.
M

INTEGER.

عدد الصفوف في المصفوفة A. يجب أن يكون M >= 0.
N

INTEGER.

عدد الأعمدة في المصفوفة A. يجب أن يكون N >= 0.
A

REAL مصفوفة، حجم (LDA, *).

على الإدخال، تحتوي على مصفوفة الإدخال A.
على الخرج، إذا كان JOBU = 'A' أو 'S' أو 'C'، أو JOBV = 'V'، فقد تم استبدال A بنسخة محسوبة منها.
LDA

INTEGER.

حجم الصف الرئيسي للمصفوفة A. يجب أن يكون LDA >= max(1, M).
SVA

REAL مصفوفة، حجم (MAX(1, MIN(M,N))).

على الخرج، تحتوي على القيم المفردة المحسوبة لـ A بالترتيب غير المتزايد.
U

REAL مصفوفة، حجم (LDU, *).

على الخرج، تحتوي على المتجهات المفردة اليسرى المتعامدة.
LDU

INTEGER.

حجم الصف الرئيسي للمصفوفة U.
إذا كان JOBU = 'A' أو 'C'، فإن LDU >= max(1, M).
إذا كان JOBU = 'S'، فإن LDU >= max(1, M).
V

REAL مصفوفة، حجم (LDV, *).

على الخرج، تحتوي على المتجهات المفردة اليمنى المتعامدة.
LDV

INTEGER.

حجم الصف الرئيسي للمصفوفة V.
إذا كان JOBV = 'V'، فإن LDV >= max(1, N).
WORK

REAL مصفوفة، حجم (MAX(1,LWORK)).

إذا كان JOBU = 'C':
WORK(1) = CTOL، حيث يحدد CTOL عتبة التقارب.
        تتوقف العملية إذا كانت جميع أعمدة A متعامدة بشكل متبادل حتى CTOL*EPS، EPS=SLAMCH('E').
        يجب أن يكون CTOL >= ONE، أي لا يُسمح بإجبار الروتين على الحصول على التعامد أقل من EPSILON.

على الخرج:
WORK(1) = SCALE هو عامل القياس بحيث يكون SCALE*SVA(1:N)
        هي القيم المفردة المحسوبة لـ A.
        (انظر وصف SVA().)
WORK(2) = NINT(WORK(2)) هو عدد القيم المفردة غير الصفرية المحسوبة.
WORK(3) = NINT(WORK(3)) هو عدد القيم المفردة المحسوبة الأكبر من عتبة الانخفاض.
WORK(4) = NINT(WORK(4)) هو عدد دورات دوران Jacobi اللازمة للتقارب العددي.
WORK(5) = max_{i.NE.j} |COS(A(:,i),A(:,j))| في الدورة الأخيرة.
        هذه معلومات مفيدة في الحالات التي لم تتقارب فيها SGESVJ، حيث يمكن استخدامها لتقدير ما إذا كانت
        المخرجات لا تزال مفيدة وللتحليل اللاحق.
WORK(6) = أكبر قيمة مطلقة على جميع جيوب زوايا دوران Jacobi في الدورة الأخيرة. يمكن أن تكون
        مفيدة للتحليل اللاحق.
LWORK

INTEGER.

طول WORK.
LWORK >= 1، إذا كان MIN(M,N) = 0، و LWORK >= MAX(6,M+N)، بخلاف ذلك.

إذا كان LWORK = -1 عند الإدخال، فسيتم افتراض استعلام مساحة العمل ولن يتم إجراء أي حساب؛
سيتم تعيين WORK(1) إلى الطول الأدنى (والأمثل) لـ WORK.
INFO

INTEGER.

= 0: خروج ناجح.
< 0: إذا كان INFO = -i، فإن المعلمة i كان لها قيمة غير قانونية.
> 0: لم تتقارب SGESVJ في الحد الأقصى المسموح به (30) من الدورات. قد تكون المخرجات لا تزال مفيدة. انظر وصف WORK.

تفاصيل إضافية:

يتم الحصول على المصفوفة المتعامدة N-by-N V كناتج لدورانات مستوى Jacobi. يتم تنفيذ الدورانات كدورانات قياس سريعة لـ Anda و Park [1]. في حالة انخفاض زاوية Jacobi، يتم استخدام تحويل Jacobi معدل لـ Drmac [4]. تستخدم استراتيجية الارتكاز تبادلات الأعمدة لـ de Rijk [2]. دقة القيم المفردة المحسوبة ودقة المتجهات المفردة المحسوبة (بمقياس الزاوية) هي كما تضمنها نظرية Demmel و Veselic [3]. رقم الحالة الذي يحدد الدقة في حالة الرتبة الكاملة هو أساسًا min_{D=diag} kappa(A*D)، حيث kappa(.) هو رقم الحالة الطيفي. أفضل أداء لروتين Jacobi SVD هذا يتحقق إذا تم استخدامه في نسخة مسرعة من Drmac و Veselic [5,6]، وهو الروتين الأساسي في مكتبة SIGMA [7]. تتوفر بعض معلمات الضبط (مُشار إليها بـ [TP]) للمنفذ. النطاق الحسابي للقيم المفردة غير الصفرية هو فترة الأرقام الآلية ( UNDERFLOW ، OVERFLOW ). في الحالات القصوى، يمكن حتى حساب القيم المفردة المنحلة مع فقدان مماثل للدقة الدقيقة.
