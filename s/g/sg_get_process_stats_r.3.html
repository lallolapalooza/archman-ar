الوصف

توفر الدوال sg_get_process_stats إحصائيات حول العمليات قيد التشغيل حاليًا. كلتا الدالتين، sg_get_process_stats() و sg_get_process_stats_r()، تأخذ معلمة إدخالات اختيارية، تشير (عند تقديمها) إلى حجمsize_t لأخذ عدد إدخالات المتجه المُرجعة.

توفر الدالتان sg_get_process_count_of() و sg_get_process_count_r() عرضًا مجمعًا لجدول العمليات - إنهما تقدمان كمية العمليات لكل حالة عملية. sg_get_process_count() هي في الواقع وحدة ماكرو معالجة مسبقة للتوافق الرجعي وتستدعي sg_get_process_count_of() مع المعلمةpcs من sg_entire_process_count لمحاكاة السلوك حتى الإصدار 0.17.

تقارن هذه الدوال إدخالاتsg_process_stats اثنتين، وترجعint لتمثيل أيهما أكبر. الاستخدام الرئيسي لهذه الدوال هو تمريرها إلىqsort لفرزsg_process_stats حسب النوع المحدد.

مثال

size_t entries;
sg_process_stats *proc_stats = NULL;
while( NULL != ( proc_stats = sg_get_process_stats_r(&entries) ) ) {
    /* ترتيب الإدخالات بمقارنة معرف العملية */
    qsort( proc_stats, entries, sizeof(proc_stats[0]), &sg_process_compare_pid );
    show_proc_stats( proc_stats );
    sg_free_process_stats( proc_stats );
}

قيم الإرجاع

الهيكل الذي يتم إرجاعه بواسطة sg_get_process_stats هو من النوع sg_process_stats.

typedef struct {
        char *process_name;
        char *proctitle;
        pid_t pid;    /* معرف العملية */
        pid_t parent; /* معرف العملية الأب */
        pid_t pgid;   /* معرف العملية لقائد مجموعة العمليات */
        pid_t sessid; /* معرف الجلسة التي تنتمي إليها العملية */
        uid_t uid;
        uid_t euid;
        gid_t gid;
        gid_t egid;
        unsigned long long context_switches;
        unsigned long long voluntary_context_switches;
        unsigned long long involuntary_context_switches;
        unsigned long long proc_size; /* بالبايت */
        unsigned long long proc_resident; /* بالبايت */
        time_t start_time; /* متى بدأت العملية */
        time_t time_spent; /* الوقت المستغرق في التشغيل بالثواني */
        double cpu_percent;
        int nice;
        sg_process_state state;
        time_t systime;
} sg_process_stats;

typedef enum {

process_name
اسم الأمر الذي تم تشغيله. يعتمد محتوى هذا الحقل بشكل كبير على نظام التشغيل الأساسي، يخزن البعض اسم الملف التنفيذي الأساسي الذي تمرره استدعاء نظام exec(2)، ويخزن البعض الآخر المسار الكامل. معظم أنظمة التشغيل تقيد حجم هذا الحقل - بعضها مثل عائلة *BSD إلى قيمة منخفضة جدًا تبلغ 15 بايت.

عادة ما يكون هذا الحقل غير قابل للتغيير للعمليات في مساحة المستخدم.

proctitle
سطر الأوامر ("العنوان") للعملية. لاحظ - يمكن تعديل هذا بواسطة العملية، لذا لا يضمن أن يكون سطر الأوامر الأصلي.
pid
معرف العملية.
parent
معرف العملية الأب.
pgid
معرف العملية لقائد مجموعة العمليات.
sessid
معرف الجلسة التي تنتمي إليها العملية.
uid
معرف المستخدم الذي تعمل العملية باسمه.
euid
معرف المستخدم الفعلي الذي تعمل العملية باسمه.
gid
معرف المجموعة التي تعمل العملية باسمها.
egid
معرف المجموعة الفعلية التي تعمل العملية باسمها.
context_switches
عدد تبديلات السياق لهذه العملية (طوعية وغير طوعية).
voluntary_context_switches
عدد تبديلات السياق الطوعية لهذه العملية (على سبيل المثال، عن طريق استدعاء sched_yield() أو sleep()).
involuntary_context_switches
عدد تبديلات السياق غير الطوعية لهذه العملية (على سبيل المثال، استنفاد شريحة الوقت أو إرسال إشارة).
proc_size
حجم الذاكرة الافتراضية للعملية بالبايت.
