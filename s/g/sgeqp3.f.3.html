توثيق الوظيفة/الروتين الفرعي

subroutine sgeqp3 (integer m, integer n, real, dimension( lda, * ) a, integer lda, integer, dimension( * ) jpvt, real, dimension( * ) tau, real, dimension( * ) work, integer lwork, integer info)

SGEQP3

الغرض:

!>
!> SGEQP3 يحسب تفكيك QR مع تدوير الأعمدة لمصفوفة A:  A*P = Q*R  باستخدام BLAS المستوى 3.
!>

المعلمات

M 
!>          M هو عدد صحيح (INTEGER)
!>          عدد صفوف المصفوفة A. M >= 0.
!>

N

!>          N هو عدد صحيح (INTEGER)
!>          عدد أعمدة المصفوفة A.  N >= 0.
!>

A

!>          A هو مصفوفة حقيقية (REAL array)، بأبعاد (LDA,N)
!>          عند الإدخال، المصفوفة A ذات الأبعاد M × N.
!>          عند الخروج، يحتوي الجزء العلوي من المصفوفة على المصفوفة شبه المنحرفة العلوية R ذات الأبعاد min(M,N) × N؛ العناصر الموجودة أسفل
!>          القطر، جنبًا إلى جنب مع المصفوفة TAU، تمثل المصفوفة المتعامدة Q كناتج ضرب min(M,N) عاكسات أولية.
!>

LDA

!>          LDA هو عدد صحيح (INTEGER)
!>          البعد الرئيسي للمصفوفة A. LDA >= max(1,M).
!>

JPVT

!>          JPVT هي مصفوفة أعداد صحيحة (INTEGER array)، بأبعاد (N)
!>          عند الإدخال، إذا كان JPVT(J).ne.0، فإن العمود J من A يتم تدويره
!>          إلى مقدمة A*P (عمود رائد)؛ إذا كان JPVT(J)=0،
!>          فإن العمود J من A هو عمود حر.
!>          عند الخروج، إذا كان JPVT(J)=K، فإن العمود J من A*P كان
!>          العمود K من A.
!>

TAU

!>          TAU هي مصفوفة حقيقية (REAL array)، بأبعاد (min(M,N))
!>          العوامل القياسية للعواكس الأولية.
!>

WORK

!>          WORK هي مصفوفة حقيقية (REAL array)، بأبعاد (MAX(1,LWORK))
!>          عند الخروج، إذا كان INFO=0، فإن WORK(1) يعيد قيمة LWORK المثلى.
!>

LWORK

!>          LWORK هو عدد صحيح (INTEGER)
!>          بعد المصفوفة WORK. LWORK >= 3*N+1.
!>          للحصول على أفضل أداء، LWORK >= 2*N+( N+1 )*NB، حيث NB
!>          هو حجم الكتلة الأمثل.
!>
!>          إذا كان LWORK = -1، فسيتم افتراض استعلام مساحة عمل؛ يقوم الروتين
!>          بحساب حجم مصفوفة WORK الأمثل فقط، ويعيد
!>          هذه القيمة كإدخال أول في مصفوفة WORK، ولا يصدر أي رسالة خطأ
!>          متعلقة بـ LWORK بواسطة XERBLA.
!>

INFO

!>          INFO هو عدد صحيح (INTEGER)
!>          = 0: انتهى بنجاح.
!>          < 0: إذا كان INFO = -i، فإن المعلمة i كانت لها قيمة غير قانونية.
!>

تفاصيل إضافية:

!>
!>  يتم تمثيل المصفوفة Q كناتج ضرب عواكس أولية
!>
!>     Q = H(1) H(2) . . . H(k)، حيث k = min(m,n).
!>
!>  كل H(i) لها الشكل
!>
!>     H(i) = I - tau * v * v**T
!>
!>  حيث tau هو ثابت حقيقي، و v هو متجه حقيقي/مركب
!>  مع v(1:i-1) = 0 و v(i) = 1؛ يتم تخزين v(i+1:m) عند الخروج في
!>  A(i+1:m,i)، و tau في TAU(i).
!>
