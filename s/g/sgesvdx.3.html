subroutine sgesvdx (character jobu, character jobvt, character range, integer m, integer n, real, dimension( lda, * ) a, integer lda, real vl, real vu, integer il, integer iu, integer ns, real, dimension( * ) s, real, dimension( ldu, * ) u, integer ldu, real, dimension( ldvt, * ) vt, integer ldvt, real, dimension( * ) work, integer lwork, integer, dimension( * ) iwork, integer info)

SGESVDX يحسب تحليل القيم المفردة (SVD) لمصفوفات GE

الغرض:

!>
!>  SGESVDX يحسب تحليل القيم المفردة (SVD) لمصفوفة A الحقيقية ذات الأبعاد M×N، مع حساب اختياري للمتجهات المفردة اليسرى و/أو اليمنى. يتم كتابة SVD على النحو التالي:
!>
!>      A = U * SIGMA * transpose(V)
!>
!>  حيث SIGMA هي مصفوفة ذات الأبعاد M×N وتكون صفرية باستثناء عناصرها القطرية min(m,n)، و U هي مصفوفة متعامدة ذات الأبعاد M×M، و V هي مصفوفة متعامدة ذات الأبعاد N×N. العناصر القطرية لـ SIGMA هي القيم المفردة لـ A؛ وهي أعداد حقيقية وغير سالبة، ويتم إرجاعها بترتيب تنازلي. الأعمدة min(m,n) الأولى في U و V هي المتجهات المفردة اليسرى واليمنى لـ A.
!>
!>  تستخدم SGESVDX مشكلة القيم الذاتية للحصول على SVD، مما يسمح بحساب مجموعة فرعية من القيم والمتجهات المفردة. انظر SBDSVDX للتفاصيل.
!>
!>  لاحظ أن الروتين يرجع V**T، وليس V.
!>

المعاملات

JOBU 
!>          JOBU هو CHARACTER*1
!>          يحدد خيارات لحساب كل أو جزء من المصفوفة U:
!>          = 'V': يتم إرجاع الأعمدة min(m,n) الأولى من U (المتجهات المفردة اليسرى) أو كما هو محدد بواسطة RANGE في المصفوفة U؛
!>          = 'N': لا يتم حساب أي أعمدة من U (لا توجد متجهات مفردة يسرى).
!>

JOBVT

!>          JOBVT هو CHARACTER*1
!>           يحدد خيارات لحساب كل أو جزء من المصفوفة V**T:
!>           = 'V': يتم إرجاع الصفوف min(m,n) الأولى من V**T (المتجهات المفردة اليمنى) أو كما هو محدد بواسطة RANGE في المصفوفة VT؛
!>           = 'N': لا يتم حساب أي صفوف من V**T (لا توجد متجهات مفردة يمنى).
!>

RANGE

!>          RANGE هو CHARACTER*1
!>          = 'A': سيتم العثور على جميع القيم المفردة.
!>          = 'V': سيتم العثور على جميع القيم المفردة في الفترة نصف المفتوحة (VL,VU].
!>          = 'I': سيتم العثور على القيم المفردة من المرتبة IL إلى IU.
!>

M

!>          M هو INTEGER
!>          عدد صفوف المصفوفة المدخلة A. M >= 0.
!>

N

!>          N هو INTEGER
!>          عدد أعمدة المصفوفة المدخلة A. N >= 0.
!>

A

!>          A هي مصفوفة REAL، أبعادها (LDA,N)
!>          عند الإدخال، المصفوفة A ذات الأبعاد M×N.
!>          عند الخروج، يتم تدمير محتويات A.
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرئيسي للمصفوفة A. LDA >= max(1,M).
!>

VL

!>          VL هو REAL
!>          إذا كان RANGE='V'، الحد الأدنى للفترة التي
!>          سيتم البحث فيها عن القيم المفردة. VU > VL.
!>          لا يتم الرجوع إليها إذا كان RANGE = 'A' أو 'I'.
!>

VU

!>          VU هو REAL
!>          إذا كان RANGE='V'، الحد الأعلى للفترة التي
!>          سيتم البحث فيها عن القيم المفردة. VU > VL.
!>          لا يتم الرجوع إليها إذا كان RANGE = 'A' أو 'I'.
!>

IL

!>          IL هو INTEGER
!>          إذا كان RANGE='I'، فهرس
!>          أصغر قيمة مفردة سيتم إرجاعها.
!>          1 <= IL <= IU <= min(M,N)، إذا كان min(M,N) > 0.
!>          لا يتم الرجوع إليها إذا كان RANGE = 'A' أو 'V'.
!>

IU

!>          IU هو INTEGER
!>          إذا كان RANGE='I'، فهرس
!>          أكبر قيمة مفردة سيتم إرجاعها.
!>          1 <= IL <= IU <= min(M,N)، إذا كان min(M,N) > 0.
!>          لا يتم الرجوع إليها إذا كان RANGE = 'A' أو 'V'.
!>

NS

!>          NS هو INTEGER
!>          العدد الإجمالي للقيم المفردة التي تم العثور عليها،
!>          0 <= NS <= min(M,N).
!>          إذا كان RANGE = 'A'، فإن NS = min(M,N)؛ إذا كان RANGE = 'I'، فإن NS = IU-IL+1.
!>

S

!>          S هي مصفوفة REAL، أبعادها (min(M,N))
!>          القيم المفردة لـ A، مرتبة بحيث S(i) >= S(i+1).
!>

U

!>          U هي مصفوفة REAL، أبعادها (LDU,UCOL)
!>          إذا كان JOBU = 'V'، فإن U تحتوي على أعمدة U (المتجهات المفردة اليسرى،
!>          مخزنة كأعمدة) كما هو محدد بواسطة RANGE؛ إذا كان
!>          JOBU = 'N'، لا يتم الرجوع إلى U.
!>          ملاحظة: يجب على المستخدم التأكد من أن UCOL >= NS؛ إذا كان RANGE = 'V'،
!>          فإن القيمة الدقيقة لـ NS غير معروفة مقدمًا ويجب استخدام حد أعلى.
!>

LDU

!>          LDU هو INTEGER
!>          البعد الرئيسي للمصفوفة U. LDU >= 1؛ إذا كان
!>          JOBU = 'V'، فإن LDU >= M.
!>

VT

!>          VT هي مصفوفة REAL، أبعادها (LDVT,N)
!>          إذا كان JOBVT = 'V'، فإن VT تحتوي على صفوف V**T (المتجهات المفردة اليمنى،
!>          مخزنة كصفوف) كما هو محدد بواسطة RANGE؛ إذا كان JOBVT = 'N'،
!>          لا يتم الرجوع إلى VT.
!>          ملاحظة: يجب على المستخدم التأكد من أن LDVT >= NS؛ إذا كان RANGE = 'V'،
!>          فإن القيمة الدقيقة لـ NS غير معروفة مقدمًا ويجب استخدام حد أعلى.
!>

LDVT

!>          LDVT هو INTEGER
!>          البعد الرئيسي للمصفوفة VT. LDVT >= 1؛ إذا كان
!>          JOBVT = 'V'، فإن LDVT >= NS (انظر أعلاه).
!>

WORK

!>          WORK هي مصفوفة REAL، أبعادها (MAX(1,LWORK))
!>          عند الخروج، إذا كان INFO = 0، فإن WORK(1) يرجع LWORK المثلى؛
!>

LWORK

!>          LWORK هو INTEGER
!>          بعد المصفوفة WORK.
!>          LWORK >= MAX(1,MIN(M,N)*(MIN(M,N)+4)) للمسارات (انظر
!>          التعليقات داخل الكود):
!>             - المسار 1 (M أكبر بكثير من N)
!>             - المسار 1t (N أكبر بكثير من M)
!>          LWORK >= MAX(1,MIN(M,N)*2+MAX(M,N)) للمسارات الأخرى.
!>          للحصول على أداء جيد، يجب أن يكون LWORK أكبر بشكل عام.
!>
!>          إذا كان LWORK = -1، فيفترض استعلام مساحة العمل؛ تقوم الروتين
!>          فقط بحساب الحجم الأمثل لمصفوفة WORK، وترجع
!>          هذه القيمة كالإدخال الأول في مصفوفة WORK، ولا يتم إصدار أي رسالة خطأ
!>          متعلقة بـ LWORK بواسطة XERBLA.
!>

IWORK

!>          IWORK هي مصفوفة INTEGER، أبعادها (12*MIN(M,N))
!>          إذا كان INFO = 0، فإن العناصر الأولى NS في IWORK تكون صفرية. إذا كان INFO > 0،
!>          فإن IWORK تحتوي على فهارس المتجهات الذاتية التي فشلت
!>          في التقارب في SBDSVDX/SSTEVX.
!>

INFO

!>     INFO هو INTEGER
!>           = 0:  انتهاء ناجح
!>           < 0:  إذا كان INFO = -i، فإن المعامل i كان له قيمة غير قانونية
!>           > 0:  إذا كان INFO = i، فإن i من المتجهات الذاتية فشلت في التقارب
!>                 في SBDSVDX/SSTEVX.
!>                 إذا كان INFO = N*2 + 1، حدث خطأ داخلي في
!>                 SBDSVDX
!>
