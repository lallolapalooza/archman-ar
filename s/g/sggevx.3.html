subroutine sggevx (character balanc, character jobvl, character jobvr, character sense, integer n, real, dimension( lda, * ) a, integer lda, real, dimension( ldb, * ) b, integer ldb, real, dimension( * ) alphar, real, dimension( * ) alphai, real, dimension( * ) beta, real, dimension( ldvl, * ) vl, integer ldvl, real, dimension( ldvr, * ) vr, integer ldvr, integer ilo, integer ihi, real, dimension( * ) lscale, real, dimension( * ) rscale, real abnrm, real bbnrm, real, dimension( * ) rconde, real, dimension( * ) rcondv, real, dimension( * ) work, integer lwork, integer, dimension( * ) iwork, logical, dimension( * ) bwork, integer info)

SGGEVX تحسب القيم الذاتية، اختياريًا، المتجهات الذاتية اليسرى و/أو اليمنى لمصفوفات GE

الغرض:

!>
!> SGGEVX تحسب أزواج المصفوفات الحقيقية غير المتناظرة N × N (A,B)
!> القيم الذاتية المعممة، واختياريًا، المتجهات الذاتية المعممة اليسرى و/أو اليمنى.
!>
!> اختياريًا، تحسب أيضًا تحويل موازنة لتحسين
!> شرط القيم الذاتية والمتجهات الذاتية (ILO، IHI،
!> LSCALE، RSCALE، ABNRM، و BBNRM)، الأعداد الشرطية المعكوسة لـ
!> القيم الذاتية (RCONDE)، والأعداد الشرطية المعكوسة لـ
!> المتجهات الذاتية اليمنى (RCONDV).
!>
!> القيمة الذاتية المعممة لزوج المصفوفات (A,B) هي كمية قياسية
!> lambda أو نسبة alpha/beta = lambda، بحيث A - lambda*B تكون
!> شاذة. عادة ما يتم تمثيلها بالزوج (alpha,beta)، حيث
!> يوجد تفسير معقول لـ beta=0، وحتى لكليهما
!> أن يكونا صفرًا.
!>
!> المتجه الذاتي الأيمن v(j) المقابل للقيمة الذاتية lambda(j)
!> لـ (A,B) يحقق
!>
!>                  A * v(j) = lambda(j) * B * v(j) .
!>
!> المتجه الذاتي الأيسر u(j) المقابل للقيمة الذاتية lambda(j)
!> لـ (A,B) يحقق
!>
!>                  u(j)**H * A  = lambda(j) * u(j)**H * B.
!>
!> حيث u(j)**H هو المرافق المترافق لـ u(j).
!>
!>

المعلمات

BALANC 
!>          BALANC هو CHARACTER*1
!>          يحدد خيار الموازنة المراد تنفيذه.
!>          = 'N':  لا تقم بالتحجيم أو التبديل القطري؛
!>          = 'P':  قم بالتبديل فقط؛
!>          = 'S':  قم بالتحجيم فقط؛
!>          = 'B':  قم بالتبديل والتحجيم معًا.
!>          سيتم حساب الأعداد الشرطية المعكوسة للمصفوفات
!>          بعد التبديل و/أو الموازنة. التبديل لا
!>          يغير الأعداد الشرطية (في الحساب الدقيق)، لكن
!>          الموازنة تغيرها.
!>

JOBVL

!>          JOBVL هو CHARACTER*1
!>          = 'N':  لا تحسب المتجهات الذاتية المعممة اليسرى؛
!>          = 'V':  احسب المتجهات الذاتية المعممة اليسرى.
!>

JOBVR

!>          JOBVR هو CHARACTER*1
!>          = 'N':  لا تحسب المتجهات الذاتية المعممة اليمنى؛
!>          = 'V':  احسب المتجهات الذاتية المعممة اليمنى.
!>

SENSE

!>          SENSE هو CHARACTER*1
!>          يحدد أي الأعداد الشرطية المعكوسة سيتم حسابها.
!>          = 'N': لا يتم حساب أي منها؛
!>          = 'E': يتم حسابها للقيم الذاتية فقط؛
!>          = 'V': يتم حسابها للمتجهات الذاتية فقط؛
!>          = 'B': يتم حسابها للقيم الذاتية والمتجهات الذاتية معًا.
!>

N

!>          N هو INTEGER
!>          رتبة المصفوفات A، B، VL، و VR.  N >= 0.
!>

A

!>          A هو مصفوفة REAL، ذات أبعاد (LDA, N)
!>          عند الدخول، المصفوفة A في الزوج (A,B).
!>          عند الخروج، تم استبدال A. إذا كان JOBVL='V' أو JOBVR='V'
!>          أو كلاهما، فإن A تحتوي على الجزء الأول من الصيغة الشوراتية الحقيقية
!>          للتشكيلات المدخلة A و B.
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرئيسي لـ A.  LDA >= max(1,N).
!>

B

!>          B هو مصفوفة REAL، ذات أبعاد (LDB, N)
!>          عند الدخول، المصفوفة B في الزوج (A,B).
!>          عند الخروج، تم استبدال B. إذا كان JOBVL='V' أو JOBVR='V'
!>          أو كلاهما، فإن B تحتوي على الجزء الثاني من الصيغة الشوراتية الحقيقية
!>          للتشكيلات المدخلة A و B.
!>

LDB

!>          LDB هو INTEGER
!>          البعد الرئيسي لـ B.  LDB >= max(1,N).
!>

ALPHAR

!>          ALPHAR هي مصفوفة REAL، ذات أبعاد (N)
!>

ALPHAI

!>          ALPHAI هي مصفوفة REAL، ذات أبعاد (N)
!>

BETA

!>          BETA هي مصفوفة REAL، ذات أبعاد (N)
!>          عند الخروج، (ALPHAR(j) + ALPHAI(j)*i)/BETA(j)، j=1,...,N، ستكون
!>          القيم الذاتية المعممة. إذا كانت ALPHAI(j) صفرًا، فإن
!>          القيمة الذاتية j حقيقية؛ إذا كانت موجبة، فإن القيمتين الذاتيتين j و
!>          (j+1) تشكلان زوجًا مترافقًا مركبًا، مع
!>          ALPHAI(j+1) سالبًا.
!>
!>          ملاحظة: النسب ALPHAR(j)/BETA(j) و ALPHAI(j)/BETA(j)
!>          قد تتجاوز أو تقل عن النطاق بسهولة، وقد تكون BETA(j) صفرًا.
!>          لذلك، يجب على المستخدم تجنب حساب النسبة بشكل ساذج
!>          ALPHA/BETA. ومع ذلك، ستكون ALPHAR و ALPHAI دائمًا أقل
!>          من magnitude(A) وعادة ما تكون قابلة للمقارنة بها،
!>          و BETA دائمًا أقل من magnitude(B) وعادة ما تكون قابلة للمقارنة بها.
!>

VL

!>          VL هي مصفوفة REAL، ذات أبعاد (LDVL,N)
!>          إذا كان JOBVL = 'V'، يتم تخزين المتجهات الذاتية اليسرى u(j) واحدة تلو الأخرى
!>          في أعمدة VL، بنفس ترتيب القيم الذاتية المقابلة لها. إذا كانت القيمة الذاتية j حقيقية،
!>          فإن u(j) = VL(:,j)، العمود j من VL. إذا شكلت القيمتان الذاتيتان j و
!>          (j+1) زوجًا مترافقًا مركبًا، فإن
!>          u(j) = VL(:,j)+i*VL(:,j+1) و u(j+1) = VL(:,j)-i*VL(:,j+1).
!>          سيتم تحجيم كل متجه ذاتي بحيث يكون أكبر مكون له
!>          abs(الجزء الحقيقي) + abs(الجزء التخيلي) = 1.
!>          لا تتم الإشارة إليه إذا كان JOBVL = 'N'.
!>

LDVL

!>          LDVL هو INTEGER
!>          البعد الرئيسي للمصفوفة VL. LDVL >= 1، و
!>          إذا كان JOBVL = 'V'، فإن LDVL >= N.
!>

VR

!>          VR هي مصفوفة REAL، ذات أبعاد (LDVR,N)
!>          إذا كان JOBVR = 'V'، يتم تخزين المتجهات الذاتية اليمنى v(j) واحدة تلو الأخرى
!>          في أعمدة VR، بنفس ترتيب القيم الذاتية المقابلة لها. إذا كانت القيمة الذاتية j حقيقية،
!>          فإن v(j) = VR(:,j)، العمود j من VR. إذا شكلت القيمتان الذاتيتان j و
!>          (j+1) زوجًا مترافقًا مركبًا، فإن
!>          v(j) = VR(:,j)+i*VR(:,j+1) و v(j+1) = VR(:,j)-i*VR(:,j+1).
!>          سيتم تحجيم كل متجه ذاتي بحيث يكون أكبر مكون له
!>          abs(الجزء الحقيقي) + abs(الجزء التخيلي) = 1.
!>          لا تتم الإشارة إليه إذا كان JOBVR = 'N'.
!>

LDVR

!>          LDVR هو INTEGER
!>          البعد الرئيسي للمصفوفة VR. LDVR >= 1، و
!>          إذا كان JOBVR = 'V'، فإن LDVR >= N.
!>
