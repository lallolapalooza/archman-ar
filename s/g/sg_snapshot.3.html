#include <statgrab.h>

(void);

الوصف

sg_init() إما تهيئة مكتبة statgrab وتعيين عداد التهيئة إلى 1، إذا لم يتم ذلك بالفعل، أو زيادة عداد التهيئة. إذا تم استدعاؤها للمرة الأولى (في الواقع: إذا تم استدعاؤها عندما يكون عداد التهيئة 0)، يتم استدعاء الدالة المنشئة (روتين التهيئة) لكل مكون مسجل، ويتم تهيئة بعض المتغيرات العامة، ويتم تهيئة أقفال التبادل الحصري المطلوبة وما إلى ذلك. sg_init() محمية بالقفص المسمى "statgrab".

sg_log_init() تسمح ببعض تكوينات التسجيل الخاصة بالتطبيق. وهي مخصصة للمشاريع التي تحتوي على عدة تطبيقات أو أوامر قد تكون لها ملحقات مختلفة. لاحظ أنه يجب استدعاء sg_log_init قبل sg_init، ولكن بعد أن يقوم تطبيقك بتهيئة إطار عمل التسجيل.

sg_snapshot() خاص بنظام Win32 ومن المحتمل اختفائه.

sg_shutdown() تقلل عداد التهيئة. إذا وصل العداد إلى الصفر، يتم استدعاء الدوال المدمرة للمكونات، وتدمير أقفال التبادل الحصري (باستثناء "statgrab") وما إلى ذلك. sg_shutdown() محمية بالقفص المسمى "statgrab".

sg_drop_privileges() تسقط الامتيازات المرتفعة.

قيم الإرجاع

جميع الدوال ترجع رمز خطأ statgrab. إما SG_ERROR_NONE عندما كان كل شيء على ما يرام أو رمز الخطأ المناسب من دالة منشئة/مدمرة.

مثال

تسلسل التهيئة/إلغاء التهيئة النموذجي عند الاستخدام مع log4cplus:

static void *l4cplus_initializer;
static void
cleanup_logging(void)
{
       log4cplus_deinitialize(l4cplus_initializer);
}
int
main(int argc, char const *argv[])
{
    l4cplus_initializer = log4cplus_initialize();
    atexit((void (*)(void))cleanup_logging);
    sg_log_init("saidar", "SAIDAR_LOG_PROPERTIES", argv[0]);
    sg_init(1);
    if(sg_drop_privileges() != 0) {
	die("Failed to drop setuid/setgid privileges");
    }
    do_something();
    sg_shutdown();
    return 0;
}
