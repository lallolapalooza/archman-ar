subroutine sgges (character jobvsl, character jobvsr, character sort, external selctg, integer n, real, dimension( lda, * ) a, integer lda, real, dimension( ldb, * ) b, integer ldb, integer sdim, real, dimension( * ) alphar, real, dimension( * ) alphai, real, dimension( * ) beta, real, dimension( ldvsl, * ) vsl, integer ldvsl, real, dimension( ldvsr, * ) vsr, integer ldvsr, real, dimension( * ) work, integer lwork, logical, dimension( * ) bwork, integer info)

SGGES تحسب القيم الذاتية، وشكل شتور، واختيارياً، مصفوفة متجهات شتور لمصفوفات GE

الغرض:

!>
!> SGGES تحسب للزوج من المصفوفات الحقيقية غير المتناظرة N-by-N (A,B)،
!> القيم الذاتية المعممة، شكل شتور المعممة (S,T)،
!> اختياريًا، مصفوفات المتجهات اليسرى و/أو اليمنى لشتور (VSL و
!> VSR). هذا يعطي تحلل شتور المعممة
!>
!>          (A,B) = ( (VSL)*S*(VSR)**T, (VSL)*T*(VSR)**T )
!>
!> اختياريًا، تقوم أيضًا بترتيب القيم الذاتية بحيث تظهر مجموعة
!> مختارة من القيم الذاتية في الكتل القطرية الأمامية للمصفوفة العليا
!> شبه المثلثية S والمصفوفة العليا المثلثية T. الأعمدة الأمامية
!> لـ VSL و VSR تشكل بعد ذلك قاعدة متعامدة لـ
!> مساحات المتجهات الذاتية اليسرى واليمنى المقابلة (مساحات الانحدار).
!>
!> (إذا كانت القيم الذاتية المعممة مطلوبة فقط، استخدم المشغل
!> SGGEV بدلاً من ذلك، فهو أسرع.)
!>
!> القيمة الذاتية المعممة لزوج من المصفوفات (A,B) هي قيمة قياسية w
!> أو نسبة alpha/beta = w، بحيث تكون A - w*B مفردة. يتم
!> تمثيلها عادةً كزوج (alpha,beta)، حيث يوجد تفسير معقول لـ beta=0
!> أو كلاهما صفر.
!>
!> زوج المصفوفات (S,T) يكون في شكل شتور المعممة الحقيقي إذا كانت T
!> عليا مثلثية بقطر موجب غير سالب و S تكون كتلة عليا مثلثية
!> بكتل 1x1 و 2x2. الكتل 1x1 تقابل
!> قيم ذاتية معممة حقيقية، بينما الكتل 2x2 لـ S ستكون
!>
!>          [  a  0  ]
!>          [  0  b  ]
!>
!> وزوج الكتل 2x2 المقابلة في S و T سيكون له زوج
!> مرافق معقد من القيم الذاتية المعممة.
!>
!>

المعلمات

JOBVSL 
!>          JOBVSL هو CHARACTER*1
!>          = 'N':  لا تحسب متجهات شتور اليسرى؛
!>          = 'V':  احسب متجهات شتور اليسرى.
!>

JOBVSR

!>          JOBVSR هو CHARACTER*1
!>          = 'N':  لا تحسب متجهات شتور اليمنى؛
!>          = 'V':  احسب متجهات شتور اليمنى.
!>

SORT

!>          SORT هو CHARACTER*1
!>          يحدد ما إذا كان سيتم ترتيب القيم الذاتية على
!>          القطر لشكل شتور المعممة أم لا.
!>          = 'N':  القيم الذاتية غير مرتبة؛
!>          = 'S':  القيم الذاتية مرتبة (انظر SELCTG)؛
!>

SELCTG

!>          SELCTG هي وظيفة LOGICAL بثلاث وسائط REAL
!>          يجب تعريف SELCTG بشكل خارجي في الإجراء الفرعي المستدعي.
!>          إذا كان SORT = 'N'، فلن يتم استخدام SELCTG.
!>          إذا كان SORT = 'S'، يتم استخدام SELCTG لتحديد القيم الذاتية لفرزها
!>          إلى أعلى اليسار من شكل شتور.
!>          يتم تحديد القيمة الذاتية (ALPHAR(j)+ALPHAI(j))/BETA(j) إذا
!>          كان SELCTG(ALPHAR(j),ALPHAI(j),BETA(j)) صحيحًا؛ أي إذا كان
!>          واحد من زوج من القيم الذاتية المترافقة المعقدة يتم تحديده،
!>          فإن كلتا القيمتين الذاتيتين المعقدتين يتم تحديدهما.
!>
!>          لاحظ أنه في الحالة سيئة التكييف، قد لا تستوفي القيمة الذاتية المحددة
!>          SELCTG(ALPHAR(j),ALPHAI(j),
!>          BETA(j)) = .TRUE. بعد الترتيب. يجب تعيين INFO إلى N+2
!>          في هذه الحالة.
!>

N

!>          N هو INTEGER
!>          ترتيب المصفوفات A، B، VSL، و VSR. N >= 0.
!>

A

!>          A هي مصفوفة REAL، ذات البعد (LDA, N)
!>          عند الدخول، الجزء الأول من زوج المصفوفات.
!>          عند الخروج، تم استبدال A بشكل شتور المعممة.
!>          S.
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرائد لـ A. LDA >= max(1,N).
!>

B

!>          B هي مصفوفة REAL، ذات البعد (LDB, N)
!>          عند الدخول، الجزء الثاني من زوج المصفوفات.
!>          عند الخروج، تم استبدال B بشكل شتور المعممة.
!>          T.
!>

LDB

!>          LDB هو INTEGER
!>          البعد الرائد لـ B. LDB >= max(1,N).
!>

SDIM

!>          SDIM هو INTEGER
!>          إذا كان SORT = 'N'، فإن SDIM = 0.
!>          إذا كان SORT = 'S'، فإن SDIM = عدد القيم الذاتية (بعد الترتيب)
!>          التي يكون SELCTG صحيحًا لها. (الأزواج المترافقة المعقدة التي
!>          يكون SELCTG صحيحًا لأي من القيمتين الذاتيتين تُحسب على أنها 2.)
!>

ALPHAR

!>          ALPHAR هي مصفوفة REAL، ذات البعد (N)
!>

ALPHAI

!>          ALPHAI هي مصفوفة REAL، ذات البعد (N)
!>

BETA

!>          BETA هي مصفوفة REAL، ذات البعد (N)
!>          عند الخروج، (ALPHAR(j) + ALPHAI(j)*i)/BETA(j)، j=1,...,N، ستكون
!>          القيم الذاتية المعممة. ALPHAR(j) + ALPHAI(j)*i،
!>          و BETA(j)، j=1,...,N هي أقطار شكل شتور المعقد
!>          الذي سينتج إذا تم اختزال الكتل القطرية 2x2 لـ
!>          شكل شتور الحقيقي لـ (A,B) إلى
!>          شكل مثلثي باستخدام تحويلات الوحدة المعقدة 2x2.
!>          إذا كان ALPHAI(j) صفرًا، فإن القيمة الذاتية j حقيقية؛ إذا
!>          كانت موجبة، فإن القيمتين الذاتيتين j و (j+1) هما زوج
!>          مترافق معقد، مع ALPHAI(j+1) سالب.
!>
!>          ملاحظة: قد تتجاوز النسب ALPHAR(j)/BETA(j) و ALPHAI(j)/BETA(j)
!>          أو تقل بسهولة، وقد تكون BETA(j) حتى صفرًا.
!>          لذلك، يجب على المستخدم تجنب حساب النسبة بشكل ساذج.
!>          ومع ذلك، سيكون ALPHAR و ALPHAI دائمًا أقل من، وعادة ما تكون قابلة للمقارنة مع،
!>          معيار (A) في القيمة المطلقة، و BETA دائمًا أقل من، وعادة ما تكون قابلة للمقارنة مع،
!>          معيار (B).
!>

VSL

!>          VSL هي مصفوفة REAL، ذات البعد (LDVSL,N)
!>          إذا كان JOBVSL = 'V'، ستحتوي VSL على متجهات شتور اليسرى.
!>          لا يتم استخدامها إذا كان JOBVSL = 'N'.
!>

LDVSL

!>          LDVSL هو INTEGER
!>          البعد الرائد للمصفوفة VSL. LDVSL >=1، و
!>          إذا كان JOBVSL = 'V'، فإن LDVSL >= N.
!>

VSR

!>          VSR هي مصفوفة REAL، ذات البعد (LDVSR,N)
!>          إذا كان JOBVSR = 'V'، ستحتوي VSR على متجهات شتور اليمنى.
!>          لا يتم استخدامها إذا كان JOBVSR = 'N'.
!>

LDVSR

!>          LDVSR هو INTEGER
!>          البعد الرائد للمصفوفة VSR. LDVSR >= 1، و
!>          إذا كان JOBVSR = 'V'، فإن LDVSR >= N.
!>

WORK

!>          WORK هي مصفوفة REAL، ذات البعد (MAX(1,LWORK))
!>          عند الخروج، إذا كان INFO = 0، فإن WORK(1) يعيد LWORK المثلى.
!>

LWORK

!>          LWORK هو INTEGER
!>          بعد المصفوفة WORK.
!>          إذا كان N = 0، فإن LWORK >= 1، وإلا فإن LWORK >= max(8*N,6*N+16).
!>

NAME

sgges - Compute the generalized eigenvalues and eigenvectors for a pair of general matrices

SYNOPSIS

  SUBROUTINE SGGEES(
     CHARACTER*1        JOBVL, JOBVR, SENSE, N, A, LDA, B, LDB,
     INTEGER            ALPHA, BETA, VL, LDVL, VR, LDVR, WORK,
     LWORK, BWORK, INFO
  )

Parameters:

JOBVL
CHARACTER*1.

Default: 'V'

If JOBVL = 'V', compute the left Schur vectors of the generalized Schur form.
If JOBVL = 'N', do not compute the left Schur vectors.

JOBVR
CHARACTER*1.

Default: 'V'

If JOBVR = 'V', compute the right Schur vectors of the generalized Schur form.
If JOBVR = 'N', do not compute the right Schur vectors.

SENSE
CHARACTER*1.

Default: 'N'

Determines which eigenvectors are computed.
If SENSE = 'E', compute the eigenvectors of selected eigenvalues.
If SENSE = 'V', compute the eigenvectors of selected eigenvalues.
If SENSE = 'B', compute the eigenvectors of selected eigenvalues.
If SENSE = 'N', none of the eigenvectors of the selected eigenvalues are computed.

N
INTEGER.

The order of the matrices A and B. N >= 0.

A
REAL array, dimension (LDA, N).

On entry, the matrix A.

On exit, A is overwritten by its generalized Schur form.

LDA
INTEGER.

Leading dimension of the array A. LDA >= max(1, N).

B
REAL array, dimension (LDB, N).

On entry, the matrix B.

On exit, B is overwritten by its generalized Schur form.

LDB
INTEGER.

Leading dimension of the array B. LDB >= max(1, N).

ALPHA
INTEGER array, dimension (N).

On exit, ALPHA(i) and BETA(i) contain the parameters of the generalized eigenvalue Ai/Bi for i=1,...,N.

ALPHA contains the real part of the eigenvalues.

BETA
INTEGER array, dimension (N).

On exit, ALPHA(i) and BETA(i) contain the parameters of the generalized eigenvalue Ai/Bi for i=1,...,N.

BETA contains the imaginary part of the eigenvalues.

VL
REAL array, dimension (LDVL, N) if JOBVL = 'V', and 0 otherwise.

If JOBVL = 'V', VL contains the left Schur vectors of the generalized Schur form.
If JOBVL = 'N', VL is not referenced.

LDVL
INTEGER.

Leading dimension of the array VL. LDVL >= 1, and if JOBVL = 'V', LDVL >= N.

VR
REAL array, dimension (LDVR, N) if JOBVR = 'V', and 0 otherwise.

If JOBVR = 'V', VR contains the right Schur vectors of the generalized Schur form.
If JOBVR = 'N', VR is not referenced.

LDVR
INTEGER.

Leading dimension of the array VR. LDVR >= 1, and if JOBVR = 'V', LDVR >= N.

WORK
REAL array, dimension (LWORK).

Workspace on entry. On exit, WORK(1) contains the optimal LWORK.

LWORK
INTEGER.

Length of the array WORK. LWORK >= max(1, 2*N).
For good performance, LWORK must generally be larger.

If LWORK = -1, then a workspace query is assumed; the routine only calculates the optimal size of the WORK array, returns this value as the first entry of the WORK array, and no error message related to LWORK is issued by XERBLA.

BWORK
LOGICAL array, dimension (N).

Not referenced if SORT = 'N'.

INFO
INTEGER.

= 0: successful exit.
< 0: if INFO = -i, the i-th argument had an illegal value.
= 1,...,N:
The QZ iteration failed. (A,B) are not in Schur form, but ALPHAR(j), ALPHAI(j), and BETA(j) should be correct for j=INFO+1,...,N.
> N: =N+1: other than QZ iteration failed in SHGEQZ.
=N+2: after reordering, roundoff changed values of some complex eigenvalues so that leading eigenvalues in the Generalized Schur form no longer satisfy SELCTG=.TRUE. This could also be caused due to scaling.
=N+3: reordering failed in STGSEN.
