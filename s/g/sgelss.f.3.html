توثيق الوظيفة/الروتين الفرعي

subroutine sgelss (integer m, integer n, integer nrhs, real, dimension( lda, * ) a, integer lda, real, dimension( ldb, * ) b, integer ldb, real, dimension( * ) s, real rcond, integer rank, real, dimension( * ) work, integer lwork, integer info)

SGELSS يحل أنظمة مصفوفات GE الزائدة أو الناقصة

الغرض:

!>
!> SGELSS يحسب الحل بأقل معيار لمشكلة المربعات الصغرى الخطية الحقيقية:
!>
!> تصغير المعيار 2 لـ(| b - A*x |).
!>
!> باستخدام تحليل القيم المنفردة (SVD) للمصفوفة A. A هي مصفوفة M-by-N
!> قد تكون ذات رتبة ناقصة.
!>
!> يمكن التعامل مع عدة متجهات طرف أيمن b ومتجهات حل x في استدعاء واحد؛
!> يتم تخزينها كأعمدة في المصفوفة B ذات الطرف الأيمن M-by-NRHS ومصفوفة الحل
!> N-by-NRHS X.
!>
!> يتم تحديد الرتبة الفعالة للمصفوفة A عن طريق معاملة القيم المنفردة
!> التي تقل عن RCOND مضروبة في أكبر قيمة منفردة كصفر.
!>

المعلمات

M 
!>          M هو عدد صحيح (INTEGER)
!>          عدد الصفوف في المصفوفة A. M >= 0.
!>

N

!>          N هو عدد صحيح (INTEGER)
!>          عدد الأعمدة في المصفوفة A. N >= 0.
!>

NRHS

!>          NRHS هو عدد صحيح (INTEGER)
!>          عدد الأطراف اليمنى، أي عدد الأعمدة
!>          للمصفوفتين B و X. NRHS >= 0.
!>

A

!>          A هي مصفوفة حقيقية (REAL array)، ذات بُعد (LDA,N)
!>          عند الدخول، يتم استبدال الصفوف min(m,n) الأولى من A
!>          بمتجهاتها المنفردة اليمنى، مخزنة صفًا بصف.
!>

LDA

!>          LDA هو عدد صحيح (INTEGER)
!>          البعد الرائد للمصفوفة A. LDA >= max(1,M).
!>

B

!>          B هي مصفوفة حقيقية (REAL array)، ذات بُعد (LDB,NRHS)
!>          عند الدخول، المصفوفة B ذات الطرف الأيمن M-by-NRHS.
!>          عند الخروج، يتم استبدال B بمصفوفة الحل X ذات البعد N-by-NRHS.
!>          إذا كانت m >= n و RANK = n، فإن مجموع مربعات البواقي
!>          للحل في العمود i يُعطى بمجموع مربعات العناصر n+1:m في ذلك العمود.
!>

LDB

!>          LDB هو عدد صحيح (INTEGER)
!>          البعد الرائد للمصفوفة B. LDB >= max(1,max(M,N)).
!>

S

!>          S هي مصفوفة حقيقية (REAL array)، ذات بُعد (min(M,N))
!>          القيم المنفردة للمصفوفة A بترتيب تنازلي.
!>          رقم شرط A في المعيار 2 = S(1)/S(min(m,n)).
!>

RCOND

!>          RCOND هو قيمة حقيقية (REAL)
!>          تُستخدم RCOND لتحديد الرتبة الفعالة للمصفوفة A.
!>          تُعامل القيم المنفردة S(i) <= RCOND*S(1) على أنها صفر.
!>          إذا كانت RCOND < 0، يتم استخدام دقة الآلة بدلاً من ذلك.
!>

RANK

!>          RANK هو عدد صحيح (INTEGER)
!>          الرتبة الفعالة للمصفوفة A، أي عدد القيم المنفردة
!>          التي هي أكبر من RCOND*S(1).
!>

WORK

!>          WORK هي مصفوفة حقيقية (REAL array)، ذات بُعد (MAX(1,LWORK))
!>          عند الخروج، إذا كان INFO = 0، فإن WORK(1) يعيد LWORK الأمثل.
!>

LWORK

!>          LWORK هو عدد صحيح (INTEGER)
!>          بعد المصفوفة WORK. LWORK >= 1، وأيضًا:
!>          LWORK >= 3*min(M,N) + max( 2*min(M,N), max(M,N), NRHS )
!>          لأداء جيد، يجب أن تكون LWORK أكبر بشكل عام.
!>
!>          إذا كانت LWORK = -1، فسيتم افتراض استعلام مساحة العمل؛
!>          يحسب الروتين فقط الحجم الأمثل للمصفوفة WORK، ويعيد
!>          هذه القيمة كالإدخال الأول للمصفوفة WORK، ولا يتم إصدار أي رسالة خطأ
!>          متعلقة بـ LWORK بواسطة XERBLA.
!>

INFO

!>          INFO هو عدد صحيح (INTEGER)
!>          = 0: انتهاء ناجح
!>          < 0: إذا كان INFO = -i، فإن المعامل i كان لديه قيمة غير قانونية.
!>          > 0: فشل الخوارزمية لحساب SVD في التقارب؛
!>                إذا كان INFO = i، فإن i عنصرًا خارج القطر لمصفوفة ثنائية
!>                وسيطة لم تتقارب إلى الصفر.
!>
