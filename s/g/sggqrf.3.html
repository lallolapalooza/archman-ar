توثيق الوظيفة/الروتين الفرعي

subroutine sggqrf (integer n, integer m, integer p, real, dimension( lda, * ) a, integer lda, real, dimension( * ) taua, real, dimension( ldb, * ) b, integer ldb, real, dimension( * ) taub, real, dimension( * ) work, integer lwork, integer info)

SGGQRF

الغرض:

!>
!> SGGQRF يحسب عامل QR المعمم لمصفوفة N-by-M A ومصفوفة N-by-P B:
!>
!>             A = Q*R,        B = Q*T*Z,
!>
!> حيث Q هي مصفوفة متعامدة N-by-N، Z هي مصفوفة متعامدة P-by-P، و R و T تأخذ أحد الأشكال:
!>
!> إذا كان N >= M،  R = ( R11 ) M  ,   أو إذا كان N < M،  R = ( R11  R12 ) N,
!>                 (  0  ) N-M                         N   M-N
!>                    M
!>
!> حيث R11 هي مصفوفة مثلثية عليا، و
!>
!> إذا كان N <= P،  T = ( 0  T12 ) N,   أو إذا كان N > P،  T = ( T11 ) N-P,
!>                  P-N  N                           ( T21 ) P
!>                                                      P
!>
!> حيث T12 أو T21 هي مصفوفة مثلثية عليا.
!>
!> وبشكل خاص، إذا كانت B مربعة وغير مفردة، فإن عامل GQR للمصفوفة A و B يعطي ضمنيًا عامل QR للمصفوفة inv(B)*A:
!>
!>              inv(B)*A = Z**T*(inv(T)*R)
!>
!> حيث يمثل inv(B) معكوس المصفوفة B، ويمثل Z**T منقول المصفوفة Z.
!>

المعلمات

N 
!>          N هو عدد صحيح (INTEGER)
!>          عدد صفوف المصفوفات A و B. N >= 0.
!>

M

!>          M هو عدد صحيح (INTEGER)
!>          عدد أعمدة المصفوفة A.  M >= 0.
!>

P

!>          P هو عدد صحيح (INTEGER)
!>          عدد أعمدة المصفوفة B.  P >= 0.
!>

A

!>          A هي مصفوفة من النوع REAL، البعد (LDA,M)
!>          عند الدخول، تحتوي المصفوفة A ذات الأبعاد N-by-M.
!>          عند الخروج، تحتوي العناصر على القطر وفوقه في المصفوفة على مصفوفة شبه منحرفة عليا R ذات الأبعاد min(N,M)-by-M (R هي مصفوفة مثلثية عليا إذا كان N >= M)؛ العناصر الموجودة تحت القطر، مع المصفوفة TAUA، تمثل المصفوفة المتعامدة Q كحاصل ضرب min(N,M) من الانعكاسات الأولية (انظر التفاصيل الإضافية).
!>

LDA

!>          LDA هو عدد صحيح (INTEGER)
!>          البعد الرائد للمصفوفة A. LDA >= max(1,N).
!>

TAUA

!>          TAUA هي مصفوفة من النوع REAL، البعد (min(N,M))
!>          العوامل القياسية للانعكاسات الأولية التي تمثل المصفوفة المتعامدة Q (انظر التفاصيل الإضافية).
!>

B

!>          B هي مصفوفة من النوع REAL، البعد (LDB,P)
!>          عند الدخول، المصفوفة B ذات الأبعاد N-by-P.
!>          عند الخروج، إذا كان N <= P، فإن الجزء المثلثي العلوي من المصفوفة الفرعية B(1:N,P-N+1:P) يحتوي على المصفوفة المثلثية العليا T ذات الأبعاد N-by-N؛ إذا كان N > P، فإن العناصر على وتحت القطر الفرعي (N-P) تحتوي على المصفوفة شبه المنحرفة العليا T ذات الأبعاد N-by-P؛ العناصر المتبقية، مع المصفوفة TAUB، تمثل المصفوفة المتعامدة Z كحاصل ضرب انعكاسات أولية (انظر التفاصيل الإضافية).
!>

LDB

!>          LDB هو عدد صحيح (INTEGER)
!>          البعد الرائد للمصفوفة B. LDB >= max(1,N).
!>

TAUB

!>          TAUB هي مصفوفة من النوع REAL، البعد (min(N,P))
!>          العوامل القياسية للانعكاسات الأولية التي تمثل المصفوفة المتعامدة Z (انظر التفاصيل الإضافية).
!>

WORK

!>          WORK هي مصفوفة من النوع REAL، البعد (MAX(1,LWORK))
!>          عند الخروج، إذا كان INFO = 0، فإن WORK(1) يعيد قيمة LWORK المثلى.
!>

LWORK

!>          LWORK هو عدد صحيح (INTEGER)
!>          بعد المصفوفة WORK. LWORK >= max(1,N,M,P).
!>          للحصول على الأداء الأمثل LWORK >= max(N,M,P)*max(NB1,NB2,NB3)،
!>          حيث NB1 هو حجم الكتلة الأمثل لعامل QR لمصفوفة N-by-M، NB2 هو حجم الكتلة الأمثل لعامل RQ لمصفوفة N-by-P، و NB3 هو حجم الكتلة الأمثل لاستدعاء SORMQR.
!>
!>          إذا كان LWORK = -1، فسيتم افتراض استعلام عن مساحة العمل؛ سيقوم الروتين فقط بحساب الحجم الأمثل للمصفوفة WORK، ويعيد هذا الحجم كالإدخال الأول للمصفوفة WORK، ولن يتم إصدار رسالة خطأ متعلقة بـ LWORK بواسطة XERBLA.
!>

INFO

!>          INFO هو عدد صحيح (INTEGER)
!>          = 0:  انتهاء ناجح
!>          < 0:  إذا كان INFO = -i، فإن الوسيط i كان له قيمة غير قانونية.
!>

التفاصيل الإضافية:

!>
!>  يتم تمثيل المصفوفة Q كحاصل ضرب انعكاسات أولية
!>
!>     Q = H(1) H(2) . . . H(k)، حيث k = min(n,m).
!>
!>  كل H(i) تأخذ الشكل
!>
!>     H(i) = I - taua * v * v**T
!>
!>  حيث taua هو عامل قياسي حقيقي، و v هو متجه حقيقي ذو
!>  v(1:i-1) = 0 و v(i) = 1؛ يتم تخزين v(i+1:n) عند الخروج في A(i+1:n,i)،
!>  ويتم تخزين taua في TAUA(i).
!>  لتشكيل Q بشكل صريح، استخدم الروتين الفرعي SORGQR من LAPACK.
!>  لاستخدام Q لتحديث مصفوفة أخرى، استخدم الروتين الفرعي SORMQR من LAPACK.
!>
!>  يتم تمثيل المصفوفة Z كحاصل ضرب انعكاسات أولية
!>
!>     Z = H(1) H(2) . . . H(k)، حيث k = min(n,p).
!>
!>  كل H(i) تأخذ الشكل
!>
!>     H(i) = I - taub * v * v**T
!>
!>  حيث taub هو عامل قياسي حقيقي، و v هو متجه حقيقي ذو
!>  v(p-k+i+1:p) = 0 و v(p-k+i) = 1؛ يتم تخزين v(1:p-k+i-1) عند الخروج في
!>  B(n-k+i,1:p-k+i-1)، ويتم تخزين taub في TAUB(i).
!>  لتشكيل Z بشكل صريح، استخدم الروتين الفرعي SORGRQ من LAPACK.
!>  لاستخدام Z لتحديث مصفوفة أخرى، استخدم الروتين الفرعي SORMRQ من LAPACK.
!>
