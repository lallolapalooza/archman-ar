subroutine sgbrfs (character trans, integer n, integer kl, integer ku, integer nrhs, real, dimension( ldab, * ) ab, integer ldab, real, dimension( ldafb, * ) afb, integer ldafb, integer, dimension( * ) ipiv, real, dimension( ldb, * ) b, integer ldb, real, dimension( ldx, * ) x, integer ldx, real, dimension( * ) ferr, real, dimension( * ) berr, real, dimension( * ) work, integer, dimension( * ) iwork, integer info)

SGBRFS

الغرض:

!>
!> SGBRFS يحسن الحل المحسوب لنظام المعادلات الخطية
!> عندما تكون مصفوفة المعاملات شريطية، ويوفر
!> حدود الخطأ وتقديرات الخطأ الخلفي للحل.
!>

المعلمات

TRANS 
!>          TRANS is CHARACTER*1
!>          يحدد شكل نظام المعادلات:
!>          = 'N':  A * X = B     (بدون تبديل)
!>          = 'T':  A**T * X = B  (تبديل)
!>          = 'C':  A**H * X = B  (تبديل مترافق = تبديل)
!>

N

!>          N is INTEGER
!>          رتبة المصفوفة A.  N >= 0.
!>

KL

!>          KL is INTEGER
!>          عدد الأقطار الفرعية داخل الشريط A.  KL >= 0.
!>

KU

!>          KU is INTEGER
!>          عدد الأقطار الفائقة داخل الشريط A.  KU >= 0.
!>

NRHS

!>          NRHS is INTEGER
!>          عدد الجوانب اليمنى، أي عدد أعمدة
!>          المصفوفتين B و X.  NRHS >= 0.
!>

AB

!>          AB is REAL array, dimension (LDAB,N)
!>          مصفوفة الشريط الأصلية A، مخزنة في الصفوف من 1 إلى KL+KU+1.
!>          يتم تخزين العمود j للمصفوفة A في العمود j للمصفوفة
!>          AB كما يلي:
!>          AB(ku+1+i-j,j) = A(i,j) for max(1,j-ku)<=i<=min(n,j+kl).
!>

LDAB

!>          LDAB is INTEGER
!>          البعد الرائد للمصفوفة AB.  LDAB >= KL+KU+1.
!>

AFB

!>          AFB is REAL array, dimension (LDAFB,N)
!>          تفاصيل التحليل LU للمصفوفة الشريطية A، كما
!>          تم حسابها بواسطة SGBTRF. يتم تخزين U كمصفوفة شريطية مثلثة عليا
!>          مع KL+KU أقطار فائقة في الصفوف من 1 إلى KL+KU+1، و
!>          يتم تخزين المضاعفات المستخدمة أثناء التحليل في
!>          الصفوف من KL+KU+2 إلى 2*KL+KU+1.
!>

LDAFB

!>          LDAFB is INTEGER
!>          البعد الرائد للمصفوفة AFB.  LDAFB >= 2*KL*KU+1.
!>

IPIV

!>          IPIV is INTEGER array, dimension (N)
!>          مؤشرات المحور من SGBTRF؛ من أجل 1<=i<=N، تم تبديل الصف i للمصفوفة
!>          مع الصف IPIV(i).
!>

B

!>          B is REAL array, dimension (LDB,NRHS)
!>          مصفوفة الجانب الأيمن B.
!>

LDB

!>          LDB is INTEGER
!>          البعد الرائد للمصفوفة B.  LDB >= max(1,N).
!>

X

!>          X is REAL array, dimension (LDX,NRHS)
!>          عند الإدخال، مصفوفة الحل X، كما تم حسابها بواسطة SGBTRS.
!>          عند الخروج، مصفوفة الحل المحسنة X.
!>

LDX

!>          LDX is INTEGER
!>          البعد الرائد للمصفوفة X.  LDX >= max(1,N).
!>

FERR

!>          FERR is REAL array, dimension (NRHS)
!>          تقدير حد الخطأ الأمامي لكل متجه حل
!>          X(j) (العمود j للمصفوفة الحل X).
!>          إذا كانت XTRUE هي الحل الحقيقي المقابل لـ X(j)، فإن FERR(j)
!>          هو حد أعلى مقدر لمقدار أكبر
!>          عنصر في (X(j) - XTRUE) مقسومًا على مقدار أكبر
!>          عنصر في X(j). التقدير موثوق به مثل
!>          التقدير لـ RCOND، وهو دائمًا تقريبًا زيادة طفيفة
!>          للخطأ الحقيقي.
!>

BERR

!>          BERR is REAL array, dimension (NRHS)
!>          خطأ الخلفية النسبي المكوني لكل متجه حل
!>          X(j) (أي، أصغر تغيير نسبي في
!>          أي عنصر في A أو B يجعل X(j) حلاً دقيقًا).
!>

WORK

!>          WORK is REAL array, dimension (3*N)
!>

IWORK

!>          IWORK is INTEGER array, dimension (N)
!>

INFO

!>          INFO is INTEGER
!>          = 0:  نجاح الخروج
!>          < 0:  إذا كان INFO = -i، فإن المعلمة i كانت قيمة غير قانونية
!>

المعلمات الداخلية:

!>  ITMAX هو الحد الأقصى لعدد خطوات التحسين التكراري.
!>
