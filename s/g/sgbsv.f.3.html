وثائق الوظيفة/الروتين الفرعي

subroutine sgbsv (integer n, integer kl, integer ku, integer nrhs, real, dimension( ldab, * ) ab, integer ldab, integer, dimension( * ) ipiv, real, dimension( ldb, * ) b, integer ldb, integer info)

SGBSV تحسب الحل لنظام المعادلات الخطية A * X = B لمصفوفات GB (مشغل بسيط)

الغرض:

!>
!> SGBSV تحسب الحل لنظام حقيقي من المعادلات الخطية
!> A * X = B، حيث A مصفوفة شريطية من الرتبة N مع KL تحت أقطار
!> و KU فوق أقطار، و X و B مصفوفتان N × NRHS.
!>
!> يستخدم تفكيك LU مع التنبييه الجزئي وتبادلات الصفوف
!> لتحليل A إلى A = L * U، حيث L هو حاصل ضرب تبادلات
!> ومصفوفات مثلثية سفلية موحدة مع KL تحت أقطار، و U
!> مثلثية علوية مع KL+KU فوق أقطار. ثم يستخدم الشكل المحلل لـ A
!> لحل نظام المعادلات A * X = B.
!>

المعاملات

N 
!>          N هو صحيح (INTEGER)
!>          عدد المعادلات الخطية، أي رتبة المصفوفة A. N >= 0.
!>

KL

!>          KL هو صحيح (INTEGER)
!>          عدد الأقطار السفلية ضمن شريط A. KL >= 0.
!>

KU

!>          KU هو صحيح (INTEGER)
!>          عدد الأقطار العلوية ضمن شريط A. KU >= 0.
!>

NRHS

!>          NRHS هو صحيح (INTEGER)
!>          عدد الجوانب اليمنى، أي عدد أعمدة المصفوفة B. NRHS >= 0.
!>

AB

!>          AB هو مصفوفة حقيقية (REAL array)، أبعادها (LDAB,N)
!>          عند الإدخال، المصفوفة A بتخزين الشريط، في الصفوف KL+1 إلى
!>          2*KL+KU+1؛ لا يلزم تعيين الصفوف 1 إلى KL من المصفوفة.
!>          يتم تخزين العمود j من A في العمود j من
!>          المصفوفة AB كما يلي:
!>          AB(KL+KU+1+i-j,j) = A(i,j) لـ max(1,j-KU)<=i<=min(N,j+KL)
!>          عند الخروج، تفاصيل التحلل: يتم تخزين U كمصفوفة شريطية
!>          مثلثة علوية مع KL+KU فوق أقطار في
!>          الصفوف 1 إلى KL+KU+1، وتخزن المضاعفات المستخدمة أثناء
!>          التحلل في الصفوف KL+KU+2 إلى 2*KL+KU+1.
!>          انظر أدناه لمزيد من التفاصيل.
!>

LDAB

!>          LDAB هو صحيح (INTEGER)
!>          البعد الرائد للمصفوفة AB. LDAB >= 2*KL+KU+1.
!>

IPIV

!>          IPIV هي مصفوفة صحيحة (INTEGER array)، أبعادها (N)
!>          مؤشرات التنبييه التي تحدد مصفوفة التبادل P؛
!>          تم تبديل الصف i من المصفوفة مع الصف IPIV(i).
!>

B

!>          B هي مصفوفة حقيقية (REAL array)، أبعادها (LDB,NRHS)
!>          عند الإدخال، مصفوفة الجانب الأيمن B ذات الأبعاد N×NRHS.
!>          عند الخروج، إذا كان INFO = 0، فإن مصفوفة الحل X ذات الأبعاد N×NRHS.
!>

LDB

!>          LDB هو صحيح (INTEGER)
!>          البعد الرائد للمصفوفة B. LDB >= max(1,N).
!>

INFO

!>          INFO هو صحيح (INTEGER)
!>          = 0: خروج ناجح
!>          < 0: إذا كان INFO = -i، فإن المعامل i كان له قيمة غير قانونية
!>          > 0: إذا كان INFO = i، فإن U(i,i) يساوي صفرًا تمامًا. تم إكمال التحلل،
!>                ولكن العامل U منفرد تمامًا، ولم يتم حساب الحل.
!>

تفاصيل إضافية:

!>
!>  يتم توضيح مخطط تخزين الشريط بالمثال التالي، عندما
!>  M = N = 6، KL = 2، KU = 1:
!>
!>  عند الإدخال:                       عند الخروج:
!>
!>      *    *    *    +    +    +       *    *    *   u14  u25  u36
!>      *    *    +    +    +    +       *    *   u13  u24  u35  u46
!>      *   a12  a23  a34  a45  a56      *   u12  u23  u34  u45  u56
!>     a11  a22  a33  a44  a55  a66     u11  u22  u33  u44  u55  u66
!>     a21  a32  a43  a54  a65   *      m21  m32  m43  m54  m65   *
!>     a31  a42  a53  a64   *    *      m31  m42  m53  m64   *    *
!>
!>  لا يتم استخدام عناصر المصفوفة المميزة بـ * بواسطة الروتين؛ العناصر المميزة
!>  بـ + لا يلزم تعيينها عند الإدخال، ولكنها مطلوبة بواسطة الروتين لتخزين
!>  عناصر U بسبب الملء الناتج عن تبادلات الصفوف.
!>
