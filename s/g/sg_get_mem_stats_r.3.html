sg_get_mem_stats_r(3)

الحزمة: libstatgrab

التراخيص: LGPL, GPL

sg_get_mem_stats(3) sg_get_mem_stats(3)

الاسم

sg_get_mem_stats, sg_get_mem_stats_r, sg_free_mem_stats, sg_get_swap_stats, sg_get_swap_stats_r, sg_free_swap_stats - الحصول على إحصائيات الذاكرة الظاهرية (VM)

الملخص

#include <statgrab.h>


sg_mem_stats *sg_get_mem_stats
(size_t *entries);


sg_mem_stats *sg_get_mem_stats_r
(size_t *entries);


sg_error sg_free_mem_stats
(sg_mem_stats *data);


sg_swap_stats *sg_get_swap_stats
(size_t *entries);


sg_swap_stats *sg_get_swap_stats_r
(size_t *entries);


sg_error sg_free_swap_stats
(sg_swap_stats *data);

الوصف

يتم الوصول إلى إحصائيات الذاكرة من خلال الدالة sg_get_mem_stats() والدالة sg_get_mem_stats_r(). كلتاهما تعيدان مؤشراً إلى مخزن مؤقت من نوع sg_mem_stats. لاحظ أن هذه الإحصائية لا تحتاج إلى تعيين استخدام الذاكرة الفعلية بالكامل، بل تمثل الذاكرة الحقيقية القابلة للاستخدام بواسطة نظام التشغيل.

يتم الوصول إلى إحصائيات التبديل (swap) من خلال الدالة sg_get_swap_stats() والدالة sg_get_swap_stats_r(). كلتاهما تعيدان مؤشراً إلى مخزن مؤقت من نوع sg_swap_stats.

اختصار الواجهة البرمجية (API Shortcut)

الدالة تعيد مالك البيانات
sg_get_mem_stats sg_mem_stats * libstatgrab (محلي للخيط)
sg_get_mem_stats_r sg_mem_stats * المتصل
sg_get_swap_stats sg_swap_stats * libstatgrab (محلي للخيط)
sg_get_swap_stats_r sg_swap_stats * المتصل

يجب تحرير المخزن المؤقت sg_mem_stats الذي تم الحصول عليه من sg_get_mem_stats_r() والمخزن المؤقت sg_swap_stats الذي تم الحصول عليه من sg_get_swap_stats_r() باستخدام sg_free_mem_stats() أو sg_free_swap_stats() على التوالي، عندما لا تكون هناك حاجة إليها بعد الآن. المتصل مسؤول عن القيام بذلك.

في نظام التشغيل FreeBSD، يلزم امتيازات مرتفعة للوصول إلى إحصائيات التبديل. جعل البرنامج setgid kmem يجب أن يكون كافياً. البرامج التي تعمل بصلاحيات الجذر لن تواجه هذه المشكلة.

قيم الإرجاع

يمكن لاستدعاءات نظام الذاكرة الظاهرية (VM) إرجاع مؤشر إما إلى sg_mem_stats أو sg_swap_stats.

typedef struct{
        unsigned long long total;
        unsigned long long free;
        unsigned long long used;
        unsigned long long cache;
        time_t systime;
} sg_mem_stats;
total
إجمالي كمية الذاكرة الحقيقية بالبايت.
free
كمية الذاكرة الحقيقية الحرة بالبايت.
used
كمية الذاكرة الحقيقية المستخدمة بالبايت.
cache
كمية الذاكرة الحقيقية بالبايت المستخدمة للتخزين المؤقت.
systime
الطابع الزمني الذي تم فيه جمع الإحصائيات المذكورة أعلاه بالثواني منذ بداية العصر (epoch)
typedef struct {
        unsigned long long total;
        unsigned long long used;
        unsigned long long free;
        time_t systime;
} sg_swap_stats;
total
إجمالي مساحة التبديل (swap) بالبايت.
used
مساحة التبديل (swap) المستخدمة بالبايت.
free
مساحة التبديل (swap) الحرة بالبايت.
systime
الطابع الزمني الذي تم فيه جمع الإحصائيات المذكورة أعلاه بالثواني منذ بداية العصر (epoch)

ما يجب فعله

إضافة دالة لفتح واصف الملف (file descriptor) لهياكل ذاكرة النواة (kernel memory structures). القيام بذلك سيسمح بإسقاط الامتيازات المرتفعة في وقت مبكر.
