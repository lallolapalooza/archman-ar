subroutine sggesx (character jobvsl, character jobvsr, character sort, external selctg, character sense, integer n, real, dimension( lda, * ) a, integer lda, real, dimension( ldb, * ) b, integer ldb, integer sdim, real, dimension( * ) alphar, real, dimension( * ) alphai, real, dimension( * ) beta, real, dimension( ldvsl, * ) vsl, integer ldvsl, real, dimension( ldvsr, * ) vsr, integer ldvsr, real, dimension( 2 ) rconde, real, dimension( 2 ) rcondv, real, dimension( * ) work, integer lwork, integer, dimension( * ) iwork, integer liwork, logical, dimension( * ) bwork, integer info)

SGGESX تحسب القيم الذاتية، وشكل شور، واختياريًا، مصفوفة متجهات شور لمصفوفات GE

الغرض:

!>
!> SGGESX تحسب لمجموعة من مصفوفتين حقيقيتين غير متماثلتين N × N
!> (A,B)، القيم الذاتية المعممة، وشكل شور الحقيقي (S,T)، واختياريًا،
!> مصفوفات متجهات شور اليسرى و/أو اليمنى (VSL و
!> VSR). وهذا يعطي تحليل شور المعمم
!>
!>      (A,B) = ( (VSL) S (VSR)**T, (VSL) T (VSR)**T )
!>
!> اختياريًا، تقوم أيضًا بترتيب القيم الذاتية بحيث تظهر مجموعة مختارة
!> من القيم الذاتية في القطاعات القطرية الأمامية للمصفوفة شبه المثلثية العليا
!> S والمصفوفة المثلثية العليا T؛ تحسب
!> رقم الشرط العكسي لمتوسط القيم الذاتية المختارة
!> (RCONDE)؛ وتحسب رقم الشرط العكسي للفضاءات المنحلة اليمنى واليسرى المقابلة للقيم الذاتية
!> المختارة (RCONDV). الأعمدة الأمامية لـ VSL و VSR تشكل
!> أساسًا متعامدًا للفضاءات الذاتية المقابلة (فضاءات منحلة) اليسرى واليمنى.
!>
!> القيمة الذاتية المعممة لمجموعة من المصفوفات (A,B) هي قيمة قياسية w
!> أو نسبة alpha/beta = w، بحيث يكون A - w*B مفردًا. وهي
!> تمثل عادةً بالزوج (alpha,beta)، حيث يوجد تفسير معقول
!> لـ beta=0 أو لكليهما صفر.
!>
!> مجموعة من المصفوفات (S,T) في شكل شور معممة حقيقية بحيث T
!> مثلثية عليا بأقطار غير سالبة و S مصفوفة عليا كتلية بـ
!> قطاعات 1 × 1 و 2 × 2. القطاعات 1 × 1 تقابل
!> القيم الذاتية المعممة الحقيقية، بينما القطاعات 2 × 2 من S ستكون
!>  بجعل العناصر المقابلة لـ T بالشكل:
!>         [  a  0  ]
!>         [  0  b  ]
!>
!> والزوج المقابل من القطاعات 2 × 2 في S و T سيكون له
!> زوج مترافق مركب من القيم الذاتية المعممة.
!>
!>

المعلمات

JOBVSL 
!>          JOBVSL هو CHARACTER*1
!>          = 'N':  لا تحسب متجهات شور اليسرى؛
!>          = 'V':  احسب متجهات شور اليسرى.
!>

JOBVSR

!>          JOBVSR هو CHARACTER*1
!>          = 'N':  لا تحسب متجهات شور اليمنى؛
!>          = 'V':  احسب متجهات شور اليمنى.
!>

SORT

!>          SORT هو CHARACTER*1
!>          يحدد ما إذا كان سيتم ترتيب القيم الذاتية على
!>          القطر لشكل شور المعمم أم لا.
!>          = 'N':  القيم الذاتية غير مرتبة؛
!>          = 'S':  القيم الذاتية مرتبة (انظر SELCTG).
!>

SELCTG

!>          SELCTG هي وظيفة منطقية لثلاثة متغيرات حقيقية
!>          يجب تعريف SELCTG خارجيًا في الروتين الفرعي المستدعي.
!>          إذا كان SORT = 'N'، فلا يتم الرجوع إلى SELCTG.
!>          إذا كان SORT = 'S'، يتم استخدام SELCTG لتحديد القيم الذاتية للترتيب
!>          إلى أعلى اليسار لشكل شور.
!>          يتم تحديد قيمة ذاتية (ALPHAR(j) + ALPHAI(j))/BETA(j) إذا
!>          كان SELCTG(ALPHAR(j),ALPHAI(j),BETA(j)) صحيحًا؛ أي إذا كان
!>          أحد زوج من القيم الذاتية المترافقة المركبة محددًا،
!>          فإن كلا القيمتين الذاتيتين المركبتين محددتين.
!>          لاحظ أن القيمة الذاتية المركبة المحددة قد لا تحقق
!>          SELCTG(ALPHAR(j),ALPHAI(j),BETA(j)) = .TRUE. بعد الترتيب،
!>          لأن الترتيب قد يغير قيمة القيم الذاتية المركبة
!>          (خاصة إذا كانت القيمة الذاتية غير مستقرة)، في هذه
!>          الحالة يتم تعيين INFO إلى N+3.
!>

SENSE

!>          SENSE هو CHARACTER*1
!>          يحدد أرقام الشروط العكسية التي يتم حسابها.
!>          = 'N':  لا يتم حساب أي منها؛
!>          = 'E':  يتم حسابها لمتوسط القيم الذاتية المحددة فقط؛
!>          = 'V':  يتم حسابها للفضاءات المنحلة المحددة فقط؛
!>          = 'B':  يتم حسابها لكليهما.
!>          إذا كان SENSE = 'E'، 'V'، أو 'B'، يجب أن يساوي SORT 'S'.
!>

N

!>          N هو INTEGER
!>          رتبة المصفوفات A، B، VSL، و VSR. N >= 0.
!>

A

!>          A هو مصفوفة REAL، البعد (LDA, N)
!>          عند الإدخال، الأولى من زوج المصفوفات.
!>          عند الخروج، تم استبدال A بشكل شور المعمم الخاص بها
!>          S.
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرئيسي لـ A. LDA >= max(1,N).
!>

B

!>          B هو مصفوفة REAL، البعد (LDB, N)
!>          عند الإدخال، الثانية من زوج المصفوفات.
!>          عند الخروج، تم استبدال B بشكل شور المعمم الخاص بها
!>          T.
!>

SDIM

!>          SDIM هو INTEGER
!>          إذا كان SORT = 'N'، فإن SDIM = 0.
!>          إذا كان SORT = 'S'، فإن SDIM = عدد القيم الذاتية (بعد الترتيب)
!>          التي تكون SELCTG صحيحة لها. (الأزواج المترافقة المركبة التي
!>          تكون SELCTG صحيحة لأي من القيمتين الذاتيتين تُحتسب كـ 2.)
!>

ALPHAR

!>          ALPHAR هو مصفوفة REAL، البعد (N)
!>

ALPHAI

!>          ALPHAI هو مصفوفة REAL، البعد (N)
!>

BETA

!>          BETA هو مصفوفة REAL، البعد (N)
!>          عند الخروج، (ALPHAR(j) + ALPHAI(j)*i)/BETA(j)، j=1,...,N، ستكون
!>          القيم الذاتية المعممة. ALPHAR(j) + ALPHAI(j)*i
!>          و BETA(j)، j=1,...,N هي أقطار شكل شور المركب
!>          (S,T) التي ستنتج إذا تم اختزال القطاعات القطرية 2 × 2
!>          لشكل شور الحقيقي لـ (A,B) إلى
!>          الشكل المثلثي باستخدام تحويلات وحدوية مركبة 2 × 2.
!>          إذا كان ALPHAI(j) صفرًا، فإن القيمة الذاتية j حقيقية؛ إذا
!>          كانت موجبة، فإن القيمتين الذاتيتين j و (j+1) هما زوج
!>          مترافق مركب، مع ALPHAI(j+1) سالبًا.
!>
!>          ملاحظة: القسمة ALPHAR(j)/BETA(j) و ALPHAI(j)/BETA(j)
!>          قد تتجاوز أو تقلل بسهولة، وقد يكون BETA(j) صفرًا.
!>          لذلك، يجب على المستخدم تجنب حساب النسبة بشكل ساذج.
!>          ومع ذلك، سيكون ALPHAR و ALPHAI دائمًا أقل من
!>          وغالبًا ما يكونان مقارنين بـ norm(A) في المقدار، و BETA دائمًا
!>          أقل من وغالبًا ما يكون مقارنًا بـ norm(B).
!>

VSL

!>          VSL هو مصفوفة REAL، البعد (LDVSL,N)
!>          إذا كان JOBVSL = 'V'، فسيحتوي VSL على شور الأيسر
