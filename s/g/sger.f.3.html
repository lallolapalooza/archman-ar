subroutine sger (integer m, integer n, real alpha, real, dimension(*) x, integer incx, real, dimension(*) y, integer incy, real, dimension(lda,*) a, integer lda)

SGER

الغرض:

!>
!> SGER   تقوم بالعملية ذات الرتبة 1
!>
!>    A := alpha*x*y**T + A,
!>
!> حيث alpha هو قياسي، x هو متجه من m عنصر، y هو متجه من n عنصر، و A هي مصفوفة m × n.
!>

المعلمات

M 
!>          M هو عدد صحيح (INTEGER)
!>           عند الإدخال، يحدد M عدد الصفوف في المصفوفة A.
!>           يجب أن يكون M على الأقل صفرًا.
!>

N

!>          N هو عدد صحيح (INTEGER)
!>           عند الإدخال، يحدد N عدد الأعمدة في المصفوفة A.
!>           يجب أن يكون N على الأقل صفرًا.
!>

ALPHA

!>          ALPHA هو عدد حقيقي (REAL)
!>           عند الإدخال، يحدد ALPHA القياسي alpha.
!>

X

!>          X هو مصفوفة أعداد حقيقية (REAL array)، أبعادها على الأقل
!>           ( 1 + ( m - 1 )*abs( INCX ) ).
!>           قبل الإدخال، يجب أن تحتوي المصفوفة X المعدلة على المتجه x ذي m عنصر.
!>

INCX

!>          INCX هو عدد صحيح (INTEGER)
!>           عند الإدخال، يحدد INCX الزيادة لعناصر X.
!>           يجب ألا يكون INCX صفرًا.
!>

Y

!>          Y هو مصفوفة أعداد حقيقية (REAL array)، أبعادها على الأقل
!>           ( 1 + ( n - 1 )*abs( INCY ) ).
!>           قبل الإدخال، يجب أن تحتوي المصفوفة Y المعدلة على المتجه y ذي n عنصر.
!>

INCY

!>          INCY هو عدد صحيح (INTEGER)
!>           عند الإدخال، يحدد INCY الزيادة لعناصر Y.
!>           يجب ألا يكون INCY صفرًا.
!>

A

!>          A هي مصفوفة أعداد حقيقية (REAL array)، أبعادها ( LDA, N )
!>           قبل الإدخال، يجب أن يحتوي الجزء الرائد m × n من المصفوفة A على مصفوفة المعاملات.
!>           عند الخروج، يتم الكتابة فوق المصفوفة A بالمصفوفة المحدثة.
!>

LDA

!>          LDA هو عدد صحيح (INTEGER)
!>           عند الإدخال، يحدد LDA البعد الأول للمصفوفة A كما هو معلن في البرنامج (الفرعي) المستدعي.
!>           يجب أن يكون LDA على الأقل max( 1, m ).
!>
