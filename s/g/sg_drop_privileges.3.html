sg_drop_privileges(3)

الحزمة: libstatgrab

التراخيص: LGPL, GPL

الاسم

sg_init, sg_snapshot, sg_shutdown, sg_drop_privileges - تهيئة libstatgrab

الملخص

#include <statgrab.h>


void sg_log_init
(const char *properties_pfx, const char *env_name, const char *argv0);


sg_error sg_init
(int ignore_init_errors);


sg_error sg_snapshot
(void);


sg_error sg_shutdown
(void);


sg_error sg_drop_privileges
(void);

الوصف

تقوم sg_init() إما بتهيئة مكتبة statgrab وتعيين عداد التهيئة إلى 1، إذا لم يتم القيام بذلك بالفعل، أو بزيادة عداد التهيئة. إذا تم استدعاؤها لأول مرة (في الواقع: عند استدعائها عندما يكون عداد التهيئة 0)، يتم استدعاء المُنشئ (روتين التهيئة) لكل مكون مسجل، ويتم تهيئة بعض المتغيرات العامة، وتهيئة أقفال الاستبعاد المتبادل المطلوبة وما إلى ذلك. تتم حماية sg_init() بواسطة القفل المسمى "statgrab".

تسمح sg_log_init() بتكوين تسجيل خاص بالتطبيق. وهي مخصصة للمشاريع التي تحتوي على العديد من التطبيقات أو الأوامر التي قد يكون لديها ملحقات مختلفة. لاحظ أنه يجب استدعاء sg_log_init قبل sg_init، ولكن بعد قيام تطبيقك بتهيئة إطار عمل التسجيل.

sg_snapshot() خاصة بنظام Win32 ومن المحتمل أن يتم إزالتها.

تقوم sg_shutdown() بتقليل عداد التهيئة. إذا وصل العداد إلى الصفر، يتم استدعاء مُدمرات المكونات، ويتم تدمير أقفال الاستبعاد المتبادل (باستثناء "statgrab") وما إلى ذلك. تتم حماية sg_shutdown() بواسطة القفل المسمى "statgrab".

تقوم sg_drop_privileges() بإسقاط الامتيازات المرتفعة.

قيم الإرجاع

جميع الدوال تُرجع رمز خطأ statgrab. إما SG_ERROR_NONE عندما يكون كل شيء على ما يرام أو رمز الخطأ المناسب من مُنشئ/مُدمر.

مثال

تسلسل تهيئة/إلغاء تهيئة نموذجي عند الاستخدام مع log4cplus:

static void *l4cplus_initializer;
static void
cleanup_logging(void)
{
       log4cplus_deinitialize(l4cplus_initializer);
}
int
main(int argc, char const *argv[])
{
    l4cplus_initializer = log4cplus_initialize();
    atexit((void (*)(void))cleanup_logging);
    sg_log_init("saidar", "SAIDAR_LOG_PROPERTIES", argv[0]);
    sg_init(1);
    if(sg_drop_privileges() != 0) {
	die("Failed to drop setuid/setgid privileges");
    }
    do_something();
    sg_shutdown();
    return 0;
}
