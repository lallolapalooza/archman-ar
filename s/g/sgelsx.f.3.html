subroutine sgelsx (integer m, integer n, integer nrhs, real, dimension( lda, * ) a, integer lda, real, dimension( ldb, * ) b, integer ldb, integer, dimension( * ) jpvt, real rcond, integer rank, real, dimension( * ) work, integer info)

SGELSX يحل أنظمة مفرطة التحديد أو ناقصة التحديد لمصفوفات GE

الغرض:

!>
!> هذا الإجراء الفرعي مهمل وقد تم استبداله بالإجراء الفرعي SGELSY.
!>
!> SGELSX يحسب الحل ذو المعيار الأدنى لمشكلة أقل المربعات الخطية الحقيقية:
!>     تصغير || A * X - B ||
!> باستخدام تفكيك متعامد كامل لـ A. A هي مصفوفة M-by-N
!> قد تكون ذات رتبة ناقصة.
!>
!> يمكن التعامل مع عدة متجهات للجانب الأيمن b ومتجهات الحل x
!> في استدعاء واحد؛ يتم تخزينها كأعمدة للمصفوفة B ذات الجانب الأيمن
!> M-by-NRHS ومصفوفة الحل X ذات الجانب الأيمن N-by-NRHS.
!>
!> يقوم الإجراء الفرعي أولاً بحساب تفكيك QR مع تناوب الأعمدة:
!>     A * P = Q * [ R11 R12 ]
!>                 [  0  R22 ]
!> مع تعريف R11 كأكبر مصفوفة فرعية رائدة لها رقم حالة تقديري
!> أقل من 1/RCOND. ترتيب R11، RANK،
!> هو الرتبة الفعالة لـ A.
!>
!> بعد ذلك، تعتبر R22 مهملة، ويتم إلغاء R12
!> بواسطة تحويلات متعامدة من اليمين، مما يؤدي إلى
!> تفكيك متعامد كامل:
!>    A * P = Q * [ T11 0 ] * Z
!>                [  0  0 ]
!> الحل ذو المعيار الأدنى هو بعد ذلك
!>    X = P * Z**T [ inv(T11)*Q1**T*B ]
!>                 [        0         ]
!> حيث تتكون Q1 من أول RANK أعمدة من Q.
!>

المعلمات

M 
!>          M هو عدد صحيح
!>          عدد صفوف المصفوفة A.  M >= 0.
!>

N

!>          N هو عدد صحيح
!>          عدد أعمدة المصفوفة A.  N >= 0.
!>

NRHS

!>          NRHS هو عدد صحيح
!>          عدد الأطراف اليمنى، أي عدد
!>          أعمدة المصفوفات B و X. NRHS >= 0.
!>

A

!>          A هي مصفوفة حقيقية، البعد (LDA,N)
!>          عند الدخول، المصفوفة A بحجم M-by-N.
!>          عند الخروج، تم استبدال A بتفاصيل
!>          تفكيكها المتعامد الكامل.
!>

LDA

!>          LDA هو عدد صحيح
!>          البعد الرائد للمصفوفة A.  LDA >= max(1,M).
!>

B

!>          B هي مصفوفة حقيقية، البعد (LDB,NRHS)
!>          عند الدخول، مصفوفة الجانب الأيمن B بحجم M-by-NRHS.
!>          عند الخروج، مصفوفة الحل X بحجم N-by-NRHS.
!>          إذا كان m >= n و RANK = n، فإن مجموع مربعات البواقي
!>          للحل في العمود i يُعطى بمجموع
!>          المربعات للعناصر N+1:M في ذلك العمود.
!>

LDB

!>          LDB هو عدد صحيح
!>          البعد الرائد للمصفوفة B. LDB >= max(1,M,N).
!>

JPVT

!>          JPVT هي مصفوفة أعداد صحيحة، البعد (N)
!>          عند الدخول، إذا كان JPVT(i) .ne. 0، فإن العمود i من A هو
!>          عمود أولي، وإلا فهو عمود حر. قبل
!>          تفكيك QR لـ A، يتم تبديل جميع الأعمدة الأولية
!>          إلى المواقع الرائدة؛ فقط الأعمدة الحرة المتبقية
!>          تتحرك نتيجة لتناوب الأعمدة
!>          أثناء التفكيك.
!>          عند الخروج، إذا كان JPVT(i) = k، فإن العمود i من A*P
!>          كان العمود k من A.
!>

RCOND

!>          RCOND هي قيمة حقيقية
!>          يستخدم RCOND لتحديد الرتبة الفعالة لـ A، والتي
!>          يتم تعريفها بأنها ترتيب أكبر مصفوفة فرعية مثلثة
!>          رائدة R11 في تفكيك QR مع تناوب A،
!>          التي رقم حالتها المقدر < 1/RCOND.
!>

RANK

!>          RANK هو عدد صحيح
!>          الرتبة الفعالة لـ A، أي ترتيب المصفوفة الفرعية
!>          R11. هذا هو نفس ترتيب المصفوفة الفرعية T11
!>          في تفكيك A المتعامد الكامل.
!>

WORK

!>          WORK هي مصفوفة حقيقية، البعد
!>                      (max( min(M,N)+3*N, 2*min(M,N)+NRHS )),
!>

INFO

!>          INFO هو عدد صحيح
!>          = 0:  انتهاء ناجح
!>          < 0:  إذا كان INFO = -i، فإن المعلمة i كانت ذات قيمة غير قانونية
!>
