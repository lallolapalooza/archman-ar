subroutine sget37 (real, dimension( 3 ) rmax, integer, dimension( 3 ) lmax, integer, dimension( 3 ) ninfo, integer knt, integer nin)

SGET37

الغرض:

!>
!> SGET37 تختبر STRSNA، وهو روتين لتقدير أرقام الشروط للقيم الذاتية
!> و/أو المتجهات الذاتية اليمنى لمصفوفة.
!>
!> تتم قراءة مصفوفات الاختبار من ملف برقم وحدة منطقية NIN.
!>

المعلمات

RMAX 
!>          RMAX هي مصفوفة حقيقية، بُعدها (3)
!>          قيمة أكبر نسبة اختبار.
!>          RMAX(1) = أكبر نسبة تقارن بين استدعاءات مختلفة لـ STRSNA
!>          RMAX(2) = أكبر خطأ في معكوس أرقام الشروط
!>                    مع الأخذ في الاعتبار شروطها
!>          RMAX(3) = أكبر خطأ في معكوس أرقام الشروط
!>                    دون الأخذ في الاعتبار شروطها
!>                    (قد تكون أكبر من RMAX(2))
!>

LMAX

!>          LMAX هي مصفوفة صحيحة، بُعدها (3)
!>          LMAX(i) هو رقم المثال حيث يتم تحقيق أكبر نسبة اختبار
!>          RMAX(i). أيضًا:
!>          إذا أعادت SGEHRD قيمة غير صفرية على المثال i، فإن LMAX(1)=i
!>          إذا أعادت SHSEQR قيمة غير صفرية على المثال i، فإن LMAX(2)=i
!>          إذا أعادت STRSNA قيمة غير صفرية على المثال i، فإن LMAX(3)=i
!>

NINFO

!>          NINFO هي مصفوفة صحيحة، بُعدها (3)
!>          NINFO(1) = عدد المرات التي أعادت فيها SGEHRD قيمة غير صفرية
!>          NINFO(2) = عدد المرات التي أعادت فيها SHSEQR قيمة غير صفرية
!>          NINFO(3) = عدد المرات التي أعادت فيها STRSNA قيمة غير صفرية
!>

KNT

!>          KNT هو عدد صحيح
!>          إجمالي عدد الأمثلة التي تم اختبارها.
!>

NIN

!>          NIN هو عدد صحيح
!>          رقم وحدة الإدخال المنطقية
!>
