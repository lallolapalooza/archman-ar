subroutine sgtt02 (character trans, integer n, integer nrhs, real, dimension( * ) dl, real, dimension( * ) d, real, dimension( * ) du, real, dimension( ldx, * ) x, integer ldx, real, dimension( ldb, * ) b, integer ldb, real resid)

SGTT02

الغرض:

!>
!> SGTT02 يحسب الباقي لحل نظام معادلات ثلاثي الأقطار:
!>    RESID = norm(B - op(A)*X) / (norm(op(A)) * norm(X) * EPS)،
!> حيث EPS هو الخطأ المطبعي.
!> المعيار المستخدم هو المعيار 1.
!>

المعلمات

TRANS 
!>          TRANS is CHARACTER
!>          يحدد شكل الباقي.
!>          = 'N':  B - A    * X  (بدون تبديل)
!>          = 'T':  B - A**T * X  (تبديل)
!>          = 'C':  B - A**H * X  (تبديل مترافق = تبديل)
!>

N

!>          N is INTEGER
!>          ترتيب المصفوفة A.  N >= 0.
!>

NRHS

!>          NRHS is INTEGER
!>          عدد الأطراف اليمنى، أي عدد أعمدة المصفوفات B و X.  NRHS >= 0.
!>

DL

!>          DL is REAL array, dimension (N-1)
!>          العناصر الفرعية تحت القطر (N-1) للمصفوفة A.
!>

D

!>          D is REAL array, dimension (N)
!>          العناصر القطرية للمصفوفة A.
!>

DU

!>          DU is REAL array, dimension (N-1)
!>          العناصر فوق القطرية (N-1) للمصفوفة A.
!>

X

!>          X is REAL array, dimension (LDX,NRHS)
!>          الحلول المحسوبة للمتجهات X.
!>

LDX

!>          LDX is INTEGER
!>          البعد الرئيسي للمصفوفة X.  LDX >= max(1,N).
!>

B

!>          B is REAL array, dimension (LDB,NRHS)
!>          عند الإدخال، المتجهات الطرفية اليمنى لنظام المعادلات الخطية.
!>          عند الخروج، يتم استبدال B بالفرق B - op(A)*X.
!>

LDB

!>          LDB is INTEGER
!>          البعد الرئيسي للمصفوفة B.  LDB >= max(1,N).
!>

RESID

!>          RESID is REAL
!>          norm(B - op(A)*X) / (norm(op(A)) * norm(X) * EPS)
!>
