subroutine sgeqrs (integer m, integer n, integer nrhs, real, dimension( lda, * ) a, integer lda, real, dimension( * ) tau, real, dimension( ldb, * ) b, integer ldb, real, dimension( lwork ) work, integer lwork, integer info)

SGEQRS

الغرض:

!>
!> حل مشكلة أقل المربعات
!>     min || A*X - B ||
!> باستخدام تحليل QR
!>     A = Q*R
!> الذي تم حسابه بواسطة SGEQRF.
!>

المعلمات

M 
!>          M هو عدد صحيح (INTEGER)
!>          عدد صفوف المصفوفة A.  M >= 0.
!>

N

!>          N هو عدد صحيح (INTEGER)
!>          عدد أعمدة المصفوفة A.  M >= N >= 0.
!>

NRHS

!>          NRHS هو عدد صحيح (INTEGER)
!>          عدد أعمدة B.  NRHS >= 0.
!>

A

!>          A هي مصفوفة حقيقية (REAL array)، ذات أبعاد (LDA,N)
!>          تفاصيل تحليل QR للمصفوفة الأصلية A كما تم إرجاعها بواسطة SGEQRF.
!>

LDA

!>          LDA هو عدد صحيح (INTEGER)
!>          البعد الرئيسي للمصفوفة A.  LDA >= M.
!>

TAU

!>          TAU هي مصفوفة حقيقية (REAL array)، ذات أبعاد (N)
!>          تفاصيل المصفوفة المتعامدة Q.
!>

B

!>          B هي مصفوفة حقيقية (REAL array)، ذات أبعاد (LDB,NRHS)
!>          عند الدخول، مصفوفة الجانب الأيمن B ذات الأبعاد m-by-nrhs.
!>          عند الخروج، مصفوفة الحل X ذات الأبعاد n-by-nrhs.
!>

LDB

!>          LDB هو عدد صحيح (INTEGER)
!>          البعد الرئيسي للمصفوفة B. LDB >= M.
!>

WORK

!>          WORK هي مصفوفة حقيقية (REAL array)، ذات أبعاد (LWORK)
!>

LWORK

!>          LWORK هو عدد صحيح (INTEGER)
!>          طول المصفوفة WORK.  يجب أن يكون LWORK على الأقل NRHS،
!>          ويجب أن يكون على الأقل NRHS*NB، حيث NB هو حجم الكتلة
!>          للبيئة الحالية.
!>

INFO

!>          INFO هو عدد صحيح (INTEGER)
!>          = 0: خروج ناجح
!>          < 0: إذا كان INFO = -i، فإن الوسيط i كان له قيمة غير قانونية
!>
