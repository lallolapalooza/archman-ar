sg_get_host_info_r(3)

الحزمة: libstatgrab

التراخيص: LGPL, GPL

sg_get_host_info(3) sg_get_host_info(3)

الاسم

sg_get_host_info، sg_get_host_info_r، sg_free_host_info - الحصول على إحصائيات نظام التشغيل العامة

الملخص

#include <statgrab.h>


sg_host_info *sg_get_host_info
(size_t *entries);


sg_host_info *sg_get_host_info_r
(size_t *entries);


sg_error sg_free_host_info
(sg_host_info *data);

الوصف

تقوم هذه الاستدعاءات بإرجاع تفاصيل حول نظام التشغيل والجهاز الذي يعمل عليه.

اختصار واجهة برمجة التطبيقات (API Shortcut)

الدالة ترجع مالك البيانات
sg_get_host_info sg_host_info * libstatgrab (خاص بالخيط)
sg_get_host_info_r sg_host_info * المستدعي

يجب تحرير المخزن المؤقت sg_host_info الذي تم استلامه من sg_get_host_info_r() باستخدام sg_free_host_info() عند عدم الحاجة إليه بعد الآن. المستدعي هو المسؤول عن القيام بذلك.

قيم الإرجاع

الهيكل الذي تم إرجاعه هو من النوع sg_host_info.

typedef enum {
        sg_unknown_configuration = 0,
        sg_physical_host,
        sg_virtual_machine,
        sg_paravirtual_machine,
        sg_hardware_virtualized
} sg_host_state;

typedef struct{
        char *os_name;
        char *os_release;
        char *os_version;
        char *platform;
        char *hostname;
        unsigned bitwidth;
        sg_host_state host_state;
        unsigned ncpus;
        unsigned maxcpus;
        time_t uptime;
        time_t systime;
}sg_host_info;
os_name
اسم نظام التشغيل. (مثال: SunOS أو Linux)
os_release
إصدار نظام التشغيل. (مثال: 5.8 أو 5.9 أو Solaris)
os_version
مستوى إصدار نظام التشغيل.
platform
منصة الجهاز (الهندسة المعمارية) التي يعمل عليها نظام التشغيل.
hostname
اسم الجهاز.
uptime
وقت تشغيل الجهاز بالثواني.
systime
الطابع الزمني الذي تم فيه جمع الإحصائيات أعلاه بالثواني منذ الحقبة.
