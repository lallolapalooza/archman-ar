subroutine sget38 (real, dimension( 3 ) rmax, integer, dimension( 3 ) lmax, integer, dimension( 3 ) ninfo, integer knt, integer nin)

SGET38

الغرض:

!>
!> SGET38 تختبر STRSEN، وهي روتينية لتقدير أرقام شروط مجموعة من القيم الذاتية
!> و/أو مساحتها الجزئية اليمنى المرتبطة بها.
!>
!> تتم قراءة مصفوفات الاختبار من ملف برقم وحدة منطقي NIN.
!>

المعلمات

RMAX 
!>          RMAX هو مصفوفة REAL، أبعادها (3)
!>          قيم أكبر نسب الاختبار.
!>          RMAX(1) = أكبر البواقي من SHST01 أو المقارنة بين
!>                    استدعاءات مختلفة لـ STRSEN
!>          RMAX(2) = أكبر خطأ في أرقام الشروط المقلوبة
!>                    مع الأخذ في الاعتبار شروطها
!>          RMAX(3) = أكبر خطأ في أرقام الشروط المقلوبة
!>                    بدون الأخذ في الاعتبار شروطها (قد تكون أكبر من RMAX(2))
!>

LMAX

!>          LMAX هو مصفوفة INTEGER، أبعادها (3)
!>          LMAX(i) هو رقم المثال حيث يتم تحقيق أكبر نسبة اختبار
!>          RMAX(i). أيضاً:
!>          إذا أعادت SGEHRD قيمة INFO غير صفرية في المثال i، فإن LMAX(1)=i
!>          إذا أعادت SHSEQR قيمة INFO غير صفرية في المثال i، فإن LMAX(2)=i
!>          إذا أعادت STRSEN قيمة INFO غير صفرية في المثال i، فإن LMAX(3)=i
!>

NINFO

!>          NINFO هي مصفوفة INTEGER، أبعادها (3)
!>          NINFO(1) = عدد المرات التي أعادت فيها SGEHRD قيمة INFO غير صفرية
!>          NINFO(2) = عدد المرات التي أعادت فيها SHSEQR قيمة INFO غير صفرية
!>          NINFO(3) = عدد المرات التي أعادت فيها STRSEN قيمة INFO غير صفرية
!>

KNT

!>          KNT هو INTEGER
!>          العدد الإجمالي للأمثلة التي تم اختبارها.
!>

NIN

!>          NIN هو INTEGER
!>          رقم الوحدة المنطقية للإدخال.
!>
