subroutine sgtts2 (integer itrans, integer n, integer nrhs, real, dimension( * ) dl, real, dimension( * ) d, real, dimension( * ) du, real, dimension( * ) du2, integer, dimension( * ) ipiv, real, dimension( ldb, * ) b, integer ldb)

SGTTS2 يحل نظامًا من المعادلات الخطية مع مصفوفة ثلاثية الأقطار باستخدام تحليل LU الذي تم حسابه بواسطة sgttrf.

الغرض:

!>
!> SGTTS2 يحل أحد أنظمة المعادلات
!>    A*X = B  أو  A**T*X = B,
!> مع مصفوفة ثلاثية الأقطار A باستخدام تحليل LU
!> الذي تم حسابه بواسطة SGTTRF.
!>

المعلمات

ITRANS 
!>          ITRANS هو عدد صحيح (INTEGER)
!>          يحدد شكل نظام المعادلات.
!>          = 0:  A * X = B  (بدون تبديل)
!>          = 1:  A**T* X = B  (تبديل)
!>          = 2:  A**T* X = B  (تبديل مترافق = تبديل)
!>

N

!>          N هو عدد صحيح (INTEGER)
!>          رتبة المصفوفة A.
!>

NRHS

!>          NRHS هو عدد صحيح (INTEGER)
!>          عدد الأطراف اليمنى، أي عدد أعمدة
!>          المصفوفة B.  NRHS >= 0.
!>

DL

!>          DL هو مصفوفة حقيقية (REAL array)، البعد (N-1)
!>          المضاعفات (n-1) التي تحدد المصفوفة L من
!>          تحليل LU للمصفوفة A.
!>

D

!>          D هو مصفوفة حقيقية (REAL array)، البعد (N)
!>          العناصر القطرية n للمصفوفة المثلثية العليا U من
!>          تحليل LU للمصفوفة A.
!>

DU

!>          DU هو مصفوفة حقيقية (REAL array)، البعد (N-1)
!>          العناصر (n-1) للقطر العلوي الأول لـ U.
!>

DU2

!>          DU2 هو مصفوفة حقيقية (REAL array)، البعد (N-2)
!>          العناصر (n-2) للقطر العلوي الثاني لـ U.
!>

IPIV

!>          IPIV هو مصفوفة أعداد صحيحة (INTEGER array)، البعد (N)
!>          فهارس المحور؛ لـ 1 <= i <= n، تم تبديل الصف i للمصفوفة
!>          مع الصف IPIV(i).  سيكون IPIV(i) دائمًا إما
!>          i أو i+1؛ يشير IPIV(i) = i إلى عدم الحاجة إلى تبديل صف.
!>

B

!>          B هي مصفوفة حقيقية (REAL array)، البعد (LDB,NRHS)
!>          عند الدخول، المصفوفة التي تحتوي على متجهات الطرف الأيمن B.
!>          عند الخروج، يتم استبدال B بمتجهات الحل X.
!>

LDB

!>          LDB هو عدد صحيح (INTEGER)
!>          البعد الرئيسي للمصفوفة B.  LDB >= max(1,N).
!>
