subroutine sgglse (integer m, integer n, integer p, real, dimension( lda, * ) a, integer lda, real, dimension( ldb, * ) b, integer ldb, real, dimension( * ) c, real, dimension( * ) d, real, dimension( * ) x, real, dimension( * ) work, integer lwork, integer info)

SGGLSE يحل أنظمة إفراط في التحديد أو نقص في التحديد لمصفوفات أخرى

الغرض:

!>
!> SGGLSE يحل مشكلة المربعات الصغرى ذات القيود الخطية (LSE)
!> المشكلة:
!>
!>         تقليل || c - A*x ||_2   بشرط   B*x = d
!>
!> حيث A مصفوفة M-by-N، و B مصفوفة P-by-N، و c هو متجه M معطى،
!> و d هو متجه P معطى. يُفترض أن
!> P <= N <= M+P، و
!>
!>          rank(B) = P و  rank( (A) ) = N.
!>                               ( (B) )
!>
!> تضمن هذه الشروط أن مشكلة LSE لها حل فريد،
!> والذي يتم الحصول عليه باستخدام تحليل RQ المعمم للمصفوفات
!> (B, A) المعطاة بواسطة
!>
!>    B = (0 R)*Q,   A = Z*T*Q.
!>
!> يجب على المستدعين لهذه الروتين الفرعي ملاحظة أن التحقق من التفرد/نقص الرتبة
!> المنفذ في هذه الروتين الفرعي هو بدائي. الروتين STRTRS الذي تستدعيه هذه
!> الروتين الفرعي يشير فقط إلى فشل بسبب التفرد إذا كانت المشكلة مفردة تمامًا.
!>
!> من الممكن أن يكون أحد العوامل (أو أكثر) المشاركة في التحليل المعمم RQ
!> للزوج (B، A) قريبًا بشكل غير طبيعي من التفرد دون أن تشير هذه
!> الروتين الفرعي إلى خطأ. قد تكون الحلول المحسوبة لمثل هذه المشاكل ذات الرتبة شبه الناقصة
!> أقل دقة بسبب فقدان الدقة العددية.
!> 
!>

المعلمات

M 
!>          M هو عدد صحيح (INTEGER)
!>          عدد صفوف المصفوفة A.  M >= 0.
!>

N

!>          N هو عدد صحيح (INTEGER)
!>          عدد أعمدة المصفوفات A و B. N >= 0.
!>

P

!>          P هو عدد صحيح (INTEGER)
!>          عدد صفوف المصفوفة B. 0 <= P <= N <= M+P.
!>

A

!>          A هي مصفوفة أعداد حقيقية (REAL array)، الأبعاد (LDA,N)
!>          عند الإدخال، المصفوفة A بحجم M-by-N.
!>          عند الخروج، العناصر على القطر وفوقه في المصفوفة
!>          تحتوي على المصفوفة الشبه منحرفة العلوية T بحجم min(M,N)-by-N.
!>

LDA

!>          LDA هو عدد صحيح (INTEGER)
!>          البعد الرئيسي للمصفوفة A. LDA >= max(1,M).
!>

B

!>          B هي مصفوفة أعداد حقيقية (REAL array)، الأبعاد (LDB,N)
!>          عند الإدخال، المصفوفة B بحجم P-by-N.
!>          عند الخروج، المثلث العلوي من المصفوفة الفرعية B(1:P,N-P+1:N)
!>          يحتوي على المصفوفة المثلثية العلوية R بحجم P-by-P.
!>

LDB

!>          LDB هو عدد صحيح (INTEGER)
!>          البعد الرئيسي للمصفوفة B. LDB >= max(1,P).
!>

C

!>          C هي مصفوفة أعداد حقيقية (REAL array)، الأبعاد (M)
!>          عند الإدخال، تحتوي C على متجه الطرف الأيمن لجزء
!>          المربعات الصغرى من مشكلة LSE.
!>          عند الخروج، مجموع مربعات البواقي للحل
!>          معطى بمجموع مربعات العناصر من N-P+1 إلى M
!>          من المتجه C.
!>

D

!>          D هي مصفوفة أعداد حقيقية (REAL array)، الأبعاد (P)
!>          عند الإدخال، يحتوي D على متجه الطرف الأيمن للمعادلة
!>          المقيدة.
!>          عند الخروج، يتم تدمير D.
!>

X

!>          X هي مصفوفة أعداد حقيقية (REAL array)، الأبعاد (N)
!>          عند الخروج، X هو حل مشكلة LSE.
!>

WORK

!>          WORK هي مصفوفة أعداد حقيقية (REAL array)، الأبعاد (MAX(1,LWORK))
!>          عند الخروج، إذا كان INFO = 0، فإن WORK(1) تعيد LWORK الأمثل.
!>

LWORK

!>          LWORK هو عدد صحيح (INTEGER)
!>          بعد المصفوفة WORK. LWORK >= max(1,M+N+P).
!>          للحصول على الأداء الأمثل LWORK >= P+min(M,N)+max(M,N)*NB,
!>          حيث NB هو حد أعلى لأحجام الكتل المثلى لـ
!>          SGEQRF، SGERQF، SORMQR و SORMRQ.
!>
!>          إذا كان LWORK = -1، فيُفترض استعلام مساحة العمل؛ الروتين
!>          يحسب فقط الحجم الأمثل للمصفوفة WORK، ويعيد
!>          هذه القيمة كالإدخال الأول للمصفوفة WORK، ولا يصدر أي رسالة خطأ
!>          متعلقة بـ LWORK بواسطة XERBLA.
!>

INFO

!>          INFO هو عدد صحيح (INTEGER)
!>          = 0:  انتهاء ناجح.
!>          < 0:  إذا كان INFO = -i، فإن الوسيط i كان له قيمة غير قانونية.
!>          = 1:  العامل المثلثي العلوي R المرتبط بـ B في
!>                التحليل المعمم RQ للزوج (B، A) مفرد تمامًا، بحيث rank(B) < P؛ لم يتمكن من حساب
!>                حل المربعات الصغرى.
!>          = 2:  الجزء (N-P) by (N-P) من العامل الشبه منحرف العلوي
!>                T المرتبط بـ A في التحليل المعمم RQ
!>                للزوج (B، A) مفرد تمامًا، بحيث
!>                rank( (A) ) < N؛ لم يتمكن من حساب
!>                    ( (B) )
!>                حل المربعات الصغرى.
!>
