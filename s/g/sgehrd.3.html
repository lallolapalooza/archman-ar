وثائق الوظيفة/الإجراء الفرعي

subroutine sgehrd (integer n, integer ilo, integer ihi, real, dimension( lda, * ) a, integer lda, real, dimension( * ) tau, real, dimension( * ) work, integer lwork, integer info)

SGEHRD

الغرض:

!>
!> SGEHRD تقلل مصفوفة عامة حقيقية A إلى شكل هسنبرغ علوي H بواسطة
!> تحويل تشابه متعامد:  Q**T * A * Q = H .
!>

المعلمات

N 
!>          N هو عدد صحيح (INTEGER)
!>          رتبة المصفوفة A.  N >= 0.
!>

ILO

!>          ILO هو عدد صحيح (INTEGER)
!>

IHI

!>          IHI هو عدد صحيح (INTEGER)
!>
!>          من المفترض أن تكون A بالفعل مثلثية علوية في الصفوف
!>          والأعمدة 1:ILO-1 و IHI+1:N. عادة ما يتم تعيين ILO و IHI
!>          بواسطة استدعاء سابق لـ SGEBAL؛ وإلا فيجب تعيينهما
!>          إلى 1 و N على التوالي. انظر التفاصيل الإضافية.
!>          1 <= ILO <= IHI <= N, إذا كان N > 0؛ ILO=1 و IHI=0, إذا كان N=0.
!>

A

!>          A هي مصفوفة حقيقية (REAL array)، ذات أبعاد (LDA,N)
!>          عند الإدخال، المصفوفة العامة N-by-N المراد تقليلها.
!>          عند الخروج، يتم استبدال الجزء العلوي المثلثي والقطر السفلي الأول من A
!>          بالمصفوفة هسنبرغ العلوية H، والعناصر الموجودة أسفل القطر السفلي الأول،
!>          مع المصفوفة TAU، تمثل المصفوفة المتعامدة Q كناتج ضرب من
!>          عواكس أولية. انظر التفاصيل الإضافية.
!>

LDA

!>          LDA هو عدد صحيح (INTEGER)
!>          البعد الرائد للمصفوفة A.  LDA >= max(1,N).
!>

TAU

!>          TAU هي مصفوفة حقيقية (REAL array)، ذات أبعاد (N-1)
!>          العوامل القياسية للعواكس الأولية (انظر التفاصيل الإضافية).
!>          تم تعيين العناصر 1:ILO-1 و IHI:N-1 من TAU إلى صفر.
!>

WORK

!>          WORK هي مصفوفة حقيقية (REAL array)، ذات أبعاد (MAX(1,LWORK))
!>          عند الخروج، إذا كان INFO = 0، فإن WORK(1) تعيد LWORK المثلى.
!>

LWORK

!>          LWORK هو عدد صحيح (INTEGER)
!>          طول المصفوفة WORK.  LWORK >= max(1,N).
!>          لأداء جيد، يجب أن يكون LWORK أكبر بشكل عام.
!>
!>          إذا كان LWORK = -1، فمن المفترض استعلام مساحة العمل؛ تقوم الروتين
!>          فقط بحساب الحجم الأمثل للمصفوفة WORK، وتعيد
!>          هذه القيمة كالإدخال الأول للمصفوفة WORK، ولا يتم إصدار أي رسالة خطأ
!>          متعلقة بـ LWORK بواسطة XERBLA.
!>

INFO

!>          INFO هو عدد صحيح (INTEGER)
!>          = 0:  خروج ناجح
!>          < 0:  إذا كان INFO = -i، فإن الوسيط i كان له قيمة غير قانونية.
!>

تفاصيل إضافية:

!>
!>  يتم تمثيل المصفوفة Q كناتج ضرب لـ (ihi-ilo) من العواكس
!>  الأولية
!>
!>     Q = H(ilo) H(ilo+1) . . . H(ihi-1).
!>
!>  كل H(i) له الشكل
!>
!>     H(i) = I - tau * v * v**T
!>
!>  حيث tau هو عامل قياسي حقيقي، و v هو متجه حقيقي مع
!>  v(1:i) = 0، v(i+1) = 1 و v(ihi+1:n) = 0؛ يتم تخزين v(i+2:ihi) عند
!>  الخروج في A(i+2:ihi,i)، و tau في TAU(i).
!>
!>  يتم توضيح محتويات A بالمثال التالي، مع
!>  n = 7، ilo = 2، و ihi = 6:
!>
!>  عند الإدخال،                        عند الخروج،
!>
!>  ( a   a   a   a   a   a   a )    (  a   a   h   h   h   h   a )
!>  (     a   a   a   a   a   a )    (      a   h   h   h   h   a )
!>  (     a   a   a   a   a   a )    (      h   h   h   h   h   h )
!>  (     a   a   a   a   a   a )    (      v2  h   h   h   h   h )
!>  (     a   a   a   a   a   a )    (      v2  v3  h   h   h   h )
!>  (     a   a   a   a   a   a )    (      v2  v3  v4  h   h   h )
!>  (                         a )    (                          a )
!>
!>  حيث a يمثل عنصرًا من المصفوفة الأصلية A، h يمثل عنصرًا معدلاً
!>  من مصفوفة هسنبرغ العلوية H، و vi يمثل عنصرًا من المتجه الذي يحدد H(i).
!>
!>  هذا الملف هو تعديل طفيف لإجراء LAPACK-3.0's SGEHRD
!>  يدمج التحسينات المقترحة بواسطة Quintana-Orti و
!>  Van de Geijn (2006). (انظر SLAHR2.)
!>
