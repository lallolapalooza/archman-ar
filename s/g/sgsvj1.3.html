subroutine sgsvj1 (character*1 jobv, integer m, integer n, integer n1, real, dimension( lda, * ) a, integer lda, real, dimension( n ) d, real, dimension( n ) sva, integer mv, real, dimension( ldv, * ) v, integer ldv, real eps, real sfmin, real tol, integer nsweep, real, dimension( lwork ) work, integer lwork, integer info)

SGSVJ1 هو معالج مسبق للروتين sgesvj، ويطبق دورانات يعقوبية تستهدف محاور محددة فقط.

الغرض:

!>
!> يتم استدعاء SGSVJ1 من SGESVJ كمعالج مسبق وهذا هو غرضه الرئيسي.
!> يطبق دورانات يعقوبية بنفس الطريقة التي يطبقها SGESVJ، ولكنه يستهدف
!> محاور محددة فقط ولا يتحقق من التقارب (معيار التوقف). تتوفر بعض معلمات
!> الضبط (مميزة بـ [TP]) للمنفذ.
!>
!> تفاصيل إضافية
!> ~~~~~~~~~~~~~~~
!> يطبق SGSVJ1 بعض دورات من دورانات يعقوبية في فضاء الأعمدة للمصفوفة
!> المدخلة M-by-N A. يتم أخذ أزواج المحاور من الكتلة القطرية الخارجية (1,2)
!> في مصفوفة جرام المقابلة A^T * A. الكتل المدخلة (البلاطات) للكتلة
!> القطرية الخارجية (1,2) مميزة بـ [x] في المخطط التالي:
!>
!>    | *  *  * [x] [x] [x]|
!>    | *  *  * [x] [x] [x]|    تكرار الصفوف في كتل [x] nblr-by-nblc.
!>    | *  *  * [x] [x] [x]|    محور الصفوف داخل كل كتلة [x].
!>    |[x] [x] [x] *  *  * |
!>    |[x] [x] [x] *  *  * |
!>    |[x] [x] [x] *  *  * |
!>
!> فيما يتعلق بأعمدة A، يتم تدوير أول N1 عمود 'مقابل' الأعمدة المتبقية N-N1
!> من A، في محاولة لزيادة الزاوية بين الفضاءات الفرعية المقابلة. الكتلة
!> القطرية الخارجية هي N1-by-(N-N1) ويتم تبليطها باستخدام بلاطات مربعة
!> ذات طول جانبي KBL. هنا، KBL هي معلمة ضبط. عدد الدورات معطى في
!> NSWEEP وعتبة التعامد معطاة في TOL.
!>

المعلمات

JOBV 
!>          JOBV هو CHARACTER*1
!>          يحدد ما إذا كان سيتم استخدام المخرجات من هذا الإجراء
!>          لحساب المصفوفة V:
!>          = 'V': يتم تجميع حاصل ضرب دورانات يعقوبية
!>                 بضرب المصفوفة V ذات الأبعاد N-by-N لاحقًا.
!>                (انظر وصف V.)
!>          = 'A': يتم تجميع حاصل ضرب دورانات يعقوبية
!>                 بضرب المصفوفة MV-by-N V لاحقًا.
!>                (انظر أوصاف MV و V.)
!>          = 'N': لا يتم تجميع دورانات يعقوبية.
!>

M

!>          M هو INTEGER
!>          عدد الصفوف في المصفوفة المدخلة A.  M >= 0.
!>

N

!>          N هو INTEGER
!>          عدد الأعمدة في المصفوفة المدخلة A.
!>          M >= N >= 0.
!>

N1

!>          N1 هو INTEGER
!>          N1 يحدد تقسيم الكتلة 2x2، حيث يتم تدوير أول N1 عمود 'مقابل'
!>          للأعمدة المتبقية N-N1 من A.
!>

A

!>          A هو مصفوفة REAL، ذات أبعاد (LDA,N)
!>          عند الدخول، مصفوفة A بأبعاد M-by-N، بحيث تمثل A*diag(D)
!>          المصفوفة المدخلة.
!>          عند الخروج،
!>          A_onexit * D_onexit تمثل المصفوفة المدخلة A*diag(D)
!>          مضروبة لاحقًا في تسلسل من دورانات يعقوبية، حيث يتم إعطاء
!>          عتبة الدوران وإجمالي عدد الدورات في
!>          TOL و NSWEEP، على التوالي.
!>          (انظر أوصاف N1، D، TOL و NSWEEP.)
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرئيسي للمصفوفة A.  LDA >= max(1,M).
!>

D

!>          D هي مصفوفة REAL، ذات أبعاد (N)
!>          تجمع المصفوفة D عوامل القياس من دورانات يعقوبية سريعة مقاسة.
!>          عند الدخول، تمثل A*diag(D) المصفوفة المدخلة.
!>          عند الخروج، تمثل A_onexit*diag(D_onexit) المصفوفة المدخلة
!>          مضروبة لاحقًا في تسلسل من دورانات يعقوبية، حيث يتم إعطاء
!>          عتبة الدوران وعدد الدورات الإجمالي في
!>          TOL و NSWEEP، على التوالي.
!>          (انظر أوصاف N1، A، TOL و NSWEEP.)
!>

SVA

!>          SVA هي مصفوفة REAL، ذات أبعاد (N)
!>          عند الدخول، تحتوي SVA على المعايير الإقليدية لأعمدة
!>          المصفوفة A*diag(D).
!>          عند الخروج، تحتوي SVA على المعايير الإقليدية لأعمدة
!>          المصفوفة onexit*diag(D_onexit).
!>

MV

!>          MV هو INTEGER
!>          إذا كان JOBV = 'A'، فسيتم ضرب MV صفًا من V لاحقًا في تسلسل
!>                           من دورانات يعقوبية.
!>          إذا كان JOBV = 'N'،   فلن يتم استخدام MV.
!>

V

!>          V هي مصفوفة REAL، ذات أبعاد (LDV,N)
!>          إذا كان JOBV = 'V'، فسيتم ضرب N صفًا من V لاحقًا في تسلسل
!>                           من دورانات يعقوبية.
!>          إذا كان JOBV = 'A'، فسيتم ضرب MV صفًا من V لاحقًا في تسلسل
!>                           من دورانات يعقوبية.
!>          إذا كان JOBV = 'N'،   فلن يتم استخدام V.
!>

LDV

!>          LDV هو INTEGER
!>          البعد الرئيسي للمصفوفة V، LDV >= 1.
!>          إذا كان JOBV = 'V'، فإن LDV >= N.
!>          إذا كان JOBV = 'A'، فإن LDV >= MV.
!>

EPS

!>          EPS هو REAL
!>          EPS = SLAMCH('Epsilon')
!>

SFMIN

!>          SFMIN هو REAL
!>          SFMIN = SLAMCH('Safe Minimum')
!>

TOL

!>          TOL هو REAL
!>          TOL هي عتبة دورانات يعقوبية. لزوج
!>          من أعمدة المحاور A(:,p)، A(:,q)، يتم تطبيق دوران يعقوبي فقط
!>          إذا كان ABS(COS(angle(A(:,p),A(:,q)))) > TOL.
!>

NSWEEP

!>          NSWEEP هو INTEGER
!>          NSWEEP هو عدد دورانات يعقوبية التي سيتم إجراؤها.
!>

WORK

!>          WORK هي مصفوفة REAL، ذات أبعاد (LWORK)
!>

LWORK

!>          LWORK هو INTEGER
!>          LWORK هو بعد WORK. LWORK >= M.
!>

INFO

!>          INFO هو INTEGER
!>          = 0:  خروج ناجح.
!>          < 0:  إذا كان INFO = -i، فإن المعامل i كان له قيمة غير قانونية
!>
