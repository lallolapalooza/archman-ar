الوصف

sg_vector_create() تقوم بتخصيص وتهيئة متجه statgrab جديد مع عناصر initial_used جاهزة للاستخدام. يتم تخصيص مساحة مبدئية لـ alloc_count عناصر (لتجنب العديد من استدعاءات realloc() خلال استدعاءات sg_vector_resize() اللاحقة). يجب أن يكون حجم block_size قوة للرقم 2، ويتم تقريبه لأعلى إلى أقرب قوة للرقم 2 عندما لا يكون كذلك. إذا لم يكن alloc_count مضاعفًا لـ block_size، فسيتم تقريبه لأعلى إلى أقرب مضاعف لـ block_size. تقوم بإرجاع مؤشر إلى المتجه المنشأ حديثًا.

sg_vector_clear() تدمر جميع العناصر الموجودة في المتجه المعطى. على عكس sg_vector_resize( x, 0 )، يبقى حجم المتجه المخصص دون تغيير.

sg_vector_resize() تزيد أو تقلل كمية العناصر المخصصة في المتجه المحدد. كمية العناصر المخصصة هي دائمًا مضاعف للمعامل block_size المعطى في التهيئة. في حالة خاصة، عندما يتم استدعاء sg_vector_resize() مع القيمة 0 في الوسيط new_count، يتم تحرير المتجه بعد تدمير جميع عناصر المتجه. تقوم بإرجاع مؤشر إلى المتجه الذي تم تغيير حجمه.

sg_vector_free() تدمر جميع عناصر المتجه وتقوم بإلغاء تخصيص التخزين الخاص بالمتجه المعطى.

sg_vector_clone() تقوم بنسخ جميع عناصر المتجه المعطى إلى متجه جديد تم إنشاؤه بنفس مواصفات المتجه الأصلي. تقوم بإرجاع مؤشر إلى المتجه المنسوخ.

sg_vector_clone_into() تقوم بنسخ جميع عناصر متجه المصدر المعطى إلى متجه الوجهة المعطى. يجب أن يكون متجه الوجهة قد تم إنشاؤه لنفس نوع بيانات العناصر مثل متجه المصدر. تقوم بإرجاع رمز خطأ غير SG_ERROR_NONE إذا حدث خطأ ما.

sg_vector_compute_diff() تقوم بحساب متجه فرق بين المتجه الذي يحتوي على الإحصائيات الحالية ومتجه آخر يحتوي على إحصائيات أقدم. إذا كان عنصر موجودًا في المتجه الحالي ولكن ليس في المتجه المقابل، يتم نسخه إلى متجه النتيجة. إذا كان عنصر موجودًا فقط في المتجه المقابل، فإنه لا يظهر في متجه الوجهة. sg_vector_compute_diff() تقوم بإرجاع رمز خطأ غير SG_ERROR_NONE إذا حدث خطأ ما.

sg_prove_vector() تثبت ما إذا كان مؤشر إلى متجه يشير فعليًا إلى متجه. في حالة كان مؤشر المتجه يشير إلى بيانات تالفة، يتم إنهاء البرنامج. عندما تعود sg_prove_vector()، فإنها تعيد SG_ERROR_NONE.

sg_get_nelements() تقوم بإرجاع عدد العناصر التي تحتوي عليها منطقة البيانات المعطاة، والتي يشملها متجه statgrab. يتم حساب رأس المتجه داخليًا من المؤشر المعطى للعنصر الأول في المتجه.

sg_free_stats_buf() تقوم بتحرير المتجه الذي يشمل منطقة البيانات المعطاة.

ملاحظات

باستثناء sg_get_nelements() و sg_free_stats_buf()، لا يمكن استدعاء أي من الدوال المذكورة أعلاه من خارج مصادر libstatgrab. قد تتغير الهياكل وواجهات برمجة التطبيقات الموثقة دون سابق إنذار. وصف جميع واجهات برمجة التطبيقات الأخرى مخصص لمطوري libstatgrab فقط.

كل متجه يتم إنشاؤه من عنصرين: معلومات المتجه وقائمة العناصر:

template <class T, class Impl>
struct sg_vector {
        size_t used_count;
        size_t alloc_count;
        size_t block_shift;
        Impl vector_implementation;
        T elements[alloc_count];
};

بالطبع، هذا ليس C صحيحًا، لذا كان الحل هو استخدام حيل:

typedef struct sg_vector {
	size_t used_count;
	size_t alloc_count;
	size_t block_shift;
	struct sg_vector_init_info info;
} sg_vector;
struct sg_vector_size_helper {
	struct sg_vector v;
	long long ll;
};
#define VECTOR_SIZE offsetof(struct sg_vector_size_helper,ll)
/* إرجاع مؤشر بيانات المتجه */
#define VECTOR_DATA(vector) \
	(vector ? (void *)(((char *)vector)+VECTOR_SIZE) : NULL)
#define VECTOR_ADDR_ARITH(ptr) \
	(sg_vector *)(((char *)(ptr))-VECTOR_SIZE)
/* إرجاع المتجه للبيانات */
#define VECTOR_ADDRESS(ptr) \
	((ptr) ? (SG_ERROR_NONE == sg_prove_vector(VECTOR_ADDR_ARITH(ptr)) ? VECTOR_ADDR_ARITH(ptr) : NULL ) : NULL)

هذا يسمح أيضًا للدوال المستخدمة مثل sg_get_nelements() و sg_free_stats_buf() بالتبديل بسهولة بين بنية المتجه والمحتوى.

هيكل تخصيص المتجه

كما ذكرنا، يستخدم تطبيق المتجه استراتيجيات من مفهوم البرمجة كائنية التوجه المسماة "تعدد الأشكال". يتم وصف المتجه بواسطة كائن صغير يحتوي على خصائص متأصلة مثل حجم العنصر ومجموعة من "الأساليب الافتراضية" للقيام بمهام متعلقة بالعناصر مثل تهيئة أو تدمير العناصر.

typedef void (*vector_init_function)(void *item);
typedef sg_error (*vector_copy_function)(const void *src, void *dst);
typedef sg_error (*vector_compute_diff_function)(void *dst, const void *src);
typedef int (*vector_compare_function)(const void *a, const void *b);
typedef void (*vector_destroy_function)(void *item);
struct sg_vector_init_info {
        size_t item_size;
        vector_init_function init_fn;
        vector_copy_function copy_fn;
        vector_compute_diff_function compute_diff_fn;
        vector_compare_function compare_fn;
        vector_destroy_function destroy_fn;
};

يتم تهيئة مثيلات struct sg_vector_init_info بشكل مفاهيمي ثابت باستخدام الماكرو المعالج المسبق VECTOR_INIT_INFO_FULL_INIT(type) أو VECTOR_INIT_INFO_EMPTY_INIT(type). إليك بعض الأمثلة لتوضيح كيفية استخدامها:

تهيئة وصف متجه إحصائيات وحدة المعالجة المركزية

VECTOR_INIT_INFO_EMPTY_INIT(sg_cpu_stats);

تهيئة وصف متجه إحصائيات معلومات المضيف

static void sg_os_stats_item_init(sg_os_stats *d);
static void sg_os_stats_item_destroy(sg_os_stats *d);
#define sg_os_stats_item_copy NULL
#define sg_os_stats_item_compute_diff NULL
#define sg_os_stats_item_compare NULL
VECTOR_INIT_INFO_FULL_INIT(sg_os_stats);

تهيئة وصف متجه إحصائيات Disk-IO

static void sg_disk_io_stats_item_init(sg_disk_io_stats *d);
static sg_error sg_disk_io_stats_item_copy(sg_disk_io_stats *d, const sg_disk_io_stats *s);
static sg_error sg_disk_io_stats_item_compute_diff(const sg_disk_io_stats *s, sg_disk_io_stats *d);
static int sg_disk_io_stats_item_compare(const sg_disk_io_stats *a, const sg_disk_io_stats *b);
