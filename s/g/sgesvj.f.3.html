subroutine sgesvj (character*1 joba, character*1 jobu, character*1 jobv, integer m, integer n, real, dimension( lda, * ) a, integer lda, real, dimension( n ) sva, integer mv, real, dimension( ldv, * ) v, integer ldv, real, dimension( lwork ) work, integer lwork, integer info)

SGESVJ

الغرض:

!>
!> SGESVJ تحسب تفكيك القيم المفردة (SVD) لمصفوفة حقيقية A بحجم M × N، حيث M >= N. يُكتب تفكيك القيم المفردة لـ A كـ
!>                                    [++]   [xx]   [x0]   [xx]
!>              A = U * SIGMA * V^t,  [++] = [xx] * [ox] * [xx]
!>                                    [++]   [xx]
!> حيث SIGMA مصفوفة قطرية بحجم N × N، U مصفوفة متعامدة بحجم M × N، و V مصفوفة متعامدة بحجم N × N. العناصر القطرية
!> لـ SIGMA هي القيم المفردة للمصفوفة A. أعمدة U و V هي المتجهات المفردة اليسرى واليمنى للمصفوفة A، على التوالي.
!> يمكن لـ SGESVJ في بعض الأحيان حساب قيم مفردة صغيرة جدًا ومتجهاتها بشكل أكثر دقة من روتينات SVD الأخرى، انظر أدناه تحت التفاصيل الإضافية.
!>

المعلمات

JOBA 
!>          JOBA هو CHARACTER*1
!>          يحدد بنية المصفوفة A.
!>          = 'L': المصفوفة المدخلة A هي مصفوفة مثلثية سفلية؛
!>          = 'U': المصفوفة المدخلة A هي مصفوفة مثلثية علوية؛
!>          = 'G': المصفوفة المدخلة A هي مصفوفة عامة بحجم M × N، حيث M >= N.
!>

JOBU

!>          JOBU هو CHARACTER*1
!>          يحدد ما إذا كان سيتم حساب المتجهات المفردة اليسرى
!>          (أعمدة U):
!>          = 'U': يتم حساب المتجهات المفردة اليسرى المقابلة للقيم المفردة غير الصفرية وإعادتها في الأعمدة الرائدة للمصفوفة A. انظر المزيد من التفاصيل في وصف A.
!>                 يتم تعيين عتبة التعامد العددية الافتراضية إلى
!>                 حوالي TOL=CTOL*EPS، حيث CTOL=SQRT(M)، EPS=SLAMCH('E').
!>          = 'C': مشابه لـ JOBU='U'، باستثناء أنه يمكن للمستخدم التحكم في
!>                 مستوى التعامد العددي للمتجهات المفردة اليسرى المحسوبة. يمكن تعيين TOL إلى TOL = CTOL*EPS، حيث
!>                 يتم إعطاء CTOL في الإدخال في المصفوفة WORK.
!>                 لا يُسمح بأي CTOL أصغر من ONE. CTOL أكبر
!>                 من 1 / EPS لا معنى له. يمكن استخدام الخيار 'C'
!>                 إذا كان M*EPS يوفر تعامدًا مرضيًا
!>                 للمتجهات المفردة اليسرى المحسوبة، لذا يمكن لـ CTOL=M
!>                 توفير بضع دورات دوران جاكوبي.
!>                 انظر أوصاف A و WORK(1).
!>          = 'N': لم يتم حساب المصفوفة U. ومع ذلك، انظر وصف A.
!>

JOBV

!>          JOBV هو CHARACTER*1
!>          يحدد ما إذا كان سيتم حساب المتجهات المفردة اليمنى، أي
!>          المصفوفة V:
!>          = 'V':  يتم حساب المصفوفة V وإعادتها في المصفوفة V
!>          = 'A':  يتم تطبيق دورانات جاكوبي على المصفوفة MV × N
!>                  المخزنة في V. بمعنى آخر، لم يتم حساب المصفوفة المتجهة المفردة اليمنى V بشكل صريح؛ بدلاً من ذلك، يتم
!>                  تطبيقها على مصفوفة MV × N مخزنة مبدئيًا في
!>                  الصفوف MV الأولى من V.
!>          = 'N':  لم يتم حساب المصفوفة V ولا يتم الرجوع إلى المصفوفة V
!>

M

!>          M هو INTEGER
!>          عدد صفوف المصفوفة المدخلة A. 1/SLAMCH('E') > M >= 0.
!>

N

!>          N هو INTEGER
!>          عدد أعمدة المصفوفة المدخلة A.
!>          M >= N >= 0.
!>

A

!>          A هي مصفوفة REAL، ذات بعد (LDA,N)
!>          عند الدخول، المصفوفة A بحجم M × N.
!>          عند الخروج،
!>          إذا كان JOBU = 'U' .OR. JOBU = 'C':
!>                 إذا كان INFO = 0:
!>                 RANKA من الأعمدة المتعامدة لـ U يتم إرجاعها في
!>                 الأعمدة الرائدة RANKA للمصفوفة A. هنا RANKA <= N
!>                 هو عدد القيم المفردة المحسوبة للمصفوفة A التي هي
!>                 أعلى من عتبة الانخفاض SLAMCH('S'). المتجهات المفردة المقابلة للقيم المفردة المنخفضة جدًا أو الصفرية
!>                 لا يتم حسابها. يتم إرجاع قيمة RANKA
!>                 في المصفوفة WORK كـ RANKA=NINT(WORK(2)). انظر أيضًا أوصاف SVA و WORK. أعمدة U المحسوبة متعامدة رقميًا بشكل متبادل حتى حوالي
!>                 TOL=SQRT(M)*EPS (الافتراضي)؛ أو TOL=CTOL*EPS (JOBU = 'C')،
!>                 انظر وصف JOBU.
!>                 إذا كان INFO > 0،
!>                 لم يتقارب الروتين الفرعي SGESVJ في العدد المحدد
!>                 من الدورات (المرات). في هذه الحالة، قد لا تكون أعمدة U المحسوبة متعامدة حتى عتبة TOL. يظل الإخراج
!>                 U (المخزن في A)، SIGMA (المعطاة بالقيم المفردة المحسوبة
!>                 في SVA(1:N)) و V تفكيكًا للمصفوفة المدخلة A
!>                 بمعنى أن الباقي
!>                 ||A-SCALE*U*SIGMA*V^T||_2 / ||A||_2 صغير.
!>          إذا كان JOBU = 'N':
!>                 إذا كان INFO = 0:
!>                 لاحظ أن المتجهات المفردة اليسرى متاحة "مجانًا" في
!>                 خوارزمية SVD بجاكوبي أحادية الجانب. ومع ذلك، إذا كانت
!>                 القيم المفردة فقط هي المطلوبة، فإن مستوى التعامد العددي
!>                 لـ U ليس مصدر قلق، وتتوقف الدورات
!>                 عندما تكون أعمدة المصفوفة المتكررة
!>                 متعامدة عدديًا حتى حوالي M*EPS. وبالتالي،
!>                 عند الخروج، تحتوي A على أعمدة U مضروبة في
!>                 القيم المفردة المقابلة.
!>                 إذا كان INFO > 0:
!>                 لم يتقارب الروتين الفرعي SGESVJ في العدد المحدد
!>                 من الدورات (المرات).
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرائد للمصفوفة A. LDA >= max(1,M).
!>

SVA

!>          SVA هي مصفوفة REAL، ذات بعد (N)
!>          عند الخروج،
!>          إذا كان INFO = 0:
!>          اعتمادًا على قيمة SCALE = WORK(1)، لدينا:
!>                 إذا كان SCALE = ONE:
!>                 تحتوي SVA(1:N) على القيم المفردة المحسوبة للمصفوفة A.
!>                 أثناء الحساب، تحتوي SVA على معايير أعمدة الإقليدية
!>                 للمصفوفات المتكررة في المصفوفة A.
!>                 إذا كان SCALE .NE. ONE:
!>                 القيم المفردة للمصفوفة A هي SCALE*SVA(1:N)، وهذا
!>                 التمثيل المضروب يرجع إلى حقيقة أن بعض
!>                 القيم المفردة للمصفوفة A قد تنخفض أو تفيض.
!>
!>          إذا كان INFO > 0:
!>          لم يتقارب الروتين الفرعي SGESVJ في العدد المحدد
!>          من الدورات (المرات) وقد لا تكون SCALE*SVA(1:N) دقيقة.
!>

MV

!>          MV هو INTEGER
!>          إذا كان JOBV = 'A'، فإن حاصل ضرب دورانات جاكوبي في SGESVJ
!>          يتم تطبيقه على الصفوف MV الأولى من V. انظر وصف JOBV.
!>

V

!>          V هي مصفوفة REAL، ذات بعد (LDV,N)
!>          إذا كان JOBV = 'V'، فإن V تحتوي عند الخروج على المصفوفة N × N لـ
!>                         المتجهات المفردة اليمنى؛
!>          إذا كان JOBV = 'A'، فإن V تحتوي على حاصل ضرب مصفوفة المتجهات المفردة اليمنى المحسوبة والمصفوفة الأولية في
!>                         المصفوفة V.
!>          إذا كان JOBV = 'N'، فلا يتم الرجوع إلى V.
!>

LDV

!>          LDV هو INTEGER
!>          البعد الرائد للمصفوفة V، LDV >= 1.
!>          إذا كان JOBV = 'V'، فإن LDV >= max(1,N).
!>          إذا كان JOBV = 'A'، فإن LDV >= max(1,MV) .
!>

WORK

!>          WORK هي مصفوفة REAL، ذات بعد (LWORK)
!>          يستخدم هذا الروتين الفرعي كمصفوفة عمل.
!>          إذا كان JOBU = 'C'، فإن WORK(1) يحتوي على عتبة التعامد CTOL.
!>          عند الخروج، WORK(2) يحتوي على عدد القيم المفردة فوق عتبة الانخفاض.
!>

LWORK

!>          LWORK هو INTEGER
!>          حجم مصفوفة العمل WORK.
!>          إذا كان LWORK = -1، فسيتم حساب البعد الأدنى المطلوب لـ WORK.
!>          (يجب أن يكون LWORK >= MAX(1, M) لـ LDV = N عند استخدام JOBV = 'V' أو 'A'،
!>          أو LWORK >= MAX(1, M) لـ LDV = MV عند استخدام JOBV = 'A'.)
!>

INFO

!>          INFO هو INTEGER
!>          = 0: نجاح.
!>          > 0: فشل التقارب.
!>

sgesvj.f(3)

NAME

sgesvj - the LAPACK routine to compute the singular value decomposition (SVD) of a general matrix.

SYNOPSIS

SUBROUTINE SGESVJ( JOBZ, M, N, A, LDA, SVA, WORK, LWORK, INFO )
      LOGICAL            JOBZ
      INTEGER            M, N, LDA, LWORK, INFO
      REAL               A( LDA, N ), SVA( MIN( M, N ) )
      REAL               WORK( LWORK )

Function/Subroutine Documentation

JOBZ

 !> JOBZ is LOGICAL
 !> = .TRUE. : The Jacobi SVD algorithm will compute the vectors U and V.
 !> = .FALSE.: The Jacobi SVD algorithm will not compute the vectors U and V.

M

 !> M is INTEGER
 !> The number of rows of the matrix A. M >= 0.

N

 !> N is INTEGER
 !> The number of columns of the matrix A. N >= 0.

A

 !> A is REAL array, dimension (LDA, N)
 !> On entry, the M-by-N matrix A whose singular values and singular
 !> vectors are to be computed.
 !> On exit,
 !> If JOBZ = .TRUE., A is overwritten by the orthogonal matrix U.
 !> If JOBZ = .FALSE., the contents of A are not specified.

LDA

 !> LDA is INTEGER
 !> The leading dimension of the array A. LDA >= MAX( 1, M ).

SVA

 !> SVA is REAL array, dimension (MIN( M, N ))
 !> On exit, the singular values of A, sorted in descending order.
 !> For non-negative integers K and L, the singular values are partitioned
 !> into the two sets
 !> SVA(1:K) and SVA(K+1:M) (0 <= K <= M).
 !> SVA(1:K) contains the computed singular values that are larger than
 !> the underflow threshold.
 !> SVA(K+1:M) contains the computed singular values that are smaller than
 !> the underflow threshold.

WORK

 !>          WORK is REAL array, dimension (MAX(1,LWORK))
 !>          On entry,
 !>          If JOBU = 'C' :
 !>          WORK(1) = CTOL, where CTOL defines the threshold for convergence.
 !>                    The process stops if all columns of A are mutually
 !>                    orthogonal up to CTOL*EPS, EPS=SLAMCH('E').
 !>                    It is required that CTOL >= ONE, i.e. it is not
 !>                    allowed to force the routine to obtain orthogonality
 !>                    below EPSILON.
 !>          On exit,
 !>          WORK(1) = SCALE is the scaling factor such that SCALE*SVA(1:N)
 !>                    are the computed singular vcalues of A.
 !>                    (See description of SVA().)
 !>          WORK(2) = NINT(WORK(2)) is the number of the computed nonzero
 !>                    singular values.
 !>          WORK(3) = NINT(WORK(3)) is the number of the computed singular
 !>                    values that are larger than the underflow threshold.
 !>          WORK(4) = NINT(WORK(4)) is the number of sweeps of Jacobi
 !>                    rotations needed for numerical convergence.
 !>          WORK(5) = max_{i.NE.j} |COS(A(:,i),A(:,j))| in the last sweep.
 !>                    This is useful information in cases when SGESVJ did
 !>                    not converge, as it can be used to estimate whether
 !>                    the output is still useful and for post festum analysis.
 !>          WORK(6) = the largest absolute value over all sines of the
 !>                    Jacobi rotation angles in the last sweep. It can be
 !>                    useful for a post festum analysis.

LWORK

 !>          LWORK is INTEGER
 !>          Length of WORK.
 !>          LWORK >= 1, if MIN(M,N) = 0, and LWORK >= MAX(6,M+N), otherwise.
 !>
 !>          If on entry LWORK = -1, then a workspace query is assumed and
 !>          no computation is done; WORK(1) is set to the minial (and optimal)
 !>          length of WORK.

INFO

 !>          INFO is INTEGER
 !>          = 0:  successful exit.
 !>          < 0:  if INFO = -i, then the i-th argument had an illegal value
 !>          > 0:  SGESVJ did not converge in the maximal allowed number (30)
 !>                of sweeps. The output may still be useful. See the
 !>                description of WORK.
