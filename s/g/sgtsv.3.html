subroutine sgtsv (integer n, integer nrhs, real, dimension( * ) dl, real, dimension( * ) d, real, dimension( * ) du, real, dimension( ldb, * ) b, integer ldb, integer info)

SGTSV تحسب حل نظام المعادلات الخطية A * X = B لمصفوفات GT

الغرض:

!>
!> SGTSV  تحل المعادلة
!>
!>    A*X = B,
!>
!> حيث A مصفوفة ثلاثية قطريّة n × n، عن طريق الحذف الغاوسي مع التقسيم الجزئي.
!>
!> لاحظ أنه يمكن حل المعادلة  A**T*X = B  عن طريق تبديل ترتيب الوسيطات DU و DL.
!>

المعلمات

N 
!>          N هو عدد صحيح (INTEGER)
!>          رتبة المصفوفة A.  N >= 0.
!>

NRHS

!>          NRHS هو عدد صحيح (INTEGER)
!>          عدد الأطراف اليمنى، أي عدد الأعمدة في المصفوفة B.  NRHS >= 0.
!>

DL

!>          DL هو مصفوفة حقيقية (REAL array)، الأبعاد (N-1)
!>          عند الإدخال، يجب أن تحتوي DL على العناصر دون القطرية (n-1) للمصفوفة A.
!>
!>          عند الخروج، يتم استبدال DL بالعناصر (n-2) من القطر الفوقي الثاني للمصفوفة المثلثية العليا U من التحليل LU للمصفوفة A، في DL(1)، ..., DL(n-2).
!>

D

!>          D هو مصفوفة حقيقية (REAL array)، الأبعاد (N)
!>          عند الإدخال، يجب أن تحتوي D على العناصر القطرية للمصفوفة A.
!>
!>          عند الخروج، يتم استبدال D بالعناصر القطرية n للمصفوفة U.
!>

DU

!>          DU هو مصفوفة حقيقية (REAL array)، الأبعاد (N-1)
!>          عند الإدخال، يجب أن تحتوي DU على العناصر فوق القطرية (n-1) للمصفوفة A.
!>
!>          عند الخروج، يتم استبدال DU بالعناصر (n-1) من القطر الفوقي الأول للمصفوفة U.
!>

B

!>          B هي مصفوفة حقيقية (REAL array)، الأبعاد (LDB,NRHS)
!>          عند الإدخال، مصفوفة الطرف الأيمن B ذات الأبعاد n × NRHS.
!>          عند الخروج، إذا كان INFO = 0، فإن مصفوفة الحل X ذات الأبعاد n × NRHS.
!>

LDB

!>          LDB هو عدد صحيح (INTEGER)
!>          البعد الرائد للمصفوفة B.  LDB >= max(1,N).
!>

INFO

!>          INFO هو عدد صحيح (INTEGER)
!>          = 0: خروج ناجح
!>          < 0: إذا كان INFO = -i، فإن الوسيط i كان له قيمة غير قانونية
!>          > 0: إذا كان INFO = i، فإن U(i,i) يساوي صفر تمامًا، ولم يتم حساب الحل. لم يكتمل التحليل إلا إذا كان i = N.
!>
