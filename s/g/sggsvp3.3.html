subroutine sggsvp3 (character jobu, character jobv, character jobq, integer m, integer p, integer n, real, dimension( lda, * ) a, integer lda, real, dimension( ldb, * ) b, integer ldb, real tola, real tolb, integer k, integer l, real, dimension( ldu, * ) u, integer ldu, real, dimension( ldv, * ) v, integer ldv, real, dimension( ldq, * ) q, integer ldq, integer, dimension( * ) iwork, real, dimension( * ) tau, real, dimension( * ) work, integer lwork, integer info)

SGGSVP3

الغرض:

!>
!> SGGSVP3 يحسب المصفوفات المتعامدة U، V و Q بحيث
!>
!>                    N-K-L  K    L
!>  U**T*A*Q =     K ( 0    A12  A13 )  إذا كان M-K-L >= 0؛
!>                 L ( 0     0   A23 )
!>             M-K-L ( 0     0    0  )
!>
!>                  N-K-L  K    L
!>         =     K ( 0    A12  A13 )  إذا كان M-K-L < 0؛
!>             M-K ( 0     0   A23 )
!>
!>                  N-K-L  K    L
!>  V**T*B*Q =   L ( 0     0   B13 )
!>             P-L ( 0     0    0  )
!>
!> حيث تكون المصفوفة A12 بحجم K×K والمصفوفة B13 بحجم L×L غير منفردتين
!> مثلثيتين علويتين؛ A23 مثلثية علوية بحجم L×L إذا كان M-K-L >= 0،
!> وإلا فإن A23 تكون شبه منحرفة علوية بحجم (M-K)×L. K+L = الرتبة العددية الفعالة
!> للمصفوفة (A**T,B**T)**T التي حجمها (M+P)×N.
!>
!> هذا التحلل هو خطوة المعالجة المسبقة لحساب التحلل المتفرد المعمم (GSVD)،
!> راجع الروتين الفرعي SGGSVD3.
!>

المعلمات

JOBU 
!>          JOBU هو CHARACTER*1
!>          = 'U':  يتم حساب المصفوفة المتعامدة U؛
!>          = 'N':  لا يتم حساب U.
!>

JOBV

!>          JOBV هو CHARACTER*1
!>          = 'V':  يتم حساب المصفوفة المتعامدة V؛
!>          = 'N':  لا يتم حساب V.
!>

JOBQ

!>          JOBQ هو CHARACTER*1
!>          = 'Q':  يتم حساب المصفوفة المتعامدة Q؛
!>          = 'N':  لا يتم حساب Q.
!>

M

!>          M هو INTEGER
!>          عدد صفوف المصفوفة A. M >= 0.
!>

P

!>          P هو INTEGER
!>          عدد صفوف المصفوفة B. P >= 0.
!>

N

!>          N هو INTEGER
!>          عدد أعمدة المصفوفتين A و B. N >= 0.
!>

A

!>          A هي مصفوفة REAL، البعد (LDA,N)
!>          عند الإدخال، المصفوفة A بحجم M×N.
!>          عند الخروج، تحتوي A على المصفوفة المثلثية (أو شبه المنحرفة)
!>          الموصوفة في قسم الغرض.
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرئيسي للمصفوفة A. LDA >= max(1,M).
!>

B

!>          B هي مصفوفة REAL، البعد (LDB,N)
!>          عند الإدخال، المصفوفة B بحجم P×N.
!>          عند الخروج، تحتوي B على المصفوفة المثلثية الموصوفة في
!>          قسم الغرض.
!>

LDB

!>          LDB هو INTEGER
!>          البعد الرئيسي للمصفوفة B. LDB >= max(1,P).
!>

TOLA

!>          TOLA هو REAL
!>

TOLB

!>          TOLB هو REAL
!>
!>          TOLA و TOLB هما الحدان الأقصيان لتحديد الرتبة العددية الفعالة
!>          للمصفوفة B وللمصفوفة الفرعية A. بشكل عام،
!>          يتم تعيينهما إلى:
!>             TOLA = MAX(M,N)*norm(A)*MACHEPS,
!>             TOLB = MAX(P,N)*norm(B)*MACHEPS.
!>          قد يؤثر حجم TOLA و TOLB على حجم الأخطاء الخلفية للتحلل.
!>

K

!>          K هو INTEGER
!>

L

!>          L هو INTEGER
!>
!>          عند الخروج، يحدد K و L بعد المصفوفات الفرعية الموصوفة في
!>          قسم الغرض.
!>          K + L = الرتبة العددية الفعالة للمصفوفة (A**T,B**T)**T.
!>

U

!>          U هي مصفوفة REAL، البعد (LDU,M)
!>          إذا كان JOBU = 'U'، فإن U تحتوي على المصفوفة المتعامدة U.
!>          إذا كان JOBU = 'N'، فلا يتم الرجوع إلى U.
!>

LDU

!>          LDU هو INTEGER
!>          البعد الرئيسي للمصفوفة U. LDU >= max(1,M) إذا كان
!>          JOBU = 'U'. LDU >= 1 بخلاف ذلك.
!>

V

!>          V هي مصفوفة REAL، البعد (LDV,P)
!>          إذا كان JOBV = 'V'، فإن V تحتوي على المصفوفة المتعامدة V.
!>          إذا كان JOBV = 'N'، فلا يتم الرجوع إلى V.
!>

LDV

!>          LDV هو INTEGER
!>          البعد الرئيسي للمصفوفة V. LDV >= max(1,P) إذا كان
!>          JOBV = 'V'. LDV >= 1 بخلاف ذلك.
!>

Q

!>          Q هي مصفوفة REAL، البعد (LDQ,N)
!>          إذا كان JOBQ = 'Q'، فإن Q تحتوي على المصفوفة المتعامدة Q.
!>          إذا كان JOBQ = 'N'، فلا يتم الرجوع إلى Q.
!>

LDQ

!>          LDQ هو INTEGER
!>          البعد الرئيسي للمصفوفة Q. LDQ >= max(1,N) إذا كان
!>          JOBQ = 'Q'. LDQ >= 1 بخلاف ذلك.
!>

IWORK

!>          IWORK هي مصفوفة INTEGER، البعد (N)
!>

TAU

!>          TAU هي مصفوفة REAL، البعد (N)
!>

WORK

!>          WORK هي مصفوفة REAL، البعد (MAX(1,LWORK))
!>          عند الخروج، إذا كان INFO = 0، فإن WORK(1) تعود بالحجم الأمثل LWORK.
!>

LWORK

!>          LWORK هو INTEGER
!>          بعد المصفوفة WORK. LWORK >= 1.
!>
!>          إذا كان LWORK = -1، فإنه يُفترض استعلام عن مساحة العمل؛ يقوم الروتين
!>          فقط بحساب الحجم الأمثل لمصفوفة WORK، ويعيد هذا الحجم كإدخال أول
!>          لمصفوفة WORK، ولا يتم إصدار أي رسالة خطأ متعلقة بـ LWORK بواسطة XERBLA.
!>

INFO

!>          INFO هو INTEGER
!>          = 0:  خروج ناجح
!>          < 0:  إذا كان INFO = -i، فإن الوسيط i كان له قيمة غير قانونية.
!>
