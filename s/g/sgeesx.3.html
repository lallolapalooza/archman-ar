توثيق الوظيفة/الروتين الفرعي

subroutine sgeesx (character jobvs, character sort, external select, character sense, integer n, real, dimension( lda, * ) a, integer lda, integer sdim, real, dimension( * ) wr, real, dimension( * ) wi, real, dimension( ldvs, * ) vs, integer ldvs, real rconde, real rcondv, real, dimension( * ) work, integer lwork, integer, dimension( * ) iwork, integer liwork, logical, dimension( * ) bwork, integer info)

SGEESX تحسب القيم الذاتية، والصيغة شتور، واختيارياً، مصفوفة متجهات شتور لمصفوفات GE

الغرض:

!>
!> SGEESX تحسب لمصفوفة A الحقيقية غير المتماثلة ذات البعد N×N، القيم الذاتية،
!> والصيغة شتور الحقيقية T، واختيارياً، مصفوفة متجهات شتور Z. هذا يعطي
!> تحليل شتور A = Z*T*(Z**T).
!>
!> اختيارياً، تقوم أيضًا بترتيب القيم الذاتية على القطر للصيغة شتور
!> بحيث تكون القيم الذاتية المختارة في أعلى اليسار؛
!> تحسب رقم شرط معكوس لمتوسط القيم الذاتية المختارة (RCONDE)؛ وتحسب رقم شرط معكوس
!> للفضاء الجزئي الأيمن المناظر للقيم الذاتية المختارة (RCONDV).
!> تشكل الأعمدة الرائدة لـ Z أساسًا متعامدًا لهذا الفضاء الجزئي.
!>
!> لمزيد من الشرح حول أرقام الشرط المعكوس RCONDE و RCONDV، انظر القسم 4.10
!> من دليل مستخدمي LAPACK (حيث تسمى هذه الكميات s و sep على التوالي).
!>
!> المصفوفة الحقيقية تكون في صيغة شتور الحقيقية إذا كانت شبه مثلثة عليا
!> مع كتل 1×1 و 2×2. سيتم توحيد الكتل 2×2 في الشكل:
!>           [  a  b  ]
!>           [  c  a  ]
!>
!> حيث b*c < 0. القيم الذاتية لهذه الكتلة هي a +- sqrt(bc).
!>

المعلمات

JOBVS 
!>          JOBVS هو CHARACTER*1
!>          = 'N': متجهات شتور غير محسوبة؛
!>          = 'V': متجهات شتور محسوبة.
!>

SORT

!>          SORT هو CHARACTER*1
!>          يحدد ما إذا كان سيتم ترتيب القيم الذاتية على قطر الصيغة شتور أم لا.
!>          = 'N': القيم الذاتية غير مرتبة؛
!>          = 'S': القيم الذاتية مرتبة (انظر SELECT).
!>

SELECT

!>          SELECT هي وظيفة LOGICAL وسيطتين REAL
!>          يجب تعريف SELECT كـ EXTERNAL في الروتين الفرعي المستدعي.
!>          إذا كان SORT = 'S'، يتم استخدام SELECT لفرز القيم الذاتية
!>          إلى أعلى اليسار من الصيغة شتور.
!>          إذا كان SORT = 'N'، لا يتم الرجوع إلى SELECT.
!>          يتم اختيار قيمة ذاتية WR(j)+sqrt(-1)*WI(j) إذا كانت
!>          SELECT(WR(j),WI(j)) صحيحة؛ أي، إذا تم اختيار إحدى قيمتين ذاتيتين
!>          مترافقتين، فسيتم اختيار كلتيهما. لاحظ أن قيمة ذاتية مختارة
!>          معقدة قد لا تحقق SELECT(WR(j),WI(j)) = .TRUE. بعد الترتيب،
!>          حيث أن الترتيب قد يغير قيمة القيم الذاتية المعقدة
!>          (خاصة إذا كانت القيمة الذاتية سيئة التكييف)؛ في هذه الحالة
!>          قد يتم تعيين INFO إلى N+3 (انظر INFO أدناه).
!>

SENSE

!>          SENSE هو CHARACTER*1
!>          يحدد أرقام الشرط المعكوس التي تم حسابها.
!>          = 'N': لا شيء محسوب؛
!>          = 'E': محسوب لمتوسط القيم الذاتية المختارة فقط؛
!>          = 'V': محسوب للفضاء الجزئي الأيمن المختار فقط؛
!>          = 'B': محسوب لكليهما.
!>          إذا كان SENSE = 'E'، 'V' أو 'B'، يجب أن يساوي SORT 'S'.
!>

N

!>          N هو INTEGER
!>          رتبة المصفوفة A. N >= 0.
!>

A

!>          A هي مصفوفة REAL، البعد (LDA, N)
!>          عند الإدخال، المصفوفة A ذات البعد N×N.
!>          عند الخروج، يتم استبدال A بصيغة شتور T الخاصة بها.
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرئيسي للمصفوفة A. LDA >= max(1,N).
!>

SDIM

!>          SDIM هو INTEGER
!>          إذا كان SORT = 'N'، فإن SDIM = 0.
!>          إذا كان SORT = 'S'، فإن SDIM = عدد القيم الذاتية (بعد الترتيب)
!>                         التي تكون فيها SELECT صحيحة. (الزوج المترافق
!>                         المعقدة التي تكون فيها SELECT صحيحة لأي من
!>                         القيمتين الذاتيتين تعتبر 2.)
!>

WR

!>          WR هي مصفوفة REAL، البعد (N)
!>

WI

!>          WI هي مصفوفة REAL، البعد (N)
!>          تحتوي WR و WI على الأجزاء الحقيقية والتخيلية، على التوالي،
!>          للقيم الذاتية المحسوبة، بنفس الترتيب الذي تظهر به على قطر
!>          صيغة شتور الناتجة T. تظهر أزواج القيم الذاتية المترافقة
!>          المعقدة بشكل متتابع مع القيمة الذاتية ذات الجزء التخيلي الموجب أولاً.
!>

VS

!>          VS هي مصفوفة REAL، البعد (LDVS,N)
!>          إذا كان JOBVS = 'V'، تحتوي VS على المصفوفة المتعامدة Z لمتجهات شتور.
!>          إذا كان JOBVS = 'N'، لا يتم الرجوع إلى VS.
!>

LDVS

!>          LDVS هو INTEGER
!>          البعد الرئيسي للمصفوفة VS. LDVS >= 1، وإذا كان
!>          JOBVS = 'V'، فإن LDVS >= N.
!>

RCONDE

!>          RCONDE هي REAL
!>          إذا كان SENSE = 'E' أو 'B'، تحتوي RCONDE على رقم الشرط
!>          المعكوس لمتوسط القيم الذاتية المختارة.
!>          لا يتم الرجوع إليها إذا كان SENSE = 'N' أو 'V'.
!>

RCONDV

!>          RCONDV هي REAL
!>          إذا كان SENSE = 'V' أو 'B'، تحتوي RCONDV على رقم الشرط
!>          المعكوس للفضاء الجزئي الأيمن المختار.
!>          لا يتم الرجوع إليها إذا كان SENSE = 'N' أو 'E'.
!>

WORK

!>          WORK هي مصفوفة REAL، البعد (MAX(1,LWORK))
!>          عند الخروج، إذا كان INFO = 0، فإن WORK(1) تعيد LWORK المثلى.
!>

LWORK

!>          LWORK هو INTEGER
!>          بعد المصفوفة WORK. LWORK >= max(1,3*N).
!>          أيضًا، إذا كان SENSE = 'E' أو 'V' أو 'B'،
!>          LWORK >= N+2*SDIM*(N-SDIM)، حيث SDIM هو عدد
!>          القيم الذاتية المختارة التي تم حسابها بواسطة هذا الروتين.
!>          لاحظ أن N+2*SDIM*(N-SDIM) <= N+N*N/2. لاحظ أيضًا أنه يتم إرجاع خطأ
!>          فقط إذا كان LWORK < max(1,3*N)، ولكن إذا كان SENSE = 'E' أو 'V' أو
!>          'B' فقد لا يكون هذا كافيًا.
!>          لأداء جيد، يجب أن يكون LWORK أكبر بشكل عام.
!>
!>          إذا كان LWORK = -1، فسيتم افتراض استعلام مساحة العمل؛ الروتين
!>          يحسب فقط الحدود العليا للأحجام المثلى للمصفوفات
!>          WORK و IWORK، ويعيد هذه القيم كمدخلات أولى
!>          لمصفوفتي WORK و IWORK، ولا تصدر رسائل خطأ
!>          متعلقة بـ LWORK أو LIWORK بواسطة XERBLA.
!>

IWORK

!>          IWORK هي مصفوفة INTEGER، البعد (MAX(1,LIWORK))
!>          عند الخروج، إذا كان INFO = 0، فإن IWORK(1) تعيد LIWORK المثلى.
!>

LIWORK

!>          LIWORK هو INTEGER
!>          بعد المصفوفة IWORK.
!>          LIWORK >= 1؛ إذا كان SENSE = 'V' أو 'B'، فإن LIWORK >= SDIM*(N-SDIM).
!>          لاحظ أن SDIM*(N-SDIM) <= N*N/4. لاحظ أيضًا أنه يتم إرجاع خطأ
!>          فقط إذا كان LIWORK < 1، ولكن إذا كان SENSE = 'V' أو 'B' فقد لا يكون
!>          هذا كافيًا.
!>
!>          إذا كان LIWORK = -1، فسيتم افتراض استعلام مساحة العمل؛
!>

هذا الموقع غير مرتبط بـ Arch Linux ولا تابع له بأي شكل من الأشكال. المحتوى هو ترجمة فقط لصفحات دليل Arch المنشورة على man.archlinux.org.

يقتصر الروتين على حساب الحدود العليا للأحجام المثلى للمصفوفات WORK و IWORK، ويعيد هذه القيم كمدخلات أولى للمصفوفات WORK و IWORK، ولا تصدر أي رسائل خطأ متعلقة بـ LWORK أو LIWORK بواسطة XERBLA. !>

BWORK

!>          BWORK هي مصفوفة LOGICAL، ذات بعد (N)
!>          لا تتم الإشارة إليها إذا كان SORT = 'N'.
!>

INFO

!>          INFO هو عدد صحيح (INTEGER)
!>          = 0: خروج ناجح
!>          < 0: إذا كان INFO = -i، فإن الوسيط i كان له قيمة غير قانونية.
!>          > 0: إذا كان INFO = i، و i هو
!>             <= N: فشلت خوارزمية QR في حساب جميع القيم الذاتية؛ العناصر 1:ILO-1 و i+1:N من WR و WI
!>                   تحتوي على تلك القيم الذاتية التي تقاربت؛ إذا كان
!>                   JOBVS = 'V'، فإن VS تحتوي على التحويل الذي
!>                   يقلل A إلى شكله الجزئي المتقارب من Schur.
!>             = N+1: لم يكن من الممكن إعادة ترتيب القيم الذاتية لأن بعض
!>                   القيم الذاتية كانت قريبة جدًا من الفصل (المشكلة
!>                   غير مستقرة للغاية)؛
!>             = N+2: بعد إعادة الترتيب، قامت تقريبات التقريب بتغيير قيم بعض
!>                   القيم الذاتية المعقدة بحيث لم تعد القيم الذاتية الرائدة في
!>                   شكل Schur تحقق SELECT=.TRUE. يمكن أن يكون هذا أيضًا
!>                   ناجمًا عن التقريب بسبب القياس.
!>
