الاسم

sg_comp_init, sg_comp_destroy, sg_comp_get_tls, sg_global_lock, sg_global_unlock - إدارة تسليم إحصائيات النظام

الاستخدام

#include "tools.h"


void *get_global_static
(unsigned int id);


sg_error sg_global_lock
(void);


sg_error sg_global_unlock
(void);

الوصف

يتم استدعاء sg_comp_init() بواسطة sg_init() لتشغيل التهيئة لمكون العولمة ولكل مكون libstatgrab مسجل. يتم هذا التسجيل بشكل ثابت عن طريق إلحاق بنية معلومات تهيئة المكون (نسخة من sg_comp_info) بقائمة comp_info في src/libstatgrab/globals.c. عادةً ما يتم تعريف نسخة sg_comp_info باستخدام إما EASY_COMP_SETUP() أو EXTENDED_COMP_SETUP() على التوالي.

يتم استدعاء sg_comp_destroy() بواسطة sg_shutdown() لتدمير جميع الموارد العامة، على سبيل المثال، جداول الترجمة لأسماء الأجهزة أو التعبيرات العادية المترجمة لمطابقة أسماء المسارات وما إلى ذلك. تتم إدارة موارد الإحصائيات في مكان آخر ويتم تحريرها (مفاهيميًا وعادةً) قبل استدعاء sg_comp_destroy().

sg_comp_get_tls() هي الدالة الداخلية للوصول إلى تخزين المكون المحلي الخاص بالخيط (المتغيرات العامة الثابتة سابقًا). عادةً ما يكون من الأسهل الاعتماد على الماكرو المغلف GLOBAL_GET_TLS().

ملاحظات

توصيل إحصائيات النظام هو مهمة libstatgrab، وإدارة التسليم هي مهمة مكون العولمة. للوفاء بهذه المهمة، يجب تجهيز المكونات المراد إدارتها:

1.
تعريف بنية بيانات المكون العالمية والمحلية الخاصة بالخيط (ربما فقط نظريًا، وليس في الكود)
2.
تعريف دوال تهيئة العولمة، وتدمير الخيط، وتدمير العملية (إذا لزم الأمر بواسطة 1.)
3.
تعريف بنية معلومات المكون باستخدام *_COMP_SETUP()
4.
تعريف وحدات الوصول للمكون باستخدام واحدة أو أكثر من

EASY_COMP_ACCESS()
EASY_COMP_DIFF()
MULTI_COMP_ACCESS()
MULTI_COMP_DIFF()

عند الانتهاء من هذه الخطوات، يولد مكون جديد يقدم إحصائيات جديدة ويحتاج إلى "الإعلان" عنه. بافتراض أن اسم المكون هو cpu، قم بإلحاق السطر { &sg_cpu_init, 0 } بقائمة comp_info المسماة أعلاه.

معلومات تهيئة المكون بالتفصيل:

typedef sg_error (*comp_global_init_function)(unsigned id);
typedef void (*comp_global_destroy_function)(void);
typedef void (*comp_global_cleanup_function)(void *);
struct sg_comp_status {
        sg_error init_error;
};
struct sg_comp_init {
        comp_global_init_function init_fn;
        comp_global_destroy_function destroy_fn;
        comp_global_cleanup_function cleanup_fn;
        size_t static_buf_size;
#if defined(ENABLE_THREADS) && defined(HAVE_PTHREAD)
        const char **required_locks;
#endif
        struct sg_comp_status *status;
};

المكونات التي لا تحتاج إلى شيء خاص يمكنها الاعتماد على EASY_COMP_SETUP():

تهيئة مكون الذاكرة

EASY_COMP_SETUP(mem,1,NULL);

عند الحاجة إلى تهيئة خاصة، فإن القيام بذلك أكثر تعقيدًا بعض الشيء:

يهتم EASY_COMP_SETUP() بكل ما يلزم تلقائيًا لإنشاء بنية معلومات مكون للمكون comp المحدد. سيحتوي تخزين TLS المنشأ على nvect عنصر مؤشر وهذا كل شيء. يتم إنشاء جميع روتينات التهيئة والتدمير والتنظيف حسب الحاجة باستخدام EASY_COMP_INIT_FN() و EASY_COMP_DESTROY_FN() و EASY_COMP_CLEANUP_FN(). بعد مقدار مؤشرات المتجه المطلوبة للتخزين، يجب تحديد قائمة الأقفال المطلوبة، مع الانتهاء بمؤشر NULL.

يهتم EXTENDED_COMP_SETUP() بكل ما يلزم تلقائيًا لإنشاء بنية معلومات مكون للمكون comp المحدد، ولكن مع التعريف المطلوب لروتينات التهيئة والتدمير والتنظيف. سيحتوي تخزين TLS المنشأ على nvect عنصر مؤشر وهذا كل شيء. بعد مقدار مؤشرات المتجه المطلوبة للتخزين، يجب تحديد قائمة الأقفال المطلوبة، مع الانتهاء بمؤشر NULL. يمكن إنشاء جميع الروتينات القياسية شبه تلقائيًا باستخدام EASY_COMP_INIT_FN() و EASY_COMP_DESTROY_FN() و EASY_COMP_CLEANUP_FN().

يقوم DEFAULT_INIT_COMP() فقط بتعريف النماذج الأولية لروتينات التهيئة والتدمير والتنظيف، ويحدد مخزن مؤقت لحالة التهيئة، وقائمة بأسماء الأقفال، وأخيراً يملأ بنية تهيئة المكون. استخدم هذا عندما يحتوي تخزين TLS الخاص بك على أكثر من مجرد مؤشرات متجه.

GLOBAL_GET_TLS() يعيد المؤشر إلى تخزين المكون المحلي الخاص بالخيط.

GLOBAL_SET_ID() يخزن معرف المكون، وهو مطلوب على سبيل المثال للوصول إلى TLS الخاص به.

يعرّف EASY_COMP_INIT_FN() روتين تهيئة مكون افتراضي. يقوم بتخزين معرف المكون ويعود بـ SG_ERROR_NONE.

يعرّف EASY_COMP_DESTROY_FN() مدمر مكون افتراضي، يتم استدعاؤه في نهاية العملية بأكملها (أو عند استدعاء sg_shutdown() الأخير). المدمر الافتراضي لا يفعل شيئًا وعادةً ما يتطلب روتين تهيئة فردي مدمرًا فرديًا أيضًا.

يعرّف EASY_COMP_CLEANUP_FN() روتين تنظيف TLS افتراضي، يتم استدعاؤه دائمًا عند انتهاء خيط لتحرير المتجهات الموجودة في تخزين المحلي الخاص بالخيط.

يعرّف EASY_COMP_ACCESS() وحدات وصول إلى إحصائية معينة تحتوي على عنصر واحد يقدمه المكون: الدالتان fn() و fn##_r(). يجب أن تكون الدالة التالية موجودة:
sg_error fn##_int
(sg_vector *name##_vect);
تقوم بالوصول إلى المتجه idx من TLS للمكون comp وتعيد

هذا الموقع غير مرتبط بـ Arch Linux ولا تابع له بأي شكل من الأشكال. المحتوى هو ترجمة فقط لصفحات دليل Arch المنشورة على man.archlinux.org.

الاسم

sg_global_lock - آليات القفل العامة لـ libstatgrab

ملخص

#include <statgrab.h>

void sg_global_lock();
void sg_global_unlock();
void sg_global_rwlock_rdlock();
void sg_global_rwlock_wrlock();
void sg_global_rwlock_unlock();

الوصف

تمتلك libstatgrab قفلًا عامًا أو قفل قراءة/كتابة للوصول المتزامن الآمن إلى بياناتها.

تمنح الدالة sg_global_lock() القفل العام (الكتابة)، وتمنعه الدالة sg_global_unlock().

تمنح الدالة sg_global_rwlock_rdlock() قفل القراءة، وتمنحه الدالة sg_global_rwlock_wrlock() قفل الكتابة. يسمح قفل القراءة لعدة قراء بالوصول في نفس الوقت، بينما يتطلب قفل الكتابة وصولًا حصريًا.

تقوم الدالة sg_global_rwlock_unlock() بإلغاء قفل القراءة أو الكتابة.

ملاحظات

يجب استخدام هذه الدوال فقط من قبل المطورين الذين يقومون بتوسيع libstatgrab.

EASY_COMP_DIFF() و MULTI_COMP_DIFF() هي وحدات ماكرو تساعد في إدارة إحصائيات المكون.

EASY_COMP_DIFF() تُرجع الفرق بين تشغيلين لجمع الإحصائيات. النسخة التي تتعامل مع إحصائيات خاصة بـ statgrab تُرجع الفرق بين المحتوى الحالي في المتجه المحدد بواسطة nowidx والمتجه الناتج من getfn(). يتم تخزين النتيجة في المتجه diffidx. إذا لم تكن هناك نتيجة حالية، فسيتم ببساطة إرجاع نتيجة getfn().

MULTI_COMP_ACCESS() تحدد الوصول إلى إحصائية معينة تحتوي على 0..n عنصر مقدمة من المكون: الدوال fn() و fn##_r(). يجب أن توجد الدالة التالية:
sg_error fn##_int
(sg_vector **name##_vect);
تقوم بالوصول إلى فهرس المتجه من TLS الخاص بالمكون comp وترجع sg_##name##_stats. تدير جميع الأشياء القياسية مثل إدارة الذاكرة والأخطاء، قيم الإرجاع، إلخ.

MULTI_COMP_DIFF() تفعل نفس الشيء الذي تفعله EASY_COMP_DIFF() ولكن للمتجهات التي تحتوي على أكثر من عنصر واحد.
