subroutine sgsvts3 (integer m, integer p, integer n, real, dimension( lda, * ) a, real, dimension( lda, * ) af, integer lda, real, dimension( ldb, * ) b, real, dimension( ldb, * ) bf, integer ldb, real, dimension( ldu, * ) u, integer ldu, real, dimension( ldv, * ) v, integer ldv, real, dimension( ldq, * ) q, integer ldq, real, dimension( * ) alpha, real, dimension( * ) beta, real, dimension( ldr, * ) r, integer ldr, integer, dimension( * ) iwork, real, dimension( lwork ) work, integer lwork, real, dimension( * ) rwork, real, dimension( 6 ) result)

SGSVTS3

الغرض:

!>
!> SGSVTS3 تختبر SGGSVD3، الذي يحسب GSVD للمصفوفة A ذات الأبعاد M*N
!> والمصفوفة B ذات الأبعاد P*N:
!>              U'*A*Q = D1*R و V'*B*Q = D2*R.
!>

المعلمات

M 
!>          M هو عدد صحيح
!>          عدد صفوف المصفوفة A. M >= 0.
!>

P

!>          P هو عدد صحيح
!>          عدد صفوف المصفوفة B. P >= 0.
!>

N

!>          N هو عدد صحيح
!>          عدد أعمدة المصفوفتين A و B. N >= 0.
!>

A

!>          A هي مصفوفة حقيقية، أبعادها (LDA,M)
!>          المصفوفة A ذات الأبعاد M*N.
!>

AF

!>          AF هي مصفوفة حقيقية، أبعادها (LDA,N)
!>          تفاصيل GSVD للمصفوفتين A و B، كما تم إرجاعها بواسطة SGGSVD3،
!>          انظر SGGSVD3 لمزيد من التفاصيل.
!>

LDA

!>          LDA هو عدد صحيح
!>          البعد الرئيسي للمصفوفات A و AF.
!>          LDA >= max( 1,M ).
!>

B

!>          B هي مصفوفة حقيقية، أبعادها (LDB,P)
!>          عند الإدخال، المصفوفة B ذات الأبعاد P*N.
!>

BF

!>          BF هي مصفوفة حقيقية، أبعادها (LDB,N)
!>          تفاصيل GSVD للمصفوفتين A و B، كما تم إرجاعها بواسطة SGGSVD3،
!>          انظر SGGSVD3 لمزيد من التفاصيل.
!>

LDB

!>          LDB هو عدد صحيح
!>          البعد الرئيسي للمصفوفتين B و BF.
!>          LDB >= max(1,P).
!>

U

!>          U هي مصفوفة حقيقية، أبعادها (LDU,M)
!>          المصفوفة المتعامدة U ذات الأبعاد M*M.
!>

LDU

!>          LDU هو عدد صحيح
!>          البعد الرئيسي للمصفوفة U. LDU >= max(1,M).
!>

V

!>          V هي مصفوفة حقيقية، أبعادها (LDV,M)
!>          المصفوفة المتعامدة V ذات الأبعاد P*P.
!>

LDV

!>          LDV هو عدد صحيح
!>          البعد الرئيسي للمصفوفة V. LDV >= max(1,P).
!>

Q

!>          Q هي مصفوفة حقيقية، أبعادها (LDQ,N)
!>          المصفوفة المتعامدة Q ذات الأبعاد N*N.
!>

LDQ

!>          LDQ هو عدد صحيح
!>          البعد الرئيسي للمصفوفة Q. LDQ >= max(1,N).
!>

ALPHA

!>          ALPHA هي مصفوفة حقيقية، أبعادها (N)
!>

BETA

!>          BETA هي مصفوفة حقيقية، أبعادها (N)
!>
!>          أزواج القيم المفردة المعممة للمصفوفتين A و B،
!>          المصفوفات "القطرية" D1 و D2 يتم بناؤها من
!>          ALPHA و BETA، انظر الإجراء الفرعي SGGSVD3 للحصول على التفاصيل.
!>

R

!>          R هي مصفوفة حقيقية، أبعادها (LDQ,N)
!>          المصفوفة المثلثية العليا R.
!>

LDR

!>          LDR هو عدد صحيح
!>          البعد الرئيسي للمصفوفة R. LDR >= max(1,N).
!>

IWORK

!>          IWORK هي مصفوفة أعداد صحيحة، أبعادها (N)
!>

WORK

!>          WORK هي مصفوفة حقيقية، أبعادها (LWORK)
!>

LWORK

!>          LWORK هو عدد صحيح
!>          بعد المصفوفة WORK،
!>          LWORK >= max(M,P,N)*max(M,P,N).
!>

RWORK

!>          RWORK هي مصفوفة حقيقية، أبعادها (max(M,P,N))
!>

RESULT

!>          RESULT هي مصفوفة حقيقية، أبعادها (6)
!>          نسب الاختبار:
!>          RESULT(1) = norm( U'*A*Q - D1*R ) / ( MAX(M,N)*norm(A)*ULP)
!>          RESULT(2) = norm( V'*B*Q - D2*R ) / ( MAX(P,N)*norm(B)*ULP)
!>          RESULT(3) = norm( I - U'*U ) / ( M*ULP )
!>          RESULT(4) = norm( I - V'*V ) / ( P*ULP )
!>          RESULT(5) = norm( I - Q'*Q ) / ( N*ULP )
!>          RESULT(6) = 0        إذا كان ALPHA في ترتيب تنازلي؛
!>                    = ULPINV   بخلاف ذلك.
!>
