subroutine sgemqr (character side, character trans, integer m, integer n, integer k, real, dimension( lda, * ) a, integer lda, real, dimension( * ) t, integer tsize, real, dimension( ldc, * ) c, integer ldc, real, dimension( * ) work, integer lwork, integer info)

SGEMQR

الغرض:

!>
!> SGEMQR يستبدل المصفوفة C الحقيقية العامة M-by-N بـ
!>
!>                      SIDE = 'L'     SIDE = 'R'
!>     TRANS = 'N':      Q * C          C * Q
!>     TRANS = 'T':      Q**T * C       C * Q**T
!>
!> حيث Q هي مصفوفة متعامدة حقيقية معرفة كناتج
!> عوامل انعكاس عنصرية مجمعة تم حسابها بواسطة التحلل القطري الضيق (tall skinny)
!> QR (SGEQR)
!>
!>

المعاملات

SIDE 
!>          SIDE is CHARACTER*1
!>          = 'L': تطبيق Q أو Q**T من اليسار؛
!>          = 'R': تطبيق Q أو Q**T من اليمين.
!>

TRANS

!>          TRANS is CHARACTER*1
!>          = 'N':  بدون تبديل، تطبيق Q؛
!>          = 'T':  تبديل، تطبيق Q**T.
!>

M

!>          M is INTEGER
!>          عدد صفوف المصفوفة A.  M >=0.
!>

N

!>          N is INTEGER
!>          عدد أعمدة المصفوفة C. N >= 0.
!>

K

!>          K is INTEGER
!>          عدد عوامل الانعكاس الأولية التي تحدد حاصل الضرب
!>          بالمصفوفة Q.
!>          إذا كانت SIDE = 'L', M >= K >= 0؛
!>          إذا كانت SIDE = 'R', N >= K >= 0.
!>

A

!>          A is REAL array, dimension (LDA,K)
!>          جزء من بنية البيانات لتمثيل Q كما تم إرجاعه بواسطة SGEQR.
!>

LDA

!>          LDA is INTEGER
!>          البعد الرائد للمصفوفة A.
!>          إذا كانت SIDE = 'L', LDA >= max(1,M)؛
!>          إذا كانت SIDE = 'R', LDA >= max(1,N).
!>

T

!>          T is REAL array, dimension (MAX(5,TSIZE)).
!>          جزء من بنية البيانات لتمثيل Q كما تم إرجاعه بواسطة SGEQR.
!>

TSIZE

!>          TSIZE is INTEGER
!>          بعد المصفوفة T. TSIZE >= 5.
!>

C

!>          C is REAL array, dimension (LDC,N)
!>          عند الإدخال، المصفوفة C بالحجم M-by-N.
!>          عند الخروج، يتم استبدال C بـ Q*C أو Q**T*C أو C*Q**T أو C*Q.
!>

LDC

!>          LDC is INTEGER
!>          البعد الرائد للمصفوفة C. LDC >= max(1,M).
!>

WORK

!>         (مساحة عمل) REAL array, dimension (MAX(1,LWORK))
!>

LWORK

!>          LWORK is INTEGER
!>          بعد المصفوفة WORK.
!>          إذا كانت LWORK = -1، فسيتم افتراض استعلام مساحة العمل. تقوم الوظيفة
!>          بحساب حجم المصفوفة WORK فقط، وتُرجع هذا
!>          القيمة كـ WORK(1)، ولا يتم إصدار أي رسالة خطأ متعلقة بـ WORK
!>          بواسطة XERBLA.
!>

INFO

!>          INFO is INTEGER
!>          = 0:  انتهاء ناجح
!>          < 0:  إذا كانت INFO = -i، فإن المعامل i كان له قيمة غير قانونية
!>

تفاصيل إضافية

!>
!> هذه التفاصيل خاصة بتنفيذ LAPACK هذا. لا ينبغي على المستخدمين
!> أخذها كأمر واقع. قد تتغير هذه التفاصيل في المستقبل، ومن غير المرجح
!> أن تكون صحيحة لتنفيذ LAPACK آخر. هذه التفاصيل ذات صلة إذا أراد
!> أحدهم محاولة فهم الكود. وهي ليست جزءًا من الواجهة.
!>
!> في هذا الإصدار،
!>
!>          T(2): حجم كتلة الصف (MB)
!>          T(3): حجم كتلة العمود (NB)
!>          T(6:TSIZE): بنية البيانات اللازمة لـ Q، محسوبة بواسطة
!>                           SLATSQR أو SGEQRT
!>
!>  اعتمادًا على أبعاد المصفوفة M و N، وأحجام الكتل للصفوف والأعمدة
!>  MB و NB التي تم إرجاعها بواسطة ILAENV، ستستخدم SGEQR إما
!>  SLATSQR (إذا كانت المصفوفة طويلة وضيقة) أو SGEQRT لحساب
!>  التحلل القطري.
!>  سيستخدم هذا الإصدار من SGEMQR إما SLAMTSQR أو SGEMQRT لـ
!>  ضرب المصفوفة Q بمصفوفة أخرى.
!>  تفاصيل إضافية في SLAMTSQR أو SGEMQRT.
!>
!>
