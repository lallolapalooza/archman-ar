subroutine sgesvxx (fact, trans, n, nrhs, a, lda, af, ldaf, ipiv, equed, r, c, b, ldb, x, ldx, rcond, rpvgrw, berr, n_err_bnds, err_bnds_norm, err_bnds_comp, nparams, params, work, iwork, info)

SGESVXX تحسب حل نظام المعادلات الخطية A * X = B لمصفوفات GE

الغرض:

!>
!>    تستخدم SGESVXX تحليل LU لحساب حل نظام المعادلات الخطية الحقيقي A * X = B،
!>    حيث A مصفوفة N*N و X و B مصفوفتان N*NRHS.
!>
!>    إذا طُلب، يتم إرجاع حدود خطأ معيارية وخطأ مكوني أقصى.
!>    ستُرجع SGESVXX حلاً بخطأ صغير مضمون (O(eps) حيث eps هو دقة الآلة العاملة)
!>    ما لم تكن المصفوفة سيئة التكييف للغاية، وفي هذه الحالة يتم إرجاع تحذير.
!>    يتم أيضًا حساب أرقام التكييف ذات الصلة وإرجاعها.
!>
!>    تقبل SGESVXX التحليلات وعوامل المعايرة المقدمة من المستخدم؛ انظر
!>    تعريفات خيارات FACT و EQUED.
!>    سيؤدي الحل مع التنقيح واستخدام تحليل من استدعاء SGESVXX سابق أيضًا
!>    إلى حل بخطأ O(eps) أو تحذيرات، ولكن لا يمكننا المطالبة بذلك للتحليلات
!>    وعوامل المعايرة المقدمة من المستخدم بشكل عام إذا اختلفت عن ما
!>    تنتجه SGESVXX بنفسها.
!>

الوصف:

!>
!>    يتم تنفيذ الخطوات التالية:
!>
!>    1. إذا كان FACT = 'E'، يتم حساب عوامل قياس حقيقية لمعايرة النظام:
!>
!>      TRANS = 'N':  diag(R)*A*diag(C)     *inv(diag(C))*X = diag(R)*B
!>      TRANS = 'T': (diag(R)*A*diag(C))**T *inv(diag(R))*X = diag(C)*B
!>      TRANS = 'C': (diag(R)*A*diag(C))**H *inv(diag(R))*X = diag(C)*B
!>
!>    يعتمد ما إذا كان سيتم معايرة النظام أم لا على قياس المصفوفة A،
!>    ولكن إذا تم استخدام المعايرة، يتم الكتابة فوق A بواسطة diag(R)*A*diag(C)
!>    و B بواسطة diag(R)*B (إذا كان TRANS='N') أو diag(C)*B (إذا كان TRANS = 'T' أو 'C').
!>
!>    2. إذا كان FACT = 'N' أو 'E'، يتم استخدام تحليل LU لتحليل
!>    المصفوفة A (بعد المعايرة إذا كان FACT = 'E') كما يلي:
!>
!>      A = P * L * U,
!>
!>    حيث P هي مصفوفة تبديل، L هي مصفوفة مثلثة سفلية أحادية، و U هي مصفوفة مثلثة عليا.
!>
!>    3. إذا كان U(i,i)=0 لبعض i، مما يجعل U مفردة تمامًا، فإن
!>    الوحدة الفرعية تُرجع بـ INFO = i. بخلاف ذلك، يتم استخدام الشكل المحلل لـ A
!>    لتقدير رقم تكييف المصفوفة A (انظر الوسيط RCOND). إذا كان مقلوب
!>    رقم التكييف أقل من دقة الآلة، فإن الوحدة الفرعية لا تزال تستمر في الحل لـ X
!>    وحساب حدود الخطأ كما هو موضح أدناه.
!>
!>    4. يتم حل نظام المعادلات لـ X باستخدام الشكل المحلل لـ A.
!>
!>    5. افتراضيًا (ما لم يتم تعيين PARAMS(LA_LINRX_ITREF_I) إلى صفر)،
!>    ستستخدم الوحدة الفرعية التنقيح التكراري لمحاولة الحصول على خطأ صغير
!>    وحدود خطأ. يقوم التنقيح بحساب الباقي بدقة مرتين على الأقل من دقة العمل.
!>
!>    6. إذا تم استخدام المعايرة، يتم ضرب المصفوفة X مسبقًا بواسطة
!>    diag(C) (إذا كان TRANS = 'N') أو diag(R) (إذا كان TRANS = 'T' أو 'C')
!>    بحيث تحل النظام الأصلي قبل المعايرة.
!> 
!>     يتم تجميع بعض المعلمات الاختيارية في مصفوفة PARAMS. هذه الإعدادات
!>     تحدد كيفية إجراء التنقيح، ولكن في كثير من الأحيان تكون الإعدادات الافتراضية مقبولة.
!>     إذا كانت الإعدادات الافتراضية مقبولة، يمكن للمستخدمين تمرير NPARAMS = 0
!>     مما يمنع الشفرة المصدرية من الوصول إلى الوسيط PARAMS.
!>

المعلمات

FACT 
!>          FACT هو CHARACTER*1
!>     يحدد ما إذا كان الشكل المحلل للمصفوفة A مزودًا عند الإدخال، وإذا لم يكن الأمر كذلك،
!>     فهل يجب معايرة المصفوفة A قبل تحليلها.
!>       = 'F':  عند الإدخال، تحتوي AF و IPIV على الشكل المحلل لـ A.
!>               إذا لم يكن EQUED = 'N'، فقد تم معايرة المصفوفة A
!>               بعوامل قياس معطاة بواسطة R و C.
!>               لا يتم تعديل A، AF، و IPIV.
!>       = 'N':  سيتم نسخ المصفوفة A إلى AF وتحليلها.
!>       = 'E':  سيتم معايرة المصفوفة A إذا لزم الأمر، ثم
!>               نسخها إلى AF وتحليلها.
!>

TRANS

!>          TRANS هو CHARACTER*1
!>     يحدد شكل نظام المعادلات:
!>       = 'N':  A * X = B     (بدون تبديل)
!>       = 'T':  A**T * X = B  (تبديل)
!>       = 'C':  A**H * X = B  (تبديل مرافق = تبديل)
!>

N

!>          N هو INTEGER
!>     عدد المعادلات الخطية، أي ترتيب
!>     المصفوفة A.  N >= 0.
!>

NRHS

!>          NRHS هو INTEGER
!>     عدد الأطراف اليمنى، أي عدد الأعمدة
!>     للمصفوفتين B و X.  NRHS >= 0.
!>

A

!>          A هي مصفوفة REAL، البعد (LDA,N)
!>     عند الإدخال، المصفوفة A ذات البعد N*N.  إذا كان FACT = 'F' و EQUED
!>     ليس 'N'، فيجب أن تكون A قد تمت معايرتها بواسطة عوامل القياس
!>     في R و/أو C.
!>     لا يتم تعديل A إذا كان FACT = 'F' أو 'N'، أو إذا كان FACT = 'E' و EQUED = 'N' عند الخروج.
!>
!>     عند الخروج، إذا كان EQUED .ne. 'N'، يتم قياس A كالتالي:
!>     EQUED = 'R':  A := diag(R) * A
!>     EQUED = 'C':  A := A * diag(C)
!>     EQUED = 'B':  A := diag(R) * A * diag(C).
!>

LDA

!>          LDA هو INTEGER
!>     البعد الرئيسي للمصفوفة A.  LDA >= max(1,N).
!>

AF

!>          AF هي مصفوفة REAL، البعد (LDAF,N)
!>     إذا كان FACT = 'F'، فإن AF هي وسيط إدخال وعند الإدخال
!>     تحتوي على العوامل L و U من تحليل
!>     A = P*L*U كما تم حسابها بواسطة SGETRF.  إذا كان EQUED .ne. 'N'، فإن
!>     AF هو الشكل المحلل للمصفوفة A المعايرة.
!>
!>     إذا كان FACT = 'N'، فإن AF هي وسيط إخراج وعند الخروج
!>     ترجع العوامل L و U من تحليل A = P*L*U
!>     للمصفوفة الأصلية A.
!>
!>     إذا كان FACT = 'E'، فإن AF هي وسيط إخراج وعند الخروج
!>     ترجع العوامل L و U من تحليل A = P*L*U
!>     للمصفوفة A المعايرة (انظر وصف A لشكل المصفوفة المعايرة).
!>

LDAF

!>          LDAF هو INTEGER
!>     البعد الرئيسي للمصفوفة AF.  LDAF >= max(1,N).
!>

IPIV

!>          IPIV هي مصفوفة INTEGER، البعد (N)
!>     إذا كان FACT = 'F'، فإن IPIV هي وسيط إدخال وعند الإدخال
!>     تحتوي على مؤشرات المحور من تحليل
