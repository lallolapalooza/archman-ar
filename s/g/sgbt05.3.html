توثيق الوظيفة/الإجراء الفرعي

subroutine sgbt05 (character trans, integer n, integer kl, integer ku, integer nrhs, real, dimension( ldab, * ) ab, integer ldab, real, dimension( ldb, * ) b, integer ldb, real, dimension( ldx, * ) x, integer ldx, real, dimension( ldxact, * ) xact, integer ldxact, real, dimension( * ) ferr, real, dimension( * ) berr, real, dimension( * ) reslts)

SGBT05

الغرض:

!>
!> يقوم SGBT05 باختبار حدود الخطأ للتحسين التكراري للحل المحسوب لنظام المعادلات
!> op(A)*X = B، حيث A هي مصفوفة نطاق عامة من الرتبة n مع kl أقطار فرعية و ku
!> أقطار فائقة و op(A) = A أو A**T، اعتمادًا على TRANS.
!>
!> RESLTS(1) = اختبار حد الخطأ
!>           = norm(X - XACT) / ( norm(X) * FERR )
!>
!> يتم إرجاع قيمة كبيرة إذا لم تكن هذه النسبة أقل من واحد.
!>
!> RESLTS(2) = المتبقي من روتين التحسين التكراري
!>           = الحد الأقصى لـ BERR / ( NZ*EPS + (*) )، حيث
!>             (*) = NZ*UNFL / (min_i (abs(op(A))*abs(X) +abs(b))_i )
!>             و NZ = الحد الأقصى لعدد العناصر غير الصفرية في أي صف من A، بالإضافة إلى 1
!>

المعلمات

TRANS 
!>          TRANS هو CHARACTER*1
!>          يحدد شكل نظام المعادلات.
!>          = 'N':  A    * X = B  (بدون تبديل)
!>          = 'T':  A**T * X = B  (تبديل)
!>          = 'C':  A**H * X = B  (تبديل مترافق = تبديل)
!>

N

!>          N هو INTEGER
!>          عدد صفوف المصفوفات X، B، و XACT، وترتيب المصفوفة A. N >= 0.
!>

KL

!>          KL هو INTEGER
!>          عدد الأقطار الفرعية داخل نطاق A. KL >= 0.
!>

KU

!>          KU هو INTEGER
!>          عدد الأقطار الفائقة داخل نطاق A. KU >= 0.
!>

NRHS

!>          NRHS هو INTEGER
!>          عدد أعمدة المصفوفات X، B، و XACT. NRHS >= 0.
!>

AB

!>          AB هي مصفوفة REAL، ذات بعد (LDAB,N)
!>          مصفوفة النطاق الأصلية A، مخزنة في الصفوف من 1 إلى KL+KU+1.
!>          يتم تخزين العمود j من A في العمود j من المصفوفة AB كما يلي:
!>          AB(ku+1+i-j,j) = A(i,j) لـ max(1,j-ku) <= i <= min(n,j+kl).
!>

LDAB

!>          LDAB هو INTEGER
!>          البعد الرائد للمصفوفة AB. LDAB >= KL+KU+1.
!>

B

!>          B هي مصفوفة REAL، ذات بعد (LDB,NRHS)
!>          متجهات الطرف الأيمن لنظام المعادلات الخطية.
!>

LDB

!>          LDB هو INTEGER
!>          البعد الرائد للمصفوفة B. LDB >= max(1,N).
!>

X

!>          X هي مصفوفة REAL، ذات بعد (LDX,NRHS)
!>          متجهات الحل المحسوبة. يتم تخزين كل متجه كعمود في المصفوفة X.
!>

LDX

!>          LDX هو INTEGER
!>          البعد الرائد للمصفوفة X. LDX >= max(1,N).
!>

XACT

!>          XACT هي مصفوفة REAL، ذات بعد (LDX,NRHS)
!>          متجهات الحل الدقيقة. يتم تخزين كل متجه كعمود في المصفوفة XACT.
!>

LDXACT

!>          LDXACT هو INTEGER
!>          البعد الرائد للمصفوفة XACT. LDXACT >= max(1,N).
!>

FERR

!>          FERR هي مصفوفة REAL، ذات بعد (NRHS)
!>          حدود الخطأ الأمامي المقدرة لكل متجه حل X. إذا كان XTRUE هو الحل الحقيقي،
!>          فإن FERR يحد حجم أكبر إدخال في (X - XTRUE) مقسومًا على حجم أكبر
!>          إدخال في X.
!>

BERR

!>          BERR هي مصفوفة REAL، ذات بعد (NRHS)
!>          الخطأ الخلفي النسبي المكوني لكل متجه حل (أي، أصغر تغيير نسبي في أي إدخال
!>          في A أو B يجعل X حلاً دقيقًا).
!>

RESLTS

!>          RESLTS هي مصفوفة REAL، ذات بعد (2)
!>          الحد الأقصى على متجهات الحل NRHS للنسب:
!>          RESLTS(1) = norm(X - XACT) / ( norm(X) * FERR )
!>          RESLTS(2) = BERR / ( NZ*EPS + (*) )
!>
