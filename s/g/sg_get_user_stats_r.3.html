sg_get_user_stats_r(3)

الحزمة: libstatgrab

التراخيص: LGPL, GPL

sg_get_user_stats(3) sg_get_user_stats(3)

الاسم

sg_get_user_stats, sg_get_user_stats_r, sg_free_user_stats - الحصول على إحصائيات المستخدمين المسجلين حاليًا

الملخص

#include <statgrab.h>


sg_user_stats *sg_get_user_stats
(size_t *entries);


sg_user_stats *sg_get_user_stats_r
(size_t *entries);


sg_error sg_free_user_stats
(sg_user_stats *data);

الوصف

تقدم وحدة إحصائيات المستخدم معلومات حول المستخدمين المسجلين. يتم التعامل مع كل تسجيل دخول بشكل منفصل. كلا دالتين الحصول على الإحصائيات، sg_get_user_stats() و sg_get_user_stats_r()، تُرجع متجهًا من sg_user_stats يحتوي على عدد elements.

اختصار واجهة برمجة التطبيقات (API)

الدالة تُرجع مالك البيانات
sg_get_user_stats sg_user_stats * libstatgrab (خاص بالمسار)
sg_get_user_stats_r sg_user_stats * المستدعي

يجب تحرير متجه sg_user_stats المستلم من sg_get_user_stats_r() باستخدام sg_free_user_stats() عندما لا يعود ضروريًا. المستدعي مسؤول عن القيام بذلك.

قيم الإرجاع

الهيكل الذي يتم إرجاعه هو من النوع sg_user_stats.

typedef struct {
        char *login_name;
        char *record_id;
        size_t record_id_size;
        char *device;
        char *hostname;
        pid_t pid;
        time_t login_time;
        time_t systime;
} sg_user_stats;
login_name
اسم المستخدم الذي تم استخدامه لتسجيل الدخول
record_id
معرف السجل لقاعدة بيانات المضيف التي تحتوي على معلومات تسجيل الدخول (ليس بالضرورة منتهية بـ 0)
record_id_size
حجم معرف السجل
device
اسم الجهاز (مثل "pts/0") الخاص بالطرفية المعينة لجلسة تسجيل الدخول
hostname
اسم المضيف (البعيد) الذي سجل المستخدم دخوله منه، على سبيل المثال "infoterm7.some.kind.of.domain.local"، "localhost"، "10.42.17.4" أو ":0.0" (في حالة تسجيل الدخول المحلي عبر xterm جديد)
pid
معرف العملية الذي قام بالإدخال إلى قاعدة بيانات المستخدمين المسجلين
login_time
الطابع الزمني (الوقت بالثواني منذ بداية الحقبة) عند تسجيل المستخدم دخوله
systime
الطابع الزمني الذي تم فيه جمع الإحصائيات المذكورة أعلاه بالثواني منذ بداية الحقبة
