توثيق الوظيفة/الإجراء الفرعي

subroutine sgetsls (character trans, integer m, integer n, integer nrhs, real, dimension( lda, * ) a, integer lda, real, dimension( ldb, * ) b, integer ldb, real, dimension( * ) work, integer lwork, integer info)

SGETSLS

الغرض:

!>
!> SGETSLS يحل أنظمة خطية مفرطة التحديد أو ناقصة التحديد حقيقية
!> تتضمن مصفوفة A بحجم M × N ، باستخدام عامل A QR الضيق الطويل أو LQ القصير العريض.
!>
!> يُفترض أن A لها رتبة كاملة، ويتم توفير حماية أولية فقط ضد المصفوفات ذات الرتبة الناقصة.
!> هذا الإجراء الفرعي يكتشف فقط عدم اكتمال الرتبة التام، حيث يكون عنصر قطري
!> للعامل المثلثي لـ A صفرًا تمامًا.
!>
!> من الممكن أن يكون أحد (أو أكثر) من العناصر القطرية للعامل المثلثي لـ A
!> أرقامًا صغيرة جدًا دون أن يشير هذا الإجراء الفرعي إلى خطأ. قد تكون الحلول المحسوبة
!> لمثل هذه المصفوفات شبه ناقصة الرتبة أقل دقة بسبب فقدان الدقة الرقمية.
!>
!>
!> الخيارات التالية متاحة:
!>
!> 1. إذا كان TRANS = 'N' و m >= n: البحث عن حل المربعات الصغرى لنظام
!>    مفرط التحديد، أي حل مشكلة المربعات الصغرى
!>                 تقليل || B - A*X ||.
!>
!> 2. إذا كان TRANS = 'N' و m < n: البحث عن الحل ذي المعيار الأدنى لنظام
!>    ناقص التحديد A * X = B.
!>
!> 3. إذا كان TRANS = 'T' و m >= n: البحث عن الحل ذي المعيار الأدنى لنظام
!>    ناقص التحديد A**T * X = B.
!>
!> 4. إذا كان TRANS = 'T' و m < n: البحث عن حل المربعات الصغرى لنظام
!>    مفرط التحديد، أي حل مشكلة المربعات الصغرى
!>                 تقليل || B - A**T * X ||.
!>
!> يمكن معالجة عدة متجهات جانبية يمنى b ومتجهات حل x في استدعاء واحد؛
!> يتم تخزينها كأعمدة في مصفوفة الجانب الأيمن B بحجم M × NRHS
!> ومصفوفة الحل X بحجم N × NRHS.
!>

المعاملات

TRANS 
!>          TRANS هو CHARACTER*1
!>          = 'N': النظام الخطي يتضمن A؛
!>          = 'T': النظام الخطي يتضمن A**T.
!>

M

!>          M هو INTEGER
!>          عدد صفوف المصفوفة A. M >= 0.
!>

N

!>          N هو INTEGER
!>          عدد أعمدة المصفوفة A. N >= 0.
!>

NRHS

!>          NRHS هو INTEGER
!>          عدد الجوانب اليمنى، أي عدد أعمدة المصفوفات B و X. NRHS >=0.
!>

A

!>          A هو REAL array, dimension (LDA,N)
!>          عند الدخول، مصفوفة A بحجم M × N.
!>          عند الخروج،
!>          يتم استبدال A بتفاصيل عاملها QR أو LQ
!>          كما تم إرجاعها بواسطة SGEQR أو SGELQ.
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرائد للمصفوفة A. LDA >= max(1,M).
!>

B

!>          B هو REAL array, dimension (LDB,NRHS)
!>          عند الدخول، مصفوفة B لمتجهات الجانب الأيمن، مخزنة
!>          بشكل عمودي؛ B بحجم M × NRHS إذا كان TRANS = 'N'، أو N × NRHS
!>          إذا كان TRANS = 'T'.
!>          عند الخروج، إذا كان INFO = 0، يتم استبدال B بمتجهات الحل
!>          المخزنة بشكل عمودي:
!>          إذا كان TRANS = 'N' و m >= n، فإن الصفوف من 1 إلى n في B تحتوي على متجهات حل المربعات الصغرى.
!>          إذا كان TRANS = 'N' و m < n، فإن الصفوف من 1 إلى N في B تحتوي على متجهات الحل ذي المعيار الأدنى؛
!>          إذا كان TRANS = 'T' و m >= n، فإن الصفوف من 1 إلى M في B تحتوي على متجهات الحل ذي المعيار الأدنى؛
!>          إذا كان TRANS = 'T' و m < n، فإن الصفوف من 1 إلى M في B تحتوي على متجهات حل المربعات الصغرى.
!>

LDB

!>          LDB هو INTEGER
!>          البعد الرائد للمصفوفة B. LDB >= MAX(1,M,N).
!>

WORK

!>          (مساحة عمل) REAL array, dimension (MAX(1,LWORK))
!>          عند الخروج، إذا كان INFO = 0، فإن WORK(1) تحتوي على LWORK المثلى (أو الحد الأدنى
!>          أو الأمثل، إذا تم افتراض الاستعلام).
!>          راجع LWORK لمزيد من التفاصيل.
!>

LWORK

!>          LWORK هو INTEGER
!>          بعد المصفوفة WORK. LWORK >= 1.
!>          إذا كان LWORK = -1 أو -2، فسيتم افتراض استعلام مساحة عمل.
!>          إذا كان LWORK = -1، فإن الروتين يحسب الحجم الأمثل لـ WORK لتحقيق الأداء الأمثل
!>          ويعيد هذا القيمة في WORK(1).
!>          إذا كان LWORK = -2، فإن الروتين يحسب الحد الأدنى لحجم WORK
!>          ويعيد هذه القيمة في WORK(1).
!>

INFO

!>          INFO هو INTEGER
!>          = 0: اكتمال ناجح
!>          < 0: إذا كان INFO = -i، فإن الوسيط i كان له قيمة غير قانونية
!>          > 0: إذا كان INFO = i، فإن العنصر القطري i من
!>                العامل المثلثي لـ A هو صفر تمامًا، لذا فإن A ليس لديها
!>                رتبة كاملة؛ لم يكن من الممكن حساب حل المربعات الصغرى.
!>
