subroutine sgemv (character trans, integer m, integer n, real alpha, real, dimension(lda,*) a, integer lda, real, dimension(*) x, integer incx, real beta, real, dimension(*) y, integer incy)

SGEMV

الغرض:

!>
!> SGEMV  تقوم بإحدى عمليات المصفوفة والمتجه التالية:
!>
!>    y := alpha*A*x + beta*y,   أو   y := alpha*A**T*x + beta*y,
!>
!> حيث alpha و beta قيمتان قياسيتان، x و y متجهان، و A مصفوفة m × n.
!>

المعلمات

TRANS 
!>          TRANS هو CHARACTER*1
!>           عند الدخول، تحدد TRANS العملية المراد تنفيذها كما يلي:
!>
!>              TRANS = 'N' أو 'n'   y := alpha*A*x + beta*y.
!>
!>              TRANS = 'T' أو 't'   y := alpha*A**T*x + beta*y.
!>
!>              TRANS = 'C' أو 'c'   y := alpha*A**T*x + beta*y.
!>

M

!>          M هو INTEGER
!>           عند الدخول، تحدد M عدد صفوف المصفوفة A.
!>           يجب أن تكون M أكبر من أو تساوي صفر.
!>

N

!>          N هو INTEGER
!>           عند الدخول، تحدد N عدد أعمدة المصفوفة A.
!>           يجب أن تكون N أكبر من أو تساوي صفر.
!>

ALPHA

!>          ALPHA هو REAL
!>           عند الدخول، تحدد ALPHA القيمة القياسية alpha.
!>

A

!>          A هو مصفوفة REAL، ذات بعد ( LDA, N )
!>           قبل الدخول، يجب أن يحتوي الجزء الرئيسي m × n من المصفوفة A على
!>           مصفوفة المعاملات.
!>

LDA

!>          LDA هو INTEGER
!>           عند الدخول، تحدد LDA البعد الأول للمصفوفة A كما تم تعريفه
!>           في البرنامج (الفرعي) المستدعي. يجب أن تكون LDA أكبر من أو تساوي
!>           max( 1, m ).
!>

X

!>          X هو مصفوفة REAL، ذات بعد لا يقل عن
!>           ( 1 + ( n - 1 )*abs( INCX ) ) عندما TRANS = 'N' أو 'n'
!>           ولا يقل عن
!>           ( 1 + ( m - 1 )*abs( INCX ) ) بخلاف ذلك.
!>           قبل الدخول، يجب أن تحتوي المصفوفة X ذات الأبعاد المتزايدة على
!>           المتجه x.
!>

INCX

!>          INCX هو INTEGER
!>           عند الدخول، تحدد INCX الزيادة لعناصر X. يجب ألا تكون INCX صفرًا.
!>

BETA

!>          BETA هو REAL
!>           عند الدخول، تحدد BETA القيمة القياسية beta. عندما يتم
!>           توفير BETA على أنها صفر، فلا حاجة لتعيين Y عند الدخول.
!>

Y

!>          Y هو مصفوفة REAL، ذات بعد لا يقل عن
!>           ( 1 + ( m - 1 )*abs( INCY ) ) عندما TRANS = 'N' أو 'n'
!>           ولا يقل عن
!>           ( 1 + ( n - 1 )*abs( INCY ) ) بخلاف ذلك.
!>           قبل الدخول مع BETA غير صفر، يجب أن تحتوي المصفوفة Y ذات الأبعاد المتزايدة على
!>           المتجه y. عند الخروج، يتم الكتابة فوق Y بالمتجه y المحدث.
!>           إذا كان m أو n يساوي صفرًا، فلن يتم الرجوع إلى Y وسيتم تنفيذ
!>           استدعاء سريع للدالة.
!>

INCY

!>          INCY هو INTEGER
!>           عند الدخول، تحدد INCY الزيادة لعناصر Y. يجب ألا تكون INCY صفرًا.
!>
