توثيق الوظيفة/الروتين الفرعي

subroutine sgebal (character job, integer n, real, dimension( lda, * ) a, integer lda, integer ilo, integer ihi, real, dimension( * ) scale, integer info)

SGEBAL

الغرض:

!>
!> SGEBAL يوزن مصفوفة حقيقية عامة A. يتضمن ذلك، أولاً،
!> تبديل A بتحويل تشابه لعزل القيم الذاتية في العناصر من 1 إلى ILO-1 ومن IHI+1 إلى N على
!> القطر؛ وثانياً، تطبيق تحويل تشابه قطري
!> للصفوف والأعمدة ILO إلى IHI لجعل الصفوف والأعمدة أقرب ما يمكن في المعيار.
!> كلا الخطوتين اختياريتان.
!>
!> قد يؤدي الوزن إلى تقليل المعيار 1 للمصفوفة، وتحسين
!> دقة القيم الذاتية و/أو المتجهات الذاتية المحسوبة.
!>

المعلمات:

JOB 
!>          JOB هو CHARACTER*1
!>          يحدد العمليات التي سيتم إجراؤها على A:
!>          = 'N':  لا شيء: ببساطة اضبط ILO = 1, IHI = N, SCALE(I) = 1.0
!>                  من أجل i = 1,...,N;
!>          = 'P':  تبديل فقط;
!>          = 'S':  قياس فقط;
!>          = 'B':  تبديل وقياس كلاهما.
!>

N

!>          N هو INTEGER
!>          ترتيب المصفوفة A. N >= 0.
!>

A

!>          A هي مصفوفة REAL، البعد (LDA,N)
!>          عند الدخول، المصفوفة A المدخلة.
!>          عند الخروج، يتم استبدال A بالمصفوفة الموزونة.
!>          إذا كان JOB = 'N'، لا يتم الرجوع إلى A.
!>          انظر التفاصيل الإضافية.
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرئيسي للمصفوفة A. LDA >= max(1,N).
!>

ILO

!>          ILO هو INTEGER
!>

IHI

!>          IHI هو INTEGER
!>          يتم تعيين ILO و IHI إلى أعداد صحيحة بحيث عند الخروج
!>          A(i,j) = 0 إذا كان i > j و j = 1,...,ILO-1 أو i = IHI+1,...,N.
!>          إذا كان JOB = 'N' أو 'S'، فإن ILO = 1 و IHI = N.
!>

SCALE

!>          SCALE هي مصفوفة REAL، البعد (N)
!>          تفاصيل التباديل وعوامل القياس المطبقة على
!>          A. إذا كان P(j) هو فهرس الصف والعمود الذي تم تبديله
!>          مع الصف والعمود j وكان D(j) هو عامل القياس
!>          المطبق على الصف والعمود j، فإن
!>          SCALE(j) = P(j)    من أجل j = 1,...,ILO-1
!>                   = D(j)    من أجل j = ILO,...,IHI
!>                   = P(j)    من أجل j = IHI+1,...,N.
!>          ترتيب التبادلات التي تم إجراؤها هو N إلى IHI+1،
!>          ثم 1 إلى ILO-1.
!>

INFO

!>          INFO هو INTEGER
!>          = 0:  انتهاء ناجح.
!>          < 0:  إذا كان INFO = -i، كان المعامل i له قيمة غير قانونية.
!>

تفاصيل إضافية:

!>
!>  تتكون التباديل من تبادلات الصفوف والأعمدة التي تضع
!>  المصفوفة في الشكل
!>
!>             ( T1   X   Y  )
!>     P A P = (  0   B   Z  )
!>             (  0   0   T2 )
!>
!>  حيث T1 و T2 مصفوفتان مثلثيتان علويتان تقع قيمهما الذاتية
!>  على طول القطر. تحدد فهارس الأعمدة ILO و IHI بداية
!>  ونهاية الأعمدة في المصفوفة الفرعية B. الوزن يتكون من تطبيق
!>  تحويل تشابه قطري inv(D) * B * D لجعل
!>  المعايير 1 لكل صف من B والعمود المقابل له متساوية تقريبًا.
!>  المصفوفة الناتجة هي
!>
!>     ( T1     X*D          Y    )
!>     (  0  inv(D)*B*D  inv(D)*Z ).
!>     (  0      0           T2   )
!>
!>  تتم إعادة المعلومات حول التباديل P والمصفوفة القطرية D
!>  في المتجه SCALE.
!>
!>  يعتمد هذا الروتين الفرعي على روتين EISPACK BALANC.
!>
!>  تم تعديله بواسطة Tzu-Yi Chen، قسم علوم الكمبيوتر، جامعة
!>    كاليفورنيا في بيركلي، الولايات المتحدة الأمريكية
!>
!>  تمت إعادة هيكلته بواسطة Evert Provoost، قسم علوم الكمبيوتر،
!>    KU Leuven، بلجيكا
!>
