subroutine sgesvx (fact, trans, n, nrhs, a, lda, af, ldaf, ipiv, equed, r, c, b, ldb, x, ldx, rcond, ferr, berr, work, iwork, info)

SGESVX تحسب حل نظام المعادلات الخطية A * X = B لمصفوفات GE

الغرض:

!>
!> SGESVX يستخدم تحليل LU لحساب حل نظام المعادلات الخطية الحقيقية
!>    A * X = B,
!> حيث A مصفوفة N × N و X و B مصفوفتان N × NRHS.
!>
!> يتم أيضًا توفير حدود خطأ للحل وتقدير شرط.
!>

الوصف:

!>
!> يتم تنفيذ الخطوات التالية:
!>
!> 1. إذا كانت FACT = 'E'، يتم حساب عوامل قياس حقيقية لموازنة
!>    النظام:
!>       TRANS = 'N':  diag(R)*A*diag(C)     *inv(diag(C))*X = diag(R)*B
!>       TRANS = 'T': (diag(R)*A*diag(C))**T *inv(diag(R))*X = diag(C)*B
!>       TRANS = 'C': (diag(R)*A*diag(C))**H *inv(diag(R))*X = diag(C)*B
!>    يعتمد ما إذا كان سيتم موازنة النظام أم لا على
!>    قياس المصفوفة A، ولكن إذا تم استخدام الموازنة، يتم
!>    الكتابة فوق A بواسطة diag(R)*A*diag(C) و B بواسطة diag(R)*B (إذا كان TRANS='N')
!>    أو diag(C)*B (إذا كان TRANS = 'T' أو 'C').
!>
!> 2. إذا كانت FACT = 'N' أو 'E'، يتم استخدام تحليل LU لتحليل
!>    المصفوفة A (بعد الموازنة إذا كانت FACT = 'E') كـ
!>       A = P * L * U,
!>    حيث P مصفوفة تبديل، L مصفوفة مثلثية سفلية موحدة، و U مصفوفة مثلثية عليا.
!>
!> 3. إذا كان بعض U(i,i)=0، بحيث تكون U مفردة تمامًا، فإن الروتين
!>    يعود بـ INFO = i. بخلاف ذلك، يتم استخدام الشكل المحلل لـ A
!>    لتقدير رقم شرط المصفوفة A. إذا كان
!>    مقلوب رقم الشرط أقل من دقة الآلة،
!>    يتم إرجاع INFO = N+1 كتحذير، ولكن الروتين لا يزال يواصل
!>    الحل لـ X وحساب حدود الخطأ كما هو موضح أدناه.
!>
!> 4. يتم حل نظام المعادلات لـ X باستخدام الشكل المحلل لـ
!>    A.
!>
!> 5. يتم تطبيق التحسين التكراري لتحسين الحل المحسوب
!>    وتحسين المصفوفة وحساب حدود الخطأ وتقديرات الخطأ الخلفي
!>    له.
!>
!> 6. إذا تم استخدام الموازنة، يتم ضرب المصفوفة X مسبقًا بـ
!>    diag(C) (إذا كان TRANS = 'N') أو diag(R) (إذا كان TRANS = 'T' أو 'C') بحيث
!>    يحل النظام الأصلي قبل الموازنة.
!>

المعلمات

FACT 
!>          FACT is CHARACTER*1
!>          يحدد ما إذا كان الشكل المحلل للمصفوفة A مزودًا عند الدخول، وإذا لم يكن كذلك،
!>          فهل يجب موازنة المصفوفة A قبل تحليلها.
!>          = 'F':  عند الدخول، تحتوي AF و IPIV على الشكل المحلل لـ A.
!>                  إذا كانت EQUED ليست 'N'، فقد تمت موازنة المصفوفة A
!>                  بعوامل القياس المعطاة بواسطة R و C.
!>                  لا يتم تعديل A، AF، و IPIV.
!>          = 'N':  سيتم نسخ المصفوفة A إلى AF وتحليلها.
!>          = 'E':  سيتم موازنة المصفوفة A إذا لزم الأمر، ثم
!>                  نسخها إلى AF وتحليلها.
!>

TRANS

!>          TRANS is CHARACTER*1
!>          يحدد شكل نظام المعادلات:
!>          = 'N':  A * X = B     (بدون تبديل)
!>          = 'T':  A**T * X = B  (تبديل)
!>          = 'C':  A**H * X = B  (تبديل)
!>

N

!>          N is INTEGER
!>          عدد المعادلات الخطية، أي رتبة
!>          المصفوفة A.  N >= 0.
!>

NRHS

!>          NRHS is INTEGER
!>          عدد الأطراف اليمنى، أي عدد الأعمدة
!>          للمصفوفتين B و X.  NRHS >= 0.
!>

A

!>          A is REAL array, dimension (LDA,N)
!>          عند الدخول، المصفوفة N × N A.  إذا كانت FACT = 'F' و EQUED
!>          ليست 'N'، فيجب أن تكون A قد تمت موازنتها بواسطة عوامل القياس
!>          في R و/أو C.  لا يتم تعديل A إذا كانت FACT = 'F' أو
!>          'N'، أو إذا كانت FACT = 'E' و EQUED = 'N' عند الخروج.
!>
!>          عند الخروج، إذا كانت EQUED .ne. 'N'، تتم موازنة A كالتالي:
!>          EQUED = 'R':  A := diag(R) * A
!>          EQUED = 'C':  A := A * diag(C)
!>          EQUED = 'B':  A := diag(R) * A * diag(C).
!>

LDA

!>          LDA is INTEGER
!>          البعد الرئيسي للمصفوفة A.  LDA >= max(1,N).
!>

AF

!>          AF is REAL array, dimension (LDAF,N)
!>          إذا كانت FACT = 'F'، فإن AF هي معلمة إدخال وتحتوي عند الدخول
!>          على العوامل L و U من تحليل
!>          A = P*L*U كما تم حسابه بواسطة SGETRF.  إذا كانت EQUED .ne. 'N'، فإن
!>          AF هو الشكل المحلل للمصفوفة A المتوازنة.
!>
!>          إذا كانت FACT = 'N'، فإن AF هي معلمة إخراج وتعود عند الخروج
!>          بالعوامل L و U من تحليل A = P*L*U
!>          للمصفوفة الأصلية A.
!>
!>          إذا كانت FACT = 'E'، فإن AF هي معلمة إخراج وتعود عند الخروج
!>          بالعوامل L و U من تحليل A = P*L*U
!>          للمصفوفة المتوازنة A (انظر وصف A لـ
!>          شكل المصفوفة المتوازنة).
!>

LDAF

!>          LDAF is INTEGER
!>          البعد الرئيسي للمصفوفة AF.  LDAF >= max(1,N).
!>

IPIV

!>          IPIV is INTEGER array, dimension (N)
!>          إذا كانت FACT = 'F'، فإن IPIV هي معلمة إدخال وتحتوي عند الدخول
!>          على مؤشرات المحور من تحليل A = P*L*U
!>          كما تم حسابه بواسطة SGETRF؛ تم تبديل الصف i للمصفوفة
!>          مع الصف IPIV(i).
!>
!>          إذا كانت FACT = 'N'، فإن IPIV هي معلمة إخراج وتعود عند الخروج
!>          بمؤشرات المحور من تحليل A = P*L*U
!>          للمصفوفة الأصلية A.
!>
!>          إذا كانت FACT = 'E'، فإن IPIV هي معلمة إخراج وتعود عند الخروج
!>          بمؤشرات المحور من تحليل A = P*L*U
!>          للمصفوفة المتوازنة A.
!>

EQUED

!>          EQUED is CHARACTER*1
!>          يحدد شكل الموازنة التي تم إجراؤها.
!>          = 'N':  لا توجد موازنة (صحيحة دائمًا إذا كانت FACT = 'N').
!>          = 'R':  موازنة الصف، أي تم ضرب A مسبقًا بـ
!>                  diag(R).
!>          = 'C':  موازنة العمود، أي تمت موازنة A بشكل لاحق بـ
!>                  diag(C).
!>          = 'B':  موازنة الصف والعمود، أي تم استبدال A بـ
!>                  diag(R) * A * diag(C).
!>          EQUED هي معلمة إدخال إذا كانت FACT = 'F'؛ بخلاف ذلك، فهي معلمة
!>          إخراج.
!>

R

!>          R is REAL array, dimension (N)
!>          عوامل قياس الصف لـ A.  إذا كانت EQUED = 'R' أو 'B'، يتم ضرب A
!>          من اليسار بـ diag(R)؛ إذا كانت EQUED = 'N' أو 'C'، لا يتم الوصول إلى R.
!>          R هي معلمة إدخال إذا كانت FACT = 'F'؛
!>          بخلاف ذلك، R هي معلمة إخراج.  إذا كانت FACT = 'F' و
!>
