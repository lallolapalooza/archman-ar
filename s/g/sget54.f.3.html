subroutine sget54 (integer n, real, dimension( lda, * ) a, integer lda, real, dimension( ldb, * ) b, integer ldb, real, dimension( lds, * ) s, integer lds, real, dimension( ldt, * ) t, integer ldt, real, dimension( ldu, * ) u, integer ldu, real, dimension( ldv, * ) v, integer ldv, real, dimension( * ) work, real result)

SGET54

الغرض:

!>
!> SGET54 يتحقق من تفكيك معمّم للشكل
!>
!>          A = U*S*V'  و B = U*T* V'
!>
!> حيث ' تعني تبديل الصفوف والأعمدة و U و V متعامدتان.
!>
!> بالتحديد،
!>
!>  RESULT = ||( A - U*S*V', B - U*T*V' )|| / (||( A, B )||*n*ulp )
!>

المعاملات

N 
!>          N هو عدد صحيح (INTEGER)
!>          حجم المصفوفة. إذا كان صفرًا، فإن SGET54 لا تفعل شيئًا.
!>          يجب أن يكون على الأقل صفرًا.
!>

A

!>          A هي مصفوفة حقيقية (REAL array)، أبعادها (LDA, N)
!>          المصفوفة الأصلية (غير المحللة) A.
!>

LDA

!>          LDA هو عدد صحيح (INTEGER)
!>          البعد الرائد لـ A. يجب أن يكون على الأقل 1
!>          وعلى الأقل N.
!>

B

!>          B هي مصفوفة حقيقية (REAL array)، أبعادها (LDB, N)
!>          المصفوفة الأصلية (غير المحللة) B.
!>

LDB

!>          LDB هو عدد صحيح (INTEGER)
!>          البعد الرائد لـ B. يجب أن يكون على الأقل 1
!>          وعلى الأقل N.
!>

S

!>          S هي مصفوفة حقيقية (REAL array)، أبعادها (LDS, N)
!>          المصفوفة المحللة S.
!>

LDS

!>          LDS هو عدد صحيح (INTEGER)
!>          البعد الرائد لـ S. يجب أن يكون على الأقل 1
!>          وعلى الأقل N.
!>

T

!>          T هي مصفوفة حقيقية (REAL array)، أبعادها (LDT, N)
!>          المصفوفة المحللة T.
!>

LDT

!>          LDT هو عدد صحيح (INTEGER)
!>          البعد الرائد لـ T. يجب أن يكون على الأقل 1
!>          وعلى الأقل N.
!>

U

!>          U هي مصفوفة حقيقية (REAL array)، أبعادها (LDU, N)
!>          المصفوفة المتعامدة على الجانب الأيسر في
!>          التفكيك.
!>

LDU

!>          LDU هو عدد صحيح (INTEGER)
!>          البعد الرائد لـ U. يجب أن يكون LDU على الأقل N
!>          وعلى الأقل 1.
!>

V

!>          V هي مصفوفة حقيقية (REAL array)، أبعادها (LDV, N)
!>          المصفوفة المتعامدة على الجانب الأيسر في
!>          التفكيك.
!>

LDV

!>          LDV هو عدد صحيح (INTEGER)
!>          البعد الرائد لـ V. يجب أن يكون LDV على الأقل N
!>          وعلى الأقل 1.
!>

WORK

!>          WORK هي مصفوفة حقيقية (REAL array)، أبعادها (3*N**2)
!>

RESULT

!>          RESULT هو عدد حقيقي (REAL)
!>          القيمة RESULT، وهي محدودة حاليًا بـ 1/ulp، لتجنب
!>          الفيضان. يتم الإبلاغ عن الأخطاء بواسطة RESULT=10/ulp.
!>
