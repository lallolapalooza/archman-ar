subroutine sgemm (character transa, character transb, integer m, integer n, integer k, real alpha, real, dimension(lda,*) a, integer lda, real, dimension(ldb,*) b, integer ldb, real beta, real, dimension(ldc,*) c, integer ldc)

SGEMM

الغرض:

!>
!> SGEMM  تقوم بإحدى عمليات ضرب المصفوفات:
!>
!>    C := alpha*op( A )*op( B ) + beta*C,
!>
!> حيث  op( X )  هي إحدى العمليات التالية:
!>
!>    op( X ) = X   أو   op( X ) = X**T,
!>
!> alpha و beta هما قيمتان قياسيتان، و A، B، C هي مصفوفات، حيث  op( A )  مصفوفة
!>  m × k ،  op( B )  مصفوفة  k × n  ، و C مصفوفة m × n.
!>

المعلمات

TRANSA 
!>          TRANSA is CHARACTER*1
!>           عند الإدخال، تحدد TRANSA شكل op( A ) المستخدم في
!>           ضرب المصفوفات كالتالي:
!>
!>              TRANSA = 'N' أو 'n'،  op( A ) = A.
!>
!>              TRANSA = 'T' أو 't'،  op( A ) = A**T.
!>
!>              TRANSA = 'C' أو 'c'،  op( A ) = A**T.
!>

TRANSB

!>          TRANSB is CHARACTER*1
!>           عند الإدخال، تحدد TRANSB شكل op( B ) المستخدم في
!>           ضرب المصفوفات كالتالي:
!>
!>              TRANSB = 'N' أو 'n'،  op( B ) = B.
!>
!>              TRANSB = 'T' أو 't'،  op( B ) = B**T.
!>
!>              TRANSB = 'C' أو 'c'،  op( B ) = B**T.
!>

M

!>          M is INTEGER
!>           عند الإدخال، تحدد M عدد صفوف المصفوفة op( A ) والمصفوفة C.
!>           يجب أن تكون M على الأقل صفر.
!>

N

!>          N is INTEGER
!>           عند الإدخال، تحدد N عدد أعمدة المصفوفة op( B ) وعدد أعمدة
!>           المصفوفة C. يجب أن تكون N على الأقل صفر.
!>

K

!>          K is INTEGER
!>           عند الإدخال، تحدد K عدد أعمدة المصفوفة op( A ) وعدد صفوف
!>           المصفوفة op( B ). يجب أن تكون K على الأقل صفر.
!>

ALPHA

!>          ALPHA is REAL
!>           عند الإدخال، تحدد ALPHA القيمة القياسية alpha.
!>

A

!>          A is REAL array, dimension ( LDA, ka ), where ka is
!>           k  when  TRANSA = 'N' or 'n',  and is  m  otherwise.
!>           قبل الإدخال مع  TRANSA = 'N' or 'n'، يجب أن تحتوي الجزء
!>           الرئيسي m × k من المصفوفة A على المصفوفة A، وإلا يجب أن
!>           يحتوي الجزء الرئيسي k × m من المصفوفة A على المصفوفة A.
!>

LDA

!>          LDA is INTEGER
!>           عند الإدخال، تحدد LDA البعد الأول للمصفوفة A كما تم الإعلان عنه
!>           في البرنامج (الروتين الفرعي) المستدعي. عندما تكون TRANSA = 'N'
!>           أو 'n'، يجب أن يكون LDA على الأقل max( 1, m )، وإلا يجب أن يكون
!>           LDA على الأقل max( 1, k ).
!>

B

!>          B is REAL array, dimension ( LDB, kb ), where kb is
!>           n  when  TRANSB = 'N' or 'n',  and is  k  otherwise.
!>           قبل الإدخال مع  TRANSB = 'N' or 'n'، يجب أن تحتوي الجزء
!>           الرئيسي k × n من المصفوفة B على المصفوفة B، وإلا يجب أن
!>           يحتوي الجزء الرئيسي n × k من المصفوفة B على المصفوفة B.
!>

LDB

!>          LDB is INTEGER
!>           عند الإدخال، تحدد LDB البعد الأول للمصفوفة B كما تم الإعلان عنه
!>           في البرنامج (الروتين الفرعي) المستدعي. عندما تكون TRANSB = 'N'
!>           أو 'n'، يجب أن يكون LDB على الأقل max( 1, k )، وإلا يجب أن يكون
!>           LDB على الأقل max( 1, n ).
!>

BETA

!>          BETA is REAL
!>           عند الإدخال، تحدد BETA القيمة القياسية beta. عندما تكون BETA
!>           مُقدمة كصفر، لا يلزم تعيين C عند الإدخال.
!>

C

!>          C is REAL array, dimension ( LDC, N )
!>           قبل الإدخال، يجب أن يحتوي الجزء الرئيسي m × n من المصفوفة C على
!>           المصفوفة C، إلا عندما تكون beta صفرًا، وفي هذه الحالة لا يلزم
!>           تعيين C عند الإدخال.
!>           عند الخروج، يتم استبدال المصفوفة C بالمصفوفة m × n
!>           ( alpha*op( A )*op( B ) + beta*C ).
!>

LDC

!>          LDC is INTEGER
!>           عند الإدخال، تحدد LDC البعد الأول للمصفوفة C كما تم الإعلان عنه
!>           في البرنامج (الروتين الفرعي) المستدعي. يجب أن يكون LDC على
!>           الأقل max( 1, m ).
!>
