recursive subroutine sgelqt3 (m, n, a, lda, t, ldt, info)
SGELQT3

توثيق الوظيفة/الإجراء الفرعي

recursive subroutine sgelqt3 (integer m, integer n, real, dimension( lda, * ) a, integer lda, real, dimension( ldt, * ) t, integer ldt, integer info)

SGELQT3

الغرض:

!>
!> SGELQT3 يحسب بشكل متكرر تحليل LQ لمصفوفة حقيقية M-by-N،
!> باستخدام تمثيل WY المضغوط لـ Q.
!>
!> يعتمد على خوارزمية Elmroth و Gustavson،
!> IBM J. Res. Develop. Vol 44 No. 4 July 2000.
!>

المعلمات

M 
!>          M هو عدد صحيح (INTEGER)
!>          عدد الصفوف في المصفوفة A.  M =< N.
!>

N

!>          N هو عدد صحيح (INTEGER)
!>          عدد الأعمدة في المصفوفة A.  N >= 0.
!>

A

!>          A هي مصفوفة حقيقية (REAL array)، ذات بُعد (LDA,N)
!>          عند الإدخال، تحتوي العناصر الموجودة على القطر وتحته على المصفوفة المثلثية السفلية L ذات البعد N-by-N؛
!>          تحتوي العناصر فوق القطر على صفوف V. انظر أدناه لمزيد من التفاصيل.
!>

LDA

!>          LDA هو عدد صحيح (INTEGER)
!>          البُعد الرائد للمصفوفة A.  LDA >= max(1,M).
!>

T

!>          T هي مصفوفة حقيقية (REAL array)، ذات بُعد (LDT,N)
!>          المصفوفة المثلثية العلوية ذات البعد N-by-N للمعكسر الكتلي.
!>          تحتوي العناصر الموجودة على القطر وفوقه على المعكسر الكتلي T؛
!>          العناصر الموجودة أسفل القطر غير مستخدمة. انظر أدناه لمزيد من التفاصيل.
!>

LDT

!>          LDT هو عدد صحيح (INTEGER)
!>          البُعد الرائد للمصفوفة T.  LDT >= max(1,N).
!>

INFO

!>          INFO هو عدد صحيح (INTEGER)
!>          = 0: اكتمال ناجح
!>          < 0: إذا كان INFO = -i، فإن الوسيط i كان له قيمة غير قانونية
!>

تفاصيل إضافية:

!>
!>  تخزن المصفوفة V المعكسرات الأولية H(i) في الصف i فوق القطر. على سبيل المثال، إذا كان M=5 و N=3، فإن المصفوفة V هي
!>
!>               V = (  1  v1 v1 v1 v1 )
!>                   (     1  v2 v2 v2 )
!>                   (     1  v3 v3 v3 )
!>
!>
!>  حيث تمثل vi المتجهات التي تحدد H(i)، والتي يتم إرجاعها
!>  في المصفوفة A.  لا يتم تخزين الرقم 1 على طول قطر V في A.  المعكسر الكتلي H يُعطى بعد ذلك بواسطة
!>
!>               H = I - V * T * V**T
!>
!>  حيث V**T هو منقول V.
!>
!>  للحصول على تفاصيل الخوارزمية، انظر Elmroth و Gustavson (المشار إليهما أعلاه).
!>
