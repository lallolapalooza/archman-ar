subroutine sggev (character jobvl, character jobvr, integer n, real, dimension( lda, * ) a, integer lda, real, dimension( ldb, * ) b, integer ldb, real, dimension( * ) alphar, real, dimension( * ) alphai, real, dimension( * ) beta, real, dimension( ldvl, * ) vl, integer ldvl, real, dimension( ldvr, * ) vr, integer ldvr, real, dimension( * ) work, integer lwork, integer info)

SGGEV تحسب القيم الذاتية، اختياريًا، المتجهات الذاتية اليسرى و/أو اليمنى لمصفوفات GE

الغرض:

!>
!> SGGEV تحسب لأزواج من المصفوفات الحقيقية غير المتماثلة N-by-N (A,B)
!> القيم الذاتية المعممة، واختياريًا، المتجهات الذاتية المعممة اليسرى و/أو اليمنى.
!>
!> القيمة الذاتية المعممة لزوج من المصفوفات (A,B) هي قيمة قياسية
!> lambda أو نسبة alpha/beta = lambda، بحيث A - lambda*B تكون
!> شاذة. وعادة ما يتم تمثيلها كزوج (alpha,beta)، نظرًا لوجود تفسير معقول
!> لـ beta=0، وحتى لكليهما صفراً.
!>
!> المتجه الذاتي الأيمن v(j) المقابل للقيمة الذاتية lambda(j)
!> لـ (A,B) يحقق
!>
!>                  A * v(j) = lambda(j) * B * v(j).
!>
!> المتجه الذاتي الأيسر u(j) المقابل للقيمة الذاتية lambda(j)
!> لـ (A,B) يحقق
!>
!>                  u(j)**H * A  = lambda(j) * u(j)**H * B .
!>
!> حيث u(j)**H هو المترافق المستعرض لـ u(j).
!>
!>

المعلمات

JOBVL 
!>          JOBVL هو CHARACTER*1
!>          = 'N':  لا تحسب المتجهات الذاتية المعممة اليسرى؛
!>          = 'V':  تحسب المتجهات الذاتية المعممة اليسرى.
!>

JOBVR

!>          JOBVR هو CHARACTER*1
!>          = 'N':  لا تحسب المتجهات الذاتية المعممة اليمنى؛
!>          = 'V':  تحسب المتجهات الذاتية المعممة اليمنى.
!>

N

!>          N هو INTEGER
!>          رتبة المصفوفات A، B، VL، و VR. N >= 0.
!>

A

!>          A هو مصفوفة REAL، البعد (LDA, N)
!>          عند الدخول، المصفوفة A في الزوج (A,B).
!>          عند الخروج، تم استبدال A.
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرئيسي لـ A. LDA >= max(1,N).
!>

B

!>          B هو مصفوفة REAL، البعد (LDB, N)
!>          عند الدخول، المصفوفة B في الزوج (A,B).
!>          عند الخروج، تم استبدال B.
!>

LDB

!>          LDB هو INTEGER
!>          البعد الرئيسي لـ B. LDB >= max(1,N).
!>

ALPHAR

!>          ALPHAR هو مصفوفة REAL، البعد (N)
!>

ALPHAI

!>          ALPHAI هو مصفوفة REAL، البعد (N)
!>

BETA

!>          BETA هو مصفوفة REAL، البعد (N)
!>          عند الخروج، (ALPHAR(j) + ALPHAI(j)*i)/BETA(j)، j=1,...,N، ستكون
!>          القيم الذاتية المعممة. إذا كانت ALPHAI(j) صفرًا، فإن
!>          القيمة الذاتية j حقيقية؛ إذا كانت موجبة، فإن القيمتين الذاتيتين j و
!>          (j+1) تشكلان زوجًا مترافقًا مركبًا، مع
!>          ALPHAI(j+1) سالبًا.
!>
!>          ملاحظة: قد تتجاوز النسب ALPHAR(j)/BETA(j) و ALPHAI(j)/BETA(j)
!>          أو تقلل بسهولة، وقد تكون BETA(j) صفرًا.
!>          لذلك، يجب على المستخدم تجنب حساب النسبة بشكل أعمى
!>          alpha/beta. ومع ذلك، ستكون ALPHAR و ALPHAI دائمًا أقل
!>          من norm(A) وعادة ما تكون قابلة للمقارنة به في المقدار،
!>          و BETA دائمًا أقل من norm(B) وعادة ما تكون قابلة للمقارنة به.
!>

VL

!>          VL هو مصفوفة REAL، البعد (LDVL,N)
!>          إذا كان JOBVL = 'V'، يتم تخزين المتجهات الذاتية اليسرى u(j)
!>          واحدة تلو الأخرى في أعمدة VL، بنفس ترتيب
!>          قيمها الذاتية. إذا كانت القيمة الذاتية j حقيقية، فإن
!>          u(j) = VL(:,j)، العمود j من VL. إذا كانت القيمتان الذاتيتان j و
!>          (j+1) تشكلان زوجًا مترافقًا مركبًا، فإن
!>          u(j) = VL(:,j)+i*VL(:,j+1) و u(j+1) = VL(:,j)-i*VL(:,j+1).
!>          يتم قياس كل متجه ذاتي بحيث يكون أكبر مكون له
!>          abs(الجزء الحقيقي) + abs(الجزء التخيلي) = 1.
!>          غير مستخدم إذا كان JOBVL = 'N'.
!>

LDVL

!>          LDVL هو INTEGER
!>          البعد الرئيسي للمصفوفة VL. LDVL >= 1، و
!>          إذا كان JOBVL = 'V'، فإن LDVL >= N.
!>

VR

!>          VR هو مصفوفة REAL، البعد (LDVR,N)
!>          إذا كان JOBVR = 'V'، يتم تخزين المتجهات الذاتية اليمنى v(j)
!>          واحدة تلو الأخرى في أعمدة VR، بنفس ترتيب
!>          قيمها الذاتية. إذا كانت القيمة الذاتية j حقيقية، فإن
!>          v(j) = VR(:,j)، العمود j من VR. إذا كانت القيمتان الذاتيتان j و
!>          (j+1) تشكلان زوجًا مترافقًا مركبًا، فإن
!>          v(j) = VR(:,j)+i*VR(:,j+1) و v(j+1) = VR(:,j)-i*VR(:,j+1).
!>          يتم قياس كل متجه ذاتي بحيث يكون أكبر مكون له
!>          abs(الجزء الحقيقي) + abs(الجزء التخيلي) = 1.
!>          غير مستخدم إذا كان JOBVR = 'N'.
!>

LDVR

!>          LDVR هو INTEGER
!>          البعد الرئيسي للمصفوفة VR. LDVR >= 1، و
!>          إذا كان JOBVR = 'V'، فإن LDVR >= N.
!>

WORK

!>          WORK هو مصفوفة REAL، البعد (MAX(1,LWORK))
!>          عند الخروج، إذا كان INFO = 0، فإن WORK(1) تعيد قيمة LWORK المثلى.
!>

LWORK

!>          LWORK هو INTEGER
!>          بعد المصفوفة WORK. LWORK >= max(1,8*N).
!>          لأداء جيد، يجب أن يكون LWORK أكبر عمومًا.
!>
!>          إذا كان LWORK = -1، فسيتم افتراض استعلام مساحة العمل؛ يقوم الإجراء
!>          فقط بحساب الحجم الأمثل لمصفوفة WORK، ويعيد
!>          هذه القيمة كالإدخال الأول في مصفوفة WORK، ولا يصدر رسالة خطأ
!>          متعلقة بـ LWORK بواسطة XERBLA.
!>

INFO

!>          INFO هو INTEGER
!>          = 0:  تم الانتهاء بنجاح
!>          < 0:  إذا كان INFO = -i، فإن المعامل i كان له قيمة غير قانونية.
!>          = 1,...,N:
!>                فشلت تكرارات QZ. لم يتم حساب أي متجهات ذاتية،
!>                ولكن يجب أن تكون ALPHAR(j)، ALPHAI(j)، و BETA(j)
!>                صحيحة لـ j=INFO+1,...,N.
!>          > N:  =N+1: فشل تكرار QZ بشكل آخر في SHGEQZ.
!>                =N+2: إرجاع خطأ من STGEVC.
!>
