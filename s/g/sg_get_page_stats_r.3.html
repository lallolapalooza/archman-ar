الحزمة: libstatgrab

التراخيص: LGPL, GPL

sg_get_page_stats(3) sg_get_page_stats(3)

الاسم

sg_get_page_stats, sg_get_page_stats_r, sg_get_page_stats_diff, sg_get_page_stats_diff_between, sg_free_page_stats - الحصول على إحصائيات التصفح

الملخص

#include <statgrab.h>


sg_page_stats *sg_get_page_stats
(size_t *entries);


sg_page_stats *sg_get_page_stats_r
(size_t *entries);


sg_page_stats *sg_get_page_stats_diff
(size_t *entries);


sg_page_stats *sg_get_page_stats_diff_between
(const sg_page_stats *cur, const sg_page_stats *last, size_t *entries);


sg_error sg_free_page_stats
(sg_page_stats *data);

الوصف

وحدة إحصائيات الصفحات تقدم عدد الصفحات التي تم تبديلها للداخل والخارج، إما منذ تم تشغيل الجهاز (بواسطة الدوال sg_get_page_stats() و sg_get_page_stats_r()) أو الفرق بين إحصائيتين مقاستين (بواسطة الدوال sg_get_page_stats_diff() و sg_get_page_stats_diff_between()).

الدالة sg_get_page_stats_diff() تحسب الفرق بين آخر استدعاء للدالة sg_get_page_stats() في هذا الخيط (يتم حفظ إحصائيات الاستدعاء الأقدم ويتم استدعاء sg_get_page_stats() مباشرة بعد ذلك).

اختصار واجهة برمجة التطبيقات (API Shortcut)

الدالة تعيد مالك البيانات
sg_get_page_stats sg_page_stats * libstatgrab (محلية للخيط)
sg_get_page_stats_r sg_page_stats * المستدعي
sg_get_page_stats_diff sg_page_stats * libstatgrab (محلية للخيط)
sg_get_page_stats_diff_between sg_page_stats * المستدعي

يجب تحرير مخزن sg_page_stats الذي تم استلامه من sg_get_page_stats_r() و sg_get_page_stats_diff_between() باستخدام sg_free_page_stats() عند عدم الحاجة إليه بعد الآن. المستدعي مسؤول عن القيام بذلك.

قيم الإرجاع

typedef struct{
        unsigned long long pages_pagein;
        unsigned long long pages_pageout;
        time_t systime;
}sg_page_stats;
pages_pagein
عدد الصفحات التي تم تبديلها إلى الذاكرة.
pages_pageout
عدد الصفحات التي تم تبديلها من الذاكرة (إلى مساحة التبديل).
systime
الطابع الزمني الذي تم فيه جمع الإحصائيات المذكورة أعلاه بالثواني منذ الحقبة أو الفترة الزمنية التي تم فيها نقل pages_pagein و pages_pageout.

أخطاء

يبدو أن Solaris لا يقدم تقارير دقيقة. يقوم بالإبلاغ عن عدد الصفحات التي تم تبديلها إلى الذاكرة، وليس بالضرورة من مساحة التبديل. هذه الميزة ليست موثوقة تمامًا.
