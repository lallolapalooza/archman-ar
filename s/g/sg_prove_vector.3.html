الوصف

sg_vector_create() تقوم بتخصيص وتهيئة متجه statgrab جديد مع عناصر initial_used جاهزة للاستخدام. يتم تخصيص مساحة لعناصر alloc_count في البداية (لتجنب العديد من استدعاءات realloc() أثناء استدعاءات sg_vector_resize() اللاحقة). يجب أن يكون قيمة block_size قوة للرقم 2، ويتم تقريبها إلى أقرب قوة للرقم 2 عندما لا تكون كذلك. إذا لم يكن alloc_count مضاعفًا لـ block_size، فسيتم تقريبه إلى أقرب مضاعف لـ block_size. تعيد مؤشرًا إلى المتجه المنشأ حديثًا.

sg_vector_clear() تدمر جميع العناصر الموجودة في المتجه المعطى. على عكس sg_vector_resize( x, 0 )، يظل حجم المتجه المخصص دون تغيير.

sg_vector_resize() تزيد أو تقلل من عدد العناصر المخصصة في المتجه المحدد. يمثل عدد العناصر المخصصة دائمًا مضاعفًا لمعامل التهيئة block_size. في الحالة الخاصة، يتم استدعاء sg_vector_resize() بـ 0 في الوسيط new_count، يتم تحرير المتجه بعد تدمير جميع عناصر المتجه. تعيد مؤشرًا إلى المتجه المعاد تحجيمه.

sg_vector_free() تدمر جميع عناصر المتجه وتحرر المساحة المخصصة للمتجه المعطى.

sg_vector_clone() تستنسخ جميع عناصر المتجه المعطى إلى متجه جديد تم إنشاؤه بنفس مواصفات المتجه الأصلي. تعيد مؤشرًا إلى المتجه المستنسخ.

sg_vector_clone_into() تستنسخ جميع عناصر المتجه المصدر المعطى إلى المتجه الهدف المعطى. يجب أن يكون المتجه الهدف قد تم إنشاؤه لنفس نوع بيانات العناصر مثل المتجه المصدر. تعيد رمز خطأ غير SG_ERROR_NONE إذا حدث خطأ ما.

sg_vector_compute_diff() تحسب متجه فرق بين المتجه الذي يحتوي على الإحصائيات الحالية ومتجه آخر يحتوي على إحصائيات أقدم. إذا كان عنصر موجودًا في المتجه الحالي ولكنه غير موجود في المتجه المقابل، فسيتم استنساخه إلى متجه النتيجة. إذا كان عنصر موجودًا فقط في المتجه المقابل، فلن يظهر في المتجه الهدف. تعيد sg_vector_compute_diff() رمز خطأ غير SG_ERROR_NONE إذا حدث خطأ ما.

sg_prove_vector() تثبت ما إذا كان المؤشر إلى متجه يشير حقًا إلى متجه. في حالة إشارة مؤشر المتجه المعطى إلى بيانات تالفة، يتم إنهاء البرنامج. عندما تعيد sg_prove_vector()، فإنها تعيد SG_ERROR_NONE.

sg_get_nelements() تعيد عدد العناصر التي تشملها منطقة البيانات المعطاة، والتي يحيط بها متجه statgrab. يتم حساب رأس المتجه داخليًا من المؤشر المعطى لأول عنصر في المتجه.

sg_free_stats_buf() تحرر المتجه الذي يحيط بمنطقة البيانات المعطاة.

ملاحظات

باستثناء sg_get_nelements() و sg_free_stats_buf()، لا يمكن استدعاء أي من الدوال المذكورة أعلاه من خارج مصادر libstatgrab. قد تتغير الهياكل وواجهات برمجة التطبيقات الموثقة دون سابق إنذار. وصف جميع واجهات برمجة التطبيقات الأخرى مخصص لمطوري libstatgrab فقط.

كل متجه يتم إنشاؤه من عنصرين: معلومات المتجه وقائمة العناصر:

template <class T, class Impl>
struct sg_vector {
        size_t used_count;
        size_t alloc_count;
        size_t block_shift;
        Impl vector_implementation;
        T elements[alloc_count];
};

بالطبع، هذا ليس C صالحًا، لذا كان الحل هو الحيل:

typedef struct sg_vector {
	size_t used_count;
	size_t alloc_count;
	size_t block_shift;
	struct sg_vector_init_info info;
} sg_vector;
struct sg_vector_size_helper {
	struct sg_vector v;
	long long ll;
};
#define VECTOR_SIZE offsetof(struct sg_vector_size_helper,ll)
/* Return the data ptr of a vector */
#define VECTOR_DATA(vector) \
	(vector ? (void *)(((char *)vector)+VECTOR_SIZE) : NULL)
#define VECTOR_ADDR_ARITH(ptr) \
	(sg_vector *)(((char *)(ptr))-VECTOR_SIZE)
/* Return the vector for a data */
#define VECTOR_ADDRESS(ptr) \
	((ptr) ? (SG_ERROR_NONE == sg_prove_vector(VECTOR_ADDR_ARITH(ptr)) ? VECTOR_ADDR_ARITH(ptr) : NULL ) : NULL)

هذا يسمح أيضًا لوظائف المستخدم مثل sg_get_nelements() و sg_free_stats_buf() بالتبديل بسهولة بين بنية المتجه والمحتوى.

هيكل تخصيص المتجه

كما ذكرنا، تستخدم آلية المتجه استراتيجيات من مفهوم البرمجة كائنية التوجه المسمى "تعدد الأشكال". يتم وصف المتجه بواسطة كائن صغير يحتوي على سمات متأصلة مثل حجم العنصر ومجموعة من "الدوال الافتراضية" لأداء مهام متعلقة بالعناصر مثل تهيئة أو تدمير العناصر.

typedef void (*vector_init_function)(void *item);
typedef sg_error (*vector_copy_function)(const void *src, void *dst);
typedef sg_error (*vector_compute_diff_function)(void *dst, const void *src);
typedef int (*vector_compare_function)(const void *a, const void *b);
typedef void (*vector_destroy_function)(void *item);
struct sg_vector_init_info {
        size_t item_size;
        vector_init_function init_fn;
        vector_copy_function copy_fn;
        vector_compute_diff_function compute_diff_fn;
        vector_compare_function compare_fn;
        vector_destroy_function destroy_fn;
};

تُهيأ مثيلات struct sg_vector_init_info بشكل مفاهيمي ثابتًا باستخدام الماكرو الذي يعالج النصوص المسبقة VECTOR_INIT_INFO_FULL_INIT(type) أو VECTOR_INIT_INFO_EMPTY_INIT(type). إليك بعض الأمثلة لتوضيح الغرض:

تهيئة وصف متجه إحصائيات وحدة المعالجة المركزية

VECTOR_INIT_INFO_EMPTY_INIT(sg_cpu_stats);

تهيئة وصف متجه إحصائيات معلومات المضيف

static void sg_os_stats_item_init(sg_os_stats *d);
static void sg_os_stats_item_destroy(sg_os_stats *d);
#define sg_os_stats_item_copy NULL
#define sg_os_stats_item_compute_diff NULL
#define sg_os_stats_item_compare NULL
VECTOR_INIT_INFO_FULL_INIT(sg_os_stats);

تهيئة وصف متجه إحصائيات الإدخال/الإخراج للقرص

static void sg_disk_io_stats_item_init(sg_disk_io_stats *d);
static sg_error sg_disk_io_stats_item_copy(sg_disk_io_stats *d, const sg_disk_io_stats *s);
static sg_error sg_disk_io_stats_item_compute_diff(const sg_disk_io_stats *s, sg_disk_io_stats *d);
static int sg_disk_io_stats_item_compare(const sg_disk_io_stats *a, const sg_disk_io_stats *b);
static void sg_disk_io_stats_item_destroy(sg_disk_io_stats *d);
VECTOR_INIT_INFO_FULL_INIT(sg_disk_io_stats);
