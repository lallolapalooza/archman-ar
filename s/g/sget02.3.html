subroutine sget02 (character trans, integer m, integer n, integer nrhs, real, dimension( lda, * ) a, integer lda, real, dimension( ldx, * ) x, integer ldx, real, dimension( ldb, * ) b, integer ldb, real, dimension( * ) rwork, real resid)

SGET02

الغرض:

!>
!> SGET02 يحسب الباقي لحل نظام المعادلات الخطية op(A)*X = B:
!>    RESID = norm(B - op(A)*X) / ( norm(op(A)) * norm(X) * EPS ),
!> حيث op(A) = A أو A**T، اعتمادًا على TRANS، و EPS هو
!> خطأ الآلة.
!> يتم استخدام المعيار (norm) وهو المعيار 1.
!>

المعلمات

TRANS 
!>          TRANS هو CHARACTER*1
!>          يحدد شكل نظام المعادلات:
!>          = 'N':  A    * X = B  (بدون تبديل)
!>          = 'T':  A**T * X = B  (تبديل)
!>          = 'C':  A**H * X = B  (مرافق تبديل = تبديل)
!>

M

!>          M هو INTEGER
!>          عدد صفوف المصفوفة A.  M >= 0.
!>

N

!>          N هو INTEGER
!>          عدد أعمدة المصفوفة A.  N >= 0.
!>

NRHS

!>          NRHS هو INTEGER
!>          عدد أعمدة B، مصفوفة الطرف الأيمن.
!>          NRHS >= 0.
!>

A

!>          A هي مصفوفة REAL، الأبعاد (LDA,N)
!>          المصفوفة الأصلية M x N المسماة A.
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرائد للمصفوفة A.  LDA >= max(1,M).
!>

X

!>          X هي مصفوفة REAL، الأبعاد (LDX,NRHS)
!>          الحل المحسوب للمتجهات لنظام المعادلات الخطية.
!>

LDX

!>          LDX هو INTEGER
!>          البعد الرائد للمصفوفة X.  إذا كان TRANS = 'N'،
!>          LDX >= max(1,N)؛ إذا كان TRANS = 'T' أو 'C'، LDX >= max(1,M).
!>

B

!>          B هي مصفوفة REAL، الأبعاد (LDB,NRHS)
!>          عند الإدخال، متجهات الطرف الأيمن لنظام المعادلات الخطية.
!>          عند الخروج، يتم استبدال B بالفرق B - op(A)*X.
!>

LDB

!>          LDB هو INTEGER
!>          البعد الرائد للمصفوفة B.  إذا كان TRANS = 'N'،
!>          LDB >= max(1,M)؛ إذا كان TRANS = 'T' أو 'C'، LDB >= max(1,N).
!>

RWORK

!>          RWORK هي مصفوفة REAL، الأبعاد (M)
!>

RESID

!>          RESID هو REAL
!>          القيمة القصوى لعدد الأطراف اليمنى من
!>          norm(B - op(A)*X) / ( norm(op(A)) * norm(X) * EPS ).
!>
