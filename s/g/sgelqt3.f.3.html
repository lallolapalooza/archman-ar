recursive subroutine sgelqt3 (m, n, a, lda, t, ldt, info)
SGELQT3

توثيق الوظيفة/الإجراء الفرعي

recursive subroutine sgelqt3 (integer m, integer n, real, dimension( lda, * ) a, integer lda, real, dimension( ldt, * ) t, integer ldt, integer info)

SGELQT3

الغرض:

!>
!> SGELQT3 تحسب بشكل تكراري عامل تصنيف LQ لمصفوفة حقيقية M-by-N
!> A، باستخدام تمثيل WY المضغوط لـ Q.
!>
!> بناءً على خوارزمية Elmroth و Gustavson،
!> IBM J. Res. Develop. Vol 44 No. 4 July 2000.
!>

المعلمات

M 
!>          M هو عدد صحيح (INTEGER)
!>          عدد صفوف المصفوفة A.  M =< N.
!>

N

!>          N هو عدد صحيح (INTEGER)
!>          عدد أعمدة المصفوفة A.  N >= 0.
!>

A

!>          A هي مصفوفة حقيقية (REAL array)، البعد (LDA,N)
!>          عند الإدخال، تحتوي العناصر على القطر وفي الأسفل على المصفوفة المثلثية السفلية L بحجم N-by-N؛
!>          العناصر الموجودة فوق القطر هي صفوف V. انظر أدناه لمزيد من التفاصيل.
!>

LDA

!>          LDA هو عدد صحيح (INTEGER)
!>          البعد الرئيسي للمصفوفة A.  LDA >= max(1,M).
!>

T

!>          T هي مصفوفة حقيقية (REAL array)، البعد (LDT,N)
!>          المصفوفة المثلثية العلوية بحجم N-by-N للمن عاكس الكتلة.
!>          تحتوي العناصر على القطر وفي الأعلى على المن عاكس الكتلة T؛
!>          العناصر الموجودة أسفل القطر لا يتم استخدامها. انظر أدناه لمزيد من التفاصيل.
!>

LDT

!>          LDT هو عدد صحيح (INTEGER)
!>          البعد الرئيسي للمصفوفة T.  LDT >= max(1,N).
!>

INFO

!>          INFO هو عدد صحيح (INTEGER)
!>          = 0: الخروج بنجاح
!>          < 0: إذا كان INFO = -i، فقد كان الوسيط i ذا قيمة غير قانونية
!>

تفاصيل إضافية:

!>
!>  تخزن المصفوفة V المنعكسات الأولية H(i) في الصف i
!>  فوق القطر. على سبيل المثال، إذا كان M=5 و N=3، فإن المصفوفة V هي
!>
!>               V = (  1  v1 v1 v1 v1 )
!>                   (     1  v2 v2 v2 )
!>                   (     1  v3 v3 v3 )
!>
!>
!>  حيث تمثل قيم vi المتجهات التي تحدد H(i)، والتي يتم إرجاعها
!>  في المصفوفة A. لا يتم تخزين القيم 1 على طول قطر V في A.
!>  المنعكس الكتلي H يُعطى بعد ذلك بواسطة
!>
!>               H = I - V * T * V**T
!>
!>  حيث V**T هو منقول V.
!>
!>  للحصول على تفاصيل الخوارزمية، انظر Elmroth و Gustavson (المذكور أعلاه).
!>
