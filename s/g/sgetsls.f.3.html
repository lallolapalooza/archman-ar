توثيق الوظيفة/الروتين الفرعي

subroutine sgetsls (character trans, integer m, integer n, integer nrhs, real, dimension( lda, * ) a, integer lda, real, dimension( ldb, * ) b, integer ldb, real, dimension( * ) work, integer lwork, integer info)

SGETSLS

الغرض:

!>
!> SGETSLS يحل أنظمة خطية مفرطة التحديد أو ناقصة التحديد حقيقية
!> تتضمن مصفوفة A بحجم M-by-N، باستخدام عامل QR نحيف طويل أو عامل LQ قصير عريض
!> للمصفوفة A.
!>
!> يُفترض أن المصفوفة A كاملة الرتبة، ويتم توفير حماية أساسية فقط
!> ضد المصفوفات ناقصة الرتبة. هذا الروتين الفرعي يكتشف فقط
!> نقص الرتبة التام، حيث يكون عنصر قطري في العامل المثلثي
!> للمصفوفة A يساوي صفرًا تمامًا.
!>
!> من الممكن أن تكون إحدى (أو أكثر) العناصر القطرية للعامل المثلثي
!> للمصفوفة A أرقامًا صغيرة جدًا دون أن يقوم هذا الروتين الفرعي بالإشارة
!> إلى خطأ. قد تكون الحلول المحسوبة لمثل هذه المصفوفات شبه ناقصة الرتبة
!> أقل دقة بسبب فقدان الدقة الرقمية.
!>
!>
!> تتوفر الخيارات التالية:
!>
!> 1. إذا كان TRANS = 'N' و m >= n: إيجاد حل المربعات الصغرى لنظام
!>    مفرط التحديد، أي حل مشكلة المربعات الصغرى
!>                 minimizear || B - A*X ||.
!>
!> 2. إذا كان TRANS = 'N' و m < n: إيجاد حل المعيار الأدنى لنظام
!>    ناقص التحديد A * X = B.
!>
!> 3. إذا كان TRANS = 'T' و m >= n: إيجاد حل المعيار الأدنى لنظام
!>    ناقص التحديد A**T * X = B.
!>
!> 4. إذا كان TRANS = 'T' و m < n: إيجاد حل المربعات الصغرى لنظام
!>    مفرط التحديد، أي حل مشكلة المربعات الصغرى
!>                 minimizear || B - A**T * X ||.
!>
!> يمكن التعامل مع عدة متجهات جانبية يمنى b ومتجهات حل x في استدعاء واحد؛
!> وهي مخزنة كأعمدة في مصفوفة المتجهات الجانبية اليمنى B بحجم M-by-NRHS
!> ومصفوفة الحل X بحجم N-by-NRHS.
!>

المعلمات

TRANS 
!>          TRANS هو CHARACTER*1
!>          = 'N': يتضمن النظام الخطي A؛
!>          = 'T': يتضمن النظام الخطي A**T.
!>

M

!>          M هو INTEGER
!>          عدد صفوف المصفوفة A. M >= 0.
!>

N

!>          N هو INTEGER
!>          عدد أعمدة المصفوفة A. N >= 0.
!>

NRHS

!>          NRHS هو INTEGER
!>          عدد الجوانب اليمنى، أي عدد
!>          أعمدة المصفوفتين B و X. NRHS >=0.
!>

A

!>          A هو صفيف REAL، البعد (LDA,N)
!>          عند الإدخال، المصفوفة A بحجم M-by-N.
!>          عند الخروج،
!>          يتم استبدال A بتفاصيل عاملها QR أو LQ
!>          الذي تم إرجاعه بواسطة SGEQR أو SGELQ.
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرئيسي للصفيف A. LDA >= max(1,M).
!>

B

!>          B هو صفيف REAL، البعد (LDB,NRHS)
!>          عند الإدخال، مصفوفة المتجهات الجانبية اليمنى B، مخزنة
!>          عموديًا؛ B بحجم M-by-NRHS إذا كان TRANS = 'N'، أو N-by-NRHS
!>          إذا كان TRANS = 'T'.
!>          عند الخروج، إذا كان INFO = 0، يتم استبدال B بمتجهات
!>          الحل، مخزنة عموديًا:
!>          إذا كان TRANS = 'N' و m >= n، فإن الصفوف من 1 إلى n في B تحتوي على متجهات
!>          حل المربعات الصغرى.
!>          إذا كان TRANS = 'N' و m < n، فإن الصفوف من 1 إلى N في B تحتوي على
!>          متجهات الحل بالمعيار الأدنى؛
!>          إذا كان TRANS = 'T' و m >= n، فإن الصفوف من 1 إلى M في B تحتوي على
!>          متجهات الحل بالمعيار الأدنى؛
!>          إذا كان TRANS = 'T' و m < n، فإن الصفوف من 1 إلى M في B تحتوي على
!>          متجهات حل المربعات الصغرى.
!>

LDB

!>          LDB هو INTEGER
!>          البعد الرئيسي للصفيف B. LDB >= MAX(1,M,N).
!>

WORK

!>          (مساحة العمل) صفيف REAL، البعد (MAX(1,LWORK))
!>          عند الخروج، إذا كان INFO = 0، فإن WORK(1) يحتوي على LWORK الأمثل
!>          (أو الحد الأدنى أو الأمثل، إذا افترض استعلام).
!>          راجع LWORK لمزيد من التفاصيل.
!>

LWORK

!>          LWORK هو INTEGER
!>          بعد الصفيف WORK. LWORK >= 1.
!>          إذا كان LWORK = -1 أو -2، فسيتم افتراض استعلام مساحة العمل.
!>          إذا كان LWORK = -1، يقوم الروتين بحساب الحجم الأمثل لـ WORK
!>          لأداء أمثل ويعيد هذا الحجم في WORK(1).
!>          إذا كان LWORK = -2، يقوم الروتين بحساب الحجم الأدنى لـ WORK
!>          ويعيد هذا الحجم في WORK(1).
!>

INFO

!>          INFO هو INTEGER
!>          = 0: خروج ناجح
!>          < 0: إذا كان INFO = -i، فإن الوسيط i كان لديه قيمة غير قانونية
!>          > 0: إذا كان INFO = i، فإن العنصر القطري i للعامل
!>                المثلثي لـ A يساوي صفرًا تمامًا، وبالتالي فإن A ليس لديها
!>                رتبة كاملة؛ لم يكن من الممكن حساب حل المربعات الصغرى.
!>
