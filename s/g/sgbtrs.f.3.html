وثائق الوظيفة/الروتين الفرعي

subroutine sgbtrs (character trans, integer n, integer kl, integer ku, integer nrhs, real, dimension( ldab, * ) ab, integer ldab, integer, dimension( * ) ipiv, real, dimension( ldb, * ) b, integer ldb, integer info)

SGBTRS

الغرض:

!>
!> SGBTRS تحل نظام المعادلات الخطية
!>    A * X = B  أو  A**T * X = B
!> مع مصفوفة نطاق عامة A باستخدام تحليل LU الذي تم حسابه بواسطة SGBTRF.
!>

المعلمات

TRANS 
!>          TRANS هو CHARACTER*1
!>          يحدد شكل نظام المعادلات.
!>          = 'N':  A * X = B  (بدون تبديل)
!>          = 'T':  A**T* X = B  (تبديل)
!>          = 'C':  A**T* X = B  (تبديل مرافق = تبديل)
!>

N

!>          N هو INTEGER
!>          رتبة المصفوفة A.  N >= 0.
!>

KL

!>          KL هو INTEGER
!>          عدد الأقطار الفرعية داخل نطاق A.  KL >= 0.
!>

KU

!>          KU هو INTEGER
!>          عدد الأقطار الفائقة داخل نطاق A.  KU >= 0.
!>

NRHS

!>          NRHS هو INTEGER
!>          عدد الأطراف اليمنى، أي عدد أعمدة المصفوفة B.  NRHS >= 0.
!>

AB

!>          AB هو مصفوفة REAL، ذات بعد (LDAB,N)
!>          تفاصيل تحليل LU للمصفوفة النطاقية A، كما
!>          تم حسابها بواسطة SGBTRF.  U مخزنة كمصفوفة نطاقية علوية
!>          مع KL+KU أقطار فائقة في الصفوف 1 إلى KL+KU+1، و
!>          المضاعفات المستخدمة أثناء التحليل مخزنة في
!>          الصفوف KL+KU+2 إلى 2*KL+KU+1.
!>

LDAB

!>          LDAB هو INTEGER
!>          البعد الرائد للمصفوفة AB.  LDAB >= 2*KL+KU+1.
!>

IPIV

!>          IPIV هي مصفوفة INTEGER، ذات بعد (N)
!>          مؤشرات المحور؛ من أجل 1 <= i <= N، تم تبديل الصف i للمصفوفة
!>          مع الصف IPIV(i).
!>

B

!>          B هي مصفوفة REAL، ذات بعد (LDB,NRHS)
!>          عند الإدخال، مصفوفة الطرف الأيمن B.
!>          عند الخروج، مصفوفة الحل X.
!>

LDB

!>          LDB هو INTEGER
!>          البعد الرائد للمصفوفة B.  LDB >= max(1,N).
!>

INFO

!>          INFO هو INTEGER
!>          = 0:  انتهاء ناجح
!>          < 0: إذا كان INFO = -i، فإن المعامل i كان له قيمة غير قانونية
!>
