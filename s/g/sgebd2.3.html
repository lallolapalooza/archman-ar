subroutine sgebd2 (m, n, a, lda, d, e, tauq, taup, work, info)
SGEBD2 تقلل مصفوفة عامة إلى شكل ثنائي الأقطار باستخدام خوارزمية غير مجمعة.

توثيق الوظيفة/الروتين الفرعي

subroutine sgebd2 (integer m, integer n, real, dimension( lda, * ) a, integer lda, real, dimension( * ) d, real, dimension( * ) e, real, dimension( * ) tauq, real, dimension( * ) taup, real, dimension( * ) work, integer info)

SGEBD2 تقلل مصفوفة عامة m × n إلى شكل ثنائي الأقطار باستخدام خوارزمية غير مجمعة.

الغرض:

!>
!> SGEBD2 تقلل مصفوفة عامة حقيقية m × n A إلى شكل ثنائي الأقطار علوي أو سفلي B
!> بواسطة تحويل متعامد: Q**T * A * P = B.
!>
!> إذا كان m >= n، تكون B ثنائية الأقطار علوية؛ إذا كان m < n، تكون B ثنائية
!> الأقطار سفلية.
!>

المعلمات

M 
!>          M هو عدد صحيح
!>          عدد الصفوف في المصفوفة A.  M >= 0.
!>

N

!>          N هو عدد صحيح
!>          عدد الأعمدة في المصفوفة A.  N >= 0.
!>

A

!>          A هي مصفوفة حقيقية، ذات أبعاد (LDA,N)
!>          عند الإدخال، المصفوفة العامة m × n المراد تقليلها.
!>          عند الخروج:
!>          إذا كان m >= n، يتم الكتابة فوق القطر الرئيسي والقطر الفرعي الأول
!>            بالمصفوفة ثنائية الأقطار العلوية B؛ العناصر أسفل القطر الرئيسي،
!>            مع المصفوفة TAUQ، تمثل المصفوفة المتعامدة Q كحاصل ضرب عواكس
!>            عنصرية، والعناصر فوق القطر الفرعي الأول، مع المصفوفة TAUP، تمثل
!>            المصفوفة المتعامدة P كحاصل ضرب عواكس عنصرية؛
!>          إذا كان m < n، يتم الكتابة فوق القطر الرئيسي والقطر الفرعي الأول
!>            بالمصفوفة ثنائية الأقطار السفلية B؛ العناصر أسفل القطر الفرعي الأول،
!>            مع المصفوفة TAUQ، تمثل المصفوفة المتعامدة Q كحاصل ضرب عواكس
!>            عنصرية، والعناصر فوق القطر الرئيسي، مع المصفوفة TAUP، تمثل
!>            المصفوفة المتعامدة P كحاصل ضرب عواكس عنصرية.
!>          انظر التفاصيل الإضافية.
!>

LDA

!>          LDA هو عدد صحيح
!>          البعد الرائد للمصفوفة A.  LDA >= max(1,M).
!>

D

!>          D هي مصفوفة حقيقية، ذات أبعاد (min(M,N))
!>          العناصر القطرية للمصفوفة ثنائية الأقطار B:
!>          D(i) = A(i,i).
!>

E

!>          E هي مصفوفة حقيقية، ذات أبعاد (min(M,N)-1)
!>          العناصر غير القطرية للمصفوفة ثنائية الأقطار B:
!>          إذا كان m >= n، E(i) = A(i,i+1) لـ i = 1,2,...,n-1؛
!>          إذا كان m < n، E(i) = A(i+1,i) لـ i = 1,2,...,m-1.
!>

TAUQ

!>          TAUQ هي مصفوفة حقيقية، ذات أبعاد (min(M,N))
!>          العوامل العددية للعواكس العنصرية التي تمثل المصفوفة المتعامدة Q.
!>          انظر التفاصيل الإضافية.
!>

TAUP

!>          TAUP هي مصفوفة حقيقية، ذات أبعاد (min(M,N))
!>          العوامل العددية للعواكس العنصرية التي تمثل المصفوفة المتعامدة P.
!>          انظر التفاصيل الإضافية.
!>

WORK

!>          WORK هي مصفوفة حقيقية، ذات أبعاد (max(M,N))
!>

INFO

!>          INFO هو عدد صحيح
!>          = 0: خروج ناجح.
!>          < 0: إذا كان INFO = -i، فإن المعلمة i كانت لها قيمة غير قانونية.
!>

التفاصيل الإضافية:

!>
!>  المصفوفتان Q و P ممثلتان كحاصل ضرب عواكس عنصرية:
!>
!>  إذا كان m >= n،
!>
!>     Q = H(1) H(2) . . . H(n)  و  P = G(1) G(2) . . . G(n-1)
!>
!>  كل H(i) و G(i) لها الشكل:
!>
!>     H(i) = I - tauq * v * v**T  و G(i) = I - taup * u * u**T
!>
!>  حيث tauq و taup عوامل عددية حقيقية، و v و u متجهات حقيقية؛
!>  v(1:i-1) = 0، v(i) = 1، ويتم تخزين v(i+1:m) عند الخروج في A(i+1:m,i)؛
!>  u(1:i) = 0، u(i+1) = 1، ويتم تخزين u(i+2:n) عند الخروج في A(i,i+2:n)؛
!>  يتم تخزين tauq في TAUQ(i) و taup في TAUP(i).
!>
!>  إذا كان m < n،
!>
!>     Q = H(1) H(2) . . . H(m-1)  و  P = G(1) G(2) . . . G(m)
!>
!>  كل H(i) و G(i) لها الشكل:
!>
!>     H(i) = I - tauq * v * v**T  و G(i) = I - taup * u * u**T
!>
!>  حيث tauq و taup عوامل عددية حقيقية، و v و u متجهات حقيقية؛
!>  v(1:i) = 0، v(i+1) = 1، ويتم تخزين v(i+2:m) عند الخروج في A(i+2:m,i)؛
!>  u(1:i-1) = 0، u(i) = 1، ويتم تخزين u(i+1:n) عند الخروج في A(i,i+1:n)؛
!>  يتم تخزين tauq في TAUQ(i) و taup في TAUP(i).
!>
!>  يتم توضيح محتويات A عند الخروج بالأمثلة التالية:
!>
!>  m = 6 و n = 5 (m > n):          m = 5 و n = 6 (m < n):
!>
!>    (  d   e   u1  u1  u1 )           (  d   u1  u1  u1  u1  u1 )
!>    (  v1  d   e   u2  u2 )           (  e   d   u2  u2  u2  u2 )
!>    (  v1  v2  d   e   u3 )           (  v1  e   d   u3  u3  u3 )
!>    (  v1  v2  v3  d   e  )           (  v1  v2  e   d   u4  u4 )
!>    (  v1  v2  v3  v4  d  )           (  v1  v2  v3  e   d   u5 )
!>    (  v1  v2  v3  v4  v5 )
!>
!>  حيث d و e تمثل عناصر القطر الرئيسي والعناصر غير القطرية لـ B،
!>  vi يمثل عنصرًا من المتجه الذي يعرف H(i)، و ui عنصرًا من المتجه
!>  الذي يعرف G(i).
!>
