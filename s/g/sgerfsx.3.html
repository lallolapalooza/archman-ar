SGERFSX

الغرض:

!>
!>    SGERFSX يحسن الحل المحسوب لنظام المعادلات الخطية ويوفر حدود الخطأ وتقديرات الخطأ الخلفي
!>    للحل. بالإضافة إلى حد الخطأ المعياري، يقوم الكود
!>    بتوفير الحد الأقصى للخطأ المكوني إذا أمكن. انظر
!>    التعليقات لـ ERR_BNDS_NORM و ERR_BNDS_COMP للحصول على تفاصيل
!>    حدود الخطأ.
!>
!>    قد تم موازنة نظام المعادلات الخطية الأصلي
!>    قبل استدعاء هذا الروتين، كما هو موضح بواسطة الوسيطات EQUED، R
!>    و C أدناه. في هذه الحالة، الحل وحدود الخطأ المعادة
!>    هي للنظام الأصلي غير المتوازن.
!> 
!>     يتم تجميع بعض المعلمات الاختيارية في مصفوفة PARAMS.
!>     تحدد هذه الإعدادات كيفية إجراء التحسين، ولكن غالباً ما تكون
!>     القيم الافتراضية مقبولة. إذا كانت القيم الافتراضية مقبولة، يمكن للمستخدمين
!>     تمرير NPARAMS = 0 مما يمنع الكود المصدري من الوصول
!>     إلى الوسيط PARAMS.
!>

المعلمات:

TRANS 
!>          TRANS هو CHARACTER*1
!>     يحدد شكل نظام المعادلات:
!>       = 'N':  A * X = B     (بدون تبديل)
!>       = 'T':  A**T * X = B  (تبديل)
!>       = 'C':  A**H * X = B  (مترافق تبديل = تبديل)
!>

EQUED

!>          EQUED هو CHARACTER*1
!>     يحدد شكل الموازنة التي تم إجراؤها على A
!>     قبل استدعاء هذا الروتين. هذا ضروري لحساب
!>     الحل وحدود الخطأ بشكل صحيح.
!>       = 'N':  لا موازنة
!>       = 'R':  موازنة الصفوف، أي تم ضرب A مسبقًا بـ
!>               diag(R).
!>       = 'C':  موازنة الأعمدة، أي تم ضرب A لاحقًا بـ
!>               diag(C).
!>       = 'B':  موازنة الصفوف والأعمدة، أي تم استبدال A بـ
!>               diag(R) * A * diag(C).
!>               تم تغيير الجانب الأيمن B وفقًا لذلك.
!>

N

!>          N هو INTEGER
!>     ترتيب المصفوفة A.  N >= 0.
!>

NRHS

!>          NRHS هو INTEGER
!>     عدد الأطراف اليمنى، أي عدد أعمدة
!>     المصفوفتين B و X.  NRHS >= 0.
!>

A

!>          A هي مصفوفة REAL، البعد (LDA,N)
!>     المصفوفة الأصلية A بحجم N-by-N.
!>

LDA

!>          LDA هو INTEGER
!>     البعد الرائد للمصفوفة A.  LDA >= max(1,N).
!>

AF

!>          AF هي مصفوفة REAL، البعد (LDAF,N)
!>     العوامل L و U من تحليل A = P*L*U
!>     كما تم حسابها بواسطة SGETRF.
!>

LDAF

!>          LDAF هو INTEGER
!>     البعد الرائد للمصفوفة AF.  LDAF >= max(1,N).
!>

IPIV

!>          IPIV هي مصفوفة INTEGER، البعد (N)
!>     مؤشرات المحور من SGETRF؛ لـ 1<=i<=N، تم تبديل الصف i للمصفوفة
!>     مع الصف IPIV(i).
!>

R

!>          R هي مصفوفة REAL، البعد (N)
!>     عوامل مقياس الصف لـ A.  إذا كان EQUED = 'R' أو 'B'، تم
!>     ضرب A مسبقًا بـ diag(R)؛ إذا كان EQUED = 'N' أو 'C'،
!>     فإن R لا يتم الوصول إليها.
!>     إذا تم الوصول إلى R، يجب أن يكون كل عنصر في R قوة للجذر
!>     لضمان حل موثوق وتقديرات خطأ. المقياس بـ
!>     قوى الجذر لا يسبب أخطاء تقريب إلا إذا
!>     كانت النتيجة تحت التدفق أو فيضانه. أخطاء التقريب أثناء القياس
!>     تؤدي إلى التحسين بمصفوفة ليست مكافئة للمصفوفة
!>     المدخلة، مما ينتج تقديرات خطأ قد لا تكون
!>     موثوقة.
!>

C

!>          C هي مصفوفة REAL، البعد (N)
!>     عوامل مقياس العمود لـ A.  إذا كان EQUED = 'C' أو 'B'، تم
!>     ضرب A لاحقًا بـ diag(C)؛ إذا كان EQUED = 'N' أو 'R'،
!>     فإن C لا يتم الوصول إليها.
!>     إذا تم الوصول إلى C، يجب أن يكون كل عنصر في C قوة للجذر
!>     لضمان حل موثوق وتقديرات خطأ. المقياس بـ
!>     قوى الجذر لا يسبب أخطاء تقريب إلا إذا
!>     كانت النتيجة تحت التدفق أو فيضانه. أخطاء التقريب أثناء القياس
!>     تؤدي إلى التحسين بمصفوفة ليست مكافئة للمصفوفة
!>     المدخلة، مما ينتج تقديرات خطأ قد لا تكون
!>     موثوقة.
!>

B

!>          B هي مصفوفة REAL، البعد (LDB,NRHS)
!>     مصفوفة الطرف الأيمن B.
!>

LDB

!>          LDB هو INTEGER
!>     البعد الرائد للمصفوفة B.  LDB >= max(1,N).
!>

X

!>          X هي مصفوفة REAL، البعد (LDX,NRHS)
!>     عند الإدخال، مصفوفة الحل X، كما تم حسابها بواسطة SGETRS.
!>     عند الخروج، مصفوفة الحل المحسنة X.
!>

LDX

!>          LDX هو INTEGER
!>     البعد الرائد للمصفوفة X.  LDX >= max(1,N).
!>

RCOND

!>          RCOND هو REAL
!>     مقلوب عدد الشرط المقاس. هذا تقدير لـ
!>     مقلوب عدد الشرط Skeel للمصفوفة A بعد
!>     الموازنة (إذا تم إجراؤها). إذا كان هذا أقل من دقة الجهاز
!>     (على وجه الخصوص، إذا كان صفرًا)، فإن المصفوفة منفردة
!>     لدقة العمل. لاحظ أن الخطأ قد لا يزال صغيرًا حتى
!>     إذا كان هذا الرقم صغيرًا جدًا وبدت المصفوفة غير مستقرة.
!>

BERR

!>          BERR هي مصفوفة REAL، البعد (NRHS)
!>     الخطأ الخلفي النسبي المكوني. هذا هو
!>     الخطأ الخلفي النسبي المكوني لكل متجه حل X(j)
!>     (أي، أصغر تغيير نسبي في أي عنصر من A أو B يجعل X(j)
!>     حلاً دقيقًا).
!>

N_ERR_BNDS

!>          N_ERR_BNDS هو INTEGER
!>     عدد حدود الخطأ لإعادتها لكل طرف أيمن
!>     وكل نوع (معياري أو مكوني). انظر ERR_BNDS_NORM و
!>     ERR_BNDS_COMP أدناه.
!>

ERR_BNDS_NORM

!>          ERR_BNDS_NORM هي مصفوفة REAL، البعد (NRHS, N_ERR_BNDS)
!>     لكل طرف أيمن، تحتوي هذه المصفوفة على معلومات حول
!>     حدود الخطأ المختلفة وأعداد الشرط المقابلة لـ
!>     الخطأ النسبي المعياري، والذي يتم تعريفه على النحو التالي:
!>
!>     الخطأ النسبي المعياري في متجه الحل i:
!>             max_j (abs(XTRUE(j,i) - X(j,i)))
!>            ------------------------------
!>                  max_j abs(X(j,i))
!>
!>     يتم فهرسة المصفوفة حسب نوع الخطأ

هذا الموقع غير مرتبط بـ Arch Linux ولا تابع له بأي شكل من الأشكال. المحتوى هو ترجمة فقط لصفحات دليل Arch المنشورة على man.archlinux.org.

ERR_BNDS_NORM

!>     ERR_BNDS_NORM is REAL array, dimension (NRHS, N_ERR_BNDS)
!> For each right-hand side, this array contains information about
!> various error bounds and condition numbers corresponding to the
!> normwise relative error, which is defined as follows:
!>
!> normwise relative error in the ith solution vector:
!>                abs(XTRUE(j,i) - X(j,i))
!>         max_j ----------------------
!>                    norm(X(j,i), inf)
!>
!> The array is indexed by the right-hand side i (on which the
!> normwise relative error depends), and the type of error
!> information as described below. There currently are up to three pieces of information
!> returned.
!>
!> The first index in ERR_BNDS_NORM(i,:) corresponds to the ith
!> right-hand side.
!>
!> The second index in ERR_BNDS_NORM(:,err) contains the following
!> three fields:
!> err = 1  boolean. Trust the answer if the
!>          reciprocal condition number is less than the threshold
!>          sqrt(n) * slamch('Epsilon').
!>
!> err = 2  error bound: The estimated forward error,
!>          almost certainly within a factor of 10 of the true error
!>          so long as the next entry is greater than the threshold
!>          sqrt(n) * slamch('Epsilon'). This error bound should only
!>          be trusted if the previous boolean is true.
!>
!> err = 3  Reciprocal condition number: Estimated normwise
!>          reciprocal condition number.  Compared with the threshold
!>          sqrt(n) * slamch('Epsilon') to determine if the error
!>          estimate is . These reciprocal condition
!>          numbers are 1 / (norm(Z^{-1},inf) * norm(Z,inf)) for some
!>          appropriately scaled matrix Z.
!>          Let Z = S*A, where S scales each row by a power of the
!>          radix so all absolute row sums of Z are approximately 1.
!>
!> See Lapack Working Note 165 for further details and extra
!> cautions.

ERR_BNDS_COMP

!>          ERR_BNDS_COMP is REAL array, dimension (NRHS, N_ERR_BNDS)
!> For each right-hand side, this array contains information about
!> various error bounds and condition numbers corresponding to the
!> componentwise relative error, which is defined as follows:
!>
!> Componentwise relative error in the ith solution vector:
!>                abs(XTRUE(j,i) - X(j,i))
!>         max_j ----------------------
!>                     abs(X(j,i))
!>
!> The array is indexed by the right-hand side i (on which the
!> componentwise relative error depends), and the type of error
!> information as described below. There currently are up to three
!> pieces of information returned for each right-hand side. If
!> componentwise accuracy is not requested (PARAMS(3) = 0.0), then
!> ERR_BNDS_COMP is not accessed.  If N_ERR_BNDS < 3, then at most
!> the first (:,N_ERR_BNDS) entries are returned.
!>
!> The first index in ERR_BNDS_COMP(i,:) corresponds to the ith
!> right-hand side.
!>
!> The second index in ERR_BNDS_COMP(:,err) contains the following
!> three fields:
!> err = 1  boolean. Trust the answer if the
!>          reciprocal condition number is less than the threshold
!>          sqrt(n) * slamch('Epsilon').
!>
!> err = 2  error bound: The estimated forward error,
!>          almost certainly within a factor of 10 of the true error
!>          so long as the next entry is greater than the threshold
!>          sqrt(n) * slamch('Epsilon'). This error bound should only
!>          be trusted if the previous boolean is true.
!>
!> err = 3  Reciprocal condition number: Estimated componentwise
!>          reciprocal condition number.  Compared with the threshold
!>          sqrt(n) * slamch('Epsilon') to determine if the error
!>          estimate is . These reciprocal condition
!>          numbers are 1 / (norm(Z^{-1},inf) * norm(Z,inf)) for some
!>          appropriately scaled matrix Z.
!>          Let Z = S*(A*diag(x)), where x is the solution for the
!>          current right-hand side and S scales each row of
!>          A*diag(x) by a power of the radix so all absolute row
!>          sums of Z are approximately 1.
!>
!> See Lapack Working Note 165 for further details and extra
!> cautions.

NPARAMS

!>          NPARAMS is INTEGER
!>     Specifies the number of parameters set in PARAMS.  If <= 0, the
!>     PARAMS array is never referenced and default values are used.

PARAMS

!>          PARAMS is REAL array, dimension NPARAMS
!>     Specifies algorithm parameters.  If an entry is < 0.0, then
!>     that entry will be filled with default value used for that
!>     parameter.  Only positions up to NPARAMS are accessed; defaults
!>     are used for higher-numbered parameters.
!>
!>       PARAMS(LA_LINRX_ITREF_I = 1) : Whether to perform iterative
!>            refinement or not.
!>         Default: 1.0
!>            = 0.0:  No refinement is performed, and no error bounds are
!>                    computed.
!>            = 1.0:  Use the double-precision refinement algorithm,
!>                    possibly with doubled-single computations if the
!>                    compilation environment does not support DOUBLE
!>                    PRECISION.
!>              (other values are reserved for future use)
!>
!>       PARAMS(LA_LINRX_ITHRESH_I = 2) : Maximum number of residual
!>            computations allowed for refinement.
!>         Default: 10
!>         Aggressive: Set to 100 to permit convergence using approximate
!>                     factorizations or factorizations other than LU. If
!>                     the factorization uses a technique other than
!>                     Gaussian elimination, the guarantees in
!>                     err_bnds_norm and err_bnds_comp may no longer be
!>                     trustworthy.
!>
!>       PARAMS(LA_LINRX_CWISE_I = 3) : Flag determining if the code
!>            will attempt to find a solution with small componentwise
!!>            relative error in the double-precision algorithm.  Positive
!>            is true, 0.0 is false.
!>         Default: 1.0 (attempt componentwise convergence)

WORK

!>          WORK is REAL array, dimension (4*N)

IWORK

!>          IWORK is INTEGER array, dimension (N)

INFO

!>          INFO is INTEGER
!>       = 0:  Successful exit. The solution to every right-hand side is
!>         guaranteed.
!>       < 0:  If INFO = -i, the i-th argument had an illegal value
!>       > 0 and <= N:  U(INFO,INFO) is exactly zero.  The factorization
!>         has been completed, but the factor U is exactly singular, so
!>         the solution and error bounds could not be computed. RCOND = 0
!>         is returned.
!>       = N+J: The solution corresponding to the Jth right-hand side is
!>         not guaranteed. The solutions corresponding to other right-
!>         hand sides K with K > J may not be guaranteed as well, but
!>         only the first such right-hand side is reported. If a small
!>         componentwise error is not requested (PARAMS(3) = 0.0) then
!>         the Jth right-hand side is the first with a normwise error
!>         bound that is not guaranteed (the smallest J such
!>         that ERR_BNDS_NORM(J,1) = 0.0). By default (PARAMS(3) = 1.0)
!>         the Jth right-hand side is the first with either a normwise or
!>         componentwise error bound that is not guaranteed (the smallest
!>         J such that either ERR_BNDS_NORM(J,1) = 0.0 or
!>         ERR_BNDS_COMP(J,1) = 0.0). See the definition of
!>         ERR_BNDS_NORM(:,1) and ERR_BNDS_COMP(:,1). To get information
!>         about all of the right-hand sides check ERR_BNDS_NORM or
!>         ERR_BNDS_COMP.
