توثيق الوظيفة/الروتين الفرعي

subroutine sgelqf (integer m, integer n, real, dimension( lda, * ) a, integer lda, real, dimension( * ) tau, real, dimension( * ) work, integer lwork, integer info)

SGELQF

الغرض:

!>
!> SGELQF يحسب تحليل LQ لمصفوفة A الحقيقية من النوع M × N:
!>
!>    A = ( L 0 ) *  Q
!>
!> حيث:
!>
!>    Q هي مصفوفة متعامدة من النوع N × N؛
!>    L هي مصفوفة مثلثة سفلية من النوع M × M؛
!>    0 هي مصفوفة صفرية من النوع M × (N-M)، إذا كانت M < N.
!>
!>

المعاملات

M 
!>          M هو عدد صحيح (INTEGER)
!>          عدد صفوف المصفوفة A.  M >= 0.
!>

N

!>          N هو عدد صحيح (INTEGER)
!>          عدد أعمدة المصفوفة A.  N >= 0.
!>

A

!>          A هي مصفوفة حقيقية (REAL array)، ذات بعد (LDA,N)
!>          عند الإدخال، مصفوفة A من النوع M × N.
!>          عند الخروج، تحتوي العناصر على القطر وفي الأسفل من المصفوفة
!>          على المصفوفة شبه المنحرفة السفلية L من النوع m × min(m,n) (L
!>          مثلثة سفلية إذا كانت m <= n)؛ العناصر في الأعلى من القطر،
!>          مع المصفوفة TAU، تمثل المصفوفة المتعامدة Q كـ
!>          ضرب للمنكسرات الأولية (انظر التفاصيل الإضافية).
!>

LDA

!>          LDA هو عدد صحيح (INTEGER)
!>          البعد الرئيسي للمصفوفة A.  LDA >= max(1,M).
!>

TAU

!>          TAU هي مصفوفة حقيقية (REAL array)، ذات بعد (min(M,N))
!>          العوامل القياسية للمنكسرات الأولية (انظر التفاصيل الإضافية).
!>

WORK

!>          WORK هي مصفوفة حقيقية (REAL array)، ذات بعد (MAX(1,LWORK))
!>          عند الخروج، إذا كان INFO = 0، فإن WORK(1) تعيد LWORK الأمثل.
!>

LWORK

!>          LWORK هو عدد صحيح (INTEGER)
!>          بعد المصفوفة WORK.
!>          LWORK >= 1، إذا كان MIN(M,N) = 0، و LWORK >= M، بخلاف ذلك.
!>          لأداء أمثل LWORK >= M*NB، حيث NB هو
!>          حجم الكتلة الأمثل.
!>
!>          إذا كان LWORK = -1، فإنه يفترض استعلام مساحة عمل؛ الروتين
!>          يحسب فقط الحجم الأمثل للمصفوفة WORK، ويعيد
!>          هذه القيمة كمدخل أول للمصفوفة WORK، ولا يتم إصدار رسالة خطأ
!>          متعلقة بـ LWORK بواسطة XERBLA.
!>

INFO

!>          INFO هو عدد صحيح (INTEGER)
!>          = 0:  انتهاء ناجح
!>          < 0:  إذا كان INFO = -i، فإن المعامل i كان له قيمة غير قانونية
!>

تفاصيل إضافية:

!>
!>  يتم تمثيل المصفوفة Q كضرب للمنكسرات الأولية
!>
!>     Q = H(k) . . . H(2) H(1)، حيث k = min(m,n).
!>
!>  كل H(i) له الشكل
!>
!>     H(i) = I - tau * v * v**T
!>
!>  حيث tau هو عامل قياسي حقيقي، و v هو متجه حقيقي مع
!>  v(1:i-1) = 0 و v(i) = 1؛ يتم تخزين v(i+1:n) عند الخروج في A(i,i+1:n)،
!>  و tau في TAU(i).
!>
