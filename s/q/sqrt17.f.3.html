real function sqrt17 (character trans, integer iresid, integer m, integer n, integer nrhs, real, dimension( lda, * ) a, integer lda, real, dimension( ldx, * ) x, integer ldx, real, dimension( ldb, * ) b, integer ldb, real, dimension( ldb, * ) c, real, dimension( lwork ) work, integer lwork)

SQRT17

الغرض:

!>
!> SQRT17 تحسب النسبة
!>
!>    norm(R**T * op(A)) / ( norm(A) * alpha * max(M,N,NRHS) * EPS ),
!>
!> حيث R = B - op(A)*X، op(A) هي A أو A**T، اعتمادًا على TRANS، EPS
!> هي أدنى قيمة لعتبة الآلة، و
!>
!>    alpha = norm(B) إذا كان IRESID = 1 (مشكلة صفرية الباقي)
!>    alpha = norm(R) إذا كان IRESID = 2 (خلاف ذلك).
!>
!> المعيار المستخدم هو المعيار 1.
!>

المعلمات

TRANS 
!>          TRANS هو CHARACTER*1
!>          تحدد ما إذا كانت منقول A مستخدمة أم لا.
!>          = 'N':  لا يوجد منقول، op(A) = A.
!>          = 'T':  منقول، op(A) = A**T.
!>

IRESID

!>          IRESID هو INTEGER
!>          IRESID = 1 يشير إلى مشكلة صفرية الباقي.
!>          IRESID = 2 يشير إلى باقٍ غير صفري.
!>

M

!>          M هو INTEGER
!>          عدد صفوف المصفوفة A.
!>          إذا كان TRANS = 'N'، فإن عدد صفوف المصفوفة B.
!>          إذا كان TRANS = 'T'، فإن عدد صفوف المصفوفة X.
!>

N

!>          N هو INTEGER
!>          عدد أعمدة المصفوفة A.
!>          إذا كان TRANS = 'N'، فإن عدد صفوف المصفوفة X.
!>          إذا كان TRANS = 'T'، فإن عدد صفوف المصفوفة B.
!>

NRHS

!>          NRHS هو INTEGER
!>          عدد أعمدة المصفوفات X و B.
!>

A

!>          A هي مصفوفة REAL، الأبعاد (LDA,N)
!>          المصفوفة A ذات الأبعاد m-by-n.
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرئيسي للمصفوفة A. LDA >= M.
!>

X

!>          X هي مصفوفة REAL، الأبعاد (LDX,NRHS)
!>          إذا كان TRANS = 'N'، فإن المصفوفة X ذات الأبعاد n-by-nrhs.
!>          إذا كان TRANS = 'T'، فإن المصفوفة X ذات الأبعاد m-by-nrhs.
!>

LDX

!>          LDX هو INTEGER
!>          البعد الرئيسي للمصفوفة X.
!>          إذا كان TRANS = 'N'، فإن LDX >= N.
!>          إذا كان TRANS = 'T'، فإن LDX >= M.
!>

B

!>          B هي مصفوفة REAL، الأبعاد (LDB,NRHS)
!>          إذا كان TRANS = 'N'، فإن المصفوفة B ذات الأبعاد m-by-nrhs.
!>          إذا كان TRANS = 'T'، فإن المصفوفة B ذات الأبعاد n-by-nrhs.
!>

LDB

!>          LDB هو INTEGER
!>          البعد الرئيسي للمصفوفة B.
!>          إذا كان TRANS = 'N'، فإن LDB >= M.
!>          إذا كان TRANS = 'T'، فإن LDB >= N.
!>

C

!>          C هي مصفوفة REAL، الأبعاد (LDB,NRHS)
!>

WORK

!>          WORK هي مصفوفة REAL، الأبعاد (LWORK)
!>

LWORK

!>          LWORK هو INTEGER
!>          طول المصفوفة WORK.  LWORK >= NRHS*(M+N).
!>
