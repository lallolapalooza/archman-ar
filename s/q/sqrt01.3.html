توثيق الدالة/الروتين الفرعي

subroutine sqrt01 (integer m, integer n, real, dimension( lda, * ) a, real, dimension( lda, * ) af, real, dimension( lda, * ) q, real, dimension( lda, * ) r, integer lda, real, dimension( * ) tau, real, dimension( lwork ) work, integer lwork, real, dimension( * ) rwork, real, dimension( * ) result)

SQRT01

الغرض:

!>
!> SQRT01 تختبر SGEQRF، التي تحسب تحليل QR للمصفوفة A ذات الأبعاد m×n،
!> وتختبر جزئيًا SORGQR التي تشكل المصفوفة المتعامدة Q ذات الأبعاد m×m.
!>
!> تقارن SQRT01 بين R و Q'*A، وتتحقق من أن Q متعامدة.
!>

المعلمات

M 
!>          M هو عدد صحيح (INTEGER)
!>          عدد صفوف المصفوفة A.  M >= 0.
!>

N

!>          N هو عدد صحيح (INTEGER)
!>          عدد أعمدة المصفوفة A.  N >= 0.
!>

A

!>          A هي مصفوفة حقيقية (REAL array)، ذات البعد (LDA,N)
!>          المصفوفة A ذات الأبعاد m×n.
!>

AF

!>          AF هي مصفوفة حقيقية (REAL array)، ذات البعد (LDA,N)
!>          تفاصيل تحليل QR للمصفوفة A، كما تم إرجاعها بواسطة SGEQRF.
!>          راجع SGEQRF لمزيد من التفاصيل.
!>

Q

!>          Q هي مصفوفة حقيقية (REAL array)، ذات البعد (LDA,M)
!>          المصفوفة المتعامدة Q ذات الأبعاد m×m.
!>

R

!>          R هي مصفوفة حقيقية (REAL array)، ذات البعد (LDA,max(M,N))
!>

LDA

!>          LDA هو عدد صحيح (INTEGER)
!>          البعد الرائد للمصفوفات A, AF, Q, و R.
!>          LDA >= max(M,N).
!>

TAU

!>          TAU هي مصفوفة حقيقية (REAL array)، ذات البعد (min(M,N))
!>          العوامل القياسية للمنعكسات الأولية، كما تم إرجاعها
!>          بواسطة SGEQRF.
!>

WORK

!>          WORK هي مصفوفة حقيقية (REAL array)، ذات البعد (LWORK)
!>

LWORK

!>          LWORK هو عدد صحيح (INTEGER)
!>          بعد المصفوفة WORK.
!>

RWORK

!>          RWORK هي مصفوفة حقيقية (REAL array)، ذات البعد (M)
!>

RESULT

!>          RESULT هي مصفوفة حقيقية (REAL array)، ذات البعد (2)
!>          نسب الاختبار:
!>          RESULT(1) = norm( R - Q'*A ) / ( M * norm(A) * EPS )
!>          RESULT(2) = norm( I - Q'*Q ) / ( M * EPS )
!>
