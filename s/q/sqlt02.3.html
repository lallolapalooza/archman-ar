توثيق الوظيفة/الإجراء الفرعي

subroutine sqlt02 (integer m, integer n, integer k, real, dimension( lda, * ) a, real, dimension( lda, * ) af, real, dimension( lda, * ) q, real, dimension( lda, * ) l, integer lda, real, dimension( * ) tau, real, dimension( lwork ) work, integer lwork, real, dimension( * ) rwork, real, dimension( * ) result)

SQLT02

الغرض:

!>
!> SQLT02 يختبر SORGQL، الذي ينشئ مصفوفة m-by-n Q بأعمدة متعامدة
!> تُعرّف بأنها حاصل ضرب k عواكس أولية.
!>
!> بالنظر إلى تحليل QL لمصفوفة m-by-n A، يقوم SQLT02 بإنشاء
!> المصفوفة المتعامدة Q التي يحددها تحليل الأعمدة k الأخيرة
!> من A؛ يقارن L(m-n+1:m,n-k+1:n) مع
!> Q(1:m,m-n+1:m)'*A(1:m,n-k+1:n)، ويتحقق من أن أعمدة Q
!> متعامدة.
!>

المعلمات

M 
!>          M هو عدد صحيح (INTEGER).
!>          عدد الصفوف للمصفوفة Q المراد إنشاؤها.  M >= 0.
!>

N

!>          N هو عدد صحيح (INTEGER).
!>          عدد الأعمدة للمصفوفة Q المراد إنشاؤها.
!>          M >= N >= 0.
!>

K

!>          K هو عدد صحيح (INTEGER).
!>          عدد العواكس الأولية التي تحدد حاصل ضربها
!>          المصفوفة Q. N >= K >= 0.
!>

A

!>          A هي مصفوفة حقيقية (REAL array)، البعد (LDA,N)
!>          المصفوفة A التي تم تحليلها بواسطة SQLT01.
!>

AF

!>          AF هي مصفوفة حقيقية (REAL array)، البعد (LDA,N)
!>          تفاصيل تحليل QL لـ A، كما تم إرجاعها بواسطة SGEQLF.
!>          انظر SGEQLF لمزيد من التفاصيل.
!>

Q

!>          Q هي مصفوفة حقيقية (REAL array)، البعد (LDA,N)
!>

L

!>          L هي مصفوفة حقيقية (REAL array)، البعد (LDA,N)
!>

LDA

!>          LDA هو عدد صحيح (INTEGER).
!>          البعد الرئيسي للمصفوفات A، AF، Q، و L. LDA >= M.
!>

TAU

!>          TAU هي مصفوفة حقيقية (REAL array)، البعد (N)
!>          العوامل القياسية للعواكس الأولية المقابلة
!>          لتحليل QL في AF.
!>

WORK

!>          WORK هي مصفوفة حقيقية (REAL array)، البعد (LWORK)
!>

LWORK

!>          LWORK هو عدد صحيح (INTEGER).
!>          بعد المصفوفة WORK.
!>

RWORK

!>          RWORK هي مصفوفة حقيقية (REAL array)، البعد (M)
!>

RESULT

!>          RESULT هي مصفوفة حقيقية (REAL array)، البعد (2)
!>          نسب الاختبار:
!>          RESULT(1) = norm( L - Q'*A ) / ( M * norm(A) * EPS )
!>          RESULT(2) = norm( I - Q'*Q ) / ( M * EPS )
!>
