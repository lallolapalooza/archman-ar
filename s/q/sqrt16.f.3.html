توثيق الوظيفة/الروتين الفرعي

subroutine sqrt16 (character trans, integer m, integer n, integer nrhs, real, dimension( lda, * ) a, integer lda, real, dimension( ldx, * ) x, integer ldx, real, dimension( ldb, * ) b, integer ldb, real, dimension( * ) rwork, real resid)

SQRT16

الغرض:

!>
!> SQRT16 تحسب الباقي لحل نظام المعادلات الخطية  A*x = b  أو  A'*x = b:
!>    RESID = norm(B - A*X) / ( max(m,n) * norm(A) * norm(X) * EPS ),
!> حيث EPS هو الخطأ الآلي.
!>

المعلمات

TRANS 
!>          TRANS هو CHARACTER*1
!>          يحدد شكل نظام المعادلات:
!>          = 'N':  A *x = b
!>          = 'T':  A'*x = b, حيث A' هو منقول A
!>          = 'C':  A'*x = b, حيث A' هو منقول A
!>

M

!>          M هو INTEGER
!>          عدد صفوف المصفوفة A.  M >= 0.
!>

N

!>          N هو INTEGER
!>          عدد أعمدة المصفوفة A.  N >= 0.
!>

NRHS

!>          NRHS هو INTEGER
!>          عدد أعمدة B، مصفوفة الأطراف اليمنى.
!>          NRHS >= 0.
!>

A

!>          A هي مصفوفة REAL، الأبعاد (LDA,N)
!>          المصفوفة الأصلية A بحجم M x N.
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرائد للمصفوفة A.  LDA >= max(1,M).
!>

X

!>          X هي مصفوفة REAL، الأبعاد (LDX,NRHS)
!>          حلول متجهات الحل المحسوبة لنظام المعادلات الخطية.
!>

LDX

!>          LDX هو INTEGER
!>          البعد الرائد للمصفوفة X.  إذا كان TRANS = 'N'،
!>          LDX >= max(1,N)؛ إذا كان TRANS = 'T' أو 'C'، LDX >= max(1,M).
!>

B

!>          B هي مصفوفة REAL، الأبعاد (LDB,NRHS)
!>          عند الدخول، متجهات الطرف الأيمن لنظام المعادلات الخطية.
!>          عند الخروج، يتم استبدال B بالفرق B - A*X.
!>

LDB

!>          LDB هو INTEGER
!>          البعد الرائد للمصفوفة B.  إذا كان TRANS = 'N'،
!>          LDB >= max(1,M)؛ إذا كان TRANS = 'T' أو 'C'، LDB >= max(1,N).
!>

RWORK

!>          RWORK هي مصفوفة REAL، الأبعاد (M)
!>

RESID

!>          RESID هو REAL
!>          الحد الأقصى لعدد الأطراف اليمنى لـ
!>          norm(B - A*X) / ( max(m,n) * norm(A) * norm(X) * EPS ).
!>
