توثيق الوظيفة/الروتين الفرعي

subroutine sqlt03 (integer m, integer n, integer k, real, dimension( lda, * ) af, real, dimension( lda, * ) c, real, dimension( lda, * ) cc, real, dimension( lda, * ) q, integer lda, real, dimension( * ) tau, real, dimension( lwork ) work, integer lwork, real, dimension( * ) rwork, real, dimension( * ) result)

SQLT03

الغرض:

!>
!> SQLT03 تختبر SORMQL، التي تحسب Q*C، Q'*C، C*Q أو C*Q'.
!>
!> تقارن SQLT03 نتائج استدعاء SORMQL مع نتائج
!> تشكيل Q صراحةً عن طريق استدعاء SORGQL ثم إجراء ضرب المصفوفات
!> عن طريق استدعاء SGEMM.
!>

المعلمات

M 
!>          M هو عدد صحيح (INTEGER)
!>          ترتيب المصفوفة المتعامدة Q.  M >= 0.
!>

N

!>          N هو عدد صحيح (INTEGER)
!>          عدد الصفوف أو الأعمدة للمصفوفة C؛ C هي m-by-n إذا
!>          تم تطبيق Q من اليسار، أو n-by-m إذا تم تطبيق Q من
!>          اليمين.  N >= 0.
!>

K

!>          K هو عدد صحيح (INTEGER)
!>          عدد المعكوسات الأولية التي يُعرّف حاصل ضربها
!>          المصفوفة المتعامدة Q.  M >= K >= 0.
!>

AF

!>          AF هي مصفوفة حقيقية (REAL array)، البعد (LDA,N)
!>          تفاصيل تحليل QL لمصفوفة m-by-n، كما
!>          تم إرجاعها بواسطة SGEQLF. انظر SGEQLF لمزيد من التفاصيل.
!>

C

!>          C هي مصفوفة حقيقية (REAL array)، البعد (LDA,N)
!>

CC

!>          CC هي مصفوفة حقيقية (REAL array)، البعد (LDA,N)
!>

Q

!>          Q هي مصفوفة حقيقية (REAL array)، البعد (LDA,M)
!>

LDA

!>          LDA هو عدد صحيح (INTEGER)
!>          البعد الرئيسي للمصفوفات AF، C، CC، و Q.
!>

TAU

!>          TAU هي مصفوفة حقيقية (REAL array)، البعد (min(M,N))
!>          العوامل القياسية للمعكوسات الأولية المقابلة
!>          لتحليل QL في AF.
!>

WORK

!>          WORK هي مصفوفة حقيقية (REAL array)، البعد (LWORK)
!>

LWORK

!>          LWORK هو عدد صحيح (INTEGER)
!>          طول WORK. يجب أن يكون LWORK على الأقل M، ويجب أن يكون
!>          M*NB، حيث NB هو حجم الكتلة لهذه البيئة.
!>

RWORK

!>          RWORK هي مصفوفة حقيقية (REAL array)، البعد (M)
!>

RESULT

!>          RESULT هي مصفوفة حقيقية (REAL array)، البعد (4)
!>          تقارن نسب الاختبار بين تقنيتين لضرب مصفوفة عشوائية C
!>          بمصفوفة متعامدة m-by-m Q.
!>          RESULT(1) = norm( Q*C - Q*C )  / ( M * norm(C) * EPS )
!>          RESULT(2) = norm( C*Q - C*Q )  / ( M * norm(C) * EPS )
!>          RESULT(3) = norm( Q'*C - Q'*C )/ ( M * norm(C) * EPS )
!>          RESULT(4) = norm( C*Q' - C*Q' )/ ( M * norm(C) * EPS )
!>
