وثائق الوظيفة/الروتين الفرعي

subroutine sqrt02 (integer m, integer n, integer k, real, dimension( lda, * ) a, real, dimension( lda, * ) af, real, dimension( lda, * ) q, real, dimension( lda, * ) r, integer lda, real, dimension( * ) tau, real, dimension( lwork ) work, integer lwork, real, dimension( * ) rwork, real, dimension( * ) result)

SQRT02

الغرض:

!>
!> SQRT02 تختبر SORGQR، والتي تولد مصفوفة Q بحجم m×n بأعمدة متعامدة
!> معرفة كناتج لـ k عواكس ابتدائية.
!>
!> بالنظر إلى تحليل QR للمصفوفة A بحجم m×n، تقوم SQRT02 بتوليد
!> المصفوفة المتعامدة Q المعرفة بتحليل أول k أعمدة من A؛ وهي تقارن
!> R(1:n,1:k) مع Q(1:m,1:n)'*A(1:m,1:k)،
!> وتتحقق من أن أعمدة Q متعامدة.
!>

المعلمات:

M 
!>          M هو عدد صحيح (INTEGER)
!>          عدد صفوف المصفوفة Q المراد توليدها.  M >= 0.
!>

N

!>          N هو عدد صحيح (INTEGER)
!>          عدد أعمدة المصفوفة Q المراد توليدها.
!>          M >= N >= 0.
!>

K

!>          K هو عدد صحيح (INTEGER)
!>          عدد العواكس الابتدائية التي يحدد ناتجها المصفوفة Q.
!>          N >= K >= 0.
!>

A

!>          A هي مصفوفة حقيقية (REAL)، ذات بعد (LDA,N)
!>          المصفوفة A بحجم m×n والتي تم تحليلها بواسطة SQRT01.
!>

AF

!>          AF هي مصفوفة حقيقية (REAL)، ذات بعد (LDA,N)
!>          تفاصيل تحليل QR للمصفوفة A، كما تم إرجاعها بواسطة SGEQRF.
!>          راجع SGEQRF لمزيد من التفاصيل.
!>

Q

!>          Q هي مصفوفة حقيقية (REAL)، ذات بعد (LDA,N)
!>

R

!>          R هي مصفوفة حقيقية (REAL)، ذات بعد (LDA,N)
!>

LDA

!>          LDA هو عدد صحيح (INTEGER)
!>          البعد الرائد للمصفوفات A، AF، Q، و R. LDA >= M.
!>

TAU

!>          TAU هي مصفوفة حقيقية (REAL)، ذات بعد (N)
!>          العوامل القياسية للعواكس الابتدائية المقابلة لتحليل QR في AF.
!>

WORK

!>          WORK هي مصفوفة حقيقية (REAL)، ذات بعد (LWORK)
!>

LWORK

!>          LWORK هو عدد صحيح (INTEGER)
!>          بعد المصفوفة WORK.
!>

RWORK

!>          RWORK هي مصفوفة حقيقية (REAL)، ذات بعد (M)
!>

RESULT

!>          RESULT هي مصفوفة حقيقية (REAL)، ذات بعد (2)
!>          نسب الاختبار:
!>          RESULT(1) = norm( R - Q'*A ) / ( M * norm(A) * EPS )
!>          RESULT(2) = norm( I - Q'*Q ) / ( M * EPS )
!>
