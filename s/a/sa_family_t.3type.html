    sa_family_t     ss_family;      /* عائلة العنوان */
};

مقابس مجال الإنترنت

#include <netinet/in.h>
struct sockaddr_in {
    sa_family_t     sin_family;     /* AF_INET */
    in_port_t       sin_port;       /* رقم المنفذ */
    struct in_addr  sin_addr;       /* عنوان IPv4 */
};
struct sockaddr_in6 {
    sa_family_t     sin6_family;    /* AF_INET6 */
    in_port_t       sin6_port;      /* رقم المنفذ */
    uint32_t        sin6_flowinfo;  /* معلومات تدفق IPv6 */
    struct in6_addr sin6_addr;      /* عنوان IPv6 */
    uint32_t        sin6_scope_id;  /* مجموعة الواجهات للنطاق */
};
struct in_addr {
    in_addr_t s_addr;
};
struct in6_addr {
    uint8_t   s6_addr[16];
};
typedef uint32_t in_addr_t;
typedef uint16_t in_port_t;

مقابس مجال UNIX

#include <sys/un.h>
struct sockaddr_un {
    sa_family_t     sun_family;     /* عائلة العنوان */
    char            sun_path[];     /* مسار اسم المقبس */
};

الوصف

sockaddr
يصف عنوان المقبس.
sockaddr_storage
هيكل بحجم كافٍ على الأقل لأي هياكل عنوان أخرى من نوع sockaddr_*. تمت محاذاته بحيث يمكن تحويل مؤشر إليه كمؤشر إلى هياكل sockaddr_* أخرى واستخدامه للوصول إلى حقوله.
socklen_t
يصف طول عنوان المقبس. هذا نوع صحيح لا يقل عن 32 بت.
sa_family_t
يصف عائلة بروتوكول المقبس. هذا نوع صحيح غير موقع.

مقابس مجال الإنترنت

sockaddr_in
يصف عنوان مقبس مجال إنترنت IPv4. يتم تخزين عضوين sin_port و sin_addr بترتيب بايت الشبكة.
sockaddr_in6
يصف عنوان مقبس مجال إنترنت IPv6. يستخدم مصفوفة sin6_addr.s6_addr لتحتوي على عنوان IPv6 بحجم 128 بت، ويتم تخزينه بترتيب بايت الشبكة.

مقابس مجال UNIX

sockaddr_un
يصف عنوان مقبس مجال UNIX.

المعايير

POSIX.1-2024.

التاريخ

POSIX.1-2001.

تم اختراع socklen_t بواسطة POSIX. انظر أيضًا accept(2).

تم اختراع هذه الهياكل قبل قواعد ISO C الحديثة الخاصة بالتحويل المباشر (strict-aliasing). إذا تم تطبيق قواعد التحويل المباشر بصرامة، فسيكون استخدام هذه الهياكل صعبًا للغاية دون استدعاء سلوك غير محدد. قام POSIX.1-2024 بتصحيح ذلك من خلال مطالبة التطبيقات بالتأكد من أنه يمكن استخدام هذه الهياكل بأمان كما تم تصميمها.

ملاحظات

يتم تعريف socklen_t أيضًا في <netdb.h>.

يتم تعريف sa_family_t أيضًا في <netinet/in.h> و <sys/un.h>.
