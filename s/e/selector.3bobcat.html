FBB::Selector(3bobcat) تأخيرات زمنية، إدخال/إخراج ملفات متعددة FBB::Selector(3bobcat)

الاسم

FBB::Selector - تأخيرات زمنية، إنذارات وإدخال/إخراج ملفات متعددة.

الملخص

#include <bobcat/selector>
خيار الربط: -lbobcat

الوصف

تعد كائنات FBB::Selector أغلفة حول استدعاءات نظام select(2) وتسمح بالتأخيرات الزمنية، وظائف الإنذار و/أو إدخال/إخراج ملفات متعددة. يتطلب استخدام واصفات الملفات، التي ليست جزءًا رسميًا من C++. ومع ذلك، تقدم معظم أنظمة التشغيل واصفات الملفات. المقابس هي واصفات ملفات معروفة.

مساحة الاسم

FBB
جميع المنشئات، الأعضاء، المعاملات والمعدلات المذكورة في صفحة الدليل هذه، معرّفة في مساحة الاسم FBB.

int exceptFd():
يعيد -1 إذا لم تعد هناك واصفات ملفات متاحة في فئة الاستثناء. بخلاف ذلك، يتم إعادة واصف الملف التالي المتاح في فئة الاستثناء. عند العودة من wait، يمكن استدعاء هذه الدالة بشكل متكرر حتى يتم إرجاع -1، مع خدمة كل واصف ملف متاح بالتتابع.
o
void noAlarm():
يمنع هذا العضو أي إنذار بسبب انتهاء الوقت من الحدوث.
o
int nReady():
يعيد عدد واصفات الملفات المتاحة. يتم إرجاع 0 عند انتهاء الوقت، ويتم إرجاع -1: عندما فشل select(2) نفسه.
o
int readFd():
يعيد -1 إذا لم تعد هناك واصفات ملفات متاحة للقراءة. بخلاف ذلك، يتم إعادة واصف الملف التالي المتاح للقراءة. عند العودة من wait، يمكن استدعاء هذه الدالة بشكل متكرر حتى يتم إرجاع -1، مع خدمة كل واصف ملف متاح بالتتابع. لاحظ أن الملف الذي يتم إرجاع واصف الملف الخاص به بواسطة readFd قد يكون أيضًا في موضع نهاية الملف الخاص به. الملف "جاهز للقراءة"، ولكن لن يتم إرجاع أي أحرف عند محاولة القراءة منه بسبب حالة نهاية الملف الخاصة به. في هذه الحالة، من الأفضل على الأرجح إزالة واصف الملف من مجموعة واصفات الملفات النشطة.

o
void setAlarm(int sec, int usec = 0):
يضبط هذا العضو الإنذار على الثواني والميكرو ثواني المحددة. إذا لم يحدث أي إجراء على أحد واصفات الملفات التي تتم مراقبتها بعد انقضاء الوقت المحدد، فسوف يعود wait مع إرجاع nReady بالقيمة 0. قد لا يكون وقت الإنذار المطلوب (sec + usec / 1e+6) سالبًا وقد لا يتجاوز std::numeric_limits<int>::max() أو سيتم طرح استثناء FBB::Exception. يؤدي تحديد وقت إنذار بقيمة 0 إلى عودة wait فورًا. لإيقاف وقت الإنذار استخدم noAlarm.
o
int wait():
يجب استدعاء هذا العضو للانتظار حتى تحدث أنشطة على واصفات الملفات المثبتة أو فترة انتهاء الوقت. تظهر الأعضاء exceptFd, nReady, readFd و writeFd سلوكياتها المحددة فقط بعد عودة wait.
يقوم بطرح استثناء FBB::Exception عند فشل select(2)، مما قد يشير بقوة إلى نهاية أي إدخال متاح. يتم طرح استثناء أيضًا إذا تلقى البرنامج إشارة.
إذا عاد wait بشكل طبيعي، فإن قيمة الإرجاع تمثل عدد واصفات الملفات المتاحة. لاحظ أن wait قد يعود أيضًا مع واصف ملف إدخال تم إرجاعه بواسطة readFd لملف في موضع نهاية الملف الخاص به. الملف "جاهز للقراءة"، ولكن لن يتم إرجاع أي أحرف عند محاولة القراءة منه بسبب حالة نهاية الملف الخاصة به.
o
int writeFd():
يعيد -1 إذا لم تعد هناك واصفات ملفات متاحة للكتابة. بخلاف ذلك، يتم إعادة واصف الملف التالي المتاح للكتابة. عند العودة من wait، يمكن استدعاء هذه الدالة بشكل متكرر حتى يتم إرجاع -1، مع خدمة كل واصف ملف متاح بالتتابع.

مثال

#include <string>
#include <iostream>
#include <bobcat/selector>
#include <bobcat/exception>
using namespace std;
using namespace FBB;
int main(int argc, char **argv, char **envp)
try
{
    Selector selector;
    selector.setAlarm(5);               // كل 5 ثوانٍ: الإنذار يطلق
    selector.addReadFd(STDIN_FILENO);   // انظر أيضًا إلى cin
    while (true)
    {
        if (!selector.wait())           // 0: الإنذار يطلق
            cout << "Are you still there?" << endl;
        else
        {
            string s;
            if (!getline(cin, s) || !s.length())
                return 0;
            cout << "Thank you for: " << s << endl;
        }
    }
}
catch (Exception const &e)
{
    cout << e.what() << ’\n’;
    return 1;
}
