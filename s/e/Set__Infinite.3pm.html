Set__Infinite(3pm)

Set::Infinite(3) توثيق Perl من مساهمة المستخدم Set::Infinite(3)

الاسم

Set::Infinite - مجموعات الفترات

الملخص

use Set::Infinite;
$set = Set::Infinite->new(1,2);    # [1..2]
print $set->union(5,6);            # [1..2],[5..6]

الوصف

Set::Infinite هي وحدة نظرية المجموعات للمجموعات اللانهائية.

المجموعة هي مجموعة من الكائنات. الكائنات التي تنتمي إلى مجموعة تسمى أعضائها، أو "عناصرها".

نسمح (تقريبًا) بأي شيء ككائنات: الأعداد الحقيقية، الصحيحة، والكائنات (مثل التواريخ).

نسمح للمجموعات بأن تكون لانهائية.

لا يتم حساب ترتيب العناصر. على سبيل المثال، {1,2} = {2,1}.

لا يتم حساب تكرار العناصر. على سبيل المثال، {1,2,2} = {1,1,1,2} = {1,2}.

المُنشئ

جديد

ينشئ كائن مجموعة جديد:

$set = Set::Infinite->new;             # مجموعة فارغة
$set = Set::Infinite->new( 10 );       # عنصر واحد
$set = Set::Infinite->new( 10, 20 );   # فترة واحدة
$set = Set::Infinite->new( 
          [ 10, 20 ], [ 50, 70 ] );    # فترتان
مجموعة فارغة
$set = Set::Infinite->new;
مجموعة بعنصر واحد
$set = Set::Infinite->new( 10 );
$set = Set::Infinite->new( [ 10 ] );
مجموعة بفاصل زمني واحد
$set = Set::Infinite->new( 10, 20 );
$set = Set::Infinite->new( [ 10, 20 ] );
# 10 <= x <= 20
مجموعة بفاصل زمني واحد، مفتوح
$set = Set::Infinite->new(
    {
        a => 10, open_begin => 0,
        b => 20, open_end => 1,
    }
);
# 10 <= x < 20
مجموعة بفواصل متعددة
$set = Set::Infinite->new( 10, 20,  100, 200 );
$set = Set::Infinite->new( [ 10, 20 ], [ 100, 200 ] );
$set = Set::Infinite->new(
    {
        a => 10, open_begin => 0,
        b => 20, open_end => 0,
    },
    {
        a => 100, open_begin => 0,
        b => 200, open_end => 0,
    }
);

تتوقع الطريقة new() معاملات *مرتبة*.

إذا كانت لديك فترات غير مرتبة، يمكنك بناء المجموعة باستخدام "union":

@ranges = ( [ 10, 20 ], [ -10, 1 ] );
$set = Set::Infinite->new;
$set = $set->union( @$_ ) for @ranges;

يجب أن تكون هياكل البيانات التي تم تمريرها إلى "new" *غير قابلة للتغيير*. لذلك هذا ليس ممارسة جيدة:

$set = Set::Infinite->new( $object_a, $object_b );
$object_a->set_value( 10 );

هذه هي الطريقة الموصى بها للقيام بذلك:

$set = Set::Infinite->new( $object_a->clone, $object_b->clone );
$object_a->set_value( 10 );

استنساخ / نسخ

ينشئ كائنًا جديدًا، وينسخ بيانات الكائن.

مجموعة فارغة

ينشئ مجموعة فارغة.

إذا تم استدعاؤه من مجموعة موجودة، ترث المجموعة الفارغة خصائص "النوع" و"الكثافة".

المجموعة الشاملة

ينشئ مجموعة تحتوي على "جميع" العناصر الممكنة.

إذا تم استدعاؤه من مجموعة موجودة، ترث المجموعة الشاملة خصائص "النوع" و"الكثافة".

وظائف المجموعات

اتحاد

$set = $set->union($b);

يعيد المجموعة التي تحتوي على جميع العناصر من كلا المجموعتين.

تعمل هذه الوظيفة مثل عملية "أو".

$set1 = new Set::Infinite( [ 1, 4 ], [ 8, 12 ] );
$set2 = new Set::Infinite( [ 7, 20 ] );
print $set1->union( $set2 );
# الإخراج: [1..4],[7..20]

تقاطع

$set = $set->intersection($b);

يعيد المجموعة التي تحتوي على العناصر المشتركة بين المجموعتين.

تعمل هذه الوظيفة مثل عملية "و".

$set1 = new Set::Infinite( [ 1, 4 ], [ 8, 12 ] );
$set2 = new Set::Infinite( [ 7, 20 ] );
print $set1->intersection( $set2 );
# الإخراج: [8..12]

المكمل

ناقص

فرق

$set = $set->complement;

يعيد المجموعة التي تحتوي على جميع العناصر التي لا تنتمي إلى المجموعة.

$set1 = new Set::Infinite( [ 1, 4 ], [ 8, 12 ] );
print $set1->complement;
# الإخراج: (-inf..1),(4..8),(12..inf)

قد تأخذ وظيفة المكمل معاملًا:

$set = $set->minus($b);

يعيد فرق المجموعات، أي العناصر التي لا تنتمي إلى المجموعة المعطاة.

$set1 = new Set::Infinite( [ 1, 4 ], [ 8, 12 ] );
$set2 = new Set::Infinite( [ 7, 20 ] );
print $set1->minus( $set2 );
# الإخراج: [1..4]

فرق متناظر

يعيد مجموعة تحتوي على العناصر الموجودة في إحدى المجموعتين، ولكن ليس كليهما. هذا هو إصدار "مجموعة" من "XOR".

طرق الكثافة

حقيقي

$set1 = $set->real;

يعيد مجموعة بكثافة "0".

صحيح

$set1 = $set->integer;

يعيد مجموعة بكثافة "1".

وظائف المنطق

يتقاطع

$logic = $set->intersects($b);

يحتوي

$logic = $set->contains($b);

هل فارغة

هل صفرية

$logic = $set->is_null;

هل غير فارغة

هذه المجموعة التي تحتوي على عنصر واحد على الأقل.

هل فاصل

هذه المجموعة التي لها فاصل زمني أو فترة واحدة.

هل مفردة

هذه المجموعة التي تحتوي على عنصر واحد.

هل مجموعة فرعية ( $set )

كل عنصر في هذه المجموعة هو عضو في المجموعة المعطاة.

هل مجموعة فرعية صحيحة ( $set )

كل عنصر في هذه المجموعة هو عضو في المجموعة المعطاة. بعض أعضاء المجموعة المعطاة ليسوا عناصر في هذه المجموعة.

هل منفصلة ( $set )

المجموعة المعطاة لا تشترك في أي عناصر مع هذه المجموعة.

هل معقدة جدًا

في بعض الأحيان قد تكون المجموعة معقدة للغاية بحيث لا يمكن تعدادها أو طباعتها.

يحدث هذا مع المجموعات التي تمثل تكرارات لا نهائية، كما هو الحال عند طلب تكميم على مجموعة محدودة بـ -inf أو inf.

دوال المجموعات الخاصة

span

$set1 = $set->span;

تعيد نطاق المجموعة.

until

توسع مجموعة حتى أخرى:

0,5,7 -> until 2,6,10

ينتج عنه:

[0..2), [5..6), [7..10)

start_set

تقوم هاتان الدالتان بعكس عمل الدالة "until".

نظرًا لـ:

[0..2), [5..6), [7..10)

يكون start_set:

0,5,7

ويكون end_set:

2,6,10

intersected_spans

$set = $set1->intersected_spans( $set2 );

تعيد الدالة مجموعة جديدة تحتوي على جميع النطاقات التي تتقاطع معها المجموعة المعطاة.

بخلاف الدالة "intersection"، لا يتم تعديل النطاقات. انظر الرسم البياني أدناه:

             set1   [....]   [....]   [....]   [....]
             set2      [................]
     intersection      [.]   [....]   [.]
intersected_spans   [....]   [....]   [....]

iterate

iterate ( sub { } , @args )

تكرر على نطاقات المجموعة، باستخدام روتين فرعي لاستدعاء رد الاتصال. تعيد اتحاد جميع النتائج الجزئية.

وسيطة رد الاتصال $_[0] هي نطاق. إذا كانت هناك وسائط إضافية، يتم تمريرها إلى رد الاتصال.

يمكن لرد الاتصال أن يعيد نطاقًا، أو hashref (انظر "Set::Infinite::Basic")، أو قيمة قياسية، أو كائنًا، أو "undef".

[تجريبي] تقبل "iterate" وسيطة اختيارية "backtrack_callback". الغرض من "backtrack_callback" هو *عكس* تشغيل iterate()، متجاوزًا قيود خوارزمية التراجع الداخلية. التنسيق هو:

iterate ( sub { } , backtrack_callback => sub { }, @args )

يمكن لـ "backtrack_callback" أن يعيد نطاقًا، أو hashref، أو قيمة قياسية، أو كائنًا، أو "undef".

على سبيل المثال، يضيف المقطع التالي قيمة ثابتة إلى كل عنصر في مجموعة غير محدودة:

$set1 = $set->iterate(
             sub { $_[0]->min + 54, $_[0]->max + 54 },
          backtrack_callback =>
             sub { $_[0]->min - 54, $_[0]->max - 54 },
          );

ملاحظات

  • ترميز "span" للمنشئ 
    $set = Set::Infinite->new(10,1);

    سيتم تفسيرها على أنها [1..10]

  • ترميز "multiple-span" للمنشئ 
    $set = Set::Infinite->new(1,2,3,4);

    سيتم تفسيرها على أنها [1..2],[3..4] بدلاً من [1,2,3,4]. ربما تريد ->new([1],[2],[3],[4]) أو ->new(1,4)

  • معامل "range" للمنشئ 
    $set = Set::Infinite->new(1..3);

    سيتم تفسيرها على أنها [1..2],3 بدلاً من [1,2,3]. ربما تريد ->new(1,3) بدلاً من ذلك.

تفاصيل داخلية

الكائن الأساسي set بدون تكرارات هو "Set::Infinite::Basic".

يتم تمثيل مجموعة التكرارات بواسطة اسم الدالة، وواحد أو اثنين من الكائنات الأصل، ومعلمات إضافية. يتم تعيين "list" إلى مصفوفة فارغة، ويتم تعيين "too_complex" إلى 1.

هذا هو هيكل يحتوي على اتحاد اثنين من المجموعات المعقدة:

{
  too_complex => 1,             # "this is a recurrence"
  list   => [ ],                # غير مستخدم
  method => 'union',            # اسم الدالة
  parent => [ $set1, $set2 ],   # "leaves" في شجرة الصيغة
  param  => [ ]                 # معلمات اختيارية للدالة
}

هذا هو هيكل يحتوي على مكمل مجموعة معقدة:

{
  too_complex => 1,             # "this is a recurrence"
  list   => [ ],                # غير مستخدم
  method => 'complement',       # اسم الدالة
  parent => $set,               # "leaf" في شجرة الصيغة
  param  => [ ]                 # معلمات اختيارية للدالة
}
