encrypt(3) دليل دوال المكتبة encrypt(3)

الاسم

encrypt, setkey, encrypt_r, setkey_r - تشفير رسائل 64 بت

المكتبة

مكتبة تجزئة كلمات المرور (libcrypt, -lcrypt)

الملخص

#define _XOPEN_SOURCE       /* انظر feature_test_macros(7) */
#include <unistd.h>
[[deprecated]] void encrypt(char block[64], int edflag);
#define _XOPEN_SOURCE       /* انظر feature_test_macros(7) */
#include <stdlib.h>
[[deprecated]] void setkey(const char *key);
#define _GNU_SOURCE         /* انظر feature_test_macros(7) */
#include <crypt.h>
[[deprecated]] void setkey_r(const char *key, struct crypt_data *data);
[[deprecated]] void encrypt_r(char *block, int edflag,
                              struct crypt_data *data);

الوصف

تقوم هذه الدوال بتشفير وفك تشفير رسائل 64 بت. تقوم الدالة setkey() بتعيين المفتاح المستخدم بواسطة encrypt(). المفتاح key المستخدم هنا هو مصفوفة من 64 بايت، كل منها له قيمة رقمية 1 أو 0. البايتات key[n] حيث n=8*i-1 يتم تجاهلها، بحيث يكون طول المفتاح الفعال 56 بت.

تقوم الدالة encrypt() بتعديل المخزن المؤقت الممرر، تشفير إذا كان edflag يساوي 0، وفك تشفير إذا تم تمرير 1. مثل وسيط المفتاح key، block هو تمثيل متجه بت للقيمة الفعلية التي يتم تشفيرها. يتم إرجاع النتيجة في نفس المتجه.

هاتان الدالتان ليستا قابلتين لإعادة الدخول، أي أن بيانات المفتاح موجودة في تخزين ثابت. الدوال setkey_r() و encrypt_r() هي الإصدارات القابلة لإعادة الدخول. يستخدمان البنية التالية لتخزين بيانات المفتاح:


struct crypt_data {
    char keysched[16 * 8];
    char sb0[32768];
    char sb1[32768];
    char sb2[32768];
    char sb3[32768];
    char crypt_3_buf[14];
    char current_salt[2];
    long current_saltbits;
    int  direction;
    int  initialized;
};

قبل استدعاء setkey_r() قم بتعيين data->initialized إلى صفر.

قيمة الإرجاع

هذه الدوال لا تعيد أي قيمة.

الأخطاء

قم بتعيين errno إلى الصفر قبل استدعاء الدوال أعلاه. عند النجاح، errno لا تتغير.

ENOSYS
الدالة غير متوفرة. (على سبيل المثال بسبب قيود التصدير الأمريكية السابقة.)

السمات

لشرح المصطلحات المستخدمة في هذا القسم، انظر attributes(7).

الواجهة السمة القيمة
encrypt (), setkey () أمان الخيط MT-Unsafe race:crypt
encrypt_r (), setkey_r () أمان الخيط MT-Safe

المعايير

encrypt()
setkey()
POSIX.1-2008.
encrypt_r()
setkey_r()
لا شيء.

التاريخ

تمت الإزالة في glibc 2.28.

نظرًا لأنها تستخدم خوارزمية DES لتشفير الكتل، والتي لم تعد تعتبر آمنة، فقد تمت إزالة هذه الدوال من glibc. يجب على التطبيقات التبديل إلى مكتبة تشفير حديثة، مثل libgcrypt.

encrypt()
setkey()
POSIX.1-2001, SUS, SVr4.

التوفر في glibc

انظر crypt(3).

الميزات في glibc

في glibc 2.2، تستخدم هذه الدوال خوارزمية DES.

أمثلة

#define _XOPEN_SOURCE
#include <crypt.h>
#include <stdio.h>
#include <stdlib.h>
#include <unistd.h>
int
main(void)
{
    char key[64];
    char orig[9] = "eggplant";
    char buf[64];
    char txt[9];
    for (size_t i = 0; i < 64; i++) {
        key[i] = rand() & 1;
    }
    for (size_t i = 0; i < 8; i++) {
        for (size_t j = 0; j < 8; j++) {
            buf[i * 8 + j] = orig[i] >> j & 1;
        }
        setkey(key);
    }
    printf("Before encrypting: %s\n", orig);
    encrypt(buf, 0);
    for (size_t i = 0; i < 8; i++) {
        for (size_t j = 0, txt[i] = '\0'; j < 8; j++) {
            txt[i] |= buf[i * 8 + j] << j;
        }
        txt[8] = '\0';
    }
    printf("After encrypting:  %s\n", txt);
    encrypt(buf, 1);
    for (size_t i = 0; i < 8; i++) {
        for (size_t j = 0, txt[i] = '\0'; j < 8; j++) {
            txt[i] |= buf[i * 8 + j] << j;
        }
        txt[8] = '\0';
    }
    printf("After decrypting:  %s\n", txt);
    exit(EXIT_SUCCESS);
}
