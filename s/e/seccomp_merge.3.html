seccomp_merge(3)

الحزمة: libseccomp

التراخيص: LGPL-2.1-only

seccomp_merge(3) وثائق libseccomp seccomp_merge(3)

الاسم

seccomp_merge - دمج مرشحين seccomp

الملخص

#include <seccomp.h>

typedef void * scmp_filter_ctx;

int seccomp_merge(scmp_filter_ctx dst, scmp_filter_ctx src);

الربط مع -lseccomp.

الوصف

تقوم الدالة seccomp_merge() بدمج مرشح seccomp الموجود في src مع المرشح الموجود في dst وتخزين النتيجة في المرشح dst. في حال النجاح، يتم تحرير مرشح seccomp src وتحرير جميع الذاكرة الداخلية المرتبطة بالمرشح؛ لا حاجة لاستدعاء seccomp_release(3) على src ويجب على المتصل تجاهل أي مراجع للمرشح.

لدمج مرشحي seccomp، يجب أن يكون لكل من المرشحين نفس قيم السمات وعدم تداخل البنيات (architectures).

قيمة الإرجاع

تعيد صفر عند النجاح أو أحد رموز الخطأ التالية عند الفشل:

-EDOM
تعذر دمج المرشحات بسبب مشاكل في البنية، على سبيل المثال، عدم تطابق ترتيب البايت.
-EEXIST
البنية موجودة بالفعل في المرشح.
-EINVAL
أحد المرشحات غير صالح.
-ENOMEM
لم تتمكن المكتبة من تخصيص ذاكرة كافية.

أمثلة

#include <seccomp.h>
int main(int argc, char *argv[])
{
	int rc = -1;
	scmp_filter_ctx ctx_32, ctx_64;
	ctx_32 = seccomp_init(SCMP_ACT_KILL);
	if (ctx_32 == NULL)
		goto out_all;
	ctx_64 = seccomp_init(SCMP_ACT_KILL);
	if (ctx_64 == NULL)
		goto out_all;
	if (seccomp_arch_exist(ctx_32, SCMP_ARCH_X86) == -EEXIST) {
		rc = seccomp_arch_add(ctx_32, SCMP_ARCH_X86);
		if (rc != 0)
			goto out_all;
		rc = seccomp_arch_remove(ctx_32, SCMP_ARCH_NATIVE);
		if (rc != 0)
			goto out_all;
	}
	if (seccomp_arch_exist(ctx_64, SCMP_ARCH_X86_64) == -EEXIST) {
		rc = seccomp_arch_add(ctx_64, SCMP_ARCH_X86_64);
		if (rc != 0)
			goto out_all;
		rc = seccomp_arch_remove(ctx_64, SCMP_ARCH_NATIVE);
		if (rc != 0)
			goto out_all;
	}
	/* ... */
	rc = seccomp_merge(ctx_64, ctx_32);
	if (rc != 0)
		goto out_all;
	/* ملاحظة: المرشح 'ctx_32' لم يعد صالحًا في هذه النقطة */
	/* ... */
out:
	seccomp_release(ctx_64);
	return -rc;
out_all:
	seccomp_release(ctx_32);
	goto out;
}

ملاحظات

بينما يمكن إنشاء مرشح seccomp بشكل مستقل عن النواة، فإن دعم النواة مطلوب لتحميل وتطبيق مرشح seccomp الذي تم إنشاؤه بواسطة libseccomp.

يمكن العثور على موقع مشروع libseccomp، مع مزيد من المعلومات ومستودع الشيفرة المصدرية، على https://github.com/seccomp/libseccomp. هذه الأداة، بالإضافة إلى مكتبة libseccomp، قيد التطوير حاليًا، يرجى الإبلاغ عن أي أخطاء في موقع المشروع أو مباشرة إلى المؤلف.
