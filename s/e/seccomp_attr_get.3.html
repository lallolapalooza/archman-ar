seccomp_attr_get(3)

الحزمة: libseccomp

التراخيص: LGPL-2.1-only

seccomp_attr_set(3) وثائق libseccomp seccomp_attr_set(3)

الاسم

seccomp_attr_set, seccomp_attr_get - إدارة سمات مرشح seccomp

الملخص

#include <seccomp.h>

typedef void * scmp_filter_ctx;
enum scmp_filter_attr;

int seccomp_attr_set(scmp_filter_ctx ctx,
                     enum scmp_filter_attr attr, uint32_t value);
int seccomp_attr_get(scmp_filter_ctx ctx,
                     enum scmp_filter_attr attr, uint32_t *value);

اربط مع -lseccomp.

الوصف

تقوم الدالة seccomp_attr_set() بتعيين سمات مرشح seccomp المختلفة، بينما تقوم الدالة seccomp_attr_get() بجلب سمات المرشح. سمات مرشح seccomp هي قيم قابلة للضبط تؤثر على كيفية عمل المكتبة عند إنشاء مرشح seccomp وتحميله في النواة. تتم إعادة تعيين السمات إلى قيمها الافتراضية في كل مرة يتم فيها تهيئة المرشح أو إعادة تعيينه عبر seccomp_init(3) أو seccomp_reset(3).

سياق المرشح ctx هو القيمة التي تم إرجاعها من استدعاء seccomp_init(3).

قيم attr الصالحة هي كما يلي:

SCMP_FLTATR_ACT_DEFAULT
الإجراء الافتراضي للمرشح كما هو محدد في استدعاء seccomp_init(3) أو seccomp_reset(3). هذه السمة للقراءة فقط.
SCMP_FLTATR_ACT_BADARCH
الإجراء الذي يتم اتخاذه للمرشح عندما لا يتطابق المرشح المحمل مع بنية التطبيق قيد التنفيذ. افتراضياً يكون الإجراء SCMP_ACT_KILL.
SCMP_FLTATR_CTL_NNP
علم لتحديد ما إذا كان سيتم تمكين وظيفة NO_NEW_PRIVS قبل تحميل مرشح seccomp في النواة. يؤدي تعيين هذا العلم إلى إيقاف التشغيل (value == 0) إلى عدم اتخاذ أي إجراء، مما يعني أن تحميل مرشح seccomp في النواة سيفشل إذا كان CAP_SYS_ADMIN مفقودًا ولم يتم تعيين NO_NEW_PRIVS خارجيًا. افتراضياً يكون التشغيل (value == 1).
SCMP_FLTATR_CTL_TSYNC
علم لتحديد ما إذا كان يجب على النواة محاولة مزامنة المرشحات عبر جميع الخيوط عند seccomp_load(3). إذا لم تتمكن النواة من مزامنة جميع الخيوط، فستفشل عملية التحميل. هذا العلم متاح فقط على نواة لينكس 3.17 أو أحدث؛ محاولة تمكين هذا العلم على النواة الأقدم ستؤدي إلى إرجاع خطأ. افتراضياً يكون الإيقاف (value == 0).
SCMP_FLTATR_API_TSKIP
علم لتحديد ما إذا كانت مكتبة libseccomp يجب أن تسمح بإنشاء قواعد مرشح لنداء النظام -1. يمكن استخدام قيمة نداء النظام -1 بواسطة برامج التتبع لتخطي استدعاءات النظام المحددة، انظر seccomp(2) لمزيد من المعلومات. افتراضياً يكون الإيقاف (value == 0).
SCMP_FLTATR_CTL_LOG
علم لتحديد ما إذا كان يجب على النواة تسجيل جميع إجراءات المرشح المتخذة باستثناء إجراء SCMP_ACT_ALLOW. افتراضياً يكون الإيقاف (value == 0).
SCMP_FLTATR_CTL_SSB
علم لتعطيل تخفيفات Speculative Store Bypass لهذا المرشح. افتراضياً يكون الإيقاف (value == 0).
SCMP_FLTATR_CTL_OPTIMIZE
علم لتحديد مستوى التحسين لمرشح seccomp. افتراضياً، تنشئ مكتبة libseccomp مجموعة من عبارات ´if´ المتسلسلة لكل قاعدة في المرشح. يمكن استخدام seccomp_syscall_priority(3) لتحديد أولوية الترتيب للسبب الافتراضي. يقوم تحسين الشجرة الثنائية بالفرز حسب أرقام نداء النظام وينشئ اجتيازًا ثابتًا للمرشح O(log n) لكل قاعدة في المرشح. قد تكون الشجرة الثنائية مفيدة للمرشحات الكبيرة. لاحظ أن seccomp_syscall_priority(3) يتم تجاهله عندما SCMP_FLTATR_CTL_OPTIMIZE == 2.

مستويات التحسين المختلفة موصوفة أدناه:

0
قيمة محجوزة، غير مستخدمة حاليًا.
1
القواعد مرتبة حسب الأولوية والتعقيد (افتراضي).
2
شجرة ثنائية مرتبة حسب رقم نداء النظام.
SCMP_FLTATR_API_SYSRAWRC
علم لتحديد ما إذا كانت مكتبة libseccomp يجب أن تمرر رموز خطأ النظام مرة أخرى إلى المتصل بدلاً من -ECANCELED الافتراضي. افتراضياً يكون الإيقاف (value == 0).

قيمة الإرجاع

تُرجع صفرًا عند النجاح أو أحد رموز الخطأ التالية عند الفشل:

-EACCES
تعيين السمة بالقيمة المعطاة غير مسموح به.
-EEXIST
السمة غير موجودة.
-EINVAL
مدخلات غير صالحة، إما السياق أو رمز البنية غير صالح.
-EOPNOTSUPP
المكتبة لا تدعم العملية المحددة.

أمثلة

#include <seccomp.h>
int main(int argc, char *argv[])
{
	int rc = -1;
	scmp_filter_ctx ctx;
	ctx = seccomp_init(SCMP_ACT_ALLOW);
	if (ctx == NULL)
		goto out;
	/* ... */
	rc = seccomp_attr_set(ctx, SCMP_FLTATR_ACT_BADARCH, SCMP_ACT_TRAP);
	if (rc < 0)
		goto out;
	/* ... */
out:
	seccomp_release(ctx);
	return -rc;
}

seccomp_attr_get

seccomp_attr_get - الحصول على سمة لمرشح seccomp

SYNOPSIS


#include <seccomp.h>

int seccomp_attr_get(uint32_t typ, struct seccomp_attr_metadata *meta);

DESCRIPTION

seccomp_attr_get هي واجهة نظام تأخذ سمة معينة من مرشح seccomp. تُستخدم هذه الوظيفة بشكل أساسي لأغراض التصحيح أو الحصول على معلومات حول كيفية عمل مرشح seccomp.

يحدد المعامل typ نوع السمة المطلوب الحصول عليها. يجب أن تكون إحدى القيم التالية:

  • SECCOMP_ATTR_METADATA_VERSION: يطلب إصدار البيانات الوصفية لـ seccomp. سيتم تخزين البيانات في meta->data.version.
  • SECCOMP_ATTR_METADATA_MAX_RETRIES: يطلب الحد الأقصى لعدد محاولات التحقق من الصحة المسموح بها لمرشح seccomp. سيتم تخزين البيانات في meta->data.max_retries.

يجب أن يشير المعامل meta إلى بنية struct seccomp_attr_metadata حيث سيتم تخزين البيانات المستردة.

RETURN VALUE

عند النجاح، تُرجع seccomp_attr_get 0.

عند الفشل، تُرجع قيمة سالبة تشير إلى رمز خطأ.

ERRORS

  • EINVAL: نوع السمة غير صالح أو بنية meta غير صالحة.
  • ENOSYS: الوظيفة غير مدعومة على هذا النظام.

EXAMPLES


#include <seccomp.h>
#include <stdio.h>
#include <errno.h>

int main() {
    struct seccomp_attr_metadata meta;
    int ret;

    ret = seccomp_attr_get(SECCOMP_ATTR_METADATA_VERSION, &meta);
    if (ret == 0) {
        printf("Seccomp metadata version: %u\n", meta.data.version);
    } else {
        perror("seccomp_attr_get failed");
    }

    ret = seccomp_attr_get(SECCOMP_ATTR_METADATA_MAX_RETRIES, &meta);
    if (ret == 0) {
        printf("Seccomp max retries: %u\n", meta.data.max_retries);
    } else {
        perror("seccomp_attr_get failed");
    }

    return 0;
}

NOTES

هذه الوظيفة مخصصة بشكل أساسي للاستخدام الداخلي أو لأغراض التصحيح. قد لا تكون المعلومات التي تم الحصول عليها متاحة أو ذات مغزى في جميع سيناريوهات الاستخدام.

تتطلب استخدام هذه الوظيفة أذونات مناسبة.
