seccomp_reset(3)

الحزمة: libseccomp

التراخيص: LGPL-2.1-only

seccomp_init(3) وثائق libseccomp seccomp_init(3)

الاسم

seccomp_init, seccomp_reset - تهيئة حالة فلتر seccomp

الملخص

#include <seccomp.h>

typedef void * scmp_filter_ctx;

scmp_filter_ctx seccomp_init(uint32_t def_action);
int seccomp_reset(scmp_filter_ctx ctx, uint32_t def_action);

اربط مع -lseccomp.

الوصف

تقوم الدالتان seccomp_init() و seccomp_reset() (بإعادة) تهيئة حالة فلتر seccomp الداخلية، وتجهيزها للاستخدام، وتعيين الإجراء الافتراضي بناءً على المعامل def_action. يجب استدعاء الدالة seccomp_init() قبل أي وظائف أخرى في libseccomp لأن بقية واجهة برمجة التطبيقات للمكتبة ستفشل إذا لم يتم تهيئة سياق الفلتر بشكل صحيح. تقوم الدالة seccomp_reset() بتحرير حالة سياق الفلتر الموجودة قبل إعادة تهيئتها ولا يمكن استدعاؤها إلا بعد نجاح استدعاء seccomp_init(). إذا تم استدعاء seccomp_reset() مع سياق فلتر فارغ (NULL)، فإنها تعيد تهيئة حالة المهمة العامة للمكتبة، بما في ذلك أي واصفات ملفات إشعار تم استردادها بواسطة seccomp_notify_fd(3). عادةً لا يكون هذا ضروريًا، ولكنه قد يكون مطلوبًا للاستمرار في استخدام المكتبة بعد استدعاء fork() أو clone() لضمان إعادة تعيين مستوى واجهة برمجة التطبيقات وحالة إشعار المستخدم بشكل صحيح.

عندما ينتهي المستدعي من تكوين فلتر seccomp وتحميله في النواة، يجب على المستدعي استدعاء seccomp_release(3) لتحرير جميع حالات سياق الفلتر.

القيم الصالحة للمعامل def_action هي كما يلي:

SCMP_ACT_KILL
سيتم إنهاء الخيط بواسطة النواة باستخدام SIGSYS عندما يقوم باستدعاء نظام لا يتطابق مع أي من قواعد فلتر seccomp المكونة. لن يتمكن الخيط من التقاط الإشارة.
SCMP_ACT_KILL_PROCESS
سيتم إنهاء العملية بأكملها بواسطة النواة باستخدام SIGSYS عندما تقوم باستدعاء نظام لا يتطابق مع أي من قواعد فلتر seccomp المكونة.
SCMP_ACT_TRAP
سيتم إرسال إشارة SIGSYS إلى الخيط عندما يقوم باستدعاء نظام لا يتطابق مع أي من قواعد فلتر seccomp المكونة. يمكنه التقاط هذه الإشارة وتغيير سلوكه وفقًا لذلك. عند استخدام SA_SIGINFO مع sigaction(2)، سيتم تعيين si_code إلى SYS_SECCOMP، وسيتم تعيين si_syscall إلى استدعاء النظام الذي فشل في القواعد، وسيتم تعيين si_arch إلى AUDIT_ARCH لواجهة برمجة التطبيقات النشطة.
SCMP_ACT_ERRNO(uint16_t errno)
سيحصل الخيط على قيمة إرجاع errno عند استدعاء نظام لا يتطابق مع أي من قواعد فلتر seccomp المكونة.
SCMP_ACT_TRACE(uint16_t msg_num)
إذا كان الخيط قيد التتبع وحدد عملية التتبع الخيار PTRACE_O_TRACESECCOMP في استدعاء ptrace(2)، فسيتم إخطار عملية التتبع، عبر PTRACE_EVENT_SECCOMP، ويمكن استرداد القيمة المقدمة في msg_num باستخدام الخيار PTRACE_GETEVENTMSG.
SCMP_ACT_LOG
لن يكون لفلتر seccomp أي تأثير على الخيط الذي يستدعي استدعاء النظام إذا لم يتطابق مع أي من قواعد فلتر seccomp المكونة، ولكن سيتم تسجيل استدعاء النظام.
SCMP_ACT_ALLOW
لن يكون لفلتر seccomp أي تأثير على الخيط الذي يستدعي استدعاء النظام إذا لم يتطابق مع أي من قواعد فلتر seccomp المكونة.

قيمة الإرجاع

تعيد الدالة seccomp_init() سياق فلتر عند النجاح، وقيمة فارغة (NULL) عند الفشل. تعيد الدالة seccomp_reset() صفرًا عند النجاح أو أحد رموز الخطأ التالية عند الفشل:

-EINVAL
مدخلات غير صالحة، إما السياق أو الإجراء غير صالح.
-ENOMEM
لم تتمكن المكتبة من تخصيص ذاكرة كافية.

أمثلة

#include <seccomp.h>
int main(int argc, char *argv[])
{
	int rc = -1;
	scmp_filter_ctx ctx;
	ctx = seccomp_init(SCMP_ACT_KILL);
	if (ctx == NULL)
		goto out;
	/* ... */
	rc = seccomp_reset(ctx, SCMP_ACT_KILL);
	if (rc < 0)
		goto out;
	/* ... */
out:
	seccomp_release(ctx);
	return -rc;
}

ملاحظات
