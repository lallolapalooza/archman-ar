SHADOW(3) استدعاءات المكتبة SHADOW(3)

الاسم

shadow, getspnam - إجراءات ملف كلمة المرور المشفرة

الصيغة

#include <shadow.h>

struct spwd *getspent();

struct spwd *getspnam(char *name);

void setspent();

void endspent();

struct spwd *fgetspent(FILE *fp);

struct spwd *sgetspent(char *cp);

int putspent(struct spwd *p, FILE *fp);

int lckpwdf();

int ulckpwdf();

الوصف

shadow يتعامل مع محتويات ملف كلمة المرور المشفر، /etc/shadow. الهيكل في ملف #include هو:

struct spwd {
      char		*sp_namp; /* اسم تسجيل الدخول للمستخدم */
      char		*sp_pwdp; /* كلمة المرور المشفرة */
      long int		sp_lstchg; /* آخر تغيير لكلمة المرور */
      long int		sp_min; /* عدد الأيام المسموح بها للتغيير. */
      long int		sp_max; /* عدد الأيام قبل أن يكون تغيير كلمة المرور مطلوبًا */
      long int		sp_warn; /* عدد الأيام للتحذير من انتهاء الصلاحية */
      long int		sp_inact; /* عدد الأيام قبل أن يصبح الحساب غير نشط */
      long int		sp_expire; /* تاريخ انتهاء صلاحية الحساب */
      unsigned long int	sp_flag; /* محجوز للاستخدام المستقبلي */
}

معاني كل حقل هي:

•sp_namp - مؤشر إلى اسم مستخدم منتهي بـ null
•sp_pwdp - مؤشر إلى كلمة مرور منتهية بـ null
•sp_lstchg - عدد الأيام منذ 1 يناير 1970 تم تغيير كلمة المرور آخر مرة
•sp_min - عدد الأيام التي لا يمكن قبلها تغيير كلمة المرور
•sp_max - عدد الأيام بعد التي يجب فيها تغيير كلمة المرور
•sp_warn - عدد الأيام قبل انتهاء صلاحية كلمة المرور يتم تحذير المستخدم من انتهاء صلاحية كلمة المرور الوشيك
•sp_inact - عدد الأيام بعد انتهاء صلاحية كلمة المرور يعتبر الحساب غير نشط ومعطل
•sp_expire - عدد الأيام منذ 1 يناير 1970 سيتم تعطيل الحساب
•sp_flag - محجوز للاستخدام المستقبلي

الوصف

getspent, getspname, fgetspent, و sgetspent كل منها تعيد مؤشرًا إلى struct spwd. getspent تعيد الإدخال التالي من الملف، و fgetspent تعيد الإدخال التالي من الدفق المعطى، والذي يفترض أنه ملف بالتنسيق الصحيح. sgetspent تعيد مؤشرًا إلى struct spwd باستخدام السلسلة النصية المقدمة كمدخل. getspnam تبحث من الموضع الحالي في الملف عن إدخال يطابق name.

setspent و endspent يمكن استخدامهما لبدء وإنهاء، على التوالي، الوصول إلى ملف كلمة المرور المشفر.

يجب استخدام الإجراءات lckpwdf و ulckpwdf لضمان الوصول الحصري إلى ملف /etc/shadow. lckpwdf تحاول الحصول على قفل باستخدام pw_lock لمدة تصل إلى 15 ثانية. تستمر عن طريق محاولة الحصول على قفل ثانٍ باستخدام spw_lock للمدة المتبقية من الـ 15 ثانية الأولية. في حالة فشل أي من المحاولتين بعد إجمالي 15 ثانية، تعيد lckpwdf -1. عند الحصول على كلا القفلين، يتم إرجاع 0.

التشخيصات

تعيد الإجراءات NULL إذا لم تكن هناك إدخالات أخرى متاحة أو إذا حدث خطأ أثناء المعالجة. الإجراءات التي لها int كقيمة إرجاع تعيد 0 للنجاح و -1 للفشل.

تنبيهات

قد يتم استخدام هذه الإجراءات فقط من قبل المستخدم الخارق (superuser) حيث أن الوصول إلى ملف كلمة المرور المشفر مقيد.
