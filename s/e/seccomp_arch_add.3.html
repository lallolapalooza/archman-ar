seccomp_arch_add(3)

الحزمة: libseccomp

التراخيص: LGPL-2.1-only

seccomp_arch_add(3) وثائق libseccomp seccomp_arch_add(3)

الاسم

seccomp_arch_add, seccomp_arch_remove, seccomp_arch_exist, seccomp_arch_native - إدارة معماريات مرشح seccomp

الملخص

#include <seccomp.h>

typedef void * scmp_filter_ctx;

#define SCMP_ARCH_NATIVE
#define SCMP_ARCH_X86
#define SCMP_ARCH_X86_64
#define SCMP_ARCH_X32
#define SCMP_ARCH_ARM
#define SCMP_ARCH_AARCH64
#define SCMP_ARCH_MIPS
#define SCMP_ARCH_MIPS64
#define SCMP_ARCH_MIPS64N32
#define SCMP_ARCH_MIPSEL
#define SCMP_ARCH_MIPSEL64
#define SCMP_ARCH_MIPSEL64N32
#define SCMP_ARCH_PPC
#define SCMP_ARCH_PPC64
#define SCMP_ARCH_PPC64LE
#define SCMP_ARCH_S390
#define SCMP_ARCH_S390X
#define SCMP_ARCH_PARISC
#define SCMP_ARCH_PARISC64
#define SCMP_ARCH_RISCV64

uint32_t seccomp_arch_resolve_name(const char *arch_name);
uint32_t seccomp_arch_native();
int seccomp_arch_exist(const scmp_filter_ctx ctx, uint32_t arch_token);
int seccomp_arch_add(scmp_filter_ctx ctx, uint32_t arch_token);
int seccomp_arch_remove(scmp_filter_ctx ctx, uint32_t arch_token);

اربط باستخدام -lseccomp.

الوصف

تقوم الدالة seccomp_arch_exist() بالتحقق مما إذا كانت معمارية معينة قد تمت إضافتها إلى مرشح seccomp في ctx، حيث تقوم seccomp_arch_add() و seccomp_arch_remove() بإضافة وإزالة المعماريات من مرشح seccomp على التوالي. في جميع الدوال الثلاث، يجب أن تكون قيم المعمارية المعطاة في arch_token هي الثوابت المعرفة SCMP_ARCH_*؛ مع الثابت SCMP_ARCH_NATIVE الذي يشير دائمًا إلى المعمارية المترجمة الأصلية. تقوم الدالة seccomp_arch_native() بإرجاع معمارية النظام بحيث تتطابق مع أحد الثوابت SCMP_ARCH_*. بينما تقوم الدالة seccomp_arch_resolve_name() أيضًا بإرجاع ثابت SCMP_ARCH_*، إلا أن الرمز المميز الذي يتم إرجاعه يتطابق مع اسم المعمارية التي تم تمريرها كوسيط للدالة.

عند تهيئة مرشح seccomp باستدعاء seccomp_init(3)، تتم إضافة المعمارية الأصلية تلقائيًا إلى المرشح.

على الرغم من أنه من الممكن إزالة جميع المعماريات من مرشح، إلا أن معظم واجهات برمجة تطبيقات libseccomp ستفشل إذا لم يحتوِ المرشح على معمارية واحدة على الأقل.

عند إضافة معمارية جديدة إلى مرشح موجود، لن تتم إضافة القواعد الحالية إلى المعمارية الجديدة. ومع ذلك، ستتم إضافة القواعد التي تتم إضافتها بعد إضافة المعمارية الجديدة إلى جميع المعماريات في المرشح.

قيمة الإرجاع

تُرجع الدوال seccomp_arch_add() و seccomp_arch_remove() و seccomp_arch_exist() صفرًا عند النجاح أو أحد رموز الخطأ التالية عند الفشل:

-EDOM
فشل خاص بالمعمارية.
-EEXIST
في حالة seccomp_arch_add()، المعمارية موجودة بالفعل، وفي حالة seccomp_arch_remove()، المعمارية غير موجودة.
-EINVAL
مدخل غير صالح، إما السياق أو رمز المعمارية غير صالح.
-ENOMEM
لم تتمكن المكتبة من تخصيص ذاكرة كافية.

أمثلة

#include <seccomp.h>
int main(int argc, char *argv[])
{
	int rc = -1;
	scmp_filter_ctx ctx;
	ctx = seccomp_init(SCMP_ACT_KILL);
	if (ctx == NULL)
		goto out;
	if (seccomp_arch_exist(ctx, SCMP_ARCH_X86) == -EEXIST) {
		rc = seccomp_arch_add(ctx, SCMP_ARCH_X86);
		if (rc != 0)
			goto out_all;
		rc = seccomp_arch_remove(ctx, SCMP_ARCH_NATIVE);
		if (rc != 0)
			goto out_all;
	}
	/* ... */
out:
	seccomp_release(ctx);
	return -rc;
}

ملاحظات

على الرغم من أنه يمكن إنشاء مرشح seccomp بشكل مستقل عن النواة، إلا أن دعم النواة مطلوب لتحميل وإنفاذ مرشح seccomp الذي تم إنشاؤه بواسطة libseccomp.

يمكن العثور على موقع مشروع libseccomp، مع مزيد من المعلومات ومستودع الشفرة المصدرية، على https://github.com/seccomp/libseccomp. هذه الأداة، وكذلك مكتبة libseccomp، قيد التطوير حاليًا، يرجى الإبلاغ عن أي أخطاء في موقع المشروع أو مباشرة إلى المؤلف.
