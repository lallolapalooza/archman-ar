seccomp_version(3)

الحزمة: libseccomp

التراخيص: LGPL-2.1-only

seccomp_version(3) وثائق libseccomp seccomp_version(3)

الاسم

seccomp_version - الاستعلام عن معلومات إصدار libseccomp

الملخص

#include <seccomp.h>

struct scmp_version {
	unsigned int major;
	unsigned int minor;
	unsigned int micro;
}

const struct scmp_version *seccomp_version(void);

ربط مع -lseccomp.

الوصف

الدالتان seccomp_version() و seccomp_reset() ترجعان مؤشرًا إلى هيكل scmp_version الذي يحتوي على معلومات الإصدار الخاصة بمكتبة libseccomp المحملة حاليًا. يمكن للتطبيقات التي تحتاج إلى التحقق من أنها مرتبطة بإصدار معين من libseccomp وقت التشغيل استخدام هذه الدالة.

لا ينبغي للمستدعي محاولة تحرير هيكل scmp_version المُرجع عند الانتهاء.

قيمة الإرجاع

الدالة seccomp_version() ترجع مؤشرًا إلى هيكل scmp_version عند النجاح، و NULL عند الفشل. لا ينبغي للمستدعي محاولة تحرير الهيكل المُرجع.

أمثلة

#include <seccomp.h>
int main(int argc, char *argv[])
{
	const struct scmp_version *ver;
	ver = seccomp_version();
	if (ver == NULL)
		goto err;
	/* ... */
	return 0;
err:
	return 1;
}

ملاحظات

بينما يمكن إنشاء مرشح seccomp بشكل مستقل عن النواة، فإن دعم النواة مطلوب لتحميل وتطبيق مرشح seccomp الذي تم إنشاؤه بواسطة libseccomp.

يمكن العثور على موقع مشروع libseccomp، مع مزيد من المعلومات ومستودع شفرة المصدر، على https://github.com/seccomp/libseccomp. هذه الأداة، بالإضافة إلى مكتبة libseccomp، قيد التطوير حاليًا، يرجى الإبلاغ عن أي أخطاء في موقع المشروع أو مباشرة إلى المؤلف.
