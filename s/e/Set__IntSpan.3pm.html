Set__IntSpan(3pm)

;, "السابق", و "الحالي". في جميع الأوقات، يكون المكرر إما عنصرًا من المجموعة، أو "غير معرف".

"الأول"، "الأخير"، و "البداية" تعين المكرر؛ "التالي"، و "السابق" تحركه؛ و "الحالي" تعيده. يمكن خلط استدعاءات هذه الطرق بحرية.

باستخدام "التالي" و "السابق"، يمكن لحلقة واحدة التحرك للأمام والخلف عبر مجموعة. باستخدام "البداية"، يمكن لحلقة تكرار أجزاء من مجموعة لا نهائية.

الطرق

الإنشاء

$set = "جديد" "Set::IntSpan" $set_spec
$set = "جديد" "Set::IntSpan" @set_specs
ينشئ ويعيد كائن "Set::IntSpan".

يتم تحديد المحتويات الأولية للمجموعة بواسطة $set_spec، أو باتحاد جميع @set_specs.

$ok = "صالح" "Set::IntSpan" $run_list
يعيد صحيحًا إذا كان $run_list قائمة تشغيل صالحة. بخلاف ذلك، يعيد خطأ ويترك رسالة خطأ في $@.
$set = "نسخ" $set $set_spec
ينسخ $set_spec إلى $set. يتم فقدان المحتويات السابقة لـ $set. للراحة، يعيد "نسخ" $set.
$run_list = "قائمة_تشغيل" $set
يعيد قائمة تشغيل تمثل $set. لن تحتوي قائمة التشغيل على مسافات بيضاء. $set غير متأثر.

بشكل افتراضي، يتم تنسيق المجموعة الفارغة كـ '-'. يمكن تحديد سلسلة مختلفة في $Set::IntSpan::Empty_String.

@elements = "عناصر" $set
يعيد مصفوفة تحتوي على عناصر $set. سيتم فرز العناصر بترتيب رقمي. في سياق قياسي، يعيد مرجع مصفوفة. $set غير متأثر.
@sets = "مجموعات" $set
يعيد التشغيلات في $set، كقائمة من كائنات "Set::IntSpan". المجموعات في القائمة مرتبة.
@spans = "فترات" $set
يعيد التشغيلات في $set، كقائمة بالشكل 
([$a1, $b1],
 [$a2, $b2],
 ...
 [$aN, $bN])

إذا كان التشغيل يحتوي على عدد صحيح واحد فقط، فإن الحدود العليا والدنيا للفترة المقابلة ستكون متساوية.

إذا لم يكن للمجموعة حد أدنى، فإن $a1 سيكون "غير معرف". وبالمثل، إذا لم يكن للمجموعة حد أعلى، فإن $bN سيكون "غير معرف".

التشغيلات في القائمة مرتبة.

عمليات المجموعة

لهذه العمليات، يتم إنشاء وإعادة كائن "Set::IntSpan" جديد. لا تتأثر المعاملات.

$u_set = "اتحاد" $set $set_spec
يعيد مجموعة الأعداد الصحيحة في $set أو $set_spec.
$i_set = "تقاطع" $set $set_spec
يعيد مجموعة الأعداد الصحيحة في كل من $set و $set_spec.
$x_set = "فرق_متماثل" $set $set_spec
يعيد مجموعة الأعداد الصحيحة في $set أو $set_spec، ولكن ليس كليهما.
$d_set = "فرق" $set $set_spec
يعيد مجموعة الأعداد الصحيحة في $set ولكن ليس في $set_spec.
$c_set = "مكمل" $set
يعيد مجموعة الأعداد الصحيحة التي ليست في $set.

مغيرات

حسب الطلب الشعبي، أصبح لدى "Set::IntSpan" الآن أشكال مغيرة لعمليات المجموعة الثنائية. هذه الطرق تغير الكائن الذي يتم استدعاؤها عليه.

$set->"U"($set_spec)
يجعل $set اتحاد $set و $set_spec. يعيد $set.
$set->"I"($set_spec)
يجعل $set تقاطع $set و $set_spec. يعيد $set.
$set->"X"($set_spec)
يجعل $set الفرق المتماثل لـ $set و $set_spec. يعيد $set.
$set->"D"($set_spec)
يجعل $set الفرق بين $set و $set_spec. يعيد $set.
$set->"C"
يحول $set إلى مكملته الخاصة. يعيد $set.

المقارنة

"متساوٍ" $set $set_spec
يعيد صحيحًا إذا وفقط إذا كانت $set و $set_spec تحتوي على نفس العناصر.
"متكافئ" $set $set_spec
يعيد صحيحًا إذا وفقط إذا كانت $set و $set_spec تحتوي على نفس العدد من العناصر. جميع المجموعات اللانهائية متكافئة.
"مجموعة_شاملة" $set $set_spec
يعيد صحيحًا إذا وفقط إذا كانت $set مجموعة شاملة لـ $set_spec.
"مجموعة_جزئية" $set $set_spec
يعيد صحيحًا إذا وفقط إذا كانت $set مجموعة جزئية لـ $set_spec.

الحجم

$n = "حجم" $set
$n = "عدد" $set
يعيد عدد العناصر في $set. يعيد -1 للمجموعات اللانهائية. يتم توفير "عدد" كاسم بديل لـ "حجم".
"فارغة" $set
يعيد صحيحًا إذا وفقط إذا كانت $set فارغة.
"منتهية" $set
يعيد صحيحًا إذا وفقط إذا كانت $set منتهية.
"سالب_لانهائية" $set
يعيد صحيحًا إذا وفقط إذا كانت $set تحتوي على {x | x<n} لبعض n.
"موجب_لانهائية" $set
يعيد صحيحًا إذا وفقط إذا كانت $set تحتوي على {x | x>n} لبعض n.
"لانهائية" $set
يعيد صحيحًا إذا وفقط إذا كانت $set لانهائية.
"عالمية" $set
يعيد صحيحًا إذا وفقط إذا كانت $set تحتوي على جميع الأعداد الصحيحة.

العضوية

"عضو" $set $n
يعيد صحيحًا إذا وفقط إذا كان العدد الصحيح $n عضوًا في $set.
"إدراج" $set $n
يدرج العدد الصحيح $n في $set. لا يفعل شيئًا إذا كان $n عضوًا بالفعل في $set.
"إزالة" $set $n
يزيل العدد الصحيح $n من $set. لا يفعل شيئًا إذا كان $n ليس عضوًا في $set.

القيم القصوى

"الحد_الأدنى" $set
يعيد أصغر عنصر في $set، أو "غير معرف" إذا لم يكن هناك.
"الحد_الأقصى" $set
يعيد أكبر عنصر في $set، أو "غير معرف"
$holes = "holes" $set
تُرجع مجموعة تحتوي على جميع الثقوب في $set، أي جميع الأعداد الصحيحة الموجودة بين نطاقات $set.

"holes" هي دائمًا مجموعة محدودة.

$cover = "cover" $set
تُرجع مجموعة تتكون من نطاق واحد من $set->"min" إلى $set->"max". هذا هو نفس 
union $set $set->holes
$inset = "inset" $set $n
$smaller = "trim" $set $n
$bigger = "pad" $set $n
"inset" تُرجع مجموعة مبنية عن طريق إزالة $n من الأعداد الصحيحة من كل طرف من كل نطاق في $set. إذا كان $n سالبًا، فإنه يتم إضافة -$n من الأعداد الصحيحة إلى كل طرف من كل نطاق.

في الحالة الأولى، قد تختفي النطاقات من المجموعة؛ في الحالة الثانية، قد تختفي الثقوب.

"trim" مُقدم كمرادف لـ "inset".

"pad" $set $n هو نفس "inset" $set -$n.

المكررات

$set->"first"
يعيّن المكرر لـ $set إلى أصغر عنصر في $set. إذا لم يكن هناك أصغر عنصر، يعيّن المكرر إلى "undef". يُرجع المكرر.
$set->"last"
يعيّن المكرر لـ $set إلى أكبر عنصر في $set. إذا لم يكن هناك أكبر عنصر، يعيّن المكرر إلى "undef". يُرجع المكرر.
$set->"start"($n)
يعيّن المكرر لـ $set إلى $n. إذا لم يكن $n عنصرًا في $set، يعيّن المكرر إلى "undef". يُرجع المكرر.
$set->"next"
يعيّن المكرر لـ $set إلى العنصر التالي في $set. إذا لم يكن هناك عنصر تالي، يعيّن المكرر إلى "undef". يُرجع المكرر.

"next" سيرجع "undef" مرة واحدة فقط؛ الاستدعاء التالي لـ "next" سيعيد تعيين المكرر إلى أصغر عنصر في $set.

$set->"prev"
يعيّن المكرر لـ $set إلى العنصر السابق في $set. إذا لم يكن هناك عنصر سابق، يعيّن المكرر إلى "undef". يُرجع المكرر.

"prev" سيرجع "undef" مرة واحدة فقط؛ الاستدعاء التالي لـ "prev" سيعيد تعيين المكرر إلى أكبر عنصر في $set.

$set->"current"
تُرجع المكرر لـ $set.

الفهرسة

يتم الاحتفاظ بعناصر المجموعة بترتيب رقمي. تحدد هذه الطرق فهرسة المجموعة بناءً على هذا الترتيب.

$n = $set->"at"($i)
تُرجع العنصر الـ $i من $set، أو "undef" إذا لم يكن هناك عنصر بالـ $i. الفهارس السالبة تعد تنازليًا من نهاية المجموعة.

يموت إذا

  • $i غير سالب و $set هو "neg_inf"
  • $i سالب و $set هو "pos_inf"
$slice = $set->"slice"($from, $to)
تُرجع كائن "Set::IntSpan" يحتوي على عناصر $set في الفهارس $from..$to. الفهارس السالبة تعد تنازليًا من نهاية المجموعة.

يموت إذا

  • $from غير سالب و $set هو "neg_inf"
  • $from سالب و $set هو "pos_inf"
$i = $set->"ord"($n)
عكس "at".

تُرجع الفهرس $i للعدد الصحيح $n في $set، أو "undef" إذا لم يكن $n عنصرًا في $set.

يموت إذا كانت $set هي "neg_inf".

$i = $set->"span_ord"($n)
تُرجع الفهرس $i للنطاق الذي يحتوي على العدد الصحيح $n، أو "undef" إذا لم يكن $n عنصرًا في $set.

لاستعادة النطاق الذي يحتوي على $n، اكتب

($set->spans)[$i]

تحميلات المعاملات

للراحة، تم تحميل بعض المعاملات على كائنات "Set::IntSpan".

عمليات المجموعات

يجب أن يكون أحد المعاملين كائن "Set::IntSpan". يمكن أن يكون المعامل الآخر كائن "Set::IntSpan" أو مواصفات مجموعة.

$u_set =  $set + $set_spec;   # union
$i_set =  $set * $set_spec;   # intersect
$x_set =  $set ^ $set_spec;   # xor
$d_set =  $set - $set_spec;   # diff
$c_set = ~$set;               # complement
$set += $set_spec;            # union
$set *= $set_spec;            # intersect
$set ^= $set_spec;            # xor
$set -= $set_spec;            # diff

المساواة

تم تحميل عمليات المقارنة النصية لمقارنة المجموعات من حيث المساواة والاحتواء. يجب أن يكون أحد المعاملين كائن "Set::IntSpan". يمكن أن يكون المعامل الآخر كائن "Set::IntSpan" أو مواصفات مجموعة.

$set eq $set_spec             # equal
$set ne $set_spec             # not equal
$set le $set_spec             # subset
$set lt $set_spec             # proper subset
$set ge $set_spec             # superset
$set gt $set_spec             # proper superset

التكافؤ

تم تحميل عمليات المقارنة الرقمية لمقارنة المجموعات من حيث العدد. يجب أن يكون أحد المعاملين كائن "Set::IntSpan". يمكن أن يكون المعامل الآخر كائن "Set::IntSpan" أو عددًا صحيحًا.

$set1 ==  $set2
$set1 !=  $set2
$set1 <=  $set2
$set1 <   $set2
$set1 >=  $set2
$set1 >   $set2
$set1 <=> $set2
$set1 cmp $set2
$set1 ==  $n
$set1 !=  $n
$set1 <=  $n
$set1 <   $n
$set1 >=  $n
$set1 >   $n
$set1 <=> $n
$set1 cmp $n

ملاحظة: تم تحميل معامل "cmp" لمقارنة المجموعات حسب العدد، وليس حسب الاحتواء. تم القيام بذلك حتى

sort @sets

سيقوم بفرز قائمة من المجموعات حسب العدد.

التحويل

في السياق المنطقي، تقيّم كائن "Set::IntSpan" على أنه صحيح إذا لم يكن فارغًا.

يتحول كائن "Set::IntSpan" إلى سلسلة نصية تمثل قائمة نطاقاته.

الدوال

$sub_set = "grep_set" { ... } $set
يقوم بتقييم الكتلة لكل عدد صحيح في $set (تعيين $_ محليًا لكل عدد صحيح) ويُرجع كائن "Set::IntSpan" يحتوي على تلك الأعداد الصحيحة التي تُرجع فيها الكتلة TRUE.
