seccomp_notify_free(3)

الحزمة: libseccomp

التراخيص: LGPL-2.1-only

seccomp_notify_alloc(3) توثيق libseccomp seccomp_notify_alloc(3)

الاسم

seccomp_notify_alloc, seccomp_notify_free, seccomp_notify_receive, seccomp_notify_respond, seccomp_notify_id_valid, seccomp_notify_fd - إدارة إشعارات seccomp

ملخص

#include <seccomp.h>

int seccomp_notify_alloc(struct seccomp_notif **req, struct seccomp_notif_resp **resp)
void seccomp_notify_free(struct seccomp_notif *req, struct seccomp_notif_resp *resp)
int seccomp_notify_receive(int fd, struct seccomp_notif *req)
int seccomp_notify_respond(int fd, struct seccomp_notif_resp *resp)
int seccomp_notify_id_valid(int fd, uint64_t id)
int seccomp_notify_fd(const scmp_filter_ctx ctx)

اربط مع -lseccomp.

الوصف

تقوم الدالة seccomp_notify_alloc() بتخصيص ذاكرة كافية ديناميكيًا لإشعار seccomp والاستجابة له. لاحظ أنه يجب على المرء دائمًا استخدام هذه الدوال وعدم الاعتماد على أحجام الهياكل في الرؤوس، نظرًا لأن الحجم يمكن أن يختلف اعتمادًا على إصدار النواة. تتولى هذه الدالة مهمة الاستعلام من النواة عن حجم كل هيكل مطلوب، وتقوم بتخصيص الكمية الصحيحة من الذاكرة. تقوم الدالة seccomp_notify_free() بتحرير الذاكرة التي تم تخصيصها بواسطة seccomp_notify_alloc().

تستقبل الدالة seccomp_notify_receive() إشعارًا من واصف ملف إشعار seccomp (تم الحصول عليه من seccomp_notify_fd()).

ترسل الدالة seccomp_notify_respond() استجابة إلى إشعار معين. يجب أن يكون حقل المعرف (id) هو نفسه المعرف (id) من الطلب، حتى تعرف النواة إلى أي طلب تتوافق هذه الاستجابة.

تتحقق الدالة seccomp_notify_id_valid() لمعرفة ما إذا كانت استدعاء النظام (syscall) من طلب إشعار معين لا تزال صالحة، أي ما إذا كانت المهمة (task) لا تزال قيد التشغيل. انظر الملاحظات أدناه للحصول على تفاصيل حول حالات السباق (race conditions).

تعيد الدالة seccomp_notify_fd() واصف الملف للإشعار الخاص بالفلتر بعد تحميله.

قيمة الإرجاع

تعيد الدالة seccomp_notify_fd() واصف ملف الإشعار الخاص بالفلتر المحمل، و -1 إذا لم يتم إنشاء واصف ملف للإشعار بعد، و -EINVAL إذا كان سياق الفلتر غير صالح.

تعيد الدالة seccomp_notify_id_valid() القيمة 0 إذا كان المعرف صالحًا، و -ENOENT إذا لم يكن كذلك.

تعيد الدوال seccomp_notify_alloc() و seccomp_notify_receive() و seccomp_notify_respond() القيمة صفر عند النجاح، أو أحد أكواد الخطأ التالية عند الفشل:

-ECANCELED
حدث فشل في النظام خارج عن سيطرة المكتبة، تحقق من قيمة errno لمزيد من المعلومات.
-EFAULT
فشل داخلي في libseccomp.
-ENOMEM
لم تتمكن المكتبة من تخصيص ذاكرة كافية.
-EOPNOTSUPP
لا تدعم المكتبة العملية المحددة.

ملاحظات

يجب توخي الحذر لتجنب خطأين مختلفين في وقت التحقق/وقت الاستخدام. أولاً، بعد فتح أي موارد ذات صلة بمعرف العملية (pid) للإشعار (مثل /proc/pid/mem لقراءة ذاكرة المتتبع (tracee) لاتخاذ قرارات بشأن السياسة)، يجب على التطبيقات استدعاء seccomp_notify_id_valid() للتأكد من أن الموارد التي فتحتها التطبيق تتوافق مع معرف العملية الصحيح، أي أن معرف العملية لم يمت وأن مهمة مختلفة لم تأخذ مكانه.

ثانياً، يجب أيضًا تجنب مشكلة وقت التحقق/وقت الاستخدام الكلاسيكية مع ذاكرة seccomp: يجب على التطبيقات نسخ أي ذاكرة ترغب في استخدامها لاتخاذ قرارات من المتتبع إلى مساحة العنوان الخاصة بها قبل تطبيق أي قرارات سياسة، نظرًا لأن متتبعًا متعدد الخيوط (multi-threaded tracee) قد يعدل الذاكرة في أي وقت، بما في ذلك بعد استخدامها لاتخاذ قرار سياسة.

يتوفر مثال كامل لكيفية تجنب هاتين الحالتين في شجرة مصدر نواة لينكس في /samples/seccomp/user-trap.c.
