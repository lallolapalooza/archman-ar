seccomp_attr_set(3)

الحزمة: libseccomp

التراخيص: LGPL-2.1-only

seccomp_attr_set(3) توثيق libseccomp seccomp_attr_set(3)

الاسم

seccomp_attr_set, seccomp_attr_get - إدارة سمات مرشح seccomp

الملخص

#include <seccomp.h>

typedef void * scmp_filter_ctx;
enum scmp_filter_attr;

int seccomp_attr_set(scmp_filter_ctx ctx,
                     enum scmp_filter_attr attr, uint32_t value);
int seccomp_attr_get(scmp_filter_ctx ctx,
                     enum scmp_filter_attr attr, uint32_t *value);

ربط باستخدام -lseccomp.

الوصف

تقوم الدالة seccomp_attr_set() بتعيين سمات مرشح seccomp المختلفة بينما تقوم الدالة seccomp_attr_get() بجلب سمات المرشح. سمات مرشح seccomp هي قيم قابلة للتعديل تؤثر على كيفية سلوك المكتبة عند إنشاء وتحميل مرشح seccomp إلى النواة. يتم إعادة تعيين السمات إلى قيمها الافتراضية كلما تم تهيئة المرشح أو إعادة تعيينه عبر seccomp_init(3) أو seccomp_reset(3).

سياق المرشح ctx هو القيمة التي تم إرجاعها من استدعاء seccomp_init(3).

القيم الصالحة لـ attr هي كما يلي:

SCMP_FLTATR_ACT_DEFAULT
الإجراء الافتراضي للمرشح كما هو محدد في استدعاء seccomp_init(3) أو seccomp_reset(3). هذه السمة للقراءة فقط.
SCMP_FLTATR_ACT_BADARCH
إجراء المرشح الذي يتم اتخاذه عندما لا يتطابق المرشح المحمل مع بنية التطبيق قيد التنفيذ. القيمة الافتراضية هي إجراء SCMP_ACT_KILL.
SCMP_FLTATR_CTL_NNP
علم لتحديد ما إذا كان يجب تمكين وظيفة NO_NEW_PRIVS قبل تحميل مرشح seccomp إلى النواة. يؤدي تعيين هذا إلى إيقاف (value == 0) إلى عدم اتخاذ أي إجراء، مما يعني أن تحميل مرشح seccomp إلى النواة سيفشل إذا كان CAP_SYS_ADMIN مفقودًا ولم يتم تعيين NO_NEW_PRIVS خارجيًا. القيمة الافتراضية هي تشغيل (value == 1).
SCMP_FLTATR_CTL_TSYNC
علم لتحديد ما إذا كان يجب على النواة محاولة مزامنة المرشحات عبر جميع الخيوط عند seccomp_load(3). إذا كانت النواة غير قادرة على مزامنة جميع الخيوط، فسيفشل عملية التحميل. هذا العلم متاح فقط على نواة Linux 3.17 أو أحدث؛ محاولة تمكين هذا العلم على أنوية أقدم ستؤدي إلى إرجاع خطأ. القيمة الافتراضية هي إيقاف (value == 0).
SCMP_FLTATR_API_TSKIP
علم لتحديد ما إذا كانت libseccomp يجب أن تسمح بإنشاء قواعد مرشح لـ syscall -1. يمكن استخدام قيمة syscall -1 بواسطة برامج التتبع لتخطي استدعاءات syscall محددة، انظر seccomp(2) لمزيد من المعلومات. القيمة الافتراضية هي إيقاف (value == 0).
SCMP_FLTATR_CTL_LOG
علم لتحديد ما إذا كان يجب على النواة تسجيل جميع إجراءات المرشح المتخذة باستثناء إجراء SCMP_ACT_ALLOW. القيمة الافتراضية هي إيقاف (value == 0).
SCMP_FLTATR_CTL_SSB
علم لتعطيل تخفيف هجوم Speculative Store Bypass لهذا المرشح. القيمة الافتراضية هي إيقاف (value == 0).
SCMP_FLTATR_CTL_OPTIMIZE
علم لتحديد مستوى تحسين مرشح seccomp. افتراضيًا، تقوم libseccomp بإنشاء مجموعة من عبارات ´if´ متسلسلة لكل قاعدة في المرشح. يمكن استخدام seccomp_syscall_priority(3) لتحديد أولوية الترتيب للسبب الافتراضي. يقوم تحسين الشجرة الثنائية بالفرز حسب أرقام syscall وينشئ اجتيازًا ثابتًا للمرشح O(log n) لكل قاعدة في المرشح. قد تكون الشجرة الثنائية مفيدة للمرشحات الكبيرة. لاحظ أن seccomp_syscall_priority(3) يتم تجاهله عندما يكون SCMP_FLTATR_CTL_OPTIMIZE == 2.

تتمثل مستويات التحسين المختلفة فيما يلي:

0
قيمة محجوزة، غير مستخدمة حاليًا.
1
القواعد مرتبة حسب الأولوية والتعقيد (افتراضي).
2
شجرة ثنائية مرتبة حسب رقم syscall.
SCMP_FLTATR_API_SYSRAWRC
علم لتحديد ما إذا كانت libseccomp يجب أن تمرر رموز خطأ النظام إلى المتصل بدلاً من القيمة الافتراضية -ECANCELED. القيمة الافتراضية هي إيقاف (value == 0).

قيمة الإرجاع

ترجع صفرًا عند النجاح أو أحد رموز الخطأ التالية عند الفشل:

-EACCES
تعيين السمة بالقيمة المعطاة غير مسموح به.
-EEXIST
السمة غير موجودة.
-EINVAL
مدخل غير صالح، إما السياق أو رمز البنية غير صالح.
-EOPNOTSUPP
المكتبة لا تدعم العملية المحددة.

أمثلة

#include <seccomp.h>
int main(int argc, char *argv[])
{
	int rc = -1;
	scmp_filter_ctx ctx;
	ctx = seccomp_init(SCMP_ACT_ALLOW);
	if (ctx == NULL)
		goto out;
	/* ... */
	rc = seccomp_attr_set(ctx, SCMP_FLTATR_ACT_BADARCH, SCMP_ACT_TRAP);
	if (rc < 0)
		goto out;
	/* ... */
out:
	seccomp_release(ctx);
	return -rc;
}
