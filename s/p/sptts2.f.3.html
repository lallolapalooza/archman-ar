توثيق الوظيفة/الروتين الفرعي

subroutine sptts2 (integer n, integer nrhs, real, dimension( * ) d, real, dimension( * ) e, real, dimension( ldb, * ) b, integer ldb)

SPTTS2 تحل نظامًا ثلاثي الأقطار من الشكل AX=B باستخدام تحليل L D LH الذي تم حسابه بواسطة spttrf.

الغرض:

!>
!> SPTTS2 تحل نظامًا ثلاثي الأقطار من الشكل
!>    A * X = B
!> باستخدام تحليل L*D*L**T للمصفوفة A الذي تم حسابه بواسطة SPTTRF.  D هي
!> مصفوفة قطرية محددة في المتجه D، L هي مصفوفة ثنائية القطر وحدوية
!> عناصرها تحت القطرية محددة في المتجه E، و X و B
!> هما مصفوفتان من حجم N في NRHS.
!>

المعلمات

N 
!>          N هو عدد صحيح (INTEGER)
!>          رتبة المصفوفة ثلاثية الأقطار A.  N >= 0.
!>

NRHS

!>          NRHS هو عدد صحيح (INTEGER)
!>          عدد الأطراف اليمنى، أي عدد أعمدة
!>          المصفوفة B.  NRHS >= 0.
!>

D

!>          D هي مصفوفة حقيقية (REAL array)، حجمها (N)
!>          العناصر القطرية الـ N للمصفوفة القطرية D من
!>          تحليل L*D*L**T للمصفوفة A.
!>

E

!>          E هي مصفوفة حقيقية (REAL array)، حجمها (N-1)
!>          العناصر تحت القطرية الـ (N-1) للعامل الوحدوي ثنائي القطر
!>          L من تحليل L*D*L**T للمصفوفة A.  يمكن اعتبار E أيضًا
!>          على أنها العناصر فوق القطرية للعامل الوحدوي ثنائي القطر U من
!>          تحليل A = U**T*D*U.
!>

B

!>          B هي مصفوفة حقيقية (REAL array)، حجمها (LDB,NRHS)
!>          عند الدخول، متجهات الطرف الأيمن B لأنظمة
!>          المعادلات الخطية.
!>          عند الخروج، متجهات الحل X.
!>

LDB

!>          LDB هو عدد صحيح (INTEGER)
!>          البعد الرائد للمصفوفة B.  LDB >= max(1,N).
!>
