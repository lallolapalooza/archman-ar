subroutine sppt03 (character uplo, integer n, real, dimension( * ) a, real, dimension( * ) ainv, real, dimension( ldwork, * ) work, integer ldwork, real, dimension( * ) rwork, real rcond, real resid)

SPPT03

الغرض:

!>
!> SPPT03 تحسب الباقي لمصفوفة متماثلة مجمعة مضروبة في معكوسها:
!>    norm( I - A*AINV ) / ( N * norm(A) * norm(AINV) * EPS ),
!> حيث EPS هو الجزء العشري للآلة.
!>

المعاملات

UPLO 
!>          UPLO هو CHARACTER*1
!>          يحدد ما إذا كان الجزء العلوي أو السفلي المثلثي للمصفوفة المتماثلة
!>          A مخزن:
!>          = 'U':  المثلثي العلوي
!>          = 'L':  المثلثي السفلي
!>

N

!>          N هو INTEGER
!>          عدد الصفوف والأعمدة للمصفوفة A.  N >= 0.
!>

A

!>          A هو REAL array, dimension (N*(N+1)/2)
!>          المصفوفة المتماثلة الأصلية A، مخزنة كمصفوفة
!>          مثلثية مجمعة.
!>

AINV

!>          AINV هو REAL array, dimension (N*(N+1)/2)
!>          المعكوس (المتماثل) للمصفوفة A، مخزن كمصفوفة
!>          مثلثية مجمعة.
!>

WORK

!>          WORK هو REAL array, dimension (LDWORK,N)
!>

LDWORK

!>          LDWORK هو INTEGER
!>          البعد الرئيسي للمصفوفة WORK.  LDWORK >= max(1,N).
!>

RWORK

!>          RWORK هو REAL array, dimension (N)
!>

RCOND

!>          RCOND هو REAL
!>          مقلوب رقم الشرط لـ A، محسوب كـ
!>          ( 1/norm(A) ) / norm(AINV).
!>

RESID

!>          RESID هو REAL
!>          norm(I - A*AINV) / ( N * norm(A) * norm(AINV) * EPS )
!>
