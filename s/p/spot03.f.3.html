subroutine spot03 (character uplo, integer n, real, dimension( lda, * ) a, integer lda, real, dimension( ldainv, * ) ainv, integer ldainv, real, dimension( ldwork, * ) work, integer ldwork, real, dimension( * ) rwork, real rcond, real resid)

SPOT03

الغرض:

!>
!> SPOT03 يحسب الباقي لضرب مصفوفة متماثلة في معكوسها:
!>    norm( I - A*AINV ) / ( N * norm(A) * norm(AINV) * EPS ),
!> حيث EPS هو البت الآلي.
!>

المعلمات

UPLO 
!>          UPLO is CHARACTER*1
!>          يحدد ما إذا كان الجزء العلوي أو السفلي المثلثي للمصفوفة المتماثلة A مخزنًا:
!>          = 'U':  مثلثي علوي
!>          = 'L':  مثلثي سفلي
!>

N

!>          N is INTEGER
!>          عدد الصفوف والأعمدة للمصفوفة A.  N >= 0.
!>

A

!>          A is REAL array, dimension (LDA,N)
!>          المصفوفة المتماثلة الأصلية A.
!>

LDA

!>          LDA is INTEGER
!>          البعد الرائد للمصفوفة A.  LDA >= max(1,N)
!>

AINV

!>          AINV is REAL array, dimension (LDAINV,N)
!>          عند الإدخال، معكوس المصفوفة A، مخزن كمصفوفة متماثلة
!>          بنفس تنسيق A.
!>          في هذا الإصدار، يتم توسيع AINV إلى مصفوفة كاملة
!>          وضربها في A، لذلك سيتم تغيير المثلث المقابل لـ AINV؛
!>          أي، إذا تم تخزين الجزء المثلثي العلوي من AINV،
!>          فإن الجزء المثلثي السفلي سيستخدم كمساحة عمل.
!>

LDAINV

!>          LDAINV is INTEGER
!>          البعد الرائد للمصفوفة AINV.  LDAINV >= max(1,N).
!>

WORK

!>          WORK is REAL array, dimension (LDWORK,N)
!>

LDWORK

!>          LDWORK is INTEGER
!>          البعد الرائد للمصفوفة WORK.  LDWORK >= max(1,N).
!>

RWORK

!>          RWORK is REAL array, dimension (N)
!>

RCOND

!>          RCOND is REAL
!>          مقلوب رقم الشرط للمصفوفة A، محسوب كـ
!>          ( 1/norm(A) ) / norm(AINV).
!>

RESID

!>          RESID is REAL
!>          norm(I - A*AINV) / ( N * norm(A) * norm(AINV) * EPS )
!>
