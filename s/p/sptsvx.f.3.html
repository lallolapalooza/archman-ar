subroutine sptsvx (character fact, integer n, integer nrhs, real, dimension( * ) d, real, dimension( * ) e, real, dimension( * ) df, real, dimension( * ) ef, real, dimension( ldb, * ) b, integer ldb, real, dimension( ldx, * ) x, integer ldx, real rcond, real, dimension( * ) ferr, real, dimension( * ) berr, real, dimension( * ) work, integer info)

SPTSVX يحسب حل نظام المعادلات الخطية A * X = B لمصفوفات PT

الغرض:

!>
!> SPTSVX يستخدم التحليل A = L*D*L**T لحساب الحل
!> لنظام المعادلات الخطية الحقيقي A*X = B، حيث A مصفوفة ثلاثية قطرية متماثلة وموجبة محددة من الرتبة N، و X و B مصفوفتان من الرتبة N × NRHS.
!>
!> يتم أيضًا توفير حدود الخطأ للحل وتقدير الشرط.
!>

الوصف:

!>
!> يتم تنفيذ الخطوات التالية:
!>
!> 1. إذا كانت FACT = 'N'، يتم تحليل المصفوفة A إلى A = L*D*L**T، حيث L
!>    هي مصفوفة ثنائية قطرية سفلية ووحدية، و D مصفوفة قطرية. يمكن أيضًا
!>    اعتبار التحليل له الشكل
!>    A = U**T*D*U.
!>
!> 2. إذا كان الحاصل الرئيسي الأولي من الرتبة i غير موجب،
!>    فإن الروتين يعود بـ INFO = i. بخلاف ذلك، يتم استخدام الشكل المحلل
!>    لـ A لتقدير رقم شرط المصفوفة A. إذا كان مقلوب رقم الشرط أقل من دقة الآلة،
!>    يتم إرجاع INFO = N+1 كتحذير، ولكن الروتين
!>    يستمر في حل X وحساب حدود الخطأ كما
!>    هو موضح أدناه.
!>
!> 3. يتم حل نظام المعادلات لـ X باستخدام الشكل المحلل
!>    لـ A.
!>
!> 4. يتم تطبيق التحسين التكراري لتحسين الحل المحسوب
!>    للمصفوفة وحساب حدود الخطأ وتقديرات الخطأ الخلفي
!>    لها.
!>

المعلمات

FACT 
!>          FACT هو CHARACTER*1
!>          يحدد ما إذا كان الشكل المحلل لـ A قد تم توفيره عند الدخول أم لا.
!>          = 'F':  عند الدخول، تحتوي DF و EF على الشكل المحلل لـ A.
!>                  لن يتم تعديل D و E و DF و EF.
!>          = 'N':  سيتم نسخ المصفوفة A إلى DF و EF و
!>                  تحليلها.
!>

N

!>          N هو INTEGER
!>          رتبة المصفوفة A. N >= 0.
!>

NRHS

!>          NRHS هو INTEGER
!>          عدد الأطراف اليمنى، أي عدد الأعمدة
!>          للمصفوفتين B و X. NRHS >= 0.
!>

D

!>          D هو مصفوفة REAL، البعد (N)
!>          عناصر القطر الرئيسية N للمصفوفة الثلاثية القطرية A.
!>

E

!>          E هو مصفوفة REAL، البعد (N-1)
!>          عناصر القطر الفرعية N-1 للمصفوفة الثلاثية القطرية A.
!>

DF

!>          DF هو مصفوفة REAL، البعد (N)
!>          إذا كانت FACT = 'F'، فإن DF هي معلمة إدخال وعند الدخول
!>          تحتوي على العناصر القطرية الرئيسية N للمصفوفة القطرية D
!>          من التحليل L*D*L**T للمصفوفة A.
!>          إذا كانت FACT = 'N'، فإن DF هي معلمة إخراج وعند الخروج
!>          تحتوي على العناصر القطرية الرئيسية N للمصفوفة القطرية D
!>          من التحليل L*D*L**T للمصفوفة A.
!>

EF

!>          EF هو مصفوفة REAL، البعد (N-1)
!>          إذا كانت FACT = 'F'، فإن EF هي معلمة إدخال وعند الدخول
!>          تحتوي على عناصر القطر الفرعية N-1 للعامل الأحادي
!>          الثنائي القطري L من التحليل L*D*L**T للمصفوفة A.
!>          إذا كانت FACT = 'N'، فإن EF هي معلمة إخراج وعند الخروج
!>          تحتوي على عناصر القطر الفرعية N-1 للعامل الأحادي
!>          الثنائي القطري L من التحليل L*D*L**T للمصفوفة A.
!>

B

!>          B هي مصفوفة REAL، البعد (LDB,NRHS)
!>          مصفوفة الطرف الأيمن B من الرتبة N × NRHS.
!>

LDB

!>          LDB هو INTEGER
!>          البعد الرئيسي للمصفوفة B. LDB >= max(1,N).
!>

X

!>          X هي مصفوفة REAL، البعد (LDX,NRHS)
!>          إذا كانت INFO = 0 أو INFO = N+1، فإن مصفوفة الحل X من الرتبة N × NRHS.
!>

LDX

!>          LDX هو INTEGER
!>          البعد الرئيسي للمصفوفة X. LDX >= max(1,N).
!>

RCOND

!>          RCOND هو REAL
!>          مقلوب رقم شرط المصفوفة A. إذا كان RCOND
!>          أقل من دقة الآلة (خاصة، إذا
!>          كان RCOND = 0)، فإن المصفوفة منفردة بدقة العمل.
!>          يتم الإشارة إلى هذا الشرط برمز عودة INFO > 0.
!>

FERR

!>          FERR هي مصفوفة REAL، البعد (NRHS)
!>          حد الخطأ الأمامي لكل متجه حل
!>          X(j) (العمود j لمصفوفة الحل X).
!>          إذا كان XTRUE هو الحل الحقيقي المقابل لـ X(j)، فإن FERR(j)
!>          هو حد علوي مقدر لحجم أكبر
!>          عنصر في (X(j) - XTRUE) مقسومًا على حجم أكبر
!>          عنصر في X(j).
!>

BERR

!>          BERR هي مصفوفة REAL، البعد (NRHS)
!>          الخطأ الخلفي النسبي لكل مكون لمتجه الحل
!>          X(j) (أي، أصغر تغيير نسبي في أي
!>          عنصر في A أو B يجعل X(j) حلاً دقيقًا).
!>

WORK

!>          WORK هي مصفوفة REAL، البعد (2*N)
!>

INFO

!>          INFO هو INTEGER
!>          = 0:  انتهاء ناجح
!>          < 0:  إذا كان INFO = -i، فإن المعلمة i كانت لها قيمة غير قانونية
!>          > 0:  إذا كان INFO = i، و i هو
!>                <= N:  الحاصل الرئيسي الأولي من الرتبة i لـ A
!>                       ليس موجبًا، لذلك لم يكن من الممكن إكمال التحليل،
!>                       ولم يتم حساب الحل. يتم إرجاع RCOND = 0.
!>                = N+1: U غير منفردة، ولكن RCOND أقل من دقة الآلة، مما يعني أن المصفوفة
!>                       منفردة لدقة العمل. ومع ذلك، يتم
!>                       حساب الحل وحدود الخطأ لأنه
!>                       هناك عدد من الحالات التي يمكن أن يكون فيها
!>                       الحل المحسوب أكثر دقة من
!>                       قيمة RCOND.
!>
