subroutine spftrf (character transr, character uplo, integer n, real, dimension( 0: * ) a, integer info)

SPFTRF

الغرض:

!>
!> SPFTRF يحسب تحليل تشوليسكي لمصفوفة A الحقيقية المتناظرة الموجبة المحددة.
!>
!> يتكون التحليل من الشكل
!>    A = U**T * U,  إذا كان UPLO = 'U', أو
!>    A = L  * L**T,  إذا كان UPLO = 'L',
!> حيث U مصفوفة مثلثية عليا و L مصفوفة مثلثية سفلى.
!>
!> هذا هو الإصدار المقطعي من الخوارزمية، الذي يستدعي BLAS من المستوى 3.
!>

المعلمات

TRANSR 
!>          TRANSR هو CHARACTER*1
!>          = 'N':  يتم تخزين TRANSR العادي للمصفوفة A بتنسيق RFP؛
!>          = 'T':  يتم تخزين TRANSR المنقول للمصفوفة A بتنسيق RFP.
!>

UPLO

!>          UPLO هو CHARACTER*1
!>          = 'U':  يتم تخزين المثلث العلوي للمصفوفة A بتنسيق RFP؛
!>          = 'L':  يتم تخزين المثلث السفلي للمصفوفة A بتنسيق RFP.
!>

N

!>          N هو INTEGER
!>          رتبة المصفوفة A.  N >= 0.
!>

A

!>          A هو مصفوفة REAL، ذات بُعد ( N*(N+1)/2 )؛
!>          عند الإدخال، المصفوفة المتناظرة A بتنسيق RFP. يتم وصف تنسيق RFP
!>          بواسطة TRANSR و UPLO و N كما يلي: إذا كان TRANSR = 'N'
!>          فإن RFP A تكون (0:N,0:k-1) عندما يكون N زوجياً؛ k=N/2. RFP A تكون
!>          (0:N-1,0:k) عندما يكون N فردياً؛ k=N/2. إذا كان TRANSR = 'T' فإن RFP
!>          هي المنقولة لـ RFP A كما هو معرف عند
!>          TRANSR = 'N'. يتم تحديد محتويات RFP A بواسطة UPLO كما يلي:
!>          إذا كان UPLO = 'U' فإن RFP A تحتوي على عناصر NT للمصفوفة A المعبأة علوياً.
!>          إذا كان UPLO = 'L' فإن RFP A تحتوي على عناصر
!>          للمصفوفة A المعبأة سفلياً. LDA لـ RFP A هو (N+1)/2 عندما يكون TRANSR =
!>          'T'. عندما يكون TRANSR = 'N' فإن LDA هو N+1 عندما يكون N زوجياً و
!>          N فردياً. راجع الملاحظة أدناه لمزيد من التفاصيل.
!>
!>          عند الخروج، إذا كان INFO = 0، فإن العامل U أو L من تحليل تشوليسكي
!>          RFP A = U**T*U أو RFP A = L*L**T.
!>

INFO

!>          INFO هو INTEGER
!>          = 0:  خروج ناجح
!>          < 0:  إذا كان INFO = -i، فإن المعامل i كان له قيمة غير قانونية
!>          > 0:  إذا كان INFO = i، فإن أصغر بادئة رئيسية من الرتبة i
!>                ليست موجبة، ولم يكن من الممكن إكمال التحليل.
!>

تفاصيل إضافية:

!>
!>  ننظر أولاً في تنسيق التعبئة الكاملة المستطيلة (RFP) عندما يكون N
!>  زوجيًا. نقدم مثالاً حيث N = 6.
!>
!>      AP علوي             AP سفلي
!>
!>   00 01 02 03 04 05       00
!>      11 12 13 14 15       10 11
!>         22 23 24 25       20 21 22
!>            33 34 35       30 31 32 33
!>               44 45       40 41 42 43 44
!>                  55       50 51 52 53 54 55
!>
!>
!>  لتكن TRANSR = 'N'. RFP يخزن AP كما يلي:
!>  لـ UPLO = 'U' فإن شبه المنحرف العلوي A(0:5,0:2) يتكون من الأعمدة الثلاثة الأخيرة
!>  من AP العلوي. المثلث السفلي A(4:6,0:2) يتكون من
!>  منقول الأعمدة الثلاثة الأولى من AP العلوي.
!>  لـ UPLO = 'L' فإن شبه المنحرف السفلي A(1:6,0:2) يتكون من الأعمدة الثلاثة الأولى
!>  من AP السفلي. المثلث العلوي A(0:2,0:2) يتكون من
!>  منقول الأعمدة الثلاثة الأخيرة من AP السفلي.
!>  هذا يغطي الحالة N زوجي و TRANSR = 'N'.
!>
!>         RFP A                   RFP A
!>
!>        03 04 05                33 43 53
!>        13 14 15                00 44 54
!>        23 24 25                10 11 55
!>        33 34 35                20 21 22
!>        00 44 45                30 31 32
!>        01 11 55                40 41 42
!>        02 12 22                50 51 52
!>
!>
!>  الآن لتكن TRANSR = 'T'. RFP A في كلتا حالتي UPLO هي مجرد
!>  منقولة RFP A أعلاه. لذلك نحصل على:
!>
!>
!>           RFP A                   RFP A
!>
!>     03 13 23 33 00 01 02    33 00 10 20 30 40 50
!>     04 14 24 34 44 11 12    43 44 11 21 31 41 51
!>     05 15 25 35 45 55 22    53 54 55 22 32 42 52
!>
!>
!>  ثم ننظر في تنسيق التعبئة الكاملة المستطيلة (RFP) عندما يكون N
!>  فرديًا. نقدم مثالاً حيث N = 5.
!>
!>     AP علوي                 AP سفلي
!>
!>   00 01 02 03 04              00
!>      11 12 13 14              10 11
!>         22 23 24              20 21 22
!>            33 34              30 31 32 33
!>               44              40 41 42 43 44
!>
!>
!>  لتكن TRANSR = 'N'. RFP يخزن AP كما يلي:
!>  لـ UPLO = 'U' فإن شبه المنحرف العلوي A(0:4,0:2) يتكون من الأعمدة الثلاثة الأخيرة
!>  من AP العلوي. المثلث السفلي A(3:4,0:1) يتكون من
!>  منقول العمودين الأولين من AP العلوي.
!>  لـ UPLO = 'L' فإن شبه المنحرف السفلي A(0:4,0:2) يتكون من الأعمدة الثلاثة الأولى
!>  من AP السفلي. المثلث العلوي A(0:1,1:2) يتكون من
!>  منقول العمودين الأخيرين من AP السفلي.
!>  هذا يغطي الحالة N فردي و TRANSR = 'N'.
!>
!>         RFP A                   RFP A
!>
!>        02 03 04                00 33 43
!>        12 13 14                10 11 44
!>        22 23 24                20 21 22
!>        00 33 34                30 31 32
!>        01 11 44                40 41 42
!>
!>  الآن لتكن TRANSR = 'T'. RFP A في كلتا حالتي UPLO هي مجرد
!>  منقولة RFP A أعلاه. لذلك نحصل على:
!>
!>           RFP A                   RFP A
!>
!>     02 12 22 00 01             00 10 20 30 40 50
!>     03 13 23 33 11             33 11 21 31 41 51
!>     04 14 24 34 44             43 44 22 32 42 52
!>
