subroutine spbsv (character uplo, integer n, integer kd, integer nrhs, real, dimension( ldab, * ) ab, integer ldab, real, dimension( ldb, * ) b, integer ldb, integer info)

SPBSV تحسب حل نظام المعادلات الخطية A * X = B للمصفوفات الأخرى

الغرض:

!>
!> SPBSV يحسب حل نظام المعادلات الخطية الحقيقي
!>    A * X = B,
!> حيث A هي مصفوفة ذات نطاق متماثل وموجب محدد من الدرجة N×N، و X
!> و B هما مصفوفتان من الدرجة N×NRHS.
!>
!> يتم استخدام تفكيك تشولسكي لتحليل A إلى
!>    A = U**T * U، إذا كان UPLO = 'U'، أو
!>    A = L * L**T، إذا كان UPLO = 'L'،
!> حيث U هي مصفوفة ذات نطاق مثلثي علوي، و L هي مصفوفة ذات نطاق مثلثي
!> سفلي، بنفس عدد الأقطار العلوية أو السفلية مثل A.  ثم يتم استخدام الشكل المفكك لـ A لحل
!> نظام المعادلات A * X = B.
!>

المعلمات

UPLO 
!>          UPLO هو CHARACTER*1
!>          = 'U':  يتم تخزين الجزء العلوي من A؛
!>          = 'L':  يتم تخزين الجزء السفلي من A.
!>

N

!>          N هو INTEGER
!>          عدد المعادلات الخطية، أي درجة
!>          المصفوفة A.  N >= 0.
!>

KD

!>          KD هو INTEGER
!>          عدد الأقطار العلوية للمصفوفة A إذا كان UPLO = 'U'،
!>          أو عدد الأقطار السفلية إذا كان UPLO = 'L'.  KD >= 0.
!>

NRHS

!>          NRHS هو INTEGER
!>          عدد الأيادي اليمنى، أي عدد الأعمدة
!>          للمصفوفة B.  NRHS >= 0.
!>

AB

!>          AB هو مصفوفة REAL، البعد (LDAB,N)
!>          عند الدخول، الجزء العلوي أو السفلي من مصفوفة النطاق المتماثل
!>          A، مخزن في أول KD+1 صفوف من المصفوفة.  الـ
!>          العمود j من A مخزن في العمود j من المصفوفة AB
!>          كما يلي:
!>          إذا كان UPLO = 'U'، AB(KD+1+i-j,j) = A(i,j) لـ max(1,j-KD)<=i<=j؛
!>          إذا كان UPLO = 'L'، AB(1+i-j,j)    = A(i,j) لـ j<=i<=min(N,j+KD).
!>          انظر أدناه لمزيد من التفاصيل.
!>
!>          عند الخروج، إذا كان INFO = 0، فإن العامل المثلثي U أو L من
!>          تفكيك تشولسكي A = U**T*U أو A = L*L**T لمصفوفة النطاق
!>          A، بنفس تنسيق التخزين مثل A.
!>

LDAB

!>          LDAB هو INTEGER
!>          البعد الرائد للمصفوفة AB.  LDAB >= KD+1.
!>

B

!>          B هو مصفوفة REAL، البعد (LDB,NRHS)
!>          عند الدخول، المصفوفة B ذات الأيادي اليمنى من الدرجة N×NRHS.
!>          عند الخروج، إذا كان INFO = 0، فإن المصفوفة الحل X من الدرجة N×NRHS.
!>

LDB

!>          LDB هو INTEGER
!>          البعد الرائد للمصفوفة B.  LDB >= max(1,N).
!>

INFO

!>          INFO هو INTEGER
!>          = 0:  انتهاء ناجح
!>          < 0:  إذا كان INFO = -i، فإن الوسيط i كان له قيمة غير قانونية
!>          > 0:  إذا كان INFO = i، فإن الأصغر الرئيسي الرائد من الدرجة i
!>                لـ A ليس موجبًا، وبالتالي لم يكن من الممكن إكمال التفكيك،
!>                ولم يتم حساب الحل.
!>

تفاصيل إضافية:

!>
!>  مخطط تخزين النطاق موضح بالمثال التالي، عندما
!>  N = 6، KD = 2، و UPLO = 'U':
!>
!>  عند الدخول:                       عند الخروج:
!>
!>      *    *   a13  a24  a35  a46      *    *   u13  u24  u35  u46
!>      *   a12  a23  a34  a45  a56      *   u12  u23  u34  u45  u56
!>     a11  a22  a33  a44  a55  a66     u11  u22  u33  u44  u55  u66
!>
!>  بالمثل، إذا كان UPLO = 'L'، فإن تنسيق A هو كما يلي:
!>
!>  عند الدخول:                       عند الخروج:
!>
!>     a11  a22  a33  a44  a55  a66     l11  l22  l33  l44  l55  l66
!>     a21  a32  a43  a54  a65   *      l21  l32  l43  l54  l65   *
!>     a31  a42  a53  a64   *    *      l31  l42  l53  l64   *    *
!>
!>  العناصر * في المصفوفة لا تستخدمها الروتين.
!>
