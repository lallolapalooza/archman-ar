subroutine spot02 (character uplo, integer n, integer nrhs, real, dimension( lda, * ) a, integer lda, real, dimension( ldx, * ) x, integer ldx, real, dimension( ldb, * ) b, integer ldb, real, dimension( * ) rwork, real resid)

SPOT02

الغرض:

!>
!> SPOT02 تحسب الباقي لحل نظام المعادلات الخطية المتناظرة A*x = b:
!>
!>    RESID = norm(B - A*X) / ( norm(A) * norm(X) * EPS ),
!>
!> حيث EPS هي إبسيلون الآلة.
!>

المعاملات

UPLO 
!>          UPLO is CHARACTER*1
!>          يحدد ما إذا كان الجزء العلوي أو السفلي المثلثي من المصفوفة المتناظرة A مخزنًا:
!>          = 'U':  مثلثي علوي
!>          = 'L':  مثلثي سفلي
!>

N

!>          N is INTEGER
!>          عدد الصفوف والأعمدة للمصفوفة A. N >= 0.
!>

NRHS

!>          NRHS is INTEGER
!>          عدد أعمدة B، مصفوفة الطرف الأيمن.
!>          NRHS >= 0.
!>

A

!>          A is REAL array, dimension (LDA,N)
!>          المصفوفة المتناظرة الأصلية A.
!>

LDA

!>          LDA is INTEGER
!>          البعد الرائد للمصفوفة A. LDA >= max(1,N)
!>

X

!>          X is REAL array, dimension (LDX,NRHS)
!>          متجهات الحل المحسوبة لنظام المعادلات الخطية.
!>

LDX

!>          LDX is INTEGER
!>          البعد الرائد للمصفوفة X. LDX >= max(1,N).
!>

B

!>          B is REAL array, dimension (LDB,NRHS)
!>          عند الإدخال، متجهات الطرف الأيمن لنظام المعادلات الخطية.
!>          عند الخروج، يتم الكتابة فوق B بالفرق B - A*X.
!>

LDB

!>          LDB is INTEGER
!>          البعد الرائد للمصفوفة B. LDB >= max(1,N).
!>

RWORK

!>          RWORK is REAL array, dimension (N)
!>

RESID

!>          RESID is REAL
!>          الحد الأقصى لعدد الأطراف اليمنى من
!>          norm(B - A*X) / ( norm(A) * norm(X) * EPS ).
!>
