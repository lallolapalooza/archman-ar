توثيق الوظيفة/الإجراء الفرعي

subroutine sptt05 (integer n, integer nrhs, real, dimension( * ) d, real, dimension( * ) e, real, dimension( ldb, * ) b, integer ldb, real, dimension( ldx, * ) x, integer ldx, real, dimension( ldxact, * ) xact, integer ldxact, real, dimension( * ) ferr, real, dimension( * ) berr, real, dimension( * ) reslts)

SPTT05

الغرض:

!>
!> SPTT05 تختبر حدود الخطأ من التنقيح التكراري للحل
!> المحسوب لنظام المعادلات A*X = B، حيث A مصفوفة ثلاثية قطرية متناظرة من الرتبة n.
!>
!> RESLTS(1) = اختبار حد الخطأ
!>           = norm(X - XACT) / ( norm(X) * FERR )
!>
!> تُعاد قيمة كبيرة إذا لم تكن هذه النسبة أقل من واحد.
!>
!> RESLTS(2) = الخطأ المتبقي من إجراء التنقيح التكراري
!>           = الحد الأقصى لـ BERR / ( NZ*EPS + (*) )، حيث
!>             (*) = NZ*UNFL / (min_i (abs(A)*abs(X) +abs(b))_i )
!>             و NZ = الحد الأقصى لعدد العناصر غير الصفرية في أي صف من A، زائد 1
!>

المعلمات:

N 
!>          N هو عدد صحيح (INTEGER)
!>          عدد صفوف المصفوفات X، B، و XACT، ورتبة المصفوفة A. N >= 0.
!>

NRHS

!>          NRHS هو عدد صحيح (INTEGER)
!>          عدد أعمدة المصفوفات X، B، و XACT. NRHS >= 0.
!>

D

!>          D هي مصفوفة حقيقية (REAL array)، البعد (N)
!>          العناصر القطرية n للمصفوفة الثلاثية القطرية A.
!>

E

!>          E هي مصفوفة حقيقية (REAL array)، البعد (N-1)
!>          العناصر دون القطرية (n-1) للمصفوفة الثلاثية القطرية A.
!>

B

!>          B هي مصفوفة حقيقية (REAL array)، البعد (LDB,NRHS)
!>          متجهات الطرف الأيمن لنظام المعادلات الخطية.
!>

LDB

!>          LDB هو عدد صحيح (INTEGER)
!>          البعد الرئيسي للمصفوفة B. LDB >= max(1,N).
!>

X

!>          X هي مصفوفة حقيقية (REAL array)، البعد (LDX,NRHS)
!>          متجهات الحل المحسوبة. كل متجه مخزن كعمود في المصفوفة X.
!>

LDX

!>          LDX هو عدد صحيح (INTEGER)
!>          البعد الرئيسي للمصفوفة X. LDX >= max(1,N).
!>

XACT

!>          XACT هي مصفوفة حقيقية (REAL array)، البعد (LDX,NRHS)
!>          متجهات الحل الدقيقة. كل متجه مخزن كعمود في المصفوفة XACT.
!>

LDXACT

!>          LDXACT هو عدد صحيح (INTEGER)
!>          البعد الرئيسي للمصفوفة XACT. LDXACT >= max(1,N).
!>

FERR

!>          FERR هي مصفوفة حقيقية (REAL array)، البعد (NRHS)
!>          حدود الخطأ الأمامي المقدرة لكل متجه حل X. إذا كانت XTRUE هي الحل الحقيقي،
!>          فإن FERR تحد من قيمة أكبر مدخل في (X - XTRUE) مقسومة على قيمة أكبر مدخل في X.
!>

BERR

!>          BERR هي مصفوفة حقيقية (REAL array)، البعد (NRHS)
!>          الخطأ الخلفي النسبي المكوني لكل متجه حل
!>          (أي، أصغر تغيير نسبي في أي مدخل في A أو B يجعل X حلاً دقيقًا).
!>

RESLTS

!>          RESLTS هي مصفوفة حقيقية (REAL array)، البعد (2)
!>          الحد الأقصى لمتجهات الحل NRHS للنسب التالية:
!>          RESLTS(1) = norm(X - XACT) / ( norm(X) * FERR )
!>          RESLTS(2) = BERR / ( NZ*EPS + (*) )
!>
