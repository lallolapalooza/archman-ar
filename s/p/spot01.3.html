subroutine spot01 (character uplo, integer n, real, dimension( lda, * ) a, integer lda, real, dimension( ldafac, * ) afac, integer ldafac, real, dimension( * ) rwork, real resid)

SPOT01

الغرض:

!>
!> SPOT01 يعيد بناء مصفوفة متماثلة موجبة محددة A من
!> تحليلها L*L' أو U'*U ويحسب الباقي
!>    norm( L*L' - A ) / ( N * norm(A) * EPS ) أو
!>    norm( U'*U - A ) / ( N * norm(A) * EPS ),
!> حيث EPS هو الخطأ المطبعي الآلي.
!>

المعلمات

UPLO 
!>          UPLO هو CHARACTER*1
!>          يحدد ما إذا كان الجزء العلوي أو السفلي المثلثي للمصفوفة المتماثلة A مخزناً:
!>          = 'U':  مثلث علوي
!>          = 'L':  مثلث سفلي
!>

N

!>          N هو INTEGER
!>          عدد الصفوف والأعمدة للمصفوفة A.  N >= 0.
!>

A

!>          A هو مصفوفة REAL، ذات بعد (LDA,N)
!>          المصفوفة المتماثلة الأصلية A.
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرئيسي للمصفوفة A.  LDA >= max(1,N)
!>

AFAC

!>          AFAC هو مصفوفة REAL، ذات بعد (LDAFAC,N)
!>          عند الإدخال، العامل L أو U من تحليل L * L**T أو U**T * U
!>          للمصفوفة A.
!>          يتم الكتابة فوقها بالمصفوفة المعاد بناؤها، ثم بالفرق
!>          L * L**T - A (أو U**T * U - A).
!>

LDAFAC

!>          LDAFAC هو INTEGER
!>          البعد الرئيسي للمصفوفة AFAC.  LDAFAC >= max(1,N).
!>

RWORK

!>          RWORK هي مصفوفة REAL، ذات بعد (N)
!>

RESID

!>          RESID هو REAL
!>          إذا كان UPLO = 'L'، فإن norm(L * L**T - A) / ( N * norm(A) * EPS )
!>          إذا كان UPLO = 'U'، فإن norm(U**T * U - A) / ( N * norm(A) * EPS )
!>
