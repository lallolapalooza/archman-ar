SPLAY_HEAD(3bsd)

الحزمة: libbsd

التراخيص: custom

الوصف

تُستخدم هذه الماكرو لإنشاء واستخدام الأشجار المستقرة (splay trees) وأشجار RB (red-black trees) في C.

يُعرّف الماكرو SPLAY_HEAD() نوع رأس الشجرة. يجب استدعاء هذا الماكرو مرة واحدة لكل شجرة. HEADNAME هو اسم نوع رأس الشجرة، و TYPE هو اسم نوع العقدة التي ستُخزن في الشجرة.

يُعرّف الماكرو SPLAY_ENTRY() نوع الإدخال الذي سيُستخدم في بنية العقدة. يتطلب هذا وسيطين: TYPE وهو نوع بنية العقدة، و FIELD وهو اسم الحقل الذي يحتوي على بنية الإدخال.

الماكرو SPLAY_INIT() يقوم بتهيئة الشجرة المشار إليها بواسطة head.

يمكن أيضًا تهيئة الشجرة المستقرة بشكل ثابت باستخدام الماكرو SPLAY_INITIALIZER() على النحو التالي:

SPLAY_HEAD(HEADNAME, TYPE) head = SPLAY_INITIALIZER(&head);

يقوم الماكرو SPLAY_INSERT() بإدراج العنصر الجديد elm في الشجرة. عند النجاح، يُرجع NULL. إذا كان هناك عنصر مطابق موجود بالفعل في الشجرة، يتم إلغاء الإدراج، ويتم إرجاع مؤشر إلى العنصر الموجود.

يقوم الماكرو SPLAY_REMOVE() بإزالة العنصر elm من الشجرة المشار إليها بواسطة head. يُرجع SPLAY_REMOVE() elm.

يمكن استخدام الماكرو SPLAY_FIND() للبحث عن عنصر معين في الشجرة. يبحث SPLAY_FIND() عن العقدة ذات المفتاح المطابق لـ elm.

struct TYPE find, *res;
find.key = 30;
res = SPLAY_FIND(NAME, &head, &find);

يمكن استخدام الماكرو SPLAY_ROOT() للحصول على جذر الشجرة.

يمكن استخدام الماكرو SPLAY_MIN() للحصول على أصغر عنصر في الشجرة.

يمكن استخدام الماكرو SPLAY_MAX() للحصول على أكبر عنصر في الشجرة.

يمكن استخدام الماكرو SPLAY_NEXT() للانتقال إلى العنصر التالي في الشجرة.

يمكن استخدام الماكرو SPLAY_PREV() للانتقال إلى العنصر السابق في الشجرة.

أو، لتبسيط الأمر، يمكن استخدام الماكرو SPLAY_FOREACH() للتكرار عبر الشجرة:

SPLAY_FOREACH(np, NAME, &head)

يجب استخدام الماكرو SPLAY_EMPTY() للتحقق مما إذا كانت الشجرة المستقرة فارغة.

أشجار RB (Red-Black Trees)

توفر واجهة مماثلة لأشجار RB. تحتاج إلى تعريف بنية RB_ENTRY() في بنية العقدة الخاصة بك، مثل:

struct mynode {
    RB_ENTRY(mynode) next;
    /* other data */
};

ثم، قم بتعريف رأس الشجرة باستخدام RB_HEAD(). على سبيل المثال:

RB_HEAD(mynodes, mynode) head = RB_INITIALIZER(&head);

لتعريف دالة المقارنة، قم بإنشاء وظيفة مثل:

int
mynode_cmp(struct mynode *a, struct mynode *b)
{
    /* Compare a and b based on their key fields */
    return (a->key < b->key ? -1 : a->key > b->key);
}

بعد ذلك، قم بتعريف نموذج دالة المقارنة باستخدام RB_PROTOTYPE() وقم بتوليد الماكرو باستخدام RB_GENERATE(). يجب أن تكون هذه التعريفات متاحة في ملف .c الخاص بك.

RB_PROTOTYPE(mynodes, mynode, next, mynode_cmp)
RB_GENERATE(mynodes, mynode, next, mynode_cmp)

يجب أن تحتوي بنية العقدة على حقل مفتاح للمقارنة. يشبه استخدام الماكرو RB_INSERT() و RB_REMOVE() و RB_FIND() و RB_NEXT() وما إلى ذلك، إلى حد كبير، استخداماتهم في أشجار Splay، ولكن يجب تحديد اسم رأس الشجرة (مثل mynodes) كوسيط أول.

يجب استدعاء وظائف المقارنة فقط مرة واحدة. تمت مناقشة الوسيط CMP بشكل شامل في قسم أشجار Splay. يجب استخدامه مرة واحدة فقط.

أخيرًا، الوسيط CMP هو اسم وظيفة تُستخدم لمقارنة عقد الأشجار ببعضها البعض. تأخذ الوظيفة وسيطين من نوع struct TYPE *. إذا كان الوسيط الأول أصغر من الثاني، فإن الوظيفة تُرجع قيمة أصغر من الصفر. إذا كانت متساوية، فإن الوظيفة تُرجع صفرًا. بخلاف ذلك، يجب أن تُرجع قيمة أكبر من الصفر. تحدد وظيفة المقارنة ترتيب عناصر الشجرة.

الماكرو RB_INIT() يقوم بتهيئة الشجرة المشار إليها بواسطة head.

يمكن أيضًا تهيئة شجرة RB بشكل ثابت باستخدام الماكرو RB_INITIALIZER() على النحو التالي:

RB_HEAD(HEADNAME, TYPE) head = RB_INITIALIZER(&head);

يقوم الماكرو RB_INSERT() بإدراج العنصر الجديد elm في الشجرة. عند النجاح، يُرجع NULL. إذا كان هناك عنصر مطابق موجود بالفعل في الشجرة، يتم إلغاء الإدراج، ويتم إرجاع مؤشر إلى العنصر الموجود.

يقوم الماكرو RB_REMOVE() بإزالة العنصر elm من الشجرة المشار إليها بواسطة head. يُرجع RB_REMOVE() elm.

يمكن استخدام الماكروين RB_FIND() و RB_NFIND() للبحث عن عنصر معين في الشجرة. يبحث RB_FIND() عن العقدة ذات المفتاح المطابق لـ elm. يبحث RB_NFIND() عن أول عقدة أكبر من أو تساوي مفتاح البحث.

struct TYPE find, *res;
find.key = 30;
res = RB_FIND(NAME, &head, &find);

يمكن استخدام الماكروات RB_ROOT() و RB_MIN() و RB_MAX() و RB_NEXT() و RB_PREV() للتنقل في الشجرة:

for (np = RB_MIN(NAME, &head); np != NULL; np = RB_NEXT(NAME, &head, np))

أو، لتبسيط الأمر، يمكن استخدام الماكروين RB_FOREACH() أو RB_FOREACH_REVERSE():

RB_FOREACH(np, NAME, &head)

الماكروات RB_FOREACH_SAFE() و RB_FOREACH_REVERSE_SAFE() تتنقل في الشجرة المشار إليها بواسطة head في اتجاه أمامي أو عكسي على التوالي، وتعين كل عنصر بدوره إلى np. ومع ذلك، على عكس نظائرها غير الآمنة، فإنها تسمح بإزالة np وكذلك تحريره من داخل الحلقة بأمان دون التداخل مع عملية التنقل.

يجب استخدام الماكرو RB_EMPTY() للتحقق مما إذا كانت شجرة RB فارغة.

أمثلة

يوضح المثال التالي كيفية تعريف شجرة RB تحتوي على أعداد صحيحة. يتم إدراج القيم فيها وطباعة محتويات الشجرة بترتيبها. أخيرًا، تتم طباعة الهيكل الداخلي للشجرة.

#include <sys/tree.h>
#include <err.h>
#include <stdio.h>
#include <stdlib.h>

struct node {
	RB_ENTRY(node) entry;
	int i;
};

int	intcmp(struct node *, struct node *);
void	print_tree(struct node *);

int
intcmp(struct node *e1, struct node *e2)
{
	return (e1->i < e2->i ? -1 : e1->i > e2->i);
}

RB_HEAD(inttree, node) head = RB_INITIALIZER(&head);
RB_PROTOTYPE(inttree, node, entry, intcmp)
RB_GENERATE(inttree, node, entry, intcmp)

int testdata[] = {
	20, 16, 17, 13, 3, 6, 1, 8, 2, 4, 10, 19, 5, 9, 12, 15, 18,
	7, 11, 14
};

void
print_tree(struct node *n)
{
	struct node *left, *right;

	if (n == NULL) {
		printf("nil");
		return;
	}
	left = RB_LEFT(n, entry);
	right = RB_RIGHT(n, entry);
	if (left == NULL && right == NULL)
		printf("%d", n->i);
	else {
		printf("%d(", n->i);
		print_tree(left);
		printf(",");
		print_tree(right);
		printf(")");
	}
}

int
main(void)
{
	int i;
	struct node *n;

	for (i = 0; i < sizeof(testdata) / sizeof(testdata[0]); i++) {
		if ((n = malloc(sizeof(struct node))) == NULL)
			err(1, NULL);
		n->i = testdata[i];
		RB_INSERT(inttree, &head, n);
	}

	RB_FOREACH(n, inttree, &head) {
		printf("%d\n", n->i);
	}
	print_tree(RB_ROOT(&head));
	printf("\n");
	return (0);
}

ملاحظات

محاولة تحرير شجرة بالطريقة التالية خطأ شائع:

SPLAY_FOREACH(var, NAME, &head) {
	SPLAY_REMOVE(NAME, &head, var);
	free(var);
}

نظرًا لأنه تم تحرير var، فإن الماكرو FOREACH() يشير إلى مؤشر ربما تم إعادة تخصيصه بالفعل. يتطلب الكود الصحيح متغيرًا ثانيًا.

for (var = SPLAY_MIN(NAME, &head); var != NULL; var = nxt) {
	nxt = SPLAY_NEXT(NAME, &head, var);
	SPLAY_REMOVE(NAME, &head, var);
	free(var);
}
