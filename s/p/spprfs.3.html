subroutine spprfs (character uplo, integer n, integer nrhs, real, dimension( * ) ap, real, dimension( * ) afp, real, dimension( ldb, * ) b, integer ldb, real, dimension( ldx, * ) x, integer ldx, real, dimension( * ) ferr, real, dimension( * ) berr, real, dimension( * ) work, integer, dimension( * ) iwork, integer info)

SPPRFS

الغرض:

!>
!> SPPRFS يحسن الحل المحسوب لنظام المعادلات الخطية
!> عندما تكون مصفوفة المعاملات متماثلة موجبة محددة
!> ومعبأة، ويوفر حدود الخطأ وتقديرات الخطأ الخلفي
!> للحل.
!>

المعلمات

UPLO 
!>          UPLO is CHARACTER*1
!>          = 'U':  مثلث علوي من A مخزن؛
!>          = 'L':  مثلث سفلي من A مخزن.
!>

N

!>          N is INTEGER
!>          ترتيب المصفوفة A.  N >= 0.
!>

NRHS

!>          NRHS is INTEGER
!>          عدد الأطراف اليمنى، أي عدد الأعمدة
!>          للمصفوفتين B و X.  NRHS >= 0.
!>

AP

!>          AP is REAL array, dimension (N*(N+1)/2)
!>          المثلث العلوي أو السفلي للمصفوفة المتماثلة A، معبأة
!>          حسب العمود في مصفوفة خطية.  يتم تخزين العمود j من A
!>          في المصفوفة AP كما يلي:
!>          إذا كان UPLO = 'U'، فإن AP(i + (j-1)*j/2) = A(i,j) لـ 1<=i<=j؛
!>          إذا كان UPLO = 'L'، فإن AP(i + (j-1)*(2n-j)/2) = A(i,j) لـ j<=i<=n.
!>

AFP

!>          AFP is REAL array, dimension (N*(N+1)/2)
!>          العامل المثلثي U أو L من التحليل Cholesky
!>          A = U**T*U أو A = L*L**T، كما تم حسابه بواسطة SPPTRF/CPPTRF،
!>          معبأة حسب العمود في مصفوفة خطية بنفس تنسيق A
!>          (انظر AP).
!>

B

!>          B is REAL array, dimension (LDB,NRHS)
!>          مصفوفة الطرف الأيمن B.
!>

LDB

!>          LDB is INTEGER
!>          البعد الرئيسي للمصفوفة B.  LDB >= max(1,N).
!>

X

!>          X is REAL array, dimension (LDX,NRHS)
!>          عند الإدخال، مصفوفة الحل X، كما تم حسابها بواسطة SPPTRS.
!>          عند الخروج، مصفوفة الحل المحسنة X.
!>

LDX

!>          LDX is INTEGER
!>          البعد الرئيسي للمصفوفة X.  LDX >= max(1,N).
!>

FERR

!>          FERR is REAL array, dimension (NRHS)
!>          الحد الأمامي المقدر للخطأ لكل متجه حل
!>          X(j) (العمود j من مصفوفة الحل X).
!>          إذا كان XTRUE هو الحل الحقيقي المقابل لـ X(j)، فإن FERR(j)
!>          هو حد أعلى مقدر لقيمة أكبر
!>          عنصر في (X(j) - XTRUE) مقسومًا على قيمة أكبر
!>          عنصر في X(j).  التقدير موثوق به مثل
!>          تقدير RCOND، وهو تقريبًا دائمًا
!>          زيادة تقديرية للخطأ الحقيقي.
!>

BERR

!>          BERR is REAL array, dimension (NRHS)
!>          خطأ الخلفية النسبي على مستوى المكون لكل متجه حل
!>          X(j) (أي، أصغر تغيير نسبي في
!>          أي عنصر في A أو B يجعل X(j) حلاً دقيقًا).
!>

WORK

!>          WORK is REAL array, dimension (3*N)
!>

IWORK

!>          IWORK is INTEGER array, dimension (N)
!>

INFO

!>          INFO is INTEGER
!>          = 0:  انتهاء ناجح
!>          < 0:  إذا كان INFO = -i، فإن المعامل i لديه قيمة غير قانونية
!>
