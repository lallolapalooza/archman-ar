توثيق الوظيفة/الروتين الفرعي

subroutine spteqr (character compz, integer n, real, dimension( * ) d, real, dimension( * ) e, real, dimension( ldz, * ) z, integer ldz, real, dimension( * ) work, integer info)

SPTEQR

الغرض:

!>
!> SPTEQR تحسب جميع القيم الذاتية، وبشكل اختياري، المتجهات الذاتية لمصفوفة ثلاثية قطرية متناظرة وموجبة تمامًا عن طريق تحليل المصفوفة أولاً باستخدام SPTTRF، ثم استدعاء SBDSQR لحساب قيم التفرد للعامل الثنائي.
!>
!> يحسب هذا الروتين الفرعي القيم الذاتية للمصفوفة الثلاثية القطرية الموجبة تمامًا بدقة نسبية عالية. هذا يعني أنه إذا تراوحت القيم الذاتية عبر العديد من رتب الحجم، فسيتم حساب القيم الذاتية الصغيرة والمتجهات الذاتية المقابلة بدقة أكبر من، على سبيل المثال، الطريقة المربعة القياسية.
!>
!> يمكن أيضًا العثور على المتجهات الذاتية لمصفوفة متناظرة كاملة أو نطاقية موجبة تمامًا إذا تم استخدام SSYTRD أو SSPTRD أو SSBTRD لتقليل هذه المصفوفة إلى شكل ثلاثي الأقطار. (ومع ذلك، فإن تقليل المصفوفة إلى شكل ثلاثي الأقطار قد يمنع إمكانية الحصول على دقة نسبية عالية في القيم الذاتية الصغيرة للمصفوفة الأصلية، إذا تراوحت هذه القيم الذاتية عبر العديد من رتب الحجم.)
!>

المعلمات

COMPZ 
!>          COMPZ is CHARACTER*1
!>          = 'N':  احسب القيم الذاتية فقط.
!>          = 'V':  احسب المتجهات الذاتية للمصفوفة المتناظرة الأصلية
!>                  أيضًا. المصفوفة Z تحتوي على المصفوفة المتعامدة
!>                  المستخدمة لتقليل المصفوفة الأصلية إلى
!>                  شكل ثلاثي الأقطار.
!>          = 'I':  احسب المتجهات الذاتية للمصفوفة ثلاثية الأقطار
!>                  أيضًا.
!>

N

!>          N is INTEGER
!>          ترتيب المصفوفة.  N >= 0.
!>

D

!>          D is REAL array, dimension (N)
!>          عند الإدخال، عناصر القطر n للمصفوفة ثلاثية الأقطار.
!>          عند الخروج الطبيعي، تحتوي D على القيم الذاتية، بترتيب تنازلي.
!>

E

!>          E is REAL array, dimension (N-1)
!>          عند الإدخال، عناصر القطر الفرعي (n-1) للمصفوفة ثلاثية الأقطار.
!>          عند الخروج، تم تدمير E.
!>

Z

!>          Z is REAL array, dimension (LDZ, N)
!>          عند الإدخال، إذا كان COMPZ = 'V'، المصفوفة المتعامدة المستخدمة في
!>          التقليل إلى شكل ثلاثي الأقطار.
!>          عند الخروج، إذا كان COMPZ = 'V'، المتجهات الذاتية المتعامدة الطبيعية
!>          للمصفوفة المتناظرة الأصلية؛
!>          إذا كان COMPZ = 'I'، المتجهات الذاتية المتعامدة الطبيعية لـ
!>          المصفوفة ثلاثية الأقطار.
!>          إذا كان INFO > 0 عند الخروج، تحتوي Z على المتجهات الذاتية المرتبطة
!>          فقط بالقيم الذاتية المخزنة.
!>          إذا كان COMPZ = 'N'، فلا يتم الرجوع إلى Z.
!>

LDZ

!>          LDZ is INTEGER
!>          البعد الرائد للمصفوفة Z.  LDZ >= 1، وإذا
!>          كان COMPZ = 'V' أو 'I'، فإن LDZ >= max(1,N).
!>

WORK

!>          WORK is REAL array, dimension (4*N)
!>

INFO

!>          INFO is INTEGER
!>          = 0:  انتهاء ناجح.
!>          < 0:  إذا كان INFO = -i، فقد كان الوسيط i ذو قيمة غير قانونية.
!>          > 0:  إذا كان INFO = i، و i هو:
!>                <= N  لم يكن من الممكن إجراء تحليل Cholesky للمصفوفة
!>                      لأن القطر الرئيسي الرائد من الدرجة i لم يكن موجبًا.
!>                > N   فشل خوارزمية SVD في التقارب؛
!>                      إذا كان INFO = N+i، فإن i عناصر خارج القطر من
!>                      العامل الثنائي لم تتقارب إلى الصفر.
!>
