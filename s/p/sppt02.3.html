subroutine sppt02 (character uplo, integer n, integer nrhs, real, dimension( * ) a, real, dimension( ldx, * ) x, integer ldx, real, dimension( ldb, * ) b, integer ldb, real, dimension( * ) rwork, real resid)

SPPT02

الغرض:

!>
!> SPPT02 يحسب الباقي في حل نظام المعادلات الخطية المتناظرة  A*x = b  عند استخدام
!> تخزين مضغوط للمصفوفة المعاملة. النسبة المحسوبة هي
!>
!>    RESID = norm(B - A*X) / ( norm(A) * norm(X) * EPS),
!>
!> حيث EPS هو دقة الآلة.
!>

المعلمات

UPLO 
!>          UPLO is CHARACTER*1
!>          يحدد ما إذا كان الجزء العلوي أو السفلي من المصفوفة المتناظرة A
!>          مخزنًا:
!>          = 'U':  مثلثي علوي
!>          = 'L':  مثلثي سفلي
!>

N

!>          N is INTEGER
!>          عدد الصفوف والأعمدة للمصفوفة A.  N >= 0.
!>

NRHS

!>          NRHS is INTEGER
!>          عدد أعمدة B، مصفوفة الأطراف اليمنى.
!>          NRHS >= 0.
!>

A

!>          A is REAL array, dimension (N*(N+1)/2)
!>          المصفوفة المتناظرة الأصلية A، مخزنة كمصفوفة
!>          مثلثية مضغوطة.
!>

X

!>          X is REAL array, dimension (LDX,NRHS)
!>          متجهات الحل المحسوبة لنظام المعادلات الخطية.
!>

LDX

!>          LDX is INTEGER
!>          البعد الرائد للمصفوفة X.   LDX >= max(1,N).
!>

B

!>          B is REAL array, dimension (LDB,NRHS)
!>          عند الإدخال، متجهات الطرف الأيمن لنظام المعادلات الخطية.
!>          عند الخروج، يتم استبدال B بالفرق B - A*X.
!>

LDB

!>          LDB is INTEGER
!>          البعد الرائد للمصفوفة B.  LDB >= max(1,N).
!>

RWORK

!>          RWORK is REAL array, dimension (N)
!>

RESID

!>          RESID is REAL
!>          الحد الأقصى على عدد الأطراف اليمنى
!>          norm(B - A*X) / ( norm(A) * norm(X) * EPS ).
!>
