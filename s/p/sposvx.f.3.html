subroutine sposvx (character fact, character uplo, integer n, integer nrhs, real, dimension( lda, * ) a, integer lda, real, dimension( ldaf, * ) af, integer ldaf, character equed, real, dimension( * ) s, real, dimension( ldb, * ) b, integer ldb, real, dimension( ldx, * ) x, integer ldx, real rcond, real, dimension( * ) ferr, real, dimension( * ) berr, real, dimension( * ) work, integer, dimension( * ) iwork, integer info)

SPOSVX تحسب حل نظام المعادلات الخطية A * X = B لمصفوفات PO

الغرض:

!>
!> يستخدم SPOSVX تحليل Cholesky A = U**T*U أو A = L*L**T لحساب الحل لنظام المعادلات الخطية الحقيقية
!>    A * X = B،
!> حيث A هي مصفوفة N-by-N متناظرة وموجبة قطعيًا، و X و B هما مصفوفتان N-by-NRHS.
!>
!> يتم أيضًا توفير حدود الخطأ للحل وتقدير شرطي.
!>

الوصف:

!>
!> يتم تنفيذ الخطوات التالية:
!>
!> 1. إذا كان FACT = 'E'، يتم حساب عوامل قياس حقيقية لمعادلة النظام:
!>       diag(S) * A * diag(S) * inv(diag(S)) * X = diag(S) * B
!>    يعتمد ما إذا كان النظام سيتم معادلته أم لا على قياس المصفوفة A، ولكن إذا تم استخدام المعادلة،
!>    فإن A يتم استبدالها بـ diag(S)*A*diag(S) و B بـ diag(S)*B.
!>
!> 2. إذا كان FACT = 'N' أو 'E'، يتم استخدام تحليل Cholesky لتحليل المصفوفة A (بعد المعادلة إذا كان FACT = 'E') بالشكل:
!>       A = U**T* U،  إذا كان UPLO = 'U'، أو
!>       A = L * L**T،  إذا كان UPLO = 'L'،
!>    حيث U هي مصفوفة مثلثة عليا و L هي مصفوفة مثلثة سفلى.
!>
!> 3. إذا كان الحد الرئيسي الأولي من الرتبة i غير موجب، فإن الروتين يعود بـ INFO = i.
!>    وإلا، يتم استخدام الشكل المحلل لـ A لتقدير رقم شرط المصفوفة A.
!>    إذا كان مقلوب رقم الشرط أقل من دقة الآلة، يتم إرجاع INFO = N+1 كتحذير،
!>    ولكن الروتين يستمر في حل X وحساب حدود الخطأ كما هو موضح أدناه.
!>
!> 4. يتم حل نظام المعادلات لـ X باستخدام الشكل المحلل لـ A.
!>
!> 5. يتم تطبيق تحسين تكراري لتحسين الحل المحسوب وحساب حدود الخطأ وتقديرات الخطأ الخلفي له.
!>
!> 6. إذا تم استخدام المعادلة، يتم ضرب المصفوفة X مسبقًا بـ diag(S) بحيث تحل النظام الأصلي
!>    قبل المعادلة.
!>

المعلمات

FACT 
!>          FACT هو CHARACTER*1
!>          يحدد ما إذا كان الشكل المحلل للمصفوفة A معروضًا عند الدخول، وإذا لم يكن كذلك،
!>          فهل يجب معادلة المصفوفة A قبل تحليلها.
!>          = 'F':  عند الدخول، يحتوي AF على الشكل المحلل.
!>                  إذا كان EQUED = 'Y'، فقد تم معادلة المصفوفة A
!>                  بعوامل قياس معطاة بواسطة S. لن يتم تعديل A و AF.
!>          = 'N':  سيتم نسخ المصفوفة A إلى AF وتحليلها.
!>          = 'E':  سيتم معادلة المصفوفة A إذا لزم الأمر، ثم نسخها إلى AF وتحليلها.
!>

UPLO

!>          UPLO هو CHARACTER*1
!>          = 'U':  يتم تخزين المثلث العلوي لـ A؛
!>          = 'L':  يتم تخزين المثلث السفلي لـ A.
!>

N

!>          N هو INTEGER
!>          عدد المعادلات الخطية، أي رتبة المصفوفة A.  N >= 0.
!>

NRHS

!>          NRHS هو INTEGER
!>          عدد الأطراف اليمنى، أي عدد أعمدة المصفوفتين B و X.  NRHS >= 0.
!>

A

!>          A هو مصفوفة REAL، ذات البعد (LDA,N)
!>          عند الدخول، المصفوفة المتناظرة A، باستثناء إذا كان FACT = 'F' و
!>          EQUED = 'Y'، فإن A يجب أن تحتوي على المصفوفة المعادلة
!>          diag(S)*A*diag(S).  إذا كان UPLO = 'U'، فإن الجزء العلوي
!>          المثلثي من A بحجم N-by-N يحتوي على الجزء العلوي
!>          المثلثي للمصفوفة A، والجزء السفلي المثلثي الصارم لـ A
!>          لا يتم الرجوع إليه.  إذا كان UPLO = 'L'، فإن الجزء السفلي
!>          المثلثي من A بحجم N-by-N يحتوي على الجزء السفلي
!>          المثلثي للمصفوفة A، والجزء العلوي المثلثي الصارم لـ A
!>          لا يتم الرجوع إليه.  لا يتم تعديل A إذا كان FACT = 'F' أو 'N'،
!>          أو إذا كان FACT = 'E' و EQUED = 'N' عند الخروج.
!>
!>          عند الخروج، إذا كان FACT = 'E' و EQUED = 'Y'، يتم استبدال A بـ
!>          diag(S)*A*diag(S).
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرئيسي للمصفوفة A.  LDA >= max(1,N).
!>

AF

!>          AF هو مصفوفة REAL، ذات البعد (LDAF,N)
!>          إذا كان FACT = 'F'، فإن AF هو معامل إدخال ويحتوي عند الدخول
!>          على العامل المثلثي U أو L من تحليل Cholesky
!>          A = U**T*U أو A = L*L**T، بنفس تنسيق التخزين لـ A.
!>          إذا كان EQUED .ne. 'N'، فإن AF هو الشكل المحلل
!>          للمصفوفة المعادلة diag(S)*A*diag(S).
!>
!>          إذا كان FACT = 'N'، فإن AF هو معامل إخراج ويحتوي عند الخروج
!>          على العامل المثلثي U أو L من تحليل Cholesky
!>          A = U**T*U أو A = L*L**T للمصفوفة الأصلية
!>          A.
!>
!>          إذا كان FACT = 'E'، فإن AF هو معامل إخراج ويحتوي عند الخروج
!>          على العامل المثلثي U أو L من تحليل Cholesky
!>          A = U**T*U أو A = L*L**T للمصفوفة المعادلة
!>          A (انظر وصف A لشكل المصفوفة المعادلة).
!>

LDAF

!>          LDAF هو INTEGER
!>          البعد الرئيسي للمصفوفة AF.  LDAF >= max(1,N).
!>

EQUED

!>          EQUED هو CHARACTER*1
!>          يحدد شكل المعادلة التي تم إجراؤها.
!>          = 'N':  لا يوجد معادلة (دائمًا صحيح إذا كان FACT = 'N').
!>          = 'Y':  تم إجراء المعادلة، أي تم استبدال A بـ
!>                  diag(S) * A * diag(S).
!>          EQUED هو معامل إدخال إذا كان FACT = 'F'؛ وإلا، فهو معامل إخراج.
!>

S

!>          S هو مصفوفة REAL، ذات البعد (N)
!>          عوامل القياس لـ A؛ لا يتم الوصول إليها إذا كان EQUED = 'N'. S هو
!>          معامل إدخال إذا كان FACT = 'F'؛ وإلا، S هو معامل إخراج.
!>          إذا كان FACT = 'F' و EQUED = 'Y'، يجب أن يكون كل عنصر من S
!>          موجبًا.
!>

B

!>          B هو مصفوفة REAL، ذات البعد (LDB,NRHS)
!>          عند الدخول، مصفوفة الطرف الأيمن B بحجم N-by-NRHS.
!>          عند الخروج، إذا كان EQUED = 'N'، لا يتم تعديل B؛ إذا كان EQUED = 'Y'،
!>          يتم استبدال B بـ diag(S) * B.
!>

LDB

!>          LDB هو INTEGER
!>          البعد الرئيسي للمصفوفة B.  LDB >= max(1,N).
!>

X

!>          X هو مصفوفة REAL، ذات البعد (LDX,NRHS)
!>          إذا كان INFO = 0 أو INFO = N+1، فإن مصفوفة الحل X بحجم N-by-NRHS
!>          للنظام الأصلي للمعادلات.  لاحظ أنه إذا كان EQUED = 'Y'،
!>          يتم تعديل A و B عند الخروج، والحل للنظام
!>          المعادلة هو inv(diag(S))*X.
!>

LDX

!>          LDX هو INTEGER
!>          البعد الرئيسي للمصفوفة X.  LDX >= max(1,N).
!>

RCOND

!>          RCOND is REAL
!>          The estimate of the reciprocal condition number of the matrix
!>          A after equilibration (if done).  If RCOND is less than the
!>          machine precision (in particular, if RCOND = 0), the matrix
!>          is singular to working precision.  This condition is
!>          indicated by a return code of INFO > 0.
!>

FERR

!>          FERR is REAL array, dimension (NRHS)
!>          The estimated forward error bound for each solution vector
!>          X(j) (the j-th column of the solution matrix X).
!>          If XTRUE is the true solution corresponding to X(j), FERR(j)
!>          is an estimated upper bound for the magnitude of the largest
!>          element in (X(j) - XTRUE) divided by the magnitude of the
!>          largest element in X(j).  The estimate is as reliable as
!>          the estimate for RCOND, and is almost always a slight
!>          overestimate of the true error.
!>

BERR

!>          BERR is REAL array, dimension (NRHS)
!>          The componentwise relative backward error of each solution
!>          vector X(j) (i.e., the smallest relative change in
!>          any element of A or B that makes X(j) an exact solution).
!>

WORK

!>          WORK is REAL array, dimension (3*N)
!>

IWORK

!>          IWORK is INTEGER array, dimension (N)
!>

INFO

!>          INFO is INTEGER
!>          = 0: successful exit
!>          < 0: if INFO = -i, the i-th argument had an illegal value
!>          > 0: if INFO = i, and i is
!>                <= N:  the leading principal minor of order i of A
!>                       is not positive, so the factorization could not
!>                       be completed, and the solution has not been
!>                       computed. RCOND = 0 is returned.
!>                = N+1: U is nonsingular, but RCOND is less than machine
!>                       precision, meaning that the matrix is singular
!>                       to working precision.  Nevertheless, the
!>                       solution and error bounds are computed because
!>                       there are a number of situations where the
!>                       computed solution can be more accurate than the
!>                       value of RCOND would suggest.
!>
