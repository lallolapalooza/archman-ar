subroutine spftrs (character transr, character uplo, integer n, integer nrhs, real, dimension( 0: * ) a, real, dimension( ldb, * ) b, integer ldb, integer info)

SPFTRS

الغرض:

!>
!> SPFTRS يحل نظام المعادلات الخطية A*X = B لمصفوفة متماثلة موجبة قطعية A
!> باستخدام تحليل Cholesky
!> A = U**T*U أو A = L*L**T الذي تم حسابه بواسطة SPFTRF.
!>

المعلمات

TRANSR 
!>          TRANSR هو CHARACTER*1
!>          = 'N':  يتم تخزين TRANSR العادي للمصفوفة RFP A؛
!>          = 'T':  يتم تخزين TRANSR المنقول للمصفوفة RFP A.
!>

UPLO

!>          UPLO هو CHARACTER*1
!>          = 'U':  يتم تخزين المثلث العلوي للمصفوفة RFP A؛
!>          = 'L':  يتم تخزين المثلث السفلي للمصفوفة RFP A.
!>

N

!>          N هو INTEGER
!>          رتبة المصفوفة A.  N >= 0.
!>

NRHS

!>          NRHS هو INTEGER
!>          عدد الأطراف اليمنى، أي عدد أعمدة المصفوفة B.  NRHS >= 0.
!>

A

!>          A هو مصفوفة REAL، ذات بُعد ( N*(N+1)/2 )
!>          العامل المثلثي U أو L من تحليل Cholesky للمصفوفة RFP A = U**H*U
!>          أو RFP A = L*L**T، كما تم حسابه بواسطة SPFTRF.
!>          انظر الملاحظة أدناه لمزيد من التفاصيل حول RFP A.
!>

B

!>          B هي مصفوفة REAL، ذات بُعد (LDB,NRHS)
!>          عند الإدخال، مصفوفة الطرف الأيمن B.
!>          عند الخروج، مصفوفة الحل X.
!>

LDB

!>          LDB هو INTEGER
!>          البُعد الرئيسي للمصفوفة B.  LDB >= max(1,N).
!>

INFO

!>          INFO هو INTEGER
!>          = 0:  انتهاء ناجح
!>          < 0:  إذا كانت INFO = -i، فإن المعلمة i لها قيمة غير قانونية
!>

تفاصيل إضافية:

!>
!>  ننظر أولاً إلى التنسيق المعبأ الكامل المستطيل (RFP) عندما تكون N
!>  زوجية. نقدم مثالاً حيث N = 6.
!>
!>      AP مثلث علوي             AP مثلث سفلي
!>
!>   00 01 02 03 04 05       00
!>      11 12 13 14 15       10 11
!>         22 23 24 25       20 21 22
!>            33 34 35       30 31 32 33
!>               44 45       40 41 42 43 44
!>                  55       50 51 52 53 54 55
!>
!>
!>  إذا كانت TRANSR = 'N'. تحتفظ RFP بـ AP على النحو التالي:
!>  بالنسبة لـ UPLO = 'U'، فإن شبه المنحرف العلوي A(0:5,0:2) يتكون من الأعمدة الثلاثة الأخيرة
!>  من AP العلوي. يتكون المثلث السفلي A(4:6,0:2) من
!>  منقول الأعمدة الثلاثة الأولى من AP العلوي.
!>  بالنسبة لـ UPLO = 'L'، فإن شبه المنحرف السفلي A(1:6,0:2) يتكون من
!>  الأعمدة الثلاثة الأولى من AP السفلي. يتكون المثلث العلوي A(0:2,0:2) من
!>  منقول الأعمدة الثلاثة الأخيرة من AP السفلي.
!>  هذا يغطي الحالة N زوجية و TRANSR = 'N'.
!>
!>         RFP A                   RFP A
!>
!>        03 04 05                33 43 53
!>        13 14 15                00 44 54
!>        23 24 25                10 11 55
!>        33 34 35                20 21 22
!>        00 44 45                30 31 32
!>        01 11 55                40 41 42
!>        02 12 22                50 51 52
!>
!>
!>  الآن إذا كانت TRANSR = 'T'. RFP A في كلتا حالتي UPLO هي مجرد
!>  منقول RFP A أعلاه. لذلك يحصل المرء على:
!>
!>
!>           RFP A                   RFP A
!>
!>     03 13 23 33 00 01 02    33 00 10 20 30 40 50
!>     04 14 24 34 44 11 12    43 44 11 21 31 41 51
!>     05 15 25 35 45 55 22    53 54 55 22 32 42 52
!>
!>
!>  ثم ننظر إلى التنسيق المعبأ الكامل المستطيل (RFP) عندما تكون N
!>  فردية. نقدم مثالاً حيث N = 5.
!>
!>     AP مثلث علوي             AP مثلث سفلي
!>
!>   00 01 02 03 04              00
!>      11 12 13 14              10 11
!>         22 23 24              20 21 22
!>            33 34              30 31 32 33
!>               44              40 41 42 43 44
!>
!>
!>  إذا كانت TRANSR = 'N'. تحتفظ RFP بـ AP على النحو التالي:
!>  بالنسبة لـ UPLO = 'U'، فإن شبه المنحرف العلوي A(0:4,0:2) يتكون من الأعمدة الثلاثة الأخيرة
!>  من AP العلوي. يتكون المثلث السفلي A(3:4,0:1) من
!>  منقول العمودين الأولين من AP العلوي.
!>  بالنسبة لـ UPLO = 'L'، فإن شبه المنحرف السفلي A(0:4,0:2) يتكون من
!>  الأعمدة الثلاثة الأولى من AP السفلي. يتكون المثلث العلوي A(0:1,1:2) من
!>  منقول العمودين الأخيرين من AP السفلي.
!>  هذا يغطي الحالة N فردية و TRANSR = 'N'.
!>
!>         RFP A                   RFP A
!>
!>        02 03 04                00 33 43
!>        12 13 14                10 11 44
!>        22 23 24                20 21 22
!>        00 33 34                30 31 32
!>        01 11 44                40 41 42
!>
!>  الآن إذا كانت TRANSR = 'T'. RFP A في كلتا حالتي UPLO هي مجرد
!>  منقول RFP A أعلاه. لذلك يحصل المرء على:
!>
!>           RFP A                   RFP A
!>
!>     02 12 22 00 01             00 10 20 30 40 50

!>     04 14 24 34 44             43 44 22 32 42 52
!>
