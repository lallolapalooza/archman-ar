subroutine spbt01 (character uplo, integer n, integer kd, real, dimension( lda, * ) a, integer lda, real, dimension( ldafac, * ) afac, integer ldafac, real, dimension( * ) rwork, real resid)

SPBT01

الغرض:

!>
!> SPBT01 يعيد بناء مصفوفة متماثلة موجبة محددة A من
!> تحليلها L*L' أو U'*U ويحسب الباقي
!>    norm( L*L' - A ) / ( N * norm(A) * EPS ) أو
!>    norm( U'*U - A ) / ( N * norm(A) * EPS ),
!> حيث EPS هي وحدة الآلة، L' هي منقولة مترافقة لـ L، و U' هي منقولة مترافقة لـ U.
!>

المعلمات

UPLO 
!>          UPLO هي CHARACTER*1
!>          تحدد ما إذا كان الجزء العلوي أو السفلي المثلثي للمصفوفة المتماثلة A مخزناً:
!>          = 'U':  مثلث علوي
!>          = 'L':  مثلث سفلي
!>

N

!>          N هو INTEGER
!>          عدد الصفوف والأعمدة للمصفوفة A.  N >= 0.
!>

KD

!>          KD هو INTEGER
!>          عدد الأقطار الفوقية للمصفوفة A إذا كان UPLO = 'U'،
!>          أو عدد الأقطار الفرعية إذا كان UPLO = 'L'.  KD >= 0.
!>

A

!>          A هي REAL array، حجم (LDA,N)
!>          المصفوفة المتماثلة الأصلية A.  إذا كان UPLO = 'U'، يتم تخزين
!>          الجزء المثلثي العلوي لـ A كمصفوفة شريطية؛ إذا كان UPLO = 'L'،
!>          يتم تخزين الجزء المثلثي السفلي لـ A.  يتم تخزين أعمدة المثلث المناسب
!>          في أعمدة A وتخزين أقطار المثلث في صفوف A.  انظر SPBTRF لمزيد
!>          من التفاصيل.
!>

LDA

!>          LDA هو INTEGER.
!>          البعد الرائد للمصفوفة A.  LDA >= max(1,KD+1).
!>

AFAC

!>          AFAC هي REAL array، حجم (LDAFAC,N)
!>          الشكل المحلل للمصفوفة A.  تحتوي AFAC على العامل
!>          L أو U من تحليل L*L' أو U'*U بتنسيق تخزين شريطي،
!>          كما تم حسابه بواسطة SPBTRF.
!>

LDAFAC

!>          LDAFAC هو INTEGER
!>          البعد الرائد للمصفوفة AFAC.
!>          LDAFAC >= max(1,KD+1).
!>

RWORK

!>          RWORK هي REAL array، حجم (N)
!>

RESID

!>          RESID هو REAL
!>          إذا كان UPLO = 'L'، norm(L*L' - A) / ( N * norm(A) * EPS )
!>          إذا كان UPLO = 'U'، norm(U'*U - A) / ( N * norm(A) * EPS )
!>
