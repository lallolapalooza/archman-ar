وثائق الوظيفة/الروتين الفرعي

subroutine spbcon (character uplo, integer n, integer kd, real, dimension( ldab, * ) ab, integer ldab, real anorm, real rcond, real, dimension( * ) work, integer, dimension( * ) iwork, integer info)

SPBCON

الغرض:

!>
!> SPBCON يقدر مقلوب رقم الشرط (بالـ
!> 1-norm) لمصفوفة حقيقية متماثلة موجبة محددة شريطية باستخدام
!> تحليل Cholesky A = U**T*U أو A = L*L**T الذي تم حسابه بواسطة SPBTRF.
!>
!> يتم الحصول على تقدير لـ norm(inv(A))، ويتم حساب مقلوب
!> رقم الشرط على النحو RCOND = 1 / (ANORM * norm(inv(A))).
!>

المعلمات

UPLO 
!>          UPLO is CHARACTER*1
!>          = 'U':  المعامل المثلثي العلوي مخزن في AB؛
!>          = 'L':  المعامل المثلثي السفلي مخزن في AB.
!>

N

!>          N is INTEGER
!>          رتبة المصفوفة A.  N >= 0.
!>

KD

!>          KD is INTEGER
!>          عدد الأقطار الفائقة للمصفوفة A إذا كان UPLO = 'U'،
!>          أو عدد الأقطار الفرعية إذا كان UPLO = 'L'.  KD >= 0.
!>

AB

!>          AB is REAL array, dimension (LDAB,N)
!>          المعامل المثلثي U أو L من تحليل Cholesky
!>          A = U**T*U أو A = L*L**T للمصفوفة الشريطية A، مخزن في
!>          أول KD+1 صفوف من المصفوفة. يتم تخزين العمود j من U أو L
!>          في العمود j من المصفوفة AB على النحو التالي:
!>          إذا كان UPLO ='U'، AB(kd+1+i-j,j) = U(i,j) لـ max(1,j-kd)<=i<=j؛
!>          إذا كان UPLO ='L'، AB(1+i-j,j)    = L(i,j) لـ j<=i<=min(n,j+kd).
!>

LDAB

!>          LDAB is INTEGER
!>          البعد الرائد للمصفوفة AB.  LDAB >= KD+1.
!>

ANORM

!>          ANORM is REAL
!>          الـ 1-norm (أو infinity-norm) للمصفوفة الشريطية المتماثلة A.
!>

RCOND

!>          RCOND is REAL
!>          مقلوب رقم شرط المصفوفة A،
!>          المحسوب على النحو RCOND = 1/(ANORM * AINVNM)، حيث AINVNM هو
!>          تقدير للـ 1-norm للمصفوفة المعكوسة (inv(A)) محسوب في هذا الروتين.
!>

WORK

!>          WORK is REAL array, dimension (3*N)
!>

IWORK

!>          IWORK is INTEGER array, dimension (N)
!>

INFO

!>          INFO is INTEGER
!>          = 0:  انتهاء ناجح
!>          < 0:  إذا كان INFO = -i، فإن المعامل i كان له قيمة غير قانونية
!>
