subroutine spttrs (integer n, integer nrhs, real, dimension( * ) d, real, dimension( * ) e, real, dimension( ldb, * ) b, integer ldb, integer info)

SPTTRS

الغرض:

!>
!> SPTTRS يحل نظام ثلاثي الأقطار من الشكل
!>    A * X = B
!> باستخدام تحليل L*D*L**T للمصفوفة A الذي تم حسابه بواسطة SPTTRF. D هي
!> مصفوفة قطرية محددة في المتجه D، L هي مصفوفة ثنائية القطر موحدة
!> عناصرها تحت القطرية محددة في المتجه E، و X و B
!> هما مصفوفتان N × NRHS.
!>

المعلمات

N 
!>          N هو عدد صحيح
!>          ترتيب المصفوفة ثلاثية الأقطار A. N >= 0.
!>

NRHS

!>          NRHS هو عدد صحيح
!>          عدد الأطراف اليمنى، أي عدد أعمدة
!>          المصفوفة B. NRHS >= 0.
!>

D

!>          D هي مصفوفة حقيقية، البعد (N)
!>          عناصر القطر N للمصفوفة القطرية D من
!>          تحليل L*D*L**T للمصفوفة A.
!>

E

!>          E هي مصفوفة حقيقية، البعد (N-1)
!>          عناصر تحت القطرية (N-1) للعامل ثنائي القطر الموحد
!>          L من تحليل L*D*L**T للمصفوفة A. يمكن اعتبار E أيضًا
!>          كقطرية فائقة للعامل ثنائي القطر الموحد U من
!>          تحليل A = U**T*D*U.
!>

B

!>          B هي مصفوفة حقيقية، البعد (LDB,NRHS)
!>          عند الدخول، متجهات الطرف الأيمن B لنظام
!>          المعادلات الخطية.
!>          عند الخروج، متجهات الحل X.
!>

LDB

!>          LDB هو عدد صحيح
!>          البعد الرائد للمصفوفة B. LDB >= max(1,N).
!>

INFO

!>          INFO هو عدد صحيح
!>          = 0: خروج ناجح
!>          < 0: إذا كان INFO = -k، فإن الوسيط k له قيمة غير قانونية
!>
