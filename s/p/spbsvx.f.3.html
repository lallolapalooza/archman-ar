subroutine spbsvx (character fact, character uplo, integer n, integer kd, integer nrhs, real, dimension( ldab, * ) ab, integer ldab, real, dimension( ldafb, * ) afb, integer ldafb, character equed, real, dimension( * ) s, real, dimension( ldb, * ) b, integer ldb, real, dimension( ldx, * ) x, integer ldx, real rcond, real, dimension( * ) ferr, real, dimension( * ) berr, real, dimension( * ) work, integer, dimension( * ) iwork, integer info)

SPBSVX تحسب حل نظام المعادلات الخطية A * X = B للمصفوفات الأخرى

الغرض:

!>
!> يستخدم SPBSVX تحليل Cholesky A = U**T*U أو A = L*L**T لحساب الحل لنظام المعادلات الخطية الحقيقي
!>    A * X = B,
!> حيث A مصفوفة فرقة متماثلة موجبة محددة ذات حجم N*N، و X و B هما مصفوفتان بحجم N*NRHS.
!>
!> يتم أيضًا توفير حدود الخطأ للحل وتقدير للموضع.
!>

الوصف:

!>
!> يتم تنفيذ الخطوات التالية:
!>
!> 1. إذا كانت FACT = 'E'، يتم حساب عوامل قياس حقيقية لمعادلة النظام:
!>       diag(S) * A * diag(S) * inv(diag(S)) * X = diag(S) * B
!>    يعتمد ما إذا كان سيتم معادلة النظام أم لا على قياس المصفوفة A، ولكن إذا تم استخدام المعادلة،
!>    فإن A يتم استبدالها بـ diag(S)*A*diag(S) و B بـ diag(S)*B.
!>
!> 2. إذا كانت FACT = 'N' أو 'E'، يتم استخدام تحليل Cholesky لتحليل المصفوفة A (بعد المعادلة إذا كانت FACT = 'E') كـ
!>       A = U**T * U,  إذا كان UPLO = 'U'، أو
!>       A = L * L**T,  إذا كان UPLO = 'L'،
!>    حيث U هي مصفوفة فرقة مثلثة عليا، و L هي مصفوفة فرقة مثلثة سفلى.
!>
!> 3. إذا كان الحد الرئيسي الرئيسي من الرتبة i غير موجب،
!>    فإن الروتين الفرعي يعود بـ INFO = i. وإلا، يتم استخدام الشكل المحلل لـ A لتقدير
!>    رقم موضع المصفوفة A. إذا كان مقلوب رقم الموضع أقل من الدقة الآلية،
!>    يتم إرجاع INFO = N+1 كتحذير، ولكن الروتين الفرعي يستمر في حل X وحساب حدود الخطأ كما هو موضح أدناه.
!>
!> 4. يتم حل نظام المعادلات لـ X باستخدام الشكل المحلل لـ A.
!>
!> 5. يتم تطبيق التنقيح التكراري لتحسين الحل المحسوب
!>    وحساب حدود الخطأ وتقديرات الخطأ العكسي له.
!>
!> 6. إذا تم استخدام المعادلة، يتم ضرب المصفوفة X مسبقًا بـ
!>    diag(S) بحيث تحل النظام الأصلي قبل
!>    المعادلة.
!>

المعلمات

FACT 
!>          FACT هو CHARACTER*1
!>          يحدد ما إذا كان الشكل المحلل للمصفوفة A مقدمًا عند الإدخال، وإذا لم يكن كذلك،
!>          ما إذا كان يجب معادلة المصفوفة A قبل تحليلها.
!>          = 'F':  عند الإدخال، يحتوي AFB على الشكل المحلل لـ A.
!>                  إذا كان EQUED = 'Y'، فقد تمت معادلة المصفوفة A
!>                  بعوامل قياس معطاة بواسطة S. لن يتم تعديل AB و AFB.
!>          = 'N':  سيتم نسخ المصفوفة A إلى AFB وتحليلها.
!>          = 'E':  سيتم معادلة المصفوفة A إذا لزم الأمر، ثم
!>                  نسخها إلى AFB وتحليلها.
!>

UPLO

!>          UPLO هو CHARACTER*1
!>          = 'U':  يتم تخزين المثلث العلوي لـ A؛
!>          = 'L':  يتم تخزين المثلث السفلي لـ A.
!>

N

!>          N هو INTEGER
!>          عدد المعادلات الخطية، أي رتبة
!>          المصفوفة A. N >= 0.
!>

KD

!>          KD هو INTEGER
!>          عدد الأقطار الفائقة للمصفوفة A إذا كان UPLO = 'U'،
!>          أو عدد الأقطار الفرعية إذا كان UPLO = 'L'. KD >= 0.
!>

NRHS

!>          NRHS هو INTEGER
!>          عدد الأطراف اليمنى، أي عدد الأعمدة
!>          للمصفوفتين B و X. NRHS >= 0.
!>

AB

!>          AB هو مصفوفة REAL، البعد (LDAB,N)
!>          عند الإدخال، المثلث العلوي أو السفلي للمصفوفة الفرقة المتماثلة
!>          A، مخزن في أول KD+1 صفوف من المصفوفة، باستثناء
!>          إذا كانت FACT = 'F' و EQUED = 'Y'، فيجب أن تحتوي A على
!>          المصفوفة المعادلة diag(S)*A*diag(S). العمود j للمصفوفة A
!>          مخزن في العمود j من المصفوفة AB كالتالي:
!>          إذا كان UPLO = 'U'، AB(KD+1+i-j,j) = A(i,j) لـ max(1,j-KD)<=i<=j؛
!>          إذا كان UPLO = 'L'، AB(1+i-j,j)    = A(i,j) لـ j<=i<=min(N,j+KD).
!>          انظر أدناه لمزيد من التفاصيل.
!>
!>          عند الخروج، إذا كانت FACT = 'E' و EQUED = 'Y'، يتم استبدال A بـ
!>          diag(S)*A*diag(S).
!>

LDAB

!>          LDAB هو INTEGER
!>          البعد الرئيسي للمصفوفة A. LDAB >= KD+1.
!>

AFB

!>          AFB هو مصفوفة REAL، البعد (LDAFB,N)
!>          إذا كانت FACT = 'F'، فإن AFB هي معامل إدخال وتحتوي عند الإدخال
!>          على العامل المثلثي U أو L من تحليل Cholesky
!>          A = U**T*U أو A = L*L**T للمصفوفة الفرقة
!>          A، بنفس تنسيق التخزين مثل A (راجع AB). إذا كان EQUED = 'Y'،
!>          فإن AFB هي الشكل المحلل للمصفوفة المعادلة A.
!>
!>          إذا كانت FACT = 'N'، فإن AFB هي معامل إخراج وتعود عند الخروج
!>          بالعامل المثلثي U أو L من تحليل Cholesky
!>          A = U**T*U أو A = L*L**T.
!>
!>          إذا كانت FACT = 'E'، فإن AFB هي معامل إخراج وتعود عند الخروج
!>          بالعامل المثلثي U أو L من تحليل Cholesky
!>          A = U**T*U أو A = L*L**T للمصفوفة المعادلة A (راجع وصف A لشكل
!>          المصفوفة المعادلة).
!>

LDAFB

!>          LDAFB هو INTEGER
!>          البعد الرئيسي للمصفوفة AFB. LDAFB >= KD+1.
!>

EQUED

!>          EQUED هو CHARACTER*1
!>          يحدد شكل المعادلة التي تم إجراؤها.
!>          = 'N':  لم تتم المعادلة (صحيح دائمًا إذا كانت FACT = 'N').
!>          = 'Y':  تمت المعادلة، أي تم استبدال A بـ
!>                  diag(S) * A * diag(S).
!>          EQUED هو معامل إدخال إذا كانت FACT = 'F'؛ وإلا، فهو معامل إخراج.
!>

S

!>          S هو مصفوفة REAL، البعد (N)
!>          عوامل القياس لـ A؛ لا يتم الوصول إليها إذا كان EQUED = 'N'. S
!>          هو معامل إدخال إذا كانت FACT = 'F'؛ وإلا، S هو معامل إخراج.
!>          إذا كانت FACT = 'F' و EQUED = 'Y'، يجب أن يكون كل عنصر من S
!>          موجبًا.
!>

B

!>          B هو مصفوفة REAL، البعد (LDB,NRHS)
!>          عند الإدخال، مصفوفة الطرف الأيمن B بحجم N*NRHS.
!>          عند الخروج، إذا كان EQUED = 'N'، لم يتم تعديل B؛ إذا كان EQUED = 'Y'،
!>          تم استبدال B بـ diag(S) * B.
!>

LDB

!>          LDB هو INTEGER
!>          البعد الرئيسي للمصفوفة B. LDB >= max(1,N).
!>

X

!>          X هو مصفوفة REAL، البعد (LDX,NRHS)
!>          إذا كان INFO = 0 أو INFO = N+1، فإن مصفوفة الحل X بحجم N*NRHS
!>          للنظام الأصلي للمعادلات. لاحظ أنه إذا كان EQUED = 'Y'،
!>          تم تعديل A و B عند الخروج، والحل للنظام
!>          المعاد هو inv(diag(S))*X.
!&
