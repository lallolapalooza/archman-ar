subroutine spbtrs (character uplo, integer n, integer kd, integer nrhs, real, dimension( ldab, * ) ab, integer ldab, real, dimension( ldb, * ) b, integer ldb, integer info)

SPBTRS

الغرض:

!>
!> SPBTRS يحل نظام المعادلات الخطية A*X = B لمصفوفة نطاق متناظرة موجبة محددة A
!> باستخدام تحليل Cholesky A = U**T*U أو A = L*L**T الذي تم حسابه بواسطة SPBTRF.
!>

المعلمات

UPLO 
!>          UPLO is CHARACTER*1
!>          = 'U':  عامل المثلث العلوي مخزن في AB؛
!>          = 'L':  عامل المثلث السفلي مخزن في AB.
!>

N

!>          N is INTEGER
!>          رتبة المصفوفة A.  N >= 0.
!>

KD

!>          KD is INTEGER
!>          عدد الأقطار الفائقة للمصفوفة A إذا كان UPLO = 'U'،
!>          أو عدد الأقطار الفرعية إذا كان UPLO = 'L'.  KD >= 0.
!>

NRHS

!>          NRHS is INTEGER
!>          عدد الأطراف اليمنى، أي عدد أعمدة
!>          المصفوفة B.  NRHS >= 0.
!>

AB

!>          AB is REAL array, dimension (LDAB,N)
!>          عامل المثلث العلوي U أو السفلي L من تحليل Cholesky
!>          A = U**T*U أو A = L*L**T للمصفوفة النطاقية A، مخزنة في
!>          الصفين KD+1 الأولين من المصفوفة.  العمود j من U أو L
!>          مخزن في العمود j من المصفوفة AB على النحو التالي:
!>          إذا كان UPLO ='U'، فإن AB(kd+1+i-j,j) = U(i,j) لـ max(1,j-kd)<=i<=j؛
!>          إذا كان UPLO ='L'، فإن AB(1+i-j,j)    = L(i,j) لـ j<=i<=min(n,j+kd).
!>

LDAB

!>          LDAB is INTEGER
!>          البعد الرائد للمصفوفة AB.  LDAB >= KD+1.
!>

B

!>          B is REAL array, dimension (LDB,NRHS)
!>          عند الدخول، مصفوفة الطرف الأيمن B.
!>          عند الخروج، مصفوفة الحل X.
!>

LDB

!>          LDB is INTEGER
!>          البعد الرائد للمصفوفة B.  LDB >= max(1,N).
!>

INFO

!>          INFO is INTEGER
!>          = 0:  خروج ناجح
!>          < 0:  إذا كان INFO = -i، فإن المعامل i كان له قيمة غير قانونية
!>
