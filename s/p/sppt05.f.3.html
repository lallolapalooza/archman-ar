subroutine sppt05 (character uplo, integer n, integer nrhs, real, dimension( * ) ap, real, dimension( ldb, * ) b, integer ldb, real, dimension( ldx, * ) x, integer ldx, real, dimension( ldxact, * ) xact, integer ldxact, real, dimension( * ) ferr, real, dimension( * ) berr, real, dimension( * ) reslts)

SPPT05

الغرض:

!>
!> SPPT05 تختبر حدود الخطأ من التنقيح التكراري للحل الذي تم حسابه
!> لنظام المعادلات A*X = B، حيث A مصفوفة متناظرة في صيغة تخزين مجمعة.
!>
!> RESLTS(1) = اختبار حد الخطأ
!>           = norm(X - XACT) / ( norm(X) * FERR )
!>
!> يتم إرجاع قيمة كبيرة إذا لم تكن هذه النسبة أقل من واحد.
!>
!> RESLTS(2) = الخطأ المتبقي من روتين التنقيح التكراري
!>           = الحد الأقصى لـ BERR / ( (n+1)*EPS + (*) )، حيث
!>             (*) = (n+1)*UNFL / (min_i (abs(A)*abs(X) +abs(b))_i )
!>

المعلمات

UPLO 
!>          UPLO is CHARACTER*1
!>          يحدد ما إذا كان الجزء العلوي أو السفلي المثلثي للمصفوفة المتناظرة A
!>          مخزنًا.
!>          = 'U':  مثلثي علوي
!>          = 'L':  مثلثي سفلي
!>

N

!>          N is INTEGER
!>          عدد الصفوف في المصفوفات X، B، و XACT، ورتبة المصفوفة A.  N >= 0.
!>

NRHS

!>          NRHS is INTEGER
!>          عدد الأعمدة في المصفوفات X، B، و XACT.  NRHS >= 0.
!>

AP

!>          AP is REAL array, dimension (N*(N+1)/2)
!>          الجزء العلوي أو السفلي المثلثي للمصفوفة المتناظرة A، مجمع
!>          عموديًا في مصفوفة خطية.  يتم تخزين العمود j من A
!>          في المصفوفة AP على النحو التالي:
!>          إذا كان UPLO = 'U'، AP(i + (j-1)*j/2) = A(i,j) لـ 1<=i<=j؛
!>          إذا كان UPLO = 'L'، AP(i + (j-1)*(2n-j)/2) = A(i,j) لـ j<=i<=n.
!>

B

!>          B is REAL array, dimension (LDB,NRHS)
!>          متجهات الطرف الأيمن لنظام المعادلات الخطية.
!>

LDB

!>          LDB is INTEGER
!>          البعد الرئيسي للمصفوفة B.  LDB >= max(1,N).
!>

X

!>          X is REAL array, dimension (LDX,NRHS)
!>          متجهات الحل المحسوبة.  يتم تخزين كل متجه كعمود في المصفوفة X.
!>

LDX

!>          LDX is INTEGER
!>          البعد الرئيسي للمصفوفة X.  LDX >= max(1,N).
!>

XACT

!>          XACT is REAL array, dimension (LDX,NRHS)
!>          متجهات الحل الدقيقة.  يتم تخزين كل متجه كعمود في المصفوفة XACT.
!>

LDXACT

!>          LDXACT is INTEGER
!>          البعد الرئيسي للمصفوفة XACT.  LDXACT >= max(1,N).
!>

FERR

!>          FERR is REAL array, dimension (NRHS)
!>          تقديرات حدود الخطأ الأمامي لكل متجه حل X.  إذا كان XTRUE هو الحل الحقيقي،
!>          فإن FERR يحد من حجم أكبر مدخل في (X - XTRUE) مقسومًا على حجم أكبر مدخل في X.
!>

BERR

!>          BERR is REAL array, dimension (NRHS)
!>          الخطأ الخلفي النسبي على مستوى المكون لكل متجه حل
!>          (أي، أصغر تغيير نسبي في أي مدخل في A أو B يجعل X حلاً دقيقًا).
!>

RESLTS

!>          RESLTS is REAL array, dimension (2)
!>          الحد الأقصى على متجهات الحل NRHS للنسب التالية:
!>          RESLTS(1) = norm(X - XACT) / ( norm(X) * FERR )
!>          RESLTS(2) = BERR / ( (n+1)*EPS + (*) )
!>
