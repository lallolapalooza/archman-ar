subroutine sptt02 (integer n, integer nrhs, real, dimension( * ) d, real, dimension( * ) e, real, dimension( ldx, * ) x, integer ldx, real, dimension( ldb, * ) b, integer ldb, real resid)

SPTT02

الغرض:

!>
!> SPTT02 تحسب الباقي لحل نظام معادلات ثلاثي الأقطار متماثل:
!>    RESID = norm(B - A*X) / (norm(A) * norm(X) * EPS),
!> حيث EPS هو ثابت الآلة.
!>

المعاملات

N 
!>          N هو عدد صحيح (INTEGER)
!>          رتبة المصفوفة A.
!>

NRHS

!>          NRHS هو عدد صحيح (INTEGER)
!>          عدد الأطراف اليمنى، أي عدد أعمدة المصفوفتين B و X.
!>          NRHS >= 0.
!>

D

!>          D هو مصفوفة حقيقية (REAL array)، أبعادها (N)
!>          عناصر القطر الرئيسية الـ n للمصفوفة ثلاثية الأقطار A.
!>

E

!>          E هو مصفوفة حقيقية (REAL array)، أبعادها (N-1)
!>          العناصر تحت القطر الرئيسي الـ (n-1) للمصفوفة ثلاثية الأقطار A.
!>

X

!>          X هي مصفوفة حقيقية (REAL array)، أبعادها (LDX,NRHS)
!>          مصفوفة الحلول X ذات الأبعاد n × nrhs.
!>

LDX

!>          LDX هو عدد صحيح (INTEGER)
!>          البعد الرئيسي للمصفوفة X.  LDX >= max(1,N).
!>

B

!>          B هي مصفوفة حقيقية (REAL array)، أبعادها (LDB,NRHS)
!>          عند الدخول: مصفوفة الأطراف اليمنى B ذات الأبعاد n × nrhs.
!>          عند الخروج: يتم استبدال B بالفرق B - A*X.
!>

LDB

!>          LDB هو عدد صحيح (INTEGER)
!>          البعد الرئيسي للمصفوفة B.  LDB >= max(1,N).
!>

RESID

!>          RESID هو عدد حقيقي (REAL)
!>          norm(B - A*X) / (norm(A) * norm(X) * EPS)
!>
