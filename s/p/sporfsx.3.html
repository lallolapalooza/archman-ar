SPORFSX

الغرض:

!>
!>    SPORFSX يقوم بتحسين الحل المحسوب لنظام المعادلات الخطية
!>    عندما تكون مصفوفة المعاملات متماثلة موجبة
!>    محددة، ويوفر حدود الخطأ وتقديرات الخطأ الخلفي
!>    للحل. بالإضافة إلى حد الخطأ المعياري، يقوم الكود
!>    بتوفير حد خطأ مكوني أقصى إذا كان ذلك ممكنًا. انظر
!>    التعليقات الخاصة بـ ERR_BNDS_NORM و ERR_BNDS_COMP للحصول على
!>    تفاصيل حول حدود الخطأ.
!>
!>    قد تم موازنة نظام المعادلات الخطية الأصلي
!>    قبل استدعاء هذا الإجراء، كما هو موضح بواسطة المعاملات EQUED و S
!>    أدناه. في هذه الحالة، الحل وحدود الخطأ التي تم إرجاعها
!>    هي للنظام الأصلي غير المتوازن.
!> 
!>     يتم تجميع بعض المعاملات الاختيارية في مصفوفة PARAMS. هذه
!>     الإعدادات تحدد كيفية أداء التنقيح، ولكن غالبًا ما تكون
!>     الافتراضات مقبولة. إذا كانت الافتراضات مقبولة، يمكن للمستخدمين
!>     تمرير NPARAMS = 0 مما يمنع الكود المصدري من الوصول
!>     إلى المعامل PARAMS.
!>

المعاملات

UPLO 
!>          UPLO هو CHARACTER*1
!>       = 'U':  المثلث العلوي من A مخزن؛
!>       = 'L':  المثلث السفلي من A مخزن.
!>

EQUED

!>          EQUED هو CHARACTER*1
!>     يحدد شكل الموازنة التي تم إجراؤها على A
!>     قبل استدعاء هذا الإجراء. هذا ضروري لحساب
!>     الحل وحدود الخطأ بشكل صحيح.
!>       = 'N':  لا يوجد موازنة
!>       = 'Y':  موازنة الصفوف والأعمدة، أي تم استبدال A
!>               بـ diag(S) * A * diag(S).
!>               تم تغيير الطرف الأيمن B وفقًا لذلك.
!>

N

!>          N هو INTEGER
!>     ترتيب المصفوفة A.  N >= 0.
!>

NRHS

!>          NRHS هو INTEGER
!>     عدد الأطراف اليمنى، أي عدد أعمدة
!>     المصفوفتين B و X.  NRHS >= 0.
!>

A

!>          A هي مصفوفة REAL، البعد (LDA,N)
!>     المصفوفة المتماثلة A.  إذا كان UPLO = 'U'، فإن الجزء
!>     العلوي المثلثي الأيسر الذي حجمه N-by-N من A يحتوي على الجزء
!>     المثلثي العلوي من المصفوفة A، ولا تتم الإشارة إلى الجزء
!>     المثلثي السفلي الصارم من A.  إذا كان UPLO = 'L'، فإن الجزء
!>     السفلي المثلثي الأيسر الذي حجمه N-by-N من A يحتوي على الجزء
!>     المثلثي السفلي من المصفوفة A، ولا تتم الإشارة إلى الجزء
!>     المثلثي العلوي الصارم من A.
!>

LDA

!>          LDA هو INTEGER
!>     البعد الرئيسي للمصفوفة A.  LDA >= max(1,N).
!>

AF

!>          AF هي مصفوفة REAL، البعد (LDAF,N)
!>     العامل المثلثي U أو L من تحليل Cholesky
!>     A = U**T*U أو A = L*L**T، كما تم حسابه بواسطة SPOTRF.
!>

LDAF

!>          LDAF هو INTEGER
!>     البعد الرئيسي للمصفوفة AF.  LDAF >= max(1,N).
!>

S

!>          S هي مصفوفة REAL، البعد (N)
!>     عوامل القياس للمصفوفة A.  إذا كان EQUED = 'Y'، يتم ضرب A
!>     من اليسار واليمين بواسطة diag(S).  S هو معامل إدخال إذا كان FACT =
!>     'F'؛ بخلاف ذلك، S هو معامل إخراج.  إذا كان FACT = 'F' و EQUED
!>     = 'Y'، يجب أن يكون كل عنصر من S موجبًا.  إذا كان S إخراجًا، فإن كل عنصر
!>     من S هو قوة للجذر.  إذا كان S إدخالًا، فيجب أن يكون كل عنصر
!>     من S قوة للجذر لضمان حل موثوق
!>     وتقديرات خطأ.  القياس بقوى الجذر لا يسبب
!>     أخطاء تقريب إلا إذا كانت النتيجة تحت التدفق أو تجاوزته.
!>     تؤدي أخطاء التقريب أثناء القياس إلى تنقيح بمصفوفة
!>     غير مكافئة للمصفوفة المدخلة، مما ينتج تقديرات خطأ
!>     قد لا تكون موثوقة.
!>

B

!>          B هي مصفوفة REAL، البعد (LDB,NRHS)
!>     مصفوفة الطرف الأيمن B.
!>

LDB

!>          LDB هو INTEGER
!>     البعد الرئيسي للمصفوفة B.  LDB >= max(1,N).
!>

X

!>          X هي مصفوفة REAL، البعد (LDX,NRHS)
!>     عند الإدخال، مصفوفة الحل X، كما تم حسابها بواسطة SGETRS.
!>     عند الخروج، مصفوفة الحل المحسنة X.
!>

LDX

!>          LDX هو INTEGER
!>     البعد الرئيسي للمصفوفة X.  LDX >= max(1,N).
!>

RCOND

!>          RCOND هو REAL
!>     المقلوب المقياس لرقم الشرط. هذا تقدير لـ
!>     مقلوب رقم شرط Skeel للمصفوفة A بعد
!>     الموازنة (إذا تم إجراؤها).  إذا كان هذا أقل من دقة الآلة
!>     (على وجه الخصوص، إذا كان صفرًا)، فإن المصفوفة شاذة
!>     لدقة العمل.  لاحظ أن الخطأ قد لا يزال صغيرًا حتى
!>     إذا كان هذا الرقم صغيرًا جدًا وتبدو المصفوفة مشروطة بشكل سيء.
!>

BERR

!>          BERR هي مصفوفة REAL، البعد (NRHS)
!>     الخطأ الخلفي النسبي المكوني. هذا هو
!>     الخطأ الخلفي النسبي المكوني لكل متجه حل X(j)
!>     (أي، أصغر تغيير نسبي في أي عنصر من A أو B يجعل
!>     X(j) حلًا دقيقًا).
!>

N_ERR_BNDS

!>          N_ERR_BNDS هو INTEGER
!>     عدد حدود الخطأ لإرجاعها لكل طرف أيمن
!>     ولكل نوع (معياري أو مكوني).  انظر ERR_BNDS_NORM و
!>     ERR_BNDS_COMP أدناه.
!>

ERR_BNDS_NORM

!>          ERR_BNDS_NORM هي مصفوفة REAL، البعد (NRHS, N_ERR_BNDS)
!>     لكل طرف أيمن، تحتوي هذه المصفوفة على معلومات حول
!>     حدود الخطأ المختلفة وأرقام الشرط المقابلة لـ
!>     الخطأ النسبي المعياري، والذي يتم تعريفه على النحو التالي:
!>
!>     الخطأ النسبي المعياري في متجه الحل i:
!>             max_j (abs(XTRUE(j,i) - X(j,i)))
!>            ------------------------------
!>                  max_j abs(X(j,i))
!>
!>     يتم فهرسة المصفوفة حسب نوع معلومات الخطأ كما هو موضح
!>     أدناه. حاليًا هناك ما يصل إلى ثلاثة أجزاء من المعلومات
!>     التي يتم إرجاعها.
!>
!>     الفهرس الأول في ERR_BNDS_NORM(i,:) يتوافق مع الطرف الأيمن
!>     i.
!>
!>     الفهرس الثاني في ERR_BNDS_NORM(:,err) يحتوي على المجالات
!>     الثلاثة التالية:
!>     err = 1  قيمة منطقية. ثق بالحل إذا كان
!>              مقلوب رقم الشرط أقل من الحد
!>              sqrt(n) * slamch('Epsilon').
!>
!>     err = 2  حد الخطأ: الخطأ الأمامي المقدر،
!>              بالتأكيد ضمن عامل 10 من الخطأ الحقيقي
!>              طالما أن الإدخال التالي أكبر من الحد
!>              sqrt(n) * slamch('Epsilon'). Thi
