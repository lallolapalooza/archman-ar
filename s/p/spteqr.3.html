توثيق الوظيفة/الروتين الفرعي

subroutine spteqr (character compz, integer n, real, dimension( * ) d, real, dimension( * ) e, real, dimension( ldz, * ) z, integer ldz, real, dimension( * ) work, integer info)

SPTEQR

الغرض:

!>
!> SPTEQR يحسب جميع القيم الذاتية، واختيارياً، المتجهات الذاتية لمصفوفة
!> ثلاثية القطر متماثلة موجبة محددة عن طريق تحليل المصفوفة أولاً باستخدام SPTTRF،
!> ثم استدعاء SBDSQR لحساب القيم المفردة للعامل ثنائي القطر.
!>
!> يقوم هذا الروتين بحساب القيم الذاتية لمصفوفة القطر الثلاثي الموجبة المحددة
!> بدقة نسبية عالية. هذا يعني أنه إذا تراوحت القيم الذاتية عبر العديد من
!> أوامر الحجم في الحجم، فسيتم حساب القيم الذاتية الصغيرة والمتجهات الذاتية
!> المقابلة بدقة أكبر من، على سبيل المثال، باستخدام طريقة QR القياسية.
!>
!> يمكن أيضاً العثور على المتجهات الذاتية لمصفوفة ثلاثية القطر متماثلة كاملة أو
!> نطاقية محددة موجبة إذا تم استخدام SSYTRD أو SSPTRD أو SSBTRD
!> لتقليل هذه المصفوفة إلى شكل ثلاثي القطر. (ومع ذلك، فإن تقليل إلى شكل ثلاثي القطر
!> قد يستبعد إمكانية الحصول على دقة نسبية عالية في القيم الذاتية الصغيرة للمصفوفة الأصلية،
!> إذا تراوحت هذه القيم الذاتية عبر العديد من أوامر الحجم.)
!>

المعلمات

COMPZ 
!>          COMPZ is CHARACTER*1
!>          = 'N':  حساب القيم الذاتية فقط.
!>          = 'V':  حساب المتجهات الذاتية للمصفوفة المتماثلة الأصلية
!>                  أيضاً. المصفوفة Z تحتوي على المصفوفة المتعامدة
!>                  المستخدمة لتقليل المصفوفة الأصلية إلى
!>                  شكل ثلاثي القطر.
!>          = 'I':  حساب المتجهات الذاتية للمصفوفة ثلاثية القطر
!>                  أيضاً.
!>

N

!>          N is INTEGER
!>          رتبة المصفوفة. N >= 0.
!>

D

!>          D is REAL array, dimension (N)
!>          عند الإدخال، عناصر القطر n للمصفوفة ثلاثية القطر.
!>          عند الخروج العادي، تحتوي D على القيم الذاتية، بترتيب تنازلي.
!>

E

!>          E is REAL array, dimension (N-1)
!>          عند الإدخال، عناصر القطر الفرعي (n-1) للمصفوفة ثلاثية القطر.
!>          عند الخروج، تم تدمير E.
!>

Z

!>          Z is REAL array, dimension (LDZ, N)
!>          عند الإدخال، إذا كان COMPZ = 'V'، فإن المصفوفة المتعامدة المستخدمة في
!>          التقليل إلى شكل ثلاثي القطر.
!>          عند الخروج، إذا كان COMPZ = 'V'، فإن المتجهات الذاتية المتعامدة الطبيعية
!>          للمصفوفة المتماثلة الأصلية؛
!>          إذا كان COMPZ = 'I'، فإن المتجهات الذاتية المتعامدة الطبيعية
!>          للمصفوفة ثلاثية القطر.
!>          إذا كان INFO > 0 عند الخروج، فإن Z تحتوي على المتجهات الذاتية المرتبطة
!>          بالقيم الذاتية المخزنة فقط.
!>          إذا كان COMPZ = 'N'، فلا يتم الرجوع إلى Z.
!>

LDZ

!>          LDZ is INTEGER
!>          البعد الرئيسي للمصفوفة Z. LDZ >= 1، وإذا
!>          كان COMPZ = 'V' أو 'I'، فإن LDZ >= max(1,N).
!>

WORK

!>          WORK is REAL array, dimension (4*N)
!>

INFO

!>          INFO is INTEGER
!>          = 0:  خروج ناجح.
!>          < 0:  إذا كان INFO = -i، فإن الوسيط i كان له قيمة غير قانونية.
!>          > 0:  إذا كان INFO = i، وكان i هو:
!>                <= N  فشل تحليل Cholesky للمصفوفة
!>                      لأن العامل الرئيسي الرئيسي ذو الترتيب i لم يكن موجباً.
!>                > N   فشل خوارزمية SVD في التقارب؛
!>                      إذا كان INFO = N+i، فإن i عنصر قطر فرعي من
!>                      العامل ثنائي القطر لم يتقارب إلى الصفر.
!>
