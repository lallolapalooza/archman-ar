subroutine sptrfs (integer n, integer nrhs, real, dimension( * ) d, real, dimension( * ) e, real, dimension( * ) df, real, dimension( * ) ef, real, dimension( ldb, * ) b, integer ldb, real, dimension( ldx, * ) x, integer ldx, real, dimension( * ) ferr, real, dimension( * ) berr, real, dimension( * ) work, integer info)

SPTRFS

الغرض:

!>
!> SPTRFS يحسّن الحل المحسوب لنظام المعادلات الخطية
!> عندما تكون المصفوفة المعاملة متماثلة موجبة محددة
!> وثلاثية الأقطار، ويوفر حدودًا للأخطاء وتقديرات للأخطاء اللاحقة
!> للحل.
!>

المعلمات

N 
!>          N هو عدد صحيح (INTEGER)
!>          رتبة المصفوفة A. N >= 0.
!>

NRHS

!>          NRHS هو عدد صحيح (INTEGER)
!>          عدد الأطراف اليمنى، أي عدد الأعمدة
!>          للمصفوفة B. NRHS >= 0.
!>

D

!>          D هي مصفوفة حقيقية (REAL array)، البعد (N)
!>          العناصر القطرية n للمصفوفة ثلاثية الأقطار A.
!>

E

!>          E هي مصفوفة حقيقية (REAL array)، البعد (N-1)
!>          العناصر تحت القطرية (n-1) للمصفوفة ثلاثية الأقطار A.
!>

DF

!>          DF هي مصفوفة حقيقية (REAL array)، البعد (N)
!>          العناصر القطرية n للمصفوفة القطرية D من
!>          التحليل الذي تم حسابه بواسطة SPTTRF.
!>

EF

!>          EF هي مصفوفة حقيقية (REAL array)، البعد (N-1)
!>          العناصر تحت القطرية (n-1) للعامل ثنائي الأقطار الوحيد
!>          L من التحليل الذي تم حسابه بواسطة SPTTRF.
!>

B

!>          B هي مصفوفة حقيقية (REAL array)، البعد (LDB,NRHS)
!>          مصفوفة الطرف الأيمن B.
!>

LDB

!>          LDB هو عدد صحيح (INTEGER)
!>          البعد الرئيسي للمصفوفة B. LDB >= max(1,N).
!>

X

!>          X هي مصفوفة حقيقية (REAL array)، البعد (LDX,NRHS)
!>          عند الإدخال، مصفوفة الحل X، كما تم حسابها بواسطة SPTTRS.
!>          عند الخروج، مصفوفة الحل المحسّنة X.
!>

LDX

!>          LDX هو عدد صحيح (INTEGER)
!>          البعد الرئيسي للمصفوفة X. LDX >= max(1,N).
!>

FERR

!>          FERR هي مصفوفة حقيقية (REAL array)، البعد (NRHS)
!>          حد الخطأ الأمامي لكل متجه حل
!>          X(j) (العمود j للمصفوفة الحل X).
!>          إذا كان XTRUE هو الحل الحقيقي المقابل لـ X(j)، فإن FERR(j)
!>          هو تقدير للحد العلوي لمقدار العنصر الأكبر
!>          في (X(j) - XTRUE) مقسومًا على مقدار العنصر الأكبر
!>          في X(j).
!>

BERR

!>          BERR هي مصفوفة حقيقية (REAL array)، البعد (NRHS)
!>          الخطأ النسبي للمكونات لكل متجه حل
!>          X(j) (أي، أصغر تغيير نسبي في
!>          أي عنصر من A أو B يجعل X(j) حلاً دقيقًا).
!>

WORK

!>          WORK هي مصفوفة حقيقية (REAL array)، البعد (2*N)
!>

INFO

!>          INFO هو عدد صحيح (INTEGER)
!>          = 0: الخروج الناجح
!>          < 0: إذا كان INFO = -i، فإن المعامل i كان له قيمة غير قانونية
!>
