الملخص

package MyApp::Type::Library;
use parent 'Specio::Exporter';
use Specio::Declare;
use Specio::Library::Builtins;
declare(
    'Foo',
    parent => t('Str'),
    where  => sub { $_[0] =~ /foo/i },
);
declare(
    'ArrayRefOfInt',
    parent => t( 'ArrayRef', of => t('Int') ),
);
my $even = anon(
    parent => t('Int'),
    inline => sub {
        my $type      = shift;
        my $value_var = shift;
        return $value_var . ' % 2 == 0';
    },
);
coerce(
    t('ArrayRef'),
    from  => t('Foo'),
    using => sub { [ $_[0] ] },
);
coerce(
    $even,
    from  => t('Int'),
    using => sub { $_[0] % 2 ? $_[0] + 1 : $_[0] },
);
# اسم النوع هو DateTime
any_isa_type('DateTime');
# اسم النوع هو DateTimeObject
object_isa_type( 'DateTimeObject', class => 'DateTime' );
any_can_type(

    methods => [ 'duck_walk', 'quack' ],
);
object_can_type(
    'DuckObject',
    methods => [ 'duck_walk', 'quack' ],
);
enum(
    'Colors',

);

);

الوصف

يقوم هذا الحزمة بتصدير مجموعة من الأدوات المساعدة لإعلان الأنواع. استيراد هذه الحزمة ينشئ أيضًا روتينية فرعية "t" في الحزمة المسماة.

الروتينيات الفرعية

تُصدِّر هذه الوحدة الروتينيات الفرعية التالية.

t('الاسم')

تسمح لك هذه الروتينية الفرعية بالوصول إلى أي نوع قمت بتعريفه من قبل، وكذلك أي نوع تم استيراده من مكتبة أنواع أخرى.

إذا تم تمرير اسم غير معروف، فسوف تطرح استثناءً.

declare(...)

تُعرّف هذه الروتينية الفرعية نوعًا مُسمى. الوسيط الأول هو اسم النوع، يليه مجموعة من أزواج المفتاح/القيمة:

  • parent => $type

    يجب أن يكون الوالد كائن نوع آخر. على وجه التحديد، يمكن أن يكون أي شيء يقوم بدور Specio::Constraint::Role::Interface. يمكن أن يكون الوالد نوعًا مُسمى أو مجهولاً.

  • where => sub { ... }

    هذه روتينية فرعية تحدد قيد النوع. سيتم تمرير وسيط واحد لها، وهي القيمة المراد التحقق منها، ويجب أن تُرجع صحيحًا أو خاطئًا للإشارة إلى ما إذا كانت القيمة صالحة للنوع.

    هذا الوسيط متبادل حصريًا مع الوسيط "inline".

  • inline => sub { ... }

    هذه روتينية فرعية تُستدعى لإنشاء كود مضمن للتحقق من النوع. يمكن أن يكون الترميز المضمن أسرع بكثير من مجرد توفير روتينية فرعية مع الوسيط "where"، ولكنه قد يكون أكثر تعقيدًا للحصول عليه بشكل صحيح.

    يتم استدعاء توليد الكود المضمن كطريقة على النوع مع وسيط واحد. هذا الوسيط هو سلسلة نصية تحتوي على اسم المتغير المراد استخدامه في الكود المُنشأ. بشكل عام، هذا شيء مثل '$_[0]' أو '$value'.

    يجب أن يُرجع توليد الكود المضمن سلسلة نصية من الكود تمثل حدًا واحدًا، ويجب * ألا * ينتهي بفاصلة. هذا يسمح بتضمين الكود المضمن بشكل آمن في عبارة "if"، على سبيل المثال. يمكنك استخدام كتل "do { }" والرموز الثلاثية لتضمين كل شيء في حد واحد. لا تقم بتعيين المتغير الذي تختبره. يجب أن يُقيّم هذا الحد الواحد إلى صحيح أو خاطئ.

    يجب أن يتضمن توليد الكود المضمن كودًا لتنفيذ كل من النوع الحالي وجميع آبائه. أبسط طريقة للقيام بذلك هي كتابة روتينية فرعية مثل:

    sub {
    my $self = shift;
    my $var  = shift;
    return $self->parent->inline_check($var)
        . ' المزيد من الكود الذي يتحقق من $var';
}

    أو، بشكل أكثر إيجازًا:

    sub { $_[0]->parent->inline_check( $_[1] ) . ' المزيد من الكود الذي يتحقق من $_[1]' }

    الوسيط "inline" متبادل حصريًا مع الوسيط "where".

  • message_generator => sub { ... }

    روتينية فرعية تُنشئ رسالة خطأ عند فشل فحص النوع. الرسالة الافتراضية تقول شيئًا مثل "فشل التحقق من النوع المسمى Int في مكتبة الأنواع Specio::Library::Builtins (.../Specio/blib/lib/Specio/Library/Builtins.pm) في السطر 147 في الروتينية الفرعية (eval) بالقيمة 1.1".

    يمكنك تجاوز هذه الرسالة لتقديم شيء أكثر تحديدًا حول سبب فشل النوع.

    تُستدعى الروتينية الفرعية التي تقدمها كطريقة على النوع مع وسيطين. الأول هو وصف النوع (الجزء في الرسالة أعلاه الذي يبدأ بـ "type named Int ..." وينتهي بـ "... in subroutine (eval)". يخبر هذا الوصف ما هو النوع وأين تم تعريفه.

    الوسيط الثاني هو القيمة التي فشلت في فحص النوع، بعد أي تحويلات تم تطبيقها.

anon(...)

تُعرّف هذه الروتينية الفرعية نوعًا مجهولًا. وهي متطابقة مع "declare" باستثناء أنها تتوقع قائمة من أزواج المفتاح/القيمة بدون اسم نوع كوسيط أول.

coerce(...)

تُعرّف هذه الروتينية الفرعية تحويلًا من نوع إلى آخر. يجب أن يكون الوسيط الأول كائنًا يقوم بدور Specio::Constraint::Role::Interface. يمكن أن يكون هذا نوعًا مُسمى أو مجهولًا. هذا هو النوع الذي يتم التحويل إليه.

الوسائط المتبقية هي أزواج المفتاح/القيمة:

  • from => $type

    يجب أن يكون هذا كائنًا يقوم بدور Specio::Constraint::Role::Interface. هذا هو النوع الذي يتم التحويل *منه*. مرة أخرى، يمكن أن يكون هذا نوعًا مُسمى أو مجهولًا.

  • using => sub { ... }

    هذه روتينية فرعية تحدد تحويل النوع. سيتم تمرير وسيط واحد لها، وهي القيمة المراد تحويلها. يجب أن تُرجع قيمة جديدة للنوع الذي يتم التحويل إليه.

    هذا الوسيط متبادل حصريًا مع الوسيط "inline".

  • inline => sub { ... }

    هذه روتينية فرعية تُستدعى لإنشاء كود مضمن لأداء التحويل.

    يتم استدعاء توليد الكود المضمن كطريقة على النوع مع وسيط واحد. هذا الوسيط هو سلسلة نصية تحتوي على اسم المتغير المراد استخدامه في الكود المُنشأ. بشكل عام، هذا شيء مثل '$_[0]' أو '$value'.

    يجب أن يُرجع توليد الكود المضمن سلسلة نصية من الكود تمثل حدًا واحدًا، ويجب * ألا * ينتهي بفاصلة. هذا يسمح بتضمين الكود المضمن بشكل آمن في عبارة "if"، على سبيل المثال. يمكنك استخدام كتل "do { }" والرموز الثلاثية لتضمين كل شيء في حد واحد. يجب أن يُقيّم هذا الحد الواحد إلى القيمة الجديدة.

أدوات مساعدة للإعلان

تُصدِّر هذه الحزمة أيضًا بعض الأدوات المساعدة للإعلان عن أنواع معينة:

any_isa_type, object_isa_type

تُنشئ الأداة المساعدة "any_isa_type" نوعًا يقبل اسم فئة أو كائنًا من الفئة المعطاة. تُنشئ الأداة المساعدة "object_isa_type" نوعًا يقبل فقط كائنًا من الفئة المعطاة.
