spcat(1)

الحزمة: opensp

التراخيص: BSD

spcat(1) مرجع spcat(1)

الاسم

spcat - طباعة كتالوج SGML

الملخص

spcat [-b الاسم] [--bctf=الاسم] [-C] [--catalogs] [-c sysid] [--catalog=sysid] [-f] [--error-file=الملف] [-D الدليل] [--directory=الدليل] [-R] [--restricted] [-v] [-h] [--help] [-P الحرف] [--public-id=الحرف] [-S الحرف] [--system_id=الحرف] [-p الاسم] [--parameter-entity=الاسم] [-d الاسم] [--doctype=الاسم] [-l الاسم] [--linktype=الاسم] [-e الاسم] [--entity=الاسم] [-n الاسم] [--notation=الاسم] [-s الاسم] [--declaration=الاسم]

الوصف

ospcat يطبع المعرفات النظامية الفعالة الموجودة في الكتالوجات إلى المخرج القياسي. لا يتحقق من أن المعرفات النظامية تتوافق مع ملفات فعلية. قيمة الإرجاع هي 0 إذا تم إنشاء المعرف النظامي الأخير بنجاح و 1 بخلاف ذلك.

الخيارات

الخيارات التالية متاحة:

-b الاسم, --bctf=الاسم

استخدم اسم bctf الاسم للمخرج.

-f الملف, --error-file=الملف

أضف رسائل الخطأ إلى الملف الملف.

-v, --version

عرض إصدار البرنامج.

-h, --help

عرض نص المساعدة هذا.

-c sysid, --catalog=sysid

استخدم كتالوج sysid.

-C, --catalogs

الوسائط هي الكتالوجات، استخدم إدخال DOCUMENT.

-D الدليل, --directory=الدليل

ابحث عن الملفات في الدليل الدليل.

-R, --restricted

تقييد قراءة الملفات (للاستخدام في تطبيقات الويب).

-P الحرف, --public-id=الحرف

اطبع المعرف النظامي الفعال للمعرف العام الحرف.

-S الحرف, --system-id=الحرف

اطبع المعرف النظامي الفعال للمعرفة النظامية الحرف.

-p الاسم, --parameter-entity=الاسم

اطبع المعرف النظامي الفعال لمعرف المعلمة الاسم.

-d الاسم, --doctype=الاسم

اطبع المعرف النظامي الفعال للنوع المستند الاسم.

-l الاسم, --linktype=الاسم

اطبع المعرف النظامي الفعال لنوع الارتباط الاسم.

-e الاسم, --entity=الاسم

اطبع المعرف النظامي الفعال للكيان الاسم.

-n الاسم, --notation=الاسم

اطبع المعرف النظامي الفعال للترميز الاسم.

-s الاسم, --declaration=الاسم

اطبع المعرف النظامي الفعال لإعلان SGML للنوع المستند الاسم.
