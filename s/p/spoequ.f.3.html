subroutine spoequ (integer n, real, dimension( lda, * ) a, integer lda, real, dimension( * ) s, real scond, real amax, integer info)

SPOEQU

الغرض:

!>
!> SPOEQU يحسب عوامل القياس للصف والأعمدة المقصودة لموازنة مصفوفة متناظرة موجبة محددة A وتقليل عدد شرطها
!> (فيما يتعلق بمعيار الثنين). S يحتوي على عوامل القياس،
!> S(i) = 1/sqrt(A(i,i))، يتم اختيارها بحيث تكون المصفوفة المتوازنة B بعناصر
!> B(i,j) = S(i)*A(i,j)*S(j) لها واحدات على القطر. هذا الاختيار لـ S يضع عدد شرط B
!> ضمن عامل N للعدد الشرطي الأصغر الممكن على جميع عمليات القياس القطرية الممكنة.
!>

المعلمات

N 
!>          N هو صحيح (INTEGER)
!>          رتبة المصفوفة A.  N >= 0.
!>

A

!>          A هي مصفوفة حقيقية (REAL)، بأبعاد (LDA,N)
!>          المصفوفة المتناظرة الموجبة المحددة N×N التي يتم حساب عوامل القياس الخاصة بها.
!>          يتم فقط الرجوع إلى العناصر القطرية للمصفوفة A.
!>

LDA

!>          LDA هو صحيح (INTEGER)
!>          البعد الرائد للمصفوفة A.  LDA >= max(1,N).
!>

S

!>          S هي مصفوفة حقيقية (REAL)، بأبعاد (N)
!>          إذا كان INFO = 0، فإن S يحتوي على عوامل القياس للمصفوفة A.
!>

SCOND

!>          SCOND هي قيمة حقيقية (REAL)
!>          إذا كان INFO = 0، فإن S يحتوي على نسبة أصغر S(i) إلى أكبر S(i).
!>          إذا كان SCOND >= 0.1 ولم يكن AMAX كبيرًا جدًا أو صغيرًا جدًا، فلن يكون من المفيد القياس بـ S.
!>

AMAX

!>          AMAX هي قيمة حقيقية (REAL)
!>          القيمة المطلقة لأكبر عنصر في المصفوفة. إذا كان AMAX قريبًا جدًا من الفيضان (overflow)
!>          أو قريبًا جدًا من التدفق الناقص (underflow)، فيجب قياس المصفوفة.
!>

INFO

!>          INFO هو صحيح (INTEGER)
!>          = 0:  خروج ناجح
!>          < 0:  إذا كان INFO = -i، فإن المعلمة i كانت قيمة غير قانونية
!>          > 0:  إذا كان INFO = i، فإن العنصر القطري i غير موجب.
!>
