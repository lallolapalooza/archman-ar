SPLAY_ENTRY(3bsd)

الحزمة: libbsd

التراخيص: custom

NAME

RB_HEAD, RB_INITIALIZER, RB_INSERT, RB_REMOVE, RB_FIND, RB_NFIND, RB_ROOT, RB_MIN, RB_MAX, RB_NEXT, RB_PREV, RB_FOREACH, RB_FOREACH_SAFE, RB_EMPTY

LIBRARY

libbsd

SYNOPSIS

#include <sys/tree.h>

DESCRIPTION

تُستخدم وحدات الماكرو RB_HEAD و RB_INITIALIZER و RB_INSERT و RB_REMOVE و RB_FIND و RB_NFIND و RB_ROOT و RB_MIN و RB_MAX و RB_NEXT و RB_PREV و RB_FOREACH و RB_FOREACH_SAFE و RB_EMPTY للتعامل مع أشجار SPLAY وأشجار الأحمر والأسود في C. الهدف هو الحفاظ على بنية الملف.

SPLAY TREES

تُستخدم وحدات الماكرو RB_HEAD و RB_INITIALIZER و RB_INSERT و RB_REMOVE و RB_FIND و RB_NFIND و RB_ROOT و RB_MIN و RB_MAX و RB_NEXT و RB_PREV و RB_FOREACH و RB_FOREACH_SAFE و RB_EMPTY للتعامل مع أشجار SPLAY وأشجار الأحمر والأسود في C. الهدف هو الحفاظ على بنية الملف.

RED-BLACK TREES

تُستخدم وحدات الماكرو RB_HEAD و RB_INITIALIZER و RB_INSERT و RB_REMOVE و RB_FIND و RB_NFIND و RB_ROOT و RB_MIN و RB_MAX و RB_NEXT و RB_PREV و RB_FOREACH و RB_FOREACH_SAFE و RB_EMPTY للتعامل مع أشجار SPLAY وأشجار الأحمر والأسود في C. الهدف هو الحفاظ على بنية الملف.

EXAMPLES

يوضح المثال التالي كيفية تعريف شجرة SPLAY تحتوي على أعداد صحيحة. يتم إدراج القيم فيها وطباعتها بترتيب. أخيرًا، يتم طباعة البنية الداخلية للشجرة.

#include <sys/tree.h>
#include <err.h>
#include <stdio.h>
#include <stdlib.h>

struct node {
	RB_ENTRY(node) entry;
	int i;
};

int intcmp(struct node *, struct node *);
void print_tree(struct node *);

int
intcmp(struct node *e1, struct node *e2)
{
	return (e1->i < e2->i ? -1 : e1->i > e2->i);
}

RB_HEAD(inttree, node) head = RB_INITIALIZER(&head);
RB_PROTOTYPE(inttree, node, entry, intcmp)
RB_GENERATE(inttree, node, entry, intcmp)

int testdata[] = {
	20, 16, 17, 13, 3, 6, 1, 8, 2, 4, 10, 19, 5, 9, 12, 15, 18,
	7, 11, 14
};

void
print_tree(struct node *n)
{
	struct node *left, *right;

	if (n == NULL) {
		printf("nil");
		return;
	}
	left = RB_LEFT(n, entry);
	right = RB_RIGHT(n, entry);
	if (left == NULL && right == NULL)
		printf("%d", n->i);
	else {
		printf("%d(", n->i);
		print_tree(left);
		printf(",");
		print_tree(right);
		printf(")");
	}
}

int
main(void)
{
	int i;
	struct node *n;

	for (i = 0; i < sizeof(testdata) / sizeof(testdata[0]); i++) {
		if ((n = malloc(sizeof(struct node))) == NULL)
			err(1, NULL);
		n->i = testdata[i];
		RB_INSERT(inttree, &head, n);
	}

	RB_FOREACH(n, inttree, &head) {
		printf("%d\n", n->i);
	}
	print_tree(RB_ROOT(&head));
	printf("\n");
	return (0);
}

HISTORY

ظهرت وحدات الماكرو لأشجار SPLAY لأول مرة في FreeBSD 4.6.

NOTES

محاولة تحرير شجرة في الوضع التالي هو خطأ شائع:

SPLAY_FOREACH(var, NAME, &head) {
	SPLAY_REMOVE(NAME, &head, var);
	free(var);
}
free(head);

نظرًا لأن var يتم تحريرها، فإن وحدة الماكرو FOREACH() تشير إلى مؤشر ربما تم إعادة تخصيصه بالفعل. يتطلب الكود الصحيح متغيرًا ثانيًا.

for (var = SPLAY_MIN(NAME, &head); var != NULL; var = nxt) {
	nxt = SPLAY_NEXT(NAME, &head, var);
	SPLAY_REMOVE(NAME, &head, var);
	free(var);
}
