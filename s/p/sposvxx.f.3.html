الإجراء الفرعي sposvxx (character fact, character uplo, integer n, integer nrhs, real, dimension( lda, * ) a, integer lda, real, dimension( ldaf, * ) af, integer ldaf, character equed, real, dimension( * ) s, real, dimension( ldb, * ) b, integer ldb, real, dimension( ldx, * ) x, integer ldx, real rcond, real rpvgrw, real, dimension( * ) berr, integer n_err_bnds, real, dimension( nrhs, * ) err_bnds_norm, real, dimension( nrhs, * ) err_bnds_comp, integer nparams, real, dimension( * ) params, real, dimension( * ) work, integer, dimension( * ) iwork, integer info)

يحسب SPOSVXX حل نظام المعادلات الخطية A * X = B لمصفوفات PO

الغرض:

!>
!>    يستخدم SPOSVXX تحليل تشوليسكي A = U**T*U أو A = L*L**T
!>    لحساب حل نظام حقيقي من المعادلات الخطية
!>    A * X = B، حيث A مصفوفة متماثلة موجبة محددة من الرتبة N-by-N
!>    و X و B مصفوفتان من الرتبة N-by-NRHS.
!>
!>    عند الطلب، يتم إرجاع حدود الخطأ المعيارية والحد الأقصى للمكونات
!>    على حد سواء. سيرجع SPOSVXX حلاً بخطأ صغير
!>    مضمون (O(eps) حيث eps هي دقة الآلة العاملة)
!>    ما لم تكن المصفوفة سيئة التكييف جداً، وفي هذه الحالة
!>    يتم إرجاع تحذير. كما يتم حساب وإرجاع أرقام
!>    التكييف ذات الصلة.
!>
!>    يقبل SPOSVXX تحليلات وعوامل موازنة مقدمة من المستخدم؛
!>    انظر تعريفات خيارات FACT و EQUED.
!>    الحل مع التحسين واستخدام تحليل من استدعاء SPOSVXX سابق
!>    سينتج أيضاً حلاً بأخطاء O(eps) أو تحذيرات،
!>    لكن لا يمكننا تقديم هذا الادعاء للتحليلات وعوامل الموازنة
!>    العامة المقدمة من المستخدم إذا كانت تختلف عما
!>    ينتجه SPOSVXX نفسه.
!>

الوصف:

!>
!>    يتم تنفيذ الخطوات التالية:
!>
!>    1. إذا كان FACT = 'E'، يتم حساب عوامل تحجيم حقيقية لموازنة
!>    النظام:
!>
!>      diag(S)*A*diag(S)     *inv(diag(S))*X = diag(S)*B
!>
!>    يعتمد ما إذا كان النظام سيتم موازنته أم لا على
!>    تحجيم المصفوفة A، ولكن إذا تم استخدام الموازنة، يتم
!>    استبدال A بـ diag(S)*A*diag(S) و B بـ diag(S)*B.
!>
!>    2. إذا كان FACT = 'N' أو 'E'، يتم استخدام تحليل تشوليسكي
!>    لتحليل المصفوفة A (بعد الموازنة إذا كان FACT = 'E') كـ
!>       A = U**T* U،  إذا كان UPLO = 'U'، أو
!>       A = L * L**T،  إذا كان UPLO = 'L'،
!>    حيث U مصفوفة مثلثية عليا و L مصفوفة مثلثية
!>    سفلى.
!>
!>    3. إذا لم يكن القاصر الرئيسي الرئيسي من الرتبة i موجباً،
!>    فإن الإجراء يرجع مع INFO = i. وإلا، يتم استخدام
!>    الشكل المحلل لـ A لتقدير رقم تكييف المصفوفة
!>    A (انظر الوسيط RCOND).  إذا كان مقلوب رقم التكييف
!>    أقل من دقة الآلة، فإن الإجراء يستمر في حل
!>    X وحساب حدود الخطأ كما هو موضح أدناه.
!>
!>    4. يتم حل نظام المعادلات لـ X باستخدام الشكل المحلل
!>    لـ A.
!>
!>    5. افتراضياً (ما لم يتم تعيين PARAMS(LA_LINRX_ITREF_I) إلى صفر)،
!>    سيستخدم الإجراء التحسين التكراري لمحاولة الحصول على خطأ
!>    صغير وحدود خطأ.  يحسب التحسين الباقي على
!>    الأقل ضعف دقة العمل.
!>
!>    6. إذا تم استخدام الموازنة، يتم ضرب المصفوفة X من اليسار بـ
!>    diag(S) بحيث تحل النظام الأصلي قبل
!>    الموازنة.
!> 
!>     يتم تجميع بعض المعاملات الاختيارية في مصفوفة PARAMS.  تحدد
!>     هذه الإعدادات كيفية تنفيذ التحسين، ولكن غالباً ما تكون
!>     القيم الافتراضية مقبولة.  إذا كانت القيم الافتراضية مقبولة، يمكن للمستخدمين
!>     تمرير NPARAMS = 0 مما يمنع الكود المصدري من الوصول
!>     إلى وسيط PARAMS.
!>

المعاملات

FACT 
!>          FACT هو CHARACTER*1
!>     يحدد ما إذا كان الشكل المحلل للمصفوفة A
!>     مقدم عند الإدخال، وإذا لم يكن كذلك، ما إذا كان يجب موازنة المصفوفة A
!>     قبل تحليلها.
!>       = 'F':  عند الإدخال، يحتوي AF على الشكل المحلل لـ A.
!>               إذا كان EQUED ليس 'N'، فقد تمت موازنة
!>               المصفوفة A بعوامل تحجيم معطاة بواسطة S.
!>               لا يتم تعديل A و AF.
!>       = 'N':  سيتم نسخ المصفوفة A إلى AF وتحليلها.
!>       = 'E':  سيتم موازنة المصفوفة A إذا لزم الأمر، ثم
!>               نسخها إلى AF وتحليلها.
!>

UPLO

!>          UPLO هو CHARACTER*1
!>       = 'U':  يتم تخزين المثلث العلوي من A؛
!>       = 'L':  يتم تخزين المثلث السفلي من A.
!>

N

!>          N هو INTEGER
!>     عدد المعادلات الخطية، أي رتبة
!>     المصفوفة A.  N >= 0.
!>

NRHS

!>          NRHS هو INTEGER
!>     عدد الأطراف اليمنى، أي عدد أعمدة
!>     المصفوفتين B و X.  NRHS >= 0.
!>

A

!>          A هو مصفوفة REAL، البعد (LDA,N)
!>     عند الإدخال، المصفوفة المتماثلة A، باستثناء إذا كان FACT = 'F' و EQUED =
!>     'Y'، فيجب أن تحتوي A على المصفوفة الموازنة
!>     diag(S)*A*diag(S).  إذا كان UPLO = 'U'، فإن الجزء العلوي
!>     المثلثي الرئيسي N-by-N من A يحتوي على الجزء العلوي المثلثي من
!>     المصفوفة A، والجزء السفلي المثلثي الصارم من A غير
!>     مشار إليه.  إذا كان UPLO = 'L'، فإن الجزء السفلي المثلثي
!>     الرئيسي N-by-N من A يحتوي على الجزء السفلي المثلثي من المصفوفة A، و
!>     الجزء العلوي المثلثي الصارم من A غير مشار إليه.  لا يتم
!>     تعديل A إذا كان FACT = 'F' أو 'N'، أو إذا كان FACT = 'E' و EQUED =
!>     'N' عند الخروج.
!>
!>     عند الخروج، إذا كان FACT = 'E' و EQUED = 'Y'، يتم استبدال A بـ
!>     diag(S)*A*diag(S).
!>

LDA

!>          LDA هو INTEGER
!>     البعد الرئيسي للمصفوفة A.  LDA >= max(1,N).
!>

AF

!>          AF هو مصفوفة REAL، البعد (LDAF,N)
!>     إذا كان FACT = 'F'، فإن AF وسيط إدخال وعند الإدخال
!>     يحتوي على العامل المثلثي U أو L من تحليل
!>     تشوليسكي A = U**T*U أو A = L*L**T، بنفس تنسيق
!>     التخزين مثل A.  إذا كان EQUED .ne. 'N'، فإن AF هو الشكل
!>     المحلل للمصفوفة الموازنة diag(S)*A*diag(S).
!>
!>     إذا كان FACT = 'N'، فإن AF وسيط إخراج وعند الخروج
!>     يرجع العامل المثلثي U أو L من تحليل تشوليسكي
!>     A = U**T*U أو A = L*L**T للمصفوفة
!>     الأصلية A.
!>
!>     إذا كان FACT = 'E'، فإن AF وسيط إخراج وعند الخروج
!>     يرجع العامل المثلثي U أو L من تحليل تشوليسكي
!>     A = U**T*U أو A = L*L**T للمصفوفة الموازنة
!>     A (انظر وصف A لشكل المصفوفة
!>     الموازنة).
!>

LDAF

!>          LDAF هو INTEGER
!>     البعد الرئيسي للمصفوفة AF.  LDAF >= max(1,N).
!>

EQUED

!>          EQUED هو CHARACTER*1
!>     يحدد شكل الموازنة التي تم إجراؤها.
!>       = 'N':  لا موازنة (صحيح دائماً إذا كان FACT = 'N').
!>       = 'Y':  كل من الصف و coموازنة الأعمدة، أي أنه تم استبدال A بـ
!>               diag(S) * A * diag(S).
!>     EQUED هي وسيطة إدخال إذا كان FACT = 'F'؛ وإلا فهي وسيطة
!>     إخراج.
!>

S

!>          S هي مصفوفة REAL، البُعد (N)
!>     عوامل قياس الصفوف للمصفوفة A. إذا كان EQUED = 'Y'، تُضرب A من
!>     اليسار واليمين بـ diag(S). S هي وسيطة إدخال إذا كان FACT =
!>     'F'؛ وإلا فإن S هي وسيطة إخراج. إذا كان FACT = 'F' و EQUED
!>     = 'Y'، يجب أن يكون كل عنصر من S موجبًا. إذا كانت S إخراجًا، فإن كل
!>     عنصر من S هو قوة للأساس. إذا كانت S إدخالًا، يجب أن يكون كل عنصر
!>     من S قوة للأساس لضمان حل موثوق
!>     وتقديرات أخطاء موثوقة. القياس بقوى الأساس لا يسبب
!>     أخطاء تقريب إلا إذا كانت النتيجة تحتفيضًا أو فوقفيضًا.
!>     أخطاء التقريب أثناء القياس تؤدي إلى التحسين بمصفوفة
!>     غير مكافئة للمصفوفة المدخلة، مما ينتج تقديرات أخطاء
!>     قد لا تكون موثوقة.
!>

B

!>          B هي مصفوفة REAL، البُعد (LDB,NRHS)
!>     عند الإدخال، مصفوفة الطرف الأيمن B ذات البُعد N-by-NRHS.
!>     عند الإخراج،
!>     إذا كان EQUED = 'N'، لا تُعدَّل B؛
!>     إذا كان EQUED = 'Y'، تُكتب فوق B بـ diag(S)*B؛
!>

LDB

!>          LDB هو INTEGER
!>     البُعد القائد للمصفوفة B.  LDB >= max(1,N).
!>

X

!>          X هي مصفوفة REAL، البُعد (LDX,NRHS)
!>     إذا كان INFO = 0، مصفوفة الحل X ذات البُعد N-by-NRHS للنظام
!>     الأصلي من المعادلات. لاحظ أن A و B تُعدَّلان عند الإخراج إذا
!>     كان EQUED .ne. 'N'، والحل للنظام الموازَن هو
!>     inv(diag(S))*X.
!>

LDX

!>          LDX هو INTEGER
!>     البُعد القائد للمصفوفة X.  LDX >= max(1,N).
!>

RCOND

!>          RCOND هو REAL
!> مقلوب رقم الشرط المُقاس. هذا تقدير لمقلوب
!> رقم شرط Skeel للمصفوفة A بعد
!> الموازنة (إذا تمت). إذا كان هذا أقل من دقة
!> الآلة (على وجه الخصوص، إذا كان صفرًا)، فإن المصفوفة شاذة
!> بالنسبة لدقة العمل. لاحظ أن الخطأ قد يظل صغيرًا حتى
!> إذا كان هذا الرقم صغيرًا جدًا وبدت المصفوفة سيئة
!> الشرط.
!>

RPVGRW

!>          RPVGRW هو REAL
!> مقلوب نمو المحور. عند الإخراج، يحتوي هذا على مقلوب
!> عامل نمو المحور norm(A)/norm(U). يُستخدم
!> norm. إذا كان هذا أقل بكثير من 1، فإن استقرار
!> تحليل LU للمصفوفة (الموازَنة) A قد يكون ضعيفًا.
!> هذا يعني أيضًا أن الحل X، وأرقام الشرط المقدَّرة،
!> وحدود الخطأ قد تكون غير موثوقة. إذا فشل التحليل مع
!>     0<INFO<=N، فإن هذا يحتوي على مقلوب عامل نمو المحور
!> لأعمدة INFO الأولى من A.
!>

BERR

!>          BERR هي مصفوفة REAL، البُعد (NRHS)
!> الخطأ الخلفي النسبي المركّب. هذا هو
!> الخطأ الخلفي النسبي المركّب لكل متجه حل X(j)
!> (أي أصغر تغيير نسبي في أي عنصر من A أو B يجعل
!> X(j) حلًا دقيقًا).
!>

N_ERR_BNDS

!>          N_ERR_BNDS هو INTEGER
!> عدد حدود الخطأ التي تُعاد لكل طرف أيمن
!> ولكل نوع (معياري أو مركّب). انظر ERR_BNDS_NORM و
!> ERR_BNDS_COMP أدناه.
!>

ERR_BNDS_NORM

!>          ERR_BNDS_NORM هي مصفوفة REAL، البُعد (NRHS, N_ERR_BNDS)
!> لكل طرف أيمن، تحتوي هذه المصفوفة على معلومات حول
!> حدود خطأ وأرقام شرط متنوعة تقابل
!> الخطأ النسبي المعياري، المعرَّف كما يلي:
!>
!>     الخطأ النسبي المعياري في متجه الحل i:
!>             max_j (abs(XTRUE(j,i) - X(j,i)))
!>            ------------------------------
!>                  max_j abs(X(j,i))
!>
!>     تُفهرَس المصفوفة حسب نوع معلومات الخطأ كما هو موصوف
!>     أدناه. يوجد حاليًا ما يصل إلى ثلاث قطع من المعلومات
!>     تُعاد.
!>
!>     الفهرس الأول في ERR_BNDS_NORM(i,:) يقابل الطرف الأيمن
!>     i.
!>
!>     الفهرس الثاني في ERR_BNDS_NORM(:,err) يحتوي على الحقول
!>     الثلاثة التالية:
!>     err = 1  منطقي. ثق بالإجابة إذا كان
!>              مقلوب رقم الشرط أقل من العتبة
!>              sqrt(n) * slamch('Epsilon').
!>
!>     err = 2  حد الخطأ: الخطأ الأمامي المقدَّر،
!>              على الأرجح ضمن عامل 10 من الخطأ الحقيقي
!>              طالما أن الإدخال التالي أكبر من العتبة
!>              sqrt(n) * slamch('Epsilon'). يجب الوثوق بحد الخطأ هذا فقط
!>              إذا كان المنطقي السابق صحيحًا.
!>
!>     err = 3  مقلوب رقم الشرط: مقلوب رقم الشرط
!>              المعياري المقدَّر. يُقارَن مع العتبة
!>              sqrt(n) * slamch('Epsilon') لتحديد ما إذا كان تقدير
!>              الخطأ . أرقام الشرط المقلوبة
!>              هذه هي 1 / (norm(Z^{-1},inf) * norm(Z,inf)) لبعض
!>              المصفوفات Z المقيسة بشكل مناسب.
!>              لنفترض Z = S*A، حيث تقيس S كل صف بقوة من
!>              الأساس بحيث تكون جميع مجاميع الصفوف المطلقة لـ Z تقريبًا 1.
!>
!>     انظر Lapack Working Note 165 لمزيد من التفاصيل والتحذيرات
!>     الإضافية.
!>

ERR_BNDS_COMP

!>          ERR_BNDS_COMP هي مصفوفة REAL، البُعد (NRHS, N_ERR_BNDS)
!> لكل طرف أيمن، تحتوي هذه المصفوفة على معلومات حول
!> حدود خطأ وأرقام شرط متنوعة تقابل
!> الخطأ النسبي المركّب، المعرَّف كما يلي:
!>
!>     الخطأ النسبي المركّب في متجه الحل i:
!>                    abs(XTRUE(j,i) - X(j,i))
!>             max_j ----------------------
!>                         abs(X(j,i))
!>
!>     تُفهرَس المصفوفة حسب الطرف الأيمن i (الذي يعتمد عليه
!>     الخطأ النسبي المركّب)، ونوع معلومات الخطأ
!>     كما هو موصوف أدناه. يوجد حاليًا ما يصل إلى ثلاث
!>     قطع من المعلومات تُعاد لكل طرف أيمن. إذا
!>     لم تُطلب الدقة المركّبة (PARAMS(3) = 0.0)، فإن
!>     ERR_BNDS_COMP لا يُوصَل إليها. إذا كان N_ERR_BNDS < 3، فإنه يُعاد
!>     على الأكثر الإدخالات الأولى (:,N_ERR_BNDS).
!>
!>     الفهرس الأول في ERR_BNDS_COMP(i,:) يقابل الطرف الأيمن
!>     i.
!>
!>     الفهرس الثاني في ERR_BNDS_COMP(:,err) يحتوي على الحقول
!>     الثلاثة التالية:
!>     err = 1  منطقي. ثق بالإجابة إذا كان
!>              مقلوب رقم الشرط أقل من العتبة
!>              sqrt(n) * slamch('Epsilon').
!>
!>     err = 2  حد الخطأ: الخطأ الأمامي المقدَّر،
!>              على الأرجح ضمن عامل 10 من الخطأ الحقيقي
!>              طالما أن الإدخال التالي أكبر من العتبة
!>              sqrt(n) * slamch('Epsilon'). يجب الوثوق بحد الخطأ هذا فقط
!>              إذا كان المنطقي السابق صحيحًا.
!>
!>     err = 3  مقلوب رقم الشرط: مقلوب رقم الشرط
!>              المركّب المقدَّر. يُقارَن مع العتبة
!>              sqrt(n) * slamch('Epsilon') لتحديد ما إذا كان تقدير
!>              الخطأ . أرقام الشرط المقلوبة
!>              هذه هي 1 / (norm(Z^{-1},inf) * norm(Z,inf)) لبعض
!>              المصفوفات Z المقيسة بشكل مناسب.
!>              لنفترض Z = S*(A*diag(x))، حيث x هو الحل للطرف
!>              الأيمن الحالي و S تقيس كل صف من
!>              A*diag(x) بقوة من الأساس بحيث تكون جميع مجاميع الصفوف المطلقة
!>              لـ Z تقريبًا 1.
!>
!>     انظر Lapack Working Note 165 لمزيد من التفاصيل والتحذيرات
!>     الإضافية.
!>

NPARAMS

!>          NPARAMS هو INTEGER
!>     يحدد عدد المعاملات المضبوطة في PARAMS. إذا كان <= 0، فإن
!>     مصفوفة PARAMS لا يُشار إليها أبدًا وتُستخدم القيم الافتراضية.
!>

PARAMS

!>          PARAMS هي مصفوفة REAL، البُعد NPARAMS
!>     تحدد معاملات الخوارزمية. إذا كان إدخال < 0.0، فإن
!>     ذلك الإدخال سيُملأ بالقيمة الافتراضية المستخدمة لذلك
!>     المعامل. لا يُوصَل إلا إلى المواضع حتى NPARAMS؛ تُستخدم
!>     القيم الافتراضية للمعاملات ذات الأرقام الأعلى.
!>
!>       PARAMS(LA_LINRX_ITREF_I = 1) : ما إذا كان سيُجرى تحسين
!>            تكراري أم لا.
!>         الافتراضي: 1.0
!>            = 0.0:  لا يُجرى تحسين، ولا تُحسب حدود
!>                    خطأ.
!>            = 1.0:  استخدم خوارزمية التحسين ذات الدقة المزدوجة،
!>                    ربما مع حسابات أحادية مزدوجة إذا كانت
!>                    بيئة الترجمة لا تدعم الدقة
!>                    المزدوجة.
!>              (القيم الأخرى محجوزة للاستخدام المستقبلي)
!>
!>       PARAMS(LA_LINRX_ITHRESH_I = 2) : الحد الأقصى لعدد حسابات
!>            البواقي المسموح بها للتحسين.
!>         الافتراضي: 10
!>         عدواني: اضبطه على 100 للسماح بالتقارب باستخدام تحليلات
!>                     تقريبية أو تحليلات غير LU. إذا
!>                     استخدم التحليل تقنية غير
!>                     الحذف الغاوسي، فإن الضمانات في
!>                     err_bnds_norm و err_bnds_comp قد لا تعود
!>                     موثوقة.
!>
!>       PARAMS(LA_LINRX_CWISE_I = 3) : علامة تحدد ما إذا كان الكود
!>            سيحاول إيجاد حل بخطأ نسبي مركّب
!>            صغير في خوارزمية الدقة المزدوجة. الموجب
!>            صحيح، 0.0 خطأ.
!>         الافتراضي: 1.0 (محاولة التقارب المركّب)
!>

WORK

!>          WORK هي مصفوفة REAL، البُعد (4*N)
!>

IWORK

!>          IWORK هي مصفوفة INTEGER، البُعد (N)
!>

INFO

!>          INFO هو INTEGER
!>       = 0:  خروج ناجح. الحل لكل طرف أيمن
!>         مضمون.
!>       < 0:  إذا كان INFO = -i، فإن الوسيطة i لها قيمة غير قانونية
!>       > 0 و <= N:  U(INFO,INFO) صفر تمامًا. اكتمل
!>         التحليل، لكن العامل U شاذ تمامًا، لذا
!>         لم يمكن حساب الحل وحدود الخطأ. يُعاد RCOND = 0
!>         .
!>       = N+J: الحل المقابل للطرف الأيمن J
!>         غير مضمون. الحلول المقابلة لأطراف أيمن
!>         أخرى K حيث K > J قد لا تكون مضمونة أيضًا، لكن
!>         يُبلَّغ فقط عن أول طرف أيمن من هذا القبيل. إذا لم يُطلب
!>         خطأ مركّب صغير (PARAMS(3) = 0.0) فإن
!>         الطرف الأيمن J هو الأول بحد خطأ
!>         معياري غير مضمون (أصغر J بحيث
!>         ERR_BNDS_NORM(J,1) = 0.0). افتراضيًا (PARAMS(3) = 1.0)
!>         الطرف الأيمن J هو الأول بحد خطأ معياري أو
!>         مركّب غير مضمون (أصغر
!>         J بحيث إما ERR_BNDS_NORM(J,1) = 0.0 أو
!>ERR_BNDS_COMP(J,1) = 0.0). راجع تعريف
!>         ERR_BNDS_NORM(:,1) و ERR_BNDS_COMP(:,1). للحصول على معلومات
!>         حول جميع الأطراف اليمنى، تحقق من ERR_BNDS_NORM أو
!>         ERR_BNDS_COMP.
!>
