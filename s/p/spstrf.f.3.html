subroutine spstrf (character uplo, integer n, real, dimension( lda, * ) a, integer lda, integer, dimension( n ) piv, integer rank, real tol, real, dimension( 2*n ) work, integer info)

SPSTRF تحسب مفكوك Cholesky مع تحويل كامل لمصفوفة حقيقية متناظرة شبه موجبة.

الغرض:

!>
!> SPSTRF تحسب مفكوك Cholesky مع تحويل كامل
!> لمصفوفة حقيقية متناظرة شبه موجبة A.
!>
!> المفكوك له الصيغة
!>    P**T * A * P = U**T * U ,  إذا كان UPLO = 'U',
!>    P**T * A * P = L  * L**T,  إذا كان UPLO = 'L',
!> حيث U مصفوفة مثلثية علوية و L مصفوفة مثلثية سفلية، و
!> P مخزنة كمتجه PIV.
!>
!> هذه الخوارزمية لا تحاول التحقق من أن A شبه موجبة.
!> هذا الإصدار من الخوارزمية يستدعي BLAS المستوى 3.
!>

المعلمات

UPLO 
!>          UPLO هو CHARACTER*1
!>          يحدد ما إذا كان الجزء المثلثي العلوي أو السفلي من المصفوفة
!>          المتناظرة A مخزن.
!>          = 'U':  مثلثي علوي
!>          = 'L':  مثلثي سفلي
!>

N

!>          N هو INTEGER
!>          رتبة المصفوفة A.  N >= 0.
!>

A

!>          A هو مصفوفة REAL، بعد (LDA,N)
!>          عند الإدخال، المصفوفة المتناظرة A.  إذا كان UPLO = 'U'، فإن الجزء
!>          المثلثي العلوي n x n العلوي من A يحتوي على الجزء المثلثي
!>          العلوي للمصفوفة A، والجزء المثلثي السفلي الصارم لـ A لا تتم
!>          الإشارة إليه.  إذا كان UPLO = 'L'، فإن الجزء المثلثي
!>          السفلي n x n السفلي من A يحتوي على الجزء المثلثي السفلي
!>          للمصفوفة A، والجزء المثلثي العلوي الصارم لـ A لا تتم الإشارة إليه.
!>
!>          عند الخروج، إذا كان INFO = 0، فإن العامل U أو L من مفكوك Cholesky
!>          المذكور أعلاه.
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرائد للمصفوفة A.  LDA >= max(1,N).
!>

PIV

!>          PIV هو مصفوفة INTEGER، بعد (N)
!>          PIV بحيث تكون الإدخالات غير الصفرية هي P( PIV(K), K ) = 1.
!>

RANK

!>          RANK هو INTEGER
!>          رتبة A المعطاة بواسطة عدد الخطوات التي أكملتها الخوارزمية.
!>

TOL

!>          TOL هو REAL
!>          الحد الأدنى المحدد من قبل المستخدم. إذا كان TOL < 0، فسيتم استخدام
!>          N*U*MAX( A(K,K) ). تنتهي الخوارزمية عند الخطوة (K-1) إذا كان
!>          المحور (pivot) <= TOL.
!>

WORK

!>          WORK هو مصفوفة REAL، بعد (2*N)
!>          مساحة العمل.
!>

INFO

!>          INFO هو INTEGER
!>          < 0: إذا كان INFO = -K، فإن المعلمة K كان لها قيمة غير قانونية.
!>          = 0: اكتملت الخوارزمية بنجاح.
!>          > 0: المصفوفة A إما ناقصة الرتبة مع رتبة محسوبة كما هو
!>               مُرجع في RANK، أو ليست شبه موجبة. انظر القسم 7 من
!>               مذكرة العمل #161 من LAPACK لمزيد من المعلومات.
!>
