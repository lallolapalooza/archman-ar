subroutine spbt05 (character uplo, integer n, integer kd, integer nrhs, real, dimension( ldab, * ) ab, integer ldab, real, dimension( ldb, * ) b, integer ldb, real, dimension( ldx, * ) x, integer ldx, real, dimension( ldxact, * ) xact, integer ldxact, real, dimension( * ) ferr, real, dimension( * ) berr, real, dimension( * ) reslts)

SPBT05

الغرض:

!>
!> يقوم SPBT05 باختبار حدود الخطأ للحل المحسوب لنظام المعادلات A*X = B، حيث A هي
!> مصفوفة فرقة متناظرة.
!>
!> RESLTS(1) = اختبار حد الخطأ
!>           = norm(X - XACT) / ( norm(X) * FERR )
!>
!> يتم إرجاع قيمة كبيرة إذا لم تكن هذه النسبة أقل من واحد.
!>
!> RESLTS(2) = الباقي من روتين التحسين التكراري
!>           = الحد الأقصى لـ BERR / ( NZ*EPS + (*) )، حيث
!>             (*) = NZ*UNFL / (min_i (abs(A)*abs(X) +abs(b))_i )
!>             و NZ = الحد الأقصى لعدد العناصر غير الصفرية في أي صف من A، بالإضافة إلى 1
!>

المعلمات

UPLO 
!>          UPLO هو CHARACTER*1
!>          يحدد ما إذا كان الجزء العلوي أو السفلي المثلثي من المصفوفة المتناظرة A
!>          مخزنًا.
!>          = 'U':  المثلث العلوي
!>          = 'L':  المثلث السفلي
!>

N

!>          N هو INTEGER
!>          عدد صفوف المصفوفات X، B، و XACT، ورتبة المصفوفة A.  N >= 0.
!>

KD

!>          KD هو INTEGER
!>          عدد الأقطار الفائقة للمصفوفة A إذا كان UPLO = 'U'،
!>          أو عدد الأقطار الفرعية إذا كان UPLO = 'L'.  KD >= 0.
!>

NRHS

!>          NRHS هو INTEGER
!>          عدد أعمدة المصفوفات X، B، و XACT.
!>          NRHS >= 0.
!>

AB

!>          AB هو مصفوفة REAL، البعد (LDAB,N)
!>          الجزء العلوي أو السفلي المثلثي من المصفوفة الفرقة المتناظرة A،
!>          مخزن في أول KD+1 صفوف من المصفوفة.  يتم تخزين العمود j من A
!>          في العمود j من المصفوفة AB كما يلي:
!>          إذا كان UPLO = 'U'، فإن AB(kd+1+i-j,j) = A(i,j) لـ max(1,j-kd)<=i<=j؛
!>          إذا كان UPLO = 'L'، فإن AB(1+i-j,j)    = A(i,j) لـ j<=i<=min(n,j+kd).
!>

LDAB

!>          LDAB هو INTEGER
!>          البعد الرائد للمصفوفة AB.  LDAB >= KD+1.
!>

B

!>          B هي مصفوفة REAL، البعد (LDB,NRHS)
!>          متجهات الجانب الأيمن لنظام المعادلات الخطية.
!>

LDB

!>          LDB هو INTEGER
!>          البعد الرائد للمصفوفة B.  LDB >= max(1,N).
!>

X

!>          X هي مصفوفة REAL، البعد (LDX,NRHS)
!>          متجهات الحل المحسوبة.  يتم تخزين كل متجه كعمود في المصفوفة X.
!>

LDX

!>          LDX هو INTEGER
!>          البعد الرائد للمصفوفة X.  LDX >= max(1,N).
!>

XACT

!>          XACT هي مصفوفة REAL، البعد (LDX,NRHS)
!>          متجهات الحل الدقيقة.  يتم تخزين كل متجه كعمود في المصفوفة XACT.
!>

LDXACT

!>          LDXACT هو INTEGER
!>          البعد الرائد للمصفوفة XACT.  LDXACT >= max(1,N).
!>

FERR

!>          FERR هي مصفوفة REAL، البعد (NRHS)
!>          التقديرات لحدود الخطأ الأمامي لكل متجه حل X.  إذا كان XTRUE هو الحل الحقيقي،
!>          فإن FERR يحد من مقدار أكبر إدخال في (X - XTRUE) مقسومًا على مقدار
!>          أكبر إدخال في X.
!>

BERR

!>          BERR هي مصفوفة REAL، البعد (NRHS)
!>          الخطأ الخلفي النسبي المكون لكل متجه حل (أي، أصغر تغيير نسبي في أي إدخال
!>          في A أو B يجعل X حلاً دقيقًا).
!>

RESLTS

!>          RESLTS هي مصفوفة REAL، البعد (2)
!>          الحد الأقصى عبر متجهات الحل NRHS للنسب:
!>          RESLTS(1) = norm(X - XACT) / ( norm(X) * FERR )
!>          RESLTS(2) = BERR / ( NZ*EPS + (*) )
!>
