subroutine sptcon (integer n, real, dimension( * ) d, real, dimension( * ) e, real anorm, real rcond, real, dimension( * ) work, integer info)

SPTCON

الغرض:

!>
!> SPTCON يحسب مقلوب رقم الشرط (في المعيار 1) لمصفوفة مثلثة متماثلة حقيقية موجبة محددة
!> باستخدام التحلل A = L*D*L**T أو A = U**T*D*U المحسوب بواسطة SPTTRF.
!>
!> يتم حساب Norm(inv(A)) بطريقة مباشرة، ويتم حساب مقلوب
!> رقم الشرط على النحو التالي:
!>              RCOND = 1 / (ANORM * norm(inv(A))).
!>

المعلمات

N 
!>          N هو عدد صحيح (INTEGER)
!>          ترتيب المصفوفة A.  N >= 0.
!>

D

!>          D هو مصفوفة حقيقية (REAL array)، البعد (N)
!>          العناصر القطرية n للمصفوفة القطرية D من
!>          تحلل A، كما هو محسوب بواسطة SPTTRF.
!>

E

!>          E هو مصفوفة حقيقية (REAL array)، البعد (N-1)
!>          العناصر خارج القطر (n-1) للعامل المتشابه قطرياً أحادي الوحدة
!>          U أو L من تحلل A، كما هو محسوب بواسطة SPTTRF.
!>

ANORM

!>          ANORM هو عدد حقيقي (REAL)
!>          المعيار 1 للمصفوفة الأصلية A.
!>

RCOND

!>          RCOND هو عدد حقيقي (REAL)
!>          مقلوب رقم شرط المصفوفة A،
!>          محسوب على النحو RCOND = 1/(ANORM * AINVNM)، حيث AINVNM هو
!>          المعيار 1 لـ inv(A) المحسوب في هذا الروتين.
!>

WORK

!>          WORK هي مصفوفة حقيقية (REAL array)، البعد (N)
!>

INFO

!>          INFO هو عدد صحيح (INTEGER)
!>          = 0:  خروج ناجح
!>          < 0:  إذا كان INFO = -i، فإن المعلمة i كانت ذات قيمة غير قانونية
!>
