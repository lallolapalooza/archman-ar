subroutine sppsvx (character fact, character uplo, integer n, integer nrhs, real, dimension( * ) ap, real, dimension( * ) afp, character equed, real, dimension( * ) s, real, dimension( ldb, * ) b, integer ldb, real, dimension( ldx, * ) x, integer ldx, real rcond, real, dimension( * ) ferr, real, dimension( * ) berr, real, dimension( * ) work, integer, dimension( * ) iwork, integer info)

SPPSVX يحسب حل نظام المعادلات الخطية A * X = B لمصفوفات أخرى

الغرض:

!>
!> يستخدم SPPSVX تحليل Cholesky A = U**T*U أو A = L*L**T لحساب حل نظام
!> المعادلات الخطية الحقيقية
!>    A * X = B,
!> حيث A مصفوفة متماثلة محددة موجبة من الحجم N*N مخزنة بتنسيق مكدس، و X و B
!> مصفوفتان من الحجم N*NRHS.
!>
!> يتم أيضًا توفير حدود الخطأ للحل وتقدير للحالة.
!>

الوصف:

!>
!> يتم تنفيذ الخطوات التالية:
!>
!> 1. إذا كانت FACT = 'E'، يتم حساب عوامل قياس حقيقية لتحقيق التوازن
!>    للنظام:
!>       diag(S) * A * diag(S) * inv(diag(S)) * X = diag(S) * B
!>    يعتمد ما إذا كان سيتم تحقيق التوازن للنظام أم لا على
!>    قياس المصفوفة A، ولكن إذا تم استخدام تحقيق التوازن، يتم
!>    الكتابة فوق A بواسطة diag(S)*A*diag(S) و B بواسطة diag(S)*B.
!>
!> 2. إذا كانت FACT = 'N' أو 'E'، يتم استخدام تحليل Cholesky
!>    لتحليل المصفوفة A (بعد تحقيق التوازن إذا كانت FACT = 'E') على النحو التالي:
!>       A = U**T* U,  إذا كانت UPLO = 'U'، أو
!>       A = L * L**T,  إذا كانت UPLO = 'L'،
!>    حيث U مصفوفة مثلثة علوية و L مصفوفة مثلثة سفلية.
!>
!> 3. إذا كان الحد الرئيسي الأصغر من الرتبة i غير موجب،
!>    فإن الإجراء الفرعي يعود بـ INFO = i. وإلا، يتم استخدام الشكل المحلل
!>    لـ A لتقدير رقم حالة المصفوفة A. إذا كان مقلوب رقم الحالة أقل من
!>    دقة الآلة، يتم إرجاع INFO = N+1 كتحذير، ولكن الإجراء الفرعي
!>    يستمر في الحل لـ X وحساب حدود الخطأ كما هو موضح أدناه.
!>
!> 4. يتم حل نظام المعادلات لـ X باستخدام الشكل المحلل لـ A.
!>
!> 5. يتم تطبيق تحسين تكراري لتحسين الحل المحسوب
!>    وحساب حدود الخطأ وتقديرات الخطأ الخلفي له.
!>
!> 6. إذا تم استخدام تحقيق التوازن، يتم ضرب المصفوفة X مسبقًا
!>    بـ diag(S) بحيث تحل النظام الأصلي قبل
!>    تحقيق التوازن.
!>

المعلمات

FACT 
!>          FACT is CHARACTER*1
!>          يحدد ما إذا كان الشكل المحلل للمصفوفة A مقدمًا عند الإدخال،
!>          وإذا لم يكن كذلك، فهل يجب تحقيق التوازن للمصفوفة A قبل تحليلها.
!>          = 'F':  عند الإدخال، يحتوي AFP على الشكل المحلل لـ A.
!>                  إذا كانت EQUED = 'Y'، فقد تم تحقيق التوازن للمصفوفة A
!>                  باستخدام عوامل القياس المعطاة بواسطة S. لن يتم تعديل
!>                  AP و AFP.
!>          = 'N':  سيتم نسخ المصفوفة A إلى AFP وتحليلها.
!>          = 'E':  سيتم تحقيق التوازن للمصفوفة A إذا لزم الأمر، ثم
!>                  نسخها إلى AFP وتحليلها.
!>

UPLO

!>          UPLO is CHARACTER*1
!>          = 'U':  يتم تخزين الجزء العلوي من A؛
!>          = 'L':  يتم تخزين الجزء السفلي من A.
!>

N

!>          N is INTEGER
!>          عدد المعادلات الخطية، أي ترتيب المصفوفة A. N >= 0.
!>

NRHS

!>          NRHS is INTEGER
!>          عدد الأطراف اليمنى، أي عدد أعمدة المصفوفتين B و X. NRHS >= 0.
!>

AP

!>          AP is REAL array, dimension (N*(N+1)/2)
!>          عند الإدخال، الجزء العلوي أو السفلي من المصفوفة المتماثلة
!>          A، مكدس حسب العمود في مصفوفة خطية، باستثناء إذا كانت FACT = 'F'
!>          و EQUED = 'Y'، فإن A يجب أن تحتوي على المصفوفة المتوازنة
!>          diag(S)*A*diag(S). يتم تخزين العمود j من A في
!>          المصفوفة AP كالتالي:
!>          إذا كانت UPLO = 'U'، AP(i + (j-1)*j/2) = A(i,j) لـ 1<=i<=j؛
!>          إذا كانت UPLO = 'L'، AP(i + (j-1)*(2n-j)/2) = A(i,j) لـ j<=i<=n.
!>          انظر أدناه لمزيد من التفاصيل. لا يتم تعديل A إذا
!>          كانت FACT = 'F' أو 'N'، أو إذا كانت FACT = 'E' و EQUED = 'N' عند الخروج.
!>
!>          عند الخروج، إذا كانت FACT = 'E' و EQUED = 'Y'، يتم الكتابة فوق A بواسطة
!>          diag(S)*A*diag(S).
!>

AFP

!>          AFP is REAL array, dimension (N*(N+1)/2)
!>          إذا كانت FACT = 'F'، فإن AFP هو معلمة إدخال ويحتوي عند الإدخال
!>          على العامل المثلثي U أو L من تحليل Cholesky
!>          A = U**T*U أو A = L*L**T، بنفس تنسيق التخزين الخاص بـ A.
!>          إذا كانت EQUED .ne. 'N'، فإن AFP هو الشكل المحلل
!>          للمصفوفة المتوازنة A.
!>
!>          إذا كانت FACT = 'N'، فإن AFP هو معلمة إخراج ويحتوي عند الخروج
!>          على العامل المثلثي U أو L من تحليل Cholesky
!>          A = U**T * U أو A = L * L**T للمصفوفة الأصلية A.
!>
!>          إذا كانت FACT = 'E'، فإن AFP هو معلمة إخراج ويحتوي عند الخروج
!>          على العامل المثلثي U أو L من تحليل Cholesky
!>          A = U**T * U أو A = L * L**T للمصفوفة المتوازنة A (انظر وصف AP
!>          لشكل المصفوفة المتوازنة).
!>

EQUED

!>          EQUED is CHARACTER*1
!>          يحدد شكل تحقيق التوازن الذي تم إجراؤه.
!>          = 'N':  لم يتم تحقيق التوازن (دائمًا صحيح إذا كانت FACT = 'N').
!>          = 'Y':  تم تحقيق التوازن، أي تم استبدال A بـ
!>                  diag(S) * A * diag(S).
!>          EQUED هي معلمة إدخال إذا كانت FACT = 'F'؛ بخلاف ذلك، فهي معلمة
!>          إخراج.
!>

S

!>          S is REAL array, dimension (N)
!>          عوامل القياس لـ A؛ لا يتم الوصول إليها إذا كانت EQUED = 'N'. S هو
!>          معلمة إدخال إذا كانت FACT = 'F'؛ بخلاف ذلك، S هو معلمة
!>          إخراج. إذا كانت FACT = 'F' و EQUED = 'Y'، يجب أن يكون كل عنصر من S
!>          موجبًا.
!>

B

!>          B is REAL array, dimension (LDB,NRHS)
!>          عند الإدخال، مصفوفة الطرف الأيمن B بالحجم N*NRHS.
!>          عند الخروج، إذا كانت EQUED = 'N'، لا يتم تعديل B؛ إذا كانت EQUED = 'Y'،
!>          يتم الكتابة فوق B بواسطة diag(S) * B.
!>

LDB

!>          LDB is INTEGER
!>          البعد الرائد للمصفوفة B. LDB >= max(1,N).
!>

X

!>          X is REAL array, dimension (LDX,NRHS)
!>          إذا كانت INFO = 0 أو INFO = N+1، مصفوفة الحل X بالحجم N*NRHS
!>          للنظام الأصلي للمعادلات. لاحظ أنه إذا كانت EQUED = 'Y'،
!>          فإن A و B يتم تعديلهما عند الخروج، وحل النظام
!>          المتوازن هو inv(diag(S))*X.
!>

LDX

!>          LDX is INTEGER
!>          البعد الرائد للمصفوفة X. LDX >= max(1,N).
!>

RCOND

!>          RCOND is REAL
!>          تقدير رقم الحالة المقلوب للمصفوفة A بعد تحقيق التوازن (إذا تم).
!>          إذا كان RCOND أقل من دقة الآلة (على وجه الخصوص، إذا كان RCOND = 0)،
!>          فإن المصفوفة منفردة بالنسبة لدقة العمل. يتم الإشارة إلى هذه الحالة
!>          برمز عودة INFO > 0.
!>

Function/Subroutine Documentation

The core of the description for the parameters of the `sppsvx` subroutine, including explanations for `FERR`, `BERR`, `WORK`, `IWORK`, and `INFO`, is as follows:

FERR

!>          FERR is REAL array, dimension (NRHS)
!>          The estimated forward error bound for each solution vector
!>          X(j) (the j-th column of the solution matrix X).
!>          If XTRUE is the true solution corresponding to X(j), FERR(j)
!>          is an estimated upper bound for the magnitude of the largest
!>          element in (X(j) - XTRUE) divided by the magnitude of the
!>          largest element in X(j).  The estimate is as reliable as
!>          the estimate for RCOND, and is almost always a slight
!>          overestimate of the true error.
!>

BERR

!>          BERR is REAL array, dimension (NRHS)
!>          The componentwise relative backward error of each solution
!>          vector X(j) (i.e., the smallest relative change in
!>          any element of A or B that makes X(j) an exact solution).
!>

WORK

!>          WORK is REAL array, dimension (3*N)
!>

IWORK

!>          IWORK is INTEGER array, dimension (N)
!>

INFO

!>          INFO is INTEGER
!>          = 0:  successful exit
!>          < 0:  if INFO = -i, the i-th argument had an illegal value
!>          > 0:  if INFO = i, and i is
!>                <= N:  the leading principal minor of order i of A
!>                       is not positive, so the factorization could not
!>                       be completed, and the solution has not been
!>                       computed. RCOND = 0 is returned.
!>                = N+1: U is nonsingular, but RCOND is less than machine
!>                       precision, meaning that the matrix is singular
!>                       to working precision.  Nevertheless, the
!>                       solution and error bounds are computed because
!>                       there are a number of situations where the
!>                       computed solution can be more accurate than the
!>                       value of RCOND would suggest.
!>
