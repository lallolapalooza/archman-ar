الاسم

apptainer-sif-add - إضافة كائن بيانات

الملخص

apptainer sif add [خيارات]

الوصف

أضف كائن بيانات إلى صورة SIF.

الخيارات

--alignment=0 تعيين المحاذاة [الافتراضي: 4096 مع --datatype 4-Partition، 0 بخلاف ذلك]

--datatype=0 نوع البيانات المراد إضافتها [مطلوب، لا يوجد افتراضي]:
1-Deffile، 2-EnvVar، 3-Labels،
4-Partition، 5-Signature، 6-GenericJSON،
7-Generic، 8-CryptoMessage، 9-SBOM،
10-OCI.RootIndex، 11-OCI.Blob

--filename="" تعيين اسم ملف منطقي/مقبض [الافتراضي: اسم ملف الإدخال]

--groupid=0 تعيين معرف المجموعة [الافتراضي: 0]

-h, --help[=false] المساعدة للإضافة

--link=0 تعيين مؤشر الارتباط [الافتراضي: 0]

--partarch=0 البنية الرئيسية المستخدمة (مع --datatype 4-Partition) [مطلوب، لا يوجد افتراضي]:
1-386، 2-amd64، 3-arm،
4-arm64، 5-ppc64، 6-ppc64le،
7-mips، 8-mipsle، 9-mips64،
10-mips64le، 11-s390x، 12-riscv64

--partfs=0 نظام الملفات المستخدم (مع --datatype 4-Partition) [مطلوب، لا يوجد افتراضي]:
1-Squash، 2-Ext3، 3-ImmuObj،
4-Raw

--parttype=0 نوع القسم (مع --datatype 4-Partition) [مطلوب، لا يوجد افتراضي]:
1-System، 2-PrimSys، 3-Data،
4-Overlay

--sbomformat="" تنسيق SBOM (مع --datatype 9-sbom):
cyclonedx-json، cyclonedx-xml، github-json،
spdx-json، spdx-rdf، spdx-tag-value،
spdx-yaml، syft-json

--signentity="" الكيان الذي يوقع (مع --datatype 5-Signature) [مطلوب، لا يوجد افتراضي]:
مثال: 433FE984155206BD962725E20E8713472A879943

--signhash=0 تجزئة التوقيع المستخدمة (مع --datatype 5-Signature) [مطلوب، لا يوجد افتراضي]:
1-SHA256، 2-SHA384، 3-SHA512،
4-BLAKE2s_256، 5-BLAKE2b_256

مثال

sif add image.sif recipe.def --datatype 1
sif add image.sif rootfs.squashfs --datatype 4 --parttype 1 --partfs 1 --partarch 2
sif add image.sif signature.bin --datatype 5 --signentity 433FE984155206BD962725E20E8713472A879943 --signhash 1
