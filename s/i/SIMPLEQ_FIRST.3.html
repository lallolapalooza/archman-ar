STAILQ(3) أدلة الوظائف القياسية STAILQ(3)

الاسم

SIMPLEQ_EMPTY, SIMPLEQ_ENTRY, SIMPLEQ_FIRST, SIMPLEQ_FOREACH, SIMPLEQ_HEAD, SIMPLEQ_HEAD_INITIALIZER, SIMPLEQ_INIT, SIMPLEQ_INSERT_AFTER, SIMPLEQ_INSERT_HEAD, SIMPLEQ_INSERT_TAIL, SIMPLEQ_NEXT, SIMPLEQ_REMOVE, SIMPLEQ_REMOVE_HEAD, - تطبيق قائمة ذات ارتباط مزدوج

المكتبة

مكتبة C القياسية (libc, -lc)

الملخص

#include <sys/queue.h>
STAILQ_ENTRY(TYPE);
STAILQ_HEAD(HEADNAME, TYPE);
STAILQ_HEAD STAILQ_HEAD_INITIALIZER(STAILQ_HEAD head);
void STAILQ_INIT(STAILQ_HEAD *head);
int STAILQ_EMPTY(STAILQ_HEAD *head);
void STAILQ_INSERT_HEAD(STAILQ_HEAD *head,
                         struct TYPE *elm, STAILQ_ENTRY NAME);
void STAILQ_INSERT_TAIL(STAILQ_HEAD *head,
                         struct TYPE *elm, STAILQ_ENTRY NAME);
void STAILQ_INSERT_AFTER(STAILQ_HEAD *head, struct TYPE *listelm,
                         struct TYPE *elm, STAILQ_ENTRY NAME);
struct TYPE *STAILQ_FIRST(STAILQ_HEAD *head);
struct TYPE *STAILQ_NEXT(struct TYPE *elm, STAILQ_ENTRY NAME);
STAILQ_FOREACH(struct TYPE *var, STAILQ_HEAD *head, STAILQ_ENTRY NAME);
void STAILQ_REMOVE(STAILQ_HEAD *head, struct TYPE *elm, TYPE,
                         STAILQ_ENTRY NAME);
void STAILQ_REMOVE_HEAD(STAILQ_HEAD *head,
                         STAILQ_ENTRY NAME);
void STAILQ_CONCAT(STAILQ_HEAD *head1, STAILQ_HEAD *head2);

ملاحظة: توجد وحدات ماكرو متطابقة مسبوقة بـ SIMPLEQ بدلاً من STAILQ؛ انظر VERSIONS.

الوصف

تعرف هذه الوحدات وتعمل على قائمة ذات ارتباط مزدوج.

في تعريفات وحدات الماكرو، TYPE هو اسم بنية محددة من قبل المستخدم، والتي يجب أن تحتوي على حقل من النوع STAILQ_ENTRY، يسمى NAME. الوسيط HEADNAME هو اسم بنية محددة من قبل المستخدم، والتي يجب تعريفها باستخدام الماكرو STAILQ_HEAD().

الإنشاء

يتم تنظيم قائمة ذات ارتباط مزدوج بواسطة بنية محددة بواسطة الماكرو STAILQ_HEAD(). تحتوي هذه البنية على زوج من المؤشرات، واحد إلى العنصر الأول في القائمة والآخر إلى العنصر الأخير في القائمة. ترتبط العناصر بشكل بسيط، مما يعني الحد الأدنى من المساحة المهدرة والعبء المنخفض على معالجة المؤشرات. يمكن إضافة العناصر إلى القائمة بعد عنصر موجود، في مقدمة القائمة، أو في نهاية القائمة. يتم تعريف بنية STAILQ_HEAD على النحو التالي:


STAILQ_HEAD(HEADNAME, TYPE) head;

حيث struct HEADNAME هي البنية التي سيتم تعريفها و struct TYPE هو نوع العناصر المرتبطة في القائمة. يمكن لاحقًا تعريف مؤشر إلى رأس القائمة على هذا النحو:


struct HEADNAME *headp;

(أسماء head و headp قابلة للاختيار من قبل المستخدم.)

STAILQ_ENTRY() تعرف بنية تربط عناصر القائمة.

STAILQ_HEAD_INITIALIZER() تقيم إلى مُهيئ للقائمة head.

STAILQ_INIT() تهيئ القائمة المشار إليها بواسطة head.

STAILQ_EMPTY() تقيم إلى صحيح إذا لم تكن هناك عناصر في القائمة.

الإدراج

STAILQ_INSERT_HEAD() تدرج العنصر elm في مقدمة القائمة.

STAILQ_INSERT_TAIL() تدرج العنصر elm في نهاية القائمة.

STAILQ_INSERT_AFTER() تدرج العنصر elm بعد العنصر listelm.

التكرار

STAILQ_FIRST() ترجع العنصر الأول في القائمة أو NULL إذا كانت القائمة فارغة.

STAILQ_NEXT() ترجع العنصر التالي في القائمة، أو NULL إذا كان هذا هو الأخير.

STAILQ_FOREACH() تتكرر على القائمة المشار إليها بواسطة head في اتجاه الأمام، وتعيين كل عنصر إلى var.

الحذف

STAILQ_REMOVE() تزيل العنصر elm من القائمة.

STAILQ_REMOVE_HEAD() تزيل العنصر الموجود في مقدمة القائمة. للحصول على الأداء الأمثل، يجب على العناصر التي يتم إزالتها من مقدمة القائمة استخدام هذه الماكرو بشكل صريح بدلاً من الماكرو العامة STAILQ_REMOVE().

وظائف أخرى

STAILQ_CONCAT() ستلحق القائمة ذات الارتباط المزدوج المشار إليها بواسطة head2 بنهاية القائمة المشار إليها بواسطة head1، مع إزالة جميع العناصر من الأولى.

قيمة الإرجاع

STAILQ_EMPTY() ترجع قيمة غير صفرية إذا كانت القائمة فارغة، وصفر إذا كانت تحتوي على عنصر واحد على الأقل.

STAILQ_FIRST() و STAILQ_NEXT() ترجعان مؤشرًا إلى بنية TYPE الأولى أو التالية، على التوالي.

STAILQ_HEAD_INITIALIZER() ترجع مُهيئًا يمكن تعيينه إلى رأس القائمة head.

الإصدارات

تقدم بعض أنظمة BSD البديلة SIMPLEQ بدلاً من STAILQ. وهما متطابقان، ولكن لأسباب تاريخية، تم تسميتهما بشكل مختلف في أنظمة BSD المختلفة. نشأت STAILQ في FreeBSD، و SIMPLEQ في NetBSD. للتوافق، توفر بعض الأنظمة كلتا السلسلتين من وحدات الماكرو. يوفر glibc كلاً من STAILQ و SIMPLEQ، وهما متطابقان، باستثناء عدم وجود نظير SIMPLEQ لـ STAILQ_CONCAT.

مشاكل

STAILQ_FOREACH() لا تسمح بإزالة أو تحرير var داخل الحلقة، حيث يمكن أن تتداخل مع التكرار. STAILQ_FOREACH_SAFE()، الموجود في أنظمة BSD ولكن ليس في glibc، يصلح هذا الحد من خلال السماح بإزالة var من القائمة وتحريرها داخل الحلقة دون التدخل في التكرار.

أمثلة

#include <stddef.h>
#include <stdio.h>
#include <stdlib.h>
#include <sys/queue.h>
struct entry {
    int data;
    STAILQ_ENTRY(entry) entries;        /* قائمة مزدوجة الارتباط */
};
STAILQ_HEAD(stailhead, entry);
int
main(void)
{
    struct entry *n1, *n2, *n3, *np;
    struct stailhead head;                  /* رأس قائمة مزدوجة الارتباط */
    STAILQ_INIT(&head);                     /* تهيئة القائمة */
    n1 = malloc(sizeof(struct entry));      /* الإدراج في المقدمة */
    STAILQ_INSERT_HEAD(&head, n1, entries);
    n1 = malloc(sizeof(struct entry));      /* الإدراج في النهاية */
    STAILQ_INSERT_TAIL(&head, n1, entries);
    n2 = malloc(sizeof(struct entry));      /* الإدراج بعد */
    STAILQ_INSERT_AFTER(&head, n1, n2, entries);
    STAILQ_REMOVE(&head, n2, entry, entries); /* الحذف */
    free(n2);
    n3 = STAILQ_FIRST(&head);
    STAILQ_REMOVE_HEAD(&head, entries);     /* الحذف من المقدمة */
    free(n3);
    n1 = STAILQ_FIRST(&head);
    n1->data = 0;
    for (unsigned int i = 1; i < 5; i++) {
        n1 = malloc(sizeof(struct entry));
        STAILQ_INSERT_HEAD(&head, n1, entries);
        n1->data = i;
    }

/* اجتياز أمامي */ STAILQ_FOREACH(np, &head, entries) printf("%i\n", np->data); /* حذف TailQ */ n1 = STAILQ_FIRST(&head); while (n1 != NULL) { n2 = STAILQ_NEXT(n1, entries); free(n1); n1 = n2; } STAILQ_INIT(&head); exit(EXIT_SUCCESS); }
