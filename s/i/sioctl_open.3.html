sioctl_open(3)

الحزمة: sndio

التراخيص: ISC

SIOCTL_OPEN(3) دليل وظائف المكتبة SIOCTL_OPEN(3)

الاسم

sioctl_open، sioctl_close، sioctl_ondesc، sioctl_onval، sioctl_setval، sioctl_nfds، sioctl_pollfd، sioctl_eof — واجهة لمعلمات الصوت

المُلَخَّص

#include <sndio.h>

struct sioctl_hdl *
sioctl_open(const char *name, unsigned int mode, int nbio_flag);

void
sioctl_close(struct sioctl_hdl *hdl);

int
sioctl_ondesc(struct sioctl_hdl *hdl, void (*cb)(void *arg, struct sioctl_desc *desc, int val), void *arg);

int
sioctl_onval(struct sioctl_hdl *hdl, void (*cb)(void *arg, unsigned int addr, unsigned int val), void *arg);

int
sioctl_setval(struct sioctl_hdl *hdl, unsigned int addr, unsigned int val);

int
sioctl_nfds(struct sioctl_hdl *hdl);

int
sioctl_pollfd(struct sioctl_hdl *hdl, struct pollfd *pfd, int events);

int
sioctl_revents(struct sioctl_hdl *hdl, struct pollfd *pfd);

int
sioctl_eof(struct sioctl_hdl *hdl);

الوصف

قد تكشف أجهزة الصوت عن عدد من الضوابط، مثل التحكم في مستوى صوت التشغيل. لكل تحكم عنوان عددي وقيمة عددية. بعض القيم منطقية ولا يمكن تبديلها إلا إلى 0 (إيقاف) أو 1 (تشغيل). يمكن تغيير أي تحكم عن طريق إرسال قيمة جديدة إلى عنوانه. عند تغيير القيم، يتم إرسال إشعارات غير متزامنة.

أوصاف الضوابط متاحة، مما يسمح بتجميعها وتمثيلها بشكل مفهوم للبشر.

فتح وإغلاق جهاز التحكم

أولاً، يجب على التطبيق استدعاء الدالة sioctl_open() للحصول على مقبض سيتم تمريره كمعامل hdl إلى الدوال الأخرى.

يحدد المعامل name سلسلة الجهاز التي نوقشت في sndio(7). في معظم الحالات، يجب تعيينه إلى SIO_DEVANY للسماح للمستخدم بتحديده باستخدام متغير البيئة AUDIODEVICE. يحدد المعامل mode خريطة بت من الثوابت SIOCTL_READ و SIOCTL_WRITE التي تشير إلى ما إذا كان يمكن قراءة قيم التحكم وتعديلها على التوالي.

إذا كانت قيمة المعامل nbio_flag هي 1، فقد تفشل الدالة sioctl_setval() (انظر أدناه) بدلاً من حظر، والدالة sioctl_ondesc() لا تحظر.

الدالة sioctl_close() تغلق جهاز التحكم وتحرر أي موارد محجوزة مرتبطة بالمقبض.

أوصاف الضوابط

يمكن استخدام الدالة sioctl_ondesc() للحصول على وصف لجميع الضوابط المتاحة وقيمها الأولية. تقوم بتسجيل دالة رد اتصال يتم استدعاؤها فورًا لجميع الضوابط. يتم استدعاؤها مرة واحدة مع معامل NULL للإشارة إلى أنه تم إرسال الوصف الكامل وأن للمتصل تمثيلاً متسقًا لمجموعة الضوابط.

ثم، كلما تغير وصف تحكم، يتم استدعاء دالة رد الاتصال بالمعلومات المحدثة متبوعة باستدعاء بمعامل NULL.

يتم وصف الضوابط بواسطة البنية sioctl_desc على النحو التالي:

struct sioctl_node {
	char name[SIOCTL_NAMEMAX];	/* ex. "spkr" */
	int unit;			/* optional number or -1 */
};

struct sioctl_desc {
	unsigned int addr;		/* control address */
#define SIOCTL_NONE		0	/* deleted */
#define SIOCTL_NUM		2	/* integer in the maxval range */
#define SIOCTL_SW		3	/* on/off switch (1 or 0) */
#define SIOCTL_VEC		4	/* number, element of vector */
#define SIOCTL_LIST		5	/* switch, element of a list */
#define SIOCTL_SEL		6	/* element of a selector */
	unsigned int type;		/* one of above */
	char func[SIOCTL_NAMEMAX];	/* function name, ex. "level" */
	char group[SIOCTL_NAMEMAX];	/* group this control belongs to */
	struct sioctl_node node0;	/* affected node */
	struct sioctl_node node1;	/* dito for SIOCTL_{VEC,LIST,SEL} */
	unsigned int maxval;		/* max value */
	char display[SIOCTL_DISPLAYMAX];	/* free-format hint */
};

السمة addr هي عنوان التحكم، يمكن استخدامه مع sioctl_setval() لتعيين قيمته.

تشير السمة type إلى ما تصفه البنية. الأنواع الممكنة هي:

SIOCTL_NONE
تم حذف تحكم كان صالحًا سابقًا.
SIOCTL_NUM
تحكم عددي في النطاق من 0 إلى maxval، شامل. على سبيل المثال، مستوى صوت مكبر الصوت.
SIOCTL_SW
تحكم منطقي. على سبيل المثال، مفتاح كتم صوت مكبر الصوت.
SIOCTL_VEC
عنصر في مصفوفة ضوابط عددية. على سبيل المثال، مقبض للتحكم في مقدار الإشارة المتدفقة من مدخل الخط إلى مكبر الصوت.
SIOCTL_LIST
عنصر في مصفوفة مفاتيح منطقية. على سبيل المثال، موضع خط الإدخال في محدد مصدر مكبر الصوت.
SIOCTL_SEL
نفس SIOCTL_LIST ولكن يتم تحديد عنصر واحد بالضبط في كل مرة.

السمة func هي اسم المعلمة التي يتم التحكم فيها. قد لا يكون هناك معلمات من أنواع مختلفة بنفس الاسم.

السمتان node0 و node1 تشيران إلى أسماء العقد التي يتم التحكم فيها، عادةً قنوات تدفقات الصوت. node1 ذات مغزى لـ SIOCTL_VEC و SIOCTL_LIST و SIOCTL_SEL فقط.

الأسماء في السمتين node0 و node1 و func هي سلاسل يمكن استخدامها كمعرفات فريدة داخل group المحدد.

تشير السمة maxval إلى الحد الأقصى لقيمة هذا التحكم. بالنسبة لأنواع التحكم المنطقية، يتم تعيينها إلى 1.

تحتوي السمة display على سلسلة اختيارية بتنسيق حر توفر تلميحات إضافية حول التحكم، مثل طراز الأجهزة، أو الوحدات.

تغيير وقراءة قيم الضوابط
