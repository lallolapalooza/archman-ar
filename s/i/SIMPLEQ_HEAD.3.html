SIMPLEQ_EMPTY, SIMPLEQ_ENTRY, SIMPLEQ_FIRST, SIMPLEQ_FOREACH, SIMPLEQ_HEAD, SIMPLEQ_HEAD_INITIALIZER, SIMPLEQ_INIT, SIMPLEQ_INSERT_AFTER, SIMPLEQ_INSERT_HEAD, SIMPLEQ_INSERT_TAIL, SIMPLEQ_NEXT, SIMPLEQ_REMOVE, SIMPLEQ_REMOVE_HEAD, STAILQ_CONCAT, STAILQ_EMPTY, STAILQ_ENTRY, STAILQ_FIRST, STAILQ_FOREACH, STAILQ_HEAD, STAILQ_HEAD_INITIALIZER, STAILQ_INIT, STAILQ_INSERT_AFTER, STAILQ_INSERT_HEAD, STAILQ_INSERT_TAIL, STAILQ_NEXT, STAILQ_REMOVE, STAILQ_REMOVE_HEAD, - تطبيق قائمة انتظار ذات وصلة فردية مع ذيل

المكتبة

مكتبة C القياسية (libc, -lc)

ملخص

#include <sys/queue.h>
STAILQ_ENTRY(TYPE);
STAILQ_HEAD(HEADNAME, TYPE);
STAILQ_HEAD STAILQ_HEAD_INITIALIZER(STAILQ_HEAD head);
void STAILQ_INIT(STAILQ_HEAD *head);
int STAILQ_EMPTY(STAILQ_HEAD *head);
void STAILQ_INSERT_HEAD(STAILQ_HEAD *head,
                         struct TYPE *elm, STAILQ_ENTRY NAME);
void STAILQ_INSERT_TAIL(STAILQ_HEAD *head,
                         struct TYPE *elm, STAILQ_ENTRY NAME);
void STAILQ_INSERT_AFTER(STAILQ_HEAD *head, struct TYPE *listelm,
                         struct TYPE *elm, STAILQ_ENTRY NAME);
struct TYPE *STAILQ_FIRST(STAILQ_HEAD *head);
struct TYPE *STAILQ_NEXT(struct TYPE *elm, STAILQ_ENTRY NAME);
STAILQ_FOREACH(struct TYPE *var, STAILQ_HEAD *head, STAILQ_ENTRY NAME);
void STAILQ_REMOVE(STAILQ_HEAD *head, struct TYPE *elm, TYPE,
                         STAILQ_ENTRY NAME);
void STAILQ_REMOVE_HEAD(STAILQ_HEAD *head,
                         STAILQ_ENTRY NAME);
void STAILQ_CONCAT(STAILQ_HEAD *head1, STAILQ_HEAD *head2);

ملاحظة: توجد وحدات ماكرو متطابقة تحمل البادئة SIMPLEQ بدلاً من STAILQ؛ انظر الإصدارات.

الوصف

تُعرّف هذه الوحدات ماكرو وتشغل قوائم انتظار ذات وصلة فردية مع ذيل.

في تعريفات وحدات الماكرو، TYPE هو اسم بنية معرفة من قبل المستخدم، يجب أن تحتوي على حقل من النوع STAILQ_ENTRY، واسمه NAME. الوسيط HEADNAME هو اسم بنية معرفة من قبل المستخدم يجب الإعلان عنها باستخدام الماكرو STAILQ_HEAD().

الإنشاء

تُدار قائمة انتظار ذات وصلة فردية مع ذيل بواسطة بنية يُعرّفها الماكرو STAILQ_HEAD(). تحتوي هذه البنية على زوج من المؤشرات، أحدهما يشير إلى العنصر الأول في قائمة الانتظار والآخر يشير إلى العنصر الأخير في قائمة الانتظار. يتم ربط العناصر ببعضها البعض بشكل فردي لتقليل استهلاك المساحة وتكاليف معالجة المؤشرات عند إزالة عناصر عشوائية، مقابل تكلفة O(n). يمكن إضافة عناصر جديدة إلى قائمة الانتظار بعد عنصر موجود، أو في بداية قائمة الانتظار، أو في نهاية قائمة الانتظار. يتم الإعلان عن بنية STAILQ_HEAD كالتالي:


STAILQ_HEAD(HEADNAME, TYPE) head;

حيث struct HEADNAME هي البنية المراد تعريفها، و struct TYPE هو نوع العناصر التي سيتم ربطها في قائمة الانتظار. يمكن لاحقًا الإعلان عن مؤشر إلى رأس قائمة الانتظار كالتالي:


struct HEADNAME *headp;

(الأسماء head و headp قابلة للاختيار من قبل المستخدم.)

يعلن STAILQ_ENTRY() عن بنية تربط العناصر في قائمة الانتظار.

يقوم STAILQ_HEAD_INITIALIZER() بتقييم قيمة تهيئة لقائمة الانتظار head.

يقوم STAILQ_INIT() بتهيئة قائمة الانتظار المشار إليها بواسطة head.

يقوم STAILQ_EMPTY() بتقييم قيمة صحيحة إذا لم تكن هناك عناصر في قائمة الانتظار.

الإدخال

يقوم STAILQ_INSERT_HEAD() بإدخال العنصر الجديد elm في بداية قائمة الانتظار.

يقوم STAILQ_INSERT_TAIL() بإدخال العنصر الجديد elm في نهاية قائمة الانتظار.

يقوم STAILQ_INSERT_AFTER() بإدخال العنصر الجديد elm بعد العنصر listelm.

اجتياز

يقوم STAILQ_FIRST() بإرجاع العنصر الأول في قائمة الانتظار أو NULL إذا كانت قائمة الانتظار فارغة.

يقوم STAILQ_NEXT() بإرجاع العنصر التالي في قائمة الانتظار، أو NULL إذا كان هذا العنصر هو الأخير.

يجتاز STAILQ_FOREACH() قائمة الانتظار المشار إليها بواسطة head في الاتجاه الأمامي، مع تعيين كل عنصر بدوره إلى var.

الإزالة

يقوم STAILQ_REMOVE() بإزالة العنصر elm من قائمة الانتظار.

يقوم STAILQ_REMOVE_HEAD() بإزالة العنصر الموجود في بداية قائمة الانتظار. للحصول على أقصى قدر من الكفاءة، يجب على العناصر التي تتم إزالتها من بداية قائمة الانتظار استخدام هذا الماكرو صراحةً بدلاً من الماكرو العام STAILQ_REMOVE().

ميزات أخرى

يقوم STAILQ_CONCAT() بدمج قائمة الانتظار التي يرأسها head2 في نهاية قائمة الانتظار التي يرأسها head1، مع إزالة جميع الإدخالات من الأولى.

قيمة الإرجاع

يقوم STAILQ_EMPTY() بإرجاع قيمة غير صفرية إذا كانت القائمة فارغة، وصفر إذا كانت القائمة تحتوي على إدخال واحد على الأقل.

يقوم STAILQ_FIRST() و STAILQ_NEXT() بإرجاع مؤشر إلى البنية الأولى أو التالية TYPE، على التوالي.

يقوم STAILQ_HEAD_INITIALIZER() بإرجاع قيمة تهيئة يمكن تعيينها إلى رأس القائمة head.

الإصدارات

توفر بعض أنظمة BSD الوظيفة SIMPLEQ بدلاً من STAILQ. هما متطابقان، ولكن لأسباب تاريخية تم تسميتهما بشكل مختلف في أنظمة BSD المختلفة. نشأت STAILQ في FreeBSD، ونشأت SIMPLEQ في NetBSD. لأسباب التوافق، توفر بعض الأنظمة مجموعتي وحدات الماكرو. يوفر glibc كلاً من STAILQ و SIMPLEQ، وهما متطابقان باستثناء عدم وجود مكافئ لـ SIMPLEQ للوظيفة STAILQ_CONCAT().

أخطاء

لا يسمح STAILQ_FOREACH() بإزالة أو تحرير var داخل الحلقة، حيث قد يتداخل ذلك مع الاجتياز. تقوم STAILQ_FOREACH_SAFE()، الموجودة في أنظمة BSD ولكنها غير موجودة في glibc، بإصلاح هذا القيد عن طريق السماح بإزالة var بأمان من القائمة وتحريرها من داخل الحلقة دون التأثير على الاجتياز.

أمثلة

#include <stddef.h>
#include <stdio.h>
#include <stdlib.h>
#include <sys/queue.h>
struct entry {
    int data;
    STAILQ_ENTRY(entry) entries;        /* قائمة انتظار ذات وصلة فردية مع ذيل */
};
STAILQ_HEAD(stailhead, entry);
int
main(void)
{
    struct entry *n1, *n2, *n3, *np;
    struct stailhead head;                  /* رأس قائمة انتظار ذات وصلة فردية مع ذيل */
    STAILQ_INIT(&head);                     /* تهيئة القائمة */
    n1 = malloc(sizeof(struct entry));      /* إدخال في البداية */
    STAILQ_INSERT_HEAD(&head, n1, entries);
    n1 = malloc(sizeof(struct entry));      /* إدخال في النهاية */
    STAILQ_INSERT_TAIL(&head, n1, entries);
    n2 = malloc(sizeof(struct entry));      /* إدخال بعد */
    STAILQ_INSERT_AFTER(&head, n1, n2, entries);
    STAILQ_REMOVE(&head, n2, entry, entries); /* حذف */
    free(n2);
    n3 = STAILQ_FIRST(&head);
    STAILQ_REMOVE_HEAD(&head, entries);     /* حذف من البداية */
    free(n3);
    n1 = STAILQ_FIRST(&head);
    n1->data = 0;
    for (unsigned int i = 1; i < 5; i++) {
        n1 = malloc(sizeof(struct entry));
        STAILQ_INSERT_HEAD(&head, n1, entries);
        n1->data = i;
    }
