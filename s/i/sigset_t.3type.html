    int      si_signo;  /* رقم الإشارة */

sigset_t
يتضمن: <signal.h>. بدلاً من ذلك، <spawn.h>، أو <sys/select.h>.

هذا نوع يمثل مجموعة من الإشارات. وفقًا لـ POSIX، يجب أن يكون نوع عدد صحيح أو هيكل.

يتوافق مع: POSIX.1-2001 وما بعده.

انظر أيضاً: epoll_pwait(2), ppoll(2), pselect(2), sigaction(2), signalfd(2), sigpending(2), sigprocmask(2), sigsuspend(2), sigwaitinfo(2), signal(7)

ملاحظات

يجب أن تحتوي الهياكل الموضحة في صفحة الدليل هذه على الأعضاء الموضحة في تعريفها على الأقل، بترتيب غير محدد.

معظم أنواع الأعداد الصحيحة الموضحة في هذه الصفحة لا تحتوي على معدل طول مطابق لعائلات الوظائف printf(3) و scanf(3). لطباعة قيمة من نوع عدد صحيح لا يحتوي على معدل طول، يجب تحويلها إلى intmax_t أو uintmax_t عن طريق إلقاء صريح. لمسح متغير من نوع عدد صحيح لا يحتوي على معدل طول، يجب استخدام متغير مؤقت وسيط من النوع intmax_t أو uintmax_t. عند النسخ من المتغير المؤقت إلى متغير الوجهة، قد تحدث فيضانات. إذا كان النوع يحتوي على حدود عليا وسفلى، فيجب على المستخدم التحقق من أن القيمة ضمن تلك الحدود، قبل نسخ القيمة فعليًا. يوضح المثال أدناه كيف يجب إجراء هذه التحويلات.

الاصطلاحات المستخدمة في هذه الصفحة

في قسم "يتوافق مع"، نهتم فقط بـ C99 والإصدارات الأحدث و POSIX.1-2001 والإصدارات الأحدث. قد تكون بعض الأنواع محددة في إصدارات سابقة من أحد هذه المعايير، ولكن من أجل البساطة، نتجاهل التفاصيل من المعايير السابقة.

في قسم "يتضمن"، نلاحظ أولاً رؤوس "أساسية" التي تحدد النوع وفقًا لمعايير C أو POSIX.1. تحت "بدلاً من ذلك"، نلاحظ رؤوسًا إضافية تحددها المعايير بأنها يجب أن تحدد النوع.

أمثلة

يقوم البرنامج المعروض أدناه بالمسح من سلسلة نصية ويطبع قيمة مخزنة في متغير من نوع عدد صحيح لا يحتوي على معدل طول. يتم استخدام التحويلات المناسبة من وإلى intmax_t، والتحقق المناسب من النطاق، كما هو موضح في قسم الملاحظات أعلاه.

#include <stdint.h>
#include <stdio.h>
#include <stdlib.h>
#include <sys/types.h>
int
main (void)
{
    static const char *const str = "500000 us in half a second";
    suseconds_t us;
    intmax_t    tmp;
    /* امسح الرقم من السلسلة إلى المتغير المؤقت.  */
    sscanf(str, "%jd", &tmp);
    /* تحقق من أن القيمة ضمن النطاق الصالح لـ suseconds_t.  */
    if (tmp < -1 || tmp > 1000000) {
        fprintf(stderr, "قيمة المسح خارج النطاق الصالح!\n");
        exit(EXIT_FAILURE);
    }
    /* انسخ القيمة إلى المتغير suseconds_t 'us'.  */
    us = tmp;
    /* على الرغم من أن suseconds_t يمكن أن تحمل القيمة -1، إلا أن هذا ليس
       عددًا معقولًا من الميكروثواني.  */
    if (us < 0) {
        fprintf(stderr, "يجب ألا تكون قيمة المسح سالبة!\n");
        exit(EXIT_FAILURE);
    }
    /* اطبع القيمة.  */
    printf("هناك %jd ميكروثانية في نصف ثانية.\n",
            (intmax_t) us);
    exit(EXIT_SUCCESS);
}
