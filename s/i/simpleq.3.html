الاسم

simpleq - قائمة مرتبطة بعناصر بسيطة

المكتبة

مكتبة C القياسية (libc)

الملخص

#include <sys/queue.h>

struct simpleq;

struct simpleq_item {
    int data;
    struct simpleq_item *next;
};

struct simpleq_item *
simpleq_alloc(void);

void
simpleq_free(struct simpleq_item *item);

void
simpleq_insert(struct simpleq *queue, struct simpleq_item *item);

struct simpleq_item *
simpleq_pop(struct simpleq *queue);

الوصف

تمثل simpleq قائمة مرتبطة بالبرامج. يتم إدارتها بواسطة دالات simpleq_alloc و simpleq_free و simpleq_insert و simpleq_pop.

يتم تعريف عنصر القائمة كـ struct simpleq_item، والذي يحتوي على حقل بيانات من النوع int وحقل next يشير إلى العنصر التالي في القائمة.

  • simpleq_alloc: تخصص وتهيئة عنصر قائمة جديد.
  • simpleq_free: تحرر عنصر قائمة تم تخصيصه مسبقًا.
  • simpleq_insert: تقوم بإدراج عنصر قائمة في نهاية القائمة.
  • simpleq_pop: تزيل وتُرجع العنصر الأول من القائمة.

قيمة الإرجاع

تُرجع simpleq_alloc مؤشرًا إلى عنصر قائمة جديد، أو NULL إذا فشل التخصيص.

تُرجع simpleq_pop مؤشرًا إلى العنصر الذي تم إزالته، أو NULL إذا كانت القائمة فارغة.

أمثلة

#include <stdio.h>
#include <stdlib.h>
#include <sys/queue.h>

struct my_item {
    int data;
    STAILQ_ENTRY(my_item) entries;
};

int main(void) {
    struct my_item *np;
    struct my_item *n1, *n2;
    STAILQ_HEAD(mylisthead, my_item) head = STAILQ_HEAD_INITIALIZER(head);

    /* Add some items */
    for (int i = 0; i < 5; i++) {
        np = malloc(sizeof(struct my_item));
        if (np == NULL) {
            perror("malloc");
            exit(EXIT_FAILURE);
        }
        np->data = i;
        STAILQ_INSERT_TAIL(&head, np, entries);
    }

    /* Forward traversal */
    STAILQ_FOREACH(np, &head, entries)
        printf("%i\n", np->data);

    /* TailQ deletion */
    n1 = STAILQ_FIRST(&head);
    while (n1 != NULL) {
        n2 = STAILQ_NEXT(n1, entries);
        free(n1);
        n1 = n2;
    }
    STAILQ_INIT(&head);
    exit(EXIT_SUCCESS);
}
