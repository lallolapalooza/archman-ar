الحزمة: sudo

SUDO_LOGSRV.PROTO(5) File Formats Manual SUDO_LOGSRV.PROTO(5)

NAME

sudo_logsrv.proto — بروتوكول خادم تسجيل sudo

DESCRIPTION

بدءًا من الإصدار 1.9.0، يدعم sudo إرسال سجلات الأحداث والإدخال/الإخراج إلى خادم تسجيل. البروتوكول المستخدم مكتوب بلغة Google Protocol Buffers الخاصة بالمجال. يتضمن قسم الأمثلة وصفًا كاملاً للبروتوكول بتنسيق Protocol Buffers.

نظرًا لعدم وجود طريقة لتحديد حدود الرسائل عند استخدام Protocol Buffers، يتم إرسال حجم السلك لكل رسالة مسبقًا مباشرةً قبل الرسالة نفسها كرقم صحيح غير موقع 32 بت بترتيب بايت الشبكة. يُشار إلى هذا باسم "التأطير البادئ بالطول" وهو الطريقة التي تقترحها Google للتعامل مع نقص محددات الرسائل.

يتكون البروتوكول من رسالتين أساسيتين، ClientMessage و ServerMessage، الموصوفتين أدناه. يجب أن يقبل الخادم الرسائل التي يصل حجمها إلى ميغابايت. قد يُرجع الخادم خطأ إذا حاول العميل إرسال رسالة أكبر من ميغابايت.

رسائل العميل

إن ClientMessage هو حاوية تُستخدم لتغليف جميع أنواع الرسائل الممكنة التي يمكن للعميل إرسالها إلى الخادم.

message ClientMessage {
  oneof type {
    AcceptMessage accept_msg = 1;
    RejectMessage reject_msg = 2;
    ExitMessage exit_msg = 3;
    RestartMessage restart_msg = 4;
    AlertMessage alert_msg = 5;
    IoBuffer ttyin_buf = 6;
    IoBuffer ttyout_buf = 7;
    IoBuffer stdin_buf = 8;
    IoBuffer stdout_buf = 9;
    IoBuffer stderr_buf = 10;
    ChangeWindowSize winsize_event = 11;
    CommandSuspend suspend_event = 12;
    ClientHello hello_msg = 13;
  }
}

الرسائل الفرعية المختلفة من نوع ClientMessage التي يمكن للعميل إرسالها إلى الخادم موصوفة أدناه.

TimeSpec

message TimeSpec {
    int64 tv_sec = 1;
    int32 tv_nsec = 2;
}

إن TimeSpec هو المكافئ لبنية POSIX struct timespec، يحتوي على أعضاء الثواني والنانوثواني. العضو tv_sec هو عدد صحيح 64 بت لدعم التواريخ بعد عام 2038.

InfoMessage

message InfoMessage {
  message StringList {
    repeated string strings = 1;
  }
  message NumberList {
    repeated int64 numbers = 1;
  }
  string key = 1;
  oneof value {
    int64 numval = 2;
    string strval = 3;
    StringList strlistval = 4;
    NumberList numlistval = 5;
  }
}

يُستخدم InfoMessage لتمثيل معلومات حول المستخدم المستدعي وكذلك بيئة التنفيذ التي يعمل فيها الأمر على شكل أزواج مفتاح-قيمة. المفتاح هو دائمًا سلسلة نصية ولكن القيمة قد تكون عددًا صحيحًا 64 بت، أو سلسلة نصية، أو مصفوفة من السلاسل النصية، أو مصفوفة من الأعداد الصحيحة 64 بت. يتكون سجل الأحداث من إدخالات InfoMessage. انظر قسم متغيرات سجل الأحداث لمزيد من المعلومات.

ClientHello hello_msg

message ClientHello {
  string client_id = 1;
}

تتكون رسالة ClientHello من معلومات العميل التي قد يتم إرسالها إلى الخادم عند اتصال العميل لأول مرة.

client_id
وصف حر للعميل. يتضمن هذا عادةً اسم وإصدار تطبيق العميل.

AcceptMessage accept_msg

message AcceptMessage {
  TimeSpec submit_time = 1;
  repeated InfoMessage info_msgs = 2;
  bool expect_iobufs = 3;
}

يتم إرسال AcceptMessage بواسطة العميل عندما يُسمح للأمر بموجب سياسة الأمان. تحتوي على الأعضاء التالية:

submit_time
وقت الساعة الجداري عندما تم تقديم الأمر إلى سياسة الأمان.
info_msgs
مصفوفة من InfoMessage تصف المستخدم الذي قدم الأمر بالإضافة إلى بيئة تنفيذ الأمر. تُستخدم هذه المعلومات لإنشاء إدخال سجل أحداث وقد يستخدمها الخادم أيضًا لتحديد مكان وكيفية تخزين سجل الإدخال/الإخراج.
expect_iobufs
تم تعيينه إلى true إذا كان يجب أن يتوقع الخادم رسائل IoBuffer لتتبعها (لتسجيل الإدخال/الإخراج) أو false إذا كان يجب على الخادم تخزين سجل الأحداث فقط.

إذا تم إرسال AcceptMessage، يجب على العميل عدم إرسال RejectMessage أو RestartMessage.

RejectMessage reject_msg

message RejectMessage {
  TimeSpec submit_time = 1;
  string reason = 2;
  repeated InfoMessage info_msgs = 3;
}

يتم إرسال RejectMessage بواسطة العميل عندما يتم رفض الأمر بموجب سياسة الأمان. تحتوي على الأعضاء التالية:

submit_time
وقت الساعة الجداري عندما تم تقديم الأمر إلى سياسة الأمان.
reason
السبب الذي قدمته سياسة الأمان لرفض الأمر.
info_msgs
مصفوفة من InfoMessage تصف المستخدم الذي قدم الأمر بالإضافة إلى بيئة تنفيذ الأمر. تُستخدم هذه المعلومات لإنشاء إدخال سجل أحداث.

إذا تم إرسال RejectMessage، يجب على العميل عدم إرسال AcceptMessage أو RestartMessage.

ExitMessage exit_msg

message ExitMessage {
  TimeSpec run_time = 1;
  int32 exit_value = 2;
  bool dumped_core = 3;
  string signal = 4;
  string error = 5;
}

يتم إرسال ExitMessage بواسطة العميل بعد خروج الأمر أو إنهاءه بإشارة. تحتوي على الأعضاء التالية:

run_time
إجمالي الوقت المنقضي منذ بدء الأمر، محسوبًا باستخدام ساعة رتيبة حيثما أمكن. هذا ليس وقت الساعة الجداري.
exit_value
قيمة خروج الأمر في النطاق 0-255.
dumped_core
صحيح إذا تم إنهاء الأمر بإشارة وتم تفريغ الذاكرة الأساسية.
signal
إذا تم إنهاء الأمر بإشارة، يتم تعيين هذا إلى اسم الإشارة بدون البادئة "SIG". على سبيل المثال، INT، TERM، KILL، SEGV.
error
رسالة من العميل تشير إلى أن الأمر قد تم إنهاؤه بشكل غير متوقع بسبب خطأ.

عند إجراء تسجيل الإدخال/الإخراج، يجب على العميل انتظار commit_point المقابل لـ IoBuffer النهائي قبل إغلاق الاتصال ما لم يتم استلام commit_point النهائي بالفعل.

RestartMessage restart_msg

message RestartMessage {
  string log_id = 1;
  TimeSpec resume_point = 2;
}

يتم إرسال RestartMessage بواسطة العميل لاستئناف إرسال سجل إدخال/إخراج موجود تم مقاطعته سابقًا. يحتوي على الأعضاء التالية:

log_id
الاسم من جانب الخادم لسجل الإدخال/الإخراج الذي تم إرساله سابقًا من الخادم إلى العميل. قد يكون هذا اسم مسار على الخادم أو نوعًا آخر من المعرفات من جانب الخادم.
resume_point
النقطة الزمنية التي يجب استئناف سجل الإدخال/الإخراج بعدها. هذا على شكل TimeSpec يمثل مقدار الوقت منذ بدء الأمر، وليس وقت الساعة الجداري. يجب أن يتوافق resume_point مع commit_point تم إرساله سابقًا من الخادم إلى العميل. إذا تلقى الخادم RestartMessage يحتوي على resume_point لم يره من قبل، فسيتم إرجاع خطأ إلى العميل وسيتم إنهاء الاتصال.

AlertMessage alert_msg

message AlertMessage {
  string message = 1;
  repeated InfoMessage info_msgs = 2;
}

يتم إرسال AlertMessage بواسطة العميل لإعلام الخادم بوجود مشكلة لم تؤد إلى إنهاء الأمر. تحتوي على الأعضاء التالية:

message
رسالة التنبيه.
info_msgs
مصفوفة من InfoMessage تصف المستخدم الذي قدم الأمر بالإضافة إلى بيئة تنفيذ الأمر.

IoBuffer

message IoBuffer {
  TimeSpec timestamp = 1;
  bytes buffer = 2;
  bool is_commit_point = 3;
}

يمثل IoBuffer المخزن المؤقت لبيانات الإدخال/الإخراج التي تم التقاطها. تحتوي على الأعضاء التالية:

timestamp
الطابع الزمني لبيانات الإدخال/الإخراج.
buffer
بيانات الإدخال/الإخراج الفعلية.
is_commit_point
يتم تعيينه إلى true إذا كان المخزن المؤقت يمثل نقطة التزام، مما يعني أن جميع البيانات السابقة قد تم تسجيلها. تُستخدم هذه لتحديد نقاط الاستئناف.

ChangeWindowSize winsize_event

message ChangeWindowSize {
  int32 rows = 1;
  int32 cols = 2;
  int32 x = 3;
  int32 y = 4;
}

يتم إرسال ChangeWindowSize بواسطة العميل لإعلام الخادم بأن حجم نافذة الجهاز الظاهري قد تغير.

rows
عدد الصفوف الجديدة.
cols
عدد الأعمدة الجديدة.
x
موضع x الجديد للنافذة (خاص بـ xterm).
y
موضع y الجديد للنافذة (خاص بـ xterm).

CommandSuspend suspend_event

message CommandSuspend {
  bool suspended = 1;
}

يتم إرسال CommandSuspend بواسطة العميل للإشارة إلى ما إذا كان الأمر قد تم تعليقه أو استئنافه.

suspended
صحيح إذا تم تعليق الأمر، وخطأ إذا تم استئنافه.

رسائل الخادم

إن ServerMessage هو حاوية تُستخدم لتغليف جميع أنواع الرسائل الممكنة التي يمكن للخادم إرسالها إلى العميل.

message ServerMessage {
  oneof type {
    LogIdMessage log_id_msg = 1;
    CommitPointMessage commit_point_msg = 2;
    ErrorMessage error_msg = 3;
    ServerHello hello_msg = 4;
  }
}

الرسائل الفرعية المختلفة من نوع ServerMessage التي يمكن للخادم إرسالها إلى العميل موصوفة أدناه.

ServerHello hello_msg

message ServerHello {
  string server_id = 1;
  repeated string supported_api_versions = 2;
}

رسالة ServerHello يتم إرسالها بواسطة الخادم كاستجابة لـ ClientHello. تحتوي على معرف الخادم وإصدارات واجهة برمجة التطبيقات التي يدعمها.

server_id
معرف حر للخادم.
supported_api_versions
قائمة بالسلاسل النصية التي تمثل إصدارات واجهة برمجة التطبيقات التي يدعمها الخادم.

LogIdMessage log_id_msg

message LogIdMessage {
  string log_id = 1;
}

يتم إرسال LogIdMessage بواسطة الخادم عند إنشاء معرف فريد لسجل الإدخال/الإخراج.

log_id
الاسم من جانب الخادم لسجل الإدخال/الإخراج. قد يكون هذا اسم مسار على الخادم أو معرفًا آخر من جانب الخادم.

CommitPointMessage commit_point_msg

message CommitPointMessage {
  TimeSpec commit_point = 1;
}

يتم إرسال CommitPointMessage بواسطة الخادم للإشارة إلى نقطة في سجل الإدخال/الإخراج تم حفظها بشكل دائم. تُستخدم هذه النقاط كنقاط استئناف محتملة.

commit_point
نقطة الالتزام في شكل TimeSpec تمثل الوقت المنقضي منذ بدء الأمر.

ErrorMessage error_msg

message ErrorMessage {
  string error = 1;
}

يتم إرسال ErrorMessage بواسطة الخادم للإشارة إلى خطأ حدث على جانب الخادم. يؤدي هذا إلى إنهاء الاتصال.

error
رسالة خطأ من الخادم.

EXAMPLES

يحتوي ملف /etc/sudo.conf على مثال لتعريف بروتوكول Protocol Buffers:

sudo_logsrv.proto

يمكنك تضمين هذا الملف في ملفات .proto الأخرى باستخدام البناء import.

EVENT LOG VARIABLES

تُستخدم متغيرات سجل الأحداث لتخزين معلومات حول الأمر الذي تم تشغيله. يتم تمثيل هذه المعلومات كأزواج مفتاح-قيمة في رسائل InfoMessage. فيما يلي قائمة ببعض متغيرات سجل الأحداث الشائعة:

  • COMMAND: الأمر الكامل الذي تم تشغيله.
  • ARGUMENTS: قائمة بالوسائط التي تم تمريرها إلى الأمر.
  • CWD: دليل العمل الحالي عند تشغيل الأمر.
  • USER: اسم المستخدم الذي قام بتشغيل الأمر.
  • HOME: الدليل الرئيسي للمستخدم.
  • SHELL: الغلاف المستخدم لتشغيل الأمر.
  • HOST: اسم المضيف الذي تم تشغيل الأمر عليه.
  • TTY: الجهاز الطرفي الذي تم استخدامه.
  • REMOTEHOST: اسم المضيف البعيد إذا تم تشغيل الأمر عن بعد.
  • COLOR: يشير إلى ما إذا كان لون الإخراج مدعومًا.
  • PID: معرف العملية للأمر.
  • SESSION_ID: معرف الجلسة.
  • LOGIN_ID: معرف تسجيل الدخول.
  • KERNEL_VERSION: إصدار النواة.
  • Os: اسم نظام التشغيل.
  • ARCHITECTURE: بنية النظام.
  • PLUGIN_NAME: اسم المكون الإضافي المستخدم.
  • PLUGIN_OPTIONS: خيارات المكون الإضافي المستخدم.
  • LOGIN_USER: المستخدم الذي قام بتسجيل الدخول.
  • LOGIN_GROUP: المجموعة التي قام بتسجيل الدخول إليها.
  • LOGIN_GROUPS: قائمة المجموعات التي ينتمي إليها المستخدم.
  • LOGIN_HOMEDIR: الدليل الرئيسي للمستخدم المسجل.
  • LOGIN_SHELL: الغلاف الخاص بالمستخدم المسجل.
  • Login_REALM: مجال تسجيل الدخول.
  • LOGIN_TTY: الجهاز الطرفي للمستخدم المسجل.
  • LOGIN_REMOTEHOST: اسم المضيف البعيد للمستخدم المسجل.
  • LOGIN_SSHHOSTKEY FINGERPRINT: بصمة مفتاح SSH لمضيف تسجيل الدخول.
  • LOGIN_SSHHOSTKEY ALGORITHM: خوارزمية مفتاح SSH لمضيف تسجيل الدخول.
  • LOGIN_SSHHOSTKEY BASE64: بصمة مفتاح SSH الأساسية 64 لمضيف تسجيل الدخول.

يمكن أن تتضمن سياسات sudo تكوين متغيرات سجل أحداث إضافية.

DESCRIPTION

The following InfoMessage entries are required:

Description Type Command Run User Submit Host Submit User
command string command that was submitted
runuser string name of user the command was run as
submithost string name of host the command was submitted on
submituser string name of user submitting the command

The following InfoMessage entries are recognized, but not required:

Key Type Description
clientargv StringList client's original argument vector
clientpid int64 client's process ID
clientppid int64 client's parent process ID
clientsid int64 client's terminal session ID
columns int64 number of columns in the terminal
lines int64 number of lines in the terminal
runargv StringList argument vector of command to run
runchroot string root directory of command to run
runcwd string running command's working directory
runenv StringList the running command's environment
rungid int64 primary group-ID of the command
rungids NumberList supplementary group-IDs for the command
rungroup string primary group name of the command
rungroups StringList supplementary group names for the command
runuid int64 run user's user-ID
submitcwd string submit user's current working directory
submitenv StringList the submit user's environment
submitgid int64 submit user's primary group-ID
submitgids NumberList submit user's supplementary group-IDs
submitgroup string submitting user's primary group name
submitgroups StringList submit user's supplementary group names
submituid int64 submit user's user-ID
ttyname string terminal the command was submitted from

The server must accept other variables not listed above but may ignore them.

EXAMPLES

The Protocol Buffers description of the log server protocol, using "proto3" syntax, is included in full below.

syntax = "proto3";

/*
 * Client message to the server.  Messages on the wire are
 * prefixed with a 32-bit size in network byte order.
 */
message ClientMessage {
  oneof type {
    AcceptMessage accept_msg = 1;
    RejectMessage reject_msg = 2;
    ExitMessage exit_msg = 3;
    RestartMessage restart_msg = 4;
    AlertMessage alert_msg = 5;
    IoBuffer ttyin_buf = 6;
    IoBuffer ttyout_buf = 7;
    IoBuffer stdin_buf = 8;
    IoBuffer stdout_buf = 9;
    IoBuffer stderr_buf = 10;
    ChangeWindowSize winsize_event = 11;
    CommandSuspend suspend_event = 12;
  }
}

/* Equivalent of POSIX struct timespec */
message TimeSpec {
    int64 tv_sec = 1;		/* seconds */
    int32 tv_nsec = 2;		/* nanoseconds */
}

/* I/O buffer with keystroke data */
message IoBuffer {
  TimeSpec delay = 1;		/* elapsed time since last record */
  bytes data = 2;		/* keystroke data */
}

/*
 * Key/value pairs, like Privilege Manager struct info.
 * The value may be a number, a string, or a list of strings.
 */
message InfoMessage {
  message StringList {
    repeated string strings = 1;
  }
  message NumberList {
    repeated int64 numbers = 1;
  }
  string key = 1;
  oneof value {
    int64 numval = 2;
    string strval = 3;
    StringList strlistval = 4;
    NumberList numlistval = 5;
  }
}

/*
 * Event log data for command accepted by the policy.
 */
message AcceptMessage {
  TimeSpec submit_time = 1;		/* when command was submitted */
  repeated InfoMessage info_msgs = 2;	/* key,value event log data */
  bool expect_iobufs = 3;		/* true if I/O logging enabled */
}

/*
 * Event log data for command rejected by the policy.
 */
message RejectMessage {
  TimeSpec submit_time = 1;		/* when command was submitted */
  string reason = 2;			/* reason command was rejected */
  repeated InfoMessage info_msgs = 3;	/* key,value event log data */
}

/* Message sent by client when command exits. */
/* Might revisit runtime and use end_time instead */
message ExitMessage {
  TimeSpec run_time = 1;	/* total elapsed run time */
  int32 exit_value = 2;		/* 0–255 */
  bool dumped_core = 3;		/* true if command dumped core */
  string signal = 4;		/* signal name if killed by signal */
  string error = 5;		/* if killed due to other error */
}

/* Alert message, policy module-specific. */
message AlertMessage {
  TimeSpec alert_time = 1;		/* time alert message occurred */
  string reason = 2;			/* policy alert error string */
  repeated InfoMessage info_msgs = 3;	/* key,value event log data */
}

/* Used to restart an existing I/O log on the server. */
message RestartMessage {
  string log_id = 1;		/* ID of log being restarted */
  TimeSpec resume_point = 2;	/* resume point (elapsed time) */
}

/* Window size change event. */
message ChangeWindowSize {
  TimeSpec delay = 1;		/* elapsed time since last record */
  int32 rows = 2;		/* new number of rows */
  int32 cols = 3;		/* new number of columns */
}

/* Command suspend/resume event. */
message CommandSuspend {
  TimeSpec delay = 1;		/* elapsed time since last record */
  string signal = 2;		/* signal that caused suspend/resume */
}

/*
 * Server messages to the client.  Messages on the wire are
 * prefixed with a 32-bit size in network byte order.
 */
message ServerMessage {
  oneof type {
    ServerHello hello = 1;	/* server hello message */
    TimeSpec commit_point = 2;	/* cumulative time of records stored */
    string log_id = 3;		/* ID of server-side I/O log */
    string error = 4;		/* error message from server */
    string abort = 5;		/* abort message, kill command */
  }
}

/* Hello message from server when client connects. */
message ServerHello {
  string server_id = 1;		/* free-form server description */
  string redirect = 2;		/* optional redirect if busy */
  repeated string servers = 3;	/* optional list of known servers */
}

sudo_logsrv.proto(5) - صفحة الدليل

NAME

sudo_logsrv.proto - صيغة بروتوكول سجل sudo

DESCRIPTION

الملف /etc/sudo-plugins/sudo-logsrv.proto يحتوي على وصف بصيغة بروتوكول نصي عادي لبنية رسائل سجل sudoers.

هذا الملف اختياري. إذا لم يكن موجودًا، فإن sudo يستخدم صيغة افتراضية. إذا كان موجودًا، فسيتم تفسيره بواسطة sudo log server.

يجب أن يكون الملف قابلاً للقراءة بواسطة المستخدم الذي يقوم بتشغيل sudo log server.

رسائل العميل

تصف بنية الرسائل التي يرسلها عميل sudo إلى خادم السجل.

رسائل الخادم

تصف بنية الرسائل التي يرسلها خادم السجل إلى عميل sudo.

تدفق التحكم في البروتوكول

يصف تسلسل الرسائل المتبادلة بين عميل sudo وخادم السجل.

متغيرات سجل الأحداث

قائمة بالمتغيرات المتاحة التي يمكن تضمينها في رسائل السجل.

أمثلة

أمثلة توضح كيفية استخدام ملف البروتوكول.

انظر أيضًا

sudoers(5)، sudo.conf(5)
