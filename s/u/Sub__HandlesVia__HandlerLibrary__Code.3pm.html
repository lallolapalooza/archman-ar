Sub__HandlesVia__HandlerLibrary__Code(3pm)

الحزمة: perl-sub-handlesvia

Sub::HandlesVia::HandlerLibrary::Code(3) وثائق Perl التي ساهم بها المستخدم Sub::HandlesVia::HandlerLibrary::Code(3)

الاسم

Sub::HandlesVia::HandlerLibrary::Code - مكتبة طرق متعلقة بالكود

الملخص

package My::Class {
  use Moo;
  use Sub::HandlesVia;
  use Types::Standard 'CodeRef';
  has attr => (
    is => 'rwp',
    isa => CodeRef,
    handles_via => 'Code',
    handles => {
      'my_execute' => 'execute',
      'my_execute_list' => 'execute_list',
      'my_execute_method' => 'execute_method',
      'my_execute_method_list' => 'execute_method_list',
      'my_execute_method_scalar' => 'execute_method_scalar',
      'my_execute_method_void' => 'execute_method_void',
      'my_execute_scalar' => 'execute_scalar',
      'my_execute_void' => 'execute_void',
    },
  );
}

الوصف

هذه مكتبة طرق لـ Sub::HandlesVia.

الطرق القابلة للتفويض

execute( @args )

يستدعي مرجع الكود، ويمرر له أي وسائط.

my $coderef = sub { 'code' };
my $object  = My::Class->new( attr => $coderef );

# يستدعي: $coderef->( 1, 2, 3 )
$object->my_execute( 1, 2, 3 );

execute_list( @args )

يستدعي مرجع الكود، ويمرر له أي وسائط، ويجبر سياق القائمة. إذا تم استدعاؤه في سياق قياسي، فإنه يعيد مرجع مصفوفة.

my $context;
my $coderef = sub { $context = wantarray(); 'code' };
my $object  = My::Class->new( attr => $coderef );

# يستدعي: $coderef->( 1, 2, 3 )
my $result = $object->my_execute_list( 1, 2, 3 );

say Dumper( $result );  ## ==> [ 'code' ]
say $context;           ## ==> true

execute_method( @args )

يستدعي مرجع الكود كما لو كان طريقة، ويمرر أي وسائط.

my $coderef = sub { 'code' };
my $object  = My::Class->new( attr => $coderef );

# يستدعي: $coderef->( $object, 1, 2, 3 )
$object->my_execute_method( 1, 2, 3 );

execute_method_list( @args )

يستدعي مرجع الكود كما لو كان طريقة، ويمرر أي وسائط، ويجبر سياق القائمة. إذا تم استدعاؤه في سياق قياسي، فإنه يعيد مرجع مصفوفة.

my $context;
my $coderef = sub { $context = wantarray(); 'code' };
my $object  = My::Class->new( attr => $coderef );

# يستدعي: $coderef->( $object, 1, 2, 3 )
my $result = $object->my_execute_method_list( 1, 2, 3 );

say Dumper( $result );  ## ==> [ 'code' ]
say $context;           ## ==> true

execute_method_scalar( @args )

يستدعي مرجع الكود كما لو كان طريقة، ويمرر أي وسائط، ويجبر سياق قياسي.

my $context;
my $coderef = sub { $context = wantarray(); 'code' };
my $object  = My::Class->new( attr => $coderef );

# يستدعي: $coderef->( $object, 1, 2, 3 )
my $result = $object->my_execute_method_scalar( 1, 2, 3 );

say $result;  ## ==> 'code'
say $context; ## ==> false

execute_method_void( @args )

يستدعي مرجع الكود كما لو كان طريقة، ويمرر أي وسائط، ويجبر سياق فارغ. يعيد undef.

my $context;
my $coderef = sub { $context = wantarray(); 'code' };
my $object  = My::Class->new( attr => $coderef );

# يستدعي: $coderef->( $object, 1, 2, 3 )
my $result = $object->my_execute_method_void( 1, 2, 3 );

say $result;  ## ==> undef
say $context; ## ==> undef

execute_scalar( @args )

يستدعي مرجع الكود، ويمرر له أي وسائط، ويجبر سياق قياسي.

my $context;
my $coderef = sub { $context = wantarray(); 'code' };
my $object  = My::Class->new( attr => $coderef );

# يستدعي: $coderef->( 1, 2, 3 )
my $result = $object->my_execute_scalar( 1, 2, 3 );

say $result;  ## ==> 'code'
say $context; ## ==> false

execute_void( @args )

يستدعي مرجع الكود، ويمرر له أي وسائط، ويجبر سياق فارغ. يعيد undef.

my $context;
my $coderef = sub { $context = wantarray(); 'code' };
my $object  = My::Class->new( attr => $coderef );

# يستدعي: $coderef->( 1, 2, 3 )
my $result = $object->my_execute_void( 1, 2, 3 );

say $result;  ## ==> undef
say $context; ## ==> undef

أمثلة موسعة

استخدام execute_method

يسمح معالج execute_method لفئة بتوفير طرق معينة يمكن تجاوزها بواسطة معلمات في المنشئ.

use strict;
use warnings;
use Data::Dumper;

package My::Processor {
  use Moo;
  use Sub::HandlesVia;
  use Types::Standard qw( Str CodeRef );

  has name => (
    is => 'ro',
    isa => Str,
    default => 'Main Process',
  );

  my $NULL_CODEREF = sub {};

  has _debug => (
    is => 'ro',
    isa => CodeRef,
    handles_via => 'Code',
    handles => { debug => 'execute_method' },
    default => sub { $NULL_CODEREF },
    init_arg => 'debug',
  );

  sub _do_stuff {
    my $self = shift;
    $self->debug( 'continuing process' );
    return;
  }

  sub run_process {
    my $self = shift;
    $self->debug( 'starting process' );
    $self->_do_stuff;
    $self->debug( 'ending process' );
  }
}

my $p1 = My::Processor->new( name => 'First Process' );
$p1->run_process; # لا يوجد إخراج

my @got;
my $p2 = My::Processor->new(
  name => 'Second Process',
  debug => sub {
    my ( $processor, $message ) = @_;
    push @got, sprintf( '%s: %s', $processor->name, $message );
  },
);
$p2->run_process; # إخراج مسجل

my @expected = (
  'Second Process: starting process',
  'Second Process: continuing process',
  'Second Process: ending process',
);
say Dumper( \@got ); ## ==> \@expected
