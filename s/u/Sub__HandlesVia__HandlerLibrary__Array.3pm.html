الوصف

هذه مكتبة من الطرق لـ Sub::HandlesVia.

طرق قابلة للتفويض

"accessor( $index, $value? )"

المعاملات: Int، Optional[Any].

تعمل مثل "get" إذا تم تمرير فهرس فقط، أو "set" إذا تم تمرير فهرس وقيمة.

my $object = My::Class->new( attr => [ 'foo', 'bar', 'baz' ] );
$object->my_accessor( 1, 'quux' );
say Dumper( $object->attr ); ## ==> [ 'foo', 'quux', 'baz' ]
say $object->my_accessor( 2 ); ## ==> 'baz'

all()

جميع العناصر في المصفوفة، في سياق قائمة.

my $object = My::Class->new( attr => [ 'foo', 'bar' ] );
my @list = $object->my_all;
say Dumper( \@list ); ## ==> [ 'foo', 'bar' ]

all_true( $coderef )

المعاملات: CodeRef.

مثل List::Util::all().

any( $coderef )

المعاملات: CodeRef.

مثل List::Util::any().

my $object = My::Class->new( attr => [ 'foo', 'bar', 'baz' ] );
my $truth  = $object->my_any( sub { /a/ } );
say $truth; ## ==> true

apply( $coderef )

المعاملات: CodeRef.

يقوم بتنفيذ الدالة المرجعية (التي يجب أن تعدل $_) على كل عنصر من عناصر المصفوفة؛ يعيد المصفوفة الناتجة في سياق قائمة.

clear()

تفرغ المصفوفة.

my $object = My::Class->new( attr => [ 'foo' ] );
$object->my_clear;
say Dumper( $object->attr ); ## ==> []

count()

عدد العناصر في المصفوفة المشار إليها.

my $object = My::Class->new( attr => [ 'foo', 'bar' ] );
say $object->my_count; ## ==> 2

delete( $index )

المعاملات: Int.

يزيل العنصر المحدد بالفهرس من المصفوفة ويعيده. سيتم "نقل" العناصر اللاحقة.

elements()

جميع العناصر في المصفوفة، في سياق قائمة. (في الأساس نفس "all".)

my $object = My::Class->new( attr => [ 'foo', 'bar' ] );
my @list = $object->my_elements;
say Dumper( \@list ); ## ==> [ 'foo', 'bar' ]

first( $coderef )

المعاملات: CodeRef.

مثل List::Util::first().

my $object = My::Class->new( attr => [ 'foo', 'bar', 'baz' ] );
my $found  = $object->my_first( sub { /a/ } );
say $found; ## ==> 'bar'

first_index( $coderef )

المعاملات: CodeRef.

مثل List::MoreUtils::first_index().

my $object = My::Class->new( attr => [ 'foo', 'bar', 'baz' ] );
my $found  = $object->my_first_index( sub { /z$/ } );
say $found; ## ==> 2

flatten()

جميع العناصر في المصفوفة، في سياق قائمة. (في الأساس نفس "all".)

my $object = My::Class->new( attr => [ 'foo', 'bar' ] );
my @list = $object->my_flatten;
say Dumper( \@list ); ## ==> [ 'foo', 'bar' ]

flatten_deep( $depth? )

المعاملات: Optional[Int].

تسطح المصفوفة المشار إليها إلى قائمة، بما في ذلك أي مصفوفات متداخلة. (لديها إمكانية حدوث حلقة لا نهائية.)

my $object = My::Class->new( attr => [ 'foo', [ 'bar', [ 'baz' ] ] ] );
say Dumper( [ $object->my_flatten_deep ] ); ## ==> [ 'foo', 'bar', 'baz' ]
my $object2 = My::Class->new( attr => [ 'foo', [ 'bar', [ 'baz' ] ] ] );
say Dumper( [ $object->my_flatten_deep(1) ] ); ## ==> [ 'foo', 'bar', [ 'baz' ] ]

for_each( $coderef )

المعاملات: CodeRef.

طريقة قابلة للسلسلة تنفذ الدالة المرجعية على كل عنصر من عناصر المصفوفة. سيتم تمرير قيمتين إلى الدالة المرجعية: العنصر وفهرسه.

my $object = My::Class->new( attr => [ 'foo', 'bar', 'baz' ] );
$object->my_for_each( sub { say "Item $_[1] is $_[0]." } );

for_each_pair( $coderef )

المعاملات: CodeRef.

طريقة قابلة للسلسلة تنفذ الدالة المرجعية على كل زوج من العناصر في المصفوفة. سيتم تمرير العنصرين إلى الدالة المرجعية.

المعاملات: Int.

يعيد عنصرًا واحدًا من المصفوفة بواسطة الفهرس.

my $object = My::Class->new( attr => [ 'foo', 'bar', 'baz' ] );
say $object->my_get(  0 ); ## ==> 'foo'
say $object->my_get(  1 ); ## ==> 'bar'

Sub::HandlesVia::HandlerLibrary::Array(3pm)

NAME

Sub::HandlesVia::HandlerLibrary::Array - Array handling library

SYNOPSIS

  package My::Class {
    use Moo;
    use Sub::HandlesVia;
    use Sub::HandlesVia::HandlerLibrary::Array qw( HandleQueue );

    has things => (
      is          => 'ro',
      handles_via => 'Array',
      handles     => HandleQueue,
      default     => sub { [] },
    );
  }

DESCRIPTION

This library provides handlers for array attributes.

DELEGATABLE METHODS

clear()

Clears the array.

delete_at( @indexes )

Deletes elements from the array at the given indexes.

delete_head( @indexes )

Deletes elements from the array at the given indexes.

delete_list( @indexes )

Deletes elements from the array at the given indexes.

delete_tail( @indexes )

Deletes elements from the array at the given indexes.

dequeue( $n )

Removes and returns the first $n elements from the array in list context. Returns undef in scalar context.

duplicates_only()

Returns the array filtered to contain only duplicate elements; can be called in list context or scalar context and will return an arrayref in the latter case.

duplicates_only_in_place()

Filters the array to contain only duplicate elements, and changes the attribute to point to the filtered array.

empty()

Returns true if the array is empty, false otherwise.

exists( $index )

Returns true if an element exists at the given index.

extend( @values )

Adds elements to the end of the array.

my $object = My::Class->new( attr => [ 'foo' ] );
$object->my_extend( 'bar', 'baz' );
say Dumper( $object->attr ); ## ==> [ 'foo', 'bar', 'baz' ]

fill( $size, $_value )

Resizes the array to $size, filling new elements with $_value.

flush()

Clears the array.

for_each( $_coderef )

Calls the subroutine $_coderef for each element of the array, passing the element as the only argument.

head( $n )

Returns the first $n elements of the array in list context.

insert_after( $element | $_coderef, @values )

Inserts elements into the array after the first occurrence of $element, or after the element for which $_coderef returns true.

insert_before( $element | $_coderef, @values )

Inserts elements into the array before the first occurrence of $element, or before the element for which $_coderef returns true.

is_empty()

Returns true if the array is empty, false otherwise.

pop( $n )

Removes and returns the last $n elements from the array in list context. Returns undef in scalar context.

pop_all()

Removes and returns all elements from the array.

push( @values )

Adds elements to the end of the array.

my $object = My::Class->new( attr => [ 'foo' ] );
$object->my_push( 'bar', 'baz' );
say Dumper( $object->attr ); ## ==> [ 'foo', 'bar', 'baz' ]

queue( $n )

Returns the first $n elements of the array in list context.

peek( $n )

Returns the last $n elements of the array in list context.

shift( $n )

Removes and returns the first $n elements from the array in list context. Returns undef in scalar context.

shift_all()

Removes and returns all elements from the array.

size()

Returns the number of elements in the array.

slice( @indexes )

Returns the elements of the array at the specified indexes.

splice( $offset, $length, @values )

Replaces $length elements starting at $offset with @values.

tail( $n )

Returns the last $n elements of the array in list context.

uniq()

Returns the array filtered to remove duplicates; can be called in list context or scalar context and will return an arrayref in the latter case.

uniq_in_place()

Filters the array to remove duplicates, and changes the attribute to point to the filtered array.

uniqnum()

Returns the array filtered to remove duplicates numerically; can be called in list context or scalar context and will return an arrayref in the latter case.

uniqnum_in_place()

Filters the array to remove duplicates numerically, and changes the attribute to point to the filtered array.

uniqstr()

Returns the array filtered to remove duplicates stringwise; can be called in list context or scalar context and will return an arrayref in the latter case.

uniqstr_in_place()

Filters the array to remove duplicates stringwise, and changes the attribute to point to the filtered array.

unshift( @values )

Adds an element to the start of the array.

my $object = My::Class->new( attr => [ 'foo' ] );
$object->my_unshift( 'bar', 'baz' );
say Dumper( $object->attr ); ## ==> [ 'bar', 'baz', 'foo' ]

SHORTCUT CONSTANTS

This module provides some shortcut constants for indicating a list of delegations.

package My::Class {
  use Moo;
  use Sub::HandlesVia;
  use Sub::HandlesVia::HandlerLibrary::Array qw( HandleQueue );

  has things => (
    is          => 'ro',
    handles_via => 'Array',
    handles     => HandleQueue,
    default     => sub { [] },
  );
}

These shortcuts can be combined using the " | " operator.

has things => (
  is          => 'ro',
  handles_via => 'Array',
  handles     => HandleQueue | HandleStack,
  default     => sub { [] },
);

"HandleQueue"

Creates delegations named like "things_is_empty", "things_size", "things_enqueue", "things_dequeue", and "things_peek".

"HandleStack"

Creates delegations named like "things_is_empty", "things_size", "things_push", "things_pop", and "things_peek".

EXTENDED EXAMPLES

Using for_each

use strict;
use warnings;

package My::Plugin {
  use Moo::Role;
  sub initialize {}
  sub finalize {}
}

package My::Processor {
  use Moo;
  use Sub::HandlesVia;
  use Types::Standard qw( ArrayRef ConsumerOf );

  has plugins => (
    is => 'ro',
    isa => ArrayRef[ ConsumerOf['My::Plugin'] ],
    handles_via => 'Array',
    handles => {
      add_plugin => 'push',
      plugin_do => 'for_each',
    },
    default => sub { [] },
  );

  sub _do_stuff {
    return;
  }

  sub run_process {
    my ( $self, @args ) = @_;
    $self->plugin_do( sub {
      my $plugin = shift;
      $plugin->initialize( $self, @args );
    } );
    $self->_do_stuff( @args );
    $self->plugin_do( sub {
      my $plugin = shift;
      $plugin->finalize( $self, @args );
    } );
  }
}

my $p = My::Processor->new();

package My::Plugin::Noisy {
  use Moo; with 'My::Plugin';
  sub initialize {
    my ( $self, $processor, @args ) = @_;
    say "initialize @args"; ## ==> 'initialize 1 2 3'
  }
  sub finalize {
    my ( $self, $processor, @args ) = @_;
    say "finalize @args"; ## ==> 'finalize 1 2 3'
  }
}

$p->add_plugin( My::Plugin::Noisy->new );

$p->run_process( 1, 2, 3 );

Job queue using push and shift

use strict;
use warnings;
use Try::Tiny;

package My::JobQueue {
  use Moo;
  use Sub::HandlesVia;
  use Types::Standard qw( Bool ArrayRef CodeRef HasMethods is_Object );
  use Try::Tiny;

  has auto_requeue => (
    is => 'ro',
    isa => Bool,
    default => 0,
  );

  has jobs => (
    is => 'ro',
    isa => ArrayRef[ CodeRef | HasMethods['run'] ],
    handles_via => 'Array',
    handles => {
      add_job => 'push',
      _get_job => 'shift',
      is_empty => 'is_empty',
    },
    default => sub { [] },
  );

  sub _handle_failed_job {
    my ( $self, $job ) = @_;
    $self->add_job( $job ) if $self->auto_requeue;
  }

  sub run_jobs {
    my $self = shift;
    while ( not $self->is_empty ) {
      my $job = $self->_get_job;
      try {
        is_Object($job) ? $job->run() : $job->();
      }
      catch {
        $self->_handle_failed_job( $job );
      };
    }
  }
}

my $q = My::JobQueue->new();

my $str = '';
$q->add_job( sub { $str .= 'A' } );
$q->add_job( sub { $str .= 'B' } );
$q->add_job( sub { $str .= 'C' } );

$q->run_jobs;

say $str; ## ==> 'ABC'

# Attempt to push invalid value on the queue
#
try {
  $q->add_job( "jobs cannot be strings" );
}
catch {
  say $q->is_empty;  ## ==> true
};
