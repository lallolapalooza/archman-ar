Sub__Exporter__Cookbook(3pm)

الحزمة: perl-sub-exporter

التراخيص: PerlArtistic, GPL

Sub::Exporter::Cookbook(3) وثائق Perl المقدمة من المستخدم Sub::Exporter::Cookbook(3)

الاسم

Sub::Exporter::Cookbook - استخدامات قابلة للاستخدام، توضيحية، أو حيل سخيفة لـ Sub::Exporter

الإصدار

الإصدار 0.991

نظرة عامة

Sub::Exporter هي أداة بسيطة جدًا، ويمكن استخدامها لتحقيق بعض الأهداف البسيطة جدًا. يتم وصف سلوكها الأساسي وتطبيقها الأساسي (أي، تصدير الدوال التقليدي) في Sub::Exporter::Tutorial و Sub::Exporter. تقدم هذه الوثيقة تطبيقات غير فورية أو يمكن أن توضح كيف يمكن استخدام ميزات معينة (للخير أو للشر).

إصدار Perl

يجب أن تعمل هذه المكتبة على إصدارات Perl التي صدرت حتى منذ فترة طويلة. يجب أن تعمل على أي إصدار Perl صدر خلال السنوات الخمس الماضية.

على الرغم من أنها قد تعمل على إصدارات Perl القديمة، إلا أنه لا يُضمن أن الحد الأدنى للإصدار المطلوب لن يزداد. يمكن زيادة الإصدار لأي سبب، ولا يُضمن قبول الترقيعات لتقليل الحد الأدنى لإصدار Perl المطلوب.

الوصفات

تصدير الدوال كروتينات

بينما تصدر وحدات التصدير عادةً دوال ليتم استدعاؤها كدوال، فمن السهل استخدام Sub::Exporter لتصدير دوال يمكن استدعاؤها كدوال في حزمة الاستيراد أو كائناتهم.

إليك مثال بسيط (وساذج):

package Mixin::DumpObj;
use Data::Dumper;
use Sub::Exporter -setup => {
  exports => [ qw(dump) ]
};
sub dump {
  my ($self) = @_;
  return Dumper($self);
}

عندما نكتب فئة كائنات خاصة بنا، يمكننا استيراد dump كدالة، استدعاؤها على النحو التالي:

$object->dump;

بافتراض أن حزمة الاستيراد ستوفر واجهة محددة، يمكن استخدام وحدة تصدير الدوال كبرنامج إضافي بسيط:

package Number::Plugin::Upto;
use Sub::Exporter -setup => {
  into    => 'Number',
  exports => [ qw(upto) ],
  groups  => [ default => [ qw(upto) ] ],
};
sub upto {
  my ($self) = @_;
  return 1 .. abs($self->as_integer);
}

يشير السطر "into" في التكوين إلى أن هذا البرنامج الإضافي سيصدر افتراضيًا في الحزمة "Number"، وليس في الحزمة التي تستخدم "use". يمكن تصديره بأي طريقة، ولكنه يعمل طالما أن الوجهة توفر دالة "as_integer" كما هو متوقع. لاستيراده إلى وجهة مختلفة، ما عليك سوى كتابة:

use Number::Plugin::Upto { into => 'Quantity' };

خلط سلوك خارجي معقد

عندما نصدر دوال لاستخدامها كدوال (انظر أعلاه)، فإن خيارًا قويًا جدًا هو تصدير دوال تحتفظ بمرجع محيط بوحدة المصدر. هذا يسمح للمستخدم باستيراد دالة واحدة يتم تنفيذها من حيث حزمة كاملة ومنظمة بشكل جيد.

إليك مثال صغير جدًا:

package Data::Analyzer;
use Sub::Exporter -setup => {
  exports => [ analyze => \'_generate_analyzer' ],
};
sub _generate_analyzer {
  my ($mixin, $name, $arg, $col) = @_;
  return sub {
    my ($self) = @_;
    my $values = [ $self->values ];
    my $analyzer = $mixin->new($values);
    $analyzer->perform_analysis;
    $analyzer->aggregate_results;
    return $analyzer->summary;
  };
}

إذا تم استيراده بواسطة أي حزمة توفر دالة "values"، فإن هذا البرنامج الإضافي سيوفر دالة "analyze" واحدة تعمل كواجهة بسيطة لمجموعة من السلوكيات الأكثر تعقيدًا.

الأهم من ذلك، لأن القيمة $mixin ستكون المستدعي الذي تم استدعاء "import" عليه، يمكن أن ترث فئة "Data::Analyzer" ويتم استبدال أجزاء فردية فقط من السلوك المعقد، مما يجعل من السهل كتابة أدوات معقدة قابلة للوراثة بنقاط دخول بسيطة للواجهات الخارجية.

تصدير الثوابت

بينما Sub::Exporter لا يشارك في تصدير الثوابت، فمن السهل تصدير الثوابت باستخدام إحدى وحدات أخواته، Package::Generator.

package Important::Constants;

use Sub::Exporter -setup => {
  collectors => [ constants => \'_set_constants' ],
};

sub _set_constants {
  my ($class, $value, $data) = @_;

  Package::Generator->assign_symbols(
    $data->{into},
    [
      MEANING_OF_LIFE => \42,
      ONE_TRUE_BASE   => \13,
      FACTORS         => [ 6, 9 ],
    ],
  );
  return 1;
}

بعد ذلك، يمكن لشخص ما كتابة:

use Important::Constants 'constants';

print "The factors @FACTORS produce $MEANING_OF_LIFE in $ONE_TRUE_BASE.";

(يجب تصدير الثوابت من خلال جامع، لأنها فعليًا تعدل حزمة الاستيراد بطريقة مختلفة عن تثبيت الدوال.)

تعديل @ISA الخاص بالمستورد

من السهل إنشاء جامع يغير وراثة حزمة الاستيراد:

use Sub::Exporter -setup => {
  collectors => { -base => \'_make_base' },
};
sub _make_base {
  my ($class, $value, $data) = @_;
  my $target = $data->{into};
  push @{"$target\::ISA"}, $class;
}

بعد ذلك، يمكن لمستخدم فئتك كتابة:

use Some::Class -base;

وتصبح فئة فرعية. يمكن أن يكون هذا مفيدًا جدًا في بناء، على سبيل المثال، وحدة تساعد في بناء البرامج الإضافية. قد نرغب في استيراد بعض الأدوات المساعدة، ولكننا نريد أيضًا أن نرث سلوكًا من فئة أساسية لبرنامج إضافي؛

package Framework::Util;
use Sub::Exporter -setup => {
  exports    => [ qw(log global_config) ],
  groups     => [ _plugin => [ qw(log global_config) ]
  collectors => { '-plugin' => \'_become_plugin' },
};
sub _become_plugin {
  my ($class, $value, $data) = @_;
  my $target = $data->{into};
  push @{"$target\::ISA"}, $class->plugin_base_class;
  push @{ $data->{import_args} }, '-_plugin';
}

الآن، يمكنك كتابة برنامج إضافي على النحو التالي:

package Framework::Plugin::AirFreshener;
use Framework::Util -plugin;
