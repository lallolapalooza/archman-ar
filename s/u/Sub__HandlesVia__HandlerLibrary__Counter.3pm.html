Sub__HandlesVia__HandlerLibrary__Counter(3pm)

الحزمة: perl-sub-handlesvia

التراخيص: PerlArtistic, GPL

Sub::HandlesVia::HandlerLibrary::Counter(3) وثائق Perl التي ساهم بها المستخدم Sub::HandlesVia::HandlerLibrary::Counter(3)

الاسم

Sub::HandlesVia::HandlerLibrary::Counter - مكتبة طرق متعلقة بالعداد

الملخص

package My::Class {
  use Moo;
  use Sub::HandlesVia;
  use Types::Standard 'Int';
  has attr => (
    is => 'rwp',
    isa => Int,
    handles_via => 'Counter',
    handles => {
      'my_dec' => 'dec',
      'my_inc' => 'inc',
      'my_reset' => 'reset',
      'my_set' => 'set',
    },
  );
}

الوصف

هذه مكتبة طرق لـ Sub::HandlesVia.

الطرق القابلة للتفويض

dec( $amount? )

الوسائط: Optional[Int].

يقلل العداد بمقدار $amount، أو بمقدار 1 إذا لم يتم إعطاء قيمة.

my $object = My::Class->new( attr => 10 );
$object->my_dec;
$object->my_dec;
say $object->attr; ## ==> 8
$object->my_dec( 5 );
say $object->attr; ## ==> 3

inc( $amount? )

الوسائط: Optional[Int].

يزيد العداد بمقدار $amount، أو بمقدار 1 إذا لم يتم إعطاء قيمة.

my $object = My::Class->new( attr => 0 );
$object->my_inc;
$object->my_inc;
say $object->attr; ## ==> 2
$object->my_inc( 3 );
say $object->attr; ## ==> 5

reset()

يضبط العداد إلى قيمته الافتراضية، أو 0 إذا لم يكن له قيمة افتراضية.

my $object = My::Class->new( attr => 10 );
$object->my_reset;
say $object->attr; ## ==> 0

set( $value )

الوسائط: Int.

يضبط العداد إلى القيمة المعطاة.

my $object = My::Class->new( attr => 0 );
$object->my_set( 5 );
say $object->attr; ## ==> 5
