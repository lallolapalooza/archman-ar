Sub__HandlesVia__Declare(3pm)

الحزمة: perl-sub-handlesvia

التراخيص: PerlArtistic, GPL

Sub::HandlesVia::Declare(3) وثائق Perl من مساهمي المستخدم Sub::HandlesVia::Declare(3)

الاسم

Sub::HandlesVia::Declare - تعريف التفويضات في وقت الترجمة

الملخص

use Sub::HandlesVia::Declare( $attr, $via => %delegations );

هذا يعادل تقريبًا ما يلي:

use Sub::HandlesVia qw(delegations);

BEGIN {
  delegations(
    attribute    => $attr,
    handles_via  => $via,
    handles      => \%delegations,
  );
};

باستثناء أنه لا يستورد الدالة "delegations" إلى مساحة اسمك.

الوصف

مفيد لـ Object::Pad وهو لطيف إلى حد ما لـ Class::Tiny. بشكل أساسي أي مُنشئ فئة يقوم بعمله في وقت الترجمة.

Object::Pad

use Object::Pad;

class Kitchen {
  has @foods;
  use Sub::HandlesVia::Declare '@foods', Array => (
    all_foods => 'all',
    add_food  => 'push',
  );
}

إذا بدأ سمة بـ '@' أو '%'، يمكن حذف $via.

use Object::Pad;

class Kitchen {
  has @foods;
  use Sub::HandlesVia::Declare '@foods', (
    all_foods => 'all',
    add_food  => 'push',
  );
}

Class::Tiny

package Kitchen;
use Class::Tiny {
  foods  => sub { [] },
  drinks => sub { [ 'water' ] },
};
use Sub::HandlesVia::Declare 'foods', Array => (
  all_foods  => 'all',
  add_food   => 'push',
);
use Sub::HandlesVia::Declare 'drinks', Array => (
  all_drinks => 'all',
  add_drink  => 'push',
);
