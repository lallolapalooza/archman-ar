Sub__Exporter(3pm)

الحزمة: perl-sub-exporter

التراخيص: PerlArtistic, GPL

Sub::Exporter(3) توثيق بيرل من مساهمة المستخدم Sub::Exporter(3)

الاسم

Sub::Exporter - مصدر تصدير متطور للروتينات المبنية خصيصًا

الإصدار

الإصدار 0.991

ملخص

يجب استخدام Sub::Exporter في مكانين. أولاً، في وحدة تصدير:

# في الوحدة المصدرة:
package Text::Tweaker;
use Sub::Exporter -setup => {
  exports => [
    qw(squish titlecase), # يعمل دائمًا بنفس الطريقة
    reformat => \&build_reformatter, # مولد لبناء الدالة المصدرة
    trim     => \&build_trimmer,
    indent   => \&build_indenter,
  ],
  collectors => [ 'defaults' ],
};

ثم، في وحدة مستوردة:

# في الوحدة المستوردة:
use Text::Tweaker
  'squish',
  indent   => { margin => 5 },
  reformat => { width => 79, justify => 'full', -as => 'prettify_text' },
  defaults => { eol => 'CRLF' };

مع هذا الإعداد، تنتهي الوحدة المستوردة بثلاثة روتينات: "squish"، "indent"، و "prettify_text". تم بناء الاثنين الأخيرين وفقًا لمواصفات المستورد - فهما ليسا مجرد نسخ من الكود الموجود في الحزمة المصدرة.

الوصف

انتبه! إذا لم تكن مألوفًا بـ Exporter أو التصدير، فاقرأ Sub::Exporter::Tutorial أولاً!

لماذا المولدات؟

أكبر فائدة لـ Sub::Exporter مقارنة بالمصدرين الحاليين (بما في ذلك Exporter.pm المنتشر) هي قدرته على بناء rofefs جديدة للتصدير، بدلاً من مجرد تصدير كود مطابق لما هو موجود في الحزمة المصدرة.

إذا حصل مستهلكو وحدتك على روتين يعمل كالتالي:

use Data::Analyze qw(analyze);
my $value = analyze($data, $tolerance, $passes);

وهم يمررون باستمرار مجموعة واحدة أو اثنتين من القيم للوسائط غير $data، يمكن للكود الخاص بك الاستفادة من Sub::Exporter. من خلال كتابة مولد بسيط، يمكنك السماح لهم بالقيام بما يلي بدلاً من ذلك:

use Data::Analyze
  analyze => { tolerance => 0.10, passes => 10, -as => analyze10 },
  analyze => { tolerance => 0.15, passes => 50, -as => analyze50 };
my $value = analyze10($data);

ستبدو الحزمة التي تحتوي على مولد لذلك شيئًا كهذا:

package Data::Analyze;
use Sub::Exporter -setup => {
  exports => [
    analyze => \&build_analyzer,
  ],
};
sub build_analyzer {
  my ($class, $name, $arg) = @_;
  return sub {
    my $data      = shift;
    my $tolerance = shift || $arg->{tolerance};
    my $passes    = shift || $arg->{passes};
    analyze($data, $tolerance, $passes);
  }
}

يجب على مستخدم وحدتك الآن بذل جهد أقل للاستفادة منها - وتذكر، غالبًا ما تكون أنت مستخدمك! الاستثمار في الإجراءات المخصصة هو استثمار في الكسل المستقبلي.

يتجنب هذا أيضًا شكلاً شائعًا من القبح الذي يُرى في العديد من الوحدات: تكوين على مستوى الحزمة. أي، ربما رأيت شيئًا مثل ما سبق تم تنفيذه على النحو التالي:

use Data::Analyze qw(analyze);
$Data::Analyze::default_tolerance = 0.10;
$Data::Analyze::default_passes    = 10;

قد يوفر هذا الوقت، حتى يكون لديك وحدات متعددة تستخدم Data::Analyze. نظرًا لوجود تكوين عام واحد فقط، فإنها تتداخل مع بعضها البعض ويبدأ الكود الخاص بك في الحصول على أخطاء غامضة.

يمكن للمولدات أيضًا السماح لك بتصدير أساليب الفئة ليتم استدعاؤها كروتينات:

package Data::Methodical;
use Sub::Exporter -setup => { exports => { some_method => \&_curry_class } };
sub _curry_class {
  my ($class, $name) = @_;
  sub { $class->$name(@_); };
}

نظرًا للطريقة التي تعمل بها المصدرون و Sub::Exporter، يمكن لأي حزمة ترث من Data::Methodical أن ترث المصدر الخاص بها وتتجاوز "some_method". إذا قام مستخدم باستيراد "some_method" من تلك الحزمة، فسوف يتلقى روتينًا يستدعي الأسلوب على الفئة الفرعية، بدلاً من Data::Methodical نفسها. ضع في اعتبارك أنه إذا قمت بإعادة إعداد Sub::Exporter في حزمة ترث من Data::Methodical، فستقوم بالطبع باستبدال المصدر من Data::Methodical بالكامل. "import" هو أسلوب، ويتم إخفاؤه بنفس الطرق التي يتم بها إخفاء أي أسلوب آخر.

تخصيصات أخرى

إن بناء الروتينات المخصصة باستخدام المولدات ليس هو الطريقة الوحيدة التي تسمح بها Sub::Exporters للكود المستورد بتحسين استخدامه للروتينات المصدرة. يمكن أيضًا إعادة تسميتها لتجنب تضارب الأسماء.

ضع في اعتبارك الكود التالي:

# هذا البرنامج يحدد إلى أي دائرة من الجحيم سيتم إدانتك
use Morality qw(sin virtue); # لحساب الرذيلة
use Math::Trig qw(:all);     # للتعامل مع الدوائر

قام المبرمج عن طريق الخطأ باستيراد روتينين "sin". الحل، في الوحدات المستندة إلى Exporter.pm، سيكون استيراد واحد فقط ثم استدعاء الآخر باسمه المؤهل بالكامل. بدلاً من ذلك، يمكن للمستورد كتابة روتين يفعل ذلك، أو يمكنه العبث بأنواع البيانات.

كم هو أسهل الكتابة:

# هذا البرنامج يحدد إلى أي دائرة من الجحيم سيتم إدانتك
use Morality qw(virtue), sin => { -as => 'offense' };
use Math::Trig -all => { -prefix => 'trig_' };

وتوفر لديك "offense" و "trig_sin" - ناهيك عن "trig_cos" و "trig_tan".

إصدار بيرل

يجب أن تعمل هذه المكتبة على إصدارات بيرل التي تم إصدارها حتى منذ فترة طويلة. يجب أن تعمل على أي إصدار من بيرل تم إصداره في السنوات الخمس الماضية.

على الرغم من أنها قد تعمل على إصدارات أقدم من بيرل، لا يوجد ضمان بأن الحد الأدنى للإصدار المطلوب لن يتم زيادته. قد يتم زيادة الإصدار لأي سبب، ولا يوجد وعد بقبول تعديلات لخفض الحد الأدنى المطلوب من بيرل.

تكوين المصدر

يمكنك تكوين مصدر لحزمتك باستخدام Sub::Exporter على النحو التالي:

package Tools;
use Sub::Exporter
  -setup => { exports => [ qw(function1 function2 function3) ] };

هذه هي أبسط طريقة لاستخدام المصدر، وهي أساسًا مكافئة لهذا:

package Tools;
use base qw(Exporter);
our @EXPORT_OK = qw(function1 function2 function3);

أي استخدام أساسي لـ Sub::Exporter سيبدو كالتالي:

package Tools;
use Sub::Exporter -setup => \%config;

المفاتيح التالية صالحة في %config:

exports - قائمة بالروتينات لتوفيرها للتصدير؛ يمكن أن يتبع كل روتين مولد
groups  - قائمة بالمجموعات لتوفيرها للتصدير؛ يجب أن يتبع كل منها إما (أ) قائمة بالصادرات، مع الوسائط المحتملة لكل تصدير، أو (ب) مولد
collectors - قائمة بالأسماء التي يتم جمع القيم فيها لاستخدامها في توليد الروتين؛ يمكن أن يتبع كل اسم مدقق

بالإضافة إلى الخيارات الأساسية المذكورة أعلاه، قد يتم تمرير بعض الخيارات الأكثر تقدمًا:

into_level - إلى أي مدى في المكدس المستدعي للبحث عن هدف (الافتراضي 0)
into       - هدف صريح (حزمة) للتصدير إليه الروتينات

بمعنى آخر: يقوم Sub::Exporter بتثبيت روتين "import" والذي، عند استدعائه، يصدر الروتينات إلى مساحة اسم المستدعي. الخيارات "into" و "into_level" تغير مكان تثبيت هذه الروتينات المصدرة.

generator  - رد اتصال يستخدم لإنتاج الكود الذي سيتم تثبيته
             الافتراضي: Sub::Exporter::default_generator
installer  - رد اتصال يستخدم لتثبيت الكود الذي أنتجه المولد
             الافتراضي: Sub::Exporter::default_installer

للحصول على معلومات حول كيفية استخدام ردود الاتصال هذه، راجع وثائق "default_generator" و "default_installer".
