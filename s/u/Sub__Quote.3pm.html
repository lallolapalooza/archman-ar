Sub__Quote(3pm)

الحزمة: perl-sub-quote

التراخيص: PerlArtistic, GPL

Sub::Quote(3) وثائق Perl المساهمة من المستخدمين Sub::Quote(3)

الاسم

Sub::Quote - توليد فعّال للروتينات الفرعية عبر تقييم النصوص

الملخص

package Silly;
use Sub::Quote qw(quote_sub unquote_sub quoted_from_sub);
quote_sub 'Silly::kitty', q{ print "meow" };
quote_sub 'Silly::doggy', q{ print "woof" };
my $sound = 0;
quote_sub 'Silly::dagron',
  q{ print ++$sound % 2 ? 'burninate' : 'roar' },
  { '$sound' => \$sound };

وفي مكان آخر:

Silly->kitty;  # meow
Silly->doggy;  # woof
Silly->dagron; # burninate
Silly->dagron; # roar
Silly->dagron; # burninate

الوصف

توفر هذه الحزمة طرقًا فعّالة لتوليد الروتينات الفرعية من النصوص.

الروتينات الفرعية

quote_sub

my $coderef = quote_sub 'Silly::kitty', q{ print "meow" }, { '$x' => \0 };

الوسائط: ?$name، $code، ?\%captures، ?\%options

$name هو الروتين الفرعي الذي سيتم تثبيت مرجع الكود فيه.

$code هو نص سيتم تحويله إلى كود.

"\%captures" هو hashref للمتغيرات التي ستُستخدم في الكود. يجب أن تكون المفاتيح الاسم الكامل للمتغير المراد إتاحته، بما في ذلك الرمز. يجب أن تكون القيم مراجع إلى القيم. ستحتوي المتغيرات على نسخ من القيم. راجع مثال "Silly::dagron" لاستخدام الالتقاطات.

يُصدَّر افتراضيًا.

الخيارات

"no_install"
قيمة منطقية. اضبط هذا الخيار لعدم تثبيت مرجع الكود في الروتين الفرعي الممرر عند إلغاء التأجيل.
"no_defer"
قيمة منطقية. يمنع توليد غلاف Sub::Defer للروتين الفرعي الموسوم. إذا كان الروتين الفرعي سيُستدعى في مرحلة ما، فمن المستحسن ذلك. أما بالنسبة لروتين فرعي يُرجَّح أن يُضمَّن، فلا يُنصح به.
"package"
الحزمة التي سيُقيَّم فيها الروتين الفرعي الموسوم. إذا لم يُحدَّد، ستُستخدم حزمة استدعاء "quote_sub".
"hints"
قيمة $^H المراد استخدامها للكود الذي يُقيَّم. هذا يلتقط إعدادات pragma strict. إذا لم يُحدَّد، ستُستخدم قيمة الاستدعاء الحالي.
"warning_bits"
قيمة "${^WARNING_BITS}" المراد استخدامها للكود الذي يُقيَّم. هذا يلتقط التحذيرات المضبوطة. إذا لم يُحدَّد، ستُستخدم تحذيرات الاستدعاء الحالي.
"hintshash"
قيمة "%^H" المراد استخدامها للكود الذي يُقيَّم. هذا يلتقط إعدادات pragmas الأخرى. إذا لم يُحدَّد، ستُستخدم قيمة الاستدعاء الحالي إن أمكن (في perl 5.10+).
"attributes"
سمات الروتينات الفرعية في perlsub المراد تطبيقها على الروتين الفرعي المولَّد. يجب تحديدها كمصفوفة من المراجع. ستُطبَّق السمات على الروتين الفرعي المولَّد وعلى غلاف التأجيل المؤجل، إن استُخدم أحدها.
"file"
الاسم الظاهري المراد استخدامه للكود الذي يُقيَّم.
"line"
رقم السطر الظاهري المراد استخدامه للكود الذي يُقيَّم.

unquote_sub

my $coderef = unquote_sub $sub;

يستبدل قسرًا روتينًا فرعيًا موسومًا بكود حقيقي.

إذا لم يكن $sub روتينًا فرعيًا موسومًا، فهذه عملية لا تفعل شيئًا.

يُصدَّر افتراضيًا.

quoted_from_sub

my $data = quoted_from_sub $sub;
my ($name, $code, $captures, $compiled_sub) = @$data;

يعيد الوسائط الأصلية إلى quote_sub، بالإضافة إلى النسخة المترجمة إذا كان هذا الروتين الفرعي قد أُزيلت وسمه بالفعل.

ملاحظة: $sub قد يكون النسخة الأصلية أو المترجمة للتسهيل.

يُصدَّر افتراضيًا.

inlinify

my $prelude = capture_unroll '$captures', {
  '$x' => 1,
  '$y' => 2,
}, 4;
my $inlined_code = inlinify q{
  my ($x, $y) = @_;
  print $x + $y . "\n";
}, '$x, $y', $prelude;

يأخذ نص كود، ونص وسائط، ونص كود يعمل كـ "مقدمة"، وقيمة منطقية تشير إلى ما إذا كان يجب توطين الوسائط أم لا.

quotify

my $quoted_value = quotify $value;

ينسخ قيمة مفردة (غير مرجعية) للاستخدام في نص كود. يجب أن تعيد النتيجة إنتاج القيمة الأصلية، بما في ذلك النصوص وundef والأعداد الصحيحة والأعداد العشرية. يجب أن تكون الأعداد العشرية (بما في ذلك اللانهايات وليس رقم) مطابقة تمامًا للأصل، إن أمكن. لا يجب الاعتماد على التنسيق الدقيق للرقم الناتج، لأنه قد يتضمن أعدادًا عشرية سداسية أو تعبيرات رياضية.

capture_unroll

my $prelude = capture_unroll '$captures', {
  '$x' => 1,
  '$y' => 2,
}, 4;

الوسائط: $from، \%captures، $indent

يولّد مقتطف كود مناسبًا للاستخدام كـ "مقدمة" لـ inlinify. $from هو نص سيُستخدم كـ hashref في الكود الناتج. مفاتيح %captures هي أسماء المتغيرات والقيم يتم تجاهلها. $indent هو عدد المسافات لإزاحة النتيجة.

qsub

my $hash = {
 coderef => qsub q{ print "hello"; },
 other   => 5,
};

الوسائط: $code

يعمل تمامًا مثل quote_sub، لكنه يتضمن نموذجًا أوليًا لقبول وسيط واحد فقط. هذا يجعل من الأسهل تضمينه في هياكل التجزئة أو القوائم.

يُصدَّر افتراضيًا.

sanitize_identifier

my $var_name = '$variable_for_' . sanitize_identifier('@name');
quote_sub qq{ print \$${var_name} }, { $var_name => \$value };

الوسائط: $identifier

يعقّم قيمة بحيث يمكن استخدامها في معرّف.

البيئة

SUB_QUOTE_DEBUG

يطبع الكود قبل تقييمه. تُدعم عدة أشكال:

1
تُطبع جميع الروتينات الفرعية.
"/foo/"
يُطبع روتين فرعي إذا كان كوده يطابق التعبير النمطي المعطى.
"simple_identifier"
يُطبع روتين فرعي بالاسم المعطى.
"Full::identifier"
يُطبع روتين فرعي يطابق الاسم الكامل.
"Package::Name::"
تُطبع جميع الروتينات الفرعية في الحزمة المعطاة (بما في ذلك الروتينات الفرعية المجهولة).

تحذيرات

الكثير من هذه المعلومات هو مجرد توليد كود قائم على النصوص، ونتيجة لذلك، تنطبق بعض المحاذير.

return

استدعاء "return" من روتين فرعي موسوم لن يفعل ما هو متوقع. بدلاً من العودة من الكود المحدد في "quote_sub"، سيعود من الدالة التي يُركَّب فيها.

لذلك، عند تمرير:

quote_sub q{  return 1 if $condition; $morecode }

قد يظهر في السياق المطلوب كما يلي:

sub foo {
  <important code a>
  do {
    return 1 if $condition;
    $morecode
  };
  <important code b>
}

والذي سيعود من foo، بينما كل ما كان مطلوبًا هو العودة من

هذا الموقع غير مرتبط بـ Arch Linux ولا تابع له بأي شكل من الأشكال. المحتوى هو ترجمة فقط لصفحات دليل Arch المنشورة على man.archlinux.org.
