Sub__HandlesVia__HandlerLibrary__Number(3pm)

الحزمة: perl-sub-handlesvia

التراخيص: PerlArtistic, GPL

Sub::HandlesVia::HandlerLibrary::Number(3) وثائق Perl مقدمة من المستخدم Sub::HandlesVia::HandlerLibrary::Number(3)

الاسم

Sub::HandlesVia::HandlerLibrary::Number - مكتبة طرق متعلقة بالأرقام

الملخص

package My::Class {
  use Moo;
  use Sub::HandlesVia;
  use Types::Standard 'Num';
  has attr => (
    is => 'rwp',
    isa => Num,
    handles_via => 'Number',
    handles => {
      'my_abs' => 'abs',
      'my_add' => 'add',
      'my_ceil' => 'ceil',
      'my_cmp' => 'cmp',
      'my_div' => 'div',
      'my_eq' => 'eq',
      'my_floor' => 'floor',
      'my_ge' => 'ge',
      'my_get' => 'get',
      'my_gt' => 'gt',
      'my_le' => 'le',
      'my_lt' => 'lt',
      'my_mod' => 'mod',
      'my_mul' => 'mul',
      'my_ne' => 'ne',
      'my_set' => 'set',
      'my_sub' => 'sub',
    },
  );
}

الوصف

هذه مكتبة طرق لـ Sub::HandlesVia.

الطرق القابلة للتفويض

abs()

تجد القيمة المطلقة للرقم الحالي، مع تحديث السمة.

my $object = My::Class->new( attr => -5 );
$object->my_abs;
say $object->attr; ## ==> 5

add( $addend )

الوسائط: Num.

تضيف رقماً إلى الرقم الحالي، مع تحديث السمة.

my $object = My::Class->new( attr => 4 );
$object->my_add( 5 );
say $object->attr; ## ==> 9

ceil()

تجد سقف الرقم الحالي، مع تحديث السمة. مثل "ceil" من المدمجات، ولكن في مكانه.

cmp( $num )

الوسائط: Num.

ترجع "$object->attr <=> $num".

div( $divisor )

الوسائط: Num.

تقسم الرقم الحالي على رقم، مع تحديث السمة.

my $object = My::Class->new( attr => 6 );
$object->my_div( 2 );
say $object->attr; ## ==> 3

eq( $num )

الوسائط: Num.

ترجع "$object->attr == $num".

floor()

تجد أرضية الرقم الحالي، مع تحديث السمة. مثل "floor" من المدمجات، ولكن في مكانه.

ge( $num )

الوسائط: Num.

ترجع "$object->attr >= $num".

get()

ترجع القيمة الحالية للرقم.

my $object = My::Class->new( attr => 4 );
say $object->my_get; ## ==> 4

gt( $num )

الوسائط: Num.

ترجع "$object->attr > $num".

le( $num )

الوسائط: Num.

ترجع "$object->attr <= $num".

lt( $num )

الوسائط: Num.

ترجع "$object->attr < $num".

mod( $divisor )

الوسائط: Num.

تجد باقي قسمة الرقم الحالي على المقسوم عليه، مع تحديث السمة.

my $object = My::Class->new( attr => 5 );
$object->my_mod( 2 );
say $object->attr; ## ==> 1

mul( $factor )

الوسائط: Num.

تضرب الرقم الحالي في رقم، مع تحديث السمة.

my $object = My::Class->new( attr => 2 );
$object->my_mul( 5 );
say $object->attr; ## ==> 10

ne( $num )

الوسائط: Num.

ترجع "$object->attr != $num".

set( $value )

الوسائط: Num.

تعيين الرقم إلى قيمة جديدة.

my $object = My::Class->new( attr => 4 );
$object->my_set( 5 );
say $object->attr; ## ==> 5

sub( $subtrahend )

الوسائط: Num.

تطرح رقماً من الرقم الحالي، مع تحديث السمة.

my $object = My::Class->new( attr => 9 );
$object->my_sub( 6 );
say $object->attr; ## ==> 3
