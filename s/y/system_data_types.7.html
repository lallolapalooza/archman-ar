الاسم

system_data_types - نظرة عامة على أنواع بيانات النظام

الوصف

siginfo_t
تضمين: <signal.h>. بدلاً من ذلك، <sys/wait.h>. 
typedef struct {
    int      si_signo;  /* رقم الإشارة */
    int      si_code;   /* رمز الإشارة */
    pid_t    si_pid;    /* معرف عملية الإرسال */
    uid_t    si_uid;    /* معرف المستخدم الحقيقي لعملية الإرسال */
    void    *si_addr;   /* موقع الذاكرة الذي تسبب في الخطأ */
    int      si_status; /* قيمة الخروج أو الإشارة */
    union sigval si_value;  /* قيمة الإشارة */
} siginfo_t;

sigset_t
تضمين: <signal.h>. بدلاً من ذلك، <spawn.h>، أو <sys/select.h>.

هذا نوع يمثل مجموعة من الإشارات. وفقًا لـ POSIX، يجب أن يكون نوعًا صحيحًا أو هيكليًا.

متوافق مع: POSIX.1-2001 والإصدارات الأحدث.

انظر أيضًا: epoll_pwait(2)، ppoll(2)، pselect(2)، sigaction(2)، signalfd(2)، sigpending(2)، sigprocmask(2)، sigsuspend(2)، sigwaitinfo(2)، signal(7)

ملاحظات

يجب أن تحتوي الهياكل الموصوفة في صفحة الدليل هذه، على الأقل، على الأعضاء الموضحة في تعريفها، بترتيب غير محدد.

معظم الأنواع الصحيحة الموصوفة في هذه الصفحة لا يوجد لها معدل طول مقابل لعائلات الوظائف printf(3) و scanf(3). لطباعة قيمة من نوع صحيح لا يوجد له معدل طول، يجب تحويله إلى intmax_t أو uintmax_t عن طريق تحويل صريح. لمسح متغير من نوع صحيح لا يوجد له معدل طول، يجب استخدام متغير مؤقت وسيط من نوع intmax_t أو uintmax_t. عند النسخ من المتغير المؤقت إلى المتغير الوجهة، قد تتجاوز القيمة الحد. إذا كان النوع له حدود عليا وسفلى، يجب على المستخدم التحقق من أن القيمة ضمن هذه الحدود، قبل نسخ القيمة فعليًا. يوضح المثال أدناه كيفية إجراء هذه التحويلات.

الاصطلاحات المستخدمة في هذه الصفحة

في قسم "متوافق مع"، نهتم فقط بإصدارات C99 وما بعدها و POSIX.1-2001 وما بعدها. قد يتم تحديد بعض الأنواع في إصدارات سابقة من أحد هذين المعيارين، ولكن لتجنب التعقيد، نتجاهل التفاصيل من المعايير السابقة.

في قسم "تضمين"، نذكر أولاً رأس (رؤوس) "أساسي" (رؤوس) التي تحدد النوع وفقًا لمعايير C أو POSIX.1. تحت "بدلاً من ذلك"، نذكر رؤوسًا إضافية تنص المعايير على أنها يجب أن تحدد النوع.

أمثلة

البرنامج المعروض أدناه يمسح من سلسلة نصية ويطبع قيمة مخزنة في متغير من نوع صحيح لا يوجد له معدل طول. يتم استخدام التحويلات المناسبة من وإلى intmax_t، وفحوصات النطاق المناسبة، كما هو موضح في قسم الملاحظات أعلاه.

#include <stdint.h>
#include <stdio.h>
#include <stdlib.h>
#include <sys/types.h>
int
main (void)
{
    static const char *const str = "500000 us in half a second";
    suseconds_t us;
    intmax_t    tmp;
    /* امسح الرقم من السلسلة النصية في المتغير المؤقت.  */
    sscanf(str, "%jd", &tmp);
    /* تحقق من أن القيمة ضمن النطاق الصحيح لـ suseconds_t.  */
    if (tmp < -1 || tmp > 1000000) {
        fprintf(stderr, "Scanned value outside valid range!\n");
        exit(EXIT_FAILURE);
    }
    /* انسخ القيمة إلى المتغير suseconds_t 'us'.  */
    us = tmp;
    /* على الرغم من أن suseconds_t يمكنه استيعاب القيمة -1، إلا أن هذا ليس
       عددًا معقولًا من الميكروثواني.  */
    if (us < 0) {
        fprintf(stderr, "Scanned value shouldn't be negative!\n");
        exit(EXIT_FAILURE);
    }
    /* اطبع القيمة.  */
    printf("There are %jd microseconds in half a second.\n",
            (intmax_t) us);
    exit(EXIT_SUCCESS);
}
