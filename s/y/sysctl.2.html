sysctl(2) دليل استدعاءات النظام sysctl(2)

الاسم

sysctl - قراءة/كتابة معلمات النظام

الملخص

#include <unistd.h>
#include <linux/sysctl.h>
[[deprecated]] int _sysctl(struct __sysctl_args *args);

الوصف

استدعاء النظام هذا لم يعد موجودًا في الأنوية الحالية! انظر ملاحظات.

يقوم استدعاء _sysctl() بقراءة و/أو كتابة معلمات النواة. على سبيل المثال، اسم المضيف، أو الحد الأقصى لعدد الملفات المفتوحة. يكون الوسيط بالشكل


struct __sysctl_args {
    int    *name;    /* متجه الأعداد الصحيحة يصف المتغير */
    int     nlen;    /* عدد عناصر هذا المتجه */
    void   *oldval;  /* 0 أو عنوان لتخزين القيمة القديمة */
    size_t *oldlenp; /* المساحة المتاحة للقيمة القديمة،
                        سيتم الكتابة فوقها بالحجم الفعلي للقيمة القديمة */
    void   *newval;  /* 0 أو عنوان القيمة الجديدة */
    size_t  newlen;  /* حجم القيمة الجديدة */
};

يقوم هذا الاستدعاء بإجراء بحث في بنية شجرية، قد تشبه بنية شجرة الدلائل تحت /proc/sys، وإذا تم العثور على العنصر المطلوب، فإنه يستدعي بعض الروتينات المناسبة لقراءة القيمة أو تعديلها.

قيمة الإرجاع

عند الإكمال بنجاح، يعيد _sysctl() القيمة 0. بخلاف ذلك، يتم إرجاع القيمة -1 ويتم تعيين errno للإشارة إلى الخطأ.

الأخطاء

EACCES
EPERM
لا يوجد إذن بالبحث في أحد "الدلائل" التي تم العثور عليها، أو لا يوجد إذن قراءة حيث كان oldval غير NULL، أو لا يوجد إذن كتابة حيث كان newval غير NULL.
EFAULT
طلب الاستدعاء القيمة السابقة عن طريق تعيين oldval إلى قيمة غير NULL، ولكنه سمح بمساحة صفرية في oldlenp.
ENOTDIR
لم يتم العثور على name.

المعايير

Linux.

التاريخ

Linux 1.3.57. تمت إزالته في Linux 5.5، glibc 2.32.

نشأ في 4.4BSD. Linux فقط لديه نسخة /proc/sys المشابهة، وتختلف مخططات تسمية الكائنات بين Linux و 4.4BSD، لكن تعريف استدعاء sysctl() هو نفسه في كليهما.

ملاحظات

كان استخدام استدعاء النظام هذا غير مستحسن لفترة طويلة: منذ Linux 2.6.24، تؤدي استخدامات استدعاء النظام هذا إلى تحذيرات في سجل النواة، وفي Linux 5.5، تمت إزالة استدعاء النظام هذا أخيرًا. استخدم واجهة /proc/sys بدلاً من ذلك.

لاحظ أنه في الأنوية الأقدم حيث لا يزال استدعاء النظام هذا موجودًا، فإنه متاح فقط إذا تم تكوين النواة مع خيار CONFIG_SYSCTL_SYSCALL. علاوة على ذلك، لا يوفر glibc غلافًا لاستدعاء النظام هذا، مما يستلزم استخدام syscall(2).

أخطاء

تختلف أسماء الكائنات بين إصدارات النواة، مما يجعل استدعاء النظام هذا عديم الفائدة للتطبيقات.

لم يتم توثيق جميع الكائنات المتاحة بشكل صحيح.

لم يعد من الممكن تغيير نظام التشغيل عن طريق الكتابة إلى /proc/sys/kernel/ostype.

أمثلة

#define _GNU_SOURCE
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <sys/syscall.h>
#include <unistd.h>
#include <linux/sysctl.h>
#define NITEMS(arr)  (sizeof(arr) / sizeof((arr)[0]))
int _sysctl(struct __sysctl_args *args);
#define OSNAMESZ 100
int
main(void)
{
    int                   name[] = { CTL_KERN, KERN_OSTYPE };
    char                  osname[OSNAMESZ];
    size_t                osnamelth;
    struct __sysctl_args  args;
    memset(&args, 0, sizeof(args));
    args.name = name;
    args.nlen = NITEMS(name);
    args.oldval = osname;
    args.oldlenp = &osnamelth;
    osnamelth = sizeof(osname);
    if (syscall(SYS__sysctl, &args) == -1) {
        perror("_sysctl");
        exit(EXIT_FAILURE);
    }
    printf("This machine is running %*s\n", (int) osnamelth, osname);
    exit(EXIT_SUCCESS);
}
