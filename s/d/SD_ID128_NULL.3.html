الوصف

sd-id128.h جزء من libsystemd(3) ويوفر واجهات برمجة التطبيقات لإنشاء وتحويل ومقارنة قيم المعرفات ذات 128 بت. قيم المعرفات ذات 128 بت التي تتم معالجتها وإنشاؤها بواسطة واجهات برمجة التطبيقات هذه هي تعميم لمعرفات OSF UUIDs كما هو محدد بواسطة RFC 4122[1] ولكنه يستخدم تنسيق سلسلة أبسط. لا تفرض هذه الدوال أي هيكل على المعرفات المستخدمة، على عكس معرفات OSF UUIDs أو GUIDs من Microsoft، ولكنها متوافقة إلى حد كبير مع تلك الأنواع من المعرفات.

يتم تنفيذ المعرف ذي 128 بت كنوع اتحاد على النحو التالي:

typedef union sd_id128 {
  uint8_t bytes[16];
  uint64_t qwords[2];
} sd_id128_t;

يسمح نوع الاتحاد هذا بالوصول إلى المعرف ذي 128 بت كـ 16 بايت منفصلة أو كلمتين 64 بت. بشكل عام، من الأكثر أمانًا الوصول إلى مكونات المعرف من خلال مصفوفة 8 بت لتجنب مشاكل ترتيب البايتات. هذا الاتحاد مخصص ليتم تمريره حسب القيمة (على عكس التمرير بالإشارة) ويمكن معالجته مباشرة بواسطة العملاء.

تم تعريف عدد قليل من الماكرو للإشارة إلى المعرفات ذات 128 بت وفك ترميزها:

SD_ID128_MAKE() يستخدم لكتابة معرف ثابت في شيفرة المصدر. الاستخدام الشائع هو تعيين اسم لمعرف باستخدام هذا الماكرو:

#define SD_MESSAGE_COREDUMP SD_ID128_MAKE(fc,2e,22,bc,6e,e6,47,b6,b9,07,29,ab,34,a2,50,b1)

SD_ID128_NULL يحدد معرفًا يتكون فقط من بايتات NUL (أي، كل البتات مطفأة).

SD_ID128_ALLF يحدد معرفًا يتكون فقط من بايتات 0xFF (أي، كل البتات مضاءة).

SD_ID128_MAKE_STR() مشابه لـ SD_ID128_MAKE()، ولكنه ينشئ تعبير const char* يمكن استخدامه بشكل ملائم في تنسيقات الرسائل وما شابه ذلك:

#include <stdio.h>
#define SD_MESSAGE_COREDUMP_STR SD_ID128_MAKE_STR(fc,2e,22,bc,6e,e6,47,b6,b9,07,29,ab,34,a2,50,b1)
int main(int argc, char **argv) {
  puts("Match for coredumps: MESSAGE_ID=" SD_MESSAGE_COREDUMP_STR);
}

SD_ID128_CONST_STR() يحول المعرفات الثابتة إلى سلاسل ثابتة للإخراج. الكود المثال التالي سيخرج السلسلة "fc2e22bc6ee647b6b90729ab34a250b1":

int main(int argc, char *argv[]) {
  puts("Match for coredumps: %s", SD_ID128_CONST_STR(SD_MESSAGE_COREDUMP));
}

SD_ID128_FORMAT_STR و SD_ID128_FORMAT_VAL() يستخدمان لتنسيق معرف في سلسلة تنسيق printf(3)، كما هو موضح في المثال التالي:

int main(int argc, char *argv[]) {
  sd_id128_t id;
  id = SD_ID128_MAKE(ee,89,be,71,bd,6e,43,d6,91,e6,c5,5d,eb,03,02,07);
  printf("The ID encoded in this C file is " SD_ID128_FORMAT_STR ".\n", SD_ID128_FORMAT_VAL(id));
  return 0;
}

SD_ID128_UUID_FORMAT_STR و SD_ID128_MAKE_UUID_STR() مشابهان لـ SD_ID128_FORMAT_STR و SD_ID128_MAKE_STR()، ولكنهما يتضمنان شرطات فاصلة لتتوافق مع "تمثيل UUID القياسي[2]". يقومان بتنسيق السلسلة بناءً على قواعد RFC4122[1] Variant 1، أي التحويل من ترتيب بايت Big Endian. هذا يطابق سلوك معظم البنية التحتية الأخرى لنظام Linux. من الأفضل تجنب UUIDs من متغيرات أخرى، لتجنب الغموض غير الضروري. جميع المعرفات ذات 128 بت التي تم إنشاؤها بواسطة واجهات برمجة التطبيقات sd-id128 تتوافق تمامًا مع Variant 1 Version 4 UUIDs، وفقًا لـ RFC 4122.

sd_id128_equal() تقارن معرفين ذوي 128 بت:

int main(int argc, char *argv[]) {
  sd_id128_t a, b, c;
  a = SD_ID128_MAKE(ee,89,be,71,bd,6e,43,d6,91,e6,c5,5d,eb,03,02,07);
  b = SD_ID128_MAKE(f2,28,88,9c,5f,09,44,15,9d,d7,04,77,58,cb,e7,3e);
  c = a;
  assert(sd_id128_equal(a, c));
  assert(!sd_id128_equal(a, b));
  return 0;
}

sd_id128_string_equal() مشابه لـ sd_id128_equal()، ولكن المعرف الأول يتم تنسيقه كسلسلة const char*. تنطبق نفس القيود كما على الوسيط الأول لـ sd_id128_from_string().

sd_id128_is_null() يتحقق مما إذا كان المعرف يتكون فقط من بايتات NUL:

assert(sd_id128_is_null(SD_ID128_NULL));

بالمثل، sd_id128_is_allf() يتحقق مما إذا كان المعرف يتكون فقط من بايتات 0xFF (كل البتات مضاءة):

assert(sd_id128_is_allf(SD_ID128_ALLF));

sd_id128_in_set_sentinel() تأخذ قائمة من المعرفات وتعيد صحيحًا إذا كان الوسيط الأول يساوي أيًا من الوسائط اللاحقة. يتم إنهاء قائمة الوسائط بـ SD_ID128_NULL كعلامة نهاية، والتي يجب أن تكون موجودة.

sd_id128_in_set() هي وظيفة سهلة الاستخدام تأخذ قائمة من المعرفات وتعيد صحيحًا إذا كان الوسيط الأول يساوي أيًا من الوسائط اللاحقة:

int main(int argc, char *argv[]) {
  sd_id12_t a = SD_ID128_MAKE(ee,89,be,71,bd,6e,43,d6,91,e6,c5,5d,eb,03,02,07);
  assert(sd_id128_in_set(a, a));
  assert(sd_id128_in_set(a, a, a));
  assert(!sd_id128_in_set(a));
  assert(!sd_id128_in_set(a,
                          SD_ID128_MAKE(f2,28,88,9c,5f,09,44,15,9d,d7,04,77,58,cb,e7,3e)
                          SD_ID128_MAKE(2f,88,28,5f,9c,44,09,9d,d7,15,77,04,bc,85,7e,e3)
                          SD_ID128_ALLF));
  return 0;
}

sd_id128_in_set() معرفة كـ ماكرو فوق sd_id128_in_set_sentinel()، مما يضيف العلامة SD_ID128_NULL تلقائيًا. نظرًا لأن sd_id128_in_set_sentinel() تستخدم SD_ID128_NULL كعلامة نهاية، فلا يمكن وضع SD_ID128_NULL بخلاف ذلك في قائمة الوسائط.

sd_id128_in_setv() مشابه لـ sd_id128_in_set_sentinel()، ولكنه يأخذ وسيط struct varargs.

يمكن إنشاء معرفات عشوائية جديدة باستخدام أمر new الخاص بـ systemd-id128(1).

انظر sd_id128_to_string(3)، sd_id128_randomize(3) و sd_id128_get_machine(3) للحصول على معلومات حول تطبيقات أخرى.

SD_ID128_NULL(3) — systemd.258.2

NAME

sd_id128_null, sd_id128_string_null, sd_id128_is_null, sd_id128_is_allf, sd_id128_in_setv, sd_id128_in_set_sentinel, sd_id128_in_set - Check if an ID128 is the NULL or all-ones ID, or if it is contained in a set of IDs.

SYNOPSIS

#include <systemd/sd-id128.h>

sd_id128_t sd_id128_null(void);
int sd_id128_string_null(const char *s);
int sd_id128_is_null(sd_id128_t id);
int sd_id128_is_allf(sd_id128_t id);
int sd_id128_in_setv(sd_id128_t id, const sd_id128_t *set, size_t set_len);
int sd_id128_in_set_sentinel(sd_id128_t id, const sd_id128_t *set);
int sd_id128_in_set(sd_id128_t id, ...);

DESCRIPTION

These functions allow checking if a 128-bit identifier (ID128) is the special NULL ID or the all-ones ID, or if it is part of a set of IDs.

sd_id128_null() returns the special ID128 value that represents the NULL ID. This is a 128-bit value where all bits are zero. It is returned by various functions when no ID is available or defined.

sd_id128_string_null() checks if the string s is a canonical representation of the NULL ID. It returns 1 if it is, and 0 otherwise.

sd_id128_is_null() checks if the given id is the NULL ID. It returns 1 if it is, and 0 otherwise.

sd_id128_is_allf() checks if the given id is the all-ones ID (all bits set to 1). It returns 1 if it is, and 0 otherwise.

sd_id128_in_setv() checks if the given id is present in the set of IDs passed in the set array, which has set_len elements. It returns 1 if the ID is found, and 0 otherwise.

sd_id128_in_set_sentinel() is similar to sd_id128_in_setv(), but the set array is expected to be NULL-terminated. That is, the end of the array is marked by an ID128 value where all bits are zero (the NULL ID).

sd_id128_in_set() is a convenience wrapper around sd_id128_in_set_sentinel(). It takes a variable number of sd_id128_t arguments, followed by a sentinel NULL ID, and checks if the first argument id is present among the subsequent arguments. This function uses setenv(3) internally, which is not thread-safe.

Note that the implementation of these functions might use the getenv(3) function, which is not thread-safe. This means that if you call these functions from multiple threads, you must ensure that no thread calls setenv(3) concurrently. NOTES

Functions described here are available as a shared library, which can be compiled against and linked to with the libsystemd pkg-config(1) file.

The code described here uses getenv(3), which is declared to be not multi-thread-safe. This means that the code calling the functions described here must not call setenv(3) from a parallel thread. It is recommended to only do calls to setenv() from an early phase of the program when no other threads have been started.
