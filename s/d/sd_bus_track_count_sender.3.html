sd_bus_track_count_sender(3)

SD_BUS_TRACK_ADD_NAME(3) sd_bus_track_add_name SD_BUS_TRACK_ADD_NAME(3)

الاسم

sd_bus_track_add_name, sd_bus_track_add_sender, sd_bus_track_remove_name, sd_bus_track_remove_sender, sd_bus_track_count, sd_bus_track_count_sender, sd_bus_track_count_name, sd_bus_track_contains, sd_bus_track_first, sd_bus_track_next - إضافة، إزالة، واسترجاع الأقران المتتبعين في كائن تتبع أقران الناقل.

ملخص

#include <systemd/sd-bus.h>

int sd_bus_track_remove_sender(sd_bus_track* t, sd_bus_message* message);

unsigned sd_bus_track_count(sd_bus_track* t);

int sd_bus_track_count_name(sd_bus_track* t, const char* name);

int sd_bus_track_count_sender(sd_bus_track* t, sd_bus_message* message);

const char* sd_bus_track_next(sd_bus_track* t);

الوصف

sd_bus_track_add_name() تضيف نظيرًا للتتبع إلى كائن تتبع أقران الناقل. يجب أن يشير الوسيط الأول إلى كائن تتبع أقران الناقل الذي تم إنشاؤه بواسطة sd_bus_track_new(3)، يجب أن يشير الاسم الثاني إلى اسم نظير D-Bus لتتبعه، سواء بتنسيق الخدمة الفريد أو المعروف. إذا لم يكن الاسم متتبعًا بعد، فسيتم إضافته إلى قائمة الأسماء المراد تتبعها. إذا كان يتم تتبعه بالفعل وتم تمكين الوضع غير التكراري، فلا يتم تنفيذ أي عملية بواسطة هذا الاستدعاء. إذا تم تمكين الوضع التكراري، يتم زيادة عداد لكل اسم بمقدار واحد في كل مرة يتم فيها استدعاء هذا الاستدعاء، ويجب استدعاء sd_bus_track_remove_name() بنفس عدد مرات استدعاء sd_bus_track_add_name() قبل إيقاف تتبع الاسم. استخدم sd_bus_track_set_recursive(3) للتبديل من الوضع الافتراضي غير التكراري إلى الوضع التكراري، أو العودة إليه. لاحظ أن الاسم المحدد يتم تتبعه كما هو، ولا يتم حل الأسماء المعروفة إلى أسماء فريدة بواسطة هذا الاستدعاء. لاحظ أن كائنات تتبع أقران الناقل المتعددة قد تتتبع نفس الاسم.

sd_bus_track_remove_name() يلغي تأثير sd_bus_track_add_name() ويزيل اسم نظير الناقل من قائمة الأقران المراد مراقبتهم. اعتمادًا على ما إذا كان الوضع غير التكراري أو التكراري ممكّنًا لكائن تتبع أقران الناقل، سيقوم هذا الاستدعاء إما بإزالة الاسم بالكامل من كائن التتبع، أو سيقوم ببساطة بتقليل العداد لكل اسم بمقدار واحد، وإزالة الاسم فقط عند وصول العداد إلى الصفر (انظر أعلاه). لاحظ أن فصل نظير الناقل عن الناقل سيقوم ضمنيًا بإزالة أسمائه بالكامل من كائن تتبع أقران الناقل، بغض النظر عن العداد الحالي لكل اسم.

sd_bus_track_add_sender() و sd_bus_track_remove_sender() مشابهان لـ sd_bus_track_add_name() و sd_bus_track_remove_name() ولكنهما يأخذان رسالة ناقل كوسيط. يتم تحديد مرسل رسالة الناقل هذه وإضافته إلى/إزالته من كائن تتبع أقران الناقل. نظرًا لأن الرسائل تنشأ دائمًا من أسماء فريدة، وليس من أسماء معروفة، فهذا يعني أن هذا الاستدعاء سيضيف فعليًا الأسماء الفريدة فقط إلى كائن تتبع أقران الناقل.

sd_bus_track_count() يرجع عدد الأسماء التي يتم تتبعها حاليًا بواسطة كائن تتبع أقران الناقل المحدد. لاحظ أن هذه الدالة ترجع دائمًا العدد الفعلي للأسماء المتتبعة، وبالتالي إذا تم استدعاء sd_bus_track_add_name() عدة مرات لنفس الاسم، فسيتم حسابه كواحد فقط، بغض النظر عما إذا كان الوضع التكراري مستخدمًا أم لا.

sd_bus_track_count_name() يرجع العداد الحالي لكل اسم للاسم المحدد. إذا تم استخدام الوضع غير التكراري، فسترجع هذه الدالة إما 1 أو 0، اعتمادًا على ما إذا كان الاسم المحدد قد تمت إضافته إلى كائن التتبع من قبل، أم لا. إذا تم تمكين الوضع التكراري، فقد يتم إرجاع قيم أكبر من 1 أيضًا، في حالة استدعاء sd_bus_track_add_name() عدة مرات لنفس الاسم.

sd_bus_track_count_sender() مشابه لـ sd_bus_track_count_name()، ولكنه يأخذ كائن رسالة ناقل ويرجع العداد لكل اسم مطابق لمرسل الرسالة.

sd_bus_track_contains() يمكن استخدامه لتحديد ما إذا كان الاسم المحدد قد تمت إضافته مرة واحدة على الأقل إلى كائن تتبع أقران الناقل المحدد.

sd_bus_track_first() و sd_bus_track_next() يمكن استخدامهما لتعداد جميع الأسماء التي يتم تتبعها حاليًا بواسطة كائن تتبع أقران الناقل الممرر. ترجع sd_bus_track_first() الإدخال الأول في الكائن، وتعيد تعيين مؤشر قراءة داخلي. كل استدعاء لاحق لـ sd_bus_track_next() يرجع الاسم التالي الموجود في كائن الناقل. إذا تم الوصول إلى النهاية، فسيتم إرجاع NULL. إذا لم تتم إضافة أي أسماء إلى الكائن بعد، فسترجع sd_bus_track_first() NULL على الفور. ترتيب إرجاع الأسماء غير محدد؛ وعلى وجه الخصوص، فإن الاسم الذي يعتبر الأول الذي تم إرجاعه غير محدد. إذا تم تمكين الوضع التكراري وتمت إضافة نفس الاسم عدة مرات إلى كائن تتبع أقران الناقل، فسيتم إرجاعه مرة واحدة فقط بواسطة هذا التعداد. إذا تمت إضافة أسماء جديدة إلى كائن تتبع أقران الناقل أو تمت إزالة الأسماء الموجودة أثناء تعدادها، ينتهي التعداد عند الاستدعاء التالي لـ sd_bus_track_next() حيث يتم إرجاع NULL.

قيمة الإرجاع

عند النجاح، ترجع sd_bus_track_add_name() و sd_bus_track_add_sender() القيمة 0 إذا كان الاسم المحدد قد تمت إضافته بالفعل إلى كائن تتبع أقران الناقل من قبل، وقيمة موجبة إذا لم يكن كذلك. عند الفشل، ترجع رمز خطأ سالبًا على نمط errno.

sd_bus_track_remove_name() و sd_bus_track_remove_sender() ترجع قيمة موجبة إذا كان الاسم المحدد قد تم تتبعه سابقًا بواسطة كائن تتبع أقران الناقل وتمت إزالته الآن. في الوضع غير التكراري، يتم إرجاع 0 إذا لم يتم تتبع الاسم المحدد بعد. في الوضع التكراري، يتم إرجاع -EUNATCH في هذه الحالة. عند الفشل، ترجع رمز خطأ سالبًا على نمط errno.

sd_bus_track_count() يرجع عدد الأسماء التي يتم تتبعها حاليًا، أو 0 عند الفشل.

sd_bus_track_count_name() و sd_bus_track_count_sender() يرجعان العداد الحالي لكل اسم للاسم المحدد أو لمرسل الرسالة المحددة. يتم إرجاع الصفر للأسماء التي لم يتم تتبعها بعد، وقيمة موجبة للأسماء التي تمت إضافتها مرة واحدة على الأقل. القيم الأكبر من 1 يتم إرجاعها فقط في الوضع التكراري. عند الفشل، يتم إرجاع رمز خطأ سالبًا على نمط errno.

sd_bus_track_contains() قد يرجع الاسم الممرر إذا كان موجودًا في كائن تتبع أقران الناقل. عند الفشل، وإذا لم تتم إضافة الاسم بعد، يتم إرجاع NULL.

sd_bus_track_first() و sd_bus_track_next() يرجعان الاسم الأول/التالي الموجود في كائن تتبع أقران الناقل، و NULL إذا تم الوصول إلى نهاية التعداد وعند حدوث خطأ.

الأخطاء

قد تشير الأخطاء المرجعة إلى المشكلات التالية:

-EUNATCH

تم استدعاء sd_bus_track_remove_name() أو sd_bus_track_remove_sender() لاسم لم تتم إضافته مسبقًا إلى كائن الناقل.

-EINVAL

الوسيط المحدد غير صالح.

-ENOMEM

فشل تخصيص الذاكرة.
