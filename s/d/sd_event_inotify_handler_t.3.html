#include <systemd/sd-event.h>

الوصف

sd_event_add_inotify() تضيف مصدر حدث جديد لنظام ملفات inotify(7) إلى حلقة الأحداث. يتم تحديد كائن حلقة الأحداث في المعامل event، ويتم إرجاع كائن مصدر الحدث في المعامل source. يحدد المعامل path مسار عقدة نظام الملفات المراد مراقبتها. يحدد المعامل mask أنواع أحداث العقدة المراد مراقبتها بشكل خاص. يجب أن يحتوي على تركيبة ORed من الأعلام IN_ACCESS، IN_ATTRIB، IN_CLOSE_WRITE، ... انظر inotify(7) لمزيد من المعلومات.

يجب أن يشير handler إلى دالة سيتم استدعاؤها عند تغيير العقدة أو NULL. سيتم تمرير المؤشر userdata إلى الدالة، والتي يمكن اختيارها بحرية من قبل المتصل. تتلقى الدالة أيضًا مؤشرًا إلى بنية struct inotify_event تحتوي على معلومات حول حدث العقدة. قد تُرجع الدالة قيمة سالبة للإشارة إلى خطأ (انظر أدناه)، وسيتم تجاهل القيم الأخرى التي تم إرجاعها. إذا كان handler هو NULL، فسيتم استخدام معالج افتراضي يستدعي sd_event_exit(3).

sd_event_add_inotify_fd() مماثل لـ sd_event_add_inotify()، باستثناء أنه يأخذ واصف ملف لعقدة (ربما واحد O_PATH، ولكن أي واحد آخر سيفي بالغرض أيضًا) بدلاً من مسار في نظام الملفات.

إذا تم تثبيت مصادر أحداث متعددة لنفس العقدة، فسيتم مشاركة واصف مراقبة inotify الأساسي تلقائيًا. قد يحتوي المعامل mask على أي علم تم تعريفه بواسطة واجهة برمجة تطبيقات inotify، باستثناء IN_MASK_ADD.

يتم تمكين المعالج بشكل مستمر (SD_EVENT_ON)، ولكن يمكن تغيير ذلك باستخدام sd_event_source_set_enabled(3). بدلاً من ذلك، يمكن استخدام علامة القناع IN_ONESHOT لطلب وضع SD_EVENT_ONESHOT. إذا أعادت الدالة handler رمز خطأ سالبًا، فسيتم تعطيلها بعد الاستدعاء، حتى لو تم طلب وضع SD_EVENT_ON من قبل.

كمحدد خاص، يمكن تغيير أولوية مصادر أحداث inotify فقط (انظر sd_event_source_set_priority(3)) في الوقت بين إنشاء كائن مصدر الحدث باستخدام sd_event_add_inotify() وبداية تكرار حلقة الأحداث التالية. سيتم رفض محاولات تغيير الأولوية لاحقًا. ضع في اعتبارك تحرير وتخصيص مصدر حدث inotify جديد لتغيير الأولوية في تلك المرحلة.

لتدمير كائن مصدر الحدث استخدم sd_event_source_unref(3)، ولكن لاحظ أنه يتم إزالة مصدر الحدث من حلقة الأحداث فقط عندما يتم إسقاط جميع المراجع إلى مصدر الحدث. للتأكد من أن مصدر الحدث لا يعمل بعد الآن، حتى عندما لا يزال هناك مرجع إليه، فكر في تعطيله باستخدام sd_event_source_set_enabled(3).

إذا تم تمرير المعامل الثاني لـ sd_event_add_inotify() كـ NULL، فلن يتم إرجاع أي مرجع لكائن مصدر الحدث. في هذه الحالة، يعتبر مصدر الحدث "عائمًا"، وسيتم تدميره ضمنيًا عند تدمير حلقة الأحداث نفسها.

إذا كان المعامل handler لـ sd_event_add_inotify() هو NULL، وسيتم تشغيل مصدر الحدث، فسيتم اعتبار ذلك طلبًا للخروج من حلقة الأحداث. في هذه الحالة، يتم تمرير المعامل userdata، بعد تحويله إلى عدد صحيح، كمعامل رمز الخروج إلى sd_event_exit(3).

sd_event_source_get_inotify_mask() تسترجع قناع مراقبة inotify المُعد لـ مصدر حدث تم إنشاؤه مسبقًا باستخدام sd_event_add_inotify(). تأخذ كائن مصدر الحدث كالمعامل source ومؤشر إلى متغير uint32_t لإرجاع القناع فيه.

sd_event_source_get_inotify_path() تسترجع المسار الهدف لمراقبة inotify المُعدة لـ مصدر حدث تم إنشاؤه مسبقًا باستخدام sd_event_add_inotify(). تأخذ كائن مصدر الحدث كالمعامل source ومؤشر إلى متغير const char ** لإرجاع المسار فيه. يجب على المتصل عدم تحرير المسار المُرجع.

قيمة الإرجاع

عند النجاح، تُرجع هذه الدوال 0 أو عددًا صحيحًا موجبًا. عند الفشل، تُرجع رمز خطأ سالبًا بأسلوب errno.

الأخطاء

قد تشير الأخطاء المُرجعة إلى المشاكل التالية:

-ENOMEM

لا توجد ذاكرة كافية لتخصيص كائن.

-EINVAL

تم تمرير وسيط غير صالح. يشمل ذلك تحديد قناع مع تعيين IN_MASK_ADD.

-ESTALE

تم إرجاعها بواسطة sd_event_source_add_inotify() أو sd_event_source_add_inotify_fd() عندما تكون حلقة الأحداث قد انتهت بالفعل. تم إرجاعها بواسطة sd_event_source_get_inotify_path() عندما لا توجد بيانات عقدة نشطة مرتبطة بمصدر الحدث، على سبيل المثال، عندما يكون مصدر الحدث معطلاً.

-ECHILD

تم إنشاء حلقة الأحداث في عملية مختلفة، أو مكتبة، أو مثيل وحدة.

-EDOM

مصدر الحدث المُمرر ليس مصدر حدث عملية inotify.

-EBADF

واصف الملف المُمرر غير صالح.

-ENOSYS

تم استدعاء sd_event_add_inotify_fd() أو sd_event_source_get_inotify_path() بدون تحميل /proc/.

أمثلة

مثال 1. برنامج بسيط يستخدم inotify لمراقبة دليل واحد أو اثنين

/* SPDX-License-Identifier: MIT-0 */
#include <stdio.h>
#include <string.h>
#include <sys/inotify.h>
#include <systemd/sd-event.h>
#define _cleanup_(f) __attribute__((cleanup(f)))
static int inotify_handler(sd_event_source *source,
                           const struct inotify_event *event,
                           void *userdata) {
  const char *desc = NULL;
  sd_event_source_get_description(source, &desc);
  if (event->mask & IN_Q_OVERFLOW)
    printf("inotify-handler <%s>: overflow\n", desc);
  else if (event->mask & IN_CREATE)
    printf("inotify-handler <%s>: create on %s\n", desc, event->name);
  else if (event->mask & IN_DELETE)
    printf("inotify-handler <%s>: delete on %s\n", desc, event->name);
  else if (event->mask & IN_MOVED_TO)
    printf("inotify-handler <%s>: moved-to on %s\n", desc, event->name);
  /* Terminate the program if an "exit" file appears */
  if ((event->mask & (IN_CREATE|IN_MOVED_TO)) &&
      strcmp(event->name, "exit") == 0)
    sd_event_exit(sd_event_source_get_event(source), 0);
  return 1;
}
int main(int argc, char **argv) {
  _cleanup_(sd_event_unrefp) sd_event *event = NULL;
  _cleanup_(sd_event_source_unrefp) sd_event_source *source1 = NULL, *source2 = NULL;
  const char *path1 = argc > 1 ? argv[1] : "/tmp";
  const char *path2 = argc > 2 ? argv[2] : "/var";
  int r;
  r = sd_event_new(&event);
  if (r < 0) {
    fprintf(stderr, "Failed to create event loop: %s\n", strerror(-r));
    return 1;
  }
  r = sd_event_add_inotify(event, &source1, path1,
                           IN_CREATE|IN_DELETE|IN_MOVED_TO, inotify_handler, NULL);
  if (r < 0) {
    fprintf(stderr, "Failed to add inotify watch for %s: %s\n", path1, strerror(-r));
    goto finish;
  }
  sd_event_source_set_description(source1, path1);
  r = sd_event_add_inotify(event, &source2, path2,
                           IN_CREATE|IN_DELETE|IN_MOVED_TO, inotify_handler, NULL);
  if (r < 0) {
    fprintf(stderr, "Failed to add inotify watch for %s: %s\n", path2, strerror(-r));
    goto finish;
  }
  sd_event_source_set_description(source2, path2);
  r = sd_event_loop(event);
finish:
  return r < 0 ? 1 : 0;
}

This program registers two directories for inotify events: $1 and $2 (or /tmp and /var if not specified). If a file is created, deleted, or moved into one of the directories, a message is printed to standard output. If a file named "exit" is created or moved into one of the directories, the program terminates.

To compile this program: gcc -o inotify-example inotify-example.c -lsystemd

To run this program: ./inotify-example /path/to/dir1 /path/to/dir2

When the program is running, you can create, delete, or move files in the specified directories. For example:

touch /path/to/dir1/newfile
rm /path/to/dir1/newfile
mv /path/to/dir1/newfile /path/to/dir1/anothername
touch /path/to/dir1/exit

The program will print messages indicating the events that occurred. Creating or moving the file "exit" will cause the program to terminate.
