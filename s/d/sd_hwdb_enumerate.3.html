sd_hwdb_enumerate(3)

الحزمة: systemd-libs

التراخيص: LGPL-2.1-or-later, CC0-1.0, GPL-2.0-or-later WITH Linux-syscall-note

SD_HWDB_GET(3) sd_hwdb_get SD_HWDB_GET(3)

الاسم

sd_hwdb_get, sd_hwdb_seek, sd_hwdb_enumerate, SD_HWDB_FOREACH_PROPERTY - البحث في hwdb أو الوصول إلى الإدخالات

الملخص

#include <systemd/sd-hwdb.h>

int sd_hwdb_get(sd_hwdb *hwdb, const char *modalias, const char *key, const char **value);

int sd_hwdb_seek(sd_hwdb *hwdb, const char *modalias);

int sd_hwdb_enumerate(sd_hwdb *hwdb, const char **key, const char **value);

SD_HWDB_FOREACH_PROPERTY(hwdb, modalias, key, value);

الوصف

sd_hwdb_get() تستعلم عن الكائن hwdb الذي تم إنشاؤه سابقًا باستخدام sd_hwdb_new(3) للإدخالات التي تطابق السلسلة المحددة modalias، وتعيد القيمة المقابلة للمفتاح key. تُعاد القيمة كسلسلة منتهية بـ NUL في value. يجب ألا يعدلها المستدعي وهي صالحة طالما أن هناك مرجعًا إلى hwdb. عندما تتطابق العديد من الأنماط في قاعدة البيانات مع modalias، يتم استخدام النمط ذو الأولوية الأعلى. انظر hwdb(7) للحصول على التفاصيل.

sd_hwdb_seek() تحدد الإدخالات التي تطابق السلسلة المحددة modalias. الاستعلامات اللاحقة باستخدام sd_hwdb_enumerate() ستصل إلى أزواج المفتاح-القيمة لهذه السلسلة.

sd_hwdb_enumerate() تعيد (بالترتيب) جميع أزواج المفتاح-القيمة المعرفة للسلسلة المستخدمة مع sd_hwdb_seek(). تُعاد كل زوج كسلاسل منتهية بـ NUL في key و value. يجب ألا يعدلها المستدعي وهي صالحة طالما أن هناك مرجعًا إلى hwdb. عندما تتطابق العديد من الأنماط في قاعدة البيانات مع modalias، يتم استخدام مزيج من جميع أزواج المفتاح-القيمة المتطابقة. انظر hwdb(7) للحصول على التفاصيل.

الماكرو SD_HWDB_FOREACH_PROPERTY() يجمع بين sd_hwdb_seek() و sd_hwdb_enumerate(). لا يتم إجراء معالجة للأخطاء ويتوقف التكرار ببساطة عند حدوث خطأ. انظر المثال أدناه.

قيمة الإرجاع

عند النجاح، تعيد sd_hwdb_get() و sd_hwdb_seek() عددًا صحيحًا غير سالب. عند الفشل، تعيدان رمز خطأ سالبًا بنمط errno.

sd_hwdb_enumerate() تعيد عددًا صحيحًا موجبًا إذا تم العثور على زوج مفتاح-قيمة آخر أو صفر إذا تم تعداد جميع الإدخالات بالفعل. عند الفشل، تعيد رمز خطأ سالبًا بنمط errno.

الأخطاء

قد تشير الأخطاء التي تم إرجاعها إلى المشاكل التالية:

-EINVAL

أحد المعلمات هو NULL.

تمت الإضافة في الإصدار 246.

-ENOENT

لم يتم العثور على إدخال لـ modalias المحدد.

تمت الإضافة في الإصدار 246.

-EAGAIN

لم يتم استدعاء sd_hwdb_seek() قبل sd_hwdb_enumerate().

تمت الإضافة في الإصدار 246.

ملاحظات

الوظائف الموصوفة هنا متاحة كمكتبة مشتركة، والتي يمكن تجميعها وربطها مع ملف pkg-config(1) الخاص بـ libsystemd.

الكود الموصوف هنا يستخدم getenv(3)، والذي تم الإعلان عنه بأنه ليس آمنًا للاستخدام مع الخيوط المتوازية. هذا يعني أن الكود الذي يستدعي الوظائف الموصوفة هنا يجب ألا يستدعي setenv(3) من خيط متوازٍ. يوصى بإجراء استدعاءات لـ setenv() فقط من مرحلة مبكرة من البرنامج عندما لم تبدأ أي خيوط أخرى.

أمثلة

مثال 1. البحث عن إدخالات hwdb لجهاز USB

/* SPDX-License-Identifier: MIT-0 */
#define _GNU_SOURCE 1
#include <stdio.h>
#include <stdint.h>
#include <systemd/sd-hwdb.h>
int print_usb_properties(uint16_t vid, uint16_t pid) {
  char match[128];
  sd_hwdb *hwdb;
  const char *key, *value;
  int r;
  /* المطابقة بمعرف المورد ومعرف المنتج هذين */
  snprintf(match, sizeof match, "usb:v%04Xp%04X", vid, pid);
  r = sd_hwdb_new(&hwdb);
  if (r < 0)
    return r;
  SD_HWDB_FOREACH_PROPERTY(hwdb, match, key, value)
    printf("%s: \"%s\" → \"%s\"\n", match, key, value);
  sd_hwdb_unref(hwdb);
  return 0;
}
int main(int argc, char **argv) {
  print_usb_properties(0x046D, 0xC534);
  return 0;
}

التأثير مشابه لاستدعاء systemd-hwdb query usb:v046DpC534.
