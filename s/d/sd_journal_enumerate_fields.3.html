sd_journal_enumerate_fields(3)

الحزمة: systemd-libs

التراخيص: LGPL-2.1-or-later, CC0-1.0, GPL-2.0-or-later WITH Linux-syscall-note

SD_JOURNAL_ENUMERATE_FIELDS(3) sd_journal_enumerate_fields SD_JOURNAL_ENUMERATE_FIELDS(3)

الاسم

sd_journal_enumerate_fields, sd_journal_restart_fields, SD_JOURNAL_FOREACH_FIELD - قراءة أسماء الحقول المستخدمة من السجل

الملخص

#include <systemd/sd-journal.h>

int sd_journal_enumerate_fields(sd_journal *j, const char **field);

void sd_journal_restart_fields(sd_journal *j);

SD_JOURNAL_FOREACH_FIELD(sd_journal *j, const char *field);

الوصف

يمكن استخدام sd_journal_enumerate_fields() للتكرار عبر جميع أسماء الحقول المستخدمة في ملفات السجل المفتوحة. في كل استدعاء، يتم إرجاع اسم الحقل التالي. ترتيب أسماء الحقول المرجعة غير محدد. تأخذ وسيطين: كائن سياق السجل، بالإضافة إلى مؤشر إلى مؤشر سلسلة نصية ثابتة حيث يتم تخزين اسم الحقل. البيانات المرجعة موجودة في خريطة ذاكرة للقراءة فقط وهي صالحة فقط حتى الاستدعاء التالي لـ sd_journal_enumerate_fields(). لاحظ أن هذا الاستدعاء يخضع لعتبة حجم حقل البيانات كما يتم التحكم فيه بواسطة sd_journal_set_data_threshold().

sd_journal_restart_fields() يعيد تعيين فهرس تعداد أسماء الحقول إلى بداية القائمة. الاستدعاء التالي لـ sd_journal_enumerate_fields() سيعيد اسم الحقل الأول مرة أخرى.

يمكن استخدام الماكرو SD_JOURNAL_FOREACH_FIELD() كغلاف مفيد حول sd_journal_restart_fields() و sd_journal_enumerate_fields().

لا تتأثر هذه الدوال حاليًا بالمطابقات التي تم تعيينها باستخدام sd_journal_add_match() ولكن هذا قد يتغير في إصدار لاحق من هذا البرنامج.

لاسترداد القيم الممكنة التي يمكن أن يأخذها حقل معين، استخدم sd_journal_query_unique(3).

قيمة الإرجاع

sd_journal_enumerate_fields() تعيد عددًا صحيحًا موجبًا إذا تم قراءة اسم الحقل التالي، 0 عندما لا تكون هناك أسماء حقول معروفة أخرى، أو رمز خطأ سالب بأسلوب errno. sd_journal_restart_fields() لا تعيد شيئًا.

ملاحظات

جميع الدوال المدرجة هنا لا تتأثر بالخيوط، ويمكن لخيط واحد فقط العمل على كائن معين في أي وقت. قد تصل الخيوط المختلفة إلى نفس الكائن في أوقات مختلفة. يمكن استخدام كائنات مستقلة متعددة من خيوط مختلفة بالتوازي.

الدوال الموصوفة هنا متاحة كمكتبة مشتركة، والتي يمكن تجميعها وربطها باستخدام ملف libsystemd pkg-config(1).

أمثلة

استخدم الماكرو SD_JOURNAL_FOREACH_FIELD() للتكرار عبر جميع أسماء الحقول المستخدمة في السجل الحالي.

/* SPDX-License-Identifier: MIT-0 */
#include <errno.h>
#include <stdio.h>
#include <systemd/sd-journal.h>
int main(int argc, char *argv[]) {
  sd_journal *j;
  const char *field;
  int r;
  r = sd_journal_open(&j, SD_JOURNAL_LOCAL_ONLY);
  if (r < 0) {
    fprintf(stderr, "Failed to open journal: %s\n", strerror(-r));
    return 1;
  }
  SD_JOURNAL_FOREACH_FIELD(j, field)
    printf("%s\n", field);
  sd_journal_close(j);
  return 0;
}
