الوصف

sd-id128.h هو جزء من libsystemd(3) ويوفر واجهات برمجة التطبيقات لإنشاء وتحويل ومقارنة قيم المعرفات المكونة من 128 بت. قيم المعرفات المكونة من 128 بت التي تتم معالجتها وتوليدها بواسطة واجهات برمجة التطبيقات هذه هي تعميم لمعرفات UUIDs OSF كما تم تعريفها في RFC 4122[1]، ولكنها تستخدم تنسيق سلسلة أبسط. لا تفرض هذه الدوال أي بنية على المعرفات المستخدمة، على عكس UUIDs OSF أو GUIDs Microsoft، ولكنها متوافقة بشكل عام مع هذه الأنواع من المعرفات.

يتم تنفيذ معرف 128 بت على شكل الاتحاد التالي:

typedef union sd_id128 {
  uint8_t bytes[16];
  uint64_t qwords[2];
} sd_id128_t;

يسمح هذا الاتحاد بالوصول إلى المعرف المكون من 128 بت كسلسلة من 16 بايت منفصلة أو كلمتين من 64 بت. من الآمن عمومًا الوصول إلى مكونات المعرف عبر مصفوفة 8 بت لتجنب مشاكل ترتيب البايت (endianness). يُقصد تمرير هذا الاتحاد بالقيمة (بدلاً من المرجع) ويمكن للعملاء التعامل معه مباشرة.

تم تعريف العديد من الماكروات للإشارة إلى المعرفات المكونة من 128 بت وفك تشفيرها:

SD_ID128_MAKE() تستخدم لكتابة معرف ثابت في الكود المصدري. تقنية شائعة هي تعيين اسم لمعرف باستخدام هذا الماكرو:

#define SD_MESSAGE_COREDUMP SD_ID128_MAKE(fc,2e,22,bc,6e,e6,47,b6,b9,07,29,ab,34,a2,50,b1)

SD_ID128_NULL يحدد معرفًا يتكون فقط من بايتات NUL (أي، جميع البتات غير نشطة).

SD_ID128_ALLF يحدد معرفًا يتكون فقط من بايتات 0xFF (أي، جميع البتات نشطة).

SD_ID128_MAKE_STR() مشابه لـ SD_ID128_MAKE()، ولكنه ينشئ تعبيرًا ثابتًا من نوع const char* يمكن استخدامه بشكل ملائم في تنسيقات الرسائل:

#include <stdio.h>
#define SD_MESSAGE_COREDUMP_STR SD_ID128_MAKE_STR(fc,2e,22,bc,6e,e6,47,b6,b9,07,29,ab,34,a2,50,b1)
int main(int argc, char **argv) {
  puts("Match for coredumps: MESSAGE_ID=" SD_MESSAGE_COREDUMP_STR);
}

SD_ID128_CONST_STR() يحول المعرفات الثابتة إلى سلاسل ثابتة للإخراج. المثال التالي للكود سيطبع السلسلة "fc2e22bc6ee647b6b90729ab34a250b1":

int main(int argc, char *argv[]) {
  puts("Match for coredumps: %s", SD_ID128_CONST_STR(SD_MESSAGE_COREDUMP));
}

SD_ID128_FORMAT_STR و SD_ID128_FORMAT_VAL() تستخدم لتنسيق معرف في تنسيق سلسلة printf(3)، كما هو موضح في المثال التالي:

int main(int argc, char *argv[]) {
  sd_id128_t id;
  id = SD_ID128_MAKE(ee,89,be,71,bd,6e,43,d6,91,e6,c5,5d,eb,03,02,07);
  printf("The ID encoded in this C file is " SD_ID128_FORMAT_STR ".\n", SD_ID128_FORMAT_VAL(id));
  return 0;
}

SD_ID128_UUID_FORMAT_STR و SD_ID128_MAKE_UUID_STR() مشابهان لـ SD_ID128_FORMAT_STR و SD_ID128_MAKE_STR()، ولكنهما يشتملان على شرطات في UUIDs للامتثال للتمثيل القياسي لـ UUID[2]. يعتمد تنسيق السلسلة على قواعد RFC4122[1]، أي، تحويل ترتيب البايت Big Endian. هذا يتطابق مع سلوك البنية التحتية لمساحة المستخدم في أنظمة Linux الأخرى. من المحتمل أن يكون من الأفضل تجنب UUIDs من متغيرات أخرى لتجنب الغموض غير الضروري. تتبع جميع المعرفات المكونة من 128 بت التي تم إنشاؤها بواسطة واجهات برمجة تطبيقات sd-id128 بشكل صارم قواعد المتغير 1 الإصدار 4 من UUIDs، كما هو الحال في RFC 4122.

sd_id128_equal() تقارن معرفين مكونين من 128 بت:

int main(int argc, char *argv[]) {
  sd_id128_t a, b, c;
  a = SD_ID128_MAKE(ee,89,be,71,bd,6e,43,d6,91,e6,c5,5d,eb,03,02,07);
  b = SD_ID128_MAKE(f2,28,88,9c,5f,09,44,15,9d,d7,04,77,58,cb,e7,3e);
  c = a;
  assert(sd_id128_equal(a, c));
  assert(!sd_id128_equal(a, b));
  return 0;
}

sd_id128_string_equal() مشابه لـ sd_id128_equal()، ولكن المعرف الأول يتم تشكيله كسلسلة const char*. تنطبق نفس القيود على الوسيط الأول لـ sd_id128_from_string().

sd_id128_is_null() تتحقق مما إذا كان المعرف يتكون فقط من بايتات NUL:

assert(sd_id128_is_null(SD_ID128_NULL));

ببساطة، sd_id128_is_allf() تتحقق مما إذا كان
