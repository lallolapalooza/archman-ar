الاسم

الهيكل

#include <systemd/sd-journal.h>

int sd_journal_open_directory_fd(sd_journal **ret, int fd, int flags);

int sd_journal_open_files(sd_journal **ret, const char **paths, int flags);

int sd_journal_open_files_fd(sd_journal **ret, int fds[], unsigned n_fds, int flags);

void sd_journal_close(sd_journal *j);

الوصف

sd_journal_open() يفتح سجل السجل (log journal) للقراءة. سيجد جميع ملفات السجل تلقائيًا وسيوزعها تلقائيًا عند القراءة. كأول وسيط، يأخذ مؤشرًا إلى مؤشر sd_journal، والذي، عند النجاح، سيحتوي على كائن سياق السجل. الوسيط الثاني هو حقل أعلام (flags)، والذي قد يتكون من الأعلام التالية مدمجة معًا: SD_JOURNAL_LOCAL_ONLY يضمن أنه سيتم فتح ملفات السجل التي تم إنشاؤها على الجهاز المحلي فقط. SD_JOURNAL_RUNTIME_ONLY يضمن أنه سيتم فتح ملفات السجل المتطايرة فقط، باستثناء تلك المخزنة على التخزين الدائم. SD_JOURNAL_SYSTEM سيؤدي إلى فتح ملفات السجل للخدمات النظامية والنواة (على عكس عمليات جلسة المستخدم). SD_JOURNAL_CURRENT_USER سيؤدي إلى فتح ملفات السجل للمستخدم الحالي. إذا لم يتم تحديد SD_JOURNAL_SYSTEM أو SD_JOURNAL_CURRENT_USER، فسيتم فتح جميع أنواع ملفات السجل.

sd_journal_open_namespace() مشابه لـ sd_journal_open() ولكنه يأخذ وسيطًا إضافيًا namespace يحدد مساحة اسم السجل التي سيتم العمل عليها. إذا تم تحديده كـ NULL، فإن الاستدعاء مطابق لـ sd_journal_open(). إذا كان غير NULL، يتم الوصول فقط إلى البيانات من مساحة الاسم المحددة بالمعلم المحدد. يفهم هذا الاستدعاء علامتين إضافيتين: إذا تم تحديد SD_JOURNAL_ALL_NAMESPACES، يتم تجاهل المعلم namespace ويتم الوصول إلى جميع مساحات الأسماء المعرفة في وقت واحد؛ إذا تم تحديد SD_JOURNAL_INCLUDE_DEFAULT_NAMESPACE، يتم الوصول إلى مساحة الاسم المحددة ومساحة الاسم الافتراضية ولكن لا شيء آخر (ليس لهذه العلامة أي تأثير عند تمرير namespace كـ NULL). للحصول على تفاصيل حول مساحات أسماء السجل، انظر systemd-journald.service(8).

sd_journal_open_directory() مشابه لـ sd_journal_open() ولكنه يأخذ مسار دليل مطلق كوسيط. سيتم فتح جميع ملفات السجل في هذا الدليل وتوزيعها تلقائيًا. يأخذ هذا الاستدعاء أيضًا وسيط أعلام. الأعلام المقبولة بواسطة هذا الاستدعاء هي SD_JOURNAL_OS_ROOT، SD_JOURNAL_SYSTEM، و SD_JOURNAL_CURRENT_USER. إذا تم تحديد SD_JOURNAL_OS_ROOT، فسيتم البحث عن ملفات السجل أسفل /var/log/journal و /run/log/journal بالنسبة للمسار المحدد، بدلاً من تحتها مباشرة. العلامتان الأخريان تحددان الملفات التي سيتم فتحها، بنفس طريقة sd_journal_open().

sd_journal_open_directory_fd() مشابه لـ sd_journal_open_directory()، ولكنه يأخذ واصف ملف يشير إلى دليل في نظام الملفات بدلاً من مسار نظام ملفات مطلق. بالإضافة إلى الأعلام التي تقبلها sd_journal_open_directory()، تقبل هذه الوظيفة أيضًا SD_JOURNAL_TAKE_DIRECTORY_FD. إذا تم تحديد SD_JOURNAL_TAKE_DIRECTORY_FD، فستتولى الوظيفة ملكية واصف الملف المحدد عند النجاح، وسيتم إغلاقه بواسطة sd_journal_close()، وبالتالي يجب على مستدعي الوظيفة عدم إغلاق واصف الملف. عندما لا يتم تحديد العلامة، لن تقوم sd_journal_close() بإغلاق واصف الملف، لذا يجب على المستدعي إغلاقه بعد sd_journal_close().

sd_journal_open_files() مشابه لـ sd_journal_open() ولكنه يأخذ قائمة مسارات ملفات منتهية بـ NULL لفتحها. سيتم فتح جميع الملفات وتوزيعها تلقائيًا. يأخذ هذا الاستدعاء أيضًا وسيط أعلام، ولكن يجب تمريره كـ 0 حيث لا توجد علامات مفهومة حاليًا لهذا الاستدعاء. يرجى ملاحظة أنه في حالة السجل المباشر، هذه الوظيفة مفيدة فقط للتصحيح، لأن ملفات السجل الفردية يمكن تدويرها في أي لحظة، وفتح ملفات محددة غير آمن بطبيعته.

sd_journal_open_files_fd() مشابه لـ sd_journal_open_files() ولكنه يأخذ مصفوفة من واصفات الملفات المفتوحة التي يجب أن تشير إلى ملفات السجل، بدلاً من مصفوفة من مسارات نظام الملفات. مرر مصفوفة واصفات الملفات كوسيط ثانٍ، وعدد عناصر المصفوفة في الثالث. يجب تمرير وسيط الأعلام كـ 0.

لا يمكن استخدام كائنات sd_journal في الطفل بعد عملية fork. الوظائف التي تأخذ كائن سجل كوسيط (sd_journal_next() وغيرها) ستعيد -ECHILD بعد عملية fork.

sd_journal_close() سيغلق سياق السجل الذي تم تخصيصه بواسطة sd_journal_open() أو sd_journal_open_directory() وسيحرر موارده.

عند فتح السجل، سيتم فتح ملفات السجل التي يمكن للمستخدم المتصل الوصول إليها فقط. إذا لم تكن ملفات السجل متاحة للمستدعي، فسيتم تجاهل ذلك بصمت.

انظر sd_journal_next(3) للحصول على مثال حول كيفية التكرار عبر السجل بعد فتحه باستخدام sd_journal_open().

يشير كائن سياق السجل الذي تم إرجاعه بواسطة sd_journal_open() إلى إدخال سجل محدد كإدخال حالى، مشابه لمؤشر ملف في ملف نظام ملفات كلاسيكي، ولكن بدون مواضع مطلقة. يمكن تعديله بواسطة sd_journal_next(3) و sd_journal_seek_head(3) واستدعاءات ذات صلة. قد يتم تصدير موضع الإدخال الحالي في سلاسل cursor، كما هو متاح عبر sd_journal_get_cursor(3). يمكن استخدام سلاسل المؤشر لتحديد إدخال سجل معين عالميًا بطريقة مستقرة ثم لاحقًا للبحث عنه (أو إذا كان الإدخال المحدد غير متاح محليًا، إلى أقرب إدخال له في الوقت المناسب) sd_journal_seek_cursor(3).

الإخطار بتغييرات السجل متاح عبر sd_journal_get_fd() والاستدعاءات ذات الصلة.

قيمة الإرجاع

تعيد استدعاءات sd_journal_open()، sd_journal_open_directory()، و sd_journal_open_files() القيمة 0 عند النجاح أو رمز خطأ سلبي بأسلوب errno. sd_journal_close() لا تعيد شيئًا.

ملاحظات

جميع الوظائف المدرجة هنا غير حساسة للخيوط (thread-agnostic) ويمكن لخيط واحد فقط العمل على كائن معين في أي وقت. قد تصل خيوط مختلفة إلى نفس الكائن في أوقات مختلفة. يمكن استخدام كائنات مستقلة متعددة من خيوط مختلفة بالتوازي.

الوظائف الموضحة هنا متاحة كمكتبة مشتركة، والتي يمكن تجميعها وربطها باستخدام ملف libsystemd pkg-config(1).
