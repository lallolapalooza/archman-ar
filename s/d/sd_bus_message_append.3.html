sd_bus_message_append(3)

الحزمة: systemd-libs

التراخيص: LGPL-2.1-or-later, CC0-1.0, GPL-2.0-or-later WITH Linux-syscall-note

SD_BUS_MESSAGE_APPEND(3) sd_bus_message_append SD_BUS_MESSAGE_APPEND(3)

الاسم

sd_bus_message_append, sd_bus_message_appendv - إرفاق الحقول برسالة D-Bus بناءً على سلسلة الأنواع

الملخص

#include <systemd/sd-bus.h>

int sd_bus_message_append(sd_bus_message *m, const char *types, ...);

int sd_bus_message_appendv(sd_bus_message *m, const char *types, va_list ap);

الوصف

تقوم الدالة sd_bus_message_append() بإلحاق سلسلة من الحقول بكائن رسالة D-Bus m. تصف سلسلة الأنواع types أنواع وسائط الحقول التي تليها. لكل نوع محدد في سلسلة الأنواع، يجب تحديد وسيط واحد أو أكثر، بنفس الترتيب المعلن في سلسلة الأنواع.

تتكون سلسلة الأنواع من العناصر الموضحة في الجدول أدناه. تحتوي على صفر أو أكثر من "الأنواع الكاملة" المفردة. قد يكون كل نوع كامل أحد الأنواع الأساسية أو نوع حاوية موصوف بالكامل. قد يكون نوع الحاوية بنية مع الأنواع المحتواة، أو متغيرًا، أو مصفوفة مع نوع عنصرها، أو مدخل قاموس مع الأنواع المحتواة. سلسلة الأنواع منتهية بـ NUL.

في حالة النوع الأساسي، يُتوقع وسيط واحد من النوع المقابل.

يُشار إلى البنية بسلسلة من الأنواع الكاملة بين "(" و ")". لا يمكن أن تكون هذه السلسلة فارغة — يجب أن تحتوي على نوع واحد على الأقل. تتبع الوسائط المقابلة لهذه السلسلة المتداخلة نفس القواعد كما لو لم تكن متداخلة.

يُشار إلى المتغير بـ "v". يجب أن تبدأ الوسائط المقابلة بسلسلة أنواع تشير إلى نوع كامل، وبعد ذلك، وسائط مقابلة للنوع المحدد.

يُشار إلى المصفوفة بـ "a" متبوعة بنوع كامل. يجب أن تبدأ الوسائط المقابلة بعدد العناصر في المصفوفة، متبوعة بالعناصر نفسها، بما يطابق نوع عنصر المصفوفة.

القاموس هو مصفوفة من مدخلات القاموس، يُشار إليه بـ "a" متبوعة بزوج من الأنواع الكاملة بين "{" و "}". يجب أن يكون أول هذه الأنواع نوعًا أساسيًا. يجب أن تبدأ الوسائط المقابلة بعدد مدخلات القاموس، متبوعة بزوج من القيم لكل مدخل بما يطابق نوع عنصر مدخلات القاموس.

sd_bus_message_appendv() مكافئة لـ sd_bus_message_append()، باستثناء أنها تُستدعى بـ "va_list" بدلاً من عدد متغير من الوسائط. لا تستدعي هذه الدالة الماكرو va_end(). ولأنها تستدعي الماكرو va_arg()، فإن قيمة ap غير معرّفة بعد الاستدعاء.

لمزيد من التفاصيل حول نظام أنواع D-Bus، يُرجى الرجوع إلى مواصفات D-Bus[1].

بالنسبة للأنواع
"s" و "g" (سلسلة يونيكود أو توقيع)، قد يكون المؤشر NULL، وهو ما يعادل سلسلة فارغة. بالنسبة لـ "h" (واصف ملف UNIX)، يتم تكرار الواصف بواسطة هذا الاستدعاء ويبقى الواصف المُمرَّر في حيازة المستدعي. راجع sd_bus_message_append_basic(3) للتفسير الدقيق لهذه الأنواع وغيرها.

قواعد سلسلة الأنواع

types ::= complete_type*
complete_type ::= basic_type | variant | structure | array | dictionary
basic_type ::= "y" | "n" | "q" | "u" | "i" | "x" | "t" | "d" |
               "b" | "h" |
               "s" | "o" | "g"
variant ::= "v"
structure ::= "(" complete_type+ ")"
array ::= "a" complete_type
dictionary ::= "a" "{" basic_type complete_type "}"

أمثلة

إلحاق نوع أساسي واحد (السلسلة "a string"):

sd_bus_message *m;
...
sd_bus_message_append(m, "s", "a string");

إلحاق جميع أنواع الأعداد الصحيحة:

uint8_t y = 1;
int16_t n = 2;
uint16_t q = 3;
int32_t i = 4;
uint32_t u = 5;
int32_t x = 6;
uint32_t t = 7;
double d = 8.0;
sd_bus_message_append(m, "ynqiuxtd", y, n, q, i, u, x, t, d);

إلحاق بنية مكونة من سلسلة ومسار D-Bus:

sd_bus_message_append(m, "(so)", "a string", "/a/path");

إلحاق مصفوفة من واصفات ملف UNIX:

sd_bus_message_append(m, "ah", 3, STDIN_FILENO, STDOUT_FILENO, STDERR_FILENO);

إلحاق متغير، بالنوع الحقيقي "g" (توقيع)، والقيمة "sdbusisgood":

sd_bus_message_append(m, "v", "g", "sdbusisgood");

إلحاق قاموس يحتوي على التعيين {1=>"a", 2=>"b", 3=>""}:

sd_bus_message_append(m, "a{is}", 3, 1, "a", 2, "b", 3, NULL);

القيمة المُرجعة

عند النجاح، تُرجع هذه الدوال عددًا صحيحًا غير سالب. وعند الفشل، تُرجع رمز خطأ بأسلوب errno سالبًا.

الأخطاء

قد تشير الأخطاء المُرجعة إلى المشكلات التالية:

-EINVAL

المعامل المحدد غير صالح.

-EPERM

تم ختم الرسالة.

-ESTALE

الرسالة في حالة غير صالحة.

-ENXIO

لا يمكن إلحاق الرسالة.

-ENOMEM

فشل تخصيص الذاكرة.

ملاحظات

الدوال الموصوفة هنا متاحة كمكتبة مشتركة، يمكن تجميعها وربطها باستخدام ملف libsystemd pkg-config(1).

يستخدم الكود الموصوف هنا getenv(3)، والذي يُعلن أنه غير آمن متعدد الخيوط. هذا يعني أن الكود الذي يستدعي الدوال الموصوفة هنا يجب ألا يستدعي setenv(3) من خيط متوازٍ. يُنصح بإجراء استدعاءات setenv() فقط من مرحلة مبكرة من البرنامج عندما لا تكون قد بدأت خيوط أخرى.
