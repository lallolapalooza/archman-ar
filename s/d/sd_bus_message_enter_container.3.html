sd_bus_message_enter_container(3)

الحزمة: systemd-libs

التراخيص: LGPL-2.1-or-later, CC0-1.0, GPL-2.0-or-later WITH Linux-syscall-note

SD_BUS_MESSAGE_OPEN_CONTAINER(3) sd_bus_message_open_container SD_BUS_MESSAGE_OPEN_CONTAINER(3)

الاسم

sd_bus_message_open_container, sd_bus_message_close_container, sd_bus_message_enter_container, sd_bus_message_exit_container - إنشاء حاويات والانتقال بينها في رسائل D-Bus

الملخص

#include <systemd/sd-bus.h>

int sd_bus_message_open_container(sd_bus_message *m, char type, const char *contents);

int sd_bus_message_close_container(sd_bus_message *m);

int sd_bus_message_enter_container(sd_bus_message *m, char type, const char *contents);

int sd_bus_message_exit_container(sd_bus_message *m);

الوصف

sd_bus_message_open_container() تضيف حاوية جديدة إلى الرسالة m. بعد فتح حاوية جديدة، يمكن ملؤها بالمحتوى باستخدام sd_bus_message_append(3) ووظائف مشابهة. تعمل الحاويات مثل المكدس. لتداخل الحاويات داخل بعضها البعض، قم باستدعاء sd_bus_message_open_container() عدة مرات دون استدعاء sd_bus_message_close_container() بينهما. سيتم تداخل كل حاوية داخل الحاوية السابقة. type يمثل نوع الحاوية ويجب أن يكون أحد "r"، "a"، "v" أو "e" كما هو موضح في sd_bus_message_append(3). بدلاً من القيم الحرفية، يمكن أيضًا استخدام الثوابت المقابلة SD_BUS_TYPE_STRUCT، SD_BUS_TYPE_ARRAY، SD_BUS_TYPE_VARIANT أو SD_BUS_TYPE_DICT_ENTRY. contents يصف نوع عناصر الحاوية ويجب أن يكون سلسلة نوع D-Bus وفقًا للقواعد الموضحة في sd_bus_message_append(3).

sd_bus_message_close_container() تغلق آخر حاوية مفتوحة باستخدام sd_bus_message_open_container(). عند النجاح، يتم وضع مؤشر الكتابة للرسالة m بعد الحاوية المغلقة في حاويتها الأم أو في m نفسها إذا لم تكن هناك حاوية أم.

sd_bus_message_enter_container() تدخل الحاوية التالية في الرسالة m للقراءة. تعمل بشكل مشابه جدًا لـ sd_bus_message_open_container(). يسمح الدخول إلى حاوية بقراءة محتوياتها باستخدام sd_bus_message_read(3) ووظائف مشابهة. type و contents هي نفسها كما في sd_bus_message_open_container().

sd_bus_message_exit_container() تخرج من نطاق آخر حاوية تم الدخول إليها بواسطة sd_bus_message_enter_container(). تعمل بشكل مشابه جدًا لـ sd_bus_message_close_container(). لاحظ أنه يمكن استدعاء sd_bus_message_exit_container() فقط بعد التكرار عبر جميع أعضاء الحاوية، أي قراءة أو تخطيها. استخدم sd_bus_message_skip(3) لتخطي الحقول من حاوية لكي تتمكن من الخروج من الحاوية باستخدام sd_bus_message_exit_container() دون قراءة جميع الأعضاء.

قيمة الإرجاع

عند النجاح، تعيد هذه الوظائف عددًا صحيحًا غير سالب. sd_bus_message_open_container() و sd_bus_message_close_container() تعيد 0. sd_bus_message_enter_container() تعيد 1 إذا نجحت في فتح حاوية جديدة، و 0 إذا لم يكن ذلك ممكنًا بسبب الوصول إلى نهاية الحاوية المفتوحة حاليًا أو الرسالة. sd_bus_message_exit_container() تعيد 1 عند النجاح. عند الفشل، تعيد جميع هذه الوظائف رمز خطأ سالب بأسلوب errno.

الأخطاء

قد تشير الأخطاء المرجعة إلى المشاكل التالية:

-EINVAL

m أو contents هي NULL أو type غير صالحة.

تمت الإضافة في الإصدار 246.

-EBADMSG

الرسالة m ذات بنية غير صالحة.

تمت الإضافة في الإصدار 254.

-ENXIO

الرسالة m لا تحتوي على حاوية من النوع type في الموضع الحالي، أو أن المحتويات لا تتطابق مع contents.

تمت الإضافة في الإصدار 254.

-EPERM

الرسالة m مختومة بالفعل.

تمت الإضافة في الإصدار 246.

-ESTALE

الرسالة m في حالة غير صالحة.

تمت الإضافة في الإصدار 246.

-ENOMEM

فشل تخصيص الذاكرة.

تمت الإضافة في الإصدار 246.

-EBUSY

تم استدعاء sd_bus_message_exit_container() ولكن لا تزال هناك أعضاء غير مقروءة متبقية في الحاوية.

تمت الإضافة في الإصدار 247.

ملاحظات

الوظائف الموضحة هنا متاحة كمكتبة مشتركة، والتي يمكن تجميعها ضدها وربطها بملف libsystemd pkg-config(1).

الكود الموضح هنا يستخدم getenv(3)، والذي تم الإعلان عنه بأنه ليس آمنًا للاستخدام مع الخيوط المتعددة. يعني هذا أن الكود الذي يستدعي الوظائف الموضحة هنا يجب ألا يستدعي setenv(3) من خيط متوازي. يوصى بإجراء استدعاءات لـ setenv() فقط من مرحلة مبكرة من البرنامج عندما لم يتم بدء أي خيوط أخرى.

أمثلة

مثال 1. إضافة مصفوفة من السلاسل النصية إلى رسالة

/* SPDX-License-Identifier: MIT-0 */
#include <systemd/sd-bus.h>
int append_strings_to_message(sd_bus_message *m, const char *const *arr) {
  const char *s;
  int r;
  r = sd_bus_message_open_container(m, 'a', "s");
  if (r < 0)
    return r;
  for (s = *arr; *s; s++) {
    r = sd_bus_message_append(m, "s", s);
    if (r < 0)
      return r;
  }
  return sd_bus_message_close_container(m);
}

مثال 2. قراءة مصفوفة من السلاسل النصية من رسالة

/* SPDX-License-Identifier: MIT-0 */
#include <stdio.h>
#include <systemd/sd-bus.h>
int read_strings_from_message(sd_bus_message *m) {
  int r;
  r = sd_bus_message_enter_container(m, 'a', "s");
  if (r < 0)
    return r;
  for (;;) {
    const char *s;
    r = sd_bus_message_read(m, "s", &s);
    if (r < 0)
      return r;
    if (r == 0)
      break;
    printf("%s\n", s);
  }
  return sd_bus_message_exit_container(m);
}
