sd_bus_message_unref(3)

الحزمة: systemd-libs

التراخيص: LGPL-2.1-or-later, CC0-1.0, GPL-2.0-or-later WITH Linux-syscall-note

SD_BUS_MESSAGE_NEW(3) sd_bus_message_new SD_BUS_MESSAGE_NEW(3)

الاسم

sd_bus_message_new, sd_bus_message_ref, sd_bus_message_unref, sd_bus_message_unrefp, SD_BUS_MESSAGE_METHOD_CALL, SD_BUS_MESSAGE_METHOD_RETURN, SD_BUS_MESSAGE_METHOD_ERROR, SD_BUS_MESSAGE_SIGNAL, sd_bus_message_get_bus - إنشاء كائن رسالة ناقل جديدة وإنشاء أو تدمير مراجع لها

ملخص

#include <systemd/sd-bus.h>
enum {
      SD_BUS_MESSAGE_METHOD_CALL,
      SD_BUS_MESSAGE_METHOD_RETURN,
      SD_BUS_MESSAGE_METHOD_ERROR,
      SD_BUS_MESSAGE_SIGNAL,
};

int sd_bus_message_new(sd_bus *bus, sd_bus_message **m, uint8_t type);

sd_bus_message *sd_bus_message_ref(sd_bus_message *m);

sd_bus_message *sd_bus_message_unref(sd_bus_message *m);

void sd_bus_message_unrefp(sd_bus_message **mp);

sd_bus *sd_bus_message_get_bus(sd_bus_message *m);

الوصف

sd_bus_message_new() ينشئ كائن رسالة ناقل جديد متصل بالناقل bus ويعيده في المعامل الإخراجي m. هذا الكائن يتم عد مرافعه، وسيتم تدميره عندما تختفي جميع المراجع. في البداية، يمتلك المستدعي لهذه الدالة المرجع الوحيد لكائن الرسالة. لاحظ أن كائن الرسالة يحتفظ بمرجع لكائن الناقل، لذلك لن يتم تدمير كائن الناقل طالما كانت الرسالة موجودة.

ملاحظة: هذا استدعاء منخفض المستوى. في معظم الحالات، تكون الدوال مثل sd_bus_message_new_method_call(3)، sd_bus_message_new_method_error(3)، sd_bus_message_new_method_return(3)، وsd_bus_message_new_signal(3) التي تنشئ رسالة من نوع معين وتُهيئ حقولًا مختلفة، أسهل في الاستخدام.

يحدد المعامل type نوع الرسالة. يجب أن يكون أحد الخيارات التالية: SD_BUS_MESSAGE_METHOD_CALL - استدعاء طريقة، SD_BUS_MESSAGE_METHOD_RETURN - رد على استدعاء طريقة، SD_BUS_MESSAGE_METHOD_ERROR - رد خطأ على استدعاء طريقة، SD_BUS_MESSAGE_SIGNAL - رسالة بث بدون رد.

يتم تهيئة علامة السماح بالتفويض التفاعلي بناءً على القيمة الحالية المعينة في كائن الناقل، انظر sd_bus_set_allow_interactive_authorization(3). يمكن تغيير ذلك باستخدام sd_bus_message_set_allow_interactive_authorization(3).

sd_bus_message_ref() يزيد عداد المراجع الداخلي لـ m بمقدار واحد.

sd_bus_message_unref() يقلل عداد المراجع الداخلي لـ m بمقدار واحد. بمجرد انخفاض عدد المراجع إلى الصفر، يتم تدمير كائن الرسالة ولا يمكن استخدامه مرة أخرى، لذا فإن أي استدعاءات إضافية لـ sd_bus_message_ref() أو sd_bus_message_unref() غير قانونية.

sd_bus_message_unrefp() مشابهة لـ sd_bus_message_unref() ولكنها تأخذ مؤشرًا إلى مؤشر لكائن sd_bus_message. هذه الدالة مفيدة بالاقتران مع سمة متغير التنظيف[1] في GCC و LLVM. انظر sd_bus_new(3) للحصول على مثال حول كيفية استخدام سمة التنظيف.

sd_bus_message_ref() و sd_bus_message_unref() لا تقومان بأي عملية إذا كان عنوان كائن رسالة الناقل المُمرر هو NULL. sd_bus_message_unrefp() ستقوم أولاً بإلغاء مرجعية الوسيط الخاص بها، والذي يجب ألا يكون NULL، ولن تنفذ أي عملية إذا كان ذلك NULL.

sd_bus_message_get_bus() تعيد كائن الناقل الذي ترتبط به الرسالة m.

قيمة الإرجاع

عند النجاح، تعيد sd_bus_message_new() القيمة 0 أو عددًا صحيحًا موجبًا. عند الفشل، تعيد رمز خطأ سالبًا بنمط errno.

sd_bus_message_ref() تعيد دائمًا الوسيط.

sd_bus_message_unref() تعيد دائمًا NULL.

sd_bus_message_get_bus() تعيد دائمًا كائن الناقل.

الأخطاء

قد تشير الأخطاء المرتجعة إلى المشاكل التالية:

-EINVAL

النوع type المحدد غير صالح.

-ENOTCONN

الوسيط bus هو NULL أو الناقل غير متصل.

-ENOMEM

فشل تخصيص الذاكرة.

ملاحظات

الدوال الموصوفة هنا متاحة كمكتبة مشتركة، والتي يمكن تجميعها وربطها باستخدام ملف pkg-config(1) الخاص بـ libsystemd.

يستخدم الكود الموصوف هنا getenv(3)، والذي يُعلن أنه غير آمن للاستخدام في خيوط متعددة. هذا يعني أن الكود الذي يستدعي الدوال الموصوفة هنا يجب ألا يستدعي setenv(3) من خيط متوازي. يُنصح فقط بإجراء استدعاءات لـ setenv() من مرحلة مبكرة من البرنامج عندما لم تبدأ خيوط أخرى.
