sd_device_enumerator_new(3)

الحزمة: systemd-libs

SD_DEVICE_ENUMERATOR_NEW(3) sd_device_enumerator_new SD_DEVICE_ENUMERATOR_NEW(3)

الاسم

sd_device_enumerator_new, sd_device_enumerator_ref, sd_device_enumerator_unref, sd_device_enumerator_unrefp - إنشاء، إسناد مرجعي، وتحرير كائن تعداد الأجهزة

الملخص

#include <systemd/sd-device.h>

int sd_device_enumerator_new(sd_device_enumerator **ret);

sd_device_enumerator* sd_device_enumerator_ref(sd_device_enumerator *enumerator);

sd_device_enumerator* sd_device_enumerator_unref(sd_device_enumerator *enumerator);

void sd_device_enumerator_unrefp(sd_device_enumerator **enumerator);

الوصف

توفر عائلة دوال sd_device_enumerator طريقة للتكرار فوق الأجهزة التي تم التعرف عليها بواسطة systemd-udevd(8). يسمح التعداد بتصفية ومطابقة الأجهزة حسب النظام الفرعي، والخصائص، والسمات الأخرى.

sd_device_enumerator_new() تنشئ كائن تعداد أجهزة جديد وتخزن النتيجة في المؤشر المشار إليه بواسطة ret. تعيد 0 عند النجاح، أو رمز خطأ سالب بأسلوب errno عند الفشل.

sd_device_enumerator_ref() تزيد عدد المراجع للكائن enumerator المحدد بمقدار واحد.

sd_device_enumerator_unref() تقلل عدد المراجع للكائن enumerator بمقدار واحد. عندما يصل عدد المراجع إلى الصفر، يتم تدمير كائن التعداد ولا يمكن استخدامه بعد ذلك، لذلك فإن الاستدعاءات اللاحقة لـ sd_device_enumerator_unref() أو sd_device_enumerator_unrefp() غير قانونية.

sd_device_enumerator_unrefp() تشبه sd_device_enumerator_unref() ولكنها تأخذ مؤشرًا إلى مؤشر لكائن sd_device_enumerator. هذه الدالة مفيدة بالاقتران مع سمة متغير التنظيف[1] من GCC و LLVM. لاحظ أن هذه الدالة معرفة كدالة مضمنة. استخدم تصريحًا مثل التالي، لتخصيص كائن sd_device_enumerator يتم تحريره تلقائيًا عند مغادرة كتلة الكود:

{
  __attribute__((cleanup(sd_device_enumerator_unrefp))) sd_device_enumerator *enumerator = NULL;
  int r;
  ...
  r = sd_device_enumerator_new(&enumerator);
  if (r < 0)
    fprintf(stderr, "فشل تخصيص sd_device_enumerator: %s\n", strerror(-r));
  ...
}

sd_device_enumerator_ref() و sd_device_enumerator_unref() لا تنفذان أي عملية إذا كان enumerator هو NULL. sd_device_enumerator_unrefp() ستفك مرجع الوسيط الخاص بها أولاً، والذي يجب ألا يكون NULL، ولن تنفذ أي عملية إذا كان *ذلك* NULL.

قيمة الإرجاع

sd_device_enumerator_new() تعيد 0 عند النجاح أو رمز خطأ سالب بأسلوب errno عند الفشل.

sd_device_enumerator_ref() تعيد دائمًا مؤشر التعداد.

sd_device_enumerator_unref() تعيد دائمًا NULL.

الأخطاء

قد تشير الأخطاء التي تم إرجاعها إلى المشاكل التالية:

-ENOMEM

فشل تخصيص الذاكرة.

-EINVAL

الوسيط غير صالح.

مثال

مثال 1. استخدام sd_device_enumerator_new()

/* SPDX-License-Identifier: MIT-0 */
#include <stdio.h>
#include <systemd/sd-device.h>
int main(void) {
    sd_device_enumerator *enumerator;
    int r;
    r = sd_device_enumerator_new(&enumerator);
    if (r < 0) {
        fprintf(stderr, "فشل إنشاء التعداد: %s\n", strerror(-r));
        return 1;
    }
    sd_device_enumerator_ref(enumerator);
    sd_device_enumerator_unref(enumerator);
    sd_device_enumerator_unref(enumerator);
    return 0;
}
