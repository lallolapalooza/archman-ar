sd_bus_call_method(3)

الحزمة: systemd-libs

التراخيص: LGPL-2.1-or-later, CC0-1.0, GPL-2.0-or-later WITH Linux-syscall-note

SD_BUS_CALL_METHOD(3) sd_bus_call_method SD_BUS_CALL_METHOD(3)

الاسم

sd_bus_call_method, sd_bus_call_methodv, sd_bus_call_method_async, sd_bus_call_method_asyncv - تهيئة كائن رسالة الناقل واستدعاء طريقة D-Bus المقابلة

الملخص

#include <systemd/sd-bus.h>

typedef int (*sd_bus_message_handler_t)(sd_bus_message *m, void *userdata, sd_bus_error *ret_error);

int sd_bus_call_method(sd_bus *bus, const char *destination, const char *path, const char *interface, const char *member, sd_bus_error *ret_error, sd_bus_message **reply, const char *types, ...);

int sd_bus_call_methodv(sd_bus *bus, const char *destination, const char *path, const char *interface, const char *member, sd_bus_error *ret_error, sd_bus_message **reply, const char *types, va_list ap);

int sd_bus_call_method_async(sd_bus *bus, sd_bus_slot **slot, const char *destination, const char *path, const char *interface, const char *member, sd_bus_message_handler_t callback, void *userdata, const char *types, ...);

int sd_bus_call_method_asyncv(sd_bus *bus, sd_bus_slot **slot, const char *destination, const char *path, const char *interface, const char *member, sd_bus_message_handler_t callback, void *userdata, const char *types, va_list ap);

الوصف

sd_bus_call_method() هي دالة مساعدة لتهيئة كائن رسالة الناقل واستدعاء طريقة D-Bus المقابلة. تجمع بين وظائف sd_bus_message_new_method_call(3) و sd_bus_message_append(3) و sd_bus_call(3) في استدعاء وظيفة واحدة.

sd_bus_call_method_async() هي دالة مساعدة لتهيئة كائن رسالة الناقل واستدعاء طريقة D-Bus المقابلة بشكل غير متزامن. تجمع بين وظائف sd_bus_message_new_method_call(3) و sd_bus_message_append(3) و sd_bus_call_async(3) في استدعاء وظيفة واحدة.

قيمة الإرجاع

عند النجاح، ترجع هذه الدوال عدداً صحيحاً غير سالب. عند الفشل، ترجع رمز خطأ سالب بأسلوب errno.

الأخطاء

راجع صفحات الدليل sd_bus_message_new_method_call(3) و sd_bus_message_append(3) و sd_bus_call(3) و sd_bus_call_async(3) للحصول على قائمة بالأخطاء المحتملة.

ملاحظات

الوظائف الموصوفة هنا متاحة كمكتبة مشتركة، يمكن الترجمة ضدها والربط مع ملف pkg-config(1) الخاص بـ libsystemd.

الكود الموصوف هنا يستخدم getenv(3)، والذي صُرح بأنه غير آمن للاستخدام مع الخيوط المتوازية. هذا يعني أن الكود الذي يستدعي الوظائف الموصوفة هنا يجب ألا يستدعي setenv(3) من خيط موازٍ. يوصى فقط بإجراء استدعاءات لـ setenv() من مرحلة مبكرة من البرنامج عندما لم يتم بدء أي خيوط أخرى.

أمثلة

مثال 1. إجراء استدعاء لطريقة D-Bus تأخذ معامل واحد

/* SPDX-License-Identifier: MIT-0 */
/* This is equivalent to:
 * busctl call org.freedesktop.systemd1 /org/freedesktop/systemd1 \
 *       org.freedesktop.systemd1.Manager GetUnitByPID $$
 *
 * Compile with 'cc print-unit-path-call-method.c -lsystemd'
 */
#include <errno.h>
#include <stdio.h>
#include <sys/types.h>
#include <unistd.h>
#include <systemd/sd-bus.h>
#define _cleanup_(f) __attribute__((cleanup(f)))
#define DESTINATION "org.freedesktop.systemd1"
#define PATH        "/org/freedesktop/systemd1"
#define INTERFACE   "org.freedesktop.systemd1.Manager"
#define MEMBER      "GetUnitByPID"
static int log_error(int error, const char *message) {
  fprintf(stderr, "%s: %s\n", message, strerror(-error));
  return error;
}
int main(int argc, char **argv) {
  _cleanup_(sd_bus_flush_close_unrefp) sd_bus *bus = NULL;
  _cleanup_(sd_bus_error_free) sd_bus_error error = SD_BUS_ERROR_NULL;
  _cleanup_(sd_bus_message_unrefp) sd_bus_message *reply = NULL;
  int r;
  r = sd_bus_open_system(&bus);
  if (r < 0)
    return log_error(r, "Failed to acquire bus");
  r = sd_bus_call_method(bus, DESTINATION, PATH, INTERFACE, MEMBER, &error, &reply, "u", (unsigned) getpid());
  if (r < 0)
    return log_error(r, MEMBER " call failed");
  const char *ans;
  r = sd_bus_message_read(reply, "o", &ans);
  if (r < 0)
    return log_error(r, "Failed to read reply");
  printf("Unit path is \"%s\".\n", ans);
  return 0;
}

هذا يعرف برنامجاً ذا فائدة دنيا سيفتح اتصالاً بالناقل، ويستدعي طريقة، وينتظر الرد، وأخيراً يستخرج ويطبع الإجابة. يقوم بمعالجة الأخطاء وإدارة الذاكرة بشكل صحيح.

NAME

sd_bus_call_method, sd_bus_call_method_async, sd_bus_call_method_delays, sd_bus_call_method_strv, sd_bus_call_method_alloc - Invoke a bus method

                                     const char *types, ...);

DESCRIPTION

sd_bus_call_method() may be used to invoke a D-Bus method on a remote object. bus is the connection to use. destination is the bus name to invoke the method on. path is the object path to invoke the method on. interface is the interface name, and member is the method name. ret_error is an error object to store any error that occurs. If the method call is successful, the return value is positive. If an error occurs, the return value is negative and ret_error is filled.

The arguments to the method are passed after types. types is a string describing the argument types, similar to the format string in sd_bus_message_append(). The arguments are appended to the message in the order they are specified.

sd_bus_call_method_alloc() is similar to sd_bus_call_method(), but it allocates and returns the reply message in reply_message. The caller is responsible for unref'ing the message.

sd_bus_call_method_async() is similar to sd_bus_call_method(), but it invokes the method asynchronously. The result will be delivered to the callback function. The செயல argument should be a valid sd_bus_ செயல_t object. The userdata is passed to the callback.

sd_bus_call_method_delays() is similar to sd_bus_call_method_alloc(), but it includes a timeout. max_usec specifies the maximum time in microseconds to wait for a reply. If the timeout is reached, -EAGAIN is returned.

sd_bus_call_method_strv() is similar to sd_bus_call_method(), but it accepts the arguments as a NULL-terminated array of strings argv. This is useful for passing string arguments.

ARGUMENTS

  • bus: A pointer to the connection object.
  • destination: The bus name to send the call to.
  • path: The object path.
  • interface: The interface name.
  • member: The method name.
  • ret_error: Pointer to an sd_bus_error object to store errors.
  • callback: A callback function to handle asynchronous replies.
  • userdata: User data for the callback function.
  • reply_message: Pointer to an sd_bus_message pointer to store the reply message.
  • max_usec: Maximum time to wait for a reply in microseconds.
  • types: A D-Bus type string describing the arguments.
  • ...: The arguments to be passed to the method.
  • argv: A NULL-terminated array of strings for arguments.

RETURN VALUE

On success, a positive value is returned. On failure, a negative value is returned and ret_error is filled with details about the error.

NOTES

The types string must be correctly formatted according to the D-Bus type system. For example, for a method that takes an integer and a string, types would be "is".

The arguments are appended to the message in the order they appear in the ... or argv.

For asynchronous calls, the callback function will be invoked with the reply message or an error. The caller should handle the message or error appropriately.

EXAMPLES

Example of a synchronous call:


#include <stdio.h>
#include <stdlib.h>
#include <systemd/sd-bus.h>

int main(int argc, char *argv[]) {
    sd_bus_error error = SD_BUS_ERROR_NULL;
    sd_bus_message *reply = NULL;
    sd_bus *bus = NULL;
    int r;

    r = sd_bus_open_user(&bus);
    if (r < 0) {
        fprintf(stderr, "Failed to connect to system bus: %s\n", strerror(-r));
        return r;
    }

    r = sd_bus_call_method(bus,
                           "org.freedesktop.DBus",
                           "/org/freedesktop/DBus",
                           "org.freedesktop.DBus",
                           "Hello",
                           &error,
                           "s",
                           "World");
    if (r < 0) {
        fprintf(stderr, "Failed to call method: %s\n", error.message);
        sd_bus_error_free(&error);
        sd_bus_unref(bus);
        return r;
    }

    printf("Method call successful!\n");

    sd_bus_unref(bus);
    return 0;
}
