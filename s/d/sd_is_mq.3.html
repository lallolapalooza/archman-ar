sd_is_mq(3)

الحزمة: systemd-libs

التراخيص: LGPL-2.1-or-later, CC0-1.0, GPL-2.0-or-later WITH Linux-syscall-note

SD_IS_FIFO(3) sd_is_fifo SD_IS_FIFO(3)

الاسم

sd_is_fifo, sd_is_socket, sd_is_socket_inet, sd_is_socket_unix, sd_is_socket_sockaddr, sd_is_mq, sd_is_special - التحقق من نوع واصف الملف

الملخص

#include <systemd/sd-daemon.h>

int sd_is_fifo(int fd, const char *path);

int sd_is_socket(int fd, int family, int type, int listening);

int sd_is_socket_inet(int fd, int family, int type, int listening, uint16_t port);

int sd_is_socket_sockaddr(int fd, int type, const struct sockaddr *addr, unsigned addr_len, int listening);

int sd_is_socket_unix(int fd, int type, int listening, const char *path, size_t length);

int sd_is_mq(int fd, const char *path);

int sd_is_special(int fd, const char *path);

الوصف

sd_is_fifo() يمكن استدعاؤها للتحقق مما إذا كان واصف الملف المحدد يشير إلى FIFO أو أنبوب. إذا كان المعامل path ليس NULL، يتم التحقق مما إذا كان FIFO مرتبطًا بمسار نظام الملفات المحدد.

sd_is_socket() يمكن استدعاؤها للتحقق مما إذا كان واصف الملف المحدد يشير إلى مقبس. إذا كان المعامل family ليس AF_UNSPEC، يتم التحقق مما إذا كان المقبس من العائلة المحددة (AF_UNIX، AF_INET، ...). إذا كان المعامل type ليس 0، يتم التحقق مما إذا كان المقبس من النوع المحدد (SOCK_STREAM، SOCK_DGRAM، ...). إذا كان المعامل listening موجبًا، يتم التحقق مما إذا كان المقبس في وضع القبول، أي تم استدعاء listen() له. إذا كان listening يساوي 0، يتم التحقق مما إذا كان المقبس ليس في هذا الوضع. إذا كان المعامل سالبًا، فلا يتم إجراء هذا الفحص. يجب استخدام المعامل listening فقط لمقابس التدفق ويجب تعيينه إلى قيمة سالبة بخلاف ذلك.

sd_is_socket_inet() مشابهة لـ sd_is_socket()، ولكنها تتحقق اختياريًا من رقم منفذ IPv4 أو IPv6 الذي يرتبط به المقبس، ما لم يكن port صفرًا. بالنسبة لهذا الاستدعاء، يجب تمرير family كـ AF_UNSPEC، أو AF_INET، أو AF_INET6.

sd_is_socket_sockaddr() مشابهة لـ sd_is_socket_inet()، ولكنها تتحقق مما إذا كان المقبس مرتبطًا بالعنوان المحدد بواسطة addr. يجب أن تكون family المحددة بواسطة addr إما AF_INET أو AF_INET6 ويجب أن يكون addr_len كبيرًا بما يكفي لتلك العائلة. إذا حدد addr منفذًا غير صفري، يتم أيضًا التحقق مما إذا كان المقبس مرتبطًا بهذا المنفذ. بالإضافة إلى ذلك، بالنسبة لـ IPv6، إذا حدد addr sin6_flowinfo أو sin6_scope_id غير صفريين، يتم التحقق مما إذا كان المقبس يحتوي على نفس القيم.

sd_is_socket_unix() مشابهة لـ sd_is_socket() ولكنها تتحقق اختياريًا من مسار AF_UNIX الذي يرتبط به المقبس، ما لم يكن المعامل path هو NULL. بالنسبة لمقابس نظام الملفات العادية AF_UNIX، قم بتعيين المعامل length إلى 0. بالنسبة لمقابس مساحة الاسم المجردة في Linux، قم بتعيين length إلى حجم العنوان، بما في ذلك البايت الأولي 0، وقم بتعيين path إلى البايت الأولي 0 لعنوان المقبس.

sd_is_mq() يمكن استدعاؤها للتحقق مما إذا كان واصف الملف المحدد يشير إلى طابور رسائل POSIX. إذا كان المعامل path ليس NULL، يتم التحقق مما إذا كان طابور الرسائل مرتبطًا بالاسم المحدد.

sd_is_special() يمكن استدعاؤها للتحقق مما إذا كان واصف الملف المحدد يشير إلى ملف خاص. إذا كان المعامل path ليس NULL، يتم التحقق مما إذا كان واصف الملف مرتبطًا باسم الملف المحدد. الملفات الخاصة في هذا السياق هي عقد جهاز الحرف وملفات في /proc/ أو /sys/.

قيمة الإرجاع

عند الفشل، تعيد هذه الاستدعاءات رمز خطأ سالبًا على نمط errno. إذا كان واصف الملف من النوع المحدد ومرتبطًا بالعنوان المحدد، يتم إرجاع قيمة موجبة، وإلا يتم إرجاع صفر.

ملاحظات

الوظائف الموصوفة هنا متاحة كمكتبة مشتركة، والتي يمكن تجميعها والربط بها باستخدام ملف libsystemd pkg-config(1).

داخليًا، تستخدم هذه الوظائف مزيجًا من fstat(2) و getsockname(2) للتحقق من نوع واصف الملف والمكان الذي يرتبط به.
