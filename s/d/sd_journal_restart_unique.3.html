sd_journal_restart_unique(3)

الحزمة: systemd-libs

التراخيص: LGPL-2.1-or-later, CC0-1.0, GPL-2.0-or-later WITH Linux-syscall-note

SD_JOURNAL_QUERY_UNIQUE(3) sd_journal_query_unique SD_JOURNAL_QUERY_UNIQUE(3)

الاسم

sd_journal_query_unique, sd_journal_enumerate_unique, sd_journal_enumerate_available_unique, sd_journal_restart_unique, SD_JOURNAL_FOREACH_UNIQUE - قراءة حقول بيانات فريدة من السجل

الملخص

#include <systemd/sd-journal.h>

int sd_journal_query_unique(sd_journal *j, const char *field);

int sd_journal_enumerate_available_unique(sd_journal *j, const void **data, size_t *length);

int sd_journal_enumerate_unique(sd_journal *j, const void **data, size_t *length);

void sd_journal_restart_unique(sd_journal *j);

SD_JOURNAL_FOREACH_UNIQUE(sd_journal *j, const void *data, size_t length);

الوصف

sd_journal_query_unique() تستعلم السجل عن جميع القيم الفريدة التي يمكن أن يأخذها الحقل المحدد. تأخذ وسيطين: السجل للاستعلام واسم الحقل للبحث عنه. يتم سرد أسماء الحقول المعروفة في systemd.journal-fields(7)، ولكن يمكن تحديد أي حقل. يجب تحديد أسماء الحقول بدون حرف "=" لاحق. بعد أن يتم تنفيذ هذه الدالة بنجاح، يمكن الاستعلام عن قيم الحقول باستخدام sd_journal_enumerate_unique() و sd_journal_enumerate_available_unique(). سيؤدي استدعاء أحد هذه الاستدعاءات إلى تغيير اسم الحقل الذي يتم الاستعلام عنه وإعادة تعيين مؤشر التعداد إلى أول قيمة حقل تطابق.

يمكن استخدام sd_journal_enumerate_unique() للتكرار عبر جميع حقول البيانات التي تطابق اسم الحقل المحدد مسبقًا كما تم تعيينه باستخدام sd_journal_query_unique(). في كل استدعاء، يتم إرجاع بيانات الحقل التالية المطابقة. ترتيب حقول البيانات المرجعة غير محدد. تأخذ ثلاث وسائط: كائن السجل، بالإضافة إلى زوج من المؤشرات إلى متغيرات مؤشر/حجم حيث يجب تخزين كائن البيانات وحجمه. البيانات المرجعة موجودة في خريطة ذاكرة للقراءة فقط وهي صالحة فقط حتى الاستدعاء التالي لـ sd_journal_enumerate_unique(). لاحظ أن البيانات المرجعة سيسبقها اسم الحقل وعلامة "=". لاحظ أن هذا الاستدعاء يخضع للحد الأقصى لحجم حقل البيانات الذي يتم التحكم فيه بواسطة sd_journal_set_data_threshold() ويتم إرجاع الجزء الأولي فقط من الحقل حتى الحد الأقصى. يتم إرجاع خطأ للحقول التي لا يمكن استرجاعها. انظر قائمة الأخطاء أدناه للحصول على التفاصيل.

sd_journal_enumerate_available_unique() مشابه لـ sd_journal_enumerate_unique()، ولكنه يتجاوز بصمت أي حقول قد تكون صالحة، ولكنها كبيرة جدًا أو غير مدعومة من قبل التنفيذ الحالي.

sd_journal_restart_unique() يعيد تعيين مؤشر تعداد البيانات إلى بداية القائمة. الاستدعاء التالي لـ sd_journal_enumerate_unique() سيعيد بيانات الحقل الأولى التي تطابق اسم الحقل مرة أخرى.

لاحظ أن الماكرو SD_JOURNAL_FOREACH_UNIQUE() يمكن استخدامه كغلاف مفيد حول sd_journal_restart_unique() و sd_journal_enumerate_available_unique().

لاحظ أن هذه الدوال حاليًا لا تتأثر بالمطابقات التي تم تعيينها باستخدام sd_journal_add_match() ولكن هذا قد يتغير في إصدار لاحق من هذا البرنامج.

لتعداد جميع أسماء الحقول المستخدمة حاليًا (وبالتالي جميع معلمات الحقل المناسبة لـ sd_journal_query_unique())، استخدم استدعاء sd_journal_enumerate_fields(3).

قيمة الإرجاع

sd_journal_query_unique() تعيد 0 عند النجاح أو رمز خطأ سالب بنمط errno. تعيد sd_journal_enumerate_unique() و sd_journal_query_available_unique() عددًا صحيحًا موجبًا إذا تم قراءة بيانات الحقل التالية، 0 عندما لا توجد حقول متبقية، أو رمز خطأ سالب بنمط errno. لا تعيد sd_journal_restart_unique() شيئًا.

الأخطاء

قد تشير الأخطاء المرجعة إلى المشكلات التالية:

-EINVAL

أحد المعلمات المطلوبة هو NULL أو غير صالح.

تمت الإضافة في الإصدار 246.

-ECHILD

تم إنشاء كائن السجل في عملية مختلفة، أو مكتبة، أو مثيل وحدة.

تمت الإضافة في الإصدار 246.

-EADDRNOTAVAIL

مؤشر القراءة ليس في إدخال صالح؛ لم يتم استدعاء sd_journal_next(3) أو استدعاء ذي صلة مرة واحدة على الأقل.

تمت الإضافة في الإصدار 246.

-ENOENT

لا يتضمن الإدخال الحالي الحقل المحدد.

تمت الإضافة في الإصدار 246.

-ENOBUFS

إدخال مضغوط كبير جدًا.

تمت الإضافة في الإصدار 246.

-E2BIG

حقل البيانات كبير جدًا بالنسبة لبنية الكمبيوتر هذه (على سبيل المثال، فوق 4 جيجابايت على بنية 32 بت).

تمت الإضافة في الإصدار 246.

-EPROTONOSUPPORT

تم ضغط السجل بطريقة غير مدعومة أو يستخدم السجل ميزة غير مدعومة.

تمت الإضافة في الإصدار 246.

-EBADMSG

السجل تالف (ربما الإدخال الذي يتم التكرار فوقه فقط).

تمت الإضافة في الإصدار 246.

-EIO

تم الإبلاغ عن خطأ في الإدخال/الإخراج بواسطة النواة.

تمت الإضافة في الإصدار 246.

أمثلة

استخدم الماكرو SD_JOURNAL_FOREACH_UNIQUE() للتكرار عبر جميع القيم التي يمكن أن يأخذها حقل السجل (والتي يمكن الوصول إليها على البنية الحالية ولا يتم ضغطها بآلية غير مدعومة). يسرد المثال التالي جميع أسماء الوحدات المشار إليها في السجل:

/* SPDX-License-Identifier: MIT-0 */
#include <errno.h>
#include <stdio.h>
#include <systemd/sd-journal.h>
int main(int argc, char *argv[]) {
  sd_journal *j;
  const void *d;
  size_t l;
  int r;
  r = sd_journal_open(&j, SD_JOURNAL_LOCAL_ONLY);
  if (r < 0) {
    fprintf(stderr, "فشل فتح السجل: %s\n", strerror(-r));
    return 1;
  }
  r = sd_journal_query_unique(j, "_SYSTEMD_UNIT");
  if (r < 0) {
    fprintf(stderr, "فشل الاستعلام عن السجل: %s\n", strerror(-r));
    return 1;
  }
  SD_JOURNAL_FOREACH_UNIQUE(j, d, l)
    printf("%.*s\n", (int) l, (const char*) d);
  sd_journal_close(j);
  return 0;
}
