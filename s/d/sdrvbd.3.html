subroutine sdrvbd (integer nsizes, integer, dimension( * ) mm, integer, dimension( * ) nn, integer ntypes, logical, dimension( * ) dotype, integer, dimension( 4 ) iseed, real thresh, real, dimension( lda, * ) a, integer lda, real, dimension( ldu, * ) u, integer ldu, real, dimension( ldvt, * ) vt, integer ldvt, real, dimension( lda, * ) asav, real, dimension( ldu, * ) usav, real, dimension( ldvt, * ) vtsav, real, dimension( * ) s, real, dimension( * ) ssav, real, dimension( * ) e, real, dimension( * ) work, integer lwork, integer, dimension( * ) iwork, integer nout, integer info)

SDRVBD

الغرض:

!>
!> يقوم SDRVBD بفحص إجراءات تحليل القيم المفردة (SVD) SGESVD, SGESDD, SGESVDQ, SGESVJ, SGEJSV, و DGESVDX.
!>
!> كل من SGESVD و SGESDD تقوم بتحليل A = U diag(S) VT، حيث U و VT متعامدان و diag(S) قطري مع مدخلات المصفوفة S على قطرها. مدخلات S هي القيم المفردة، غير سالبة ومخزنة بترتيب تنازلي. يمكن اختيار عدم حساب U و VT، أو الكتابة فوقهما في A، أو حسابهما جزئيًا.
!>
!> A هي M × N. لنفترض MNMIN = min( M, N ). حجم S هو MNMIN.
!> يمكن أن تكون U بحجم M × M أو M × MNMIN. يمكن أن تكون VT بحجم N × N أو MNMIN × N.
!>
!> عند استدعاء SDRVBD، يتم تحديد عدد من أبعاد المصفوفة (M's و N's) وعدد من أنواع المصفوفات. لكل حجم (M,N) ولكل نوع مصفوفة، وللحد الأدنى من مساحة العمل بالإضافة إلى مساحة العمل الكافية للسماح بالحجب، سيتم إنشاء مصفوفة M × N لاستخدامها في اختبار إجراءات SVD. لكل مصفوفة، سيتم تحليل A إلى A = U diag(S) VT وسيتم حساب الاختبارات الـ 12 التالية:
!>
!> اختبار لـ SGESVD:
!>
!> (1)    | A - U diag(S) VT | / ( |A| max(M,N) ulp )
!>
!> (2)    | I - U'U | / ( M ulp )
!>
!> (3)    | I - VT VT' | / ( N ulp )
!>
!> (4)    تحتوي S على MNMIN قيم غير سالبة بترتيب تنازلي.
!>        (إرجاع 0 إذا كان صحيحًا، 1/ULP إذا كان خاطئًا.)
!>
!> (5)    | U - Upartial | / ( M ulp ) حيث Upartial هي U محسوبة جزئيًا.
!>
!> (6)    | VT - VTpartial | / ( N ulp ) حيث VTpartial هي VT محسوبة جزئيًا.
!>
!> (7)    | S - Spartial | / ( MNMIN ulp |S| ) حيث Spartial هو
!>        متجه القيم المفردة من SVD الجزئي
!>
!> اختبار لـ SGESDD:
!>
!> (8)    | A - U diag(S) VT | / ( |A| max(M,N) ulp )
!>
!> (9)    | I - U'U | / ( M ulp )
!>
!> (10)   | I - VT VT' | / ( N ulp )
!>
!> (11)   تحتوي S على MNMIN قيم غير سالبة بترتيب تنازلي.
!>        (إرجاع 0 إذا كان صحيحًا، 1/ULP إذا كان خاطئًا.)
!>
!> (12)   | U - Upartial | / ( M ulp ) حيث Upartial هي U محسوبة جزئيًا.
!>
!> (13)   | VT - VTpartial | / ( N ulp ) حيث VTpartial هي VT محسوبة جزئيًا.
!>
!> (14)   | S - Spartial | / ( MNMIN ulp |S| ) حيث Spartial هو
!>        متجه القيم المفردة من SVD الجزئي
!>
!> اختبار لـ SGESVDQ:
!>
!> (36)   | A - U diag(S) VT | / ( |A| max(M,N) ulp )
!>
!> (37)   | I - U'U | / ( M ulp )
!>
!> (38)   | I - VT VT' | / ( N ulp )
!>
!> (39)   تحتوي S على MNMIN قيم غير سالبة بترتيب تنازلي.
!>        (إرجاع 0 إذا كان صحيحًا، 1/ULP إذا كان خاطئًا.)
!>
!> اختبار لـ SGESVJ:
!>
!> (15)   | A - U diag(S) VT | / ( |A| max(M,N) ulp )
!>
!> (16)   | I - U'U | / ( M ulp )
!>
!> (17)   | I - VT VT' | / ( N ulp )
!>
!> (18)   تحتوي S على MNMIN قيم غير سالبة بترتيب تنازلي.
!>        (إرجاع 0 إذا كان صحيحًا، 1/ULP إذا كان خاطئًا.)
!>
!> اختبار لـ SGEJSV:
!>
!> (19)   | A - U diag(S) VT | / ( |A| max(M,N) ulp )
!>
!> (20)   | I - U'U | / ( M ulp )
!>
!> (21)   | I - VT VT' | / ( N ulp )
!>
!> (22)   تحتوي S على MNMIN قيم غير سالبة بترتيب تنازلي.
!>        (إرجاع 0 إذا كان صحيحًا، 1/ULP إذا كان خاطئًا.)
!>
!> اختبار لـ SGESVDX( 'V', 'V', 'A' )/SGESVDX( 'N', 'N', 'A' )
!>
!> (23)   | A - U diag(S) VT | / ( |A| max(M,N) ulp )
!>
!> (24)   | I - U'U | / ( M ulp )
!>
!> (25)   | I - VT VT' | / ( N ulp )
!>
!> (26)   تحتوي S على MNMIN قيم غير سالبة بترتيب تنازلي.
!>        (إرجاع 0 إذا كان صحيحًا، 1/ULP إذا كان خاطئًا.)
!>
!> (27)   | U - Upartial | / ( M ulp ) حيث Upartial هي U محسوبة جزئيًا.
!>
!> (28)   | VT - VTpartial | / ( N ulp ) حيث VTpartial هي VT محسوبة جزئيًا.
!>
!> (29)   | S - Spartial | / ( MNMIN ulp |S| ) حيث Spartial هو
!>        متجه القيم المفردة من SVD الجزئي
!>
!> اختبار لـ SGESVDX( 'V', 'V', 'I' )
!>
!> (30)   | U' A VT''' - diag(S) | / ( |A| max(M,N) ulp )
!>
!> (31)   | I - U'U | / ( M ulp )
!>
!> (32)   | I - VT VT' | / ( N ulp )
!>
!> اختبار لـ SGESVDX( 'V', 'V', 'V' )
!>
!> (33)   | U' A VT''' - diag(S) | / ( |A| max(M,N) ulp )
!>
!> (34)   | I - U'U | / ( M ulp )
!>
!> (35)   | I - VT VT' | / ( N ulp )
!>
!> يتم تحديد أحجام المصفوفات (M's و N's) بواسطة المتجهين MM و NN؛ قيمة كل زوج من العناصر (MM(j),NN(j)) تحدد حجمًا واحدًا. يتم تحديد أنواع المصفوفات بواسطة مصفوفة منطقية DOTYPE( 1:NTYPES )؛ إذا كان DOTYPE(j) هو .TRUE.، فسيتم إنشاء مصفوفة من النوع j.
!> حاليًا، قائمة الأنواع الممكنة هي:
!>
!> (1)  المصفوفة الصفرية.
!! (2)  المصفوفة الوحدة.
!> (3)  مصفوفة من الشكل U D V، حيث U و V متعامدتان و D لها مدخلات متباعدة بالتساوي 1, ..., ULP مع علامات عشوائية
!>      على القطر.
!> (4)  نفس (3)، ولكن مضروبة في عتبة التقوض / ULP.
!> (5)  نفس (3)، ولكن مضروبة في عتبة الفائض * ULP.
!>

المعلمات

NSIZES 
!>          NSIZES هو عدد صحيح (INTEGER)
!>          عدد أبعاد المصفوفات (M,N) الموجودة في المتجهين
!>          MM و NN.
!>

MM

!>          MM هو مصفوفة أعداد صحيحة (INTEGER array)، حجمها (NSIZES)
!>          قيم البعد الصفّي للمصفوفة M.
!>

NN

!>          NN هو مصفوفة أعداد صحيحة (INTEGER array)، حجمها (NSIZES)
!>          قيم البعد العمودي للمصفوفة N.
!>

NTYPES

!>          NTYPES هو عدد صحيح (INTEGER)
!>          عدد العناصر في DOTYPE. إذا كان صفرًا، فإن SDRVBD
!>          لا تفعل شيئًا. يجب أن يكون على الأقل صفرًا. إذا كان MAXTYP+1
!>          و NSIZES هو 1، فسيتم تعريف نوع إضافي، MAXTYP+1،
!>          وهو استخدام المصفوفات الموجودة في A و B.
!>          هذا مفيد فقط إذا كانت DOTYPE(1:MAXTYP) هي .FALSE. و
!>          DOTYPE(MAXTYP+1) هي .TRUE. .
!>

DOTYPE

!>          DOTYPE هي مصفوفة منطقية (LOGICAL array)، حجمها (NTYPES)
!>          إذا كانت DOTYPE(j) هي .TRUE.، فسيتم إنشاء مصفوفة
!>          من النوع j لكل حجم (m,n). إذا كان NTYPES أصغر من
!>          الحد الأقصى لعدد الأنواع المعرفة (PARAMETER MAXTYP)، فسيتم
!>          تجاهل الأنواع NTYPES+1 إلى MAXTYP. إذا
!>          كان NTYPES أكبر من MAXTYP، فسيتم تجاهل DOTYPE(MAXTYP+1)
!>          إلى DOTYPE(NTYPES).
!>

ISEED

!>          ISEED هي مصفوفة أعداد صحيحة (INTEGER array)، حجمها (4)
!>          عند الدخول، هي بذرة مولد الأرقام العشوائية. يجب أن تكون عناصر المصفوفة
!>          بين 0 و 4095؛ إذا لم تكن كذلك، فسيتم تقليلها حسب 4096.
!>          أيضًا، يجب أن يكون ISEED(4) فرديًا.
!>          عند الخروج، يتغير ISEED ويمكن استخدامه في الاستدعاء التالي لـ
!>
