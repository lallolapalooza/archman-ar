sd_bus_message_readv(3)

الحزمة: systemd-libs

التراخيص: LGPL-2.1-or-later, CC0-1.0, GPL-2.0-or-later WITH Linux-syscall-note

SD_BUS_MESSAGE_READ(3) sd_bus_message_read SD_BUS_MESSAGE_READ(3)

الاسم

sd_bus_message_read, sd_bus_message_readv, sd_bus_message_peek_type - قراءة تسلسل من القيم من رسالة

ملخص

#include <systemd/sd-bus.h>

int sd_bus_message_read(sd_bus_message *m, const char *types, ...);

int sd_bus_message_readv(sd_bus_message *m, const char *types, va_list ap);

int sd_bus_message_peek_type(sd_bus_message *m, char *type, const char **contents);

الوصف

sd_bus_message_read() تقرأ تسلسلًا من الحقول من كائن رسالة D-Bus m وتتقدم بموضع القراءة في الرسالة. سلسلة الأنواع types تصف أنواع العناصر المتوقعة في الرسالة وحجج الحقول التي تليها. يمكن أن تكون سلسلة الأنواع NULL أو فارغة، وفي هذه الحالة لا تتم قراءة أي شيء.

تتكون سلسلة الأنواع من العناصر الموصوفة في sd_bus_message_append(3)، أي الأنواع الأساسية وأنواع الحاويات. يجب أن تحتوي على صفر أو أكثر من "الأنواع الكاملة" الفردية. سلسلة الأنواع تنتهي بـ NUL.

لكل نوع محدد في سلسلة الأنواع، يجب تحديد وسيط واحد أو أكثر بعد معامل types، بنفس الترتيب. يجب أن تكون الوسائط مؤشرات إلى أنواع مناسبة (مؤشر إلى int8_t لـ "y" في سلسلة الأنواع، مؤشر إلى int32_t لـ "i"، مؤشر إلى const char* لـ "s"، ...) والتي يتم تعيينها بناءً على القيم الموجودة في الرسالة. كاستثناء، في حالة أنواع المصفوفات والمتغيرات، فإن الوسيط الأول هو وسيط "إدخال" يحدد بشكل إضافي كيفية قراءة الرسالة. راجع الجدول أدناه للحصول على قائمة كاملة بالوسائط المسموح بها وأنواعها. لاحظ أنه إذا كان النوع الأساسي مؤشرًا (على سبيل المثال، const char * في حالة السلسلة)، فإن الوسيط هو مؤشر إلى مؤشر، كما أن قيمة المؤشر التي يتم كتابتها هي مجرد استعارة ويجب نسخ المحتويات إذا كان سيتم استخدامها بعد انتهاء عمر الرسالة. إذا كان النوع "h" (واصف ملف يونكس)، فإن الواصف لا يتم تكراره بواسطة هذا الاستدعاء ويبقى الواصف المُرجع في حوزة كائن الرسالة، ويجب تكراره بواسطة المتصل للاحتفاظ بمرجع مفتوح له بعد تحرير كائن الرسالة.

يمكن أن يكون كل وسيط أيضًا NULL، وفي هذه الحالة يتم قراءة القيمة وتجاهلها.

جدول 1. محددات عناصر النوع

محدد ثابت وصف نوع الوسيط الأول أنواع الوسائط اللاحقة، إن وجدت
"y" SD_BUS_TYPE_BYTE عدد صحيح غير موقع بحجم 8 بت uint8_t *  
"b" SD_BUS_TYPE_BOOLEAN قيمة منطقية int * (ملاحظة: ليس bool *)  
"n" SD_BUS_TYPE_INT16 عدد صحيح موقع بحجم 16 بت int16_t *  
"q" SD_BUS_TYPE_UINT16 عدد صحيح غير موقع بحجم 16 بت uint16_t *  
"i" SD_BUS_TYPE_INT32 عدد صحيح موقع بحجم 32 بت int32_t *  
"u" SD_BUS_TYPE_UINT32 عدد صحيح غير موقع بحجم 32 بت uint32_t *  
"x" SD_BUS_TYPE_INT64 عدد صحيح موقع بحجم 64 بت int64_t *  
"t" SD_BUS_TYPE_UINT64 عدد صحيح غير موقع بحجم 64 بت uint64_t *  
"d" SD_BUS_TYPE_DOUBLE فاصلة عائمة مزدوجة الدقة IEEE 754 double *  
"s" SD_BUS_TYPE_STRING سلسلة UTF-8 const char **  
"o" SD_BUS_TYPE_OBJECT_PATH سلسلة مسار كائن D-Bus const char **  
"g" SD_BUS_TYPE_SIGNATURE سلسلة توقيع D-Bus const char **  
"h" SD_BUS_TYPE_UNIX_FD واصف ملف يونكس int *  
"a" SD_BUS_TYPE_ARRAY مصفوفة int، والذي يحدد الطول المتوقع n للمصفوفة n مجموعات من الوسائط المناسبة لنوع عنصر المصفوفة
"v" SD_BUS_TYPE_VARIANT متغير سلسلة توقيع وسائط مناسبة للأنواع المحددة بالتوقيع
"(" SD_BUS_TYPE_STRUCT_BEGIN بداية مصفوفة وسائط مناسبة لعناصر الهيكل
")" SD_BUS_TYPE_STRUCT_END نهاية مصفوفة
"{" SD_BUS_TYPE_DICT_ENTRY_BEGIN بداية إدخال القاموس وسائط مناسبة للنوع الأول في الزوج وسائط مناسبة للنوع الثاني في الزوج
"}" SD_BUS_TYPE_DICT_ENTRY_END نهاية إدخال القاموس

إذا لم تكن الكائنات من الأنواع المحددة موجودة في الموضع الحالي في الرسالة، يتم إرجاع خطأ.

sd_bus_message_readv() مكافئ لـ sd_bus_message_read()، باستثناء أنه يتم استدعاؤه مع "va_list" بدلاً من عدد متغير من الوسائط. هذه الدالة لا تستدعي الماكرو va_end(). نظرًا لأنها تستدعي الماكرو va_arg()، فإن قيمة ap غير محددة بعد الاستدعاء.

sd_bus_message_peek_type() تحدد نوع العنصر التالي في m ليتم قراءته بواسطة sd_bus_message_read() أو دوال مشابهة. عند النجاح، يتم تخزين النوع في type، إذا لم يكن NULL. إذا كان النوع نوع حاوية، يتم تخزين نوع عناصره في contents، إذا لم يكن NULL. إذا نجحت هذه الدالة في تحديد نوع العنصر التالي في m، فإنها تُرجع عددًا صحيحًا موجبًا. إذا لم تكن هناك عناصر أخرى للقراءة، فإنها تُرجع صفرًا.

قيمة الإرجاع

عند النجاح، تُرجع هذه الدوال عددًا صحيحًا غير سالب. عند الفشل، تُرجع رمز خطأ سالبًا بأسلوب errno.

الأخطاء

قد تشير الأخطاء المُرجعة إلى المشاكل التالية:

-EINVAL

سلسلة النوع المحددة غير صالحة أو أن معامل الرسالة هو NULL.

-ENXIO

الرسالة لا تحتوي على النوع المحدد في الموضع الحالي.

-EBADMSG

لا يمكن تحليل الرسالة.

-EBUSY

عند القراءة من حاوية، سيتم إرجاع هذا الخطأ إذا بقيت عناصر غير مقروءة في الحاوية.

تمت الإضافة في الإصدار 251.

ملاحظات

الوظائف الموصوفة هنا متاحة كمكتبة مشتركة، والتي يمكن تجميعها معها وربطها بملف pkg-config(1) الخاص بـ libsystemd.

الكود الموصوف هنا يستخدم getenv(3)، والذي تم تعريفه على أنه غير آمن للاستخدام في خيوط متعددة. هذا يعني أن الكود الذي يستدعي الدوال الموصوفة هنا يجب ألا يستدعي setenv(3) من خيط موازٍ. يوصى بإجراء استدعاءات لـ setenv() فقط في مرحلة مبكرة من البرنامج عندما لم يتم بدء أي خيوط أخرى.
