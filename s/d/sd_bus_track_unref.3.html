sd_bus_track_unref(3)

الحزمة: systemd-libs

التراخيص: LGPL-2.1-or-later، CC0-1.0، GPL-2.0-or-later WITH Linux-syscall-note

SD_BUS_TRACK_NEW(3) sd_bus_track_new SD_BUS_TRACK_NEW(3)

الاسم

#include <systemd/sd-bus.h>

typedef int (*sd_bus_track_handler_t)(sd_bus_track *track, void *userdata);

int sd_bus_track_new(sd_bus* bus, sd_bus_track** ret, sd_bus_track_handler_t handler, void* userdata);

sd_bus_track *sd_bus_track_unref(sd_bus_track *t);

void sd_bus_track_unrefp(sd_bus_track **t);

int sd_bus_track_get_recursive(sd_bus_track *t);

int sd_bus_track_set_recursive(sd_bus_track *t, int b);

الوصف

تُنشئ sd_bus_track_new() كائن تتبع أقران ناقل جديدًا. يُخصص الكائن للناقل المحدد، ويُعاد في المعامل *ret. بعد الاستخدام، يجب تحرير الكائن مرة أخرى بإسقاط المرجع المكتسب باستخدام sd_bus_track_unref() (انظر أدناه). يمكن استخدام كائن تتبع أقران الناقل لتتبع الأقران على ناقل IPC محدد، في الحالات التي يستخدم فيها الأقران كائنًا محليًا واحدًا أو أكثر، للتحكم في دورة حياة الكائنات المحلية وضمان بقائها طالما بقي الأقران المحتاجون إليها موجودين، و إلغاء الإشارة إليها (وربما تدميرها) بمجرد اختفاء جميع الأقران المعنيين. يمكن استخدام كل كائن تتبع أقران ناقل لتتبع صفر أو واحد أو أكثر من الأقران في المرة الواحدة. تُضاف المراجع إلى أقران ناقل محددين عبر sd_bus_track_add_name(3) أو sd_bus_track_add_sender(). يمكن إسقاطها مرة أخرى عبر sd_bus_track_remove_name() و sd_bus_track_remove_sender(). بدلاً من ذلك، تُزال المراجع على الأقران تلقائيًا عند فصلهم عن الناقل. إذا كان handler غير NULL فقد يحدد دالة تُستدعى كلما أُسقط المرجع الأخير، بغض النظر عما إذا كان المرجع قد أُسقط صراحةً عبر sd_bus_track_remove_name() أو ضمنيًا لأن القرين قد فصل عن الناقل. يمكن استخدام المعامل الأخير userdata لإرفاق مؤشر بيانات مستخدم عام بالكائن. يُمرَّر هذا المؤشر إلى دالة الاستدعاء handler عند استدعائها.

تُنشئ sd_bus_track_ref() مرجعًا جديدًا لكائن تتبع أقران الناقل. لن يُدمَّر هذا الكائن حتى تُستدعى sd_bus_track_unref() بعدد المرات نفسها زائد مرة أخرى. بمجرد انخفاض عدد المراجع إلى الصفر، لا يمكن استخدام الكائن المحدد بعد الآن، والاستدعاءات الإضافية لـ sd_bus_track_ref() أو sd_bus_track_unref() على الكائن نفسه غير قانونية.

تُدمّر sd_bus_track_unref() مرجعًا لكائن تتبع أقران الناقل.

sd_bus_track_unrefp() مشابهة لـ sd_bus_track_unref() لكنها تأخذ مؤشرًا إلى مؤشر إلى كائن sd_bus_track. هذا الاستدعاء مفيد بالاقتران مع سمة المتغير للتنظيف[1] في GCC وLLVM. لاحظ أن هذه الدالة معرّفة كدالة inline.

sd_bus_track_ref() وsd_bus_track_unref() و sd_bus_track_unrefp() لا تنفذ أي عملية إذا كان كائن تتبع أقران الناقل الممرر هو NULL.

قد توجد كائنات تتبع أقران الناقل في وضعين: افتراضيًا تعمل في الوضع غير التكراري، لكن يمكن اختياريًا تبديلها إلى الوضع التكراري. إذا كانت تعمل في الوضع غير التكراري الافتراضي فإن القرين إما متتبع أو غير متتبع. في هذا الوضع فإن استدعاء sd_bus_track_add_name() عدة مرات متتالية لنفس القرين يعادل تمامًا استدعاءها مرة واحدة فقط، لأن الاستدعاء يضيف القرين إلى مجموعة الأقران المتتبعة إذا لزم الأمر، ولا ينفذ أي عملية إذا كان القرين متتبعًا بالفعل. استدعاء واحد لـ sd_bus_track_remove_name() يزيل المرجع على القرين مرة أخرى، بغض النظر عن عدد المرات التي استُدعيت فيها sd_bus_track_add_name() من قبل. إذا كانت تعمل في الوضع التكراري، فإن عدد المرات التي تُستدعى فيها sd_bus_track_add_name() لنفس اسم القرين يُحسب ويجب استدعاء sd_bus_track_remove_name() بنفس عدد المرات قبل أن يتوقف تتبع القرين، باستثناء عندما يختفي القرين المتتبع من الناقل، وفي هذه الحالة يكون العدد غير ذي صلة ويُزال تتبع القرين المحدد فورًا. يمكن استخدام sd_bus_track_get_recursive() لتحديد ما إذا كان كائن تتبع أقران الناقل يعمل في الوضع التكراري. يمكن استخدام sd_bus_track_set_recursive() لتمكين أو تعطيل الوضع التكراري. افتراضيًا، يعمل كائن تتبع أقران الناقل في الوضع غير التكراري، وبالتالي فإن sd_bus_track_get_recursive() لكائن مُخصص حديثًا تُرجع قيمة تساوي صفرًا. استخدم sd_bus_track_set_recursive() لتمكين الوضع التكراري، مباشرة بعد التخصيص. تأخذ معاملًا منطقيًا لتمكين أو تعطيل الوضع التكراري. لاحظ أن كائنات التتبع التي استُدعيت لها sd_bus_track_add_name() بالفعل مرة واحدة على الأقل (والتي تتتبع بالفعل قرينًا واحدًا أو أكثر) لا يمكن تبديلها من الوضع التكراري إلى غير التكراري بعد الآن.

تُرجع sd_bus_track_get_bus() كائن الناقل الذي ينتمي إليه كائن تتبع أقران الناقل. تُرجع كائن الناقل الممرر في البداية إلى sd_bus_track_new() عند تخصيص الكائن.

تُرجع sd_bus_track_get_userdata() مؤشر بيانات المستخدم العام المعيّن على كائن تتبع أقران الناقل وقت الإنشاء باستخدام sd_bus_track_new() أو في وقت لاحق، باستخدام sd_bus_track_set_userdata().

القيمة المُعادة

عند النجاح، تُرجع sd_bus_track_new() و sd_bus_track_set_recursive() القيمة 0 أو عددًا صحيحًا موجبًا. عند الفشل، تُرجعان رمز خطأ بأسلوب errno سالبًا.

تُرجع sd_bus_track_ref() دائمًا المعامل.

تُرجع sd_bus_track_unref() دائمًا NULL.

تُرجع sd_bus_track_get_recursive() القيمة 0 إذا كان الوضع غير التكراري محددًا (الافتراضي)، وقيمة أكبر من 0 إذا كان الوضع التكراري محددًا. عند الفشل يُرجع رمز خطأ بأسلوب errno سالبًا.

تُرجع sd_bus_track_get_bus() كائن الناقل المرتبط بكائن تتبع أقران الناقل.

تُرجع sd_bus_track_get_userdata() مؤشر بيانات المستخدم العام المرتبط بكائن تتبع أقران الناقل. تُرجع sd_bus_track_set_userdata() مؤشر بيانات المستخدم السابق المعيّن.

ملكية المرجع

تُنشئ دالة sd_bus_track_new() كائنًا جديدًا ويملك المستدعي المرجع الوحيد. عندما لا تكون هناك حاجة إليه بعد الآن، يجب تدمير هذا المرجع باستخدام sd_bus_track_unref().

الأخطاء

قد تشير الأخطاء المُعادة إلى المشكلات التالية:

-EBUSY

تمت إضافة أقران الناقل بالفعل إلى كائن تتبع أقران الناقل واستُدعيت sd_bus_track_set_recursive() لتغيير وضع التتبع.

-EINVAL

المعامل المحدد غير صالح (NULL في حالة معاملات الإخراج).

-ENOMEM

فشل تخصيص الذاكرة.
