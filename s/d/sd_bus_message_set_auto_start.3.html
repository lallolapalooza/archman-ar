sd_bus_message_set_auto_start(3)

الحزمة: systemd-libs

التراخيص: LGPL-2.1-or-later, CC0-1.0, GPL-2.0-or-later WITH Linux-syscall-note

SD_BUS_MESSAGE_SET_EXPECT_REPLY(3) sd_bus_message_set_expect_reply SD_BUS_MESSAGE_SET_EXPECT_REPLY(3)

الاسم

sd_bus_message_set_expect_reply, sd_bus_message_get_expect_reply, sd_bus_message_set_auto_start, sd_bus_message_get_auto_start, sd_bus_message_set_allow_interactive_authorization, sd_bus_message_get_allow_interactive_authorization - تعيين والاستعلام عن بيانات تعريف رسالة الناقل

المُلَخَّص

#include <systemd/sd-bus.h>

int sd_bus_message_set_expect_reply(sd_bus_message *message, int b);

int sd_bus_message_get_expect_reply(sd_bus_message *message);

int sd_bus_message_set_auto_start(sd_bus_message *message, int b);

int sd_bus_message_get_auto_start(sd_bus_message *message);

int sd_bus_message_set_allow_interactive_authorization(sd_bus_message *message, int b);

int sd_bus_message_get_allow_interactive_authorization(sd_bus_message *message);

الوصف

sd_bus_message_set_expect_reply() تعيين أو إلغاء تعيين العلامة NO_REPLY_EXPECTED على الرسالة m. هذه العلامة مهمة فقط لرسائل استدعاء الطريقة وتستخدم لتحديد أنه لا يُتوقع رد للطريقة أو رد خطأ. يتم تجاهلها للأنواع الأخرى. وبالتالي، بالنسبة لرسالة استدعاء طريقة، فإن استدعاء

sd_bus_message_set_expect_reply(..., 0)

يعين العلامة ويلغي الرد.

sd_bus_message_get_expect_reply() يتحقق مما إذا كانت العلامة NO_REPLY_EXPECTED معيّنة على الرسالة m. ستُرجع قيمة موجبة إذا لم تكن معيّنة، وصفرًا إذا كانت كذلك.

sd_bus_message_set_auto_start() تعيين أو إلغاء تعيين العلامة NO_AUTO_START على الرسالة m. عندما تكون العلامة معيّنة، لا يجب على الناقل تشغيل مالك لاسم الوجهة استجابةً لهذه الرسالة. استدعاء

sd_bus_message_set_auto_start(..., 0)

يعين العلامة.

sd_bus_message_get_auto_start() يتحقق مما إذا كانت العلامة NO_AUTO_START معيّنة على الرسالة m. ستُرجع قيمة موجبة إذا لم تكن معيّنة، وصفرًا إذا كانت كذلك.

sd_bus_message_set_allow_interactive_authorization() تعيين أو إلغاء تعيين العلامة ALLOW_INTERACTIVE_AUTHORIZATION على الرسالة m. إعداد هذه العلامة يُعلم المستلم بأن المتصل مستعد للانتظار للحصول على ترخيص تفاعلي عبر polkit أو إطار عمل مشابه. لاحظ أن إعداد هذه العلامة لا يضمن أن المستلم سيقوم بالفعل بإجراء ترخيص تفاعلي. أيضًا، تأكد من تعيين مهلة رسالة مناسبة عند استخدام هذه العلامة نظرًا لأن الترخيص التفاعلي قد يستغرق وقتًا طويلاً لأنه يعتمد على إدخال المستخدم. إذا كان b غير صفري، فسيتم تعيين العلامة.

sd_bus_message_get_allow_interactive_authorization() يتحقق مما إذا كانت العلامة ALLOW_INTERACTIVE_AUTHORIZATION معيّنة على الرسالة m. ستُرجع عددًا صحيحًا موجبًا إذا كانت العلامة معيّنة. بخلاف ذلك، ستُرجع صفرًا.

قيمة الإرجاع

عند النجاح، تُرجع هذه الدوال عددًا صحيحًا غير سالب. عند الفشل، تُرجع رمز خطأ سالبًا بنمط errno.

الأخطاء

قد تشير الأخطاء المرجعة إلى المشاكل التالية:

-EINVAL

المعامل message هو NULL.

-EPERM

الرسالة message مختومة عند محاولة تعيين علامة.

الرسالة message ذات نوع خاطئ.

ملاحظات

الوظائف الموصوفة هنا متاحة كمكتبة مشتركة، والتي يمكن ترجمتها والربط بها باستخدام ملف pkg-config(1) الخاص بـ libsystemd.

الكود الموصوف هنا يستخدم getenv(3)، والذي تم تعريفه على أنه غير آمن للخيوط المتعددة. هذا يعني أن الكود الذي يستدعي الدوال الموصوفة هنا يجب ألا يستدعي setenv(3) من خيط موازٍ. يوصى بإجراء استدعاءات لـ setenv() فقط من مرحلة مبكرة من البرنامج عندما لم يتم بدء أي خيوط أخرى.
