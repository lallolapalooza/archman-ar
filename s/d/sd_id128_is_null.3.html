الوصف

sd-id128.h جزء من libsystemd(3) ويوفر واجهات برمجة التطبيقات لإنشاء وتحويل ومقارنة معرفات 128 بت. المعرفات 128 بت التي تتم معالجتها وإنشاؤها بواسطة واجهات برمجة التطبيقات هذه هي تعميم لـ OSF UUIDs كما هو معرف بواسطة RFC 4122[1] ولكنها تستخدم تنسيق سلسلة أبسط. لا تفرض هذه الوظائف أي بنية على المعرفات المستخدمة، على عكس OSF UUIDs أو Microsoft GUIDs، ولكنها متوافقة إلى حد كبير مع هذه الأنواع من المعرفات.

يتم تنفيذ المعرف 128 بت كاتحاد أنواع التالي:

typedef union sd_id128 {
  uint8_t bytes[16];
  uint64_t qwords[2];
} sd_id128_t;

يسمح هذا النوع الاتحادي بالوصول إلى المعرف 128 بت كـ 16 بايت منفصلة أو كلمتين 64 بت. بشكل عام، يكون الوصول إلى مكونات المعرف من خلال مصفوفة 8 بت الخاصة به أكثر أمانًا لتجنب مشاكل ترتيب البايتات (endianness). يهدف هذا الاتحاد إلى تمريره بالقيمة (على عكس التمرير بالمرجع) ويمكن معالجته مباشرة بواسطة العملاء.

تم تعريف زوج من الماكروات للدلالة على وفك تشفير معرفات 128 بت:

SD_ID128_MAKE() يستخدم لكتابة معرف ثابت في كود المصدر. الاستخدام الشائع هو تعيين اسم لمعرف باستخدام هذا الماكرو:

#define SD_MESSAGE_COREDUMP SD_ID128_MAKE(fc,2e,22,bc,6e,e6,47,b6,b9,07,29,ab,34,a2,50,b1)

SD_ID128_NULL يحدد معرفًا يتكون فقط من بايتات NUL (أي جميع البتات قبالة).

SD_ID128_ALLF يحدد معرفًا يتكون فقط من بايتات 0xFF (أي جميع البتات قيد التشغيل).

SD_ID128_MAKE_STR() مشابه لـ SD_ID128_MAKE()، ولكنه ينشئ تعبير const char* يمكن استخدامه بشكل مريح في تنسيقات الرسائل وما شابه ذلك:

#include <stdio.h>
#define SD_MESSAGE_COREDUMP_STR SD_ID128_MAKE_STR(fc,2e,22,bc,6e,e6,47,b6,b9,07,29,ab,34,a2,50,b1)
int main(int argc, char **argv) {
  puts("Match for coredumps: MESSAGE_ID=" SD_MESSAGE_COREDUMP_STR);
}

SD_ID128_CONST_STR() يحول المعرفات الثابتة إلى سلاسل نصية ثابتة للإخراج. سيقوم كود المثال التالي بإخراج السلسلة النصية "fc2e22bc6ee647b6b90729ab34a250b1":

int main(int argc, char *argv[]) {
  puts("Match for coredumps: %s", SD_ID128_CONST_STR(SD_MESSAGE_COREDUMP));
}

SD_ID128_FORMAT_STR و SD_ID128_FORMAT_VAL() يستخدمان لتنسيق معرف في سلسلة تنسيق printf(3)، كما هو موضح في المثال التالي:

int main(int argc, char *argv[]) {
  sd_id128_t id;
  id = SD_ID128_MAKE(ee,89,be,71,bd,6e,43,d6,91,e6,c5,5d,eb,03,02,07);
  printf("The ID encoded in this C file is " SD_ID128_FORMAT_STR ".\n", SD_ID128_FORMAT_VAL(id));
  return 0;
}

SD_ID128_UUID_FORMAT_STR و SD_ID128_MAKE_UUID_STR() مشابهان لـ SD_ID128_FORMAT_STR و SD_ID128_MAKE_STR()، ولكنهما يتضمنان الشرطات الفاصلة للامتثال لـ "تمثيل UUID القياسي[2]". يقومان بتنسيق السلسلة بناءً على قواعد Variant 1 لـ RFC4122[1]، أي التحويل من ترتيب البايت الكبير (Big Endian). هذا يتطابق مع سلوك معظم البنية التحتية الأخرى لنظام Linux. من الأفضل تجنب UUIDs من متغيرات أخرى، لتجنب الغموض غير الضروري. جميع المعرفات 128 بت التي تم إنشاؤها بواسطة واجهات برمجة التطبيقات sd-id128 تتوافق تمامًا مع Variant 1 Version 4 UUIDs، وفقًا لـ RFC 4122.

sd_id128_equal() يقارن معرفين 128 بت:

int main(int argc, char *argv[]) {
  sd_id128_t a, b, c;
  a = SD_ID128_MAKE(ee,89,be,71,bd,6e,43,d6,91,e6,c5,5d,eb,03,02,07);
  b = SD_ID128_MAKE(f2,28,88,9c,5f,09,44,15,9d,d7,04,77,58,cb,e7,3e);
  c = a;
  assert(sd_id128_equal(a, c));
  assert(!sd_id128_equal(a, b));
  return 0;
}

sd_id128_string_equal() مشابه لـ sd_id128_equal()، ولكن المعرف الأول يتم تنسيقه كسلسلة const char*. تنطبق نفس القيود كما على الوسيط الأول لـ sd_id128_from_string().

sd_id128_is_null() يتحقق مما إذا كان المعرف يتكون فقط من بايتات NUL:

assert(sd_id128_is_null(SD_ID128_NULL));

وبالمثل، sd_id128_is_allf() يتحقق مما إذا كان المعرف يتكون فقط من بايتات 0xFF (جميع البتات قيد التشغيل):

assert(sd_id128_is_allf(SD_ID128_ALLF));

sd_id128_in_set_sentinel() يأخذ قائمة من المعرفات ويعيد صحيحًا إذا كان الوسيط الأول يساوي أيًا من الوسائط اللاحقة. يتم إنهاء قائمة الوسائط بواسطة علامة SD_ID128_NULL، والتي يجب أن تكون موجودة.

sd_id128_in_set() هي وظيفة مساعدة تأخذ قائمة من المعرفات وتعيد صحيحًا إذا كان الوسيط الأول يساوي أيًا من الوسائط اللاحقة:

int main(int argc, char *argv[]) {
  sd_id128_t a = SD_ID128_MAKE(ee,89,be,71,bd,6e,43,d6,91,e6,c5,5d,eb,03,02,07);
  assert(sd_id128_in_set(a, a));
  assert(sd_id128_in_set(a, a, a));
  assert(!sd_id128_in_set(a));
  assert(!sd_id128_in_set(a,
                          SD_ID128_MAKE(f2,28,88,9c,5f,09,44,15,9d,d7,04,77,58,cb,e7,3e)
                          SD_ID128_MAKE(2f,88,28,5f,9c,44,09,9d,d7,15,77,04,bc,85,7e,e3)
                          SD_ID128_ALLF));
  return 0;
}

sd_id128_in_set() معرف كـ ماكرو فوق sd_id128_in_set_sentinel()، مضيفًا علامة SD_ID128_NULL تلقائيًا. نظرًا لأن sd_id128_in_set_sentinel() يستخدم SD_ID128_NULL كعلامة، لا يمكن وضع SD_ID128_NULL بشكل آخر في قائمة الوسائط.

sd_id128_in_setv() مشابه لـ sd_id128_in_set_sentinel()، ولكنه يأخذ وسيط varargs من نوع struct.

يمكن إنشاء معرفات عشوائية جديدة باستخدام أمر new في systemd-id128(1).

راجع sd_id128_to_string(3)، sd_id128_randomize(3) و sd_id128_get_machine(3) لمعلومات حول أشخاص آخرين

sd_id128_is_null(3)

NAME

int sd_id128_is_allf(sd_id128_t id);
int sd_id128_equal(sd_id128_t id1, sd_id128_t id2);

DESCRIPTION

The sd_id128_t type represents a 128-bit identifier. This type is used to store unique identifiers such as machine IDs, boot IDs, and other UUIDs.

The functions described in this man page allow for the manipulation and comparison of these 128-bit IDs.

sd_id128_from_string converts a string representation of a 128-bit ID (e.g., "5515275a-580a-4a0d-969c-8821069b533a") into the sd_id128_t type. It returns the converted ID on success, or a null ID if the string is invalid.

sd_id128_to_string converts an sd_id128_t ID into its canonical string representation. The buffer pointed to by ret must be large enough to hold the string plus the null terminator (37 bytes).

sd_id128_to_bytes converts an sd_id128_t ID into an array of 16 bytes.

sd_id128_from_bytes converts an array of 16 bytes into an sd_id128_t ID.

sd_id128_is_null checks if an ID is the null ID (all zeros).

sd_id128_is_allf checks if an ID is the all-ones ID.

sd_id128_equal compares two sd_id128_t IDs for equality.

sd_id128_string_equal compares two string representations of IDs for equality.

sd_id128_in_setv checks if an ID is present in a NULL-terminated array of IDs. The n_ids parameter specifies the number of IDs in the ids array.

sd_id128_in_set_sentinel and sd_id128_in_set are varargs functions that check if an ID is present in a list of IDs. The list is terminated by a sentinel value. sd_id128_in_set_sentinel expects the sentinel value to be the last argument, while sd_id128_in_set takes the sentinel value as a separate argument.

sd_id128_get_machine retrieves the system's machine ID. This ID is typically stored in /etc/machine-id.

sd_id128_get_boot retrieves the system's boot ID. This ID changes with each boot.

sd_id128_randomize generates a new random 128-bit ID.

sd_id128_nullify sets an sd_id128_t ID to the null ID (all zeros).

These functions are part of the systemd library and are typically used for managing unique identifiers within the systemd ecosystem, especially for journaling and other services that require unique identification.

The functions return 1 if the condition is met, and 0 otherwise. Error conditions are indicated by negative return values.

The sd_id128_get_machine and sd_id128_get_boot functions use getenv(3) internally, which is not thread-safe. Therefore, these functions should not be called from parallel threads, especially if setenv(3) is used elsewhere.

The format of the string representation of the 128-bit ID follows RFC 4122.

These functions are available as a shared library, which can be compiled against and linked to with the pkg-config(1) file.

The code described here uses getenv(3), which is declared to be not multi-thread-safe. This means that the code calling the functions described here must not call setenv(3) from a parallel thread. It is recommended to only do calls to setenv() from an early phase of the program when no other threads have been started.
