#include <systemd/sd-bus.h>

int sd_bus_set_trusted(sd_bus *bus, int b);

int sd_bus_set_allow_interactive_authorization(sd_bus *bus, int b);

int sd_bus_get_scope(sd_bus *bus, const char **scope);

int sd_bus_get_tid(sd_bus *bus, pid_t *tid);

int sd_bus_get_unique_name(sd_bus *bus, const char **unique);

الوصف

sd_bus_set_description() تعين سلسلة الوصف المستخدمة في التسجيل إلى السلسلة المحددة. يتم نسخ السلسلة داخليًا وتحريرها عند إلغاء تخصيص كائن الناقل. قد تكون الوسيطة description NULL، وفي هذه الحالة يتم إلغاء تعيين الوصف. يجب استدعاء هذه الوظيفة قبل بدء تشغيل الناقل.

sd_bus_get_description() تُرجع سلسلة وصف في description. قد تم تعيين هذه السلسلة مسبقًا باستخدام sd_bus_set_description() أو sd_bus_open_with_description(3) أو ما شابه. إذا لم يتم تعيينها بهذه الطريقة، فسيتم إرجاع سلسلة افتراضية مثل "system" أو "user" للناقلات النظام أو المستخدم، و -ENXIO خلاف ذلك.

sd_bus_set_anonymous() تمكن أو تعطل "المصادقة المجهولة"، أي عدم وجود مصادقة، للطرف المتصل بالناقل. يجب استدعاء هذه الوظيفة قبل بدء تشغيل الناقل. انظر قسم آليات مصادقة D-Bus[1] في مواصفات D-Bus للحصول على التفاصيل.

sd_bus_is_anonymous() تُرجع true إذا كان اتصال الناقل يسمح بالمصادقة المجهولة (بالمعنى الموصوف في الفقرة السابقة).

sd_bus_set_trusted() تعين حالة "موثوق به" على الكائن bus. إذا كانت صحيحة، فسيتم الوثوق بجميع الاتصالات على الناقل وسيتم منح الوصول إلى جميع الطرق المميزة وغير المميزة. يجب استدعاء هذه الوظيفة قبل بدء تشغيل الناقل.

sd_bus_is_trusted() تُرجع true إذا كان اتصال الناقل موثوقًا به (بالمعنى الموصوف في الفقرة السابقة).

sd_bus_set_allow_interactive_authorization() تمكن أو تعطل التفويض التفاعلي لاستدعاءات الطريقة. إذا كانت صحيحة، فسيتم تمييز الرسائل بالعلم ALLOW_INTERACTIVE_AUTHORIZATION المحدد بواسطة مواصفات D-Bus[2]، مما يعلم الجانب المستلم بأن المتصل مستعد للانتظار للحصول على التفويض التفاعلي، والذي قد يستغرق وقتًا طويلاً لإكماله. إذا تم تعيين هذا العلم، فقد يُطلب من المستخدم كلمات المرور أو التأكيد عبر polkit[3] أو إطار عمل مماثل.

sd_bus_get_allow_interactive_authorization() تُرجع true إذا كان التفويض التفاعلي مسموحًا به و false إذا لم يكن كذلك.

sd_bus_get_scope() تخزن نطاق كائن الناقل المحدد في scope. نطاق ناقل النظام هو "system". نطاق ناقل جلسة المستخدم هو "user". إذا لم يكن كائن الناقل المحدد هو ناقل النظام أو ناقل جلسة مستخدم، فإن sd_bus_get_scope() تُرجع خطأ.

sd_bus_get_tid() تخزن معرف مؤشر ترابط النواة للمؤشر المرتبط بكائن الناقل المحدد في tid. إذا كان bus كائن ناقل افتراضي تم الحصول عليه عن طريق استدعاء إحدى وظائف عائلة sd_bus_default(3)، فإنه يخزن معرف مؤشر ترابط للمؤشر الذي تم إنشاء كائن الناقل فيه. خلاف ذلك، إذا كان كائن الناقل متصلاً بحلقة أحداث، فإنه يخزن معرف مؤشر ترابط للمؤشر الذي تم إنشاء كائن حلقة الأحداث فيه. إذا لم يكن bus كائن ناقل افتراضي ولم يكن متصلاً بحلقة أحداث، فإن sd_bus_get_tid() تُرجع خطأ.

sd_bus_get_unique_name() تخزن الاسم الفريد لكائن الناقل على الناقل في unique. انظر مواصفات D-Bus[4] لمزيد من المعلومات حول أسماء الناقلات. لاحظ أن المتصل لا يملك السلسلة المخزنة في unique ولا ينبغي له تحريرها.

قيمة الإرجاع

عند النجاح، تُرجع هذه الدوال عددًا صحيحًا غير سالب. عند الفشل، تُرجع رمز خطأ سالبًا بأسلوب errno.

الأخطاء

قد تشير الأخطاء المُرجعة إلى المشاكل التالية:

-EINVAL

وسيطة غير صالحة.

-ENOPKG

لا يمكن حل الناقل.

-EPERM

تم بدء تشغيل الناقل بالفعل.

-ECHILD

تم إنشاء الناقل في عملية أو مكتبة أو مثيل وحدة مختلف.

-ENOMEM

فشل تخصيص الذاكرة.

-ENODATA

كائن الناقل المُمرر إلى sd_bus_get_scope() لم يكن ناقل نظام أو ناقل جلسة مستخدم.

تمت الإضافة في الإصدار 246.

-ENXIO

كائن الناقل المُمرر إلى sd_bus_get_tid() لم يكن كائن ناقل افتراضي وليس متصلاً بحلقة أحداث.

كائن الناقل المُمرر إلى sd_bus_get_description() لم يكن لديه description.

تمت الإضافة في الإصدار 246.

ملاحظات

الدوال الموصوفة هنا متاحة كمكتبة مشتركة، والتي يمكن تجميعها وربطها مع ملف pkg-config(1) الخاص بـ libsystemd.

الكود الموصوف هنا يستخدم getenv(3)، والتي تم الإعلان عنها بأنها ليست آمنة للاستخدام في مواضيع متعددة. هذا يعني أن الكود الذي يستدعي الدوال الموصوفة هنا يجب ألا يستدعي setenv(3) من مؤشر ترابط موازٍ. يوصى فقط بإجراء استدعاءات لـ setenv() من مرحلة مبكرة من البرنامج عندما لم يتم بدء أي مؤشرات ترابط أخرى.
