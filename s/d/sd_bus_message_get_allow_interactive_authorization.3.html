sd_bus_message_get_allow_interactive_authorization(3)

الحزمة: systemd-libs

التراخيص: LGPL-2.1-or-later, CC0-1.0, GPL-2.0-or-later WITH Linux-syscall-note

SD_BUS_MESSAGE_SET_EXPECT_REPLY(3) sd_bus_message_set_expect_reply SD_BUS_MESSAGE_SET_EXPECT_REPLY(3)

الاسم

sd_bus_message_set_expect_reply, sd_bus_message_get_expect_reply, sd_bus_message_set_auto_start, sd_bus_message_get_auto_start, sd_bus_message_set_allow_interactive_authorization, sd_bus_message_get_allow_interactive_authorization - تعيين والاستعلام عن بيانات وصفية لرسالة الناقل

المُلَخَّص

#include <systemd/sd-bus.h>

int sd_bus_message_set_expect_reply(sd_bus_message *message, int b);

int sd_bus_message_get_expect_reply(sd_bus_message *message);

int sd_bus_message_set_auto_start(sd_bus_message *message, int b);

int sd_bus_message_get_auto_start(sd_bus_message *message);

int sd_bus_message_set_allow_interactive_authorization(sd_bus_message *message, int b);

int sd_bus_message_get_allow_interactive_authorization(sd_bus_message *message);

الوصف

sd_bus_message_set_expect_reply() تعيّن أو تزيل علم NO_REPLY_EXPECTED على الرسالة m. هذا العلم مهم فقط لرسائل استدعاء التابع ولا يتم استخدامه لتحديد عدم توقع رد تابع أو رد خطأ. يتم تجاهله للأنواع الأخرى. وبالتالي، لرسالة استدعاء تابع، فإن استدعاء

sd_bus_message_set_expect_reply(..., 0)

يعيّن العلم ويلغي الرد.

sd_bus_message_get_expect_reply() يتحقق مما إذا كان علم NO_REPLY_EXPECTED معينًا على الرسالة m. سيعيد قيمة موجبة إذا لم يكن معينًا، وصفر إذا كان كذلك.

sd_bus_message_set_auto_start() تعيّن أو تزيل علم NO_AUTO_START على الرسالة m. عند تعيين العلم، لا يجب على الناقل تشغيل مالك للاسم الوجهة استجابة لهذه الرسالة. استدعاء

sd_bus_message_set_auto_start(..., 0)

يعيّن العلم.

sd_bus_message_get_auto_start() يتحقق مما إذا كان علم NO_AUTO_START معينًا على الرسالة m. سيعيد قيمة موجبة إذا لم يكن معينًا، وصفر إذا كان كذلك.

sd_bus_message_set_allow_interactive_authorization() تعيّن أو تزيل علم ALLOW_INTERACTIVE_AUTHORIZATION على الرسالة m. تعيين هذا العلم يبلغ المستلم بأن المتصل مستعد للانتظار للحصول على تفويض تفاعلي عبر polkit أو إطار عمل مشابه. لاحظ أن تعيين هذا العلم لا يضمن أن المستلم سيقوم بالفعل بإجراء تفويض تفاعلي. أيضًا، تأكد من تعيين مهلة رسالة مناسبة عند استخدام هذا العلم حيث أن التفويض التفاعلي يمكن أن يستغرق وقتًا طويلاً لأنه يعتمد على إدخال المستخدم. إذا كان b غير صفري، يتم تعيين العلم.

sd_bus_message_get_allow_interactive_authorization() يتحقق مما إذا كان علم ALLOW_INTERACTIVE_AUTHORIZATION معينًا على الرسالة m. سيعيد عددًا صحيحًا موجبًا إذا تم تعيين العلم. بخلاف ذلك، يعيد صفرًا.

قيمة الإرجاع

عند النجاح، تعيد هذه الدوال عددًا صحيحًا غير سالب. عند الفشل، تعيد رمز خطأ سالبًا بنمط errno.

الأخطاء

قد تشير الأخطاء المُعادة إلى المشكلات التالية:

-EINVAL

المعلمة message هي NULL.

-EPERM

الرسالة message مختومة عند محاولة تعيين علم.

الرسالة message لديها نوع خاطئ.

ملاحظات

الدوال الموصوفة هنا متاحة كمكتبة مشتركة، والتي يمكن ترجمتها وربطها باستخدام ملف libsystemd pkg-config(1).

يستخدم الكود الموصوف هنا getenv(3)، والذي تم تعريفه بأنه ليس آمنًا للاستخدام متعدد المسارات. هذا يعني أن الكود الذي يستدعي الدوال الموصوفة هنا يجب ألا يستدعي setenv(3) من مسار متوازٍ. يوصى بإجراء استدعاءات لـ setenv() فقط من مرحلة مبكرة من البرنامج عندما لم يتم بدء أي مسارات أخرى.
