#include <systemd/sd-login.h>

int sd_login_monitor_flush(sd_login_monitor *m);

int sd_login_monitor_get_fd(sd_login_monitor *m);

int sd_login_monitor_get_events(sd_login_monitor *m);

int sd_login_monitor_get_timeout(sd_login_monitor *m, uint64_t *timeout_usec);

الوصف

sd_login_monitor_new() يمكن استخدامها لمراقبة جلسات تسجيل الدخول، المستخدمين، المقاعد، والأجهزة الافتراضية/الحاويات. من خلال كائن مراقبة، يمكن دمج واصف ملف في حلقة أحداث محددة من قبل التطبيق والتي يتم إيقاظها في كل مرة يقوم فيها مستخدم بتسجيل الدخول، تسجيل الخروج، أو تتم إضافة أو إزالة مقعد، أو يتغير حالة جلسة، مستخدم، مقعد، أو جهاز افتراضي/حاوية. المعامل الأول يأخذ سلسلة نصية يمكن أن تكون "seat" (للحصول فقط على إشعارات حول المقاعد التي تتم إضافتها، إزالتها، أو تغييرها)، "session" (للحصول فقط على إشعارات حول الجلسات التي يتم إنشاؤها أو إزالتها أو تغييرها)، "uid" (للحصول فقط على إشعارات عند تغيير حالة مستخدم فيما يتعلق بتسجيلات الدخول)، أو "machine" (للحصول فقط على إشعارات عند بدء أو إيقاف جهاز افتراضي أو حاوية). إذا كان المطلوب إنشاء إشعارات في جميع هذه الظروف، يمكن تمرير NULL. لاحظ أنه في المستقبل قد يتم تعريف فئات إضافية. المعامل الثاني يعيد كائن مراقبة ويحتاج إلى تحريره باستخدام استدعاء sd_login_monitor_unref() بعد الاستخدام.

sd_login_monitor_unref() يمكن استخدامها لتدمير كائن مراقبة. لاحظ أن هذا سيجعل أي واصف ملف تم إرجاعه بواسطة sd_login_monitor_get_fd() غير صالح.

sd_login_monitor_unrefp() مشابه لـ sd_login_monitor_unref() ولكنه يأخذ مؤشرًا إلى مؤشر لكائن sd_login_monitor. هذا الاستدعاء مفيد بالاقتران مع سمة المتغير Cleanup من GCC و LLVM[1]. لاحظ أن هذه الوظيفة معرفة كوظيفة مضمنة. استخدم تعريفًا مثل التالي، وذلك لتخصيص كائن مراقبة تسجيل دخول يتم تحريره تلقائيًا عند مغادرة كتلة التعليمات البرمجية:

{
  __attribute__((cleanup(sd_login_monitor_unrefp))) sd_login_monitor *m = NULL;
  int r;
  ...
  r = sd_login_monitor_new(NULL, &m);
  if (r < 0) {
    errno = -r;
    fprintf(stderr, "فشل تخصيص كائن مراقبة تسجيل الدخول: %m\n");
  }
  ...
}

sd_login_monitor_flush() يمكن استخدامها لإعادة تعيين حالة الاستيقاظ لكائن المراقبة. كلما تسببت حدث في استيقاظ المراقبة لحلقة الأحداث عبر واصف الملف، يجب استدعاء هذه الوظيفة لإعادة تعيين حالة الاستيقاظ. إذا لم يتم استدعاء هذا الاستدعاء، فسوف يقوم واصف الملف بإيقاظ حلقة الأحداث مرة أخرى فورًا.

sd_login_monitor_unref() و sd_login_monitor_unrefp() لا تنفذان أي عملية إذا كان كائن المراقبة الممرر هو NULL.

sd_login_monitor_get_fd() يمكن استخدامها لاسترداد واصف الملف لكائن المراقبة الذي يمكن دمجه في حلقة أحداث محددة من قبل التطبيق، المبنية حول poll(2) أو واجهة مشابهة. يجب على التطبيق تضمين واصف الملف المُعاد كمصدر للاستيقاظ لقناع الأحداث الذي تم إرجاعه بواسطة sd_login_monitor_get_events(). يجب عليه تمرير قيمة مهلة كما تم إرجاعها بواسطة sd_login_monitor_get_timeout(). كلما تم تشغيل استيقاظ، يجب إعادة تعيين واصف الملف عبر sd_login_monitor_flush(). يحتاج التطبيق إلى إعادة قراءة حالة تسجيل الدخول باستخدام وظيفة مثل sd_get_seats(3) أو ما شابهها لتحديد ما تغير.

sd_login_monitor_get_events() ستعيد قناع poll() للانتظار. هذه الوظيفة ستعيد مزيجًا من POLLIN، POLLOUT وما شابه ذلك لملء حقل ".events" في struct pollfd.

sd_login_monitor_get_timeout() ستعيد قيمة مهلة للاستخدام في poll(). هذا يعيد قيمة بالميكروثانية منذ عصر CLOCK_MONOTONIC لوقت انتهاء poll() في timeout_usec. انظر clock_gettime(2) للحصول على تفاصيل حول CLOCK_MONOTONIC. إذا لم تكن هناك مهلة للانتظار، فسيتم ملء (uint64_t) -1 بدلاً من ذلك. لاحظ أن poll() يأخذ مهلة نسبية بالمللي ثانية وليس مهلة مطلقة بالميكروثانية. لتحويل مهلة 'ميكروثانية' المطلقة إلى 'مللي ثانية' نسبية، استخدم كودًا مثل التالي:

uint64_t t;
int msec;
sd_login_monitor_get_timeout(m, &t);
if (t == (uint64_t) -1)
  msec = -1;
else {
  struct timespec ts;
  uint64_t n;
  clock_gettime(CLOCK_MONOTONIC, &ts);
  n = (uint64_t) ts.tv_sec * 1000000 + ts.tv_nsec / 1000;
  msec = t > n ? (int) ((t - n + 999) / 1000) : 0;
}

الكود أعلاه لا يقوم بأي فحص للأخطاء للاختصار. يمكن تمرير المتغير msec المحسوب مباشرة كمعامل مهلة لـ poll().

قيمة الإرجاع

عند النجاح، تعيد sd_login_monitor_new()، sd_login_monitor_flush()، و sd_login_monitor_get_timeout() 0 أو عددًا صحيحًا موجبًا. عند النجاح، تعيد sd_login_monitor_get_fd() واصف ملف يونكس. عند النجاح، تعيد sd_login_monitor_get_events() مزيجًا من POLLIN، POLLOUT وما شابه. عند الفشل، تعيد هذه الاستدعاءات رمز خطأ سالب من نوع errno.

sd_login_monitor_unref() تعيد دائمًا NULL.

الأخطاء

قد تشير الأخطاء المُعادة إلى المشاكل التالية:

-EINVAL

كان معامل الإدخال غير صالح (خارج النطاق، أو NULL، حيث أن ذلك غير مقبول). الفئة المحددة للمراقبة غير معروفة.

-ENOMEM

فشل تخصيص الذاكرة.

sd_login_monitor_get_fd

الاسم

sd_login_monitor_get_fd - الحصول على معرف ملف واصف لإدخال/إخراج مراقب تسجيل الدخول

الملخص

#include <systemd/sd-login.h>

int sd_login_monitor_get_fd(sd_login_monitor *monitor);

الوصف

يقوم الاستدعاء sd_login_monitor_get_fd() بإرجاع معرف ملف واصف (file descriptor) يمكن استخدامه للمراقبة على الأحداث المتعلقة بتسجيل الدخول من كائن sd_login_monitor. هذا المعرف مخصص فقط للقراءة.

إذا كان monitor هو NULL، يتم إرجاع -EBADFD.

يجب على المتصل عدم إغلاق معرف الملف الواصف الذي تم إرجاعه. سيتم إغلاقه تلقائيًا عند تدمير الكائن sd_login_monitor.

قيمة الإرجاع

عند النجاح، يتم إرجاع معرف ملف واصف غير سالب. عند الفشل، يتم إرجاع خطأ سالب.

يمكن إرجاع الأخطاء التالية:

  • -EBADFD: monitor هو NULL.
  • -EAGAIN: لم يتم بعد تعيين معرف الملف الواصف (غير محتمل).

ملاحظات

يجب أن يتم تمرير معرف الملف الواصف الذي تم الحصول عليه من هذه الدالة إلى poll(2)، epoll_wait(2)، sd_event_add_poll(3)، أو آلية مماثلة. عندما يصبح معرف الملف الواصف قابلاً للقراءة، فهذا يعني أن حدث تسجيل دخول قد وقع.

للحصول على تفاصيل حول الأحداث التي وقعت، يجب استدعاء sd_login_monitor_get_next(3).
