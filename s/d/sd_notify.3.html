sd_notify(3)

الحزمة: systemd-libs

التراخيص: LGPL-2.1-or-later, CC0-1.0, GPL-2.0-or-later WITH Linux-syscall-note

/section>

البيئة

$NOTIFY_SOCKET

يتم تعيينها بواسطة مدير الخدمة للعمليات التي تتم مراقبتها لإشعار الحالة واكتمال البدء. تحدد متغير البيئة هذا المقبس الذي تتواصل معه sd_notify(). انظر أعلاه للتفاصيل.

أمثلة

مثال 1. إشعار بدء التشغيل

عندما تنتهي الخدمة من بدء التشغيل، قد تصدر الاستدعاء التالي لإعلام مدير الخدمة:

sd_notify(0, "READY=1");

مثال 2. إشعار بدء تشغيل موسع

يمكن للخدمة إرسال ما يلي بعد إكمال التهيئة:

sd_notifyf(0, "READY=1\n"
              "STATUS=Processing requests...\n"
              "MAINPID=%lu",
           (unsigned long) getpid());

مثال 3. إشعار سبب الخطأ

يمكن للخدمة إرسال ما يلي قبل الخروج مباشرة، عند الفشل:

sd_notifyf(0, "STATUS=Failed to start up: %s\n"
              "ERRNO=%i",
           strerror_r(errnum, (char[1024]){}, 1024),
           errnum);

مثال 4. تخزين واصف ملف في مدير الخدمة

لتخزين واصف ملف مفتوح في مدير الخدمة، لمواصلة التشغيل بعد إعادة تشغيل الخدمة دون فقدان الحالة، استخدم "FDSTORE=1":

sd_pid_notify_with_fds(0, 0, "FDSTORE=1\nFDNAME=foobar", &fd, 1);

مثال 5. التخلص من حالات السباق

عندما لا يتم إنشاء العميل الذي يرسل الإشعارات بواسطة مدير الخدمة، فقد يخرج بسرعة كبيرة وقد يفشل مدير الخدمة في تعيينها بشكل صحيح للوحدة. لمنع مثل هذه السباقات، استخدم sd_notify_barrier() للمزامنة مع استقبال جميع الإشعارات المرسلة قبل إجراء هذا الاستدعاء.

sd_notify(0, "READY=1");
/* set timeout to 5 seconds */
sd_notify_barrier(0, 5 * 1000000);
