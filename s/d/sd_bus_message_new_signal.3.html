sd_bus_message_new_signal(3)

الحزمة: systemd-libs

التراخيص: LGPL-2.1-or-later, CC0-1.0, GPL-2.0-or-later WITH Linux-syscall-note

SD_BUS_MESSAGE_NEW_SIGNAL(3) sd_bus_message_new_signal SD_BUS_MESSAGE_NEW_SIGNAL(3)

الاسم

sd_bus_message_new_signal, sd_bus_message_new_signal_to - إنشاء رسالة إشارة

مُلَخَّص

#include <systemd/sd-bus.h>

int sd_bus_message_new_signal(sd_bus *bus, sd_bus_message **m, const char *path, const char *interface, const char *member);

int sd_bus_message_new_signal_to(sd_bus *bus, sd_bus_message **m, const char *destination, const char *path, const char *interface, const char *member);

الوصف

تقوم الدالة sd_bus_message_new_signal() بإنشاء كائن رسالة حافلة جديد يغلف إشارة D-Bus، وتعيده في المعلمة الإخراجية m. سيتم إرسال الإشارة إلى المسار path، على الواجهة interface، والعضو member. عند إرسال هذه الرسالة، لا يُتوقع رد. راجع sd_bus_message_new_method_call(3) لوصف موجز لمعنى المعلمات path و interface و member.

sd_bus_message_new_signal_to() هي اختصار لإنشاء رسالة حافلة جديدة إلى وجهة محددة. سلوكها مشابه لاستدعاء sd_bus_message_new_signal() متبوعًا باستدعاء sd_bus_message_set_destination(3).

قيمة الإرجاع

تعيد هذه الدالة 0 إذا تم إنشاء كائن الرسالة بنجاح، ورمز خطأ سلبي بأسلوب errno بخلاف ذلك.

أخطاء

قد تشير الأخطاء المُعادة إلى المشاكل التالية:

-EINVAL

المعلمة الإخراجية m هي NULL.

المعلمة path ليست مسار D-Bus صالحًا ("/an/object/path")، المعلمة interface ليست اسم واجهة D-Bus صالحًا ("an.interface.name")، أو المعلمة member ليست عضو D-Bus صالحًا ("Name").

-ENOTCONN

المعلمة bus هي NULL أو الحافلة غير متصلة.

-ENOMEM

فشل تخصيص الذاكرة.

ملاحظات

الدوال الموصوفة هنا متاحة كمكتبة مشتركة، والتي يمكن تجميعها وربطها مع ملف pkg-config(1) الخاص بـ libsystemd.

الكود الموصوف هنا يستخدم getenv(3)، والذي تم تعريفه بأنه غير آمن للاستخدام المتزامن بين الخيوط. هذا يعني أن الكود الذي يستدعي الدوال الموصوفة هنا يجب ألا يستدعي setenv(3) من خيط متوازٍ. يوصى بالقيام باستدعاءات setenv() فقط في مرحلة مبكرة من البرنامج عندما لم يتم بدء أي خيوط أخرى.

أمثلة

مثال 1. إرسال إشارة بسيطة

/* SPDX-License-Identifier: MIT-0 */
#include <systemd/sd-bus.h>
#define _cleanup_(f) __attribute__((cleanup(f)))
int send_unit_files_changed(sd_bus *bus) {
  _cleanup_(sd_bus_message_unrefp) sd_bus_message *message = NULL;
  int r;
  r = sd_bus_message_new_signal(bus, &message,
                                "/org/freedesktop/systemd1",
                                "org.freedesktop.systemd1.Manager",
                                "UnitFilesChanged");
  if (r < 0)
    return r;
  return sd_bus_send(bus, message, NULL);
}

هذه الدالة في مصادر systemd تُستخدم لإصدار إشارة "UnitFilesChanged" عند تغيير ملفات الوحدة.
