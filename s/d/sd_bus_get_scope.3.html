#include <systemd/sd-bus.h>

int sd_bus_set_description(sd_bus *bus, const char *description);

int sd_bus_get_description(sd_bus *bus, const char **description);

int sd_bus_set_anonymous(sd_bus *bus, int b);

int sd_bus_set_trusted(sd_bus *bus, int b);

int sd_bus_is_trusted(sd_bus *bus);

int sd_bus_set_allow_interactive_authorization(sd_bus *bus, int b);

int sd_bus_get_scope(sd_bus *bus, const char **scope);

int sd_bus_get_tid(sd_bus *bus, pid_t *tid);

int sd_bus_get_unique_name(sd_bus *bus, const char **unique);

الوصف

sd_bus_set_description() تعيين سلسلة الوصف التي تستخدم في التسجيل إلى السلسلة المحددة. يتم نسخ السلسلة داخليًا وتحريرها عند إلغاء تخصيص كائن الناقل. قد تكون الوسيطة description NULL، وفي هذه الحالة يتم إلغاء تعيين الوصف. يجب استدعاء هذه الدالة قبل بدء تشغيل الناقل.

sd_bus_get_description() تُرجع سلسلة وصف في description. قد تكون هذه السلسلة قد تم تعيينها مسبقًا باستخدام sd_bus_set_description() أو sd_bus_open_with_description(3) أو ما شابه. إذا لم يتم تعيينها بهذه الطريقة، سيتم إرجاع سلسلة افتراضية مثل "system" أو "user" للناقلات النظامية أو للمستخدم، و -ENXIO بخلاف ذلك.

sd_bus_set_anonymous() تمكين أو تعطيل "المصادقة المجهولة"، أي عدم وجود مصادقة، لنظير الناقل. يجب استدعاء هذه الدالة قبل بدء تشغيل الناقل. راجع قسم آليات مصادقة D-Bus[1] من مواصفات D-Bus للتفاصيل.

sd_bus_is_anonymous() تُرجع صحيحة إذا كان اتصال الناقل يسمح بالمصادقة المجهولة (بالمعنى الموضح في الفقرة السابقة).

sd_bus_set_trusted() تعيين حالة "موثوق به" على الكائن bus. إذا كانت صحيحة، فسيتم الوثوق بجميع الاتصالات على الناقل وسيتم منح الوصول إلى جميع الطرق المميزة وغير المميزة. يجب استدعاء هذه الدالة قبل بدء تشغيل الناقل.

sd_bus_is_trusted() تُرجع صحيحة إذا كان اتصال الناقل موثوقًا به (بالمعنى الموضح في الفقرة السابقة).

sd_bus_set_allow_interactive_authorization() تمكين أو تعطيل التفويض التفاعلي لاستدعاءات الأساليب. إذا كانت صحيحة، فسيتم تمييز الرسائل بالعلامة ALLOW_INTERACTIVE_AUTHORIZATION المحددة بواسطة مواصفات D-Bus[2]، مما يشير إلى الجانب المستقبل بأن المتصل مستعد للانتظار للحصول على التفويض التفاعلي، والذي قد يستغرق وقتًا طويلاً لإكماله. إذا تم تعيين هذه العلامة، فقد يُطلب من المستخدم كلمات المرور أو التأكيد عبر polkit[3] أو إطار عمل مماثل.

sd_bus_get_allow_interactive_authorization() تُرجع صحيحة إذا كان التفويض التفاعلي مسموحًا به وخطأ إذا لم يكن كذلك.

sd_bus_get_scope() يخزن نطاق كائن الناقل المحدد في scope. نطاق ناقل النظام هو "system". نطاق ناقل جلسة المستخدم هو "user". إذا لم يكن كائن الناقل المحدد هو ناقل النظام أو ناقل جلسة مستخدم، فستُرجع sd_bus_get_scope() خطأ.

sd_bus_get_tid() يخزن معرف مؤشر ترابط النواة للمؤشر المرتبط بكائن الناقل المحدد في tid. إذا كان bus كائن ناقل افتراضي تم الحصول عليه عن طريق استدعاء إحدى دوال عائلة sd_bus_default(3)، فإنه يخزن معرف المؤشر الخاص بالمؤشر الذي تم إنشاء كائن الناقل فيه. بخلاف ذلك، إذا تم ربط كائن الناقل بحلقة الأحداث، فإنه يخزن معرف المؤشر الخاص بالمؤشر الذي تم إنشاء كائن حلقة الأحداث فيه. إذا لم يكن bus كائن ناقل افتراضي ولم يكن مرتبطًا بحلقة أحداث، فستُرجع sd_bus_get_tid() خطأ.

sd_bus_get_unique_name() يخزن الاسم الفريد لكائن الناقل على الناقل في unique. انظر مواصفات D-Bus[4] لمزيد من المعلومات حول أسماء النواقل. لاحظ أن المتصل لا يمتلك السلسلة المخزنة في unique ولا ينبغي تحريرها.

قيمة الإرجاع

عند النجاح، تُرجع هذه الدوال عددًا صحيحًا غير سالب. عند الفشل، تُرجع رمز خطأ سالبًا بنمط errno.

الأخطاء

قد تشير الأخطاء المرجعة إلى المشاكل التالية:

-EINVAL

وسيطة غير صالحة.

-ENOPKG

لا يمكن حل الناقل.

-EPERM

تم بدء تشغيل الناقل بالفعل.

-ECHILD

تم إنشاء الناقل في عملية أو مكتبة أو وحدة أخرى.

-ENOMEM

فشل تخصيص الذاكرة.

-ENODATA

لم يكن كائن الناقل الممرر إلى sd_bus_get_scope() ناقل نظام أو ناقل جلسة مستخدم.

تمت الإضافة في الإصدار 246.

-ENXIO

لم يكن كائن الناقل الممرر إلى sd_bus_get_tid() كائن ناقل افتراضي ولم يكن مرتبطًا بحلقة أحداث.

لم يكن لكائن الناقل الممرر إلى sd_bus_get_description() description.

تمت الإضافة في الإصدار 246.

ملاحظات

الوظائف الموصوفة هنا متاحة كمكتبة مشتركة، والتي يمكن تجميعها وربطها باستخدام ملف pkg-config(1) الخاص بـ libsystemd.

يستخدم الكود الموصوف هنا getenv(3)، والذي تم الإعلان عنه بأنه غير آمن للخيوط المتوازية. هذا يعني أن الكود الذي يستدعي الدوال الموصوفة هنا يجب ألا يستدعي setenv(3) من خيط موازٍ. يوصى بإجراء استدعاءات لـ setenv() فقط من مرحلة مبكرة من البرنامج عندما لم يتم بدء أي خيوط أخرى.

sd_bus_get_scope(3) - صفحة الدليل

NAME

sd_bus_get_scope - استرجاع نطاق اتصال D-Bus

SYNOPSIS

sd_bus_get_scope(3)


sd_bus_get_scope(3) – استرجاع نطاق اتصال D-Bus


sd_bus_get_scope(3)


sd_bus_get_scope(3) – استرجاع نطاق اتصال D-Bus

DESCRIPTION

يسترجع وظيفة sd_bus_get_scope() نطاق عنوان bus الذي تم تعيينه لاتصال bus المحدد b. يعتمد النطاق على السياق الذي تم فيه إنشاء الاتصال.

إذا تم إنشاء الاتصال ليكون متاحًا فقط محليًا (على سبيل المثال، عبر sd_bus_open_user() أو sd_bus_open_system())، فإن النطاق سيكون SD_BUS_SCOPE_USER أو SD_BUS_SCOPE_SYSTEM على التوالي. إذا تم إنشاء الاتصال للتواصل عبر شبكة (على سبيل المثال، عبر sd_bus_open_user_with_creds() أو sd_bus_open_system_with_creds())، فإن النطاق سيكون SD_BUS_SCOPE_SYSTEM. بشكل عام، يتم تعيين النطاق لتحديد ما إذا كان اتصال bus يعتبر اتصالًا عامًا (خاصة بالشبكة) أم اتصالًا خاصًا (فقط محليًا).

يتم تمرير مؤشر إلى قيمة النطاق إلى المتغير الذي تشير إليه ret. يجب أن يتم تهيئة ret بـ NULL.

RETURN VALUE

عند النجاح، يتم إرجاع 0. عند الفشل، يتم إرجاع خطأ سالب.

NOTES

يجب ملاحظة أن قيمة النطاق المرجعة قد تتغير ديناميكيًا بناءً على سياسات polkit.

HISTORY

تمت إضافة sd_bus_get_scope() في الإصدار 2.46.
