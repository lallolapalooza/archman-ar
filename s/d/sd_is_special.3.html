sd_is_special(3)

الحزمة: systemd-libs

التراخيص: LGPL-2.1-or-later, CC0-1.0, GPL-2.0-or-later WITH Linux-syscall-note

SD_IS_FIFO(3) sd_is_fifo SD_IS_FIFO(3)

الاسم

sd_is_fifo, sd_is_socket, sd_is_socket_inet, sd_is_socket_unix, sd_is_socket_sockaddr, sd_is_mq, sd_is_special - التحقق من نوع واصف الملف

الملخص

#include <systemd/sd-daemon.h>

int sd_is_fifo(int fd, const char *path);

int sd_is_socket(int fd, int family, int type, int listening);

int sd_is_socket_inet(int fd, int family, int type, int listening, uint16_t port);

int sd_is_socket_sockaddr(int fd, int type, const struct sockaddr *addr, unsigned addr_len, int listening);

int sd_is_socket_unix(int fd, int type, int listening, const char *path, size_t length);

int sd_is_mq(int fd, const char *path);

int sd_is_special(int fd, const char *path);

الوصف

يمكن استدعاء sd_is_fifo() للتحقق مما إذا كان واصف الملف المحدد يشير إلى FIFO أو أنبوب (pipe). إذا لم يكن المعامل path هو NULL، يتم التحقق مما إذا كان FIFO مرتبطًا بمسار نظام الملفات المحدد.

يمكن استدعاء sd_is_socket() للتحقق مما إذا كان واصف الملف المحدد يشير إلى مقبس (socket). إذا لم يكن المعامل family هو AF_UNSPEC، يتم التحقق مما إذا كان المقبس من العائلة المحددة (AF_UNIX، AF_INET، ...). إذا لم يكن المعامل type هو 0، يتم التحقق مما إذا كان المقبس من النوع المحدد (SOCK_STREAM، SOCK_DGRAM، ...). إذا كان المعامل listening موجبًا، يتم التحقق مما إذا كان المقبس في وضع الاستقبال، أي تم استدعاء listen() له. إذا كان listening هو 0، يتم التحقق مما إذا كان المقبس ليس في هذا الوضع. إذا كان المعامل سالبًا، لا يتم إجراء هذا التحقق. يجب استخدام المعامل listening فقط لمقابس التدفق ويجب تعيينه إلى قيمة سالبة بخلاف ذلك.

sd_is_socket_inet() مشابه لـ sd_is_socket()، ولكنه يتحقق اختياريًا من رقم منفذ IPv4 أو IPv6 الذي يرتبط به المقبس، إلا إذا كان port صفرًا. بالنسبة لهذا الاستدعاء، يجب تمرير family كـ AF_UNSPEC، أو AF_INET، أو AF_INET6.

sd_is_socket_sockaddr() مشابه لـ sd_is_socket_inet()، ولكنه يتحقق مما إذا كان المقبس مرتبطًا بالعنوان المحدد بواسطة addr. يجب أن تكون family المحددة بواسطة addr إما AF_INET أو AF_INET6 ويجب أن يكون addr_len كبيرًا بما يكفي لتلك العائلة. إذا حدد addr منفذًا غير صفري، يتم أيضًا التحقق مما إذا كان المقبس مرتبطًا بهذا المنفذ. بالإضافة إلى ذلك، بالنسبة لـ IPv6، إذا حدد addr قيمًا غير صفرية لـ sin6_flowinfo أو sin6_scope_id، يتم التحقق مما إذا كان للمقبس نفس القيم.

sd_is_socket_unix() مشابه لـ sd_is_socket() ولكنه يتحقق اختياريًا من مسار AF_UNIX الذي يرتبط به المقبس، إلا إذا كان المعامل path هو NULL. بالنسبة لمقابس نظام الملفات العادية AF_UNIX، قم بتعيين المعامل length إلى 0. بالنسبة لمقابس مساحة الأسماء المجردة في Linux، قم بتعيين length إلى حجم العنوان، بما في ذلك البايت الصفري الأولي، وقم بتعيين path إلى البايت الصفري الأولي لعنوان المقبس.

يمكن استدعاء sd_is_mq() للتحقق مما إذا كان واصف الملف المحدد يشير إلى طابور رسائل POSIX. إذا لم يكن المعامل path هو NULL، يتم التحقق مما إذا كان طابور الرسائل مرتبطًا بالاسم المحدد.

يمكن استدعاء sd_is_special() للتحقق مما إذا كان واصف الملف المحدد يشير إلى ملف خاص. إذا لم يكن المعامل path هو NULL، يتم التحقق مما إذا كان واصف الملف مرتبطًا باسم الملف المحدد. الملفات الخاصة في هذا السياق هي عقد الجهاز الحرفي والملفات في /proc/ أو /sys/.

قيمة الإرجاع

عند الفشل، تعيد هذه الاستدعاءات رمز خطأ سالبًا بأسلوب errno. إذا كان واصف الملف من النوع المحدد ومرتبطًا بالعنوان المحدد، يتم إرجاع قيمة موجبة، وإلا صفر.

ملاحظات

الوظائف الموصوفة هنا متاحة كمكتبة مشتركة، والتي يمكن تجميعها وربطها باستخدام ملف libsystemd pkg-config(1).

داخليًا، تستخدم هذه الوظائف مزيجًا من fstat(2) و getsockname(2) للتحقق من نوع واصف الملف ومكان ارتباطه.
