sd_journal_sendv(3)

الحزمة: systemd-libs

التراخيص: LGPL-2.1-or-later, CC0-1.0, GPL-2.0-or-later WITH Linux-syscall-note

SD_JOURNAL_PRINT(3) sd_journal_print SD_JOURNAL_PRINT(3)

الاسم

sd_journal_print, sd_journal_printv, sd_journal_send, sd_journal_sendv, sd_journal_perror, SD_JOURNAL_SUPPRESS_LOCATION, sd_journal_print_with_location, sd_journal_printv_with_location, sd_journal_send_with_location, sd_journal_sendv_with_location, sd_journal_perror_with_location - إرسال إدخالات السجل إلى دفتر اليومية

الملخص

#include <systemd/sd-journal.h>

int sd_journal_print(int priority, const char *format, ...);

int sd_journal_printv(int priority, const char *format, va_list ap);

int sd_journal_send(const char *format, ...);

int sd_journal_sendv(const struct iovec *iov, int n);

int sd_journal_perror(const char *message);

int sd_journal_print_with_location(int priority, const char *file, const char *line, const char *func, const char *format, ...);

int sd_journal_printv_with_location(int priority, const char *file, const char *line, const char *func, const char *format, va_list ap);

int sd_journal_send_with_location(const char *file, const char *line, const char *func, const char *format, ...);

int sd_journal_sendv_with_location(const char *file, const char *line, const char *func, const struct iovec *iov, int n);

int sd_journal_perror_with_location(const char *file, const char *line, const char *func, const char *message);

الوصف

يمكن استخدام sd_journal_print() لإرسال إدخالات سجل نصية بسيطة إلى سجل النظام. الوسيط الأول هو قيمة الأولوية. يتبع ذلك سلسلة تنسيق ومعاملاتها، على غرار printf(3) أو syslog(3). لاحظ أن الرسالة الناتجة سيتم اقتطاعها حاليًا إلى LINE_MAX - 8. قيمة الأولوية هي واحدة من LOG_EMERG، LOG_ALERT، LOG_CRIT، LOG_ERR، LOG_WARNING، LOG_NOTICE، LOG_INFO، LOG_DEBUG، كما هو معرّف في syslog.h، راجع syslog(3) للتفاصيل. يُوصى باستخدام هذا الاستدعاء لإرسال رسائل السجل بلغة التطبيق أو لغة النظام وبصيغة UTF-8، ولكن لا تُفرض مثل هذه القيود. لاحظ أن رسائل السجل المكتوبة باستخدام هذه الدالة لا يُتوقع عمومًا أن تنتهي بحرف سطر جديد. ومع ذلك، نظرًا لأن جميع المسافات البيضاء اللاحقة (بما في ذلك المسافات والأسطر الجديدة وعلامات الجدولة وحروف الإرجاع) تُزال تلقائيًا من السلسلة المسجلة، فمن المقبول تحديد واحدة (أو أكثر). تُحجب الأسطر الفارغة (بعد إزالة المسافات البيضاء اللاحقة). في الأسطر غير الفارغة، تُترك المسافات البيضاء البادئة (وكذلك المسافات البيضاء الداخلية) دون تعديل.

sd_journal_printv() مشابهة لـ sd_journal_print() لكنها تأخذ قائمة وسائط متغيرة مغلفة في كائن من النوع va_list (راجع stdarg(3) لمزيد من المعلومات) بدلاً من سلسلة التنسيق. وهي متكافئة في السلوك بخلاف ذلك.

يمكن استخدام sd_journal_send() لإرسال إدخالات سجل منظمة إلى سجل النظام. تأخذ سلسلة من سلاسل التنسيق، كل منها يتبعها مباشرة معاملاتها المرتبطة، وتنتهي بـ NULL. يجب أن تكون السلاسل الممررة بصيغة "VARIABLE=value". يجب أن يكون اسم المتغير بأحرف كبيرة وأن يتكون فقط من الأحرف والأرقام والشرطات السفلية، ويجب ألا يبدأ بشرطة سفلية. (سيتم تجاهل جميع التعيينات التي لا تتبع هذا التركيب.) يمكن أن تكون القيمة بأي حجم وصيغة. يُوصى بشدة بإرسال سلاسل نصية منسقة بترميز أحرف UTF-8 فقط، وإرسال الحقول الثنائية فقط عندما لا يكون التنسيق في سلاسل UTF-8 منطقيًا. تم تعريف عدد من الحقول المعروفة، راجع systemd.journal-fields(7) للتفاصيل، ولكن يمكن استخدام حقول إضافية محددة من التطبيق. يمكن تعيين أكثر من قيمة لمتغير واحد لكل إدخال. إذا تم استخدام هذه الدالة، تُزال المسافات البيضاء اللاحقة تلقائيًا من كل حقل منسق.

sd_journal_sendv() مشابهة لـ sd_journal_send() لكنها تأخذ مصفوفة من struct iovec (كما هو معرّف في uio.h، راجع readv(3) للتفاصيل) بدلاً من سلسلة التنسيق. يجب أن يشير كل هيكل إلى حقل واحد من الإدخال المراد إرساله. يحدد الوسيط الثاني عدد الهياكل في المصفوفة. sd_journal_sendv() مفيدة بشكل خاص لإرسال كائنات ثنائية إلى دفتر اليومية حيث يكون ذلك ضروريًا. لاحظ أن هذه الدالة لن تزيل المسافات البيضاء اللاحقة من الحقول الممررة، بل تمرر البيانات المحددة دون تعديل. وهذا يختلف عن كل من sd_journal_print() و sd_journal_send() الموصوفتين أعلاه، اللتين تعتمدان على سلاسل التنسيق، وتزيلان المسافات البيضاء اللاحقة.

sd_journal_perror() مشابهة لـ perror(3) و تكتب رسالة إلى دفتر اليومية تتكون من السلسلة الممررة، متبوعة بـ ": " وتمثيل مقروء للبشر لرمز الخطأ الحالي المخزن في errno(3). إذا تم تمرير سلسلة الرسالة كـ NULL أو سلسلة فارغة، فسيتم كتابة تمثيل سلسلة الخطأ فقط، بدون بادئة. يتم تضمين حقل دفتر يومية إضافي ERRNO= في الإدخال يحتوي على رمز الخطأ الرقمي منسقًا كسلسلة عشرية. أولوية السجل المستخدمة هي LOG_ERR (3).

لاحظ أن sd_journal_send() هي غلاف حول sd_journal_sendv() لتسهيل استخدامها عندما لا يُراد إرسال سوى سلاسل نصية. أيضًا، الاستدعاءان التاليان متكافئان تقريبًا:

sd_journal_print(LOG_INFO, "Hello World, this is PID %lu!", (unsigned long) getpid());
sd_journal_send("MESSAGE=Hello World, this is PID %lu!", (unsigned long) getpid(),
                "PRIORITY=%i", LOG_INFO,
                NULL);

لاحظ أن هذه الاستدعاءات تضيف ضمنيًا حقولاً لملف المصدر واسم الدالة وسطر الكود حيث تم استدعاؤها. يتم تنفيذ ذلك باستخدام وحدات الماكرو. إذا لم يكن ذلك مرغوبًا، يمكن إيقافه بتعريف SD_JOURNAL_SUPPRESS_LOCATION قبل تضمين sd-journal.h.

sd_journal_print_with_location()، sd_journal_printv_with_location()، sd_journal_send_with_location()، sd_journal_sendv_with_location()، و sd_journal_perror_with_location() مشابهة لنظيراتها بدون "_with_location"، لكنها تقبل معاملات إضافية لتعيين اسم ملف المصدر والدالة والسطر بشكل صريح. يجب أن تحتوي الوسائط "file" و "line" على إدخالات دفتر يومية صالحة بما في ذلك اسم المتغير، مثل "CODE_FILE=src/foo.c" و "CODE_LINE=666"، بينما يجب أن تحتوي "func" على اسم الدالة فقط، أي القيمة بدون "CODE_FUNC=". هذه المتغيرات مفيدة بشكل أساسي عند كتابة أغلفة مخصصة، على سبيل المثال في الارتباطات للغة مختلفة.

يمكن استخدام syslog(3) و sd_journal_print() بشكل متبادل إلى حد كبير من حيث الوظائف. ومع ذلك، لاحظ أن رسائل السجل المسجلة عبر الأولى تسلك مسارًا مختلفًا إلى خادم دفتر اليومية عن الثانية، وبالتالي لا يمكن ضمان الترتيب الزمني العالمي بين التدفقين. استخدام sd_journal_print() له ميزة تسجيل سطر الكود المصدري وأسماء الملفات والدوال كبيانات وصفية مع جميع الإدخالات، وضمان الترتيب الزمني مع إدخالات السجل المنظمة التي يتم إنشاؤها عبر sd_journal_send(). استخدام syslog() له ميزة كونه أكثر قابلية للنقل.

تنفذ هذه الدوال عميلاً لـ بروتوكول دفتر اليومية الأصلي[1].

القيمة المُرجعة

تُرجع الدوال العشر 0 عند النجاح أو رمز خطأ بأسلوب errno سالبًا. متغير errno(3) الخاص بهإلف لم يتم تغييره.

إذا لم يكن systemd-journald(8) قيد التشغيل (المقبس غير موجود)، فإن تلك الدوال لا تفعل شيئًا، وتعيد أيضًا 0.

أمان الخيوط

جميع الدوال المدرجة هنا آمنة للخيوط ويمكن استدعاؤها بالتوازي من خيوط متعددة.

sd_journal_sendv() وsd_journal_sendv_with_location() "آمنة للإشارات غير المتزامنة" بمعنى signal-safety(7).

sd_journal_print() وsd_journal_printv() وsd_journal_send() وsd_journal_perror() ونظيراتها مع "_with_location" ليست آمنة للإشارات غير المتزامنة.

ملاحظات

الدوال الموصوفة هنا متاحة كمكتبة مشتركة، يمكن الترجمة مقابلها والربط بها باستخدام ملف libsystemd pkg-config(1).

يستخدم الكود الموصوف هنا getenv(3)، والذي يُعلن أنه غير آمن متعدد الخيوط. هذا يعني أن الكود الذي يستدعي الدوال الموصوفة هنا يجب ألا يستدعي setenv(3) من خيط متوازٍ. يُوصى بإجراء استدعاءات setenv() فقط من مرحلة مبكرة من البرنامج عندما لم تكن قد بدأت خيوط أخرى.
