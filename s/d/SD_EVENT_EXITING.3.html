#include <systemd/sd-event.h>
enum {
        SD_EVENT_INITIAL,

int sd_event_prepare(sd_event *event);

int sd_event_dispatch(sd_event *event);

int sd_event_get_state(sd_event *event);

int sd_event_get_iteration(sd_event *event, uint64_t *ret);

الوصف

يمكن استخدام الدوال منخفضة المستوى sd_event_prepare() و sd_event_wait() و sd_event_dispatch() لتنفيذ مراحل محددة من حلقة الأحداث. راجع sd_event_run(3) و sd_event_loop(3) للدوال عالية المستوى التي تنفذ تكرارات فردية ولكن كاملة لحلقة الأحداث أو تشغلها باستمرار.

تقوم sd_event_prepare() بالتحقق من الأحداث المعلقة وتجهيز المؤقتات اللازمة. إذا كانت أي أحداث جاهزة للمعالجة ( "معلقة")، فإنها تُرجع قيمة موجبة وغير صفرية، ويجب على المتصل معالجة هذه الأحداث باستخدام sd_event_dispatch().

تقوم sd_event_dispatch() بتوزيع مصدر الحدث ذي الأولوية الأعلى الذي لديه حدث معلق. عند النجاح، تُرجع sd_event_dispatch() إما صفرًا، مما يشير إلى أنه لا يمكن توزيع المزيد من مصادر الأحداث وأنه تم طلب الخروج من حلقة الأحداث عبر sd_event_exit(3)؛ أو قيمة موجبة غير صفرية، مما يعني أنه تم توزيع مصدر حدث وعادت الحلقة إلى حالتها الأولية، ويجب على المتصل بدء تكرار حلقة الأحداث التالي عن طريق استدعاء sd_event_prepare() مرة أخرى.

في حال أعادت sd_event_prepare() صفرًا، يجب استدعاء sd_event_wait() للانتظار لمزيد من الأحداث أو انتهاء المهلة. إذا كانت أي أحداث جاهزة للمعالجة، فإنها تُرجع قيمة موجبة غير صفرية، ويجب توزيع الأحداث باستخدام sd_event_dispatch(). بخلاف ذلك، عادت حلقة الأحداث إلى حالتها الأولية ويجب بدء تكرار حلقة الأحداث التالي عن طريق استدعاء sd_event_prepare() مرة أخرى.

يمكن استخدام sd_event_get_state() لتحديد الحالة التي تكون فيها حلقة الأحداث حاليًا. تُرجع إحدى الحالات الموضحة أدناه.

يمكن استخدام sd_event_get_iteration() لتحديد التكرار الحالي لحلقة الأحداث. تُرجع عددًا صحيحًا موجبًا غير موقع بحجم 64 بت يحتوي على عداد يزداد بشكل رتيب مع كل تكرار لحلقة الأحداث، بدءًا من 0. يتم زيادة العداد في وقت استدعاء sd_event_prepare().

تأخذ جميع الدوال الخمس، كوسيط أول، كائن حلقة الأحداث event الذي تم إنشاؤه باستخدام sd_event_new(). يتم تحديد المهلة لـ sd_event_wait() بالـ usec بالميكروثانية. يمكن استخدام (uint64_t) -1 لتحديد مهلة غير محدودة.

آلة الحالة

تعرف حلقة الأحداث الحالات التالية، التي يمكن الاستعلام عنها باستخدام sd_event_get_state().

SD_EVENT_INITIAL

الحالة الأولية التي تكون فيها حلقة الأحداث، قبل كل تكرار لحلقة الأحداث. استخدم sd_event_prepare() للانتقال بحلقة الأحداث إلى الحالتين SD_EVENT_ARMED أو SD_EVENT_PENDING.

تمت الإضافة في الإصدار 229.

SD_EVENT_PREPARING

يتم حاليًا تجهيز مصدر حدث، أي أنه يتم حاليًا تنفيذ معالج التجهيز، كما تم تعيينه باستخدام sd_event_source_set_prepare(3). هذه الحالة تُرى فقط في معالج تجهيز مصدر الحدث الذي يتم استدعاؤه من استدعاء sd_event_prepare() ويتبعه مباشرة SD_EVENT_ARMED أو SD_EVENT_PENDING.

تمت الإضافة في الإصدار 229.

SD_EVENT_ARMED

تم استدعاء sd_event_prepare() ولم تكن هناك مصادر أحداث جاهزة للتوزيع. استخدم sd_event_wait() للانتظار لأحداث جديدة، والانتقال إلى SD_EVENT_PENDING أو العودة إلى SD_EVENT_INITIAL.

تمت الإضافة في الإصدار 229.

SD_EVENT_PENDING

تم استدعاء sd_event_prepare() أو sd_event_wait() وكانت هناك مصادر أحداث بأحداث معلقة. استخدم sd_event_dispatch() لتوزيع مصدر الحدث ذي الأولوية الأعلى والانتقال إلى SD_EVENT_INITIAL أو SD_EVENT_FINISHED.

تمت الإضافة في الإصدار 229.

SD_EVENT_RUNNING

يتم حاليًا توزيع مصدر حدث منتظم. هذه الحالة تُرى فقط في معالج مصدر الحدث الذي يتم استدعاؤه من استدعاء sd_event_dispatch()، ويتبعه مباشرة SD_EVENT_INITIAL أو SD_EVENT_FINISHED بمجرد عودة معالج مصدر الحدث. لاحظ أنه أثناء توزيع مصادر أحداث الخروج، تُرى الحالة SD_EVENT_EXITING بدلاً من ذلك.

تمت الإضافة في الإصدار 229.

SD_EVENT_EXITING

على غرار SD_EVENT_RUNNING ولكنها الحالة السارية أثناء توزيع مصادر أحداث الخروج. تتبعها SD_EVENT_INITIAL أو SD_EVENT_FINISHED بمجرد عودة معالج الحدث.

تمت الإضافة في الإصدار 229.

SD_EVENT_FINISHED

انتهت حلقة الأحداث. تم تشغيل جميع مصادر أحداث الخروج. إذا كانت حلقة الأحداث في هذه الحالة، فإنها لا فائدة منها بعد الآن، ويجب تحريرها.

تمت الإضافة في الإصدار 229.

يوضح الرسم البياني المبسط للحالات والاستدعاءات للانتقال بينها أدناه. لاحظ أن SD_EVENT_PREPARING و SD_EVENT_RUNNING و SD_EVENT_EXITING غير موضحة هنا.

قيمة الإرجاع

عند النجاح، تُرجع هذه الدوال 0 أو عددًا صحيحًا موجبًا. عند الفشل، تُرجع رمز خطأ سالبًا بأسلوب errno. في حالة sd_event_prepare() و sd_event_wait()، يشير رمز إرجاع موجب وغير صفري إلى أن الأحداث جاهزة للمعالجة وصفر يشير إلى عدم وجود أحداث جاهزة. في حالة sd_event_dispatch()، يشير رمز إرجاع موجب وغير صفري إلى أن حلقة الأحداث عادت إلى حالتها الأولية وصفر يشير إلى أن حلقة الأحداث قد انتهت. تُرجع sd_event_get_state() حالة موجبة أو صفر عند النجاح.

الأخطاء

قد تشير الأخطاء المرجعة إلى المشاكل التالية:

-EINVAL

المعامل event غير صالح أو NULL.

-EBUSY

كائن حلقة الأحداث ليس في الحالة الصحيحة.

-ESTALE

حلقة الأحداث منتهية بالفعل.
