الخلاصة

#include <systemd/sd-journal.h>

int sd_journal_get_fd(sd_journal *j);

int sd_journal_get_events(sd_journal *j);

int sd_journal_get_timeout(sd_journal *j, uint64_t *timeout_usec);

int sd_journal_wait(sd_journal *j, uint64_t timeout_usec);

int sd_journal_reliable_fd(sd_journal *j);

الوصف

sd_journal_get_fd() تُرجع واصف ملف يمكن الاستقصاء بشكل غير متزامن في حلقة أحداث خارجية ويتم تنبيهه بمجرد تغيير السجل، بسبب إضافة إدخالات أو ملفات جديدة، أو حدوث تدوير، أو حذف ملفات، وما إلى ذلك. واصف الملف مناسب للاستخدام في poll(2). استخدم sd_journal_get_events() لقناع الأحداث للمراقبة. تأخذ الدالة وسيطًا واحدًا: كائن سياق السجل. لاحظ أن ليست كل أنظمة الملفات قادرة على إنشاء الأحداث اللازمة للتنبيه من واصف الملف هذا لتلاحظ التغييرات على الفور. في الواقع، لا تنشئ أنظمة الملفات الشبكية أحداث تغيير ملف مناسبة في جميع الحالات. يمكن اكتشاف حالات مثل هذه باستخدام sd_journal_reliable_fd()، أدناه. ستضمن sd_journal_get_timeout() في هذه الحالات أن تحدث التنبيهات بشكل متكرر بما يكفي لملاحظة التغييرات، على الرغم من وجود تأخير معين.

sd_journal_get_events() ستُرجع قناع poll() للانتظار. ستُرجع هذه الدالة مجموعة من POLLIN و POLLOUT وما شابه لملء حقل ".events" في struct pollfd.

sd_journal_get_timeout() ستُرجع قيمة مهلة للاستخدام في poll(). تُرجع هذه القيمة بالميكروثانية منذ بداية فترة CLOCK_MONOTONIC لمهلة poll() في timeout_usec. انظر clock_gettime(2) للحصول على تفاصيل حول CLOCK_MONOTONIC. إذا لم تكن هناك مهلة للانتظار، فستملأ (uint64_t) -1 بدلاً من ذلك. لاحظ أن poll() تأخذ مهلة نسبية بالميللي ثانية بدلاً من مهلة مطلقة بالميكروثانية. لتحويل المهلة المطلقة "us" إلى "ms" نسبية، استخدم كودًا مثل التالي:

uint64_t t;
int msec;
sd_journal_get_timeout(m, &t);
if (t == (uint64_t) -1)
  msec = -1;
else {
  struct timespec ts;
  uint64_t n;
  clock_gettime(CLOCK_MONOTONIC, &ts);
  n = (uint64_t) ts.tv_sec * 1000000 + ts.tv_nsec / 1000;
  msec = t > n ? (int) ((t - n + 999) / 1000) : 0;
}

الكود أعلاه لا يقوم بأي فحص للأخطاء للاختصار. يمكن تمرير المتغير msec المحسوب مباشرة كمعامل مهلة لـ poll().

بعد كل تنبيه من poll()، يجب استدعاء sd_journal_process() لمعالجة الأحداث. ستبين هذه الاستدعاء أيضًا ما هو نوع التغيير الذي تم اكتشافه (انظر أدناه؛ لاحظ أن التنبيهات الزائفة ممكنة).

بديل متزامن لاستخدام sd_journal_get_fd() و sd_journal_get_events() و sd_journal_get_timeout() و sd_journal_process() هو sd_journal_wait(). ستنتظر هذه الدالة بشكل متزامن حتى يتغير السجل. يمكن التحكم في الحد الأقصى للوقت الذي تنام فيه هذه الدالة باستخدام المعامل timeout_usec. قم بتمرير (uint64_t) -1 للانتظار إلى أجل غير مسمى. داخليًا، تجمع هذه الدالة ببساطة sd_journal_get_fd() و sd_journal_get_events() و sd_journal_get_timeout() و poll() و sd_journal_process() في دالة واحدة.

يمكن استخدام sd_journal_reliable_fd() للتحقق مما إذا كانت أحداث التنبيه من واصف الملف الذي تم إرجاعه بواسطة sd_journal_get_fd() معروفة بأنها يتم تشغيلها بسرعة. على أنظمة الملفات معينة حيث لا تتوفر أحداث تغيير الملف من نظام التشغيل (مثل NFS)، يجب استقصاء التغييرات بشكل متكرر، وبالتالي لا يتم اكتشافها إلا مع تأخير كبير. ستُرجع هذه الدالة قيمة موجبة إذا تم اكتشاف تغييرات السجل بسرعة وصفر إذا كانت تحتاج إلى استقصاء. لاحظ أنه لا يوجد عادةً حاجة لاستدعاء هذه الدالة مباشرة لأن sd_journal_get_timeout() ستطلب مهلات مناسبة على أي حال.

لاحظ أن جميع واجهات إشعار التغيير المذكورة أعلاه لا تُبلغ عن التغييرات فورًا. يتم تقديم التأخيرات لأسباب متعددة: كما ذكرنا، تتطلب بعض الواجهات الخلفية للتخزين الاستقصاء الزمني، وفي حالات أخرى، يتم تحسين التنبيهات عن طريق تجميع الأحداث، ويقدم نظام التشغيل تأخيرات إضافية في جدولة الإدخال/الإخراج/وحدة المعالجة المركزية.

قيمة الإرجاع

sd_journal_get_fd() تُرجع واصف ملف صالحًا عند النجاح أو رمز خطأ سالبًا بأسلوب errno.

sd_journal_get_events() تُرجع مجموعة من POLLIN و POLLOUT وما شابهها عند النجاح أو رمز خطأ سالبًا بأسلوب errno.

sd_journal_reliable_fd() تُرجع عددًا صحيحًا موجبًا إذا كان واصف الملف الذي تم إرجاعه بواسطة sd_journal_get_fd() سيُنشئ تنبيهات فورية لجميع تغييرات السجل. تُرجع 0 إذا كان قد يكون هناك تأخير متضمن.

sd_journal_process() و sd_journal_wait() تُرجعان رمز خطأ سالبًا بأسلوب errno، أو أحد SD_JOURNAL_NOP و SD_JOURNAL_APPEND و SD_JOURNAL_INVALIDATE عند النجاح:

•إذا تم إرجاع SD_JOURNAL_NOP، لم يتغير السجل منذ الاستدعاء الأخير.
•إذا تم إرجاع SD_JOURNAL_APPEND، تمت إضافة إدخالات جديدة إلى نهاية السجل. في هذه الحالة، يكفي ببساطة الاستمرار في القراءة من الموقع النهائي السابق للسجل، لقراءة الإدخالات المضافة حديثًا.
•إذا تم إرجاع SD_JOURNAL_INVALIDATE، تمت إضافة ملفات سجل أو إزالتها من مجموعة ملفات السجل التي تمت مراقبتها (على سبيل المثال، بسبب التدوير أو التنظيف)، وبالتالي قد تكون الإدخالات قد ظهرت أو اختفت في أماكن عشوائية في تدفق السجل، ربما قبل أو بعد النهاية السابقة لتدفق السجل. إذا تم إرجاع SD_JOURNAL_INVALIDATE، فيجب على واجهات المستخدم ذات العرض المباشر التي ترغب في عكس حالة بيانات السجل على القرص على الشاشة تحديث شاشتها بالكامل (بالنسبة لمؤشر إدخال السجل في أعلى الشاشة). البرامج المهتمة فقط بتدفق بيانات السجل المتسلسل بشكل صارم قد تعامل SD_JOURNAL_INVALIDATE بنفس طريقة SD_JOURNAL_APPEND، وبالتالي تتجاهل أي تغييرات في عرض السجل قبل نهاية السجل السابقة.

سلامة الإشارة

بشكل عام، sd_journal_get_fd() و sd_journal_get_events() و sd_journal_get_timeout() ليست "آمنة للإشارة غير المتزامنة" بالمعنى المقصود في signal-safety(7). ومع ذلك، فإن الاستدعاء الأول لأي من هذه الدوال الثلاث فقط يقوم بعمليات غير آمنة، لذا فإن الاستدعاءات اللاحقة *آمنة*.

sd_journal_process() و sd_journal_wait() غير آمنتين. sd_journal_reliable_fd() آمنة.

ملاحظات

جميع الدوال المدرجة هنا غير مرتبطة بالخيوط (thread-agnostic)، وخيط واحد فقط يمكنه العمل على كائن معين في أي وقت. يمكن للخيوط المختلفة الوصول إلى نفس الكائن في أوقات مختلفة. يمكن استخدام كائنات مستقلة متعددة من خيوط مختلفة بالتوازي.

الدوال الموصوفة هنا متاحة كمكتبة مشتركة، والتي يمكن تجميعها والربط بها باستخدام ملف libsystemd pkg-config(1).

sd_journal_wait(3) — systemd manual pages

NAME

sd_journal_wait, sd_journal_process, sd_journal_get_fd, sd_journal_get_events, sd_journal_get_timeout, sd_journal_reliable_fd - Wait for journal entries, process queued entries, get journal file descriptor and event mask

SYNOPSIS

sd_journal_wait(3);
sd_journal_process(3);
sd_journal_get_fd(3);
sd_journal_get_events(3);
sd_journal_get_timeout(3);
sd_journal_reliable_fd(3);

Headers:


#include <systemd/sd-journal.h>

DESCRIPTION

The function sd_journal_wait() blocks until new journal entries are available, or a timeout occurs. It takes a pointer to a journal object j and a timeout value usec in microseconds. If usec is 0, the function returns immediately. If usec is (uint64_t)-1, it will block indefinitely until new entries are available.

The function sd_journal_process() should be called after sd_journal_wait() returns with a positive value or after sd_journal_get_fd() returns a valid file descriptor and sd_journal_get_events() returns events that indicate data is ready. It processes any pending journal entries and returns the number of new entries that are available for reading. This count may be zero if the event was a non-data event (e.g. a journal rotation).

The function sd_journal_get_fd() returns a file descriptor suitable for use with poll(2). This file descriptor may be used to wait for journal entries to become available.

The function sd_journal_get_events() returns a bitmask of events that should be monitored using poll(2) on the file descriptor returned by sd_journal_get_fd(). This bitmask may include POLLIN to indicate that new entries are available, and POLLPRI to indicate that the journal has rotated.

The function sd_journal_get_timeout() returns the recommended timeout in microseconds for polling. This value should be used as the timeout argument for poll(2). If the journal is configured to block indefinitely, this function returns (uint64_t)-1.

The function sd_journal_reliable_fd() is a synonym for sd_journal_get_fd(). It is provided for backward compatibility and should not be used in new code.

RETURN VALUE

sd_journal_wait() returns the number of new journal entries available, or a negative error code.

sd_journal_process() returns the number of new journal entries available for reading, or a negative error code.

sd_journal_get_fd() returns a file descriptor on success, or a negative error code on failure.

sd_journal_get_events() returns a bitmask of events to monitor on success, or a negative error code on failure.

sd_journal_get_timeout() returns the timeout in microseconds on success, or a negative error code on failure.

SIGNAL SAFETY

sd_journal_wait(), sd_journal_process(), sd_journal_get_fd(), sd_journal_get_events(), sd_journal_get_timeout(), and sd_journal_reliable_fd() are async-signal-safe.

NOTES

The file descriptor returned by sd_journal_get_fd() should not be closed or modified. It is owned by the sd-journal(3) library.

The timeout returned by sd_journal_get_timeout() is the time until the next journal event is expected. If this timeout is reached, sd_journal_process() should still be called to handle potential journal rotations or other non-data events.

The journal entries are buffered internally by the sd-journal(3) library. When sd_journal_wait() returns with a positive value, it indicates that new entries are available. These entries must be read using functions like sd_journal_get_data(3) or sd_journal_get_message(3) until the count returned by sd_journal_process() reaches zero.

It is recommended to use sd_journal_get_fd(), sd_journal_get_events(), and sd_journal_get_timeout() in combination with poll(2) for efficient event driven processing of journal entries.

Note that the file descriptor returned by sd_journal_get_fd() is not always readable. You should always use sd_journal_get_events() to determine which events to wait for, and then call sd_journal_process() after the event occurs. Relying on the file descriptor becoming readable without consulting the event mask may lead to incorrect behavior.

EXAMPLES

The following example shows how to read all available journal entries:

#include <stdio.h>
#include <string.h>
#include <systemd/sd-journal.h>
int main(int argc, char *argv[]) {
  sd_journal *j = NULL;
  const char *data;
  size_t length;
  int r;
  /* Open the journal in the default mode */
  r = sd_journal_open(&j, SD_JOURNAL_LOCAL_ONLY);
  if (r < 0) {
    fprintf(stderr, "Failed to open journal: %s\n", strerror(-r));
    return 1;
  }
  /* Loop over all available entries */
  for (;;) {
    /* Wait for new entries, with indefinite timeout */
    r = sd_journal_wait(j, (uint64_t) -1);
    if (r < 0) {
      fprintf(stderr, "Failed to wait for journal entry: %s\n", strerror(-r));
      break;
    }
    /* Process the entries, and get the count of new entries */
    r = sd_journal_process(j);
    if (r < 0) {
      fprintf(stderr, "Failed to process journal: %s\n", strerror(-r));
      break;
    }
    if (r == 0) {
      /* No new entries, continue waiting */
      continue;
    }
    /* Read the MESSAGE field */
    r = sd_journal_get_data(j, "MESSAGE", &data, &length);
    if (r < 0) {
      fprintf(stderr, "Failed to get MESSAGE field: %s\n", strerror(-r));
      continue;
    }
    /* Print the message */
    printf("%.*s\n", (int) length, data);
  }
  sd_journal_close(j);
  return 0;
}

Waiting with poll() (this example lacks all error checking for the sake of simplicity):

/* SPDX-License-Identifier: MIT-0 */
#define _GNU_SOURCE 1
#include <poll.h>
#include <time.h>
#include <systemd/sd-journal.h>
int wait_for_changes(sd_journal *j) {
  uint64_t t;
  int msec;
  struct pollfd pollfd;
  sd_journal_get_timeout(j, &t);
  if (t == (uint64_t) -1)
    msec = -1;
  else {
    struct timespec ts;
    uint64_t n;
    clock_gettime(CLOCK_MONOTONIC, &ts);
    n = (uint64_t) ts.tv_sec * 1000000 + ts.tv_nsec / 1000;
    msec = t > n ? (int) ((t - n + 999) / 1000) : 0;
  }
  pollfd.fd = sd_journal_get_fd(j);
  pollfd.events = sd_journal_get_events(j);
  poll(&pollfd, 1, msec);
  return sd_journal_process(j);
}
