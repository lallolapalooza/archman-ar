sd_hwdb_seek(3)

الحزمة: systemd-libs

التراخيص: LGPL-2.1-or-later, CC0-1.0, GPL-2.0-or-later WITH Linux-syscall-note

SD_HWDB_GET(3) sd_hwdb_get SD_HWDB_GET(3)

الاسم

sd_hwdb_get, sd_hwdb_seek, sd_hwdb_enumerate, SD_HWDB_FOREACH_PROPERTY - الانتقال إلى موقع في hwdb أو الوصول إلى الإدخالات

الملخص

#include <systemd/sd-hwdb.h>

int sd_hwdb_get(sd_hwdb *hwdb, const char *modalias, const char *key, const char **value);

int sd_hwdb_seek(sd_hwdb *hwdb, const char *modalias);

int sd_hwdb_enumerate(sd_hwdb *hwdb, const char **key, const char **value);

SD_HWDB_FOREACH_PROPERTY(hwdb, modalias, key, value);

الوصف

تقوم sd_hwdb_get() بالاستعلام عن كائن hwdb الذي تم إنشاؤه مسبقًا باستخدام sd_hwdb_new(3) للحصول على الإدخالات التي تطابق السلسلة المحددة modalias، وتعيد القيمة المقابلة للمفتاح key. يتم إرجاع القيمة كسلسلة منتهية بـ NUL في value. لا يجب تعديلها بواسطة المتصل وهي صالحة طالما تم الاحتفاظ بمرجع إلى hwdb. عندما تطابق أنماط متعددة في قاعدة البيانات modalias، يتم استخدام النمط ذي الأولوية الأعلى. انظر hwdb(7) للتفاصيل.

تقوم sd_hwdb_seek() بتحديد الإدخالات التي تطابق السلسلة المحددة modalias. ستصل الاستعلامات اللاحقة باستخدام sd_hwdb_enumerate() إلى أزواج المفاتيح والقيم لهذه السلسلة.

تقوم sd_hwdb_enumerate() بإرجاع (بالتتابع) جميع أزواج المفاتيح والقيم المعرفة للسلسلة المستخدمة مع sd_hwdb_seek(). يتم إرجاع كل زوج كسلاسل منتهية بـ NUL في key و value. لا يجب تعديل السلاسل بواسطة المتصل وهي صالحة طالما تم الاحتفاظ بمرجع إلى hwdb. عندما تطابق أنماط متعددة في قاعدة البيانات modalias، يتم استخدام مجموعة جميع أزواج المفاتيح والقيم المطابقة. انظر hwdb(7) للتفاصيل.

الماكرو SD_HWDB_FOREACH_PROPERTY() يجمع بين sd_hwdb_seek() و sd_hwdb_enumerate(). لا يتم إجراء معالجة للأخطاء ويتوقف التكرار ببساطة عند حدوث خطأ. انظر المثال أدناه.

قيمة الإرجاع

عند النجاح، تعيد sd_hwdb_get() و sd_hwdb_seek() عددًا صحيحًا غير سالب. عند الفشل، يعيدان رمز خطأ سالب بنمط errno.

تعيد sd_hwdb_enumerate() عددًا صحيحًا موجبًا إذا تم العثور على زوج مفتاح-قيمة آخر أو صفر إذا تم تعداد جميع الإدخالات بالفعل. عند الفشل، تعيد رمز خطأ سالب بنمط errno.

الأخطاء

قد تشير الأخطاء المرجعة إلى المشاكل التالية:

-EINVAL

أحد المعلمات هو NULL.

تمت الإضافة في الإصدار 246.

-ENOENT

لم يتم العثور على إدخال للمجلد modalias المحدد.

تمت الإضافة في الإصدار 246.

-EAGAIN

لم يتم استدعاء sd_hwdb_seek() قبل sd_hwdb_enumerate().

تمت الإضافة في الإصدار 246.

ملاحظات

الوظائف الموصوفة هنا متاحة كمكتبة مشتركة، والتي يمكن الترجمة ضدها والربط مع ملف pkg-config(1) الخاص بـ libsystemd.

الكود الموصوف هنا يستخدم getenv(3)، والتي تم الإعلان عنها على أنها غير آمنة للاستخدام في خيوط متعددة. هذا يعني أن الكود الذي يستدعي الوظائف الموصوفة هنا يجب ألا يستدعي setenv(3) من خيط موازٍ. يوصى بالقيام بالاستدعاءات لـ setenv() فقط من مرحلة مبكرة من البرنامج عندما لم يتم بدء أي خيوط أخرى.

أمثلة

مثال 1. البحث عن إدخالات hwdb لجهاز USB

/* SPDX-License-Identifier: MIT-0 */
#define _GNU_SOURCE 1
#include <stdio.h>
#include <stdint.h>
#include <systemd/sd-hwdb.h>
int print_usb_properties(uint16_t vid, uint16_t pid) {
  char match[128];
  sd_hwdb *hwdb;
  const char *key, *value;
  int r;
  /* تطابق معرف البائع ومعرف المنتج هذا لـ USB */
  snprintf(match, sizeof match, "usb:v%04Xp%04X", vid, pid);
  r = sd_hwdb_new(&hwdb);
  if (r < 0)
    return r;
  SD_HWDB_FOREACH_PROPERTY(hwdb, match, key, value)
    printf("%s: \"%s\" → \"%s\"\n", match, key, value);
  sd_hwdb_unref(hwdb);
  return 0;
}
int main(int argc, char **argv) {
  print_usb_properties(0x046D, 0xC534);
  return 0;
}

التأثير مشابه لاستدعاء systemd-hwdb query usb:v046DpC534.
