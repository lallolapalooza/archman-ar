sd_bus_message_read(3)

الحزمة: systemd-libs

التراخيص: LGPL-2.1-or-later, CC0-1.0, GPL-2.0-or-later WITH Linux-syscall-note

NAME

sd_bus_message_read, sd_bus_message_readv, sd_bus_message_peek_type — قراءة البيانات من رسالة D-Bus

SYNOPSIS

#include <systemd/sd-bus.h>
int64_t sd_bus_message_read(sd_bus_message *m,
                                 const char *fmt, ...);
int sd_bus_message_readv(sd_bus_message *m,
                                const char *fmt,
                                va_list ap);
int sd_bus_message_peek_type(sd_bus_message *m,
                                   const char *type,
                                   size_t *n);

DESCRIPTION

تستخدم الدالة sd_bus_message_read() لقراءة البيانات من رسالة D-Bus. تأخذ الدالة كمعاملات مؤشرًا إلى رسالة D-Bus (m) و سلسلة تنسيق (fmt) تصف أنواع البيانات المراد قراءتها، تليها قائمة بالمتغيرات التي سيتم كتابة القيم المقروءة إليها.

تتطابق سلسلة التنسيق مع مجموعة من الرموز التي تمثل أنواع بيانات D-Bus. يتم قراءة البيانات من الرسالة بالترتيب المحدد في سلسلة التنسيق، ويتم تخزينها في المتغيرات المقابلة.

الدالة sd_bus_message_readv() هي نسخة من sd_bus_message_read() تأخذ قائمة متغيرات (ap) بدلاً من تمرير المتغيرات بشكل مباشر.

الدالة sd_bus_message_peek_type() تسمح بفحص نوع البيانات التالية في الرسالة دون قراءتها فعليًا. يأخذ المعامل type مؤشرًا إلى سلسلة حرفية تحتوي على نوع البيانات المتوقع. إذا تطابق النوع التالي في الرسالة مع النوع المتوقع، تقوم الدالة بتحديث المعامل n بعدد العناصر من هذا النوع (خاصة للحاويات) وتعيد 0. إذا لم يتطابق النوع، تعيد خطأ.

تتوافق رموز التنسيق مع أنواع بيانات D-Bus الأساسية: b (boolean)، y (byte)، n (int16)، q (uint16)، i (int32)، u (uint32)، x (int64)، t (uint64)، d (double)، s (string)، o (object path)، g (signature)، h (file descriptor).

يمكن أيضًا قراءة الهياكل والحاويات. تتمثل الهياكل في أزواج من الأنواع محاطة بأقواس (())، وتتمثل القواميس في مصفوفات من أزواج المفتاح/القيمة محاطة بـ a{}. يمكن تداخل الحاويات.

عند قراءة أنواع بيانات أساسية، يجب تمرير مؤشر إلى المتغير المناسب. عند قراءة السلاسل، يتم تمرير مؤشر إلى const char*. عند قراءة مسارات الكائنات، يتم تمرير مؤشر إلى const char*. عند قراءة التوقيعات، يتم تمرير مؤشر إلى const char*. عند قراءة الواصفات، يتم تمرير مؤشر إلى int.

عند قراءة هياكل، يجب تمرير مؤشرات إلى المتغيرات المقابلة لكل عنصر في الهيكل.

عند قراءة القواميس، يجب توفير عدد العناصر كمعامل إضافي قبل مؤشرات العناصر.

عند قراءة متغير (v)، يجب توفير توقيع المتغير كمعامل إضافي، متبوعًا بمؤشر إلى المتغير الفعلي.

RETURN VALUE

تعيد الدالة sd_bus_message_read() و sd_bus_message_readv() عددًا غير سالب في حالة النجاح، أو قيمة سالبة في حالة الخطأ. القيم الإيجابية تعكس عدد العناصر التي تم قراءتها بنجاح.

تعيد الدالة sd_bus_message_peek_type() 0 في حالة النجاح، أو قيمة سالبة في حالة الخطأ.

NOTES

يجب التأكد من أن أنواع البيانات الممررة إلى sd_bus_message_read() تتطابق مع رموز التنسيق. قد يؤدي عدم التطابق إلى سلوك غير محدد أو تعطل البرنامج.

عند قراءة أنواع الأعداد الصحيحة، تأكد من استخدام الأنواع المناسبة (مثل int32_t لـ i، uint64_t لـ t).

تذكر تحرير الذاكرة المخصصة للسلاسل ومسارات الكائنات المرجعة من الرسالة عند الانتهاء منها (عادةً باستخدام sd_string_free أو sd_path_free).

عند قراءة واصفات الملفات (h)، يتم نقل ملكية الواصف من الرسالة إلى المتغير. إذا كنت بحاجة إلى الاحتفاظ بالواصف بعد تحرير الرسالة، فيجب عليك تكراره (على سبيل المثال، باستخدام fcntl(fd, FD_DUPFD_CLOEXEC, 3)).

EXAMPLES

اقرأ نوعًا أساسيًا واحدًا (عدد صحيح 64 بت):

sd_bus_message *m;
int64_t x;
sd_bus_message_read(m, "x", &x);

اقرأ قيمة منطقية:

sd_bus_message *m;
int x; /* لا تستخدم نوع "bool" الخاص بـ C99 هنا، فهو عادةً أصغر
          في الذاكرة وقد يسبب تلفًا في الذاكرة */
sd_bus_message_read(m, "b", &x);

اقرأ جميع أنواع الأعداد الصحيحة:

uint8_t y;
int16_t n;
uint16_t q;
int32_t i;
uint32_t u;
int64_t x;
uint64_t t;
double d;
sd_bus_message_read(m, "ynqiuxtd", &y, &n, &q, &i, &u, &x, &t, &d);

اقرأ هيكلًا يتكون من سلسلة ومسار D-Bus:

const char *s, *p;
sd_bus_message_read(m, "(so)", &s, &p);

اقرأ متغيرًا، مع النوع الحقيقي "gt" (توقيع، عدد صحيح غير موقع):

const char *s;
uint64_t *v;
sd_bus_message_read(m, "v", "gt", &s, &v);

اقرأ قاموسًا يحتوي على ثلاثة أزواج من النوع {عدد صحيح=>سلسلة}:

int i, j, k;
const char *s, *t, *u;
sd_bus_message_read(m, "a{is}", 3, &i, &s, &j, &t, &k, &u);

اقرأ واصف ملف واحد، وقم بتكراره للاحتفاظ به مفتوحًا بعد تحرير الرسالة.

sd_bus_message *m;
int fd, fd_copy;
sd_bus_message_read(m, "h", &fd);
fd_copy = fcntl(fd, FD_DUPFD_CLOEXEC, 3);

HISTORY

تمت إضافة sd_bus_message_peek_type() في الإصدار 246.
