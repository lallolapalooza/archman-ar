sd_bus_message_close_container(3)

الحزمة: systemd-libs

التراخيص: LGPL-2.1-or-later, CC0-1.0, GPL-2.0-or-later WITH Linux-syscall-note

SD_BUS_MESSAGE_OPEN_CONTAINER(3) sd_bus_message_open_container SD_BUS_MESSAGE_OPEN_CONTAINER(3)

الاسم

sd_bus_message_open_container, sd_bus_message_close_container, sd_bus_message_enter_container, sd_bus_message_exit_container - إنشاء حاويات والتحرك بينها في رسائل D-Bus

ملخص

#include <systemd/sd-bus.h>

int sd_bus_message_open_container(sd_bus_message *m, char type, const char *contents);

int sd_bus_message_close_container(sd_bus_message *m);

int sd_bus_message_enter_container(sd_bus_message *m, char type, const char *contents);

int sd_bus_message_exit_container(sd_bus_message *m);

الوصف

تضيف sd_bus_message_open_container() حاوية جديدة إلى الرسالة m. بعد فتح حاوية جديدة، يمكن ملؤها بالمحتوى باستخدام sd_bus_message_append(3) ووظائف مشابهة. الحاويات تعمل مثل المكدس. لتداخل الحاويات داخل بعضها البعض، قم باستدعاء sd_bus_message_open_container() عدة مرات دون استدعاء sd_bus_message_close_container() بينها. سيتم تداخل كل حاوية داخل الحاوية السابقة. يمثل type نوع الحاوية ويجب أن يكون أحد "r" أو "a" أو "v" أو "e" كما هو موضح في sd_bus_message_append(3). بدلاً من النصوص الحرفية، يمكن أيضًا استخدام الثوابت المقابلة SD_BUS_TYPE_STRUCT، SD_BUS_TYPE_ARRAY، SD_BUS_TYPE_VARIANT أو SD_BUS_TYPE_DICT_ENTRY. يصف contents نوع عناصر الحاوية ويجب أن يكون سلسلة أنواع D-Bus تتبع القواعد الموضحة في sd_bus_message_append(3).

تغلق sd_bus_message_close_container() آخر حاوية تم فتحها باستخدام sd_bus_message_open_container(). عند النجاح، يتم وضع مؤشر الكتابة للرسالة m بعد الحاوية المغلقة في حاويتها الأم أو في m نفسها إذا لم تكن هناك حاوية أم.

تدخل sd_bus_message_enter_container() الحاوية التالية في الرسالة m للقراءة. تعمل بشكل مشابه جدًا لـ sd_bus_message_open_container(). يسمح الدخول إلى حاوية بقراءة محتوياتها باستخدام sd_bus_message_read(3) ووظائف مشابهة. type و contents هما نفس ما في sd_bus_message_open_container().

تخرج sd_bus_message_exit_container() نطاق آخر حاوية تم الدخول إليها باستخدام sd_bus_message_enter_container(). تعمل بشكل مشابه جدًا لـ sd_bus_message_close_container(). لاحظ أنه لا يمكن استدعاء sd_bus_message_exit_container() إلا بعد المرور عبر جميع أعضاء الحاوية، أي القراءة أو التخطي فوقها. استخدم sd_bus_message_skip(3) لتخطي الحقول من حاوية لتكون قادرًا على الخروج من الحاوية باستخدام sd_bus_message_exit_container() دون قراءة جميع الأعضاء.

قيمة الإرجاع

عند النجاح، تعيد هذه الوظائف قيمة صحيحة غير سالبة. تعيد sd_bus_message_open_container() و sd_bus_message_close_container() القيمة 0. تعيد sd_bus_message_enter_container() القيمة 1 إذا نجحت في فتح حاوية جديدة، و 0 إذا لم يكن ذلك ممكنًا لأنه تم الوصول إلى نهاية الحاوية المفتوحة حاليًا أو الرسالة. تعيد sd_bus_message_exit_container() القيمة 1 عند النجاح. عند الفشل، تعيد جميع هذه الوظائف قيمة خطأ سالبة بأسلوب errno.

الأخطاء

قد تشير الأخطاء المرجعة إلى المشاكل التالية:

-EINVAL

m أو contents هي NULL أو type غير صالحة.

تمت الإضافة في الإصدار 246.

-EBADMSG

الرسالة m ذات بنية غير صالحة.

تمت الإضافة في الإصدار 254.

-ENXIO

الرسالة m لا تحتوي على حاوية من النوع type في الموضع الحالي، أو أن المحتويات لا تتطابق مع contents.

تمت الإضافة في الإصدار 254.

-EPERM

الرسالة m مختومة بالفعل.

تمت الإضافة في الإصدار 246.

-ESTALE

الرسالة m في حالة غير صالحة.

تمت الإضافة في الإصدار 246.

-ENOMEM

فشل تخصيص الذاكرة.

تمت الإضافة في الإصدار 246.

-EBUSY

تم استدعاء sd_bus_message_exit_container() ولكن لا تزال هناك أعضاء غير مقروءة في الحاوية.

تمت الإضافة في الإصدار 247.

ملاحظات

الوظائف الموصوفة هنا متاحة كمكتبة مشتركة، والتي يمكن تجميعها وربطها باستخدام ملف pkg-config(1) الخاص بـ libsystemd.

الكود الموصوف هنا يستخدم getenv(3)، والذي تم تعريفه على أنه غير آمن للاستخدام المتعدد الخيوط. هذا يعني أن الكود الذي يستدعي الوظائف الموصوفة هنا يجب ألا يستدعي setenv(3) من خيط موازٍ. يوصى بإجراء استدعاءات لـ setenv() فقط من مرحلة مبكرة من البرنامج عندما لم يتم بدء أي خيوط أخرى.

أمثلة

مثال 1. إضافة مصفوفة من السلاسل إلى رسالة

/* SPDX-License-Identifier: MIT-0 */
#include <systemd/sd-bus.h>
int append_strings_to_message(sd_bus_message *m, const char *const *arr) {
  const char *s;
  int r;
  r = sd_bus_message_open_container(m, 'a', "s");
  if (r < 0)
    return r;
  for (s = *arr; *s; s++) {
    r = sd_bus_message_append(m, "s", s);
    if (r < 0)
      return r;
  }
  return sd_bus_message_close_container(m);
}

مثال 2. قراءة مصفوفة من السلاسل من رسالة

/* SPDX-License-Identifier: MIT-0 */
#include <stdio.h>
#include <systemd/sd-bus.h>
int read_strings_from_message(sd_bus_message *m) {
  int r;
  r = sd_bus_message_enter_container(m, 'a', "s");
  if (r < 0)
    return r;
  for (;;) {
    const char *s;
    r = sd_bus_message_read(m, "s", &s);
    if (r < 0)
      return r;
    if (r == 0)
      break;
    printf("%s\n", s);
  }
  return sd_bus_message_exit_container(m);
}
