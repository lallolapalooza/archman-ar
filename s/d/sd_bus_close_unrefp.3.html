sd_bus_close_unrefp(3)

الحزمة: systemd-libs

التراخيص: LGPL-2.1-or-later, CC0-1.0, GPL-2.0-or-later WITH Linux-syscall-note

SD_BUS_NEW(3) sd_bus_new SD_BUS_NEW(3)

الاسم

sd_bus_new, sd_bus_ref, sd_bus_unref, sd_bus_unrefp, sd_bus_close_unref, sd_bus_close_unrefp, sd_bus_flush_close_unref, sd_bus_flush_close_unrefp - إنشاء كائن ناقل جديد وإنشاء أو تدمير مراجع له

الملخص

#include <systemd/sd-bus.h>

int sd_bus_new(sd_bus **bus);

sd_bus *sd_bus_ref(sd_bus *bus);

sd_bus *sd_bus_unref(sd_bus *bus);

sd_bus *sd_bus_close_unref(sd_bus *bus);

sd_bus *sd_bus_flush_close_unref(sd_bus *bus);

void sd_bus_unrefp(sd_bus **busp);

void sd_bus_close_unrefp(sd_bus **busp);

void sd_bus_flush_close_unrefp(sd_bus **busp);

الوصف

sd_bus_new() تنشئ كائن ناقل جديد. هذا الكائن يتم عد مرافعه، وسيتم تدميره عندما تختفي جميع المراجع. في البداية، يمتلك المتصل بهذه الوظيفة المرجع الوحيد ولن يكون كائن الناقل متصلاً بأي ناقل. لتوصيله بناقل، تأكد من تعيين عنوان باستخدام sd_bus_set_address(3) أو استدعاء ذي صلة، ثم ابدأ الاتصال باستخدام sd_bus_start(3).

في معظم الحالات، من الأفضل استخدام sd_bus_default_user(3)، sd_bus_default_system(3) أو استدعاءات ذات صلة بدلاً من sd_bus_new() و sd_bus_start() الأكثر انخفاضًا. لا تقوم الوظائف ذات المستوى الأعلى بتخصيص كائن ناقل فحسب، بل تبدأ أيضًا الاتصال بناقل معروف جيدًا في استدعاء وظيفي واحد.

sd_bus_ref() تزيد عداد المراجع لـ bus بمقدار واحد.

sd_bus_unref() تقلل عداد المراجع لـ bus بمقدار واحد. بمجرد انخفاض عدد المراجع إلى الصفر، يتم تدمير bus ولا يمكن استخدامه مرة أخرى، لذا فإن الاستدعاءات الإضافية لـ sd_bus_ref() أو sd_bus_unref() غير قانونية.

sd_bus_unrefp() تشبه sd_bus_unref() ولكنها تأخذ مؤشرًا إلى مؤشر إلى كائن sd_bus. هذا الاستدعاء مفيد بالاقتران مع ** سمة متغير التنظيف **[1] الخاصة بـ GCC و LLVM. لاحظ أن هذه الوظيفة معرفة كوظيفة مضمنة. استخدم تعريفًا مثل التالي، وذلك لتخصيص كائن ناقل يتم تحريره تلقائيًا عند مغادرة كتلة التعليمات البرمجية:

{
  __attribute__((cleanup(sd_bus_unrefp))) sd_bus *bus = NULL;
  int r;
  ...
  r = sd_bus_default(&bus);
  if (r < 0) {
    errno = -r;
    fprintf(stderr, "فشل تخصيص الناقل: %m\n");
  }
  ...
}

sd_bus_ref() و sd_bus_unref() لا تقومان بأي عملية إذا كان الوسيط هو NULL. sd_bus_unrefp() ستقوم أولاً بإلغاء إشارة الوسيط الخاص بها، والذي يجب ألا يكون NULL، ولن تقوم بأي عملية إذا كان ذلك هو NULL.

sd_bus_close_unref() تشبه sd_bus_unref()، ولكنها تنفذ أولاً sd_bus_close(3)، مما يضمن إنهاء الاتصال قبل إسقاط المرجع إلى الاتصال وربما تحرير الكائن.

sd_bus_flush_close_unref() تشبه sd_bus_unref()، ولكنها تنفذ أولاً sd_bus_flush(3) وكذلك sd_bus_close(3)، مما يضمن كتابة أي رسائل معلقة بشكل متزامن قبل إسقاط المرجع إلى الاتصال وربما تحرير الكائن. هذا الاستدعاء مفيد بشكل خاص فور الخروج من برنامج لأنه يضمن كتابة أي رسائل صادرة معلقة، وإلغاء معالجة الرسائل الواردة ولكن المكدسة قبل إنهاء الاتصال وتحريره.

sd_bus_close_unrefp() تشبه sd_bus_close_unref()، ولكن يمكن استخدامها في سمة متغير التنظيف الخاصة بـ GCC و LLVM، انظر أعلاه. وبالمثل، sd_bus_flush_close_unrefp() تشبه sd_bus_flush_close_unref().

قيمة الإرجاع

عند النجاح، ترجع sd_bus_new() 0 أو عددًا صحيحًا موجبًا. عند الفشل، ترجع رمز خطأ سالبًا بنمط errno.

sd_bus_ref() ترجع الوسيط دائمًا.

sd_bus_unref() و sd_bus_flush_close_unref() ترجعان دائمًا NULL.

الأخطاء

قد تشير الأخطاء المرجعة إلى المشاكل التالية:

-ENOMEM

فشل تخصيص الذاكرة.

ملاحظات

الوظائف الموصوفة هنا متاحة كمكتبة مشتركة، والتي يمكن تجميعها وربطها بملف libsystemd pkg-config(1).

الكود الموصوف هنا يستخدم getenv(3)، والذي تم تعريفه بأنه غير آمن للعمليات متعددة الخيوط. هذا يعني أن الكود الذي يستدعي الوظائف الموصوفة هنا يجب ألا يستدعي setenv(3) من خيط موازٍ. يوصى بإجراء استدعاءات لـ setenv() فقط من مرحلة مبكرة من البرنامج عندما لم يتم بدء خيوط أخرى.
