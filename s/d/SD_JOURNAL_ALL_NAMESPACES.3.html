SD_JOURNAL_ALL_NAMESPACES(3)

الحزمة: systemd-libs

التراخيص: LGPL-2.1-or-later, CC0-1.0, GPL-2.0-or-later WITH Linux-syscall-note

SD_JOURNAL_OPEN(3) sd_journal_open SD_JOURNAL_OPEN(3)

الاسم

sd_journal_open, sd_journal_open_directory, sd_journal_open_directory_fd, sd_journal_open_files, sd_journal_open_files_fd, sd_journal_open_namespace, sd_journal_close, sd_journal, SD_JOURNAL_LOCAL_ONLY, SD_JOURNAL_RUNTIME_ONLY, SD_JOURNAL_SYSTEM, SD_JOURNAL_CURRENT_USER, SD_JOURNAL_OS_ROOT, SD_JOURNAL_ALL_NAMESPACES, SD_JOURNAL_INCLUDE_DEFAULT_NAMESPACE, SD_JOURNAL_TAKE_DIRECTORY_FD - فتح سجل النظام للقراءة

الملخص

#include <systemd/sd-journal.h>

int sd_journal_open(sd_journal **ret, int flags);

int sd_journal_open_namespace(sd_journal **ret, const char *namespace, int flags);

int sd_journal_open_directory(sd_journal **ret, const char *path, int flags);

int sd_journal_open_directory_fd(sd_journal **ret, int fd, int flags);

int sd_journal_open_files(sd_journal **ret, const char **paths, int flags);

int sd_journal_open_files_fd(sd_journal **ret, int fds[], unsigned n_fds, int flags);

void sd_journal_close(sd_journal *j);

الوصف

sd_journal_open() يفتح سجل السجل (log journal) للقراءة. سيقوم بتحديد جميع ملفات السجل تلقائيًا ودمجها تلقائيًا عند القراءة. كأول وسيط، يأخذ مؤشرًا إلى مؤشر sd_journal، والذي سيحتوي عند النجاح على كائن سياق السجل. الوسيط الثاني هو حقل الأعلام (flags)، والذي قد يتكون من الأعلام التالية مدمجة معًا: SD_JOURNAL_LOCAL_ONLY يتأكد من فتح ملفات السجل التي تم إنشاؤها محليًا فقط. SD_JOURNAL_RUNTIME_ONLY يتأكد من فتح ملفات السجل المؤقتة فقط، باستثناء تلك المخزنة على التخزين الدائم. SD_JOURNAL_SYSTEM سيؤدي إلى فتح ملفات السجل للخدمات النظامية والنواة (على عكس عمليات جلسة المستخدم). SD_JOURNAL_CURRENT_USER سيؤدي إلى فتح ملفات السجل للمستخدم الحالي. إذا لم يتم تحديد أي من SD_JOURNAL_SYSTEM أو SD_JOURNAL_CURRENT_USER، سيتم فتح جميع أنواع ملفات السجل.

sd_journal_open_namespace() مشابه لـ sd_journal_open() ولكنه يأخذ وسيطًا إضافيًا namespace يحدد مساحة اسم السجل التي سيتم العمل عليها. إذا تم تحديده كـ NULL، يكون الاستدعاء مطابقًا لـ sd_journal_open(). إذا لم يكن NULL، يتم الوصول فقط إلى البيانات من مساحة الاسم المحددة بالوسيط المحدد. يفهم هذا الاستدعاء علامتين إضافيتين: إذا تم تحديد SD_JOURNAL_ALL_NAMESPACES، يتم تجاهل الوسيط namespace ويتم الوصول إلى جميع مساحات الأسماء المحددة في وقت واحد؛ إذا تم تحديد SD_JOURNAL_INCLUDE_DEFAULT_NAMESPACE، يتم الوصول إلى مساحة الاسم المحددة ومساحة الاسم الافتراضية ولكن لا شيء آخر (هذه العلامة لا يكون لها تأثير عندما يتم تمرير namespace كـ NULL). للحصول على تفاصيل حول مساحات أسماء السجل، انظر systemd-journald.service(8).

sd_journal_open_directory() مشابه لـ sd_journal_open() ولكنه يأخذ مسار دليل مطلق كوسيط. سيتم فتح جميع ملفات السجل في هذا الدليل ودمجها تلقائيًا. يأخذ هذا الاستدعاء أيضًا وسيط أعلام. الأعلام المقبولة في هذا الاستدعاء هي SD_JOURNAL_OS_ROOT، SD_JOURNAL_SYSTEM، و SD_JOURNAL_CURRENT_USER. إذا تم تحديد SD_JOURNAL_OS_ROOT، يتم البحث عن ملفات السجل أسفل /var/log/journal و /run/log/journal العادية بالنسبة للمسار المحدد، بدلاً من أسفلها مباشرة. العلامتان الأخريان تحددان الملفات التي سيتم فتحها، بنفس الطريقة كما في sd_journal_open().

sd_journal_open_directory_fd() مشابه لـ sd_journal_open_directory()، ولكنه يأخذ واصف ملف يشير إلى دليل في نظام الملفات بدلاً من مسار نظام ملفات مطلق. بالإضافة إلى الأعلام التي تقبلها sd_journal_open_directory()، تقبل هذه الدالة أيضًا SD_JOURNAL_TAKE_DIRECTORY_FD. إذا تم تحديد SD_JOURNAL_TAKE_DIRECTORY_FD، ستأخذ الدالة ملكية واصف الملف المحدد عند النجاح، وسيتم إغلاقه بواسطة sd_journal_close()، وبالتالي يجب على مستدعي الدالة عدم إغلاق واصف الملف. عندما لا يتم تحديد العلامة، لن تقوم sd_journal_close() بإغلاق واصف الملف، لذلك يجب على المستدعي إغلاقه بعد sd_journal_close().

sd_journal_open_files() مشابه لـ sd_journal_open() ولكنه يأخذ قائمة من مسارات الملفات المنتهية بـ NULL لفتحها. سيتم فتح جميع الملفات ودمجها تلقائيًا. يأخذ هذا الاستدعاء أيضًا وسيط أعلام، ولكنه يجب تمريره كـ 0 حيث لا توجد أعلام مفهومة حاليا لهذا الاستدعاء. يرجى ملاحظة أنه في حالة السجل المباشر، هذه الدالة مفيدة فقط لأغراض التصحيح، لأن ملفات السجل الفردية يمكن تدويرها في أي لحظة، وفتح ملفات معينة يعد سباقيًا بطبيعته.

sd_journal_open_files_fd() مشابه لـ sd_journal_open_files() ولكنه يأخذ مصفوفة من واصفات الملفات المفتوحة التي يجب أن تشير إلى ملفات السجل، بدلاً من مصفوفة مسارات نظام الملفات. مرر مصفوفة واصفات الملفات كوسيط ثانٍ، وعدد عناصر المصفوفة في الثالث. يجب تمرير وسيط الأعلام كـ 0.

لا يمكن استخدام كائنات sd_journal في الابن بعد عملية fork. الوظائف التي تأخذ كائن سجل كوسيط (sd_journal_next() وغيرها) ستعيد -ECHILD بعد عملية fork.

sd_journal_close() سيغلق سياق السجل المخصص باستخدام sd_journal_open() أو sd_journal_open_directory() ويحرر موارده.

عند فتح السجل، سيتم فتح ملفات السجل التي يمكن للمستخدم المتصل الوصول إليها فقط. إذا لم تكن ملفات السجل متاحة للمتصل، فسيتم تجاهلها بصمت.

انظر sd_journal_next(3) للحصول على مثال حول كيفية التكرار عبر السجل بعد فتحه باستخدام sd_journal_open().

يشير كائن سياق السجل الذي تم إرجاعه بواسطة sd_journal_open() إلى إدخال سجل محدد كإدخال حالي، مشابه لمؤشر ملف في ملف نظام ملفات كلاسيكي، ولكن بدون مواضع مطلقة. يمكن تعديله باستخدام sd_journal_next(3) و sd_journal_seek_head(3) والمكالمات ذات الصلة. يمكن تصدير موضع الإدخال الحالي في سلاسل cursor، كما هو متاح عبر sd_journal_get_cursor(3). يمكن استخدام سلاسل المؤشر لتحديد إدخال سجل محدد عالميًا بطريقة مستقرة ثم لاحقًا للانتقال إليه (أو إذا لم يكن الإدخال المحدد متاحًا محليًا، إلى أقرب إدخال له في الوقت) sd_journal_seek_cursor(3).

يتوفر الإخطار بتغييرات السجل عبر sd_journal_get_fd() والمكالمات ذات الصلة.

قيمة الإرجاع

تعيد استدعاءات sd_journal_open() و sd_journal_open_directory() و sd_journal_open_files() القيمة 0 عند النجاح أو رمز خطأ سالب بنمط errno. sd_journal_close() لا تعيد شيئًا.

ملاحظات

الاسم

sd_journal_all_namespaces - الحصول على قائمة بجميع مساحات الأسماء النشطة في سجل النظام

ملخص

#include <systemd/sd-journal.h>

int sd_journal_all_namespaces(char ***ret_namespaces);

الوصف

تقوم الدالة sd_journal_all_namespaces() بإرجاع مصفوفة من السلاسل النصية تحتوي على أسماء جميع مساحات الأسماء النشطة في سجل النظام. مساحات الأسماء هي آليات لتقسيم السجلات إلى مجموعات منطقية.

يتم تخصيص الذاكرة للمصفوفة التي سيتم إرجاعها في *ret_namespaces. يجب على المستدعي تحرير هذه الذاكرة باستخدام free() عندما لا تكون هناك حاجة إليها.

إذا لم تكن هناك مساحات أسماء نشطة، فسيتم إرجاع مصفوفة فارغة (أي NULL في *ret_namespaces و 0 في حجم المصفوفة). في حالة حدوث خطأ، سيتم إرجاع قيمة سالبة.

قيمة الإرجاع

عند النجاح، تُرجع الدالة sd_journal_all_namespaces() عدد مساحات الأسماء التي تم العثور عليها. في حالة الخطأ، تُرجع قيمة سالبة.

ملاحظات

يجب استدعاء هذه الدالة بعد فتح اتصال بالسجل باستخدام sd_journal_open().
