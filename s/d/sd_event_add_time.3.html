sd_event_add_time(3)

الحزمة: systemd-libs

التراخيص: LGPL-2.1-or-later, CC0-1.0, GPL-2.0-or-later WITH Linux-syscall-note

NAME

sd_event_add_time, sd_event_source_get_time, sd_event_source_set_time, sd_event_source_get_time_accuracy, sd_event_source_set_time_accuracy, sd_event_source_get_time_clock, sd_event_add_time_relative, sd_event_source_set_time_relative - إضافة مؤقت أحداث أو تعديله

SYNOPSIS

#include <systemd/sd-event.h>

sd_event_source_t* sd_event_add_time(sd_event *loop, uint64_t usec, sd_event_handler_t handler, void *userdata, sd_event_source **source);
sd_event_source_t* sd_event_add_time_relative(sd_event *loop, uint64_t usec, sd_event_handler_t handler, void *userdata, sd_event_source **source);
int sd_event_source_set_time(sd_event_source *source, uint64_t usec);
int sd_event_source_set_time_relative(sd_event_source *source, uint64_t usec);
int sd_event_source_get_time(sd_event_source *source, uint64_t *usec);
int sd_event_source_set_time_accuracy(sd_event_source *source, uint64_t accuracy_usec);
int sd_event_source_get_time_accuracy(sd_event_source *source, uint64_t *accuracy_usec);
int sd_event_source_get_time_clock(sd_event_source *source, clockid_t *clock_id);

DESCRIPTION

sd_event_add_time() يضيف مؤقتًا إلى حلقة الأحداث loop. عند انقضاء الوقت المحدد usec (باليوثانية) بدءًا من الوقت الحالي، يتم استدعاء الدالة handler. يتم تمرير userdata كوسيطة إلى الدالة handler.

sd_event_add_time_relative() يعمل بنفس الطريقة، ولكن usec هو وقت نسبي بدلاً من وقت مطلق. يتم تشغيل المؤقت بعد usec يوثانية من الآن.

sd_event_source_set_time() و sd_event_source_set_time_relative() يضبطان وقت تشغيل مؤقت أحداث موجود. usec هو الوقت المطلق أو النسبي، كما هو موضح أعلاه.

sd_event_source_get_time() يسترجع الوقت المحدد لمؤقت أحداث تم إنشاؤه مسبقًا باستخدام sd_event_add_time(). يأخذ كائن مصدر الحدث ومؤشرًا إلى متغير لتخزين الوقت فيه.

sd_event_source_set_time_accuracy() يضبط دقة المؤقت. يتم استخدام accuracy_usec (باليوثانية) لطلب أن يتم تنشيط المؤقت في غضون هذه الفترة الزمنية.

sd_event_source_get_time_accuracy() يسترجع دقة المؤقت التي تم ضبطها مسبقًا باستخدام sd_event_source_set_time_accuracy(). يأخذ كائن مصدر الحدث ومؤشرًا إلى متغير لتخزين الدقة فيه.

sd_event_source_get_time_clock() يسترجع الساعة المكوّنة لمصدر الحدث الذي تم إنشاؤه مسبقًا باستخدام sd_event_add_time(). يأخذ كائن مصدر الحدث ومؤشرًا إلى متغير لتخزين معرف الساعة فيه.

RETURN VALUE

عند النجاح، تعيد هذه الدوال 0 أو عددًا صحيحًا موجبًا. عند الفشل، تعيد رمز خطأ سالبًا بأسلوب errno.

Errors

قد تشير القيم المعادة إلى المشاكل التالية:

-ENOMEM

لا توجد ذاكرة كافية لتخصيص كائن.

-EINVAL

تم تمرير وسيطة غير صالحة.

-ESTALE

تم إنهاء حلقة الأحداث بالفعل.

-ECHILD

تم إنشاء حلقة الأحداث في عملية أو مكتبة أو وحدة مختلفة.

-EOPNOTSUPP

الساعة المحددة غير مدعومة من قبل تنفيذ حلقة الأحداث.

-EDOM

مصدر الحدث الممرر ليس مصدر حدث مؤقت.

-EOVERFLOW

الوقت النسبي الممرر خارج النطاق المسموح به لقيم الوقت (أي أن القيمة المحددة مضافة إلى الوقت الحالي تقع خارج نطاق الأعداد الصحيحة غير الموقعة ذات 64 بت).

تمت الإضافة في الإصدار 247.

NOTES

الوظائف الموصوفة هنا متاحة كمكتبة مشتركة، والتي يمكن تجميعها والربط بها باستخدام ملف pkg-config(1) الخاص بـ libsystemd.

الكود الموصوف هنا يستخدم getenv(3)، والذي تم الإعلان عنه بأنه غير آمن للاستخدام المتعدد الخيوط. هذا يعني أن الكود الذي يستدعي الوظائف الموصوفة هنا يجب ألا يستدعي setenv(3) من خيط متوازٍ. يوصى بإجراء استدعاءات لـ setenv() فقط في مرحلة مبكرة من البرنامج عندما لم يتم بدء أي خيوط أخرى.
