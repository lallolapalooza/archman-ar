الوصف

sd-id128.h جزء من libsystemd(3) ويوفر واجهات برمجة التطبيقات لإنشاء وتحويل ومقارنة معرفات 128 بت. معرفات 128 بت التي تعالجها وتنتجها واجهات برمجة التطبيقات هذه هي تعميم لمعرفات OSF UUIDs كما هو محدد في RFC 4122[1] ولكنها تستخدم تنسيق سلسلة أبسط. لا تفرض هذه الدوال أي بنية على المعرفات المستخدمة، على عكس معرفات OSF UUIDs أو Microsoft GUIDs، ولكنها متوافقة إلى حد كبير مع هذه الأنواع من المعرفات.

يتم تنفيذ المعرف 128 بت على أنه نوع اتحاد (union) بالشكل التالي:

typedef union sd_id128 {
  uint8_t bytes[16];
  uint64_t qwords[2];
} sd_id128_t;

يسمح نوع الاتحاد هذا بالوصول إلى المعرف 128 بت كـ 16 بايت منفصلة أو كلمتين 64 بت. بشكل عام، من الأكثر أمانًا الوصول إلى مكونات المعرف عن طريق مصفوفة 8 بت لتجنب مشاكل ترتيب البايت (endianness). يُقصد بهذا الاتحاد تمريره بالقيمة (على عكس التمرير بالإشارة) ويمكن للعملاء معالجته مباشرة.

تم تعريف بعض الماكروats لتمثيل وفك ترميز معرفات 128 بت:

SD_ID128_MAKE() يُستخدم لكتابة معرف ثابت في كود المصدر. الاستخدام الشائع هو تعيين اسم لمعرف باستخدام هذا الماكرو:

#define SD_MESSAGE_COREDUMP SD_ID128_MAKE(fc,2e,22,bc,6e,e6,47,b6,b9,07,29,ab,34,a2,50,b1)

SD_ID128_NULL يعرّف معرفًا يتكون فقط من بايتات NUL (أي كل البتات مطفأة).

SD_ID128_ALLF يعرّف معرفًا يتكون فقط من بايتات 0xFF (أي كل البتات مضاءة).

SD_ID128_MAKE_STR() مشابه لـ SD_ID128_MAKE()، ولكنه ينشئ تعبير const char* يمكن استخدامه بسهولة في تنسيقات الرسائل وما شابه:

#include <stdio.h>
#define SD_MESSAGE_COREDUMP_STR SD_ID128_MAKE_STR(fc,2e,22,bc,6e,e6,47,b6,b9,07,29,ab,34,a2,50,b1)
int main(int argc, char **argv) {
  puts("Match for coredumps: MESSAGE_ID=" SD_MESSAGE_COREDUMP_STR);
}

SD_ID128_CONST_STR() يحول المعرفات الثابتة إلى سلاسل ثابتة للإخراج. سيعرض الكود المثال التالي السلسلة "fc2e22bc6ee647b6b90729ab34a250b1":

int main(int argc, char *argv[]) {
  puts("Match for coredumps: %s", SD_ID128_CONST_STR(SD_MESSAGE_COREDUMP));
}

SD_ID128_FORMAT_STR و SD_ID128_FORMAT_VAL() تُستخدم لتنسيق معرف في سلسلة تنسيق printf(3)، كما هو موضح في المثال التالي:

int main(int argc, char *argv[]) {
  sd_id128_t id;
  id = SD_ID128_MAKE(ee,89,be,71,bd,6e,43,d6,91,e6,c5,5d,eb,03,02,07);
  printf("The ID encoded in this C file is " SD_ID128_FORMAT_STR ".\n", SD_ID128_FORMAT_VAL(id));
  return 0;
}

SD_ID128_UUID_FORMAT_STR و SD_ID128_MAKE_UUID_STR() مشابهان لـ SD_ID128_FORMAT_STR و SD_ID128_MAKE_STR()، ولكنهما يتضمنان شرطات فاصلة للامتثال لـ "تمثيل UUID الأساسي[2]". يقومان بتنسيق السلسلة بناءً على قواعد RFC4122[1] Variant 1، أي التحويل من ترتيب البايت Big Endian. هذا يطابق سلوك معظم البنية التحتية الأخرى لنظام Linux. من الأفضل تجنب UUIDs من متغيرات أخرى، لتجنب الغموض غير الضروري. جميع المعرفات 128 بت التي تنتجها واجهات برمجة تطبيقات sd-id128 تتوافق تمامًا مع Variant 1 Version 4 UUIDs، وفقًا لـ RFC 4122.

sd_id128_equal() تقارن بين معرفين 128 بت:

int main(int argc, char *argv[]) {
  sd_id128_t a, b, c;
  a = SD_ID128_MAKE(ee,89,be,71,bd,6e,43,d6,91,e6,c5,5d,eb,03,02,07);
  b = SD_ID128_MAKE(f2,28,88,9c,5f,09,44,15,9d,d7,04,77,58,cb,e7,3e);
  c = a;
  assert(sd_id128_equal(a, c));
  assert(!sd_id128_equal(a, b));
  return 0;
}

sd_id128_string_equal() مشابه لـ sd_id128_equal()، ولكن المعرف الأول يتم تنسيقه كسلسلة const char*. تنطبق نفس القيود على الوسيط الأول لـ sd_id128_from_string().

sd_id128_is_null() تتحقق مما إذا كان المعرف يتكون فقط من بايتات NUL:

assert(sd_id128_is_null(SD_ID128_NULL));

بشكل مشابه، sd_id128_is_allf() تتحقق مما إذا كان المعرف يتكون فقط من بايتات 0xFF (كل البتات مضاءة):

assert(sd_id128_is_allf(SD_ID128_ALLF));

sd_id128_in_set_sentinel() تأخذ قائمة بمعرفات وتعيد صحيحًا إذا كان الوسيط الأول مساويًا لأي من الوسائط اللاحقة. قائمة الوسائط تنتهي بـ SD_ID128_NULL كعلامة نهاية، والتي يجب أن تكون موجودة.

sd_id128_in_set() هي دالة مساعدة تأخذ قائمة بمعرفات وتعيد صحيحًا إذا كان الوسيط الأول مساويًا لأي من الوسائط اللاحقة:

int main(int argc, char *argv[]) {
  sd_id12_t a = SD_ID128_MAKE(ee,89,be,71,bd,6e,43,d6,91,e6,c5,5d,eb,03,02,07);
  assert(sd_id128_in_set(a, a));
  assert(sd_id128_in_set(a, a, a));
  assert(!sd_id128_in_set(a));
  assert(!sd_id128_in_set(a,
                          SD_ID128_MAKE(f2,28,88,9c,5f,09,44,15,9d,d7,04,77,58,cb,e7,3e)
                          SD_ID128_MAKE(2f,88,28,5f,9c,44,09,9d,d7,15,77,04,bc,85,7e,e3)
                          SD_ID128_ALLF));
  return 0;
}

sd_id128_in_set() معرفة كـ ماكرو فوق sd_id128_in_set_sentinel()، وتضيف علامة النهاية SD_ID128_NULL تلقائيًا. بما أن sd_id128_in_set_sentinel() تستخدم SD_ID128_NULL كعلامة نهاية، فلا يمكن وضع SD_ID128_NULL بشكل آخر في قائمة الوسائط.

sd_id128_in_setv() مشابه لـ sd_id128_in_set_sentinel()، ولكنه يأخذ وسيط varargs (قائمة وسائط متغيرة).

يمكن إنشاء معرفات عشوائية جديدة باستخدام أمر new من systemd-id128(1).
