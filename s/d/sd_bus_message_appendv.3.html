sd_bus_message_appendv(3)

الحزمة: systemd-libs

التراخيص: LGPL-2.1-or-later, CC0-1.0, GPL-2.0-or-later WITH Linux-syscall-note

SD_BUS_MESSAGE_APPEND(3) sd_bus_message_append SD_BUS_MESSAGE_APPEND(3)

الاسم

sd_bus_message_append, sd_bus_message_appendv - إلحاق حقول برسالة D-Bus بناءً على سلسلة أنواع

المُلخَّص

#include <systemd/sd-bus.h>

int sd_bus_message_append(sd_bus_message *m, const char *types, ...);

int sd_bus_message_appendv(sd_bus_message *m, const char *types, va_list ap);

الوصف

تقوم الدالة sd_bus_message_append() بإلحاق سلسلة من الحقول بكائن رسالة D-Bus m. تصف سلسلة الأنواع types أنواع وسائط الحقول التي تليها. لكل نوع محدد في سلسلة الأنواع، يجب تحديد وسيط واحد أو أكثر، بنفس الترتيب المعلن في سلسلة الأنواع.

تتكون سلسلة الأنواع من العناصر المعروضة في الجدول أدناه. تحتوي على صفر أو أكثر من "الأنواع الكاملة" المفردة. قد يكون كل نوع كامل أحد الأنواع الأساسية أو نوع حاوية موصوف بالكامل. قد يكون نوع الحاوية بنية بالأنواع المحتواة، أو متغيرًا، أو مصفوفة بنوع عنصرها، أو إدخال قاموس بالأنواع المحتواة. تنتهي سلسلة الأنواع بـ NUL.

في حالة النوع الأساسي، يُتوقع وسيط من النوع المقابل.

يتم تمييز البنية بتسلسل من الأنواع الكاملة بين "(" و ")". لا يمكن أن يكون هذا التسلسل فارغًا — يجب أن يحتوي على نوع واحد على الأقل. الوسائط المقابلة لهذا التسلسل المتداخل تتبع نفس القواعد كما لو لم تكن متداخلة.

يتم تمييز المتغير بـ "v". يجب أن تبدأ الوسائط المقابلة بسلسلة أنواع تصف نوعًا كاملاً، وبعد ذلك، الوسائط المقابلة للنوع المحدد.

يتم تمييز المصفوفة بـ "a" متبوعة بنوع كامل. يجب أن تبدأ الوسائط المقابلة بعدد الإدخالات في المصفوفة، متبوعة بالإدخالات نفسها، مطابقة لنوع عنصر المصفوفة.

القاموس هو مصفوفة من إدخالات القاموس، يتم تمييزها بـ "a" متبوعة بزوج من الأنواع الكاملة بين "{" و "}". يجب أن يكون الأول من هذه الأنواع نوعًا أساسيًا. يجب أن تبدأ الوسائط المقابلة بعدد إدخالات القاموس، متبوعة بزوج من القيم لكل إدخال مطابق لنوع عنصر إدخالات القاموس.

sd_bus_message_appendv() مكافئة لـ sd_bus_message_append()، باستثناء أنها تُستدعى باستخدام "va_list" بدلاً من عدد متغير من الوسائط. هذه الدالة لا تستدعي ماكرو va_end(). نظرًا لأنها تستدعي ماكرو va_arg()، فإن قيمة ap غير محددة بعد المكالمة.

للمزيد من التفاصيل حول نظام أنواع D-Bus، يرجى استشارة D-Bus Specification[1].

"v" SD_BUS_TYPE_VARIANT متغير يحددها الوسيط النوع سلسلة توقيع، متبوعة بمحتويات المتغير

"{" SD_BUS_TYPE_DICT_ENTRY_BEGIN بداية إدخال قاموس يحددها الأنواع المتداخلة محتويات القاموس
"}" SD_BUS_TYPE_DICT_ENTRY_END نهاية إدخال قاموس

بالنسبة للأنواع
"s" و "g" (سلسلة يونيكود أو توقيع)، قد يكون المؤشر NULL، وهو مكافئ لسلسلة فارغة. بالنسبة لـ "h" (واصف ملف UNIX)، يتم تكرار الواصف بواسطة هذه المكالمة ويظل الواصف الذي تم تمريره في حيازة المتصل. راجع sd_bus_message_append_basic(3) للتفسير الدقيق لتلك الأنواع وغيرها.

قواعد بناء سلسلة الأنواع

types ::= complete_type*
complete_type ::= basic_type | variant | structure | array | dictionary
basic_type ::= "y" | "n" | "q" | "u" | "i" | "x" | "t" | "d" |
               "b" | "h" |
               "s" | "o" | "g"
variant ::= "v"
structure ::= "(" complete_type+ ")"
array ::= "a" complete_type
dictionary ::= "a" "{" basic_type complete_type "}"

أمثلة

إلحاق نوع أساسي واحد (السلسلة "a string"):

sd_bus_message *m;
...
sd_bus_message_append(m, "s", "a string");

إلحاق جميع أنواع الأعداد الصحيحة:

uint8_t y = 1;
int16_t n = 2;
uint16_t q = 3;
int32_t i = 4;
uint32_t u = 5;
int32_t x = 6;
uint32_t t = 7;
double d = 8.0;
sd_bus_message_append(m, "ynqiuxtd", y, n, q, i, u, x, t, d);

إلحاق بنية تتكون من سلسلة ومسار D-Bus:

sd_bus_message_append(m, "(so)", "a string", "/a/path");

إلحاق مصفوفة من واصفات ملفات UNIX:

sd_bus_message_append(m, "ah", 3, STDIN_FILENO, STDOUT_FILENO, STDERR_FILENO);

إلحاق متغير، مع النوع الحقيقي "g" (توقيع)، والقيمة "sdbusisgood":

sd_bus_message_append(m, "v", "g", "sdbusisgood");

إلحاق قاموس يحتوي على الربط {1=>"a", 2=>"b", 3=>""}:

sd_bus_message_append(m, "a{is}", 3, 1, "a", 2, "b", 3, NULL);

قيمة الإرجاع

عند النجاح، تعيد هذه الدوال عددًا صحيحًا غير سالب. عند الفشل، فإنها تعيد رمز خطأ سالبًا على نمط errno.

هذا الموقع غير مرتبط بـ Arch Linux ولا تابع له بأي شكل من الأشكال. المحتوى هو ترجمة فقط لصفحات دليل Arch المنشورة على man.archlinux.org.

lass="Ss" id='Errors'>أخطاء

قد تشير الأخطاء المُرجعة إلى المشاكل التالية:

-EINVAL

المعلمة المحددة غير صالحة.

-EPERM

تم ختم الرسالة.

-ESTALE

الرسالة في حالة غير صالحة.

-ENXIO

لا يمكن إلحاق الرسالة.

-ENOMEM

فشل تخصيص الذاكرة.

ملاحظات

الوظائف الموضحة هنا متاحة كمكتبة مشتركة، والتي يمكن تجميعها وربطها بملف libsystemd pkg-config(1).

يستخدم الكود الموضح هنا getenv(3)، والذي تم تعريفه بأنه غير آمن للاستخدام في خيوط متعددة. هذا يعني أن الكود الذي يستدعي الوظائف الموضحة هنا يجب ألا يستدعي setenv(3) من خيط موازٍ. يوصى بإجراء استدعاءات لـ setenv() فقط من مرحلة مبكرة من البرنامج عندما لم يتم بدء خيوط أخرى.
