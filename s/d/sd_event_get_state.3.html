#include <systemd/sd-event.h>
enum {

};

int sd_event_get_iteration(sd_event *event, uint64_t *ret);

الوصف

يمكن استخدام الدوال منخفضة المستوى sd_event_prepare() و sd_event_wait() و sd_event_dispatch() لتنفيذ مراحل محددة من حلقة الأحداث. راجع sd_event_run(3) و sd_event_loop(3) للدوال عالية المستوى التي تنفذ تكرارات فردية ولكن كاملة لحلقة الأحداث أو تشغيلها بشكل مستمر.

تقوم sd_event_prepare() بالتحقق من الأحداث المعلقة وتجهيز المؤقتات الضرورية. إذا كانت أي أحداث جاهزة للمعالجة ("معلقة")، فإنها تعيد قيمة موجبة غير صفرية، ويجب على المتصل معالجة هذه الأحداث باستخدام sd_event_dispatch().

تقوم sd_event_dispatch() بتوزيع مصدر الحدث ذي الأولوية القصوى الذي لديه حدث معلق. عند النجاح، تعيد sd_event_dispatch() إما صفرًا، مما يشير إلى أنه لا يمكن توزيع أي مصادر أحداث أخرى وأنه تم طلب إنهاء حلقة الأحداث عبر sd_event_exit(3)؛ أو قيمة موجبة غير صفرية، مما يعني أنه تم توزيع مصدر حدث وعادت الحلقة إلى حالتها الأولية، ويجب على المتصل بدء تكرار حلقة الأحداث التالي عن طريق استدعاء sd_event_prepare() مرة أخرى.

في حالة عودة sd_event_prepare() بصفر، يجب استدعاء sd_event_wait() للانتظار لأحداث أخرى أو مهلة. إذا كانت أي أحداث جاهزة للمعالجة، فإنها تعيد قيمة موجبة غير صفرية، ويجب توزيع الأحداث باستخدام sd_event_dispatch(). بخلاف ذلك، عادت حلقة الأحداث إلى حالتها الأولية ويجب بدء تكرار حلقة الأحداث التالي عن طريق استدعاء sd_event_prepare() مرة أخرى.

يمكن استخدام sd_event_get_state() لتحديد الحالة التي توجد بها حلقة الأحداث حاليًا. إنها تعيد إحدى الحالات الموضحة أدناه.

يمكن استخدام sd_event_get_iteration() لتحديد التكرار الحالي لحلقة الأحداث. تعيد عددًا صحيحًا غير موقع بعرض 64 بت يحتوي على عداد يزداد بشكل رتيب مع كل تكرار لحلقة الأحداث، بدءًا من 0. يتم زيادة العداد في وقت استدعاء sd_event_prepare().

جميع الدوال الخمس تأخذ، كوسيط أول، كائن حلقة الأحداث event الذي تم إنشاؤه باستخدام sd_event_new(). يتم تحديد المهلة لـ sd_event_wait() في usec بالميكروثانية. يمكن استخدام (uint64_t) -1 لتحديد مهلة لا نهائية.

آلة الحالة

تعرف حلقة الأحداث الحالات التالية، والتي يمكن الاستعلام عنها باستخدام sd_event_get_state().

SD_EVENT_INITIAL

الحالة الأولية التي تكون عليها حلقة الأحداث، قبل كل تكرار لحلقة الأحداث. استخدم sd_event_prepare() للانتقال بحلقة الأحداث إلى حالات SD_EVENT_ARMED أو SD_EVENT_PENDING.

تمت الإضافة في الإصدار 229.

SD_EVENT_PREPARING

يجري حاليًا إعداد مصدر حدث، أي أنه يتم حاليًا تنفيذ معالج الإعداد، كما تم تعيينه باستخدام sd_event_source_set_prepare(3). هذه الحالة تُرى فقط في معالج إعداد مصدر الحدث الذي يتم استدعاؤه من استدعاء sd_event_prepare() ويتبعه مباشرة SD_EVENT_ARMED أو SD_EVENT_PENDING.

تمت الإضافة في الإصدار 229.

SD_EVENT_ARMED

تم استدعاء sd_event_prepare() ولم تكن هناك مصادر أحداث جاهزة للتوزيع. استخدم sd_event_wait() للانتظار لأحداث جديدة، والانتقال إلى SD_EVENT_PENDING أو العودة إلى SD_EVENT_INITIAL.

تمت الإضافة في الإصدار 229.

SD_EVENT_PENDING

تم استدعاء sd_event_prepare() أو sd_event_wait() وكانت هناك مصادر أحداث مع أحداث معلقة. استخدم sd_event_dispatch() لتوزيع مصدر الحدث ذي الأولوية القصوى والانتقال إلى SD_EVENT_INITIAL أو SD_EVENT_FINISHED.

تمت الإضافة في الإصدار 229.

SD_EVENT_RUNNING

يجري حاليًا توزيع مصدر حدث عادي. تُرى هذه الحالة فقط في معالج مصدر الحدث الذي يتم استدعاؤه من استدعاء sd_event_dispatch()، ويتبعه مباشرة SD_EVENT_INITIAL أو SD_EVENT_FINISHED بمجرد عودة معالج مصدر الحدث. لاحظ أنه أثناء توزيع مصادر أحداث الإنهاء، تُرى الحالة SD_EVENT_EXITING بدلاً من ذلك.

تمت الإضافة في الإصدار 229.

SD_EVENT_EXITING

على غرار SD_EVENT_RUNNING ولكنها الحالة السارية أثناء توزيع مصادر أحداث الإنهاء. تتبعها SD_EVENT_INITIAL أو SD_EVENT_FINISHED بمجرد عودة معالج الحدث.

تمت الإضافة في الإصدار 229.

SD_EVENT_FINISHED

انتهت حلقة الأحداث. تم تشغيل جميع مصادر أحداث الإنهاء. إذا كانت حلقة الأحداث في هذه الحالة، فهي لا تقدم أي فائدة بعد الآن، ويجب تحريرها.

تمت الإضافة في الإصدار 229.

يوضح المخطط الانسيابي المبسط للحالات والاستدعاءات للانتقال بينها أدناه. لاحظ أن SD_EVENT_PREPARING و SD_EVENT_RUNNING و SD_EVENT_EXITING غير معروضة هنا.

          INITIAL <---<---<---<---<---<---<---<---<---<---<---\
             |                                                     |
             |                                                     ^
             |                                                     |
             v                 ret == 0                            |
      sd_event_prepare() >--->--->--->---> ARMED                  |
             |                                |                    ^
             | ret > 0                        |                    |
             |                                |                    |
             v                                v          ret == 0  |
          PENDING <---<---<---<---< sd_event_wait() >--->--->--+
             |           ret > 0                                   ^
             |                                                     |
             |                                                     |
             v                                                     |
      sd_event_dispatch() >--->--->--->--->--->--->--->--->--->--->/
             |                             ret > 0
             | ret == 0
             |
             v
          FINISHED

قيمة الإرجاع

عند النجاح، تعيد هذه الدوال 0 أو عددًا صحيحًا موجبًا. عند الفشل، تعيد رمز خطأ سالبًا على غرار errno. في حالة sd_event_prepare() و sd_event_wait()، تشير رمز إرجاع موجب وغير صفري إلى أن الأحداث جاهزة للمعالجة وصفر يشير إلى عدم جاهزية أي أحداث. في حالة sd_event_dispatch()، تشير قيمة إرجاع موجبة غير صفرية إلى أن حلقة الأحداث عادت إلى حالتها الأولية وصفر يشير إلى أن حلقة الأحداث قد انتهت. تعيد sd_event_get_state() حالة موجبة أو صفرية عند النجاح.

الأخطاء

قد تشير الأخطاء المعادة إلى المشاكل التالية:

-EINVAL

الوسيط event غير صالح أو NULL.

-EBUSY

كائن حلقة الأحداث ليس في الحالة الصحيحة.

-ESTALE

تم إنهاء حلقة الأحداث بالفعل.

ERRORES

-ECHILD:: تم إنشاء حلقة الأحداث في عملية مختلفة، أو مكتبة، أو مثيل وحدة نمطية. -ECHILD:: The event loop has been created in a different process, library or module instance.

قد تحدث أخطاء أخرى أيضاً.
