sd_bus_message_get_expect_reply(3)

الحزمة: systemd-libs

التراخيص: LGPL-2.1-or-later, CC0-1.0, GPL-2.0-or-later WITH Linux-syscall-note

SD_BUS_MESSAGE_SET_EXPECT_REPLY(3) sd_bus_message_set_expect_reply SD_BUS_MESSAGE_SET_EXPECT_REPLY(3)

الاسم

sd_bus_message_set_expect_reply, sd_bus_message_get_expect_reply, sd_bus_message_set_auto_start, sd_bus_message_get_auto_start, sd_bus_message_set_allow_interactive_authorization, sd_bus_message_get_allow_interactive_authorization - تعيين الاستعلام عن البيانات الوصفية لرسالة الناقل

ملخص

#include <systemd/sd-bus.h>

int sd_bus_message_set_expect_reply(sd_bus_message *message, int b);

int sd_bus_message_get_expect_reply(sd_bus_message *message);

int sd_bus_message_set_auto_start(sd_bus_message *message, int b);

int sd_bus_message_get_auto_start(sd_bus_message *message);

int sd_bus_message_set_allow_interactive_authorization(sd_bus_message *message, int b);

int sd_bus_message_get_allow_interactive_authorization(sd_bus_message *message);

الوصف

sd_bus_message_set_expect_reply() تقوم بتعيين أو مسح علامة NO_REPLY_EXPECTED على الرسالة m. هذه العلامة مهمة فقط لرسائل استدعاء الطرق وتستخدم لتحديد أنه لا يُتوقع رد طريق أو رد خطأ. يتم تجاهلها لأنواع أخرى. بالتالي، لرسالة استدعاء طريقة، فإن استدعاء

sd_bus_message_set_expect_reply(..., 0)

يقوم بتعيين العلامة وكتم الرد.

sd_bus_message_get_expect_reply() تتحقق مما إذا كانت علامة NO_REPLY_EXPECTED معين على الرسالة m. ستعيد قيمة موجبة إذا لم تكن معين، وصفر إذا كانت كذلك.

sd_bus_message_set_auto_start() تقوم بتعيين أو مسح علامة NO_AUTO_START على الرسالة m. عندما تكون العلامة معين، يجب على الناقل عدم تشغيل مالك لاسم الوجهة استجابة لهذه الرسالة. استدعاء

sd_bus_message_set_auto_start(..., 0)

يقوم بتعيين العلامة.

sd_bus_message_get_auto_start() تتحقق مما إذا كانت علامة NO_AUTO_START معين على الرسالة m. ستعيد قيمة موجبة إذا لم تكن معين، وصفر إذا كانت كذلك.

sd_bus_message_set_allow_interactive_authorization() تقوم بتعيين أو مسح علامة ALLOW_INTERACTIVE_AUTHORIZATION على الرسالة m. تعيين هذه العلامة يبلغ المستلم بأن المتصل مستعد للانتظار للمصادقة التفاعلية عبر polkit أو إطار عمل مشابه. لاحظ أن تعيين هذه العلامة لا يضمن أن المستلم سيقوم فعليًا بالمصادقة التفاعلية. أيضًا، تأكد من تعيين مهلة رسالة مناسبة عند استخدام هذه العلامة نظرًا لأن المصادقة التفاعلية قد تستغرق وقتًا طويلاً لأنها تعتمد على إدخال المستخدم. إذا كانت b غير صفرية، يتم تعيين العلامة.

sd_bus_message_get_allow_interactive_authorization() تتحقق مما إذا كانت علامة ALLOW_INTERACTIVE_AUTHORIZATION معين على الرسالة m. ستعيد عددًا صحيحًا موجبًا إذا كانت العلامة معين. بخلاف ذلك، ستعيد صفرًا.

قيمة الإرجاع

عند النجاح، تعيد هذه الدوال عددًا صحيحًا غير سالب. عند الفشل، تعيد رمز خطأ سالبًا بنمط errno.

الأخطاء

قد تشير الأخطاء المرتجعة إلى المشاكل التالية:

-EINVAL

المعلمة message هي NULL.

-EPERM

الرسالة message مختومة عند محاولة تعيين علامة.

الرسالة message ذات نوع خاطئ.

ملاحظات

الدوال الموصوفة هنا متاحة كمكتبة مشتركة، والتي يمكن تجميعها ضدها وربطها باستخدام ملف pkg-config(1) الخاص بـ libsystemd.

الكود الموصوف هنا يستخدم getenv(3)، والذي يُعلن أنه ليس آمنًا للاستخدام في خيوط متعددة. هذا يعني أن الكود الذي يستدعي الدوال الموصوفة هنا يجب ألا يستدعي setenv(3) من خيط متوازي. يوصى بالقيام باستدعاءات لـ setenv() فقط من مرحلة مبكرة من البرنامج عندما لم تبدأ أي خيوط أخرى.
