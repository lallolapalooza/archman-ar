SD_JOURNAL_FOREACH(3)

الحزمة: systemd-libs

التراخيص: LGPL-2.1-or-later, CC0-1.0, GPL-2.0-or-later WITH Linux-syscall-note

SD_JOURNAL_NEXT(3) sd_journal_next SD_JOURNAL_NEXT(3)

الاسم

sd_journal_next, sd_journal_previous, sd_journal_step_one, sd_journal_next_skip, sd_journal_previous_skip, SD_JOURNAL_FOREACH, SD_JOURNAL_FOREACH_BACKWARDS - تقدم أو ترجع مؤشر القراءة في السجل

ملخص

#include <systemd/sd-journal.h>

int sd_journal_next(sd_journal *j);

int sd_journal_previous(sd_journal *j);

int sd_journal_step_one(sd_journal *j, int advanced);

int sd_journal_next_skip(sd_journal *j, uint64_t skip);

int sd_journal_previous_skip(sd_journal *j, uint64_t skip);

SD_JOURNAL_FOREACH(sd_journal *j);

SD_JOURNAL_FOREACH_BACKWARDS(sd_journal *j);

الوصف

sd_journal_next() يتقدم بمؤشر القراءة في السجل بمقدار مدخل واحد. الوسيط الوحيد الذي يأخذه هو كائن سياق السجل كما تم تخصيصه عبر sd_journal_open(3). بعد الاستدعاء الناجح، يمكن قراءة المدخل باستخدام دوال مثل sd_journal_get_data(3).

وبالمثل، sd_journal_previous() يرجع مؤشر القراءة بمقدار مدخل واحد.

sd_journal_step_one() يحرك أيضًا مؤشر القراءة. إذا كان الموقع الحالي هو رأس السجل، على سبيل المثال عند استدعاء هذه الدالة بعد sd_journal_seek_head(3)، فإنها تعادل sd_journal_next()، وسيتم تجاهل الوسيط advanced. وبالمثل، إذا كان الموقع الحالي هو ذيل السجل، على سبيل المثال عند استدعاء هذه الدالة بعد sd_journal_seek_tail(3)، فإنها تعادل sd_journal_previous()، وسيتم تجاهل advanced. خلاف ذلك، فإنها تعادل sd_journal_next() عندما تكون advanced غير صفرية، و sd_journal_previous() عندما تكون advanced صفرية.

sd_journal_next_skip() و sd_journal_previous_skip() يتقدمان/يرجعان مؤشر القراءة بمقدار مدخلات متعددة مرة واحدة، كما هو محدد في المعلمة skip. يجب أن تكون المعلمة skip أقل من أو تساوي 2147483647 (2³¹-1).

السجل مرتب بشكل صارم حسب وقت الاستقبال، وبالتالي فإن التقدم إلى المدخل التالي يضمن أن المدخل الذي يشير إليه بعد ذلك هو في وقت لاحق من السابق، أو له نفس الطابع الزمني.

لاحظ أن sd_journal_get_data(3) والدعوات ذات الصلة ستفشل ما لم يتم استدعاء sd_journal_next() مرة واحدة على الأقل لوضع مؤشر القراءة على مدخل السجل.

لاحظ أن الماكرو SD_JOURNAL_FOREACH() يمكن استخدامه كغلاف حول sd_journal_seek_head(3) و sd_journal_next() لتسهيل التكرار عبر السجل. انظر أدناه مثالاً. وبالمثل، يمكن استخدام SD_JOURNAL_FOREACH_BACKWARDS() للتكرار عبر السجل بترتيب عكسي.

قيمة الإرجاع

تعيد الدوال الأربع عدد المدخلات التي تم التقدم بها/ارجاعها عند النجاح أو رمز خطأ بسالب بنمط errno. عندما يتم الوصول إلى نهاية أو بداية السجل، يتم إرجاع رقم أصغر من المطلوب. وبشكل أكثر تحديدًا، إذا وصلت sd_journal_next() أو sd_journal_previous() إلى نهاية/بداية السجل، فسوف ترجع 0، بدلاً من 1 عند نجاحها. يجب اعتبار هذا علامة نهاية الملف (EOF).

ملاحظات

جميع الدوال المذكورة هنا مستقلة عن الخيوط (thread-agnostic) ويمكن لخيط واحد فقط العمل على كائن معين في أي وقت. قد تصل خيوط مختلفة إلى نفس الكائن في أوقات مختلفة. يمكن استخدام كائنات مستقلة متعددة من خيوط مختلفة بالتوازي.

الدوال الموصوفة هنا متاحة كمكتبة مشتركة، والتي يمكن تجميعها معها وربطها بملف libsystemd pkg-config(1).

أمثلة

التكرار عبر السجل:

/* SPDX-License-Identifier: MIT-0 */
#include <errno.h>
#include <stdio.h>
#include <systemd/sd-journal.h>
int main(int argc, char *argv[]) {
  int r;
  sd_journal *j;
  r = sd_journal_open(&j, SD_JOURNAL_LOCAL_ONLY);
  if (r < 0) {
    fprintf(stderr, "فشل فتح السجل: %s\n", strerror(-r));
    return 1;
  }
  SD_JOURNAL_FOREACH(j) {
    const char *d;
    size_t l;
    r = sd_journal_get_data(j, "MESSAGE", (const void **)&d, &l);
    if (r < 0) {
      fprintf(stderr, "فشل قراءة حقل الرسالة: %s\n", strerror(-r));
      continue;
    }
    printf("%.*s\n", (int) l, d);
  }
  sd_journal_close(j);
  return 0;
}
