DESCRIPTION

sd-id128.h هو جزء من libsystemd(3) ويوفر واجهات برمجة تطبيقات لإنشاء وتحويل ومقارنة قيم المعرفات 128 بت. قيم المعرفات 128 بت التي تعالجها وتنشئها واجهات برمجة التطبيقات هذه هي تعميم لـ UUIDs المعرفة في RFC 4122[1]، ولكنها تستخدم تنسيق سلسلة أبسط. لا تفرض هذه الدوال أي بنية على المعرفات المستخدمة، على عكس UUIDs OSF أو GUIDs Microsoft، ولكنها متوافقة بشكل عام مع هذه الأنواع من المعرفات.

يتم تنفيذ معرف 128 بت كنوع اتحاد (union) التالي:

typedef union sd_id128 {
  uint8_t bytes[16];
  uint64_t qwords[2];
} sd_id128_t;

يتيح نوع الاتحاد هذا الوصول إلى المعرف 128 بت كسلسلة من 16 بايتات منفصلة أو كلمتين 64 بت. بشكل عام، يكون أكثر أمانًا الوصول إلى مكونات المعرف من خلال مصفوفة 8 بت الخاصة بها لتجنب مشكلات ترتيب البايت (endianness). يُقصد بنوع الاتحاد هذا أن يتم تمريره بواسطة القيمة (بدلاً من المرجع) ويمكن معالجته مباشرة بواسطة العملاء.

تم تعريف عدة وحدات ماكرو لتمثيل وفك ترميز المعرفات 128 بت:

SD_ID128_MAKE() يُستخدم لكتابة معرف ثابت في الكود المصدري. لغة شائعة هي تعيين اسم لمعرف باستخدام وحدة الماكرو هذه:

#define SD_MESSAGE_COREDUMP SD_ID128_MAKE(fc,2e,22,bc,6e,e6,47,b6,b9,07,29,ab,34,a2,50,b1)

SD_ID128_NULL يحدد معرفًا يتكون فقط من بايتات NUL (أي، جميع البتات معطلة).

SD_ID128_ALLF يحدد معرفًا يتكون فقط من بايتات 0xFF (أي، جميع البتات ممكّنة).

SD_ID128_MAKE_STR() يشبه SD_ID128_MAKE()، ولكنه ينشئ تعبير const char* يمكن استخدامه بشكل ملائم في تنسيقات الرسائل:

#include <stdio.h>
#define SD_MESSAGE_COREDUMP_STR SD_ID128_MAKE_STR(fc,2e,22,bc,6e,e6,47,b6,b9,07,29,ab,34,a2,50,b1)
int main(int argc, char **argv) {
  puts("Match for coredumps: MESSAGE_ID=" SD_MESSAGE_COREDUMP_STR);
}

SD_ID128_CONST_STR() يحول المعرفات الثابتة إلى سلاسل ثابتة للإخراج. سيعرض مثال الكود التالي السلسلة "fc2e22bc6ee647b6b90729ab34a250b1":

int main(int argc, char *argv[]) {
  puts("Match for coredumps: %s", SD_ID128_CONST_STR(SD_MESSAGE_COREDUMP));
}

SD_ID128_FORMAT_STR و SD_ID128_FORMAT_VAL() يُستخدمان لتنسيق معرف في تنسيق printf(3) كما هو موضح في المثال التالي:

int main(int argc, char *argv[]) {
  sd_id128_t id;
  id = SD_ID128_MAKE(ee,89,be,71,bd,6e,43,d6,91,e6,c5,5d,eb,03,02,07);
  printf("The ID encoded in this C file is " SD_ID128_FORMAT_STR ".\n", SD_ID128_FORMAT_VAL(id));
  return 0;
}

SD_ID128_UUID_FORMAT_STR و SD_ID128_MAKE_UUID_STR() مشابهان لـ SD_ID128_FORMAT_STR و SD_ID128_MAKE_STR()، ولكنهما يتضمنان فواصل شرطية للامتثال للتمثيل القياسي لـ UUID[2]. يعتمد تنسيق السلسلة على قواعد RFC4122[1]، أي التحويل من ترتيب البايت Big Endian. يتوافق هذا السلوك مع معظم البنية التحتية لمساحة المستخدم في Linux. من المحتمل أن يكون من الأفضل تجنب UUIDs من متغيرات أخرى لتجنب الغموض غير الضروري. تتبع جميع المعرفات 128 بت التي تم إنشاؤها بواسطة واجهات برمجة تطبيقات sd-id128 بدقة المتغيرات 1 الإصدار 4 من UUIDs وفقًا لـ RFC 4122.

sd_id128_equal() تقارن معرفين 128 بت:

int main(int argc, char *argv[]) {
  sd_id128_t a, b, c;
  a = SD_ID128_MAKE(ee,89,be,71,bd,6e,43,d6,91,e6,c5,5d,eb,03,02,07);
  b = SD_ID128_MAKE(f2,28,88,9c,5f,09,44,15,9d,d7,04,77,58,cb,e7,3e);
  c = a;
  assert(sd_id128_equal(a, c));
  assert(!sd_id128_equal(a, b));
  return 0;
}

sd_id128_string_equal() مشابه لـ sd_id128_equal()، ولكن المعرف الأول يتم تشكيله كسلسلة const char*. تنطبق نفس القيود على الوسيط الأول لـ sd_id128_from_string().

sd_id128_is_null() يتحقق مما إذا كان المعرف يتكون فقط من بايتات NUL:

assert(sd_id128_is_null(SD_ID128_NULL));

بشكل مماثل، sd_id128_is_allf() يتحقق مما إذا كان المعرف يتكون فقط من بايتات 0xFF (جميع البتات ممكّنة):

assert(sd_id128_is_allf(SD_ID128_ALLF));

sd_id128_in_set_sentinel() يأخذ قائمة من المعرفات ويعيد قيمة صحيحة إذا كان الوسيط الأول يساوي أيًا من الوسائط التالية. يتم إنهاء قائمة الوسائط بواسطة محدد SD_ID128_NULL، والذي يجب أن يكون موجودًا.
