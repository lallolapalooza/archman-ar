sd_event_source_send_child_signal(3)

الحزمة: systemd-libs

التراخيص: LGPL-2.1-or-later, CC0-1.0, GPL-2.0-or-later WITH Linux-syscall-note

NAME

sd_event_source_send_child_signal, sd_event_add_child, sd_event_add_child_pidfd, sd_event_child_handler_t, sd_event_source_get_child_pid, sd_event_source_get_child_pidfd, sd_event_source_get_child_pidfd_own, sd_event_source_set_child_pidfd_own, sd_event_source_get_child_process_own, sd_event_source_set_child_process_own — وظائف لإدارة مصادر أحداث العمليات الفرعية

SYNOPSIS

#include <systemd/sd-event.h>
int sd_event_add_child(sd_event *event, sd_event_source **source, pid_t pid, uint32_t events, sd_event_child_handler_t *callback, void *userdata);
int sd_event_add_child_pidfd(sd_event *event, sd_event_source **source, int pidfd, uint32_t events, sd_event_child_handler_t *callback, void *userdata);
typedef int (*sd_event_child_handler_t)(sd_event *event, pid_t pid, int wstatus, void *userdata);
int sd_event_source_get_child_pid(sd_event_source *source, pid_t *pid);
int sd_event_source_get_child_pidfd(sd_event_source *source, int *pidfd);
int sd_event_source_get_child_pidfd_own(sd_event_source *source, bool *own);
int sd_event_source_set_child_pidfd_own(sd_event_source *source, bool own);
int sd_event_source_get_child_process_own(sd_event_source *source, bool *own);
int sd_event_source_set_child_process_own(sd_event_source *source, bool own);
int sd_event_source_send_child_signal(sd_event_source *source, pid_t pid, int sig, const siginfo_t *info, int flags);

DESCRIPTION

تتعامل هذه الدوال مع مراقبة الأحداث المتعلقة بالعمليات الفرعية. وظيفة sd_event_add_child تتيح للمستخدمين إضافة مراقب لعملية فرعية محددة بواسطة pid. عندما تنتهي العملية الفرعية، يتم استدعاء callback المحدد مع userdata.

تتيح وظيفة sd_event_add_child_pidfd للمستخدمين إضافة مراقب لعملية فرعية محددة بواسطة pidfd. هذا مشابه لوظيفة sd_event_add_child ولكنه يستخدم pidfd_open(2) بدلاً من pid.

sd_event_child_handler_t هو نوع دالة يسمح للمستخدمين بتحديد معالج لحدث انتهاء العملية الفرعية. يأخذ event، pid للعملية الفرعية، wstatus الذي يحتوي على معلومات حالة انتهاء العملية الفرعية، و userdata.

توفر دوال sd_event_source_get_* و sd_event_source_set_* وصولاً إلى معلومات حول العملية الفرعية التي يراقبها المصدر، مثل pid، pidfd، وما إذا كان مصدر الحدث يمتلك العملية الفرعية أو pidfd.

تتيح وظيفة sd_event_source_send_child_signal إرسال إشارة UNIX إلى العملية الفرعية المراقبة. تستخدم هذه الوظيفة pidfd_send_signal(2).

يمكن استخدام sd_event_source_set_child_pidfd_own() و sd_event_source_set_child_process_own() لتغيير إعداد الملكية. تأخذ هذه الدوال معلمة منطقية واحدة مع الإعداد الجديد. لاحظ أنه في الوقت الحالي، إذا انتهت العملية المتصلة بشكل غير طبيعي، فقد تبقى العملية المراقبة على قيد الحياة على الرغم من توقف مصدر الحدث عن الوجود. قد يتغير هذا السلوك لاحقًا.

يمكن استخدام sd_event_source_send_child_signal() لإرسال إشارة UNIX إلى العملية المراقبة عبر pidfd_send_signal(2). تتطابق المعلمات المحددة مع تلك الخاصة بالاستدعاء النظامي الأساسي، باستثناء أن info لا يتم تعديلها أبدًا (وبالتالي تم الإعلان عنها كـ ثابتة). كما هو الحال مع الاستدعاء النظامي الأساسي، يجب أن تكون المعلمة flags صفرًا حاليًا.

RETURN VALUE

عند النجاح، تعيد هذه الدوال 0 أو عددًا صحيحًا موجبًا. عند الفشل، تعيد رمز خطأ سالبًا بأسلوب errno.

Errors

قد تشير الأخطاء المعادة إلى المشاكل التالية:

-ENOMEM

لا توجد ذاكرة كافية لتخصيص كائن.

-EINVAL

تم تمرير وسيط غير صالح. يشمل ذلك تحديد قناع فارغ في options أو قناع يحتوي على قيم بخلاف تركيبة WEXITED، WSTOPPED، و WCONTINUED.

-EBUSY

تم تثبيت معالج بالفعل لهذه العملية الفرعية، أو أن SIGCHLD غير محظور.

-ESTALE

حلقة الأحداث قد انتهت بالفعل.

-ECHILD

تم إنشاء حلقة الأحداث في عملية مختلفة، أو مكتبة، أو نسخة وحدة.

-EDOM

مصدر الحدث الذي تم تمريره ليس مصدر حدث عملية فرعية.

NOTES

الوظائف الموصوفة هنا متاحة كمكتبة مشتركة، والتي يمكن تجميعها وربطها باستخدام ملف pkg-config(1) الخاص بـ libsystemd.

يستخدم الكود الموصوف هنا getenv(3)، والذي تم الإعلان عنه على أنه غير آمن للاستخدام مع الخيوط المتوازية. هذا يعني أن الكود الذي يستدعي الدوال الموصوفة هنا يجب ألا يستدعي setenv(3) من خيط متوازي. يوصى فقط بإجراء استدعاءات لـ setenv() من مرحلة مبكرة من البرنامج عندما لم يتم بدء أي خيوط أخرى.

EXAMPLE

Example 1. Exit loop when the child terminates

/* SPDX-License-Identifier: MIT-0 */
#define _GNU_SOURCE 1
#include <assert.h>
#include <stdio.h>
#include <unistd.h>
#include <systemd/sd-event.h>
int main(int argc, char **argv) {
  pid_t pid = fork();
  assert(pid >= 0);
  /* SIGCHLD signal must be blocked for sd_event_add_child to work */
  sigset_t ss;
  sigemptyset(&ss);
  sigaddset(&ss, SIGCHLD);
  sigprocmask(SIG_BLOCK, &ss, NULL);
  if (pid == 0)  /* child */
    sleep(1);
  else {         /* parent */
    sd_event *e = NULL;
    int r;
    /* Create the default event loop */
    sd_event_default(&e);
    assert(e);
    /* We create a floating child event source (attached to 'e').
     * The default handler will be called with 666 as userdata, which
     * will become the exit value of the loop. */
    r = sd_event_add_child(e, NULL, pid, WEXITED, NULL, (void*) 666);
    assert(r >= 0);
    r = sd_event_loop(e);
    assert(r == 666);
    sd_event_unref(e);
  }
  return 0;
}
