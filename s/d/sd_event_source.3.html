#include <systemd/sd-event.h>
typedef struct sd_event_source sd_event_source;

typedef int (*sd_event_io_handler_t)(sd_event_source *s, int fd, uint32_t revents, void *userdata);

الوصف

sd_event_add_io() يضيف مصدر حدث I/O جديد إلى حلقة أحداث. يتم تحديد كائن حلقة الأحداث في المعلمة event، ويتم إرجاع كائن مصدر الحدث في المعلمة source. تأخذ المعلمة fd وصف ملف UNIX لمراقبته، والذي قد يشير إلى مقبس، أو FIFO، أو قائمة انتظار رسائل، أو اتصال تسلسلي، أو جهاز حرفي، أو أي واصف ملف آخر متوافق مع epoll(7) في Linux. تأخذ المعلمة events قناع بت للأحداث المراد مراقبتها، وهو مزيج من أعلام الأحداث التالية: EPOLLIN، EPOLLOUT، EPOLLRDHUP، EPOLLPRI، و EPOLLET، راجع epoll_ctl(2) للحصول على التفاصيل. لاحظ أن ليست كل واصفات الملفات متوافقة مع epoll، على سبيل المثال الملفات العادية أو المجلدات ليست كذلك. إذا تم استدعاء هذه الدالة مع واصف ملف لا يدعم epoll، فسيتم إرجاع -EPERM (انظر أدناه أيضًا). في معظم الحالات، يمكن التعامل مع واصفات الملفات هذه على أنها قابلة للقراءة دائمًا أو قابلة للكتابة دائمًا، بحيث لا تكون مراقبة أحداث I/O ضرورية.

handler هي دالة يتم استدعاؤها عند تشغيل مصدر الحدث أو NULL. سيتم تمرير المؤشر userdata إلى الدالة المعالجة، ويمكن أن يختاره المتصل بحرية. سيتم أيضًا تمرير واصف الملف الذي شوهد عليه الحدث، بالإضافة إلى علامات الحدث الفعلية، إلى المعالج. عادة ما تكون هذه مجموعة فرعية من الأحداث التي تمت مراقبتها، ولكنها قد تشمل أيضًا EPOLLERR و EPOLLHUP. يمكن للمعالج إرجاع قيمة سالبة للإشارة إلى خطأ (انظر أدناه)، وسيتم تجاهل القيم المرجعة الأخرى. إذا كان handler هو NULL، فسيتم استخدام معالج افتراضي يستدعي sd_event_exit(3).

افتراضيًا، يظل مصدر الحدث ممكّنًا بشكل مستمر (SD_EVENT_ON)، ولكن يمكن تغيير ذلك باستخدام sd_event_source_set_enabled(3). إذا أعادت الدالة المعالجة رمز خطأ سالبًا، فسيتم تعطيلها بعد الاستدعاء، حتى لو تم طلب وضع SD_EVENT_ON من قبل، أو سيؤدي ذلك إلى إنهاء الحلقة، انظر sd_event_source_set_exit_on_failure(3). لاحظ أن مصدر الحدث الذي تم تعيينه على SD_EVENT_ON سيتم تشغيله باستمرار ما لم يتم قراءة البيانات من أو كتابتها إلى واصف الملف لإعادة تعيين قناع الأحداث التي تم رؤيتها.

تعيين قناع حدث I/O للمراقبة إلى 0 لا يعني أن مصدر الحدث لن يتم تشغيله بعد الآن، حيث قد يتم تشغيل EPOLLHUP و EPOLLERR حتى مع قناع حدث صفري. لتعطيل مصدر حدث I/O مؤقتًا، استخدم sd_event_source_set_enabled(3) مع SD_EVENT_OFF بدلاً من ذلك.

لتدمير كائن مصدر الحدث، استخدم sd_event_source_unref(3)، ولكن لاحظ أنه لا يتم إزالة مصدر الحدث من حلقة الأحداث إلا عند إسقاط جميع المراجع إلى مصدر الحدث. للتأكد من أن مصدر الحدث لا يتم تشغيله بعد الآن، حتى لو كان لا يزال مرجعيًا، قم بتعطيل مصدر الحدث باستخدام sd_event_source_set_enabled(3) مع SD_EVENT_OFF.

إذا كانت المعلمة الثانية لـ sd_event_add_io() هي NULL، فلا يتم إرجاع أي مرجع لكائن مصدر الحدث. في هذه الحالة، يعتبر مصدر الحدث "عائمًا"، وسيتم تدميره ضمنيًا عند تدمير حلقة الأحداث نفسها.

إذا كانت handler لـ sd_event_add_io() هي NULL، وتم تشغيل مصدر الحدث، فسيتم اعتبار هذا طلبًا للخروج من حلقة الأحداث. في هذه الحالة، يتم تمرير المعلمة userdata، بعد تحويلها إلى عدد صحيح، كمعلمة رمز الخروج إلى sd_event_exit(3).

لاحظ أن هذه المكالمة لا تأخذ ملكية واصف الملف المقدم، تظل الملكية (وبالتالي واجب إغلاقه عند عدم الحاجة إليه) مع المتصل. ومع ذلك، مع المكالمة sd_event_source_set_io_fd_own() (انظر أدناه)، يمكن لمصدر الحدث اختياريًا أخذ ملكية واصف الملف بعد إنشاء مصدر الحدث. في هذه الحالة، يتم إغلاق واصف الملف تلقائيًا بمجرد تحرير مصدر الحدث.

يوصى باستخدام sd_event_add_io() فقط بالاشتراك مع واصفات الملفات التي تم تعيين O_NONBLOCK لها، لضمان أن جميع عمليات I/O من المعالجات المستدعاة غير متزامنة وغير حاجزة بشكل صحيح. قد يؤدي استخدام واصفات الملفات بدون O_NONBLOCK إلى تجويع غير متوقع لمصادر الأحداث الأخرى. انظر fcntl(2) للحصول على التفاصيل حول تمكين وضع O_NONBLOCK.

sd_event_source_get_io_events() يسترد قناع أحداث I/O المراقب والمكوّن لمصدر حدث تم إنشاؤه مسبقًا باستخدام sd_event_add_io(). يأخذ كائن مصدر الحدث ومؤشرًا إلى متغير لتخزين القناع فيه.

sd_event_source_set_io_events() يقوم بتكوين قناع أحداث I/O المراقب لمصدر حدث تم إنشاؤه مسبقًا باستخدام sd_event_add_io(). يأخذ كائن مصدر الحدث وقناع الحدث الجديد.

sd_event_source_get_io_revents() يسترد قناع أحداث I/O للأحداث التي تم رؤيتها حاليًا ولكن لم يتم إرسالها بعد من مصدر حدث تم إنشاؤه مسبقًا باستخدام sd_event_add_io(). يأخذ كائن مصدر الحدث ومؤشرًا إلى متغير لتخزين قناع الحدث فيه. عند استدعائه من دالة معالجة على كائن مصدر الحدث الخاص بالمعالج، سيعيد هذا نفس القناع الذي تم تمريره إلى معلمة revents الخاصة بالمعالج. هذه المكالمة مفيدة بشكل أساسي للتحقق من الأحداث غير المرسلة لمصدر حدث من معالج حدث غير ذي صلة (ربما ذي أولوية أعلى). لاحظ العلاقة بين sd_event_source_get_pending() و sd_event_source_get_io_revents(): ستعيد كلتا الدالتين نتائج غير صفرية عندما يكون هناك حدث معلق لمصدر الحدث، ولكن الأولى تنطبق على جميع أنواع مصادر الأحداث، والأخيرة تنطبق فقط على مصادر أحداث I/O.

sd_event_source_get_io_fd() يسترد واصف ملف UNIX لمصدر حدث تم إنشاؤه مسبقًا باستخدام sd_event_add_io(). يأخذ كائن مصدر الحدث ويعيد واصف الملف غير السالب أو رقم خطأ سالب عند حدوث خطأ (انظر أدناه).

sd_event_source_set_io_fd() يغير واصف ملف UNIX لمصدر حدث I/O تم إنشاؤه مسبقًا باستخدام sd_event_add_io(). يأخذ كائن مصدر الحدث وواصف الملف الجديد كمعلمات. إذا كان مصدر الحدث يمتلك واصف الملف السابق، أي إذا تم استدعاء sd_event_source_set_io_fd_own() لمصدر الحدث بقيمة غير صفرية، فسيتم إغلاق واصف الملف السابق وسيأخذ مصدر الحدث أيضًا ملكية واصف الملف الجديد بنجاح.

sd_event_source_set_io_fd_own() يتحكم فيما إذا كان واصف الملف الخاص بمصدر الحدث سيأخذ ملكية واصف الملف. يأخذ معلمة منطقية b. عندما تكون صحيحة (غير صفرية)، فإن واصف الملف

هذا الموقع غير مرتبط بـ Arch Linux ولا تابع له بأي شكل من الأشكال. المحتوى هو ترجمة فقط لصفحات دليل Arch المنشورة على man.archlinux.org.

criptor سيتم إغلاقه تلقائيًا عند تحرير مصدر الحدث أو عند استبدال واصف الملف بواسطة sd_event_source_set_io_fd(). بشكل افتراضي، الواصف لا يملكه مصدر الحدث، وعلى التطبيق القيام بإغلاقه بنفسه إذا لزم الأمر.

يمكن استخدام sd_event_source_get_io_fd_own() للاستعلام عن الإعداد الحالي لعلامة ملكية واصف الملف كما تم تعيينها بواسطة sd_event_source_set_io_fd_own(). يعيد قيمة موجبة إذا تم إغلاق واصف الملف تلقائيًا عند تدمير مصدر الحدث، وصفر إذا لم يتم ذلك، وقيمة سالبة في حالة حدوث خطأ.

قيمة الإرجاع

عند النجاح، تعيد هذه الدوال 0 أو عددًا صحيحًا موجبًا. عند الفشل، تعيد رمز خطأ سالبًا بأسلوب errno.

الأخطاء

قد تشير القيم المعادة إلى المشاكل التالية:

-ENOMEM

لا توجد ذاكرة كافية لتخصيص كائن.

-EINVAL

تم تمرير وسيط غير صالح.

-ESTALE

تم إنهاء حلقة الأحداث بالفعل.

-ECHILD

تم إنشاء حلقة الأحداث في عملية مختلفة، أو مكتبة، أو مثيل وحدة.

-EDOM

مصدر الحدث الذي تم تمريره ليس مصدر حدث إدخال/إخراج.

-EPERM

واصف الملف الذي تم تمريره لا يدعم واجهة برمجة التطبيقات epoll(7)، على سبيل المثال لأنه ملف عادي أو دليل. راجع epoll_ctl(2) للحصول على التفاصيل.

تمت الإضافة في الإصدار 255.

ملاحظات

الدوال الموصوفة هنا متاحة كمكتبة مشتركة، والتي يمكن الترجمة عليها وربطها باستخدام ملف libsystemd pkg-config(1).

الكود الموصوف هنا يستخدم getenv(3)، والتي تم الإعلان عنها على أنها ليست آمنة للاستخدام المتعدد الخيوط. هذا يعني أنه يجب على الكود الذي يستدعي الدوال الموصوفة هنا عدم استدعاء setenv(3) من خيط موازٍ. يوصى فقط بإجراء استدعاءات لـ setenv() من مرحلة مبكرة من البرنامج عندما لم يتم بدء أي خيوط أخرى.
