sd_journal_seek_cursor(3)

الحزمة: systemd-libs

التراخيص: LGPL-2.1-or-later, CC0-1.0, GPL-2.0-or-later WITH Linux-syscall-note

SD_JOURNAL_SEEK_HEAD(3) sd_journal_seek_head SD_JOURNAL_SEEK_HEAD(3)

الاسم

sd_journal_seek_head، sd_journal_seek_tail، sd_journal_seek_monotonic_usec، sd_journal_seek_realtime_usec، sd_journal_seek_cursor - الانتقال إلى موضع في السجل

الملخص

#include <systemd/sd-journal.h>

int sd_journal_seek_head(sd_journal *j);

int sd_journal_seek_tail(sd_journal *j);

int sd_journal_seek_monotonic_usec(sd_journal *j, sd_id128_t boot_id, uint64_t usec);

int sd_journal_seek_realtime_usec(sd_journal *j, uint64_t usec);

int sd_journal_seek_cursor(sd_journal *j, const char *cursor);

الوصف

sd_journal_seek_head() ينتقل إلى بداية السجل، أي إلى الموضع قبل أقدم إدخال متاح.

وبالمثل، يمكن استخدام sd_journal_seek_tail() للانتقال إلى نهاية السجل، أي الموضع بعد أحدث إدخال متاح.

sd_journal_seek_monotonic_usec() ينتقل إلى موضع بالطابع الزمني الوحيد المحدد، أي CLOCK_MONOTONIC. نظرًا لأن الوقت الوحيد يعاد تعيينه مع كل إعادة تشغيل، يجب تحديد معرف إعادة التشغيل أيضًا.

sd_journal_seek_realtime_usec() ينتقل إلى موضع بالطابع الزمني الحقيقي (ساعة الحائط) المحدد، أي CLOCK_REALTIME. لاحظ أن الساعة الحقيقية ليست بالضرورة وحيدة. إذا كان الطابع الزمني الحقيقي غامضًا، فليس من المحدد الموضع الذي تم الانتقال إليه.

sd_journal_seek_cursor() ينتقل إلى الموضع المحدد في سلسلة المؤشر المحددة. لمزيد من التفاصيل حول المؤشرات، راجع sd_journal_get_cursor(3). إذا لم يتم العثور على إدخال يطابق المؤشر المحدد، فسيتم الانتقال إلى أقرب إدخال تالية (من حيث الوقت) بدلاً من ذلك.

لاحظ أن هذه الاستدعاءات لا تجعل أي إدخال هو الإدخال الحالي الجديد فعليًا، يجب القيام بذلك في خطوة منفصلة باستدعاء sd_journal_next(3) لاحق (أو استدعاء مشابه). عندها فقط، يمكن استرداد بيانات الإدخال عبر sd_journal_get_data(3) أو استرداد مؤشر إدخال عبر sd_journal_get_cursor(3). إذا لم يكن هناك إدخال موجود يطابق عنوان الانتقال المحدد تمامًا، يتم البحث عن أقرب إدخال تالية. إذا تم استخدام sd_journal_next(3)، فسيتم البحث عن أقرب إدخال تالية، إذا تم استخدام sd_journal_previous(3) فسيتم البحث عن أقرب إدخال سابقة.

بعد الانتهاء من الانتقال، وبعد استدعاء sd_journal_next(3) أو استدعاء مشابه، يمكن استخدام sd_journal_test_cursor(3) للتحقق مما إذا كان الإدخال المحدد حديثًا يطابق المؤشر بالفعل.

قيمة الإرجاع

تعيد الدوال 0 عند النجاح أو رمز خطأ سالب بأسلوب errno.

ملاحظات

جميع الدوال المدرجة هنا مستقلة عن الخيوط، ويمكن لخيط واحد فقط العمل على كائن معين في أي وقت. قد تصل خيوط مختلفة إلى نفس الكائن في أوقات مختلفة. يمكن استخدام كائنات مستقلة متعددة من خيوط مختلفة بالتوازي.

الدوال الموصوفة هنا متاحة كمكتبة مشتركة، والتي يمكن تجميعها وربطها باستخدام ملف libsystemd pkg-config(1).
