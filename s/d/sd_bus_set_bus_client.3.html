sd_bus_set_bus_client(3)

الحزمة: systemd-libs

التراخيص: LGPL-2.1-or-later, CC0-1.0, GPL-2.0-or-later WITH Linux-syscall-note

SD_BUS_SET_SERVER(3) sd_bus_set_server SD_BUS_SET_SERVER(3)

الاسم

sd_bus_set_server, sd_bus_is_server, sd_bus_get_bus_id, sd_bus_set_bus_client, sd_bus_is_bus_client, sd_bus_set_monitor, sd_bus_is_monitor - تهيئة وضع الاتصال لكائن الناقل

ملخص

#include <systemd/sd-bus.h>

int sd_bus_set_server(sd_bus *bus, int b, sd_id128_t id);

int sd_bus_is_server(sd_bus *bus);

int sd_bus_get_bus_id(sd_bus *bus, sd_id128_t *id);

int sd_bus_set_bus_client(sd_bus *bus, int b);

int sd_bus_is_bus_client(sd_bus *bus);

int sd_bus_set_monitor(sd_bus *bus, int b);

int sd_bus_is_monitor(sd_bus *bus);

الوصف

sd_bus_set_server() يهيئ كائن الناقل كخادم لاتصالات D-Bus المباشرة. b يمكّن/يعطّل وضع الخادم. إذا كان صفرًا، يتم تعطيل وضع الخادم. بخلاف ذلك، يتم تمكين وضع الخادم. يلزم تهيئة كائن الناقل كخادم للسماح بإقامة اتصالات مباشرة بين طرفين دون المرور عبر برنامج D-Bus. يجب أن يحتوي id على معرف رقم صحيح 128 بت للخادم. إذا أضاف العملاء حقل guid إلى سلسلة عنوان D-Bus الخاص بهم، فيجب أن يتطابق معرف الخادم مع هذا guid أو ستفشل مصافحة مصادقة D-Bus. إذا لم يتم تحديد معرف معين للخادم، فيمكن استخدام sd_id128_randomize(3) لتوليد معرف عشوائي بدلاً من ذلك.

sd_bus_is_server() يعيد ما إذا كان وضع الخادم ممكّنًا لكائن الناقل المحدد.

sd_bus_get_bus_id() يخزن معرف خادم D-Bus الذي تم تكوينه باستخدام sd_bus_set_server() (لكائنات ناقل الخادم) أو الذي تم استلامه أثناء مصادقة D-Bus (لكائنات ناقل العميل) في id.

sd_bus_set_bus_client() يهيئ كائن الناقل كعميل لبرنامج D-Bus. b يمكّن/يعطّل وضع العميل. إذا كان صفرًا، يتم تعطيل وضع العميل ويجب أن يتصل كائن الناقل مباشرة بخادم D-Bus. بخلاف ذلك، يتم تمكين وضع العميل ويجب أن يتصل كائن الناقل ببرنامج D-Bus. عند الاتصال بناقل موجود باستخدام أي من الوظائف في عائلة وظائف sd_bus_open(3) أو أي من الوظائف في عائلة وظائف sd_bus_default(3)، يتم تكوين كائن الناقل تلقائيًا كعميل ناقل. ومع ذلك، عند الاتصال ببرنامج D-Bus عن طريق استدعاء sd_bus_set_address(3) متبوعًا بـ sd_bus_start(3)، يجب تكوين كائن الناقل يدويًا كعميل ناقل باستخدام sd_bus_set_bus_client(). بشكل افتراضي، لا يتم تكوين كائن الناقل كعميل لبرنامج D-Bus.

sd_bus_is_bus_client() يعيد ما إذا كان وضع العميل ممكّنًا/معطّلاً لكائن الناقل المحدد.

sd_bus_set_monitor() يهيئ كائن الناقل ككائن مراقبة D-Bus. b يمكّن/يعطّل وضع المراقبة. إذا كان صفرًا، يتم تعطيل وضع المراقبة. إذا كان غير صفري، يتم تمكين وضع المراقبة. عند تمكين وضع المراقبة، لا يجوز إرسال أي رسائل عبر كائن الناقل ولا يجوز له كشف أي كائنات على الناقل. لبدء مراقبة الرسائل، استدعِ طريقة org.freedesktop.DBus.Monitoring.BecomeMonitor لبرنامج D-Bus ومرر قائمة بالمطابقات التي تشير إلى الرسائل التي يجب اعتراضها. راجع مواصفات D-Bus[1] لمزيد من المعلومات.

sd_bus_is_monitor() يعيد ما إذا كان وضع المراقبة ممكّنًا/معطّلاً لكائن الناقل المحدد.

قيمة الإرجاع

عند النجاح، تعيد sd_bus_set_server()، sd_bus_get_bus_id()، sd_bus_set_bus_client() و sd_bus_set_monitor() عددًا صحيحًا غير سالب. عند الفشل، تعيد رمز خطأ سلبي بنمط errno.

sd_bus_is_server()، sd_bus_is_bus_client() و sd_bus_is_monitor() تعيد عددًا صحيحًا موجبًا عندما يكون وضع الخادم أو العميل ممكّنًا، على التوالي. بخلاف ذلك، تعيد صفرًا.

الأخطاء

قد تشير الأخطاء المعادة إلى المشاكل التالية:

-ECHILD

تم إنشاء اتصال الناقل في عملية أو مكتبة أو مثيل وحدة نمطية مختلف.

تمت الإضافة في الإصدار 246.

-EPERM

تم تشغيل اتصال الناقل بالفعل.

تمت الإضافة في الإصدار 246.

-ENOPKG

لا يمكن حل الناقل.

تمت الإضافة في الإصدار 246.

-EINVAL

كان أحد المعلمات المطلوبة NULL أو كان b صفرًا و id لم يساوي SD_ID128_NULL.

تمت الإضافة في الإصدار 246.

-ENOTCONN

الناقل غير متصل.

تمت الإضافة في الإصدار 246.

ملاحظات

الوظائف الموصوفة هنا متاحة كمكتبة مشتركة، والتي يمكن تجميعها وارتباطها باستخدام ملف pkg-config(1) الخاص بـ libsystemd.

الكود الموصوف هنا يستخدم getenv(3)، وهو معلن على أنه ليس آمنًا للاستخدام في المواضيع المتوازية. هذا يعني أن الكود الذي يستدعي الوظائف الموصوفة هنا يجب ألا يستدعي setenv(3) من موضوع موازٍ. يوصى بإجراء استدعاءات لـ setenv() فقط من مرحلة مبكرة من البرنامج عندما لم يتم بدء أي مواضيع أخرى.
