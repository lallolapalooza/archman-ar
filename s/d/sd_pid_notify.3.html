الاسم

sd_notify, sd_notifyf, sd_pid_notify, sd_pid_notifyf, sd_pid_notify_with_fds, sd_pid_notifyf_with_fds, sd_notify_barrier, sd_pid_notify_barrier - إعلام مدير الخدمة باكتمال البدء وتغييرات حالة الخدمة الأخرى

المُلَخَّص

#include <systemd/sd-daemon.h>

int sd_notify(int unset_environment, const char *state);

int sd_notifyf(int unset_environment, const char *format, ...);

int sd_pid_notify(pid_t pid, int unset_environment, const char *state);

int sd_pid_notifyf(pid_t pid, int unset_environment, const char *format, ...);

int sd_pid_notify_with_fds(pid_t pid, int unset_environment, const char *state, const int *fds, unsigned n_fds);

int sd_pid_notifyf_with_fds(pid_t pid, int unset_environment, const int *fds, size_t n_fds, const char *format, ...);

int sd_notify_barrier(int unset_environment, uint64_t timeout);

int sd_pid_notify_barrier(pid_t pid, int unset_environment, uint64_t timeout);

الوصف

sd_notify() يمكن استدعاؤها بواسطة خدمة لإعلام مدير الخدمة بتغييرات الحالة. يمكن استخدامها لإرسال معلومات عشوائية، مشفرة في سلسلة تشبه كتلة البيئة. الأهم من ذلك، يمكن استخدامها لإشعارات اكتمال البدء أو إعادة التحميل.

إذا كان المعامل unset_environment غير صفري، فإن sd_notify() ستقوم بإلغاء تعيين متغير البيئة $NOTIFY_SOCKET قبل العودة (بغض النظر عما إذا كان استدعاء الوظيفة نفسه ناجحًا أم لا). الاستدعاءات اللاحقة لـ sd_notify() لن تفعل شيئًا بصمت، ولن يتم توريث المتغير بعد ذلك إلى العمليات الفرعية.

يجب أن تحتوي المعامل state على قائمة بفصل الأسطر من تعيينات المتغيرات، مشابهة في الأسلوب لكتلة البيئة. يتم استنتاج سطر فاصل لاحق إذا لم يتم تحديده. قد تحتوي السلسلة على أي نوع من تعيينات المتغيرات، ولكن انظر القسم التالي للحصول على قائمة بالتعيينات التي يفهمها مدير الخدمة.

لاحظ أن systemd سيقبل بيانات الحالة المرسلة من خدمة فقط إذا تم تعيين خيار NotifyAccess= بشكل صحيح في ملف تعريف الخدمة. انظر systemd.service(5) للتفاصيل.

لاحظ أن إشعارات sd_notify() قد تُنسب إلى الوحدات بشكل صحيح فقط إذا كانت العملية المرسلة لا تزال موجودة عند معالجة PID 1 للرسالة، أو إذا كانت العملية المرسلة يتم تتبعها بشكل صريح بواسطة مدير الخدمة. الحالة الأخيرة هي إذا كان مدير الخدمة قد أنشأ العملية في الأصل، أي على جميع العمليات التي تتطابق مع NotifyAccess=main أو NotifyAccess=exec. على العكس من ذلك، إذا أرسلت عملية مساعدة للوحدة رسالة sd_notify() وخرجت فورًا، فقد لا يتمكن مدير الخدمة من نسب الرسالة بشكل صحيح إلى الوحدة، وبالتالي سيتجاهلها، حتى لو تم تعيين NotifyAccess=all لها.

لذلك، للقضاء على جميع حالات التسابق التي تتضمن البحث عن وحدة العميل ونسب الإشعارات إلى الوحدات بشكل صحيح، يمكن استخدام sd_notify_barrier(). تعمل هذه المكالمة كنقطة مزامنة وتضمن أن جميع الإشعارات المرسلة قبل هذه المكالمة قد تم التقاطها بواسطة مدير الخدمة عند عودتها بنجاح. يلزم استخدام sd_notify_barrier() للعملاء الذين لا يتم استدعاؤهم بواسطة مدير الخدمة، وإلا فإن آلية المزامنة هذه غير ضرورية لنسب الإشعارات إلى الوحدة.

sd_notifyf() مشابه لـ sd_notify() ولكنه يأخذ سلسلة تنسيق مشابهة لـ printf() بالإضافة إلى وسائط.

sd_pid_notify() و sd_pid_notifyf() مشابهان لـ sd_notify() و sd_notifyf() ولكنهما يأخذان معرف عملية (PID) لاستخدامه كمعرف أصل للرسالة كوسيط أول. هذا مفيد لإرسال رسائل إعلام نيابة عن عمليات أخرى، شريطة توفر الامتيازات المناسبة. فعليًا، هذا يعني أن استدعاء مميز لـ sd_pid_notify() قد يتجاوز قيود NotifyAccess=main أو NotifyAccess=exec المفروضة على الخدمة. إذا تم تحديد وسيط PID كـ 0، يتم استخدام معرف العملية للمتصل، وفي هذه الحالة تكون الاستدعاءات مكافئة تمامًا لـ sd_notify() و sd_notifyf().

sd_pid_notify_with_fds() مشابه لـ sd_pid_notify() ولكنه يأخذ مصفوفة إضافية من واصفات الملفات. يتم إرسال واصفات الملفات هذه مع رسالة الإعلام إلى مدير الخدمة. هذا مفيد بشكل خاص لإرسال رسائل "FDSTORE=1"، كما هو موضح أعلاه. الوسائط الإضافية هي مؤشر إلى مصفوفة واصفات الملفات بالإضافة إلى عدد واصفات الملفات في المصفوفة. إذا تم تمرير عدد واصفات الملفات كـ 0، فإن المكالمة مكافئة تمامًا لـ sd_pid_notify()، أي لا يتم تمرير أي واصفات ملفات. لاحظ أنه يتم إغلاق واصفات الملفات المرسلة إلى مدير الخدمة في رسالة بدون "FDSTORE=1" على الفور عند الاستلام.

sd_pid_notifyf_with_fds() هو مزيج من sd_pid_notify_with_fds() و sd_notifyf()، أي أنه يقبل كلاً من PID ومجموعة من واصفات الملفات كمدخلات، ويعالج سلسلة تنسيق لإنشاء سلسلة الحالة.

sd_notify_barrier() تسمح للمستدعي بالمزامنة ضد استلام رسائل الإعلام المرسلة سابقًا وتستخدم الأمر BARRIER=1. يأخذ قيمة timeout نسبية بالميكروثانية يتم تمريرها إلى ppoll(2). يتم تفسير القيمة UINT64_MAX على أنها مهلة غير محدودة.

sd_pid_notify_barrier() هي تمامًا مثل sd_notify_barrier()، ولكنها تسمح بتحديد PID المصدر لرسالة الإعلام.

التعيينات المعروفة

التعيينات التالية لها معنى محدد:

READY=1

يخبر مدير الخدمة أن بدء تشغيل الخدمة قد اكتمل، أو أن الخدمة أكملت إعادة تحميل تكوينها. يستخدم هذا فقط بواسطة systemd إذا كان ملف تعريف الخدمة يحتوي على Type=notify أو Type=notify-reload. نظرًا لعدم وجود قيمة كبيرة للإشارة إلى عدم الجاهزية، فإن القيمة الوحيدة التي يجب أن ترسلها الخدمات هي "READY=1" (أي أن "READY=0" غير معرّف).

RELOADING=1

يخبر مدير الخدمة أن الخدمة بدأت في إعادة تحميل تكوينها. هذا مفيد للسماح لمدير الخدمة بتتبع الحالة الداخلية للخدمة، وعرضها للمستخدم. لاحظ أن الخدمة التي ترسل هذا الإشعار يجب أن ترسل أيضًا إشعار "READY=1" عند اكتمال إعادة تحميل تكوينها. تتم نشر عمليات إعادة التحميل التي يتم إعلام مدير الخدمة بها باستخدام هذه الآليات بنفس الطريقة التي يتم بها عند بدء تشغيلها في الأصل من خلال مدير الخدمة. هذه الرسالة ذات أهمية خاصة لخدمات Type=notify-reload، لإبلاغ مدير الخدمة بأنه تم استلام طلب إعادة تحميل الخدمة ويتم الآن معالجته.

تمت الإضافة في الإصدار 217.

STOPPING=1

يخبر مدير الخدمة أن الخدمة بدأت في إيقاف تشغيلها. هذا مفيد للسماح لمدير الخدمة بتتبع الحالة الداخلية للخدمة، وعرضها للمستخدم.

تمت الإضافة في الإصدار 217.

MONOTONIC_USEC=...

حقل يحمل الطابع الزمني المنتظم (وفقًا لـ CLOCK_MONOTONIC) منسقًا بالعشرية بالميكروثانية، عند إنشاء رسالة الإعلام بواسطة العميل. يستخدم هذا عادةً بالاشتراك مع "RELOADING=1"، للسماح لمدير الخدمة بمزامنة دورات إعادة التحميل بشكل صحيح. انظر systemd.service(5) للحصول على التفاصيل، وخاصة "Type=notify-reload".

تمت الإضافة في الإصدار 253.

STATUS=...

يمرر سلسلة حالة UTF-8 ذات سطر واحد مرة أخرى إلى مدير الخدمة تصف حالة الخدمة. هذا مجاني ويمكن استخدامه لأغراض مختلفة: يمكن لبرامج مثل fsck تمرير نسب الإنجاز، وبرامج الفشل يمكنها تمرير خطأ قابل للقراءة بشريًا.
message. مثال: "STATUS=Completed 66% of file system check..."

تمت الإضافة في الإصدار 233.

NOTIFYACCESS=...

إعادة تعيين الوصول إلى مقبس إشعار حالة الخدمة أثناء التشغيل، متجاوزًا إعداد NotifyAccess= في ملف وحدة الخدمة. راجع systemd.service(5) للحصول على التفاصيل، وخاصة "NotifyAccess=" للحصول على قائمة بالقيم المقبولة.

تمت الإضافة في الإصدار 254.

ERRNO=...

إذا فشلت خدمة، يتم تنسيق رمز خطأ نمط errno كسلسلة نصية. مثال: "ERRNO=2" لـ ENOENT.

تمت الإضافة في الإصدار 233.

BUSERROR=...

إذا فشلت خدمة، يتم عرض رمز خطأ نمط D-Bus. مثال: "BUSERROR=org.freedesktop.DBus.Error.TimedOut".

تمت الإضافة في الإصدار 233.

VARLINKERROR=...

إذا فشلت خدمة، يتم عرض رمز خطأ نمط Varlink. مثال: "VARLINKERROR=org.varlink.service.InvalidParameter".

تمت الإضافة في الإصدار 257.

EXIT_STATUS=...

حالة خروج الخدمة أو المدير نفسه. لاحظ أن systemd حاليًا لا يستهلك هذه القيمة عند إرسالها بواسطة الخدمات، لذلك هذا التعيين هو للمعلومات فقط. سيرسل المدير هذا الإشعار إلى مقبس الإشعار الخاص به، والذي يمكن استخدامه لجمع حالة خروج من النظام (حاوية أو جهاز افتراضي) أثناء إيقاف تشغيله. على سبيل المثال، mkosi(1) يستفيد من هذا. يمكن تعيين القيمة المراد إرجاعها عبر systemctl(1) الأمر exit.

تمت الإضافة في الإصدار 254.

MAINPID=...

تغيير معرف العملية الرئيسية (PID) للخدمة. هذا مفيد بشكل خاص في حالة عدم إنشاء العملية الرئيسية الحقيقية مباشرة بواسطة مدير الخدمة. مثال: "MAINPID=4711".

تمت الإضافة في الإصدار 233.

MAINPIDFDID=...

رقم عقدة نظام الملفات (inode number) لمقبض العملية الرئيسي الجديد (المحدد عبر MAINPID=). يمكن الحصول على هذه المعلومات من خلال sd_pidfd_get_inode_id(3) على مقبض العملية وتستخدم لتحديد العملية بطريقة خالية من السباقات. بدلاً من ذلك، يمكن إرسال مقبض العملية مباشرة إلى مدير الخدمة (انظر MAINPIDFD=1 أدناه).

تمت الإضافة في الإصدار 257.

MAINPIDFD=1

مشابه لـ MAINPID= مع MAINPIDFDID=، ولكن يتم الإشارة إلى العملية مباشرة بواسطة مقبض العملية الذي تم تمريره إلى مدير الخدمة. هذا مفيد إذا لم يكن معرف مقبض العملية مدعومًا على النظام. يُتوقع وجود مقبس عملية واحد بالضبط لهذا الإشعار.

تمت الإضافة في الإصدار 257.

WATCHDOG=1

يطلب من مدير الخدمة تحديث طابع watchdog الزمني. هذا هو إشعار "keep-alive" الذي تحتاج الخدمات إلى إصداره على فترات منتظمة إذا تم تمكين WatchdogSec= لها. راجع systemd.service(5) للحصول على معلومات حول كيفية تمكين هذه الوظيفة و sd_watchdog_enabled(3) للحصول على تفاصيل حول كيفية تحقق الخدمة مما إذا كان watchdog ممكّنًا.

WATCHDOG=trigger

يخبر مدير الخدمة بأن الخدمة اكتشفت خطأ داخليًا يجب التعامل معه بواسطة خيارات watchdog المهيأة. سيؤدي هذا إلى تشغيل نفس السلوك كما لو كان WatchdogSec= ممكّنًا ولم ترسل الخدمة "WATCHDOG=1" في الوقت المناسب. لاحظ أنه لا يلزم تمكين WatchdogSec= لتشغيل "WATCHDOG=trigger" إجراء watchdog. راجع systemd.service(5) للحصول على معلومات حول سلوك watchdog.

تمت الإضافة في الإصدار 243.

WATCHDOG_USEC=...

إعادة تعيين قيمة watchdog_usec أثناء التشغيل. لاحظ أن هذا غير متاح عند استخدام sd_event_set_watchdog() أو sd_watchdog_enabled(). مثال: "WATCHDOG_USEC=20000000"

تمت الإضافة في الإصدار 233.

EXTEND_TIMEOUT_USEC=...

يطلب من مدير الخدمة تمديد مهلة بدء التشغيل أو التشغيل أو إيقاف تشغيل الخدمة المقابلة للحالة الحالية. القيمة المحددة هي وقت بالميكروثانية يجب على الخدمة خلالها إرسال رسالة جديدة. سيحدث انتهاء مهلة الخدمة إذا لم يتم استلام الرسالة، ولكن فقط إذا تجاوز وقت تشغيل الحالة الحالية الأوقات القصوى الأصلية لـ TimeoutStartSec= و RuntimeMaxSec= و TimeoutStopSec=. راجع systemd.service(5) لتأثيرات مهلات الخدمة.

تمت الإضافة في الإصدار 236.

RESTART_RESET=1

إعادة تعيين عداد إعادة التشغيل للخدمة، مما يؤدي إلى استعادة مدة إعادة التشغيل إلى RestartSec= إذا تم استخدام RestartSteps= و RestartMaxDelaySec=. لمزيد من المعلومات، راجع systemd.service(5).

تمت الإضافة في الإصدار 258.

FDSTORE=1

تخزين واصفات الملفات في مدير الخدمة. سيتم الاحتفاظ بواصفات الملفات المرسلة بهذه الطريقة للخدمة بواسطة مدير الخدمة وسيتم إعادتها لاحقًا باستخدام منطق تمرير واصفات الملفات المعتاد عند بدء تشغيل الخدمة أو إعادة تشغيلها، راجع sd_listen_fds(3). أي مقابس مفتوحة وواصفات ملفات أخرى لا ينبغي إغلاقها أثناء إعادة التشغيل يمكن تخزينها بهذه الطريقة. عند إيقاف الخدمة، يتم تجاهل مخزن واصفات الملفات الخاص بها وإغلاق جميع واصفات الملفات فيه، باستثناء ما تم تجاوزه باستخدام FileDescriptorStorePreserve=، راجع systemd.service(5).

سيقبل مدير الخدمة رسائل لخدمة فقط إذا كان إعداد FileDescriptorStoreMax= الخاص بها غير صفري (افتراضيًا صفر، راجع systemd.service(5)). سيقوم مدير الخدمة بتعيين متغير البيئة $FDSTORE للخدمات التي تم تمكين مخزن واصفات الملفات الخاص بها، راجع systemd.exec(5).

إذا لم يتم تعيين FDPOLL=0 وكانت واصفات الملفات قابلة للاستجواب (pollable) (راجع epoll_ctl(2))، فإن أي حدث EPOLLHUP أو EPOLLERR يُرى عليها سيؤدي إلى إزالتها تلقائيًا من المخزن.

يمكن إرسال مجموعات متعددة من واصفات الملفات في رسائل منفصلة، وفي هذه الحالة يتم دمج المجموعات. يزيل مدير الخدمة واصفات الملفات المكررة (تلك التي تشير إلى نفس الكائن) قبل تمريرها إلى الخدمة.

يجب استخدام هذه الوظيفة لتنفيذ الخدمات التي يمكنها إعادة التشغيل بعد طلب صريح أو عطل دون فقدان الحالة. يمكن تسلسل حالة التطبيق إلى ملف في /run/، أو بشكل أفضل، تخزينها في واصف ملف ذاكرة memfd_create(2). استخدم sd_pid_notify_with_fds() لإرسال رسائل مع "FDSTORE=1". يوصى بالجمع بين FDSTORE= و FDNAME= لتسهيل إدارة واصفات الملفات المخزنة.

لمزيد من المعلومات حول مخزن واصفات الملفات، راجع نظرة عامة على مخزن واصفات الملفات[1].

تمت الإضافة في الإصدار 219.

FDSTOREREMOVE=1

يزيل واصفات الملفات من مخزن واصفات الملفات. يحتاج هذا الحقل إلى دمجه مع FDNAME= لتحديد اسم واصفات الملفات المراد إزالتها.

تمت الإضافة في الإصدار 236.

FDNAME=...

عند استخدامه بالاشتراك مع FDSTORE=1، يحدد اسمًا لواصفات الملفات المقدمة. عند استخدامه مع FDSTOREREMOVE=1، يحدد الاسم لواصفات الملفات المراد إزالتها. يتم تمرير هذا الاسم إلى الخدمة أثناء التفعيل، ويمكن الاستعلام عنه باستخدام sd_listen_fds_with_names(3). سيتم تسمية واصفات الملفات المقدمة بدون هذا الحقل "مخزنة".

يمكن أن يتكون الاسم من أي أحرف ASCII باستثناء أحرف التحكم أو ":". يجب ألا يتجاوز طوله 255 حرفًا. إذا كان الاسم المقدم لا يتبع هذه القيود، فسيتم تجاهله.

لاحظ أنه إذا تم تقديم واصفات ملفات متعددة في رسالة واحدة، فسيتم استخدام الاسم المحدد لجميعها. لتعيين أسماء مختلفة لواصفات الملفات المقدمة، قم بتقديمها في رسائل منفصلة.

تمت الإضافة في الإصدار 233.

FDPOLL=0

عند استخدامه بالاشتراك مع FDSTORE=1، يقوم بتعطيل استجواب واصفات الملفات المقدمة بغض النظر عما إذا كانت قابلة للاستجواب أم لا. نظرًا لأن هذا الخيار يعطل التنظيف التلقائي لواصفات الملفات المقدمة عند EPOLLERR و EPOLLHUP، يجب توخي الحذر لضمان التنظيف اليدوي المناسب. لا يُنصح عمومًا باستخدام هذا الخيار إلا عندما يكون للتنظيف التلقائي سلوك غير مرغوب فيه مثل التخلص المبكر من واصفات الملفات من المخزن.

تمت الإضافة في الإصدار 246.

BARRIER=1

يطلب من مدير الخدمة أن العميل يطلب بشكل صريح المزامنة عن طريق إغلاق واصف الملف المرسل مع هذا الأمر. يضمن مدير الخدمة أن معالجة أمر BARRIER=1 لن تحدث إلا بعد معالجة جميع رسائل الإشعارات السابقة المرسلة قبل هذا الأمر. وبالتالي، يمكن استخدام هذا الأمر المقترن بواصف ملف واحد للمزامنة مع استقبال جميع رسائل الحالة السابقة. لاحظ أنه لا يمكن خلط هذا الأمر مع إشعارات أخرى، ويجب إرساله في رسالة منفصلة إلى مدير الخدمة، وإلا سيتم تجاهل جميع التعيينات. لاحظ أن إرسال 0 أو أكثر من واصف ملف واحد مع هذا الأمر هو انتهاك للبروتوكول.

تمت الإضافة في الإصدار 246.

يتم تفسير رسائل الإشعار التي ترسلها الخدمات بواسطة مدير الخدمة. يتم تجاهل التعيينات غير المعروفة. وبالتالي، من الآمن (ولكن غالبًا بدون تأثير) إرسال التعيينات التي ليست في هذه القائمة. البروتوكول قابل للتوسيع، ولكن يجب توخي الحذر لضمان أن الامتدادات الخاصة يمكن التعرف عليها على هذا النحو. على وجه الخصوص، يوصى ببادئتها بـ "X_" متبوعة بمعرف مساحة اسم. يرسل مدير الخدمة أيضًا بعض الرسائل إلى مقبس الإشعار الخاص به، والذي قد يتم استهلاكه بواسطة مدير آلة مشرفة أو مدير حاويات أعلى في المكدس. يرسل مدير الخدمة عددًا من حقول الامتداد، على سبيل المثال X_SYSTEMD_UNIT_ACTIVE=، للحصول على التفاصيل راجع systemd(1).

قيمة الإرجاع

عند الفشل، تعيد هذه الاستدعاءات رمز خطأ نمط errno سالب. إذا لم يتم تعيين $NOTIFY_SOCKET وبالتالي لم يكن بالإمكان إرسال رسالة حالة، يتم إرجاع 0. إذا تم إرسال الحالة، تعيد هذه الدوال قيمة موجبة. لدعم مديري الخدمات الذين ينفذون هذه المخططات والذين لا ينفذونها، يوصى عمومًا بتجاهل قيمة الإرجاع لهذه الاستدعاءات. لاحظ أن قيمة الإرجاع تشير ببساطة إلى ما إذا كانت رسالة الإشعار قد تم وضعها في قائمة الانتظار بشكل صحيح، ولا تعكس ما إذا كانت العملية

الوصف

تُستخدم الدالة sd_notifyf لتنفيذ بروتوكول الإخطار الخاص بـ systemd، والذي يسمح للتطبيقات بإعلام systemd بحالتها. وهي تقبل سلسلة تنسيق بتنسيق printf(3)، وتُرسل الحالة إلى المقبس المذكور في متغير البيئة $NOTIFY_SOCKET.

تُعد sd_notifyf في الأساس التفافًا حول sd_notify، مع إضافة القدرة على تنسيق رسالة الحالة باستخدام سلسلة تنسيق.

تُرسل هذه الدوال رسالة بيانات (datagram) فردية مع سلسلة الحالة كحمولة إلى المقبس المشار إليه في متغير البيئة $NOTIFY_SOCKET. إذا كان الحرف الأول من $NOTIFY_SOCKET هو "/" أو "@"، يتم تفسير السلسلة على أنها مقبس AF_UNIX أو مقبس مساحة الأسماء المجردة في لينكس (على التوالي)، وفي كلتا الحالتين، تُرفق رسالة البيانات ببيانات اعتماد العملية الخاصة بالخدمة المرسلة، باستخدام SCM_CREDENTIALS.

إذا بدأت السلسلة بـ "vsock:"، فسيتم تفسيرها على أنها عنوان AF_VSOCK، وهو مفيد للمشرفين الفائقين (hypervisors) / مديري الأجهزة الافتراضية (VMMs) أو العمليات الأخرى على المضيف لتلقي إشعار عند الانتهاء من تشغيل جهاز افتراضي. لاحظ أنه في حالة عدم دعم المشرف الفائق لـ SOCK_DGRAM عبر AF_VSOCK، فسيتم استخدام SOCK_SEQPACKET بدلاً من ذلك. يمكن استخدام "vsock-stream" أو "vsock-dgram" أو "vsock-seqpacket" بدلاً من "vsock" لفرض استخدام نوع المقبس المقابل. يجب أن يكون العنوان بالتنسيق: "vsock:CID:PORT". لاحظ أنه بخلاف الاستخدامات الأخرى لـ vsock، فإن CID إلزامي ولا يمكن أن يكون "VMADDR_CID_ANY". لاحظ أن PID1 سيرسل حزم VSOCK من منفذ مميز (أي: أقل من 1024)، كمحاولة لمعالجة المخاوف من أن العمليات غير المميزة في الضيف قد تحاول إرسال إشعارات ضارة إلى المضيف، مما يدفعه إلى اتخاذ قرارات مدمرة بناءً عليها.

إذا كانت قيمة $NOTIFY_SOCKET غير موجودة، فسيتم تجاهل الإشعار بصمت. إذا فشل إرسال الرسالة، فسيتم إرجاع رمز خطأ.

لا تُرجع هذه الدالة شيئًا عندما تكون sd_notify قيد التشغيل، باستثناء حالة عدم وجود متصل بالمقبس $NOTIFY_SOCKET. يمكن أن تحدث هذه الحالة إذا كانت sd_notify تُستدعى قبل أو بعد تهيئة المخزن المؤقت الخاص بالمقبس، أو إذا تم إغلاق المقبس.

الأخطاء

قد تشير الأخطاء المُرجعة إلى المشاكل التالية:

-EINVAL

تم تمرير وسيطة غير صالحة (مثل NULL أو سلسلة فارغة).

-EAFNOSUPPORT

تم تكوين $NOTIFY_SOCKET لاستخدام نوع مقبس غير مدعوم.

-ENOENT

المقبس المشار إليه في $NOTIFY_SOCKET غير موجود.

-ECONNREFUSED

لا يوجد مستمع على المقبس المشار إليه في $NOTIFY_SOCKET.

-E2BIG

تم تمرير عدد كبير جدًا من واصفات الملفات مرة واحدة مما يسمح به النظام. على لينكس، يبلغ عدد واصفات الملفات التي يمكن تمريرها عبر مقابس AF_UNIX مرة واحدة 253، راجع unix(7) للحصول على التفاصيل.

تطبيقات مستقلة

لاحظ أنه، في حين أنه يجب تفضيل استخدام هذه المكتبة لتجنب تكرار التعليمات البرمجية، فمن الممكن أيضًا إعادة تنفيذ بروتوكول إخطار الاستعداد البسيط دون تبعيات خارجية، كما هو موضح في الأمثلة التالية المتضمنة من لغات مختلفة:

C 

/* SPDX-License-Identifier: MIT-0 */
/* Implement the systemd notify protocol without external dependencies.
 * Supports both readiness notification on startup and on reloading,
 * according to the protocol defined at:
 * https://www.freedesktop.org/software/systemd/man/latest/sd_notify.html
 * This protocol is guaranteed to be stable as per:
 * https://systemd.io/PORTABILITY_AND_STABILITY/ */
#define _GNU_SOURCE 1
#include <errno.h>
#include <inttypes.h>
#include <signal.h>
#include <stdbool.h>
#include <stddef.h>
#include <stdlib.h>
#include <stdio.h>
#include <string.h>
#include <sys/socket.h>
#include <sys/un.h>
#include <time.h>
#include <unistd.h>
#define _cleanup_(f) __attribute__((cleanup(f)))
static void closep(int *fd) {
  if (!fd || *fd < 0)
    return;
  close(*fd);
  *fd = -1;
}
static int notify(const char *message) {
  union sockaddr_union {
    struct sockaddr sa;
    struct sockaddr_un sun;
  } socket_addr = {
    .sun.sun_family = AF_UNIX,
  };
  size_t path_length, message_length;
  _cleanup_(closep) int fd = -1;
  const char *socket_path;
  /* Verify the argument first */
  if (!message)
    return -EINVAL;
  message_length = strlen(message);
  if (message_length == 0)
    return -EINVAL;
  /* If the variable is not set, the protocol is a noop */
  socket_path = getenv("NOTIFY_SOCKET");
  if (!socket_path)
    return 0; /* Not set? Nothing to do */
  /* Only AF_UNIX is supported, with path or abstract sockets */
  if (socket_path[0] != '/' && socket_path[0] != '@')
    return -EAFNOSUPPORT;
  path_length = strlen(socket_path);
  /* Ensure there is room for NUL byte */
  if (path_length >= sizeof(socket_addr.sun.sun_path))
    return -E2BIG;
  memcpy(socket_addr.sun.sun_path, socket_path, path_length);
  /* Support for abstract socket */
  if (socket_addr.sun.sun_path[0] == '@')
    socket_addr.sun.sun_path[0] = 0;
  fd = socket(AF_UNIX, SOCK_DGRAM|SOCK_CLOEXEC, 0);
  if (fd < 0)
    return -errno;
  if (connect(fd, &socket_addr.sa, offsetof(struct sockaddr_un, sun_path) + path_length) != 0)
    return -errno;
  ssize_t written = write(fd, message, message_length);
  if (written != (ssize_t) message_length)
    return written < 0 ? -errno : -EPROTO;
  return 1; /* Notified! */
}
static int notify_ready(void) {
  return notify("READY=1");
}
static int notify_reloading(void) {
  /* A buffer with length sufficient to format the maximum UINT64 value. */
  char reload_message[sizeof("RELOADING=1\nMONOTONIC_USEC=18446744073709551615")];
  struct timespec ts;
  uint64_t now;
  /* Notify systemd that we are reloading, including a CLOCK_MONOTONIC timestamp in usec
   * so that the program is compatible with a Type=notify-reload service. */
  if (clock_gettime(CLOCK_MONOTONIC, &ts) < 0)
    return -errno;
  if (ts.tv_sec < 0 || ts.tv_nsec < 0 ||
      (uint64_t) ts.tv_sec > (UINT64_MAX - (ts.tv_nsec / 1000ULL)) / 1000000ULL)
    return -EINVAL;
  now = (uint64_t) ts.tv_sec * 1000000ULL + (uint64_t) ts.tv_nsec / 1000ULL;
  if (snprintf(reload_message, sizeof(reload_message), "RELOADING=1\nMONOTONIC_USEC=%" PRIu64, now) < 0)
    return -EINVAL;
  return notify(reload_message);
}
static int notify_stopping(void) {
  return notify("STOPPING=1");
}
static volatile sig_atomic_t reloading = 0;
static volatile sig_atomic_t terminating = 0;
static void signal_handler(int sig) {
  if (sig == SIGHUP)
    reloading = 1;
  else if (sig == SIGINT || sig == SIGTERM)
    terminating = 1;
}
int main(int argc, char **argv) {
  struct sigaction sa = {
    .sa_handler = signal_handler,
    .sa_flags = SA_RESTART,
  };
  int r;
  /* Setup signal handlers */
  sigemptyset(&sa.sa_mask);
  sigaction(SIGHUP, &sa, NULL);
  sigaction(SIGINT, &sa, NULL);
  sigaction(SIGTERM, &sa, NULL);
  /* Do more service initialization work here ... */
  /* Now that all the preparations steps are done, signal readiness */
  r = notify_ready();
  if (r < 0) {
    fprintf(stderr, "Failed to notify readiness to $NOTIFY_SOCKET: %s\n", strerror(-r));
    return EXIT_FAILURE;
  }
  while (!terminating) {
    if (reloading) {
      reloading = false;
      /* As a separate but related feature, we can also notify the manager
       * when reloading configuration. This allows accurate state-tracking,
       * and also automated hook-in of 'systemctl reload' without having to
       * specify manually an ExecReload= line in the unit file. */
      r = notify_reloading();
      if (r < 0) {
        fprintf(stderr, "Failed to notify reloading to $NOTIFY_SOCKET: %s\n", strerror(-r));
        return EXIT_FAILURE;
      }
      /* Do some reconfiguration work here ... */
      r = notify_ready();
      if (r < 0) {
        fprintf(stderr, "Failed to notify readiness to $NOTIFY_SOCKET: %s\n", strerror(-r));
        return EXIT_FAILURE;
      }
    }
    /* Do some daemon work here ... */
    sleep(5);
  }
  r = notify_stopping();
  if (r < 0) {
    fprintf(stderr, "Failed to report termination to $NOTIFY_SOCKET: %s\n", strerror(-r));
    return EXIT_FAILURE;
  }
  /* Do some shutdown work here ... */
  return EXIT_SUCCESS;
}

Python 

#!/usr/bin/python
# SPDX-License-Identifier: MIT-0
#
# Implement the systemd notify protocol without external dependencies.
# Supports both readiness notification on startup and on reloading,
# according to the protocol defined at:
# https://www.freedesktop.org/software/systemd/man/latest/sd_notify.html
# This protocol is guaranteed to be stable as per:
# https://systemd.io/PORTABILITY_AND_STABILITY/
import errno
import os
import signal
import socket
import sys
import time
reloading = False
terminating = False
def notify(message):
    if not message:
        raise ValueError("notify() requires a message")
    socket_path = os.environ.get("NOTIFY_SOCKET")
    if not socket_path:
        return
    if socket_path[0] not in ("/", "@"):
        raise OSError(errno.EAFNOSUPPORT, "Unsupported socket type")
    # Handle abstract socket.
    if socket_path[0] == "@":
        socket_path = "\0" + socket_path[1:]
    with socket.socket(socket.AF_UNIX, socket.SOCK_DGRAM | socket.SOCK_CLOEXEC) as sock:
        sock.connect(socket_path)
        sock.sendall(message)
def notify_ready():
    notify(b"READY=1")
def notify_reloading():
    microsecs = time.clock_gettime_ns(time.CLOCK_MONOTONIC) // 1000
    notify(f"RELOADING=1\nMONOTONIC_USEC={microsecs}".encode())
def notify_stopping():
    notify(b"STOPPING=1")
def reload(signum, frame):
    global reloading
    reloading = True
def terminate(signum, frame):
    global terminating
    terminating = True
def main():
    print("Doing initial setup")
    global reloading, terminating
    # Set up signal handlers.
    print("Setting up signal handlers")
    signal.signal(signal.SIGHUP, reload)
    signal.signal(signal.SIGINT, terminate)
    signal.signal(signal.SIGTERM, terminate)
    # Do any other setup work here.
    # Once all setup is done, signal readiness.
    print("Done setting up")
    notify_ready()
    print("Starting loop")
    while not terminating:
        if reloading:
            print("Reloading")
            reloading = False
            # Support notifying the manager when reloading configuration.
            # This allows accurate state tracking as well as automatically
            # enabling 'systemctl reload' without needing to manually
            # specify an ExecReload= line in the unit file.
            notify_reloading()
            # Do some reconfiguration work here.
            print("Done reloading")
            notify_ready()
        # Do the real work here ...
        print("Sleeping for five seconds")
        time.sleep(5)
    print("Terminating")
    notify_stopping()
if __name__ == "__main__":
    sys.stdout.reconfigure(line_buffering=True)
    print("Starting app")
    main()
    print("Stopped app")

البيئة

$NOTIFY_SOCKET

يتم تعيينه بواسطة مدير الخدمة للعمليات الخاضعة للإشراف للحصول على حالة وإعلام اكتمال البدء. تحدد متغير البيئة هذا المقبس الذي تتحدث إليه sd_notify(). انظر أعلاه للتفاصيل.

أمثلة

مثال 1. إعلام بدء التشغيل

عندما تنتهي الخدمة من بدء التشغيل، قد تصدر الاستدعاء التالي لإعلام مدير الخدمة:

sd_notify(0, "READY=1");

مثال 2. إعلام بدء تشغيل موسع

يمكن للخدمة إرسال ما يلي بعد الانتهاء من التهيئة:

sd_notifyf(0, "READY=1\n"
              "STATUS=Processing requests...\n"
              "MAINPID=%lu",
           (unsigned long) getpid());

مثال 3. إعلام سبب الخطأ

يمكن للخدمة إرسال ما يلي قبل الخروج بوقت قصير، عند الفشل:

sd_notifyf(0, "STATUS=Failed to start up: %s\n"
              "ERRNO=%i",
           strerror_r(errnum, (char[1024]){}, 1024),
           errnum);

مثال 4. تخزين واصف ملف في مدير الخدمة

لتخزين واصف ملف مفتوح في مدير الخدمة، لمواصلة التشغيل بعد إعادة تشغيل الخدمة دون فقدان الحالة، استخدم "FDSTORE=1":

sd_pid_notify_with_fds(0, 0, "FDSTORE=1\nFDNAME=foobar", &fd, 1);

مثال 5. التخلص من حالات السباق

عندما لا يتم إنشاء العميل الذي يرسل الإشعارات بواسطة مدير الخدمة، فقد يخرج بسرعة كبيرة وقد يفشل مدير الخدمة في ربطها بشكل صحيح بالوحدة. لمنع مثل هذه السباقات، استخدم sd_notify_barrier() للمزامنة مع استقبال جميع الإشعارات المرسلة قبل استدعاء هذه المكالمة.

sd_notify(0, "READY=1");
/* set timeout to 5 seconds */
sd_notify_barrier(0, 5 * 1000000);
