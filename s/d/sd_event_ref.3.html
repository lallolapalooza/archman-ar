sd_event_ref(3)

الحزمة: systemd-libs

التراخيص: LGPL-2.1-or-later, CC0-1.0, GPL-2.0-or-later WITH Linux-syscall-note

SD_EVENT_NEW(3) sd_event_new SD_EVENT_NEW(3)

الاسم

sd_event_new, sd_event_default, sd_event_ref, sd_event_unref, sd_event_unrefp, sd_event_get_tid, sd_event - الحصول على كائن حلقة الأحداث وإلغاء الإشارة إليه

الملخص

#include <systemd/sd-event.h>
typedef struct sd_event sd_event;

int sd_event_new(sd_event **event);

int sd_event_default(sd_event **event);

sd_event *sd_event_ref(sd_event *event);

sd_event *sd_event_unref(sd_event *event);

void sd_event_unrefp(sd_event **event);

int sd_event_get_tid(sd_event *event, pid_t *ret);

الوصف

sd_event_new() تخصص كائن حلقة أحداث جديد. يتم إرجاع كائن حلقة الأحداث في المعامل event. بعد الاستخدام، قم بإسقاط الإشارة المرجعة باستخدام sd_event_unref(). عندما يتم إسقاط آخر إشارة، يتم تحرير الكائن.

sd_event_default() تحصل على إشارة إلى كائن حلقة الأحداث الافتراضي للثريد المتصل، مما قد يخصص كائنًا جديدًا إذا لم يتم تخصيص كائن حلقة أحداث افتراضي بعد للثريد. بعد الاستخدام، قم بإسقاط الإشارة المرجعة باستخدام sd_event_unref(). عندما يتم إسقاط آخر إشارة، يتم تحرير حلقة الأحداث. إذا تم استدعاء هذه الدالة بينما الكائن المرجع من استدعاء سابق من نفس الثريد لا يزال قيد الإشارة، يتم إرجاع نفس الكائن مرة أخرى، ولكن يتم زيادة الإشارة بواحد. يوصى باستخدام هذا الاستدعاء بدلاً من sd_event_new() لمشاركة كائنات حلقات الأحداث بين المكونات المختلفة التي يتم توزيعها في نفس الثريد. جميع الثريدات لديها بالضبط صفر أو واحد من كائنات حلقات الأحداث الافتراضية المرتبطة، ولا أكثر.

بعد تخصيص كائن حلقة أحداث، أضف مصادر الأحداث إليه باستخدام sd_event_add_io(3)، sd_event_add_time(3)، sd_event_add_signal(3)، sd_event_add_child(3)، sd_event_add_inotify(3)، sd_event_add_defer(3)، sd_event_add_post(3) أو sd_event_add_exit(3)، ثم قم بتنفيذ حلقة الأحداث باستخدام sd_event_loop(3).

sd_event_ref() تزيد عدد الإشارات للكائن حلقة الأحداث المحدد بمقدار واحد.

sd_event_unref() تقلل عدد الإشارات للكائن حلقة الأحداث المحدد بمقدار واحد. إذا وصل العدد إلى الصفر، يتم تحرير الكائن. لاحظ أنه يتم تحريره بغض النظر عما إذا كان كائن حلقة الأحداث الافتراضي لثريد أم لا. هذا يعني أن تخصيص حلقة أحداث باستخدام sd_event_default()، ثم تحريرها، ثم الحصول على حلقة جديدة باستخدام sd_event_default() سيؤدي إلى كائنين مختلفين. لاحظ أنه لتحرير كائن حلقة أحداث، يجب أيضًا تحرير جميع مصادر الأحداث المتبقية لحلقة الأحداث حيث أن كل منها يحتفظ بإشارة إليه.

sd_event_unrefp() مشابهة لـ sd_event_unref() ولكنها تأخذ مؤشرًا إلى مؤشر لكائن sd_event. هذه الدالة معرفة كدالة مضمنة. استخدم تعريفًا مثل التالي، لتخصيص كائن حلقة أحداث يتم تحريره تلقائيًا عند مغادرة كتلة التعليمات البرمجية:

{
        __attribute__((cleanup(sd_event_unrefp))) sd_event *event = NULL;
        int r;
        ...
        r = sd_event_default(&event);
        if (r < 0) {
          errno = -r;
          fprintf(stderr, "فشل تخصيص حلقة الأحداث: %m\n");
        }
        ...
}

sd_event_ref()، sd_event_unref() و sd_event_unrefp() لا تنفذ أي عملية إذا كان كائن حلقة الأحداث الممرر هو NULL.

sd_event_get_tid() تسترجع معرف الثريد (TID) للثريد الذي يرتبط به كائن حلقة الأحداث المحدد. هذه الدالة مدعومة فقط لحلقات الأحداث المخصصة باستخدام sd_event_default()، وترجع المعرف للثريد الذي تعد حلقة الأحداث هي الحلقة الافتراضية له. انظر gettid(2) لمزيد من المعلومات حول معرفات الثريدات.

قيمة الإرجاع

عند النجاح، ترجع sd_event_new()، sd_event_default() و sd_event_get_tid() 0 أو عددًا صحيحًا موجبًا. عند الفشل، ترجع رمز خطأ سالبًا من نوع errno. ترجع sd_event_ref() دائمًا مؤشرًا إلى كائن حلقة الأحداث الممرر. ترجع sd_event_unref() دائمًا NULL.

الأخطاء

قد تشير الأخطاء المرجعة إلى المشكلات التالية:

-ENOMEM

لا توجد ذاكرة كافية لتخصيص الكائن.

-EMFILE

تم تخصيص الحد الأقصى لعدد حلقات الأحداث.

-ENXIO

تم استدعاء sd_event_get_tid() على كائن حلقة أحداث لم يتم تخصيصه باستخدام sd_event_default().

ملاحظات

الوظائف الموصوفة هنا متاحة كمكتبة مشتركة، والتي يمكن تجميعها وربطها باستخدام ملف pkg-config(1) الخاص بـ libsystemd.

التعليمات البرمجية الموصوفة هنا تستخدم getenv(3)، والذي تم الإعلان عنه بأنه غير آمن للاستخدام المتزامن مع الثريدات. هذا يعني أن التعليمات البرمجية التي تستدعي الوظائف الموصوفة هنا يجب ألا تستدعي setenv(3) من ثريد متوازٍ. يوصى بإجراء استدعاءات لـ setenv() فقط من مرحلة مبكرة من البرنامج عندما لم يتم بدء أي ثريدات أخرى.
