real function sdot (integer n, real, dimension(*) sx, integer incx, real, dimension(*) sy, integer incy)

SDOT

الغرض:

!>
!>    SDOT تشكل المنتج النقطي لمتجهين.
!>    تستخدم حلقات غير ملفوفة للزيادات تساوي واحد.
!>

المعلمات

N 
!>          N هو عدد صحيح
!>         عدد العناصر في المتجه (المتجهات) المدخلة
!>

SX

!>          SX هو مصفوفة حقيقية، البعد ( 1 + ( N - 1 )*abs( INCX ) )
!>

INCX

!>          INCX هو عدد صحيح
!>         مسافة التخزين بين عناصر SX
!>

SY

!>          SY هو مصفوفة حقيقية، البعد ( 1 + ( N - 1 )*abs( INCY ) )
!>

INCY

!>          INCY هو عدد صحيح
!>         مسافة التخزين بين عناصر SY
!>
