الوصف

sd-id128.h هو جزء من libsystemd(3) ويوفر واجهات برمجة تطبيقات لتوليد وتحويل ومقارنة قيم معرفات 128 بت. قيم المعرفات 128 بت التي تتم معالجتها وتوليدها بواسطة واجهات برمجة التطبيقات هذه هي تعميم لـ OSF UUIDs كما هو محدد بواسطة RFC 4122[1] ولكنها تستخدم تنسيق سلسلة أبسط. لا تفرض هذه الوظائف أي هيكل على المعرفات المستخدمة، على عكس OSF UUIDs أو Microsoft GUIDs، ولكنها متوافقة إلى حد كبير مع هذه الأنواع من المعرفات.

يتم تنفيذ المعرف 128 بت على أنه نوع اتحاد التالي:

typedef union sd_id128 {
  uint8_t bytes[16];
  uint64_t qwords[2];
} sd_id128_t;

يسمح نوع الاتحاد هذا بالوصول إلى المعرف 128 بت كـ 16 بايت منفصلة أو كلمتين 64 بت. بشكل عام، يكون الوصول إلى مكونات المعرف بواسطة مصفوفة 8 بت الخاصة بها أكثر أمانًا لتجنب مشاكل ترتيب البايتات. يهدف هذا الاتحاد إلى أن يتم تمريره عن طريق القيمة (على عكس التمرير عن طريق المرجع) ويمكن تعديله مباشرة بواسطة العملاء.

تم تعريف بعض وحدات الماكرو للإشارة إلى وفك تشفير معرفات 128 بت:

SD_ID128_MAKE() يستخدم لكتابة معرف ثابت في شفرة المصدر. الاستخدام الشائع هو تعيين اسم لمعرف باستخدام هذه الماكرو:

#define SD_MESSAGE_COREDUMP SD_ID128_MAKE(fc,2e,22,bc,6e,e6,47,b6,b9,07,29,ab,34,a2,50,b1)

SD_ID128_NULL يعرف معرفًا يتكون فقط من بايتات NUL (أي جميع البتات مطفأة).

SD_ID128_ALLF يعرف معرفًا يتكون فقط من بايتات 0xFF (أي جميع البتات مضاءة).

SD_ID128_MAKE_STR() مشابه لـ SD_ID128_MAKE()، ولكنه ينشئ تعبير const char* يمكن استخدامه بسهولة في تنسيقات الرسائل وما شابه ذلك:

#include <stdio.h>
#define SD_MESSAGE_COREDUMP_STR SD_ID128_MAKE_STR(fc,2e,22,bc,6e,e6,47,b6,b9,07,29,ab,34,a2,50,b1)
int main(int argc, char **argv) {
  puts("Match for coredumps: MESSAGE_ID=" SD_MESSAGE_COREDUMP_STR);
}

SD_ID128_CONST_STR() تحول المعرفات الثابتة إلى سلاسل ثابتة للإخراج. سيقوم مثال الكود التالي بإخراج السلسلة "fc2e22bc6ee647b6b90729ab34a250b1":

int main(int argc, char *argv[]) {
  puts("Match for coredumps: %s", SD_ID128_CONST_STR(SD_MESSAGE_COREDUMP));
}

SD_ID128_FORMAT_STR و SD_ID128_FORMAT_VAL() يستخدمان لتنسيق معرف في سلسلة تنسيق printf(3)، كما هو موضح في المثال التالي:

int main(int argc, char *argv[]) {
  sd_id128_t id;
  id = SD_ID128_MAKE(ee,89,be,71,bd,6e,43,d6,91,e6,c5,5d,eb,03,02,07);
  printf("The ID encoded in this C file is " SD_ID128_FORMAT_STR ".\n", SD_ID128_FORMAT_VAL(id));
  return 0;
}

SD_ID128_UUID_FORMAT_STR و SD_ID128_MAKE_UUID_STR() مشابهان لـ SD_ID128_FORMAT_STR و SD_ID128_MAKE_STR()، ولكنهما يتضمنان واصلات فاصلة للامتثال لـ "UUID canonical representation[2]". يقومان بتنسيق السلسلة بناءً على قواعد RFC4122[1] Variant 1، أي التحويل من ترتيب بايت Big Endian. هذا يطابق سلوك معظم البنية التحتية الأخرى لنظام Linux. من الأفضل تجنب UUIDs من المتغيرات الأخرى، لتجنب الغموض غير الضروري. جميع المعرفات 128 بت التي تم إنشاؤها بواسطة واجهات برمجة تطبيقات sd-id128 تتوافق بشكل صارم مع Variant 1 Version 4 UUIDs، وفقًا لـ RFC 4122.

sd_id128_equal() تقارن معرفين 128 بت:

int main(int argc, char *argv[]) {
  sd_id128_t a, b, c;
  a = SD_ID128_MAKE(ee,89,be,71,bd,6e,43,d6,91,e6,c5,5d,eb,03,02,07);
  b = SD_ID128_MAKE(f2,28,88,9c,5f,09,44,15,9d,d7,04,77,58,cb,e7,3e);
  c = a;
  assert(sd_id128_equal(a, c));
  assert(!sd_id128_equal(a, b));
  return 0;
}

sd_id128_string_equal() مشابه لـ sd_id128_equal()، ولكن المعرف الأول يتم تنسيقه كسلسلة const char*. تنطبق نفس القيود كما في الوسيط الأول لـ sd_id128_from_string().

sd_id128_is_null() يتحقق مما إذا كان المعرف يتكون فقط من بايتات NUL:

assert(sd_id128_is_null(SD_ID128_NULL));

بالمثل، يتحقق sd_id128_is_allf() مما إذا كان المعرف يتكون فقط من بايتات 0xFF (جميع البتات مضاءة):

assert(sd_id128_is_allf(SD_ID128_ALLF));

sd_id128_in_set_sentinel() تأخذ قائمة من المعرفات وتعيد صحيحًا إذا كان الوسيط الأول يساوي أيًا من الوسائط اللاحقة. يتم إنهاء قائمة الوسائط بواسطة فاصل SD_ID128_NULL، والذي يجب أن يكون موجودًا.

sd_id128_in_set() هي دالة مساعدة تأخذ قائمة من المعرفات وتعيد صحيحًا إذا كان الوسيط الأول يساوي أيًا من الوسائط اللاحقة:

int main(int argc, char *argv[]) {
  sd_id12_t a = SD_ID128_MAKE(ee,89,be,71,bd,6e,43,d6,91,e6,c5,5d,eb,03,02,07);
  assert(sd_id128_in_set(a, a));
  assert(sd_id128_in_set(a, a, a));
  assert(!sd_id128_in_set(a));
  assert(!sd_id128_in_set(a,
                          SD_ID128_MAKE(f2,28,88,9c,5f,09,44,15,9d,d7,04,77,58,cb,e7,3e)
                          SD_ID128_MAKE(2f,88,28,5f,9c,44,09,9d,d7,15,77,04,bc,85,7e,e3)
                          SD_ID128_ALLF));
  return 0;
}

sd_id128_in_set() معرفة على أنها ماكرو فوق sd_id128_in_set_sentinel()، تضيف الفاصل SD_ID128_NULL تلقائيًا. نظرًا لأن sd_id128_in_set_sentinel() تستخدم SD_ID128_NULL كفاصل، لا يمكن وضع SD_ID128_NULL بشكل آخر في قائمة الوسائط.

sd_id128_in_setv() مشابه لـ sd_id128_in_set_sentinel()، ولكنه يأخذ بنية varargs كوسيط.

يمكن توليد معرفات عشوائية جديدة باستخدام الأمر new الخاص بـ systemd-id128(1).

انظر sd_id128_to_string(3)، sd_id128_randomize(3) و sd_id128_get_machine(3) للحصول على معلومات حول معلومات أخرى

ملاحظات

الوظائف الموصوفة هنا متاحة كمكتبة مشتركة، والتي يمكن الترجمة ضدها والربط بها باستخدام ملف libsystemd pkg-config(1).

الكود الموصوف هنا يستخدم getenv(3)، والذي تم تعريفه على أنه ليس آمنًا للاستخدام المتزامن بين الخيوط. هذا يعني أن الكود الذي يستدعي الوظائف الموصوفة هنا يجب ألا يستدعي setenv(3) من خيط موازٍ. يوصى بإجراء استدعاءات لـ setenv() فقط من مرحلة مبكرة من البرنامج عندما لم يتم بدء أي خيوط أخرى.
