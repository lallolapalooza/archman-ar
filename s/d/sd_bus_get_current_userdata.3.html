sd_bus_get_current_userdata(3)

الحزمة: systemd-libs

التراخيص: LGPL-2.1-or-later, CC0-1.0, GPL-2.0-or-later WITH Linux-syscall-note

SD_BUS_GET_CURRENT_HANDLER(3) sd_bus_get_current_handler SD_BUS_GET_CURRENT_HANDLER(3)

الاسم

sd_bus_get_current_handler, sd_bus_get_current_message, sd_bus_get_current_slot, sd_bus_get_current_userdata - الاستعلام عن معلومات الدالة الاستدعائية التي يعمل عليها كائن الناقل حاليًا

الاختصار

#include <systemd/sd-bus.h>

typedef int (*sd_bus_message_handler_t)(sd_bus_message *m, void *userdata, sd_bus_error *ret_error);

sd_bus_message_handler_t sd_bus_get_current_handler(sd_bus *bus);

sd_bus_message* sd_bus_get_current_message(sd_bus *bus);

sd_bus_slot* sd_bus_get_current_slot(sd_bus *bus);

void* sd_bus_get_current_userdata(sd_bus *bus);

الوصف

عندما يكون sd-bus على وشك استدعاء دالة استدعائية مقدمة من المستخدم، فإنه يخزن الدالة الاستدعائية الحالية، رسالة D-Bus، الفتحة، ومؤشر بيانات المستخدم ويسمح بالاستعلام عنها عبر sd_bus_get_current_handler()، sd_bus_get_current_message()، sd_bus_get_current_slot()، و sd_bus_get_current_userdata()، على التوالي. إذا تعذر حل bus أو إذا كان التنفيذ لا يقع ضمن دالة استدعائية مقدمة من المستخدم لـ bus، فإن هذه الدوال ترجع NULL.

قيمة الإرجاع

عند النجاح، ترجع هذه الدوال الكائن المطلوب. عند الفشل، ترجع NULL.

ملاحظات

الوظائف الموضحة هنا متاحة كمكتبة مشتركة، والتي يمكن تجميعها والربط بها باستخدام ملف pkg-config(1) الخاص بـ libsystemd.

الكود الموضح هنا يستخدم getenv(3)، والتي تم الإعلان عنها على أنها غير آمنة للعمليات المتوازية. هذا يعني أن الكود الذي يستدعي الوظائف الموضحة هنا يجب ألا يستدعي setenv(3) من خيط متوازي. يوصى بالقيام باستدعاءات لـ setenv() فقط من مرحلة مبكرة من البرنامج عندما لم يتم بدء خيوط أخرى بعد.
