sd_bus_track_add_name(3)

الحزمة: systemd-libs

التراخيص: LGPL-2.1-or-later, CC0-1.0, GPL-2.0-or-later WITH Linux-syscall-note

SD_BUS_TRACK_ADD_NAME(3) sd_bus_track_add_name SD_BUS_TRACK_ADD_NAME(3)

الاسم

sd_bus_track_add_name، sd_bus_track_add_sender، sd_bus_track_remove_name، sd_bus_track_remove_sender، sd_bus_track_count، sd_bus_track_count_sender، sd_bus_track_count_name، sd_bus_track_contains، sd_bus_track_first، sd_bus_track_next - إضافة، إزالة واسترجاع أقران الناقل المتعقبين في كائن تتبع أقران الناقل

الملخص

#include <systemd/sd-bus.h>

int sd_bus_track_add_name(sd_bus_track* t، const char* name);

int sd_bus_track_add_sender(sd_bus_track* t، sd_bus_message* message);

int sd_bus_track_remove_name(sd_bus_track* t، const char* name);

int sd_bus_track_remove_sender(sd_bus_track* t، sd_bus_message* message);

unsigned sd_bus_track_count(sd_bus_track* t);

int sd_bus_track_count_name(sd_bus_track* t، const char* name);

int sd_bus_track_count_sender(sd_bus_track* t، sd_bus_message* message);

int sd_bus_track_contains(sd_bus_track* t، const char* name);

const char* sd_bus_track_first(sd_bus_track* t);

const char* sd_bus_track_next(sd_bus_track* t);

الوصف

sd_bus_track_add_name() تضيف نديًا لتتبعه إلى كائن تتبع أقران الناقل. يجب أن يشير الوسيط الأول إلى كائن تتبع أقران الناقل الذي تم إنشاؤه بواسطة sd_bus_track_new(3)، ويجب أن يشير الاسم الثاني إلى اسم ندي D-Bus لتتبعه، سواء كان بتنسيق خدمة فريد أو معروف. إذا لم يكن الاسم قيد التتبع بالفعل، فسيتم إضافته إلى قائمة الأسماء المراد تتبعها. إذا كان قيد التتبع بالفعل وتم تمكين الوضع غير التكراري، فلن يتم تنفيذ أي عملية بواسطة هذا الاستدعاء. إذا تم تمكين الوضع التكراري، يتم زيادة عداد لكل اسم بمقدار واحد في كل مرة يتم فيها استدعاء هذا الاستدعاء، ويجب استدعاء sd_bus_track_remove_name() بنفس عدد المرات التي تم فيها استدعاء sd_bus_track_add_name() قبل إيقاف تتبع الاسم. استخدم sd_bus_track_set_recursive(3) للتبديل من الوضع الافتراضي غير التكراري إلى الوضع التكراري، أو العودة. لاحظ أن الاسم المحدد يتم تتبعه كما هو، ولا يتم حل الأسماء المعروفة إلى أسماء فريدة بواسطة هذا الاستدعاء. لاحظ أن كائنات تتبع أقران الناقل المتعددة قد تتتبع نفس الاسم.

sd_bus_track_remove_name() يلغي تأثير sd_bus_track_add_name() ويزيل اسم ندي الناقل من قائمة الأقران للمراقبة. اعتمادًا على ما إذا كان الوضع غير التكراري أو التكراري ممكّنًا لكائن تتبع أقران الناقل، سيقوم هذا الاستدعاء إما بإزالة الاسم بالكامل من كائن التتبع، أو سيقوم ببساطة بتقليل العداد لكل اسم بمقدار واحد، وإزالة الاسم فقط عندما يصل العداد إلى الصفر (انظر أعلاه). لاحظ أن ندي الناقل الذي ينفصل عن الناقل سيزيل ضمنيًا أسماءه بالكامل من كائن تتبع أقران الناقل، بغض النظر عن العداد الحالي لكل اسم.

sd_bus_track_add_sender() و sd_bus_track_remove_sender() مماثلة لـ sd_bus_track_add_name() و sd_bus_track_remove_name() ولكنها تأخذ رسالة ناقل كوسيط. يتم تحديد مرسل رسالة الناقل هذه وإضافته إلى/إزالته من كائن تتبع أقران الناقل. نظرًا لأن الرسائل تنشأ دائمًا من أسماء فريدة، ولا تنشأ أبدًا من أسماء معروفة، فهذا يعني أن هذا الاستدعاء سيضيف فقط الأسماء الفريدة إلى كائن تتبع أقران الناقل.

sd_bus_track_count() يعيد عدد الأسماء التي يتم تتبعها حاليًا بواسطة كائن تتبع أقران الناقل المحدد. لاحظ أن هذه الوظيفة تعيد دائمًا العدد الفعلي للأسماء المتتبعة، وبالتالي إذا تم استدعاء sd_bus_track_add_name() عدة مرات لنفس الاسم، فسيتم عدها مرة واحدة فقط، بغض النظر عما إذا كان الوضع التكراري مستخدمًا أم لا.

sd_bus_track_count_name() يعيد العداد الحالي لكل اسم للاسم المحدد. إذا تم استخدام الوضع غير التكراري، فسيعيد إما 1 أو 0، اعتمادًا على ما إذا كان الاسم المحدد قد تمت إضافته إلى كائن التتبع من قبل، أم لا. إذا تم تمكين الوضع التكراري، فقد يتم أيضًا إرجاع قيم أكبر من 1، في حالة استدعاء sd_bus_track_add_name() عدة مرات لنفس الاسم.

sd_bus_track_count_sender() مماثلة لـ sd_bus_track_count_name()، ولكنها تأخذ كائن رسالة ناقل وتعيد العداد لكل اسم مطابق لمرسل الرسالة.

sd_bus_track_contains() يمكن استخدامها لتحديد ما إذا كان الاسم المحدد قد تمت إضافته مرة واحدة على الأقل إلى كائن تتبع أقران الناقل المحدد.

sd_bus_track_first() و sd_bus_track_next() يمكن استخدامها لتعداد جميع الأسماء التي يتم تتبعها حاليًا بواسطة كائن تتبع أقران الناقل الممرر. sd_bus_track_first() يعيد الإدخال الأول في الكائن، ويعيد تعيين مؤشر قراءة داخلي. كل استدعاء لاحق لـ sd_bus_track_next() يعيد الاسم التالي الموجود في كائن الناقل. إذا تم الوصول إلى النهاية، يتم إرجاع NULL. إذا لم تتم إضافة أي أسماء إلى الكائن بعد، فستعيد sd_bus_track_first() NULL على الفور. ترتيب الأسماء التي يتم إرجاعها غير محدد؛ على وجه الخصوص، الاسم الذي يعتبر الأول المعروض غير محدد. إذا تم تمكين الوضع التكراري وتمت إضافة نفس الاسم عدة مرات إلى كائن تتبع أقران الناقل، فسيتم إرجاعه مرة واحدة فقط بواسطة هذا التعداد. إذا تمت إضافة أسماء جديدة أو إزالة أسماء موجودة من كائن تتبع أقران الناقل أثناء التعداد، ينتهي التعداد عند الاستدعاء التالي لـ sd_bus_track_next() حيث يتم إرجاع NULL.

قيمة الإرجاع

عند النجاح، تعيد sd_bus_track_add_name() و sd_bus_track_add_sender() القيمة 0 إذا تمت إضافة الاسم المحدد بالفعل إلى كائن تتبع أقران الناقل من قبل، وقيمة موجبة إذا لم يتم إضافته. عند الفشل، تعيد رمز خطأ سالبًا بأسلوب errno.

sd_bus_track_remove_name() و sd_bus_track_remove_sender() تعيد قيمة موجبة إذا تم تتبع الاسم المحدد سابقًا بواسطة كائن تتبع أقران الناقل وتمت إزالته الآن. في الوضع غير التكراري، يتم إرجاع 0 إذا لم يتم تتبع الاسم المحدد بعد. في الوضع التكراري، يتم إرجاع -EUNATCH في هذه الحالة. عند الفشل، تعيد رمز خطأ سالبًا بأسلوب errno.

sd_bus_track_count() تعيد عدد الأسماء التي يتم تتبعها حاليًا، أو 0 عند الفشل.

sd_bus_track_count_name() و sd_bus_track_count_sender() تعيد العداد الحالي لكل اسم للاسم المحدد أو لمرسل الرسالة المحددة. يتم إرجاع صفر للأسماء التي لم يتم تتبعها بعد، وقيمة موجبة للأسماء التي تمت إضافتها مرة واحدة على الأقل. القيم الأكبر من 1 يتم إرجاعها فقط في الوضع التكراري. عند الفشل، يتم إرجاع رمز خطأ سالب بأسلوب errno.

sd_bus_track_contains() تعيد الاسم الممرر إذا كان موجودًا في كائن تتبع أقران الناقل. عند الفشل، وإذا لم تتم إضافة الاسم بعد، يتم إرجاع NULL.

sd_bus_track_first() و sd_bus_track_next() تعيد الاسم الأول/التالي الموجود في كائن تتبع أقران الناقل، و NULL إذا تم الوصول إلى نهاية التعداد وعند حدوث خطأ.

الأخطاء

قد تشير الأخطاء التي تم إرجاعها إلى المشاكل التالية:

-EUNATCH

تم استدعاء sd_bus_track_remove_name() أو sd_bus_track_remove_sender() لاسم لم تتم إضافته سابقًا إلى كائن الناقل.

-EINVAL

المعلمة المحددة غير صالحة.

-ENOMEM

فشل تخصيص الذاكرة.
