sd_journal_query_unique(3)

الحزمة: systemd-libs

التراخيص: LGPL-2.1-or-later, CC0-1.0, GPL-2.0-or-later WITH Linux-syscall-note

SD_JOURNAL_QUERY_UNIQUE(3) sd_journal_query_unique SD_JOURNAL_QUERY_UNIQUE(3)

الاسم

sd_journal_query_unique, sd_journal_enumerate_unique, sd_journal_enumerate_available_unique, sd_journal_restart_unique, SD_JOURNAL_FOREACH_UNIQUE - قراءة حقول بيانات فريدة من سجل النظام

الملخص

#include <systemd/sd-journal.h>

int sd_journal_query_unique(sd_journal *j, const char *field);

int sd_journal_enumerate_available_unique(sd_journal *j, const void **data, size_t *length);

int sd_journal_enumerate_unique(sd_journal *j, const void **data, size_t *length);

void sd_journal_restart_unique(sd_journal *j);

SD_JOURNAL_FOREACH_UNIQUE(sd_journal *j, const void *data, size_t length);

الوصف

تقوم الدالة sd_journal_query_unique() بالبحث في سجل النظام عن جميع القيم الفريدة التي يمكن أن يأخذها الحقل المحدد. تأخذ وسيطتين: سجل النظام للبحث فيه واسم الحقل المطلوب. الأسماء المعروفة للحقول مدرجة في systemd.journal-fields(7)، ولكن يمكن تحديد أي حقل. يجب تحديد أسماء الحقول بدون علامة "=" في نهايتها. بعد تنفيذ هذه الدالة بنجاح، يمكن الاستعلام عن قيم الحقول باستخدام sd_journal_enumerate_unique() و sd_journal_enumerate_available_unique(). سيؤدي استدعاء أحد هذه الاستدعاءات إلى تغيير اسم الحقل الذي يتم الاستعلام عنه وإعادة تعيين فهرس التعداد إلى أول قيمة حقل مطابقة.

يمكن استخدام الدالة sd_journal_enumerate_unique() للتكرار عبر جميع حقول البيانات التي تطابق اسم الحقل المحدد مسبقًا باستخدام sd_journal_query_unique(). في كل استدعاء، يتم إرجاع بيانات الحقل التالية المطابقة. ترتيب حقول البيانات المرجعة غير محدد. تأخذ ثلاث وسيطات: كائن سجل النظام، بالإضافة إلى زوج من المؤشرات إلى مؤشرات / متغيرات حجم حيث يجب تخزين كائن البيانات وحجمه. البيانات المرجعة موجودة في خريطة ذاكرة للقراءة فقط وهي صالحة فقط حتى الاستدعاء التالي لـ sd_journal_enumerate_unique(). لاحظ أن البيانات المرجعة ستكون مسبوقة باسم الحقل و "=". لاحظ أن هذا الاستدعاء يخضع للحد الأقصى لحجم حقل البيانات الذي يتم التحكم فيه بواسطة sd_journal_set_data_threshold() ويتم إرجاع الجزء الأولي من الحقل فقط حتى الحد الأقصى. يتم إرجاع خطأ للحقول التي لا يمكن استرجاعها. انظر قائمة الأخطاء أدناه للحصول على التفاصيل.

sd_journal_enumerate_available_unique() مشابهة لـ sd_journal_enumerate_unique()، ولكنها تتخطى بصمت أي حقول قد تكون صالحة، ولكنها كبيرة جدًا أو غير مدعومة بالتنفيذ الحالي.

تعمل الدالة sd_journal_restart_unique() على إعادة تعيين فهرس تعداد البيانات إلى بداية القائمة. الاستدعاء التالي لـ sd_journal_enumerate_unique() سيعيد بيانات الحقل الأولى المطابقة للحقل مرة أخرى.

لاحظ أن الماكرو SD_JOURNAL_FOREACH_UNIQUE() يمكن استخدامه كغلاف مفيد حول sd_journal_restart_unique() و sd_journal_enumerate_available_unique().

لاحظ أن هذه الدوال حاليًا لا تتأثر بالمطابقات التي تم تعيينها باستخدام sd_journal_add_match() ولكن هذا قد يتغير في إصدار لاحق من هذا البرنامج.

لتعداد جميع أسماء الحقول المستخدمة حاليًا (وبالتالي جميع معلمات الحقول المناسبة لـ sd_journal_query_unique())، استخدم استدعاء sd_journal_enumerate_fields(3).

قيمة الإرجاع

sd_journal_query_unique() ترجع 0 عند النجاح أو رمز خطأ سلبي من نوع errno. sd_journal_enumerate_unique() و sd_journal_query_available_unique() ترجعان عددًا صحيحًا موجبًا إذا تم قراءة بيانات الحقل التالية، 0 عندما لا تبقى حقول أخرى، أو رمز خطأ سلبي من نوع errno. sd_journal_restart_unique() لا ترجع شيئًا.

-E2BIG

تمت الإضافة في الإصدار 246.

-EPROTONOSUPPORT

تم ضغط سجل النظام بطريقة غير مدعومة أو يستخدم سجل النظام ميزة غير مدعومة.

-EBADMSG

تمت الإضافة في الإصدار 246.

-EIO

تم الإبلاغ عن خطأ إدخال / إخراج بواسطة النواة.

تمت الإضافة في الإصدار 246.

أمثلة

استخدم الماكرو SD_JOURNAL_FOREACH_UNIQUE() للتكرار عبر جميع القيم التي يمكن أن يأخذها حقل في سجل النظام (والتي يمكن الوصول إليها على البنية الحالية وليست مضغوطة بآلية غير مدعومة). المثال التالي يسرد جميع أسماء الوحدات المشار إليها في سجل النظام:

/* SPDX-License-Identifier: MIT-0 */
#include <errno.h>
#include <stdio.h>
#include <systemd/sd-journal.h>
int main(int argc, char *argv[]) {
  sd_journal *j;
  const void *d;
  size_t l;
  int r;
  r = sd_journal_open(&j, SD_JOURNAL_LOCAL_ONLY);
  if (r < 0) {
    fprintf(stderr, "فشل فتح سجل النظام: %s\n", strerror(-r));
    return 1;
  }
  r = sd_journal_query_unique(j, "_SYSTEMD_UNIT");
  if (r < 0) {
    fprintf(stderr, "فشل الاستعلام عن سجل النظام: %s\n", strerror(-r));
    return 1;
  }
  SD_JOURNAL_FOREACH_UNIQUE(j, d, l)
    printf("%.*s\n", (int) l, (const char*) d);
  sd_journal_close(j);
  return 0;
}

sd_journal_query_unique(3)

الاسم

sd_journal_query_unique - الحصول على قيم فريدة للحقل من سجل النظام

الملخص

#include <systemd/sd-journal.h>

int sd_journal_query_unique(sd_journal *j, const char *field,
                            sd_journal_enumerator *enumerator);

الوصف

تُستخدم الدالة sd_journal_query_unique() للحصول على قائمة بالقيم الفريدة لحقل معين من سجل النظام. تأخذ مؤشرًا إلى بنية sd_journal (تم الحصول عليها بواسطة sd_journal_open())، واسم الحقل الذي تريد الاستعلام عنه، ومؤشرًا إلى بنية sd_journal_enumerator (تم الحصول عليها بواسطة sd_journal_enumerate_init()).

سيتم ملء بنية sd_journal_enumerator بالقيم الفريدة للحقل المحدد. يمكنك بعد ذلك استخدام sd_journal_enumerate_next() للتكرار فوق هذه القيم.

يجب أن يتم تهيئة sd_journal_enumerator باستخدام sd_journal_enumerate_init() قبل استدعاء sd_journal_query_unique(). بعد الانتهاء من استخدام القيم، يجب تحرير sd_journal_enumerator باستخدام sd_journal_enumerate_reset().

قيمة الإرجاع

عند النجاح، تُرجع الدالة sd_journal_query_unique() قيمة غير سالبة. عند الفشل، تُرجع قيمة سالبة تشير إلى خطأ.

  • < 0: عند حدوث خطأ.

ملاحظات

يجب أن يكون سجل النظام متاحًا للقراءة. قد لا تكون جميع الحقول متاحة للاستعلام عن قيمها الفريدة.

أمثلة

#include <stdio.h>
#include <systemd/sd-journal.h>

int main(void) {
    sd_journal *j = NULL;
    sd_journal_enumerator *enumerator = NULL;
    const char *field = "PRIORITY";
    int ret;

    ret = sd_journal_open(&j, SD_JOURNAL_READ_ANY);
    if (ret < 0) {
        fprintf(stderr, "Failed to open journal: %s\n", strerror(-ret));
        return 1;
    }

    ret = sd_journal_enumerate_init(&enumerator);
    if (ret < 0) {
        fprintf(stderr, "Failed to initialize enumerator: %s\n", strerror(-ret));
        sd_journal_close(j);
        return 1;
    }

    ret = sd_journal_query_unique(j, field, enumerator);
    if (ret < 0) {
        fprintf(stderr, "Failed to query unique values for '%s': %s\n", field, strerror(-ret));
        sd_journal_enumerate_reset(enumerator);
        sd_journal_close(j);
        return 1;
    }

    printf("Unique values for '%s':\n", field);
    while ((ret = sd_journal_enumerate_next(enumerator)) > 0) {
        printf("- %s\n", sd_journal_enumerator_get_value(enumerator));
    }

    if (ret < 0) {
        fprintf(stderr, "Error during enumeration: %s\n", strerror(-ret));
    }

    sd_journal_enumerate_reset(enumerator);
    sd_journal_close(j);

    return 0;
}
