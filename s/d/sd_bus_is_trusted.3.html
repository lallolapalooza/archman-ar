sd_bus_is_trusted(3)

الحزمة: systemd-libs

التراخيص: LGPL-2.1-or-later, CC0-1.0, GPL-2.0-or-later WITH Linux-syscall-note

الاسم

الملخص

#include <systemd/sd-bus.h>

int sd_bus_set_description(sd_bus *bus, const char *description);

int sd_bus_get_description(sd_bus *bus, const char **description);

int sd_bus_set_anonymous(sd_bus *bus, int b);

int sd_bus_is_anonymous(sd_bus *bus);

int sd_bus_set_trusted(sd_bus *bus, int b);

int sd_bus_is_trusted(sd_bus *bus);

int sd_bus_set_allow_interactive_authorization(sd_bus *bus, int b);

int sd_bus_get_allow_interactive_authorization(sd_bus *bus);

int sd_bus_get_scope(sd_bus *bus, const char **scope);

int sd_bus_get_tid(sd_bus *bus, pid_t *tid);

int sd_bus_get_unique_name(sd_bus *bus, const char **unique);

الوصف

sd_bus_set_description() تقوم بتعيين سلسلة الوصف التي تستخدم في التسجيل إلى السلسلة المحددة. يتم نسخ السلسلة داخليًا وتحريرها عند إلغاء تخصيص كائن الناقل. قد يكون الوسيط description هو NULL، وفي هذه الحالة يتم إلغاء تعيين الوصف. يجب استدعاء هذه الدالة قبل بدء تشغيل الناقل.

sd_bus_get_description() تُرجع سلسلة وصف في description. قد تكون هذه السلسلة قد تم تعيينها مسبقًا باستخدام sd_bus_set_description() أو sd_bus_open_with_description(3) أو ما شابه. إذا لم يتم تعيينها بهذه الطريقة، فسيتم إرجاع سلسلة افتراضية مثل "system" أو "user" للناقلات النظامية أو للمستخدم، و -ENXIO بخلاف ذلك.

sd_bus_set_anonymous() تمكن أو تعطيل "المصادقة المجهولة"، أي عدم وجود مصادقة، للطرف الآخر في الناقل. يجب استدعاء هذه الدالة قبل بدء تشغيل الناقل. راجع قسم آليات مصادقة D-Bus[1] في مواصفات D-Bus للحصول على التفاصيل.

sd_bus_is_anonymous() تُرجع صحيحًا إذا كان اتصال الناقل يسمح بالمصادقة المجهولة (بالمعنى الموصوف في الفقرة السابقة).

sd_bus_set_trusted() تقوم بتعيين حالة "موثوق" على الكائن bus. إذا كانت صحيحة، فإن جميع الاتصالات على الناقل موثوق بها ويتم منح الوصول إلى جميع الطرق المميزة وغير المميزة. يجب استدعاء هذه الدالة قبل بدء تشغيل الناقل.

sd_bus_is_trusted() تُرجع صحيحًا إذا كان اتصال الناقل موثوقًا به (بالمعنى الموصوف في الفقرة السابقة).

sd_bus_set_allow_interactive_authorization() تمكن أو تعطيل التفويض التفاعلي لاستدعاءات الطرق. إذا كانت صحيحة، يتم تمييز الرسائل بعلامة ALLOW_INTERACTIVE_AUTHORIZATION المحددة بواسطة مواصفات D-Bus[2]، لإعلام الجانب المستقبل بأن المتصل مستعد للانتظار للحصول على التفويض التفاعلي، والذي قد يستغرق وقتًا طويلاً لإكماله. إذا تم تعيين هذه العلامة، قد يُطلب من المستخدم كلمات مرور أو تأكيد عبر polkit[3] أو إطار عمل مماثل.

sd_bus_get_allow_interactive_authorization() تُرجع صحيحًا إذا كان التفويض التفاعلي مسموحًا به وخطأ إذا لم يكن كذلك.

sd_bus_get_scope() تخزن نطاق كائن الناقل المعطى في scope. نطاق ناقل النظام هو "system". نطاق ناقل جلسة المستخدم هو "user". إذا لم يكن كائن الناقل المعطى ناقل نظام أو ناقل جلسة مستخدم، فإن sd_bus_get_scope() تُرجع خطأ.

sd_bus_get_tid() تخزن معرف مؤشر ترابط النواة للمؤشر المرتبط بكائن الناقل المعطى في tid. إذا كان bus كائن ناقل افتراضي تم الحصول عليه عن طريق استدعاء إحدى دوال عائلة sd_bus_default(3)، فإنه يخزن معرف مؤشر الترابط الذي تم إنشاء كائن الناقل فيه. بخلاف ذلك، إذا كان كائن الناقل مرتبطًا بحلقة أحداث، فإنه يخزن معرف مؤشر الترابط الذي تم إنشاء كائن حلقة الأحداث فيه. إذا لم يكن bus كائن ناقل افتراضي ولم يكن مرتبطًا بحلقة أحداث، فإن sd_bus_get_tid() تُرجع خطأ.

sd_bus_get_unique_name() تخزن الاسم الفريد لكائن الناقل على الناقل في unique. راجع مواصفات D-Bus[4] لمزيد من المعلومات حول أسماء الناقلات. لاحظ أن المتصل لا يمتلك السلسلة المخزنة في unique ويجب ألا يقوم بتحريرها.

قيمة الإرجاع

عند النجاح، تُرجع هذه الدوال عددًا صحيحًا غير سالب. عند الفشل، تُرجع رمز خطأ سالبًا بنمط errno.

الأخطاء

قد تشير الأخطاء المرجعة إلى المشاكل التالية:

-EINVAL

وسيط غير صالح.

-ENOPKG

لا يمكن حل الناقل.

-EPERM

تم تشغيل الناقل بالفعل.

-ECHILD

تم إنشاء الناقل في عملية أو مكتبة أو وحدة أخرى.

-ENOMEM

فشل تخصيص الذاكرة.

-ENODATA

لم يكن كائن الناقل الذي تم تمريره إلى sd_bus_get_scope() ناقل نظام أو ناقل جلسة مستخدم.

تمت الإضافة في الإصدار 246.

-ENXIO

لم يكن كائن الناقل الذي تم تمريره إلى sd_bus_get_tid() كائن ناقل افتراضي ولم يكن مرتبطًا بحلقة أحداث.

لم يكن كائن الناقل الذي تم تمريره إلى sd_bus_get_description() يحتوي على description.

تمت الإضافة في الإصدار 246.

ملاحظات

الوظائف الموصوفة هنا متاحة كمكتبة مشتركة، يمكن تجميعها وربطها بملف libsystemd pkg-config(1).

الكود الموصوف هنا يستخدم getenv(3)، والتي تم الإعلان عنها على أنها غير آمنة للاستخدام المتزامن. هذا يعني أن الكود الذي يستدعي الوظائف الموصوفة هنا يجب ألا يستدعي setenv(3) من مؤشر ترابط متوازي. يوصى بإجراء استدعاءات لـ setenv() فقط من مرحلة مبكرة من البرنامج عندما لم يتم بدء مؤشرات ترابط أخرى.
