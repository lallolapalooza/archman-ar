sd_journal_previous_skip(3)

الحزمة: systemd-libs

التراخيص: LGPL-2.1-or-later, CC0-1.0, GPL-2.0-or-later WITH Linux-syscall-note

SD_JOURNAL_NEXT(3) sd_journal_next SD_JOURNAL_NEXT(3)

الاسم

sd_journal_next, sd_journal_previous, sd_journal_step_one, sd_journal_next_skip, sd_journal_previous_skip, SD_JOURNAL_FOREACH, SD_JOURNAL_FOREACH_BACKWARDS - تقدم أو ارجع مؤشر القراءة في السجل

المُلخص

#include <systemd/sd-journal.h>

int sd_journal_next(sd_journal *j);

int sd_journal_previous(sd_journal *j);

int sd_journal_step_one(sd_journal *j, int advanced);

int sd_journal_next_skip(sd_journal *j, uint64_t skip);

int sd_journal_previous_skip(sd_journal *j, uint64_t skip);

SD_JOURNAL_FOREACH(sd_journal *j);

SD_JOURNAL_FOREACH_BACKWARDS(sd_journal *j);

الوصف

sd_journal_next() تقدم مؤشر القراءة في السجل بمقدار إدخال واحد. الوسيط الوحيد الذي تأخذه هو كائن سياق السجل كما تم تخصيصه عبر sd_journal_open(3). بعد الاستدعاء الناجح، يمكن قراءة الإدخال باستخدام دوال مثل sd_journal_get_data(3).

بالمثل، sd_journal_previous() ترجع مؤشر القراءة بمقدار إدخال واحد.

sd_journal_step_one() تحرك أيضًا مؤشر القراءة. إذا كان الموقع الحالي هو رأس السجل، على سبيل المثال عند استدعاء هذه الدالة بعد sd_journal_seek_head(3)، فإنها تكون مكافئة لـ sd_journal_next()، وسيتم تجاهل الوسيط advanced. وبالمثل، إذا كان الموقع الحالي هو نهاية السجل، على سبيل المثال عند استدعاء هذه الدالة بعد sd_journal_seek_tail(3)، فإنها تكون مكافئة لـ sd_journal_previous()، وسيتم تجاهل advanced. بخلاف ذلك، فهي مكافئة لـ sd_journal_next() عندما تكون advanced غير صفرية، و sd_journal_previous() عندما تكون advanced صفرية.

sd_journal_next_skip() و sd_journal_previous_skip() تقدمان / ترجعان مؤشر القراءة بعدة إدخالات في وقت واحد، كما هو محدد في المعامل skip. يجب أن يكون المعامل skip أقل من أو يساوي 2147483647 (2³¹-1).

السجل مرتب ترتيبًا صارمًا حسب وقت الاستقبال، وبالتالي فإن التقدم إلى الإدخال التالي يضمن أن الإدخال الذي يشير إليه بعد ذلك هو أحدث زمنيًا من الإدخال السابق، أو له نفس الطابع الزمني.

لاحظ أن sd_journal_get_data(3) والدوال ذات الصلة ستفشل ما لم يتم استدعاء sd_journal_next() مرة واحدة على الأقل لوضع مؤشر القراءة على إدخال سجل.

لاحظ أنه يمكن استخدام الماكرو SD_JOURNAL_FOREACH() كغلاف حول sd_journal_seek_head(3) و sd_journal_next() لجعل التكرار عبر السجل أسهل. انظر أدناه للحصول على مثال. وبالمثل، يمكن استخدام SD_JOURNAL_FOREACH_BACKWARDS() للتكرار عبر السجل بترتيب عكسي.

قيمة الإرجاع

تعيد الاستدعاءات الأربعة عدد الإدخالات التي تم تقديمها/إرجاعها بنجاح أو رمز خطأ سالب بنمط errno. عند الوصول إلى نهاية السجل أو بدايته، يتم إرجاع عدد أقل من المطلوب. وبشكل أكثر تحديدًا، إذا وصل sd_journal_next() أو sd_journal_previous() إلى نهاية/بداية السجل، فسيتم إرجاع 0، بدلاً من 1 عند نجاحهما. يجب اعتبار ذلك علامة نهاية الملف.

ملاحظات

جميع الدوال المذكورة هنا مستقلة عن الخيوط، ويمكن لخيط واحد فقط العمل على كائن معين في أي وقت. قد تصل الخيوط المختلفة إلى نفس الكائن في أوقات مختلفة. يمكن استخدام كائنات مستقلة متعددة من خيوط مختلفة بالتوازي.

الدوال الموصوفة هنا متاحة كمكتبة مشتركة، والتي يمكن تجميعها والربط بها باستخدام ملف libsystemd pkg-config(1).

أمثلة

التكرار عبر السجل:

/* SPDX-License-Identifier: MIT-0 */
#include <errno.h>
#include <stdio.h>
#include <systemd/sd-journal.h>
int main(int argc, char *argv[]) {
  int r;
  sd_journal *j;
  r = sd_journal_open(&j, SD_JOURNAL_LOCAL_ONLY);
  if (r < 0) {
    fprintf(stderr, "فشل فتح السجل: %s\n", strerror(-r));
    return 1;
  }
  SD_JOURNAL_FOREACH(j) {
    const char *d;
    size_t l;
    r = sd_journal_get_data(j, "MESSAGE", (const void **)&d, &l);
    if (r < 0) {
      fprintf(stderr, "فشل قراءة حقل الرسالة: %s\n", strerror(-r));
      continue;
    }
    printf("%.*s\n", (int) l, d);
  }
  sd_journal_close(j);
  return 0;
}
