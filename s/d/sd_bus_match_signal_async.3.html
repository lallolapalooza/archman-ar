sd_bus_match_signal_async(3)

الحزمة: systemd-libs

التراخيص: LGPL-2.1-or-later, CC0-1.0, GPL-2.0-or-later WITH Linux-syscall-note

SD_BUS_ADD_MATCH(3) sd_bus_add_match SD_BUS_ADD_MATCH(3)

الاسم

sd_bus_add_match, sd_bus_add_match_async, sd_bus_match_signal, sd_bus_match_signal_async - إضافة قاعدة مطابقة لتوزيع الرسائل الواردة

الملخص

#include <systemd/sd-bus.h>

typedef int (*sd_bus_message_handler_t)(sd_bus *m, void *userdata, sd_bus_error *ret_error);

int sd_bus_add_match(sd_bus *bus, sd_bus_slot **slot, const char *match, sd_bus_message_handler_t callback, void *userdata);

int sd_bus_add_match_async(sd_bus *bus, sd_bus_slot **slot, const char *match, sd_bus_message_handler_t callback, sd_bus_message_handler_t install_callback, void *userdata);

int sd_bus_match_signal(sd_bus *bus, sd_bus_slot **slot, const char *sender, const char *path, const char *interface, const char *member, sd_bus_message_handler_t callback, void *userdata);

int sd_bus_match_signal_async(sd_bus *bus, sd_bus_slot **slot, const char *sender, const char *path, const char *interface, const char *member, sd_bus_message_handler_t callback, sd_bus_message_handler_t install_callback, void *userdata);

الوصف

sd_bus_add_match() تثبت قاعدة مطابقة للرسائل المستلمة على كائن اتصال الناقل المحدد bus. صيغة تعبير قاعدة المطابقة الممررة في match موصوفة في مواصفات D-Bus[1]. يتم استدعاء الدالة المعالجة المحددة callback لكل رسالة واردة تطابق التعبير المحدد، ويتم تمرير المعلمة userdata كما هي إلى الدالة المعالجة. يتم تثبيت المطابقة بشكل متزامن عند الاتصال بوسيط الناقل، أي أن المكالمة ترسل رسالة تحكم تطلب إضافة المطابقة إلى الوسيط وتنتظر حتى يؤكد الوسيط بنجاح تثبيت المطابقة.

sd_bus_add_match_async() تعمل بشكل مشابه جداً لـ sd_bus_add_match()، ولكنها تثبت المطابقة بشكل غير متزامن، بطريقة غير معوقة: يتم إرسال طلب إلى الوسيط، ولكن المكالمة لا تنتظر رداً. يتم استدعاء الدالة install_callback عند استلام الرد لاحقًا، مع رسالة الرد من الوسيط كمعلمة. إذا تم تحديد هذه الدالة كـ NULL، يتم استخدام تطبيق افتراضي ينهي اتصال الناقل إذا فشل تثبيت المطابقة.

sd_bus_match_signal() مشابهة جداً لـ sd_bus_add_match()، ولكنها تطابق الإشارات فقط، وبدلاً من تعبير المطابقة تقبل أربع معلمات: sender (اسم الخدمة للمرسل)، path (مسار الكائن للكائن الباعث)، interface (الواجهة التي تنتمي إليها الإشارة)، member (اسم الإشارة)، والتي يتم من خلالها إنشاء سلسلة المطابقة داخليًا. اختياريًا، يمكن تحديد هذه المعلمات كـ NULL، وفي هذه الحالة لا يتم اختبار الحقل ذي الصلة للإشارات الواردة.

sd_bus_match_signal_async() تجمع منطق مطابقة الإشارات لـ sd_bus_match_signal() مع السلوك غير المتزامن لـ sd_bus_add_match_async().

عند النجاح، وإذا لم تكن NULL، فسيتم تعيين المعلمة المرجعة slot إلى كائن فتحة يمكن استخدامه كمرجع للمطابقة المثبتة، ويمكن استخدامه لإزالته مرة أخرى لاحقًا باستخدام sd_bus_slot_unref(3). إذا تم تحديده كـ NULL، فإن عمر المطابقة مقيد بعمر كائن الناقل نفسه، وعادةً لا تتم إزالة المطابقة بشكل مستقل. انظر sd_bus_slot_set_floating(3) للتفاصيل.

الرسالة m الممررة إلى الدالة المعالجة مستعارة فقط، أي أنه لا ينبغي للدالة المعالجة استدعاء sd_bus_message_unref(3) عليها. إذا أرادت الدالة المعالجة الاحتفاظ بالرسالة بعد انتهاء عمر الدالة المعالجة، فإنها تحتاج إلى استدعاء sd_bus_message_ref(3) لإنشاء مرجع جديد.

إذا حدث خطأ أثناء استدعاء الدالة المعالجة، يجب على الدالة المعالجة إرجاع رقم خطأ سالب (اختياريًا، يمكن إرجاع خطأ أكثر دقة في ret_error، كذلك). إذا أرادت استدعاء دوال معالجة أخرى تطابق نفس القاعدة، فيجب عليها إرجاع 0. بخلاف ذلك، يجب عليها إرجاع عدد صحيح موجب.

إذا كان bus يشير إلى اتصال مباشر (أي ليس اتصال ناقل، كما تم تعيينه باستخدام sd_bus_set_bus_client(3)) يتم تثبيت المطابقة فقط على جانب العميل، وتعمل الدوال المتزامنة وغير المتزامنة بنفس الطريقة.

قيمة الإرجاع

عند النجاح، ترجع sd_bus_add_match() والمكالمات الأخرى 0 أو عددًا صحيحًا موجبًا. عند الفشل، ترجع رمز خطأ سالب على نمط errno.

ملاحظات

الدوال الموصوفة هنا متاحة كمكتبة مشتركة، والتي يمكن تجميعها وربطها باستخدام ملف pkg-config(1) الخاص بـ libsystemd.

الكود الموصوف هنا يستخدم getenv(3)، والذي تم تعريفه على أنه غير آمن للاستخدام المتعدد الخيوط. هذا يعني أن الكود الذي يستدعي الدوال الموصوفة هنا يجب ألا يستدعي setenv(3) من خيط موازٍ. يوصى بإجراء استدعاءات لـ setenv() فقط من مرحلة مبكرة من البرنامج عندما لم يتم بدء أي خيوط أخرى.
