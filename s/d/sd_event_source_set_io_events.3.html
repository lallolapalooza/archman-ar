sd_event_source_set_io_events(3)

الحزمة: systemd-libs

التراخيص: LGPL-2.1-or-later, CC0-1.0, GPL-2.0-or-later WITH Linux-syscall-note

SD_EVENT_ADD_IO(3) sd_event_add_io SD_EVENT_ADD_IO(3)

الاسم

sd_event_add_io, sd_event_source_get_io_events, sd_event_source_set_io_events, sd_event_source_get_io_revents, sd_event_source_get_io_fd, sd_event_source_set_io_fd, sd_event_source_get_io_fd_own, sd_event_source_set_io_fd_own, sd_event_source, sd_event_io_handler_t - إضافة مصدر حدث إدخال/إخراج إلى حلقة أحداث

الملخص

#include <systemd/sd-event.h>
typedef struct sd_event_source sd_event_source;

typedef int (*sd_event_io_handler_t)(sd_event_source *s, int fd, uint32_t revents, void *userdata);

int sd_event_add_io(sd_event *event, sd_event_source **source, int fd, uint32_t events, sd_event_io_handler_t handler, void *userdata);

int sd_event_source_get_io_events(sd_event_source *source, uint32_t *ret);

int sd_event_source_set_io_events(sd_event_source *source, uint32_t events);

int sd_event_source_set_io_fd_own(sd_event_source *source, int b);

الوصف

تضيف sd_event_add_io() مصدر حدث إدخال/إخراج جديد إلى حلقة أحداث. يُحدَّد كائن حلقة الأحداث في المعامل event، ويُعاد كائن مصدر الحدث في المعامل source. يأخذ المعامل fd واصف الملف UNIX المراد مراقبته، والذي قد يشير إلى مقبس أو FIFO أو قائمة انتظار رسائل أو اتصال تسلسلي أو جهاز حرفي أو أي واصف ملف آخر متوافق مع Linux epoll(7). يأخذ المعامل events قناع بتات للأحداث المراد مراقبتها، وهو مزيج من أعلام الأحداث التالية: EPOLLIN وEPOLLOUT وEPOLLRDHUP وEPOLLPRI وEPOLLET، راجع epoll_ctl(2) للتفاصيل. لاحظ أن ليس كل واصفات الملفات متوافقة مع epoll، فمثلاً الملفات العادية أو الأدلة ليست كذلك. إذا استُدعيت هذه الدالة بواصف ملف لا يدعم epoll، فسيُعاد -EPERM (انظر أيضاً أدناه). في معظم الحالات يمكن معاملة واصفات الملفات هذه على أنها قابلة للقراءة دائماً أو قابلة للكتابة دائماً، بحيث لا تكون مراقبة أحداث الإدخال/الإخراج ضرورية.

المعامل handler هو دالة تُستدعى عند تفعيل مصدر الحدث أو NULL. سيُمرَّر مؤشر userdata إلى دالة المعالج، ويمكن للمستدعي اختياره بحرية. سيُمرَّر أيضاً إلى المعالج واصف الملف الذي شوهد الحدث عليه، بالإضافة إلى أعلام الأحداث الفعلية. وهي عموماً مجموعة فرعية من الأحداث المراقبة، لكنها قد تتضمن إضافةً إلى ذلك EPOLLERR وEPOLLHUP. يمكن للمعالج إرجاع قيمة سالبة للإشارة إلى خطأ (انظر أدناه)، وتُتجاهل قيم الإرجاع الأخرى. إذا كان handler هو NULL، فسيُستخدم معالج افتراضي يستدعي sd_event_exit(3).

بشكل افتراضي، يبقى مصدر الحدث ممكَّناً باستمرار (SD_EVENT_ON)، لكن يمكن تغيير ذلك باستخدام sd_event_source_set_enabled(3). إذا أعادت دالة المعالج رمز خطأ سالباً، فسيُعطَّل إما بعد الاستدعاء، حتى لو كان وضع SD_EVENT_ON مطلوباً سابقاً، أو سيتسبب في إنهاء الحلقة، راجع sd_event_source_set_exit_on_failure(3). لاحظ أن مصدر حدث مضبوطاً على SD_EVENT_ON سيُفعَّل باستمرار ما لم تُقرأ بيانات من واصف الملف أو تُكتب إليه لإعادة ضبط قناع الأحداث المشاهدة.

ضبط قناع أحداث الإدخال/الإخراج المراد مراقبتها على 0 لا يعني أن مصدر الحدث لن يُفعَّل بعد الآن، لأن EPOLLHUP وEPOLLERR قد يُفعَّلان حتى مع قناع أحداث صفري. لتعطيل مصدر حدث إدخال/إخراج مؤقتاً، استخدم sd_event_source_set_enabled(3) مع SD_EVENT_OFF بدلاً من ذلك.

لتدمير كائن مصدر حدث، استخدم sd_event_source_unref(3)، لكن لاحظ أن مصدر الحدث لا يُزال من حلقة الأحداث إلا عندما تُسقط جميع المراجع إلى مصدر الحدث. للتأكد من أن مصدر الحدث لن يُفعَّل بعد الآن، حتى لو كانت لا تزال هناك مراجع إليه، عطِّل مصدر الحدث باستخدام sd_event_source_set_enabled(3) مع SD_EVENT_OFF.

إذا كان المعامل الثاني لـ sd_event_add_io() هو NULL، فلن يُعاد أي مرجع إلى كائن مصدر الحدث. في هذه الحالة، يُعتبر مصدر الحدث "عائماً"، وسيُدمَّر ضمنياً عند تدمير حلقة الأحداث نفسها.

إذا كان handler الممرَّر إلى sd_event_add_io() هو NULL، وفُعِّل مصدر الحدث، فسيُعتبر ذلك طلباً للخروج من حلقة الأحداث. في هذه الحالة، يُمرَّر المعامل userdata، محوَّلاً إلى عدد صحيح، كمعامل رمز الخروج إلى sd_event_exit(3).

لاحظ أن هذا الاستدعاء لا يستحوذ على واصف الملف الممرَّر، فتبقى الملكية (وبالتالي واجب إغلاقه عند عدم الحاجة إليه) مع المستدعي. ومع ذلك، باستخدام الاستدعاء sd_event_source_set_io_fd_own() (انظر أدناه) يمكن لمصدر الحدث اختيارياً الاستحواذ على ملكية واصف الملف بعد إنشاء مصدر الحدث. في هذه الحالة يُغلق واصف الملف تلقائياً بمجرد تحرير مصدر الحدث.

يُنصح باستخدام sd_event_add_io() فقط بالاقتران مع واصفات الملفات التي لديها O_NONBLOCK مضبوطاً، لضمان أن جميع عمليات الإدخال/الإخراج من المعالجات المستدعاة غير متزامنة وغير حاجزة بشكل صحيح. استخدام واصفات ملفات بدون O_NONBLOCK قد يؤدي إلى تجويع غير متوقع لمصادر أحداث أخرى. راجع fcntl(2) للتفاصيل حول تمكين وضع O_NONBLOCK.

تسترد sd_event_source_get_io_events() قناع أحداث الإدخال/الإخراج المراقبة المضبوط لمصدر حدث أُنشئ سابقاً باستخدام sd_event_add_io(). تأخذ كائن مصدر الحدث ومؤشراً إلى متغير لتخزين القناع فيه.

تضبط sd_event_source_set_io_events() قناع أحداث الإدخال/الإخراج المراقبة لمصدر حدث أُنشئ سابقاً باستخدام sd_event_add_io(). تأخذ كائن مصدر الحدث وقناع الأحداث الجديد.

تسترد sd_event_source_get_io_revents() قناع أحداث الإدخال/الإخراج للأحداث المشاهدة حالياً ولكن غير المُوزَّعة من مصدر حدث أُنشئ سابقاً باستخدام sd_event_add_io(). تأخذ كائن مصدر الحدث ومؤشراً إلى متغير لتخزين قناع الأحداث فيه. عند استدعائها من دالة معالج على كائن مصدر حدث المعالج، ستعيد نفس القناع الممرَّر إلى معامل revents الخاص بالمعالج. هذا الاستدعاء مفيد بشكل أساسي للتحقق من الأحداث غير المُوزَّعة لمصدر حدث من معالج مصدر حدث غير ذي صلة (ربما ذي أولوية أعلى). لاحظ العلاقة بين sd_event_source_get_pending() وsd_event_source_get_io_revents(): ستبلغ كلتا الدالتين عن نتائج غير صفرية عندما يكون هناك حدث معلّق لمصدر الحدث، لكن الأولى تنطبق على جميع أنواع مصادر الأحداث، والثانية فقط على مصادر أحداث الإدخال/الإخراج.

تسترد sd_event_source_get_io_fd() واصف الملف UNIX لمصدر حدث أُنشئ سابقاً باستخدام sd_event_add_io(). تأخذ كائن مصدر الحدث وتعيد واصف الملف غير السالب أو رقم خطأ سالباً عند حدوث خطأ (انظر أدناه).

تغيّر sd_event_source_set_io_fd() واصف الملف UNIX لمصدر حدث إدخال/إخراج أُنشئ سابقاً باستخدام sd_event_add_io(). تأخذ كائن مصدر الحدث وواصف الملف الجديد كمعاملات. إذا كان مصدر الحدث يملك واصف الملف السابق، أي إذا استُدعيت sd_event_source_set_io_fd_own() لمصدر الحدث بقيمة غير صفرية، فسيُغلق واصف الملف السابق وسيستحوذ مصدر الحدث أيضاً على ملكية واصف الملف الجديد عند النجاح.

تتحكم sd_event_source_set_io_fd_own() فيما إذا كان ينبغي لمصدر الحدث الاستحواذ على ملكية واصف الملف. تأخذ معاملاً منطقياً b. عندما يكون صحيحاً (غير صفري)

، سيتم إغلاق واصف الملف تلقائيًا عند تحرير مصدر الحدث أو عند استبدال واصف الملف بواسطة sd_event_source_set_io_fd(). افتراضيًا، لا يكون الواصف مملوكًا لمصدر الحدث، وعلى التطبيق إغلاقه بنفسه إذا لزم الأمر.

يمكن استخدام sd_event_source_get_io_fd_own() للاستعلام عن الإعداد الحالي لعلم ملكية واصف الملف المنطقي كما تم ضبطه باستخدام sd_event_source_set_io_fd_own(). يُرجع قيمة موجبة إذا كان واصف الملف يُغلق تلقائيًا عند تدمير مصدر الحدث، وصفرًا إذا لم يكن كذلك، وقيمة سالبة عند حدوث خطأ.

القيمة المُرجعة

عند النجاح، تُرجع هذه الدوال 0 أو عددًا صحيحًا موجبًا. عند الفشل، تُرجع رمز خطأ سالبًا بنمط errno.

الأخطاء

قد تشير القيم المُرجعة إلى المشكلات التالية:

-ENOMEM

لا توجد ذاكرة كافية لتخصيص كائن.

-EINVAL

تم تمرير وسيطة غير صالحة.

-ESTALE

تم إنهاء حلقة الأحداث بالفعل.

-ECHILD

تم إنشاء حلقة الأحداث في عملية أو مكتبة أو نسخة وحدة مختلفة.

-EDOM

مصدر الحدث المُمرَّر ليس مصدر حدث إدخال/إخراج.

-EPERM

واصف الملف المُمرَّر لا يدعم واجهة epoll(7)، على سبيل المثال لأنه ملف عادي أو دليل. راجع epoll_ctl(2) للتفاصيل.

ملاحظات

الدوال الموصوفة هنا متاحة كمكتبة مشتركة، يمكن الترجمة مقابلها والربط بها باستخدام ملف libsystemd pkg-config(1).

يستخدم الكود الموصوف هنا getenv(3)، والذي يُصرَّح بأنه غير آمن متعدد الخيوط. هذا يعني أن الكود الذي يستدعي الدوال الموصوفة هنا يجب ألا يستدعي setenv(3) من خيط متوازٍ. يُنصح بإجراء استدعاءات setenv() فقط من مرحلة مبكرة من البرنامج عندما لا تكون قد بدأت خيوط أخرى.
