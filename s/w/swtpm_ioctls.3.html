swtpm_ioctls(3)

الحزمة: swtpm

التراخيص: BSD-3-Clause

sed لتحديد نهاية عملية التجزئة (hash) التي تم بدؤها باستخدام أمر PTM_HASH_START.

يتم إرجاع رمز نتيجة TPM في ptm_res.

PTM_CANCEL_CMD / CMD_CANCEL_CMD, ptm_res
يستخدم هذا الأمر لإلغاء أمر TPM.

يتم إرجاع رمز نتيجة TPM في ptm_res.

PTM_STORE_VOLATILE / CMD_STORE_VOLATILE, ptm_res
يستخدم هذا الأمر لتشغيل TPM لتخزين الحالة المتطايرة في ملف.

يتم إرجاع رمز نتيجة TPM في ptm_res.

PTM_RESET_ESTABLISHED / CMD_RESET_ESTABLISHED, ptm_reset_est
يستخدم هذا الأمر لإعادة تعيين علامة التأسيس الخاصة بـ TPM.

تبدو بنية البيانات ptm_reset_est كما يلي:

struct ptm_reset_est {
   union {
       struct {
           uint8_t loc; /* المحلى المستخدم */
       } req; /* طلب */
       struct {
           ptm_res tpm_result;
       } resp; /* استجابة */
   } u;
};

يجب تعيين المحلى الذي سيتم فيه إعادة تعيين علامة التأسيس في حقل loc. المحليات الصالحة هي في النطاق من 0 إلى 4.

يتم إرجاع رمز نتيجة TPM في حقل tpm_result.

PTM_GET_STATEBLOB / CMD_GET_STATEBLOB, ptm_getstate
يستخدم هذا الأمر لبدء استرجاع أحد مقتطفات حالة TPM.

تبدو بنية البيانات ptm_getstate كما يلي:

struct ptm_getstate {
   union {
       struct {
           uint32_t state_flags; /* قد تكون: PTM_STATE_FLAG_DECRYPTED */
           uint32_t type;        /* أي مقتطف لسحبه */
           uint32_t offset;      /* الإزاحة التي سيتم القراءة منها */
       } req; /* طلب */
       struct {
           ptm_res tpm_result;
           uint32_t state_flags; /* قد تكون: PTM_STATE_FLAG_ENCRYPTED */
           uint32_t totlength;   /* الطول الإجمالي الذي سيتم نقله */
           uint32_t length;      /* عدد البايتات في المخزن المؤقت التالي */
           uint8_t  data[PTM_STATE_BLOB_SIZE];
       } resp; /* استجابة */
   } u;
};

في الطلب، يسمح حقل state_flags للمستخدم بتعيين العلامة PTM_STATE_FLAG_DECRYPT لاسترجاع حالة TPM مفككة التشفير في حالة تم كتابة حالة TPM بصيغة مشفرة.

يسمح حقل type للمستخدم باختيار أحد مقتطفات حالة TPM، ويجب أن يكون أحد PTM_BLOB_TYPE_PERMANENT، PTM_BLOB_TYPE_VOLATILE، و PTM_BLOB_TYPE_SAVESTATE.

يشير حقل offset إلى الإزاحة التي سيتم قراءة البيانات منها. يجب أن تقوم عمليات نقل الحالة اللاحقة بزيادة حقل offset إلى البايت التالي المراد قراءته. إذا تم استخدام واجهة read()، فسيتم زيادة offset تلقائيًا.

تُرجع الاستجابة رمز خطأ TPM في حقل tpm_result.

يشير حقل state_flags في الاستجابة إلى ما إذا كان المقتطف المُرجع مشفرًا.

يشير حقل totlength إلى الطول الإجمالي لمقتطف الحالة.

يشير حقل length إلى عدد البايتات الصالحة في حقل data.

إذا لزم الأمر، يجب إجراء عمليات نقل مقتطف الحالة اللاحقة باستخدام أمر ioctl هذا أو باستخدام استدعاء read() على واصف الملف. يجب نقل جميع الحالات قبل أن تتمكن TPM من قبول الأوامر مرة أخرى.

PTM_SET_STATEBLOB / CMD_SET_STATEBLOB, ptm_setstate
يستخدم هذا الأمر لنقل أحد مقتطفات حالة TPM إلى TPM.

تبدو بنية البيانات ptm_setstate كما يلي:

struct ptm_setstate {
   union {
       struct {
           uint32_t state_flags; /* قد تكون PTM_STATE_FLAG_ENCRYPTED */
           uint32_t type;        /* أي مقتطف لتعيينه */
           uint32_t length;      /* طول البيانات؛
                                    استخدم 0 في الحزمة الأولى
                                    للتحويل باستخدام write() */
           uint8_t data[PTM_STATE_BLOB_SIZE];
       } req; /* طلب */
       struct {
           ptm_res tpm_result;
       } resp; /* استجابة */
   } u;
};

يشير حقل state_flags إلى ما إذا كانت الحالة المقدمة مشفرة. في حالة تشفيرها، يجب توفير مفتاح ترحيل لـ TPM لكي تتمكن من فك تشفير الحالة.

يشير حقل type إلى أي من مقتطفات حالة TPM التي يتم تعيينها. يجب أن يكون أحد PTM_BLOB_TYPE_PERMANENT، PTM_BLOB_TYPE_VOLATILE، و PTM_BLOB_TYPE_SAVESTATE.

يشير حقل length إلى عدد البايتات من بيانات مقتطف الحالة في حقل data. لنقل مقتطف الحالة باستخدام استدعاء write()، قم بتعيين الطول إلى 0.

تُرجع الاستجابة رمز خطأ TPM في حقل tpm_result.

PTM_STOP / CMD_STOP, ptm_res
يستخدم هذا الأمر لإيقاف TPM. على عكس إيقاف تشغيل TPM، فإن إيقاف TPM يوقف عملياته فقط دون إنهاء عملية TPM. يمكن إعادة تشغيل TPM باستخدام أمر ioctl PTM_INIT.

يتم إرجاع رمز نتيجة TPM في ptm_res.

PTM_GET_CONFIG / CMD_GET_CONFIG, ptm_getconfig
يستخدم هذا الأمر لاسترجاع التكوين الحالي لـ TPM.

تبدو بنية البيانات ptm_getconfig كما يلي:

struct ptm_getconfig {
   union {
       struct {
           ptm_res tpm_result;
           uint32_t flags;
       } resp; /* استجابة */
   } u;
};

يتم إرجاع رمز نتيجة TPM في حقل tpm_result.

يحتوي حقل flags على علامات فردية تشير إلى ما إذا كان يتم استخدام مفتاح تشفير الملف (PTM_CONFIG_FLAG_FILE_KEY) وما إذا كان يتم استخدام مفتاح ترحيل (PTM_CONFIG_FLAG_MIGRATION_KEY).

CMD_SET_DATAFD, ptm_res
يتم تنفيذ هذا الأمر فقط لقناة التحكم عبر مقبس UnixIO. يستخدم لإنشاء قناة أوامر TPM عن طريق نقل واصف ملف مقبس باستخدام قناة التحكم لمقبس UnixIO و SCM_RIGHTS. انظر أيضًا sendmsg(2) و cmsg(3).

يتم إرجاع رمز نتيجة TPM في ptm_res.

CMD_SET_BUFFERSIZE, ptm_setbuffersize
يسمح هذا الأمر للمستخدم بتعيين حجم المخزن المؤقت الذي تستخدمه TPM لمخازن الإدخال والإخراج المؤقتة والاستعلام عنه.

تبدو بنية البيانات ptm_setbuffersize كما يلي:

struct ptm_setbuffersize {
   union {
       struct {
           uint32_t buffersize; /* 0 للاستعلام عن حجم المخزن المؤقت الحالي */
       } req; /* طلب */
       struct {
           ptm_res tpm_result;
           uint32_t buffersize; /* حجم المخزن المؤقت قيد الاستخدام */
           uint32_t minsize; /* الحد الأدنى لحجم المخزن المؤقت المدعوم */
           uint32_t maxsize; /* الحد الأقصى لحجم المخزن المؤقت المدعوم */
       } resp; /* استجابة */
   } u;
};

إذا تم تعيين 0 في حجم المخزن المؤقت للطلب، فستُرجع الاستجابة حجم المخزن المؤقت قيد الاستخدام حاليًا. أي رقم آخر سيحاول تغيير حجم المخزن المؤقت، ولكن قد تقوم TPM بضبطه إلى حد أدنى أو أقصى مسموح به. يتم إرجاع الحد الأدنى والأقصى لأحجام المخازن المؤقتة المدعومة في الاستجابة.

يمكن تغيير حجم المخزن المؤقت فقط عندما تكون TPM متوقفة. يمكن قراءة حجم المخزن المؤقت المستخدم حاليًا في أي وقت.

CMD_LOCK_STORAGE, ptm_lockstorage
قم بقفل التخزين وأعد المحاولة لعدد معين من المرات مع تأخير 10 مللي ثانية بين كل محاولة. قد يكون قفل التخزين ضروريًا للقيام به بعد ترحيل حالة TPM وتحرير القفل على التخزين عند استلام مقتطف 'savestate' والآن يجب قفل التخزين مرة أخرى.

تبدو بنية البيانات ptm_lockstorage كما يلي:

struct ptm_lockstorage {
    union {
        struct {
            uint32_t retries; /* عدد المحاولات */
        } req; /* طلب */
        struct {
            ptm_res tpm_result;
        } resp; /* استجابة */
    } u;
};

يتم إرجاع رمز نتيجة TPM في حقل tpm_result.
