subroutine srotm (integer n, real, dimension(*) sx, integer incx, real, dimension(*) sy, integer incy, real, dimension(5) sparam)

SROTM

الغرض:

!>
!>    تطبيق تحويل GIVENS المعدل، H، على المصفوفة 2 × N
!>
!>    (SX**T) ، حيث **T تشير إلى المنقول. عناصر SX تكون في
!>    (SX**T)
!>
!>    SX(LX+I*INCX)، I = 0 إلى N-1، حيث LX = 1 إذا كان INCX .GE. 0، وإلا
!>    LX = (-INCX)*N، وبشكل مشابه لـ SY باستخدام LY و INCY.
!>    مع SPARAM(1)=SFLAG، H يأخذ أحد الأشكال التالية..
!>
!>    SFLAG=-1.E0     SFLAG=0.E0        SFLAG=1.E0     SFLAG=-2.E0
!>
!>      (SH11  SH12)    (1.E0  SH12)    (SH11  1.E0)    (1.E0  0.E0)
!>    H=(          )    (          )    (          )    (          )
!>      (SH21  SH22)،   (SH21  1.E0)،   (-1.E0 SH22)،   (0.E0  1.E0).
!>    انظر SROTMG لوصف تخزين البيانات في SPARAM.
!>
!>

المعاملات

N 
!>          N هو عدد صحيح (INTEGER)
!>         عدد العناصر في المتجه (المتجهات) المدخلة
!>

SX

!>          SX هو مصفوفة حقيقية (REAL array)، البعد ( 1 + ( N - 1 )*abs( INCX ) )
!>

INCX

!>          INCX هو عدد صحيح (INTEGER)
!>         المسافة بين عناصر SX
!>

SY

!>          SY هو مصفوفة حقيقية (REAL array)، البعد ( 1 + ( N - 1 )*abs( INCY ) )
!>

INCY

!>          INCY هو عدد صحيح (INTEGER)
!>         المسافة بين عناصر SY
!>

SPARAM

!>          SPARAM هي مصفوفة حقيقية (REAL array)، البعد (5)
!>     SPARAM(1)=SFLAG
!>     SPARAM(2)=SH11
!>     SPARAM(3)=SH21
!>     SPARAM(4)=SH12
!>     SPARAM(5)=SH22
!>
