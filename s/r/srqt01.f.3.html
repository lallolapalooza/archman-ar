subroutine srqt01 (integer m, integer n, real, dimension( lda, * ) a, real, dimension( lda, * ) af, real, dimension( lda, * ) q, real, dimension( lda, * ) r, integer lda, real, dimension( * ) tau, real, dimension( lwork ) work, integer lwork, real, dimension( * ) rwork, real, dimension( * ) result)

SRQT01

الغرض:

!>
!> SRQT01 يختبر SGERQF، الذي يحسب تحليل RQ للمصفوفة m-by-n A، ويختبر جزئيًا SORGRQ
!> الذي يشكل المصفوفة المتعامدة n-by-n Q.
!>
!> SRQT01 يقارن R مع A*Q'، ويتحقق من أن Q متعامدة.
!>

المعلمات

M 
!>          M هو عدد صحيح
!>          عدد صفوف المصفوفة A.  M >= 0.
!>

N

!>          N هو عدد صحيح
!>          عدد أعمدة المصفوفة A.  N >= 0.
!>

A

!>          A مصفوفة حقيقية، ذات بعد (LDA,N)
!>          المصفوفة A ذات الأبعاد m-by-n.
!>

AF

!>          AF مصفوفة حقيقية، ذات بعد (LDA,N)
!>          تفاصيل تحليل RQ للمصفوفة A، كما تم إرجاعها بواسطة SGERQF.
!>          راجع SGERQF لمزيد من التفاصيل.
!>

Q

!>          Q مصفوفة حقيقية، ذات بعد (LDA,N)
!>          المصفوفة المتعامدة Q ذات الأبعاد n-by-n.
!>

R

!>          R مصفوفة حقيقية، ذات بعد (LDA,max(M,N))
!>

LDA

!>          LDA هو عدد صحيح
!>          البعد الرائد للمصفوفات A، AF، Q و L.
!>          LDA >= max(M,N).
!>

TAU

!>          TAU مصفوفة حقيقية، ذات بعد (min(M,N))
!>          العوامل القياسية للمحددات الأولية، كما تم إرجاعها
!>          بواسطة SGERQF.
!>

WORK

!>          WORK مصفوفة حقيقية، ذات بعد (LWORK)
!>

LWORK

!>          LWORK هو عدد صحيح
!>          بعد المصفوفة WORK.
!>

RWORK

!>          RWORK مصفوفة حقيقية، ذات بعد (max(M,N))
!>

RESULT

!>          RESULT مصفوفة حقيقية، ذات بعد (2)
!>          نسب الاختبار:
!>          RESULT(1) = norm( R - A*Q' ) / ( N * norm(A) * EPS )
!>          RESULT(2) = norm( I - Q*Q' ) / ( N * EPS )
!>
