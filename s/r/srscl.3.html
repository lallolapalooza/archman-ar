subroutine srscl (integer n, real sa, real, dimension( * ) sx, integer incx)

SRSCL تضرب متجهًا بقيمة قياسية حقيقية مقلوب 1/a.

الغرض:

!>
!> SRSCL تضرب متجهًا حقيقيًا مكونًا من N عنصر x بالقيمة القياسية الحقيقية 1/a.
!> يتم ذلك دون تجاوز النطاق أو الانهيار طالما أن
!> النتيجة النهائية x/a لا تتجاوز النطاق أو تنهار.
!>

المعلمات

N 
!>          N هو عدد صحيح (INTEGER)
!>          عدد مكونات المتجه x.
!>

SA

!>          SA هي قيمة حقيقية (REAL)
!>          القيمة القياسية a التي تستخدم لتقسيم كل مكون من x.
!>          يجب أن تكون SA >= 0، وإلا فسيقوم الروتين الفرعي بالقسمة على صفر.
!>

SX

!>          SX هي مصفوفة قيم حقيقية (REAL array)، بأبعاد
!>                         (1+(N-1)*abs(INCX))
!>          المتجه x المكون من N عنصر.
!>

INCX

!>          INCX هو عدد صحيح (INTEGER)
!>          الفارق بين القيم المتتالية للمتجه SX.
!>          > 0:  SX(1) = X(1) و SX(1+(i-1)*INCX) = x(i)،     1< i<= n
!>
