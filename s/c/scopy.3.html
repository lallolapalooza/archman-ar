subroutine scopy (integer n, real, dimension(*) sx, integer incx, real, dimension(*) sy, integer incy)

SCOPY

الغرض:

!>
!>    SCOPY يقوم بنسخ متجه، x، إلى متجه، y.
!>    يستخدم حلقات غير مطوية للزيادات التي تساوي 1.
!>

المعاملات

N 
!>          N هو عدد صحيح
!>         عدد العناصر في المتجه (المتجهات) المدخلة
!>

SX

!>          SX هو مصفوفة حقيقية، البعد ( 1 + ( N - 1 )*abs( INCX ) )
!>

INCX

!>          INCX هو عدد صحيح
!>         تباعد التخزين بين عناصر SX
!>

SY

!>          SY هو مصفوفة حقيقية، البعد ( 1 + ( N - 1 )*abs( INCY ) )
!>

INCY

!>          INCY هو عدد صحيح
!>         تباعد التخزين بين عناصر SY
!>
