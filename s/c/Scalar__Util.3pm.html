Scalar::Util(3) وثائق Perl المساهمة Scalar::Util(3)

الاسم

Scalar::Util - مجموعة من الدوال القياسية العامة

الملخص

use Scalar::Util qw(blessed dualvar isdual readonly refaddr reftype
                    tainted weaken isweak isvstring looks_like_number
                    set_prototype);
                    # ودوال مساعدة أخرى مفيدة تظهر أدناه

الوصف

"Scalar::Util" يحتوي على مجموعة من الدوال التي عبر الناس عن فائدتها لوجودها في جوهر Perl، ولكن استخدامها لن يكون كافياً لتبرير استخدام كلمة مفتاحية، وسيكون حجمها صغيراً جداً لدرجة أنها ستكون غير فعالة لجعلها امتدادًا فرديًا.

افتراضيًا "Scalar::Util" لا تصدّر أي دوال.

دوال Perl الأساسية "المدمجة"

كانت العديد من دوال هذه الوحدة بمثابة إلهام لدالة تجريبية جديدة في الإصدارات الأخيرة من Perl. منذ إصدارات تطوير مختلفة، بدءًا من 5.35.7، تتوفر دوال مكافئة للعديد من هذه الأدوات المساعدة في وحدة "builtin::".

use Scalar::Util qw(blessed);
$class = blessed $obj;
$class = builtin::blessed $obj;  # مكافئ

لمزيد من المعلومات، انظر التوثيق حول "builtin".

دوال للمراجع

تقوم الدوال التالية ببعض الأنشطة المفيدة مع قيم المراجع.

blessed

my $pkg = blessed( $ref );

إذا كانت $ref مرجعًا تم مباركته، فسيتم إرجاع اسم الفئة التي تم مباركته بها. وإلا، فسيتم إرجاع "undef".

$scalar = "foo";
$class  = blessed $scalar;           # undef
$ref    = [];
$class  = blessed $ref;              # undef
$obj    = bless [], "Foo";
$class  = blessed $obj;              # "Foo"

كن حذرًا عند استخدام هذه الدالة كاختبار للحقيقة (مثل "if(blessed $ref)...") لأن اسم الفئة "0" معرّف ولكنه كاذب.

منذ إصدار Perl 5.35.7، تتوفر دالة مكافئة كـ "builtin::blessed".

refaddr

my $addr = refaddr( $ref );

إذا كانت $ref مرجعًا، فسيتم إرجاع عنوان الذاكرة الداخلي للقيمة المشار إليها كعدد صحيح بسيط. وإلا، فسيتم إرجاع "undef".

$addr = refaddr "string";           # undef
$addr = refaddr \$var;              # مثال 12345678
$addr = refaddr [];                 # مثال 23456784
$obj  = bless {}, "Foo";
$addr = refaddr $obj;               # مثال 88123488

منذ إصدار Perl 5.35.7، تتوفر دالة مكافئة كـ "builtin::refaddr".

reftype

my $type = reftype( $ref );

إذا كانت $ref مرجعًا، فسيتم إرجاع نوع المتغير الأساسي في Perl كسلسلة (مثل "ARRAY" أو "HASH"). وإلا، فسيتم إرجاع "undef".

$type = reftype "string";           # undef
$type = reftype \$var;              # SCALAR
$type = reftype [];                 # ARRAY
$obj  = bless {}, "Foo";
$type = reftype $obj;               # HASH

ملاحظة: لأسباب داخلية، جميع التعبيرات النمطية المترجمة مسبقًا ("qr/.../") هي مراجع مباركة؛ لذلك، ref() ترجع اسم الفئة "Regexp" ولكن reftype() ترجع نوع هيكل C لـ "REGEXP" بالأحرف الكبيرة.

منذ إصدار Perl 5.35.7، تتوفر دالة مكافئة كـ "builtin::reftype".

weaken

weaken( $ref );

يتم تحويل المرجع القابل للتخصيص $ref إلى مرجع ضعيف. هذا يعني أنه لن يحتفظ بعداد مراجع للكائن الذي يشير إليه. بالإضافة إلى ذلك، عندما يصل عداد المراجع للكائن إلى الصفر، سيتم تعيين المرجع إلى undef. هذه الدالة تغير المرجع القابل للتخصيص الذي يتم تمريره كمعامل ولا ترجع أي قيمة.

هذا مفيد للاحتفاظ بنسخ من المراجع، ولكنك لا ترغب في منع تدمير الكائن في الوقت المعتاد.

{
  my $var;
  $ref = \$var;
  weaken($ref);                     # جعل $ref مرجعًا ضعيفًا
}
# $ref الآن هو undef

ملاحظة: إذا قمت بنسخ مرجع ضعيف، فإن النسخة ستصبح مرجعًا عاديًا.

my $var;
my $foo = \$var;
weaken($foo);                       # جعل $foo مرجعًا ضعيفًا
my $bar = $foo;                     # $bar الآن مرجع عادي

قد يكون هذا أقل وضوحًا في مواقف أخرى، مثل grep()، على سبيل المثال، عند grep قائمة من المراجع الضعيفة لكائنات ربما تم تدميرها بالفعل:

@object = grep { defined } @object;

هذا يزيل فعليًا جميع المراجع للكائنات المدمرة، لكن المراجع المتبقية للكائنات ستظل مراجع عادية، مما يتسبب في عدم تدمير الكائنات المتبقية أبدًا لأن هناك دائمًا مرجعًا عاديًا في المصفوفة @object.

منذ إصدار Perl 5.35.7، تتوفر دالة مكافئة كـ "builtin::weaken".

unweaken

unweaken( $ref );

منذ الإصدار 1.36.

يتم تحويل المرجع القابل للتخصيص "REF" من مرجع ضعيف إلى مرجع عادي (خارج مرجع ضعيف). هذه الدالة تغير المرجع القابل للتخصيص الذي يتم تمريره كمعامل ولا ترجع أي قيمة. هذا يلغي تأثير الإجراء الذي قام به "weaken".

هذه الدالة أنظف وأكثر ملاءمة قليلاً من الكود المكافئ الذي يتم استخدامه بخلاف ذلك:

my $tmp = $REF;
undef $REF;
$REF = $tmp;

(على سبيل المثال، مجرد تعيين مرجع ضعيف إلى نفسه لا يعمل لإلغائه؛ "$REF = $REF" لا يعمل).

منذ إصدار Perl 5.35.7، تتوفر دالة مكافئة كـ "builtin::unweaken".

isweak

my $weak = isweak( $ref );

ترجع true إذا كان $ref مرجعًا ضعيفًا.

$ref  = \$foo;
$weak = isweak($ref);               # false
weaken($ref);
$weak = isweak($ref);               # true

ملاحظة: نسخ مرجع ضعيف ينشئ مرجعًا عاديًا قويًا.

$copy = $ref;
$weak = isweak($copy);              # false

منذ إصدار Perl 5.35.7، تتوفر دالة مكافئة كـ "builtin::is_weak".

دوال أخرى

dualvar

my $var = dualvar( $num, $string );

ترجع قيمة قياسية لها القيمة $num في سياق رقمي والقيمة $string في سياق سلسلة.

$foo = dualvar 10, "Hello";
$num = $foo + 2;                    # 12
$str = $foo . " world";             # Hello world

isdual

my $dual = isdual( $var );

منذ الإصدار 1.26.

إذا كانت $var قيمة قياسية لها قيم رقمية وسلسلية، فإن النتيجة تكون true.

$foo = dualvar 86, "Nix";
$dual = isdual($foo);               # true

ملاحظة: يمكن إنشاء قيمة قياسية لها محتوى سلسلي ورقمي من خلال عمليات قياسية:

$foo = "10";
$dual = isdual($foo);

# false $bar = $foo + 0; $dual = isdual($foo); # true

المتغير $! ذو قيمتين عادةً، على الرغم من أنه سحري أيضًا بطرق أخرى:

$! = 1;
$dual = isdual($!);                 # true
print("$!\n");                      # "Operation not permitted"

تحذير: هذه الوظيفة ليست مفيدة كما قد تبدو. المتغيرات المزدوجة ليست مفهومًا مميزًا في Perl، ولكنها بنية داخلية قياسية لجميع القيم العددية. يمكن اعتبار أي قيمة تقريبًا كمتغير مزدوج بواسطة هذه الوظيفة خلال العمليات العادية.

isvstring

my $vstring = isvstring( $var );

إذا كان $var قيمة عددية تم ترميزها كسلسلة v، فإن النتيجة تكون صحيحة.

$vs   = v49.46.48;
$fmt  = isvstring($vs) ? "%vd" : "%s"; #true
printf($fmt,$vs);

looks_like_number

my $isnum = looks_like_number( $var );

يعيد قيمة صحيحة إذا اعتقد perl أن $var هو رقم. انظر "looks_like_number" في perlapi.

openhandle

my $fh = openhandle( $fh );

يعيد $fh نفسه، إذا كان $fh يمكن استخدامه كمعالج ملف وهو مفتوح، أو إذا كان معالجًا مرتبطًا. بخلاف ذلك، يتم إرجاع "undef".

$fh = openhandle(*STDIN);           # \*STDIN
$fh = openhandle(\*STDIN);          # \*STDIN
$fh = openhandle(*NOTOPEN);         # undef
$fh = openhandle("scalar");         # undef

readonly

my $ro = readonly( $var );

يعيد قيمة صحيحة إذا كان $var للقراءة فقط.

sub foo { readonly($_[0]) }
$readonly = foo($bar);              # false
$readonly = foo(0);                 # true

set_prototype

my $code = set_prototype( $code, $prototype );

يقوم بتعيين النموذج الأولي للدالة المعطاة بواسطة مرجع $code، أو يحذفه إذا كان $prototype هو "undef". يعيد مرجع $code نفسه.

set_prototype \&foo, '$$';

tainted

my $t = tainted( $var );

يعيد قيمة صحيحة إذا كان $var ملوثًا.

$taint = tainted("constant");       # false
$taint = tainted($ENV{PWD});        # true if running under -T

التشخيصات

قد يعطي استخدام الوحدة النمطية أحد الأخطاء التالية أثناء الاستيراد.

سلاسل V غير مطبقة في هذا الإصدار من perl
إصدار perl الذي تستخدمه لا يطبق سلاسل V، لاستخدام "isvstring" ستحتاج إلى استخدام إصدار أحدث من perl.

الأخطاء المعروفة

يوجد خطأ في perl5.6.0 مع UVs التي تكون >= 1<<31. سيظهر هذا على أنه فشل الاختبارين 8 و 9 من dualvar.t
