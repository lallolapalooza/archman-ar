المعاملات

DOTYPE 
!>          DOTYPE هو مصفوفة منطقية، أبعادها (NTYPES)
!>          أنواع المصفوفات التي سيتم استخدامها للاختبار. المصفوفات من النوع j
!>          (لـ 1 <= j <= NTYPES) تستخدم للاختبار إذا كان DOTYPE(j) =
!>          .TRUE.؛ إذا كان DOTYPE(j) = .FALSE.، فلن يتم استخدام النوع j.
!>

NN

!>          NN هو عدد صحيح
!>          عدد القيم لـ N الموجودة في المتجه NVAL.
!>

NVAL

!>          NVAL هي مصفوفة أعداد صحيحة، أبعادها (NN)
!>          قيم بعد المصفوفة N.
!>

NNS

!>          NNS هو عدد صحيح
!>          عدد القيم لـ NRHS الموجودة في المتجه NSVAL.
!>

NSVAL

!>          NSVAL هي مصفوفة أعداد صحيحة، أبعادها (NNS)
!>          قيم عدد الأيادي اليمنى NRHS.
!>

THRESH

!>          THRESH هو عدد حقيقي
!>          القيمة الحدية لنسب الاختبار. يتم تضمين نتيجة
!>          في ملف الإخراج إذا كانت RESULT >= THRESH. لطباعة
!>          كل نسبة اختبار، استخدم THRESH = 0.
!>

TSTERR

!>          TSTERR هو منطقي
!>          علم يشير إلى ما إذا كان سيتم اختبار مخارج الخطأ.
!>

NMAX

!>          NMAX هو عدد صحيح
!>          الحد الأقصى المسموح به لـ N، المستخدم في تحديد أبعاد
!>          مصفوفات العمل.
!>

A

!>          A هي مصفوفة حقيقية، أبعادها
!>                      (NMAX*(NMAX+1)/2)
!>

AFAC

!>          AFAC هي مصفوفة حقيقية، أبعادها
!>                      (NMAX*(NMAX+1)/2)
!>

AINV

!>          AINV هي مصفوفة حقيقية، أبعادها
!>                      (NMAX*(NMAX+1)/2)
!>

B

!>          B هي مصفوفة حقيقية، أبعادها (NMAX*NSMAX)
!>          حيث NSMAX هي أكبر قيمة في NSVAL.
!>

X

!>          X هي مصفوفة حقيقية، أبعادها (NMAX*NSMAX)
!>

XACT

!>          XACT هي مصفوفة حقيقية، أبعادها (NMAX*NSMAX)
!>

WORK

!>          WORK هي مصفوفة حقيقية، أبعادها
!>                      (NMAX*max(3,NSMAX))
!>

RWORK

!>          RWORK هي مصفوفة حقيقية، أبعادها
!>                      (max(NMAX,2*NSMAX))
!>

IWORK

!>          IWORK هي مصفوفة أعداد صحيحة، أبعادها (NMAX)
!>

NOUT

!>          NOUT هو عدد صحيح
!>          رقم الوحدة للإخراج.
!>
