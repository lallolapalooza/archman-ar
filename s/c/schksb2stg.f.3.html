subroutine schksb2stg (integer nsizes, integer, dimension( * ) nn, integer nwdths, integer, dimension( * ) kk, integer ntypes, logical, dimension( * ) dotype, integer, dimension( 4 ) iseed, real thresh, integer nounit, real, dimension( lda, * ) a, integer lda, real, dimension( * ) sd, real, dimension( * ) se, real, dimension( * ) d1, real, dimension( * ) d2, real, dimension( * ) d3, real, dimension( ldu, * ) u, integer ldu, real, dimension( * ) work, integer lwork, real, dimension( * ) result, integer info)

SCHKSB2STG

الغرض:

!>
!> SCHKSB2STG تختبر اختزال مصفوفة نطاق متماثلة إلى شكل ثلاثي الأقطار،
!> تستخدم مع مشكلة القيم الذاتية المتماثلة.
!>
!> SSBTRD تقوم بتحليل مصفوفة نطاق متماثلة A بالشكل U S U' ، حيث ' تعني
!> تبديل، S ثلاثي الأقطار متماثل، و U متعامد.
!> يمكن لـ SSBTRD استخدام الجزء السفلي فقط أو الجزء العلوي فقط
!> من A؛ SCHKSB2STG تتحقق من كلا الحالتين.
!>
!> SSYTRD_SB2ST تقوم بتحليل مصفوفة نطاق متماثلة A بالشكل U S U' ،
!> حيث ' تعني تبديل، S ثلاثي الأقطار متماثل، و U متعامد.
!> يمكن لـ SSYTRD_SB2ST استخدام الجزء السفلي فقط أو الجزء العلوي فقط
!> من A؛ SCHKSB2STG تتحقق من كلا الحالتين.
!>
!> SSTEQR تقوم بتحليل S بالشكل U D1 U'.
!> D1 هي مصفوفة القيم الذاتية المحسوبة عندما لا يتم حساب U
!> و من S الناتجة عن SSBTRD (تستخدم كمرجع لـ SSYTRD_SB2ST)
!> D2 هي مصفوفة القيم الذاتية المحسوبة عندما لا يتم حساب U
!> و من S الناتجة عن SSYTRD_SB2ST.
!> D3 هي مصفوفة القيم الذاتية المحسوبة عندما لا يتم حساب U
!> و من S الناتجة عن SSYTRD_SB2ST.
!>
!> عندما يتم استدعاء SCHKSB2STG، يتم تحديد عدد من أحجام المصفوفات (NN)،
!> وعدد من عرض النطاقات (KK)، وعدد من أنواع المصفوفات (DOTYPE).
!> لكل حجم (NN)، وكل عرض نطاق (KK) أقل من أو يساوي الحجم،
!> ولكل نوع مصفوفة، سيتم إنشاء مصفوفة واحدة واستخدامها لاختبار
!> روتين الاختزال النطاقي المتماثل. لكل مصفوفة، سيتم إجراء عدد من الاختبارات:
!>
!> (1)     | A - V S V' | / ( |A| n ulp )  محسوبة بواسطة SSBTRD مع
!>                                         UPLO='U'
!>
!> (2)     | I - UU' | / ( n ulp )
!>
!> (3)     | A - V S V' | / ( |A| n ulp )  محسوبة بواسطة SSBTRD مع
!>                                         UPLO='L'
!>
!> (4)     | I - UU' | / ( n ulp )
!>
!> (5)     | D1 - D2 | / ( |D1| ulp )      حيث D1 محسوبة بواسطة
!>                                         SSBTRD مع UPLO='U' و
!>                                         D2 محسوبة بواسطة
!>                                         SSYTRD_SB2ST مع UPLO='U'
!>
!> (6)     | D1 - D3 | / ( |D1| ulp )      حيث D1 محسوبة بواسطة
!>                                         SSBTRD مع UPLO='U' و
!>                                         D3 محسوبة بواسطة
!>                                         SSYTRD_SB2ST مع UPLO='L'
!>
!> يتم تحديد أحجام المصفوفات بواسطة مصفوفة NN(1:NSIZES)؛ قيمة
!> كل عنصر NN(j) تحدد حجمًا واحدًا.
!> يتم تحديد أنواع المصفوفات بواسطة مصفوفة منطقية DOTYPE( 1:NTYPES )؛
!> إذا كانت DOTYPE(j) تساوي .TRUE.، فسيتم إنشاء مصفوفة من النوع j.
!> حاليًا، قائمة الأنواع الممكنة هي:
!>
!> (1)  مصفوفة الصفر.
!> (2)  مصفوفة الوحدة.
!>
!> (3)  مصفوفة قطرية بإدخالات متباعدة بالتساوي
!>      1, ..., ULP  وعلامات عشوائية.
!>      (ULP = (أول رقم أكبر من 1) - 1)
!> (4)  مصفوفة قطرية بإدخالات متباعدة هندسيًا
!>      1, ..., ULP  وعلامات عشوائية.
!> (5)  مصفوفة قطرية بإدخالات 1, ULP, ..., ULP
!>      وعلامات عشوائية.
!>
!> (6)  نفس (4)، ولكن مضروبة في مربع جذر (عتبة الفائض).
!> (7)  نفس (4)، ولكن مضروبة في مربع جذر (عتبة النقص).
!>
!> (8)  مصفوفة من الشكل U' D U، حيث U متعامد و
!>      D يحتوي على إدخالات متباعدة بالتساوي 1, ..., ULP مع علامات عشوائية
!>      على القطر.
!>
!> (9)  مصفوفة من الشكل U' D U، حيث U متعامد و
!>      D يحتوي على إدخالات متباعدة هندسيًا 1, ..., ULP مع علامات عشوائية
!>      على القطر.
!>
!> (10) مصفوفة من الشكل U' D U، حيث U متعامد و
!>      D يحتوي على إدخالات 1, ULP,..., ULP مع علامات عشوائية
!>      على القطر.
!>
!> (11) نفس (8)، ولكن مضروبة في مربع جذر (عتبة الفائض).
!> (12) نفس (8)، ولكن مضروبة في مربع جذر (عتبة النقص).
!>
!> (13) مصفوفة متماثلة بإدخالات عشوائية مختارة من (-1,1).
!> (14) نفس (13)، ولكن مضروبة في مربع جذر (عتبة الفائض).
!> (15) نفس (13)، ولكن مضروبة في مربع جذر (عتبة النقص).
!>

المعلمات

NSIZES 
!>          NSIZES هو عدد صحيح (INTEGER).
!>          عدد أحجام المصفوفات المراد استخدامها. إذا كان صفرًا،
!>          فإن SCHKSB2STG لا تفعل شيئًا. يجب أن يكون على الأقل صفرًا.
!>

NN

!>          NN هي مصفوفة أعداد صحيحة (INTEGER array)، البعد (NSIZES).
!>          مصفوفة تحتوي على الأحجام المراد استخدامها للمصفوفات.
!>          سيتم تخطي القيم الصفرية. يجب أن تكون القيم على الأقل
!>          صفرًا.
!>

NWDTHS

!>          NWDTHS هو عدد صحيح (INTEGER).
!>          عدد عروض النطاقات المراد استخدامها. إذا كان صفرًا،
!>          فإن SCHKSB2STG لا تفعل شيئًا. يجب أن يكون على الأقل صفرًا.
!>

KK

!>          KK هي مصفوفة أعداد صحيحة (INTEGER array)، البعد (NWDTHS).
!>          مصفوفة تحتوي على عروض النطاقات المراد استخدامها للمصفوفات
!>          النطاقية. يجب أن تكون القيم على الأقل صفرًا.
!>

NTYPES

!>          NTYPES هو عدد صحيح (INTEGER).
!>          عدد العناصر في DOTYPE. إذا كان صفرًا، فإن SCHKSB2STG
!>          لا تفعل شيئًا. يجب أن يكون على الأقل صفرًا. إذا كان MAXTYP+1
!>          و NSIZES هو 1، فسيتم تعريف نوع إضافي، MAXTYP+1،
!>          وهو استخدام أي مصفوفة موجودة في A. هذا
!>          مفيد فقط إذا كانت DOTYPE(1:MAXTYP) تساوي .FALSE. و
!>          DOTYPE(MAXTYP+1) تساوي .TRUE. .
!>

DOTYPE

!>          DOTYPE هي مصفوفة منطقية (LOGICAL array)، البعد (NTYPES).
!>          إذا كانت DOTYPE(j) تساوي .TRUE.، فلكل حجم في NN سيتم إنشاء
!>          مصفوفة بهذا الحجم ومن النوع j.
!>          إذا كان NTYPES أصغر من الحد الأقصى لعدد الأنواع
!>          المعرفة (PARAMETER MAXTYP)، فلن يتم إنشاء الأنواع NTYPES+1 إلى
!>          MAXTYP. إذا كان NTYPES أكبر
!>          من MAXTYP، فسيتم تجاهل DOTYPE(MAXTYP+1) إلى DOTYPE(NTYPES).
!>

ISEED

!>          ISEED هي مصفوفة أعداد صحيحة (INTEGER array)، البعد (4).
!>          عند الإدخال، تحدد ISEED البذرة لمولد الأرقام العشوائية.
!>          يجب أن تكون عناصر المصفوفة بين 0 و 4095؛
!>          إذا لم تكن كذلك، فسيتم تقليلها بمقدار 4096. أيضًا، يجب أن يكون ISEED(4)
!>          فرديًا. يستخدم مولد الأرقام العشوائية تسلسل خطي متوافق
!>          محدود بأعداد صحيحة صغيرة، وبالتالي
!>          يجب أن ينتج أرقامًا عشوائية مستقلة عن الجهاز.
!>          تتغير قيم ISEED عند الخروج، ويمكن استخدامها في
!>          الاستدعاء التالي لـ SCHKSB2STG لمواصلة نفس تسلسل الأرقام العشوائية.
!>

THRESH

!>          THRESH هو عدد حقيقي (REAL).
!>          سيتم احتساب اختبار على أنه فشل إذا تجاوزت النسبة،
!>          المحسوبة كما هو موضح أعلاه، THRESH. لاحظ أن الخطأ
!>          يتم قياسه ليكون O(1)، لذلك يجب أن يكون THRESH مضاعفًا معقولًا
!>          لـ 1، على سبيل المثال، 10 أو 100. على وجه الخصوص،
!>          لا يجب أن يعتمد على الدقة (مفردة مقابل مزدوجة)
!>          أو حجم المصفوفة.
!>

NOUN IT

!>          عدد الأعداد الصحيحة غير الصفرية في مصفوفة نطاق.
!>

A

!>          تحتوي هذه المصفوفة على مصفوفة النطاق المتماثلة المراد اختبارها.
!>          يتم تخزين الجزء السفلي أو العلوي من المصفوفة (حسب UPLO).
!>

LDA

!>

SD

!>          SD هي مصفوفة أعداد حقيقية (REAL array)، البعد (*).
!>          تخزن المصفوفة ثلاثية الأقطار الناتجة عن الاختزال.
!>

SE

!>          SE هي مصفوفة أعداد حقيقية (REAL array)، البعد (*).
!>          تخزن مصفوفة النطاق القطرية الناتجة عن الاختزال.
!>

D1

!>          D1 هي مصفوفة أعداد حقيقية (REAL array)، البعد (*).
!>          تخزن القيم الذاتية من SSBTRD (UPLO='U').
!>

D2

!>          D2 هي مصفوفة أعداد حقيقية (REAL array)، البعد (*).
!>          تخزن القيم الذاتية من SSYTRD_SB2ST (UPLO='U').
!>

D3

!>          D3 هي مصفوفة أعداد حقيقية (REAL array)، البعد (*).
!>          تخزن القيم الذاتية من SSYTRD_SB2ST (UPLO='L').
!>

U

!>          U هي مصفوفة أعداد حقيقية (REAL array)، البعد (LDU, *).

!>

LDU

WORK

LWORK

!>          طول مصفوفة العمل WORK.
!>

RESULT

!>          RESULT هي مصفوفة أعداد حقيقية (REAL array)، البعد (*).
!>          تخزن نتائج الاختبارات.
!>

INFO
