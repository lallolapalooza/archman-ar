subroutine slaed1 (n, d, q, ldq, indxq, rho, cutpnt, work, iwork, info)
SLAED1 يستخدم بواسطة SSTEDC. تحسب النظام الهندسي المحدث لمصفوفة قطرية بعد التعديل بواسطة مصفوفة متناظرة ذات رتبة واحد. تستخدم عندما تكون المصفوفة الأصلية ثلاثية الأقطار.

توثيق الوظيفة/الإجراء الفرعي

subroutine slaed1 (integer n, real, dimension( * ) d, real, dimension( ldq, * ) q, integer ldq, integer, dimension( * ) indxq, real rho, integer cutpnt, real, dimension( * ) work, integer, dimension( * ) iwork, integer info)

SLAED1 يستخدم بواسطة SSTEDC. تحسب النظام الهندسي المحدث لمصفوفة قطرية بعد التعديل بواسطة مصفوفة متناظرة ذات رتبة واحد. تستخدم عندما تكون المصفوفة الأصلية ثلاثية الأقطار.

الغرض:

!>
!> SLAED1 تحسب النظام الهندسي المحدث لمصفوفة قطرية بعد التعديل بواسطة مصفوفة متناظرة ذات رتبة واحد. تستخدم هذه
!> الروتين فقط للمشكلة الهندسية التي تتطلب جميع القيم الهندسية والمتجهات الهندسية لمصفوفة ثلاثية الأقطار. SSTEDC
!> يتعامل مع الحالة التي تكون فيها القيم الهندسية فقط أو القيم الهندسية والمتجهات الهندسية لمصفوفة متناظرة كاملة (التي تم
!> اختزالها إلى شكل ثلاثي الأقطار) مرغوبة.
!>
!>   T = Q(in) ( D(in) + RHO * Z*Z**T ) Q**T(in) = Q(out) * D(out) * Q**T(out)
!>
!>    حيث Z = Q**T*u، u هو متجه طوله N مع وحدات في العناصر CUTPNT و CUTPNT + 1 وأصفار في أماكن أخرى.
!>
!>    يتم تخزين المتجهات الهندسية للمصفوفة الأصلية في Q، والقيم الهندسية في D. تتكون الخوارزمية من ثلاث مراحل:
!>
!>       تتكون المرحلة الأولى من تخفيف حجم المشكلة
!>       عندما تكون هناك قيم هندسية متعددة أو إذا كان هناك صفر في
!>       متجه Z. لكل حدوث من هذا القبيل، يتم تقليل بعد مشكلة المعادلة العلمانية.
!>       يتم تنفيذ هذه المرحلة بواسطة الروتين SLAED2.
!>
!>       تتكون المرحلة الثانية من حساب القيم الهندسية المحدثة. يتم ذلك عن طريق إيجاد جذور المعادلة العلمانية
!>       عبر الروتين SLAED4 (كما تم استدعاؤه بواسطة SLAED3).
!>       يحسب هذا الروتين أيضًا المتجهات الهندسية للمشكلة الحالية.
!>
!>       تتكون المرحلة النهائية من حساب المتجهات الهندسية المحدثة
!>       مباشرة باستخدام القيم الهندسية المحدثة. يتم ضرب المتجهات الهندسية للمشكلة
!>       الحالية مع المتجهات الهندسية من المشكلة العامة.
!>

المعلمات

N 
!>          N هو عدد صحيح (INTEGER)
!>         بعد المصفوفة المتناظرة ثلاثية الأقطار. N >= 0.
!>

D

!>          D هو مصفوفة حقيقية (REAL array)، البعد (N)
!>         عند الإدخال، القيم الهندسية للمصفوفة المعدلة برتبة-1.
!>         عند الخروج، القيم الهندسية للمصفوفة المُصلحة.
!>

Q

!>          Q هو مصفوفة حقيقية (REAL array)، البعد (LDQ,N)
!>         عند الإدخال، المتجهات الهندسية للمصفوفة المعدلة برتبة-1.
!>         عند الخروج، المتجهات الهندسية للمصفوفة ثلاثية الأقطار المُصلحة.
!>

LDQ

!>          LDQ هو عدد صحيح (INTEGER)
!>         البعد الرئيسي للمصفوفة Q. LDQ >= max(1,N).
!>

INDXQ

!>          INDXQ هو مصفوفة أعداد صحيحة (INTEGER array)، البعد (N)
!>         عند الإدخال، التبديل الذي يقوم بفرز مشكلتي الفرعية في D بشكل منفصل
!>         إلى ترتيب تصاعدي.
!>         عند الخروج، التبديل الذي سيعيد دمج مشكلتي الفرعية مرة أخرى إلى الترتيب المصنف،
!>         أي أن D( INDXQ( I = 1, N ) ) سيكون بترتيب تصاعدي.
!>

RHO

!>          RHO هو حقيقي (REAL)
!>         مدخلة ما دون القطر المستخدمة لإنشاء التعديل ذي الرتبة-1.
!>

CUTPNT

!>          CUTPNT هو عدد صحيح (INTEGER)
!>         موقع آخر قيمة هندسية في المصفوفة الفرعية الرائدة.
!>         min(1,N) <= CUTPNT <= N/2.
!>

WORK

!>          WORK هو مصفوفة حقيقية (REAL array)، البعد (4*N + N**2)
!>

IWORK

!>          IWORK هو مصفوفة أعداد صحيحة (INTEGER array)، البعد (4*N)
!>

INFO

!>          INFO هو عدد صحيح (INTEGER)
!>          = 0:  إنهاء ناجح.
!>          < 0:  إذا كان INFO = -i، فإن الوسيط i كان له قيمة غير قانونية.
!>          > 0:  إذا كان INFO = 1، لم تتقارب قيمة هندسية.
!>
