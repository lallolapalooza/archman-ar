subroutine slagtf (n, a, lambda, b, c, tol, d, in, info)
SLAGTF تحسب تحليل LU للمصفوفة T-λI، حيث T مصفوفة ثلاثية قطرية عامة، و λ قياسي، باستخدام التنقيط الجزئي مع تبديلات الصفوف.

توثيق الوظيفة/الإجراء الفرعي

subroutine slagtf (integer n, real, dimension( * ) a, real lambda, real, dimension( * ) b, real, dimension( * ) c, real tol, real, dimension( * ) d, integer, dimension( * ) in, integer info)

SLAGTF تحسب تحليل LU للمصفوفة T-λI، حيث T مصفوفة ثلاثية قطرية عامة، و λ قياسي، باستخدام التنقيط الجزئي مع تبديلات الصفوف.

الغرض:

!>
!> SLAGTF تقوم بتحليل المصفوفة (T - lambda*I)، حيث T مصفوفة ثلاثية قطرية
!> n × n و lambda عدد قياسي، إلى:
!>
!>    T - lambda*I = PLU،
!>
!> حيث P مصفوفة تبديل، L مصفوفة ثلاثية قطرية سفلية وحدوية مع ما لا يزيد عن
!> عنصر واحد غير صفري تحت القطر الرئيسي في كل عمود، و U مصفوفة مثلثة علوية
!> مع ما لا يزيد عن عنصرين غير صفريين فوق القطر الرئيسي في كل عمود.
!>
!> يتم الحصول على التحليل عن طريق الحذف الغاوسي مع التنقيط الجزئي
!> والقياس الضمني للصفوف.
!>
!> يتم تضمين المعامل LAMBDA في الإجراء الفرعي حتى يمكن استخدام SLAGTF،
!> بالاقتران مع SLAGTS، للحصول على المتجهات الذاتية لـ T عن طريق التكرار العكسي.
!>

المعاملات

N 
!>          N هو عدد صحيح (INTEGER)
!>          رتبة المصفوفة T.
!>

A

!>          A هو مصفوفة أعداد حقيقية (REAL array)، البعد (N)
!>          عند الدخول، يجب أن تحتوي A على العناصر القطرية لـ T.
!>
!>          عند الخروج، يتم استبدال A بالعناصر القطرية n للمصفوفة
!>          المثلثة العلوية U لتحليل T.
!>

LAMBDA

!>          LAMBDA هو عدد حقيقي (REAL)
!>          عند الدخول، العدد القياسي lambda.
!>

B

!>          B هو مصفوفة أعداد حقيقية (REAL array)، البعد (N-1)
!>          عند الدخول، يجب أن تحتوي B على العناصر فوق القطرية (n-1) لـ
!>          T.
!>
!>          عند الخروج، يتم استبدال B بالعناصر فوق القطرية (n-1)
!>          للمصفوفة U لتحليل T.
!>

C

!>          C هو مصفوفة أعداد حقيقية (REAL array)، البعد (N-1)
!>          عند الدخول، يجب أن تحتوي C على العناصر تحت القطرية (n-1) لـ
!>          T.
!>
!>          عند الخروج، يتم استبدال C بالعناصر تحت القطرية (n-1)
!>          للمصفوفة L لتحليل T.
!>

TOL

!>          TOL هو عدد حقيقي (REAL)
!>          عند الدخول، قيمة تقريبية نسبية تستخدم للإشارة إلى ما إذا كانت
!>          المصفوفة (T - lambda*I) قريبة من التفرد أم لا. يجب اختيار TOL
!>          عادةً كخطأ نسبي تقريبي في عناصر T. على سبيل المثال، إذا كانت
!>          عناصر T صحيحة لحوالي 4 أرقام معنوية، فيجب ضبط TOL على حوالي
!>          5*10**(-4). إذا تم توفير TOL بقيمة أقل من eps، حيث eps هي الدقة
!>          النسبية للآلة، فسيتم استخدام القيمة eps بدلاً من TOL.
!>

D

!>          D هو مصفوفة أعداد حقيقية (REAL array)، البعد (N-2)
!>          عند الخروج، يتم استبدال D بالعناصر فوق القطرية الثانية (n-2)
!>          للمصفوفة U لتحليل T.
!>

IN

!>          IN هو مصفوفة أعداد صحيحة (INTEGER array)، البعد (N)
!>          عند الخروج، تحتوي IN على تفاصيل مصفوفة التبديل P. إذا حدث
!>          تبديل في الخطوة k من الحذف، فإن IN(k) = 1، وإلا فإن IN(k) = 0.
!>          العنصر IN(n) يعيد أصغر عدد صحيح موجب j بحيث
!>
!>             abs( u(j,j) ) <= norm( (T - lambda*I)(j) )*TOL،
!>
!>          حيث يشير norm( A(j) ) إلى مجموع القيم المطلقة للصف j من
!>          المصفوفة A. إذا لم يوجد مثل هذا j، فسيتم إعادة IN(n) كصفر.
!>          إذا تم إرجاع IN(n) بقيمة موجبة، فهذا يشير إلى أن عنصراً قطرياً
!>          من U صغير، مما يعني أن (T - lambda*I) منفردة أو قريبة من
!>          التفرد.
!>

INFO

!>          INFO هو عدد صحيح (INTEGER)
!>          = 0: خروج ناجح
!>          < 0: إذا كان INFO = -k، فإن المعامل k كان له قيمة غير قانونية
!>
