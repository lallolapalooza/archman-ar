المعلمات

ILSCHUR 
!>          ILSCHUR is LOGICAL
!>              يحدد ما إذا كان سيتم تحديث الشكل Schur الكامل أم لا
!>

ILQ

!>          ILQ is LOGICAL
!>              يحدد ما إذا كان سيتم تحديث المصفوفة Q أم لا
!>

ILZ

!>          ILZ is LOGICAL
!>              يحدد ما إذا كان سيتم تحديث المصفوفة Z أم لا
!>

N

!>          N is INTEGER
!>          ترتيب المصفوفات A، B، Q، و Z.  N >= 0.
!>

ILO

!>          ILO is INTEGER
!>

IHI

!>          IHI is INTEGER
!>          يحدد ILO و IHI الصفوف والأعمدة من (A,B) التي
!>          يجب تسويتها
!>

NW

!>          NW is INTEGER
!>          الحجم المطلوب لنافذة التفريغ.
!>

A

!>          A is REAL array, dimension (LDA, N)
!>

LDA

!>          LDA is INTEGER
!>          البعد الرئيسي للمصفوفة A.  LDA >= max( 1, N ).
!>

B

!>          B is REAL array, dimension (LDB, N)
!>

LDB

!>          LDB is INTEGER
!>          البعد الرئيسي للمصفوفة B.  LDB >= max( 1, N ).
!>

Q

!>          Q is REAL array, dimension (LDQ, N)
!>

LDQ

!>          LDQ is INTEGER
!>

Z

!>          Z is REAL array, dimension (LDZ, N)
!>

LDZ

!>          LDZ is INTEGER
!>

NS

!>          NS is INTEGER
!>          عدد القيم الذاتية غير المتقاربة المتاحة
!>          لاستخدامها كتحولات.
!>

ND

!>          ND is INTEGER
!>          عدد القيم الذاتية المتقاربة التي تم العثور عليها.
!>

ALPHAR

!>          ALPHAR is REAL array, dimension (N)
!>          الأجزاء الحقيقية لكل ثابت ألفا يحدد قيمة ذاتية
!>          لـ GNEP.
!>

ALPHAI

!>          ALPHAI is REAL array, dimension (N)
!>          الأجزاء التخيلية لكل ثابت ألفا يحدد قيمة ذاتية
!>          لـ GNEP.
!>          إذا كانت ALPHAI(j) صفرًا، فإن القيمة الذاتية j حقيقية؛ إذا
!>          كانت موجبة، فإن القيمتين الذاتيتين j و (j+1) هما زوج
!>          مركب مترافق، مع ALPHAI(j+1) = -ALPHAI(j).
!>

BETA

!>          BETA is REAL array, dimension (N)
!>          الثوابت بيتا التي تحدد القيم الذاتية لـ GNEP.
!>          معًا، الكميات ألفا = (ALPHAR(j),ALPHAI(j)) و
!>          بيتا = BETA(j) تمثل القيمة الذاتية j لزوج المصفوفات
!>          (A,B)، بأحد الأشكال lambda = alpha/beta أو
!>          mu = beta/alpha. بما أن lambda أو mu قد يحدث فيهما فيضان،
!>          لا ينبغي حسابهما، بشكل عام.
!>

QC

!>          QC is REAL array, dimension (LDQC, NW)
!>

LDQC

!>          LDQC is INTEGER
!>

ZC

!>          ZC is REAL array, dimension (LDZC, NW)
!>

LDZC

!>          LDZ is INTEGER
!>

WORK

!>          WORK is REAL array, dimension (MAX(1,LWORK))
!>          عند الخروج، إذا كان INFO >= 0، فإن WORK(1) يعيد LWORK المثلى.
!>

LWORK

!>          LWORK is INTEGER
!>          بعد المصفوفة WORK.  LWORK >= max(1,N).
!>
!>          إذا كان LWORK = -1، فسيتم افتراض استعلام مساحة العمل؛ الروتين
!>          يحسب فقط الحجم الأمثل لمصفوفة WORK، ويعيد
!>          هذه القيمة كالإدخال الأول لمصفوفة WORK، ولا تصدر رسالة خطأ
!>          متعلقة بـ LWORK بواسطة XERBLA.
!>

REC

!>          REC is INTEGER
!>             يشير REC إلى مستوى الاستدعاء الحالي. يجب تعيينه
!>             على 0 عند أول استدعاء.
!>

INFO

!>          INFO is INTEGER
!>          = 0: خروج ناجح
!>          < 0: إذا كان INFO = -i، فإن المعلمة i كانت لها قيمة غير قانونية
!>
