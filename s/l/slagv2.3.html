subroutine slagv2 (real, dimension( lda, * ) a, integer lda, real, dimension( ldb, * ) b, integer ldb, real, dimension( 2 ) alphar, real, dimension( 2 ) alphai, real, dimension( 2 ) beta, real csl, real snl, real csr, real snr)

SLAGV2 تقوم بحساب تحليل Schur المعمم لقلم مصفوفة حقيقي 2 × 2 (A,B) حيث B مثلثية علوية.

الغرض

!>
!> SLAGV2 تقوم بحساب تحليل Schur المعمم لقلم مصفوفة حقيقي 2 × 2
!> (A,B) حيث B مثلثية علوية. هذا الروتين
!> يحسب مصفوفتي دوران متعامدتين (تدوير) تُعطى بواسطة CSL، SNL و CSR،
!> SNR بحيث
!>
!> 1) إذا كان القلم (A,B) له قيمتان ذاتيان حقيقيان (بما في ذلك أنواع
!>    0/0 أو 1/0)، فإن
!>
!>    [ a11 a12 ] := [  CSL  SNL ] [ a11 a12 ] [  CSR -SNR ]
!>    [  0  a22 ]    [ -SNL  CSL ] [ a21 a22 ] [  SNR  CSR ]
!>
!>    [ b11 b12 ] := [  CSL  SNL ] [ b11 b12 ] [  CSR -SNR ]
!>    [  0  b22 ]    [ -SNL  CSL ] [  0  b22 ] [  SNR  CSR ],
!>
!> 2) إذا كان القلم (A,B) له زوج من القيم الذاتية المترافقة المعقدة،
!>    فإن
!>
!>    [ a11 a12 ] := [  CSL  SNL ] [ a11 a12 ] [  CSR -SNR ]
!>    [ a21 a22 ]    [ -SNL  CSL ] [ a21 a22 ] [  SNR  CSR ]
!>
!>    [ b11  0  ] := [  CSL  SNL ] [ b11 b12 ] [  CSR -SNR ]
!>    [  0  b22 ]    [ -SNL  CSL ] [  0  b22 ] [  SNR  CSR ]
!>
!>    حيث b11 >= b22 > 0.
!>
!>

المعلمات

A 
!>          A هي مصفوفة REAL، البعد (LDA, 2)
!>          عند الدخول، المصفوفة 2 × 2 A.
!>          عند الخروج، يتم استبدال A بـ "جزء A" من نموذج Schur المعمم.
!>

LDA

!>          LDA هو عدد صحيح
!>          البعد الرائد للمصفوفة A. LDA >= 2.
!>

B

!>          B هي مصفوفة REAL، البعد (LDB, 2)
!>          عند الدخول، المصفوفة المثلثية العلوية 2 × 2 B.
!>          عند الخروج، يتم استبدال B بـ "جزء B" من نموذج Schur المعمم.
!>

LDB

!>          LDB هو عدد صحيح
!>          البعد الرائد للمصفوفة B. LDB >= 2.
!>

ALPHAR

!>          ALPHAR هي مصفوفة REAL، البعد (2)
!>

ALPHAI

!>          ALPHAI هي مصفوفة REAL، البعد (2)
!>

BETA

!>          BETA هي مصفوفة REAL، البعد (2)
!>          (ALPHAR(k)+i*ALPHAI(k))/BETA(k) هي القيم الذاتية للقلم
!>          (A,B)، k=1,2، i = الجذر التربيعي لـ (-1). لاحظ أن BETA(k) قد
!>          يكون صفرًا.
!>

CSL

!>          CSL هو REAL
!>          جيب تمام مصفوفة الدوران اليسرى.
!>

SNL

!>          SNL هو REAL
!>          جيب مصفوفة الدوران اليسرى.
!>

CSR

!>          CSR هو REAL
!>          جيب تمام مصفوفة الدوران اليمنى.
!>

SNR

!>          SNR هو REAL
!>          جيب مصفوفة الدوران اليمنى.
!>
