subroutine slaqge (m, n, a, lda, r, c, rowcnd, colcnd, amax, equed)
SLAQGE تقوم بقياس مصفوفة مستطيلة عامة، باستخدام عوامل قياس الصف والعمود التي تم حسابها بواسطة sgeequ.

توثيق الوظيفة/الإجراء الفرعي

subroutine slaqge (integer m, integer n, real, dimension( lda, * ) a, integer lda, real, dimension( * ) r, real, dimension( * ) c, real rowcnd, real colcnd, real amax, character equed)

SLAQGE تقوم بقياس مصفوفة مستطيلة عامة، باستخدام عوامل قياس الصف والعمود التي تم حسابها بواسطة sgeequ.

الغرض:

!>
!> SLAQGE تقوم بموازنة مصفوفة عامة M في N باستخدام عوامل القياس الصفية والعمودية في المتجهين R و C.
!>

المعلمات

M 
!>          M هو عدد صحيح (INTEGER)
!>          عدد صفوف المصفوفة A.  M >= 0.
!>

N

!>          N هو عدد صحيح (INTEGER)
!>          عدد أعمدة المصفوفة A.  N >= 0.
!>

A

!>          A هي مصفوفة حقيقية (REAL array)، ذات بعد (LDA,N)
!>          عند الدخول: المصفوفة A البعد M في N.
!>          عند الخروج: المصفوفة المتوازنة. انظر EQUED لشكل المصفوفة المتوازنة.
!>

LDA

!>          LDA هو عدد صحيح (INTEGER)
!>          البعد الرائد للمصفوفة A.  LDA >= max(M,1).
!>

R

!>          R هي مصفوفة حقيقية (REAL array)، ذات بعد (M)
!>          عوامل قياس الصف للمصفوفة A.
!>

C

!>          C هي مصفوفة حقيقية (REAL array)، ذات بعد (N)
!>          عوامل قياس العمود للمصفوفة A.
!>

ROWCND

!>          ROWCND هو عدد حقيقي (REAL)
!>          نسبة أصغر R(i) إلى أكبر R(i).
!>

COLCND

!>          COLCND هو عدد حقيقي (REAL)
!>          نسبة أصغر C(i) إلى أكبر C(i).
!>

AMAX

!>          AMAX هو عدد حقيقي (REAL)
!>          القيمة المطلقة لأكبر عنصر في المصفوفة.
!>

EQUED

!>          EQUED هو حرف واحد (CHARACTER*1)
!>          يحدد شكل الموازنة التي تم إجراؤها.
!>          = 'N':  لا يوجد موازنة
!>          = 'R':  موازنة الصف، أي تم ضرب A مسبقًا بـ
!>                  diag(R).
!>          = 'C':  موازنة العمود، أي تم ضرب A لاحقًا بـ
!>                  diag(C).
!>          = 'B':  موازنة الصف والعمود، أي تم استبدال A بـ
!>                  diag(R) * A * diag(C).
!>

المعلمات الداخلية:

!>  THRESH هي قيمة حدية تستخدم لتحديد ما إذا كان يجب إجراء قياس الصف أو العمود بناءً على
!>  نسبة عوامل قياس الصف أو العمود.  إذا كان ROWCND < THRESH، يتم إجراء قياس الصف، وإذا كان
!>  COLCND < THRESH، يتم إجراء قياس العمود.
!>
!>  LARGE و SMALL هي قيم حدية تستخدم لتحديد ما إذا كان يجب إجراء قياس الصف
!>  بناءً على الحجم المطلق لأكبر عنصر في المصفوفة.  إذا كان AMAX > LARGE أو AMAX < SMALL، يتم إجراء قياس الصف.
!>
