وثائق الوظيفة/الروتين الفرعي

subroutine slahqr (logical wantt, logical wantz, integer n, integer ilo, integer ihi, real, dimension( ldh, * ) h, integer ldh, real, dimension( * ) wr, real, dimension( * ) wi, integer iloz, integer ihiz, real, dimension( ldz, * ) z, integer ldz, integer info)

SLAHQR تحسب القيم الذاتية والتحليل Schur لمصفوفة هسنبرغ علوية، باستخدام خوارزمية QR المزدوجة/المفردة.

الغرض:

!>
!>    SLAHQR هو روتين مساعد يتم استدعاؤه بواسطة SHSEQR لتحديث القيم الذاتية والتحليل Schur المحسوبة بالفعل بواسطة SHSEQR، عن طريق
!>    التعامل مع المصفوفة الفرعية لهسنبرغ في الصفوف والأعمدة ILO إلى IHI.
!>

المعلمات

WANTT 
!>          WANTT هو LOGICAL
!>          = .TRUE. : المطلوب هو الشكل Schur الكامل T؛
!>          = .FALSE.: مطلوب فقط القيم الذاتية.
!>

WANTZ

!>          WANTZ هو LOGICAL
!>          = .TRUE. : مطلوب مصفوفة متجهات Schur Z؛
!>          = .FALSE.: متجهات Schur غير مطلوبة.
!>

N

!>          N هو INTEGER
!>          ترتيب المصفوفة H. N >= 0.
!>

ILO

!>          ILO هو INTEGER
!>

IHI

!>          IHI هو INTEGER
!>          من المفترض أن تكون H بالفعل شبه مثلثية علوية في
!>          الصفوف والأعمدة IHI+1:N، وأن H(ILO,ILO-1) = 0 (ما لم
!>          يكن ILO = 1). تعمل SLAHQR بشكل أساسي مع المصفوفة الفرعية لهسنبرغ
!>          في الصفوف والأعمدة ILO إلى IHI، ولكنها تطبق
!>          تحويلات على H بأكملها إذا كان WANTT هو .TRUE..
!>          1 <= ILO <= max(1,IHI)؛ IHI <= N.
!>

H

!>          H هي مصفوفة REAL، أبعادها (LDH,N)
!>          عند الدخول، مصفوفة هسنبرغ العلوية H.
!>          عند الخروج، إذا كان INFO يساوي صفر وإذا كان WANTT هو .TRUE.، تكون H شبه مثلثية
!>          علوية في الصفوف والأعمدة ILO:IHI، مع أي
!>          كتل قطرية 2x2 في الشكل القياسي. إذا كان INFO يساوي صفر
!>          وكان WANTT هو .FALSE.، فإن محتويات H غير محددة عند الخروج.
!>          حالة الخروج لـ H إذا كان INFO غير صفري موصوفة أدناه تحت وصف INFO.
!>

LDH

!>          LDH هو INTEGER
!>          البعد القيادي للمصفوفة H. LDH >= max(1,N).
!>

WR

!>          WR هي مصفوفة REAL، أبعادها (N)
!>

WI

!>          WI هي مصفوفة REAL، أبعادها (N)
!>          يتم تخزين الأجزاء الحقيقية والتخيلية، على التوالي، للقيم الذاتية المحسوبة
!>          ILO إلى IHI في العناصر المقابلة في WR و WI. إذا تم حساب قيمتين ذاتيتين كزوج
!>          مترافق مركب، يتم تخزينهما في عناصر متتالية
!>          في WR و WI، مثل i و (i+1)، مع
!>          WI(i) > 0 و WI(i+1) < 0. إذا كان WANTT هو .TRUE.، فإن
!>          القيم الذاتية مخزنة بنفس ترتيب القطر
!>          ل لشكل Schur الذي تم إرجاعه في H، مع WR(i) = H(i,i)، وإذا
!>          كانت H(i:i+1,i:i+1) كتلة قطرية 2x2،
!>          WI(i) = sqrt(H(i+1,i)*H(i,i+1)) و WI(i+1) = -WI(i).
!>

ILOZ

!>          ILOZ هو INTEGER
!>

IHIZ

!>          IHIZ هو INTEGER
!>          حدد الصفوف في Z التي يجب تطبيق التحويلات عليها إذا كان WANTZ هو .TRUE..
!>          1 <= ILOZ <= ILO؛ IHI <= IHIZ <= N.
!>

Z

!>          Z هي مصفوفة REAL، أبعادها (LDZ,N)
!>          إذا كان WANTZ هو .TRUE.، عند الدخول يجب أن تحتوي Z على
!>          مصفوفة التحويلات الحالية Z التي تراكمت بواسطة SHSEQR، وعند
!>          الخروج تم تحديثها؛ يتم تطبيق التحويلات فقط على
!>          المصفوفة الفرعية Z(ILOZ:IHIZ,ILO:IHI).
!>          إذا كان WANTZ هو .FALSE.، فلا يتم الإشارة إلى Z.
!>

LDZ

!>          LDZ هو INTEGER
!>          البعد القيادي للمصفوفة Z. LDZ >= max(1,N).
!>

INFO

!>          INFO هو INTEGER
!>           = 0:   خروج ناجح
!>           > 0:   إذا كان INFO = i، فشلت SLAHQR في حساب جميع
!>                  القيم الذاتية ILO إلى IHI في إجمالي 30 تكرارًا
!>                  لكل قيمة ذاتية؛ العناصر i+1:ihi من WR و WI
!>                  تحتوي على تلك القيم الذاتية التي تم
!>                  حسابها بنجاح.
!>
!>                  إذا كان INFO > 0 وكان WANTT هو .FALSE.، فعند الخروج،
!>                  القيم الذاتية المتبقية غير المتقاربة هي
!>                  القيم الذاتية لمصفوفة هسنبرغ العلوية في
!>                  الصفوف والأعمدة ILO إلى INFO من القيمة النهائية، الخارجة
!>                  من H.
!>
!>                  إذا كان INFO > 0 وكان WANTT هو .TRUE.، فعند الخروج
!>          (*)       (القيمة الأولية لـ H)*U  = U*(القيمة النهائية لـ H)
!>                  حيث U هي مصفوفة متعامدة. القيمة النهائية
!>                  لـ H هي هسنبرغ علوية ومثلثية في
!>                  الصفوف والأعمدة INFO+1 إلى IHI.
!>
!>                  إذا كان INFO > 0 وكان WANTZ هو .TRUE.، فعند الخروج
!>                      (القيمة النهائية لـ Z)  = (القيمة الأولية لـ Z)*U
!>                  حيث U هي المصفوفة المتعامدة في (*)
!>                  (بغض النظر عن قيمة WANTT.)
!>
