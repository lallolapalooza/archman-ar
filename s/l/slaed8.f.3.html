subroutine slaed8 (integer icompq, integer k, integer n, integer qsiz, real, dimension( * ) d, real, dimension( ldq, * ) q, integer ldq, integer, dimension( * ) indxq, real rho, integer cutpnt, real, dimension( * ) z, real, dimension( * ) dlambda, real, dimension( ldq2, * ) q2, integer ldq2, real, dimension( * ) w, integer, dimension( * ) perm, integer givptr, integer, dimension( 2, * ) givcol, real, dimension( 2, * ) givnum, integer, dimension( * ) indxp, integer, dimension( * ) indx, integer info)

SLAED8 تستخدم بواسطة SSTEDC. تدمج القيم الذاتية وتُسطِّح المعادلة العلمانية. تستخدم عندما تكون المصفوفة الأصلية كثيفة.

الغرض:

!>
!> SLAED8 تدمج مجموعتي القيم الذاتية معًا في مجموعة مرتبة واحدة. ثم تحاول
!> تسطيح حجم المشكلة. هناك طريقتان يمكن أن يحدث بهما التسطيح: عندما تكون
!> قيمتان ذاتيتان أو أكثر قريبتين من بعضهما البعض أو إذا كان هناك عنصر صغير جدًا
!> في المتجه Z. لكل حدوث من هذا القبيل، يتم تقليل رتبة مشكلة المعادلة العلمانية
!> ذات الصلة بمقدار واحد.
!>

المعلمات

ICOMPQ 
!>          ICOMPQ هو عدد صحيح (INTEGER)
!>          = 0:  حساب القيم الذاتية فقط.
!>          = 1:  حساب المتجهات الذاتية للمصفوفة المتماثلة الكثيفة الأصلية
!>                أيضًا. عند الإدخال، تحتوي Q على المصفوفة المتعامدة المستخدمة
!>                لتقليل المصفوفة الأصلية إلى شكل ثلاثي الأقطار.
!>

K

!>          K هو عدد صحيح (INTEGER)
!>         عدد القيم الذاتية غير المسطحة، ورتبة المعادلة العلمانية
!>         ذات الصلة.
!>

N

!>          N هو عدد صحيح (INTEGER)
!>         بعد المصفوفة المتماثلة ثلاثية الأقطار. N >= 0.
!>

QSIZ

!>          QSIZ هو عدد صحيح (INTEGER)
!>         بعد المصفوفة المتعامدة المستخدمة لتقليل
!>         المصفوفة الكاملة إلى شكل ثلاثي الأقطار. QSIZ >= N إذا كان ICOMPQ = 1.
!>

D

!>          D هو مصفوفة حقيقية (REAL array)، البعد (N)
!>         عند الإدخال، القيم الذاتية للمصفوفتين الفرعيتين المراد
!>         دمجهما. عند الخروج، القيم الذاتية المحدثة اللاحقة (N-K)
!>         (تلك التي تم تسطيحها) مرتبة بترتيب تصاعدي.
!>

Q

!>          Q هو مصفوفة حقيقية (REAL array)، البعد (LDQ,N)
!>         إذا كان ICOMPQ = 0، لا يتم الرجوع إلى Q. خلاف ذلك،
!>         عند الإدخال، تحتوي Q على المتجهات الذاتية للنظام
!>         الذي تم حله جزئيًا والذي تم تحديثه سابقًا في ضربات
!>         المصفوفة مع أنظمة ذاتية أخرى تم حلها جزئيًا.
!>         عند الخروج، تحتوي Q على المتجهات الذاتية المحدثة اللاحقة (N-K)
!>         (تلك التي تم تسطيحها) في أعمدتها الأخيرة N-K.
!>

LDQ

!>          LDQ هو عدد صحيح (INTEGER)
!>         البعد الرائد للمصفوفة Q. LDQ >= max(1,N).
!>

INDXQ

!>          INDXQ هو مصفوفة أعداد صحيحة (INTEGER array)، البعد (N)
!>         التبديل الذي يرتب بشكل منفصل المشكلتين الفرعيتين
!>         في D بترتيب تصاعدي. لاحظ أن العناصر في النصف الثاني
!>         من هذا التبديل يجب أن يكون لها CUTPNT مضافة إلى
!>         قيمها لتكون دقيقة.
!>

RHO

!>          RHO هو قيمة حقيقية (REAL)
!>         عند الإدخال، العنصر خارج القطر المرتبط بالقطع ذي الرتبة 1
!>         الذي قسم في الأصل المصفوفتين الفرعيتين اللتين يتم
!>         إعادة دمجهما الآن.
!>         عند الخروج، تم تعديل RHO إلى القيمة المطلوبة بواسطة
!>         SLAED3.
!>

CUTPNT

!>          CUTPNT هو عدد صحيح (INTEGER)
!>         موقع القيمة الذاتية الأخيرة في المصفوفة الفرعية الرائدة.
!>         min(1,N) <= CUTPNT <= N.
!>

Z

!>          Z هو مصفوفة حقيقية (REAL array)، البعد (N)
!>         عند الإدخال، تحتوي Z على المتجه المحدث (الصف الأخير
!>         من مصفوفة المتجهات الذاتية الفرعية الأولى والصف الأول
!>         من مصفوفة المتجهات الذاتية الفرعية الثانية).
!>         عند الخروج، يتم تدمير محتويات Z بواسطة عملية التحديث.
!>

DLAMBDA

!>          DLAMBDA هو مصفوفة حقيقية (REAL array)، البعد (N)
!>         نسخة من أول K قيمة ذاتية سيتم استخدامها بواسطة
!>         SLAED3 لتشكيل المعادلة العلمانية.
!>

Q2

!>          Q2 هو مصفوفة حقيقية (REAL array)، البعد (LDQ2,N)
!>         إذا كان ICOMPQ = 0، لا يتم الرجوع إلى Q2. خلاف ذلك،
!>         نسخة من أول K متجه ذاتي سيتم استخدامها بواسطة
!>         SLAED7 في ضرب المصفوفة (SGEMM) لتحديث المتجهات الذاتية الجديدة.
!>

LDQ2

!>          LDQ2 هو عدد صحيح (INTEGER)
!>         البعد الرائد للمصفوفة Q2. LDQ2 >= max(1,N).
!>

W

!>          W هو مصفوفة حقيقية (REAL array)، البعد (N)
!>         أول K قيم للمتجه Z المعدل النهائي بالتسطيح وسيتم
!>         تمريرها إلى SLAED3.
!>

PERM

!>          PERM هو مصفوفة أعداد صحيحة (INTEGER array)، البعد (N)
!>         التبديلات (من التسطيح والترتيب) التي سيتم تطبيقها
!>         على كل كتلة ذاتية.
!>

GIVPTR

!>          GIVPTR هو عدد صحيح (INTEGER)
!>         عدد دورانات جيفنز التي حدثت في هذه المشكلة الفرعية.
!>

GIVCOL

!>          GIVCOL هو مصفوفة أعداد صحيحة (INTEGER array)، البعد (2, N)
!>         كل زوج من الأرقام يشير إلى زوج من الأعمدة التي ستحدث
!>         في دوران جيفنز.
!>

GIVNUM

!>          GIVNUM هو مصفوفة حقيقية (REAL array)، البعد (2, N)
!>         كل رقم يشير إلى قيمة S التي سيتم استخدامها في
!>         دوران جيفنز المقابل.
!>

INDXP

!>          INDXP هو مصفوفة أعداد صحيحة (INTEGER array)، البعد (N)
!>         التبديل المستخدم لوضع القيم المسطحة لـ D في نهاية
!>         المصفوفة. INDXP(1:K) يشير إلى قيم D غير المسطحة
!>         و INDXP(K+1:N) يشير إلى القيم الذاتية المسطحة.
!>

INDX

!>          INDX هو مصفوفة أعداد صحيحة (INTEGER array)، البعد (N)
!>         التبديل المستخدم لترتيب محتويات D بترتيب تصاعدي.
!>

INFO

!>          INFO هو عدد صحيح (INTEGER)
!>          = 0:  خروج ناجح.
!>          < 0:  إذا كان INFO = -i، كان الوسيط i له قيمة غير قانونية.
!>
