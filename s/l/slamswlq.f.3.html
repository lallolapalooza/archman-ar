وثائق الوظيفة/الروتين الفرعي

subroutine slamswlq (character side, character trans, integer m, integer n, integer k, integer mb, integer nb, real, dimension( lda, * ) a, integer lda, real, dimension( ldt, * ) t, integer ldt, real, dimension( ldc, * ) c, integer ldc, real, dimension( * ) work, integer lwork, integer info)

SLAMSWLQ

الغرض:

!>
!>    SLAMSWLQ يستبدل مصفوفة C الحقيقية العامة M-by-N بـ
!>
!>
!>                    SIDE = 'L'     SIDE = 'R'
!>    TRANS = 'N':      Q * C          C * Q
!>    TRANS = 'T':      Q**T * C       C * Q**T
!>    حيث Q مصفوفة متعامدة حقيقية معرفة كناتج للمستقلين الأوليين المحصورين
!>    التي تم حسابها بواسطة التحلل الضيق الواسع (SLASWLQ)
!>

المعلمات:

SIDE 
!>          SIDE is CHARACTER*1
!>          = 'L': طبق Q أو Q**T من اليسار؛
!>          = 'R': طبق Q أو Q**T من اليمين.
!>

TRANS

!>          TRANS is CHARACTER*1
!>          = 'N':  بدون تبديل، طبق Q؛
!>          = 'T':  تبديل، طبق Q**T.
!>

M

!>          M is INTEGER
!>          عدد صفوف المصفوفة C. M >=0.
!>

N

!>          N is INTEGER
!>          عدد أعمدة المصفوفة C. N >= 0.
!>

K

!>          K is INTEGER
!>          عدد المستقلين الأوليين الذين يحدد حاصل ضربهم
!>          المصفوفة Q.
!>          M >= K >= 0;
!>
!>

MB

!>          MB is INTEGER
!>          حجم الكتلة للصف المستخدم في LQ المحصور.
!>          M >= MB >= 1
!>

NB

!>          NB is INTEGER
!>          حجم الكتلة للعمود المستخدم في LQ المحصور.
!>          NB > M.
!>

A

!>          A is REAL array, dimension
!>                               (LDA,M) if SIDE = 'L',
!>                               (LDA,N) if SIDE = 'R'
!>          يجب أن يحتوي الصف i على المتجه الذي يحدد المستقل المحصور
!>          H(i)، للصف i = 1,2,...,k، كما تم إرجاعه بواسطة
!>          SLASWLQ في الصفوف k الأولى من مصفوفة الوسيط A.
!>

LDA

!>          LDA is INTEGER
!>          البعد الرائد للمصفوفة A. LDA >= max(1,K).
!>

T

!>          T is REAL array, dimension
!>          ( M * Number of blocks(CEIL(N-K/NB-K)),
!>          مستقلات الكتلة المثلثية العلوية المحصورة المخزنة في شكل مضغوط
!>          كسلسلة من الكتل المثلثية العلوية. انظر أدناه
!>          لمزيد من التفاصيل.
!>

LDT

!>          LDT is INTEGER
!>          البعد الرائد للمصفوفة T. LDT >= MB.
!>

C

!>          C is REAL array, dimension (LDC,N)
!>          عند الدخول، المصفوفة C حجمها M-by-N.
!>          عند الخروج، يتم استبدال C بـ Q*C أو Q**T*C أو C*Q**T أو C*Q.
!>

LDC

!>          LDC is INTEGER
!>          البعد الرائد للمصفوفة C. LDC >= max(1,M).
!>

WORK

!>          (مساحة عمل) REAL array, dimension (MAX(1,LWORK))
!>          عند الخروج، إذا كان INFO = 0، فإن WORK(1) ترجع الحد الأدنى LWORK.
!>

LWORK

!>          LWORK is INTEGER
!>          بعد مصفوفة WORK.
!>
!>          إذا كان MIN(M,N,K) = 0، فإن LWORK >= 1.
!>          إذا كان SIDE = 'L'، فإن LWORK >= max(1,NB*MB).
!>          إذا كان SIDE = 'R'، فإن LWORK >= max(1,M*MB).
!>          إذا كان LWORK = -1، فسيتم افتراض استعلام مساحة العمل؛ الروتين
!>          يحسب فقط الحد الأدنى لحجم مصفوفة WORK، ويرجع
!>          هذه القيمة كالإدخال الأول لمصفوفة WORK، ولا يتم إصدار أي رسالة خطأ
!>          بسبب LWORK بواسطة XERBLA.
!>

INFO

!>          INFO is INTEGER
!>          = 0:  انتهاء ناجح
!>          < 0:  إذا كان INFO = -i، كان الوسيط i يحتوي على قيمة غير قانونية
!>

تفاصيل إضافية:

!> يطبق تحليل LQ الضيق الواسع (SWLQ) LQ بواسطة سلسلة من التحويلات المتعامدة،
!> يمثل Q كناتج لمصفوفات متعامدة أخرى
!>   Q = Q(1) * Q(2) * . . . * Q(k)
!> حيث يقوم كل Q(i) بتحويل الإدخالات القطرية العلوية إلى صفر لكتلة من صفوف NB من A:
!>   Q(1) يقوم بتحويل الإدخالات القطرية العلوية للصفوف 1:NB من A إلى صفر
!>   Q(2) يقوم بتحويل الصفوف MB-N السفلية للصفوف [1:M,NB+1:2*NB-M] من A إلى صفر
!>   Q(3) يقوم بتحويل الصفوف MB-N السفلية للصفوف [1:M,2*NB-M+1:3*NB-2*M] من A إلى صفر
!>   . . .
!>
!> يتم حساب Q(1) بواسطة GELQT، الذي يمثل Q(1) بواسطة متجهات هاوسهولدر
!> المخزنة تحت القطر في الصفوف 1:MB من A، وبواسطة كتل مثلثية علوية
!> مستقلات، مخزنة في المصفوفة T(1:LDT,1:N).
!> لمزيد من المعلومات، انظر التفاصيل الإضافية في GELQT.
!>
!> يتم حساب Q(i) لـ i>1 بواسطة TPLQT، الذي يمثل Q(i) بواسطة متجهات هاوسهولدر
!> المخزنة في الأعمدة [(i-1)*(NB-M)+M+1:i*(NB-M)+M] من A، وبواسطة كتل مثلثية علوية
!> مستقلات، مخزنة في المصفوفة T(1:LDT,(i-1)*M+1:i*M).
!> قد تستخدم Q(k) الأخيرة عددًا أقل من الصفوف.
!> لمزيد من المعلومات، انظر التفاصيل الإضافية في TPLQT.
!>
!> لمزيد من التفاصيل حول الخوارزمية الشاملة، راجع وصف
!> التحليلات المتوازية والتسلسلية المتوافقة مع الاتصالات QR و LU
!> في القسم 2.2 من [1].
!>
!> [1] “Communication-Optimal Parallel and Sequential QR and LU Factorizations,”
!>     J. Demmel, L. Grigori, M. Hoemmen, J. Langou,
!>     SIAM J. Sci. Comput, vol. 34, no. 1, 2012
!>
