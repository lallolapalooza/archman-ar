subroutine slagsy (integer n, integer k, real, dimension( * ) d, real, dimension( lda, * ) a, integer lda, integer, dimension( 4 ) iseed, real, dimension( * ) work, integer info)

SLAGSY

الغرض:

!>
!> SLAGSY تقوم بإنشاء مصفوفة متماثلة حقيقية A، عن طريق الضرب المسبق واللاحق
!> لمصفوفة قطرية حقيقية D بمصفوفة متعامدة عشوائية:
!> A = U*D*U'. ثم يمكن تقليل عرض النطاق شبه إلى k عن طريق تحويلات متعامدة إضافية.
!>

المعاملات

N 
!>          N هو عدد صحيح (INTEGER)
!>          مرتبة المصفوفة A.  N >= 0.
!>

K

!>          K هو عدد صحيح (INTEGER)
!>          عدد الأقطار الفرعية غير الصفرية داخل نطاق A.
!>          0 <= K <= N-1.
!>

D

!>          D هو مصفوفة حقيقية (REAL array)، البعد (N)
!>          العناصر القطرية للمصفوفة القطرية D.
!>

A

!>          A هي مصفوفة حقيقية (REAL array)، البعد (LDA,N)
!>          المصفوفة المتماثلة n x n المولدة A (يتم تخزين المصفوفة الكاملة).
!>

LDA

!>          LDA هو عدد صحيح (INTEGER)
!>          البعد الرائد للمصفوفة A.  LDA >= N.
!>

ISEED

!>          ISEED هي مصفوفة أعداد صحيحة (INTEGER array)، البعد (4)
!>          عند الإدخال، البذرة لمولد الأرقام العشوائية؛ يجب أن تكون عناصر المصفوفة بين 0 و 4095، ويجب أن يكون ISEED(4)
!>          فرديًا.
!>          عند الخروج، يتم تحديث البذرة.
!>

WORK

!>          WORK هي مصفوفة حقيقية (REAL array)، البعد (2*N)
!>

INFO

!>          INFO هو عدد صحيح (INTEGER)
!>          = 0: الخروج بنجاح
!>          < 0: إذا كان INFO = -i، فإن الوسيط i كان له قيمة غير قانونية
!>
