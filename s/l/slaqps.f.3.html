subroutine slaqps (integer m, integer n, integer offset, integer nb, integer kb, real, dimension( lda, * ) a, integer lda, integer, dimension( * ) jpvt, real, dimension( * ) tau, real, dimension( * ) vn1, real, dimension( * ) vn2, real, dimension( * ) auxv, real, dimension( ldf, * ) f, integer ldf)

SLAQPS تقوم بحساب خطوة من تحليل QR مع تبديل الأعمدة لمصفوفة حقيقية m-by-n A باستخدام BLAS مستوى 3.

الغرض:

!>
!> SLAQPS يقوم بحساب خطوة من تحليل QR مع تبديل الأعمدة
!> لمصفوفة حقيقية M-by-N A باستخدام Blas-3.  يحاول تحليل
!> NB عمودًا من A بدءًا من الصف OFFSET+1، ويقوم بتحديث كل
!> المصفوفة باستخدام Blas-3 xGEMM.
!>
!> في بعض الحالات، بسبب الإلغاءات الكارثية، لا يمكنه
!> تحليل NB عمودًا.  لذلك، يتم إرجاع العدد الفعلي للأعمدة المحللة
!> في KB.
!>
!> الجزء A(1:OFFSET,1:N) يتم تبديله وفقًا لذلك، ولكنه غير محلل.
!>

المعلمات

M 
!>          M هو عدد صحيح INTEGER
!>          عدد صفوف المصفوفة A. M >= 0.
!>

N

!>          N هو عدد صحيح INTEGER
!>          عدد أعمدة المصفوفة A. N >= 0
!>

OFFSET

!>          OFFSET هو عدد صحيح INTEGER
!>          عدد صفوف A التي تم تحليلها في
!>          الخطوات السابقة.
!>

NB

!>          NB هو عدد صحيح INTEGER
!>          عدد الأعمدة المراد تحليلها.
!>

KB

!>          KB هو عدد صحيح INTEGER
!>          عدد الأعمدة التي تم تحليلها فعليًا.
!>

A

!>          A هو مصفوفة حقيقية REAL array, dimension (LDA,N)
!>          عند الدخول، المصفوفة A بحجم M-by-N.
!>          عند الخروج، الجزء A(OFFSET+1:M,1:KB) هو الجزء المثلثي
!>          المحصل عليه والجزء A(1:OFFSET,1:N) تم تبديله
!>          وفقًا لذلك، ولكنه لم يتم تحليله.
!>          بقية المصفوفة، الجزء A(OFFSET+1:M,KB+1:N) تم
!>          تحديثه.
!>

LDA

!>          LDA هو عدد صحيح INTEGER
!>          البعد الرئيسي للمصفوفة A. LDA >= max(1,M).
!>

JPVT

!>          JPVT هي مصفوفة أعداد صحيحة INTEGER array, dimension (N)
!>          JPVT(I) = K <==> العمود K من المصفوفة الكاملة A تم
!>          تبديله إلى الموضع I في AP.
!>

TAU

!>          TAU هي مصفوفة حقيقية REAL array, dimension (KB)
!>          العوامل القياسية للمنشئات الأولية.
!>

VN1

!>          VN1 هي مصفوفة حقيقية REAL array, dimension (N)
!>          المتجه بمعايير الأعمدة الجزئية.
!>

VN2

!>          VN2 هي مصفوفة حقيقية REAL array, dimension (N)
!>          المتجه بمعايير الأعمدة الدقيقة.
!>

AUXV

!>          AUXV هي مصفوفة حقيقية REAL array, dimension (NB)
!>          متجه مساعد.
!>

F

!>          F هي مصفوفة حقيقية REAL array, dimension (LDF,NB)
!>          المصفوفة F**T = L*Y**T*A.
!>

LDF

!>          LDF هو عدد صحيح INTEGER
!>          البعد الرئيسي للمصفوفة F. LDF >= max(1,N).
!>
