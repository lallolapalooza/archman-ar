subroutine slaebz (integer ijob, integer nitmax, integer n, integer mmax, integer minp, integer nbmin, real abstol, real reltol, real pivmin, real, dimension( * ) d, real, dimension( * ) e, real, dimension( * ) e2, integer, dimension( * ) nval, real, dimension( mmax, * ) ab, real, dimension( * ) c, integer mout, integer, dimension( mmax, * ) nab, real, dimension( * ) work, integer, dimension( * ) iwork, integer info)

SLAEBZ تحسب عدد القيم الذاتية لمصفوفة ثلاثية القطر متماثلة حقيقية والتي هي أقل من أو تساوي قيمة معينة، وتقوم بمهام أخرى مطلوبة بواسطة الروتين sstebz.

الغرض:

!>
!> SLAEBZ يحتوي على حلقات التكرار التي تحسب وتستخدم الدالة N(w)،
!> وهي عدد القيم الذاتية لمصفوفة ثلاثية القطر متماثلة T الأقل من أو
!> تساوي الوسيط w. يقوم بعمل اختيار من بين نوعين من الحلقات:
!>
!> IJOB=1، يتبعه
!> IJOB=2: يأخذ كمدخل قائمة من الفترات ويعيد قائمة من الفترات الصغيرة
!>         بشكل كافٍ والتي تحتوي وحدتها على نفس القيم الذاتية لوحدة
!>         الفترات الأصلية.
!>         الفترات المدخلة هي (AB(j,1),AB(j,2)]، j=1,...,MINP.
!>         الفترة المخرجة (AB(j,1),AB(j,2)] ستحتوي على
!>         القيم الذاتية NAB(j,1)+1,...,NAB(j,2)، حيث 1 <= j <= MOUT.
!>
!> IJOB=3: يقوم ببحث ثنائي في كل فترة مدخلة
!>         (AB(j,1),AB(j,2)] لنقطة w(j) بحيث
!>         N(w(j))=NVAL(j)، ويستخدم C(j) كنقطة بداية
!>         للبحث. إذا تم العثور على مثل هذه w(j)، فعند الخرج
!>         AB(j,1)=AB(j,2)=w. إذا لم يتم العثور على مثل هذه w(j)، فعند الخرج
!>         ستكون (AB(j,1),AB(j,2)] فترة صغيرة تحتوي على
!>         النقطة التي تقفز فيها N(w) خلال NVAL(j)، ما لم تكن تلك النقطة
!>         خارج الفترة الأولية.
!>
!> لاحظ أن الفترات في جميع الحالات هي فترات نصف مفتوحة،
!> أي على شكل (a,b]، والتي تشمل b ولكن ليس a.
!>
!> لتجنب الانهيار، يجب قياس المصفوفة بحيث لا يتجاوز أكبر عنصر فيها
!> قيمة overflow**(1/2) * underflow**(1/4)
!> بالقيمة المطلقة. لضمان أدق حساب
!> للقيم الذاتية الصغيرة، يجب قياس المصفوفة بحيث لا تكون أصغر بكثير
!> من ذلك.
!>
!> انظر W. Kahan، تقرير CS41، قسم علوم الكمبيوتر، جامعة ستانفورد
!> ، 21 يوليو 1966
!>
!> ملاحظة: الوسائط، بشكل عام، *لا* يتم التحقق منها للقيم غير المعقولة.
!>

SLAEBZ(3) — LAPACK — Arch Linux

NAME

SLAEBZ — Finds eigenvalues in a specific interval (a,b] for a symmetric or Hermitian tridiagonal matrix.

SYNOPSIS

 SUBROUTINE SLAEBZ(
  IJOB, NAB, MINIM, MMAX, NBANDS,
  RTOL, ABSTOL, M, NZ,
  WORK, IWORK, INFO
 )

 IMPLICIT NONE

 INTEGER IJOB, NAB(*), MINIM, MMAX, NBANDS, M, NZ, INFO
 REAL RTOL, ABSTOL
 REAL WORK(*), IWORK(*)

Function/Subroutine Documentation

!>
!>  IJOB : INTEGER
!>         Specifies the operation to be performed:
!>         = 1: Compute NAB(i,1) and NAB(i,2) for each interval i.
!>              NAB(i,1) is the number of eigenvalues less than or equal to AB(i,1),
!>              and NAB(i,2) is the number of eigenvalues less than or equal to AB(i,2).
!>              If N is the total number of eigenvalues, the number of eigenvalues
!>              in the interval (AB(i,1), AB(i,2)] is NAB(i,2) - NAB(i,1).
!>              The input values of NAB are ignored.
!>              On output, M is the number of intervals, and NZ is the total number of
!>              eigenvalues in all intervals.
!>         = 2: Compute approximate eigenvalues for intervals where NAB(i,2) - NAB(i,1) > 1.
!>              The intervals are updated such that eigenvalues satisfy
!>              NAB(i,1) <= NAB(i,1) <= NAB(i,2) <= NAB(i,2),
!>              which means that in interval i only eigenvalues
!>              NAB(i,1)+1,...,NAB(i,2) will be considered.  Usually,
!>              NAB(i,j)=N(AB(i,j)), from a previous call to SLAEBZ with
!>              IJOB=1.
!>              On output, NAB(i,j) will contain
!>              max(na(k),min(nb(k),N(AB(i,j)))), where k is the index of
!>              the input interval that the output interval
!>              (AB(j,1),AB(j,2)] came from, and na(k) and nb(k) are the
!>              the input values of NAB(k,1) and NAB(k,2).
!>          If IJOB=3, then on output, NAB(i,j) contains N(AB(i,j)),
!>             unless N(w) > NVAL(i) for all search points  w , in which
!>             case NAB(i,1) will not be modified, i.e., the output
!>             value will be the same as the input value (modulo
!>             reorderings -- see NVAL and AB), or unless N(w) < NVAL(i)
!>             for all search points  w , in which case NAB(i,2) will
!>             not be modified.  Normally, NAB should be set to some
!>             distinctive value(s) before SLAEBZ is called.
!>
!>  NAB    : INTEGER array, dimension (MMAX, 2)
!>         On input, if IJOB=2 or 3, NAB(i,1) and NAB(i,2) specify the
!>         desired range of eigenvalues for the i-th interval.
!>         On output, NAB(i,1) and NAB(i,2) contain the updated range.
!>
!>  MINIM  : INTEGER
!>         The minimum number of intervals to be generated.
!>
!>  MMAX   : INTEGER
!>         The maximum number of intervals to be generated.
!>
!>  NBANDS : INTEGER
!>         The number of bands in the tridiagonal matrix. (Not used in this routine,
!>         but part of the LAPACK interface for tridiagonal matrices).
!>
!>  RTOL   : REAL
!>         Relative tolerance for convergence.
!>
!>  ABSTOL : REAL
!>         Absolute tolerance for convergence.
!>
!>  M      : INTEGER
!>         On output, the number of intervals generated.
!>
!>  NZ     : INTEGER
!>         On output, the total number of eigenvalues in all intervals.
!>
!>  WORK   : REAL array, dimension (MMAX)
!>         Workspace.
!>
!>  IWORK  : INTEGER array, dimension (MMAX)
!>         Workspace.
!>
!>  INFO   : INTEGER
!>         = 0:       All intervals converged.
!>         = 1--MMAX: The last INFO intervals did not converge.
!>         = MMAX+1:  More than MMAX intervals were generated.
!>

Further Details:

!>
!>      This routine is intended to be called only by other LAPACK
!>  routines, thus the interface is less user-friendly.  It is intended
!>  for two purposes:
!>
!>  (a) finding eigenvalues.  In this case, SLAEBZ should have one or
!>      more initial intervals set up in AB, and SLAEBZ should be called
!>      with IJOB=1.  This sets up NAB, and also counts the eigenvalues.
!>      Intervals with no eigenvalues would usually be thrown out at
!>      this point.  Also, if not all the eigenvalues in an interval i
!>      are desired, NAB(i,1) can be increased or NAB(i,2) decreased.
!>      For example, set NAB(i,1)=NAB(i,2)-1 to get the largest
!>      eigenvalue.  SLAEBZ is then called with IJOB=2 and MMAX
!>      no smaller than the value of MOUT returned by the call with
!>      IJOB=1.  After this (IJOB=2) call, eigenvalues NAB(i,1)+1
!>      through NAB(i,2) are approximately AB(i,1) (or AB(i,2)) to the
!>      tolerance specified by ABSTOL and RELTOL.
!>
!>  (b) finding an interval (a',b'] containing eigenvalues w(f),...,w(l).
!>      In this case, start with a Gershgorin interval  (a,b).  Set up
!>      AB to contain 2 search intervals, both initially (a,b).  One
!>      NVAL element should contain  f-1  and the other should contain  l
!>      , while C should contain a and b, resp.  NAB(i,1) should be -1
!>      and NAB(i,2) should be N+1, to flag an error if the desired
!>      interval does not lie in (a,b).  SLAEBZ is then called with
!>      IJOB=3.  On exit, if w(f-1) < w(f), then one of the intervals --
!>      j -- will have AB(j,1)=AB(j,2) and NAB(j,1)=NAB(j,2)=f-1, while
!>      if, to the specified tolerance, w(f-k)=...=w(f+r), k > 0 and r
!>      >= 0, then the interval will have  N(AB(j,1))=NAB(j,1)=f-k and
!>      N(AB(j,2))=NAB(j,2)=f+r.  The cases w(l) < w(l+1) and
!!>      w(l-r)=...=w(l+k) are handled similarly.
!>
