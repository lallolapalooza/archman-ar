subroutine slasd6 (integer icompq, integer nl, integer nr, integer sqre, real, dimension( * ) d, real, dimension( * ) vf, real, dimension( * ) vl, real alpha, real beta, integer, dimension( * ) idxq, integer, dimension( * ) perm, integer givptr, integer, dimension( ldgcol, * ) givcol, integer ldgcol, real, dimension( ldgnum, * ) givnum, integer ldgnum, real, dimension( ldgnum, * ) poles, real, dimension( * ) difl, real, dimension( * ) difr, real, dimension( * ) z, integer k, real c, real s, real, dimension( * ) work, integer, dimension( * ) iwork, integer info)

SLASD6 تحسب SVD لمصفوفة ثنائية عليا محدثة تم الحصول عليها عن طريق دمج مصفوفتين أصغر عن طريق إلحاق صف. تستخدم بواسطة sbdsdc.

الغرض:

!>
!> SLASD6 تحسب SVD لمصفوفة ثنائية عليا محدثة B
!> تم الحصول عليها عن طريق دمج مصفوفتين أصغر عن طريق إلحاق صف. هذه
!> الوحدة تستخدم فقط للمسألة التي تتطلب جميع القيم الفردية
!> واختياريا مصفوفات المتجهات الفردية في شكل مضروب.
!> B هي مصفوفة N-by-M حيث N = NL + NR + 1 و M = N + SQRE.
!> وحدة فرعية ذات صلة، SLASD1، تتعامل مع الحالة التي تكون فيها جميع القيم الفردية
!> والمتجهات الفردية للمصفوفة الثنائية مطلوبة.
!>
!> SLASD6 تحسب SVD على النحو التالي:
!>
!>               ( D1(in)    0    0       0 )
!>   B = U(in) * (   Z1**T   a   Z2**T    b ) * VT(in)
!>               (   0       0   D2(in)   0 )
!>
!>     = U(out) * ( D(out) 0) * VT(out)
!>
!> حيث Z**T = (Z1**T a Z2**T b) = u**T VT**T، و u هو متجه من البعد M
!> مع ALPHA و BETA في المدخلات NL+1 و NL+2 وأصفار
!> في أماكن أخرى؛ والإدخال b فارغ إذا كان SQRE = 0.
!>
!> يمكن حساب القيم الفردية لـ B باستخدام D1، D2، المكونات الأولى
!> لجميع المتجهات الفردية اليمنى للكتلة السفلية، و
!> المكونات الأخيرة لجميع المتجهات الفردية اليمنى للكتلة العلوية. هذه المكونات مخزنة
!> ويتم تحديثها في VF و VL،
!> على التوالي، في SLASD6. وبالتالي لا يتم الرجوع إلى U و VT صراحة.
!>
!> القيم الفردية مخزنة في D. تتكون الخوارزمية من مرحلتين:
!>
!>       تتكون المرحلة الأولى من تقليل حجم المشكلة
!>       عندما تكون هناك قيم فردية متعددة أو إذا كان هناك صفر
!>       في المتجه Z. لكل حدوث من هذا القبيل، يتم تقليل
!>       بعد مشكلة المعادلة العلمانية بمقدار واحد. يتم إجراء هذه المرحلة
!>       بواسطة الوحدة SLASD7.
!>
!>       تتكون المرحلة الثانية من حساب
!>       القيم الفردية المحدثة. يتم ذلك عن طريق إيجاد جذور
!>       المعادلة العلمانية من خلال الوحدة SLASD4 (كما تم استدعاؤها بواسطة SLASD8).
!>       تقوم هذه الوحدة أيضًا بتحديث VF و VL وحساب المسافات
!>       بين القيم الفردية المحدثة والقيم الفردية القديمة.
!>
!> يتم استدعاء SLASD6 من SLASDA.
!>

المعلمات

ICOMPQ 
!>          ICOMPQ هو INTEGER
!>         يحدد ما إذا كان سيتم حساب المتجهات الفردية
!>         في شكل مضروب:
!>         = 0: حساب القيم الفردية فقط.
!>         = 1: حساب المتجهات الفردية في شكل مضروب أيضًا.
!>

NL

!>          NL هو INTEGER
!>         بعد الصف للكتلة العلوية. NL >= 1.
!>

NR

!>          NR هو INTEGER
!>         بعد الصف للكتلة السفلية. NR >= 1.
!>

SQRE

!>          SQRE هو INTEGER
!>         = 0: الكتلة السفلية هي مصفوفة مربعة NR-by-NR.
!>         = 1: الكتلة السفلية هي مصفوفة مستطيلة NR-by-(NR+1).
!>
!>         المصفوفة الثنائية لها بعد صف N = NL + NR + 1،
!>         وبعد عمود M = N + SQRE.
!>

D

!>          D هي مصفوفة REAL، البعد (NL+NR+1).
!>         عند الإدخال D(1:NL,1:NL) تحتوي على القيم الفردية
!>         للكتلة العلوية، و D(NL+2:N) تحتوي على القيم الفردية
!>         للكتلة السفلية. عند الخروج D(1:N) تحتوي على القيم الفردية
!>         للمصفوفة المعدلة.
!>

VF

!>          VF هي مصفوفة REAL، البعد (M)
!>         عند الإدخال، VF(1:NL+1) تحتوي على المكونات الأولى لجميع
!>         المتجهات الفردية اليمنى للكتلة العلوية؛ و VF(NL+2:M)
!>         تحتوي على المكونات الأولى لجميع المتجهات الفردية اليمنى
!>         للكتلة السفلية. عند الخروج، تحتوي VF على المكونات الأولى
!>         لجميع المتجهات الفردية اليمنى للمصفوفة الثنائية.
!>

VL

!>          VL هي مصفوفة REAL، البعد (M)
!>         عند الإدخال، VL(1:NL+1) تحتوي على المكونات الأخيرة لجميع
!>         المتجهات الفردية اليمنى للكتلة العلوية؛ و VL(NL+2:M)
!>         تحتوي على المكونات الأخيرة لجميع المتجهات الفردية اليمنى
!>         للكتلة السفلية. عند الخروج، تحتوي VL على المكونات الأخيرة
!>         لجميع المتجهات الفردية اليمنى للمصفوفة الثنائية.
!>

ALPHA

!>          ALPHA هو REAL
!>         يحتوي على العنصر القطري المرتبط بالصف المضاف.
!>

BETA

!>          BETA هو REAL
!>         يحتوي على العنصر غير القطري المرتبط بالصف المضاف.
!>

IDXQ

!>          IDXQ هي مصفوفة INTEGER، البعد (N)
!>         تحتوي هذه على التبديل الذي سيعيد دمج المشكلة الفرعية التي تم حلها للتو
!>         في ترتيب مرتب، أي D( IDXQ( I = 1, N ) ) سيكون في ترتيب تصاعدي.
!>

PERM

!>          PERM هي مصفوفة INTEGER، البعد ( N )
!>         التبديلات (من التقليل والفرز) التي سيتم تطبيقها
!>         على كل كتلة. لا تتم الإشارة إليها إذا كان ICOMPQ = 0.
!>

GIVPTR

!>          GIVPTR هو INTEGER
!>         عدد دورات جيفنز التي حدثت في هذه
!>         المشكلة الفرعية. لا تتم الإشارة إليها إذا كان ICOMPQ = 0.
!>

GIVCOL

!>          GIVCOL هي مصفوفة INTEGER، البعد ( LDGCOL, 2 )
!>         يشير كل زوج من الأرقام إلى زوج من الأعمدة التي سيتم أخذها في
!>         دورة جيفنز. لا تتم الإشارة إليها إذا كان ICOMPQ = 0.
!>

LDGCOL

!>          LDGCOL هو INTEGER
!>         البعد الرائد لـ GIVCOL، يجب أن يكون على الأقل N.
!>

GIVNUM

!>          GIVNUM هي مصفوفة REAL، البعد ( LDGNUM, 2 )
!>         يشير كل رقم إلى قيمة C أو S التي سيتم استخدامها في
!>         دورة جيفنز المقابلة. لا تتم الإشارة إليها إذا كان ICOMPQ = 0.
!>

LDGNUM

!>          LDGNUM هو INTEGER
!>         البعد الرائد لـ GIVNUM و POLES، يجب أن يكون على الأقل N.
!>

POLES

!>          POLES هي مصفوفة REAL، البعد ( LDGNUM, 2 )
!>         عند الخروج، POLES(1,*) هي مصفوفة تحتوي على القيم الفردية الجديدة
!>         التي تم الحصول عليها من حل المعادلة العلمانية، و
!>         POLES(2,*) هي مصفوفة تحتوي على الأقطاب في
!>         المعادلة العلمانية. لا تتم الإشارة إليها إذا كان ICOMPQ = 0.
!>

DIFL

!>          DIFL هي مصفوفة REAL، البعد ( N )
!>         عند الخروج، DIFL(I) هي المسافة بين i-th المحدثة
!>
