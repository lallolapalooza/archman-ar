توثيق الوظيفة/الروتين الفرعي

subroutine slattr (integer imat, character uplo, character trans, character diag, integer, dimension( 4 ) iseed, integer n, real, dimension( lda, * ) a, integer lda, real, dimension( * ) b, real, dimension( * ) work, integer info)

SLATTR

الغرض

!>
!> SLATTR ينشئ مصفوفة اختبار مثلثة.
!> IMAT و UPLO يحددان بشكل فريد خصائص مصفوفة الاختبار،
!> والتي يتم إرجاعها في المصفوفة A.
!>

المعلمات

IMAT 
!>          IMAT هو عدد صحيح
!>          مفتاح عدد صحيح يصف أي مصفوفة سيتم إنشاؤها لهذا
!>          المسار.
!>

UPLO

!>          UPLO هو حرف*1
!>          يحدد ما إذا كانت المصفوفة A ستكون مثلثية عليا أو سفلى.
!>          = 'U':  مثلثية عليا
!>          = 'L':  مثلثية سفلى
!>

TRANS

!>          TRANS هو حرف*1
!>          يحدد ما إذا كانت المصفوفة أو منقولها سيتم استخدامهما.
!>          = 'N':  بدون نقل
!>          = 'T':  منقول
!>          = 'C':  منقول مترافق (= منقول)
!>

DIAG

!>          DIAG هو حرف*1
!>          يحدد ما إذا كانت المصفوفة A مثلثة وحدية أم لا.
!>          = 'N':  مثلثية غير وحدية
!>          = 'U':  مثلثية وحدية
!>

ISEED

!>          ISEED هي مصفوفة أعداد صحيحة، بأبعاد (4)
!>          متجه البذرة لمولد الأرقام العشوائية (المستخدم في
!>          SLATMS).  يتم تعديله عند الخروج.
!>

N

!>          N هو عدد صحيح
!>          ترتيب المصفوفة التي سيتم إنشاؤها.
!>

A

!>          A هي مصفوفة حقيقية، بأبعاد (LDA,N)
!>          المصفوفة المثلثية A.  إذا كان UPLO = 'U'، فإن الجزء العلوي
!>          المثلثي العلوي الذي تصل أبعاده إلى n × n في المصفوفة A يحتوي
!>          على المصفوفة المثلثية العليا، والجزء السفلي المثلثي الصارم
!>          لـ A غير مستخدم.  إذا كان UPLO = 'L'، فإن الجزء السفلي
!>          المثلثي السفلي الذي تصل أبعاده إلى n × n في المصفوفة A يحتوي
!>          على المصفوفة المثلثية السفلى، والجزء العلوي المثلثي الصارم
!>          لـ A غير مستخدم.  إذا كان DIAG = 'U'، يتم ضبط العناصر القطرية
!>          لـ A بحيث يكون A(k,k) = k لـ 1 <= k <= n.
!>

LDA

!>          LDA هو عدد صحيح
!>          البعد الرائد للمصفوفة A.  LDA >= max(1,N).
!>

B

!>          B هي مصفوفة حقيقية، بأبعاد (N)
!>          المتجه الطرف الأيمن، إذا كان IMAT > 10.
!>

WORK

!>          WORK هي مصفوفة حقيقية، بأبعاد (3*N)
!>

INFO

!>          INFO هو عدد صحيح
!>          = 0:  خروج ناجح
!>          < 0: إذا كان INFO = -k، فإن المعلمة k كانت لها قيمة غير قانونية
!>
