subroutine slatmt (integer m, integer n, character dist, integer, dimension( 4 ) iseed, character sym, real, dimension( * ) d, integer mode, real cond, real dmax, integer rank, integer kl, integer ku, character pack, real, dimension( lda, * ) a, integer lda, real, dimension( * ) work, integer info)

الغرض:

!>
!>    SLATMT تنشئ مصفوفات عشوائية بقيم فردية محددة
!>    (أو متناظرة/هرميتية بقيم ذاتية محددة)
!>    لاختبار برامج LAPACK.
!>
!>    تعمل SLATMT عن طريق تطبيق التسلسل التالي من
!!>    العمليات:
!>
!>      تعيين القطر إلى D، حيث يمكن أن تكون D مدخلة أو
!>         محسوبة وفقًا لـ MODE، COND، DMAX، وSYM
!>         كما هو موضح أدناه.
!>
!>      إنشاء مصفوفة ذات بنية نطاق مناسبة، عن طريق إحدى
!>         الطريقتين:
!>
!>      الطريقة أ:
!>          إنشاء مصفوفة كثيفة M x N عن طريق ضرب D على اليسار
!>              وعلى اليمين بمصفوفات وحدوية عشوائية، ثم:
!>
!>          تقليل عرض النطاق وفقًا لـ KL و KU، باستخدام
!>          تحويلات هاوسهولدر.
!>
!>      الطريقة ب:
!>          تحويل المصفوفة ذات عرض النطاق الصفري (أي قطرية) إلى
!>              مصفوفة ذات عرض نطاق واحد باستخدام دورانات جيفنز، 
!>              العناصر خارج النطاق مرة أخرى، إلى حد كبير كما في QR؛ ثم
!>              تحويل عرض النطاق الواحد إلى عرض نطاق اثنان، وهكذا.
!>              لاحظ أنه بالنسبة لعروض النطاق الصغيرة بشكل معقول (مقارنة بـ
!>              M و N) يتطلب هذا تخزينًا أقل، حيث لا يتم إنشاء مصفوفة كثيفة.
!>              أيضًا، بالنسبة للمصفوفات المتناظرة، يتم إنشاء مثلث واحد فقط.
!>
!>      يتم اختيار الطريقة أ إذا كان عرض النطاق يمثل جزءًا كبيرًا من
!>          ترتيب المصفوفة، وكان LDA على الأقل M (بحيث يمكن تخزين مصفوفة كثيفة).
!>          يتم اختيار الطريقة ب إذا كان عرض النطاق
!>          صغيرًا (< 1/2 N للمتناظرة، < .3 N+M لغير المتناظرة)،
!>          أو كان LDA أقل من M وليس أقل من
!>          عرض النطاق.
!>
!>      تعبئة المصفوفة إذا رغبت في ذلك. الخيارات المحددة بواسطة PACK هي:
!>         لا تعبئة
!>         تصفير النصف العلوي (إذا كانت متناظرة)
!>         تصفير النصف السفلي (إذا كانت متناظرة)
!>         تخزين النصف العلوي عموديًا (إذا كانت متناظرة أو مثلثية عليا)
!>         تخزين النصف السفلي عموديًا (إذا كانت متناظرة أو مثلثية سفلى)
!>         تخزين المثلث السفلي بتنسيق نطاق (إذا كانت متناظرة
!>               أو مثلثية سفلى)
!>         تخزين المثلث العلوي بتنسيق نطاق (إذا كانت متناظرة
!>               أو مثلثية عليا)
!>         تخزين المصفوفة بأكملها بتنسيق نطاق
!>      إذا تم اختيار الطريقة ب، وتم تحديد تنسيق النطاق، فسيتم
!>         إنشاء المصفوفة بتنسيق النطاق، لذلك لن تكون هناك حاجة لإعادة التعبئة.
!>

المعلمات

M 
!>          M هو عدد صحيح (INTEGER)
!>           عدد صفوف A. لم يتم تعديله.
!>

N

!>          N هو عدد صحيح (INTEGER)
!>           عدد أعمدة A. لم يتم تعديله.
!>

DIST

!>          DIST هو حرف واحد (CHARACTER*1)
!>           عند الدخول، يحدد DIST نوع التوزيع الذي سيتم استخدامه
!>           لتوليد القيم الذاتية/الفردية العشوائية.
!>           'U' => موزع بانتظام (UNIFORM) (0, 1) ( 'U' لـ uniform )
!>           'S' => موزع بانتظام (UNIFORM) (-1, 1) ( 'S' لـ symmetric )
!>           'N' => طبيعي (NORMAL) (0, 1) ( 'N' لـ normal )
!>           لم يتم تعديله.
!>

ISEED

!>          ISEED هو مصفوفة أعداد صحيحة (INTEGER array)، البعد (4)
!>           عند الدخول، يحدد ISEED البذرة لمولد الأرقام
!>           العشوائية. يجب أن تكون بين 0 و 4095 شاملة،
!>           ويجب أن يكون ISEED(4) فرديًا. يستخدم مولد الأرقام
!>           العشوائية سلسلة خطية متطابقة محدودة بأعداد صحيحة صغيرة،
!>           لذا يجب أن تنتج أرقامًا عشوائية مستقلة عن الجهاز.
!>           تتغير قيم ISEED عند الخروج، ويمكن استخدامها في
!>           الاستدعاء التالي لـ SLATMT لمواصلة نفس تسلسل الأرقام العشوائية.
!>           تم تعديله عند الخروج.
!>

SYM

!>          SYM هو حرف واحد (CHARACTER*1)
!>           إذا كانت SYM='S' أو 'H'، فإن المصفوفة التي تم إنشاؤها تكون متناظرة، مع
!>             قيم ذاتية محددة بواسطة D، COND، MODE، و DMAX؛ يمكن أن تكون
!>             موجبة أو سالبة أو صفرية.
!>           إذا كانت SYM='P'، فإن المصفوفة التي تم إنشاؤها تكون متناظرة، مع
!>             قيم ذاتية ( = قيم فردية) محددة بواسطة D، COND،
!>             MODE، و DMAX؛ لن تكون سالبة.
!>           إذا كانت SYM='N'، فإن المصفوفة التي تم إنشاؤها تكون غير متناظرة، مع
!>             قيم فردية محددة بواسطة D، COND، MODE، و DMAX؛
!>             لن تكون سالبة.
!>           لم يتم تعديله.
!>

D

!>          D هي مصفوفة حقيقية (REAL array)، البعد (MIN( M , N ))
!>           تستخدم هذه المصفوفة لتحديد القيم الفردية أو
!>           القيم الذاتية لـ A (انظر SYM، أعلاه). إذا كان MODE=0، فإن D
!>           يعتبر أنه يحتوي على القيم الفردية/الذاتية، وإلا
!>           فسيتم حسابها وفقًا لـ MODE، COND، و DMAX،
!>           ووضعها في D.
!>           تم تعديله إذا كان MODE غير صفري.
!>

MODE

!>          MODE هو عدد صحيح (INTEGER)
!>           عند الدخول، يصف هذا كيفية تحديد القيم الفردية/الذاتية
!>           :
!>           MODE = 0 يعني استخدام D كمدخل
!>
!>           MODE = 1 يحدد D(1)=1 و D(2:RANK)=1.0/COND
!>           MODE = 2 يحدد D(1:RANK-1)=1 و D(RANK)=1.0/COND
!>           MODE = 3 يحدد D(I)=COND**(-(I-1)/(RANK-1))
!>
!>           MODE = 4 يحدد D(i)=1 - (i-1)/(N-1)*(1 - 1/COND)
!>           MODE = 5 يحدد D لأرقام عشوائية في النطاق
!>                    ( 1/COND , 1 ) بحيث تكون لوغاريتماتها
!>                    موزعة بشكل موحد.
!>           MODE = 6 يحدد D لأرقام عشوائية من نفس التوزيع
!>                    مثل بقية المصفوفة.
!>           MODE < 0 له نفس المعنى مثل ABS(MODE)، باستثناء أن
!>              ترتيب عناصر D معكوس.
!>           لذلك، إذا كان MODE موجبًا، فإن D يحتوي على مدخلات تتراوح من
!>              1 إلى 1/COND، إذا كان سالبًا، من 1/COND إلى 1،
!>           إذا كانت SYM='S' أو 'H'، وكان MODE غير 0 أو 6 أو -6، فإن
!>              سيتم ضرب عناصر D أيضًا بإشارة عشوائية (أي +1 أو -1).
!>           لم يتم تعديله.
!>

COND

!>          COND هو قيمة حقيقية (REAL)
!>           عند الدخول، يتم استخدامه كما هو موضح تحت MODE أعلاه.
!>           إذا تم استخدامه، يجب أن يكون >= 1. لم يتم تعديله.
!>

DMAX

!>          DMAX هو قيمة حقيقية (REAL)
!>           إذا كان MODE ليس -6 أو 0 أو 6، فسيتم قياس محتويات D، كما
!>           تم حسابها وفقًا لـ MODE و COND، بواسطة
!>           DMAX / max(abs(D(i)))؛ وبالتالي، سيكون الحد الأقصى للقيمة الذاتية أو
!>           الفردية المطلقة (وهو المعيار) هو abs(DMAX).
!>           لاحظ أنه ليس من الضروري أن يكون DMAX موجبًا: إذا كان DMAX سالبًا
!>           (أو صفرًا)، فسيتم قياس D برقم سالب (أو صفر).
!>           لم يتم تعديله.
!>

RANK

!>          RANK هو عدد صحيح (INTEGER)
!>           رتبة المصفوفة التي سيتم إنشاؤها للأوضاع 1، 2، 3 فقط.
!>           D( RANK+1:N ) = 0.
!>           لم يتم تعديله.
!>

KL

!>          KL هو عدد صحيح (INTEGER)
!>           يحدد هذا عرض النطاق السفلي للمصفوفة. على سبيل المثال،
!>           KL=0 يعني مثلثية عليا، KL=1 يعني هسنبرغ عليا،
!>           و KL أكبر من أو يساوي M-1 يعني أن المصفوفة
!>           لها عرض نطاق سفلي كامل. يجب أن يساوي KL قيمة KU إذا كانت المصفوفة
!>           متناظرة.
!>           لم يتم تعديله.
!>

NAME

slatmt - generates a matrix with a specified structure and distribution

SYNOPSIS

SUBROUTINE SLATMT(
     & M, N, DIST, D S, SYM, MODE, COND, DMAX, PACK, A, LDA, WORK, INFO )
>     CHARACTER *(*), DIMENSION( * ) :: DIST, DS, SYM, PACK
>     INTEGER :: M, N, MODE, INFO
>     REAL :: COND, DMAX
>     REAL, DIMENSION( * ) :: D S, WORK
>     CHARACTER :: A( LDA, * )
>     INTEGER :: LDA

Function/Subroutine Documentation

M

!>          M is INTEGER
!>           The number of rows of the matrix A.
!>           Not modified.
!>

N

!>          N is INTEGER
!>           The number of columns of the matrix A.
!>           Not modified.
!>

DIST

!>          DIST is CHARACTER*1
!>           Specifies the distribution of the elements of A.
!>           'U' => uniform distribution
!>           'N' => normal distribution
!>           'S' => symmetrically distributed, with one of the following
!>                  distributions for the first N elements of the diagonal:
!>                  'U' => uniform distribution
!>                  'N' => normal distribution
!>                  'S' => symmetrically distributed
!>                  'L' => logarithmically distributed (uniform between log and 1)
!>                  'E' => exponentially distributed
!>                  'P' => p-th power distribution (uniform between 0 and 1),
!>                         with p = 1/2
!>           Not modified.
!>

D S

!>          D S is REAL array, dimension ( N )
!>           Contains the diagonal elements of the matrix.
!>           Not modified.
!>

SYM

!>          SYM is CHARACTER*1
!>           Specifies the structure of the matrix A.
!>           'N' => no structure, random entries
!>           'U' => upper triangular matrix
!>           'L' => lower triangular matrix
!>           'S' => symmetric matrix
!>           'H' => upper Hessenberg matrix
!>           'P' => lower Hessenberg matrix
!>           Not modified.
!>

MODE

!>          MODE is INTEGER
!>           Specifies the mode of the test matrix.
!>           MODE = 0 => all entries equal
!>           MODE = 1 => diagonal entries are 1, off-diagonal entries are 0
!>           MODE = 2 => diagonal entries are 1, off-diagonal entries are random
!>           MODE = 3 => diagonal entries are random, off-diagonal entries are 0
!>           MODE = 4 => diagonal entries are random, off-diagonal entries are random
!>           MODE = 5 => diagonal entries are random, off-diagonal entries are
!>                      random and non-zero
!>           MODE = 6 => diagonal entries are random, off-diagonal entries are
!>                      random and non-zero, and have the same sign as the
!>                      corresponding diagonal entry.
!>           MODE = -1 => all entries equal (for symmetric matrices)
!>           MODE = -2 => diagonal entries are 1, off-diagonal entries are 0 (for
!>                      symmetric matrices)
!>           MODE = -3 => diagonal entries are 1, off-diagonal entries are random (for
!>                      symmetric matrices)
!>           MODE = -4 => diagonal entries are random, off-diagonal entries are 0 (for
!>                      symmetric matrices)
!>           MODE = -5 => diagonal entries are random, off-diagonal entries are
!>                      random (for symmetric matrices)
!>           MODE = -6 => diagonal entries are random, off-diagonal entries are
!>                      random and non-zero (for symmetric matrices)
!>           Not modified.
!>

COND

!>          COND is REAL
!>           Used only when MODE is not 0. If COND is positive, the matrix
!>           is scaled so that its condition number is equal to COND.
!>           Not modified.
!>

DMAX

!>          DMAX is REAL
!>           Used only when MODE is not 0. If DMAX is positive, the matrix
!>           is scaled so that the maximum absolute value of any matrix
!>           element is equal to DMAX.
!>           Not modified.
!>

KU

!>          KU is INTEGER
!>           This specifies the upper bandwidth of the  matrix. For
!>           example, KU=0 implies lower triangular, KU=1 implies lower
!>           Hessenberg, and KU being at least N-1 means that the matrix
!>           has full upper bandwidth.  KL must equal KU if the matrix
!>           is symmetric.
!>           Not modified.
!>

PACK

!>          PACK is CHARACTER*1
!>           This specifies packing of matrix as follows:
!>           'N' => no packing
!>           'U' => zero out all subdiagonal entries (if symmetric)
!>           'L' => zero out all superdiagonal entries (if symmetric)
!>           'C' => store the upper triangle columnwise
!>                  (only if the matrix is symmetric or upper triangular)
!>           'R' => store the lower triangle columnwise
!>                  (only if the matrix is symmetric or lower triangular)
!>           'B' => store the lower triangle in band storage scheme
!>                  (only if matrix symmetric or lower triangular)
!>           'Q' => store the upper triangle in band storage scheme
!>                  (only if matrix symmetric or upper triangular)
!>           'Z' => store the entire matrix in band storage scheme
!>                      (pivoting can be provided for by using this
!>                      option to store A in the trailing rows of
!>                      the allocated storage)
!>
!>           Using these options, the various LAPACK packed and banded
!>           storage schemes can be obtained:
!>           GB               - use 'Z'
!>           PB, SB or TB     - use 'B' or 'Q'
!>           PP, SP or TP     - use 'C' or 'R'
!>
!>           If two calls to SLATMT differ only in the PACK parameter,
!>           they will generate mathematically equivalent matrices.
!>           Not modified.
!>

A

!>          A is REAL array, dimension ( LDA, N )
!>           On exit A is the desired test matrix.  A is first generated
!>           in full (unpacked) form, and then packed, if so specified
!>           by PACK.  Thus, the first M elements of the first N
!>           columns will always be modified.  If PACK specifies a
!>           packed or banded storage scheme, all LDA elements of the
!>           first N columns will be modified; the elements of the
!>           array which do not correspond to elements of the generated
!>           matrix are set to zero.
!>           Modified.
!>

LDA

!>          LDA is INTEGER
!>           LDA specifies the first dimension of A as declared in the
!>           calling program.  If PACK='N', 'U', 'L', 'C', or 'R', then
!>           LDA must be at least M.  If PACK='B' or 'Q', then LDA must
!>           be at least MIN( KL, M-1) (which is equal to MIN(KU,N-1)).
!>           If PACK='Z', LDA must be large enough to hold the packed
!>           array: MIN( KU, N-1) + MIN( KL, M-1) + 1.
!>           Not modified.
!>

WORK

!>          WORK is REAL array, dimension ( 3*MAX( N , M ) )
!>           Workspace.
!>           Modified.
!>

INFO

!>          INFO is INTEGER
!>           Error code.  On exit, INFO will be set to one of the
!>           following values:
!>             0 => normal return
!>            -1 => M negative or unequal to N and SYM='S', 'H', or 'P'
!>            -2 => N negative
!>            -3 => DIST illegal string
!>            -5 => SYM illegal string
!>            -7 => MODE not in range -6 to 6
!>            -8 => COND less than 1.0, and MODE neither -6, 0 nor 6
!>           -10 => KL negative
!>           -11 => KU negative, or SYM='S' or 'H' and KU not equal to KL
!>           -12 => PACK illegal string, or PACK='U' or 'L', and SYM='N';
!>                  or PACK='C' or 'Q' and SYM='N' and KL is not zero;
!>                  or PACK='R' or 'B' and SYM='N' and KU is not zero;
!>                  or PACK='U', 'L', 'C', 'R', 'B', or 'Q', and M is not
!!>                  N.
!>           -14 => LDA is less than M, or PACK='Z' and LDA is less than
!>                  MIN(KU,N-1) + MIN(KL,M-1) + 1.
!>            1  => Error return from SLATM7
!>            2  => Cannot scale to DMAX (max. sing. value is 0)
!>            3  => Error return from SLAGGE or SLAGSY
!>
