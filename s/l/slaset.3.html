توثيق الوظيفة/الروتين الفرعي

subroutine slaset (character uplo, integer m, integer n, real alpha, real beta, real, dimension( lda, * ) a, integer lda)

SLASET يقوم بتهيئة العناصر القطرية والعناصر غير القطرية لمصفوفة بقيم معطاة.

الغرض:

!>
!> SLASET تهيئة مصفوفة A بحجم m x n بالقيمة BETA على القطر
!> و ALPHA على العناصر غير القطرية.
!>

المعاملات

UPLO 
!>          UPLO هو CHARACTER*1
!>          يحدد الجزء من المصفوفة A المراد تعيينه.
!>          = 'U':      يتم تعيين الجزء العلوي المثلثي؛ الجزء السفلي تمامًا
!>                      من المصفوفة A لا يتغير.
!>          = 'L':      يتم تعيين الجزء السفلي المثلثي؛ الجزء العلوي تمامًا
!>                      من المصفوفة A لا يتغير.
!>          بخلاف ذلك:  يتم تعيين المصفوفة A بأكملها.
!>

M

!>          M هو INTEGER
!>          عدد صفوف المصفوفة A.  M >= 0.
!>

N

!>          N هو INTEGER
!>          عدد أعمدة المصفوفة A.  N >= 0.
!>

ALPHA

!>          ALPHA هو REAL
!>          القيمة الثابتة التي سيتم تعيين العناصر غير القطرية لها.
!>

BETA

!>          BETA هو REAL
!>          القيمة الثابتة التي سيتم تعيين العناصر القطرية لها.
!>

A

!>          A هو مصفوفة REAL، ذات بعد (LDA,N)
!>          عند الخروج، يتم تعيين الجزء الرائد من المصفوفة A بحجم m x n كما يلي:
!>
!>          إذا كان UPLO = 'U'، فإن A(i,j) = ALPHA، 1<=i<=j-1، 1<=j<=n،
!>          إذا كان UPLO = 'L'، فإن A(i,j) = ALPHA، j+1<=i<=m، 1<=j<=n،
!>          بخلاف ذلك،     A(i,j) = ALPHA، 1<=i<=m، 1<=j<=n، i.ne.j،
!>
!>          و، لجميع قيم UPLO، A(i,i) = BETA، 1<=i<=min(m,n).
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرائد للمصفوفة A.  LDA >= max(1,M).
!>
