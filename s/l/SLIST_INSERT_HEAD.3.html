الاسم

SLIST_EMPTY, SLIST_ENTRY, SLIST_FIRST, SLIST_FOREACH, SLIST_HEAD, SLIST_HEAD_INITIALIZER, SLIST_INIT, SLIST_INSERT_AFTER, SLIST_INSERT_HEAD, SLIST_NEXT, SLIST_REMOVE, SLIST_REMOVE_HEAD - تنفيذ قائمة مرتبطة مفردة

المكتبة

المكتبة القياسية (libc)، libc، -lc

الملخص

#include <sys/queue.h>
SLIST_ENTRY(TYPE);
SLIST_HEAD(HEADNAME, TYPE);
SLIST_HEAD SLIST_HEAD_INITIALIZER(SLIST_HEAD head);
void SLIST_INIT(SLIST_HEAD *head);
int SLIST_EMPTY(SLIST_HEAD *head);
void SLIST_INSERT_HEAD(SLIST_HEAD *head,
                        struct TYPE *elm, SLIST_ENTRY NAME);
void SLIST_INSERT_AFTER(struct TYPE *listelm,
                        struct TYPE *elm, SLIST_ENTRY NAME);
struct TYPE *SLIST_FIRST(SLIST_HEAD *head);
struct TYPE *SLIST_NEXT(struct TYPE *elm, SLIST_ENTRY NAME);
SLIST_FOREACH(struct TYPE *var, SLIST_HEAD *head, SLIST_ENTRY NAME);
void SLIST_REMOVE(SLIST_HEAD *head, struct TYPE *elm, TYPE,
                        SLIST_ENTRY NAME);

                        SLIST_ENTRY NAME);

الوصف

تعرّف هذه الماكرو قوائم مرتبطة مفردة وتعمل عليها.

في تعريفات الماكرو، TYPE هو اسم هيكل معرف من قبل المستخدم يجب أن يحتوي على حقل من نوع SLIST_ENTRY، ويسمى NAME. الوسيط HEADNAME هو اسم هيكل معرف من قبل المستخدم يجب تعريفه باستخدام الماكرو SLIST_HEAD().

الإنشاء

يتم ترؤيس قائمة مرتبطة مفردة بهيكل يتم تعريفه باستخدام الماكرو SLIST_HEAD(). يحتوي هذا الهيكل على مؤشر فريد إلى العنصر الأول في القائمة. ترتبط العناصر بطريقة مفردة لتقليل مساحة المؤشر والنفقات العامة للمعالجة، على الرغم من أن هذا يأتي بتكلفة O(n) عند إزالة العناصر العشوائية. يمكن إضافة عناصر جديدة إلى القائمة بعد عنصر موجود أو في بداية القائمة. يتم تعريف هيكل SLIST_HEAD كالتالي:


SLIST_HEAD(HEADNAME, TYPE) head;

حيث struct HEADNAME هو الهيكل المراد تعريفه و struct TYPE هو نوع العناصر في القائمة. يمكن بعد ذلك تعريف عنوان مؤشر إلى بداية القائمة كالتالي:


struct HEADNAME *headp;

(الأسماء head و headp قابلة للاختيار من قبل المستخدم.)

SLIST_ENTRY() تعرّف هيكلاً يربط عناصر القائمة.

SLIST_HEAD_INITIALIZER() تقيّم إلى مُهيئ للقائمة head.

SLIST_INIT() تهيئ القائمة المشار إليها بواسطة head.

SLIST_EMPTY() تقيّم إلى TRUE إذا كانت القائمة فارغة، و FALSE بخلاف ذلك.

الإدراج

SLIST_INSERT_HEAD() تُدرج العنصر الجديد elm في بداية القائمة.

SLIST_INSERT_AFTER() تُدرج العنصر الجديد elm بعد العنصر listelm.

التجوال

SLIST_FIRST() تُرجع العنصر الأول في القائمة، أو NULL إذا كانت القائمة فارغة.

SLIST_NEXT() تُرجع العنصر التالي في القائمة.

SLIST_FOREACH() تتجول عبر القائمة المشار إليها بواسطة head في الاتجاه الأمامي، مع تعيين كل عنصر إلى var.

الإزالة

SLIST_REMOVE() تزيل العنصر elm من القائمة.

SLIST_REMOVE_HEAD() تزيل العنصر elm من بداية القائمة. للحصول على أقصى كفاءة، يجب أن تستخدم العناصر التي تتم إزالتها من بداية القائمة هذه الماكرو بشكل صريح بدلاً من الماكرو العام SLIST_REMOVE().

قيمة الإرجاع

SLIST_EMPTY() ترجع FALSE إذا كانت القائمة فارغة، و TRUE بخلاف ذلك.

SLIST_FIRST() و SLIST_NEXT() ترجعان مؤشراً إلى هيكل TYPE للعنصر الأول أو التالي، على التوالي.

SLIST_HEAD_INITIALIZER() ترجع مُهيئًا يمكن تعيينه للقائمة head.

المعايير

BSD.

التاريخ

4.4BSD.

أخطاء

SLIST_FOREACH() لا تسمح بإزالة أو تحرير var داخل الحلقة، لأن ذلك سيتداخل مع عملية التجوال. SLIST_FOREACH_SAFE()، الموجودة في BSDs ولكن ليس في glibc، تتجاوز هذا القيد بالسماح بإزالة var من القائمة وتحريرها داخل الحلقة دون تعطيل عملية التجوال.

أمثلة

#include <stddef.h>
#include <stdio.h>
#include <stdlib.h>
#include <sys/queue.h>
struct entry {
    int data;
    SLIST_ENTRY(entry) entries;             /* قائمة مرتبطة مفردة */
};
SLIST_HEAD(slisthead, entry);
int
main(void)
{
    struct entry *n1, *n2, *n3, *np;
    struct slisthead head;                  /* رأس قائمة مرتبطة مفردة */
    SLIST_INIT(&head);                      /* تهيئة القائمة */
    n1 = malloc(sizeof(struct entry));      /* إدراج في البداية */
    SLIST_INSERT_HEAD(&head, n1, entries);
    n2 = malloc(sizeof(struct entry));      /* إدراج بعد */
    SLIST_INSERT_AFTER(n1, n2, entries);
    SLIST_REMOVE(&head, n2, entry, entries);/* حذف */
    free(n2);
    n3 = SLIST_FIRST(&head);
    SLIST_REMOVE_HEAD(&head, entries);      /* حذف من البداية */
    free(n3);
    for (unsigned int i = 0; i < 5; i++) {
        n1 = malloc(sizeof(struct entry));
        SLIST_INSERT_HEAD(&head, n1, entries);
        n1->data = i;
    }
                                            /* التجوال الأمامي */
    SLIST_FOREACH(np, &head, entries)
        printf("%i\n", np->data);
    while (!SLIST_EMPTY(&head)) {           /* تفريغ القائمة */
        n1 = SLIST_FIRST(&head);
        SLIST_REMOVE_HEAD(&head, entries);
        free(n1);
    }
    SLIST_INIT(&head);
    exit(EXIT_SUCCESS);
}
