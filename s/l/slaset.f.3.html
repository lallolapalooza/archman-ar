وثائق الوظيفة/الإجراء الفرعي

subroutine slaset (character uplo, integer m, integer n, real alpha, real beta, real, dimension( lda, * ) a, integer lda)

SLASET تهيئة العناصر القطرية وغير القطرية في مصفوفة بقيم معينة.

الغرض:

!>
!> SLASET تهيئة مصفوفة A بحجم m x n إلى BETA على القطر و
!> ALPHA في الأجزاء غير القطرية.
!>

المعاملات

UPLO 
!>          UPLO هو CHARACTER*1
!>          يحدد الجزء من المصفوفة A الذي سيتم تعيينه.
!>          = 'U':      يتم تعيين الجزء المثلث العلوي؛ الجزء السفلي تماماً
!>                      من المصفوفة A لا يتم تغييره.
!>          = 'L':      يتم تعيين الجزء المثلث السفلي؛ الجزء العلوي تماماً
!>                      من المصفوفة A لا يتم تغييره.
!>          بخلاف ذلك:  يتم تعيين كل المصفوفة A.
!>

M

!>          M هو INTEGER
!>          عدد صفوف المصفوفة A.  M >= 0.
!>

N

!>          N هو INTEGER
!>          عدد أعمدة المصفوفة A.  N >= 0.
!>

ALPHA

!>          ALPHA هو REAL
!>          القيمة الثابتة التي يجب تعيين العناصر غير القطرية إليها.
!>

BETA

!>          BETA هو REAL
!>          القيمة الثابتة التي يجب تعيين العناصر القطرية إليها.
!>

A

!>          A هو مصفوفة REAL، ذات بعد (LDA,N)
!>          عند الخروج، الجزء الرائد بحجم m x n من A يتم تعيينه على النحو التالي:
!>
!>          إذا UPLO = 'U'، A(i,j) = ALPHA، 1<=i<=j-1، 1<=j<=n،
!>          إذا UPLO = 'L'، A(i,j) = ALPHA، j+1<=i<=m، 1<=j<=n،
!>          بخلاف ذلك،     A(i,j) = ALPHA، 1<=i<=m، 1<=j<=n، i.ne.j،
!>
!>          و، لجميع UPLO، A(i,i) = BETA، 1<=i<=min(m,n).
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرائد للمصفوفة A.  LDA >= max(1,M).
!>
