subroutine slarre (character range, integer n, real vl, real vu, integer il, integer iu, real, dimension( * ) d, real, dimension( * ) e, real, dimension( * ) e2, real rtol1, real rtol2, real spltol, integer nsplit, integer, dimension( * ) isplit, integer m, real, dimension( * ) w, real, dimension( * ) werr, real, dimension( * ) wgap, integer, dimension( * ) iblock, integer, dimension( * ) indexw, real, dimension( * ) gers, real pivmin, real, dimension( * ) work, integer, dimension( * ) iwork, integer info)

SLARRE بالنظر إلى المصفوفة ثلاثية الأقطار T، يقوم بتصفير العناصر الصغيرة خارج القطر، ولكل كتلة غير مختزلة Ti، يجد التمثيل الأساسي والقيم الذاتية.

الغرض:

!>
!> لإيجاد القيم الذاتية المطلوبة لمصفوفة ثلاثية الأقطار متناظرة حقيقية T،
!> تقوم SLARRE بتصفير أي عناصر خارج القطر، ولكل كتلة غير مختزلة T_i،
!> فإنها تجد (أ) إزاحة مناسبة في أحد طرفي طيف الكتلة،
!> (ب) التمثيل الأساسي، T_i - sigma_i I = L_i D_i L_i^T، و
!> (ج) القيم الذاتية لكل L_i D_i L_i^T.
!> تُستخدم التمثيلات والقيم الذاتية التي تم العثور عليها بعد ذلك بواسطة
!> SSTEMR لحساب المتجهات الذاتية لـ T.
!> تختلف الدقة اعتمادًا على ما إذا كان يتم استخدام التنصيف للعثور على
!> عدد قليل من القيم الذاتية أو خوارزمية dqds (الإجراء الفرعي SLASQ2) لحساب
!> جميع القيم الذاتية ثم التخلص من أي قيمة غير مرغوبة.
!> كميزة إضافية، تخرج SLARRE أيضًا n
!> فترات Gerschgorin للمصفوفات L_i D_i L_i^T.
!>

المعلمات

RANGE 
!>          RANGE هو CHARACTER*1
!>          = 'A': ()   سيتم العثور على جميع القيم الذاتية.
!>          = 'V': () سيتم العثور على جميع القيم الذاتية في الفترة نصف المفتوحة
!>                           (VL, VU].
!>          = 'I': () سيتم العثور على القيم الذاتية من IL-th إلى IU-th (للمصفوفة
!>                           بأكملها).
!>

N

!>          N هو INTEGER
!>          رتبة المصفوفة. N > 0.
!>

VL

!>          VL هو REAL
!>          إذا كانت RANGE='V'، فإنها تمثل الحد الأدنى للقيم الذاتية.
!>          القيم الذاتية الأقل من أو تساوي VL، أو الأكبر من VU،
!>          لن يتم إرجاعها. VL < VU.
!>          إذا كانت RANGE='I' أو ='A'، فإن SLARRE تحسب حدودًا على الجزء المطلوب
!>          من الطيف.
!>

VU

!>          VU هو REAL
!>          إذا كانت RANGE='V'، فإنها تمثل الحد الأعلى للقيم الذاتية.
!>          القيم الذاتية الأقل من أو تساوي VL، أو الأكبر من VU،
!>          لن يتم إرجاعها. VL < VU.
!>          إذا كانت RANGE='I' أو ='A'، فإن SLARRE تحسب حدودًا على الجزء المطلوب
!>          من الطيف.
!>

IL

!>          IL هو INTEGER
!>          إذا كانت RANGE='I'، فإنها تمثل فهرس
!>          أصغر قيمة ذاتية سيتم إرجاعها.
!>          1 <= IL <= IU <= N.
!>

IU

!>          IU هو INTEGER
!>          إذا كانت RANGE='I'، فإنها تمثل فهرس
!>          أكبر قيمة ذاتية سيتم إرجاعها.
!>          1 <= IL <= IU <= N.
!>

D

!>          D هو مصفوفة REAL، البعد (N)
!>          عند الإدخال، العناصر القطرية N للمصفوفة ثلاثية الأقطار
!>          T.
!>          عند الخروج، العناصر القطرية N للمصفوفات القطرية
!>          D_i.
!>

E

!>          E هو مصفوفة REAL، البعد (N)
!>          عند الإدخال، تحتوي العناصر (N-1) الأولى على العناصر تحت القطرية
!>          للمصفوفة ثلاثية الأقطار T؛ E(N) لا تحتاج إلى تعيين.
!>          عند الخروج، تحتوي E على العناصر تحت القطرية للمصفوفات الثنائية الوحدة
!>          L_i. العناصر E( ISPLIT( I ) )،
!>          1 <= I <= NSPLIT، تحتوي على نقاط الأساس sigma_i عند الخروج.
!>

E2

!>          E2 هو مصفوفة REAL، البعد (N)
!>          عند الإدخال، تحتوي العناصر (N-1) الأولى على مربعات
!>          العناصر تحت القطرية للمصفوفة ثلاثية الأقطار T؛
!>          E2(N) لا تحتاج إلى تعيين.
!>          عند الخروج، تم تعيين العناصر E2( ISPLIT( I ) )،
!>          1 <= I <= NSPLIT، إلى الصفر
!>

RTOL1

!>          RTOL1 هو REAL
!>

RTOL2

!>          RTOL2 هو REAL
!>           معلمات للتنصيف.
!>           فترة [LEFT,RIGHT] متقاربة إذا كان
!>           RIGHT-LEFT < MAX( RTOL1*GAP, RTOL2*MAX(|LEFT|,|RIGHT|) )
!>

SPLTOL

!>          SPLTOL هو REAL
!>          عتبة التقسيم.
!>

NSPLIT

!>          NSPLIT هو INTEGER
!>          عدد الكتل التي تنقسم إليها T. 1 <= NSPLIT <= N.
!>

ISPLIT

!>          ISPLIT هي مصفوفة INTEGER، البعد (N)
!>          نقاط التقسيم، حيث تنقسم T إلى كتل.
!>          الكتلة الأولى تتكون من الصفوف/الأعمدة من 1 إلى ISPLIT(1)،
!>          الثانية من الصفوف/الأعمدة ISPLIT(1)+1 حتى ISPLIT(2)،
!>          إلخ، والـ NSPLIT-th تتكون من الصفوف/الأعمدة
!>          ISPLIT(NSPLIT-1)+1 حتى ISPLIT(NSPLIT)=N.
!>

M

!>          M هو INTEGER
!>          العدد الإجمالي للقيم الذاتية (لجميع L_i D_i L_i^T)
!>          التي تم العثور عليها.
!>

W

!>          W هي مصفوفة REAL، البعد (N)
!>          تحتوي العناصر M الأولى على القيم الذاتية.
!>          القيم الذاتية لكل كتلة، L_i D_i L_i^T، مرتبة
!>          تصاعديًا (قد تستخدم SLARRE العناصر المتبقية N-M كمسافة عمل).
!>

WERR

!>          WERR هي مصفوفة REAL، البعد (N)
!>          حد الخطأ على القيمة الذاتية المقابلة في W.
!>

WGAP

!>          WGAP هي مصفوفة REAL، البعد (N)
!>          الفصل عن القيمة الذاتية المجاورة اليمنى في W.
!>          الفجوة هي فقط بالنسبة للقيم الذاتية لنفس الكتلة، حيث أن لكل كتلة
!>          شجرة تمثيل خاصة بها.
!>          استثناء: عند الطرف الأيمن للكتلة، نخزن الفجوة اليسرى
!>

IBLOCK

!>          IBLOCK هي مصفوفة INTEGER، البعد (N)
!>          فهارس الكتل (المصفوفات الفرعية) المرتبطة بالقيم الذاتية
!>          المقابلة في W؛ IBLOCK(i)=1 إذا كانت القيمة الذاتية
!>          W(i) تنتمي إلى الكتلة الأولى من الأعلى، =2 إذا كانت W(i)
!>          تنتمي إلى الكتلة الثانية، إلخ.
!>

INDEXW

!>          INDEXW هي مصفوفة INTEGER، البعد (N)
!>          فهارس القيم الذاتية داخل كل كتلة (مصفوفة فرعية)؛
!>          على سبيل المثال، INDEXW(i)= 10 و IBLOCK(i)=2 يعني أن
!>          القيمة الذاتية i-th W(i) هي القيمة الذاتية رقم 10 في الكتلة 2
!>

GERS

!>          GERS هي مصفوفة REAL، البعد (2*N)
!>          فترات Gerschgorin N (الفترة Gerschgorin i-th
!>          هي (GERS(2*i-1)، GERS(2*i)).
!>

PIVMIN

!>          PIVMIN هو REAL
!>          المنحنى الأدنى في تسلسل Sturm لـ T.
!>

WORK

!>          WORK هي مصفوفة REAL، البعد (6*N)
!>          مساحة عمل.
!>

IWORK

!>          IWORK هي مصفوفة INTEGER، البعد (N)
!>          مساحة عمل.
!>

INFO

!>          INFO هو INTEGER
!>          = 0: نجاح.
!>          < 0: إذا كان INFO = -i، فإن المعلمة i-th التي تم تمريرها
!>               إلى SLARRE كانت غير صالحة.
!>          > 0: فشل SLARRE في العثور على جميع القيم الذاتية المطلوبة.
!>               على سبيل المثال، إذا كانت INFO > 0، فإن DQDS
!>               (SLASQ2) فشل في التقارب.
!>
