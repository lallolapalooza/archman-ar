توثيق الوظيفة/الروتين الفرعي

subroutine slargv (integer n, real, dimension( * ) x, integer incx, real, dimension( * ) y, integer incy, real, dimension( * ) c, integer incc)

SLARGV يولد متجهًا من دورانات مستوية مع جيوب تمام حقيقية وجيوب حقيقية.

الغرض:

!>
!> SLARGV يولد متجهًا من الدورانات المستوية الحقيقية، يتم تحديده بواسطة
!> عناصر المتجهات الحقيقية x و y. من أجل i = 1,2,...,n
!>
!>    (  c(i)  s(i) ) ( x(i) ) = ( a(i) )
!>    ( -s(i)  c(i) ) ( y(i) ) = (   0  )
!>

المعلمات

N 
!>          N هو عدد صحيح (INTEGER).
!>          عدد الدورانات المستوية التي سيتم توليدها.
!>

X

!>          X هو مصفوفة حقيقية (REAL array)،
!>                         dimension (1+(N-1)*INCX)
!>          عند الدخول، المتجه x.
!>          عند الخروج، يتم استبدال x(i) بـ a(i)، من أجل i = 1,...,n.
!>

INCX

!>          INCX هو عدد صحيح (INTEGER).
!>          الزيادة بين عناصر X. INCX > 0.
!>

Y

!>          Y هو مصفوفة حقيقية (REAL array)،
!>                         dimension (1+(N-1)*INCY)
!>          عند الدخول، المتجه y.
!>          عند الخروج، جيوب الدورانات المستوية.
!>

INCY

!>          INCY هو عدد صحيح (INTEGER).
!>          الزيادة بين عناصر Y. INCY > 0.
!>

C

!>          C هو مصفوفة حقيقية (REAL array)، dimension (1+(N-1)*INCC)
!>          جيوب تمام الدورانات المستوية.
!>

INCC

!>          INCC هو عدد صحيح (INTEGER).
!>          الزيادة بين عناصر C. INCC > 0.
!>
