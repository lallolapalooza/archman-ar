subroutine slatdf (ijob, n, z, ldz, rhs, rdsum, rdscal, ipiv, jpiv)
SLATDF يستخدم التفكيك LU للمصفوفة n-by-n التي تم حسابها بواسطة sgetc2 ويحسب مساهمة في تقدير Dif العكسي.

توثيق الوظيفة/الروتين الفرعي

subroutine slatdf (integer ijob, integer n, real, dimension( ldz, * ) z, integer ldz, real, dimension( * ) rhs, real rdsum, real rdscal, integer, dimension( * ) ipiv, integer, dimension( * ) jpiv)

SLATDF يستخدم التفكيك LU للمصفوفة n-by-n التي تم حسابها بواسطة sgetc2 ويحسب مساهمة في تقدير Dif العكسي.

الغرض:

!>
!> SLATDF يستخدم التفكيك LU للمصفوفة Z التي بحجم n-by-n والتي تم حسابها بواسطة
!> SGETC2 ويحسب مساهمة في تقدير Dif العكسي
!> عن طريق حل Z * x = b لـ x، واختيار b للطرف الأيمن بحيث
!> تكون معيار x أكبر ما يمكن. عند الإدخال، يحتوي RHS = b على
!> المساهمة من الأنظمة الفرعية التي تم حلها سابقًا، وعند الخروج، يحتوي RHS = x.
!>
!> يتكون تفكيك Z الذي أعادته SGETC2 من Z = P*L*U*Q،
!> حيث P و Q مصفوفتان تبادل. L هي مصفوفة ثلاثية سفلية ذات
!> عناصر قطرية وحدة و U هي مصفوفة عليا ثلاثية.
!>

المعلمات

IJOB 
!>          IJOB هو عدد صحيح (INTEGER)
!>          IJOB = 2: أولاً، قم بحساب متجه فارغ تقريبي (null-vector) e
!>              لـ Z باستخدام SGECON، يتم تطبيع e وحل لـ
!>              Zx = +-e - f مع الإشارة التي تعطي القيمة الأكبر
!>              للمعيار 2-norm(x). أغلى بحوالي 5 مرات من الوضع الافتراضي.
!>          IJOB .ne. 2: استراتيجية البحث الأمامي المحلية حيث يتم اختيار جميع مدخلات
!>              الطرف الأيمن b لتكون إما +1 أو -1 (الوضع الافتراضي).
!>

N

!>          N هو عدد صحيح (INTEGER)
!>          عدد أعمدة المصفوفة Z.
!>

Z

!>          Z هي مصفوفة حقيقية (REAL array)، الأبعاد (LDZ, N)
!>          عند الإدخال، الجزء LU من تفكيك المصفوفة n-by-n
!>          Z التي تم حسابها بواسطة SGETC2:  Z = P * L * U * Q
!>

LDZ

!>          LDZ هو عدد صحيح (INTEGER)
!>          البعد الرائد للمصفوفة Z.  LDA >= max(1, N).
!>

RHS

!>          RHS هي مصفوفة حقيقية (REAL array)، الأبعاد N.
!>          عند الإدخال، يحتوي RHS على مساهمات من الأنظمة الفرعية الأخرى.
!>          عند الخروج، يحتوي RHS على حل النظام الفرعي مع
!>          مدخلات وفقًا لقيمة IJOB (انظر أعلاه).
!>

RDSUM

!>          RDSUM هو عدد حقيقي (REAL)
!>          عند الإدخال، مجموع مربعات المساهمات المحسوبة لـ
!>          تقدير Dif قيد الحساب بواسطة STGSYL، حيث تم
!>          استبعاد عامل القياس RDSCAL (انظر أدناه).
!>          عند الخروج، مجموع المربعات المقابل المحدث بـ
!>          المساهمات من النظام الفرعي الحالي.
!>          إذا كان TRANS = 'T'، فلن يتم المساس بـ RDSUM.
!>          ملاحظة: RDSUM له معنى فقط عندما يتم استدعاء STGSY2 بواسطة STGSYL.
!>

RDSCAL

!>          RDSCAL هو عدد حقيقي (REAL)
!>          عند الإدخال، عامل القياس المستخدم لمنع تجاوز السعة في RDSUM.
!>          عند الخروج، يتم تحديث RDSCAL فيما يتعلق بالمساهمات
!>          الحالية في RDSUM.
!>          إذا كان TRANS = 'T'، فلن يتم المساس بـ RDSCAL.
!>          ملاحظة: RDSCAL له معنى فقط عندما يتم استدعاء STGSY2 بواسطة
!>                STGSYL.
!>

IPIV

!>          IPIV هي مصفوفة أعداد صحيحة (INTEGER array)، الأبعاد (N).
!>          فهارس المحور؛ لـ 1 <= i <= N، تم تبديل الصف i من
!>          المصفوفة مع الصف IPIV(i).
!>

JPIV

!>          JPIV هي مصفوفة أعداد صحيحة (INTEGER array)، الأبعاد (N).
!>          فهارس المحور؛ لـ 1 <= j <= N، تم تبديل العمود j من
!>          المصفوفة مع العمود JPIV(j).
!>
