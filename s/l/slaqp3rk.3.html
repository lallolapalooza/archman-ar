subroutine slaqp3rk (integer m, integer n, integer nrhs, integer ioffset, integer nb, real abstol, real reltol, integer kp1, real maxc2nrm, real, dimension( lda, * ) a, integer lda, logical done, integer kb, real maxc2nrmk, real relmaxc2nrmk, integer, dimension( * ) jpiv, real, dimension( * ) tau, real, dimension( * ) vn1, real, dimension( * ) vn2, real, dimension( * ) auxv, real, dimension( ldf, * ) f, integer ldf, integer, dimension( * ) iwork, integer info)

SLAQP3RK يحسب خطوة من التحليل المتدرج QR مع تبديل الأعمدة لمصفوفة حقيقية m-by-n A ويقوم بالكتابة فوق مصفوفة حقيقية m-by-nrhs B بـ Q**T * B.

الغرض:

!>
!> SLAQP3RK يحسب خطوة من التحليل المتدرج QR مع تبديل الأعمدة
!> لمصفوفة حقيقية M-by-N A بلوك A(IOFFSET+1:M,1:N)
!> باستخدام BLAS المستوى 3 كـ
!>
!>   A * P(KB) = Q(KB) * R(KB).
!>
!> الإجراء الفرعي يحاول تحليل NB عمودًا من A بدءًا من
!> الصف IOFFSET+1 وتحديث المصفوفة المتبقية باستخدام BLAS 3
!> xGEMM. يتم إرجاع عدد الأعمدة المحللة بالفعل
!> أصغر من NB.
!>
!> يتم تبديل بلوك A(1:IOFFSET,1:N) بشكل مناسب، ولكن لا يتم تحليله.
!>
!> يقوم الإجراء الفرعي أيضًا بالكتابة فوق الجوانب اليمنى لمصفوفة B المخزنة
!> في A(IOFFSET+1:M,1:N+1:N+NRHS) بـ Q(KB)**T * B.
!>
!> حالات يكون فيها عدد الأعمدة المحللة KB < NB:
!>
!> (1) في بعض الحالات، بسبب الإلغاء الكارثي، لا يمكنه
!> تحليل جميع أعمدة NB وتحتاج إلى تحديث المصفوفة المتبقية.
!> وبالتالي، فإن العدد الفعلي للأعمدة المحللة في البلوك المرجع
!> في KB أصغر من NB. يتم إرجاع المنطقي DONE كـ FALSE.
!> يجب أن تستمر عملية تحليل المصفوفة الأصلية A_orig بأكملها
!> مع البلوك التالي.
!>
!> (2) كلما تم استيفاء معيار التوقف ABSTOL أو RELTOL،
!> تتوقف عملية تحليل المصفوفة الأصلية A_orig بأكملها،
!> ويتم إرجاع المنطقي DONE كـ TRUE. يتم إرجاع عدد الأعمدة المحللة
!> والذي هو أصغر من NB في KB.
!>
!> (3) في حالة عدم استخدام كلا معيري التوقف ABSTOL أو RELTOL،
!> وعندما تكون المصفوفة المتبقية مصفوفة صفرية في خطوة تحليل
!> KB، تتوقف عملية تحليل المصفوفة الأصلية A_orig بأكملها،
!> ويتم إرجاع المنطقي DONE كـ TRUE. يتم إرجاع عدد الأعمدة المحللة
!> والذي هو أصغر من NB في KB.
!>
!> (4) كلما تم اكتشاف NaN في المصفوفة A أو في المصفوفة TAU،
!> تتوقف عملية تحليل المصفوفة الأصلية A_orig بأكملها،
!> ويتم إرجاع المنطقي DONE كـ TRUE. يتم إرجاع عدد الأعمدة المحللة
!> والذي هو أصغر من NB في KB. يتم تعيين المعلمة INFO
!> إلى فهرس العمود لأول ظهور لـ NaN.
!>
!>

!>
!>          التفاوت المطلق (عتبة التوقف) لـ
!>          معيار 2-norm للأعمدة القصوى للمصفوفة المتبقية.
!>          تتقارب الخوارزمية (توقف التحليل) عندما
!>          يكون معيار 2-norm للأعمدة القصوى للمصفوفة المتبقية
!>          أقل من أو يساوي ABSTOL.
!>
!>          أ) إذا كان ABSTOL < 0.0، فإن معيار التوقف هذا
!>                غير مستخدم، ويقوم الإجراء الفرعي بتحليل الأعمدة اعتمادًا
!>                على NB و RELTOL.
!>                هذا يشمل الحالة ABSTOL = -Inf.
!>
!>          ب) إذا كان 0.0 <= ABSTOL، يتم استخدام القيمة المدخلة
!>                لـ ABSTOL.
!>

RELTOL

!>          RELTOL is REAL, cannot be NaN.
!>
!>          التفاوت (عتبة التوقف) لنسبة
!>          معيار 2-norm للأعمدة القصوى للمصفوفة المتبقية إلى
!>          معيار 2-norm للأعمدة القصوى للمصفوفة الأصلية A_orig. تتقارب الخوارزمية
!>          (تتوقف عملية التحليل)، عندما تكون هذه النسبة
!>          أقل من أو تساوي RELTOL.
!>
!>          أ) إذا كان RELTOL < 0.0، فإن معيار التوقف هذا
!>                غير مستخدم، ويقوم الإجراء الفرعي بتحليل الأعمدة اعتمادًا
!>                على NB و ABSTOL.
!>                هذا يشمل الحالة RELTOL = -Inf.
!>
!>          د) إذا كان 0.0 <= RELTOL، يتم استخدام القيمة المدخلة
!>                لـ RELTOL.
!>

KP1

!>          KP1 is INTEGER

!>          المصفوفة الأصلية A_orig بأكملها والمحسوب في

!>          الإجراء الرئيسي SGEQP3RK.
!>          MAXC2NRM >= 0.
!>

!>          A is REAL array, dimension (LDA,N+NRHS)
!>          عند الإدخال:
!>              المصفوفة A بقياس M-by-N والمصفوفة B بقياس M-by-NRHS، كما في
!>
!>                                  N     NRHS
!>              array_A   =   M  [ mat_A, mat_B ]
!>
!>          عند الخروج:
!>          1. العناصر في بلوك A(IOFFSET+1:M,1:KB) أسفل
!>             القطر مع مصفوفة TAU تمثل

NAME

Slaqp3rk

Rank-revealing QR factorization of a general matrix using block Wythe

SYNOPSIS

SUBROUTINE SLAQP3RK(
     M, N, NB, A, LDA, SRCDIST, IPIV, TAU, AF, LDAF, WORK, LWORK,
     PERM, RESID, COLAVL, THRESH, PIVNUM, LOGICAL, KB, MAXC2NRMK,
     RELMAXC2NRMK, JPIV, AUXV, F, LDF, IWORK, INFO
)

Parameters:

M
INTEGER
The number of rows of the matrix A.
N
INTEGER
The number of columns of the matrix A.
NB
INTEGER
The blocking size for the loop.
A
REAL array, dimension (LDA, N)
On entry, the M-by-N matrix A.
LDA
INTEGER
The leading dimension of the array A.
SRCDIST
REAL
The column distances of the columns of A. SRCDIST(j) is the
distance of column j from the next column that has a larger
2-norm.
IPIV
INTEGER array, dimension (N)
pivot indices: if IPIV(j) = k, then the j-th column was swapped
with the k-th column.
TAU
REAL array, dimension (min(M, N))
The scalar factors of the elementary reflectors.
AF
REAL array, dimension (LDAF, N)
On exit, AF is an upper triangular matrix whose diagonal elements
are computed from the block LU factorization with partial pivoting.
AF is stored in the upper triangle of the array A.
LDAF
INTEGER
The leading dimension of the array AF.
WORK
REAL array, dimension (LWORK)
Workspace array.
LWORK
INTEGER
The dimension of the array WORK. LWORK >= max(3*N, N+NB*(N+1)).
PERM
INTEGER array, dimension (N)
The permutation vector which permutes the columns of A.
RESID
REAL
The 2-norm of the residual matrix A*Q - U.
COLAVL
REAL array, dimension (N)
The vector of the column norms.
THRESH
REAL
The threshold used to define the rank of the matrix.
PIVNUM
REAL
The pivot number, which is the maximum of the column norms.
LOGICAL
LOGICAL
LOGICAL flag:
TRUE: a) if the factorization completed before processing
all min(M-IOFFSET,NB,N) columns due to ABSTOL
or RELTOL criterion,
b) if the factorization completed before processing
all min(M-IOFFSET,NB,N) columns due to the
residual matrix being a ZERO matrix.
c) when NaN was detected in the matrix A
or in the array TAU.
FALSE: otherwise.
KB
INTEGER
Factorization rank of the matrix A, i.e. the rank of
the factor R, which is the same as the number of non-zero
rows of the factor R. 0 <= KB <= min(M-IOFFSET,NB,N).
KB also represents the number of non-zero Householder
vectors.
MAXC2NRMK
REAL
The maximum column 2-norm of the residual matrix,
when the factorization stopped at rank KB. MAXC2NRMK >= 0.
RELMAXC2NRMK
REAL
The ratio MAXC2NRMK / MAXC2NRM of the maximum column
2-norm of the residual matrix (when the factorization
stopped at rank KB) to the maximum column 2-norm of the
original matrix A_orig. RELMAXC2NRMK >= 0.
JPIV
INTEGER array, dimension (N)
Column pivot indices, for 1 <= j <= N, column j
of the matrix A was interchanged with column JPIV(j).
TAU
REAL array, dimension (min(M-IOFFSET,N))
The scalar factors of the elementary reflectors.
VN1
REAL array, dimension (N)
The vector with the partial column norms.
VN2
REAL array, dimension (N)
The vector with the exact column norms.
AUXV
REAL array, dimension (NB)
Auxiliary vector.
F
REAL array, dimension (LDF,NB)
Matrix F**T = L*(Y**T)*A.
LDF
INTEGER
The leading dimension of the array F. LDF >= max(1,N+NRHS).
IWORK
INTEGER array, dimension (N-1).
Is a work array. ( IWORK is used to store indices
of  columns for norm downdating in the residual
matrix ).
INFO
INTEGER
1) INFO = 0: successful exit.
2) If INFO = j_1, where 1 <= j_1 <= N, then NaN was
detected and the routine stops the computation.
The j_1-th column of the matrix A or the j_1-th
element of array TAU contains the first occurrence
of NaN in the factorization step KB+1 ( when KB columns
have been factorized ).
On exit:
KB                  is set to the number of
factorized columns without
exception.
MAXC2NRMK           is set to NaN.
RELMAXC2NRMK        is set to NaN.
TAU(KB+1:min(M,N))     is not set and contains undefined
elements. If j_1=KB+1, TAU(KB+1)
may contain NaN.
3) If INFO = j_2, where N+1 <= j_2 <= 2*N, then no NaN
was detected, but +Inf (or -Inf) was detected and
the routine continues the computation until completion.
The (j_2-N)-th column of the matrix A contains the first
occurrence of +Inf (or -Inf) in the actorization
step KB+1 ( when KB columns have been factorized ).
