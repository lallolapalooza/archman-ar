وثائق الوظائف/الروتينات الفرعية

subroutine slaptm (integer n, integer nrhs, real alpha, real, dimension( * ) d, real, dimension( * ) e, real, dimension( ldx, * ) x, integer ldx, real beta, real, dimension( ldb, * ) b, integer ldb)

SLAPTM

الغرض:

!>
!> SLAPTM تضرب مصفوفة X ذات الأبعاد N × NRHS بمصفوفة متماثلة ثلاثية الأقطار A
!> وتخزن النتيجة في مصفوفة B. العملية لها الشكل
!>
!>    B := alpha * A * X + beta * B
!>
!> حيث يمكن أن يكون alpha إما 1. أو -1. ويمكن أن يكون beta 0. أو 1. أو -1.
!>

المعلمات

N 
!>          N هو عدد صحيح (INTEGER)
!>          رتبة المصفوفة A. N >= 0.
!>

NRHS

!>          NRHS هو عدد صحيح (INTEGER)
!>          عدد الأطراف اليمنى، أي عدد أعمدة المصفوفتين X و B.
!>

ALPHA

!>          ALPHA هو عدد حقيقي (REAL)
!>          الثابت alpha. يجب أن يكون ALPHA 1. أو -1.؛ بخلاف ذلك،
!>          يُفترض أنه 0.
!>

D

!>          D هو مصفوفة حقيقية (REAL array)، الأبعاد (N)
!>          العناصر القطرية n للمصفوفة الثلاثية الأقطار A.
!>

E

!>          E هو مصفوفة حقيقية (REAL array)، الأبعاد (N-1)
!>          العناصر دون القطرية أو فوق القطرية (n-1) للمصفوفة A.
!>

X

!>          X هي مصفوفة حقيقية (REAL array)، الأبعاد (LDX,NRHS)
!>          المصفوفة X ذات الأبعاد N × NRHS.
!>

LDX

!>          LDX هو عدد صحيح (INTEGER)
!>          البعد الرائد للمصفوفة X. LDX >= max(N,1).
!>

BETA

!>          BETA هو عدد حقيقي (REAL)
!>          الثابت beta. يجب أن يكون BETA 0. أو 1. أو -1.؛ بخلاف ذلك،
!>          يُفترض أنه 1.
!>

B

!>          B هي مصفوفة حقيقية (REAL array)، الأبعاد (LDB,NRHS)
!>          عند الإدخال، المصفوفة B ذات الأبعاد N × NRHS.
!>          عند الخروج، يتم استبدال B بالتعبير المصفوفي
!>          B := alpha * A * X + beta * B.
!>

LDB

!>          LDB هو عدد صحيح (INTEGER)
!>          البعد الرائد للمصفوفة B. LDB >= max(N,1).
!>
