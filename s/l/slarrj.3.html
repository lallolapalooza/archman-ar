توثيق الوظيفة/الروتين الفرعي

subroutine slarrj (integer n, real, dimension( * ) d, real, dimension( * ) e2, integer ifirst, integer ilast, real rtol, integer offset, real, dimension( * ) w, real, dimension( * ) werr, real, dimension( * ) work, integer, dimension( * ) iwork, real pivmin, real spdiam, integer info)

SLARRJ تقوم بتحسين التقديرات الأولية للقيم الذاتية للمصفوفة T.

الغرض:

!>
!> بالنظر إلى التقديرات الأولية للقيم الذاتية لـ T، تقوم SLARRJ
!> بتنفيذ التنصيف لتحسين القيم الذاتية لـ T،
!> W( IFIRST-OFFSET ) إلى W( ILAST-OFFSET )، بدقة أكبر. التخمينات الأولية
!> لهذه القيم الذاتية يتم إدخالها في W، والتقدير المقابل
!> للخطأ في هذه التخمينات في WERR. أثناء التنصيف، يتم الحفاظ على الفواصل
!> [left, right] عن طريق تخزين نقاط منتصفها و
!> أنصاف عرضها في المصفوفات W و WERR على التوالي.
!>

المعلمات

N 
!>          N هو عدد صحيح (INTEGER)
!>          رتبة المصفوفة.
!>

D

!>          D هي مصفوفة حقيقية (REAL array)، ذات بعد (N)
!>          العناصر القطرية N للمصفوفة T.
!>

E2

!>          E2 هي مصفوفة حقيقية (REAL array)، ذات بعد (N-1)
!>          مربعات العناصر الفرعية القطرية (N-1) للمصفوفة T.
!>

IFIRST

!>          IFIRST هو عدد صحيح (INTEGER)
!>          فهرس أول قيمة ذاتية سيتم حسابها.
!>

ILAST

!>          ILAST هو عدد صحيح (INTEGER)
!>          فهرس آخر قيمة ذاتية سيتم حسابها.
!>

RTOL

!>          RTOL هو عدد حقيقي (REAL)
!>          التسامح لتقارب فواصل التنصيف.
!>          فاصل [LEFT,RIGHT] يتقارب إذا كان
!>          RIGHT-LEFT < RTOL*MAX(|LEFT|,|RIGHT|).
!>

OFFSET

!>          OFFSET هو عدد صحيح (INTEGER)
!>          إزاحة لمصفوفات W و WERR، أي العناصر IFIRST-OFFSET
!>          إلى ILAST-OFFSET في هذه المصفوفات سيتم استخدامها.
!>

W

!>          W هي مصفوفة حقيقية (REAL array)، ذات بعد (N)
!>          عند الإدخال، W( IFIRST-OFFSET ) إلى W( ILAST-OFFSET ) هي
!>          تقديرات للقيم الذاتية للمصفوفة L D L^T المفهرسة IFIRST إلى
!>          ILAST.
!>          عند الإخراج، يتم تحسين هذه التقديرات.
!>

WERR

!>          WERR هي مصفوفة حقيقية (REAL array)، ذات بعد (N)
!>          عند الإدخال، WERR( IFIRST-OFFSET ) إلى WERR( ILAST-OFFSET ) هي
!>          الأخطاء في تقديرات العناصر المقابلة في W.
!>          عند الإخراج، يتم تحسين هذه الأخطاء.
!>

WORK

!>          WORK هي مصفوفة حقيقية (REAL array)، ذات بعد (2*N)
!>          مساحة عمل.
!>

IWORK

!>          IWORK هي مصفوفة أعداد صحيحة (INTEGER array)، ذات بعد (2*N)
!>          مساحة عمل.
!>

PIVMIN

!>          PIVMIN هو عدد حقيقي (REAL)
!>          الارتكاز الأدنى في متتالية ستورم للمصفوفة T.
!>

SPDIAM

!>          SPDIAM هو عدد حقيقي (REAL)
!>          القطر الطيفي للمصفوفة T.
!>

INFO

!>          INFO هو عدد صحيح (INTEGER)
!>          علم الخطأ.
!>
