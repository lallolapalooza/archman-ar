subroutine slartv (integer n, real, dimension( * ) x, integer incx, real, dimension( * ) y, integer incy, real, dimension( * ) c, real, dimension( * ) s, integer incc)

SLARTV يطبق متجه دوران مستوي مع جيب تمام حقيقي وجيب حقيقي على عناصر زوج من المتجهات.

الغرض:

!>
!> SLARTV يطبق متجه دوران مستوي حقيقي على عناصر المتجهات الحقيقية x و y. لكل i = 1,2,...,n
!>
!>    ( x(i) ) := (  c(i)  s(i) ) ( x(i) )
!>    ( y(i) )    ( -s(i)  c(i) ) ( y(i) )
!>

المعلمات

N 
!>          N هو عدد صحيح (INTEGER)
!>          عدد دورانات المستوي المطلوب تطبيقها.
!>

X

!>          X هو مصفوفة حقيقية (REAL array)،
!>                         البعد (1+(N-1)*INCX)
!>          المتجه x.
!>

INCX

!>          INCX هو عدد صحيح (INTEGER)
!>          الزيادة بين عناصر X. INCX > 0.
!>

Y

!>          Y هو مصفوفة حقيقية (REAL array)،
!>                         البعد (1+(N-1)*INCY)
!>          المتجه y.
!>

INCY

!>          INCY هو عدد صحيح (INTEGER)
!>          الزيادة بين عناصر Y. INCY > 0.
!>

C

!>          C هي مصفوفة حقيقية (REAL array)، البعد (1+(N-1)*INCC)
!>          جيب تمام دورانات المستوي.
!>

S

!>          S هي مصفوفة حقيقية (REAL array)، البعد (1+(N-1)*INCC)
!>          جيب دورانات المستوي.
!>

INCC

!>          INCC هو عدد صحيح (INTEGER)
!>          الزيادة بين عناصر C و S. INCC > 0.
!>
