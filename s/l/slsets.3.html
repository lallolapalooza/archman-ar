subroutine slsets (integer m, integer p, integer n, real, dimension( lda, * ) a, real, dimension( lda, * ) af, integer lda, real, dimension( ldb, * ) b, real, dimension( ldb, * ) bf, integer ldb, real, dimension( * ) c, real, dimension( * ) cf, real, dimension( * ) d, real, dimension( * ) df, real, dimension( * ) x, real, dimension( lwork ) work, integer lwork, real, dimension( * ) rwork, real, dimension( 2 ) result)

SLSETS

الغرض:

!>
!> SLSETS تختبر SGGLSE - وهي روتين فرعي لحل مشكلة الحد الأدنى المربعات المقيد بالمعادلات الخطية (LSE).
!>

المعلمات

M 
!>          M هو عدد صحيح
!>          عدد صفوف المصفوفة A.  M >= 0.
!>

P

!>          P هو عدد صحيح
!>          عدد صفوف المصفوفة B.  P >= 0.
!>

N

!>          N هو عدد صحيح
!>          عدد أعمدة المصفوفتين A و B.  N >= 0.
!>

A

!>          A هي مصفوفة REAL، أبعادها (LDA,N)
!>          المصفوفة A بحجم M×N.
!>

AF

!>          AF هي مصفوفة REAL، أبعادها (LDA,N)
!>

LDA

!>          LDA هو عدد صحيح
!>          البعد الرئيسي للمصفوفات A، AF، Q، و R.
!>          LDA >= max(M,N).
!>

B

!>          B هي مصفوفة REAL، أبعادها (LDB,N)
!>          المصفوفة A بحجم P×N.
!>

BF

!>          BF هي مصفوفة REAL، أبعادها (LDB,N)
!>

LDB

!>          LDB هو عدد صحيح
!>          البعد الرئيسي للمصفوفات B، BF، V، و S.
!>          LDB >= max(P,N).
!>

C

!>          C هي مصفوفة REAL، أبعادها ( M )
!>          المتجه C في مشكلة LSE.
!>

CF

!>          CF هي مصفوفة REAL، أبعادها ( M )
!>

D

!>          D هي مصفوفة REAL، أبعادها ( P )
!>          المتجه D في مشكلة LSE.
!>

DF

!>          DF هي مصفوفة REAL، أبعادها ( P )
!>

X

!>          X هي مصفوفة REAL، أبعادها ( N )
!>          متجه الحل X في مشكلة LSE.
!>

WORK

!>          WORK هي مصفوفة REAL، أبعادها (LWORK)
!>

LWORK

!>          LWORK هو عدد صحيح
!>          بعد المصفوفة WORK.
!>

RWORK

!>          RWORK هي مصفوفة REAL، أبعادها (M)
!>

RESULT

!>          RESULT هي مصفوفة REAL، أبعادها (2)
!>          نسب الاختبار:
!>            RESULT(1) = norm( A*x - c )/ norm(A)*norm(X)*EPS
!>            RESULT(2) = norm( B*x - d )/ norm(B)*norm(X)*EPS
!>
