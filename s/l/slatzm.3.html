توثيق الوظيفة/الروتين الفرعي

subroutine slatzm (character side, integer m, integer n, real, dimension( * ) v, integer incv, real tau, real, dimension( ldc, * ) c1, real, dimension( ldc, * ) c2, integer ldc, real, dimension( * ) work)

SLATZM

الغرض:

!>
!> هذا الروتين مهمل وتم استبداله بالروتين SORMRZ.
!>
!> SLATZM يطبق مصفوفة هاوسهولدر تم إنشاؤها بواسطة STZRQF على مصفوفة.
!>
!> دع P = I - tau*u*u**T,   u = ( 1 ),
!>                               ( v )
!> حيث v هو متجه (m-1) إذا كان SIDE = 'L'، أو متجه (n-1) إذا كان
!> SIDE = 'R'.
!>
!> إذا كان SIDE يساوي 'L'، دع
!>        C = [ C1 ] 1
!>            [ C2 ] m-1
!>              n
!> إذن C يتم استبدالها بـ P*C.
!>
!> إذا كان SIDE يساوي 'R'، دع
!>        C = [ C1, C2 ] m
!>               1  n-1
!> إذن C يتم استبدالها بـ C*P.
!>

المعلمات

SIDE 
!>          SIDE هو CHARACTER*1
!>          = 'L': تكوين P * C
!>          = 'R': تكوين C * P
!>

M

!>          M هو INTEGER
!>          عدد صفوف المصفوفة C.
!>

N

!>          N هو INTEGER
!>          عدد أعمدة المصفوفة C.
!>

V

!>          V هو مصفوفة REAL، البعد
!>                  (1 + (M-1)*abs(INCV)) إذا كان SIDE = 'L'
!>                  (1 + (N-1)*abs(INCV)) إذا كان SIDE = 'R'
!>          المتجه v في تمثيل P. لا يتم استخدام V
!>          إذا كان TAU = 0.
!>

INCV

!>          INCV هو INTEGER
!>          الزيادة بين عناصر v. INCV <> 0
!>

TAU

!>          TAU هو REAL
!>          القيمة tau في تمثيل P.
!>

C1

!>          C1 هو مصفوفة REAL، البعد
!>                         (LDC,N) إذا كان SIDE = 'L'
!>                         (M,1)   إذا كان SIDE = 'R'
!>          عند الإدخال، المتجه n C1 إذا كان SIDE = 'L'، أو المتجه m C1
!>          إذا كان SIDE = 'R'.
!>
!>          عند الخروج، الصف الأول من P*C إذا كان SIDE = 'L'، أو العمود الأول
!>          من C*P إذا كان SIDE = 'R'.
!>

C2

!>          C2 هو مصفوفة REAL، البعد
!>                         (LDC, N)   إذا كان SIDE = 'L'
!>                         (LDC, N-1) إذا كان SIDE = 'R'
!>          عند الإدخال، المصفوفة (m - 1) x n C2 إذا كان SIDE = 'L'، أو
!>          المصفوفة m x (n - 1) C2 إذا كان SIDE = 'R'.
!>
!>          عند الخروج، الصفوف 2:m من P*C إذا كان SIDE = 'L'، أو الأعمدة 2:m من C*P
!>          إذا كان SIDE = 'R'.
!>

LDC

!>          LDC هو INTEGER
!>          البعد الرائد للمصفوفات C1 و C2. LDC >= (1,M).
!>

WORK

!>          WORK هو مصفوفة REAL، البعد
!>                      (N) إذا كان SIDE = 'L'
!>                      (M) إذا كان SIDE = 'R'
!>
