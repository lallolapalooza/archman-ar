subroutine slarrb (integer n, real, dimension( * ) d, real, dimension( * ) lld, integer ifirst, integer ilast, real rtol1, real rtol2, integer offset, real, dimension( * ) w, real, dimension( * ) wgap, real, dimension( * ) werr, real, dimension( * ) work, integer, dimension( * ) iwork, real pivmin, real spdiam, integer twist, integer info)

SLARRB توفر تقسيمًا محدودًا لتحديد القيم الذاتية لمزيد من الدقة.

الغرض:

!>
!> نظرًا للتمثيل النسبي القوي (RRR) L D L^T، فإن SLARRB
!> يقوم بعمل تقسيم لتنقيح القيم الذاتية لـ L D L^T،
!> W( IFIRST-OFFSET ) حتى W( ILAST-OFFSET )، لمزيد من الدقة. المدخلات
!> التقديرية لهذه القيم الذاتية موجودة في W، وتقدير الخطأ المقابل
!> لها وفجواتها موجود في WERR و WGAP على التوالي. أثناء التقسيم،
!> يتم الحفاظ على الفترات [left, right] عن طريق تخزين نقاط المنتصف
!> ونصف القطر في المصفوفات W و WERR على التوالي.
!>

المعلمات

N 
!>          N هو عدد صحيح
!>          رتبة المصفوفة.
!>

D

!>          D هي مصفوفة حقيقية، البعد (N)
!>          عناصر القطر N للمصفوفة القطرية D.
!>

LLD

!>          LLD هي مصفوفة حقيقية، البعد (N-1)
!>          العناصر (N-1) L(i)*L(i)*D(i).
!>

IFIRST

!>          IFIRST هو عدد صحيح
!>          فهرس أول قيمة ذاتية سيتم حسابها.
!>

ILAST

!>          ILAST هو عدد صحيح
!>          فهرس آخر قيمة ذاتية سيتم حسابها.
!>

RTOL1

!>          RTOL1 هو عدد حقيقي
!>

RTOL2

!>          RTOL2 هو عدد حقيقي
!>          التسامح لتقارب فترات التقسيم.
!>          فترة [LEFT,RIGHT] تعتبر متقاربة إذا
!>          RIGHT-LEFT < MAX( RTOL1*GAP, RTOL2*MAX(|LEFT|,|RIGHT|) )
!>          حيث GAP هي المسافة (المقدرة) إلى أقرب
!>          قيمة ذاتية.
!>

OFFSET

!>          OFFSET هو عدد صحيح
!>          الإزاحة لمصفوفات W، WGAP، و WERR، أي، سيتم استخدام العناصر
!>          من IFIRST-OFFSET إلى ILAST-OFFSET لهذه المصفوفات.
!>

W

!>          W هي مصفوفة حقيقية، البعد (N)
!>          عند الإدخال، W( IFIRST-OFFSET ) حتى W( ILAST-OFFSET ) هي
!>          تقديرات للقيم الذاتية لـ L D L^T المرقمة IFIRST حتى
!>          ILAST.
!>          عند الإخراج، يتم تنقيح هذه التقديرات.
!>

WGAP

!>          WGAP هي مصفوفة حقيقية، البعد (N-1)
!>          عند الإدخال، فجوات (تقديرية) بين المتتالية
!>          القيم الذاتية لـ L D L^T، أي، WGAP(I-OFFSET) هي الفجوة بين
!>          القيم الذاتية I و I+1. لاحظ أنه إذا كان IFIRST = ILAST
!>          فيجب تعيين WGAP(IFIRST-OFFSET) إلى الصفر.
!>          عند الإخراج، يتم تنقيح هذه الفجوات.
!>

WERR

!>          WERR هي مصفوفة حقيقية، البعد (N)
!>          عند الإدخال، WERR( IFIRST-OFFSET ) حتى WERR( ILAST-OFFSET ) هي
!>          الأخطاء في تقديرات العناصر المقابلة في W.
!>          عند الإخراج، يتم تنقيح هذه الأخطاء.
!>

WORK

!>          WORK هي مصفوفة حقيقية، البعد (2*N)
!>          مساحة عمل.
!>

IWORK

!>          IWORK هي مصفوفة أعداد صحيحة، البعد (2*N)
!>          مساحة عمل.
!>

PIVMIN

!>          PIVMIN هو عدد حقيقي
!>          أصغر محور في متتالية ستورم.
!>

SPDIAM

!>          SPDIAM هو عدد حقيقي
!>          القطر الطيفي للمصفوفة.
!>

TWIST

!>          TWIST هو عدد صحيح
!>          مؤشر الالتواء للتحليل الملوي المستخدم
!>          لحساب العدد السلبي (negcount).
!>          TWIST = N: حساب العدد السلبي من L D L^T - LAMBDA I = L+ D+ L+^T
!>          TWIST = 1: حساب العدد السلبي من L D L^T - LAMBDA I = U- D- U-^T
!>          TWIST = R: حساب العدد السلبي من L D L^T - LAMBDA I = N(r) D(r) N(r)
!>

INFO

!>          INFO هو عدد صحيح
!>          علامة الخطأ.
!>
