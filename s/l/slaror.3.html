subroutine slaror (side, init, m, n, a, lda, iseed, x, info)
SLAROR

وثائق الوظيفة/الروتين الفرعي

subroutine slaror (character side, character init, integer m, integer n, real, dimension( lda, * ) a, integer lda, integer, dimension( 4 ) iseed, real, dimension( * ) x, integer info)

SLAROR

الغرض:

!>
!> SLAROR يضرب مسبقًا أو لاحقًا مصفوفة M × N A بمصفوفة متعامدة عشوائية
!> U، مع الكتابة فوق A. قد يتم تهيئة A اختياريًا إلى مصفوفة الوحدة
!> قبل ضربها بـ U. يتم إنشاء U باستخدام
!> طريقة G.W. Stewart (SIAM J. Numer. Anal. 17, 1980, 403-409).
!>

المعلمات

SIDE 
!>          SIDE هو CHARACTER*1
!>          يحدد ما إذا كانت A مضروبة على اليسار أو اليمين بواسطة U.
!>          = 'L':         اضرب A على اليسار (اضرب مسبقًا) بواسطة U
!>          = 'R':         اضرب A على اليمين (اضرب لاحقًا) بواسطة U'
!>          = 'C' أو 'T':  اضرب A على اليسار بواسطة U وعلى اليمين
!>                          بواسطة U' (هنا، U' تعني U-transpose.)
!>

INIT

!>          INIT هو CHARACTER*1
!>          يحدد ما إذا كان ينبغي تهيئة A إلى
!>          مصفوفة الوحدة أم لا.
!>          = 'I':  تهيئة A إلى (قسم من) مصفوفة الوحدة
!>                   قبل تطبيق U.
!>          = 'N':  لا توجد تهيئة. قم بتطبيق U على المصفوفة المدخلة A.
!>
!>          يمكن استخدام INIT = 'I' لإنشاء
!>          مصفوفات متعامدة مربعة أو مستطيلة:
!>
!>          لـ M = N و SIDE = 'L' أو 'R'، ستكون الصفوف متعامدة
!>          على بعضها البعض، وكذلك الأعمدة.
!>
!>          إذا كانت M < N، فإن SIDE = 'R' ينتج مصفوفة كثيفة صفوفها
!>          متعامدة وأعمدتها ليست كذلك، بينما SIDE = 'L'
!>          ينتج مصفوفة صفوفها متعامدة، وأعمدتها الأولى
!>          M متعامدة، وأعمدتها المتبقية
!>          صفر.
!>
!>          إذا كانت M > N، فإن SIDE = 'L' ينتج مصفوفة كثيفة أعمدتها
!>          متعامدة وصفوفها ليست كذلك، بينما SIDE = 'R'
!>          ينتج مصفوفة أعمدتها متعامدة، وصفوفها الأولى
!>          M متعامدة، وصفوفها المتبقية
!>          صفر.
!>

M

!>          M هو INTEGER
!>          عدد صفوف A.
!>

N

!>          N هو INTEGER
!>          عدد أعمدة A.
!>

A

!>          A هو مصفوفة REAL، بعد (LDA, N)
!>          عند الإدخال، المصفوفة A.
!>          عند الخروج، يتم الكتابة فوقها بواسطة U A ( إذا كان SIDE = 'L' )،
!>           أو بواسطة A U ( إذا كان SIDE = 'R' )،
!>           أو بواسطة U A U' ( إذا كان SIDE = 'C' أو 'T' ).
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرائد للمصفوفة A. LDA >= max(1,M).
!>

ISEED

!>          ISEED هي مصفوفة INTEGER، بعد (4)
!>          عند الإدخال، تحدد ISEED بذرة مولد الأرقام
!>          العشوائية. يجب أن تكون عناصر المصفوفة بين 0 و 4095؛
!>          إذا لم تكن كذلك، فسيتم تقليلها حسب 4096. أيضًا، يجب أن يكون ISEED(4)
!>          فرديًا. يستخدم مولد الأرقام العشوائية تسلسلًا خطيًا متطابقًا
!>          محدودًا بالأعداد الصحيحة الصغيرة، وبالتالي
!>          يجب أن ينتج أرقامًا عشوائية مستقلة عن الآلة. القيم
!>          الخاصة بـ ISEED تتغير عند الخروج، ويمكن استخدامها في
!>          الاستدعاء التالي لـ SLAROR لمواصلة نفس التسلسل العشوائي
!>          للأرقام.
!>

X

!>          X هي مصفوفة REAL، بعد (3*MAX( M, N ))
!>          مساحة عمل بطول
!>              2*M + N إذا كان SIDE = 'L'،
!>              2*N + M إذا كان SIDE = 'R'،
!>              3*N     إذا كان SIDE = 'C' أو 'T'.
!>

INFO

!>          INFO هو INTEGER
!>          علم خطأ. يتم تعيينه إلى:
!>          = 0:  عودة طبيعية
!>          < 0:  إذا كان INFO = -k، فإن المعامل k لديه قيمة غير قانونية
!>          = 1:  إذا كانت الأرقام العشوائية التي تم إنشاؤها بواسطة SLARND سيئة.
!>
