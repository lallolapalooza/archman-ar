subroutine slag2d (m, n, sa, ldsa, a, lda, info)
SLAG2D تقوم بتحويل مصفوفة بدقة فردية إلى مصفوفة بدقة مزدوجة.

توثيق الوظيفة/الإجراء الفرعي

subroutine slag2d (integer m, integer n, real, dimension( ldsa, * ) sa, integer ldsa, double precision, dimension( lda, * ) a, integer lda, integer info)

SLAG2D تقوم بتحويل مصفوفة بدقة فردية، SA، إلى مصفوفة بدقة مزدوجة، A.

الغرض:

!>
!> SLAG2D تقوم بتحويل مصفوفة بدقة فردية، SA، إلى مصفوفة بدقة
!> مزدوجة، A.
!>
!> ملاحظة أنه بينما من الممكن حدوث تجاوز عند التحويل
!> من الدقة المزدوجة إلى الفردية، فإنه ليس من الممكن حدوث تجاوز عند
!> التحويل من الدقة الفردية إلى المزدوجة.
!>
!> هذه روتين مساعدة لذلك لا يوجد فحص للمعاملات.
!>

المعاملات

M 
!>          M هو عدد صحيح (INTEGER)
!>          عدد أسطر المصفوفة A.  M >= 0.
!>

N

!>          N هو عدد صحيح (INTEGER)
!>          عدد أعمدة المصفوفة A.  N >= 0.
!>

SA

!>          SA هي مصفوفة حقيقية (REAL)، الأبعاد (LDSA,N)
!>          عند الإدخال، مصفوفة المعاملات SA ذات الأبعاد M-by-N.
!>

LDSA

!>          LDSA هو عدد صحيح (INTEGER)
!>          البعد الرائد للمصفوفة SA.  LDSA >= max(1,M).
!>

A

!>          A هي مصفوفة بدقة مزدوجة (DOUBLE PRECISION)، الأبعاد (LDA,N)
!>          عند الخروج، مصفوفة المعاملات A ذات الأبعاد M-by-N.
!>

LDA

!>          LDA هو عدد صحيح (INTEGER)
!>          البعد الرائد للمصفوفة A.  LDA >= max(1,M).
!>

INFO

!>          INFO هو عدد صحيح (INTEGER)
!>          = 0:  الخروج بنجاح
!>
