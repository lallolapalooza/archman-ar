subroutine slavsy_rook (character uplo, character trans, character diag, integer n, integer nrhs, real, dimension( lda, * ) a, integer lda, integer, dimension( * ) ipiv, real, dimension( ldb, * ) b, integer ldb, integer info)

SLAVSY_ROOK

الغرض:

!>
!> SLAVSY_ROOK  يقوم بإحدى عمليات ضرب المصفوفة في المتجه:
!>    x := A*x  أو  x := A'*x,
!> حيث x هو متجه من N عنصر و A هو أحد العوامل
!> من التحلل الناتج عن SSYTRF_ROOK وهو U*D*U' أو L*D*L'.
!>
!> إذا كانت TRANS = 'N'، الضرب في U  أو U * D  (أو L  أو L * D)
!> إذا كانت TRANS = 'T'، الضرب في U' أو D * U' (أو L' أو D * L')
!> إذا كانت TRANS = 'C'، الضرب في U' أو D * U' (أو L' أو D * L')
!>

المعلمات:

UPLO 
!>          UPLO هو CHARACTER*1
!>          تحدد ما إذا كان العامل المخزن في A مثلثيًا علويًا أم سفليًا.
!>          = 'U':  مثلثي علوي
!>          = 'L':  مثلثي سفلي
!>

TRANS

!>          TRANS هو CHARACTER*1
!>          تحدد العملية التي سيتم تنفيذها:
!>          = 'N':  x := A*x
!>          = 'T':  x := A'*x
!>          = 'C':  x := A'*x
!>

DIAG

!>          DIAG هو CHARACTER*1
!>          تحدد ما إذا كانت كتل القطر وحدة أم لا.
!>          إذا كانت كتل القطر تعتبر وحدات، فإن A = U أو A = L، وإلا فإن A = U*D أو A = L*D.
!>          = 'U':  تُعتبر كتل القطر مصفوفات وحدة.
!>          = 'N':  تُعتبر كتل القطر مصفوفات غير وحدة.
!>

N

!>          N هو INTEGER
!>          عدد الصفوف والأعمدة للمصفوفة A.  N >= 0.
!>

NRHS

!>          NRHS هو INTEGER
!>          عدد الأطراف اليمنى، أي عدد المتجهات x التي سيتم ضربها في A.  NRHS >= 0.
!>

A

!>          A هو مصفوفة REAL، أبعادها (LDA,N)
!>          المصفوفة القطرية المتفرعة D والمعاملات المستخدمة للحصول على العامل U أو L
!>          كما تم حسابها بواسطة SSYTRF_ROOK.
!>          مخزنة كمصفوفة مثلثية ثنائية الأبعاد.
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرائد للمصفوفة A.  LDA >= max(1,N).
!>

IPIV

!>          IPIV هو مصفوفة INTEGER، أبعادها (N)
!>          تفاصيل التبادلات والهيكل القطري لـ D،
!>          كما تم تحديده بواسطة SSYTRF_ROOK.
!>
!>          إذا كانت UPLO = 'U':
!>               إذا كان IPIV(k) > 0، فقد تم تبديل الصفوف والأعمدة k و IPIV(k)
!>               وكان D(k,k) كتلة قطرية بحجم 1x1.
!>               (إذا كان IPIV( k ) = k، لم يتم إجراء تبديل).
!>
!>               إذا كان IPIV(k) < 0 و IPIV(k-1) < 0، فقد تم تبديل الصفوف
!>               والأعمدة k و -IPIV(k) وتم تبديل الصفوف
!>               والأعمدة k-1 و -IPIV(k-1)،
!>               وكان D(k-1:k,k-1:k) كتلة قطرية بحجم 2x2.
!>
!>          إذا كانت UPLO = 'L':
!>               إذا كان IPIV(k) > 0، فقد تم تبديل الصفوف والأعمدة k و IPIV(k)
!>               وكان D(k,k) كتلة قطرية بحجم 1x1.
!>               (إذا كان IPIV( k ) = k، لم يتم إجراء تبديل).
!>
!>               إذا كان IPIV(k) < 0 و IPIV(k+1) < 0، فقد تم تبديل الصفوف
!>               والأعمدة k و -IPIV(k) وتم تبديل الصفوف
!>               والأعمدة k+1 و -IPIV(k+1)،
!>               وكان D(k:k+1,k:k+1) كتلة قطرية بحجم 2x2.
!>

B

!>          B هي مصفوفة REAL، أبعادها (LDB,NRHS)
!>          عند الدخول، تحتوي B على NRHS متجهًا بطول N.
!>          عند الخروج، يتم استبدال B بمنتج A * B.
!>

LDB

!>          LDB هو INTEGER
!>          البعد الرائد للمصفوفة B.  LDB >= max(1,N).
!>

INFO

!>          INFO هو INTEGER
!>          = 0: تم الخروج بنجاح
!>          < 0: إذا كان INFO = -k، كان الوسيط k-th ذو قيمة غير قانونية
!>
