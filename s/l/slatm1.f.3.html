subroutine slatm1 (integer mode, real cond, integer irsign, integer idist, integer, dimension( 4 ) iseed, real, dimension( * ) d, integer n, integer info)

SLATM1

الغرض:

!>
!>    SLATM1 يحسب مدخلات D(1..N) حسب المواصفات
!>    MODE، COND و IRSIGN. IDIST و ISEED يحددان توليد
!>    أرقام عشوائية. SLATM1 تستدعى بواسطة SLATMR لتوليد
!>    مصفوفات اختبار عشوائية لبرامج LAPACK.
!>

المعلمات

MODE 
!>          MODE هو عدد صحيح (INTEGER)
!>           عند الإدخال يصف كيف يتم حساب D:
!>           MODE = 0 يعني عدم تغيير D.
!>           MODE = 1 يضبط D(1)=1 و D(2:N)=1.0/COND
!>           MODE = 2 يضبط D(1:N-1)=1 و D(N)=1.0/COND
!>           MODE = 3 يضبط D(I)=COND**(-(I-1)/(N-1))
!>           MODE = 4 يضبط D(i)=1 - (i-1)/(N-1)*(1 - 1/COND)
!>           MODE = 5 يضبط D لأرقام عشوائية في النطاق
!>                    ( 1/COND , 1 ) بحيث تكون لوغاريتماتها
!>                    موزعة بشكل منتظم.
!>           MODE = 6 يضبط D لأرقام عشوائية من نفس التوزيع
!>                    مثل بقية المصفوفة.
!>           MODE < 0 لها نفس المعنى مثل ABS(MODE)، باستثناء أن
!>              ترتيب عناصر D معكوس.
!>           لذلك إذا كان MODE موجبًا، فإن عناصر D تتراوح من
!>              1 إلى 1/COND، وإذا كان سالبًا، من 1/COND إلى 1،
!>           لا يتم تعديله.
!>

COND

!>          COND هو عدد حقيقي (REAL)
!>           عند الإدخال، يستخدم كما هو موضح تحت MODE أعلاه.
!>           إذا تم استخدامه، يجب أن يكون >= 1. لا يتم تعديله.
!>

IRSIGN

!>          IRSIGN هو عدد صحيح (INTEGER)
!>           عند الإدخال، إذا كان MODE ليس -6 أو 0 أو 6، يحدد إشارة
!>           مدخلات D
!>           0 => ترك مدخلات D دون تغيير
!>           1 => ضرب كل مدخل في D بـ 1 أو -1 باحتمالية 0.5
!>

IDIST

!>          IDIST هو عدد صحيح (INTEGER)
!>           عند الإدخال، يحدد IDIST نوع التوزيع الذي سيتم
!>           استخدامه لتوليد مصفوفة عشوائية.
!>           1 => منتظم ( 0, 1 )
!>           2 => منتظم ( -1, 1 )
!>           3 => طبيعي ( 0, 1 )
!>           لا يتم تعديله.
!>

ISEED

!>          ISEED هو مصفوفة أعداد صحيحة (INTEGER array)، البعد ( 4 )
!>           عند الإدخال، يحدد ISEED بذرة مولد الأرقام العشوائية.
!>           يستخدم مولد الأرقام العشوائية تسلسل خطي متطابق محدود
!>           لأعداد صحيحة صغيرة، وبالتالي يجب أن ينتج أرقامًا عشوائية
!>           مستقلة عن الجهاز. تتغير قيم ISEED عند الخروج، ويمكن
!>           استخدامها في الاستدعاء التالي لـ SLATM1
!>           لمواصلة نفس تسلسل الأرقام العشوائية.
!>           تتغير عند الخروج.
!>

D

!>          D هي مصفوفة أعداد حقيقية (REAL array)، البعد ( N )
!>           المصفوفة التي سيتم حسابها وفقًا لـ MODE و COND و IRSIGN.
!>           قد تتغير عند الخروج إذا كان MODE غير صفري.
!>

N

!>          N هو عدد صحيح (INTEGER)
!>           عدد مدخلات D. لا يتم تعديله.
!>

INFO

!>          INFO هو عدد صحيح (INTEGER)
!>            0  => إنهاء طبيعي
!>           -1  => إذا كان MODE ليس في النطاق -6 إلى 6
!>           -2  => إذا كان MODE ليس -6 أو 0 أو 6، و
!>                  IRSIGN ليس 0 أو 1
!>           -3  => إذا كان MODE ليس -6 أو 0 أو 6 و COND أقل من 1
!>           -4  => إذا كان MODE يساوي 6 أو -6 و IDIST ليس في النطاق 1 إلى 3
!>           -7  => إذا كان N سالبًا
!>
