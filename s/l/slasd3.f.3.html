توثيق الوظيفة/الروتين الفرعي

subroutine slasd3 (integer nl, integer nr, integer sqre, integer k, real, dimension( * ) d, real, dimension( ldq, * ) q, integer ldq, real, dimension( * ) dsigma, real, dimension( ldu, * ) u, integer ldu, real, dimension( ldu2, * ) u2, integer ldu2, real, dimension( ldvt, * ) vt, integer ldvt, real, dimension( ldvt2, * ) vt2, integer ldvt2, integer, dimension( * ) idxc, integer, dimension( * ) ctot, real, dimension( * ) z, integer info)

SLASD3 تجد جميع الجذور التربيعية لجذور المعادلة العلمانية، كما هي محددة بالقيم في D و Z، ثم تقوم بتحديث المتجهات المفردة عن طريق ضرب المصفوفات. تستخدم بواسطة sbdsdc.

الغرض:

!>
!> SLASD3 تجد جميع الجذور التربيعية لجذور المعادلة العلمانية، كما هي محددة
!> بالقيم في D و Z. تقوم بإجراء الاستدعاءات المناسبة لـ SLASD4 ثم تقوم بتحديث
!> المتجهات المفردة عن طريق ضرب المصفوفات.
!>
!> يتم استدعاء SLASD3 من SLASD1.
!>

المعلمات

NL 
!>          NL هو عدد صحيح (INTEGER)
!>         بعد الصف للمجموعة العليا. NL >= 1.
!>

NR

!>          NR هو عدد صحيح (INTEGER)
!>         بعد الصف للمجموعة السفلى. NR >= 1.
!>

SQRE

!>          SQRE هو عدد صحيح (INTEGER)
!>         = 0: المجموعة السفلى هي مصفوفة مربعة بقياس NR-by-NR.
!>         = 1: المجموعة السفلى هي مصفوفة مستطيلة بقياس NR-by-(NR+1).
!>
!>         المصفوفة ثنائية القطر لها N = NL + NR + 1 صفوف و
!>         M = N + SQRE >= N أعمدة.
!>

K

!>          K هو عدد صحيح (INTEGER)
!>         حجم المعادلة العلمانية، 1 =< K = < N.
!>

D

!>          D هي مصفوفة حقيقية (REAL)، ذات بعد (K)
!>         عند الخروج، الجذور التربيعية لجذور المعادلة العلمانية،
!>         بترتيب تصاعدي.
!>

Q

!>          Q هي مصفوفة حقيقية (REAL)، ذات بعد (LDQ,K)
!>

LDQ

!>          LDQ هو عدد صحيح (INTEGER)
!>         البعد الرائد للمصفوفة Q. LDQ >= K.
!>

DSIGMA

!>          DSIGMA هي مصفوفة حقيقية (REAL)، ذات بعد (K)
!>         تحتوي العناصر K الأولى من هذه المصفوفة على الجذور القديمة
!>         لمشكلة التحديث المخفضة. هذه هي الأقطاب
!>         للمعادلة العلمانية.
!>

U

!>          U هي مصفوفة حقيقية (REAL)، ذات بعد (LDU, N)
!>         الأعمدة N - K الأخيرة من هذه المصفوفة تحتوي على المتجهات
!>         المفردة اليسرى المخفضة.
!>

LDU

!>          LDU هو عدد صحيح (INTEGER)
!>         البعد الرائد للمصفوفة U. LDU >= N.
!>

U2

!>          U2 هي مصفوفة حقيقية (REAL)، ذات بعد (LDU2, N)
!>         الأعمدة K الأولى من هذه المصفوفة تحتوي على المتجهات
!>         المفردة اليسرى غير المخفضة
!>         للمشكلة المقسمة.
!>

LDU2

!>          LDU2 هو عدد صحيح (INTEGER)
!>         البعد الرائد للمصفوفة U2. LDU2 >= N.
!>

VT

!>          VT هي مصفوفة حقيقية (REAL)، ذات بعد (LDVT, M)
!>         الأعمدة M - K الأخيرة من VT**T تحتوي على المتجهات
!>         المفردة اليمنى المخفضة.
!>

LDVT

!>          LDVT هو عدد صحيح (INTEGER)
!>         البعد الرائد للمصفوفة VT. LDVT >= N.
!>

VT2

!>          VT2 هي مصفوفة حقيقية (REAL)، ذات بعد (LDVT2, N)
!>         الأعمدة K الأولى من VT2**T تحتوي على المتجهات
!>         المفردة اليمنى غير المخفضة
!>         للمشكلة المقسمة.
!>

LDVT2

!>          LDVT2 هو عدد صحيح (INTEGER)
!>         البعد الرائد للمصفوفة VT2. LDVT2 >= N.
!>

IDXC

!>          IDXC هي مصفوفة أعداد صحيحة (INTEGER)، ذات بعد (N)
!>         التبديل المستخدم لترتيب أعمدة U (وصفوف VT) إلى ثلاث مجموعات:
!>         المجموعة الأولى تحتوي على إدخالات غير صفرية فقط عند NL + 1
!>         (أو قبلها)؛ المجموعة الثانية تحتوي على إدخالات غير صفرية فقط
!>         عند NL+2 (أو بعدها)؛ والثالثة كثيفة. يعامل العمود الأول من U وصف
!>         VT بشكل منفصل.
!>
!>         يجب تبديل صفوف المتجهات المفردة التي تم العثور عليها بواسطة SLASD4
!>         بنفس الطريقة قبل إجراء ضربات المصفوفات.
!>

CTOT

!>          CTOT هي مصفوفة أعداد صحيحة (INTEGER)، ذات بعد (4)
!>         عدد إجمالي لأنواع الأعمدة المختلفة
!>         في U (أو صفوف في VT)، كما هو موضح في IDXC. العمود الرابع
!>         النوع هو أي عمود تم تخفيضه.
!>

Z

!>          Z هي مصفوفة حقيقية (REAL)، ذات بعد (K)
!>         تحتوي العناصر K الأولى من هذه المصفوفة على مكونات
!>         متجه التحديث المعدل بالتخفيض.
!>

INFO

!>          INFO هو عدد صحيح (INTEGER)
!>         = 0: الخروج بنجاح.
!>         < 0: إذا كان INFO = -i، فإن الوسيط i كان له قيمة غير قانونية.
!>         > 0: إذا كان INFO = 1، لم تتقارب قيمة مفردة.
!>
