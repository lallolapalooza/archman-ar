!>
!>    SLAROT يطبق دوران (غيفنز) على صفين أو عمودين متجاورين، حيث يكون أحد عناصر العمود/الصف الأول و/أو الأخير
!>    للاستخدام على المصفوفات المخزنة بتنسيق غير GE، بحيث يمكن استخدام العناصر الخاصة بالمصفوفة أو تعديلها
!>    التي لا توجد لها عناصر مصفوفة.
!>
!>    أحد الأمثلة هو مصفوفة متناظرة بتنسيق SB (عرض النطاق=4)،
!>    حيث UPLO='L': سيكون للصفين المتجاورين التنسيق التالي:
!>
!>    صف j:     C> C> C> C> C> .  .  .  .
!>    صف j+1:      C> C> C> C> C> .  .  .  .
!>
!>    يشير '*' إلى العناصر التي يتم توفير تخزين لها،
!>    يشير '.' إلى العناصر التي لا يتم توفير تخزين لها،
!>    ولكنها ليست بالضرورة صفرية؛ يتم تحديد قيمها بواسطة
!>    التناظر. يشير ' ' إلى العناصر التي هي بالضرورة صفرية،
!>     ولا يوجد لها تخزين موفر.
!>    تلك الأعمدة التي تحتوي على نجمتين (*) يمكن التعامل معها بواسطة SROT.
!>    يمكن تجاهل تلك الأعمدة التي لا تحتوي على نجوم (*)، لأنه طالما
!>    تم تطبيق دورانات غيفنز بعناية للحفاظ على
!>    التناظر، فإن قيمها محددة.
!>    تلك الأعمدة التي تحتوي على نجمة واحدة (*) يجب التعامل معها بشكل منفصل،
!>    باستخدام متغيرات منفصلة و :
!>
!>    صف j:     C> C> C> C> C> p  .  .  .
!>    صف j+1:   q  C> C> C> C> C> .  .  .  .
!>
!>    سيتم تعيين العنصر p بشكل صحيح، ثم سيتم تدوير هذا العمود،
!>    وتعيين p إلى قيمته الجديدة. سيقوم الاستدعاء التالي إلى
!>    SLAROT بتدوير العمودين j و j+1، باستخدام p، واستعادة
!>    التناظر. سيبدأ العنصر q بصفر، وسيصبح غير صفري
!>    بواسطة الدوران. لاحقًا، من المفترض أن تكون الدورانات
!>    مختارة لتصفير q.
!>
!>    تسلسلات الاستدعاء النموذجية: تدوير الصف i و i+1.
!>    ------- ------- ---------
!>
!>      مصفوفة كثيفة عامة:
!>
!>              CALL SLAROT(.TRUE.,.FALSE.,.FALSE., N, C,S,
!>                      A(i,1),LDA, DUMMY, DUMMY)
!>
!>      مصفوفة نطاق عامة بتنسيق GB:
!>
!>              j = MAX(1, i-KL )
!>              NL = MIN( N, i+KU+1 ) + 1-j
!>              CALL SLAROT( .TRUE., i-KL.GE.1, i+KU.LT.N, NL, C,S,
!>                      A(KU+i+1-j,j),LDA-1, XLEFT, XRIGHT )
!>
!>              [ لاحظ أن i+1-j هو ببساطة MIN(i,KL+1) ]
!>
!>      مصفوفة نطاق متناظرة بتنسيق SY، عرض النطاق K،
!>      المثلث السفلي فقط:
!>
!>              j = MAX(1, i-K )
!>              NL = MIN( K+1, i ) + 1
!>              CALL SLAROT( .TRUE., i-K.GE.1, .TRUE., NL, C,S,
!>                      A(i,j), LDA, XLEFT, XRIGHT )
!>
!>      نفس الشيء، ولكن المثلث العلوي فقط:
!>
!>              NL = MIN( K+1, N-i ) + 1
!>              CALL SLAROT( .TRUE., .TRUE., i+K.LT.N, NL, C,S,
!>                      A(i,i), LDA, XLEFT, XRIGHT )
!>
!>      مصفوفة نطاق متناظرة بتنسيق SB، عرض النطاق K،
!>      المثلث السفلي فقط:
!>
!>              [ نفس الشيء كما هو الحال مع SY، باستثناء:]
!>                  . . .
!>                      A(i+1-j,j), LDA-1, XLEFT, XRIGHT )
!>
!>              [ لاحظ أن i+1-j هو ببساطة MIN(i,K+1) ]
!>
!>      نفس الشيء، ولكن المثلث العلوي فقط:
!>                   . . .
!>                      A(K+1,i), LDA-1, XLEFT, XRIGHT )
!>
!>      تدوير الأعمدة هو مجرد تبديل لتدوير الصفوف، باستثناء
!>      لـ GB و SB: (تدوير العمودين i و i+1)
!>
!>      GB:
!>              j = MAX(1, i-KU )
!>              NL = MIN( N, i+KL+1 ) + 1-j
!>              CALL SLAROT( .TRUE., i-KU.GE.1, i+KL.LT.N, NL, C,S,
!>                      A(KU+j+1-i,i),LDA-1, XTOP, XBOTTM )
!>
!>              [ لاحظ أن KU+j+1-i هو ببساطة MAX(1,KU+2-i)]
!>
!>      SB: (المثلث العلوي)
!>
!>                   . . . . . .
!>                      A(K+j+1-i,i),LDA-1, XTOP, XBOTTM )
!>
!>      SB: (المثلث السفلي)
!>
!>                   . . . . . .
!>                      A(1,i),LDA-1, XTOP, XBOTTM )
!> 
!>  LROWS  - LOGICAL
!>           إذا كان .TRUE.، فإن SLAROT سيقوم بتدوير صفين. إذا كان .FALSE.،
!>           فسيقوم بتدوير عمودين.
!>           غير معدل.
!>
!>  LLEFT  - LOGICAL
!>           إذا كان .TRUE.، فسيتم استخدام XLEFT بدلاً من
!>           العنصر المقابل في A للعنصر الأول في
!>           الصف الثاني (إذا كان LROWS=.FALSE.) أو العمود (إذا كان LROWS=.TRUE.)
!>           إذا كان .FALSE.، فسيتم استخدام العنصر المقابل في A.
!>           غير معدل.
!>
!>  LRIGHT - LOGICAL
!>           إذا كان .TRUE.، فسيتم استخدام XRIGHT بدلاً من
!>           العنصر المقابل في A للعنصر الأخير في
!>           الصف الأول (إذا كان LROWS=.FALSE.) أو العمود (إذا كان LROWS=.TRUE.) إذا
!>           كان .FALSE.، فسيتم استخدام العنصر المقابل في A.
!>           غير معدل.
!>
!>  NL     - INTEGER
!>           طول الصفوف (إذا كان LROWS=.TRUE.) أو الأعمدة (إذا كان
!>           LROWS=.FALSE.) المراد تدويرها. إذا تم استخدام XLEFT و/أو XRIGHT،
!>           فيجب تضمين الأعمدة/الصفوف التي توجد فيها في
!>           NL، على سبيل المثال، إذا كان LLEFT = LRIGHT = .TRUE.، فيجب أن يكون NL
!>           على الأقل 2. يجب ألا يكون عدد الصفوف/الأعمدة المراد تدويرها
!>           باستثناء تلك التي تشمل XLEFT و/أو XRIGHT، سالبًا، أي أن NL مطروحًا منه عدد القيم .TRUE.
!>           في XLEFT و XRIGHT يجب أن يكون على الأقل صفرًا؛ وإلا، سيتم استدعاء XERBLA.
!>           غير معدل.
!>
!>  C, S   - REAL
!>           تحديد دوران غيفنز المراد تطبيقه. إذا كان LROWS
!>           صحيحًا، فإن المصفوفة ( c  s )
!>                                 (-s  c )  يتم تطبيقها من اليسار؛
!>           إذا كانت خاطئة، فسيتم تطبيق تبديلها من
!>           اليمين. لدوران غيفنز، يجب أن يكون C**2 + S**2 = 1،
!>           ولكن هذا لا يتم التحقق منه.
!>           غير معدل.
!>
!>  A      - REAL array.
!>           المصفوفة التي تحتوي على الصفوف/الأعمدة المراد تدويرها. يجب أن يكون العنصر الأول
!>           من A هو العنصر العلوي الأيسر المراد تدويره.
!>           مقروء ومعدل.
!>
!>  LDA    - INTEGER
!>           البعد الرائد لـ A. إذا كانت A تحتوي على
!>           مصفوفة مخزنة بتنسيق GE أو SY، فإن هذا هو ببساطة
!>           البعد الرائد لـ A كما هو محدد في الروتين المستدعي.
!>           إذا كانت A تحتوي على مصفوفة مخزنة بتنسيق نطاق (GB أو SB)،
!>           فيجب أن يكون هذا *أقل بواحد* من البعد الرائد
!>           المستخدم في الروتين المستدعي. وبالتالي، إذا
!>           كانت A مصممة A(LDA,*) في SLAROT، فإن A(1,j) ستكون
!>           العنصر j في الأول من الصفين
!>           المراد تدويرهما، و A(2,j) ستكون العنصر j في الثاني،
!>           بغض النظر عن كيفية تخزين المصفوفة في الروتين المستدعي.
!>           [ ومع ذلك، لا يمكن تصميم A فعليًا هكذا،
!>           لأنه بالنسبة لتنسيق النطاق، قد يتجاوز رقم الصف LDA، وهو
!>           غير قانوني في FORTRAN.]
!>           إذا كان LROWS=.TRUE.، فيجب أن يكون LDA على الأقل 1، وإلا
!>           فيجب أن يكون على الأقل NL مطروحًا منه عدد القيم .TRUE.
!>           في XLEFT و XRIGHT.
!>           غير معدل.
!>
!>  XLEFT  - REAL
!>           إذا كان LLEFT .TRUE.، فسيتم استخدام XLEFT وتعديله
!>           بدلاً من A(2,1) (إذا كان LROWS=.TRUE.) أو A(1,2)
!>           (إذا كان LROWS=.FALSE.).
!>           مقروء ومعدل.
!>
!>  XRIGHT - REAL
!>           إذا كان LRIGHT .TRUE.، فسيتم استخدام XRIGHT وتعديله
!>           بدلاً من A(1,NL) (إذا كان LROWS=.TRUE.) أو A(NL,1)
!>           (إذا كان LROWS=.FALSE.).
!>           مقروء ومعدل.
!>

NAME

slarot – applies a single rotation to a pair of vectors

SYNOPSIS

SUBROUTINE SLAROT(
     N,
     X,
     INCX,
     Y,
     INCY,
     C,
     S
)
INTEGER, INTENT(IN) :: N
REAL, INTENT(INOUT) :: X(:), Y(:)
INTEGER, INTENT(IN) :: INCX, INCY
REAL, INTENT(IN) :: C, S

Function/Subroutine Documentation

SLAROT applies a single rotation of the form 

   [ x ] = [ c  s ] [ x ]
   [ y ]   [-s  c ] [ y ]

where c**2 + s**2 = 1.

Arguments:

N (input)

The number of elements in the vectors X and Y to which the rotation is to be applied.

X (input/output)

The first vector.

INCX (input)

The increment between elements of X.

Y (input/output)

The second vector.

INCY (input)

The increment between elements of Y.

C (input)

The cosine of the rotation.

S (input)

The sine of the rotation.
