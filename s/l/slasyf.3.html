subroutine slasyf (uplo, n, nb, kb, a, lda, ipiv, w, ldw, info)
SLASYF تحسب عامل تجزئة جزئي لمصفوفة متماثلة حقيقية باستخدام طريقة الـ Bunch-Kaufman للـ pivoting القطري.

توثيق الوظيفة/الإجراء الفرعي

subroutine slasyf (character uplo, integer n, integer nb, integer kb, real, dimension( lda, * ) a, integer lda, integer, dimension( * ) ipiv, real, dimension( ldw, * ) w, integer ldw, integer info)

SLASYF تحسب عامل تجزئة جزئي لمصفوفة متماثلة حقيقية باستخدام طريقة الـ Bunch-Kaufman للـ pivoting القطري.

الغرض:

!>
!> SLASYF تحسب عامل تجزئة جزئي لمصفوفة متماثلة حقيقية A
!> باستخدام طريقة الـ Bunch-Kaufman للـ pivoting القطري. عامل التجزئة الجزئي
!> له الشكل:
!>
!> A  =  ( I  U12 ) ( A11  0  ) (  I       0    )  إذا كان UPLO = 'U'، أو:
!>       ( 0  U22 ) (  0   D  ) ( U12**T U22**T )
!>
!> A  =  ( L11  0 ) (  D   0  ) ( L11**T L21**T )  إذا كان UPLO = 'L'
!>       ( L21  I ) (  0  A22 ) (  0       I    )
!>
!> حيث ترتيب D هو على الأكثر NB. يتم إرجاع الترتيب الفعلي في
!> المتغير KB، وهو إما NB أو NB-1، أو N إذا كان N <= NB.
!>
!> SLASYF هي روتين مساعد يتم استدعاؤه بواسطة SSYTRF. تستخدم رموزًا مجمعة
!> (استدعاء BLAS من المستوى 3) لتحديث المصفوفة الفرعية A11 (إذا كان UPLO = 'U') أو
!> A22 (إذا كان UPLO = 'L').
!>

المتغيرات

UPLO 
!>          UPLO is CHARACTER*1
!>          يحدد ما إذا كان الجزء المثلثي العلوي أو السفلي للمصفوفة المتماثلة A
!>          مخزنًا:
!>          = 'U':  مثلثي علوي
!>          = 'L':  مثلثي سفلي
!>

N

!>          N is INTEGER
!>          ترتيب المصفوفة A. N >= 0.
!>

NB

!>          NB is INTEGER
!>          الحد الأقصى لعدد الأعمدة للمصفوفة A التي يجب
!>          تجزئتها. يجب أن يكون NB على الأقل 2 للسماح بكتل Pivot بحجم 2x2.
!>

KB

!>          KB is INTEGER
!>          عدد الأعمدة من A التي تم تجزئتها فعليًا.
!>          KB هو إما NB-1 أو NB، أو N إذا كان N <= NB.
!>

A

!>          A is REAL array, dimension (LDA,N)
!>          عند الدخول، المصفوفة المتماثلة A. إذا كان UPLO = 'U'، فإن الجزء
!>          المثلثي العلوي N x N الرائد في A يحتوي على الجزء المثلثي العلوي
!>          للمصفوفة A، والجزء المثلثي السفلي الصارم في A لا يتم الرجوع إليه.
!>          إذا كان UPLO = 'L'، فإن الجزء المثلثي السفلي N x N الرائد في A
!>          يحتوي على الجزء المثلثي السفلي للمصفوفة A، والجزء المثلثي العلوي
!>          الصارم في A لا يتم الرجوع إليه.
!>          عند الخروج، تحتوي A على تفاصيل عامل التجزئة الجزئي.
!>

LDA

!>          LDA is INTEGER
!>          البعد الرائد للمصفوفة A. LDA >= max(1,N).
!>

IPIV

!>          IPIV is INTEGER array, dimension (N)
!>          تفاصيل التبادلات وهيكل الكتلة D.
!>
!>          إذا كان UPLO = 'U':
!>             يتم تعيين العناصر KB الأخيرة فقط من IPIV.
!>
!>             إذا كان IPIV(k) > 0، فتم تبديل الصفوف والأعمدة k و IPIV(k)
!>             وكان D(k,k) كتلة قطرية بحجم 1x1.
!>
!>             إذا كان IPIV(k) = IPIV(k-1) < 0، فتم تبديل الصفوف والأعمدة
!>             k-1 و -IPIV(k) وكانت D(k-1:k,k-1:k)
!>             كتلة قطرية بحجم 2x2.
!>
!>          إذا كان UPLO = 'L':
!>             يتم تعيين العناصر KB الأولى فقط من IPIV.
!>
!>             إذا كان IPIV(k) > 0، فتم تبديل الصفوف والأعمدة k و IPIV(k)
!>             وكان D(k,k) كتلة قطرية بحجم 1x1.
!>
!>             إذا كان IPIV(k) = IPIV(k+1) < 0، فتم تبديل الصفوف والأعمدة
!>             k+1 و -IPIV(k) وكانت D(k:k+1,k:k+1)
!>             كتلة قطرية بحجم 2x2.
!>

W

!>          W is REAL array, dimension (LDW,NB)
!>

LDW

!>          LDW is INTEGER
!>          البعد الرائد للمصفوفة W. LDW >= max(1,N).
!>

INFO

!>          INFO is INTEGER
!>          = 0: خروج ناجح
!>          > 0: إذا كان INFO = k، فإن D(k,k) يساوي صفرًا تمامًا. تم
!>               إكمال عامل التجزئة، ولكن المصفوفة القطرية المجمعة D
!>               متفردة تمامًا.
!>
