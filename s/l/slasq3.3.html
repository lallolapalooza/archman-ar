subroutine slasq3 (integer i0, integer n0, real, dimension( * ) z, integer pp, real dmin, real sigma, real desig, real qmax, integer nfail, integer iter, integer ndiv, logical ieee, integer ttype, real dmin1, real dmin2, real dn, real dn1, real dn2, real g, real tau)

SLASQ3 تتحقق من الانكماش، وتحسب إزاحة وتستدعي dqds. تستخدم بواسطة sbdsqr.

الغرض:

!>
!> SLASQ3 تتحقق من الانكماش، وتحسب إزاحة (TAU) وتستدعي dqds.
!> في حالة الفشل تقوم بتغيير الإزاحات، وتحاول مرة أخرى حتى يكون الناتج
!> موجبًا.
!>

المعلمات

I0 
!>          I0 هو عدد صحيح (INTEGER)
!>         المؤشر الأول.
!>

N0

!>          N0 هو عدد صحيح (INTEGER)
!>         المؤشر الأخير.
!>

Z

!>          Z هو مصفوفة حقيقية (REAL array)، ذات بعد ( 4*N0 )
!>         Z تحمل مصفوفة qd.
!>

PP

!>          PP هو عدد صحيح (INTEGER)
!>         PP=0 من أجل ping، PP=1 من أجل pong.
!>         PP=2 يشير إلى أنه تم تطبيق التبديل على مصفوفة Z
!>         وأنه لا ينبغي إجراء الاختبارات الأولية للانكماش.
!>

DMIN

!>          DMIN هو عدد حقيقي (REAL)
!>         الحد الأدنى لقيمة d.
!>

SIGMA

!>          SIGMA هو عدد حقيقي (REAL)
!>         مجموع الإزاحات المستخدمة في المقطع الحالي.
!>

DESIG

!>          DESIG هو عدد حقيقي (REAL)
!>         الجزء السفلي الرتبة من SIGMA
!>

QMAX

!>          QMAX هو عدد حقيقي (REAL)
!>         الحد الأقصى لقيمة q.
!>

NFAIL

!>          NFAIL هو عدد صحيح (INTEGER)
!>         قم بزيادة NFAIL بمقدار 1 في كل مرة كانت فيها الإزاحة كبيرة جدًا.
!>

ITER

!>          ITER هو عدد صحيح (INTEGER)
!>         قم بزيادة ITER بمقدار 1 لكل تكرار.
!>

NDIV

!>          NDIV هو عدد صحيح (INTEGER)
!>         قم بزيادة NDIV بمقدار 1 لكل قسمة.
!>

IEEE

!>          IEEE هو قيمة منطقية (LOGICAL)
!>         علامة للعمليات الحسابية IEEE أو غير IEEE (تم تمريرها إلى SLASQ5).
!>

TTYPE

!>          TTYPE هو عدد صحيح (INTEGER)
!>         نوع الإزاحة.
!>

DMIN1

DMIN2

DN

G
