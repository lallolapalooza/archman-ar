subroutine slatrz (m, n, l, a, lda, tau, work)

SLATRZ تقوم بتحليل مصفوفة شبه منحرفة علوية بواسطة تحويلات متعامدة.

توثيق الوظيفة/الروتين الفرعي

subroutine slatrz (integer m, integer n, integer l, real, dimension( lda, * ) a, integer lda, real, dimension( * ) tau, real, dimension( * ) work)

SLATRZ تقوم بتحليل مصفوفة شبه منحرفة علوية بواسطة تحويلات متعامدة.

الغرض:

!>
!> SLATRZ تحلل المصفوفة شبه المنحرفة العلوية الحقيقية M-by-(M+L)
!> [ A1 A2 ] = [ A(1:M,1:M) A(1:M,N-L+1:N) ] كـ ( R  0 ) * Z، بواسطة
!> تحويلات متعامدة. Z هي مصفوفة متعامدة (M+L)-by-(M+L) و R و A1 هما
!> مصفوفتان مثلثيتان علويتان M-by-M.
!>

المعلمات

M 
!>          M هو عدد صحيح (INTEGER)
!>          عدد الصفوف في المصفوفة A. M >= 0.
!>

N

!>          N هو عدد صحيح (INTEGER)
!>          عدد الأعمدة في المصفوفة A. N >= 0.
!>

L

!>          L هو عدد صحيح (INTEGER)
!>          عدد الأعمدة في المصفوفة A التي تحتوي على الجزء الهادف
!>          من متجهات هاوسهولدر. N-M >= L >= 0.
!>

A

!>          A هي مصفوفة حقيقية (REAL)، ذات بعد (LDA,N)
!>          عند الدخول، الجزء العلوي شبه المنحرف M-by-N الرائد في
!>          المصفوفة A يجب أن يحتوي على المصفوفة المطلوب تحليلها.
!>          عند الخروج، الجزء العلوي المثلثي M-by-M الرائد في A
!>          يحتوي على المصفوفة المثلثية العلوية R، والعناصر N-L+1 إلى
!>          N من الصفوف M الأولى في A، مع المصفوفة TAU، تمثل
!>          المصفوفة المتعامدة Z كناتج لـ M عاكسات ابتدائية.
!>

LDA

!>          LDA هو عدد صحيح (INTEGER)
!>          البعد الرائد للمصفوفة A. LDA >= max(1,M).
!>

TAU

!>          TAU هي مصفوفة حقيقية (REAL)، ذات بعد (M)
!>          العوامل القياسية للعواكس الابتدائية.
!>

WORK

!>          WORK هي مصفوفة حقيقية (REAL)، ذات بعد (M)
!>

تفاصيل إضافية:

!>
!> يتم الحصول على التحليل بطريقة هاوسهولدر. مصفوفة التحويل
!> k-th، Z( k )، والتي تستخدم لإدخال الأصفار في الصف
!> ( m - k + 1 ) من A، تعطى بالشكل
!>
!>    Z( k ) = ( I     0   )،
!>             ( 0  T( k ) )
!>
!> حيث
!>
!>    T( k ) = I - tau*u( k )*u( k )**T،   u( k ) = (   1    )،
!>                                            (   0    )
!>                                            ( z( k ) )
!>
!> tau هو عامل قياسي و z( k ) هو متجه ذو l عنصر. tau و z( k )
!> يتم اختيارهما لإفناء عناصر الصف k من A2.
!>
!> يتم إرجاع العامل tau في العنصر k من TAU والمتجه
!> u( k ) في الصف k من A2، بحيث تكون عناصر z( k )
!> في a( k, l + 1 )، ..., a( k, n ). يتم إرجاع عناصر R في
!> الجزء المثلثي العلوي من A1.
!>
!> Z معطاة بواسطة
!>
!>    Z =  Z( 1 ) * Z( 2 ) * ... * Z( m ).
!>
