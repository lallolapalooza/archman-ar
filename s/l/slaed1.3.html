توثيق الوظيفة/الروتين الفرعي

subroutine slaed1 (integer n, real, dimension( * ) d, real, dimension( ldq, * ) q, integer ldq, integer, dimension( * ) indxq, real rho, integer cutpnt, real, dimension( * ) work, integer, dimension( * ) iwork, integer info)

SLAED1 تستخدم بواسطة SSTEDC. تحسب النظام المشترك المحدث لمصفوفة قطرية بعد تعديلها بواسطة مصفوفة متناظرة ذات رتبة واحد. تستخدم عندما تكون المصفوفة الأصلية ثلاثية الأقطار.

الغرض:

!>
!> SLAED1 تحسب النظام المشترك المحدث لمصفوفة قطرية بعد تعديلها بواسطة مصفوفة متناظرة ذات رتبة واحد.  يستخدم هذا
!> الروتين فقط لمشكلة القيمة الذاتية التي تتطلب جميع القيم الذاتية والمتجهات الذاتية
!> لمصفوفة ثلاثية الأقطار.  SLAED7 تتعامل مع
!> الحالة التي تكون فيها القيم الذاتية فقط أو القيم الذاتية والمتجهات الذاتية
!> لمصفوفة متناظرة كاملة (والتي تم اختزالها إلى شكل ثلاثي الأقطار)
!> مطلوبة.
!>
!>   T = Q(in) ( D(in) + RHO * Z*Z**T ) Q**T(in) = Q(out) * D(out) * Q**T(out)
!>
!>    حيث Z = Q**T*u، u هو متجه بطول N مع وحدات في
!>    العناصر CUTPNT و CUTPNT + 1 وأصفار في الأماكن الأخرى.
!>
!>    تخزن المتجهات الذاتية للمصفوفة الأصلية في Q، والقيم الذاتية
!>    في D. تتكون الخوارزمية من ثلاث مراحل:
!>
!>       تتكون المرحلة الأولى من تفكيك حجم المشكلة
!>       عندما تكون هناك قيم ذاتية متعددة أو إذا كان هناك صفر في
!>       المتجه Z. لكل حدوث من هذا القبيل، يتم تقليل بُعد
!>       مشكلة المعادلة العلمانية بمقدار واحد.  تتم هذه المرحلة
!>       بواسطة الروتين SLAED2.
!>
!>       تتكون المرحلة الثانية من حساب
!>       القيم الذاتية المحدثة. يتم ذلك عن طريق إيجاد جذور
!>       المعادلة العلمانية عبر الروتين SLAED4 (كما تم استدعاؤه بواسطة SLAED3).
!>       هذا الروتين يحسب أيضًا المتجهات الذاتية للمشكلة الحالية.
!>
!>       تتكون المرحلة النهائية من حساب المتجهات الذاتية
!>       المحدثة مباشرة باستخدام القيم الذاتية المحدثة.  يتم ضرب المتجهات الذاتية
!>       للمشكلة الحالية مع المتجهات الذاتية للمشكلة الشاملة.
!>

المعلمات

N 
!>          N هو عدد صحيح (INTEGER)
!>         بُعد المصفوفة المتناظرة ثلاثية الأقطار.  N >= 0.
!>

D

!>          D هو مصفوفة حقيقية (REAL)، بُعدها (N)
!>         عند الإدخال، القيم الذاتية للمصفوفة المعدلة ذات الرتبة-1.
!>         عند الخروج، القيم الذاتية للمصفوفة المُصلحة.
!>

Q

!>          Q هو مصفوفة حقيقية (REAL)، بُعدها (LDQ,N)
!>         عند الإدخال، المتجهات الذاتية للمصفوفة المعدلة ذات الرتبة-1.
!>         عند الخروج، المتجهات الذاتية للمصفوفة ثلاثية الأقطار المُصلحة.
!>

LDQ

!>          LDQ هو عدد صحيح (INTEGER)
!>         البُعد الرئيسي للمصفوفة Q.  LDQ >= max(1,N).
!>

INDXQ

!>          INDXQ هي مصفوفة أعداد صحيحة (INTEGER)، بُعدها (N)
!>         عند الإدخال، التبديل الذي يقوم بفرز المشكلتين الفرعيتين في D
!>         بشكل منفصل بترتيب تصاعدي.
!>         عند الخروج، التبديل الذي سيعيد دمج المشكلتين الفرعيتين
!>         مرة أخرى في ترتيب مُصنّف،
!>         أي أن D( INDXQ( I = 1, N ) ) ستكون بترتيب تصاعدي.
!>

RHO

!>          RHO هو عدد حقيقي (REAL)
!>         عنصر ما تحت القطر الرئيسي المستخدم لإنشاء التعديل ذي الرتبة-1.
!>

CUTPNT

!>          CUTPNT هو عدد صحيح (INTEGER)
!>         موقع آخر قيمة ذاتية في المصفوفة الفرعية الرائدة.
!>         min(1,N) <= CUTPNT <= N/2.
!>

WORK

!>          WORK هي مصفوفة حقيقية (REAL)، بُعدها (4*N + N**2)
!>

IWORK

!>          IWORK هي مصفوفة أعداد صحيحة (INTEGER)، بُعدها (4*N)
!>

INFO

!>          INFO هو عدد صحيح (INTEGER)
!>          = 0:  الخروج ناجح.
!>          < 0:  إذا كان INFO = -i، فإن الوسيط i كان له قيمة غير قانونية.
!>          > 0:  إذا كان INFO = 1، لم تتقارب قيمة ذاتية
!>
