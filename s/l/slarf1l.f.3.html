subroutine slarf1l (character side, integer m, integer n, real, dimension( * ) v, integer incv, real tau, real, dimension( ldc, * ) c, integer ldc, real, dimension( * ) work)

SLARF1L يطبق انعكاسًا أوليًا حقيقيًا H على مصفوفة حقيقية m × n C، من اليسار أو اليمين. H ممثلة في الصورة

الغرض:

!>
!> SLARF1L يطبق انعكاسًا أوليًا حقيقيًا H على مصفوفة حقيقية C
!> بأبعاد m × n، من اليسار أو اليمين. H ممثلة في الصورة
!>
!>       H = I - tau * v * v**T
!>
!> حيث tau هو عدد قياسي حقيقي و v هو متجه حقيقي يفترض أن v(lastv) = 1،
!> حيث lastv هو آخر عنصر غير صفري.
!>
!> إذا كان tau = 0، فإن H تعتبر مصفوفة الوحدة.
!>

المعلمات

SIDE 
!>          SIDE هو CHARACTER*1
!>          = 'L': تشكل  H * C
!>          = 'R': تشكل  C * H
!>

M

!>          M هو INTEGER
!>          عدد صفوف المصفوفة C.
!>

N

!>          N هو INTEGER
!>          عدد أعمدة المصفوفة C.
!>

V

!>          V هو مصفوفة REAL، البعد
!>                     (1 + (M-1)*abs(INCV)) إذا كانت SIDE = 'L'
!>                  أو (1 + (N-1)*abs(INCV)) إذا كانت SIDE = 'R'
!>          المتجه v في تمثيل H. لا يتم استخدام V إذا كانت
!>          TAU = 0.
!>

INCV

!>          INCV هو INTEGER
!>          الزيادة بين عناصر v. INCV > 0.
!>

TAU

!>          TAU هو REAL
!>          القيمة tau في تمثيل H.
!>

C

!>          C هي مصفوفة REAL، البعد (LDC,N)
!>          عند الإدخال، المصفوفة C بأبعاد m × n.
!>          عند الخروج، يتم استبدال C بالمصفوفة H * C إذا كانت SIDE = 'L'،
!>          أو C * H إذا كانت SIDE = 'R'.
!>

LDC

!>          LDC هو INTEGER
!>          البعد الرائد للمصفوفة C. LDC >= max(1,M).
!>

WORK

!>          WORK هي مصفوفة REAL، البعد
!>                         (N) إذا كانت SIDE = 'L'
!>                      أو (M) إذا كانت SIDE = 'R'
!>
