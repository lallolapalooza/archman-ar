subroutine slauu2 (uplo, n, a, lda, info)
SLAUU2 تحسب حاصل الضرب UUH أو LHL، حيث U و L هما مصفوفتان مثلثيتان علويتان أو سفليتان (خوارزمية غير مجمعة).

توثيق الوظيفة/الإجراء الفرعي

subroutine slauu2 (character uplo, integer n, real, dimension( lda, * ) a, integer lda, integer info)

SLAUU2 تحسب حاصل الضرب UUH أو LHL، حيث U و L هما مصفوفتان مثلثيتان علويتان أو سفليتان (خوارزمية غير مجمعة).

الغرض:

!>
!> SLAUU2 تحسب حاصل الضرب U * U**T أو L**T * L، حيث يتم تخزين العامل المثلثي U أو L
!> في الجزء العلوي أو السفلي من المصفوفة A.
!>
!> إذا كان UPLO = 'U' أو 'u' فسيتم تخزين المثلث العلوي للنتيجة،
!> مما يؤدي إلى الكتابة فوق العامل U في A.
!> إذا كان UPLO = 'L' أو 'l' فسيتم تخزين المثلث السفلي للنتيجة،
!> مما يؤدي إلى الكتابة فوق العامل L في A.
!>
!> هذا هو الشكل غير المجمع للخوارزمية، والذي يستدعي BLAS من المستوى 2.
!>

المعلمات

UPLO 
!>          UPLO هو CHARACTER*1
!>          يحدد ما إذا كان العامل المثلثي المخزن في المصفوفة A
!>          هو مثلث علوي أم سفلي:
!>          = 'U':  مثلث علوي
!>          = 'L':  مثلث سفلي
!>

N

!>          N هو INTEGER
!>          ترتيب العامل المثلثي U أو L.  N >= 0.
!>

A

!>          A هي مصفوفة REAL، ذات أبعاد (LDA,N)
!>          عند الإدخال، العامل المثلثي U أو L.
!>          عند الخروج، إذا كان UPLO = 'U'، يتم الكتابة فوق الجزء العلوي من A
!>          بالجزء العلوي من حاصل الضرب U * U**T؛
!>          إذا كان UPLO = 'L'، يتم الكتابة فوق الجزء السفلي من A
!>          بالجزء السفلي من حاصل الضرب L**T * L.
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرائد للمصفوفة A.  LDA >= max(1,N).
!>

INFO

!>          INFO هو INTEGER
!>          = 0: خروج ناجح
!>          < 0: إذا كان INFO = -k، فإن المعلمة k كانت لها قيمة غير قانونية
!>
