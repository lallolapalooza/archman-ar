subroutine slasd0 (n, sqre, d, e, u, ldu, vt, ldvt, smlsiz, iwork, work, info)
SLASD0 تحسب القيم المفردة لمصفوفة B حقيقية علوية ثنائية القطر n-by-m مع قطر رئيسي d وقطر فرعي e. تستخدم بواسطة sbdsdc.

توثيق الوظيفة/الإجراء الفرعي

subroutine slasd0 (integer n, integer sqre, real, dimension( * ) d, real, dimension( * ) e, real, dimension( ldu, * ) u, integer ldu, real, dimension( ldvt, * ) vt, integer ldvt, integer smlsiz, integer, dimension( * ) iwork, real, dimension( * ) work, integer info)

SLASD0 تحسب القيم المفردة لمصفوفة B حقيقية علوية ثنائية القطر n-by-m مع قطر رئيسي d وقطر فرعي e. تستخدم بواسطة sbdsdc.

الغرض:

!>
!> باستخدام نهج تقسيم الغزو، تقوم SLASD0 بحساب
!> تحلل القيم المفردة (SVD) لمصفوفة B ثنائية القطر علوية حقيقية N-by-M
!> مع قطر رئيسي D وقطر فرعي E، حيث M = N + SQRE.
!> تحسب الخوارزمية مصفوفات متعامدة U و VT بحيث
!> B = U * S * VT. يتم استبدال القيم المفردة S في D.
!>
!> إجراء فرعي ذو صلة، SLASDA، يحسب فقط القيم المفردة،
!> واختياريًا، المتجهات المفردة في شكل مضغوط.
!>

المعاملات

N 
!>          N هو INTEGER
!>         عند الدخول، يمثل البعد الصفّي للمصفوفة ثنائية القطر العلوية.
!>         هذا هو أيضًا بعد مصفوفة القطر الرئيسي D.
!>

SQRE

!>          SQRE هو INTEGER
!>         يحدد البعد العمودي للمصفوفة ثنائية القطر.
!>         = 0: المصفوفة ثنائية القطر لها بعد عمودي M = N؛
!>         = 1: المصفوفة ثنائية القطر لها بعد عمودي M = N+1؛
!>

D

!>          D هي مصفوفة REAL، حجمها (N)
!>         عند الدخول تحتوي D على القطر الرئيسي للمصفوفة ثنائية القطر.
!>         عند الخروج D، إذا كان INFO = 0، تحتوي على قيمها المفردة.
!>

E

!>          E هي مصفوفة REAL، حجمها (M-1)
!>         تحتوي على مدخلات القطر الفرعي للمصفوفة ثنائية القطر.
!>         عند الخروج، تم تدمير E.
!>

U

!>          U هي مصفوفة REAL، حجمها (LDU, N)
!>         عند الخروج، تحتوي U على المتجهات المفردة اليسرى،
!>          إذا تم تمرير U كمصفوفة هوية (N, N).
!>

LDU

!>          LDU هو INTEGER
!>         عند الدخول، البعد الرائد لـ U.
!>

VT

!>          VT هي مصفوفة REAL، حجمها (LDVT, M)
!>         عند الخروج، تحتوي VT**T على المتجهات المفردة اليمنى،
!>          إذا تم تمرير VT كمصفوفة هوية (M, M).
!>

LDVT

!>          LDVT هو INTEGER
!>         عند الدخول، البعد الرائد لـ VT.
!>

SMLSIZ

!>          SMLSIZ هو INTEGER
!>         عند الدخول، الحد الأقصى لحجم المشاكل الفرعية في
!>         أسفل شجرة الحساب.
!>

IWORK

!>          IWORK هي مصفوفة INTEGER، حجمها (8*N)
!>

WORK

!>          WORK هي مصفوفة REAL، حجمها (3*M**2+2*M)
!>

INFO

!>          INFO هو INTEGER
!>          = 0:  انتهاء ناجح.
!>          < 0:  إذا كان INFO = -i، فإن المعامل i كان له قيمة غير قانونية.
!>          > 0:  إذا كان INFO = 1، لم تتقارب قيمة مفردة.
!>
