subroutine slapll (integer n, real, dimension( * ) x, integer incx, real, dimension( * ) y, integer incy, real ssmin)

SLAPLL يقيس التبعية الخطية لمتجهين.

الغرض:

!>
!> بالنظر إلى المتجهين العموديين X و Y ، فإن
!>
!>                      A = ( X Y ).
!>
!> تقوم الروتين الفرعي أولاً بحساب تحليل QR للمصفوفة A = Q*R،
!> ثم تقوم بحساب SVD للمصفوفة المثلثية العليا 2x2 R.
!> يتم إرجاع أصغر قيمة مفردة لـ R في SSMIN، والتي تستخدم
!> كقياس للتبعية الخطية للمتجهين X و Y.
!>

المعلمات

N 
!>          N هو عدد صحيح (INTEGER)
!>          طول المتجهين X و Y.
!>

X

!>          X هو مصفوفة حقيقية (REAL array)،
!>                         dimension (1+(N-1)*INCX)
!>          عند الدخول، تحتوي X على المتجه N X.
!>          عند الخروج، يتم الكتابة فوق X.
!>

INCX

!>          INCX هو عدد صحيح (INTEGER)
!>          الزيادة بين العناصر المتتالية لـ X. INCX > 0.
!>

Y

!>          Y هي مصفوفة حقيقية (REAL array)،
!>                         dimension (1+(N-1)*INCY)
!>          عند الدخول، تحتوي Y على المتجه N Y.
!>          عند الخروج، يتم الكتابة فوق Y.
!>

INCY

!>          INCY هو عدد صحيح (INTEGER)
!>          الزيادة بين العناصر المتتالية لـ Y. INCY > 0.
!>

SSMIN

!>          SSMIN هي قيمة حقيقية (REAL)
!>          أصغر قيمة مفردة للمصفوفة N x 2 A = ( X Y ).
!>
