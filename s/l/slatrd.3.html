subroutine slatrd (uplo, n, nb, a, lda, e, tau, w, ldw)
SLATRD تقلل أول nb صفًا وأعمدة من مصفوفة متماثلة/هيرميتية A إلى شكل ثلاثي الأقطار حقيقي عبر تحويل تشابه متعامد.

توثيق الوظيفة/الروتين الفرعي

subroutine slatrd (character uplo, integer n, integer nb, real, dimension( lda, * ) a, integer lda, real, dimension( * ) e, real, dimension( * ) tau, real, dimension( ldw, * ) w, integer ldw)

SLATRD تقلل أول nb صفًا وأعمدة من مصفوفة متماثلة/هيرميتية A إلى شكل ثلاثي الأقطار حقيقي عبر تحويل تشابه متعامد.

الغرض:

!>
!> SLATRD تقلل NB صفًا وأعمدة من مصفوفة متماثلة حقيقية A إلى
!> شكل ثلاثي الأقطار متماثل عبر تحويل تشابه متعامد Q**T * A * Q، وتعيد
!> المصفوفتين V و W اللازمتين لتطبيق التحويل على الجزء غير المنفصل من A.
!>
!> إذا كان UPLO = 'U'، تقوم SLATRD بتقليل آخر NB صفًا وأعمدة من مصفوفة،
!> يتم توفير المثلث العلوي لها؛
!> إذا كان UPLO = 'L'، تقوم SLATRD بتقليل أول NB صفًا وأعمدة من مصفوفة،
!> يتم توفير المثلث السفلي لها.
!>
!> هذه روتينية مساعدة يتم استدعاؤها بواسطة SSYTRD.
!>

المعلمات

UPLO 
!>          UPLO هو CHARACTER*1
!>          يحدد ما إذا كان الجزء المثلثي العلوي أو السفلي من المصفوفة
!>          المتماثلة A مخزنًا:
!>          = 'U': مثلث علوي
!>          = 'L': مثلث سفلي
!>

N

!>          N هو INTEGER
!>          رتبة المصفوفة A.
!>

NB

!>          NB هو INTEGER
!>          عدد الصفوف والأعمدة المراد تقليلها.
!>

A

!>          A هو مصفوفة REAL، ذات أبعاد (LDA,N)
!>          عند الإدخال، المصفوفة المتماثلة A. إذا كان UPLO = 'U'، فإن الجزء
!>          المثلثي العلوي من n-by-n في A يحتوي على الجزء المثلثي العلوي
!>          من المصفوفة A، والجزء المثلثي السفلي الصارم من A لا يتم الرجوع إليه.
!>          إذا كان UPLO = 'L'، فإن الجزء المثلثي السفلي من n-by-n في A يحتوي
!>          على الجزء المثلثي السفلي من المصفوفة A، والجزء المثلثي العلوي الصارم
!>          من A لا يتم الرجوع إليه.
!>          عند الخروج:
!>          إذا كان UPLO = 'U'، تم تقليل آخر NB أعمدة إلى
!>            شكل ثلاثي الأقطار، مع تجاوز العناصر القطرية
!>            للعناصر القطرية لـ A؛ العناصر فوق القطر،
!>            مع المصفوفة TAU، تمثل المصفوفة المتعامدة Q كـ
!>            ناتج لعاكسات أساسية؛
!>          إذا كان UPLO = 'L'، تم تقليل أول NB أعمدة إلى
!>            شكل ثلاثي الأقطار، مع تجاوز العناصر القطرية
!>            للعناصر القطرية لـ A؛ العناصر تحت القطر،
!>            مع المصفوفة TAU، تمثل المصفوفة المتعامدة Q كـ
!>            ناتج لعاكسات أساسية.
!>          انظر المزيد من التفاصيل.
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرائد للمصفوفة A. LDA >= (1,N).
!>

E

!>          E هو مصفوفة REAL، ذات أبعاد (N-1)
!>          إذا كان UPLO = 'U'، فإن E(n-nb:n-1) تحتوي على عناصر فوق القطر
!>          العلوي لآخر NB أعمدة للمصفوفة المخفضة؛
!>          إذا كان UPLO = 'L'، فإن E(1:nb) تحتوي على عناصر تحت القطر
!>          لأول NB أعمدة للمصفوفة المخفضة.
!>

TAU

!>          TAU هو مصفوفة REAL، ذات أبعاد (N-1)
!>          عوامل القياس للعاكسات الأساسية، مخزنة في
!>          TAU(n-nb:n-1) إذا كان UPLO = 'U'، وفي TAU(1:nb) إذا كان UPLO = 'L'.
!>          انظر المزيد من التفاصيل.
!>

W

!>          W هو مصفوفة REAL، ذات أبعاد (LDW,NB)
!>          المصفوفة n-by-nb W المطلوبة لتحديث الجزء غير المنفصل
!>          من A.
!>

LDW

!>          LDW هو INTEGER
!>          البعد الرائد للمصفوفة W. LDW >= max(1,N).
!>

المزيد من التفاصيل:

!>
!>  إذا كان UPLO = 'U'، يتم تمثيل المصفوفة Q كناتج لـ
!>  عاكسات أساسية
!>
!>     Q = H(n) H(n-1) . . . H(n-nb+1).
!>
!>  كل H(i) لها الشكل
!>
!>     H(i) = I - tau * v * v**T
!>
!>  حيث tau هو عدد قياسي حقيقي، و v هو متجه حقيقي مع
!>  v(i:n) = 0 و v(i-1) = 1؛ v(1:i-1) مخزن عند الخروج في A(1:i-1,i)،
!>  و tau في TAU(i-1).
!>
!>  إذا كان UPLO = 'L'، يتم تمثيل المصفوفة Q كناتج لـ
!>  عاكسات أساسية
!>
!>     Q = H(1) H(2) . . . H(nb).
!>
!>  كل H(i) لها الشكل
!>
!>     H(i) = I - tau * v * v**T
!>
!>  حيث tau هو عدد قياسي حقيقي، و v هو متجه حقيقي مع
!>  v(1:i) = 0 و v(i+1) = 1؛ v(i+1:n) مخزن عند الخروج في A(i+1:n,i)،
!>  و tau في TAU(i).
!>
!>  عناصر المتجهات v تشكل معًا المصفوفة V بحجم n-by-nb
!>  المطلوبة، مع W، لتطبيق التحويل على الجزء غير المنفصل
!>  من المصفوفة، باستخدام تحديث متماثل من الرتبة 2k بالشكل:
!>  A := A - V*W**T - W*V**T.
!>
!>  محتويات A عند الخروج موضحة بالأمثلة التالية
!>  مع n = 5 و nb = 2:
!>
!>  إذا كان UPLO = 'U':                       إذا كان UPLO = 'L':
!>
!>    (  a   a   a   v4  v5 )              (  d                  )
!>    (      a   a   v4  v5 )              (  1   d              )
!>    (          a   1   v5 )              (  v1  1   a          )
!>    (              d   1  )              (  v1  v2  a   a      )
!>    (                  d  )              (  v1  v2  a   a   a  )
!>
!>  حيث d يمثل عنصرًا قطريًا للمصفوفة المخفضة، a يمثل
!>  عنصرًا من المصفوفة الأصلية لم يتغير، و vi يمثل
!>  عنصرًا من المتجه الذي يحدد H(i).
!>
