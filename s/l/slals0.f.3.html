subroutine slals0 (integer icompq, integer nl, integer nr, integer sqre, integer nrhs, real, dimension( ldb, * ) b, integer ldb, real, dimension( ldbx, * ) bx, integer ldbx, integer, dimension( * ) perm, integer givptr, integer, dimension( ldgcol, * ) givcol, integer ldgcol, real, dimension( ldgnum, * ) givnum, integer ldgnum, real, dimension( ldgnum, * ) poles, real, dimension( * ) difl, real, dimension( ldgnum, * ) difr, real, dimension( * ) z, integer k, real c, real s, real, dimension( * ) work, integer info)

SLALS0 تطبق عوامل الضرب العكسية في حل مشكلة المربعات الصغرى باستخدام نهج SVD القائم على التقسيم والغزو. تستخدم بواسطة sgelsd.

الغرض:

!>
!> SLALS0 تطبق عوامل الضرب العكسية إما لمصفوفة المتجهات المنفردة اليسرى أو اليمنى لمصفوفة قطرية مضاف إليها صف إلى مصفوفة الطرف الأيمن B في حل مشكلة المربعات الصغرى باستخدام نهج SVD القائم على التقسيم والغزو.
!>
!> بالنسبة لمصفوفة المتجهات المنفردة اليسرى، تتضمن ثلاثة أنواع من المصفوفات المتعامدة:
!>
!> (1L) دورانات جيفنز: عدد هذه الدورانات هو GIVPTR؛ أزواج الأعمدة/الصفوف التي تم تطبيقها عليها مخزنة في GIVCOL؛ وقيم C و S لهذه الدورانات مخزنة في GIVNUM.
!>
!> (2L) التبديل. الصف (NL+1) من B سيتم نقله إلى الصف الأول، ولـ J=2:N، سيتم نقل الصف PERM(J) من B إلى الصف J.
!>
!> (3L) مصفوفة المتجهات المنفردة اليسرى للمصفوفة المتبقية.
!>
!> بالنسبة لمصفوفة المتجهات المنفردة اليمنى، تتضمن أربعة أنواع من المصفوفات المتعامدة:
!>
!> (1R) مصفوفة المتجهات المنفردة اليمنى للمصفوفة المتبقية.
!>
!> (2R) إذا كان SQRE = 1، دورنة جيفنز إضافية لتوليد الفضاء الصفري الأيمن.
!>
!> (3R) التحويل العكسي لـ (2L).
!>
!> (4R) التحويل العكسي لـ (1L).
!>

المعلمات

ICOMPQ 
!>          ICOMPQ هو INTEGER
!>         يحدد ما إذا كان سيتم حساب المتجهات المنفردة في شكل مُقسم:
!>         = 0: مصفوفة المتجهات المنفردة اليسرى.
!>         = 1: مصفوفة المتجهات المنفردة اليمنى.
!>

NL

!>          NL هو INTEGER
!>         البعد الصفي للكتلة العلوية. NL >= 1.
!>

NR

!>          NR هو INTEGER
!>         البعد الصفي للكتلة السفلية. NR >= 1.
!>

SQRE

!>          SQRE هو INTEGER
!>         = 0: الكتلة السفلية هي مصفوفة مربعة بحجم NR-by-NR.
!>         = 1: الكتلة السفلية هي مصفوفة مستطيلة بحجم NR-by-(NR+1).
!>
!>         المصفوفة ثنائية القطر لها البعد الصفي N = NL + NR + 1،
!>         والبعد العمودي M = N + SQRE.
!>

NRHS

!>          NRHS هو INTEGER
!>         عدد أعمدة B و BX. يجب أن يكون NRHS على الأقل 1.
!>

B

!>          B هو مصفوفة REAL، البعد ( LDB, NRHS )
!>         عند الإدخال، تحتوي B على الأطراف اليمنى لمشكلة المربعات الصغرى في الصفوف من 1 إلى M. عند الإخراج، تحتوي B على الحل X في الصفوف من 1 إلى N.
!>

LDB

!>          LDB هو INTEGER
!>         البعد الرائد لـ B. يجب أن يكون LDB على الأقل
!>         max(1,MAX( M, N ) ).
!>

BX

!>          BX هو مصفوفة REAL، البعد ( LDBX, NRHS )
!>

LDBX

!>          LDBX هو INTEGER
!>         البعد الرائد لـ BX.
!>

PERM

!>          PERM هي مصفوفة INTEGER، البعد ( N )
!>         التبديلات (من التقويض والفرز) المطبقة
!>         على الكتل.
!>

GIVPTR

!>          GIVPTR هو INTEGER
!>         عدد دورانات جيفنز التي حدثت في هذه المشكلة الفرعية.
!>

GIVCOL

!>          GIVCOL هي مصفوفة INTEGER، البعد ( LDGCOL, 2 )
!>         كل زوج من الأرقام يشير إلى زوج من الصفوف/الأعمدة
!>         المشاركة في دورنة جيفنز.
!>

LDGCOL

!>          LDGCOL هو INTEGER
!>         البعد الرائد لـ GIVCOL، يجب أن يكون على الأقل N.
!>

GIVNUM

!>          GIVNUM هي مصفوفة REAL، البعد ( LDGNUM, 2 )
!>         كل رقم يشير إلى قيمة C أو S المستخدمة في
!>         دورة جيفنز المقابلة.
!>

LDGNUM

!>          LDGNUM هو INTEGER
!>         البعد الرائد للمصفوفات DIFR، POLES و
!>         GIVNUM، يجب أن يكون على الأقل K.
!>

POLES

!>          POLES هي مصفوفة REAL، البعد ( LDGNUM, 2 )
!>         عند الإدخال، تحتوي POLES(1:K, 1) على القيم المنفردة الجديدة
!>         التي تم الحصول عليها من حل المعادلة العلمانية، و
!>         POLES(1:K, 2) هي مصفوفة تحتوي على الأقطاب في
!>         المعادلة العلمانية.
!>

DIFL

!>          DIFL هي مصفوفة REAL، البعد ( K ).
!>         عند الإدخال، DIFL(I) هي المسافة بين القيمة المنفردة المحدثة (غير المفككة) بالترتيب I
!>         والقيمة المنفردة القديمة (غير المفككة) بالترتيب I.
!>

DIFR

!>          DIFR هي مصفوفة REAL، البعد ( LDGNUM, 2 ).
!>         عند الإدخال، تحتوي DIFR(I, 1) على المسافات بين القيمة المنفردة المحدثة (غير المفككة) بالترتيب I والقيمة المنفردة القديمة (غير المفككة) بالترتيب I+1. و DIFR(I, 2) هي
!>         عامل التطبيع للمتجه المنفرد الأيمن بالترتيب I.
!>

Z

!>          Z هي مصفوفة REAL، البعد ( K )
!>         تحتوي على مكونات متجه الصف المُعدل بفعل التقويض.
!>

K

!>          K هو INTEGER
!>         يحتوي على بعد المصفوفة غير المفككة،

!>

C

!>         تحتوي C على بيانات عشوائية إذا كان SQRE = 0 وقيمة C لدورنة جيفنز
