وثائق الوظيفة/الإجراء الفرعي

subroutine slaqr0 (logical wantt, logical wantz, integer n, integer ilo, integer ihi, real, dimension( ldh, * ) h, integer ldh, real, dimension( * ) wr, real, dimension( * ) wi, integer iloz, integer ihiz, real, dimension( ldz, * ) z, integer ldz, real, dimension( * ) work, integer lwork, integer info)

SLAQR0 تحسب القيم الذاتية لمصفوفة هسنبرغ، واختيارياً، مصفوفات التحليل الشوري.

الغرض:

!>
!>    SLAQR0 تحسب القيم الذاتية لمصفوفة هسنبرغ H
!>    واختيارياً، المصفوفتين T و Z من التحليل الشوري
!>    H = Z T Z**T، حيث T مصفوفة شبه قطرية عليا (الشكل الشوري)،
!>    و Z هي مصفوفة المتجهات الشورية المتعامدة.
!>
!>    اختيارياً يمكن ضرب Z مسبقاً في مصفوفة متعامدة دخل Q
!>    بحيث يمكن لهذا الإجراء الفرعي إعطاء التحليل الشوري
!>    لمصفوفة A التي تم اختزالها إلى شكل هسنبرغ H
!>    بواسطة المصفوفة المتعامدة Q:  A = Q*H*Q**T = (QZ)*T*(QZ)**T.
!>

المعلمات

WANTT 
!>          WANTT هو LOGICAL
!>          = .TRUE. : مطلوب الشكل الشوري الكامل T؛
!>          = .FALSE.: القيم الذاتية فقط مطلوبة.
!>

WANTZ

!>          WANTZ هو LOGICAL
!>          = .TRUE. : مطلوب مصفوفة المتجهات الشورية Z؛
!>          = .FALSE.: المتجهات الشورية غير مطلوبة.
!>

N

!>          N هو INTEGER
!>           رتبة المصفوفة H.  N >= 0.
!>

ILO

!>          ILO هو INTEGER
!>

IHI

!>          IHI هو INTEGER
!>           يفترض أن H هي بالفعل مصفوفة قطرية عليا في الصفوف
!>           والأعمدة 1:ILO-1 و IHI+1:N، وإذا كان ILO > 1،
!>           فإن H(ILO,ILO-1) يساوي صفرًا. عادةً ما يتم تعيين ILO و IHI بواسطة
!>           استدعاء سابق لـ SGEBAL، ثم يتم تمريرهما إلى SGEHRD عند
!>           اختزال ناتج المصفوفة من SGEBAL إلى شكل هسنبرغ.
!>           بخلاف ذلك، يجب تعيين ILO و IHI إلى 1 و N،
!>           على التوالي.  إذا كان N > 0، فإن 1 <= ILO <= IHI <= N.
!>           إذا كان N = 0، فإن ILO = 1 و IHI = 0.
!>

H

!>          H هي مصفوفة REAL، البعد (LDH,N)
!>           عند الإدخال، مصفوفة هسنبرغ العليا H.
!>           عند الخروج، إذا كان INFO = 0 وكان WANTT هو .TRUE.، فإن H تحتوي
!>           على المصفوفة شبه القطرية العليا T من التحليل الشوري
!>           (الشكل الشوري)؛ يتم إرجاع الكتل القطرية 2x2
!>           (المقابلة لأزواج القيم الذاتية المترافقة المعقدة)
!>           في الشكل القياسي، مع H(i,i) = H(i+1,i+1)
!>           و H(i+1,i)*H(i,i+1) < 0. إذا كان INFO = 0 وكان WANTT هو
!>           .FALSE.، فإن محتويات H غير محددة عند الخروج.
!>           (قيمة H عند الخروج عندما يكون INFO > 0 معطاة تحت
!>           وصف INFO أدناه.)
!>
!>           قد تقوم هذه الوحدة الفرعية بتعيين H(i,j) = 0 صراحةً لـ i > j و
!>           j = 1، 2، ... ILO-1 أو j = IHI+1، IHI+2، ... N.
!>

LDH

!>          LDH هو INTEGER
!>           البعد الرائد للمصفوفة H. LDH >= max(1,N).
!>

WR

!>          WR هي مصفوفة REAL، البعد (IHI)
!>

WI

!>          WI هي مصفوفة REAL، البعد (IHI)
!>           يتم تخزين الأجزاء الحقيقية والمخيلية، على التوالي، للقيم الذاتية
!>           المحسوبة للمصفوفة H(ILO:IHI,ILO:IHI) في WR(ILO:IHI)
!>           و WI(ILO:IHI). إذا تم حساب قيمتين ذاتيتين كزوج
!>           مترافق معقد، فسيتم تخزينهما في عناصر متتالية
!>           في WR و WI، على سبيل المثال العنصر i و (i+1)، مع
!>           WI(i) > 0 و WI(i+1) < 0. إذا كان WANTT هو .TRUE.، فإن
!>           القيم الذاتية مخزنة بنفس ترتيب القطر الرئيسي
!>           للشكل الشوري الذي تم إرجاعه في H، مع
!>           WR(i) = H(i,i) و، إذا كانت H(i:i+1,i:i+1) كتلة قطرية 2x2،
!>           فإن WI(i) = sqrt(-H(i+1,i)*H(i,i+1)) و
!>           WI(i+1) = -WI(i).
!>

ILOZ

!>          ILOZ هو INTEGER
!>

IHIZ

!>          IHIZ هو INTEGER
!>           تحديد صفوف Z التي يجب تطبيق التحويلات عليها إذا كان WANTZ هو .TRUE.
!>           1 <= ILOZ <= ILO؛ IHI <= IHIZ <= N.
!>

Z

!>          Z هي مصفوفة REAL، البعد (LDZ,IHI)
!>           إذا كان WANTZ هو .FALSE.، فلن يتم الرجوع إلى Z.
!>           إذا كان WANTZ هو .TRUE.، فسيتم استبدال Z(ILO:IHI,ILOZ:IHIZ)
!>           بـ Z(ILO:IHI,ILOZ:IHIZ)*U حيث U هي
!>           عامل الشوري المتعامد لـ H(ILO:IHI,ILO:IHI).
!>           (قيمة Z عند الخروج عندما يكون INFO > 0 معطاة تحت
!>           وصف INFO أدناه.)
!>

LDZ

!>          LDZ هو INTEGER
!>           البعد الرائد للمصفوفة Z.  إذا كان WANTZ هو .TRUE.
!>           فإن LDZ >= MAX(1,IHIZ).  بخلاف ذلك، LDZ >= 1.
!>

WORK

!>          WORK هي مصفوفة REAL، البعد LWORK
!>           عند الخروج، إذا كان LWORK = -1، فإن WORK(1) يرجع تقديرًا لـ
!>           القيمة المثلى لـ LWORK.
!>

LWORK

!>          LWORK هو INTEGER
!>           بعد المصفوفة WORK.  LWORK >= max(1,N)
!>           كافٍ، ولكن LWORK عادةً ما يصل إلى 6*N قد
!>           يكون مطلوبًا للحصول على الأداء الأمثل.
!>           يُوصى بالاستعلام عن مساحة العمل لتحديد حجم مساحة العمل المثلى.
!>
!>           إذا كان LWORK = -1، فإن SLAQR0 يقوم باستعلام مساحة العمل.
!>           في هذه الحالة، تتحقق SLAQR0 من معلمات الإدخال وتقدر
!>           حجم مساحة العمل الأمثل للقيم المعطاة
!>           لـ N، ILO، و IHI.  يتم إرجاع التقدير
!>           في WORK(1).  لا يتم إصدار رسالة خطأ تتعلق بـ LWORK
!>           بواسطة XERBLA.  لا يتم الوصول إلى H ولا Z.
!>

INFO

!>          INFO هو INTEGER
!>             = 0:  خروج ناجح
!>             > 0:  إذا كان INFO = i، فشلت SLAQR0 في حساب جميع
!>                القيم الذاتية.  العناصر 1:ilo-1 و i+1:n من WR
!>                و WI تحتوي على تلك القيم الذاتية التي تم
!>                حسابها بنجاح.  (الفشل نادر.)
!>
!>                إذا كان INFO > 0 ولم يكن WANT .FALSE.، فعند الخروج،
!>                القيم الذاتية غير المتقاربة المتبقية هي القيم الذاتية
!>                لمصفوفة هسنبرغ العليا الصفوف والأعمدة ILO إلى
!>                INFO من القيمة النهائية، الناتجة
!>                لـ H.
!>
!>                إذا كان INFO > 0 وكان WANTT هو .TRUE.، فعند الخروج
!>
!>           (*)  (القيمة الأولية لـ H)*U  = U*(القيمة النهائية لـ H)
!>
!>                حيث U هي مصفوفة متعامدة.  القيمة النهائية
!>                لـ H هي عليا هسنبرغ وشبه قطرية
!>                في الصفوف والأعمدة INFO+1 إلى IHI.
!>
!>                إذا كان INFO > 0 وكان WANTZ هو .TRUE.، فعند الخروج
!>
!>                  (القيمة النهائية لـ Z(ILO:IHI,ILOZ:IHIZ)
!>                   =  (القيمة الأولية لـ Z(ILO:IHI,ILOZ:IHIZ)*U
!>
!>                حيث U هي المصفوفة المتعامدة في (*) (بغض النظر
!>                عن قيمة WANTT.)
!>
!>                إذا كان INFO > 0 وكان WANTZ هو .FALSE.، فلن يتم الوصول إلى Z.
!>
