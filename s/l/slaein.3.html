SLAEIN تحسب متجهة ذاتية يمينية أو يسارية محددة لمصفوفة هسنبرغ علوية باستخدام التكرار العكسي.

توثيق الوظيفة/الإجراء الفرعي

subroutine slaein (logical rightv, logical noinit, integer n, real, dimension( ldh, * ) h, integer ldh, real wr, real wi, real, dimension( * ) vr, real, dimension( * ) vi, real, dimension( ldb, * ) b, integer ldb, real, dimension( * ) work, real eps3, real smlnum, real bignum, integer info)

SLAEIN تحسب متجهة ذاتية يمينية أو يسارية محددة لمصفوفة هسنبرغ علوية باستخدام التكرار العكسي.

الغرض:

!>
!> SLAEIN يستخدم التكرار العكسي لإيجاد متجهة ذاتية يمينية أو يسارية
!> مقابلة للقيمة الذاتية (WR,WI) لمصفوفة هسنبرغ علوية حقيقية H.
!>

المعلمات

RIGHTV 
!>          RIGHTV هو LOGICAL
!>          = .TRUE. : حساب المتجهة الذاتية اليمنى؛
!>          = .FALSE.: حساب المتجهة الذاتية اليسرى.
!>

NOINIT

!>          NOINIT هو LOGICAL
!>          = .TRUE. : لم يتم توفير متجه أولي في (VR,VI).
!>          = .FALSE.: تم توفير متجه أولي في (VR,VI).
!>

N

!>          N هو INTEGER
!>          ترتيب المصفوفة H.  N >= 0.
!>

H

!>          H هو مصفوفة REAL، البعد (LDH,N)
!>          مصفوفة هسنبرغ العلوية H.
!>

LDH

!>          LDH هو INTEGER
!>          البعد الرائد للمصفوفة H.  LDH >= max(1,N).
!>

WR

!>          WR هو REAL
!>

WI

!>          WI هو REAL
!>          الأجزاء الحقيقية والخيالية للقيمة الذاتية لـ H التي يتم
!>          حساب المتجهة الذاتية اليمنى أو اليسرى المقابلة لها.
!>

VR

!>          VR هو مصفوفة REAL، البعد (N)
!>

VI

!>          VI هو مصفوفة REAL، البعد (N)
!>          عند الإدخال، إذا كان NOINIT = .FALSE. و WI = 0.0، يجب أن تحتوي VR على
!>          متجه بداية حقيقي للتكرار العكسي باستخدام القيمة الذاتية الحقيقية
!>          WR؛ إذا كان NOINIT = .FALSE. و WI.ne.0.0، يجب أن تحتوي VR و VI على
!>          الأجزاء الحقيقية والخيالية لمتجه بداية مركب
!>          للتكرار العكسي باستخدام القيمة الذاتية المركبة
!>          (WR,WI)؛ بخلاف ذلك، لا يلزم تعيين VR و VI.
!>          عند الخروج، إذا كان WI = 0.0 (قيمة ذاتية حقيقية)، تحتوي VR على
!>          المتجهة الذاتية الحقيقية المحسوبة؛ إذا كان WI.ne.0.0 (قيمة ذاتية مركبة)،
!>          تحتوي VR و VI على الأجزاء الحقيقية والخيالية للمتجهة الذاتية
!>          المركبة المحسوبة. يتم تطبيع المتجهة الذاتية بحيث يكون للمكون ذي
!>          أكبر قدر حجم يساوي 1؛ هنا، يتم أخذ حجم العدد المركب (x,y)
!>          ليكون |x| + |y|.
!>          لا يتم الرجوع إلى VI إذا كان WI = 0.0.
!>

B

!>          B هو مصفوفة REAL، البعد (LDB,N)
!>

LDB

!>          LDB هو INTEGER
!>          البعد الرائد للمصفوفة B.  LDB >= N+1.
!>

WORK

!>          WORK هو مصفوفة REAL، البعد (N)
!>

EPS3

!>          EPS3 هو REAL
!>          قيمة صغيرة تعتمد على الآلة تستخدم لـ perturbation
!>          للقيم الذاتية القريبة، وللاستبدال بالمحاور الصفرية.
!>

SMLNUM

!>          SMLNUM هو REAL
!>          قيمة تعتمد على الآلة قريبة من عتبة التقريب للأدنى.
!>

BIGNUM

!>          BIGNUM هو REAL
!>          قيمة تعتمد على الآلة قريبة من عتبة التقريب للأعلى.
!>

INFO

!>          INFO هو INTEGER
!>          = 0:  انتهاء ناجح
!>          = 1:  لم يتقارب التكرار العكسي؛ تم تعيين VR إلى آخر تكرار،
!>                وكذلك VI إذا كان WI.ne.0.0.
!>
