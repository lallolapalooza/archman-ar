subroutine slagts (job, n, a, b, c, d, in, y, tol, info)
SLAGTS تقوم بحل نظام المعادلات (T-λI)x = y أو (T-λI)^Tx = y، حيث T مصفوفة ثلاثية عامة و λ قيمة قياسية، باستخدام عامل LU المحسوب بواسطة slagtf.

توثيق الوظيفة/الوحدة الفرعية

subroutine slagts (integer job, integer n, real, dimension( * ) a, real, dimension( * ) b, real, dimension( * ) c, real, dimension( * ) d, integer, dimension( * ) in, real, dimension( * ) y, real tol, integer info)

SLAGTS تقوم بحل نظام المعادلات (T-λI)x = y أو (T-λI)^Tx = y، حيث T مصفوفة ثلاثية عامة و λ قيمة قياسية، باستخدام عامل LU المحسوب بواسطة slagtf.

الغرض:

!>
!> قد يتم استخدام SLAGTS لحل أحد أنظمة المعادلات
!>
!>    (T - lambda*I)*x = y   أو   (T - lambda*I)**T*x = y,
!>
!> حيث T مصفوفة ثلاثية n × n، لحساب x، بعد عامل (T - lambda*I) على الصورة
!>
!>    (T - lambda*I) = P*L*U ,
!>
!> بواسطة الروتين SLAGTF. يتم التحكم في اختيار المعادلة التي سيتم حلها
!> بواسطة الوسيط JOB، وفي كل حالة هناك خيار
!> لتعطيل العناصر القطرية الصفرية أو الصغيرة جدًا في U، وهذا الخيار
!> مخصص للاستخدام في تطبيقات مثل التكرار العكسي.
!>

المعاملات

JOB 
!>          JOB هو عدد صحيح (INTEGER)
!>          يحدد المهمة التي سيتم تنفيذها بواسطة SLAGTS كما يلي:
!>          =  1: يتم حل المعادلات  (T - lambda*I)x = y ،
!>                ولكن لا يتم تعطيل العناصر القطرية لـ U.
!>          = -1: يتم حل المعادلات  (T - lambda*I)x = y
!>                وإذا حدث تجاوز، يتم تعطيل العناصر القطرية
!>                لـ U. انظر الوسيط TOL أدناه.
!>          =  2: يتم حل المعادلات  (T - lambda*I)**Tx = y ،
!>                ولكن لا يتم تعطيل العناصر القطرية لـ U.
!>          = -2: يتم حل المعادلات  (T - lambda*I)**Tx = y
!>                وإذا حدث تجاوز، يتم تعطيل العناصر القطرية
!>                لـ U. انظر الوسيط TOL أدناه.
!>

N

!>          N هو عدد صحيح (INTEGER)
!>          رتبة المصفوفة T.
!>

A

!>          A هي مصفوفة حقيقية (REAL array)، أبعادها (N)
!>          عند الإدخال، يجب أن تحتوي A على العناصر القطرية لـ U كما
!>          تم إرجاعها من SLAGTF.
!>

B

!>          B هي مصفوفة حقيقية (REAL array)، أبعادها (N-1)
!>          عند الإدخال، يجب أن تحتوي B على عناصر القطر العلوي الأول لـ
!>          U كما تم إرجاعها من SLAGTF.
!>

C

!>          C هي مصفوفة حقيقية (REAL array)، أبعادها (N-1)
!>          عند الإدخال، يجب أن تحتوي C على عناصر القطر السفلي لـ L كما
!>          تم إرجاعها من SLAGTF.
!>

D

!>          D هي مصفوفة حقيقية (REAL array)، أبعادها (N-2)
!>          عند الإدخال، يجب أن تحتوي D على عناصر القطر العلوي الثاني
!>          لـ U كما تم إرجاعها من SLAGTF.
!>

IN

!>          IN هي مصفوفة أعداد صحيحة (INTEGER array)، أبعادها (N)
!>          عند الإدخال، يجب أن تحتوي IN على تفاصيل المصفوفة P كما تم
!>          إرجاعها من SLAGTF.
!>

Y

!>          Y هي مصفوفة حقيقية (REAL array)، أبعادها (N)
!>          عند الإدخال، المتجه الطرف الأيمن y.
!>          عند الخروج، يتم الكتابة فوق Y بالمتجه الحل x.
!>

TOL

!>          TOL هي قيمة حقيقية (REAL)
!>          عند الإدخال، عندما يكون JOB < 0، يجب أن يكون TOL هو الحد الأدنى
!>          للتعطيل الذي يجب إجراؤه للعناصر القطرية الصغيرة جدًا في U.
!>          يجب عادةً اختيار TOL ليكون حوالي eps*norm(U)، حيث eps
!>          هي الدقة النسبية للآلة، ولكن إذا تم توفير TOL كـ
!>          قيمة غير موجبة، فسيتم إعادة تعيينها إلى eps*max( abs( u(i,j) ) ).
!>          إذا كان JOB > 0، فلن يتم استخدام TOL.
!>
!>          عند الخروج، تتغير TOL كما هو موضح أعلاه، فقط إذا كانت
!>          TOL غير موجبة عند الإدخال. بخلاف ذلك، تظل TOL دون تغيير.
!>

INFO

!>          INFO هو عدد صحيح (INTEGER)
!>          = 0: خروج ناجح
!>          < 0: إذا كان INFO = -i، فإن الوسيط i كان له قيمة غير قانونية
!>          > 0: سيحدث تجاوز عند حساب العنصر INFO
!>               من متجه الحل x. يمكن أن يحدث هذا فقط
!>               عندما يتم توفير JOB بشكل إيجابي ويعني إما
!>               أن عنصرًا قطريًا في U صغير جدًا، أو أن
!>               عناصر المتجه الطرف الأيمن y كبيرة جدًا.
!>
