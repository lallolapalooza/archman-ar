توثيق الوظيفة/الإجراء الفرعي

subroutine slaqr3 (logical wantt, logical wantz, integer n, integer ktop, integer kbot, integer nw, real, dimension( ldh, * ) h, integer ldh, integer iloz, integer ihiz, real, dimension( ldz, * ) z, integer ldz, integer ns, integer nd, real, dimension( * ) sr, real, dimension( * ) si, real, dimension( ldv, * ) v, integer ldv, integer nh, real, dimension( ldt, * ) t, integer ldt, integer nv, real, dimension( ldwv, * ) wv, integer ldwv, real, dimension( * ) work, integer lwork)

SLAQR3 تجري تحويل التشابه المتعامد لمصفوفة هسنبرغ لاكتشاف القيم الذاتية التي تتقارب بالكامل من مصفوفة فرعية رئيسية لاحقة (ازالة مبكرة شرسة).

الغرض:

!>
!>    ازالة مبكرة شرسة:
!>
!>    تقبل SLAQR3 كمُدخل مصفوفة هسنبرغ علوية
!>    H وتجري تحويل تشابه متعامد
!>    مصمم لاكتشاف القيم الذاتية التي تتقارب بالكامل وازالتها من
!>    مصفوفة فرعية رئيسية لاحقة. على المخرج تم استبدال H بـ
!>    مصفوفة هسنبرغ جديدة هي اضطراب لـ
!>    تحويل تشابه متعامد لـ H. من المأمول أن يكون الإصدار النهائي لـ H
!>    به العديد من الإدخالات الفرعية القطرية الصفرية.
!>

المعلمات

WANTT 
!>          WANTT هو LOGICAL (منطقي)
!>          إذا كانت .TRUE.، فسيتم تحديث مصفوفة هسنبرغ H بالكامل
!>          بحيث يمكن حساب عامل شور شبه المثلث (بالتعاون مع الإجراء الفرعي المستدعي).
!>          إذا كانت .FALSE.، فلن يتم تحديث سوى جزء كافٍ من H للحفاظ
!>          على القيم الذاتية.
!>

WANTZ

!>          WANTZ هو LOGICAL (منطقي)
!>          إذا كانت .TRUE.، فسيتم تحديث المصفوفة المتعامدة Z
!>          بحيث يمكن حساب عامل شور المتعامد (بالتعاون مع الإجراء الفرعي المستدعي).
!>          إذا كانت .FALSE.، فلن يتم الاستشهاد بـ Z.
!>

N

!>          N هو INTEGER (عدد صحيح)
!>          ترتيب المصفوفة H و (إذا كانت WANTZ هي .TRUE.) ترتيب المصفوفة المتعامدة Z.
!>

KTOP

!>          KTOP هو INTEGER (عدد صحيح)
!>          من المفترض أن KTOP = 1 أو H(KTOP,KTOP-1)=0.
!>          يحدد KBOT و KTOP معًا كتلة معزولة
!>          على طول قطر مصفوفة هسنبرغ.
!>

KBOT

!>          KBOT هو INTEGER (عدد صحيح)
!>          من المفترض بدون فحص أن KBOT = N أو H(KBOT+1,KBOT)=0.
!>          يحدد KBOT و KTOP معًا كتلة معزولة
!>          على طول قطر مصفوفة هسنبرغ.
!>

NW

!>          NW هو INTEGER (عدد صحيح)
!>          حجم نافذة الإزالة. 1 <= NW <= (KBOT-KTOP+1).
!>

H

!>          H هي مصفوفة REAL (حقيقية)، أبعادها (LDH,N)
!>          على المُدخل، الجزء العلوي N × N من H يخزن
!>          مصفوفة هسنبرغ التي تخضع لإزالة مبكرة شرسة.
!>          على المخرج، تم تحويل H بواسطة تحويل تشابه متعامد،
!>          وتم إحداث اضطراب، ثم إعادتها إلى شكل هسنبرغ (من المأمول أن يكون به
!>          بعض الإدخالات الفرعية القطرية الصفرية).
!>

LDH

!>          LDH هو INTEGER (عدد صحيح)
!>          البعد الرائد لـ H كما هو معلن في الإجراء الفرعي المستدعي.
!>          N <= LDH
!>

ILOZ

!>          ILOZ هو INTEGER (عدد صحيح)
!>

IHIZ

!>          IHIZ هو INTEGER (عدد صحيح)
!>          حدد صفوف Z التي يجب تطبيق التحويلات عليها إذا كانت WANTZ هي .TRUE..
!>          1 <= ILOZ <= IHIZ <= N.
!>

Z

!>          Z هي مصفوفة REAL (حقيقية)، أبعادها (LDZ,N)
!>          إذا كانت WANTZ هي .TRUE.، فعلى المخرج، تم تجميع
!>          تحويل التشابه المتعامد المذكور أعلاه في Z(ILOZ:IHIZ,ILOZ:IHIZ) من اليمين.
!>          إذا كانت WANTZ هي .FALSE.، فلن يتم الاستشهاد بـ Z.
!>

LDZ

!>          LDZ هو INTEGER (عدد صحيح)
!>          البعد الرائد لـ Z كما هو معلن في الإجراء الفرعي المستدعي.
!>          1 <= LDZ.
!>

NS

!>          NS هو INTEGER (عدد صحيح)
!>          عدد القيم الذاتية غير المتقاربة (أي التقريبية)
!>          التي تم إرجاعها في SR و SI والتي يمكن استخدامها كتحولات بواسطة
!>          الإجراء الفرعي المستدعي.
!>

ND

!>          ND هو INTEGER (عدد صحيح)
!>          عدد القيم الذاتية المتقاربة التي تم اكتشافها بواسطة هذا
!>          الإجراء الفرعي.
!>

SR

!>          SR هي مصفوفة REAL (حقيقية)، أبعادها (KBOT)
!>

SI

!>          SI هي مصفوفة REAL (حقيقية)، أبعادها (KBOT)
!>          على المخرج، يتم تخزين الأجزاء الحقيقية والخيالية للقيم الذاتية التقريبية
!>          التي يمكن استخدامها للتحولات في
!>          SR(KBOT-ND-NS+1) حتى SR(KBOT-ND) و
!>          SI(KBOT-ND-NS+1) حتى SI(KBOT-ND)، على التوالي.
!>          يتم تخزين الأجزاء الحقيقية والخيالية للقيم الذاتية المتقاربة
!>          في SR(KBOT-ND+1) حتى SR(KBOT) و
!>          SI(KBOT-ND+1) حتى SI(KBOT)، على التوالي.
!>

V

!>          V هي مصفوفة REAL (حقيقية)، أبعادها (LDV,NW)
!>          مصفوفة عمل NW × NW.
!>

LDV

!>          LDV هو INTEGER (عدد صحيح)
!>          البعد الرائد لـ V كما هو معلن في الإجراء الفرعي المستدعي.
!>          NW <= LDV
!>

NH

!>          NH هو INTEGER (عدد صحيح)
!>          عدد أعمدة T. NH >= NW.
!>

T

!>          T هي مصفوفة REAL (حقيقية)، أبعادها (LDT,NW)
!>

LDT

!>          LDT هو INTEGER (عدد صحيح)
!>          البعد الرائد لـ T كما هو معلن في الإجراء الفرعي المستدعي.
!>          NW <= LDT
!>

NV

!>          NV هو INTEGER (عدد صحيح)
!>          عدد صفوف مصفوفة العمل WV المتاحة كـ
!>          مساحة عمل. NV >= NW.
!>

WV

!>          WV هي مصفوفة REAL (حقيقية)، أبعادها (LDWV,NW)
!>

LDWV

!>          LDWV هو INTEGER (عدد صحيح)
!>          البعد الرائد لـ W كما هو معلن في الإجراء الفرعي المستدعي.
!>          NW <= LDV
!>

WORK

!>          WORK هي مصفوفة REAL (حقيقية)، أبعادها (LWORK)
!>          عند الخروج، يتم تعيين WORK(1) إلى تقدير للقيمة المثلى
!>          لـ LWORK للقيم المعطاة لـ N، NW، KTOP، و KBOT.
!>

LWORK

!>          LWORK هو INTEGER (عدد صحيح)
!>          بعد مصفوفة العمل WORK. 2*NW كافٍ،
!>          ولكن قد تنتج كفاءة أكبر من قيم LWORK الأكبر.
!>
!>          إذا كان LWORK = -1، فمن المفترض وجود استعلام مساحة عمل؛ SLAQR3
!>          يقدر فقط حجم مساحة العمل المثلى للقيم المعطاة
!>          لـ N، NW، KTOP، و KBOT. يتم إرجاع التقدير
!>          في WORK(1). لا يتم إصدار رسالة خطأ متعلقة بـ LWORK
!>          بواسطة XERBLA. لا يتم الوصول إلى H أو Z.
!>
