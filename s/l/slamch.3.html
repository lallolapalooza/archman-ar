SLAMCH

الغرض:

!>
!> SLAMCH تحدد معلمات الآلة ذات الدقة الأحادية.
!>

المعلمات

CMACH 
!>          CMACH هو CHARACTER*1
!>          يحدد القيمة التي سيتم إرجاعها بواسطة SLAMCH:
!>          = 'E' أو 'e'،   SLAMCH := eps
!>          = 'S' أو 's'،   SLAMCH := sfmin
!>          = 'B' أو 'b'،   SLAMCH := base
!>          = 'P' أو 'p'،   SLAMCH := eps*base
!>          = 'N' أو 'n'،   SLAMCH := t
!>          = 'R' أو 'r'،   SLAMCH := rnd
!>          = 'M' أو 'm'،   SLAMCH := emin
!>          = 'U' أو 'u'،   SLAMCH := rmin
!>          = 'L' أو 'l'،   SLAMCH := emax
!>          = 'O' أو 'o'،   SLAMCH := rmax
!>          حيث
!>          eps   = الدقة النسبية للآلة
!>          sfmin = الحد الأدنى الآمن، بحيث لا يتجاوز 1/sfmin
!>          base  = أساس الآلة
!>          prec  = eps*base
!>          t     = عدد أرقام (الأساس) في الصيغة
!>          rnd   = 1.0 عند حدوث التقريب في الجمع، 0.0 بخلاف ذلك
!>          emin  = أقل أس قبل (التناقص التدريجي)
!>          rmin  = عتبة التناقص - base**(emin-1)
!>          emax  = أكبر أس قبل الفائض
!>          rmax  = عتبة الفائض - (base**emax)*(1-eps)
!>
