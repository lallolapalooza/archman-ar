subroutine slarrd (character range, character order, integer n, real vl, real vu, integer il, integer iu, real, dimension( * ) gers, real reltol, real, dimension( * ) d, real, dimension( * ) e, real, dimension( * ) e2, real pivmin, integer nsplit, integer, dimension( * ) isplit, integer m, real, dimension( * ) w, real, dimension( * ) werr, real wl, real wu, integer, dimension( * ) iblock, integer, dimension( * ) indexw, real, dimension( * ) work, integer, dimension( * ) iwork, integer info)

SLARRD يحسب القيم الذاتية لمصفوفة متناظرة ثلاثية الأقطار بدقة مناسبة.

الغرض:

!>
!> SLARRD يحسب القيم الذاتية لمصفوفة T المتناظرة ثلاثية الأقطار
!> بدقة مناسبة. هذا رمز مساعد يتم استدعاؤه من SSTEMR.
!> قد يطلب المستخدم جميع القيم الذاتية، أو جميع القيم الذاتية
!> في الفترة نصف المفتوحة (VL, VU]، أو القيم الذاتية من IL-th إلى IU-th.
!>
!> لتجنب تجاوز سعة الذاكرة، يجب قياس المصفوفة بحيث لا يتجاوز أكبر عنصر فيها
!> قيمة overflow**(1/2) * underflow**(1/4) بالقيمة المطلقة، وللحصول على
!> أقصى دقة، يجب ألا تكون أقل من ذلك بكثير.
!>
!> انظر W. Kahan ، التقرير CS41، قسم علوم الحاسوب، جامعة ستانفورد
!> ، 21 يوليو 1966.
!>

المعاملات

RANGE 
!>          RANGE is CHARACTER*1
!>          = 'A': ()   سيتم العثور على جميع القيم الذاتية.
!>          = 'V': () سيتم العثور على جميع القيم الذاتية في الفترة
!>                           نصف المفتوحة (VL, VU].
!>          = 'I': () سيتم العثور على القيم الذاتية من IL-th إلى IU-th (للمصفوفة
!>                           بأكملها).
!>

ORDER

!>          ORDER is CHARACTER*1
!>          = 'B': () سيتم تجميع القيم الذاتية حسب
!>                              الكتلة المنفصلة (انظر IBLOCK، ISPLIT) و
!>                              سيتم ترتيبها من الأصغر إلى الأكبر داخل
!>                              الكتلة.
!>          = 'E': ()
!>                              سيتم ترتيب القيم الذاتية للمصفوفة
!>                              بأكملها من الأصغر إلى
!>                              الأكبر.
!>

N

!>          N is INTEGER
!>          ترتيب المصفوفة ثلاثية الأقطار T. N >= 0.
!>

VL

!>          VL is REAL
!>          إذا كانت RANGE='V'، فإن الحد الأدنى للفترة المراد
!>          البحث فيها عن القيم الذاتية. القيم الذاتية الأقل من أو تساوي
!>          VL، أو الأكبر من VU، لن يتم إرجاعها. VL < VU.
!>          لا يتم الرجوع إليها إذا كانت RANGE = 'A' أو 'I'.
!>

VU

!>          VU is REAL
!>          إذا كانت RANGE='V'، فإن الحد الأعلى للفترة المراد
!>          البحث فيها عن القيم الذاتية. القيم الذاتية الأقل من أو تساوي
!>          VL، أو الأكبر من VU، لن يتم إرجاعها. VL < VU.
!>          لا يتم الرجوع إليها إذا كانت RANGE = 'A' أو 'I'.
!>

IL

!>          IL is INTEGER
!>          إذا كانت RANGE='I'، فإن مؤشر
!>          أصغر قيمة ذاتية سيتم إرجاعها.
!>          1 <= IL <= IU <= N، إذا كانت N > 0؛ IL = 1 و IU = 0 إذا كانت N = 0.
!>          لا يتم الرجوع إليها إذا كانت RANGE = 'A' أو 'V'.
!>

IU

!>          IU is INTEGER
!>          إذا كانت RANGE='I'، فإن مؤشر
!>          أكبر قيمة ذاتية سيتم إرجاعها.
!>          1 <= IL <= IU <= N، إذا كانت N > 0؛ IL = 1 و IU = 0 إذا كانت N = 0.
!>          لا يتم الرجوع إليها إذا كانت RANGE = 'A' أو 'V'.
!>

GERS

!>          GERS is REAL array, dimension (2*N)
!>          فترات Gerschgorin الـ N (الفترة i-th Gerschgorin
!>          هي (GERS(2*i-1)، GERS(2*i)).
!>

RELTOL

!>          RELTOL is REAL
!>          الحد الأدنى للعرض النسبي للفترة. عندما تكون الفترة
!>          أضيق من RELTOL مضروبة في الطرف الأكبر (بالقيمة المطلقة)
!>          من الطرفين، تعتبر أنها
!>          صغيرة بما يكفي، أي، متقاربة. ملاحظة: يجب أن يكون هذا
!>          دائمًا على الأقل radix*machine epsilon.
!>

D

!>          D is REAL array, dimension (N)
!>          عناصر القطر الرئيسية الـ n للمصفوفة ثلاثية الأقطار T.
!>

E

!>          E is REAL array, dimension (N-1)
!>          العناصر خارج القطر الرئيسية الـ (n-1) للمصفوفة ثلاثية الأقطار T.
!>

E2

!>          E2 is REAL array, dimension (N-1)
!>          مربعات العناصر خارج القطر الرئيسية الـ (n-1) للمصفوفة ثلاثية الأقطار T.
!>

PIVMIN

!>          PIVMIN is REAL
!>          أصغر محور دوران مسموح به في متتالية ستورم لـ T.
!>

NSPLIT

!>          NSPLIT is INTEGER
!>          عدد كتل القطر في المصفوفة T.
!>          1 <= NSPLIT <= N.
!>

ISPLIT

!>          ISPLIT is INTEGER array, dimension (N)
!>          نقاط التقسيم، التي تنقسم عندها T إلى مصفوفات فرعية.
!>          المصفوفة الفرعية الأولى تتكون من الصفوف/الأعمدة من 1 إلى ISPLIT(1)،
!>          الثانية من الصفوف/الأعمدة ISPLIT(1)+1 إلى ISPLIT(2)،
!>          إلخ، والـ NSPLIT-th تتكون من الصفوف/الأعمدة
!>          ISPLIT(NSPLIT-1)+1 إلى ISPLIT(NSPLIT)=N.
!>          (سيتم استخدام أول NSPLIT عنصر فقط فعليًا، ولكن
!>          نظرًا لأنه لا يمكن للمستخدم معرفة قيمة NSPLIT مسبقًا،
!>          يجب حجز N كلمة لـ ISPLIT.)
!>

M

!>          M is INTEGER
!>          العدد الفعلي للقيم الذاتية التي تم العثور عليها. 0 <= M <= N.
!>          (انظر أيضًا وصف INFO=2,3.)
!>

W

!>          W is REAL array, dimension (N)
!>          عند الخروج، ستحتوي العناصر الـ M الأولى من W على
!>          تقريب القيم الذاتية. SLARRD يحسب فترة
!>          I_j = (a_j, b_j] التي تشمل القيمة الذاتية j. التقريب
!>          للقيمة الذاتية معطى بمنتصف الفترة
!>          W(j)= ( a_j + b_j)/2. الخطأ المقابل محدود بـ
!>          WERR(j) = abs( a_j - b_j)/2
!>

WERR

!>          WERR is REAL array, dimension (N)
!>          حد الخطأ على التقريب المقابل للقيمة الذاتية
!>          في W.
!>

WL

!>          WL is REAL
!>

WU

!>          WU is REAL
!>          الفترة (WL, WU] تحتوي على جميع القيم الذاتية المطلوبة.
!>          إذا كانت RANGE='V'، فإن WL=VL و WU=VU.
!>          إذا كانت RANGE='A'، فإن WL و WU هما حدود Gerschgorin العامة
!>                        على الطيف.
!>          إذا كانت RANGE='I'، فإن WL و WU يتم حسابهما بواسطة SLAEBZ من
!>                        نطاق المؤشر المحدد.
!>

IBLOCK

!>          IBLOCK is INTEGER array, dimension (N)
!>          عند كل صف/عمود j حيث يكون E(j) صفرًا أو صغيرًا،
!>          تعتبر المصفوفة T منقسمة إلى مصفوفة قطرية مقسمة.
!>          عند الخروج، إذا كانت INFO = 0، فإن IBLOCK(i) تحدد إلى أي
!>          كتلة (من 1 إلى عدد الكتل) تنتمي القيمة الذاتية W(i).
!>          (قد تستخدم SLARRD العناصر المتبقية N-M كـ
!>          مساحة عمل.)
!>
