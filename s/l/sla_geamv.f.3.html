subroutine sla_geamv (integer trans, integer m, integer n, real alpha, real, dimension( lda, * ) a, integer lda, real, dimension( * ) x, integer incx, real beta, real, dimension( * ) y, integer incy)

SLA_GEAMV تحسب حاصل ضرب مصفوفة-متجه باستخدام مصفوفة عامة لحساب حدود الخطأ.

الغرض:

!>
!> SLA_GEAMV  تقوم بإحدى عمليات المصفوفة-المتجه التالية:
!>
!>         y := alpha*abs(A)*abs(x) + beta*abs(y),
!>    أو   y := alpha*abs(A)**T*abs(x) + beta*abs(y),
!>
!> حيث alpha و beta قيم قياسية، x و y متجهات، و A مصفوفة من حجم m × n.
!>
!> هذه الوظيفة تستخدم بشكل أساسي في حساب حدود الخطأ.
!> للحماية من انخفاض القيمة إلى الصفر أثناء التقييم، يتم إزاحة المكونات في
!> المتجه الناتج بعيدًا عن الصفر بمقدار (N+1)
!> مرة من حد الانخفاض إلى الصفر. لمنع الأخطاء غير الضرورية الكبيرة للمصفوفات
!> ذات البنية الكتلية المضمنة في المصفوفات العامة،
!>  لا يتم إزاحة المكونات الصفرية. يعتبر الإدخال الصفري
!> إذا كانت جميع عمليات الضرب المتضمنة
!> في حساب هذا الإدخال تحتوي على عامل ضرب صفري واحد على الأقل.
!>

المعلمات

TRANS 
!>          TRANS هو عدد صحيح (INTEGER)
!>           عند الإدخال، يحدد TRANS العملية المراد تنفيذها كما يلي:
!>
!>             BLAS_NO_TRANS      y := alpha*abs(A)*abs(x) + beta*abs(y)
!>             BLAS_TRANS         y := alpha*abs(A**T)*abs(x) + beta*abs(y)
!>             BLAS_CONJ_TRANS    y := alpha*abs(A**T)*abs(x) + beta*abs(y)
!>
!>           لا تتغير عند الخروج.
!>

M

!>          M هو عدد صحيح (INTEGER)
!>           عند الإدخال، يحدد M عدد صفوف المصفوفة A.
!>           يجب أن يكون M أكبر من أو يساوي صفرًا.
!>           لا يتغير عند الخروج.
!>

N

!>          N هو عدد صحيح (INTEGER)
!>           عند الإدخال، يحدد N عدد أعمدة المصفوفة A.
!>           يجب أن يكون N أكبر من أو يساوي صفرًا.
!>           لا يتغير عند الخروج.
!>

ALPHA

!>          ALPHA هو قيمة حقيقية (REAL)
!>           عند الإدخال، تحدد ALPHA القيمة القياسية alpha.
!>           لا تتغير عند الخروج.
!>

A

!>          A هو مصفوفة قيم حقيقية (REAL array)، أبعادها ( LDA, n )
!>           قبل الإدخال، يجب أن يحتوي الجزء الرئيسي من المصفوفة A بحجم m × n على
!>           مصفوفة المعاملات.
!>           لا تتغير عند الخروج.
!>

LDA

!>          LDA هو عدد صحيح (INTEGER)
!>           عند الإدخال، يحدد LDA البعد الأول للمصفوفة A كما تم تعريفه
!>           في البرنامج (الروتين الفرعي) المستدعي. يجب أن يكون LDA أكبر من أو يساوي
!>           max( 1, m ).
!>           لا يتغير عند الخروج.
!>

X

!>          X هو مصفوفة قيم حقيقية (REAL array)، أبعادها
!>           ( 1 + ( n - 1 )*abs( INCX ) ) عندما TRANS = 'N' أو 'n'
!>           وعلى الأقل
!>           ( 1 + ( m - 1 )*abs( INCX ) ) خلاف ذلك.
!>           قبل الإدخال، يجب أن تحتوي المصفوفة X المزودة بالزيادات على
!>           المتجه x.
!>           لا تتغير عند الخروج.
!>

INCX

!>          INCX هو عدد صحيح (INTEGER)
!>           عند الإدخال، يحدد INCX الزيادة للعناصر في
!>           X. يجب أن لا يكون INCX صفرًا.
!>           لا يتغير عند الخروج.
!>

BETA

!>          BETA هو قيمة حقيقية (REAL)
!>           عند الإدخال، تحدد BETA القيمة القياسية beta. عندما يتم
!>           توفير BETA كصفر، لا يلزم تعيين Y عند الإدخال.
!>           لا تتغير عند الخروج.
!>

Y

!>          Y هي مصفوفة قيم حقيقية (REAL array)،
!>           أبعادها على الأقل
!>           ( 1 + ( m - 1 )*abs( INCY ) ) عندما TRANS = 'N' أو 'n'
!>           وعلى الأقل
!>           ( 1 + ( n - 1 )*abs( INCY ) ) خلاف ذلك.
!>           قبل الإدخال مع وجود BETA غير صفري، يجب أن تحتوي المصفوفة Y المزودة بالزيادات على
!>           المتجه y. عند الخروج، يتم استبدال Y بالمتجه y
!>           المحدث.
!>           إذا كان m أو n يساوي صفرًا، فلن يتم الرجوع إلى Y وستقوم الوظيفة
!>           بإجراء عودة سريعة.
!>

INCY

!>          INCY هو عدد صحيح (INTEGER)
!>           عند الإدخال، يحدد INCY الزيادة للعناصر في
!>           Y. يجب أن لا يكون INCY صفرًا.
!>           لا يتغير عند الخروج.
!>
!>  روتين BLAS من المستوى 2.
!>
