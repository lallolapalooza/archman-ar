subroutine slasdq (character uplo, integer sqre, integer n, integer ncvt, integer nru, integer ncc, real, dimension( * ) d, real, dimension( * ) e, real, dimension( ldvt, * ) vt, integer ldvt, real, dimension( ldu, * ) u, integer ldu, real, dimension( ldc, * ) c, integer ldc, real, dimension( * ) work, integer info)

SLASDQ تحسب SVD لمصفوفة ثنائية القطر حقيقية مع قطري d وقطري ثانوي e. تستخدم بواسطة sbdsdc.

الغرض:

!>
!> SLASDQ تحسب تحليل القيم المفردة (SVD) لمصفوفة ثنائية القطر حقيقية
!> (علوية أو سفلية) مع قطري D وقطري ثانوي E، وتجميع التحويلات إذا رغبت.
!> لتكن B المصفوفة ثنائية القطر المدخلة، يقوم الخوارزم بتعيين
!> المصفوفات المتعامدة Q و P بحيث B = Q * S * P**T (P**T تعني منقول P).
!> القيم المفردة S يتم الكتابة فوقها على D.
!>
!> يتم تغيير المصفوفة المدخلة U إلى U * Q إذا رغبت.
!> يتم تغيير المصفوفة المدخلة VT إلى P**T * VT إذا رغبت.
!> يتم تغيير المصفوفة المدخلة C إلى Q**T * C إذا رغبت.
!>
!> انظر  بواسطة J. Demmel و W. Kahan،
!> مذكرة عمل LAPACK #3، لوصف مفصل للخوارزم.
!>

المعاملات

UPLO 
!>          UPLO is CHARACTER*1
!>        عند الدخول، تحدد UPLO ما إذا كانت المصفوفة ثنائية القطر المدخلة
!>        ثنائية القطر علوية أم سفلية، وما إذا كانت مربعة أم لا.
!>           UPLO = 'U' or 'u'   B هي ثنائية القطر علوية.
!>           UPLO = 'L' or 'l'   B هي ثنائية القطر سفلية.
!>

SQRE

!>          SQRE is INTEGER
!>        = 0: إذن المصفوفة المدخلة هي N-by-N.
!>        = 1: إذن المصفوفة المدخلة هي N-by-(N+1) إذا كانت UPLU = 'U' و
!>             (N+1)-by-N إذا كانت UPLU = 'L'.
!>
!>        المصفوفة ثنائية القطر لديها
!>        N = NL + NR + 1 صفوف و
!>        M = N + SQRE >= N أعمدة.
!>

N

!>          N is INTEGER
!>        عند الدخول، تحدد N عدد الصفوف والأعمدة
!>        في المصفوفة. يجب أن تكون N على الأقل 0.
!>

NCVT

!>          NCVT is INTEGER
!>        عند الدخول، تحدد NCVT عدد أعمدة
!>        المصفوفة VT. يجب أن تكون NCVT على الأقل 0.
!>

NRU

!>          NRU is INTEGER
!>        عند الدخول، تحدد NRU عدد صفوف
!>        المصفوفة U. يجب أن تكون NRU على الأقل 0.
!>

NCC

!>          NCC is INTEGER
!>        عند الدخول، تحدد NCC عدد أعمدة
!>        المصفوفة C. يجب أن تكون NCC على الأقل 0.
!>

D

!>          D is REAL array, dimension (N)
!>        عند الدخول، تحتوي D على مدخلات القطر
!>        للمصفوفة ثنائية القطر التي يراد حساب SVD لها. عند الخروج الطبيعي،
!>        تحتوي D على القيم المفردة بترتيب تصاعدي.
!>

E

!>          E is REAL array.
!>        dimension is (N-1) if SQRE = 0 and N if SQRE = 1.
!>        عند الدخول، تحتوي مدخلات E على مدخلات القطر الثانوي
!>        للمصفوفة ثنائية القطر التي يراد حساب SVD لها. عند الخروج الطبيعي،
!>        ستحتوي E على 0. إذا لم تتقارب الخوارزمية،
!>        فستحتوي D و E على مدخلات القطر والقطر العلوي
!>        لمصفوفة ثنائية القطر مكافئة متعامدة للمدخلة.
!>

VT

!>          VT is REAL array, dimension (LDVT, NCVT)
!>        عند الدخول، تحتوي على مصفوفة تم
!>        ضربها مسبقاً بـ P**T عند الخروج، حجمها N-by-NCVT إذا كانت SQRE = 0
!>        و (N+1)-by-NCVT إذا كانت SQRE = 1 (لا تتم الإشارة إليها إذا كانت NCVT=0).
!>

LDVT

!>          LDVT is INTEGER
!>        عند الدخول، تحدد LDVT البعد الرئيسي لـ VT كما
!>        تم تعريفه في البرنامج (الفرعي) المستدعي. يجب أن تكون LDVT على
!>        الأقل 1. إذا كانت NCVT غير صفرية، يجب أن تكون LDVT أيضًا على الأقل N.
!>

U

!>          U is REAL array, dimension (LDU, N)
!>        عند الدخول، تحتوي على مصفوفة تم
!>        ضربها لاحقاً بـ Q عند الخروج، حجمها NRU-by-N إذا كانت SQRE = 0
!>        و NRU-by-(N+1) إذا كانت SQRE = 1 (لا تتم الإشارة إليها إذا كانت NRU=0).
!>

LDU

!>          LDU is INTEGER
!>        عند الدخول، تحدد LDU البعد الرئيسي لـ U كما
!>        تم تعريفه في البرنامج (الفرعي) المستدعي. يجب أن تكون LDU على
!>        الأقل max( 1, NRU ).
!>

C

!>          C is REAL array, dimension (LDC, NCC)
!>        عند الدخول، تحتوي على مصفوفة N-by-NCC تم
!>        ضربها مسبقاً بـ Q**T عند الخروج، حجمها N-by-NCC إذا كانت SQRE = 0
!>        و (N+1)-by-NCC إذا كانت SQRE = 1 (لا تتم الإشارة إليها إذا كانت NCC=0).
!>

LDC

!>          LDC is INTEGER
!>        عند الدخول، تحدد LDC البعد الرئيسي لـ C كما
!>        تم تعريفه في البرنامج (الفرعي) المستدعي. يجب أن تكون LDC على
!>        الأقل 1. إذا كانت NCC غير صفرية، يجب أن تكون LDC أيضًا على الأقل N.
!>

WORK

!>          WORK is REAL array, dimension (4*N)
!>        مساحة عمل. تتم الإشارة إليها فقط إذا كانت NCVT أو NRU أو NCC غير صفرية،
!>        وإذا كانت N على الأقل 2.
!>

INFO

!>          INFO is INTEGER
!>        عند الخروج، قيمة 0 تشير إلى خروج ناجح.
!>        إذا كانت INFO < 0، فإن رقم المعامل -INFO غير قانوني.
!>        إذا كانت INFO > 0، فإن الخوارزمية لم تتقارب، و INFO
!>        تحدد عدد الأقطار العلوية التي لم تتقارب.
!>
