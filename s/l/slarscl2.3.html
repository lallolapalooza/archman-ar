subroutine slarscl2 (integer m, integer n, real, dimension( * ) d, real, dimension( ldx, * ) x, integer ldx)

SLARSCL2 تقوم بالتحجيم القطري العكسي لمصفوفة.

الغرض:

!>
!> SLARSCL2 تقوم بالتحجيم القطري العكسي لمصفوفة:
!>   x <-- inv(D) * x
!> حيث يتم تخزين المصفوفة القطرية D كمتجه.
!>
!> سيتم استبدالها في النهاية بـ BLAS_sge_diag_scale في معيار BLAS الجديد.
!>

المعلمات

M 
!>          M هو عدد صحيح (INTEGER)
!>     عدد صفوف D و X. M >= 0.
!>

N

!>          N هو عدد صحيح (INTEGER)
!>     عدد أعمدة X. N >= 0.
!>

D

!>          D هو مصفوفة حقيقية (REAL array)، طولها M
!>     المصفوفة القطرية D، مخزنة كمتجه بطول M.
!>

X

!>          X هي مصفوفة حقيقية (REAL array)، أبعادها (LDX,N)
!>     عند الإدخال، المصفوفة X التي سيتم تحجيمها بواسطة D.
!>     عند الخروج، المصفوفة المحجمة.
!>

LDX

!>          LDX هو عدد صحيح (INTEGER)
!>     البعد الرئيسي للمصفوفة X. LDX >= M.
!>
