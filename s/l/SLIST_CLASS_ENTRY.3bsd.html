الماكرو SLIST_NEXT يُرجع العنصر التالي في القائمة.

الماكرو SLIST_REMOVE_AFTER يزيل العنصر الموجود بعد elm من القائمة. على عكس SLIST_REMOVE, هذا الماكرو لا يجتاز القائمة بأكملها.

الماكرو SLIST_REMOVE_HEAD يزيل العنصر elm من بداية القائمة. لتحقيق أقصى قدر من الكفاءة، يجب على العناصر التي تتم إزالتها من بداية القائمة استخدام هذا الماكرو صراحةً بدلاً من الماكرو العام SLIST_REMOVE.

الماكرو SLIST_REMOVE يزيل العنصر elm من القائمة. يجب تجنب استخدام هذا الماكرو لأنه يجتاز القائمة بأكملها. يجب استخدام قائمة مرتبطة مزدوجة إذا كان هذا الماكرو مطلوبًا في مسارات التعليمات البرمجية ذات الاستخدام العالي أو للعمل على قوائم طويلة.

الماكرو SLIST_SWAP يبادل محتويات head1 و head2.

مثال على قائمة مرتبطة أحادية

SLIST_HEAD(slisthead, entry) head =
    SLIST_HEAD_INITIALIZER(head);
struct slisthead *headp;		/* رأس قائمة مرتبطة أحادية. */
struct entry {
	...
	SLIST_ENTRY(entry) entries;	/* قائمة مرتبطة أحادية. */
	...
} *n1, *n2, *n3, *np;

SLIST_INIT(&head);			/* تهيئة القائمة. */

n1 = malloc(sizeof(struct entry));	/* الإدراج في البداية. */
SLIST_INSERT_HEAD(&head, n1, entries);

n2 = malloc(sizeof(struct entry));	/* الإدراج بعد. */
SLIST_INSERT_AFTER(n1, n2, entries);

SLIST_REMOVE(&head, n2, entry, entries);/* حذف. */
free(n2);

n3 = SLIST_FIRST(&head);
SLIST_REMOVE_HEAD(&head, entries);	/* حذف من البداية. */
free(n3);
					/* اجتياز أمامي. */
SLIST_FOREACH(np, &head, entries)
	np-> ...
					/* اجتياز أمامي آمن. */
SLIST_FOREACH_SAFE(np, &head, entries, np_temp) {
	np->do_stuff();
	...
	SLIST_REMOVE(&head, np, entry, entries);
	free(np);
}

while (!SLIST_EMPTY(&head)) {		/* حذف القائمة. */
	n1 = SLIST_FIRST(&head);
	SLIST_REMOVE_HEAD(&head, entries);
	free(n1);
}

قوائم ذات ذيل مرتبطة أحاديًا

قائمة ذات ذيل مرتبطة أحاديًا يرأسها هيكل معرف بواسطة الماكرو STAILQ_HEAD. يحتوي هذا الهيكل على زوج من المؤشرات، واحد إلى العنصر الأول في قائمة الذيل والآخر إلى العنصر الأخير في قائمة الذيل. العناصر مرتبطة أحاديًا لتقليل المساحة وتكاليف معالجة المؤشرات على حساب وقت إزالة O(n) للعناصر العشوائية. يمكن إضافة عناصر جديدة إلى قائمة الذيل بعد عنصر موجود، أو في بداية قائمة الذيل، أو في نهاية قائمة الذيل. يتم تعريف هيكل STAILQ_HEAD على النحو التالي:

STAILQ_HEAD(HEADNAME, TYPE) head;

حيث HEADNAME هو اسم الهيكل المراد تعريفه، و TYPE هو نوع العناصر المراد ربطها في قائمة الذيل. يمكن بعد ذلك تعريف مؤشر إلى رأس قائمة الذيل على النحو التالي:

struct HEADNAME *headp;

(الاسمين head و headp قابلان للاختيار من قبل المستخدم.)

الماكرو STAILQ_HEAD_INITIALIZER يُقيّم إلى مُهيئ لقائمة الذيل head.

الماكرو STAILQ_CONCAT يربط قائمة الذيل التي يرأسها head2 بنهاية القائمة التي يرأسها head1 ويزيل جميع الإدخالات من القائمة الأولى.

الماكرو STAILQ_EMPTY يُقيّم إلى صحيح إذا لم تكن هناك عناصر في قائمة الذيل.

الماكرو STAILQ_ENTRY يُعرّف هيكلًا يربط العناصر في قائمة الذيل.

الماكرو STAILQ_FIRST يُرجع العنصر الأول في قائمة الذيل أو NULL إذا كانت قائمة الذيل فارغة.

الماكرو STAILQ_FOREACH يجتاز قائمة الذيل المشار إليها بواسطة head في الاتجاه الأمامي، مع تعيين كل عنصر بدوره إلى var.

الماكرو STAILQ_FOREACH_FROM يعمل بشكل مطابق لـ STAILQ_FOREACH عندما يكون var هو NULL، وإلا فإنه يعامل var كعنصر STAILQ تم العثور عليه سابقًا ويبدأ الحلقة عند var بدلاً من العنصر الأول في STAILQ المشار إليه بواسطة head.

الماكرو STAILQ_FOREACH_SAFE يجتاز قائمة الذيل المشار إليها بواسطة head في الاتجاه الأمامي، مع تعيين كل عنصر بدوره إلى var. ومع ذلك، على عكس STAILQ_FOREACH() ، يُسمح هنا بإزالة var وكذلك تحريره من داخل الحلقة بأمان دون التدخل في الاجتياز.

الماكرو STAILQ_FOREACH_FROM_SAFE يعمل بشكل مطابق لـ STAILQ_FOREACH_SAFE عندما يكون var هو NULL، وإلا فإنه يعامل var كعنصر STAILQ تم العثور عليه سابقًا ويبدأ الحلقة عند var بدلاً من العنصر الأول في STAILQ المشار إليه بواسطة head.

الماكرو STAILQ_INIT يُهيئ قائمة الذيل المشار إليها بواسطة head.

الماكرو STAILQ_INSERT_HEAD يُدرج العنصر الجديد elm في بداية قائمة الذيل.

الماكرو STAILQ_INSERT_TAIL يُدرج العنصر الجديد elm في نهاية قائمة الذيل.

الماكرو STAILQ_INSERT_AFTER يُدرج العنصر الجديد elm بعد العنصر listelm.

الماكرو STAILQ_LAST يُرجع العنصر الأخير في قائمة الذيل. إذا كانت قائمة الذيل فارغة، فإن القيمة المرجعة هي NULL.

الماكرو STAILQ_NEXT يُرجع العنصر التالي في قائمة الذيل، أو NULL إذا كان هذا العنصر هو الأخير.

الماكرو STAILQ_REMOVE_AFTER يزيل العنصر الموجود بعد elm من قائمة الذيل. على عكس STAILQ_REMOVE, هذا الماكرو لا يجتاز قائمة الذيل بأكملها.

الماكرو STAILQ_REMOVE_HEAD يزيل العنصر الموجود في بداية قائمة الذيل. لتحقيق أقصى قدر من الكفاءة، يجب على العناصر التي تتم إزالتها من بداية قائمة الذيل استخدام هذا الماكرو صراحةً بدلاً من الماكرو العام STAILQ_REMOVE.

الماكرو STAILQ_REMOVE يزيل العنصر elm من قائمة الذيل. يجب تجنب استخدام هذا الماكرو لأنه يجتاز القائمة بأكملها. يجب استخدام قائمة ذيل مرتبطة مزدوجة إذا كان هذا الماكرو مطلوبًا في مسارات التعليمات البرمجية ذات الاستخدام العالي أو للعمل على قوائم ذيل طويلة.

الماكرو STAILQ_SWAP يبادل محتويات head1 و head2.

مثال على قائمة ذيل مرتبطة أحاديًا

STAILQ_HEAD(stailhead, entry) head =
    STAILQ_HEAD_INITIALIZER(head);
struct stailhead *headp;		/* رأس قائمة ذيل مرتبطة أحادية. */
struct entry {
	...
	STAILQ_ENTRY(entry) entries;	/* قائمة ذيل. */
	...
} *n1, *n2, *n3, *np;

STAILQ_INIT(&head);			/* تهيئة القائمة. */

n1 = malloc(sizeof(struct entry));	/* الإدراج في البداية. */
STAILQ_INSERT_HEAD(&head, n1, entries);

n1 = malloc(sizeof(struct entry));	/* الإدراج في النهاية. */
STAILQ_INSERT_TAIL(&head, n1, entries);

n2 = malloc(sizeof(struct entry));	/* الإدراج بعد. */
STAILQ_INSERT_AFTER(&head, n1, n2, entries);
					/* حذف. */
STAILQ_REMOVE(&head, n2, entry, entries);
free(n2);
					/* حذف من البداية. */
n3 = STAILQ_FIRST(&head);
STAILQ_REMOVE_HEAD(&head, entries);
free(n3);
					/* اجتياز أمامي. */
STAILQ_FOREACH(np, &head, entries)
	np-> ...
					/* اجتياز أمامي آمن. */
STAILQ_FOREACH_SAFE(np, &head, entries, np_temp) {
	np->do_stuff();
	...
	STAILQ_REMOVE(&head, np, entry, entries);
	free(np);
}
					/* حذف قائمة الذيل. */
while (!STAILQ_EMPTY(&head)) {
	n1 = STAILQ_FIRST(&head);
	STAILQ_REMOVE_HEAD(&head, entries);
	free(n1);
}
					/* حذف قائمة ذيل أسرع. */
n1 = STAILQ_FIRST(&head);
while (n1 != NULL) {
	n2 = STAILQ_NEXT(n1, entries);
	free(n1);
	n1 = n2;
}
STAILQ_INIT(&head);

قوائم

قائمة يرأسها هيكل معرف بواسطة الماكرو LIST_HEAD. يحتوي هذا الهيكل على مؤشر واحد إلى العنصر الأول في القائمة. العناصر مرتبطة بشكل مزدوج بحيث يمكن إزالة عنصر عشوائي دون اجتياز القائمة. يمكن إضافة عناصر جديدة إلى القائمة بعد عنصر موجود، أو قبل عنصر موجود، أو في بداية القائمة. يتم تعريف هيكل LIST_HEAD على النحو التالي:

LIST_HEAD(HEADNAME, TYPE) head;

حيث HEADNAME هو اسم الهيكل المراد تعريفه، و TYPE هو نوع العناصر المراد ربطها في القائمة. يمكن بعد ذلك تعريف مؤشر إلى رأس القائمة على النحو التالي:

struct HEADNAME *headp;

(الاسمين head و headp قابلان للاختيار من قبل المستخدم.)

الماكرو LIST_HEAD_INITIALIZER يُقيّم إلى مُهيئ للقائمة head.

الماكرو LIST_CONCAT يربط القائمة التي يرأسها head2 بنهاية القائمة التي يرأسها head1 ويزيل جميع الإدخالات من القائمة الأولى. يجب تجنب استخدام هذا الماكرو لأنه يجتاز القائمة بأكملها لـ head1. يجب استخدام قائمة ذيل إذا كان هذا الماكرو مطلوبًا في مسارات التعليمات البرمجية ذات الاستخدام العالي أو للعمل على قوائم طويلة.

الماكرو LIST_EMPTY يُقيّم إلى صحيح إذا لم تكن هناك عناصر في القائمة.

الماكرو LIST_ENTRY يُعرّف هيكلًا يربط العناصر في القائمة.

الماكرو LIST_FIRST يُرجع العنصر الأول في القائمة أو NULL إذا كانت القائمة فارغة.

الماكرو LIST_FOREACH يجتاز القائمة المشار إليها بواسطة

قوائم الذيل

قائمة الذيل يترأسها هيكل معرف بواسطة الماكرو TAILQ_HEAD. يحتوي هذا الهيكل على زوج من المؤشرات، واحد للعنصر الأول في قائمة الذيل والآخر للعنصر الأخير في قائمة الذيل. العناصر مرتبطة بروابط مزدوجة بحيث يمكن إزالة عنصر عشوائي دون اجتياز قائمة الذيل. يمكن إضافة عناصر جديدة إلى قائمة الذيل بعد عنصر موجود، قبل عنصر موجود، في رأس قائمة الذيل، أو في نهاية قائمة الذيل. يتم تعريف هيكل TAILQ_HEAD على النحو التالي:

TAILQ_HEAD(HEADNAME, TYPE) head;

حيث HEADNAME هو اسم الهيكل المراد تعريفه، و TYPE هو نوع العناصر التي سيتم ربطها في قائمة الذيل. يمكن لاحقًا تعريف مؤشر إلى رأس قائمة الذيل على النحو التالي:

struct HEADNAME *headp;

(الأسماء head و headp قابلة للاختيار من قبل المستخدم.)

الماكرو TAILQ_HEAD_INITIALIZER يقيم إلى مُهيئ لقائمة الذيل head.

الماكرو TAILQ_CONCAT يربط قائمة الذيل التي يترأسها head2 بنهاية القائمة التي يترأسها head1، ويزيل جميع الإدخالات من القائمة الأولى.

الماكرو TAILQ_EMPTY يقيم إلى صحيح إذا لم تكن هناك عناصر في قائمة الذيل.

الماكرو TAILQ_ENTRY يعرف هيكلًا يربط العناصر في قائمة الذيل.

الماكرو TAILQ_FIRST يعيد العنصر الأول في قائمة الذيل أو NULL إذا كانت قائمة الذيل فارغة.

الماكرو TAILQ_FOREACH يجتاز قائمة الذيل المشار إليها بواسطة head في الاتجاه الأمامي، ويعين كل عنصر بدوره إلى var. يتم تعيين var إلى NULL إذا اكتملت الحلقة بشكل طبيعي، أو إذا لم تكن هناك عناصر.

الماكرو TAILQ_FOREACH_FROM يعمل بنفس طريقة TAILQ_FOREACH عندما يكون var هو NULL، وإلا فإنه يعامل var كعنصر TAILQ تم العثور عليه سابقًا ويبدأ الحلقة عند var بدلاً من العنصر الأول في TAILQ المشار إليه بواسطة head.

الماكرو TAILQ_FOREACH_REVERSE يجتاز قائمة الذيل المشار إليها بواسطة head في الاتجاه العكسي، ويعين كل عنصر بدوره إلى var.

الماكرو TAILQ_FOREACH_REVERSE_FROM يعمل بنفس طريقة TAILQ_FOREACH_REVERSE عندما يكون var هو NULL، وإلا فإنه يعامل var كعنصر TAILQ تم العثور عليه سابقًا ويبدأ الحلقة العكسية عند var بدلاً من العنصر الأخير في TAILQ المشار إليه بواسطة head.

الماكرو TAILQ_FOREACH_SAFE و TAILQ_FOREACH_REVERSE_SAFE يجتازان القائمة المشار إليها بواسطة head في الاتجاه الأمامي أو العكسي على التوالي، ويعينان كل عنصر بدوره إلى var. ومع ذلك، على عكس نظيريهما غير الآمنين، فإن TAILQ_FOREACH و TAILQ_FOREACH_REVERSE يتيحان إمكانية إزالة var وكذلك تحريره من داخل الحلقة بأمان دون التأثير على الاجتياز.

الماكرو TAILQ_FOREACH_FROM_SAFE يعمل بنفس طريقة TAILQ_FOREACH_SAFE عندما يكون var هو NULL، وإلا فإنه يعامل var كعنصر TAILQ تم العثور عليه سابقًا ويبدأ الحلقة عند var بدلاً من العنصر الأول في TAILQ المشار إليه بواسطة head.

الماكرو TAILQ_FOREACH_REVERSE_FROM_SAFE يعمل بنفس طريقة TAILQ_FOREACH_REVERSE_SAFE عندما يكون var هو NULL، وإلا فإنه يعامل var كعنصر TAILQ تم العثور عليه سابقًا ويبدأ الحلقة العكسية عند var بدلاً من العنصر الأخير في TAILQ المشار إليه بواسطة head.

الماكرو TAILQ_INIT يقوم بتهيئة قائمة الذيل المشار إليها بواسطة head.

الماكرو TAILQ_INSERT_HEAD يدخل العنصر الجديد elm في رأس قائمة الذيل.

الماكرو TAILQ_INSERT_TAIL يدخل العنصر الجديد elm في نهاية قائمة الذيل.

الماكرو TAILQ_INSERT_AFTER يدخل العنصر الجديد elm بعد العنصر listelm.

الماكرو TAILQ_INSERT_BEFORE يدخل العنصر الجديد elm قبل العنصر listelm.

الماكرو TAILQ_LAST يعيد العنصر الأخير في قائمة الذيل. إذا كانت قائمة الذيل فارغة، تكون القيمة المعادة NULL.

الماكرو TAILQ_NEXT يعيد العنصر التالي في قائمة الذيل، أو NULL إذا كان هذا العنصر هو الأخير.

الماكرو TAILQ_PREV يعيد العنصر السابق في قائمة الذيل، أو NULL إذا كان هذا العنصر هو الأول.

الماكرو TAILQ_REMOVE يزيل العنصر elm من قائمة الذيل.

الماكرو TAILQ_SWAP يبادل محتويات head1 و head2.

التشخيص

عند تصحيح أخطاء queue(3)، قد يكون من المفيد تتبع تغييرات قائمة الانتظار. لتمكين التتبع، قم بتعريف الماكرو QUEUE_MACRO_DEBUG_TRACE في وقت الترجمة.

قد يكون من المفيد أيضًا إتلاف المؤشرات التي تم فصلها من قائمة انتظار، لاكتشاف الاستخدام بعد الإزالة. لتمكين إتلاف المؤشرات، قم بتعريف الماكرو QUEUE_MACRO_DEBUG_TRASH في وقت الترجمة. الماكرو QMD_IS_TRASHED(void *ptr) يعيد القيمة true إذا كان ptr قد تم إتلافه بواسطة خيار QUEUE_MACRO_DEBUG_TRASH.
