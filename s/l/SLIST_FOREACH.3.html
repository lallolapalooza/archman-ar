SLIST(3) دليل وظائف المكتبة SLIST(3)

الاسم

SLIST_EMPTY, SLIST_ENTRY, SLIST_FIRST, SLIST_FOREACH, SLIST_HEAD, SLIST_HEAD_INITIALIZER, SLIST_INIT, SLIST_INSERT_AFTER, SLIST_INSERT_HEAD, SLIST_NEXT, SLIST_REMOVE, SLIST_REMOVE_HEAD - تنفيذ قائمة مرتبطة أحاديًا

المكتبة

مكتبة C القياسية (libc, -lc)

الملخص

#include <sys/queue.h>
SLIST_ENTRY(TYPE);
SLIST_HEAD(HEADNAME, TYPE);
SLIST_HEAD SLIST_HEAD_INITIALIZER(SLIST_HEAD head);
void SLIST_INIT(SLIST_HEAD *head);
int SLIST_EMPTY(SLIST_HEAD *head);
void SLIST_INSERT_HEAD(SLIST_HEAD *head,
                        struct TYPE *elm, SLIST_ENTRY NAME);
void SLIST_INSERT_AFTER(struct TYPE *listelm,
                        struct TYPE *elm, SLIST_ENTRY NAME);
struct TYPE *SLIST_FIRST(SLIST_HEAD *head);
struct TYPE *SLIST_NEXT(struct TYPE *elm, SLIST_ENTRY NAME);
SLIST_FOREACH(struct TYPE *var, SLIST_HEAD *head, SLIST_ENTRY NAME);
void SLIST_REMOVE(SLIST_HEAD *head, struct TYPE *elm, TYPE,
                        SLIST_ENTRY NAME);
void SLIST_REMOVE_HEAD(SLIST_HEAD *head,
                        SLIST_ENTRY NAME);

الوصف

تعرف هذه الماكروات وتعمل على القوائم المرتبطة أحاديًا.

في تعريفات الماكرو، TYPE هو اسم هيكل معرف من قبل المستخدم، والذي يجب أن يحتوي على حقل من النوع SLIST_ENTRY، باسم NAME. الوسيط HEADNAME هو اسم هيكل معرف من قبل المستخدم والذي يجب تعريفه باستخدام الماكرو SLIST_HEAD().

الإنشاء

يتم رأس القائمة المرتبطة أحاديًا بواسطة هيكل معرف بواسطة الماكرو SLIST_HEAD(). يحتوي هذا الهيكل على مؤشر واحد للعنصر الأول في القائمة. يتم ربط العناصر بشكل أحادي لتقليل النفقات العامة للمساحة ومعالجة المؤشرات على حساب إزالة O(n) للعناصر العشوائية. يمكن إضافة عناصر جديدة إلى القائمة بعد عنصر موجود أو في رأس القائمة. يتم تعريف هيكل SLIST_HEAD على النحو التالي:


SLIST_HEAD(HEADNAME, TYPE) head;

حيث struct HEADNAME هو الهيكل المراد تعريفه، و struct TYPE هو نوع العناصر المراد ربطها في القائمة. يمكن لاحقًا الإعلان عن مؤشر لرأس القائمة على النحو التالي:


struct HEADNAME *headp;

(الأسماء head و headp قابلة للاختيار من قبل المستخدم.)

يعرف SLIST_ENTRY() هيكلًا يربط العناصر في القائمة.

يقيم SLIST_HEAD_INITIALIZER() قيمة ابتدائية للقائمة head.

يقوم SLIST_INIT() بتهيئة القائمة التي يشير إليها head.

يقيم SLIST_EMPTY() قيمة صحيحة إذا لم تكن هناك عناصر في القائمة.

الإدراج

يقوم SLIST_INSERT_HEAD() بإدراج العنصر الجديد elm في رأس القائمة.

يقوم SLIST_INSERT_AFTER() بإدراج العنصر الجديد elm بعد العنصر listelm.

التجوال

يعيد SLIST_FIRST() العنصر الأول في القائمة، أو NULL إذا كانت القائمة فارغة.

يعيد SLIST_NEXT() العنصر التالي في القائمة.

يقوم SLIST_FOREACH() بتجوال القائمة المشار إليها بواسطة head في الاتجاه الأمامي، مع تخصيص كل عنصر بدوره لـ var.

الإزالة

يقوم SLIST_REMOVE() بإزالة العنصر elm من القائمة.

يقوم SLIST_REMOVE_HEAD() بإزالة العنصر elm من رأس القائمة. لتحقيق أقصى قدر من الكفاءة، يجب على العناصر التي تتم إزالتها من رأس القائمة استخدام هذا الماكرو بشكل صريح بدلاً من SLIST_REMOVE() العام.

قيمة الإرجاع

يعيد SLIST_EMPTY() قيمة غير صفرية إذا كانت القائمة فارغة، وصفرًا إذا كانت القائمة تحتوي على إدخال واحد على الأقل.

يعيد SLIST_FIRST() و SLIST_NEXT() مؤشرًا إلى بنية TYPE الأولى أو التالية، على التوالي.

يعيد SLIST_HEAD_INITIALIZER() قيمة ابتدائية يمكن تعيينها لرأس القائمة head.

المعايير

BSD.

التاريخ

4.4BSD.

الأخطاء

لا يسمح SLIST_FOREACH() بإزالة أو تحرير var داخل الحلقة، لأنه سيتداخل مع عملية التجوال. SLIST_FOREACH_SAFE()، والذي يوجد في BSD ولكنه غير موجود في glibc، يصلح هذا القيد بالسماح بإزالة var بأمان من القائمة وتحريرها من داخل الحلقة دون التأثير على عملية التجوال.

أمثلة

#include <stddef.h>
#include <stdio.h>
#include <stdlib.h>
#include <sys/queue.h>
struct entry {
    int data;
    SLIST_ENTRY(entry) entries;             /* قائمة مرتبطة أحاديًا */
};
SLIST_HEAD(slisthead, entry);
int
main(void)
{
    struct entry *n1, *n2, *n3, *np;
    struct slisthead head;                  /* رأس القائمة المرتبطة أحاديًا */
    SLIST_INIT(&head);                      /* تهيئة الطابور */
    n1 = malloc(sizeof(struct entry));      /* الإدراج في الرأس */
    SLIST_INSERT_HEAD(&head, n1, entries);
    n2 = malloc(sizeof(struct entry));      /* الإدراج بعد */
    SLIST_INSERT_AFTER(n1, n2, entries);
    SLIST_REMOVE(&head, n2, entry, entries);/* الحذف */
    free(n2);
    n3 = SLIST_FIRST(&head);
    SLIST_REMOVE_HEAD(&head, entries);      /* الحذف من الرأس */
    free(n3);
    for (unsigned int i = 0; i < 5; i++) {
        n1 = malloc(sizeof(struct entry));
        SLIST_INSERT_HEAD(&head, n1, entries);
        n1->data = i;
    }
                                            /* التجوال الأمامي */
    SLIST_FOREACH(np, &head, entries)
        printf("%i\n", np->data);
    while (!SLIST_EMPTY(&head)) {           /* حذف القائمة */
        n1 = SLIST_FIRST(&head);
        SLIST_REMOVE_HEAD(&head, entries);
        free(n1);
    }
    SLIST_INIT(&head);
    exit(EXIT_SUCCESS);
}
