توثيق الوظيفة/الروتين الفرعي

subroutine slaqr1 (integer n, real, dimension( ldh, * ) h, integer ldh, real sr1, real si1, real sr2, real si2, real, dimension( * ) v)

SLAQR1 يضبط مضاعفًا قياسيًا للعمود الأول من حاصل ضرب مصفوفة 2x2 أو 3x3 H والإزاحات المحددة.

الغرض:

!>
!>      بالنظر إلى مصفوفة H بحجم 2x2 أو 3x3، يضبط SLAQR1 المتجه v ليكون مضاعفًا
!>      قياسيًا للعمود الأول من حاصل الضرب
!>
!>      (*)  K = (H - (sr1 + i*si1)*I)*(H - (sr2 + i*si2)*I)
!>
!>      مع تحجيم لتجنب الفائض ومعظم النقصان. يُفترض أن إحدى الحالتين التاليتين
!>      تتحقق:
!>
!>              1) sr1 = sr2 و si1 = -si2
!>          أو
!>              2) si1 = si2 = 0.
!>
!>      هذا مفيد لبدء نتوءات الإزاحة المزدوجة الضمنية في خوارزمية QR.
!>

المعاملات

N 
!>          N هو عدد صحيح (INTEGER)
!>              رتبة المصفوفة H. يجب أن تكون N إما 2 أو 3.
!>

H

!>          H هي مصفوفة حقيقية (REAL array)، ذات بعد (LDH,N)
!>              المصفوفة H بحجم 2x2 أو 3x3 في (*).
!>

LDH

!>          LDH هو عدد صحيح (INTEGER)
!>              البعد الرائد للمصفوفة H كما هو معلن في
!>              الإجراء المستدعي.  LDH >= N
!>

SR1

!>          SR1 هو عدد حقيقي (REAL)
!>

SI1

!>          SI1 هو عدد حقيقي (REAL)
!>

SR2

!>          SR2 هو عدد حقيقي (REAL)
!>

SI2

!>          SI2 هو عدد حقيقي (REAL)
!>              الإزاحات في (*).
!>

V

!>          V هي مصفوفة حقيقية (REAL array)، ذات بعد (N)
!>              مضاعف قياسي للعمود الأول للمصفوفة K في (*).
!>
