subroutine slar1v (integer n, integer b1, integer bn, real lambda, real, dimension( * ) d, real, dimension( * ) l, real, dimension( * ) ld, real, dimension( * ) lld, real pivmin, real gaptol, real, dimension( * ) z, logical wantnc, integer negcnt, real ztz, real mingma, integer r, integer, dimension( * ) isuppz, real nrminv, real resid, real rqcorr, real, dimension( * ) work)

SLAR1V يحسب العمود (المُقاس) r لمصفوفة معكوس المصفوفة الجزئية في الصفوف b1 إلى bn للمصفوفة الثلاثية LDLT - λI.

الغرض:

!>
!> SLAR1V يحسب العمود (المُقاس) r لمصفوفة معكوس المصفوفة الجزئية
!> في الصفوف B1 إلى BN للمصفوفة الثلاثية
!> L D L**T - sigma I. عندما تكون sigma قريبة من قيمة ذاتية،
!> يكون المتجه المحسوب متجهًا ذاتيًا دقيقًا. عادةً، يتوافق r
!> مع الفهرس حيث يكون المتجه الذاتي هو الأكبر من حيث القيمة المطلقة.
!> الخطوات التالية تحقق هذه الحسابات:
!> (أ) تحويل qd الثابت، L D L**T - sigma I = L(+) D(+) L(+)**T،
!> (ب) تحويل qd التقدمي، L D L**T - sigma I = U(-) D(-) U(-)**T،
!> (ج) حساب العناصر القطرية لمعكوس
!>     L D L**T - sigma I عن طريق دمج التحويلات المذكورة أعلاه، واختيار
!>     r كفهرس حيث يكون قطر المعكوس (واحد من)
!>     الأكبر من حيث القيمة المطلقة.
!> (د) حساب العمود (المُقاس) r للمعكوس باستخدام
!>     التحليل الملتوي الذي تم الحصول عليه عن طريق دمج الجزء العلوي من
!>     التحويل الثابت والجزء السفلي من التحويل التقدمي.
!>

المعلمات

N 
!>          N هو عدد صحيح
!>           رتبة المصفوفة L D L**T.
!>

B1

!>          B1 هو عدد صحيح
!>           فهرس البداية للمصفوفة الجزئية L D L**T.
!>

BN

!>          BN هو عدد صحيح
!>           فهرس النهاية للمصفوفة الجزئية L D L**T.
!>

LAMBDA

!>          LAMBDA هو عدد حقيقي
!>           الإزاحة. لحساب متجه ذاتي دقيق،
!>           يجب أن تكون LAMBDA تقريبًا جيدًا لقيمة ذاتية
!>           لـ L D L**T.
!>

L

!>          L هو مصفوفة حقيقية، ذات بُعد (N-1)
!>           العناصر الفرعية (n-1) للمصفوفة الأحادية ثنائية القطر
!>           L، في العناصر من 1 إلى N-1.
!>

D

!>          D هو مصفوفة حقيقية، ذات بُعد (N)
!>           العناصر القطرية n للمصفوفة القطرية D.
!>

LD

!>          LD هو مصفوفة حقيقية، ذات بُعد (N-1)
!>           العناصر L(i)*D(i) البالغ عددها n-1.
!>

LLD

!>          LLD هو مصفوفة حقيقية، ذات بُعد (N-1)
!>           العناصر L(i)*L(i)*D(i) البالغ عددها n-1.
!>

PIVMIN

!>          PIVMIN هو عدد حقيقي
!>           أصغر محور في متتالية ستورم.
!>

GAPTOL

!>          GAPTOL هو عدد حقيقي
!>           المسامحة التي تشير إلى متى تكون مدخلات المتجه الذاتي ضئيلة
!>           بالمقارنة مع مساهمتها في الباقي.
!>

Z

!>          Z هي مصفوفة حقيقية، ذات بُعد (N)
!>           عند الإدخال، يجب تعيين جميع مدخلات Z إلى 0.
!>           عند الإخراج، تحتوي Z على العمود (المُقاس) r للمعكوس.
!>           المقياس بحيث يكون Z(R) يساوي 1.
!>

WANTNC

!>          WANTNC هو منطقي
!>           يحدد ما إذا كان يجب حساب NEGCNT.
!>

NEGCNT

!>          NEGCNT هو عدد صحيح
!>           إذا كان WANTNC هو .TRUE. فإن NEGCNT = عدد المحاور < pivmin
!>           في تحليل المصفوفة L D L**T، و NEGCNT = -1 خلاف ذلك.
!>

ZTZ

!>          ZTZ هو عدد حقيقي
!>           مربع المعيار 2 للمتجه Z.
!>

MINGMA

!>          MINGMA هو عدد حقيقي
!>           مقلوب العنصر القطري الأكبر (من حيث القيمة المطلقة)
!>           لمعكوس L D L**T - sigma I.
!>

R

!>          R هو عدد صحيح
!>           فهرس الالتواء للتحليل الملتوي المستخدم ل
!>           حساب Z.
!>           عند الإدخال، 0 <= R <= N. إذا تم إدخال R كـ 0، يتم تعيين R إلى
!>           الفهرس حيث يكون (L D L**T - sigma I)^{-1} هو الأكبر
!>           من حيث القيمة المطلقة. إذا كان 1 <= R <= N، يظل R دون تغيير.
!>           عند الإخراج، يحتوي R على فهرس الالتواء المستخدم لحساب Z.
!>           من الناحية المثالية، يحدد R موقع أكبر مدخل في
!>           المتجه الذاتي.
!>

ISUPPZ

!>          ISUPPZ هي مصفوفة عدد صحيح، ذات بُعد (2)
!>           دعم المتجه في Z، أي أن المتجه Z
!>           غير صفري فقط في العناصر من ISUPPZ(1) إلى ISUPPZ( 2 ).
!>

NRMINV

!>          NRMINV هو عدد حقيقي
!>           NRMINV = 1/SQRT( ZTZ )
!>

RESID

!>          RESID هو عدد حقيقي
!>           باقي المتجه النقطي العائم.
!>           RESID = ABS( MINGMA )/SQRT( ZTZ )
!>

RQCORR

!>          RQCORR هو عدد حقيقي
!>           تصحيح حاصل قسمة رايلي لـ LAMBDA.
!>           RQCORR = MINGMA*TMP
!>

WORK

!>          WORK هي مصفوفة حقيقية، ذات بُعد (4*N)
!>
