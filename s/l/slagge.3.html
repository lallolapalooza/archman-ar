وثائق الوظيفة/الإجراء الفرعي

subroutine slagge (integer m, integer n, integer kl, integer ku, real, dimension( * ) d, real, dimension( lda, * ) a, integer lda, integer, dimension( 4 ) iseed, real, dimension( * ) work, integer info)

SLAGGE

الغرض:

!>
!> SLAGGE تنشئ مصفوفة حقيقية عامة m × n A، عن طريق الضرب المسبق واللاحق
!> لمصفوفة قطرية حقيقية D بمصفوفات متعامدة عشوائية:
!> A = U*D*V. قد يتم تقليل النطاقين السفلي والعلوي بعد ذلك إلى
!> kl و ku عن طريق تحويلات متعامدة إضافية.
!>

المعاملات

M 
!>          M هو عدد صحيح (INTEGER)
!>          عدد الصفوف في المصفوفة A. M >= 0.
!>

N

!>          N هو عدد صحيح (INTEGER)
!>          عدد الأعمدة في المصفوفة A. N >= 0.
!>

KL

!>          KL هو عدد صحيح (INTEGER)
!>          عدد الأقطار الفرعية غير الصفرية داخل نطاق A.
!>          0 <= KL <= M-1.
!>

KU

!>          KU هو عدد صحيح (INTEGER)
!>          عدد الأقطار فوق الصفرية داخل نطاق A.
!>          0 <= KU <= N-1.
!>

D

!>          D هو مصفوفة حقيقية (REAL array)، البعد (min(M,N))
!>          عناصر القطر للمصفوفة القطرية D.
!>

A

!>          A هي مصفوفة حقيقية (REAL array)، البعد (LDA,N)
!>          المصفوفة m × n المتولدة A.
!>

LDA

!>          LDA هو عدد صحيح (INTEGER)
!>          البعد الرئيسي للمصفوفة A. LDA >= M.
!>

ISEED

!>          ISEED هي مصفوفة أعداد صحيحة (INTEGER array)، البعد (4)
!>          عند الإدخال، بذرة مولد الأرقام العشوائية؛ يجب أن تكون عناصر المصفوفة
!>          بين 0 و 4095، ويجب أن يكون ISEED(4) فردياً.
!>          عند الخروج، يتم تحديث البذرة.
!>

WORK

!>          WORK هي مصفوفة حقيقية (REAL array)، البعد (M+N)
!>

INFO

!>          INFO هو عدد صحيح (INTEGER)
!>          = 0: خروج ناجح
!>          < 0: إذا كان INFO = -i، فإن المعامل i كان له قيمة غير قانونية
!>
