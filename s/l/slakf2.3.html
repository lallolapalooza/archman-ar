subroutine slakf2 (integer m, integer n, real, dimension( lda, * ) a, integer lda, real, dimension( lda, * ) b, real, dimension( lda, * ) d, real, dimension( lda, * ) e, real, dimension( ldz, * ) z, integer ldz)

SLAKF2

الغرض:

!>
!> تشكيل المصفوفة 2*M*N × 2*M*N
!>
!>        Z = [ kron(In, A)  -kron(B', Im) ]
!>            [ kron(In, D)  -kron(E', Im) ],
!>
!> حيث In هي مصفوفة الوحدة بالحجم n و X' هو منقول X. kron(X, Y) هو حاصل ضرب كرونيكر بين المصفوفتين X
!> و Y.
!>

المعاملات

M 
!>          M is INTEGER
!>          حجم المصفوفة، يجب أن يكون >= 1.
!>

N

!>          N is INTEGER
!>          حجم المصفوفة، يجب أن يكون >= 1.
!>

A

!>          A is REAL, dimension ( LDA, M )
!>          المصفوفة A في المصفوفة الناتجة Z.
!>

LDA

!>          LDA is INTEGER
!>          البعد الرائد للمصفوفتين A، B، D، و E. ( LDA >= M+N )
!>

B

!>          B is REAL, dimension ( LDA, N )
!>

D

!>          D is REAL, dimension ( LDA, M )
!>

E

!>          E is REAL, dimension ( LDA, N )
!>
!>          المصفوفات المستخدمة في تشكيل المصفوفة الناتجة Z.
!>

Z

!>          Z is REAL, dimension ( LDZ, 2*M*N )
!>          المصفوفة الناتجة كرونيكر M*N*2 × M*N*2 (انظر أعلاه.)
!>

LDZ

!>          LDZ is INTEGER
!>          البعد الرائد للمصفوفة Z. ( LDZ >= 2*M*N )
!>
