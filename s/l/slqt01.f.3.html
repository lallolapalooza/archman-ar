توثيق الدالة/الإجراء الفرعي

subroutine slqt01 (integer m, integer n, real, dimension( lda, * ) a, real, dimension( lda, * ) af, real, dimension( lda, * ) q, real, dimension( lda, * ) l, integer lda, real, dimension( * ) tau, real, dimension( lwork ) work, integer lwork, real, dimension( * ) rwork, real, dimension( * ) result)

SLQT01

الغرض:

!>
!> SLQT01 تختبر SGELQF، التي تحسب تحليل LQ لمصفوفة m-by-n A، وتختبر جزئيًا SORGLQ
!> التي تشكل المصفوفة المتعامدة n-by-n Q.
!>
!> SLQT01 تقارن L بـ A*Q'، وتتحقق من أن Q متعامدة.
!>

المعلمات:

M 
!>          M هو عدد صحيح (INTEGER)
!>          عدد صفوف المصفوفة A. M >= 0.
!>

N

!>          N هو عدد صحيح (INTEGER)
!>          عدد أعمدة المصفوفة A. N >= 0.
!>

A

!>          A هي مصفوفة REAL، البعد (LDA,N)
!>          المصفوفة m-by-n A.
!>

AF

!>          AF هي مصفوفة REAL، البعد (LDA,N)
!>          تفاصيل تحليل LQ للمصفوفة A، كما تم إرجاعها بواسطة SGELQF.
!>          راجع SGELQF لمزيد من التفاصيل.
!>

Q

!>          Q هي مصفوفة REAL، البعد (LDA,N)
!>          المصفوفة المتعامدة n-by-n Q.
!>

L

!>          L هي مصفوفة REAL، البعد (LDA,max(M,N))
!>

LDA

!>          LDA هو عدد صحيح (INTEGER)
!>          البعد الرائد للمصفوفات A، AF، Q و L.
!>          LDA >= max(M,N).
!>

TAU

!>          TAU هي مصفوفة REAL، البعد (min(M,N))
!>          العوامل القياسية للمنعكسات الأولية، كما تم إرجاعها
!>          بواسطة SGELQF.
!>

WORK

!>          WORK هي مصفوفة REAL، البعد (LWORK)
!>

LWORK

!>          LWORK هو عدد صحيح (INTEGER)
!>          بعد المصفوفة WORK.
!>

RWORK

!>          RWORK هي مصفوفة REAL، البعد (max(M,N))
!>

RESULT

!>          RESULT هي مصفوفة REAL، البعد (2)
!>          نسب الاختبار:
!>          RESULT(1) = norm( L - A*Q' ) / ( N * norm(A) * EPS )
!>          RESULT(2) = norm( I - Q*Q' ) / ( N * EPS )
!>
