subroutine slalsd (uplo, smlsiz, n, nrhs, d, e, b, ldb, rcond, rank, work, iwork, info)
SLALSD يستخدم التحلل القيمي المفرد (singular value decomposition) لحل مشكلة المربعات الصغرى.

توثيق الوظيفة/الروتين الفرعي

subroutine slalsd (character uplo, integer smlsiz, integer n, integer nrhs, real, dimension( * ) d, real, dimension( * ) e, real, dimension( ldb, * ) b, integer ldb, real rcond, integer rank, real, dimension( * ) work, integer, dimension( * ) iwork, integer info)

SLALSD يستخدم التحلل القيمي المفرد (singular value decomposition) لحل مشكلة المربعات الصغرى.

الغرض:

!>
!> SLALSD يستخدم التحلل القيمي المفرد (singular value decomposition) لحل
!> مشكلة المربعات الصغرى لإيجاد X التي تقلل من المعيار الإقليدي لكل
!> عمود من A*X-B، حيث A هي مصفوفة عليا ثنائية القطر بحجم N*N، و X و B
!> هما بحجم N*NRHS. الحل X يحل محل B.
!>
!> يتم التعامل مع القيم المفردة للمصفوفة A الأصغر من RCOND مضروبة في أكبر
!> قيمة مفردة على أنها صفر عند حل مشكلة المربعات الصغرى؛ في هذه الحالة يتم
!> إرجاع حل الحد الأدنى للمعايير.
!> يتم إرجاع القيم المفردة الفعلية في D بترتيب تصاعدي.
!>
!>

المعلمات

UPLO 
!>          UPLO is CHARACTER*1
!>         = 'U': D و E تحددان مصفوفة ثنائية القطر عليا.
!>         = 'L': D و E تحددان مصفوفة ثنائية القطر سفلى.
!>

SMLSIZ

!>          SMLSIZ is INTEGER
!>         أقصى حجم للمشاكل الفرعية في أسفل شجرة الحساب.
!>

N

!>          N is INTEGER
!>         بعد المصفوفة ثنائية القطر. N >= 0.
!>

NRHS

!>          NRHS is INTEGER
!>         عدد أعمدة B. يجب أن يكون NRHS على الأقل 1.
!>

D

!>          D is REAL array, dimension (N)
!>         عند الإدخال تحتوي D على القطر الرئيسي للمصفوفة ثنائية القطر.
!>         عند الخروج، إذا كان INFO = 0، تحتوي D على القيم المفردة الخاصة بها.
!>

E

!>          E is REAL array, dimension (N-1)
!>         تحتوي على المدخلات فوق القطرية للمصفوفة ثنائية القطر.
!>         عند الخروج، تم تدمير E.
!>

B

!>          B is REAL array, dimension (LDB,NRHS)
!>         عند الإدخال، تحتوي B على الأطراف اليمنى لمشكلة المربعات الصغرى.
!>         عند الخروج، تحتوي B على الحل X.
!>

LDB

!>          LDB is INTEGER
!>         البعد الرائد لـ B في البرنامج الفرعي المستدعي.
!>         يجب أن يكون LDB على الأقل max(1,N).
!>

RCOND

!>          RCOND is REAL
!>         القيم المفردة للمصفوفة A الأقل من أو تساوي RCOND مضروبة في
!>         أكبر قيمة مفردة يتم التعامل معها على أنها صفر عند حل مشكلة
!>         المربعات الصغرى. إذا كان RCOND سالبًا،
!>         يتم استخدام الدقة الآلية بدلاً من ذلك.
!>         على سبيل المثال، إذا كانت diag(S)*X=B هي مشكلة المربعات الصغرى،
!>         حيث diag(S) هي مصفوفة قطرية من القيم المفردة، فإن
!>         الحل سيكون X(i) = B(i) / S(i) إذا كانت S(i) أكبر من
!>         RCOND*max(S)، و X(i) = 0 إذا كانت S(i) أقل من أو تساوي
!>         RCOND*max(S).
!>

RANK

!>          RANK is INTEGER
!>         عدد القيم المفردة للمصفوفة A الأكبر من RCOND مضروبة في
!>         أكبر قيمة مفردة.
!>

WORK

!>          WORK is REAL array, dimension at least
!>         (9*N + 2*N*SMLSIZ + 8*N*NLVL + N*NRHS + (SMLSIZ+1)**2),
!>         حيث NLVL = max(0, INT(log_2 (N/(SMLSIZ+1))) + 1).
!>

IWORK

!>          IWORK is INTEGER array, dimension at least
!>         (3*N*NLVL + 11*N)
!>

INFO

!>          INFO is INTEGER
!>         = 0:  خروج ناجح.
!>         < 0:  إذا كان INFO = -i، كان الوسيط i ذو قيمة غير قانونية.
!>         > 0:  فشل الخوارزمية في حساب قيمة مفردة أثناء العمل على
!>               المصفوفة الفرعية الموجودة في الصفوف والأعمدة
!>               INFO/(N+1) إلى MOD(INFO,N+1).
!>
