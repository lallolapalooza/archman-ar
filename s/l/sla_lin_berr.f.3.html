subroutine sla_lin_berr (integer n, integer nz, integer nrhs, real, dimension( n, nrhs ) res, real, dimension( n, nrhs ) ayb, real, dimension( nrhs ) berr)

SLA_LIN_BERR تحسب خطأ رجعي نسبي مكون بمكون.

الغرض:

!>
!>    SLA_LIN_BERR تحسب الخطأ الرجعي النسبي المكون بمكون من
!>    الصيغة
!>        max(i) ( abs(R(i)) / ( abs(op(A_s))*abs(Y) + abs(B_s) )(i) )
!>    حيث abs(Z) هي القيمة المطلقة المكونة بمكون للمصفوفة
!>    أو المتجه Z.
!>

المعاملات

N 
!>          N هو عدد صحيح (INTEGER)
!>     عدد المعادلات الخطية، أي، رتبة
!>     المصفوفة A. N >= 0.
!>

NZ

!>          NZ هو عدد صحيح (INTEGER)
!>     نضيف (NZ+1)*SLAMCH( 'Safe minimum' ) إلى R(i) في البسط لتجنب
!>     البقايا الصفرية بشكل خاطئ. القيمة الافتراضية هي N.
!>

NRHS

!>          NRHS هو عدد صحيح (INTEGER)
!>     عدد الأطراف اليمنى، أي، عدد أعمدة
!>     المصفوفتين AYB و RES و BERR. NRHS >= 0.
!>

RES

!>          RES هي مصفوفة حقيقية (REAL array)، أبعادها (N,NRHS)
!>     مصفوفة البقايا، أي، المصفوفة R في صيغة الخطأ الرجعي النسبي أعلاه.
!>

AYB

!>          AYB هي مصفوفة حقيقية (REAL array)، أبعادها (N, NRHS)
!>     المقام في صيغة الخطأ الرجعي النسبي أعلاه، أي،
!>     المصفوفة abs(op(A_s))*abs(Y) + abs(B_s). المصفوفات A و Y و B
!>     تأتي من التحسين التكراري (انظر sla_gerfsx_extended.f).
!>

BERR

!>          BERR هي مصفوفة حقيقية (REAL array)، أبعادها (NRHS)
!>     الخطأ الرجعي النسبي المكون بمكون من الصيغة أعلاه.
!>
