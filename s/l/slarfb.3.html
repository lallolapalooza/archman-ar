subroutine slarfb (side, trans, direct, storev, m, n, k, v, ldv, t, ldt, c, ldc, work, ldwork)
SLARFB يطبق عاكس كتلة أو معكوسه على مصفوفة مستطيلة عامة.

توثيق الوظيفة/الروتين الفرعي

subroutine slarfb (character side, character trans, character direct, character storev, integer m, integer n, integer k, real, dimension( ldv, * ) v, integer ldv, real, dimension( ldt, * ) t, integer ldt, real, dimension( ldc, * ) c, integer ldc, real, dimension( ldwork, * ) work, integer ldwork)

SLARFB يطبق عاكس كتلة حقيقي H أو معكوسه H**T على مصفوفة مستطيلة حقيقية C، من اليسار أو اليمين.

الغرض:

!>
!> SLARFB يطبق عاكس كتلة حقيقي H أو معكوسه H**T على مصفوفة مستطيلة حقيقية C،
!> من اليسار أو اليمين.
!>

المعلمات

SIDE 
!>          SIDE هو CHARACTER*1
!>          = 'L': تطبيق H أو H**T من اليسار
!>          = 'R': تطبيق H أو H**T من اليمين
!>

TRANS

!>          TRANS هو CHARACTER*1
!>          = 'N': تطبيق H (بدون تبديل)
!>          = 'T': تطبيق H**T (تبديل)
!>

DIRECT

!>          DIRECT هو CHARACTER*1
!>          يشير إلى كيفية تكوين H من حاصل ضرب انعكاسات أولية
!>          = 'F': H = H(1) H(2) . . . H(k) (للأمام)
!>          = 'B': H = H(k) . . . H(2) H(1) (للخلف)
!>

STOREV

!>          STOREV هو CHARACTER*1
!>          يشير إلى كيفية تخزين المتجهات التي تحدد الانعكاسات الأولية:
!>          = 'C': عمودياً
!>          = 'R': صفياً
!>

M

!>          M هو INTEGER
!>          عدد صفوف المصفوفة C.
!>

N

!>          N هو INTEGER
!>          عدد أعمدة المصفوفة C.
!>

K

!>          K هو INTEGER
!>          ترتيب المصفوفة T (= عدد الانعكاسات الأولية التي يشكل حاصل ضربها
!>          العاكس الكتلي).
!>          إذا كانت SIDE = 'L'، فإن M >= K >= 0؛
!>          إذا كانت SIDE = 'R'، فإن N >= K >= 0.
!>

V

!>          V هو مصفوفة REAL، البعد
!>                                (LDV,K) إذا كانت STOREV = 'C'
!>                                (LDV,M) إذا كانت STOREV = 'R' و SIDE = 'L'
!>                                (LDV,N) إذا كانت STOREV = 'R' و SIDE = 'R'
!>          المصفوفة V. انظر التفاصيل الإضافية.
!>

LDV

!>          LDV هو INTEGER
!>          البعد الرائد للمصفوفة V.
!>          إذا كانت STOREV = 'C' و SIDE = 'L'، فإن LDV >= max(1,M)؛
!>          إذا كانت STOREV = 'C' و SIDE = 'R'، فإن LDV >= max(1,N)؛
!>          إذا كانت STOREV = 'R'، فإن LDV >= K.
!>

T

!>          T هي مصفوفة REAL، البعد (LDT,K)
!>          المصفوفة المثلثية k × k، T، في تمثيل العاكس الكتلي.
!>

LDT

!>          LDT هو INTEGER
!>          البعد الرائد للمصفوفة T. LDT >= K.
!>

C

!>          C هي مصفوفة REAL، البعد (LDC,N)
!>          عند الإدخال، المصفوفة C ذات البعد m × n.
!>          عند الخروج، يتم الكتابة فوق C بواسطة H*C أو H**T*C أو C*H أو C*H**T.
!>

LDC

!>          LDC هو INTEGER
!>          البعد الرائد للمصفوفة C. LDC >= max(1,M).
!>

WORK

!>          WORK هي مصفوفة REAL، البعد (LDWORK,K)
!>

LDWORK

!>          LDWORK هو INTEGER
!>          البعد الرائد للمصفوفة WORK.
!>          إذا كانت SIDE = 'L'، فإن LDWORK >= max(1,N)؛
!>          إذا كانت SIDE = 'R'، فإن LDWORK >= max(1,M).
!>

تفاصيل إضافية:

!>
!>  شكل المصفوفة V وتخزين المتجهات التي تحدد H(i) موضح بشكل أفضل
!>  بالمثال التالي مع n = 5 و k = 3. الجزء المثلثي من V (بما في ذلك قطره)
!>  غير مستخدم.
!>
!>  DIRECT = 'F' و STOREV = 'C':         DIRECT = 'F' و STOREV = 'R':
!>
!>               V = (  1       )                 V = (  1 v1 v1 v1 v1 )
!>                   ( v1  1    )                     (     1 v2 v2 v2 )
!>                   ( v1 v2  1 )                     (        1 v3 v3 )
!>                   ( v1 v2 v3 )
!>                   ( v1 v2 v3 )
!>
!>  DIRECT = 'B' و STOREV = 'C':         DIRECT = 'B' و STOREV = 'R':
!>
!>               V = ( v1 v2 v3 )                 V = ( v1 v1  1       )
!>                   ( v1 v2 v3 )                     ( v2 v2 v2  1    )
!>                   (  1 v2 v3 )                     ( v3 v3 v3 v3  1 )
!>                   (     1 v3 )
!>                   (        1 )
!>
