subroutine slasd0 (n, sqre, d, e, u, ldu, vt, ldvt, smlsiz, iwork, work, info)
SLASD0 تحسب القيم المنفردة لمصفوفة B فوق قطرية حقيقية بحجم n-by-m مع القطر d والمصفوفات الخارجية e. تستخدم بواسطة sbdsdc.

وثائق الوظيفة/الروتين الفرعي

subroutine slasd0 (integer n, integer sqre, real, dimension( * ) d, real, dimension( * ) e, real, dimension( ldu, * ) u, integer ldu, real, dimension( ldvt, * ) vt, integer ldvt, integer smlsiz, integer, dimension( * ) iwork, real, dimension( * ) work, integer info)

SLASD0 تحسب القيم المنفردة لمصفوفة B فوق قطرية حقيقية بحجم n-by-m مع القطر d والمصفوفات الخارجية e. تستخدم بواسطة sbdsdc.

الغرض:

!>
!> باستخدام نهج الانقسام والغزو، تحسب SLASD0 تفكيك القيم المنفردة (SVD)
!> لمصفوفة B فوق قطرية حقيقية من حجم N-by-M مع قطر D ومصفوفات خارجية E، حيث M = N + SQRE.
!> تحسب الخوارزمية مصفوفتين متعامدتين U و VT بحيث B = U * S * VT.
!> يتم الكتابة فوق القيم المنفردة S على D.
!>
!> روتين فرعي مشابه، SLASDA، يحسب فقط القيم المنفردة،
!> واختياريًا، المتجهات المنفردة بشكل مضغوط.
!>

المعلمات

N 
!>          N هو عدد صحيح (INTEGER)
!>         عند الدخول، بعد البعد الصفوف للمصفوفة فوق القطرية.
!>         هذا هو أيضًا بعد مصفوفة القطر الرئيسي D.
!>

SQRE

!>          SQRE هو عدد صحيح (INTEGER)
!>         يحدد بعد الأعمدة للمصفوفة فوق القطرية.
!>         = 0: تحتوي المصفوفة فوق القطرية على بعد أعمدة M = N؛
!>         = 1: تحتوي المصفوفة فوق القطرية على بعد أعمدة M = N+1؛
!>

D

!>          D هي مصفوفة حقيقية (REAL array)، حجمها (N)
!>         عند الدخول، تحتوي D على القطر الرئيسي للمصفوفة فوق القطرية.
!>         عند الخروج، إذا كان INFO = 0، تحتوي D على القيم المنفردة.
!>

E

!>          E هي مصفوفة حقيقية (REAL array)، حجمها (M-1)
!>         تحتوي على مدخلات القطر الفرعي للمصفوفة فوق القطرية.
!>         عند الخروج، تم تدمير E.
!>

U

!>          U هي مصفوفة حقيقية (REAL array)، حجمها (LDU, N)
!>         عند الخروج، تحتوي U على المتجهات المنفردة اليسرى،
!>          إذا تم تمرير U على أنها مصفوفة الوحدة (N, N).
!>

LDU

!>          LDU هو عدد صحيح (INTEGER)
!>         عند الدخول، البعد الرائد لـ U.
!>

VT

!>          VT هي مصفوفة حقيقية (REAL array)، حجمها (LDVT, M)
!>         عند الخروج، تحتوي VT**T على المتجهات المنفردة اليمنى،
!>          إذا تم تمرير VT على أنها مصفوفة الوحدة (M, M).
!>

LDVT

!>          LDVT هو عدد صحيح (INTEGER)
!>         عند الدخول، البعد الرائد لـ VT.
!>

SMLSIZ

!>          SMLSIZ هو عدد صحيح (INTEGER)
!>         عند الدخول، الحد الأقصى لحجم المشاكل الفرعية في
!>         أسفل شجرة الحساب.
!>

IWORK

!>          IWORK هي مصفوفة أعداد صحيحة (INTEGER array)، حجمها (8*N)
!>

WORK

!>          WORK هي مصفوفة حقيقية (REAL array)، حجمها (3*M**2+2*M)
!>

INFO

!>          INFO هو عدد صحيح (INTEGER)
!>          = 0:  انتهاء ناجح.
!>          < 0:  إذا كان INFO = -i، فإن المعلمة i كانت لها قيمة غير قانونية.
!>          > 0:  إذا كان INFO = 1، لم تتقارب قيمة منفردة.
!>
