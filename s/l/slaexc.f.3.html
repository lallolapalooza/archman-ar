subroutine slaexc (wantq, n, t, ldt, q, ldq, j1, n1, n2, work, info)
SLAEXC تبادل الكتل القطرية المتجاورة لمصفوفة حقيقية شبه قطرية علوية في الشكل القانوني لشور، عن طريق تحويل تشابه متعامد.

توثيق الوظيفة/الإجراء الفرعي

subroutine slaexc (logical wantq, integer n, real, dimension( ldt, * ) t, integer ldt, real, dimension( ldq, * ) q, integer ldq, integer j1, integer n1, integer n2, real, dimension( * ) work, integer info)

SLAEXC تبادل الكتل القطرية المتجاورة T11 و T22 من الرتبة 1 أو 2 في مصفوفة حقيقية شبه قطرية علوية T في الشكل القانوني لشور، عن طريق تحويل تشابه متعامد.

الغرض:

!>
!> SLAEXC تبادل الكتل القطرية المتجاورة T11 و T22 من الرتبة 1 أو 2 في
!> مصفوفة شبه قطرية علوية T عن طريق تحويل تشابه متعامد.
!>
!> يجب أن تكون T في الشكل القانوني لشور، أي، كتلة مثلثة علوية
!> مع كتل قطرية 1-بواسطة-1 و 2-بواسطة-2؛ كل كتلة قطرية 2-بواسطة-2
!> لها عناصرها القطرية متساوية وعناصرها غير القطرية ذات
!> إشارات متعاكسة.
!>

المعلمات

WANTQ 
!>          WANTQ هو LOGICAL
!>          = .TRUE. : تجميع التحويل في المصفوفة Q؛
!>          = .FALSE.: لا تجمع التحويل.
!>

N

!>          N هو INTEGER
!>          رتبة المصفوفة T. N >= 0.
!>

T

!>          T هو مصفوفة REAL، البعد (LDT,N)
!>          عند الدخول، المصفوفة شبه القطرية العلوية T، في الشكل القانوني
!>          لـ Schur.
!> عند الخروج، المصفوفة المحدثة T، مرة أخرى في الشكل القانوني لشور.
!>

LDT

!>          LDT هو INTEGER
!>          البعد الرائد للمصفوفة T. LDT >= max(1,N).
!>

Q

!>          Q هو مصفوفة REAL، البعد (LDQ,N)
!>          عند الدخول، إذا كانت WANTQ هي .TRUE.، المصفوفة المتعامدة Q.
!>          عند الخروج، إذا كانت WANTQ هي .TRUE.، المصفوفة المحدثة Q.
!>          إذا كانت WANTQ هي .FALSE.، لا يتم الاستشهاد بـ Q.
!>

LDQ

!>          LDQ هو INTEGER
!>          البعد الرائد للمصفوفة Q.
!>          LDQ >= 1؛ وإذا كانت WANTQ هي .TRUE.، LDQ >= N.
!>

J1

!>          J1 هو INTEGER
!>          مؤشر الصف الأول للكتلة الأولى T11.
!>

N1

!>          N1 هو INTEGER
!>          رتبة الكتلة الأولى T11. N1 = 0، 1 أو 2.
!>

N2

!>          N2 هو INTEGER
!>          رتبة الكتلة الثانية T22. N2 = 0، 1 أو 2.
!>

WORK

!>          WORK هو مصفوفة REAL، البعد (N)
!>

INFO

!>          INFO هو INTEGER
!>          = 0: الخروج بنجاح
!>          = 1: المصفوفة المحولة T ستكون بعيدة جدًا عن الشكل القانوني
!>               لـ Schur؛ لم يتم تبديل الكتل وظلت T و Q دون تغيير.
!>
