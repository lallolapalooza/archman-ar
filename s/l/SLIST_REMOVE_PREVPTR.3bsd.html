	...

	...

n1 = malloc(sizeof(struct entry));	/* إدراج في الرأس. */
SLIST_INSERT_HEAD(&head, n1, entries);

n2 = malloc(sizeof(struct entry));	/* إدراج بعد. */
SLIST_INSERT_AFTER(n1, n2, entries);

free(n2);

SLIST_REMOVE_HEAD(&head, entries);	/* حذف من الرأس. */
free(n3);
					/* اجتياز أمامي. */
SLIST_FOREACH(np, &head, entries)
	np-> ...
					/* اجتياز أمامي آمن. */
SLIST_FOREACH_SAFE(np, &head, entries, np_temp) {

طوابير ذيل مرتبطة أحادية

طابور ذيل مرتب أحادي يتم توجيهه بواسطة بنية محددة بواسطة الماكرو STAILQ_HEAD. تحتوي هذه البنية على زوج من المؤشرات، أحدهما إلى العنصر الأول في طابور الذيل والآخر إلى العنصر الأخير في طابور الذيل. ترتبط العناصر بشكل أحادي لتقليل مساحة الاستهلاك وفوقية معالجة المؤشرات عند الحذف العشوائي للعناصر بتكلفة O(n). يمكن إضافة عناصر جديدة إلى طابور الذيل بعد عنصر موجود، أو في رأس طابور الذيل، أو في نهاية طابور الذيل. يتم الإعلان عن بنية STAILQ_HEAD على النحو التالي:

STAILQ_HEAD(HEADNAME, TYPE) head;

حيث HEADNAME هو اسم البنية التي سيتم تعريفها، و TYPE هو نوع العناصر التي سيتم ربطها في طابور الذيل. يمكن لاحقًا الإعلان عن مؤشر إلى رأس طابور الذيل على النحو التالي:

struct HEADNAME *headp;

(الأسماء head و headp قابلة للاختيار من قبل المستخدم.)

الماكرو STAILQ_HEAD_INITIALIZER يقيم إلى مُهيئ لطابور الذيل head.

الماكرو STAILQ_CONCAT يربط طابور الذيل الموجه بواسطة head2 بنهاية الطابور الموجه بواسطة head1، مما يزيل جميع الإدخالات من الأول.

الماكرو STAILQ_EMPTY يقيم إلى صحيح إذا لم تكن هناك عناصر في طابور الذيل.

الماكرو STAILQ_ENTRY يعلن عن بنية تربط العناصر في طابور الذيل.

الماكرو STAILQ_FIRST يُرجع العنصر الأول في طابور الذيل أو NULL إذا كان طابور الذيل فارغًا.

الماكرو STAILQ_FOREACH يجتاز طابور الذيل المشار إليه بواسطة head في الاتجاه الأمامي، ويعين كل عنصر بدوره إلى var.

الماكرو STAILQ_FOREACH_FROM يعمل بشكل مطابق للماكرو STAILQ_FOREACH عندما يكون var هو NULL، وإلا فإنه يعامل var كعنصر STAILQ تم العثور عليه سابقًا ويبدأ الحلقة عند var بدلاً من العنصر الأول في STAILQ المشار إليه بواسطة head.

الماكرو STAILQ_FOREACH_SAFE يجتاز طابور الذيل المشار إليه بواسطة head في الاتجاه الأمامي، ويعين كل عنصر بدوره إلى var. ومع ذلك، على عكس STAILQ_FOREACH() هنا يُسمح بإزالة var وكذلك تحريره من داخل الحلقة بأمان دون التدخل في الاجتياز.

الماكرو STAILQ_FOREACH_FROM_SAFE يعمل بشكل مطابق للماكرو STAILQ_FOREACH_SAFE عندما يكون var هو NULL، وإلا فإنه يعامل var كعنصر STAILQ تم العثور عليه سابقًا ويبدأ الحلقة عند var بدلاً من العنصر الأول في STAILQ المشار إليه بواسطة head.

الماكرو STAILQ_INIT يهيئ طابور الذيل المشار إليه بواسطة head.

الماكرو STAILQ_INSERT_HEAD يُدرج العنصر الجديد elm في رأس طابور الذيل.

الماكرو STAILQ_INSERT_TAIL يُدرج العنصر الجديد elm في نهاية طابور الذيل.

الماكرو STAILQ_INSERT_AFTER يُدرج العنصر الجديد elm بعد العنصر listelm.

الماكرو STAILQ_LAST يُرجع العنصر الأخير في طابور الذيل. إذا كان طابور الذيل فارغًا، فإن القيمة المرجعة هي NULL.

الماكرو STAILQ_NEXT يُرجع العنصر التالي في طابور الذيل، أو NULL إذا كان هذا العنصر هو الأخير.

الماكرو STAILQ_REMOVE_AFTER يزيل العنصر الذي يلي elm من طابور الذيل. على عكس STAILQ_REMOVE، لا يجتاز هذا الماكرو طابور الذيل بأكمله.

الماكرو STAILQ_REMOVE_HEAD يزيل العنصر الموجود في رأس طابور الذيل. للحصول على أقصى قدر من الكفاءة، يجب على العناصر التي تتم إزالتها من رأس طابور الذيل استخدام هذا الماكرو صراحةً بدلاً من الماكرو العام STAILQ_REMOVE.

الماكرو STAILQ_REMOVE يزيل العنصر elm من طابور الذيل. يجب تجنب استخدام هذا الماكرو لأنه يجتاز القائمة بأكملها. يجب استخدام طابور ذيل مزدوج الارتباط إذا كان هذا الماكرو مطلوبًا في مسارات التعليمات البرمجية عالية الاستخدام أو للعمل على طوابير ذيل طويلة.

الماكرو STAILQ_SWAP يتبادل محتويات head1 و head2.

مثال على طابور ذيل مرتب أحادي

STAILQ_HEAD(stailhead, entry) head =
    STAILQ_HEAD_INITIALIZER(head);
struct stailhead *headp;		/* رأس طابور الذيل المرتبطة أحادية. */
struct entry {
	...
	STAILQ_ENTRY(entry) entries;	/* طابور الذيل. */
	...
} *n1, *n2, *n3, *np;

STAILQ_INIT(&head);			/* تهيئة الطابور. */

n1 = malloc(sizeof(struct entry));	/* إدراج في الرأس. */
STAILQ_INSERT_HEAD(&head, n1, entries);

n1 = malloc(sizeof(struct entry));	/* إدراج في الذيل. */
STAILQ_INSERT_TAIL(&head, n1, entries);

n2 = malloc(sizeof(struct entry));	/* إدراج بعد. */
STAILQ_INSERT_AFTER(&head, n1, n2, entries);
					/* حذف. */
STAILQ_REMOVE(&head, n2, entry, entries);
free(n2);
					/* حذف من الرأس. */
n3 = STAILQ_FIRST(&head);
STAILQ_REMOVE_HEAD(&head, entries);
free(n3);
					/* اجتياز أمامي. */
STAILQ_FOREACH(np, &head, entries)
	np-> ...
					/* اجتياز أمامي آمن. */
STAILQ_FOREACH_SAFE(np, &head, entries, np_temp) {
	np->do_stuff();
	...
	STAILQ_REMOVE(&head, np, entry, entries);
	free(np);
}
					/* حذف طابور الذيل. */
while (!STAILQ_EMPTY(&head)) {
	n1 = STAILQ_FIRST(&head);
	STAILQ_REMOVE_HEAD(&head, entries);
	free(n1);
}
					/* حذف أسرع لطابور الذيل. */
n1 = STAILQ_FIRST(&head);
while (n1 != NULL) {
	n2 = STAILQ_NEXT(n1, entries);
	free(n1);
	n1 = n2;
}
STAILQ_INIT(&head);

قوائم

يتم توجيه القائمة بواسطة بنية محددة بواسطة الماكرو LIST_HEAD. تحتوي هذه البنية على مؤشر واحد إلى العنصر الأول في القائمة. ترتبط العناصر بشكل مزدوج بحيث يمكن إزالة عنصر عشوائي دون اجتياز القائمة. يمكن إضافة عناصر جديدة إلى القائمة بعد عنصر موجود، أو قبل عنصر موجود، أو في رأس القائمة. يتم الإعلان عن بنية LIST_HEAD على النحو التالي:

LIST_HEAD(HEADNAME, TYPE) head;

حيث HEADNAME هو اسم البنية التي سيتم تعريفها، و TYPE هو نوع العناصر التي سيتم ربطها في القائمة. يمكن لاحقًا الإعلان عن مؤشر إلى رأس القائمة على النحو التالي:

struct HEADNAME *headp;

(الأسماء head و headp قابلة للاختيار من قبل المستخدم.)

الماكرو LIST_HEAD_INITIALIZER يقيم إلى مُهيئ للقائمة head.

الماكرو LIST_CONCAT يربط القائمة الموجهة بواسطة head2 بنهاية القائمة الموجهة بواسطة head1، مما يزيل جميع الإدخالات من القائمة الأولى. يجب تجنب استخدام هذا الماكرو لأنه يجتاز القائمة head1 بأكملها. يجب استخدام طابور ذيل إذا كان هذا الماكرو مطلوبًا في مسارات التعليمات البرمجية عالية الاستخدام أو للعمل على قوائم طويلة.

الماكرو LIST_EMPTY يقيم إلى صحيح إذا لم تكن هناك عناصر في القائمة.

الماكرو LIST_ENTRY يعلن عن بنية تربط العناصر في القائمة.

الماكرو LIST_FIRST يُرجع العنصر الأول في القائمة أو NULL إذا كانت القائمة فارغة.

الماكرو LIST_FOREACH يجتاز القائمة المشار إليها بواسطة head في الاتجاه الأمامي، ويعين كل عنصر بدوره إلى var.

الماكرو LIST_FOREACH_FROM يعمل بشكل مطابق للماكرو LIST_FOREACH عندما يكون var هو NULL، وإلا فإنه يعامل var كعنصر LIST تم العثور عليه سابقًا ويبدأ الحلقة عند var بدلاً من العنصر الأول في LIST المشار إليه بواسطة head.

الماكرو LIST_FOREACH_SAFE يجتاز القائمة المشار إليها بواسطة head في الاتجاه الأمامي، ويعين كل عنصر بدوره إلى var. ومع ذلك، على عكس LIST_FOREACH() هنا يُسمح بإزالة var وكذلك تحريره من داخل الحلقة بأمان دون التدخل في الاجتياز.

الماكرو LIST_FOREACH_FROM_SAFE يعمل بشكل مطابق للماكرو LIST_FOREACH_SAFE عندما يكون var هو NULL، وإلا فإنه يعامل var كعنصر LIST تم العثور عليه سابقًا ويبدأ الحلقة عند var بدلاً من العنصر الأول في LIST المشار إليه بواسطة head.

التشخيصات

عند تصحيح أخطاء queue(3)، قد يكون من المفيد تتبع التغييرات في قائمة الانتظار. لتمكين التتبع، قم بتعريف الماكرو QUEUE_MACRO_DEBUG_TRACE عند التجميع.

قد يكون من المفيد أيضًا التخلص من المؤشرات التي تم فصلها عن قائمة الانتظار للكشف عن الاستخدام بعد الإزالة. لتمكين التخلص من المؤشرات، قم بتعريف الماكرو QUEUE_MACRO_DEBUG_TRASH عند التجميع. الماكرو QMD_IS_TRASHED(void *ptr) يعيد true إذا تم التخلص من ptr بواسطة الخيار QUEUE_MACRO_DEBUG_TRASH.
