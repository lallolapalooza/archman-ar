subroutine slaswlq (integer m, integer n, integer mb, integer nb, real, dimension( lda, * ) a, integer lda, real, dimension( ldt, * ) t, integer ldt, real, dimension( * ) work, integer lwork, integer info)

SLASWLQ

الغرض:

!>
!> SLASWLQ يحسب تحليل LQ المكدس الضيق الطويل (Tall-Skinny LQ)
!> لمصفوفة حقيقية A بحجم M × N حيث M <= N:
!>
!>    A = ( L 0 ) *  Q,
!>
!> حيث:
!>
!>    Q هي مصفوفة متعامدة N × N، مخزنة عند الخروج في شكل ضمني
!>    في العناصر فوق القطرية للمصفوفة A والعناصر في المصفوفة T؛
!>    L هي مصفوفة مثلثة سفلية M × M مخزنة عند الخروج في
!>    العناصر على القطر وتحته في المصفوفة A.
!>    0 هي مصفوفة صفرية M × (N-M)، إذا كان M < N، وليست مخزنة.
!>
!>

المعاملات

M 
!>          M هو عدد صحيح (INTEGER)
!>          عدد صفوف المصفوفة A.  M >= 0.
!>

N

!>          N هو عدد صحيح (INTEGER)
!>          عدد أعمدة المصفوفة A.  N >= M >= 0.
!>

MB

!>          MB هو عدد صحيح (INTEGER)
!>          حجم الكتلة الصفية المستخدمة في تحليل QR المكدس.
!>          M >= MB >= 1
!>

NB

!>          NB هو عدد صحيح (INTEGER)
!>          حجم الكتلة العمودية المستخدمة في تحليل QR المكدس.
!>          NB > 0.
!>

A

!>          A هي مصفوفة حقيقية (REAL array)، بأبعاد (LDA,N)
!>          عند الدخول، مصفوفة A بحجم M × N.
!>          عند الخروج، تحتوي العناصر على القطر وتحته
!>          للمصفوفة على مصفوفة مثلثة سفلية L بحجم N × N؛
!>          العناصر فوق القطر تمثل Q بواسطة صفوف
!>          V المكدسة (انظر التفاصيل الإضافية).
!>
!>

LDA

!>          LDA هو عدد صحيح (INTEGER)
!>          البعد الرائد للمصفوفة A.  LDA >= max(1,M).
!>

T

!>          T هي مصفوفة حقيقية (REAL array)،
!>          بأبعاد (LDT, N * Number_of_row_blocks)
!>          حيث Number_of_row_blocks = CEIL((N-M)/(NB-M))
!>          المتجهات العاكسة الكتلية المثلثية العلوية المخزنة في شكل مضغوط
!>          كسلسلة من الكتل المثلثية العلوية.
!>          انظر التفاصيل الإضافية أدناه.
!>

LDT

!>          LDT هو عدد صحيح (INTEGER)
!>          البعد الرائد للمصفوفة T.  LDT >= MB.
!>

WORK

!>          (مساحة عمل) مصفوفة حقيقية (REAL array)، بأبعاد (MAX(1,LWORK))
!>          عند الخروج، إذا كان INFO = 0، فإن WORK(1) تعيد الحد الأدنى LWORK.
!>

LWORK

!>          LWORK هو عدد صحيح (INTEGER)
!>          بعد المصفوفة WORK.
!>          LWORK >= 1، إذا كان MIN(M,N) = 0، و LWORK >= MB*M، بخلاف ذلك.
!>
!>          إذا كان LWORK = -1، فسيتم افتراض استعلام عن مساحة العمل؛ يقوم الروتين
!>          بحساب الحد الأدنى لحجم مصفوفة WORK فقط، ويعيد
!>          هذه القيمة كالإدخال الأول لمصفوفة WORK، ولا يتم إصدار أي رسالة خطأ
!>          متعلقة بـ LWORK بواسطة XERBLA.
!>

INFO

!>          INFO هو عدد صحيح (INTEGER)
!>          = 0:  انتهاء ناجح
!>          < 0:  إذا كان INFO = -i، فإن المعامل i لديه قيمة غير قانونية
!>

تفاصيل إضافية:

!> تحليل LQ ضيق طويل (SWLQ) يقوم بتحليل LQ عبر سلسلة من التحويلات المتعامدة،
!> يمثل Q كناتج لمصفوفات متعامدة أخرى
!>   Q = Q(1) * Q(2) * . . . * Q(k)
!> حيث كل Q(i) تصفر المدخلات فوق القطرية لكتلة من NB صفًا من A:
!>   Q(1) تصفر المدخلات فوق القطرية للصفوف 1:NB من A
!>   Q(2) تصفر الصفوف السفلية MB-N من الصفوف [1:M,NB+1:2*NB-M] من A
!>   Q(3) تصفر الصفوف السفلية MB-N من الصفوف [1:M,2*NB-M+1:3*NB-2*M] من A
!>   . . .
!>
!> يتم حساب Q(1) بواسطة GELQT، التي تمثل Q(1) بواسطة متجهات هاوسهولدر
!> مخزنة تحت القطر في صفوف 1:MB من A، وبواسطة كتل عاكسة علوية مثلثية،
!> مخزنة في المصفوفة T(1:LDT,1:N).
!> لمزيد من المعلومات انظر التفاصيل الإضافية في GELQT.
!>
!> Q(i) لـ i>1 يتم حسابها بواسطة TPLQT، التي تمثل Q(i) بواسطة متجهات هاوسهولدر
!> مخزنة في الأعمدة [(i-1)*(NB-M)+M+1:i*(NB-M)+M] من A، وبواسطة كتل عاكسة علوية مثلثية،
!> مخزنة في المصفوفة T(1:LDT,(i-1)*M+1:i*M).
!> قد تستخدم Q(k) الأخيرة عددًا أقل من الصفوف.
!> لمزيد من المعلومات انظر التفاصيل الإضافية في TPQRT.
!>
!> لمزيد من التفاصيل حول الخوارزمية الشاملة، انظر وصف
!> QR و LU Factorizations المتوازية والمتسلسلة المثلى من حيث الاتصال
!> في القسم 2.2 من [1].
!>
!> [1] "Communication-Optimal Parallel and Sequential QR and LU Factorizations,"
!>     J. Demmel, L. Grigori, M. Hoemmen, J. Langou,
!>     SIAM J. Sci. Comput, vol. 34, no. 1, 2012
!>
