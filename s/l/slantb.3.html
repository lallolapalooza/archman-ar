function real slantb (norm, uplo, diag, n, k, ab, ldab, work)
SLANTB تُرجع قيمة المعيار 1، أو معيار فروبنيوس، أو المعيار اللانهائي، أو العنصر ذو القيمة المطلقة الأكبر لمصفوفة مثلثة نطاقية.

وثائق الوظيفة/الوحدة الفرعية

real function slantb (character norm, character uplo, character diag, integer n, integer k, real, dimension( ldab, * ) ab, integer ldab, real, dimension( * ) work)

SLANTB تُرجع قيمة المعيار 1، أو معيار فروبنيوس، أو المعيار اللانهائي، أو العنصر ذو القيمة المطلقة الأكبر لمصفوفة مثلثة نطاقية.

الغرض:

!>
!> SLANTB  تُرجع قيمة المعيار واحد، أو معيار فروبنيوس، أو
!> المعيار اللانهائي، أو العنصر ذو القيمة المطلقة الأكبر لمصفوفة
!> مثلثة نطاقية A، بحجم n × n، مع ( k + 1 ) أقطار.
!>

الإرجاع

SLANTB 
!>
!>    SLANTB = ( max(abs(A(i,j))), NORM = 'M' or 'm'
!>             (
!>             ( norm1(A),         NORM = '1', 'O' or 'o'
!>             (
!>             ( normI(A),         NORM = 'I' or 'i'
!>             (
!>             ( normF(A),         NORM = 'F', 'f', 'E' or 'e'
!>
!> حيث  norm1  تمثل المعيار واحد لمصفوفة (أقصى مجموع أعمدة)،
!> normI  تمثل المعيار اللانهائي لمصفوفة (أقصى مجموع صفوف) و
!> normF  تمثل معيار فروبنيوس لمصفوفة (الجذر التربيعي لمجموع المربعات).
!> لاحظ أن  max(abs(A(i,j)))  ليس معيار مصفوفة متسقًا.
!>

المعلمات

NORM 
!>          NORM is CHARACTER*1
!>          يحدد القيمة التي سيتم إرجاعها في SLANTB كما هو موضح
!>          أعلاه.
!>

UPLO

!>          UPLO is CHARACTER*1
!>          يحدد ما إذا كانت المصفوفة A مثلثة عليا أو سفلى.
!>          = 'U':  مثلثة عليا
!>          = 'L':  مثلثة سفلى
!>

DIAG

!>          DIAG is CHARACTER*1
!>          يحدد ما إذا كانت المصفوفة A مثلثة موحدة أم لا.
!>          = 'N':  مثلثة غير موحدة
!>          = 'U':  مثلثة موحدة
!>

N

!>          N is INTEGER
!>          رتبة المصفوفة A.  N >= 0.  عندما N = 0، يتم
!>          تعيين SLANTB إلى صفر.
!>

K

!>          K is INTEGER
!>          عدد الأقطار العلوية للمصفوفة A إذا كانت UPLO = 'U'،
!>          أو عدد الأقطار السفلية للمصفوفة A إذا كانت UPLO = 'L'.
!>          K >= 0.
!>

AB

!>          AB is REAL array, dimension (LDAB,N)
!>          المصفوفة النطاقية المثلثة العليا أو السفلى A، مخزنة في
!>          أول k+1 صفوف من AB.  يتم تخزين العمود j من A
!>          في العمود j من الصف AB كما يلي:
!>          إذا كانت UPLO = 'U'، AB(k+1+i-j,j) = A(i,j) لـ max(1,j-k)<=i<=j؛
!>          إذا كانت UPLO = 'L'، AB(1+i-j,j)   = A(i,j) لـ j<=i<=min(n,j+k).
!>          لاحظ أنه عندما يكون DIAG = 'U'، فإن العناصر في الصف AB
!>          المقابلة للعناصر القطرية للمصفوفة A
!>          لا يتم الإشارة إليها، ولكن يفترض أنها واحد.
!>

LDAB

!>          LDAB is INTEGER
!>          البعد الرائد للصف AB.  LDAB >= K+1.
!>

WORK

!>          WORK is REAL array, dimension (MAX(1,LWORK)),
!>          حيث LWORK >= N عندما NORM = 'I'؛ بخلاف ذلك، لا يتم
!>          الإشارة إلى WORK.
!>
