subroutine slaorhr_col_getrfnp (integer m, integer n, real, dimension( lda, * ) a, integer lda, real, dimension( * ) d, integer info)

SLAORHR_COL_GETRFNP

الغرض:

!>
!> SLAORHR_COL_GETRFNP يحسب تحلل LU المعدل بدون
!> محورية لمصفوفة حقيقية عامة M-by-N A. التحلل له الشكل:
!>
!>     A - S = L * U,
!>
!> حيث:
!>    S هي مصفوفة إشارة قطرية m-by-n مع القطر D، بحيث أن
!>    D(i) = S(i,i)، 1 <= i <= min(M,N). القطر D مبني
!>    كـ D(i)=-SIGN(A(i,i))، حيث A(i,i) هي القيمة بعد إجراء
!>    i-1 خطوات من الحذف الغاوسي. هذا يعني أن العنصر القطري
!>    في كل خطوة من خطوات الحذف الغاوسي
!>    هو على الأقل واحد في القيمة المطلقة (بحيث لا يكون القسمة على صفر
!>    ممكنة أثناء القسمة على العنصر القطري)؛
!>
!>    L هي مصفوفة مثلثية سفلية M-by-N مع عناصر قطرية وحدوية
!>    (شبه منحرف سفلي إذا كان M > N)؛
!>
!>    و U هي مصفوفة مثلثية علوية M-by-N
!>    (شبه منحرف علوي إذا كان M < N).
!>
!> هذا الروتين هو روتين مساعد يستخدم في روتين إعادة بناء هاوسهولدر
!> SORHR_COL. في SORHR_COL، يتم تطبيق هذا الروتين على مصفوفة
!> M-by-N A ذات أعمدة متعامدة، حيث يتم تحديد كل عنصر
!> بحد أقصى واحد في القيمة المطلقة. مع اختيار
!> المصفوفة S أعلاه، يمكن إظهار أن العنصر القطري في كل
!> خطوة من خطوات الحذف الغاوسي هو الأكبر (في القيمة المطلقة) في
!> العمود على أو أسفل القطر، بحيث لا تكون المحورية مطلوبة
!> للاستقرار العددي [1].
!>
!> لمزيد من التفاصيل حول خوارزمية إعادة بناء هاوسهولدر،
!> بما في ذلك تحلل LU المعدل، انظر [1].
!>
!> هذه هي النسخة المحظورة الموجهة لليمين من الخوارزمية،
!> تستدعي BLAS المستوى 3 لتحديث المصفوفة الفرعية. لتحليل كتلة،
!> يستدعي هذا الروتين الروتين الاستدعائي الذاتي SLAORHR_COL_GETRFNP2.
!>
!> [1] ،
!>     G. Ballard, J. Demmel, L. Grigori, M. Jacquelin, H.D. Nguyen,
!>     E. Solomonik, J. Parallel Distrib. Comput.,
!>     vol. 85, pp. 3-31, 2015.
!>

المعلمات

M 
!>          M هو عدد صحيح (INTEGER)
!>          عدد صفوف المصفوفة A.  M >= 0.
!>

N

!>          N هو عدد صحيح (INTEGER)
!>          عدد أعمدة المصفوفة A.  N >= 0.
!>

A

!>          A هي مصفوفة حقيقية (REAL array)، أبعادها (LDA,N)
!>          عند الإدخال، المصفوفة M-by-N المطلوب تحليلها.
!>          عند الخروج، عوامل L و U من التحلل
!>          A-S=L*U؛ لا يتم تخزين عناصر الوحدة القطرية لـ L.
!>

LDA

!>          LDA هو عدد صحيح (INTEGER)
!>          البعد الأمامي للمصفوفة A.  LDA >= max(1,M).
!>

D

!>          D هي مصفوفة حقيقية (REAL array)، أبعادها min(M,N)
!>          العناصر القطرية لمصفوفة الإشارة القطرية M-by-N S،
!>          D(i) = S(i,i)، حيث 1 <= i <= min(M,N). يمكن أن تكون العناصر
!>          فقط زائد أو ناقص واحد.
!>

INFO

!>          INFO هو عدد صحيح (INTEGER)
!>          = 0:  انتهاء ناجح
!>          < 0:  إذا كان INFO = -i، فإن المعلمة i كانت ذات قيمة غير قانونية
!>
