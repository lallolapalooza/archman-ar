subroutine slafts (character*3 type, integer m, integer n, integer imat, integer ntests, real, dimension( * ) result, integer, dimension( 4 ) iseed, real thresh, integer iounit, integer ie)

SLAFTS

الغرض:

!>
!>    SLAFTS تختبر متجه النتيجة مقابل قيمة الحد لرؤية أي اختبارات لهذا النوع من المصفوفات فشلت في اجتياز الحد.
!>    الإخراج يكون إلى الملف المحدد بواسطة الوحدة IOUNIT.
!> 
!>  TYPE   - CHARACTER*3
!>           عند الدخول، يحدد TYPE نوع المصفوفة المراد استخدامها في
!>           الرسائل المطبوعة.
!>           غير معدل.
!>
!>  N      - INTEGER
!>           عند الدخول، يحدد N رتبة مصفوفة الاختبار.
!>           غير معدل.
!>
!>  IMAT   - INTEGER
!>           عند الدخول، يحدد IMAT نوع مصفوفة الاختبار.
!>           سيتم طباعة قائمة بأنواع مختلفة بواسطة SLAHD2
!>           إلى ملف الإخراج إذا فشل اختبار في اجتياز الحد.
!>           غير معدل.
!>
!>  NTESTS - INTEGER
!>           عند الدخول، NTESTS هو عدد الاختبارات التي تم إجراؤها على
!>           الإجراءات الفرعية في المسار المحدد بواسطة TYPE.
!>           غير معدل.
!>
!>  RESULT - REAL               مصفوفة ذات بعد ( NTESTS )
!>           عند الدخول، تحتوي RESULT على نسب الاختبار من الاختبارات
!>           التي تم إجراؤها في البرنامج المستدعي.
!>           غير معدل.
!>
!>  ISEED  - INTEGER            مصفوفة ذات بعد ( 4 )
!>           تحتوي على بذرة العشوائية التي ولّدت المصفوفة المستخدمة
!>           للاختبارات التي توجد نسبها في RESULT.
!>           غير معدل.
!>
!>  THRESH - REAL
!>           عند الدخول، يحدد THRESH الحد المقبول لنسب الاختبار.
!>           إذا كان RESULT( K ) > THRESH، فإن الاختبار K لم يجتز الحد وسيتم طباعة رسالة.
!>           غير معدل.
!>
!>  IOUNIT - INTEGER
!>           عند الدخول، يحدد IOUNIT رقم وحدة الملف
!>           الذي سيتم طباعة الرسائل إليه.
!>           غير معدل.
!>
!>  IE     - INTEGER
!>           عند الدخول، يحتوي IE على عدد الاختبارات التي
!>           فشلت في اجتياز الحد حتى الآن.
!>           يتم تحديثه عند الخروج إذا فشلت أي من النسب في RESULT أيضًا.
!>
