subroutine slarfy (character uplo, integer n, real, dimension( * ) v, integer incv, real tau, real, dimension( ldc, * ) c, integer ldc, real, dimension( * ) work)

SLARFY

الغرض:

!>
!> SLARFY يطبق عاكسًا عنصريًا، أو مصفوفة هووسهولدر، H،
!> على مصفوفة متماثلة n x n هي C، من اليسار واليمين.
!>
!> يتم تمثيل H بالشكل
!>
!>    H = I - tau * v * v'
!>
!> حيث tau هو قيمة قياسية و v هو متجه.
!>
!> إذا كانت tau صفرًا، فإن H تُعتبر مصفوفة الوحدة.
!>

المعلمات

UPLO 
!>          UPLO هو CHARACTER*1
!>          يحدد ما إذا كان الجزء العلوي أو السفلي المثلثي للمصفوفة المتماثلة C مخزنًا.
!>          = 'U':  المثلث العلوي
!>          = 'L':  المثلث السفلي
!>

N

!>          N هو INTEGER
!>          عدد الصفوف والأعمدة للمصفوفة C. N >= 0.
!>

V

!>          V هو مصفوفة REAL، البعد
!>                  (1 + (N-1)*abs(INCV))
!>          المتجه v كما هو موضح أعلاه.
!>

INCV

!>          INCV هو INTEGER
!>          الزيادة بين العناصر المتتالية لـ v. يجب ألا يكون INCV صفرًا.
!>

TAU

!>          TAU هو REAL
!>          القيمة tau كما هو موضح أعلاه.
!>

C

!>          C هي مصفوفة REAL، البعد (LDC, N)
!>          عند الدخول، المصفوفة C.
!>          عند الخروج، يتم الكتابة فوق C بواسطة H * C * H'.
!>

LDC

!>          LDC هو INTEGER
!>          البعد الرائد للمصفوفة C. LDC >= max( 1, N ).
!>

WORK

!>          WORK هي مصفوفة REAL، البعد (N)
!>
