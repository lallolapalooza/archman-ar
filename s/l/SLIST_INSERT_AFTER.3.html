SLIST(3) دليل وظائف المكتبة SLIST(3)

الاسم

SLIST_EMPTY, SLIST_ENTRY, SLIST_FIRST, SLIST_FOREACH, SLIST_HEAD, SLIST_HEAD_INITIALIZER, SLIST_INIT, SLIST_INSERT_AFTER, SLIST_INSERT_HEAD, SLIST_NEXT, SLIST_REMOVE, SLIST_REMOVE_HEAD - تنفيذ قائمة مرتبطة مفردة

المكتبة

مكتبة C القياسية (libc, -lc)

الملخص

#include <sys/queue.h>
SLIST_ENTRY(TYPE);
SLIST_HEAD(HEADNAME, TYPE);
SLIST_HEAD SLIST_HEAD_INITIALIZER(SLIST_HEAD head);
void SLIST_INIT(SLIST_HEAD *head);
int SLIST_EMPTY(SLIST_HEAD *head);
void SLIST_INSERT_HEAD(SLIST_HEAD *head,
                        struct TYPE *elm, SLIST_ENTRY NAME);
void SLIST_INSERT_AFTER(struct TYPE *listelm,
                        struct TYPE *elm, SLIST_ENTRY NAME);
struct TYPE *SLIST_FIRST(SLIST_HEAD *head);
struct TYPE *SLIST_NEXT(struct TYPE *elm, SLIST_ENTRY NAME);
SLIST_FOREACH(struct TYPE *var, SLIST_HEAD *head, SLIST_ENTRY NAME);
void SLIST_REMOVE(SLIST_HEAD *head, struct TYPE *elm, TYPE,
                        SLIST_ENTRY NAME);
void SLIST_REMOVE_HEAD(SLIST_HEAD *head,
                        SLIST_ENTRY NAME);

الوصف

تحدد هذه الماكرو وتعمل على قوائم مرتبطة مفردة.

في تعريفات الماكرو، TYPE هو اسم هيكل معرف من قبل المستخدم، يجب أن يحتوي على حقل من النوع SLIST_ENTRY، اسمه NAME. الوسيط HEADNAME هو اسم هيكل معرف من قبل المستخدم يجب تعريفه باستخدام الماكرو SLIST_HEAD().

الإنشاء

يتم ترؤس قائمة مرتبطة مفردة بواسطة هيكل معرف بواسطة الماكرو SLIST_HEAD(). يحتوي هذا الهيكل على مؤشر واحد للعنصر الأول في القائمة. يتم ربط العناصر بشكل مفرد لتقليل الحمل على المساحة ومعالجة المؤشرات، مقابل تكلفة O(n) للإزالة للعناصر العشوائية. يمكن إضافة عناصر جديدة إلى القائمة بعد عنصر موجود أو في رأس القائمة. يتم تعريف هيكل SLIST_HEAD كما يلي:


SLIST_HEAD(HEADNAME, TYPE) head;

حيث struct HEADNAME هو الهيكل الذي سيتم تعريفه، و struct TYPE هو نوع العناصر التي سيتم ربطها في القائمة. يمكن لاحقًا تعريف مؤشر إلى رأس القائمة كالتالي:


struct HEADNAME *headp;

(الأسماء head و headp اختيارية للمستخدم.)

SLIST_ENTRY() يعرف هيكلًا يربط العناصر في القائمة.

SLIST_HEAD_INITIALIZER() يقيم إلى مُهيئ للقائمة head.

SLIST_INIT() تهيئة القائمة المشار إليها بواسطة head.

SLIST_EMPTY() يقيم إلى صحيح إذا لم تكن هناك عناصر في القائمة.

الإدخال

SLIST_INSERT_HEAD() يدخل العنصر الجديد elm في رأس القائمة.

SLIST_INSERT_AFTER() يدخل العنصر الجديد elm بعد العنصر listelm.

التنقل

SLIST_FIRST() يعيد العنصر الأول في القائمة، أو NULL إذا كانت القائمة فارغة.

SLIST_NEXT() يعيد العنصر التالي في القائمة.

SLIST_FOREACH() يتنقل في القائمة المشار إليها بواسطة head في الاتجاه الأمامي، مع تعيين كل عنصر بدوره إلى var.

الإزالة

SLIST_REMOVE() يزيل العنصر elm من القائمة.

SLIST_REMOVE_HEAD() يزيل العنصر elm من رأس القائمة. لتحقيق أقصى قدر من الكفاءة، يجب على العناصر التي يتم إزالتها من رأس القائمة استخدام هذا الماكرو بشكل صريح بدلاً من SLIST_REMOVE() العام.

قيمة الإرجاع

SLIST_EMPTY() يعيد قيمة غير صفرية إذا كانت القائمة فارغة، وصفر إذا كانت القائمة تحتوي على إدخال واحد على الأقل.

SLIST_FIRST() و SLIST_NEXT() يعيدان مؤشرًا إلى هيكل TYPE الأول أو التالي، على التوالي.

SLIST_HEAD_INITIALIZER() يعيد مُهيئ يمكن تعيينه للقائمة head.

المعايير

BSD.

التاريخ

4.4BSD.

الأخطاء

SLIST_FOREACH() لا تسمح بإزالة var أو تحريرها داخل الحلقة، لأن ذلك سيتداخل مع التنقل. SLIST_FOREACH_SAFE()، الموجودة في BSD ولكنها غير موجودة في glibc، تصلح هذا القيد عن طريق السماح لـ var بالإزالة من القائمة وتحريرها بأمان من داخل الحلقة دون التداخل مع التنقل.

أمثلة

#include <stddef.h>
#include <stdio.h>
#include <stdlib.h>
#include <sys/queue.h>
struct entry {
    int data;
    SLIST_ENTRY(entry) entries;             /* قائمة مرتبطة مفردة */
};
SLIST_HEAD(slisthead, entry);
int
main(void)
{
    struct entry *n1, *n2, *n3, *np;
    struct slisthead head;                  /* رأس القائمة المرتبطة المفردة */
    SLIST_INIT(&head);                      /* تهيئة الطابور */
    n1 = malloc(sizeof(struct entry));      /* الإدخال في الرأس */
    SLIST_INSERT_HEAD(&head, n1, entries);
    n2 = malloc(sizeof(struct entry));      /* الإدخال بعد */
    SLIST_INSERT_AFTER(n1, n2, entries);
    SLIST_REMOVE(&head, n2, entry, entries);/* الحذف */
    free(n2);
    n3 = SLIST_FIRST(&head);
    SLIST_REMOVE_HEAD(&head, entries);      /* الحذف من الرأس */
    free(n3);
    for (unsigned int i = 0; i < 5; i++) {
        n1 = malloc(sizeof(struct entry));
        SLIST_INSERT_HEAD(&head, n1, entries);
        n1->data = i;
    }
                                            /* التنقل الأمامي */
    SLIST_FOREACH(np, &head, entries)
        printf("%i\n", np->data);
    while (!SLIST_EMPTY(&head)) {           /* حذف القائمة */
        n1 = SLIST_FIRST(&head);
        SLIST_REMOVE_HEAD(&head, entries);
        free(n1);
    }
    SLIST_INIT(&head);
    exit(EXIT_SUCCESS);
}
