subroutine slarfy (character uplo, integer n, real, dimension( * ) v, integer incv, real tau, real, dimension( ldc, * ) c, integer ldc, real, dimension( * ) work)

SLARFY

الغرض:

!>
!> SLARFY يطبق عاكس أساسي، أو مصفوفة هوت هاوسر، H،
!> على مصفوفة متناظرة n x n هي C، من اليسار واليمين.
!>
!> H ممثلة في الشكل
!>
!>    H = I - tau * v * v'
!>
!> حيث  tau  هي قيمة قياسية و  v  هو متجه.
!>
!> إذا كانت  tau  صفرًا، فإن H تؤخذ لتكون مصفوفة الوحدة.
!>

المعلمات

UPLO 
!>          UPLO هي CHARACTER*1
!>          يحدد ما إذا كان الجزء العلوي أو السفلي الثلاثي من المصفوفة المتناظرة C
!>          مخزنًا.
!>          = 'U':  مثلث علوي
!>          = 'L':  مثلث سفلي
!>

N

!>          N هي INTEGER
!>          عدد الصفوف والأعمدة للمصفوفة C.  N >= 0.
!>

V

!>          V هي مصفوفة REAL، الأبعاد
!>                  (1 + (N-1)*abs(INCV))
!>          المتجه v كما هو موضح أعلاه.
!>

INCV

!>          INCV هي INTEGER
!>          الفارق بين العناصر المتتالية في v. يجب أن تكون INCV
!>          غير صفرية.
!>

TAU

!>          TAU هي REAL
!>          القيمة tau كما هو موضح أعلاه.
!>

C

!>          C هي مصفوفة REAL، الأبعاد (LDC, N)
!>          عند الإدخال، المصفوفة C.
!>          عند الخروج، يتم استبدال C بـ H * C * H'.
!>

LDC

!>          LDC هي INTEGER
!>          البعد الرئيسي للمصفوفة C.  LDC >= max( 1, N ).
!>

WORK

!>          WORK هي مصفوفة REAL، الأبعاد (N)
!>
