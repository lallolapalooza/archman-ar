subroutine slasyf_aa (character uplo, integer j1, integer m, integer nb, real, dimension( lda, * ) a, integer lda, integer, dimension( * ) ipiv, real, dimension( ldh, * ) h, integer ldh, real, dimension( * ) work)

SLASYF_AA

الغرض:

!>
!> DLATRF_AA يحلل لوحة من مصفوفة متماثلة حقيقية A باستخدام
!> خوارزمية Aasen. تتكون اللوحة من مجموعة من الصفوف NB من A
!> عندما تكون UPLO هي U، أو مجموعة من الأعمدة NB عندما تكون UPLO هي L.
!>
!> لتحليل اللوحة، تتطلب خوارزمية Aasen الصف الأخير، أو العمود،
!> من اللوحة السابقة. يتم تعيين الصف الأول، أو العمود، من A ليكون الصف الأول، أو العمود،
!> من مصفوفة الوحدة، والتي تستخدم لتحليل اللوحة الأولى.
!>
!> يتم تخزين الصف J الناتج من U، أو العمود J من L، في
!> الصف (J-1)، أو العمود (J-1) من A (بدون الأقطار الوحدوية)، بينما
!> يتم الكتابة فوق القطر والقطر الفرعي لـ A بواسطة تلك الخاصة بـ T.
!>
!>

المعلمات

UPLO 
!>          UPLO is CHARACTER*1
!>          = 'U':  يتم تخزين المثلث العلوي لـ A؛
!>          = 'L':  يتم تخزين المثلث السفلي لـ A.
!>

J1

!>          J1 is INTEGER
!>          موقع الصف الأول، أو العمود الأول، من اللوحة
!>          ضمن المصفوفة الفرعية لـ A، التي تم تمريرها إلى هذا الروتين، على سبيل المثال،
!>          عند الاستدعاء بواسطة SSYTRF_AA، بالنسبة للوحة الأولى، J1 هو 1،
!>          بينما بالنسبة للوحات المتبقية، J1 هو 2.
!>

M

!>          M is INTEGER
!>          بعد المصفوفة الفرعية. M >= 0.
!>

NB

!>          NB is INTEGER
!>          بعد اللوحة التي سيتم تحليلها.
!>

A

!>          A is REAL array, dimension (LDA,M) for
!>          the first panel, while dimension (LDA,M+1) for the
!>          remaining panels.
!>
!>          عند الإدخال، يحتوي A على الصف الأخير، أو العمود الأخير،
!>          من اللوحة السابقة، والمصفوفة الفرعية اللاحقة لـ A
!>          التي سيتم تحليلها، باستثناء اللوحة الأولى، يتم تمرير اللوحة فقط.
!>
!>          عند الخروج، تم تحليل اللوحة الرائدة.
!>

LDA

!>          LDA is INTEGER
!>          البعد الرئيسي للمصفوفة A. LDA >= max(1,M).
!>

IPIV

!>          IPIV is INTEGER array, dimension (M)
!>          تفاصيل التبادلات الصفية والعمودية،
!>          تم تبديل الصف والعمود k مع الصف
!>          والعمود IPIV(k).
!>

H

!>          H is REAL workspace, dimension (LDH,NB).
!>
!>

LDH

!>          LDH is INTEGER
!>          البعد الرئيسي لمساحة العمل H. LDH >= max(1,M).
!>

WORK

!>          WORK is REAL workspace, dimension (M).
!>
