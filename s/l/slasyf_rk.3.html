توثيق الوظيفة/الروتين الفرعي

subroutine slasyf_rk (character uplo, integer n, integer nb, integer kb, real, dimension( lda, * ) a, integer lda, real, dimension( * ) e, integer, dimension( * ) ipiv, real, dimension( ldw, * ) w, integer ldw, integer info)

SLASYF_RK تحسب تجزئة جزئية لمصفوفة حقيقية متماثلة غير محددة باستخدام طريقة الارتكاز القطري المحدود من نوع Bunch-Kaufman (rook).

الغرض:

!> SLASYF_RK تحسب تجزئة جزئية لمصفوفة حقيقية متماثلة
!> باستخدام طريقة الارتكاز القطري المحدود من نوع Bunch-Kaufman (rook).
!> التجزئة الجزئية لها الشكل التالي:
!>
!> A  =  ( I  U12 ) ( A11  0  ) (  I       0    )  إذا كان UPLO = 'U'، أو:
!>       ( 0  U22 ) (  0   D  ) ( U12**T U22**T )
!>
!> A  =  ( L11  0 ) (  D   0  ) ( L11**T L21**T )  إذا كان UPLO = 'L'،
!>       ( L21  I ) (  0  A22 ) (  0       I    )
!>
!> حيث ترتيب D هو على الأكثر NB. يتم إرجاع الترتيب الفعلي في
!> المعامل KB، وهو إما NB أو NB-1، أو N إذا كانت N <= NB.
!>
!> SLASYF_RK هو روتين مساعد يتم استدعاؤه بواسطة SSYTRF_RK. يستخدم
!> رموزًا مجمعة (يستدعي BLAS من المستوى 3) لتحديث المصفوفة الفرعية
!> A11 (إذا كان UPLO = 'U') أو A22 (إذا كان UPLO = 'L').
!>

المعاملات

UPLO 
!>          UPLO هو CHARACTER*1
!>          يحدد ما إذا كان الجزء المثلثي العلوي أو السفلي للمصفوفة المتماثلة A
!>          مخزنًا:
!>          = 'U':  مثلثي علوي
!>          = 'L':  مثلثي سفلي
!>

N

!>          N هو INTEGER
!>          ترتيب المصفوفة A. N >= 0.
!>

NB

!>          NB هو INTEGER
!>          العدد الأقصى لأعمدة المصفوفة A التي يجب أن يتم
!>          تجزئتها. يجب أن يكون NB على الأقل 2 للسماح بكتل الارتكاز 2x2.
!>

KB

!>          KB هو INTEGER
!>          عدد أعمدة A التي تم تجزئتها فعليًا.
!>          KB هو إما NB-1 أو NB، أو N إذا كانت N <= NB.
!>

A

!>          A هو REAL array, dimension (LDA,N)
!>          عند الإدخال، المصفوفة المتماثلة A.
!>            إذا كان UPLO = 'U': الجزء العلوي المثلثي N×N الرائد
!>            من A يحتوي على الجزء المثلثي العلوي للمصفوفة A،
!>            والجزء السفلي المثلثي الصارم من A غير مرجعي.
!>
!>            إذا كان UPLO = 'L': الجزء السفلي المثلثي N×N الرائد
!>            من A يحتوي على الجزء السفلي المثلثي للمصفوفة A،
!>            والجزء العلوي المثلثي الصارم من A غير مرجعي.
!>
!>          عند الخروج، يحتوي على:
!>            أ) عناصر قطرية فقط للمصفوفة القطرية المتماثلة المقسمة
!>               D على القطر الخاص بـ A، أي D(k,k) = A(k,k)؛
!>               (العناصر فوق القطرية (أو تحت القطرية) لـ D
!>                مخزنة عند الخروج في المصفوفة E)، و
!>            ب) إذا كان UPLO = 'U': العامل U في الجزء فوق القطري لـ A.
!>               إذا كان UPLO = 'L': العامل L في الجزء تحت القطري لـ A.
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرائد للمصفوفة A. LDA >= max(1,N).
!>

E

!>          E هو REAL array, dimension (N)
!>          عند الخروج، يحتوي على العناصر فوق القطرية (أو تحت القطرية)
!>          للمصفوفة القطرية المتماثلة المقسمة D
!>          مع كتل قطرية 1x1 أو 2x2، حيث
!>          إذا كان UPLO = 'U': E(i) = D(i-1,i), i=2:N, E(1) تم تعيينها إلى 0؛
!>          إذا كان UPLO = 'L': E(i) = D(i+1,i), i=1:N-1, E(N) تم تعيينها إلى 0.
!>
!>          ملاحظة: بالنسبة لكتلة قطرية 1x1 D(k)، حيث
!>          1 <= k <= N، يتم تعيين العنصر E(k) إلى 0 في كل من
!>          حالات UPLO = 'U' أو UPLO = 'L'.
!>

IPIV

!>          IPIV هو INTEGER array, dimension (N)
!>          يصف IPIV مصفوفة التبديل P في تجزئة
!>          المصفوفة A على النحو التالي. تمثل القيمة المطلقة لـ IPIV(k)
!>          فهرس الصف والعمود اللذين تم
!>          تبادلهما مع الصف والعمود k. قيمة UPLO
!>          تصف ترتيب تطبيق التبادلات.
!>          أيضًا، يشير توقيع IPIV إلى الهيكل المقسم
!>          للمصفوفة القطرية المتماثلة المقسمة D مع كتل قطرية 1x1 أو 2x2
!>          والتي تتوافق مع تبادل واحد أو اثنين
!>          في كل خطوة تجزئة.
!>
!>          إذا كان UPLO = 'U'،
!>          (بترتيب التجزئة، k تتناقص من N إلى 1):
!>            أ) مدخل موجب واحد IPIV(k) > 0 يعني:
!>               D(k,k) هي كتلة قطرية 1x1.
!>               إذا كان IPIV(k) != k، تم تبادل الصفوف والأعمدة k و IPIV(k)
!>               في المصفوفة الفرعية A(1:N,N-KB+1:N)؛
!>               إذا كان IPIV(k) = k، لم يحدث تبادل.
!>
!>
!>            ب) زوج من المدخلات المتتالية السالبة
!>               IPIV(k) < 0 و IPIV(k-1) < 0 يعني:
!>               D(k-1:k,k-1:k) هي كتلة قطرية 2x2.
!>               (ملاحظة: المدخلات السالبة في IPIV تظهر في أزواج فقط).
!>               1) إذا كان -IPIV(k) != k، تم تبادل الصفوف والأعمدة
!>                  k و -IPIV(k)
!>                  في المصفوفة A(1:N,N-KB+1:N).
!>                  إذا كان -IPIV(k) = k، لم يحدث تبادل.
!>               2) إذا كان -IPIV(k-1) != k-1، تم تبادل الصفوف والأعمدة
!>                  k-1 و -IPIV(k-1)
!>                  في المصفوفة الفرعية A(1:N,N-KB+1:N).
!>                  إذا كان -IPIV(k-1) = k-1، لم يحدث تبادل.
!>
!>            ج) في كلتا الحالتين أ) وب) يكون دائمًا | IPIV(k) | <= k.
!>
!>            د) ملاحظة: أي مدخل IPIV(k) يكون دائمًا غير صفري عند الخروج.
!>
!>          إذا كان UPLO = 'L'،
!>          (بترتيب التجزئة، k تزداد من 1 إلى N):
!>            أ) مدخل موجب واحد IPIV(k) > 0 يعني:
!>               D(k,k) هي كتلة قطرية 1x1.
!>               إذا كان IPIV(k) != k، تم تبادل الصفوف والأعمدة k و IPIV(k)
!>               في المصفوفة الفرعية A(1:N,1:KB).
!>               إذا كان IPIV(k) = k، لم يحدث تبادل.
!>
!>            ب) زوج من المدخلات المتتالية السالبة
!>               IPIV(k) < 0 و IPIV(k+1) < 0 يعني:
!>               D(k:k+1,k:k+1) هي كتلة قطرية 2x2.
!>               (ملاحظة: المدخلات السالبة في IPIV تظهر في أزواج فقط).
!>               1) إذا كان -IPIV(k) != k، تم تبادل الصفوف والأعمدة
!>                  k و -IPIV(k)
!>                  في المصفوفة الفرعية A(1:N,1:KB).
!>                  إذا كان -IPIV(k) = k، لم يحدث تبادل.
!>               2) إذا كان -IPIV(k+1) != k+1، تم تبادل الصفوف والأعمدة
!>                  k-1 و -IPIV(k-1)
!>                  في المصفوفة الفرعية A(1:N,1:KB).
!>                  إذا كان -IPIV(k+1) = k+1، لم يحدث تبادل.
!>
!>            ج) في كلتا الحالتين أ) وب) يكون دائمًا | IPIV(k) | >= k.
!>
!>            د) ملاحظة: أي مدخل IPIV(k) يكون دائمًا غير صفري عند الخروج.
!>

W

!>          W هو REAL array, dimension (LDW,NB)
!>

LDW

!>          LDW هو INTEGER
!>          البعد الرائد للمصفوفة W. LDW >= max(1,N).
!>

INFO

!>          INFO هو INTEGER
!>          = 0: خروج ناجح
!>
!>          < 0: إذا كان INFO = -k، فإن المعامل k كان له قيمة غير قانونية
!>
!>          > 0: إذا كان INFO = k، فإن المصفوفة A منفردة، بسبب:
!>                 إذا كان UPLO = 'U': العمود k في الجزء
!>                 المثلثي العلوي لـ A يحتوي على أصفار.
!>                 إذا كان UPLO = 'L': العمود k في الجزء
!>                 المثلثي السفلي لـ A يحتوي على أصفار.
!>
!>               لذلك D(k,k) هو صفر بالضبط، والعناصر فوق القطرية
!>               للعمود k من U (أو العناصر تحت القطرية لـ
!>               العمود k من L) كلها أصفار. تم إكمال التجزئة.
