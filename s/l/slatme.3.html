وثائق الوظيفة/الروتين الفرعي

subroutine slatme (integer n, character dist, integer, dimension( 4 ) iseed, real, dimension( * ) d, integer mode, real cond, real dmax, character, dimension( * ) ei, character rsign, character upper, character sim, real, dimension( * ) ds, integer modes, real conds, integer kl, integer ku, real anorm, real, dimension( lda, * ) a, integer lda, real, dimension( * ) work, integer info)

SLATME

الغرض:

!>
!>    SLATME تولد مصفوفات مربعة عشوائية غير متماثلة بقيم ذاتية
!>    محددة لاختبار برامج LAPACK.
!>
!>    تعمل SLATME عن طريق تطبيق التسلسل التالي من
!>    العمليات:
!>
!>    1. تعيين القطر الرئيسي إلى D، حيث يمكن أن يكون D مدخلاً أو
!>         يتم حسابه وفقًا لـ MODE، COND، DMAX، و RSIGN
!>         كما هو موضح أدناه.
!>
!>    2. إذا كانت الأزواج المترافقة المركبة مطلوبة (MODE=0 و EI(1)='R'،
!>         أو MODE=5)، سيتم تفسير أزواج معينة من العناصر المتجاورة في D
!>         كالجزء الحقيقي والجزء التخيلي لزوج مترافق مركب؛ وبالتالي تصبح A
!>         قطرية جزئية، مع كتل 1x1 و 2x2.
!>
!>    3. إذا كان UPPER='T'، فسيتم تعيين المثلث العلوي للمصفوفة A بقيم عشوائية
!>         من التوزيع DIST.
!>
!>    4. إذا كان SIM='T'، فسيتم ضرب A على اليسار بمصفوفة عشوائية
!>         X، بقيم مفردة محددة بواسطة DS، MODES، و CONDS، وعلى اليمين
!>         بمعكوس X.
!>
!>    5. إذا كان KL < N-1، فسيتم تقليل النطاق السفلي إلى KL باستخدام
!>         تحويلات هاوسهولدر. إذا كان KU < N-1، فسيتم تقليل النطاق العلوي
!>         إلى KU.
!>
!>    6. إذا كان ANORM ليس سالباً، فسيتم قياس المصفوفة لتكون لديها
!>         أقصى قيمة مقياس مطلقة ANORM.
!>
!>    (ملاحظة: بما أن المصفوفة لا يمكن تقليلها إلى ما هو أبعد من الشكل هيسينبيرغ،
!!>     لا تتوفر خيارات التعبئة.)
!>

المعلمات

N 
!>          N هو عدد صحيح (INTEGER)
!>           عدد الأعمدة (أو الصفوف) للمصفوفة A. لم يتم تعديله.
!>

DIST

!>          DIST هو حرف (CHARACTER*1)
!>           عند الإدخال، تحدد DIST نوع التوزيع الذي سيتم استخدامه
!>           لتوليد القيم الذاتية/المفردة العشوائية، وللمثلث العلوي (انظر UPPER).
!>           'U' => منتظم ( 0, 1 )  ( 'U' لـ uniform )
!>           'S' => منتظم ( -1, 1 ) ( 'S' لـ symmetric )
!>           'N' => طبيعي ( 0, 1 )   ( 'N' لـ normal )
!>           لم يتم تعديله.
!>

ISEED

!>          ISEED هو مصفوفة أعداد صحيحة (INTEGER array)، حجمها ( 4 )
!>           عند الإدخال، تحدد ISEED بذرة مولد الأرقام العشوائية.
!>           يجب أن تكون القيم بين 0 و 4095 شاملة، ويجب أن يكون ISEED(4) فردياً.
!>           يستخدم مولد الأرقام العشوائية سلسلة خطية متطابقة محدودة بأعداد صحيحة صغيرة،
!>           ولذلك يجب أن ينتج أرقاماً عشوائية مستقلة عن الآلة.
!>           تتغير قيم ISEED عند الخروج، ويمكن استخدامها في الاستدعاء التالي لـ SLATME
!>           لمتابعة نفس تسلسل الأرقام العشوائية.
!>           تم تعديله عند الخروج.
!>

D

!>          D هي مصفوفة أعداد حقيقية (REAL array)، حجمها ( N )
!>           تستخدم هذه المصفوفة لتحديد القيم الذاتية للمصفوفة A.
!>           إذا كان MODE=0، فسيتم اعتبار D تحتوي على القيم الذاتية (ولكن
!>           راجع وصف EI)، وإلا فسيتم حسابها وفقًا لـ MODE، COND، DMAX،
!>           و RSIGN ووضعها في D.
!>           تم تعديله إذا كان MODE غير صفري.
!>

MODE

!>          MODE هو عدد صحيح (INTEGER)
!>           عند الإدخال، يصف هذا كيف سيتم تحديد القيم الذاتية:
!>           MODE = 0 يعني استخدام D (مع EI) كمدخل
!>           MODE = 1 يعين D(1)=1 و D(2:N)=1.0/COND
!>           MODE = 2 يعين D(1:N-1)=1 و D(N)=1.0/COND
!>           MODE = 3 يعين D(I)=COND**(-(I-1)/(N-1))
!>           MODE = 4 يعين D(i)=1 - (i-1)/(N-1)*(1 - 1/COND)
!>           MODE = 5 يعين D لأرقام عشوائية في النطاق
!>                    ( 1/COND , 1 ) بحيث تكون لوغاريتماتها
!>                    موزعة بشكل موحد. سيتم استخدام كل زوج متجاور (فردي-زوجي)
!>                    إما كقيمة ذاتية حقيقية أو كجزء حقيقي وجزء تخيلي
!>                    لزوج مترافق مركب؛ يتم اختيار أيهما بشكل عشوائي،
!>                    باحتمالية 50-50، لكل زوج.
!>           MODE = 6 يعين D لأرقام عشوائية من نفس توزيع
!>                    بقية المصفوفة.
!>           MODE < 0 له نفس معنى ABS(MODE)، باستثناء أن
!>              ترتيب عناصر D معكوس.
!>           وبالتالي، إذا كان MODE بين 1 و 4، تحتوي D على عناصر تتراوح
!>              من 1 إلى 1/COND، وإذا كان بين -1 و -4، تحتوي D على عناصر
!>              تتراوح من 1/COND إلى 1،
!>           لم يتم تعديله.
!>

COND

!>          COND هو عدد حقيقي (REAL)
!>           عند الإدخال، يستخدم كما هو موضح تحت MODE أعلاه.
!>           إذا تم استخدامه، يجب أن يكون >= 1. لم يتم تعديله.
!>

DMAX

!>          DMAX هو عدد حقيقي (REAL)
!>           إذا كان MODE ليس -6، 0، أو 6، فسيتم قياس محتويات D، كما
!>           تم حسابها وفقًا لـ MODE و COND، بواسطة
!>           DMAX / max(abs(D(i))). لاحظ أن DMAX لا يلزم أن يكون
!>           موجبًا: إذا كان DMAX سالبًا (أو صفرًا)، فسيتم قياس D
!>           بعدد سالب (أو صفر).
!>           لم يتم تعديله.
!>

EI

!>          EI هي مصفوفة حروف (CHARACTER*1 array)، حجمها ( N )
!>           إذا كان MODE هو 0، وإذا لم يكن EI(1)=' ' (مسافة)،
!>           تحدد هذه المصفوفة أي عناصر في D (عند الإدخال) هي
!>           قيم ذاتية حقيقية وأيها هي الجزء الحقيقي والجزء التخيلي
!>           لزوج مترافق مركب. قد تحتوي عناصر EI فقط على القيم 'R' و 'I'.
!>           إذا كان EI(j)='R' و EI(j+1)='I'، فإن القيمة الذاتية j هي
!>           CMPLX( D(j) , D(j+1) )، والقيمة (j+1) هي المرافق المركب لها.
!>           إذا كان EI(j)=EI(j+1)='R'، فإن القيمة الذاتية j هي D(j) (أي حقيقية).
!>           لا يمكن أن يكون EI(1)='I'، ولا يمكن أن يكون عنصران متجاوران في EI لهما القيمة 'I'.
!>           إذا كان MODE ليس 0، فسيتم تجاهل EI. إذا كان MODE هو 0 و
!>           EI(1)=' '، فإن جميع القيم الذاتية ستكون حقيقية.
!>           لم يتم تعديله.
!>

RSIGN

!>          RSIGN هو حرف (CHARACTER*1)
!>           إذا كان MODE ليس 0، 6، أو -6، وكان RSIGN='T'، فسيتم ضرب
!>           عناصر D، كما تم حسابها وفقًا لـ MODE و COND،
!>           في علامة عشوائية (+1 أو -1). إذا كان RSIGN='F'،
!>           فلن يتم ضربها. يمكن أن يكون لـ RSIGN فقط القيم 'T' أو 'F'.
!>           لم يتم تعديله.
!>

UPPER

!>          UPPER هو حرف (CHARACTER*1)
!>           إذا كان UPPER='T'، فسيتم تعيين عناصر A فوق القطر الرئيسي
!>           (وفوق الكتل القطرية 2x2، إذا كانت A تحتوي على قيم ذاتية مركبة)
!>           إلى أرقام عشوائية من DIST.
!>           إذا كان UPPER='F'، فلن يتم ذلك. يمكن أن يكون لـ UPPER فقط
!>           القيم 'T' أو 'F'.
!>           لم يتم تعديله.
!>

SIM

!>          SIM هو حرف (CHARACTER*1)
!>           إذا كان SIM='T'، فسيتم تطبيق عملية على A، أي ضربها على اليسار بمصفوفة X
!>           وعلى اليمين بمعكوس X. X = U S V، حيث U و V
!>           مصفوفتان وحدويتان عشوائيتان و S مصفوفة (قطرية) ذات
!>           قيم مفردة محددة بواسطة DS، MODES، و CONDS. إذا
!>           كان SIM='F'، فلن يتم تطبيق العملية على A.
!>           لم يتم تعديله.
!>

DS

!>          DS هي مصفوفة أعداد حقيقية (REAL array)، حجمها ( N )
!>           إذا كان SIM='T'، تحدد DS القيم المفردة للمصفوفة X.
!>           إذا كان MODES ليس 0، فسيتم حساب القيم المفردة وفقًا لـ MODES و CONDS
!>           ووضعها في DS.
!>           تم تعديله إذا كان SIM='T' و MODES غير صفر.
!>

MODES

!>          MODES هو عدد صحيح (INTEGER)
!>           يصف هذا كيف سيتم تحديد القيم المفردة في DS (إذا كان SIM='T').
!>           MODES = 0 يعني استخدام DS (مع CONDS) كمدخل.
!>           MODES = 1 يعين DS(1)=1 و DS(2:N)=1.0/CONDS.
!>           MODES = 2 يعين DS(1:N-1)=1 و DS(N)=1.0/CONDS.
!>           MODES = 3 يعين DS(I)=CONDS**(-(I-1)/(N-1)).
!>           MODES = 4 يعين DS(i)=1 - (i-1)/(N-1)*(1 - 1/CONDS).
!>           MODES = 5 يعين DS لأرقام عشوائية في النطاق
!>                    ( 1/CONDS , 1 ) بحيث تكون لوغاريتماتها
!>                    موزعة بشكل موحد.
!>           MODES < 0 له نفس معنى ABS(MODES)، باستثناء أن
!>              ترتيب عناصر DS معكوس.
!>           لم يتم تعديله.
!>

CONDS

!>          CONDS هو عدد حقيقي (REAL)
!>           عند الإدخال، يستخدم كما هو موضح تحت MODES أعلاه.
!>           إذا تم استخدامه، يجب أن يكون >= 1. لم يتم تعديله.
!>

KL

!>          KL هو عدد صحيح (INTEGER)
!>           على الإدخال، يحدد النطاق السفلي للمصفوفة. إذا كان KL < N-1،
!>           فسيتم تقليل النطاق السفلي إلى KL باستخدام تحويلات هاوسهولدر.
!>           لم يتم تعديله.
!>

KU

!>          KU هو عدد صحيح (INTEGER)
!>           على الإدخال، يحدد النطاق العلوي للمصفوفة. إذا كان KU < N-1،
!>           فسيتم تقليل النطاق العلوي إلى KU باستخدام تحويلات هاوسهولدر.
!>           لم يتم تعديله.
!>

ANORM

!>          ANORM هو عدد حقيقي (REAL)
!>           إذا كان ANORM >= 0، فسيتم قياس المصفوفة A بحيث تكون
!>           أقصى قيمة مقياس مطلقة لها هي ANORM.
!>           لم يتم تعديله.
!>

A

!>          A هي مصفوفة أعداد حقيقية (REAL array)، حجمها ( LDA, * )
!>           على الخروج، تحتوي على المصفوفة العشوائية المطلوبة.
!>           تم تعديله.
!>

LDA

!>          LDA هو عدد صحيح (INTEGER)
!>           على الإدخال، هو طول الصف الأول للمصفوفة A.
!>           لم يتم تعديله.
!>

WORK

!>          WORK هي مصفوفة أعداد حقيقية (REAL array)، حجمها ( * )
!>           مساحة عمل.
!>           تم تعديله.
!>

INFO

!>          INFO هو عدد صحيح (INTEGER)
!>           معلومات الإخراج.
!>           = 0: صحيح
!>           < 0: إذا كان INFO = -k، فإن المعلمة k-th تحتوي على قيمة غير صالحة.
!>           تم تعديله.
!>
