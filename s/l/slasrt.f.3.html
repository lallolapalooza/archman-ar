subroutine slasrt (character id, integer n, real, dimension( * ) d, integer info)

SLASRT تقوم بفرز الأرقام بترتيب تصاعدي أو تنازلي.

الغرض:

!>
!> قم بفرز الأرقام في D بترتيب تصاعدي (إذا كان ID = 'I') أو
!> بترتيب تنازلي (إذا كان ID = 'D').
!>
!> استخدم Quick Sort، بالعودة إلى Insertion sort للمصفوفات ذات
!> الحجم <= 20. بُعد STACK يحد N إلى حوالي 2**32.
!>

المعلمات

ID 
!>          ID هو CHARACTER*1
!>          = 'I': فرز D بترتيب تصاعدي؛
!>          = 'D': فرز D بترتيب تنازلي.
!>

N

!>          N هو INTEGER
!>          طول المصفوفة D.
!>

D

!>          D هي مصفوفة REAL، بُعدها (N)
!>          عند الدخول، المصفوفة المراد فرزها.
!>          عند الخروج، تم فرز D بترتيب تصاعدي
!>          (D(1) <= ... <= D(N)) أو بترتيب تنازلي
!>          (D(1) >= ... >= D(N))، اعتمادًا على ID.
!>

INFO

!>          INFO هو INTEGER
!>          = 0:  انتهاء ناجح
!>          < 0:  إذا كان INFO = -i، فإن المعلمة i كان لها قيمة غير قانونية
!>
