توثيق الوظيفة/الإجراء الفرعي

function حقيقية slantp (character norm, character uplo, character diag, integer n, real, dimension( * ) ap, real, dimension( * ) work)

SLANTP تُرجع قيمة المعيار 1، أو معيار فروبنيوس، أو المعيار اللانهائي، أو العنصر ذي القيمة المطلقة الأكبر لمصفوفة مثلثية مقدمة بشكل مضغوط.

الغرض:

!>
!> SLANTP  تُرجع قيمة المعيار واحد،  أو معيار فروبنيوس،  أو
!> المعيار اللانهائي،  أو  العنصر ذي القيمة المطلقة الأكبر  لمصفوفة
!> مثلثية A،  مقدمة بشكل مضغوط.
!>

تُرجع

SLANTP 
!>
!>    SLANTP = ( max(abs(A(i,j))), NORM = 'M' or 'm'
!>             (
!>             ( norm1(A),         NORM = '1', 'O' or 'o'
!>             (
!>             ( normI(A),         NORM = 'I' or 'i'
!>             (
!>             ( normF(A),         NORM = 'F', 'f', 'E' or 'e'
!>
!> حيث  norm1  يمثل  المعيار واحد لمصفوفة (أقصى مجموع عمود)،
!> normI  يمثل  المعيار اللانهائي  لمصفوفة  (أقصى مجموع صف) و
!> normF  يمثل  معيار فروبنيوس  لمصفوفة (الجذر التربيعي لمجموع
!> المربعات).  لاحظ أن  max(abs(A(i,j)))  ليس معيار مصفوفة ثابت.
!>

المعلمات

NORM 
!>          NORM هو CHARACTER*1
!>          يحدد القيمة التي سيتم إرجاعها في SLANTP كما هو موضح
!>          أعلاه.
!>

UPLO

!>          UPLO هو CHARACTER*1
!>          يحدد ما إذا كانت المصفوفة A مثلثية علوية أو سفلية.
!>          = 'U':  مثلثية علوية
!>          = 'L':  مثلثية سفلية
!>

DIAG

!>          DIAG هو CHARACTER*1
!>          يحدد ما إذا كانت المصفوفة A مثلثية وحدوية أم لا.
!>          = 'N':  مثلثية غير وحدوية
!>          = 'U':  مثلثية وحدوية
!>

N

!>          N هو INTEGER
!>          رتبة المصفوفة A.  N >= 0.  عندما N = 0، يتم تعيين SLANTP
!>          إلى الصفر.
!>

AP

!>          AP هو مصفوفة REAL، بُعدها (N*(N+1)/2)
!>          المصفوفة المثلثية العلوية أو السفلية A، مضغوطة عمودياً في
!>          مصفوفة خطية.  يتم تخزين العمود j من A في المصفوفة
!>          AP كما يلي:
!>          إذا كان UPLO = 'U'، فإن AP(i + (j-1)*j/2) = A(i,j) لـ 1<=i<=j؛
!>          إذا كان UPLO = 'L'، فإن AP(i + (j-1)*(2n-j)/2) = A(i,j) لـ j<=i<=n.
!>          لاحظ أنه عندما يكون DIAG = 'U'، فإن عناصر المصفوفة AP
!>          المقابلة للعناصر القطرية للمصفوفة A
!>          لا يتم الرجوع إليها، ولكن يُفترض أنها تساوي واحد.
!>

WORK

!>          WORK هي مصفوفة REAL، بُعدها (MAX(1,LWORK))،
!>          حيث LWORK >= N عندما يكون NORM = 'I'؛ بخلاف ذلك، لا يتم
!>          الرجوع إلى WORK.
!>
