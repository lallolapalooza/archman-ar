subroutine slattb (integer imat, character uplo, character trans, character diag, integer, dimension( 4 ) iseed, integer n, integer kd, real, dimension( ldab, * ) ab, integer ldab, real, dimension( * ) b, real, dimension( * ) work, integer info)

SLATTB

الغرض:

!>
!> SLATTB تولد مصفوفة اختبار مثلثية في تخزين ثنائي الأبعاد.
!> IMAT و UPLO يحددان بشكل فريد خصائص مصفوفة الاختبار،
!> والتي يتم إرجاعها في المصفوفة A.
!>

المعلمات

IMAT 
!>          IMAT هو عدد صحيح
!>          مفتاح عدد صحيح يصف أي مصفوفة سيتم إنشاؤها لهذا
!>          المسار.
!>

UPLO

!>          UPLO هو حرف واحد
!>          يحدد ما إذا كانت المصفوفة A ستكون علوية أو سفلية
!>          مثلثة.
!>          = 'U':  مثلثية علوية
!>          = 'L':  مثلثية سفلية
!>

TRANS

!>          TRANS هو حرف واحد
!>          يحدد ما إذا كانت المصفوفة أو منقولتها سيتم استخدامها.
!>          = 'N':  بدون منقولة
!>          = 'T':  منقولة
!>          = 'C':  منقولة مترافقة (= منقولة)
!>

DIAG

!>          DIAG هو حرف واحد
!>          يحدد ما إذا كانت المصفوفة A مثلثة وحدوية أم لا.
!>          = 'N':  مثلثية غير وحدوية
!>          = 'U':  مثلثية وحدوية
!>

ISEED

!>          ISEED هي مصفوفة أعداد صحيحة، البعد (4)
!>          متجه البذور لمولد الأرقام العشوائية (يستخدم في
!>          SLATMS).  يتم تعديله عند الخروج.
!>

N

!>          N هو عدد صحيح
!>          ترتيب المصفوفة التي سيتم إنشاؤها.
!>

KD

!>          KD هو عدد صحيح
!>          عدد الأقطار العليا أو الأقطار السفلية للمصفوفة
!>          المثلثية ذات النطاق A.  KD >= 0.
!>

AB

!>          AB هي مصفوفة حقيقية، البعد (LDAB,N)
!>          المصفوفة المثلثية العليا أو السفلية ذات النطاق A، مخزنة في
!>          أول KD+1 صف من AB.  لتكن j عمودًا من A، 1<=j<=n.
!>          إذا كان UPLO = 'U'، فإن AB(kd+1+i-j,j) = A(i,j) لـ max(1,j-kd)<=i<=j.
!>          إذا كان UPLO = 'L'، فإن AB(1+i-j,j)    = A(i,j) لـ j<=i<=min(n,j+kd).
!>

LDAB

!>          LDAB هو عدد صحيح
!>          البعد الرائد للمصفوفة AB.  LDAB >= KD+1.
!>

B

!>          B هي مصفوفة حقيقية، البعد (N)
!>

WORK

!>          WORK هي مصفوفة حقيقية، البعد (2*N)
!>

INFO

!>          INFO هو عدد صحيح
!>          = 0:  انتهاء ناجح
!>          < 0: إذا كان INFO = -k، فإن المعلمة k-th كانت ذات قيمة غير قانونية
!>
