subroutine slaptm (integer n, integer nrhs, real alpha, real, dimension( * ) d, real, dimension( * ) e, real, dimension( ldx, * ) x, integer ldx, real beta, real, dimension( ldb, * ) b, integer ldb)

SLAPTM

الغرض:

!>
!> SLAPTM تضرب مصفوفة X ذات الأبعاد N × NRHS بالمصفوفة المتناظرة ثلاثية القطر A
!> وتخزن النتيجة في مصفوفة B. العملية تأخذ الشكل
!>
!>    B := alpha * A * X + beta * B
!>
!> حيث يمكن أن تكون alpha إما 1. أو -1. ويمكن أن تكون beta 0. أو 1. أو -1.
!>

المعلمات

N 
!>          N هو عدد صحيح (INTEGER)
!>          رتبة المصفوفة A. N >= 0.
!>

NRHS

!>          NRHS هو عدد صحيح (INTEGER)
!>          عدد الأطراف اليمنى، أي عدد أعمدة المصفوفتين X و B.
!>

ALPHA

!>          ALPHA هو قيمة حقيقية (REAL)
!>          المقدار الثابت alpha. يجب أن تكون ALPHA 1. أو -1.؛ بخلاف ذلك،
!>          يفترض أنها 0.
!>

D

!>          D هي مصفوفة حقيقية (REAL array)، بأبعاد (N)
!>          العناصر القطرية N للمصفوفة ثلاثية القطر A.
!>

E

!>          E هي مصفوفة حقيقية (REAL array)، بأبعاد (N-1)
!>          العناصر تحت القطرية أو فوق القطرية (N-1) للمصفوفة A.
!>

X

!>          X هي مصفوفة حقيقية (REAL array)، بأبعاد (LDX,NRHS)
!>          المصفوفة X ذات الأبعاد N × NRHS.
!>

LDX

!>          LDX هو عدد صحيح (INTEGER)
!>          البعد الرائد للمصفوفة X. LDX >= max(N,1).
!>

BETA

!>          BETA هو قيمة حقيقية (REAL)
!>          المقدار الثابت beta. يجب أن تكون BETA 0. أو 1. أو -1.؛ بخلاف ذلك،
!>          يفترض أنها 1.
!>

B

!>          B هي مصفوفة حقيقية (REAL array)، بأبعاد (LDB,NRHS)
!>          عند الدخول، المصفوفة B ذات الأبعاد N × NRHS.
!>          عند الخروج، يتم استبدال B بالتعبير المصفوفي
!>          B := alpha * A * X + beta * B.
!>

LDB

!>          LDB هو عدد صحيح (INTEGER)
!>          البعد الرائد للمصفوفة B. LDB >= max(N,1).
!>
