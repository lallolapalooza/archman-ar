real function slantr (norm, uplo, diag, m, n, a, lda, work)
SLANTR تُرجع قيمة المعيار 1، أو معيار فروبنيوس، أو المعيار اللانهائي، أو العنصر ذي القيمة المطلقة الأكبر لمصفوفة شبه منحرفة أو مثلثية.

توثيق الوظيفة/الروتين الفرعي

real function slantr (character norm, character uplo, character diag, integer m, integer n, real, dimension( lda, * ) a, integer lda, real, dimension( * ) work)

SLANTR تُرجع قيمة المعيار 1، أو معيار فروبنيوس، أو المعيار اللانهائي، أو العنصر ذي القيمة المطلقة الأكبر لمصفوفة شبه منحرفة أو مثلثية.

الغرض:

!>
!> SLANTR  تُرجع قيمة المعيار واحد،  أو معيار فروبنيوس،  أو
!> المعيار اللانهائي،  أو  العنصر ذي القيمة المطلقة الأكبر  لمصفوفة
!> شبه منحرفة أو مثلثية A.
!>

الإرجاع

SLANTR 
!>
!>    SLANTR = ( max(abs(A(i,j))), NORM = 'M' or 'm'
!>             (
!>             ( norm1(A),         NORM = '1', 'O' or 'o'
!>             (
!>             ( normI(A),         NORM = 'I' or 'i'
!>             (
!>             ( normF(A),         NORM = 'F', 'f', 'E' or 'e'
!>
!> حيث  norm1  تعني معيار المصفوفة واحد (أقصى مجموع للأعمدة)،
!> normI  تعني المعيار اللانهائي  للمصفوفة  (أقصى مجموع للصفوف)  و
!> normF  تعني معيار فروبنيوس  للمصفوفة  (الجذر التربيعي لمجموع
!> المربعات).  لاحظ أن  max(abs(A(i,j)))  ليس معيار مصفوفة ثابت.
!>

المعاملات

NORM 
!>          NORM is CHARACTER*1
!>          يحدد القيمة التي سيتم إرجاعها في SLANTR كما هو موضح
!>          أعلاه.
!>

UPLO

!>          UPLO is CHARACTER*1
!>          يحدد ما إذا كانت المصفوفة A شبه منحرفة علوية أم سفلية.
!>          = 'U':  شبه منحرفة علوية
!>          = 'L':  شبه منحرفة سفلية
!>          لاحظ أن A تكون مثلثية بدلاً من شبه منحرفة إذا كان M = N.
!>

DIAG

!>          DIAG is CHARACTER*1
!>          يحدد ما إذا كانت المصفوفة A لها قطر وحيد أم لا.
!>          = 'N':  قطر غير وحيد
!>          = 'U':  قطر وحيد
!>

M

!>          M is INTEGER
!>          عدد صفوف المصفوفة A.  M >= 0, وإذا
!>          UPLO = 'U', M <= N.  عندما M = 0، يتم تعيين SLANTR إلى صفر.
!>

N

!>          N is INTEGER
!>          عدد أعمدة المصفوفة A.  N >= 0, وإذا
!>          UPLO = 'L', N <= M.  عندما N = 0، يتم تعيين SLANTR إلى صفر.
!>

A

!>          A is REAL array, dimension (LDA,N)
!>          المصفوفة شبه المنحرفة A (A تكون مثلثية إذا كان M = N).
!>          إذا كان UPLO = 'U', الجزء العلوي شبه المنحرف الذي أبعاده m × n في
!>          المصفوفة A يحتوي على المصفوفة شبه المنحرفة العلوية، والجزء
!>          المثلثي السفلي تمامًا من A لا يتم الإشارة إليه.
!>          إذا كان UPLO = 'L', الجزء السفلي شبه المنحرف الذي أبعاده m × n في
!>          المصفوفة A يحتوي على المصفوفة شبه المنحرفة السفلية، والجزء
!>          المثلثي العلوي تمامًا من A لا يتم الإشارة إليه.  لاحظ
!>          أنه عند DIAGONAL = 'U', لا يتم الإشارة إلى العناصر القطرية
!>          من A ويُفترض أنها واحد.
!>

LDA

!>          LDA is INTEGER
!>          البعد الرئيسي للمصفوفة A.  LDA >= max(M,1).
!>

WORK

!>          WORK is REAL array, dimension (MAX(1,LWORK)),
!>          حيث LWORK >= M عندما يكون NORM = 'I'; بخلاف ذلك، لا يتم
!>          الإشارة إلى WORK.
!>
