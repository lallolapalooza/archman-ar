subroutine slatm5 (integer prtype, integer m, integer n, real, dimension( lda, * ) a, integer lda, real, dimension( ldb, * ) b, integer ldb, real, dimension( ldc, * ) c, integer ldc, real, dimension( ldd, * ) d, integer ldd, real, dimension( lde, * ) e, integer lde, real, dimension( ldf, * ) f, integer ldf, real, dimension( ldr, * ) r, integer ldr, real, dimension( ldl, * ) l, integer ldl, real alpha, integer qblcka, integer qblckb)

SLATM5

الغرض:

!>
!> SLATM5 ينشئ مصفوفات مشاركة في معادلة سيلفستر المعممة:
!>
!>     A * R - L * B = C
!>     D * R - L * E = F
!>
!> وهي تحقق أيضًا (شرط التقطير)
!>
!>  [ I -L ] ( [ A  -C ], [ D -F ] ) [ I  R ] = ( [ A    ], [ D    ] )
!>  [    I ] ( [     B ]  [    E ] ) [    I ]   ( [    B ]  [    E ] )
!>
!>

!>           لنوع معين من المصفوفات المراد إنشاؤها
!>          (انظر المزيد من التفاصيل).
!>

!>          M هو عدد صحيح (INTEGER)

تفاصيل إضافية:

!>
!>  PRTYPE = 1: A و B هما كتل جوردان، D و E هما مصفوفات وحدة
!>
!>             A : إذا كان (i == j) فإن A(i, j) = 1.0
!>                 إذا كان (j == i + 1) فإن A(i, j) = -1.0
!>                 وإلا فإن A(i, j) = 0.0،            i, j = 1...M
!>
!>             B : إذا كان (i == j) فإن B(i, j) = 1.0 - ALPHA
!>                 إذا كان (j == i + 1) فإن B(i, j) = 1.0
!>                 وإلا فإن B(i, j) = 0.0،            i, j = 1...N
!>
!>             D : إذا كان (i == j) فإن D(i, j) = 1.0
!>                 وإلا فإن D(i, j) = 0.0،            i, j = 1...M
!>
!>             E : إذا كان (i == j) فإن E(i, j) = 1.0
!>                 وإلا فإن E(i, j) = 0.0،            i, j = 1...N
!>
!>             L = R يتم اختيارهما من [-10...10]،
!>                  مما يحدد الأطراف اليمنى (C, F).
!>
!>  PRTYPE = 2 أو 3: مثلثية و/أو شبه مثلثية.
!>
!>             A : إذا كان (i <= j) فإن A(i, j) = [-1...1]
!>                 وإلا فإن A(i, j) = 0.0،             i, j = 1...M
!>
!>                 إذا كان (PRTYPE = 3) فإن
!>                    A(k + 1, k + 1) = A(k, k)
!>                    A(k + 1, k) = [-1...1]
!>                    sign(A(k, k + 1) = -(sin(A(k + 1, k))
!>                        k = 1, M - 1, QBLCKA
!>
!>             B : إذا كان (i <= j) فإن B(i, j) = [-1...1]
!>                 وإلا فإن B(i, j) = 0.0،            i, j = 1...N
!>
!>                 إذا كان (PRTYPE = 3) فإن
!>                    B(k + 1, k + 1) = B(k, k)
!>                    B(k + 1, k) = [-1...1]
!>                    sign(B(k, k + 1) = -(sign(B(k + 1, k))
!>                        k = 1, N - 1, QBLCKB
!>
!>             D : إذا كان (i <= j) فإن D(i, j) = [-1...1].
!>                 وإلا فإن D(i, j) = 0.0،            i, j = 1...M
!>
!>
!>             E : إذا كان (i <= j) فإن D(i, j) = [-1...1]
!>                 وإلا فإن E(i, j) = 0.0،            i, j = 1...N
!>
!>                 يتم اختيار L و R من [-10...10]،
!>                 مما يحدد الأطراف اليمنى (C, F).
!>
!>  PRTYPE = 4 كاملة
!>             A(i, j) = [-10...10]
!>             D(i, j) = [-1...1]    i,j = 1...M
!>             B(i, j) = [-10...10]
!>             E(i, j) = [-1...1]    i,j = 1...N
!>             R(i, j) = [-10...10]
!>             L(i, j) = [-1...1]    i = 1..M ,j = 1...N
!>
!>             L و R يحددان الأطراف اليمنى (C, F).
!>
!>  PRTYPE = 5 حالة خاصة القيم الذاتية المشتركة و/أو القريبة.
!>
