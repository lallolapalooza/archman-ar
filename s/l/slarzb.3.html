SLARZB تطبق عاكس كتلة أو معكوسها على مصفوفة عامة.

توثيق الوظيفة/الإجراء الفرعي

subroutine slarzb (character side, character trans, character direct, character storev, integer m, integer n, integer k, integer l, real, dimension( ldv, * ) v, integer ldv, real, dimension( ldt, * ) t, integer ldt, real, dimension( ldc, * ) c, integer ldc, real, dimension( ldwork, * ) work, integer ldwork)

SLARZB تطبق عاكس كتلة حقيقي H أو معكوسه H**T على مصفوفة عامة حقيقية موزعة C من اليسار أو اليمين.

الغرض:

!>
!> SLARZB تطبق عاكس كتلة حقيقي H أو معكوسه H**T على
!> مصفوفة عامة حقيقية موزعة M-by-N  C من اليسار أو اليمين.
!>
!> حالياً، يتم دعم STOREV = 'R' و DIRECT = 'B' فقط.
!>

المعلمات

SIDE 
!>          SIDE هو CHARACTER*1
!>          = 'L': تطبيق H أو H**T من اليسار
!>          = 'R': تطبيق H أو H**T من اليمين
!>

TRANS

!>          TRANS هو CHARACTER*1
!>          = 'N': تطبيق H (بدون تبديل)
!>          = 'C': تطبيق H**T (تبديل)
!>

DIRECT

!>          DIRECT هو CHARACTER*1
!>          يشير إلى كيفية تشكيل H من حاصل ضرب عواكس أولية
!>          = 'F': H = H(1) H(2) . . . H(k) (أمامي، غير مدعوم بعد)
!>          = 'B': H = H(k) . . . H(2) H(1) (خلفي)
!>

STOREV

!>          STOREV هو CHARACTER*1
!>          يشير إلى كيفية تخزين المتجهات التي تحدد العواكس الأولية:
!>          = 'C': عمودي                        (غير مدعوم بعد)
!>          = 'R': صفّي
!>

M

!>          M هو INTEGER
!>          عدد صفوف المصفوفة C.
!>

N

!>          N هو INTEGER
!>          عدد أعمدة المصفوفة C.
!>

K

!>          K هو INTEGER
!>          رتبة المصفوفة T (= عدد العواكس الأولية التي يشكل حاصل ضربها
!>          عاكس الكتلة).
!>

L

!>          L هو INTEGER
!>          عدد أعمدة المصفوفة V التي تحتوي على الجزء ذي المعنى
!>          من عواكس هاوسهولدر.
!>          إذا كانت SIDE = 'L'، فإن M >= L >= 0، وإذا كانت SIDE = 'R'، فإن N >= L >= 0.
!>

V

!>          V هي مصفوفة REAL، أبعادها (LDV,NV).
!>          إذا كانت STOREV = 'C'، فإن NV = K؛ وإذا كانت STOREV = 'R'، فإن NV = L.
!>

LDV

!>          LDV هو INTEGER
!>          البعد الرئيسي للمصفوفة V.
!>          إذا كانت STOREV = 'C'، فإن LDV >= L؛ وإذا كانت STOREV = 'R'، فإن LDV >= K.
!>

T

!>          T هي مصفوفة REAL، أبعادها (LDT,K)
!>          المصفوفة المثلثية K-by-K  T في تمثيل عاكس الكتلة.
!>

LDT

!>          LDT هو INTEGER
!>          البعد الرئيسي للمصفوفة T. LDT >= K.
!>

C

!>          C هي مصفوفة REAL، أبعادها (LDC,N)
!>          عند الدخول، المصفوفة C ذات الأبعاد M-by-N.
!>          عند الخروج، يتم استبدال C بـ H*C أو H**T*C أو C*H أو C*H**T.
!>

LDC

!>          LDC هو INTEGER
!>          البعد الرئيسي للمصفوفة C. LDC >= max(1,M).
!>

WORK

!>          WORK هي مصفوفة REAL، أبعادها (LDWORK,K)
!>

LDWORK

!>          LDWORK هو INTEGER
!>          البعد الرئيسي للمصفوفة WORK.
!>          إذا كانت SIDE = 'L'، فإن LDWORK >= max(1,N)؛
!>          إذا كانت SIDE = 'R'، فإن LDWORK >= max(1,M).
!>
