subroutine slatm4 (integer itype, integer n, integer nz1, integer nz2, integer isign, real amagn, real rcond, real triang, integer idist, integer, dimension( 4 ) iseed, real, dimension( lda, * ) a, integer lda)

SLATM4

الغرض:

!>
!> SLATM4 ينشئ مصفوفات مربعة أساسية، والتي يمكن لاحقًا ضربها
!> بأخرى لإنتاج مصفوفات اختبار. وهي
!> مخصصة بشكل أساسي لاختبار إجراءات القيم الذاتية المعممة.
!>
!> تقوم أولاً بإنشاء القطر الرئيسي (وربما) القطر الفرعي،
!> وفقًا لقيمة ITYPE، NZ1، NZ2، ISIGN، AMAGN، و RCOND.
!> ثم تقوم بملء المثلث العلوي بأرقام عشوائية، إذا كانت TRIANG
!> غير صفرية.
!>

المعاملات

ITYPE 
!>          ITYPE هو عدد صحيح (INTEGER).
!>          نوع المصفوفة على القطر الرئيسي والقطر الفرعي.
!>          إذا كانت ITYPE < 0، يتم إنشاء النوع abs(ITYPE) ثم
!>             يتم تبديله من النهاية إلى النهاية (A(I,J) := A'(N-J,N-I).) انظر أيضًا
!>             وصف AMAGN و ISIGN.
!>
!>          أنواع خاصة:
!>          = 0:  مصفوفة الأصفار.
!>          = 1:  مصفوفة الوحدة.
!>          = 2:  كتلة جوردان منقولة (transposed Jordan block).
!>          = 3:  إذا كانت N فردية، فإن كتلة جوردان منقولة بحجم k+1 × k+1
!>                متبوعة بكتلة وحدة بحجم k × k، حيث k=(N-1)/2.
!>                إذا كانت N زوجية، فإن k=(N-2)/2، ويتم إضافة إدخال قطري صفري
!>                في النهاية.
!>
!>          أنواع القطر الرئيسي. يتكون القطر الرئيسي من NZ1 أصفار، ثم
!>             k=N-NZ1-NZ2 إدخالات غير صفرية. القطر الفرعي هو صفر. تحدد ITYPE
!>             إدخالات القطر الرئيسي غير الصفرية على النحو التالي:
!>          = 4:  1، ...، k
!>          = 5:  1، RCOND، ...، RCOND
!>          = 6:  1، ...، 1، RCOND
!>          = 7:  1، a، a^2، ...، a^(k-1)=RCOND
!>          = 8:  1، 1-d، 1-2*d، ...، 1-(k-1)*d=RCOND
!>          = 9:  أرقام عشوائية مختارة من (RCOND,1)
!>          = 10: أرقام عشوائية بتوزيع IDIST (انظر SLARND).
!>

N

!>          N هو عدد صحيح (INTEGER).
!>          رتبة المصفوفة.
!>

NZ1

!>          NZ1 هو عدد صحيح (INTEGER).
!>          إذا كانت abs(ITYPE) > 3، فإن أول NZ1 إدخالات قطرية ستكون
!>          أصفارًا.
!>

NZ2

!>          NZ2 هو عدد صحيح (INTEGER).
!>          إذا كانت abs(ITYPE) > 3، فإن آخر NZ2 إدخالات قطرية ستكون
!>          أصفارًا.
!>

ISIGN

!>          ISIGN هو عدد صحيح (INTEGER).
!>          = 0: سيتم ترك إشارة إدخالات القطر الرئيسي والقطر الفرعي
!>               دون تغيير.
!>          = 1: سيتم تغيير إشارة إدخالات القطر الرئيسي والقطر الفرعي
!>               عشوائيًا.
!>          = 2: إذا كانت ITYPE تساوي 2 أو 3، فإنها تكون مثل ISIGN=1.
!>               بخلاف ذلك، باحتمالية 0.5، سيتم تحويل أزواج الإدخالات القطرية
!>               الفردية-الزوجية A(2*j-1,2*j-1)، A(2*j,2*j) إلى كتلة 2×2 عن طريق
!>               الضرب المسبق واللاحق بدورانات متعامدة عشوائية مميزة.
!>               سيتم تغيير إشارة الإدخالات القطرية المتبقية عشوائيًا.
!>

AMAGN

!>          AMAGN هو عدد حقيقي (REAL).
!>          سيتم ضرب إدخالات القطر الرئيسي والقطر الفرعي في
!>          AMAGN.
!>

RCOND

!>          RCOND هو عدد حقيقي (REAL).
!>          إذا كانت abs(ITYPE) > 4، فإن أصغر إدخال قطري سيكون
!>          إدخالًا سيكون RCOND. يجب أن يكون RCOND بين 0 و 1.
!>

TRIANG

!>          TRIANG هو عدد حقيقي (REAL).
!>          سيتم ملء الإدخالات فوق القطر الرئيسي بأرقام عشوائية
!>          بمقدار محدد بـ TRIANG (أي، أرقام عشوائية مضروبة
!>          في TRIANG).
!>

IDIST

!>          IDIST هو عدد صحيح (INTEGER).
!>          يحدد نوع التوزيع الذي سيتم استخدامه لإنشاء
!>          مصفوفة عشوائية.
!>          = 1:  توزيع منتظم (0, 1)
!>          = 2:  توزيع منتظم (-1, 1)
!>          = 3:  توزيع طبيعي (0, 1)
!>

ISEED

!>          ISEED هو مصفوفة أعداد صحيحة (INTEGER array)، بأبعاد (4).
!>          عند الإدخال، تحدد ISEED بذرة مولد الأرقام العشوائية.
!>          تتغير قيم ISEED عند الخروج، ويمكن
!>          استخدامها في الاستدعاء التالي لـ SLATM4 لمواصلة نفس
!>          تسلسل الأرقام العشوائية.
!>          ملاحظة: يجب أن يكون ISEED(4) فرديًا، لمولد الأرقام العشوائية
!>          المستخدم حاليًا.
!>

A

!>          A هي مصفوفة أعداد حقيقية (REAL array)، بأبعاد (LDA, N).
!>          المصفوفة المراد حسابها.
!>

LDA

!>          LDA هو عدد صحيح (INTEGER).
!>          البعد الرائد للمصفوفة A. يجب أن يكون على الأقل 1 وعلى الأقل N.
!>
