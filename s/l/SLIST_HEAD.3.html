SLIST(3) دليل وظائف المكتبة SLIST(3)

الاسم

SLIST_EMPTY, SLIST_ENTRY, SLIST_FIRST, SLIST_FOREACH, SLIST_HEAD, SLIST_HEAD_INITIALIZER, SLIST_INIT, SLIST_INSERT_AFTER, SLIST_INSERT_HEAD, SLIST_NEXT, SLIST_REMOVE, SLIST_REMOVE_HEAD - تطبيق قائمة مفردة متصلة

المكتبة

مكتبة C القياسية (libc، -lc)

ملخص

#include <sys/queue.h>
SLIST_ENTRY(TYPE);
SLIST_HEAD(HEADNAME, TYPE);
SLIST_HEAD SLIST_HEAD_INITIALIZER(SLIST_HEAD head);
void SLIST_INIT(SLIST_HEAD *head);
int SLIST_EMPTY(SLIST_HEAD *head);
void SLIST_INSERT_HEAD(SLIST_HEAD *head,
                        struct TYPE *elm, SLIST_ENTRY NAME);
void SLIST_INSERT_AFTER(struct TYPE *listelm,
                        struct TYPE *elm, SLIST_ENTRY NAME);
struct TYPE *SLIST_FIRST(SLIST_HEAD *head);
struct TYPE *SLIST_NEXT(struct TYPE *elm, SLIST_ENTRY NAME);
SLIST_FOREACH(struct TYPE *var, SLIST_HEAD *head, SLIST_ENTRY NAME);
void SLIST_REMOVE(SLIST_HEAD *head, struct TYPE *elm, TYPE,
                        SLIST_ENTRY NAME);
void SLIST_REMOVE_HEAD(SLIST_HEAD *head,
                        SLIST_ENTRY NAME);

الوصف

تعرّف هذه الماكرويات وتعمل على قوائم مفردة متصلة.

في تعريفات الماكرو، TYPE هو اسم بنية معرفة من قبل المستخدم، والتي يجب أن تحتوي على حقل من نوع SLIST_ENTRY، اسمه NAME. الوسيط HEADNAME هو اسم بنية معرفة من قبل المستخدم والتي يجب تعريفها باستخدام الماكرو SLIST_HEAD().

الإنشاء

يتم استهلال قائمة مفردة متصلة بواسطة بنية معرفة بواسطة الماكرو SLIST_HEAD(). تحتوي هذه البنية على مؤشر واحد للعنصر الأول في القائمة. تتصل العناصر بشكل مفرد لتقليل استهلاك المساحة وتعقيد معالجة المؤشرات، مقابل تكلفة O(n) للإزالة للعناصر العشوائية. يمكن إضافة عناصر جديدة إلى القائمة بعد عنصر موجود أو في بداية القائمة. يتم تعريف بنية SLIST_HEAD على النحو التالي:


SLIST_HEAD(HEADNAME, TYPE) head;

حيث struct HEADNAME هي البنية التي سيتم تعريفها، و struct TYPE هو نوع العناصر التي سيتم ربطها في القائمة. يمكن لاحقًا تعريف مؤشر إلى رأس القائمة على النحو التالي:


struct HEADNAME *headp;

(الأسماء head و headp قابلة للاختيار من قبل المستخدم.)

تعرّف SLIST_ENTRY() بنية تربط العناصر في القائمة.

SLIST_HEAD_INITIALIZER() يُقيّم إلى مُهيئ للقائمة head.

SLIST_INIT() تهيئ القائمة المشار إليها بواسطة head.

SLIST_EMPTY() يُقيّم إلى صحيح إذا لم تكن هناك عناصر في القائمة.

الإدراج

SLIST_INSERT_HEAD() يُدرج العنصر الجديد elm في بداية القائمة.

SLIST_INSERT_AFTER() يُدرج العنصر الجديد elm بعد العنصر listelm.

اجتياز

SLIST_FIRST() تُرجع العنصر الأول في القائمة، أو NULL إذا كانت القائمة فارغة.

SLIST_NEXT() تُرجع العنصر التالي في القائمة.

SLIST_FOREACH() تجتاز القائمة المشار إليها بواسطة head في الاتجاه الأمامي، وتعيّن كل عنصر بدوره إلى var.

الإزالة

SLIST_REMOVE() تزيل العنصر elm من القائمة.

SLIST_REMOVE_HEAD() تزيل العنصر elm من بداية القائمة. لتحقيق أقصى قدر من الكفاءة، يجب على العناصر التي تتم إزالتها من بداية القائمة استخدام هذه الماكرو بشكل صريح بدلاً من SLIST_REMOVE() العامة.

قيمة الإرجاع

SLIST_EMPTY() تُرجع غير صفري إذا كانت القائمة فارغة، وصفر إذا كانت القائمة تحتوي على إدخال واحد على الأقل.

SLIST_FIRST() و SLIST_NEXT() تُرجع مؤشرًا إلى بنية TYPE الأولى أو التالية، على التوالي.

SLIST_HEAD_INITIALIZER() تُرجع مُهيئًا يمكن تعيينه إلى القائمة head.

المعايير

BSD.

التاريخ

4.4BSD.

أخطاء

SLIST_FOREACH() لا تسمح بإزالة var أو تحريرها داخل الحلقة، لأن ذلك قد يتعارض مع الاجتياز. SLIST_FOREACH_SAFE()، الموجودة في BSD ولكن غير موجودة في glibc، تعالج هذا القيد عن طريق السماح بإزالة var بأمان من القائمة وتحريرها من داخل الحلقة دون التأثير على الاجتياز.

أمثلة

#include <stddef.h>
#include <stdio.h>
#include <stdlib.h>
#include <sys/queue.h>
struct entry {
    int data;
    SLIST_ENTRY(entry) entries;             /* قائمة مفردة متصلة */
};
SLIST_HEAD(slisthead, entry);
int
main(void)
{
    struct entry *n1, *n2, *n3, *np;
    struct slisthead head;                  /* رأس قائمة مفردة متصلة */
    SLIST_INIT(&head);                      /* تهيئة القائمة */
    n1 = malloc(sizeof(struct entry));      /* الإدراج في البداية */
    SLIST_INSERT_HEAD(&head, n1, entries);
    n2 = malloc(sizeof(struct entry));      /* الإدراج بعد */
    SLIST_INSERT_AFTER(n1, n2, entries);
    SLIST_REMOVE(&head, n2, entry, entries);/* حذف */
    free(n2);
    n3 = SLIST_FIRST(&head);
    SLIST_REMOVE_HEAD(&head, entries);      /* حذف من البداية */
    free(n3);
    for (unsigned int i = 0; i < 5; i++) {
        n1 = malloc(sizeof(struct entry));
        SLIST_INSERT_HEAD(&head, n1, entries);
        n1->data = i;
    }
                                            /* الاجتياز الأمامي */
    SLIST_FOREACH(np, &head, entries)
        printf("%i\n", np->data);
    while (!SLIST_EMPTY(&head)) {           /* حذف القائمة */
        n1 = SLIST_FIRST(&head);
        SLIST_REMOVE_HEAD(&head, entries);
        free(n1);
    }
    SLIST_INIT(&head);
    exit(EXIT_SUCCESS);
}
