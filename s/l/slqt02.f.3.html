توثيق الوظيفة/الإجراء الفرعي

subroutine slqt02 (integer m, integer n, integer k, real, dimension( lda, * ) a, real, dimension( lda, * ) af, real, dimension( lda, * ) q, real, dimension( lda, * ) l, integer lda, real, dimension( * ) tau, real, dimension( lwork ) work, integer lwork, real, dimension( * ) rwork, real, dimension( * ) result)

SLQT02

الغرض:

!>
!> SLQT02 تختبر SORGLQ، التي تولد مصفوفة Q بحجم m*n ذات صفوف
!> متعامدة معرفة كناتج لمُنعكسات أولية k.
!>
!> بالنظر إلى تحليل LQ للمصفوفة A بحجم m*n، تولد SLQT02
!> المصفوفة المتعامدة Q المعرفة بتحليل أول k صفوف
!> من A؛ وهي تقارن L(1:k,1:m) مع A(1:k,1:n)*Q(1:m,1:n)'،
!> وتتحقق من أن صفوف Q متعامدة.
!>

المعلمات

M 
!>          M هو عدد صحيح (INTEGER)
!>          عدد صفوف المصفوفة Q التي سيتم إنشاؤها. M >= 0.
!>

N

!>          N هو عدد صحيح (INTEGER)
!>          عدد أعمدة المصفوفة Q التي سيتم إنشاؤها.
!>          N >= M >= 0.
!>

K

!>          K هو عدد صحيح (INTEGER)
!>          عدد المُنعكسات الأولية التي يحدد ناتجها المصفوفة Q.
!>          M >= K >= 0.
!>

A

!>          A هي مصفوفة أعداد حقيقية (REAL array)، البعد (LDA,N)
!>          المصفوفة A بحجم m*n التي تم تحليلها بواسطة SLQT01.
!>

AF

!>          AF هي مصفوفة أعداد حقيقية (REAL array)، البعد (LDA,N)
!>          تفاصيل تحليل LQ للمصفوفة A، كما تم إرجاعها بواسطة SGELQF.
!>          راجع SGELQF لمزيد من التفاصيل.
!>

Q

!>          Q هي مصفوفة أعداد حقيقية (REAL array)، البعد (LDA,N)
!>

L

!>          L هي مصفوفة أعداد حقيقية (REAL array)، البعد (LDA,M)
!>

LDA

!>          LDA هو عدد صحيح (INTEGER)
!>          البعد الرائد للمصفوفات A، AF، Q و L. LDA >= N.
!>

TAU

!>          TAU هي مصفوفة أعداد حقيقية (REAL array)، البعد (M)
!>          العوامل القياسية للمُنعكسات الأولية المقابلة
!>          لتحليل LQ في AF.
!>

WORK

!>          WORK هي مصفوفة أعداد حقيقية (REAL array)، البعد (LWORK)
!>

LWORK

!>          LWORK هو عدد صحيح (INTEGER)
!>          بعد المصفوفة WORK.
!>

RWORK

!>          RWORK هي مصفوفة أعداد حقيقية (REAL array)، البعد (M)
!>

RESULT

!>          RESULT هي مصفوفة أعداد حقيقية (REAL array)، البعد (2)
!>          نسب الاختبار:
!>          RESULT(1) = norm( L - A*Q' ) / ( N * norm(A) * EPS )
!>          RESULT(2) = norm( I - Q*Q' ) / ( N * EPS )
!>
