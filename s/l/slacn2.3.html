توثيق الوظيفة/الروتين الفرعي

subroutine slacn2 (integer n, real, dimension( * ) v, real, dimension( * ) x, integer, dimension( * ) isgn, real est, integer kase, integer, dimension( 3 ) isave)

SLACN2 يقدر المعيار 1 لمصفوفة مربعة حقيقية، باستخدام الاتصال العكسي لتقييم نواتج ضرب المصفوفة والمتجه.

الغرض:

!>
!> SLACN2 يقدر المعيار 1 لمصفوفة مربعة حقيقية A.
!> يتم استخدام الاتصال العكسي لتقييم نواتج ضرب المصفوفة والمتجه.
!>

المعلمات

N 
!>          N هو عدد صحيح (INTEGER)
!>         رتبة المصفوفة.  N >= 1.
!>

V

!>          V هي مصفوفة حقيقية (REAL array)، البعد (N)
!>         عند العودة النهائية، V = A*W،  حيث  EST = norm(V)/norm(W)
!>         (W لا يتم إرجاعه).
!>

X

!>          X هي مصفوفة حقيقية (REAL array)، البعد (N)
!>         عند العودة الوسيطة، يجب استبدال X بـ
!>               A * X,   إذا كان KASE=1،
!>               A**T * X,  إذا كان KASE=2،
!>         ويجب إعادة استدعاء SLACN2 مع بقاء جميع المعلمات الأخرى
!>         دون تغيير.
!>

ISGN

!>          ISGN هي مصفوفة أعداد صحيحة (INTEGER array)، البعد (N)
!>

EST

!>          EST هو قيمة حقيقية (REAL)
!>         عند الإدخال مع KASE = 1 أو 2 و ISAVE(1) = 3، يجب أن تظل EST
!>         دون تغيير من الاستدعاء السابق لـ SLACN2.
!>         عند الخروج، EST هو تقدير (حد أدنى) للمعيار (norm) لـ (A).
!>

KASE

!>          KASE هو عدد صحيح (INTEGER)
!>         عند الاستدعاء الأولي لـ SLACN2، يجب أن يكون KASE هو 0.
!>         عند العودة الوسيطة، سيكون KASE هو 1 أو 2، مما يشير
!>         إلى ما إذا كان يجب استبدال X بـ A * X  أو A**T * X.
!>         عند العودة النهائية من SLACN2، سيكون KASE مرة أخرى 0.
!>

ISAVE

!>          ISAVE هي مصفوفة أعداد صحيحة (INTEGER array)، البعد (3)
!>         يتم استخدام ISAVE لحفظ المتغيرات بين استدعاءات SLACN2
!>
