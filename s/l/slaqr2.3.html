subroutine slaqr2 (wantt, wantz, n, ktop, kbot, nw, h, ldh, iloz, ihiz, z, ldz, ns, nd, sr, si, v, ldv, nh, t, ldt, nv, wv, ldwv, work, lwork)
SLAQR2 تنفذ التحويل المتشابه المتعامد لمصفوفة هسنبرغ للكشف عن قيم ذاتية متقاربة بالكامل وتفكيكها من مصفوفة فرعية رئيسية تابعة (فك تجميع مبكر عدواني).

توثيق الوظيفة/الإجراء الفرعي

الإجراء الفرعي slaqr2 (logical wantt, logical wantz, integer n, integer ktop, integer kbot, integer nw, real, dimension( ldh, * ) h, integer ldh, integer iloz, integer ihiz, real, dimension( ldz, * ) z, integer ldz, integer ns, integer nd, real, dimension( * ) sr, real, dimension( * ) si, real, dimension( ldv, * ) v, integer ldv, integer nh, real, dimension( ldt, * ) t, integer ldt, integer nv, real, dimension( ldwv, * ) wv, integer ldwv, real, dimension( * ) work, integer lwork)

SLAQR2 تنفذ التحويل المتشابه المتعامد لمصفوفة هسنبرغ للكشف عن قيم ذاتية متقاربة بالكامل وتفكيكها من مصفوفة فرعية رئيسية تابعة (فك تجميع مبكر عدواني).

الغرض:

!>
!>    SLAQR2 مطابق لـ SLAQR3 باستثناء أنه يتجنب
!>    الاستدعاء التكراري عن طريق استدعاء SLAHQR بدلاً من SLAQR4.
!>
!>    فك تجميع مبكر عدواني:
!>
!>    يقبل هذا الإجراء الفرعي كمُدخل مصفوفة هسنبرغ عليا
!>    H وينفذ تحويلاً متشابهاً متعامداً
!>    مصمماً للكشف عن القيم الذاتية المتقاربة بالكامل وتفكيكها من
!>    مصفوفة فرعية رئيسية تابعة. عند الخروج، تم استبدال H بـ
!>    مصفوفة هسنبرغ جديدة تمثل اضطراباً لـ
!>    تحويل متشابه متعامد لـ H. يُؤمل أن يكون الإصدار النهائي لـ H يحتوي على العديد من الإدخالات الفرعية القطرية الصفرية.
!>

المعلمات

WANTT 
!>          WANTT هو LOGICAL (منطقي)
!>          إذا كانت .TRUE.، يتم تحديث مصفوفة هسنبرغ H بالكامل
!>          بحيث يمكن حساب العامل شبه المائل لـ Schur (بالتعاون مع الإجراء الفرعي المستدعي).
!>          إذا كانت .FALSE.، يتم تحديث جزء فقط من H للحفاظ على
!>          القيم الذاتية.
!>

WANTZ

!>          WANTZ هو LOGICAL (منطقي)
!>          إذا كانت .TRUE.، يتم تحديث المصفوفة المتعامدة Z
!>          بحيث يمكن حساب عامل Schur المتعامد (بالتعاون مع الإجراء الفرعي المستدعي).
!>          إذا كانت .FALSE.، فلن يتم الرجوع إلى Z.
!>

N

!>          N هو INTEGER (عدد صحيح)
!>          رتبة المصفوفة H (وإذا كانت WANTZ هي .TRUE.) رتبة المصفوفة المتعامدة Z.
!>

KTOP

!>          KTOP هو INTEGER (عدد صحيح)
!>          يُفترض أن KTOP = 1 أو H(KTOP,KTOP-1)=0.
!>          تحدد KBOT و KTOP معًا كتلة معزولة
!>          على طول قطر مصفوفة هسنبرغ.
!>

KBOT

!>          KBOT هو INTEGER (عدد صحيح)
!>          يُفترض دون فحص أن KBOT = N أو H(KBOT+1,KBOT)=0.
!>          تحدد KBOT و KTOP معًا كتلة معزولة
!>          على طول قطر مصفوفة هسنبرغ.
!>

NW

!>          NW هو INTEGER (عدد صحيح)
!>          حجم نافذة فك التجميع. 1 <= NW <= (KBOT-KTOP+1).
!>

H

!>          H هو مصفوفة REAL (حقيقية)، ذات بُعد (LDH,N)
!>          عند الإدخال، يخزن القسم N×N الأولي لـ H
!>          مصفوفة هسنبرغ التي تخضع لفك تجميع مبكر عدواني.
!>          عند الخروج، تم تحويل H بواسطة تحويل متشابه متعامد،
!>          تم إزاحتها، وأعيدت
!>          إلى شكل هسنبرغ الذي (يُؤمل) يحتوي على بعض
!>          الإدخالات الفرعية القطرية الصفرية.
!>

LDH

!>          LDH هو INTEGER (عدد صحيح)
!>          البعد الرائد لـ H كما تم التصريح عنه في الإجراء الفرعي المستدعي.
!>          N <= LDH
!>

ILOZ

!>          ILOZ هو INTEGER (عدد صحيح)
!>

IHIZ

!>          IHIZ هو INTEGER (عدد صحيح)
!>          حدد صفوف Z التي يجب تطبيق التحويلات عليها إذا كانت WANTZ هي .TRUE..
!>          1 <= ILOZ <= IHIZ <= N.
!>

Z

!>          Z هو مصفوفة REAL (حقيقية)، ذات بُعد (LDZ,N)
!>          إذا كانت WANTZ هي .TRUE.، فعند الخروج، تم تجميع التحويل
!>          المتشابه المتعامد المذكور أعلاه
!>          في Z(ILOZ:IHIZ,ILOZ:IHIZ) من اليمين.
!>          إذا كانت WANTZ هي .FALSE.، فلن يتم الرجوع إلى Z.
!>

LDZ

!>          LDZ هو INTEGER (عدد صحيح)
!>          البعد الرائد لـ Z كما تم التصريح عنه في
!>          الإجراء الفرعي المستدعي. 1 <= LDZ.
!>

NS

!>          NS هو INTEGER (عدد صحيح)
!>          عدد القيم الذاتية غير المتقاربة (أي التقريبية)
!>          التي تم إرجاعها في SR و SI والتي يمكن استخدامها كإزاحات بواسطة
!>          الإجراء الفرعي المستدعي.
!>

ND

!>          ND هو INTEGER (عدد صحيح)
!>          عدد القيم الذاتية المتقاربة التي كشفها هذا
!>          الإجراء الفرعي.
!>

SR

!>          SR هو مصفوفة REAL (حقيقية)، ذات بُعد (KBOT)
!>

SI

!>          SI هو مصفوفة REAL (حقيقية)، ذات بُعد (KBOT)
!>          عند الخروج، تم تخزين الأجزاء الحقيقية والتخيلية للقيم
!>          الذاتية التقريبية التي يمكن استخدامها للإزاحات في
!>          SR(KBOT-ND-NS+1) حتى SR(KBOT-ND) و
!>          SI(KBOT-ND-NS+1) حتى SI(KBOT-ND)، على التوالي.
!>          تم تخزين الأجزاء الحقيقية والتخيلية للقيم الذاتية
!>          المتقاربة في SR(KBOT-ND+1) حتى SR(KBOT) و
!>          SI(KBOT-ND+1) حتى SI(KBOT)، على التوالي.
!>

V

!>          V هو مصفوفة REAL (حقيقية)، ذات بُعد (LDV,NW)
!>          مصفوفة عمل NW×NW.
!>

LDV

!>          LDV هو INTEGER (عدد صحيح)
!>          البعد الرائد لـ V كما تم التصريح عنه في
!>          الإجراء الفرعي المستدعي. NW <= LDV
!>

NH

!>          NH هو INTEGER (عدد صحيح)
!>          عدد أعمدة T. NH >= NW.
!>

T

!>          T هو مصفوفة REAL (حقيقية)، ذات بُعد (LDT,NW)
!>

LDT

!>          LDT هو INTEGER (عدد صحيح)
!>          البعد الرائد لـ T كما تم التصريح عنه في
!>          الإجراء الفرعي المستدعي. NW <= LDT
!>

NV

!>          NV هو INTEGER (عدد صحيح)
!>          عدد صفوف مصفوفة العمل WV المتاحة كـ
!>          مساحة عمل. NV >= NW.
!>

WV

!>          WV هو مصفوفة REAL (حقيقية)، ذات بُعد (LDWV,NW)
!>

LDWV

!>          LDWV هو INTEGER (عدد صحيح)
!>          البعد الرائد لـ W كما تم التصريح عنه في
!>          الإجراء الفرعي المستدعي. NW <= LDV
!>

WORK

!>          WORK هو مصفوفة REAL (حقيقية)، ذات بُعد (LWORK)
!>          عند الخروج، يتم تعيين WORK(1) إلى تقدير للقيمة المثلى
!>          لـ LWORK للقيم المعطاة لـ N, NW, KTOP و KBOT.
!>

LWORK

!>          LWORK هو INTEGER (عدد صحيح)
!>          بُعد مصفوفة العمل WORK. LWORK = 2*NW
!>          يكفي، ولكن قد ينتج عن القيم الأكبر لـ LWORK كفاءة أكبر.
!>
!>          إذا كانت LWORK = -1، فيُفترض استعلام عن مساحة العمل؛ SLAQR2
!>          يقدر فقط حجم مساحة العمل المثلى للقيم المعطاة
!>          لـ N, NW, KTOP و KBOT. يتم إرجاع التقدير
!>          في WORK(1). لا يتم إصدار أي رسالة خطأ متعلقة بـ LWORK
!>          بواسطة XERBLA. لا يتم الوصول إلى H أو Z.
!>
