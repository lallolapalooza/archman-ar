توثيق الوظيفة/الروتين الفرعي

subroutine slaexc (logical wantq, integer n, real, dimension( ldt, * ) t, integer ldt, real, dimension( ldq, * ) q, integer ldq, integer j1, integer n1, integer n2, real, dimension( * ) work, integer info)

SLAEXC تبادل الكتل القطرية المتجاورة لمصفوفة شبه مثلثة علوية حقيقية في الشكل القانوني لشور، عن طريق تحويل تشابه متعامد.

الغرض:

!>
!> SLAEXC تبادل الكتل القطرية المتجاورة T11 و T22 ذات الرتبة 1 أو 2 في
!> مصفوفة شبه مثلثة علوية T عن طريق تحويل تشابه متعامد.
!>
!> يجب أن تكون T في الشكل القانوني لشور، أي، مصفوفة مثلثة علوية مقسمة
!> إلى كتل ذات كتل قطرية بحجم 1x1 و 2x2؛ كل كتلة قطرية بحجم 2x2
!> لها عناصر قطرية متساوية وعناصر خارج القطر متعاكسة.
!>

المعاملات

WANTQ 
!>          WANTQ هو LOGICAL
!>          = .TRUE. : تجميع التحويل في المصفوفة Q؛
!>          = .FALSE.: عدم تجميع التحويل.
!>

N

!>          N هو INTEGER
!>          رتبة المصفوفة T. N >= 0.
!>

T

!>          T هي مصفوفة REAL، بعد(LDT,N)
!>          عند الإدخال، المصفوفة شبه المثلثة العلوية T، في الشكل
!>          القانوني لشور.
!>          عند الخروج، المصفوفة المحدثة T، مرة أخرى في الشكل القانوني لشور.
!>

LDT

!>          LDT هو INTEGER
!>          البعد الرئيسي للمصفوفة T. LDT >= max(1,N).
!>

Q

!>          Q هي مصفوفة REAL، بعد(LDQ,N)
!>          عند الإدخال، إذا كانت WANTQ هي .TRUE.، المصفوفة المتعامدة Q.
!>          عند الخروج، إذا كانت WANTQ هي .TRUE.، المصفوفة المحدثة Q.
!>          إذا كانت WANTQ هي .FALSE.، فإن Q غير مستخدمة.
!>

LDQ

!>          LDQ هو INTEGER
!>          البعد الرئيسي للمصفوفة Q.
!>          LDQ >= 1؛ وإذا كانت WANTQ هي .TRUE.، LDQ >= N.
!>

J1

!>          J1 هو INTEGER
!>          مؤشر الصف الأول للكتلة الأولى T11.
!>

N1

!>          N1 هو INTEGER
!>          رتبة الكتلة الأولى T11. N1 = 0، 1 أو 2.
!>

N2

!>          N2 هو INTEGER
!>          رتبة الكتلة الثانية T22. N2 = 0، 1 أو 2.
!>

WORK

!>          WORK هي مصفوفة REAL، بعد(N)
!>

INFO

!>          INFO هو INTEGER
!>          = 0: الخروج بنجاح
!>          = 1: المصفوفة المحولة T ستكون بعيدة جدًا عن الشكل القانوني لشور؛
!>               لم يتم تبديل الكتل و T و Q لم تتغير.
!>
