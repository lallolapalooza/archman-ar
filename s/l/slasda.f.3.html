subroutine slasda (integer icompq, integer smlsiz, integer n, integer sqre, real, dimension( * ) d, real, dimension( * ) e, real, dimension( ldu, * ) u, integer ldu, real, dimension( ldu, * ) vt, integer, dimension( * ) k, real, dimension( ldu, * ) difl, real, dimension( ldu, * ) difr, real, dimension( ldu, * ) z, real, dimension( ldu, * ) poles, integer, dimension( * ) givptr, integer, dimension( ldgcol, * ) givcol, integer ldgcol, integer, dimension( ldgcol, * ) perm, real, dimension( ldu, * ) givnum, real, dimension( * ) c, real, dimension( * ) s, real, dimension( * ) work, integer, dimension( * ) iwork, integer info)

SLASDA تحسب التحلل المنفرد (SVD) لمصفوفة علوية ثنائية قطرية حقيقية ذات قطر d وحافة e. تستخدم بواسطة sbdsdc.

الغرض:

!>
!> باستخدام نهج التقسيم والغزو، تحسب SLASDA التحلل المنفرد (SVD) لمصفوفة N-by-M علوية ثنائية قطرية حقيقية
!> B ذات قطر D وحافة E، حيث M = N + SQRE. تقوم الخوارزمية بحساب القيم المنفردة في SVD B = U * S * VT.
!> يتم حساب المتجهات المتعامدة U و VT اختياريًا في شكل مضغوط.
!>
!> روتين فرعي ذو صلة، SLASD0، يحسب القيم المنفردة والمتجهات المنفردة في شكل صريح.
!>

المعلمات

ICOMPQ 
!>          ICOMPQ هو عدد صحيح (INTEGER)
!>         يحدد ما إذا كان سيتم حساب المتجهات المنفردة
!>         في شكل مضغوط، على النحو التالي
!>         = 0: حساب القيم المنفردة فقط.
!>         = 1: حساب المتجهات المنفردة للمصفوفة العلوية الثنائية القطرية
!>              في شكل مضغوط.
!>

SMLSIZ

!>          SMLSIZ هو عدد صحيح (INTEGER)
!>         الحد الأقصى لحجم المشاكل الفرعية في أسفل
!>         شجرة الحساب.
!>

N

!>          N هو عدد صحيح (INTEGER)
!>         البعد الصف للمصفوفة العلوية الثنائية القطرية. هذا هو
!>         أيضًا بعد المصفوفة القطرية الرئيسية D.
!>

SQRE

!>          SQRE هو عدد صحيح (INTEGER)
!>         يحدد البعد العمودي للمصفوفة الثنائية القطرية.
!>         = 0: المصفوفة الثنائية القطرية لها بعد عمودي M = N؛
!>         = 1: المصفوفة الثنائية القطرية لها بعد عمودي M = N + 1.
!>

D

!>          D هي مصفوفة حقيقية (REAL array)، حجمها ( N )
!>         عند الإدخال تحتوي D على القطر الرئيسي للمصفوفة الثنائية القطرية.
!>         عند الخروج D، إذا كان INFO = 0، تحتوي على قيمها المنفردة.
!>

E

!>          E هي مصفوفة حقيقية (REAL array)، حجمها ( M-1 )
!>         تحتوي على مداخل المصفوفة الفرعية القطرية للمصفوفة الثنائية القطرية.
!>         عند الخروج، تم تدمير E.
!>

U

!>          U هي مصفوفة حقيقية (REAL array)،
!>         حجمها ( LDU, SMLSIZ ) إذا كان ICOMPQ = 1، ولا تتم الإشارة إليها
!>         إذا كان ICOMPQ = 0. إذا كان ICOMPQ = 1، عند الخروج، تحتوي U على
!>         مصفوفات المتجهات المنفردة اليسرى لجميع المشاكل الفرعية في
!>         المستوى السفلي.
!>

LDU

!>          LDU هو عدد صحيح (INTEGER)، LDU = > N.
!>         البعد الرئيسي للمصفوفات U، VT، DIFL، DIFR، POLES،
!>         GIVNUM، و Z.
!>

VT

!>          VT هي مصفوفة حقيقية (REAL array)،
!>         حجمها ( LDU, SMLSIZ+1 ) إذا كان ICOMPQ = 1، ولا تتم الإشارة إليها
!>         إذا كان ICOMPQ = 0. إذا كان ICOMPQ = 1، عند الخروج، تحتوي VT**T على
!>         مصفوفات المتجهات المنفردة اليمنى لجميع المشاكل الفرعية في
!>         المستوى السفلي.
!>

K

!>          K هي مصفوفة أعداد صحيحة (INTEGER array)، حجمها ( N )
!>         إذا كان ICOMPQ = 1، وحجمها 1 إذا كان ICOMPQ = 0.
!>         إذا كان ICOMPQ = 1، عند الخروج، K(I) هو بعد المعادلة العلمانية (secular equation)
!>         رقم I في شجرة الحساب.
!>

DIFL

!>          DIFL هي مصفوفة حقيقية (REAL array)، حجمها ( LDU, NLVL )،
!>         حيث NLVL = floor(log_2 (N/SMLSIZ))).
!>

DIFR

!>          DIFR هي مصفوفة حقيقية (REAL array)،
!>                  حجمها ( LDU, 2 * NLVL ) إذا كان ICOMPQ = 1 و
!>                  حجمها ( N ) إذا كان ICOMPQ = 0.
!>         إذا كان ICOMPQ = 1، عند الخروج، تسجل DIFL(1:N, I) و DIFR(1:N, 2 * I - 1)
!>         المسافات بين القيم المنفردة في المستوى I
!>         والقيم المنفردة في المستوى (I -1)، و
!>         تحتوي DIFR(1:N, 2 * I ) على عوامل التطبيع
!>         لمصفوفة المتجهات المنفردة اليمنى. انظر SLASD8 للتفاصيل.
!>

Z

!>          Z هي مصفوفة حقيقية (REAL array)،
!>                  حجمها ( LDU, NLVL ) إذا كان ICOMPQ = 1 و
!>                  حجمها ( N ) إذا كان ICOMPQ = 0.
!>         أول K عنصر في Z(1, I) تحتوي على مكونات
!>         متجه الصف المحدث بعد تعديل الانكماش (deflation-adjusted) للمشاكل الفرعية
!>         في المستوى I.
!>

POLES

!>          POLES هي مصفوفة حقيقية (REAL array)،
!>         حجمها ( LDU, 2 * NLVL ) إذا كان ICOMPQ = 1، ولا تتم الإشارة إليها
!>         إذا كان ICOMPQ = 0. إذا كان ICOMPQ = 1، عند الخروج، تحتوي POLES(1, 2*I - 1) و
!>         POLES(1, 2*I) على القيم المنفردة الجديدة والقديمة
!>         المشاركة في المعادلات العلمانية في المستوى I.
!>

GIVPTR

!>          GIVPTR هي مصفوفة أعداد صحيحة (INTEGER array)،
!>         حجمها ( N ) إذا كان ICOMPQ = 1، ولا تتم الإشارة إليها إذا
!>         كان ICOMPQ = 0. إذا كان ICOMPQ = 1، عند الخروج، يسجل GIVPTR( I )
!>         عدد دورانات جيفينز (Givens rotations) التي تم إجراؤها على المشكلة رقم I
!>         في شجرة الحساب.
!>

GIVCOL

!>          GIVCOL هي مصفوفة أعداد صحيحة (INTEGER array)،
!>         حجمها ( LDGCOL, 2 * NLVL ) إذا كان ICOMPQ = 1، ولا
!>         تتم الإشارة إليها إذا كان ICOMPQ = 0. إذا كان ICOMPQ = 1، عند الخروج، لكل I،
!>         تسجل GIVCOL(1, 2 *I - 1) و GIVCOL(1, 2 *I) مواقع
!>         دورانات جيفينز التي تم إجراؤها على المستوى I في
!>         شجرة الحساب.
!>

LDGCOL

!>          LDGCOL هو عدد صحيح (INTEGER)، LDGCOL = > N.
!>         البعد الرئيسي للمصفوفتين GIVCOL و PERM.
!>

PERM

!>          PERM هي مصفوفة أعداد صحيحة (INTEGER array)، حجمها ( LDGCOL, NLVL )
!>         إذا كان ICOMPQ = 1، ولا تتم الإشارة إليها
!>         إذا كان ICOMPQ = 0. إذا كان ICOMPQ = 1، عند الخروج، يسجل PERM(1, I)
!>         التباديل التي تم إجراؤها على المستوى I من شجرة الحساب.
!>

GIVNUM

!>          GIVNUM هي مصفوفة حقيقية (REAL array)،
!>         حجمها ( LDU,  2 * NLVL ) إذا كان ICOMPQ = 1، ولا
!>         تتم الإشارة إليها إذا كان ICOMPQ = 0. إذا كان ICOMPQ = 1، عند الخروج، لكل I،
!>         تسجل GIVNUM(1, 2 *I - 1) و GIVNUM(1, 2 *I) قيم
!>         C و S لدورانات جيفينز التي تم إجراؤها على المستوى I في
!>         شجرة الحساب.
!>

C

!>          C هي مصفوفة حقيقية (REAL array)،
!>         حجمها ( N ) إذا كان ICOMPQ = 1، وحجمها 1 إذا كان ICOMPQ = 0.
!>         إذا كان ICOMPQ = 1 ولم تكن المشكلة الفرعية I مربعة، عند الخروج،
!>         تحتوي C( I ) على قيمة C لدوران جيفينز مرتبط
!>         بالفضاء الصفري الأيمن للمشكلة الفرعية I.
!>

S

!>
