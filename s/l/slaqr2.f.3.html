توثيق الوظيفة/الروتين الفرعي

subroutine slaqr2 (logical wantt, logical wantz, integer n, integer ktop, integer kbot, integer nw, real, dimension( ldh, * ) h, integer ldh, integer iloz, integer ihiz, real, dimension( ldz, * ) z, integer ldz, integer ns, integer nd, real, dimension( * ) sr, real, dimension( * ) si, real, dimension( ldv, * ) v, integer ldv, integer nh, real, dimension( ldt, * ) t, integer ldt, integer nv, real, dimension( ldwv, * ) wv, integer ldwv, real, dimension( * ) work, integer lwork)

SLAQR2 يقوم بتحويل التشابه المتعامد لمصفوفة هسنبرغ لاكتشاف قيم ذاتية متقاربة بالكامل من مصفوفة فرعية رئيسية لاحقة (تقليص مبكر شره).

الغرض:

!>
!>    SLAQR2 مطابق لـ SLAQR3 باستثناء أنه يتجنب
!>    الاستدعاء المتكرر عن طريق استدعاء SLAHQR بدلاً من SLAQR4.
!>
!>    تقليص مبكر شره:
!>
!>    يستقبل هذا الروتين الفرعي كمُدخل مصفوفة هسنبرغ علوية
!>    H ويقوم بتحويل تشابه متعامد
!>    مصمم للكشف عن القيم الذاتية المتقاربة بالكامل وإزالتها من
!>    مصفوفة فرعية رئيسية لاحقة. عند الخرج، تم استبدال H بـ
!>    مصفوفة هسنبرغ جديدة وهي اضطراب لـ
!>    تحويل تشابه متعامد لـ H. يُؤمل أن النسخة النهائية لـ H بها العديد من
!>    الإدخالات الفرعية القطرية الصفرية.
!>

المعلمات

WANTT 
!>          WANTT هو LOGICAL
!>          إذا كانت .TRUE. ، فسيتم تحديث مصفوفة هسنبرغ H بالكامل
!>          بحيث يمكن حساب عامل Schur شبه المثلث (بالتعاون مع الروتين الفرعي المستدعي).
!>          إذا كانت .FALSE. ، فسيتم تحديث جزء كافٍ فقط من H للحفاظ على
!>          القيم الذاتية.
!>

WANTZ

!>          WANTZ هو LOGICAL
!>          إذا كانت .TRUE. ، فسيتم تحديث المصفوفة المتعامدة Z
!>          بحيث يمكن حساب عامل Schur المتعامد (بالتعاون مع الروتين الفرعي المستدعي).
!>          إذا كانت .FALSE. ، فلن تتم الإشارة إلى Z.
!>

N

!>          N هو INTEGER
!>          ترتيب المصفوفة H، و (إذا كانت WANTZ هي .TRUE.) ترتيب المصفوفة المتعامدة Z.
!>

KTOP

!>          KTOP هو INTEGER
!>          من المفترض أن KTOP = 1 أو H(KTOP,KTOP-1)=0.
!>          يحدد KBOT و KTOP معًا كتلة معزولة
!>          على طول قطر مصفوفة هسنبرغ.
!>

KBOT

!>          KBOT هو INTEGER
!>          من المفترض دون فحص أن KBOT = N أو H(KBOT+1,KBOT)=0.
!>          يحدد KBOT و KTOP معًا كتلة معزولة
!>          على طول قطر مصفوفة هسنبرغ.
!>

NW

!>          NW هو INTEGER
!>          حجم نافذة التقليص. 1 <= NW <= (KBOT-KTOP+1).
!>

H

!>          H هي مصفوفة REAL، أبعادها (LDH,N)
!>          عند الإدخال، يخزن القسم N-by-N الأولي لـ H
!>          مصفوفة هسنبرغ التي تخضع لتقليص مبكر شره.
!>          عند الخرج، تم تحويل H بواسطة تشابه متعامد
!>          اضطراب، وعادت
!>          إلى شكل هسنبرغ الذي (يؤمل) لديه بعض
!>          الإدخالات الفرعية القطرية الصفرية.
!>

LDH

!>          LDH هو INTEGER
!>          البعد الرائد لـ H تمامًا كما تم تعريفه في الروتين الفرعي المستدعي.
!>          N <= LDH
!>

ILOZ

!>          ILOZ هو INTEGER
!>

IHIZ

!>          IHIZ هو INTEGER
!>          حدد صفوف Z التي يجب تطبيق التحويلات عليها
!>          إذا كانت WANTZ هي .TRUE.. 1 <= ILOZ <= IHIZ <= N.
!>

Z

!>          Z هي مصفوفة REAL، أبعادها (LDZ,N)
!>          إذا كانت WANTZ هي .TRUE.، فعند الخرج، تم تجميع
!>          تحويل التشابه المتعامد المذكور أعلاه
!>          في Z(ILOZ:IHIZ,ILOZ:IHIZ) من اليمين.
!>          إذا كانت WANTZ هي .FALSE.، فلن تتم الإشارة إلى Z.
!>

LDZ

!>          LDZ هو INTEGER
!>          البعد الرائد لـ Z تمامًا كما تم تعريفه في
!>          الروتين الفرعي المستدعي. 1 <= LDZ.
!>

NS

!>          NS هو INTEGER
!>          عدد القيم الذاتية غير المتقاربة (أي التقريبية)
!>          المُرجعة في SR و SI والتي يمكن استخدامها كإزاحات بواسطة
!>          الروتين الفرعي المستدعي.
!>

ND

!>          ND هو INTEGER
!>          عدد القيم الذاتية المتقاربة التي كشفتها هذه
!>          الروتين الفرعي.
!>

SR

!>          SR هي مصفوفة REAL، أبعادها (KBOT)
!>

SI

!>          SI هي مصفوفة REAL، أبعادها (KBOT)
!>          عند الخرج، الأجزاء الحقيقية والمخيلة للقيم الذاتية التقريبية
!>          التي يمكن استخدامها للإزاحات مخزنة في
!>          SR(KBOT-ND-NS+1) حتى SR(KBOT-ND) و
!>          SI(KBOT-ND-NS+1) حتى SI(KBOT-ND)، على التوالي.
!>          الأجزاء الحقيقية والمخيلة للقيم الذاتية المتقاربة
!>          مخزنة في SR(KBOT-ND+1) حتى SR(KBOT) و
!>          SI(KBOT-ND+1) حتى SI(KBOT)، على التوالي.
!>

V

!>          V هي مصفوفة REAL، أبعادها (LDV,NW)
!>          مصفوفة عمل بحجم NW-by-NW.
!>

LDV

!>          LDV هو INTEGER
!>          البعد الرائد لـ V تمامًا كما تم تعريفه في
!>          الروتين الفرعي المستدعي. NW <= LDV
!>

NH

!>          NH هو INTEGER
!>          عدد أعمدة T. NH >= NW.
!>

T

!>          T هي مصفوفة REAL، أبعادها (LDT,NW)
!>

LDT

!>          LDT هو INTEGER
!>          البعد الرائد لـ T تمامًا كما تم تعريفه في
!>          الروتين الفرعي المستدعي. NW <= LDT
!>

NV

!>          NV هو INTEGER
!>          عدد صفوف مصفوفة العمل WV المتاحة لـ
!>          مساحة العمل. NV >= NW.
!>

WV

!>          WV هي مصفوفة REAL، أبعادها (LDWV,NW)
!>

LDWV

!>          LDWV هو INTEGER
!>          البعد الرائد لـ W تمامًا كما تم تعريفه في
!>          الروتين الفرعي المستدعي. NW <= LDV
!>

WORK

!>          WORK هي مصفوفة REAL، أبعادها (LWORK)
!>          عند الخروج، يتم تعيين WORK(1) إلى تقدير للقيمة المثلى
!>          لـ LWORK للقيم المعطاة لـ N، NW، KTOP، و KBOT.
!>

LWORK

!>          LWORK هو INTEGER
!>          بعد مصفوفة العمل WORK. LWORK = 2*NW
!>          كافٍ، ولكن قد ينتج عن قيم أكبر لـ LWORK كفاءة أكبر.
!>
!>          إذا كانت LWORK = -1، فعندئذٍ يُفترض استعلام مساحة العمل؛ SLAQR2
!>          يقدر فقط حجم مساحة العمل المثلى للقيم المعطاة
!>          لـ N، NW، KTOP، و KBOT. يتم إرجاع التقدير
!>          في WORK(1). لا يتم إصدار رسالة خطأ متعلقة بـ LWORK
!>          بواسطة XERBLA. لا يتم الوصول إلى H ولا Z.
!>
