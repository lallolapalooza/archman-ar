subroutine slarz (character side, integer m, integer n, integer l, real, dimension( * ) v, integer incv, real tau, real, dimension( ldc, * ) c, integer ldc, real, dimension( * ) work)

SLARZ يطبق عاكس أولي (كما تم إرجاعه بواسطة stzrzf) على مصفوفة عامة.

الغرض:

!>
!> SLARZ يطبق عاكس أولي حقيقي H على مصفوفة C عامة حقيقية M-by-N
!> ، من اليسار أو اليمين. يتم تمثيل H بالشكل
!>
!>       H = I - tau * v * v**T
!>
!> حيث tau عدد حقيقي و v متجه حقيقي.
!>
!> إذا كان tau = 0، يتم اعتبار H كمصفوفة الوحدة.
!>
!>
!> H هو حاصل ضرب k من العواكس الأولية كما تم إرجاعه بواسطة STZRZF.
!>

المعلمات

SIDE 
!>          SIDE هو CHARACTER*1
!>          = 'L': تشكيل  H * C
!>          = 'R': تشكيل  C * H
!>

M

!>          M هو INTEGER
!>          عدد صفوف المصفوفة C.
!>

N

!>          N هو INTEGER
!>          عدد أعمدة المصفوفة C.
!>

L

!>          L هو INTEGER
!>          عدد عناصر المتجه V التي تحتوي على
!>          الجزء الهادف من متجهات Householder.
!>          إذا كان SIDE = 'L'، فإن M >= L >= 0، إذا كان SIDE = 'R'، فإن N >= L >= 0.
!>

V

!>          V هو مصفوفة REAL، البعد (1+(L-1)*abs(INCV))
!>          المتجه v في تمثيل H كما تم إرجاعه بواسطة
!>          STZRZF. لا يتم استخدام V إذا كان TAU = 0.
!>

INCV

!>          INCV هو INTEGER
!>          الزيادة بين عناصر v. INCV <> 0.
!>

TAU

!>          TAU هو REAL
!>          القيمة tau في تمثيل H.
!>

C

!>          C هو مصفوفة REAL، البعد (LDC,N)
!>          عند الإدخال، المصفوفة C بحجم M-by-N.
!>          عند الخروج، يتم الكتابة فوق C بالمصفوفة H * C إذا كان SIDE = 'L'،
!>          أو C * H إذا كان SIDE = 'R'.
!>

LDC

!>          LDC هو INTEGER
!>          البعد الرئيسي للمصفوفة C. LDC >= max(1,M).
!>

WORK

!>          WORK هو مصفوفة REAL، البعد
!>                         (N) إذا كان SIDE = 'L'
!>                      أو (M) إذا كان SIDE = 'R'
!>
