subroutine slafts (character*3 type, integer m, integer n, integer imat, integer ntests, real, dimension( * ) result, integer, dimension( 4 ) iseed, real thresh, integer iounit, integer ie)

SLAFTS

الغرض:

!>
!>    SLAFTS تختبر متجه النتيجة مقابل قيمة الحد لرؤية الاختبارات
!>    التي فشلت في اجتياز الحد لنوع المصفوفة هذا.
!>    يتم الإخراج إلى الملف المحدد بالوحدة IOUNIT.
!> 
!>  TYPE   - CHARACTER*3
!>           عند الدخول، يحدد TYPE نوع المصفوفة المستخدم في
!>           رسائل الطباعة.
!>           لم يتم التعديل.
!>
!>  N      - INTEGER
!>           عند الدخول، يحدد N رتبة مصفوفة الاختبار.
!>           لم يتم التعديل.
!>
!>  IMAT   - INTEGER
!>           عند الدخول، يحدد IMAT نوع مصفوفة الاختبار.
!>           سيتم طباعة قائمة بالأنواع المختلفة بواسطة SLAHD2
!>           إلى ملف الإخراج إذا فشل اختبار في اجتياز الحد.
!>           لم يتم التعديل.
!>
!>  NTESTS - INTEGER
!>           عند الدخول، NTESTS هو عدد الاختبارات التي تم إجراؤها على
!>           الإجراءات الفرعية في المسار المحدد بواسطة TYPE.
!>           لم يتم التعديل.
!>
!>  RESULT - REAL               array of dimension( NTESTS )
!>           عند الدخول، تحتوي RESULT على نسب الاختبارات
!>           التي تم إجراؤها في البرنامج المستدعي.
!>           لم يتم التعديل.
!>
!>  ISEED  - INTEGER            array of dimension( 4 )
!>           يحتوي على البذرة العشوائية التي ولّدت المصفوفة المستخدمة
!>           للاختبارات التي تكون نسبها في RESULT.
!>           لم يتم التعديل.
!>
!>  THRESH - REAL
!>           عند الدخول، يحدد THRESH الحد المقبول لـ
!>           نسب الاختبار. إذا كان RESULT( K ) > THRESH، فإن الاختبار K
!>           لم يجتاز الحد وسيتم طباعة رسالة.
!>           لم يتم التعديل.
!>
!>  IOUNIT - INTEGER
!>           عند الدخول، يحدد IOUNIT رقم الوحدة للملف
!>           الذي تتم طباعة الرسائل إليه.
!>           لم يتم التعديل.
!>
!>  IE     - INTEGER
!>           عند الدخول، يحتوي IE على عدد الاختبارات التي
!>           فشلت في اجتياز الحد حتى الآن.
!>           يتم التحديث عند الخروج إذا فشلت أي من النسب في RESULT أيضًا.
!>
