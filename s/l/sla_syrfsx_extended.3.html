subroutine sla_syrfsx_extended (integer prec_type, character uplo, integer n, integer nrhs, real, dimension( lda, * ) a, integer lda, real, dimension( ldaf, * ) af, integer ldaf, integer, dimension( * ) ipiv, logical colequ, real, dimension( * ) c, real, dimension( ldb, * ) b, integer ldb, real, dimension( ldy, * ) y, integer ldy, real, dimension( * ) berr_out, integer n_norms, real, dimension( nrhs, * ) err_bnds_norm, real, dimension( nrhs, * ) err_bnds_comp, real, dimension( * ) res, real, dimension( * ) ayb, real, dimension( * ) dy, real, dimension( * ) y_tail, real rcond, integer ithresh, real rthresh, real dz_ub, logical ignore_cwise, integer info)

SLA_SYRFSX_EXTENDED يقوم بتحسين الحل المحسوب لنظام المعادلات الخطية للمصفوفات المتناظرة غير المحددة عن طريق إجراء تنقيح تكراري دقيق إضافي ويوفر حدودًا للأخطاء وتقديرات للأخطاء الخلفية للحل.

الغرض:

!>
!>
!> SLA_SYRFSX_EXTENDED يقوم بتحسين الحل المحسوب لنظام المعادلات الخطية
!> عن طريق إجراء تنقيح تكراري دقيق إضافي
!> ويوفر حدودًا للأخطاء وتقديرات للأخطاء الخلفية للحل.
!> يتم استدعاء هذا الإجراء الفرعي بواسطة SSYRFSX لإجراء التنقيح التكراري.
!> بالإضافة إلى حد الخطأ المعياري، يوفر الرمز حدًا أقصى
!> للخطأ المكوني إذا كان ذلك ممكنًا. راجع التعليقات لـ ERR_BNDS_NORM
!> و ERR_BNDS_COMP لتفاصيل حدود الأخطاء. لاحظ أن هذا
!> الإجراء الفرعي مسؤول فقط عن تعيين الحقول الثانية لـ
!> ERR_BNDS_NORM و ERR_BNDS_COMP.
!>

ERR_BNDS_NORM

!>          ERR_BNDS_NORM هو مصفوفة REAL، البعد (NRHS, N_ERR_BNDS)

!>     حدود أخطاء مختلفة وأرقام شرطية مقابلة لـ
!>     الخطأ النسبي المعياري، والذي يتم تعريفه على النحو التالي:
!>
!>     الخطأ النسبي المعياري في متجه الحل i:
!>             max_j (abs(XTRUE(j,i) - X(j,i)))
!>            ------------------------------
!>                  max_j abs(X(j,i))
!>
!>     يتم فهرسة المصفوفة حسب نوع معلومات الخطأ كما هو موضح
!>     أدناه. يوجد حاليًا ما يصل إلى ثلاثة أجزاء من المعلومات
!>     يتم إرجاعها.
!>
!>     الفهرس الأول في ERR_BNDS_NORM(i,:) يقابل الطرف
!>     الأيمن i.
!>
!>     الفهرس الثاني في ERR_BNDS_NORM(:,err) يحتوي على الثلاثة
!>     حقول التالية:
!>     err = 1  منطقية. وثق الإجابة إذا كان
!>              المقلوب للرقم الشرطي أقل من الحد
!>              sqrt(n) * slamch('Epsilon').
!>
!>     err = 2  حد الخطأ: الخطأ الأمامي المقدر،
!>              موجود تقريبًا ضمن عامل 10 للخطأ الحقيقي
!>              طالما أن الإدخال التالي أكبر من الحد
!>              sqrt(n) * slamch('Epsilon'). يجب أن يكون حد الخطأ هذا موثوقًا به فقط
!>              إذا كانت القيمة المنطقية السابقة صحيحة.
!>
!>     err = 3  مقلوب الرقم الشرطي: تقديري معياري
!>

ERR_BNDS_COMP

العدد الشرطي المتبادل. مقارنة بالحد !> sqrt(n) * slamch('Epsilon') لتحديد ما إذا كان تقدير !> الخطأ هو . هذه الأعداد الشرطية المتبادلة !> هي 1 / (norm(Z^{-1},inf) * norm(Z,inf)) لبعض !> المصفوفات Z المقيّسة بشكل مناسب. !> لتكن Z = S*A، حيث يقوم S بقياس كل صف بقوة الأساس بحيث تكون !> جميع مجاميع الصفوف المطلقة لـ Z تقريبًا 1. !> !> هذه الوحدة الفرعية مسؤولة فقط عن تعيين الحقل الثاني !> أعلاه. !> انظر مذكرة عمل لاباك 165 لمزيد من التفاصيل والاحتياطات الإضافية. !>

ERR_BNDS_COMP

!>          ERR_BNDS_COMP هو مصفوفة حقيقية، ذات أبعاد (NRHS, N_ERR_BNDS)
!>     لكل جانب أيمن، تحتوي هذه المصفوفة على معلومات حول
!>     حدود الخطأ المختلفة والأعداد الشرطية المقابلة لـ
!>     الخطأ النسبي المكون، والذي يتم تعريفه على النحو التالي:
!>
!>     الخطأ النسبي المكون في متجه الحل i:
!>                    abs(XTRUE(j,i) - X(j,i))
!>             max_j ----------------------
!>                         abs(X(j,i))
!>
!>     يتم فهرسة المصفوفة حسب الجانب الأيمن i (الذي يعتمد عليه الخطأ
!>     النسبي المكون)، ونوع معلومات الخطأ كما هو موضح أدناه. يوجد حاليًا
!>     ما يصل إلى ثلاثة أجزاء من المعلومات التي تم إرجاعها لكل جانب أيمن. إذا
!>     لم يتم طلب الدقة المكونة (PARAMS(3) = 0.0)، فلن يتم الوصول إلى
!>     ERR_BNDS_COMP. إذا كانت N_ERR_BNDS < 3، فلن يتم إرجاع سوى أول
!>     (:,N_ERR_BNDS) إدخالات كحد أقصى.
!>
!>     الفهرس الأول في ERR_BNDS_COMP(i,:) يتوافق مع الجانب الأيمن i.
!>
!>     الفهرس الثاني في ERR_BNDS_COMP(:,err) يحتوي على المجالات الثلاثة
!>     التالية:
!>     err = 1 منطقي. ثق بالإجابة إذا كان
!>              العدد الشرطي المتبادل أقل من الحد
!>              sqrt(n) * slamch('Epsilon').
!>
!>     err = 2 حد الخطأ: الخطأ الأمامي المقدر،
!>              من المؤكد تقريبًا أنه ضمن عامل 10 من الخطأ الحقيقي
!>              طالما أن الإدخال التالي أكبر من الحد
!>              sqrt(n) * slamch('Epsilon'). يجب أن يكون حد الخطأ هذا موثوقًا به
!>              فقط إذا كان المنطقي السابق صحيحًا.
!>
!>     err = 3 العدد الشرطي المتبادل: العدد الشرطي المتبادل المقيّس المقدر.
!>              مقارنة بالحد
!>              sqrt(n) * slamch('Epsilon') لتحديد ما إذا كان تقدير
!>              الخطأ هو . هذه الأعداد الشرطية المتبادلة
!>              هي 1 / (norm(Z^{-1},inf) * norm(Z,inf)) لبعض
!>              المصفوفات Z المقيّسة بشكل مناسب.
!>              لتكن Z = S*(A*diag(x))، حيث x هو الحل
!>              للجانب الأيمن الحالي و S يقيس كل صف من
!>              A*diag(x) بقوة الأساس بحيث تكون جميع مجاميع الصفوف المطلقة لـ Z
!>              تقريبًا 1.
!>
!>     هذه الوحدة الفرعية مسؤولة فقط عن تعيين الحقل الثاني
!>     أعلاه.
!>     انظر مذكرة عمل لاباك 165 لمزيد من التفاصيل والاحتياطات الإضافية.
!>

AYB

!>

DY

!>          DY هي مصفوفة حقيقية، ذات أبعاد (N)

!>

Y_TAIL

!>          Y_TAIL هي مصفوفة حقيقية، ذات أبعاد (N)
!>     مساحة عمل للاحتفاظ بالأجزاء الذيلية للحل الوسيط.
!>

RCOND

!>          RCOND هو رقم حقيقي
!>     العدد الشرطي المقيّس المتبادل. هذا هو تقدير لـ
!>     العدد الشرطي لسكيل المتبادل للمصفوفة A بعد
!>     الموازنة (إذا تم إجراؤها). إذا كان أقل من دقة الآلة
!>     (على وجه الخصوص، إذا كان صفرًا)، فإن المصفوفة مفردة
!>     بدقة العمل. لاحظ أن الخطأ قد يظل صغيرًا حتى لو كان هذا الرقم صغيرًا جدًا
!>     وتبدو المصفوفة سيئة التكييف.
!>

ITHRESH

!>          ITHRESH هو عدد صحيح
!>     العدد الأقصى لحسابات الباقي المسموح بها
!>     للتحسين. القيمة الافتراضية هي 10. بالنسبة لـ 'aggressive' اضبطها على 100 للسماح
!>     بالتقارب باستخدام عوامل تقريبية أو
!>     عوامل غير LU. إذا استخدمت العامل
!>     تقنية أخرى غير الاستبعاد الغاوسي، فقد لا تكون الضمانات في
!>     ERR_BNDS_NORM و ERR_BNDS_COMP موثوقة بعد الآن.
!>

RTHRESH

!>          RTHRESH هو رقم حقيقي
!>     يحدد متى يجب إيقاف التحسين إذا توقف تقدير الخطأ عن
!>     التناقص. سيتوقف التحسين عندما لا يحقق الحل التالي
!>     norm(dx_{i+1}) < RTHRESH * norm(dx_i) حيث norm(Z) هو
!>     المعيار اللانهائي لـ Z. RTHRESH يحقق 0 < RTHRESH <= 1. القيمة
!>     الافتراضية هي 0.5. بالنسبة لـ 'aggressive' اضبطها على 0.9 للسماح
!>     بالتقارب على مصفوفات سيئة التكييف للغاية. انظر LAWN 165
!>     لمزيد من التفاصيل.
!>

DZ_UB

!>          DZ_UB هو رقم حقيقي
