subroutine slasyf_aa (character uplo, integer j1, integer m, integer nb, real, dimension( lda, * ) a, integer lda, integer, dimension( * ) ipiv, real, dimension( ldh, * ) h, integer ldh, real, dimension( * ) work)

SLASYF_AA

الغرض:

!>
!> DLATRF_AA تقوم بتحليل لوحة من مصفوفة متماثلة حقيقية A باستخدام
!> خوارزمية Aasen. تتكون اللوحة من مجموعة من الصفوف NB من A
!> عندما يكون UPLO هو U، أو مجموعة من الأعمدة NB عندما يكون UPLO هو L.
!>
!> من أجل تحليل اللوحة، تتطلب خوارزمية Aasen الصف الأخير، أو العمود،
!!> من اللوحة السابقة. يتم تعيين الصف الأول، أو العمود،
!> من A ليكون الصف الأول، أو العمود، من مصفوفة الوحدة،
!> والتي تستخدم لتحليل اللوحة الأولى.
!>
!> يتم تخزين الصف J الناتج من U، أو العمود J من L، في
!> الصف (J-1)، أو العمود، من A (بدون الأقطار الوحدية)، بينما
!> يتم الكتابة فوق القطر والقطر الفرعي لـ A بواسطة تلك الخاصة بـ T.
!>
!>

المعلمات

UPLO 
!>          UPLO هو CHARACTER*1
!>          = 'U':  يتم تخزين المثلث العلوي لـ A؛
!>          = 'L':  يتم تخزين المثلث السفلي لـ A.
!>

J1

!>          J1 هو INTEGER
!>          موقع الصف الأول، أو العمود، للوحة
!>          ضمن المصفوفة الفرعية لـ A، الممررة لهذه الروتين، على سبيل المثال،
!>          عند استدعائه بواسطة SSYTRF_AA، للوحة الأولى، J1 هو 1،
!>          بينما للوحات المتبقية، J1 هو 2.
!>

M

!>          M هو INTEGER
!>          بعد المصفوفة الفرعية. M >= 0.
!>

NB

!>          NB هو INTEGER
!>          بعد اللوحة المراد تحليلها.
!>

A

!>          A هو مصفوفة REAL، البعد (LDA,M) لـ
!>          اللوحة الأولى، بينما البعد (LDA,M+1) لـ
!>          اللوحات المتبقية.
!>
!>          عند الدخول، تحتوي A على الصف الأخير، أو العمود،
!>          من اللوحة السابقة، والمصفوفة الفرعية اللاحقة لـ A
!>          لتحليلها، باستثناء اللوحة الأولى، فقط
!>          يتم تمرير اللوحة.
!>
!>          عند الخروج، يتم تحليل اللوحة الرائدة.
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرائد للمصفوفة A.  LDA >= max(1,M).
!>

IPIV

!>          IPIV هو مصفوفة INTEGER، البعد (M)
!>          تفاصيل تبادلات الصفوف والأعمدة،
!>          تم تبديل الصف والعمود k بالصف والعمود IPIV(k).
!>

H

!>          H هي مساحة عمل REAL، البعد (LDH,NB).
!>
!>

LDH

!>          LDH هو INTEGER
!>          البعد الرائد لمساحة العمل H. LDH >= max(1,M).
!>

WORK

!>          WORK هي مساحة عمل REAL، البعد (M).
!>
