subroutine slarhs (character*3 path, character xtype, character uplo, character trans, integer m, integer n, integer kl, integer ku, integer nrhs, real, dimension( lda, * ) a, integer lda, real, dimension( ldx, * ) x, integer ldx, real, dimension( ldb, * ) b, integer ldb, integer, dimension( 4 ) iseed, integer info)

SLARHS

الغرض:

!>
!> SLARHS يختار مجموعة من NRHS ناقلات حل عشوائية ويجهز
!> الأطراف اليمنى للنظام الخطي
!>    op(A) * X = B,
!> حيث op(A) = A أو A**T، حسب TRANS.
!>

المعلمات

PATH 
!>          PATH هو CHARACTER*3
!>          نوع المصفوفة الحقيقية A. يمكن تقديم PATH بأي
!>          تركيبة من الأحرف الكبيرة والصغيرة. الأنواع الصالحة تشمل
!>             xGE:  مصفوفة عامة m x n
!>             xGB:  مصفوفة نطاقية عامة
!>             xPO:  محددة موجبة متماثلة، تخزين ثنائي الأبعاد
!>             xPP:  محددة موجبة متماثلة مدمجة
!>             xPB:  محددة موجبة متماثلة نطاقية
!>             xSY:  غير محددة متماثلة، تخزين ثنائي الأبعاد
!>             xSP:  غير محددة متماثلة مدمجة
!>             xSB:  غير محددة متماثلة نطاقية
!>             xTR:  مثلثة
!>             xTP:  مثلثة مدمجة
!>             xTB:  مثلثة نطاقية
!>             xQR:  مصفوفة عامة m x n
!>             xLQ:  مصفوفة عامة m x n
!>             xQL:  مصفوفة عامة m x n
!>             xRQ:  مصفوفة عامة m x n
!>          حيث يشير الحرف الأول إلى الدقة.
!>

XTYPE

!>          XTYPE هو CHARACTER*1
!>          يحدد كيفية تحديد الحل الدقيق X:
!>          = 'N':  حل جديد؛ إنشاء X عشوائي.
!>          = 'C':  محسوب؛ استخدام قيمة X عند الإدخال.
!>

UPLO

!>          UPLO هو CHARACTER*1
!>          يحدد ما إذا كان الجزء العلوي أو السفلي من المصفوفة A
!>          مخزنًا، إذا كانت A متماثلة.
!>          = 'U':  مثلثي علوي
!>          = 'L':  مثلثي سفلي
!>

TRANS

!>          TRANS هو CHARACTER*1
!>          تستخدم فقط إذا كانت A غير متماثلة؛ تحدد العملية
!>          المطبقة على المصفوفة A.
!>          = 'N':  B := A    * X  (بدون تبديل)
!>          = 'T':  B := A**T * X  (تبديل)
!>          = 'C':  B := A**H * X  (تبديل مترافق = تبديل)
!>

M

!>          M هو INTEGER
!>          عدد الصفوف في المصفوفة A.  M >= 0.
!>

N

!>          N هو INTEGER
!>          عدد الأعمدة في المصفوفة A.  N >= 0.
!>

KL

!>          KL هو INTEGER
!>          يستخدم فقط إذا كانت A مصفوفة نطاقية؛ يحدد عدد
!>          الأقطار الفرعية للمصفوفة A إذا كانت A مصفوفة نطاقية عامة أو إذا كانت A
!>          متماثلة أو مثلثة و UPLO = 'L'؛ يحدد عدد
!>          الأقطار الفرعية العلوية للمصفوفة A إذا كانت A متماثلة أو مثلثة و
!>          UPLO = 'U'.  0 <= KL <= M-1.
!>

KU

!>          KU هو INTEGER
!>          يستخدم فقط إذا كانت A مصفوفة نطاقية عامة أو إذا كانت A
!>          مثلثة.
!>
!>          إذا كان PATH = xGB، يحدد عدد الأقطار الفرعية العلوية لـ A،
!>          و 0 <= KU <= N-1.
!>
!>          إذا كان PATH = xTR، xTP، أو xTB، يحدد ما إذا كانت
!>          المصفوفة ذات قطر وحيد أم لا:
!>          = 1:  المصفوفة ذات قطر غير وحيد (الافتراضي)
!>          = 2:  المصفوفة ذات قطر وحيد
!>

NRHS

!>          NRHS هو INTEGER
!>          عدد متجهات الطرف الأيمن في النظام A*X = B.
!>

A

!>          A هو مصفوفة REAL، البعد (LDA,N)
!>          مصفوفة الاختبار التي يحدد نوعها PATH.
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرئيسي للمصفوفة A.
!>          إذا كان PATH = xGB، LDA >= KL+KU+1.
!>          إذا كان PATH = xPB، xSB، xHB، أو xTB، LDA >= KL+1.
!>          بخلاف ذلك، LDA >= max(1,M).
!>

X

!>          X هي مصفوفة REAL (إدخال أو إخراج)، البعد(LDX,NRHS)
!>          عند الإدخال، إذا كان XTYPE = 'C' (لـ 'Computed')، فإن X تحتوي على
!>          الحل الدقيق لنظام المعادلات الخطية.
!>          عند الإخراج، إذا كان XTYPE = 'N' (لـ 'New')، يتم تهيئة X
!>          بقِيم عشوائية.
!>

LDX

!>          LDX هو INTEGER
!>          البعد الرئيسي للمصفوفة X. إذا كان TRANS = 'N'،
!>          LDX >= max(1,N)؛ إذا كان TRANS = 'T'، LDX >= max(1,M).
!>

B

!>          B هي مصفوفة REAL، البعد (LDB,NRHS)
!>          متجه (متجهات) الطرف الأيمن لنظام المعادلات،
!>          محسوبة من B = op(A) * X، حيث يتم تحديد op(A)
!>          بواسطة TRANS.
!>

LDB

!>          LDB هو INTEGER
!>          البعد الرئيسي للمصفوفة B. إذا كان TRANS = 'N'،
!>          LDB >= max(1,M)؛ إذا كان TRANS = 'T'، LDB >= max(1,N).
!>

ISEED

!>          ISEED هي مصفوفة INTEGER، البعد (4)
!>          متجه البذور لمولد الأرقام العشوائية (المستخدم في
!>          SLATMS). يتم تعديله عند الإخراج.
!>

INFO

!>          INFO هو INTEGER
!>          = 0: خروج ناجح
!>          < 0: إذا كان INFO = -i، فإن المعلمة i-th كان لها قيمة غير صالحة
!>
