subroutine slahrd (n, k, nb, a, lda, tau, t, ldt, y, ldy)
SLAHRD تقوم بتقليص الأعمدة NB الأولى لمصفوفة عامة مستطيلة A بحيث تصبح العناصر الموجودة تحت القطر الفرعي k صفرًا، وتعيد مصفوفات مساعدة مطلوبة لتطبيق التحويل على الجزء غير المختزل من A.

توثيق الوظيفة/الإجراء الفرعي

subroutine slahrd (integer n, integer k, integer nb, real, dimension( lda, * ) a, integer lda, real, dimension( nb ) tau, real, dimension( ldt, nb ) t, integer ldt, real, dimension( ldy, nb ) y, integer ldy)

SLAHRD تقوم بتقليص الأعمدة NB الأولى لمصفوفة عامة مستطيلة A بحيث تصبح العناصر الموجودة تحت القطر الفرعي k صفرًا، وتعيد مصفوفات مساعدة مطلوبة لتطبيق التحويل على الجزء غير المختزل من A.

الغرض:

!>
!> هذا الإجراء الفرعي مهمل وتم استبداله بالإجراء SLAHR2.
!>
!> SLAHRD تقوم بتقليص الأعمدة NB الأولى لمصفوفة عامة n×(n-k+1)
!> A بحيث تصبح العناصر الموجودة تحت القطر الفرعي k صفرًا. يتم
!> إجراء التقليص بواسطة تحويل تشابه متعامد
!> Q**T * A * Q. تعيد الوحدة المصفوفتين V و T التي تحددان
!> Q كمُعكِس كتلي I - V*T*V**T، وكذلك المصفوفة Y = A * V * T.
!>

المعلمات

!>          رتبة المصفوفة A.

!>          K هو عدد صحيح INTEGER
!>          الإزاحة للتقليص. يتم تقليص العناصر الموجودة تحت القطر الفرعي
!>          k في الأعمدة NB الأولى إلى صفر.
!>

!>          NB هو عدد صحيح INTEGER
!>          عدد الأعمدة التي سيتم تقليصها.
!>

!>          A هي مصفوفة حقيقية REAL، حجمها (LDA,N-K+1)
!>          عند الإدخال، المصفوفة العامة A بحجم n×(n-k+1).
!>          عند الخروج، يتم الكتابة فوق العناصر الموجودة على القطر الفرعي k وما فوقه في
!>          الأعمدة NB الأولى بالعناصر المقابلة للمصفوفة المختزلة؛ العناصر الموجودة تحت القطر الفرعي k،
!>          مع المصفوفة TAU، تمثل المصفوفة Q كناتج للمُعكِسات الأولية. الأعمدة الأخرى للمصفوفة A
!>          تبقى دون تغيير. انظر التفاصيل الإضافية.
!>

!>

T

!>          T هي مصفوفة حقيقية REAL، حجمها (LDT,NB)
!>          المصفوفة المثلثية العليا T.
!>

LDT

!>          LDT هو عدد صحيح INTEGER
!>          البعد القيادي للمصفوفة T.  LDT >= NB.

Y

!>

!>          البعد القيادي للمصفوفة Y. LDY >= N.

تفاصيل إضافية:

!>
!>  تُمثل المصفوفة Q كناتج لـ nb من المُعكِسات الأولية
!>
!>     Q = H(1) H(2) . . . H(nb).
!>
!>  كل H(i) له الشكل
!>
!>     H(i) = I - tau * v * v**T
!>
!>  حيث tau هو عامل قياسي حقيقي، و v هو متجه حقيقي له
!>  v(1:i+k-1) = 0، v(i+k) = 1؛ يتم تخزين v(i+k+1:n) عند الخروج في
!>  A(i+k+1:n,i)، و tau في TAU(i).
!>
!>  عناصر المتجهات v تشكل معًا المصفوفة V بحجم (n-k+1)×nb
!>  المطلوبة، مع T و Y، لتطبيق التحويل على
!>  الجزء غير المختزل من المصفوفة، باستخدام تحديث من الشكل:
!>  A := (I - V*T*V**T) * (A - Y*V**T).
!>
!>  محتويات A عند الخروج موضح بال مثال التالي
!>  عندما n = 7، k = 3 و nb = 2:
!>
!>     ( a   h   a   a   a )
!>     ( a   h   a   a   a )
!>     ( a   h   a   a   a )
!>     ( h   h   a   a   a )
!>     ( v1  h   a   a   a )
!>     ( v1  v2  a   a   a )
!>     ( v1  v2  a   a   a )
!>
!>  حيث a يمثل عنصرًا من المصفوفة الأصلية A، h يمثل عنصرًا معدلاً
!>  من المصفوفة الهيسنبرجية العليا H، و vi يمثل عنصرًا
!>  من المتجه الذي يحدد H(i).
!>
