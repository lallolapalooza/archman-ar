subroutine slarzb (side, trans, direct, storev, m, n, k, l, v, ldv, t, ldt, c, ldc, work, ldwork)
SLARZB يطبق عاكس كتلة أو منقوله على مصفوفة عامة.

توثيق الوظيفة/الإجراء الفرعي

subroutine slarzb (character side, character trans, character direct, character storev, integer m, integer n, integer k, integer l, real, dimension( ldv, * ) v, integer ldv, real, dimension( ldt, * ) t, integer ldt, real, dimension( ldc, * ) c, integer ldc, real, dimension( ldwork, * ) work, integer ldwork)

SLARZB يطبق عاكس كتلة حقيقي H أو منقوله H**T على مصفوفة عامة موزعة حقيقية C من اليسار أو اليمين.

الهدف:

!>
!> SLARZB يطبق عاكس كتلة حقيقي H أو منقوله H**T على
!> مصفوفة عامة موزعة حقيقية M-by-N  C من اليسار أو اليمين.
!>
!> حاليًا، يتم دعم STOREV = 'R' و DIRECT = 'B' فقط.
!>

المعاملات

SIDE 
!>          SIDE is CHARACTER*1
!>          = 'L': apply H or H**T from the Left (طبق H أو H**T من اليسار)
!>          = 'R': apply H or H**T from the Right (طبق H أو H**T من اليمين)
!>

TRANS

!>          TRANS is CHARACTER*1
!>          = 'N': apply H (No transpose) (طبق H (بدون تدوير))
!>          = 'C': apply H**T (Transpose) (طبق H**T (تدوير))
!>

DIRECT

!>          DIRECT is CHARACTER*1
!>          Indicates how H is formed from a product of elementary (يشير إلى كيفية تكوين H من حاصل ضرب)
!>          reflectors (عاكسات أولية)
!>          = 'F': H = H(1) H(2) . . . H(k) (Forward, not supported yet) (أمامي، غير مدعوم بعد)
!>          = 'B': H = H(k) . . . H(2) H(1) (Backward) (خلفي)
!>

STOREV

!>          STOREV is CHARACTER*1
!>          Indicates how the vectors which define the elementary (يشير إلى كيفية تخزين المتجهات التي تحدد العاكسات الأولية)
!>          reflectors are stored: (مخزنة:)
!>          = 'C': Columnwise                        (not supported yet) (عموديًا (غير مدعوم بعد))
!>          = 'R': Rowwise (سطريًا)
!>

M

!>          M is INTEGER
!>          The number of rows of the matrix C. (عدد صفوف المصفوفة C.)
!>

N

!>          N is INTEGER
!>          The number of columns of the matrix C. (عدد أعمدة المصفوفة C.)
!>

K

!>          K is INTEGER
!>          The order of the matrix T (= the number of elementary (ترتيب المصفوفة T (= عدد العاكسات الأولية)
!>          reflectors whose product defines the block reflector). (التي يحدد حاصل ضربها عاكس الكتلة).)
!>

L

!>          L is INTEGER
!>          The number of columns of the matrix V containing the (عدد أعمدة المصفوفة V التي تحتوي على)
!>          meaningful part of the Householder reflectors. (الجزء الهادف من عاكسات هاوسهولدر.)
!>          If SIDE = 'L', M >= L >= 0, if SIDE = 'R', N >= L >= 0.
!>

V

!>          V is REAL array, dimension (LDV,NV).
!>          If STOREV = 'C', NV = K; if STOREV = 'R', NV = L.
!>

LDV

!>          LDV is INTEGER
!>          The leading dimension of the array V. (البعد الرائد للمصفوفة V.)
!>          If STOREV = 'C', LDV >= L; if STOREV = 'R', LDV >= K.
!>

T

!>          T is REAL array, dimension (LDT,K)
!>          The triangular K-by-K matrix T in the representation of the (المصفوفة المثلثية K-by-K T في تمثيل)
!>          block reflector. (عاكس الكتلة.)
!>

LDT

!>          LDT is INTEGER
!>          The leading dimension of the array T. LDT >= K. (البعد الرائد للمصفوفة T. LDT >= K.)
!>

C

!>          C is REAL array, dimension (LDC,N)
!>          On entry, the M-by-N matrix C. (عند الدخول، المصفوفة C ذات الأبعاد M-by-N.)
!>          On exit, C is overwritten by H*C or H**T*C or C*H or C*H**T. (عند الخروج، يتم الكتابة فوق C بـ H*C أو H**T*C أو C*H أو C*H**T.)
!>

LDC

!>          LDC is INTEGER
!>          The leading dimension of the array C. LDC >= max(1,M). (البعد الرائد للمصفوفة C. LDC >= max(1,M).)
!>

WORK

!>          WORK is REAL array, dimension (LDWORK,K)
!>

LDWORK

!>          LDWORK is INTEGER
!>          The leading dimension of the array WORK. (البعد الرائد للمصفوفة WORK.)
!>          If SIDE = 'L', LDWORK >= max(1,N);
!>          if SIDE = 'R', LDWORK >= max(1,M).
!>
