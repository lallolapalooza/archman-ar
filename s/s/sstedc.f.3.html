subroutine sstedc (character compz, integer n, real, dimension( * ) d, real, dimension( * ) e, real, dimension( ldz, * ) z, integer ldz, real, dimension( * ) work, integer lwork, integer, dimension( * ) iwork, integer liwork, integer info)

SSTEDC

الغرض:

!>
!> SSTEDC تحسب جميع القيم الذاتية، واختيارياً، المتجهات الذاتية لمصفوفة متماثلة ثلاثية الأقطار باستخدام طريقة الانقسام والغزو.
!> يمكن أيضًا إيجاد المتجهات الذاتية لمصفوفة متماثلة كاملة أو نطاقية إذا تم استخدام SSYTRD أو SSPTRD أو SSBTRD لتقليل هذه
!> المصفوفة إلى شكل ثلاثي الأقطار.
!>
!>

المعلمات

COMPZ 
!>          COMPZ is CHARACTER*1
!>          = 'N':  حساب القيم الذاتية فقط.
!>          = 'I':  حساب المتجهات الذاتية للمصفوفة ثلاثية الأقطار أيضًا.
!>          = 'V':  حساب المتجهات الذاتية للمصفوفة المتماثلة الكثيفة الأصلية
!>                  أيضًا. عند الإدخال، تحتوي Z على المصفوفة المتعامدة
!>                  المستخدمة لتقليل المصفوفة الأصلية إلى
!>                  شكل ثلاثي الأقطار.
!>

N

!>          N is INTEGER
!>          بعد المصفوفة المتماثلة ثلاثية الأقطار. N >= 0.
!>

D

!>          D is REAL array, dimension (N)
!>          عند الإدخال، العناصر القطرية للمصفوفة ثلاثية الأقطار.
!>          عند الخروج، إذا كان INFO = 0، فإن القيم الذاتية بترتيب تصاعدي.
!>

E

!>          E is REAL array, dimension (N-1)
!>          عند الإدخال، العناصر تحت القطرية للمصفوفة ثلاثية الأقطار.
!>          عند الخروج، تم تدمير E.
!>

Z

!>          Z is REAL array, dimension (LDZ,N)
!>          عند الإدخال، إذا كان COMPZ = 'V'، فإن Z تحتوي على المتعامدة
!>          المصفوفة المستخدمة في التقليل إلى شكل ثلاثي الأقطار.
!>          عند الخروج، إذا كان INFO = 0، فإن إذا كان COMPZ = 'V'، فإن Z تحتوي على
!>          المتجهات الذاتية المتعامدة للمصفوفة المتماثلة الأصلية،
!>          وإذا كان COMPZ = 'I'، فإن Z تحتوي على المتجهات الذاتية المتعامدة
!>          للمصفوفة المتماثلة ثلاثية الأقطار.
!>          إذا كان COMPZ = 'N'، فإن Z لا تتم الإشارة إليها.
!>

LDZ

!>          LDZ is INTEGER
!>          البعد الرئيسي للمصفوفة Z. LDZ >= 1.
!>          إذا كانت المتجهات الذاتية مطلوبة، فإن LDZ >= max(1,N).
!>

WORK

!>          WORK is REAL array, dimension (MAX(1,LWORK))
!>          عند الخروج، إذا كان INFO = 0، فإن WORK(1) تعيد LWORK الأمثل.
!>

LWORK

!>          LWORK is INTEGER
!>          بعد المصفوفة WORK.
!>          إذا كان COMPZ = 'N' أو N <= 1، فيجب أن يكون LWORK على الأقل 1.
!>          إذا كان COMPZ = 'V' و N > 1، فيجب أن يكون LWORK على الأقل
!>                         ( 1 + 3*N + 2*N*lg N + 4*N**2 )،
!>                         حيث lg( N ) = أصغر عدد صحيح k بحيث
!>                         2**k >= N.
!>          إذا كان COMPZ = 'I' و N > 1، فيجب أن يكون LWORK على الأقل
!>                         ( 1 + 4*N + N**2 ).
!>          لاحظ أنه بالنسبة لـ COMPZ = 'I' أو 'V'، إذا كان N أقل من أو
!>          يساوي الحد الأدنى لحجم التقسيم، عادة 25، فإن LWORK يحتاج فقط
!>          أن يكون max(1,2*(N-1)).
!>
!>          إذا كان LWORK = -1، فسيتم افتراض استعلام مساحة العمل؛ الروتين
!>          يحسب فقط الحجم الأمثل للمصفوفة WORK، ويعيد
!>          هذه القيمة كإدخال أول للمصفوفة WORK، ولا يصدر أي رسالة خطأ
!>          متعلقة بـ LWORK بواسطة XERBLA.
!>

IWORK

!>          IWORK is INTEGER array, dimension (MAX(1,LIWORK))
!>          عند الخروج، إذا كان INFO = 0، فإن IWORK(1) تعيد LIWORK الأمثل.
!>

LIWORK

!>          LIWORK is INTEGER
!>          بعد المصفوفة IWORK.
!>          إذا كان COMPZ = 'N' أو N <= 1، فيجب أن يكون LIWORK على الأقل 1.
!>          إذا كان COMPZ = 'V' و N > 1، فيجب أن يكون LIWORK على الأقل
!>                         ( 6 + 6*N + 5*N*lg N ).
!>          إذا كان COMPZ = 'I' و N > 1، فيجب أن يكون LIWORK على الأقل
!>                         ( 3 + 5*N ).
!>          لاحظ أنه بالنسبة لـ COMPZ = 'I' أو 'V'، إذا كان N أقل من أو
!>          يساوي الحد الأدنى لحجم التقسيم، عادة 25، فإن LIWORK
!>          يحتاج فقط أن يكون 1.
!>
!>          إذا كان LIWORK = -1، فسيتم افتراض استعلام مساحة العمل؛ الروتين
!>          يحسب فقط الحجم الأمثل للمصفوفة IWORK،
!>          ويعيد هذه القيمة كإدخال أول للمصفوفة IWORK،
!>          ولا يصدر أي رسالة خطأ متعلقة بـ LIWORK بواسطة XERBLA.
!>

INFO

!>          INFO is INTEGER
!>          = 0:  خروج ناجح.
!>          < 0:  إذا كان INFO = -i، فإن المعلمة i كانت لها قيمة غير قانونية.
!>          > 0:  فشل الخوارزمية في حساب قيمة ذاتية أثناء العمل
!>                على المصفوفة الفرعية الموجودة في الصفوف والأعمدة
!>                INFO/(N+1) حتى mod(INFO,N+1).
!>
