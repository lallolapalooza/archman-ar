subroutine ssyevr_2stage (character jobz, character range, character uplo, integer n, real, dimension( lda, * ) a, integer lda, real vl, real vu, integer il, integer iu, real abstol, integer m, real, dimension( * ) w, real, dimension( ldz, * ) z, integer ldz, integer, dimension( * ) isuppz, real, dimension( * ) work, integer lwork, integer, dimension( * ) iwork, integer liwork, integer info)

SSYEVR_2STAGE تحسب القيم الذاتية، واختياريًا، المتجهات الذاتية لمصفوفات SY

الغرض:

!>
!> SSYEVR_2STAGE تحسب القيم الذاتية المختارة، واختياريًا، المتجهات الذاتية
!> لمصفوفة متماثلة حقيقية A باستخدام تقنية المرحلتين (2stage)
!> للتحويل إلى مصفوفة ثلاثية الأقطار. يمكن اختيار القيم الذاتية والمتجهات الذاتية
!> عن طريق تحديد نطاق قيم أو نطاق
!> مؤشرات للقيم الذاتية المطلوبة.
!>
!> تقوم SSYEVR_2STAGE أولاً بتحويل المصفوفة A إلى صيغة ثلاثية الأقطار T باستدعاء
!> SSYTRD. بعد ذلك، عندما يكون ذلك ممكنًا، تستدعي SSYEVR_2STAGE SSTEMR لحساب
!> الطيف الذاتي باستخدام التمثيلات الثابتة نسبيًا (Relatively Robust Representations).
!> تحسب SSTEMR القيم الذاتية بواسطة خوارزمية dqds، بينما يتم حساب المتجهات الذاتية المتعامدة
!> من تمثيلات L D L^T مختلفة (تُعرف أيضًا بالتمثيلات الثابتة نسبيًا).
!> يتم تجنب التعامد بواسطة الجرام-شميدت قدر الإمكان. وبشكل أكثر تحديدًا،
!> فإن الخطوات المختلفة للخوارزمية هي كما يلي.
!>
!> لكل كتلة غير مختزلة (مصفوفة فرعية) من T،
!>    (أ) حساب T - sigma I = L D L^T، بحيث تحدد L و D
!>        جميع القيم الذاتية المطلوبة بدقة نسبية عالية.
!>        هذا يعني أن التغييرات النسبية الصغيرة في مدخلات D و L
!>        تسبب تغييرات نسبية صغيرة فقط في القيم الذاتية والمتجهات الذاتية.
!>        الصيغة القياسية (غير المفككة) للمصفوفة ثلاثية الأقطار T لا تتمتع بهذه الخاصية بشكل عام.
!>    (ب) حساب القيم الذاتية بدقة مناسبة.
!>        إذا كانت المتجهات الذاتية مطلوبة، فإن الخوارزمية تحقق دقة كاملة
!>        للقيم الذاتية المحسوبة فقط قبل أن يتم حساب المتجهات المقابلة، انظر الخطوات ج) ود).
!>    (ج) لكل مجموعة من القيم الذاتية القريبة، اختر قيمة تحول جديدة
!>        قريبة من المجموعة، وابحث عن مفكك جديد، وصقل
!>        القيم الذاتية المتحولة إلى دقة مناسبة.
!>    (د) لكل قيمة ذاتية ذات فصل نسبي كبير بما فيه الكفاية، احسب
!>        المتجه الذاتي المقابل عن طريق تكوين مفكك ملتوي يكشف الرتبة (rank revealing twisted factorization).
!>        ارجع إلى (ج) لأي مجموعات متبقية.
!>
!> يمكن تحديد الدقة المطلوبة للمخرجات بواسطة المعلمة
!> المدخلة ABSTOL.
!>
!> لمزيد من التفاصيل، راجع توثيق SSTEMR و:
!> - Inderjit S. Dhillon and Beresford N. Parlett: 
!>   Linear Algebra and its Applications, 387(1), pp. 1-28, August 2004.
!> - Inderjit Dhillon and Beresford Parlett:  SIAM Journal on Matrix Analysis and Applications, Vol. 25,
!>   2004.  Also LAPACK Working Note 154.
!> - Inderjit Dhillon: ,
!>   Computer Science Division Technical Report No. UCB/CSD-97-971,
!>   UC Berkeley, May 1997.
!>
!>
!> ملاحظة 1: تستدعي SSYEVR_2STAGE SSTEMR عندما يتم طلب الطيف الكامل
!> على الأجهزة التي تتوافق مع معيار النقطة العائمة ieee-754.
!> تستدعي SSYEVR_2STAGE SSTEBZ و SSTEIN على الأجهزة غير ieee وعند
!> إجراء طلبات طيف جزئي.
!>
!> قد يؤدي التنفيذ العادي لـ SSTEMR إلى إنشاء قيم NaN (ليست رقمًا) ولا نهائية وقد
!> يتسبب في تعطل بسبب استثناء نقطة عائمة في البيئات
!> التي لا تتعامل مع قيم NaN واللانهائية بالوضع الافتراضي لمعيار ieee.
!>

المعلمات

JOBZ 
!>          JOBZ is CHARACTER*1
!>          = 'N':  حساب القيم الذاتية فقط؛
!>          = 'V':  حساب القيم الذاتية والمتجهات الذاتية.
!>                  غير متوفر في هذا الإصدار.
!>

RANGE

!>          RANGE is CHARACTER*1
!>          = 'A': سيتم العثور على جميع القيم الذاتية.
!>          = 'V': سيتم العثور على جميع القيم الذاتية في النطاق (VL,VU]
!>                 (الفترة نصف المفتوحة).
!>          = 'I': سيتم العثور على القيم الذاتية من المؤشر IL إلى IU.
!>          بالنسبة لـ RANGE = 'V' أو 'I' و IU - IL < N - 1، يتم استدعاء SSTEBZ و
!>          SSTEIN
!>

UPLO

!>          UPLO is CHARACTER*1
!>          = 'U':  يتم تخزين الجزء العلوي من A؛
!>          = 'L':  يتم تخزين الجزء السفلي من A.
!>

N

!>          N is INTEGER
!>          رتبة المصفوفة A. N >= 0.
!>

A

!>          A is REAL array, dimension (LDA, N)
!>          عند الدخول، تحتوي المصفوفة المتماثلة A. إذا كان UPLO = 'U'، فإن
!>          الجزء العلوي الأيسر بحجم N*N من A يحتوي على
!>          الجزء العلوي من المصفوفة A. إذا كان UPLO = 'L'، فإن
!>          الجزء السفلي الأيسر بحجم N*N من A يحتوي على
!>          الجزء السفلي من المصفوفة A.
!>          عند الخروج، يتم تدمير الجزء السفلي (إذا كان UPLO='L') أو الجزء
!>          العلوي (إذا كان UPLO='U') من A، بما في ذلك القطر.
!>

LDA

!>          LDA is INTEGER
!>          البعد الرئيسي للمصفوفة A. LDA >= max(1,N).
!>

VL

!>          VL is REAL
!>          إذا كان RANGE='V'، فإن الحد الأدنى للنطاق للبحث عن
!>          القيم الذاتية. VL < VU.
!>          لا يتم الرجوع إليه إذا كان RANGE = 'A' أو 'I'.
!>

VU

!>          VU is REAL
!>          إذا كان RANGE='V'، فإن الحد الأعلى للنطاق للبحث عن
!>          القيم الذاتية. VL < VU.
!>          لا يتم الرجوع إليه إذا كان RANGE = 'A' أو 'I'.
!>

IL

!>          IL is INTEGER
!>          إذا كان RANGE='I'، فإن مؤشر
!>          أصغر قيمة ذاتية سيتم إرجاعها.
!>          1 <= IL <= IU <= N، إذا كان N > 0؛ IL = 1 و IU = 0 إذا كان N = 0.
!>          لا يتم الرجوع إليه إذا كان RANGE = 'A' أو 'V'.
!>

IU

!>          IU is INTEGER
!>          إذا كان RANGE='I'، فإن مؤشر
!>          أكبر قيمة ذاتية سيتم إرجاعها.
!>          1 <= IL <= IU <= N، إذا كان N > 0؛ IL = 1 و IU = 0 إذا كان N = 0.
!>          لا يتم الرجوع إليه إذا كان RANGE = 'A' أو 'V'.
!>

ABSTOL

!>          ABSTOL is REAL
!>          التفاوت المطلق في الخطأ للقيم الذاتية.
!>          يتم قبول قيمة ذاتية تقريبية على أنها متقاربة
!>          عندما يتم تحديد أنها تقع في فترة [a,b]
!>          بعرض أقل من أو يساوي
!>
!>                  ABSTOL + EPS *   max( |a|,|b| ) ،
!>
!>          حيث EPS هي دقة الآلة. إذا كانت ABSTOL أقل من
!>          أو تساوي صفرًا، فسيتم استخدام EPS*|T| بدلاً منها،
!>          حيث |T| هو معيار 1 للمصفوفة ثلاثية الأقطار التي تم الحصول عليها
!>          عن طريق تحويل A إلى صيغة ثلاثية الأقطار.
!>
!>          راجع by Demmel and
!>          Kahan, LAPACK Working Note #3.
!>
!>          إذا كانت الدقة النسبية العالية مهمة، اضبط ABSTOL على
!>          SLAMCH( 'Safe minimum' ). سيضمن ذلك أن
!>          القيم الذاتية يتم حسابها بدقة نسبية عالية عندما
!>          يكون ذلك ممكنًا في الإصدارات المستقبلية. الكود الحالي لا
!>          يقدم أي ضمانات بشأن الدقة النسبية العالية، لكن
!>          الإصدارات المستقبلية ستفعل. راجع J. Barlow and J. Demmel,
!>          , LAPACK Working Note #7، لمناقشة
!>          أي المصفوفات تحدد قيمها الذاتية بدقة نسبية عالية.
!>
