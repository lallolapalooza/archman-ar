subroutine ssterf (integer n, real, dimension( * ) d, real, dimension( * ) e, integer info)

SSTERF

الغرض:

!>
!> SSTERF يحسب جميع القيم الذاتية لمصفوفة قطرية ثلاثية متناظرة
!> باستخدام متغير Pal-Walker-Kahan من خوارزمية QL أو QR.
!>

المعلمات

N 
!>          N هو عدد صحيح (INTEGER)
!>          ترتيب المصفوفة.  N >= 0.
!>

D

!>          D هو مصفوفة حقيقية (REAL array)، البعد (N)
!>          عند الإدخال، العناصر القطرية n للمصفوفة القطرية الثلاثية.
!>          عند الخروج، إذا كان INFO = 0، القيم الذاتية بترتيب تصاعدي.
!>

E

!>          E هو مصفوفة حقيقية (REAL array)، البعد (N-1)
!>          عند الإدخال، العناصر القطرية الفرعية n-1 للمصفوفة القطرية
!>          الثلاثية.
!>          عند الخروج، تم تدمير E.
!>

INFO

!>          INFO هو عدد صحيح (INTEGER)
!>          = 0:  خروج ناجح
!>          < 0:  إذا كان INFO = -i، كان المعامل i ذو قيمة غير قانونية
!>          > 0:  فشلت الخوارزمية في إيجاد جميع القيم الذاتية في
!>                ما مجموعه 30*N تكرار؛ إذا كان INFO = i، فإن i
!>                من عناصر E لم تتقارب إلى الصفر.
!>
