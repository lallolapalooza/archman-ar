subroutine sspgvx (integer itype, character jobz, character range, character uplo, integer n, real, dimension( * ) ap, real, dimension( * ) bp, real vl, real vu, integer il, integer iu, real abstol, integer m, real, dimension( * ) w, real, dimension( ldz, * ) z, integer ldz, real, dimension( * ) work, integer, dimension( * ) iwork, integer, dimension( * ) ifail, integer info)

SSPGVX

الغرض:

!>
!> SSPGVX يحسب قيم eigenvalue المختارة، واختياريًا، المتجهات المميزة
!> لمشكلة eigenvalue متمايزة عامة حقيقية، من الشكل
!> A*x=(lambda)*B*x,  A*Bx=(lambda)*x,  أو B*A*x=(lambda)*x. هنا A
!> و B مفترضان أن يكونا متماثلين، مخزنين بتخزين مضغوط، و B
!> موجب محدد أيضًا. يمكن تحديد قيم eigenvalue والمتجهات المميزة
!> عن طريق تحديد نطاق قيم أو نطاق مؤشرات
!> لقيم eigenvalue المطلوبة.
!>

المعلمات

ITYPE 
!>          ITYPE هو عدد صحيح (INTEGER)
!>          يحدد نوع المشكلة المراد حلها:
!>          = 1:  A*x = (lambda)*B*x
!>          = 2:  A*B*x = (lambda)*x
!>          = 3:  B*A*x = (lambda)*x
!>

JOBZ

!>          JOBZ هو حرف (CHARACTER*1)
!>          = 'N':  حساب قيم eigenvalue فقط؛
!>          = 'V':  حساب قيم eigenvalue والمتجهات المميزة.
!>

RANGE

!>          RANGE هو حرف (CHARACTER*1)
!>          = 'A': سيتم العثور على جميع قيم eigenvalue.
!>          = 'V': سيتم العثور على جميع قيم eigenvalue في النطاق النصفي المفتوح (VL,VU]
!>                 .
!>          = 'I': سيتم العثور على قيم eigenvalue من IL إلى IU.
!>

UPLO

!>          UPLO هو حرف (CHARACTER*1)
!>          = 'U':  المثلث العلوي لـ A و B مخزن؛
!>          = 'L':  المثلث السفلي لـ A و B مخزن.
!>

N

!>          N هو عدد صحيح (INTEGER)
!>          رتبة زوج المصفوفات (A,B).  N >= 0.
!>

AP

!>          AP هو مصفوفة حقيقية (REAL array)، البعد (N*(N+1)/2)
!>          عند الدخول، المثلث العلوي أو السفلي للمصفوفة المتماثلة
!>          A، مضغوط عموديًا في مصفوفة خطية. العمود j من A
!>          مخزن في المصفوفة AP كالتالي:
!>          إذا كان UPLO = 'U'، فإن AP(i + (j-1)*j/2) = A(i,j) لـ 1<=i<=j؛
!>          إذا كان UPLO = 'L'، فإن AP(i + (j-1)*(2*n-j)/2) = A(i,j) لـ j<=i<=n.
!>
!>          عند الخروج، يتم تدمير محتويات AP.
!>

BP

!>          BP هو مصفوفة حقيقية (REAL array)، البعد (N*(N+1)/2)
!>          عند الدخول، المثلث العلوي أو السفلي للمصفوفة المتماثلة
!>          B، مضغوط عموديًا في مصفوفة خطية. العمود j من B
!>          مخزن في المصفوفة BP كالتالي:
!>          إذا كان UPLO = 'U'، فإن BP(i + (j-1)*j/2) = B(i,j) لـ 1<=i<=j؛
!>          إذا كان UPLO = 'L'، فإن BP(i + (j-1)*(2*n-j)/2) = B(i,j) لـ j<=i<=n.
!>
!>          عند الخروج، المثلث العامل U أو L من التحليل Cholesky
!>          B = U**T*U أو B = L*L**T، بنفس تنسيق التخزين
!>          مثل B.
!>

VL

!>          VL هو عدد حقيقي (REAL)
!>
!>          إذا كان RANGE='V'، الحد الأدنى للنطاق الذي
!>          سيتم البحث فيه عن قيم eigenvalue. VL < VU.
!>          لا يتم الإشارة إليه إذا كان RANGE = 'A' أو 'I'.
!>

VU

!>          VU هو عدد حقيقي (REAL)
!>
!>          إذا كان RANGE='V'، الحد الأعلى للنطاق الذي
!>          سيتم البحث فيه عن قيم eigenvalue. VL < VU.
!>          لا يتم الإشارة إليه إذا كان RANGE = 'A' أو 'I'.
!>

IL

!>          IL هو عدد صحيح (INTEGER)
!>
!>          إذا كان RANGE='I'، مؤشر
!>          أصغر eigenvalue سيتم إرجاعه.
!>          1 <= IL <= IU <= N، إذا كان N > 0؛ IL = 1 و IU = 0 إذا كان N = 0.
!>          لا يتم الإشارة إليه إذا كان RANGE = 'A' أو 'V'.
!>

IU

!>          IU هو عدد صحيح (INTEGER)
!>
!>          إذا كان RANGE='I'، مؤشر
!>          أكبر eigenvalue سيتم إرجاعه.
!>          1 <= IL <= IU <= N، إذا كان N > 0؛ IL = 1 و IU = 0 إذا كان N = 0.
!>          لا يتم الإشارة إليه إذا كان RANGE = 'A' أو 'V'.
!>

ABSTOL

!>          ABSTOL هو عدد حقيقي (REAL)
!>          التسامح المطلق للخطأ لقيم eigenvalue.
!>          يتم قبول eigenvalue تقريبية على أنها متقاربة
!>          عندما يتم تحديد أنها تقع في فترة [a,b]
!>          بعرض أقل من أو يساوي
!>
!>                  ABSTOL + EPS *   max( |a|,|b| ) ,
!>
!>          حيث EPS هي دقة الآلة. إذا كان ABSTOL أقل من
!>          أو يساوي صفر، فسيتم استخدام EPS*|T| بدلاً منه،
!>          حيث |T| هو معيار 1 للمصفوفة ثلاثية الأقطار التي تم الحصول عليها
!>          عن طريق تقليل A إلى شكل ثلاثي الأقطار.
!>
!>          سيتم حساب قيم eigenvalue بأقصى دقة عندما يتم
!>          تعيين ABSTOL إلى ضعف عتبة التدفق السفلي 2*SLAMCH('S')، وليس صفرًا.
!>          إذا عادت هذه الروتين الفرعي مع INFO>0، مما يشير إلى أن بعض
!>          المتجهات المميزة لم تتقارب، حاول تعيين ABSTOL إلى
!>          2*SLAMCH('S').
!>

M

!>          M هو عدد صحيح (INTEGER)
!>          العدد الإجمالي لقيم eigenvalue التي تم العثور عليها.  0 <= M <= N.
!>          إذا كان RANGE = 'A'، فإن M = N، وإذا كان RANGE = 'I'، فإن M = IU-IL+1.
!>

W

!>          W هي مصفوفة حقيقية (REAL array)، البعد (N)
!>          عند الخروج العادي، تحتوي العناصر M الأولى على قيم eigenvalue المختارة
!>          مرتبة تصاعديًا.
!>

Z

!>          Z هي مصفوفة حقيقية (REAL array)، البعد (LDZ, max(1,M))
!>          إذا كان JOBZ = 'N'، فلن يتم الإشارة إلى Z.
!>          إذا كان JOBZ = 'V'، فعندما يكون INFO = 0، تحتوي الأعمدة M الأولى من Z
!>          على المتجهات المميزة المتعامدة للمصفوفة A
!>          المقابلة لقيم eigenvalue المختارة، مع احتواء العمود i
!>          من Z على المتجه المميز المرتبط بـ W(i).
!>          يتم تطبيع المتجهات المميزة كالتالي:
!>          إذا كان ITYPE = 1 أو 2، فإن Z**T*B*Z = I؛
!>          إذا كان ITYPE = 3، فإن Z**T*inv(B)*Z = I.
!>
!>          إذا فشل متجه مميز في التقارب، فإن هذا العمود من Z
!>          يحتوي على أحدث تقريب للمتجه المميز، ويتم
!>          إرجاع فهرس المتجه المميز في IFAIL.
!>          ملاحظة: يجب على المستخدم التأكد من توفير عمود واحد على الأقل (max(1,M))
!>          في المصفوفة Z؛ إذا كان RANGE = 'V'، فإن القيمة الدقيقة لـ M
!>          غير معروفة مسبقًا ويجب استخدام حد أعلى.
!>

LDZ

!>          LDZ هو عدد صحيح (INTEGER)
!>          البعد الرائد للمصفوفة Z.  LDZ >= 1، وإذا
!>          كان JOBZ = 'V'، فإن LDZ >= max(1,N).
!>

WORK

!>          WORK هي مصفوفة حقيقية (REAL array)، البعد (8*N)
!>

IWORK

!>          IWORK هي مصفوفة أعداد صحيحة (INTEGER array)، البعد (5*N)
!>

IFAIL

!>          IFAIL هي مصفوفة أعداد صحيحة (INTEGER array)، البعد (N)
!>          إذا كان JOBZ = 'V'، فعندما يكون INFO = 0، فإن العناصر M الأولى من
!>          IFAIL تكون صفرًا.  إذا كان INFO > 0، فإن IFAIL يحتوي على
!>          فهارس المتجهات المميزة التي فشلت في التقارب.
!>          إذا كان JOBZ = 'N'، فلن يتم الإشارة إلى IFAIL.
!>

INFO

!>          INFO هو عدد صحيح (INTEGER)
!>          = 0:  انتهاء ناجح
!>          < 0:  إذا كان INFO = -i، فإن المعلمة i كانت قيمة غير قانونية
!>          > 0:  SPPTRF أو SSPEVX أرجع رمز خطأ:
!>             <= N:  إذا كان INFO = i، فإن SSPEVX فشل في التقارب؛
!>                    فشل i من المتجهات المميزة في التقارب.  تم تخزين فهارسها
!>                    في المصفوفة IFAIL.
!>             > N:   إذا كان INFO = N + i، لـ 1 <= i <= N، فإن
!>                    الرئيسي الأولي m
