subroutine ssysv_aa_2stage (character uplo, integer n, integer nrhs, real, dimension( lda, * ) a, integer lda, real, dimension( * ) tb, integer ltb, integer, dimension( * ) ipiv, integer, dimension( * ) ipiv2, real, dimension( ldb, * ) b, integer ldb, real, dimension( * ) work, integer lwork, integer info)

SSYSV_AA_2STAGE تحسب حل نظام المعادلات الخطية A * X = B للمصفوفات SY

الغرض:

!>
!> SSYSV_AA_2STAGE تحسب حل نظام المعادلات الخطية الحقيقي
!>    A * X = B,
!> حيث A مصفوفة متماثلة من الرتبة N × N، و X و B مصفوفتان من الرتبة N × NRHS.
!>
!> يتم استخدام خوارزمية Aasen ذات المرحلتين لتحليل A إلى:
!>    A = U**T * T * U,  إذا كان UPLO = 'U', أو
!>    A = L * T * L**T,  إذا كان UPLO = 'L',
!> حيث U (أو L) هي حاصل ضرب مصفوفات التبديل والمصفوفات المثلثية العليا (السفلى) الأحادية، و T
!> هي مصفوفة متماثلة وشريطية. ثم يتم تحليل T باستخدام LU مع تحريك جزئي.
!> بعد ذلك، يتم استخدام الشكل المحلل لـ A لحل نظام المعادلات A * X = B.
!>
!> هذه هي النسخة المحزمة من الخوارزمية، والتي تستدعي BLAS من المستوى 3.
!>

المعلمات

UPLO 
!>          UPLO is CHARACTER*1
!>          = 'U':  المثلث العلوي من A مخزن؛
!>          = 'L':  المثلث السفلي من A مخزن.
!>

N

!>          N is INTEGER
!>          رتبة المصفوفة A.  N >= 0.
!>

NRHS

!>          NRHS is INTEGER
!>          عدد الأطراف اليمنى، أي عدد أعمدة المصفوفة B.  NRHS >= 0.
!>

A

!>          A is REAL array, dimension (LDA,N)
!>          عند الإدخال، المصفوفة المتماثلة A.  إذا كان UPLO = 'U'، فإن الجزء
!>          العلوي المثلثي من A الذي بحجم N×N يحتوي على الجزء المثلثي العلوي
!>          من المصفوفة A، ولا يتم الرجوع إلى الجزء السفلي المثلثي الصارم من A.
!>          إذا كان UPLO = 'L'، فإن الجزء السفلي المثلثي من A الذي بحجم N×N
!>          يحتوي على الجزء المثلثي السفلي من المصفوفة A، ولا يتم الرجوع إلى الجزء
!>          العلوي المثلثي الصارم من A.
!>
!>          عند الخروج، يتم تخزين L تحت (أو فوق) كتل تحت القطر،
!>          عندما يكون UPLO = 'L' (أو 'U').
!>

LDA

!>          LDA is INTEGER
!>          البعد الرائد للمصفوفة A.  LDA >= max(1,N).
!>

TB

!>          TB is REAL array, dimension (MAX(1,LTB))
!>          عند الخروج، تفاصيل تحليل LU للمصفوفة الشريطية.
!>

LTB

!>          LTB is INTEGER
!>          حجم المصفوفة TB. LTB >= MAX(1,4*N)، ويستخدم داخليًا
!>          لاختيار NB بحيث يكون LTB >= (3*NB+1)*N.
!>
!>          إذا كان LTB = -1، فسيتم افتراض استعلام مساحة عمل؛
!>          يقوم الروتين الفرعي فقط بحساب الحجم الأمثل لـ LTB،
!>          يعيد هذا الحجم كمدخل أول لـ TB،
!>          ولا يصدر أي رسالة خطأ متعلقة بـ LTB بواسطة XERBLA.
!>

IPIV

!>          IPIV is INTEGER array, dimension (N)
!>          عند الخروج، تحتوي على تفاصيل التبادلات، أي،
!>          تم تبديل الصف والعمود k من A مع
!>          الصف والعمود IPIV(k).
!>

IPIV2

!>          IPIV2 is INTEGER array, dimension (N)
!>          عند الخروج، تحتوي على تفاصيل التبادلات، أي،
!>          تم تبديل الصف والعمود k من T مع
!>          الصف والعمود IPIV(k).
!>

B

!>          B is REAL array, dimension (LDB,NRHS)
!>          عند الإدخال، مصفوفة الطرف الأيمن B.
!>          عند الخروج، مصفوفة الحل X.
!>

LDB

!>          LDB is INTEGER
!>          البعد الرائد للمصفوفة B.  LDB >= max(1,N).
!>

WORK

!>          WORK is REAL workspace of size (MAX(1,LWORK))
!>          عند الخروج، إذا كان INFO = 0، فإن WORK(1) يعيد LWORK الأمثل.
!>

LWORK

!>          LWORK is INTEGER
!>          حجم WORK. LWORK >= MAX(1,N)، ويستخدم داخليًا
!>          لاختيار NB بحيث يكون LWORK >= N*NB.
!>
!>          إذا كان LWORK = -1، فسيتم افتراض استعلام مساحة عمل؛
!>          يقوم الروتين الفرعي فقط بحساب الحجم الأمثل لمصفوفة WORK،
!>          يعيد هذا الحجم كمدخل أول لمصفوفة WORK،
!>          ولا يصدر أي رسالة خطأ متعلقة بـ LWORK بواسطة XERBLA.
!>

INFO

!>          INFO is INTEGER
!>          = 0:  خروج ناجح
!>          < 0:  إذا كان INFO = -i، فإن المعامل i كان له قيمة غير قانونية.
!>          > 0:  إذا كان INFO = i، فشل تحليل LU الشريطي في العمود i.
!>
