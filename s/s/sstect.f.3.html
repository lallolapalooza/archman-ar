subroutine sstect (integer n, real, dimension( * ) a, real, dimension( * ) b, real shift, integer num)

SSTECT

الغرض:

!>
!>    SSTECT يحسب عدد NUM من القيم الذاتية لمصفوفة ثلاثية الأقطار T
!>    التي تكون أقل من أو تساوي SHIFT. T لها
!>    إدخالات قطرية A(1), ... , A(N), وإدخالات خارج القطر
!>    B(1), ..., B(N-1).
!>    انظر W. Kahan , Report CS41, Computer Science Dept., Stanford
!>    University, July 21, 1966
!>

المعلمات

N 
!>          N هو عدد صحيح (INTEGER)
!>          بعد المصفوفة T ثلاثية الأقطار.
!>

A

!>          A هو مصفوفة حقيقية (REAL array)، البعد (N)
!>          الإدخالات القطرية للمصفوفة T ثلاثية الأقطار.
!>

B

!>          B هو مصفوفة حقيقية (REAL array)، البعد (N-1)
!>          الإدخالات خارج القطر للمصفوفة T ثلاثية الأقطار.
!>

SHIFT

!>          SHIFT هو قيمة حقيقية (REAL)
!>          الإزاحة، تستخدم كما هو موضح تحت الغرض.
!>

NUM

!>          NUM هو عدد صحيح (INTEGER)
!>          عدد القيم الذاتية لـ T الأقل من أو تساوي
!>          SHIFT.
!>
