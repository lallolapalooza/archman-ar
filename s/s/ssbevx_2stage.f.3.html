SSBEVX_2STAGE يحسب القيم الذاتية، واختيارياً، المتجهات الذاتية اليسرى و/أو اليمنى لمصفوفات أخرى

وثائق الوظيفة/الروتين الفرعي

subroutine ssbevx_2stage (character jobz, character range, character uplo, integer n, integer kd, real, dimension( ldab, * ) ab, integer ldab, real, dimension( ldq, * ) q, integer ldq, real vl, real vu, integer il, integer iu, real abstol, integer m, real, dimension( * ) w, real, dimension( ldz, * ) z, integer ldz, real, dimension( * ) work, integer lwork, integer, dimension( * ) iwork, integer, dimension( * ) ifail, integer info)

SSBEVX_2STAGE يحسب القيم الذاتية، واختيارياً، المتجهات الذاتية لمصفوفة شريطية متناظرة A باستخدام تقنية المرحلتين (2stage) للاختزال إلى مصفوفة ثلاثية القطر. يمكن اختيار القيم الذاتية والمتجهات الذاتية عن طريق تحديد نطاق من القيم أو نطاق من المؤشرات للقيم الذاتية المطلوبة.

الغرض:

!>
!> SSBEVX_2STAGE يحسب القيم الذاتية المحددة، واختيارياً، المتجهات الذاتية
!> لمصفوفة شريطية متناظرة حقيقية A باستخدام تقنية المرحلتين (2stage) للاختزال
!> إلى مصفوفة ثلاثية القطر. يمكن اختيار القيم الذاتية والمتجهات الذاتية
!> عن طريق تحديد نطاق من القيم أو نطاق من المؤشرات للقيم الذاتية المطلوبة.
!>

المعلمات

JOBZ 
!>          JOBZ هو CHARACTER*1
!>          = 'N':  حساب القيم الذاتية فقط؛
!>          = 'V':  حساب القيم الذاتية والمتجهات الذاتية.
!>                  غير متوفر في هذا الإصدار.
!>

RANGE

!>          RANGE هو CHARACTER*1
!>          = 'A': سيتم إيجاد جميع القيم الذاتية؛
!>          = 'V': سيتم إيجاد جميع القيم الذاتية في الفترة نصف المفتوحة (VL,VU]
!>                 ؛
!>          = 'I': سيتم إيجاد القيم الذاتية من IL-th إلى IU-th.
!>

UPLO

!>          UPLO هو CHARACTER*1
!>          = 'U':  يتم تخزين المثلث العلوي للمصفوفة A؛
!>          = 'L':  يتم تخزين المثلث السفلي للمصفوفة A.
!>

N

!>          N هو INTEGER
!>          رتبة المصفوفة A.  N >= 0.
!>

KD

!>          KD هو INTEGER
!>          عدد الأقطار الفائقة للمصفوفة A إذا كان UPLO = 'U'،
!>          أو عدد الأقطار الفرعية إذا كان UPLO = 'L'.  KD >= 0.
!>

AB

!>          AB هو مصفوفة REAL، البعد (LDAB, N)
!>          عند الإدخال، المثلث العلوي أو السفلي للمصفوفة الشريطية المتناظرة
!>          A، مخزنة في أول KD+1 صف من المصفوفة.  العمود j من A
!>          مخزن في العمود j من المصفوفة AB
!>          كالتالي:
!>          إذا كان UPLO = 'U'، فإن AB(kd+1+i-j,j) = A(i,j) لـ max(1,j-kd) <= i <= j؛
!>          إذا كان UPLO = 'L'، فإن AB(1+i-j,j) = A(i,j) لـ j <= i <= min(n,j+kd).
!>
!>          عند الخروج، يتم الكتابة فوق AB بقيم تم إنشاؤها أثناء
!>          الاختزال إلى شكل ثلاثي الأقطار. إذا كان UPLO = 'U'، فإن
!>          القطر الفائق الأول والقطر الرئيسي للمصفوفة ثلاثية الأقطار T
!>          يتم إرجاعهما في الصفوف KD و KD+1 من AB، وإذا كان UPLO = 'L'،
!>          فإن القطر الرئيسي والقطر الفرعي الأول لـ T يتم إرجاعهما في
!>          الصفين الأولين من AB.
!>

LDAB

!>          LDAB هو INTEGER
!>          البعد الرائد للمصفوفة AB.  LDAB >= KD + 1.
!>

Q

!>          Q هي مصفوفة REAL، البعد (LDQ, N)
!>          إذا كان JOBZ = 'V'، فإن المصفوفة المتعامدة N-by-N المستخدمة في
!>                         الاختزال إلى شكل ثلاثي الأقطار.
!>          إذا كان JOBZ = 'N'، فلا يتم الرجوع إلى المصفوفة Q.
!>

LDQ

!>          LDQ هو INTEGER
!>          البعد الرائد للمصفوفة Q.  إذا كان JOBZ = 'V'، فإن
!>          LDQ >= max(1,N).
!>

VL

!>          VL هو REAL
!>          إذا كان RANGE='V'، الحد الأدنى للفترة التي
!>          سيتم البحث فيها عن القيم الذاتية. VL < VU.
!>          لا يتم الرجوع إليه إذا كان RANGE = 'A' أو 'I'.
!>

VU

!>          VU هو REAL
!>          إذا كان RANGE='V'، الحد الأعلى للفترة التي
!>          سيتم البحث فيها عن القيم الذاتية. VL < VU.
!>          لا يتم الرجوع إليه إذا كان RANGE = 'A' أو 'I'.
!>

IL

!>          IL هو INTEGER
!>          إذا كان RANGE='I'، مؤشر
!>          أصغر قيمة ذاتية سيتم إرجاعها.
!>          1 <= IL <= IU <= N، إذا كان N > 0؛ IL = 1 و IU = 0 إذا كان N = 0.
!>          لا يتم الرجوع إليه إذا كان RANGE = 'A' أو 'V'.
!>

IU

!>          IU هو INTEGER
!>          إذا كان RANGE='I'، مؤشر
!>          أكبر قيمة ذاتية سيتم إرجاعها.
!>          1 <= IL <= IU <= N، إذا كان N > 0؛ IL = 1 و IU = 0 إذا كان N = 0.
!>          لا يتم الرجوع إليه إذا كان RANGE = 'A' أو 'V'.
!>

ABSTOL

!>          ABSTOL هو REAL
!>          تحمل الخطأ المطلق للقيمة الذاتية.
!>          يتم قبول القيمة الذاتية التقريبية على أنها متقاربة
!>          عندما يتم تحديد أنها تقع في فترة [a,b]
!>          بعرض أقل من أو يساوي
!>
!>                  ABSTOL + EPS *   max( |a|,|b| ) ،
!>
!>          حيث EPS هي دقة الآلة.  إذا كانت ABSTOL أقل من
!>          أو تساوي صفر، فسيتم استخدام EPS*|T| مكانها،
!>          حيث |T| هو المعيار 1 للمصفوفة ثلاثية الأقطار التي تم
!>          الحصول عليها عن طريق اختزال AB إلى شكل ثلاثي الأقطار.
!>
!>          سيتم حساب القيم الذاتية بأكبر دقة عندما يتم
!>          تعيين ABSTOL إلى ضعف حد القيمة الأدنى 2*SLAMCH('S')، وليس صفر.
!>          إذا عادت هذه الروتينات الفرعية مع INFO>0، مما يشير إلى أن بعض
!>          المتجهات الذاتية لم تتقارب، حاول تعيين ABSTOL إلى
!>          2*SLAMCH('S').
!>
!>          انظر  بواسطة Demmel و
!>          Kahan، مذكرة عمل LAPACK #3.
!>

M

!>          M هو INTEGER
!>          إجمالي عدد القيم الذاتية التي تم العثور عليها.  0 <= M <= N.
!>          إذا كان RANGE = 'A'، فإن M = N، وإذا كان RANGE = 'I'، فإن M = IU-IL+1.
!>

W

!>          W هي مصفوفة REAL، البعد (N)
!>          تحتوي العناصر الأولى M على القيم الذاتية المحددة في
!>          ترتيب تصاعدي.
!>

Z

!>          Z هي مصفوفة REAL، البعد (LDZ, max(1,M))
!>          إذا كان JOBZ = 'V'، فإذا كان INFO = 0، فإن الأعمدة الأولى لـ Z
!>          تحتوي على المتجهات الذاتية المتعامدة للمصفوفة A
!>          المقابلة للقيم الذاتية المحددة، حيث يحمل العمود i من Z
!>          المتجه الذاتي المرتبط بـ W(i).
!>          إذا فشل متجه ذاتي في التقارب، فإن هذا العمود من Z
!>          يحتوي على أحدث تقريب للمتجه الذاتي، وسيتم إرجاع مؤشر
!>          المتجه الذاتي في IFAIL.
!>          إذا كان JOBZ = 'N'، فلا يتم الرجوع إلى Z.
!>          ملاحظة: يجب على المستخدم التأكد من توفير عمود واحد على الأقل كحد أقصى (max(1,M))
!>          في المصفوفة Z؛ إذا كان RANGE = 'V'، فإن القيمة الدقيقة لـ M
!>          غير معروفة مسبقًا ويجب استخدام حد أعلى.
!>

LDZ

!>          LDZ هو INTEGER
!>          البعد الرائد للمصفوفة Z.  LDZ >= 1، وإذا
!>          كان JOBZ = 'V'، فإن LDZ >= max(1,N).
!>

WORK

!>          WORK هي مصفوفة REAL، البعد (LWORK)
!>

LWORK

!>          LWORK هو INTEGER
!>          طول المصفوفة WORK. LWORK >= 1، عندما يكون N <= 1؛
!>          غير ذلك
!>          إذا كان JOBZ = 'N' و N > 1، فيجب الاستعلام عن LWORK.
!>                                   LWORK = MAX(1, 7*N, dimension) حيث
!>                                   dimension = (2KD+1)*N + KD*NTHREADS + 2*N
!>                                   حيث KD هو حجم النطاق.
!>
