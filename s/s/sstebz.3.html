توثيق الدالة/الإجراء الفرعي

subroutine sstebz (character range, character order, integer n, real vl, real vu, integer il, integer iu, real abstol, real, dimension( * ) d, real, dimension( * ) e, integer m, integer nsplit, real, dimension( * ) w, integer, dimension( * ) iblock, integer, dimension( * ) isplit, real, dimension( * ) work, integer, dimension( * ) iwork, integer info)

SSTEBZ

الغرض:

!>
!> SSTEBZ تحسب القيم الذاتية لمصفوفة متماثلة ثلاثية القطر T. يمكن للمستخدم طلب جميع القيم الذاتية، أو جميع القيم الذاتية في الفترة النصف مفتوحة (VL, VU]، أو القيم الذاتية من المرتبة IL إلى IU.
!>
!> لتجنب تجاوز السعة، يجب قياس المصفوفة بحيث لا يتجاوز أكبر عنصر فيها القيمة overflow**(1/2) * underflow**(1/4) بالقيمة المطلقة، وللحصول على أدق دقة، يجب ألا تكون أصغر بكثير من ذلك.
!>
!> انظر W. Kahan, Report CS41, Computer Science Dept., Stanford
!> University, July 21, 1966.
!>

المعلمات

RANGE 
!>          RANGE is CHARACTER*1
!>          = 'A': ()   سيتم العثور على جميع القيم الذاتية.
!>          = 'V': () سيتم العثور على جميع القيم الذاتية في الفترة النصف مفتوحة
!>                           (VL, VU].
!>          = 'I': () سيتم العثور على القيم الذاتية من المرتبة IL إلى IU (للمصفوفة بأكملها).
!>

ORDER

!>          ORDER is CHARACTER*1
!>          = 'B': () سيتم تجميع القيم الذاتية حسب
!>                              الكتلة المنفصلة (انظر IBLOCK, ISPLIT) وترتيبها من الأصغر إلى الأكبر داخل
!>                              الكتلة.
!>          = 'E': ()
!>                              سيتم ترتيب القيم الذاتية للمصفوفة بأكملها
!>                              من الأصغر إلى
!>                              الأكبر.
!>

N

!>          N is INTEGER
!>          ترتيب المصفوفة ثلاثية القطر T. N >= 0.
!>

VL

!>          VL is REAL
!>
!>          إذا كان RANGE='V'، الحد الأدنى للفترة
!>          التي سيتم البحث فيها عن القيم الذاتية. القيم الذاتية الأقل من أو تساوي
!>          VL، أو أكبر من VU، لن يتم إرجاعها. VL < VU.
!>          غير مستخدمة إذا كان RANGE = 'A' أو 'I'.
!>

VU

!>          VU is REAL
!>
!>          إذا كان RANGE='V'، الحد الأعلى للفترة
!>          التي سيتم البحث فيها عن القيم الذاتية. القيم الذاتية الأقل من أو تساوي
!>          VL، أو أكبر من VU، لن يتم إرجاعها. VL < VU.
!>          غير مستخدمة إذا كان RANGE = 'A' أو 'I'.
!>

IL

!>          IL is INTEGER
!>
!>          إذا كان RANGE='I'، مؤشر
!>          أصغر قيمة ذاتية سيتم إرجاعها.
!>          1 <= IL <= IU <= N، إذا كان N > 0؛ IL = 1 و IU = 0 إذا كان N = 0.
!>          غير مستخدمة إذا كان RANGE = 'A' أو 'V'.
!>

IU

!>          IU is INTEGER
!>
!>          إذا كان RANGE='I'، مؤشر
!>          أكبر قيمة ذاتية سيتم إرجاعها.
!>          1 <= IL <= IU <= N، إذا كان N > 0؛ IL = 1 و IU = 0 إذا كان N = 0.
!>          غير مستخدمة إذا كان RANGE = 'A' أو 'V'.
!>

ABSTOL

!>          ABSTOL is REAL
!>          التفاوت المطلق للقيم الذاتية. تعتبر القيمة الذاتية
!>          (أو المجموعة) موجودة إذا تم تحديد أنها تقع في فترة عرضها ABSTOL أو
!>          أقل. إذا كان ABSTOL أقل من أو يساوي صفرًا، فسيتم استخدام ULP*|T|
!>          حيث يعني |T| معيار T ذي المسار الواحد.
!>
!>          سيتم حساب القيم الذاتية بأكبر دقة عندما يتم تعيين ABSTOL
!>          إلى ضعف عتبة تجاوز السعة الدنيا 2*SLAMCH('S')، وليس صفرًا.
!>

D

!>          D is REAL array, dimension (N)
!>          عناصر القطر الرئيسية الـ N للمصفوفة ثلاثية القطر T.
!>

E

!>          E is REAL array, dimension (N-1)
!>          العناصر الفرعية غير القطرية الـ (N-1) للمصفوفة ثلاثية القطر T.
!>

M

!>          M is INTEGER
!>          العدد الفعلي للقيم الذاتية التي تم العثور عليها. 0 <= M <= N.
!>          (انظر أيضًا وصف INFO=2,3.)
!>

NSPLIT

!>          NSPLIT is INTEGER
!>          عدد الكتل القطرية في المصفوفة T.
!>          1 <= NSPLIT <= N.
!>

W

!>          W is REAL array, dimension (N)
!>          عند الخروج، ستحتوي العناصر M الأولى من W على
!>          القيم الذاتية. (قد تستخدم SSTEBZ العناصر المتبقية N-M كمساحة عمل.)
!>

IBLOCK

!>          IBLOCK is INTEGER array, dimension (N)
!>          عند كل صف/عمود j حيث يكون E(j) صفرًا أو صغيرًا، تعتبر المصفوفة T
!>          منقسمة إلى مصفوفة قطرية كتلة. عند الخروج، إذا كان INFO = 0، يحدد IBLOCK(i)
!>          إلى أي كتلة (من 1 إلى عدد الكتل) تنتمي القيمة الذاتية W(i).
!>          (قد تستخدم SSTEBZ العناصر المتبقية N-M كمساحة عمل.)
!>

ISPLIT

!>          ISPLIT is INTEGER array, dimension (N)
!>          نقاط الانقسام، التي تنقسم عندها T إلى مصفوفات فرعية.
!>          تتكون المصفوفة الفرعية الأولى من الصفوف/الأعمدة من 1 إلى ISPLIT(1)،
!>          الثانية من الصفوف/الأعمدة ISPLIT(1)+1 حتى ISPLIT(2)،
!>          وهكذا، وتتكون الكتلة رقم NSPLIT من الصفوف/الأعمدة
!>          ISPLIT(NSPLIT-1)+1 حتى ISPLIT(NSPLIT)=N.
!>          (سيتم استخدام أول NSPLIT عنصر فقط فعليًا، ولكن
!>          نظرًا لأن المستخدم لا يمكنه معرفة قيمة NSPLIT مسبقًا، يجب حجز N كلمة لـ ISPLIT.)
!>

WORK

!>          WORK is REAL array, dimension (4*N)
!>

IWORK

!>          IWORK is INTEGER array, dimension (3*N)
!>

INFO

!>          INFO is INTEGER
!>          = 0:  خروج ناجح
!>          < 0:  إذا كان INFO = -i، فقد كان الوسيط i غير قانوني
!>          > 0:  فشلت بعض أو كل القيم الذاتية في التقارب أو
!>                لم يتم حسابها:
!>                =1 أو 3: فشل التنصيف في التقارب لبعض
!>                        القيم الذاتية؛ يتم تمييز هذه القيم الذاتية برقم كتلة سالب. التأثير هو أن
!>                        القيم الذاتية قد لا تكون دقيقة مثل
!>                        التفاوتات المطلقة والنسبية. يحدث هذا بشكل عام بسبب
!>                        حسابات غير دقيقة بشكل غير متوقع.
!>                =2 أو 3: RANGE='I' فقط: لم يتم العثور على جميع القيم الذاتية
!>                        IL:IU.
!>                        التأثير: M < IU+1-IL
!>                        السبب: حسابات غير رتيبة، مما يتسبب في أن يكون تسلسل ستورم غير رتيب.
!>                        الحل: أعد الحساب، باستخدام RANGE='A'، واختر
!>                               القيم الذاتية IL:IU. في بعض الحالات،
!>                               زيادة PARAMETER قد
!>                               تنجح الأمور.
!>                = 4:    RANGE='I'، وفترة غيرشورين
!>                        المستخدمة مبدئيًا كانت صغيرة جدًا. لم يتم حساب أي قيم ذاتية.
!>                        السبب المحتمل: جهازك لديه حسابات نقطة عائمة غير دقيقة.
!>                        الحل: زيادة PARAMETER،
!>                              إعادة التجميع، والمحاولة مرة أخرى.
!>

المعلمات الداخلية:

!>  RELFAC  REAL, default = 2.0e0
!>          The relative tolerance.  An interval (a,b] lies within
!>           if  b-a < RELFAC*ulp*max(|a|,|b|),
!>          where  is the machine precision (distance from 1 to
!>          the next larger floating point number.)
!>
!>  FUDGE   REAL, default = 2
!>          A  to widen the Gershgorin intervals.  Ideally,
!>          a value of 1 should work, but on machines with sloppy
!>          arithmetic, this needs to be larger.  The default for
!>          publicly released versions should be large enough to handle
!>          the worst machine around.  Note that this has no effect
!>          on accuracy of the solution.
!>
