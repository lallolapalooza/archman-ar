قيم الإرجاع

تُرجع الدوال SSL_CTX_dane_enable() و SSL_CTX_dane_mtype_set() و SSL_dane_enable() و SSL_dane_tlsa_add() قيمة موجبة عند النجاح. تشير قيم الإرجاع السالبة إلى مشاكل في الموارد (نفاد الذاكرة، إلخ) في مكتبة SSL، بينما تشير قيمة الإرجاع 0 إلى استخدام غير صحيح أو مدخلات غير صالحة، مثل استخدام شهادة سجل TLSA غير مدعوم، أو محدد، أو نوع مطابق. تشمل المدخلات غير الصالحة أيضًا البيانات المشوهة، سواء كان طول التجزئة لا يتطابق مع خوارزمية التجزئة، أو شهادة Full(0) (صيغة ASN.1 DER الثنائية) أو مفتاح عام يفشل في التحليل.

تُرجع الدوال SSL_get0_dane_authority() و SSL_get0_dane_tlsa() قيمة سالبة عند فشل مصادقة DANE أو عدم تمكينها. تشير قيمة غير سالبة إلى عمق السلسلة التي تطابق سجل TLSA شهادة في السلسلة، أو عمق الشهادة الأعلى، عندما يكون سجل TLSA مفتاحًا عامًا كاملاً هو موقع التوقيع الخاص به.

تُرجع الدوال SSL_CTX_dane_set_flags() و SSL_CTX_dane_clear_flags() و SSL_dane_set_flags() و SSL_dane_clear_flags() قيمة flags السارية قبل استدعائها.

أمثلة

لنفترض أن "smtp.example.com" هو مضيف MX للمجال "example.com"، ولديه سجلات TLSA تم التحقق منها بواسطة DNSSEC. ستقوم الاستدعاءات أدناه بإجراء مصادقة DANE وترتيب مطابقة إما اسم مضيف MX أو اسم المجال الوجهة في شهادة خادم SMTP. يتم دعم الأحرف البديلة، ولكن يجب أن تطابق الملصق بأكمله. يتم استرداد الاسم الفعلي الذي تمت مطابقته في الشهادة (والذي قد يكون حرفًا بديلًا)، ويجب على التطبيق نسخه إذا كان سيتم الاحتفاظ به بعد فترة صلاحية اتصال SSL.

SSL_CTX *ctx;
SSL *ssl;
int (*verify_cb)(int ok, X509_STORE_CTX *sctx) = NULL;
int num_usable = 0;
const char *nexthop_domain = "example.com";
const char *dane_tlsa_domain = "smtp.example.com";
uint8_t usage, selector, mtype;
if ((ctx = SSL_CTX_new(TLS_client_method())) == NULL)
    /* خطأ */
if (SSL_CTX_dane_enable(ctx) <= 0)
    /* خطأ */
if ((ssl = SSL_new(ctx)) == NULL)
    /* خطأ */
if (SSL_dane_enable(ssl, dane_tlsa_domain) <= 0)
    /* خطأ */
/*
 * للعديد من التطبيقات، من الآمن تخطي عمليات التحقق من اسم DANE-EE(3). لا
 * تقم بتعطيل عمليات التحقق ما لم تكن هجمات "مشاركة المفاتيح غير المعروفة"
 * لا تشكل خطرًا على تطبيقك.
 */
SSL_dane_set_flags(ssl, DANE_FLAG_NO_DANE_EE_NAMECHECKS);
if (!SSL_add1_host(ssl, nexthop_domain))
    /* خطأ */
SSL_set_hostflags(ssl, X509_CHECK_FLAG_NO_PARTIAL_WILDCARDS);
for (... كل سجل TLSA ...) {
    unsigned char *data;
    size_t len;
    int ret;
    /* تعيين الاستخدام، والمحدد، ونوع الوسيط، والبيانات، والطول */
    /*
     * عملاء DANE TLS الفرصيون يدعمون فقط DANE-TA(2) أو DANE-EE(3).
     * يعاملون جميع استخدامات الشهادات الأخرى، وخاصة PKIX-TA(0)
     * و PKIX-EE(1)، على أنها غير قابلة للاستخدام.
     */
    switch (usage) {
    default:
    case 0:     /* PKIX-TA(0) */
    case 1:     /* PKIX-EE(1) */
        continue;
    case 2:     /* DANE-TA(2) */
    case 3:     /* DANE-EE(3) */
        break;
    }
    ret = SSL_dane_tlsa_add(ssl, usage, selector, mtype, data, len);
    /* تحرير البيانات حسب الاقتضاء */
    if (ret < 0)
        /* معالجة خطأ داخلي في مكتبة SSL */
    else if (ret == 0)
        /* معالجة سجل TLSA غير قابل للاستخدام */
    else
        ++num_usable;
}
/*
 * في هذه المرحلة، لا يزال وضع التحقق هو الإعداد الافتراضي SSL_VERIFY_NONE.
 * يستخدم عملاء DANE الفرصيون TLS غير المصادق عليه عندما تكون جميع سجلات TLSA
 * غير قابلة للاستخدام، لذا استمر في المصافحة حتى لو فشلت المصادقة.
 */
if (num_usable == 0) {
    /* تسجيل جميع السجلات غير القابلة للاستخدام؟ */
    /* اختياريًا، قم بتعيين verify_cb إلى رد اتصال مناسب غير فارغ. */
    SSL_set_verify(ssl, SSL_VERIFY_NONE, verify_cb);
} else {
    /* سجل واحد على الأقل قابل للاستخدام. نتوقع التحقق من الطرف المقابل */
    /* اختياريًا، قم بتعيين verify_cb إلى رد اتصال مناسب غير فارغ. */
    /*
     * أدناه، نختار فشل المصافحة عند فشل التحقق من الطرف المقابل.
     * بدلاً من ذلك، استخدم وضع التحقق المتساهل SSL_VERIFY_NONE،
     * أكمل المصافحة، تحقق من حالة التحقق، وإذا لم يتم التحقق،
     * قم بقطع الاتصال بأمان على مستوى التطبيق، خاصة إذا كان
     * بروتوكول التطبيق يدعم إبلاغ الخادم بفشل المصادقة.
     */
    SSL_set_verify(ssl, SSL_VERIFY_PEER, verify_cb);
}
/*
 * تحميل أي جلسة محفوظة للاستئناف، مع التأكد من أن الجلسة السابقة
 * طبقت نفس متطلبات الأمان والمصادقة التي كانت متوقعة لاتصال جديد.
 */
/* قم بإجراء مصافحة SSL_connect() ومعالجة الأخطاء هنا */
if (SSL_session_reused(ssl)) {
    if (SSL_get_verify_result(ssl) == X509_V_OK) {
        /*
         * تمت مصادقة الجلسة المستأنفة في الأصل، هذا الاتصال
         * مصادق عليه.
         */
    } else {
        /*
         * لم تتم مصادقة الجلسة المستأنفة في الأصل، هذا الاتصال ليس
         * مصادقًا عليه.
         */
    }
} else if (SSL_get_verify_result(ssl) == X509_V_OK) {
    const char *peername = SSL_get0_peername(ssl);
    EVP_PKEY *mspki = NULL;
    int depth = SSL_get0_dane_authority(ssl, NULL, &mspki);
    if (depth >= 0) {
        (void) SSL_get0_dane_tlsa(ssl, &usage, &selector, &mtype, NULL, NULL);
        printf("DANE TLSA %d %d %d ", usage, selector, mtype);
        if (SSL_get0_peer_rpk(ssl) == NULL)
            printf("%s certificate at depth %d\n",
                   (mspki != NULL) ? "signed the peer" :
                   mdpth ? "matched the TA" : "matched the EE", mdpth);
        else
            printf(bio, "matched the peer raw public key\n");
    }
    if (peername != NULL) {
        /* كانت عمليات التحقق من الاسم ضمن النطاق وطابقت اسم الطرف المقابل */
        printf("Verified peername: %s\n", peername);
    }
} else {
    /*
     * غير مصادق عليه، على الأرجح جميع سجلات TLSA غير قابلة للاستخدام،
     * ولكن ربما يكون رد الاتصال قد قمع إنهاء الاتصال على الرغم من وجود
     * سجلات TLSA قابلة للاستخدام لم تتطابق أي منها. قم بما هو مناسب
     * للاتصالات الجديدة غير المصادق عليها.
     */
}

ملاحظات

من المتوقع أن يكون غالبية العملاء الذين يستخدمون DANE TLS هم "عملاء DANE TLS الفرصيون" بالمعنى المقصود في RFC7672 و RFC7435. أي، سيستخدمون مصادقة DANE عندما يتم نشر سجلات TLSA تم التحقق منها بواسطة DNSSEC لجهة مقابلة معينة، وبخلاف ذلك سيستخدمون TLS غير المصادق عليه أو حتى النص العادي.

يجب على مثل هذه التطبيقات عمومًا التعامل مع أي سجلات TLSA ينشرها الطرف المقابل مع استخدامات PKIX-TA(0) و PKIX-EE(1) على أنها "غير قابلة للاستخدام"، ويجب ألا تشملها ضمن سجلات TLSA المستخدمة لمصادقة اتصالات الأقران. بالإضافة إلى ذلك، قد تكون بعض سجلات TLSA ذات الاستخدامات المدعومة "غير قابلة للاستخدام" نتيجة لمعلمات غير صالحة أو غير مدعومة.

عندما يكون للطرف المقابل سجلات TLSA، ولكن لا يمكن استخدام أي منها، يجب على التطبيق الفرصي تجنب النص العادي، ولكنه لا يستطيع مصادقة الطرف المقابل، لذلك يجب عليه عمومًا المتابعة باتصال غير مصادق عليه. يحتاج التطبيق الفرصي إلى ملاحظة قيمة الإرجاع لكل استدعاء لـ SSL_dane_tlsa_add()، وإذا كانت جميعها ترجع 0 (بسبب معلمات غير صالحة أو غير مدعومة)، فقم بتعطيل مصادقة الطرف المقابل عن طريق استدعاء SSL_set_verify(3) مع mode يساوي SSL_VERIFY_NONE.

    SSL_CTX_dane_enable(3ssl)

    NAME

    SSL_CTX_dane_enable, SSL_CTX_dane_enable_ローバル, SSL_CTX_dane_auth_enable, SSL_CTX_dane_auth_enable_ローバル, SSL_CTX_dane_set_security_errors, SSL_CTX_dane_set_security_errors_ローバル, SSL_CTX_dane_tlsa_add, SSL_CTX_dane_tlsa_add_ローバル - DANE support for TLS

    SYNOPSIS

                #include <openssl/ssl.h>
            
            int SSL_CTX_dane_enable(SSL_CTX *ctx);
            int SSL_CTX_dane_enable_ローバル(SSL_CTX *ctx);
            int SSL_CTX_dane_auth_enable(SSL_CTX *ctx);
            int SSL_CTX_dane_auth_enable_ローバル(SSL_CTX *ctx);
            int SSL_CTX_dane_set_security_errors(SSL_CTX *ctx, int options);
            int SSL_CTX_dane_set_security_errors_ローバル(SSL_CTX *ctx, int options);
            int SSL_CTX_dane_tlsa_add(SSL_CTX *ctx, const char *name, const char *certfile, const char *keyfile, const char *newtoname, const char *selector, const char *mtype);
            int SSL_CTX_dane_tlsa_add_ローバル(SSL_CTX *ctx, const char *name, const char *certfile, const char *keyfile, const char *newtoname, const char *selector, const char *mtype);

    DESCRIPTION

    DANE (DNS-based Authentication of Named Entities) is a mechanism that uses DNS records to specify which certificates or public keys should be considered valid for a given service. It is described in RFC 6698. DANE is an extension of TLS, and it is primarily used to enhance the security of STARTTLS-enabled services, such as SMTP (email) and XMPP (instant messaging).

    The functions provided allow an application to configure DANE support in an SSL_CTX object.

    SSL_CTX_dane_enable() enables DANE checking for the specified SSL_CTX object. When enabled, the client will perform DANE checks on the server's certificate before trusting it.

    SSL_CTX_dane_enable_ローバル() is the same as SSL_CTX_dane_enable(), but it also enables the DANE TLSA lookup for the specified context.

    SSL_CTX_dane_auth_enable() enables DANE authentication for the specified SSL_CTX object. This means that the client will use DANE to authenticate the server's certificate.

    SSL_CTX_dane_auth_enable_ローバル() is the same as SSL_CTX_dane_auth_enable(), but it also enables the DANE TLSA lookup for the specified context.

    SSL_CTX_dane_set_security_errors() sets the security error options for DANE checking. The options are a bitmask. Currently, the following options are supported:

    • DANE_FLAG_NO_SECURITY_ERRORS: Do not report DANE security errors.

    SSL_CTX_dane_set_security_errors_ローバル() is the same as SSL_CTX_dane_set_security_errors(), but it also affects the DANE TLSA lookup.

    SSL_CTX_dane_tlsa_add() adds a TLSA record to the context. The TLSA record is used to specify the certificate or public key to be used for DANE. The parameters are:

    • ctx: The SSL_CTX object.
    • name: The hostname for which the TLSA record is valid.
    • certfile: The path to the certificate file.
    • keyfile: The path to the key file.
    • newtoname: The DNS name of the newton.
    • selector: The selector for the TLSA record.
    • mtype: The matching type for the TLSA record.

    SSL_CTX_dane_tlsa_add_ローバル() is the same as SSL_CTX_dane_tlsa_add(), but it also affects the DANE TLSA lookup.

    RETURN VALUES

    All functions return 1 on success, and 0 on failure.

    NOTES

    DANE support is experimental and may not be available in all OpenSSL builds.
