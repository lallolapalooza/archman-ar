SSBEVD_2STAGE يحسب القيم الذاتية، واختيارياً، المتجهات الذاتية اليسرى و/أو اليمنى لمصفوفات أخرى

توثيق الوظيفة/الإجراء الفرعي

subroutine ssbevd_2stage (character jobz, character uplo, integer n, integer kd, real، dimension( ldab, * ) ab, integer ldab, real، dimension( * ) w, real، dimension( ldz, * ) z, integer ldz, real، dimension( * ) work, integer lwork, integer، dimension( * ) iwork, integer liwork, integer info)

SSBEVD_2STAGE يحسب القيم الذاتية، واختيارياً، المتجهات الذاتية لمصفوفة فرعية متناظرة حقيقية A باستخدام تقنية المرحلتين (2stage) للاختزال إلى مصفوفة ثلاثية الأقطار. إذا تم طلب المتجهات الذاتية، فإنه يستخدم خوارزمية الانقسام والسيطرة.

الغرض:

!>
!> SSBEVD_2STAGE يحسب جميع القيم الذاتية، واختيارياً، المتجهات الذاتية لمصفوفة
!> فرعية متناظرة حقيقية A باستخدام تقنية المرحلتين (2stage) للاختزال إلى مصفوفة ثلاثية الأقطار.
!> إذا تم طلب المتجهات الذاتية، فإنه يستخدم خوارزمية الانقسام والسيطرة.
!>
!>

المعلمات

JOBZ 
!>          JOBZ is CHARACTER*1
!>          = 'N':  حساب القيم الذاتية فقط؛
!>          = 'V':  حساب القيم الذاتية والمتجهات الذاتية.
!>                  غير متوفر في هذا الإصدار.
!>

UPLO

!>          UPLO is CHARACTER*1
!>          = 'U':  يتم تخزين المثلث العلوي للمصفوفة A؛
!>          = 'L':  يتم تخزين المثلث السفلي للمصفوفة A.
!>

N

!>          N is INTEGER
!>          رتبة المصفوفة A. N >= 0.
!>

KD

!>          KD is INTEGER
!>          عدد الأقطار الفائقة للمصفوفة A إذا كان UPLO = 'U'،
!>          أو عدد الأقطار الفرعية إذا كان UPLO = 'L'. KD >= 0.
!>

AB

!>          AB is REAL array, dimension (LDAB, N)
!>          عند الإدخال، المثلث العلوي أو السفلي للمصفوفة الفرعية المتناظرة
!>          A، مخزنة في أول KD+1 صف من المصفوفة. العمود j للمصفوفة A
!>          مخزن في العمود j من المصفوفة AB
!>          على النحو التالي:
!>          إذا كان UPLO = 'U'، فإن AB(kd+1+i-j,j) = A(i,j) لـ max(1,j-kd)<=i<=j؛
!>          إذا كان UPLO = 'L'، فإن AB(1+i-j,j) = A(i,j) لـ j<=i<=min(n,j+kd).
!>
!>          عند الخروج، يتم الكتابة فوق AB بالقيم المتولدة أثناء
!>          الاختزال إلى الشكل الثلاثي الأقطار. إذا كان UPLO = 'U'، يتم إرجاع
!>          القطر الفائق الأول والقطر للمصفوفة الثلاثية الأقطار T
!>          في الصفوف KD و KD+1 من AB، وإذا كان UPLO = 'L'،
!>          يتم إرجاع القطر والقطر الفرعي الأول لـ T في
!>          الصفين الأولين من AB.
!>

LDAB

!>          LDAB is INTEGER
!>          البعد الرائد للمصفوفة AB. LDAB >= KD + 1.
!>

W

!>          W is REAL array, dimension (N)
!>          إذا كان INFO = 0، فإن القيم الذاتية مرتبة تصاعديًا.
!>

Z

!>          Z is REAL array, dimension (LDZ, N)
!>          إذا كان JOBZ = 'V'، فعندما يكون INFO = 0، تحتوي Z على المتجهات الذاتية
!>          ال متعامدة للمصفوفة A، حيث يحتوي العمود i من Z
!>          على المتجه الذاتي المرتبط بـ W(i).
!>          إذا كان JOBZ = 'N'، فلن يتم الرجوع إلى Z.
!>

LDZ

!>          LDZ is INTEGER
!>          البعد الرائد للمصفوفة Z. LDZ >= 1، وإذا
!>          كان JOBZ = 'V'، فإن LDZ >= max(1,N).
!>

WORK

!>          WORK is REAL array, dimension LWORK
!>          عند الخروج، إذا كان INFO = 0، فإن WORK(1) تعيد LWORK المثلى.
!>

LWORK

!>          LWORK is INTEGER
!>          طول المصفوفة WORK. LWORK >= 1، عندما N <= 1؛
!>          بخلاف ذلك
!>          إذا كان JOBZ = 'N' و N > 1، يجب استعلام LWORK.
!>                                   LWORK = MAX(1, dimension) حيث
!>                                   dimension = (2KD+1)*N + KD*NTHREADS + N
!>                                   حيث KD هو حجم النطاق.
!>                                   NTHREADS هو عدد الخيوط المستخدمة عند
!>                                   تفعيل الترجمة المفتوحة openMP، وإلا =1.
!>          إذا كان JOBZ = 'V' و N > 1، يجب استعلام LWORK. غير متاح بعد.
!>
!>          إذا كان LWORK = -1، فإنه يُفترض وجود استعلام مساحة العمل؛ تقوم الروتين
!>          بحساب أبعاد مساحة العمل المثلى لمصفوفتي WORK و IWORK فقط،
!>          وتعيد هذه القيم كمدخلات أولى لمصفوفتي WORK و IWORK، ولا تصدر
!>          رسالة خطأ متعلقة بـ LWORK أو LIWORK بواسطة XERBLA.
!>

IWORK

!>          IWORK is INTEGER array, dimension (MAX(1,LIWORK))
!>          عند الخروج، إذا كان INFO = 0، فإن IWORK(1) تعيد LIWORK المثلى.
!>

LIWORK

!>          LIWORK is INTEGER
!>          بعد مصفوفة IWORK.
!>          إذا كان JOBZ = 'N' أو N <= 1، يجب أن يكون LIWORK على الأقل 1.
!>          إذا كان JOBZ = 'V' و N > 2، يجب أن يكون LIWORK على الأقل 3 + 5*N.
!>
!>          إذا كان LIWORK = -1، فإنه يُفترض وجود استعلام مساحة العمل؛ تقوم الروتين
!>          بحساب أبعاد مساحة العمل المثلى لمصفوفتي WORK و IWORK فقط،
!>          وتعيد هذه القيم كمدخلات أولى لمصفوفتي WORK و IWORK، ولا تصدر
!>          رسالة خطأ متعلقة بـ LWORK أو LIWORK بواسطة XERBLA.
!>

INFO

!>          INFO is INTEGER
!>          = 0:  الخروج بنجاح
!>          < 0:  إذا كان INFO = -i، فإن المعلمة i كانت لها قيمة غير قانونية
!>          > 0:  إذا كان INFO = i، فشلت الخوارزمية في التقارب؛ لم تتقارب العناصر
!>                غير القطرية لشكل ثلاثي الأقطار وسيط إلى الصفر.
!>
