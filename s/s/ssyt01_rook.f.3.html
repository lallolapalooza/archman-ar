توثيق الوظيفة/الروتين الفرعي

subroutine ssyt01_rook (character uplo, integer n, real, dimension( lda, * ) a, integer lda, real, dimension( ldafac, * ) afac, integer ldafac, integer, dimension( * ) ipiv, real, dimension( ldc, * ) c, integer ldc, real, dimension( * ) rwork, real resid)

SSYT01_ROOK

الغرض:

!>
!> SSYT01_ROOK يعيد بناء مصفوفة متناظرة غير محددة A من تجزئتها L*D*L' أو U*D*U'
!> ويحسب معيار الباقي
!>    norm( C - A ) / ( N * norm(A) * EPS ),
!> حيث C هي المصفوفة المعاد بناؤها و EPS هو خطأ الآلة.
!>

المعلمات

UPLO 
!>          UPLO هو CHARACTER*1
!>          يحدد ما إذا كان الجزء العلوي أو السفلي المثلثي للمصفوفة المتناظرة A
!>          مخزنًا:
!>          = 'U':  مثلث علوي
!>          = 'L':  مثلث سفلي
!>

N

!>          N هو INTEGER
!>          عدد الصفوف والأعمدة للمصفوفة A.  N >= 0.
!>

A

!>          A هي مصفوفة REAL، ذات أبعاد (LDA,N)
!>          المصفوفة المتناظرة الأصلية A.
!>

LDA

!>          LDA هو INTEGER
!>          البعد الأساسي للمصفوفة A.  LDA >= max(1,N)
!>

AFAC

!>          AFAC هي مصفوفة REAL، ذات أبعاد (LDAFAC,N)
!>          الشكل المجزأ للمصفوفة A.  AFAC تحتوي على المصفوفة القطرية الجزئية D
!>          والمضاعفات المستخدمة للحصول على العامل L أو U من التجزئة
!>          L*D*L' أو U*D*U' كما تم حسابها بواسطة SSYTRF_ROOK.
!>

LDAFAC

!>          LDAFAC هو INTEGER
!>          البعد الأساسي للمصفوفة AFAC.  LDAFAC >= max(1,N).
!>

IPIV

!>          IPIV هي مصفوفة INTEGER، ذات أبعاد (N)
!>          فهارس المحور من SSYTRF_ROOK.
!>

C

!>          C هي مصفوفة REAL، ذات أبعاد (LDC,N)
!>

LDC

!>          LDC هو INTEGER
!>          البعد الأساسي للمصفوفة C.  LDC >= max(1,N).
!>

RWORK

!>          RWORK هي مصفوفة REAL، ذات أبعاد (N)
!>

RESID

!>          RESID هو REAL
!>          إذا كان UPLO = 'L'، فإن norm(L*D*L' - A) / ( N * norm(A) * EPS )
!>          إذا كان UPLO = 'U'، فإن norm(U*D*U' - A) / ( N * norm(A) * EPS )
!>
