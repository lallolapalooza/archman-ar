subroutine sspgst (integer itype, character uplo, integer n, real, dimension( * ) ap, real, dimension( * ) bp, integer info)

SSPGST

الغرض:

!>
!> SSPGST يقلل مشكلة القيمة الذاتية المتناظرة المحددة القياسية
!> إلى الصيغة القياسية، باستخدام تخزين مضغوط.
!>
!> إذا كان ITYPE = 1، فالمشكلة هي A*x = lambda*B*x،
!> ويتم استبدال A بـ inv(U**T)*A*inv(U) أو inv(L)*A*inv(L**T)
!>
!> إذا كان ITYPE = 2 أو 3، فالمشكلة هي A*B*x = lambda*x أو
!> B*A*x = lambda*x، ويتم استبدال A بـ U*A*U**T أو L**T*A*L.
!>
!> يجب أن تكون B قد تم تحليلها مسبقًا كـ U**T*U أو L*L**T بواسطة SPPTRF.
!>

المعاملات

ITYPE 
!>          ITYPE هو عدد صحيح
!>          = 1: حساب inv(U**T)*A*inv(U) أو inv(L)*A*inv(L**T)؛
!>          = 2 أو 3: حساب U*A*U**T أو L**T*A*L.
!>

UPLO

!>          UPLO هو حرف واحد
!>          = 'U':  يتم تخزين المثلث العلوي لـ A وتحليل B كـ
!>                  U**T*U؛
!>          = 'L':  يتم تخزين المثلث السفلي لـ A وتحليل B كـ
!>                  L*L**T.
!>

N

!>          N هو عدد صحيح
!>          رتبة المصفوفتين A و B.  N >= 0.
!>

AP

!>          AP هو مصفوفة حقيقية، ذات بعد (N*(N+1)/2)
!>          عند الإدخال، المثلث العلوي أو السفلي للمصفوفة المتناظرة
!>          A، مضغوطة حسب العمود في مصفوفة خطية.  العمود j للمصفوفة A
!>          مخزن في المصفوفة AP كالتالي:
!>          إذا كان UPLO = 'U'، فإن AP(i + (j-1)*j/2) = A(i,j) لـ 1<=i<=j؛
!>          إذا كان UPLO = 'L'، فإن AP(i + (j-1)*(2n-j)/2) = A(i,j) لـ j<=i<=n.
!>
!>          عند الخروج، إذا كان INFO = 0، فإن المصفوفة المحولة، المخزنة بنفس
!>          التنسيق مثل A.
!>

BP

!>          BP هي مصفوفة حقيقية، ذات بعد (N*(N+1)/2)
!>          العامل المثلثي من التحليل الشولسكي لـ B،
!>          مخزن بنفس تنسيق A، كما تم إرجاعه بواسطة SPPTRF.
!>

INFO

!>          INFO هو عدد صحيح
!>          = 0:  خروج ناجح
!>          < 0:  إذا كان INFO = -i، فإن المعامل i كان له قيمة غير قانونية
!>
