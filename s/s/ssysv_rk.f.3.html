subroutine ssysv_rk (character uplo, integer n, integer nrhs, real, dimension( lda, * ) a, integer lda, real, dimension( * ) e, integer, dimension( * ) ipiv, real, dimension( ldb, * ) b, integer ldb, real, dimension( * ) work, integer lwork, integer info)

SSYSV_RK تحسب حل نظام المعادلات الخطية A * X = B لمصفوفات SY

الغرض:

!> SSYSV_RK تحسب حل نظام المعادلات الخطية الحقيقي A * X = B، حيث A مصفوفة متماثلة
!> بحجم N*N و X و B مصفوفتان بحجم N*NRHS.
!>
!> يتم استخدام طريقة المحورية القطرية المحدودة لـ Bunch-Kaufman (rook)
!> لتحليل A إلى
!>    A = P*U*D*(U**T)*(P**T)،  إذا كان UPLO = 'U'، أو
!>    A = P*L*D*(L**T)*(P**T)،  إذا كان UPLO = 'L'،
!> حيث U (أو L) مصفوفة مثلثة عليا (أو سفلية) وحدة،
!> U**T (أو L**T) هي منقولة U (أو L)، P مصفوفة تبديل، P**T هي منقولة P،
!> و D مصفوفة متماثلة وقطرية كتلية تحتوي على كتل قطرية بحجم 1*1 و 2*2.
!>
!> يتم استدعاء SSYTRF_RK لحساب تحليل مصفوفة متماثلة حقيقية.
!> يتم بعد ذلك استخدام الصيغة المحللة لـ A لحل نظام المعادلات A * X = B
!> عن طريق استدعاء روتين BLAS3 SSYTRS_3.
!>

المعلمات

UPLO 
!>          UPLO هي CHARACTER*1
!>          تحدد ما إذا كان الجزء المثلثي العلوي أو السفلي من المصفوفة المتماثلة A مخزنًا:
!>          = 'U':  يتم تخزين المثلث العلوي لـ A؛
!>          = 'L':  يتم تخزين المثلث السفلي لـ A؛
!>

N

!>          N هي INTEGER
!>          عدد المعادلات الخطية، أي ترتيب المصفوفة A. N >= 0.
!>

NRHS

!>          NRHS هي INTEGER
!>          عدد الأطراف اليمنى، أي عدد أعمدة المصفوفة B. NRHS >= 0.
!>

A

!>          A هي مصفوفة REAL، البعد (LDA,N)
!>          عند الدخول، المصفوفة المتماثلة A.
!>            إذا كان UPLO = 'U': الجزء العلوي المثلثي A بحجم N*N يحتوي على الجزء المثلثي العلوي للمصفوفة A،
!>            والجزء السفلي المثلثي الصارم لـ A لا يتم الرجوع إليه.
!>
!>            إذا كان UPLO = 'L': الجزء السفلي المثلثي A بحجم N*N يحتوي على الجزء المثلثي السفلي للمصفوفة A،
!>            والجزء العلوي المثلثي الصارم لـ A لا يتم الرجوع إليه.
!>
!>          عند الخروج، إذا كان INFO = 0، القطر للمصفوفة القطرية الكتلية
!>          D والعوامل U أو L كما تم حسابها بواسطة SSYTRF_RK:
!>            أ) فقط العناصر القطرية للمصفوفة القطرية الكتلية المتماثلة
!>               D على قطر A، أي D(k,k) = A(k,k)؛
!>               (تم تخزين عناصر فوق القطر (أو تحت القطر) عند الخروج في المصفوفة E)، و
!>            ب) إذا كان UPLO = 'U': العامل U في الجزء فوق القطري لـ A.
!>               إذا كان UPLO = 'L': العامل L في الجزء تحت القطري لـ A.
!>
!>          لمزيد من المعلومات، راجع وصف روتين DSYTRF_RK.
!>

LDA

!>          LDA هي INTEGER
!>          البعد الرائد للمصفوفة A. LDA >= max(1,N).
!>

E

!>          E هي مصفوفة REAL، البعد (N)
!>          عند الخروج، تحتوي على المخرجات التي تم حسابها بواسطة روتين التحليل
!>          DSYTRF_RK، أي العناصر فوق القطرية (أو تحت القطرية)
!>          للمصفوفة القطرية الكتلية المتماثلة D
!>          مع كتل قطرية بحجم 1*1 أو 2*2، حيث
!>          إذا كان UPLO = 'U': E(i) = D(i-1,i)، i=2:N، E(1) يتم تعيينها إلى 0؛
!>          إذا كان UPLO = 'L': E(i) = D(i+1,i)، i=1:N-1، E(N) يتم تعيينها إلى 0.
!>
!>          ملاحظة: للكتلة القطرية بحجم 1*1 D(k)، حيث
!>          1 <= k <= N، يتم تعيين العنصر E(k) إلى 0 في كل من
!>          حالات UPLO = 'U' أو UPLO = 'L'.
!>
!>          لمزيد من المعلومات، راجع وصف روتين DSYTRF_RK.
!>

IPIV

!>          IPIV هي مصفوفة INTEGER، البعد (N)
!>          تفاصيل التبادلات وهيكل الكتلة لـ D،
!>          كما تم تحديده بواسطة SSYTRF_RK.
!>
!>          لمزيد من المعلومات، راجع وصف روتين DSYTRF_RK.
!>

B

!>          B هي مصفوفة REAL، البعد (LDB,NRHS)
!>          عند الدخول، مصفوفة الطرف الأيمن B بحجم N*NRHS.
!>          عند الخروج، إذا كان INFO = 0، فإن مصفوفة الحل X بحجم N*NRHS.
!>

LDB

!>          LDB هي INTEGER
!>          البعد الرائد للمصفوفة B. LDB >= max(1,N).
!>

WORK

!>          WORK هي مصفوفة REAL، البعد ( MAX(1,LWORK) ).
!>          مصفوفة العمل المستخدمة في مرحلة التحليل.
!>          عند الخروج، إذا كان INFO = 0، فإن WORK(1) ترجع LWORK المثلى.
!>

LWORK

!>          LWORK هي INTEGER
!>          طول WORK. LWORK >= 1. للحصول على أفضل أداء
!>          لمرحلة التحليل LWORK >= max(1,N*NB)، حيث NB هو
!>          حجم الكتلة الأمثل لـ DSYTRF_RK.
!>
!>          إذا كان LWORK = -1، فإنه يُفترض استعلام مساحة العمل؛
!>          تقوم الروتين فقط بحساب الحجم الأمثل لمصفوفة WORK
!>          لمرحلة التحليل، وترجع هذه القيمة كـ
!>          الإدخال الأول لمصفوفة WORK، ولا يتم إصدار رسالة خطأ
!>          متعلقة بـ LWORK بواسطة XERBLA.
!>

INFO

!>          INFO هي INTEGER
!>          = 0: إنهاء ناجح
!>
!>          < 0: إذا كان INFO = -k، فإن المعلمة k كانت لها قيمة غير قانونية
!>
!>          > 0: إذا كان INFO = k، فإن المصفوفة A منفردة، بسبب:
!>                 إذا كان UPLO = 'U': العمود k في الجزء
!>                 المثلثي العلوي لـ A يحتوي على جميع الأصفار.
!>                 إذا كان UPLO = 'L': العمود k في الجزء
!>                 المثلثي السفلي لـ A يحتوي على جميع الأصفار.
!>
!>               لذلك D(k,k) تساوي صفرًا بالضبط، وعناصر ما فوق القطر
!>               للفصل k من U (أو عناصر ما تحت القطر
!>               للعمود k من L) جميعها أصفار. تم الانتهاء من التحليل،
!>               ولكن المصفوفة القطرية الكتلية D منفردة تمامًا،
!>               وسيحدث قسمة على صفر إذا تم استخدامها لحل نظام معادلات.
!>
!>               ملاحظة: INFO يخزن فقط أول حدوث
!>               لتفرد، أي حدوث لاحق للتفرد
!>               لا يتم تخزينه في INFO على الرغم من أن التحليل
!>               يكتمل دائمًا.
!>
