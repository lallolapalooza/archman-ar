subroutine ssysv_rook (character uplo, integer n, integer nrhs, real, dimension( lda, * ) a, integer lda, integer, dimension( * ) ipiv, real, dimension( ldb, * ) b, integer ldb, real, dimension( * ) work, integer lwork, integer info)

SSYSV_ROOK تحسب حل نظام المعادلات الخطية A * X = B للمصفوفات SY

الغرض:

!>
!> SSYSV_ROOK تحسب حل نظام المعادلات الخطية الحقيقية
!>    A * X = B,
!> حيث A مصفوفة متناظرة N×N و X و B مصفوفتان N×NRHS.
!>
!> يتم استخدام طريقة التقطيع القطري لتحليل A كـ
!>    A = U * D * U**T,  إذا كان UPLO = 'U', أو
!>    A = L * D * L**T,  إذا كان UPLO = 'L',
!> حيث U (أو L) هو ناتج ضرب مصفوفات التباديل والمصفوفات المثلثية العلوية (السفلية) الوحدوية، و D متناظرة وقطرية كتلة ذات كتل قطرية 1×1 و 2×2.
!>
!> يتم استدعاء SSYTRF_ROOK لحساب تحليل مصفوفة متناظرة حقيقية A باستخدام طريقة التقطيع القطري لـ Bunch-Kaufman () المحدودة.
!>
!> يتم استخدام الشكل المحلل لـ A لحل نظام
!> المعادلات A * X = B عن طريق استدعاء SSYTRS_ROOK.
!>

المعلمات

UPLO 
!>          UPLO هو CHARACTER*1
!>          = 'U':  يتم تخزين المثلث العلوي من A؛
!>          = 'L':  يتم تخزين المثلث السفلي من A؛
!>

N

!>          N هو INTEGER
!>          عدد المعادلات الخطية، أي رتبة المصفوفة A.  N >= 0.
!>

NRHS

!>          NRHS هو INTEGER
!>          عدد الأطراف اليمنى، أي عدد أعمدة المصفوفة B.  NRHS >= 0.
!>

A

!>          A هي مصفوفة REAL، حجمها (LDA,N)
!>          عند الإدخال، المصفوفة المتناظرة A.  إذا كان UPLO = 'U'، فإن الجزء المثلثي العلوي الذي حجمه N×N من A يحتوي على الجزء المثلثي العلوي للمصفوفة A، ولا تتم الإشارة إلى الجزء المثلثي السفلي الصارم لـ A.  إذا كان UPLO = 'L'، فإن الجزء المثلثي السفلي الذي حجمه N×N من A يحتوي على الجزء المثلثي السفلي للمصفوفة A، ولا تتم الإشارة إلى الجزء المثلثي العلوي الصارم لـ A.
!>
!>          عند الخروج، إذا كان INFO = 0، فإن المصفوفة القطرية الكتلة D والمضاعفات المستخدمة للحصول على العامل U أو L من التحليل A = U*D*U**T أو A = L*D*L**T كما تم حسابها بواسطة SSYTRF_ROOK.
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرائد للمصفوفة A.  LDA >= max(1,N).
!>

IPIV

!>          IPIV هي مصفوفة INTEGER، حجمها (N)
!>          تفاصيل التبادلات وهيكل الكتلة لـ D، كما تم تحديدها بواسطة SSYTRF_ROOK.
!>
!>          إذا كان UPLO = 'U':
!>               إذا كان IPIV(k) > 0، فتم تبديل الصفوف والأعمدة k و IPIV(k) و D(k,k) هي كتلة قطرية 1×1.
!>
!>               إذا كان IPIV(k) < 0 و IPIV(k-1) < 0، فتم تبديل الصفوف والأعمدة k و -IPIV(k) وتم تبديل الصفوف والأعمدة k-1 و -IPIV(k-1)،
!>               D(k-1:k,k-1:k) هي كتلة قطرية 2×2.
!>
!>          إذا كان UPLO = 'L':
!>               إذا كان IPIV(k) > 0، فتم تبديل الصفوف والأعمدة k و IPIV(k) و D(k,k) هي كتلة قطرية 1×1.
!>
!>               إذا كان IPIV(k) < 0 و IPIV(k+1) < 0، فتم تبديل الصفوف والأعمدة k و -IPIV(k) وتم تبديل الصفوف والأعمدة k+1 و -IPIV(k+1)،
!>               D(k:k+1,k:k+1) هي كتلة قطرية 2×2.
!>

B

!>          B هي مصفوفة REAL، حجمها (LDB,NRHS)
!>          عند الإدخال، المصفوفة B ذات الطرف الأيمن N×NRHS.
!>          عند الخروج، إذا كان INFO = 0، فإن المصفوفة الحل X ذات الحجم N×NRHS.
!>

LDB

!>          LDB هو INTEGER
!>          البعد الرائد للمصفوفة B.  LDB >= max(1,N).
!>

WORK

!>          WORK هي مصفوفة REAL، حجمها (MAX(1,LWORK))
!>          عند الخروج، إذا كان INFO = 0، فإن WORK(1) تعيد قيمة LWORK المثلى.
!>

LWORK

!>          LWORK هو INTEGER
!>          طول WORK.  LWORK >= 1، ولأفضل أداء
!>          LWORK >= max(1,N*NB)، حيث NB هو حجم الكتلة الأمثل لـ
!>          SSYTRF_ROOK.
!>
!>          سيتم إجراء TRS باستخدام BLAS المستوى 2
!>
!>          إذا كان LWORK = -1، فسيتم افتراض استعلام عن مساحة العمل؛ يقوم الروتين
!>          بحساب حجم مصفوفة WORK الأمثل فقط، ويعيد
!>          هذه القيمة كالإدخال الأول لمصفوفة WORK، ولا يتم إصدار أي رسالة خطأ
!>          متعلقة بـ LWORK بواسطة XERBLA.
!>

INFO

!>          INFO هو INTEGER
!>          = 0: انتهاء ناجح
!>          < 0: إذا كان INFO = -i، فإن المعامل i كان له قيمة غير قانونية
!>          > 0: إذا كان INFO = i، فإن D(i,i) يساوي صفرًا تمامًا. لقد اكتمل التحليل،
!>               ولكن المصفوفة القطرية الكتلة D منفردة تمامًا، لذلك لم يكن من الممكن حساب الحل.
!>
