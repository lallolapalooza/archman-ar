subroutine sspt01 (character uplo, integer n, real, dimension( * ) a, real, dimension( * ) afac, integer, dimension( * ) ipiv, real, dimension( ldc, * ) c, integer ldc, real, dimension( * ) rwork, real resid)

SSPT01

الغرض:

!>
!> تعيد SSPT01 بناء مصفوفة A غير المحددة والمتناظرة المخزنة A من عامل L*D*L'
!> أو U*D*U' الخاص بها وتحسب الباقي
!>      norm( C - A ) / ( N * norm(A) * EPS ),
!> حيث C هي المصفوفة المعاد بناؤها و EPS هي ثابت العائمة للجهاز.
!>

المعاملات

UPLO 
!>          UPLO is CHARACTER*1
!>          تحدد ما إذا كان الجزء المثلثي العلوي أو السفلي للمصفوفة
!>          المتناظرة A مخزنًا:
!>          = 'U':  مثلثي علوي
!>          = 'L':  مثلثي سفلي
!>

N

!>          N is INTEGER
!>          عدد الصفوف والأعمدة للمصفوفة A.  N >= 0.
!>

A

!>          A is REAL array, dimension (N*(N+1)/2)
!>          المصفوفة المتناظرة الأصلية A، مخزنة كمصفوفة
!>          مثلثية مخزنة.
!>

AFAC

!>          AFAC is REAL array, dimension (N*(N+1)/2)
!>          الشكل المُعامل للمصفوفة A، مخزن كمصفوفة
!>          مثلثية مخزنة. يحتوي AFAC على المصفوفة القطرية الكتلة D
!>          والمُضاعفات المستخدمة للحصول على العامل L أو U من
!>          التحليل الكتل L*D*L' أو U*D*U' كما هو محسوب بواسطة SSPTRF.
!>

IPIV

!>          IPIV is INTEGER array, dimension (N)
!>          مؤشرات المحور من SSPTRF.
!>

C

!>          C is REAL array, dimension (LDC,N)
!>

LDC

!>          LDC is INTEGER
!>          البعد الرئيسي للمصفوفة C.  LDC >= max(1,N).
!>

RWORK

!>          RWORK is REAL array, dimension (N)
!>

RESID

!>          RESID is REAL
!>          If UPLO = 'L', norm(L*D*L' - A) / ( N * norm(A) * EPS )
!>          If UPLO = 'U', norm(U*D*U' - A) / ( N * norm(A) * EPS )
!>
