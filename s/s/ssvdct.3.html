subroutine ssvdct (integer n, real, dimension( * ) s, real, dimension( * ) e, real shift, integer num)

SSVDCT

الغرض:

!>
!> SSVDCT يحسب عدد NUM من القيم الذاتية لمصفوفة ثلاثية قطرية 2*N × 2*N
!> T التي تكون أقل من أو تساوي SHIFT. يتم تشكيل T بوضع أصفار على القطر الرئيسي
!> وجعل الأقطار غير الرئيسية مساوية لـ S(1), E(1), S(2), E(2), ... , E(N-1), S(N).
!> إذا كانت SHIFT موجبة، فإن NUM تساوي N بالإضافة إلى عدد القيم المنفردة لمصفوفة
!> ثنائية القطر B الأقل من أو تساوي SHIFT. هنا B لديها مدخلات قطرية
!> S(1), ..., S(N) ومدخلات فوق قطرية E(1), ... E(N-1).
!> إذا كانت SHIFT سالبة، فإن NUM تساوي عدد القيم المنفردة لـ B الأكبر من أو تساوي -SHIFT.
!>
!> انظر W. Kahan ، تقرير CS41، قسم علوم الحاسوب، جامعة ستانفورد،
!> 21 يوليو 1966
!>

المعاملات

N 
!>          N هو عدد صحيح (INTEGER)
!>          بعد المصفوفة ثنائية القطر B.
!>

S

!>          S هو مصفوفة حقيقية (REAL)، البعد (N)
!>          المدخلات القطرية للمصفوفة ثنائية القطر B.
!>

E

!>          E هي مصفوفة حقيقية (REAL) بالبعد (N-1)
!>          المدخلات فوق القطرية للمصفوفة ثنائية القطر B.
!>

SHIFT

!>          SHIFT هو عدد حقيقي (REAL)
!>          الإزاحة، المستخدمة كما هو موضح تحت الغرض.
!>

NUM

!>          NUM هو عدد صحيح (INTEGER)
!>          عدد القيم الذاتية لـ T الأقل من أو تساوي SHIFT.
!>
