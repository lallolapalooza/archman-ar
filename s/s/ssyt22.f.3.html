subroutine ssyt22 (integer itype, character uplo, integer n, integer m, integer kband, real, dimension( lda, * ) a, integer lda, real, dimension( * ) d, real, dimension( * ) e, real, dimension( ldu, * ) u, integer ldu, real, dimension( ldv, * ) v, integer ldv, real, dimension( * ) tau, real, dimension( * ) work, real, dimension( 2 ) result)

SSYT22

الغرض:

!>
!>      SSYT22  بشكل عام يتحقق من تفكيك بالشكل
!>
!>              A U = U S
!>
!>      حيث A متماثل، وأعمدة U متعامدة، و S قطرية (إذا كان KBAND=0) أو
!>      متماثل ثلاثي الأقطار (إذا كان KBAND=1). إذا كان ITYPE=1، فإن U ممثلة
!>      كمصفوفة كثيفة، وإلا فإن U معبر عنها كناتج تحويلات هوت هاوس،
!>      تُخزن متجهات تحويلات هوت هاوس في المصفوفة  و
!>      ثوابت القياس الخاصة بها في . سنستخدم الحرف
!>       للإشارة إلى ناتج تحويلات هوت هاوس (والذي يجب أن يساوي U).
!>
!>      على وجه التحديد، إذا كان ITYPE=1، فإن:
!>
!>              RESULT(1) = | U**T A U - S | / ( |A| m ulp ) و
!>              RESULT(2) = | I - U**T U | / ( m ulp )
!> 
!>  ITYPE   INTEGER
!>          يحدد نوع الاختبارات التي سيتم إجراؤها.
!>          1: U معبر عنها كمصفوفة متعامدة كثيفة:
!>             RESULT(1) = | A - U S U**T | / ( |A| n ulp ) و
!>             RESULT(2) = | I - U U**T | / ( n ulp )
!>
!>  UPLO    CHARACTER
!>          إذا كان UPLO='U'، سيتم استخدام المثلث العلوي لـ A ولن
!>          يتم الرجوع إلى المثلث السفلي (المطلق). إذا كان UPLO='L'،
!>          سيتم استخدام المثلث السفلي لـ A ولن يتم الرجوع إلى المثلث
!>          العلوي (المطلق).
!>          غير معدل.
!>
!>  N       INTEGER
!>          حجم المصفوفة. إذا كان صفرًا، فإن SSYT22 لا تفعل شيئًا.
!>          يجب أن يكون على الأقل صفرًا.
!>          غير معدل.
!>
!>  M       INTEGER
!>          عدد أعمدة U. إذا كان صفرًا، فإن SSYT22 لا تفعل شيئًا.
!>          يجب أن يكون على الأقل صفرًا.
!>          غير معدل.
!>
!>  KBAND   INTEGER
!>          عرض النطاق للمصفوفة. يمكن أن يكون صفرًا أو واحدًا فقط.
!>          إذا كان صفرًا، فإن S قطرية، ولا يتم الرجوع إلى E. إذا كان
!>          واحدًا، فإن S متماثل ثلاثي الأقطار.
!>          غير معدل.
!>
!>  A       REAL array, dimension (LDA , N)
!>          المصفوفة الأصلية (غير المفككة). يُفترض أنها متماثلة، وسيتم
!>          الرجوع فقط إلى الجزء العلوي (UPLO='U') أو الجزء السفلي
!>          (UPLO='L') فقط.
!>          غير معدل.
!>
!>  LDA     INTEGER
!>          البعد الرئيسي لـ A. يجب أن يكون على الأقل 1 وعلى الأقل N.
!>          غير معدل.
!>
!>  D       REAL array, dimension (N)
!>          القطر للمصفوفة (المتماثلة ثلاثية) القطرية.
!>          غير معدل.
!>
!>  E       REAL array, dimension (N)
!>          العنصر خارج القطر للمصفوفة (المتماثلة ثلاثية) القطرية.
!>          يتم تجاهل E(1)، و E(2) هو العنصر (1,2) و (2,1)، وما إلى ذلك.
!>          لا يتم الرجوع إليه إذا كان KBAND=0.
!>          غير معدل.
!>
!>  U       REAL array, dimension (LDU, N)
!>          إذا كان ITYPE=1 أو 3، تحتوي هذه المصفوفة على المصفوفة المتعامدة
!>          في التفكيك، معبرًا عنها كمصفوفة كثيفة. إذا كان ITYPE=2،
!>          فلا يتم الرجوع إليها.
!>          غير معدل.
!>
!>  LDU     INTEGER
!>          البعد الرئيسي لـ U. يجب أن يكون LDU على الأقل N وعلى الأقل 1.
!>          غير معدل.
!>
!>  V       REAL array, dimension (LDV, N)
!>          إذا كان ITYPE=2 أو 3، يحتوي المثلث السفلي لهذه المصفوفة على
!>          متجهات هوت هاوس التي تصف المصفوفة المتعامدة في التفكيك.
!>          إذا كان ITYPE=1، فلا يتم الرجوع إليها.
!>          غير معدل.
!>
!>  LDV     INTEGER
!>          البعد الرئيسي لـ V. يجب أن يكون LDV على الأقل N وعلى الأقل 1.
!>          غير معدل.
!>
!>  TAU     REAL array, dimension (N)
!>          إذا كان ITYPE >= 2، فإن TAU(j) هو عامل القياس لـ
!>          v(j) v(j)**T في تحويل هوت هاوس H(j) للناتج  U = H(1)...H(n-2)
!>          إذا كان ITYPE < 2، فلا يتم الرجوع إلى TAU.
!>          غير معدل.
!>
!>  WORK    REAL array, dimension (2*N**2)
!>          مساحة عمل.
!>          معدل.
!>
!>  RESULT  REAL array, dimension (2)
!>          القيم المحسوبة بواسطة الاختبارين الموصوفين أعلاه.
!>          القيم محدودة حاليًا بـ 1/ulp، لتجنب تجاوز السعة.
!>          يتم تعديل RESULT(1) دائمًا. يتم تعديل RESULT(2) فقط
!>          إذا كان LDU على الأقل N.
!>          معدل.
!>
