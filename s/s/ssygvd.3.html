توثيق الوظيفة/الإجراء الفرعي

subroutine ssygvd (integer itype, character jobz, character uplo, integer n, real, dimension( lda, * ) a, integer lda, real, dimension( ldb, * ) b, integer ldb, real, dimension( * ) w, real, dimension( * ) work, integer lwork, integer, dimension( * ) iwork, integer liwork, integer info)

SSYGVD

الغرض:

!>
!> SSYGVD تحسب جميع القيم الذاتية، اختياريا، المتجهات الذاتية
!> لمشكلة قياسية متناظرة معرفة من النوع A*x=(lambda)*B*x,  A*Bx=(lambda)*x,  أو B*A*x=(lambda)*x. هنا A و
!> B مفترض أنهما متناظرتان و B موجبة قطعية.
!> إذا كانت المتجهات الذاتية مطلوبة، فإنها تستخدم خوارزمية التقسيم والغزو.
!>
!>

المعاملات

ITYPE 
!>          ITYPE هو عدد صحيح (INTEGER)
!>          يحدد نوع المشكلة المراد حلها:
!>          = 1:  A*x = (lambda)*B*x
!>          = 2:  A*B*x = (lambda)*x
!>          = 3:  B*A*x = (lambda)*x
!>

JOBZ

!>          JOBZ هو حرف واحد (CHARACTER*1)
!>          = 'N':  حساب القيم الذاتية فقط؛
!>          = 'V':  حساب القيم الذاتية والمتجهات الذاتية.
!>

UPLO

!>          UPLO هو حرف واحد (CHARACTER*1)
!>          = 'U':  المثلث العلوي لمصفوفتي A و B مخزن؛
!>          = 'L':  المثلث السفلي لمصفوفتي A و B مخزن.
!>

N

!>          N هو عدد صحيح (INTEGER)
!>          رتبة المصفوفتين A و B.  N >= 0.
!>

A

!>          A هي مصفوفة حقيقية (REAL array)، ذات أبعاد (LDA, N)
!>          عند الدخول، المصفوفة المتناظرة A. إذا كان UPLO = 'U'، فإن
!>          الجزء العلوي المثلثي لـ N-by-N في A يحتوي على
!>          الجزء العلوي المثلثي للمصفوفة A. إذا كان UPLO = 'L'، فإن
!>          الجزء السفلي المثلثي لـ N-by-N في A يحتوي على
!>          الجزء السفلي المثلثي للمصفوفة A.
!>
!>          عند الخروج، إذا كان JOBZ = 'V'، وإذا كان INFO = 0، فإن A تحتوي على
!>          المصفوفة Z للمتجهات الذاتية. المتجهات الذاتية طبيعية
!>          كالتالي:
!>          إذا كان ITYPE = 1 أو 2، Z**T*B*Z = I؛
!>          إذا كان ITYPE = 3، Z**T*inv(B)*Z = I.
!>          إذا كان JOBZ = 'N'، فإن الجزء العلوي (إذا كان UPLO='U')
!>          أو الجزء السفلي (إذا كان UPLO='L') للمصفوفة A، بما في ذلك
!>          القطر، قد تم تدميره.
!>

LDA

!>          LDA هو عدد صحيح (INTEGER)
!>          البعد الرئيسي للمصفوفة A.  LDA >= max(1,N).
!>

B

!>          B هي مصفوفة حقيقية (REAL array)، ذات أبعاد (LDB, N)
!>          عند الدخول، المصفوفة المتناظرة B. إذا كان UPLO = 'U'، فإن
!>          الجزء العلوي المثلثي لـ N-by-N في B يحتوي على
!>          الجزء العلوي المثلثي للمصفوفة B. إذا كان UPLO = 'L'، فإن
!>          الجزء السفلي المثلثي لـ N-by-N في B يحتوي على
!>          الجزء السفلي المثلثي للمصفوفة B.
!>
!>          عند الخروج، إذا كان INFO <= N، فإن الجزء من B الذي يحتوي على المصفوفة
!>          يتم الكتابة فوقه بواسطة العامل المثلثي U أو L من تحليل Cholesky
!>          B = U**T*U أو B = L*L**T.
!>

LDB

!>          LDB هو عدد صحيح (INTEGER)
!>          البعد الرئيسي للمصفوفة B.  LDB >= max(1,N).
!>

W

!>          W هي مصفوفة حقيقية (REAL array)، ذات أبعاد (N)
!>          إذا كان INFO = 0، فإن القيم الذاتية مرتبة تصاعديًا.
!>

WORK

!>          WORK هي مصفوفة حقيقية (REAL array)، ذات أبعاد (MAX(1,LWORK))
!>          عند الخروج، إذا كان INFO = 0، فإن WORK(1) يعيد LWORK الأمثل.
!>

LWORK

!>          LWORK هو عدد صحيح (INTEGER)
!>          بعد المصفوفة WORK.
!>          إذا كان N <= 1،               LWORK >= 1.
!>          إذا كان JOBZ = 'N' و N > 1، LWORK >= 2*N+1.
!>          إذا كان JOBZ = 'V' و N > 1، LWORK >= 1 + 6*N + 2*N**2.
!>
!>          إذا كان LWORK = -1، فيفترض استعلام مساحة العمل؛ الوظيفة
!>          تحسب فقط الأحجام المثلى لمصفوفتي WORK و IWORK،
!>          وتعيد هذه القيم كأول مدخلات في مصفوفتي WORK و IWORK، ولا يتم إصدار
!>          رسالة خطأ تتعلق بـ LWORK أو LIWORK بواسطة XERBLA.
!>

IWORK

!>          IWORK هي مصفوفة أعداد صحيحة (INTEGER array)، ذات أبعاد (MAX(1,LIWORK))
!>          عند الخروج، إذا كان INFO = 0، فإن IWORK(1) يعيد LIWORK الأمثل.
!>

LIWORK

!>          LIWORK هو عدد صحيح (INTEGER)
!>          بعد المصفوفة IWORK.
!>          إذا كان N <= 1،                LIWORK >= 1.
!>          إذا كان JOBZ  = 'N' و N > 1، LIWORK >= 1.
!>          إذا كان JOBZ  = 'V' و N > 1، LIWORK >= 3 + 5*N.
!>
!>          إذا كان LIWORK = -1، فيفترض استعلام مساحة العمل؛ الوظيفة
!>          تحسب فقط الأحجام المثلى لمصفوفتي WORK و IWORK،
!>          وتعيد هذه القيم كأول مدخلات في مصفوفتي WORK و IWORK، ولا يتم إصدار
!>          رسالة خطأ تتعلق بـ LWORK أو LIWORK بواسطة XERBLA.
!>

INFO

!>          INFO هو عدد صحيح (INTEGER)
!>          = 0:  انتهاء ناجح
!>          < 0:  إذا كان INFO = -i، فإن المعامل i كان له قيمة غير قانونية
!>          > 0:  فشل SPOTRF أو SSYEVD في العودة؛
!>             <= N:  إذا كان INFO = i و JOBZ = 'N'، فإن الخوارزمية
!>                    فشلت في التقارب؛ i عناصر غير قطرية لشكل ثلاثي قطري وسيط لم تتقارب إلى
!>                    صفر؛
!>                    إذا كان INFO = i و JOBZ = 'V'، فإن الخوارزمية
!>                    فشلت في حساب قيمة ذاتية أثناء العمل على المصفوفة الفرعية الموجودة في الصفوف والأعمدة INFO/(N+1)
!>                    حتى mod(INFO,N+1)؛
!>             > N:   إذا كان INFO = N + i، حيث 1 <= i <= N، فإن
!>                    المدخل الرئيسي الأصغر من الرتبة i للمصفوفة B ليس موجبًا.
!>                    لم يتم إكمال تحليل B ولم يتم حساب أي قيم ذاتية أو متجهات ذاتية.
!>
