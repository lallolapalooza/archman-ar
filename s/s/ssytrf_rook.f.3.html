subroutine ssytrf_rook (character uplo, integer n, real, dimension( lda, * ) a, integer lda, integer, dimension( * ) ipiv, real, dimension( * ) work, integer lwork, integer info)

SSYTRF_ROOK

الغرض:

!>
!> SSYTRF_ROOK يحسب تحليل مصفوفة متناظرة حقيقية A
!> باستخدام طريقة التنقيط القطري المحدودة لبونش-كوفمان ().
!> شكل التحليل هو
!>
!>    A = U*D*U**T  أو  A = L*D*L**T
!>
!> حيث U (أو L) هو ناتج تبادلات وزوايا عليا (سفلى) جزئية
!> المصفوفات، و D متناظرة ومقسمة قطريًا مع
!> كتل قطرية بحجم 1x1 و 2x2.
!>
!> هذه هي النسخة المجمعة من الخوارزمية، والتي تستدعي BLAS المستوى 3.
!>

المعاملات

UPLO 
!>          UPLO هو CHARACTER*1
!>          = 'U':  يتم تخزين المثلث العلوي من A؛
!>          = 'L':  يتم تخزين المثلث السفلي من A.
!>

N

!>          N هو INTEGER
!>          رتبة المصفوفة A.  N >= 0.
!>

A

!>          A هي مصفوفة REAL، ذات البعد (LDA,N)
!>          عند الدخول، المصفوفة المتناظرة A.  إذا كان UPLO = 'U'، فإن الجزء
!>          المثلثي العلوي بحجم N×N من A يحتوي على الجزء
!>          المثلثي العلوي للمصفوفة A، ولا يتم الرجوع إلى الجزء
!>          المثلثي السفلي الصارم من A.  إذا كان UPLO = 'L'، فإن الجزء
!>          المثلثي السفلي بحجم N×N من A يحتوي على الجزء
!>          المثلثي السفلي للمصفوفة A، ولا يتم الرجوع إلى الجزء
!>          المثلثي العلوي الصارم من A.
!>
!>          عند الخروج، المصفوفة القطرية المقسمة D والمعاملات المستخدمة
!>          للحصول على العامل U أو L (انظر أدناه لمزيد من التفاصيل).
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرائد للمصفوفة A.  LDA >= max(1,N).
!>

IPIV

!>          IPIV هي مصفوفة INTEGER، ذات البعد (N)
!>          تفاصيل التبادلات وهيكل الكتلة لـ D.
!>
!>          إذا كان UPLO = 'U':
!>               إذا كان IPIV(k) > 0، فتم تبادل الصفوف والأعمدة k و IPIV(k)
!>               و D(k,k) هي كتلة قطرية بحجم 1x1.
!>
!>               إذا كان IPIV(k) < 0 و IPIV(k-1) < 0، فتم تبادل الصفوف
!>               والأعمدة k و -IPIV(k) وتم تبادل الصفوف
!>               والأعمدة k-1 و -IPIV(k-1)،
!>               D(k-1:k,k-1:k) هي كتلة قطرية بحجم 2x2.
!>
!>          إذا كان UPLO = 'L':
!>               إذا كان IPIV(k) > 0، فتم تبادل الصفوف والأعمدة k و IPIV(k)
!>               و D(k,k) هي كتلة قطرية بحجم 1x1.
!>
!>               إذا كان IPIV(k) < 0 و IPIV(k+1) < 0، فتم تبادل الصفوف
!>               والأعمدة k و -IPIV(k) وتم تبادل الصفوف
!>               والأعمدة k+1 و -IPIV(k+1)،
!>               D(k:k+1,k:k+1) هي كتلة قطرية بحجم 2x2.
!>

WORK

!>          WORK هي مصفوفة REAL، ذات البعد (MAX(1,LWORK)).
!>          عند الخروج، إذا كان INFO = 0، فإن WORK(1) يعيد LWORK المثلى.
!>

LWORK

!>          LWORK هو INTEGER
!>          طول WORK.  LWORK >= 1.  لأفضل أداء
!>          LWORK >= N*NB، حيث NB هو حجم الكتلة الذي تعيده ILAENV.
!>
!>          إذا كان LWORK = -1، فسيتم افتراض استعلام مساحة العمل؛ يقوم الروتين
!>          بحساب حجم مساحة العمل المثلى فقط، ويعيد
!>          هذه القيمة كمدخل أول لمصفوفة WORK، ولا يتم إصدار أي رسالة خطأ
!>          متعلقة بـ LWORK بواسطة XERBLA.
!>

INFO

!>          INFO هو INTEGER
!>          = 0:  انتهاء ناجح
!>          < 0:  إذا كان INFO = -i، فإن المعامل i كان له قيمة غير قانونية
!>          > 0:  إذا كان INFO = i، فإن D(i,i) يساوي صفرًا تمامًا.  التحليل
!>                قد اكتمل، ولكن المصفوفة القطرية المقسمة D
!>                غير قابلة للانعكاس تمامًا، وسيحدث قسمة على صفر إذا
!>                تم استخدامها لحل نظام من المعادلات.
!>

تفاصيل إضافية:

!>
!>  إذا كان UPLO = 'U'، فإن A = U*D*U**T، حيث
!>     U = P(n)*U(n)* ... *P(k)U(k)* ...,
!>  أي، U هو ناتج المصطلحات P(k)*U(k)، حيث k يتناقص من n إلى
!>  1 بخطوات 1 أو 2، و D هي مصفوفة قطرية مقسمة مع كتل
!>  قطرية بحجم 1x1 و 2x2 D(k).  P(k) هي مصفوفة تبادل كما
!>  تحددها IPIV(k)، و U(k) هي مصفوفة مثلثية عليا موحدة، بحيث إذا كانت الكتلة
!>  القطرية D(k) من الرتبة s (s = 1 أو 2)، فإن
!>
!>             (   I    v    0   )   k-s
!>     U(k) =  (   0    I    0   )   s
!>             (   0    0    I   )   n-k
!>                k-s   s   n-k
!>
!>  إذا كان s = 1، فإن D(k) تستبدل A(k,k)، و v تستبدل A(1:k-1,k).
!>  إذا كان s = 2، فإن المثلث العلوي لـ D(k) يستبدل A(k-1,k-1)، A(k-1,k)،
!>  و A(k,k)، و v تستبدل A(1:k-2,k-1:k).
!>
!>  إذا كان UPLO = 'L'، فإن A = L*D*L**T، حيث
!>     L = P(1)*L(1)* ... *P(k)*L(k)* ...,
!>  أي، L هو ناتج المصطلحات P(k)*L(k)، حيث k يزداد من 1 إلى
!>  n بخطوات 1 أو 2، و D هي مصفوفة قطرية مقسمة مع كتل
!>  قطرية بحجم 1x1 و 2x2 D(k).  P(k) هي مصفوفة تبادل كما
!>  تحددها IPIV(k)، و L(k) هي مصفوفة مثلثية سفلية موحدة، بحيث إذا كانت الكتلة
!>  القطرية D(k) من الرتبة s (s = 1 أو 2)، فإن
!>
!>             (   I    0     0   )  k-1
!>     L(k) =  (   0    I     0   )  s
!>             (   0    v     I   )  n-k-s+1
!>                k-1   s  n-k-s+1
!>
!>  إذا كان s = 1، فإن D(k) تستبدل A(k,k)، و v تستبدل A(k+1:n,k).
!>  إذا كان s = 2، فإن المثلث السفلي لـ D(k) يستبدل A(k,k)، A(k+1,k)،
!>  و A(k+1,k+1)، و v تستبدل A(k+2:n,k:k+1).
!>
