subroutine ssyt01_3 (character uplo, integer n, real, dimension( lda, * ) a, integer lda, real, dimension( ldafac, * ) afac, integer ldafac, real, dimension( * ) e, integer, dimension( * ) ipiv, real, dimension( ldc, * ) c, integer ldc, real, dimension( * ) rwork, real resid)

SSYT01_3

الغرض:

!>
!> SSYT01_3 يعيد بناء مصفوفة متناظرة غير محددة A من تحليلها الكتل L*D*L' أو U*D*U'
!> المحسوب بواسطة SSYTRF_RK (أو SSYTRF_BK) ويحسب الباقي
!>    norm( C - A ) / ( N * norm(A) * EPS ),
!> حيث C هي المصفوفة المعاد بناؤها و EPS هو الخطأ المطلق للآلة.
!>

المعلمات

UPLO 
!>          UPLO is CHARACTER*1
!>          يحدد ما إذا كان الجزء العلوي أو السفلي المثلثي من المصفوفة المتناظرة A مخزنًا:
!>          = 'U':  مثلثي علوي
!>          = 'L':  مثلثي سفلي
!>

N

!>          N is INTEGER
!>          عدد الصفوف والأعمدة للمصفوفة A.  N >= 0.
!>

A

!>          A is DOUBLE PRECISION array, dimension (LDA,N)
!>          المصفوفة المتناظرة الأصلية A.
!>

LDA

!>          LDA is INTEGER
!>          البعد الرئيسي للمصفوفة A.  LDA >= max(1,N)
!>

AFAC

!>          AFAC is DOUBLE PRECISION array, dimension (LDAFAC,N)
!>          القطر للمصفوفة القطرية المتناظرة D والعوامل U أو L
!>          كما تم حسابهما بواسطة SSYTRF_RK و SSYTRF_BK:
!>            أ) فقط العناصر القطرية للمصفوفة القطرية المتناظرة D
!>               على قطر A، أي D(k,k) = A(k,k);
!>               (يجب توفير عناصر فوق القطر (أو تحت القطر) في D
!>                في مصفوفة الإدخال E)، و
!>            ب) إذا كان UPLO = 'U': العامل U في الجزء فوق القطري لـ A.
!>               إذا كان UPLO = 'L': العامل L في الجزء تحت القطري لـ A.
!>

LDAFAC

!>          LDAFAC is INTEGER
!>          البعد الرئيسي للمصفوفة AFAC.
!>          LDAFAC >= max(1,N).
!>

E

!>          E is DOUBLE PRECISION array, dimension (N)
!>          عند الإدخال، تحتوي على عناصر فوق القطر (أو تحت القطر)
!>          للمصفوفة القطرية المتناظرة D
!>          بكتل قطرية بحجم 1x1 أو 2x2، حيث
!>          إذا كان UPLO = 'U': E(i) = D(i-1,i),i=2:N, E(1) غير مستخدم؛
!>          إذا كان UPLO = 'L': E(i) = D(i+1,i),i=1:N-1, E(N) غير مستخدم.
!>

IPIV

!>          IPIV is INTEGER array, dimension (N)
!>          فهارس المحور من SSYTRF_RK (أو SSYTRF_BK).
!>

C

!>          C is DOUBLE PRECISION array, dimension (LDC,N)
!>

LDC

!>          LDC is INTEGER
!>          البعد الرئيسي للمصفوفة C.  LDC >= max(1,N).
!>

RWORK

!>          RWORK is DOUBLE PRECISION array, dimension (N)
!>

RESID

!>          RESID is DOUBLE PRECISION
!>          إذا كان UPLO = 'L'، norm(L*D*L' - A) / ( N * norm(A) * EPS )
!>          إذا كان UPLO = 'U'، norm(U*D*U' - A) / ( N * norm(A) * EPS )
!>
