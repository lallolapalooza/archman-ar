subroutine ssyr (character uplo, integer n, real alpha, real, dimension(*) x, integer incx, real, dimension(lda,*) a, integer lda)

SSYR

الغرض:

!>
!> SSYR   تنفذ عملية المتماثل الرتبة 1
!>
!>    A := alpha*x*x**T + A,
!>
!> حيث alpha هو مقياس حقيقي، x هو متجه n عنصر، و A هي مصفوفة متماثلة n × n.
!>

المعاملات

UPLO 
!>          UPLO is CHARACTER*1
!>           عند الإدخال، تحدد UPLO ما إذا كان الجزء العلوي أو السفلي
!>           من المصفوفة A سيتم الرجوع إليه كما يلي:
!>
!>              UPLO = 'U' or 'u'   يتم الرجوع فقط إلى الجزء العلوي من A
!>                                  .
!>
!>              UPLO = 'L' or 'l'   يتم الرجوع فقط إلى الجزء السفلي من A
!>                                  .
!>

N

!>          N is INTEGER
!>           عند الإدخال، تحدد N رتبة المصفوفة A.
!>           يجب أن تكون N على الأقل صفر.
!>

ALPHA

!>          ALPHA is REAL
!>           عند الإدخال، تحدد ALPHA المقياس alpha.
!>

X

!>          X is REAL array, dimension at least
!>           ( 1 + ( n - 1 )*abs( INCX ) ).
!>           قبل الإدخال، يجب أن يحتوي المصفوفة X المتزايدة على متجه n
!>           x.
!>

INCX

!>          INCX is INTEGER
!>           عند الإدخال، تحدد INCX الزيادة لعناصر X.
!>           يجب ألا تكون INCX صفرًا.
!>

A

!>          A is REAL array, dimension ( LDA, N )
!>           قبل الإدخال مع UPLO = 'U' أو 'u'، يجب أن يحتوي الجزء العلوي
!>           الرائد n × n من المصفوفة A على الجزء العلوي
!>           من المصفوفة المتماثلة، والجزء السفلي المثلثي الصارم
!>           من A لا يتم الرجوع إليه. عند الخروج، الجزء العلوي
!>           من المصفوفة A يتم استبداله بالجزء العلوي
!>           من المصفوفة المحدثة.
!>           قبل الإدخال مع UPLO = 'L' أو 'l'، يجب أن يحتوي الجزء السفلي
!>           الرائد n × n من المصفوفة A على الجزء السفلي
!>           من المصفوفة المتماثلة، والجزء العلوي المثلثي الصارم
!>           من A لا يتم الرجوع إليه. عند الخروج، الجزء السفلي
!>           من المصفوفة A يتم استبداله بالجزء السفلي
!>           من المصفوفة المحدثة.
!>

LDA

!>          LDA is INTEGER
!>           عند الإدخال، تحدد LDA البعد الأول لـ A كما تم تعريفه
!>           في البرنامج (الفرعي) المستدعي. يجب أن يكون LDA على الأقل
!>           max( 1, n ).
!>
