subroutine ssytrs_3 (character uplo, integer n, integer nrhs, real, dimension( lda, * ) a, integer lda, real, dimension( * ) e, integer, dimension( * ) ipiv, real, dimension( ldb, * ) b, integer ldb, integer info)

SSYTRS_3

الغرض:

!> SSYTRS_3 يحل نظام المعادلات الخطية A * X = B بمصفوفة متناظرة حقيقية A باستخدام
!> التحليل الذي تم حسابه بواسطة SSYTRF_RK أو SSYTRF_BK:
!>
!>    A = P*U*D*(U**T)*(P**T) أو A = P*L*D*(L**T)*(P**T)،
!>
!> حيث U (أو L) هي مصفوفة مثلثية علوية (أو سفلية) وحدوية،
!> U**T (أو L**T) هي منقولة U (أو L)، P هي مصفوفة تبديل، P**T هي منقولة P،
!> و D هي مصفوفة قطرية متناظرة وكتلة مع كتل قطرية بحجم 1x1 و 2x2.
!>
!> تستخدم هذه الخوارزمية BLAS من المستوى 3.
!>

المعلمات

UPLO 
!>          UPLO هو CHARACTER*1
!>          يحدد ما إذا كانت تفاصيل التحليل مخزنة كمصفوفة مثلثية علوية أم سفلية:
!>          = 'U':  مثلثية علوية، الشكل هو A = P*U*D*(U**T)*(P**T)؛
!>          = 'L':  مثلثية سفلية، الشكل هو A = P*L*D*(L**T)*(P**T).
!>

N

!>          N هو INTEGER
!>          ترتيب المصفوفة A.  N >= 0.
!>

NRHS

!>          NRHS هو INTEGER
!>          عدد الأطراف اليمنى، أي عدد أعمدة المصفوفة B.  NRHS >= 0.
!>

A

!>          A هي مصفوفة REAL، البعد (LDA,N)
!>          قطر المصفوفة القطرية الكتلة D والعوامل U أو L
!>          كما تم حسابها بواسطة SSYTRF_RK و SSYTRF_BK:
!>            أ) العناصر القطرية فقط للمصفوفة القطرية الكتلة المتناظرة
!>               D على قطر A، أي D(k,k) = A(k,k)؛
!>               (يجب توفير العناصر فوق القطرية (أو تحت القطرية) لـ D
!>                في المصفوفة E)، و
!>            ب) إذا كان UPLO = 'U': العامل U في الجزء فوق القطري لـ A.
!>               إذا كان UPLO = 'L': العامل L في الجزء تحت القطري لـ A.
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرائد للمصفوفة A.  LDA >= max(1,N).
!>

E

!>          E هي مصفوفة REAL، البعد (N)
!>          عند الإدخال، تحتوي على العناصر فوق القطرية (أو تحت القطرية)
!>          للمصفوفة القطرية الكتلة المتناظرة D
!>          مع كتل قطرية بحجم 1x1 أو 2x2، حيث
!>          إذا كان UPLO = 'U': E(i) = D(i-1,i),i=2:N, E(1) لا يتم الرجوع إليها؛
!>          إذا كان UPLO = 'L': E(i) = D(i+1,i),i=1:N-1, E(N) لا يتم الرجوع إليها.
!>
!>          ملاحظة: لكتلة قطرية بحجم 1x1 D(k)، حيث
!>          1 <= k <= N، لا يتم الرجوع إلى العنصر E(k) في كلتا
!>          الحالتين UPLO = 'U' أو UPLO = 'L'.
!>

IPIV

!>          IPIV هي مصفوفة INTEGER، البعد (N)
!>          تفاصيل التبادلات وهيكل الكتلة لـ D
!>          كما تم تحديدها بواسطة SSYTRF_RK أو SSYTRF_BK.
!>

B

!>          B هي مصفوفة REAL، البعد (LDB,NRHS)
!>          عند الإدخال، مصفوفة الطرف الأيمن B.
!>          عند الخروج، مصفوفة الحل X.
!>

LDB

!>          LDB هو INTEGER
!>          البعد الرائد للمصفوفة B.  LDB >= max(1,N).
!>

INFO

!>          INFO هو INTEGER
!>          = 0:  انتهاء ناجح
!>          < 0:  إذا كان INFO = -i، فإن المعامل i كان له قيمة غير قانونية
!>
