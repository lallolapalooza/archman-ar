subroutine ssyequb (character uplo, integer n, real, dimension( lda, * ) a, integer lda, real, dimension( * ) s, real scond, real amax, real, dimension( * ) work, integer info)

SSYEQUB

الغرض:

!>
!> SSYEQUB تحسب عوامل قياس للصف والأعمدة مخصصة لموازنة مصفوفة متناظرة A
!> (بالنسبة للمعيار الإقليدي) وتقليل رقم شرطها. يتم حساب عوامل القياس S
!> بواسطة خوارزمية BIN (انظر المراجع) بحيث تكون المصفوفة المقاسة B
!> بعناصر B(i,j) = S(i)*A(i,j)*S(j) لها رقم شرط في حدود عامل N
!> من أصغر رقم شرط ممكن عبر جميع القياسات القطرية الممكنة.
!>

المعاملات

UPLO 
!>          UPLO هو CHARACTER*1
!>          = 'U':  المثلث العلوي لـ A مخزن؛
!>          = 'L':  المثلث السفلي لـ A مخزن.
!>

N

!>          N هو INTEGER
!>          رتبة المصفوفة A. N >= 0.
!>

A

!>          A هو مصفوفة REAL، حجمها (LDA,N)
!>          المصفوفة المتناظرة N-by-N التي سيتم حساب عوامل قياسها.
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرائد للمصفوفة A. LDA >= max(1,N).
!>

S

!>          S هو مصفوفة REAL، حجمها (N)
!>          إذا كان INFO = 0، تحتوي S على عوامل القياس لـ A.
!>

SCOND

!>          SCOND هو REAL
!>          إذا كان INFO = 0، تحتوي S على نسبة أصغر S(i) إلى
!>          أكبر S(i). إذا كان SCOND >= 0.1 و AMAX ليس كبيرًا جدًا
!>          ولا صغيرًا جدًا، فلا يستحق القياس بواسطة S.
!>

AMAX

!>          AMAX هو REAL
!>          أكبر قيمة مطلقة لأي عنصر مصفوفة. إذا كان AMAX قريبًا جدًا
!>          من الفيضان أو قريبًا جدًا من التدفق السفلي، فيجب قياس
!>          المصفوفة.
!>

WORK

!>          WORK هو مصفوفة REAL، حجمها (2*N)
!>

INFO

!>          INFO هو INTEGER
!>          = 0:  انتهاء ناجح
!>          < 0:  إذا كان INFO = -i، كان الوسيط i غير صالح
!>          > 0:  إذا كان INFO = i، فإن العنصر القطري i غير موجب.
!>
