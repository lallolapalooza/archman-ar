subroutine ssysv_aa (character uplo, integer n, integer nrhs, real, dimension( lda, * ) a, integer lda, integer, dimension( * ) ipiv, real, dimension( ldb, * ) b, integer ldb, real, dimension( * ) work, integer lwork, integer info)

SSYSV_AA يحسب حل نظام المعادلات الخطية A * X = B لمصفوفات SY

الغرض:

!>
!> SSYSV يحسب حل لنظام حقيقي من المعادلات الخطية
!>    A * X = B,
!> حيث A مصفوفة متماثلة N×N و X و B مصفوفتان N×NRHS.
!>
!> يتم استخدام خوارزمية Aasen لتحليل A إلى
!>    A = U**T * T * U,  إذا كان UPLO = 'U', أو
!>    A = L * T * L**T,  إذا كان UPLO = 'L',
!> حيث U (أو L) هي حاصل ضرب تبديلات ومصفوفات مثلثة علوية (سفلية) وحدة،
!> و T مصفوفة متماثلة ثلاثية الأقطار. ثم يتم استخدام الشكل المحلل لـ A
!> لحل نظام المعادلات A * X = B.
!>

المعاملات

UPLO 
!>          UPLO هو CHARACTER*1
!>          = 'U':  يتم تخزين المثلث العلوي من A؛
!>          = 'L':  يتم تخزين المثلث السفلي من A.
!>

N

!>          N هو INTEGER
!>          عدد المعادلات الخطية، أي رتبة المصفوفة A.  N >= 0.
!>

NRHS

!>          NRHS هو INTEGER
!>          عدد الأطراف اليمنى، أي عدد أعمدة المصفوفة B.  NRHS >= 0.
!>

A

!>          A هي مصفوفة REAL، البعد (LDA,N)
!>          عند الدخول، المصفوفة المتماثلة A.  إذا كان UPLO = 'U'، فإن الجزء
!>          العلوي المثلثي الأيسر N×N من A يحتوي على الجزء العلوي
!>          المثلثي للمصفوفة A، والجزء السفلي المثلثي الصارم
!>          من A لا تتم الإشارة إليه.  إذا كان UPLO = 'L'، فإن الجزء
!>          السفلي المثلثي الأيسر N×N من A يحتوي على الجزء السفلي
!>          المثلثي للمصفوفة A، والجزء العلوي المثلثي الصارم
!>          من A لا تتم الإشارة إليه.
!>
!>          عند الخروج، إذا كان INFO = 0، فإن المصفوفة ثلاثية الأقطار T
!>          والمضاعفات المستخدمة للحصول على العامل U أو L من
!>          التحليل A = U**T*T*U أو A = L*T*L**T كما تم حسابه بواسطة
!>          SSYTRF.
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرائد للمصفوفة A.  LDA >= max(1,N).
!>

IPIV

!>          IPIV هي مصفوفة INTEGER، البعد (N)
!>          عند الخروج، تحتوي على تفاصيل التبادلات، أي
!>          الصف والعمود k من A تم تبديلهما بالصف
!>          والعمود IPIV(k).
!>

B

!>          B هي مصفوفة REAL، البعد (LDB,NRHS)
!>          عند الدخول، مصفوفة الطرف الأيمن B من N×NRHS.
!>          عند الخروج، إذا كان INFO = 0، فإن مصفوفة الحل X من N×NRHS.
!>

LDB

!>          LDB هو INTEGER
!>          البعد الرائد للمصفوفة B.  LDB >= max(1,N).
!>

WORK

!>          WORK هي مصفوفة REAL، البعد (MAX(1,LWORK))
!>          عند الخروج، إذا كان INFO = 0، ترجع WORK(1) قيمة LWORK المثلى.
!>

LWORK

!>          LWORK هو INTEGER
!>          طول WORK.  LWORK >= MAX(1,2*N,3*N-2)، وللحصول على
!>          أفضل أداء، LWORK >= MAX(1,N*NB)، حيث NB
!>          هو الحجم الأمثل للكتلة لـ SSYTRF_AA.
!>
!>          إذا كان LWORK = -1، فسيتم افتراض استعلام مساحة العمل؛ الروتين
!>          يحسب فقط الحجم الأمثل لمصفوفة WORK، ويرجع
!>          هذه القيمة كإدخال أول لمصفوفة WORK، ولا يتم إصدار أي رسالة خطأ
!>          متعلقة بـ LWORK بواسطة XERBLA.
!>

INFO

!>          INFO هو INTEGER
!>          = 0: إنهاء ناجح
!>          < 0: إذا كان INFO = -i، فإن المعامل i كان له قيمة غير قانونية
!>          > 0: إذا كان INFO = i، فإن D(i,i) يساوي صفرًا تمامًا. تم إكمال التحليل،
!>               ولكن المصفوفة القطرية الفرعية D مفردة تمامًا، لذلك لا يمكن
!>               حساب الحل.
!>
