subroutine ssycon (character uplo, integer n, real, dimension( lda, * ) a, integer lda, integer, dimension( * ) ipiv, real anorm, real rcond, real, dimension( * ) work, integer, dimension( * ) iwork, integer info)

SSYCON

الغرض:

!>
!> SSYCON يقدر معكوس عدد الشرط (بالقاعدة 1-norm) لمصفوفة متماثلة حقيقية A باستخدام
!> التفكيك A = U*D*U**T أو A = L*D*L**T الذي تم حسابه بواسطة SSYTRF.
!>
!> يتم الحصول على تقدير لـ norm(inv(A))، ويتم حساب معكوس عدد الشرط على أنه
!> RCOND = 1 / (ANORM * norm(inv(A))).
!>

المعلمات

UPLO 
!>          UPLO هو CHARACTER*1
!>          يحدد ما إذا كانت تفاصيل التفكيك مخزنة
!>          كمصفوفة مثلثية علوية أو سفلية.
!>          = 'U':  مثلثية علوية، الشكل هو A = U*D*U**T؛
!>          = 'L':  مثلثية سفلية، الشكل هو A = L*D*L**T.
!>

N

!>          N هو INTEGER
!>          رتبة المصفوفة A.  N >= 0.
!>

A

!>          A هو مصفوفة REAL، البعد (LDA,N)
!>          المصفوفة القطرية الكتلة D والمضاعفات المستخدمة
!>          للحصول على العامل U أو L كما تم حسابه بواسطة SSYTRF.
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرئيسي للمصفوفة A.  LDA >= max(1,N).
!>

IPIV

!>          IPIV هو مصفوفة INTEGER، البعد (N)
!>          تفاصيل التبادلات وهيكل الكتلة لـ D
!>          كما تم تحديده بواسطة SSYTRF.
!>

ANORM

!>          ANORM هو REAL
!>          القاعدة 1-norm للمصفوفة الأصلية A.
!>

RCOND

!>          RCOND هو REAL
!>          معكوس عدد الشرط للمصفوفة A،
!>          محسوب كـ RCOND = 1/(ANORM * AINVNM)، حيث AINVNM هو
!>          تقدير للقاعدة 1-norm لـ inv(A) محسوب في هذه الوحدة.
!>

WORK

!>          WORK هو مصفوفة REAL، البعد (2*N)
!>

IWORK

!>          IWORK هو مصفوفة INTEGER، البعد (N)
!>

INFO

!>          INFO هو INTEGER
!>          = 0:  خروج ناجح
!>          < 0:  إذا كان INFO = -i، فإن المعلمة i كانت ذات قيمة غير قانونية
!>
