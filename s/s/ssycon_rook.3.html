توثيق الوظيفة/الروتين الفرعي

subroutine ssycon_rook (character uplo, integer n, real, dimension( lda, * ) a, integer lda, integer, dimension( * ) ipiv, real anorm, real rcond, real, dimension( * ) work, integer, dimension( * ) iwork, integer info)

SSYCON_ROOK

الغرض:

!>
!> SSYCON_ROOK تقدّر مقلوب عدد الشرط (بالنسبة لمعيار 1) لمصفوفة متماثلة حقيقية A باستخدام التحليل
!> A = U*D*U**T أو A = L*D*L**T التي تم حسابها بواسطة SSYTRF_ROOK.
!>
!> يتم الحصول على تقدير لـ norm(inv(A))، ويتم حساب مقلوب عدد الشرط على أنه RCOND = 1 / (ANORM * norm(inv(A))).
!>

المعلمات

UPLO 
!>          UPLO هو CHARACTER*1
!>          يحدد ما إذا كانت تفاصيل التحليل مخزنة
!>          كمصفوفة مثلثة عليا أو سفلى.
!>          = 'U':  مثلثة عليا، الشكل هو A = U*D*U**T؛
!>          = 'L':  مثلثة سفلى، الشكل هو A = L*D*L**T.
!>

N

!>          N هو INTEGER
!>          رتبة المصفوفة A.  N >= 0.
!>

A

!>          A هي مصفوفة REAL، البعد (LDA,N)
!>          المصفوفة القطرية الجزئية D والمضاعفات المستخدمة
!>          للحصول على العامل U أو L كما تم حسابه بواسطة SSYTRF_ROOK.
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرئيسي للمصفوفة A.  LDA >= max(1,N).
!>

IPIV

!>          IPIV هي مصفوفة INTEGER، البعد (N)
!>          تفاصيل التبادلات والهيكل الجزئي لـ D
!>          كما تم تحديده بواسطة SSYTRF_ROOK.
!>

ANORM

!>          ANORM هي REAL
!>          معيار 1 للمصفوفة الأصلية A.
!>

RCOND

!>          RCOND هي REAL
!>          مقلوب عدد شرط المصفوفة A،
!>          تم حسابه على أنه RCOND = 1/(ANORM * AINVNM)، حيث AINVNM هو
!>          تقدير لمعيار 1 لـ inv(A) المحسوب في هذا الروتين.
!>

WORK

!>          WORK هي مصفوفة REAL، البعد (2*N)
!>

IWORK

!>          IWORK هي مصفوفة INTEGER، البعد (N)
!>

INFO

!>          INFO هو INTEGER
!>          = 0:  انتهاء ناجح
!>          < 0:  إذا كان INFO = -i، فإن المعامل i كان له قيمة غير قانونية
!>
