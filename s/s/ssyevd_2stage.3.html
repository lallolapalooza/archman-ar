subroutine ssyevd_2stage (character jobz, character uplo, integer n, real, dimension( lda, * ) a, integer lda, real, dimension( * ) w, real, dimension( * ) work, integer lwork, integer, dimension( * ) iwork, integer liwork, integer info)

SSYEVD_2STAGE يحسب القيم الذاتية، اختياريًا، المتجهات الذاتية اليسرى و/أو اليمنى لمصفوفات SY

الغرض:

!>
!> SSYEVD_2STAGE يحسب جميع القيم الذاتية، واختياريًا، المتجهات الذاتية لمصفوفة
!> متناظرة حقيقية A باستخدام تقنية 2-stage
!> للاختزال إلى مصفوفة ثلاثية قطرية. إذا كانت المتجهات الذاتية مطلوبة، فإنه يستخدم
!> خوارزمية التفرع والالتقاء.
!>
!>

المعلمات

JOBZ 
!>          JOBZ هو CHARACTER*1
!>          = 'N':  احسب القيم الذاتية فقط؛
!>          = 'V':  احسب القيم الذاتية والمتجهات الذاتية.
!>                  غير متاح في هذا الإصدار.
!>

UPLO

!>          UPLO هو CHARACTER*1
!>          = 'U':  يتم تخزين المثلث العلوي من A؛
!>          = 'L':  يتم تخزين المثلث السفلي من A.
!>

N

!>          N هو INTEGER
!>          رتبة المصفوفة A. N >= 0.
!>

A

!>          A هو مصفوفة REAL، ذات أبعاد (LDA, N)
!>          عند الإدخال، المصفوفة المتناظرة A. إذا كان UPLO = 'U'، فإن
!>          الجزء العلوي المثلثي الرئيسي بحجم N×N من A يحتوي على
!>          الجزء العلوي المثلثي من المصفوفة A. إذا كان UPLO = 'L'، فإن
!>          الجزء السفلي المثلثي الرئيسي بحجم N×N من A يحتوي على
!>          الجزء السفلي المثلثي من المصفوفة A.
!>          عند الخروج، إذا كان JOBZ = 'V'، وإذا كان INFO = 0، فإن A تحتوي على
!>          المتجهات الذاتية المتعامدة للمصفوفة A.
!>          إذا كان JOBZ = 'N'، فإنه عند الخروج، يتم تدمير المثلث السفلي (إذا كان UPLO='L')
!>          أو المثلث العلوي (إذا كان UPLO='U') من A، بما في ذلك
!>          القطر الرئيسي.
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرئيسي للمصفوفة A. LDA >= max(1,N).
!>

W

!>          W هي مصفوفة REAL، ذات أبعاد (N)
!>          إذا كان INFO = 0، فالقيم الذاتية مرتبة تصاعديًا.
!>

WORK

!>          WORK هي مصفوفة REAL،
!>                                         ذات أبعاد (LWORK)
!>          عند الخروج، إذا كان INFO = 0، فإن WORK(1) يعيد قيمة LWORK المثلى.
!>

LWORK

!>          LWORK هو INTEGER
!>          بعد المصفوفة WORK.
!>          إذا كان N <= 1، يجب أن يكون LWORK على الأقل 1.
!>          إذا كان JOBZ = 'N' و N > 1، يجب الاستعلام عن LWORK.
!>                                   LWORK = MAX(1, dimension) حيث
!>                                   dimension = max(stage1,stage2) + (KD+1)*N + 2*N+1
!>                                             = N*KD + N*max(KD+1,FACTOPTNB)
!>                                               + max(2*KD*KD, KD*NTHREADS)
!>                                               + (KD+1)*N + 2*N+1
!>                                   حيث KD هو حجم الحظر للاختزال،
!>                                   FACTOPTNB هو الحجم المستخدم بواسطة خوارزمية QR أو LQ
!>                                   عادةً ما يكون FACTOPTNB=128 خيارًا جيدًا
!>                                   NTHREADS هو عدد الخيوط المستخدمة عندما
!>                                   تم تفعيل تجميع openMP، وإلا =1.
!>          إذا كان JOBZ = 'V' و N > 1، يجب أن يكون LWORK على الأقل
!>                                                1 + 6*N + 2*N**2.
!>
!>          إذا كان LWORK = -1، فسيتم افتراض استعلام عن مساحة العمل؛ الروتين
!>          يقوم فقط بحساب الأحجام المثلى لمصفوفتي WORK و IWORK،
!>          يعيد هذه القيم كأول إدخالات في مصفوفتي WORK و IWORK، ولا يتم إصدار
!>          أي رسالة خطأ متعلقة بـ LWORK أو LIWORK بواسطة XERBLA.
!>

IWORK

!>          IWORK هي مصفوفة INTEGER، ذات أبعاد (MAX(1,LIWORK))
!>          عند الخروج، إذا كان INFO = 0، فإن IWORK(1) يعيد قيمة LIWORK المثلى.
!>

LIWORK

!>          LIWORK هو INTEGER
!>          بعد المصفوفة IWORK.
!>          إذا كان N <= 1، يجب أن يكون LIWORK على الأقل 1.
!>          إذا كان JOBZ  = 'N' و N > 1، يجب أن يكون LIWORK على الأقل 1.
!>          إذا كان JOBZ  = 'V' و N > 1، يجب أن يكون LIWORK على الأقل 3 + 5*N.
!>
!>          إذا كان LIWORK = -1، فسيتم افتراض استعلام عن مساحة العمل؛ الروتين
!>          يقوم فقط بحساب الأحجام المثلى لمصفوفتي WORK و IWORK،
!>          يعيد هذه القيم كأول إدخالات في مصفوفتي WORK و IWORK، ولا يتم إصدار
!>          أي رسالة خطأ متعلقة بـ LWORK أو LIWORK بواسطة XERBLA.
!>

INFO

!>          INFO هو INTEGER
!>          = 0:  الخروج ناجح
!>          < 0:  إذا كان INFO = -i، فإن الوسيط i كان له قيمة غير قانونية
!>          > 0:  إذا كان INFO = i و JOBZ = 'N'، فإن الخوارزمية فشلت
!>                في التقارب؛ لم تتقارب i عناصر غير قطرية في شكل ثلاثي قطري وسيط
!>                إلى الصفر؛
!>                إذا كان INFO = i و JOBZ = 'V'، فإن الخوارزمية فشلت
!>                في حساب قيمة ذاتية أثناء العمل على المصفوفة الفرعية
!>                التي تقع في الصفوف والأعمدة INFO/(N+1) حتى
!>                mod(INFO,N+1).
!>
