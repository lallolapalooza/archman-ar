subroutine ssytrf_aa_2stage (character uplo, integer n, real, dimension( lda, * ) a, integer lda, real, dimension( * ) tb, integer ltb, integer, dimension( * ) ipiv, integer, dimension( * ) ipiv2, real, dimension( * ) work, integer lwork, integer info)

SSYTRF_AA_2STAGE

الغرض:

!>
!> SSYTRF_AA_2STAGE يحسب تحليل مصفوفة حقيقية متناظرة A
!> باستخدام خوارزمية Aasen. شكل التحليل هو
!>
!>    A = U**T*T*U  أو  A = L*T*L**T
!>
!> حيث U (أو L) هو ناتج ضرب مصفوفة تبديل ومصفوفة ثلاثية علوية (سفلية)
!> وحدوية، و T هي مصفوفة نطاق متناظرة بعرض نطاق NB (يتم اختيار NB داخلياً وتخزينه في TB( 1 )، ويتم
!> تحليل T إلى LU مع تحويل جزئي.
!>
!> هذه هي النسخة المحظورة من الخوارزمية، والتي تستدعي BLAS من المستوى 3.
!>

المعاملات

UPLO 
!>          UPLO هو CHARACTER*1
!>          = 'U':  يتم تخزين المثلث العلوي لـ A؛
!>          = 'L':  يتم تخزين المثلث السفلي لـ A؛
!>

N

!>          N هو INTEGER
!>          ترتيب المصفوفة A. N >= 0.
!>

A

!>          A هو مصفوفة REAL، حجمها (LDA,N)
!>          عند الإدخال، المصفوفة المتناظرة A. إذا كان UPLO = 'U'، فإن
!>          الجزء العلوي المثلثي من A بحجم N-by-N يحتوي على الجزء
!>          المثلثي العلوي للمصفوفة A، ولا يتم الإشارة إلى الجزء المثلثي السفلي
!>          بشكل صارم لـ A. إذا كان UPLO = 'L'، فإن
!>          الجزء السفلي المثلثي من A بحجم N-by-N يحتوي على الجزء
!>          المثلثي السفلي للمصفوفة A، ولا يتم الإشارة إلى الجزء المثلثي العلوي
!>          بشكل صارم لـ A.
!>
!>          عند الخروج، يتم تخزين L أسفل (أو فوق) كتل القطر الفرعي،
!>          عندما يكون UPLO هو 'L' (أو 'U').
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرئيسي للمصفوفة A. LDA >= max(1,N).
!>

TB

!>          TB هو مصفوفة REAL، حجمها (LTB)
!>          عند الخروج، تفاصيل تحليل LU لمصفوفة النطاق.
!>

LTB

!>          LTB هو INTEGER
!>          حجم المصفوفة TB. LTB >= MAX(1,4*N)، يتم استخدامه داخلياً
!>          لاختيار NB بحيث LTB >= (3*NB+1)*N.
!>
!>          إذا كان LTB = -1، فإنه يُفترض وجود استعلام مساحة عمل؛
!>          يحسب الروتين فقط الحجم الأمثل لـ LTB،
!>          يعيد هذا الحجم كمدخل أول لـ TB،
!>          ولا يتم إصدار أي رسالة خطأ متعلقة بـ LTB بواسطة XERBLA.
!>

IPIV

!>          IPIV هو مصفوفة INTEGER، حجمها (N)
!>          عند الخروج، تحتوي على تفاصيل التبديلات، أي
!>          تم تبديل الصف والعمود k من A مع
!>          الصف والعمود IPIV(k).
!>

IPIV2

!>          IPIV2 هو مصفوفة INTEGER، حجمها (N)
!>          عند الخروج، تحتوي على تفاصيل التبديلات، أي
!>          تم تبديل الصف والعمود k من T مع
!>          الصف والعمود IPIV(k).
!>

WORK

!>          WORK هو مساحة عمل REAL بحجم (MAX(1,LWORK))
!>

LWORK

!>          LWORK هو INTEGER
!>          حجم WORK. LWORK >= MAX(1,N)، يتم استخدامه داخلياً
!>          لاختيار NB بحيث LWORK >= N*NB.
!>
!>          إذا كان LWORK = -1، فإنه يُفترض وجود استعلام مساحة عمل؛
!>          يحسب الروتين فقط الحجم الأمثل لمصفوفة WORK،
!>          يعيد هذا الحجم كمدخل أول لمصفوفة WORK،
!>          ولا يتم إصدار أي رسالة خطأ متعلقة بـ LWORK بواسطة XERBLA.
!>

INFO

!>          INFO هو INTEGER
!>          = 0:  انتهاء ناجح
!>          < 0:  إذا كان INFO = -i، فإن المعامل i كان له قيمة غير قانونية.
!>          > 0:  إذا كان INFO = i، فشل تحليل LU النطاقي في العمود i
!>
