subroutine ssytd2 (uplo, n, a, lda, d, e, tau, info)
SSYTD2 تقلل مصفوفة متناظرة إلى شكل متماثل ثلاثي الأقطار بواسطة تحويل تشابه متعامد (خوارزمية غير مجمعة).

توثيق الوظيفة/الإجراء الفرعي

subroutine ssytd2 (character uplo, integer n, real, dimension( lda, * ) a, integer lda, real, dimension( * ) d, real, dimension( * ) e, real, dimension( * ) tau, integer info)

SSYTD2 تقلل مصفوفة متناظرة حقيقية A إلى شكل ثلاثي الأقطار متناظر T بواسطة تحويل تشابه متعامد: Q**T * A * Q = T.

الغرض:

!>
!> SSYTD2 تقلل مصفوفة متناظرة حقيقية A إلى شكل ثلاثي الأقطار
!> متناظر T بواسطة تحويل تشابه متعامد: Q**T * A * Q = T.
!>

المعلمات

UPLO 
!>          UPLO is CHARACTER*1
!>          تحدد ما إذا كان الجزء المثلثي العلوي أو السفلي من المصفوفة
!>          المتناظرة A مخزنًا:
!>          = 'U':  مثلثي علوي
!>          = 'L':  مثلثي سفلي
!>

N

!>          N is INTEGER
!>          ترتيب المصفوفة A. N >= 0.
!>

A

!>          A is REAL array, dimension (LDA,N)
!>          عند الدخول، المصفوفة المتناظرة A. إذا كان UPLO = 'U'، فإن الجزء
!>          المثلثي العلوي n-by-n الرائد من A يحتوي على الجزء المثلثي
!>          العلوي من المصفوفة A، والجزء السفلي الصارم من A لا تتم الإشارة إليه.
!>          إذا كان UPLO = 'L'، فإن الجزء المثلثي السفلي n-by-n الرائد من A
!>          يحتوي على الجزء المثلثي السفلي من المصفوفة A، والجزء العلوي
!>          الصارم من A لا تتم الإشارة إليه.
!>          عند الخروج، إذا كان UPLO = 'U'، يتم استبدال القطر والعناصر فوق القطر
!>          الأولى بالعناصر المقابلة للمصفوفة ثلاثية الأقطار T، وتمثل العناصر
!>          فوق القطر الأول، مع المصفوفة TAU، المصفوفة المتعامدة Q كناتج
!>          للانعكاسات الأولية؛ إذا كان UPLO = 'L'، يتم استبدال القطر والعناصر
!>          تحت القطر الأولى بالعناصر المقابلة للمصفوفة ثلاثية الأقطار T، وتمثل
!>          العناصر تحت القطر الأول، مع المصفوفة TAU، المصفوفة المتعامدة Q كناتج
!>          للانعكاسات الأولية. انظر التفاصيل الإضافية.
!>

LDA

!>          LDA is INTEGER
!>          البعد الرائد للمصفوفة A. LDA >= max(1,N).
!>

D

!>          D is REAL array, dimension (N)
!>          عناصر القطر للمصفوفة ثلاثية الأقطار T:
!>          D(i) = A(i,i).
!>

E

!>          E is REAL array, dimension (N-1)
!>          عناصر خارج القطر للمصفوفة ثلاثية الأقطار T:
!>          E(i) = A(i,i+1) إذا كان UPLO = 'U'، E(i) = A(i+1,i) إذا كان UPLO = 'L'.
!>

TAU

!>          TAU is REAL array, dimension (N-1)
!>          العوامل القياسية للانعكاسات الأولية (انظر التفاصيل الإضافية).
!>

INFO

!>          INFO is INTEGER
!>          = 0:  خروج ناجح
!>          < 0:  إذا كان INFO = -i، فإن المعلمة i كانت لها قيمة غير قانونية.
!>

التفاصيل الإضافية:

!>
!>  إذا كان UPLO = 'U'، يتم تمثيل المصفوفة Q كناتج للانعكاسات الأولية
!>
!>     Q = H(n-1) . . . H(2) H(1).
!>
!>  كل H(i) له الشكل
!>
!>     H(i) = I - tau * v * v**T
!>
!>  حيث tau هو عامل حقيقي، و v هو متجه حقيقي مع
!>  v(i+1:n) = 0 و v(i) = 1؛ يتم تخزين v(1:i-1) عند الخروج في
!>  A(1:i-1,i+1)، و tau في TAU(i).
!>
!>  إذا كان UPLO = 'L'، يتم تمثيل المصفوفة Q كناتج للانعكاسات الأولية
!>
!>     Q = H(1) H(2) . . . H(n-1).
!>
!>  كل H(i) له الشكل
!>
!>     H(i) = I - tau * v * v**T
!>
!>  حيث tau هو عامل حقيقي، و v هو متجه حقيقي مع
!>  v(1:i) = 0 و v(i+1) = 1؛ يتم تخزين v(i+2:n) عند الخروج في A(i+2:n,i)،
!>  و tau في TAU(i).
!>
!>  يتم توضيح محتويات A عند الخروج بالأمثلة التالية
!>  مع n = 5:
!>
!>  إذا كان UPLO = 'U':                       إذا كان UPLO = 'L':
!>
!>    (  d   e   v2  v3  v4 )              (  d                  )
!>    (      d   e   v3  v4 )              (  e   d              )
!>    (          d   e   v4 )              (  v1  e   d          )
!>    (              d   e  )              (  v1  v2  e   d      )
!>    (                  d  )              (  v1  v2  v3  e   d  )
!>
!>  حيث d و e يمثلان عناصر القطر والعناصر خارج القطر لـ T، و vi
!!>  يمثل عنصرًا من المتجه الذي يحدد H(i).
!>
