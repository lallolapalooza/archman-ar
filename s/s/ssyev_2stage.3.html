توثيق الوظيفة/الروتين الفرعي

subroutine ssyev_2stage (character jobz, character uplo, integer n, real, dimension( lda, * ) a, integer lda, real, dimension( * ) w, real, dimension( * ) work, integer lwork, integer info)

SSYEV_2STAGE تحسب القيم الذاتية، اختيارياً، المتجهات الذاتية اليسرى و/أو اليمنى لمصفوفات SY

الغرض:

!>
!> SSYEV_2STAGE تحسب جميع القيم الذاتية، واختيارياً، المتجهات الذاتية لمصفوفة متناظرة حقيقية A باستخدام تقنية المرحلتين
!> للاختزال إلى ثلاثية الأقطار.
!>

المعاملات

JOBZ 
!>          JOBZ هو CHARACTER*1
!>          = 'N':  حساب القيم الذاتية فقط؛
!>          = 'V':  حساب القيم الذاتية والمتجهات الذاتية.
!>                  غير متوفر في هذا الإصدار.
!>

UPLO

!>          UPLO هو CHARACTER*1
!>          = 'U':  المثلث العلوي من A مخزن؛
!>          = 'L':  المثلث السفلي من A مخزن.
!>

N

!>          N هو INTEGER
!>          رتبة المصفوفة A.  N >= 0.
!>

A

!>          A هو مصفوفة REAL، البعد (LDA, N)
!>          عند الإدخال، المصفوفة المتناظرة A.  إذا كان UPLO = 'U'، فإن
!>          الجزء العلوي المثلثي العلوي الذي يبلغ حجمه N-by-N من A يحتوي على
!>          الجزء العلوي المثلثي من المصفوفة A.  إذا كان UPLO = 'L'، فإن
!>          الجزء السفلي المثلثي السفلي الذي يبلغ حجمه N-by-N من A يحتوي على
!>          الجزء السفلي المثلثي من المصفوفة A.
!>          عند الخروج، إذا كان JOBZ = 'V'، وإذا كان INFO = 0، فإن A تحتوي على
!>          المتجهات الذاتية المتعامدة القياسية للمصفوفة A.
!>          إذا كان JOBZ = 'N'، فإن عند الخروج، المثلث السفلي (إذا كان UPLO='L')
!>          أو المثلث العلوي (إذا كان UPLO='U') من A، بما في ذلك
!>          القطر، يتم تدميره.
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرئيسي للمصفوفة A.  LDA >= max(1,N).
!>

W

!>          W هو مصفوفة REAL، البعد (N)
!>          إذا كان INFO = 0، فإن القيم الذاتية مرتبة تصاعدياً.
!>

WORK

!>          WORK هو مصفوفة REAL، البعد LWORK
!>          عند الخروج، إذا كان INFO = 0، فإن WORK(1) يعيد LWORK الأمثل.
!>

LWORK

!>          LWORK هو INTEGER
!>          طول المصفوفة WORK. LWORK >= 1، عندما N <= 1؛
!>          بخلاف ذلك
!>          إذا كان JOBZ = 'N' و N > 1، يجب الاستعلام عن LWORK.
!>                                   LWORK = MAX(1, dimension) حيث
!>                                   dimension = max(stage1,stage2) + (KD+1)*N + 2*N
!>                                             = N*KD + N*max(KD+1,FACTOPTNB)
!>                                               + max(2*KD*KD, KD*NTHREADS)
!>                                               + (KD+1)*N + 2*N
!>                                   حيث KD هو حجم الحجب للاختزال،
!>                                   FACTOPTNB هو الحجب المستخدم بواسطة خوارزمية QR أو LQ
!>                                   عادةً ما يكون FACTOPTNB=128 خياراً جيداً
!>                                   NTHREADS هو عدد الخيوط المستخدمة عند
!>                                   تم تمكين تجميع openMP، وإلا =1.
!>          إذا كان JOBZ = 'V' و N > 1، يجب الاستعلام عن LWORK. غير متوفر بعد
!>
!>          إذا كان LWORK = -1، فسيتم افتراض استعلام مساحة العمل؛ الروتين
!>          يحسب فقط الحجم الأمثل للمصفوفة WORK، ويعيد
!>          هذه القيمة كالإدخال الأول للمصفوفة WORK، ولا يتم إصدار رسالة خطأ
!>          متعلقة بـ LWORK بواسطة XERBLA.
!>

INFO

!>          INFO هو INTEGER
!>          = 0:  الخروج بنجاح
!>          < 0:  إذا كان INFO = -i، كان المعامل i غير قانوني
!>          > 0:  إذا كان INFO = i، فشلت الخوارزمية في التقارب؛ i
!>                العناصر غير القطرية لشكل ثلاثي الأقطار وسيط
!>                لم تتقارب إلى الصفر.
!>
