subroutine ssysv_aa_2stage (character uplo, integer n, integer nrhs, real, dimension( lda, * ) a, integer lda, real, dimension( * ) tb, integer ltb, integer, dimension( * ) ipiv, integer, dimension( * ) ipiv2, real, dimension( ldb, * ) b, integer ldb, real, dimension( * ) work, integer lwork, integer info)

SSYSV_AA_2STAGE يحسب الحل لنظام المعادلات الخطية A * X = B لمصفوفات SY

الغرض:

!>
!> SSYSV_AA_2STAGE يحسب الحل لنظام حقيقي من
!> المعادلات الخطية
!>    A * X = B,
!> حيث A مصفوفة متناظرة من البعد N×N و X و B مصفوفتان من البعد N×NRHS.
!>
!> تستخدم خوارزمية Aasen ذات المرحلتين لتحليل A إلى
!>    A = U**T * T * U,  إذا كان UPLO = 'U', أو
!>    A = L * T * L**T,  إذا كان UPLO = 'L',
!> حيث U (أو L) هو حاصل ضرب مصفوفة تبديل مصفوفات ومصفوفة مثلثية عليا (سفلى) وحيدة،
!> و T متناظرة ونطاقية. ثم يتم تحليل T بالـ LU مع الارتكاز الجزئي.
!> بعد ذلك، يستخدم الشكل المحلل لـ A لحل نظام المعادلات A * X = B.
!>
!> هذا هو الإصدار المجمع من الخوارزمية، والذي يستدعي BLAS المستوى 3.
!>

المعلمات

UPLO 
!>          UPLO هو CHARACTER*1
!>          = 'U':  يتم تخزين المثلث العلوي من A؛
!>          = 'L':  يتم تخزين المثلث السفلي من A.
!>

N

!>          N هو INTEGER
!>          رتبة المصفوفة A.  N >= 0.
!>

NRHS

!>          NRHS هو INTEGER
!>          عدد الأطراف اليمنى، أي عدد أعمدة
!>          المصفوفة B.  NRHS >= 0.
!>

A

!>          A هو مصفوفة REAL، ذات الأبعاد (LDA,N)
!>          عند الإدخال، المصفوفة المتناظرة A.  إذا كان UPLO = 'U'، فإن الجزء
!>          المثلثي العلوي من A الذي أبعاده N×N يحتوي على الجزء
!>          المثلثي العلوي للمصفوفة A، ولا يتم الإشارة إلى الجزء السفلي
!>          المثلثي الصارم لـ A.  إذا كان UPLO = 'L'، فإن الجزء
!>          المثلثي السفلي من A الذي أبعاده N×N يحتوي على الجزء
!>          المثلثي السفلي للمصفوفة A، ولا يتم الإشارة إلى الجزء العلوي
!>          المثلثي الصارم لـ A.
!>
!>          عند الخروج، يتم تخزين L تحت (أو فوق) كتل القطر السفلي،
!>          عندما يكون UPLO = 'L' (أو 'U').
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرائد للمصفوفة A.  LDA >= max(1,N).
!>

TB

!>          TB هو مصفوفة REAL، ذات الأبعاد (MAX(1,LTB))
!>          عند الخروج، تفاصيل تحليل LU للمصفوفة النطاقية.
!>

LTB

!>          LTB هو INTEGER
!>          حجم المصفوفة TB. LTB >= MAX(1,4*N)، يستخدم داخليًا
!>          لاختيار NB بحيث يكون LTB >= (3*NB+1)*N.
!>
!>          إذا كان LTB = -1، فيفترض استعلام مساحة العمل؛
!>          الوظيفة الفرعية تحسب فقط الحجم الأمثل لـ LTB،
!>          وتعيد هذا القيمة كإدخال أول في TB،
!>          ولا تصدر رسالة خطأ متعلقة بـ LTB بواسطة XERBLA.
!>

IPIV

!>          IPIV هو مصفوفة INTEGER، ذات الأبعاد (N)
!>          عند الخروج، تحتوي على تفاصيل التبادلات، أي،
!>          تم تبديل الصف والعمود k من A مع
!>          الصف والعمود IPIV(k).
!>

IPIV2

!>          IPIV2 هو مصفوفة INTEGER، ذات الأبعاد (N)
!>          عند الخروج، تحتوي على تفاصيل التبادلات، أي،
!>          تم تبديل الصف والعمود k من T مع
!>          الصف والعمود IPIV(k).
!>

B

!>          B هو مصفوفة REAL، ذات الأبعاد (LDB,NRHS)
!>          عند الإدخال، المصفوفة الطرفية اليمنى B.
!>          عند الخروج، مصفوفة الحل X.
!>

LDB

!>          LDB هو INTEGER
!>          البعد الرائد للمصفوفة B.  LDB >= max(1,N).
!>

WORK

!>          WORK هي مساحة عمل REAL بحجم (MAX(1,LWORK))
!>          عند الخروج، إذا كان INFO = 0، فإن WORK(1) تعيد LWORK الأمثل.
!>

LWORK

!>          LWORK هو INTEGER
!>          حجم WORK. LWORK >= MAX(1,N)، يستخدم داخليًا
!>          لاختيار NB بحيث يكون LWORK >= N*NB.
!>
!>          إذا كان LWORK = -1، فيفترض استعلام مساحة العمل؛
!>          الوظيفة الفرعية تحسب فقط الحجم الأمثل للمصفوفة WORK،
!>          وتعيد هذا القيمة كإدخال أول في المصفوفة WORK،
!>          ولا تصدر رسالة خطأ متعلقة بـ LWORK بواسطة XERBLA.
!>

INFO

!>          INFO هو INTEGER
!>          = 0:  انتهاء ناجح
!>          < 0:  إذا كان INFO = -i، فإن الوسيط i كان له قيمة غير قانونية.
!>          > 0:  إذا كان INFO = i، فشل تحليل LU النطاقي في العمود i
!>
