subroutine ssprfs (character uplo, integer n, integer nrhs, real, dimension( * ) ap, real, dimension( * ) afp, integer, dimension( * ) ipiv, real, dimension( ldb, * ) b, integer ldb, real, dimension( ldx, * ) x, integer ldx, real, dimension( * ) ferr, real, dimension( * ) berr, real, dimension( * ) work, integer, dimension( * ) iwork, integer info)

SSPRFS

الغرض:

!>
!> SSPRFS يحسّن الحل المحسوب لنظام معادلات خطية
!> عندما تكون مصفوفة المعاملات متماثلة غير محددة
!> ومعبأة، ويوفر حدود خطأ وتقديرات خطأ خلفية
!> للحل.
!>

المعلمات

UPLO 
!>          UPLO is CHARACTER*1
!>          = 'U':  يتم تخزين المثلث العلوي لـ A؛
!>          = 'L':  يتم تخزين المثلث السفلي لـ A.
!>

N

!>          N is INTEGER
!>          رتبة المصفوفة A.  N >= 0.
!>

NRHS

!>          NRHS is INTEGER
!>          عدد الأطراف اليمنى، أي عدد الأعمدة
!>          للمصفوفتين B و X.  NRHS >= 0.
!>

AP

!>          AP is REAL array, dimension (N*(N+1)/2)
!>          المثلث العلوي أو السفلي للمصفوفة المتماثلة A، معبأ
!>          عمودياً في مصفوفة خطية.  يتم تخزين العمود j للمصفوفة A
!>          في المصفوفة AP كما يلي:
!>          إذا كان UPLO = 'U'، فإن AP(i + (j-1)*j/2) = A(i,j) لـ 1<=i<=j؛
!>          إذا كان UPLO = 'L'، فإن AP(i + (j-1)*(2*n-j)/2) = A(i,j) لـ j<=i<=n.
!>

AFP

!>          AFP is REAL array, dimension (N*(N+1)/2)
!>          الشكل المقلوب للمصفوفة A.  تحتوي AFP على المصفوفة
!>          القطرية الكتلية D والعوامل المستخدمة للحصول على
!>          المثلث U أو L من تحلل A = U*D*U**T أو
!>          A = L*D*L**T كما تم حسابه بواسطة SSPTRF، مخزنة كمصفوفة
!>          مثلثة معبأة.
!>

IPIV

!>          IPIV is INTEGER array, dimension (N)
!>          تفاصيل التبادلات والبنية الكتلية لـ D
!>          كما تم تحديدها بواسطة SSPTRF.
!>

B

!>          B is REAL array, dimension (LDB,NRHS)
!>          مصفوفة الطرف الأيمن B.
!>

LDB

!>          LDB is INTEGER
!>          البعد الرائد للمصفوفة B.  LDB >= max(1,N).
!>

X

!>          X is REAL array, dimension (LDX,NRHS)
!>          عند الإدخال، مصفوفة الحل X، كما تم حسابها بواسطة SSPTRS.
!>          عند الخروج، مصفوفة الحل المحسنة X.
!>

LDX

!>          LDX is INTEGER
!>          البعد الرائد للمصفوفة X.  LDX >= max(1,N).
!>

FERR

!>          FERR is REAL array, dimension (NRHS)
!>          حد الخطأ الأمامي المقدر لكل متجه حل
!>          X(j) (العمود j للمصفوفة الحل X).
!>          إذا كان XTRUE هو الحل الحقيقي المقابل لـ X(j)، فإن FERR(j)
!>          هو حد أعلى مقدر للقيمة المطلقة لأكبر
!>          عنصر في (X(j) - XTRUE) مقسومًا على القيمة المطلقة لـ
!>          أكبر عنصر في X(j).  التقدير موثوق به مثل
!>          التقدير لـ RCOND، وهو دائمًا تقريبًا
!>          مبالغة طفيفة في الخطأ الحقيقي.
!>

BERR

!>          BERR is REAL array, dimension (NRHS)
!>          الخطأ الخلفي النسبي المكوني لكل متجه حل
!>          X(j) (أي، أصغر تغيير نسبي في
!>          أي عنصر في A أو B يجعل X(j) حلاً دقيقًا).
!>

WORK

!>          WORK is REAL array, dimension (3*N)
!>

IWORK

!>          IWORK is INTEGER array, dimension (N)
!>

INFO

!>          INFO is INTEGER
!>          = 0:  خروج ناجح
!>          < 0:  إذا كان INFO = -i، فإن المعلمة i كانت لها قيمة غير قانونية
!>
