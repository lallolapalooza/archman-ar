subroutine ssygv (integer itype, character jobz, character uplo, integer n, real, dimension( lda, * ) a, integer lda, real, dimension( ldb, * ) b, integer ldb, real, dimension( * ) w, real, dimension( * ) work, integer lwork, integer info)

SSYGV

الغرض:

!>
!> SSYGV يحسب جميع القيم الذاتية، اختيارياً، المتجهات الذاتية
!> لمشكلة القياسية المتماثلة المحددة الحقيقية المعممة، من الشكل
!> A*x=(lambda)*B*x,  A*Bx=(lambda)*x,  أو B*A*x=(lambda)*x.
!> هنا يُفترض أن A و B متماثلتان وأن B أيضًا
!> موجبة محددة.
!>

المعاملات

ITYPE 
!>          ITYPE هو عدد صحيح (INTEGER)
!>          يحدد نوع المشكلة المطلوب حلها:
!>          = 1:  A*x = (lambda)*B*x
!>          = 2:  A*B*x = (lambda)*x
!>          = 3:  B*A*x = (lambda)*x
!>

JOBZ

!>          JOBZ هو حرف واحد (CHARACTER*1)
!>          = 'N':  حساب القيم الذاتية فقط؛
!>          = 'V':  حساب القيم الذاتية والمتجهات الذاتية.
!>

UPLO

!>          UPLO هو حرف واحد (CHARACTER*1)
!>          = 'U':  المثلثات العلوية لـ A و B مخزنة؛
!>          = 'L':  المثلثات السفلية لـ A و B مخزنة.
!>

N

!>          N هو عدد صحيح (INTEGER)
!>          رتبة المصفوفتين A و B.  N >= 0.
!>

A

!>          A هي مصفوفة حقيقية (REAL array)، البعد (LDA, N)
!>          عند الإدخال، المصفوفة المتماثلة A.  إذا كان UPLO = 'U'، فإن
!>          الجزء المثلث العلوي العلوي N-by-N من A يحتوي على
!>          الجزء المثلث العلوي للمصفوفة A.  إذا كان UPLO = 'L'، فإن
!>          الجزء المثلث السفلي السفلي N-by-N من A يحتوي على
!>          الجزء المثلث السفلي للمصفوفة A.
!>
!>          عند الخروج، إذا كان JOBZ = 'V'، فإذا كان INFO = 0، فإن A تحتوي على
!>          المصفوفة Z للمتجهات الذاتية.  تُطبع المتجهات الذاتية
!>          بالشكل التالي:
!>          إذا كان ITYPE = 1 أو 2، فإن Z**T*B*Z = I؛
!>          إذا كان ITYPE = 3، فإن Z**T*inv(B)*Z = I.
!>          إذا كان JOBZ = 'N'، فعند الخروج، المثلث العلوي (إذا كان UPLO='U')
!>          أو المثلث السفلي (إذا كان UPLO='L') من A، بما في ذلك
!>          القطر، يتم تدميره.
!>

LDA

!>          LDA هو عدد صحيح (INTEGER)
!>          البعد الرائد للمصفوفة A.  LDA >= max(1,N).
!>

B

!>          B هي مصفوفة حقيقية (REAL array)، البعد (LDB, N)
!>          عند الإدخال، المصفوفة المتماثلة الموجبة المحددة B.
!>          إذا كان UPLO = 'U'، فإن الجزء المثلث العلوي العلوي N-by-N من B
!>          يحتوي على الجزء المثلث العلوي للمصفوفة B.
!>          إذا كان UPLO = 'L'، فإن الجزء المثلث السفلي السفلي N-by-N من B
!>          يحتوي على الجزء المثلث السفلي للمصفوفة B.
!>
!>          عند الخروج، إذا كان INFO <= N، فإن الجزء من B الذي يحتوي على المصفوفة
!>          يُستبدل بالعامل المثلثي U أو L من تحليل Cholesky
!>          B = U**T*U أو B = L*L**T.
!>

LDB

!>          LDB هو عدد صحيح (INTEGER)
!>          البعد الرائد للمصفوفة B.  LDB >= max(1,N).
!>

W

!>          W هي مصفوفة حقيقية (REAL array)، البعد (N)
!>          إذا كان INFO = 0، فإن القيم الذاتية بترتيب تصاعدي.
!>

WORK

!>          WORK هي مصفوفة حقيقية (REAL array)، البعد (MAX(1,LWORK))
!>          عند الخروج، إذا كان INFO = 0، فإن WORK(1) يُرجع LWORK الأمثل.
!>

LWORK

!>          LWORK هو عدد صحيح (INTEGER)
!>          طول المصفوفة WORK.  LWORK >= max(1,3*N-1).
!>          للكفاءة المثلى، LWORK >= (NB+2)*N,
!>          حيث NB هو حجم الكتلة لـ SSYTRD الذي تم إرجاعه بواسطة ILAENV.
!>
!>          إذا كان LWORK = -1، فيُفترض استعلام مساحة العمل؛ الروتين
!>          يحسب فقط الحجم الأمثل للمصفوفة WORK، ويُرجع
!>          هذه القيمة كالإدخال الأول للمصفوفة WORK، ولا يتم إصدار أي رسالة خطأ
!>          متعلقة بـ LWORK بواسطة XERBLA.
!>

INFO

!>          INFO هو عدد صحيح (INTEGER)
!>          = 0:  انتهاء ناجح
!>          < 0:  إذا كان INFO = -i، فإن المعامل i كان له قيمة غير قانونية
!>          > 0:  SPOTRF أو SSYEV أرجعت رمز خطأ:
!>             <= N:  إذا كان INFO = i، فشلت SSYEV في التقارب؛
!>                    i عناصر خارج القطر لمصفوفة ثلاثية وسيطة
!>                    لم تتقارب إلى الصفر؛
!>             > N:   إذا كان INFO = N + i، لـ 1 <= i <= N، فإن
!>                    المُحدد الرئيسي الرائد من الرتبة i لـ B ليس موجبًا.
!>                    لم يكن من الممكن إكمال تحليل B
!>                    ولم يتم حساب أي قيم ذاتية أو متجهات ذاتية.
!>
