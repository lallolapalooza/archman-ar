subroutine sspevx (character jobz, character range, character uplo, integer n, real, dimension( * ) ap, real vl, real vu, integer il, integer iu, real abstol, integer m, real, dimension( * ) w, real, dimension( ldz, * ) z, integer ldz, real, dimension( * ) work, integer, dimension( * ) iwork, integer, dimension( * ) ifail, integer info)

SSPEVX تحسب القيم الذاتية، وبشكل اختياري، المتجهات الذاتية اليسرى و/أو اليمنى للمصفوفات الأخرى

الغرض:

!>
!> SSPEVX يحسب القيم الذاتية المحددة، وبشكل اختياري، المتجهات الذاتية
!> لمصفوفة متماثلة حقيقية A في تخزين مكدس. يمكن تحديد القيم/المتجهات الذاتية
!> عن طريق تحديد نطاق قيم أو نطاق
!> مؤشرات للقيم الذاتية المطلوبة.
!>

المعلمات

JOBZ 
!>          JOBZ هو CHARACTER*1
!>          = 'N':  حساب القيم الذاتية فقط؛
!>          = 'V':  حساب القيم الذاتية والمتجهات الذاتية.
!>

RANGE

!>          RANGE هو CHARACTER*1
!>          = 'A': سيتم العثور على جميع القيم الذاتية؛
!>          = 'V': سيتم العثور على جميع القيم الذاتية في الفترة شبه المفتوحة (VL,VU]
!>                 ؛
!>          = 'I': سيتم العثور على القيم الذاتية من IL إلى IU.
!>

UPLO

!>          UPLO هو CHARACTER*1
!>          = 'U':  يتم تخزين المثلث العلوي لـ A؛
!>          = 'L':  يتم تخزين المثلث السفلي لـ A.
!>

N

!>          N هو INTEGER
!>          ترتيب المصفوفة A.  N >= 0.
!>

AP

!>          AP هو مصفوفة REAL، ذات أبعاد (N*(N+1)/2)
!>          عند الإدخال، المثلث العلوي أو السفلي للمصفوفة المتماثلة
!>          A، مكدس عموديًا في مصفوفة خطية. يتم تخزين العمود j للمصفوفة A
!>          في المصفوفة AP كما يلي:
!>          إذا كان UPLO = 'U'، فإن AP(i + (j-1)*j/2) = A(i,j) لـ 1<=i<=j؛
!>          إذا كان UPLO = 'L'، فإن AP(i + (j-1)*(2*n-j)/2) = A(i,j) لـ j<=i<=n.
!>
!>          عند الخروج، سيتم الكتابة فوق AP بالقيم التي تم إنشاؤها أثناء
!>          الاختزال إلى شكل ثلاثي الأقطار. إذا كان UPLO = 'U'، فإن القطر
!>          والقطر فوق القطري الأول للمصفوفة ثلاثية الأقطار T سيكتبان فوق
!>          العناصر المقابلة لـ A، وإذا كان UPLO = 'L'، فإن القطر
!>          والقطر تحت القطري الأول لـ T سيكتبان فوق
!>          العناصر المقابلة لـ A.
!>

VL

!>          VL هو REAL
!>          إذا كانت RANGE='V'، فإن الحد الأدنى للفترة للبحث عن القيم الذاتية.
!>          VL < VU.
!>          لا يتم الرجوع إليه إذا كانت RANGE = 'A' أو 'I'.
!>

VU

!>          VU هو REAL
!>          إذا كانت RANGE='V'، فإن الحد الأعلى للفترة للبحث عن القيم الذاتية.
!>          VL < VU.
!>          لا يتم الرجوع إليه إذا كانت RANGE = 'A' أو 'I'.
!>

IL

!>          IL هو INTEGER
!>          إذا كانت RANGE='I'، فإن مؤشر أصغر قيمة ذاتية سيتم إرجاعها.
!>          1 <= IL <= IU <= N، إذا كان N > 0؛ IL = 1 و IU = 0 إذا كان N = 0.
!>          لا يتم الرجوع إليه إذا كانت RANGE = 'A' أو 'V'.
!>

IU

!>          IU هو INTEGER
!>          إذا كانت RANGE='I'، فإن مؤشر أكبر قيمة ذاتية سيتم إرجاعها.
!>          1 <= IL <= IU <= N، إذا كان N > 0؛ IL = 1 و IU = 0 إذا كان N = 0.
!>          لا يتم الرجوع إليه إذا كانت RANGE = 'A' أو 'V'.
!>

ABSTOL

!>          ABSTOL هو REAL
!>          التفاوت المطلق للخطأ في القيم الذاتية.
!>          يتم قبول قيمة ذاتية تقريبية على أنها متقاربة
!>          عندما يتم تحديد أنها تقع في فترة [a,b]
!>          بعرض يساوي أو أقل من
!>
!>                  ABSTOL + EPS *   max( |a|,|b| ) ,
!>
!>          حيث EPS هي دقة الآلة. إذا كان ABSTOL أقل من أو يساوي صفرًا،
!>          فسيتم استخدام EPS*|T| بدلاً منه،
!>          حيث |T| هو معيار 1 للمصفوفة ثلاثية الأقطار التي تم الحصول عليها
!>          عن طريق اختزال AP إلى شكل ثلاثي الأقطار.
!>
!>          سيتم حساب القيم الذاتية بدقة أكبر عندما يتم تعيين ABSTOL
!>          إلى ضعف عتبة الانخفاض 2*SLAMCH('S')، وليس صفرًا.
!>          إذا عادت هذه الوظيفة بمعامل INFO>0، مما يشير إلى أن بعض
!>          المتجهات الذاتية لم تتقارب، حاول تعيين ABSTOL إلى
!>          2*SLAMCH('S').
!>
!>          راجع  بواسطة Demmel و
!>          Kahan، مذكرة عمل LAPACK #3.
!>

M

!>          M هو INTEGER
!>          العدد الإجمالي للقيم الذاتية التي تم العثور عليها.  0 <= M <= N.
!>          إذا كانت RANGE = 'A'، فإن M = N، وإذا كانت RANGE = 'I'، فإن M = IU-IL+1.
!>

W

!>          W هي مصفوفة REAL، ذات أبعاد (N)
!>          إذا كان INFO = 0، فإن القيم الذاتية المحددة بترتيب تصاعدي.
!>

Z

!>          Z هي مصفوفة REAL، ذات أبعاد (LDZ, max(1,M))
!>          إذا كان JOBZ = 'V'، فعندما يكون INFO = 0، فإن الأعمدة M الأولى من Z
!>          تحتوي على المتجهات الذاتية المتعامدة للمصفوفة A
!>          المقابلة للقيم الذاتية المحددة، مع العمود i
!>          من Z يحمل المتجه الذاتي المرتبط بـ W(i).
!>          إذا فشل المتجه الذاتي في التقارب، فإن هذا العمود من Z
!>          يحتوي على أحدث تقريب للمتجه الذاتي، وسيتم إرجاع مؤشر
!>          المتجه الذاتي في IFAIL.
!>          إذا كان JOBZ = 'N'، فلا يتم الرجوع إلى Z.
!>          ملاحظة: يجب على المستخدم التأكد من توفير عمود واحد على الأقل (max(1,M))
!>          في المصفوفة Z؛ إذا كانت RANGE = 'V'، فإن القيمة الدقيقة لـ M
!>          غير معروفة مسبقًا ويجب استخدام حد أعلى.
!>

LDZ

!>          LDZ هو INTEGER
!>          البعد الرائد للمصفوفة Z.  LDZ >= 1، وإذا كان
!>          JOBZ = 'V'، فإن LDZ >= max(1,N).
!>

WORK

!>          WORK هي مصفوفة REAL، ذات أبعاد (8*N)
!>

IWORK

!>          IWORK هي مصفوفة INTEGER، ذات أبعاد (5*N)
!>

IFAIL

!>          IFAIL هي مصفوفة INTEGER، ذات أبعاد (N)
!>          إذا كان JOBZ = 'V'، فعندما يكون INFO = 0، فإن العناصر M الأولى من
!>          IFAIL هي صفر. إذا كان INFO > 0، فإن IFAIL يحتوي على
!>          مؤشرات المتجهات الذاتية التي فشلت في التقارب.
!>          إذا كان JOBZ = 'N'، فلا يتم الرجوع إلى IFAIL.
!>

INFO

!>          INFO هو INTEGER
!>          = 0:  اكتمال ناجح
!>          < 0:  إذا كان INFO = -i، فإن المعلمة i كانت ذات قيمة غير قانونية
!>          > 0:  إذا كان INFO = i، فقد فشلت i متجهات ذاتية في التقارب.
!>                تم تخزين مؤشراتها في المصفوفة IFAIL.
!>
