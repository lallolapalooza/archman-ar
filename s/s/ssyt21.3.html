subroutine ssyt21 (integer itype, character uplo, integer n, integer kband, real, dimension( lda, * ) a, integer lda, real, dimension( * ) d, real, dimension( * ) e, real, dimension( ldu, * ) u, integer ldu, real, dimension( ldv, * ) v, integer ldv, real, dimension( * ) tau, real, dimension( * ) work, real, dimension( 2 ) result)

SSYT21

الغرض:

!>
!> تتحقق SSYT21 بشكل عام من تفكيك بالشكل
!>
!>    A = U S U**T
!>
!> حيث **T تعني تبديل، A متناظرة، U متعامدة، و S قطرية (إذا كان KBAND=0)
!> أو متناظرة ثلاثية القطر (إذا كان KBAND=1).
!>
!> إذا كان ITYPE=1، فإن U ممثلة كمصفوفة كثيفة؛ بخلاف ذلك، يتم التعبير عن U
!> كناتج تحويلات هاوسهولدر، التي يتم تخزين متجهات 
!> فيها وثوابتها القياسية في .
!> سنستخدم الحرف  للإشارة إلى ناتج تحويلات هاوسهولدر
!> (التي يجب أن تساوي U).
!>
!> على وجه التحديد، إذا كان ITYPE=1، فإن:
!>
!>    RESULT(1) = | A - U S U**T | / ( |A| n ulp ) و
!>    RESULT(2) = | I - U U**T | / ( n ulp )
!>
!> إذا كان ITYPE=2، فإن:
!>
!>    RESULT(1) = | A - V S V**T | / ( |A| n ulp )
!>
!> إذا كان ITYPE=3، فإن:
!>
!>    RESULT(1) = | I - V U**T | / ( n ulp )
!>
!> بالنسبة لـ ITYPE > 1، يتم التعبير عن التحويل U كناتج 
!> V = H(1)...H(n-2)، حيث H(j) = I - tau(j) v(j) v(j)**T وكل متجه v(j)
!> له عناصره الأولى j تساوي 0 والعناصر المتبقية n-j مخزنة في V(j+1:n,j).
!>

المعلمات

ITYPE 
!>          ITYPE هو عدد صحيح (INTEGER)
!>          يحدد نوع الاختبارات التي سيتم إجراؤها.
!>          1: يتم التعبير عن U كمصفوفة متعامدة كثيفة:
!>             RESULT(1) = | A - U S U**T | / ( |A| n ulp ) و
!>             RESULT(2) = | I - U U**T | / ( n ulp )
!>
!>          2: يتم التعبير عن U كناتج V لتحويلات هاوسهولدر:
!>             RESULT(1) = | A - V S V**T | / ( |A| n ulp )
!>
!>          3: يتم التعبير عن U كمصفوفة متعامدة كثيفة وكـ
!>             كناتج لتحويلات هاوسهولدر:
!>             RESULT(1) = | I - V U**T | / ( n ulp )
!>

UPLO

!>          UPLO هو حرف (CHARACTER)
!>          إذا كان UPLO='U'، فسيتم استخدام المثلث العلوي لـ A و V
!>          ولن يتم الرجوع إلى المثلث السفلي (بشكل صارم).
!>          إذا كان UPLO='L'، فسيتم استخدام المثلث السفلي لـ A و V
!>          ولن يتم الرجوع إلى المثلث العلوي (بشكل صارم).
!>

N

!>          N هو عدد صحيح (INTEGER)
!>          حجم المصفوفة. إذا كان صفرًا، فإن SSYT21 لا تفعل شيئًا.
!>          يجب أن يكون أكبر من أو يساوي صفر.
!>

KBAND

!>          KBAND هو عدد صحيح (INTEGER)
!>          عرض النطاق الترددي للمصفوفة. قد يكون صفرًا أو واحدًا فقط.
!>          إذا كان صفرًا، فإن S تكون قطرية، ولا يتم الرجوع إلى E.
!>          إذا كان واحدًا، فإن S تكون ثلاثية الأقطار متناظرة.
!>

A

!>          A هي مصفوفة حقيقية (REAL array)، أبعادها (LDA, N)
!>          المصفوفة الأصلية (غير المفككة). يفترض أنها متناظرة،
!>          وسيتم الرجوع فقط إلى الجزء العلوي (UPLO='U') أو الجزء السفلي
!>          (UPLO='L') فقط.
!>

LDA

!>          LDA هو عدد صحيح (INTEGER)
!>          البعد الرئيسي لـ A. يجب أن يكون على الأقل 1
!>          وعلى الأقل N.
!>

D

!>          D هي مصفوفة حقيقية (REAL array)، أبعادها (N)
!>          القطر الرئيسي للمصفوفة (ثلاثية الأقطار المتناظرة).
!>

E

!>          E هي مصفوفة حقيقية (REAL array)، أبعادها (N-1)
!>          العناصر خارج القطر الرئيسي للمصفوفة (ثلاثية الأقطار المتناظرة).
!>          E(1) هو العنصر (1,2) و (2,1)، E(2) هو العنصر (2,3) و
!>          (3,2)، إلخ.
!>          لا يتم الرجوع إليه إذا كان KBAND=0.
!>

U

!>          U هي مصفوفة حقيقية (REAL array)، أبعادها (LDU, N)
!>          إذا كان ITYPE=1 أو 3، فإنها تحتوي على المصفوفة المتعامدة
!>          في التفكيك، ممثلة كمصفوفة كثيفة. إذا كان ITYPE=2،
!>          فلن يتم الرجوع إليها.
!>

LDU

!>          LDU هو عدد صحيح (INTEGER)
!>          البعد الرئيسي لـ U. يجب أن يكون LDU على الأقل N
!>          وعلى الأقل 1.
!>

V

!>          V هي مصفوفة حقيقية (REAL array)، أبعادها (LDV, N)
!>          إذا كان ITYPE=2 أو 3، فإن أعمدة هذه المصفوفة تحتوي على
!>          متجهات هاوسهولدر المستخدمة لوصف المصفوفة المتعامدة
!>          في التفكيك. إذا كان UPLO='L'، فسيتم تخزين المتجهات
!>          في المثلث السفلي، وإذا كان UPLO='U'، فسيتم تخزينها
!>          في المثلث العلوي.
!>          *ملاحظة* إذا كان ITYPE=2 أو 3، فسيتم تعديل V واستعادتها.
!>          سيتم تعيين تحت القطر الرئيسي (إذا كان UPLO='L') أو فوق القطر
!>          الرئيسي (إذا كان UPLO='U') إلى واحد، ثم سيتم إعادة تعيينه لاحقًا
!>          إلى قيمته الأصلية، خلال مسار الحساب.
!>          إذا كان ITYPE=1، فلن يتم الرجوع إليها أو تعديلها.
!>

LDV

!>          LDV هو عدد صحيح (INTEGER)
!>          البعد الرئيسي لـ V. يجب أن يكون LDV على الأقل N
!>          وعلى الأقل 1.
!>

TAU

!>          TAU هي مصفوفة حقيقية (REAL array)، أبعادها (N)
!>          إذا كان ITYPE >= 2، فإن TAU(j) هو العامل القياسي لـ
!>          v(j) v(j)**T في تحويل هاوسهولدر H(j) لناتج
!>          U = H(1)...H(n-2)
!>          إذا كان ITYPE < 2، فلن يتم الرجوع إلى TAU.
!>

WORK

!>          WORK هي مصفوفة حقيقية (REAL array)، أبعادها (2*N**2)
!>

RESULT

!>          RESULT هي مصفوفة حقيقية (REAL array)، أبعادها (2)
!>          القيم المحسوبة بواسطة الاختبارين الموصوفين أعلاه.
!>          القيم محدودة حاليًا بـ 1/ulp، لتجنب تجاوز السعة.
!>          يتم تعديل RESULT(1) دائمًا. يتم تعديل RESULT(2) فقط
!>          إذا كان ITYPE=1.
!>
