subroutine ssbgv (character jobz, character uplo, integer n, integer ka, integer kb, real, dimension( ldab, * ) ab, integer ldab, real, dimension( ldbb, * ) bb, integer ldbb, real, dimension( * ) w, real, dimension( ldz, * ) z, integer ldz, real, dimension( * ) work, integer info)

SSBGV

الغرض:

!>
!> SSBGV يحسب جميع القيم الذاتية، وبشكل اختياري، المتجهات الذاتية
!> لمشكلة قيمة ذاتية مصفوفة متماثلة محددة متعامدة،
!> بالشكل A*x=(lambda)*B*x. هنا يُفترض أن A و B متماثلة
!> ومزدوجة، و B موجبة بشكل محدد.
!>

المعاملات

JOBZ 
!>          JOBZ هو CHARACTER*1
!>          = 'N':  حساب القيم الذاتية فقط؛
!>          = 'V':  حساب القيم الذاتية والمتجهات الذاتية.
!>

UPLO

!>          UPLO هو CHARACTER*1
!>          = 'U':  المثلثات العلوية لـ A و B مخزنة؛
!>          = 'L':  المثلثات السفلية لـ A و B مخزنة.
!>

N

!>          N هو INTEGER
!>          رتبة المصفوفتين A و B.  N >= 0.
!>

KA

!>          KA هو INTEGER
!>          عدد الأقطار الفائقة للمصفوفة A إذا كانت UPLO = 'U'،
!>          أو عدد الأقطار الفرعية إذا كانت UPLO = 'L'. KA >= 0.
!>

KB

!>          KB هو INTEGER
!>          عدد الأقطار الفائقة للمصفوفة B إذا كانت UPLO = 'U'،
!>          أو عدد الأقطار الفرعية إذا كانت UPLO = 'L'. KB >= 0.
!>

AB

!>          AB هي مصفوفة REAL، بأبعاد (LDAB, N)
!>          عند الإدخال، المثلث العلوي أو السفلي للمصفوفة المزدوجة المتماثلة
!>          A، مخزنة في أول ka+1 صف من المصفوفة.  العمود
!>          j للمصفوفة A مخزن في العمود j للمصفوفة AB
!>          كالتالي:
!>          إذا كانت UPLO = 'U'، فإن AB(ka+1+i-j,j) = A(i,j) لـ max(1,j-ka)<=i<=j؛
!>          إذا كانت UPLO = 'L'، فإن AB(1+i-j,j) = A(i,j) لـ j<=i<=min(n,j+ka).
!>
!>          عند الخروج، يتم تدمير محتويات AB.
!>

LDAB

!>          LDAB هو INTEGER
!>          البعد الرائد للمصفوفة AB.  LDAB >= KA+1.
!>

BB

!>          BB هي مصفوفة REAL، بأبعاد (LDBB, N)
!>          عند الإدخال، المثلث العلوي أو السفلي للمصفوفة المزدوجة المتماثلة
!>          B، مخزنة في أول kb+1 صف من المصفوفة.  العمود
!>          j للمصفوفة B مخزن في العمود j للمصفوفة BB
!>          كالتالي:
!>          إذا كانت UPLO = 'U'، فإن BB(kb+1+i-j,j) = B(i,j) لـ max(1,j-kb)<=i<=j؛
!>          إذا كانت UPLO = 'L'، فإن BB(1+i-j,j) = B(i,j) لـ j<=i<=min(n,j+kb).
!>
!>          عند الخروج، العامل S من التحليل Cholesky المنفصل
!>          B = S**T*S، كما تم إرجاعه بواسطة SPBSTF.
!>

LDBB

!>          LDBB هو INTEGER
!>          البعد الرائد للمصفوفة BB.  LDBB >= KB+1.
!>

W

!>          W هي مصفوفة REAL، بأبعاد (N)
!>          إذا كان INFO = 0، فإن القيم الذاتية بترتيب تصاعدي.
!>

Z

!>          Z هي مصفوفة REAL، بأبعاد (LDZ, N)
!>          إذا كانت JOBZ = 'V'، فعندما يكون INFO = 0، تحتوي Z على المصفوفة Z للمتجهات
!>          الذاتية، حيث يحمل العمود i من Z المتجه
!>          الذاتي المرتبط بـ W(i). يتم تطبيع المتجهات الذاتية
!>          بحيث Z**T*B*Z = I.
!>          إذا كانت JOBZ = 'N'، فلا يتم استخدام Z.
!>

LDZ

!>          LDZ هو INTEGER
!>          البعد الرائد للمصفوفة Z.  LDZ >= 1، وإذا
!>          كانت JOBZ = 'V'، فإن LDZ >= N.
!>

WORK

!>          WORK هي مصفوفة REAL، بأبعاد (3*N)
!>

INFO

!>          INFO هو INTEGER
!>          = 0:  انتهاء ناجح
!>          < 0:  إذا كان INFO = -i، فإن المعامل i كان له قيمة غير قانونية
!>          > 0:  إذا كان INFO = i، و i هو:
!>             <= N:  فشلت الخوارزمية في التقارب:
!>                    i عنصر غير قطري من صيغة ثلاثية
!>                    وسيطة لم تتقارب إلى الصفر؛
!>             > N:   إذا كان INFO = N + i، لـ 1 <= i <= N، فإن SPBSTF
!>                    أرجع INFO = i: B ليست موجبة بشكل محدد.
!>                    لم يتمكن من إكمال تحليل B
!>                    ولم يتم حساب أي قيم ذاتية أو متجهات ذاتية.
!>
