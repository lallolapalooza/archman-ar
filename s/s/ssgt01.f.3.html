subroutine ssgt01 (integer itype, character uplo, integer n, integer m, real, dimension( lda, * ) a, integer lda, real, dimension( ldb, * ) b, integer ldb, real, dimension( ldz, * ) z, integer ldz, real, dimension( * ) d, real, dimension( * ) work, real, dimension( * ) result)

SSGT01

الغرض:

!>
!> SSGT01 يتحقق من تفكيك للشكل
!>
!>    A Z   =  B Z D أو
!>    A B Z =  Z D أو
!>    B A Z =  Z D
!>
!> حيث A مصفوفة متناظرة، B مصفوفة متناظرة موجبة محددة، Z مصفوفة متعامدة، و D مصفوفة قطرية.
!>
!> يتم حساب إحدى نسب الاختبار التالية:
!>
!> ITYPE = 1:  RESULT(1) = | A Z - B Z D | / ( |A| |Z| n ulp )
!>
!> ITYPE = 2:  RESULT(1) = | A B Z - Z D | / ( |A| |Z| n ulp )
!>
!> ITYPE = 3:  RESULT(1) = | B A Z - Z D | / ( |A| |Z| n ulp )
!>

المعلمات

ITYPE 
!>          ITYPE هو عدد صحيح
!>          شكل مسألة القيم الذاتية المعممة المتناظرة.
!>          = 1:  A*z = (lambda)*B*z
!>          = 2:  A*B*z = (lambda)*z
!>          = 3:  B*A*z = (lambda)*z
!>

UPLO

!>          UPLO هو حرف واحد
!>          يحدد ما إذا كان الجزء العلوي أو السفلي المثلثي للمصفوفتين المتناظرتين A و B مخزنًا.
!>          = 'U':  مثلث علوي
!>          = 'L':  مثلث سفلي
!>

N

!>          N هو عدد صحيح
!>          رتبة المصفوفة A.  N >= 0.
!>

M

!>          M هو عدد صحيح
!>          عدد القيم الذاتية التي تم العثور عليها.  0 <= M <= N.
!>

A

!>          A هي مصفوفة حقيقية، بعد (LDA, N)
!>          المصفوفة الأصلية المتناظرة A.
!>

LDA

!>          LDA هو عدد صحيح
!>          البعد الرئيسي للمصفوفة A.  LDA >= max(1,N).
!>

B

!>          B هي مصفوفة حقيقية، بعد (LDB, N)
!>          المصفوفة الأصلية المتناظرة الموجبة المحددة B.
!>

LDB

!>          LDB هو عدد صحيح
!>          البعد الرئيسي للمصفوفة B.  LDB >= max(1,N).
!>

Z

!>          Z هي مصفوفة حقيقية، بعد (LDZ, M)
!>          المتجهات الذاتية المحسوبة لمسألة القيم الذاتية المعممة.
!>

LDZ

!>          LDZ هو عدد صحيح
!>          البعد الرئيسي للمصفوفة Z.  LDZ >= max(1,N).
!>

D

!>          D هي مصفوفة حقيقية، بعد (M)
!>          القيم الذاتية المحسوبة لمسألة القيم الذاتية المعممة.
!>

WORK

!>          WORK هي مصفوفة حقيقية، بعد (N*N)
!>

RESULT

!>          RESULT هي مصفوفة حقيقية، بعد (1)
!>          نسبة الاختبار كما هو موضح أعلاه.
!>
