توثيق الوظيفة/الروتين الفرعي

subroutine ssygst (integer itype, character uplo, integer n, real, dimension( lda, * ) a, integer lda, real, dimension( ldb, * ) b, integer ldb, integer info)

SSYGST

الغرض:

!>
!> SSYGST تقلل مشكلة القيم الذاتية المعممة المتناظرة المحددة الحقيقية
!> إلى الشكل القياسي.
!>
!> إذا كان ITYPE = 1، فالمشكلة هي A*x = lambda*B*x,
!> ويتم الكتابة فوق A بـ inv(U**T)*A*inv(U) أو inv(L)*A*inv(L**T)
!>
!> إذا كان ITYPE = 2 أو 3، فالمشكلة هي A*B*x = lambda*x أو
!> B*A*x = lambda*x، ويتم الكتابة فوق A بـ U*A*U**T أو L**T*A*L.
!>
!> يجب أن يكون B قد تم تحليله سابقًا كـ U**T*U أو L*L**T بواسطة SPOTRF.
!>

المعلمات

ITYPE 
!>          ITYPE هو عدد صحيح
!>          = 1: حساب inv(U**T)*A*inv(U) أو inv(L)*A*inv(L**T);
!>          = 2 أو 3: حساب U*A*U**T أو L**T*A*L.
!>

UPLO

!>          UPLO هو حرف واحد
!>          = 'U':  يتم تخزين المثلث العلوي لـ A وتحليل B كـ
!>                  U**T*U;
!>          = 'L':  يتم تخزين المثلث السفلي لـ A وتحليل B كـ
!>                  L*L**T.
!>

N

!>          N هو عدد صحيح
!>          رتبة المصفوفتين A و B.  N >= 0.
!>

A

!>          A هي مصفوفة حقيقية، أبعادها (LDA,N)
!>          عند الإدخال، المصفوفة المتناظرة A.  إذا كان UPLO = 'U'، فإن
!>          الجزء العلوي المثلثي العلوي N-by-N من A يحتوي على الجزء
!>          المثلثي العلوي للمصفوفة A، ولا يتم الرجوع إلى الجزء المثلثي
!>          السفلي الصارم من A.  إذا كان UPLO = 'L'، فإن الجزء
!>          المثلثي السفلي السفلي N-by-N من A يحتوي على الجزء
!>          المثلثي السفلي للمصفوفة A، ولا يتم الرجوع إلى الجزء المثلثي
!>          العلوي الصارم من A.
!>
!>          عند الخروج، إذا كان INFO = 0، فإن المصفوفة المحولة، مخزنة في
!>          نفس تنسيق A.
!>

LDA

!>          LDA هو عدد صحيح
!>          البعد الرائد للمصفوفة A.  LDA >= max(1,N).
!>

B

!>          B هي مصفوفة حقيقية، أبعادها (LDB,N)
!>          العامل المثلثي من تحليل Cholesky لـ B،
!>          كما تم إرجاعه بواسطة SPOTRF.
!>

LDB

!>          LDB هو عدد صحيح
!>          البعد الرائد للمصفوفة B.  LDB >= max(1,N).
!>

INFO

!>          INFO هو عدد صحيح
!>          = 0:  خروج ناجح
!>          < 0:  إذا كان INFO = -i، فإن المعامل i كان له قيمة غير قانونية
!>
