توثيق الوظيفة/الروتين الفرعي

subroutine sstedc (character compz, integer n, real, dimension( * ) d, real, dimension( * ) e, real, dimension( ldz, * ) z, integer ldz, real, dimension( * ) work, integer lwork, integer, dimension( * ) iwork, integer liwork, integer info)

SSTEDC

الغرض:

!>
!> SSTEDC تحسب جميع القيم الذاتية، واختياريًا، المتجهات الذاتية لمصفوفة ثلاثية قطرية متناظرة باستخدام طريقة الانقسام والغزو.
!> يمكن أيضًا العثور على المتجهات الذاتية لمصفوفة متناظرة كاملة أو نطاقية إذا تم استخدام SSYTRD أو SSPTRD أو SSBTRD لتقليل هذه
!> المصفوفة إلى شكل ثلاثي قطري.
!>
!>

المعلمات:

COMPZ 
!>          COMPZ is CHARACTER*1
!>          = 'N':  حساب القيم الذاتية فقط.
!>          = 'I':  حساب المتجهات الذاتية للمصفوفة ثلاثية القطرية أيضًا.
!>          = 'V':  حساب المتجهات الذاتية للمصفوفة المتناظرة الكثيفة الأصلية
!>                  أيضًا. عند الدخول، تحتوي Z على المصفوفة المتعامدة
!>                  المستخدمة لتقليل المصفوفة الأصلية إلى
!>                  شكل ثلاثي قطري.
!>

N

!>          N is INTEGER
!>          بعد المصفوفة المتناظرة ثلاثية القطرية. N >= 0.
!>

D

!>          D is REAL array, dimension (N)
!>          عند الدخول، العناصر القطرية للمصفوفة ثلاثية القطرية.
!>          عند الخروج، إذا كان INFO = 0، فإن القيم الذاتية بترتيب تصاعدي.
!>

E

!>          E is REAL array, dimension (N-1)
!>          عند الدخول، العناصر تحت القطرية للمصفوفة ثلاثية القطرية.
!>          عند الخروج، تم تدمير E.
!>

Z

!>          Z is REAL array, dimension (LDZ,N)
!>          عند الدخول، إذا كان COMPZ = 'V'، فإن Z تحتوي على
!>          المصفوفة المتعامدة المستخدمة في الاختزال إلى شكل ثلاثي قطري.
!>          عند الخروج، إذا كان INFO = 0، فإن إذا كان COMPZ = 'V'، فإن Z تحتوي على
!>          المتجهات الذاتية المتعامدة العادية للمصفوفة المتناظرة الأصلية،
!>          وإذا كان COMPZ = 'I'، فإن Z تحتوي على المتجهات الذاتية المتعامدة العادية
!>          للمصفوفة ثلاثية القطرية المتناظرة.
!>          إذا كان COMPZ = 'N'، فلا يتم الرجوع إلى Z.
!>

LDZ

!>          LDZ is INTEGER
!>          البعد الرائد للمصفوفة Z. LDZ >= 1.
!>          إذا كانت المتجهات الذاتية مطلوبة، فإن LDZ >= max(1,N).
!>

WORK

!>          WORK is REAL array, dimension (MAX(1,LWORK))
!>          عند الخروج، إذا كان INFO = 0، فإن WORK(1) ترجع LWORK المثلى.
!>

LWORK

!>          LWORK is INTEGER
!>          بعد المصفوفة WORK.
!>          إذا كان COMPZ = 'N' أو N <= 1، فيجب أن يكون LWORK على الأقل 1.
!>          إذا كان COMPZ = 'V' و N > 1، فيجب أن يكون LWORK على الأقل
!>                         ( 1 + 3*N + 2*N*lg N + 4*N**2 )،
!>                         حيث lg( N ) = أصغر عدد صحيح k بحيث
!>                         2**k >= N.
!>          إذا كان COMPZ = 'I' و N > 1، فيجب أن يكون LWORK على الأقل
!>                         ( 1 + 4*N + N**2 ).
!>          لاحظ أنه بالنسبة لـ COMPZ = 'I' أو 'V'، إذا كان N أقل من أو
!!>          يساوي حجم الانقسام الأدنى، عادةً 25، فيجب أن يكون LWORK فقط
!>          max(1,2*(N-1)).
!>
!>          إذا كان LWORK = -1، فيفترض استعلام عن مساحة العمل؛ الروتين
!>          يحسب فقط الحجم الأمثل لمصفوفة WORK، ويرجع
!>          هذه القيمة كالإدخال الأول لمصفوفة WORK، ولا يتم إصدار أي رسالة خطأ
!>          متعلقة بـ LWORK بواسطة XERBLA.
!>

IWORK

!>          IWORK is INTEGER array, dimension (MAX(1,LIWORK))
!>          عند الخروج، إذا كان INFO = 0، فإن IWORK(1) ترجع LIWORK المثلى.
!>

LIWORK

!>          LIWORK is INTEGER
!>          بعد المصفوفة IWORK.
!>          إذا كان COMPZ = 'N' أو N <= 1، فيجب أن يكون LIWORK على الأقل 1.
!>          إذا كان COMPZ = 'V' و N > 1، فيجب أن يكون LIWORK على الأقل
!>                         ( 6 + 6*N + 5*N*lg N ).
!>          إذا كان COMPZ = 'I' و N > 1، فيجب أن يكون LIWORK على الأقل
!>                         ( 3 + 5*N ).
!>          لاحظ أنه بالنسبة لـ COMPZ = 'I' أو 'V'، إذا كان N أقل من أو
!>          يساوي حجم الانقسام الأدنى، عادةً 25، فيجب أن يكون LIWORK
!>          فقط 1.
!>
!>          إذا كان LIWORK = -1، فيفترض استعلام عن مساحة العمل؛ الروتين
!>          يحسب فقط الحجم الأمثل لمصفوفة IWORK، ويرجع
!>          هذه القيمة كالإدخال الأول لمصفوفة IWORK، ولا يتم إصدار أي رسالة خطأ
!>          متعلقة بـ LIWORK بواسطة XERBLA.
!>

INFO

!>          INFO is INTEGER
!>          = 0:  خروج ناجح.
!>          < 0:  إذا كان INFO = -i، فإن المعلمة i-th كانت لها قيمة غير قانونية.
!>          > 0:  فشلت الخوارزمية في حساب قيمة ذاتية أثناء
!>                العمل على المصفوفة الفرعية الواقعة في الصفوف والأعمدة
!>                INFO/(N+1) حتى mod(INFO,N+1).
!>
