subroutine ssyt22 (integer itype, character uplo, integer n, integer m, integer kband, real, dimension( lda, * ) a, integer lda, real, dimension( * ) d, real, dimension( * ) e, real, dimension( ldu, * ) u, integer ldu, real, dimension( ldv, * ) v, integer ldv, real, dimension( * ) tau, real, dimension( * ) work, real, dimension( 2 ) result)

SSYT22

الغرض:

!>
!>      SSYT22  بشكل عام يتحقق من تفكيك بالشكل
!>
!>              A U = U S
!>
!>      حيث A متماثل، أعمدة U متعامدة، و S
!>      قطري (إذا كان KBAND=0) أو شبه ثلاثي متماثل (إذا كان
!>      KBAND=1). إذا كان ITYPE=1، فإن U يتم تمثيله كمصفوفة كثيفة،
!>      وإلا فإن U يتم التعبير عنه كحاصل ضرب تحويلات هاوسهولدر،
!>      التي يتم تخزين متجهات Vectors الخاصة بها في المصفوفة  
!>      وثوابت القياس الخاصة بها في .  سنستخدم الحرف
!>       للإشارة إلى حاصل ضرب تحويلات هاوسهولدر
!>      (الذي يجب أن يساوي U).
!>
!>      على وجه التحديد، إذا كان ITYPE=1، فإن:
!>
!>              RESULT(1) = | U**T A U - S | / ( |A| m ulp ) و
!>              RESULT(2) = | I - U**T U | / ( m ulp )
!> 
!>  ITYPE   INTEGER
!>          يحدد نوع الاختبارات التي سيتم إجراؤها.
!>          1: U معبر عنها كمصفوفة متعامدة كثيفة:
!>             RESULT(1) = | A - U S U**T | / ( |A| n ulp ) و
!>             RESULT(2) = | I - U U**T | / ( n ulp )
!>
!>  UPLO    CHARACTER
!>          إذا كان UPLO='U'، فسيتم استخدام المثلث العلوي لـ A و
!>          المثلث السفلي (تمامًا) لن تتم الإشارة إليه. إذا كان
!>          UPLO='L'، فسيتم استخدام المثلث السفلي لـ A و
!>          المثلث العلوي (تمامًا) لن تتم الإشارة إليه.
!>          لم يتم تعديله.
!>
!>  N       INTEGER
!>          حجم المصفوفة. إذا كان صفرًا، فإن SSYT22 لا تفعل شيئًا.
!>          يجب أن يكون على الأقل صفرًا.
!>          لم يتم تعديله.
!>
!>  M       INTEGER
!>          عدد أعمدة U. إذا كان صفرًا، فإن SSYT22 لا تفعل شيئًا.
!>          يجب أن يكون على الأقل صفرًا.
!>          لم يتم تعديله.
!>
!>  KBAND   INTEGER
!>          عرض النطاق للمصفوفة. يمكن أن يكون صفرًا أو واحدًا فقط.
!>          إذا كان صفرًا، فإن S قطري، و E لا تتم الإشارة إليه. إذا كان
!>          واحدًا، فإن S شبه ثلاثي متماثل.
!>          لم يتم تعديله.
!>
!>  A       REAL array, dimension (LDA , N)
!>          المصفوفة الأصلية (غير المفككة). يُفترض أنها
!>          متماثلة، وسيتم الإشارة فقط إلى الجزء العلوي (UPLO='U') أو الجزء السفلي (UPLO='L')
!>          فقط.
!>          لم يتم تعديله.
!>
!>  LDA     INTEGER
!>          البعد الرائد لـ A. يجب أن يكون على الأقل 1
!>          وعلى الأقل N.
!>          لم يتم تعديله.
!>
!>  D       REAL array, dimension (N)
!>          القطر للمصفوفة (شبه الثلاثية المتماثلة).
!>          لم يتم تعديله.
!>
!>  E       REAL array, dimension (N)
!>          العناصر خارج القطر للمصفوفة (شبه الثلاثية المتماثلة).
!>          يتم تجاهل E(1)، E(2) هو العنصر (1,2) و (2,1)، إلخ.
!>          لا تتم الإشارة إليه إذا كان KBAND=0.
!>          لم يتم تعديله.
!>
!>  U       REAL array, dimension (LDU, N)
!>          إذا كان ITYPE=1 أو 3، فإن هذه تحتوي على المصفوفة المتعامدة في
!>          التفكيك، معبرًا عنها كمصفوفة كثيفة. إذا كان ITYPE=2،
!>          فلن تتم الإشارة إليها.
!>          لم يتم تعديله.
!>
!>  LDU     INTEGER
!>          البعد الرائد لـ U. يجب أن يكون LDU على الأقل N و
!>          على الأقل 1.
!>          لم يتم تعديله.
!>
!>  V       REAL array, dimension (LDV, N)
!>          إذا كان ITYPE=2 أو 3، فإن المثلث السفلي لهذه المصفوفة يحتوي على
!>          متجهات هاوسهولدر التي تصف المصفوفة المتعامدة في
!>          التفكيك. إذا كان ITYPE=1، فلن تتم الإشارة إليها.
!>          لم يتم تعديله.
!>
!>  LDV     INTEGER
!>          البعد الرائد لـ V. يجب أن يكون LDV على الأقل N و
!>          على الأقل 1.
!>          لم يتم تعديله.
!>
!>  TAU     REAL array, dimension (N)
!>          إذا كان ITYPE >= 2، فإن TAU(j) هو العامل القياسي لـ
!>          v(j) v(j)**T في تحويل هاوسهولدر H(j) لـ
!>          حاصل الضرب  U = H(1)...H(n-2)
!>          إذا كان ITYPE < 2، فلن تتم الإشارة إلى TAU.
!>          لم يتم تعديله.
!>
!>  WORK    REAL array, dimension (2*N**2)
!>          مساحة العمل.
!>          تم تعديله.
!>
!>  RESULT  REAL array, dimension (2)
!>          القيم المحسوبة بواسطة الاختبارين الموصوفين أعلاه.
!>          القيم حاليًا محدودة بـ 1/ulp، لتجنب تجاوز سعة الذاكرة.
!>          RESULT(1) يتم تعديله دائمًا. RESULT(2) يتم تعديله فقط
!>          إذا كان LDU على الأقل N.
!>          تم تعديله.
!>
