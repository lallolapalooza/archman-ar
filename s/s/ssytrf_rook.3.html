subroutine ssytrf_rook (character uplo, integer n, real, dimension( lda, * ) a, integer lda, integer, dimension( * ) ipiv, real, dimension( * ) work, integer lwork, integer info)

SSYTRF_ROOK

الغرض:

!>
!> SSYTRF_ROOK تحسب تحليل مصفوفة متناظرة حقيقية A
!> باستخدام طريقة محورية قطرية Bunch-Kaufman المحدودة ().
!> شكل التحليل هو
!>
!>    A = U*D*U**T  أو  A = L*D*L**T
!>
!> حيث U (أو L) هو حاصل ضرب مصفوفات التبديل والمصفوفات المثلثية العلوية (السفلية)
!>، و D مصفوفة متناظرة وقطرية جزئية ذات كتل قطرية 1x1 و 2x2.
!>
!> هذه هي النسخة المجمعة من الخوارزمية، تستدعي BLAS المستوى 3.
!>

المعلمات

UPLO 
!>          UPLO هو CHARACTER*1
!>          = 'U':  المثلث العلوي من A مخزن؛
!>          = 'L':  المثلث السفلي من A مخزن.
!>

N

!>          N هو INTEGER
!>          رتبة المصفوفة A.  N >= 0.
!>

A

!>          A هي مصفوفة REAL، البعد (LDA,N)
!>          عند الدخول، المصفوفة المتناظرة A.  إذا كان UPLO = 'U'، الجزء العلوي
!>          المثلثي من A (حجم N×N) يحتوي على الجزء العلوي
!>          المثلثي للمصفوفة A، والجزء المثلثي السفلي الصارم
!>          لـ A لا تتم الإشارة إليه.  إذا كان UPLO = 'L'، الجزء الرائد
!>          المثلثي السفلي من A (حجم N×N) يحتوي على الجزء السفلي
!>          المثلثي للمصفوفة A، والجزء المثلثي العلوي الصارم
!>          لـ A لا تتم الإشارة إليه.
!>
!>          عند الخروج، مصفوفة الكتلة القطرية D والعوامل المستخدمة
!>          للحصول على العامل U أو L (انظر أدناه لمزيد من التفاصيل).
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرائد للمصفوفة A.  LDA >= max(1,N).
!>

IPIV

!>          IPIV هي مصفوفة INTEGER، البعد (N)
!>          تفاصيل التبادلات وهيكل الكتلة لـ D.
!>
!>          إذا كان UPLO = 'U':
!>               إذا كان IPIV(k) > 0، فإن الصفوف والأعمدة k و IPIV(k)
!>               تم تبديلها و D(k,k) هي كتلة قطرية 1x1.
!>
!>               إذا كان IPIV(k) < 0 و IPIV(k-1) < 0، فإن الصفوف
!>               والأعمدة k و -IPIV(k) تم تبديلها و الصفوف
!>               والأعمدة k-1 و -IPIV(k-1) تم تبديلها،
!>               D(k-1:k,k-1:k) هي كتلة قطرية 2x2.
!>
!>          إذا كان UPLO = 'L':
!>               إذا كان IPIV(k) > 0، فإن الصفوف والأعمدة k و IPIV(k)
!>               تم تبديلها و D(k,k) هي كتلة قطرية 1x1.
!>
!>               إذا كان IPIV(k) < 0 و IPIV(k+1) < 0، فإن الصفوف
!>               والأعمدة k و -IPIV(k) تم تبديلها و الصفوف
!>               والأعمدة k+1 و -IPIV(k+1) تم تبديلها،
!>               D(k:k+1,k:k+1) هي كتلة قطرية 2x2.
!>

WORK

!>          WORK هي مصفوفة REAL، البعد (MAX(1,LWORK)).
!>          عند الخروج، إذا كان INFO = 0، فإن WORK(1) تعيد LWORK المثلى.
!>

LWORK

!>          LWORK هو INTEGER
!>          طول WORK.  LWORK >= 1.  للحصول على أفضل أداء
!>          LWORK >= N*NB، حيث NB هي حجم الكتلة التي تعيدها ILAENV.
!>
!>          إذا كان LWORK = -1، فإنه يُفترض استعلام مساحة العمل؛ الروتين
!>          يحسب فقط الحجم الأمثل لمصفوفة WORK، ويعيد
!>          هذه القيمة كإدخال أول لمصفوفة WORK، ولا يتم إصدار رسالة خطأ
!>          متعلقة بـ LWORK بواسطة XERBLA.
!>

INFO

!>          INFO هو INTEGER
!>          = 0:  انتهاء ناجح
!>          < 0:  إذا كان INFO = -i، فإن المعامل i كان له قيمة غير قانونية
!>          > 0:  إذا كان INFO = i، فإن D(i,i) يساوي صفرًا تمامًا. التحليل
!>                اكتمل، ولكن المصفوفة القطرية الجزئية D
!>                منفردة تمامًا، وسيحدث قسمة على صفر إذا
!>                تم استخدامها لحل نظام من المعادلات.
!>

تفاصيل إضافية:

!>
!>  إذا كان UPLO = 'U'، فإن A = U*D*U**T، حيث
!>     U = P(n)*U(n)* ... *P(k)U(k)* ...,
!>  أي، U هو حاصل ضرب المصطلحات P(k)*U(k)، حيث k تتناقص من n إلى
!>  1 بخطوات 1 أو 2، و D هي مصفوفة قطرية جزئية ذات كتل 1x1
!>  و 2x2 قطرية D(k).  P(k) هي مصفوفة تبديل كما
!>  يتم تعريفها بواسطة IPIV(k)، و U(k) هي مصفوفة مثلثة علوية وحدوية،
!>  بحيث إذا كانت الكتلة القطرية D(k) من الرتبة s (s = 1 أو 2)، فإن
!>
!>             (   I    v    0   )   k-s
!>     U(k) =  (   0    I    0   )   s
!>             (   0    0    I   )   n-k
!>                k-s   s   n-k
!>
!>  إذا كان s = 1، فإن D(k) تطغى على A(k,k)، و v تطغى على A(1:k-1,k).
!>  إذا كان s = 2، فإن المثلث العلوي من D(k) يطغى على A(k-1,k-1)، A(k-1,k)،
!>  و A(k,k)، و v تطغى على A(1:k-2,k-1:k).
!>
!>  إذا كان UPLO = 'L'، فإن A = L*D*L**T، حيث
!>     L = P(1)*L(1)* ... *P(k)*L(k)* ...,
!>  أي، L هو حاصل ضرب المصطلحات P(k)*L(k)، حيث k تتزايد من 1 إلى
!>  n بخطوات 1 أو 2، و D هي مصفوفة قطرية جزئية ذات كتل 1x1
!>  و 2x2 قطرية D(k).  P(k) هي مصفوفة تبديل كما
!>  يتم تعريفها بواسطة IPIV(k)، و L(k) هي مصفوفة مثلثة سفلية وحدوية،
!>  بحيث إذا كانت الكتلة القطرية D(k) من الرتبة s (s = 1 أو 2)، فإن
!>
!>             (   I    0     0   )  k-1
!>     L(k) =  (   0    I     0   )  s
!>             (   0    v     I   )  n-k-s+1
!>                k-1   s  n-k-s+1
!>
!>  إذا كان s = 1، فإن D(k) تطغى على A(k,k)، و v تطغى على A(k+1:n,k).
!>  إذا كان s = 2، فإن المثلث السفلي من D(k) يطغى على A(k,k)، A(k+1,k)،
!>  و A(k+1,k+1)، و v تطغى على A(k+2:n,k:k+1).
!>
