subroutine ssbt21 (character uplo, integer n, integer ka, integer ks, real, dimension( lda, * ) a, integer lda, real, dimension( * ) d, real, dimension( * ) e, real, dimension( ldu, * ) u, integer ldu, real, dimension( * ) work, real, dimension( 2 ) result)

SSBT21

الغرض:

!>
!> SSBT21  بشكل عام يتحقق من تفكيك من الشكل
!>
!>         A = U S U**T
!>
!> حيث **T تعني تبديل، A هي مصفوفة متناظرة ذات نطاق، U هي مصفوفة متعامدة،
!> و S هي مصفوفة قطرية (إذا كان KS=0) أو مصفوفة متناظرة ثلاثية الأقطار
!> (إذا كان KS=1).
!>
!> على وجه التحديد:
!>
!>         RESULT(1) = | A - U S U**T | / ( |A| n ulp ) و
!>         RESULT(2) = | I - U U**T | / ( n ulp )
!>

المعلمات

UPLO 
!>          UPLO هو CHARACTER
!>          إذا كان UPLO='U'، يتم استخدام المثلث العلوي لـ A و V ولن تتم
!>          الإشارة إلى المثلث السفلي (الصارم).
!>          إذا كان UPLO='L'، يتم استخدام المثلث السفلي لـ A و V ولن تتم
!>          الإشارة إلى المثلث العلوي (الصارم).
!>

N

!>          N هو INTEGER
!>          حجم المصفوفة. إذا كان صفرًا، لا تفعل SSBT21 شيئًا.
!>          يجب أن يكون على الأقل صفرًا.
!>

KA

!>          KA هو INTEGER
!>          عرض النطاق للمصفوفة A. يجب أن يكون على الأقل صفرًا. إذا
!>          كان أكبر من N-1، فسيتم استخدام max( 0, N-1 ).
!>

KS

!>          KS هو INTEGER
!>          عرض النطاق للمصفوفة S. قد يكون فقط صفرًا أو واحدًا.
!>          إذا كان صفرًا، فإن S تكون قطرية، ولا تتم الإشارة إلى E. إذا
!>          كان واحدًا، فإن S تكون ثلاثية الأقطار متناظرة.
!>

A

!>          A هي مصفوفة REAL، أبعادها (LDA, N)
!>          المصفوفة الأصلية (غير المفككة). يُفترض أنها
!>          متناظرة، وسيتم الإشارة فقط إلى الجزء العلوي (UPLO='U') أو
!>          الجزء السفلي (UPLO='L') فقط.
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرئيسي لـ A. يجب أن يكون على الأقل 1
!>          وعلى الأقل min( KA, N-1 ).
!>

D

!>          D هي مصفوفة REAL، أبعادها (N)
!>          القطر للمصفوفة S (ثلاثية الأقطار المتناظرة).
!>

E

!>          E هي مصفوفة REAL، أبعادها (N-1)
!>          العناصر خارج القطر للمصفوفة S (ثلاثية الأقطار المتناظرة).
!>          E(1) هو العنصر (1,2) و (2,1)، E(2) هو العنصر (2,3) و
!>          (3,2)، إلخ.
!>          لا تتم الإشارة إليه إذا كان KS=0.
!>

U

!>          U هي مصفوفة REAL، أبعادها (LDU, N)
!>          المصفوفة المتعامدة في التفكيك، معبرًا عنها
!>          كمصفوفة كثيفة (أي، ليست كناتج تحويلات هاوسهولدر،
!>          تحويلات جيفنز، إلخ.)
!>

LDU

!>          LDU هو INTEGER
!>          البعد الرئيسي لـ U. يجب أن يكون LDU على الأقل N
!>          وعلى الأقل 1.
!>

WORK

!>          WORK هي مصفوفة REAL، أبعادها (N**2+N)
!>

RESULT

!>          RESULT هي مصفوفة REAL، أبعادها (2)
!>          القيم المحسوبة من الاختبارين الموصوفين أعلاه.
!>          القيم محدودة حاليًا بـ 1/ulp، لتجنب تجاوز السعة.
!>
