SSL_load_client_CA_file(3ssl)

الحزمة: openssl

التراخيص: Apache-2.0

SSL_LOAD_CLIENT_CA_FILE(3ssl) OpenSSL SSL_LOAD_CLIENT_CA_FILE(3ssl)

الاسم

SSL_load_client_CA_file_ex, SSL_load_client_CA_file, SSL_add_file_cert_subjects_to_stack, SSL_add_dir_cert_subjects_to_stack, SSL_add_store_cert_subjects_to_stack - تحميل أسماء الشهادات

الملخص

#include <openssl/ssl.h>
STACK_OF(X509_NAME) *SSL_load_client_CA_file_ex(const char *file,
                                                OSSL_LIB_CTX *libctx,
                                                const char *propq);
STACK_OF(X509_NAME) *SSL_load_client_CA_file(const char *file);
int SSL_add_file_cert_subjects_to_stack(STACK_OF(X509_NAME) *stack,
                                        const char *file);
int SSL_add_dir_cert_subjects_to_stack(STACK_OF(X509_NAME) *stack,
                                       const char *dir);
int SSL_add_store_cert_subjects_to_stack(STACK_OF(X509_NAME) *stack,
                                         const char *store);

الوصف

SSL_load_client_CA_file_ex() تقرأ الشهادات من file وتعيد STACK_OF(X509_NAME) بأسماء الموضوع الموجودة. يتم استخدام سياق المكتبة libctx واستعلام الخصائص propq عند جلب الخوارزميات من الموفرين.

SSL_load_client_CA_file() مشابه لـ SSL_load_client_CA_file_ex() ولكنه يستخدم NULL لسياق المكتبة libctx واستعلام الخصائص propq.

SSL_add_file_cert_subjects_to_stack() تقرأ الشهادات من file، وتضيف اسم موضوعها إلى stack الموجود بالفعل.

SSL_add_dir_cert_subjects_to_stack() تقرأ الشهادات من كل ملف في الدليل dir، وتضيف اسم موضوعها إلى stack الموجود بالفعل.

SSL_add_store_cert_subjects_to_stack() تقوم بتحميل الشهادات من عنوان URI store، وتضيف اسم موضوعها إلى stack الموجود بالفعل.

ملاحظات

SSL_load_client_CA_file() تقرأ ملفًا من الشهادات بصيغة PEM وتستخرج أسماء X509_NAMES للشهادات الموجودة. بينما يوحي الاسم بالاستخدام الخاص كوظيفة دعم لـ SSL_CTX_set_client_CA_list(3)، فإنها لا تقتصر على شهادات CA.

قيم الإرجاع

يمكن أن تحدث قيم الإرجاع التالية لـ SSL_load_client_CA_file_ex()، و SSL_load_client_CA_file():

NULL
فشلت العملية، تحقق من مكدس الأخطاء لمعرفة السبب.
مؤشر إلى STACK_OF(X509_NAME)
مؤشر إلى أسماء الموضوع للشهادات التي تم قراءتها بنجاح.

يمكن أن تحدث قيم الإرجاع التالية لـ SSL_add_file_cert_subjects_to_stack()، SSL_add_dir_cert_subjects_to_stack()، و SSL_add_store_cert_subjects_to_stack():

0 (فشل)
فشلت العملية.
1 (نجاح)
نجحت العملية.

أمثلة

تحميل أسماء CAs من ملف واستخدامه كقائمة CA للعميل:

SSL_CTX *ctx;
STACK_OF(X509_NAME) *cert_names;
...
cert_names = SSL_load_client_CA_file("/path/to/CAfile.pem");
if (cert_names != NULL)
    SSL_CTX_set_client_CA_list(ctx, cert_names);
else
    /* خطأ */
...
