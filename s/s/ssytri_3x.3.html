subroutine ssytri_3x (character uplo, integer n, real, dimension( lda, * ) a, integer lda, real, dimension( * ) e, integer, dimension( * ) ipiv, real, dimension( n+nb+1, * ) work, integer nb, integer info)

SSYTRI_3X

الغرض:

!> SSYTRI_3X يحسب معكوس مصفوفة حقيقية غير محددة متماثلة
!> A باستخدام التحليل الذي تم حسابه بواسطة SSYTRF_RK أو SSYTRF_BK:
!>
!>     A = P*U*D*(U**T)*(P**T) أو A = P*L*D*(L**T)*(P**T),
!>
!> حيث U (أو L) هي مصفوفة مثلثة علوية (أو سفلية) وحدة،
!> U**T (أو L**T) هي منقولة U (أو L)، P هي مصفوفة تبديل، P**T هي منقولة P، و D هي متماثلة وكتلة قطرية
!> مع كتل قطرية 1x1 و 2x2.
!>
!> هذا هو الإصدار المعبأ من الخوارزمية، يستدعي المستوى 3 BLAS.
!>

المعلمات

UPLO 
!>          UPLO هو CHARACTER*1
!>          يحدد ما إذا كانت تفاصيل التحليل مخزنة كمصفوفة مثلثية علوية أو سفلية.
!>          = 'U':  يتم تخزين المثلث العلوي لـ A؛
!>          = 'L':  يتم تخزين المثلث السفلي لـ A.
!>

N

!>          N هو INTEGER
!>          ترتيب المصفوفة A.  N >= 0.
!>

A

!>          A هي مصفوفة REAL، ذات أبعاد (LDA,N)
!>          عند الإدخال، القطر للمصفوفة القطرية الكتلة D والعوامل U أو L كما تم حسابها بواسطة SYTRF_RK و SSYTRF_BK:
!>            أ) عناصر القطر فقط للمصفوفة القطرية الكتلة المتماثلة D
!>               على قطر A، أي D(k,k) = A(k,k)؛
!>               (يجب توفير عناصر فوق القطر (أو تحت القطر) لـ D
!>                في المصفوفة E عند الإدخال)، و
!>            ب) إذا كان UPLO = 'U': العامل U في الجزء فوق القطر لـ A.
!>               إذا كان UPLO = 'L': العامل L في الجزء تحت القطر لـ A.
!>
!>          عند الخروج، إذا كان INFO = 0، فإن معكوس المصفوفة الأصلية المتماثلة.
!>             إذا كان UPLO = 'U': يتم تكوين الجزء العلوي المثلثي للمعكوس
!>             ولا يتم الرجوع إلى الجزء السفلي من A؛
!>             إذا كان UPLO = 'L': يتم تكوين الجزء السفلي المثلثي للمعكوس
!>             ولا يتم الرجوع إلى الجزء العلوي من A.
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرائد للمصفوفة A.  LDA >= max(1,N).
!>

E

!>          E هي مصفوفة REAL، ذات أبعاد (N)
!>          عند الإدخال، تحتوي على عناصر فوق القطر (أو تحت القطر)
!>          للمصفوفة القطرية الكتلة المتماثلة D
!>          مع كتل قطرية 1x1 أو 2x2، حيث
!>          إذا كان UPLO = 'U': E(i) = D(i-1,i)، i=2:N، E(1) غير مشار إليها؛
!>          إذا كان UPLO = 'L': E(i) = D(i+1,i)، i=1:N-1، E(N) غير مشار إليها.
!>
!>          ملاحظة: لكتلة قطرية 1x1 D(k)، حيث
!>          1 <= k <= N، لا تتم الإشارة إلى العنصر E(k) في كلتا
!>          حالات UPLO = 'U' أو UPLO = 'L'.
!>

IPIV

!>          IPIV هي مصفوفة INTEGER، ذات أبعاد (N)
!>          تفاصيل التبادلات وهيكل الكتلة لـ D
!>          كما تم تحديده بواسطة SSYTRF_RK أو SSYTRF_BK.
!>

WORK

!>          WORK هي مصفوفة REAL، ذات أبعاد (N+NB+1,NB+3).
!>

NB

!>          NB هو INTEGER
!>          حجم الكتلة.
!>

INFO

!>          INFO هو INTEGER
!>          = 0: الخروج الناجح
!>          < 0: إذا كان INFO = -i، فإن الوسيط i كان له قيمة غير قانونية
!>          > 0: إذا كان INFO = i، فإن D(i,i) = 0؛ المصفوفة منفردة ولا يمكن حساب معكوسها.
!>
