subroutine ssgt01 (integer itype, character uplo, integer n, integer m, real, dimension( lda, * ) a, integer lda, real, dimension( ldb, * ) b, integer ldb, real, dimension( ldz, * ) z, integer ldz, real, dimension( * ) d, real, dimension( * ) work, real, dimension( * ) result)

SSGT01

الغرض:

!>
!> SSGT01 يتحقق من تحليل للشكل
!>
!>    A Z   =  B Z D أو
!>    A B Z =  Z D أو
!>    B A Z =  Z D
!>
!> حيث A مصفوفة متناظرة، B
!> موجبة محددة ومتناظرة، Z متعامدة، و D قطرية.
!>
!> يتم حساب إحدى نسب الاختبار التالية:
!>
!> ITYPE = 1:  RESULT(1) = | A Z - B Z D | / ( |A| |Z| n ulp )
!>
!> ITYPE = 2:  RESULT(1) = | A B Z - Z D | / ( |A| |Z| n ulp )
!>
!> ITYPE = 3:  RESULT(1) = | B A Z - Z D | / ( |A| |Z| n ulp )
!>

المعلمات

ITYPE 
!>          ITYPE هو عدد صحيح (INTEGER)
!>          شكل مشكلة القيمة الذاتية المعممة المتناظرة.
!>          = 1:  A*z = (lambda)*B*z
!>          = 2:  A*B*z = (lambda)*z
!>          = 3:  B*A*z = (lambda)*z
!>

UPLO

!>          UPLO هو حرف (CHARACTER*1)
!>          يحدد ما إذا كان الجزء العلوي أو السفلي المثلثي من المصفوفتين المتناظرتين A و B مخزنًا.
!>          = 'U':  مثلثي علوي
!>          = 'L':  مثلثي سفلي
!>

N

!>          N هو عدد صحيح (INTEGER)
!>          رتبة المصفوفة A. N >= 0.
!>

M

!>          M هو عدد صحيح (INTEGER)
!>          عدد القيم الذاتية التي تم العثور عليها.  0 <= M <= N.
!>

A

!>          A هي مصفوفة حقيقية (REAL array)، حجمها (LDA, N)
!>          المصفوفة المتناظرة الأصلية A.
!>

LDA

!>          LDA هو عدد صحيح (INTEGER)
!>          البعد الرائد للمصفوفة A.  LDA >= max(1,N).
!>

B

!>          B هي مصفوفة حقيقية (REAL array)، حجمها (LDB, N)
!>          المصفوفة المتناظرة الموجبة المحددة الأصلية B.
!>

LDB

!>          LDB هو عدد صحيح (INTEGER)
!>          البعد الرائد للمصفوفة B.  LDB >= max(1,N).
!>

Z

!>          Z هي مصفوفة حقيقية (REAL array)، حجمها (LDZ, M)
!>          المتجهات الذاتية المحسوبة لمشكلة القيمة الذاتية المعممة.
!>

LDZ

!>          LDZ هو عدد صحيح (INTEGER)
!>          البعد الرائد للمصفوفة Z.  LDZ >= max(1,N).
!>

D

!>          D هي مصفوفة حقيقية (REAL array)، حجمها (M)
!>          القيم الذاتية المحسوبة لمشكلة القيمة الذاتية المعممة.
!>

WORK

!>          WORK هي مصفوفة حقيقية (REAL array)، حجمها (N*N)
!>

RESULT

!>          RESULT هي مصفوفة حقيقية (REAL array)، حجمها (1)
!>          نسبة الاختبار كما هو موضح أعلاه.
!>
