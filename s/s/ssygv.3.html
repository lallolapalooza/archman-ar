subroutine ssygv (integer itype, character jobz, character uplo, integer n, real, dimension( lda, * ) a, integer lda, real, dimension( ldb, * ) b, integer ldb, real, dimension( * ) w, real, dimension( * ) work, integer lwork, integer info)

SSYGV

الغرض:

!>
!> SSYGV يحسب جميع القيم الذاتية، واختياريًا، المتجهات الذاتية
!> لمشكلة قياسات المتماثل المحدد العام الحقيقية، بالشكل
!> A*x=(lambda)*B*x,  A*Bx=(lambda)*x,  أو B*A*x=(lambda)*x.
!> هنا يفترض أن A و B متماثلان وأن B أيضًا
!> محدد إيجابيًا.
!>

المعلمات

ITYPE 
!>          ITYPE هو عدد صحيح (INTEGER)
!>          يحدد نوع المشكلة المراد حلها:
!>          = 1:  A*x = (lambda)*B*x
!>          = 2:  A*B*x = (lambda)*x
!>          = 3:  B*A*x = (lambda)*x
!>

JOBZ

!>          JOBZ هو حرف واحد (CHARACTER*1)
!>          = 'N':  حساب القيم الذاتية فقط؛
!>          = 'V':  حساب القيم الذاتية والمتجهات الذاتية.
!>

UPLO

!>          UPLO هو حرف واحد (CHARACTER*1)
!>          = 'U':  المثلثات العلوية لـ A و B مخزنة؛
!>          = 'L':  المثلثات السفلية لـ A و B مخزنة.
!>

N

!>          N هو عدد صحيح (INTEGER)
!>          ترتيب المصفوفات A و B.  N >= 0.
!>

A

!>          A هو مصفوفة أرقام حقيقية (REAL array)، بأبعاد (LDA, N)
!>          عند الإدخال، المصفوفة المتماثلة A.  إذا كانت UPLO = 'U'، فإن
!>          الجزء العلوي الأيسر من A بحجم N*N يحتوي على
!>          الجزء العلوي من المصفوفة A.  إذا كانت UPLO = 'L'، فإن
!>          الجزء السفلي الأيسر من A بحجم N*N يحتوي على
!>          الجزء السفلي من المصفوفة A.
!>
!>          عند الخروج، إذا كانت JOBZ = 'V'، فإذا كانت INFO = 0، فإن A تحتوي على
!>          المصفوفة Z للمتجهات الذاتية.  يتم تطبيع المتجهات الذاتية كما يلي:
!>          إذا كانت ITYPE = 1 أو 2، فإن Z**T*B*Z = I؛
!>          إذا كانت ITYPE = 3، فإن Z**T*inv(B)*Z = I.
!>          إذا كانت JOBZ = 'N'، فعند الخروج، يتم تدمير المثلث العلوي (إذا كانت UPLO='U')
!>          أو المثلث السفلي (إذا كانت UPLO='L') من A، بما في ذلك القطر.
!>

LDA

!>          LDA هو عدد صحيح (INTEGER)
!>          البعد الرئيسي للمصفوفة A.  LDA >= max(1,N).
!>

B

!>          B هو مصفوفة أرقام حقيقية (REAL array)، بأبعاد (LDB, N)
!>          عند الإدخال، المصفوفة المتماثلة المحددة إيجابيًا B.
!>          إذا كانت UPLO = 'U'، فإن الجزء العلوي الأيسر من B بحجم N*N يحتوي على
!>          الجزء العلوي من المصفوفة B.
!>          إذا كانت UPLO = 'L'، فإن الجزء السفلي الأيسر من B بحجم N*N يحتوي على
!>          الجزء السفلي من المصفوفة B.
!>
!>          عند الخروج، إذا كانت INFO <= N، يتم استبدال جزء B الذي يحتوي على المصفوفة
!>          بالعامل المثلثي U أو L من تحليل Cholesky
!>          B = U**T*U أو B = L*L**T.
!>

LDB

!>          LDB هو عدد صحيح (INTEGER)
!>          البعد الرئيسي للمصفوفة B.  LDB >= max(1,N).
!>

W

!>          W هي مصفوفة أرقام حقيقية (REAL array)، بأبعاد (N)
!>          إذا كانت INFO = 0، فإن القيم الذاتية مرتبة تصاعديًا.
!>

WORK

!>          WORK هي مصفوفة أرقام حقيقية (REAL array)، بأبعاد (MAX(1,LWORK))
!>          عند الخروج، إذا كانت INFO = 0، فإن WORK(1) تعيد LWORK الأمثل.
!>

LWORK

!>          LWORK هو عدد صحيح (INTEGER)
!>          طول المصفوفة WORK.  LWORK >= max(1,3*N-1).
!>          للكفاءة المثلى، LWORK >= (NB+2)*N،
!>          حيث NB هو حجم الكتلة لـ SSYTRD الذي تعيده ILAENV.
!>
!>          إذا كانت LWORK = -1، فسيتم افتراض استعلام مساحة عمل؛ يقوم الروتين
!>          بحساب حجم مصفوفة WORK الأمثل فقط، ويعيد
!>          هذه القيمة كأول إدخال في مصفوفة WORK، ولا يتم إصدار أي رسالة خطأ
!>          متعلقة بـ LWORK بواسطة XERBLA.
!>

INFO

!>          INFO هو عدد صحيح (INTEGER)
!>          = 0:  الانتهاء بنجاح
!>          < 0:  إذا كانت INFO = -i، فإن المعلمة i كانت قيمة غير قانونية
!>          > 0:  عادت SPOTRF أو SSYEV برمز خطأ:
!>             <= N:  إذا كانت INFO = i، فشلت SSYEV في التقارب؛
!>                    i عناصر غير قطرية لشكل ثلاثي قطري وسيط
!>                    لم تتقارب إلى الصفر؛
!>             > N:   إذا كانت INFO = N + i، لـ 1 <= i <= N، فإن
!>                    المinal الرئيسي الرائد من الرتبة i لـ B ليس موجبًا.
!>                    لم يكن من الممكن إكمال تحليل B ولم يتم حساب
!>                    أي قيم ذاتية أو متجهات ذاتية.
!>
