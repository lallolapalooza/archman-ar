subroutine ssyr2k (character uplo, character trans, integer n, integer k, real alpha, real, dimension(lda,*) a, integer lda, real, dimension(ldb,*) b, integer ldb, real beta, real, dimension(ldc,*) c, integer ldc)

SSYR2K

الغرض:

!>
!> SSYR2K  تقوم بإحدى عمليات الرتبة 2k المتماثلة
!>
!>    C := alpha*A*B**T + alpha*B*A**T + beta*C,
!>
!> أو
!>
!>    C := alpha*A**T*B + alpha*B**T*A + beta*C,
!>
!> حيث  alpha و beta  هما قيمتان عدديتان، C هي مصفوفة متماثلة بحجم n × n
!> و A و B هما مصفوفتان بحجم n × k  في الحالة الأولى ومصفوفتان بحجم k × n
!> في الحالة الثانية.
!>

المعلمات

UPLO 
!>          UPLO هو CHARACTER*1
!>           عند الإدخال، يحدد  UPLO  ما إذا كان يجب الرجوع إلى الجزء
!>           المثلثي العلوي أو السفلي من المصفوفة  C  كما يلي:
!>
!>              UPLO = 'U' أو 'u'   يجب الرجوع فقط إلى الجزء المثلثي العلوي من C.
!>
!>              UPLO = 'L' أو 'l'   يجب الرجوع فقط إلى الجزء المثلثي السفلي من C.
!>

TRANS

!>          TRANS هو CHARACTER*1
!>           عند الإدخال، يحدد  TRANS  العملية التي يجب تنفيذها كما يلي:
!>
!>              TRANS = 'N' أو 'n'   C := alpha*A*B**T + alpha*B*A**T +
!>                                        beta*C.
!>
!>              TRANS = 'T' أو 't'   C := alpha*A**T*B + alpha*B**T*A +
!>                                        beta*C.
!>
!>              TRANS = 'C' أو 'c'   C := alpha*A**T*B + alpha*B**T*A +
!>                                        beta*C.
!>

N

!>          N هو INTEGER
!>           عند الإدخال، يحدد N  رتبة المصفوفة C. يجب أن يكون N
!>           على الأقل صفر.
!>

K

!>          K هو INTEGER
!>           عند الإدخال مع  TRANS = 'N' أو 'n'، يحدد K  عدد
!>           الأعمدة للمصفوفتين A و B، وعند الإدخال مع
!>           TRANS = 'T' أو 't' أو 'C' أو 'c'، يحدد K  عدد
!>           الصفوف للمصفوفتين A و B. يجب أن يكون K على الأقل صفر.
!>

ALPHA

!>          ALPHA هو REAL
!>           عند الإدخال، يحدد ALPHA  القيمة العددية alpha.
!>

A

!>          A هي مصفوفة REAL، ذات بعد ( LDA, ka )، حيث ka هي
!>           k  عندما  TRANS = 'N' أو 'n'، و n  خلاف ذلك.
!>           قبل الإدخال مع  TRANS = 'N' أو 'n'، يجب أن يحتوي الجزء
!>           الرائد بحجم n × k  من المصفوفة A  على المصفوفة A، وإلا
!>           فيجب أن يحتوي الجزء الرائد بحجم k × n  من المصفوفة A على
!>           المصفوفة A.
!>

LDA

!>          LDA هو INTEGER
!>           عند الإدخال، يحدد LDA  البعد الأول لـ A كما هو معلن
!>           في  البرنامج (الفرعي) المستدعي.  عندما  TRANS = 'N' أو 'n'
!>           يجب أن يكون LDA  على الأقل  max( 1, n )، وإلا يجب أن يكون LDA
!>           على الأقل  max( 1, k ).
!>

B

!>          B هي مصفوفة REAL، ذات بعد ( LDB, kb )، حيث kb هي
!>           k  عندما  TRANS = 'N' أو 'n'، و n  خلاف ذلك.
!>           قبل الإدخال مع  TRANS = 'N' أو 'n'، يجب أن يحتوي الجزء
!>           الرائد بحجم n × k  من المصفوفة B  على المصفوفة B، وإلا
!>           فيجب أن يحتوي الجزء الرائد بحجم k × n  من المصفوفة B على
!>           المصفوفة B.
!>

LDB

!>          LDB هو INTEGER
!>           عند الإدخال، يحدد LDB  البعد الأول لـ B كما هو معلن
!>           في  البرنامج (الفرعي) المستدعي.  عندما  TRANS = 'N' أو 'n'
!>           يجب أن يكون LDB  على الأقل  max( 1, n )، وإلا يجب أن يكون LDB
!>           على الأقل  max( 1, k ).
!>

BETA

!>          BETA هو REAL
!>           عند الإدخال، يحدد BETA  القيمة العددية beta.
!>

C

!>          C هي مصفوفة REAL، ذات بعد ( LDC, N )
!>           قبل الإدخال مع  UPLO = 'U' أو 'u'، يجب أن يحتوي الجزء
!>           الرائد بحجم n × n  المثلثي العلوي من المصفوفة C على الجزء
!>           المثلثي العلوي  من المصفوفة المتماثلة، ولا يتم الرجوع إلى الجزء
!>           المثلثي السفلي الصارم من C. عند الخروج، يتم الكتابة فوق الجزء
!>           المثلثي العلوي من المصفوفة C بالجزء المثلثي العلوي من المصفوفة المحدثة.
!>           قبل الإدخال مع  UPLO = 'L' أو 'l'، يجب أن يحتوي الجزء
!>           الرائد بحجم n × n  المثلثي السفلي من المصفوفة C على الجزء
!>           المثلثي السفلي  من المصفوفة المتماثلة، ولا يتم الرجوع إلى الجزء
!>           المثلثي العلوي الصارم من C. عند الخروج، يتم الكتابة فوق الجزء
!>           المثلثي السفلي من المصفوفة C بالجزء المثلثي السفلي من المصفوفة المحدثة.
!>

LDC

!>          LDC هو INTEGER
!>           عند الإدخال، يحدد LDC  البعد الأول لـ C كما هو معلن
!>           في  البرنامج (الفرعي) المستدعي.  يجب أن يكون LDC  على الأقل
!>           max( 1, n ).
!>
