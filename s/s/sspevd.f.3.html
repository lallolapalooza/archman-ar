subroutine sspevd (jobz, uplo, n, ap, w, z, ldz, work, lwork, iwork, liwork, info)
SSPEVD تحسب القيم الذاتية، واختياريًا، المتجهات الذاتية اليسرى و/أو اليمنى للمصفوفات الأخرى

توثيق الدالة/الإجراء الفرعي

subroutine sspevd (character jobz, character uplo, integer n, real, dimension( * ) ap, real, dimension( * ) w, real, dimension( ldz, * ) z, integer ldz, real, dimension( * ) work, integer lwork, integer, dimension( * ) iwork, integer liwork, integer info)

SSPEVD تحسب القيم الذاتية، واختياريًا، المتجهات الذاتية للمصفوفة المتماثلة الحقيقية A المخزنة بتنسيق مضغوط. إذا كانت المتجهات الذاتية مطلوبة، فإنها تستخدم خوارزمية الانقسام والغزو.

الغرض:

!>
!> SSPEVD تحسب جميع القيم الذاتية، واختياريًا، المتجهات الذاتية
!> لمصفوفة متماثلة حقيقية A في تخزين مضغوط. إذا كانت المتجهات الذاتية
!> مطلوبة، فإنها تستخدم خوارزمية الانقسام والغزو.
!>
!>

المعلمات

JOBZ 
!>          JOBZ هو CHARACTER*1
!>          = 'N':  حساب القيم الذاتية فقط؛
!>          = 'V':  حساب القيم الذاتية والمتجهات الذاتية.
!>

UPLO

!>          UPLO هو CHARACTER*1
!>          = 'U':  يتم تخزين المثلث العلوي للمصفوفة A؛
!>          = 'L':  يتم تخزين المثلث السفلي للمصفوفة A.
!>

N

!>          N هو INTEGER
!>          رتبة المصفوفة A.  N >= 0.
!>

AP

!>          AP هو مصفوفة REAL، البعد (N*(N+1)/2)
!>          عند الإدخال، المثلث العلوي أو السفلي للمصفوفة المتماثلة
!>          A، مخزن بشكل عمودي في مصفوفة خطية.  يتم تخزين العمود j للمصفوفة A
!>          في المصفوفة AP كما يلي:
!>          إذا كان UPLO = 'U'، فإن AP(i + (j-1)*j/2) = A(i,j) لـ 1<=i<=j؛
!>          إذا كان UPLO = 'L'، فإن AP(i + (j-1)*(2*n-j)/2) = A(i,j) لـ j<=i<=n.
!>
!>          عند الخروج، يتم استبدال AP بقيم تم إنشاؤها أثناء
!>          الاختزال إلى شكل ثلاثي القطر.  إذا كان UPLO = 'U'، فإن القطر
!>          والقطر الخارق الأول للمصفوفة ثلاثية القطر T يتم استبدالهما
!>          بالعناصر المقابلة في A، وإذا كان UPLO = 'L'، فإن
!>          القطر والقطر الفرعي الأول لـ T يتم استبدالهما
!>          بالعناصر المقابلة في A.
!>

W

!>          W هو مصفوفة REAL، البعد (N)
!>          إذا كان INFO = 0، فإن القيم الذاتية مرتبة تصاعديًا.
!>

Z

!>          Z هي مصفوفة REAL، البعد (LDZ, N)
!>          إذا كان JOBZ = 'V'، فعندما يكون INFO = 0، تحتوي Z على
!>          المتجهات الذاتية المتعامدة للمصفوفة A، حيث يمثل العمود i لـ Z
!>          المتجه الذاتي المرتبط بـ W(i).
!>          إذا كان JOBZ = 'N'، فلا يتم الرجوع إلى Z.
!>

LDZ

!>          LDZ هو INTEGER
!>          البعد الرائد للمصفوفة Z.  LDZ >= 1، وإذا
!>          كان JOBZ = 'V'، فإن LDZ >= max(1,N).
!>

WORK

!>          WORK هي مصفوفة REAL، البعد (MAX(1,LWORK))
!>          عند الخروج، إذا كان INFO = 0، فإن WORK(1) يرجع LWORK المطلوبة.
!>

LWORK

!>          LWORK هو INTEGER
!>          بعد المصفوفة WORK.
!>          إذا كان N <= 1، يجب أن يكون LWORK على الأقل 1.
!>          إذا كان JOBZ = 'N' و N > 1، يجب أن يكون LWORK على الأقل 2*N.
!>          إذا كان JOBZ = 'V' و N > 1، يجب أن يكون LWORK على الأقل
!>                                                 1 + 6*N + N**2.
!>
!>          إذا كان LWORK = -1، فإنه يفترض استعلام مساحة العمل؛ تقوم الدالة
!>          فقط بحساب الأحجام المطلوبة للمصفوفات WORK و IWORK،
!>          وتعيد هذه القيم كأول إدخالات في مصفوفتي WORK و IWORK، ولا
!>          يتم إصدار أي رسالة خطأ متعلقة بـ LWORK أو LIWORK بواسطة XERBLA.
!>

IWORK

!>          IWORK هي مصفوفة INTEGER، البعد (MAX(1,LIWORK))
!>          عند الخروج، إذا كان INFO = 0، فإن IWORK(1) يرجع LIWORK المطلوبة.
!>

LIWORK

!>          LIWORK هو INTEGER
!>          بعد المصفوفة IWORK.
!>          إذا كان JOBZ  = 'N' أو N <= 1، يجب أن يكون LIWORK على الأقل 1.
!>          إذا كان JOBZ  = 'V' و N > 1، يجب أن يكون LIWORK على الأقل 3 + 5*N.
!>
!>          إذا كان LIWORK = -1، فإنه يفترض استعلام مساحة العمل؛ تقوم الدالة
!>          فقط بحساب الأحجام المطلوبة للمصفوفتين WORK و IWORK،
!>          وتعيد هذه القيم كأول إدخالات في مصفوفتي WORK و IWORK، ولا
!>          يتم إصدار أي رسالة خطأ متعلقة بـ LWORK أو LIWORK بواسطة XERBLA.
!>

INFO

!>          INFO هو INTEGER
!>          = 0:  انتهاء ناجح
!>          < 0:  إذا كان INFO = -i، فإن المعلمة i كانت لها قيمة غير قانونية.
!>          > 0:  إذا كان INFO = i، فشلت الخوارزمية في التقارب؛
!>                لم تتقارب عناصر القطر غير القطرية
!>                لشكل ثلاثي القطر الوسيط إلى الصفر.
!>
