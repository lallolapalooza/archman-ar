subroutine ssytri2 (character uplo, integer n, real, dimension( lda, * ) a, integer lda, integer, dimension( * ) ipiv, real, dimension( * ) work, integer lwork, integer info)

SSYTRI2

الغرض:

!>
!> SSYTRI2 يحسب معكوس مصفوفة REAL متماثلة غير محددة
!> A باستخدام التحليل A = U*D*U**T أو A = L*D*L**T الذي تم حسابه بواسطة
!> SSYTRF. SSYTRI2 تحدد البعد الرائد لمساحة العمل
!> قبل استدعاء SSYTRI2X الذي يحسب المعكوس فعليًا.
!>

المعلمات

UPLO 
!>          UPLO هو CHARACTER*1
!>          يحدد ما إذا كانت تفاصيل التحليل مخزنة
!>          كمصفوفة مثلثة علوية أو سفلية.
!>          = 'U':  مثلثة علوية، الشكل هو A = U*D*U**T;
!>          = 'L':  مثلثة سفلية، الشكل هو A = L*D*L**T.
!>

N

!>          N هو INTEGER
!>          ترتيب المصفوفة A.  N >= 0.
!>

A

!>          A هي مصفوفة REAL، البعد (LDA,N)
!>          عند الدخول، المصفوفة القطرية المقطعية D والعوامل
!>          المستخدمة للحصول على العامل U أو L كما تم حسابه بواسطة SSYTRF.
!>
!>          عند الخروج، إذا كان INFO = 0، معكوس (المتماثل) للمصفوفة الأصلية.
!>          إذا كان UPLO = 'U'، يتم تشكيل الجزء المثلثي العلوي من المعكوس
!>          ولا يتم الرجوع إلى الجزء السفلي من A تحت القطر؛ إذا كان UPLO = 'L'
!>          يتم تشكيل الجزء المثلثي السفلي من المعكوس ولا يتم الرجوع إلى الجزء
!>          العلوي من A فوق القطر.
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرائد للمصفوفة A.  LDA >= max(1,N).
!>

IPIV

!>          IPIV هي مصفوفة INTEGER، البعد (N)
!>          تفاصيل التبادلات وهيكل المقطع D
!>          كما تم تحديده بواسطة SSYTRF.
!>

WORK

!>          WORK هي مصفوفة REAL، البعد (MAX(1,LWORK))
!>

LWORK

!>          LWORK هو INTEGER
!>          بعد مصفوفة WORK.
!>          إذا كان N = 0، LWORK >= 1، وإلا LWORK >= (N+NB+1)*(NB+3).
!>          إذا كان LWORK = -1، فسيتم افتراض استعلام مساحة عمل؛ تقوم الروتين
!>          بحساب:
!>              - البعد الأمثل لمصفوفة WORK، تعيد
!>          هذه القيمة كأول إدخال في مصفوفة WORK،
!>              - ولا يتم إصدار أي رسالة خطأ متعلقة بـ LWORK بواسطة XERBLA.
!>

INFO

!>          INFO هو INTEGER
!>          = 0: إنهاء ناجح
!>          < 0: إذا كان INFO = -i، كان الوسيط i له قيمة غير قانونية
!>          > 0: إذا كان INFO = i، D(i,i) = 0؛ المصفوفة منفردة ولم يتم حساب معكوسها.
!>
