توثيق الوظيفة/الوحدة الفرعية

subroutine ssytd2 (character uplo, integer n, real, dimension( lda, * ) a, integer lda, real, dimension( * ) d, real, dimension( * ) e, real, dimension( * ) tau, integer info)

SSYTD2 تقلل مصفوفة متناظرة حقيقية إلى صيغة متناظرة ثلاثية الأقطار بواسطة تحويل تشابه متعامد (خوارزمية غير مجمعة).

الغرض:

!>
!> SSYTD2 تقلل مصفوفة متناظرة حقيقية A إلى صيغة متناظرة ثلاثية الأقطار T
!> بواسطة تحويل تشابه متعامد: Q**T * A * Q = T.
!>

المعلمات

UPLO 
!>          UPLO هو CHARACTER*1
!>          يحدد ما إذا كان الجزء المثلثي العلوي أو السفلي من المصفوفة
!>          المتناظرة A مخزنًا:
!>          = 'U':  مثلثي علوي
!>          = 'L':  مثلثي سفلي
!>

N

!>          N هو INTEGER
!>          رتبة المصفوفة A.  N >= 0.
!>

A

!>          A هو مصفوفة REAL، البعد (LDA,N)
!>          عند الإدخال، المصفوفة المتناظرة A.  إذا كان UPLO = 'U'، فإن الجزء
!>          المثلثي العلوي n-by-n من A يحتوي على الجزء المثلثي العلوي
!>          للمصفوفة A، والجزء السفلي الصارم للمصفوفة A لا يتم الإشارة إليه.
!>          إذا كان UPLO = 'L'، فإن الجزء المثلثي السفلي n-by-n من A يحتوي على
!>          الجزء المثلثي السفلي للمصفوفة A، والجزء العلوي الصارم للمصفوفة A لا
!>          يتم الإشارة إليه.
!>          عند الخروج، إذا كان UPLO = 'U'، فإن الأقطار والقطر العلوي الأول
!>          للمصفوفة A يتم استبدالها بالعناصر المقابلة للمصفوفة ثلاثية الأقطار T،
!>          والعناصر فوق القطر العلوي الأول، مع المصفوفة TAU، تمثل المصفوفة
!>          المتعامدة Q كناتج للانعكاسات الابتدائية؛ إذا كان UPLO = 'L'، فإن
!>          الأقطار والقطر السفلي الأول للمصفوفة A يتم استبدالها بالعناصر
!>          المقابلة للمصفوفة ثلاثية الأقطار T، والعناصر تحت القطر السفلي الأول،
!>          مع المصفوفة TAU، تمثل المصفوفة المتع
!>          اددة Q كناتج للانعكاسات الابتدائية. انظر التفاصيل الإضافية.
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرائد للمصفوفة A.  LDA >= max(1,N).
!>

D

!>          D هي مصفوفة REAL، البعد (N)
!>          عناصر الأقطار للمصفوفة ثلاثية الأقطار T:
!>          D(i) = A(i,i).
!>

E

!>          E هي مصفوفة REAL، البعد (N-1)
!>          عناصر خارج القطر للمصفوفة ثلاثية الأقطار T:
!>          E(i) = A(i,i+1) إذا كان UPLO = 'U'، E(i) = A(i+1,i) إذا كان UPLO = 'L'.
!>

TAU

!>          TAU هي مصفوفة REAL، البعد (N-1)
!>          العوامل القياسية للانعكاسات الابتدائية (انظر التفاصيل الإضافية).
!>

INFO

!>          INFO هو INTEGER
!>          = 0:  خروج ناجح
!>          < 0:  إذا كان INFO = -i، فإن المعامل i كان له قيمة غير قانونية.
!>

تفاصيل إضافية:

!>
!>  إذا كان UPLO = 'U'، يتم تمثيل المصفوفة Q كناتج للانعكاسات الابتدائية
!>
!>     Q = H(n-1) . . . H(2) H(1).
!>
!>  كل H(i) له الصيغة
!>
!>     H(i) = I - tau * v * v**T
!>
!>  حيث tau هو رقم قياسي حقيقي، و v هو متجه حقيقي مع
!>  v(i+1:n) = 0 و v(i) = 1؛ v(1:i-1) مخزن عند الخروج في
!>  A(1:i-1,i+1)، و tau في TAU(i).
!>
!>  إذا كان UPLO = 'L'، يتم تمثيل المصفوفة Q كناتج للانعكاسات الابتدائية
!>
!>     Q = H(1) H(2) . . . H(n-1).
!>
!>  كل H(i) له الصيغة
!>
!>     H(i) = I - tau * v * v**T
!>
!>  حيث tau هو رقم قياسي حقيقي، و v هو متجه حقيقي مع
!>  v(1:i) = 0 و v(i+1) = 1؛ v(i+2:n) مخزن عند الخروج في A(i+2:n,i)،
!>  و tau في TAU(i).
!>
!>  محتويات A عند الخروج توضحها الأمثلة التالية
!>  مع n = 5:
!>
!>  إذا كان UPLO = 'U':                       إذا كان UPLO = 'L':
!>
!>    (  d   e   v2  v3  v4 )              (  d                  )
!>    (      d   e   v3  v4 )              (  e   d              )
!>    (          d   e   v4 )              (  v1  e   d          )
!>    (              d   e  )              (  v1  v2  e   d      )
!>    (                  d  )              (  v1  v2  v3  e   d  )
!>
!>  حيث d و e يمثلان عناصر الأقطار وخارج القطر لـ T، و vi
!>  يمثل عنصرًا في المتجه الذي يحدد H(i).
!>
