توثيق الوظيفة/الروتين الفرعي

subroutine ssyev_2stage (character jobz, character uplo, integer n, real, dimension( lda, * ) a, integer lda, real, dimension( * ) w, real, dimension( * ) work, integer lwork, integer info)

SSYEV_2STAGE يحسب القيم الذاتية، واختياريًا، المتجهات الذاتية اليسرى و/أو اليمنى لمصفوفات SY

الغرض:

!>
!> SSYEV_2STAGE يحسب جميع القيم الذاتية، واختياريًا، المتجهات الذاتية لمصفوفة A المتماثلة الحقيقية
!> باستخدام تقنية 2stage للاختزال إلى المصفوفة الثلاثية القطرية.
!>

المعلمات

JOBZ 
!>          JOBZ هو CHARACTER*1
!>          = 'N':  حساب القيم الذاتية فقط؛
!>          = 'V':  حساب القيم الذاتية والمتجهات الذاتية.
!>                  غير متوفر في هذا الإصدار.
!>

UPLO

!>          UPLO هو CHARACTER*1
!>          = 'U':  المثلث العلوي للمصفوفة A مخزن؛
!>          = 'L':  المثلث السفلي للمصفوفة A مخزن.
!>

N

!>          N هو INTEGER
!>          رتبة المصفوفة A.  N >= 0.
!>

A

!>          A هي مصفوفة REAL، أبعادها (LDA, N)
!>          عند الإدخال، المصفوفة المتماثلة A.  إذا كان UPLO = 'U'، فإن
!>          الجزء المثلثي العلوي من A ذو الأبعاد N × N يحتوي على
!>          الجزء المثلثي العلوي للمصفوفة A.  إذا كان UPLO = 'L'، فإن
!>          الجزء المثلثي السفلي من A ذو الأبعاد N × N يحتوي على
!>          الجزء المثلثي السفلي للمصفوفة A.
!>          عند الخروج، إذا كان JOBZ = 'V'، وإذا كان INFO = 0، فإن A تحتوي على
!>          المتجهات الذاتية المتعامدة للمصفوفة A.
!>          إذا كان JOBZ = 'N'، فإن عند الخروج المثلث السفلي (إذا كان UPLO='L')
!>          أو المثلث العلوي (إذا كان UPLO='U') من A، بما في ذلك
!>          القطر، يتم تدميره.
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرئيسي للمصفوفة A.  LDA >= max(1,N).
!>

W

!>          W هي مصفوفة REAL، أبعادها (N)
!>          إذا كان INFO = 0، فإن القيم الذاتية بترتيب تصاعدي.
!>

WORK

!>          WORK هي مصفوفة REAL، أبعادها LWORK
!>          عند الخروج، إذا كان INFO = 0، فإن WORK(1) تعيد قيمة LWORK المثلى.
!>

LWORK

!>          LWORK هو INTEGER
!>          طول المصفوفة WORK. LWORK >= 1، عندما N <= 1؛
!>          بخلاف ذلك
!>          إذا كان JOBZ = 'N' و N > 1، يجب الاستعلام عن LWORK.
!>                                   LWORK = MAX(1, dimension) حيث
!>                                   dimension = max(stage1,stage2) + (KD+1)*N + 2*N
!>                                             = N*KD + N*max(KD+1,FACTOPTNB)
!>                                               + max(2*KD*KD, KD*NTHREADS)
!>                                               + (KD+1)*N + 2*N
!>                                   حيث KD هو حجم التجميع للاختزال،
!>                                   FACTOPTNB هو التجميع المستخدم بواسطة خوارزمية QR أو LQ
!>                                   عادةً ما يكون FACTOPTNB=128 خيارًا جيدًا
!>                                   NTHREADS هو عدد الخيوط المستخدمة عند
!>                                   تم تمكين الترجمة البرمجية openMP، وإلا =1.
!>          إذا كان JOBZ = 'V' و N > 1، يجب الاستعلام عن LWORK. غير متوفر بعد
!>
!>          إذا كان LWORK = -1، فسيتم افتراض استعلام مساحة العمل؛ الروتين
!>          يحسب فقط الحجم الأمثل للمصفوفة WORK، ويعيد
!>          هذه القيمة كأول إدخال في المصفوفة WORK، ولا يصدر أي رسالة خطأ
!>          متعلقة بـ LWORK بواسطة XERBLA.
!>

INFO

!>          INFO هو INTEGER
!>          = 0:  انتهاء ناجح
!>          < 0:  إذا كان INFO = -i، فإن الوسيط i كان لديه قيمة غير قانونية
!>          > 0:  إذا كان INFO = i، فشلت الخوارزمية في التقارب؛ i
!>                العناصر خارج القطر لمصفوفة ثلاثية قطرية وسيطة
!>                لم تتقارب إلى الصفر.
!>
