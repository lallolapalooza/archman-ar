توثيق الوظيفة/الروتين الفرعي

subroutine ssyt01_3 (character uplo, integer n, real, dimension( lda, * ) a, integer lda, real, dimension( ldafac, * ) afac, integer ldafac, real, dimension( * ) e, integer, dimension( * ) ipiv, real, dimension( ldc, * ) c, integer ldc, real, dimension( * ) rwork, real resid)

SSYT01_3

الغرض:

!>
!> SSYT01_3 يعيد بناء مصفوفة متماثلة غير محددة A من عامل التجزئة الخاص بها
!> L*D*L' أو U*D*U' والذي تم حسابه بواسطة SSYTRF_RK
!> (أو SSYTRF_BK) ويحسب الباقي
!>    norm( C - A ) / ( N * norm(A) * EPS ),
!> حيث C هي المصفوفة المعاد بناؤها و EPS هو ثابت الآلة.
!>

المعلمات

UPLO 
!>          UPLO هو CHARACTER*1
!>          يحدد ما إذا كان الجزء المثلثي العلوي أو السفلي من المصفوفة
!>          المتماثلة A مخزنًا:
!>          = 'U':  مثلثي علوي
!>          = 'L':  مثلثي سفلي
!>

N

!>          N هو INTEGER
!>          عدد الصفوف والأعمدة في المصفوفة A.  N >= 0.
!>

A

!>          A هو مصفوفة DOUBLE PRECISION، بأبعاد (LDA,N)
!>          المصفوفة المتماثلة الأصلية A.
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرائد للمصفوفة A.  LDA >= max(1,N)
!>

AFAC

!>          AFAC هو مصفوفة DOUBLE PRECISION، بأبعاد (LDAFAC,N)
!>          القطر للمصفوفة القطرية المتعددة الكتل D والعوامل U أو L
!>          كما تم حسابها بواسطة SSYTRF_RK و SSYTRF_BK:
!>            أ) عناصر القطر فقط للمصفوفة المتماثلة القطرية المتعددة الكتل
!>               D على قطر A، أي D(k,k) = A(k,k)؛
!>               (يجب توفير عناصر ما فوق القطر (أو تحت القطر) لـ D
!>                عند الإدخال في المصفوفة E)، و
!>            ب) إذا كان UPLO = 'U': العامل U في الجزء فوق القطري لـ A.
!>               إذا كان UPLO = 'L': العامل L في الجزء تحت القطري لـ A.
!>

LDAFAC

!>          LDAFAC هو INTEGER
!>          البعد الرائد للمصفوفة AFAC.
!>          LDAFAC >= max(1,N).
!>

E

!>          E هي مصفوفة DOUBLE PRECISION، بأبعاد (N)
!>          عند الإدخال، تحتوي على عناصر ما فوق القطر (أو تحت القطر)
!>          للمصفوفة المتماثلة القطرية المتعددة الكتل D
!>          مع كتل قطرية بحجم 1x1 أو 2x2، حيث
!>          إذا كان UPLO = 'U': E(i) = D(i-1,i)، i=2:N، E(1) غير مستخدمة؛
!>          إذا كان UPLO = 'L': E(i) = D(i+1,i)، i=1:N-1، E(N) غير مستخدمة.
!>

IPIV

!>          IPIV هي مصفوفة INTEGER، بأبعاد (N)
!>          فهارس المحور من SSYTRF_RK (أو SSYTRF_BK).
!>

C

!>          C هي مصفوفة DOUBLE PRECISION، بأبعاد (LDC,N)
!>

LDC

!>          LDC هو INTEGER
!>          البعد الرائد للمصفوفة C.  LDC >= max(1,N).
!>

RWORK

!>          RWORK هي مصفوفة DOUBLE PRECISION، بأبعاد (N)
!>

RESID

!>          RESID هو DOUBLE PRECISION
!>          إذا كان UPLO = 'L'، norm(L*D*L' - A) / ( N * norm(A) * EPS )
!>          إذا كان UPLO = 'U'، norm(U*D*U' - A) / ( N * norm(A) * EPS )
!>
