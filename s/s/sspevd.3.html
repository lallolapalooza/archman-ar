subroutine sspevd (jobz, uplo, n, ap, w, z, ldz, work, lwork, iwork, liwork, info)
SSPEVD تحسب القيم الذاتية، اختياريًا، المتجهات الذاتية اليسرى و/أو اليمنى لمصفوفات أخرى

توثيق الوظيفة/الروتين الفرعي

subroutine sspevd (character jobz, character uplo, integer n, real, dimension( * ) ap, real, dimension( * ) w, real, dimension( ldz, * ) z, integer ldz, real, dimension( * ) work, integer lwork, integer, dimension( * ) iwork, integer liwork, integer info)

SSPEVD تحسب القيم الذاتية، اختياريًا، المتجهات الذاتية اليسرى و/أو اليمنى لمصفوفات أخرى

الغرض:

!>
!> SSPEVD تحسب جميع القيم الذاتية، اختياريًا، والمتجهات الذاتية
!> لمصفوفة متناظرة حقيقية A في تخزين مضغوط. إذا كانت المتجهات الذاتية مطلوبة،
!> فإنها تستخدم خوارزمية الانقسام والغزو.
!>
!>

المعاملات

JOBZ 
!>          JOBZ هو CHARACTER*1
!>          = 'N':  حساب القيم الذاتية فقط؛
!>          = 'V':  حساب القيم الذاتية والمتجهات الذاتية.
!>

UPLO

!>          UPLO هو CHARACTER*1
!>          = 'U':  المثلث العلوي من A مخزن؛
!>          = 'L':  المثلث السفلي من A مخزن.
!>

N

!>          N هو INTEGER
!>          رتبة المصفوفة A.  N >= 0.
!>

AP

!>          AP هو مصفوفة REAL، حجمها (N*(N+1)/2)
!>          عند الإدخال، المثلث العلوي أو السفلي للمصفوفة المتناظرة
!>          A، مضغوطة حسب الأعمدة في مصفوفة خطية. العمود j من A
!>          مخزن في المصفوفة AP كالتالي:
!>          إذا كان UPLO = 'U'، فإن AP(i + (j-1)*j/2) = A(i,j) لـ 1<=i<=j؛
!>          إذا كان UPLO = 'L'، فإن AP(i + (j-1)*(2*n-j)/2) = A(i,j) لـ j<=i<=n.
!>
!>          عند الخروج، يتم استبدال AP بقيم تم إنشاؤها أثناء
!>          الاختزال إلى شكل ثلاثي الأقطار. إذا كان UPLO = 'U'، فإن القطر
!>          والقطر الرئيسي العلوي للمصفوفة ثلاثية الأقطار T يستبدلان
!>          العناصر المقابلة في A، وإذا كان UPLO = 'L'، فإن
!>          القطر والعناصر الفرعية الأولى لـ T تستبدل
!>          العناصر المقابلة في A.
!>

W

!>          W هو مصفوفة REAL، حجمها (N)
!>          إذا كان INFO = 0، فإن القيم الذاتية مرتبة تصاعدياً.
!>

Z

!>          Z هو مصفوفة REAL، حجمها (LDZ, N)
!>          إذا كان JOBZ = 'V'، فعندما يكون INFO = 0، تحتوي Z على المتجهات الذاتية
!>          العمودية للمصفوفة A، حيث يحمل العمود i من Z
!>          المتجه الذاتي المرتبط بـ W(i).
!>          إذا كان JOBZ = 'N'، فلا يتم استخدام Z.
!>

LDZ

!>          LDZ هو INTEGER
!>          البعد الرئيسي للمصفوفة Z.  LDZ >= 1، وإذا كان
!>          JOBZ = 'V'، فإن LDZ >= max(1,N).
!>

WORK

!>          WORK هو مصفوفة REAL، حجمها (MAX(1,LWORK))
!>          عند الخروج، إذا كان INFO = 0، فإن WORK(1) يعيد LWORK المطلوبة.
!>

LWORK

!>          LWORK هو INTEGER
!>          حجم المصفوفة WORK.
!>          إذا كان N <= 1،               يجب أن تكون LWORK على الأقل 1.
!>          إذا كان JOBZ = 'N' و N > 1، يجب أن تكون LWORK على الأقل 2*N.
!>          إذا كان JOBZ = 'V' و N > 1، يجب أن تكون LWORK على الأقل
!>                                                 1 + 6*N + N**2.
!>
!>          إذا كان LWORK = -1، فيفترض استعلام عن مساحة العمل؛ الروتين
!>          يحسب فقط الأحجام المطلوبة للمصفوفتين WORK و IWORK،
!>          يعيد هذه القيم كأول إدخالات في المصفوفتين WORK
!>          و IWORK، ولا يتم إصدار أي رسالة خطأ متعلقة بـ LWORK أو
!>          LIWORK بواسطة XERBLA.
!>

IWORK

!>          IWORK هو مصفوفة INTEGER، حجمها (MAX(1,LIWORK))
!>          عند الخروج، إذا كان INFO = 0، فإن IWORK(1) يعيد LIWORK المطلوبة.
!>

LIWORK

!>          LIWORK هو INTEGER
!>          حجم المصفوفة IWORK.
!>          إذا كان JOBZ  = 'N' أو N <= 1، يجب أن تكون LIWORK على الأقل 1.
!>          إذا كان JOBZ  = 'V' و N > 1، يجب أن تكون LIWORK على الأقل 3 + 5*N.
!>
!>          إذا كان LIWORK = -1، فيفترض استعلام عن مساحة العمل؛ الروتين
!>          يحسب فقط الأحجام المطلوبة للمصفوفتين WORK و IWORK،
!>          يعيد هذه القيم كأول إدخالات في المصفوفتين WORK
!>          و IWORK، ولا يتم إصدار أي رسالة خطأ متعلقة بـ LWORK أو
!>          LIWORK بواسطة XERBLA.
!>

INFO

!>          INFO هو INTEGER
!>          = 0:  اكتمال ناجح
!>          < 0:  إذا كان INFO = -i، فإن المعامل i كان له قيمة غير قانونية.
!>          > 0:  إذا كان INFO = i، فشلت الخوارزمية في التقارب؛ i
!>                عناصر خارج القطر لمصفوفة ثلاثية الأقطار وسيطة
!>                لم تتقارب إلى الصفر.
!>
