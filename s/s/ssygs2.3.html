subroutine ssygs2 (integer itype, character uplo, integer n, real, dimension( lda, * ) a, integer lda, real, dimension( ldb, * ) b, integer ldb, integer info)

SSYGS2 تختزل مشكلة قيمة خاصة معمم متماثل حقيقي محدد إلى شكل قياسي، باستخدام نتائج التحليل التي تم الحصول عليها من spotrf (خوارزمية غير محظورة).

الغرض:

!>
!> SSYGS2 تختزل مشكلة قيمة خاصة معمم متماثل حقيقي محدد
!> إلى شكل قياسي.
!>
!> إذا كان ITYPE = 1، فإن المشكلة هي A*x = lambda*B*x،
!> ويتم الكتابة فوق A بـ inv(U**T)*A*inv(U) أو inv(L)*A*inv(L**T)
!>
!> إذا كان ITYPE = 2 أو 3، فإن المشكلة هي A*B*x = lambda*x أو
!> B*A*x = lambda*x، ويتم الكتابة فوق A بـ U*A*U**T أو L**T *A*L.
!>
!> يجب أن يكون B قد تم تحليله مسبقًا كـ U**T *U أو L*L**T بواسطة SPOTRF.
!>

المعلمات

ITYPE 
!>          ITYPE هو عدد صحيح (INTEGER)
!>          = 1: حساب inv(U**T)*A*inv(U) أو inv(L)*A*inv(L**T);
!>          = 2 أو 3: حساب U*A*U**T أو L**T *A*L.
!>

UPLO

!>          UPLO هو حرف (CHARACTER*1)
!>          يحدد ما إذا كان الجزء العلوي أو السفلي المثلثي للمصفوفة المتماثلة A
!>          مخزنًا، وكيف تم تحليل B.
!>          = 'U':  المثلث العلوي
!>          = 'L':  المثلث السفلي
!>

N

!>          N هو عدد صحيح (INTEGER)
!>          رتبة المصفوفتين A و B.  N >= 0.
!>

A

!>          A هي مصفوفة حقيقية (REAL)، أبعادها (LDA,N)
!>          عند الإدخال، المصفوفة المتماثلة A.  إذا كان UPLO = 'U'، فإن الجزء
!>          العلوي المثلثي بحجم n × n في A يحتوي على الجزء المثلثي العلوي
!>          للمصفوفة A، والجزء السفلي المثلثي الصارم لـ A لا يتم الإشارة إليه.
!>          إذا كان UPLO = 'L'، فإن الجزء السفلي المثلثي بحجم n × n في A يحتوي
!>          على الجزء المثلثي السفلي للمصفوفة A، والجزء العلوي المثلثي الصارم
!>          لـ A لا يتم الإشارة إليه.
!>
!>          عند الخروج، إذا كان INFO = 0، فإن المصفوفة المحولة، المخزنة بنفس
!>          تنسيق A.
!>

LDA

!>          LDA هو عدد صحيح (INTEGER)
!>          البعد الرئيسي للمصفوفة A.  LDA >= max(1,N).
!>

B

!>          B هي مصفوفة حقيقية (REAL)، أبعادها (LDB,N)
!>          العامل المثلثي من التحليل Cholesky لـ B،
!>          كما تم إرجاعه بواسطة SPOTRF.
!>

LDB

!>          LDB هو عدد صحيح (INTEGER)
!>          البعد الرئيسي للمصفوفة B.  LDB >= max(1,N).
!>

INFO

!>          INFO هو عدد صحيح (INTEGER)
!>          = 0:  خروج ناجح.
!>          < 0:  إذا كان INFO = -i، فإن الوسيط i كان له قيمة غير قانونية.
!>
