subroutine ssbevx_2stage (character jobz, character range, character uplo, integer n, integer kd, real, dimension( ldab, * ) ab, integer ldab, real, dimension( ldq, * ) q, integer ldq, real vl, real vu, integer il, integer iu, real abstol, integer m, real, dimension( * ) w, real, dimension( ldz, * ) z, integer ldz, real, dimension( * ) work, integer lwork, integer, dimension( * ) iwork, integer, dimension( * ) ifail, integer info)

SSBEVX_2STAGE تحسب القيم الذاتية المختارة، وبشكل اختياري، المتجهات الذاتية لمصفوفة نطاق متماثلة حقيقية A باستخدام تقنية المرحلتين (2stage) للاختزال إلى مصفوفة ثلاثية الأقطار. يمكن اختيار القيم الذاتية والمتجهات الذاتية عن طريق تحديد نطاق من القيم أو نطاق من المؤشرات للقيم الذاتية المطلوبة.

الغرض:

!>
!> SSBEVX_2STAGE تحسب القيم الذاتية المختارة، وبشكل اختياري، المتجهات الذاتية
!> لمصفوفة نطاق متماثلة حقيقية A باستخدام تقنية المرحلتين (2stage) للاختزال إلى مصفوفة ثلاثية الأقطار. يمكن اختيار القيم الذاتية والمتجهات الذاتية
!> عن طريق تحديد نطاق من القيم أو نطاق من المؤشرات للقيم الذاتية المطلوبة.
!>

المعاملات

JOBZ 
!>          JOBZ is CHARACTER*1
!>          = 'N':  احسب القيم الذاتية فقط؛
!>          = 'V':  احسب القيم الذاتية والمتجهات الذاتية.
!>                  غير متاحة في هذا الإصدار.
!>

RANGE

!>          RANGE is CHARACTER*1
!>          = 'A': سيتم العثور على جميع القيم الذاتية؛
!>          = 'V': سيتم العثور على جميع القيم الذاتية في الفترة نصف المفتوحة (VL,VU]
!>                 ؛
!>          = 'I': سيتم العثور على القيم الذاتية من IL إلى IU.
!>

UPLO

!>          UPLO is CHARACTER*1
!>          = 'U':  يتم تخزين المثلث العلوي للمصفوفة A؛
!>          = 'L':  يتم تخزين المثلث السفلي للمصفوفة A.
!>

N

!>          N is INTEGER
!>          رتبة المصفوفة A.  N >= 0.
!>

KD

!>          KD is INTEGER
!>          عدد النطاقات العلوية للمصفوفة A إذا كانت UPLO = 'U'،
!>          أو عدد النطاقات السفلية إذا كانت UPLO = 'L'.  KD >= 0.
!>

AB

!>          AB is REAL array, dimension (LDAB, N)
!>          عند الدخول، يتم تخزين المثلث العلوي أو السفلي للمصفوفة النطاقية المتماثلة
!>          A في أول KD+1 صفوف للمصفوفة. يتم تخزين العمود j للمصفوفة A
!>          في العمود j للمصفوفة AB
!>          كالتالي:
!>          إذا كانت UPLO = 'U'، فإن AB(kd+1+i-j,j) = A(i,j) حيث max(1,j-kd)<=i<=j؛
!>          إذا كانت UPLO = 'L'، فإن AB(1+i-j,j)    = A(i,j) حيث j<=i<=min(n,j+kd).
!>
!>          عند الخروج، يتم استبدال AB بقيم تم إنشاؤها أثناء
!>          الاختزال إلى شكل ثلاثي الأقطار. إذا كانت UPLO = 'U'، فإن
!>          النطاق العلوي الأول والقطر الرئيسي للمصفوفة ثلاثية الأقطار T
!>          يتم إرجاعها في الصفين KD و KD+1 للمصفوفة AB، وإذا كانت UPLO = 'L'،
!>          يتم إرجاع القطر الرئيسي والنطاق الفرعي الأول لـ T في
!>          أول صفين من AB.
!>

LDAB

!>          LDAB is INTEGER
!>          البعد الرئيسي للمصفوفة AB.  LDAB >= KD + 1.
!>

Q

!>          Q is REAL array, dimension (LDQ, N)
!>          إذا كانت JOBZ = 'V'، فإن المصفوفة المتعامدة N×N المستخدمة في
!>                         الاختزال إلى شكل ثلاثي الأقطار.
!>          إذا كانت JOBZ = 'N'، فلا يتم الرجوع إلى المصفوفة Q.
!>

LDQ

!>          LDQ is INTEGER
!>          البعد الرئيسي للمصفوفة Q.  إذا كانت JOBZ = 'V'، فإن
!>          LDQ >= max(1,N).
!>

VL

!>          VL is REAL
!>          إذا كانت RANGE = 'V'، فإن الحد الأدنى للفترة
!>          التي سيتم البحث فيها عن القيم الذاتية. VL < VU.
!>          لا يتم الرجوع إليها إذا كانت RANGE = 'A' أو 'I'.
!>

VU

!>          VU is REAL
!>          إذا كانت RANGE = 'V'، فإن الحد الأعلى للفترة
!>          التي سيتم البحث فيها عن القيم الذاتية. VL < VU.
!>          لا يتم الرجوع إليها إذا كانت RANGE = 'A' أو 'I'.
!>

IL

!>          IL is INTEGER
!>          إذا كانت RANGE = 'I'، فإن مؤشر
!>          أصغر قيمة ذاتية سيتم إرجاعها.
!>          1 <= IL <= IU <= N، إذا كانت N > 0؛ IL = 1 و IU = 0 إذا كانت N = 0.
!>          لا يتم الرجوع إليها إذا كانت RANGE = 'A' أو 'V'.
!>

IU

!>          IU is INTEGER
!>          إذا كانت RANGE = 'I'، فإن مؤشر
!>          أكبر قيمة ذاتية سيتم إرجاعها.
!>          1 <= IL <= IU <= N، إذا كانت N > 0؛ IL = 1 و IU = 0 إذا كانت N = 0.
!>          لا يتم الرجوع إليها إذا كانت RANGE = 'A' أو 'V'.
!>

ABSTOL

!>          ABSTOL is REAL
!>          التفاوت المطلق للأخطاء في القيم الذاتية.
!>          يتم قبول قيمة ذاتية تقريبية على أنها متقاربة
!>          عندما يتم تحديد أنها تقع في فترة [a,b]
!>          بعرض يساوي أو أقل من
!>
!>                  ABSTOL + EPS *   max( |a|,|b| ) ,
!>
!>          حيث EPS هي دقة الآلة. إذا كانت ABSTOL أقل من
!>          أو تساوي الصفر، فسيتم استخدام EPS*|T| بدلاً منها،
!>          حيث |T| هو معيار النطاق 1 للمصفوفة ثلاثية الأقطار التي تم الحصول عليها
!>          عن طريق اختزال AB إلى شكل ثلاثي الأقطار.
!>
!>          سيتم حساب القيم الذاتية بأكبر دقة عندما تكون ABSTOL
!>          مضبوطة على ضعف حد التدفق السفلي 2*SLAMCH('S')، وليس صفرًا.
!>          إذا عادت هذه الوظيفة بمعلومة INFO > 0، مما يشير إلى أن بعض
!>          المتجهات الذاتية لم تتقارب، حاول ضبط ABSTOL على
!>          2*SLAMCH('S').
!>
!>          انظر  بواسطة Demmel و
!>          Kahan، مذكرة عمل LAPACK #3.
!>

M

!>          M is INTEGER
!>          إجمالي عدد القيم الذاتية التي تم العثور عليها.  0 <= M <= N.
!>          إذا كانت RANGE = 'A'، فإن M = N، وإذا كانت RANGE = 'I'، فإن M = IU-IL+1.
!>

W

!>          W is REAL array, dimension (N)
!>          تحتوي العناصر M الأولى على القيم الذاتية المختارة بترتيب تصاعدي.
!>

Z

!>          Z is REAL array, dimension (LDZ, max(1,M))
!>          إذا كانت JOBZ = 'V'، فإذا كانت INFO = 0، فإن الأعمدة M الأولى من Z
!>          تحتوي على المتجهات الذاتية المتعامدة للمصفوفة A
!>          المقابلة للقيم الذاتية المختارة، مع احتواء العمود i من Z
!>          على المتجه الذاتي المرتبط بـ W(i).
!>          إذا فشل متجه ذاتي في التقارب، فإن العمود المقابل في Z
!>          يحتوي على آخر تقريب للمتجه الذاتي، وسيتم إرجاع مؤشر
!>          المتجه الذاتي في IFAIL.
!>          إذا كانت JOBZ = 'N'، فلا يتم الرجوع إلى Z.
!>          ملاحظة: يجب على المستخدم التأكد من توفير ما لا يقل عن max(1,M) عمود في
!>          المصفوفة Z؛ إذا كانت RANGE = 'V'، فإن القيمة الدقيقة لـ M
!>          غير معروفة مسبقًا ويجب استخدام حد أعلى.
!>

LDZ

!>          LDZ is INTEGER
!>          البعد الرئيسي للمصفوفة Z.  LDZ >= 1، وإذا كانت
!>          JOBZ = 'V'، فإن LDZ >= max(1,N).
!>

WORK

!>          WORK is REAL array, dimension (LWORK)
!>

LWORK

!>          LWORK is INTEGER
!>          طول المصفوفة WORK. LWORK >= 1، عندما تكون N <= 1؛
!>          وإلا
!>          إذا كانت JOBZ = 'N' و N > 1، يجب الاستعلام عن LWORK.
!>                                   LWORK = MAX(1, 7*N, dimension) حيث
!>                                   dimension = (2KD+1)*N + KD*NTHREADS + 2*N
!>                                   حيث KD هو حجم النطاق.
!>

IWORK

!>

IFAIL

!>          IFAIL هي مصفوفة INTEGER، البعد (N)
!>          إذا كان JOBZ = 'V'، فعندما يكون INFO = 0، فإن العناصر M الأولى في
!>          IFAIL تكون صفرًا. إذا كان INFO > 0، فإن IFAIL يحتوي على
!>          فهارس المتجهات الذاتية التي فشلت في التقارب.
!>          إذا كان JOBZ = 'N'، فلن يتم الرجوع إلى IFAIL.
!>

INFO

!>          INFO هو INTEGER
!>          = 0:  خروج ناجح.
!>          < 0:  إذا كان INFO = -i، فإن الوسيط i كان له قيمة غير قانونية.
!>          > 0:  إذا كان INFO = i، فإن i من المتجهات الذاتية فشلت في التقارب.
!>                تم تخزين فهارسها في المصفوفة IFAIL.
!>
