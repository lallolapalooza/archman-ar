subroutine sstevd (character jobz, integer n, real, dimension( * ) d, real, dimension( * ) e, real, dimension( ldz, * ) z, integer ldz, real, dimension( * ) work, integer lwork, integer, dimension( * ) iwork, integer liwork, integer info)

SSTEVD يحسب القيم الذاتية، واختياريًا، المتجهات الذاتية اليسرى و/أو اليمنى لمصفوفات أخرى

الغرض:

!>
!> SSTEVD يحسب جميع القيم الذاتية، واختياريًا، المتجهات الذاتية لمصفوفة ثلاثية قطرية متماثلة حقيقية. إذا كانت المتجهات الذاتية مطلوبة، فإنها
!> تستخدم خوارزمية تقسيم واستحواذ.
!>
!>

المعلمات

JOBZ 
!>          JOBZ هو CHARACTER*1
!>          = 'N':  حساب القيم الذاتية فقط؛
!>          = 'V':  حساب القيم الذاتية والمتجهات الذاتية.
!>

N

!>          N هو INTEGER
!>          ترتيب المصفوفة. N >= 0.
!>

D

!>          D هو REAL array، البعد (N)
!>          عند الإدخال، العناصر القطرية n للمصفوفة ثلاثية القطرية
!>          A.
!>          عند الخروج، إذا كان INFO = 0، فإن القيم الذاتية مرتبة تصاعديًا.
!>

E

!>          E هو REAL array، البعد (N-1)
!>          عند الإدخال، العناصر الفرعية القطرية (n-1) للمصفوفة ثلاثية القطرية
!>          A، مخزنة في العناصر 1 إلى N-1 من E.
!>          عند الخروج، يتم تدمير محتويات E.
!>

Z

!>          Z هو REAL array، البعد (LDZ, N)
!>          إذا كان JOBZ = 'V'، فعندما يكون INFO = 0، تحتوي Z على المصفوفة المتعامدة
!>          المتجهات الذاتية للمصفوفة A، مع احتواء العمود i من Z
!>          على المتجه الذاتي المرتبط بـ D(i).
!>          إذا كان JOBZ = 'N'، فلن يتم الرجوع إلى Z.
!>

LDZ

!>          LDZ هو INTEGER
!>          البعد الرائد للمصفوفة Z. LDZ >= 1، وإذا كان
!>          JOBZ = 'V'، فإن LDZ >= max(1,N).
!>

WORK

!>          WORK هو REAL array،
!>                                         البعد (LWORK)
!>          عند الخروج، إذا كان INFO = 0، فإن WORK(1) يعيد LWORK الأمثل.
!>

LWORK

!>          LWORK هو INTEGER
!>          بعد المصفوفة WORK.
!>          إذا كان JOBZ  = 'N' أو N <= 1، فيجب أن يكون LWORK على الأقل 1.
!>          إذا كان JOBZ  = 'V' و N > 1، فيجب أن يكون LWORK على الأقل
!>                         ( 1 + 4*N + N**2 ).
!>
!>          إذا كان LWORK = -1، فسيتم افتراض استعلام عن مساحة العمل؛ الروتين
!>          يحسب فقط الأحجام المثلى لمصفوفتي WORK و IWORK،
!>          يعيد هذه القيم كأول إدخالات في مصفوفتي WORK و IWORK، ولا يتم
!>          إصدار رسالة خطأ تتعلق بـ LWORK أو LIWORK بواسطة XERBLA.
!>

IWORK

!>          IWORK هو INTEGER array، البعد (MAX(1,LIWORK))
!>          عند الخروج، إذا كان INFO = 0، فإن IWORK(1) يعيد LIWORK الأمثل.
!>

LIWORK

!>          LIWORK هو INTEGER
!>          بعد المصفوفة IWORK.
!>          إذا كان JOBZ  = 'N' أو N <= 1، فيجب أن يكون LIWORK على الأقل 1.
!>          إذا كان JOBZ  = 'V' و N > 1، فيجب أن يكون LIWORK على الأقل 3+5*N.
!>
!>          إذا كان LIWORK = -1، فسيتم افتراض استعلام عن مساحة العمل؛ الروتين
!>          يحسب فقط الأحجام المثلى لمصفوفتي WORK و IWORK،
!>          يعيد هذه القيم كأول إدخالات في مصفوفتي WORK و IWORK، ولا يتم
!>          إصدار رسالة خطأ تتعلق بـ LWORK أو LIWORK بواسطة XERBLA.
!>

INFO

!>          INFO هو INTEGER
!>          = 0:  اكتمال ناجح
!>          < 0:  إذا كان INFO = -i، فإن المعلمة i كانت لها قيمة غير قانونية
!>          > 0:  إذا كان INFO = i، فشلت الخوارزمية في التقارب؛ i
!>                عناصر خارج القطر من E لم تتقارب إلى الصفر.
!>
