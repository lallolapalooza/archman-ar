توثيق الوظيفة/الروتين الفرعي

subroutine ssyt01 (character uplo, integer n, real, dimension( lda, * ) a, integer lda, real, dimension( ldafac, * ) afac, integer ldafac, integer, dimension( * ) ipiv, real, dimension( ldc, * ) c, integer ldc, real, dimension( * ) rwork, real resid)

SSYT01

Purpose:

!>
!> SSYT01 يعيد بناء مصفوفة متناظرة غير محددة A من تحليلها
!> L*D*L' الكتلة أو U*D*U' ويحسب ثابت الخطأ
!>    norm( C - A ) / ( N * norm(A) * EPS ),
!> حيث C هي المصفوفة المعاد بناؤها و EPS هو الجزء العشري من الآلة.
!>

Parameters

UPLO 
!>          UPLO is CHARACTER*1
!>          يحدد ما إذا كان الجزء المثلثي العلوي أو السفلي للمصفوفة
!>          المتناظرة A مخزنًا:
!>          = 'U':  مثلثي علوي
!>          = 'L':  مثلثي سفلي
!>

N

!>          N is INTEGER
!>          عدد الصفوف والأعمدة للمصفوفة A.  N >= 0.
!>

A

!>          A is REAL array, dimension (LDA,N)
!>          المصفوفة المتناظرة الأصلية A.
!>

LDA

!>          LDA is INTEGER
!>          البعد الرائد للمصفوفة A.  LDA >= max(1,N)
!>

AFAC

!>          AFAC is REAL array, dimension (LDAFAC,N)
!>          الشكل المفكك للمصفوفة A.  AFAC يحتوي على المصفوفة
!>          القطرية الكتلة D والمضاعفات المستخدمة للحصول على
!>          العامل L أو U من تحليل L*D*L' أو U*D*U' الكتلة
!>          كما تم حسابه بواسطة SSYTRF.
!>

LDAFAC

!>          LDAFAC is INTEGER
!>          البعد الرائد للمصفوفة AFAC.  LDAFAC >= max(1,N).
!>

IPIV

!>          IPIV is INTEGER array, dimension (N)
!>          مؤشرات المحور من SSYTRF.
!>

C

!>          C is REAL array, dimension (LDC,N)
!>

LDC

!>          LDC is INTEGER
!>          البعد الرائد للمصفوفة C.  LDC >= max(1,N).
!>

RWORK

!>          RWORK is REAL array, dimension (N)
!>

RESID

!>          RESID is REAL
!>          If UPLO = 'L', norm(L*D*L' - A) / ( N * norm(A) * EPS )
!>          If UPLO = 'U', norm(U*D*U' - A) / ( N * norm(A) * EPS )
!>
