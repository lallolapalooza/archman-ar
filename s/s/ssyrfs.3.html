subroutine ssyrfs (character uplo, integer n, integer nrhs, real, dimension( lda, * ) a, integer lda, real, dimension( ldaf, * ) af, integer ldaf, integer, dimension( * ) ipiv, real, dimension( ldb, * ) b, integer ldb, real, dimension( ldx, * ) x, integer ldx, real, dimension( * ) ferr, real, dimension( * ) berr, real, dimension( * ) work, integer, dimension( * ) iwork, integer info)

SSYRFS

الغرض:

!>
!> SSYRFS يحسن الحل المحسوب لنظام المعادلات الخطية
!> عندما تكون مصفوفة المعاملات متماثلة وغير محددة، و
!> يوفر حدود الخطأ وتقديرات الخطأ الخلفي للحل.
!>

المعلمات

UPLO 
!>          UPLO is CHARACTER*1
!>          = 'U':  يتم تخزين المثلث العلوي من A؛
!>          = 'L':  يتم تخزين المثلث السفلي من A.
!>

N

!>          N is INTEGER
!>          ترتيب المصفوفة A.  N >= 0.
!>

NRHS

!>          NRHS is INTEGER
!>          عدد الأطراف اليمنى، أي عدد الأعمدة
!>          للمصفوفتين B و X.  NRHS >= 0.
!>

A

!>          A is REAL array, dimension (LDA,N)
!>          المصفوفة المتماثلة A.  إذا كان UPLO = 'U'، فإن الجزء العلوي
!>          الرئيسي N-by-N من A يحتوي على الجزء العلوي
!>          من المصفوفة A، ولا يتم الإشارة إلى الجزء السفلي الصارم من A.
!>          إذا كان UPLO = 'L'، فإن الجزء السفلي الرئيسي
!>          N-by-N من A يحتوي على الجزء السفلي
!>          من المصفوفة A، ولا يتم الإشارة إلى الجزء العلوي الصارم من A.
!>

LDA

!>          LDA is INTEGER
!>          البعد الرئيسي للمصفوفة A.  LDA >= max(1,N).
!>

AF

!>          AF is REAL array, dimension (LDAF,N)
!>          الشكل المكسور للمصفوفة A.  AF يحتوي على المصفوفة
!>          القطرية الفرعية D والمضاعفات المستخدمة للحصول على
!>          U أو L من تحليل A = U*D*U**T أو
!>          A = L*D*L**T كما تم حسابه بواسطة SSYTRF.
!>

LDAF

!>          LDAF is INTEGER
!>          البعد الرئيسي للمصفوفة AF.  LDAF >= max(1,N).
!>

IPIV

!>          IPIV is INTEGER array, dimension (N)
!>          تفاصيل التبادلات وهيكل D الفرعي
!>          كما تم تحديده بواسطة SSYTRF.
!>

B

!>          B is REAL array, dimension (LDB,NRHS)
!>          مصفوفة الطرف الأيمن B.
!>

LDB

!>          LDB is INTEGER
!>          البعد الرئيسي للمصفوفة B.  LDB >= max(1,N).
!>

X

!>          X is REAL array, dimension (LDX,NRHS)
!>          عند الإدخال، مصفوفة الحل X، كما تم حسابها بواسطة SSYTRS.
!>          عند الخروج، مصفوفة الحل المحسنة X.
!>

LDX

!>          LDX is INTEGER
!>          البعد الرئيسي للمصفوفة X.  LDX >= max(1,N).
!>

FERR

!>          FERR is REAL array, dimension (NRHS)
!>          تقدير الحد الأعلى لخطأ الخطأ الأمامي لكل متجه حل
!>          X(j) (العمود j للمصفوفة الحل X).
!>          إذا كان XTRUE هو الحل الحقيقي المقابل لـ X(j)، فإن FERR(j)
!>          هو حد أعلى مقدر لقيمة الجزء الأكبر
!>          من (X(j) - XTRUE) مقسومًا على قيمة الجزء الأكبر
!>          من X(j).  التقدير موثوق به مثل
!>          التقدير لـ RCOND، وهو دائمًا تقريبًا خطأ طفيف
!>          زائد عن الخطأ الحقيقي.
!>

BERR

!>          BERR is REAL array, dimension (NRHS)
!>          خطأ الخلفية النسبي الجزئي لكل متجه حل
!>          X(j) (أي، أصغر تغيير نسبي في
!>          أي عنصر من A أو B يجعل X(j) حلاً دقيقًا).
!>

WORK

!>          WORK is REAL array, dimension (3*N)
!>

IWORK

!>          IWORK is INTEGER array, dimension (N)
!>

INFO

!>          INFO is INTEGER
!>          = 0:  خروج ناجح
!>          < 0:  إذا كان INFO = -i، فإن المعامل i كان له قيمة غير قانونية
!>

المعلمات الداخلية:

!>
