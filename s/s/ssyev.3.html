subroutine ssyev (jobz, uplo, n, a, lda, w, work, lwork, info)
SSYEV تحسب القيم الذاتية، اختيارياً، المتجهات الذاتية اليسرى و/أو اليمنى لمصفوفات SY

توثيق الوظيفة/الروتين الفرعي

subroutine ssyev (character jobz, character uplo, integer n, real, dimension( lda, * ) a, integer lda, real, dimension( * ) w, real, dimension( * ) work, integer lwork, integer info)

SSYEV تحسب القيم الذاتية، اختيارياً، المتجهات الذاتية لمصفوفات SY

الغرض:

!>
!> SSYEV تحسب جميع القيم الذاتية، واختيارياً، المتجهات الذاتية لمصفوفة متناظرة حقيقية A.
!>

المعلمات

JOBZ 
!>          JOBZ هو CHARACTER*1
!>          = 'N':  حساب القيم الذاتية فقط؛
!>          = 'V':  حساب القيم الذاتية والمتجهات الذاتية.
!>

UPLO

!>          UPLO هو CHARACTER*1
!>          = 'U':  المثلث العلوي من A مخزن؛
!>          = 'L':  المثلث السفلي من A مخزن.
!>

N

!>          N هو INTEGER
!>          رتبة المصفوفة A.  N >= 0.
!>

A

!>          A هو مصفوفة REAL، ذات البعد (LDA, N)
!>          عند الإدخال، المصفوفة المتناظرة A.  إذا كان UPLO = 'U'، فإن
!>          الجزء المثلثي العلوي الرئيسي N-by-N من A يحتوي على
!>          الجزء المثلثي العلوي للمصفوفة A.  إذا كان UPLO = 'L'، فإن
!>          الجزء المثلثي السفلي الرئيسي N-by-N من A يحتوي على
!>          الجزء المثلثي السفلي للمصفوفة A.
!>          عند الخروج، إذا كان JOBZ = 'V'، وإذا كان INFO = 0، فإن A تحتوي على
!>          المتجهات الذاتية المتعامدة للمصفوفة A.
!>          إذا كان JOBZ = 'N'، فعند الخروج يتم تدمير المثلث السفلي (إذا كان UPLO='L')
!>          أو المثلث العلوي (إذا كان UPLO='U') من A، بما في ذلك
!>          القطر.
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرئيسي للمصفوفة A.  LDA >= max(1,N).
!>

W

!>          W هي مصفوفة REAL، ذات البعد (N)
!>          إذا كان INFO = 0، فإن القيم الذاتية مرتبة تصاعديًا.
!>

WORK

!>          WORK هي مصفوفة REAL، ذات البعد (MAX(1,LWORK))
!>          عند الخروج، إذا كان INFO = 0، فإن WORK(1) تعيد قيمة LWORK المثلى.
!>

LWORK

!>          LWORK هو INTEGER
!>          طول المصفوفة WORK.  LWORK >= max(1,3*N-1).
!>          للكفاءة المثلى، LWORK >= (NB+2)*N،
!>          حيث NB هو حجم الكتلة لـ SSYTRD الذي تم إرجاعه بواسطة ILAENV.
!>
!>          إذا كان LWORK = -1، فسيتم افتراض استعلام عن مساحة العمل؛ الروتين
!>          يحسب فقط الحجم الأمثل لمصفوفة WORK، ويعيد
!>          هذه القيمة كالإدخال الأول لمصفوفة WORK، ولا يتم إصدار أي رسالة خطأ
!>          متعلقة بـ LWORK بواسطة XERBLA.
!>

INFO

!>          INFO هو INTEGER
!>          = 0:  خروج ناجح
!>          < 0:  إذا كان INFO = -i، فإن الوسيط i كان له قيمة غير قانونية
!>          > 0:  إذا كان INFO = i، فشلت الخوارزمية في التقارب؛ i
!>                العناصر غير القطرية لمصفوفة ثلاثية وسيطة
!>                لم تتقارب إلى الصفر.
!>
