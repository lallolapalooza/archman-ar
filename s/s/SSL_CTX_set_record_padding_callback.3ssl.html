SSL_CTX_set_record_padding_callback(3ssl)

الحزمة: openssl

التراخيص: Apache-2.0

SSL_CTX_SET_RECORD_PADDING_CALLBACK(3ssl) OpenSSL SSL_CTX_SET_RECORD_PADDING_CALLBACK(3ssl)

الاسم

SSL_CTX_set_record_padding_callback, SSL_set_record_padding_callback, SSL_CTX_set_record_padding_callback_arg, SSL_set_record_padding_callback_arg, SSL_CTX_get_record_padding_callback_arg, SSL_get_record_padding_callback_arg, SSL_CTX_set_block_padding, SSL_CTX_set_block_padding_ex, SSL_set_block_padding, SSL_set_block_padding_ex - تثبيت دالة استدعاء لتحديد حشو سجل TLS 1.3

الملخص

#include <openssl/ssl.h>
void SSL_CTX_set_record_padding_callback(SSL_CTX *ctx, size_t (*cb)(SSL *s, int type, size_t len, void *arg));
int SSL_set_record_padding_callback(SSL *ssl, size_t (*cb)(SSL *s, int type, size_t len, void *arg));
void SSL_CTX_set_record_padding_callback_arg(SSL_CTX *ctx, void *arg);
void *SSL_CTX_get_record_padding_callback_arg(const SSL_CTX *ctx);
void SSL_set_record_padding_callback_arg(SSL *ssl, void *arg);
void *SSL_get_record_padding_callback_arg(const SSL *ssl);
int SSL_CTX_set_block_padding(SSL_CTX *ctx, size_t block_size);
int SSL_set_block_padding(SSL *ssl, size_t block_size);
int SSL_CTX_set_block_padding_ex(SSL_CTX *ctx, size_t app_block_size, size_t hs_block_size);
int SSL_set_block_padding_ex(SSL *ssl, size_t app_block_size, size_t hs_block_size);

الوصف

يمكن استخدام SSL_CTX_set_record_padding_callback() أو SSL_set_record_padding_callback() لتعيين دالة استدعاء cb لتحديد حشو سجلات TLS 1.3. يتم نسخ القيمة المعينة في ctx إلى SSL جديد بواسطة SSL_new(). لا يمكن استخدام TLS الأساسي إذا تم تعيين دالة استدعاء الحشو، ولا يمكن تعيين دالة الاستدعاء إذا تم تكوين TLS الأساسي بالفعل للكائن SSL الحالي.

تقوم SSL_CTX_set_record_padding_callback_arg() و SSL_set_record_padding_callback_arg() بتعيين قيمة arg التي يتم تمريرها إلى دالة الاستدعاء عند استدعائها. يتم نسخ القيمة المعينة في ctx إلى SSL جديد بواسطة SSL_new().

تسترد SSL_CTX_get_record_padding_callback_arg() و SSL_get_record_padding_callback_arg() قيمة arg التي يتم تمريرها إلى دالة الاستدعاء.

تقوم SSL_CTX_set_block_padding() و SSL_set_block_padding() بحشو السجل إلى مضاعف block_size. قيمة block_size 0 أو 1 تقوم بتعطيل حشو الكتل. الحد الأقصى لـ block_size هو SSL3_RT_MAX_PLAIN_LENGTH.

تقوم SSL_CTX_set_block_padding_ex() و SSL_set_block_padding_ex() بفعل الشيء نفسه ولكنها تسمح للمتصل بتحديد حجم كتلة الحشو بشكل منفصل ليتم تطبيقه على رسائل المصافحة وبيانات التطبيق.

يتم استدعاء دالة الاستدعاء لكل سجل قبل التشفير. المعلمة type هي نوع سجل TLS الذي تتم معالجته؛ قد تكون واحدة من SSL3_RT_APPLICATION_DATA، SSL3_RT_HANDSHAKE، أو SSL3_RT_ALERT. المعلمة len هي الطول الحالي للنص العادي للسجل قبل التشفير. المعلمة arg هي القيمة المعينة عبر SSL_CTX_set_record_padding_callback_arg() أو SSL_set_record_padding_callback_arg().

لا يمكن استخدام هذه الدوال مع كائنات QUIC SSL. تفشل SSL_set_record_padding_callback() و SSL_set_block_padding() إذا تم استدعاؤها على كائن QUIC SSL.

قيم الإرجاع

تقوم الدوال SSL_CTX_get_record_padding_callback_arg() و SSL_get_record_padding_callback_arg() بإرجاع قيمة arg التي تم تعيينها في دوال الإعداد المقابلة.

تقوم الدوال SSL_CTX_set_block_padding() و SSL_set_block_padding() بإرجاع 1 عند النجاح أو 0 إذا كان block_size كبيرًا جدًا.

تقوم cb بإرجاع عدد البايتات المراد إضافتها إلى السجل. قيمة إرجاع 0 تشير إلى عدم إضافة أي حشو. قيمة إرجاع تتسبب في تجاوز السجل للحجم الأقصى للسجل (SSL3_RT_MAX_PLAIN_LENGTH) ستقوم بالحشو حتى الحد الأقصى لحجم السجل.

ترجع دالة SSL_CTX_get_record_padding_callback_arg() القيمة 1 عند النجاح أو 0 إذا لم يتم تعيين دالة الاستدعاء لأن TLS الأساسي تم تكوينه لكائن SSL.

ملاحظات

السلوك الافتراضي هو عدم إضافة أي حشو إلى السجل.

ستقوم دالة استدعاء الحشو التي يقدمها المستخدم بتجاوز السلوك الذي تم تعيينه بواسطة SSL_set_block_padding() أو SSL_CTX_set_block_padding(). تعيين دالة الاستدعاء التي يقدمها المستخدم إلى NULL سيعيد سلوك حشو الكتلة المحدد.

تنطبق هذه الدوال فقط على سجلات TLS 1.3 التي تتم كتابتها.

لا تتم إضافة بايتات الحشو بوقت ثابت.

انظر أيضًا

التاريخ

تمت إضافة واجهة برمجة تطبيقات حشو السجل لدعم TLS 1.3 في OpenSSL 1.1.1.

تم تغيير نوع الإرجاع لدالة SSL_CTX_set_record_padding_callback() إلى int في OpenSSL 3.0.

تمت إضافة الدوال SSL_set_block_padding_ex() و SSL_CTX_set_block_padding_ex() في OpenSSL 3.4.
