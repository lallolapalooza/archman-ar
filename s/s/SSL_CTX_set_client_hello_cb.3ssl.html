SSL_CTX_set_client_hello_cb(3ssl)

الحزمة: openssl

التراخيص: Apache-2.0

SSL_CTX_SET_CLIENT_HELLO_CB(3ssl) OpenSSL SSL_CTX_SET_CLIENT_HELLO_CB(3ssl)

الاسم

SSL_CTX_set_client_hello_cb، SSL_client_hello_cb_fn، SSL_client_hello_isv2، SSL_client_hello_get0_legacy_version، SSL_client_hello_get0_random، SSL_client_hello_get0_session_id، SSL_client_hello_get0_ciphers، SSL_client_hello_get0_compression_methods، SSL_client_hello_get1_extensions_present، SSL_client_hello_get_extension_order، SSL_client_hello_get0_ext - دوال رد النداء لمعالجة ClientHello المبكرة من جانب الخادم

ملخص

typedef int (*SSL_client_hello_cb_fn)(SSL *s, int *al, void *arg);
void SSL_CTX_set_client_hello_cb(SSL_CTX *c, SSL_client_hello_cb_fn *f,
                                 void *arg);
int SSL_client_hello_isv2(SSL *s);
unsigned int SSL_client_hello_get0_legacy_version(SSL *s);
size_t SSL_client_hello_get0_random(SSL *s, const unsigned char **out);
size_t SSL_client_hello_get0_session_id(SSL *s, const unsigned char **out);
size_t SSL_client_hello_get0_ciphers(SSL *s, const unsigned char **out);
size_t SSL_client_hello_get0_compression_methods(SSL *s,
                                                 const unsigned char **out);
int SSL_client_hello_get1_extensions_present(SSL *s, int **out,
                                             size_t *outlen);
int SSL_client_hello_get_extension_order(SSL *s, uint16_t *exts,
                                         size_t *num_exts);
int SSL_client_hello_get0_ext(SSL *s, unsigned int type, const unsigned char **out,
                              size_t *outlen);

الوصف

SSL_CTX_set_client_hello_cb() تقوم بتعيين دالة رد النداء، والتي يتم استدعاؤها تلقائيًا خلال المراحل المبكرة من معالجة ClientHello على الخادم. يتم تمرير الوسيطة المقدمة عند تعيين رد النداء مرة أخرى إلى رد النداء وقت التشغيل. رد النداء الذي يعيد فشلاً (0) سيؤدي إلى إنهاء الاتصال، ويجب على ردود النداء التي تعيد فشلاً الإشارة إلى قيمة التنبيه التي سيتم إرسالها في المعلمة al. يمكن لرد النداء أيضًا إعادة قيمة سالبة لتعليق المصافحة، وستعود دالة المصافحة على الفور. SSL_get_error(3) ستعيد SSL_ERROR_WANT_CLIENT_HELLO_CB للإشارة إلى أن المصافحة تم تعليقها. تقع على عاتق دالة رد النداء ClientHello تخزين المعلومات حول حالة آخر استدعاء إذا لزم الأمر للمتابعة. في الاستدعاء التالي لدالة المصافحة، سيتم استدعاء دالة رد النداء ClientHello مرة أخرى، وإذا أعادت نجاحًا، فستستمر معالجة المصافحة العادية من تلك النقطة.

SSL_client_hello_isv2() تشير إلى ما إذا كان ClientHello قد تم نقله في سجل SSLv2 وهو بتنسيق SSLv2. تنسيق SSLv2 له اختلافات كبيرة عن تنسيق SSLv3 العادي، بما في ذلك استخدام ثلاثة بايتات لكل مجموعة تشفير، وعدم السماح بالملحقات. بالإضافة إلى ذلك، يتم الكشف عن حقل "challenge" بتنسيق SSLv2 عبر SSL_client_hello_get0_random()، مع إضافة أصفار إذا لزم الأمر إلى SSL3_RANDOM_SIZE بايت. بالنسبة لـ ClientHellos بتنسيق SSLv2، تعيد SSL_client_hello_get0_compression_methods() قائمة وهمية تتضمن فقط طريقة الضغط الفارغة، حيث أن تنسيق SSLv2 لا يتضمن آلية للتفاوض بشأن الضغط.

SSL_client_hello_get0_random()، SSL_client_hello_get0_session_id()، SSL_client_hello_get0_ciphers()، و SSL_client_hello_get0_compression_methods() توفر الوصول إلى حقول ClientHello المقابلة، مع إرجاع طول الحقل واختيارياً تعيين مؤشر خارجي إلى بايتات هذا الحقل.

بالمثل، توفر SSL_client_hello_get0_ext() الوصول إلى الملحقات الفردية من ClientHello على أساس كل ملحق. بالنسبة لقيمة نوع الملحق في بروتوكول السلك المقدم، يتم إرجاع قيمة الملحق وطوله في معلمات الإخراج (إذا كانت موجودة).

يمكن استخدام SSL_client_hello_get1_extensions_present() قبل SSL_client_hello_get0_ext()، لتحديد الملحقات الموجودة في ClientHello قبل الاستعلام عنها. المعلمتان out و outlen مطلوبتان، وعند النجاح، يجب على المتصل تحرير المساحة المخصصة لـ *out باستخدام OPENSSL_free(). محتويات *out هي مصفوفة من الأعداد الصحيحة التي تحمل القيمة الرقمية لأنواع ملحقات TLS بالترتيب الذي تظهر به في ClientHello. تحتوي *outlen على عدد العناصر في المصفوفة. في الحالات التي لا تحتوي فيها ClientHello على ملحقات، ستعيد الدالة نجاحًا مع تعيين *out إلى NULL و *outlen إلى 0.

SSL_client_hello_get_extension_order() مشابهة لـ SSL_client_hello_get1_extensions_present()، بدون تخصيص ذاكرة داخلي. عند استدعائها مع تعيين exts إلى NULL، تعيد عدد الملحقات (على سبيل المثال، لتخصيص مساحة لاستدعاء لاحق). بخلاف ذلك، يتم ملء *exts بقيم ExtensionType بالترتيب الذي ظهرت به الملحقات المقابلة في ClientHello. *num_exts هي معلمة إدخال / إخراج، تستخدم كإدخال لتوفير حجم المساحة المخصصة من قبل المتصل، وكإخراج للإشارة إلى عدد قيم ExtensionType التي تم كتابتها. إذا كان إدخال *num_exts أصغر من عدد الملحقات المعنية، فيتم التعامل مع ذلك على أنه خطأ. يمكن استدعاء لاحق مع تعيين exts إلى NULL لاسترداد حجم المساحة المطلوبة. يتم التعامل مع ClientHello الذي لم يحتوي على ملحقات كنجاح، مع تعيين *num_exts إلى 0.

ملاحظات

توفر دالة رد النداء ClientHello نافذة واسعة من الإمكانيات لرمز التطبيق للتأثير على مصافحة TLS. أحد الاستخدامات الأساسية لرد النداء هو السماح للخادم بفحص ملحق اسم الخادم المشار إليه الذي قدمه العميل من أجل اختيار شهادة مناسبة لتقديمها، وإجراء تعديلات تكوين أخرى ذات صلة باسم الخادم وتكوينه. يمكن أن تشمل تغييرات التكوين هذه تبديل مؤشر SSL_CTX المرتبط، وتعديل قائمة إصدارات TLS المسموح بها للخادم، وتغيير قائمة التشفير للخادم استجابةً لقائمة تشفير العميل، وما إلى ذلك.

يوصى أيضًا بأن تستخدم التطبيقات دالة رد نداء ClientHello وعدم استخدام دالة رد نداء servername، لتجنب السلوك غير المتوقع الذي يحدث بسبب الترتيب النسبي للمعالجة بين أشياء مثل استئناف الجلسة ودالة رد النداء servername التاريخية.

يمكن استدعاء عائلة دوال SSL_client_hello_* فقط من الرمز الذي يتم تنفيذه داخل رد نداء ClientHello.

قيم الإرجاع

تعيد دالة رد النداء ClientHello التي قدمها التطبيق SSL_CLIENT_HELLO_SUCCESS عند النجاح، و SSL_CLIENT_HELLO_ERROR عند الفشل، و SSL_CLIENT_HELLO_RETRY لتعليق المعالجة.

SSL_client_hello_isv2() تعيد 1 لـ ClientHellos بتنسيق SSLv2 و 0 خلاف ذلك.

SSL_client_hello_get0_random()، SSL_client_hello_get0_session_id()، SSL_client_hello_get0_ciphers()، و SSL_client_hello_get0_compression_methods() تعيد طول حقول ClientHello المقابلة. إذا تم إرجاع صفر، فلا يجب افتراض صحة مؤشر الإخراج.

SSL_client_hello_get0_ext() تعيد 1 إذا كان الملحق من النوع 'type' موجودًا، و 0 خلاف ذلك.

SSL_client_hello_get1_extensions_present() تعيد 1 عند النجاح و 0 عند الفشل.

SSL_client_hello_get_extension_order() تعيد 1 عند النجاح و 0 عند الفشل.
