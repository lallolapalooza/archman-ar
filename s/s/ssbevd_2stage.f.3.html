subroutine ssbevd_2stage (character jobz, character uplo, integer n, integer kd, real, dimension( ldab, * ) ab, integer ldab, real, dimension( * ) w, real, dimension( ldz, * ) z, integer ldz, real, dimension( * ) work, integer lwork, integer, dimension( * ) iwork, integer liwork, integer info)

SSBEVD_2STAGE تحسب القيم الذاتية، واختياريًا، المتجهات الذاتية اليسرى و/أو اليمنى لمصفوفات أخرى

الغرض:

!>
!> SSBEVD_2STAGE تحسب جميع القيم الذاتية، واختياريًا، المتجهات الذاتية لمصفوفة فريدة متماثلة حقيقية A باستخدام تقنية 2stage للاختزال إلى مصفوفة ثلاثية الأقطار. إذا تم طلب المتجهات الذاتية، فإنها تستخدم خوارزمية تقسيم وحل.
!>
!>

المعلمات

JOBZ 
!>          JOBZ هو CHARACTER*1
!>          = 'N':  حساب القيم الذاتية فقط؛
!>          = 'V':  حساب القيم الذاتية والمتجهات الذاتية.
!>                  غير متاح في هذا الإصدار.
!>

UPLO

!>          UPLO هو CHARACTER*1
!>          = 'U':  يتم تخزين الجزء العلوي من A؛
!>          = 'L':  يتم تخزين الجزء السفلي من A.
!>

N

!>          N هو INTEGER
!>          رتبة المصفوفة A. N >= 0.
!>

KD

!>          KD هو INTEGER
!>          عدد الأقطار الفوقية للمصفوفة A إذا كانت UPLO = 'U'،
!>          أو عدد الأقطار الفرعية إذا كانت UPLO = 'L'. KD >= 0.
!>

AB

!>          AB هو مصفوفة REAL، ذات البعد (LDAB, N)
!>          عند الإدخال، الجزء العلوي أو السفلي من المصفوفة الفريدة المتماثلة A، مخزنة في أول KD+1 صفوف من المصفوفة. يتم
!>          تخزين العمود j من A في العمود j من المصفوفة AB
!>          كما يلي:
!>          إذا كانت UPLO = 'U'، فإن AB(kd+1+i-j,j) = A(i,j) لـ max(1,j-kd) <= i <= j؛
!>          إذا كانت UPLO = 'L'، فإن AB(1+i-j,j) = A(i,j) لـ j <= i <= min(n,j+kd).
!>
!>          عند الخروج، يتم استبدال AB بقيم تم إنشاؤها أثناء
!>          الاختزال إلى شكل ثلاثي الأقطار. إذا كانت UPLO = 'U'، فإن
!>          القطر الفوقي الأول والقطر الرئيسي للمصفوفة ثلاثية الأقطار T
!>          يتم إرجاعهما في الصفين KD و KD+1 من AB، وإذا كانت UPLO = 'L'،
!>          يتم إرجاع القطر الرئيسي والقطر الفرعي الأول لـ T في
!>          أول صفين من AB.
!>

LDAB

!>          LDAB هو INTEGER
!>          البعد الرائد للمصفوفة AB. LDAB >= KD + 1.
!>

W

!>          W هي مصفوفة REAL، ذات البعد (N)
!>          إذا كانت INFO = 0، فإن القيم الذاتية مرتبة تصاعديًا.
!>

Z

!>          Z هي مصفوفة REAL، ذات البعد (LDZ, N)
!>          إذا كانت JOBZ = 'V'، فعندما تكون INFO = 0، تحتوي Z على المتجهات الذاتية المتعامدة
!>          للمصفوفة A، مع احتواء العمود i من Z
!>          على المتجه الذاتي المرتبط بـ W(i).
!>          إذا كانت JOBZ = 'N'، فلا يتم الرجوع إلى Z.
!>

LDZ

!>          LDZ هو INTEGER
!>          البعد الرائد للمصفوفة Z. LDZ >= 1، وإذا
!>          كانت JOBZ = 'V'، فإن LDZ >= max(1,N).
!>

WORK

!>          WORK هي مصفوفة REAL، ذات البعد LWORK
!>          عند الخروج، إذا كانت INFO = 0، فإن WORK(1) ترجع LWORK الأمثل.
!>

LWORK

!>          LWORK هو INTEGER
!>          طول المصفوفة WORK. LWORK >= 1، عندما تكون N <= 1؛
!>          بخلاف ذلك
!>          إذا كانت JOBZ = 'N' و N > 1، يجب الاستعلام عن LWORK.
!>                                   LWORK = MAX(1, dimension) حيث
!>                                   dimension = (2KD+1)*N + KD*NTHREADS + N
!>                                   حيث KD هو حجم النطاق.
!>                                   NTHREADS هو عدد الخيوط المستخدمة عند
!>                                   تمكين التجميع باستخدام openMP، وإلا فهو =1.
!>          إذا كانت JOBZ = 'V' و N > 1، يجب الاستعلام عن LWORK. غير متاح حاليًا.
!>
!>          إذا كانت LWORK = -1، فسيتم افتراض استعلام مساحة العمل؛ تقوم الإجراء الفرعي
!>          بحساب أحجام LWORK و IWORK المثلى فقط،
!>          ترجع هذه القيم كمدخلات أولى لمصفوفتي WORK و IWORK، ولا يتم إصدار رسالة خطأ متعلقة بـ LWORK أو
!>          LIWORK بواسطة XERBLA.
!>

IWORK

!>          IWORK هي مصفوفة INTEGER، ذات البعد (MAX(1,LIWORK))
!>          عند الخروج، إذا كانت INFO = 0، فإن IWORK(1) ترجع LIWORK الأمثل.
!>

LIWORK

!>          LIWORK هو INTEGER
!>          بعد المصفوفة IWORK.
!>          إذا كانت JOBZ = 'N' أو N <= 1، يجب أن تكون LIWORK على الأقل 1.
!>          إذا كانت JOBZ = 'V' و N > 2، يجب أن تكون LIWORK على الأقل 3 + 5*N.
!>
!>          إذا كانت LIWORK = -1، فسيتم افتراض استعلام مساحة العمل؛ تقوم الإجراء الفرعي
!>          بحساب أحجام LWORK و IWORK المثلى فقط،
!>          ترجع هذه القيم كمدخلات أولى لمصفوفتي WORK و IWORK، ولا يتم إصدار رسالة خطأ متعلقة بـ
!>          LWORK أو LIWORK بواسطة XERBLA.
!>

INFO

!>          INFO هو INTEGER
!>          = 0:  انتهاء ناجح
!>          < 0:  إذا كانت INFO = -i، فإن الوسيط i كان له قيمة غير قانونية
!>          > 0:  إذا كانت INFO = i، فشلت الخوارزمية في التقارب؛ i
!>                عناصر خارج القطر لمصفوفة وسيطة ثلاثية الأقطار
!>                لم تتقارب إلى الصفر.
!>
