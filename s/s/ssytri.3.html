subroutine ssytri (character uplo, integer n, real, dimension( lda, * ) a, integer lda, integer, dimension( * ) ipiv, real, dimension( * ) work, integer info)

SSYTRI

الغرض:

!>
!> SSYTRI يحسب معكوس مصفوفة حقيقية متماثلة غير محددة
!> A باستخدام التحليل A = U*D*U**T أو A = L*D*L**T الذي تم حسابه بواسطة
!> SSYTRF.
!>

المعلمات

UPLO 
!>          UPLO is CHARACTER*1
!>          يحدد ما إذا كانت تفاصيل التحليل مخزنة
!>          كمصفوفة مثلثة علوية أو سفلية.
!>          = 'U':  مثلثة علوية، الشكل هو A = U*D*U**T;
!>          = 'L':  مثلثة سفلية، الشكل هو A = L*D*L**T.
!>

N

!>          N is INTEGER
!>          رتبة المصفوفة A.  N >= 0.
!>

A

!>          A is REAL array, dimension (LDA,N)
!>          عند الدخول، المصفوفة القطرية الفرعية D والمضاعفات
!>          المستخدمة للحصول على العامل U أو L كما تم حسابه بواسطة SSYTRF.
!>
!>          عند الخروج، إذا كان INFO = 0، فإن معكوس المصفوفة الأصلية
!>          (المتماثل) موجود. إذا كان UPLO = 'U'، يتم تكوين الجزء العلوي
!>          المثلث من المعكوس ولا يتم الرجوع إلى الجزء A أسفل القطر؛
!>          إذا كان UPLO = 'L'، يتم تكوين الجزء السفلي المثلث من المعكوس
!>          ولا يتم الرجوع إلى الجزء A فوق القطر.
!>

LDA

!>          LDA is INTEGER
!>          البعد الرائد للمصفوفة A.  LDA >= max(1,N).
!>

IPIV

!>          IPIV is INTEGER array, dimension (N)
!>          تفاصيل المبادلات وهيكل D الفرعي
!>          كما تم تحديده بواسطة SSYTRF.
!>

WORK

!>          WORK is REAL array, dimension (N)
!>

INFO

!>          INFO is INTEGER
!>          = 0: نجاح الخروج
!>          < 0: إذا كان INFO = -i، فإن المعلمة i كانت لها قيمة غير قانونية
!>          > 0: إذا كان INFO = i، فإن D(i,i) = 0؛ المصفوفة منفردة ولا يمكن
!>               حساب معكوسها.
!>
