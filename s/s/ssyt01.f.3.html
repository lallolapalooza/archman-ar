وثائق الوظيفة/الروتين الفرعي

subroutine ssyt01 (character uplo, integer n, real, dimension( lda, * ) a, integer lda, real, dimension( ldafac, * ) afac, integer ldafac, integer, dimension( * ) ipiv, real, dimension( ldc, * ) c, integer ldc, real, dimension( * ) rwork, real resid)

SSYT01

الغرض:

!>
!> SSYT01 يعيد بناء مصفوفة غير محددة متماثلة A من تحليلها
!> L*D*L' أو U*D*U' الكتلة ويحسب مقدار الباقي
!>    norm( C - A ) / ( N * norm(A) * EPS ),
!> حيث C هي المصفوفة المعاد بناؤها و EPS هي وحدة الآلة.
!>

المعلمات

UPLO 
!>          UPLO هو CHARACTER*1
!>          يحدد ما إذا كان الجزء العلوي أو السفلي المثلثي للمصفوفة
!>          المتماثلة A مخزنًا:
!>          = 'U':  مثلثي علوي
!>          = 'L':  مثلثي سفلي
!>

N

!>          N هو INTEGER
!>          عدد الصفوف والأعمدة للمصفوفة A.  N >= 0.
!>

A

!>          A هو مصفوفة REAL، البعد (LDA,N)
!>          المصفوفة المتماثلة الأصلية A.
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرئيسي للمصفوفة A.  LDA >= max(1,N)
!>

AFAC

!>          AFAC هو مصفوفة REAL، البعد (LDAFAC,N)
!>          الشكل المفكك للمصفوفة A.  AFAC يحتوي على مصفوفة قطرية
!>          قطرية D والمعاملات المستخدمة للحصول على L
!>          أو U من L*D*L' أو U*D*U' الكتلة
!>          كما تم حسابه بواسطة SSYTRF.
!>

LDAFAC

!>          LDAFAC هو INTEGER
!>          البعد الرئيسي للمصفوفة AFAC.  LDAFAC >= max(1,N).
!>

IPIV

!>          IPIV هو مصفوفة INTEGER، البعد (N)
!>          مؤشرات المحور من SSYTRF.
!>

C

!>          C هو مصفوفة REAL، البعد (LDC,N)
!>

LDC

!>          LDC هو INTEGER
!>          البعد الرئيسي للمصفوفة C.  LDC >= max(1,N).
!>

RWORK

!>          RWORK هو مصفوفة REAL، البعد (N)
!>

RESID

!>          RESID هو REAL
!>          إذا كان UPLO = 'L'، norm(L*D*L' - A) / ( N * norm(A) * EPS )
!>          إذا كان UPLO = 'U'، norm(U*D*U' - A) / ( N * norm(A) * EPS )
!>
