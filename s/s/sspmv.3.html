subroutine sspmv (character uplo, integer n, real alpha, real, dimension(*) ap, real, dimension(*) x, integer incx, real beta, real, dimension(*) y, integer incy)

SSPMV

الغرض:

!>
!> SSPMV  تقوم بعملية المصفوفة والمتجه
!>
!>    y := alpha*A*x + beta*y,
!>
!> حيث alpha و beta عددان قياسيان، x و y متجهان يحتويان على n عنصر،
!> و A هي مصفوفة متناظرة بحجم n × n، مقدمة في شكل مضغوط.
!>

المعاملات

UPLO 
!>          UPLO هو CHARACTER*1
!>           عند الإدخال، يحدد UPLO ما إذا كان الجزء العلوي أو السفلي
!>           من المثلث من المصفوفة A مقدم في المصفوفة AP كما يلي:
!>
!>              UPLO = 'U' or 'u'   الجزء العلوي من المثلث من A مقدم في
!>                                  AP.
!>
!>              UPLO = 'L' or 'l'   الجزء السفلي من المثلث من A مقدم في
!>                                  AP.
!>

N

!>          N هو INTEGER
!>           عند الإدخال، يحدد N ترتيب المصفوفة A.
!>           يجب أن يكون N على الأقل صفرًا.
!>

ALPHA

!>          ALPHA هو REAL
!>           عند الإدخال، يحدد ALPHA العدد القياسي alpha.
!>

AP

!>          AP هي مصفوفة REAL، بأبعاد على الأقل
!>           ( ( n*( n + 1 ) )/2 ).
!>           قبل الإدخال مع UPLO = 'U' أو 'u'، يجب أن تحتوي المصفوفة AP
!>           على الجزء العلوي من المثلث من المصفوفة المتناظرة
!>           مضغوطًا بشكل متسلسل، عمودًا بعمود، بحيث يحتوي AP( 1 )
!>           على a( 1, 1 )، ويحتوي AP( 2 ) و AP( 3 ) على a( 1, 2 )
!>           و a( 2, 2 ) على التوالي، وهكذا.
!>           قبل الإدخال مع UPLO = 'L' أو 'l'، يجب أن تحتوي المصفوفة AP
!>           على الجزء السفلي من المثلث من المصفوفة المتناظرة
!>           مضغوطًا بشكل متسلسل، عمودًا بعمود، بحيث يحتوي AP( 1 )
!>           على a( 1, 1 )، ويحتوي AP( 2 ) و AP( 3 ) على a( 2, 1 )
!>           و a( 3, 1 ) على التوالي، وهكذا.
!>

X

!>          X هي مصفوفة REAL، بأبعاد على الأقل
!>           ( 1 + ( n - 1 )*abs( INCX ) ).
!>           قبل الإدخال، يجب أن تحتوي المصفوفة X المزودة بزيادة على المتجه
!>           x المكون من n عنصر.
!>

INCX

!>          INCX هو INTEGER
!>           عند الإدخال، يحدد INCX الزيادة لعناصر X.
!>           يجب ألا يكون INCX صفرًا.
!>

BETA

!>          BETA هو REAL
!>           عند الإدخال، يحدد BETA العدد القياسي beta. عندما يتم توفير BETA
!>           على أنه صفر، فلا يلزم تعيين Y عند الإدخال.
!>

Y

!>          Y هي مصفوفة REAL، بأبعاد على الأقل
!>           ( 1 + ( n - 1 )*abs( INCY ) ).
!>           قبل الإدخال، يجب أن تحتوي المصفوفة Y المزودة بزيادة على المتجه
!>           y المكون من n عنصر. عند الخروج، يتم الكتابة فوق Y بالمتجه
!>           y المحدث.
!>

INCY

!>          INCY هو INTEGER
!>           عند الإدخال، يحدد INCY الزيادة لعناصر Y.
!>           يجب ألا يكون INCY صفرًا.
!>
