subroutine ssyevx (character jobz, character range, character uplo, integer n, real, dimension( lda, * ) a, integer lda, real vl, real vu, integer il, integer iu, real abstol, integer m, real, dimension( * ) w, real, dimension( ldz, * ) z, integer ldz, real, dimension( * ) work, integer lwork, integer, dimension( * ) iwork, integer, dimension( * ) ifail, integer info)

SSYEVX يحسب القيم الذاتية، واختياريًا، المتجهات الذاتية اليسرى و/أو اليمنى لمصفوفات SY

الغرض:

!>
!> SSYEVX يحسب القيم الذاتية المحددة، واختياريًا، المتجهات الذاتية
!> لمصفوفة متناظرة حقيقية A. يمكن تحديد القيم الذاتية والمتجهات الذاتية
!> عن طريق تحديد نطاق من القيم أو نطاق من المؤشرات
!> للقيم الذاتية المطلوبة.
!>

المعلمات

JOBZ 
!>          JOBZ is CHARACTER*1
!>          = 'N':  حساب القيم الذاتية فقط؛
!>          = 'V':  حساب القيم الذاتية والمتجهات الذاتية.
!>

RANGE

!>          RANGE is CHARACTER*1
!>          = 'A': سيتم العثور على جميع القيم الذاتية.
!>          = 'V': سيتم العثور على جميع القيم الذاتية في الفترة نصف المفتوحة (VL,VU]
!>          = 'I': سيتم العثور على القيم الذاتية من IL إلى IU.
!>

UPLO

!>          UPLO is CHARACTER*1
!>          = 'U':  يتم تخزين الجزء العلوي من A؛
!>          = 'L':  يتم تخزين الجزء السفلي من A.
!>

N

!>          N is INTEGER
!>          رتبة المصفوفة A.  N >= 0.
!>

A

!>          A is REAL array, dimension (LDA, N)
!>          عند الدخول، يحتوي الجزء العلوي من المصفوفة A بحجم N*N على
!>          الجزء العلوي من المصفوفة A. إذا كان UPLO = 'L'،
!>          يحتوي الجزء السفلي من المصفوفة A بحجم N*N على
!>          الجزء السفلي من المصفوفة A.
!>          عند الخروج، يتم تدمير الجزء السفلي (إذا كان UPLO='L') أو الجزء العلوي
!>          (إذا كان UPLO='U') من A، بما في ذلك القطر.
!>

LDA

!>          LDA is INTEGER
!>          البعد الرئيسي للمصفوفة A.  LDA >= max(1,N).
!>

VL

!>          VL is REAL
!>          إذا كان RANGE='V'، الحد الأدنى للفترة التي
!>          سيتم البحث فيها عن القيم الذاتية. VL < VU.
!>          غير مستخدمة إذا كان RANGE = 'A' أو 'I'.
!>

VU

!>          VU is REAL
!>          إذا كان RANGE='V'، الحد الأعلى للفترة التي
!>          سيتم البحث فيها عن القيم الذاتية. VL < VU.
!>          غير مستخدمة إذا كان RANGE = 'A' أو 'I'.
!>

IL

!>          IL is INTEGER
!>          إذا كان RANGE='I'، مؤشر
!>          أصغر قيمة ذاتية سيتم إرجاعها.
!>          1 <= IL <= IU <= N، إذا كان N > 0؛ IL = 1 و IU = 0 إذا كان N = 0.
!>          غير مستخدمة إذا كان RANGE = 'A' أو 'V'.
!>

IU

!>          IU is INTEGER
!>          إذا كان RANGE='I'، مؤشر
!>          أكبر قيمة ذاتية سيتم إرجاعها.
!>          1 <= IL <= IU <= N، إذا كان N > 0؛ IL = 1 و IU = 0 إذا كان N = 0.
!>          غير مستخدمة إذا كان RANGE = 'A' أو 'V'.
!>

ABSTOL

!>          ABSTOL is REAL
!>          التسامح المطلق للخطأ في القيم الذاتية.
!>          يتم قبول قيمة ذاتية تقريبية على أنها متقاربة
!>          عندما يتم تحديد أنها تقع في فترة [a,b]
!>          بعرض أقل من أو يساوي
!>
!>                  ABSTOL + EPS *   max( |a|,|b| ) ,
!>
!>          حيث EPS هو الدقة الآلية. إذا كانت ABSTOL أقل من
!>          أو تساوي الصفر، فسيتم استخدام EPS*|T| بدلاً منها،
!>          حيث |T| هو المعيار 1 للمصفوفة ثلاثية القطر التي تم الحصول عليها
!>          عن طريق تقليل A إلى الشكل ثلاثي القطر.
!>
!>          سيتم حساب القيم الذاتية بأكبر دقة عندما يتم تعيين ABSTOL إلى
!>          ضعف عتبة التدفق السفلي 2*SLAMCH('S')، وليس الصفر.
!>          إذا عادت هذه الروتين الفرعي مع INFO>0، مما يشير إلى أن بعض
!>          المتجهات الذاتية لم تتقارب، حاول تعيين ABSTOL إلى
!>          2*SLAMCH('S').
!>
!>          انظر  بواسطة Demmel و
!>          Kahan، مذكرة عمل LAPACK #3.
!>

M

!>          M is INTEGER
!>          العدد الإجمالي للقيم الذاتية التي تم العثور عليها.  0 <= M <= N.
!>          إذا كان RANGE = 'A'، فإن M = N، وإذا كان RANGE = 'I'، فإن M = IU-IL+1.
!>

W

!>          W is REAL array, dimension (N)
!>          عند الخروج الطبيعي، تحتوي العناصر M الأولى على القيم الذاتية المحددة
!>          بترتيب تصاعدي.
!>

Z

!>          Z is REAL array, dimension (LDZ, max(1,M))
!>          إذا كان JOBZ = 'V'، فعندما يكون INFO = 0، تحتوي الأعمدة M الأولى من Z
!>          على المتجهات الذاتية المتعامدة للمصفوفة A
!>          المقابلة للقيم الذاتية المحددة، حيث يحمل العمود i من Z
!>          المتجه الذاتي المرتبط بـ W(i).
!>          إذا فشل متجه ذاتي في التقارب، فإن هذا العمود من Z
!>          يحتوي على أحدث تقريب للمتجه الذاتي، وسيتم إرجاع مؤشر
!>          المتجه الذاتي في IFAIL.
!>          إذا كان JOBZ = 'N'، فلن يتم استخدام Z.
!>          ملاحظة: يجب على المستخدم التأكد من توفير عمود واحد على الأقل (max(1,M))
!>          في المصفوفة Z؛ إذا كان RANGE = 'V'، فإن القيمة الدقيقة لـ M
!>          غير معروفة مقدمًا ويجب استخدام حد أعلى.
!>

LDZ

!>          LDZ is INTEGER
!>          البعد الرئيسي للمصفوفة Z.  LDZ >= 1، وإذا
!>          كان JOBZ = 'V'، فإن LDZ >= max(1,N).
!>

WORK

!>          WORK is REAL array, dimension (MAX(1,LWORK))
!>          عند الخروج، إذا كان INFO = 0، فإن WORK(1) يرجع LWORK الأمثل.
!>

LWORK

!>          LWORK is INTEGER
!>          طول المصفوفة WORK.  LWORK >= 1، عندما يكون N <= 1؛
!>          بخلاف ذلك 8*N.
!>          للكفاءة المثلى، LWORK >= (NB+3)*N،
!>          حيث NB هو الحد الأقصى لحجم الكتلة لـ SSYTRD و SORMTR
!>          الذي تم إرجاعه بواسطة ILAENV.
!>
!>          إذا كان LWORK = -1، فسيتم افتراض استعلام عن مساحة العمل؛ الروتين الفرعي
!>          يحسب فقط الحجم الأمثل لمصفوفة WORK، ويعيد
!>          هذه القيمة كالإدخال الأول لمصفوفة WORK، ولا يتم إصدار أي رسالة خطأ
!>          متعلقة بـ LWORK بواسطة XERBLA.
!>

IWORK

!>          IWORK is INTEGER array, dimension (5*N)
!>

IFAIL

!>          IFAIL is INTEGER array, dimension (N)
!>          إذا كان JOBZ = 'V'، فعندما يكون INFO = 0، فإن العناصر M الأولى من
!>          IFAIL تكون صفرًا. إذا كان INFO > 0، فإن IFAIL يحتوي على
!>          مؤشرات المتجهات الذاتية التي فشلت في التقارب.
!>          إذا كان JOBZ = 'N'، فلن يتم استخدام IFAIL.
!>

INFO

!>          INFO is INTEGER
!>          = 0:  انتهاء ناجح
!>          < 0:  إذا كان INFO = -i، فإن المعلمة i كانت لها قيمة غير قانونية
!>          > 0:  إذا كان INFO = i، فقد فشلت i من المتجهات الذاتية في التقارب.
!>                تم تخزين مؤشراتها في المصفوفة IFAIL.
!>
