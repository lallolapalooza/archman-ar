subroutine ssbevd (jobz, uplo, n, kd, ab, ldab, w, z, ldz, work, lwork, iwork, liwork, info)
SSBEVD يحسب القيم الذاتية، واختياريًا، المتجهات الذاتية اليسرى و/أو اليمنى للمصفوفات الأخرى

توثيق الوظيفة/الروتين الفرعي

subroutine ssbevd (character jobz, character uplo, integer n, integer kd, real, dimension( ldab, * ) ab, integer ldab, real, dimension( * ) w, real, dimension( ldz, * ) z, integer ldz, real, dimension( * ) work, integer lwork, integer, dimension( * ) iwork, integer liwork, integer info)

SSBEVD يحسب جميع القيم الذاتية، واختياريًا، المتجهات الذاتية لمصفوفة فرقة متناظرة حقيقية A. إذا تم طلب المتجهات الذاتية، فإنها تستخدم خوارزمية divide and conquer.

الغرض:

!>
!> SSBEVD يحسب جميع القيم الذاتية، واختياريًا، المتجهات الذاتية
!> لمصفوفة فرقة متناظرة حقيقية A. إذا تم طلب المتجهات الذاتية، فإنها تستخدم
!> خوارزمية divide and conquer.
!>
!>

المعلمات

JOBZ 
!>          JOBZ is CHARACTER*1
!>          = 'N':  احسب القيم الذاتية فقط؛
!>          = 'V':  احسب القيم الذاتية والمتجهات الذاتية.
!>

UPLO

!>          UPLO is CHARACTER*1
!>          = 'U':  المثلث العلوي من A مخزن؛
!>          = 'L':  المثلث السفلي من A مخزن.
!>

N

!>          N is INTEGER
!>          رتبة المصفوفة A. N >= 0.
!>

KD

!>          KD is INTEGER
!>          عدد الأقطار الفائقة للمصفوفة A إذا كان UPLO = 'U'،
!>          أو عدد الأقطار الفرعية إذا كان UPLO = 'L'. KD >= 0.
!>

AB

!>          AB is REAL array, dimension (LDAB, N)
!>          عند الإدخال، المثلث العلوي أو السفلي للمصفوفة الفرقة المتناظرة
!>          A، مخزن في أول KD+1 صفوف من المصفوفة.
!>          يتم تخزين العمود j من A في العمود j من المصفوفة AB
!>          على النحو التالي:
!>          إذا كان UPLO = 'U'، فإن AB(kd+1+i-j,j) = A(i,j) لـ max(1,j-kd) <= i <= j؛
!>          إذا كان UPLO = 'L'، فإن AB(1+i-j,j) = A(i,j) لـ j <= i <= min(n,j+kd).
!>
!>          عند الخروج، يتم الكتابة فوق AB بالقيم التي تم إنشاؤها أثناء
!>          الاختزال إلى شكل ثلاثي الأقطار. إذا كان UPLO = 'U'، فإن القطر الفائق الأول
!>          والقطر للمصفوفة ثلاثية الأقطار T
!>          يتم إرجاعهما في الصفين KD و KD+1 من AB، وإذا كان UPLO = 'L'،
!>          يتم إرجاع القطر والقطر الفرعي الأول لـ T في
!>          أول صفين من AB.
!>

LDAB

!>          LDAB is INTEGER
!>          البعد الرائد للمصفوفة AB. LDAB >= KD + 1.
!>

W

!>          W is REAL array, dimension (N)
!>          إذا كان INFO = 0، فإن القيم الذاتية بترتيب تصاعدي.
!>

Z

!>          Z is REAL array, dimension (LDZ, N)
!>          إذا كان JOBZ = 'V'، فعندما يكون INFO = 0، تحتوي Z على المتجهات الذاتية
!>          المتعامدة للمصفوفة A، مع احتواء العمود i من Z
!>          على المتجه الذاتي المرتبط بـ W(i).
!>          إذا كان JOBZ = 'N'، فلا يتم الرجوع إلى Z.
!>

LDZ

!>          LDZ is INTEGER
!>          البعد الرائد للمصفوفة Z. LDZ >= 1، وإذا كان
!>          JOBZ = 'V'، فإن LDZ >= max(1,N).
!>

WORK

!>          WORK is REAL array,
!>                                         dimension (LWORK)
!>          عند الخروج، إذا كان INFO = 0، فإن WORK(1) يعيد LWORK الأمثل.
!>

LWORK

!>          LWORK is INTEGER
!>          بعد المصفوفة WORK.
!>          إذا كان N <= 1، فيجب أن يكون LWORK على الأقل 1.
!>          إذا كان JOBZ = 'N' و N > 2، فيجب أن يكون LWORK على الأقل 2*N.
!>          إذا كان JOBZ = 'V' و N > 2، فيجب أن يكون LWORK على الأقل
!>                         ( 1 + 5*N + 2*N**2 ).
!>
!>          إذا كان LWORK = -1، فيفترض استعلام مساحة العمل؛ الروتين
!>          يحسب فقط الأحجام المثلى لمصفوفتي WORK و IWORK،
!>          ويعيد هذه القيم كأول إدخالات في مصفوفتي WORK
!>          و IWORK، ولا يتم إصدار أي رسالة خطأ متعلقة بـ LWORK أو
!>          LIWORK بواسطة XERBLA.
!>

IWORK

!>          IWORK is INTEGER array, dimension (MAX(1,LIWORK))
!>          عند الخروج، إذا كان INFO = 0، فإن IWORK(1) يعيد LIWORK الأمثل.
!>

LIWORK

!>          LIWORK is INTEGER
!>          بعد المصفوفة IWORK.
!>          إذا كان JOBZ = 'N' أو N <= 1، فيجب أن يكون LIWORK على الأقل 1.
!>          إذا كان JOBZ = 'V' و N > 2، فيجب أن يكون LIWORK على الأقل 3 + 5*N.
!>
!>          إذا كان LIWORK = -1، فيفترض استعلام مساحة العمل؛ الروتين
!>          يحسب فقط الأحجام المثلى لمصفوفتي WORK و
!>          IWORK، ويعيد هذه القيم كأول إدخالات في
!>          مصفوفتي WORK و IWORK، ولا يتم إصدار أي رسالة خطأ متعلقة بـ
!>          LWORK أو LIWORK بواسطة XERBLA.
!>

INFO

!>          INFO is INTEGER
!>          = 0:  نجاح الخروج
!>          < 0:  إذا كان INFO = -i، فإن المعامل i كان لديه قيمة غير قانونية
!>          > 0:  إذا كان INFO = i، فشلت الخوارزمية في التقارب؛ i
!>                العناصر خارج القطر لمصفوفة ثلاثية الأقطار وسيطة
!>                لم تتقارب إلى الصفر.
!>
