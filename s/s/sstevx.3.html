subroutine sstevx (jobz, range, n, d, e, vl, vu, il, iu, abstol, m, w, z, ldz, work, iwork, ifail, info)
SSTEVX تحسب القيم الذاتية، وبشكل اختياري، المتجهات الذاتية اليسرى و/أو اليمنى لمصفوفات أخرى

توثيق الوظيفة/الإجراء الفرعي

subroutine sstevx (character jobz, character range, integer n, real, dimension( * ) d, real, dimension( * ) e, real vl, real vu, integer il, integer iu, real abstol, integer m, real, dimension( * ) w, real, dimension( ldz, * ) z, integer ldz, real, dimension( * ) work, integer, dimension( * ) iwork, integer, dimension( * ) ifail, integer info)

SSTEVX تحسب القيم الذاتية المحددة، وبشكل اختياري، المتجهات الذاتية لمصفوفة ثلاثية متناظرة حقيقية A. يمكن تحديد القيم الذاتية والمتجهات الذاتية عن طريق تحديد نطاق من القيم أو نطاق من الفهارس للقيم الذاتية المطلوبة.

الغرض:

!>
!> SSTEVX تحسب القيم الذاتية المحددة، وبشكل اختياري، المتجهات الذاتية
!> لمصفوفة ثلاثية متناظرة حقيقية A.  يمكن تحديد القيم الذاتية والمتجهات الذاتية
!> عن طريق تحديد إما نطاق من القيم
!> أو نطاق من الفهارس للقيم الذاتية المطلوبة.
!>

المعلمات

JOBZ 
!>          JOBZ is CHARACTER*1
!>          = 'N':  حساب القيم الذاتية فقط؛
!>          = 'V':  حساب القيم الذاتية والمتجهات الذاتية.
!>

RANGE

!>          RANGE is CHARACTER*1
!>          = 'A': سيتم العثور على جميع القيم الذاتية.
!>          = 'V': سيتم العثور على جميع القيم الذاتية في الفترة شبه المفتوحة (VL,VU]
!>                 .
!>          = 'I': سيتم العثور على القيم الذاتية من IL إلى IU.
!>

N

!>          N is INTEGER
!>          رتبة المصفوفة.  N >= 0.
!>

D

!>          D is REAL array, dimension (N)
!>          عند الدخول، العناصر القطرية n للمصفوفة الثلاثية
!>          A.
!>          عند الخروج، قد يتم ضرب D بمعامل ثابت تم اختياره
!>          لتجنب الحمل الزائد/الناقص في حساب القيم الذاتية.
!>

E

!>          E is REAL array, dimension (max(1,N-1))
!>          عند الدخول، العناصر تحت القطرية (n-1) للمصفوفة الثلاثية
!>          A في العناصر من 1 إلى N-1 من E.
!>          عند الخروج، قد يتم ضرب E بمعامل ثابت تم اختياره
!>          لتجنب الحمل الزائد/الناقص في حساب القيم الذاتية.
!>

VL

!>          VL is REAL
!>          إذا كانت RANGE='V'، فإن الحد الأدنى للفترة التي
!>          سيتم البحث فيها عن القيم الذاتية. VL < VU.
!>          لا يتم الرجوع إليها إذا كانت RANGE = 'A' أو 'I'.
!>

VU

!>          VU is REAL
!>          إذا كانت RANGE='V'، فإن الحد الأعلى للفترة التي
!>          سيتم البحث فيها عن القيم الذاتية. VL < VU.
!>          لا يتم الرجوع إليها إذا كانت RANGE = 'A' أو 'I'.
!>

IL

!>          IL is INTEGER
!>          إذا كانت RANGE='I'، فهرس
!>          أصغر قيمة ذاتية سيتم إرجاعها.
!>          1 <= IL <= IU <= N، إذا كانت N > 0؛ IL = 1 و IU = 0 إذا كانت N = 0.
!>          لا يتم الرجوع إليها إذا كانت RANGE = 'A' أو 'V'.
!>

IU

!>          IU is INTEGER
!>          إذا كانت RANGE='I'، فهرس
!>          أكبر قيمة ذاتية سيتم إرجاعها.
!>          1 <= IL <= IU <= N، إذا كانت N > 0؛ IL = 1 و IU = 0 إذا كانت N = 0.
!>          لا يتم الرجوع إليها إذا كانت RANGE = 'A' أو 'V'.
!>

ABSTOL

!>          ABSTOL is REAL
!>          التفاوت المطلق في الخطأ للقيم الذاتية.
!>          يتم قبول قيمة ذاتية تقريبية على أنها متقاربة
!>          عندما يتم تحديد أنها تقع في فترة [a,b]
!>          بعرض أقل من أو يساوي
!>
!>                  ABSTOL + EPS *   max( |a|,|b| ) ,
!>
!>          حيث EPS هي دقة الآلة.  إذا كان ABSTOL أقل
!>          من أو يساوي صفرًا، فسيتم استخدام EPS*|T|
!!>          بدلاً منه، حيث |T| هو المعيار 1 للمصفوفة الثلاثية
!>          .
!>
!>          سيتم حساب القيم الذاتية بدقة أكبر عندما يتم تعيين ABSTOL
!>          إلى ضعف حد التدفق السفلي 2*SLAMCH('S')، وليس صفرًا.
!>          إذا عادت هذه الوظيفة مع INFO>0، مما يشير إلى أن بعض
!>          المتجهات الذاتية لم تتقارب، حاول تعيين ABSTOL إلى
!>          2*SLAMCH('S').
!>
!>          انظر  بواسطة Demmel و
!>          Kahan، LAPACK Working Note #3.
!>

M

!>          M is INTEGER
!>          العدد الإجمالي للقيم الذاتية التي تم العثور عليها.  0 <= M <= N.
!>          إذا كانت RANGE = 'A'، فإن M = N، وإذا كانت RANGE = 'I'، فإن M = IU-IL+1.
!>

W

!>          W is REAL array, dimension (N)
!>          تحتوي العناصر الأولى M على القيم الذاتية المحددة في
!>          ترتيب تصاعدي.
!>

Z

!>          Z is REAL array, dimension (LDZ, max(1,M) )
!>          إذا كان JOBZ = 'V'، فعندما يكون INFO = 0، فإن الأعمدة M الأولى من Z
!>          تحتوي على المتجهات الذاتية المتعامدة للمصفوفة A
!>          المقابلة للقيم الذاتية المحددة، مع العمود i
!>          من Z يحمل المتجه الذاتي المرتبط بـ W(i).
!>          إذا فشل متجه ذاتي في التقارب (INFO > 0)، فإن هذا
!>          العمود من Z يحتوي على أحدث تقريب للمتجه الذاتي،
!>          ويتم إرجاع فهرس المتجه الذاتي في IFAIL.  إذا كان JOBZ = 'N'، فلا يتم الرجوع إلى Z.
!>          ملاحظة: يجب على المستخدم التأكد من توفير عمود واحد على الأقل كحد أقصى (1,M)
!>          في المصفوفة Z؛ إذا كانت RANGE = 'V'، فإن القيمة الدقيقة لـ M
!>          غير معروفة مسبقًا ويجب استخدام حد أعلى.
!>

LDZ

!>          LDZ is INTEGER
!>          البعد الرائد للمصفوفة Z.  LDZ >= 1، وإذا
!>          كان JOBZ = 'V'، فإن LDZ >= max(1,N).
!>

WORK

!>          WORK is REAL array, dimension (5*N)
!>

IWORK

!>          IWORK is INTEGER array, dimension (5*N)
!>

IFAIL

!>          IFAIL is INTEGER array, dimension (N)
!>          إذا كان JOBZ = 'V'، فعندما يكون INFO = 0، فإن العناصر M الأولى من
!>          IFAIL هي صفر.  إذا كان INFO > 0، فإن IFAIL يحتوي على
!>          فهارس المتجهات الذاتية التي فشلت في التقارب.
!>          إذا كان JOBZ = 'N'، فلا يتم الرجوع إلى IFAIL.
!>

INFO

!>          INFO is INTEGER
!>          = 0:  انتهاء ناجح
!>          < 0:  إذا كان INFO = -i، فإن الوسيط i كان له قيمة غير قانونية
!>          > 0:  إذا كان INFO = i، فقد فشلت i متجهات ذاتية في التقارب.
!>                يتم تخزين فهارسها في المصفوفة IFAIL.
!>
