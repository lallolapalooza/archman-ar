!>          M هو عدد صحيح
!>          إجمالي عدد القيم الذاتية التي تم العثور عليها.  0 <= M <= N.
!>          إذا كان RANGE = 'A'، فإن M = N، وإذا كان RANGE = 'I'، فإن M = IU-IL+1.
!>

W

!>          W هو مصفوفة حقيقية، بُعدها (N)
!>          تحتوي العناصر القليلة الأولى على القيم الذاتية المحددة في
!>          ترتيب تصاعدي.
!>

Z

!>          Z هي مصفوفة حقيقية، بُعدها (LDZ, max(1,M))
!>          إذا كان JOBZ = 'V'، فعندما يكون INFO = 0، تحتوي الأعمدة القليلة الأولى من Z
!>          على المتجهات الذاتية المتعامدة للمصفوفة A
!>          المقابلة للقيم الذاتية المحددة، حيث يحمل العمود i من Z المتجه الذاتي المرتبط بـ W(i).
!>          إذا كان JOBZ = 'N'، فلا يتم الرجوع إلى Z.
!>          ملاحظة: يجب على المستخدم التأكد من توفير عمود واحد على الأقل في المصفوفة Z؛ إذا كان RANGE = 'V'، فإن القيمة الدقيقة لـ M
!>          غير معروفة مسبقًا ويجب استخدام حد أعلى.
!>          توفير N أعمدة آمن دائمًا.
!>

LDZ

!>          LDZ هو عدد صحيح
!>          البعد الرئيسي للمصفوفة Z.  LDZ >= 1، وإذا
!>          كان JOBZ = 'V'، فإن LDZ >= max(1,N).
!>

ISUPPZ

!>          ISUPPZ هي مصفوفة أعداد صحيحة، بُعدها ( 2*max(1,M) )
!>          دعم المتجهات الذاتية في Z، أي الفهارس
!>          التي تشير إلى العناصر غير الصفرية في Z. المتجه الذاتي i يكون غير صفري فقط في العناصر ISUPPZ( 2*i-1 ) إلى
!>          ISUPPZ( 2*i ). هذا هو ناتج SSTEMR (مصفوفة ثلاثية القطر).
!>          دعم المتجهات الذاتية للمصفوفة A هو عادة 1:N بسبب التحويلات المتعامدة التي تطبقها SORMTR.
!>          مُنفذ فقط لـ RANGE = 'A' أو 'I' و IU - IL = N - 1
!>

WORK

!>          WORK هي مصفوفة حقيقية، بُعدها (MAX(1,LWORK))
!>          عند الخروج، إذا كان INFO = 0، فإن WORK(1) يعيد LWORK الأمثل.
!>

LWORK

!>          LWORK هو عدد صحيح
!>          بُعد المصفوفة WORK.
!>          إذا كان N <= 1، يجب أن يكون LWORK >= 1.
!>          إذا كان JOBZ = 'N' و N > 1، يجب الاستعلام عن LWORK.
!>                                   LWORK = MAX(1, 26*N, dimension) حيث
!>                                   dimension = max(stage1,stage2) + (KD+1)*N + 5*N
!>                                             = N*KD + N*max(KD+1,FACTOPTNB)
!>                                               + max(2*KD*KD, KD*NTHREADS)
!>                                               + (KD+1)*N + 5*N
!>                                   حيث KD هو حجم التجميع للتقليل،
!>                                   FACTOPTNB هو التجميع المستخدم بواسطة خوارزمية QR أو LQ،
!>                                   عادةً ما يكون FACTOPTNB=128 خيارًا جيدًا.
!>                                   NTHREADS هو عدد الخيوط المستخدمة عند
!>                                   تم تفعيل تجميع openMP، وإلا =1.
!>          إذا كان JOBZ = 'V' و N > 1، يجب الاستعلام عن LWORK. غير متاح حاليًا.
!>
!>          إذا كان LWORK = -1، فإنه يُفترض استعلام عن مساحة العمل؛ تقوم الروتين
!>          فقط بحساب الحجم الأمثل لمصفوفة WORK، وتعيد
!>          هذه القيمة كإدخال أول لمصفوفة WORK، ولا يصدر رسالة خطأ
!>          متعلقة بـ LWORK بواسطة XERBLA.
!>

IWORK

!>          IWORK هي مصفوفة أعداد صحيحة، بُعدها (MAX(1,LIWORK))
!>          عند الخروج، إذا كان INFO = 0، فإن IWORK(1) يعيد LIWORK الأمثل.
!>

LIWORK

!>          LIWORK هو عدد صحيح
!>          بُعد المصفوفة IWORK.
!>          إذا كان N <= 1، فإن LIWORK >= 1، وإلا فإن LIWORK >= 10*N.
!>
!>          إذا كان LIWORK = -1، فإنه يُفترض استعلام عن مساحة العمل؛ تقوم الروتين
!>          فقط بحساب الحجم الأمثل لمصفوفة IWORK،
!>          وتعيد هذه القيمة كإدخال أول لمصفوفة IWORK، و
!>          لا يصدر رسالة خطأ متعلقة بـ LIWORK بواسطة XERBLA.
!>

INFO

!>          INFO هو عدد صحيح
!>          = 0:  خروج ناجح
!>          < 0:  إذا كان INFO = -i، فإن الوسيط i كان له قيمة غير قانونية
!>          > 0:  خطأ داخلي
!>
