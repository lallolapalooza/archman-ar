subroutine ssysv_rook (character uplo, integer n, integer nrhs, real, dimension( lda, * ) a, integer lda, integer, dimension( * ) ipiv, real, dimension( ldb, * ) b, integer ldb, real, dimension( * ) work, integer lwork, integer info)

SSYSV_ROOK تحسب حل نظام المعادلات الخطية A * X = B لمصفوفات SY

الغرض:

!>
!> SSYSV_ROOK تحسب حل نظام المعادلات الخطية الحقيقية
!>    A * X = B,
!> حيث A مصفوفة متناظرة N × N و X و B مصفوفتان N × NRHS.
!>
!> يتم استخدام طريقة التنقيط القطري لتحليل A إلى
!>    A = U * D * U**T,  إذا كان UPLO = 'U', أو
!>    A = L * D * L**T,  إذا كان UPLO = 'L',
!> حيث U (أو L) هو حاصل ضرب مصفوفات التبديل والمصفوفات المثلثية العلوية (السفلية) الأحادية، و D مصفوفة متناظرة وقطرية كتلية مع كتل قطرية 1 × 1 و 2 × 2.
!>
!> يتم استدعاء SSYTRF_ROOK لحساب تحليل مصفوفة متناظرة حقيقية A باستخدام طريقة التنقيط القطري المحظورة لـ Bunch-Kaufman ().
!>
!> يتم استخدام الشكل المحلل لـ A لحل نظام
!> المعادلات A * X = B عن طريق استدعاء SSYTRS_ROOK.
!>

المعلمات

UPLO 
!>          UPLO هو CHARACTER*1
!>          = 'U':  يتم تخزين الجزء العلوي من A؛
!>          = 'L':  يتم تخزين الجزء السفلي من A؛
!>

N

!>          N هو INTEGER
!>          عدد المعادلات الخطية، أي رتبة المصفوفة A.  N >= 0.
!>

NRHS

!>          NRHS هو INTEGER
!>          عدد الأطراف اليمنى، أي عدد أعمدة المصفوفة B.  NRHS >= 0.
!>

A

!>          A هي مصفوفة REAL، البعد (LDA,N)
!>          عند الإدخال، المصفوفة المتناظرة A.  إذا كان UPLO = 'U'، فإن الجزء العلوي
!>          المثلثي N × N من A يحتوي على الجزء العلوي
!>          المثلثي للمصفوفة A، والجزء السفلي الصارم
!>          المثلثي لـ A لا تتم الإشارة إليه.  إذا كان UPLO = 'L'، فإن الجزء
!>          العلوي N × N من A يحتوي على الجزء السفلي
!>          المثلثي للمصفوفة A، والجزء العلوي الصارم
!>          المثلثي لـ A لا تتم الإشارة إليه.
!>
!>          عند الخروج، إذا كان INFO = 0، فإن المصفوفة القطرية الكتلية D والمضروبات
!>          المستخدمة للحصول على العامل U أو L من
!>          التحليل A = U*D*U**T أو A = L*D*L**T كما تم حسابها بواسطة
!>          SSYTRF_ROOK.
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرئيسي للمصفوفة A.  LDA >= max(1,N).
!>

IPIV

!>          IPIV هي مصفوفة INTEGER، البعد (N)
!>          تفاصيل التبادلات والهيكل الكتلي لـ D،
!>          كما تم تحديدها بواسطة SSYTRF_ROOK.
!>
!>          إذا كان UPLO = 'U':
!>               إذا كان IPIV(k) > 0، فإن الصفوف والأعمدة k و IPIV(k)
!>               تم تبديلها و D(k,k) هي كتلة قطرية 1 × 1.
!>
!>               إذا كان IPIV(k) < 0 و IPIV(k-1) < 0، فإن الصفوف و
!>               الأعمدة k و -IPIV(k) تم تبديلها والصفوف و
!>               الأعمدة k-1 و -IPIV(k-1) تم تبديلها،
!>               D(k-1:k,k-1:k) هي كتلة قطرية 2 × 2.
!>
!>          إذا كان UPLO = 'L':
!>               إذا كان IPIV(k) > 0، فإن الصفوف والأعمدة k و IPIV(k)
!>               تم تبديلها و D(k,k) هي كتلة قطرية 1 × 1.
!>
!>               إذا كان IPIV(k) < 0 و IPIV(k+1) < 0، فإن الصفوف و
!>               الأعمدة k و -IPIV(k) تم تبديلها والصفوف و
!>               الأعمدة k+1 و -IPIV(k+1) تم تبديلها،
!>               D(k:k+1,k:k+1) هي كتلة قطرية 2 × 2.
!>

B

!>          B هي مصفوفة REAL، البعد (LDB,NRHS)
!>          عند الإدخال، المصفوفة B ذات الطرف الأيمن N × NRHS.
!>          عند الخروج، إذا كان INFO = 0، فإن المصفوفة X الحل N × NRHS.
!>

LDB

!>          LDB هو INTEGER
!>          البعد الرئيسي للمصفوفة B.  LDB >= max(1,N).
!>

WORK

!>          WORK هي مصفوفة REAL، البعد (MAX(1,LWORK))
!>          عند الخروج، إذا كان INFO = 0، فإن WORK(1) يعيد LWORK الأمثل.
!>

LWORK

!>          LWORK هو INTEGER
!>          طول WORK.  LWORK >= 1، ولأفضل أداء
!>          LWORK >= max(1,N*NB)، حيث NB هو الحجم الكتلي الأمثل لـ
!>          SSYTRF_ROOK.
!>
!>          سيتم إجراء TRS باستخدام BLAS المستوى 2
!>
!>          إذا كان LWORK = -1، فيفترض أنه استعلام عن مساحة عمل؛ تقوم الروتين فقط
!>          بحساب الحجم الأمثل لمصفوفة WORK، وتعيد
!>          هذه القيمة كالإدخال الأول لمصفوفة WORK، ولا يتم إصدار رسالة خطأ
!>          متعلقة بـ LWORK بواسطة XERBLA.
!>

INFO

!>          INFO هو INTEGER
!>          = 0: خروج ناجح
!>          < 0: إذا كان INFO = -i، فإن الوسيط i لديه قيمة غير قانونية
!>          > 0: إذا كان INFO = i، فإن D(i,i) يساوي صفر تمامًا. تم اكتمال التحليل،
!>               ولكن المصفوفة القطرية الكتلية D منفردة تمامًا، لذلك لم يكن من الممكن حساب الحل.
!>
