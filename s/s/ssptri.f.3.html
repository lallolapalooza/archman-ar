subroutine ssptri (character uplo, integer n, real, dimension( * ) ap, integer, dimension( * ) ipiv, real, dimension( * ) work, integer info)

SSPTRI

الغرض:

!>
!> SSPTRI يحسب معكوس مصفوفة حقيقية متناظرة غير محددة
!> A بتخزين مضغوط باستخدام التحليل A = U*D*U**T أو
!> A = L*D*L**T المحسوب بواسطة SSPTRF.
!>

المعاملات

UPLO 
!>          UPLO هو CHARACTER*1
!>          يحدد ما إذا كانت تفاصيل التحليل مخزنة
!>          كمصفوفة مثلثية عليا أو سفلى.
!>          = 'U':  مثلثية عليا، الشكل هو A = U*D*U**T;
!>          = 'L':  مثلثية سفلى، الشكل هو A = L*D*L**T.
!>

N

!>          N هو INTEGER
!>          رتبة المصفوفة A.  N >= 0.
!>

AP

!>          AP هو REAL array, dimension (N*(N+1)/2)
!>          عند الإدخال، المصفوفة القطرية الماسية D والمضاعفات
!>          المستخدمة للحصول على العامل U أو L كما تم حسابه بواسطة SSPTRF،
!>          مخزنة كمصفوفة مثلثية مضغوطة.
!>
!>          عند الخروج، إذا كان INFO = 0، معكوس المصفوفة الأصلية (المتناظرة)
!>          مخزن كمصفوفة مثلثية مضغوطة. يتم تخزين العمود j من inv(A)
!>          في المصفوفة AP كما يلي:
!>          إذا كان UPLO = 'U'، AP(i + (j-1)*j/2) = inv(A)(i,j) لـ 1<=i<=j؛
!>          إذا كان UPLO = 'L'،
!>             AP(i + (j-1)*(2n-j)/2) = inv(A)(i,j) لـ j<=i<=n.
!>

IPIV

!>          IPIV هو INTEGER array, dimension (N)
!>          تفاصيل التبادلات وهيكل D الماسي كما تم تحديده بواسطة SSPTRF.
!>

WORK

!>          WORK هو REAL array, dimension (N)
!>

INFO

!>          INFO هو INTEGER
!>          = 0: خروج ناجح
!>          < 0: إذا كان INFO = -i، كان المعامل i له قيمة غير قانونية
!>          > 0: إذا كان INFO = i، فإن D(i,i) = 0؛ المصفوفة مفردة ولا يمكن حساب معكوسها.
!>
