توثيق الوظيفة/الإجراء الفرعي

subroutine sstein (integer n, real, dimension( * ) d, real, dimension( * ) e, integer m, real, dimension( * ) w, integer, dimension( * ) iblock, integer, dimension( * ) isplit, real, dimension( ldz, * ) z, integer ldz, real, dimension( * ) work, integer, dimension( * ) iwork, integer, dimension( * ) ifail, integer info)

SSTEIN

الغرض:

!>
!> SSTEIN يحسب المتجهات الذاتية لمصفوفة ثلاثية الأقطار متماثلة حقيقية T
!> المقابلة للقيم الذاتية المحددة، باستخدام التكرار العكسي.
!>
!> الحد الأقصى لعدد التكرارات المسموح بها لكل متجه ذاتي محدد بواسطة
!> معلمة داخلية MAXITS (حاليا مضبوطة على 5).
!>

المعلمات:

N 
!>          N هو عدد صحيح (INTEGER)
!>          ترتيب المصفوفة. N >= 0.
!>

D

!>          D هو مصفوفة حقيقية (REAL array)، البعد (N)
!>          العناصر القطرية n للمصفوفة ثلاثية الأقطار T.
!>

E

!>          E هو مصفوفة حقيقية (REAL array)، البعد (N-1)
!>          العناصر تحت القطرية (n-1) للمصفوفة ثلاثية الأقطار
!>          T، في العناصر من 1 إلى N-1.
!>

M

!>          M هو عدد صحيح (INTEGER)
!>          عدد المتجهات الذاتية المطلوب إيجادها. 0 <= M <= N.
!>

W

!>          W هو مصفوفة حقيقية (REAL array)، البعد (N)
!>          العناصر M الأولى في W تحتوي على القيم الذاتية التي
!>          سيتم حساب المتجهات الذاتية المقابلة لها. يجب تجميع القيم الذاتية
!>          حسب الكتلة المنفصلة وترتيبها من الأصغر إلى الأكبر ضمن الكتلة.
!>          (نتوقع مصفوفة الإخراج W من SSTEBZ مع ORDER = 'B' هنا.)
!>

IBLOCK

!>          IBLOCK هي مصفوفة أعداد صحيحة (INTEGER array)، البعد (N)
!>          فهارس المصفوفات الفرعية المرتبطة بالقيم الذاتية المقابلة
!>          في W؛ IBLOCK(i)=1 إذا كانت القيمة الذاتية W(i) تنتمي إلى
!>          المصفوفة الفرعية الأولى من الأعلى، =2 إذا كانت W(i) تنتمي إلى
!>          المصفوفة الفرعية الثانية، إلخ.
!>          (نتوقع مصفوفة الإخراج IBLOCK من SSTEBZ هنا.)
!>

ISPLIT

!>          ISPLIT هي مصفوفة أعداد صحيحة (INTEGER array)، البعد (N)
!>          نقاط الفصل، التي تنقسم عندها T إلى مصفوفات فرعية.
!>          تتكون المصفوفة الفرعية الأولى من الصفوف/الأعمدة 1 إلى
!>          ISPLIT( 1 )، الثانية من الصفوف/الأعمدة ISPLIT( 1 )+1
!>          حتى ISPLIT( 2 )، وهكذا.
!>          (نتوقع مصفوفة الإخراج ISPLIT من SSTEBZ هنا.)
!>

Z

!>          Z هي مصفوفة حقيقية (REAL array)، البعد (LDZ, M)
!>          المتجهات الذاتية المحسوبة. يتم تخزين المتجه الذاتي المرتبط
!>          بالقيمة الذاتية W(i) في العمود i من Z.
!>          أي متجه يفشل في التقارب يتم ضبطه على تكراره الحالي
!>          بعد MAXITS تكرارات.
!>

LDZ

!>          LDZ هو عدد صحيح (INTEGER)
!>          البعد الرائد للمصفوفة Z. LDZ >= max(1,N).
!>

WORK

!>          WORK هي مصفوفة حقيقية (REAL array)، البعد (5*N)
!>

IWORK

!>          IWORK هي مصفوفة أعداد صحيحة (INTEGER array)، البعد (N)
!>

IFAIL

!>          IFAIL هي مصفوفة أعداد صحيحة (INTEGER array)، البعد (M)
!>          عند الخروج الطبيعي، تكون جميع عناصر IFAIL صفرًا.
!>          إذا فشل متجه ذاتي واحد أو أكثر في التقارب بعد
!>          MAXITS تكرارات، يتم تخزين فهارسها في
!>          المصفوفة IFAIL.
!>

INFO

!>          INFO هو عدد صحيح (INTEGER)
!>          = 0: خروج ناجح.
!>          < 0: إذا كان INFO = -i، فإن المعلمة i كان لها قيمة غير قانونية
!>          > 0: إذا كان INFO = i، فإن i متجه ذاتي فشل في التقارب في
!>               MAXITS تكرارات. يتم تخزين فهارسها في
!>               المصفوفة IFAIL.
!>

المعلمات الداخلية:

!>  MAXITS  عدد صحيح (INTEGER)، الافتراضي = 5
!>          الحد الأقصى لعدد التكرارات المنفذة.
!>
!>  EXTRA   عدد صحيح (INTEGER)، الافتراضي = 2
!>          عدد التكرارات المنفذة بعد تلبية معيار نمو المعيار، يجب أن يكون على الأقل 1.
!>
