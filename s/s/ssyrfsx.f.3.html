subroutine ssyrfsx (character uplo, character equed, integer n, integer nrhs, real, dimension( lda, * ) a, integer lda, real, dimension( ldaf, * ) af, integer ldaf, integer, dimension( * ) ipiv, real, dimension( * ) s, real, dimension( ldb, * ) b, integer ldb, real, dimension( ldx, * ) x, integer ldx, real rcond, real, dimension( * ) berr, integer n_err_bnds, real, dimension( nrhs, * ) err_bnds_norm, real, dimension( nrhs, * ) err_bnds_comp, integer nparams, real, dimension( * ) params, real, dimension( * ) work, integer, dimension( * ) iwork, integer info)

SSYRFSX

الغرض:

!>
!>    SSYRFSX تقوم بتحسين الحل المحسوب لنظام المعادلات الخطية
!>    عندما تكون مصفوفة المعاملات متماثلة غير محددة، وتوفر حدود خطأ وتقديرات
!>    للخطأ العكسي للحل. بالإضافة إلى حد خطأ المعيار، يوفر الرمز
!>    حدود خطأ قصوى مكونة إذا كان ذلك ممكنًا. انظر التعليقات لـ
!>    ERR_BNDS_NORM و ERR_BNDS_COMP لتفاصيل حدود الخطأ.
!>
!>    قد يكون نظام المعادلات الخطية الأصلي قد تم موازنته
!>    قبل استدعاء هذا الروتين، كما هو موضح بواسطة الوسيطات EQUED و S
!>    أدناه. في هذه الحالة، يتم إرجاع الحل وحدود الخطأ
!>    للنظام الأصلي غير الموزون.
!> 
!>     بعض المعلمات الاختيارية مجمعة في مصفوفة PARAMS. هذه
!>     الإعدادات تحدد كيفية إجراء التحسين، ولكن غالبًا ما تكون
!>     الإعدادات الافتراضية مقبولة. إذا كانت الإعدادات الافتراضية مقبولة، يمكن للمستخدمين
!>     تمرير NPARAMS = 0 مما يمنع الكود المصدري من الوصول
!>     إلى وسيطة PARAMS.
!>

المعلمات

UPLO 
!>          UPLO هي CHARACTER*1
!>       = 'U':  يتم تخزين المثلث العلوي من A؛
!>       = 'L':  يتم تخزين المثلث السفلي من A.
!>

EQUED

!>          EQUED هي CHARACTER*1
!>     تحدد شكل الموازنة التي تم إجراؤها على A
!>     قبل استدعاء هذا الروتين. هذا ضروري لحساب
!>     الحل وحدود الخطأ بشكل صحيح.
!>       = 'N':  لا توجد موازنة
!>       = 'Y':  تمت موازنة الصفوف والأعمدة، أي تم استبدال A بـ
!>               diag(S) * A * diag(S).
!>               تم تغيير الجانب الأيمن B وفقًا لذلك.
!>

N

!>          N هو INTEGER
!>     رتبة المصفوفة A. N >= 0.
!>

NRHS

!>          NRHS هو INTEGER
!>     عدد الجوانب اليمنى، أي عدد الأعمدة
!>     للمصفوفتين B و X. NRHS >= 0.
!>

A

!>          A هي مصفوفة REAL، البعد (LDA,N)
!>     المصفوفة المتماثلة A. إذا كان UPLO = 'U'، فإن الجزء العلوي
!>     المثلثي من A الذي يبلغ حجمه N*N يحتوي على الجزء العلوي
!>     المثلثي للمصفوفة A، ولا يتم الرجوع إلى الجزء السفلي
!>     المثلثي الصارم من A. إذا كان UPLO = 'L'، فإن الجزء السفلي
!>     المثلثي من A الذي يبلغ حجمه N*N يحتوي على الجزء السفلي
!>     المثلثي للمصفوفة A، ولا يتم الرجوع إلى الجزء العلوي
!>     المثلثي الصارم من A.
!>

LDA

!>          LDA هو INTEGER
!>     البعد الرائد للمصفوفة A. LDA >= max(1,N).
!>

AF

!>          AF هي مصفوفة REAL، البعد (LDAF,N)
!>     الشكل المكسور للمصفوفة A. تحتوي AF على المصفوفة
!>     القطرية الفرعية D والعوامل المستخدمة للحصول على
!>     المثلث U أو L من تكسير A = U*D*U**T أو A =
!>     L*D*L**T كما تم حسابه بواسطة SSYTRF.
!>

LDAF

!>          LDAF هو INTEGER
!>     البعد الرائد للمصفوفة AF. LDAF >= max(1,N).
!>

IPIV

!>          IPIV هي مصفوفة INTEGER، البعد (N)
!>     تفاصيل التبادلات والبنية الفرعية لـ D
!>     كما تم تحديده بواسطة SSYTRF.
!>

S

!>          S هي مصفوفة REAL، البعد (N)
!>     عوامل القياس لـ A. إذا كان EQUED = 'Y'، يتم ضرب A على
!>     اليسار واليمين بواسطة diag(S). S هي وسيطة إدخال إذا كانت FACT =
!>     'F'. بخلاف ذلك، S هي وسيطة إخراج. إذا كانت FACT = 'F' و EQUED
!>     = 'Y'، يجب أن يكون كل عنصر من عناصر S موجبًا. إذا كانت S إخراج، فإن كل عنصر
!>     من عناصر S هو قوة للجذر. إذا كانت S إدخال، فيجب أن يكون كل عنصر
!>     من عناصر S قوة للجذر لضمان حل موثوق به
!>     وتقديرات الخطأ. لا تسبب الموازنة بالقوى للجذر
!>     أخطاء تقريب ما لم تنخفض النتيجة أو تفيض.
!>     تؤدي أخطاء التقريب أثناء الموازنة إلى التحسين بمصفوفة
!>     غير مكافئة للمصفوفة المدخلة، مما ينتج عنه تقديرات خطأ
!>     قد لا تكون موثوقة.
!>

B

!>          B هي مصفوفة REAL، البعد (LDB,NRHS)
!>     مصفوفة الجانب الأيمن B.
!>

LDB

!>          LDB هو INTEGER
!>     البعد الرائد للمصفوفة B. LDB >= max(1,N).
!>

X

!>          X هي مصفوفة REAL، البعد (LDX,NRHS)
!>     عند الدخول، مصفوفة الحل X، كما تم حسابها بواسطة SGETRS.
!>     عند الخروج، مصفوفة الحل المحسنة X.
!>

LDX

!>          LDX هو INTEGER
!>     البعد الرائد للمصفوفة X. LDX >= max(1,N).
!>

RCOND

!>          RCOND هو REAL
!>     مقلوب عدد الشرط المقاس. هذا تقدير لـ
!>     مقلوب عدد الشرط Skeel للمصفوفة A بعد
!>     الموازنة (إذا تمت). إذا كان هذا أقل من دقة الجهاز
!>     (على وجه الخصوص، إذا كان صفرًا)، فإن المصفوفة مفردة
!>     بدقة العمل. لاحظ أن الخطأ قد لا يزال صغيرًا حتى
!>     إذا كان هذا الرقم صغيرًا جدًا وتبدو المصفوفة غير مشروطة.
!>

BERR

!>          BERR هي مصفوفة REAL، البعد (NRHS)
!>     الخطأ العكسي النسبي المكون. هذا هو
!>     الخطأ العكسي النسبي المكون لكل متجه حل X(j)
!>     (أي، أصغر تغيير نسبي في أي عنصر من A أو B يجعل
!>     X(j) حلاً دقيقًا).
!>

N_ERR_BNDS

!>          N_ERR_BNDS هو INTEGER
!>     عدد حدود الخطأ التي يجب إرجاعها لكل جانب أيمن
!>     ولكل نوع (معيار أو مكون). انظر ERR_BNDS_NORM و
!>     ERR_BNDS_COMP أدناه.
!>

ERR_BNDS_NORM

!>          ERR_BNDS_NORM هي مصفوفة REAL، البعد (NRHS, N_ERR_BNDS)
!>     لكل جانب أيمن، تحتوي هذه المصفوفة على معلومات حول
!>     حدود الخطأ المختلفة وأعداد الشرط المقابلة لـ
!>     الخطأ النسبي المعياري، والذي يتم تعريفه على النحو التالي:
!>
!>     الخطأ النسبي المعياري في متجه الحل i:
!>             max_j (abs(XTRUE(j,i) - X(j,i)))
!>            ------------------------------
!>                  max_j abs(X(j,i))
!>
!>     يتم فهرست المصفوفة حسب نوع معلومات الخطأ كما هو موضح
!>     أدناه. يوجد حاليًا ما يصل إلى ثلاث قطع من المعلومات
!>     يتم إرجاعها.
!>
!>     الفهرس الأول في ERR_BNDS_NORM(i,:) يتوافق مع الجانب الأيمن i.
!>
!>     الفهرس الثاني في ERR_BNDS_NORM(:,err) يحتوي على الحقول الثلاثة التالية:
!>     err = 1  قيمة منطقية. ثق بالإجابة
