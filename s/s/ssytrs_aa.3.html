توثيق الوظيفة/الروتين الفرعي

subroutine ssytrs_aa (character uplo, integer n, integer nrhs, real, dimension( lda, * ) a, integer lda, integer, dimension( * ) ipiv, real, dimension( ldb, * ) b, integer ldb, real, dimension( * ) work, integer lwork, integer info)

SSYTRS_AA

الغرض:

!>
!> SSYTRS_AA يحل نظام المعادلات الخطية A*X = B حيث A مصفوفة حقيقية متماثلة
!> باستخدام التحليل A = U**T*T*U أو A = L*T*L**T الذي تم حسابه بواسطة SSYTRF_AA.
!>

المعلمات

UPLO 
!>          UPLO هو CHARACTER*1
!>          يحدد ما إذا كانت تفاصيل التحليل مخزنة
!>          كمصفوفة مثلثية عليا أو سفلى.
!>          = 'U':  مثلثية عليا، الشكل هو A = U**T*T*U؛
!>          = 'L':  مثلثية سفلى، الشكل هو A = L*T*L**T.
!>

N

!>          N هو INTEGER
!>          رتبة المصفوفة A.  N >= 0.
!>

NRHS

!>          NRHS هو INTEGER
!>          عدد الأطراف اليمنى، أي عدد أعمدة
!>          المصفوفة B.  NRHS >= 0.
!>

A

!>          A هي مصفوفة REAL، ذات البعد (LDA,N)
!>          تفاصيل العوامل التي تم حسابها بواسطة SSYTRF_AA.
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرائد للمصفوفة A.  LDA >= max(1,N).
!>

IPIV

!>          IPIV هي مصفوفة INTEGER، ذات البعد (N)
!>          تفاصيل التبادلات كما تم حسابها بواسطة SSYTRF_AA.
!>

B

!>          B هي مصفوفة REAL، ذات البعد (LDB,NRHS)
!>          عند الدخول، مصفوفة الطرف الأيمن B.
!>          عند الخروج، مصفوفة الحل X.
!>

LDB

!>          LDB هو INTEGER
!>          البعد الرائد للمصفوفة B.  LDB >= max(1,N).
!>

WORK

!>          WORK هي مصفوفة REAL، ذات البعد (MAX(1,LWORK))
!>

LWORK

!>          LWORK هو INTEGER
!>          بعد المصفوفة WORK.
!>          إذا كان MIN(N,NRHS) = 0، فإن LWORK >= 1، وإلا فإن LWORK >= 3*N-2.
!>
!>          إذا كان LWORK = -1، فسيتم افتراض استعلام مساحة العمل؛ الروتين
!>          يحسب فقط الحد الأدنى لحجم مصفوفة WORK، ويعيد
!>          هذه القيمة كإدخال أول في مصفوفة WORK، ولا يتم إصدار أي رسالة خطأ
!>          متعلقة بـ LWORK بواسطة XERBLA.
!>

INFO

!>          INFO هو INTEGER
!>          = 0:  اكتمال ناجح
!>          < 0:  إذا كان INFO = -i، فقد كان الوسيط i له قيمة غير قانونية
!>
