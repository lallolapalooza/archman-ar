subroutine ssysv_aa (character uplo, integer n, integer nrhs, real, dimension( lda, * ) a, integer lda, integer, dimension( * ) ipiv, real, dimension( ldb, * ) b, integer ldb, real, dimension( * ) work, integer lwork, integer info)

SSYSV_AA تحسب حل نظام المعادلات الخطية A * X = B لمصفوفات SY

الغرض:

!>
!> SSYSV تحسب حل نظام المعادلات الخطية الحقيقي
!>    A * X = B,
!> حيث A مصفوفة متناظرة N x N و X و B مصفوفتان N x NRHS.
!>
!> تستخدم خوارزمية Aasen لتحليل A إلى
!>    A = U**T * T * U,  إذا كان UPLO = 'U', أو
!>    A = L * T * L**T,  إذا كان UPLO = 'L',
!> حيث U (أو L) هي حاصل ضرب مصفوفات تبديلية ومصفوفات علوية (سفلية) أحادية القطر,
!> و T هي مصفوفة متناظرة ثلاثية الأقطار. ثم تستخدم الصيغة المحللة لـ A
!> لحل نظام المعادلات A * X = B.
!>

المعاملات

UPLO 
!>          UPLO هي CHARACTER*1
!>          = 'U':  الجزء العلوي من A مخزن؛
!>          = 'L':  الجزء السفلي من A مخزن.
!>

N

!>          N هي INTEGER
!>          عدد المعادلات الخطية، أي رتبة المصفوفة A.  N >= 0.
!>

NRHS

!>          NRHS هي INTEGER
!>          عدد الأطراف اليمنى، أي عدد أعمدة المصفوفة B.  NRHS >= 0.
!>

A

!>          A هي مصفوفة REAL، ذات أبعاد (LDA,N)
!>          عند الإدخال، المصفوفة المتناظرة A.  إذا كان UPLO = 'U'، فإن الجزء
!>          العلوي ذو الأبعاد N x N من A يحتوي على الجزء العلوي
!>          المثلثي من المصفوفة A، والجزء السفلي الصارم
!>          من A لا يتم الرجوع إليه.  إذا كان UPLO = 'L'، فإن الجزء
!>          السفلي ذو الأبعاد N x N من A يحتوي على الجزء السفلي
!>          المثلثي من المصفوفة A، والجزء العلوي الصارم
!>          من A لا يتم الرجوع إليه.
!>
!>          عند الخروج، إذا كان INFO = 0، فإن المصفوفة ثلاثية الأقطار T
!>          والمضاعفات المستخدمة للحصول على العامل U أو L من
!>          التحليل A = U**T*T*U أو A = L*T*L**T كما تم حسابها بواسطة
!>          SSYTRF.
!>

LDA

!>          LDA هي INTEGER
!>          البعد الرائد للمصفوفة A.  LDA >= max(1,N).
!>

IPIV

!>          IPIV هي مصفوفة INTEGER، ذات أبعاد (N)
!>          عند الخروج، تحتوي على تفاصيل التبادلات، أي
!>          تم تبديل الصف والعمود k من A مع
!>          الصف والعمود IPIV(k).
!>

B

!>          B هي مصفوفة REAL، ذات أبعاد (LDB,NRHS)
!>          عند الإدخال، مصفوفة الطرف الأيمن B ذات الأبعاد N x NRHS.
!>          عند الخروج، إذا كان INFO = 0، فإن مصفوفة الحل X ذات الأبعاد N x NRHS.
!>

LDB

!>          LDB هي INTEGER
!>          البعد الرائد للمصفوفة B.  LDB >= max(1,N).
!>

WORK

!>          WORK هي مصفوفة REAL، ذات أبعاد (MAX(1,LWORK))
!>          عند الخروج، إذا كان INFO = 0، فإن WORK(1) تعيد قيمة LWORK المثلى.
!>

LWORK

!>          LWORK هي INTEGER
!>          طول WORK.  LWORK >= MAX(1,2*N,3*N-2), و
!>          للحصول على أفضل أداء، LWORK >= MAX(1,N*NB), حيث NB
!>          هي حجم الكتلة الأمثل لـ SSYTRF_AA.
!>
!>          إذا كان LWORK = -1، فيفترض استعلام عن مساحة العمل؛ الروتين
!>          يحسب فقط الحجم الأمثل لمصفوفة WORK، ويعيد
!>          هذه القيمة كإدخال أول لمصفوفة WORK، ولا يصدر أي رسالة خطأ
!>          متعلقة بـ LWORK بواسطة XERBLA.
!>

INFO

!>          INFO هي INTEGER
!>          = 0: خروج ناجح
!>          < 0: إذا كان INFO = -i، فإن المعامل i كان له قيمة غير قانونية
!>          > 0: إذا كان INFO = i، فإن D(i,i) يساوي صفر بالضبط. تم اكتمال التحليل،
!>               ولكن المصفوفة القطرية الجزئية D منفردة تمامًا، لذلك لم يكن من الممكن حساب الحل.
!>
