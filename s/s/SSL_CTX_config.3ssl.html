SSL_CTX_config(3ssl)

الحزمة: openssl

التراخيص: Apache-2.0

SSL_CTX_CONFIG(3ssl) OpenSSL SSL_CTX_CONFIG(3ssl)

الاسم

SSL_CTX_config, SSL_config - تكوين هيكل SSL_CTX أو SSL

الملخص

#include <openssl/ssl.h>
int SSL_CTX_config(SSL_CTX *ctx, const char *name);
int SSL_config(SSL *s, const char *name);

الوصف

تقوم الدالتان SSL_CTX_config() و SSL_config() بتكوين هيكل SSL_CTX أو SSL باستخدام اسم التكوين name.

من خلال استدعاء SSL_CTX_config() أو SSL_config() يمكن للتطبيق تنفيذ العديد من المهام المعقدة بناءً على محتويات ملف التكوين: مما يبسط بشكل كبير كود تكوين التطبيق. يمكن أيضًا تحقيق درجة من المستقبلية: يمكن للتطبيق دعم ميزات التكوين في الإصدارات الأحدث من OpenSSL تلقائيًا.

يجب أن يكون ملف تكوين قد تم تحميله مسبقًا، على سبيل المثال باستخدام CONF_modules_load_file(). انظر config(5) للحصول على تفاصيل صيغة ملف التكوين.

قيم الإرجاع

SSL_CTX_config() و SSL_config() ترجعان 1 للنجاح أو 0 في حالة حدوث خطأ.

أمثلة

إذا كان الملف "config.cnf" يحتوي على ما يلي:

testapp = test_sect
[test_sect]
# list of configuration modules
ssl_conf = ssl_sect
[ssl_sect]
server = server_section
[server_section]
RSA.Certificate = server-rsa.pem
ECDSA.Certificate = server-ecdsa.pem
Ciphers = ALL:!RC4

يمكن للتطبيق استدعاء:

if (CONF_modules_load_file("config.cnf", "testapp", 0) <= 0) {
    fprintf(stderr, "Error processing config file\n");
    goto err;
}
ctx = SSL_CTX_new(TLS_server_method());
if (SSL_CTX_config(ctx, "server") == 0) {
    fprintf(stderr, "Error configuring server.\n");
    goto err;
}

في هذا المثال، تم تكوين شهادتين وقائمة التشفير دون الحاجة إلى أي كود تطبيق إضافي.

انظر أيضاً

ssl(7)، config(5)، SSL_CONF_cmd(3)، CONF_modules_load_file(3)
