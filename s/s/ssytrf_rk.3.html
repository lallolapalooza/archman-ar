توثيق الوظيفة/الروتين الفرعي

subroutine ssytrf_rk (character uplo, integer n, real, dimension( lda, * ) a, integer lda, real, dimension( * ) e, integer, dimension( * ) ipiv, real, dimension( * ) work, integer lwork, integer info)

SSYTRF_RK تحسب تحليل مصفوفة متماثلة غير محددة حقيقية باستخدام طريقة المحور القطري المقيد من بونش-كوفمان (طريقة الكومة) (خوارزمية BLAS3 المحظورة).

الغرض:

!> SSYTRF_RK تحسب تحليل مصفوفة متماثلة حقيقية A
!> باستخدام طريقة المحور القطري المقيد من بونش-كوفمان (طريقة الكومة):
!>
!>    A = P*U*D*(U**T)*(P**T) أو A = P*L*D*(L**T)*(P**T),
!>
!> حيث U (أو L) هي مصفوفة مثلثية علوية (أو سفلية) موحدة،
!> U**T (أو L**T) هي منقولة U (أو L)، P هي مصفوفة تبديل،
!> P**T هي منقولة P، و D هي مصفوفة متماثلة وقطرية كتلية
!> ذات كتل قطرية بحجم 1x1 و 2x2.
!>
!> هذه هي النسخة المحظورة من الخوارزمية، والتي تستدعي BLAS من المستوى 3.
!> لمزيد من المعلومات انظر قسم التفاصيل الإضافية.
!>

المعلمات

UPLO 
!>          UPLO هو CHARACTER*1
!>          تحدد ما إذا كان الجزء العلوي أو السفلي المثلثي للمصفوفة المتماثلة A
!>          مخزنًا:
!>          = 'U':  مثلثية علوية
!>          = 'L':  مثلثية سفلية
!>

N

!>          N هو INTEGER
!>          رتبة المصفوفة A.  N >= 0.
!>

A

!>          A هي مصفوفة REAL، أبعادها (LDA,N)
!>          عند الإدخال، المصفوفة المتماثلة A.
!>            إذا كان UPLO = 'U': الجزء المثلثي العلوي N×N الرائد
!>            من A يحتوي على الجزء المثلثي العلوي للمصفوفة A،
!>            والجزء السفلي المثلثي الصارم من A لا يتم الرجوع إليه.
!>
!>            إذا كان UPLO = 'L': الجزء المثلثي السفلي N×N الرائد
!>            من A يحتوي على الجزء المثلثي السفلي للمصفوفة A،
!>            والجزء العلوي المثلثي الصارم من A لا يتم الرجوع إليه.
!>
!>          عند الخروج، تحتوي على:
!>            أ) فقط العناصر القطرية للمصفوفة المتماثلة القطرية الكتلية
!>               D على قطر A، أي D(k,k) = A(k,k)؛
!>               (العناصر فوق القطرية (أو تحت القطرية) لـ D
!>                مخزنة عند الخروج في المصفوفة E)، و
!>            ب) إذا كان UPLO = 'U': العامل U في الجزء فوق القطري لـ A.
!>               إذا كان UPLO = 'L': العامل L في الجزء تحت القطري لـ A.
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرائد للمصفوفة A.  LDA >= max(1,N).
!>

E

!>          E هي مصفوفة REAL، أبعادها (N)
!>          عند الخروج، تحتوي على العناصر فوق القطرية (أو تحت القطرية)
!>          للمصفوفة المتماثلة القطرية الكتلية D
!>          ذات الكتل القطرية بحجم 1x1 أو 2x2، حيث
!>          إذا كان UPLO = 'U': E(i) = D(i-1,i)، i=2:N، E(1) مضبوط على 0؛
!>          إذا كان UPLO = 'L': E(i) = D(i+1,i)، i=1:N-1، E(N) مضبوط على 0.
!>
!>          ملاحظة: بالنسبة للكتلة القطرية بحجم 1x1 D(k)، حيث
!>          1 <= k <= N، يتم ضبط العنصر E(k) على 0 في كل من
!>          حالات UPLO = 'U' أو UPLO = 'L'.
!>

IPIV

!>          IPIV هي مصفوفة INTEGER، أبعادها (N)
!>          تصف IPIV مصفوفة التبديل P في تحليل
!>          المصفوفة A كالتالي. تمثل القيمة المطلقة لـ IPIV(k)
!>          مؤشر الصف والعمود اللذين تم تبديلهما مع الصف والعمود k.
!>          تصف قيمة UPLO ترتيب تطبيق التبديلات.
!>          أيضًا، إشارة IPIV تمثل البنية الكتلية
!>          للمصفوفة المتماثلة القطرية الكتلية D ذات الكتل
!>          القطرية بحجم 1x1 أو 2x2 والتي تتوافق مع تبديل واحد أو اثنين
!>          في كل خطوة تحليل. لمزيد من المعلومات انظر قسم التفاصيل الإضافية.
!>
!>          إذا كان UPLO = 'U'،
!>          ( بترتيب التحليل، k تتناقص من N إلى 1 ):
!>            أ) إدخال موجب واحد IPIV(k) > 0 يعني:
!>               D(k,k) هي كتلة قطرية بحجم 1x1.
!>               إذا كان IPIV(k) != k، فقد تم تبديل الصفوف والأعمدة k و IPIV(k)
!>               في المصفوفة A(1:N,1:N)؛
!>               إذا كان IPIV(k) = k، لم يحدث تبديل.
!>
!>            ب) زوج من الإدخالات المتتالية السالبة
!>               IPIV(k) < 0 و IPIV(k-1) < 0 يعني:
!>               D(k-1:k,k-1:k) هي كتلة قطرية بحجم 2x2.
!>               (ملاحظة: الإدخالات السالبة في IPIV تظهر فقط في أزواج).
!>               1) إذا كان -IPIV(k) != k، فقد تم تبديل الصفوف والأعمدة
!>                  k و -IPIV(k)
!>                  في المصفوفة A(1:N,1:N).
!>                  إذا كان -IPIV(k) = k، لم يحدث تبديل.
!>               2) إذا كان -IPIV(k-1) != k-1، فقد تم تبديل الصفوف والأعمدة
!>                  k-1 و -IPIV(k-1)
!>                  في المصفوفة A(1:N,1:N).
!>                  إذا كان -IPIV(k-1) = k-1، لم يحدث تبديل.
!>
!>            ج) في كلتا الحالتين أ) وب)، يكون دائمًا ABS( IPIV(k) ) <= k.
!>
!>            د) ملاحظة: أي إدخال IPIV(k) يكون دائمًا غير صفري عند الخروج.
!>
!>          إذا كان UPLO = 'L'،
!>          ( بترتيب التحليل، k تزداد من 1 إلى N ):
!>            أ) إدخال موجب واحد IPIV(k) > 0 يعني:
!>               D(k,k) هي كتلة قطرية بحجم 1x1.
!>               إذا كان IPIV(k) != k، فقد تم تبديل الصفوف والأعمدة k و IPIV(k)
!>               في المصفوفة A(1:N,1:N).
!>               إذا كان IPIV(k) = k، لم يحدث تبديل.
!>
!>            ب) زوج من الإدخالات المتتالية السالبة
!>               IPIV(k) < 0 و IPIV(k+1) < 0 يعني:
!>               D(k:k+1,k:k+1) هي كتلة قطرية بحجم 2x2.
!>               (ملاحظة: الإدخالات السالبة في IPIV تظهر فقط في أزواج).
!>               1) إذا كان -IPIV(k) != k، فقد تم تبديل الصفوف والأعمدة
!>                  k و -IPIV(k)
!>                  في المصفوفة A(1:N,1:N).
!>                  إذا كان -IPIV(k) = k، لم يحدث تبديل.
!>               2) إذا كان -IPIV(k+1) != k+1، فقد تم تبديل الصفوف والأعمدة
!>                  k-1 و -IPIV(k-1)
!>                  في المصفوفة A(1:N,1:N).
!>                  إذا كان -IPIV(k+1) = k+1، لم يحدث تبديل.
!>
!>            ج) في كلتا الحالتين أ) وب)، يكون دائمًا ABS( IPIV(k) ) >= k.
!>
!>            د) ملاحظة: أي إدخال IPIV(k) يكون دائمًا غير صفري عند الخروج.
!>

WORK

!>          WORK هي مصفوفة REAL، أبعادها (MAX(1,LWORK)).
!>          عند الخروج، إذا كان INFO = 0، تعيد WORK(1) قيمة LWORK المثلى.
!>

LWORK

!>          LWORK هو INTEGER
!>          طول WORK.  LWORK >= 1.  للحصول على أفضل أداء
!>          LWORK >= N*NB، حيث NB هو حجم الكتلة الذي تعيده
!>          الدالة ILAENV.
!>
!>          إذا كان LWORK = -1، فسيتم افتراض استعلام مساحة العمل؛
!>          تقوم الروتين فقط بحساب الحجم الأمثل لمصفوفة WORK،
!>          وتعيد هذه القيمة كالإدخال الأول لمصفوفة WORK،
!>          ولا يصدر XERBLA أي رسالة خطأ تتعلق بـ LWORK.
!>

INFO

!>          INFO هو INTEGER
!>          = 0: تم التنفيذ بنجاح
!>
!>          < 0: إذا كان INFO = -k، فإن المعامل k كان له قيمة غير قانونية
!>
!>          > 0: إذا كان INFO = k، فإن المصفوفة A منفردة، بسبب:
!>                 إذا كان UPLO = 'U': العمود k في الجزء
!>                 المثلثي العلوي لـ A يحتوي على جميع الأصفار.
!>                 إذا كان UPLO = 'L': العمود k في الجزء
!>                 المثلثي السفلي لـ A يحتوي على جميع الأصفار.
!>
!>               لذلك فإن D(k,k) تساوي صفرًا تمامًا، والعناصر فوق القطرية
!>               للعمود k من U (أو العناصر تحت القطرية
!>               للعمود k من L) هي جميعها أصفار. تم إكمال التحليل،
!>               ولكن المصفوفة القطرية الكتلية D منفردة تمامًا،
!>               وسيحدث قسمة على صفر إذا تم استخدامها لحل نظام
!>
