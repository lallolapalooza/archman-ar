subroutine ssytri2 (character uplo, integer n, real, dimension( lda, * ) a, integer lda, integer, dimension( * ) ipiv, real, dimension( * ) work, integer lwork, integer info)

SSYTRI2

الغرض:

!>
!> SSYTRI2 يحسب معكوس مصفوفة **REAL** متماثلة غير محددة
!> A باستخدام التحليل A = U*D*U**T أو A = L*D*L**T الذي تم حسابه بواسطة
!> SSYTRF. SSYTRI2 يقوم بتعيين البعد الرائد لمساحة العمل
!> قبل استدعاء SSYTRI2X الذي يحسب المعكوس فعليًا.
!>

المعلمات

UPLO 
!>          UPLO هو CHARACTER*1
!>          يحدد ما إذا كانت تفاصيل التحليل مخزنة
!>          كمصفوفة مثلثة عليا أو سفلى.
!>          = 'U':  مثلثة عليا، الشكل هو A = U*D*U**T؛
!>          = 'L':  مثلثة سفلى، الشكل هو A = L*D*L**T.
!>

N

!>          N هو INTEGER
!>          رتبة المصفوفة A.  N >= 0.
!>

A

!>          A هي مصفوفة REAL، البعد (LDA,N)
!>          عند الدخول، المصفوفة القطرية الكتلية D والمعاملات
!>          المستخدمة للحصول على العامل U أو L كما تم حسابه بواسطة SSYTRF.
!>
!>          عند الخروج، إذا كان INFO = 0، فإن المعكوس (المتماثل) للمصفوفة الأصلية.
!>          إذا كان UPLO = 'U'، فإن الجزء المثلث العلوي من المعكوس
!>          يتم تشكيله والجزء A أسفل القطر لا يتم الرجوع إليه؛ إذا كان UPLO = 'L'
!>          الجزء المثلث السفلي من المعكوس يتم تشكيله والجزء A فوق القطر
!>          لا يتم الرجوع إليه.
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرائد للمصفوفة A.  LDA >= max(1,N).
!>

IPIV

!>          IPIV هي مصفوفة INTEGER، البعد (N)
!>          تفاصيل التبادلات وهيكل الكتلة لـ D
!>          كما تم تحديده بواسطة SSYTRF.
!>

WORK

!>          WORK هي مصفوفة REAL، البعد (MAX(1,LWORK))
!>

LWORK

!>          LWORK هو INTEGER
!>          بعد مصفوفة WORK.
!>          إذا كان N = 0، LWORK >= 1، وإلا LWORK >= (N+NB+1)*(NB+3).
!>          إذا كان LWORK = -1، فإنه يفترض استعلام مساحة العمل؛ الروتين
!>          يحسب:
!>              - الحجم الأمثل لمصفوفة WORK، ويعيد
!>          هذه القيمة كأول إدخال في مصفوفة WORK،
!>              - ولا تصدر رسالة خطأ متعلقة بـ LWORK بواسطة XERBLA.
!>

INFO

!>          INFO هو INTEGER
!>          = 0: خروج ناجح
!>          < 0: إذا كان INFO = -i، فإن المعلمة i كانت قيمة غير قانونية
!>          > 0: إذا كان INFO = i، فإن D(i,i) = 0؛ المصفوفة مفردة ولا يمكن حساب معكوسها.
!>
