توثيق الوظيفة/الروتين الفرعي

subroutine ssygvx (integer itype, character jobz, character range, character uplo, integer n, real, dimension( lda, * ) a, integer lda, real, dimension( ldb, * ) b, integer ldb, real vl, real vu, integer il, integer iu, real abstol, integer m, real, dimension( * ) w, real, dimension( ldz, * ) z, integer ldz, real, dimension( * ) work, integer lwork, integer, dimension( * ) iwork, integer, dimension( * ) ifail, integer info)

SSYGVX

الغرض:

!>
!> SSYGVX يحسب قيمًا ذاتية مختارة، واختياريًا، متجهات ذاتية
!> لمشكلة معكوس متماثلة معرفة عامة، بالشكل
!> A*x=(lambda)*B*x,  A*Bx=(lambda)*x,  or B*A*x=(lambda)*x. هنا A
!> و B مفترضان أن يكونا متماثلين و B موجب محدد أيضًا.
!> يمكن تحديد القيم الذاتية والمتجهات الذاتية عن طريق تحديد إما نطاق
!> من القيم أو نطاق من المؤشرات للقيم الذاتية المطلوبة.
!>

المعلمات

ITYPE 
!>          ITYPE هو عدد صحيح
!>          يحدد نوع المشكلة المطلوب حلها:
!>          = 1:  A*x = (lambda)*B*x
!>          = 2:  A*B*x = (lambda)*x
!>          = 3:  B*A*x = (lambda)*x
!>

JOBZ

!>          JOBZ هو حرف واحد
!>          = 'N':  حساب القيم الذاتية فقط؛
!>          = 'V':  حساب القيم الذاتية والمتجهات الذاتية.
!>

RANGE

!>          RANGE هو حرف واحد
!>          = 'A': سيتم العثور على جميع القيم الذاتية.
!>          = 'V': سيتم العثور على جميع القيم الذاتية في الفترة نصف المفتوحة (VL,VU]
!>                 .
!>          = 'I': سيتم العثور على القيم الذاتية من IL إلى IU.
!>

UPLO

!>          UPLO هو حرف واحد
!>          = 'U':  يتم تخزين الجزء العلوي من A و B؛
!>          = 'L':  يتم تخزين الجزء السفلي من A و B.
!>

N

!>          N هو عدد صحيح
!>          ترتيب القلم (A,B).  N >= 0.
!>

A

!>          A هي مصفوفة حقيقية، البعد (LDA, N)
!>          عند الإدخال، المصفوفة المتماثلة A. إذا كان UPLO = 'U'، فإن
!>          الجزء العلوي المثلثي من A بحجم N*N يحتوي على
!>          الجزء العلوي المثلثي للمصفوفة A. إذا كان UPLO = 'L'، فإن
!>          الجزء السفلي المثلثي من A بحجم N*N يحتوي على
!>          الجزء السفلي المثلثي للمصفوفة A.
!>
!>          عند الخروج، الجزء السفلي (إذا كان UPLO='L') أو الجزء العلوي
!>          (إذا كان UPLO='U') من A، بما في ذلك القطر، يتم تدميره.
!>

LDA

!>          LDA هو عدد صحيح
!>          البعد الرائد للمصفوفة A.  LDA >= max(1,N).
!>

B

!>          B هي مصفوفة حقيقية، البعد (LDB, N)
!>          عند الإدخال، المصفوفة المتماثلة B. إذا كان UPLO = 'U'، فإن
!>          الجزء العلوي المثلثي من B بحجم N*N يحتوي على
!>          الجزء العلوي المثلثي للمصفوفة B. إذا كان UPLO = 'L'، فإن
!>          الجزء السفلي المثلثي من B بحجم N*N يحتوي على
!>          الجزء السفلي المثلثي للمصفوفة B.
!>
!>          عند الخروج، إذا كان INFO <= N، فإن الجزء من B الذي يحتوي على المصفوفة
!>          مُعاد كتابته بالعامل المثلثي U أو L من تحلل Cholesky
!>          B = U**T*U أو B = L*L**T.
!>

LDB

!>          LDB هو عدد صحيح
!>          البعد الرائد للمصفوفة B.  LDB >= max(1,N).
!>

VL

!>          VL هو عدد حقيقي
!>          إذا كان RANGE='V'، الحد الأدنى للفترة المراد
!>          البحث فيها عن القيم الذاتية. VL < VU.
!>          لا يتم الاستشهاد به إذا كان RANGE = 'A' أو 'I'.
!>

VU

!>          VU هو عدد حقيقي
!>          إذا كان RANGE='V'، الحد الأعلى للفترة المراد
!>          البحث فيها عن القيم الذاتية. VL < VU.
!>          لا يتم الاستشهاد به إذا كان RANGE = 'A' أو 'I'.
!>

IL

!>          IL هو عدد صحيح
!>          إذا كان RANGE='I'، مؤشر
!>          أصغر قيمة ذاتية سيتم إرجاعها.
!>          1 <= IL <= IU <= N، إذا كان N > 0؛ IL = 1 و IU = 0 إذا كان N = 0.
!>          لا يتم الاستشهاد به إذا كان RANGE = 'A' أو 'V'.
!>

IU

!>          IU هو عدد صحيح
!>          إذا كان RANGE='I'، مؤشر
!>          أكبر قيمة ذاتية سيتم إرجاعها.
!>          1 <= IL <= IU <= N، إذا كان N > 0؛ IL = 1 و IU = 0 إذا كان N = 0.
!>          لا يتم الاستشهاد به إذا كان RANGE = 'A' أو 'V'.
!>

ABSTOL

!>          ABSTOL هو عدد حقيقي
!>          تسامح الخطأ المطلق للقيم الذاتية.
!>          يتم قبول قيمة ذاتية تقريبية على أنها متقاربة
!>          عندما يتم تحديد أنها تقع في فترة [a,b]
!>          بعرض أقل من أو يساوي
!>
!>                  ABSTOL + EPS *   max( |a|,|b| ) ,
!>
!>          حيث EPS هي دقة الآلة. إذا كان ABSTOL أقل من
!>          أو يساوي صفرًا، فسيتم استخدام EPS * |T| بدلاً منه،
!>          حيث |T| هو المعيار 1 للمصفوفة الثلاثية القطرية التي تم الحصول عليها
!>          عن طريق اختزال C إلى شكل ثلاثي القطر، حيث C هي المصفوفة المتماثلة
!>          للمشكلة المتماثلة القياسية التي تم تحويل المشكلة
!>          العامة إليها.
!>
!>          سيتم حساب القيم الذاتية بأكبر دقة عندما يتم تعيين ABSTOL إلى
!>          ضعف عتبة التحويل السفلي 2*DLAMCH('S')، وليس صفرًا.
!>          إذا عادت هذه الروتين الفرعي مع INFO>0، مما يشير إلى أن بعض
!>          المتجهات الذاتية لم تتقارب، حاول تعيين ABSTOL إلى
!>          2*SLAMCH('S').
!>

M

!>          M هو عدد صحيح
!>          العدد الإجمالي للقيم الذاتية التي تم العثور عليها.  0 <= M <= N.
!>          إذا كان RANGE = 'A'، فإن M = N، وإذا كان RANGE = 'I'، فإن M = IU-IL+1.
!>

W

!>          W هي مصفوفة حقيقية، البعد (N)
!>          عند الخروج الطبيعي، تحتوي العناصر الأولى M على القيم الذاتية
!>          المختارة بترتيب تصاعدي.
!>

Z

!>          Z هي مصفوفة حقيقية، البعد (LDZ, max(1,M))
!>          إذا كان JOBZ = 'N'، فلا يتم الاستشهاد بـ Z.
!>          إذا كان JOBZ = 'V'، وإذا كان INFO = 0، فإن الأعمدة M الأولى من Z
!>          تحتوي على المتجهات الذاتية المتعامدة للمصفوفة A
!>          المقابلة للقيم الذاتية المختارة، مع العمود i
!>          من Z يحمل المتجه الذاتي المرتبط بـ W(i).
!>          يتم تطبيع المتجهات الذاتية كالتالي:
!>          إذا كان ITYPE = 1 أو 2، فإن Z**T*B*Z = I؛
!>          إذا كان ITYPE = 3، فإن Z**T*inv(B)*Z = I.
!>
!>          إذا فشل المتجه الذاتي في التقارب، فإن هذا العمود من Z
!>          يحتوي على أحدث تقريب للمتجه الذاتي، ويتم إرجاع
!>          مؤشر المتجه الذاتي في IFAIL.
!>          ملاحظة: يجب على المستخدم التأكد من توفير ما لا يقل عن max(1,M) أعمدة
!>          في المصفوفة Z؛ إذا كان RANGE = 'V'، فإن القيمة الدقيقة لـ M
!>          غير معروفة مقدمًا ويجب استخدام حد أعلى.
!>

LDZ

!>          LDZ هو عدد صحيح
!>          البعد الرائد للمصفوفة Z.  LDZ >= 1، وإذا كان
!>          JOBZ = 'V'، فإن LDZ >= max(1,N).
!>

WORK

!>          WORK هي مصفوفة حقيقية، البعد (MAX(1,LWORK))
!>          عند الخروج، إذا كان INFO = 0، فإن WORK(1) تُرجع LWORK الأمثل.
!>

LWORK

!>          LWORK هو عدد صحيح
!>          طول المصفوفة WORK.  LWORK >= max(1,8*N).
!>          للكفاءة المثلى، LWORK >= (NB+3)*N،
!>          حيث NB هو حجم الكتلة لـ SSYTRD الذي تم إرجاعه بواسطة ILAENV.
!>
!>          إذا كان LWORK = -1، فسيتم تخصيص مساحة العمل
