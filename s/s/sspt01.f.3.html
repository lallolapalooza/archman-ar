subroutine sspt01 (character uplo, integer n, real, dimension( * ) a, real, dimension( * ) afac, integer, dimension( * ) ipiv, real, dimension( ldc, * ) c, integer ldc, real, dimension( * ) rwork, real resid)

SSPT01

الغرض:

!>
!> SSPT01 تعيد بناء مصفوفة متماثلة غير محددة A من تحليلها L*D*L' أو U*D*U'
!> وتحسب الباقي
!>      norm( C - A ) / ( N * norm(A) * EPS ),
!> حيث C هي المصفوفة المعاد بناؤها و EPS هي مدخلات الآلة.
!>

المعلمات

UPLO 
!>          UPLO is CHARACTER*1
!>          يحدد ما إذا كان الجزء العلوي أو السفلي المثلثي للمصفوفة
!>          المتماثلة A مخزنًا:
!>          = 'U':  علوي مثلثي
!>          = 'L':  سفلي مثلثي
!>

N

!>          N is INTEGER
!>          عدد الصفوف والأعمدة للمصفوفة A.  N >= 0.
!>

A

!>          A is REAL array, dimension (N*(N+1)/2)
!>          المصفوفة المتماثلة الأصلية A، مخزنة كمصفوفة
!>          مثلثة مدمجة.
!>

AFAC

!>          AFAC is REAL array, dimension (N*(N+1)/2)
!>          الشكل المفكك للمصفوفة A، مخزن كمصفوفة
!>          مثلثة مدمجة.  AFAC يحتوي على المصفوفة القطرية الكتلية D
!>          والعوامل المستخدمة للحصول على العامل L أو U من
!>          تحليل L*D*L' أو U*D*U' كما تم حسابه بواسطة SSPTRF.
!>

IPIV

!>          IPIV is INTEGER array, dimension (N)
!>          فهارس المحور من SSPTRF.
!>

C

!>          C is REAL array, dimension (LDC,N)
!>

LDC

!>          LDC is INTEGER
!>          البعد الرائد للمصفوفة C.  LDC >= max(1,N).
!>

RWORK

!>          RWORK is REAL array, dimension (N)
!>

RESID

!>          RESID is REAL
!>          إذا كان UPLO = 'L'، norm(L*D*L' - A) / ( N * norm(A) * EPS )
!>          إذا كان UPLO = 'U'، norm(U*D*U' - A) / ( N * norm(A) * EPS )
!>
