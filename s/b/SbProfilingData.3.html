SbProfilingData(3)

الحزمة: coin

التراخيص: BSD-3-Clause

الاسم

SbProfilingData — معلومات حول الملف الشخصي.

الملخص

#include <Inventor/SbProfilingData.h>

#include <Inventor/details/SoDetail.h>

#include <Inventor/paths/SoPath.h>

#include <Inventor/SbBool.h>

#include <cstdio>

class SbProfilingData {
public:
  SbProfilingData(void);
  SbProfilingData(const SbProfilingData & rhs);
  virtual ~SbProfilingData(void);
  const SoDetail * getNodeDetail(int idx) const;
  int getNodeTiming(const SoPath * path, unsigned int flags = 0) const;
  int getIndex(const SoPath * path, SbBool create = FALSE);
  int getParentIndex(int idx) const;
  void reset(void);
  SbProfilingData & operator=(const SbProfilingData & rhs);
  SbProfilingData & operator+=(const SbProfilingData & rhs);
  size_t getProfilingDataSize(void) const;
};

الوصف التفصيلي

هذه الفئة هي كائن تفصيلي يستخدم مع <a href="SoProfilingData.html">SoProfilingData</a>.

توثيق وظائف الأعضاء

const SoDetail * SbProfilingData::getNodeDetail (int idx) const

يعيد مؤشرًا إلى تفاصيل العقدة في الفهرس المحدد.

int SbProfilingData::getNodeTiming (const SoPath * path, unsigned int flags = 0) const

يعيد التوقيت لعقدة.

int SbProfilingData::getIndex (const SoPath * path, SbBool create = FALSE)

يعيد فهرس العقدة الذيلية في المسار. إذا لم يتم تسجيل العقدة، فقم بإضافتها وإعادة هذا الفهرس.

int SbProfilingData::getParentIndex (int idx) const

يعيد فهرس الأصل لإدخال العقدة في الفهرس idx. إذا كان الإدخال إدخالًا جذريًا، فسيتم إرجاع -1.

void SbProfilingData::reset (void )

إزالة جميع البيانات المخزنة.

SbProfilingData & SbProfilingData::operator= (const SbProfilingData & rhs)

عامل التعيين.

SbProfilingData & SbProfilingData::operator+= (const SbProfilingData & rhs)

إضافة بيانات الملف الشخصي من مجموعة بيانات أخرى.

size_t SbProfilingData::getProfilingDataSize (void ) const

يعيد كمية الذاكرة المخصصة لهيكل البيانات هذا.
