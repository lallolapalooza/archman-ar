توثيق الوظيفة/الروتين الفرعي

subroutine sbdt01 (integer m, integer n, integer kd, real, dimension( lda, * ) a, integer lda, real, dimension( ldq, * ) q, integer ldq, real, dimension( * ) d, real, dimension( * ) e, real, dimension( ldpt, * ) pt, integer ldpt, real, dimension( * ) work, real resid)

SBDT01

الغرض:

!>
!> SBDT01 يعيد بناء مصفوفة عامة A من شكلها الثنائي القطري
!>    A = Q * B * P**T
!> حيث Q (m بـ min(m,n)) و P**T (min(m,n) بـ n) مصفوفات متعامدة
!> و B ثنائية القطر.
!>
!> نسبة الاختبار لاختبار الاختزال هي
!>    RESID = norm(A - Q * B * P**T) / ( n * norm(A) * EPS )
!> حيث EPS الدقة الآلية.
!>

المعلمات

M 
!>          M هو عدد صحيح (INTEGER)
!>          عدد صفوف المصفوفات A و Q.
!>

N

!>          N هو عدد صحيح (INTEGER)
!>          عدد أعمدة المصفوفات A و P**T.
!>

KD

!>          KD هو عدد صحيح (INTEGER)
!>          إذا كان KD = 0، فإن B قطرية والمصفوفة E لا يتم الرجوع إليها.
!>          إذا كان KD = 1، تم إجراء الاختزال بواسطة xGEBRD؛ B علوية
!>          ثنائية القطر إذا كان M >= N، وسفلية ثنائية القطر إذا كان M < N.
!>          إذا كان KD = -1، تم إجراء الاختزال بواسطة xGBBRD؛ B
!>          دائمًا علوية ثنائية القطر.
!>

A

!>          A هي مصفوفة حقيقية (REAL array)، البعد (LDA,N)
!>          المصفوفة A بقياس m × n.
!>

LDA

!>          LDA هو عدد صحيح (INTEGER)
!>          البعد الرئيسي للمصفوفة A. LDA >= max(1,M).
!>

Q

!>          Q هي مصفوفة حقيقية (REAL array)، البعد (LDQ,N)
!>          المصفوفة المتعامدة Q بقياس m × min(m,n) في الاختزال
!>          A = Q * B * P**T.
!>

LDQ

!>          LDQ هو عدد صحيح (INTEGER)
!>          البعد الرئيسي للمصفوفة Q. LDQ >= max(1,M).
!>

D

!>          D هي مصفوفة حقيقية (REAL array)، البعد (min(M,N))
!>          العناصر القطرية للمصفوفة ثنائية القطر B.
!>

E

!>          E هي مصفوفة حقيقية (REAL array)، البعد (min(M,N)-1)
!>          العناصر فوق القطرية للمصفوفة ثنائية القطر B إذا كان
!>          m >= n، أو العناصر تحت القطرية لـ B إذا كان m < n.
!>

PT

!>          PT هي مصفوفة حقيقية (REAL array)، البعد (LDPT,N)
!>          المصفوفة المتعامدة P**T بقياس min(m,n) × n في الاختزال
!>          A = Q * B * P**T.
!>

LDPT

!>          LDPT هو عدد صحيح (INTEGER)
!>          البعد الرئيسي للمصفوفة PT.
!>          LDPT >= max(1,min(M,N)).
!>

WORK

!>          WORK هي مصفوفة حقيقية (REAL array)، البعد (M+N)
!>

RESID

!>          RESID هي قيمة حقيقية (REAL)
!>          نسبة الاختبار:
!>          norm(A - Q * B * P**T) / ( n * norm(A) * EPS )
!>
