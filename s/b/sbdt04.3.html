توثيق الوظيفة/الروتين الفرعي

subroutine sbdt04 (character uplo, integer n, real, dimension( * ) d, real, dimension( * ) e, real, dimension( * ) s, integer ns, real, dimension( ldu, * ) u, integer ldu, real, dimension( ldvt, * ) vt, integer ldvt, real, dimension( * ) work, real resid)

SBDT04

الغرض:

!>
!> SBDT04 يعيد بناء مصفوفة ثنائية القطر B من تحليلها (الجزئي) للقيمة المفردة SVD:
!>    S = U' * B * V
!> حيث U و V مصفوفتان متعامدتان و S مصفوفة قطرية.
!>
!> نسبة الاختبار لاختبار تحليل القيمة المفردة هي
!>    RESID = norm( S - U' * B * V ) / ( n * norm(B) * EPS )
!> حيث VT = V' و EPS هي دقة الآلة.
!>

المعلمات

UPLO 
!>          UPLO is CHARACTER*1
!>          تحدد ما إذا كانت المصفوفة B علوية أو سفلية ثنائية القطر.
!>          = 'U':  علوية ثنائية القطر
!>          = 'L':  سفلية ثنائية القطر
!>

N

!>          N is INTEGER
!>          رتبة المصفوفة B.
!>

D

!>          D is REAL array, dimension (N)
!>          عناصر القطر الرئيسية n للمصفوفة ثنائية القطر B.
!>

E

!>          E is REAL array, dimension (N-1)
!>          العناصر فوق القطرية n-1 للمصفوفة B إذا كان UPLO = 'U'، أو العناصر
!>          تحت القطرية n-1 للمصفوفة B إذا كان UPLO = 'L'.
!>

S

!>          S is REAL array, dimension (NS)
!>          القيم المفردة من تحليل القيمة المفردة (الجزئي) لـ B، مرتبة بترتيب تنازلي.
!>

NS

!>          NS is INTEGER
!>          عدد القيم/المتجهات المفردة من تحليل القيمة المفردة (الجزئي) لـ B.
!>

U

!>          U is REAL array, dimension (LDU,NS)
!>          المصفوفة المتعامدة U ذات الأبعاد n × ns في S = U' * B * V.
!>

LDU

!>          LDU is INTEGER
!>          البعد الرائد للمصفوفة U. LDU >= max(1,N)
!>

VT

!>          VT is REAL array, dimension (LDVT,N)
!>          المصفوفة المتعامدة V ذات الأبعاد n × ns في S = U' * B * V.
!>

LDVT

!>          LDVT is INTEGER
!>          البعد الرائد للمصفوفة VT.
!>

WORK

!>          WORK is REAL array, dimension (2*N)
!>

RESID

!>          RESID is REAL
!>          نسبة الاختبار: norm(S - U' * B * V) / ( n * norm(B) * EPS )
!>
