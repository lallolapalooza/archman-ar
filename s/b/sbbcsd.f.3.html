subroutine sbbcsd (character jobu1, character jobu2, character jobv1t, character jobv2t, character trans, integer m, integer p, integer q, real, dimension( * ) theta, real, dimension( * ) phi, real, dimension( ldu1, * ) u1, integer ldu1, real, dimension( ldu2, * ) u2, integer ldu2, real, dimension( ldv1t, * ) v1t, integer ldv1t, real, dimension( ldv2t, * ) v2t, integer ldv2t, real, dimension( * ) b11d, real, dimension( * ) b11e, real, dimension( * ) b12d, real, dimension( * ) b12e, real, dimension( * ) b21d, real, dimension( * ) b21e, real, dimension( * ) b22d, real, dimension( * ) b22e, real, dimension( * ) work, integer lwork, integer info)

SBBCSD

الغرض:

!>
!> تحسب SBBCSD تحليل CS لمصفوفة متعامدة في
!> شكل الكتلة ثنائية القطر،
!>
!>
!>     [ B11 | B12 0  0 ]
!>     [  0  |  0 -I  0 ]
!> X = [----------------]
!>     [ B21 | B22 0  0 ]
!>     [  0  |  0  0  I ]
!>
!>                               [  C | -S  0  0 ]
!>                   [ U1 |    ] [  0 |  0 -I  0 ] [ V1 |    ]**T
!>                 = [---------] [---------------] [---------]   .
!>                   [    | U2 ] [  S |  C  0  0 ] [    | V2 ]
!>                               [  0 |  0  0  I ]
!>
!> X هي M في M، وكتلتها العلوية اليسرى هي P في Q، ويجب ألا يكون Q أكبر
!> من P أو M-P أو M-Q. (إذا لم يكن Q أصغر فهرس، فيجب نقل X
!> و/أو تبديله. يمكن القيام بذلك في وقت ثابت باستخدام
!> خيارات TRANS وSIGNS. راجع SORCSD للتفاصيل.)
!>
!> يتم تمثيل المصفوفات ثنائية القطر B11 وB12 وB21 وB22
!> ضمنيًا بواسطة الزوايا THETA(1:Q) وPHI(1:Q-1).
!>
!> المصفوفات المتعامدة U1 وU2 وV1T وV2T هي إدخال/إخراج.
!> يتم ضرب المصفوفات المدخلة مسبقًا أو لاحقًا بمصفوفات
!> المتجهات المفردة المناسبة.
!>

المعاملات

JOBU1 
!>          JOBU1 هو CHARACTER
!>          = 'Y':      يتم تحديث U1;
!>          خلاف ذلك:  لا يتم تحديث U1.
!>

JOBU2

!>          JOBU2 هو CHARACTER
!>          = 'Y':      يتم تحديث U2;
!>          خلاف ذلك:  لا يتم تحديث U2.
!>

JOBV1T

!>          JOBV1T هو CHARACTER
!>          = 'Y':      يتم تحديث V1T;
!>          خلاف ذلك:  لا يتم تحديث V1T.
!>

JOBV2T

!>          JOBV2T هو CHARACTER
!>          = 'Y':      يتم تحديث V2T;
!>          خلاف ذلك:  لا يتم تحديث V2T.
!>

TRANS

!>          TRANS هو CHARACTER
!>          = 'T':      يتم تخزين X وU1 وU2 وV1T وV2T بترتيب
!>                      الصفوف الرئيسي;
!>          خلاف ذلك:  يتم تخزين X وU1 وU2 وV1T وV2T بترتيب
!>                      الأعمدة الرئيسي.
!>

M

!>          M هو INTEGER
!>          عدد الصفوف والأعمدة في X، المصفوفة المتعامدة في
!>          شكل الكتلة ثنائية القطر.
!>

P

!>          P هو INTEGER
!>          عدد الصفوف في الكتلة العلوية اليسرى من X. 0 <= P <= M.
!>

Q

!>          Q هو INTEGER
!>          عدد الأعمدة في الكتلة العلوية اليسرى من X.
!>          0 <= Q <= MIN(P,M-P,M-Q).
!>

THETA

!>          THETA هو مصفوفة REAL، البعد (Q)
!>          عند الإدخال، الزوايا THETA(1),...,THETA(Q) التي، جنبًا إلى جنب مع
!>          PHI(1), ...,PHI(Q-1)، تحدد المصفوفة في شكل الكتلة
!>          ثنائية القطر. عند الإخراج، الزوايا التي تحدد جيوب تمامها
!>          وجيوبها الكتل القطرية في تحليل CS.
!>

PHI

!>          PHI هو مصفوفة REAL، البعد (Q-1)
!>          الزوايا PHI(1),...,PHI(Q-1) التي، جنبًا إلى جنب مع THETA(1),...,
!>          THETA(Q)، تحدد المصفوفة في شكل الكتلة ثنائية القطر.
!>

U1

!>          U1 هو مصفوفة REAL، البعد (LDU1,P)
!>          عند الإدخال، مصفوفة P في P. عند الإخراج، يتم ضرب U1 لاحقًا
!>          بمصفوفة المتجهات المفردة اليسرى المشتركة بين [ B11 ; 0 ] و
!>          [ B12 0 0 ; 0 -I 0 0 ].
!>

LDU1

!>          LDU1 هو INTEGER
!>          البعد الرئيسي للمصفوفة U1، LDU1 >= MAX(1,P).
!>

U2

!>          U2 هو مصفوفة REAL، البعد (LDU2,M-P)
!>          عند الإدخال، مصفوفة (M-P) في (M-P). عند الإخراج، يتم ضرب U2
!>          لاحقًا بمصفوفة المتجهات المفردة اليسرى المشتركة بين
!>          [ B21 ; 0 ] و [ B22 0 0 ; 0 0 I ].
!>

LDU2

!>          LDU2 هو INTEGER
!>          البعد الرئيسي للمصفوفة U2، LDU2 >= MAX(1,M-P).
!>

V1T

!>          V1T هو مصفوفة REAL، البعد (LDV1T,Q)
!>          عند الإدخال، مصفوفة Q في Q. عند الإخراج، يتم ضرب V1T مسبقًا
!>          بمنقول مصفوفة المتجهات المفردة اليمنى
!>          المشتركة بين [ B11 ; 0 ] و [ B21 ; 0 ].
!>

LDV1T

!>          LDV1T هو INTEGER
!>          البعد الرئيسي للمصفوفة V1T، LDV1T >= MAX(1,Q).
!>

V2T

!>          V2T هو مصفوفة REAL، البعد (LDV2T,M-Q)
!>          عند الإدخال، مصفوفة (M-Q) في (M-Q). عند الإخراج، يتم ضرب V2T
!>          مسبقًا بمنقول مصفوفة المتجهات المفردة اليمنى
!>          المشتركة بين [ B12 0 0 ; 0 -I 0 ] و
!>          [ B22 0 0 ; 0 0 I ].
!>

LDV2T

!>          LDV2T هو INTEGER
!>          البعد الرئيسي للمصفوفة V2T، LDV2T >= MAX(1,M-Q).
!>

B11D

!>          B11D هو مصفوفة REAL، البعد (Q)
!>          عندما تتقارب SBBCSD، تحتوي B11D على جيوب تمام THETA(1),
!>          ..., THETA(Q). إذا فشلت SBBCSD في التقارب، فإن B11D
!>          تحتوي على قطر الكتلة العلوية اليسرى
!>          المخفضة جزئيًا.
!>

B11E

!>          B11E هو مصفوفة REAL، البعد (Q-1)
!>          عندما تتقارب SBBCSD، تحتوي B11E على أصفار. إذا فشلت SBBCSD
!>          في التقارب، فإن B11E تحتوي على القطر الفوقي للكتلة
!>          العلوية اليسرى المخفضة جزئيًا.
!>

B12D

!>          B12D هو مصفوفة REAL، البعد (Q)
!>          عندما تتقارب SBBCSD، تحتوي B12D على جيوب الجيب السالبة لـ
!>          THETA(1), ..., THETA(Q). إذا فشلت SBBCSD في التقارب، فإن
!>          B12D تحتوي على قطر الكتلة العلوية اليمنى
!>          المخفضة جزئيًا.
!>

B12E

!>          B12E هو مصفوفة REAL، البعد (Q-1)
!>          عندما تتقارب SBBCSD، تحتوي B12E على أصفار. إذا فشلت SBBCSD
!>          في التقارب، فإن B12E تحتوي على القطر التحتي للكتلة
!>          العلوية اليمنى المخفضة جزئيًا.
!>

B21D

!>          B21D هو مصفوفة REAL، البعد (Q)
!>          عندما تتقارب SBBCSD، تحتوي B21D على جيوب الجيب السالبة لـ
!>          THETA(1), ..., THETA(Q). إذا فشلت SBBCSD في التقارب، فإن
!>          B21D تحتوي على قطر الكتلة السفلية اليسرى
!>          المخفضة جزئيًا.
!>

B21E

!>          B21E هو مصفوفة REAL، البُعد (Q-1)
!>          عندما تتقارب SBBCSD، تحتوي B21E على أصفار. إذا فشلت SBBCSD
!>          في التقارب، فإن B21E تحتوي على القطر الفرعي للكتلة
!>          السفلية اليسرى المخفّضة جزئياً.
!>

B22D

!>          B22D هو مصفوفة REAL، البُعد (Q)
!>          عندما تتقارب SBBCSD، تحتوي B22D على جيوب THETA(1)، ...،
!>          THETA(Q) السالبة. إذا فشلت SBBCSD في التقارب، فإن
!>          B22D تحتوي على قطر الكتلة السفلية اليمنى المخفّضة جزئياً.
!>

B22E

!>          B22E هو مصفوفة REAL، البُعد (Q-1)
!>          عندما تتقارب SBBCSD، تحتوي B22E على أصفار. إذا فشلت SBBCSD
!>          في التقارب، فإن B22E تحتوي على القطر الفرعي للكتلة
!>          السفلية اليمنى المخفّضة جزئياً.
!>

WORK

!>          WORK هو مصفوفة REAL، البُعد (MAX(1,LWORK))
!>          عند الخروج، إذا كان INFO = 0، فإن WORK(1) يُرجع LWORK الأمثل.
!>

LWORK

!>          LWORK هو INTEGER
!>          بُعد المصفوفة WORK. LWORK >= MAX(1,8*Q).
!>
!>          إذا كان LWORK = -1، فسيُفترض استعلام عن مساحة العمل؛
!>          فالروتين يحسب فقط الحجم الأمثل لمصفوفة WORK،
!>          ويُرجع هذه القيمة كأول عنصر في مصفوفة العمل، ولا
!>          تُصدر رسالة خطأ متعلقة بـ LWORK من قبل XERBLA.
!>

INFO

!>          INFO هو INTEGER
!>          = 0:  خروج ناجح.
!>          < 0:  إذا كان INFO = -i، فإن الوسيط رقم i له قيمة غير قانونية.
!>          > 0:  إذا لم تتقارب SBBCSD، فإن INFO يحدد عدد
!>                العناصر غير الصفرية في PHI، وتحتوي B11D، B11E، إلخ،
!>                على المصفوفة المخفّضة جزئياً.
!>

المعاملات الداخلية:

!>  TOLMUL  REAL، القيمة الافتراضية = MAX(10,MIN(100,EPS**(-1/8)))
!>          يتحكم TOLMUL في معيار التقارب لحلقة QR.
!>          تُقرَّب الزوايا THETA(i)، PHI(i) إلى 0 أو PI/2 عندما
!>          تكون ضمن TOLMUL*EPS من أي من الحدين.
!>
