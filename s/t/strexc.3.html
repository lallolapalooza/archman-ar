توثيق الوظيفة/الروتين الفرعي

subroutine strexc (character compq, integer n, real, dimension( ldt, * ) t, integer ldt, real, dimension( ldq, * ) q, integer ldq, integer ifst, integer ilst, real, dimension( * ) work, integer info)

STREXC

الغرض:

!>
!> STREXC يعيد ترتيب شجرة Schur الحقيقية لمصفوفة حقيقية
!> A = Q*T*Q**T، بحيث يتم نقل الكتلة القطرية لـ T ذات فهرس الصف IFST
!> إلى الصف ILST.
!>
!> يتم إعادة ترتيب شكل Schur الحقيقي T عن طريق تحويل تشابه متعامد
!> Z**T*T*Z، ويمكن اختياريًا تحديث مصفوفة Q لأشعة Schur
!> عن طريق ضربها لاحقًا بـ Z.
!>
!> يجب أن تكون T في شكل Schur القانوني (كما تم إرجاعها بواسطة SHSEQR)، أي
!> الكتلة المثلثية العلوية مع كتل قطرية 1x1 و 2x2؛ كل
!> كتلة قطرية 2x2 لها عناصرها القطرية متساوية والعناصر
!> خارج القطر ذات إشارات متعاكسة.
!>

المعاملات

COMPQ 
!>          COMPQ هو CHARACTER*1
!>          = 'V':  تحديث مصفوفة Q لأشعة Schur؛
!>          = 'N':  لا تقم بتحديث Q.
!>

N

!>          N هو INTEGER
!>          رتبة المصفوفة T. N >= 0.
!>          إذا كانت N == 0، فقد تكون وسيطات ILST و IFST أي قيمة.
!>

T

!>          T هي مصفوفة REAL، ذات أبعاد (LDT,N)
!>          عند الدخول، المصفوفة شبه المثلثية العلوية T، في شكل Schur
!>          القانوني.
!>          عند الخروج، المصفوفة شبه المثلثية العلوية المعاد ترتيبها، مرة أخرى
!>          في شكل Schur القانوني.
!>

LDT

!>          LDT هو INTEGER
!>          البعد الرائد للمصفوفة T. LDT >= max(1,N).
!>

Q

!>          Q هي مصفوفة REAL، ذات أبعاد (LDQ,N)
!>          عند الدخول، إذا كانت COMPQ = 'V'، مصفوفة Q لأشعة Schur.
!>          عند الخروج، تم ضرب Q لاحقًا بالمصفوفة التحويلية المتعامدة Z
!>          التي تعيد ترتيب T.
!>          إذا كانت COMPQ = 'N'، لا يتم الرجوع إلى Q.
!>

LDQ

!>          LDQ هو INTEGER
!>          البعد الرائد للمصفوفة Q. LDQ >= 1، وإذا
!>          كانت COMPQ = 'V'، فإن LDQ >= max(1,N).
!>

IFST

!>          IFST هو INTEGER
!>

ILST

!>          ILST هو INTEGER
!>
!>          تحديد إعادة ترتيب الكتل القطرية لـ T.
!>          يتم نقل الكتلة ذات فهرس الصف IFST إلى الصف ILST، عن طريق
!>          سلسلة من عمليات التبديل بين الكتل المجاورة.
!>          عند الخروج، إذا كان IFST يشير عند الدخول إلى الصف الثاني لكتلة
!>          2x2، فإنه يتغير ليشير إلى الصف الأول؛ يشير ILST دائمًا
!>          إلى الصف الأول للكتلة في موقعها النهائي
!>          (والذي قد يختلف عن قيمته المدخلة بـ +1 أو -1).
!>          1 <= IFST <= N؛ 1 <= ILST <= N.
!>

WORK

!>          WORK هي مصفوفة REAL، ذات أبعاد (N)
!>

INFO

!>          INFO هو INTEGER
!>          = 0:  خروج ناجح
!>          < 0:  إذا كان INFO = -i، فإن الوسيط i كان له قيمة غير قانونية
!>          = 1:  كانت هناك كتلتان متجاورتان قريبتين جدًا للتبديل (المشكلة
!>                غير مستقرة للغاية)؛ قد تكون T قد أعيد ترتيبها جزئيًا،
!>                ويشير ILST إلى الصف الأول للموقع الحالي للكتلة التي يتم نقلها.
!>
