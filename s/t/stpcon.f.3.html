subroutine stpcon (character norm, character uplo, character diag, integer n, real, dimension( * ) ap, real rcond, real, dimension( * ) work, integer, dimension( * ) iwork, integer info)

STPCON

الغرض:

!>
!> STPCON يقدر مقلوب رقم الشرط لمصفوفة مثلثة معبأة A، سواء في المعيار 1-norm أو
!> المعيار اللانهائي.
!>
!> يتم حساب معيار A ويتم الحصول على تقدير لـ
!> norm(inv(A))، ثم يتم حساب مقلوب رقم الشرط كـ
!>    RCOND = 1 / ( norm(A) * norm(inv(A)) ).
!>

المعاملات

NORM 
!>          NORM is CHARACTER*1
!>          يحدد ما إذا كان رقم الشرط 1-norm أو رقم الشرط اللانهائي مطلوبًا:
!>          = '1' or 'O':  1-norm;
!>          = 'I':         Infinity-norm.
!>

UPLO

!>          UPLO is CHARACTER*1
!>          = 'U':  A is upper triangular;
!>          = 'L':  A is lower triangular.
!>

DIAG

!>          DIAG is CHARACTER*1
!>          = 'N':  A is non-unit triangular;
!>          = 'U':  A is unit triangular.
!>

N

!>          N is INTEGER
!>          رتبة المصفوفة A.  N >= 0.
!>

AP

!>          AP is REAL array, dimension (N*(N+1)/2)
!>          المصفوفة المثلثية العليا أو السفلى A، معبأة عموديًا في
!>          مصفوفة خطية.  يتم تخزين العمود j من A في
!>          المصفوفة AP كالتالي:
!>          if UPLO = 'U', AP(i + (j-1)*j/2) = A(i,j) for 1<=i<=j;
!>          if UPLO = 'L', AP(i + (j-1)*(2n-j)/2) = A(i,j) for j<=i<=n.
!>          If DIAG = 'U', يتم تجاهل العناصر القطرية لـ A
!>          ويفترض أنها 1.
!>

RCOND

!>          RCOND is REAL
!>          مقلوب رقم شرط المصفوفة A،
!>          محسوب كـ RCOND = 1/(norm(A) * norm(inv(A))).
!>

WORK

!>          WORK is REAL array, dimension (3*N)
!>

IWORK

!>          IWORK is INTEGER array, dimension (N)
!>

INFO

!>          INFO is INTEGER
!>          = 0:  successful exit
!>          < 0:  if INFO = -i, the i-th argument had an illegal value
!>
