توثيق الدالة/الروتين الفرعي

subroutine strti2 (character uplo, character diag, integer n, real, dimension( lda, * ) a, integer lda, integer info)

STRTI2 تحسب معكوس مصفوفة مثلثة (خوارزمية غير مجمعة).

الغرض:

!>
!> STRTI2 تحسب معكوس مصفوفة حقيقية علوية أو سفلية مثلثة.
!>
!> هذه هي النسخة من المستوى 2 من الخوارزمية BLAS.
!>

المعاملات

UPLO 
!>          UPLO هو CHARACTER*1
!>          تحدد ما إذا كانت المصفوفة A علوية أم سفلية مثلثة.
!>          = 'U':  علوية مثلثة
!>          = 'L':  سفلية مثلثة
!>

DIAG

!>          DIAG هو CHARACTER*1
!>          تحدد ما إذا كانت المصفوفة A وحدة مثلثة أم لا.
!>          = 'N':  غير وحدة مثلثة
!>          = 'U':  وحدة مثلثة
!>

N

!>          N هو INTEGER
!>          ترتيب المصفوفة A.  N >= 0.
!>

A

!>          A هو مصفوفة REAL، بُعدها (LDA,N)
!>          عند الإدخال، المصفوفة المثلثة A.  إذا كان UPLO = 'U'، فإن
!>          الجزء العلوي المثلثي العلوي الرئيسي بمقاس n × n في المصفوفة A يحتوي
!>          على المصفوفة المثلثة العلوية، والجزء السفلي المثلثي الصارم
!>          من A لا يتم الرجوع إليه.  إذا كان UPLO = 'L'، فإن
!>          الجزء السفلي المثلثي السفلي الرئيسي بمقاس n × n في المصفوفة A يحتوي
!>          على المصفوفة المثلثة السفلية، والجزء العلوي المثلثي الصارم
!>          من A لا يتم الرجوع إليه.  إذا كان DIAG = 'U'، فإن
!>          العناصر القطرية لـ A لا يتم الرجوع إليها أيضًا ويفترض أنها 1.
!>
!>          عند الخروج، معكوس (المثلثة) للمصفوفة الأصلية، في
!>          نفس صيغة التخزين.
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرئيسي للمصفوفة A.  LDA >= max(1,N).
!>

INFO

!>          INFO هو INTEGER
!>          = 0: نجاح الخروج
!>          < 0: إذا كان INFO = -k، فإن المعامل k كان له قيمة غير قانونية
!>
