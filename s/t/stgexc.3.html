subroutine stgexc (logical wantq, logical wantz, integer n, real, dimension( lda, * ) a, integer lda, real, dimension( ldb, * ) b, integer ldb, real, dimension( ldq, * ) q, integer ldq, real, dimension( ldz, * ) z, integer ldz, integer ifst, integer ilst, real, dimension( * ) work, integer lwork, integer info)

STGEXC

الغرض:

!>
!> تعيد STGEXC ترتيب التحلل الهرمي الحقيقي المعمم لمصفوفة زوج (A,B)
!> باستخدام تحويل تعادل متعامد
!>
!>                (A, B) = Q * (A, B) * Z**T,
!>
!> بحيث يتم نقل القطعة القطرية من (A, B) ذات فهرس الصف IFST إلى الصف ILST.
!>
!> يجب أن تكون (A, B) في شكل كانونيكي هرمي حقيقي معمّم (كما تم إرجاعه
!> بواسطة SGGES)، أي أن A هي قطعة مثلثية عليا مع قطع قطرية بحجم 1x1 و 2x2.
!> B هي قطعة مثلثية عليا.
!>
!> بشكل اختياري، يتم تحديث مصفوفتي Q و Z من المتجهات الهرمية المعممة.
!>
!>        Q(in) * A(in) * Z(in)**T = Q(out) * A(out) * Z(out)**T
!>        Q(in) * B(in) * Z(in)**T = Q(out) * B(out) * Z(out)**T
!>
!>

المعلمات

WANTQ 
!>          WANTQ هي LOGICAL
!>          .TRUE. : تحديث مصفوفة التحويل اليسرى Q؛
!>          .FALSE.: لا تحديث Q.
!>

WANTZ

!>          WANTZ هي LOGICAL
!>          .TRUE. : تحديث مصفوفة التحويل اليمنى Z؛
!>          .FALSE.: لا تحديث Z.
!>

N

!>          N هي INTEGER
!>          رتبة المصفوفتين A و B. N >= 0.
!>

A

!>          A هي مصفوفة REAL، الأبعاد (LDA,N)
!>          عند الإدخال، المصفوفة A في شكل هرمي حقيقي معمّم.
!>          عند الخروج، المصفوفة A المحدثة، مرة أخرى في شكل هرمي حقيقي معمّم.
!>

LDA

!>          LDA هي INTEGER
!>          البعد الرئيسي للمصفوفة A. LDA >= max(1,N).
!>

B

!>          B هي مصفوفة REAL، الأبعاد (LDB,N)
!>          عند الإدخال، المصفوفة B في شكل هرمي حقيقي معمّم (A,B).
!>          عند الخروج، المصفوفة B المحدثة، مرة أخرى في شكل هرمي حقيقي معمّم (A,B).
!>

LDB

!>          LDB هي INTEGER
!>          البعد الرئيسي للمصفوفة B. LDB >= max(1,N).
!>

Q

!>          Q هي مصفوفة REAL، الأبعاد (LDQ,N)
!>          عند الإدخال، إذا كان WANTQ = .TRUE.، فإن المصفوفة المتعامدة Q.
!>          عند الخروج، المصفوفة Q المحدثة.
!>          إذا كان WANTQ = .FALSE.، لا يتم الرجوع إلى Q.
!>

LDQ

!>          LDQ هي INTEGER
!>          البعد الرئيسي للمصفوفة Q. LDQ >= 1.
!>          إذا كان WANTQ = .TRUE.، LDQ >= N.
!>

Z

!>          Z هي مصفوفة REAL، الأبعاد (LDZ,N)
!>          عند الإدخال، إذا كان WANTZ = .TRUE.، فإن المصفوفة المتعامدة Z.
!>          عند الخروج، المصفوفة Z المحدثة.
!>          إذا كان WANTZ = .FALSE.، لا يتم الرجوع إلى Z.
!>

LDZ

!>          LDZ هي INTEGER
!>          البعد الرئيسي للمصفوفة Z. LDZ >= 1.
!>          إذا كان WANTZ = .TRUE.، LDZ >= N.
!>

IFST

!>          IFST هي INTEGER
!>

ILST

!>          ILST هي INTEGER
!>          تحديد إعادة ترتيب القطع القطرية لـ (A, B).
!>          يتم نقل القطعة ذات فهرس الصف IFST إلى الصف ILST، عن طريق
!>          سلسلة من التبادلات بين القطع المتجاورة.
!>          عند الخروج، إذا كان IFST يشير عند الإدخال إلى الصف الثاني من
!>          قطعة 2x2، فإنه يتغير للإشارة إلى الصف الأول؛
!>          يشير ILST دائمًا إلى الصف الأول للقطعة في موضعها النهائي
!>          (الذي قد يختلف عن قيمته المدخلة بـ +1 أو -1). 1 <= IFST, ILST <= N.
!>

WORK

!>          WORK هي مصفوفة REAL، الأبعاد (MAX(1,LWORK))
!>          عند الخروج، إذا كان INFO = 0، ترجع WORK(1) قيمة LWORK المثلى.
!>

LWORK

!>          LWORK هي INTEGER
!>          بعد المصفوفة WORK.
!>          LWORK >= 1 عندما N <= 1، وإلا LWORK >= 4*N + 16.
!>
!>          إذا كان LWORK = -1، فيفترض استعلام مساحة العمل؛ تقوم الروتين
!>          بحساب حجم مساحة العمل المثلى فقط، وترجع هذه القيمة كالإدخال الأول
!>          للمصفوفة WORK، ولا يتم إصدار أي رسالة خطأ متعلقة بـ LWORK بواسطة XERBLA.
!>

INFO

!>          INFO هي INTEGER
!>           =0:  خروج ناجح.
!>           <0:  إذا كان INFO = -i، فإن الوسيط i لديه قيمة غير قانونية.
!>           =1:  ستكون المصفوفة المزدوجة المحولة (A, B) بعيدة جدًا
!>                عن الشكل الهرمي المعمم؛ المشكلة غير مستقرة.
!>                قد تكون (A, B) قد تم إعادة ترتيبها جزئيًا،
!>                ويشير ILST إلى الصف الأول من الموضع الحالي
!>                للقطعة التي يتم نقلها.
!>
