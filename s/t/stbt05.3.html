توثيق الوظيفة/الروتين الفرعي

subroutine stbt05 (character uplo, character trans, character diag, integer n, integer kd, integer nrhs, real, dimension( ldab, * ) ab, integer ldab, real, dimension( ldb, * ) b, integer ldb, real, dimension( ldx, * ) x, integer ldx, real, dimension( ldxact, * ) xact, integer ldxact, real, dimension( * ) ferr, real, dimension( * ) berr, real, dimension( * ) reslts)

STBT05

الغرض:

!>
!> تختبر STBT05 حدود الخطأ من التنقيح التكراري للحل المحسوب لنظام
!> المعادلات A*X = B، حيث A عبارة عن مصفوفة نطاق مثلثة.
!>
!> RESLTS(1) = اختبار حد الخطأ
!>           = norm(X - XACT) / ( norm(X) * FERR )
!>
!> يتم إرجاع قيمة كبيرة إذا لم تكن هذه النسبة أقل من واحد.
!>
!> RESLTS(2) = الباقي من روتين التنقيح التكراري
!>           = الحد الأقصى لـ BERR / ( NZ*EPS + (*) )، حيث
!>             (*) = NZ*UNFL / (min_i (abs(A)*abs(X) +abs(b))_i )
!>             و NZ = الحد الأقصى لعدد العناصر غير الصفرية في أي صف من A، بالإضافة إلى 1
!>

المعلمات

UPLO 
!>          UPLO is CHARACTER*1
!>          يحدد ما إذا كانت المصفوفة A علوية أو سفلية مثلثة.
!>          = 'U':  علوية مثلثة
!>          = 'L':  سفلية مثلثة
!>

TRANS

!>          TRANS is CHARACTER*1
!>          يحدد شكل نظام المعادلات.
!>          = 'N':  A * X = B  (بدون تبديل)
!>          = 'T':  A'* X = B  (تبديل)
!>          = 'C':  A'* X = B  (تبديل مرافق = تبديل)
!>

DIAG

!>          DIAG is CHARACTER*1
!>          يحدد ما إذا كانت المصفوفة A مثلثة وحدوية أم لا.
!>          = 'N':  غير وحدوية مثلثة
!>          = 'U':  وحدوية مثلثة
!>

N

!>          N is INTEGER
!>          عدد صفوف المصفوفات X، B، و XACT، وترتيب المصفوفة A.  N >= 0.
!>

KD

!>          KD is INTEGER
!>          عدد الأقطار الفائقة للمصفوفة A إذا كان UPLO = 'U'،
!>          أو عدد الأقطار الفرعية إذا كان UPLO = 'L'.  KD >= 0.
!>

NRHS

!>          NRHS is INTEGER
!>          عدد أعمدة المصفوفات X، B، و XACT.
!>          NRHS >= 0.
!>

AB

!>          AB is REAL array, dimension (LDAB,N)
!>          مصفوفة النطاق المثلثة العلوية أو السفلية A، مخزنة في
!>          أول kd+1 صفوف من المصفوفة. يتم تخزين العمود j من A
!>          في العمود j من المصفوفة AB على النحو التالي:
!>          إذا كان UPLO = 'U'، فإن AB(kd+1+i-j,j) = A(i,j) لـ max(1,j-kd)<=i<=j؛
!>          إذا كان UPLO = 'L'، فإن AB(1+i-j,j)    = A(i,j) لـ j<=i<=min(n,j+kd).
!>          إذا كان DIAG = 'U'، لا تتم الإشارة إلى العناصر القطرية لـ A
!>          ويفترض أنها 1.
!>

LDAB

!>          LDAB is INTEGER
!>          البعد الرئيسي للمصفوفة AB.  LDAB >= KD+1.
!>

B

!>          B is REAL array, dimension (LDB,NRHS)
!>          متجهات الطرف الأيمن لنظام المعادلات الخطية.
!>

LDB

!>          LDB is INTEGER
!>          البعد الرئيسي للمصفوفة B.  LDB >= max(1,N).
!>

X

!>          X is REAL array, dimension (LDX,NRHS)
!>          متجهات الحل المحسوبة. يتم تخزين كل متجه كعمود من المصفوفة X.
!>

LDX

!>          LDX is INTEGER
!>          البعد الرئيسي للمصفوفة X.  LDX >= max(1,N).
!>

XACT

!>          XACT is REAL array, dimension (LDX,NRHS)
!>          متجهات الحل الدقيقة. يتم تخزين كل متجه كعمود من المصفوفة XACT.
!>

LDXACT

!>          LDXACT is INTEGER
!>          البعد الرئيسي للمصفوفة XACT.  LDXACT >= max(1,N).
!>

FERR

!>          FERR is REAL array, dimension (NRHS)
!>          تقديرات حدود الخطأ الأمامي لكل متجه حل X. إذا كان XTRUE هو الحل الحقيقي،
!>          فإن FERR يحد من حجم أكبر مدخل في (X - XTRUE) مقسومًا على حجم أكبر مدخل في X.
!>

BERR

!>          BERR is REAL array, dimension (NRHS)
!>          الخطأ الخلفي النسبي المكون لكل متجه حل (أي، أصغر تغيير نسبي في أي مدخل
!>          لـ A أو B يجعله حلاً دقيقًا).
!>

RESLTS

!>          RESLTS is REAL array, dimension (2)
!>          الحد الأقصى عبر متجهات الحل NRHS للنسب التالية:
!>          RESLTS(1) = norm(X - XACT) / ( norm(X) * FERR )
!>          RESLTS(2) = BERR / ( NZ*EPS + (*) )
!>
