subroutine strt03 (character uplo, character trans, character diag, integer n, integer nrhs, real, dimension( lda, * ) a, integer lda, real scale, real, dimension( * ) cnorm, real tscal, real, dimension( ldx, * ) x, integer ldx, real, dimension( ldb, * ) b, integer ldb, real, dimension( * ) work, real resid)

STRT03

الغرض:

!>
!> STRT03 يحسب الباقي لحل نظام معادلات مثلثي مقياس A*x = s*b  أو  A'*x = s*b.
!> هنا A مصفوفة مثلثية، A' هي منقول A، s هو معامل قياس، و x و b هما مصفوفتان N في NRHS.  نسبة الاختبار هي
!> الحد الأقصى على عدد الأطراف اليمنى لـ
!>    norm(s*b - op(A)*x) / ( norm(op(A)) * norm(x) * EPS ),
!> حيث op(A) تمثل A أو A' و EPS هو الابسلون الآلي.
!>

المعلمات

UPLO 
!>          UPLO is CHARACTER*1
!>          يحدد ما إذا كانت المصفوفة A علوية أم سفلية مثلثية.
!>          = 'U':  مثلثة علوية
!>          = 'L':  مثلثة سفلية
!>

TRANS

!>          TRANS is CHARACTER*1
!>          يحدد العملية المطبقة على A.
!>          = 'N':  A *x = s*b  (بدون منقول)
!>          = 'T':  A'*x = s*b  (منقول)
!>          = 'C':  A'*x = s*b  (منقول مترافق = منقول)
!>

DIAG

!>          DIAG is CHARACTER*1
!>          يحدد ما إذا كانت المصفوفة A مثلثية وحدة أم لا.
!>          = 'N':  غير وحدة مثلثية
!>          = 'U':  وحدة مثلثية
!>

N

!>          N is INTEGER
!>          رتبة المصفوفة A.  N >= 0.
!>

NRHS

!>          NRHS is INTEGER
!>          عدد الأطراف اليمنى، أي عدد الأعمدة
!>          للمصفوفتين X و B.  NRHS >= 0.
!>

A

!>          A is REAL array, dimension (LDA,N)
!>          المصفوفة المثلثية A.  إذا كان UPLO = 'U'، فإن الجزء العلوي المثلثي العلوي
!>          المقدم للمصفوفة A يحتوي على المصفوفة المثلثية العلوية، والجزء السفلي المثلثي
!>          الصارم للمصفوفة A لا تتم الإشارة إليه.  إذا كان UPLO = 'L'، فإن الجزء السفلي المثلثي السفلي
!>          المقدم للمصفوفة A يحتوي على المصفوفة المثلثية السفلية، والجزء العلوي المثلثي
!>          الصارم للمصفوفة A لا تتم الإشارة إليه.  إذا كان DIAG = 'U'، فإن العناصر القطرية لـ A
!>          لا تتم الإشارة إليها أيضًا وتُفترض أنها 1.
!>

LDA

!>          LDA is INTEGER
!>          البعد الرائد للمصفوفة A.  LDA >= max(1,N).
!>

SCALE

!>          SCALE is REAL
!>          عامل القياس s المستخدم في حل النظام المثلثي.
!>

CNORM

!>          CNORM is REAL array, dimension (N)
!>          المعايير 1 لأعمدة A، لا تشمل القطر.
!>

TSCAL

!>          TSCAL is REAL
!>          عامل القياس المستخدم في حساب المعايير 1 في CNORM.
!>          CNORM يحتوي فعليًا على معايير أعمدة TSCAL*A.
!>

X

!>          X is REAL array, dimension (LDX,NRHS)
!>          متجهات الحل المحسوبة لنظام المعادلات الخطية.
!>

LDX

!>          LDX is INTEGER
!>          البعد الرائد للمصفوفة X.  LDX >= max(1,N).
!>

B

!>          B is REAL array, dimension (LDB,NRHS)
!>          متجهات الطرف الأيمن لنظام المعادلات الخطية.
!>

LDB

!>          LDB is INTEGER
!>          البعد الرائد للمصفوفة B.  LDB >= max(1,N).
!>

WORK

!>          WORK is REAL array, dimension (N)
!>

RESID

!>          RESID is REAL
!>          الحد الأقصى على عدد الأطراف اليمنى لـ
!>          norm(op(A)*x - s*b) / ( norm(op(A)) * norm(x) * EPS ).
!>
