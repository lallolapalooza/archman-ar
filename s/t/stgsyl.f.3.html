توثيق الوظيفة/الروتين الفرعي

subroutine stgsyl (character trans, integer ijob, integer m, integer n, real, dimension( lda, * ) a, integer lda, real, dimension( ldb, * ) b, integer ldb, real, dimension( ldc, * ) c, integer ldc, real, dimension( ldd, * ) d, integer ldd, real, dimension( lde, * ) e, integer lde, real, dimension( ldf, * ) f, integer ldf, real scale, real dif, real, dimension( * ) work, integer lwork, integer, dimension( * ) iwork, integer info)

STGSYL

الغرض:

!>
!> STGSYL يحل المعادلة التفاضلية المعممة:
!>
!>             A * R - L * B = scale * C                 (1)
!>             D * R - L * E = scale * F
!>
!> حيث R و L مصفوفتان غير معروفتين بحجم m × n، (A, D)، (B, E) و
!> (C, F) أزواج مصفوفات معطاة بأبعاد m × m، n × n و m × n، على التوالي،
!> تحتوي على مدخلات حقيقية. يجب أن تكون (A, D) و (B, E) في
!> شكل شوري المعمم (الحقيقي) القياسي، أي A، B هي مصفوفات شبه علوية
!> مثلثية و D، E هي مصفوفات علوية مثلثية.
!>
!> الحل (R, L) يكتب فوق (C, F). 0 <= SCALE <= 1 هو عامل قياس ناتج
!> تم اختياره لتجنب الفيضان.
!>
!> في ترميز المصفوفة (1) يكافئ حل Zx = scale b، حيث
!>
!>            Z = [ kron(In, A)  -kron(B**T, Im) ]         (2)
!>                [ kron(In, D)  -kron(E**T, Im) ].
!>
!> هنا Ik هي المصفوفة الوحدة بحجم k و X**T هي منقول المصفوفة X. kron(X, Y) هو حاصل كرونكر بين المصفوفتين X و Y.
!>
!> إذا كان TRANS = 'T'، تحل STGSYL النظام المنقول Z**T*y = scale*b،
!> والتي تكافئ الحل لـ R و L في
!>
!>             A**T * R + D**T * L = scale * C           (3)
!>             R * B**T + L * E**T = scale * -F
!>
!> هذه الحالة (TRANS = 'T') تستخدم لحساب تقدير مبني على المعيار الواحد
!> لـ Dif[(A,D), (B,E)]، الفصل بين أزواج المصفوفات (A,D) و (B,E)، باستخدام SLACON.
!>
!> إذا كان IJOB >= 1، تحسب STGSYL تقديرًا مبنيًا على معيار فروبينيوس
!> لـ Dif[(A,D),(B,E)]. أي، مقلوب الحد الأدنى لمقلوب
!> أصغر قيمة مفردة لـ Z. انظر [1-2] لمزيد من المعلومات.
!>
!> هذه خوارزمية BLAS مستوى 3.
!>

المعلمات

TRANS 
!>          TRANS is CHARACTER*1
!>          = 'N': حل المعادلة التفاضلية المعممة (1).
!>          = 'T': حل النظام 'المنقول' (3).
!>

IJOB

!>          IJOB is INTEGER
!>          يحدد نوع الوظيفة التي سيتم تنفيذها.
!>          = 0: حل (1) فقط.
!>          = 1: وظيفة 0 و 3.
!>          = 2: وظيفة 0 و 4.
!>          = 3: يتم حساب تقدير لـ Dif[(A,D), (B,E)] فقط.
!>               (يتم استخدام استراتيجية النظر للأمام IJOB = 1).
!>          = 4: يتم حساب تقدير لـ Dif[(A,D), (B,E)] فقط.
!>               (يتم استخدام SGECON على الأنظمة الفرعية).
!>          لا يتم الرجوع إليه إذا كان TRANS = 'T'.
!>

M

!>          M is INTEGER
!>          ترتيب المصفوفتين A و D، وبعد الصفوف للمصفوفتين C، F، R و L.
!>

N

!>          N is INTEGER
!>          ترتيب المصفوفتين B و E، وبعد الأعمدة للمصفوفتين C، F، R و L.
!>

A

!>          A is REAL array, dimension (LDA, M)
!>          المصفوفة شبه العلوية المثلثية A.
!>

LDA

!>          LDA is INTEGER
!>          البعد الرائد للمصفوفة A. LDA >= max(1, M).
!>

B

!>          B is REAL array, dimension (LDB, N)
!>          المصفوفة شبه العلوية المثلثية B.
!>

LDB

!>          LDB is INTEGER
!>          البعد الرائد للمصفوفة B. LDB >= max(1, N).
!>

C

!>          C is REAL array, dimension (LDC, N)
!>          عند الدخول، تحتوي C على الطرف الأيمن للمعادلة المصفوفية الأولى في (1) أو (3).
!>          عند الخروج، إذا كان IJOB = 0، 1 أو 2، تم الكتابة فوق C بالحل R. إذا كان IJOB = 3 أو 4 وكان TRANS = 'N'، فإن C تحتوي على R،
!>          الحل الذي تم تحقيقه أثناء حساب تقدير Dif.
!>

LDC

!>          LDC is INTEGER
!>          البعد الرائد للمصفوفة C. LDC >= max(1, M).
!>

D

!>          D is REAL array, dimension (LDD, M)
!>          المصفوفة العلوية المثلثية D.
!>

LDD

!>          LDD is INTEGER
!>          البعد الرائد للمصفوفة D. LDD >= max(1, M).
!>

E

!>          E is REAL array, dimension (LDE, N)
!>          المصفوفة العلوية المثلثية E.
!>

LDE

!>          LDE is INTEGER
!>          البعد الرائد للمصفوفة E. LDE >= max(1, N).
!>

F

!>          F is REAL array, dimension (LDF, N)
!>          عند الدخول، تحتوي F على الطرف الأيمن للمعادلة المصفوفية الثانية في (1) أو (3).
!>          عند الخروج، إذا كان IJOB = 0، 1 أو 2، تم الكتابة فوق F بالحل L. إذا كان IJOB = 3 أو 4 وكان TRANS = 'N'، فإن F تحتوي على L،
!>          الحل الذي تم تحقيقه أثناء حساب تقدير Dif.
!>

LDF

!>          LDF is INTEGER
!>          البعد الرائد للمصفوفة F. LDF >= max(1, M).
!>

DIF

!>          DIF is REAL
!>          عند الخروج DIF هو مقلوب حد أدنى لمقلوب
!>          دالة Dif، أي DIF هو حد أعلى لـ
!>          Dif[(A,D), (B,E)] = sigma_min(Z)، حيث Z كما في (2).
!>          إذا كان IJOB = 0 أو TRANS = 'T'، لا يتم المساس بـ DIF.
!>

SCALE

!>          SCALE is REAL
!>          عند الخروج SCALE هو عامل القياس في (1) أو (3).
!>          إذا كان 0 < SCALE < 1، تحتوي C و F على الحلول R و L، على التوالي،
!>          لنظام مضطرب قليلاً ولكن لم يتم تغيير المصفوفات المدخلة A، B، D
!>          و E. إذا كان SCALE = 0، تحتوي C و F على
!>          الحلول R و L، على التوالي، للنظام المتجانس
!>          مع C = F = 0. عادة، SCALE = 1.
!>

WORK

!>          WORK is REAL array, dimension (MAX(1,LWORK))
!>          عند الخروج، إذا كان INFO = 0، فإن WORK(1) يعيد LWORK الأمثل.
!>

LWORK

!>          LWORK is INTEGER
!>          بعد المصفوفة WORK. LWORK > = 1.
!>          إذا كان IJOB = 1 أو 2 وكان TRANS = 'N'، فإن LWORK >= max(1,2*M*N).
!>
!>          إذا كان LWORK = -1، فيفترض استعلام مساحة العمل؛ يقوم الروتين
!>          فقط بحساب الحجم الأمثل لمصفوفة WORK، ويعيد
!>          هذه القيمة كإدخال أول في مصفوفة WORK، ولا يصدر أي رسالة خطأ
!>          متعلقة بـ LWORK بواسطة XERBLA.
!>

IWORK

!>          IWORK is INTEGER array, dimension (M+N+6)
!>

INFO

!>          INFO is INTEGER
!>            =0: خروج ناجح
!>            <0: إذا كان INFO = -i، فإن الوسيط i كان لديه قيمة غير قانونية.
!>            >0: (A, D) و (B, E) لهما قيم ذاتية مشتركة أو قريبة.
!>
