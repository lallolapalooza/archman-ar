subroutine stpt06 (real rcond, real rcondc, character uplo, character diag, integer n, real, dimension( * ) ap, real, dimension( * ) work, real rat)

STPT06

الغرض:

!>
!> STPT06 يحسب نسبة اختبار تقارن RCOND (الرقم الشرطي المتبادل لمصفوفة مثلثية A) و RCONDC، التقدير المحسوب بواسطة STPCON.  تستخدم معلومات حول المصفوفة المثلثية A إذا كان أحد التقديرين صفراً والآخر غير صفري لتحديد ما إذا كان نقص التدفق في التقدير مبرراً.
!>

المعلمات

RCOND 
!>          RCOND هو REAL
!>          تقدير الرقم الشرطي المتبادل الناتج عن تكوين معكوس صريح للمصفوفة A وحساب
!>          RCOND = 1/( norm(A) * norm(inv(A)) ).
!>

RCONDC

!>          RCONDC هو REAL
!>          تقدير الرقم الشرطي المتبادل المحسوب بواسطة
!>          STPCON.
!>

UPLO

!>          UPLO هو CHARACTER
!>          يحدد ما إذا كانت المصفوفة A مثلثية عليا أم سفلى.
!>          = 'U':  مثلثية عليا
!>          = 'L':  مثلثية سفلى
!>

DIAG

!>          DIAG هو CHARACTER
!>          يحدد ما إذا كانت المصفوفة A مثلثية موحدة أم لا.
!>          = 'N':  مثلثية غير موحدة
!>          = 'U':  مثلثية موحدة
!>

N

!>          N هو INTEGER
!>          رتبة المصفوفة A.  N >= 0.
!>

AP

!>          AP هو مصفوفة REAL، البعد (N*(N+1)/2)
!>          المصفوفة المثلثية العليا أو السفلى A، مكدسة عمودياً في
!>          مصفوفة خطية.  يتم تخزين العمود j من A في المصفوفة
!>          AP كما يلي:
!>          إذا كان UPLO = 'U'، AP((j-1)*j/2 + i) = A(i,j) لـ 1<=i<=j؛
!>          إذا كان UPLO = 'L'،
!>             AP((j-1)*(n-j) + j*(j+1)/2 + i-j) = A(i,j) لـ j<=i<=n.
!>

WORK

!>          WORK هي مصفوفة REAL، البعد (N)
!>

RAT

!>          RAT هو REAL
!>          نسبة الاختبار.  إذا كان كل من RCOND و RCONDC غير صفريين،
!>             RAT = MAX( RCOND, RCONDC )/MIN( RCOND, RCONDC ) - 1.
!>          إذا كان RAT = 0، فإن التقديرين متطابقان تماماً.
!>
