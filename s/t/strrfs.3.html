subroutine strrfs (character uplo, character trans, character diag, integer n, integer nrhs, real, dimension( lda, * ) a, integer lda, real, dimension( ldb, * ) b, integer ldb, real, dimension( ldx, * ) x, integer ldx, real, dimension( * ) ferr, real, dimension( * ) berr, real, dimension( * ) work, integer, dimension( * ) iwork, integer info)

STRRFS

الغرض:

!>
!> يوفر STRRFS حدود الخطأ وتقديرات الخطأ العكسي لحل نظام من المعادلات الخطية
!> بمصفوفة معاملات مثلثية.
!>
!> يجب حساب مصفوفة الحل X بواسطة STRTRS أو بطريقة أخرى
!> قبل الدخول إلى هذا الروتين. لا يقوم STRRFS بالتحسين التكراري
!> لأن القيام بذلك لا يمكن أن يحسن الخطأ العكسي.
!>

المعلمات

UPLO 
!>          UPLO is CHARACTER*1
!>          = 'U':  A is upper triangular; (A مثلثية عليا)
!>          = 'L':  A is lower triangular. (A مثلثية سفلى)
!>

TRANS

!>          TRANS is CHARACTER*1
!>          Specifies the form of the system of equations: (يحدد شكل نظام المعادلات)
!>          = 'N':  A * X = B  (No transpose) (بدون تبديل)
!>          = 'T':  A**T * X = B  (Transpose) (تبديل)
!>          = 'C':  A**H * X = B  (Conjugate transpose = Transpose) (تبديل مرافق = تبديل)
!>

DIAG

!>          DIAG is CHARACTER*1
!>          = 'N':  A is non-unit triangular; (A مثلثية غير وحدوية)
!>          = 'U':  A is unit triangular. (A مثلثية وحدوية)
!>

N

!>          N is INTEGER
!>          The order of the matrix A.  N >= 0. (رتبة المصفوفة A. N >= 0)
!>

NRHS

!>          NRHS is INTEGER
!>          The number of right hand sides, i.e., the number of columns
!>          of the matrices B and X.  NRHS >= 0. (عدد الأيادي اليمنى، أي عدد أعمدة
!>          المصفوفتين B و X. NRHS >= 0)
!>

A

!>          A is REAL array, dimension (LDA,N)
!>          The triangular matrix A.  If UPLO = 'U', the leading N-by-N
!>          upper triangular part of the array A contains the upper
!>          triangular matrix, and the strictly lower triangular part of
!>          A is not referenced.  If UPLO = 'L', the leading N-by-N lower
!>          triangular part of the array A contains the lower triangular
!>          matrix, and the strictly upper triangular part of A is not
!>          referenced.  If DIAG = 'U', the diagonal elements of A are
!>          also not referenced and are assumed to be 1.
!>          (المصفوفة المثلثية A. إذا كان UPLO = 'U'، فإن الجزء العلوي الأيسر بحجم N×N
!>          من المصفوفة A يحتوي على المصفوفة المثلثية العليا، والجزء السفلي القطري الصارم
!>          لـ A لا تتم الإشارة إليه. إذا كان UPLO = 'L'، فإن الجزء السفلي الأيسر بحجم N×N
!>          من المصفوفة A يحتوي على المصفوفة المثلثية السفلى، والجزء العلوي القطري الصارم
!>          لـ A لا تتم الإشارة إليه. إذا كان DIAG = 'U'، فإن عناصر القطر لـ A
!>          لا تتم الإشارة إليها أيضًا ويفترض أنها 1.)
!>

LDA

!>          LDA is INTEGER
!>          The leading dimension of the array A.  LDA >= max(1,N). (البعد الرائد للمصفوفة A. LDA >= max(1,N))
!>

B

!>          B is REAL array, dimension (LDB,NRHS)
!>          The right hand side matrix B. (مصفوفة الجانب الأيمن B)
!>

LDB

!>          LDB is INTEGER
!>          The leading dimension of the array B.  LDB >= max(1,N). (البعد الرائد للمصفوفة B. LDB >= max(1,N))
!>

X

!>          X is REAL array, dimension (LDX,NRHS)
!>          The solution matrix X. (مصفوفة الحل X)
!>

LDX

!>          LDX is INTEGER
!>          The leading dimension of the array X.  LDX >= max(1,N). (البعد الرائد للمصفوفة X. LDX >= max(1,N))
!>

FERR

!>          FERR is REAL array, dimension (NRHS)
!>          The estimated forward error bound for each solution vector
!>          X(j) (the j-th column of the solution matrix X).
!>          If XTRUE is the true solution corresponding to X(j), FERR(j)
!>          is an estimated upper bound for the magnitude of the largest
!>          element in (X(j) - XTRUE) divided by the magnitude of the
!>          largest element in X(j).  The estimate is as reliable as
!>          the estimate for RCOND, and is almost always a slight
!>          overestimate of the true error.
!>          (حد الخطأ الأمامي المقدر لكل متجه حل
!>          X(j) (العمود j من مصفوفة الحل X).
!>          إذا كان XTRUE هو الحل الحقيقي المقابل لـ X(j)، فإن FERR(j)
!>          هو حد علوي مقدر لقيمة أكبر عنصر في (X(j) - XTRUE) مقسومًا على قيمة
!>          أكبر عنصر في X(j). التقدير موثوق به مثل
!>          تقدير RCOND، وهو دائمًا تقريبًا تقدير زائد قليلًا للخطأ الحقيقي.)
!>

BERR

!>          BERR is REAL array, dimension (NRHS)
!>          The componentwise relative backward error of each solution
!>          vector X(j) (i.e., the smallest relative change in
!>          any element of A or B that makes X(j) an exact solution).
!>          (الخطأ العكسي النسبي للمكونات لكل متجه حل
!>          X(j) (أي، أصغر تغيير نسبي في
!>          أي عنصر من A أو B يجعل X(j) حلاً دقيقًا).)
!>

WORK

!>          WORK is REAL array, dimension (3*N) (مصفوفة WORK هي مصفوفة حقيقية، حجمها 3*N)
!>

IWORK

!>          IWORK is INTEGER array, dimension (N) (المصفوفة IWORK هي مصفوفة صحيحة، حجمها N)
!>

INFO

!>          INFO is INTEGER
!>          = 0:  successful exit (خروج ناجح)
!>          < 0:  if INFO = -i, the i-th argument had an illegal value (إذا كان INFO = -i، فإن الوسيط i كان له قيمة غير قانونية)
!>
