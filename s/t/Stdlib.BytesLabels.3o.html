val create : int -> bytes

create n يعيد تسلسل بايت بطول n، مهيأ ويحتوي على بايتات عشوائية.

استثناء عالي Invalid_argument إذا كان n < 0 أو n > Sys.max_string_length.

val make : int -> char -> bytes

make n c يعيد تسلسل بايت بطول n، مملوءًا بالبايت c.

استثناء عالي Invalid_argument إذا كان n < 0 أو n > Sys.max_string_length.

استثناء عالي Invalid_argument إذا كان n < 0 أو n > Sys.max_string_length.

val sub : bytes -> pos:int -> len:int -> bytes

sub s ~pos ~len يعيد تسلسل بايت جديد بطول len، يحتوي على التسلسل الفرعي لـ s الذي يبدأ من الموضع pos ويبلغ طوله len.

استثناء عالي Invalid_argument إذا كان pos و len لا يحددان نطاقًا صالحًا لـ s.

val sub_string : bytes -> pos:int -> len:int -> string

هو نفس BytesLabels.sub ولكنه يعيد سلسلة نصية بدلاً من تسلسل بايت.

val extend : bytes -> left:int -> right:int -> bytes

extend s ~left ~right يعيد تسلسل بايت جديد يحتوي على بايتات s، مع left بايت غير مهيأ في البداية و right بايت غير مهيأ في النهاية. إذا كان left أو right سالبًا، يتم حذف بايتات (بدلاً من إضافتها) من الجوانب المقابلة لـ s.

val blit : src:bytes -> src_pos:int -> dst:bytes -> dst_pos:int -> len:int -> unit

blit ~src ~src_pos ~dst ~dst_pos ~len ينسخ len بايت من تسلسل البايتات src، بدءًا من الفهرس src_pos، إلى تسلسل البايتات dst، بدءًا من الفهرس dst_pos. يعمل بشكل صحيح حتى لو كان src و dst نفس تسلسل البايتات، والنطاقات المصدر والوجهة تتداخل.

استثناء عالي Invalid_argument إذا كان src_pos و len لا يحددان نطاقًا صالحًا لـ src، أو إذا كان dst_pos و len لا يحددان نطاقًا صالحًا لـ dst.

تحويلات غير آمنة (للمستخدمين المتقدمين)

يصف هذا القسم وظائف تحويل غير آمنة ومنخفضة المستوى بين bytes و string. لا تقوم بنسخ البيانات الداخلية؛ عند استخدامها بشكل غير صحيح، يمكنها كسر ثابت الثبات على السلاسل النصية. إنها متاحة لمؤلفي المكتبات الخبراء، ولكن لمعظم الأغراض يجب استخدام BytesLabels.to_string و BytesLabels.of_string الصحيحة دائمًا بدلاً من ذلك.

val unsafe_to_string : bytes -> string

تحويل تسلسل بايت بشكل غير آمن إلى سلسلة نصية.

للتفكير في استخدام unsafe_to_string، من المفيد النظر إلى نظام "الملكية" الانضباطي. قطعة من الكود تتعامل مع بعض البيانات "تمتلكها"؛ هناك عدة أوضاع ملكية منفصلة، بما في ذلك:

- ملكية فريدة: يمكن الوصول إلى البيانات وتعديلها

- ملكية مشتركة: البيانات لديها العديد من المالكين، الذين يمكنهم فقط الوصول إليها، وليس تعديلها.

الملكية الفريدة خطية: تمرير البيانات إلى قطعة أخرى من الكود يعني التخلي عن الملكية (لا يمكننا كتابة البيانات مرة أخرى). يمكن للمالك الفريد أن يقرر جعل البيانات مشتركة (التخلي عن حقوق التعديل عليها)، ولكن لا يمكن للبيانات المشتركة أن تصبح مملوكة بشكل فريد مرة أخرى.

unsafe_to_string s يمكن استخدامها فقط عندما يمتلك المتصل تسلسل البايت s - إما بشكل فريد أو كبيانات ثابتة مشتركة. يتخلى المتصل عن ملكية s، ويكتسب ملكية السلسلة النصية المرجعة.

هناك حالتان استخداميتان صالحتان تحترمان هذا الانضباط في الملكية:

1. إنشاء سلسلة نصية عن طريق تهيئة وتعديل تسلسل بايت لا يتم تغييره أبدًا بعد إتمام التهيئة.

let string_init len f : string =
  let s = Bytes.create len in
  for i = 0 to len - 1 do Bytes.set s i (f i) done;
  Bytes.unsafe_to_string s

هذه الوظيفة آمنة لأن تسلسل البايت s لن يتم الوصول إليه أو تعديله أبدًا بعد استدعاء unsafe_to_string. يتخلى كود string_init عن ملكية s، ويعيد ملكية السلسلة النصية الناتجة إلى المتصل الخاص به.

لاحظ أنه سيكون غير آمن إذا تم تمرير s كوسيط إضافي إلى الوظيفة f لأنه يمكن أن يتسرب بهذه الطريقة ويتم تعديله في المستقبل - ستتخلى string_init عن ملكية s لتمريرها إلى f، ولا يمكنها استدعاء unsafe_to_string بأمان.

لقد وفرنا وظائف String.init و String.map و String.mapi لتغطية معظم حالات بناء سلاسل نصية جديدة. يجب تفضيلها على to_string أو unsafe_to_string كلما كان ذلك ممكنًا.

2. إعارة مؤقتة لملكية تسلسل بايت إلى وظيفة تتوقع سلسلة نصية مملوكة بشكل فريد وتعيد الملكية، حتى نتمكن من تعديل التسلسل مرة أخرى بعد انتهاء الاستدعاء.

let bytes_length (s : bytes) =
  String.length (Bytes.unsafe_to_string s)

في حالة الاستخدام هذه، لا نعد بأن s لن يتم تعديلها أبدًا بعد استدعاء bytes_length s. تستعير وظيفة String.length مؤقتًا الملكية الفريدة لتسلسل البايت (وتراها كسلسلة نصية string)، ولكنها تعيد هذه الملكية إلى المتصل، الذي قد يفترض أن s لا يزال تسلسل بايت صالحًا بعد الاستدعاء. لاحظ أن هذا صحيح فقط لأننا نعلم أن String.length لا تلتقط وسيطها - يمكن أن تتسرب عبر قناة جانبية مثل مُجمِّع التخزين المؤقت.

لا يجوز للمتصل تعديل s أثناء استعارة السلسلة النصية (لقد تخلى مؤقتًا عن الملكية). يؤثر هذا على البرامج المتزامنة، ولكنه يؤثر أيضًا على الدوال ذات الرتب الأعلى: إذا أعادت String.length إغلاقًا ليتم استدعاؤه لاحقًا، فلا ينبغي تعديل s حتى يتم تطبيق هذا الإغلاق بالكامل ويعيد الملكية.

val unsafe_of_string : string -> bytes

تحويل سلسلة نصية مشتركة بشكل غير آمن إلى تسلسل بايت يجب عدم تعديله.

نفس انضباط الملكية الذي يجعل unsafe_to_string صحيحًا ينطبق على unsafe_of_string: يمكنك استخدامه إذا كنت مالك قيمة string، وستمتلك bytes المرجعة بنفس الوضع.

من الناحية العملية، من الصعب للغاية التفكير بشكل صحيح في الملكية الفريدة لقيم السلاسل النصية. يجب عليك دائمًا افتراض أن السلاسل النصية مشتركة، وليست مملوكة بشكل فريد.

على سبيل المثال، يتم مشاركة الحروف النصية ضمنيًا بواسطة المترجم، لذلك لا تمتلكها بشكل فريد أبدًا.

let incorrect = Bytes.unsafe_of_string "hello"
let s = Bytes.of_string "hello"

الإعلان الأول غير صحيح، لأن الحرف النصي "hello" يمكن مشاركته بواسطة المترجم مع أجزاء أخرى من البرنامج، وتعديل incorrect هو خطأ. يجب عليك دائمًا استخدام الإصدار الثاني، الذي يقوم بنسخ وهو صحيح وبالتالي.

افتراض الملكية الفريدة للسلاسل النصية التي ليست حروفًا نصية، ولكنها مبنية (جزئيًا) من حروف نصية، هو أيضًا غير صحيح. على سبيل المثال، تعديل unsafe_of_string ("foo" ^ s) يمكن أن يعدل السلسلة النصية المشتركة "foo" - بافتراض تمثيل يشبه الحبال للسلاسل النصية. بشكل عام، تفترض الوظائف التي تعمل على السلاسل النصية الملكية المشتركة، ولا تحافظ على الملكية الفريدة. لذلك، من غير الصحيح افتراض الملكية الفريدة للنتيجة.

set_utf_16be_uchar b i u يرمّز u عند الموضع i في b ويعيد عدد البايتات n التي تم كتابتها بدءًا من i. إذا كان n يساوي 0، فلا توجد مساحة كافية لترميز u عند i، ويبقى b دون تغيير. خلاف ذلك، قد يكتب حرفًا جديدًا عند i + n.

val is_valid_utf_16be : t -> bool

is_valid_utf_16be b هو true إذا وفقط إذا كان b يحتوي على بيانات UTF-16BE صالحة.

ترميز وفك ترميز الأعداد الصحيحة

تقوم الدوال في هذا القسم بترميز وفك ترميز الأعداد الصحيحة إلى ومن تسلسلات البايتات.

تطلق جميع الدوال التالية Invalid_argument إذا لم تكن هناك مساحة كافية في الموضع i لفك ترميز أو ترميز عدد صحيح.

يشير الترميز بنظام endian (أو فك الترميز بنظام endian) إلى أن البايتات الأقل أهمية (أو الأكثر أهمية) يتم تخزينها أولاً. يُعرف نظام endian الكبير أيضًا بترتيب بايتات الشبكة. الترميز الأصلي هو، بناءً على Sys.big_endian، إما نظام endian صغير أو نظام endian كبير.

يتم تمثيل الأعداد الصحيحة 32 بت و 64 بت بواسطة الأنواع int32 و int64، والتي يمكن تفسيرها كأعداد صحيحة موقعة وغير موقعة.

يتم تمثيل الأعداد الصحيحة 8 بت و 16 بت بواسطة النوع int، الذي يحتوي على بتات أكثر من الترميز الثنائي. يتم التعامل مع البتات الإضافية على النحو التالي:

- الدوال التي تفك ترميز أعداد صحيحة 8 بت (أو 16 بت) ممثلة بقيم int تقوم بتوسيع نتيجتها بالإشارة (أو توسيع بالصفر).

- الدوال التي ترمّز أعدادًا صحيحة 8 بت (أو 16 بت) ممثلة بقيم int تقوم باقتطاع مدخلاتها إلى بايتاتها الأكثر أهمية.
