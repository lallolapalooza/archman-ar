subroutine strevc (character side, character howmny, logical, dimension( * ) select, integer n, real, dimension( ldt, * ) t, integer ldt, real, dimension( ldvl, * ) vl, integer ldvl, real, dimension( ldvr, * ) vr, integer ldvr, integer mm, integer m, real, dimension( * ) work, integer info)

STREVC

الغرض:

!>
!> STREVC يحسب بعض أو كل المتجهات الذاتية اليمنى واليسرى لمصفوفة T شبه المثلثية الحقيقية.
!> يتم إنتاج مصفوفات هذا النوع عن طريق تحليل شور لمصفوفة حقيقية عامة:  A = Q*T*Q**T، كما تم حسابها بواسطة SHSEQR.
!>
!> يتم تعريف المتجه الذاتي الأيمن x والمتجه الذاتي الأيسر y للمصفوفة T المقابلة لقيمة ذاتية w بواسطة:
!>
!>    T*x = w*x,     (y**H)*T = w*(y**H)
!>
!> حيث يشير y**H إلى المرافق التبادلي لـ y.
!> القيم الذاتية ليست مدخلات لهذا الروتين، ولكن يتم قراءتها مباشرة
!> من الكتل القطرية لـ T.
!>
!> يعيد هذا الروتين المصفوفات X و/أو Y للمتجهات الذاتية اليمنى واليسرى
!> للمصفوفة T، أو المنتجات Q*X و/أو Q*Y، حيث Q هي مصفوفة إدخال. إذا كانت Q هي العامل المتعامد الذي يقلل المصفوفة
!> A إلى شكل شور T، فإن Q*X و Q*Y هما مصفوفتي المتجهات الذاتية اليمنى واليسرى
!> للمصفوفة A.
!>

المعلمات

SIDE 
!>          SIDE هو CHARACTER*1
!>          = 'R':  حساب المتجهات الذاتية اليمنى فقط؛
!>          = 'L':  حساب المتجهات الذاتية اليسرى فقط؛
!>          = 'B':  حساب كل من المتجهات الذاتية اليمنى واليسرى.
!>

HOWMNY

!>          HOWMNY هو CHARACTER*1
!>          = 'A':  حساب جميع المتجهات الذاتية اليمنى و/أو اليسرى؛
!>          = 'B':  حساب جميع المتجهات الذاتية اليمنى و/أو اليسرى،
!>                  معاد تحويلها بواسطة المصفوفات في VR و/أو VL؛
!>          = 'S':  حساب المتجهات الذاتية اليمنى و/أو اليسرى المختارة،
!>                  كما هو محدد بواسطة المصفوفة المنطقية SELECT.
!>

SELECT

!>          SELECT هي مصفوفة LOGICAL، بعد (N)
!>          إذا كانت HOWMNY = 'S'، فإن SELECT تحدد المتجهات الذاتية التي سيتم
!>          حسابها.
!>          إذا كانت w(j) قيمة ذاتية حقيقية، يتم حساب المتجه الذاتي الحقيقي المقابل إذا كانت SELECT(j) تساوي .TRUE..
!>          إذا كانت w(j) و w(j+1) هما الجزء الحقيقي والتخيلي لقيمة ذاتية
!>          مركبة، يتم حساب المتجه الذاتي المركب إذا كانت SELECT(j) أو SELECT(j+1) تساوي .TRUE.،
!>          وعند الخروج يتم تعيين SELECT(j) إلى .TRUE. وتعيين SELECT(j+1) إلى
!>          .FALSE..
!>          غير مستخدمة إذا كانت HOWMNY = 'A' أو 'B'.
!>

N

!>          N هو INTEGER
!>          رتبة المصفوفة T. N >= 0.
!>

T

!>          T هي مصفوفة REAL، بعد (LDT,N)
!>          المصفوفة شبه المثلثية العلوية T في شكل شور القياسي.
!>

LDT

!>          LDT هو INTEGER
!>          البعد الرائد للمصفوفة T. LDT >= max(1,N).
!>

VL

!>          VL هي مصفوفة REAL، بعد (LDVL,MM)
!>          عند الإدخال، إذا كانت SIDE = 'L' أو 'B' وكانت HOWMNY = 'B'، يجب أن
!>          تحتوي VL على مصفوفة Q بحجم N*N (عادةً ما تكون المصفوفة المتعامدة Q
!>          لمتجهات شور التي أعادتها SHSEQR).
!>          عند الخروج، إذا كانت SIDE = 'L' أو 'B'، تحتوي VL على:
!>          إذا كانت HOWMNY = 'A'، فإن المصفوفة Y للمتجهات الذاتية اليسرى لـ T؛
!>          إذا كانت HOWMNY = 'B'، فإن المصفوفة Q*Y؛
!>          إذا كانت HOWMNY = 'S'، فإن المتجهات الذاتية اليسرى لـ T المحددة بواسطة
!>                           SELECT، مخزنة بشكل متتالي في أعمدة
!>                           VL، بنفس ترتيب القيم الذاتية الخاصة بها.
!>          يتم تخزين المتجه الذاتي المركب المقابل لقيمة ذاتية مركبة
!>          في عمودين متتاليين، يحمل الأول الجزء الحقيقي، والثاني الجزء التخيلي.
!>          غير مستخدمة إذا كانت SIDE = 'R'.
!>

LDVL

!>          LDVL هو INTEGER
!>          البعد الرائد للمصفوفة VL. LDVL >= 1، وإذا
!>          كانت SIDE = 'L' أو 'B'، فإن LDVL >= N.
!>

VR

!>          VR هي مصفوفة REAL، بعد (LDVR,MM)
!>          عند الإدخال، إذا كانت SIDE = 'R' أو 'B' وكانت HOWMNY = 'B'، يجب أن
!>          تحتوي VR على مصفوفة Q بحجم N*N (عادةً ما تكون المصفوفة المتعامدة Q
!>          لمتجهات شور التي أعادتها SHSEQR).
!>          عند الخروج، إذا كانت SIDE = 'R' أو 'B'، تحتوي VR على:
!>          إذا كانت HOWMNY = 'A'، فإن المصفوفة X للمتجهات الذاتية اليمنى لـ T؛
!>          إذا كانت HOWMNY = 'B'، فإن المصفوفة Q*X؛
!>          إذا كانت HOWMNY = 'S'، فإن المتجهات الذاتية اليمنى لـ T المحددة بواسطة
!>                           SELECT، مخزنة بشكل متتالي في أعمدة
!>                           VR، بنفس ترتيب القيم الذاتية الخاصة بها.
!>          يتم تخزين المتجه الذاتي المركب المقابل لقيمة ذاتية مركبة
!>          في عمودين متتاليين، يحمل الأول الجزء الحقيقي والثاني الجزء التخيلي.
!>          غير مستخدمة إذا كانت SIDE = 'L'.
!>

LDVR

!>          LDVR هو INTEGER
!>          البعد الرائد للمصفوفة VR. LDVR >= 1، وإذا
!>          كانت SIDE = 'R' أو 'B'، فإن LDVR >= N.
!>

MM

!>          MM هو INTEGER
!>          عدد الأعمدة في المصفوفات VL و/أو VR. MM >= M.
!>

M

!>          M هو INTEGER
!>          عدد الأعمدة في المصفوفات VL و/أو VR المستخدمة فعليًا
!>          لتخزين المتجهات الذاتية.
!>          إذا كانت HOWMNY = 'A' أو 'B'، يتم تعيين M إلى N.
!>          يشغل كل متجه ذاتي حقيقي مختار عمودًا واحدًا ويشغل كل
!>          متجه ذاتي مركب مختار عمودين.
!>

WORK

!>          WORK هي مصفوفة REAL، بعد (3*N)
!>

INFO

!>          INFO هو INTEGER
!>          = 0:  انتهاء ناجح
!>          < 0:  إذا كانت INFO = -i، فإن المعلمة i كان لها قيمة غير قانونية
!>
