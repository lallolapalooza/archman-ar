stpncpy(3) دليل وظائف المكتبة stpncpy(3)

الاسم

stpncpy, strncpy - ملء مخزن مؤقت بحجم ثابت ببايتات غير خالية من سلسلة نصية، مع ملء ببايتات خالية حسب الحاجة

المكتبة

مكتبة C القياسية (libc،  -lc)

الملخص

#include <string.h>
char *strncpy(size_t dsize;
              char dst[restrict dsize], const char *restrict src,
              size_t dsize);
char *stpncpy(size_t dsize;
              char dst[restrict dsize], const char *restrict src,
              size_t dsize);
متطلبات ميزة الاختبار Macro لـ glibc (انظر feature_test_macros(7)):

stpncpy():

    منذ glibc 2.10:
        _POSIX_C_SOURCE >= 200809L
    قبل glibc 2.10:
        _GNU_SOURCE

الوصف

تقوم هذه الدوال بنسخ البايتات غير الخالية من السلسلة النصية التي يشير إليها src إلى المصفوفة التي يشير إليها dst. إذا كان المصدر يحتوي على عدد قليل جدًا من البايتات غير الخالية لملء الوجهة، تقوم الدوال بملء الوجهة ببايتات خالية لاحقة. إذا لم يكن المخزن المؤقت للوجهة، والمحدود بحجمه، كبيرًا بما يكفي لاستيعاب النسخة، يتم اقتطاع تسلسل الأحرف الناتج. للاطلاع على الفرق بين الدالتين، انظر قيمة الإرجاع.

قد يكون تنفيذ هذه الدوال كالتالي:


char *
strncpy(char *restrict dst, const char *restrict src, size_t dsize)
{
    stpncpy(dst, src, dsize);
    return dst;
}
char *
stpncpy(char *restrict dst, const char *restrict src, size_t dsize)
{
    size_t  dlen;
    dlen = strnlen(src, dsize);
    return memset(mempcpy(dst, src, dlen), 0, dsize - dlen);
}

قيمة الإرجاع

strncpy()
تُرجع dst.
stpncpy()
تُرجع مؤشرًا إلى ما بعد آخر حرف واسع غير خالي تم كتابته، أي dest + strnlen(src, n).

تحذيرات

اسم هذه الدوال مربك. تنتج هذه الدوال تسلسل أحرف مملوء ببايتات خالية، وليس سلسلة نصية (انظر string_copying(7)). على سبيل المثال:


strncpy(buf, "1", 5);       // { '1',   0,   0,   0,   0 }
strncpy(buf, "1234", 5);    // { '1', '2', '3', '4',   0 }
strncpy(buf, "12345", 5);   // { '1', '2', '3', '4', '5' }
strncpy(buf, "123456", 5);  // { '1', '2', '3', '4', '5' }

من المستحيل التمييز بين الاقتطاع الناتج عن استدعاء الدالة، وبين تسلسل أحرف يناسب تمامًا المخزن المؤقت للوجهة؛ يجب اكتشاف الاقتطاع عن طريق مقارنة طول السلسلة النصية المدخلة مع حجم المخزن المؤقت للوجهة.

إذا كنت ستستخدم هذه الدالة في استدعاءات متسلسلة، فسيكون من المفيد تطوير دالة مماثلة تقبل مؤشرًا إلى نهاية (ما بعد العنصر الأخير) المخزن المؤقت للجهة بدلاً من حجمه.

أمثلة

#include <err.h>
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
int
main(void)
{
    char    *p;
    char    buf1[20];
    char    buf2[20];
    size_t  len;
    if (sizeof(buf2) < strlen("Hello world!"))
        errx("strncpy: truncating character sequence");
    strncpy(buf2, "Hello world!", sizeof(buf2));
    len = strnlen(buf2, sizeof(buf2));
    printf("[len = %zu]: ", len);
    fwrite(buf2, 1, len, stdout);
    putchar('\n');
    if (sizeof(buf1) < strlen("Hello world!"))
        errx("stpncpy: truncating character sequence");
    p = stpncpy(buf1, "Hello world!", sizeof(buf1));
    len = p - buf1;
    printf("[len = %zu]: ", len);
    fwrite(buf1, 1, len, stdout);
    putchar('\n');
    exit(EXIT_SUCCESS);
}
