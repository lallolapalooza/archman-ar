وثائق الوظيفة/الإجراء الفرعي

subroutine stfttp (character transr, character uplo, integer n, real, dimension( 0: * ) arf, real, dimension( 0: * ) ap, integer info)

STFTTP يقوم بنسخ مصفوفة مثلثية من تنسيق الحزم الكامل المستطيل (TF) إلى تنسيق الحزم القياسي (TP).

الغرض:

!>
!> STFTTP يقوم بنسخ مصفوفة مثلثية A من تنسيق الحزم الكامل المستطيل
!> (TF) إلى تنسيق الحزم القياسي (TP).
!>

المعلمات

TRANSR 
!>          TRANSR هو CHARACTER*1
!>          = 'N':  ARF بتنسيق عادي؛
!>          = 'T':  ARF بتنسيق منقول؛
!>

UPLO

!>          UPLO هو CHARACTER*1
!>          = 'U':  A مصفوفة مثلثية عليا؛
!>          = 'L':  A مصفوفة مثلثية سفلى.
!>

N

!>          N هو INTEGER
!>          رتبة المصفوفة A. N >= 0.
!>

ARF

!>          ARF هو مصفوفة REAL، حجمها ( N*(N+1)/2 )،
!>          عند الإدخال، المصفوفة المثلثية A العليا أو السفلى مخزنة
!>          بتنسيق RFP. لمزيد من المناقشة، انظر الملاحظات أدناه.
!>

AP

!>          AP هو مصفوفة REAL، حجمها ( N*(N+1)/2 )،
!>          عند الخروج، المصفوفة المثلثية A العليا أو السفلى، محسوبة
!>          عمودياً في مصفوفة خطية. يتم تخزين العمود j من A
!>          في المصفوفة AP على النحو التالي:
!>          إذا كان UPLO = 'U'، فإن AP(i + (j-1)*j/2) = A(i,j) لـ 1<=i<=j؛
!>          إذا كان UPLO = 'L'، فإن AP(i + (j-1)*(2n-j)/2) = A(i,j) لـ j<=i<=n.
!>

INFO

!>          INFO هو INTEGER
!>          = 0:  انتهاء ناجح
!>          < 0:  إذا كان INFO = -i، فإن الوسيط i كان له قيمة غير قانونية
!>

تفاصيل إضافية:

!>
!>  ننظر أولاً إلى تنسيق الحزم الكامل المستطيل (RFP) عندما تكون N
!>  زوجية. نقدم مثالاً حيث N = 6.
!>
!>      AP عليا             AP سفلى
!>
!>   00 01 02 03 04 05       00
!>      11 12 13 14 15       10 11
!>         22 23 24 25       20 21 22
!>            33 34 35       30 31 32 33
!>               44 45       40 41 42 43 44
!>                  55       50 51 52 53 54 55
!>
!>
!>  ليكن TRANSR = 'N'. يخزن RFP المصفوفة AP على النحو التالي:
!>  بالنسبة لـ UPLO = 'U'، فإن شبه المنحرف العلوي A(0:5,0:2) يتكون من
!>  الأعمدة الثلاثة الأخيرة من AP العليا. يتكون المثلث السفلي A(4:6,0:2) من
!>  منقول الأعمدة الثلاثة الأولى من AP العليا.
!>  بالنسبة لـ UPLO = 'L'، فإن شبه المنحرف السفلي A(1:6,0:2) يتكون من
!>  الأعمدة الثلاثة الأولى من AP السفلى. يتكون المثلث العلوي A(0:2,0:2) من
!>  منقول الأعمدة الثلاثة الأخيرة من AP السفلى.
!>  هذا يغطي حالة N زوجية و TRANSR = 'N'.
!>
!>         RFP A                   RFP A
!>
!>        03 04 05                33 43 53
!>        13 14 15                00 44 54
!>        23 24 25                10 11 55
!>        33 34 35                20 21 22
!>        00 44 45                30 31 32
!>        01 11 55                40 41 42
!>        02 12 22                50 51 52
!>
!>
!>  الآن ليكن TRANSR = 'T'. RFP A في كلتا حالتي UPLO هو فقط
!>  منقول RFP A أعلاه. لذلك يحصل المرء على:
!>
!>
!>           RFP A                   RFP A
!>
!>     03 13 23 33 00 01 02    33 00 10 20 30 40 50
!>     04 14 24 34 44 11 12    43 44 11 21 31 41 51
!>     05 15 25 35 45 55 22    53 54 55 22 32 42 52
!>
!>
!>  ثم ننظر إلى تنسيق الحزم الكامل المستطيل (RFP) عندما تكون N
!>  فردية. نقدم مثالاً حيث N = 5.
!>
!>     AP عليا                 AP سفلى
!>
!>   00 01 02 03 04              00
!>      11 12 13 14              10 11
!>         22 23 24              20 21 22
!>            33 34              30 31 32 33
!>               44              40 41 42 43 44
!>
!>
!>  ليكن TRANSR = 'N'. يخزن RFP المصفوفة AP على النحو التالي:
!>  بالنسبة لـ UPLO = 'U'، فإن شبه المنحرف العلوي A(0:4,0:2) يتكون من
!>  الأعمدة الثلاثة الأخيرة من AP العليا. يتكون المثلث السفلي A(3:4,0:1) من
!>  منقول العمودين الأولين من AP العليا.
!>  بالنسبة لـ UPLO = 'L'، فإن شبه المنحرف السفلي A(0:4,0:2) يتكون من
!>  الأعمدة الثلاثة الأولى من AP السفلى. يتكون المثلث العلوي A(0:1,1:2) من
!>  منقول العمودين الأخيرين من AP السفلى.
!>  هذا يغطي حالة N فردية و TRANSR = 'N'.
!>
!>         RFP A                   RFP A
!>
!>        02 03 04                00 33 43
!>        12 13 14                10 11 44
!>        22 23 24                20 21 22
!>        00 33 34                30 31 32
!>        01 11 44                40 41 42
!>
!>  الآن ليكن TRANSR = 'T'. RFP A في كلتا حالتي UPLO هو فقط
!>  منقول RFP A أعلاه. لذلك يحصل المرء على:
!>
!>           RFP A                   RFP A
!>
!>     02 12 22 00 01             00 10 20 30 40 50
!>     03 13 23 33 11             33 11 21 31 41 51
!>     04 14 24 34 44             43 44 22 32 42 52
!>
