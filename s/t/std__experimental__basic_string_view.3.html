NAME

std::experimental::basic_string_view - a non-owning view over a sequence of characters

SYNOPSIS

#include <experimental/string_view>

template<class CharT, class Traits = std::char_traits<CharT>>
class basic_string_view {
public:
    // types
    using char_type = CharT;
    using traits_type = Traits;
    using size_type = std::size_t;
    using difference_type = std::ptrdiff_t;
    using reference = char_type&;
    using const_reference = const char_type&;
    using pointer = char_type*;
    using const_pointer = const char_type*;
    using const_iterator = implementation_defined;
    using iterator = const_iterator;
    using reverse_iterator = std::reverse_iterator<iterator>;
    using const_reverse_iterator = reverse_iterator;

    // construct/destroy
    constexpr basic_string_view() noexcept;
    constexpr basic_string_view(const basic_string_view& other) noexcept;
    template<class Allocator>
    explicit basic_string_view(const std::basic_string<char_type, traits_type, Allocator>& str);
    template<class ArrayLike>
    explicit basic_string_view(const ArrayLike& arr);
    constexpr basic_string_view(const_pointer s, size_type count);
    constexpr basic_string_view(const_pointer s);

    // assignment
    constexpr basic_string_view& operator=(const basic_string_view& other) noexcept;

    // iterators
    constexpr const_iterator begin() const noexcept;
    constexpr const_iterator cbegin() const noexcept;
    constexpr const_iterator end() const noexcept;
    constexpr const_iterator cend() const noexcept;
    constexpr auto rbegin() const noexcept;
    constexpr auto rend() const noexcept;
    constexpr auto crbegin() const noexcept;
    constexpr auto crend() const noexcept;

    // capacity
    constexpr size_type size() const noexcept;
    constexpr bool empty() const noexcept;

    // element access
    constexpr const_reference operator[](size_type pos) const;
    constexpr const_reference at(size_type pos) const;
    constexpr const_pointer data() const noexcept;

    // modifiers
    constexpr void remove_prefix(size_type n);
    constexpr void remove_suffix(size_type n);
    constexpr void swap(basic_string_view& other) noexcept;

    // comparison
    constexpr int compare(basic_string_view other) const;
    constexpr int compare(size_type pos1, size_type count1, basic_string_view other) const;
    constexpr int compare(size_type pos1, size_type count1, basic_string_view other, size_type pos2, size_type count2) const;
    constexpr int compare(const_pointer s) const;
    constexpr int compare(size_type pos1, size_type count1, const_pointer s) const;
    constexpr int compare(size_type pos1, size_type count1, const_pointer s, size_type count2) const;

    // observers
    constexpr size_type length() const noexcept;
    constexpr size_type max_size() const noexcept;
    constexpr const_reference front() const;
    constexpr const_reference back() const;
    constexpr const_pointer c_str() const noexcept; // Not a C++ standard function, but often implemented

    // substring
    constexpr basic_string_view substr(size_type pos = 0, size_type count = npos) const;

    // searching
    constexpr size_type find(basic_string_view v, size_type pos = 0) const noexcept;
    constexpr size_type find(const_pointer s, size_type pos = 0) const noexcept;
    constexpr size_type find(const_pointer s, size_type pos, size_type count) const noexcept;
    constexpr size_type find(char_type c, size_type pos = 0) const noexcept;

    constexpr size_type rfind(basic_string_view v, size_type pos = npos) const noexcept;
    constexpr size_type rfind(const_pointer s, size_type pos = npos) const noexcept;
    constexpr size_type rfind(const_pointer s, size_type pos, size_type count) const noexcept;
    constexpr size_type rfind(char_type c, size_type pos = npos) const noexcept;

    constexpr size_type find_first_of(basic_string_view v, size_type pos = 0) const noexcept;
    constexpr size_type find_first_of(const_pointer s, size_type pos = 0) const noexcept;
    constexpr size_type find_first_of(const_pointer s, size_type pos, size_type count) const noexcept;
    constexpr size_type find_first_of(char_type c, size_type pos = 0) const noexcept;

    constexpr size_type find_first_not_of(basic_string_view v, size_type pos = 0) const noexcept;
    constexpr size_type find_first_not_of(const_pointer s, size_type pos = 0) const noexcept;
    constexpr size_type find_first_not_of(const_pointer s, size_type pos, size_type count) const noexcept;
    constexpr size_type find_first_not_of(char_type c, size_type pos = 0) const noexcept;

    constexpr size_type find_last_of(basic_string_view v, size_type pos = npos) const noexcept;
    constexpr size_type find_last_of(const_pointer s, size_type pos = npos) const noexcept;
    constexpr size_type find_last_of(const_pointer s, size_type pos, size_type count) const noexcept;
    constexpr size_type find_last_of(char_type c, size_type pos = npos) const noexcept;

    constexpr size_type find_last_not_of(basic_string_view v, size_type pos = npos) const noexcept;
    constexpr size_type find_last_not_of(const_pointer s, size_type pos = npos) const noexcept;

    constexpr size_type find_last_not_of(char_type c, size_type pos = npos) const noexcept;

    static constexpr size_type npos = implementation_defined;
};

DESCRIPTION

std::experimental::basic_string_view is a class template that provides a non-owning view over a sequence of characters. It is similar to std::basic_string, but it does not own the underlying character sequence. This makes it more efficient for operations that do not require modification of the string, such as passing strings to functions or iterating over parts of a string.

A basic_string_view typically stores a pointer to the beginning of the character sequence and its length. It does not manage the lifetime of the character data, meaning the caller is responsible for ensuring the data remains valid for the duration of the string_view's use.

NOTES

  • The std::experimental::basic_string_view is part of the C++ Technical Specification for library fundamentals V1. It is intended to be integrated into the C++ standard in a future revision.
  • Always ensure that the underlying character data pointed to by the string_view remains valid. Dangling references can occur if the original string or character array is destroyed or modified while the string_view still points to it.
  • string_view can be constructed from std::string, C-style strings, and character arrays.
  • Comparison operators are provided for comparing string_view objects with each other and with C-style strings.
  • The npos member is a static member constant value representing a size greater than any possible string length.

الوصف

std::experimental::basic_string_view هو قالب فئة يوفر عرضًا غير مملوك لتسلسل من الأحرف. وهو مشابه لـ std::basic_string، ولكنه لا يمتلك تسلسل الأحرف الأساسي. وهذا يجعله أكثر كفاءة للعمليات التي لا تتطلب تعديل السلسلة، مثل تمرير السلاسل إلى الدوال أو التكرار فوق أجزاء من السلسلة.

يقوم basic_string_view عادةً بتخزين مؤشر إلى بداية تسلسل الأحرف وطوله. لا يقوم بإدارة دورة حياة بيانات الأحرف، مما يعني أن المتصل مسؤول عن ضمان بقاء البيانات صالحة طوال مدة استخدام string_view.

ملاحظات

  • std::experimental::basic_string_view هو جزء من المواصفات التقنية لـ C++ لأساسيات المكتبة الإصدار 1. وهو مخصص للدمج في المعيار C++ في مراجعة مستقبلية.
  • تأكد دائمًا من أن بيانات الأحرف الأساسية التي يشير إليها string_view تظل صالحة. يمكن أن تحدث مراجع معلقة إذا تم تدمير السلسلة الأصلية أو مصفوفة الأحرف أو تعديلها بينما لا يزال string_view يشير إليها.
  • يمكن إنشاء string_view من std::string، سلاسل C-style، ومصفوفات الأحرف.
  • تم توفير عوامل المقارنة لمقارنة كائنات string_view ببعضها البعض ومع سلاسل C-style.
  • العضو npos هو قيمة عضو ثابتة تمثل حجمًا أكبر من أي طول سلسلة ممكن.
