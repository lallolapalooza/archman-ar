subroutine stbtrs (character uplo, character trans, character diag, integer n, integer kd, integer nrhs, real, dimension( ldab, * ) ab, integer ldab, real, dimension( ldb, * ) b, integer ldb, integer info)

STBTRS

الغرض:

!>
!> STBTRS يحل نظامًا مثلثيًا من الشكل
!>
!>    A * X = B  أو  A**T * X = B,
!>
!> حيث A مصفوفة شريط مثلثي من الرتبة N، و B مصفوفة N-by-NRHS.
!>
!> يتحقق هذا الإجراء الفرعي من أن A غير منفردة، ولكن يجب ملاحظة أن عدم الانفرادية التام
!!> فقط هو الذي يتم اكتشافه. من الممكن أن تكون عناصر قطرية واحدة أو أكثر في A أرقامًا صغيرة
!> جدًا دون أن يشير هذا الإجراء الفرعي إلى خطأ.
!>
!> إذا كان فقدان الدقة العددية المحتمل بسبب المصفوفات القريبة من المنفردة يمثل مصدر قلق،
!> فيجب على المتصل التحقق من أن A غير منفردة ضمن نطاق معين قبل استدعاء هذا الإجراء الفرعي.
!>

المعلمات

UPLO 
!>          UPLO هو CHARACTER*1
!>          = 'U':  A هي مثلثية عليا؛
!>          = 'L':  A هي مثلثية سفلى.
!>

TRANS

!>          TRANS هو CHARACTER*1
!>          يحدد شكل نظام المعادلات:
!>          = 'N':  A * X = B  (بدون تبديل)
!>          = 'T':  A**T * X = B  (تبديل)
!>          = 'C':  A**H * X = B  (تبديل مترافق = تبديل)
!>

DIAG

!>          DIAG هو CHARACTER*1
!>          = 'N':  A هي مثلثية غير وحدوية؛
!>          = 'U':  A هي مثلثية وحدوية.
!>

N

!>          N هو INTEGER
!>          رتبة المصفوفة A.  N >= 0.
!>

KD

!>          KD هو INTEGER
!>          عدد الأقطار الفائقة أو الأقطار الفرعية للمصفوفة
!>          الشريطية المثلثية A.  KD >= 0.
!>

NRHS

!>          NRHS هو INTEGER
!>          عدد الأطراف اليمنى، أي عدد أعمدة
!>          المصفوفة B.  NRHS >= 0.
!>

AB

!>          AB هي مصفوفة REAL، البعد (LDAB,N)
!>          المصفوفة الشريطية المثلثية العليا أو السفلى A، مخزنة في
!>          أول kd+1 صفوف من AB.  يتم تخزين العمود j من A
!>          في العمود j من المصفوفة AB على النحو التالي:
!>          إذا كان UPLO = 'U'، فإن AB(kd+1+i-j,j) = A(i,j) لـ max(1,j-kd) <= i <= j؛
!>          إذا كان UPLO = 'L'، فإن AB(1+i-j,j)    = A(i,j) لـ j <= i <= min(n,j+kd).
!>          إذا كان DIAG = 'U'، فإن العناصر القطرية لـ A غير مرجعية
!>          ويفترض أنها 1.
!>

LDAB

!>          LDAB هو INTEGER
!>          البعد الرئيسي للمصفوفة AB.  LDAB >= KD+1.
!>

B

!>          B هي مصفوفة REAL، البعد (LDB,NRHS)
!>          عند الدخول، مصفوفة الطرف الأيمن B.
!>          عند الخروج، إذا كان INFO = 0، فإن مصفوفة الحل X.
!>

LDB

!>          LDB هو INTEGER
!>          البعد الرئيسي للمصفوفة B.  LDB >= max(1,N).
!>

INFO

!>          INFO هو INTEGER
!>          = 0:  انتهاء ناجح
!>          < 0:  إذا كان INFO = -i، فإن المعامل i كان له قيمة غير قانونية
!>          > 0:  إذا كان INFO = i، فإن العنصر القطري i من A هو صفر تمامًا،
!>                مما يشير إلى أن المصفوفة منفردة وأن
!>                الحلول X لم يتم حسابها.
!>
