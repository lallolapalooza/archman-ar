هذا الموقع غير مرتبط بـ Arch Linux ولا تابع له بأي شكل من الأشكال. المحتوى هو ترجمة فقط لصفحات دليل Arch المنشورة على man.archlinux.org.

مثال القائمة

LIST_HEAD(listhead, entry) head =
    LIST_HEAD_INITIALIZER(head);
struct listhead *headp;			/* رأس القائمة. */
struct entry {
	...
	LIST_ENTRY(entry) entries;	/* القائمة. */
	...
} *n1, *n2, *n3, *np, *np_temp;

LIST_INIT(&head);			/* تهيئة القائمة. */

n1 = malloc(sizeof(struct entry));	/* إدراج في الرأس. */
LIST_INSERT_HEAD(&head, n1, entries);

n2 = malloc(sizeof(struct entry));	/* إدراج بعد. */
LIST_INSERT_AFTER(n1, n2, entries);

n3 = malloc(sizeof(struct entry));	/* إدراج قبل. */
LIST_INSERT_BEFORE(n2, n3, entries);

LIST_REMOVE(n2, entries);		/* حذف. */
free(n2);
					/* اجتياز أمامي. */
LIST_FOREACH(np, &head, entries)
	np-> ...

					/* اجتياز أمامي آمن. */
LIST_FOREACH_SAFE(np, &head, entries, np_temp) {
	np->do_stuff();
	...
	LIST_REMOVE(np, entries);
	free(np);
}

while (!LIST_EMPTY(&head)) {		/* حذف القائمة. */
	n1 = LIST_FIRST(&head);
	LIST_REMOVE(n1, entries);
	free(n1);
}

n1 = LIST_FIRST(&head);			/* حذف القائمة أسرع. */
while (n1 != NULL) {
	n2 = LIST_NEXT(n1, entries);
	free(n1);
	n1 = n2;
}
LIST_INIT(&head);

طوابير الذيل

تُدار طوابير الذيل بواسطة بنية تُعرّف باستخدام الماكرو TAILQ_HEAD. تحتوي هذه البنية على زوج من المؤشرات، أحدهما إلى العنصر الأول في طابور الذيل والآخر إلى العنصر الأخير في طابور الذيل. ترتبط العناصر بروابط مزدوجة، بحيث يمكن إزالة عنصر عشوائي دون الحاجة إلى اجتياز طابور الذيل. يمكن إضافة عناصر جديدة إلى طابور الذيل بعد عنصر موجود، أو قبل عنصر موجود، أو في مقدمة طابور الذيل، أو في نهاية طابور الذيل. تُعرّف بنية TAILQ_HEAD على النحو التالي:

TAILQ_HEAD(HEADNAME, TYPE) head;

حيث HEADNAME هو اسم البنية المراد تعريفها و TYPE هو نوع العناصر المرتبطة بطابور الذيل. يمكن لاحقًا تعريف مؤشر إلى مقدمة طابور الذيل:

struct HEADNAME *headp;

(أسماء head و headp قابلة للاختيار من قبل المستخدم.)

الماكرو TAILQ_HEAD_INITIALIZER يُقيّم إلى مُهيئ لطابور الذيل head.

الماكرو TAILQ_CONCAT يربط طابور الذيل المُدار بواسطة head2 بنهاية طابور الذيل المُدار بواسطة head1، مزيلًا جميع العناصر من طابور الذيل الأول.

الماكرو TAILQ_EMPTY يُقيّم إلى صحيح إذا لم يكن هناك عناصر في طابور الذيل.

الماكرو TAILQ_ENTRY يُعرّف بنية تربط عناصر طابور الذيل.

الماكرو TAILQ_FIRST يُرجع العنصر الأول في طابور الذيل أو NULL إذا كان طابور الذيل فارغًا.

الماكرو TAILQ_FOREACH يجتاز طابور الذيل المُدار بواسطة head في الاتجاه الأمامي، مع تعيين كل عنصر بالترتيب إلى var. يتم تعيين var إلى NULL إذا انتهت الحلقة بشكل طبيعي، أو إذا لم تكن هناك عناصر.

الماكرو TAILQ_FOREACH_FROM يتصرف بنفس طريقة TAILQ_FOREACH عندما يكون var هو NULL، حيث يُعتبر var عنصرًا تم العثور عليه مسبقًا في طابور الذيل وتبدأ الحلقة من var بدلاً من العنصر الأول في طابور الذيل المُدار بواسطة head.

الماكرو TAILQ_FOREACH_REVERSE يجتاز طابور الذيل المُدار بواسطة head في الاتجاه العكسي، مع تعيين كل عنصر بالترتيب إلى var.

الماكرو TAILQ_FOREACH_REVERSE_FROM يتصرف بنفس طريقة TAILQ_FOREACH_REVERSE عندما يكون var هو NULL، حيث يُعتبر var عنصرًا تم العثور عليه مسبقًا في طابور الذيل وتبدأ حلقة العكس من var بدلاً من العنصر الأخير في طابور الذيل المُدار بواسطة head.

الماكرو TAILQ_FOREACH_SAFE و TAILQ_FOREACH_REVERSE_SAFE يجتازون القائمة المُدارة بواسطة head في الاتجاه الأمامي أو العكسي على التوالي، مع تعيين كل عنصر بالترتيب إلى var. ومع ذلك، على عكس نظائرهم غير الآمنة، تسمح TAILQ_FOREACH و TAILQ_FOREACH_REVERSE بإزالة var وتحرير محتواه من داخل الحلقة بأمان دون التدخل في الاجتياز.

الماكرو TAILQ_FOREACH_FROM_SAFE يتصرف بنفس طريقة TAILQ_FOREACH_SAFE عندما يكون var هو NULL، حيث يُعتبر var عنصرًا تم العثور عليه مسبقًا في طابور الذيل وتبدأ الحلقة من var بدلاً من العنصر الأول في طابور الذيل المُدار بواسطة head.

الماكرو TAILQ_FOREACH_REVERSE_FROM_SAFE يتصرف بنفس طريقة TAILQ_FOREACH_REVERSE_SAFE عندما يكون var هو NULL، حيث يُعتبر var عنصرًا تم العثور عليه مسبقًا في طابور الذيل وتبدأ حلقة العكس من var بدلاً من العنصر الأخير في طابور الذيل المُدار بواسطة head.

الماكرو TAILQ_INIT يُهيئ طابور الذيل المُدار بواسطة head.

الماكرو TAILQ_INSERT_HEAD يُدرج العنصر الجديد elm في مقدمة طابور الذيل.

الماكرو TAILQ_INSERT_TAIL يُدرج العنصر الجديد elm في نهاية طابور الذيل.

الماكرو TAILQ_INSERT_AFTER يُدرج العنصر الجديد elm بعد العنصر listelm.

الماكرو TAILQ_INSERT_BEFORE يُدرج العنصر الجديد elm قبل العنصر listelm.

الماكرو TAILQ_LAST يُرجع العنصر الأخير في طابور الذيل. إذا كان طابور الذيل فارغًا، فإن القيمة المرجعة هي NULL.

الماكرو TAILQ_NEXT يُرجع العنصر التالي في طابور الذيل، أو NULL إذا كان هذا هو الأخير.

الماكرو TAILQ_PREV يُرجع العنصر السابق في طابور الذيل، أو NULL إذا كان هذا هو الأول.

الماكرو TAILQ_REMOVE يُزيل العنصر elm من طابور الذيل.

الماكرو TAILQ_SWAP يتبادل محتويات head1 و head2.

مثال طابور الذيل

TAILQ_HEAD(tailhead, entry) head =
    TAILQ_HEAD_INITIALIZER(head);
struct tailhead *headp;			/* رأس طابور الذيل. */
struct entry {
	...
	TAILQ_ENTRY(entry) entries;	/* طابور الذيل. */
	...
} *n1, *n2, *n3, *np;

TAILQ_INIT(&head);			/* تهيئة الطابور. */

n1 = malloc(sizeof(struct entry));	/* إدراج في الرأس. */
TAILQ_INSERT_HEAD(&head, n1, entries);

n1 = malloc(sizeof(struct entry));	/* إدراج في الذيل. */
TAILQ_INSERT_TAIL(&head, n1, entries);

n2 = malloc(sizeof(struct entry));	/* إدراج بعد. */
TAILQ_INSERT_AFTER(&head, n1, n2, entries);

n3 = malloc(sizeof(struct entry));	/* إدراج قبل. */
TAILQ_INSERT_BEFORE(n2, n3, entries);

TAILQ_REMOVE(&head, n2, entries);	/* حذف. */
free(n2);
					/* اجتياز أمامي. */
TAILQ_FOREACH(np, &head, entries)
	np-> ...
					/* اجتياز أمامي آمن. */
TAILQ_FOREACH_SAFE(np, &head, entries, np_temp) {
	np->do_stuff();
	...
	TAILQ_REMOVE(&head, np, entries);
	free(np);
}
					/* اجتياز عكسي. */
TAILQ_FOREACH_REVERSE(np, &head, tailhead, entries)
	np-> ...
					/* حذف طابور الذيل. */
while (!TAILQ_EMPTY(&head)) {
	n1 = TAILQ_FIRST(&head);
	TAILQ_REMOVE(&head, n1, entries);
	free(n1);
}
					/* حذف طابور الذيل أسرع. */
n1 = TAILQ_FIRST(&head);
while (n1 != NULL) {
	n2 = TAILQ_NEXT(n1, 
entries);
	free(n1);
	n1 = n2;
}
TAILQ_INIT(&head);

التشخيص

عند تصحيح أخطاء queue(3)، قد يكون من المفيد تتبع التغييرات في الطوابير. لتمكين التتبع، قم بتعريف الماكرو QUEUE_MACRO_DEBUG_TRACE في وقت الترجمة.

قد يكون من المفيد أيضًا تدمير المؤشرات التي تم فصلها من طابور، للكشف عن الاستخدام بعد الإزالة. لتمكين تدمير المؤشرات، قم بتعريف الماكرو QUEUE_MACRO_DEBUG_TRASH في وقت الترجمة. الماكرو QMD_IS_TRASHED(void *ptr) يعيد صحيحًا إذا تم تدمير ptr بواسطة الخيار QUEUE_MACRO_DEBUG_TRASH.
