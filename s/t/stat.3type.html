stat(3type) stat(3type)

الاسم

stat - حالة الملف

المكتبة

مكتبة C القياسية (libc)

المُلخص

#include <sys/stat.h>
struct stat {
    dev_t      st_dev;      /* معرف الجهاز الذي يحتوي على الملف */
    ino_t      st_ino;      /* رقم الـ inode */
    mode_t     st_mode;     /* نوع الملف ووضعه */
    nlink_t    st_nlink;    /* عدد الروابط الثابتة */
    uid_t      st_uid;      /* معرف المستخدم للمالك */
    gid_t      st_gid;      /* معرف المجموعة للمالك */
    dev_t      st_rdev;     /* معرف الجهاز (إذا كان ملفًا خاصًا) */
    off_t      st_size;     /* الحجم الإجمالي، بالبايت */
    blksize_t  st_blksize;  /* حجم القطعة لعمليات إدخال/إخراج نظام الملفات */
    blkcnt_t   st_blocks;   /* عدد القطع بحجم 512 بايت المخصصة */
    /* منذ POSIX.1-2008، يدعم هذا الهيكل دقة النانو ثانية للحقول الزمنية التالية.
       للتفاصيل قبل POSIX.1-2008، انظر VERSIONS.  */
    struct timespec  st_atim;  /* وقت آخر وصول */
    struct timespec  st_mtim;  /* وقت آخر تعديل */
    struct timespec  st_ctim;  /* وقت آخر تغيير للحالة */
#define st_atime  st_atim.tv_sec  /* توافق رجعي */
#define st_mtime  st_mtim.tv_sec
#define st_ctime  st_ctim.tv_sec
};
متطلبات ماكرو اختبار الميزة لـ glibc (انظر feature_test_macros(7)):

st_atim، st_mtim، st_ctim:

    منذ glibc 2.12:
        _POSIX_C_SOURCE >= 200809L || _XOPEN_SOURCE >= 700
    glibc 2.19 والإصدارات السابقة:
        _BSD_SOURCE || _SVID_SOURCE

الوصف

يصف المعلومات حول ملف.

الحقول هي كالتالي:

st_dev
يصف هذا الحقل الجهاز الذي يقع عليه هذا الملف. (قد تكون وحدات major(3) و minor(3) مفيدة لتحليل معرف الجهاز في هذا الحقل.)
st_ino
يحتوي هذا الحقل على رقم الـ inode الخاص بالملف.
st_mode
يحتوي هذا الحقل على نوع الملف ووضعه. انظر inode(7) لمزيد من المعلومات.
st_nlink
يحتوي هذا الحقل على عدد الروابط الثابتة للملف.
st_uid
يحتوي هذا الحقل على معرف المستخدم لمالك الملف.
st_gid
يحتوي هذا الحقل على معرف المجموعة لمالك الملف.
st_rdev
يصف هذا الحقل الجهاز الذي يمثله هذا الملف (inode).
st_size
يعطي هذا الحقل حجم الملف (إذا كان ملفًا عاديًا أو رابطًا رمزيًا) بالبايت. حجم الرابط الرمزي هو طول المسار الذي يحتويه، بدون حرف null نهائي.
st_blksize
يعطي هذا الحقل حجم القطعة "المفضل" لعمليات إدخال/إخراج نظام الملفات بكفاءة.
st_blocks
يشير هذا الحقل إلى عدد القطع المخصصة للملف، بوحدات 512 بايت. (قد يكون هذا أصغر من st_size/512 عندما يحتوي الملف على فجوات.)
st_atime
هذا هو وقت آخر وصول لبيانات الملف.
st_mtime
هذا هو وقت آخر تعديل لبيانات الملف.
st_ctime
هذا هو طابع زمني لآخر تغيير لحالة الملف (وقت آخر تغيير في الـ inode).

لمزيد من المعلومات حول الحقول المذكورة أعلاه، انظر inode(7).

المعايير

POSIX.1-2024.

التاريخ

POSIX.1-2001.

لم تدعم نواة الأنظمة القديمة والمعايير القديمة حقول الطوابع الزمنية بالنانو ثانية. بدلاً من ذلك، كانت هناك ثلاثة حقول للطوابع الزمنية — st_atime، st_mtime، و st_ctime — من نوع time_t سجلت الطوابع الزمنية بدقة ثانية واحدة.

منذ Linux 2.5.48، يدعم هيكل stat دقة النانو ثانية للحقول الزمنية الثلاثة للملفات. مكونات النانو ثانية لكل طابع زمني متاحة عبر أسماء بالشكل st_atim.tv_nsec، إذا تم تعريف ماكرو الاختبار المناسب. تم توحيد الطوابع الزمنية بالنانو ثانية في POSIX.1-2008، واعتبارًا من glibc 2.12، يكشف glibc عن أسماء مكونات النانو ثانية إذا تم تعريف _POSIX_C_SOURCE بالقيمة 200809L أو أكبر، أو تم تعريف _XOPEN_SOURCE بالقيمة 700 أو أكبر. حتى glibc 2.19 والإصدارات السابقة، تم تعريف مكونات النانو ثانية أيضًا إذا تم تعريف _BSD_SOURCE أو _SVID_SOURCE. إذا لم يتم تعريف أي من الماكرو المذكورة أعلاه، فسيتم الكشف عن قيم النانو ثانية بأسماء بالشكل st_atimensec.
