subroutine stpmv (character uplo, character trans, character diag, integer n, real, dimension(*) ap, real, dimension(*) x, integer incx)

STPMV

الغرض:

!>
!> STPMV  يجري إحدى عمليات المصفوفة والمتجه التالية:
!>
!>    x := A*x,   أو   x := A**T*x,
!>
!> حيث x هو متجه n عنصر و A هي مصفوفة مثلثة علوية أو سفلية من الحجم n × n،
!> موحدة أو غير موحدة، مقدمة بشكل مضغوط.
!>

المعلمات

UPLO 
!>          UPLO هو CHARACTER*1
!>           عند الإدخال، يحدد UPLO ما إذا كانت المصفوفة مثلثة علوية أو سفلية
!>           على النحو التالي:
!>
!>              UPLO = 'U' أو 'u'   A مصفوفة مثلثة علوية.
!>
!>              UPLO = 'L' أو 'l'   A مصفوفة مثلثة سفلية.
!>

TRANS

!>          TRANS هو CHARACTER*1
!>           عند الإدخال، يحدد TRANS العملية المراد تنفيذها على النحو التالي:
!>
!>              TRANS = 'N' أو 'n'   x := A*x.
!>
!>              TRANS = 'T' أو 't'   x := A**T*x.
!>
!>              TRANS = 'C' أو 'c'   x := A**T*x.
!>

DIAG

!>          DIAG هو CHARACTER*1
!>           عند الإدخال، يحدد DIAG ما إذا كانت A موحدة مثلثة أم لا
!>           على النحو التالي:
!>
!>              DIAG = 'U' أو 'u'   يفترض أن A موحدة مثلثة.
!>
!>              DIAG = 'N' أو 'n'   لا يفترض أن A موحدة
!>                                  مثلثة.
!>

N

!>          N هو INTEGER
!>           عند الإدخال، يحدد N رتبة المصفوفة A.
!>           يجب أن يكون N على الأقل صفر.
!>

AP

!>          AP هو مصفوفة REAL، بأبعاد على الأقل
!>           ( ( n*( n + 1 ) )/2 ).
!>           قبل الإدخال مع UPLO = 'U' أو 'u'، يجب أن تحتوي المصفوفة AP
!>           على المصفوفة المثلثة العلوية مضغوطة بالتتابع،
!>           عمودًا بعمود، بحيث يحتوي AP( 1 ) على a( 1, 1 )،
!>           ويحتوي AP( 2 ) و AP( 3 ) على a( 1, 2 ) و a( 2, 2 )
!>           على التوالي، وهكذا.
!>           قبل الإدخال مع UPLO = 'L' أو 'l'، يجب أن تحتوي المصفوفة AP
!>           على المصفوفة المثلثة السفلية مضغوطة بالتتابع،
!>           عمودًا بعمود، بحيث يحتوي AP( 1 ) على a( 1, 1 )،
!>           ويحتوي AP( 2 ) و AP( 3 ) على a( 2, 1 ) و a( 3, 1 )
!>           على التوالي، وهكذا.
!>           لاحظ أنه عندما يكون DIAG = 'U' أو 'u'، لا يتم الرجوع إلى العناصر القطرية
!>           لـ A، ولكن يفترض أنها الوحدة.
!>

X

!>          X هي مصفوفة REAL، بأبعاد على الأقل
!>           ( 1 + ( n - 1 )*abs( INCX ) ).
!>           قبل الإدخال، يجب أن تحتوي المصفوفة X ذات الفواصل المتزايدة على المتجه x
!>           ذي العناصر n. عند الخروج، سيتم الكتابة فوق X بالمتجه المحول x.
!>

INCX

!>          INCX هو INTEGER
!>           عند الإدخال، يحدد INCX الفاصل بين عناصر X.
!>           يجب ألا يكون INCX صفرًا.
!>
