subroutine stgsna (job الطابع, howmny الطابع, select منطقي, dimension( * ) select, integer n, real, dimension( lda, * ) a, integer lda, real, dimension( ldb, * ) b, integer ldb, real, dimension( ldvl, * ) vl, integer ldvl, real, dimension( ldvr, * ) vr, integer ldvr, real, dimension( * ) s, real, dimension( * ) dif, integer mm, integer m, real, dimension( * ) work, integer lwork, integer, dimension( * ) iwork, integer info)

STGSNA

الغرض:

!>
!> STGSNA تقدّر أرقام شروط عكسية للقيم الذاتية و/أو المتجهات الذاتية المحددة
!> لمصفوفة زوج (A, B) في شكل Schur العام الحقيقي (أو لأي زوج مصفوفات
!> (Q*A*Z**T, Q*B*Z**T) مع مصفوفات متعامدة Q و Z، حيث Z**T هو منقول Z.
!>
!> يجب أن تكون (A, B) في شكل Schur العام الحقيقي (كما تم إرجاعها بواسطة SGGES)،
!> أي أن A تكون شبه علوية بشكل كتل مع كتل قطرية بحجم 1x1 و 2x2.
!> B تكون علوية مثلثية.
!>
!>

المعلمات

JOB 
!>          JOB هو CHARACTER*1
!>          يحدد ما إذا كانت أرقام الشروط مطلوبة للقيم الذاتية (S) أو المتجهات الذاتية (DIF):
!>          = 'E': للقيم الذاتية فقط (S).
!>          = 'V': للمتجهات الذاتية فقط (DIF).
!>          = 'B': لكل من القيم الذاتية والمتجهات الذاتية (S و DIF).
!>

HOWMNY

!>          HOWMNY هو CHARACTER*1
!>          = 'A': حساب أرقام الشروط لجميع الأزواج الذاتية.
!>          = 'S': حساب أرقام الشروط للأزواج الذاتية المحددة
!>                 المحددة بواسطة المصفوفة SELECT.
!>

SELECT

!>          SELECT هي مصفوفة منطقية، dimension (N)
!>          إذا كان HOWMNY = 'S'، فإن SELECT تحدد الأزواج الذاتية المطلوبة
!>          لأجلها أرقام الشروط. لتحديد أرقام الشروط
!>          المقابلة لقيمة ذاتية حقيقية w(j)،
!>          يجب تعيين SELECT(j) إلى .TRUE.. لتحديد أرقام الشروط
!>          المقابلة لزوج مرافق معقد من القيم الذاتية w(j)
!>          و w(j+1)، يجب تعيين SELECT(j) أو SELECT(j+1) أو كليهما
!>          إلى .TRUE..
!>          إذا كان HOWMNY = 'A'، فلا يتم الرجوع إلى SELECT.
!>

N

!>          N هو INTEGER
!>          رتبة زوج المصفوفات المربع (A, B). N >= 0.
!>

A

!>          A هي مصفوفة REAL، dimension (LDA,N)
!>          المصفوفة شبه المثلثية العليا A في الزوج (A,B).
!>

LDA

!>          LDA هو INTEGER
!>          البعد القيادي للمصفوفة A. LDA >= max(1,N).
!>

B

!>          B هي مصفوفة REAL، dimension (LDB,N)
!>          المصفوفة المثلثية العليا B في الزوج (A,B).
!>

LDB

!>          LDB هو INTEGER
!>          البعد القيادي للمصفوفة B. LDB >= max(1,N).
!>

VL

!>          VL هي مصفوفة REAL، dimension (LDVL,M)
!>          إذا كان JOB = 'E' أو 'B'، يجب أن تحتوي VL على المتجهات الذاتية اليسرى
!>          للأزواج الذاتية (A, B)، المقابلة للأزواج الذاتية المحددة بواسطة HOWMNY
!>          و SELECT. يجب تخزين المتجهات الذاتية في أعمدة متجاورة من VL،
!>          كما تم إرجاعها بواسطة STGEVC.
!>          إذا كان JOB = 'V'، فلا يتم الرجوع إلى VL.
!>

LDVL

!>          LDVL هو INTEGER
!>          البعد القيادي للمصفوفة VL. LDVL >= 1.
!>          إذا كان JOB = 'E' أو 'B'، فإن LDVL >= N.
!>

VR

!>          VR هي مصفوفة REAL، dimension (LDVR,M)
!>          إذا كان JOB = 'E' أو 'B'، يجب أن تحتوي VR على المتجهات الذاتية اليمنى
!>          للأزواج الذاتية (A, B)، المقابلة للأزواج الذاتية المحددة بواسطة HOWMNY
!>          و SELECT. يجب تخزين المتجهات الذاتية في أعمدة متجاورة من VR،
!>          كما تم إرجاعها بواسطة STGEVC.
!>          إذا كان JOB = 'V'، فلا يتم الرجوع إلى VR.
!>

LDVR

!>          LDVR هو INTEGER
!>          البعد القيادي للمصفوفة VR. LDVR >= 1.
!>          إذا كان JOB = 'E' أو 'B'، فإن LDVR >= N.
!>

S

!>          S هي مصفوفة REAL، dimension (MM)
!>          إذا كان JOB = 'E' أو 'B'، فإن أرقام الشروط العكسية للقيم الذاتية المحددة،
!>          مخزنة في عناصر متجاورة من المصفوفة. بالنسبة لزوج مرافق معقد
!>          من القيم الذاتية، يتم تعيين عنصرين متجاورين من S إلى نفس القيمة.
!>          وبالتالي، S(j)، DIF(j)، والأعمدة j من VL و VR كلها تتوافق مع
!>          نفس الزوج الذاتي (ولكن ليس في العموم الزوج الذاتي j، إلا إذا تم
!>          تحديد جميع الأزواج الذاتية).
!>          إذا كان JOB = 'V'، فلا يتم الرجوع إلى S.
!>

DIF

!>          DIF هي مصفوفة REAL، dimension (MM)
!>          إذا كان JOB = 'V' أو 'B'، فإن تقديرات أرقام الشروط العكسية للمتجهات الذاتية المحددة،
!>          مخزنة في عناصر متجاورة من المصفوفة. بالنسبة لمتجه ذاتي معقد،
!>          يتم تعيين عنصرين متجاورين من DIF إلى نفس القيمة. إذا لم يكن من الممكن
!>          إعادة ترتيب القيم الذاتية لحساب DIF(j)، يتم تعيين DIF(j) إلى 0؛
!>          يمكن أن يحدث هذا فقط عندما تكون القيمة الحقيقية صغيرة جدًا على أي حال.
!>          إذا كان JOB = 'E'، فلا يتم الرجوع إلى DIF.
!>

MM

!>          MM هو INTEGER
!>          عدد العناصر في المصفوفتين S و DIF. MM >= M.
!>

M

!>          M هو INTEGER
!>          عدد العناصر في المصفوفتين S و DIF المستخدمة لتخزين
!>          أرقام الشروط المحددة؛ لكل قيمة ذاتية حقيقية محددة، يتم استخدام عنصر واحد،
!>          ولكل زوج مرافق معقد من القيم الذاتية، يتم استخدام عنصرين.
!>          إذا كان HOWMNY = 'A'، يتم تعيين M إلى N.
!>

WORK

!>          WORK هي مصفوفة REAL، dimension (MAX(1,LWORK))
!>          عند الخروج، إذا كان INFO = 0، فإن WORK(1) ترجع LWORK المثلى.
!>

LWORK

!>          LWORK هو INTEGER
!>          بعد المصفوفة WORK. LWORK >= max(1,N).
!>          إذا كان JOB = 'V' أو 'B'، فإن LWORK >= 2*N*(N+2)+16.
!>
!>          إذا كان LWORK = -1، فمن المفترض وجود استعلام مساحة عمل؛ يقوم الروتين
!>          فقط بحساب الحجم الأمثل للمصفوفة WORK، ويرجع هذه القيمة كأول إدخال
!>          للمصفوفة WORK، ولا يتم إصدار أي رسالة خطأ متعلقة بـ LWORK بواسطة XERBLA.
!>

IWORK

!>          IWORK هي مصفوفة INTEGER، dimension (N + 6)
!>          إذا كان JOB = 'E'، فلا يتم الرجوع إلى IWORK.
!>

INFO

!>          INFO هو INTEGER
!>          =0: خروج ناجح
!>          <0: إذا كان INFO = -i، فإن المعلمة i كانت لها قيمة غير قانونية
!>

!>
!>  يتم تعريف مقلوب رقم شرط قيمة ذاتية عامة
!>  w = (a, b) على أنه
!>
!>       S(w) = (|u**TAv|**2 + |u**TBv|**2)**(1/2) / (norm(u)*norm(v))
!>
!>  حيث u و v هما المتجهان الذاتيان الأيسر والأيمن لـ (A, B)
!>  المقابلان لـ w؛ |z| تعني القيمة المطلقة للعدد المعقد،
!>  و norm(u) تعني المعيار 2 للمتجه u.
!>  الزوج (a, b) يقابل قيمة ذاتية w = a/b (= u**TAv/u**TBv)
!>  للمصفوفة الزوجية (A, B). إذا كان كل من a و b يساويان الصفر، فإن (A B)
!>  تكون شاذة ويتم إرجاع S(I) = -1.
!
