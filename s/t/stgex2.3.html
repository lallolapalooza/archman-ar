توثيق الوظيفة/الروتين الفرعي

subroutine stgex2 (logical wantq, logical wantz, integer n, real, dimension( lda, * ) a, integer lda, real, dimension( ldb, * ) b, integer ldb, real, dimension( ldq, * ) q, integer ldq, real, dimension( ldz, * ) z, integer ldz, integer j1, integer n1, integer n2, real, dimension( * ) work, integer lwork, integer info)

STGEX2 يبدل الكتل القطرية المتجاورة في زوج مصفوفة مثلثية عليا (شبه) عبر تحويل تكافؤ متعامد.

الغرض:

!>
!> STGEX2 يبدل الكتل القطرية المتجاورة (A11, B11) و (A22, B22)
!> بحجم 1x1 أو 2x2 في زوج مصفوفة مثلثية عليا (شبه) (A, B) عبر تحويل
!> تكافؤ متعامد.
!>
!> يجب أن تكون (A, B) في شكل شوري عام حقيقي (كما تم إرجاعه بواسطة
!> SGGES)، أي A مصفوفة مثلثية عليا مقسمة إلى كتل مع كتل قطرية بحجم
!> 1x1 و 2x2. B مصفوفة مثلثية عليا.
!>
!> اختياريًا، يتم تحديث مصفوفتي المتجهات الشورية العامة Q و Z.
!>
!>        Q(in) * A(in) * Z(in)**T = Q(out) * A(out) * Z(out)**T
!>        Q(in) * B(in) * Z(in)**T = Q(out) * B(out) * Z(out)**T
!>
!>

المعلمات

WANTQ 
!>          WANTQ هو LOGICAL (منطقي)
!>          .TRUE. : تحديث مصفوفة التحويل اليسرى Q؛
!>          .FALSE.: لا تحديث Q.
!>

WANTZ

!>          WANTZ هو LOGICAL (منطقي)
!>          .TRUE. : تحديث مصفوفة التحويل اليمنى Z؛
!>          .FALSE.: لا تحديث Z.
!>

N

!>          N هو INTEGER (عدد صحيح)
!>          رتبة المصفوفتين A و B. N >= 0.
!>

A

!>          A هي مصفوفة REAL (حقيقية)، ذات أبعاد (LDA,N)
!>          عند الدخول، المصفوفة A في الزوج (A, B).
!>          عند الخروج، المصفوفة A المحدثة.
!>

LDA

!>          LDA هو INTEGER (عدد صحيح)
!>          البعد الرائد للمصفوفة A. LDA >= max(1,N).
!>

B

!>          B هي مصفوفة REAL (حقيقية)، ذات أبعاد (LDB,N)
!>          عند الدخول، المصفوفة B في الزوج (A, B).
!>          عند الخروج، المصفوفة B المحدثة.
!>

LDB

!>          LDB هو INTEGER (عدد صحيح)
!>          البعد الرائد للمصفوفة B. LDB >= max(1,N).
!>

Q

!>          Q هي مصفوفة REAL (حقيقية)، ذات أبعاد (LDQ,N)
!>          عند الدخول، إذا كان WANTQ = .TRUE.، فإن المصفوفة المتعامدة Q.
!>          عند الخروج، المصفوفة Q المحدثة.
!>          لا يتم الرجوع إليها إذا كان WANTQ = .FALSE..
!>

LDQ

!>          LDQ هو INTEGER (عدد صحيح)
!>          البعد الرائد للمصفوفة Q. LDQ >= 1.
!>          إذا كان WANTQ = .TRUE.، فإن LDQ >= N.
!>

Z

!>          Z هي مصفوفة REAL (حقيقية)، ذات أبعاد (LDZ,N)
!>          عند الدخول، إذا كان WANTZ = .TRUE.، فإن المصفوفة المتعامدة Z.
!>          عند الخروج، المصفوفة Z المحدثة.
!>          لا يتم الرجوع إليها إذا كان WANTZ = .FALSE..
!>

LDZ

!>          LDZ هو INTEGER (عدد صحيح)
!>          البعد الرائد للمصفوفة Z. LDZ >= 1.
!>          إذا كان WANTZ = .TRUE.، فإن LDZ >= N.
!>

J1

!>          J1 هو INTEGER (عدد صحيح)
!>          المؤشر إلى الكتلة الأولى (A11, B11). 1 <= J1 <= N.
!>

N1

!>          N1 هو INTEGER (عدد صحيح)
!>          رتبة الكتلة الأولى (A11, B11). N1 = 0، 1 أو 2.
!>

N2

!>          N2 هو INTEGER (عدد صحيح)
!>          رتبة الكتلة الثانية (A22, B22). N2 = 0، 1 أو 2.
!>

WORK

!>          WORK هي مصفوفة REAL (حقيقية)، ذات أبعاد (MAX(1,LWORK)).
!>

LWORK

!>          LWORK هو INTEGER (عدد صحيح)
!>          بعد المصفوفة WORK.
!>          LWORK >=  MAX( N*(N2+N1), (N2+N1)*(N2+N1)*2 )
!>

INFO

!>          INFO هو INTEGER (عدد صحيح)
!>            =0: خروج ناجح
!>            >0: إذا كان INFO = 1، ستكون المصفوفة المحولة (A, B)
!>                بعيدة جدًا عن الشكل الشوري العام؛ لم يتم تبديل الكتل
!>                و (A, B) و (Q, Z) لم تتغير.
!>                مشكلة التبديل غير مستقرة للغاية.
!>            <0: إذا كان INFO = -16: LWORK صغير جدًا. القيمة المناسبة
!>                لـ LWORK يتم إرجاعها في WORK(1).
!>
