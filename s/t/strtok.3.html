strtok(3) دليل دوال المكتبة strtok(3)

الاسم

strtok, strtok_r - استخراج أجزاء (tokens) من السلاسل النصية

المكتبة

مكتبة C القياسية (libc, -lc)

ملخص

#include <string.h>
char *strtok(char *_Nullable restrict str, const char *restrict delim);
char *strtok_r(char *_Nullable restrict str, const char *restrict delim,
               char **restrict saveptr);
متطلبات ميزة الاختبار Macro لـ glibc (انظر feature_test_macros(7)):

strtok_r():

    _POSIX_C_SOURCE
        || /* glibc <= 2.19: */ _BSD_SOURCE || _SVID_SOURCE

الوصف

تقوم الدالة strtok() بتقسيم سلسلة نصية إلى سلسلة من الأجزاء غير الفارغة أو الصفر. في أول استدعاء لـ strtok()، يجب تحديد السلسلة النصية المراد تحليلها في str. في كل استدعاء لاحق يجب أن يحلل نفس السلسلة النصية، يجب أن تكون str هي NULL.

يحدد الوسيط delim مجموعة من البايتات التي تفصل الأجزاء في السلسلة النصية التي يتم تحليلها. يجوز للمستدعي تحديد سلاسل مختلفة في delim في استدعاءات متعاقبة تحلل نفس السلسلة النصية.

كل استدعاء لـ strtok() يعيد مؤشرًا إلى سلسلة نصية منتهية بـ null تحتوي على الجزء التالي. هذه السلسلة النصية لا تشمل البايت الفاصل. إذا لم يتم العثور على أجزاء أخرى، تعيد strtok() القيمة NULL.

سلسلة من استدعاءات strtok() التي تعمل على نفس السلسلة النصية تحافظ على مؤشر يحدد النقطة التي يبدأ منها البحث عن الجزء التالي. أول استدعاء لـ strtok() يجعل هذا المؤشر يشير إلى البايت الأول من السلسلة النصية. يتم تحديد بداية الجزء التالي عن طريق المسح للأمام بحثًا عن البايت التالي غير الفاصل في str. إذا تم العثور على مثل هذا البايت، فإنه يؤخذ كبداية للجزء التالي. إذا لم يتم العثور على مثل هذا البايت، فهذا يعني عدم وجود أجزاء أخرى، وتعيد strtok() القيمة NULL. (سلسلة نصية فارغة أو تحتوي فقط على فواصل ستؤدي بالتالي إلى قيام strtok() بإعادة القيمة NULL عند أول استدعاء).

يتم العثور على نهاية كل جزء عن طريق المسح للأمام حتى يتم العثور على البايت الفاصل التالي أو حتى يتم الوصول إلى البايت null المنتهي ('\0'). إذا تم العثور على بايت فاصل، يتم استبداله ببايت null لإنهاء الجزء الحالي، وتقوم strtok() بحفظ مؤشر إلى البايت التالي؛ سيتم استخدام هذا المؤشر كنقطة بداية عند البحث عن الجزء التالي. في هذه الحالة، تعيد strtok() مؤشرًا إلى بداية الجزء الذي تم العثور عليه.

من الوصف أعلاه، يتبع أن سلسلة من بايتين متتاليين أو أكثر من الفواصل في السلسلة النصية التي يتم تحليلها تعتبر فاصلًا واحدًا، ويتم تجاهل الفواصل في بداية أو نهاية السلسلة النصية. بعبارة أخرى: الأجزاء التي تعيدها strtok() هي دائمًا سلاسل نصية غير فارغة. وبالتالي، على سبيل المثال، بالنظر إلى السلسلة النصية "aaa;;bbb,"، فإن الاستدعاءات المتتالية لـ strtok() التي تحدد سلسلة الفواصل ";," ستعيد السلاسل النصية "aaa" و "bbb"، ثم مؤشر null.

الدالة strtok_r() هي نسخة قابلة لإعادة الدخول (reentrant) من strtok(). الوسيط saveptr هو مؤشر إلى متغير من النوع char * يتم استخدامه داخليًا بواسطة strtok_r() للحفاظ على السياق بين الاستدعاءات المتتالية التي تحلل نفس السلسلة النصية.

في أول استدعاء لـ strtok_r()، يجب أن يشير str إلى السلسلة النصية المراد تحليلها، ويتم تجاهل قيمة *saveptr (ولكن انظر VERSIONS). في الاستدعاءات اللاحقة، يجب أن تكون str هي NULL، ويجب أن تظل saveptr (والمخزن المؤقت الذي يشير إليه) دون تغيير منذ الاستدعاء السابق.

يمكن تحليل سلاسل نصية مختلفة بشكل متزامن باستخدام سلاسل من استدعاءات strtok_r() التي تحدد وسيطات saveptr مختلفة.

قيمة الإرجاع

تعيد الدالتان strtok() و strtok_r() مؤشرًا إلى الجزء التالي، أو NULL إذا لم تكن هناك أجزاء أخرى.

الأخطاء

كن حذرًا عند استخدام هذه الدوال. إذا قمت باستخدامها، لاحظ أن:

•
هذه الدوال تعدل وسيطها الأول.
•
لا يمكن استخدام هذه الدوال على السلاسل النصية الثابتة.
•
هوية البايت الفاصل مفقودة.
•
تستخدم الدالة strtok() مخزنًا مؤقتًا ثابتًا أثناء التحليل، لذا فهي ليست آمنة للخيوط. استخدم strtok_r() إذا كان هذا يهمك.

أمثلة

يستخدم البرنامج أدناه حلقات متداخلة تستخدم strtok_r() لتقسيم سلسلة نصية إلى تسلسل هرمي من مستويين من الأجزاء. يحدد وسيط سطر الأوامر الأول السلسلة النصية المراد تحليلها. يحدد الوسيط الثاني البايتات الفاصلة التي سيتم استخدامها لفصل هذه السلسلة إلى أجزاء "رئيسية". يحدد الوسيط الثالث البايتات الفاصلة التي سيتم استخدامها لفصل الأجزاء "الرئيسية" إلى أجزاء فرعية.

مثال على الإخراج الناتج عن هذا البرنامج هو كالتالي:


$ ./a.out 'a/bbb///cc;xxx:yyy:' ':;' '/'
1: a/bbb///cc
         --> a
         --> bbb
         --> cc
2: xxx
         --> xxx
3: yyy
         --> yyy

مصدر البرنامج

#include <stdio.h>
#include <stdlib.h>
#include <string.h>
int
main(int argc, char *argv[])
{
    char *str1, *str2, *token, *subtoken;
    char *saveptr1, *saveptr2;
    int j;
    if (argc != 4) {
        fprintf(stderr, "Usage: %s string delim subdelim\n",
                argv[0]);
        exit(EXIT_FAILURE);
    }
    for (j = 1, str1 = argv[1]; ; j++, str1 = NULL) {
        token = strtok_r(str1, argv[2], &saveptr1);
        if (token == NULL)
            break;
        printf("%d: %s\n", j, token);
        for (str2 = token; ; str2 = NULL) {
            subtoken = strtok_r(str2, argv[3], &saveptr2);
            if (subtoken == NULL)
                break;
            printf("\t --> %s\n", subtoken);
        }
    }
    exit(EXIT_SUCCESS);
}

يمكن العثور على مثال آخر لبرنامج يستخدم strtok() في getaddrinfo_a(3).
