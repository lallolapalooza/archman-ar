الاسم

std::__numeric_limits_base - جزء من std::numeric_limits.

ملخص

#include <limits>

موروث بواسطة std::numeric_limits< bool >، std::numeric_limits< char >، std::numeric_limits< char16_t >، std::numeric_limits< char32_t >، std::numeric_limits< const _Tp >، std::numeric_limits< const volatile _Tp >، std::numeric_limits< double >، std::numeric_limits< float >، std::numeric_limits< int >، std::numeric_limits< long >، std::numeric_limits< long double >، std::numeric_limits< long long >، std::numeric_limits< short >، std::numeric_limits< signed char >، std::numeric_limits< unsigned char >، std::numeric_limits< unsigned int >، std::numeric_limits< unsigned long >، std::numeric_limits< unsigned long long >، std::numeric_limits< unsigned short >، std::numeric_limits< volatile _Tp >، std::numeric_limits< wchar_t >، و std::numeric_limits< _Tp >.

توثيق بيانات الأعضاء

int std::__numeric_limits_base::digits [static], [constexpr]

عدد الأرقام الأساسية التي يمكن تمثيلها دون تغيير: لأنواع الأعداد الصحيحة، عدد البتات غير الإشارية في المانتيس؛ لأنواع الفاصلة العائمة، عدد أرقام الأساس في المانتيس.

تمت الإشارة إليها بواسطة std::generate_canonical()، std::independent_bits_engine< _RandomNumberEngine, __w, _UIntType >::operator()()، std::operator<<()، و std::linear_congruential_engine< uint_fast32_t, 16807UL, 0UL, 2147483647UL >::seed().

int std::__numeric_limits_base::digits10 [static], [constexpr]

عدد أرقام الأساس 10 التي يمكن تمثيلها دون تغيير.

float_denorm_style std::__numeric_limits_base::has_denorm [static], [constexpr]

انظر std::float_denorm_style لمزيد من المعلومات.

bool std::__numeric_limits_base::has_denorm_loss [static], [constexpr]

صحيح إذا تم اكتشاف فقدان الدقة كفقدان إزالة الاستواء، بدلاً من كونه نتيجة غير دقيقة.

bool std::__numeric_limits_base::has_infinity [static], [constexpr]

صحيح إذا كان للنوع تمثيل لـ +infinity.

bool std::__numeric_limits_base::has_quiet_NaN [static], [constexpr]

صحيح إذا كان للنوع تمثيل لـ "ليس رقمًا هادئًا" (غير مولد للإشارة).

bool std::__numeric_limits_base::has_signaling_NaN [static], [constexpr]

صحيح إذا كان للنوع تمثيل لـ "ليس رقمًا مولدًا للإشارة".

bool std::__numeric_limits_base::is_bounded [static], [constexpr]

صحيح إذا كانت مجموعة القيم التي يمكن تمثيلها بواسطة النوع محدودة. جميع الأنواع المدمجة محدودة، ستكون هذه القيمة خاطئة لأنواع الدقة التعسفية.

bool std::__numeric_limits_base::is_exact [static], [constexpr]

صحيح إذا كان النوع يستخدم تمثيلًا دقيقًا. جميع أنواع الأعداد الصحيحة دقيقة، ولكن ليست كل الأنواع الدقيقة أعدادًا صحيحة. على سبيل المثال، التمثيلات ذات الأسس العقلانية والثابتة دقيقة ولكنها ليست صحيحة.

bool std::__numeric_limits_base::is_iec559 [static], [constexpr]

صحيح إذا وفقط إذا كان النوع يلتزم بمعيار IEC 559، المعروف أيضًا باسم IEEE 754. (له معنى فقط لأنواع الفاصلة العائمة.)

bool std::__numeric_limits_base::is_integer [static], [constexpr]

صحيح إذا كان النوع عددًا صحيحًا.

bool std::__numeric_limits_base::is_modulo [static], [constexpr]

صحيح إذا كان النوع متوافقًا مع المعامل. النوع يكون متوافقًا مع المعامل إذا، لأي عملية تتضمن + أو - أو * على قيم من هذا النوع والتي تقع نتيجتها خارج النطاق [min(), max()]، فإن القيمة المرجعة تختلف عن القيمة الحقيقية بمضاعف صحيح لـ max() - min() + 1. في معظم الأجهزة، هذا خاطئ لأنواع الفاصلة العائمة، صحيح للأعداد الصحيحة غير الموقعة، وصحيح للأعداد الصحيحة الموقعة. انظر PR22200 بخصوص الأعداد الصحيحة الموقعة.

bool std::__numeric_limits_base::is_signed [static], [constexpr]

صحيح إذا كان النوع موقعًا.

bool std::__numeric_limits_base::is_specialized [static], [constexpr]

سيكون هذا صحيحًا لجميع الأنواع الأساسية (التي لديها تجسيدات)، وخاطئًا لكل شيء آخر.

int std::__numeric_limits_base::max_digits10 [static], [constexpr]

عدد الأرقام بالنظام العشري المطلوبة لضمان التمييز بين القيم المختلفة دائمًا.

int std::__numeric_limits_base::max_exponent [static], [constexpr]

أقصى عدد صحيح موجب بحيث يكون الأساس مرفوعًا للقوة (أقل بواحد من هذا العدد) هو رقم فاصل عائم محدود قابل للتمثيل.

int std::__numeric_limits_base::max_exponent10 [static], [constexpr]

أقصى عدد صحيح موجب بحيث يكون 10 مرفوعًا لهذه القوة ضمن نطاق الأرقام الفاصلة العائمة المحدودة القابلة للتمثيل.

int std::__numeric_limits_base::min_exponent [static], [constexpr]

أصغر عدد صحيح سالب بحيث يكون الأساس مرفوعًا للقوة (أقل بواحد من هذا العدد) هو رقم فاصل عائم طبيعي.

int std::__numeric_limits_base::min_exponent10 [static], [constexpr]

أصغر عدد صحيح سالب بحيث يكون 10 مرفوعًا لهذه القوة ضمن نطاق الأرقام الفاصلة العائمة الطبيعية.

int std::__numeric_limits_base::radix [static], [constexpr]

بالنسبة لأنواع الأعداد الصحيحة، تحدد أساس التمثيل. بالنسبة لأنواع الأرقام الفاصلة، تحدد أساس تمثيل الأس.

float_round_style std::__numeric_limits_base::round_style [static], [constexpr]

انظر std::float_round_style لمزيد من المعلومات. هذا له معنى فقط لأنواع الأرقام الفاصلة؛ ستكون جميع أنواع الأعداد الصحيحة round_toward_zero.

bool std::__numeric_limits_base::tinyness_before [static], [constexpr]

صحيح إذا تم اكتشاف الصغر قبل التقريب. (راجع IEC 559)

bool std::__numeric_limits_base::traps [static], [constexpr]

صحيح إذا كان الإمساك مطبقًا لهذا النوع.
