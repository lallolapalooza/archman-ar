NAME

STAILQ_INSERT_HEAD, STAILQ_INSERT_TAIL, STAILQ_INSERT_AFTER, STAILQ_INSERT_BEFORE — add entries to a singly linked tail queue

LIBRARY

libbsd

SYNOPSIS

#include <sys/queue.h>

#include <libbsd.h>

void
STAILQ_INSERT_HEAD(STAILQ_HEAD_T(head, type) *headp, type *elm,
    entries);

void
STAILQ_INSERT_TAIL(STAILQ_HEAD_T(head, type) *headp, type *elm,
    entries);

void
STAILQ_INSERT_AFTER(STAILQ_HEAD_T(head, type) *headp, type *listelm,
    type *elm, entries);

void
STAILQ_INSERT_BEFORE(STAILQ_HEAD_T(head, type) *headp, type *listelm,
    type *elm, entries);

DESCRIPTION

These functions are used to insert elements into a singly linked tail queue. A singly linked tail queue is a queue where elements are added to the tail (end) of the queue, but elements can be added to the head (beginning) or tail. The order of elements is maintained, and it is possible to traverse the list using an iterator.

STAILQ_INSERT_HEAD inserts the element elm at the head of the queue pointed to by headp. The entries argument is the name of the macro that defines the next pointer field in the structure type.

STAILQ_INSERT_TAIL inserts the element elm at the tail of the queue pointed to by headp. The entries argument is the name of the macro that defines the next pointer field in the structure type.

STAILQ_INSERT_AFTER inserts the element elm after the element listelm in the queue pointed to by headp. The entries argument is the name of the macro that defines the next pointer field in the structure type.

STAILQ_INSERT_BEFORE inserts the element elm before the element listelm in the queue pointed to by headp. The entries argument is the name of the macro that defines the next pointer field in the structure type.

RETURN VALUE

None.

VERSIONS

These functions first appeared in FreeBSD 3.0.

BUGS

There are no known bugs.

STANDARDS

These functions are POSIX compliant.

HISTORY

The STAILQ macros were introduced in BSD 4.4.

EXAMPLES

The following example demonstrates how to use STAILQ_INSERT_HEAD and STAILQ_FOREACH to insert and traverse a singly linked tail queue.

/* Define a structure for the queue elements */
struct my_entry {
    int data;
    STAILQ_ENTRY(my_entry) entries;
};

int
main() {
    STAILQ_HEAD(my_list, my_entry) head;
    struct my_entry *np, *n1, *n2;

    /* Initialize the queue head */
    STAILQ_INIT(&head);

    /* Allocate and insert elements at the head */
    np = malloc(sizeof(*np));
    np->data = 1;
    STAILQ_INSERT_HEAD(&head, np, entries);

    np = malloc(sizeof(*np));
    np->data = 2;
    STAILQ_INSERT_HEAD(&head, np, entries);

    np = malloc(sizeof(*np));
    np->data = 3;
    STAILQ_INSERT_HEAD(&head, np, entries);

    /* Forward traversal */
    STAILQ_FOREACH(np, &head, entries)
        printf("%i\n", np->data);
                                            /* TailQ deletion */
    n1 = STAILQ_FIRST(&head);
    while (n1 != NULL) {
        n2 = STAILQ_NEXT(n1, entries);
        free(n1);
        n1 = n2;
    }
    STAILQ_INIT(&head);
    exit(EXIT_SUCCESS);
}

STAILQ_INSERT_HEAD، STAILQ_INSERT_TAIL، STAILQ_INSERT_AFTER، STAILQ_INSERT_BEFORE — إضافة إدخالات إلى صف ذيل مرتبط بشكل فردي

الوصف

تُستخدم هذه الدوال لإدراج عناصر في صف ذيل مرتبط بشكل فردي. صف الذيل المرتبط بشكل فردي هو صف تُضاف فيه العناصر إلى الذيل (نهاية) الصف، ولكن يمكن إضافة العناصر إلى الرأس (البداية) أو الذيل. يتم الحفاظ على ترتيب العناصر، ومن الممكن اجتياز القائمة باستخدام مكرر.

STAILQ_INSERT_HEAD تُدرج العنصر elm في رأس الصف المشار إليه بواسطة headp. الوسيط entries هو اسم الماكرو الذي يحدد حقل المؤشر التالي في البنية type.

STAILQ_INSERT_TAIL تُدرج العنصر elm في ذيل الصف المشار إليه بواسطة headp. الوسيط entries هو اسم الماكرو الذي يحدد حقل المؤشر التالي في البنية type.

STAILQ_INSERT_AFTER تُدرج العنصر elm بعد العنصر listelm في الصف المشار إليه بواسطة headp. الوسيط entries هو اسم الماكرو الذي يحدد حقل المؤشر التالي في البنية type.

STAILQ_INSERT_BEFORE تُدرج العنصر elm قبل العنصر listelm في الصف المشار إليه بواسطة headp. الوسيط entries هو اسم الماكرو الذي يحدد حقل المؤشر التالي في البنية type.

أمثلة

يوضح المثال التالي كيفية استخدام STAILQ_INSERT_HEAD و STAILQ_FOREACH لإدراج واجتياز صف ذيل مرتبط بشكل فردي.

/* تعريف بنية لعناصر الصف */
struct my_entry {
    int data;
    STAILQ_ENTRY(my_entry) entries;
};

int
main() {
    STAILQ_HEAD(my_list, my_entry) head;
    struct my_entry *np, *n1, *n2;

    /* تهيئة رأس الصف */
    STAILQ_INIT(&head);

    /* تخصيص وإدراج عناصر في الرأس */
    np = malloc(sizeof(*np));
    np->data = 1;
    STAILQ_INSERT_HEAD(&head, np, entries);

    np = malloc(sizeof(*np));
    np->data = 2;
    STAILQ_INSERT_HEAD(&head, np, entries);

    np = malloc(sizeof(*np));
    np->data = 3;
    STAILQ_INSERT_HEAD(&head, np, entries);

                                          /* اجتياز أمامي */
    STAILQ_FOREACH(np, &head, entries)
        printf("%i\n", np->data);
                                            /* حذف صف الذيل */
    n1 = STAILQ_FIRST(&head);
    while (n1 != NULL) {
        n2 = STAILQ_NEXT(n1, entries);
        free(n1);
        n1 = n2;
    }
    STAILQ_INIT(&head);
    exit(EXIT_SUCCESS);
}
