stat(3)

الحزمة: physfs

التراخيص: Zlib

الاسم

PHYSFS_Archiver - بنية بيانات لتعريف أرشيف للوصول إليه.

الملخص

#include <physfs.h>

typedef struct PHYSFS_Archiver {
    PHYSFS_uint32 version;
    int (*init) (void **opaque, const char *modulename);
    void (*destroy) (void *opaque);
    const char *(*getMountPoint) (void *opaque);
    int (*enumerate) (void *opaque, PHYSFS_EnumerateCallBack callback,
                      const char *dirname, void *callbackdata);
    PHYSFS_FileIO *(*openRead) (void *opaque, const char *fnm);
    PHYSFS_FileIO *(*openWrite) (void *opaque, const char *filename);
    int (*remove) (void *opaque, const char *filename);
    int (*stat) (void *opaque, const char *fn, PHYSFS_Stat *stat);
} PHYSfs_archiver;

الوصف المفصل

هذه البنية هي واجهة لأرشيفات بديلة. يتم استخدامها لتحديد كيفية تنفيذ عمليات على نوع معين من الأرشيفات. كل حقل هو مؤشر إلى وظيفة تنفذ عملية محددة. إذا لم يكن نوع أرشيف معين يدعم عملية ما، فيجب تعيين المؤشر المقابل إلى NULL.

يجب أن يقوم كل تطبيق لاستخدام أرشيف باستدعاء PHYSFS_registerArchiver() لتسجيل بنيته PHYSFS_Archiver. عند محاولة جبل (mount) أرشيف، ستقوم PhysFS بالمرور عبر قائمة الأرشيفات المسجلة، واستدعاء حقل init() لكل منها، ثم محاولة تحديد ما إذا كان يمكنها التعامل مع الأرشيف.

يجب أن تقوم الوظيفة init() بتخصيص أي بيانات خاصة بنوع الأرشيف، وتعيينها إلى `*opaque`. إذا فشلت، يجب أن تعيد 0، وتترك `*opaque` كما هو. إذا نجحت، يجب أن تعيد 1، وتترك `*opaque` يشير إلى البيانات المخصصة. يجب أن تكون قيمة `opaque` الأولى في PHYSFS_mount() إما NULL أو قيمة تم تمريرها سابقًا إلى init().

يجب أن تقوم الوظيفة destroy() بتحرير أي بيانات تم تخصيصها في init().

يجب أن تقوم الوظيفة getMountPoint() بإعادة سلسلة نصية تمثل نقطة جبل (mount point) الأرشيف. يجب ألا تكون السلسلة النصية مؤقتة؛ يجب أن تكون حرفيًا.

يجب أن تقوم الوظيفة enumerate() بإعادة قائمة بالملفات والدلائل داخل الدليل المحدد (dirname). يتم استدعاء `callback` لكل إدخال، مع تمرير `callbackdata` إليها.

يجب أن تقوم الوظيفة openRead() بفتح ملف للقراءة. يجب أن تعيد مؤشرًا إلى PHYSFS_Io على النجاح، أو NULL عند الفشل. يجب أن يتم تمرير المؤشر الذي تم إرجاعه كـ `opaque` لاستدعاءات الملفات اللاحقة.

يجب أن تقوم الوظيفة openWrite() بفتح ملف للكتابة. إذا لم يكن الملف موجودًا، فيجب إنشاؤه. إذا كان موجودًا، فيجب اقتطاعه إلى صفر بايت. يجب أن يكون إزاحة الكتابة في بداية الملف. إذا كان الأرشيف للقراءة فقط، فيجب أن تفشل هذه العملية. يجب أن تعيد مؤشرًا إلى PHYSFS_Io عند النجاح، أو NULL عند الفشل.

يجب أن تقوم الوظيفة remove() بحذف ملف أو دليل في الأرشيف. يتم استخدام نفس الاستدعاء لكل من الملفات والأدلة؛ لا توجد وظيفة منفصلة لإزالة الدليل (rmdir()). يُقصد فقط إزالة الأدلة إذا كانت فارغة. إذا كان الأرشيف للقراءة فقط، فيجب أن تفشل هذه العملية.

يجب أن تقوم الوظيفة stat() بالحصول على البيانات الوصفية الأساسية للملف. عند النجاح، يجب ملء جميع الحقول في (stat)، باستخدام القيم الافتراضية المعقولة للحقول التي تنطبق على الأرشيف الخاص بك.

يجب أن تعيد الوظيفة `init` قيمة غير صفرية إذا نجحت في التعامل مع البيانات، و 0 إذا لم تستطع. إذا تعاملت مع البيانات، فيجب تعيين الحقل `claimed` في بنية PHYSFS_MountInfo إلى 1. يجب تعيين هذا الحقل إلى 1 إذا كان هذا بالتأكيد أرشيفًا يمكن لتنفيذ الأرشفة الخاص بك التعامل معه، حتى لو فشل. نستخدمه لتحديد ما إذا كان يجب التوقف عن تجربة أرشفات أخرى إذا فشلت في فتحه. على سبيل المثال: سيقوم أرشيف .zip بتعيين هذا إلى 1 لشيء يحمل توقيع ملف .zip، حتى لو فشل لأنه تم اقتطاع الملف أيضًا. لا معنى لمحاولة أرشفات أخرى هنا، فقد حاولنا بالفعل التعامل معه بالتنفيذ المناسب!. إعادة NULL عند الفشل وتعيين (claimed) بشكل مناسب. إذا لم يفتح أي أرشيف الأرشيف أو قام بتعيين (claimed)، فستبلغ PHYSFS_mount() عن PHYSFS_ERR_UNSUPPORTED. بخلاف ذلك، ستبلغ عن الخطأ من الأرشيف الذي ادعى البيانات من خلال (claimed). إعادة قيمة غير NULL عند النجاح. سيتم تمرير المؤشر المُرجع كمعلمة "opaque" لاستدعاءات لاحقة.

PHYSFS_Io *(* PHYSFS_Archiver::openRead) (void *opaque, const char *fnm)

افتح ملفًا في هذا الأرشيف للقراءة. اسم الملف هذا، (fnm)، هو بتنسيق مستقل عن النظام الأساسي. فشل إذا لم يكن الملف موجودًا. تعيد NULL عند الفشل، وتستدعي PHYSFS_setErrorCode(). تعيد قيمة غير NULL عند النجاح. سيتم تمرير المؤشر المُرجع كمعلمة "opaque" لاستدعاءات الملفات اللاحقة.

PHYSFS_Io *(* PHYSFS_Archiver::openWrite) (void *opaque, const char *filename)

افتح ملفًا في هذا الأرشيف للكتابة. إذا لم يكن الملف موجودًا، فيجب إنشاؤه. إذا كان موجودًا، فيجب اقتطاعه إلى صفر بايت. يجب أن تكون إزاحة الكتابة في بداية الملف. إذا كان الأرشيف للقراءة فقط، فيجب أن تفشل هذه العملية. اسم الملف هذا هو بتنسيق مستقل عن النظام الأساسي. تعيد NULL عند الفشل، وتستدعي PHYSFS_setErrorCode(). تعيد قيمة غير NULL عند النجاح. سيتم تمرير المؤشر المُرجع كمعلمة "opaque" لاستدعاءات الملفات اللاحقة.

int(* PHYSFS_Archiver::remove) (void *opaque, const char *filename)

احذف ملفًا أو دليلًا في الأرشيف. يتم استخدام هذا الاستدعاء نفسه لكل من الملفات والأدلة؛ لا توجد وظيفة منفصلة لإزالة الدليل (rmdir()). يُقصد فقط إزالة الأدلة إذا كانت فارغة. إذا كان الأرشيف للقراءة فقط، فيجب أن تفشل هذه العملية.

أعد قيمة غير صفرية عند النجاح، وصفرًا عند الفشل. اسم الملف هذا هو بتنسيق مستقل عن النظام الأساسي. عند الفشل، استدعِ PHYSFS_setErrorCode().

int(* PHYSFS_Archiver::stat) (void *opaque, const char *fn, PHYSFS_Stat *stat)

احصل على البيانات الوصفية الأساسية للملف. عند النجاح، املأ جميع الحقول في (stat)، باستخدام القيم الافتراضية المعقولة للحقول التي تنطبق على الأرشيف الخاص بك.

أعد قيمة غير صفرية عند النجاح، وصفرًا عند الفشل. اسم الملف هذا هو بتنسيق مستقل عن النظام الأساسي. عند الفشل، استدعِ PHYSFS_setErrorCode().

PHYSFS_uint32 PHYSFS_Archiver::version

معلومات التوافق الثنائي. يجب تعيين هذا إلى صفر في هذا الوقت. ستزيد الإصدارات المستقبلية من هذه البنية هذا الحقل، لذلك نعرف ما يدعمه تنفيذ معين. من المحتمل أن نستمر في دعم الإصدارات الأقدم أثناء تقديم ميزات جديدة.
