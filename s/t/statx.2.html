الاسم

statx - الحصول على حالة الملف (موسعة)

المكتبة

مكتبة C القياسية (libc، -lc)

الملخص

#define _GNU_SOURCE          /* انظر feature_test_macros(7) */
#include <fcntl.h>           /* تعريف ثوابت AT_* */
#include <sys/stat.h>
int statx(int dirfd, const char *_Nullable restrict path,
          int flags, unsigned int mask,
          struct statx *restrict statxbuf);

الوصف

تقوم هذه الدالة بإرجاع معلومات حول ملف، وتخزينها في المخزن المؤقت الذي يشير إليه statxbuf. المخزن المؤقت الذي تم إرجاعه هو هيكل من النوع التالي:


struct statx {
    __u32 stx_mask;        /* قناع البتات التي تشير إلى
                              الحقول المكتملة */
    __u32 stx_blksize;     /* حجم الكتلة لعمليات I/O لنظام الملفات */
    __u64 stx_attributes;  /* مؤشرات سمات الملف الإضافية */
    __u32 stx_nlink;       /* عدد الروابط الصلبة */
    __u32 stx_uid;         /* معرف المستخدم للمالك */
    __u32 stx_gid;         /* معرف المجموعة للمالك */
    __u16 stx_mode;        /* نوع الملف ووضعه */
    __u64 stx_ino;         /* رقم inode */
    __u64 stx_size;        /* الحجم الكلي بالبايت */
    __u64 stx_blocks;      /* عدد الكتل المخصصة بحجم 512 بايت */
    __u64 stx_attributes_mask;
                           /* قناع لإظهار ما هو مدعوم
                              في stx_attributes */
    /* الحقول التالية هي طوابع زمن الملف */
    struct statx_timestamp stx_atime;  /* آخر وصول */
    struct statx_timestamp stx_btime;  /* الإنشاء */
    struct statx_timestamp stx_ctime;  /* آخر تغيير للحالة */
    struct statx_timestamp stx_mtime;  /* آخر تعديل */
    /* إذا كان هذا الملف يمثل جهازًا، فإن الحقلين التاليين
       يحتويان على معرف الجهاز */
    __u32 stx_rdev_major;  /* المعرف الرئيسي */
    __u32 stx_rdev_minor;  /* المعرف الثانوي */
    /* الحقلين التاليين يحتويان على معرف الجهاز
       الذي يحتوي على نظام الملفات الذي يوجد به الملف */
    __u32 stx_dev_major;   /* المعرف الرئيسي */
    __u32 stx_dev_minor;   /* المعرف الثانوي */
    __u64 stx_mnt_id;      /* معرف نقطة التركيب */
    /* قيود محاذاة I/O المباشر */
    __u32 stx_dio_mem_align;
    __u32 stx_dio_offset_align;
    __u64 stx_subvol;      /* معرف الحجم الفرعي */
    /* حدود الكتابة الذرية لـ I/O المباشر */
    __u32 stx_atomic_write_unit_min;
    __u32 stx_atomic_write_unit_max;
    __u32 stx_atomic_write_segments_max;
    /* محاذاة إزاحة الملف لقراءات I/O المباشر */
    __u32 stx_dio_read_offset_align;
    /* الحد الأقصى لحد الكتابة الذرية لـ I/O المباشر */
    __u32 stx_atomic_write_unit_max_opt;
};

طوابع زمن الملف هي هياكل من النوع التالي:


struct statx_timestamp {
    __s64 tv_sec;    /* الثواني منذ العصر (وقت يونكس) */
    __u32 tv_nsec;   /* نانو ثواني منذ tv_sec */
};

(ملاحظة: تم حذف المساحة المحجوزة والحشو.)

استدعاء statx():

للوصول إلى حالة الملف، لا يلزم وجود أذونات على الملف نفسه، ولكن في حالة statx() مع مسار، يلزم وجود إذن التنفيذ (البحث) على جميع الدلائل في path التي تؤدي إلى الملف.

تستخدم statx() path و dirfd و flags لتحديد الملف الهدف بإحدى الطرق التالية:

مسار مطلق
إذا بدأ path بشرطة مائلة، فهو مسار مطلق يحدد الملف الهدف. في هذه الحالة، يتم تجاهل dirfd.
مسار نسبي
إذا كان path سلسلة تبدأ بحرف غير الشرطة المائلة و dirfd هو AT_FDCWD، فإن path هو مسار نسبي يتم تفسيره بالنسبة لدليل العمل الحالي للعملية.
مسار نسبي للدليل
إذا كان path سلسلة تبدأ بحرف غير الشرطة المائلة و dirfd هو واصف ملف يشير إلى دليل، فإن path هو مسار نسبي يتم تفسيره بالنسبة للدليل الذي يشير إليه dirfd. (انظر openat(2) لشرح سبب فائدة هذا.)
بواسطة واصف ملف
إذا كان path سلسلة فارغة (أو NULL منذ Linux 6.11) وتم تحديد الراية AT_EMPTY_PATH في flags (انظر أدناه)، فإن الملف الهدف هو الملف الذي يشير إليه واصف الملف dirfd.

يمكن استخدام flags للتأثير على البحث المستند إلى المسار. يتم بناء قيمة لـ flags عن طريق ORing صفرًا أو أكثر من الثوابت التالية:

AT_EMPTY_PATH
إذا كان path سلسلة فارغة (أو NULL منذ Linux 6.11)، قم بالعمل على الملف الذي يشير إليه dirfd (الذي ربما تم الحصول عليه باستخدام راية open(2) O_PATH). في هذه الحالة، يمكن أن يشير dirfd إلى أي نوع من الملفات، وليس مجرد دليل.
إذا كان dirfd هو AT_FDCWD، فإن الاستدعاء يعمل على دليل العمل الحالي.
AT_NO_AUTOMOUNT
لا تقم بتركيب نقطة التركيب النهائية ("basename") تلقائيًا إذا كان دليلًا لنقطة تركيب تلقائي. هذا يسمح للمستدعي بجمع سمات نقطة التركيب التلقائي (بدلاً من الموقع الذي سيتم تركيبه). ليس لهذه الراية أي تأثير إذا تم تركيب نقطة التركيب بالفعل.
يمكن استخدام الراية AT_NO_AUTOMOUNT في الأدوات التي تقوم بمسح الأدلة لمنع التركيب التلقائي الجماعي للأدلة التي تحتوي على نقاط تركيب تلقائي.
جميع من stat(2) و lstat(2) و fstatat(2) تعمل كما لو تم تعيين AT_NO_AUTOMOUNT.
AT_SYMLINK_NOFOLLOW
إذا كان path رابطًا رمزيًا، فلا تقم بفك الإشارة إليه: بدلاً من ذلك، قم بإرجاع معلومات حول الرابط نفسه، مثل lstat(2).

يمكن أيضًا استخدام flags للتحكم في نوع المزامنة التي سيقوم بها النواة عند الاستعلام عن ملف في نظام ملفات عن بعد. يتم ذلك عن طريق ORing بإحدى القيم التالية:

AT_STATX_SYNC_AS_STAT
قم بما يقوم به stat(2). هذا هو الافتراضي وهو يعتمد بشكل كبير على نظام الملفات.
AT_STATX_FORCE_SYNC
فرض مزامنة السمات مع الخادم. قد يتطلب هذا أن يقوم نظام ملفات الشبكة بإرجاع البيانات لكتابتها للتأكد من صحة الطوابع الزمنية.
AT_STATX_DONT_SYNC
لا تقم بمزامنة أي شيء، بل خذ فقط ما هو موجود في ذاكرة التخزين المؤقت للنظام إن أمكن. قد يعني هذا أن المعلومات التي تم إرجاعها تقريبية، ولكن، على نظام ملفات شبكة، قد لا تتضمن رحلة ذهاب وعودة إلى الخادم - حتى لو لم يكن هناك تأجير.

يستخدم الوسيط mask لـ statx() لإخبار النواة بالحقول التي يهتم بها المستدعي. mask هو مزيج ORed من الثوابت التالية:


STATX_TYPE طلب stx_mode & S_IFMT
STATX_MODE طلب stx_mode & ~S_IFMT
STATX_NLINK طلب stx_nlink
STATX_UID طلب stx_uid
STATX_GID طلب stx_gid
STATX_ATIME طلب stx_atime
STATX_MTIME طلب stx_mtime
STATX_CTIME طلب stx_ctime
STATX_INO طلب stx_ino
STATX_SIZE طلب stx_size
STATX_BLOCKS طلب stx_blocks
STATX_BASIC_STATS [كل ما سبق]
STATX_BTIME طلب stx_btime

هذا الموقع غير مرتبط بـ Arch Linux ولا تابع له بأي شكل من الأشكال. المحتوى هو ترجمة فقط لصفحات دليل Arch المنشورة على man.archlinux.org.

STATX_ALL نفس STATX_BASIC_STATS | STATX_BTIME. تم إهمالها ويجب عدم استخدامها. STATX_MNT_ID طلب stx_mnt_id (منذ Linux 5.8) STATX_DIOALIGN طلب stx_dio_mem_align و stx_dio_offset_align. (منذ Linux 6.1؛ الدعم يختلف حسب نظام الملفات) STATX_MNT_ID_UNIQUE طلب stx_mnt_id فريد (منذ Linux 6.8) STATX_SUBVOL طلب stx_subvol (منذ Linux 6.10؛ الدعم يختلف حسب نظام الملفات) STATX_WRITE_ATOMIC طلب stx_atomic_write_unit_min، stx_atomic_write_unit_max، stx_atomic_write_segments_max، و stx_atomic_write_unit_max_opt. (منذ Linux 6.11؛ الدعم يختلف حسب نظام الملفات) STATX_DIO_READ_ALIGN طلب stx_dio_read_offset_align. (منذ Linux 6.14؛ الدعم يختلف حسب نظام الملفات)

لاحظ أنه بشكل عام، لا يرفض النواة القيم في mask بخلاف ما تم ذكره أعلاه. (للحصول على استثناء، انظر EINVAL في الأخطاء.) بدلاً من ذلك، تقوم ببساطة بإعلام المتصل بالقيم التي تدعمها هذه النواة ونظام الملفات عبر الحقل statx.stx_mask. لذلك، لا تقم ببساطة بتعيين mask إلى UINT_MAX (جميع البتات محددة)، حيث قد يتم استخدام بت واحد أو أكثر في المستقبل لتحديد امتداد للمخزن المؤقت.

المعلومات المعادة

يتم إرجاع معلومات الحالة للملف المستهدف في البنية statx التي يشير إليها statxbuf. يتضمن ذلك stx_mask الذي يشير إلى المعلومات الأخرى التي تم إرجاعها. stx_mask له نفس تنسيق الوسيط mask ويتم تعيين البتات فيه للإشارة إلى الحقول التي تم ملؤها.

يجب ملاحظة أن النواة قد تعيد حقولًا لم يتم طلبها وقد تفشل في إعادة حقول تم طلبها، اعتمادًا على ما يدعمه نظام الملفات الأساسي. (يمكن تجاهل الحقول التي تحصل على قيم على الرغم من عدم طلبها). في كلتا الحالتين، لن يكون stx_mask مساويًا لـ mask.

إذا لم يدعم نظام ملفات حقلاً ما أو إذا كان لديه قيمة غير قابلة للتمثيل (على سبيل المثال، ملف بنوع غريب)، فسيتم مسح بت القناع المقابل لذلك الحقل في stx_mask حتى لو طلبه المستخدم وسيتم ملء قيمة وهمية للتوافق إذا كانت متاحة (على سبيل المثال، قد يتم تحديد معرف مستخدم (UID) ومعرف مجموعة (GID) وهمي تحت بعض الظروف).

قد يقوم نظام الملفات أيضًا بملء حقول لم يطلبها المستدعي إذا كان لديه قيم متاحة لها ولم تكن المعلومات متاحة بتكلفة إضافية. إذا حدث هذا، فسيتم تعيين البتات المقابلة في stx_mask.

ملاحظة: لأسباب تتعلق بالأداء والبساطة، قد تحتوي حقول مختلفة في البنية statx على معلومات حالة من لحظات مختلفة أثناء تنفيذ استدعاء النظام. على سبيل المثال، إذا تم تغيير stx_mode أو stx_uid بواسطة عملية أخرى عن طريق استدعاء chmod(2) أو chown(2)، فقد تعيد stat() قيمة stx_mode القديمة مع stx_uid الجديدة، أو stx_uid القديمة مع stx_mode الجديدة.

بصرف النظر عن stx_mask (الموصوف أعلاه)، فإن الحقول في البنية statx هي:

stx_blksize
حجم الكتلة "المفضل" لإدخال/إخراج فعال لنظام الملفات. (الكتابة إلى ملف بأجزاء أصغر قد تسبب إعادة قراءة-تعديل-إعادة كتابة غير فعال.)
stx_attributes
معلومات حالة إضافية للملف (انظر أدناه لمزيد من المعلومات).
stx_nlink
عدد الروابط الصلبة للملف.
stx_uid
يحتوي هذا الحقل على معرف المستخدم (UID) لمالك الملف.
stx_gid
يحتوي هذا الحقل على معرف المجموعة (GID) لمالك الملف.
stx_mode
نوع الملف ووضعه. انظر inode(7) للتفاصيل.
stx_ino
رقم inode للملف.
stx_size
حجم الملف (إذا كان ملفًا عاديًا أو رابطًا رمزيًا) بالبايت. حجم الرابط الرمزي هو طول المسار الذي يحتويه، بدون حرف إنهاء صفري.
stx_blocks
عدد الكتل المخصصة للملف على الوسيط، بوحدات 512 بايت. (قد يكون هذا أصغر من stx_size/512 عندما يحتوي الملف على ثقوب.)
stx_attributes_mask
قناع يشير إلى البتات المدعومة في stx_attributes من قبل VFS ونظام الملفات.
stx_atime
الطابع الزمني لآخر وصول للملف.
stx_btime
الطابع الزمني لإنشاء الملف.
stx_ctime
الطابع الزمني لآخر تغيير لحالة الملف.
stx_mtime
الطابع الزمني لآخر تعديل للملف.
stx_dev_major
stx_dev_minor
الجهاز الذي يوجد عليه هذا الملف (inode).
stx_rdev_major
stx_rdev_minor
الجهاز الذي يمثله هذا الملف (inode) إذا كان الملف من نوع جهاز كتلة أو جهاز حرفي.
stx_mnt_id
إذا كنت تستخدم STATX_MNT_ID، فهذا هو معرف التركيب (mount ID) للتركيب الذي يحتوي على الملف. هذا هو نفس الرقم الذي يتم الإبلاغ عنه بواسطة name_to_handle_at(2) ويتوافق مع الرقم في الحقل الأول في أحد السجلات في /proc/self/mountinfo.
إذا كنت تستخدم STATX_MNT_ID_UNIQUE، فهذا هو معرف التركيب الفريد للتركيب الذي يحتوي على الملف. هذا هو الرقم الذي يتم الإبلاغ عنه بواسطة listmount(2) وهو المعرف المستخدم للاستعلام عن التركيب باستخدام statmount(2). ومن المضمون عدم إعادة استخدامه أثناء تشغيل النظام.
stx_dio_mem_align
المحاذاة (بالبايت) المطلوبة لمخازن الذاكرة المؤقتة للمستخدم للإدخال/الإخراج المباشر (O_DIRECT) على هذا الملف، أو 0 إذا لم يكن الإدخال/الإخراج المباشر مدعومًا على هذا الملف.
STATX_DIOALIGN (stx_dio_mem_align و stx_dio_offset_align) مدعوم على أجهزة الكتل منذ Linux 6.1. الدعم على الملفات العادية يختلف حسب نظام الملفات؛ تدعمه ext4 و f2fs و xfs منذ Linux 6.1.
stx_dio_offset_align
المحاذاة (بالبايت) المطلوبة لإزاحات الملفات وأطوال مقاطع الإدخال/الإخراج للإدخال/الإخراج المباشر (O_DIRECT) على هذا الملف، أو 0 إذا لم يكن الإدخال/الإخراج المباشر مدعومًا على هذا الملف. سيكون هذا غير صفري فقط إذا كان stx_dio_mem_align غير صفري، والعكس صحيح.
stx_dio_read_offset_align
المحاذاة (بالبايت) المطلوبة لإزاحات الملفات وأطوال مقاطع الإدخال/الإخراج لقراءات الإدخال/الإخراج المباشر (O_DIRECT) على هذا الملف. إذا كانت صفرًا، فإن الحد في stx_dio_offset_align ينطبق على القراءات أيضًا. إذا كانت غير صفرية، فيجب أن تكون هذه القيمة أصغر من أو تساوي stx_dio_offset_align ويجب أن يوفرها نظام الملفات إذا طلبها التطبيق. لا تتأثر محاذاة الذاكرة في stx_dio_mem_align بهذه القيمة.
STATX_DIO_READ_ALIGN (stx_dio_offset_align) مدعوم من قبل xfs على الملفات العادية منذ Linux 6.14.
stx_subvol
رقم الحجم الفرعي للملف الحالي.
الأحجام الفرعية هي أدلة متطورة، أي أنها تشكل هيكل شجرة يمكن المشي عبره بشكل متكرر. يختلف الدعم حسب نظام الملفات؛ تدعمه bcachefs و btrfs منذ Linux 6.10.
stx_atomic_write_unit_min
stx_atomic_write_unit_max
الحد الأدنى والأقصى للأحجام (بالبايت) المدعومة للإدخال/الإخراج المباشر (O_DIRECT) على الملف ليتم كتابته بحماية من الكتابة الممزقة. يضمن أن تكون هذه القيم كلتاهما قوة للعدد 2.
STATX_WRITE_ATOMIC (stx_atomic_write_unit_min، stx_atomic_write_unit_max، و stx_atomic_write_segments_max) مدعوم على أجهزة الكتل منذ Linux 6.11. يختلف الدعم على الملفات العادية حسب نظام الملفات؛ تدعمه xfs و ext4 منذ Linux 6.13.
stx_atomic_write_unit_max_opt
الحد الأقصى للحجم (بالبايت) المحسن للكتابات الصادرة مع حماية من الكتابة الممزقة. إذا كانت غير صفرية، فلن تتجاوز هذه القيمة القيمة في stx_atomic_write_unit_max ولن تقل عن القيمة في stx_atomic_write_unit_min. تشير القيمة صفر إلى أن stx_atomic_write_unit_max هو الحد الأمثل. قد يتم تجربة كتابات أبطأ عندما يتجاوز حجم الكتابة stx_atomic_write_unit_max_opt (عندما تكون غير صفرية).
stx_atomic_write_segments_max

statx(2)

get file status

LIBRARY

Linux API (libc, version 2.28)

SYNOPSIS

#include <sys/stat.h>
int statx(int dirfd, const char *pathname, int flags,
          unsigned int mask, struct statx *statxbuf);

DESCRIPTION

The statx() system call is an extension of stat(2). It provides a more flexible and extensible interface for retrieving file and filesystem information.

The call takes a path, specified by the combination of dirfd and pathname, and returns file information in the structure pointed to by statxbuf. The flags argument controls the path resolution, and the mask argument specifies which fields of the statxbuf structure the caller is interested in.

If pathname is not an absolute path, it is interpreted relative to the directory referred to by the file descriptor dirfd. If pathname is absolute, then dirfd is ignored. If pathname is NULL and flags contains AT_EMPTY_PATH, then a reference to the file described by dirfd is used.

The flags argument can be 0 or include any of the following (in addition to flags used for path resolution, see openat(2)):

AT_EMPTY_PATH
If pathname is an empty string or NULL, a reference to the file described by dirfd is used. This flag is intended to allow operations on the file described by dirfd without requiring it to be a directory (e.g., used with fstatat()). It is also used to allow statx() to be called on a file descriptor representing a mount point.
AT_NO_AUTOMNT
Do not follow a automount point by transitioning into the subdirectory and then performing the operation on the mounted file system.
AT_SYMLINK_NOFOLLOW
Do not follow symbolic links. If pathname is a symbolic link, then return information about the link itself rather than the file it references.

The mask argument is a bitmask specifying which fields of the struct statx structure should be filled in. If a bit is set in mask, the corresponding field in statxbuf will be filled in by the kernel. If the bit is not set, the value in the statxbuf field is undefined. The available bits are defined as constants in <sys/stat.h> and begin with STATX_.

The struct statx structure is defined as follows:

struct statx {
    unsigned int     stx_mask;     /* Which fields are valid */
    unsigned int     stx_attributes; /* File attributes      */
    /* ... other fields ... */
};

The fields available for querying via the mask argument are:

STATX_BASIC_STATS
The following fields are valid: stx_mode, stx_nlink, stx_uid, stx_gid, stx_filetype, stx_size, stx_atime, stx_mtime, stx_ctime.
STATX_FORCE_SYNC
Update times before returning. This flag is intended to provide consistency with stat(2) and futimes(3) when dealing with filesystems that might delay updates. Use with caution, as it can impact performance.
STATX_USER_RESERVED
User-space can use this mask to specify flags that are not known to the kernel. This can be useful for forward compatibility.
STATX_TYPE
stx_mode (file type and mode bits), stx_nlink (number of hard links).
STATX_MODE
stx_mode (permission bits).
STATX_NLINK
stx_nlink (number of hard links).
STATX_INTERNAL_ALL
Includes all available fields, including internal filesystem-specific information.
STATX_UID
stx_uid (user ID of owner).
STATX_GID
stx_gid (group ID of owner).
STATX_ATIME
stx_atime (time of last access).
STATX_MTIME
stx_mtime (time of last modification).
STATX_CTIME
stx_ctime (time of last status change).
STATX_SIZE
stx_size (total size, in bytes).
STATX_BLOCKS
stx_blocks (number of 512-byte blocks allocated).
STATX_BTIME
stx_btime (time of last birth/creation).
STATX_RDEV
stx_rdev (ID of device node).
STATX_ALL
The union of all the above flags, excluding any internal flags.

The struct statx structure contains the following fields:

unsigned int stx_mask
This field reflects the mask argument passed to statx(), indicating which fields in struct statx are valid. It can also be used to indicate unsupported attributes (see stx_attributes).
unsigned int stx_attributes
This field contains a bitmask of file attributes. The interpretation of these bits depends on the underlying filesystem. See the File attributes section below.
uint64_t stx_ino
Inode number.
uint32_t stx_uid
User ID of owner.
uint32_t stx_gid
Group ID of owner.
uint16_t stx_mode
File type and mode bits (permissions).
uint16_t stx_nlink
Number of hard links.
uint64_t stx_size
Total size, in bytes.
uint64_t stx_blocks
Number of blocks allocated. These are typically filesystem block sizes, which may not be 512 bytes.
uint32_t stx_attributes_mask
Mask of supported file attributes. This field indicates which bits in stx_attributes are supported by the filesystem. Any attribute bit not set in this mask should be ignored in stx_attributes.
uint64_t stx_nbytes
The number of bytes that can be written in a single atomic operation. This is relevant for filesystems that support atomic writes. A value of 0 indicates that atomic writes are not supported or that the size is unknown.
struct timespec64 stx_atime
Time of last access.
struct timespec64 stx_mtime
Time of last modification.
struct timespec64 stx_ctime
Time of last status change.
struct timespec64 stx_btime
Time of last birth/creation.
uint32_t stx_dev_major
Major device number (for device files).
uint32_t stx_dev_minor
Minor device number (for device files).
uint32_t stx_rdev_major
Major device number (for device files, if different from stx_dev_major).
uint32_t stx_rdev_minor
Minor device number (for device files, if different from stx_dev_minor).
uint32_t stx_mnt_id
Filesystem mount ID.
uint64_t stx_realino
The real inode number, which may differ from stx_ino for special file types like hard links or bind mounts.

File attributes

The stx_attributes field contains a set of ORed flags that indicate additional attributes of the file. Note that any attribute that is not indicated as supported by stx_attributes_mask has no usable value here. The bits in stx_attributes_mask correspond bit-by-bit to stx_attributes.

The flags are as follows:

STATX_ATTR_COMPRESSED
The file is compressed by the filesystem and may take extra resources to access.
STATX_ATTR_IMMUTABLE
The file cannot be modified: it cannot be deleted or renamed, no hard links can be created to this file and no data can be written to it. See chattr(1).
STATX_ATTR_APPEND
The file can only be opened in append mode for writing. Random access writing is not permitted. See chattr(1).
STATX_ATTR_NODUMP
File is not a candidate for backup when a backup program such as dump(8) is run. See chattr(1).
STATX_ATTR_ENCRYPTED
A key is required for the file to be encrypted by the filesystem.
STATX_ATTR_VERITY (since Linux 5.5)
The file has fs-verity enabled. It cannot be written to, and all reads from it will be verified against a cryptographic hash that covers the entire file (e.g., via a Merkle tree).
STATX_ATTR_WRITE_ATOMIC (since Linux 6.11)
The file supports torn-write protection.
STATX_ATTR_DAX (since Linux 5.8)
The file is in the DAX (cpu direct access) state. DAX state attempts to minimize software cache effects for both I/O and memory mappings of this file. It requires a file system which has been configured to support DAX.
DAX generally assumes all accesses are via CPU load / store instructions which can minimize overhead for small accesses, but may adversely affect CPU utilization for large transfers.
File I/O is done directly to/from user-space buffers and memory mapped I/O may be performed with direct memory mappings that bypass the kernel page cache.
While the DAX property tends to result in data being transferred synchronously, it does not give the same guarantees as the O_SYNC flag (see open(2)), where data and the necessary metadata are transferred together.
A DAX file may support being mapped with the MAP_SYNC flag, which enables a program to use CPU cache flush instructions to persist CPU store operations without an explicit fsync(2). See mmap(2) for more information.
STATX_ATTR_MOUNT_ROOT (since Linux 5.8)
The file is the root of a mount.

RETURN VALUE

On success, zero is returned. On error, -1 is returned, and errno is set to indicate the error.

ERRORS

EACCES
Search permission is denied for one of the directories in the path prefix of path. (See also path_resolution(7).)
EBADF
path is relative but dirfd is neither AT_FDCWD nor a valid file descriptor.
EFAULT
path or statxbuf points to a location outside the process's accessible address space or is NULL (except since Linux 6.11 if AT_EMPTY_PATH is specified in flags, path is allowed to be NULL).
EINVAL
Invalid flag specified in flags.
EINVAL
Reserved flag specified in mask. (Currently, there is one such flag, designated by the constant STATX__RESERVED, with the value 0x80000000U.)
ELOOP
Too many symbolic links encountered while traversing the pathname.
ENAMETOOLONG
path is too long.
ENOENT
A component of path does not exist, or path is an empty string and AT_EMPTY_PATH was not specified in flags.
ENOMEM
Out of memory (i.e., kernel memory).
ENOTDIR
A component of the path prefix of path is not a directory or path is relative and dirfd is a file descriptor referring to a file other than a directory.

STANDARDS

Linux.

statx(2)

الحصول على حالة الملف

المكتبة

واجهة برمجة تطبيقات لينكس (libc، الإصدار 2.28)

الملخص

#include <sys/stat.h>
int statx(int dirfd, const char *pathname, int flags,
          unsigned int mask, struct statx *statxbuf);

الوصف

statx() هو استدعاء نظام يمثل امتدادًا لـ stat(2). يوفر واجهة أكثر مرونة وقابلية للتوسيع لاسترداد معلومات الملف ونظام الملفات.

يقوم الاستدعاء بأخذ مسار، يتم تحديده بواسطة الجمع بين dirfd و pathname، ويعيد معلومات الملف في البنية المشار إليها بواسطة statxbuf. يتحكم الوسيط flags في تحليل المسار، ويحدد الوسيط mask أي حقول في بنية statxbuf مهتم بها المتصل. إذا لم يكن pathname مسارًا مطلقًا، فسيتم تفسيره بالنسبة للدليل المشار إليه بواسطة واصف الملف dirfd. إذا كان pathname مطلقًا، فسيتم تجاهل dirfd. إذا كان pathname فارغًا (NULL) وكان flags يحتوي على AT_EMPTY_PATH، فسيتم استخدام مرجع للملف الموصوف بواسطة dirfd.

يمكن أن يكون الوسيط flags 0 أو يتضمن أيًا مما يلي (بالإضافة إلى الأعلام المستخدمة لتحليل المسار، انظر openat(2)):

AT_EMPTY_PATH
إذا كان pathname سلسلة فارغة أو NULL، فسيتم استخدام مرجع للملف الموصوف بواسطة dirfd. هذا العلم مخصص للسماح بالعمليات على الملف الموصوف بواسطة dirfd دون الحاجة إلى أن يكون دليلًا (على سبيل المثال، يستخدم مع fstatat()). كما أنه يستخدم للسماح باستدعاء statx() على واصف ملف يمثل نقطة تثبيت.
AT_NO_AUTOMNT
لا تتبع نقطة تثبيت تلقائية عن طريق الانتقال إلى الدليل الفرعي ثم إجراء العملية على نظام الملفات المثبت.
AT_SYMLINK_NOFOLLOW
لا تتبع الروابط الرمزية. إذا كان pathname رابطًا رمزيًا، فأعد المعلومات المتعلقة بالرابط نفسه بدلاً من الملف الذي يشير إليه.

الوسيط mask هو قناع بت يحدد الحقول في بنية struct statx التي يجب ملؤها. إذا تم تعيين بت في mask، فسيتم ملء الحقل المقابل في statxbuf بواسطة النواة. إذا لم يتم تعيين البت، فإن القيمة الموجودة في حقل statxbuf غير محددة. يتم تعريف البتات المتاحة كثوابت في <sys/stat.h> وتبدأ بـ STATX_.

تحتوي بنية struct statx على التعريف التالي:

struct statx {
    unsigned int     stx_mask;     /* أي الحقول صالحة */
    unsigned int     stx_attributes; /* سمات الملف      */
    /* ... حقول أخرى ... */
};

الحقول المتاحة للاستعلام عبر الوسيط mask هي:

STATX_BASIC_STATS
الحقول التالية صالحة: stx_mode، stx_nlink، stx_uid، stx_gid، stx_filetype، stx_size، stx_atime، stx_mtime، stx_ctime.
STATX_FORCE_SYNC
تحديث الأوقات قبل الإرجاع. هذا العلم مخصص لتوفير الاتساق مع stat(2) و futimes(3) عند التعامل مع أنظمة الملفات التي قد تؤخر التحديثات. استخدمه بحذر، لأنه قد يؤثر على الأداء.
STATX_USER_RESERVED
يمكن لمساحة المستخدم استخدام هذا القناع لتحديد الأعلام غير المعروفة للنواة. يمكن أن يكون هذا مفيدًا للتوافق المستقبلي.
STATX_TYPE
stx_mode (نوع الملف وبتات الوضع)، stx_nlink (عدد الروابط الصلبة).
STATX_MODE
stx_mode (بتات الأذونات).
STATX_NLINK
stx_nlink (عدد الروابط الصلبة).
STATX_INTERNAL_ALL
يتضمن جميع الحقول المتاحة، بما في ذلك المعلومات الداخلية الخاصة بنظام الملفات.
STATX_UID
stx_uid (معرف المستخدم للمالك).
STATX_GID
stx_gid (معرف المجموعة للمالك).
STATX_ATIME
stx_atime (وقت آخر وصول).
STATX_MTIME
stx_mtime (وقت آخر تعديل).
STATX_CTIME
stx_ctime (وقت آخر تغيير في الحالة).
STATX_SIZE
stx_size (الحجم الإجمالي، بالبايت).
STATX_BLOCKS
stx_blocks (عدد الكتل بحجم 512 بايت المخصصة).
STATX_BTIME
stx_btime (وقت آخر ولادة/إنشاء).
STATX_RDEV
stx_rdev (معرف عقدة الجهاز).
STATX_ALL
اتحاد جميع الأعلام المذكورة أعلاه، باستثناء أي أعلام داخلية.

تحتوي بنية struct statx على الحقول التالية:

unsigned int stx_mask
يعكس هذا الحقل الوسيط mask الذي تم تمريره إلى statx()، مشيرًا إلى الحقول الصالحة في struct statx. يمكن استخدامه أيضًا للإشارة إلى السمات غير المدعومة (انظر stx_attributes).
unsigned int stx_attributes
يحتوي هذا الحقل على قناع بت لسمات الملف. تعتمد تفسيرات هذه البتات على نظام الملفات الأساسي. انظر قسم سمات الملف أدناه.
uint64_t stx_ino
رقم i-node.
uint32_t stx_uid
معرف المستخدم للمالك.
uint32_t stx_gid
معرف المجموعة للمالك.
uint16_t stx_mode
نوع الملف وبتات الوضع (الأذونات).
uint16_t stx_nlink
عدد الروابط الصلبة.
uint64_t stx_size
الحجم الإجمالي، بالبايت.
uint64_t stx_blocks
عدد الكتل المخصصة. هذه عادةً أحجام كتل نظام الملفات، والتي قد لا تكون 512 بايت.
uint32_t stx_attributes_mask
قناع السمات المدعومة للملف. يشير هذا الحقل إلى أي البتات في stx_attributes مدعومة من قبل نظام الملفات. يجب تجاهل أي بت سمة غير معين في هذا القناع في stx_attributes.
uint64_t stx_nbytes

uint32_t stx_rdev_minor

سمات الملف

يحتوي الحقل stx_attributes على مجموعة من الأعلام المدمجة (ORed) التي تشير إلى سمات إضافية للملف. لاحظ أن أي سمة غير مشار إليها كمدعومة بواسطة stx_attributes_mask ليس لها قيمة قابلة للاستخدام هنا. البتات في stx_attributes_mask تتوافق بتًا بتًا مع stx_attributes.

الأعلام هي كما يلي:

STATX_ATTR_COMPRESSED
الملف مضغوط بواسطة نظام الملفات وقد يتطلب موارد إضافية للوصول إليه.
STATX_ATTR_IMMUTABLE
لا يمكن تعديل الملف: لا يمكن حذفه أو إعادة تسميته، ولا يمكن إنشاء روابط صلبة إليه، ولا يمكن كتابة بيانات عليه. انظر chattr(1).
STATX_ATTR_APPEND
يمكن فتح الملف في وضع الإلحاق للكتابة فقط. الكتابة العشوائية غير مسموح بها. انظر chattr(1).
STATX_ATTR_NODUMP
الملف ليس مرشحًا للنسخ الاحتياطي عند تشغيل برنامج نسخ احتياطي مثل dump(8). انظر chattr(1).
STATX_ATTR_ENCRYPTED
مفتاح مطلوب لتشفير الملف بواسطة نظام الملفات.
STATX_ATTR_VERITY (منذ لينكس 5.5)
تم تمكين fs-verity للملف. لا يمكن الكتابة إليه، وسيتم التحقق من جميع عمليات القراءة منه مقابل تجزئة تشفيرية تغطي الملف بالكامل (على سبيل المثال، عبر شجرة ميركل).
STATX_ATTR_WRITE_ATOMIC (منذ لينكس 6.11)
يدعم الملف حماية الكتابة المتقطعة.
STATX_ATTR_DAX (منذ لينكس 5.8)
الملف في حالة DAX (وصول مباشر لوحدة المعالجة المركزية). تحاول حالة DAX تقليل تأثيرات ذاكرة التخزين المؤقت للبرنامج لكل من عمليات الإدخال / الإخراج وربطات الذاكرة لهذا الملف. يتطلب نظام ملفات تم تكوينه لدعم DAX.
تفترض DAX عمومًا أن جميع عمليات الوصول تتم عبر تعليمات تحميل / تخزين لوحدة المعالجة المركزية والتي يمكن أن تقلل الحمل الزائد للوصول الصغير، ولكن قد تؤثر سلبًا على استخدام وحدة المعالجة المركزية لنقل البيانات الكبيرة.
تتم عمليات الإدخال / الإخراج للملف مباشرة من / إلى مخازن المستخدم المؤقتة، وقد تتم عمليات الإدخال / الإخراج المربوطة بالذاكرة باستخدام ربطات ذاكرة مباشرة تتجاوز ذاكرة التخزين المؤقت لصفحة النواة.
بينما تميل خاصية DAX إلى أن تؤدي إلى نقل البيانات بشكل متزامن، فإنها لا تعطي نفس ضمانات العلم O_SYNC (انظر open(2))، حيث يتم نقل البيانات والبيانات الوصفية الضرورية معًا.
قد يدعم ملف DAX الربط بالعلم MAP_SYNC، والذي يمكّن البرنامج من استخدام تعليمات مسح ذاكرة التخزين المؤقت لوحدة المعالجة المركزية لتثبيت عمليات تخزين وحدة المعالجة المركزية دون الحاجة إلى fsync(2) صريح. انظر mmap(2) لمزيد من المعلومات.
STATX_ATTR_MOUNT_ROOT (منذ لينكس 5.8)
الملف هو جذر تثبيت.

قيمة الإرجاع

عند النجاح، يتم إرجاع الصفر. عند حدوث خطأ، يتم إرجاع -1، ويتم تعيين errno للإشارة إلى الخطأ.

الأخطاء

EACCES
تم رفض إذن البحث لأحد الأدلة في البادئة الأمامية لـ path. (انظر أيضًا path_resolution(7).)
EBADF
path نسبي ولكن dirfd ليس AT_FDCWD وليس واصف ملف صالحًا.
EFAULT
path أو statxbuf يشير إلى موقع خارج مساحة العنوان التي يمكن الوصول إليها في العملية أو أنه NULL (باستثناء ما بعد لينكس 6.11 إذا تم تحديد AT_EMPTY_PATH في flags، يُسمح بأن يكون path NULL).
EINVAL
تم تحديد علم غير صالح في flags.
EINVAL
تم تحديد علم محجوز في mask. (حاليًا، يوجد علم واحد من هذا القبيل، يُشار إليه بالثابت STATX__RESERVED، بقيمة 0x80000000U.)
ELOOP
تم العثور على عدد كبير جدًا من الروابط الرمزية أثناء اجتياز اسم المسار.
ENAMETOOLONG
path طويل جدًا.
ENOENT
أحد مكونات path غير موجود، أو أن path سلسلة فارغة ولم يتم تحديد AT_EMPTY_PATH في flags.
ENOMEM
نفاد الذاكرة (أي ذاكرة النواة).
ENOTDIR
أحد مكونات البادئة الأمامية لـ path ليس دليلًا أو أن path نسبي وأن dirfd هو واصف ملف يشير إلى ملف غير دليل.
