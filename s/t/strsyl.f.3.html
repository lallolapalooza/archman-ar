subroutine strsyl (character trana, character tranb, integer isgn, integer m, integer n, real, dimension( lda, * ) a, integer lda, real, dimension( ldb, * ) b, integer ldb, real, dimension( ldc, * ) c, integer ldc, real scale, integer info)

STRSYL

الغرض:

!>
!> STRSYL يحل معادلة Sylvester المصفوفية الحقيقية:
!>
!>    op(A)*X + X*op(B) = scale*C أو
!>    op(A)*X - X*op(B) = scale*C,
!>
!> حيث op(A) = A أو A**T، و A و B كلاهما شبه مثلثية عليا.
!> A هي M × M و B هي N × N؛ الطرف الأيمن C والحل X هما M × N؛
!> و scale هو عامل قياس للإخراج، يتم تعيينه <= 1 لتجنب تجاوز السعة في X.
!>
!> يجب أن تكون A و B في الشكل القياسي لـ Schur (كما تم إرجاعه بواسطة SHSEQR)،
!> أي، شبه مصفوفة عليا كتلية مع كتل قطرية 1 × 1 و 2 × 2؛
!> كل كتلة قطرية 2 × 2 لها عناصرها القطرية متساوية وعناصرها غير القطرية
!> ذات إشارة متعاكسة.
!>

المعلمات

TRANA 
!>          TRANA هو CHARACTER*1
!>          يحدد خيار op(A):
!>          = 'N': op(A) = A    (بدون تبديل)
!>          = 'T': op(A) = A**T (تبديل)
!>          = 'C': op(A) = A**H (تبديل مرافق = تبديل)
!>

TRANB

!>          TRANB هو CHARACTER*1
!>          يحدد خيار op(B):
!>          = 'N': op(B) = B    (بدون تبديل)
!>          = 'T': op(B) = B**T (تبديل)
!>          = 'C': op(B) = B**H (تبديل مرافق = تبديل)
!>

ISGN

!>          ISGN هو INTEGER
!>          يحدد الإشارة في المعادلة:
!>          = +1: حل op(A)*X + X*op(B) = scale*C
!>          = -1: حل op(A)*X - X*op(B) = scale*C
!>

M

!>          M هو INTEGER
!>          رتبة المصفوفة A، وعدد الصفوف في المصفوفتين X و C. M >= 0.
!>

N

!>          N هو INTEGER
!>          رتبة المصفوفة B، وعدد الأعمدة في المصفوفتين X و C. N >= 0.
!>

A

!>          A هي مصفوفة REAL، ذات بُعد (LDA,M)
!>          المصفوفة شبه المثلثية العليا A، في شكل Schur القياسي.
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرئيسي للمصفوفة A. LDA >= max(1,M).
!>

B

!>          B هي مصفوفة REAL، ذات بُعد (LDB,N)
!>          المصفوفة شبه المثلثية العليا B، في شكل Schur القياسي.
!>

LDB

!>          LDB هو INTEGER
!>          البعد الرئيسي للمصفوفة B. LDB >= max(1,N).
!>

C

!>          C هي مصفوفة REAL، ذات بُعد (LDC,N)
!>          عند الدخول، مصفوفة الطرف الأيمن C ذات الأبعاد M × N.
!>          عند الخروج، يتم استبدال C بمصفوفة الحل X.
!>

LDC

!>          LDC هو INTEGER
!>          البعد الرئيسي للمصفوفة C. LDC >= max(1,M)
!>

SCALE

!>          SCALE هو REAL
!>          عامل القياس، scale، يتم تعيينه <= 1 لتجنب تجاوز السعة في X.
!>

INFO

!>          INFO هو INTEGER
!>          = 0: خروج ناجح
!>          < 0: إذا كان INFO = -i، فإن المعلمة i-th كانت لها قيمة غير قانونية
!>          = 1: A و B لهما قيم ذاتية مشتركة أو قريبة جدًا؛ تم استخدام قيم معدلة لحل المعادلة
!>               (ولكن المصفوفات A و B لم تتغير).
!>
