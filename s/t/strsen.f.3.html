توثيق الوظيفة/الروتين الفرعي

subroutine strsen (character job, character compq, logical, dimension( * ) select, integer n, real, dimension( ldt, * ) t, integer ldt, real, dimension( ldq, * ) q, integer ldq, real, dimension( * ) wr, real, dimension( * ) wi, integer m, real s, real sep, real, dimension( * ) work, integer lwork, integer, dimension( * ) iwork, integer liwork, integer info)

STRSEN

الغرض:

!>
!> STRSEN يعيد ترتيب تحليل شور الحقيقي لمصفوفة حقيقية
!> A = Q*T*Q**T، بحيث تظهر مجموعة مختارة من القيم الذاتية في
!> الكتل القطرية الأمامية للمصفوفة شبه المثلثية العلوية T،
!> والأعمدة الأمامية لـ Q تشكل أساسًا متعامدًا للمساحة المتجهية الصحيحة المقابلة.
!>
!> اختياريًا، تقوم الروتين بحساب الأعداد الشرطية المقلوبة للمجموعة المختارة من القيم الذاتية
!> و/أو المساحة المتجهية.
!>
!> يجب أن تكون T في شكل كانونيكي شور (كما تم إرجاعه بواسطة SHSEQR)، أي،
!> مصفوفة شبه مثلثية قطريًا مع كتل قطرية بحجم 1x1 و 2x2؛ كل
!> كتلة قطرية بحجم 2x2 لها عناصرها القطرية متساوية و
!> عناصرها غير القطرية ذات إشارة معاكسة.
!>

المعلمات

JOB 
!>          JOB هو CHARACTER*1
!>          يحدد ما إذا كانت الأعداد الشرطية مطلوبة للمجموعة المختارة من القيم الذاتية (S)
!>          أو المساحة المتجهية (SEP):
!>          = 'N': لا شيء؛
!>          = 'E': للقيم الذاتية فقط (S)؛
!>          = 'V': للمساحة المتجهية فقط (SEP)؛
!>          = 'B': لكل من القيم الذاتية والمساحة المتجهية (S و SEP).
!>

COMPQ

!>          COMPQ هو CHARACTER*1
!>          = 'V': تحديث المصفوفة Q لمتجهات شور؛
!>          = 'N': لا تحديث Q.
!>

SELECT

!>          SELECT هي مصفوفة LOGICAL، حجمها (N)
!>          تحدد SELECT القيم الذاتية في المجموعة المختارة. لتحديد قيمة ذاتية حقيقية w(j)،
!>          يجب تعيين SELECT(j) إلى .TRUE.. لتحديد زوج من القيم الذاتية المترافقة المعقدة
!>          w(j) و w(j+1)، المقابلة لكتلة قطرية بحجم 2x2، يجب تعيين إما SELECT(j) أو SELECT(j+1)
!>          أو كليهما إلى .TRUE.؛ يجب أن يكون زوج من القيم الذاتية المترافقة المعقدة إما
!>          مضمنًا في المجموعة أو مستبعدًا منها.
!>

N

!>          N هو INTEGER
!>          رتبة المصفوفة T. N >= 0.
!>

T

!>          T هي مصفوفة REAL، حجمها (LDT,N)
!>          عند الدخول، المصفوفة شبه المثلثية العلوية T، في شكل شور الكانونيكي.
!>          عند الخروج، يتم استبدال T بالمصفوفة المعاد ترتيبها T، مرة أخرى في
!>          شكل شور الكانونيكي، مع القيم الذاتية المختارة في الكتل القطرية الأمامية.
!>

LDT

!>          LDT هو INTEGER
!>          البعد الرائد للمصفوفة T. LDT >= max(1,N).
!>

Q

!>          Q هي مصفوفة REAL، حجمها (LDQ,N)
!>          عند الدخول، إذا كان COMPQ = 'V'، فإن المصفوفة Q لمتجهات شور.
!>          عند الخروج، إذا كان COMPQ = 'V'، تم ضرب Q بشكل لاحق بالمصفوفة
!>          التحويلية المتعامدة التي تعيد ترتيب T؛ الأعمدة الأمامية M لـ Q تشكل
!>          أساسًا متعامدًا للمساحة المتجهية الصحيحة المحددة.
!>          إذا كان COMPQ = 'N'، لا يتم الرجوع إلى Q.
!>

LDQ

!>          LDQ هو INTEGER
!>          البعد الرائد للمصفوفة Q.
!>          LDQ >= 1؛ وإذا كان COMPQ = 'V'، فإن LDQ >= N.
!>

WR

!>          WR هي مصفوفة REAL، حجمها (N)
!>

WI

!>          WI هي مصفوفة REAL، حجمها (N)
!>
!>          الأجزاء الحقيقية والتخيلية، على التوالي، للقيم الذاتية المعاد ترتيبها لـ T.
!>          تُخزن القيم الذاتية بنفس ترتيب عناصر القطر الرئيسي لـ T، حيث WR(i) = T(i,i)
!>          وإذا كانت T(i:i+1,i:i+1) كتلة قطرية بحجم 2x2، فإن WI(i) > 0 و WI(i+1) = -WI(i).
!>          لاحظ أنه إذا كانت قيمة ذاتية معقدة غير مستقرة بشكل كافٍ، فقد تختلف قيمتها
!>          بشكل كبير عن قيمتها قبل إعادة الترتيب.
!>

M

!>          M هو INTEGER
!>          بعد المساحة المتجهية المحددة.
!>          0 < = M <= N.
!>

S

!>          S هو REAL
!>          إذا كان JOB = 'E' أو 'B'، فإن S هو حد أدنى على المقلوب
!>          للرقم الشرطي للمجموعة المختارة من القيم الذاتية.
!>          لا يمكن أن يقلل S عن الرقم الشرطي الحقيقي بمقدار يزيد عن عامل sqrt(N).
!>          إذا كان M = 0 أو N، فإن S = 1.
!>          إذا كان JOB = 'N' أو 'V'، لا يتم الرجوع إلى S.
!>

SEP

!>          SEP هو REAL
!>          إذا كان JOB = 'V' أو 'B'، فإن SEP هو تقدير للمقلوب
!>          للرقم الشرطي للمساحة المتجهية المحددة.
!>          إذا كان M = 0 أو N، فإن SEP = norm(T).
!>          إذا كان JOB = 'N' أو 'E'، لا يتم الرجوع إلى SEP.
!>

WORK

!>          WORK هي مصفوفة REAL، حجمها (MAX(1,LWORK))
!>          عند الخروج، إذا كان INFO = 0، فإن WORK(1) يرجع LWORK الأمثل.
!>

LWORK

!>          LWORK هو INTEGER
!>          بعد المصفوفة WORK.
!>          إذا كان JOB = 'N'، LWORK >= max(1,N)؛
!>          إذا كان JOB = 'E'، LWORK >= max(1,M*(N-M))؛
!>          إذا كان JOB = 'V' أو 'B'، LWORK >= max(1,2*M*(N-M)).
!>
!>          إذا كان LWORK = -1، فسيتم افتراض استعلام مساحة العمل؛ الروتين
!>          يحسب فقط الحجم الأمثل للمصفوفة WORK، ويرجع
!>          هذه القيمة كإدخال أول للمصفوفة WORK، ولا يتم إصدار أي رسالة خطأ
!>          متعلقة بـ LWORK بواسطة XERBLA.
!>

IWORK

!>          IWORK هي مصفوفة INTEGER، حجمها (MAX(1,LIWORK))
!>          عند الخروج، إذا كان INFO = 0، فإن IWORK(1) يرجع LIWORK الأمثل.
!>

LIWORK

!>          LIWORK هو INTEGER
!>          بعد المصفوفة IWORK.
!>          إذا كان JOB = 'N' أو 'E'، LIWORK >= 1؛
!>          إذا كان JOB = 'V' أو 'B'، LIWORK >= max(1,M*(N-M)).
!>
!>          إذا كان LIWORK = -1، فسيتم افتراض استعلام مساحة العمل؛ الروتين
!>          يحسب فقط الحجم الأمثل للمصفوفة IWORK، ويرجع
!>          هذه القيمة كإدخال أول للمصفوفة IWORK، ولا يتم إصدار أي رسالة خطأ
!>          متعلقة بـ LIWORK بواسطة XERBLA.
!>

INFO

!>          INFO هو INTEGER
!>          = 0: خروج ناجح
!>          < 0: إذا كان INFO = -i، فإن المعلمة i كانت ذات قيمة غير قانونية
!>          = 1: فشل إعادة ترتيب T لأن بعض القيم الذاتية قريبة جدًا من بعضها البعض
!>               (المشكلة غير مستقرة للغاية)؛
!>               ربما تم إعادة ترتيب T جزئيًا، وتحتوي WR و WI
!>               على القيم الذاتية بنفس الترتيب كما في T؛ يتم تعيين S و
!>               SEP (إذا طُلبت) إلى الصفر.
!>

!>
!>  تقوم STRSEN أولاً بتجميع القيم الذاتية المختارة عن طريق حساب تحويل متعامد Z
!>  لنقلها إلى الزاوية العلوية اليسرى من T.
!>  بمعنى آخر، القيم الذاتية المختارة هي القيم الذاتية لـ T11
!>  في:
!>
!>          Z**T * T * Z = ( T11 T12 ) n1
!>                         (  0  T22 ) n2
!>                            n1  n2
!>
!>  حيث N = n1+n2 و Z**T تعني منقول Z. تشكل الأعمدة n1 الأولى من Z
!>  أساسًا متعامدًا للمساحة المتجهية الصحيحة المحددة
