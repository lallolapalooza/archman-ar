std::iterator< _Category, _Tp, _Distance, _Pointer, _Reference >(3) دليل وظائف المكتبة std::iterator< _Category, _Tp, _Distance, _Pointer, _Reference >(3)

الاسم

std::iterator< _Category, _Tp, _Distance, _Pointer, _Reference > - فئة مكرر مشتركة.

الملخص

#include <iterator>

تورث من std::reverse_iterator< iterator >، و std::reverse_iterator< const_iterator >.

أنواع عامة


typedef _Distance difference_type
يتم تمثيل المسافة بين المكررات بهذا النوع. typedef _Category iterator_category
أحد أنواع العلامات. typedef _Pointer pointer
يمثل هذا النوع مؤشرًا إلى نوع القيمة. typedef _Reference reference
يمثل هذا النوع مرجعًا لنوع القيمة. typedef _Tp value_type
النوع الذي يشير إليه المكرر.

وصف تفصيلي

template<typename _Category, typename _Tp, typename _Distance = ptrdiff_t, typename _Pointer = _Tp*, typename _Reference = _Tp&>

struct std::iterator< _Category, _Tp, _Distance, _Pointer, _Reference >"فئة مكرر مشتركة.

هذه الفئة لا تفعل شيئًا سوى تعريف أنواع `typedef` متداخلة. يمكن لفئات المكررات الوراثة من هذه الفئة لتوفير بعض العمل. تُستخدم أنواع `typedef` بعد ذلك في التخصصات والتجاوزات.

على وجه الخصوص، لا توجد تطبيقات افتراضية للمتطلبات مثل `operator++` وما شابه. (كيف يمكن أن تكون؟)

توثيق `typedef` الأعضاء

template<typename _Category, typename _Tp, typename _Distance = ptrdiff_t, typename _Pointer = _Tp*, typename _Reference = _Tp&> typedef _Distance std::iterator< _Category, _Tp, _Distance, _Pointer, _Reference >::difference_type

يتم تمثيل المسافة بين المكررات بهذا النوع.

template<typename _Category, typename _Tp, typename _Distance = ptrdiff_t, typename _Pointer = _Tp*, typename _Reference = _Tp&> typedef _Category std::iterator< _Category, _Tp, _Distance, _Pointer, _Reference >::iterator_category

أحد أنواع العلامات.

template<typename _Category, typename _Tp, typename _Distance = ptrdiff_t, typename _Pointer = _Tp*, typename _Reference = _Tp&> typedef _Pointer std::iterator< _Category, _Tp, _Distance, _Pointer, _Reference >::pointer

يمثل هذا النوع مؤشرًا إلى نوع القيمة.

template<typename _Category, typename _Tp, typename _Distance = ptrdiff_t, typename _Pointer = _Tp*, typename _Reference = _Tp&> typedef _Reference std::iterator< _Category, _ Tp, _Distance, _Pointer, _Reference >::reference

يمثل هذا النوع مرجعًا لنوع القيمة.

template<typename _Category, typename _Tp, typename _Distance = ptrdiff_t, typename _Pointer = _Tp*, typename _Reference = _Tp&> typedef _Tp std::iterator< _Category, _Tp, _Distance, _Pointer, _Reference >::value_type

النوع الذي "يشير إليه" المكرر.
