static_assert(3) دليل وظائف المكتبة static_assert(3)

الاسم

static_assert, _Static_assert - فشل التصريف إذا كان التأكيد خاطئًا

المكتبة

مكتبة C القياسية (libc)

الملخص

#include <assert.h>
void static_assert(bool constant-expression, const char *msg);
/* منذ C23: */
void static_assert(bool constant-expression);

الوصف

هذا الماكرو مشابه لـ assert(3)، ولكنه يعمل في وقت التصريف، ويصدر خطأ في التصريف (مع رسالة اختيارية) عندما يكون الإدخال خاطئًا (أي، يساوي صفرًا).

إذا كان الإدخال غير صفري، فلن يتم إصدار أي كود.

msg يجب أن يكون حرفًا نصيًا. منذ C23، هذه الوسيطة اختيارية.

يوجد كلمة مفتاحية، _Static_assert()، والتي تتصرف بشكل متطابق، ويمكن استخدامها دون تضمين <assert.h>.

قيمة الإرجاع

لا يتم إرجاع أي قيمة.

الإصدارات

في C11، كانت الوسيطة الثانية (msg) إلزامية؛ منذ C23، يمكن حذفها.

المعايير

C11 وما بعده.

أمثلة

لا يمكن استخدام static_assert() في بعض الأماكن، مثل على سبيل المثال على النطاق العام. لهذا، يمكن كتابة ماكرو must_be() باستخدام static_assert(). يقوم البرنامج التالي باستخدام الماكرو للحصول على حجم مصفوفة بأمان.


#include <assert.h>
#include <stdint.h>
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
/*
 * يتصرف هذا الماكرو مثل static_assert()، مما يؤدي إلى
 * فشل التصريف إذا لم يكن وسيطه صحيحًا. ومع ذلك، فإنه
 * يعيد دائمًا 0، مما يسمح باستخدامه في أي مكان يمكن
 * استخدام تعبير فيه.
 */
#define must_be(e)                                        \
(                                                         \
    0 * (int) sizeof(                                     \
        struct {                                          \
            static_assert(e);                             \
            int  ISO_C_forbids_a_struct_with_no_members;  \
        }                                                 \
    )                                                     \
)
#define is_same_type(a, b)  \
    __builtin_types_compatible_p(typeof(a), typeof(b))
#define is_array(arr)       (!is_same_type((arr), &*(arr)))
#define must_be_array(arr)  must_be(is_array(arr))
#define sizeof_array(arr)   (sizeof(arr) + must_be_array(arr))
#define NITEMS(arr)         (sizeof((arr)) / sizeof((arr)[0]) \
                             + must_be_array(arr))
int     foo[10];
int8_t  bar[sizeof_array(foo)];
int
main(void)
{
    for (size_t i = 0; i < NITEMS(foo); i++) {
        foo[i] = i;
    }
    memcpy(bar, foo, sizeof_array(bar));
    for (size_t i = 0; i < NITEMS(bar); i++) {
        printf("%d,", bar[i]);
    }
    exit(EXIT_SUCCESS);
}
