subroutine dstevr (character jobz, character range, integer n, double precision, dimension( * ) d, double precision, dimension( * ) e, double precision vl, double precision vu, integer il, integer iu, double precision abstol, integer m, double precision, dimension( * ) w, double precision, dimension( ldz, * ) z, integer ldz, integer, dimension( * ) isuppz, double precision, dimension( * ) work, integer lwork, integer, dimension( * ) iwork, integer liwork, integer info)

DSTEVR تحسب القيم الذاتية، وبشكل اختياري، المتجهات الذاتية اليسرى و/أو اليمنى لمصفوفة قطرية ثلاثية متماثلة T. القيم الذاتية والمتجهات الذاتية يمكن اختيارها بتحديد نطاق قيم أو نطاق مؤشرات للقيم الذاتية المطلوبة.

الغرض:

!>
!> DSTEVR تحسب القيم الذاتية المختارة، وبشكل اختياري، المتجهات الذاتية
!> لمصفوفة قطرية ثلاثية متماثلة T. يمكن اختيار القيم الذاتية
!> والمتجهات الذاتية بتحديد نطاق قيم أو نطاق مؤشرات للقيم الذاتية المطلوبة.
!>
!> كلما أمكن، تستدعي DSTEVR الدالة DSTEMR لحساب الطيف الذاتي
!> باستخدام تمثيلات قوية نسبياً. تحسب DSTEMR القيم الذاتية بواسطة خوارزمية dqds،
!> بينما تحسب المتجهات الذاتية المتعامدة من تمثيلات L D L^T مختلفة
!> (تُعرف أيضًا بالتمثيلات القوية نسبياً). يتم تجنب التعامد باستخدام
!> Gram-Schmidt قدر الإمكان. بشكل أكثر تحديدًا،
!> خطوات الخوارزمية المختلفة هي كالتالي. بالنسبة للكتلة غير المختزلة i من T،
!>    (a) حساب T - sigma_i = L_i D_i L_i^T، بحيث تكون L_i D_i L_i^T
!>         تمثيلاً قوياً نسبياً،
!>    (b) حساب القيم الذاتية، lambda_j، لـ L_i D_i L_i^T بدقة نسبية عالية
!>        باستخدام خوارزمية dqds،
!>    (c) إذا كان هناك تجمع من القيم الذاتية القريبة، sigma_i
!>        قريب من التجمع، انتقل إلى الخطوة (a)،
!>    (d) مع الأخذ في الاعتبار القيمة الذاتية التقريبية lambda_j لـ L_i D_i L_i^T،
!>        حساب المتجه الذاتي المقابل عن طريق تكوين
!>        تحليل ملتوٍ يكشف الرتبة.
!> يمكن تحديد دقة المخرجات المطلوبة بواسطة المعامل المدخل
!> ABSTOL.
!>
!> لمزيد من التفاصيل، انظر ، بواسطة Inderjit Dhillon،
!> تقرير فني لقسم علوم الحاسوب رقم UCB//CSD-97-971،
!> UC Berkeley، مايو 1997.
!>
!>
!> ملاحظة 1: تستدعي DSTEVR الدالة DSTEMR عند طلب الطيف الكامل
!> على الأجهزة التي تتوافق مع معيار الفاصلة العائمة ieee-754.
!> تستدعي DSTEVR الدالتين DSTEBZ و DSTEIN على الأجهزة غير ieee
!> وعند طلب طيف جزئي.
!>
!> التنفيذ الطبيعي لـ DSTEMR قد ينشئ قيم NaN و infinity وقد يتوقف
!> بسبب استثناء فاصلة عائمة في البيئات
!> التي لا تتعامل مع قيم NaN و infinity بالطريقة الافتراضية لمعيار ieee.
!>

المعاملات

JOBZ 
!>          JOBZ هو CHARACTER*1
!>          = 'N':  حساب القيم الذاتية فقط؛
!>          = 'V':  حساب القيم الذاتية والمتجهات الذاتية.
!>

RANGE

!>          RANGE هو CHARACTER*1
!>          = 'A': سيتم العثور على جميع القيم الذاتية.
!>          = 'V': سيتم العثور على جميع القيم الذاتية في الفاصلة النصف مفتوحة (VL,VU].
!>          = 'I': سيتم العثور على القيم الذاتية من IL إلى IU.
!>          بالنسبة لـ RANGE = 'V' أو 'I' و IU - IL < N - 1، يتم استدعاء DSTEBZ و
!>          DSTEIN
!>

N

!>          N هو INTEGER
!>          رتبة المصفوفة.  N >= 0.
!>

D

!>          D هو مصفوفة DOUBLE PRECISION، البعد (N)
!>          عند الإدخال، العناصر القطرية n للمصفوفة الثلاثية
!>          A.
!>          عند الخروج، قد يتم ضرب D بعامل ثابت يتم اختياره
!>          لتجنب الحمل الزائد/التقصير في حساب القيم الذاتية.
!>

E

!>          E هو مصفوفة DOUBLE PRECISION، البعد (max(1,N-1))
!>          عند الإدخال، عناصر تحت القطر n للمصفوفة الثلاثية
!>          A في العناصر 1 إلى N-1 من E.
!>          عند الخروج، قد يتم ضرب E بعامل ثابت يتم اختياره
!>          لتجنب الحمل الزائد/التقصير في حساب القيم الذاتية.
!>

VL

!>          VL هو DOUBLE PRECISION
!>          إذا كان RANGE='V'، الحد الأدنى للفاصلة
!>          التي سيتم البحث فيها عن القيم الذاتية. VL < VU.
!>          لا تتم الإشارة إليه إذا كان RANGE = 'A' أو 'I'.
!>

VU

!>          VU هو DOUBLE PRECISION
!>          إذا كان RANGE='V'، الحد الأعلى للفاصلة
!>          التي سيتم البحث فيها عن القيم الذاتية. VL < VU.
!>          لا تتم الإشارة إليه إذا كان RANGE = 'A' أو 'I'.
!>

IL

!>          IL هو INTEGER
!>          إذا كان RANGE='I'، مؤشر
!>          أصغر قيمة ذاتية سيتم إرجاعها.
!>          1 <= IL <= IU <= N، إذا كان N > 0؛ IL = 1 و IU = 0 إذا كان N = 0.
!>          لا تتم الإشارة إليه إذا كان RANGE = 'A' أو 'V'.
!>

IU

!>          IU هو INTEGER
!>          إذا كان RANGE='I'، مؤشر
!>          أكبر قيمة ذاتية سيتم إرجاعها.
!>          1 <= IL <= IU <= N، إذا كان N > 0؛ IL = 1 و IU = 0 إذا كان N = 0.
!>          لا تتم الإشارة إليه إذا كان RANGE = 'A' أو 'V'.
!>

ABSTOL

!>          ABSTOL هو DOUBLE PRECISION
!>          التفاوت المطلق للخطأ في القيم الذاتية.
!>          يتم قبول قيمة ذاتية تقريبية على أنها متقاربة
!>          عندما تحدد أنها تقع في فاصلة [a,b]
!>          بعرض يساوي أو أقل من
!>
!>                  ABSTOL + EPS *   max( |a|,|b| ) ،
!>
!>          حيث EPS هي دقة الآلة. إذا كان ABSTOL أقل من
!>          أو يساوي صفرًا، فسيتم استخدام EPS*|T| بدلاً منه،
!>          حيث |T| هو المعيار 1 للمصفوفة الثلاثية التي تم الحصول عليها
!>          عن طريق تقليل A إلى شكل ثلاثي.
!>
!>          انظر  بواسطة Demmel و
!>          Kahan، مذكرة عمل LAPACK #3.
!>
!>          إذا كانت الدقة النسبية العالية مهمة، اضبط ABSTOL على
!>          DLAMCH ('الحد الأدنى الآمن'). سيضمن هذا أن
!>          القيم الذاتية يتم حسابها بدقة نسبية عالية عندما
!>          يكون ذلك ممكنًا في الإصدارات المستقبلية. لا يقوم
!>          الرمز الحالي بأي ضمانات بشأن الدقة النسبية العالية،
!>          لكن الإصدارات المستقبلية ستقوم بذلك. انظر J. Barlow و J. Demmel،
!>          ، مذكرة عمل LAPACK #7، لمناقشة
!>          المصفوفات التي تحدد قيمها الذاتية بدقة نسبية عالية.
!>

M

!>          M هو INTEGER
!>          إجمالي عدد القيم الذاتية التي تم العثور عليها.  0 <= M <= N.
!>          إذا كان RANGE = 'A'، فإن M = N، وإذا كان RANGE = 'I'، فإن M = IU-IL+1.
!>

W

!>          W هو مصفوفة DOUBLE PRECISION، البعد (N)
!>          تحتوي العناصر الأولى M على القيم الذاتية المختارة بترتيب
!>          تصاعدي.
!>

Z

!>          Z هو مصفوفة DOUBLE PRECISION، البعد (LDZ, max(1,M) )
!>          إذا كان JOBZ = 'V'، فعندما يكون INFO = 0، فإن الأعمدة الأولى M من Z
!>          تحتوي على المتجهات الذاتية المتعامدة للمصفوفة A
!>          المقابلة للقيم الذاتية المختارة، مع احتواء العمود i من Z
!>          على المتجه الذاتي المرتبط بـ W(i).
!>          ملاحظة: المستخدم
