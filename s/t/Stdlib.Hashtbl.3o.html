(* 
   Example of hash table usage.

   This module provides a function to count characters in a sequence,
   and then display the counts sorted by character.
*)

(* Type alias for clarity *)
module Char_tbl = Hashtbl

(** [count_chars seq] returns a list of pairs (character, count)
    for each character present in the input sequence [seq].
    The list is sorted alphabetically by character.

    @param seq The input sequence of characters.
    @return A list of (Char.t * int) pairs, sorted by character. *)
let count_chars (seq : char Seq.t) : (Char.t * int) list =
        let counts = Char_tbl.create 16 in
        Seq.iter
          (fun c ->
            let count_c =
              Char_tbl.find_opt counts c
              |> Option.value ~default:0
            in
            Char_tbl.replace counts c (count_c + 1))
          seq;
        (* turn into a list *)
        Char_tbl.fold (fun c n l -> (c,n) :: l) counts []
          |> List.sort (fun (c1,_)(c2,_) -> Char.compare c1 c2)

(* Define the type for the count_chars function *)
val count_chars : Char_tbl.key Seq.t -> (Char.t * int) list

(* Create a sequence from a string *)
# let seq = String.to_seq "hello world, and all the camels in it!"
val seq : char Seq.t = 

(* Count characters in the first sequence *)
# count_chars seq
- : (Char.t * int) list =
[(' ', 7); ('!', 1); (',', 1); ('a', 3); ('c', 1); ('d', 2); ('e', 3);
 ('h', 2); ('i', 2); ('l', 6); ('m', 1); ('n', 2); ('o', 2); ('r', 1);
 ('s', 1); ('t', 2); ('w', 1)]

(* Create a second, repeating sequence *)
# let seq2 =
    Seq.cycle (String.to_seq "abc") |> Seq.take 31
val seq2 : char Seq.t = 

(* Convert the second sequence to a string to display it *)
# String.of_seq seq2
- : String.t = "abcabcabcabcabcabcabcabcabcabca"

(* Count characters in the second sequence *)
# count_chars seq2
- : (Char.t * int) list = [('a', 11); ('b', 10); ('c', 10)]
