subroutine strsna (character job, character howmny, logical, dimension( * ) select, integer n, real, dimension( ldt, * ) t, integer ldt, real, dimension( ldvl, * ) vl, integer ldvl, real, dimension( ldvr, * ) vr, integer ldvr, real, dimension( * ) s, real, dimension( * ) sep, integer mm, integer m, real, dimension( ldwork, * ) work, integer ldwork, integer, dimension( * ) iwork, integer info)

STRSNA

الغرض:

!>
!> STRSNA تقدر أرقام الشروط المعكوسة للقيم الذاتية و/أو المتجهات الذاتية اليمنى المحددة
!> لمصفوفة T شبه قطرية حقيقية عليا (أو لأي مصفوفة Q*T*Q**T مع Q
!> متعامدة).
!>
!> يجب أن تكون T في الشكل القانوني لشور (كما تم إرجاعه بواسطة SHSEQR)، أي،
!> شبه قطرية مربعة مع كتل قطرية 1x1 و 2x2؛ كل
!> كتلة قطرية 2x2 لها عناصرها القطرية متساوية وعناصرها
!> خارج القطر ذات إشارة معاكسة.
!>

المعاملات

JOB 
!>          JOB is CHARACTER*1
!>          يحدد ما إذا كانت أرقام الشروط مطلوبة للقيم الذاتية
!>          (S) أو المتجهات الذاتية (SEP):
!>          = 'E': للقيم الذاتية فقط (S)؛
!>          = 'V': للمتجهات الذاتية فقط (SEP)؛
!>          = 'B': لكلا من القيم الذاتية والمتجهات الذاتية (S و SEP).
!>

HOWMNY

!>          HOWMNY is CHARACTER*1
!>          = 'A': حساب أرقام الشروط لجميع الأزواج الذاتية؛
!>          = 'S': حساب أرقام الشروط للأزواج الذاتية المحددة
!>                 المحددة بواسطة المصفوفة SELECT.
!>

SELECT

!>          SELECT is LOGICAL array, dimension (N)
!>          إذا كان HOWMNY = 'S'، فإن SELECT يحدد الأزواج الذاتية التي
!>          تُطلب أرقام الشروط لها. لتحديد أرقام الشروط
!>          للزوج الذاتي المقابل لقيمة ذاتية حقيقية w(j)،
!>          يجب تعيين SELECT(j) إلى .TRUE.. لتحديد أرقام الشروط
!>          المقابلة لزوج مترافق مركب من قيم ذاتية w(j)
!>          و w(j+1)، يجب تعيين SELECT(j) أو SELECT(j+1) أو كليهما
!>          إلى .TRUE..
!>          إذا كان HOWMNY = 'A'، فلن يتم استخدام SELECT.
!>

N

!>          N is INTEGER
!>          رتبة المصفوفة T. N >= 0.
!>

T

!>          T is REAL array, dimension (LDT,N)
!>          المصفوفة شبه القطرية العليا T، في شكل شور القانوني.
!>

LDT

!>          LDT is INTEGER
!>          البعد الرئيسي للمصفوفة T. LDT >= max(1,N).
!>

VL

!>          VL is REAL array, dimension (LDVL,M)
!>          إذا كان JOB = 'E' أو 'B'، فيجب أن تحتوي VL على المتجهات الذاتية اليسرى لـ T
!>          (أو لأي Q*T*Q**T مع Q متعامدة)، المقابلة للأزواج الذاتية
!>          المحددة بواسطة HOWMNY و SELECT. يجب تخزين المتجهات الذاتية
!>          في أعمدة متتالية في VL، كما تم إرجاعها بواسطة
!>          SHSEIN أو STREVC.
!>          إذا كان JOB = 'V'، فلن يتم استخدام VL.
!>

LDVL

!>          LDVL is INTEGER
!>          البعد الرئيسي للمصفوفة VL.
!>          LDVL >= 1؛ وإذا كان JOB = 'E' أو 'B'، فإن LDVL >= N.
!>

VR

!>          VR is REAL array, dimension (LDVR,M)
!>          إذا كان JOB = 'E' أو 'B'، فيجب أن تحتوي VR على المتجهات الذاتية اليمنى لـ T
!>          (أو لأي Q*T*Q**T مع Q متعامدة)، المقابلة للأزواج الذاتية
!>          المحددة بواسطة HOWMNY و SELECT. يجب تخزين المتجهات الذاتية
!>          في أعمدة متتالية في VR، كما تم إرجاعها بواسطة
!>          SHSEIN أو STREVC.
!>          إذا كان JOB = 'V'، فلن يتم استخدام VR.
!>

LDVR

!>          LDVR is INTEGER
!>          البعد الرئيسي للمصفوفة VR.
!>          LDVR >= 1؛ وإذا كان JOB = 'E' أو 'B'، فإن LDVR >= N.
!>

S

!>          S is REAL array, dimension (MM)
!>          إذا كان JOB = 'E' أو 'B'، فإن أرقام الشروط المعكوسة للقيم الذاتية
!>          المختارة، مخزنة في عناصر متتالية في
!>          المصفوفة. بالنسبة لزوج مترافق مركب من القيم الذاتية، يتم
!>          تعيين عنصرين متتاليين من S إلى نفس القيمة. وبالتالي
!>          فإن S(j)، SEP(j)، والأعمدة j من VL و VR كلها
!>          تتوافق مع نفس الزوج الذاتي (ولكن ليس بالضرورة الزوج الذاتي
!>          j، إلا إذا تم اختيار جميع الأزواج الذاتية).
!>          إذا كان JOB = 'V'، فلن يتم استخدام S.
!>

SEP

!>          SEP is REAL array, dimension (MM)
!>          إذا كان JOB = 'V' أو 'B'، فإن أرقام الشروط المعكوسة المقدرة
!>          للمتجهات الذاتية المختارة، مخزنة في عناصر
!>          متتالية في المصفوفة. بالنسبة لمتجه ذاتي مركب، يتم
!>          تعيين عنصرين متتاليين من SEP إلى نفس القيمة. إذا
!>          تعذر إعادة ترتيب القيم الذاتية لحساب SEP(j)، يتم
!>          تعيين SEP(j) إلى 0؛ يمكن أن يحدث هذا فقط عندما تكون القيمة الحقيقية صغيرة جدًا.
!>          إذا كان JOB = 'E'، فلن يتم استخدام SEP.
!>

MM

!>          MM is INTEGER
!>          عدد العناصر في المصفوفتين S (إذا كان JOB = 'E' أو 'B')
!>           و/أو SEP (إذا كان JOB = 'V' أو 'B'). MM >= M.
!>

M

!>          M is INTEGER
!>          عدد العناصر في المصفوفتين S و/أو SEP المستخدمة فعليًا
!>          لتخزين تقديرات أرقام الشروط.
!>          إذا كان HOWMNY = 'A'، يتم تعيين M إلى N.
!>

WORK

!>          WORK is REAL array, dimension (LDWORK,N+6)
!>          إذا كان JOB = 'E'، فلن يتم استخدام WORK.
!>

LDWORK

!>          LDWORK is INTEGER
!>          البعد الرئيسي للمصفوفة WORK.
!>          LDWORK >= 1؛ وإذا كان JOB = 'V' أو 'B'، فإن LDWORK >= N.
!>

IWORK

!>          IWORK is INTEGER array, dimension (2*(N-1))
!>          إذا كان JOB = 'E'، فلن يتم استخدام IWORK.
!>

INFO

!>          INFO is INTEGER
!>          = 0: انتهاء ناجح
!>          < 0: إذا كان INFO = -i، فإن المعامل i كان له قيمة غير قانونية
!>

تفاصيل إضافية:

!>
!>  يُعرّف مقلوب رقم شرط القيمة الذاتية lambda على أنه
!>
!>          S(lambda) = |v**T*u| / (norm(u)*norm(v))
!>
!>  حيث u و v هما المتجهان الذاتيان الأيمن والأيسر لـ T المقابلان
!>  لـ lambda؛ v**T هو منقول v، و norm(u)
!>  يمثل المعيار الإقليدي. تقع أرقام الشروط المعكوسة هذه دائمًا
!>  بين صفر (مشروط بشكل سيء للغاية) وواحد (مشروط بشكل جيد للغاية). إذا كان n = 1،
!>  يتم تعريف S(lambda) لتكون 1.
!>
!>  يُعطى تقدير تقريبي لحد الخطأ للقيمة الذاتية المحسوبة W(i) بواسطة
!>
!>                      EPS * norm(T) / S(i)
!>
!>  حيث EPS هي دقة الآلة.
!>
!>  يُعرّف مقلوب رقم شرط المتجه الذاتي الأيمن u
!>  المقابل لـ lambda على النحو التالي. افترض
!>
!>              T = ( lambda  c  )
!>                  (   0    T22 )
!>
!>  إذًا، يكون رقم الشرط المعكوس هو
!>
!>          SEP( lambda, T22 ) = sigma-min( T22 - lambda*I )
!>
!>  حيث sigma-min يمثل أصغر قيمة مفردة. نقوم بتقريب
!>  أصغر قيمة مفردة بواسطة مقلوب تقدير
!>  لمعيار الواحد لمقلوب T22 - lambda*I. إذا كان n = 1،
!>  يتم تعريف SEP(1) ليكون abs(T(1,1)).
!>
!>  تقدير تقريبي
