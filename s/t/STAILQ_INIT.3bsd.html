STAILQ_INIT(3bsd)

الحزمة: libbsd

التراخيص: custom

NAME

STAILQ_INIT, STAILQ_INSERT_HEAD, STAILQ_INSERT_TAIL, STAILQ_INSERT_AFTER, STAILQ_INSERT_BEFORE, STAILQ_REMOVE, STAILQ_SWAP — singly linked tail queue macros

LIBRARY

libbsd

SYNOPSIS

#include <bsd/sys/queue.h>

void
STAILQ_INIT(struct staq_head *head);

void
STAILQ_INSERT_HEAD(struct staq_head *head, struct staq_entry *elm,
    struct staq_entry *last);

void
STAILQ_INSERT_TAIL(struct staq_head *head, struct staq_entry *elm,
    struct staq_entry *last);

void
STAILQ_INSERT_AFTER(struct staq_head *head, struct staq_entry *listelm,
    struct staq_entry *elm, struct staq_entry *last);

void
STAILQ_INSERT_BEFORE(struct staq_head *head, struct staq_entry *listelm,
    struct staq_entry *elm, struct staq_entry *last);

void
STAILQ_REMOVE(struct staq_head *head, struct staq_entry *elm,
    struct staq_entry *last);

void
STAILQ_SWAP(struct staq_head *head1, struct staq_head *head2,
    struct staq_entry *last);

DESCRIPTION

These macros provide an implementation of singly linked tail queues. A singly linked tail queue is a singly linked list where the last element may be accessed in O(1) time. The list is accessible through the head member of the first element and the tail member of the last element. The STAILQ_INIT() macro initializes the tail queue specified by head.

The macros STAILQ_INSERT_HEAD(), STAILQ_INSERT_TAIL(), STAILQ_INSERT_AFTER(), and STAILQ_INSERT_BEFORE() insert the element elm into the queue. The last argument is the pointer to the tail member of the last element in the queue.

The macro STAILQ_REMOVE() removes the element elm from the queue. The last argument is the pointer to the tail member of the last element in the queue.

The macro STAILQ_SWAP() swaps the contents of the two queues head1 and head2. The last argument is the pointer to the tail member of the last element in the queue.

SINGLY-LINKED LISTS

Singly linked lists are implemented by the macros SLIST_HEAD, SLIST_INIT, SLIST_INSERT_HEAD, SLIST_INSERT_AFTER, SLIST_REMOVE_HEAD, SLIST_REMOVE, SLIST_FOREACH, and SLIST_FIRST.

A singly linked list requires that each element contains a pointer to the next element. The list itself is represented by a pointer to the first element.

For example:

struct my_list_entry {
    int data;
    SLIST_ENTRY(my_list_entry) next;
};

struct my_list_entry *p, *n;
struct my_list_head my_list_head = SLIST_HEAD_INITIALIZER(my_list_head);

SLIST_INSERT_HEAD(&my_list_head, p, next);
SLIST_REMOVE(&my_list_head, p, my_list_entry, next);

p = SLIST_FIRST(&my_list_head);
SLIST_FOREACH(p, &my_list_head, next) {
    /* ... */
}

SINGLY-LINKED LIST EXAMPLE

#include <stdio.h>
#include <stdlib.h>
#include <bsd/sys/queue.h>

struct my_list_entry {
    int data;
    SLIST_ENTRY(my_list_entry) next;
};

struct my_list_head {
    struct my_list_entry *slh_first;
};

int
main(void) {
    struct my_list_head head;
    struct my_list_entry *p, *n;
    struct my_list_entry *elem1, *elem2;

    SLIST_INIT(&head);

    elem1 = malloc(sizeof(*elem1));
    elem1->data = 1;
    SLIST_INSERT_HEAD(&head, elem1, next);

    elem2 = malloc(sizeof(*elem2));
    elem2->data = 2;
    SLIST_INSERT_HEAD(&head, elem2, next);

    printf("List contents:\n");
    SLIST_FOREACH(p, &head, next) {
        printf("data = %d\n", p->data);
    }

    p = SLIST_FIRST(&head);
    printf("First element data = %d\n", p->data);

    n = p;
    p = SLIST_NEXT(p, next);
    SLIST_REMOVE(&head, n, my_list_entry, next);
    free(n);

    printf("List contents after removal:\n");
    SLIST_FOREACH(p, &head, next) {
        printf("data = %d\n", p->data);
    }

    while (!SLIST_EMPTY(&head)) {
        p = SLIST_FIRST(&head);
        SLIST_REMOVE_HEAD(&head, next);
        free(p);
    }

    return 0;
}

SINGLY-LINKED TAIL QUEUES

Singly linked tail queues are implemented by the macros STAILQ_HEAD, STAILQ_INIT, STAILQ_INSERT_HEAD, STAILQ_INSERT_TAIL, STAILQ_INSERT_AFTER, STAILQ_REMOVE, STAILQ_FOREACH, and STAILQ_FIRST.

A singly linked tail queue requires that each element contains a pointer to the next element and a pointer to the tail element. The list itself is represented by a pointer to the head element and a pointer to the tail element.

For example:

struct my_tailq_entry {
    int data;
    STAILQ_ENTRY(my_tailq_entry) next;
};

struct my_tailq_head {
    struct my_tailq_entry *stqh_first;
    struct my_tailq_entry **stqh_last;
};

struct my_tailq_head my_tailq_head = STAILQ_HEAD_INITIALIZER(my_tailq_head);

STAILQ_INSERT_HEAD(&my_tailq_head, p, next);
STAILQ_INSERT_TAIL(&my_tailq_head, p, next);
STAILQ_REMOVE(&my_tailq_head, p, my_tailq_entry, next);

p = STAILQ_FIRST(&my_tailq_head);
STAILQ_FOREACH(p, &my_tailq_head, next) {
    /* ... */
}

LISTS

Doubly linked lists are implemented by the macros LIST_HEAD, LIST_INIT, LIST_INSERT_HEAD, LIST_INSERT_TAIL, LIST_INSERT_AFTER, LIST_INSERT_BEFORE, LIST_REMOVE, LIST_FOREACH, LIST_FIRST, and LIST_NEXT.

A doubly linked list requires that each element contains pointers to the next and previous elements. The list itself is represented by a pointer to the first element.

For example:

struct my_list_entry {
    int data;
    LIST_ENTRY(my_list_entry) next;
};

struct my_list_head {
    struct my_list_entry *list_first;
};

struct my_list_head my_list_head = LIST_HEAD_INITIALIZER(my_list_head);

LIST_INSERT_HEAD(&my_list_head, p, next);
LIST_INSERT_TAIL(&my_list_head, p, next);
LIST_REMOVE(&my_list_head, p, next);

p = LIST_FIRST(&my_list_head);
LIST_FOREACH(p, &my_list_head, next) {
    /* ... */
}

TAIL QUEUES

Doubly linked tail queues are implemented by the macros TAILQ_HEAD, TAILQ_INIT, TAILQ_INSERT_HEAD, TAILQ_INSERT_TAIL, TAILQ_INSERT_AFTER, TAILQ_INSERT_BEFORE, TAILQ_REMOVE, TAILQ_FOREACH, TAILQ_FIRST, and TAILQ_NEXT.

A doubly linked tail queue requires that each element contains pointers to the next and previous elements. The list itself is represented by a pointer to the head element and a pointer to the tail element.

For example:

struct my_tailq_entry {
    int data;
    TAILQ_ENTRY(my_tailq_entry) next;
};

struct my_tailq_head {
    struct my_tailq_entry *tqh_first;
    struct my_tailq_entry **tqh_last;
};

struct my_tailq_head my_tailq_head = TAILQ_HEAD_INITIALIZER(my_tailq_head);

TAILQ_INSERT_HEAD(&my_tailq_head, p, next);
TAILQ_INSERT_TAIL(&my_tailq_head, p, next);
TAILQ_REMOVE(&my_tailq_head, p, my_tailq_entry, next);

p = TAILQ_FIRST(&my_tailq_head);
TAILQ_FOREACH(p, &my_tailq_head, next) {
    /* ... */
}

DIAGNOSTICS

When debugging queue(3), it can be useful to trace queue changes. To enable tracing, define the macro QUEUE_MACRO_DEBUG_TRACE at compile time.

It can also be useful to trash pointers that have been unlinked from a queue, to detect use after removal. To enable pointer trashing, define the macro QUEUE_MACRO_DEBUG_TRASH at compile time. The macro QMD_IS_TRASHED(void *ptr) returns true if ptr has been trashed by the QUEUE_MACRO_DEBUG_TRASH option.
