توثيق الوظيفة/الإجراء الفرعي

subroutine stprfb (character side, character trans, character direct, character storev, integer m, integer n, integer k, integer l, real, dimension( ldv, * ) v, integer ldv, real, dimension( ldt, * ) t, integer ldt, real, dimension( lda, * ) a, integer lda, real, dimension( ldb, * ) b, integer ldb, real, dimension( ldwork, * ) work, integer ldwork)

STPRFB يطبق معكوس كتلة "مثلثي-خماسي" حقيقي على مصفوفة حقيقية، والتي تتكون من كتلتين.

الغرض:

!>
!> STPRFB يطبق معكوس كتلة حقيقي H أو منقوله H**T على مصفوفة حقيقية C،
!> والتي تتكون من كتلتين A و B، سواء من اليسار أو اليمين.
!>
!>

المعلمات:

SIDE 
!>          SIDE هو CHARACTER*1
!>          = 'L': تطبيق H أو H**T من اليسار
!>          = 'R': تطبيق H أو H**T من اليمين
!>

TRANS

!>          TRANS هو CHARACTER*1
!>          = 'N': تطبيق H (بدون تبديل)
!>          = 'T': تطبيق H**T (تبديل)
!>

DIRECT

!>          DIRECT هو CHARACTER*1
!>          يشير إلى كيفية تشكيل H من حاصل ضرب معكوسات أولية
!>          = 'F': H = H(1) H(2) . . . H(k) (أمامي)
!>          = 'B': H = H(k) . . . H(2) H(1) (خلفي)
!>

STOREV

!>          STOREV هو CHARACTER*1
!>          يشير إلى كيفية تخزين المتجهات التي تحدد المعكوسات الأولية:
!>          = 'C': أعمدة
!>          = 'R': صفوف
!>

M

!>          M هو INTEGER
!>          عدد صفوف المصفوفة B.
!>          M >= 0.
!>

N

!>          N هو INTEGER
!>          عدد أعمدة المصفوفة B.
!>          N >= 0.
!>

K

!>          K هو INTEGER
!>          رتبة المصفوفة T، أي عدد المعكوسات الأولية التي يشكل حاصل
!>          ضربها المعكوس الكتلي.
!>          K >= 0.
!>

L

!>          L هو INTEGER
!>          رتبة الجزء شبه المنحرف من V.
!>          K >= L >= 0. انظر التفاصيل الإضافية.
!>

V

!>          V هي مصفوفة REAL، أبعادها
!>                                (LDV,K) إذا كان STOREV = 'C'
!>                                (LDV,M) إذا كان STOREV = 'R' و SIDE = 'L'
!>                                (LDV,N) إذا كان STOREV = 'R' و SIDE = 'R'
!>          المصفوفة الخماسية V، التي تحتوي على المعكوسات الأولية
!>          H(1), H(2), ..., H(K). انظر التفاصيل الإضافية.
!>

LDV

!>          LDV هو INTEGER
!>          البعد الرائد للمصفوفة V.
!>          إذا كان STOREV = 'C' و SIDE = 'L'، فإن LDV >= max(1,M)؛
!>          إذا كان STOREV = 'C' و SIDE = 'R'، فإن LDV >= max(1,N)؛
!>          إذا كان STOREV = 'R'، فإن LDV >= K.
!>

T

!>          T هي مصفوفة REAL، أبعادها (LDT,K)
!>          المصفوفة المثلثية K-by-K T في تمثيل المعكوس الكتلي.
!>

LDT

!>          LDT هو INTEGER
!>          البعد الرائد للمصفوفة T.
!>          LDT >= K.
!>

A

!>          A هي مصفوفة REAL، أبعادها
!>          (LDA,N) إذا كان SIDE = 'L' أو (LDA,K) إذا كان SIDE = 'R'
!>          عند الإدخال، المصفوفة A بحجم K-by-N أو M-by-K.
!>          عند الخروج، يتم استبدال A بالجزء المقابل من
!>          H*C أو H**T*C أو C*H أو C*H**T. انظر التفاصيل الإضافية.
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرائد للمصفوفة A.
!>          إذا كان SIDE = 'L'، فإن LDA >= max(1,K)؛
!>          إذا كان SIDE = 'R'، فإن LDA >= max(1,M).
!>

B

!>          B هي مصفوفة REAL، أبعادها (LDB,N)
!>          عند الإدخال، المصفوفة B بحجم M-by-N.
!>          عند الخروج، يتم استبدال B بالجزء المقابل من
!>          H*C أو H**T*C أو C*H أو C*H**T. انظر التفاصيل الإضافية.
!>

LDB

!>          LDB هو INTEGER
!>          البعد الرائد للمصفوفة B.
!>          LDB >= max(1,M).
!>

WORK

!>          WORK هي مصفوفة REAL، أبعادها
!>          (LDWORK,N) إذا كان SIDE = 'L'،
!>          (LDWORK,K) إذا كان SIDE = 'R'.
!>

LDWORK

!>          LDWORK هو INTEGER
!>          البعد الرائد للمصفوفة WORK.
!>          إذا كان SIDE = 'L'، فإن LDWORK >= K؛
!>          إذا كان SIDE = 'R'، فإن LDWORK >= M.
!>

تفاصيل إضافية:

!>
!>  المصفوفة C هي مصفوفة مركبة تتكون من الكتل A و B.
!>  الكتلة B هي بحجم M-by-N؛ إذا كان SIDE = 'R'، فإن A هي بحجم M-by-K،
!>  وإذا كان SIDE = 'L'، فإن A هي بحجم K-by-N.
!>
!>  إذا كان SIDE = 'R' و DIRECT = 'F'، فإن C = [A B].
!>
!>  إذا كان SIDE = 'L' و DIRECT = 'F'، فإن C = [A]
!>                                      [B].
!>
!>  إذا كان SIDE = 'R' و DIRECT = 'B'، فإن C = [B A].
!>
!>  إذا كان SIDE = 'L' و DIRECT = 'B'، فإن C = [B]
!>                                      [A].
!>
!>  المصفوفة الخماسية V تتكون من كتلة مستطيلة V1 وكتلة شبه منحرف V2.
!>  يتم تحديد حجم الكتلة شبه المنحرفة بواسطة المعلمة L، حيث 0<=L<=K.
!>  إذا كان L=K، فإن الكتلة V2 من V تكون مثلثية؛ إذا كان L=0، فلا توجد كتلة شبه منحرف،
!>  وبالتالي V = V1 تكون مستطيلة.
!>
!>  إذا كان DIRECT = 'F' و STOREV = 'C': V = [V1]
!>                                         [V2]
!>     - V2 شبه منحرف علوي (أول L صفوف من المصفوفة المثلثية العليا K-by-K)
!>
!>  إذا كان DIRECT = 'F' و STOREV = 'R': V = [V1 V2]
!>
!>     - V2 شبه منحرف سفلي (أول L أعمدة من المصفوفة المثلثية السفلى K-by-K)
!>
!>  إذا كان DIRECT = 'B' و STOREV = 'C': V = [V2]
!>                                         [V1]
!>     - V2 شبه منحرف سفلي (آخر L صفوف من المصفوفة المثلثية السفلى K-by-K)
!>
!>  إذا كان DIRECT = 'B' و STOREV = 'R': V = [V2 V1]
!>
!>     - V2 شبه منحرف علوي (آخر L أعمدة من المصفوفة المثلثية العليا K-by-K)
!>
!>  إذا كان STOREV = 'C' و SIDE = 'L'، فإن V بحجم M-by-K مع V2 بحجم L-by-K.
!>
!>  إذا كان STOREV = 'C' و SIDE = 'R'، فإن V بحجم N-by-K مع V2 بحجم L-by-K.
!>
!>  إذا كان STOREV = 'R' و SIDE = 'L'، فإن V بحجم K-by-M مع V2 بحجم K-by-L.
!>
!>  إذا كان STOREV = 'R' و SIDE = 'R'، فإن V بحجم K-by-N مع V2 بحجم K-by-L.
!>

تم التعريف في السطر 249 من الملف stprfb.f.
