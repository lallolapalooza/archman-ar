subroutine stgexc (logical wantq, logical wantz, integer n, real, dimension( lda, * ) a, integer lda, real, dimension( ldb, * ) b, integer ldb, real, dimension( ldq, * ) q, integer ldq, real, dimension( ldz, * ) z, integer ldz, integer ifst, integer ilst, real, dimension( * ) work, integer lwork, integer info)

STGEXC

الغرض:

!>
!> يعيد STGEXC ترتيب التحلل الشوركي العام الحقيقي لمصفوفة زوج (A,B)
!> باستخدام تحويل تكافؤ متعامد
!>
!>                (A, B) = Q * (A, B) * Z**T,
!>
!> بحيث يتم نقل الكتلة القطرية من (A, B) ذات فهرس الصف IFST إلى الصف ILST.
!>
!> يجب أن تكون (A, B) في شكل شوري شونكي عام حقيقي (كما تم إرجاعه بواسطة SGGES)،
!!> أي أن A تكون ذات كتل مثلثية علوية بكتل قطرية 1x1 و 2x2. B تكون مثلثية علوية.
!>
!> اختياريًا، يتم تحديث مصفوفات Q و Z من متجهات شوركي العامة.
!>
!>        Q(in) * A(in) * Z(in)**T = Q(out) * A(out) * Z(out)**T
!>        Q(in) * B(in) * Z(in)**T = Q(out) * B(out) * Z(out)**T
!>
!>

المعلمات

WANTQ 
!>          WANTQ هو LOGICAL
!>          .TRUE. : تحديث مصفوفة التحويل اليسرى Q؛
!>          .FALSE.: عدم تحديث Q.
!>

WANTZ

!>          WANTZ هو LOGICAL
!>          .TRUE. : تحديث مصفوفة التحويل اليمنى Z؛
!>          .FALSE.: عدم تحديث Z.
!>

N

!>          N هو INTEGER
!>          ترتيب المصفوفات A و B. N >= 0.
!>

A

!>          A هي مصفوفة REAL، البعد (LDA,N)
!>          عند الإدخال، المصفوفة A في شكل شوري شونكي عام حقيقي.
!>          عند الخروج، المصفوفة A المحدثة، مرة أخرى في شكل شوري شونكي عام حقيقي.
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرئيسي للمصفوفة A. LDA >= max(1,N).
!>

B

!>          B هي مصفوفة REAL، البعد (LDB,N)
!>          عند الإدخال، المصفوفة B في شكل شوري شونكي عام حقيقي (A,B).
!>          عند الخروج، المصفوفة B المحدثة، مرة أخرى في شكل شوري شونكي عام حقيقي (A,B).
!>

LDB

!>          LDB هو INTEGER
!>          البعد الرئيسي للمصفوفة B. LDB >= max(1,N).
!>

Q

!>          Q هي مصفوفة REAL، البعد (LDQ,N)
!>          عند الإدخال، إذا كان WANTQ = .TRUE.، فإن المصفوفة المتعامدة Q.
!>          عند الخروج، المصفوفة Q المحدثة.
!>          إذا كان WANTQ = .FALSE.، لا يتم الرجوع إلى Q.
!>

LDQ

!>          LDQ هو INTEGER
!>          البعد الرئيسي للمصفوفة Q. LDQ >= 1.
!>          إذا كان WANTQ = .TRUE.، فإن LDQ >= N.
!>

Z

!>          Z هي مصفوفة REAL، البعد (LDZ,N)
!>          عند الإدخال، إذا كان WANTZ = .TRUE.، فإن المصفوفة المتعامدة Z.
!>          عند الخروج، المصفوفة Z المحدثة.
!>          إذا كان WANTZ = .FALSE.، لا يتم الرجوع إلى Z.
!>

LDZ

!>          LDZ هو INTEGER
!>          البعد الرئيسي للمصفوفة Z. LDZ >= 1.
!>          إذا كان WANTZ = .TRUE.، فإن LDZ >= N.
!>

IFST

!>          IFST هو INTEGER
!>

ILST

!>          ILST هو INTEGER
!>          يحدد إعادة ترتيب الكتل القطرية من (A, B).
!>          يتم نقل الكتلة ذات فهرس الصف IFST إلى الصف ILST، عن طريق سلسلة من
!>          التبادلات بين الكتل المتجاورة.
!>          عند الخروج، إذا كان IFST يشير عند الإدخال إلى الصف الثاني من
!>          كتلة 2x2، يتم تغييره ليشير إلى الصف الأول؛
!>          يشير ILST دائمًا إلى الصف الأول من الكتلة في موضعها النهائي
!>          (والذي قد يختلف عن قيمته المدخلة بمقدار +1 أو -1). 1 <= IFST, ILST <= N.
!>

WORK

!>          WORK هي مصفوفة REAL، البعد (MAX(1,LWORK))
!>          عند الخروج، إذا كان INFO = 0، يعيد WORK(1) قيمة LWORK المثلى.
!>

LWORK

!>          LWORK هو INTEGER
!>          بعد المصفوفة WORK.
!>          LWORK >= 1 عندما N <= 1، وإلا فإن LWORK >= 4*N + 16.
!>
!>          إذا كان LWORK = -1، فسيتم افتراض استعلام عن مساحة العمل؛ تقوم الروتين
!>          بحساب حجم مساحة العمل المثلى فقط، وتعيد
!>          هذه القيمة كإدخال أول لمصفوفة WORK، ولا يصدر XERBLA أي رسالة خطأ
!>          متعلقة بـ LWORK.
!>

INFO

!>          INFO هو INTEGER
!>           =0:  خروج ناجح.
!>           <0:  إذا كان INFO = -i، فإن الوسيط i كان له قيمة غير قانونية.
!>           =1:  ستكون المصفوفة المزدوجة المحولة (A, B) بعيدة جدًا
!>                عن الشكل الشوركي العام؛ المشكلة غير مستقرة.
!>                (A, B) ربما تم إعادة ترتيبها جزئيًا،
!>                ويشير ILST إلى الصف الأول من الموضع الحالي
!>                للكتلة التي يتم نقلها.
!>
