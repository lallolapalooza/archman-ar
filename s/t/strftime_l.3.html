strftime_l(3)

strftime(3) دليل الدوال المكتبية strftime(3)

الاسم

strftime - تنسيق التاريخ والوقت

المكتبة

مكتبة C القياسية (libc، -lc)

الملخص

#include <time.h>
size_t strftime(size_t max;
                char s[restrict max], size_t max,
                const char *restrict format,
                const struct tm *restrict tm);
size_t strftime_l(size_t max;
                char s[restrict max], size_t max,
                const char *restrict format,
                const struct tm *restrict tm,
                locale_t locale);

الوصف

تقوم الدالة strftime() بتنسيق الوقت المفكك tm وفقًا لمواصفات التنسيق format وتضع النتيجة في مصفوفة الأحرف s بحجم max. يتم تعريف بنية الوقت المفكك tm في <time.h>. انظر أيضًا ctime(3).

مواصفات التنسيق هي سلسلة منتهية بصفر وقد تحتوي على تسلسلات أحرف خاصة تسمى مواصفات التحويل، كل منها يبدأ بحرف '%' وينتهي بحرف آخر معروف باسم حرف محدد التحويل. جميع تسلسلات الأحرف الأخرى هي تسلسلات أحرف عادية.

يتم نسخ أحرف تسلسلات الأحرف العادية (بما في ذلك البايت الصفري) حرفيًا من format إلى s. ومع ذلك، يتم استبدال أحرف مواصفات التحويل كما هو موضح في القائمة أدناه. في هذه القائمة، يتم أيضًا عرض الحقل (الحقول) المستخدمة من بنية tm.

%a
الاسم المختصر ليوم الأسبوع وفقًا للموقع الحالي. (محسوب من tm_wday.) (يمكن الحصول على الأسماء المحددة المستخدمة في الموقع الحالي عن طريق استدعاء nl_langinfo(3) مع ABDAY_{1-7} كوسيط.)
%A
الاسم الكامل ليوم الأسبوع وفقًا للموقع الحالي. (محسوب من tm_wday.) (يمكن الحصول على الأسماء المحددة المستخدمة في الموقع الحالي عن طريق استدعاء nl_langinfo(3) مع DAY_{1-7} كوسيط.)
%b
الاسم المختصر للشهر وفقًا للموقع الحالي. (محسوب من tm_mon.) (يمكن الحصول على الأسماء المحددة المستخدمة في الموقع الحالي عن طريق استدعاء nl_langinfo(3) مع ABMON_{1-12} كوسيط.)
%B
الاسم الكامل للشهر وفقًا للموقع الحالي. (محسوب من tm_mon.) (يمكن الحصول على الأسماء المحددة المستخدمة في الموقع الحالي عن طريق استدعاء nl_langinfo(3) مع MON_{1-12} كوسيط.)
%c
تمثيل التاريخ والوقت المفضل للموقع الحالي. (يمكن الحصول على التنسيق المحدد المستخدم في الموقع الحالي عن طريق استدعاء nl_langinfo(3) مع D_T_FMT كوسيط لمواصفات التحويل %c، ومع ERA_D_T_FMT لمواصفات التحويل %Ec.) (في موقع POSIX، هذا يعادل %a %b %e %H:%M:%S %Y.)
%C
رقم القرن (السنة/100) كرقم صحيح مكون من رقمين. (SU) (مواصفات التحويل %EC تتوافق مع اسم العصر.) (محسوب من tm_year.)
%d
يوم الشهر كرقم عشري (النطاق 01 إلى 31). (محسوب من tm_mday.)
%D
يعادل %m/%d/%y. (يياااخ - للأمريكيين فقط. يجب ملاحظة أن بلدان أخرى تستخدم %d/%m/%y بشكل شائع. هذا يعني أنه في السياق الدولي، هذا التنسيق غامض ويجب عدم استخدامه.) (SU)
%e
مثل %d، يوم الشهر كرقم عشري، ولكن يتم استبدال الصفر البادئ بمسافة. (SU) (محسوب من tm_mday.)
%E
معدّل: استخدم تنسيقًا بديلاً ("مبني على العصر")؛ انظر أدناه. (SU)
%F
يعادل %Y-%m-%d (تنسيق التاريخ ISO 8601). (C99)
%G
السنة المبنية على الأسبوع ISO 8601 (انظر الملاحظات) مع القرن كرقم عشري. السنة المكونة من 4 أرقام المقابلة لرقم الأسبوع ISO (انظر %V). هذا له نفس التنسيق والقيمة مثل %Y، باستثناء أنه إذا كان رقم الأسبوع ISO ينتمي إلى السنة السابقة أو التالية، فسيتم استخدام تلك السنة بدلاً من ذلك. (TZ) (محسوب من tm_year، tm_yday، و tm_wday.)
%g
مثل %G، ولكن بدون القرن، أي، بسنة مكونة من رقمين (00-99). (TZ) (محسوب من tm_year، tm_yday، و tm_wday.)
%h
يعادل %b. (SU)
%H
الساعة كرقم عشري باستخدام ساعة 24 ساعة (النطاق 00 إلى 23). (محسوب من tm_hour.)
%I
الساعة كرقم عشري باستخدام ساعة 12 ساعة (النطاق 01 إلى 12). (محسوب من tm_hour.)
%j
يوم السنة كرقم عشري (النطاق 001 إلى 366). (محسوب من tm_yday.)
%k
الساعة (ساعة 24) كرقم عشري (النطاق 0 إلى 23)؛ الأرقام الفردية يسبقها فراغ. (انظر أيضًا %H.) (محسوب من tm_hour.) (TZ)
%l
الساعة (ساعة 12) كرقم عشري (النطاق 1 إلى 12)؛ الأرقام الفردية يسبقها فراغ. (انظر أيضًا %I.) (محسوب من tm_hour.) (TZ)
%m
الشهر كرقم عشري (النطاق 01 إلى 12). (محسوب من tm_mon.)
%M
الدقيقة كرقم عشري (النطاق 00 إلى 59). (محسوب من tm_min.)
%n
حرف سطر جديد. (SU)
%O
معدّل: استخدم رموزًا رقمية بديلة؛ انظر أدناه. (SU)
%p
إما "AM" أو "PM" وفقًا لقيمة الوقت المعطاة، أو السلاسل المقابلة للموقع الحالي. تعتبر الظهيرة "PM" ومنتصف الليل "AM". (محسوب من tm_hour.) (يمكن الحصول على التمثيلات السلسلة المحددة لـ "AM" و "PM" في الموقع الحالي عن طريق استدعاء nl_langinfo(3) مع AM_STR و PM_STR، على التوالي.)
%P
مثل %p ولكن بأحرف صغيرة: "am" أو "pm" أو سلسلة مقابلة للموقع الحالي. (محسوب من tm_hour.) (GNU)
%r
الوقت بتنسيق صباحًا/مساءً. (SU) (يمكن الحصول على التنسيق المحدد المستخدم في الموقع الحالي عن طريق استدعاء nl_langinfo(3) مع T_FMT_AMPM كوسيط.) (في موقع POSIX، هذا يعادل %I:%M:%S %p.)
%R
الوقت بتنسيق 24 ساعة (%H:%M). (SU) للحصول على إصدار يتضمن الثواني، انظر %T أدناه.
%s
عدد الثواني منذ نقطة البداية، 1970-01-01 00:00:00 +0000 (UTC). (TZ) (محسوب من mktime(tm).)
%S
الثانية كرقم عشري (النطاق 00 إلى 60). (النطاق يصل إلى 60 للسماح بالثواني الكبيسة العرضية.) (محسوب من tm_sec.)
%t
حرف جدولة (tab). (SU)
%T
الوقت بتنسيق 24 ساعة (%H:%M:%S). (SU)
%u
يوم الأسبوع كرقم عشري، النطاق 1 إلى 7، الاثنين هو 1. انظر أيضًا %w. (محسوب من tm_wday.) (SU)
%U
رقم الأسبوع من السنة الحالية كرقم عشري، النطاق 00 إلى 53، يبدأ بالأحد الأول كيوم الأسبوع 01. انظر أيضًا %V و %W. (محسوب من tm_yday و tm_wday.)
%V
رقم الأسبوع ISO 8601 (انظر الملاحظات) من السنة الحالية كرقم عشري، النطاق 01 إلى 53، حيث الأسبوع 1 هو الأسبوع الأول الذي يحتوي على 4 أيام على الأقل في السنة الجديدة. انظر أيضًا %U و %W. (محسوب من tm_year، tm_yday، و tm_wday.) (SU)
%w
يوم الأسبوع كرقم عشري، النطاق 0 إلى 6، الأحد هو 0. انظر أيضًا %u. (محسوب من tm_wday.)
%W
رقم الأسبوع من السنة الحالية كرقم عشري، النطاق 00 إلى 53، يبدأ بالاثنين الأول كيوم الأسبوع 01. (محسوب من tm_yday و
>tm_wday.)
%x
التمثيل المفضل للتاريخ للموقع المحلي الحالي بدون الوقت. (يمكن الحصول على التنسيق المحدد المستخدم في الموقع المحلي الحالي عن طريق استدعاء nl_langinfo(3) مع D_FMT كوسيط للمواصفات التحويلية %x، ومع ERA_D_FMT للمواصفات التحويلية %Ex.) (في الموقع المحلي POSIX، هذا يعادل %m/%d/%y.)
%X
التمثيل المفضل للوقت للموقع المحلي الحالي بدون التاريخ. (يمكن الحصول على التنسيق المحدد المستخدم في الموقع المحلي الحالي عن طريق استدعاء nl_langinfo(3) مع T_FMT كوسيط للمواصفات التحويلية %X، ومع ERA_T_FMT للمواصفات التحويلية %EX.) (في الموقع المحلي POSIX، هذا يعادل %H:%M:%S.)
%y
السنة كرقم عشري بدون قرن (النطاق 00 إلى 99). (المواصفات التحويلية %Ey تتوافق مع السنة منذ بداية العصر الذي تم تعريفه بواسطة المواصفات التحويلية %EC.) (محسوبة من tm_year)
%Y
السنة كرقم عشري بما في ذلك القرن. (المواصفات التحويلية %EY تتوافق مع التمثيل البديل الكامل للسنة.) (محسوبة من tm_year)
%z
فرق التوقيت العددي +hhmm أو -hhmm (أي، إزاحة الساعة والدقيقة عن التوقيت العالمي المنسق UTC). (SU)
%Z
اسم أو اختصار المنطقة الزمنية.
%+
التاريخ والوقت بتنسيق date(1). (TZ) (غير مدعوم في glibc2.)
%%
حرف '%' حرفي.

يمكن تعديل بعض مواصفات التحويل عن طريق وضع معدل E أو O قبل حرف محدد التحويل للإشارة إلى استخدام تنسيق بديل. إذا كان التنسيق أو المواصفات البديلة غير موجودة للموقع المحلي الحالي، فسيكون السلوك كما لو تم استخدام مواصفات التحويل غير المعدلة. (SU) يذكر مواصفات Single UNIX مواصفات %Ec، %EC، %Ex، %EX، %Ey، %EY، %Od، %Oe، %OH، %OI، %Om، %OM، %OS، %Ou، %OU، %OV، %Ow، %OW، %Oy، حيث يكون تأثير المعدل O هو استخدام رموز رقمية بديلة (مثل الأرقام الرومانية)، وتأثير المعدل E هو استخدام تمثيل بديل يعتمد على الموقع المحلي. يمكن الحصول على القواعد التي تحكم تمثيل التاريخ باستخدام المعدل E عن طريق توفير ERA كوسيط لـ nl_langinfo(3). أحد الأمثلة على هذه الأشكال البديلة هو مخطط التقويم الياباني للعصر في الموقع المحلي ja_JP glibc.

strftime_l() تعادل strftime()، باستثناء أنها تستخدم الموقع المحلي المحدد بدلاً من الموقع المحلي الحالي. السلوك غير محدد إذا كان الموقع المحلي غير صالح أو LC_GLOBAL_LOCALE.

قيمة الإرجاع

بشرط ألا يتجاوز طول السلسلة الناتجة، بما في ذلك البايت الفارغ النهائي، max بايت، تُرجع strftime() عدد البايتات (باستثناء البايت الفارغ النهائي) التي تم وضعها في المصفوفة s. إذا كان طول السلسلة الناتجة (بما في ذلك البايت الفارغ النهائي) سيتجاوز max بايت، فإن strftime() تُرجع 0، ومحتويات المصفوفة غير محددة.

لاحظ أن قيمة الإرجاع 0 لا تشير بالضرورة إلى خطأ. على سبيل المثال، في العديد من المواقع المحلية، تنتج %p سلسلة فارغة. سلسلة format فارغة ستنتج أيضًا سلسلة فارغة.

البيئة

يتم استخدام متغيرات البيئة TZ و LC_TIME.

هناك تضمينات صارمة بين مجموعة التحويلات المقدمة في ANSI C (غير معلمة) ، وتلك المقدمة في مواصفات Single UNIX (معلمة SU) ، وتلك المقدمة في حزمة Olson للتوقيت (معلمة TZ) ، وتلك المقدمة في glibc (معلمة GNU) ، باستثناء أن %+ غير مدعوم في glibc2. من ناحية أخرى، لدى glibc2 العديد من الامتدادات الإضافية. POSIX.1 تشير فقط إلى ANSI C؛ POSIX.2 تصف ضمن date(1) العديد من الامتدادات التي يمكن أن تنطبق على strftime() أيضًا. المواصفات %F موجودة في C99 و POSIX.1-2001.

في SUSv2 ، سمحت المواصفات %S بنطاق 00 إلى 61 ، للسماح بالإمكانية النظرية لدقيقة تتضمن ثانية كبيسة مزدوجة (لم تكن هناك أبدًا دقيقة من هذا القبيل).

ملاحظات

تواريخ الأسبوع ISO 8601

%G و %g و %V تعطي قيمًا محسوبة من السنة القائمة على الأسبوع المحددة بواسطة معيار ISO 8601. في هذا النظام، تبدأ الأسابيع يوم الاثنين، ويتم ترقيمها من 01 ، للأسبوع الأول ، حتى 52 أو 53 ، للأسبوع الأخير. الأسبوع 1 هو الأسبوع الأول الذي تقع فيه أربعة أيام أو أكثر ضمن العام الجديد (أو ، بعبارة أخرى ، الأسبوع 01 هو: الأسبوع الأول من العام الذي يحتوي على يوم خميس؛ أو ، الأسبوع الذي يحتوي على 4 يناير فيه). عندما تقع ثلاثة أيام أو أقل من الأسبوع التقويمي الأول من العام الجديد ضمن ذلك العام، فإن نظام الأسبوع المعتمد على ISO 8601 يعتبر هذه الأيام جزءًا من الأسبوع 52 أو 53 من العام السابق. على سبيل المثال، 1 يناير 2010 هو يوم جمعة، مما يعني أن ثلاثة أيام فقط من هذا الأسبوع التقويمي تقع في عام 2010. وبالتالي، فإن نظام الأسبوع المعتمد على ISO 8601 يعتبر هذه الأيام جزءًا من الأسبوع 53 (%V) من عام 2009 (%G)؛ يبدأ الأسبوع 01 من عام 2010 حسب ISO 8601 يوم الاثنين 4 يناير 2010. وبالمثل، يعتبر اليومان الأولان من يناير 2011 جزءًا من الأسبوع 52 من عام 2010.

ملاحظات glibc

يوفر glibc بعض الامتدادات لمواصفات التحويل. (هذه الامتدادات غير محددة في POSIX.1-2001 ، ولكن بعض الأنظمة الأخرى توفر ميزات مماثلة.) بين حرف '%' وحرف محدد التحويل ، يمكن تحديد علامة اختيارية و عرض للحقل. (هذه تسبق معدلات E أو O ، إذا كانت موجودة.)

تسمح أحرف العلامات التالية:

_
(شرطة سفلية) املأ سلسلة رقمية بمسافات.
-
(شرطة) لا تملأ سلسلة رقمية.
0
املأ سلسلة رقمية بالأصفار حتى لو كان حرف محدد التحويل يستخدم حشو المسافات افتراضيًا.
^
حول الأحرف الأبجدية في السلسلة الناتجة إلى أحرف كبيرة.
#
تبديل حالة السلسلة الناتجة. (تعمل هذه العلامة فقط مع أحرف محددات التحويل معينة ، ومن بينها ، لا تكون مفيدة حقًا إلا مع %Z.)

يمكن أن يتبع مواصف محدد عرض عشري اختياري (ربما غائب) للعرض. إذا كان الحجم الطبيعي للحقل أصغر من هذا العرض، فسيتم ملء السلسلة الناتجة (على اليسار) بالعرض المحدد.

أخطاء

إذا تجاوز طول السلسلة الناتجة max بايت، فلن يتم تعيين errno. هذا يجعل من المستحيل التمييز بين حالة الخطأ هذه والحالات التي تنتج فيها سلسلة format الشرعية مخرجات بطول صفر. POSIX.1-2001 لا تحدد أي إعدادات errno لـ strftime().

بعض إصدارات gcc(1) التي بها أخطاء تشكو من استخدام %c: warning: `%c' yields only last 2 digits of year in some locales. بالطبع يتم تشجيع المبرمجين على استخدام %c، لأنها تعطي التمثيل المفضل للتاريخ والوقت. يواجه المرء جميع أنواع التعتيمات الغريبة للتحايل على مشكلة gcc(1) هذه. إحدى الطرق النظيفة نسبيًا هي إضافة دالة وسيطة


size_t
my_strftime(char *s, size_t max, const char *fmt,
            const struct tm *tm)
{
    return strftime(s, max, fmt, tm);
}

في الوقت الحاضر، يوفر gcc(1) الخيار -Wno-format-y2k لمنع التحذير، لذلك لم تعد الحلول البديلة المذكورة أعلاه مطلوبة.

أمثلة

تنسيق التاريخ المتوافق مع RFC 2822 (مع موقع محلي باللغة الإنجليزية لـ %a و %b)


"%a, %d %b %Y %T %z"

تنسيق التاريخ المتوافق مع RFC 822 (مع موقع محلي باللغة الإنجليزية لـ %a و %b)


"%a, %d %b %y %T %z"

برنامج مثال

يمكن استخدام البرنامج أدناه للتجريب مع strftime().

بعض الأمثلة على السلسلة الناتجة التي أنتجتها طريقة glibc لتنفيذ strftime() هي كما يلي:


$ ./a.out '%m'
Result string is "11"
$ ./a.out '%5m'
Result string is "00011"
$ ./a.out '%_5m'
Result string is "   11"

#include <stdlib.h>
#include <time.h>
int
main(int argc, char *argv[])
{
    char outstr[200];
    time_t t;
    struct tm *tmp;
    if (argc != 2) {
        fprintf(stderr, "Usage
