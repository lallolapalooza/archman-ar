النتائج

type ('a, 'e) t = ('a, 'e) result =
| Ok of 'a
| Error of 'e

نوع قيم النتائج. إما قيمة Ok v أو خطأ Error e.

val ok : 'a -> ('a, 'e) result

ok v هي Ok v.

val error : 'e -> ('a, 'e) result

error e هي Error e.

val value : ('a, 'e) result -> default:'a -> 'a

value r ~default هي v إذا كان r هو Ok v و default بخلاف ذلك.

val get_ok : ('a, 'e) result -> 'a

get_ok r هي v إذا كان r هو Ok v وتثير استثناءً بخلاف ذلك.

يثير استثناء Invalid_argument إذا كان r هو Error _.

val get_ok' : ('a, string) result -> 'a

get_ok' مثل Result.get_ok ولكن في حالة الخطأ يستخدم رسالة الخطأ لإثارة Invalid_argument.

val get_error : ('a, 'e) result -> 'e

get_error r هي e إذا كان r هو Error e وتثير استثناءً بخلاف ذلك.

يثير استثناء Invalid_argument إذا كان r هو Ok _.

val error_to_failure : ('a, string) result -> 'a

error_to_failure r هي v إذا كان r هو Ok v ويثير Failure e إذا كان r هو Error e.

val bind : ('a, 'e) result -> ('a -> ('b, 'e) result) -> ('b, 'e) result

bind r f هي f v إذا كان r هو Ok v و r إذا كان r هو Error _.

val join : (('a, 'e) result, 'e) result -> ('a, 'e) result

join rr هي r إذا كان rr هو Ok r و rr إذا كان rr هو Error _.

val map : ('a -> 'b) -> ('a, 'e) result -> ('b, 'e) result

map f r هي Ok (f v) إذا كان r هو Ok v و r إذا كان r هو Error _.

val product : ('a, 'e) result -> ('b, 'e) result -> ('a * 'b, 'e) result

product r0 r1 هي Ok (v0, v1) إذا كان r0 هو Ok v0 و r1 هو Ok v2 وبخلاف ذلك تعيد خطأ r0، إن وجد، أو خطأ r1.

val map_error : ('e -> 'f) -> ('a, 'e) result -> ('a, 'f) result

map_error f r هي Error (f e) إذا كان r هو Error e و r إذا كان r هو Ok _.

val fold : ok:('a -> 'c) -> error:('e -> 'c) -> ('a, 'e) result -> 'c

fold ~ok ~error r هي ok v إذا كان r هو Ok v و error e إذا كان r هو Error e.

val retract : ('a, 'a) result -> 'a

retract r هي v إذا كان r هو Ok v أو Error v.

المدققات والمقارنات

val is_ok : ('a, 'e) result -> bool

is_ok r هي true إذا وفقط إذا كان r هو Ok _.

val is_error : ('a, 'e) result -> bool

is_error r هي true إذا وفقط إذا كان r هو Error _.

val equal : ok:('a -> 'a -> bool) -> error:('e -> 'e -> bool) -> ('a, 'e) result -> ('a, 'e) result -> bool

equal ~ok ~error r0 r1 تختبر المساواة بين r0 و r1 باستخدام ok و error لمقارنة القيم المغلفة بواسطة Ok _ و Error _ على التوالي.

val compare : ok:('a -> 'a -> int) -> error:('e -> 'e -> int) -> ('a, 'e) result -> ('a, 'e) result -> int

compare ~ok ~error r0 r1 ترتب r0 و r1 تمامًا باستخدام ok و error لمقارنة القيم المغلفة بواسطة Ok _ و Error _ على التوالي. قيم Ok _ أصغر من قيم Error _.

التحويل

val to_option : ('a, 'e) result -> 'a option

to_option r هي r كخيار، حيث يتم تحويل Ok v إلى Some v و Error _ إلى None.

val to_list : ('a, 'e) result -> 'a list

to_list r هي [v] إذا كان r هو Ok v و [] بخلاف ذلك.

val to_seq : ('a, 'e) result -> 'a Seq.t

to_seq r هي r كتسلسل. Ok v هو التسلسل المفرد الذي يحتوي على v و Error _ هو التسلسل الفارغ.
