val blit : src:bytes -> src_pos:int -> dst:bytes -> dst_pos:int -> len:int -> unit

blit ~src ~src_pos ~dst ~dst_pos ~len ينسخ len بايت من تسلسل البايت src، بدءًا من الفهرس src_pos، إلى تسلسل البايت dst، بدءًا من الفهرس dst_pos. يعمل بشكل صحيح حتى لو كان src و dst نفس تسلسل البايت، وتتداخل النطاقات المصدر والوجهة.

تحويلات غير آمنة (للمستخدمين المتقدمين)

يصف هذا القسم وظائف تحويل غير آمنة ومنخفضة المستوى بين bytes و string . لا تقوم بنسخ البيانات الداخلية؛ عند استخدامها بشكل غير صحيح، يمكنها كسر ثابت عدم القابلية للتغيير على السلاسل. وهي متاحة لمؤلفي المكتبات الخبراء، ولكن لمعظم الأغراض يجب عليك استخدام BytesLabels.to_string و BytesLabels.of_string الصحيحين دائمًا بدلاً من ذلك.

val unsafe_to_string : bytes -> string

تحويل غير آمن لتسلسل بايت إلى سلسلة.

للتفكير في استخدام unsafe_to_string ، من المفيد النظر في نظام "الملكية". قطعة من الكود التي تتعامل مع بعض البيانات "تمتلكها"؛ هناك عدة أوضاع ملكية منفصلة، بما في ذلك:

-ملكية فريدة: يمكن الوصول إلى البيانات وتعديلها

-ملكية مشتركة: البيانات لها عدة مالكين، قد يصلون إليها فقط، ولا يعدلونها.

الملكية الفريدة خطية: تمرير البيانات إلى قطعة أخرى من الكود يعني التخلي عن الملكية (لا يمكننا كتابة البيانات مرة أخرى). قد يقرر مالك فريد جعل البيانات مشتركة (التخلي عن حقوق التعديل عليها)، ولكن لا يمكن أن تصبح البيانات المشتركة مملوكة بشكل فريد مرة أخرى.

unsafe_to_string s يمكن استخدامها فقط عندما يمتلك المتصل تسلسل البايت s -- إما بشكل فريد أو كبيانات ثابتة مشتركة. يتخلى المتصل عن ملكية s ، ويكتسب ملكية السلسلة المعادة.

هناك حالتان استخدام صالحتان تحترمان هذا النظام من الملكية:

1. إنشاء سلسلة عن طريق تهيئة وتعديل تسلسل بايت لا يتغير أبدًا بعد اكتمال التهيئة.

let string_init len f : string =
  let s = Bytes.create len in
  for i = 0 to len - 1 do Bytes.set s i (f i) done;
  Bytes.unsafe_to_string s

هذه الوظيفة آمنة لأن تسلسل البايت s لن يتم الوصول إليه أو تعديله أبدًا بعد استدعاء unsafe_to_string . تتخلى وظيفة string_init عن ملكية s ، وتعيد ملكية السلسلة الناتجة إلى المتصل.

لاحظ أنه سيكون غير آمن إذا تم تمرير s كمعامل إضافي للوظيفة f حيث يمكن أن تتسرب بهذه الطريقة ويتم تعديلها في المستقبل -- ستتخلى string_init عن ملكية s لتمريرها إلى f ، ولا يمكنها استدعاء unsafe_to_string بأمان.

لقد قدمنا وظائف String.init و String.map و String.mapi لتغطية معظم حالات بناء سلاسل جديدة. يجب تفضيلها على to_string أو unsafe_to_string كلما كان ذلك ممكنًا.

2. إعارة ملكية مؤقتة لتسلسل بايت لوظيفة تتوقع سلسلة مملوكة بشكل فريد وتعيد الملكية، حتى نتمكن من تعديل التسلسل مرة أخرى بعد انتهاء الاستدعاء.

let bytes_length (s : bytes) =
  String.length (Bytes.unsafe_to_string s)

في حالة الاستخدام هذه، لا نعد بأن s لن يتم تعديلها أبدًا بعد استدعاء bytes_length s . تستعير وظيفة String.length ملكية مؤقتة لتسلسل البايت (وتراها كسلسلة string )، ولكنها تعيد هذه الملكية إلى المتصل، الذي يمكنه افتراض أن s لا تزال تسلسل بايت صالحًا بعد الاستدعاء. لاحظ أن هذا صحيح فقط لأننا نعلم أن String.length لا تلتقط معاملها -- يمكن أن تتسرب عبر قناة جانبية مثل مُجمّع الذاكرة المؤقتة.

لا يجوز للمتصل تعديل s أثناء استعارة السلسلة (لقد تخلى مؤقتًا عن الملكية). يؤثر هذا على البرامج المتزامنة، ولكنه يؤثر أيضًا على الوظائف عالية الرتبة: إذا أعادت String.length إغلاقًا ليتم استدعاؤه لاحقًا، فلا ينبغي تعديل s حتى يتم تطبيق هذا الإغلاق بالكامل ويعيد الملكية.

val unsafe_of_string : string -> bytes

تحويل غير آمن لسلسلة مشتركة إلى تسلسل بايت لا ينبغي تعديله.

نفس نظام الملكية الذي يجعل unsafe_to_string صحيحًا ينطبق على unsafe_of_string : يمكنك استخدامه إذا كنت مالك قيمة string ، وسوف تمتلك bytes المعادة بنفس الوضع.

من الناحية العملية، من الصعب للغاية التفكير بشكل صحيح في الملكية الفريدة للسلاسل. يجب عليك دائمًا افتراض أن السلاسل مشتركة، ولا تمتلكها أبدًا بشكل فريد.

على سبيل المثال، يتم مشاركة الأحرف النصية للسلاسل ضمنيًا بواسطة المترجم، لذلك لا تمتلكها أبدًا بشكل فريد.

let incorrect = Bytes.unsafe_of_string "hello"
let s = Bytes.of_string "hello"

الإعلان الأول غير صحيح، لأن الحرف النصي للسلسلة "hello" يمكن مشاركته بواسطة المترجم مع أجزاء أخرى من البرنامج، وتعديل incorrect هو خطأ. يجب عليك دائمًا استخدام الإصدار الثاني، الذي يقوم بنسخ وهو صحيح.

افتراض الملكية الفريدة للسلاسل التي ليست أحرفًا نصية، ولكنها (جزئيًا) مبنية من أحرف نصية للسلاسل، هو أيضًا غير صحيح. على سبيل المثال، تعديل unsafe_of_string ("foo" ^ s) قد يعدل السلسلة المشتركة "foo" -- بافتراض تمثيل يشبه الحبل للسلاسل. بشكل عام، الوظائف التي تعمل على السلاسل ستفترض الملكية المشتركة، فهي لا تحافظ على الملكية الفريدة. لذلك، من غير الصحيح افتراض الملكية الفريدة لنتيجة unsafe_of_string

تسلسلات البايتات وأمان التزامن

يجب توخي الحذر عند الوصول المتزامن إلى تسلسلات البايتات من مجالات متعددة: لن يؤدي الوصول إلى تسلسل بايتات أبدًا إلى تعطل البرنامج، ولكن الوصول غير المتزامن قد يؤدي إلى نتائج مفاجئة (غير متسقة تسلسليًا).

الذرية

أي عملية على تسلسل بايتات تتضمن أكثر من بايت واحد ليست ذرية. يشمل ذلك التكرار والمسح.

على سبيل المثال، ضع في اعتبارك البرنامج التالي:

let size = 100_000_000
let b = Bytes.make size ' '
let update b f ()  =
  Bytes.iteri (fun i x -> Bytes.set b i (Char.chr (f (Char.code x)))) b
let d1 = Domain.spawn (update b (fun x -> x + 1))
let d2 = Domain.spawn (update b (fun x -> 2 * x + 1))
let () = Domain.join d1; Domain.join d2
قد يحتوي تسلسل البايتات b على خليط غير حتمي من القيم '!' و 'A' و 'B' و 'C'.

بعد تنفيذ هذا الرمز، سيكون كل بايت في التسلسل b إما '!' أو 'A' أو 'B' أو 'C'. إذا كانت الذرية مطلوبة، فيجب على المستخدم تنفيذ المزامنة الخاصة به (على سبيل المثال، باستخدام Mutex.t).

سباقات البيانات

إذا كان مجالان فقط يصلان إلى أجزاء منفصلة من تسلسل بايتات، فإن السلوك المرصود يعادل بعض التداخل التسلسلي للعمليات من المجالين.

يُقال أن سباق البيانات يحدث عندما يصل مجالان إلى نفس البايت دون مزامنة وعلى الأقل أحد الوصولين هو عملية كتابة. في غياب سباقات البيانات، فإن السلوك المرصود يعادل بعض التداخل التسلسلي للعمليات من مجالات مختلفة.

حيثما أمكن، يجب تجنب سباقات البيانات باستخدام المزامنة للتوسط في الوصول إلى عناصر التسلسل.

بالفعل، في وجود سباقات البيانات، لن تتعطل البرامج ولكن السلوك المرصود قد لا يكون مكافئًا لأي تداخل تسلسلي للعمليات من مجالات مختلفة. ومع ذلك، حتى في وجود سباقات البيانات، ستعيد عملية القراءة قيمة كتابة سابقة إلى هذا الموقع.

الوصول بأحجام مختلطة

نقطة دقيقة أخرى هي أنه إذا تضمن سباق بيانات عمليات كتابة وقراءة بأحجام مختلطة إلى نفس الموقع، فإن ترتيب ملاحظة عمليات الكتابة والقراءة هذه بواسطة المجالات غير محدد. على سبيل المثال، يكتب الرمز التالي بشكل تسلسلي عددًا صحيحًا مكونًا من 32 بت و char إلى نفس الفهرس

let b = Bytes.make 10 '\000'
let d1 = Domain.spawn (fun () -> Bytes.set_int32_ne b 0 100; b.[0] <- 'd' )

في هذا الموقف، لا يُضمن أن مجالًا يلاحظ كتابة 'd' إلى b. 0 سيلاحظ أيضًا الكتابة إلى الفهارس 1 أو 2 أو 3.
