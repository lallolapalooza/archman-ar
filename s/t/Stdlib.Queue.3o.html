وصول غير متزامن

قد يؤدي الوصول غير المتزامن إلى قائمة انتظار إلى حالة غير صالحة. وبالتالي، يجب مزامنة الوصول المتزامن إلى قوائم الانتظار (على سبيل المثال، باستخدام Mutex.t ).

type !'a t

نوع قوائم الانتظار التي تحتوي على عناصر من النوع 'a .

exception Empty

يتم رفعه عندما يتم تطبيق Queue.take أو Queue.peek على قائمة انتظار فارغة.

val create : unit -> 'a t

إرجاع قائمة انتظار جديدة، فارغة مبدئيًا.

val add : 'a -> 'a t -> unit

add x q تضيف العنصر x إلى نهاية قائمة الانتظار q .

val push : 'a -> 'a t -> unit

push هو اسم مرادف لـ add .

val take : 'a t -> 'a

take q تزيل وتعيد العنصر الأول في قائمة الانتظار q ، أو ترفع Queue.Empty إذا كانت قائمة الانتظار فارغة.

val take_opt : 'a t -> 'a option

take_opt q تزيل وتعيد العنصر الأول في قائمة الانتظار q ، أو تعيد None إذا كانت قائمة الانتظار فارغة.

val pop : 'a t -> 'a

pop هو اسم مرادف لـ take .

val peek : 'a t -> 'a

peek q تعيد العنصر الأول في قائمة الانتظار q ، دون إزالته من قائمة الانتظار، أو ترفع Queue.Empty إذا كانت قائمة الانتظار فارغة.

val peek_opt : 'a t -> 'a option

peek_opt q تعيد العنصر الأول في قائمة الانتظار q ، دون إزالته من قائمة الانتظار، أو تعيد None إذا كانت قائمة الانتظار فارغة.

منذ 4.08

val top : 'a t -> 'a

top هو اسم مرادف لـ peek .

val drop : 'a t -> unit

drop q تزيل العنصر الأول في قائمة الانتظار q ، أو ترفع Queue.Empty إذا كانت قائمة الانتظار فارغة.

منذ 5.3

val clear : 'a t -> unit

تجاهل جميع العناصر من قائمة انتظار.

val copy : 'a t -> 'a t

إرجاع نسخة من قائمة الانتظار المعطاة.

val is_empty : 'a t -> bool

إرجاع true إذا كانت قائمة الانتظار المعطاة فارغة، false بخلاف ذلك.

val length : 'a t -> int

إرجاع عدد العناصر في قائمة انتظار.

val iter : ('a -> unit) -> 'a t -> unit

iter f q تطبق f بالتناوب على جميع عناصر q ، من الأقل إدخالًا مؤخرًا إلى الأكثر إدخالًا مؤخرًا. تظل قائمة الانتظار نفسها دون تغيير.

val fold : ('acc -> 'a -> 'acc) -> 'acc -> 'a t -> 'acc

fold f accu q يعادل List.fold_left f accu l ، حيث l هي قائمة عناصر q . تظل قائمة الانتظار دون تغيير.

val transfer : 'a t -> 'a t -> unit

transfer q1 q2 تضيف جميع عناصر q1 إلى نهاية قائمة الانتظار q2 ، ثم تمسح q1 . إنها تعادل التسلسل iter (fun x -> add x q2) q1; clear q1 ، ولكنها تعمل في وقت ثابت.

البحث خلال الرسم البياني

لمثال أكثر تفصيلاً، يتمثل الاستخدام التقليدي للخوارزميات لقوائم الانتظار في تنفيذ BFS (بحث في العرض الأول) عبر رسم بياني.

     type graph = {
       edges: (int, int list) Hashtbl.t
     }
    (* البحث في الرسم البياني [g] باستخدام BFS، بدءًا من العقدة [start].
       يعيد العقدة الأولى التي تحقق [p]، أو [None] إذا
       لم تحقق أي عقدة يمكن الوصول إليها من [start] الشرط [p].
    *)
    let search_for ~(g:graph) ~(start:int) (p:int -> bool) : int option =
      let to_explore = Queue.create() in
      let explored = Hashtbl.create 16 in
      Queue.push start to_explore;
      let rec loop () =
        if Queue.is_empty to_explore then None
        else
          (* العقدة التي سيتم استكشافها *)
          let node = Queue.pop to_explore in
          explore_node node
      and explore_node node =
        if not (Hashtbl.mem explored node) then (
          if p node then Some node (* تم العثور عليها *)
          else (
            Hashtbl.add explored node ();
            let children =
              Hashtbl.find_opt g.edges node
              |> Option.value ~default:[]
            in
            List.iter (fun child -> Queue.push child to_explore) children;
            loop()
          )
        ) else loop()
      in
      loop()
    (* رسم بياني نموذجي *)
    let my_graph: graph =
      let edges =
        List.to_seq [
          1, [2;3];
          2, [10; 11];
          3, [4;5];
          5, [100];
          11, [0; 20];
        ]
        |> Hashtbl.of_seq
      in {edges}
    # search_for ~g:my_graph ~start:1 (fun x -> x = 30)
    - : int option = None
    # search_for ~g:my_graph ~start:1 (fun x -> x >= 15)
    - : int option = Some 20
    # search_for ~g:my_graph ~start:1 (fun x -> x >= 50)
    - : int option = Some 100
