توثيق الوظيفة/الروتين الفرعي

subroutine stpmqrt (character side, character trans, integer m, integer n, integer k, integer l, integer nb, real, dimension( ldv, * ) v, integer ldv, real, dimension( ldt, * ) t, integer ldt, real, dimension( lda, * ) a, integer lda, real, dimension( ldb, * ) b, integer ldb, real, dimension( * ) work, integer info)

STPMQRT

الغرض:

!>
!> STPMQRT يطبق مصفوفة متعامدة حقيقية Q تم الحصول عليها من
!>  مُعرِّف كتلة حقيقي H على مصفوفة حقيقية عامة C، والتي تتكون من كتلتين A و B.
!>

المعلمات

SIDE 
!>          SIDE is CHARACTER*1
!>          = 'L': تطبيق Q أو Q^T من اليسار؛
!>          = 'R': تطبيق Q أو Q^T من اليمين.
!>

TRANS

!>          TRANS is CHARACTER*1
!>          = 'N':  بدون تبديل، تطبيق Q؛
!>          = 'T':  تبديل، تطبيق Q^T.
!>

M

!>          M is INTEGER
!>          عدد صفوف المصفوفة B. M >= 0.
!>

N

!>          N is INTEGER
!>          عدد أعمدة المصفوفة B. N >= 0.
!>

K

!>          K is INTEGER
!>          عدد المُعرِّفات الأولية التي يُعرِّف حاصل ضربها
!>          المصفوفة Q.
!>

L

!>          L is INTEGER
!>          رتبة الجزء شبه المنحرف من V.
!>          K >= L >= 0.  انظر التفاصيل الإضافية.
!>

NB

!>          NB is INTEGER
!>          حجم الكتلة المستخدم لتخزين T.  K >= NB >= 1.
!>          يجب أن تكون هذه هي نفس قيمة NB المستخدمة لتوليد T
!>          في CTPQRT.
!>

V

!>          V is REAL array, dimension (LDV,K)
!>          يجب أن يحتوي العمود i على المتجه الذي يُعرِّف
!>          المُعرِّف الأولي H(i)، لـ i = 1,2,...,k، كما تم إرجاعه بواسطة
!>          CTPQRT في B.  انظر التفاصيل الإضافية.
!>

LDV

!>          LDV is INTEGER
!>          البعد الرائد للمصفوفة V.
!>          إذا كان SIDE = 'L'، فإن LDV >= max(1,M)؛
!>          إذا كان SIDE = 'R'، فإن LDV >= max(1,N).
!>

T

!>          T is REAL array, dimension (LDT,K)
!>          العوامل المثلثية العلوية للمُعرِّفات الكتلية
!>          كما تم إرجاعها بواسطة CTPQRT، مخزنة كمصفوفة NBxK.
!>

LDT

!>          LDT is INTEGER
!>          البعد الرائد للمصفوفة T.  LDT >= NB.
!>

A

!>          A is REAL array, dimension
!>          (LDA,N) if SIDE = 'L' or
!>          (LDA,K) if SIDE = 'R'
!>          عند الإدخال، المصفوفة A بحجم KxN أو MxK.
!>          عند الخروج، يتم استبدال A بالجزء المقابل من
!>          Q*C أو Q^T*C أو C*Q أو C*Q^T.  انظر التفاصيل الإضافية.
!>

LDA

!>          LDA is INTEGER
!>          البعد الرائد للمصفوفة A.
!>          إذا كان SIDE = 'L'، فإن LDC >= max(1,K)؛
!>          إذا كان SIDE = 'R'، فإن LDC >= max(1,M).
!>

B

!>          B is REAL array, dimension (LDB,N)
!>          عند الإدخال، المصفوفة B بحجم MxN.
!>          عند الخروج، يتم استبدال B بالجزء المقابل من
!>          Q*C أو Q^T*C أو C*Q أو C*Q^T.  انظر التفاصيل الإضافية.
!>

LDB

!>          LDB is INTEGER
!>          البعد الرائد للمصفوفة B.
!>          LDB >= max(1,M).
!>

WORK

!>          WORK is REAL array. The dimension of WORK is
!>           N*NB if SIDE = 'L', or  M*NB if SIDE = 'R'.
!>

INFO

!>          INFO is INTEGER
!>          = 0:  انتهاء ناجح
!>          < 0:  إذا كان INFO = -i، كان الوسيط i ذو قيمة غير صالحة
!>

تفاصيل إضافية:

!>
!>  تحتوي أعمدة المصفوفة الخماسية V على المُعرِّفات الأولية
!>  H(1), H(2), ..., H(K)؛ تتكون V من كتلة مستطيلة V1 و
!>  كتلة شبه منحرف V2:
!>
!>        V = [V1]
!>            [V2].
!>
!>  يتم تحديد حجم الكتلة شبه المنحرفة V2 بواسطة المعلمة L،
!>  حيث 0 <= L <= K؛ V2 شبه منحرف علوي، يتكون من الصفوف L الأولى
!>  لمصفوفة علوية مثلثية KxK.  إذا كان L=K، فإن V2 علوي مثلثي؛
!>  إذا كان L=0، فلا توجد كتلة شبه منحرف، وبالتالي V = V1 مستطيل.
!>
!>  إذا كان SIDE = 'L':  C = [A]  حيث A هي KxN،  B هي MxN، و V هي MxK.
!>                      [B]
!>
!>  إذا كان SIDE = 'R':  C = [A B]  حيث A هي MxK، B هي MxN، و V هي NxK.
!>
!>  يتم تشكيل المصفوفة المتعامدة الحقيقية Q من V و T.
!>
!>  إذا كان TRANS='N' و SIDE='L'، يتم استبدال C عند الخروج بـ Q * C.
!>
!>  إذا كان TRANS='T' و SIDE='L'، يتم استبدال C عند الخروج بـ Q^T * C.
!>
!>  إذا كان TRANS='N' و SIDE='R'، يتم استبدال C عند الخروج بـ C * Q.
!>
!>  إذا كان TRANS='T' و SIDE='R'، يتم استبدال C عند الخروج بـ C * Q^T.
!>

التعريف في السطر 214 من الملف stpmqrt.f.
