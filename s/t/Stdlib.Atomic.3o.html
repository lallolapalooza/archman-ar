النوع !'a t

مرجع ذري (قابل للتغيير) لقيمة من النوع 'a.

القيمة make : 'a -> 'a t

إنشاء مرجع ذري.

القيمة make_contended : 'a -> 'a t

إنشاء مرجع ذري منفرد على سطر ذاكرة التخزين المؤقت. يشغل 4-16 ضعف حجم الذاكرة للمرجع الذي تم إنشاؤه باستخدام make v.

الغرض الأساسي هو منع المشاركة الزائفة وما يترتب عليها من تدهور في الأداء. عندما يقوم معالج بعملية ذرية، فإنه يأخذ ملكية خط ذاكرة تخزين مؤقت كامل يحتوي على المرجع الذري مؤقتًا. إذا تشاركت عدة مراجع ذرية نفس خط ذاكرة التخزين المؤقت، يصبح تعديل مناطق الذاكرة المنفصلة هذه بشكل متزامن مستحيلاً، مما قد يؤدي إلى اختناق. وبالتالي، كقاعدة عامة، إذا كان المرجع الذري يعاني من التنافس، فقد يؤدي تخصيص سطر ذاكرة تخزين مؤقت خاص به إلى تحسين الأداء.

القيمة get : 'a t -> 'a

الحصول على القيمة الحالية للمرجع الذري.

القيمة set : 'a t -> 'a -> unit

تعيين قيمة جديدة للمرجع الذري.

القيمة exchange : 'a t -> 'a -> 'a

تعيين قيمة جديدة للمرجع الذري، وإرجاع القيمة الحالية.

القيمة compare_and_set : 'a t -> 'a -> 'a -> bool

compare_and_set r seen v تعين القيمة الجديدة لـ r إلى v فقط إذا كانت قيمتها الحالية متساوية فيزيائيًا مع seen -- تتم المقارنة والتعيين بشكل ذري. تُرجع true إذا نجحت المقارنة (لذلك حدث التعيين) و false بخلاف ذلك.

القيمة fetch_and_add : int t -> int -> int

fetch_and_add r n تقوم بزيادة قيمة r بمقدار n بشكل ذري، وتُرجع القيمة الحالية (قبل الزيادة).

القيمة incr : int t -> unit

incr r تقوم بزيادة قيمة r بمقدار 1 بشكل ذري.

القيمة decr : int t -> unit

decr r تقوم بتقليل قيمة r بمقدار 1 بشكل ذري.

الوحدة Loc : sig end

"مواقع" ذرية، مثل حقول السجلات.

أمثلة

تنسيق أساسي للخيوط

حالة استخدام أساسية هي وجود عدادات عامة يتم تحديثها بطريقة آمنة للخيوط، على سبيل المثال للاحتفاظ ببعض أنواع المقاييس على عمليات الإدخال/الإخراج التي يقوم بها البرنامج. حالة استخدام أساسية أخرى هي تنسيق إنهاء الخيوط في برنامج معين، على سبيل المثال عندما يجد خيط ما إجابة، أو عندما يتم إيقاف البرنامج بواسطة المستخدم.

هنا، على سبيل المثال، سنحاول العثور على رقم يلبي الهاش الخاص به خاصية أساسية. للقيام بذلك، سنقوم بتشغيل خيوط متعددة ستحاول أرقامًا عشوائية حتى تجد واحدًا يعمل.

بالطبع، المخرجات أدناه هي عينة تشغيل وستتغير في كل مرة يتم فيها تشغيل البرنامج.

    (* استخدام للإنهاء *)
    let stop_all_threads = Atomic.make false
    (* العدد الإجمالي للمحاولات الفردية للعثور على رقم *)
    let num_attempts = Atomic.make 0
    (* ابحث عن رقم يحقق [p]، عن طريق... تجربة أرقام عشوائية
       حتى يناسب أحدهم. *)
    let find_number_where (p:int -> bool) =
      let rand = Random.State.make_self_init() in
      while not (Atomic.get stop_all_threads) do
        let n = Random.State.full_int rand max_int in
        ignore (Atomic.fetch_and_add num_attempts 1 : int);
        if p (Hashtbl.hash n) then (
          Printf.printf "found %d (hash=%d)\n%!" n (Hashtbl.hash n);
          Atomic.set stop_all_threads true; (* إشارة لجميع الخيوط للتوقف *)
        )
      done;;
    (* تشغيل مجالات متعددة للبحث عن [n] حيث [hash n <= 100] *)
    let () =
      let criterion n = n <= 100 in
      let threads =
        Array.init 8
          (fun _ -> Domain.spawn (fun () -> find_number_where criterion))
      in
      Array.iter Domain.join threads;
      Printf.printf "total number of attempts: %d\n%!"
        (Atomic.get num_attempts) ;;
    - : unit = ()
    found 1651745641680046833 (hash=33)
    total number of attempts: 30230350

مكدس Treiber

مثال آخر هو مكدس Treiber أساسي (مكدس آمن للخيوط) يمكن مشاركته بأمان بين الخيوط.

لاحظ كيف أن كل من push و pop متكرران، لأنهما يحاولان تبديل المكدس الجديد (مع عنصر واحد إضافي، أو أقل) بالمكدس القديم. هذا تزامن متفائل: كل تكرار، على سبيل المثال، push stack x يحصل على المكدس القديم l، ويأمل أنه بحلول الوقت الذي يحاول فيه استبدال l بـ x::l، لم يكن لدى أي شخص آخر وقت لتعديل القائمة. إذا فشل compare_and_set، فهذا يعني أننا كنا متفائلين للغاية، ويجب أن نحاول مرة أخرى.

    type 'a stack = 'a list Atomic.t
    let rec push (stack: _ stack) elt : unit =
      let cur = Atomic.get stack in
      let success = Atomic.compare_and_set stack cur (elt :: cur) in
      if not success then
        push stack elt
    let rec pop (stack: _ stack) : _ option =
      let cur = Atomic.get stack in
      match cur with
      | [] -> None
      | x :: tail ->
        let success = Atomic.compare_and_set stack cur tail in
        if success then Some x
        else pop stack
    # let st = Atomic.make []
    # push st 1
    - : unit = ()
    # push st 2
    - : unit = ()
    # pop st
    - : int option = Some 2
    # pop st
    - : int option = Some 1
    # pop st
    - : int option = None
