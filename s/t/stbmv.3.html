subroutine stbmv (character uplo, character trans, character diag, integer n, integer k, real, dimension(lda,*) a, integer lda, real, dimension(*) x, integer incx)

STBMV

الغرض:

!>
!> STBMV  تنفذ إحدى عمليات المصفوفة والمتجه
!>
!>    x := A*x,   أو   x := A**T*x,
!>
!> حيث x هو متجه من n عنصر و A هي مصفوفة فرقة مثلثية عليا أو سفلى، موحدة أو غير موحدة، بحجم n × n، تحتوي على ( k + 1 ) أقطار.
!>

المعلمات

UPLO 
!>          UPLO is CHARACTER*1
!>           عند الإدخال، تحدد UPLO ما إذا كانت المصفوفة مصفوفة مثلثية عليا أو سفلى على النحو التالي:
!>
!>              UPLO = 'U' أو 'u'   A مصفوفة مثلثية عليا.
!>
!>              UPLO = 'L' أو 'l'   A مصفوفة مثلثية سفلى.
!>

TRANS

!>          TRANS is CHARACTER*1
!>           عند الإدخال، تحدد TRANS العملية المراد تنفيذها على النحو التالي:
!>
!>              TRANS = 'N' أو 'n'   x := A*x.
!>
!>              TRANS = 'T' أو 't'   x := A**T*x.
!>
!>              TRANS = 'C' أو 'c'   x := A**T*x.
!>

DIAG

!>          DIAG is CHARACTER*1
!>           عند الإدخال، تحدد DIAG ما إذا كانت A موحدة مثلثية أم لا على النحو التالي:
!>
!>              DIAG = 'U' أو 'u'   يفترض أن A موحدة مثلثية.
!>
!>              DIAG = 'N' أو 'n'   لا يُفترض أن A موحدة
!>                                  مثلثية.
!>

N

!>          N is INTEGER
!>           عند الإدخال، تحدد N رتبة المصفوفة A.
!>           يجب أن تكون N على الأقل صفر.
!>

K

!>          K is INTEGER
!>           عند الإدخال مع UPLO = 'U' أو 'u'، تحدد K عدد
!>           الأقطار الفائقة للمصفوفة A.
!>           عند الإدخال مع UPLO = 'L' أو 'l'، تحدد K عدد
!>           الأقطار الفرعية للمصفوفة A.
!>           يجب أن تحقق K  0 .le. K.
!>

A

!>          A is REAL array, dimension ( LDA, N )
!>           قبل الإدخال مع UPLO = 'U' أو 'u'، يجب أن يحتوي الجزء العلوي ( k + 1 )
!>           بواسطة N من المصفوفة A على الجزء الفرقة المثلثية العليا من مصفوفة المعاملات، المقدمة عمودًا بعمود، مع القطر الرئيسي للمصفوفة في الصف
!>           ( k + 1 ) من المصفوفة، والقطر الفائق الأول يبدأ من الموضع 2 في الصف k، وهكذا. الجزء العلوي الأيسر بحجم k × k من
!>           المصفوفة A غير مرجعي.
!>           سيقوم المقطع البرمجي التالي بنقل مصفوفة فرقة مثلثية عليا من تخزين مصفوفة كاملة تقليدية
!>           إلى تخزين فرقة:
!>
!>                 DO 20, J = 1, N
!>                    M = K + 1 - J
!>                    DO 10, I = MAX( 1, J - K ), J
!>                       A( M + I, J ) = matrix( I, J )
!>              10    CONTINUE
!>              20 CONTINUE
!>
!>           قبل الإدخال مع UPLO = 'L' أو 'l'، يجب أن يحتوي الجزء العلوي ( k + 1 )
!>           بواسطة N من المصفوفة A على الجزء الفرقة المثلثية السفلى من مصفوفة المعاملات، المقدمة عمودًا بعمود، مع القطر الرئيسي للمصفوفة في الصف 1
!>           من المصفوفة، والقطر الفرعي الأول يبدأ من الموضع 1 في الصف 2، وهكذا. الجزء السفلي الأيمن بحجم k × k من
!>           المصفوفة A غير مرجعي.
!>           سيقوم المقطع البرمجي التالي بنقل مصفوفة فرقة مثلثية سفلى من تخزين مصفوفة كاملة تقليدية
!>           إلى تخزين فرقة:
!>
!>                 DO 20, J = 1, N
!>                    M = 1 - J
!>                    DO 10, I = J, MIN( N, J + K )
!>                       A( M + I, J ) = matrix( I, J )
!>              10    CONTINUE
!>              20 CONTINUE
!>
!>           لاحظ أنه عندما يكون DIAG = 'U' أو 'u'، فإن العناصر في المصفوفة A
!>           المقابلة لعناصر القطر الرئيسي للمصفوفة ليست مرجعية، ولكن يُفترض أنها وحدات.
!>

LDA

!>          LDA is INTEGER
!>           عند الإدخال، تحدد LDA البعد الأول لـ A كما هو معلن
!>           في البرنامج المستدعي. يجب أن يكون LDA على الأقل
!>           ( k + 1 ).
!>

X

!>          X is REAL array, dimension at least
!>           ( 1 + ( n - 1 )*abs( INCX ) ).
!>           قبل الإدخال، يجب أن يحتوي المتجه X المعدل على عناصر المتجه x البالغ عددها n. عند الخروج، يتم الكتابة فوق X بالمتجه المحول x.
!>

INCX

!>          INCX is INTEGER
!>           عند الإدخال، يحدد INCX الزيادة لعناصر X. يجب ألا يكون INCX صفرًا.
!>
