subroutine strcon (character norm, character uplo, character diag, integer n, real, dimension( lda, * ) a, integer lda, real rcond, real, dimension( * ) work, integer, dimension( * ) iwork, integer info)

STRCON

الغرض:

!>
!> STRCON تقدّر مقلوب رقم الشرط لمصفوفة مثلثة A، سواء في المعيار 1-norm أو
!> المعيار إنفينيتي-norm.
!>
!> يتم حساب معيار A ويتم الحصول على تقدير لـ
!> norm(inv(A))، ثم يتم حساب مقلوب رقم الشرط على أنه
!>    RCOND = 1 / ( norm(A) * norm(inv(A)) ).
!>

المعلمات

NORM 
!>          NORM هو CHARACTER*1
!>          يحدد ما إذا كان مطلوبًا رقم الشرط 1-norm أو رقم الشرط
!>          إنفينيتي-norm:
!>          = '1' أو 'O':  1-norm;
!>          = 'I':         إنفينيتي-norm.
!>

UPLO

!>          UPLO هو CHARACTER*1
!>          = 'U':  A مصفوفة مثلثة عليا؛
!>          = 'L':  A مصفوفة مثلثة سفلى.
!>

DIAG

!>          DIAG هو CHARACTER*1
!>          = 'N':  A مصفوفة مثلثة غير وحدة؛
!>          = 'U':  A مصفوفة مثلثة وحدة.
!>

N

!>          N هو INTEGER
!>          رتبة المصفوفة A.  N >= 0.
!>

A

!>          A هي مصفوفة REAL، البعد (LDA,N)
!>          المصفوفة المثلثة A.  إذا كان UPLO = 'U'، فإن الجزء العلوي
!>          المثلثي الذي بحجم N*N في المصفوفة A يحتوي على المصفوفة
!>          المثلثة العليا، والجزء السفلي المثلثي الصارم لـ
!>          A لا يتم الاستشهاد به.  إذا كان UPLO = 'L'، فإن الجزء السفلي
!>          المثلثي الذي بحجم N*N في المصفوفة A يحتوي على المصفوفة
!>          المثلثة السفلى، والجزء العلوي المثلثي الصارم لـ A لا يتم
!>          الاستشهاد به.  إذا كان DIAG = 'U'، فإن العناصر القطرية لـ A
!>          لا يتم الاستشهاد بها أيضًا ويُفترض أنها 1.
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرئيسي للمصفوفة A.  LDA >= max(1,N).
!>

RCOND

!>          RCOND هو REAL
!>          مقلوب رقم شرط المصفوفة A،
!>          محسوب كـ RCOND = 1/(norm(A) * norm(inv(A))).
!>

WORK

!>          WORK هي مصفوفة REAL، البعد (3*N)
!>

IWORK

!>          IWORK هي مصفوفة INTEGER، البعد (N)
!>

INFO

!>          INFO هو INTEGER
!>          = 0:  إنهاء ناجح
!>          < 0:  إذا كان INFO = -i، فإن المعلمة i كانت لها قيمة غير قانونية
!>
