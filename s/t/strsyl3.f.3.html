subroutine strsyl3 (character trana, character tranb, integer isgn, integer m, integer n, real, dimension( lda, * ) a, integer lda, real, dimension( ldb, * ) b, integer ldb, real, dimension( ldc, * ) c, integer ldc, real scale, integer, dimension( * ) iwork, integer liwork, real, dimension( ldswork, * ) swork, integer ldswork, integer info)

STRSYL3

الغرض:

!>
!>  STRSYL3 يحل معادلة المصفوفة السيلفستر الحقيقية:
!>
!>     op(A)*X + X*op(B) = scale*C أو
!>     op(A)*X - X*op(B) = scale*C،
!>
!>  حيث op(A) = A أو A**T، وكلا من A و B مثلثيتين شبه علويتين.
!>  A هي M*M و B هي N*N؛ الطرف الأيمن C والحل X هما M*N؛
!>  و scale هو عامل قياس ناتج، تم تعيينه <= 1 لتجنب فيضان X.
!>
!>  يجب أن تكون A و B في الشكل القانوني لشور (كما يتم إرجاعه بواسطة SHSEQR)،
!>  أي، مصفوفة مثلثية علوية كتلية مع كتل قطرية 1*1 و 2*2؛
!>  كل كتلة قطرية 2*2 لها عناصر قطرية متساوية وعناصر غير قطرية
!>  متعاكسة الإشارة.
!>
!>  هذا هو الإصدار الكتلي للخوارزمية.
!>

المعلمات:

TRANA 
!>          TRANA هي CHARACTER*1
!>          يحدد خيار op(A):
!>          = 'N': op(A) = A    (بدون تبديل)
!>          = 'T': op(A) = A**T (تبديل)
!>          = 'C': op(A) = A**H (مرافق التبديل = التبديل)
!>

TRANB

!>          TRANB هي CHARACTER*1
!>          يحدد خيار op(B):
!>          = 'N': op(B) = B    (بدون تبديل)
!>          = 'T': op(B) = B**T (تبديل)
!>          = 'C': op(B) = B**H (مرافق التبديل = التبديل)
!>

ISGN

!>          ISGN هو INTEGER
!>          يحدد الإشارة في المعادلة:
!>          = +1: حل op(A)*X + X*op(B) = scale*C
!>          = -1: حل op(A)*X - X*op(B) = scale*C
!>

M

!>          M هو INTEGER
!>          رتبة المصفوفة A، وعدد الصفوف في المصفوفتين X و C. M >= 0.
!>

N

!>          N هو INTEGER
!>          رتبة المصفوفة B، وعدد الأعمدة في المصفوفتين X و C. N >= 0.
!>

A

!>          A هي مصفوفة REAL، بأبعاد (LDA,M)
!>          المصفوفة شبه المثلثية العلوية A، في الشكل القانوني لشور.
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرائد للمصفوفة A. LDA >= max(1,M).
!>

B

!>          B هي مصفوفة REAL، بأبعاد (LDB,N)
!>          المصفوفة شبه المثلثية العلوية B، في الشكل القانوني لشور.
!>

LDB

!>          LDB هو INTEGER
!>          البعد الرائد للمصفوفة B. LDB >= max(1,N).
!>

C

!>          C هي مصفوفة REAL، بأبعاد (LDC,N)
!>          عند الدخول، مصفوفة الطرف الأيمن C ذات الأبعاد M*N.
!>          عند الخروج، يتم استبدال C بمصفوفة الحل X.
!>

LDC

!>          LDC هو INTEGER
!>          البعد الرائد للمصفوفة C. LDC >= max(1,M)
!>

SCALE

!>          SCALE هي REAL
!>          عامل القياس، scale، تم تعيينه <= 1 لتجنب فيضان X.
!>

IWORK

!>          IWORK هي مصفوفة INTEGER، بأبعاد (MAX(1,LIWORK))
!>          عند الخروج، إذا كان INFO = 0، فإن IWORK(1) ترجع LIWORK المثلى.
!>

LIWORK

!>          LIWORK هي INTEGER
!>          بعد مصفوفة IWORK. LIWORK >=  ((M + NB - 1) / NB + 1)
!>          + ((N + NB - 1) / NB + 1)، حيث NB هو حجم الكتلة الأمثل.
!>
!>          إذا كان LIWORK = -1، فيفترض استعلام مساحة العمل؛ تقوم الوحدة الفرعية
!!>          فقط بحساب البعد الأمثل لمصفوفة IWORK،
!>          وتعيد هذه القيمة كأول إدخال في مصفوفة IWORK،
!>          ولا يتم إصدار أي رسالة خطأ متعلقة بـ XERBLA بخصوص LIWORK.
!>

SWORK

!>          SWORK هي مصفوفة REAL، بأبعاد (MAX(2, ROWS),
!>          MAX(1,COLS)).
!>          عند الخروج، إذا كان INFO = 0، فإن SWORK(1) ترجع ROWS المثلى
!>          و SWORK(2) ترجع COLS المثلى.
!>

LDSWORK

!>          LDSWORK هي INTEGER
!>          LDSWORK >= MAX(2,ROWS)، حيث ROWS = ((M + NB - 1) / NB + 1)
!>          و NB هو حجم الكتلة الأمثل.
!>
!>          إذا كان LDSWORK = -1، فيفترض استعلام مساحة العمل؛ تقوم الوحدة الفرعية
!>          فقط بحساب الأبعاد المثلى لمصفوفة SWORK،
!>          وتعيد هذه القيم كأول وثاني إدخال في مصفوفة SWORK،
!>          ولا يتم إصدار أي رسالة خطأ متعلقة بـ LWORK بواسطة XERBLA.
!>

INFO

!>          INFO هو INTEGER
!>          = 0: خروج ناجح
!>          < 0: إذا كان INFO = -i، فإن المعامل i كان له قيمة غير قانونية
!>          = 1: A و B لهما قيم ذاتية مشتركة أو قريبة جدًا؛ تم استخدام
!>               قيم معدلة لحل المعادلة (لكن المصفوفتين A و B لم تتغير).
!>
