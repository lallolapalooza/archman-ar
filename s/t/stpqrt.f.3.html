subroutine stpqrt (integer m, integer n, integer l, integer nb, real, dimension( lda, * ) a, integer lda, real, dimension( ldb, * ) b, integer ldb, real, dimension( ldt, * ) t, integer ldt, real, dimension( * ) work, integer info)

الغرض:

!>
!> STPQRT تحسب تحليل QR مسدود لمصفوفة حقيقية
!>  C، والتي تتكون من
!>  كتلة مثلثة A وكتلة خماسية B، باستخدام تمثيل WY المدمج لـ Q.
!>

المعلمات

M 
!>          M هو عدد صحيح (INTEGER)
!>          عدد صفوف المصفوفة B.
!>          M >= 0.
!>

N

!>          N هو عدد صحيح (INTEGER)
!>          عدد أعمدة المصفوفة B، ورتبة

!>

L

!>          L هو عدد صحيح (INTEGER)
!>          عدد صفوف الجزء شبه المنحرف العلوي من B.

!>

NB

!>          NB هو عدد صحيح (INTEGER)

!>          مستطيلة، والصفوف L الأخيرة شبه منحرفة علوية.

T

تفاصيل إضافية:

!>
!>  المصفوفة المدخلة C هي مصفوفة (N+M) × N
!>
!>               C = [ A ]
!>                   [ B ]
!>
!>  حيث A هي مصفوفة مثلثة عليا بحجم N × N، و B هي مصفوفة خماسية بحجم M × N
!>  تتكون من مصفوفة مستطيلة B1 بحجم (M-L) × N في الأعلى ومصفوفة شبه منحرفة علوية L × N
!>  تسمى B2 في الأسفل:
!>
!>               B = [ B1 ]  <- (M-L) × N مستطيلة
!>                   [ B2 ]  <-     L × N شبه منحرفة علوية.
!>
!>  المصفوفة شبه المنحرفة العلوية B2 تتكون من أول L صفوف من مصفوفة
!>  مثلثة عليا بحجم N × N، حيث 0 <= L <= MIN(M,N).  إذا كان L=0،
!>  تكون B مستطيلة M × N؛ إذا كان M=L=N، تكون B مثلثة عليا.
!>
!>  المصفوفة W تخزن الانعكاسات الابتدائية H(i) في العمود i
!>  تحت القطر (لـ A) في المصفوفة المدخلة C بحجم (N+M) × N
!>
!>               C = [ A ]  <- مثلثة عليا N × N
!>                   [ B ]  <- خماسية M × N
!>
!>  بحيث يمكن تمثيل W كـ
!>
!>               W = [ I ]  <- هوية، N × N
!>                   [ V ]  <- M × N، بنفس شكل B.
!>
!>  وبالتالي، فإن جميع المعلومات اللازمة لـ W موجودة عند الخروج في B،
!>  والتي نسميها V أعلاه. لاحظ أن V لها نفس شكل B؛ أي،
!>
!>               V = [ V1 ] <- (M-L) × N مستطيلة
!>                   [ V2 ] <-     L × N شبه منحرفة علوية.
!>
!>  تمثل أعمدة V المتجهات التي تحدد H(i).
!>
!>  عدد الكتل هو B = سقف(N/NB)، حيث كل
!>  كتلة هي من الرتبة NB باستثناء الكتلة الأخيرة، والتي هي من الرتبة
!>  IB = N - (B-1)*NB.  لكل كتلة من الكتل B، يتم حساب عامل كتلة مثلثية عليا
!>  وهو: T1, T2, ..., TB.  يتم تخزين T ذات الأبعاد NB × NB (و IB × IB
!>  لللـ الكتلة الأخيرة) في المصفوفة T ذات الأبعاد NB × N كـ
!>
!>               T = [T1 T2 ... TB].
!>
