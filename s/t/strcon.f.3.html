توثيق الوظيفة/الروتين الفرعي

subroutine strcon (character norm, character uplo, character diag, integer n, real, dimension( lda, * ) a, integer lda, real rcond, real, dimension( * ) work, integer, dimension( * ) iwork, integer info)

STRCON

الغرض:

!>
!> STRCON يقدر مقلوب رقم الشرط لمصفوفة مثلثية A، إما في المعيار 1-norm أو
!> المعيار إنفينيتي-norm.
!>
!> يتم حساب معيار A ويتم الحصول على تقدير لـ
!> norm(inv(A))، ثم يتم حساب مقلوب رقم الشرط على أنه
!>    RCOND = 1 / ( norm(A) * norm(inv(A)) ).
!>

المعلمات

NORM 
!>          NORM is CHARACTER*1
!>          يحدد ما إذا كان رقم الشرط 1-norm أو إنفينيتي-norm مطلوبًا:
!>          = '1' or 'O':  1-norm;
!>          = 'I':         Infinity-norm.
!>

UPLO

!>          UPLO is CHARACTER*1
!>          = 'U':  A هي مصفوفة مثلثية عليا؛
!>          = 'L':  A هي مصفوفة مثلثية سفلى.
!>

DIAG

!>          DIAG is CHARACTER*1
!>          = 'N':  A هي مصفوفة مثلثية غير وحدوية؛
!>          = 'U':  A هي مصفوفة مثلثية وحدوية.
!>

N

!>          N is INTEGER
!>          رتبة المصفوفة A.  N >= 0.
!>

A

!>          A is REAL array, dimension (LDA,N)
!>          المصفوفة المثلثية A.  إذا كان UPLO = 'U'، فإن الجزء العلوي
!>          المثلثي من A بحجم N × N يحتوي على المصفوفة المثلثية العليا،
!>          والجزء السفلي المثلثي الصارم من A غير مرجعي.  إذا كان UPLO = 'L'،
!>          فإن الجزء السفلي المثلثي من A بحجم N × N يحتوي على المصفوفة
!>          المثلثية السفلى، والجزء العلوي المثلثي الصارم من A غير مرجعي.
!>          إذا كان DIAG = 'U'، فإن عناصر القطر من A غير مرجعية أيضًا
!>          ويفترض أنها 1.
!>

LDA

!>          LDA is INTEGER
!>          البعد الرئيسي للمصفوفة A.  LDA >= max(1,N).
!>

RCOND

!>          RCOND is REAL
!>          مقلوب رقم الشرط للمصفوفة A،
!>          محسوب كـ RCOND = 1/(norm(A) * norm(inv(A))).
!>

WORK

!>          WORK is REAL array, dimension (3*N)
!>

IWORK

!>          IWORK is INTEGER array, dimension (N)
!>

INFO

!>          INFO is INTEGER
!>          = 0:  انتهاء ناجح
!>          < 0:  إذا كان INFO = -i، فإن المعلمة i كانت لها قيمة غير قانونية
!>
