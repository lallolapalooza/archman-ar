المكتبة

مكتبة C القياسية (libc, -lc)

الملخص

#include <sys/queue.h>
STAILQ_ENTRY(TYPE);
STAILQ_HEAD(HEADNAME, TYPE);
STAILQ_HEAD STAILQ_HEAD_INITIALIZER(STAILQ_HEAD head);

int STAILQ_EMPTY(STAILQ_HEAD *head);
void STAILQ_INSERT_HEAD(STAILQ_HEAD *head,
                         struct TYPE *elm, STAILQ_ENTRY NAME);
void STAILQ_INSERT_TAIL(STAILQ_HEAD *head,
                         struct TYPE *elm, STAILQ_ENTRY NAME);
void STAILQ_INSERT_AFTER(STAILQ_HEAD *head, struct TYPE *listelm,
                         struct TYPE *elm, STAILQ_ENTRY NAME);
struct TYPE *STAILQ_FIRST(STAILQ_HEAD *head);
struct TYPE *STAILQ_NEXT(struct TYPE *elm, STAILQ_ENTRY NAME);
STAILQ_FOREACH(struct TYPE *var, STAILQ_HEAD *head, STAILQ_ENTRY NAME);
void STAILQ_REMOVE(STAILQ_HEAD *head, struct TYPE *elm, TYPE,
                         STAILQ_ENTRY NAME);
void STAILQ_REMOVE_HEAD(STAILQ_HEAD *head,
                         STAILQ_ENTRY NAME);
void STAILQ_CONCAT(STAILQ_HEAD *head1, STAILQ_HEAD *head2);

ملاحظة: توجد وحدات ماكرو متطابقة بنفس البادئة SIMPLEQ بدلًا من STAILQ؛ انظر الإصدارات.

الوصف

تُعرّف هذه الوحدات الماكرو وتشغل طوابير ذيل مرتبطة أحاديًا.

في تعريفات الوحدات الماكرو، TYPE هو اسم بنية يحددها المستخدم، ويجب أن تحتوي على حقل من نوع STAILQ_ENTRY، باسم NAME. الوسيط HEADNAME هو اسم بنية يحددها المستخدم ويجب أن يتم الإعلان عنها باستخدام الوحدة الماكرو STAILQ_HEAD().

الإنشاء

يتم قيادة طابور ذيل مرتبط أحاديًا ببنية معرفة بواسطة الوحدة الماكرو STAILQ_HEAD(). تحتوي هذه البنية على زوج من المؤشرات، أحدهما يشير إلى العنصر الأول في طابور الذيل والآخر يشير إلى العنصر الأخير في طابور الذيل. العناصر مرتبطة أحاديًا لتقليل استخدام المساحة وتكاليف معالجة المؤشرات للعناصر العشوائية، مقابل تكلفة O(n) للإزالة. يمكن إضافة عناصر جديدة إلى طابور الذيل بعد عنصر موجود، أو في رأس طابور الذيل، أو في نهاية طابور الذيل. يتم الإعلان عن بنية STAILQ_HEAD على النحو التالي:


STAILQ_HEAD(HEADNAME, TYPE) head;

حيث struct HEADNAME هي البنية التي سيتم تعريفها، و struct TYPE هو نوع العناصر التي سيتم ربطها في طابور الذيل. يمكن لاحقًا الإعلان عن مؤشر إلى رأس طابور الذيل كالتالي:


struct HEADNAME *headp;

(الأسماء head و headp قابلة للاختيار من قبل المستخدم.)

تُعرّف الوحدة الماكرو STAILQ_ENTRY() بنية تربط العناصر في طابور الذيل.

تقوم الوحدة الماكرو STAILQ_HEAD_INITIALIZER() بتقييم مُهيئ لطابور الذيل head.

تقوم الوحدة الماكرو STAILQ_INIT() بتهيئة طابور الذيل المشار إليه بواسطة head.

تُرجع الوحدة الماكرو STAILQ_EMPTY() قيمة صحيحة إذا لم تكن هناك عناصر في طابور الذيل.

الإدراج

تُدرج الوحدة الماكرو STAILQ_INSERT_HEAD() العنصر الجديد elm في رأس طابور الذيل.

تُدرج الوحدة الماكرو STAILQ_INSERT_TAIL() العنصر الجديد elm في نهاية طابور الذيل.

تُدرج الوحدة الماكرو STAILQ_INSERT_AFTER() العنصر الجديد elm بعد العنصر listelm.

التجوال

تُرجع الوحدة الماكرو STAILQ_FIRST() العنصر الأول في طابور الذيل أو NULL إذا كان طابور الذيل فارغًا.

تُرجع الوحدة الماكرو STAILQ_NEXT() العنصر التالي في طابور الذيل، أو NULL إذا كان هذا العنصر هو الأخير.

تجتاز الوحدة الماكرو STAILQ_FOREACH() طابور الذيل المشار إليه بواسطة head في الاتجاه الأمامي، وتعين كل عنصر بالتتابع إلى var.

الإزالة

تُزيل الوحدة الماكرو STAILQ_REMOVE() العنصر elm من طابور الذيل.

تُزيل الوحدة الماكرو STAILQ_REMOVE_HEAD() العنصر الموجود في رأس طابور الذيل. لتحقيق أقصى قدر من الكفاءة، يجب على العناصر التي تتم إزالتها من رأس طابور الذيل استخدام هذه الوحدة الماكرو بشكل صريح بدلاً من الوحدة الماكرو العامة STAILQ_REMOVE().

ميزات أخرى

تُلحق الوحدة الماكرو STAILQ_CONCAT() طابور الذيل الذي يقوده head2 بنهاية الطابور الذي يقوده head1، وتزيل جميع الإدخالات من الأول.

قيمة الإرجاع

تُرجع STAILQ_EMPTY() قيمة غير صفرية إذا كان الطابور فارغًا، وصفرًا إذا كان الطابور يحتوي على إدخال واحد على الأقل.

تُرجع STAILQ_FIRST() و STAILQ_NEXT() مؤشرًا إلى البنية الأولى أو التالية TYPE، على التوالي.

تُرجع STAILQ_HEAD_INITIALIZER() مُهيئًا يمكن تعيينه إلى الطابور head.

الإصدارات

توفر بعض أنظمة BSD وحدات SIMPLEQ بدلًا من STAILQ. هما متطابقان، ولكن لأسباب تاريخية تم تسميتهما بشكل مختلف في أنظمة BSD المختلفة. نشأت STAILQ في FreeBSD، ونشأت SIMPLEQ في NetBSD. لأسباب التوافق، توفر بعض الأنظمة كلا المجموعتين من الوحدات الماكرو. توفر glibc كلاً من STAILQ و SIMPLEQ، وهما متطابقان باستثناء عدم وجود ما يعادل SIMPLEQ للوحدة الماكرو STAILQ_CONCAT().

الأخطاء

لا تسمح الوحدة الماكرو STAILQ_FOREACH() بإزالة أو تحرير var داخل الحلقة، حيث سيؤثر ذلك على عملية التجوال. الوحدة الماكرو STAILQ_FOREACH_SAFE()، الموجودة في أنظمة BSD ولكنها غير موجودة في glibc، تصلح هذا القيد بالسماح بإزالة var بأمان من القائمة وتحريرها من داخل الحلقة دون التأثير على عملية التجوال.

أمثلة

#include <stddef.h>
#include <stdio.h>
#include <stdlib.h>
#include <sys/queue.h>
struct entry {
    int data;
    STAILQ_ENTRY(entry) entries;        /* طابور ذيل مرتبط أحاديًا */
};
STAILQ_HEAD(stailhead, entry);
int
main(void)
{
    struct entry *n1, *n2, *n3, *np;
    struct stailhead head;                  /* رأس طابور الذيل المرتبط أحاديًا */
    STAILQ_INIT(&head);                     /* تهيئة الطابور */
    n1 = malloc(sizeof(struct entry));      /* الإدراج في الرأس */
    STAILQ_INSERT_HEAD(&head, n1, entries);
    n1 = malloc(sizeof(struct entry));      /* الإدراج في الذيل */
    STAILQ_INSERT_TAIL(&head, n1, entries);
    n2 = malloc(sizeof(struct entry));      /* الإدراج بعد */
    STAILQ_INSERT_AFTER(&head, n1, n2, entries);
    STAILQ_REMOVE(&head, n2, entry, entries); /* الحذف */
    free(n2);
    n3 = STAILQ_FIRST(&head);
    STAILQ_REMOVE_HEAD(&head, entries);     /* الحذف من الرأس */
    free(n3);
    n1 = STAILQ_FIRST(&head);
    n1->data = 0;
    for (unsigned int i = 1; i < 5; i++) {
        n1 = malloc(sizeof(struct entry));
        STAILQ_INSERT_HEAD(&head, n1, entries);
        n1->data = i;
    }
