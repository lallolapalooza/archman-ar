STAILQ(3) دليل وظائف المكتبة STAILQ(3)

الاسم

SIMPLEQ_EMPTY، SIMPLEQ_ENTRY، SIMPLEQ_FIRST، SIMPLEQ_FOREACH، SIMPLEQ_HEAD، SIMPLEQ_HEAD_INITIALIZER، SIMPLEQ_INIT، SIMPLEQ_INSERT_AFTER، SIMPLEQ_INSERT_HEAD، SIMPLEQ_INSERT_TAIL، SIMPLEQ_NEXT، SIMPLEQ_REMOVE، SIMPLEQ_REMOVE_HEAD، STAILQ_CONCAT، STAILQ_EMPTY، STAILQ_ENTRY، STAILQ_FIRST، STAILQ_FOREACH، STAILQ_HEAD، STAILQ_HEAD_INITIALIZER، STAILQ_INIT، STAILQ_INSERT_AFTER، STAILQ_INSERT_HEAD، STAILQ_INSERT_TAIL، STAILQ_NEXT، STAILQ_REMOVE، STAILQ_REMOVE_HEAD، - تنفيذ طابور ذيل مرتبط بشكل أحادي

المكتبة

مكتبة C القياسية (libc، -lc)

موجز

#include <sys/queue.h>
STAILQ_ENTRY(TYPE);
STAILQ_HEAD(HEADNAME, TYPE);
STAILQ_HEAD STAILQ_HEAD_INITIALIZER(STAILQ_HEAD head);
void STAILQ_INIT(STAILQ_HEAD *head);
int STAILQ_EMPTY(STAILQ_HEAD *head);
void STAILQ_INSERT_HEAD(STAILQ_HEAD *head,
                         struct TYPE *elm, STAILQ_ENTRY NAME);
void STAILQ_INSERT_TAIL(STAILQ_HEAD *head,
                         struct TYPE *elm, STAILQ_ENTRY NAME);
void STAILQ_INSERT_AFTER(STAILQ_HEAD *head, struct TYPE *listelm,
                         struct TYPE *elm, STAILQ_ENTRY NAME);
struct TYPE *STAILQ_FIRST(STAILQ_HEAD *head);
struct TYPE *STAILQ_NEXT(struct TYPE *elm, STAILQ_ENTRY NAME);
STAILQ_FOREACH(struct TYPE *var, STAILQ_HEAD *head, STAILQ_ENTRY NAME);
void STAILQ_REMOVE(STAILQ_HEAD *head, struct TYPE *elm, TYPE,
                         STAILQ_ENTRY NAME);
void STAILQ_REMOVE_HEAD(STAILQ_HEAD *head,
                         STAILQ_ENTRY NAME);
void STAILQ_CONCAT(STAILQ_HEAD *head1, STAILQ_HEAD *head2);

ملاحظة: توجد وحدات ماكرو متطابقة مبدوءة بـ SIMPLEQ بدلاً من STAILQ؛ انظر VERSIONS.

الوصف

تعرف هذه الماكروات وتعمل على طوابير ذيل مرتبطة بشكل أحادي.

في تعريفات الماكرو، TYPE هو اسم هيكل معرف من قبل المستخدم، يجب أن يحتوي على حقل من النوع STAILQ_ENTRY، يسمى NAME. الوسيط HEADNAME هو اسم هيكل معرف من قبل المستخدم يجب الإعلان عنه باستخدام الماكرو STAILQ_HEAD().

الإنشاء

يتم ترؤيس طابور ذيل مرتبط بشكل أحادي بهيكل تم تعريفه بواسطة الماكرو STAILQ_HEAD(). يحتوي هذا الهيكل على زوج من المؤشرات، أحدهما يشير إلى العنصر الأول في طابور الذيل والآخر يشير إلى العنصر الأخير في طابور الذيل. العناصر مرتبطة بشكل أحادي لتقليل استهلاك المساحة وتعقيد معالجة المؤشرات للعناصر العشوائية، مقابل وقت O(n) للإزالة. يمكن إضافة عناصر جديدة إلى طابور الذيل بعد عنصر موجود، أو في بداية طابور الذيل، أو في نهاية طابور الذيل. يتم الإعلان عن هيكل STAILQ_HEAD على النحو التالي:


STAILQ_HEAD(HEADNAME, TYPE) head;

حيث struct HEADNAME هو الهيكل المراد تعريفه، و struct TYPE هو نوع العناصر المراد ربطها في طابور الذيل. يمكن لاحقًا الإعلان عن مؤشر إلى رأس طابور الذيل على النحو التالي:


struct HEADNAME *headp;

(الأسماء head و headp قابلة للاختيار من قبل المستخدم.)

يعلن STAILQ_ENTRY() عن هيكل يربط العناصر في طابور الذيل.

يقيّم STAILQ_HEAD_INITIALIZER() إلى مُهيئ لطابور الذيل head.

يقوم STAILQ_INIT() بتهيئة طابور الذيل الذي يشير إليه head.

يقيّم STAILQ_EMPTY() إلى true إذا لم تكن هناك عناصر في طابور الذيل.

الإدراج

يقوم STAILQ_INSERT_HEAD() بإدراج العنصر الجديد elm في بداية طابور الذيل.

يقوم STAILQ_INSERT_TAIL() بإدراج العنصر الجديد elm في نهاية طابور الذيل.

يقوم STAILQ_INSERT_AFTER() بإدراج العنصر الجديد elm بعد العنصر listelm.

التنقل

يعيد STAILQ_FIRST() العنصر الأول في طابور الذيل أو NULL إذا كان طابور الذيل فارغًا.

يعيد STAILQ_NEXT() العنصر التالي في طابور الذيل، أو NULL إذا كان هذا العنصر هو الأخير.

يقوم STAILQ_FOREACH() بالمرور عبر طابور الذيل الذي يشير إليه head في الاتجاه الأمامي، مع تعيين كل عنصر بالتناوب إلى var.

الإزالة

يقوم STAILQ_REMOVE() بإزالة العنصر elm من طابور الذيل.

يقوم STAILQ_REMOVE_HEAD() بإزالة العنصر الموجود في بداية طابور الذيل. للحصول على أقصى قدر من الكفاءة، يجب على العناصر التي تتم إزالتها من بداية طابور الذيل استخدام هذا الماكرو صراحةً بدلاً من الماكرو العام STAILQ_REMOVE().

ميزات أخرى

يقوم STAILQ_CONCAT() بربط طابور الذيل الذي يرأسه head2 بنهاية الطابور الذي يرأسه head1، مع إزالة جميع الإدخالات من الأول.

قيمة الإرجاع

يعيد STAILQ_EMPTY() قيمة غير صفرية إذا كان الطابور فارغًا، وصفرًا إذا كان الطابور يحتوي على إدخال واحد على الأقل.

يعيد STAILQ_FIRST() و STAILQ_NEXT() مؤشرًا إلى الهيكل TYPE الأول أو التالي، على التوالي.

يعيد STAILQ_HEAD_INITIALIZER() مُهيئًا يمكن تعيينه إلى رأس الطابور head.

الإصدارات

توفر بعض أنظمة BSD الواجهة SIMPLEQ بدلاً من STAILQ. وهما متطابقتان، ولكن لأسباب تاريخية تم تسميتهما بشكل مختلف في أنظمة BSD المختلفة. نشأ STAILQ في FreeBSD، ونشأ SIMPLEQ في NetBSD. لأسباب تتعلق بالتوافق، توفر بعض الأنظمة كلا المجموعتين من الماكروات. توفر glibc كلاً من STAILQ و SIMPLEQ، وهما متطابقان باستثناء عدم وجود مكافئ لـ SIMPLEQ لـ STAILQ_CONCAT().

الأخطاء

لا يسمح STAILQ_FOREACH() بإزالة أو تحرير var داخل الحلقة، لأنه قد يتداخل مع التنقل. STAILQ_FOREACH_SAFE()، وهو موجود في أنظمة BSD ولكنه غير موجود في glibc، يصلح هذا القيد عن طريق السماح بإزالة var بأمان من القائمة وتحريرها من داخل الحلقة دون التداخل مع التنقل.

أمثلة

#include <stddef.h>
#include <stdio.h>
#include <stdlib.h>
#include <sys/queue.h>
struct entry {
    int data;
    STAILQ_ENTRY(entry) entries;        /* طابور ذيل مرتبط بشكل أحادي */
};
STAILQ_HEAD(stailhead, entry);
int
main(void)
{
    struct entry *n1, *n2, *n3, *np;
    struct stailhead head;                  /* رأس طابور الذيل المرتبط
                                               بشكل أحادي */
    STAILQ_INIT(&head);                     /* تهيئة الطابور */
    n1 = malloc(sizeof(struct entry));      /* إدراج في البداية */
    STAILQ_INSERT_HEAD(&head, n1, entries);
    n1 = malloc(sizeof(struct entry));      /* إدراج في النهاية */
    STAILQ_INSERT_TAIL(&head, n1, entries);
    n2 = malloc(sizeof(struct entry));      /* إدراج بعد */
    STAILQ_INSERT_AFTER(&head, n1, n2, entries);
    STAILQ_REMOVE(&head, n2, entry, entries); /* حذف */
    free(n2);
    n3 = STAILQ_FIRST(&head);
    STAILQ_REMOVE_HEAD(&head, entries);     /* حذف من البداية */
    free(n3);
    n1 = STAILQ_FIRST(&head);
    n1->data = 0;
    for (unsigned int i = 1; i < 5; i++) {
        n1 = malloc(sizeof(struct entry));
        STAILQ_INSERT_HEAD(&head, n1, entries);
        n1->data = i;
    }
