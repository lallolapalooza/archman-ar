subroutine stgsen (ijob, wantq, wantz, select, n, a, lda, b, ldb, alphar, alphai, beta, q, ldq, z, ldz, m, pl, pr, dif, work, lwork, iwork, liwork, info)
STGSEN

توثيق الوظيفة/الروتين الفرعي

الروتين الفرعي STGSEN (integer ijob, logical wantq, logical wantz, logical, dimension( * ) select, integer n, real, dimension( lda, * ) a, integer lda, real, dimension( ldb, * ) b, integer ldb, real, dimension( * ) alphar, real, dimension( * ) alphai, real, dimension( * ) beta, real, dimension( ldq, * ) q, integer ldq, real, dimension( ldz, * ) z, integer ldz, integer m, real pl, real pr, real, dimension( * ) dif, real, dimension( * ) work, integer lwork, integer, dimension( * ) iwork, integer liwork, integer info)

STGSEN

الغرض:

!>
!> STGSEN تعيد ترتيب تفكيك Schur الحقيقي المعمم لمصفوفة زوج حقيقية (A, B)
!> (من حيث تحويل معادلة متعامدة Q**T * (A, B) * Z)، بحيث تظهر مجموعة مختارة من
!> القيم الذاتية في الكتل القطرية الرائدة للمصفوفة A شبه المثلثية العلوية
!> والمصفوفة B المثلثية العلوية. الأعمدة الرائدة لـ Q و Z تشكل قواعد متعامدة
!> لمساحات القيم الذاتية اليسرى واليمنى المقابلة (مساحات التفكيك). يجب أن تكون (A, B)
!> في شكل Schur القياسي الحقيقي المعمم (كما تم إرجاعه بواسطة SGGES)، أي A
!> هي مصفوفة شبه مثلثية علوية بكتل قطرية بحجم 1x1 و 2x2. B هي مصفوفة
!> مثلثية علوية.
!>
!> تقوم STGSEN أيضًا بحساب القيم الذاتية المعممة
!>
!>             w(j) = (ALPHAR(j) + i*ALPHAI(j))/BETA(j)
!>
!> للمصفوفة الزوجية المعاد ترتيبها (A, B).
!>
!> اختياريًا، تقوم STGSEN بحساب تقديرات للأرقام الشرطية المعكوسة
!> للقيم الذاتية ومساحات القيم الذاتية. هذه هي Difu[(A11,B11), (A22,B22)]
!> و Difl[(A11,B11), (A22,B22)]، أي الفصل (الفصل)
!> بين أزواج المصفوفات (A11, B11) و (A22,B22) التي تتوافق مع
!> المجموعة المختارة والقيم الذاتية خارج المجموعة، على التوالي،
!> ومعايير مساحات القيم الذاتية اليسرى واليمنى بالنسبة
!> للمجموعة المختارة في الكتلة (1,1).
!>

المعلمات

IJOB 
!>          IJOB هو عدد صحيح
!>          يحدد ما إذا كانت الأرقام الشرطية مطلوبة للمجموعة
!>          من القيم الذاتية (PL و PR) أو مساحات التفكيك
!>          (Difu و Difl):
!>           =0: إعادة الترتيب فقط فيما يتعلق بـ SELECT. لا إضافات.
!>           =1: مقلوب معايير مساحات القيم الذاتية اليسرى واليمنى
!>               بالنسبة للمجموعة المختارة (PL و PR).
!>           =2: حدود عليا لـ Difu و Difl. تقدير يعتمد على معيار F
!>               (DIF(1:2)).
!>           =3: تقدير لـ Difu و Difl. تقدير يعتمد على معيار 1
!>               (DIF(1:2)).
!>               حوالي 5 مرات أغلى من IJOB = 2.
!>           =4: حساب PL، PR و DIF (أي 0، 1 و 2 أعلاه): نسخة اقتصادية
!>               للحصول على كل شيء.
!>           =5: حساب PL، PR و DIF (أي 0، 1 و 3 أعلاه)
!>

WANTQ

!>          WANTQ هو منطقي
!>          .TRUE. : تحديث مصفوفة التحويل اليسرى Q؛
!>          .FALSE.: لا تحدث Q.
!>

WANTZ

!>          WANTZ هو منطقي
!>          .TRUE. : تحديث مصفوفة التحويل اليمنى Z؛
!>          .FALSE.: لا تحدث Z.
!>

SELECT

!>          SELECT هو مصفوفة منطقية، أبعادها (N)
!>          تحدد SELECT القيم الذاتية في المجموعة المختارة.
!>          لاختيار قيمة ذاتية حقيقية w(j)، يجب تعيين SELECT(j) إلى
!>          .TRUE.. لاختيار زوج مترافق من القيم الذاتية
!>          w(j) و w(j+1)، المقابلة لكتلة قطرية بحجم 2x2،
!>          يجب تعيين إما SELECT(j) أو SELECT(j+1) أو كليهما إلى
!>          .TRUE.؛ يجب أن يكون زوج مترافق من القيم الذاتية
!>          إما مضمنًا بالكامل في المجموعة أو مستبعدًا بالكامل.
!>

N

!>          N هو عدد صحيح
!>          رتبة المصفوفتين A و B. N >= 0.
!>

A

!>          A هي مصفوفة حقيقية، أبعادها(LDA,N)
!>          عند الإدخال، المصفوفة A شبه المثلثية العلوية، مع (A, B) في
!>          شكل Schur القياسي الحقيقي المعمم.
!>          عند الخروج، يتم الكتابة فوق A بالمصفوفة المعاد ترتيبها A.
!>

LDA

!>          LDA هو عدد صحيح
!>          البعد الرائد للمصفوفة A. LDA >= max(1,N).
!>

B

!>          B هي مصفوفة حقيقية، أبعادها(LDB,N)
!>          عند الإدخال، المصفوفة B المثلثية العلوية، مع (A, B) في
!>          شكل Schur القياسي الحقيقي المعمم.
!>          عند الخروج، يتم الكتابة فوق B بالمصفوفة المعاد ترتيبها B.
!>

LDB

!>          LDB هو عدد صحيح
!>          البعد الرائد للمصفوفة B. LDB >= max(1,N).
!>

ALPHAR

!>          ALPHAR هي مصفوفة حقيقية، أبعادها (N)
!>

ALPHAI

!>          ALPHAI هي مصفوفة حقيقية، أبعادها (N)
!>

BETA

!>          BETA هي مصفوفة حقيقية، أبعادها (N)
!>
!>          عند الخروج، (ALPHAR(j) + ALPHAI(j)*i)/BETA(j)، j=1,...,N، ستكون
!>          القيم الذاتية المعممة. ALPHAR(j) + ALPHAI(j)*i
!>          و BETA(j)، j=1,...,N هي أقطار الشكل Schur المعقد (S,T)
!>          الذي سينتج إذا تم تقليل الكتل القطرية بحجم 2x2 من شكل
!>          Schur الحقيقي المعمم لـ (A,B) بشكل أكبر إلى شكل مثلثي
!>          باستخدام تحويلات وحدوية معقدة.
!>          إذا كانت ALPHAI(j) صفرًا، فإن القيمة الذاتية j حقيقية؛ إذا كانت
!>          موجبة، فإن القيمتين الذاتيتين j و (j+1) زوج مترافق معقد، مع
!>          ALPHAI(j+1) سالب.
!>

Q

!>          Q هي مصفوفة حقيقية، أبعادها (LDQ,N)
!>          عند الإدخال، إذا كان WANTQ = .TRUE.، فإن Q مصفوفة بحجم N x N.
!>          عند الخروج، تم ضرب Q بشكل لاحق بمصفوفة التحويل المتعامدة اليسرى
!>          التي تعيد ترتيب (A, B)؛ الأعمدة M الرائدة لـ Q تشكل قواعد متعامدة
!>          لأزواج مساحات القيم الذاتية اليسرى (مساحات التفكيك) المحددة.
!>          إذا كان WANTQ = .FALSE.، لا يتم الرجوع إلى Q.
!>

LDQ

!>          LDQ هو عدد صحيح
!>          البعد الرائد للمصفوفة Q. LDQ >= 1؛
!>          وإذا كان WANTQ = .TRUE.، فإن LDQ >= N.
!>

Z

!>          Z هي مصفوفة حقيقية، أبعادها (LDZ,N)
!>          عند الإدخال، إذا كان WANTZ = .TRUE.، فإن Z مصفوفة بحجم N x N.
!>          عند الخروج، تم ضرب Z بشكل لاحق بمصفوفة التحويل المتعامدة اليسرى
!>          التي تعيد ترتيب (A, B)؛ الأعمدة M الرائدة لـ Z تشكل قواعد متعامدة
!>          لأزواج مساحات القيم الذاتية اليسرى (مساحات التفكيك) المحددة.
!>          إذا كان WANTZ = .FALSE.، لا يتم الرجوع إلى Z.
!>

LDZ

!>          LDZ هو عدد صحيح
!>          البعد الرائد للمصفوفة Z. LDZ >= 1؛
!>          إذا كان WANTZ = .TRUE.، فإن LDZ >= N.
!>

M

!>          M هو عدد صحيح
!>          بعد زوج مساحات القيم الذاتية اليسرى واليمنى (مساحات التفكيك)
!>          المحدد. 0 <= M <= N.
!>

PL

!>          PL هو حقيقي
!>

PR

!>          PR هو حقيقي
!>
!>          إذا كان IJOB = 1، 4 أو 5، PL، PR هي حدود سفلى لمقلوب
!>          معيار مساحات القيم الذاتية اليسرى واليمنى
!>          بالنسبة للمجموعة المختارة.
!>          0 < PL، PR <= 1.
!>          إذا كان M = 0 أو M = N، فإن PL = PR = 1.
!>          إذا كان IJOB = 0، 2 أو 3، PL و PR غير مستخدمين.
!>

DIF

!>          DIF هو حقيقي، أبعادها (*)
!>          إذا كان IJOB = 2 أو 3، DIF(1) يحتوي على تقدير لـ Difu.
!>          إذا كان IJOB = 2 أو 3، DIF(2) يحتوي على تقدير لـ Difl.
!>          0 <= DIF(1), DIF(2) <= 1.
!>          إذا كان M = 0 أو M = N، فإن DIF(1) = DIF(2) = 0.
!>          إذا كان IJOB = 0، 1، 4 أو 5، فإن DIF غير مستخدم.
!>

WORK

!>          WORK هو حقيقي، أبعادها (*)
!>          هذه هي مصفوفة العمل. يمكن أن تكون مساحة عمل مؤقتة.
!>          إذا كان LWORK = -1، يتم حساب البعد المطلوب لـ WORK
!>          وتخزينه في WORK(1).
!>

LWORK

!>          LWORK هو عدد صحيح
!>          البعد المطلوب لمصفوفة العمل WORK.
!>          للحساب الصحيح، LWORK >= max(1, 2*N).
!>          إذا كان LWORK = -1، فإن البعد المطلوب يتم حسابه وتخزينه في WORK(1).
!>

IWORK

!>          IWORK هي مصفوفة أعداد صحيحة، أبعادها (*)
!>          هذه هي مصفوفة عمل للأعداد الصحيحة.
!>          إذا كان LIWORK = -1، يتم حساب البعد المطلوب لـ IWORK
!>          وتخزينه في IWORK(1).
!>

LIWORK

!>          LIWORK هو عدد صحيح
!>          البعد المطلوب لمصفوفة العمل IWORK.
!>          إذا كان LIWORK = -1، فإن البعد المطلوب يتم حسابه وتخزينه في IWORK(1).
!>
