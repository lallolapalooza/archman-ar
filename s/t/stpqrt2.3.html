subroutine stpqrt2 (integer m, integer n, integer l, real, dimension( lda, * ) a, integer lda, real, dimension( ldb, * ) b, integer ldb, real, dimension( ldt, * ) t, integer ldt, integer info)

STPQRT2 تحسب تفاضل QR لمصفوفة حقيقية أو مركبة "مثلثة-خماسية"، والتي تتكون من كتلة مثلثة وكتلة خماسية، باستخدام تمثيل WY المدمج لـ Q.

الغرض:

!>
!> STPQRT2 تحسب تفاضل QR لمصفوفة حقيقية C، والتي تتكون من كتلة مثلثة A وكتلة خماسية B،
!> باستخدام تمثيل WY المدمج لـ Q.
!>

تفاصيل إضافية:

!>
!> المصفوفة المدخلة C هي مصفوفة بحجم (M+N)-by-N
!>
!>               C = [ A ]
!>                   [ B ]
!>
!> حيث A هي مصفوفة مثلثية علوية بحجم N*N، و B هي مصفوفة خماسية بحجم M*N
!> تتكون من مصفوفة مستطيلة B1 بحجم (M-L)*N في الأعلى فوق مصفوفة شبه منحرفة علوية B2 بحجم L*N:
!>
!>               B = [ B1 ]  <- (M-L)-by-N مستطيلة
!>                   [ B2 ]  <-     L-by-N شبه منحرفة علوية.
!>
!> تتكون المصفوفة شبه المنحرفة العلوية B2 من الصفوف L الأولى لمصفوفة مثلثية علوية بحجم N*N، حيث 0 <= L <= MIN(M,N). إذا كان L=0، فإن B مستطيلة M*N؛ إذا كان M=L=N، فإن B مثلثية علوية.
!>
!> المصفوفة W تخزن المبدلات الأولية H(i) في العمود i تحت القطر (في A) في المصفوفة المدخلة C بحجم (M+N)*N
!>
!>               C = [ A ]  <- مثلثة علوية N*N
!>                   [ B ]  <- خماسية M*N
!>
!> بحيث يمكن تمثيل W كـ
!>
!>               W = [ I ]  <- الهوية، N*N
!>                   [ V ]  <- M*N، بنفس شكل B.
!>
!> وبالتالي، تقع جميع المعلومات اللازمة لـ W عند الخروج في B، والتي
!> نسميها V أعلاه. لاحظ أن V لها نفس شكل B؛ أي،
!>
!>               V = [ V1 ] <- (M-L)-by-N مستطيلة
!>                   [ V2 ] <-     L-by-N شبه منحرفة علوية.
!>
!> أعمدة V تمثل المتجهات التي تحدد H(i).
!> المبدل الكتلي H بحجم (M+N)*(M+N) يُعطى بعد ذلك بواسطة
!>
!>               H = I - W * T * W^H
!>
!> حيث W^H هو المرافق المتعامد لـ W وهو T العامل المثلثي العلوي للمبدل الكتلي.
!>
