subroutine stgevc (character side, character howmny, logical, dimension( * ) select, integer n, real, dimension( lds, * ) s, integer lds, real, dimension( ldp, * ) p, integer ldp, real, dimension( ldvl, * ) vl, integer ldvl, real, dimension( ldvr, * ) vr, integer ldvr, integer mm, integer m, real, dimension( * ) work, integer info)

STGEVC

الغرض:

!>
!> STGEVC يحسب بعض أو كل المتجهات الذاتية اليمنى و/أو اليسرى لزوج من المصفوفات الحقيقية (S,P)، حيث S مصفوفة شبه مثلثة و P مصفوفة مثلثة عليا. أزواج المصفوفات من هذا النوع تنتج من التحلل القياسي المعمم لمصفوفة زوج (A,B):
!>
!>    A = Q*S*Z**T,  B = Q*P*Z**T
!>
!> كما تم حسابه بواسطة SGGHRD + SHGEQZ.
!>
!> المتجه الذاتي الأيمن x والمتجه الذاتي الأيسر y لـ (S,P) المقابلين لقيمة ذاتية w معرفين بـ:
!>
!>    S*x = w*P*x,  (y**H)*S = w*(y**H)*P,
!>
!> حيث y**H يمثل النسخة المترافقة لـ y.
!> القيم الذاتية ليست مدخلات لهذه الروتين، ولكن يتم حسابها مباشرة من الكتل القطرية لـ S و P.
!>
!> هذا الروتين يعيد المصفوفات X و/أو Y للمتجهات الذاتية اليمنى واليسرى لـ (S,P)، أو حاصل الضرب Z*X و/أو Q*Y،
!> حيث Z و Q مصفوفات مدخلات.
!> إذا كانت Q و Z هي العوامل المتعامدة من التحلل القياسي المعمم لزوج مصفوفات (A,B)، فإن Z*X و Q*Y
!> هي مصفوفات المتجهات الذاتية اليمنى واليسرى لـ (A,B).
!>
!>

المعلمات

SIDE 
!>          SIDE هو CHARACTER*1
!>          = 'R': حساب المتجهات الذاتية اليمنى فقط؛
!>          = 'L': حساب المتجهات الذاتية اليسرى فقط؛
!>          = 'B': حساب المتجهات الذاتية اليمنى واليسرى كليهما.
!>

HOWMNY

!>          HOWMNY هو CHARACTER*1
!>          = 'A': حساب جميع المتجهات الذاتية اليمنى و/أو اليسرى؛
!>          = 'B': حساب جميع المتجهات الذاتية اليمنى و/أو اليسرى،
!>                 محوّلة عكسياً بالمصفوفات في VR و/أو VL؛
!>          = 'S': حساب المتجهات الذاتية اليمنى و/أو اليسرى المختارة،
!>                 المحددة بواسطة المصفوفة المنطقية SELECT.
!>

SELECT

!>          SELECT هي مصفوفة LOGICAL، البعد (N)
!>          إذا كان HOWMNY='S'، فإن SELECT تحدد المتجهات الذاتية التي سيتم
!>          حسابها. إذا كانت w(j) قيمة ذاتية حقيقية، يتم حساب المتجه الذاتي
!>          الحقيقي المقابل إذا كان SELECT(j) هو .TRUE..
!>          إذا كانت w(j) و w(j+1) الأجزاء الحقيقية والتخيلية لقيمة ذاتية
!>          مركبة، يتم حساب المتجه الذاتي المركب إذا كان إما SELECT(j) أو SELECT(j+1)
!>          هو .TRUE.، وعند الخروج يتم تعيين SELECT(j) إلى .TRUE. و SELECT(j+1)
!>          إلى .FALSE..
!>          لا يتم الإشارة إليها إذا كان HOWMNY = 'A' أو 'B'.
!>

N

!>          N هو INTEGER
!>          رتبة المصفوفات S و P. N >= 0.
!>

S

!>          S هي مصفوفة REAL، البعد (LDS,N)
!>          المصفوفة شبه المثلثية العليا S من تحلل قياسي معمّم،
!>          كما تم حسابه بواسطة SHGEQZ.
!>

LDS

!>          LDS هو INTEGER
!>          البعد الرئيسي للمصفوفة S. LDS >= max(1,N).
!>

P

!>          P هي مصفوفة REAL، البعد (LDP,N)
!>          المصفوفة المثلثية العليا P من تحلل قياسي معمّم،
!>          كما تم حسابه بواسطة SHGEQZ.
!>          يجب أن تكون الكتل القطرية 2x2 في P المقابلة للكتل 2x2
!>          في S في صيغة قطرية موجبة.
!>

LDP

!>          LDP هو INTEGER
!>          البعد الرئيسي للمصفوفة P. LDP >= max(1,N).
!>

VL

!>          VL هي مصفوفة REAL، البعد (LDVL,MM)
!>          عند الإدخال، إذا كان SIDE = 'L' أو 'B' و HOWMNY = 'B'، يجب أن
!>          تحتوي VL على مصفوفة Q بحجم N*N (عادةً المصفوفة المتعامدة Q
!>          للمتجهات القياسية اليسرى التي تم إرجاعها بواسطة SHGEQZ).
!>
!>          عند الخروج، إذا كان SIDE = 'L' أو 'B'، تحتوي VL على:
!>          إذا كان HOWMNY = 'A'، المصفوفة Y للمتجهات الذاتية اليسرى لـ (S,P)؛
!>          إذا كان HOWMNY = 'B'، المصفوفة Q*Y؛
!>          إذا كان HOWMNY = 'S'، المتجهات الذاتية اليسرى لـ (S,P) المحددة
!>                      بواسطة SELECT، مخزنة بالتتابع في أعمدة
!>                      VL، بنفس ترتيب قيمها الذاتية.
!>
!>          يتم تخزين متجه ذاتي مركب مقابل قيمة ذاتية مركبة
!>          في عمودين متتاليين، الأول يحمل الجزء الحقيقي، والثاني الجزء التخيلي.
!>
!>          لا يتم الإشارة إليها إذا كان SIDE = 'R'.
!>

LDVL

!>          LDVL هو INTEGER
!>          البعد الرئيسي للمصفوفة VL. LDVL >= 1، وإذا
!>          كان SIDE = 'L' أو 'B'، فإن LDVL >= N.
!>

VR

!>          VR هي مصفوفة REAL، البعد (LDVR,MM)
!>          عند الإدخال، إذا كان SIDE = 'R' أو 'B' و HOWMNY = 'B'، يجب أن
!>          تحتوي VR على مصفوفة Z بحجم N*N (عادةً المصفوفة المتعامدة Z
!>          للمتجهات القياسية اليمنى التي تم إرجاعها بواسطة SHGEQZ).
!>
!>          عند الخروج، إذا كان SIDE = 'R' أو 'B'، تحتوي VR على:
!>          إذا كان HOWMNY = 'A'، المصفوفة X للمتجهات الذاتية اليمنى لـ (S,P)؛
!>          إذا كان HOWMNY = 'B' أو 'b'، المصفوفة Z*X؛
!>          إذا كان HOWMNY = 'S' أو 's'، المتجهات الذاتية اليمنى لـ (S,P)
!>                      المحددة بواسطة SELECT، مخزنة بالتتابع في
!>                      أعمدة VR، بنفس ترتيب قيمها الذاتية.
!>
!>          يتم تخزين متجه ذاتي مركب مقابل قيمة ذاتية مركبة
!>          في عمودين متتاليين، الأول يحمل الجزء الحقيقي والثاني الجزء التخيلي.
!>
!>          لا يتم الإشارة إليها إذا كان SIDE = 'L'.
!>

LDVR

!>          LDVR هو INTEGER
!>          البعد الرئيسي للمصفوفة VR. LDVR >= 1، وإذا
!>          كان SIDE = 'R' أو 'B'، فإن LDVR >= N.
!>

MM

!>          MM هو INTEGER
!>          عدد الأعمدة في المصفوفات VL و/أو VR. MM >= M.
!>

M

!>          M هو INTEGER
!>          عدد الأعمدة في المصفوفات VL و/أو VR المستخدمة فعليًا
!>          لتخزين المتجهات الذاتية. إذا كان HOWMNY = 'A' أو 'B'، يتم تعيين M
!>          إلى N. كل متجه ذاتي حقيقي مختار يشغل عمودًا واحدًا وكل متجه ذاتي
!>          مركب مختار يشغل عمودين.
!>

WORK

!>          WORK هي مصفوفة REAL، البعد (6*N)
!>

INFO

!>          INFO هو INTEGER
!>          = 0:  خروج ناجح.
!>          < 0:  إذا كان INFO = -i، فإن المعلمة i كانت ذات قيمة غير قانونية.
!>          > 0:  الكتلة 2x2 (INFO:INFO+1) لا تحتوي على قيمة ذاتية مركبة.
!>
