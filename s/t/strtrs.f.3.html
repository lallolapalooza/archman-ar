subroutine strtrs (character uplo, character trans, character diag, integer n, integer nrhs, real, dimension( lda, * ) a, integer lda, real, dimension( ldb, * ) b, integer ldb, integer info)

STRTRS

الغرض:

!>
!> STRTRS يحل نظامًا مثلثيًا بالشكل:
!>
!>    A * X = B  أو  A**T * X = B،
!>
!> حيث A مصفوفة مثلثية من الرتبة N، و B مصفوفة N × NRHS.
!>
!> يتحقق هذا الروتين الفرعي من أن A غير شاذة، ولكن يجب ملاحظة أن الشذوذ الدقيق فقط هو الذي يتم اكتشافه.
!> من الممكن أن تكون عناصر قطرية واحدة أو أكثر في A أرقامًا صغيرة جدًا دون أن يشير هذا الروتين الفرعي إلى خطأ.
!>
!> إذا كان فقدان الدقة العددية المحتمل بسبب المصفوفات القريبة من الشذوذ مصدر قلق،
!> فيجب على المستدعي التحقق من أن A غير شاذة ضمن بعض الحدود قبل استدعاء هذا الروتين الفرعي.
!>

المعاملات

UPLO 
!>          UPLO هو CHARACTER*1
!>          = 'U':  A مثلثية عليا؛
!>          = 'L':  A مثلثية سفلى.
!>

TRANS

!>          TRANS هو CHARACTER*1
!>          يحدد شكل نظام المعادلات:
!>          = 'N':  A * X = B  (بدون تبديل)
!>          = 'T':  A**T * X = B  (تبديل)
!>          = 'C':  A**H * X = B  (تبديل مرافق = تبديل)
!>

DIAG

!>          DIAG هو CHARACTER*1
!>          = 'N':  A مثلثية غير وحدوية؛
!>          = 'U':  A مثلثية وحدوية.
!>

N

!>          N هو INTEGER
!>          رتبة المصفوفة A.  N >= 0.
!>

NRHS

!>          NRHS هو INTEGER
!>          عدد الأطراف اليمنى، أي عدد أعمدة
!>          المصفوفة B.  NRHS >= 0.
!>

A

!>          A هي مصفوفة REAL، الأبعاد (LDA,N)
!>          المصفوفة المثلثية A.  إذا كان UPLO = 'U'، فإن الجزء العلوي المثلثي العلوي
!>          من المصفوفة A الذي يتكون من N صفًا و N عمودًا يحتوي على المصفوفة المثلثية العليا،
!>          والجزء السفلي المثلثي الصارم في A لا يتم الإشارة إليه.  إذا كان UPLO = 'L'،
!>          فإن الجزء السفلي المثلثي السفلي الذي يتكون من N صفًا و N عمودًا في المصفوفة A
!>          يحتوي على المصفوفة المثلثية السفلى، والجزء العلوي المثلثي الصارم في A لا يتم
!>          الإشارة إليه.  إذا كان DIAG = 'U'، فإن العناصر القطرية في A لا يتم
!>          الإشارة إليها أيضًا ويفترض أنها 1.
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرئيسي للمصفوفة A.  LDA >= max(1,N).
!>

B

!>          B هي مصفوفة REAL، الأبعاد (LDB,NRHS)
!>          عند الإدخال، مصفوفة الطرف الأيمن B.
!>          عند الخروج، إذا كان INFO = 0، فإن مصفوفة الحل X.
!>

LDB

!>          LDB هو INTEGER
!>          البعد الرئيسي للمصفوفة B.  LDB >= max(1,N).
!>

INFO

!>          INFO هو INTEGER
!>          = 0:  الانتهاء بنجاح
!>          < 0: إذا كان INFO = -i، فإن المعامل i كان له قيمة غير قانونية
!>          > 0:  إذا كان INFO = i، فإن العنصر القطري i في A يساوي صفرًا بالضبط،
!>               مما يشير إلى أن المصفوفة شاذة ولم يتم حساب حلول
!>               X.
!>
