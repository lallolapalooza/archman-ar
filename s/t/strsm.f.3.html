subroutine strsm (character side, character uplo, character transa, character diag, integer m, integer n, real alpha, real, dimension(lda,*) a, integer lda, real, dimension(ldb,*) b, integer ldb)

STRSM

الغرض:

!>
!> STRSM  يحل إحدى معادلات المصفوفات التالية:
!>
!>    op( A )*X = alpha*B,   أو   X*op( A ) = alpha*B,
!>
!> حيث alpha ثابت، X و B مصفوفتان m x n، A مصفوفة مثلثة عليا أو سفلى، موحدة أو غير موحدة، و op( A ) هي إحدى
!>
!>    op( A ) = A   أو   op( A ) = A**T.
!>
!> يتم استبدال المصفوفة X بالمصفوفة B.
!>

المعلمات:

SIDE 
!>          SIDE هو CHARACTER*1
!>           عند الإدخال، تحدد SIDE ما إذا كانت op( A ) تظهر على اليسار
!>           أو اليمين من X كما يلي:
!>
!>              SIDE = 'L' أو 'l'   op( A )*X = alpha*B.
!>
!>              SIDE = 'R' أو 'r'   X*op( A ) = alpha*B.
!>

UPLO

!>          UPLO هو CHARACTER*1
!>           عند الإدخال، تحدد UPLO ما إذا كانت المصفوفة A مصفوفة مثلثة عليا أو
!>           سفلى كما يلي:
!>
!>              UPLO = 'U' أو 'u'   A هي مصفوفة مثلثة عليا.
!>
!>              UPLO = 'L' أو 'l'   A هي مصفوفة مثلثة سفلى.
!>

TRANSA

!>          TRANSA هو CHARACTER*1
!>           عند الإدخال، تحدد TRANSA شكل op( A ) المستخدم في
!>           ضرب المصفوفات كما يلي:
!>
!>              TRANSA = 'N' أو 'n'   op( A ) = A.
!>
!>              TRANSA = 'T' أو 't'   op( A ) = A**T.
!>
!>              TRANSA = 'C' أو 'c'   op( A ) = A**T.
!>

DIAG

!>          DIAG هو CHARACTER*1
!>           عند الإدخال، تحدد DIAG ما إذا كانت A موحدة مثلثية أم لا
!>           كما يلي:
!>
!>              DIAG = 'U' أو 'u'   A مفترضة أنها موحدة مثلثية.
!>
!>              DIAG = 'N' أو 'n'   A غير مفترضة أنها موحدة
!>                                  مثلثية.
!>

M

!>          M هو INTEGER
!>           عند الإدخال، تحدد M عدد صفوف B. يجب أن يكون M على الأقل
!>           صفر.
!>

N

!>          N هو INTEGER
!>           عند الإدخال، تحدد N عدد أعمدة B. يجب أن يكون N على الأقل
!>           صفر.
!>

ALPHA

!>          ALPHA هو REAL
!>           عند الإدخال، يحدد ALPHA الثابت alpha. عندما يكون alpha
!>           صفرًا، فإن A لا يتم الرجوع إليها، ولا يلزم تعيين B قبل
!>           الإدخال.
!>

A

!>          A هي مصفوفة REAL، ذات البعد ( LDA, k )،
!>           حيث k يساوي m عندما تكون SIDE = 'L' أو 'l'
!>             و k يساوي n عندما تكون SIDE = 'R' أو 'r'.
!>           قبل الإدخال مع UPLO = 'U' أو 'u'، يجب أن يحتوي الجزء العلوي
!>           المثلثي من المصفوفة A ذو البعد k x k على المصفوفة
!>           المثلثة العليا، ولا يتم الرجوع إلى الجزء المثلثي السفلي
!>           الصارم من A.
!>           قبل الإدخال مع UPLO = 'L' أو 'l'، يجب أن يحتوي الجزء السفلي
!>           المثلثي من المصفوفة A ذو البعد k x k على المصفوفة
!>           المثلثة السفلى، ولا يتم الرجوع إلى الجزء المثلثي العلوي
!>           الصارم من A.
!>           لاحظ أنه عندما يكون DIAG = 'U' أو 'u'، لا يتم الرجوع إلى
!>           العناصر القطرية لـ A أيضًا، ولكنها مفترضة أنها الوحدة.
!>

LDA

!>          LDA هو INTEGER
!>           عند الإدخال، تحدد LDA البعد الأول لـ A كما تم تعريفه
!>           في البرنامج (الفرعي) المستدعي. عندما تكون SIDE = 'L' أو 'l'
!>           فيجب أن يكون LDA على الأقل max( 1, m )، عندما تكون SIDE = 'R' أو 'r'
!>           فيجب أن يكون LDA على الأقل max( 1, n ).
!>

B

!>          B هي مصفوفة REAL، ذات البعد ( LDB, N )
!>           قبل الإدخال، يجب أن يحتوي الجزء الرائد ذو البعد m x n من المصفوفة B على
!>           مصفوفة الجانب الأيمن B، وعند الخروج يتم استبداله بالمصفوفة
!>           الحل X.
!>

LDB

!>          LDB هو INTEGER
!>           عند الإدخال، تحدد LDB البعد الأول لـ B كما تم تعريفه
!>           في البرنامج (الفرعي) المستدعي. يجب أن يكون LDB على الأقل
!>           max( 1, m ).
!>
