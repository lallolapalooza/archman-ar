strcpy(3) دليل وظائف المكتبة strcpy(3)

الاسم

stpcpy, strcpy, strcat - نسخ أو ربط سلسلة نصية

المكتبة

مكتبة C القياسية (libc، -lc)

الملخص

#include <string.h>
char *stpcpy(char *restrict dst, const char *restrict src);
char *strcpy(char *restrict dst, const char *restrict src);
char *strcat(char *restrict dst, const char *restrict src);
متطلبات ميزة الاختبار (Feature Test Macro) لـ glibc (انظر feature_test_macros(7)):

stpcpy():

    منذ glibc 2.10:
        _POSIX_C_SOURCE >= 200809L
    قبل glibc 2.10:
        _GNU_SOURCE

الوصف

stpcpy()
strcpy()
تقوم هذه الوظائف بنسخ السلسلة النصية التي يشير إليها src، إلى مخزن مؤقت في الموقع المشار إليه بواسطة dst. يكون المبرمج مسؤولاً عن تخصيص مخزن مؤقت وجهة كبير بما يكفي، أي strlen(src) + 1. للحصول على الفرق بين الوظيفتين، انظر قيمة الإرجاع.
strcat()
تربط هذه الوظيفة السلسلة النصية التي يشير إليها src، بعد السلسلة النصية التي يشير إليها dst (الكتابة فوق بايتها الصفري الختامي). يكون المبرمج مسؤولاً عن تخصيص مخزن مؤقت وجهة كبير بما يكفي، أي strlen(dst) + strlen(src) + 1.

قد يكون تطبيق لهذه الوظائف كالتالي:


char *
stpcpy(char *restrict dst, const char *restrict src)
{
    char  *p;
    p = mempcpy(dst, src, strlen(src));
    *p = '\0';
    return p;
}
char *
strcpy(char *restrict dst, const char *restrict src)
{
    stpcpy(dst, src);
    return dst;
}
char *
strcat(char *restrict dst, const char *restrict src)
{
    stpcpy(dst + strlen(dst), src);
    return dst;
}

قيمة الإرجاع

stpcpy()
تعيد هذه الوظيفة مؤشراً إلى البايت الصفري الختامي للسلسلة النصية المنسوخة.
strcpy()
strcat()
تعيد هاتان الوظيفتان dst.

السمات

لشرح المصطلحات المستخدمة في هذا القسم، انظر attributes(7).

الواجهة السمة القيمة
stpcpy (), strcpy (), strcat () سلامة المسار (Thread safety) MT-Safe

POSIX.1-2008.

strcat()
C11, POSIX.1-2008.

المعايير

stpcpy()

تحذيرات

لا يجوز أن تتداخل السلاسل النصية src و dst.

إذا لم يكن المخزن المؤقت للوجهة كبيراً بما يكفي، فإن السلوك غير محدد. انظر _FORTIFY_SOURCE في feature_test_macros(7).

strcat() يمكن أن تكون غير فعالة للغاية. اقرأ عن Shlemiel the painter.

أمثلة

#include <err.h>
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
int
main(void)
{
    char    *p;
    char    *buf1;
    char    *buf2;
    size_t  len, size;
    size = strlen("Hello ") + strlen("world") + strlen("!") + 1;
    buf1 = malloc(sizeof(*buf1) * size);
    if (buf1 == NULL)
        err(EXIT_FAILURE, "malloc()");
    buf2 = malloc(sizeof(*buf2) * size);
    if (buf2 == NULL)
        err(EXIT_FAILURE, "malloc()");
    p = buf1;
    p = stpcpy(p, "Hello ");
    p = stpcpy(p, "world");
    p = stpcpy(p, "!");
    len = p - buf1;
    printf("[len = %zu]: ", len);
    puts(buf1);  // "Hello world!"
    free(buf1);
    strcpy(buf2, "Hello ");
    strcat(buf2, "world");
    strcat(buf2, "!");
    len = strlen(buf2);
    printf("[len = %zu]: ", len);
    puts(buf2);  // "Hello world!"
    free(buf2);
    exit(EXIT_SUCCESS);
}
