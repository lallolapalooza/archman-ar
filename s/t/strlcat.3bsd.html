strlcat(3bsd)

الحزمة: libbsd

التراخيص: custom

strlcpy(3bsd) 3bsd strlcpy(3bsd)

الاسم

strlcpy، strlcat — نسخ ودمج السلاسل النصية بحجم محدود

المكتبة

مكتبة “libbsd”

الملخص

#include <string.h> (راجع libbsd(7) لاستخدام تضمين.)
size_t
strlcpy(char *dst, const char *src, size_t size);

size_t
strlcat(char *dst, const char *src, size_t size);

الوصف

تقوم الدوال strlcpy() و strlcat() بنسخ ودمج السلاسل النصية على التوالي. تم تصميمها لتكون بدائل أكثر أمانًا واتساقًا وأقل عرضة للخطأ للدوال strncpy(3) و strncat(3). على عكس تلك الدوال، تأخذ strlcpy() و strlcat() الحجم الكامل للمخزن المؤقت (وليس فقط الطول) وتضمن إنهاء النتيجة بـ NUL (طالما أن size أكبر من 0 أو، في حالة strlcat()، طالما أن هناك بايت واحد على الأقل متاح في dst). لاحظ أنه يجب تضمين بايت لـ NUL في size. لاحظ أيضًا أن strlcpy() و strlcat() تعملان فقط على سلاسل “C” الحقيقية. هذا يعني أنه بالنسبة لـ strlcpy() يجب أن تكون src منتهية بـ NUL، وبالنسبة لـ strlcat() يجب أن تكون كل من src و dst منتهيتين بـ NUL.

تقوم الدالة strlcpy() بنسخ ما يصل إلى size - 1 حرف من السلسلة النصية src المنتهية بـ NUL إلى dst، وإنهاء النتيجة بـ NUL.

تقوم الدالة strlcat() بإلحاق السلسلة النصية src المنتهية بـ NUL بنهاية dst. ستقوم بإلحاق ما لا يزيد عن size - strlen(dst) - 1 بايت، وإنهاء النتيجة بـ NUL.

قيم الإرجاع

تقوم الدوال strlcpy() و strlcat() بإرجاع الطول الإجمالي للسلسلة النصية التي حاولت إنشائها. بالنسبة لـ strlcpy() يعني هذا طول src. بالنسبة لـ strlcat() يعني هذا الطول الأولي لـ dst بالإضافة إلى طول src. بينما قد يبدو هذا مربكًا بعض الشيء، فقد تم القيام به لجعل اكتشاف الاقتطاع بسيطًا.

لاحظ، مع ذلك، أنه إذا تجاوزت strlcat() size حرفًا دون العثور على NUL، فيعتبر طول السلسلة النصية size ولن يتم إنهاء السلسلة النصية الوجهة بـ NUL (نظرًا لعدم وجود مساحة لـ NUL). هذا يمنع strlcat() من الخروج عن نهاية السلسلة النصية. عمليًا، لا ينبغي أن يحدث هذا (حيث يعني أن size غير صحيحة أو أن dst ليست سلسلة نصية “C” مناسبة). الشرط موجود لمنع مشاكل أمنية محتملة في الكود غير الصحيح.

أمثلة

يوضح مقتطف الكود التالي الحالة البسيطة:

char *s, *p, buf[BUFSIZ];

...

(void)strlcpy(buf, s, sizeof(buf));
(void)strlcat(buf, p, sizeof(buf));

لاكتشاف الاقتطاع، ربما أثناء بناء اسم مسار، قد يتم استخدام شيء مثل ما يلي:

char *dir, *file, pname[MAXPATHLEN];

...

if (strlcpy(pname, dir, sizeof(pname)) >= sizeof(pname))
	goto toolong;
if (strlcat(pname, file, sizeof(pname)) >= sizeof(pname))
	goto toolong;

نظرًا لأنه من المعروف عدد الأحرف التي تم نسخها في المرة الأولى، يمكن تسريع الأمور قليلاً باستخدام النسخ بدلاً من الإلحاق:

char *dir, *file, pname[MAXPATHLEN];
size_t n;

...

n = strlcpy(pname, dir, sizeof(pname));
if (n >= sizeof(pname))
	goto toolong;
if (strlcpy(pname + n, file, sizeof(pname) - n) >= sizeof(pname) - n)
	goto toolong;

ومع ذلك، قد يتساءل المرء عن صحة مثل هذه التحسينات، لأنها تلغي الغرض الكامل من strlcpy() و strlcat(). في الواقع، حصلت النسخة الأولى من صفحة الدليل هذه على خطأ.
