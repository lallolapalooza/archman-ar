strfromd(3) دليل وظائف المكتبة strfromd(3)

الاسم

strfromd, strfromf, strfroml - تحويل قيمة نقطة عائمة إلى سلسلة نصية

المكتبة

مكتبة C القياسية (libc، -lc)

الملخص

#include <stdlib.h>
int strfromd(size_t n;
             char str[restrict n], size_t n,
             const char *restrict format, double fp);
int strfromf(size_t n;
             char str[restrict n], size_t n,
             const char *restrict format, float fp);
int strfroml(size_t n;
             char str[restrict n], size_t n,
             const char *restrict format, long double fp);
متطلبات ميزة الاختبار Macro لـ glibc (انظر feature_test_macros(7)):

strfromd(), strfromf(), strfroml():

    __STDC_WANT_IEC_60559_BFP_EXT__

الوصف

تقوم هذه الدوال بتحويل قيمة نقطة عائمة، fp، إلى سلسلة من الأحرف، str، مع سلسلة تنسيق قابلة للتكوين format. يتم تخزين n من الأحرف على الأكثر في str.

يتم كتابة حرف البايت الفارغ الختامي ('\0') فقط إذا كان n كبيرًا بما فيه الكفاية، وإلا فإن السلسلة المكتوبة سيتم اقتطاعها عند n من الأحرف.

الدوال strfromd(), strfromf(), و strfroml() تتطابق مع


snprintf(str, n, format, fp);

باستثناء سلسلة التنسيق.

تنسيق سلسلة التنسيق

يجب أن تبدأ سلسلة التنسيق format بالحرف '%'. يتبع هذا عدد اختياري للدقة يبدأ بحرف النقطة (.), يتبعه عدد صحيح اختياري عشري. إذا لم يتم تحديد عدد صحيح بعد حرف النقطة، يتم استخدام دقة صفر. أخيرًا، يجب أن تحتوي سلسلة التنسيق على أحد محددات التحويل a، A، e، E، f، F، g، أو G.

يتم تطبيق محدد التحويل بناءً على نوع النقطة العائمة الذي تشير إليه لاحقة الدالة. لذلك، على عكس snprintf()، لا تحتوي سلسلة التنسيق على حرف معدل الطول. انظر snprintf(3) لوصف تفصيلي لمحددات التحويل هذه.

يتوافق التنفيذ مع معيار C99 بشأن تحويل قيم NaN واللانهائي:

إذا كانت fp قيمة NaN، أو +NaN، أو -NaN، وكان f (أو a، e، g) هو محدد التحويل، يتم التحويل إلى "nan"، "nan"، أو "-nan"، على التوالي. إذا كان F (أو A، E، G) هو محدد التحويل، يتم التحويل إلى "NAN" أو "-NAN".

وبالمثل، إذا كانت fp قيمة لانهائية، يتم تحويلها إلى [-]inf أو [-]INF.

سلسلة تنسيق غير صحيحة تؤدي إلى سلوك غير معرف.

قيمة الإرجاع

تُرجع الدوال strfromd(), strfromf(), و strfroml() عدد الأحرف التي كان سيتم كتابتها في str إذا كان n كبيرًا بما يكفي، باستثناء البايت الفارغ الختامي. وبالتالي، فإن قيمة إرجاع قدرها n أو أكثر تعني أن الإخراج تم اقتطاعه.

أمثلة

لتحويل القيمة 12.1 كنوع float إلى سلسلة نصية باستخدام التدوين العشري، مما ينتج عنه "12.100000":


#define __STDC_WANT_IEC_60559_BFP_EXT__
#include <stdlib.h>
int ssize = 10;
char s[ssize];
strfromf(s, ssize, "%f", 12.1);

لتحويل القيمة 12.3456 كنوع float إلى سلسلة نصية باستخدام التدوين العشري بدقة رقمين، مما ينتج عنه "12.35":


#define __STDC_WANT_IEC_60559_BFP_EXT__
#include <stdlib.h>
int ssize = 10;
char s[ssize];
strfromf(s, ssize, "%.2f", 12.3456);

لتحويل القيمة 12.345e19 كنوع double إلى سلسلة نصية باستخدام التدوين العلمي بدقة صفر، مما ينتج عنه "1E+20":


#define __STDC_WANT_IEC_60559_BFP_EXT__
#include <stdlib.h>
int ssize = 10;
char s[ssize];
strfromd(s, ssize, "%.E", 12.345e19);
