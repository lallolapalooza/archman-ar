subroutine strevc3 (character side, character howmny, logical, dimension( * ) select, integer n, real, dimension( ldt, * ) t, integer ldt, real, dimension( ldvl, * ) vl, integer ldvl, real, dimension( ldvr, * ) vr, integer ldvr, integer mm, integer m, real, dimension( * ) work, integer lwork, integer info)

STREVC3

الغرض:

!>
!> STREVC3 يحسب بعض أو كل المتجهات الذاتية اليمنى واليسرى لمصفوفة T شبه ثلاثية علوية حقيقية.
!> يتم إنتاج مصفوفات هذا النوع عن طريق التحليل الشوري لمصفوفة A عامة حقيقية:  A = Q*T*Q**T، كما تم حسابها بواسطة SHSEQR.
!>
!> المتجه الذاتي الأيمن x والمتجه الذاتي الأيسر y للمصفوفة T المقابلين لقيمة ذاتية w معرفان بـ:
!>
!>    T*x = w*x,     (y**T)*T = w*(y**T)
!>
!> حيث تشير y**T إلى منقول المتجه y.
!> القيم الذاتية ليست مدخلات لهذا الروتين، ولكن يتم قراءتها مباشرة
!> من الكتل القطرية لـ T.
!>
!> يعيد هذا الروتين المصفوفتين X و/أو Y للمتجهات الذاتية اليمنى واليسرى لـ T، أو حاصل الضرب Q*X و/أو Q*Y، حيث Q مصفوفة مدخلة. إذا كانت Q هي العامل المتعامد الذي يقلل المصفوفة A إلى الشكل الشوري T، فإن Q*X و Q*Y هما مصفوفتي المتجهات الذاتية اليمنى واليسرى للمصفوفة A.
!>
!> هذا يستخدم إصدار BLAS من المستوى 3 للتحويل العكسي.
!>

المعلمات

SIDE 
!>          SIDE هو CHARACTER*1
!>          = 'R':  حساب المتجهات الذاتية اليمنى فقط؛
!>          = 'L':  حساب المتجهات الذاتية اليسرى فقط؛
!>          = 'B':  حساب المتجهات الذاتية اليمنى واليسرى معًا.
!>

HOWMNY

!>          HOWMNY هو CHARACTER*1
!>          = 'A':  حساب جميع المتجهات الذاتية اليمنى و/أو اليسرى؛
!>          = 'B':  حساب جميع المتجهات الذاتية اليمنى و/أو اليسرى،
!>                  مع تحويل عكسي بواسطة المصفوفتين في VR و/أو VL؛
!>          = 'S':  حساب المتجهات الذاتية اليمنى و/أو اليسرى المختارة،
!>                  كما هو محدد بواسطة المصفوفة المنطقية SELECT.
!>

SELECT

!>          SELECT هي مصفوفة LOGICAL، بعدها (N)
!>          إذا كان HOWMNY = 'S'، تحدد SELECT المتجهات الذاتية التي سيتم
!>          حسابها.
!>          إذا كانت w(j) قيمة ذاتية حقيقية، يتم حساب المتجه الذاتي الحقيقي
!>          المقابل إذا كانت SELECT(j) تساوي .TRUE..
!>          إذا كانت w(j) و w(j+1) هما الجزء الحقيقي والجزء التخيلي لقيمة ذاتية
!>          مركبة، يتم حساب المتجه الذاتي المركب المقابل إذا كانت أي من SELECT(j) أو SELECT(j+1)
!>          تساوي .TRUE.، وعند الخروج يتم ضبط SELECT(j) على .TRUE. و SELECT(j+1) على
!>          .FALSE..
!>          غير مستخدمة إذا كان HOWMNY = 'A' أو 'B'.
!>

N

!>          N هو INTEGER
!>          رتبة المصفوفة T. N >= 0.
!>

T

!>          T هي مصفوفة REAL، بعدها (LDT,N)
!>          المصفوفة شبه الثلاثية العلوية T في الشكل القانوني الشوري.
!>

LDT

!>          LDT هو INTEGER
!>          البعد الرئيسي للمصفوفة T. LDT >= max(1,N).
!>

VL

!>          VL هي مصفوفة REAL، بعدها (LDVL,MM)
!>          عند الدخول، إذا كانت SIDE = 'L' أو 'B' و HOWMNY = 'B'، يجب أن
!>          تحتوي VL على مصفوفة Q بحجم N*N (عادةً العامل المتعامد Q
!>          للمتجهات الشورية التي تم إرجاعها بواسطة SHSEQR).
!>          عند الخروج، إذا كانت SIDE = 'L' أو 'B'، تحتوي VL على:
!>          إذا كان HOWMNY = 'A'، المصفوفة Y للمتجهات الذاتية اليسرى لـ T؛
!>          إذا كان HOWMNY = 'B'، المصفوفة Q*Y؛
!>          إذا كان HOWMNY = 'S'، المتجهات الذاتية اليسرى لـ T المحددة بواسطة
!>                           SELECT، مخزنة بشكل متتابع في أعمدة
!>                           VL، بنفس ترتيب قيمها الذاتية.
!>          المتجه الذاتي المركب المقابل لقيمة ذاتية مركبة
!>          يخزن في عمودين متتاليين، الأول يحتوي على الجزء الحقيقي، والثاني على الجزء التخيلي.
!>          غير مستخدمة إذا كانت SIDE = 'R'.
!>

LDVL

!>          LDVL هو INTEGER
!>          البعد الرئيسي للمصفوفة VL.
!>          LDVL >= 1، وإذا كانت SIDE = 'L' أو 'B'، فإن LDVL >= N.
!>

VR

!>          VR هي مصفوفة REAL، بعدها (LDVR,MM)
!>          عند الدخول، إذا كانت SIDE = 'R' أو 'B' و HOWMNY = 'B'، يجب أن
!>          تحتوي VR على مصفوفة Q بحجم N*N (عادةً العامل المتعامد Q
!>          للمتجهات الشورية التي تم إرجاعها بواسطة SHSEQR).
!>          عند الخروج، إذا كانت SIDE = 'R' أو 'B'، تحتوي VR على:
!>          إذا كان HOWMNY = 'A'، المصفوفة X للمتجهات الذاتية اليمنى لـ T؛
!>          إذا كان HOWMNY = 'B'، المصفوفة Q*X؛
!>          إذا كان HOWMNY = 'S'، المتجهات الذاتية اليمنى لـ T المحددة بواسطة
!>                           SELECT، مخزنة بشكل متتابع في أعمدة
!>                           VR، بنفس ترتيب قيمها الذاتية.
!>          المتجه الذاتي المركب المقابل لقيمة ذاتية مركبة
!>          يخزن في عمودين متتاليين، الأول يحتوي على الجزء الحقيقي والثاني على الجزء التخيلي.
!>          غير مستخدمة إذا كانت SIDE = 'L'.
!>

LDVR

!>          LDVR هو INTEGER
!>          البعد الرئيسي للمصفوفة VR.
!>          LDVR >= 1، وإذا كانت SIDE = 'R' أو 'B'، فإن LDVR >= N.
!>

MM

!>          MM هو INTEGER
!>          عدد الأعمدة في المصفوفتين VL و/أو VR. MM >= M.
!>

M

!>          M هو INTEGER
!>          عدد الأعمدة في المصفوفتين VL و/أو VR المستخدمة فعليًا
!>          لتخزين المتجهات الذاتية.
!>          إذا كان HOWMNY = 'A' أو 'B'، يتم ضبط M على N.
!>          كل متجه ذاتي حقيقي محدد يشغل عمودًا واحدًا وكل متجه ذاتي مركب محدد
!>          يشغل عمودين.
!>

WORK

!>          WORK هي مصفوفة REAL، بعدها (MAX(1,LWORK))
!>

LWORK

!>          LWORK هو INTEGER
!>          بعد المصفوفة WORK. LWORK >= max(1,3*N).
!>          لأداء أمثل، LWORK >= N + 2*N*NB، حيث NB
!>          هو حجم الكتلة الأمثل.
!>
!>          إذا كان LWORK = -1، فمن المفترض وجود استعلام مساحة عمل؛ يقوم الروتين
!>          فقط بحساب الحجم الأمثل للمصفوفة WORK، ويعيد
!>          هذه القيمة كإدخال أول في المصفوفة WORK، ولا يتم إصدار أي رسالة خطأ
!>          متعلقة بـ LWORK بواسطة XERBLA.
!>

INFO

!>          INFO هو INTEGER
!>          = 0:  خروج ناجح
!>          < 0:  إذا كان INFO = -i، كان الوسيط i غير قانوني
!>
