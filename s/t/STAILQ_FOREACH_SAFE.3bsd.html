listelm.

الماكرو SLIST_NEXT يُرجع العنصر التالي في القائمة.

الماكرو SLIST_REMOVE_AFTER يُزيل العنصر الذي يلي elm من القائمة. على عكس SLIST_REMOVE، هذا الماكرو لا يجتاز القائمة بأكملها.

الماكرو SLIST_REMOVE_HEAD يُزيل العنصر elm من رأس القائمة. للحصول على أفضل كفاءة، يجب على العناصر التي يتم إزالتها من رأس القائمة استخدام هذا الماكرو بشكل صريح بدلاً من الماكرو العام SLIST_REMOVE.

الماكرو SLIST_REMOVE يُزيل العنصر elm من القائمة. يجب تجنب استخدام هذا الماكرو لأنه يجتاز القائمة بأكملها. يجب استخدام قائمة مرتبطة مزدوجة إذا كانت هناك حاجة لهذا الماكرو في مسارات التعليمات البرمجية ذات الاستخدام العالي أو للعمل على قوائم طويلة.

الماكرو SLIST_SWAP يبادل محتويات head1 و head2.

مثال على قائمة أحادية الارتباط

SLIST_HEAD(slisthead, entry) head =
    SLIST_HEAD_INITIALIZER(head);
struct slisthead *headp;		/* رأس القائمة أحادية الارتباط. */
struct entry {
	...
	SLIST_ENTRY(entry) entries;	/* قائمة أحادية الارتباط. */
	...
} *n1, *n2, *n3, *np;

SLIST_INIT(&head);			/* تهيئة القائمة. */

n1 = malloc(sizeof(struct entry));	/* إدراج في الرأس. */
SLIST_INSERT_HEAD(&head, n1, entries);

n2 = malloc(sizeof(struct entry));	/* إدراج بعد. */
SLIST_INSERT_AFTER(n1, n2, entries);

SLIST_REMOVE(&head, n2, entry, entries);/* حذف. */
free(n2);

n3 = SLIST_FIRST(&head);
SLIST_REMOVE_HEAD(&head, entries);	/* حذف من الرأس. */
free(n3);
					/* اجتياز أمامي. */
SLIST_FOREACH(np, &head, entries)
	np-> ...
					/* اجتياز أمامي آمن. */
SLIST_FOREACH_SAFE(np, &head, entries, np_temp) {
	np->do_stuff();
	...
	SLIST_REMOVE(&head, np, entry, entries);
	free(np);
}

while (!SLIST_EMPTY(&head)) {		/* حذف القائمة. */
	n1 = SLIST_FIRST(&head);
	SLIST_REMOVE_HEAD(&head, entries);
	free(n1);
}

قوائم الذيل أحادية الارتباط

قائمة الذيل أحادية الارتباط يتم توجيهها بواسطة هيكل معرف بواسطة الماكرو STAILQ_HEAD. يحتوي هذا الهيكل على زوج من المؤشرات، أحدهما إلى العنصر الأول في قائمة الذيل والآخر إلى العنصر الأخير في قائمة الذيل. العناصر مرتبطة أحاديًا لتقليل استهلاك المساحة والتعامل مع المؤشرات على حساب O(n) للحذف للعناصر العشوائية. يمكن إضافة عناصر جديدة إلى قائمة الذيل بعد عنصر موجود، أو في رأس قائمة الذيل، أو في نهاية قائمة الذيل. يتم تعريف هيكل STAILQ_HEAD على النحو التالي:

STAILQ_HEAD(HEADNAME, TYPE) head;

حيث HEADNAME هو اسم الهيكل الذي سيتم تعريفه، و TYPE هو نوع العناصر التي سيتم ربطها في قائمة الذيل. يمكن لاحقًا تعريف مؤشر إلى رأس قائمة الذيل على النحو التالي:

struct HEADNAME *headp;

(أسماء head و headp قابلة للاختيار من قبل المستخدم.)

الماكرو STAILQ_HEAD_INITIALIZER يُقيّم إلى مُهيئ لقائمة الذيل head.

الماكرو STAILQ_CONCAT يربط قائمة الذيل الموجهة بواسطة head2 بنهاية القائمة الموجهة بواسطة head1، ويزيل جميع الإدخالات من الأولى.

الماكرو STAILQ_EMPTY يُقيّم إلى صحيح إذا لم يكن هناك عناصر في قائمة الذيل.

الماكرو STAILQ_ENTRY يُعرّف هيكلًا يربط العناصر في قائمة الذيل.

الماكرو STAILQ_FIRST يُرجع العنصر الأول في قائمة الذيل أو NULL إذا كانت قائمة الذيل فارغة.

الماكرو STAILQ_FOREACH يجتاز قائمة الذيل المشار إليها بواسطة head في الاتجاه الأمامي، ويُعين كل عنصر بدوره إلى var.

الماكرو STAILQ_FOREACH_FROM يعمل بشكل مطابق لـ STAILQ_FOREACH عندما يكون var هو NULL، وإلا فإنه يعامل var كعنصر STAILQ تم العثور عليه مسبقًا ويبدأ الحلقة عند var بدلاً من العنصر الأول في STAILQ المشار إليه بواسطة head.

الماكرو STAILQ_FOREACH_SAFE يجتاز قائمة الذيل المشار إليها بواسطة head في الاتجاه الأمامي، ويُعين كل عنصر بدوره إلى var. ومع ذلك، على عكس STAILQ_FOREACH()، يُسمح هنا بإزالة var وكذلك تحريره من داخل الحلقة بأمان دون التأثير على الاجتياز.

الماكرو STAILQ_FOREACH_FROM_SAFE يعمل بشكل مطابق لـ STAILQ_FOREACH_SAFE عندما يكون var هو NULL، وإلا فإنه يعامل var كعنصر STAILQ تم العثور عليه مسبقًا ويبدأ الحلقة عند var بدلاً من العنصر الأول في STAILQ المشار إليه بواسطة head.

الماكرو STAILQ_INIT يُهيئ قائمة الذيل المشار إليها بواسطة head.

الماكرو STAILQ_INSERT_HEAD يُدرج العنصر الجديد elm في رأس قائمة الذيل.

الماكرو STAILQ_INSERT_TAIL يُدرج العنصر الجديد elm في نهاية قائمة الذيل.

الماكرو STAILQ_INSERT_AFTER يُدرج العنصر الجديد elm بعد العنصر listelm.

الماكرو STAILQ_LAST يُرجع العنصر الأخير في قائمة الذيل. إذا كانت قائمة الذيل فارغة، فإن القيمة المُرجعة هي NULL.

الماكرو STAILQ_NEXT يُرجع العنصر التالي في قائمة الذيل، أو NULL إذا كان هذا العنصر هو الأخير.

الماكرو STAILQ_REMOVE_AFTER يُزيل العنصر الذي يلي elm من قائمة الذيل. على عكس STAILQ_REMOVE، هذا الماكرو لا يجتاز قائمة الذيل بأكملها.

الماكرو STAILQ_REMOVE_HEAD يُزيل العنصر الموجود في رأس قائمة الذيل. للحصول على أفضل كفاءة، يجب على العناصر التي يتم إزالتها من رأس قائمة الذيل استخدام هذا الماكرو بشكل صريح بدلاً من الماكرو العام STAILQ_REMOVE.

الماكرو STAILQ_REMOVE يُزيل العنصر elm من قائمة الذيل. يجب تجنب استخدام هذا الماكرو لأنه يجتاز القائمة بأكملها. يجب استخدام قائمة ذيل مرتبطة مزدوجة إذا كانت هناك حاجة لهذا الماكرو في مسارات التعليمات البرمجية ذات الاستخدام العالي أو للعمل على قوائم ذيل طويلة.

الماكرو STAILQ_SWAP يبادل محتويات head1 و head2.

مثال على قائمة ذيل أحادية الارتباط

STAILQ_HEAD(stailhead, entry) head =
    STAILQ_HEAD_INITIALIZER(head);
struct stailhead *headp;		/* رأس قائمة الذيل أحادية الارتباط. */
struct entry {
	...
	STAILQ_ENTRY(entry) entries;	/* قائمة الذيل. */
	...
} *n1, *n2, *n3, *np;

STAILQ_INIT(&head);			/* تهيئة القائمة. */

n1 = malloc(sizeof(struct entry));	/* إدراج في الرأس. */
STAILQ_INSERT_HEAD(&head, n1, entries);

n1 = malloc(sizeof(struct entry));	/* إدراج في الذيل. */
STAILQ_INSERT_TAIL(&head, n1, entries);

n2 = malloc(sizeof(struct entry));	/* إدراج بعد. */
STAILQ_INSERT_AFTER(&head, n1, n2, entries);
					/* حذف. */
STAILQ_REMOVE(&head, n2, entry, entries);
free(n2);
					/* حذف من الرأس. */
n3 = STAILQ_FIRST(&head);
STAILQ_REMOVE_HEAD(&head, entries);
free(n3);
					/* اجتياز أمامي. */
STAILQ_FOREACH(np, &head, entries)
	np-> ...
					/* اجتياز أمامي آمن. */
STAILQ_FOREACH_SAFE(np, &head, entries, np_temp) {
	np->do_stuff();
	...
	STAILQ_REMOVE(&head, np, entry, entries);
	free(np);
}
					/* حذف قائمة الذيل. */
while (!STAILQ_EMPTY(&head)) {
	n1 = STAILQ_FIRST(&head);
	STAILQ_REMOVE_HEAD(&head, entries);
	free(n1);
}
					/* حذف أسرع لقائمة الذيل. */
n1 = STAILQ_FIRST(&head);
while (n1 != NULL) {
	n2 = STAILQ_NEXT(n1, entries);
	free(n1);
	n1 = n2;
}
STAILQ_INIT(&head);

القوائم

يتم توجيه القائمة بواسطة هيكل معرف بواسطة الماكرو LIST_HEAD. يحتوي هذا الهيكل على مؤشر واحد إلى العنصر الأول في القائمة. العناصر مرتبطة بشكل مزدوج بحيث يمكن إزالة عنصر عشوائي دون الحاجة إلى اجتياز القائمة. يمكن إضافة عناصر جديدة إلى القائمة بعد عنصر موجود، أو قبل عنصر موجود، أو في رأس القائمة. يتم تعريف هيكل LIST_HEAD على النحو التالي:

LIST_HEAD(HEADNAME, TYPE) head;

حيث HEADNAME هو اسم الهيكل الذي سيتم تعريفه، و TYPE هو نوع العناصر التي سيتم ربطها في القائمة. يمكن لاحقًا تعريف مؤشر إلى رأس القائمة على النحو التالي:

struct HEADNAME *headp;

(أسماء head و headp قابلة للاختيار من قبل المستخدم.)

الماكرو LIST_HEAD_INITIALIZER يُقيّم إلى مُهيئ للقائمة head.

الماكرو LIST_CONCAT يربط القائمة الموجهة بواسطة head2 بنهاية القائمة الموجهة بواسطة head1، ويزيل جميع الإدخالات من الأولى. يجب تجنب استخدام هذا الماكرو لأنه يجتاز القائمة بأكملها head1. يجب استخدام قائمة ذيل إذا كانت هناك حاجة لهذا الماكرو في مسارات التعليمات البرمجية ذات الاستخدام العالي أو للعمل على قوائم طويلة.

الماكرو LIST_EMPTY يُقيّم إلى صحيح إذا لم يكن هناك عناصر في القائمة.

الماكرو LIST_ENTRY يُعرّف هيكلًا يربط العناصر في القائمة.

الماكرو LIST_FIRST يُرجع العنصر الأول في القائمة أو NULL إذا كانت القائمة فارغة.

الماكرو LIST_FOREACH يجتاز القائمة المشار إليها بواسطة
