val make : int -> char -> string

make n c هي سلسلة نصية بطول n حيث يحمل كل فهرس الحرف c.

يثير Invalid_argument إذا كان n < 0 أو n > Sys.max_string_length.

val concat : sep:string -> string list -> string

concat ~sep ss يدمج قائمة السلاسل النصية ss، مع إدراج سلسلة الفاصل sep بين كل منها.

val contains_from : string -> int -> char -> bool

contains_from s start c تكون true إذا وفقط إذا كان الحرف c يظهر في s بعد الموضع start.

يثير Invalid_argument إذا كان start ليس موضعاً صالحاً في s.

val rcontains_from : string -> int -> char -> bool

rcontains_from s stop c تكون true إذا وفقط إذا كان الحرف c يظهر في s قبل الموضع stop+1.

يثير Invalid_argument إذا كان stop < 0 أو stop+1 ليس موضعاً صالحاً في s.

val contains : string -> char -> bool

contains s c هو String.contains_from s 0 c.

استخراج السلاسل الفرعية

val sub : string -> pos:int -> len:int -> string

sub s ~pos ~len هي سلسلة نصية بطول len، تحتوي على السلسلة الفرعية لـ s التي تبدأ من الموضع pos ولها طول len.

يثير Invalid_argument إذا كان pos و len لا يحددان سلسلة فرعية صالحة لـ s.

val split_on_char : sep:char -> string -> string list

split_on_char ~sep s هي قائمة بجميع السلاسل الفرعية (الممكنة أو الفارغة) لـ s التي تفصلها الحرف sep. إذا كانت s فارغة، فالنتيجة هي القائمة الأحادية [""].

تُحدد نتيجة الدالة بالثوابت التالية:

-القائمة ليست فارغة.

-دمج عناصرها باستخدام sep كفاصل يعيد سلسلة نصية مساوية للإدخال ( concat (make 1 sep)
(split_on_char sep s) = s ).

-لا تحتوي أي سلسلة نصية في النتيجة على الحرف sep.

val trim : string -> string

trim s هي s بدون المسافات البيضاء البادئة واللاحقة. الأحرف البيضاء هي: ' ' ، '\x0C' (تغذية نموذجية)، '\n' ، '\r' ، و '\t'.

منذ 4.00

val escaped : string -> string

escaped s هي s مع تمثيل الأحرف الخاصة بتسلسلات الهروب، باتباع قواعد OCaml اللفظية.

يتم هروب جميع الأحرف خارج نطاق الطباعة لـ US-ASCII [0x20;0x7E]، بالإضافة إلى الشرطة المائلة العكسية (0x2F) وعلامة الاقتباس المزدوجة (0x22).

الدالة Scanf.unescaped هي المعكوس الأيسر للدالة escaped، أي Scanf.unescaped (escaped s) = s لأي سلسلة نصية s (ما لم تفشل escaped s).

يثير Invalid_argument إذا كانت النتيجة أطول من Sys.max_string_length بايت.

val uppercase_ascii : string -> string

uppercase_ascii s هي s مع ترجمة جميع الأحرف الصغيرة إلى أحرف كبيرة، باستخدام مجموعة أحرف US-ASCII.
