subroutine stbsv (character uplo, character trans, character diag, integer n, integer k, real, dimension(lda,*) a, integer lda, real, dimension(*) x, integer incx)

STBSV

الغرض:

!>
!> STBSV  يحل أحد أنظمة المعادلات
!>
!>    A*x = b,   أو   A**T*x = b,
!>
!> حيث b و x هما متجهان بعناصر n، و A هي مصفوفة مربعة n × n، إما
!> أحادية، أو غير أحادية، علوية أو سفلية، ذات نطاق، مع ( k + 1 )
!> قطر.
!>
!> لا يتم تضمين اختبار للوحدانية أو شبه الوحدانية في هذا
!> الإجراء. يجب إجراء مثل هذه الاختبارات قبل استدعاء هذا الإجراء.
!>

المعاملات

UPLO 
!>          UPLO is CHARACTER*1
!>           عند الإدخال، يحدد UPLO ما إذا كانت المصفوفة علوية أم
!>           سفلية كما يلي:
!>
!>              UPLO = 'U' or 'u'   A هي مصفوفة علوية.
!>
!>              UPLO = 'L' or 'l'   A هي مصفوفة سفلية.
!>

TRANS

!>          TRANS is CHARACTER*1
!>           عند الإدخال، يحدد TRANS المعادلات التي سيتم حلها كما
!>           يلي:
!>
!>              TRANS = 'N' or 'n'   A*x = b.
!>
!>              TRANS = 'T' or 't'   A**T*x = b.
!>
!>              TRANS = 'C' or 'c'   A**T*x = b.
!>

DIAG

!>          DIAG is CHARACTER*1
!>           عند الإدخال، يحدد DIAG ما إذا كانت A أحادية أم لا
!>           كما يلي:
!>
!>              DIAG = 'U' or 'u'   يفترض أن A أحادية.
!>
!>              DIAG = 'N' or 'n'   لا يفترض أن A أحادية.
!>

N

!>          N is INTEGER
!>           عند الإدخال، يحدد N ترتيب المصفوفة A.
!>           يجب أن يكون N أكبر من أو يساوي صفر.
!>

K

!>          K is INTEGER
!>           عند الإدخال مع UPLO = 'U' أو 'u'، يحدد K عدد
!>           الأقطار الفوقية للمصفوفة A.
!>           عند الإدخال مع UPLO = 'L' أو 'l'، يحدد K عدد
!>           الأقطار التحتية للمصفوفة A.
!>           يجب أن يحقق K  0 .le. K.
!>

A

!>          A is REAL array, dimension ( LDA, N )
!>           قبل الإدخال مع UPLO = 'U' أو 'u'، يجب أن يحتوي الجزء
!>           الرائد ( k + 1 ) × n من المصفوفة A على الجزء العلوي
!>           المثلثي النطاقي للمصفوفة المعاملية، المقدمة عمودًا
!>           بعمود، مع القطر الرئيسي للمصفوفة في الصف
!>           ( k + 1 ) من المصفوفة، ويبدأ القطر الفوقي الأول عند
!>           الموضع 2 في الصف k، وهكذا. الجزء العلوي الأيسر k × k
!>           من المصفوفة A غير مستخدم.
!>           سيقوم المقطع البرمجي التالي بنقل مصفوفة نطاقية علوية
!>           من تخزين مصفوفة كاملة تقليدية إلى تخزين نطاقي:
!>
!>                 DO 20, J = 1, N
!>                    M = K + 1 - J
!>                    DO 10, I = MAX( 1, J - K ), J
!>                       A( M + I, J ) = matrix( I, J )
!>              10    CONTINUE
!>              20 CONTINUE
!>
!>           قبل الإدخال مع UPLO = 'L' أو 'l'، يجب أن يحتوي الجزء
!>           الرائد ( k + 1 ) × n من المصفوفة A على الجزء السفلي
!>           المثلثي النطاقي للمصفوفة المعاملية، المقدمة عمودًا
!>           بعمود، مع القطر الرئيسي للمصفوفة في الصف 1 من
!>           المصفوفة، ويبدأ القطر التحت الأول عند الموضع 1 في
!>           الصف 2، وهكذا. الجزء السفلي الأيمن k × k من
!>           المصفوفة A غير مستخدم.
!>           سيقوم المقطع البرمجي التالي بنقل مصفوفة نطاقية سفلية
!>           من تخزين مصفوفة كاملة تقليدية إلى تخزين نطاقي:
!>
!>                 DO 20, J = 1, N
!>                    M = 1 - J
!>                    DO 10, I = J, MIN( N, J + K )
!>                       A( M + I, J ) = matrix( I, J )
!>              10    CONTINUE
!>              20 CONTINUE
!>
!>           لاحظ أنه عند DIAG = 'U' أو 'u'، لا يتم استخدام عناصر
!>           المصفوفة A المقابلة للعناصر القطرية للمصفوفة، ولكن
!>           يفترض أنها تساوي الواحد.
!>

LDA

!>          LDA is INTEGER
!>           عند الإدخال، يحدد LDA البعد الأول لـ A كما هو معلن
!>           في البرنامج المستدعي. يجب أن يكون LDA على الأقل
!>           ( k + 1 ).
!>

X

!>          X is REAL array, dimension at least
!>           ( 1 + ( n - 1 )*abs( INCX ) ).
!>           قبل الإدخال، يجب أن يحتوي المصفوفة X المتزايدة على المتجه
!>           b ذي العناصر n في الطرف الأيمن. عند الخروج، يتم الكتابة
!>           فوق X بالمتجه x الناتج.
!>

INCX

!>          INCX is INTEGER
!>           عند الإدخال، يحدد INCX الزيادة لعناصر X. يجب ألا يكون INCX
!>           صفرًا.
!>
