توثيق الوظيفة/الروتين الفرعي

subroutine strt05 (character uplo, character trans, character diag, integer n, integer nrhs, real, dimension( lda, * ) a, integer lda, real, dimension( ldb, * ) b, integer ldb, real, dimension( ldx, * ) x, integer ldx, real, dimension( ldxact, * ) xact, integer ldxact, real, dimension( * ) ferr, real, dimension( * ) berr, real, dimension( * ) reslts)

STRT05

الغرض:

!>
!> STRT05 تختبر حدود الخطأ من التنقيح التكراري للحل المحسوب لنظام المعادلات A*X = B،
!> حيث A مصفوفة مثلثة n × n.
!>
!> RESLTS(1) = اختبار حد الخطأ
!>           = norm(X - XACT) / ( norm(X) * FERR )
!>
!> يتم إرجاع قيمة كبيرة إذا لم تكن هذه النسبة أقل من واحد.
!>
!> RESLTS(2) = الباقي من روتين التنقيح التكراري
!>           = الحد الأقصى لـ BERR / ( (n+1)*EPS + (*) )، حيث
!>             (*) = (n+1)*UNFL / (min_i (abs(A)*abs(X) +abs(b))_i )
!>

المعلمات

UPLO 
!>          UPLO هي CHARACTER*1
!>          تحدد ما إذا كانت المصفوفة A علوية أم سفلية.
!>          = 'U':  مثلثة علوية
!>          = 'L':  مثلثة سفلية
!>

TRANS

!>          TRANS هي CHARACTER*1
!>          تحدد شكل نظام المعادلات.
!>          = 'N':  A * X = B  (بدون تبديل)
!>          = 'T':  A'* X = B  (تبديل)
!>          = 'C':  A'* X = B  (تبديل مترافق = تبديل)
!>

DIAG

!>          DIAG هي CHARACTER*1
!>          تحدد ما إذا كانت المصفوفة A وحدة أم لا.
!>          = 'N':  غير وحدة مثلثة
!>          = 'U':  وحدة مثلثة
!>

N

!>          N هي INTEGER
!>          عدد صفوف المصفوفات X، B، و XACT، ورتبة المصفوفة A.  N >= 0.
!>

NRHS

!>          NRHS هي INTEGER
!>          عدد أعمدة المصفوفات X، B، و XACT.
!>          NRHS >= 0.
!>

A

!>          A هي مصفوفة REAL، أبعادها (LDA,N)
!>          المصفوفة المثلثة A.  إذا كان UPLO = 'U'، فإن الجزء العلوي
!>          المثلثي من المصفوفة A الذي هو n × n يحتوي على المصفوفة
!>          المثلثة العلوية، والجزء السفلي المثلثي الصارم من A غير
!>          مُشار إليه.  إذا كان UPLO = 'L'، فإن الجزء السفلي المثلثي
!>          الذي هو n × n من المصفوفة A يحتوي على المصفوفة المثلثة
!>          السفلية، والجزء العلوي المثلثي الصارم من A غير مُشار إليه.
!>          إذا كان DIAG = 'U'، فإن العناصر القطرية لـ A غير مُشار إليها
!>          وتُفترض أنها 1.
!>

LDA

!>          LDA هي INTEGER
!>          البعد الرئيسي للمصفوفة A.  LDA >= max(1,N).
!>

B

!>          B هي مصفوفة REAL، أبعادها (LDB,NRHS)
!>          متجهات الطرف الأيمن لنظام المعادلات الخطية.
!>

LDB

!>          LDB هي INTEGER
!>          البعد الرئيسي للمصفوفة B.  LDB >= max(1,N).
!>

X

!>          X هي مصفوفة REAL، أبعادها (LDX,NRHS)
!>          متجهات الحل المحسوبة.  كل متجه مخزن كعمود في المصفوفة X.
!>

LDX

!>          LDX هي INTEGER
!>          البعد الرئيسي للمصفوفة X.  LDX >= max(1,N).
!>

XACT

!>          XACT هي مصفوفة REAL، أبعادها (LDX,NRHS)
!>          متجهات الحل الدقيقة.  كل متجه مخزن كعمود في المصفوفة XACT.
!>

LDXACT

!>          LDXACT هي INTEGER
!>          البعد الرئيسي للمصفوفة XACT.  LDXACT >= max(1,N).
!>

FERR

!>          FERR هي مصفوفة REAL، أبعادها (NRHS)
!>          حدود الخطأ الأمامي المقدرة لكل متجه حل X.  إذا كان XTRUE هو الحل
!>          الحقيقي، فإن FERR يقدر مقدار أكبر إدخال في (X - XTRUE) مقسومًا
!>          على مقدار أكبر إدخال في X.
!>

BERR

!>          BERR هي مصفوفة REAL، أبعادها (NRHS)
!>          الخطأ الخلفي النسبي من حيث المكون لكل متجه حل
!>          (أي، أصغر تغيير نسبي في أي إدخال في A أو B يجعل X حلاً دقيقًا).
!>

RESLTS

!>          RESLTS هي مصفوفة REAL، أبعادها (2)
!>          الحد الأقصى لمتجهات الحل NRHS للنسب التالية:
!>          RESLTS(1) = norm(X - XACT) / ( norm(X) * FERR )
!>          RESLTS(2) = BERR / ( (n+1)*EPS + (*) )
!>
