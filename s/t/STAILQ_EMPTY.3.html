STAILQ(3) دليل وظائف المكتبة STAILQ(3)

الاسم

SIMPLEQ_EMPTY، SIMPLEQ_ENTRY، SIMPLEQ_FIRST، SIMPLEQ_FOREACH، SIMPLEQ_HEAD، SIMPLEQ_HEAD_INITIALIZER، SIMPLEQ_INIT، SIMPLEQ_INSERT_AFTER، SIMPLEQ_INSERT_HEAD، SIMPLEQ_INSERT_TAIL، SIMPLEQ_NEXT، SIMPLEQ_REMOVE، SIMPLEQ_REMOVE_HEAD، STAILQ_CONCAT، STAILQ_EMPTY، STAILQ_ENTRY، STAILQ_FIRST، STAILQ_FOREACH، STAILQ_HEAD، STAILQ_HEAD_INITIALIZER، STAILQ_INIT، STAILQ_INSERT_AFTER، STAILQ_INSERT_HEAD، STAILQ_INSERT_TAIL، STAILQ_NEXT، STAILQ_REMOVE، STAILQ_REMOVE_HEAD، - تنفيذ صف ذيل مربوط بشكل أحادي

المكتبة

مكتبة C القياسية (libc، -lc)

الخلاصة

#include <sys/queue.h>
STAILQ_ENTRY(TYPE);
STAILQ_HEAD(HEADNAME, TYPE);
STAILQ_HEAD STAILQ_HEAD_INITIALIZER(STAILQ_HEAD head);
void STAILQ_INIT(STAILQ_HEAD *head);
int STAILQ_EMPTY(STAILQ_HEAD *head);
void STAILQ_INSERT_HEAD(STAILQ_HEAD *head,
                         struct TYPE *elm, STAILQ_ENTRY NAME);
void STAILQ_INSERT_TAIL(STAILQ_HEAD *head,
                         struct TYPE *elm, STAILQ_ENTRY NAME);
void STAILQ_INSERT_AFTER(STAILQ_HEAD *head, struct TYPE *listelm,
                         struct TYPE *elm, STAILQ_ENTRY NAME);
struct TYPE *STAILQ_FIRST(STAILQ_HEAD *head);
struct TYPE *STAILQ_NEXT(struct TYPE *elm, STAILQ_ENTRY NAME);
STAILQ_FOREACH(struct TYPE *var, STAILQ_HEAD *head, STAILQ_ENTRY NAME);
void STAILQ_REMOVE(STAILQ_HEAD *head, struct TYPE *elm, TYPE,
                         STAILQ_ENTRY NAME);
void STAILQ_REMOVE_HEAD(STAILQ_HEAD *head,
                         STAILQ_ENTRY NAME);
void STAILQ_CONCAT(STAILQ_HEAD *head1, STAILQ_HEAD *head2);

ملاحظة: توجد وحدات ماكرو متطابقة مسبوقة بـ SIMPLEQ بدلاً من STAILQ؛ انظر الإصدارات.

الوصف

تُعرّف وحدات الماكرو هذه صفوف ذيل مربوطة بشكل أحادي وتشغلها.

في تعريفات الماكرو، TYPE هو اسم بنية معرفة من قبل المستخدم، يجب أن تحتوي على حقل من نوع STAILQ_ENTRY، يُطلق عليه اسم NAME. الوسيط HEADNAME هو اسم بنية معرفة من قبل المستخدم يجب الإعلان عنها باستخدام الماكرو STAILQ_HEAD().

الإنشاء

يتم تشكيل صف ذيل مربوط بشكل أحادي بواسطة بنية يحددها الماكرو STAILQ_HEAD(). تحتوي هذه البنية على زوج من المؤشرات، أحدهما يشير إلى العنصر الأول في صف الذيل والآخر إلى العنصر الأخير في صف الذيل. ترتبط العناصر بشكل أحادي لتقليل مساحة استهلاك الذاكرة والتعامل مع المؤشرات على حساب وقت O(n) للإزالة من عناصر عشوائية. يمكن إضافة عناصر جديدة إلى صف الذيل بعد عنصر موجود، أو في رأس صف الذيل، أو في نهاية صف الذيل. يتم الإعلان عن بنية STAILQ_HEAD على النحو التالي:


STAILQ_HEAD(HEADNAME, TYPE) head;

حيث struct HEADNAME هي البنية التي سيتم تعريفها، و struct TYPE هو نوع العناصر التي سيتم ربطها بصف الذيل. يمكن لاحقًا الإعلان عن مؤشر إلى رأس صف الذيل على النحو التالي:


struct HEADNAME *headp;

(الاسمين head و headp قابلان للاختيار من قبل المستخدم.)

يُعلن STAILQ_ENTRY() عن بنية تربط العناصر في صف الذيل.

STAILQ_HEAD_INITIALIZER() يُقيّم إلى مُهيئ لصف الذيل head.

يقوم STAILQ_INIT() بتهيئة صف الذيل الذي يشير إليه head.

يُقيّم STAILQ_EMPTY() إلى صحيح إذا لم تكن هناك عناصر في صف الذيل.

الإدراج

يقوم STAILQ_INSERT_HEAD() بإدراج العنصر الجديد elm في رأس صف الذيل.

يقوم STAILQ_INSERT_TAIL() بإدراج العنصر الجديد elm في نهاية صف الذيل.

يقوم STAILQ_INSERT_AFTER() بإدراج العنصر الجديد elm بعد العنصر listelm.

اجتياز

يُرجع STAILQ_FIRST() العنصر الأول في صف الذيل أو NULL إذا كان صف الذيل فارغًا.

يُرجع STAILQ_NEXT() العنصر التالي في صف الذيل، أو NULL إذا كان هذا العنصر هو الأخير.

يجتاز STAILQ_FOREACH() صف الذيل الذي يشير إليه head في الاتجاه الأمامي، ويعين كل عنصر بالتتابع إلى var.

الإزالة

يقوم STAILQ_REMOVE() بإزالة العنصر elm من صف الذيل.

يقوم STAILQ_REMOVE_HEAD() بإزالة العنصر الموجود في رأس صف الذيل. للحصول على أقصى قدر من الكفاءة، يجب على العناصر التي يتم إزالتها من رأس صف الذيل استخدام هذا الماكرو بشكل صريح بدلاً من الماكرو العام STAILQ_REMOVE().

ميزات أخرى

يقوم STAILQ_CONCAT() بدمج صف الذيل الذي يرأسه head2 في نهاية الصف الذي يرأسه head1، مما يزيل جميع الإدخالات من الأول.

قيمة الإرجاع

يُرجع STAILQ_EMPTY() قيمة غير صفرية إذا كان الصف فارغًا، وصفرًا إذا كان الصف يحتوي على إدخال واحد على الأقل.

يُرجع STAILQ_FIRST() و STAILQ_NEXT() مؤشرًا إلى بنية TYPE الأولى أو التالية، على التوالي.

يُرجع STAILQ_HEAD_INITIALIZER() مُهيئًا يمكن تعيينه إلى رأس الصف head.

الإصدارات

توفر بعض إصدارات BSD اسم SIMPLEQ بدلاً من STAILQ. وهما متطابقان، ولكن لأسباب تاريخية، تم تسميتهما بشكل مختلف في إصدارات BSD المختلفة. نشأ STAILQ في FreeBSD، ونشأ SIMPLEQ في NetBSD. لأسباب التوافق، توفر بعض الأنظمة مجموعات الماكرو هذه. توفر glibc كلاً من STAILQ و SIMPLEQ، وهما متطابقان باستثناء عدم وجود مكافئ لـ SIMPLEQ لـ STAILQ_CONCAT().

الأخطاء

لا يسمح STAILQ_FOREACH() بإزالة أو تحرير var أثناء التكرار، حيث قد يتعارض ذلك مع الاجتياز. STAILQ_FOREACH_SAFE()، الموجود في BSD ولكن غير موجود في glibc، يعالج هذا القيد من خلال السماح بإزالة var بأمان من القائمة وتحريرها من داخل الحلقة دون التأثير على الاجتياز.

أمثلة

#include <stddef.h>
#include <stdio.h>
#include <stdlib.h>
#include <sys/queue.h>
struct entry {
    int data;
    STAILQ_ENTRY(entry) entries;        /* صف ذيل مربوط بشكل أحادي */
};
STAILQ_HEAD(stailhead, entry);
int
main(void)
{
    struct entry *n1, *n2, *n3, *np;
    struct stailhead head;                  /* رأس صف ذيل مربوط بشكل أحادي */
    STAILQ_INIT(&head);                     /* تهيئة الصف */
    n1 = malloc(sizeof(struct entry));      /* الإدراج في الرأس */
    STAILQ_INSERT_HEAD(&head, n1, entries);
    n1 = malloc(sizeof(struct entry));      /* الإدراج في الذيل */
    STAILQ_INSERT_TAIL(&head, n1, entries);
    n2 = malloc(sizeof(struct entry));      /* الإدراج بعد */
    STAILQ_INSERT_AFTER(&head, n1, n2, entries);
    STAILQ_REMOVE(&head, n2, entry, entries); /* الحذف */
    free(n2);
    n3 = STAILQ_FIRST(&head);
    STAILQ_REMOVE_HEAD(&head, entries);     /* الحذف من الرأس */
    free(n3);
    n1 = STAILQ_FIRST(&head);
    n1->data = 0;
    for (unsigned int i = 1; i < 5; i++) {
        n1 = malloc(sizeof(struct entry));
        STAILQ_INSERT_HEAD(&head, n1, entries);
        n1->data = i;
    }
