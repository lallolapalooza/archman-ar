STAILQ(3) دليل الوظائف القياسية STAILQ(3)

الاسم

SIMPLEQ_EMPTY, SIMPLEQ_ENTRY, SIMPLEQ_FIRST, SIMPLEQ_FOREACH, SIMPLEQ_HEAD, SIMPLEQ_HEAD_INITIALIZER, SIMPLEQ_INIT, SIMPLEQ_INSERT_AFTER, SIMPLEQ_INSERT_HEAD, SIMPLEQ_INSERT_TAIL, SIMPLEQ_NEXT, SIMPLEQ_REMOVE, SIMPLEQ_REMOVE_HEAD, - تنفيذ طابور مؤشرات بسيط

المكتبة

مكتبة C القياسية (libc,  -lc)

الملخص

#include <sys/queue.h>
STAILQ_ENTRY(TYPE);
STAILQ_HEAD(HEADNAME, TYPE);
STAILQ_HEAD STAILQ_HEAD_INITIALIZER(STAILQ_HEAD head);
void STAILQ_INIT(STAILQ_HEAD *head);
int STAILQ_EMPTY(STAILQ_HEAD *head);
void STAILQ_INSERT_HEAD(STAILQ_HEAD *head,
                         struct TYPE *elm, STAILQ_ENTRY NAME);
void STAILQ_INSERT_TAIL(STAILQ_HEAD *head,
                         struct TYPE *elm, STAILQ_ENTRY NAME);
void STAILQ_INSERT_AFTER(STAILQ_HEAD *head, struct TYPE *listelm,
                         struct TYPE *elm, STAILQ_ENTRY NAME);
struct TYPE *STAILQ_FIRST(STAILQ_HEAD *head);
struct TYPE *STAILQ_NEXT(struct TYPE *elm, STAILQ_ENTRY NAME);
STAILQ_FOREACH(struct TYPE *var, STAILQ_HEAD *head, STAILQ_ENTRY NAME);
void STAILQ_REMOVE(STAILQ_HEAD *head, struct TYPE *elm, TYPE,
                         STAILQ_ENTRY NAME);
void STAILQ_REMOVE_HEAD(STAILQ_HEAD *head,
                         STAILQ_ENTRY NAME);
void STAILQ_CONCAT(STAILQ_HEAD *head1, STAILQ_HEAD *head2);

ملاحظة: توجد ماكرو متطابقة مسبوقة بـ SIMPLEQ بدلاً من STAILQ؛ انظر الإصدارات.

الوصف

تعرّف هذه الماكرو وتعمل على طابور مؤشرات بسيط.

في تعريفات الماكرو، TYPE هو اسم هيكل معرف من قبل المستخدم، والذي يجب أن يحتوي على حقل من النوع STAILQ_ENTRY، ويسمى NAME. الوسيط HEADNAME هو اسم هيكل معرف من قبل المستخدم يجب تعريفه باستخدام الماكرو STAILQ_HEAD().

الإنشاء

يتم توجيه طابور المؤشرات البسيط بواسطة هيكل معرف بواسطة الماكرو STAILQ_HEAD(). يحتوي هذا الهيكل على زوج من المؤشرات، أحدهما يشير إلى العنصر الأول في طابور المؤشرات والآخر يشير إلى العنصر الأخير في طابور المؤشرات. العناصر هي من نوع أحادي الارتباط لتقليل المساحة والنفقات العامة لمعالجة المؤشرات، على الرغم من أن وقت حذف العناصر التعسفية هو O(n). يمكن إضافة عناصر جديدة إلى الطابور بعد عنصر موجود، أو في بداية الطابور، أو في نهاية الطابور. يتم تعريف هيكل STAILQ_HEAD على النحو التالي:


STAILQ_HEAD(HEADNAME, TYPE) head;

حيث struct HEADNAME هو الهيكل المراد تعريفه، و struct TYPE هو نوع العناصر المرتبطة بالطابور. يمكن لاحقًا تعريف عنوان مؤشر رأس الطابور على النحو التالي:


struct HEADNAME *headp;

(أسماء head و headp قابلة للاختيار من قبل المستخدم.)

STAILQ_ENTRY() تعرّف هيكلًا يربط عناصر الطابور.

STAILQ_HEAD_INITIALIZER() يُقيّم إلى مُهيئ للطابور head.

STAILQ_INIT() يهيئ الطابور المشار إليه بواسطة head.

STAILQ_EMPTY() يُقيّم إلى صحيح إذا لم تكن هناك عناصر في الطابور.

الإدخال

STAILQ_INSERT_HEAD() يدخل العنصر الجديد elm في بداية الطابور.

STAILQ_INSERT_TAIL() يدخل العنصر الجديد elm في نهاية الطابور.

STAILQ_INSERT_AFTER() يدخل العنصر الجديد elm بعد العنصر listelm.

التجوال

STAILQ_FIRST() يعيد العنصر الأول في الطابور أو NULL إذا كان الطابور فارغًا.

STAILQ_NEXT() يعيد العنصر التالي في الطابور، أو NULL إذا كان هذا هو الأخير.

STAILQ_FOREACH() يتجول في الطابور المشار إليه بواسطة head في الاتجاه الأمامي، مع تخصيص كل عنصر لـ var.

الإزالة

STAILQ_REMOVE() يزيل العنصر elm من الطابور.

STAILQ_REMOVE_HEAD() يزيل العنصر في بداية الطابور. لتحقيق أقصى قدر من الكفاءة، يجب على العناصر التي يتم إزالتها من بداية الطابور استخدام هذه الماكرو بشكل صريح بدلاً من الماكرو العام STAILQ_REMOVE().

آليات أخرى

STAILQ_CONCAT() يربط الطابور الموجه بواسطة head2 إلى الطابور الموجه بواسطة head1، مع إزالة جميع العناصر من الأول.

قيمة الإرجاع

STAILQ_EMPTY() تعيد قيمة غير صفرية إذا كان الطابور فارغًا، وصفر إذا كان يحتوي على عنصر واحد على الأقل.

STAILQ_FIRST()، و STAILQ_NEXT() تعيد مؤشرًا إلى الهيكل من النوع TYPE، على التوالي.

STAILQ_HEAD_INITIALIZER() تعيد مُهيئ يمكن تعيينه للطابور head.

الإصدارات

تقدم بعض أنظمة BSD وظيفة SIMPLEQ بدلاً من STAILQ. وهما متطابقتان، ولكن لأسباب تاريخية تم تسميتهما بشكل مختلف في أنظمة BSD المختلفة. نشأت STAILQ في FreeBSD، و SIMPLEQ في NetBSD. لأسباب التوافق، توفر بعض الأنظمة كلا مجموعتي الماكرو. توفر glibc كلًا من STAILQ و SIMPLEQ، وهما متطابقتان، باستثناء أن مكافئ SIMPLEQ لـ STAILQ_CONCAT غير موجود.

مشاكل

STAILQ_FOREACH() لا تسمح بإزالة أو تحرير var داخل الحلقة، حيث قد يتداخل ذلك مع التجوال. STAILQ_FOREACH_SAFE()، الموجودة في أنظمة BSD ولكن ليس في glibc، تحل هذه المشكلة عن طريق السماح بإزالة var من القائمة وتحريرها داخل الحلقة دون تعطيل التجوال.

مثال

#include <stddef.h>
#include <stdio.h>
#include <stdlib.h>
#include <sys/queue.h>
struct entry {
    int data;
    STAILQ_ENTRY(entry) entries;        /* طابور مؤشرات أحادي */
};
STAILQ_HEAD(stailhead, entry);
int
main(void)
{
    struct entry *n1, *n2, *n3, *np;
    struct stailhead head;                  /* طابور مؤشرات أحادي */
    STAILQ_INIT(&head);                     /* تهيئة الطابور */
    n1 = malloc(sizeof(struct entry));      /* إدخال في البداية */
    STAILQ_INSERT_HEAD(&head, n1, entries);
    n1 = malloc(sizeof(struct entry));      /* إدخال في النهاية */
    STAILQ_INSERT_TAIL(&head, n1, entries);
    n2 = malloc(sizeof(struct entry));      /* إدخال بعد */
    STAILQ_INSERT_AFTER(&head, n1, n2, entries);
    STAILQ_REMOVE(&head, n2, entry, entries); /* إزالة */
    free(n2);
    n3 = STAILQ_FIRST(&head);
    STAILQ_REMOVE_HEAD(&head, entries);     /* إزالة من البداية */
    free(n3);
    n1 = STAILQ_FIRST(&head);
    n1->data = 0;
    for (unsigned int i = 1; i < 5; i++) {
        n1 = malloc(sizeof(struct entry));
        STAILQ_INSERT_HEAD(&head, n1, entries);
        n1->data = i;
    }
}

الملخص

/* اجتياز أمامي */
STAILQ_FOREACH(np, &head, entries)
    printf("%i\n", np->data);

/* حذف TailQ */
n1 = STAILQ_FIRST(&head);
while (n1 != NULL) {
    n2 = STAILQ_NEXT(n1, entries);
    free(n1);
    n1 = n2;
}
STAILQ_INIT(&head);
exit(EXIT_SUCCESS);

الوصف

يقوم الكود بإجراء اجتياز أمامي لهيكل STAILQ، مع طباعة قيمة data لكل عقدة. بالإضافة إلى ذلك، يقوم بحذف العقد من الهيكل بشكل تكراري وتهيئة الهيكل.

قيمة الإرجاع

قيمة الإرجاع هي NULL في حالة حدوث خطأ.
