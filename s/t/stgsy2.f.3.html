subroutine stgsy2 (character trans, integer ijob, integer m, integer n, real, dimension( lda, * ) a, integer lda, real, dimension( ldb, * ) b, integer ldb, real, dimension( ldc, * ) c, integer ldc, real, dimension( ldd, * ) d, integer ldd, real, dimension( lde, * ) e, integer lde, real, dimension( ldf, * ) f, integer ldf, real scale, real rdsum, real rdscal, integer, dimension( * ) iwork, integer pq, integer info)

STGSY2 يحل معادلة سيلفستر المعممة (خوارزمية غير محظورة).

الغرض:

!>
!> STGSY2 يحل معادلة سيلفستر المعممة:
!>
!>             A * R - L * B = scale * C                (1)
!>             D * R - L * E = scale * F,
!>
!> باستخدام BLAS المستوى 1 و 2. حيث R و L مصفوفتان غير معروفتين بحجم M-by-N،
!> (A, D)، (B, E) و (C, F) هي أزواج مصفوفات معطاة بحجم M-by-M،
!> N-by-N و M-by-N، على التوالي، بمدخلات حقيقية. يجب أن تكون (A, D) و (B, E)
!> في شكل شوري كانونيكي معمّم، أي A و B شبه مثلثة علوية و D و E مثلثية علوية.
!> الحل (R, L) يكتب فوق (C, F). 0 <= SCALE <= 1 هو عامل قياس خرج
!> تم اختياره لتجنب الحمل الزائد.
!>
!> في ترميز المصفوفة، حل المعادلة (1) يتوافق مع حل
!> Z*x = scale*b, حيث Z معرفة بـ
!>
!>        Z = [ kron(In, A)  -kron(B**T, Im) ]             (2)
!>            [ kron(In, D)  -kron(E**T, Im) ],
!>
!> Ik هي المصفوفة الهوية بحجم k و X**T هي منقولة X.
!> kron(X, Y) هو ناتج كرونيكر بين المصفوفتين X و Y.
!> في عملية حل (1)، نحل عددًا من الأنظمة المماثلة
!> حيث Dim(In)، Dim(In) = 1 أو 2.
!>
!> إذا كان TRANS = 'T'، قم بحل النظام المنقول Z**T*y = scale*b لـ y،
!> وهو ما يعادل الحل لـ R و L في
!>
!>             A**T * R  + D**T * L   = scale * C           (3)
!>             R  * B**T + L  * E**T  = scale * -F
!>
!> يستخدم هذا السيناريو لحساب تقدير لـ Dif[(A, D), (B, E)] =
!> sigma_min(Z) باستخدام اتصال عكسي مع SLACON.
!>
!> STGSY2 أيضًا (IJOB >= 1) يساهم في حساب STGSYL
!> لحد علوي على الفصل بين زوجي مصفوفات. ثم
!> (A, D) و (B, E) هي أجزاء فرعية من زوج المصفوفات في
!> STGSYL. راجع STGSYL للتفاصيل.
!>

المعلمات

TRANS 
!>          TRANS هو CHARACTER*1
!>          = 'N': حل معادلة سيلفستر المعممة (1).
!>          = 'T': حل النظام 'المنقول' (3).
!>

IJOB

!>          IJOB هو INTEGER
!>          يحدد نوع الوظيفة التي سيتم تنفيذها.
!>          = 0: حل (1) فقط.
!>          = 1: يتم حساب مساهمة من هذا النظام الفرعي لتقدير معيار فروبينيوس
!>               للمسافة بين زوجي مصفوفات. (يتم استخدام استراتيجية التطلع للمستقبل).
!>          = 2: يتم حساب مساهمة من هذا النظام الفرعي لتقدير معيار فروبينيوس
!>               للمسافة بين زوجي مصفوفات. (يتم استخدام SGECON على الأنظمة الفرعية).
!>          لا يتم الرجوع إليه إذا كان TRANS = 'T'.
!>

M

!>          M هو INTEGER
!>          عند الإدخال، يحدد M رتبة A و D، والبعد الصف للمصفوفات C، F، R و L.
!>

N

!>          N هو INTEGER
!>          عند الإدخال، يحدد N رتبة B و E، والبعد العمود للمصفوفات C، F، R و L.
!>

A

!>          A هو مصفوفة REAL، بعد (LDA, M)
!>          عند الإدخال، تحتوي A على مصفوفة شبه مثلثة علوية.
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرائد للمصفوفة A. LDA >= max(1, M).
!>

B

!>          B هو مصفوفة REAL، بعد (LDB, N)
!>          عند الإدخال، تحتوي B على مصفوفة شبه مثلثة علوية.
!>

LDB

!>          LDB هو INTEGER
!>          البعد الرائد للمصفوفة B. LDB >= max(1, N).
!>

C

!>          C هو مصفوفة REAL، بعد (LDC, N)
!>          عند الإدخال، تحتوي C على الطرف الأيمن من المعادلة المصفوفية الأولى في (1).
!>          عند الخروج، إذا كان IJOB = 0، تم الكتابة فوق C بالحل R.
!>

LDC

!>          LDC هو INTEGER
!>          البعد الرائد للمصفوفة C. LDC >= max(1, M).
!>

D

!>          D هو مصفوفة REAL، بعد (LDD, M)
!>          عند الإدخال، تحتوي D على مصفوفة مثلثية علوية.
!>

LDD

!>          LDD هو INTEGER
!>          البعد الرائد للمصفوفة D. LDD >= max(1, M).
!>

E

!>          E هو مصفوفة REAL، بعد (LDE, N)
!>          عند الإدخال، تحتوي E على مصفوفة مثلثية علوية.
!>

LDE

!>          LDE هو INTEGER
!>          البعد الرائد للمصفوفة E. LDE >= max(1, N).
!>

F

!>          F هو مصفوفة REAL، بعد (LDF, N)
!>          عند الإدخال، تحتوي F على الطرف الأيمن من المعادلة المصفوفية الثانية في (1).
!>          عند الخروج، إذا كان IJOB = 0، تم الكتابة فوق F بالحل L.
!>

LDF

!>          LDF هو INTEGER
!>          البعد الرائد للمصفوفة F. LDF >= max(1, M).
!>

SCALE

!>          SCALE هو REAL
!>          عند الخروج، 0 <= SCALE <= 1. إذا كان 0 < SCALE < 1، فإن الحلول
!>          R و L (C و F عند الإدخال) ستحتوي على حلول لنظام
!>          مضطرب قليلاً ولكن المصفوفات المدخلة A، B، D و
!>          E لم تتغير. إذا كان SCALE = 0، ستحتوي R و L على
!>          حلول للنظام المتجانس مع C = F = 0. عادة،
!>          SCALE = 1.
!>

RDSUM

!>          RDSUM هو REAL
!>          عند الإدخال، مجموع مربعات المساهمات المحسوبة لـ
!>          تقدير Dif قيد الحساب بواسطة STGSYL، حيث تم فصل
!>          عامل القياس RDSCAL (انظر أدناه).
!>          عند الخروج، مجموع المربعات المقابل المحدث بـ
!>          المساهمات من النظام الفرعي الحالي.
!>          إذا كان TRANS = 'T'، لا يتم لمس RDSUM.
!>          ملاحظة: RDSUM له معنى فقط عندما يتم استدعاء STGSY2 بواسطة STGSYL.
!>

RDSCAL

!>          RDSCAL هو REAL
!>          عند الإدخال، عامل القياس المستخدم لمنع الحمل الزائد في RDSUM.
!>          عند الخروج، يتم تحديث RDSCAL فيما يتعلق بالمساهمات الحالية
!>          في RDSUM.
!>          إذا كان TRANS = 'T'، لا يتم لمس RDSCAL.
!>          ملاحظة: RDSCAL له معنى فقط عندما يتم استدعاء STGSY2 بواسطة
!>                STGSYL.
!>

IWORK

!>          IWORK هو مصفوفة INTEGER، بعد (M+N+2)
!>

PQ

!>          PQ هو INTEGER
!>          عند الخروج، عدد الأنظمة الفرعية (بحجم 2x2، 4x4 و
!>          8x8) التي تم حلها بواسطة هذا الروتين.
!>

INFO

!>          INFO هو INTEGER
!>          عند الخروج، إذا تم تعيين INFO إلى
!>            =0: الخروج الناجح
!>            <0: إذا كان INFO = -i، فإن الوسيط i كان له قيمة غير قانونية.
!>            >0: أزواج المصفوفات (A, D) و (B, E) لها قيم ذاتية مشتركة أو قريبة جدًا.
!>
