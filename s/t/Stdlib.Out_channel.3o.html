الحزمة: ocaml

التراخيص: LGPL2.1, custom: QPL-1.0

val is_buffered : t -> bool
is_buffered oc تُرجع ما إذا كانت القناة oc مُخزنة مؤقتًا (راجع Out_channel.set_buffered).
val isatty : t -> bool
isatty oc تكون true إذا كانت oc تشير إلى طرفية أو نافذة وحدة تحكم، و false بخلاف ذلك.
منذ 5.1

أمثلة

كتابة محتوى ملف:

      let write_file file s =
        Out_channel.with_open_bin file
          (fun oc -> Out_channel.output_string oc s))
