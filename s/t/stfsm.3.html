subroutine stfsm (transr, side, uplo, trans, diag, m, n, alpha, a, b, ldb)
STFSM تحل معادلة مصفوفة (أحد المعاملات مصفوفة ثلاثية الشكل بتنسيق RFP).

توثيق الوظيفة/الروتين الفرعي

subroutine stfsm (character transr, character side, character uplo, character trans, character diag, integer m, integer n, real alpha, real, dimension( 0: * ) a, real, dimension( 0: ldb-1, 0: * ) b, integer ldb)

STFSM تحل معادلة مصفوفة (أحد المعاملات مصفوفة ثلاثية الشكل بتنسيق RFP).

الغرض:

!>
!> روتين من المستوى 3 BLAS لمصفوفة A بتنسيق RFP.
!>
!> STFSM  تحل المعادلة المصفوفية
!>
!>    op( A )*X = alpha*B  أو  X*op( A ) = alpha*B
!>
!> حيث alpha هو عدد قياسي، X و B هما مصفوفتان بحجم m × n، A هي مصفوفة
!> ثلاثية الشكل (وحدات، أو غير وحدات، علوية أو سفلية) و op( A ) هي إحدى
!>
!>    op( A ) = A   أو   op( A ) = A**T.
!>
!> A بتنسيق التعبئة الكاملة المستطيلة (RFP).
!>
!> يتم استبدال المصفوفة X في B.
!>

المعاملات

TRANSR 
!>          TRANSR هو CHARACTER*1
!>          = 'N':  يتم تخزين الشكل الطبيعي لـ RFP A؛
!>          = 'T':  يتم تخزين الشكل المنقول لـ RFP A.
!>

SIDE

!>          SIDE هو CHARACTER*1
!>           عند الدخول، تحدد SIDE ما إذا كانت op( A ) تظهر على اليسار
!>           أو اليمين لـ X كما يلي:
!>
!>              SIDE = 'L' أو 'l'   op( A )*X = alpha*B.
!>
!>              SIDE = 'R' أو 'r'   X*op( A ) = alpha*B.
!>
!>           تبقى دون تغيير عند الخروج.
!>

UPLO

!>          UPLO هو CHARACTER*1
!>           عند الدخول، تحدد UPLO ما إذا كانت المصفوفة RFP A قد جاءت من
!>           مصفوفة علوية أو سفلية ثلاثية الشكل كما يلي:
!>           UPLO = 'U' أو 'u' RFP A جاءت من مصفوفة علوية ثلاثية الشكل
!>           UPLO = 'L' أو 'l' RFP A جاءت من مصفوفة سفلية ثلاثية الشكل
!>
!>           تبقى دون تغيير عند الخروج.
!>

TRANS

!>          TRANS هو CHARACTER*1
!>           عند الدخول، يحدد TRANS شكل op( A ) المراد استخدامه
!>           في ضرب المصفوفات كما يلي:
!>
!>              TRANS  = 'N' أو 'n'   op( A ) = A.
!>
!>              TRANS  = 'T' أو 't'   op( A ) = A'.
!>
!>           يبقى دون تغيير عند الخروج.
!>

DIAG

!>          DIAG هو CHARACTER*1
!>           عند الدخول، يحدد DIAG ما إذا كانت RFP A ثلاثية الشكل وحدات
!>           أم لا كما يلي:
!>
!>              DIAG = 'U' أو 'u'   يفترض أن A ثلاثية الشكل وحدات.
!>
!>              DIAG = 'N' أو 'n'   لا يفترض أن A ثلاثية الشكل وحدات.
!>
!>           يبقى دون تغيير عند الخروج.
!>

M

!>          M هو INTEGER
!>           عند الدخول، يحدد M عدد صفوف B. يجب أن يكون M على الأقل صفر.
!>           يبقى دون تغيير عند الخروج.
!>

N

!>          N هو INTEGER
!>           عند الدخول، يحدد N عدد أعمدة B. يجب أن يكون N على الأقل صفر.
!>           يبقى دون تغيير عند الخروج.
!>

ALPHA

!>          ALPHA هو REAL
!>           عند الدخول، يحدد ALPHA العدد القياسي alpha. عندما يكون alpha
!>           صفرًا، لا يتم الإشارة إلى A وليس من الضروري تعيين B قبل
!>           الدخول.
!>           يبقى دون تغيير عند الخروج.
!>

A

!>          A هو مصفوفة REAL، ذات بعد (NT)
!>           NT = N*(N+1)/2 إذا كانت SIDE='R' و NT = M*(M+1)/2 بخلاف ذلك.
!>           عند الدخول، المصفوفة A بتنسيق RFP.
!>           يتم وصف تنسيق RFP بواسطة TRANSR و UPLO و N كما يلي:
!>           إذا كانت TRANSR='N' فإن RFP A هي (0:N,0:K-1) عندما يكون N زوجيًا؛
!>           K=N/2. RFP A هي (0:N-1,0:K) عندما يكون N فرديًا؛ K=N/2. إذا
!>           كانت TRANSR = 'T' فإن RFP هي منقول RFP A كما
!>           تم تعريفها عندما كانت TRANSR = 'N'. يتم تعريف محتويات RFP A
!>           بواسطة UPLO كما يلي: إذا كانت UPLO = 'U' فإن RFP A تحتوي على
!>           عناصر NT من A المعبأة العلوية إما بالشكل العادي أو
!>           الشكل المنقول. إذا كانت UPLO = 'L' فإن RFP A تحتوي على
!>           عناصر NT من A المعبأة السفلية إما بالشكل العادي أو
!>           الشكل المنقول. LDA لـ RFP A هو (N+1)/2 عندما
!>           يكون TRANSR = 'T'. عندما يكون TRANSR 'N' فإن LDA هو N+1 عندما
!>           يكون N زوجيًا وهو N عندما يكون فرديًا.
!>           انظر الملاحظة أدناه لمزيد من التفاصيل. تبقى دون تغيير عند الخروج.
!>

B

!>          B هي مصفوفة REAL، ذات بعد (LDB,N)
!>           قبل الدخول، يجب أن يحتوي الجزء الرئيسي m × n من المصفوفة B على
!>           مصفوفة الطرف الأيمن B، وعند الخروج يتم استبدالها بالمصفوفة
!>           الحل X.
!>

LDB

!>          LDB هو INTEGER
!>           عند الدخول، يحدد LDB البعد الأول لـ B كما تم تعريفه
!>           في البرنامج (الفرعي) المستدعي. يجب أن يكون LDB على الأقل
!>           max( 1, m ).
!>           يبقى دون تغيير عند الخروج.
!>
