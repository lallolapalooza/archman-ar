subroutine stfsm (transr, side, uplo, trans, diag, m, n, alpha, a, b, ldb)
STFSM يحل معادلة مصفوفة (أحد المعاملات مصفوفة مثلثية بتنسيق RFP).

توثيق الوظيفة/الوحدة الفرعية

subroutine stfsm (character transr, character side, character uplo, character trans, character diag, integer m, integer n, real alpha, real, dimension( 0: * ) a, real, dimension( 0: ldb-1, 0: * ) b, integer ldb)

STFSM يحل معادلة مصفوفة (أحد المعاملات مصفوفة مثلثية بتنسيق RFP).

الغرض:

!>
!> مستوى 3 BLAS مثل روتين لـ A بتنسيق RFP.
!>
!> STFSM يحل معادلة المصفوفة
!>
!>    op( A )*X = alpha*B  أو  X*op( A ) = alpha*B
!>
!> حيث alpha هو عدد قياسي، X و B هما مصفوفتان m × n، A مصفوفة مثلثية
!> (وحدة، أو غير وحدة، علوية أو سفلية) و op( A ) هو أحد
!>
!>    op( A ) = A   أو   op( A ) = A**T.
!>
!> A بتنسيق التعبئة الكاملة المستطيلة (RFP).
!>
!> يتم استبدال المصفوفة X فوق B.
!>

المعلمات

TRANSR 
!>          TRANSR هو CHARACTER*1
!>          = 'N':  يتم تخزين الشكل العادي لـ RFP A؛
!>          = 'T':  يتم تخزين الشكل المنقول لـ RFP A.
!>

SIDE

!>          SIDE هو CHARACTER*1
!>           عند الدخول، تحدد SIDE ما إذا كان op( A ) يظهر على اليسار
!>           أو اليمين لـ X كما يلي:
!>
!>              SIDE = 'L' أو 'l'   op( A )*X = alpha*B.
!>
!>              SIDE = 'R' أو 'r'   X*op( A ) = alpha*B.
!>
!>           لا يتغير عند الخروج.
!>

UPLO

!>          UPLO هو CHARACTER*1
!>           عند الدخول، تحدد UPLO ما إذا كانت المصفوفة RFP A جاءت من
!>           مصفوفة مثلثية علوية أو سفلية كما يلي:
!>           UPLO = 'U' أو 'u' جاءت RFP A من مصفوفة مثلثية علوية
!>           UPLO = 'L' أو 'l' جاءت RFP A من مصفوفة مثلثية سفلية
!>
!>           لا يتغير عند الخروج.
!>

TRANS

!>          TRANS هو CHARACTER*1
!>           عند الدخول، تحدد TRANS شكل op( A ) الذي سيتم استخدامه
!>           في ضرب المصفوفات كما يلي:
!>
!>              TRANS  = 'N' أو 'n'   op( A ) = A.
!>
!>              TRANS  = 'T' أو 't'   op( A ) = A'.
!>
!>           لا يتغير عند الخروج.
!>

DIAG

!>          DIAG هو CHARACTER*1
!>           عند الدخول، تحدد DIAG ما إذا كانت RFP A وحدة
!>           مثلثية أم لا:
!>
!>              DIAG = 'U' أو 'u'   يفترض أن A مصفوفة مثلثية وحدة.
!>
!>              DIAG = 'N' أو 'n'   لا يفترض أن A مصفوفة مثلثية وحدة.
!>
!>           لا يتغير عند الخروج.
!>

M

!>          M هو INTEGER
!>           عند الدخول، تحدد M عدد صفوف B. يجب أن تكون M على الأقل
!>           صفر.
!>           لا يتغير عند الخروج.
!>

N

!>          N هو INTEGER
!>           عند الدخول، تحدد N عدد أعمدة B. يجب أن تكون N على الأقل
!>           صفر.
!>           لا يتغير عند الخروج.
!>

ALPHA

!>          ALPHA هو REAL
!>           عند الدخول، يحدد ALPHA العدد القياسي alpha. عندما يكون alpha
!>           صفرًا، لا يتم الرجوع إلى A ولا يلزم تعيين B قبل
!>           الدخول.
!>           لا يتغير عند الخروج.
!>

A

!>          A هو مصفوفة REAL، أبعادها (NT)
!>           NT = N*(N+1)/2 إذا كان SIDE='R' و NT = M*(M+1)/2 بخلاف ذلك.
!>           عند الدخول، المصفوفة A بتنسيق RFP.
!>           يتم وصف تنسيق RFP بواسطة TRANSR و UPLO و N كما يلي:
!>           إذا كان TRANSR='N' فإن RFP A هو (0:N,0:K-1) عندما يكون N زوجيًا؛
!>           K=N/2. RFP A هو (0:N-1,0:K) عندما يكون N فرديًا؛ K=N/2. إذا
!>           كان TRANSR = 'T' فإن RFP هو منقول RFP A كما
!>           هو محدد عندما يكون TRANSR = 'N'. يتم تعريف محتويات RFP A
!>           بواسطة UPLO كما يلي: إذا كان UPLO = 'U' فإن RFP A يحتوي على
!>           NT عنصرًا من A العلوية المعبأة إما بالشكل العادي أو
!>           المنقول. إذا كان UPLO = 'L' فإن RFP A يحتوي على
!>           NT عنصرًا من A السفلية المعبأة إما بالشكل العادي أو
!>           المنقول. LDA لـ RFP A هو (N+1)/2 عندما
!>           يكون TRANSR = 'T'. عندما يكون TRANSR = 'N' فإن LDA هو N+1 عندما
!>           يكون N زوجيًا وهو N عندما يكون فرديًا.
!>           انظر الملاحظة أدناه لمزيد من التفاصيل. لا يتغير عند الخروج.
!>

B

!>          B هي مصفوفة REAL، أبعادها (LDB,N)
!>           قبل الدخول، يجب أن يحتوي الجزء الأمامي m × n من المصفوفة B على
!>           مصفوفة الطرف الأيمن B، وعند الخروج يتم استبدالها
!>           بالمصفوفة الحل X.
!>

LDB

!>          LDB هو INTEGER
!>           عند الدخول، تحدد LDB البعد الأول لـ B كما هو معلن
!>           في البرنامج (البرنامج الفرعي) المستدعي. يجب أن تكون LDB على الأقل
!>           max( 1, m ).
!>           لا يتغير عند الخروج.
!>

!>  منقول الأعمدة الثلاثة الأولى من AP العلوية.
