stdarg(3) دليل وظائف المكتبة stdarg(3)

الاسم

stdarg, va_start, va_arg, va_end, va_copy - قوائم وسائط متغيرة

المكتبة

مكتبة C القياسية (libc، -lc)

الملخص

#include <stdarg.h>
void va_start(va_list ap, last);
type va_arg(va_list ap, type);
void va_end(va_list ap);
void va_copy(va_list dest, va_list src);

الوصف

قد يتم استدعاء دالة بعدد متغير من الوسائط بأنواع متغيرة. يتضمن ملف التضمين <stdarg.h> تعريفًا لنوع va_list وتعريفًا لثلاث وحدات ماكرو للتنقل عبر قائمة الوسائط التي لا يُعرف عددها وأنواعها للدالة المستدعاة.

يجب على الدالة المستدعاة تعريف كائن من النوع va_list والذي تستخدمه وحدات الماكرو va_start() و va_arg() و va_end().

va_start()

تقوم وحدة الماكرو va_start() بتهيئة ap للاستخدام اللاحق بواسطة va_arg() و va_end()، ويجب استدعاؤها أولاً.

الوسيط last هو اسم الوسيط الأخير قبل قائمة الوسائط المتغيرة، أي الوسيط الأخير الذي تعرف الدالة المستدعية نوعه.

نظرًا لأنه قد يتم استخدام عنوان هذا الوسيط في وحدة الماكرو va_start()، فلا يجب تعريفه كمتغير مسجل، أو كنوع دالة أو مصفوفة.

va_arg()

تتوسع وحدة الماكرو va_arg() إلى تعبير له النوع والقيمة للوسيط التالي في الاستدعاء. الوسيط ap هو va_list ap الذي تم تهيئته بواسطة va_start(). كل استدعاء لـ va_arg() يعدل ap بحيث يعيد الاستدعاء التالي الوسيط التالي. الوسيط type هو اسم نوع محدد بحيث يمكن الحصول على نوع مؤشر إلى كائن له النوع المحدد ببساطة عن طريق إضافة * إلى type.

يعيد الاستخدام الأول لوحدة الماكرو va_arg() بعد استخدام وحدة الماكرو va_start() الوسيط الذي يلي last. تعيد الاستدعاءات المتتالية قيم الوسائط المتبقية.

إذا لم يكن هناك وسيط تالٍ، أو إذا لم يكن type متوافقًا مع نوع الوسيط الفعلي التالي (كما هو مروج وفقًا لترقيات الوسائط الافتراضية)، فس تحدث أخطاء عشوائية.

إذا تم تمرير ap إلى دالة تستخدم va_arg(ap,type)، فإن قيمة ap غير معرفة بعد عودة تلك الدالة.

va_end()

يجب مطابقة كل استدعاء لـ va_start() باستدعاء مقابل لـ va_end() في نفس الدالة. بعد استدعاء va_end(ap)، يكون المتغير ap غير معرف. من الممكن إجراء عمليات مسح متعددة للقائمة، كل منها محاط بـ va_start() و va_end(). قد تكون va_end() وحدة ماكرو أو دالة.

va_copy()

تقوم وحدة الماكرو va_copy() بنسخ قائمة الوسائط المتغيرة (المهيأة مسبقًا) src إلى dest. السلوك كما لو تم تطبيق va_start() على dest بنفس الوسيط last، متبوعًا بنفس عدد استدعاءات va_arg() التي تم استخدامها للوصول إلى الحالة الحالية لـ src.

قد يكون التنفيذ الواضح هو جعل va_list مؤشرًا إلى إطار المكدس للدالة المتغيرة. في مثل هذا الإعداد (وهو الأكثر شيوعًا إلى حد بعيد) لا يبدو أن هناك ما يمنع التعيين


va_list aq = ap;

للأسف، هناك أيضًا أنظمة تجعلها مصفوفة من المؤشرات (بطول 1)، وهناك تحتاج إلى


va_list aq;
*aq = *ap;

أخيرًا، في الأنظمة التي يتم فيها تمرير الوسائط في المسجلات، قد يكون من الضروري لـ va_start() تخصيص ذاكرة، وتخزين الوسائط هناك، وأيضًا مؤشر إلى الوسيط التالي، بحيث يمكن لـ va_arg() التنقل عبر القائمة. الآن يمكن لـ va_end() تحرير الذاكرة المخصصة مرة أخرى. لاستيعاب هذا الموقف، يضيف C99 وحدة ماكرو va_copy()، بحيث يمكن استبدال التعيين أعلاه بـ


va_list aq;
va_copy(aq, ap);
...
va_end(aq);

يجب مطابقة كل استدعاء لـ va_copy() باستدعاء مقابل لـ va_end() في نفس الدالة. بعض الأنظمة التي لا توفر va_copy() لديها __va_copy بدلاً من ذلك، نظرًا لأن هذا كان الاسم المستخدم في المسودة المقترحة.

تحذيرات

على عكس وحدات الماكرو varargs التاريخية، فإن وحدات الماكرو stdarg لا تسمح للمبرمجين بترميز دالة بدون وسائط ثابتة. تولد هذه المشكلة عملاً بشكل رئيسي عند تحويل كود varargs إلى كود stdarg، ولكنها تخلق أيضًا صعوبات للدوال المتغيرة التي ترغب في تمرير جميع وسائطها إلى دالة تأخذ وسيط va_list، مثل vfprintf(3).

أمثلة

الدالة foo تأخذ سلسلة من أحرف التنسيق وتطبع الوسيط المرتبط بكل حرف تنسيق بناءً على النوع.

#include <stdio.h>
#include <stdarg.h>
void
foo(char *fmt, ...)   /* '...' هو بناء جملة C لدالة متغيرة */
{
    va_list ap;
    int d;
    char c;
    char *s;
    va_start(ap, fmt);
    while (*fmt)
        switch (*fmt++) {
        case 's':              /* string */
            s = va_arg(ap, char *);
            printf("string %s\n", s);
            break;
        case 'd':              /* int */
            d = va_arg(ap, int);
            printf("int %d\n", d);
            break;
        case 'c':              /* char */
            /* نحتاج إلى تحويل هنا حيث أن va_arg
               تأخذ فقط الأنواع المروجة بالكامل */
            c = (char) va_arg(ap, int);
            printf("char %c\n", c);
            break;
        }
    va_end(ap);
}
