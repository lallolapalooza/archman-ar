strtok(3) دليل دوال المكتبة strtok(3)

الاسم

strtok, strtok_r - استخراج أجزاء من النصوص

المكتبة

مكتبة C القياسية (libc، -lc)

الملخص

#include <string.h>
char *strtok(char *_Nullable restrict str, const char *restrict delim);
char *strtok_r(char *_Nullable restrict str, const char *restrict delim,
               char **restrict saveptr);
متطلبات ماكرو اختبار الميزات لـ glibc (انظر feature_test_macros(7)):

strtok_r():

    _POSIX_C_SOURCE
        || /* glibc <= 2.19: */ _BSD_SOURCE || _SVID_SOURCE

الوصف

تقوم الدالة strtok() بتقسيم النص إلى سلسلة من الأجزاء الصفرية أو الأكثر غير الفارغة. في الاستدعاء الأول لـ strtok()، يجب تحديد النص المراد تحليله في str. في كل استدعاء لاحق يجب تحليل نفس النص، يجب أن يكون str هو NULL.

يحدد الوسيط delim مجموعة من البايتات التي تفصل الأجزاء في النص المحلل. يجوز للمستدعي تحديد سلاسل مختلفة في delim في استدعاءات متتالية تحلل نفس النص.

كل استدعاء لـ strtok() يعيد مؤشرًا إلى سلسلة منتهية بـ null تحتوي على الجزء التالي. لا يتضمن هذا النص بايت الفصل. إذا لم يتم العثور على المزيد من الأجزاء، تعيد strtok() القيمة NULL.

سلسلة من الاستدعاءات لـ strtok() التي تعمل على نفس النص تحافظ على مؤشر يحدد النقطة التي يبدأ منها البحث عن الجزء التالي. يضبط الاستدعاء الأول لـ strtok() هذا المؤشر للإشارة إلى البايت الأول من النص. يتم تحديد بداية الجزء التالي عن طريق المسح للأمام بحثًا عن البايت التالي غير الفاصل في str. إذا تم العثور على مثل هذا البايت، فيتم اعتباره بداية الجزء التالي. إذا لم يتم العثور على مثل هذا البايت، فلا توجد أجزاء أخرى، وتعيد strtok() القيمة NULL. (وبالتالي، سيؤدي النص الفارغ أو الذي يحتوي فقط على فواصل إلى أن تعيد strtok() القيمة NULL في الاستدعاء الأول.)

يتم العثور على نهاية كل جزء عن طريق المسح للأمام حتى يتم العثور على بايت الفصل التالي أو حتى يتم الوصول إلى نهاية النص (null byte ('\0')). إذا تم العثور على بايت فصل، فسيتم استبداله ببايت null لإنهاء الجزء الحالي، وتحفظ strtok() مؤشرًا إلى البايت التالي؛ وسيتم استخدام هذا المؤشر كنقطة بداية عند البحث عن الجزء التالي. في هذه الحالة، تعيد strtok() مؤشرًا إلى بداية الجزء الذي تم العثور عليه.

من الوصف أعلاه، يترتب على ذلك أن سلسلة من اثنين أو أكثر من بايتات الفصل المتجاورة في النص المحلل تعتبر فاصلًا واحدًا، ويتم تجاهل بايتات الفصل في بداية النص أو نهايته. بمعنى آخر: الأجزاء التي تعيدها strtok() هي دائمًا سلاسل غير فارغة. وبالتالي، على سبيل المثال، بالنظر إلى النص "aaa;;bbb,"، فإن الاستدعاءات المتتالية لـ strtok() التي تحدد سلسلة الفواصل ";," ستعيد النصين "aaa" و "bbb"، ثم مؤشر null.

الدالة strtok_r() هي نسخة قابلة لإعادة الدخول للدالة strtok(). الوسيط saveptr هو مؤشر إلى متغير من نوع char * يستخدم داخليًا بواسطة strtok_r() للحفاظ على السياق بين الاستدعاءات المتتالية التي تحلل نفس النص.

في الاستدعاء الأول لـ strtok_r()، يجب أن يشير str إلى النص المراد تحليله، ويتم تجاهل قيمة *saveptr (ولكن انظر VERSIONS). في الاستدعاءات اللاحقة، يجب أن يكون str هو NULL، ويجب أن يكون saveptr (والذاكرة التي يشير إليها) دون تغيير منذ الاستدعاء السابق.

يمكن تحليل النصوص المختلفة بشكل متزامن باستخدام سلاسل من الاستدعاءات لـ strtok_r() التي تحدد وسيطات saveptr مختلفة.

قيمة الإرجاع

تعيد الدالتان strtok() و strtok_r() مؤشرًا إلى الجزء التالي، أو NULL إذا لم تكن هناك أجزاء أخرى.

أخطاء

كن حذرًا عند استخدام هذه الدوال. إذا استخدمتها، لاحظ أن:

•
هذه الدوال تعدل معاملها الأول.
•
لا يمكن استخدام هذه الدوال على نصوص ثابتة.
•
هوية بايت الفصل مفقودة.
•
تستخدم الدالة strtok() ذاكرة تخزين مؤقت ثابتة أثناء التحليل، لذا فهي ليست آمنة للخيوط. استخدم strtok_r() إذا كان هذا يهمك.

أمثلة

يستخدم البرنامج أدناه حلقات متداخلة تستخدم strtok_r() لتقسيم النص إلى تسلسل هرمي من مستويين من الأجزاء. يحدد وسيط سطر الأوامر الأول النص المراد تحليله. يحدد الوسيط الثاني بايتات الفصل المستخدمة لفصل هذا النص إلى أجزاء "رئيسية". يحدد الوسيط الثالث بايتات الفصل المستخدمة لفصل الأجزاء "الرئيسية" إلى أجزاء فرعية.

مثال على المخرجات التي ينتجها هذا البرنامج هو كالتالي:


$ ./a.out 'a/bbb///cc;xxx:yyy:' ':;' '/'
1: a/bbb///cc
         --> a
         --> bbb
         --> cc
2: xxx
         --> xxx
3: yyy
         --> yyy

مصدر البرنامج

#include <stdio.h>
#include <stdlib.h>
#include <string.h>
int
main(int argc, char *argv[])
{
    char *str1, *str2, *token, *subtoken;
    char *saveptr1, *saveptr2;
    int j;
    if (argc != 4) {
        fprintf(stderr, "Usage: %s string delim subdelim\n",
                argv[0]);
        exit(EXIT_FAILURE);
    }
    for (j = 1, str1 = argv[1]; ; j++, str1 = NULL) {
        token = strtok_r(str1, argv[2], &saveptr1);
        if (token == NULL)
            break;
        printf("%d: %s\n", j, token);
        for (str2 = token; ; str2 = NULL) {
            subtoken = strtok_r(str2, argv[3], &saveptr2);
            if (subtoken == NULL)
                break;
            printf("\t --> %s\n", subtoken);
        }
    }
    exit(EXIT_SUCCESS);
}

مثال آخر على برنامج يستخدم strtok() يمكن العثور عليه في getaddrinfo_a(3).
