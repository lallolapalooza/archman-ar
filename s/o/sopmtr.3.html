subroutine sopmtr (character side, character uplo, character trans, integer m, integer n, real, dimension( * ) ap, real, dimension( * ) tau, real, dimension( ldc, * ) c, integer ldc, real, dimension( * ) work, integer info)

SOPMTR

الغرض:

!>
!> SOPMTR تستبدل المصفوفة العامة ذات الأعداد الحقيقية M-by-N المسماة C بـ
!>
!>                 SIDE = 'L'     SIDE = 'R'
!> TRANS = 'N':      Q * C          C * Q
!> TRANS = 'T':      Q**T * C       C * Q**T
!>
!> حيث Q هي مصفوفة متعامدة حقيقية من الرتبة nq، مع nq = m إذا كان
!> SIDE = 'L' و nq = n إذا كان SIDE = 'R'. تُعرّف Q بأنها حاصل ضرب
!> nq-1 عاكسات عنصرية، كما تم إرجاعها بواسطة SSPTRD باستخدام تخزين مضغوط:
!>
!> إذا كان UPLO = 'U'، فإن Q = H(nq-1) . . . H(2) H(1);
!>
!> إذا كان UPLO = 'L'، فإن Q = H(1) H(2) . . . H(nq-1).
!>

المعلمات

SIDE 
!>          SIDE هو CHARACTER*1
!>          = 'L': تطبيق Q أو Q**T من اليسار؛
!>          = 'R': تطبيق Q أو Q**T من اليمين.
!>

UPLO

!>          UPLO هو CHARACTER*1
!>          = 'U': تم استخدام التخزين المثلث العلوي المضغوط في الاستدعاء
!>                 السابق لـ SSPTRD؛
!>          = 'L': تم استخدام التخزين المثلث السفلي المضغوط في الاستدعاء
!>                 السابق لـ SSPTRD.
!>

TRANS

!>          TRANS هو CHARACTER*1
!>          = 'N':  بدون تبديل، تطبيق Q؛
!>          = 'T':  تبديل، تطبيق Q**T.
!>

M

!>          M هو INTEGER
!>          عدد صفوف المصفوفة C. M >= 0.
!>

N

!>          N هو INTEGER
!>          عدد أعمدة المصفوفة C. N >= 0.
!>

AP

!>          AP هي مصفوفة REAL، أبعادها
!>                               (M*(M+1)/2) إذا كان SIDE = 'L'
!>                               (N*(N+1)/2) إذا كان SIDE = 'R'
!>          المتجهات التي تحدد العاكسات الأولية، كما تم
!>          إرجاعها بواسطة SSPTRD.  يتم تعديل AP بواسطة الروتين ولكن
!>          يتم استعادتها عند الخروج.
!>

TAU

!>          TAU هي مصفوفة REAL، أبعادها (M-1) إذا كان SIDE = 'L'
!>                                     أو (N-1) إذا كان SIDE = 'R'
!>          يجب أن تحتوي TAU(i) على العامل القياسي للعاكس الأولي
!>          H(i)، كما تم إرجاعه بواسطة SSPTRD.
!>

C

!>          C هي مصفوفة REAL، أبعادها (LDC,N)
!>          عند الدخول، المصفوفة M-by-N المسماة C.
!>          عند الخروج، يتم استبدال C بـ Q*C أو Q**T*C أو C*Q**T أو C*Q.
!>

LDC

!>          LDC هو INTEGER
!>          البعد الرائد للمصفوفة C. LDC >= max(1,M).
!>

WORK

!>          WORK هي مصفوفة REAL، أبعادها
!>                                   (N) إذا كان SIDE = 'L'
!>                                   (M) إذا كان SIDE = 'R'
!>

INFO

!>          INFO هو INTEGER
!>          = 0:  خروج ناجح
!>          < 0:  إذا كان INFO = -i، فإن الوسيط i له قيمة غير قانونية
!>
