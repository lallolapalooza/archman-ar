subroutine sorgbr (character vect, integer m, integer n, integer k, real, dimension( lda, * ) a, integer lda, real, dimension( * ) tau, real, dimension( * ) work, integer lwork, integer info)

SORGBR

الغرض:

!>
!> SORGBR تنشئ إحدى المصفوفات المتعامدة الحقيقية Q أو P**T
!> التي تحددها SGEBRD عند اختزال مصفوفة حقيقية A إلى شكل ثنائي القطر: A = Q * B * P**T.
!> يتم تعريف Q و P**T كناتج للمنعكسات الأولية H(i) أو G(i) على التوالي.
!>
!> إذا كان VECT = 'Q'، يُفترض أن A كانت مصفوفة M-by-K، و Q
!> من الرتبة M:
!> إذا كان m >= k، فإن Q = H(1) H(2) . . . H(k) وتعيد SORGBR الأعمدة n
!> الأولى من Q، حيث m >= n >= k؛
!> إذا كان m < k، فإن Q = H(1) H(2) . . . H(m-1) وتعيد SORGBR Q كمصفوفة
!> M-by-M.
!>
!> إذا كان VECT = 'P'، يُفترض أن A كانت مصفوفة K-by-N، و P**T
!> من الرتبة N:
!> إذا كان k < n، فإن P**T = G(k) . . . G(2) G(1) وتعيد SORGBR الصفوف m
!> الأولى من P**T، حيث n >= m >= k؛
!> إذا كان k >= n، فإن P**T = G(n-1) . . . G(2) G(1) وتعيد SORGBR P**T كمصفوفة
!> N-by-N.
!>

المعلمات

VECT 
!>          VECT هو CHARACTER*1
!>          يحدد ما إذا كانت المصفوفة Q أو المصفوفة P**T مطلوبة، كما هو محدد في التحويل المطبق بواسطة SGEBRD:
!>          = 'Q':  إنشاء Q؛
!>          = 'P':  إنشاء P**T.
!>

M

!>          M هو INTEGER
!>          عدد صفوف المصفوفة Q أو P**T المراد إرجاعها.
!>          M >= 0.
!>

N

!>          N هو INTEGER
!>          عدد أعمدة المصفوفة Q أو P**T المراد إرجاعها.
!>          N >= 0.
!>          إذا كان VECT = 'Q'، فإن M >= N >= min(M,K)؛
!>          إذا كان VECT = 'P'، فإن N >= M >= min(N,K).
!>

K

!>          K هو INTEGER
!>          إذا كان VECT = 'Q'، فإن عدد الأعمدة في المصفوفة الأصلية M-by-K
!>          التي تم اختزالها بواسطة SGEBRD.
!>          إذا كان VECT = 'P'، فإن عدد الصفوف في المصفوفة الأصلية K-by-N
!>          التي تم اختزالها بواسطة SGEBRD.
!>          K >= 0.
!>

A

!>          A هي مصفوفة REAL، البعد (LDA,N)
!>          عند الإدخال، المتجهات التي تحدد المنعكسات الأولية،
!>          كما تم إرجاعها بواسطة SGEBRD.
!>          عند الخروج، المصفوفة Q أو P**T بحجم M-by-N.
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرائد للمصفوفة A. LDA >= max(1,M).
!>

TAU

!>          TAU هي مصفوفة REAL، البعد
!>                                (min(M,K)) إذا كان VECT = 'Q'
!>                                (min(N,K)) إذا كان VECT = 'P'
!>          يجب أن تحتوي TAU(i) على العامل القياسي للمنعكس الأولي
!>          H(i) أو G(i)، الذي يحدد Q أو P**T، كما
!>          تم إرجاعها بواسطة SGEBRD في وسيط مصفوفته TAUQ أو TAUP.
!>

WORK

!>          WORK هي مصفوفة REAL، البعد (MAX(1,LWORK))
!>          عند الخروج، إذا كان INFO = 0، فإن WORK(1) تعيد LWORK المثلى.
!>

LWORK

!>          LWORK هو INTEGER
!>          بعد المصفوفة WORK. LWORK >= max(1,min(M,N)).
!>          للحصول على أفضل أداء، LWORK >= min(M,N)*NB، حيث NB
!>          هو حجم الكتلة الأمثل.
!>
!>          إذا كان LWORK = -1، فسيتم افتراض استعلام عن مساحة العمل؛ يقوم الروتين
!>          بحساب حجم مصفوفة WORK الأمثل فقط، ويعيد
!>          هذه القيمة كالإدخال الأول لمصفوفة WORK، ولا يتم إصدار أي رسالة خطأ
!>          متعلقة بـ LWORK بواسطة XERBLA.
!>

INFO

!>          INFO هو INTEGER
!>          = 0:  انتهاء ناجح
!>          < 0:  إذا كان INFO = -i، فإن الوسيط i كان له قيمة غير قانونية
!>
