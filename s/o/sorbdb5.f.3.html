subroutine sorbdb5 (integer m1, integer m2, integer n, real, dimension(*) x1, integer incx1, real, dimension(*) x2, integer incx2, real, dimension(ldq1,*) q1, integer ldq1, real, dimension(ldq2,*) q2, integer ldq2, real, dimension(*) work, integer lwork, integer info)

SORBDB5

الغرض:

!>
!> SORBDB5 تعامد المتجه العمودي
!>      X = [ X1 ]
!>          [ X2 ]
!> بالنسبة للأعمدة
!>      Q = [ Q1 ] .
!>          [ Q2 ]
!> يجب أن تكون أعمدة Q متعامدة بشكل طبيعي.
!>
!> إذا كان الإسقاط صفرًا وفقًا لمعيار كاهان،
!> يتم إرجاع متجه آخر من المكمل المتعامد.
!> يتم اختيار هذا المتجه بطريقة اعتباطية ولكن حتمية.
!>
!>

المعلمات

M1 
!>          M1 هو عدد صحيح (INTEGER)
!>           بعد X1 وعدد الصفوف في Q1. 0 <= M1.
!>

M2

!>          M2 هو عدد صحيح (INTEGER)
!>           بعد X2 وعدد الصفوف في Q2. 0 <= M2.
!>

N

!>          N هو عدد صحيح (INTEGER)
!>           عدد الأعمدة في Q1 و Q2. 0 <= N.
!>

X1

!>          X1 هو مصفوفة حقيقية (REAL array)، البعد (M1)
!>           عند الإدخال، الجزء العلوي من المتجه المراد تعامده.
!>           عند الخروج، الجزء العلوي من المتجه المسقط.
!>

INCX1

!>          INCX1 هو عدد صحيح (INTEGER)
!>           زيادة لعناصر X1.
!>

X2

!>          X2 هو مصفوفة حقيقية (REAL array)، البعد (M2)
!>           عند الإدخال، الجزء السفلي من المتجه المراد
!>           تعامده. عند الخروج، الجزء السفلي من المتجه المسقط.
!>

INCX2

!>          INCX2 هو عدد صحيح (INTEGER)
!>           زيادة لعناصر X2.
!>

Q1

!>          Q1 هو مصفوفة حقيقية (REAL array)، البعد (LDQ1, N)
!>           الجزء العلوي من مصفوفة الأساس المتعامد بشكل طبيعي.
!>

LDQ1

!>          LDQ1 هو عدد صحيح (INTEGER)
!>           البعد الرئيسي لـ Q1. LDQ1 >= M1.
!>

Q2

!>          Q2 هو مصفوفة حقيقية (REAL array)، البعد (LDQ2, N)
!>           الجزء السفلي من مصفوفة الأساس المتعامد بشكل طبيعي.
!>

LDQ2

!>          LDQ2 هو عدد صحيح (INTEGER)
!>           البعد الرئيسي لـ Q2. LDQ2 >= M2.
!>

WORK

!>          WORK هو مصفوفة حقيقية (REAL array)، البعد (LWORK)
!>

LWORK

!>          LWORK هو عدد صحيح (INTEGER)
!>           بعد المصفوفة WORK. LWORK >= N.
!>

INFO

!>          INFO هو عدد صحيح (INTEGER)
!>           = 0:  خروج ناجح.
!>           < 0:  إذا كان INFO = -i، فإن المعلمة i كان لها قيمة غير قانونية.
!>
