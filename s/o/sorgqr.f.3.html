توثيق الوظيفة/الروتين الفرعي

subroutine sorgqr (integer m, integer n, integer k, real, dimension( lda, * ) a, integer lda, real, dimension( * ) tau, real, dimension( * ) work, integer lwork, integer info)

SORGQR

الغرض:

!>
!> SORGQR تنشئ مصفوفة حقيقية Q بحجم M × N بأعمدة متعامدة،
!> يتم تعريفها كأول N أعمدة لناتج K من المبدلات الأولية
!> ذات الرتبة M
!>
!>       Q  =  H(1) H(2) . . . H(k)
!>
!> كما تم إرجاعها بواسطة SGEQRF.
!>

المعلمات

M 
!>          M هو عدد صحيح (INTEGER)
!>          عدد صفوف المصفوفة Q. M >= 0.
!>

N

!>          N هو عدد صحيح (INTEGER)
!>          عدد أعمدة المصفوفة Q. M >= N >= 0.
!>

K

!>          K هو عدد صحيح (INTEGER)
!>          عدد المبدلات الأولية التي يحدد ناتجها المصفوفة Q. N >= K >= 0.
!>

A

!>          A هو مصفوفة حقيقية (REAL array)، حجمها (LDA,N)
!>          عند الإدخال، يجب أن يحتوي العمود i على المتجه الذي
!>          يحدد المبدل الأولي H(i)، لـ i = 1، 2، ...، k، كما
!>          تم إرجاعه بواسطة SGEQRF في أول k أعمدة من وسيطة مصفوفته A.
!>          عند الخروج، المصفوفة Q بحجم M × N.
!>

LDA

!>          LDA هو عدد صحيح (INTEGER)
!>          البعد الأول للمصفوفة A. LDA >= max(1,M).
!>

TAU

!>          TAU هي مصفوفة حقيقية (REAL array)، حجمها (K)
!>          يجب أن تحتوي TAU(i) على العامل القياسي للمبدل
!>          الأولي H(i)، كما تم إرجاعه بواسطة SGEQRF.
!>

WORK

!>          WORK هي مصفوفة حقيقية (REAL array)، حجمها (MAX(1,LWORK))
!>          عند الخروج، إذا كان INFO = 0، فإن WORK(1) يُرجع LWORK الأمثل.
!>

LWORK

!>          LWORK هو عدد صحيح (INTEGER)
!>          بعد المصفوفة WORK. LWORK >= max(1,N).
!>          لأفضل أداء LWORK >= N*NB، حيث NB هو
!>          حجم الكتلة الأمثل.
!>
!>          إذا كان LWORK = -1، فسيتم افتراض استعلام عن مساحة العمل؛
!>          يقوم الروتين فقط بحساب الحجم الأمثل للمصفوفة WORK،
!>          ويُرجع هذا الحجم كالإدخال الأول في المصفوفة WORK، ولا يصدر
!>          أي رسالة خطأ تتعلق بـ LWORK بواسطة XERBLA.
!>

INFO

!>          INFO هو عدد صحيح (INTEGER)
!>          = 0:  انتهاء ناجح
!>          < 0:  إذا كان INFO = -i، فإن الوسيط i لديه قيمة غير قانونية
!>
