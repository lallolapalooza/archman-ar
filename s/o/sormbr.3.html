توثيق الوظيفة/الروتين الفرعي

subroutine sormbr (character vect, character side, character trans, integer m, integer n, integer k, real, dimension( lda, * ) a, integer lda, real, dimension( * ) tau, real, dimension( ldc, * ) c, integer ldc, real, dimension( * ) work, integer lwork, integer info)

SORMBR

الغرض:

!>
!> إذا كان VECT = 'Q'، فإن SORMBR تستبدل مصفوفة C الحقيقية العامة M-by-N بـ
!>                 SIDE = 'L'     SIDE = 'R'
!> TRANS = 'N':      Q * C          C * Q
!> TRANS = 'T':      Q**T * C       C * Q**T
!>
!> إذا كان VECT = 'P'، فإن SORMBR تستبدل مصفوفة C الحقيقية العامة M-by-N بـ
!>                 SIDE = 'L'     SIDE = 'R'
!> TRANS = 'N':      P * C          C * P
!> TRANS = 'T':      P**T * C       C * P**T
!>
!> هنا Q و P**T هما المصفوفتان المتعامدتان اللتان تم تحديدهما بواسطة SGEBRD عند
!> تقليل مصفوفة حقيقية A إلى شكل ثنائي الأقطار: A = Q * B * P**T. يتم تعريف Q و
!> P**T كمنتجات من عاكسات أولية H(i) و G(i) على التوالي.
!>
!> لنفترض أن nq = m إذا كان SIDE = 'L' و nq = n إذا كان SIDE = 'R'. وبالتالي فإن nq هو
!> ترتيب المصفوفة المتعامدة Q أو P**T التي يتم تطبيقها.
!>
!> إذا كان VECT = 'Q'، يفترض أن A كانت مصفوفة NQ-by-K:
!> إذا كان nq >= k، فإن Q = H(1) H(2) . . . H(k)؛
!> إذا كان nq < k، فإن Q = H(1) H(2) . . . H(nq-1).
!>
!> إذا كان VECT = 'P'، يفترض أن A كانت مصفوفة K-by-NQ:
!> إذا كان k < nq، فإن P = G(1) G(2) . . . G(k)؛
!> إذا كان k >= nq، فإن P = G(1) G(2) . . . G(nq-1).
!>

المعلمات

VECT 
!>          VECT هو CHARACTER*1
!>          = 'Q': تطبيق Q أو Q**T؛
!>          = 'P': تطبيق P أو P**T.
!>

SIDE

!>          SIDE هو CHARACTER*1
!>          = 'L': تطبيق Q أو Q**T أو P أو P**T من اليسار؛
!>          = 'R': تطبيق Q أو Q**T أو P أو P**T من اليمين.
!>

TRANS

!>          TRANS هو CHARACTER*1
!>          = 'N':  عدم تطبيق المبدل، تطبيق Q  أو P؛
!>          = 'T':  تطبيق المبدل، تطبيق Q**T أو P**T.
!>

M

!>          M هو INTEGER
!>          عدد صفوف المصفوفة C. M >= 0.
!>

N

!>          N هو INTEGER
!>          عدد أعمدة المصفوفة C. N >= 0.
!>

K

!>          K هو INTEGER
!>          إذا كان VECT = 'Q'، فإن عدد الأعمدة في المصفوفة الأصلية
!>          التي تم تقليلها بواسطة SGEBRD.
!>          إذا كان VECT = 'P'، فإن عدد الصفوف في المصفوفة الأصلية
!>          التي تم تقليلها بواسطة SGEBRD.
!>          K >= 0.
!>

A

!>          A هو مصفوفة REAL، البعد
!>                                (LDA,min(nq,K)) إذا كان VECT = 'Q'
!>                                (LDA,nq)        إذا كان VECT = 'P'
!>          المتجهات التي تحدد العواكس الأولية H(i) و G(i)،
!>          التي تنتج عنها المصفوفتان Q و P، كما تم إرجاعها بواسطة SGEBRD.
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرائد للمصفوفة A.
!>          إذا كان VECT = 'Q'، فإن LDA >= max(1,nq)؛
!>          إذا كان VECT = 'P'، فإن LDA >= max(1,min(nq,K)).
!>

TAU

!>          TAU هو مصفوفة REAL، البعد (min(nq,K))
!>          يجب أن يحتوي TAU(i) على العامل القياسي للعكس الأولي H(i) أو G(i)
!>          الذي يحدد Q أو P، كما تم إرجاعه بواسطة SGEBRD في الوسيطTAUQ أو TAUP.
!>

C

!>          C هو مصفوفة REAL، البعد (LDC,N)
!>          عند الدخول، المصفوفة C ذات البعد M-by-N.
!>          عند الخروج، يتم استبدال C بـ Q*C أو Q**T*C أو C*Q**T أو C*Q
!>          أو P*C أو P**T*C أو C*P أو C*P**T.
!>

LDC

!>          LDC هو INTEGER
!>          البعد الرائد للمصفوفة C. LDC >= max(1,M).
!>

WORK

!>          WORK هو مصفوفة REAL، البعد (MAX(1,LWORK))
!>          عند الخروج، إذا كان INFO = 0، فإن WORK(1) ترجع LWORK المثلى.
!>

LWORK

!>          LWORK هو INTEGER
!>          بعد مصفوفة WORK.
!>          إذا كان SIDE = 'L'، فإن LWORK >= max(1,N)؛
!>          إذا كان SIDE = 'R'، فإن LWORK >= max(1,M).
!>          للحصول على أداء أمثل، LWORK >= N*NB إذا كان SIDE = 'L'، و
!>          LWORK >= M*NB إذا كان SIDE = 'R'، حيث NB هو حجم الكتلة الأمثل.
!>
!>          إذا كان LWORK = -1، فيفترض أن هناك استعلامًا عن مساحة العمل؛ تقوم الروتين الفرعي
!>          بحساب حجم مصفوفة WORK الأمثل فقط، وإرجاع
!>          هذه القيمة كالإدخال الأول في مصفوفة WORK، ولا يتم إصدار رسالة خطأ
!>          بخصوص LWORK بواسطة XERBLA.
!>

INFO

!>          INFO هو INTEGER
!>          = 0:  اكتمال ناجح
!>          < 0:  إذا كان INFO = -i، كان الوسيط i ذا قيمة غير قانونية
!>
