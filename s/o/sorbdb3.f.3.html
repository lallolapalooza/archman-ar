subroutine sorbdb3 (integer m, integer p, integer q, real, dimension(ldx11,*) x11, integer ldx11, real, dimension(ldx21,*) x21, integer ldx21, real, dimension(*) theta, real, dimension(*) phi, real, dimension(*) taup1, real, dimension(*) taup2, real, dimension(*) tauq1, real, dimension(*) work, integer lwork, integer info)

SORBDB3

الغرض:

!>
!> تقوم SORBDB3 بتقليص الكتل بشكل متزامن لمصفوفة X الطويلة والرفيعة بأعمدة متعامدة:
!>
!>                            [ B11 ]
!>      [ X11 ]   [ P1 |    ] [  0  ]
!>      [-----] = [---------] [-----] Q1**T .
!>      [ X21 ]   [    | P2 ] [ B21 ]
!>                            [  0  ]
!>
!> X11 بحجم P × Q، و X21 بحجم (M-P) × Q. يجب أن يكون M-P أصغر من أو يساوي P،
!> Q، أو M-Q. تتعامل الإجراءات SORBDB1، SORBDB2، و SORBDB4 مع الحالات التي
!> يكون فيها M-P ليس البعد الأدنى.
!>
!> المصفوفات المتعامدة P1، P2، و Q1 هي بحجم P × P، (M-P) × (M-P)،
!> و (M-Q) × (M-Q) على التوالي. يتم تمثيلها ضمنيًا بواسطة
!> متجهات هاوسهولدر.
!>
!> B11 و B12 هي مصفوفات ثنائية الأقطار بحجم (M-P) × (M-P) ممثلة
!> ضمنيًا بالزاويتين THETA و PHI.
!>
!>

المعلمات

M 
!>          M is INTEGER
!>           عدد صفوف X11 مضافًا إليه عدد صفوف X21.
!>

P

!>          P is INTEGER
!>           عدد صفوف X11. 0 <= P <= M. M-P <= min(P,Q,M-Q).
!>

Q

!>          Q is INTEGER
!>           عدد أعمدة X11 و X21. 0 <= Q <= M.
!>

X11

!>          X11 is REAL array, dimension (LDX11,Q)
!>           عند الإدخال، الكتلة العلوية من المصفوفة X المراد تقليصها. عند
!>           الخروج، تحدد أعمدة tril(X11) المعكوسات لـ P1
!>           وتحدد صفوف triu(X11,1) المعكوسات لـ Q1.
!>

LDX11

!>          LDX11 is INTEGER
!>           البعد الرئيسي لـ X11. LDX11 >= P.
!>

X21

!>          X21 is REAL array, dimension (LDX21,Q)
!>           عند الإدخال، الكتلة السفلية من المصفوفة X المراد تقليصها. عند
!>           الخروج، تحدد أعمدة tril(X21) المعكوسات لـ P2.
!>

LDX21

!>          LDX21 is INTEGER
!>           البعد الرئيسي لـ X21. LDX21 >= M-P.
!>

THETA

!>          THETA is REAL array, dimension (Q)
!>           تُعرّف مدخلات الكتل ثنائية الأقطار B11، B21 بواسطة
!>           THETA و PHI. انظر التفاصيل الإضافية.
!>

PHI

!>          PHI is REAL array, dimension (Q-1)
!>           تُعرّف مدخلات الكتل ثنائية الأقطار B11، B21 بواسطة
!>           THETA و PHI. انظر التفاصيل الإضافية.
!>

TAUP1

!>          TAUP1 is REAL array, dimension (P)
!>           العوامل القياسية للمتجهات الأساسية للمنعكسات التي تحدد
!>           P1.
!>

TAUP2

!>          TAUP2 is REAL array, dimension (M-P)
!>           العوامل القياسية للمتجهات الأساسية للمنعكسات التي تحدد
!>           P2.
!>

TAUQ1

!>          TAUQ1 is REAL array, dimension (Q)
!>           العوامل القياسية للمتجهات الأساسية للمنعكسات التي تحدد
!>           Q1.
!>

WORK

!>          WORK is REAL array, dimension (LWORK)
!>

LWORK

!>          LWORK is INTEGER
!>           بعد مصفوفة WORK. LWORK >= M-Q.
!>
!>           إذا كان LWORK = -1، فسيتم افتراض استعلام عن مساحة العمل؛
!>           تقوم الوحدة الفرعية بحساب حجم مساحة العمل المثلى فقط،
!>           وتعيد هذا الحجم كإدخال أول لمصفوفة WORK، ولا يصدر XERBLA
!>           أي رسالة خطأ تتعلق بـ LWORK.
!>

INFO

!>          INFO is INTEGER
!>           = 0:  انتهاء ناجح.
!>           < 0:  إذا كان INFO = -i، فإن المعلمة i كانت ذات قيمة غير
!>           قانونية.
!>

تفاصيل إضافية:

!>
!>  تُمثّل الكتل ثنائية الأقطار العلوية B11، B21 ضمنيًا بواسطة
!>  الزاويتين THETA(1)، ..., THETA(Q) و PHI(1)، ..., PHI(Q-1). كل مدخل
!>  في كل نطاق ثنائي الأقطار هو حاصل ضرب جيب أو جيب تمام لـ THETA
!>  مع جيب أو جيب تمام لـ PHI. انظر [1] أو SORCSD للتفاصيل.
!>
!>  تُمثّل P1، P2، و Q1 كحواصل ضرب لمتجهات المنعكسات الأولية.
!>  انظر SORCSD2BY1 لمعرفة تفاصيل إنشاء P1، P2، و Q1 باستخدام SORGQR
!>  و SORGLQ.
!>
