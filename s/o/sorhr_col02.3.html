subroutine sorhr_col02 (integer m, integer n, integer mb1, integer nb1, integer nb2, real, dimension(6) result)

SORHR_COL02

الغرض:

!>
!> SORHR_COL02 تختبر SORGTSQR_ROW و SORHR_COL داخل SGETSQRHRT
!> (التي تستدعي SLATSQR، SORGTSQR_ROW و SORHR_COL) باستخدام SGEMQRT.
!> لذلك، يجب اختبار SLATSQR (جزء من SGEQR) و SGEMQRT (جزء من SGEMQR)
!> قبل هذا الاختبار.
!>
!>

المعاملات

M 
!>          M هو عدد صحيح
!>          عدد الصفوف في مصفوفة الاختبار.
!>

N

!>          N هو عدد صحيح
!>          عدد الأعمدة في مصفوفة الاختبار.
!>

MB1

!>          MB1 هو عدد صحيح
!>          عدد الصفوف في كتلة الصف في مصفوفة اختبار الإدخال.
!>

NB1

!>          NB1 هو عدد صحيح
!>          عدد الأعمدة في كتلة العمود في مصفوفة اختبار الإدخال.
!>

NB2

!>          NB2 هو عدد صحيح
!>          عدد الأعمدة في كتلة العمود في مصفوفة الإخراج.
!>

RESULT

!>          RESULT هو مصفوفة حقيقية، بأبعاد (6)
!>          نتائج كل اختبار من الاختبارات الستة أدناه.
!>
!>            A هي مصفوفة اختبار إدخال m × n ليتم تحليلها.
!>            بحيث A = Q_gr * ( R )
!>                               ( 0 )،
!>
!>            Q_qr هي مصفوفة متعامدة ضمنية m × m، نتيجة
!>            التحليل في تمثيل WY المكتلي،
!>            مخزنة بتنسيق إخراج SGEQRT.
!>
!>            R هي مصفوفة علوية مثلثة n × n،
!>
!>            0 هي مصفوفة صفرية (m-n) × n،
!>
!>            Q هي مصفوفة متعامدة صريحة m × m Q = Q_gr * I
!>
!>            C هي مصفوفة عشوائية m × n،
!>
!>            D هي مصفوفة عشوائية n × m.
!>
!>          الاختبارات الستة هي:
!>
!>          RESULT(1) = |R - (Q**H) * A| / ( eps * m * |A| )
!>            يعادل الاختبار لـ | A - Q * R | / (eps * m * |A|)،
!>
!>          RESULT(2) = |I - (Q**H) * Q| / ( eps * m )،
!>
!>          RESULT(3) = | Q_qr * C - Q * C | / (eps * m * |C|)،
!>
!>          RESULT(4) = | (Q_gr**H) * C - (Q**H) * C | / (eps * m * |C|)
!>
!>          RESULT(5) = | D * Q_qr - D * Q | / (eps * m * |D|)
!>
!>          RESULT(6) = | D * (Q_qr**H) - D * (Q**H) | / (eps * m * |D|)،
!>
!>          حيث:
!>            Q_qr * C، (Q_gr**H) * C، D * Q_qr، D * (Q_qr**H) هي
!>            محسوبة باستخدام SGEMQRT،
!>
!>            Q * C، (Q**H) * C، D * Q، D * (Q**H) هي
!>            محسوبة باستخدام SGEMM.
!>
