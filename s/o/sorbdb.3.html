subroutine sorbdb (character trans, character signs, integer m, integer p, integer q, real, dimension( ldx11, * ) x11, integer ldx11, real, dimension( ldx12, * ) x12, integer ldx12, real, dimension( ldx21, * ) x21, integer ldx21, real, dimension( ldx22, * ) x22, integer ldx22, real, dimension( * ) theta, real, dimension( * ) phi, real, dimension( * ) taup1, real, dimension( * ) taup2, real, dimension( * ) tauq1, real, dimension( * ) tauq2, real, dimension( * ) work, integer lwork, integer info)

SORBDB

الغرض:

!>
!> يقوم SORBDB بتنقيح كتل المصفوفة المتعامدة M-by-M X بشكل متزامن:
!>
!>                                 [ B11 | B12 0  0 ]
!>     [ X11 | X12 ]   [ P1 |    ] [  0  |  0 -I  0 ] [ Q1 |    ]**T
!> X = [-----------] = [---------] [----------------] [---------]   .
!>     [ X21 | X22 ]   [    | P2 ] [ B21 | B22 0  0 ] [    | Q2 ]
!>                                 [  0  |  0  0  I ]
!>
!> X11 هو P-by-Q. يجب ألا يكون Q أكبر من P أو M-P أو M-Q. (إذا لم يكن
!> هذا هو الحال، فيجب تبديل و/أو إعادة ترتيب X. يمكن القيام بذلك
!> في وقت ثابت باستخدام خيارات TRANS و SIGNS. راجع SORCSD
!> للتفاصيل.)
!>
!> المصفوفات المتعامدة P1 و P2 و Q1 و Q2 هي P-by-P و (M-P)-by-
!> (M-P) و Q-by-Q و (M-Q)-by-(M-Q) على التوالي. يتم
!> تمثيلها ضمنيًا بواسطة متجهات هاوسهولدر.
!>
!> B11 و B12 و B21 و B22 هي مصفوفات ثنائية القطر ممثلة
!> ضمنيًا بالزوايا THETA و PHI.
!>

المعلمات

TRANS 
!>          TRANS هو CHARACTER
!>          = 'T':      يتم تخزين X و U1 و U2 و V1T و V2T بترتيب الصفوف؛
!>          غير ذلك:   يتم تخزين X و U1 و U2 و V1T و V2T بترتيب الأعمدة.
!>

SIGNS

!>          SIGNS هو CHARACTER
!>          = 'O':      تصبح الكتلة السفلية اليسرى غير موجبة (الافتراض)؛
!>          غير ذلك:   تصبح الكتلة العلوية اليمنى غير موجبة (الافتراض).
!>

M

!>          M هو INTEGER
!>          عدد الصفوف والأعمدة في X.
!>

P

!>          P هو INTEGER
!>          عدد الصفوف في X11 و X12. 0 <= P <= M.
!>

Q

!>          Q هو INTEGER
!>          عدد الأعمدة في X11 و X21. 0 <= Q <=
!>          MIN(P,M-P,M-Q).
!>

X11

!>          X11 هو مصفوفة REAL، البعد (LDX11,Q)
!>          عند الدخول، الكتلة العلوية اليسرى للمصفوفة المتعامدة المراد
!>          تخفيضها. عند الخروج، يعتمد الشكل على TRANS:
!>          إذا كان TRANS = 'N'، فإن:
!>             أعمدة tril(X11) تحدد المعكسات لـ P1،
!>             صفوف triu(X11,1) تحدد المعكسات لـ Q1؛
!>          وإلا فإن TRANS = 'T'، و:
!>             صفوف triu(X11) تحدد المعكسات لـ P1،
!>             أعمدة tril(X11,-1) تحدد المعكسات لـ Q1.
!>

LDX11

!>          LDX11 هو INTEGER
!>          البعد الرئيسي لـ X11. إذا كان TRANS = 'N'، فإن LDX11 >=
!>          P؛ وإلا فإن LDX11 >= Q.
!>

X12

!>          X12 هو مصفوفة REAL، البعد (LDX12,M-Q)
!>          عند الدخول، الكتلة العلوية اليمنى للمصفوفة المتعامدة المراد
!>          تخفيضها. عند الخروج، يعتمد الشكل على TRANS:
!>          إذا كان TRANS = 'N'، فإن:
!>             صفوف triu(X12) تحدد المعكسات الأولى لـ Q2؛
!>          وإلا فإن TRANS = 'T'، و:
!>             أعمدة tril(X12) تحدد المعكسات الأولى لـ Q2.
!>

LDX12

!>          LDX12 هو INTEGER
!>          البعد الرئيسي لـ X12. إذا كان TRANS = 'N'، فإن LDX12 >=
!>          P؛ وإلا فإن LDX11 >= M-Q.
!>

X21

!>          X21 هو مصفوفة REAL، البعد (LDX21,Q)
!>          عند الدخول، الكتلة السفلية اليسرى للمصفوفة المتعامدة المراد
!>          تخفيضها. عند الخروج، يعتمد الشكل على TRANS:
!>          إذا كان TRANS = 'N'، فإن:
!>             أعمدة tril(X21) تحدد المعكسات لـ P2؛
!>          وإلا فإن TRANS = 'T'، و:
!>             صفوف triu(X21) تحدد المعكسات لـ P2.
!>

LDX21

!>          LDX21 هو INTEGER
!>          البعد الرئيسي لـ X21. إذا كان TRANS = 'N'، فإن LDX21 >=
!>          M-P؛ وإلا فإن LDX21 >= Q.
!>

X22

!>          X22 هو مصفوفة REAL، البعد (LDX22,M-Q)
!>          عند الدخول، الكتلة السفلية اليمنى للمصفوفة المتعامدة المراد
!>          تخفيضها. عند الخروج، يعتمد الشكل على TRANS:
!>          إذا كان TRANS = 'N'، فإن:
!>             صفوف triu(X22(Q+1:M-P,P+1:M-Q)) تحدد المعكسات الأخيرة
!>             لـ Q2 (M-P-Q)،
!>          وإلا فإن TRANS = 'T'، و:
!>             أعمدة tril(X22(P+1:M-Q,Q+1:M-P)) تحدد المعكسات الأخيرة
!>             لـ P2 (M-P-Q).
!>

LDX22

!>          LDX22 هو INTEGER
!>          البعد الرئيسي لـ X22. إذا كان TRANS = 'N'، فإن LDX22 >=
!>          M-P؛ وإلا فإن LDX22 >= M-Q.
!>

THETA

!>          THETA هي مصفوفة REAL، البعد (Q)
!>          يمكن حساب مدخلات الكتل ثنائية القطر B11 و B12 و B21 و B22
!>          من الزوايا THETA و PHI. انظر التفاصيل الإضافية.
!>

PHI

!>          PHI هي مصفوفة REAL، البعد (Q-1)
!>          يمكن حساب مدخلات الكتل ثنائية القطر B11 و B12 و B21 و B22
!>          من الزوايا THETA و PHI. انظر التفاصيل الإضافية.
!>

TAUP1

!>          TAUP1 هي مصفوفة REAL، البعد (P)
!>          العوامل القياسية للمعكسات الأولية التي تحدد P1.
!>

TAUP2

!>          TAUP2 هي مصفوفة REAL، البعد (M-P)
!>          العوامل القياسية للمعكسات الأولية التي تحدد P2.
!>

TAUQ1

!>          TAUQ1 هي مصفوفة REAL، البعد (Q)
!>          العوامل القياسية للمعكسات الأولية التي تحدد Q1.
!>

TAUQ2

!>          TAUQ2 هي مصفوفة REAL، البعد (M-Q)
!>          العوامل القياسية للمعكسات الأولية التي تحدد Q2.
!>

WORK

!>          WORK هي مصفوفة REAL، البعد (LWORK)
!>

LWORK

!>          LWORK هو INTEGER
!>          بعد المصفوفة WORK. LWORK >= M-Q.
!>
!>          إذا كان LWORK = -1، فسيتم افتراض استعلام عن مساحة العمل؛
!>          يقوم الإجراء الفرعي فقط بحساب الحجم الأمثل لمصفوفة WORK،
!>          ويعيد هذا الحجم كمدخل أول لمصفوفة WORK، ولا يصدر أي رسالة
!>          خطأ متعلقة بـ LWORK بواسطة XERBLA.
!>

INFO

!>          INFO هو INTEGER
!>          = 0:  انتهاء ناجح.
!>          < 0:  إذا كان INFO = -i، فإن المعلمة i كانت لها قيمة غير قانونية.
!>
