توثيق الوظيفة/الروتين الفرعي

subroutine sormr3 (character side, character trans, integer m, integer n, integer k, integer l, real, dimension( lda, * ) a, integer lda, real, dimension( * ) tau, real, dimension( ldc, * ) c, integer ldc, real, dimension( * ) work, integer info)

SORMR3 تضرب مصفوفة عامة بواسطة المصفوفة المتعامدة من تحليل RZ الذي تم تحديده بواسطة stzrzf (خوارزمية غير محظورة).

الغرض:

!>
!> SORMR3 يستبدل المصفوفة C الحقيقية العامة m في n بـ
!>
!>       Q * C  إذا كان SIDE = 'L' و TRANS = 'N'، أو
!>
!>       Q**T* C  إذا كان SIDE = 'L' و TRANS = 'C'، أو
!>
!>       C * Q  إذا كان SIDE = 'R' و TRANS = 'N'، أو
!>
!>       C * Q**T إذا كان SIDE = 'R' و TRANS = 'C'،
!>
!> حيث Q هي مصفوفة حقيقية متعامدة معرفة كحاصل ضرب k
!> عواكس أولية
!>
!>       Q = H(1) H(2) . . . H(k)
!>
!> كما تم إرجاعها بواسطة STZRZF. Q من الرتبة m إذا كان SIDE = 'L' ومن الرتبة n
!> إذا كان SIDE = 'R'.
!>

المعلمات

SIDE 
!>          SIDE هو CHARACTER*1
!>          = 'L': تطبيق Q أو Q**T من اليسار
!>          = 'R': تطبيق Q أو Q**T من اليمين
!>

TRANS

!>          TRANS هو CHARACTER*1
!>          = 'N': تطبيق Q  (بدون تبديل)
!>          = 'T': تطبيق Q**T (تبديل)
!>

M

!>          M هو INTEGER
!>          عدد صفوف المصفوفة C. M >= 0.
!>

N

!>          N هو INTEGER
!>          عدد أعمدة المصفوفة C. N >= 0.
!>

K

!>          K هو INTEGER
!>          عدد العواكس الأولية التي يحدد حاصل ضربها
!>          المصفوفة Q.
!>          إذا كان SIDE = 'L'، فإن M >= K >= 0؛
!>          إذا كان SIDE = 'R'، فإن N >= K >= 0.
!>

L

!>          L هو INTEGER
!>          عدد أعمدة المصفوفة A التي تحتوي على
!>          الجزء الهادف من عواكس هاوسهولدر.
!>          إذا كان SIDE = 'L'، فإن M >= L >= 0؛ إذا كان SIDE = 'R'، فإن N >= L >= 0.
!>

A

!>          A هي مصفوفة REAL، الأبعاد
!>                               (LDA,M) إذا كان SIDE = 'L'،
!>                               (LDA,N) إذا كان SIDE = 'R'
!>          يجب أن يحتوي الصف i على المتجه الذي يحدد
!>          العاكس الأولي H(i)، من أجل i = 1,2,...,k، كما تم إرجاعه بواسطة
!>          STZRZF في الصفوف k الأخيرة من وسيط المصفوفة A.
!>          يتم تعديل A بواسطة الروتين ولكن يتم استعادتها عند الخروج.
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرئيسي للمصفوفة A. LDA >= max(1,K).
!>

TAU

!>          TAU هي مصفوفة REAL، الأبعاد (K)
!>          يجب أن يحتوي TAU(i) على العامل القياسي
!>          لل Becauseر الأولي H(i)، كما تم إرجاعه بواسطة STZRZF.
!>

C

!>          C هي مصفوفة REAL، الأبعاد (LDC,N)
!>          عند الدخول، المصفوفة C ذات الأبعاد m في n.
!>          عند الخروج، يتم استبدال C بـ Q*C أو Q**T*C أو C*Q**T أو C*Q.
!>

LDC

!>          LDC هو INTEGER
!>          البعد الرئيسي للمصفوفة C. LDC >= max(1,M).
!>

WORK

!>          WORK هي مصفوفة REAL، الأبعاد
!>                                   (N) إذا كان SIDE = 'L'،
!>                                   (M) إذا كان SIDE = 'R'
!>

INFO

!>          INFO هو INTEGER
!>          = 0: خروج ناجح
!>          < 0: إذا كان INFO = -i، فإن الوسيط i كان له قيمة غير قانونية
!>
