توثيق الوظيفة/الروتين الفرعي

subroutine sormtr (character side, character uplo, character trans, integer m, integer n, real, dimension( lda, * ) a, integer lda, real, dimension( * ) tau, real, dimension( ldc, * ) c, integer ldc, real, dimension( * ) work, integer lwork, integer info)

SORMTR

الغرض:

!>
!> SORMTR يستبدل المصفوفة الحقيقية M-by-N المسماة C بـ
!>
!>                 SIDE = 'L'     SIDE = 'R'
!> TRANS = 'N':      Q * C          C * Q
!> TRANS = 'T':      Q**T * C       C * Q**T
!>
!> حيث Q هي مصفوفة عمودية حقيقية من الرتبة nq، مع nq = m إذا
!> SIDE = 'L' و nq = n إذا SIDE = 'R'. تُعرف Q كناتج
!> لمُعرفات أساسية عددها nq-1، كما يتم إرجاعها بواسطة SSYTRD:
!>
!> إذا كان UPLO = 'U'، فإن Q = H(nq-1) . . . H(2) H(1);
!>
!> إذا كان UPLO = 'L'، فإن Q = H(1) H(2) . . . H(nq-1).
!>

المعلمات

SIDE 
!>          SIDE هو CHARACTER*1
!>          = 'L': تطبيق Q أو Q**T من اليسار؛
!>          = 'R': تطبيق Q أو Q**T من اليمين.
!>

UPLO

!>          UPLO هو CHARACTER*1
!>          = 'U': المثلث العلوي لـ A يحتوي على مُعرفات أساسية
!>                 من SSYTRD؛
!>          = 'L': المثلث السفلي لـ A يحتوي على مُعرفات أساسية
!>                 من SSYTRD.
!>

TRANS

!>          TRANS هو CHARACTER*1
!>          = 'N':  لا يوجد تبديل، تطبيق Q؛
!>          = 'T':  تبديل، تطبيق Q**T.
!>

M

!>          M هو INTEGER
!>          عدد صفوف المصفوفة C. M >= 0.
!>

N

!>          N هو INTEGER
!>          عدد أعمدة المصفوفة C. N >= 0.
!>

A

!>          A هو مصفوفة REAL، ذات أبعاد
!>                               (LDA,M) إذا كان SIDE = 'L'
!>                               (LDA,N) إذا كان SIDE = 'R'
!>          المتجهات التي تُعرف المُعرفات الأساسية، كما
!>          تم إرجاعها بواسطة SSYTRD.
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرائد للمصفوفة A.
!>          LDA >= max(1,M) إذا كان SIDE = 'L'؛ LDA >= max(1,N) إذا كان SIDE = 'R'.
!>

TAU

!>          TAU هو مصفوفة REAL، ذات أبعاد
!>                               (M-1) إذا كان SIDE = 'L'
!>                               (N-1) إذا كان SIDE = 'R'
!>          يجب أن تحتوي TAU(i) على العامل القياسي للمُعرف
!>          الأساسي H(i)، كما تم إرجاعه بواسطة SSYTRD.
!>

C

!>          C هو مصفوفة REAL، ذات أبعاد (LDC,N)
!>          عند الإدخال، المصفوفة C ذات الأبعاد M-by-N.
!>          عند الخروج، يتم استبدال C بـ Q*C أو Q**T*C أو C*Q**T أو C*Q.
!>

LDC

!>          LDC هو INTEGER
!>          البعد الرائد للمصفوفة C. LDC >= max(1,M).
!>

WORK

!>          WORK هي مصفوفة REAL، ذات أبعاد (MAX(1,LWORK))
!>          عند الخروج، إذا كان INFO = 0، فإن WORK(1) تُرجع LWORK المثلى.
!>

LWORK

!>          LWORK هو INTEGER
!>          بعد المصفوفة WORK.
!>          إذا كان SIDE = 'L'، فإن LWORK >= max(1,N)؛
!>          إذا كان SIDE = 'R'، فإن LWORK >= max(1,M).
!>          للحصول على أفضل أداء، LWORK >= N*NB إذا كان SIDE = 'L'، و
!>          LWORK >= M*NB إذا كان SIDE = 'R'، حيث NB هو حجم الكتلة الأمثل.
!>
!>          إذا كان LWORK = -1، فسيتم افتراض استعلام عن مساحة العمل؛
!>          يُحسب الروتين فقط الحجم الأمثل للمصفوفة WORK، ويُرجع
!!>          هذه القيمة كأول إدخال للمصفوفة WORK، ولا يتم إصدار أي رسالة خطأ
!>          متعلقة بـ LWORK بواسطة XERBLA.
!>

INFO

!>          INFO هو INTEGER
!>          = 0:  خروج ناجح
!>          < 0:  إذا كان INFO = -i، فإن الوسيط i كان له قيمة غير قانونية
!>
