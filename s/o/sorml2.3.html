subroutine sorml2 (character side, character trans, integer m, integer n, integer k, real, dimension( lda, * ) a, integer lda, real, dimension( * ) tau, real, dimension( ldc, * ) c, integer ldc, real, dimension( * ) work, integer info)

SORML2 تضرب مصفوفة عامة بمصفوفة متعامدة من تحليل LQ تحدده sgelqf (خوارزمية غير محظورة).

الغرض:

!>
!> SORML2 تستبدل المصفوفة العامة الحقيقية m × n المسماة C بـ
!>
!>       Q * C  إذا كان SIDE = 'L' و TRANS = 'N'، أو
!>
!>       Q**T* C  إذا كان SIDE = 'L' و TRANS = 'T'، أو
!>
!>       C * Q  إذا كان SIDE = 'R' و TRANS = 'N'، أو
!>
!>       C * Q**T إذا كان SIDE = 'R' و TRANS = 'T'،
!>
!> حيث Q هي مصفوفة متعامدة حقيقية معرفة كناتج لـ k
!> موصلات أساسية
!>
!>       Q = H(k) . . . H(2) H(1)
!>
!> كما تم إرجاعها بواسطة SGELQF. Q هي من الرتبة m إذا كان SIDE = 'L' ومن الرتبة n
!> إذا كان SIDE = 'R'.
!>

المعلمات

SIDE 
!>          SIDE هو CHARACTER*1
!>          = 'L': تطبيق Q أو Q**T من اليسار
!>          = 'R': تطبيق Q أو Q**T من اليمين
!>

TRANS

!>          TRANS هو CHARACTER*1
!>          = 'N': تطبيق Q  (بدون تبديل)
!>          = 'T': تطبيق Q**T (تبديل)
!>

M

!>          M هو INTEGER
!>          عدد صفوف المصفوفة C. M >= 0.
!>

N

!>          N هو INTEGER
!>          عدد أعمدة المصفوفة C. N >= 0.
!>

K

!>          K هو INTEGER
!>          عدد الموصلات الأساسية التي يحدد ناتجها
!>          المصفوفة Q.
!>          إذا كان SIDE = 'L'، فإن M >= K >= 0؛
!>          إذا كان SIDE = 'R'، فإن N >= K >= 0.
!>

A

!>          A هي مصفوفة REAL، ذات بُعد
!>                               (LDA,M) إذا كان SIDE = 'L'،
!>                               (LDA,N) إذا كان SIDE = 'R'
!>          يجب أن يحتوي الصف i على المتجه الذي يحدد
!>          الموصل الأساسي H(i)، لـ i = 1,2,...,k، كما تم إرجاعه بواسطة
!>          SGELQF في الصفوف k الأولى من مصفوفة الوسيط A.
!>          يتم تعديل A بواسطة الروتين ولكن يتم استعادتها عند الخروج.
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرئيسي للمصفوفة A. LDA >= max(1,K).
!>

TAU

!>          TAU هي مصفوفة REAL، ذات بُعد (K)
!>          يجب أن يحتوي TAU(i) على المعامل القياسي للموصل
!>          الأساسي H(i)، كما تم إرجاعه بواسطة SGELQF.
!>

C

!>          C هي مصفوفة REAL، ذات بُعد (LDC,N)
!>          عند الإدخال، مصفوفة C بحجم m × n.
!>          عند الخروج، يتم استبدال C بـ Q*C أو Q**T*C أو C*Q**T أو C*Q.
!>

LDC

!>          LDC هو INTEGER
!>          البعد الرئيسي للمصفوفة C. LDC >= max(1,M).
!>

WORK

!>          WORK هي مصفوفة REAL، ذات بُعد
!>                                   (N) إذا كان SIDE = 'L'،
!>                                   (M) إذا كان SIDE = 'R'
!>

INFO

!>          INFO هو INTEGER
!>          = 0: خروج ناجح
!>          < 0: إذا كان INFO = -i، فإن الوسيط i كان له قيمة غير قانونية
!>
