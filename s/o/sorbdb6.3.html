توثيق الوظيفة/الروتين الفرعي

subroutine sorbdb6 (integer m1, integer m2, integer n, real, dimension(*) x1, integer incx1, real, dimension(*) x2, integer incx2, real, dimension(ldq1,*) q1, integer ldq1, real, dimension(ldq2,*) q2, integer ldq2, real, dimension(*) work, integer lwork, integer info)

SORBDB6

Purpose:

!>
!> SORBDB6 يقوم بتعمود المتجه
!>      X = [ X1 ]
!>          [ X2 ]
!> بالنسبة للأعمدة من
!>      Q = [ Q1 ] .
!>          [ Q2 ]
!> يجب أن تكون أعمدة Q متعامدة. سيكون المتجه المتعامد
!> صفرًا فقط إذا كان يقع بالكامل ضمن نطاق Q.
!>
!> يتم حساب الإسقاط بما لا يزيد عن تكرارين لخوارزمية Gram-Schmidt
!> الكلاسيكية، انظر
!> * L. Giraud, J. Langou, M. Rozložník. 
!>   2002. CERFACS Technical Report No. TR/PA/02/33. URL:
!>   https://www.cerfacs.fr/algor/reports/2002/TR_PA_02_33.pdf
!>
!>

Parameters

M1 
!>          M1 is INTEGER
!>           بعد X1 وعدد الصفوف في Q1. 0 <= M1.
!>

M2

!>          M2 is INTEGER
!>           بعد X2 وعدد الصفوف في Q2. 0 <= M2.
!>

N

!>          N is INTEGER
!>           عدد الأعمدة في Q1 و Q2. 0 <= N.
!>

X1

!>          X1 is REAL array, dimension (M1)
!>           عند الإدخال، الجزء العلوي من المتجه المراد تعموده.
!>           عند الخروج، الجزء العلوي من المتجه المسقط.
!>

INCX1

!>          INCX1 is INTEGER
!>           زيادة لإدخالات X1.
!>

X2

!>          X2 is REAL array, dimension (M2)
!>           عند الإدخال، الجزء السفلي من المتجه المراد
!>           تعموده. عند الخروج، الجزء السفلي من المتجه المسقط.
!>

INCX2

!>          INCX2 is INTEGER
!>           زيادة لإدخالات X2.
!>

Q1

!>          Q1 is REAL array, dimension (LDQ1, N)
!>           الجزء العلوي من مصفوفة الأساس المتعامد.
!>

LDQ1

!>          LDQ1 is INTEGER
!>           البعد الرئيسي لـ Q1. LDQ1 >= M1.
!>

Q2

!>          Q2 is REAL array, dimension (LDQ2, N)
!>           الجزء السفلي من مصفوفة الأساس المتعامد.
!>

LDQ2

!>          LDQ2 is INTEGER
!>           البعد الرئيسي لـ Q2. LDQ2 >= M2.
!>

WORK

!>          WORK is REAL array, dimension (LWORK)
!>

LWORK

!>          LWORK is INTEGER
!>           بعد المصفوفة WORK. LWORK >= N.
!>

INFO

!>          INFO is INTEGER
!>           = 0:  خروج ناجح.
!>           < 0:  إذا كان INFO = -i، فإن الوسيط i كان له قيمة غير قانونية.
!>
