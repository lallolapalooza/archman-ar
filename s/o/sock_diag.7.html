sock_diag(7) دليل المعلومات المتنوعة sock_diag(7)

الاسم

sock_diag - الحصول على معلومات حول المقابس (sockets)

ملخص

#include <sys/socket.h>
#include <linux/sock_diag.h>
#include <linux/unix_diag.h> /* لمقابس نطاق UNIX */
#include <linux/inet_diag.h> /* لمقابس IPv4 و IPv6 */
diag_socket = socket(AF_NETLINK, socket_type, NETLINK_SOCK_DIAG);

الوصف

توفر مجموعة netlink الفرعية sock_diag آلية للحصول على معلومات حول المقابس من عائلات عناوين مختلفة من النواة. يمكن استخدام هذه المجموعة الفرعية للحصول على معلومات حول مقابس فردية أو طلب قائمة بالمقابس.

في الطلب، يمكن للمتصل تحديد معلومات إضافية يرغب في الحصول عليها حول المقبس، على سبيل المثال، معلومات الذاكرة أو معلومات خاصة بعائلة العنوان.

عند طلب قائمة بالمقابس، يمكن للمتصل تحديد عوامل تصفية سيتم تطبيقها بواسطة النواة لتحديد مجموعة فرعية من المقابس للإبلاغ عنها. في الوقت الحالي، هناك فقط القدرة على تصفية المقابس حسب الحالة (متصل، استماع، وما إلى ذلك).

لاحظ أن sock_diag يبلغ فقط عن المقابس التي لها اسم؛ أي، إما المقابس المرتبطة صراحةً باستخدام bind(2) أو المقابس التي تم ربطها تلقائيًا بعنوان (على سبيل المثال، بواسطة connect(2)). هذه هي نفس مجموعة المقابس المتاحة عبر /proc/net/unix و /proc/net/tcp و /proc/net/udp، وما إلى ذلك.

الطلب

يبدأ الطلب برأس struct nlmsghdr موصوف في netlink(7) مع حقل nlmsg_type معين إلى SOCK_DIAG_BY_FAMILY. يتبعه رأس خاص بعائلة العنوان يبدأ بجزء مشترك تشترك فيه جميع عائلات العناوين:


struct sock_diag_req {
    __u8 sdiag_family;
    __u8 sdiag_protocol;
};

الحقول في هذا الهيكل هي كما يلي:

sdiag_family
عائلة عنوان. يجب تعيينها إلى الثابت AF_* المناسب.
sdiag_protocol
يعتمد على sdiag_family. يجب تعيينه إلى الثابت IPPROTO_* المناسب لـ AF_INET و AF_INET6، وإلى 0 بخلاف ذلك.

إذا كان حقل nlmsg_flags في رأس struct nlmsghdr مضبوطًا بالعلم NLM_F_DUMP، فهذا يعني أنه يتم طلب قائمة بالمقابس؛ بخلاف ذلك، فهو استعلام عن مقبس فردي.

الاستجابة

تبدأ الاستجابة برأس struct nlmsghdr ويتبعها مصفوفة من الكائنات الخاصة بعائلة العنوان. يتم الوصول إلى المصفوفة باستخدام وحدات الماكرو NLMSG_* القياسية من واجهة برمجة تطبيقات netlink(3).

كل كائن هو قائمة NLA (سمات netlink) التي يتم الوصول إليها باستخدام وحدات الماكرو RTA_* من واجهة برمجة تطبيقات rtnetlink(3).

مقابس نطاق UNIX

بالنسبة لمقابس نطاق UNIX، يتم تمثيل الطلب في الهيكل التالي:


struct unix_diag_req {
    __u8    sdiag_family;
    __u8    sdiag_protocol;
    __u16   pad;
    __u32   udiag_states;
    __u32   udiag_ino;
    __u32   udiag_show;
    __u32   udiag_cookie[2];
};

الحقول في هذا الهيكل هي كما يلي:

sdiag_family
عائلة العنوان؛ يجب تعيينها إلى AF_UNIX.

sdiag_protocol

pad
يجب تعيين هذه الحقول إلى 0.
udiag_states
هذا قناع بت يحدد عامل تصفية لحالات المقابس. سيتم الإبلاغ فقط عن تلك المقابس التي تقع حالاتها في هذا القناع. القيم المدعومة هي:
1 << TCP_ESTABLISHED

1 << TCP_LISTEN

udiag_ino
هذا هو رقم inode عند الاستعلام عن مقبس فردي. يتم تجاهله عند الاستعلام عن قائمة المقابس.
udiag_show
هذه مجموعة من الأعلام تحدد نوع المعلومات التي يجب الإبلاغ عنها. يتم الإبلاغ عن كل نوع مطلوب من المعلومات مرة أخرى كسمة netlink كما هو موضح أدناه:
UDIAG_SHOW_NAME
السمة التي تم الإبلاغ عنها استجابةً لهذا الطلب هي UNIX_DIAG_NAME. الحمولة المرتبطة بهذه السمة هي المسار الذي تم ربط المقبس به (تسلسل بايت يصل حجمه إلى UNIX_PATH_MAX).
UDIAG_SHOW_VFS
السمة التي تم الإبلاغ عنها استجابةً لهذا الطلب هي UNIX_DIAG_VFS. يتم تمثيل الحمولة المرتبطة بهذه السمة في الهيكل التالي:

struct unix_diag_vfs {
    __u32 udiag_vfs_dev;
    __u32 udiag_vfs_ino;
};

الحقول في هذا الهيكل هي كما يلي:
udiag_vfs_dev
رقم جهاز inode المقبس المقابل على القرص.
udiag_vfs_ino
رقم inode للمقبس المقابل على القرص.
UDIAG_SHOW_PEER
السمة التي تم الإبلاغ عنها استجابةً لهذا الطلب هي UNIX_DIAG_PEER. الحمولة المرتبطة بهذه السمة هي قيمة __u32 وهي رقم inode للطرف المقابل. يتم الإبلاغ عن هذه السمة للمقابس المتصلة فقط.
UDIAG_SHOW_ICONS
السمة التي تم الإبلاغ عنها استجابةً لهذا الطلب هي UNIX_DIAG_ICONS. الحمولة المرتبطة بهذه السمة هي مصفوفة من قيم __u32 وهي أرقام inodes للمقابس التي مرت عبر استدعاء connect(2)، ولكن لم تتم معالجتها بواسطة accept(2) بعد. يتم الإبلاغ عن هذه السمة لمقابس الاستماع فقط.
UDIAG_SHOW_RQLEN
السمة التي تم الإبلاغ عنها استجابةً لهذا الطلب هي UNIX_DIAG_RQLEN. يتم تمثيل الحمولة المرتبطة بهذه السمة في الهيكل التالي:

struct unix_diag_rqlen {
    __u32 udiag_rqueue;
    __u32 udiag_wqueue;
};

الحقول في هذا الهيكل هي كما يلي:
udiag_rqueue
لمقابس الاستماع: عدد الاتصالات المعلقة. حجم المصفوفة المرتبطة بسمة UNIX_DIAG_ICONS هو مساوٍ لهذه القيمة.
للمقابس الراسخة: كمية البيانات في قائمة الانتظار الواردة.
udiag_wqueue
لمقابس الاستماع: حجم قائمة الانتظار الخلفية الذي يساوي القيمة المارة كوسيط ثانٍ لـ listen(2).
للمقابس الراسخة: مقدار الذاكرة المتاحة للإرسال.
UDIAG_SHOW_MEMINFO
السمة التي تم الإبلاغ عنها استجابةً لهذا الطلب هي UNIX_DIAG_MEMINFO. الحمولة المرتبطة بهذه السمة هي مصفوفة من قيم __u32 الموضحة أدناه في القسم الفرعي "معلومات ذاكرة المقبس".

NAME

sock_diag - socket diagnotic information interface

SYNOPSIS

#include <linux/netlink.h>
#include <linux/sock_diag.h>

DESCRIPTION

The NETLINK_SOCK_DIAG socket family provides a mechanism for dumping information about sockets on the local machine. It is intended to replace the older NETLINK_INET_DIAG interface.

Currently, the following address families are supported:

  • AF_UNIX (UNIX domain sockets)
  • AF_INET (IPv4 sockets)
  • AF_INET6 (IPv6 sockets)

The NETLINK_SOCK_DIAG socket is used with commands specified by the SOCK_DIAG_BY_FAMILY message type.

A request to dump socket information is built using the struct nlmsghdr and either struct unix_diag_req (for AF_UNIX) or struct inet_diag_req (for AF_INET and AF_INET6).

The dump request is sent using the sendmsg(2) system call to a netlink socket opened with the NETLINK_SOCK_DIAG protocol.

The dump response consists of a sequence of netlink messages. Each message starts with a struct nlmsghdr followed by a protocol-specific diagnostic message structure and optional attributes.

For AF_UNIX sockets, the response message starts with struct unix_diag_msg. The following attributes describe the socket:

UNIX_DIAG_NAME
The path name of the socket.
UNIX_DIAG_PEER
The inode number of the peer socket.
UNIX_DIAG_UID
The user ID of the socket owner.
UNIX_DIAG_VFS
VFS information about the socket.

For AF_INET and AF_INET6 sockets, the response message starts with struct inet_diag_msg. The following attributes describe the socket:

INET_DIAG_REQ
Deprecated. Use INET_DIAG_SKMEMINFO.
INET_DIAG_SKMEMINFO
Socket memory information. The payload is a struct sk_buff_info which contains the following fields:
SK_MEMINFO_WMEM_ALLOC
The amount of data in send queue.
SK_MEMINFO_SNDBUF
The send socket buffer as set by SO_SNDBUF.
SK_MEMINFO_FWD_ALLOC
The amount of memory scheduled for future use (TCP only).
SK_MEMINFO_WMEM_QUEUED
The amount of data queued by TCP, but not yet sent.
SK_MEMINFO_OPTMEM
The amount of memory allocated for the socket's service needs (e.g., socket filter).
SK_MEMINFO_BACKLOG
The amount of packets in the backlog (not yet processed).

VERSIONS

NETLINK_INET_DIAG was introduced in Linux 2.6.14 and supported AF_INET and AF_INET6 sockets only. In Linux 3.3, it was renamed to NETLINK_SOCK_DIAG and extended to support AF_UNIX sockets.

UNIX_DIAG_MEMINFO and INET_DIAG_SKMEMINFO were introduced in Linux 3.6.

STANDARDS

Linux.

EXAMPLES

The following example program prints inode number, peer's inode number, and name of all UNIX domain sockets in the current namespace.

#include <errno.h>
#include <stdio.h>
#include <string.h>
#include <unistd.h>
#include <sys/socket.h>
#include <sys/un.h>
#include <linux/netlink.h>
#include <linux/rtnetlink.h>
#include <linux/sock_diag.h>
#include <linux/unix_diag.h>
static int
send_query(int fd)
{
    struct sockaddr_nl nladdr = {
        .nl_family = AF_NETLINK
    };
    struct
    {
        struct nlmsghdr nlh;
        struct unix_diag_req udr;
    } req = {
        .nlh = {
            .nlmsg_len = sizeof(req),
            .nlmsg_type = SOCK_DIAG_BY_FAMILY,
            .nlmsg_flags = NLM_F_REQUEST | NLM_F_DUMP
        },
        .udr = {
            .sdiag_family = AF_UNIX,
            .udiag_states = -1,
            .udiag_show = UDIAG_SHOW_NAME | UDIAG_SHOW_PEER
        }
    };
    struct iovec iov = {
        .iov_base = &req,
        .iov_len = sizeof(req)
    };
    struct msghdr msg = {
        .msg_name = &nladdr,
        .msg_namelen = sizeof(nladdr),
        .msg_iov = &iov,
        .msg_iovlen = 1
    };
    for (;;) {
        if (sendmsg(fd, &msg, 0) < 0) {
            if (errno == EINTR)
                continue;
            perror("sendmsg");
            return -1;
        }
        return 0;
    }
}

print_diag(const struct unix_diag_msg *diag, unsigned int size)
{
    if (size < NLMSG_LENGTH(sizeof(*diag))) {
        fputs("short response\n", stderr);
        return -1;
    }
    if (diag->udiag_family != AF_UNIX) {
        fprintf(stderr, "unexpected family %u\n", diag->udiag_family);
        return -1;
    }
    unsigned int rta_len = size - NLMSG_LENGTH(sizeof(*diag));
    unsigned int peer = 0;
    size_t path_len = 0;
    char path[sizeof(((struct sockaddr_un *) 0)->sun_path) + 1];
    for (struct rtattr *attr = (struct rtattr *) (diag + 1);
             RTA_OK(attr, rta_len); attr = RTA_NEXT(attr, rta_len)) {
        switch (attr->rta_type) {
        case UNIX_DIAG_NAME:
            if (!path_len) {
                path_len = RTA_PAYLOAD(attr);
                if (path_len > sizeof(path) - 1)
                    path_len = sizeof(path) - 1;
                memcpy(path, RTA_DATA(attr), path_len);
                path[path_len] = '\0';
            }
            break;
        case UNIX_DIAG_PEER:
            if (RTA_PAYLOAD(attr) >= sizeof(peer))
                peer = *(unsigned int *) RTA_DATA(attr);
            break;
        }
    }
    printf("inode=%u", diag->udiag_ino);
    if (peer)
        printf(", peer=%u", peer);
    if (path_len)
        printf(", name=%s%s", *path ? "" : "@",
                *path ? path : path + 1);
    putchar('\n');
    return 0;
}
static int
receive_responses(int fd)
{
    long buf[8192 / sizeof(long)];
    struct sockaddr_nl nladdr;
    struct iovec iov = {
        .iov_base = buf,
        .iov_len = sizeof(buf)
    };
    int flags = 0;
    for (;;) {
        struct msghdr msg = {
            .msg_name = &nladdr,
            .msg_namelen = sizeof(nladdr),
            .msg_iov = &iov,
            .msg_iovlen = 1
        };
        ssize_t ret = recvmsg(fd, &msg, flags);
        if (ret < 0) {
            if (errno == EINTR)
                continue;
            perror("recvmsg");
            return -1;
        }
        if (ret == 0)
            return 0;
        if (nladdr.nl_family != AF_NETLINK) {
            fputs("!AF_NETLINK\n", stderr);
            return -1;
        }
        const struct nlmsghdr *h = (struct nlmsghdr *) buf;
        if (!NLMSG_OK(h, ret)) {
            fputs("!NLMSG_OK\n", stderr);
            return -1;
        }
        for (; NLMSG_OK(h, ret); h = NLMSG_NEXT(h, ret)) {
            if (h->nlmsg_type == NLMSG_DONE)
                return 0;
            if (h->nlmsg_type == NLMSG_ERROR) {
                const struct nlmsgerr *err = NLMSG_DATA(h);
                if (h->nlmsg_len < NLMSG_LENGTH(sizeof(*err))) {
                    fputs("NLMSG_ERROR\n", stderr);
                } else {
                    errno = -err->error;
                    perror("NLMSG_ERROR");
                }
                return -1;
            }
            if (h->nlmsg_type != SOCK_DIAG_BY_FAMILY) {
                fprintf(stderr, "unexpected nlmsg_type %u\n",
                        (unsigned) h->nlmsg_type);
                return -1;
            }
            if (print_diag(NLMSG_DATA(h), h->nlmsg_len))
                return -1;
        }
    }
}
int
main(void)
{
    int fd = socket(AF_NETLINK, SOCK_RAW, NETLINK_SOCK_DIAG);
    if (fd < 0) {
        perror("socket");
        return 1;
    }
    int ret = send_query(fd) || receive_responses(fd);
    close(fd);
    return ret;
}
