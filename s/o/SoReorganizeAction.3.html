int main(int argc, char *argv[]) {
  SbBool vrml1 = input.isFileVRML1();
  SbBool vrml2 = input.isFileVRML2();
  if (vrml2) {
    fprintf(stderr, "VRML2 no se soporta aún\n");
    return -1;
  }
  if (!root) {
    fprintf(stderr, "No se puede leer el archivo.\n");
    return -1;
  }
  root->ref();
  fprintf(stderr, "Aplicando SoReorganizeAction...\n");
  SoReorganizeAction reorg;
  reorg.apply(root);
  fprintf(stderr, "done\n");
  SoOutput out;
  if (out.openFile(argv[2])) {
    if (strip) { // eliminar nodos de tipo coord3, coord4, normal y textura
      fprintf(stderr, "Eliminando nodos antiguos de la jerarquía\n");
      strip_node(SoCoordinate3::getClassTypeId(), root);
      strip_node(SoCoordinate4::getClassTypeId(), root);
      strip_node(SoNormal::getClassTypeId(), root);
      strip_node(SoTextureCoordinate2::getClassTypeId(), root);
    }
    fprintf(stderr, "escribiendo destino\n");
    SoWriteAction wa(&out);
    wa.apply(root);
  }
  root->unref();
  return 0;
} // main()

void SoReorganizeAction::apply (SoPath * path) [virtual]

يطبق الإجراء على أجزاء الرسم البياني المحددة بواسطة path.

ملاحظة: سيحتوي SoPath أيضًا على جميع العقد التي يمكن أن تؤثر، على سبيل المثال، على عقد الهندسة في المسار. لذلك، فإن تطبيق SoGLRenderAction على SoPath سيؤدي إلى عرض المسار بشكل صحيح في العرض، حيث سيتم تطبيق المواد، والملمس، والتكوينات الأخرى للمظهر بشكل صحيح.

إذا انتهى المسار عند عقدة SoGroup، فسوف يجتاز الإجراء أيضًا أطفال العقدة النهائية.

void SoReorganizeAction::beginTraversal (SoNode * node) [protected], [virtual]

يتم استدعاء هذه الدالة الافتراضية من SoAction::apply()، وهي نقطة الدخول لاجتياز شجرة المشهد.

يمكن تجاوزها لتهيئة الإجراء في بداية الاجتياز، للتهيئة الخاصة في الفئات الفرعية لـ SoAction التي ترث SoAction.

يستدعي التنفيذ الافتراضي traverse()، والذي يجب استدعاؤه بواسطة أي تنفيذ متجاوز لهذه الدالة (أو استدعاء SoAction::beginTraversal()) لبدء اجتياز الشجرة.

تم إعادة تنفيذها من SoSimplifyAction.
