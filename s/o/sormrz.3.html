توثيق الوظيفة/الروتين الفرعي

subroutine sormrz (character side, character trans, integer m, integer n, integer k, integer l, real, dimension( lda, * ) a, integer lda, real, dimension( * ) tau, real, dimension( ldc, * ) c, integer ldc, real, dimension( * ) work, integer lwork, integer info)

SORMRZ

الغرض:

!>
!> يقوم SORMRZ بالكتابة فوق مصفوفة C الحقيقية ذات الأبعاد M × N بـ
!>
!>                 SIDE = 'L'     SIDE = 'R'
!> TRANS = 'N':      Q * C          C * Q
!> TRANS = 'T':      Q**T * C       C * Q**T
!>
!> حيث Q مصفوفة متعامدة حقيقية معرفة كناتج لـ k
!> انعكاسات أولية
!>
!>       Q = H(1) H(2) . . . H(k)
!>
!> كما تم إرجاعه بواسطة STZRZF. Q من الدرجة M إذا كانت SIDE = 'L' ومن الدرجة N
!> إذا كانت SIDE = 'R'.
!>

المعلمات

SIDE 
!>          SIDE هو CHARACTER*1
!>          = 'L': تطبيق Q أو Q**T من اليسار؛
!>          = 'R': تطبيق Q أو Q**T من اليمين.
!>

TRANS

!>          TRANS هو CHARACTER*1
!>          = 'N':  بدون تبديل، تطبيق Q؛
!>          = 'T':  تبديل، تطبيق Q**T.
!>

M

!>          M هو INTEGER
!>          عدد صفوف المصفوفة C. M >= 0.
!>

N

!>          N هو INTEGER
!>          عدد أعمدة المصفوفة C. N >= 0.
!>

K

!>          K هو INTEGER
!>          عدد الانعكاسات الأولية التي يحدد ناتجها
!>          المصفوفة Q.
!>          إذا كانت SIDE = 'L'، فإن M >= K >= 0؛
!>          إذا كانت SIDE = 'R'، فإن N >= K >= 0.
!>

L

!>          L هو INTEGER
!>          عدد أعمدة المصفوفة A التي تحتوي على
!>          الجزء الهادف للانعكاسات هاوسهولدر.
!>          إذا كانت SIDE = 'L'، فإن M >= L >= 0، إذا كانت SIDE = 'R'، فإن N >= L >= 0.
!>

A

!>          A هي مصفوفة REAL، أبعادها
!>                               (LDA,M) إذا كانت SIDE = 'L'،
!>                               (LDA,N) إذا كانت SIDE = 'R'
!>          يجب أن يحتوي الصف i على المتجه الذي يحدد الانعكاس الأولي
!>          H(i)، من أجل i = 1,2,...,k، كما تم إرجاعه بواسطة
!>          STZRZF في الصفوف k الأخيرة من وسيط المصفوفة A الخاص به.
!>          يتم تعديل A بواسطة الروتين الفرعي ولكن يتم استعادتها عند الخروج.
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرائد للمصفوفة A. LDA >= max(1,K).
!>

TAU

!>          TAU هي مصفوفة REAL، أبعادها (K)
!>          يجب أن تحتوي TAU(i) على المعامل القياسي للانعكاس الأولي
!>          H(i)، كما تم إرجاعه بواسطة STZRZF.
!>

C

!>          C هي مصفوفة REAL، أبعادها (LDC,N)
!>          عند الإدخال، المصفوفة C ذات الأبعاد M × N.
!>          عند الخروج، يتم الكتابة فوق C بواسطة Q*C أو Q**H*C أو C*Q**H أو C*Q.
!>

LDC

!>          LDC هو INTEGER
!>          البعد الرائد للمصفوفة C. LDC >= max(1,M).
!>

WORK

!>          WORK هي مصفوفة REAL، أبعادها (MAX(1,LWORK))
!>          عند الخروج، إذا كان INFO = 0، فإن WORK(1) تعيد LWORK المثلى.
!>

LWORK

!>          LWORK هو INTEGER
!>          بعد مصفوفة WORK.
!>          إذا كانت SIDE = 'L'، فإن LWORK >= max(1,N)؛
!>          إذا كانت SIDE = 'R'، فإن LWORK >= max(1,M).
!>          لأداء جيد، يجب أن تكون LWORK أكبر بشكل عام.
!>
!>          إذا كانت LWORK = -1، فيفترض استعلام عن مساحة العمل؛ يقوم الروتين الفرعي
!>          بحساب حجم مصفوفة WORK الأمثل فقط، ويعيد
!>          هذه القيمة كمدخل أول لمصفوفة WORK، ولا يتم إصدار أي رسالة خطأ
!>          متعلقة بـ LWORK بواسطة XERBLA.
!>

INFO

!>          INFO هو INTEGER
!>          = 0:  الخروج بنجاح
!>          < 0:  إذا كان INFO = -i، فإن الوسيط i كان له قيمة غير قانونية
!>
