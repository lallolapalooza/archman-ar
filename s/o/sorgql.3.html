توثيق الوظيفة/الروتين الفرعي

subroutine sorgql (integer m, integer n, integer k, real, dimension( lda, * ) a, integer lda, real, dimension( * ) tau, real, dimension( * ) work, integer lwork, integer info)

SORGQL

الغرض:

!>
!> SORGQL تنشئ مصفوفة حقيقية M-by-N بأعمدة متعامدة،
!> والتي تُعرّف على أنها الأعمدة N الأخيرة لناتج K من الانعكاسات الأولية
!> من الدرجة M
!>
!>       Q  =  H(k) . . . H(2) H(1)
!>
!> كما تم إرجاعه بواسطة SGEQLF.
!>

المعلمات

M 
!>          M هو عدد صحيح (INTEGER)
!>          عدد الصفوف في المصفوفة Q. M >= 0.
!>

N

!>          N هو عدد صحيح (INTEGER)
!>          عدد الأعمدة في المصفوفة Q. M >= N >= 0.
!>

K

!>          K هو عدد صحيح (INTEGER)
!>          عدد الانعكاسات الأولية التي يحدد ناتجها المصفوفة Q.
!>          N >= K >= 0.
!>

A

!>          A هو مصفوفة حقيقية (REAL array)، البعد (LDA,N)
!>          عند الإدخال، يجب أن يحتوي العمود (n-k+i) على المتجه الذي
!>          يحدد الانعكاس الأولي H(i)، لكل i = 1،2،...،k، كما
!>          تم إرجاعه بواسطة SGEQLF في الأعمدة K الأخيرة من وسيطة المصفوفة A الخاصة به.
!>          عند الخروج، المصفوفة Q من النوع M-by-N.
!>

LDA

!>          LDA هو عدد صحيح (INTEGER)
!>          البعد الأول للمصفوفة A. LDA >= max(1,M).
!>

TAU

!>          TAU هي مصفوفة حقيقية (REAL array)، البعد (K)
!>          يجب أن يحتوي TAU(i) على العامل العددي للانعكاس الأولي H(i)،
!>          كما تم إرجاعه بواسطة SGEQLF.
!>

WORK

!>          WORK هي مصفوفة حقيقية (REAL array)، البعد (MAX(1,LWORK))
!>          عند الخروج، إذا كان INFO = 0، فإن WORK(1) يُرجع قيمة LWORK المثلى.
!>

LWORK

!>          LWORK هو عدد صحيح (INTEGER)
!>          بعد المصفوفة WORK. LWORK >= max(1,N).
!>          للحصول على أداء مثالي LWORK >= N*NB، حيث NB هو حجم الكتلة الأمثل.
!>
!>          إذا كان LWORK = -1، فسيتم افتراض استعلام عن مساحة العمل؛
!>          يقوم الروتين فقط بحساب الحجم الأمثل للمصفوفة WORK،
!>          ويُرجع هذه القيمة كالإدخال الأول للمصفوفة WORK، ولا يتم إصدار
!>          أي رسالة خطأ متعلقة بـ LWORK بواسطة XERBLA.
!>

INFO

!>          INFO هو عدد صحيح (INTEGER)
!>          = 0:  انتهاء ناجح
!>          < 0:  إذا كان INFO = -i، فإن الوسيط i لديه قيمة غير قانونية
!>
