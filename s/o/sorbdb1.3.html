subroutine sorbdb1 (integer m, integer p, integer q, real, dimension(ldx11,*) x11, integer ldx11, real, dimension(ldx21,*) x21, integer ldx21, real, dimension(*) theta, real, dimension(*) phi, real, dimension(*) taup1, real, dimension(*) taup2, real, dimension(*) tauq1, real, dimension(*) work, integer lwork, integer info)

SORBDB1

الغرض:

!>
!> يقوم SORBDB1 بتنظيم كتل المصفوفة X ذات الأعمدة المتعامدة بشكل متزامن، حيث تكون
!> المصفوفة طويلة ورفيعة:
!>
!>                            [ B11 ]
!>      [ X11 ]   [ P1 |    ] [  0  ]
!>      [-----] = [---------] [-----] Q1**T .
!>      [ X21 ]   [    | P2 ] [ B21 ]
!>                            [  0  ]
!>
!> X11 هي P-by-Q، و X21 هي (M-P)-by-Q. يجب أن يكون Q أصغر من أو يساوي P،
!> M-P، أو M-Q. الروتينات SORBDB2، SORBDB3، و SORBDB4 تعالج الحالات التي
!> يكون فيها Q هو الحد الأدنى من الأبعاد.
!>
!> P1، P2، و Q1 هي مصفوفات متعامدة P-by-P، (M-P)-by-(M-P)،
!> و (M-Q)-by-(M-Q) على التوالي. يتم تمثيلها ضمنيًا بواسطة
!> متجهات هاوسهولدر.
!>
!> B11 و B12 هي مصفوفات ثنائية القطر Q-by-Q ممثلة ضمنيًا بواسطة
!> الزوايا THETA، PHI.
!>
!>

المعلمات

M 
!>          M is INTEGER
!>           عدد صفوف X11 بالإضافة إلى عدد صفوف X21.
!>

P

!>          P is INTEGER
!>           عدد صفوف X11. 0 <= P <= M.
!>

Q

!>          Q is INTEGER
!>           عدد أعمدة X11 و X21. 0 <= Q <=
!>           MIN(P,M-P,M-Q).
!>

X11

!>          X11 is REAL array, dimension (LDX11,Q)
!>           عند الإدخال، الكتلة العلوية للمصفوفة X المراد تخفيضها. عند
!>           الخروج، تحدد أعمدة tril(X11) معاملات الانعكاس لـ P1
!>           وتحدد صفوف triu(X11,1) معاملات الانعكاس لـ Q1.
!>

LDX11

!>          LDX11 is INTEGER
!>           البعد الرئيسي لـ X11. LDX11 >= P.
!>

X21

!>          X21 is REAL array, dimension (LDX21,Q)
!>           عند الإدخال، الكتلة السفلية للمصفوفة X المراد تخفيضها. عند
!>           الخروج، تحدد أعمدة tril(X21) معاملات الانعكاس لـ P2.
!>

LDX21

!>          LDX21 is INTEGER
!>           البعد الرئيسي لـ X21. LDX21 >= M-P.
!>

THETA

!>          THETA is REAL array, dimension (Q)
!>           يتم تعريف مدخلات الكتل ثنائية القطر B11، B21 بواسطة
!>           THETA و PHI. انظر التفاصيل الإضافية.
!>

PHI

!>          PHI is REAL array, dimension (Q-1)
!>           يتم تعريف مدخلات الكتل ثنائية القطر B11، B21 بواسطة
!>           THETA و PHI. انظر التفاصيل الإضافية.
!>

TAUP1

!>          TAUP1 is REAL array, dimension (P)
!>           العوامل القياسية للموصلات الأولية التي تحدد
!>           P1.
!>

TAUP2

!>          TAUP2 is REAL array, dimension (M-P)
!>           العوامل القياسية للموصلات الأولية التي تحدد
!>           P2.
!>

TAUQ1

!>          TAUQ1 is REAL array, dimension (Q)
!>           العوامل القياسية للموصلات الأولية التي تحدد
!>           Q1.
!>

WORK

!>          WORK is REAL array, dimension (LWORK)
!>

LWORK

!>          LWORK is INTEGER
!>           بعد المصفوفة WORK. LWORK >= M-Q.
!>
!>           إذا كان LWORK = -1، فسيتم افتراض استعلام مساحة العمل؛ سيقوم الروتين
!>           فقط بحساب الحجم الأمثل لمصفوفة WORK، وسيعيد
!>           هذه القيمة كالإدخال الأول لمصفوفة WORK، ولن يتم إصدار أي رسالة خطأ
!>           متعلقة بـ LWORK بواسطة XERBLA.
!>

INFO

!>          INFO is INTEGER
!>           = 0:  انتهاء ناجح.
!>           < 0:  إذا كان INFO = -i، فإن الوسيط i كان له قيمة غير قانونية.
!>

تفاصيل إضافية:

!>
!>  يتم تمثيل الكتل ثنائية القطر العليا B11، B21 ضمنيًا بواسطة
!>  الزوايا THETA(1)، ..., THETA(Q) و PHI(1)، ..., PHI(Q-1). كل مدخل
!>  في كل شريط ثنائي القطر هو حاصل ضرب جيب أو جيب تمام لـ THETA
!>  مع جيب أو جيب تمام لـ PHI. انظر [1] أو SORCSD للتفاصيل.
!>
!>  يتم تمثيل P1، P2، و Q1 كحاصل ضرب موصلات أولية.
!>  انظر SORCSD2BY1 للحصول على تفاصيل حول إنشاء P1، P2، و Q1 باستخدام SORGQR
!>  و SORGLQ.
!>

المراجع:

[1] Brian D. Sutton. Computing the complete CS decomposition. Numer. Algorithms, 50(1):33-65, 2009.
