وثائق الوظيفة/الإجراء الفرعي

subroutine sormhr (character side, character trans, integer m, integer n, integer ilo, integer ihi, real, dimension( lda, * ) a, integer lda, real, dimension( * ) tau, real, dimension( ldc, * ) c, integer ldc, real, dimension( * ) work, integer lwork, integer info)

SORMHR

الغرض:

!>
!> SORMHR تستبدل المصفوفة العامة الحقيقية M-by-N المسماة C بـ
!>
!>                 SIDE = 'L'     SIDE = 'R'
!> TRANS = 'N':      Q * C          C * Q
!> TRANS = 'T':      Q**T * C       C * Q**T
!>
!> حيث Q هي مصفوفة متعامدة حقيقية من الرتبة nq، حيث nq = m إذا كان
!> SIDE = 'L' و nq = n إذا كان SIDE = 'R'. يتم تعريف Q كحاصل ضرب
!> IHI-ILO من الانعكاسات الأولية، كما تم إرجاعه بواسطة SGEHRD:
!>
!> Q = H(ilo) H(ilo+1) . . . H(ihi-1).
!>

المعاملات

SIDE 
!>          SIDE هو CHARACTER*1
!>          = 'L': تطبيق Q أو Q**T من اليسار؛
!>          = 'R': تطبيق Q أو Q**T من اليمين.
!>

TRANS

!>          TRANS هو CHARACTER*1
!>          = 'N':  بدون تبديل، تطبيق Q؛
!>          = 'T':  تبديل، تطبيق Q**T.
!>

M

!>          M هو INTEGER
!>          عدد صفوف المصفوفة C. M >= 0.
!>

N

!>          N هو INTEGER
!>          عدد أعمدة المصفوفة C. N >= 0.
!>

ILO

!>          ILO هو INTEGER
!>

IHI

!>          IHI هو INTEGER
!>
!>          يجب أن يكون ILO و IHI بنفس القيم كما في الاستدعاء السابق
!>          لـ SGEHRD. Q تساوي مصفوفة الوحدة باستثناء الجزء الفرعي
!>          Q(ilo+1:ihi,ilo+1:ihi).
!>          إذا كان SIDE = 'L'، فإن 1 <= ILO <= IHI <= M، إذا كان M > 0، و
!>          ILO = 1 و IHI = 0، إذا كان M = 0؛
!>          إذا كان SIDE = 'R'، فإن 1 <= ILO <= IHI <= N، إذا كان N > 0، و
!>          ILO = 1 و IHI = 0، إذا كان N = 0.
!>

A

!>          A هو مصفوفة REAL، الأبعاد
!>                               (LDA,M) إذا كان SIDE = 'L'
!>                               (LDA,N) إذا كان SIDE = 'R'
!>          المتجهات التي تحدد الانعكاسات الأولية، كما
!>          تم إرجاعها بواسطة SGEHRD.
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرائد للمصفوفة A.
!>          LDA >= max(1,M) إذا كان SIDE = 'L'؛ LDA >= max(1,N) إذا كان SIDE = 'R'.
!>

TAU

!>          TAU هو مصفوفة REAL، الأبعاد
!>                               (M-1) إذا كان SIDE = 'L'
!>                               (N-1) إذا كان SIDE = 'R'
!>          يجب أن يحتوي TAU(i) على العامل القياسي للانعكاس الأولي
!>          H(i)، كما تم إرجاعه بواسطة SGEHRD.
!>

C

!>          C هو مصفوفة REAL، الأبعاد (LDC,N)
!>          عند الإدخال، المصفوفة C ذات الأبعاد M-by-N.
!>          عند الخروج، يتم استبدال C بـ Q*C أو Q**T*C أو C*Q**T أو C*Q.
!>

LDC

!>          LDC هو INTEGER
!>          البعد الرائد للمصفوفة C. LDC >= max(1,M).
!>

WORK

!>          WORK هو مصفوفة REAL، الأبعاد (MAX(1,LWORK))
!>          عند الخروج، إذا كان INFO = 0، فإن WORK(1) يعيد LWORK الأمثل.
!>

LWORK

!>          LWORK هو INTEGER
!>          بعد المصفوفة WORK.
!>          إذا كان SIDE = 'L'، فإن LWORK >= max(1,N)؛
!>          إذا كان SIDE = 'R'، فإن LWORK >= max(1,M).
!>          للحصول على الأداء الأمثل، LWORK >= N*NB إذا كان SIDE = 'L'، و
!>          LWORK >= M*NB إذا كان SIDE = 'R'، حيث NB هو حجم الكتلة الأمثل.
!>
!>          إذا كان LWORK = -1، فسيتم افتراض استعلام عن مساحة العمل؛ يقوم الروتين
!>          بحساب حجم مساحة العمل الأمثل فقط، ويعيد
!>          هذه القيمة كالإدخال الأول لمصفوفة WORK، ولا يصدر أي رسالة خطأ
!>          متعلقة بـ LWORK بواسطة XERBLA.
!>

INFO

!>          INFO هو INTEGER
!>          = 0:  انتهاء ناجح
!>          < 0:  إذا كان INFO = -i، فقد كان المعامل i ذا قيمة غير قانونية
!>
