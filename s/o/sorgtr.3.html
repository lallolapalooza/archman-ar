subroutine sorgtr (character uplo, integer n, real, dimension( lda, * ) a, integer lda, real, dimension( * ) tau, real, dimension( * ) work, integer lwork, integer info)

SORGTR

الغرض:

!>
!> SORGTR تولد مصفوفة متعامدة حقيقية Q معرفة كناتج لـ n-1 عاكسات أولية من الرتبة N، كما تم إرجاعها بواسطة
!> SSYTRD:
!>
!> إذا كان UPLO = 'U'، فإن Q = H(n-1) . . . H(2) H(1)،
!>
!> إذا كان UPLO = 'L'، فإن Q = H(1) H(2) . . . H(n-1).
!>

المعلمات

UPLO 
!>          UPLO هو CHARACTER*1
!>          = 'U': المثلث العلوي لـ A يحتوي على العاكسات الأولية
!>                 من SSYTRD؛
!>          = 'L': المثلث السفلي لـ A يحتوي على العاكسات الأولية
!>                 من SSYTRD.
!>

N

!>          N هو INTEGER
!>          رتبة المصفوفة Q. N >= 0.
!>

A

!>          A هو صفيف REAL، البعد (LDA,N)
!>          عند الإدخال، المتجهات التي تحدد العاكسات الأولية،
!>          كما تم إرجاعها بواسطة SSYTRD.
!>          عند الخروج، المصفوفة المتعامدة N-by-N Q.
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرائد لصفيف A. LDA >= max(1,N).
!>

TAU

!>          TAU هو صفيف REAL، البعد (N-1)
!>          يجب أن يحتوي TAU(i) على المعامل القياسي للعاكس
!>          الأولي H(i)، كما تم إرجاعه بواسطة SSYTRD.
!>

WORK

!>          WORK هو صفيف REAL، البعد (MAX(1,LWORK))
!>          عند الخروج، إذا كان INFO = 0، فإن WORK(1) يرجع LWORK الأمثل.
!>

LWORK

!>          LWORK هو INTEGER
!>          بعد صفيف WORK. LWORK >= max(1,N-1).
!>          للأداء الأمثل LWORK >= (N-1)*NB، حيث NB هو
!>          حجم الكتلة الأمثل.
!>
!>          إذا كان LWORK = -1، فسيتم افتراض استعلام مساحة عمل؛ تقوم الروتين
!>          فقط بحساب الحجم الأمثل لصفيف WORK، وترجع
!>          هذه القيمة كالإدخال الأول لصفيف WORK، ولا يتم إصدار أي رسالة خطأ
!>          متعلقة بـ LWORK بواسطة XERBLA.
!>

INFO

!>          INFO هو INTEGER
!>          = 0:  إنهاء ناجح
!>          < 0:  إذا كان INFO = -i، فقد كان الوسيط i ذا قيمة غير قانونية
!>
