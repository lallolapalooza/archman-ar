SoShapeHints(3)

الفئة SoShapeHints هي عقدة تحتوي على تلميحات حول كيفية عرض الهندسة.

تُستخدم عقدة SoShapeHints لتعيين تلميحات حول كيفية رسم ميزات الهندسة التالية في تسلسل المشهد.

هذا مثال على كيفية عرض مشهد في وضع الخطوط، والذي يعرض فقط المثلثات المتجهة نحو الكاميرا.

#include <Inventor/nodes/SoCone.h>
#include <Inventor/nodes/SoDrawStyle.h>;
#include <Inventor/nodes/SoLightModel.h>;
#include <Inventor/nodes/SoSeparator.h>;
#include <Inventor/nodes/SoShapeHints.h>;
#include <Inventor/Win/SoWin.h>;
#include <Inventor/Win/viewers/SoWinExaminerViewer.h>;
int main(int, char * argv[])
{
  HWND window = SoWin::init(argv[0]);
  SoSeparator * root = new SoSeparator;
  // wireframe
  SoDrawStyle * drawStyle = new SoDrawStyle;
  drawStyle->style = SoDrawStyle::LINES;
  root->addChild(drawStyle);
  // back-face culling
  SoShapeHints * shapeHints = new SoShapeHints;
  shapeHints->vertexOrdering = SoShapeHints::COUNTERCLOCKWISE;
  shapeHints->shapeType = SoShapeHints::SOLID;
  root->addChild(shapeHints);
  // keep a solid color
  SoLightModel * lightModel = new SoLightModel;
  lightModel->model = SoLightModel::BASE_COLOR;
  root->addChild(lightModel);
  // a cone
  SoCone * cone = new SoCone;
  root->addChild(cone);
  // set up a window
  SoWinExaminerViewer * viewer = new SoWinExaminerViewer(window);
  SbViewportRegion vpRegion = viewer->getViewportRegion();
  viewer->setSceneGraph(root);
  viewer->show();
  SoWin::mainLoop();
  delete viewer;
  return 0;
}

تم تحسين الإعدادات الافتراضية لنظام العرض لتناسب راحة المطور. على سبيل المثال، الإعداد الافتراضي هو عرض كلا جانبي جميع المضلعات لضمان عدم وجود "ثقوب" في الهندسة إذا كان ترتيب الرؤوس مختلفًا لمضلعات مختلفة.

إذا توقف مطور الكود عن استخدام هذه الراحة واستخدم عقد SoShapeHints لتحديد معلومات أكثر دقة حول هندسة تسلسل الرسم، فسيتم استخدام التلميحات التي توفرها عقد SoShapeHints بواسطة نظام العرض لتجنب العمليات المكلفة. قد ترى فوائد كبيرة في سرعة العرض.

لاحظ أن وضع إزالة الأوجه الخلفية والإضاءة ثنائية الأوجه يقرره نظام العرض وليس بديهيًا. فيما يلي القواعد الشائعة حول كيفية عرض أشكال الأشكال الأولية بناءً على كيفية تعيين حقول SoShapeHints::vertexOrdering و SoShapeHints::shapeType:

  • vertexOrdering == CLOCKWISE أو COUNTERCLOCKWISE، shapeType == SOLID: يتم إزالة الأوجه الخلفية ويتم عرض الأشكال الأولية بإضاءة أحادية الجانب.
  • vertexOrdering == CLOCKWISE أو COUNTERCLOCKWISE، shapeType == UNKNOWN_SHAPE_TYPE: لا يتم إزالة الأوجه الخلفية ويتم عرض الأشكال الأولية بإضاءة ثنائية الجانب.
  • vertexOrdering == UNKNOWN_ORDERING، أي shapeType: لا يتم إزالة الأوجه الخلفية ويتم عرض الأشكال الأولية بإضاءة أحادية الجانب. سيتم تعيين ترتيب الرؤوس في OpenGL إلى الترتيب عكس اتجاه عقارب الساعة.

يحتوي تعداد UNKNOWN_ORDERING على معنى خاص وغير بديهي. يتم في الواقع تعيين الترتيب إلى عكس اتجاه عقارب الساعة - في OpenGL وعند إنشاء المتجهات الطبيعية. ومع ذلك، إذا كنت تريد عرض الهندسة الخاصة بك بإضاءة أحادية الجانب وبدون إزالة الأوجه الخلفية، فيجب عليك استخدام تعداد القيمة هذا، ويجب أن تكون المضلعات الخاصة بك بترتيب عكس اتجاه عقارب الساعة.

تنسيق الملف/الافتراضيات:

ShapeHints {
    vertexOrdering UNKNOWN_ORDERING
    shapeType UNKNOWN_SHAPE_TYPE
    faceType CONVEX
    creaseAngle 0
}

توثيق تعداد الأعضاء

enum SoShapeHints::VertexOrdering

تعداد للطرق المختلفة لتحديد ترتيب رؤوس المضلع.

المُعَدِّد

UNKNOWN_ORDERING
الترتيب غير معروف، يتم عرض جانبي المضلع.
CLOCKWISE
يتم تحديد الرؤوس بترتيب مع اتجاه عقارب الساعة.
COUNTERCLOCKWISE
يتم تحديد الرؤوس بترتيب عكس اتجاه عقارب الساعة.

enum SoShapeHints::ShapeType

تعداد لأنواع الأشكال المختلفة.

المُعَدِّد

UNKNOWN_SHAPE_TYPE
الشكل غير معروف، سيتم عرضه بشكل احترازي.
SOLID
الأشكال التالية في الرسم البياني مغلقة بالكامل وهي أشكال ثلاثية الأبعاد صلبة. سيتم إزالة الأوجه الخلفية إذا كان ترتيب الرؤوس معروفًا.

enum SoShapeHints::FaceType

تعداد لأنواع أوجه المضلعات.

المُعَدِّد

UNKNOWN_FACE_TYPE
يشير إلى: معلومات حول بيانات المضلعات التالية غير معروفة، سيتم عرضها بشكل احترازي.
