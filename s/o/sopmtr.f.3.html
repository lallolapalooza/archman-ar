subroutine sopmtr (character side, character uplo, character trans, integer m, integer n, real, dimension( * ) ap, real, dimension( * ) tau, real, dimension( ldc, * ) c, integer ldc, real, dimension( * ) work, integer info)

SOPMTR

الغرض:

!>
!> SOPMTR يستبدل المصفوفة العامة الحقيقية M-by-N المسماة C بـ
!>
!>                 SIDE = 'L'     SIDE = 'R'
!> TRANS = 'N':      Q * C          C * Q
!> TRANS = 'T':      Q**T * C       C * Q**T
!>
!> حيث Q مصفوفة متعامدة حقيقية من الرتبة nq، حيث nq = m إذا كان
!> SIDE = 'L' و nq = n إذا كان SIDE = 'R'. تُعرف Q بأنها حاصل ضرب
!> nq-1 من الانعكاسات الأولية، كما يتم إرجاعه بواسطة SSPTRD باستخدام التخزين المعبأ:
!>
!> إذا كان UPLO = 'U'، فإن Q = H(nq-1) . . . H(2) H(1);
!>
!> إذا كان UPLO = 'L'، فإن Q = H(1) H(2) . . . H(nq-1).
!>

المعلمات:

SIDE 
!>          SIDE هو CHARACTER*1
!>          = 'L': تطبيق Q أو Q**T من اليسار؛
!>          = 'R': تطبيق Q أو Q**T من اليمين.
!>

UPLO

!>          UPLO هو CHARACTER*1
!>          = 'U': تم استخدام التخزين المثلث العلوي المعبأ في استدعاء سابق
!>                 لـ SSPTRD؛
!>          = 'L': تم استخدام التخزين المثلث السفلي المعبأ في استدعاء سابق
!>                 لـ SSPTRD.
!>

TRANS

!>          TRANS هو CHARACTER*1
!>          = 'N':  لا يوجد تبديل، تطبيق Q؛
!>          = 'T':  تبديل، تطبيق Q**T.
!>

M

!>          M هو INTEGER
!>          عدد صفوف المصفوفة C. M >= 0.
!>

N

!>          N هو INTEGER
!>          عدد أعمدة المصفوفة C. N >= 0.
!>

AP

!>          AP هي مصفوفة REAL، البعد
!>                               (M*(M+1)/2) إذا كان SIDE = 'L'
!>                               (N*(N+1)/2) إذا كان SIDE = 'R'
!>          المتجهات التي تحدد الانعكاسات الأولية، كما
!>          تم إرجاعها بواسطة SSPTRD. يتم تعديل AP بواسطة الروتين ولكن
!>          يتم استعادتها عند الخروج.
!>

TAU

!>          TAU هي مصفوفة REAL، البعد (M-1) إذا كان SIDE = 'L'
!>                                     أو (N-1) إذا كان SIDE = 'R'
!>          يجب أن يحتوي TAU(i) على العامل القياسي للانعكاس
!>          الأولي H(i)، كما تم إرجاعه بواسطة SSPTRD.
!>

C

!>          C هي مصفوفة REAL، البعد (LDC,N)
!>          عند الإدخال، المصفوفة M-by-N C.
!>          عند الخروج، يتم استبدال C بـ Q*C أو Q**T*C أو C*Q**T أو C*Q.
!>

LDC

!>          LDC هو INTEGER
!>          البعد الرائد للمصفوفة C. LDC >= max(1,M).
!>

WORK

!>          WORK هي مصفوفة REAL، البعد
!>                                   (N) إذا كان SIDE = 'L'
!>                                   (M) إذا كان SIDE = 'R'
!>

INFO

!>          INFO هو INTEGER
!>          = 0:  الخروج بنجاح
!>          < 0:  إذا كان INFO = -i، فإن الوسيط i لديه قيمة غير قانونية
!>
