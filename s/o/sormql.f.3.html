subroutine sormql (character side, character trans, integer m, integer n, integer k, real, dimension( lda, * ) a, integer lda, real, dimension( * ) tau, real, dimension( ldc, * ) c, integer ldc, real, dimension( * ) work, integer lwork, integer info)

SORMQL

الغرض:

!>
!> SORMQL تستبدل مصفوفة C العامة ذات الأبعاد M × N بالقيمة
!>
!>                 SIDE = 'L'     SIDE = 'R'
!> TRANS = 'N':      Q * C          C * Q
!> TRANS = 'T':      Q**T * C       C * Q**T
!>
!> حيث Q هي مصفوفة متعامدة حقيقية محددة كناتج لـ k
!> انعكاسات أولية
!>
!>       Q = H(k) . . . H(2) H(1)
!>
!> كما تم إرجاعها بواسطة SGEQLF. Q من الرتبة M إذا كانت SIDE = 'L' ومن الرتبة N
!> إذا كانت SIDE = 'R'.
!>

المعاملات

SIDE 
!>          SIDE هو CHARACTER*1
!>          = 'L': تطبيق Q أو Q**T من اليسار؛
!>          = 'R': تطبيق Q أو Q**T من اليمين.
!>

TRANS

!>          TRANS هو CHARACTER*1
!>          = 'N':  عدم التبديل، تطبيق Q؛
!>          = 'T':  تبديل، تطبيق Q**T.
!>

M

!>          M هو INTEGER
!>          عدد صفوف المصفوفة C. M >= 0.
!>

N

!>          N هو INTEGER
!>          عدد أعمدة المصفوفة C. N >= 0.
!>

K

!>          K هو INTEGER
!>          عدد الانعكاسات الأولية التي يحدد ناتجها
!>          المصفوفة Q.
!>          إذا كانت SIDE = 'L'، فإن M >= K >= 0؛
!>          إذا كانت SIDE = 'R'، فإن N >= K >= 0.
!>

A

!>          A هي مصفوفة REAL، الأبعاد (LDA,K)
!>          يجب أن يحتوي العمود i على المتجه الذي يحدد الانعكاس
!>          الأولي H(i)، لـ i = 1,2,...,k، كما تم إرجاعها بواسطة
!>          SGEQLF في الأعمدة k الأخيرة من وسيطتها المصفوفية A.
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرائد للمصفوفة A.
!>          إذا كانت SIDE = 'L'، فإن LDA >= max(1,M)؛
!>          إذا كانت SIDE = 'R'، فإن LDA >= max(1,N).
!>

TAU

!>          TAU هي مصفوفة REAL، الأبعاد (K)
!>          يجب أن تحتوي TAU(i) على العامل العددي للانعكاس
!>          الأولي H(i)، كما تم إرجاعها بواسطة SGEQLF.
!>

C

!>          C هي مصفوفة REAL، الأبعاد (LDC,N)
!>          عند الدخول، مصفوفة C ذات الأبعاد M × N.
!>          عند الخروج، يتم استبدال C بـ Q*C أو Q**T*C أو C*Q**T أو C*Q.
!>

LDC

!>          LDC هو INTEGER
!>          البعد الرائد للمصفوفة C. LDC >= max(1,M).
!>

WORK

!>          WORK هي مصفوفة REAL، الأبعاد (MAX(1,LWORK))
!>          عند الخروج، إذا كان INFO = 0، فإن WORK(1) يعيد حجم LWORK الأمثل.
!>

LWORK

!>          LWORK هو INTEGER
!>          بعد المصفوفة WORK.
!>          إذا كانت SIDE = 'L'، فإن LWORK >= max(1,N)؛
!>          إذا كانت SIDE = 'R'، فإن LWORK >= max(1,M).
!>          لأداء جيد، يجب أن يكون LWORK أكبر بشكل عام.
!>
!>          إذا كان LWORK = -1، فسيتم افتراض استعلام مساحة العمل؛ يقوم الروتين
!>          بحساب حجم مصفوفة WORK الأمثل فقط، ويعيد
!>          هذه القيمة كالإدخال الأول في مصفوفة WORK، ولا يتم إصدار أي رسالة خطأ
!>          متعلقة بـ LWORK بواسطة XERBLA.
!>

INFO

!>          INFO هو INTEGER
!>          = 0:  انتهاء ناجح
!>          < 0:  إذا كان INFO = -i، فإن الوسيط i كان له قيمة غير قانونية
!>
