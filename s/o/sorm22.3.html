subroutine sorm22 (character side, character trans, integer m, integer n, integer n1, integer n2, real, dimension( ldq, * ) q, integer ldq, real, dimension( ldc, * ) c, integer ldc, real, dimension( * ) work, integer lwork, integer info)

SORM22 تضرب مصفوفة عامة بمصفوفة متعامدة ذات نطاق.

الغرض

!>
!>
!>  SORM22 تستبدل المصفوفة الحقيقية العامة M-by-N المسماة C بـ
!>
!>                  SIDE = 'L'     SIDE = 'R'
!>  TRANS = 'N':      Q * C          C * Q
!>  TRANS = 'T':      Q**T * C       C * Q**T
!>
!>  حيث Q هي مصفوفة متعامدة حقيقية من الرتبة NQ، حيث NQ = M إذا كان
!>  SIDE = 'L' و NQ = N إذا كان SIDE = 'R'.
!>  المصفوفة المتعامدة Q تعالج بنية كتلة 2x2
!>
!>         [  Q11  Q12  ]
!>     Q = [            ]
!>         [  Q21  Q22  ],
!>
!>  حيث Q12 مصفوفة مثلثية سفلية بحجم N1xN1 و Q21 مصفوفة مثلثية علوية بحجم
!>  N2xN2.
!>

المعلمات

SIDE 
!>          SIDE is CHARACTER*1
!>          = 'L': تطبيق Q أو Q**T من اليسار؛
!>          = 'R': تطبيق Q أو Q**T من اليمين.
!>

TRANS

!>          TRANS is CHARACTER*1
!>          = 'N':  تطبيق Q (بدون تبديل)؛
!>          = 'C':  تطبيق Q**T (تبديل مترافق).
!>

M

!>          M is INTEGER
!>          عدد صفوف المصفوفة C. M >= 0.
!>

N

!>          N is INTEGER
!>          عدد أعمدة المصفوفة C. N >= 0.
!>

N1
N2

!>          N1 is INTEGER
!>          N2 is INTEGER
!>          أبعاد Q12 و Q21 على التوالي. N1، N2 >= 0.
!>          يجب استيفاء المتطلب التالي:
!>          N1 + N2 = M إذا كان SIDE = 'L' و N1 + N2 = N إذا كان SIDE = 'R'.
!>

Q

!>          Q is REAL array, dimension
!>                              (LDQ,M) if SIDE = 'L'
!>                              (LDQ,N) if SIDE = 'R'
!>

LDQ

!>          LDQ is INTEGER
!>          البعد الرائد للمصفوفة Q.
!>          LDQ >= max(1,M) if SIDE = 'L'; LDQ >= max(1,N) if SIDE = 'R'.
!>

C

!>          C is REAL array, dimension (LDC,N)
!>          عند الإدخال، المصفوفة C بحجم M-by-N.
!>          عند الخروج، يتم استبدال C بـ Q*C أو Q**T*C أو C*Q**T أو C*Q.
!>

LDC

!>          LDC is INTEGER
!>          البعد الرائد للمصفوفة C. LDC >= max(1,M).
!>

WORK

!>          WORK is REAL array, dimension (MAX(1,LWORK))
!>          عند الخروج، إذا كان INFO = 0، فإن WORK(1) ترجع LWORK الأمثل.
!>

LWORK

!>          LWORK is INTEGER
!>          بعد المصفوفة WORK.
!>          إذا كان SIDE = 'L'، فإن LWORK >= max(1,N)؛
!>          إذا كان SIDE = 'R'، فإن LWORK >= max(1,M).
!>          لأداء أمثل LWORK >= M*N.
!>
!>          إذا كان LWORK = -1، فسيتم افتراض استعلام عن مساحة العمل؛ تقوم الروتينية
!>          بحساب حجم مساحة العمل الأمثل فقط، وترجع
!>          هذه القيمة كالإدخال الأول في مصفوفة WORK، ولا يصدر XERBLA أي رسالة خطأ
!>          متعلقة بـ LWORK.
!>

INFO

!>          INFO is INTEGER
!>          = 0:  خروج ناجح
!>          < 0:  إذا كان INFO = -i، فإن المعامل i كان له قيمة غير قانونية
!>
