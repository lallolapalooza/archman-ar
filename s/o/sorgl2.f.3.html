subroutine sorgl2 (integer m, integer n, integer k, real, dimension( lda, * ) a, integer lda, real, dimension( * ) tau, real, dimension( * ) work, integer info)

SORGL2

الغرض:

!>
!> SORGL2 يولد مصفوفة حقيقية m x n ذات صفوف متعامدة،
!> والتي تُعرّف كأول m صفوف من حاصل ضرب k عواكس أولية من الرتبة n
!>
!>       Q  =  H(k) . . . H(2) H(1)
!>
!> كما تم إرجاعه بواسطة SGELQF.
!>

المعلمات:

M 
!>          M هو عدد صحيح (INTEGER)
!>          عدد صفوف المصفوفة Q. M >= 0.
!>

N

!>          N هو عدد صحيح (INTEGER)
!>          عدد أعمدة المصفوفة Q. N >= M.
!>

K

!>          K هو عدد صحيح (INTEGER)
!>          عدد العواكس الأولية التي يُعرّف حاصل ضربها المصفوفة Q. M >= K >= 0.
!>

A

!>          A هي مصفوفة حقيقية (REAL array)، ذات البعد (LDA,N)
!>          عند الإدخال، يجب أن يحتوي الصف i على المتجه الذي يحدد
!>          العاكس الأولي H(i)، لـ i = 1,2,...,k، كما تم إرجاعه
!>          بواسطة SGELQF في أول k صفوف من مصفوفة الوسيط A الخاصة به.
!>          عند الخروج، المصفوفة Q ذات البعد m x n.
!>

LDA

!>          LDA هو عدد صحيح (INTEGER)
!>          البعد الأول للمصفوفة A. LDA >= max(1,M).
!>

TAU

!>          TAU هي مصفوفة حقيقية (REAL array)، ذات البعد (K)
!>          يجب أن يحتوي TAU(i) على المعامل القياسي للعاكس
!>          الأولي H(i)، كما تم إرجاعه بواسطة SGELQF.
!>

WORK

!>          WORK هي مصفوفة حقيقية (REAL array)، ذات البعد (M)
!>

INFO

!>          INFO هو عدد صحيح (INTEGER)
!>          = 0: خروج ناجح
!>          < 0: إذا كان INFO = -i، فإن الوسيط i يحتوي على قيمة غير قانونية
!>
