snmp_alarm_unregister(3)

الحزمة: net-snmp

التراخيص: BSD

SNMP_ALARM(3) Net-SNMP SNMP_ALARM(3)

الاسم

snmp_alarm_register, snmp_alarm_register_hr, snmp_alarm_unregister - وظائف التنبيه

الملخص

#include <net-snmp/utilities.h>

unsigned int
snmp_alarm_register(unsigned int seconds,
unsigned int flags,
SNMPAlarmCallback *f_callback,
void *clientarg);

unsigned int
snmp_alarm_register_hr(struct timeval t,
unsigned int flags,
SNMPAlarmCallback *f_callback,
void *clientarg);

void
snmp_alarm_unregister(unsigned int reg);

الوصف

تقوم هذه الوظائف بتنفيذ دعم لآلية تعامل مع المؤقتات العامة لأجزاء متعددة من التطبيق لتسجيل استدعاءات الوظائف التي تحدث في وقت معين في المستقبل.

الاستخدام

الاستخدام بسيط ومباشر نسبيًا: ببساطة قم بإنشاء وظيفة تريد استدعائها في وقت ما في المستقبل. يجب أن يكون تعريف الوظيفة مشابهًا لـ:

void my_callback(unsigned int reg, void *clientarg);

ثم، قم باستدعاء snmp_alarm_register() لتسجيل دالة الاستدعاء الخاصة بك ليتم استدعاؤها بعد seconds من الآن. يجب أن يكون حقل flags إما SA_REPEAT أو NULL. إذا تم تعيين flags مع SA_REPEAT، فسيتم استدعاء دالة الاستدعاء المسجلة كل seconds. إذا كان flags هو NULL فسيتم استدعاء الدالة مرة واحدة فقط ثم إزالتها من تسجيل نظام الإنذار.

يتم استخدام المعلمة clientarg في وظيفة التسجيل من قبل وظيفة العميل فقط ويتم تخزينها وإعادتها مباشرة إليهم في كل مرة يتم فيها استدعاء النظام.

تعيد وظيفة snmp_alarm_register() قيمة unsigned int فريدة (والتي يتم تمريرها أيضًا كالحجة الأولى لكل استدعاء)، والتي يمكن استخدامها بعد ذلك لإزالة دالة الاستدعاء من قائمة الانتظار في وقت لاحق باستخدام وظيفة snmp_alarm_unregister(). إذا فشل استدعاء snmp_alarm_register()، فإنه يعيد صفرًا. على وجه الخصوص، لاحظ أنه مسموح تمامًا لدالة التنبيه بإلغاء تسجيل نفسها.

وظيفة snmp_alarm_register_hr() مماثلة في التشغيل لوظيفة snmp_alarm_register()، ولكنها تأخذ struct timeval كمعلمة أولى، وتقوم بجدولة الاستدعاء بعد الفترة الممثلة بـ t (الأحرف hr تعني "دقة عالية"). يعتمد تشغيل هذه الوظيفة على توفير استدعاء النظام setitimer(2) بواسطة نظام التشغيل. إذا لم يكن استدعاء النظام هذا متاحًا، فسيتم جدولة التنبيه كما لو تم استدعاء snmp_alarm_register() بمعلمة أولى تساوي قيمة العضو tv_sec في t. راجع، مع ذلك، الملاحظات أدناه.

التهيئة

تقوم وظيفة init_snmp() بتهيئة النظام الفرعي snmp_alarm عن طريق استدعاء init_snmp_alarm() ثم init_alarm_post_config() لإعداد المؤقت الأول لتهيئة دالة الاستدعاء. لا ينبغي استخدام هاتين الوظيفتين مباشرة من قبل التطبيقات.

ملاحظات

السلوك الافتراضي للنظام الفرعي snmp_alarm هو طلب إشارات SIGALRM من نظام التشغيل عبر استدعاءات النظام alarm(2) أو setitimer(2). ومع ذلك، فإن هذا له عيب يتمثل في أنه لا يمكن لأي جزء آخر من التطبيق استخدام وظائف SIGALRM (أو، إذا استخدم جزء آخر من التطبيق وظائف SIGALRM، فلن يعمل النظام الفرعي snmp_alarm بشكل صحيح).

إذا كان تطبيقك يقوم بتشغيل حلقة أحداث تعتمد على select(2)، فلا حاجة لاستخدام SIGALRM للنظام الفرعي snmp_alarm، مما يتركه متاحًا لأجزاء أخرى من التطبيق. يتم ذلك عن طريق إجراء الاستدعاء التالي:

netsnmp_ds_set_boolean(NETSNMP_DS_LIBRARY_ID, 
                       NETSNMP_DS_LIB_ALARM_DONT_USE_SIG, 1);

قبل استدعاء init_snmp(). بعد ذلك، تأخذ snmp_select_info() التنبيهات في الاعتبار عند حساب قيمة المهلة التي سيتم استخدامها لـ select(2). كل ما عليك فعله هو استدعاء run_alarms() عند انتهاء مهلة select(2) (قيمة الإرجاع صفر). هذا هو النهج المتبع في الوكيل؛ راجع snmpd.c. علاوة على ذلك، عند استخدام هذه الطريقة، لا تعتمد التنبيهات عالية الدقة على وجود استدعاء النظام setitimer(2)، على الرغم من أن الدقة الإجمالية لا تزال محددة بواسطة نظام التشغيل الأساسي. موصى به.
