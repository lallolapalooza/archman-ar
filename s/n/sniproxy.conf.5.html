الحزمة: sniproxy

التراخيص: BSD-2-Clause

SNIPROXY.CONF(5) sniproxy SNIPROXY.CONF(5)

الاسم

sniproxy.conf - ملف إعداد sniproxy

الملخص

/etc/sniproxy.conf

الوصف

/etc/sniproxy.conf هو ملف الإعداد لـ sniproxy. يتم فصل العبارات إما بسطر جديد أو بفاصلة منقوطة. الأسطر التي تبدأ بـ # هي تعليقات. يتم تقسيم الإعداد إلى مقاطع محددة بأقواس معقوفة. يمكن إلغاء الأحرف باستخدام \. يمكن اختصار توجيهات الإعداد طالما أنها غير قابلة للخلط، على سبيل المثال، user daemon بدلاً من username daemon.

اسم المستخدم

username daemon

حدد المستخدم الذي سيعمل sniproxy باسمه. عندما يتم تشغيل sniproxy كمستخدم خارق، فإنه سيسقط الأذونات إلى هذا المستخدم.

ملف PID

pidfile /var/run/sniproxy.pid

حدد المسار إلى ملف PID، ويجب أن يكون الدليل قابلاً للكتابة بواسطة المستخدم الذي يعمل sniproxy باسمه.

سجل الأخطاء

error_log {
    syslog daemon
    priority notice
}

حدد كيفية التعامل مع رسائل الخطأ. يمكن تسجيل الرسائل إما إلى syslog باستخدام توجيه syslog الذي يحدد أن السجلات يجب أن تذهب إلى مرفق syslog معين. بدلاً من ذلك، يمكن تحديد توجيه filename لتسجيل إلى ملف، هذان الخياران متعارضان. يشير توجيه priority إلى مستوى شدة الرسائل التي يجب تسجيلها. الأولويات المقبولة هي أولويات syslog القياسية، بترتيب زيادة التفاصيل: emergency, alert, critical, error, warning, notice, info, and debug.

سجل الوصول

access_log {
    filename /var/log/sniproxy/access.log
}

حدد كيفية تسجيل الاتصالات، يمكن تجاوز ذلك في مستمع معين. يتم تسجيل الاتصالات بعد إغلاق مقابسس كل من العميل والخادم. يمكن استخدام توجيه syslog و priority هنا كما في error_log.

المحلل

resolver {
    nameserver 127.0.0.1
    mode ipv6_first
}

حدد كيفية تحليل استعلامات DNS، وهذا مطلوب فقط إذا كنت تستخدم أسماء المضيفين كعناوين في الإعداد أو تستخدم خلفيات wildcard. إذا لم يتم تحديده، فسيتم إجراء استعلامات IPv4 فقط باستخدام خوادم أسماء النظام الافتراضية.

هناك أربعة أوضاع مدعومة:

ipv4_only: استعلام عن أي سجلات A، استخدم أول سجل A تم إرجاعه (بعد سجلات CNAME).

ipv6_only: استعلام عن أي سجلات AAAA، استخدم أول سجل AAAA تم إرجاعه (بعد سجلات CNAME).

ipv4_first: استعلام عن كل من سجلات A و AAAA، انتظر حتى تكتمل كلتا الاستعلامين، استخدم أول سجل A إذا كان موجودًا، وإلا استخدم أول سجل AAAA.

ipv6_first: استعلام عن كل من سجلات A و AAAA، انتظر حتى تكتمل كلتا الاستعلامين، استخدم أول سجل AAAA إذا كان موجودًا، وإلا استخدم أول سجل A.

يوصى بشدة باستخدام خادم أسماء محلي، حيث يتم إعادة استخدام مقبس واحد لجميع استعلامات DNS وبالتالي يكون رقم منفذ UDP قابلاً للتنبؤ به، مما يجعل الاستعلام محميًا فقط من الردود المزيفة بواسطة معرف الاستعلام المكون من 16 بت. بالإضافة إلى ذلك، نظرًا لعدم إجراء ذاكرة تخزين مؤقت داخلية لـ DNS، يمكن لمحلل محلي تحسين الأداء.

المستمع

listener 192.0.2.10:80 {
    protocol http
    reuseport yes
    table http_hosts
    fallback 192.0.2.100:80
    bad_requests log
    source 192.0.2.10
    access_log {
        filename /var/log/sniproxy/http_access.log
    }
}
listener [::]:80 {
    protocol http
    ipv6_v6only yes
    table http_hosts
    fallback unix:/var/run/http_fallback_unix.sock
}

حدد عنوان استماع لقبول اتصالات جديدة من العملاء. يمكن تحديد العناوين كحرف IPv4 متبوعًا برقم منفذ TCP، وحرف IPv6 متبوعًا برقم منفذ TCP، أو رقم منفذ TCP مجرد، أو مسار مقبس Unix مسبوقًا بـ 'unix:'.

يحدد البروتوكول كيفية تحليل طلب العميل للحصول على اسم المضيف المطلوب، وهناك بروتوكولان مدعومان: http و tls.

يتحكم توجيه reuseport فيما إذا كان المنفذ مفتوحًا في وضع SO_REUSEPORT، مما يسمح بتشغيل عدة مثيلات sniproxy على نفس زوج ip:port. هذا يتيح لنا موازنة تحميل الاتصالات الواردة بالتساوي بين هذه المثيلات دون الحاجة إلى أي وكيل موازنة تحميل خارجي. يتطلب نواة Linux 3.9+. تعيين reuseport إلى "yes" يتيح هذه الوظيفة.

يتحكم توجيه ipv6_v6only فيما إذا كان الاستماع على عنوان IPv6 الشامل '::' سيقبل الاتصالات إلى أي عنوان IPv4 بالإضافة إلى عنوان IPv6. هذا مفيد إذا كان المستخدم يريد إعدادات مختلفة لـ IPv4 و IPv6 أو يرغب في معالجة حركة مرور IPv4 باستخدام خادم/وكيل آخر تمامًا. هذا ينطبق فقط على مستمعي IPv6، ويتم تجاهله على المستمعين الآخرين.

تحدد table اسم الجدول المستخدم للبحث عن الخادم الذي سيتم إعادة توجيه الاتصال إليه بناءً على اسم المضيف المستخرج من طلب العميل الأولي. إذا لم يتم تحديد توجيه table، فسيتم استخدام الجدول الافتراضي غير المسمى.

يحدد توجيه fallback الخادم الذي سيتم استخدامه إذا لم يكن من الممكن تحليل طلب العميل، أو إذا تعذر العثور على خادم في الجدول لاسم المضيف المحدد، أو إذا تعذر حل اسم المضيف. يجب أن يكون هذا عنوان IP ومنفذ أو مسار مقبس Unix.

يسمح توجيه bad_requests بتسجيل محتويات طلب العميل إذا لم يكن قابلاً للتحليل، وهذا مفيد لتصحيح الأخطاء.

يسمح توجيه source بتحديد عنوان محدد للربط به قبل الاتصال بالخادم الخلفي. في معظم الحالات، من الأفضل تجاهل هذا الخيار والسماح لنظام التشغيل باختيار عنوان الخروج تلقائيًا. لا تقم بتضمين رقم منفذ في هذا العنوان، فإن القيام بذلك سيحد الوكيل من اتصال واحد متزامن بكل خادم في وقت واحد.

يمكن تجاوز تكوين سجل الوصول في كل مستمع.

الجدول

table http_hosts {
    ^example\.com$ 192.0.2.101
    ^example\.net$ 192.0.2.102
    ^example\.org$ 192.0.2.103 proxy_protocol
}

تحدد الجداول كيفية ربط كل اسم مضيف بخادم خلفي. تتم مطابقة اسم المضيف لكل طلب مع الإدخالات في الجدول بالترتيب، حتى يتم العثور على تطابق ويتم استخدام هذا الخادم. يمكن أن يكون عنوان الخادم إما IP، أو IP ومنفذ، أو مسار مقبس Unix، أو اسم مضيف، أو '*'. إذا لم يتم تحديد منفذ، فسيتم استخدام منفذ المستمع الذي تم استلام الاتصال عليه.

سيقوم الخيار الاختياري proxy_protocol بإلحاق رأس بروتوكول HAProxy PROXY v1 بالاتصال المعاد توجيهه، مما يسمح لخوادم الويب الداعمة بالحصول على عنوان IP ومنفذ المصدر والوجهة لاتصال TCP الوارد الأصلي.
