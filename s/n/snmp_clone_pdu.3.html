netsnmp_pdu *snmp_pdu_create( inttype);

netsnmp_pdu *snmp_clone_pdu( netsnmp_pdu*pdu);

netsnmp_pdu *snmp_fix_pdu( netsnmp_pdu*pdu, int idx);

netsnmp_pdu *snmp_free_pdu( netsnmp_pdu*pdu);
