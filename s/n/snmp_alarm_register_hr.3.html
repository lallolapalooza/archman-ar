snmp_alarm_register_hr(3)

الحزمة: net-snmp

التراخيص: BSD

SNMP_ALARM(3) Net-SNMP SNMP_ALARM(3)

الاسم

snmp_alarm_register, snmp_alarm_register_hr, snmp_alarm_unregister - وظائف الإنذار

الملخص

#include <net-snmp/utilities.h>

unsigned int
snmp_alarm_register(unsigned int seconds,
unsigned int flags,
SNMPAlarmCallback *f_callback,
void *clientarg);

unsigned int
snmp_alarm_register_hr(struct timeval t,
unsigned int flags,
SNMPAlarmCallback *f_callback,
void *clientarg);

void
snmp_alarm_unregister(unsigned int reg);

الوصف

تطبق هذه الوظائف دعم آلية توقيت عامة لمعالجة الاستدعاءات الوظيفية لأجزاء متعددة من التطبيق لتحدث في وقت معين في المستقبل.

الاستخدام

الاستخدام بسيط ومباشر إلى حد ما: قم ببساطة بإنشاء دالة تريد استدعاءها مرة أخرى في وقت ما في المستقبل. يجب أن يكون تعريف الدالة مشابهًا لـ:

void my_callback(unsigned int reg, void *clientarg);

ثم، قم باستدعاء snmp_alarm_register() لتسجيل استدعاءك الوظيفي ليتم استدعاؤه بعد seconds من الآن. يجب أن يكون حقل flags إما SA_REPEAT أو NULL. إذا تم تعيين flags مع SA_REPEAT، فسيتم استدعاء الدالة المسجلة كل seconds. إذا كان flags هو NULL، فسيتم استدعاء الدالة مرة واحدة فقط ثم إزالتها من تسجيل نظام الإنذار.

يتم استخدام المعامل clientarg في وظيفة التسجيل بواسطة الدالة العميل فقط ويتم تخزينه وإعادته مباشرة إليها في كل مرة يتم فيها استدعاء النظام.

تعيد الدالة snmp_alarm_register() قيمة unsigned int فريدة (والتي يتم تمريرها أيضًا كوسيط أول لكل استدعاء وظيفي)، والتي يمكن استخدامها لاحقًا لإزالة الاستدعاء الوظيفي من قائمة الانتظار في وقت لاحق باستخدام الدالة snmp_alarm_unregister(). إذا فشلت مكالمة snmp_alarm_register()، فإنها تعيد صفرًا. على وجه الخصوص، لاحظ أنه من المسموح تمامًا لدالة إنذار أن تزيل تسجيل نفسها.

الدالة snmp_alarm_register_hr() مطابقة في التشغيل للدالة snmp_alarm_register()، ولكنها تأخذ struct timeval كوسيط أول، وتقوم بجدولة الاستدعاء الوظيفي بعد الفترة التي تمثلها t (الأحرف hr تعني "دقة عالية"). يعتمد تشغيل هذه الدالة على توفير استدعاء النظام setitimer(2) بواسطة نظام التشغيل. إذا لم يكن استدعاء النظام هذا متاحًا، فسيتم جدولة الإنذار كما لو تم استدعاء snmp_alarm_register() بوسيط أول يساوي قيمة العضو tv_sec في t. انظر، مع ذلك، الملاحظات أدناه.

التهيئة

تقوم الدالة init_snmp() بتهيئة النظام الفرعي snmp_alarm عن طريق استدعاء init_snmp_alarm() ثم init_alarm_post_config() لإعداد المؤقت الأول لتهيئة دالة الاستدعاء الوظيفي. لا ينبغي استخدام هاتين الدالتين مباشرة بواسطة التطبيقات.

ملاحظات

السلوك الافتراضي للنظام الفرعي snmp_alarm هو طلب إشارات SIGALRM من نظام التشغيل عبر استدعاءات النظام alarm(2) أو setitimer(2). ومع ذلك، فإن هذا له عيب وهو أنه لا يمكن لأي جزء آخر من التطبيق استخدام وظيفة SIGALRM (أو، إذا استخدم جزء آخر من التطبيق وظيفة SIGALRM، فلن يعمل النظام الفرعي snmp_alarm بشكل صحيح).

إذا كان تطبيقك يعمل بحلقة أحداث تعتمد على select(2)، فلا داعي لاستخدام SIGALRM للنظام الفرعي snmp_alarm، مما يتركه متاحًا لأجزاء أخرى من التطبيق. يتم ذلك عن طريق إجراء المكالمة التالية:

netsnmp_ds_set_boolean(NETSNMP_DS_LIBRARY_ID, 
                       NETSNMP_DS_LIB_ALARM_DONT_USE_SIG, 1);

قبل استدعاء init_snmp(). بعد ذلك، تأخذ snmp_select_info() الإنذارات في الاعتبار عند حساب قيمة المهلة التي سيتم استخدامها لـ select(2). كل ما عليك فعله هو استدعاء run_alarms() عندما تنتهي مهلة select(2) (قيمة الإرجاع صفر). هذا هو النهج الذي تتبعه الوكيل؛ انظر snmpd.c. علاوة على ذلك، عند استخدام هذه الطريقة، لا تعتمد الإنذارات عالية الدقة على وجود استدعاء النظام setitimer(2)، على الرغم من أن الدقة الشاملة لا تزال محددة بواسطة نظام التشغيل الأساسي. موصى به.
