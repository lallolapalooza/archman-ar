subroutine cungrq (integer m, integer n, integer k, complex, dimension( lda, * ) a, integer lda, complex, dimension( * ) tau, complex, dimension( * ) work, integer lwork, integer info)

CUNGRQ

الغرض:

!>
!> CUNGRQ تولد مصفوفة معقدة M-by-N تسمى Q ذات صفوف متعامدة،
!> والتي يتم تعريفها على أنها آخر M صفوف من حاصل ضرب K من الموجهات الابتدائية
!>
!>       Q  =  H(1)**H H(2)**H . . . H(k)**H
!>
!> كما تم إرجاعها بواسطة CGERQF.
!>

المعاملات

M 
!>          M هو عدد صحيح
!>          عدد صفوف المصفوفة Q. M >= 0.
!>

N

!>          N هو عدد صحيح
!>          عدد أعمدة المصفوفة Q. N >= M.
!>

K

!>          K هو عدد صحيح
!>          عدد الموجهات الابتدائية التي تحدد حاصل ضربها المصفوفة Q. M >= K >= 0.
!>

A

!>          A هي مصفوفة معقدة، بأبعاد (LDA,N)
!>          عند الإدخال، يجب أن يحتوي الصف (m-k+i) على المتجه الذي
!>          يحدد الموجه الأولي H(i)، من أجل i = 1,2,...,k، كما
!>          تم إرجاعه بواسطة CGERQF في آخر K صفوف من مصفوفة الوسيط
!>          A.
!>          عند الخروج، المصفوفة Q بحجم M-by-N.
!>

LDA

!>          LDA هو عدد صحيح
!>          البعد الأول للمصفوفة A. LDA >= max(1,M).
!>

TAU

!>          TAU هي مصفوفة معقدة، بأبعاد (K)
!>          يجب أن يحتوي TAU(i) على العامل القياسي للموجه الأولي
!>          H(i)، كما تم إرجاعه بواسطة CGERQF.
!>

WORK

!>          WORK هي مصفوفة معقدة، بأبعاد (MAX(1,LWORK))
!>          عند الخروج، إذا كانت INFO = 0، فإن WORK(1) ترجع LWORK المثلى.
!>

LWORK

!>          LWORK هو عدد صحيح
!>          بعد مصفوفة WORK. LWORK >= max(1,M).
!>          للحصول على الأداء الأمثل LWORK >= M*NB، حيث NB هو
!>          حجم الكتلة الأمثل.
!>
!>          إذا كانت LWORK = -1، فسيتم افتراض استعلام مساحة العمل؛ الروتين
!>          يحسب فقط الحجم الأمثل لمصفوفة WORK، ويرجع
!>          هذه القيمة كأول إدخال في مصفوفة WORK، ولا يتم إصدار رسالة خطأ
!>          متعلقة بـ LWORK بواسطة XERBLA.
!>

INFO

!>          INFO هو عدد صحيح
!>          = 0:  خروج ناجح
!>          < 0:  إذا كانت INFO = -i، فإن الوسيط i له قيمة غير قانونية
!>
