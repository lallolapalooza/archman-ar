الاسم

utmp, wtmp - سجلات تسجيل الدخول

الاختصار

#include <utmp.h>

الوصف

يسمح ملف utmp باكتشاف معلومات حول من يستخدم النظام حاليًا. قد يكون هناك مستخدمون أكثر حاليًا يستخدمون النظام، لأن ليس كل البرامج تستخدم تسجيل utmp.

تحذير: يجب ألا يكون ملف utmp قابلاً للكتابة من قبل فئة المستخدم "other"، لأن العديد من برامج النظام (بشكل أحمق) تعتمد على سلامته. أنت تخاطر بملفات سجل نظام مزيفة وتعديلات على ملفات النظام إذا تركت ملف utmp قابلاً للكتابة لأي مستخدم آخر غير المالك والمجموعة المالكة للملف.

الملف هو تسلسل من هياكل utmp، معرّفة على النحو التالي في <utmp.h> (لاحظ أن هذا مجرد تعريف واحد من عدة تعريفات؛ التفاصيل تعتمد على إصدار libc):


/* القيم لحقل ut_type، أدناه */
#define EMPTY         0 /* السجل لا يحتوي على معلومات صالحة
                           (كان يُعرف سابقًا باسم UT_UNKNOWN على Linux) */
#define RUN_LVL       1 /* تغيير في مستوى تشغيل النظام (انظر
init(1)) 
*/
#define BOOT_TIME     2 /* وقت إقلاع النظام (في ut_tv) */
#define NEW_TIME      3 /* الوقت بعد تغيير ساعة النظام
                           (في ut_tv) */
#define OLD_TIME      4 /* الوقت قبل تغيير ساعة النظام
                           (في ut_tv) */
#define INIT_PROCESS  5 /* عملية تم إنشاؤها بواسطة
init(1) 
*/
#define LOGIN_PROCESS 6 /* عملية قائدة للجلسة لتسجيل دخول المستخدم */
#define USER_PROCESS  7 /* عملية عادية */
#define DEAD_PROCESS  8 /* عملية منتهية */
#define ACCOUNTING    9 /* غير مطبق */
#define UT_LINESIZE      32
#define UT_NAMESIZE      32
#define UT_HOSTSIZE     256
struct exit_status {              /* النوع لـ ut_exit، أدناه */
    short e_termination;          /* حالة إنهاء العملية */
    short e_exit;                 /* حالة خروج العملية */
};
struct utmp {
    short   ut_type;              /* نوع السجل */
    pid_t   ut_pid;               /* معرف العملية (PID) لعملية تسجيل الدخول */
    char    ut_line[UT_LINESIZE]; /* اسم الجهاز لطرفية -- "/dev/" */
    char    ut_id[4];             /* لاحقة اسم الطرفية،
                                     أو معرف inittab(5) */
    char    ut_user[UT_NAMESIZE]; /* اسم المستخدم */
    char    ut_host[UT_HOSTSIZE]; /* اسم المضيف لتسجيل الدخول عن بعد، أو
                                     إصدار النواة لرسائل مستوى التشغيل */
    struct  exit_status ut_exit;  /* حالة خروج عملية
                                     معلمة كـ DEAD_PROCESS؛ لا
                                     تستخدمها Linux
init(1) 
*/
    /* يجب أن يكون حقلا ut_session و ut_tv بنفس الحجم عند
       التجميع 32- و 64- بت. هذا يسمح بمشاركة ملفات البيانات والذاكرة المشتركة
       بين التطبيقات 32- و 64- بت. */
#if __WORDSIZE == 64 && defined __WORDSIZE_COMPAT32
    int32_t ut_session;           /* معرف الجلسة (
getsid(2))،
                                     تستخدم للنوافذ */
    struct {
        int32_t tv_sec;           /* الثواني */
        int32_t tv_usec;          /* الميكروثواني */
    } ut_tv;                      /* الوقت الذي تم فيه إنشاء السجل */
#else
     long   ut_session;           /* معرف الجلسة */
     struct timeval ut_tv;        /* الوقت الذي تم فيه إنشاء السجل */
#endif
    int32_t ut_addr_v6[4];        /* عنوان الإنترنت للمضيف
                                     البعيد؛ عنوان IPv4 يستخدم
                                     فقط ut_addr_v6[0] */
    char __unused[20];            /* محجوز للاستخدام المستقبلي */
};
/* تعديلات للتوافق الرجعي */
#define ut_name ut_user
#ifndef _NO_UT_TIME
#define ut_time ut_tv.tv_sec
#endif
#define ut_xtime ut_tv.tv_sec
#define ut_addr ut_addr_v6[0]

يقدم هذا الهيكل اسم الملف الخاص المرتبط بطرفية المستخدم، واسم تسجيل دخول المستخدم، ووقت تسجيل الدخول في شكل time(2). يتم إنهاء حقول السلسلة بحرف فارغ ('\0') إذا كانت أقصر من حجم الحقل.

نتجت الإدخالات الأولى التي تم إنشاؤها من معالجة init(1) لـ inittab(5). قبل معالجة إدخال، ومع ذلك، يقوم init(1) بتنظيف utmp عن طريق تعيين ut_type إلى DEAD_PROCESS، ومسح ut_user و ut_host و ut_time بأحرف فارغة لكل سجل يكون فيه ut_type ليس DEAD_PROCESS أو RUN_LVL ولا توجد عملية بالمعرف ut_pid. إذا لم يتم العثور على سجل فارغ بالمعرف ut_id المطلوب، يقوم init(1) بإنشاء سجل جديد. يقوم بتعيين ut_id من inittab، و ut_pid و ut_time إلى القيم الحالية، و ut_type إلى INIT_PROCESS.

يقوم mingetty(8) (أو agetty(8)) بتحديد موقع الإدخال بواسطة PID، ويغير ut_type إلى LOGIN_PROCESS، ويغير ut_time، ويضبط ut_line، وينتظر حتى يتم إنشاء الاتصال. يقوم login(1)، بعد مصادقة المستخدم، بتغيير ut_type إلى USER_PROCESS، وتغيير ut_time، وضبط ut_host و ut_addr. اعتمادًا على mingetty(8) (أو agetty(8)) و login(1)، قد يتم تحديد الإدخالات بواسطة ut_line بدلاً من ut_pid المفضل.

عندما يكتشف init(1) أن عملية قد خرجت، فإنه يحدد موقع إدخال utmp الخاص بها بواسطة ut_pid، ويضبط ut_type إلى DEAD_PROCESS، ويمسح ut_user و ut_host و ut_time بأحرف فارغة.

يقوم xterm(1) ومحاكيات الطرفية الأخرى بإنشاء سجل USER_PROCESS مباشرة وإنشاء ut_id باستخدام السلسلة التي هي الجزء اللاحق من اسم الطرفية (الأحرف التي تلي /dev/[pt]ty). إذا وجدت DEAD_PROCESS لهذا المعرف، فإنها تعيد استخدامه، وإلا فإنها تنشئ إدخالًا جديدًا. إذا استطاعت، فستعلمها كـ DEAD_PROCESS عند الخروج، ومن المستحسن أن تقوم بمسح ut_line و ut_time و ut_user و ut_host أيضًا.

يقوم telnetd(8) بإعداد إدخال LOGIN_PROCESS ويترك الباقي لـ login(1) كالمعتاد. بعد انتهاء جلسة telnet، يقوم telnetd(8) بتنظيف utmp بالطريقة الموضحة.

ملف wtmp يسجل جميع عمليات تسجيل الدخول والخروج. تنسيقه مطابق تمامًا لـ utmp باستثناء أن اسم المستخدم الفارغ يشير إلى تسجيل خروج من الطرفية المرتبطة. علاوة على ذلك، يشير اسم الطرفية ~ مع اسم المستخدم shutdown أو reboot إلى إيقاف تشغيل النظام أو إعادة تشغيله، وزوج أسماء الطرفيات |/} يسجل وقت النظام القديم/الجديد عند تغيير date(1). يتم الاحتفاظ بـ wtmp بواسطة login(1) و init(1) وبعض إصدارات getty(8) (مثل mingetty(8) أو agetty(8)). لا يقوم أي من هذه البرامج بإنشاء الملف، لذلك إذا تم إزالته، يتم إيقاف حفظ السجلات.

الملفات

/var/run/utmp
/var/log/wtmp

الإصدارات

لا يحدد POSIX.1 هيكل utmp، بل هيكل يسمى utmpx (كجزء من امتداد XSI)، مع مواصفات للحقول ut_type و ut_pid و ut_line و ut_id و ut_user و ut_tv. لا يحدد POSIX.1 أطوال حقلي ut_line و ut_user.

يحدد Linux هيكل utmpx ليكون مطابقًا لهيكل utmp.

المعايير

Linux.

التاريخ

utmp(5) — Linux man-pages (6.16)

NAME

utmp — login record file

SYNOPSIS

#include <utmp.h>

DESCRIPTION

The utmp file records user logins and logouts. It should be a regular file, not a symbolic link. It is used by programs such as who(1), last(1), and login(1).

The format of the entries in the utmp file is defined in <utmp.h>. The structure is called struct utmp and typically contains the following fields:

char ut_user[32];
Login name. Null-padded and possibly space-padded on older systems.
char ut_id[4];
Current starting identifier of the process.
char ut_line[32];
The name of the terminal device (e.g., /dev/tty1).
char ut_host[256];
Host name of a remote login (e.g. from telnet or rsh). It is empty for local logins. This field is present since Linux kernel 2.6.1.
pid_t ut_pid;
Process ID of the login process.
short ut_type;
Type of entry. Possible values are:
EMPTY
RUN_LVL
BOOT_TIME
OLD_TIME
NEW_TIME
INIT_PROCESS
LOGIN_PROCESS
USER_PROCESS
DEAD_PROCESS
ACCOUNTING
struct timeval ut_tv;
Time of last change.
int32_t ut_session;
Session ID (valid for login processes and dead processes).
int32_t ut_addr_v6[4];
Internet address of remote host. The first element is the family (AF_INET or AF_INET6). This field is present since Linux kernel 2.6.1.

The fields ut_host and ut_addr_v6 are present since Linux 2.6.1 and are only set by glibc.

For local logins, ut_host is empty. For remote logins, it contains the hostname.

The field ut_addr_v6 contains the IP address of the remote host.

The field ut_session contains the session ID of the user.

The field ut_id is used to link utmp entries to accounting records.

The field ut_type indicates the type of the entry. USER_PROCESS indicates a user login. DEAD_PROCESS indicates that the process has died. LOGIN_PROCESS indicates that the system is waiting for a login. INIT_PROCESS indicates that the system is booting. BOOT_TIME indicates that the system has just been booted. RUN_LVL indicates that the system run-level has changed. OLD_TIME and NEW_TIME indicate that the system clock has been changed.

The field ut_tv contains the time of the last change.

FILES

/var/run/utmp

/var/log/wtmp

VERSIONS

The fields ut_host and ut_addr_v6 were added in Linux kernel 2.6.1.

STANDARDS

The utmp file format is not standardized. Different Unix systems have different formats. This description is based on the Linux implementation, which is largely compatible with the System V and BSD formats.

v7/BSD has fewer fields; most importantly it lacks ut_type, which causes native v7/BSD-like programs to display (for example) dead or login entries. Further, there is no configuration file which allocates slots to sessions. BSD does so because it lacks ut_id fields.

In Linux (as in System V), the ut_id field of a record will never change once it has been set, which reserves that slot without needing a configuration file. Clearing ut_id may result in race conditions leading to corrupted utmp entries and potential security holes. Clearing the abovementioned fields by filling them with null bytes is not required by System V semantics, but makes it possible to run many programs which assume BSD semantics and which do not modify utmp. Linux uses the BSD conventions for line contents, as documented above.

System V has no ut_host or ut_addr_v6 fields.

NOTES

Unlike various other systems, where utmp logging can be disabled by removing the file, utmp must always exist on Linux. If you want to disable who(1), then do not make utmp world readable.

The file format is machine-dependent, so it is recommended that it be processed only on the machine architecture where it was created.

Note that on biarch platforms, that is, systems which can run both 32-bit and 64-bit applications (x86-64, ppc64, s390x, etc.), ut_tv is the same size in 32-bit mode as in 64-bit mode. The same goes for ut_session and ut_time if they are present. This allows data files and shared memory to be shared between 32-bit and 64-bit applications. This is achieved by changing the type of ut_session to int32_t, and that of ut_tv to a struct with two int32_t fields tv_sec and tv_usec. Since ut_tv may not be the same as struct timeval, then instead of the call:


gettimeofday((struct timeval *) &ut.ut_tv, NULL);

the following method of setting this field is recommended:


struct utmp ut;
struct timeval tv;
gettimeofday(&tv, NULL);
ut.ut_tv.tv_sec = tv.tv_sec;
ut.ut_tv.tv_usec = tv.tv_usec;

الاسم

utmp — سجل تسجيل الدخول

المُلَخَّص

#include <utmp.h>

الوصف

يسجل ملف utmp عمليات تسجيل دخول وخروج المستخدمين. يجب أن يكون ملفًا عاديًا، وليس رابطًا رمزيًا. تستخدمه البرامج مثل who(1) و last(1) و login(1).

تم تعريف تنسيق الإدخالات في ملف utmp في <utmp.h>. الهيكل يسمى struct utmp ويحتوي عادةً على الحقول التالية:

char ut_user[32];
اسم تسجيل الدخول. مبطن بالصفر وربما مبطن بالمسافات في الأنظمة القديمة.
char ut_id[4];
معرف البداية الحالي للعملية.
char ut_line[32];
اسم جهاز الطرفية (مثل /dev/tty1).
char ut_host[256];
اسم المضيف لتسجيل الدخول عن بعد (مثل من telnet أو rsh). يكون فارغًا لتسجيلات الدخول المحلية. هذا الحقل موجود منذ نواة لينكس 2.6.1.
pid_t ut_pid;
معرف العملية لعملية تسجيل الدخول.
short ut_type;
نوع الإدخال. القيم الممكنة هي:
EMPTY
RUN_LVL
BOOT_TIME
OLD_TIME
NEW_TIME
INIT_PROCESS
LOGIN_PROCESS
USER_PROCESS
DEAD_PROCESS
ACCOUNTING
struct timeval ut_tv;
وقت آخر تغيير.
int32_t ut_session;
معرف الجلسة (صالح لعمليات تسجيل الدخول والعمليات الميتة).
int32_t ut_addr_v6[4];
عنوان الإنترنت للمضيف البعيد. العنصر الأول هو العائلة (AF_INET أو AF_INET6). هذا الحقل موجود منذ نواة لينكس 2.6.1.

الحقلان ut_host و ut_addr_v6 موجودان منذ لينكس 2.6.1 ويتم تعيينهما بواسطة glibc فقط.

لتسجيلات الدخول المحلية، يكون ut_host فارغًا. لتسجيلات الدخول عن بعد، يحتوي على اسم المضيف.

يحتوي الحقل ut_addr_v6 على عنوان IP للمضيف البعيد.

يحتوي الحقل ut_session على معرف جلسة المستخدم.

يستخدم الحقل ut_id لربط إدخالات utmp بسجلات المحاسبة.

يشير الحقل ut_type إلى نوع الإدخال. USER_PROCESS يشير إلى تسجيل دخول مستخدم. DEAD_PROCESS يشير إلى أن العملية قد ماتت. LOGIN_PROCESS يشير إلى أن النظام ينتظر تسجيل الدخول. INIT_PROCESS يشير إلى أن النظام يقوم بالتمهيد. BOOT_TIME يشير إلى أن النظام قد تم تمهيده للتو. RUN_LVL يشير إلى أن مستوى تشغيل النظام قد تغير. OLD_TIME و NEW_TIME يشيران إلى أن ساعة النظام قد تغيرت.

يحتوي الحقل ut_tv على وقت آخر تغيير.

الملفات

/var/run/utmp

/var/log/wtmp

الإصدارات

تمت إضافة الحقلين ut_host و ut_addr_v6 في نواة لينكس 2.6.1.

المعايير

تنسيق ملف utmp غير قياسي. أنظمة يونكس المختلفة لديها تنسيقات مختلفة. يعتمد هذا الوصف على تنفيذ لينكس، وهو متوافق إلى حد كبير مع تنسيقات System V و BSD.

يحتوي v7/BSD على حقول أقل؛ والأهم من ذلك أنه يفتقر إلى ut_type، مما يتسبب في عرض البرامج الأصلية الشبيهة بـ v7/BSD (على سبيل المثال) للإدخالات الميتة أو إدخالات تسجيل الدخول. علاوة على ذلك، لا يوجد ملف تكوين يخصص أماكن للجلسات. يفعل BSD ذلك لأنه يفتقر إلى حقول ut_id.

في لينكس (كما في System V)، لن يتغير حقل ut_id للسجل مرة واحدة تم تعيينه، مما يحجز هذا المكان دون الحاجة إلى ملف تكوين. قد يؤدي مسح ut_id إلى ظروف سباق تؤدي إلى تلف إدخالات utmp وثغرات أمنية محتملة. لا يتطلب مسح الحقول المذكورة أعلاه عن طريق ملئها ببايتات فارغة دلالات System V، ولكنه يجعل من الممكن تشغيل العديد من البرامج التي تفترض دلالات BSD والتي لا تعدل utmp. يستخدم لينكس اتفاقيات BSD لمحتويات الخط، كما هو موضح أعلاه.

لا يحتوي System V على حقول ut_host أو ut_addr_v6.

ملاحظات

على عكس الأنظمة المختلفة، حيث يمكن تعطيل تسجيل utmp عن طريق إزالة الملف، يجب أن يوجد utmp دائمًا على لينكس. إذا كنت ترغب في تعطيل who(1)، فلا تجعل utmp قابلًا للقراءة من قبل الجميع.

تنسيق الملف يعتمد على الجهاز، لذلك يوصى بمعالجته فقط على بنية الجهاز التي تم إنشاؤه عليها.

لاحظ أنه على المنصات المزدوجة البنية، أي الأنظمة التي يمكنها تشغيل تطبيقات 32 بت و 64 بت (x86-64، ppc64، s390x، إلخ)، يكون ut_tv بنفس الحجم في وضع 32 بت كما في وضع 64 بت. وينطبق الشيء نفسه على ut_session و ut_time إذا كانت موجودة. هذا يسمح بمشاركة ملفات البيانات والذاكرة المشتركة بين تطبيقات 32 بت و 64 بت. يتم تحقيق ذلك عن طريق تغيير نوع ut_session إلى int32_t، ونوع ut_tv إلى هيكل يحتوي على حقلين int32_t هما tv_sec و tv_usec. نظرًا لأن ut_tv قد لا يكون هو نفسه struct timeval، فبدلاً من الاستدعاء:


gettimeofday((struct timeval *) &ut.ut_tv, NULL);

الطريقة التالية لتعيين هذا الحقل موصى بها:


struct utmp ut;
struct timeval tv;
gettimeofday(&tv, NULL);
ut.ut_tv.tv_sec = tv.tv_sec;
ut.ut_tv.tv_usec = tv.tv_usec;
