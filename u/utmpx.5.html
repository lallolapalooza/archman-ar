utmp(5) دليل صيغ الملفات utmp(5)

الاسم

utmp, wtmp - سجلات تسجيل الدخول

الملخص

#include <utmp.h>

الوصف

يتيح ملف utmp للمرء اكتشاف معلومات حول من يستخدم النظام حاليًا. قد يكون هناك المزيد من المستخدمين الذين يستخدمون النظام حاليًا، لأن ليس كل البرامج تستخدم تسجيل utmp.

تحذير: يجب ألا يكون utmp قابلًا للكتابة من قبل فئة المستخدمين "other"، لأن العديد من برامج النظام (بحماقة) تعتمد على سلامته. أنت تخاطر بسجلات نظام مزيفة وتعديلات على ملفات النظام إذا تركت utmp قابلًا للكتابة لأي مستخدم آخر غير مالك الملف ومجموعة مالكه.

الملف عبارة عن تسلسل من هياكل utmp، المُعلنة كما يلي في <utmp.h> (لاحظ أن هذا مجرد واحد من عدة تعريفات موجودة؛ تعتمد التفاصيل على إصدار libc):


/* قيم حقل ut_type، أدناه */
#define EMPTY         0 /* السجل لا يحتوي على معلومات صالحة
                           (المعروف سابقًا باسم UT_UNKNOWN على لينكس) */
#define RUN_LVL       1 /* تغيير في مستوى تشغيل النظام (انظر
init(1)) 
*/
#define BOOT_TIME     2 /* وقت إقلاع النظام (في ut_tv) */
#define NEW_TIME      3 /* الوقت بعد تغيير ساعة النظام
                           (في ut_tv) */
#define OLD_TIME      4 /* الوقت قبل تغيير ساعة النظام
                           (في ut_tv) */
#define INIT_PROCESS  5 /* عملية أنشأها 
init(1) 
*/
#define LOGIN_PROCESS 6 /* عملية قائد الجلسة لتسجيل دخول المستخدم */
#define USER_PROCESS  7 /* عملية عادية */
#define DEAD_PROCESS  8 /* عملية منتهية */
#define ACCOUNTING    9 /* غير مُنفذة */
#define UT_LINESIZE      32
#define UT_NAMESIZE      32
#define UT_HOSTSIZE     256
struct exit_status {              /* نوع ut_exit، أدناه */
    short e_termination;          /* حالة إنهاء العملية */
    short e_exit;                 /* حالة خروج العملية */
};
struct utmp {
    short   ut_type;              /* نوع السجل */
    pid_t   ut_pid;               /* PID لعملية تسجيل الدخول */
    char    ut_line[UT_LINESIZE]; /* اسم جهاز tty - "/dev/" */
    char    ut_id[4];             /* لاحقة اسم الطرفية،
                                     أو معرّف inittab(5) */
    char    ut_user[UT_NAMESIZE]; /* اسم المستخدم */
    char    ut_host[UT_HOSTSIZE]; /* اسم المضيف لتسجيل الدخول عن بُعد، أو
                                     إصدار النواة لرسائل مستوى
                                     التشغيل */
    struct  exit_status ut_exit;  /* حالة خروج عملية
                                     مُعلّمة كـ DEAD_PROCESS؛ لا
                                     يستخدمها 
init(1) على لينكس 
*/
    /* يجب أن يكون حقلا ut_session و ut_tv بنفس الحجم عند
       الترجمة 32- و 64-بت.  هذا يسمح بمشاركة ملفات البيانات والذاكرة
       المشتركة بين تطبيقات 32- و 64-بت.  */
#if __WORDSIZE == 64 && defined __WORDSIZE_COMPAT32
    int32_t ut_session;           /* معرّف الجلسة (
getsid(2))،
                                     يُستخدم للنوافذ */
    struct {
        int32_t tv_sec;           /* الثواني */
        int32_t tv_usec;          /* الميكروثواني */
    } ut_tv;                      /* وقت إنشاء الإدخال */
#else
     long   ut_session;           /* معرّف الجلسة */
     struct timeval ut_tv;        /* وقت إنشاء الإدخال */
#endif
    int32_t ut_addr_v6[4];        /* عنوان الإنترنت للمضيف
                                     البعيد؛ عنوان IPv4 يستخدم
                                     فقط ut_addr_v6[0] */
    char __unused[20];            /* محجوز للاستخدام المستقبلي */
};
/* حيل التوافق العكسي */
#define ut_name ut_user
#ifndef _NO_UT_TIME
#define ut_time ut_tv.tv_sec
#endif
#define ut_xtime ut_tv.tv_sec
#define ut_addr ut_addr_v6[0]

يمنح هذا الهيكل اسم الملف الخاص المرتبط بطرفية المستخدم، واسم تسجيل دخول المستخدم، ووقت تسجيل الدخول بصيغة time(2). تُنهى حقول النصوص بايت فارغ ('\0') إذا كانت أقصر من حجم الحقل.

أول الإدخالات التي تُنشأ على الإطلاق تنتج عن معالجة init(1) لـ inittab(5). قبل معالجة إدخال، ومع ذلك، يقوم init(1) بتنظيف utmp عن طريق تعيين ut_type إلى DEAD_PROCESS، ومسح ut_user وut_host وut_time ببايتات فارغة لكل سجل لا يكون فيه ut_type هو DEAD_PROCESS أو RUN_LVL وحيث لا توجد عملية بالمعرّف ut_pid. إذا لم يمكن العثور على سجل فارغ بالمعرّف ut_id المطلوب، ينشئ init(1) سجلًا جديدًا. يعيّن ut_id من inittab، وut_pid وut_time إلى القيم الحالية، وut_type إلى INIT_PROCESS.

يحدد mingetty(8) (أو agetty(8)) الإدخال بواسطة PID، ويغيّر ut_type إلى LOGIN_PROCESS، ويغيّر ut_time، ويعيّن ut_line، وينتظر إنشاء الاتصال. يقوم login(1)، بعد مصادقة المستخدم، بتغيير ut_type إلى USER_PROCESS، وتغيير ut_time، وتعيين ut_host وut_addr. اعتمادًا على mingetty(8) (أو agetty(8)) وlogin(1)، قد تُحدد السجلات بواسطة ut_line بدلًا من ut_pid المفضل.

عندما يكتشف init(1) أن عملية قد خرجت، يحدد إدخال utmp الخاص بها بواسطة ut_pid، ويعيّن ut_type إلى DEAD_PROCESS، ويمسح ut_user وut_host وut_time ببايتات فارغة.

ينشئ xterm(1) ومحاكيات الطرفية الأخرى مباشرة سجل USER_PROCESS ويولّدون ut_id باستخدام النص الذي يلحق اسم الطرفية (الأحرف التي تلي /dev/[pt]ty). إذا وجدوا DEAD_PROCESS لهذا المعرّف، فإنهم يعيدون تدويره، وإلا فإنهم ينشئون إدخالًا جديدًا. إذا استطاعوا، فسيعلّمونه كـ DEAD_PROCESS عند الخروج ويُنصح بأن يمسحوا ut_line وut_time وut_user وut_host أيضًا.

ينشئ telnetd(8) إدخال LOGIN_PROCESS ويترك الباقي لـ login(1) كالمعتاد. بعد انتهاء جلسة telnet، ينظف telnetd(8) utmp بالطريقة الموصوفة.

يسجل ملف wtmp جميع عمليات تسجيل الدخول والخروج. صيغته مطابقة تمامًا لـ utmp باستثناء أن اسم المستخدم الفارغ يشير إلى تسجيل خروج على الطرفية المرتبطة. علاوة على ذلك، يشير اسم الطرفية ~ مع اسم المستخدم shutdown أو reboot إلى إيقاف تشغيل النظام أو إعادة تشغيله، ويسجل زوج أسماء الطرفيات |/} وقت النظام القديم/الجديد عندما يغيّره date(1). يُصان wtmp بواسطة login(1) وinit(1) وبعض إصدارات getty(8) (مثل mingetty(8) أو agetty(8)). لا ينشئ أي من هذه البرامج الملف، لذا إذا أُزيل، يتم إيقاف حفظ السجلات.

الملفات

/var/run/utmp
/var/log/wtmp

الإصدارات

لا يحدد POSIX.1 هيكل utmp، بل هيكلًا يُسمى utmpx (كجزء من امتداد XSI)، مع مواصفات للحقول ut_type وut_pid وut_line وut_id وut_user وut_tv. لا يحدد POSIX.1 أطوال حقلي ut_line وut_user.

يعرّف لينكس هيكل utmpx ليكون مطابقًا لهيكل utmp.

المعايير

لينكس.

التاريخ

لا تتوافق إدخالات utmp في لينكس مع v7/BSD ولا مع System V؛ فهي مزيج من الاثنين.

يمتلك v7/BSD حقولًا أقل؛ والأهم أنه يفتقر إلى ut_type، مما يجعل البرامج الأصلية الشبيهة بـ v7/BSD تعرض (على سبيل المثال) إدخالات ميتة أو إدخالات تسجيل دخول. علاوة على ذلك، لا يوجد ملف تكوين يخصص فتحات للجلسات. يفعل BSD ذلك لأنه يفتقر إلى حقول ut_id.

في لينكس (كما في System V)، لن يتغير حقل ut_id لسجل أبدًا بمجرد تعيينه، مما يحجز تلك الفتحة دون الحاجة إلى ملف تكوين. قد يؤدي مسح ut_id إلى حالات تسابق تؤدي إلى إدخالات utmp تالفة وثغرات أمنية محتملة. مسح الحقول المذكورة أعلاه عن طريق ملء

هذا الموقع غير مرتبط بـ Arch Linux ولا تابع له بأي شكل من الأشكال. المحتوى هو ترجمة فقط لصفحات دليل Arch المنشورة على man.archlinux.org.

تعبئتها ببايتات فارغة ليس مطلوبًا من قبل دلالات System V، لكنه يجعل من الممكن تشغيل العديد من البرامج التي تفترض دلالات BSD والتي لا تعدّل utmp. يستخدم Linux اصطلاحات BSD لمحتويات السطر، كما هو موثق أعلاه.

لا يمتلك System V حقلي ut_host أو ut_addr_v6.

ملاحظات

على عكس أنظمة أخرى متنوعة، حيث يمكن تعطيل تسجيل utmp بإزالة الملف، يجب أن يكون utmp موجودًا دائمًا على Linux. إذا كنت تريد تعطيل who(1)، فلا تجعل utmp قابلًا للقراءة للجميع.

تنسيق الملف يعتمد على الآلة، لذا يُوصى بأن تتم معالجته فقط على معمارية الآلة التي أُنشئ فيها.

لاحظ أنه على منصات biarch، أي الأنظمة التي يمكنها تشغيل تطبيقات 32-بت و64-بت معًا (x86-64، ppc64، s390x، إلخ)، فإن ut_tv له نفس الحجم في وضع 32-بت كما في وضع 64-بت. الأمر نفسه ينطبق على ut_session وut_time إذا كانا موجودين. هذا يسمح بمشاركة ملفات البيانات والذاكرة المشتركة بين تطبيقات 32-بت و64-بت. ويتم تحقيق ذلك بتغيير نوع ut_session إلى int32_t، ونوع ut_tv إلى بنية تحتوي على حقلَي int32_t هما tv_sec وtv_usec. بما أن ut_tv قد لا يكون مطابقًا لـ struct timeval، فبدلًا من الاستدعاء:


gettimeofday((struct timeval *) &ut.ut_tv, NULL);

يُوصى بالطريقة التالية لضبط هذا الحقل:


struct utmp ut;
struct timeval tv;
gettimeofday(&tv, NULL);
ut.ut_tv.tv_sec = tv.tv_sec;
ut.ut_tv.tv_usec = tv.tv_usec;
