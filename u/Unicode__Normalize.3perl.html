Unicode__Normalize.3perl

الحزمة: perl

التراخيص: Artistic-1.0-Perl OR GPL-1.0-or-later

Unicode::Normalize(3perl) دليل مرجع مبرمجي Perl Unicode::Normalize(3perl)

الاسم

Unicode::Normalize - أشكال تطبيع Unicode

الملخص

(1) باستخدام أسماء الدوال المصدَّرة افتراضيًا:

use Unicode::Normalize;
$NFD_string  = NFD($string);  # شكل التطبيع D
$NFC_string  = NFC($string);  # شكل التطبيع C
$NFKD_string = NFKD($string); # شكل التطبيع KD
$NFKC_string = NFKC($string); # شكل التطبيع KC

(2) باستخدام أسماء الدوال المصدَّرة عند الطلب:

use Unicode::Normalize 'normalize';
$NFD_string  = normalize('D',  $string);  # شكل التطبيع D
$NFC_string  = normalize('C',  $string);  # شكل التطبيع C
$NFKD_string = normalize('KD', $string);  # شكل التطبيع KD
$NFKC_string = normalize('KC', $string);  # شكل التطبيع KC

الوصف

المعاملات:

يُستخدم $string كسلسلة تحت دلالات المحارف (انظر perlunicode).

ينبغي أن يكون $code_point عددًا صحيحًا غير سالب يمثل نقطة رمز Unicode.

ملاحظة: بين XSUB وPerl النقي، يوجد عدم توافق حول تفسير $code_point كعدد عشري. يحوّل XSUB $code_point إلى عدد صحيح غير سالب، لكن Perl النقي لا يفعل ذلك. لا تستخدم عددًا عشريًا ولا إشارة سالبة في $code_point.

أشكال التطبيع

"$NFD_string = NFD($string)"
يعيد شكل التطبيع D (المكوَّن بواسطة التحلل القانوني).
"$NFC_string = NFC($string)"
يعيد شكل التطبيع C (المكوَّن بواسطة التحلل القانوني متبوعًا بالتركيب القانوني).
"$NFKD_string = NFKD($string)"
يعيد شكل التطبيع KD (المكوَّن بواسطة التحلل التوافقي).
"$NFKC_string = NFKC($string)"
يعيد شكل التطبيع KC (المكوَّن بواسطة التحلل التوافقي متبوعًا بالتركيب القانوني).
"$FCD_string = FCD($string)"
إذا كانت السلسلة المعطاة في FCD (شكل "Fast C or D"؛ راجع UTN #5)، فإنه يعيد السلسلة دون تعديل؛ وإلا فإنه يعيد سلسلة FCD.

ملاحظة: FCD ليس فريدًا دائمًا، لذا قد تكون الصيغ المتعددة متكافئة مع بعضها البعض. سيعيد FCD() إحدى هذه الصيغ المتكافئة.

"$FCC_string = FCC($string)"
يعيد شكل FCC ("Fast C Contiguous"؛ راجع UTN #5).

ملاحظة: FCC فريد، وكذلك أشكال التطبيع الأربعة (NF*).

"$normalized_string = normalize($form_name, $string)"
يعيد شكل التطبيع الخاص بـ $form_name.

باعتباره $form_name، يجب إعطاء أحد الأسماء التالية.

'C'  أو 'NFC'  لشكل التطبيع C  (UAX #15)
'D'  أو 'NFD'  لشكل التطبيع D  (UAX #15)
'KC' أو 'NFKC' لشكل التطبيع KC (UAX #15)
'KD' أو 'NFKD' لشكل التطبيع KD (UAX #15)
'FCD'          لشكل "Fast C or D"  (UTN #5)
'FCC'          لشكل "Fast C Contiguous" (UTN #5)

التحلل والتركيب

"$decomposed_string = decompose($string [, $useCompatMapping])"
يعيد تسلسل تحلل كل محرف في السلسلة.

إذا حُذف المعامل الثاني (قيمة منطقية) أو كان خطأ، فإن التحلل هو التحلل القانوني؛ وإذا كان المعامل الثاني (قيمة منطقية) صحيحًا، فإن التحلل هو التحلل التوافقي.

السلسلة المعادة ليست دائمًا في NFD/NFKD. قد تكون إعادة الترتيب مطلوبة.

$NFD_string  = reorder(decompose($string));       # مكافئ لـ NFD()
$NFKD_string = reorder(decompose($string, TRUE)); # مكافئ لـ NFKD()
"$reordered_string = reorder($string)"
يعيد نتيجة إعادة ترتيب المحارف المدمجة وفقًا لسلوك الترتيب القانوني.

على سبيل المثال، عندما تكون لديك قائمة من سلاسل NFD/NFKD، يمكنك الحصول على سلسلة NFD/NFKD المتسلسلة منها بقولك

$concat_NFD  = reorder(join '', @NFD_strings);
$concat_NFKD = reorder(join '', @NFKD_strings);
"$composed_string = compose($string)"
يعيد نتيجة التركيب القانوني دون تطبيق أي تحلل.

على سبيل المثال، عندما تكون لديك سلسلة NFD/NFKD، يمكنك الحصول على سلسلة NFC/NFKC الخاصة بها بقولك

$NFC_string  = compose($NFD_string);
$NFKC_string = compose($NFKD_string);
"($processed, $unprocessed) = splitOnLastStarter($normalized)"
يعيد سلسلتين: الأولى، $processed، هي الجزء قبل آخر بادئ، والثانية، $unprocessed، هي جزء آخر بعد الجزء الأول. البادئ هو محرف له فئة دمج تساوي صفرًا (راجع UAX #15).

لاحظ أن $processed قد تكون فارغة (عندما لا تحتوي $normalized على بادئ أو تبدأ بآخر بادئ)، وعندها ينبغي أن تكون $unprocessed مساوية لكامل $normalized.

عندما تكون لديك سلسلة $normalized وسلسلة $unnormalized تتبعها، فإن التسلسل البسيط خاطئ:

$concat = $normalized . normalize($form, $unnormalized); # خاطئ!

بدلًا من ذلك، افعل هكذا:

($processed, $unprocessed) = splitOnLastStarter($normalized);
$concat = $processed . normalize($form,$unprocessed.$unnormalized);

ينبغي استدعاء splitOnLastStarter() بمعامل مُطبَّع مسبقًا $normalized، أي في نفس الشكل الذي تريده $form.

إذا كانت لديك مصفوفة @string ينبغي تسلسلها ثم تطبيعها، يمكنك فعل ذلك هكذا:

my $result = "";
my $unproc = "";
foreach my $str (@string) {
    $unproc .= $str;
    my $n = normalize($form, $unproc);
    my($p, $u) = splitOnLastStarter($n);
    $result .= $p;
    $unproc  = $u;
}
$result .= $unproc;
# بدلًا من normalize($form, join('', @string))
"$processed = normalize_partial($form, $unprocessed)"
غلاف لتركيبة normalize() وsplitOnLastStarter(). لاحظ أن $unprocessed ستُعدَّل كأثر جانبي.

إذا كانت لديك مصفوفة @string ينبغي تسلسلها ثم تطبيعها، يمكنك فعل ذلك هكذا:

my $result = "";
my $unproc = "";
foreach my $str (@string) {
    $unproc .= $str;
    $result .= normalize_partial($form, $unproc);
}
$result .= $unproc;
# بدلًا من normalize($form, join('', @string))
"$processed = NFD_partial($unprocessed)"
يفعل مثل "normalize_partial('NFD', $unprocessed)". لاحظ أن $unprocessed ستُعدَّل كأثر جانبي.
"$processed = NFC_partial($unprocessed)"
يفعل مثل "normalize_partial('NFC', $unprocessed)". لاحظ أن $unprocessed ستُعدَّل كأثر جانبي.
"$processed = NFKD_partial($unprocessed)"
يفعل مثل "normalize_partial('NFKD', $unprocessed)". لاحظ أن $unprocessed ستُعدَّل كأثر جانبي.
"$processed = NFKC_partial($unprocessed)"
يفعل مثل "normalize_partial('NFKC', $unprocessed)". لاحظ أن $unprocessed ستُعدَّل كأثر جانبي.

التحقق السريع

(راجع الملحق 8، UAX #15؛ وlib/unicore/DerivedNormalizationProps.txt)

تتحقق الدوال التالية مما إذا كانت السلسلة في شكل التطبيع ذلك.

ستكون النتيجة المعادة واحدة مما يلي:

YES     السلسلة في شكل التطبيع ذلك.
NO      السلسلة ليست في شكل التطبيع ذلك.
MAYBE   مشكوك فيه. ربما نعم، ربما لا.
"$result = checkNFD($string)"
يعيد صحيحًا (1) إذا "YES"؛ false ("سلسلة فارغة") إذا "NO".
"$result = checkNFC($string)"
تُرجع true (1) إذا "YES"؛ false ("سلسلة فارغة") إذا "NO"؛ "undef" إذا "MAYBE".
"$result = checkNFKD($string)"
تُرجع true (1) إذا "YES"؛ false ("سلسلة فارغة") إذا "NO".
"$result = checkNFKC($string)"
تُرجع true (1) إذا "YES"؛ false ("سلسلة فارغة") إذا "NO"؛ "undef" إذا "MAYBE".
"$result = checkFCD($string)"
تُرجع true (1) إذا "YES"؛ false ("سلسلة فارغة") إذا "NO".
"$result = checkFCC($string)"
تُرجع true (1) إذا "YES"؛ false ("سلسلة فارغة") إذا "NO"؛ "undef" إذا "MAYBE".

ملاحظة: إذا لم تكن السلسلة في FCD، فيجب ألا تكون في FCC. لذا يجب أن تُرجع checkFCC($not_FCD_string) "NO".

"$result = check($form_name, $string)"
تُرجع true (1) إذا "YES"؛ false ("سلسلة فارغة") إذا "NO"؛ "undef" إذا "MAYBE".

بصفتها $form_name، يجب إعطاء أحد الأسماء التالية.

'C'  أو 'NFC'  لـ Normalization Form C  (UAX #15)
'D'  أو 'NFD'  لـ Normalization Form D  (UAX #15)
'KC' أو 'NFKC' لـ Normalization Form KC (UAX #15)
'KD' أو 'NFKD' لـ Normalization Form KD (UAX #15)
'FCD'          لـ "Fast C or D" Form  (UTN #5)
'FCC'          لـ "Fast C Contiguous" (UTN #5)

ملاحظة

في حالات NFD وNFKD وFCD، يجب أن تكون الإجابة إما "YES" أو "NO". قد تُرجع الإجابة "MAYBE" في حالات NFC وNFKC وFCC.

يجب أن تحتوي سلسلة "MAYBE" على حرف تركيبي واحد على الأقل أو ما شابه. على سبيل المثال، "COMBINING ACUTE ACCENT" لها خاصية MAYBE_NFC/MAYBE_NFKC.

كل من "checkNFC("A\N{COMBINING ACUTE ACCENT}")" و "checkNFC("B\N{COMBINING ACUTE ACCENT}")" ستُرجع "MAYBE". "A\N{COMBINING ACUTE ACCENT}" ليست في NFC (قيمتها NFC هي "\N{LATIN CAPITAL LETTER A WITH ACUTE}")، بينما "B\N{COMBINING ACUTE ACCENT}" في NFC.

إذا أردت التحقق بدقة، قارن السلسلة مع NFC/NFKC/FCC الخاصة بها.

if ($string eq NFC($string)) {
    # $string مطبَّعة بدقة في NFC;
} else {
    # $string غير مطبَّعة في NFC;
}
if ($string eq NFKC($string)) {
    # $string مطبَّعة بدقة في NFKC;
} else {
    # $string غير مطبَّعة في NFKC;
}

بيانات الأحرف

هذه الدوال هي واجهة لبيانات الأحرف المستخدمة داخليًا. إذا أردت فقط الحصول على أشكال تطبيع Unicode، فلست بحاجة لاستدعائها بنفسك.

"$canonical_decomposition = getCanon($code_point)"
إذا كان الحرف قابلًا للتحلل القانوني (بما في ذلك مقاطع Hangul)، فإنها تُرجع التحلل القانوني (الكامل) كسلسلة. وإلا فإنها تُرجع "undef".

ملاحظة: وفقًا لمعيار Unicode، فإن التحلل القانوني للحرف غير القابل للتحلل القانوني هو نفس الحرف نفسه.

"$compatibility_decomposition = getCompat($code_point)"
إذا كان الحرف قابلًا للتحلل التوافقي (بما في ذلك مقاطع Hangul)، فإنها تُرجع التحلل التوافقي (الكامل) كسلسلة. وإلا فإنها تُرجع "undef".

ملاحظة: وفقًا لمعيار Unicode، فإن التحلل التوافقي للحرف غير القابل للتحلل التوافقي هو نفس الحرف نفسه.

"$code_point_composite = getComposite($code_point_here, $code_point_next)"
إذا كان الحرفان هنا والتالي (كنقاط رمز) قابلين للتركيب (بما في ذلك Hangul Jamo/Syllables واستثناءات التركيب)، فإنها تُرجع نقطة رمز المركّب.

إذا لم يكونا قابلين للتركيب، فإنها تُرجع "undef".

"$combining_class = getCombinClass($code_point)"
تُرجع فئة التركيب (كعدد صحيح) للحرف.
"$may_be_composed_with_prev_char = isComp2nd($code_point)"
تُرجع قيمة منطقية حول ما إذا كان حرف نقطة الرمز المحددة قد يُركَّب مع الحرف السابق في تركيب معين (بما في ذلك تركيبات Hangul، ولكن باستثناء استثناءات التركيب والتحللات غير البادئة).
"$is_exclusion = isExclusion($code_point)"
تُرجع قيمة منطقية حول ما إذا كانت نقطة الرمز استثناء تركيب.
"$is_singleton = isSingleton($code_point)"
تُرجع قيمة منطقية حول ما إذا كانت نقطة الرمز مفردة
"$is_non_starter_decomposition = isNonStDecomp($code_point)"
تُرجع قيمة منطقية حول ما إذا كانت نقطة الرمز لديها تحلل غير بادئ.
"$is_Full_Composition_Exclusion = isComp_Ex($code_point)"
تُرجع قيمة منطقية للخاصية المشتقة Comp_Ex (Full_Composition_Exclusion). تُولَّد هذه الخاصية من استثناءات التركيب + المفردات + التحللات غير البادئة.
"$NFD_is_NO = isNFD_NO($code_point)"
تُرجع قيمة منطقية للخاصية المشتقة NFD_NO (NFD_Quick_Check=No).
"$NFC_is_NO = isNFC_NO($code_point)"
تُرجع قيمة منطقية للخاصية المشتقة NFC_NO (NFC_Quick_Check=No).
"$NFC_is_MAYBE = isNFC_MAYBE($code_point)"
تُرجع قيمة منطقية للخاصية المشتقة NFC_MAYBE (NFC_Quick_Check=Maybe).
"$NFKD_is_NO = isNFKD_NO($code_point)"
تُرجع قيمة منطقية للخاصية المشتقة NFKD_NO (NFKD_Quick_Check=No).
"$NFKC_is_NO = isNFKC_NO($code_point)"
تُرجع قيمة منطقية للخاصية المشتقة NFKC_NO (NFKC_Quick_Check=No).
"$NFKC_is_MAYBE = isNFKC_MAYBE($code_point)"
تُرجع قيمة منطقية للخاصية المشتقة NFKC_MAYBE (NFKC_Quick_Check=Maybe).

EXPORT

"NFC"، "NFD"، "NFKC"، "NFKD": افتراضيًا.

"normalize" وبعض الدوال الأخرى: عند الطلب.

تصحيح تعيين التحلل
في إصدارات Unicode الأقدم، قد يكون لعدد صغير من الأحرف (وكلها من الأيديوغرافات التوافقية CJK بقدر ما تم العثور عليها) تعيين تحلل خاطئ (انظر lib/unicore/NormalizationCorrections.txt). على أي حال، لن تشير هذه الوحدة إلى lib/unicore/NormalizationCorrections.txt ولن توفر أي إصدار محدد من التطبيع. لذلك قد تستخدم هذه الوحدة التي تعمل على perl أقدم مع قاعدة بيانات Unicode أقدم تعيين التحلل الخاطئ بشكل أعمى وفقًا لقاعدة بيانات Unicode.
التعريف المنقح للتركيب القانوني
في Unicode 4.1.0، تم تغيير التعريف D2 للتركيب القانوني (الذي يؤثر على NFC وNFKC) (انظر Public Review Issue #29 و UAX #15 الحديث). استخدمت هذه الوحدة التعريف الأحدث منذ الإصدار 0.07 (31 أكتوبر 2001). لن تدعم هذه الوحدة التطبيع وفقًا للتعريف الأقدم، حتى لو كان إصدار Unicode المنفَّذ بواسطة perl أقل من 4.1.0.
