upmpdcli.conf(5)

الحزمة: upmpdcli

التراخيص: LGPL-2.1-or-later

UPMPDCLI.CONF(5) Upmpdcli UPMPDCLI.CONF(5)

الاسم

upmpdcli.conf - توثيق upmpdcli.conf

الوصف

upmpdcli.conf هو ملف الإعدادات لـ upmpdcli(1). يجب الإشارة إلى موقعه للبرنامج إما من خلال متغير البيئة UPMPD_CONFIG أو من خلال خيار سطر الأوامر -c.

يحتوي كل سطر في ملف الإعدادات على اسم إعداد وقيمته، مثال:

logfilename = /tmp/upmpdcli.log

لدى جميع متغيرات الإعدادات تقريبًا قيم افتراضية مناسبة، ويشمل توزيع البرنامج ملفًا نموذجيًا، مثبتًا عادةً في /etc، حيث يتم سرد المتغيرات وتعليقها.

يجب عليك إلقاء نظرة على الدليل عبر الإنترنت لمزيد من المعلومات الكاملة: http://www.lesbonscomptes.com/upmpdcli/upmpdcli-manual.html

المعلمات

logfilename = fn
اسم ملف السجل. القيمة الافتراضية هي stderr. يمكن تحديد هذا أيضًا كـ -d logfilename.
loglevel = int
مستوى السجل. يمكن تحديده أيضًا كـ -l loglevel.
upnplogfilename = fn
اسم ملف السجل لمكتبة UPnP ذات المستوى المنخفض (libnpupnp). هذا السجل غير نشط إذا لم يتم تعيين هذا المتغير على الإطلاق. استخدم قيمة فارغة للرسائل لتذهب إلى stderr.
upnploglevel = int
مستوى السجل لمكتبة UPnP ذات المستوى المنخفض. لأسباب تاريخية، القيم المتطابقة هي أقل بواحد من القيم المكافئة لسجل upmpdcli. 4 هو إسهابي للغاية.
pkgdatadir = dfn
دليل تخزين البيانات الثابتة مثل XML وغيرها. القيمة الافتراضية: '$prefix/share/upmpdcli' (مثال: '/usr/share/upmpdcli').
pidfile = fn
اسم ملف القفل المستخدم لتخزين معرّف العملية الحالي. يُستخدم فقط إذا تم تشغيل العملية بواسطة المستخدم الجذر. يُستخدم بواسطة نظام التشغيل، ولتجنب تشغيل نسخ متعددة. السبب الوحيد لتغيير القيمة هو إذا كنت تريد بالفعل تشغيل عدة نسخ (غيّر cachedir أيضًا في هذه الحالة).
upnpiface = cstr
واجهة (واجهات) الشبكة لاستخدامها لـ UPnP. يمكن تركها فارغة لاستخدام الواجهة الأولى المناسبة التي تم العثور عليها، أو تعيينها على حرف "*" واحد لاستخدام جميع الواجهات. خلاف ذلك، قم بتعيينها إلى قائمة مفصولة بمسافات لأسماء الواجهات (استخدم علامات اقتباس مزدوجة إذا كانت الأسماء تحتوي على مسافات).
upnpip = string
عنوان IP V4 لاستخدامه لـ UPnP، بديل لاستخدام اسم واجهة.
upnpport = int
رقم المنفذ المستخدم لـ UPnP. libupnp/pupnp افتراضيًا يستخدم أول منفذ مجاني بعد 49152. لاحظ أن العملاء لا يحتاجون إلى معرفة القيمة، حيث يتم اكتشافها تلقائيًا.
useipv6 = bool
تمكين استخدام IPV6. افتراضيًا، نستخدم IPV4 فقط.
friendlyname = string
"اسم ودود" لـ Media Renderer. سيتم عرضه بواسطة معظم نقاط التحكم لتحديد Renderer. قد تعرض بعض نقاط التحكم OpenHome قيمة 'ohproductroom' بدلاً من ذلك.
avfriendlyname = string
اسم ودود محدد لـ UPnP/AV Media Renderer. الافتراضي هو إضافة "-av" إلى قيمة friendlyname.
upnpav = bool
تمكين خدمات UPnP AV (0/1). هذا مضبوط افتراضيًا، ولكنه قد يكون مفيدًا لتعطيله لتجنب ازدحام خيارات Renderer إذا كنت تستخدم OpenHome فقط.
openhome = bool
تمكين خدمات OpenHome (0/1). هذا افتراضيًا 1، نظرًا لأن قوائم انتظار OpenHome أفضل لمعظم الاستخدامات. يجب أن يكون هناك سبب قليل لتعطيله.
lumincompat = bool
تعطيل خدمة OpenHome Credentials. في وقت ما، كان هذا ضروريًا لتمكين Lumin من رؤية مثيلات upmpdcli: لسبب غير معروف، لم يكتشف Lumin 1.10 مثيلات upmpdcli مع تمكين Credentials.
ohinfotexttodata = bool
تخزين البيانات الوصفية في متغير OpenHome "metatext". من المفترض أن تحتوي خدمات OpenHome Info على بيانات مختلفة، معظمها للراديو، مع بيانات راديو دائمة (أيقونة، عنوان URL، اسم الراديو) محددة في متغير "Metadata"، ومعلومات حول العنوان الحالي المسجل في متغير "Metatext". سيؤدي تعيين هذا الخيار إلى تخزين upmpdcli لأي بيانات Metatext غير فارغة في متغير Metadata. سيساعد هذا بعض المشغلات على عرض غلاف الألبوم للعنوان الحالي (على سبيل المثال، Kazoo).
checkcontentformat = bool
التحقق من أن تنسيق الإدخال مدعوم. استخراج معلومات protocolinfo من البيانات الوصفية المدخلة والتحقق منها مقابل التنسيقات المدعومة لدينا. قم بتعيين هذا الخيار إلى 0 إذا كان نقطة التحكم أو خادم الوسائط يرسل بيانات صوتية جيدة مع بيانات وصفية سيئة.
iconpath = fn
المسار إلى أيقونة Renderer. سيتم عرض الصورة بواسطة نقاط التحكم التي تدعمها. يحتوي بروتوكول UPnP على أحكام لـ renderer لإرسال عنوان URL لأيقونة وصفية كجزء من وصف الجهاز. نظرًا لقيود *upmpdcli* الحالية (وربما الدائمة)، يجب أن يكون ملف الصورة ملف png بدقة 64x64 و 32 بت لكل بكسل. القيمة الافتراضية: '$pkgdatadir/icon.png'. سيتم قراءة الأيقونة مرة واحدة فقط، عند بدء تشغيل upmpdcli.
saveohcredentials = bool
حفظ معلمات تسجيل الدخول لخدمات البث على القرص. هذا يسمح بمشاركة كلمة المرور مع وكيل بث خادم وسائط upmpdcli، إذا كنت تستخدمه أيضًا. يتجنب الحاجة إلى إدخال كلمة المرور في ملف الإعدادات العادي. اعتمادًا على وضعك، قد ترى هذا كخطر أمني. القيمة الافتراضية هي true، لأنني لا أرى هذا كقضية حقيقية في الحالة العامة.
opensslcmd = fn
المسار إلى أمر openssl. خدمة OHCredentials تحتاج openssl 1.1.x أو أحدث. المسار إلى أمر openssl للاستخدام إذا كان النظام (في /usr/bin) قديمًا جدًا. غير معين افتراضيًا.
cachedir = dfn
الدليل المستخدم لتخزين البيانات المخزنة مؤقتًا. يُستخدم فقط لبيانات تعريف قوائم انتظار OpenHome في الوقت الحالي. القيمة الافتراضية هي ~/.cache/upmpdcli للمستخدمين العاديين أو /var/cache/upmpdcli عندما يتم تشغيل upmpdcli كجذر.
presentationhtml = fn
المسار إلى مستند HTML للعرض. يمكنك تغييره لاستبدال صفحة العرض الافتراضية. سيتم قراءة الصفحة مرة واحدة فقط، عند بدء تشغيل upmpdcli. لا يمكن استخدامه حاليًا لتحديثات الحالة (ولكن أعتقد أنه يمكنك وضع إعادة توجيه فيه، إلى شيء أكثر ديناميكية يخدمه خادم HTTP حقيقي). القيمة الافتراضية: '$pkgdatadir/presentation.html'.
enablel16 = bool
الإعلان عن دعم تنسيق L16. L16 هو تيار صوتي خطي خام 16 بت ولا يتضمن معلومات تعريف. كان معطلاً افتراضيًا، بسبب خطأ في mpd أدى إلى الكثير من الضوضاء. إنه الآن ممكّن افتراضيًا، ولكن يمكن تعطيله إذا لزم الأمر. يبدو أنه يتداخل أحيانًا بشكل سيء مع ميزة "cast to device" في win10، لذا قد تحاول تعطيله إذا واجهت مشاكل مع هذا.
mpdhost = string
المضيف الذي يعمل عليه MPD. الافتراضي هو localhost. يمكن تحديد هذا أيضًا كـ -h host
mpdport = int
منفذ IP الذي يستخدمه MPD. يمكن تحديده أيضًا كـ -p port. الافتراضي هو منفذ MPD العادي، 6600

هذا الموقع غير مرتبط بـ Arch Linux ولا تابع له بأي شكل من الأشكال. المحتوى هو ترجمة فقط لصفحات دليل Arch المنشورة على man.archlinux.org.

mpdpassword = string
كلمة مرور MPD. كلمة المرور للاتصال بـ MPD (مطلوبة فقط إذا كان الوصول بكلمة مرور ممكّنًا في ملف تكوين MPD).
mpdtimeoutms = int
مهلة اتصال MPD بالمللي ثانية. إذا لم تكن هذه القيمة صفرًا، فستنتهي مهلة الحوارات مع mpd بعد القيمة. سيسمح هذا لـ upmpdcli بالبقاء مستجيبًا (إرجاع أخطاء) إذا أصبح MPD عالقًا في أي وقت.
ownqueue = bool
يتم تعيينه إذا كنا نمتلك قائمة انتظار MPD. إذا تم تعيين هذا (مُمكّن افتراضيًا)، فإننا نمتلك قائمة انتظار MPD وسنقوم بمسحها بجرأة. يمكن أيضًا تحديده كـ -q 0|1.
onstart = fn
الأمر المراد تشغيله عند بدء التشغيل. حدد المسار الكامل للبرنامج، مثل /usr/bin/logger.
onplay = fn
الأمر المراد تشغيله عند تحويل حالة MPD إلى "PLAY". حدد المسار الكامل للبرنامج، مثل /usr/bin/logger.
onpause = fn
الأمر المراد تشغيله عند تحويل حالة MPD إلى "PAUSE". حدد المسار الكامل للبرنامج، مثل /usr/bin/logger.
onstop = fn
الأمر المراد تشغيله عند تحويل حالة MPD إلى "STOP". حدد المسار الكامل للبرنامج، مثل /usr/bin/logger.
onstandby = fn
الأمر المراد تشغيله عند استدعاء الإجراء setstandby. حدد المسار الكامل للبرنامج. يتم استدعاؤه مع وسيط واحد 0/1 لتغيير حالة الاستعداد (1 لتنشيط الاستعداد)، وبدون وسيط للاستعلام عن الحالة. في جميع الحالات، يجب طباعة حالة الاستعداد (0/1) إلى الإخراج القياسي قبل الخروج.
externalvolumecontrol = fn
استخدام أمر خارجي لإدارة مستوى صوت النظام (0/1). يتم استخدام هذا في الحالة التي لا يستطيع فيها MPD التحكم في مستوى الصوت، ولكن يمكن لأمر آخر ذلك، على سبيل المثال إذا كان لديك مضخم صوت خارجي يمكن تعيينه وقراءة مستوى الصوت من خلال البرامج النصية. إذا تم تعيينه، فسيتم استبدال استدعاءات MPD لتعيين وقراءة مستوى الصوت بتنفيذ 'onvolumechange' و 'getexternalvolume'
getexternalvolume = fn
الأمر المراد تشغيله لقراءة مستوى صوت النظام. يجب أن يكتب الأمر قيمة رقمية من 0-100 إلى الإخراج القياسي.
onvolumechange = fn
الأمر المراد تشغيله لتعيين مستوى الصوت. يستخدم عند تعيين 'externalvolumecontrol'. حدد المسار الكامل للبرنامج، والذي يتم استدعاؤه مع مستوى الصوت كوسيط أول، مثل /some/script 85.
avtautoplay = string
محاكاة أمر تشغيل تلقائيًا عند تعيين المسار. يقوم UPnP/AV SetAVTransportURI، المستخدم لتعيين المسار للتشغيل، عادةً بتغيير حالة النقل الحالية: يلزم أمر تشغيل صريح لبدء التشغيل إذا كانت حالة النقل متوقفة. سيؤدي تعيين هذه المعلمة إلى إنشاء أمر تشغيل بعد تلقي SetAVTransportURI. هذا مطلوب من بعض نقاط التحكم التي لا ترسل أمر التشغيل.
ohproductroom = string
اسم الغرفة التي يوجد بها المنتج. يتم تعيينه افتراضيًا على "الغرفة الرئيسية"، ويتم عرضه بدلاً من "الاسم الودي" بواسطة بعض نقاط التحكم. يستخدم ProductRoom لتجميع المنتج مع المنتجات الأخرى ذات الصلة في نفس الغرفة المادية (مثل مصدر مع مضخم صوت مسبق). يجب أن تشترك المنتجات المرتبطة ماديًا دائمًا في نفس اسم ProductRoom.
radiolist = fn
مسار إلى ملف خارجي بتعريفات الراديو. يمكن تعريف محطات الراديو في هذا الملف (ملف التكوين الرئيسي)، أو في ملف خارجي (أو كليهما). يمكن أن يؤدي استخدام ملف خارجي إلى تسهيل التوزيع على أجهزة متعددة، ويجعل الأمور أكثر وضوحًا وهو الآن الافتراضي. راجع الملف الموزع (الاسم أدناه) للحصول على معلومات حول كيفية تعريف محطة راديو. في حال كنت ترغب في تعريف أجهزة راديو في ملف التكوين الرئيسي، يجب أن تحدث في نهاية الملف (لأنها معرفة كأقسام فرعية، يمتد آخرها إلى نهاية الملف).
radioscripts = fn
دليل برامج النصوص لبيانات وصفية الراديو. مسار دليل حيث توجد البرامج النصية المستخدمة لجلب البيانات الوصفية للقناة. الافتراضي هو /usr/share/upmpdcli/radio_scripts. لاحظ أنه يتم البحث عن البرامج النصية دائمًا أولاً في المواقع المحددة بواسطة $PATH.
ohmanufacturername = string
اسم الشركة المصنعة.
ohmanufacturerinfo = string
معلومات الشركة المصنعة.
ohmanufacturerurl = string
عنوان URL لموقع الشركة المصنعة على الويب.
ohmanufacturerimageuri = string
عنوان URI لشعار الشركة المصنعة.
ohmodelname = string
اسم الطراز.
ohmodelinfo = string
معلومات الطراز.
ohmodelurl = string
عنوان URL لموقع الطراز على الويب.
ohmodelimageuri = string
عنوان URI لأيقونة الطراز.
ohproductname = string
اسم المنتج المرئي للمستخدم. افتراضيًا، يتم تعيين هذا على ModelName.
ohproductinfo = string
معلومات المنتج.
ohproducturl = string
عنوان URL لموقع المنتج على الويب. قد تكون هذه صفحة عرض UPnP.
ohproductimageuri = string
عنوان URI لصورة المنتج.
ohmetapersist = bool
حفظ البيانات الوصفية لقائمة التشغيل على القرص (0/1). هذا يسمح بالاستمرارية لمعلومات البيانات الوصفية عبر عمليات إعادة التشغيل، الافتراضي هو 1، ولا يوجد سبب لإيقاف تشغيله.
ohmetasleep = int
الحد الأدنى للفترة الفاصلة (بالثواني) بين عمليتي حفظ ذاكرة التخزين المؤقت. قد يؤدي زيادة هذا إلى تحسين سرعة تحميل قائمة التشغيل على جهاز بطيء. الافتراضي هو بدء حفظ جديد بمجرد الانتهاء من السابق (إذا تغيرت القائمة مرة أخرى في هذه الأثناء).
msfriendlyname = string
اسم سهل لخادم الوسائط (إذا تم تمكينه). الافتراضي هو إضافة "-mediaserver" إلى قيمة friendlyname.
msrootalias = string
معرف كائن اسم مستعار لجذر خادم الوسائط. استخدم الحاوية المحددة لخادم الوسائط كجذر. سيقوم نقطة التحكم بإظهار هذه الحاوية مباشرة عند الاتصال، مما يتجنب الحاجة إلى تخطي المستويات العليا إذا لم تستخدمها. (على سبيل المثال، قائمة المكونات الإضافية النشطة إذا كان هناك مكون إضافي واحد فقط). القيمة هي معرف كائن UPnP، والذي لسوء الحظ لا يمكن الحصول عليه بشكل عام من نقاط التحكم. ومع ذلك، فإن القيم المفيدة بشكل عام بسيطة: - `0$uprcl$` وصول مباشر إلى جذر الوسائط المحلية - `0$uprcl$folders` وصول مباشر إلى شجرة [مجلدات] الوسائط المحلية.

يجب أن يبدأ الاسم المستعار بـ `0` (وهو معرف الكائن الجذر الإلزامي لـ UPnP)، متبوعًا بـ `$pluginname$` حيث يمكن أن يكون _pluginname_ أي دليل فرعي لـ `/usr/share/upmpdcli/cdplugins` باستثناء `pycommon` (على سبيل المثال، `uprcl`، `qobuz`، `upradios`...)، والباقي سيعتمد على نوع المكون الإضافي.

plgmicrohttphost = string
اسم المضيف / عنوان IP المستخدم في عناوين URL الوكيل. الافتراضي هو استخدام نفس المضيف / العنوان مثل خادم UPnP. استخدام localhost/127.0.0.1 بدلاً من ذلك يفقد القدرة على البث إلى جهاز عرض عن بعد، ولكنه يسمح بقوائم تشغيل محمولة.
plgmicrohttpport = int
منفذ IP لخدمة HTTP المحلية لـ tidal/qobuz. تشير عناوين URL للمسارات التي يتم بثها من هذه الخدمات إلى خادم microhttpd محلي يعمل على هذا المنفذ (يستخدم إعادة التوجيه إلى عنوان URL الفعلي للخدمة عند البث فعليًا). الافتراضي هو 49149.
plgproxymethod = string
قرر ما إذا كنا سنقوم بالوكالة (نسخ: جلب/تقديم)، أو إعادة توجيه تدفقات خدمات البث. استخدام إعادة التوجيه أكثر كفاءة بكثير، ولكن الوكيل لديه مرفق لإعادة المحاولة عندما يتم إسقاط تدفق بواسطة الخدمة، وهو ما يحدث خاصة مع Qobuz.
msiconpath = fn
مسار أيقونة خادم الوسائط. سيتم عرض الصورة بواسطة نقاط التحكم التي تدعمها. بسبب القيود الحالية (وربما الدائمة) لـ *upmpdcli*، يجب أن يكون ملف الصورة ملف png 64x64 بـ 32 بت لكل بكسل. الافتراضي: '$pkgdatadir/icon.png'. سيتم قراءة الأيقونة مرة واحدة فقط، عند بدء تشغيل upmpdcli.
webserverdocumentroot = fn
الدليل الذي سيقدم منه خادم HTTP الداخلي الملفات مباشرة. هذا معطل افتراضيًا للتوافق مع الإصدارات الأقدم التي لم تقدم الملفات مباشرة من نظام الملفات. تطلب بعض المكونات الإضافية لخادم الوسائط تمكينه، على سبيل المثال لتقديم ملفات الأيقونات (عادةً من الأدلة تحت /usr/share/upmpdcli/www).
hrauser = string
