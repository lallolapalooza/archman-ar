subroutine cupmtr (character side, character uplo, character trans, integer m, integer n, complex, dimension( * ) ap, complex, dimension( * ) tau, complex, dimension( ldc, * ) c, integer ldc, complex, dimension( * ) work, integer info)

CUPMTR

الغرض:

!>
!> CUPMTR تستبدل المصفوفة C العامة المعقدة M-by-N بـ
!>
!>                 SIDE = 'L'     SIDE = 'R'
!> TRANS = 'N':      Q * C          C * Q
!> TRANS = 'C':      Q**H * C       C * Q**H
!>
!> حيث Q هي مصفوفة وحدوية معقدة من الدرجة nq، حيث nq = m إذا كان
!> SIDE = 'L' و nq = n إذا كان SIDE = 'R'. تُعرّف Q كناتج
!> لعوامل الانعكاس الأولية nq-1، كما تم إرجاعها بواسطة CHPTRD باستخدام
!> تخزين مضغوط:
!>
!> إذا كان UPLO = 'U'، فإن Q = H(nq-1) . . . H(2) H(1)؛
!>
!> إذا كان UPLO = 'L'، فإن Q = H(1) H(2) . . . H(nq-1).
!>

المعلمات

SIDE 
!>          SIDE هو CHARACTER*1
!>          = 'L': تطبيق Q أو Q**H من اليسار؛
!>          = 'R': تطبيق Q أو Q**H من اليمين.
!>

UPLO

!>          UPLO هو CHARACTER*1
!>          = 'U': تخزين المثلث العلوي المضغوط المستخدم في استدعاء
!>                 سابق لـ CHPTRD؛
!>          = 'L': تخزين المثلث السفلي المضغوط المستخدم في استدعاء
!>                 سابق لـ CHPTRD.
!>

TRANS

!>          TRANS هو CHARACTER*1
!>          = 'N':  عدم النقل، تطبيق Q؛
!>          = 'C':  النقل المترافق، تطبيق Q**H.
!>

M

!>          M هو INTEGER
!>          عدد صفوف المصفوفة C. M >= 0.
!>

N

!>          N هو INTEGER
!>          عدد أعمدة المصفوفة C. N >= 0.
!>

AP

!>          AP هو مصفوفة COMPLEX، البعد
!>                               (M*(M+1)/2) إذا كان SIDE = 'L'
!>                               (N*(N+1)/2) إذا كان SIDE = 'R'
!>          المتجهات التي تحدد عوامل الانعكاس الأولية، كما
!>          تم إرجاعها بواسطة CHPTRD. يتم تعديل AP بواسطة الروتين ولكن
!>          يتم استعادتها عند الخروج.
!>

TAU

!>          TAU هو مصفوفة COMPLEX، البعد (M-1) إذا كان SIDE = 'L'
!>                                     أو (N-1) إذا كان SIDE = 'R'
!>          يجب أن يحتوي TAU(i) على العامل العددي لعامل الانعكاس
!>          الأولي H(i)، كما تم إرجاعه بواسطة CHPTRD.
!>

C

!>          C هو مصفوفة COMPLEX، البعد (LDC,N)
!>          عند الإدخال، المصفوفة C ذات البعد M-by-N.
!>          عند الخروج، يتم استبدال C بـ Q*C أو Q**H*C أو C*Q**H أو C*Q.
!>

LDC

!>          LDC هو INTEGER
!>          البعد الرائد للمصفوفة C. LDC >= max(1,M).
!>

WORK

!>          WORK هي مصفوفة COMPLEX، البعد
!>                                   (N) إذا كان SIDE = 'L'
!>                                   (M) إذا كان SIDE = 'R'
!>

INFO

!>          INFO هو INTEGER
!>          = 0:  انتهاء ناجح
!>          < 0:  إذا كان INFO = -i، فإن المعلمة i كانت قيمة غير قانونية
!>
