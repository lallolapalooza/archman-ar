unistd.h(0p) دليل مبرمج POSIX unistd.h(0p)

مقدمة

صفحة الدليل هذه هي جزء من دليل مبرمج POSIX. قد يختلف تطبيق هذا الواجهة في لينكس (راجع صفحة دليل لينكس المقابلة لمعرفة تفاصيل سلوك لينكس)، أو قد لا تكون الواجهة مطبقة في لينكس.

الاسم

unistd.h — الثوابت والأنواع الرمزية القياسية

الملخص

#include <unistd.h>

الوصف

يعرّف الرأس <unistd.h> ثوابت وأنواع رمزية متنوعة، ويعلن عن وظائف متنوعة. القيم الفعلية للثوابت غير محددة إلا كما هو موضح. محتويات هذا الرأس معروضة أدناه.

وحدات اختبار الإصدار

يجب أن يقوم الرأس <unistd.h> بتعريف الثوابت الرمزية التالية. يجب أن تكون القيم مناسبة للاستخدام في توجيهات المعالجة المسبقة #if.

_POSIX_VERSION

قيمة عدد صحيح تشير إلى إصدار هذا المعيار (ربط اللغة C) الذي يتوافق معه التنفيذ. بالنسبة للتطبيقات التي تتوافق مع POSIX.1-2008، يجب أن تكون القيمة 200809L.
_POSIX2_VERSION

قيمة عدد صحيح تشير إلى إصدار مجلد Shell والأدوات في POSIX.1 الذي يتوافق معه التنفيذ. بالنسبة للتطبيقات التي تتوافق مع POSIX.1-2008، يجب أن تكون القيمة 200809L. بالنسبة لتطبيقات الملفات التعريفية التي تعرّف _POSIX_SUBPROFILE (انظر القسم 2.1.5.1، اعتبارات الملفات التعريفية الفرعية) في <unistd.h>، قد يتم ترك _POSIX2_VERSION غير معرف أو تعريفه بالقيمة -1 للإشارة إلى أن مجلد Shell والأدوات في POSIX.1 غير مدعوم. في هذه الحالة، يجب أن يعيد استدعاء sysconf(_SC_2_VERSION) إما 200809L أو -1 للإشارة إلى أن مجلد Shell والأدوات في POSIX.1 مدعوم أو غير مدعوم، على التوالي، في وقت التشغيل.

يجب أن يقوم الرأس <unistd.h> بتعريف الثابت الرمزي التالي فقط إذا كان التنفيذ يدعم خيار XSI؛ انظر القسم 2.1.4، توافق XSI. إذا تم تعريفه، يجب أن تكون قيمته مناسبة للاستخدام في توجيهات المعالجة المسبقة #if.

_XOPEN_VERSION

قيمة عدد صحيح تشير إلى إصدار دليل X/Open للنقل الذي يتوافق معه التنفيذ. يجب أن تكون القيمة 700.

ثوابت الخيارات ومجموعات الخيارات

الثوابت الرمزية التالية، إذا تم تعريفها في <unistd.h>، يجب أن تكون قيمتها -1، 0، أو أكبر، إلا إذا تم تحديد خلاف ذلك أدناه. بالنسبة لتطبيقات الملفات التعريفية التي تعرّف _POSIX_SUBPROFILE (انظر القسم 2.1.5.1، اعتبارات الملفات التعريفية الفرعية) في <unistd.h>، لا يلزم تعريف الثوابت الموصوفة أدناه بأن لها دائمًا قيمة أكبر من الصفر، وإذا تم تعريفها، فقد تكون قيمتها -1، 0، أو أكبر. يجب أن تكون القيم مناسبة للاستخدام في توجيهات المعالجة المسبقة #if.

إذا لم يتم تعريف ثابت رمزي أو تم تعريفه بالقيمة -1، فإن الخيار غير مدعوم للتجميع. إذا تم تعريفه بقيمة أكبر من الصفر، فسيتم دائمًا دعم الخيار عند تشغيل التطبيق. إذا تم تعريفه بالقيمة صفر، فسيتم دعم الخيار للتجميع وقد يتم دعمه أو لا يتم دعمه في وقت التشغيل. انظر القسم 2.1.6، الخيارات لمزيد من المعلومات حول متطلبات التوافق لهذه الفئات الثلاث من الدعم.

استخدام التطبيق

يمكن للمطورين استخدام وحدات الماكرو الخاصة بالتحقق من الإصدار مثل _POSIX_VERSION، على سبيل المثال:

#if _POSIX_VERSION >= 200112L
/* استخدم الوظيفة الأحدث التي تتعامل مع الملفات الكبيرة. */
off_t pos=ftello(fp);
#else
/* إما أن هذا إصدار قديم من POSIX، أو أن _POSIX_VERSION غير معرف،
   لذا استخدم الوظيفة التقليدية. */
long pos=ftell(fp);
#endif

يمكن تحديد الإصدارات الأقدم من POSIX.1-2008 ومن مواصفات UNIX الوحيدة بواسطة وحدات الماكرو التالية:

معيار POSIX.1-1988

_POSIX_VERSION==198808L
معيار POSIX.1-1990

_POSIX_VERSION==199009L
معيار ISO POSIX-1:1996

_POSIX_VERSION==199506L
مواصفات UNIX الوحيدة، الإصدار 1

_XOPEN_UNIX و _XOPEN_VERSION==4
مواصفات UNIX الوحيدة، الإصدار 2

_XOPEN_UNIX و _XOPEN_VERSION==500
ISO POSIX-1:2001 ومواصفات UNIX الوحيدة، الإصدار 3

_POSIX_VERSION==200112L، بالإضافة إلى (إذا كان خيار XSI مدعومًا) _XOPEN_UNIX و _XOPEN_VERSION==600

لا يقدم POSIX.1-2008 أي محاولة لتعريف التفاعل الثنائي للتطبيقات مع نظام التشغيل الأساسي. ومع ذلك، قد يجد مطورو التطبيقات أنه من المفيد الاستعلام عن _SC_VERSION في وقت التشغيل عبر sysconf() لتحديد ما إذا كان الإصدار الحالي لنظام التشغيل يدعم الوظائف اللازمة كما في جزء البرنامج التالي:

if (sysconf(_SC_VERSION) < 200809L) {
    fprintf(stderr, "يلزم نظام POSIX.1-2008، جار الإنهاء \n");
    exit(1);
}

يجب على التطبيقات الجديدة عدم استخدام _XOPEN_SHM أو _XOPEN_ENH_I18N.

الأساس المنطقي

مع تطور POSIX.1-2008، أصبحت بعض الخيارات موحدة بما فيه الكفاية لدرجة أنه تم استنتاج أن مجرد طلب أحد خيارات الاختيار كان أبسط من الاحتفاظ بالخيار. ومع ذلك، للتوافق مع الإصدارات السابقة، يتم الاحتفاظ بعلامات الخيار (مع قيم الثوابت المطلوبة).

وحدات الماكرو لاختبار الإصدار

نظر مطورو المعايير في تغيير تعريف _POSIX_VERSION وإزالة _SC_VERSION من مواصفات sysconf() نظرًا لأن فائدة التطبيق اعتُبرت ضئيلة من قبل البعض، ولأن تنفيذ الوظائف قد يكون إشكاليًا.

في حين أن المثال الذي يستخدم _SC_VERSION في قسم استخدام التطبيق لا يوفر أكبر قدر من الفائدة المتخيلة لمطور التطبيق أو المستخدم، إلا أنه من الأفضل القول بأنه أفضل من ملف core أو أي نتيجة غامضة أخرى بالمثل. (من الممكن أيضًا للتطبيقات تشفير وتعرف الثنائيات المترجمة في بيئات متوافقة مختلفة مع POSIX.1، وتعديل دلالات النظام الأساسي ليتوافق مع توقعات التطبيق.) للأسباب الموضحة في الفقرات السابقة وفي قسم استخدام التطبيق، اختار مطورو المعايير الاحتفاظ بوظائف _POSIX_VERSION و _SC_VERSION.

الثوابت الرمزية لوقت التجميع للخيارات على مستوى النظام

يشمل POSIX.1-2008 الدعم في مجالات معينة للسياسة المعتمدة حديثًا التي تحكم الخيارات والوصلات.

توفر هذه السياسة مرونة للتطبيقات في كيفية دعمها للخيارات. كما تحدد كيفية تكييف التطبيقات المتوافقة مع التطبيقات المختلفة التي تدعم مجموعات مختلفة من الخيارات. ويسمح بما يلي:

1.
إذا لم يكن للتطبيق أي اهتمام بدعم خيار ما، فلا داعي له لتوفير أي شيء مرتبط بهذا الخيار بخلاف الإعلان عن عدم دعمه.
2.
يمكن للتطبيق دعم نسخة جزئية أو غير متوافقة من خيار (كملحق غير قياسي) طالما أنه لا يدعي دعم الخيار.
3.
يمكن للتطبيق تحديد ما إذا كان الخيار مدعومًا. يجب على التطبيق المتوافق تمامًا التحقق من آلية الإعلان هذه قبل استخدام أي شيء مرتبط بالخيار لأول مرة.

هناك آثار مهمة لهذه السياسة. لا يمكن لـ POSIX.1-2008 تحديد سلوك الواجهات المرتبطة بخيار ما عندما لا يدعي التطبيق دعم الخيار. على وجه الخصوص، لا يمكن أن يتطلب أن تفشل وظيفة مرتبطة بخيار غير مدعوم إذا لم تؤد كما هو محدد. ومع ذلك، فإن هذه السياسة لا تمنع معيارًا من طلب وجود بعض الوظائف دائمًا، ولكنها تتطلب أن تفشل دائمًا في بعض التطبيقات. على سبيل المثال، تعتبر وظيفة setpgid() في معيار POSIX.1-1990 مناسبة.

تتضمن معايير POSIX خيارات متنوعة، ويدل دعم الواجهة C للخيار على أن التطبيق يجب أن يوفر أنواع البيانات وواجهات الوظائف. يجب أن يكون التطبيق قادرًا على اكتشاف ما إذا كان التطبيق يدعم كل خيار.

يجب على أي تطبيق النظر في الحالات الثلاث التالية لكل خيار:

1.
الخيار غير مدعوم أبدًا.

يعلن التطبيق في وقت التجميع أنه لن يتم دعم الخيار أبدًا. في هذه الحالة، لا يلزم أن يوفر التطبيق أيًا من أنواع البيانات أو واجهات الوظائف المقدمة كجزء من الخيار فقط. قد يوفر التطبيق أنواع بيانات ووظائف مشابهة لتلك المحددة بواسطة POSIX.1-2008، ولكن لا يوجد ضمان لسلوك معين.

2.
الخيار مدعوم دائمًا.

يعلن التطبيق في وقت التجميع أنه سيتم دعم الخيار دائمًا. في هذه الحالة، يجب أن تكون جميع أنواع البيانات وواجهات الوظائف متاحة وأن تعمل كما هو محدد.

3.
الخيار قد يتم دعمه أو قد لا يتم دعمه.

قد لا توفر بعض التطبيقات آلية لتحديد دعم الخيارات في وقت التجميع. بالإضافة إلى ذلك، قد يكون التطبيق غير قادر أو غير راغب في تحديد الدعم أو عدم الدعم في وقت التجميع. في كلتا الحالتين، يجب أن يكون أي تطبيق قد يستخدم الخيار في وقت التشغيل قادرًا على التجميع والتنفيذ. يجب أن يوفر التطبيق، في وقت التجميع، جميع أنواع البيانات وواجهات الوظائف اللازمة للسماح بذلك. في هذا الوضع، يجب أن تكون هناك آلية تسمح للتطبيق بالاستعلام، في وقت التشغيل، عما إذا كان الخيار مدعومًا. إذا حاول التطبيق استخدام الخيار عندما لا يكون مدعومًا، فإن النتيجة غير محددة ما لم يتم تحديدها صراحةً بخلاف ذلك في POSIX.1-2008.
