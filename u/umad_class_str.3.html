UMAD_CLASS_STR(3) دليل مبرمج OpenIB UMAD_CLASS_STR(3)

الاسم

umad_*_str - فئة الدوال لإرجاع تمثيلات نصية للتعدادات

النبذة

#include <infiniband/umad_str.h>

const char * umad_class_str(uint8_t mgmt_class)
const char * umad_method_str(uint8_t mgmt_class, uint8_t method)
const char * umad_attribute_str(uint8_t mgmt_class, be16_t attr_id)
const char * umad_common_mad_status_str(be16_t status)
const char * umad_sa_mad_status_str(be16_t status)

الوصف

const char * umad_class_str(uint8_t mgmt_class) إرجاع القيمة النصية لتعداد فئة الإدارة.

const char * umad_method_str(uint8_t mgmt_class, uint8_t method) إرجاع القيمة النصية للطريقة لفئة الإدارة المحددة.

const char * umad_attribute_str(uint8_t mgmt_class, be16_t attr_id) إرجاع القيمة النصية للخاصية المحددة في attr_id بناءً على فئة الإدارة المحددة.

const char * umad_common_mad_status_str(be16_t status) إرجاع القيمة النصية لقيم حالة MAD الشائعة.

const char * umad_sa_mad_status_str(be16_t status) إرجاع القيمة النصية لقيم حالة SA MAD.

ملاحظة: لا يتم دعم جميع الفئات.

قيمة الإرجاع

ترجع تمثيلات نصية للحقول المحددة.
