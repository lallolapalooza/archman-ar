uudoc(1) دليل الأوامر العامة uudoc(1)

الاسم

uudoc - نسخ SOURCE إلى DEST أو مصادر متعددة إلى الدليل الموجود، مع تحديد أوضاع الأذونات والمالك/المجموعة

المُلَخَّص

uudoc [--backup] [-b ] [-c ] [-C|--compare] [-d|--directory] [-D ] [-g|--group] [-m|--mode] [-o|--owner] [-p|--preserve-timestamps] [-s|--strip] [--strip-program] [-S|--suffix] [-t|--target-directory] [-T|--no-target-directory] [-v|--verbose] [-P|--preserve-context] [-Z ] [--context] [-h|--help] [-V|--version] [ملفات]

الوصف

نسخ SOURCE إلى DEST أو مصادر متعددة إلى الدليل الموجود، مع تحديد أوضاع الأذونات والمالك/المجموعة

الخيارات

--backup[=<CONTROL>]
إنشاء نسخة احتياطية من كل ملف وجهة موجود
-b
مثل --backup ولكن لا يقبل وسيطة
-c
تم التجاهل
-C, --compare
مقارنة كل زوج من ملفات المصدر والوجهة، وفي بعض الحالات، عدم تعديل الوجهة على الإطلاق
-d, --directory
معاملة جميع الوسائط كأسماء أدلة. إنشاء جميع مكونات الأدلة المحددة
-D
إنشاء جميع المكونات الرئيسية لـ DEST باستثناء الأخير، ثم نسخ SOURCE إلى DEST
-g, --group <GROUP>
تعيين ملكية المجموعة، بدلاً من المجموعة الحالية للعملية
-m, --mode <MODE>
تعيين وضع الإذن (كما في chmod)، بدلاً من rwxr-xr-x
-o, --owner <OWNER>
تعيين الملكية (للمستخدم الخارق فقط)
-p, --preserve-timestamps
تطبيق أوقات الوصول/التعديل لملفات SOURCE على ملفات الوجهة المقابلة
-s, --strip
إزالة جداول الرموز (لا يوجد إجراء في Windows)
--strip-program <PROGRAM>
البرنامج المستخدم لإزالة الملفات الثنائية (لا يوجد إجراء في Windows)
-S, --suffix <SUFFIX>
تجاوز لاحقة النسخ الاحتياطي المعتادة
-t, --target-directory <DIRECTORY>
نقل جميع وسائط SOURCE إلى DIRECTORY
-T, --no-target-directory
معاملة DEST كملف عادي
-v, --verbose
شرح ما يتم القيام به
-P, --preserve-context
الحفاظ على السياق الأمني
-Z
تعيين السياق الأمني لـ SELinux لملف الوجهة وكل دليل تم إنشاؤه إلى النوع الافتراضي
--context [<CONTEXT>]
تعيين السياق الأمني للملفات والأدلة
-h, --help
طباعة المساعدة
-V, --version
طباعة الإصدار
[files]
