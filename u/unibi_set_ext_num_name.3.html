unibi_set_ext_num_name(3)

الحزمة: unibilium

التراخيص: LGPL3

doc::unibi_set_ext_num_name(3) unibilium doc::unibi_set_ext_num_name(3)

الاسم

unibi_get_ext_bool_name, unibi_set_ext_bool_name, unibi_get_ext_num_name, unibi_set_ext_num_name, unibi_get_ext_str_name, unibi_set_ext_str_name - الوصول إلى أسماء القدرات الموسعة لكائن الطرفية

الملخص

#include <unibilium.h>
const char *unibi_get_ext_bool_name(const unibi_term *ut, size_t i);
const char *unibi_get_ext_num_name(const unibi_term *ut, size_t i);
const char *unibi_get_ext_str_name(const unibi_term *ut, size_t i);

void unibi_set_ext_bool_name(unibi_term *ut, size_t i, const char *s);
void unibi_set_ext_num_name(unibi_term *ut, size_t i, const char *s);
void unibi_set_ext_str_name(unibi_term *ut, size_t i, const char *s);

الوصف

الحصول/تعيين أسماء القدرات الموسعة المنطقية والرقمية والنصية. i هو فهرس القدرة الموسعة المطلوب التصرف فيها؛ يجب أن يكون أقل من unibi_count_ext_bool(ut)، unibi_count_ext_num(ut)، أو unibi_count_ext_str(ut)، على التوالي.

لاحظ أن "unibi_set_ext_bool_name"، "unibi_set_ext_num_name"، و "unibi_set_ext_str_name" تخزن ببساطة المؤشر الذي تم تمريره لها؛ لن تقوم بتحرير s أو عمل نسخة من السلسلة.
