UPDATE(7)

الحزمة: postgresql

التراخيص: PostgreSQL

NAME

UPDATE — استرداد أو تحديث صفوف جدول

SYNOPSIS

UPDATE [ ONLY ] target_table [ FROM alias ] SET { column { = | = DEFAULT } expression | ( column [, ...] ) = ( expression [, ...] ) } [, ...] [ WHERE condition ] [ RETURNING * | output_expression [, ...] ]

UPDATE target_table SET { column { = | = DEFAULT } expression | ( column [, ...] ) = ( expression [, ...] ) } [, ...] [ FROM from_list ] [ WHERE condition ] [ RETURNING * | output_expression [, ...] ]

DESCRIPTION

UPDATE هو أمر تعديل جدول لتغيير الصفوف الموجودة.

تستخدم الصيغة الأولى، التي تفتقر إلى عبارة FROM، الجدول الهدف لتحديد الصفوف المراد تحديثها. تحدد عبارة WHERE الصفوف التي سيتم تحديثها. إذا تم حذف عبارة WHERE، فسيتم تحديث جميع الصفوف الموجودة في الجدول.

تحدد الصيغة الثانية، التي تتضمن عبارة FROM، الصفوف المراد تحديثها بناءً على نتائج الاستعلام الذي يشتمل على جداول إضافية. هذا يسمح بتحديث صفوف في جدول واحد بناءً على بيانات من جداول أخرى.

يمكن استخدام عبارة RETURNING لتحديد القيم التي سيتم إرجاعها من UPDATE. يمكن أن تتضمن هذه القيم أي أعمدة من الجدول الهدف أو تعبيرات محسوبة.

PARAMETERS

target_table
هو الجدول الذي سيتم تحديث صفوفه.
alias
اسم مستعار للجدول الهدف، يستخدم عند الحاجة إلى الإشارة إلى الجدول الهدف بطريقة معينة، مثل في عبارة WHERE.
column
هو اسم عمود في الجدول الهدف.
expression
هو تعبير يحسب القيمة الجديدة للعمود. يمكن أن يتضمن هذا التعبير قيمًا ثابتة، أو أسماء أعمدة أخرى، أو استدعاءات للدوال.
DEFAULT
يعيد تعيين العمود إلى قيمته الافتراضية.
condition
هي عبارة تحدد الصفوف التي سيتم تحديثها. إذا تم حذفها، فسيتم تحديث جميع الصفوف.
from_list
هي قائمة بالجداول الإضافية التي يمكن استخدامها في عبارة WHERE لتحديد الصفوف المراد تحديثها.
output_expression
هو تعبير يتم حسابه وإرجاعه لكل صف تم تحديثه. يمكن أن يكون هذا اسم عمود، أو تعبيرًا محسوبًا، أو حتى قيمة ثابتة.

OUTPUTS

إذا تم استخدام عبارة RETURNING، فإن UPDATE تُرجع صفًا واحدًا لكل صف تم تحديثه. تحتوي كل صف مُرجع على القيم المحددة في عبارة RETURNING.

NOTES

عند تحديث صف تم نقله بواسطة عملية أخرى، قد يتلقى الإصدار القديم من الصف خطأ serialization failure (SQLSTATE '40001'). يجب أن تكون التطبيقات قادرة على معالجة هذا الخطأ وإعادة المحاولة.

لا يمكن نقل الصفوف من قسم جدول خارجي إلى قسم آخر.

يفشل نقل الصف من قسم إلى آخر إذا كان هناك مفتاح خارجي يشير مباشرة إلى سلف للقسم المصدر، وهذا السلف ليس هو نفسه الذي تم ذكره في استعلام UPDATE.

EXAMPLES

تغيير كلمة "Drama" إلى "Dramatic" في العمود kind في جدول films:

UPDATE films SET kind = 'Dramatic' WHERE kind = 'Drama';

ضبط إدخالات درجة الحرارة وإعادة تعيين قيمة هطول الأمطار إلى قيمتها الافتراضية في صف واحد من جدول weather:

UPDATE weather SET temp_lo = temp_lo+1, temp_hi = temp_lo+15, prcp = DEFAULT
  WHERE city = 'San Francisco' AND date = '2003-07-03';

أداء نفس العملية وإرجاع الإدخالات المحدثة، وقيمة هطول الأمطار القديمة:

UPDATE weather SET temp_lo = temp_lo+1, temp_hi = temp_lo+15, prcp = DEFAULT
  WHERE city = 'San Francisco' AND date = '2003-07-03'
  RETURNING temp_lo, temp_hi, prcp, old.prcp AS old_prcp;

استخدام صيغة قائمة الأعمدة البديلة لأداء نفس التحديث:

UPDATE weather SET (temp_lo, temp_hi, prcp) = (temp_lo+1, temp_lo+15, DEFAULT)
  WHERE city = 'San Francisco' AND date = '2003-07-03';

زيادة عدد المبيعات لمدير حساب شركة Acme Corporation، باستخدام صيغة عبارة FROM:

UPDATE employees SET sales_count = sales_count + 1 FROM accounts
  WHERE accounts.name = 'Acme Corporation'
  AND employees.id = accounts.sales_person;

أداء نفس العملية، باستخدام استعلام فرعي في عبارة WHERE:

UPDATE employees SET sales_count = sales_count + 1 WHERE id =
  (SELECT sales_person FROM accounts WHERE name = 'Acme Corporation');

تحديث أسماء جهات الاتصال في جدول accounts لمطابقة مندوبي المبيعات المعينين حاليًا:

UPDATE accounts SET (contact_first_name, contact_last_name) =
    (SELECT first_name, last_name FROM employees
     WHERE employees.id = accounts.sales_person);

يمكن تحقيق نتيجة مماثلة باستخدام الربط:

UPDATE accounts SET contact_first_name = first_name,
                    contact_last_name = last_name
  FROM employees WHERE employees.id = accounts.sales_person;

ومع ذلك، قد يعطي الاستعلام الثاني نتائج غير متوقعة إذا كان employees.id ليس مفتاحًا فريدًا، بينما يضمن الاستعلام الأول حدوث خطأ إذا كانت هناك مطابقات متعددة لـ id. أيضًا، إذا لم تكن هناك مطابقة لإدخال accounts.sales_person معين، فسيقوم الاستعلام الأول بتعيين حقول الأسماء المقابلة إلى NULL، بينما لن يقوم الاستعلام الثاني بتحديث هذا الصف على الإطلاق.

تحديث الإحصائيات في جدول ملخص لمطابقة البيانات الحالية:

UPDATE summary s SET (sum_x, sum_y, avg_x, avg_y) =
    (SELECT sum(x), sum(y), avg(x), avg(y) FROM data d
     WHERE d.group_id = s.group_id);

محاولة إدراج عنصر مخزون جديد مع كمية المخزون. إذا كان العنصر موجودًا بالفعل، فقم بتحديث عدد المخزون للعنصر الموجود بدلاً من ذلك. للقيام بذلك دون فشل المعاملة بأكملها، استخدم نقاط الحفظ:

BEGIN;
-- عمليات أخرى
SAVEPOINT sp1;
INSERT INTO wines VALUES('Chateau Lafite 2003', '24');
-- افترض أن ما سبق يفشل بسبب انتهاك مفتاح فريد،
-- لذلك الآن نقوم بتنفيذ هذه الأوامر:
ROLLBACK TO sp1;
UPDATE wines SET stock = stock + 24 WHERE winename = 'Chateau Lafite 2003';
-- استمر في العمليات الأخرى، وفي النهاية
COMMIT;

تغيير العمود kind في جدول films في الصف الذي يوجد فيه المؤشر c_films حاليًا:

UPDATE films SET kind = 'Dramatic' WHERE CURRENT OF c_films;

يمكن أن يكون للتحديثات التي تؤثر على العديد من الصفوف آثار سلبية على أداء النظام، مثل تضخم الجدول، وزيادة تأخير النسخ المتماثل، وزيادة تنازع الأقفال. في مثل هذه الحالات، قد يكون من المنطقي إجراء العملية على دفعات أصغر، ربما مع عملية VACUUM على الجدول بين الدفعات. بينما لا يوجد بند LIMIT لـ UPDATE، فمن الممكن الحصول على تأثير مماثل من خلال استخدام تعبير جدول مشترك (Common Table Expression) وربط ذاتي. مع طريقة الوصول القياسية إلى جدول PostgreSQL، يكون الربط الذاتي على العمود النظامي ctid فعالًا جدًا:

WITH exceeded_max_retries AS (
  SELECT w.ctid FROM work_item AS w
    WHERE w.status = 'active' AND w.num_retries > 10
    ORDER BY w.retry_timestamp
    FOR UPDATE
    LIMIT 5000
)
UPDATE work_item SET status = 'failed'
  FROM exceeded_max_retries AS emr
  WHERE work_item.ctid = emr.ctid;

سيحتاج هذا الأمر إلى التكرار حتى لا تتبقى صفوف لتحديثها. يسمح استخدام عبارة ORDER BY للأمر بتحديد أولوية الصفوف التي سيتم تحديثها؛ يمكنه أيضًا منع التعارض (deadlock) مع عمليات التحديث الأخرى إذا استخدمت نفس الترتيب. إذا كان تنازع الأقفال مصدر قلق، فيمكن إضافة SKIP LOCKED إلى CTE لمنع الأوامر المتعددة من تحديث نفس الصف. ومع ذلك، يلزم عندئذٍ إجراء UPDATE نهائي بدون SKIP LOCKED أو LIMIT لضمان عدم إغفال أي صفوف مطابقة.

COMPATIBILITY

يتوافق هذا الأمر مع معيار SQL، باستثناء أن عبارات FROM و RETURNING هي امتدادات PostgreSQL، وكذلك القدرة على استخدام WITH مع UPDATE.

تقدم بعض أنظمة قواعد البيانات الأخرى خيار FROM حيث يُفترض إدراج الجدول الهدف مرة أخرى ضمن FROM. هذه ليست طريقة تفسير PostgreSQL لـ FROM. كن حذرًا عند نقل التطبيقات التي تستخدم هذا الامتداد.

وفقًا للمعيار، يمكن أن يكون المصدر لقائمة فرعية محاطة بأقواس من أسماء أعمدة الهدف أي تعبير ذي قيمة صفية ينتج عنه العدد الصحيح من الأعمدة. يسمح PostgreSQL فقط بأن يكون المصدر عبارة بناء صف أو استعلام فرعي. يمكن تحديد قيمة محدثة لعمود فردي كـ DEFAULT في حالة بناء الصف، ولكن ليس داخل استعلام فرعي.
