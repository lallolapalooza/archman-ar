التوثيق

الوحدة LargeFile
: sig end

عمليات على الملفات الكبيرة. توفر هذه الوحدة الفرعية نسخًا بـ 64 بت للدوال UnixLabels.LargeFile.lseek (لتحديد موضع واصف الملف)، UnixLabels.LargeFile.truncate و UnixLabels.LargeFile.ftruncate (لتغيير حجم الملف)، و UnixLabels.LargeFile.stat ، UnixLabels.LargeFile.lstat و UnixLabels.LargeFile.fstat (للحصول على معلومات حول الملفات). تمثل هذه الدوال البديلة المواضع والأحجام بأعداد صحيحة بـ 64 بت (النوع int64) بدلاً من الأعداد الصحيحة العادية (النوع int)، مما يسمح بالعمل على الملفات التي تتجاوز أحجامها max_int.

type stats = Unix.LargeFile.stats = {
st_dev : int ; (* رقم الجهاز
*)
st_ino : int ; (* رقم الـ inode
*)
st_kind : UnixLabels.file_kind ; (* نوع الملف
*)
st_perm : UnixLabels.file_perm ; (* حقوق الوصول
*)
st_nlink : int ; (* عدد الروابط
*)
st_uid : int ; (* معرف المستخدم المالك
*)
st_gid : int ; (* معرف المجموعة للملف
*)
st_rdev : int ; (* معرف الجهاز (إذا كان ملفًا خاصًا)
*)
st_size : int64 ; (* الحجم بالبايت
*)
st_atime : float ; (* وقت آخر وصول
*)
st_mtime : float ; (* وقت آخر تعديل
*)
st_ctime : float ; (* وقت آخر تغيير في الحالة
*)
}
