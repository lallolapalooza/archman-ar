Uil(3)

الحزمة: openmotif

التراخيص: LGPL-2.1-or-later

Uil(استدعاء مكتبة) Uil(استدعاء مكتبة)

الاسم

Uil — يستدعي مترجم UIL من داخل تطبيق

المُلَخَّص

#include <uil/UilDef.h>

Uil_status_type Uil(
Uil_command_type *command_desc,
Uil_compile_desc_type **compile_desc,
Uil_continue_type (*message_cb) (),
char *message_data,
Uil_continue_type (*status_cb) (),
char *status_data);

الوصف

توفر الدالة Uil نقطة دخول قابلة للاستدعاء لمترجم UIL. يمكن استخدام واجهة Uil القابلة للاستدعاء لمعالجة ملف مصدر UIL وإنشاء ملفات UID، بالإضافة إلى إرجاع وصف تفصيلي لوحدة مصدر UIL في شكل جدول رموز (شجرة تحليل).

command_desc
يحدد سطر أوامر uil.
compile_desc
يعيد نتائج الترجمة.
message_cb
يحدد دالة رد اتصال يتم استدعاؤها عندما يواجه المترجم أخطاء في مصدر UIL.
message_data
يحدد بيانات المستخدم التي يتم تمريرها إلى دالة رد اتصال الرسائل (message_cb). لاحظ أن هذه المعلمة لا يتم تفسيرها بواسطة UIL، ويتم استخدامها حصريًا بواسطة التطبيق المستدعي.
status_cb
يحدد دالة رد اتصال يتم استدعاؤها للسماح لتطبيقات X بخدمة أحداث X مثل تحديث الشاشة. يتم استدعاء هذه الدالة عند نقاط فحص مختلفة، والتي تم ترميزها في مترجم UIL. تحدد معلمة status_update_delay في command_desc عدد نقاط الفحص التي يجب تجاوزها قبل استدعاء دالة status_cb.
status_data
يحدد بيانات المستخدم التي يتم تمريرها إلى دالة رد اتصال الحالة (status_cb). لاحظ أن هذه المعلمة لا يتم تفسيرها بواسطة مترجم UIL ويتم استخدامها حصريًا بواسطة التطبيق المستدعي.

فيما يلي هياكل البيانات Uil_command_type و Uil_compile_desc_type:

typedef struct Uil_command_type {
char *source_file;
    /* ملف مصدر واحد للترجمة */
char *resource_file; /* اسم ملف الإخراج */
char *listing_file; /* اسم ملف القائمة */
unsigned int *include_dir_count;
    /* عدد المجلدات. في include_dir */
char *((*include_dir) []);
    /* المجلد. للبحث عن ملفات التضمين */
unsigned listing_file_flag: 1;
    /* إنتاج قائمة */
unsigned resource_file_flag: 1;
    /* إنشاء إخراج UID */
unsigned machine_code_flag: 1;
    /* إنشاء كود الآلة */
unsigned report_info_msg_flag: 1;
    /* الإبلاغ عن رسائل المعلومات */
unsigned report_warn_msg_flag: 1;
    /* الإبلاغ عن التحذيرات */
unsigned parse_tree_flag: 1;
    /* إنشاء شجرة تحليل */
unsigned int status_update_delay;
    /* عدد المرات التي تم فيها تجاوز نقطة الحالة */
    /* قبل استدعاء status_cb */
    /* الوظيفة 0 تعني أنه تم استدعاؤها في كل مرة */
char *database;
    /* اسم ملف قاعدة البيانات */
unsigned database_flag: 1;
    /* قراءة ملف قاعدة بيانات جديد */
unsigned use_setlocale_flag: 1;
    /* تمكين استدعاءات setlocale */
};
typedef struct Uil_compile_desc_type {
unsigned int compiler_version;
    /* رقم إصدار المترجم */
unsigned int data_version;
    /* رقم إصدار الهياكل */
char *parse_tree_root; /* إخراج شجرة التحليل */
unsigned int message_count [Uil_k_max_status+1];
/* مصفوفة من عدد الشدات */
};

فيما يلي وصف لدالة رد اتصال الرسائل المحددة بواسطة message_cb:

Uil_continue_type (*message_cb) (message_data, message_number, severity, msg_buffer,
src_buffer, ptr_buffer, loc_buffer, message_count)
        char *message_data;
        int message_number;
        int severity;
        char *msg_buffer, *src_buffer;
        char *ptr_buffer, *loc_buffer;
        int message_count[];

تحدد هذه الدالة دالة رد اتصال يستدعيها UIL بدلاً من طباعة رسالة خطأ عند مواجهة المترجم خطأ في مصدر UIL. يجب أن تعيد دالة رد الاتصال إحدى القيم التالية:

Uil_k_terminate
إنهاء معالجة ملف المصدر
Uil_k_continue
متابعة معالجة ملف المصدر

المعلمات هي

message_data
بيانات مقدمة من التطبيق كمعلمة message_data للدالة Uil. لا يفسر UIL هذه البيانات بأي شكل من الأشكال؛ بل يمررها فقط إلى دالة رد الاتصال.
message_number
فهرس في جدول رسائل الخطأ والشدات للاستخدام الداخلي بواسطة UIL.
severity
عدد صحيح يشير إلى شدة الخطأ. القيم الممكنة هي ثوابت الحالة التي تعيدها الدالة Uil. انظر قيمة الإرجاع لمزيد من المعلومات.
msg_buffer
سلسلة نصية تصف الخطأ.
src_buffer
سلسلة نصية تتكون من سطر المصدر حيث حدث الخطأ. هذه السلسلة ليست متاحة دائمًا. في هذه الحالة، تكون المعلمة NULL.
ptr_buffer
سلسلة نصية تتكون من مسافات بيضاء وحرف طباعة في الموضع الحرفي المقابل لعمود سطر المصدر حيث حدث الخطأ. يمكن طباعة هذه السلسلة أسفل سطر المصدر لتوفير إشارة مرئية للعمود الذي حدث فيه الخطأ. هذه السلسلة ليست متاحة دائمًا. في هذه الحالة، تكون المعلمة NULL.
loc_buffer
سلسلة نصية تحدد رقم السطر وملف سطر المصدر حيث حدث الخطأ. هذه ليست متاحة دائمًا؛ تكون المعلمة حينئذٍ NULL.
message_count
مصفوفة من الأعداد الصحيحة تحتوي على عدد رسائل التشخيص الصادرة حتى الآن لكل مستوى شدة. للعثور على عدد الرسائل الصادرة لمستوى الشدة الحالي، استخدم معلمة severity كفهرس لهذه المصفوفة.

فيما يلي وصف لدالة رد اتصال الحالة المحددة بواسطة status_cb:

Uil_continue_type (*status_cb) (status_data, percent_complete,
        lines_processed, current_file, message_count)
        char *status_data;
        int percent_complete;
        int lines_processed;
        char *current_file;
        int message_count[];

Uil_k_terminate
إنهاء معالجة ملف المصدر
Uil_k_continue
متابعة معالجة ملف المصدر

المعلمات هي

status_data
بيانات مقدمة من التطبيق كمعلمة status_data للدالة Uil. لا يفسر UIL هذه البيانات
