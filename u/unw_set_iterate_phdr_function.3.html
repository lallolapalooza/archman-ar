unw_set_iterate_phdr_function(3)

الحزمة: libunwind

التراخيص: MIT

UNW_SET_ITERATE_PHDR_FUNCTION(3libunwind) مكتبة البرمجة UNW_SET_ITERATE_PHDR_FUNCTION(3libunwind)

الاسم

unw_set_iterate_phdr_function -- تعيين تنفيذ dl_iterate_phdr

الملخص

#include <libunwind.h>

typedef int (*unw_iterate_phdr_callback_t)(struct dl_phdr_info *, size_t, void *);
typedef int (*unw_iterate_phdr_func_t)(unw_iterate_phdr_callback_t, void *);

void unw_set_iterate_phdr_function(unw_addr_space_t as, unw_iterate_phdr_func_t function);

الوصف

يقوم الروتين unw_set_iterate_phdr_function() بتعيين تنفيذ dl_iterate_phdr لمساحة العنوان as إلى الدالة المسماة function. سيتم استدعاء الدالة كلما احتاجت مكتبة libunwind إلى تكرار رؤوس البرنامج للتطبيق. يتم ذلك عادةً عن طريق استدعاء dl_iterate_phdr، ولكن هذه الدالة ليست آمنة للإشارة (signal safe). بمساعدة تنفيذ مخصص يقوم بتخزين رؤوس البرنامج مؤقتًا والتكرار فوقها، يمكن أيضًا القيام بذلك بطريقة آمنة للإشارة. على الرغم من أن العبء يقع على عاتق مستخدم libunwind.

السلامة للخيوط والإشارات

unw_set_iterate_phdr_function() آمن للخيوط. إذا تم تمرير مساحة العنوان المحلية كوسيطة as، فإن هذا الروتين آمن أيضًا للاستخدام من معالج الإشارة.
