الوصف التفصيلي

توثيق الوظيفة

الروتين الفرعي cunmr3 (character side, character trans, integer m, integer n, integer k, integer l, complex, dimension( lda, * ) a, integer lda, complex, dimension( * ) tau, complex, dimension( ldc, * ) c, integer ldc, complex, dimension( * ) work, integer info)

CUNMR3 تضرب مصفوفة عامة بالبُعد m*n بالمصفوفة الوحدوية من تحليل RZ الذي تحدده ctzrzf (خوارزمية غير محظورة).

الغرض:

!>
!> CUNMR3 تستبدل المصفوفة العامة المعقدة m × n المسماة C بـ
!>
!>       Q * C  إذا كان SIDE = 'L' و TRANS = 'N'، أو
!>
!>       Q**H* C  إذا كان SIDE = 'L' و TRANS = 'C'، أو
!>
!>       C * Q  إذا كان SIDE = 'R' و TRANS = 'N'، أو
!>
!>       C * Q**H إذا كان SIDE = 'R' و TRANS = 'C'،
!>
!> حيث Q هي مصفوفة وحدوية معقدة معرفة كحاصل ضرب k
!> عواكس ابتدائية
!>
!>       Q = H(1) H(2) . . . H(k)
!>
!> كما تم إرجاعها بواسطة CTZRZF. Q من الرتبة m إذا كان SIDE = 'L' ومن الرتبة n
!> إذا كان SIDE = 'R'.
!>

المعاملات

SIDE 
!>          SIDE هو CHARACTER*1
!>          = 'L': تطبيق Q أو Q**H من اليسار
!>          = 'R': تطبيق Q أو Q**H من اليمين
!>

TRANS

!>          TRANS هو CHARACTER*1
!>          = 'N': تطبيق Q (بدون تبديل)
!>          = 'C': تطبيق Q**H (تبديل مرافق)
!>

M

!>          M هو INTEGER
!>          عدد صفوف المصفوفة C. M >= 0.
!>

N

!>          N هو INTEGER
!>          عدد أعمدة المصفوفة C. N >= 0.
!>

K

!>          K هو INTEGER
!>          عدد العواكس الابتدائية التي يحدد حاصل ضربها
!>          المصفوفة Q.
!>          إذا كان SIDE = 'L', M >= K >= 0؛
!>          إذا كان SIDE = 'R', N >= K >= 0.
!>

L

!>          L هو INTEGER
!>          عدد أعمدة المصفوفة A التي تحتوي على
!>          الجزء الهام من عواكس هاوسولدر.
!>          إذا كان SIDE = 'L', M >= L >= 0، إذا كان SIDE = 'R', N >= L >= 0.
!>

A

!>          A هو مصفوفة COMPLEX، ذات البعد
!>                               (LDA,M) إذا كان SIDE = 'L'،
!>                               (LDA,N) إذا كان SIDE = 'R'
!>          الصف i يجب أن يحتوي على المتجه الذي يحدد
!>          العاكس الابتدائي H(i)، من أجل i = 1,2,...,k، كما تم إرجاعه بواسطة
!>          CTZRZF في آخر k صفوف من وسيطته A.
!>          يتم تعديل A بواسطة الروتين ولكن يتم استعادته عند الخروج.
!>

LDA

!>          LDA هو INTEGER
!>          البعد الرائد للمصفوفة A. LDA >= max(1,K).
!>

TAU

!>          TAU هو مصفوفة COMPLEX، ذات البعد (K)
!>          يجب أن يحتوي TAU(i) على العامل القياسي للعكس
!>          الابتدائي H(i)، كما تم إرجاعه بواسطة CTZRZF.
!>

C

!>          C هو مصفوفة COMPLEX، ذات البعد (LDC,N)
!>          عند الإدخال، المصفوفة C بالبُعد m × n.
!>          عند الخروج، يتم استبدال C بـ Q*C أو Q**H*C أو C*Q**H أو C*Q.
!>

LDC

!>          LDC هو INTEGER
!>          البعد الرائد للمصفوفة C. LDC >= max(1,M).
!>

WORK

!>          WORK هو مصفوفة COMPLEX، ذات البعد
!>                                   (N) إذا كان SIDE = 'L'،
!>                                   (M) إذا كان SIDE = 'R'
!>

INFO

!>          INFO هو INTEGER
!>          = 0: خروج ناجح
!>          < 0: إذا كان INFO = -i، فإن الوسيط i كان لديه قيمة غير قانونية
!>
