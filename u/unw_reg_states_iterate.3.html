unw_reg_states_iterate(3)

الحزمة: libunwind

التراخيص: MIT

UNW_REG_STATES_ITERATE(3libunwind) مكتبة البرمجة UNW_REG_STATES_ITERATE(3libunwind)

الاسم

unw_reg_states_iterate -- الحصول على معلومات حالة المسجلات للإجراء الحالي

الملخص

#include <libunwind.h>

int unw_reg_states_iterate(unw_cursor_t *cp, unw_reg_states_callbackcb, void *token);

الوصف

يقوم الروتين unw_reg_states_iterate() بتوفير معلومات حول الإجراء الذي قام بإنشاء إطار المكدس الذي تم تحديده بواسطة الوسيط cp. الوسيط cb هو مؤشر إلى دالة من نوع unw_reg_states_callback والتي تستخدم لإعادة المعلومات. الدالة unw_reg_states_callback لها التعريف التالي:

int ( *unw_reg_states_callback)(void *token, void *reg_states_data, size_t reg_states_data_size, unw_word_t start_ip, unw_word_t end_ip);

يمكن استدعاء دالة الاستدعاء عدة مرات لكل استدعاء لـ unw_reg_states_iterate. يرتبط كل استدعاء بنطاق عنوان تعليمات ومجموعة تعليمات حول كيفية تحديث قيم المسجلات عند العودة من الإجراء في نطاق العنوان هذا. لكل استدعاء، تكون الوسائط للدالة الاستدعاء:

void * token

قيمة الرمز المميز التي تم تمريرها إلى unw_reg_states_callback.
void * reg_states_data

مؤشر إلى بيانات حول تحديث قيم المسجلات. يمكن تمرير هذه البيانات، أو نسخة منها، إلى unw_apply_reg_state.
int reg_states_data_size

حجم بيانات تحديث المسجلات.
unw_word_t start_ip

عنوان أول تعليمة في نطاق العنوان.
unw_word_t end_ip

عنوان أول تعليمة *بعد* نهاية نطاق العنوان.

قيمة الإرجاع

عند الانتهاء بنجاح، يعيد unw_reg_states_iterate() القيمة 0. إذا أعادت الدالة الاستدعاء قيمة غير صفرية، فهذا يشير إلى فشل وتعود الدالة فورًا. بخلاف ذلك، يتم إرجاع القيمة السالبة لأحد رموز الخطأ أدناه.

الأمان في المواضيع والإشارات

unw_reg_states_iterate() آمن للمواضيع. إذا كان المؤشر cp في مساحة العنوان المحلية، فإن هذا الروتين آمن أيضًا للاستخدام من معالج الإشارة.

الأخطاء

UNW_EUNSPEC

حدث خطأ غير محدد.
UNW_ENOINFO

لم تتمكن Libunwind من تحديد معلومات الاسترجاع للإجراء.
UNW_EBADVERSION

معلومات الاسترجاع للإجراء لها إصدار أو تنسيق غير مفهوم بواسطة libunwind.

بالإضافة إلى ذلك، قد يعيد unw_reg_states_iterate() أي خطأ تعيده دالة الاستدعاء access_mem() (انظر unw_create_addr_space(3libunwind)).

libunwind(3libunwind), unw_apply_reg_state(3libunwind)
