subroutine cung2r (integer m, integer n, integer k, complex, dimension( lda, * ) a, integer lda, complex, dimension( * ) tau, complex, dimension( * ) work, integer info)

CUNG2R

الغرض:

!>
!> CUNG2R تولد مصفوفة معقدة Q بحجم m × n بأعمدة متعامدة،
!> والتي تُعرف بأنها أول n أعمدة لناتج k عواكس أولية
!>
!>       Q  =  H(1) H(2) . . . H(k)
!>
!> كما تم إرجاعها بواسطة CGEQRF.
!>

المعاملات

M 
!>          M هو عدد صحيح (INTEGER)
!>          عدد صفوف المصفوفة Q. M >= 0.
!>

N

!>          N هو عدد صحيح (INTEGER)
!>          عدد أعمدة المصفوفة Q. M >= N >= 0.
!>

K

!>          K هو عدد صحيح (INTEGER)
!>          عدد العواكس الأولية التي يحدد ناتجها المصفوفة Q. N >= K >= 0.
!>

A

!>          A هو مصفوفة معقدة (COMPLEX)، ذات بعد (LDA,N)
!>          عند الإدخال، يجب أن يحتوي العمود i على المتجه الذي
!>          يحدد العاكس الأولي H(i)، لـ i = 1,2,...,k، كما
!>          تم إرجاعه بواسطة CGEQRF في أول k أعمدة من حجته المصفوفية
!>          A.
!>          عند الخروج، المصفوفة Q بحجم m × n.
!>

LDA

!>          LDA هو عدد صحيح (INTEGER)
!>          البعد الأول للمصفوفة A. LDA >= max(1,M).
!>

TAU

!>          TAU هو مصفوفة معقدة (COMPLEX)، ذات بعد (K)
!>          يجب أن تحتوي TAU(i) على المعامل القياسي للعكس الأولي
!>          H(i)، كما تم إرجاعه بواسطة CGEQRF.
!>

WORK

!>          WORK هي مصفوفة معقدة (COMPLEX)، ذات بعد (N)
!>

INFO

!>          INFO هو عدد صحيح (INTEGER)
!>          = 0: خروج ناجح
!>          < 0: إذا كان INFO = -i، فإن المعامل i لديه قيمة غير قانونية
!>
