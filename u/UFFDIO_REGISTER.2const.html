UFFDIO_REGISTER(2const) UFFDIO_REGISTER(2const)

الاسم

UFFDIO_REGISTER - تسجيل نطاق عنوان ذاكرة مع كائن userfaultfd

المكتبة

مكتبة C القياسية (libc, -lc)

الملخص

#include <linux/userfaultfd.h>  /* تعريف ثوابت UFFD* */
#include <sys/ioctl.h>
int ioctl(int fd, UFFDIO_REGISTER, struct uffdio_register *argp);
#include <linux/userfaultfd.h>
struct uffdio_range {
    __u64  start;   /* بداية النطاق */
    __u64  len;     /* حجم النطاق (بايت) */
};
struct uffdio_register {
    struct uffdio_range  range;
    __u64                mode;    /* وضع التشغيل المطلوب (إدخال) */
    __u64                ioctls;  /* عمليات ioctl() المتاحة (إخراج) */
};

الوصف

تسجيل نطاق عنوان ذاكرة مع كائن userfaultfd. يجب أن تكون الصفحات في النطاق “متوافقة”. يرجى الرجوع إلى قائمة أوضاع التسجيل أدناه للواجهات الخلفية للذاكرة المتوافقة لكل وضع.

يحدد الحقل argp->range نطاق ذاكرة يبدأ من argp->range.start ويستمر لمدة argp->range.len بايت التي يجب التعامل معها بواسطة userfaultfd.

يحدد الحقل argp->mode وضع التشغيل المطلوب لهذه المنطقة من الذاكرة. يمكن دمج القيم التالية في وضع bitwise لتعيين وضع userfaultfd للنطاق المحدد:

UFFDIO_REGISTER_MODE_MISSING
تتبع أخطاء الصفحات على الصفحات المفقودة. منذ Linux 4.3، النطاقات المجهولة الخاصة فقط متوافقة. منذ Linux 4.11، نطاقات hugetlbfs والذاكرة المشتركة متوافقة أيضًا.
UFFDIO_REGISTER_MODE_WP
تتبع أخطاء الصفحات على الصفحات المحمية ضد الكتابة. منذ Linux 5.7، النطاقات المجهولة الخاصة فقط متوافقة.
UFFDIO_REGISTER_MODE_MINOR
تتبع أخطاء الصفحات الثانوية. منذ Linux 5.13، نطاقات hugetlbfs فقط متوافقة. منذ Linux 5.14، تمت إضافة التوافق مع نطاقات shmem.

إذا كانت العملية ناجحة، يقوم النواة بتعديل حقل قناع البت argp->ioctls للإشارة إلى عمليات ioctl(2) المتاحة للنطاق المحدد. يمكن أن يحتوي هذا القناع العائد على الأجزاء التالية:

1 << _UFFDIO_COPY
عملية UFFDIO_COPY مدعومة.
1 << _UFFDIO_WAKE
عملية UFFDIO_WAKE مدعومة.
1 << _UFFDIO_WRITEPROTECT
عملية UFFDIO_WRITEPROTECT مدعومة.
1 << _UFFDIO_ZEROPAGE
عملية UFFDIO_ZEROPAGE مدعومة.
1 << _UFFDIO_CONTINUE
عملية UFFDIO_CONTINUE مدعومة.
1 << _UFFDIO_POISON
عملية UFFDIO_POISON مدعومة.

قيمة الإرجاع

عند النجاح، يتم إرجاع 0. عند حدوث خطأ، يتم إرجاع -1 ويتم تعيين errno للإشارة إلى الخطأ.

الأخطاء

EBUSY
تم تسجيل تعيين في النطاق المحدد مع كائن userfaultfd آخر.
EFAULT
يشير argp إلى عنوان خارج مساحة عنوان العملية المتصلة التي يمكن الوصول إليها.
EINVAL
تم تحديد جزء غير صالح أو غير مدعوم في حقل mode؛ أو كان حقل mode فارغًا.
EINVAL
لا يوجد تعيين في نطاق العنوان المحدد.
EINVAL
range.start أو range.len ليست مضاعفًا لحجم صفحة النظام؛ أو range.len صفر؛ أو هذه الحقول غير صالحة بطريقة أخرى.
EINVAL
كان هناك تعيين غير متوافق في نطاق العنوان المحدد.
