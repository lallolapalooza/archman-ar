توثيق الدوال

subroutine cunm2r (character side, character trans, integer m, integer n, integer k, complex, dimension( lda, * ) a, integer lda, complex, dimension( * ) tau, complex, dimension( ldc, * ) c, integer ldc, complex, dimension( * ) work, integer info)

CUNM2R يضرب مصفوفة عامة بـ بالمصفوفة الوحدوية من تحليل QR محدد بواسطة cgeqrf (خوارزمية غير مجمعة).

الغرض:

!>
!> CUNM2R تستبدل المصفوفة C المعقدة العامة m-by-n بـ
!>
!>       Q * C  إذا كان SIDE = 'L' و TRANS = 'N'، أو
!>
!>       Q**H* C  إذا كان SIDE = 'L' و TRANS = 'C'، أو
!>
!>       C * Q  إذا كان SIDE = 'R' و TRANS = 'N'، أو
!>
!>       C * Q**H إذا كان SIDE = 'R' و TRANS = 'C'،
!>
!> حيث Q هي مصفوفة وحدوية معقدة معرفة كناتج لـ k
!> انعكاسات أولية
!>
!>       Q = H(1) H(2) . . . H(k)
!>
!> كما تم إرجاعها بواسطة CGEQRF. Q هي من الرتبة m إذا كان SIDE = 'L' ومن الرتبة n
!> إذا كان SIDE = 'R'.
!>

المعلمات

SIDE 
!>          SIDE هو CHARACTER*1
!>          = 'L': تطبيق Q أو Q**H من اليسار
!>          = 'R': تطبيق Q أو Q**H من اليمين
!>

TRANS

!>          TRANS هو CHARACTER*1
!>          = 'N': تطبيق Q (بدون تبديل)
!>          = 'C': تطبيق Q**H (تبديل مركب)
!>

M

!>          M هو INTEGER
!>          عدد صفوف المصفوفة C. M >= 0.
!>

N

!>          N هو INTEGER
!>          عدد أعمدة المصفوفة C. N >= 0.
!>

K

!>          K هو INTEGER
!>          عدد الانعكاسات الأولية التي يحدد ناتجها
!>          المصفوفة Q.
!>          إذا كان SIDE = 'L'، فإن M >= K >= 0؛
!>          إذا كان SIDE = 'R'، فإن N >= K >= 0.
!>

A

!>          A هو مصفوفة COMPLEX، الأبعاد (LDA,K)
!>          يجب أن يحتوي العمود i على المتجه الذي يحدد
!>          الانعكاس الأولي H(i)، لـ i = 1,2,...,k، كما تم إرجاعه بواسطة
!>          CGEQRF في الأعمدة k الأولى من وسيطة المصفوفة A الخاصة به.
!>          يتم تعديل A بواسطة الروتين ولكن يتم استعادتها عند الخروج.
!>

LDA

!>          LDA هو INTEGER
!>          البعد الأمامي للمصفوفة A.
!>          إذا كان SIDE = 'L'، فإن LDA >= max(1,M)؛
!>          إذا كان SIDE = 'R'، فإن LDA >= max(1,N).
!>

TAU

!>          TAU هي مصفوفة COMPLEX، الأبعاد (K)
!>          يجب أن يحتوي TAU(i) على العامل القياسي للانعكاس
!>          الأولي H(i)، كما تم إرجاعه بواسطة CGEQRF.
!>

C

!>          C هي مصفوفة COMPLEX، الأبعاد (LDC,N)
!>          عند الإدخال، المصفوفة C من الحجم m-by-n.
!>          عند الخروج، يتم استبدال C بـ Q*C أو Q**H*C أو C*Q**H أو C*Q.
!>

LDC

!>          LDC هو INTEGER
!>          البعد الأمامي للمصفوفة C. LDC >= max(1,M).
!>

WORK

!>          WORK هي مصفوفة COMPLEX، الأبعاد
!>                                   (N) إذا كان SIDE = 'L'،
!>                                   (M) إذا كان SIDE = 'R'
!>

INFO

!>          INFO هو INTEGER
!>          = 0: خروج ناجح
!>          < 0: إذا كان INFO = -i، فإن الوسيط i كان له قيمة غير قانونية
!>
