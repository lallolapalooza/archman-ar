udev_device_new_from_devnum(3)

الحزمة: systemd-libs

التراخيص: LGPL-2.1-or-later, CC0-1.0, GPL-2.0-or-later WITH Linux-syscall-note

UDEV_DEVICE_NEW_FROM_SYSPATH(3) udev_device_new_from_syspath UDEV_DEVICE_NEW_FROM_SYSPATH(3)

الاسم

udev_device_new_from_syspath, udev_device_new_from_devnum, udev_device_new_from_subsystem_sysname, udev_device_new_from_device_id, udev_device_new_from_environment, udev_device_ref, udev_device_unref - إنشاء، الحصول على، وتحرير كائن جهاز udev

الخلاصة

#include <libudev.h>

struct udev_device *udev_device_new_from_syspath(struct udev *udev, const char *syspath);

struct udev_device *udev_device_new_from_devnum(struct udev *udev, char type, dev_t devnum);

struct udev_device *udev_device_new_from_subsystem_sysname(struct udev *udev, const char *subsystem, const char *sysname);

struct udev_device *udev_device_new_from_device_id(struct udev *udev, const char *id);

struct udev_device *udev_device_new_from_environment(struct udev *udev);

struct udev_device *udev_device_ref(struct udev_device *udev_device);

الوصف

udev_device_new_from_syspath(), udev_device_new_from_devnum(), udev_device_new_from_subsystem_sysname(), udev_device_new_from_device_id(), و udev_device_new_from_environment() تقوم بتخصيص كائن جهاز udev جديد وتعيد مؤشرًا إليه. هذا الكائن غير شفاف ولا يجب على المتصل الوصول إليه بوسائل مختلفة عن الوظائف التي توفرها libudev. في البداية، يكون عدد المراجع للجهاز 1. يمكنك الحصول على مراجع إضافية، وإنقاص المراجع المكتسبة عبر udev_device_ref() و udev_device_unref(). بمجرد أن يصل عدد المراجع إلى 0، يتم تدمير كائن الجهاز وتحريره.

udev_device_new_from_syspath(), udev_device_new_from_devnum(), udev_device_new_from_subsystem_sysname(), و udev_device_new_from_device_id() تنشئ كائن الجهاز بناءً على المعلومات الموجودة في /sys/، مع إضافة خصائص من قاعدة بيانات الأجهزة الداخلية لـ udev. مسار syspath هو أي دليل فرعي لـ /sys/، مع القيد على أن الدليل الفرعي لـ /sys/devices (أو رابط رمزي له) يمثل جهازًا حقيقيًا ويجب أن يحتوي بالتالي على ملف uevent. تأخذ udev_device_new_from_devnum() نوع جهاز، والذي يمكن أن يكون b للأجهزة الكتلية أو c للأجهزة الحرفية، بالإضافة إلى رقم جهاز (انظر makedev(3)). تبحث udev_device_new_from_subsystem_sysname() عن الأجهزة بناءً على النظام الفرعي واسم النظام الفرعي المقدمين (انظر udev_device_get_subsystem(3) و udev_device_get_sysname(3)) وتبحث udev_device_new_from_device_id() عن الأجهزة بناءً على معرف الجهاز المقدم، وهو سلسلة نصية خاصة بأحد الأشكال الأربعة التالية:

جدول 1. سلاسل معرّف الجهاز

مثال الشرح
b8:2 الجهاز الكتلي الرئيسي: الثانوي
c128:1 الجهاز الحرفي الرئيسي: الثانوي
n3 فهرس واجهة الجهاز الشبكي
+sound:card29 النظام الفرعي لنواة المشغلات: اسم الجهاز

udev_device_new_from_environment()
ينشئ جهازًا من البيئة الحالية (انظر environ(7)). يتم تفسير كل زوج من المفاتيح والقيم بنفس الطريقة كما لو تم استلامه في uevent (انظر udev_monitor_receive_device(3)). المفاتيح DEVPATH، SUBSYSTEM، ACTION، و SEQNUM إلزامية.

قيمة الإرجاع

عند النجاح، تعيد udev_device_new_from_syspath(), udev_device_new_from_devnum(), udev_device_new_from_subsystem_sysname(), udev_device_new_from_device_id() و udev_device_new_from_environment() مؤشرًا إلى جهاز udev المخصص. عند الفشل، يتم إرجاع NULL، ويتم تعيين errno بشكل مناسب. تعيد udev_device_ref() الوسيط الذي تم تمريره إليها، دون تعديل. تعيد udev_device_unref() دائمًا NULL.
