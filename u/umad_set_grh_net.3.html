UMAD_SET_GRH_NET(3) دليل مبرمج OpenIB UMAD_SET_GRH_NET(3)

الاسم

umad_set_grh_net - ضبط حقول GRH داخل مخزن مؤقت umad باستخدام الترتيب الشبكي

الملخص

#include <infiniband/umad.h>

int umad_set_grh_net(void *umad, void *mad_addr);

الوصف

تقوم الدالة umad_set_grh_net() بضبط حقول GRH (grh_present, gid, hop_limit, traffic_class, flow_label) داخل مخزن مؤقت umad المحدد بناءً على mad_addr المقدم. يُتوقع أن تكون حقول المؤشر mad_addr المقدمة بترتيب الشبكة. إذا كان المؤشر mad_addr المقدم NULL، فلن يتم تعيين أي GRH. الوسيط mad_addr هو مؤشر إلى بنية ib_mad_addr_t، كما هو محدد في <infiniband/umad.h>. الوسيط umad هو مؤشر إلى بنية ib_user_mad_t، كما هو محدد في <infiniband/umad.h>.

typedef struct ib_mad_addr {
uint32_t qpn;
uint32_t qkey;
uint16_t lid;
uint8_t  sl;
uint8_t  path_bits;
uint8_t  grh_present;
uint8_t  gid_index;
uint8_t  hop_limit;
uint8_t  traffic_class;
uint8_t  gid[16];
uint32_t flow_label;
} ib_mad_addr_t;
typedef struct ib_user_mad {
uint32_t agent_id;
uint32_t status;
uint32_t timeout_ms;
uint32_t retries;
uint32_t length;
ib_mad_addr_t addr;
uint8_t  data[0];
} ib_user_mad_t;

قيمة الإرجاع

ترجع الدالة umad_set_grh_net() 0 عند النجاح، وقيمة سالبة عند حدوث أخطاء. حاليًا، لا توجد أخطاء مشيرة.

أخطاء معروفة
