UMAD_GET_PORT(3) دليل مبرمج OpenIB UMAD_GET_PORT(3)

الاسم

umad_get_port, umad_release_port - فتح وإغلاق منفذ InfiniBand

الملخص

#include <infiniband/umad.h>

int umad_get_port(char *ca_name, int portnum, umad_port_t *port);

int umad_release_port(umad_port_t *port);

الوصف

تقوم الدالة umad_get_port() بملء البنية port بسمات منفذ IB المحددة بواسطة ca_name و portnum ، أو المنفذ الافتراضي إذا كان ca_name يساوي NULL و portnum يساوي صفر. إذا تم تحديد واحد فقط من ca_name و portnum ، يتم استخدام الآخر كمرشح. على سبيل المثال، تمرير ca_name مساويًا لـ NULL و 2 لـ portnum يعني الحصول على منفذ من أي من أجهزة IB المحلية، طالما أنه المنفذ الثاني. لاحظ أن المكتبة قد تستخدم مخطط مرجعي لدعم تخزين المنافذ مؤقتًا، وبالتالي يجب استدعاء umad_release_port() قبل أن يمكن إلغاء تخصيص البنية port. الوسيط port هو بنية umad_port_t، كما هو محدد في <infiniband/umad.h>.

typedef struct umad_port {
char ca_name[UMAD_CA_NAME_LEN];                 /* اسم الجهاز */
int portnum;                                    /* رقم المنفذ المادي */
uint base_lid;                                  /* LID الأساسي للمنفذ */
uint lmc;                                       /* LMC الخاص بـ LID */
uint sm_lid;                                    /* SM LID */
uint sm_sl;                                     /* مستوى خدمة SM */
uint state;                                     /* الحالة المنطقية للمنفذ */
uint phys_state;                                /* الحالة المادية للمنفذ */
uint rate;                                      /* معدل بت رابط المنفذ */
uint64_t capmask;                               /* قناع قدرات المنفذ */
uint64_t gid_prefix;                            /* بادئة GID لهذا المنفذ */
uint64_t port_guid;                             /* GUID لهذا المنفذ */
} umad_port_t;

تقوم الدالة umad_release_port() بتحرير الموارد التي تم تخصيصها بواسطة الدالة umad_get_port() للمنفذ IB المحدد port.

قيمة الإرجاع

ترجع الدالتان umad_get_port() و umad_release_port() القيمة 0 عند النجاح، وقيمة سالبة عند حدوث خطأ.
