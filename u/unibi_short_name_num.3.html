unibi_short_name_num(3)

الحزمة: unibilium

التراخيص: LGPL3

doc::unibi_short_name_num(3) unibilium doc::unibi_short_name_num(3)

اسم

unibi_name_bool, unibi_name_num, unibi_name_str, unibi_short_name_bool, unibit_short_name_num, unibi_short_name_str - ترجمة تعدادات القدرات إلى أسماء

ملخص

#include <unibilium.h>

const char *unibi_name_bool(enum unibi_boolean b);
const char *unibi_name_num(enum unibi_numeric n);
const char *unibi_name_str(enum unibi_string s);

const char *unibi_short_name_bool(enum unibi_boolean b);
const char *unibi_short_name_num(enum unibi_numeric n);
const char *unibi_short_name_str(enum unibi_string s);

وصف

تعيد هذه الدوال أسماء القدرات كسلاسل نصية. افتراضيًا، يتم إرجاع الأسماء الطويلة ("أسماء المتغيرات")؛ تستخدم المتغيرات "unibi_short_name_*" "أسماء القدرات" الأقصر.

مثال

#include <stdio.h>
#include <unibilium.h>

int main(void) {
  printf("%s - %s\n", unibi_name_bool(unibi_has_status_line), unibi_short_name_bool(unibi_has_status_line));
  /* الإخراج:
     has_status_line - hs
  */
}
