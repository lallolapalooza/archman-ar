rtnetlink(7)

rtnetlink(7) دليل المعلومات المتنوعة rtnetlink(7)

الاسم

rtnetlink - مقبس توجيه لينكس

الملخص

#include <asm/types.h>
#include <linux/netlink.h>
#include <linux/rtnetlink.h>
#include <sys/socket.h>
rtnetlink_socket = socket(AF_NETLINK, int socket_type, NETLINK_ROUTE);

الوصف

يسمح Rtnetlink بقراءة وتعديل جداول توجيه نواة لينكس. يُستخدم داخل النواة للتواصل بين الأنظمة الفرعية المختلفة، على الرغم من أن هذا الاستخدام غير موثق هنا، وللتواصل مع برامج مساحة المستخدم. يمكن التحكم في مسارات الشبكة، وعناوين IP، ومعلمات الارتباط، وإعدادات الجوار، وتخصصات قوائم الانتظار، وفئات حركة المرور، ومصنفات الحزم عبر مقابس NETLINK_ROUTE. يعتمد على رسائل netlink؛ انظر netlink(7) لمزيد من المعلومات.

سمات التوجيه

تحتوي بعض رسائل rtnetlink على سمات اختيارية بعد الرأس الأولي:


struct rtattr {
    unsigned short rta_len;    /* طول الخيار */
    unsigned short rta_type;   /* نوع الخيار */
    /* البيانات تتبع */
};

يجب معالجة هذه السمات باستخدام وحدها ماكرو RTA_* أو libnetlink، انظر rtnetlink(3).

الرسائل

يتكون Rtnetlink من أنواع الرسائل هذه (بالإضافة إلى رسائل netlink القياسية):

RTM_NEWLINK
RTM_DELLINK
RTM_GETLINK
إنشاء أو إزالة أو الحصول على معلومات حول واجهة شبكة معينة. تحتوي هذه الرسائل على بنية ifinfomsg تتبعها سلسلة من سمات rtattr.
struct ifinfomsg {
    unsigned char  ifi_family; /* AF_UNSPEC */
    unsigned short ifi_type;   /* نوع الجهاز */
    int            ifi_index;  /* فهرس الواجهة */
    unsigned int   ifi_flags;  /* علامات الجهاز */
    unsigned int   ifi_change; /* قناع التغيير */
};
ifi_flags تحتوي على علامات الجهاز، انظر netdevice(7)؛ ifi_index هو فهرس الواجهة الفريد (منذ لينكس 3.7، من الممكن تغذية قيمة غير صفرية برسالة RTM_NEWLINK، وبالتالي إنشاء ارتباط بالفهرس ifindex المحدد)؛ ifi_change محجوز للاستخدام المستقبلي ويجب أن يتم تعيينه دائمًا إلى 0xFFFFFFFF.
سمات التوجيه
rta_type نوع القيمة الوصف
IFLA_UNSPEC - غير محدد
IFLA_ADDRESS عنوان الجهاز عنوان L2 للواجهة
IFLA_BROADCAST عنوان الجهاز عنوان البث L2
IFLA_IFNAME سلسلة نصية اسم الجهاز
IFLA_MTU unsigned int MTU للجهاز
IFLA_LINK int نوع الارتباط
IFLA_QDISC سلسلة نصية تخصص قائمة الانتظار
IFLA_STATS انظر أدناه إحصائيات الواجهة
IFLA_PERM_ADDRESS عنوان الجهاز عنوان الجهاز الذي يوفره الجهاز (منذ لينكس 5.5)
نوع القيمة لـ IFLA_STATS هو struct rtnl_link_stats (struct net_device_stats في لينكس 2.4 وما قبله).
RTM_NEWADDR
RTM_DELADDR
RTM_GETADDR
إضافة أو إزالة أو استقبال معلومات حول عنوان IP مرتبط بواجهة. في لينكس 2.2، يمكن للواجهة حمل عناوين IP متعددة، وهذا يحل محل مفهوم الجهاز البديل في لينكس 2.0. في لينكس 2.2، تدعم هذه الرسائل عناوين IPv4 و IPv6. تحتوي على بنية ifaddrmsg، تتبعها اختياريًا سمات توجيه rtattr.
struct ifaddrmsg {
    unsigned char ifa_family;    /* نوع العنوان */
    unsigned char ifa_prefixlen; /* طول بادئة العنوان */
    unsigned char ifa_flags;     /* علامات العنوان */
    unsigned char ifa_scope;     /* نطاق العنوان */
    unsigned int  ifa_index;     /* فهرس الواجهة */
};
ifa_family هو نوع عائلة العنوان (حالياً AF_INET أو AF_INET6)، ifa_prefixlen هو طول قناع العنوان للعنون إذا تم تعريفه للعائلة (مثل IPv4)، ifa_scope هو نطاق العنوان، ifa_index هو فهرس الواجهة التي يرتبط بها العنوان. ifa_flags هي كلمة علامة لـ IFA_F_SECONDARY للعنوان الثانوي (واجهة بديلة قديمة)، IFA_F_PERMANENT لعنوان دائم تم تعيينه بواسطة المستخدم وعلامات أخرى غير موثقة.
السمات
rta_type نوع القيمة الوصف
IFA_UNSPEC - غير محدد
IFA_ADDRESS عنوان بروتوكول خام عنوان الواجهة
IFA_LOCAL عنوان بروتوكول خام العنوان المحلي
IFA_LABEL سلسلة نصية اسم الواجهة
IFA_BROADCAST عنوان بروتوكول خام عنوان البث
IFA_ANYCAST عنوان بروتوكول خام عنوان anycast
IFA_CACHEINFO struct ifa_cacheinfo معلومات العنوان 
struct ifa_cacheinfo {
    __u32 ifa_prefered; // عمر مفضل متبقي، بالثواني
    __u32 ifa_valid;    // عمر صالح متبقي، بالثواني
    __u32 cstamp;       // ختم وقت الإنشاء، بالسنتمتر ثانية
    __u32 tstamp;       // ختم وقت التحديث، بالسنتمتر ثانية
};
ifa_valid لا يمكن أن يكون صفراً، و ifa_prefered لا يمكن أن يكون أكبر من ifa_valid. قيمة UINT32_MAX تمثل عمرًا لا نهائيًا.
RTM_NEWROUTE
RTM_DELROUTE
RTM_GETROUTE
إنشاء أو إزالة أو استقبال معلومات حول مسار شبكة. تحتوي هذه الرسائل على بنية rtmsg مع تسلسل اختياري من سمات rtattr تتبعها. بالنسبة لـ RTM_GETROUTE، فإن تعيين rtm_dst_len و rtm_src_len إلى 0 يعني أنك تحصل على جميع الإدخالات لجدول التوجيه المحدد. بالنسبة للحقول الأخرى، باستثناء rtm_table و rtm_protocol، فإن 0 هي علامة استفهام.
struct rtmsg {
    unsigned char rtm_family;   /* عائلة عنوان المسار */
    unsigned char rtm_dst_len;  /* طول الوجهة */
    unsigned char rtm_src_len;  /* طول المصدر */
    unsigned char rtm_tos;      /* فلتر TOS */
    unsigned char rtm_table;    /* معرف جدول التوجيه؛
                                   انظر RTA_TABLE أدناه */
    unsigned char rtm_protocol; /* بروتوكول التوجيه؛ انظر أدناه */
    unsigned char rtm_scope;    /* انظر أدناه */
    unsigned char rtm_type;     /* انظر أدناه */
    unsigned int  rtm_flags;
};
