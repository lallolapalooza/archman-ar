ranger.3bobcat

الحزمة: libbobcat

التراخيص: GPL

FBB::Ranger(3bobcat) مكررات الحلقة التكرارية FBB::Ranger(3bobcat)

الاسم

FBB::Ranger - تعميم النطاقات لحلقات for المستندة إلى النطاق

موجز

#include <bobcat/ranger>

الوصف

الفئة الـ Ranger تحدد النطاقات التي يمكن استخدامها مع حلقات for المستندة إلى النطاق. تتطلب حلقة for القياسية المستندة إلى النطاق، لمواصفات النطاق الخاص بها، مصفوفة أو نطاق مكرر كما تقدمه، على سبيل المثال، الحاويات (من خلال أعضائها begin و end). النطاقات المحددة بزوج من المؤشرات أو بنطاق فرعي محدد بتعبيرات المكرر لا يمكن استخدامها حاليًا بالاقتران مع حلقات for المستندة إلى النطاق.

تمد Ranger نطاق تطبيق حلقات for المستندة إلى النطاق عن طريق تحويل أزواج المؤشرات، مؤشر ابتدائي وعدد مؤشرات، أو زوج من المكررات إلى نطاق يمكن استخدامه بواسطة حلقات for المستندة إلى النطاق.

Ranger هي فئة قالب تتطلب معلمة نوع قالب واحدة: Iterator، وهو مكرر أو نوع مؤشر يصل إلى البيانات عند إلغاء الإشارة إليها.

لا يحتاج مستخدمو Ranger إلى تحديد نوع قالب Ranger. قالب الدالة ranger يعيد كائن Ranger المناسب.

الدالة الحرة

عند استخدام الدوال الحرة التالية، يمكن استخدام أي (نطاق فرعي) من المكررات أو المؤشرات. مع المكررات، يمكن أيضًا تحديد نطاقات فرعية من المكررات المعكوسة. يوضح قسم المثال أدناه استخدام قوالب دوال ranger.

o
Ranger<Iterator> ranger(Iterator &&begin, Iterator &&end):
يعيد قالب الدالة هذا كائن Ranger للنطاق (الفرعي) المحدد بواسطة مكررين (معكوسين)؛
o
Ranger<Iterator> ranger(Iterator &&begin, size_t count):
يعيد قالب الدالة هذا كائن Ranger للنطاق (الفرعي) المحدد بواسطة المكرر (المعكوس) begin والعدد count (أي النطاق من begin إلى begin + count)؛
o
Ranger<Data *> ranger(Data *begin, Data *end):
يعيد قالب الدالة هذا كائن Ranger للنطاق (الفرعي) المحدد بواسطة المؤشرين begin و end؛
o
Ranger<Data *> ranger(Data *begin, size_t count):
يعيد قالب الدالة هذا كائن Ranger للنطاق (الفرعي) المحدد بواسطة المؤشرين begin و begin + count.

المنشئات

أدناه، يشير Iterator إلى معلمة نوع الفئة الـ Ranger. على الرغم من تسميته بـ "Iterator"، يمكن أن يكون أيضًا مؤشرًا لنوع بيانات معين (مثل std::string *).

o
Ranger(Iterator const &begin, Iterator const &end):
يمكن تمرير كائن Ranger كمحدد للنطاق في حلقة for مستندة إلى النطاق. سيتم بعد ذلك زيارة جميع العناصر المحددة بواسطة النطاق بواسطة حلقة for المستندة إلى النطاق.

تتوفر منشئات النسخ والنقل (وعمليات الإسناد).

الدوال الأعضاء

o
Iterator const &begin() const:
يعيد (نسخة من) المكرر begin الذي تم تمريره إلى منشئ Ranger. لاحظ أنه إذا كان Iterator نوع مؤشر (مثل int *) فإن البيانات التي يشير إليها المكرر الذي يعيده begin() يمكن تعديلها بالفعل، حيث أن نوع الإرجاع للعضو (باستخدام int * لـ Iterator) يصبح int * const &، أي مرجع لمؤشر ثابت يشير إلى int. هذا مقبول تمامًا: إذا كان يجب أن تكون البيانات نفسها غير قابلة للتغيير، فيجب تعريف نوع البيانات كـ int const، وهو ما يحدث تلقائيًا عند تمرير بيانات int const *. انظر قسم المثال للتوضيح.
o
Iterator const &end() const:
يعيد (نسخة من) المكرر end الذي تم تمريره إلى منشئ Ranger. إذا تم تمرير مكررات معكوسة إلى منشئ Ranger، فإن الأعضاء begin و end يعيدون مكررات معكوسة. نظرًا لأن الاستخدام المقصود لكائنات Ranger هو تحديد نطاق لحلقات for المستندة إلى النطاق، يمكن حذف أعضاء مثل rbegin و rend من Ranger.

مثال

#include <vector>
#include <iostream>
#include <bobcat/ranger>
using namespace std;
using namespace FBB;
int main()
{
    vector<int> iv {1, 2, 3, 4, 5};
                                    // عرض وتعديل نطاق فرعي
    for(auto &el: ranger(iv.rbegin() + 1, iv.rend() - 1))
        cout << el++ << ’ ’;
    cout << ’\n’;
                                    // عرض نطاق معكوس
    for(auto &el: ranger(iv.rbegin() + 1, iv.rend() - 1))
        cout << el << ’ ’;
    cout << ’\n’;
                                    // نفس الشيء: العرض باستخدام عدد
    for(auto &el: ranger(iv.rbegin() + 1, 3))
        cout << el << ’ ’;
    cout << ’\n’;
    int intArray[] = {1, 2, 3, 4, 5};
                                    // عرض وتعديل العناصر
                                    // في نطاق يعتمد على المؤشر
    for(auto &el: ranger(intArray + 1, intArray + 3))
        cout << el++ << ’ ’;
    cout << ’\n’;
                                    // البيانات تم تعديلها الآن
    for(auto &el: ranger(intArray + 1, intArray + 3))
        cout << el << ’ ’;
    cout << ’\n’;
                                    // استخدام عدد بدلاً من
                                    // مؤشر نهاية
    for(auto &el: ranger(intArray + 1, 3))
        cout << el << ’ ’;
    cout << ’\n’;
    int const constInts[] = {1, 2, 3, 4, 5};
                                    // لا يمكن تعديل البيانات
    for(auto &el: ranger(constInts + 1, constInts + 3))
        cout << el << ’ ’;
    cout << ’\n’;
}
