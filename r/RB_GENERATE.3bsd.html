RB_GENERATE(3bsd)

الحزمة: libbsd

التراخيص: custom

الاسم

RB_HEAD

المكتبة

tree.h

الملخص

#include <sys/tree.h>

الوصف

RB_HEAD يحدد شجرة حمراء-سوداء. RB_INITIALIZER تهيئ شجرة حمراء-سوداء. RB_INSERT يدرج عنصرًا في الشجرة. RB_REMOVE يزيل عنصرًا من الشجرة. RB_FIND و RB_NFIND يبحثان عن عناصر في الشجرة. RB_ROOT، RB_MIN، RB_MAX، RB_NEXT، و RB_PREV تتنقل في الشجرة. RB_FOREACH و RB_FOREACH_REVERSE تتنقل في الشجرة. RB_FOREACH_SAFE و RB_FOREACH_REVERSE_SAFE تتنقل في الشجرة بأمان. RB_EMPTY تتحقق مما إذا كانت الشجرة فارغة.

أشجار SPLAY

SPLAY_FOREACH، SPLAY_MIN، SPLAY_MAX، SPLAY_NEXT، و SPLAY_PREV هي ماكرو للتنقل في شجرة splay.

الأشجار الحمراء-السوداء

RB_INIT، RB_INITIALIZER، RB_INSERT، RB_REMOVE، RB_FIND، RB_NFIND، RB_ROOT، RB_MIN، RB_MAX، RB_NEXT، RB_PREV، RB_FOREACH، RB_FOREACH_REVERSE، RB_FOREACH_SAFE، RB_FOREACH_REVERSE_SAFE، و RB_EMPTY هي ماكرو لمعالجة شجرة حمراء-سوداء.

أمثلة

يوضح المثال التالي كيفية تعريف شجرة حمراء-سوداء تحتوي على أعداد صحيحة. يتم إدراج القيم فيها وطباعة محتويات الشجرة بالترتيب. أخيرًا، يتم طباعة الهيكل الداخلي للشجرة.

#include <sys/tree.h>
#include <err.h>
#include <stdio.h>
#include <stdlib.h>

struct node {
	RB_ENTRY(node) entry;
	int i;
};

int intcmp(struct node *, struct node *);
void print_tree(struct node *);

int
intcmp(struct node *e1, struct node *e2)
{
	return (e1->i < e2->i ? -1 : e1->i > e2->i);
}

RB_HEAD(inttree, node) head = RB_INITIALIZER(&head);
RB_PROTOTYPE(inttree, node, entry, intcmp)
RB_GENERATE(inttree, node, entry, intcmp)

int testdata[] = {
	20, 16, 17, 13, 3, 6, 1, 8, 2, 4, 10, 19, 5, 9, 12, 15, 18,
	7, 11, 14
};

void
print_tree(struct node *n)
{
	struct node *left, *right;

	if (n == NULL) {
		printf("nil");
		return;
	}
	left = RB_LEFT(n, entry);
	right = RB_RIGHT(n, entry);
	if (left == NULL && right == NULL)
		printf("%d", n->i);
	else {
		printf("%d(", n->i);
		print_tree(left);
		printf(","); 
		print_tree(right);
		printf(")");
	}
}

int
main(void)
{
	int i;
	struct node *n;

	for (i = 0; i < sizeof(testdata) / sizeof(testdata[0]); i++) {
		if ((n = malloc(sizeof(struct node))) == NULL)
			err(1, NULL);
		n->i = testdata[i];
		RB_INSERT(inttree, &head, n);
	}

	RB_FOREACH(n, inttree, &head) {
		printf("%d\n", n->i);
	}
	print_tree(RB_ROOT(&head));
	printf("\n");
	return (0);
}

ملاحظات

محاولة تحرير شجرة بالطريقة التالية خطأ شائع:

SPLAY_FOREACH(var, NAME, &head) {
	SPLAY_REMOVE(NAME, &head, var);
	free(var);
}
free(head);

نظرًا لأنه يتم تحرير `var`، فإن ماكرو `FOREACH` يشير إلى مؤشر ربما تم إعادة تخصيصه بالفعل. يتطلب الكود الصحيح متغيرًا ثانيًا.

for (var = SPLAY_MIN(NAME, &head); var != NULL; var = nxt) {
	nxt = SPLAY_NEXT(NAME, &head, var);
	SPLAY_REMOVE(NAME, &head, var);
	free(var);
}
