rpcsec_gss(3t)

RPC_GSS_SECCREATE(3) دليل وظائف المكتبة RPC_GSS_SECCREATE(3)

الاسم

RPCSEC_GSS — المصادقة المستندة إلى GSS-API لـ RPC

الملخص

#include <rpc/rpcsec_gss.h>

الوصف

RPCSEC_GSS هي آلية أمان لبروتوكول RPC. إنها تستخدم واجهة برمجة تطبيقات خدمة الأمان العامة (GSS-API) لإنشاء سياق أمني بين عميل وخادم ولضمان أن جميع الاتصالات اللاحقة بين العميل والخادم يتم مصادقتها بشكل صحيح. اختياريًا، يمكن تطبيق حماية إضافية على الاتصال. تستخدم خدمة التكامل (integrity) المجموع الاختباري (checksums) لضمان استلام جميع البيانات المرسلة بواسطة طرف مقابل دون تعديل. تستخدم خدمة الخصوصية (privacy) التشفير لضمان عدم تمكن طرف ثالث من الوصول إلى البيانات لاتصال معين.

لاستخدام هذا النظام، يجب على التطبيق أولاً استخدام rpc_gss_seccreate() لإنشاء سياق أمني.

هياكل البيانات

هياكل البيانات المستخدمة بواسطة RPCSEC_GSS تظهر أدناه.

rpc_gss_service_t
يحدد هذا النوع أنواع خدمة الأمان المطلوبة لـ rpc_gss_seccreate(). 
typedef enum {
	rpc_gss_svc_default	= 0,
	rpc_gss_svc_none	= 1,
	rpc_gss_svc_integrity	= 2,
	rpc_gss_svc_privacy	= 3
} rpc_gss_service_t;
rpc_gss_options_req_t
يحتوي هذا الهيكل على قيم اختيارية مختلفة تُستخدم أثناء إنشاء سياق الأمان. 
typedef struct {
	int		req_flags;	/* بتات طلب GSS */
	int		time_req;	/* فترة الحياة المطلوبة */
	gss_cred_id_t	my_cred;	/* بيانات اعتماد GSS */
	gss_channel_bindings_t input_channel_bindings;
} rpc_gss_options_req_t;
rpc_gss_options_ret_t
يتم إرجاع تفاصيل مختلفة للسياق الأمني المنشأ باستخدام هذا الهيكل. 
typedef struct {
	int		major_status;
	int		minor_status;
	u_int		rpcsec_version;
	int		ret_flags;
	int		time_req;
	gss_ctx_id_t	gss_context;
	char		actual_mechanism[MAX_GSS_MECH];
} rpc_gss_options_ret_t;
rpc_gss_principal_t
يُستخدم هذا النوع للإشارة إلى كيان عميل ممثل في شكل اسم GSS-API المصدر (انظر gss_export_name(3) لمزيد من التفاصيل). قد يتم تخزين الأسماء بهذا الشكل في قوائم التحكم بالوصول أو مقارنتها بأسماء أخرى بشكل مصدر. يتم إرجاع هذا الهيكل بواسطة rpc_gss_get_principal_name() ويشار إليه أيضًا بواسطة هيكل rpc_gss_rawcred_t. 
typedef struct {
	int		len;
	char		name[1];
} *rpc_gss_principal_t;
rpc_gss_rawcred_t
يُستخدم هذا الهيكل للوصول إلى بيانات الاعتماد الأولية المرتبطة بسياق أمني. 
typedef struct {
	u_int		version;	/* رقم إصدار RPC */
	const char	*mechanism;	/* آلية الأمان */
	const char	*qop;		/* جودة الحماية */
	rpc_gss_principal_t client_principal; /* اسم العميل */
	const char	*svc_principal;	/* اسم الخادم */
	rpc_gss_service_t service;	/* نوع الخدمة */
} rpc_gss_rawcred_t;
rpc_gss_ucred_t
بيانات اعتماد Unix المشتقة من بيانات الاعتماد الأولية، يتم الوصول إليها عبر rpc_gss_getcred(). 
typedef struct {
	uid_t		uid;		/* معرف المستخدم */
	gid_t		gid;		/* معرف المجموعة */
	short		gidlen;
	gid_t		*gidlist;	/* قائمة المجموعات */
} rpc_gss_ucred_t;
rpc_gss_lock_t
الهيكل المستخدم لفرض QOP (جودة الحماية) وخدمة معينة. 
typedef struct {
	bool_t		locked;
	rpc_gss_rawcred_t *raw_cred;
} rpc_gss_lock_t;
rpc_gss_callback_t
هيكل الاستدعاء المستخدم بواسطة rpc_gss_set_callback(). 
typedef struct {
	u_int		program;	/* رقم برنامج RPC */
	u_int		version;	/* رقم إصدار RPC */
					/* استدعاء معرف من قبل المستخدم */
	bool_t		(*callback)(struct svc_req *req,
				    gss_cred_id_t deleg,
				    gss_ctx_id_t gss_context,
				    rpc_gss_lock_t *lock,
				    void **cookie);
} rpc_gss_callback_t;
rpc_gss_error_t
الهيكل المستخدم لإرجاع معلومات الخطأ بواسطة rpc_gss_get_error(). 
typedef struct {
	int		rpc_gss_error;
	int		system_error;	/* نفس قيمة errno */
} rpc_gss_error_t;

/*
 * قيم لـ rpc_gss_error
 */
#define RPC_GSS_ER_SUCCESS	0	/* لا يوجد خطأ */
#define RPC_GSS_ER_SYSTEMERROR	1	/* خطأ نظام */
