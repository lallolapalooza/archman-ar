restic-copy(1)

الحزمة: restic

التراخيص: BSD

restic backup(1) restic backup(1)

الاسم

restic-copy - نسخ اللقطات من مستودع إلى آخر

الملخص

restic copy [flags] [snapshotID ...]

الوصف

الأمر "copy" يقوم بنسخ لقطة أو أكثر من مستودع إلى آخر.

ملاحظة: هذه العملية ستحتاج إلى تحميل (قراءة) وتحميل (كتابة) اللقطات بأكملها بسبب مفاتيح التشفير المختلفة المستخدمة في مستودعات المصدر والوجهة. هذا قد يؤدي إلى استخدام نطاق ترددي وتكاليف أعلى من المتوقع أثناء عمليات النسخ الاحتياطي العادية.

ملاحظة: عملية النسخ لا تعيد تقسيم الملفات، مما قد يكسر إزالة التكرار بين الملفات التي تم نسخها والملفات المخزنة بالفعل في مستودع الوجهة. هذا يعني أن الملفات المنسوخة، التي كانت موجودة في كل من مستودع المصدر والوجهة، قد تشغل ما يصل إلى ضعف حجمها في مستودع الوجهة. يمكن تخفيف ذلك باستخدام خيار "--copy-chunker-params" عند تهيئة مستودع وجهة جديد باستخدام الأمر "init".

حالة الخروج

حالة الخروج هي 0 إذا كان الأمر ناجحًا. حالة الخروج هي 1 إذا كان هناك أي خطأ. حالة الخروج هي 10 إذا كان المستودع غير موجود. حالة الخروج هي 11 إذا كان المستودع مقفلاً بالفعل. حالة الخروج هي 12 إذا كانت كلمة المرور غير صحيحة.

خيارات

--from-insecure-no-password[=false] استخدم كلمة مرور فارغة لمستودع المصدر (غير آمن)

--from-key-hint="" معرف المفتاح لمحاولة فك تشفير مستودع المصدر أولاً (الافتراضي: $RESTIC_FROM_KEY_HINT)

--from-password-command="" أمر shell للحصول على كلمة مرور مستودع المصدر من (الافتراضي: $RESTIC_FROM_PASSWORD_COMMAND)

--from-password-file="" ملف لقراءة كلمة مرور مستودع المصدر منه (الافتراضي: $RESTIC_FROM_PASSWORD_FILE)

--from-repo="" مستودع المصدر لنسخ اللقطات منه (الافتراضي: $RESTIC_FROM_REPOSITORY)

--from-repository-file="" ملف لقراءة موقع مستودع المصدر منه لنسخ اللقطات (الافتراضي: $RESTIC_FROM_REPOSITORY_FILE)

-h, --help[=false] المساعدة لأمر copy

-H, --host=[] اعتبار اللقطات لهذا المضيف فقط (يمكن تحديده عدة مرات) (الافتراضي: $RESTIC_HOST)

--path=[] اعتبار اللقطات التي تتضمن هذا المسار (المطلق) فقط (يمكن تحديده عدة مرات، يجب أن تتضمن اللقطات جميع المسارات المحددة)

--tag=[] اعتبار اللقطات التي تتضمن العلامة (العلامات) فقط (يمكن تحديدها عدة مرات)

خيارات موروثة من الأوامر الأصل

--cacert=[] ملف لتحميل شهادات الجذر منه (الافتراضي: استخدام شهادات النظام أو $RESTIC_CACERT)

--cache-dir="" تعيين دليل ذاكرة التخزين المؤقت. (الافتراضي: استخدام دليل ذاكرة التخزين المؤقت الافتراضي للنظام)

--cleanup-cache[=false] إزالة تلقائية لدلائل ذاكرة التخزين المؤقت القديمة

--compression=auto وضع الضغط (متوفر فقط لإصدار تنسيق المستودع 2)، أحد الخيارات (auto|off|max) (الافتراضي: $RESTIC_COMPRESSION)

--http-user-agent="" تعيين وكيل مستخدم HTTP لطلبات HTTP الصادرة

--insecure-no-password[=false] استخدم كلمة مرور فارغة للمستودع، يجب تمريرها إلى كل أمر restic (غير آمن)

--insecure-tls[=false] تخطي التحقق من صحة شهادة TLS عند الاتصال بالمستودع (غير آمن)

--json[=false] تعيين وضع الإخراج إلى JSON للأوامر التي تدعمه

--key-hint="" معرف المفتاح لمحاولة فك تشفير أولاً (الافتراضي: $RESTIC_KEY_HINT)

--limit-download=0 يحد التنزيلات إلى حد أقصى معدل بالكيلو بايت/ثانية. (الافتراضي: غير محدود)

--limit-upload=0 يحد التحميلات إلى حد أقصى معدل بالكيلو بايت/ثانية. (الافتراضي: غير محدود)

--no-cache[=false] لا تستخدم ذاكرة تخزين مؤقت محلية

--no-extra-verify[=false] تخطي التحقق الإضافي من البيانات قبل التحميل (انظر التوثيق)

--no-lock[=false] لا تقفل المستودع، هذا يسمح ببعض العمليات على المستودعات للقراءة فقط

-o, --option=[] تعيين خيار ممتد (key=value، يمكن تحديده عدة مرات)

--pack-size=0 تعيين حجم الحزمة المستهدف بالميبيبايت، قد تكون ملفات الحزمة التي تم إنشاؤها أكبر (الافتراضي: $RESTIC_PACK_SIZE)

--password-command="" أمر shell للحصول على كلمة مرور المستودع من (الافتراضي: $RESTIC_PASSWORD_COMMAND)

-p, --password-file="" ملف لقراءة كلمة مرور المستودع منه (الافتراضي: $RESTIC_PASSWORD_FILE)

-q, --quiet[=false] لا تعرض تقرير تقدم شامل

-r, --repo="" المستودع للنسخ الاحتياطي إليه أو الاستعادة منه (الافتراضي: $RESTIC_REPOSITORY)

--repository-file="" ملف لقراءة موقع المستودع منه (الافتراضي: $RESTIC_REPOSITORY_FILE)

--retry-lock=0s حاول إعادة قفل المستودع إذا كان مقفلاً بالفعل، يأخذ قيمة مثل 5m أو 2h (الافتراضي: لا محاولات إعادة)

--stuck-request-timeout=5m0s مدة زمنية سيتم بعدها إعادة محاولة الطلبات العالقة

--tls-client-cert="" مسار ملف يحتوي على شهادة عميل TLS بتنسيق PEM والمفتاح الخاص (الافتراضي: $RESTIC_TLS_CLIENT_CERT)

-v, --verbose[=0] كن مفصلاً (حدد عدة مرات أو مستوى باستخدام --verbose=n، الحد الأقصى للمستوى/المرات هو 2)
