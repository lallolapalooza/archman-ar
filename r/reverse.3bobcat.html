reverse.3bobcat

الحزمة: libbobcat

التراخيص: GPL

FBB::Reverse(3bobcat) مكررات الحلقة فورية عكسية FBB::Reverse(3bobcat)

الاسم

FBB::Reverse - قالب فئة يوفر مكررات عكسية

الملخص

#include <bobcat/reverse>

الوصف

لا توفر جميع أنواع البيانات التي توفر مكررات (أو توفرها ضمنيًا) مكررات عكسية. المصفوفات لا تفعل ذلك، وكذلك قوائم التهيئة. يمكن استخدام المؤشرات إلى عناصر المصفوفة بدلاً من المكررات، ولكن لا يمكن استخدام المصفوفات في الحلقات ذات النمط فور. يمكن استخدام فئة Ranger (راجع ranger(3bobcat)) لتعريف أعضاء begin و end مما يسمح باستخدام المصفوفات في الحلقات ذات النمط فور. باستخدام كائنات Reverse والهياكل البيانات، يمكن استخدامها في الحلقات ذات النمط فور و/أو للحصول على مكررات عكسية لزيارة عناصر البيانات بترتيب معكوس.

عادةً لا يتم استخدام قوالب الفئة Reverse، و ReverseArray، و ReverseSize بشكل مباشر، ولكن يتم إرجاعها بواسطة وظائف reverse المختلفة. توفر هذه الفئات الثلاث عضوين begin، الذي يعيد reverse_iterator إلى نهاية نطاق بيانات الكائن الذي تم تمريره إلى reverse، و end الذي يعيد reverse_iterator إلى بداية نطاق بيانات الكائن الذي تم تمريره إلى reverse.

وظائف عكسية

o
reverse(Type &object):
يعيد كائن Reverse يوفر مكررات عكسية لنطاق بيانات object، والذي يجب أن يدعم الأعضاء begin و end. إذا كان Type نوعًا ثابتًا، فإن أعضاء begin و end في Reverse تعيد const_reverse_iterators، وإلا يتم إعادة reverse_iterators؛
o
reverse(Type (&array)[size]):
يعيد كائن Reverse يوفر مكررات عكسية للنطاق المعكوس لعناصر array. إذا كان Type نوعًا ثابتًا، فإن أعضاء begin و end في Reverse تعيد const_reverse_iterators، وإلا يتم إعادة reverse_iterators؛
o
reverse(Type *array, size_t size):
يعيد كائن Reverse يوفر مكررات عكسية للنطاق المعكوس لعناصر array إلى array + size. إذا كان Type نوعًا ثابتًا، فإن أعضاء begin و end في Reverse تعيد const_reverse_iterators، وإلا يتم إعادة reverse_iterators.

مثال

#include <iostream>
#include <vector>
#include <bobcat/reverse>
using namespace std;
using namespace FBB;
int main()
{
    int intArr[] = { 1, 2, 3, 4, 5};        // المصفوفات ذات الأحجام المعروفة
    for (int value: reverse(intArr))
        cout << value << ’ ’;
    cout << ’\n’;
    for (int value: reverse(intArr, 5))     // استخدام الأحجام المحددة
        cout << value << ’ ’;
    cout << ’\n’;
    int const constArr[] = { 1, 2, 3, 4, 5};// المصفوفات ذات العناصر الثابتة
    for (int value: reverse(constArr))
        cout << value << ’ ’;
    cout << ’\n’;
    for (int value: reverse(constArr, 5))   // استخدام الأحجام المحددة
        cout << value << ’ ’;
    cout << ’\n’;
    for (int value: reverse(intArr, 5))     // المصفوفات ذات الأحجام المحددة
        cout << value << ’ ’;
    cout << ’\n’;
    vector<int> vi{ 1, 2, 3, 4, 5 };        // التعامل مع الكائنات غير الثابتة
    for (int value: reverse(vi))
        cout << value << ’ ’;
    cout << ’\n’;
    vector<int> const cvi{ 1, 2, 3, 4, 5 }; // التعامل مع الكائنات الثابتة
    for (int value: reverse(cvi))
        cout << value << ’ ’;
    cout << ’\n’;
                                            // التعامل مع القيم lvalues
    for (int value: reverse(initializer_list<int>{ 1, 2, 4, 8 }))
        cout << value << ’ ’;
    cout.put(’\n’);
                                            // الكائنات الثابتة المسماة
    auto ilist = initializer_list<int>{ 1, 2, 4, 8 };
    for (int value: reverse(ilist))
        cout << value << ’ ’;
    cout.put(’\n’);
}
