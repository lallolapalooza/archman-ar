repod-file(1)

الحزمة: repod

التراخيص: GPL-3.0-or-later

REPOD-FILE(1) repod REPOD-FILE(1)

إجراءات الملفات للحزم، مستودع الإدارة، وقواعد بيانات المزامنة.

الاستخدام: repod-file [-h] [-c CONFIG] [-d] [-s] [-v] {package,repo,schema} ...

الوسائط الموضعية

الأمر الفرعي
الاختيارات الممكنة: package, repo, schema

الوسائط المسماة

-c, --config
ملف التكوين
-d, --debug
إخراج تصحيح الأخطاء

الافتراضي: False

-s, --system
وضع النظام

الافتراضي: False

-v, --verbose
إخراج مفصل

الافتراضي: False

الأوامر الفرعية

package

التفاعل مع ملفات الحزم

repod-file package [-h] {inspect} ...

الوسائط الموضعية

package
الاختيارات الممكنة: inspect

الأوامر الفرعية

inspect

فحص ملفات الحزم

repod-file package inspect [-h] [-B | -M | -P] [-p] [-s] file [file ...]

الوسائط الموضعية

file
ملفات الحزم

الوسائط المسماة

-B, --buildinfo
فحص .BUILDINFO فقط

الافتراضي: False

-M, --mtree
فحص .MTREE فقط

الافتراضي: False

-P, --pkginfo
فحص .PKGINFO فقط

الافتراضي: False

-p, --pretty
طباعة المخرجات بشكل جميل

الافتراضي: False

-s, --with-signature
تحديد موقع واستخدام ملف توقيع لكل ملف حزمة مقدم

الافتراضي: False

repo

التفاعل مع المستودعات

repod-file repo [-h] {importdb,importpkg,writedb} ...

الوسائط الموضعية

repo
الاختيارات الممكنة: importdb, importpkg, writedb

الأوامر الفرعية

importdb

استيراد الحالة من قاعدة بيانات مزامنة المستودع

repod-file repo importdb [-h] [-a ARCHITECTURE] [-D | -S | -T] file name

الوسائط الموضعية

file
قاعدة بيانات مزامنة المستودع
name
اسم المستودع المراد الاستيراد إليه

الوسائط المسماة

-a, --architecture
استهداف مستودع ببنية معمارية محددة (إذا كان هناك مستودعات متعددة بنفس الاسم ولكن ببنية معمارية مختلفة)
-D, --debug
الاستيراد إلى مستودع التصحيح

الافتراضي: False

-S, --staging
الاستيراد إلى مستودع التدريج

الافتراضي: False

-T, --testing
الاستيراد إلى مستودع الاختبار

الافتراضي: False

importpkg

استيراد الحزم إلى مستودع

repod-file repo importpkg [-h] [-a ARCHITECTURE] [-d] [-p] [-s]
                          [-u SOURCE_URL [SOURCE_URL ...]] [-D | -S | -T]
                          file [file ...] name

الوسائط الموضعية

file
ملفات الحزم
name
اسم المستودع المراد الاستيراد إليه

الوسائط المسماة

-a, --architecture
استهداف مستودع ببنية معمارية محددة (إذا كان هناك مستودعات متعددة بنفس الاسم ولكن ببنية معمارية مختلفة)
-d, --dry-run
إظهار المخرجات فقط، ولكن لا تكتب المخرجات إلى الملف

الافتراضي: False

-p, --pretty
طباعة المخرجات بشكل جميل (ينطبق فقط على وضع المحاكاة)

الافتراضي: False

-s, --with-signature
تحديد موقع واستخدام ملف توقيع لكل ملف حزمة مقدم

الافتراضي: False

-u, --source-url
قائمة بعناوين URL المصدر لقواعد الحزم المضافة (مقدمة كسلسلة واحدة أو أكثر من pkgbase=url)

الافتراضي: []

-D, --debug
الاستيراد إلى مستودع التصحيح

الافتراضي: False

-S, --staging
الاستيراد إلى مستودع التدريج

الافتراضي: False

-T, --testing
الاستيراد إلى مستودع الاختبار

الافتراضي: False

الوسائط المسماة

استهداف مستودع ببنية معمارية محددة (إذا كان هناك مستودعات متعددة بنفس الاسم ولكن ببنية معمارية مختلفة)
-D, --debug
التصدير من مستودع التصحيح

الافتراضي: False

-S, --staging
التصدير من مستودع التدريج

الافتراضي: False

-T, --testing
التصدير من مستودع الاختبار

الافتراضي: False
